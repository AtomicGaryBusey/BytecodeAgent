Registers: x0=src, x1=n, x2=out, x3=cap. Use x4 = i (src index), x5 = o (output index).

Loop:
- if i >= n: return o.
- h = ldrsb src[i]; i++.
- if h == -128: continue.
- if h >= 0, literal: len = h+1. If i+len > n, return -2. Else copy min(len, cap-o) bytes; if len > cap-o, return -1 after the partial copy. Otherwise advance i and o.
- if h < 0, repeat: len = 1-h. If i >= n, return -2 (needs 1 byte). Read the byte b, i++. Write min(len, cap-o) copies; if len > cap-o, return -1.

Error priority: the source check happens at header read, before output, so -2 wins over -1 for the same packet. The order above respects that.

Per-packet code:
- x6 = h, x7 = len, x9 = cap-o (remaining), x10 = count = min(len, rem).
- Literal copy loop: while x10 > 0, copy a byte.
- After the copy, if len > rem, return -1. Otherwise o += len.

Return -1 with o=cap already written, so the copy is complete.

packbits_decode:
    mov x4, #0
    mov x5, #0
.Lloop:
    cmp x4, x1
    b.ge .Ldone
    ldrsb x6, [x0, x4]
    add x4, x4, #1
    cmp x6, #-128
    b.eq .Lloop
    sub x9, x3, x5
    tbnz x6, #63, .Lrep
    add x7, x6, #1
    add x11, x4, x7
    cmp x11, x1
    b.gt .Lerr2
    cmp x7, x9
    csel x10, x7, x9, lt
    add x12, x0, x4
    add x13, x2, x5
    mov x14, x10
.Lcl:
    cbz x14, .Lcld
    ldrb w15, [x12], #1
    strb w15, [x13], #1
    sub x14, x14, #1
    b .Lcl
.Lcld:
    cmp x7, x9
    b.gt .Lerr1
    add x4, x4, x7
    add x5, x5, x7
    b .Lloop
.Lrep:
    mov x7, #1
    sub x7, x7, x6
    cmp x4, x1
    b.ge .Lerr2
    ldrb w15, [x0, x4]
    add x4, x4, #1
    cmp x7, x9
    csel x10, x7, x9, lt
    add x13, x2, x5
.Lrl:
    cbz x10, .Lrld
    strb w15, [x13], #1
    sub x10, x10, #1
    b .Lrl
.Lrld:
    cmp x7, x9
    b.gt .Lerr1
    add x5, x5, x7
    b .Lloop
.Ldone:
    mov x0, x5
    ret
.Lerr1:
    mov x0, #-1
    ret
.Lerr2:
    mov x0, #-2
    ret
