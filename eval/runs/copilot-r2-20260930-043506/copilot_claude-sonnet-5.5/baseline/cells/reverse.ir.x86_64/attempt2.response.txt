export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%j =l phi @start %j0, @body %j1
	%c =w csltl %i, %j
	jnz %c, @body, @end
@body
	%oi =l mul %i, 8
	%pi =l add %a, %oi
	%oj =l mul %j, 8
	%pj =l add %a, %oj
	%x =l loadl %pi
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@end
	ret 0
}
