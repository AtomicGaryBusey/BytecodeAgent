export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 1
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p1 =l add %a, %off
	%p0 =l sub %p1, 8
	%x =l loadl %p0
	%y =l loadl %p1
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%cnt2 =l add %cnt, %gl
	%i2 =l add %i, 1
	jmp @latch
@latch
	%cnt =l copy %cnt2
	%i =l copy %i2
	jmp @cond
@end
	ret %cnt
}
