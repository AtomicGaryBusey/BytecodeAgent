export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@zl
	%c =w csell %j, %n
	%p1 =l mul %j, 8
	%p =l add %row, %p1
	storel 0, %p
	%j =l add %j, 1
	%c2 =w cslel %j, %n
	jnz %c2, @zl, @outer0
@outer0
	%i =l copy 0
@outer
	%ci =w csltl %i, %m
	jnz %ci, @ob, @done
@ob
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag =l copy 0
	%j =l copy 1
	jmp @inner
@inner
	%cj =w cslel %j, %n
	jnz %cj, @ib, @onext
@ib
	%o =l mul %j, 8
	%pj =l add %row, %o
	%pm =l sub %pj, 8
	%up =l loadl %pj
	%left =l loadl %pm
	%j1 =l sub %j, 1
	%yp =l add %y, %j1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @store
@nomatch
	%gt =w csgtl %left, %up
	jnz %gt, @l, @u
@l
	%v2 =l copy %left
	jmp @store2
@u
	%v3 =l copy %up
	jmp @store3
@store
	storel %v1, %pj
	jmp @nx
@store2
	storel %v2, %pj
	jmp @nx
@store3
	storel %v3, %pj
	jmp @nx
@nx
	%diag =l copy %up
	%j =l add %j, 1
	jmp @inner
@onext
	%i =l add %i, 1
	jmp @outer
@done
	%o2 =l mul %n, 8
	%pn =l add %row, %o2
	%r =l loadl %pn
	ret %r
}
