export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zl
@zl
	%j =l phi @start 0, @zl %j1
	%p1 =l mul %j, 8
	%p =l add %row, %p1
	storel 0, %p
	%j1 =l add %j, 1
	%c2 =w cslel %j1, %n
	jnz %c2, @zl, @outer
@outer
	%i =l phi @zl 0, @onext %i1
	%ci =w csltl %i, %m
	jnz %ci, @ob, @done
@ob
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%jj =l phi @ob 1, @nx %jn
	%diag =l phi @ob 0, @nx %up
	%cj =w cslel %jj, %n
	jnz %cj, @ib, @onext
@ib
	%o =l mul %jj, 8
	%pj =l add %row, %o
	%pm =l sub %pj, 8
	%up =l loadl %pj
	%left =l loadl %pm
	%jm =l sub %jj, 1
	%yp =l add %y, %jm
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @store
@nomatch
	%gt =w csgtl %left, %up
	jnz %gt, @l, @u
@l
	jmp @store
@u
	jmp @store
@store
	%v =l phi @match %v1, @l %left, @u %up
	storel %v, %pj
	jmp @nx
@nx
	%jn =l add %jj, 1
	jmp @inner
@onext
	%i1 =l add %i, 1
	jmp @outer
@done
	%o2 =l mul %n, 8
	%pn =l add %row, %o2
	%r =l loadl %pn
	ret %r
}
