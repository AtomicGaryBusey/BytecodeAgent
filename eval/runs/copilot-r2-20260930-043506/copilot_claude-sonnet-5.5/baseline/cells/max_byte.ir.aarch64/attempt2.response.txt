export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%m =l phi @start 0, @body %m2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%a =l add %p, %i
	%b =l loadub %a
	%g =w cugtl %b, %m
	%d =l sub %b, %m
	%e =l extuw %g
	%f =l mul %d, %e
	%m2 =l add %m, %f
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %m
}
