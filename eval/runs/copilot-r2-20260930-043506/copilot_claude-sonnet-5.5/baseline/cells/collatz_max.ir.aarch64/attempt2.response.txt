export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %k2
	%best =l phi @start 0, @next %best2
	%c =w cslel %k, %n
	jnz %c, @inner_init, @done
@inner_init
	jmp @loop
@loop
	%x =l phi @inner_init %k, @step %x2
	%s =l phi @inner_init 0, @step %s2
	%d =w cnel %x, 1
	jnz %d, @body, @fin
@body
	%b =l and %x, 1
	%o =w cnel %b, 0
	jnz %o, @odd, @even
@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	jmp @step
@even
	%xe =l shr %x, 1
	jmp @step
@step
	%x2 =l phi @odd %xo, @even %xe
	%s2 =l add %s, 1
	jmp @loop
@fin
	%g =w csgtl %s, %best
	jnz %g, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%best2 =l phi @upd %s, @keep %best
	%k2 =l add %k, 1
	jmp @outer
@done
	ret %best
}
