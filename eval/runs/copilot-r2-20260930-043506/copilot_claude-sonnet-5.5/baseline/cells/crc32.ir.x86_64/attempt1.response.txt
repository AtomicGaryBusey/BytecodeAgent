export function l $crc32(l %p, l %n) {
@start
	%c =w copy 4294967295
	%i =l copy 0
	%k =w copy 0
	jmp @check
@check
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%a =l add %p, %i
	%b =w loadub %a
	%c =w xor %c, %b
	%k =w copy 0
	jmp @bit
@bit
	%m =w and %c, 1
	%neg =w sub 0, %m
	%x =w and %neg, 3988292384
	%c =w shr %c, 1
	%c =w xor %c, %x
	%k =w add %k, 1
	%kc =w csltw %k, 8
	jnz %kc, @bit, @next
@next
	%i =l add %i, 1
	jmp @check
@done
	%r =w xor %c, 4294967295
	%q =l extuw %r
	ret %q
}
