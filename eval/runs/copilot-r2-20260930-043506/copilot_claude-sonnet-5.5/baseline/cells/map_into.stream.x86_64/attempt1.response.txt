map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rbp
    sub rsp, 8
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx
    xor r15d, r15d
    xor ebp, ebp
.Lloop:
    cmp rbp, r13
    jge .Ldone
    mov rdi, qword ptr [r12 + rbp*8]
    call r14
    mov rcx, qword ptr [r12 + rbp*8]
    mov qword ptr [rbx + rbp*8], rax
    cmp rax, rcx
    setl dl
    movzx edx, dl
    add r15, rdx
    inc rbp
    jmp .Lloop
.Ldone:
    mov rax, r15
    add rsp, 8
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
