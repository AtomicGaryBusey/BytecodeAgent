    mov r9, 1
    xor r10, r10
.Louter:
    cmp r9, r8
    jg .Ldone
    mov rax, r9
    xor rcx, rcx
.Linner:
    cmp rax, 1
    je .Lend
    test al, 1
    jnz .Lodd
    shr rax, 1
    inc rcx
    jmp .Linner
.Lodd:
    lea rax, [rax + rax*2 + 1]
    inc rcx
    jmp .Linner
.Lend:
    cmp rcx, r10
    jle .Lnext
    mov r10, rcx
.Lnext:
    inc r9
    jmp .Louter
.Ldone:
    mov rax, r10
    ret
