export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
	%c =w csltl %m, 1
	jnz %c, @done, @loop
@loop
	%i =l phi @start, 0, @loop, %i2
	%cnt =l phi @start, 0, @loop, %cnt2
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%g =w csltl %y, %x
	%gl =l extuw %g
	%cnt2 =l add %cnt, %gl
	%i2 =l add %i, 1
	%d =w csltl %i2, %m
	jnz %d, @loop, @done2
@done2
	ret %cnt2
@done
	ret 0
}
