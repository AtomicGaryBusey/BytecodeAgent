export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%i0 =l copy 0
	jmp @zloop
@zloop
	%i =l phi @start %i0, @zbody %in
	%c =w csltl %i, %cap
	jnz %c, @zbody, @init
@zbody
	%o =l shl %i, 3
	%p =l add %stack, %o
	storel 0, %p
	%in =l add %i, 1
	jmp @zloop
@init
	jmp @loop
@loop
	%pc =l phi @init 0, @nx1 %pc1, @nx2 %pc2, @nx3 %pc3, @jz %pc4, @jmpb %pc5 
	%sp =l phi @init 0, @nx1 %sp1, @nx2 %sp2, @nx3 %sp3, @jz %sp4, @jmpb %sp
	%cnt =l phi @init 0, @nx1 %cnt, @nx2 %cnt, @nx3 %cnt, @jz %cnt, @jmpb %cnt
	%e =w ceql %pc, %len
	jnz %e, @done, @fetch
@fetch
	%po =l shl %pc, 3
	%pa =l add %code, %po
	%op =l loadl %pa
	%pc1 =l add %pc, 1
	%pc2 =l add %pc, 2
	%pa2 =l add %pa, 8
	%opd =l loadl %pa2
	%so =l shl %sp, 3
	%tp =l add %stack, %so
	%t1 =l sub %tp, 8
	%t2 =l sub %tp, 16
	%b =l loadl %t1
	%a =l loadl %t2
	%isz =w ceql %op, 0
	jnz %isz, @done, @d1
@d1
	%x =w ceql %op, 1
	jnz %x, @push, @d2
@push
	storel %opd, %tp
	%sp1p =l add %sp, 1
	jmp @nx1p
@nx1p
	%pc3 =l copy %pc2
	jmp @nx3
@d2
	%x =w ceql %op, 5
	jnz %x, @dup, @d3
@dup
	storel %b, %tp
	%sp5 =l add %sp, 1
	jmp @nx2
@d3
	%x =w ceql %op, 6
	jnz %x, @swap, @d4
@swap
	storel %b, %t2
	storel %a, %t1
	jmp @nx1
@d4
	%x =w ceql %op, 7
	jnz %x, @drop, @d5
@drop
	%sp7 =l sub %sp, 1
	jmp @nx3
@d5
	%x =w ceql %op, 8
	jnz %x, @over, @d6
@over
	storel %a, %tp
	%sp8 =l add %sp, 1
	jmp @nx3
@d6
	%x =w ceql %op, 10
	jnz %x, @jzb, @d7
@jzb
	%sp4 =l sub %sp, 1
	%z =w ceql %b, 0
	jnz %z, @jz1, @jz
@jz1
	jmp @jz2
@jz2
	jmp @jz
@jz
	%pc4 =l phi @jzb %pc2, @jz1 %opd
	%sp4b =l copy %sp4
	jmp @nx3
@d7
	%x =w ceql %op, 11
	jnz %x, @jmpb, @arith
@jmpb
	%pc5 =l copy %opd
	jmp @loop
@arith
	%s =l sub %sp, 1
	%r2 =w ceql %op, 2
	%r3 =w ceql %op, 3
	%r4 =w ceql %op, 4
	%ad =l add %a, %b
	%sb =l sub %a, %b
	%ml =l mul %a, %b
	%lt =w csltl %a, %b
	%ltl =l extuw %lt
	%v1 =l copy %ltl
	jnz %r2, @wr1, @c3
@c3
	jnz %r3, @wr2, @c4
@c4
	jnz %r4, @wr3, @wr4
@wr1
	storel %ad, %t2
	jmp @nx2b
@wr2
	storel %sb, %t2
	jmp @nx2b
@wr3
	storel %ml, %t2
	jmp @nx2b
@wr4
	storel %v1, %t2
	jmp @nx2b
@nx2b
	jmp @nx2
@nx1
	jmp @nx3
@nx2
	jmp @nx3
@nx3
	jmp @loop
@done
	%r =l copy 0
	ret 0
}
