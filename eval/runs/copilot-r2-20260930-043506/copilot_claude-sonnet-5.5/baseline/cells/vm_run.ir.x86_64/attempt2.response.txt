export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%sp =l loadl %spp
	%e =w ceql %pc, %len
	jnz %e, @done, @fetch
@fetch
	%po =l shl %pc, 3
	%pa =l add %code, %po
	%op =l loadl %pa
	%so =l shl %sp, 3
	%tp =l add %stack, %so
	%t1 =l sub %tp, 8
	%t2 =l sub %tp, 16
	%pc1 =l add %pc, 1
	%pc2 =l add %pc, 2
	%sp1 =l add %sp, 1
	%sm =l sub %sp, 1
	%x0 =w ceql %op, 0
	jnz %x0, @done, @d1
@d1
	%x1 =w ceql %op, 1
	jnz %x1, @push, @d2
@push
	%pa2 =l add %pa, 8
	%opd =l loadl %pa2
	storel %opd, %tp
	storel %sp1, %spp
	storel %pc2, %pcp
	jmp @loop
@d2
	%x2 =w ceql %op, 5
	jnz %x2, @dup, @d3
@dup
	%bd =l loadl %t1
	storel %bd, %tp
	storel %sp1, %spp
	storel %pc1, %pcp
	jmp @loop
@d3
	%x3 =w ceql %op, 6
	jnz %x3, @swap, @d4
@swap
	%bs =l loadl %t1
	%as =l loadl %t2
	storel %bs, %t2
	storel %as, %t1
	storel %pc1, %pcp
	jmp @loop
@d4
	%x4 =w ceql %op, 7
	jnz %x4, @drop, @d5
@drop
	storel %sm, %spp
	storel %pc1, %pcp
	jmp @loop
@d5
	%x5 =w ceql %op, 8
	jnz %x5, @over, @d6
@over
	%ao =l loadl %t2
	storel %ao, %tp
	storel %sp1, %spp
	storel %pc1, %pcp
	jmp @loop
@d6
	%x6 =w ceql %op, 10
	jnz %x6, @jzb, @d7
@jzb
	%pa3 =l add %pa, 8
	%tg =l loadl %pa3
	%bz =l loadl %t1
	storel %sm, %spp
	%z =w ceql %bz, 0
	jnz %z, @jzt, @jzf
@jzt
	storel %tg, %pcp
	jmp @loop
@jzf
	storel %pc2, %pcp
	jmp @loop
@d7
	%x7 =w ceql %op, 11
	jnz %x7, @jmpb, @arith
@jmpb
	%pa4 =l add %pa, 8
	%tg2 =l loadl %pa4
	storel %tg2, %pcp
	jmp @loop
@arith
	%b =l loadl %t1
	%a =l loadl %t2
	%r2 =w ceql %op, 2
	jnz %r2, @wadd, @c3
@wadd
	%ad =l add %a, %b
	jmp @binw
@c3
	%r3 =w ceql %op, 3
	jnz %r3, @wsub, @c4
@wsub
	%sb =l sub %a, %b
	jmp @binw
@c4
	%r4 =w ceql %op, 4
	jnz %r4, @wmul, @wlt
@wmul
	%ml =l mul %a, %b
	jmp @binw
@wlt
	%lt =w csltl %a, %b
	%ltl =l extuw %lt
	jmp @binw
@binw
	%res =l phi @wadd %ad, @wsub %sb, @wmul %ml, @wlt %ltl
	storel %res, %t2
	storel %sm, %spp
	storel %pc1, %pcp
	jmp @loop
@done
	%dsp =l loadl %spp
	%nz =w csgtl %dsp, 0
	jnz %nz, @hasv, @zinit
@hasv
	%hs =l sub %dsp, 1
	%hso =l shl %hs, 3
	%hp =l add %stack, %hso
	%hv =l loadl %hp
	jmp @zinit
@zinit
	%rv =l phi @done 0, @hasv %hv
	jmp @zloop
@zloop
	%i =l phi @zinit %dsp, @zbody %in
	%c =w csltl %i, %cap
	jnz %c, @zbody, @fin
@zbody
	%o =l shl %i, 3
	%p =l add %stack, %o
	storel 0, %p
	%in =l add %i, 1
	jmp @zloop
@fin
	ret %rv
}
