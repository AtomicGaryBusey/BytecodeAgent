dot_i32_simd:
    pxor xmm0, xmm0
    xor rax, rax
    mov rcx, rdx
    and rcx, -4
    jz .Ltail
.Lloop:
    movdqu xmm1, xmmword ptr [rdi + rax*4]
    movdqu xmm2, xmmword ptr [rsi + rax*4]
    pshufd xmm3, xmm1, 0xF5
    pshufd xmm4, xmm2, 0xF5
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm0, xmm1
    paddq xmm0, xmm3
    add rax, 4
    cmp rax, rcx
    jb .Lloop
.Ltail:
    movq r8, xmm0
    pshufd xmm1, xmm0, 0xEE
    movq r9, xmm1
    add r8, r9
.Lscalar:
    cmp rax, rdx
    jge .Ldone
    movsxd r9, dword ptr [rdi + rax*4]
    movsxd r10, dword ptr [rsi + rax*4]
    imul r9, r10
    add r8, r9
    inc rax
    jmp .Lscalar
.Ldone:
    mov rax, r8
    ret
