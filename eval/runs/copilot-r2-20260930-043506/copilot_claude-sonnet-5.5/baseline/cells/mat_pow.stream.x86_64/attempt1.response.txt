mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    mov r9, rsi
    imul r9, rsi
    xor eax, eax
.Lz:
    cmp rax, r9
    jge .Lzd
    mov qword ptr [rcx + rax*8], 0
    inc rax
    jmp .Lz
.Lzd:
    xor eax, eax
    lea r10, [rsi + 1]
.Ld:
    cmp rax, r9
    jge .Lk
    mov qword ptr [rcx + rax*8], 1
    add rax, r10
    jmp .Ld
.Lk:
    test rdx, rdx
    jle .Lfin
    xor r10, r10
    xor r12, r12
.Li:
    cmp r10, rsi
    jge .Lcopy
    xor r11, r11
.Lj:
    cmp r11, rsi
    jge .Linext
    xor eax, eax
    xor ebx, ebx
.Lt:
    cmp rbx, rsi
    jge .Ltd
    lea rbp, [r12 + rbx]
    mov r13, qword ptr [rcx + rbp*8]
    mov rbp, rbx
    imul rbp, rsi
    add rbp, r11
    imul r13, qword ptr [rdi + rbp*8]
    add rax, r13
    inc rbx
    jmp .Lt
.Ltd:
    lea rbp, [r12 + r11]
    mov qword ptr [r8 + rbp*8], rax
    inc r11
    jmp .Lj
.Linext:
    inc r10
    add r12, rsi
    jmp .Li
.Lcopy:
    xor eax, eax
.Lc:
    cmp rax, r9
    jge .Lcd
    mov r13, qword ptr [r8 + rax*8]
    mov qword ptr [rcx + rax*8], r13
    inc rax
    jmp .Lc
.Lcd:
    dec rdx
    jmp .Lk
.Lfin:
    add r9, r9
    xor eax, eax
.Lzt:
    cmp rax, r9
    jge .Lend
    mov qword ptr [r8 + rax*8], 0
    inc rax
    jmp .Lzt
.Lend:
    xor eax, eax
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
