export function l $crc32(l %p, l %n) {
@start
	jmp @check
@check
	%crc =w phi @start 4294967295, @next %x2
	%i =l phi @start 0, @next %i1
	%c =w csltl %i, %n
	jnz %c, @byte, @done
@byte
	%q =l add %p, %i
	%b =w loadub %q
	%crc1 =w xor %crc, %b
	jmp @bit
@bit
	%x =w phi @byte %crc1, @bit %x3
	%j =w phi @byte 8, @bit %j2
	%lsb =w and %x, 1
	%m =w sub 0, %lsb
	%pm =w and %m, 3988292384
	%sh =w shr %x, 1
	%x3 =w xor %sh, %pm
	%j2 =w sub %j, 1
	jnz %j2, @bit, @next
@next
	%x2 =w copy %x3
	%i1 =l add %i, 1
	jmp @check
@done
	%r =w xor %crc, 4294967295
	%rl =l extuw %r
	ret %rl
}
