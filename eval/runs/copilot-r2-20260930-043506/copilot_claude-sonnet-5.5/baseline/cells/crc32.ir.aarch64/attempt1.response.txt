export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @check
@check
	%c =w csltl %i, %n
	jnz %c, @byte, @done
@byte
	%q =l add %p, %i
	%b =w loadub %q
	%crc1 =w xor %crc, %b
	%k =w copy 8
	jmp @bit
@bit
	%x =w phi @byte %crc1, @bit %x2
	%j =w phi @byte %k, @bit %j2
	%lsb =w and %x, 1
	%m =w sub 0, %lsb
	%pm =w and %m, 3988292384
	%sh =w shr %x, 1
	%x2 =w xor %sh, %pm
	%j2 =w sub %j, 1
	jnz %j2, @bit, @next
@next
	%i1 =l add %i, 1
	jmp @check2
@check2
	jmp @check3
@check3
	jmp @loop
@loop
	jmp @check
@done
	%r =w xor %crc, 4294967295
	%rl =l extuw %r
	ret %rl
}
