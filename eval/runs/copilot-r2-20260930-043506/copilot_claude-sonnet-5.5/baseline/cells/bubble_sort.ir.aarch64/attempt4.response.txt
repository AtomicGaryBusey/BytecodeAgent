export function l $bubble_sort(l %a, l %n) {
@start
	%m0 =l sub %n, 1
	jmp @outer
@outer
	%m =l phi @start %m0, @onext %m1
	%c =w csgtl %m, 0
	jnz %c, @inner, @done
@inner
	%i =l phi @outer 0, @inc %i2
	%c2 =w csltl %i, %m
	jnz %c2, @ibody, @onext
@ibody
	%off =l mul %i, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%g =w csgtl %x, %y
	jnz %g, @swap, @inc
@swap
	storel %y, %p
	storel %x, %q
	jmp @inc
@inc
	%i2 =l add %i, 1
	jmp @inner
@onext
	%m1 =l sub %m, 1
	jmp @outer
@done
	ret 0
}
