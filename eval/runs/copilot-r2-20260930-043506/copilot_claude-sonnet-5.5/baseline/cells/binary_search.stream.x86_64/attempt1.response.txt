binary_search:
    xor ecx, ecx
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jge .Lnotfound
    mov rax, r8
    sub rax, rcx
    shr rax, 1
    add rax, rcx
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    je .Lfound
    jl .Lgreater
    mov r8, rax
    jmp .Lloop
.Lgreater:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lnotfound:
    mov rax, -1
.Lfound:
    ret
