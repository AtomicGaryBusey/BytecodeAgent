bubble_sort:
    mov x9, x1
.Louter:
    cmp x9, #1
    b.le .Ldone
    mov x10, #0
    sub x11, x9, #1
    mov x12, x0
.Linner:
    cmp x10, x11
    b.ge .Lnext
    ldr x13, [x12]
    ldr x14, [x12, #8]
    cmp x13, x14
    b.le .Lnoswap
    str x14, [x12]
    str x13, [x12, #8]
.Lnoswap:
    add x12, x12, #8
    add x10, x10, #1
    b .Linner
.Lnext:
    sub x9, x9, #1
    b .Louter
.Ldone:
    mov x0, #0
    ret
