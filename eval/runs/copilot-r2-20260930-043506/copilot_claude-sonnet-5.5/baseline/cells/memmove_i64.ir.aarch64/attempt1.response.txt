export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @fwd, @chk
@chk
	%eq =w ceql %dst, %src
	jnz %eq, @done, @bwd
@fwd
	%i =l copy 0
	jmp @fl
@fl
	%c =w csltl %i, %n
	jnz %c, @fb, @done
@fb
	%o =l shl %i, 3
	%s =l add %src, %o
	%d =l add %dst, %o
	%v =l loadl %s
	storel %v, %d
	%i =l add %i, 1
	jmp @fl
@bwd
	%j =l copy %n
	jmp @bl
@bl
	%c2 =w csgtl %j, 0
	jnz %c2, @bb, @done
@bb
	%j =l sub %j, 1
	%o2 =l shl %j, 3
	%s2 =l add %src, %o2
	%d2 =l add %dst, %o2
	%v2 =l loadl %s2
	storel %v2, %d2
	jmp @bl
@done
	ret 0
}
