export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%best =l copy 0
	jmp @outer
@outer
	%c =w cslel %k, %n
	jnz %c, @init, @done
@init
	%x =l copy %k
	%s =l copy 0
	jmp @inner
@inner
	%e =w cnel %x, 1
	jnz %e, @step, @fin
@step
	%b =l and %x, 1
	%o =w cnel %b, 0
	jnz %o, @odd, @even
@odd
	%t =l mul %x, 3
	%x1 =l add %t, 1
	jmp @cont
@even
	%x2 =l shr %x, 1
	jmp @cont
@cont
	%x =l phi @odd %x1, @even %x2
	%s =l add %s, 1
	jmp @inner
@fin
	%g =w csgtl %s, %best
	jnz %g, @upd, @next
@upd
	%best =l copy %s
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
