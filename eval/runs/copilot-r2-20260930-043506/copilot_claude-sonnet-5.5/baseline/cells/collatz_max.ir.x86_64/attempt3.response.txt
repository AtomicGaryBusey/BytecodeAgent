export function l $collatz_max(l %n) {
@start
	%pk =l alloc8 8
	%pb =l alloc8 8
	%px =l alloc8 8
	%ps =l alloc8 8
	storel 1, %pk
	storel 0, %pb
	jmp @outer
@outer
	%k =l loadl %pk
	%c =w cslel %k, %n
	jnz %c, @init, @done
@init
	%k0 =l loadl %pk
	storel %k0, %px
	storel 0, %ps
	jmp @inner
@inner
	%x =l loadl %px
	%e =w cnel %x, 1
	jnz %e, @step, @fin
@step
	%xa =l loadl %px
	%b =l and %xa, 1
	%o =w cnel %b, 0
	jnz %o, @odd, @even
@odd
	%xo =l loadl %px
	%t =l mul %xo, 3
	%x1 =l add %t, 1
	storel %x1, %px
	jmp @cont
@even
	%xe =l loadl %px
	%x2 =l shr %xe, 1
	storel %x2, %px
	jmp @cont
@cont
	%s =l loadl %ps
	%s1 =l add %s, 1
	storel %s1, %ps
	jmp @inner
@fin
	%sf =l loadl %ps
	%bf =l loadl %pb
	%g =w csgtl %sf, %bf
	jnz %g, @upd, @next
@upd
	%sf2 =l loadl %ps
	storel %sf2, %pb
	jmp @next
@next
	%kk =l loadl %pk
	%kn =l add %kk, 1
	storel %kn, %pk
	jmp @outer
@done
	%r =l loadl %pb
	ret %r
}
