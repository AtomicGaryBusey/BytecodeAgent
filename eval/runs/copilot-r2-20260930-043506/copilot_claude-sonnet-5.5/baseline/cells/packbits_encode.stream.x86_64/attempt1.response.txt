Register plan:
- rdi=src, rsi=n, rdx=out, rcx=cap
- r8=i, r9=o (output position)

Main loop:
- If i>=n, return o.
- Compute run(i) into rax: count k=1, while i+k<n and k<128 and src[i+k]==src[i], k++.
- If k>=3: need o+2<=cap, else return -1. Write 257-k as a byte (i.e. (1-k)&0xff, using neg), write the byte, i+=k, o+=2.
- Otherwise literal: j=i, and while j<n and j-i<128, compute run(j); if run>=3 break; else j++.
- L=j-i. The literal has L>=1 because run(i)<3 means the first iteration passes.
- Need o+1+L<=cap. Write header L-1, copy the bytes, and update o and i.

Run computation as inline code with r10=position p, result in eax. The cap of 128 is applied on the count. Since run is capped at 128 and only compared to 3, the literal check only needs to see whether p+2<n and src[p]==src[p+1]==src[p+2]. This is equivalent to run>=3, and it never reads past n-1 because p+2<n is checked first.

Literal loop:
- r10=j. Loop: if j>=n, done. If j-i>=128, done. If j+2<n and src[j]==src[j+1] and src[j]==src[j+2], done. Otherwise j++.

Repeat count: k=1; while k<128 and i+k<n and src[i+k]==src[i], k++. Use r10 as k, and r11b as the byte value.

```
packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movzx r11d, byte ptr [rdi + r8]
    mov r10d, 1
.Lrun:
    cmp r10, 128
    jge .Lrundone
    lea rax, [r8 + r10]
    cmp rax, rsi
    jge .Lrundone
    cmp r11b, byte ptr [rdi + rax]
    jne .Lrundone
    inc r10
    jmp .Lrun
.Lrundone:
    cmp r10, 3
    jl .Lliteral
    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lfail
    mov rax, r10
    neg rax
    inc rax
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r11b
    add r9, 2
    add r8, r10
    jmp .Lloop
.Lliteral:
    mov r10, r8
.Ljl:
    cmp r10, rsi
    jge .Ljdone
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jge .Ljdone
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Ljnext
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Ljnext
    cmp al, byte ptr [rdi + r10 + 2]
    je .Ljdone
.Ljnext:
    inc r10
    jmp .Ljl
.Ljdone:
    sub r10, r8
    lea rax, [r9 + r10 + 1]
    cmp rax, rcx
    jg .Lfail
    lea rax, [r10 - 1]
    mov byte ptr [rdx + r9], al
    inc r9
    xor eax, eax
.Lcopy:
    cmp rax, r10
    jge .Lcopied
    movzx r11d, byte ptr [rdi + r8 + rax]
    mov byte ptr [rdx + r9 + rax], r11b
    inc rax
    jmp .Lcopy
.Lcopied:
    add r9, r10
    add r8, r10
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
```
