packbits_encode:
    xor r8, r8
    xor r9, r9
.Lmain:
    cmp r8, rsi
    jge .Ldone
    movzx eax, byte ptr [rdi + r8]
    mov r10, 1
.Lrun:
    lea r11, [r8 + r10]
    cmp r11, rsi
    jge .Lrunend
    cmp r10, 128
    jge .Lrunend
    cmp al, byte ptr [rdi + r11]
    jne .Lrunend
    inc r10
    jmp .Lrun
.Lrunend:
    cmp r10, 3
    jl .Lliteral
    lea r11, [r9 + 2]
    cmp r11, rcx
    jg .Lfail
    mov r11, 1
    sub r11, r10
    mov byte ptr [rdx + r9], r11b
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lmain
.Lliteral:
    mov r10, r8
.Lljloop:
    cmp r10, rsi
    jge .Llend
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jge .Llend
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Llinc
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llinc
    cmp al, byte ptr [rdi + r10 + 2]
    je .Llend
.Llinc:
    inc r10
    jmp .Lljloop
.Llend:
    mov rax, r10
    sub rax, r8
    lea r11, [r9 + rax + 1]
    cmp r11, rcx
    jg .Lfail
    lea r11, [rax - 1]
    mov byte ptr [rdx + r9], r11b
    inc r9
    mov r11, r8
.Lcp:
    cmp r11, r10
    jge .Lcpdone
    movzx eax, byte ptr [rdi + r11]
    mov byte ptr [rdx + r9], al
    inc r9
    inc r11
    jmp .Lcp
.Lcpdone:
    mov r8, r10
    jmp .Lmain
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
