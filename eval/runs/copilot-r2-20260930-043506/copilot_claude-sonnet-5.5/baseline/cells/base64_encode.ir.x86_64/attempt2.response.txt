export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%o =l phi @start 0, @next %o2
	%rem =l sub %n, %i
	%c =w csgtl %rem, 0
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%b0 =l loadub %p
	%g1 =w csgtl %rem, 1
	%g2 =w csgtl %rem, 2
	jnz %g1, @l1, @j1
@l1
	%p1 =l add %p, 1
	%b1a =l loadub %p1
	jmp @j1
@j1
	%b1b =l phi @body 0, @l1 %b1a
	jnz %g2, @l2, @j2
@l2
	%p2 =l add %p, 2
	%b2a =l loadub %p2
	jmp @j2
@j2
	%b2b =l phi @j1 0, @l2 %b2a
	%s0 =l shl %b0, 16
	%s1 =l shl %b1b, 8
	%v0 =l or %s0, %s1
	%v =l or %v0, %b2b
	%q =l add %out, %o
	jmp @enc
@enc
	%k1 =l phi @j2 0, @encn %kn
	%sh0 =l mul %k1, 6
	%sh =l sub 18, %sh0
	%x0 =l shr %v, %sh
	%x =l and %x0, 63
	%t1 =w csltl %x, 26
	%t2 =w csltl %x, 52
	%t3 =w csltl %x, 62
	%t4 =w ceql %x, 62
	%r1 =l add %x, 65
	%r2 =l add %x, 71
	%r3 =l sub %x, 4
	jnz %t1, @c1, @n1
@c1
	jmp @st
@n1
	jnz %t2, @c2, @n2
@c2
	jmp @st
@n2
	jnz %t3, @c3, @n3
@c3
	jmp @st
@n3
	jnz %t4, @c4, @c5
@c4
	jmp @st
@c5
	jmp @st
@st
	%chr =l phi @c1 %r1, @c2 %r2, @c3 %r3, @c4 43, @c5 47
	%lim =l add %rem, 1
	%cond =w csltl %k1, %lim
	%pos =l add %q, %k1
	jnz %cond, @wr, @pad
@wr
	storeb %chr, %pos
	jmp @encn
@pad
	storeb 61, %pos
	jmp @encn
@encn
	%kn =l add %k1, 1
	%d =w csltl %kn, 4
	jnz %d, @enc, @next
@next
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @loop
@done
	ret %o
}
