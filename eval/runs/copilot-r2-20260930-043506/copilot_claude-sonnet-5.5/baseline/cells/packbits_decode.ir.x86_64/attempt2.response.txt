export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	storel 0, %pi
	storel 0, %po
	jmp @loop
@loop
	%i =l loadl %pi
	%c =w csltl %i, %n
	jnz %c, @hdr, @done
@done
	%r =l loadl %po
	ret %r
@hdr
	%i1 =l loadl %pi
	%p =l add %src, %i1
	%hw =w loadsb %p
	%h =l extsw %hw
	%i2 =l add %i1, 1
	storel %i2, %pi
	%isn =w ceql %h, -128
	jnz %isn, @loop, @c1
@c1
	%neg =w csltl %h, 0
	jnz %neg, @rep, @lit
@lit
	%cnt =l add %h, 1
	%end =l add %i2, %cnt
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @litpre
@litpre
	jmp @litl
@litl
	%k =l phi @litpre 0, @litw %k2
	%d =w csltl %k, %cnt
	jnz %d, @litc, @litdone
@litdone
	storel %end, %pi
	jmp @loop
@litc
	%o =l loadl %po
	%f =w csgel %o, %cap
	jnz %f, @err1, @litw
@litw
	%a =l add %i2, %k
	%sp =l add %src, %a
	%b =w loadub %sp
	%dp =l add %out, %o
	storeb %b, %dp
	%o2 =l add %o, 1
	storel %o2, %po
	%k2 =l add %k, 1
	jmp @litl
@rep
	%cnt2 =l sub 1, %h
	%need =l add %i2, 1
	%bad2 =w csgtl %need, %n
	jnz %bad2, @err2, @repr
@repr
	%sp2 =l add %src, %i2
	%rb =w loadub %sp2
	storel %need, %pi
	jmp @repl
@repl
	%k3 =l phi @repr 0, @repw %k4
	%d2 =w csltl %k3, %cnt2
	jnz %d2, @repc, @loop
@repc
	%o3 =l loadl %po
	%f2 =w csgel %o3, %cap
	jnz %f2, @err1, @repw
@repw
	%dp2 =l add %out, %o3
	storeb %rb, %dp2
	%o4 =l add %o3, 1
	storel %o4, %po
	%k4 =l add %k3, 1
	jmp @repl
@err1
	ret -1
@err2
	ret -2
}
