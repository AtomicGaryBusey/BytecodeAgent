packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx eax, byte ptr [rdi + r8]
    inc r8
    cmp eax, -128
    je .Lloop
    test eax, eax
    js .Lrepeat
    lea r10, [rax + 1]
    mov r11, rsi
    sub r11, r8
    cmp r11, r10
    jl .Lerr2
.Lcopy:
    test r10, r10
    jz .Lloop
    cmp r9, rcx
    jge .Lerr1
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jmp .Lcopy
.Lrepeat:
    mov r10d, 1
    sub r10d, eax
    cmp r8, rsi
    jge .Lerr2
    movzx eax, byte ptr [rdi + r8]
    inc r8
.Lrep2:
    test r10, r10
    jz .Lloop
    cmp r9, rcx
    jge .Lerr1
    mov byte ptr [rdx + r9], al
    inc r9
    dec r10
    jmp .Lrep2
.Ldone:
    mov rax, r9
    ret
.Lerr1:
    mov rax, -1
    ret
.Lerr2:
    mov rax, -2
    ret
