export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l loadl %a
	%r0 =l call %f(l %y0)
	%o0 =l and %r0, 1
	jmp @check
@check
	%i =l phi @start, 1, @body, %i2
	%mn =l phi @start, %r0, @body, %mn2
	%mx =l phi @start, %r0, @body, %mx2
	%sm =l phi @start, %r0, @body, %sm2
	%odd =l phi @start, %o0, @body, %odd2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	%mn2 =l call $dummy_never(l 0)
	jmp @end
@end
	storel %mn, %out
	ret %odd
}
