export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%r0 =l call %f(l %x0)
	%o0 =l and %r0, 1
	jmp @check
@check
	%i =l phi @start, 1, @upd, %i2
	%mn =l phi @start, %r0, @upd, %mn2
	%mx =l phi @start, %r0, @upd, %mx2
	%sm =l phi @start, %r0, @upd, %sm2
	%odd =l phi @start, %o0, @upd, %odd2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	jnz %lt, @setmin, @nomin
@setmin
	jmp @mid
@nomin
	jmp @mid
@mid
	%mn2 =l phi @setmin, %y, @nomin, %mn
	%gt =w csltl %mx, %y
	jnz %gt, @setmax, @nomax
@setmax
	jmp @upd
@nomax
	jmp @upd
@upd
	%mx2 =l phi @setmax, %y, @nomax, %mx
	%sm2 =l add %sm, %y
	%b =l and %y, 1
	%odd2 =l add %odd, %b
	%i2 =l add %i, 1
	jmp @check
@end
	storel %mn, %out
	%o1 =l add %out, 8
	storel %mx, %o1
	%o2 =l add %out, 16
	storel %sm, %o2
	ret %odd
}
