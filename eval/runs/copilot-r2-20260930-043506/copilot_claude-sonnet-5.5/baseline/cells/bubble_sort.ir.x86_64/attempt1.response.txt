export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
	jmp @outer
@outer
	%m =l sub %n, 1
	%c =w csltl %i, %m
	jnz %c, @inner_init, @done
@inner_init
	%j =l copy 0
	%lim =l sub %m, %i
	jmp @inner
@inner
	%c2 =w csltl %j, %lim
	jnz %c2, @body, @next
@body
	%off =l shl %j, 3
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%g =w csgtl %x, %y
	jnz %g, @swap, @cont
@swap
	storel %y, %p
	storel %x, %q
	jmp @cont
@cont
	%j =l add %j, 1
	jmp @inner
@next
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
