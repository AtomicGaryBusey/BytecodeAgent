memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lfwd
    lea rcx, [rsi + rdx*8]
    cmp rdi, rcx
    jae .Lfwd
.Lback:
    mov rcx, qword ptr [rsi + rdx*8 - 8]
    mov qword ptr [rdi + rdx*8 - 8], rcx
    dec rdx
    jnz .Lback
    ret
.Lfwd:
    xor ecx, ecx
.Lfl:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jb .Lfl
.Ldone:
    ret
