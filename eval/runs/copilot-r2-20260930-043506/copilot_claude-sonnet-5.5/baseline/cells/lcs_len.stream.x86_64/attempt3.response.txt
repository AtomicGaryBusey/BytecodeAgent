lcs_len:
    xor eax, eax
    xor r9d, r9d
.Lzero:
    cmp r9, rcx
    jg .Lzdone
    mov qword ptr [r8 + r9*8], 0
    inc r9
    jmp .Lzero
.Lzdone:
    xor r10d, r10d
.Louter:
    cmp r10, rsi
    jge .Ldone
    movzx r11d, byte ptr [rdi + r10]
    xor eax, eax
    mov r9d, 1
.Linner:
    cmp r9, rcx
    jg .Lnext
    mov rax, qword ptr [r8 + r9*8]
    movzx r10d, byte ptr [rdx + r9 - 1]
    cmp r10d, r11d
    mov r10, qword ptr [rsp - 8]
    jmp .Lbad
.Lbad:
    jmp .Lbad
.Lnext:
    jmp .Lnext
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    ret
