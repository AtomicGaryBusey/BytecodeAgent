mp_mul:
    push rbx
    push rbp
    push r12
    mov r9, rdx
    lea r10, [rsi + rcx]
    xor eax, eax
    xor r11d, r11d
.Lz:
    cmp r11, r10
    jge .Lzd
    mov qword ptr [r8 + r11*8], rax
    inc r11
    jmp .Lz
.Lzd:
    test rsi, rsi
    jz .Lcount
    test rcx, rcx
    jz .Lcount
    xor r11d, r11d
.Louter:
    mov rdx, qword ptr [r9 + r11*8]
    lea rbx, [r8 + r11*8]
    xor r12d, r12d
    xor r10d, r10d
.Linner:
    mulx rbp, rax, qword ptr [rdi + r10*8]
    add rax, r12
    adc rbp, 0
    add rax, qword ptr [rbx + r10*8]
    adc rbp, 0
    mov qword ptr [rbx + r10*8], rax
    mov r12, rbp
    inc r10
    cmp r10, rsi
    jl .Linner
    mov qword ptr [rbx + rsi*8], r12
    inc r11
    cmp r11, rcx
    jl .Louter
.Lcount:
    lea rax, [rsi + rcx]
.Lc:
    test rax, rax
    jz .Ldone
    cmp qword ptr [r8 + rax*8 - 8], 0
    jne .Ldone
    dec rax
    jmp .Lc
.Ldone:
    pop r12
    pop rbp
    pop rbx
    ret
