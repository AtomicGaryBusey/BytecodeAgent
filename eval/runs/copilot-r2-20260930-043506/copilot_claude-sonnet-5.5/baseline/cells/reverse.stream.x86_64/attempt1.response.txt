reverse:
    xor eax, eax
    lea rcx, [rsi - 1]
    xor rdx, rdx
.Lloop:
    cmp rdx, rcx
    jge .Ldone
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rcx*8]
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rcx*8], r8
    inc rdx
    dec rcx
    jmp .Lloop
.Ldone:
    ret
