export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l loadl %a
	%r0 =l call %f(l %y0)
	%odd0 =l and %r0, 1
	%i1 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i1, @body %inext
	%min =l phi @start %r0, @body %min2
	%max =l phi @start %r0, @body %max2
	%sum =l phi @start %r0, @body %sum2
	%odd =l phi @start %odd0, @body %odd2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%gt =w csgtl %y, %max
	%ltl =l extsw %lt
	%gtl =l extsw %gt
	%dmin =l sub %y, %min
	%dmax =l sub %y, %max
	%m1 =l mul %dmin, %ltl
	%min2 =l add %min, %m1
	%m2 =l mul %dmax, %gtl
	%max2 =l add %max, %m2
	%sum2 =l add %sum, %y
	%b =l and %y, 1
	%odd2 =l add %odd, %b
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %odd
}
