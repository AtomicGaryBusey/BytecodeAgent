export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @fwd, @chk
@chk
	%z =w cslel %n, 0
	jnz %z, @done, @bwdinit
@bwdinit
	%i =l copy %n
	jmp @bwdloop
@bwdloop
	%i1 =l phi @bwdinit %i, @bwdbody %i2
	%z2 =w cslel %i1, 0
	jnz %z2, @done, @bwdbody
@bwdbody
	%i2 =l sub %i1, 1
	%o =l shl %i2, 3
	%s =l add %src, %o
	%d =l add %dst, %o
	%v =l loadl %s
	storel %v, %d
	jmp @bwdloop
@fwd
	jmp @fwdloop
@fwdloop
	%j =l phi @fwd 0, @fwdbody %j2
	%c =w csltl %j, %n
	jnz %c, @fwdbody, @done
@fwdbody
	%o2 =l shl %j, 3
	%s2 =l add %src, %o2
	%d2 =l add %dst, %o2
	%v2 =l loadl %s2
	storel %v2, %d2
	%j2 =l add %j, 1
	jmp @fwdloop
@done
	ret 0
}
