crc32:
    mov eax, 0xFFFFFFFF
    xor rcx, rcx
.Lbyte:
    cmp rcx, rsi
    jge .Ldone
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit:
    mov edx, eax
    shr eax, 1
    and edx, 1
    neg edx
    and edx, 0xEDB88320
    xor eax, edx
    dec r8d
    jnz .Lbit
    inc rcx
    jmp .Lbyte
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
