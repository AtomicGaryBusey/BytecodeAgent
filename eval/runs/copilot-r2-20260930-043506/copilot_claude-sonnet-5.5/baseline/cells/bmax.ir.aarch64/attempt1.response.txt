export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @body %i2
	%m =l phi @start %m0, @body %m2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%g =w csgtl %v, %m
	%d =l sub %v, %m
	%g2 =l extsw %g
	%neg =l sub 0, %g2
	%dm =l and %d, %neg
	%m2 =l add %m, %dm
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %m
}
