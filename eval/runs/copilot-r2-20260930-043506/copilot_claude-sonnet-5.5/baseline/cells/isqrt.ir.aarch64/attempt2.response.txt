export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%bit =l phi @start 4611686018427387904, @next %bit2
	%res =l phi @start 0, @next %res2
	%n1 =l phi @start %n, @next %n2
	%t =l add %res, %bit
	%c =w culel %t, %n1
	jnz %c, @yes, @no
@yes
	%na =l sub %n1, %t
	%rh =l shr %res, 1
	%ra =l add %rh, %bit
	jmp @next
@no
	%rb =l shr %res, 1
	jmp @next
@next
	%n2 =l phi @yes %na, @no %n1
	%res2 =l phi @yes %ra, @no %rb
	%bit2 =l shr %bit, 2
	%z =w ceql %bit2, 0
	jnz %z, @done, @loop
@done
	ret %res2
}
