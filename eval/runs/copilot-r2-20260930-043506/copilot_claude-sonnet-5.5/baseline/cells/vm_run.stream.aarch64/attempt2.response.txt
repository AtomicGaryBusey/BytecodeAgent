The jump table was the bug. `adr x12, .Ltab` followed by `add x12, x12, x11, lsl #3` steps 8 bytes per entry, but each `b` is 4 bytes, so the dispatch landed on the wrong entries. I fixed it by scaling the opcode by 4 (`lsl #2`).

The failing loads were at `.Ladd`, which was reached for opcode 1. Because the entry stride was wrong, `.Lpush` was skipped.

The `.Ltab` label is also placed right after `br`, and `adr` to it is fine. There is no data in the table, only branch instructions, so this is allowed.

The other paths look right:
- **Zeroing:** the stack is zeroed at the start, and every pop zeroes the slot it vacates. This keeps the final buffer correct.
- **Bounds:** JZ and JMP targets are in [0, len], and the loop checks `pc >= len`.

I used `cmp x9, x1` with `b.ge`, which is a signed compare and works because pc and len are small and non-negative.

```

    mov x9, #0
    mov x10, #0
    mov x11, #0
.Lzero:
    cmp x11, x3
    b.ge .Lloop
    str xzr, [x2, x11, lsl #3]

    b .Lzero

    str x14, [x2, x12, lsl #3]

    sub x12, x10, #1
    ldr x13, [x2, x12, lsl #3]

    add x9, x9, #1
    b .Lloop
.Lover:
    sub x12, x10, #2
    ldr x13, [x2, x12, lsl #3]
    str x13, [x2, x10, lsl #3]
    add x10, x10, #1
    add x9, x9, #1
    b .Lloop
.Lswap:
    sub x12, x10, #1
    sub x14, x10, #2
    ldr x13, [x2, x12, lsl #3]
    ldr x15, [x2, x14, lsl #3]
    str x13, [x2, x14, lsl #3]
    str x15, [x2, x12, lsl #3]
    add x9, x9, #1
    b .Lloop
.Ldrop:
    sub x10, x10, #1
    str xzr, [x2, x10, lsl #3]
    add x9, x9, #1
    b .Lloop
.Ljz:
    sub x10, x10, #1
    ldr x13, [x2, x10, lsl #3]
    str xzr, [x2, x10, lsl #3]

    ldr x14, [x0, x12, lsl #3]

    cmp x13, #0
    csel x9, x14, x9, eq
    b .Lloop
.Ljmp:

    ldr x9, [x0, x12, lsl #3]
    b .Lloop
.Ldone:
    mov x0, #0
    cbz x10, .Lret
    sub x12, x10, #1
    ldr x0, [x2, x12, lsl #3]
.Lret:
    ret
```
