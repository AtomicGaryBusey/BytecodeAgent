export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @initc
@initc
	%j =l phi @start, 0, @initb, %j1
	%c =w cslel %j, %n
	jnz %c, @initb, @outer
@initb
	%o =l shl %j, 3
	%p =l add %row, %o
	storel 0, %p
	%j1 =l add %j, 1
	jmp @initc
@outer
	%i =l phi @initc, 0, @oend, %i2
	%c2 =w csltl %i, %m
	jnz %c2, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%k =l phi @obody, 1, @icont, %k2
	%diag =l phi @obody, 0, @icont, %up
	%c3 =w cslel %k, %n
	jnz %c3, @ibody, @oend
@ibody
	%ko =l shl %k, 3
	%rp =l add %row, %ko
	%up =l loadl %rp
	%yo =l add %y, %k
	%yp =l sub %yo, 1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @yes, @no
@yes
	%v1 =l add %diag, 1
	jmp @store
@no
	%lp =l sub %rp, 8
	%left =l loadl %lp
	%gt =w csgtl %left, %up
	jnz %gt, @l, @u
@l
	jmp @store2
@u
	jmp @store2
@store2
	%v2 =l phi @l, %left, @u, %up
	jmp @store
@store
	%v =l phi @yes, %v1, @store2, %v2
	storel %v, %rp
	jmp @icont
@icont
	%k2 =l add %k, 1
	jmp @inner
@oend
	%i2 =l add %i, 1
	jmp @outer
@done
	%no =l shl %n, 3
	%np =l add %row, %no
	%r =l loadl %np
	ret %r
}
