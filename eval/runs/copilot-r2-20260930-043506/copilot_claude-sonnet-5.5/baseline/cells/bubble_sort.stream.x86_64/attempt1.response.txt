bubble_sort:
    xor eax, eax
    mov rcx, rsi
.Louter:
    cmp rcx, 2
    jl .Ldone
    xor rdx, rdx
    lea r8, [rcx - 1]
.Linner:
    cmp rdx, r8
    jge .Lnext
    mov r9, qword ptr [rdi + rdx*8]
    mov r10, qword ptr [rdi + rdx*8 + 8]
    cmp r9, r10
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r10
    mov qword ptr [rdi + rdx*8 + 8], r9
.Lnoswap:
    inc rdx
    jmp .Linner
.Lnext:
    dec rcx
    jmp .Louter
.Ldone:
    ret
