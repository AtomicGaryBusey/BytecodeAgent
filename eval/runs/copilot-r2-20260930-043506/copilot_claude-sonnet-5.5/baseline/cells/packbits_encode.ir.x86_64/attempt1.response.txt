Design: a run function is inlined as a loop, used in both places.

Main loop: i = 0, o = 0.
- If i >= n, return o.
- Compute run(i): r=1; while i+r<n && r<128 && src[i+r]==src[i], r++.
- If r>=3: need o+2<=cap, else return -1. Write header (257-r) as a byte (storeb truncates), write byte, o+=2, i+=r.
- Else literal: j=i; loop while j<n and j-i<128 and run(j)<3. Run(j)<3 means not (j+2<n and src[j+1]==src[j] and src[j+2]==src[j]). The cap of 128 doesn't matter for a test against 3. After finding j, L=j-i (at least 1, since run(i)<3 holds at j=i). Need o+1+L<=cap else return -1. Write header L-1, copy bytes, o+=1+L, i=j.

Copy loop: k from 0 to L, out[o+1+k]=src[i+k].

QBE code follows. Temps must be single-assignment, so mutable state is kept in memory slots or via phi. I use phis.


export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @main_back_r %i2, @main_back_l %j
	%o =l phi @start 0, @main_back_r %o2, @main_back_l %o3
	%c =w csltl %i, %n
	jnz %c, @body, @done
@done
	ret %o
@body
	%p =l add %src, %i
	%b =w loadub %p
	jmp @rl
@rl
	%r =l phi @body 1, @rl2 %r1
	%k =l add %i, %r
	%c1 =w csltl %k, %n
	jnz %c1, @rl1, @rend
@rl1
	%c2 =w csltl %r, 128
	jnz %c2, @rl3, @rend
@rl3
	%q =l add %src, %k
	%b2 =w loadub %q
	%c3 =w ceqw %b2, %b
	jnz %c3, @rl2, @rend
@rl2
	%r1 =l add %r, 1
	jmp @rl
@rend
	%c4 =w csgel %r, 3
	jnz %c4, @rep, @lit
@rep
	%o1 =l add %o, 2
	%c5 =w cslel %o1, %cap
	jnz %c5, @rep2, @fail
@rep2
	%h =l sub 257, %r
	%po =l add %out, %o
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o2 =l copy %o1
	%i2 =l add %i, %r
	jmp @main_back_r
@main_back_r
	jmp @main
@fail
	ret -1
@lit
	jmp @ll
@ll
	%j =l phi @lit %i, @ll_next %j1
	%d1 =w csltl %j, %n
	jnz %d1, @ll1, @lend
@ll1
	%dd =l sub %j, %i
	%d2 =w csltl %dd, 128
	jnz %d2, @ll2, @lend
@ll2
	%j2 =l add %j, 2
	%d3 =w csltl %j2, %n
	jnz %d3, @ll3, @ll_next
@ll3
	%pj =l add %src, %j
	%x0 =w loadub %pj
	%pj1 =l add %pj, 1
	%x1 =w loadub %pj1
	%pj2 =l add %pj, 2
	%x2 =w loadub %pj2
	%e1 =w ceqw %x0, %x1
	%e2 =w ceqw %x0, %x2
	%e =w and %e1, %e2
	jnz %e, @lend, @ll_next
@ll_next
	%j1 =l add %j, 1
	jmp @ll
@lend
	%L =l sub %j, %i
	%t =l add %o, %L
	%t1 =l add %t, 1
	%c6 =w cslel %t1, %cap
	jnz %c6, @lit2, @fail
@lit2
	%hh =l sub %L, 1
	%pw =l add %out, %o
	storeb %hh, %pw
	jmp @cp
@cp
	%k2 =l phi @lit2 0, @cp2 %k3
	%c7 =w csltl %k2, %L
	jnz %c7, @cp2, @cpend
@cp2
	%s1 =l add %i, %k2
	%s2 =l add %src, %s1
	%v =w loadub %s2
	%d5 =l add %o, 1
	%d6 =l add %d5, %k2
	%d7 =l add %out, %d6
	storeb %v, %d7
	%k3 =l add %k2, 1
	jmp @cp
@cpend
	%o3 =l copy %t1
	jmp @main_back_l
@main_back_l
	jmp @main
}
