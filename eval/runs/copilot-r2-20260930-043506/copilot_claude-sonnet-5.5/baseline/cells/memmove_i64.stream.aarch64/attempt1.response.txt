memmove_i64:

    cmp x0, x1
    b.eq .Ldone
    b.hi .Lbackward
.Lfwd:
    ldr x9, [x1], #8
    str x9, [x0], #8
    subs x2, x2, #1
    b.ne .Lfwd
    b .Ldone
.Lbackward:
    sub x10, x2, #1
.Lbloop:
    ldr x9, [x1, x10, lsl #3]
    str x9, [x0, x10, lsl #3]
    subs x10, x10, #1
    b.hs .Lbloop
.Ldone:

    ret
