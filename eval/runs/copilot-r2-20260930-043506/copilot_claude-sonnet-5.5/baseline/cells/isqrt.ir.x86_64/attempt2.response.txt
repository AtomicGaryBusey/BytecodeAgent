export function l $isqrt(l %n) {
@start
	jmp @l1
@l1
	%bit =l phi @start 4611686018427387904, @shr %bit2
	%c =w cultl %n, %bit
	jnz %c, @shr, @l2
@shr
	%bit2 =l shr %bit, 2
	jmp @l1
@l2
	%b =l phi @l1 %bit, @next %bn
	%r =l phi @l1 0, @next %rn
	%m =l phi @l1 %n, @next %mn
	%z =w ceql %b, 0
	jnz %z, @end, @body
@body
	%s =l add %r, %b
	%ge =w cugel %m, %s
	%r1 =l shr %r, 1
	jnz %ge, @yes, @no
@yes
	%m2 =l sub %m, %s
	%r2 =l add %r1, %b
	jmp @next
@no
	jmp @next
@next
	%mn =l phi @yes %m2, @no %m
	%rn =l phi @yes %r2, @no %r1
	%bn =l shr %b, 2
	jmp @l2
@end
	ret %r
}
