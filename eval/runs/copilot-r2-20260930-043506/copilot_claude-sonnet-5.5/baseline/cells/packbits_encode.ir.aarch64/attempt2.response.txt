export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start, 0, @rep, %i1, @litend, %j
	%o =l phi @start, 0, @rep, %o1, @litend, %on
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@done
	ret %o
@body
	%p =l add %src, %i
	%b =w loadub %p
	jmp @rl
@rl
	%r =l phi @body, 1, @rlb, %r2
	%k =l add %i, %r
	%c1 =w csltl %k, %n
	jnz %c1, @rl2, @rend
@rl2
	%c2 =w csltl %r, 128
	jnz %c2, @rl3, @rend
@rl3
	%q =l add %src, %k
	%bb =w loadub %q
	%c3 =w ceqw %bb, %b
	jnz %c3, @rlb, @rend
@rlb
	%r2 =l add %r, 1
	jmp @rl
@rend
	%c4 =w csltl %r, 3
	jnz %c4, @lit, @repc
@repc
	%need =l add %o, 2
	%c5 =w cslel %need, %cap
	jnz %c5, @rep, @fail
@fail
	ret -1
@rep
	%h =l sub 257, %r
	%po =l add %out, %o
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%i1 =l add %i, %r
	%o1 =l add %o, 2
	jmp @loop
@lit
	jmp @sl
@sl
	%j =l phi @lit, %i, @slb, %j2
	%d1 =w csltl %j, %n
	jnz %d1, @sl2, @slend
@sl2
	%dd =l sub %j, %i
	%d2 =w csltl %dd, 128
	jnz %d2, @sl3, @slend
@sl3
	%j2s =l add %j, 2
	%d3 =w csltl %j2s, %n
	jnz %d3, @sl4, @slb
@sl4
	%pj =l add %src, %j
	%x0 =w loadub %pj
	%pj1 =l add %pj, 1
	%x1 =w loadub %pj1
	%pj2 =l add %pj, 2
	%x2 =w loadub %pj2
	%e1 =w ceqw %x0, %x1
	%e2 =w ceqw %x0, %x2
	%e =w and %e1, %e2
	jnz %e, @slend, @slb
@slb
	%j2 =l add %j, 1
	jmp @sl
@slend
	%L =l sub %j, %i
	%o3 =l add %o, %L
	%on =l add %o3, 1
	%c6 =w cslel %on, %cap
	jnz %c6, @lw, @fail
@lw
	%pl =l add %out, %o
	%hl =l sub %L, 1
	storeb %hl, %pl
	jmp @cl
@cl
	%cc =l phi @lw, 0, @clb, %cn
	%c7 =w csltl %cc, %L
	jnz %c7, @clb, @litend
@clb
	%sq =l add %src, %i
	%sq2 =l add %sq, %cc
	%v =w loadub %sq2
	%dp =l add %pl, 1
	%dp2 =l add %dp, %cc
	storeb %v, %dp2
	%cn =l add %cc, 1
	jmp @cl
@litend
	jmp @loop
}
