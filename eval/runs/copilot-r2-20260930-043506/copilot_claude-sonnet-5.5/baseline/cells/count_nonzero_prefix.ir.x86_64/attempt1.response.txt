export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%c =w csltl %i, %n
	jnz %c, @check, @done
@check
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @done, @body
@body
	%i2 =l add %i, 1
	jmp @loop
@done
	%r =l phi @loop %i, @check %i
	ret %r
}
