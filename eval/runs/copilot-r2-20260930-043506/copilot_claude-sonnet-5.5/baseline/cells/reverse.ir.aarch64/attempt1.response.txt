export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	jmp @loop
@loop
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%oi =l mul %i, 8
	%pi =l add %a, %oi
	%oj =l mul %j, 8
	%pj =l add %a, %oj
	%x =l loadl %pi
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @next
@next
	%i =l copy %i1
	%j =l copy %j1
	jmp @loop
@done
	ret 0
}
