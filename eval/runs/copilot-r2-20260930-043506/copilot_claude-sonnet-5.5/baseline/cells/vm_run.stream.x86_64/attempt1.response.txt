vm_run:
    push rbx
    push r12
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jge .Lend
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 11
    ja .Lend
    lea r10, [rip + .Ltab]
    movsxd r11, dword ptr [r10 + rax*4]
    add r11, r10
    jmp r11
.Ltab:
    jmp .Lend
.Lend:
    xor rax, rax
    test r9, r9
    jz .Lclr
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lclr:
    mov r10, r9
.Lzl:
    cmp r10, rcx
    jge .Ldone
    mov qword ptr [rdx + r10*8], 0
    inc r10
    jmp .Lzl
.Ldone:
    pop r12
    pop rbx
    ret
