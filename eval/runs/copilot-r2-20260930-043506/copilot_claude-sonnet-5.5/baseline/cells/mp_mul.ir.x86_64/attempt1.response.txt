export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%i0 =l copy 0
	jmp @zc
@zc
	%zi =l phi @start %i0, @zb %zi2
	%zd =w csltl %zi, %n
	jnz %zd, @zb, @outer
@zb
	%zo =l shl %zi, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%zi2 =l add %zi, 1
	jmp @zc
@outer
	%i =l phi @zc %i0, @oend %i2
	%oc =w csltl %i, %na
	jnz %oc, @obody, @fin
@obody
	%ao =l shl %i, 3
	%ap =l add %a, %ao
	%x =l loadl %ap
	%xl =l and %x, 4294967295
	%xh =l shr %x, 32
	jmp @inner
@inner
	%j =l phi @obody %i0, @ibody %j2
	%carry =l phi @obody %i0, @ibody %hi
	%ic =w csltl %j, %nb
	jnz %ic, @ibody, @oend
@ibody
	%bo =l shl %j, 3
	%bp =l add %b, %bo
	%y =l loadl %bp
	%yl =l and %y, 4294967295
	%yh =l shr %y, 32
	%ll =l mul %xl, %yl
	%lh =l mul %xl, %yh
	%hl =l mul %xh, %yl
	%hh =l mul %xh, %yh
	%t =l shr %ll, 32
	%lhl =l and %lh, 4294967295
	%hll =l and %hl, 4294967295
	%m =l add %t, %lhl
	%m =l add %m, %hll
	%lhh =l shr %lh, 32
	%hlh =l shr %hl, 32
	%hi1 =l add %hh, %lhh
	%hi2 =l add %hi1, %hlh
	%mh =l shr %m, 32
	%hi0 =l add %hi2, %mh
	%ml =l shl %m, 32
	%lll =l and %ll, 4294967295
	%lo =l or %ml, %lll
	%k =l add %i, %j
	%ko =l shl %k, 3
	%kp =l add %r, %ko
	%old =l loadl %kp
	%s1 =l add %lo, %old
	%c1 =l cultl %s1, %lo
	%c1l =l extuw %c1
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%c2l =l extuw %c2
	storel %s2, %kp
	%hia =l add %hi0, %c1l
	%hi =l add %hia, %c2l
	%j2 =l add %j, 1
	jmp @inner
@oend
	%k2 =l add %i, %nb
	%ko2 =l shl %k2, 3
	%kp2 =l add %r, %ko2
	storel %carry, %kp2
	%i2 =l add %i, 1
	jmp @outer
@fin
	jmp @cnt
@cnt
	%q =l phi @fin %n, @dec %q2
	%qz =w csgtl %q, 0
	jnz %qz, @chk, @done
@chk
	%qm =l sub %q, 1
	%qo =l shl %qm, 3
	%qp =l add %r, %qo
	%v =l loadl %qp
	%vz =w ceql %v, 0
	jnz %vz, @dec, @done
@dec
	%q2 =l sub %q, 1
	jmp @cnt
@done
	ret %q
}
