export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%i =l copy 0
	jmp @zloop
@zloop
	%c =l csltl %i, %cap
	jnz %c, @zbody, @zdone
@zbody
	%o =l shl %i, 3
	%p =l add %stack, %o
	storel 0, %p
	%i =l add %i, 1
	jmp @zloop
@zdone
	jmp @loop
@loop
	%pc =l phi @zdone 0, @next %pc2
	%sp =l phi @zdone 0, @next %sp2
	%e =l ceql %pc, %len
	jnz %e, @end, @fetch
@fetch
	%po =l shl %pc, 3
	%pa =l add %code, %po
	%op =l loadl %pa
	%pc1 =l add %pc, 1
	%pc2n =l add %pc, 2
	%so =l shl %sp, 3
	%tp =l add %stack, %so
	%t1 =l sub %tp, 8
	%t2 =l sub %tp, 16
	%is0 =w ceql %op, 0
	jnz %is0, @end, @d1
@d1
	%x =w ceql %op, 1
	jnz %x, @push, @d2
@d2
	%x =w ceql %op, 2
	jnz %x, @add, @d3
@d3
	%x =w ceql %op, 3
	jnz %x, @sub, @d4
@d4
	%x =w ceql %op, 4
	jnz %x, @mul, @d5
@d5
	%x =w ceql %op, 5
	jnz %x, @dup, @d6
@d6
	%x =w ceql %op, 6
	jnz %x, @swap, @d7
@d7
	%x =w ceql %op, 7
	jnz %x, @drop, @d8
@d8
	%x =w ceql %op, 8
	jnz %x, @over, @d9
@d9
	%x =w ceql %op, 9
	jnz %x, @lt, @d10
@d10
	%x =w ceql %op, 10
	jnz %x, @jz, @jmp
@push
	%v =l loadl %pc1s
	jmp @next
@end
	%z =l ceql %sp, 0
	jnz %z, @ret0, @retv
@ret0
	ret 0
@retv
	%so2 =l shl %sp, 3
	%q =l add %stack, %so2
	%q =l sub %q, 8
	%r =l loadl %q
	ret %r
@next
	%pc2 =l copy 0
	%sp2 =l copy 0
	jmp @loop
}
