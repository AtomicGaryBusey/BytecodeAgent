export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%sp =l loadl %spp
	%e =w ceql %pc, %len
	jnz %e, @end, @fetch
@fetch
	%po =l shl %pc, 3
	%pa =l add %code, %po
	%op =l loadl %pa
	%so =l shl %sp, 3
	%tp =l add %stack, %so
	%t1 =l sub %tp, 8
	%t2 =l sub %tp, 16
	%spm =l sub %sp, 1
	%spa =l add %sp, 1
	%oa =l add %pa, 8
	%c0 =w ceql %op, 0
	jnz %c0, @end, @d1
@d1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @d2
@d2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @d3
@d3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @d4
@d4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @d5
@d5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @d6
@d6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @d7
@d7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @d8
@d8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @d9
@d9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @d10
@d10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @jmp
@push
	%v =l loadl %oa
	storel %v, %tp
	storel %spa, %spp
	jmp @inc2
@add
	%a1 =l loadl %t2
	%b1 =l loadl %t1
	%r1 =l add %a1, %b1
	storel %r1, %t2
	storel %spm, %spp
	jmp @inc1
@sub
	%a2 =l loadl %t2
	%b2 =l loadl %t1
	%r2 =l sub %a2, %b2
	storel %r2, %t2
	storel %spm, %spp
	jmp @inc1
@mul
	%a3 =l loadl %t2
	%b3 =l loadl %t1
	%r3 =l mul %a3, %b3
	storel %r3, %t2
	storel %spm, %spp
	jmp @inc1
@dup
	%v5 =l loadl %t1
	storel %v5, %tp
	storel %spa, %spp
	jmp @inc1
@swap
	%a6 =l loadl %t2
	%b6 =l loadl %t1
	storel %b6, %t2
	storel %a6, %t1
	jmp @inc1
@drop
	storel %spm, %spp
	jmp @inc1
@over
	%v8 =l loadl %t2
	storel %v8, %tp
	storel %spa, %spp
	jmp @inc1
@lt
	%a9 =l loadl %t2
	%b9 =l loadl %t1
	%w9 =w csltl %a9, %b9
	%r9 =l extuw %w9
	storel %r9, %t2
	storel %spm, %spp
	jmp @inc1
@jz
	%vz =l loadl %t1
	%tz =l loadl %oa
	storel %spm, %spp
	%zz =w ceql %vz, 0
	jnz %zz, @jzt, @inc2
@jzt
	storel %tz, %pcp
	jmp @loop
@jmp
	%tj =l loadl %oa
	storel %tj, %pcp
	jmp @loop
@inc1
	%n1 =l add %pc, 1
	storel %n1, %pcp
	jmp @loop
@inc2
	%n2 =l add %pc, 2
	storel %n2, %pcp
	jmp @loop
@end
	%z =w ceql %sp, 0
	jnz %z, @ret0, @retv
@ret0
	jmp @zl
@retv
	jmp @zl
@zl
	%rv =l phi @ret0 0, @retv %rl
	jmp @zloop
@zloop
	%i =l phi @zl %sp, @zbody %i2
	%cc =w csltl %i, %cap
	jnz %cc, @zbody, @zdone
@zbody
	%zo =l shl %i, 3
	%zp =l add %stack, %zo
	storel 0, %zp
	%i2 =l add %i, 1
	jmp @zloop
@zdone
	ret %rv
}
