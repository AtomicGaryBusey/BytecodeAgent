export function l $qsort_pairs(l %a, l %n) {
@start
	jmp @loop
@loop
	%a1 =l phi @start %a, @fin %an
	%n1 =l phi @start %n, @fin %nn
	%c =w csltl %n1, 2
	jnz %c, @done, @part
@part
	%last =l sub %n1, 1
	%lo =l shl %last, 4
	%pp =l add %a1, %lo
	%pk =l loadl %pp
	%pvp =l add %pp, 8
	%pv =l loadl %pvp
	jmp @jl
@jl
	%i =l phi @part 0, @jl2 %i2
	%j =l phi @part 0, @jl2 %j2
	%cj =w csltl %j, %last
	jnz %cj, @body, @fin
@body
	%jo =l shl %j, 4
	%jp =l add %a1, %jo
	%kj =l loadl %jp
	%jvp =l add %jp, 8
	%vj =l loadl %jvp
	%lt =w csltl %kj, %pk
	%eq =w ceql %kj, %pk
	%vl =w csltl %vj, %pv
	%e2 =w and %eq, %vl
	%r =w or %lt, %e2
	jnz %r, @swap, @jl2n
@swap
	%io =l shl %i, 4
	%ip =l add %a1, %io
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %kj, %ip
	storel %vj, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%i3 =l add %i, 1
	jmp @jl2
@jl2n
	jmp @jl2
@jl2
	%i2 =l phi @swap %i3, @jl2n %i
	%j2 =l add %j, 1
	jmp @jl
@fin
	%fo =l shl %i, 4
	%fp =l add %a1, %fo
	%fk =l loadl %fp
	%fvp =l add %fp, 8
	%fv =l loadl %fvp
	storel %pk, %fp
	storel %pv, %fvp
	storel %fk, %pp
	storel %fv, %pvp
	%rr =l call $qsort_pairs(l %a1, l %i)
	%i4 =l add %i, 1
	%off =l shl %i4, 4
	%an =l add %a1, %off
	%nn =l sub %n1, %i4
	jmp @loop
@done
	ret 0
}
