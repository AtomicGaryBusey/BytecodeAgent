Design: Lomuto partition with the last record as pivot. Recurse on the left part, loop on the right part. Lomuto with all-equal input degrades to O(n^2), which is fine for n ≤ 150. Recursion depth is also fine, since the left part is recursed on and the right part is looped on. Worst-case depth is at most 150 frames, which is fine.

Algorithm, with a pointer and a count:
- while n >= 2:
  - pivot key pk = a[2(n-1)], pivot value pv = a[2(n-1)+1]
  - i = 0
  - for j in 0..n-1 (excluding the last): if (key_j, val_j) < (pk, pv), swap record j with record i and increment i
  - swap record i with record n-1; the pivot is now at index i
  - recurse on (a, i)
  - a = a + 16*(i+1), n = n-i-1
- return 0

The less-than test is (kj < pk) or (kj == pk and vj < pv), computed as `csltl`, `ceql`, `csltl`, then `and`, then `or`.

The pivot values are kept in temporaries across the call, but they are recomputed on each loop iteration, so no state needs to survive the call other than a and n. QBE allocates registers across calls itself, so I store nothing manually.


export function l $qsort_pairs(l %a, l %n) {
@start
	%a1 =l copy %a
	%n1 =l copy %n
	jmp @loop
@loop
	%c =w csltl %n1, 2
	jnz %c, @done, @part
@part
	%last =l sub %n1, 1
	%lo =l shl %last, 4
	%pp =l add %a1, %lo
	%pk =l loadl %pp
	%pvp =l add %pp, 8
	%pv =l loadl %pvp
	jmp @jl
@jl
	%i =l phi @part 0, @jl2 %i2
	%j =l phi @part 0, @jl2 %j2
	%cj =w csltl %j, %last
	jnz %cj, @body, @fin
@body
	%jo =l shl %j, 4
	%jp =l add %a1, %jo
	%kj =l loadl %jp
	%jvp =l add %jp, 8
	%vj =l loadl %jvp
	%lt =w csltl %kj, %pk
	%eq =w ceql %kj, %pk
	%vl =w csltl %vj, %pv
	%e2 =w and %eq, %vl
	%r =w or %lt, %e2
	jnz %r, @swap, @jl2n
@swap
	%io =l shl %i, 4
	%ip =l add %a1, %io
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %kj, %ip
	storel %vj, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%i3 =l add %i, 1
	jmp @jl2
@jl2n
	jmp @jl2
@jl2
	%i2 =l phi @swap %i3, @jl2n %i
	%j2 =l add %j, 1
	jmp @jl
@fin
	%fo =l shl %i, 4
	%fp =l add %a1, %fo
	%fk =l loadl %fp
	%fvp =l add %fp, 8
	%fv =l loadl %fvp
	storel %pk, %fp
	storel %pv, %fvp
	storel %fk, %pp
	storel %fv, %pvp
	%rr =l call $qsort_pairs(l %a1, l %i)
	%i4 =l add %i, 1
	%off =l shl %i4, 4
	%an =l add %a1, %off
	%nn =l sub %n1, %i4
	%a1 =l copy %an
	%n1 =l copy %nn
	jmp @loop
@done
	ret 0
}
