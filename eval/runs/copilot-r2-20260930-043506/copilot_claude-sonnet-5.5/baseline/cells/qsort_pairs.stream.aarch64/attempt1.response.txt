Design: recursive quicksort with Lomuto partition, using the last record as pivot. Recurse on the left part and loop on the right part. Worst-case depth is at most 150 frames, which is fine.

Registers: x19 = a, x20 = n, x21 = pivot key, x22 = pivot value, x23 = i (store index), x24 = j.

Loop: if n < 2, return 0. Load the pivot from record n-1. Set i = 0. For j in 0..n-2, if record j < pivot (key less, or key equal and value less), swap record j with record i and increment i. Finally swap record i with record n-1. The pivot ends at index i. Recurse on (a, i), then set a += (i+1)*16 and n -= i+1, and loop.

Comparison of record j (k, v) with pivot (pk, pv): if k < pk, less. If k == pk and v < pv, less. Otherwise not less. Using flags: `cmp k, pk; b.lt less; b.ne next; cmp v, pv; b.ge next`.

Swap of records j and i: load both pairs and store them crosswise. If i == j the swap is harmless.

```
qsort_pairs:
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    mov x19, x0
    mov x20, x1
.Ltop:
    cmp x20, #2
    b.lt .Ldone
    sub x9, x20, #1
    add x9, x19, x9, lsl #4
    ldp x21, x22, [x9]
    mov x23, #0
    mov x24, #0
    sub x10, x20, #1
.Lloop:
    cmp x24, x10
    b.ge .Lend
    add x11, x19, x24, lsl #4
    ldp x12, x13, [x11]
    cmp x12, x21
    b.lt .Lless
    b.ne .Lnext
    cmp x13, x22
    b.ge .Lnext
.Lless:
    add x14, x19, x23, lsl #4
    ldp x15, x9, [x14]
    stp x12, x13, [x14]
    stp x15, x9, [x11]
    add x23, x23, #1
.Lnext:
    add x24, x24, #1
    b .Lloop
.Lend:
    add x14, x19, x23, lsl #4
    ldp x12, x13, [x14]
    sub x9, x20, #1
    add x9, x19, x9, lsl #4
    stp x21, x22, [x14]
    stp x12, x13, [x9]
    mov x0, x19
    mov x1, x23
    bl qsort_pairs
    add x23, x23, #1
    add x19, x19, x23, lsl #4
    sub x20, x20, x23
    b .Ltop
.Ldone:
    mov x0, #0
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x23, x24, [sp, #48]
    ldp x29, x30, [sp], #64
    ret
```
