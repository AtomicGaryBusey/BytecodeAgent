export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%ks =l alloc8 8
	%is =l alloc8 8
	%js =l alloc8 8
	%ts =l alloc8 8
	%ss =l alloc8 8
	%xs =l alloc8 8
	%ys =l alloc8 8
	%ds =l alloc8 8
	%rs =l alloc8 8
	%nn =l mul %n, %n
	%off =l shl %nn, 3
	%P2 =l add %tmp, %off
	storel %k, %ks
	storel 0, %is
@zl
	%i =l loadl %is
	%c =w csltl %i, %nn
	jnz %c, @zb, @cpstart
@zb
	%zo =l shl %i, 3
	%zp =l add %out, %zo
	storel 0, %zp
	%zq =l add %tmp, %zo
	%zsrc =l add %a, %zo
	%zv =l loadl %zsrc
	storel %zv, %zq
	%zi1 =l add %i, 1
	storel %zi1, %is
	jmp @zl
@cpstart
	storel 0, %is
@il
	%ii =l loadl %is
	%ic =w csltl %ii, %n
	jnz %ic, @ib, @loop
@ib
	%id =l mul %ii, %n
	%id2 =l add %id, %ii
	%io =l shl %id2, 3
	%ip =l add %out, %io
	storel 1, %ip
	%ii1 =l add %ii, 1
	storel %ii1, %is
	jmp @il
@loop
	%k0 =l loadl %ks
	%lc =w ceql %k0, 0
	jnz %lc, @fin, @chk
@chk
	%kb =l loadl %ks
	%b =l and %kb, 1
	%bc =w ceql %b, 1
	jnz %bc, @s1, @after1
@s1
	storel %out, %xs
	storel %tmp, %ys
	storel %out, %ds
	storel 1, %rs
	jmp @mul
@after1
	%k1 =l loadl %ks
	%k2 =l shr %k1, 1
	storel %k2, %ks
	%ac =w ceql %k2, 0
	jnz %ac, @fin, @s2
@s2
	storel %tmp, %xs
	storel %tmp, %ys
	storel %tmp, %ds
	storel 2, %rs
	jmp @mul
@mul
	storel 0, %is
@mi
	%mi_i =l loadl %is
	%mic =w csltl %mi_i, %n
	jnz %mic, @mi2, @cp
@mi2
	storel 0, %js
@mj
	%mj_j =l loadl %js
	%mjc =w csltl %mj_j, %n
	jnz %mjc, @mj2, @mjend
@mj2
	storel 0, %ts
	storel 0, %ss
@mt
	%t =l loadl %ts
	%tc =w csltl %t, %n
	jnz %tc, @mt2, @mtend
@mt2
	%i2 =l loadl %is
	%j2 =l loadl %js
	%X =l loadl %xs
	%Y =l loadl %ys
	%r1 =l mul %i2, %n
	%e1 =l add %r1, %t
	%o1 =l shl %e1, 3
	%p1 =l add %X, %o1
	%v1 =l loadl %p1
	%r2 =l mul %t, %n
	%e2 =l add %r2, %j2
	%o2 =l shl %e2, 3
	%p2 =l add %Y, %o2
	%v2 =l loadl %p2
	%m =l mul %v1, %v2
	%s =l loadl %ss
	%sn =l add %s, %m
	storel %sn, %ss
	%t1 =l add %t, 1
	storel %t1, %ts
	jmp @mt
@mtend
	%i3 =l loadl %is
	%j3 =l loadl %js
	%r =l mul %i3, %n
	%e =l add %r, %j3
	%o =l shl %e, 3
	%p =l add %P2, %o
	%s3 =l loadl %ss
	storel %s3, %p
	%j4 =l add %j3, 1
	storel %j4, %js
	jmp @mj
@mjend
	%i5 =l loadl %is
	%i6 =l add %i5, 1
	storel %i6, %is
	jmp @mi
@cp
	storel 0, %is
@cl
	%ci =l loadl %is
	%cc =w csltl %ci, %nn
	jnz %cc, @cb, @cdone
@cb
	%co =l shl %ci, 3
	%cs =l add %P2, %co
	%cv =l loadl %cs
	%D =l loadl %ds
	%cd =l add %D, %co
	storel %cv, %cd
	%ci1 =l add %ci, 1
	storel %ci1, %is
	jmp @cl
@cdone
	%rr =l loadl %rs
	%rc =w ceql %rr, 1
	jnz %rc, @after1, @after2
@after2
	jmp @loop
@fin
	%nn2 =l shl %nn, 1
	storel 0, %is
@fl
	%fi =l loadl %is
	%fc =w csltl %fi, %nn2
	jnz %fc, @fb, @done
@fb
	%fo =l shl %fi, 3
	%fp =l add %tmp, %fo
	storel 0, %fp
	%fi1 =l add %fi, 1
	storel %fi1, %is
	jmp @fl
@done
	ret 0
}
