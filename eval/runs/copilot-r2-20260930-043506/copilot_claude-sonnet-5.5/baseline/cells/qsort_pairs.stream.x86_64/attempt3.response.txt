qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    cmp r13, 2
    jl .Ldone
    mov r14, r13
    dec r14
    shl r14, 4
    add r14, r12
    mov r8, qword ptr [r14]
    mov r9, qword ptr [r14 + 8]
    xor r15, r15
    xor rbx, rbx
.Lloop:
    mov rax, rbx
    shl rax, 4
    add rax, r12
    mov rcx, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    cmp rcx, r8
    jl .Lless
    jg .Lnext
    cmp rdx, r9
    jge .Lnext
.Lless:
    mov r10, r15
    shl r10, 4
    add r10, r12
    mov r11, qword ptr [r10]
    mov rsi, qword ptr [r10 + 8]
    mov qword ptr [r10], rcx
    mov qword ptr [r10 + 8], rdx
    mov qword ptr [rax], r11
    mov qword ptr [rax + 8], rsi
    inc r15
.Lnext:
    inc rbx
    mov rax, r13
    dec rax
    cmp rbx, rax
    jl .Lloop
    mov r10, r15
    shl r10, 4
    add r10, r12
    mov rcx, qword ptr [r10]
    mov rdx, qword ptr [r10 + 8]
    mov qword ptr [r10], r8
    mov qword ptr [r10 + 8], r9
    mov qword ptr [r14], rcx
    mov qword ptr [r14 + 8], rdx
    mov rdi, r12
    mov rsi, r15
    call qsort_pairs
    mov rax, r15
    shl rax, 4
    lea rdi, [r12 + rax + 16]
    mov rsi, r13
    sub rsi, r15
    dec rsi
    call qsort_pairs
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
