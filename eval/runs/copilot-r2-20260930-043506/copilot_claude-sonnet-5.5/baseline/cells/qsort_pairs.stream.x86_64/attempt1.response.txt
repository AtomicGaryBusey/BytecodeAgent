Design: Lomuto partition with the last record as pivot, then recurse on both sides. Worst-case depth is at most 150 frames, which is fine.

Registers: rbx = a, r12 = n, r13 = pivot key, r14 = pivot value, r15 = i (store index), rbp = j. That is 6 pushes (48 bytes) plus the return address, so rsp is 8 mod 16 + 48 = 8 mod 16 — misaligned, so I add `sub rsp, 8`.

Entry: rsp ≡ 8 (mod 16). After 6 pushes, rsp ≡ 8+48 = 56 ≡ 8 (mod 16). After `sub rsp, 8` it is 0 mod 16, which is aligned.

Algorithm:
- If n < 2, return 0.
- Pivot = record n-1. Set i = 0.
- For j in 0..n-2: if record j < pivot (key < pk, or key == pk and value < pv), swap record j with record i and increment i.
- Swap record i with record n-1.
- Recurse on (a, i), then on (a + (i+1)*16, n-i-1).

Record address: rbx + idx*16, computed via a shifted copy in rax.

Loop body: rax = j<<4 + rbx, load rcx = key, rdx = value. Compare against r13/r14. If less, compute r8 = i<<4 + rbx, load r9/r10 from it, and store crosswise.

Final swap: the record at i and the record at n-1 are exchanged. If i == n-1 the swap is a harmless self-swap.

```
qsort_pairs:
    cmp rsi, 2
    jl .Lret0
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rdi
    mov r12, rsi
    mov rax, r12
    dec rax
    shl rax, 4
    mov r13, qword ptr [rbx + rax]
    mov r14, qword ptr [rbx + rax + 8]
    xor r15d, r15d
    xor ebp, ebp
    lea r11, [r12 - 1]
.Lloop:
    cmp rbp, r11
    jge .Ldone
    mov rax, rbp
    shl rax, 4
    add rax, rbx
    mov rcx, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    cmp rcx, r13
    jl .Lless
    jg .Lnext
    cmp rdx, r14
    jge .Lnext
.Lless:
    mov r8, r15
    shl r8, 4
    add r8, rbx
    mov r9, qword ptr [r8]
    mov r10, qword ptr [r8 + 8]
    mov qword ptr [r8], rcx
    mov qword ptr [r8 + 8], rdx
    mov qword ptr [rax], r9
    mov qword ptr [rax + 8], r10
    inc r15
.Lnext:
    inc rbp
    jmp .Lloop
.Ldone:
    mov r8, r15
    shl r8, 4
    add r8, rbx
    mov rax, r11
    shl rax, 4
    add rax, rbx
    mov rcx, qword ptr [r8]
    mov rdx, qword ptr [r8 + 8]
    mov r9, qword ptr [rax]
    mov r10, qword ptr [rax + 8]
    mov qword ptr [r8], r9
    mov qword ptr [r8 + 8], r10
    mov qword ptr [rax], rcx
    mov qword ptr [rax + 8], rdx
    mov rdi, rbx
    mov rsi, r15
    call qsort_pairs
    lea rdi, [r15 + 1]
    shl rdi, 4
    add rdi, rbx
    mov rsi, r12
    sub rsi, r15
    dec rsi
    call qsort_pairs
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Lret0:
    xor eax, eax
    ret
```
