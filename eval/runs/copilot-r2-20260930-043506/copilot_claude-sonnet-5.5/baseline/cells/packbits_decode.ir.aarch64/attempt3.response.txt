export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @okk %ni, @skip %i1
	%o =l phi @start 0, @okk %o2, @skip %o
	%e =w csgel %i, %n
	jnz %e, @ret_o, @hdr
@ret_o
	ret %o
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	%hl =l extsw %h
	%isn =w csltl %hl, 0
	%nop =w ceql %hl, -128
	jnz %nop, @skip, @notnop
@skip
	jmp @loop
@notnop
	%room =l sub %cap, %o
	jnz %isn, @rep, @lit
@lit
	%cnt =l add %hl, 1
	%end =l add %i1, %cnt
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @lit1
@err2
	ret -2
@lit1
	%lt =w csltl %room, %cnt
	%ltl =l extuw %lt
	%d =l sub %room, %cnt
	%dm =l mul %d, %ltl
	%m =l add %cnt, %dm
	jmp @ll
@ll
	%k =l phi @lit1 0, @lb %k1
	%c =w csltl %k, %m
	jnz %c, @lb, @check
@lb
	%sq =l add %src, %i1
	%sq2 =l add %sq, %k
	%b =w loadub %sq2
	%dp =l add %out, %o
	%dp2 =l add %dp, %k
	storeb %b, %dp2
	%k1 =l add %k, 1
	jmp @ll
@rep
	%cnt2 =l sub 1, %hl
	%end2 =l add %i1, 1
	%bad2 =w csgtl %end2, %n
	jnz %bad2, @err2, @rep1
@rep1
	%rp =l add %src, %i1
	%cb =w loadub %rp
	%lt2 =w csltl %room, %cnt2
	%lt2l =l extuw %lt2
	%d2 =l sub %room, %cnt2
	%dm2 =l mul %d2, %lt2l
	%m2 =l add %cnt2, %dm2
	jmp @rl
@rl
	%k2 =l phi @rep1 0, @rb %k3
	%c2 =w csltl %k2, %m2
	jnz %c2, @rb, @check
@rb
	%dq =l add %out, %o
	%dq2 =l add %dq, %k2
	storeb %cb, %dq2
	%k3 =l add %k2, 1
	jmp @rl
@check
	%ni =l phi @ll %end, @rl %end2
	%cn =l phi @ll %cnt, @rl %cnt2
	%mm =l phi @ll %m, @rl %m2
	%ov =w csltl %mm, %cn
	jnz %ov, @over, @okk
@over
	ret -1
@okk
	%o2 =l add %o, %cn
	jmp @loop
}
