export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @part
@part
	%last =l sub %n, 1
	%off =l shl %last, 4
	%pp =l add %a, %off
	%pk =l loadl %pp
	%pp8 =l add %pp, 8
	%pv =l loadl %pp8
	jmp @ol
@ol
	%i =l phi @part 0, @ol2 %i2
	%j =l phi @part 0, @ol2 %j2
	%c1 =w csltl %j, %last
	jnz %c1, @body, @fin
@body
	%jo =l shl %j, 4
	%jp =l add %a, %jo
	%jk =l loadl %jp
	%jp8 =l add %jp, 8
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	%eq =w ceql %jk, %pk
	%vl =w csltl %jv, %pv
	%e2 =w and %eq, %vl
	%less =w or %lt, %e2
	jnz %less, @swap, @ol2
@swap
	%io =l shl %i, 4
	%ip =l add %a, %io
	%ik =l loadl %ip
	%ip8 =l add %ip, 8
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%i1 =l add %i, 1
	jmp @ol2
@ol2
	%i2 =l phi @body %i, @swap %i1
	%j2 =l add %j, 1
	jmp @ol
@fin
	%fo =l shl %i, 4
	%fp =l add %a, %fo
	%fk =l loadl %fp
	%fp8 =l add %fp, 8
	%fv =l loadl %fp8
	storel %pk, %fp
	storel %pv, %fp8
	storel %fk, %pp
	storel %fv, %pp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%i3 =l add %i, 1
	%no =l shl %i3, 4
	%na =l add %a, %no
	%nn =l sub %n, %i3
	%r2 =l call $qsort_pairs(l %na, l %nn)
	jmp @done
@done
	ret 0
}
