max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    pxor xmm0, xmm0
.Lvec:
    cmp rsi, 16
    jl .Ltail
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    jmp .Lvec
.Ltail:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
.Lscalar:
    test rsi, rsi
    jz .Ldone
    movzx ecx, byte ptr [rdi]
    cmp ecx, eax
    cmova eax, ecx
    inc rdi
    dec rsi
    jmp .Lscalar
.Ldone:
    ret
