export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%o =l phi @start 0, @next %o2
	%rem =l sub %n, %i
	%c =w csgtl %rem, 0
	jnz %c, @body, @end
@body
	%p =l add %src, %i
	%b0 =l loadub %p
	%c1 =w csgtl %rem, 1
	jnz %c1, @l1, @z1
@l1
	%p1 =l add %p, 1
	%x1 =l loadub %p1
	jmp @j1
@z1
	jmp @j1
@j1
	%b1 =l phi @l1 %x1, @z1 0
	%c2 =w csgtl %rem, 2
	jnz %c2, @l2, @z2
@l2
	%p2 =l add %p, 2
	%x2 =l loadub %p2
	jmp @j2
@z2
	jmp @j2
@j2
	%b2 =l phi @l2 %x2, @z2 0
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%v0 =l or %s0, %s1
	%v =l or %v0, %b2
	%t =l add %rem, 1
	%gt =w csgtl %t, 4
	%gtl =l extuw %gt
	%tm =l sub %t, 4
	%adj =l mul %tm, %gtl
	%valid =l sub %t, %adj
	%q =l add %out, %o
	jmp @cl
@cl
	%k =l phi @j2 0, @cl %k2
	%k6 =l mul %k, 6
	%sh =l sub 18, %k6
	%shw =w copy %sh
	%x =l shr %v, %shw
	%idx =l and %x, 63
	%g26 =w csgel %idx, 26
	%g52 =w csgel %idx, 52
	%g62 =w csgel %idx, 62
	%g63 =w csgel %idx, 63
	%e26 =l extuw %g26
	%e52 =l extuw %g52
	%e62 =l extuw %g62
	%e63 =l extuw %g63
	%m26 =l mul %e26, 6
	%m52 =l mul %e52, 75
	%m62 =l mul %e62, 15
	%m63 =l mul %e63, 3
	%ch0 =l add %idx, 65
	%ch1 =l add %ch0, %m26
	%ch2 =l sub %ch1, %m52
	%ch3 =l sub %ch2, %m62
	%ch =l add %ch3, %m63
	%pd =w csgel %k, %valid
	%pdl =l extuw %pd
	%df =l sub 61, %ch
	%pm =l mul %df, %pdl
	%fin =l add %ch, %pm
	%qa =l add %q, %k
	storeb %fin, %qa
	%k2 =l add %k, 1
	%more =w csltl %k2, 4
	jnz %more, @cl, @next
@next
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @loop
@end
	ret %o
}
