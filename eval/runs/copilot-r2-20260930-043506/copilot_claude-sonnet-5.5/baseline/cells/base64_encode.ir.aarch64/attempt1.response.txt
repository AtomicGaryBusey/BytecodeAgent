export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%rem =l sub %n, %i
	%c =w csgel %rem, 3
	jnz %c, @full, @tail
@full
	%p =l add %src, %i
	%p1 =l add %p, 1
	%p2 =l add %p, 2
	%b0 =l loadub %p
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%v0 =l or %s0, %s1
	%v =l or %v0, %b2
	%q =l add %out, %o
	%a0 =l shr %v, 18
	%a0 =l and %a0, 63
	%a1 =l shr %v, 12
	%a1 =l and %a1, 63
	%a2 =l shr %v, 6
	%a2 =l and %a2, 63
	%a3 =l and %v, 63
	%c0 =l call $b64c(l %a0)
	jmp @loop
@tail
	jmp @end
@end
	ret %o
}
