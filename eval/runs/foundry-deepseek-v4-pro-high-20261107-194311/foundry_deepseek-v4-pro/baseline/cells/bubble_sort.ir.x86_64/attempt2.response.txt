function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%j_init =l copy 0
	jmp @outer_loop_head

@outer_loop_head
	%j =l phi @start %j_init, @outer_end %j_next
	%j_cond =w csltl %j, %nm1
	jnz %j_cond, @outer_body, @done

@outer_body
	%inner_limit =l sub %nm1, %j
	jmp @inner_loop_init

@inner_loop_init
	%i_init =l copy 0
	jmp @inner_loop_head

@inner_loop_head
	%i =l phi @inner_loop_init %i_init, @increment %i_next
	%i_cond =w csltl %i, %inner_limit
	jnz %i_cond, @inner_body, @outer_end

@inner_body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v1 =l loadl %ptr
	%ptr2 =l add %ptr, 8
	%v2 =l loadl %ptr2
	%need_swap =w csltl %v2, %v1
	jnz %need_swap, @do_swap, @increment

@do_swap
	storel %v2, %ptr
	storel %v1, %ptr2
	jmp @increment

@increment
	%i_next =l add %i, 1
	jmp @inner_loop_head

@outer_end
	%j_next =l add %j, 1
	jmp @outer_loop_head

@done
	ret 0
}
