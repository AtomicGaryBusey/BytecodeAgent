function l $bubble_sort(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done
	%limit =l sub %n, 1
	jmp @outer_loop_start

@outer_loop_start
	%i =l copy 0
	%swapped =l copy 0

@outer_loop
	%cond =w csltl %i, %limit
	jnz %cond, @inner_loop, @after_pass

@inner_loop
	%off =l mul %i, 8
	%ptr1 =l add %a, %off
	%ptr2 =l add %ptr1, 8
	%v1 =l loadl %ptr1
	%v2 =l loadl %ptr2
	%need_swap =w csltl %v2, %v1
	jnz %need_swap, @do_swap, @next_i

@do_swap
	storel %v2, %ptr1
	storel %v1, %ptr2
	%swapped =l copy 1

@next_i
	%i =l add %i, 1
	jmp @outer_loop

@after_pass
	%swapped_cond =w ceql %swapped, 0
	jnz %swapped_cond, @done
	%limit =l sub %limit, 1
	%limit_ge1 =w csgel %limit, 1
	jnz %limit_ge1, @outer_loop_start, @done

@done
	ret 0
}
