export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bzero =w ceql %b, 0
    jnz %bzero, @b_zero, @check_minus_one
@b_zero
    storel %a, %rem
    %zero =l copy 0
    ret %zero
@check_minus_one
    %min_int =l copy -9223372036854775808
    %amin =w ceql %a, %min_int
    %bminus1 =w ceql %b, -1
    %both =w and %amin, %bminus1
    jnz %both, @overflow_case, @normal_div
@overflow_case
    storel 0, %rem
    ret %min_int
@normal_div
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
