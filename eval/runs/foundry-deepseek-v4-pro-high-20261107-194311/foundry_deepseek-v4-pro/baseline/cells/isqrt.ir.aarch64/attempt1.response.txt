export function l $isqrt(l %n) {
@start
    %low =l copy 0
    %high =l copy 4294967295
    jmp @loop
@loop
    %sum =l add %low, %high
    %mid =l shr %sum, 1
    %midsq =l mul %mid, %mid
    %cond =w culel %midsq, %n
    jnz %cond, @then, @else
@then
    %low =l add %mid, 1
    jmp @check
@else
    %high =l sub %mid, 1
    jmp @check
@check
    %loop_cond =w culel %low, %high
    jnz %loop_cond, @loop, @end
@end
    %retval =l copy %high
    ret %retval
}
