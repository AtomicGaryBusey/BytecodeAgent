count_ugt:
    test rsi, rsi
    jz .Lzero
    xor ecx, ecx
    pcmpeqd xmm1, xmm1
    psllq xmm1, 63
    movq xmm0, rdx
    punpcklqdq xmm0, xmm0
    pxor xmm0, xmm1
.Lloop:
    cmp rsi, 2
    jl .Ltail
    movups xmm2, xmmword ptr [rdi]
    pxor xmm2, xmm1
    pcmpgtq xmm2, xmm0
    movmskpd eax, xmm2
    popcnt eax, eax
    add ecx, eax
    add rdi, 16
    sub rsi, 2
    jmp .Lloop
.Ltail:
    test rsi, rsi
    jz .Ldone
    mov rax, qword ptr [rdi]
    cmp rax, rdx
    seta al
    movzx eax, al
    add ecx, eax
.Ldone:
    mov rax, rcx
    ret
.Lzero:
    xor eax, eax
    ret
