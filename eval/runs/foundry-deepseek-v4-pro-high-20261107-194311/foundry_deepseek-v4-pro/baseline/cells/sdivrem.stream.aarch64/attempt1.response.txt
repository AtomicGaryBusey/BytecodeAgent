sdivrem:
    cbz x1, .Lzero             // if b == 0, handle special case
    sdiv x3, x0, x1            // q = a / b        (truncated toward zero)
    msub x4, x3, x1, x0        // r = a - q * b
    str x4, [x2]               // *rem = r
    mov x0, x3                 // return q
    ret
.Lzero:
    str x0, [x2]               // *rem = a
    mov x0, #0                 // return 0
    ret
