vm_run:
  push rbx
  push r12
  push r13
  push r14
  push r15
  mov r12, rsi
  mov r13, rcx
  mov r14, 0
  mov r15, 0
  mov rbx, rdx
.Lloop:
  cmp r15, r12
  je .Lhalt
  mov rax, qword ptr [rdi + r15*8]
  cmp rax, 0
  je .Lhalt
  cmp rax, 1
  je .Lpush
  cmp rax, 2
  je .Ladd
  cmp rax, 3
  je .Lsub
  cmp rax, 4
  je .Lmul
  cmp rax, 5
  je .Ldup
  cmp rax, 6
  je .Lswap
  cmp rax, 7
  je .Ldrop
  cmp rax, 8
  je .Lover
  cmp rax, 9
  je .Llt
  cmp rax, 10
  je .Ljz
  cmp rax, 11
  je .Ljmp
  add r15, 1
  jmp .Lloop
.Lpush:
  mov rcx, qword ptr [rdi + r15*8 + 8]
  mov qword ptr [rbx + r14*8], rcx
  inc r14
  add r15, 2
  jmp .Lloop
.Ladd:
  mov rcx, qword ptr [rbx + r14*8 - 8]
  mov rax, qword ptr [rbx + r14*8 - 16]
  add rax, rcx
  mov qword ptr [rbx + r14*8 - 16], rax
  dec r14
  add r15, 1
  jmp .Lloop
.Lsub:
  mov rcx, qword ptr [rbx + r14*8 - 8]
  mov rax, qword ptr [rbx + r14*8 - 16]
  sub rax, rcx
  mov qword ptr [rbx + r14*8 - 16], rax
  dec r14
  add r15, 1
  jmp .Lloop
.Lmul:
  mov rax, qword ptr [rbx + r14*8 - 16]
  mov rcx, qword ptr [rbx + r14*8 - 8]
  mul rcx
  mov qword ptr [rbx + r14*8 - 16], rax
  dec r14
  add r15, 1
  jmp .Lloop
.Ldup:
  mov rax, qword ptr [rbx + r14*8 - 8]
  mov qword ptr [rbx + r14*8], rax
  inc r14
  add r15, 1
  jmp .Lloop
.Lswap:
  mov rax, qword ptr [rbx + r14*8 - 8]
  mov rcx, qword ptr [rbx + r14*8 - 16]
  mov qword ptr [rbx + r14*8 - 8], rcx
  mov qword ptr [rbx + r14*8 - 16], rax
  add r15, 1
  jmp .Lloop
.Ldrop:
  dec r14
  add r15, 1
  jmp .Lloop
.Lover:
  mov rax, qword ptr [rbx + r14*8 - 16]
  mov qword ptr [rbx + r14*8], rax
  inc r14
  add r15, 1
  jmp .Lloop
.Llt:
  mov rcx, qword ptr [rbx + r14*8 - 8]
  mov rax, qword ptr [rbx + r14*8 - 16]
  cmp rax, rcx
  setl al
  movzx rax, al
  mov qword ptr [rbx + r14*8 - 16], rax
  dec r14
  add r15, 1
  jmp .Lloop
.Ljz:
  mov rax, qword ptr [rbx + r14*8 - 8]
  dec r14
  test rax, rax
  jz .Ljz_take
  add r15, 2
  jmp .Lloop
.Ljz_take:
  mov r15, qword ptr [rdi + r15*8 + 8]
  jmp .Lloop
.Ljmp:
  mov r15, qword ptr [rdi + r15*8 + 8]
  jmp .Lloop
.Lhalt:
  mov rcx, r14
  lea rdx, [rbx + rcx*8]
  mov r8, r13
  sub r8, rcx
  jz .Lzero_done
  xor rax, rax
.Lzero_loop:
  mov qword ptr [rdx], rax
  add rdx, 8
  dec r8
  jnz .Lzero_loop
.Lzero_done:
  test r14, r14
  jz .Lreturn_zero
  mov rax, qword ptr [rbx + r14*8 - 8]
  jmp .Lcleanup
.Lreturn_zero:
  xor rax, rax
.Lcleanup:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbx
  ret
