export function l $qsort_pairs(l %a, l %n) {
@start
	%cmp =w cslel %n, 1
	jnz %cmp, @done, @pre_sort
@done
	ret 0
@pre_sort
	%n_minus_1 =l sub %n, 1
	%pivot_off =l shl %n_minus_1, 4
	%pivot_ptr =l add %a, %pivot_off
	%pivot_key =l loadl %pivot_ptr
	%pivot_val_ptr =l add %pivot_ptr, 8
	%pivot_val =l loadl %pivot_val_ptr
	%i_init =l copy -1
	%j_init =l copy 0
	jmp @loop_header
@loop_header
	%i =l phi @pre_sort %i_init, @loop_inc %i_next
	%j =l phi @pre_sort %j_init, @loop_inc %j_next
	%loop_test =w csltl %j, %n_minus_1
	jnz %loop_test, @loop_body, @partition_done
@loop_body
	%j_off =l shl %j, 4
	%recj =l add %a, %j_off
	%keyj =l loadl %recj
	%valj_ptr =l add %recj, 8
	%valj =l loadl %valj_ptr
	%key_lt =w csltl %keyj, %pivot_key
	%key_eq =w ceql %keyj, %pivot_key
	%val_lt =w csltl %valj, %pivot_val
	%val_check =w and %key_eq, %val_lt
	%is_less =w or %key_lt, %val_check
	%i_uns =l copy %i
	jnz %is_less, @swap, @no_swap
@no_swap
	jmp @loop_inc
@swap
	%i_inc =l add %i, 1
	%i_off =l shl %i_inc, 4
	%ptri =l add %a, %i_off
	%ptri_val =l add %ptri, 8
	%ptrj =l add %a, %j_off
	%ptrj_val =l add %ptrj, 8
	%tmp_key_i =l loadl %ptri
	%tmp_val_i =l loadl %ptri_val
	%tmp_key_j =l loadl %ptrj
	%tmp_val_j =l loadl %ptrj_val
	storel %tmp_key_j, %ptri
	storel %tmp_val_j, %ptri_val
	storel %tmp_key_i, %ptrj
	storel %tmp_val_i, %ptrj_val
	jmp @loop_inc
@loop_inc
	%i_next =l phi @swap %i_inc, @no_swap %i_uns
	%j_next =l add %j, 1
	jmp @loop_header
@partition_done
	%pivot_index =l add %i, 1
	%pi_off =l shl %pivot_index, 4
	%pii_ptr =l add %a, %pi_off
	%pii_val_ptr =l add %pii_ptr, 8
	%tmp_pi_key =l loadl %pii_ptr
	%tmp_pi_val =l loadl %pii_val_ptr
	storel %pivot_key, %pii_ptr
	storel %pivot_val, %pii_val_ptr
	storel %tmp_pi_key, %pivot_ptr
	storel %tmp_pi_val, %pivot_val_ptr
	%left_call =l call $qsort_pairs(l %a, l %pivot_index)
	%right_size_tmp =l sub %n, %pivot_index
	%right_size =l sub %right_size_tmp, 1
	%right_start_idx =l add %pivot_index, 1
	%right_start_off =l shl %right_start_idx, 4
	%right_ptr =l add %a, %right_start_off
	%right_call =l call $qsort_pairs(l %right_ptr, l %right_size)
	ret 0
}
