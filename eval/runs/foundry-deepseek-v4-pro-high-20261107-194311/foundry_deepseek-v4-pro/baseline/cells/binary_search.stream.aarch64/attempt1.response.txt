binary_search:
    // if n <= 0, return -1
    cmp x1, #0
    ble .Lreturn_neg1
    // lo = 0 (x3), hi = n-1 (x4)
    mov x3, #0
    sub x4, x1, #1
.Lloop:
    // if lo > hi, not found
    cmp x3, x4
    bgt .Lreturn_neg1
    // mid = (lo + hi) / 2
    add x5, x3, x4
    asr x6, x5, #1
    // load a[mid]
    ldr x5, [x0, x6, lsl #3]
    // compare with key
    cmp x5, x2
    b.lt .Lmid_less
    b.gt .Lmid_greater
    // found: return mid
    mov x0, x6
    ret
.Lmid_less:
    add x3, x6, #1
    b .Lloop
.Lmid_greater:
    sub x4, x6, #1
    b .Lloop
.Lreturn_neg1:
    mov x0, #-1
    ret
