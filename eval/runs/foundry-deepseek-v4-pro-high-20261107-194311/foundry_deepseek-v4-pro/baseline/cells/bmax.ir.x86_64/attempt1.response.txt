export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%val =l loadl %addr
	%cmp =w csgtl %val, %max
	%max =l select %cmp, %val, %max
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
