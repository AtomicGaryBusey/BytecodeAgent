export function l $bmax(l %a, l %n) {
@start
    %max =l loadl %a
    %i =l copy 1
    jmp @loop(%i, %max)
@loop(%i =l, %max =l)
    %cond =w csltl %i, %n
    jnz %cond, @body, @loop_end
@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %val =l loadl %addr
    %cmp =w csgtl %val, %max
    jnz %cmp, @update, @continue
@update
    %i_inc =l add %i, 1
    jmp @loop(%i_inc, %val)
@continue
    %i_inc2 =l add %i, 1
    jmp @loop(%i_inc2, %max)
@loop_end
    %final_max =l phi @loop %max
    ret %final_max
}
