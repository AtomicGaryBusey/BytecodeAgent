export function l $bmax(l %a, l %n) {
@start
    %max_start =l loadl %a
    %i_start =l copy 1
    jmp @loop
@loop
    %i =l phi @start %i_start, @inc %i_inc
    %max =l phi @start %max_start, @inc %max_inc
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %val =l loadl %addr
    %cmp =w csgtl %val, %max
    jnz %cmp, @set, @noset
@set
    jmp @inc
@noset
    jmp @inc
@inc
    %max_inc =l phi @set %val, @noset %max
    %i_inc =l add %i, 1
    jmp @loop
@end
    ret %max
}
