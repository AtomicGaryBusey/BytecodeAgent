export function l $reverse(l %a, l %n) {
@start
    %i =l copy 0
    %j =l sub %n, 1
@loop_cond
    %cond =w csltl %i, %j
    jnz %cond, @loop_body, @done
@loop_body
    %off_i =l mul %i, 8
    %addr_i =l add %a, %off_i
    %off_j =l mul %j, 8
    %addr_j =l add %a, %off_j
    %val_i =l loadl %addr_i
    %val_j =l loadl %addr_j
    storel %val_j, %addr_i
    storel %val_i, %addr_j
    %i =l add %i, 1
    %j =l sub %j, 1
    jmp @loop_cond
@done
    ret 0
}
