export function l $reverse(l %a, l %n) {
@start
    %c =w ceql %n, 0
    jnz %c, @done, @compute
@compute
    %n8 =l mul %n, 8
    %end =l add %a, %n8
    %right0 =l sub %end, 8
    jmp @loop(%a, %right0)
@loop(l %left, l %right)
    %cmp =w cultl %left, %right
    jnz %cmp, @continue(%left, %right), @done
@continue(l %l, l %r)
    %lval =l loadl %l
    %rval =l loadl %r
    storel %rval, %l
    storel %lval, %r
    %l2 =l add %l, 8
    %r2 =l sub %r, 8
    jmp @loop(%l2, %r2)
@done
    ret 0
}
