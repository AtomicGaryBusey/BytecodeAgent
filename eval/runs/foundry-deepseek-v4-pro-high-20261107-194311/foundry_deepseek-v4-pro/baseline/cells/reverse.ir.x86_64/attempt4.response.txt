export function l $reverse(l %a, l %n) {
@start
    %c =w ceql %n, 0
    jnz %c, @done, @compute
@compute
    %n8 =l mul %n, 8
    %end =l add %a, %n8
    %right_init =l sub %end, 8
    jmp @loop
@loop
    %left =l phi @compute %a, @body %left_next
    %right =l phi @compute %right_init, @body %right_next
    %cond =w cultl %left, %right
    jnz %cond, @body, @done
@body
    %lval =l loadl %left
    %rval =l loadl %right
    storel %rval, %left
    storel %lval, %right
    %left_next =l add %left, 8
    %right_next =l sub %right, 8
    jmp @loop
@done
    ret 0
}
