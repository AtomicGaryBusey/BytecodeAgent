export function l $xxh64(l %p, l %n, l %seed) {
@start
    %P1 =l copy -7046029288634856825
    %P2 =l copy -4417322545169274033
    %P3 =l copy 1609587929392835161
    %P4 =l copy -8796714831421723037
    %P5 =l copy 2870177450012600261
    %n_ge_32 =w csgel %n, 32
    jnz %n_ge_32, @stripe_init, @small_init

@small_init
    %h_small =l add %seed, %P5
    %q_small =l copy %p
    jmp @process_tail_start

@stripe_init
    %v1_0 =l add %seed, %P1
    %v1_init =l add %v1_0, %P2
    %v2_init =l add %seed, %P2
    %v3_init =l copy %seed
    %v4_init =l sub %seed, %P1
    %mask =l copy -32
    %n_masked =l and %n, %mask
    %end =l add %p, %n_masked
    %q_init =l copy %p
    jmp @stripe_loop

@stripe_loop
    %v1 =l phi @stripe_init %v1_init, @stripe_body %v1_new
    %v2 =l phi @stripe_init %v2_init, @stripe_body %v2_new
    %v3 =l phi @stripe_init %v3_init, @stripe_body %v3_new
    %v4 =l phi @stripe_init %v4_init, @stripe_body %v4_new
    %q =l phi @stripe_init %q_init, @stripe_body %q_new
    %loop_cond =w cultl %q, %end
    jnz %loop_cond, @stripe_body, @stripe_done

@stripe_body
    %q0 =l loadl %q
    %q8 =l add %q, 8
    %q8_val =l loadl %q8
    %q16 =l add %q, 16
    %q16_val =l loadl %q16
    %q24 =l add %q, 24
    %q24_val =l loadl %q24

    %t1 =l mul %q0, %P2
    %t2 =l add %v1, %t1
    %t2_shl =l shl %t2, 31
    %t2_shr =l shr %t2, 33
    %t2_rot =l or %t2_shl, %t2_shr
    %v1_new =l mul %t2_rot, %P1

    %t3 =l mul %q8_val, %P2
    %t4 =l add %v2, %t3
    %t4_shl =l shl %t4, 31
    %t4_shr =l shr %t4, 33
    %t4_rot =l or %t4_shl, %t4_shr
    %v2_new =l mul %t4_rot, %P1

    %t5 =l mul %q16_val, %P2
    %t6 =l add %v3, %t5
    %t6_shl =l shl %t6, 31
    %t6_shr =l shr %t6, 33
    %t6_rot =l or %t6_shl, %t6_shr
    %v3_new =l mul %t6_rot, %P1

    %t7 =l mul %q24_val, %P2
    %t8 =l add %v4, %t7
    %t8_shl =l shl %t8, 31
    %t8_shr =l shr %t8, 33
    %t8_rot =l or %t8_shl, %t8_shr
    %v4_new =l mul %t8_rot, %P1

    %q_new =l add %q, 32
    jmp @stripe_loop

@stripe_done
    %v1_shl1 =l shl %v1, 1
    %v1_shr63 =l shr %v1, 63
    %v1_rot1 =l or %v1_shl1, %v1_shr63
    %v2_shl7 =l shl %v2, 7
    %v2_shr57 =l shr %v2, 57
    %v2_rot7 =l or %v2_shl7, %v2_shr57
    %v3_shl12 =l shl %v3, 12
    %v3_shr52 =l shr %v3, 52
    %v3_rot12 =l or %v3_shl12, %v3_shr52
    %v4_shl18 =l shl %v4, 18
    %v4_shr46 =l shr %v4, 46
    %v4_rot18 =l or %v4_shl18, %v4_shr46
    %h_sum1 =l add %v1_rot1, %v2_rot7
    %h_sum2 =l add %h_sum1, %v3_rot12
    %h_sum =l add %h_sum2, %v4_rot18

    %rx1 =l mul %v1, %P2
    %rx1_shl =l shl %rx1, 31
    %rx1_shr =l shr %rx1, 33
    %rx1_rot =l or %rx1_shl, %rx1_shr
    %r01 =l mul %rx1_rot, %P1
    %h_xor1 =l xor %h_sum, %r01
    %h_mer1 =l mul %h_xor1, %P1
    %h1 =l add %h_mer1, %P4

    %rx2 =l mul %v2, %P2
    %rx2_shl =l shl %rx2, 31
    %rx2_shr =l shr %rx2, 33
    %rx2_rot =l or %rx2_shl, %rx2_shr
    %r02 =l mul %rx2_rot, %P1
    %h_xor2 =l xor %h1, %r02
    %h_mer2 =l mul %h_xor2, %P1
    %h2 =l add %h_mer2, %P4

    %rx3 =l mul %v3, %P2
    %rx3_shl =l shl %rx3, 31
    %rx3_shr =l shr %rx3, 33
    %rx3_rot =l or %rx3_shl, %rx3_shr
    %r03 =l mul %rx3_rot, %P1
    %h_xor3 =l xor %h2, %r03
    %h_mer3 =l mul %h_xor3, %P1
    %h3 =l add %h_mer3, %P4

    %rx4 =l mul %v4, %P2
    %rx4_shl =l shl %rx4, 31
    %rx4_shr =l shr %rx4, 33
    %rx4_rot =l or %rx4_shl, %rx4_shr
    %r04 =l mul %rx4_rot, %P1
    %h_xor4 =l xor %h3, %r04
    %h_mer4 =l mul %h_xor4, %P1
    %h_merged =l add %h_mer4, %P4
    %q_end =l copy %q
    jmp @process_tail_start

@process_tail_start
    %h_init =l phi @small_init %h_small, @stripe_done %h_merged
    %q_cur =l phi @small_init %q_small, @stripe_done %q_end
    %h =l add %h_init, %n
    %p_end =l add %p, %n
    jmp @tail8_loop

@tail8_loop
    %h8 =l phi @process_tail_start %h, @tail8_body %h8_next
    %q8 =l phi @process_tail_start %q_cur, @tail8_body %q8_next
    %q8_plus8 =l add %q8, 8
    %has8 =w culel %q8_plus8, %p_end
    jnz %has8, @tail8_body, @tail4_check

@tail8_body
    %x8 =l loadl %q8
    %rx8 =l mul %x8, %P2
    %rx8_shl =l shl %rx8, 31
    %rx8_shr =l shr %rx8, 33
    %rx8_rot =l or %rx8_shl, %rx8_shr
    %r0_8 =l mul %rx8_rot, %P1
    %h8_xor =l xor %h8, %r0_8
    %h8_shl27 =l shl %h8_xor, 27
    %h8_shr37 =l shr %h8_xor, 37
    %h8_rot27 =l or %h8_shl27, %h8_shr37
    %h8_mul =l mul %h8_rot27, %P1
    %h8_next =l add %h8_mul, %P4
    %q8_next =l add %q8, 8
    jmp @tail8_loop

@tail4_check
    %q4_plus4 =l add %q8, 4
    %has4 =w culel %q4_plus4, %p_end
    jnz %has4, @tail4_body, @tail_byte_loop_start

@tail4_body
    %x4_w =w loaduw %q8
    %x4 =l extuw %x4_w
    %x4_mul =l mul %x4, %P1
    %h4_xor =l xor %h8, %x4_mul
    %h4_shl23 =l shl %h4_xor, 23
    %h4_shr41 =l shr %h4_xor, 41
    %h4_rot23 =l or %h4_shl23, %h4_shr41
    %h4_mul =l mul %h4_rot23, %P2
    %h4_next =l add %h4_mul, %P3
    %q4_next =l add %q8, 4
    jmp @tail_byte_loop_start

@tail_byte_loop_start
    %h_byte_init =l phi @tail4_check %h8, @tail4_body %h4_next
    %q_byte_init =l phi @tail4_check %q8, @tail4_body %q4_next
    jmp @tail_byte_loop

@tail_byte_loop
    %h_b =l phi @tail_byte_loop_start %h_byte_init, @tail_byte_body %h_b_next
    %q_b =l phi @tail_byte_loop_start %q_byte_init, @tail_byte_body %q_b_next
    %has_byte =w cultl %q_b, %p_end
    jnz %has_byte, @tail_byte_body, @finalize

@tail_byte_body
    %byte_w =w loadub %q_b
    %byte =l extuw %byte_w
    %byte_mul =l mul %byte, %P5
    %h_b_xor =l xor %h_b, %byte_mul
    %h_b_shl11 =l shl %h_b_xor, 11
    %h_b_shr53 =l shr %h_b_xor, 53
    %h_b_rot11 =l or %h_b_shl11, %h_b_shr53
    %h_b_next =l mul %h_b_rot11, %P1
    %q_b_next =l add %q_b, 1
    jmp @tail_byte_loop

@finalize
    %h_shr33 =l shr %h_b, 33
    %f1 =l xor %h_b, %h_shr33
    %f2 =l mul %f1, %P2
    %f2_shr29 =l shr %f2, 29
    %f3 =l xor %f2, %f2_shr29
    %f4 =l mul %f3, %P3
    %f4_shr32 =l shr %f4, 32
    %result =l xor %f4, %f4_shr32
    ret %result
}
