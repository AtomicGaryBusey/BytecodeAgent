export function l $shift_sat(l %x, l %n) {
@start
    %n_lt_64 =w csltl %n, 64
    jnz %n_lt_64, @shift_lt64, @shift_ge64

@shift_lt64
    %left_lt =l shl %x, %n
    %right_lt =l sar %x, %n
    jmp @merge

@shift_ge64
    %left_ge =l copy 0
    %right_ge =l sar %x, 63
    jmp @merge

@merge
    %left =l phi @shift_lt64 %left_lt, @shift_ge64 %left_ge
    %right =l phi @shift_lt64 %right_lt, @shift_ge64 %right_ge
    %result =l add %left, %right
    ret %result
}
