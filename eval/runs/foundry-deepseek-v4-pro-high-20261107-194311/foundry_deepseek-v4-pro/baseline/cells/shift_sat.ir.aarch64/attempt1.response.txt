export function l $shift_sat(l %x, l %n) {
@start
    %n_lt_64 =w csltl %n, 64
    jnz %n_lt_64, @shift_lt64, @shift_ge64
@shift_lt64
    %left_lt =l shl %x, %n
    %right_lt =l sar %x, %n
    jmp @merge(%left_lt, %right_lt)
@shift_ge64
    %left_ge =l copy 0
    %is_neg =w csltl %x, 0
    %tmp =l extsw %is_neg
    %right_ge =l sub 0, %tmp
    jmp @merge(%left_ge, %right_ge)
@merge(l %left, l %right)
    %result =l add %left, %right
    ret %result
}
