export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zero_loop
@zero_loop
	%j0 =l phi @start 0, @zero_loop %j0_next
	%off0 =l mul %j0, 8
	%ptr0 =l add %row, %off0
	storel 0, %ptr0
	%j0_next =l add %j0, 1
	%cond0 =w cslel %j0_next, %n
	jnz %cond0, @zero_loop, @zero_done
@zero_done
	jmp @outer_loop
@outer_loop
	%i =l phi @zero_done 0, @outer_next %i_next
	%cond_outer =w csltl %i, %m
	jnz %cond_outer, @outer_body, @done
@outer_body
	%j_init =l copy 1
	%diag_init =l copy 0
	jmp @inner_loop
@inner_loop
	%j_inner =l phi @outer_body %j_init, @merge %j_next
	%diag =l phi @outer_body %diag_init, @merge %diag_next
	%inner_cond =w cslel %j_inner, %n
	jnz %inner_cond, @inner_body, @outer_next
@inner_body
	%xptr =l add %x, %i
	%xbyte =w loadub %xptr
	%jminus1 =l sub %j_inner, 1
	%yptr =l add %y, %jminus1
	%ybyte =w loadub %yptr
	%off_j =l mul %j_inner, 8
	%row_j_ptr =l add %row, %off_j
	%up =l loadl %row_j_ptr
	%off_jm1 =l mul %jminus1, 8
	%row_jm1_ptr =l add %row, %off_jm1
	%left =l loadl %row_jm1_ptr
	%eq =w ceqw %xbyte, %ybyte
	jnz %eq, @equal, @notequal
@equal
	%diag_plus =l add %diag, 1
	%newval =l copy %diag_plus
	jmp @merge
@notequal
	%up_lt_left =w csltl %up, %left
	jnz %up_lt_left, @use_left, @use_up
@use_left
	%newval =l copy %left
	jmp @merge
@use_up
	%newval =l copy %up
	jmp @merge
@merge
	storel %newval, %row_j_ptr
	%diag_next =l copy %up
	%j_next =l add %j_inner, 1
	jmp @inner_loop
@outer_next
	%i_next =l add %i, 1
	jmp @outer_loop
@done
	%off_n =l mul %n, 8
	%res_ptr =l add %row, %off_n
	%res =l loadl %res_ptr
	ret %res
}
