count_desc:
    cmp x1, #2
    b.lt .Lreturn_zero
    mov x9, x0
    mov x10, x1
    mov x2, xzr
    mov x3, xzr
.Lloop:
    add x11, x2, #1
    cmp x11, x10
    b.ge .Ldone
    ldr x4, [x9, x2, lsl #3]
    ldr x5, [x9, x11, lsl #3]
    cmp x4, x5
    b.le .Lskip
    add x3, x3, #1
.Lskip:
    add x2, x2, #1
    b .Lloop
.Ldone:
    mov x0, x3
    ret
.Lreturn_zero:
    mov x0, #0
    ret
