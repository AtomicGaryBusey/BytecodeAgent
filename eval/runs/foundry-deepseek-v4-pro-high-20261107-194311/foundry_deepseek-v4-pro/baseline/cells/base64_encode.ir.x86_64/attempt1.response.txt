export function l $base64_encode(l %src, l %n, l %out) {
@start
        %np2 =l add %n, 2
        %groups =l div %np2, 3
        %total_out =l mul %groups, 4
        %isgroups0 =w ceql %groups, 0
        jnz %isgroups0, @ret0, @init_loop

@ret0
        ret 0

@init_loop
        %zero =l copy 0
        %three =l add %zero, 3
        %i =l copy 0
        %out_pos =l copy %out
        %groups_rem =l copy %groups
        jmp @loop_cond

@loop_cond
        %cond =w csgtl %groups_rem, 0
        jnz %cond, @loop_body, @ret

@loop_body
        %rem =l sub %n, %i
        %lt3 =w csltl %rem, 3
        jnz %lt3, @m_rem_chosen, @m_3_chosen

@m_rem_chosen
        %m =l copy %rem
        jmp @after_m

@m_3_chosen
        %m =l copy %three
        jmp @after_m

@after_m
        %src_i =l add %src, %i
        %b0 =l loadub %src_i
        %b1 =l copy %zero
        %b2 =l copy %zero
        %cnd2 =w csgel %m, 2
        jnz %cnd2, @load_b1, @after_b1

@load_b1
        %src_i1 =l add %src_i, 1
        %b1 =l loadub %src_i1
        jmp @after_b1

@after_b1
        %cnd3 =w csgel %m, 3
        jnz %cnd3, @load_b2, @after_b2

@load_b2
        %src_i2 =l add %src_i, 2
        %b2 =l loadub %src_i2
        jmp @after_b2

@after_b2
        %val =l shl %b0, 16
        %tmp =l shl %b1, 8
        %val =l or %val, %tmp
        %val =l or %val, %b2
        %c0 =l shr %val, 18
        %c1 =l shr %val, 12
        %c1 =l and %c1, 63
        %c2 =l shr %val, 6
        %c2 =l and %c2, 63
        %c3 =l and %val, 63

        jmp @enc0_start

@enc0_start
        %v0 =l copy %c0
        %c0_lt26 =w csltl %v0, 26
        jnz %c0_lt26, @enc0_range1, @enc0_not_range1

@enc0_range1
        %r0_1 =l add %v0, 65
        jmp @enc0_merge

@enc0_not_range1
        %c0_lt52 =w csltl %v0, 52
        jnz %c0_lt52, @enc0_range2, @enc0_not_range2

@enc0_range2
        %r0_2 =l add %v0, 71
        jmp @enc0_merge

@enc0_not_range2
        %c0_lt62 =w csltl %v0, 62
        jnz %c0_lt62, @enc0_range3, @enc0_not_range3

@enc0_range3
        %r0_3 =l sub %v0, 4
        jmp @enc0_merge

@enc0_not_range3
        %c0_eq62 =w ceql %v0, 62
        jnz %c0_eq62, @enc0_plus, @enc0_slash

@enc0_plus
        %r0_4 =l copy 43
        jmp @enc0_merge

@enc0_slash
        %r0_5 =l copy 47
        jmp @enc0_merge

@enc0_merge
        %char0 =l phi @enc0_range1 %r0_1, @enc0_range2 %r0_2, @enc0_range3 %r0_3, @enc0_plus %r0_4, @enc0_slash %r0_5
        jmp @enc1_start

@enc1_start
        %v1 =l copy %c1
        %c1_lt26 =w csltl %v1, 26
        jnz %c1_lt26, @enc1_range1, @enc1_not_range1

@enc1_range1
        %r1_1 =l add %v1, 65
        jmp @enc1_merge

@enc1_not_range1
        %c1_lt52 =w csltl %v1, 52
        jnz %c1_lt52, @enc1_range2, @enc1_not_range2

@enc1_range2
        %r1_2 =l add %v1, 71
        jmp @enc1_merge

@enc1_not_range2
        %c1_lt62 =w csltl %v1, 62
        jnz %c1_lt62, @enc1_range3, @enc1_not_range3

@enc1_range3
        %r1_3 =l sub %v1, 4
        jmp @enc1_merge

@enc1_not_range3
        %c1_eq62 =w ceql %v1, 62
        jnz %c1_eq62, @enc1_plus, @enc1_slash

@enc1_plus
        %r1_4 =l copy 43
        jmp @enc1_merge

@enc1_slash
        %r1_5 =l copy 47
        jmp @enc1_merge

@enc1_merge
        %char1 =l phi @enc1_range1 %r1_1, @enc1_range2 %r1_2, @enc1_range3 %r1_3, @enc1_plus %r1_4, @enc1_slash %r1_5
        jmp @enc2_start

@enc2_start
        %v2 =l copy %c2
        %c2_lt26 =w csltl %v2, 26
        jnz %c2_lt26, @enc2_range1, @enc2_not_range1

@enc2_range1
        %r2_1 =l add %v2, 65
        jmp @enc2_merge

@enc2_not_range1
        %c2_lt52 =w csltl %v2, 52
        jnz %c2_lt52, @enc2_range2, @enc2_not_range2

@enc2_range2
        %r2_2 =l add %v2, 71
        jmp @enc2_merge

@enc2_not_range2
        %c2_lt62 =w csltl %v2, 62
        jnz %c2_lt62, @enc2_range3, @enc2_not_range3

@enc2_range3
        %r2_3 =l sub %v2, 4
        jmp @enc2_merge

@enc2_not_range3
        %c2_eq62 =w ceql %v2, 62
        jnz %c2_eq62, @enc2_plus, @enc2_slash

@enc2_plus
        %r2_4 =l copy 43
        jmp @enc2_merge

@enc2_slash
        %r2_5 =l copy 47
        jmp @enc2_merge

@enc2_merge
        %char2_orig =l phi @enc2_range1 %r2_1, @enc2_range2 %r2_2, @enc2_range3 %r2_3, @enc2_plus %r2_4, @enc2_slash %r2_5
        jmp @enc3_start

@enc3_start
        %v3 =l copy %c3
        %c3_lt26 =w csltl %v3, 26
        jnz %c3_lt26, @enc3_range1, @enc3_not_range1

@enc3_range1
        %r3_1 =l add %v3, 65
        jmp @enc3_merge

@enc3_not_range1
        %c3_lt52 =w csltl %v3, 52
        jnz %c3_lt52, @enc3_range2, @enc3_not_range2

@enc3_range2
        %r3_2 =l add %v3, 71
        jmp @enc3_merge

@enc3_not_range2
        %c3_lt62 =w csltl %v3, 62
        jnz %c3_lt62, @enc3_range3, @enc3_not_range3

@enc3_range3
        %r3_3 =l sub %v3, 4
        jmp @enc3_merge

@enc3_not_range3
        %c3_eq62 =w ceql %v3, 62
        jnz %c3_eq62, @enc3_plus, @enc3_slash

@enc3_plus
        %r3_4 =l copy 43
        jmp @enc3_merge

@enc3_slash
        %r3_5 =l copy 47
        jmp @enc3_merge

@enc3_merge
        %char3_orig =l phi @enc3_range1 %r3_1, @enc3_range2 %r3_2, @enc3_range3 %r3_3, @enc3_plus %r3_4, @enc3_slash %r3_5
        jmp @padding_check

@padding_check
        %m_eq1 =w ceql %m, 1
        jnz %m_eq1, @pad1, @check_pad2

@pad1
        %c2_pad1 =l copy 61
        %c3_pad1 =l copy 61
        jmp @store_chars

@check_pad2
        %m_eq2 =w ceql %m, 2
        jnz %m_eq2, @pad2, @no_pad

@pad2
        %c3_pad2 =l copy 61
        jmp @store_chars

@no_pad
        jmp @store_chars

@store_chars
        %char2 =l phi @pad1 %c2_pad1, @pad2 %char2_orig, @no_pad %char2_orig
        %char3 =l phi @pad1 %c3_pad1, @pad2 %c3_pad2, @no_pad %char3_orig
        %out0 =l add %out_pos, 0
        storeb %char0, %out0
        %out1 =l add %out_pos, 1
        storeb %char1, %out1
        %out2 =l add %out_pos, 2
        storeb %char2, %out2
        %out3 =l add %out_pos, 3
        storeb %char3, %out3

        %i =l add %i, %m
        %out_pos =l add %out_pos, 4
        %groups_rem =l sub %groups_rem, 1
        jmp @loop_cond

@ret
        ret %total_out
}
