export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@loop_k
	%cond =w csgtl %k, %n
	jnz %cond, @done, @body_k
@body_k
	%x =l copy %k
	%steps =l copy 0
@while_x
	%cmp1 =w ceql %x, 1
	jnz %cmp1, @end_while, @x_not_one
@x_not_one
	%bit =l and %x, 1
	%is_even =w ceql %bit, 0
	jnz %is_even, @even, @odd
@even
	%x =l div %x, 2
	%steps =l add %steps, 1
	jmp @while_x
@odd
	%x =l mul %x, 3
	%x =l add %x, 1
	%steps =l add %steps, 1
	jmp @while_x
@end_while
	%gt =w csgtl %steps, %max
	jnz %gt, @update_max, @next_k
@update_max
	%max =l copy %steps
	jmp @next_k
@next_k
	%k =l add %k, 1
	jmp @loop_k
@done
	ret %max
}
