export function l $fib(l %n) {
@start
	%eq0 =w ceql %n, 0
	jnz %eq0, @ret0, @preloop
@ret0
	ret 0
@preloop
	%a =l copy 0
	%b =l copy 1
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%tmp =l add %a, %b
	%a =l copy %b
	%b =l copy %tmp
	%i =l add %i, 1
	jmp @loop
@done
	ret %b
}
