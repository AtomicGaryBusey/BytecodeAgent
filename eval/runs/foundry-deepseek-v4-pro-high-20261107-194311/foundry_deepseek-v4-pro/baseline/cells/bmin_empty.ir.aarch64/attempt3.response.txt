export function l $bmin_empty(l %a, l %n) {
@start
    %iszero =w ceql %n, 0
    jnz %iszero, @retmax, @loopinit
@retmax
    %max =l copy 9223372036854775807
    ret %max
@loopinit
    %min_init =l loadl %a
    %i_init =l copy 1
    jmp @loopcond
@loopcond
    %min =l phi @loopinit %min_init, @loopinc %min_new
    %i =l phi @loopinit %i_init, @loopinc %i_next
    %cmp =w csltl %i, %n
    jnz %cmp, @loopbody, @retmin
@loopbody
    %offset =l shl %i, 3
    %addr =l add %a, %offset
    %val =l loadl %addr
    %lt =w csltl %val, %min
    jnz %lt, @setmin, @skipset
@setmin
    %newmin_from_val =l copy %val
    jmp @loopinc
@skipset
    %newmin_from_min =l copy %min
    jmp @loopinc
@loopinc
    %min_new =l phi @setmin %newmin_from_val, @skipset %newmin_from_min
    %i_next =l add %i, 1
    jmp @loopcond
@retmin
    ret %min
}
