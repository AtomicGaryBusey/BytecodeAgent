export function l $gcd(l %a, l %b) {
@start
    jmp @loop(%a, %b)
@loop(%a_loop, %b_loop)
    %cond =w cnel %b_loop, 0
    jnz %cond, @calc(%a_loop, %b_loop), @end(%a_loop)
@calc(%a_calc, %b_calc)
    %new_a =l copy %b_calc
    %new_b =l rem %a_calc, %b_calc
    jmp @loop(%new_a, %new_b)
@end(%result)
    ret %result
}
