export function l $gcd(l %a, l %b) {
@start
    %cond =w cnel %b, 0
    jnz %cond, @loop(%a, %b), @end(%a)
@loop(%a_loop, %b_loop)
    %t =l copy %b_loop
    %new_b =l rem %a_loop, %b_loop
    %new_a =l copy %t
    %cond2 =w cnel %new_b, 0
    jnz %cond2, @loop(%new_a, %new_b), @end(%new_a)
@end(%result)
    ret %result
}
