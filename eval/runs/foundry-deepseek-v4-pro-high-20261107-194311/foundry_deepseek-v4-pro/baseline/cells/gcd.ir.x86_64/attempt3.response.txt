export function l $gcd(l %a, l %b) {
@start
    jmp @loop
@loop
    %a_loop =l phi @start %a, @calc %new_a
    %b_loop =l phi @start %b, @calc %new_b
    %cond =w cnel %b_loop, 0
    jnz %cond, @calc, @end
@calc
    %new_a =l copy %b_loop
    %new_b =l rem %a_loop, %b_loop
    jmp @loop
@end
    ret %a_loop
}
