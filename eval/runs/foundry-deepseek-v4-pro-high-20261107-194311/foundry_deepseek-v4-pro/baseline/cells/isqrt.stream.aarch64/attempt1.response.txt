isqrt:
    cbz x0, .Lzero
    mov x4, x0
    mov x1, #0
    mov x2, #0x100000000
.Lloop:
    cmp x1, x2
    b.hs .Ldone
    add x3, x1, x2
    lsr x3, x3, #1
    mul x5, x3, x3
    cmp x5, x4
    b.ls .Lle
    mov x2, x3
    b .Lloop
.Lle:
    add x1, x3, #1
    b .Lloop
.Ldone:
    sub x0, x1, #1
    ret
.Lzero:
    mov x0, #0
    ret
