export function l $map_sum(l %a, l %n, l %f) {
@start
	%i.start =l copy 0
	%sum.start =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i.start, @loop_body %i.next
	%sum =l phi @start %sum.start, @loop_body %sum.next
	%cond =w csltl %i, %n
	jnz %cond, @loop_body, @end
@loop_body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%elem =l loadl %addr
	%res =l call %f(l %elem)
	%sum.next =l add %sum, %res
	%i.next =l add %i, 1
	jmp @loop
@end
	ret %sum
}
