export function l $crc32(l %p, l %n) {
@start
  %crc0 =w copy 0xFFFFFFFF
  %p0 =l copy %p
  %n0 =l copy %n
  %bit0 =l copy 0
  jmp @loop

@loop
  %crc =w phi @start %crc0, @bit_done %crc_after
  %n_rem =l phi @start %n0, @bit_done %n_pre
  %bit_rem =l phi @start %bit0, @bit_done %bit_after
  %p_cur =l phi @start %p0, @bit_done %p_pre
  %n_zero =w ceql %n_rem, 0
  %bit_zero =w ceql %bit_rem, 0
  %done =w and %n_zero, %bit_zero
  jnz %done, @done, @continue

@continue
  %need_byte =w ceql %bit_rem, 0
  jnz %need_byte, @load_byte, @process_bit

@load_byte
  %byte =w loadub %p_cur
  %crc_xor =w xor %crc, %byte
  %p_inc =l add %p_cur, 1
  %n_dec =l sub %n_rem, 1
  jmp @pre_bit

@process_bit
  jmp @pre_bit

@pre_bit
  %crc_pre =w phi @load_byte %crc_xor, @process_bit %crc
  %n_pre =l phi @load_byte %n_dec, @process_bit %n_rem
  %p_pre =l phi @load_byte %p_inc, @process_bit %p_cur
  %bit_pre =l phi @load_byte 8, @process_bit %bit_rem
  jmp @do_bit

@do_bit
  %lsb =w and %crc_pre, 1
  %is_one =w ceql %lsb, 1
  jnz %is_one, @bit_set, @bit_clear

@bit_set
  %crc_s =w shr %crc_pre, 1
  %crc_set =w xor %crc_s, 0xEDB88320
  jmp @bit_done

@bit_clear
  %crc_clear =w shr %crc_pre, 1
  jmp @bit_done

@bit_done
  %crc_after =w phi @bit_set %crc_set, @bit_clear %crc_clear
  %bit_after =l sub %bit_pre, 1
  jmp @loop

@done
  %final =w xor %crc, 0xFFFFFFFF
  %ret =l extuw %final
  ret %ret
}
