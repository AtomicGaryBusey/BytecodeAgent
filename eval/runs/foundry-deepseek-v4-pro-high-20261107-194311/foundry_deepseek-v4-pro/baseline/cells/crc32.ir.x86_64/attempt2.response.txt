export function l $crc32(l %p, l %n) {
@start
    %crc =w copy 4294967295
    %idx =l copy 0
    jmp @loop_bytes

@loop_bytes
    %idx_cur =l phi @start %idx, @byte_done %idx_next
    %crc_cur =w phi @start %crc, @byte_done %crc_inner
    %cond =w csltl %idx_cur, %n
    jnz %cond, @byte_start, @done

@byte_start
    %pbyte =l add %p, %idx_cur
    %byte =w loadub %pbyte
    %crc_xor =w xor %crc_cur, %byte
    %bitcnt =l copy 0
    jmp @loop_bits

@loop_bits
    %crc_inner =w phi @byte_start %crc_xor, @bit_done %crc_next
    %cnt =l phi @byte_start %bitcnt, @bit_done %cnt_next
    %more =w csltl %cnt, 8
    jnz %more, @bit_step, @byte_done

@bit_step
    %lsb =w and %crc_inner, 1
    %is_one =w ceql %lsb, 1
    jnz %is_one, @bit_one, @bit_zero

@bit_one
    %shifted =w shr %crc_inner, 1
    %crc_next1 =w xor %shifted, 3988292384
    jmp @bit_done

@bit_zero
    %crc_next0 =w shr %crc_inner, 1
    jmp @bit_done

@bit_done
    %crc_next =w phi @bit_one %crc_next1, @bit_zero %crc_next0
    %cnt_next =l add %cnt, 1
    jmp @loop_bits

@byte_done
    %idx_next =l add %idx_cur, 1
    jmp @loop_bytes

@done
    %final =w xor %crc_cur, 4294967295
    %ret =l extuw %final
    ret %ret
}
