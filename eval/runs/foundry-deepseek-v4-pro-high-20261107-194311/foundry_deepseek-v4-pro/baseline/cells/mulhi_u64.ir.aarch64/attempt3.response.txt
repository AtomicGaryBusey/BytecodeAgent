export function l $mulhi_u64(l %a, l %b) {
@start
    %one =l copy 1
    %shifted =l shl %one, 32
    %mask =l sub %shifted, %one
    %AH =l shr %a, 32
    %AL =l and %a, %mask
    %BH =l shr %b, 32
    %BL =l and %b, %mask
    %P0 =l mul %AL, %BL
    %P1 =l mul %AH, %BL
    %P2 =l mul %AL, %BH
    %P3 =l mul %AH, %BH
    %sum12 =l add %P1, %P2
    %carry12_w =w cultl %sum12, %P1
    %carry12 =l extuw %carry12_w
    %P0_hi =l shr %P0, 32
    %s_low =l add %sum12, %P0_hi
    %carry_add_w =w cultl %s_low, %sum12
    %carry_add =l extuw %carry_add_w
    %carry_total =l add %carry12, %carry_add
    %s_hi =l shr %s_low, 32
    %carry_shifted =l shl %carry_total, 32
    %S_shifted =l add %carry_shifted, %s_hi
    %high =l add %P3, %S_shifted
    ret %high
}
