export function l $mulhi_u64(l %a, l %b) {
@start
    %one =l copy 1
    %shifted =l shl %one, 32
    %mask =l sub %shifted, %one
    %AH =l shr %a, 32
    %AL =l and %a, %mask
    %BH =l shr %b, 32
    %BL =l and %b, %mask
    %x0 =l mul %AL, %BL
    %x1 =l mul %AH, %BL
    %x2 =l mul %AL, %BH
    %x3 =l mul %AH, %BH
    %mid =l add %x1, %x2
    %carry_mid =w cultl %mid, %x1
    %carry_midl =l extuw %carry_mid
    %mid_low =l and %mid, %mask
    %mid_high =l shr %mid, 32
    %x0_high =l shr %x0, 32
    %sum_low =l add %x0_high, %mid_low
    %carry_low =w cultl %sum_low, %x0_high
    %carry_lowl =l extuw %carry_low
    %high =l copy %x3
    %high =l add %high, %mid_high
    %high =l add %high, %carry_lowl
    %carry_mid_shifted =l shl %carry_midl, 32
    %high =l add %high, %carry_mid_shifted
    ret %high
}
