export function l $mulhi_u64(l %a, l %b) {
@start
    %one = copy 1
    %shifted_one = shl %one, 32
    %mask = sub %shifted_one, 1
    %AH = shr %a, 32
    %AL = and %a, %mask
    %BH = shr %b, 32
    %BL = and %b, %mask
    %x0 = mul %AL, %BL
    %x1 = mul %AH, %BL
    %x2 = mul %AL, %BH
    %x3 = mul %AH, %BH
    %x1_lo = and %x1, %mask
    %x1_hi = shr %x1, 32
    %t1 = shl %x1_lo, 32
    %new_acc_lo = add %x0, %t1
    %carry1 = cultl %new_acc_lo, %x0
    %carry1l = extuw %carry1
    %acc_hi = add %x3, %carry1l
    %acc_hi = add %acc_hi, %x1_hi
    %x2_lo = and %x2, %mask
    %x2_hi = shr %x2, 32
    %t2 = shl %x2_lo, 32
    %new_acc_lo2 = add %new_acc_lo, %t2
    %carry2 = cultl %new_acc_lo2, %new_acc_lo
    %carry2l = extuw %carry2
    %acc_hi = add %acc_hi, %carry2l
    %acc_hi = add %acc_hi, %x2_hi
    ret %acc_hi
}
