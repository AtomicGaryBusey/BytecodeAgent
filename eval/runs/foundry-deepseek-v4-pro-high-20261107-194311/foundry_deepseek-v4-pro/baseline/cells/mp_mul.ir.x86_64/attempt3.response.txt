export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%mask =l copy $FFFFFFFF
	%total =l add %na, %nb
	%istotzero =w ceql %total, 0
	jnz %istotzero, @ret_zero, @not_zero
@ret_zero
	%ret0 =l copy 0
	ret %ret0
@not_zero
	jmp @zero_loop_entry
@zero_loop_entry
	%i_init =l copy 0
	jmp @zero_loop
@zero_loop
	%i =l phi @zero_loop_entry %i_init, @zero_inc %i_next
	%cond_z =w csltl %i, %total
	jnz %cond_z, @zero_body, @zero_done
@zero_body
	%off_z =l mul %i, 8
	%ptr_z =l add %r, %off_z
	storel 0, %ptr_z
	jmp @zero_inc
@zero_inc
	%i_next =l add %i, 1
	jmp @zero_loop
@zero_done
	jmp @outer_loop_entry
@outer_loop_entry
	%i_out_init =l copy 0
	jmp @outer_loop
@outer_loop
	%i_out =l phi @outer_loop_entry %i_out_init, @outer_inc %i_out_next
	%cond_out =w csltl %i_out, %na
	jnz %cond_out, @outer_body, @outer_done
@outer_body
	%off_a =l mul %i_out, 8
	%ptr_a =l add %a, %off_a
	%av =l loadl %ptr_a
	jmp @inner_loop_entry
@inner_loop_entry
	%j_init =l copy 0
	jmp @inner_loop
@inner_loop
	%j =l phi @inner_loop_entry %j_init, @inner_inc %j_next
	%av_loop =l phi @inner_loop_entry %av, @inner_inc %av_loop_next
	%cond_inner =w csltl %j, %nb
	jnz %cond_inner, @inner_body, @inner_done
@inner_body
	%off_b =l mul %j, 8
	%ptr_b =l add %b, %off_b
	%bv =l loadl %ptr_b
	%a_lo =l and %av_loop, %mask
	%a_hi =l shr %av_loop, 32
	%b_lo =l and %bv, %mask
	%b_hi =l shr %bv, 32
	%m1 =l mul %a_lo, %b_lo
	%m2 =l mul %a_lo, %b_hi
	%m3 =l mul %a_hi, %b_lo
	%m4 =l mul %a_hi, %b_hi
	%sum_mid =l add %m2, %m3
	%carry_mid_w =w cultl %sum_mid, %m2
	%carry_mid_l =l extsw %carry_mid_w
	%sum_mid_shifted =l shl %sum_mid, 32
	%low1 =l add %m1, %sum_mid_shifted
	%carry_low_w =w cultl %low1, %m1
	%carry_low_l =l extsw %carry_low_w
	%mid_high_part =l shr %sum_mid, 32
	%carry_mid_shifted =l shl %carry_mid_l, 32
	%mid_high =l add %mid_high_part, %carry_mid_shifted
	%high =l add %m4, %mid_high
	%high =l add %high, %carry_low_l
	%pl =l copy %low1
	%ph =l copy %high
	%idx =l add %i_out, %j
	jmp @pl_add_entry
@pl_add_entry
	%pos_pl_init =l copy %idx
	%val_pl_init =l copy %pl
	jmp @add_pl_loop
@add_pl_loop
	%pos_pl =l phi @pl_add_entry %pos_pl_init, @add_pl_body %pos_pl_next
	%val_pl =l phi @pl_add_entry %val_pl_init, @add_pl_body %val_pl_next
	%cond_pl =w csltl %pos_pl, %total
	jnz %cond_pl, @add_pl_body, @add_pl_done
@add_pl_body
	%off_pl =l mul %pos_pl, 8
	%ptr_pl =l add %r, %off_pl
	%rval_pl =l loadl %ptr_pl
	%sum_pl =l add %rval_pl, %val_pl
	%c_pl_w =w cultl %sum_pl, %rval_pl
	%c_pl_l =l extsw %c_pl_w
	storel %sum_pl, %ptr_pl
	%pos_pl_next =l add %pos_pl, 1
	%val_pl_next =l copy %c_pl_l
	%val_nz_pl =w cnel %val_pl_next, 0
	jnz %val_nz_pl, @add_pl_loop, @add_pl_done
@add_pl_done
	jmp @ph_add_entry
@ph_add_entry
	%pos_ph_init =l add %idx, 1
	%val_ph_init =l copy %ph
	jmp @add_ph_loop
@add_ph_loop
	%pos_ph =l phi @ph_add_entry %pos_ph_init, @add_ph_body %pos_ph_next
	%val_ph =l phi @ph_add_entry %val_ph_init, @add_ph_body %val_ph_next
	%cond_ph =w csltl %pos_ph, %total
	jnz %cond_ph, @add_ph_body, @add_ph_done
@add_ph_body
	%off_ph =l mul %pos_ph, 8
	%ptr_ph =l add %r, %off_ph
	%rval_ph =l loadl %ptr_ph
	%sum_ph =l add %rval_ph, %val_ph
	%c_ph_w =w cultl %sum_ph, %rval_ph
	%c_ph_l =l extsw %c_ph_w
	storel %sum_ph, %ptr_ph
	%pos_ph_next =l add %pos_ph, 1
	%val_ph_next =l copy %c_ph_l
	%val_nz_ph =w cnel %val_ph_next, 0
	jnz %val_nz_ph, @add_ph_loop, @add_ph_done
@add_ph_done
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	%av_loop_next =l copy %av_loop
	jmp @inner_loop
@inner_done
	jmp @outer_inc
@outer_inc
	%i_out_next =l add %i_out, 1
	jmp @outer_loop
@outer_done
	jmp @scan_loop_entry
@scan_loop_entry
	%pos_init =l sub %total, 1
	jmp @scan_loop
@scan_loop
	%pos =l phi @scan_loop_entry %pos_init, @scan_zero %pos_dec
	%cond_scan =w csgel %pos, 0
	jnz %cond_scan, @scan_body, @scan_done
@scan_body
	%off_scan =l mul %pos, 8
	%ptr_scan =l add %r, %off_scan
	%val_scan =l loadl %ptr_scan
	%is_zero =w ceql %val_scan, 0
	jnz %is_zero, @scan_zero, @scan_found
@scan_zero
	%pos_dec =l sub %pos, 1
	jmp @scan_loop
@scan_found
	%k_found =l add %pos, 1
	jmp @ret_sig
@scan_done
	%k_zero =l copy 0
	jmp @ret_sig
@ret_sig
	%k =l phi @scan_found %k_found, @scan_done %k_zero
	ret %k
}
