export function l $max_byte(l %p, l %n) {
@start
    %is_pos =w csgtl %n, 0
    jnz %is_pos, @do_loop, @return_zero
@return_zero
    ret 0
@do_loop
    %slot =l alloc8 1
    storel 0, %slot
    %i =l copy 0
    jmp @loop_cond
@loop_cond
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done
@loop_body
    %ptr =l add %p, %i
    %b =l loadub %ptr
    %max =l loadl %slot
    %gt =w cugtl %b, %max
    jnz %gt, @update, @inc_i
@update
    storel %b, %slot
    jmp @inc_i
@inc_i
    %i =l add %i, 1
    jmp @loop_cond
@done
    %result =l loadl %slot
    ret %result
}
