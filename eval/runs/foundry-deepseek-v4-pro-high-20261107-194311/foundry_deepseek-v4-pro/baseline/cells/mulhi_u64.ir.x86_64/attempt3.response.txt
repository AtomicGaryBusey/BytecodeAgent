export function l $mulhi_u64(l %a, l %b) {
@start
    %A0 =l and %a, 0xffffffff
    %A1 =l shr %a, 32
    %B0 =l and %b, 0xffffffff
    %B1 =l shr %b, 32

    %p00 =l mul %A0, %B0
    %p00_hi =l shr %p00, 32

    %p10 =l mul %A1, %B0
    %p01 =l mul %A0, %B1

    %mid_sum =l add %p10, %p01
    %carry_w =w cultl %mid_sum, %p10
    %carry =l extuw %carry_w

    %mid_sum_lo =l and %mid_sum, 0xffffffff
    %mid_sum_hi =l shr %mid_sum, 32

    %t =l add %mid_sum_lo, %p00_hi
    %carry2 =l shr %t, 32

    %mid_hi_full =l add %mid_sum_hi, %carry
    %p11 =l mul %A1, %B1

    %res1 =l add %p11, %mid_hi_full
    %res =l add %res1, %carry2
    ret %res
}
