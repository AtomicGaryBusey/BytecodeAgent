export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =l and %a, 0xffffffff
    %a1 =l shr %a, 32
    %b0 =l and %b, 0xffffffff
    %b1 =l shr %b, 32

    %low =l mul %a0, %b0
    %low_hi =l shr %low, 32

    %mid1 =l mul %a1, %b0
    %mid2 =l mul %a0, %b1
    %mid_sum =l add %mid1, %mid2
    %carry_w =w cultl %mid_sum, %mid1
    %carry =l extuw %carry_w

    %mid_lo =l and %mid_sum, 0xffffffff
    %mid_hi_shift =l shr %mid_sum, 32
    %carry_shift =l shl %carry, 32
    %mid_hi =l or %mid_hi_shift, %carry_shift

    %sum_ml =l add %mid_lo, %low_hi
    %C =l shr %sum_ml, 32

    %high1 =l mul %a1, %b1
    %temp =l add %high1, %mid_hi
    %res =l add %temp, %C
    ret %res
}
