export function l $max_byte(l %p, l %n) {
@start
    %gt0 =w csgtl %n, 0
    jnz %gt0, @loop, @ret0
@ret0
    ret 0
@loop
    %max =l phi @start 0, @after %max_after
    %ptr =l phi @start %p, @after %ptr_next
    %cnt =l phi @start %n, @after %cnt_next
    %byte =l loadub %ptr
    %cond =w cultl %max, %byte
    jnz %cond, @set, @skip
@set
    jmp @after
@skip
    jmp @after
@after
    %max_after =l phi @set %byte, @skip %max
    %ptr_next =l add %ptr, 1
    %cnt_next =l sub %cnt, 1
    %cont =w csgtl %cnt_next, 0
    jnz %cont, @loop, @endloop
@endloop
    ret %max_after
}
