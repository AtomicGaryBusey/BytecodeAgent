export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i_ptr =l alloc8 1
	%min_ptr =l alloc8 1
	%max_ptr =l alloc8 1
	%sum_ptr =l alloc8 1
	%odd_ptr =l alloc8 1
	storel 0, %i_ptr
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	storel %y0, %min_ptr
	storel %y0, %max_ptr
	storel %y0, %sum_ptr
	%odd_bit0 =l andl %y0, 1
	storel %odd_bit0, %odd_ptr
	storel 1, %i_ptr
	jmp @loop_cond
@loop_cond
	%i_val =l loadl %i_ptr
	%cond =w csltl %i_val, %n
	jnz %cond, @loop_body, @done
@loop_body
	%offset =l mul %i_val, 8
	%addr =l add %a, %offset
	%x =l loadl %addr
	%y =l call %f(l %x)
	%cur_min =l loadl %min_ptr
	%is_lt =w csltl %y, %cur_min
	%new_min =l select %is_lt, %y, %cur_min
	storel %new_min, %min_ptr
	%cur_max =l loadl %max_ptr
	%is_gt =w csgtl %y, %cur_max
	%new_max =l select %is_gt, %y, %cur_max
	storel %new_max, %max_ptr
	%cur_sum =l loadl %sum_ptr
	%new_sum =l add %cur_sum, %y
	storel %new_sum, %sum_ptr
	%cur_odd =l loadl %odd_ptr
	%odd_bit =l andl %y, 1
	%new_odd =l add %cur_odd, %odd_bit
	storel %new_odd, %odd_ptr
	%new_i =l add %i_val, 1
	storel %new_i, %i_ptr
	jmp @loop_cond
@done
	%min =l loadl %min_ptr
	%max =l loadl %max_ptr
	%sum =l loadl %sum_ptr
	%odd_ret =l loadl %odd_ptr
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd_ret
}
