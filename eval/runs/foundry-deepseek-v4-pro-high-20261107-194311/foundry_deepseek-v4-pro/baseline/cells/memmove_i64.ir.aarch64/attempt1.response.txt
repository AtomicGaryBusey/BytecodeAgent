export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %n_z =w ceql %n, 0
  jnz %n_z, @done, @chk_eq
@chk_eq
  %eq =w ceql %dst, %src
  jnz %eq, @done, @check_dir
@check_dir
  %lt =w cultl %dst, %src
  jnz %lt, @fwd, @bwd
@fwd
  %rem_f =l %n
  jmp @fwd_start
@fwd_start
  %rem_f_gt0 =w cugtl %rem_f, 0
  jnz %rem_f_gt0, @fwd_body, @done
@fwd_body
  %tmp_f =l loadl %src
  storel %tmp_f, %dst
  %dst =l add %dst, 8
  %src =l add %src, 8
  %rem_f =l sub %rem_f, 1
  jmp @fwd_start
@bwd
  %bytes =l shl %n, 3
  %end_dst =l add %dst, %bytes
  %end_src =l add %src, %bytes
  %p =l sub %end_dst, 8
  %q =l sub %end_src, 8
  %rem_b =l %n
  jmp @bwd_start
@bwd_start
  %rem_b_gt0 =w cugtl %rem_b, 0
  jnz %rem_b_gt0, @bwd_body, @done
@bwd_body
  %tmp_b =l loadl %q
  storel %tmp_b, %p
  %p =l sub %p, 8
  %q =l sub %q, 8
  %rem_b =l sub %rem_b, 1
  jmp @bwd_start
@done
  ret 0
}
