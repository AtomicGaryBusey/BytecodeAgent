export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %n_z =w ceql %n, 0
  jnz %n_z, @done, @chk_eq
@chk_eq
  %eq =w ceql %dst, %src
  jnz %eq, @done, @check_dir
@check_dir
  %lt =w cultl %dst, %src
  jnz %lt, @fwd, @bwd
@fwd
  jmp @fwd_start
@fwd_start
  %dst_f =l phi @fwd %dst, @fwd_body %dst_next
  %src_f =l phi @fwd %src, @fwd_body %src_next
  %rem_f =l phi @fwd %n, @fwd_body %rem_next
  %rem_f_z =w ceql %rem_f, 0
  jnz %rem_f_z, @done, @fwd_body
@fwd_body
  %tmp_f =l loadl %src_f
  storel %tmp_f, %dst_f
  %dst_next =l add %dst_f, 8
  %src_next =l add %src_f, 8
  %rem_next =l sub %rem_f, 1
  jmp @fwd_start
@bwd
  %bytes =l shl %n, 3
  %end_dst =l add %dst, %bytes
  %end_src =l add %src, %bytes
  %p_init =l sub %end_dst, 8
  %q_init =l sub %end_src, 8
  jmp @bwd_start
@bwd_start
  %p =l phi @bwd %p_init, @bwd_body %p_next
  %q =l phi @bwd %q_init, @bwd_body %q_next
  %rem_b =l phi @bwd %n, @bwd_body %rem_b_next
  %rem_b_z =w ceql %rem_b, 0
  jnz %rem_b_z, @done, @bwd_body
@bwd_body
  %tmp_b =l loadl %q
  storel %tmp_b, %p
  %p_next =l sub %p, 8
  %q_next =l sub %q, 8
  %rem_b_next =l sub %rem_b, 1
  jmp @bwd_start
@done
  ret 0
}
