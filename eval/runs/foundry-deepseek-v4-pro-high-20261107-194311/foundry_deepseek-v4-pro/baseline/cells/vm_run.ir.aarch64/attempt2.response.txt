export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc_start =l copy 0
	%sp_start =l copy 0
	jmp @loop
@loop
	%pc =l phi @start %pc_start, @other %pc_other, @push %pc_push, @add %pc_add, @sub %pc_sub, @mul_op %pc_mul, @dup %pc_dup, @swap %pc_swap, @drop %pc_drop, @over %pc_over, @lt %pc_lt, @jz_taken %pc_jztaken, @jz_nottaken %pc_jznottaken, @jmp %pc_jmp
	%sp =l phi @start %sp_start, @other %sp_other, @push %sp_push, @add %sp_add, @sub %sp_sub, @mul_op %sp_mul, @dup %sp_dup, @swap %sp_swap, @drop %sp_drop, @over %sp_over, @lt %sp_lt, @jz_taken %sp_jztaken, @jz_nottaken %sp_jznottaken, @jmp %sp_jmp
	%donelen =w ceql %pc, %len
	jnz %donelen, @stop, @fetch
@fetch
	%pc8 =l mul %pc, 8
	%codepc =l add %code, %pc8
	%op =l loadl %codepc
	%ishalt =w ceql %op, 0
	jnz %ishalt, @stop, @not0
@not0
	%is1 =w ceql %op, 1
	jnz %is1, @push, @not1
@not1
	%is2 =w ceql %op, 2
	jnz %is2, @add, @not2
@not2
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @not3
@not3
	%is4 =w ceql %op, 4
	jnz %is4, @mul_op, @not4
@not4
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @not5
@not5
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @not6
@not6
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @not7
@not7
	%is8 =w ceql %op, 8
	jnz %is8, @over, @not8
@not8
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @not9
@not9
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @not10
@not10
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @other
@other
	%pc_other =l add %pc, 1
	%sp_other =l copy %sp
	jmp @loop
@push
	%pc1_push =l add %pc, 1
	%off_oper =l mul %pc1_push, 8
	%addr_oper =l add %code, %off_oper
	%val_push =l loadl %addr_oper
	%off_dst =l mul %sp, 8
	%dst_push =l add %stack, %off_dst
	storel %val_push, %dst_push
	%sp_push =l add %sp, 1
	%pc_push =l add %pc, 2
	jmp @loop
@add
	%top_idx_add =l sub %sp, 1
	%off_b_add =l mul %top_idx_add, 8
	%addr_b_add =l add %stack, %off_b_add
	%b_add =l loadl %addr_b_add
	%below_idx_add =l sub %sp, 2
	%off_a_add =l mul %below_idx_add, 8
	%addr_a_add =l add %stack, %off_a_add
	%a_add =l loadl %addr_a_add
	%res_add =l add %a_add, %b_add
	storel %res_add, %addr_a_add
	%sp_add =l sub %sp, 1
	%pc_add =l add %pc, 1
	jmp @loop
@sub
	%top_idx_sub =l sub %sp, 1
	%off_b_sub =l mul %top_idx_sub, 8
	%addr_b_sub =l add %stack, %off_b_sub
	%b_sub =l loadl %addr_b_sub
	%below_idx_sub =l sub %sp, 2
	%off_a_sub =l mul %below_idx_sub, 8
	%addr_a_sub =l add %stack, %off_a_sub
	%a_sub =l loadl %addr_a_sub
	%res_sub =l sub %a_sub, %b_sub
	storel %res_sub, %addr_a_sub
	%sp_sub =l sub %sp, 1
	%pc_sub =l add %pc, 1
	jmp @loop
@mul_op
	%top_idx_mul =l sub %sp, 1
	%off_b_mul =l mul %top_idx_mul, 8
	%addr_b_mul =l add %stack, %off_b_mul
	%b_mul =l loadl %addr_b_mul
	%below_idx_mul =l sub %sp, 2
	%off_a_mul =l mul %below_idx_mul, 8
	%addr_a_mul =l add %stack, %off_a_mul
	%a_mul =l loadl %addr_a_mul
	%res_mul =l mul %a_mul, %b_mul
	storel %res_mul, %addr_a_mul
	%sp_mul =l sub %sp, 1
	%pc_mul =l add %pc, 1
	jmp @loop
@dup
	%top_idx_dup =l sub %sp, 1
	%off_top_dup =l mul %top_idx_dup, 8
	%addr_top_dup =l add %stack, %off_top_dup
	%val_dup =l loadl %addr_top_dup
	%off_new_dup =l mul %sp, 8
	%addr_new_dup =l add %stack, %off_new_dup
	storel %val_dup, %addr_new_dup
	%sp_dup =l add %sp, 1
	%pc_dup =l add %pc, 1
	jmp @loop
@swap
	%idx1 =l sub %sp, 1
	%off1 =l mul %idx1, 8
	%addr1 =l add %stack, %off1
	%idx2 =l sub %sp, 2
	%off2 =l mul %idx2, 8
	%addr2 =l add %stack, %off2
	%v1 =l loadl %addr1
	%v2 =l loadl %addr2
	storel %v2, %addr1
	storel %v1, %addr2
	%sp_swap =l copy %sp
	%pc_swap =l add %pc, 1
	jmp @loop
@drop
	%sp_drop =l sub %sp, 1
	%pc_drop =l add %pc, 1
	jmp @loop
@over
	%below_idx_over =l sub %sp, 2
	%off_below =l mul %below_idx_over, 8
	%addr_below =l add %stack, %off_below
	%val_over =l loadl %addr_below
	%off_new_over =l mul %sp, 8
	%addr_new_over =l add %stack, %off_new_over
	storel %val_over, %addr_new_over
	%sp_over =l add %sp, 1
	%pc_over =l add %pc, 1
	jmp @loop
@lt
	%top_idx_lt =l sub %sp, 1
	%off_b_lt =l mul %top_idx_lt, 8
	%addr_b_lt =l add %stack, %off_b_lt
	%b_lt =l loadl %addr_b_lt
	%below_idx_lt =l sub %sp, 2
	%off_a_lt =l mul %below_idx_lt, 8
	%addr_a_lt =l add %stack, %off_a_lt
	%a_lt =l loadl %addr_a_lt
	%res_lt_w =w csltl %a_lt, %b_lt
	%res_lt =l extsw %res_lt_w
	storel %res_lt, %addr_a_lt
	%sp_lt =l sub %sp, 1
	%pc_lt =l add %pc, 1
	jmp @loop
@jz
	%top_idx_jz =l sub %sp, 1
	%off_v =l mul %top_idx_jz, 8
	%addr_v =l add %stack, %off_v
	%v =l loadl %addr_v
	%sp_popped =l sub %sp, 1
	%pc1_jz =l add %pc, 1
	%off_t =l mul %pc1_jz, 8
	%addr_t =l add %code, %off_t
	%t_val =l loadl %addr_t
	%iszero =w ceql %v, 0
	jnz %iszero, @jz_taken, @jz_nottaken
@jz_taken
	%pc_jztaken =l copy %t_val
	%sp_jztaken =l copy %sp_popped
	jmp @loop
@jz_nottaken
	%pc_jznottaken =l add %pc, 2
	%sp_jznottaken =l copy %sp_popped
	jmp @loop
@jmp
	%pc1_jmp =l add %pc, 1
	%off_jmp =l mul %pc1_jmp, 8
	%addr_jmp =l add %code, %off_jmp
	%t_jmp =l loadl %addr_jmp
	%pc_jmp =l copy %t_jmp
	%sp_jmp =l copy %sp
	jmp @loop
@stop
	%sp_stop =l phi @loop %sp, @fetch %sp
	%sp_eq0 =w ceql %sp_stop, 0
	jnz %sp_eq0, @stop_ret0, @stop_ret_top
@stop_ret0
	%top_val0 =l copy 0
	jmp @finalize
@stop_ret_top
	%top_idx_ret =l sub %sp_stop, 1
	%off_ret =l mul %top_idx_ret, 8
	%addr_ret =l add %stack, %off_ret
	%top_val1 =l loadl %addr_ret
	jmp @finalize
@finalize
	%top_val =l phi @stop_ret0 %top_val0, @stop_ret_top %top_val1
	%i_start =l copy %sp_stop
	jmp @zero_cond
@zero_cond
	%i =l phi @finalize %i_start, @zero_body %i_next
	%zerocond =w csltl %i, %cap
	jnz %zerocond, @zero_body, @zero_done
@zero_body
	%off_zero =l mul %i, 8
	%addr_zero =l add %stack, %off_zero
	storel 0, %addr_zero
	%i_next =l add %i, 1
	jmp @zero_cond
@zero_done
	ret %top_val
}
