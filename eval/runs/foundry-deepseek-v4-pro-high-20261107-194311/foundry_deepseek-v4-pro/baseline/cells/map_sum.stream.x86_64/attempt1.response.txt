map_sum:
    push r12
    push r13
    push r14
    push r15
    push rbp
    mov r12, rdx
    xor r13, r13
    mov r14, rdi
    lea r15, [rdi + rsi*8]
.Lloop:
    cmp r14, r15
    je .Ldone
    mov rdi, qword ptr [r14]
    call r12
    add r13, rax
    add r14, 8
    jmp .Lloop
.Ldone:
    mov rax, r13
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    ret
