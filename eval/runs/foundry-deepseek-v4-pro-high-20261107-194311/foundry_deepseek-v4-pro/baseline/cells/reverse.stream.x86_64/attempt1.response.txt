reverse:
    test rsi, rsi
    jle .Lret0
    lea rsi, [rdi + rsi*8 - 8]
.Lloop:
    cmp rdi, rsi
    jge .Lret0
    mov rax, qword ptr [rdi]
    mov rdx, qword ptr [rsi]
    mov qword ptr [rdi], rdx
    mov qword ptr [rsi], rax
    add rdi, 8
    sub rsi, 8
    jmp .Lloop
.Lret0:
    xor eax, eax
    ret
