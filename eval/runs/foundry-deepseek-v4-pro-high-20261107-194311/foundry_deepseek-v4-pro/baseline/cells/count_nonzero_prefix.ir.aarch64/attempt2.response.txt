export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %i0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @next %next_i
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @exit_loop
@loop_body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %elem =l loadl %addr
    %is_zero =w ceql %elem, 0
    jnz %is_zero, @found_zero, @next
@next
    %next_i =l add %i, 1
    jmp @loop
@found_zero
    ret %i
@exit_loop
    ret %i
}
