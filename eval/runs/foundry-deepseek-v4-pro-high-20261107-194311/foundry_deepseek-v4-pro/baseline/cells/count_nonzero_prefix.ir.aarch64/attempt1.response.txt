export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %i0 =l copy 0
    jmp @loop(%i0)
@loop(%i)
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @exit_loop
@loop_body
    jmp @check_element(%i)
@exit_loop
    jmp @done(%i)
@check_element(%i)
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %elem =l loadl %addr
    %is_zero =w ceql %elem, 0
    jnz %is_zero, @found_zero, @next
@found_zero
    jmp @done(%i)
@next
    %next_i =l add %i, 1
    jmp @loop(%next_i)
@done(%r)
    ret %r
}
