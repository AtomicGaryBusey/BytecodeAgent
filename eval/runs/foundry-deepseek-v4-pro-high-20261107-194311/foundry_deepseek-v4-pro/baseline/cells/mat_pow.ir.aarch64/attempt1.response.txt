export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n2 =l mul %n, %n
    %i_slot =l alloc8 8
    %j_slot =l alloc8 8
    storel 0, %i_slot
    jmp @ident_i_cond
@ident_i_cond
    %i_val =l loadl %i_slot
    %icond =w csltl %i_val, %n
    jnz %icond, @ident_i_body, @ident_i_done
@ident_i_body
    storel 0, %j_slot
    jmp @ident_j_cond
@ident_j_cond
    %j_val =l loadl %j_slot
    %jcond =w csltl %j_val, %n
    jnz %jcond, @ident_j_body, @ident_inc_i
@ident_j_body
    %off1 =l mul %i_val, %n
    %off2 =l add %off1, %j_val
    %byteoff =l mul %off2, 8
    %addr =l add %out, %byteoff
    %eq =w ceql %i_val, %j_val
    %val =l extsw %eq
    storel %val, %addr
    %new_j =l add %j_val, 1
    storel %new_j, %j_slot
    jmp @ident_j_cond
@ident_inc_i
    %new_i =l add %i_val, 1
    storel %new_i, %i_slot
    jmp @ident_i_cond
@ident_i_done
    %idx_slot =l alloc8 8
    storel 0, %idx_slot
    jmp @copy_cond
@copy_cond
    %idx =l loadl %idx_slot
    %idxcond =w csltl %idx, %n2
    jnz %idxcond, @copy_body, @copy_done
@copy_body
    %src_off =l mul %idx, 8
    %src =l add %a, %src_off
    %valc =l loadl %src
    %dst_off =l mul %idx, 8
    %dst =l add %tmp, %dst_off
    storel %valc, %dst
    %new_idx =l add %idx, 1
    storel %new_idx, %idx_slot
    jmp @copy_cond
@copy_done
    %k_slot =l alloc8 8
    storel %k, %k_slot
    jmp @exp_loop_cond
@exp_loop_cond
    %k_val =l loadl %k_slot
    %k_nonzero =w csnel %k_val, 0
    jnz %k_nonzero, @exp_body, @exp_done
@exp_body
    %lsb =l and %k_val, 1
    %lsb_ne =w csnel %lsb, 0
    jnz %lsb_ne, @exp_mult, @exp_square
@exp_mult
    %tmp2 =l mul %n2, 8
    %tmp2 =l add %tmp, %tmp2
    %i_slot2 =l alloc8 8
    %j_slot2 =l alloc8 8
    %t_slot2 =l alloc8 8
    %sum_slot2 =l alloc8 8
    storel 0, %i_slot2
    jmp @mul_RB_i_cond
@mul_RB_i_cond
    %i2 =l loadl %i_slot2
    %i2cond =w csltl %i2, %n
    jnz %i2cond, @mul_RB_i_body, @mul_RB_done
@mul_RB_i_body
    storel 0, %j_slot2
    jmp @mul_RB_j_cond
@mul_RB_j_cond
    %j2 =l loadl %j_slot2
    %j2cond =w csltl %j2, %n
    jnz %j2cond, @mul_RB_j_body, @mul_RB_inc_i
@mul_RB_j_body
    storel 0, %sum_slot2
    storel 0, %t_slot2
    jmp @mul_RB_t_cond
@mul_RB_t_cond
    %t2 =l loadl %t_slot2
    %t2cond =w csltl %t2, %n
    jnz %t2cond, @mul_RB_t_body, @mul_RB_t_done
@mul_RB_t_body
    %a_off1 =l mul %i2, %n
    %a_off2 =l add %a_off1, %t2
    %a_byte =l mul %a_off2, 8
    %a_addr =l add %out, %a_byte
    %a_val =l loadl %a_addr
    %b_off1 =l mul %t2, %n
    %b_off2 =l add %b_off1, %j2
    %b_byte =l mul %b_off2, 8
    %b_addr =l add %tmp, %b_byte
    %b_val =l loadl %b_addr
    %prod =l mul %a_val, %b_val
    %sum_cur =l loadl %sum_slot2
    %new_sum =l add %sum_cur, %prod
    storel %new_sum, %sum_slot2
    %new_t =l add %t2, 1
    storel %new_t, %t_slot2
    jmp @mul_RB_t_cond
@mul_RB_t_done
    %c_off1 =l mul %i2, %n
    %c_off2 =l add %c_off1, %j2
    %c_byte =l mul %c_off2, 8
    %c_addr =l add %tmp2, %c_byte
    %sum_val =l loadl %sum_slot2
    storel %sum_val, %c_addr
    %new_j2 =l add %j2, 1
    storel %new_j2, %j_slot2
    jmp @mul_RB_j_cond
@mul_RB_inc_i
    %new_i2 =l add %i2, 1
    storel %new_i2, %i_slot2
    jmp @mul_RB_i_cond
@mul_RB_done
    %copy_RB_slot =l alloc8 8
    storel 0, %copy_RB_slot
    jmp @copy_RB_cond
@copy_RB_cond
    %copy_RB_idx =l loadl %copy_RB_slot
    %copy_RB_cond =w csltl %copy_RB_idx, %n2
    jnz %copy_RB_cond, @copy_RB_body, @copy_RB_done
@copy_RB_body
    %src_off_RB =l mul %copy_RB_idx, 8
    %src_ad_RB =l add %tmp2, %src_off_RB
    %val_RB =l loadl %src_ad_RB
    %dst_off_RB =l mul %copy_RB_idx, 8
    %dst_ad_RB =l add %out, %dst_off_RB
    storel %val_RB, %dst_ad_RB
    %new_copy_RB =l add %copy_RB_idx, 1
    storel %new_copy_RB, %copy_RB_slot
    jmp @copy_RB_cond
@copy_RB_done
    jmp @exp_square
@exp_square
    %tmp2_sq =l mul %n2, 8
    %tmp2_sq =l add %tmp, %tmp2_sq
    %i_slot3 =l alloc8 8
    %j_slot3 =l alloc8 8
    %t_slot3 =l alloc8 8
    %sum_slot3 =l alloc8 8
    storel 0, %i_slot3
    jmp @mul_BB_i_cond
@mul_BB_i_cond
    %i3 =l loadl %i_slot3
    %i3cond =w csltl %i3, %n
    jnz %i3cond, @mul_BB_i_body, @mul_BB_done
@mul_BB_i_body
    storel 0, %j_slot3
    jmp @mul_BB_j_cond
@mul_BB_j_cond
    %j3 =l loadl %j_slot3
    %j3cond =w csltl %j3, %n
    jnz %j3cond, @mul_BB_j_body, @mul_BB_inc_i
@mul_BB_j_body
    storel 0, %sum_slot3
    storel 0, %t_slot3
    jmp @mul_BB_t_cond
@mul_BB_t_cond
    %t3 =l loadl %t_slot3
    %t3cond =w csltl %t3, %n
    jnz %t3cond, @mul_BB_t_body, @mul_BB_t_done
@mul_BB_t_body
    %a_off1_BB =l mul %i3, %n
    %a_off2_BB =l add %a_off1_BB, %t3
    %a_byte_BB =l mul %a_off2_BB, 8
    %a_addr_BB =l add %tmp, %a_byte_BB
    %a_val_BB =l loadl %a_addr_BB
    %b_off1_BB =l mul %t3, %n
    %b_off2_BB =l add %b_off1_BB, %j3
    %b_byte_BB =l mul %b_off2_BB, 8
    %b_addr_BB =l add %tmp, %b_byte_BB
    %b_val_BB =l loadl %b_addr_BB
    %prod_BB =l mul %a_val_BB, %b_val_BB
    %sum_cur_BB =l loadl %sum_slot3
    %new_sum_BB =l add %sum_cur_BB, %prod_BB
    storel %new_sum_BB, %sum_slot3
    %new_t_BB =l add %t3, 1
    storel %new_t_BB, %t_slot3
    jmp @mul_BB_t_cond
@mul_BB_t_done
    %c_off1_BB =l mul %i3, %n
    %c_off2_BB =l add %c_off1_BB, %j3
    %c_byte_BB =l mul %c_off2_BB, 8
    %c_addr_BB =l add %tmp2_sq, %c_byte_BB
    %sum_val_BB =l loadl %sum_slot3
    storel %sum_val_BB, %c_addr_BB
    %new_j3 =l add %j3, 1
    storel %new_j3, %j_slot3
    jmp @mul_BB_j_cond
@mul_BB_inc_i
    %new_i3 =l add %i3, 1
    storel %new_i3, %i_slot3
    jmp @mul_BB_i_cond
@mul_BB_done
    %copy_BB_slot =l alloc8 8
    storel 0, %copy_BB_slot
    jmp @copy_BB_cond
@copy_BB_cond
    %copy_BB_idx =l loadl %copy_BB_slot
    %copy_BB_cond =w csltl %copy_BB_idx, %n2
    jnz %copy_BB_cond, @copy_BB_body, @copy_BB_done
@copy_BB_body
    %src_off_BB =l mul %copy_BB_idx, 8
    %src_ad_BB =l add %tmp2_sq, %src_off_BB
    %val_BB =l loadl %src_ad_BB
    %dst_off_BB =l mul %copy_BB_idx, 8
    %dst_ad_BB =l add %tmp, %dst_off_BB
    storel %val_BB, %dst_ad_BB
    %new_copy_BB =l add %copy_BB_idx, 1
    storel %new_copy_BB, %copy_BB_slot
    jmp @copy_BB_cond
@copy_BB_done
    jmp @exp_shift
@exp_shift
    %k_val_shift =l loadl %k_slot
    %new_k =l shr %k_val_shift, 1
    storel %new_k, %k_slot
    jmp @exp_loop_cond
@exp_done
    %zero_idx_slot =l alloc8 8
    storel 0, %zero_idx_slot
    jmp @zero_cond
@zero_cond
    %zero_idx =l loadl %zero_idx_slot
    %max_zero =l mul 2, %n2
    %zero_cond =w csltl %zero_idx, %max_zero
    jnz %zero_cond, @zero_body, @done
@zero_body
    %zero_off =l mul %zero_idx, 8
    %zero_addr =l add %tmp, %zero_off
    storel 0, %zero_addr
    %new_zero =l add %zero_idx, 1
    storel %new_zero, %zero_idx_slot
    jmp @zero_cond
@done
    ret 0
}
