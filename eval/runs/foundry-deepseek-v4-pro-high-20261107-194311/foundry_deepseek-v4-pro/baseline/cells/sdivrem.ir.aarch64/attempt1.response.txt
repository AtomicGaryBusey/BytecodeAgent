export function l $sdivrem(l %a, l %b, l %rem_ptr) {
@start
	%is_b_zero =w ceql %b, 0
	jnz %is_b_zero, @b_zero, @check_overflow
@check_overflow
	%min =l copy -9223372036854775808
	%negone =l copy -1
	%a_is_min =w ceql %a, %min
	%b_is_negone =w ceql %b, %negone
	%overflow =w and %a_is_min, %b_is_negone
	jnz %overflow, @overflow_case, @normal
@b_zero
	%q_zero =l copy 0
	jmp @store
@overflow_case
	%q_min =l copy -9223372036854775808
	%r_zero =l copy 0
	jmp @store
@normal
	%q_normal =l div %a, %b
	%r_normal =l rem %a, %b
	jmp @store
@store
	%r =l phi @b_zero %a, @overflow_case %r_zero, @normal %r_normal
	%q =l phi @b_zero %q_zero, @overflow_case %q_min, @normal %q_normal
	storel %r, %rem_ptr
	ret %q
}
