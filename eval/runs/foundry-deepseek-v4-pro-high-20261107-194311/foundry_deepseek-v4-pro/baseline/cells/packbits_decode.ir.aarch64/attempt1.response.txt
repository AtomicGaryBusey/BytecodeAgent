export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop_head

@loop_head
    %i =l phi @start 0, @lit_done %lit_i_final, @rep_done %rep_i_final, @noop_done %noop_i_final
    %written =l phi @start 0, @lit_done %lit_written_final, @rep_done %rep_written_final, @noop_done %noop_written
    %cond_i =w csltl %i, %n
    jnz %cond_i, @process_packet, @done_normal

@process_packet
    %p =l add %src, %i
    %h_w =w loadsb %p
    %is_noop =w ceqw %h_w, -128
    jnz %is_noop, @noop_check, @not_noop

@not_noop
    %is_lit =w csgew %h_w, 0
    jnz %is_lit, @lit_check, @repeat_check

@lit_check
    %req_w =w add %h_w, 2
    %req_l =l extsw %req_w
    %rem =l sub %n, %i
    %insuf =w csltl %rem, %req_l
    jnz %insuf, @err_src, @lit_src_ok

@lit_src_ok
    %cnt_w =w add %h_w, 1
    %cnt_l =l extsw %cnt_w
    %i_after_hdr =l add %i, 1
    jmp @lit_loop_start

@lit_loop_start
    %j =l phi @lit_src_ok 0, @lit_write %j_next
    %written_cur =l phi @lit_src_ok %written, @lit_write %written_new
    %cond_j =w csltl %j, %cnt_l
    jnz %cond_j, @lit_write, @lit_done

@lit_write
    %cap_full =w ceql %written_cur, %cap
    jnz %cap_full, @err_cap
    %addr1 =l add %src, %i_after_hdr
    %addr =l add %addr1, %j
    %b =w loadub %addr
    %outaddr =l add %out, %written_cur
    storeb %b, %outaddr
    %j_next =l add %j, 1
    %written_new =l add %written_cur, 1
    jmp @lit_loop_start

@lit_done
    %lit_i_final =l add %i_after_hdr, %cnt_l
    %lit_written_final =l add %written_cur, 0
    jmp @loop_head

@repeat_check
    %rem2 =l sub %n, %i
    %insuf2 =w csltl %rem2, 2
    jnz %insuf2, @err_src
    %neg_h =w neg %h_w
    %rep_w =w add %neg_h, 1
    %rep_l =l extsw %rep_w
    %i_after_hdr_rep =l add %i, 1
    %rptr =l add %src, %i_after_hdr_rep
    %rbyte =w loadub %rptr
    jmp @rep_loop_start

@rep_loop_start
    %j_rep =l phi @repeat_check 0, @rep_write %j_next_rep
    %written_cur_rep =l phi @repeat_check %written, @rep_write %written_new_rep
    %cond_j_rep =w csltl %j_rep, %rep_l
    jnz %cond_j_rep, @rep_write, @rep_done

@rep_write
    %cap_full_rep =w ceql %written_cur_rep, %cap
    jnz %cap_full_rep, @err_cap
    %outaddr_rep =l add %out, %written_cur_rep
    storeb %rbyte, %outaddr_rep
    %j_next_rep =l add %j_rep, 1
    %written_new_rep =l add %written_cur_rep, 1
    jmp @rep_loop_start

@rep_done
    %rep_i_final =l add %i, 2
    %rep_written_final =l add %written_cur_rep, 0
    jmp @loop_head

@noop_check
    %rem_noop =l sub %n, %i
    %insuf_noop =w csltl %rem_noop, 1
    jnz %insuf_noop, @err_src
    jmp @noop_done

@noop_done
    %noop_i_final =l add %i, 1
    %noop_written =l add %written, 0
    jmp @loop_head

@done_normal
    ret %written

@err_src
    ret -2

@err_cap
    ret -1
}
