mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    mov r13, r8
    mov r14, rcx
    mov r15, rdx
    mov r8, r12
    imul r8, r8
    mov rsi, rdi
    mov rdi, r13
    mov rcx, r8
    cld
    rep movsq
    mov rdi, r14
    mov rcx, r12
    imul rcx, r12
    xor eax, eax
    rep stosq
    xor r9d, r9d
.Lid_loop:
    cmp r9, r12
    jae .Lid_done
    mov rax, r9
    imul rax, r12
    add rax, r9
    mov qword ptr [r14 + rax*8], 1
    inc r9
    jmp .Lid_loop
.Lid_done:
    jmp .Lpow_test
.Lpow_test:
    test r15, r15
    jz .Lpow_done
    test r15, 1
    jz .Lno_mult
    mov rcx, r12
    imul rcx, r12
    lea rdi, [r13 + rcx*8]
    lea rcx, [rip + .Lafter_mul1]
    mov r8, r12
    mov rsi, r14
    mov rdx, r13
    jmp multiply
.Lafter_mul1:
    mov rcx, r12
    imul rcx, r12
    lea rsi, [r13 + rcx*8]
    mov rdi, r14
    cld
    rep movsq
.Lno_mult:
.Lsquare:
    mov rcx, r12
    imul rcx, r12
    lea rdi, [r13 + rcx*8]
    lea rcx, [rip + .Lafter_square]
    mov r8, r12
    mov rsi, r13
    mov rdx, r13
    jmp multiply
.Lafter_square:
    mov rcx, r12
    imul rcx, r12
    lea rsi, [r13 + rcx*8]
    mov rdi, r13
    cld
    rep movsq
    shr r15, 1
    jmp .Lpow_test
multiply:
    xor r9d, r9d
.Li_loop:
    cmp r9, r8
    jae .Ldone_i
    xor r10d, r10d
.Lj_loop:
    cmp r10, r8
    jae .Ldone_j
    xor ebp, ebp
    xor r11d, r11d
.Lt_loop:
    cmp r11, r8
    jae .Ldone_t
    mov rax, r9
    imul rax, r8
    add rax, r11
    mov rbx, [rsi + rax*8]
    mov rax, r11
    imul rax, r8
    add rax, r10
    imul rbx, [rdx + rax*8]
    add rbp, rbx
    inc r11
    jmp .Lt_loop
.Ldone_t:
    mov rax, r9
    imul rax, r8
    add rax, r10
    mov [rdi + rax*8], rbp
    inc r10
    jmp .Lj_loop
.Ldone_j:
    inc r9
    jmp .Li_loop
.Ldone_i:
    jmp rcx
.Lpow_done:
    mov rcx, r12
    imul rcx, r12
    shl rcx, 1
    mov rdi, r13
    xor eax, eax
    rep stosq
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    xor eax, eax
    ret
