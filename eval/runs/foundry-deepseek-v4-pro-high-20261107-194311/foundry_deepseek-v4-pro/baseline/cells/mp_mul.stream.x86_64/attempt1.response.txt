mp_mul:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdx
    mov r13, rcx
    mov r14, r8
    mov r15, rdi
    test rsi, rsi
    jle .Lzero_args
    test r13, r13
    jle .Lzero_args
    lea rcx, [rsi + r13]
    test rcx, rcx
    jz .Lskip_zero
    mov rdi, r14
    xor eax, eax
    rep stosq
.Lskip_zero:
    xor r10d, r10d
.Louter:
    cmp r10, rsi
    jge .Louter_end
    mov r9, qword ptr [r15 + r10*8]
    xor edi, edi
    xor r11d, r11d
.Linner:
    cmp r11, r13
    jge .Linner_end
    lea rcx, [r10 + r11]
    mov rax, r9
    mul qword ptr [r12 + r11*8]
    add rax, rdi
    adc rdx, 0
    add qword ptr [r14 + rcx*8], rax
    adc rdx, 0
    mov rdi, rdx
    inc r11
    jmp .Linner
.Linner_end:
    lea rcx, [r10 + r13]
    lea rax, [rsi + r13]
.Lcarry_loop:
    add qword ptr [r14 + rcx*8], rdi
    jnc .Lcarry_done
    inc rcx
    mov rdi, 1
    cmp rcx, rax
    jb .Lcarry_loop
.Lcarry_done:
    inc r10
    jmp .Louter
.Louter_end:
    lea rcx, [rsi + r13]
    mov rax, rcx
    test rcx, rcx
    jz .Lreturn_zero
    dec rcx
.Lscan:
    cmp qword ptr [r14 + rcx*8], 0
    jne .Lfound
    dec rcx
    jns .Lscan
.Lreturn_zero:
    xor eax, eax
.Ldone:
    pop r15
    pop r14
    pop r13
    pop r12
    ret
.Lfound:
    lea rax, [rcx + 1]
    jmp .Ldone
.Lzero_args:
    lea rcx, [rsi + r13]
    test rcx, rcx
    jz .Lreturn_zero
    mov rdi, r14
    xor eax, eax
    rep stosq
    jmp .Lreturn_zero
