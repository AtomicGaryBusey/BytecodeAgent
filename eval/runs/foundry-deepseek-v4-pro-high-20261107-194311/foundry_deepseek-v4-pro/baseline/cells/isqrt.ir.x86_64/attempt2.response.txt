export function l $isqrt(l %n) {
@start
    %low_slot =l alloc8 8
    %high_slot =l alloc8 8
    %ans_slot =l alloc8 8
    storel 0, %low_slot
    %high_init =l copy 4294967295
    storel %high_init, %high_slot
    storel 0, %ans_slot
    jmp @loop_cond
@loop_cond
    %low =l loadl %low_slot
    %high =l loadl %high_slot
    %cond =w cslel %low, %high
    jnz %cond, @loop_body, @end
@loop_body
    %sum =l add %low, %high
    %mid =l shr %sum, 1
    %mid_zero =w ceql %mid, 0
    jnz %mid_zero, @update_ans_low, @check_mid
@check_mid
    %quot =l divu %n, %mid
    %cmp =w cslel %mid, %quot
    jnz %cmp, @update_ans_low, @update_high
@update_ans_low
    storel %mid, %ans_slot
    %newlow =l add %mid, 1
    storel %newlow, %low_slot
    jmp @loop_cond
@update_high
    %newhigh =l sub %mid, 1
    storel %newhigh, %high_slot
    jmp @loop_cond
@end
    %ans =l loadl %ans_slot
    ret %ans
}
