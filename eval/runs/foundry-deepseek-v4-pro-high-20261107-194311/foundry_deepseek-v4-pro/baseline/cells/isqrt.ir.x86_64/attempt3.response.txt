export function l $isqrt(l %n) {
@start
    jmp @loop
@loop
    %low =l phi @start 0, @update_le %low_le, @update_gt %low_gt
    %high =l phi @start 4294967295, @update_le %high_le, @update_gt %high_gt
    %ans =l phi @start 0, @update_le %ans_le, @update_gt %ans_gt
    %cond =w culel %low, %high
    jnz %cond, @body, @done
@body
    %sum =l add %low, %high
    %mid =l shr %sum, 1
    %sq =l mul %mid, %mid
    %cmp =w culel %sq, %n
    jnz %cmp, @update_le, @update_gt
@update_le
    %low_le =l add %mid, 1
    %high_le =l copy %high
    %ans_le =l copy %mid
    jmp @loop
@update_gt
    %low_gt =l copy %low
    %high_gt =l sub %mid, 1
    %ans_gt =l copy %ans
    jmp @loop
@done
    ret %ans
}
