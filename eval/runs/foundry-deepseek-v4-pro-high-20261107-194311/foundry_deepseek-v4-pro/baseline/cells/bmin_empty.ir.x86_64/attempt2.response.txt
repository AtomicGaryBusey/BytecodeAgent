export function l $bmin_empty(l %a, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @ret_max, @process

@ret_max
    %allones =l copy -1
    %max =l shr %allones, 1
    ret %max

@process
    %first =l loadl %a
    %init_idx =l copy 1
    jmp @loop_begin

@loop_begin
    %idx =l phi @process %init_idx, @loop_next %next_idx
    %min =l phi @process %first, @loop_next %min_updated
    jmp @loop_cond

@loop_cond
    %cond =w csltl %idx, %n
    jnz %cond, @loop_body, @return_min

@loop_body
    %offset =l shl %idx, 3
    %el =l loadl %a, %offset
    %lt =w csltl %el, %min
    jnz %lt, @update_min, @skip_update

@update_min
    %newmin_candidate =l copy %el
    jmp @loop_next

@skip_update
    %keep_min =l copy %min
    jmp @loop_next

@loop_next
    %min_updated =l phi @update_min %newmin_candidate, @skip_update %keep_min
    %next_idx =l add %idx, 1
    jmp @loop_begin

@return_min
    ret %min
}
