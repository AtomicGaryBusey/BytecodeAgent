dot_i32_simd:
    xor eax, eax
    test rdx, rdx
    jz .Lret
    pxor xmm4, xmm4
    mov rcx, rdx
    shr rcx, 2
    jz .Lscalar_start
.Lvector_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    psrldq xmm0, 8
    psrldq xmm1, 8
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lvector_loop
.Lscalar_start:
    mov rcx, rdx
    and rcx, 3
    jz .Lreduce_sum
.Lscalar_loop:
    movsx r10, dword ptr [rdi]
    movsx r11, dword ptr [rsi]
    imul r10, r11
    add rax, r10
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Lscalar_loop
.Lreduce_sum:
    movq r10, xmm4
    add rax, r10
    pextrq r10, xmm4, 1
    add rax, r10
.Lret:
    ret
