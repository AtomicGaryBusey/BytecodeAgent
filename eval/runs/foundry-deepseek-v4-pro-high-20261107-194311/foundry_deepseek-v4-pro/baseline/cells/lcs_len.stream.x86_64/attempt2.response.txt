lcs_len:
    push r12
    push r13
    push r14
    push r15
    push rbx
    mov r12, r8
    mov r15, rcx
    mov r13, rdi
    mov r14, rdx
    mov rbx, rsi
    mov rcx, r15
    inc rcx
    xor eax, eax
    mov rdi, r12
    rep stosq
    test rbx, rbx
    jz .Ldone
.Louter:
    movzx r8d, byte ptr [r13]
    xor eax, eax
    lea rdi, [r12 + 8]
    mov rsi, r14
    lea rdx, [r12 + r15*8 + 8]
    cmp rdi, rdx
    jae .Linner_done
.Linner:
    mov r10, qword ptr [rdi]
    movzx r9d, byte ptr [rsi]
    cmp r8b, r9b
    jne .Lnot_equal
    lea rcx, [rax + 1]
    jmp .Lupdate
.Lnot_equal:
    mov rcx, qword ptr [rdi - 8]
    cmp r10, rcx
    cmovge rcx, r10
.Lupdate:
    mov qword ptr [rdi], rcx
    mov rax, r10
    add rdi, 8
    inc rsi
    cmp rdi, rdx
    jb .Linner
.Linner_done:
    inc r13
    dec rbx
    jnz .Louter
.Ldone:
    mov rax, qword ptr [r12 + r15*8]
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
