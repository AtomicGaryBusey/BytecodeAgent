lcs_len:
    mov r10, rcx
    add r10, 1
    mov r11, r8
    xor eax, eax
.Lzero_loop:
    mov qword ptr [r11], rax
    add r11, 8
    sub r10, 1
    jnz .Lzero_loop
    push rbx
    push r12
    push r14
    lea r12, [r8 + rcx*8 + 8]
    mov rbx, rsi
.Louter:
    test rbx, rbx
    jz .Ldone_pop
    dec rbx
    xor eax, eax
    mov r11, r8
    add r11, 8
    mov r9, rdx
    xor ecx, ecx
    cmp r11, r12
    jae .Louter_inc
.Linner:
    movzx r14d, byte ptr [r9]
    mov r10, qword ptr [r11]
    lea r13, [rax+1]
    mov rax, r10
    cmp rax, rcx
    cmovl rax, rcx
    cmp byte ptr [rdi], r14b
    cmove rax, r13
    mov qword ptr [r11], rax
    mov rcx, rax
    mov rax, r10
    add r11, 8
    add r9, 1
    cmp r11, r12
    jb .Linner
.Louter_inc:
    add rdi, 1
    jmp .Louter
.Ldone_pop:
    mov rax, qword ptr [r12 - 8]
    pop r14
    pop r12
    pop rbx
    ret
