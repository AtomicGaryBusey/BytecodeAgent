export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
        %init_min =l copy 9223372036854775807
        %init_max =l copy -9223372036854775808
        %init_sum =l copy 0
        %init_count =l copy 0
        %init_i =l copy 0
        jmp @cond
@cond
        %min =l phi @start %init_min, @odd_end %min1
        %max =l phi @start %init_max, @odd_end %max1
        %sum =l phi @start %init_sum, @odd_end %sum1
        %count =l phi @start %init_count, @odd_end %count1
        %i =l phi @start %init_i, @odd_end %i1
        %cmp =w csltl %i, %n
        jnz %cmp, @body, @done
@body
        %offset =l mul %i, 8
        %addr =l add %a, %offset
        %x =l loadl %addr
        %y =l call %f(l %x)
        %lt =w csltl %y, %min
        jnz %lt, @min_t, @min_f
@min_t
        jmp @min_end
@min_f
        jmp @min_end
@min_end
        %min1 =l phi @min_t %y, @min_f %min
        %gt =w csgtl %y, %max
        jnz %gt, @max_t, @max_f
@max_t
        jmp @max_end
@max_f
        jmp @max_end
@max_end
        %max1 =l phi @max_t %y, @max_f %max
        %sum1 =l add %sum, %y
        %oddl =l and %y, 1
        %isodd =w cnel %oddl, 0
        jnz %isodd, @odd_t, @odd_f
@odd_t
        %count_inc =l add %count, 1
        jmp @odd_end
@odd_f
        %count_same =l copy %count
        jmp @odd_end
@odd_end
        %count1 =l phi @odd_t %count_inc, @odd_f %count_same
        %i1 =l add %i, 1
        jmp @cond
@done
        storel %min, %out
        %addr2 =l add %out, 8
        storel %max, %addr2
        %addr3 =l add %out, 16
        storel %sum, %addr3
        ret %count
}
