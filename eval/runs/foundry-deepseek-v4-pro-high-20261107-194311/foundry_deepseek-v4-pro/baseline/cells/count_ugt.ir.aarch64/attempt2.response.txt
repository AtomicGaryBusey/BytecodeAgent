export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%ptr =l copy %a
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%val =l loadl %ptr
	%ugt =w cugtl %val, %t
	%inc =l extsw %ugt
	%cnt =l add %cnt, %inc
	%ptr =l add %ptr, 8
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
