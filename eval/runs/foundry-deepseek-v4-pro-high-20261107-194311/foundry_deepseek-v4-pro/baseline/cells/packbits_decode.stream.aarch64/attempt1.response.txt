packbits_decode:
  mov x9, x0                  // src pointer
  mov x10, x2                 // out pointer
  mov x11, x3                 // remaining capacity
  mov x7, xzr                 // total_written = 0
.Lloop:
  cbz x1, .Ldone              // no more source bytes -> done
  ldrsb x4, [x9]              // signed header byte
  add x9, x9, #1
  sub x1, x1, #1
  cmp x4, #0
  b.ge .Lliteral
  cmn x4, #128                // compare with -128 (NOP)
  b.eq .Lloop                 // NOP, continue
  // repeat packet
  neg x5, x4
  add x5, x5, #1              // length = 1 - h
  cmp x1, #1
  b.lt .Lunderrun_src         // not enough source for value byte
  ldrb w6, [x9]               // value byte
  add x9, x9, #1
  sub x1, x1, #1
  cmp x11, x5
  b.lt .Lrepeat_overflow      // output would exceed cap
  // normal repeat fill
  mov x13, x10
  mov x14, x5
.Lrepeat_fill_loop:
  cbz x14, .Lrepeat_fill_done
  strb w6, [x13], #1
  sub x14, x14, #1
  b .Lrepeat_fill_loop
.Lrepeat_fill_done:
  mov x10, x13
  sub x11, x11, x5
  add x7, x7, x5
  b .Lloop
.Lrepeat_overflow:
  mov x13, x10
  mov x14, x11
  cbz x14, .Lrepeat_overflow_done
.Lrepeat_overflow_loop:
  strb w6, [x13], #1
  sub x14, x14, #1
  cbnz x14, .Lrepeat_overflow_loop
.Lrepeat_overflow_done:
  add x7, x7, x11
  mov x0, #-1
  ret
.Lliteral:
  add x5, x4, #1              // length = h + 1
  cmp x1, x5
  b.lt .Lunderrun_src         // source underrun
  cmp x11, x5
  b.lt .Lliteral_overflow     // output would exceed cap
  // normal literal copy
  mov x13, x9
  mov x14, x10
  mov x15, x5
.Lliteral_copy_loop:
  cbz x15, .Lliteral_copy_done
  ldrb w16, [x13], #1
  strb w16, [x14], #1
  sub x15, x15, #1
  b .Lliteral_copy_loop
.Lliteral_copy_done:
  mov x9, x13
  mov x10, x14
  sub x1, x1, x5
  sub x11, x11, x5
  add x7, x7, x5
  b .Lloop
.Lliteral_overflow:
  mov x13, x9
  mov x14, x10
  mov x15, x11
  cbz x15, .Lliteral_overflow_done
.Lliteral_overflow_loop:
  ldrb w16, [x13], #1
  strb w16, [x14], #1
  sub x15, x15, #1
  cbnz x15, .Lliteral_overflow_loop
.Lliteral_overflow_done:
  add x7, x7, x11
  mov x0, #-1
  ret
.Lunderrun_src:
  mov x0, #-2
  ret
.Ldone:
  mov x0, x7
  ret
