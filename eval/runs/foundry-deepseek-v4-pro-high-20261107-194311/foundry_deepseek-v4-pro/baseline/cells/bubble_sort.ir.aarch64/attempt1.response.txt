function l $bubble_sort(l %a, l %n) {
@start
	%n1 =l sub %n, 1
	%cond =w csltl %n1, 1
	jnz %cond, @end, @outer_cond
@outer_cond
	%i =l phi @start 0, @outer_increment %i_next
	%limit =l sub %n, 1
	%i_lt_limit =w csltl %i, %limit
	jnz %i_lt_limit, @inner_cond, @end
@inner_cond
	%j =l phi @outer_cond 0, @inner_increment %j_next
	%inner_limit =l sub %n, 1
	%inner_limit =l sub %inner_limit, %i
	%j_lt_inner_limit =w csltl %j, %inner_limit
	jnz %j_lt_inner_limit, @inner_body, @outer_increment
@inner_body
	%j8 =l mul %j, 8
	%addr_j =l add %a, %j8
	%val_j =l loadl %addr_j
	%j1 =l add %j, 1
	%j18 =l mul %j1, 8
	%addr_j1 =l add %a, %j18
	%val_j1 =l loadl %addr_j1
	%need_swap =w csgtl %val_j, %val_j1
	jnz %need_swap, @swap, @inner_increment
@swap
	storel %val_j1, %addr_j
	storel %val_j, %addr_j1
	jmp @inner_increment
@inner_increment
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_increment
	%i_next =l add %i, 1
	jmp @outer_cond
@end
	ret 0
}
