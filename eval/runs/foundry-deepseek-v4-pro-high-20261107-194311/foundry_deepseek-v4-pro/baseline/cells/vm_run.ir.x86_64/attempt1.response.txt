export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pc_mem =l alloc8 8
    %sp_mem =l alloc8 8
    storel 0, %pc_mem
    storel 0, %sp_mem
    jmp @loop

@loop
    %pc =l loadl %pc_mem
    %sp =l loadl %sp_mem
    %done =w ceql %pc, %len
    jnz %done, @done_from_len, @cont
@done_from_len
    storel %sp, %sp_mem
    jmp @done

@cont
    %off_inst =l mul %pc, 8
    %ptr_inst =l add %code, %off_inst
    %inst =l loadl %ptr_inst

    %is0 =w ceql %inst, 0
    jnz %is0, @do_halt, @chk1
@do_halt
    storel %sp, %sp_mem
    jmp @done

@chk1
    %is1 =w ceql %inst, 1
    jnz %is1, @push, @chk2
@push
    %pc1 =l add %pc, 1
    %off_op =l mul %pc1, 8
    %ptr_op =l add %code, %off_op
    %val =l loadl %ptr_op
    %sp8 =l mul %sp, 8
    %dst =l add %stack, %sp8
    storel %val, %dst
    %new_sp =l add %sp, 1
    %new_pc =l add %pc, 2
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk2
    %is2 =w ceql %inst, 2
    jnz %is2, @add, @chk3
@add
    %sp_b =l sub %sp, 1
    %off_b =l mul %sp_b, 8
    %addr_b =l add %stack, %off_b
    %b =l loadl %addr_b
    %sp_a =l sub %sp_b, 1
    %off_a =l mul %sp_a, 8
    %addr_a =l add %stack, %off_a
    %a =l loadl %addr_a
    %sum =l add %a, %b
    %off_sum =l mul %sp_a, 8
    %addr_sum =l add %stack, %off_sum
    storel %sum, %addr_sum
    %new_sp =l add %sp_a, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk3
    %is3 =w ceql %inst, 3
    jnz %is3, @sub, @chk4
@sub
    %sp_b =l sub %sp, 1
    %off_b =l mul %sp_b, 8
    %addr_b =l add %stack, %off_b
    %b =l loadl %addr_b
    %sp_a =l sub %sp_b, 1
    %off_a =l mul %sp_a, 8
    %addr_a =l add %stack, %off_a
    %a =l loadl %addr_a
    %diff =l sub %a, %b
    %off_diff =l mul %sp_a, 8
    %addr_diff =l add %stack, %off_diff
    storel %diff, %addr_diff
    %new_sp =l add %sp_a, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk4
    %is4 =w ceql %inst, 4
    jnz %is4, @mul, @chk5
@mul
    %sp_b =l sub %sp, 1
    %off_b =l mul %sp_b, 8
    %addr_b =l add %stack, %off_b
    %b =l loadl %addr_b
    %sp_a =l sub %sp_b, 1
    %off_a =l mul %sp_a, 8
    %addr_a =l add %stack, %off_a
    %a =l loadl %addr_a
    %prod =l mul %a, %b
    %off_prod =l mul %sp_a, 8
    %addr_prod =l add %stack, %off_prod
    storel %prod, %addr_prod
    %new_sp =l add %sp_a, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk5
    %is5 =w ceql %inst, 5
    jnz %is5, @dup, @chk6
@dup
    %sp_minus1 =l sub %sp, 1
    %off_src =l mul %sp_minus1, 8
    %addr_src =l add %stack, %off_src
    %val =l loadl %addr_src
    %sp8_dst =l mul %sp, 8
    %addr_dst =l add %stack, %sp8_dst
    storel %val, %addr_dst
    %new_sp =l add %sp, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk6
    %is6 =w ceql %inst, 6
    jnz %is6, @swap, @chk7
@swap
    %sp_b =l sub %sp, 1
    %off_b =l mul %sp_b, 8
    %addr_b =l add %stack, %off_b
    %b =l loadl %addr_b
    %sp_a =l sub %sp_b, 1
    %off_a =l mul %sp_a, 8
    %addr_a =l add %stack, %off_a
    %a =l loadl %addr_a
    %off_push_b =l mul %sp_a, 8
    %addr_push_b =l add %stack, %off_push_b
    storel %b, %addr_push_b
    %sp_a_plus1 =l add %sp_a, 1
    %off_push_a =l mul %sp_a_plus1, 8
    %addr_push_a =l add %stack, %off_push_a
    storel %a, %addr_push_a
    %new_pc =l add %pc, 1
    storel %sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk7
    %is7 =w ceql %inst, 7
    jnz %is7, @drop, @chk8
@drop
    %new_sp =l sub %sp, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk8
    %is8 =w ceql %inst, 8
    jnz %is8, @over, @chk9
@over
    %sp_minus2 =l sub %sp, 2
    %off_src =l mul %sp_minus2, 8
    %addr_src =l add %stack, %off_src
    %val =l loadl %addr_src
    %sp8_dst =l mul %sp, 8
    %addr_dst =l add %stack, %sp8_dst
    storel %val, %addr_dst
    %new_sp =l add %sp, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk9
    %is9 =w ceql %inst, 9
    jnz %is9, @lt, @chk10
@lt
    %sp_b =l sub %sp, 1
    %off_b =l mul %sp_b, 8
    %addr_b =l add %stack, %off_b
    %b =l loadl %addr_b
    %sp_a =l sub %sp_b, 1
    %off_a =l mul %sp_a, 8
    %addr_a =l add %stack, %off_a
    %a =l loadl %addr_a
    %condw =w csltl %a, %b
    %cond =l extsw %condw
    %off_cond =l mul %sp_a, 8
    %addr_cond =l add %stack, %off_cond
    storel %cond, %addr_cond
    %new_sp =l add %sp_a, 1
    %new_pc =l add %pc, 1
    storel %new_sp, %sp_mem
    storel %new_pc, %pc_mem
    jmp @loop

@chk10
    %is10 =w ceql %inst, 10
    jnz %is10, @jz, @chk11
@jz
    %new_sp =l sub %sp, 1
    %off_v =l mul %new_sp, 8
    %addr_v =l add %stack, %off_v
    %v =l loadl %addr_v
    %pc1 =l add %pc, 1
    %off_t =l mul %pc1, 8
    %ptr_t =l add %code, %off_t
    %t =l loadl %ptr_t
    %is_zero =w ceql %v, 0
    jnz %is_zero, @jz_take, @jz_nottake
@jz_take
    storel %t, %pc_mem
    storel %new_sp, %sp_mem
    jmp @loop
@jz_nottake
    %new_pc =l add %pc, 2
    storel %new_pc, %pc_mem
    storel %new_sp, %sp_mem
    jmp @loop

@chk11
    %is11 =w ceql %inst, 11
    jnz %is11, @jmp, @next
@jmp
    %pc1 =l add %pc, 1
    %off =l mul %pc1, 8
    %ptr =l add %code, %off
    %tgt =l loadl %ptr
    storel %tgt, %pc_mem
    storel %sp, %sp_mem
    jmp @loop

@next
    %new_pc =l add %pc, 1
    storel %new_pc, %pc_mem
    storel %sp, %sp_mem
    jmp @loop

@done
    %sp_final =l loadl %sp_mem
    %i =l copy %sp_final
    jmp @fill_cond
@fill_loop
    %i8 =l mul %i, 8
    %addr =l add %stack, %i8
    storel 0, %addr
    %i =l add %i, 1
@fill_cond
    %still =w csltl %i, %cap
    jnz %still, @fill_loop, @fill_done
@fill_done
    %has_val =w csgtl %sp_final, 0
    jnz %has_val, @ret_top, @ret_zero
@ret_top
    %top_idx =l sub %sp_final, 1
    %off_top =l mul %top_idx, 8
    %addr_top =l add %stack, %off_top
    %ret_val =l loadl %addr_top
    ret %ret_val
@ret_zero
    ret 0
}
