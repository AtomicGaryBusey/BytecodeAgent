export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %nzero =w ceql %n, 0
    jnz %nzero, @done, @eqcheck
@eqcheck
    %eq =w ceql %dst, %src
    jnz %eq, @done, @direction
@direction
    %fwd =w cultl %dst, %src
    jnz %fwd, @forward, @backward
@forward
    %i0 =l copy 0
    jmp @fw_loop
@fw_loop
    %i =l phi @forward %i0, @fw_body %inext
    %loopcond =w csltl %i, %n
    jnz %loopcond, @fw_body, @done
@fw_body
    %offset =l mul %i, 8
    %sptr =l add %src, %offset
    %dptr =l add %dst, %offset
    %val =l loadl %sptr
    storel %val, %dptr
    %inext =l add %i, 1
    jmp @fw_loop
@backward
    %imax =l sub %n, 1
    jmp @bw_loop
@bw_loop
    %i =l phi @backward %imax, @bw_body %inext
    %loopcond =w csgel %i, 0
    jnz %loopcond, @bw_body, @done
@bw_body
    %offset =l mul %i, 8
    %sptr =l add %src, %offset
    %dptr =l add %dst, %offset
    %val =l loadl %sptr
    storel %val, %dptr
    %inext =l sub %i, 1
    jmp @bw_loop
@done
    ret 0
}
