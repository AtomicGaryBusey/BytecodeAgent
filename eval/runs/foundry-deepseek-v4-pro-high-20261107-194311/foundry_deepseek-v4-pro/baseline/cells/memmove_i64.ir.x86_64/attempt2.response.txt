export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %nzero =w ceql %n, 0
    jnz %nzero, @done, @eqcheck
@eqcheck
    %eq =w ceql %dst, %src
    jnz %eq, @done, @direction
@direction
    %fwd =w cultl %dst, %src
    jnz %fwd, @forward, @backward
@forward
    %fw_i0 =l copy 0
    jmp @fw_loop
@fw_loop
    %fw_i =l phi @forward %fw_i0, @fw_body %fw_inext
    %fw_cond =w csltl %fw_i, %n
    jnz %fw_cond, @fw_body, @done
@fw_body
    %fw_offset =l mul %fw_i, 8
    %fw_sptr =l add %src, %fw_offset
    %fw_dptr =l add %dst, %fw_offset
    %fw_val =l loadl %fw_sptr
    storel %fw_val, %fw_dptr
    %fw_inext =l add %fw_i, 1
    jmp @fw_loop
@backward
    %bw_i0 =l sub %n, 1
    jmp @bw_loop
@bw_loop
    %bw_i =l phi @backward %bw_i0, @bw_body %bw_inext
    %bw_cond =w csgel %bw_i, 0
    jnz %bw_cond, @bw_body, @done
@bw_body
    %bw_offset =l mul %bw_i, 8
    %bw_sptr =l add %src, %bw_offset
    %bw_dptr =l add %dst, %bw_offset
    %bw_val =l loadl %bw_sptr
    storel %bw_val, %bw_dptr
    %bw_inext =l sub %bw_i, 1
    jmp @bw_loop
@done
    ret 0
}
