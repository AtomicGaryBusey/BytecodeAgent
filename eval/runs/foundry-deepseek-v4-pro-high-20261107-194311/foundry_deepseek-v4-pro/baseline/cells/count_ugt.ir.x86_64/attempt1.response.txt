export function l $count_ugt(l %a, l %n, l %t) {
@start
	%count =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%val =l loadl %addr
	%gt =w cultl %t, %val
	%inc =l extsw %gt
	%count =l add %count, %inc
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
