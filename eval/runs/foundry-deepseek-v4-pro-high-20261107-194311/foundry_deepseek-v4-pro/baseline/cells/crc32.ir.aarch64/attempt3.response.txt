export function l $crc32(l %p, l %n) {
@entry
    %crc_start =w copy 0xffffffff
    %poly =w copy 0xedb88320
    %i_start =l copy 0
    %p_start =l copy %p
    jmp @loop_check

@loop_check
    %crc =w phi @entry %crc_start, @loop_body %crc8
    %i =l phi @entry %i_start, @loop_body %i_next
    %p_ptr =l phi @entry %p_start, @loop_body %p_next
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done

@loop_body
    %byte =w loadub %p_ptr
    %crc0 =w xor %crc, %byte
    %lsb0 =w and %crc0, 1
    %mask0 =w neg %lsb0
    %xor0 =w and %mask0, %poly
    %crc_sh0 =w shr %crc0, 1
    %crc1 =w xor %crc_sh0, %xor0
    %lsb1 =w and %crc1, 1
    %mask1 =w neg %lsb1
    %xor1 =w and %mask1, %poly
    %crc_sh1 =w shr %crc1, 1
    %crc2 =w xor %crc_sh1, %xor1
    %lsb2 =w and %crc2, 1
    %mask2 =w neg %lsb2
    %xor2 =w and %mask2, %poly
    %crc_sh2 =w shr %crc2, 1
    %crc3 =w xor %crc_sh2, %xor2
    %lsb3 =w and %crc3, 1
    %mask3 =w neg %lsb3
    %xor3 =w and %mask3, %poly
    %crc_sh3 =w shr %crc3, 1
    %crc4 =w xor %crc_sh3, %xor3
    %lsb4 =w and %crc4, 1
    %mask4 =w neg %lsb4
    %xor4 =w and %mask4, %poly
    %crc_sh4 =w shr %crc4, 1
    %crc5 =w xor %crc_sh4, %xor4
    %lsb5 =w and %crc5, 1
    %mask5 =w neg %lsb5
    %xor5 =w and %mask5, %poly
    %crc_sh5 =w shr %crc5, 1
    %crc6 =w xor %crc_sh5, %xor5
    %lsb6 =w and %crc6, 1
    %mask6 =w neg %lsb6
    %xor6 =w and %mask6, %poly
    %crc_sh6 =w shr %crc6, 1
    %crc7 =w xor %crc_sh6, %xor6
    %lsb7 =w and %crc7, 1
    %mask7 =w neg %lsb7
    %xor7 =w and %mask7, %poly
    %crc_sh7 =w shr %crc7, 1
    %crc8 =w xor %crc_sh7, %xor7
    %p_next =l add %p_ptr, 1
    %i_next =l add %i, 1
    jmp @loop_check

@done
    %crc_final =w xor %crc, 0xffffffff
    %result =l extuw %crc_final
    ret %result
}
