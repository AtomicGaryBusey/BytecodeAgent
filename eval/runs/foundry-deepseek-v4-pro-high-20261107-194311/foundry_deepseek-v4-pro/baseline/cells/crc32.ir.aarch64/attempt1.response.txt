export function l $crc32(l %p, l %n) {
@entry
    %crc_start =w copy 0xffffffff
    %i_start =l copy 0
    %p_start =l copy %p
    jmp @loop_check

@loop_check
    %crc =w phi @entry %crc_start, @loop_body %crc_next_out
    %i =l phi @entry %i_start, @loop_body %i_next_out
    %p_ptr =l phi @entry %p_start, @loop_body %p_next_out
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done

@loop_body
    %byte =w loadub %p_ptr
    %crc_xor =w xor %crc, %byte
    %lsb1 =w and %crc_xor, 1
    %poly =w copy 0xedb88320
    %mask1 =w mul %lsb1, %poly
    %crc_sh1 =w shr %crc_xor, 1
    %crc1 =w xor %crc_sh1, %mask1
    %lsb2 =w and %crc1, 1
    %mask2 =w mul %lsb2, %poly
    %crc_sh2 =w shr %crc1, 1
    %crc2 =w xor %crc_sh2, %mask2
    %lsb3 =w and %crc2, 1
    %mask3 =w mul %lsb3, %poly
    %crc_sh3 =w shr %crc2, 1
    %crc3 =w xor %crc_sh3, %mask3
    %lsb4 =w and %crc3, 1
    %mask4 =w mul %lsb4, %poly
    %crc_sh4 =w shr %crc3, 1
    %crc4 =w xor %crc_sh4, %mask4
    %lsb5 =w and %crc4, 1
    %mask5 =w mul %lsb5, %poly
    %crc_sh5 =w shr %crc4, 1
    %crc5 =w xor %crc_sh5, %mask5
    %lsb6 =w and %crc5, 1
    %mask6 =w mul %lsb6, %poly
    %crc_sh6 =w shr %crc5, 1
    %crc6 =w xor %crc_sh6, %mask6
    %lsb7 =w and %crc6, 1
    %mask7 =w mul %lsb7, %poly
    %crc_sh7 =w shr %crc6, 1
    %crc7 =w xor %crc_sh7, %mask7
    %lsb8 =w and %crc7, 1
    %mask8 =w mul %lsb8, %poly
    %crc_sh8 =w shr %crc7, 1
    %crc_next_out =w xor %crc_sh8, %mask8
    %p_next_out =l add %p_ptr, 1
    %i_next_out =l add %i, 1
    jmp @loop_check

@done
    %crc_final =w xor %crc, 0xffffffff
    %result =l extuw %crc_final
    ret %result
}
