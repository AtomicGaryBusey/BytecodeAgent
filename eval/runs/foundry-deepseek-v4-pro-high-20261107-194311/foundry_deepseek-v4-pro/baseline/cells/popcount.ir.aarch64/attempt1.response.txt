export function l $popcount(l %x) {
@start
    %m1 =l copy 0x5555555555555555
    %shr1 =l shr %x, 1
    %and1 =l and %shr1, %m1
    %and2 =l and %x, %m1
    %val1 =l add %and1, %and2

    %m2 =l copy 0x3333333333333333
    %shr2 =l shr %val1, 2
    %and3 =l and %shr2, %m2
    %and4 =l and %val1, %m2
    %val2 =l add %and3, %and4

    %m3 =l copy 0x0F0F0F0F0F0F0F0F
    %shr3 =l shr %val2, 4
    %and5 =l and %shr3, %m3
    %and6 =l and %val2, %m3
    %val3 =l add %and5, %and6

    %m4 =l copy 0x0101010101010101
    %prod =l mul %val3, %m4
    %res =l shr %prod, 56
    ret %res
}
