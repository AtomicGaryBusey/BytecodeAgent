export function l $shift_sat(l %x, l %n) {
@start
	%big_cond =w cugel %n, 64
	jnz %big_cond, @big, @small
@small
	%left_small =l shl %x, %n
	%right_small =l sar %x, %n
	jmp @merge
@big
	%left_big =l copy 0
	%right_big =l sar %x, 63
	jmp @merge
@merge
	%left =l phi @small %left_small, @big %left_big
	%right =l phi @small %right_small, @big %right_big
	%sum =l add %left, %right
	ret %sum
}
