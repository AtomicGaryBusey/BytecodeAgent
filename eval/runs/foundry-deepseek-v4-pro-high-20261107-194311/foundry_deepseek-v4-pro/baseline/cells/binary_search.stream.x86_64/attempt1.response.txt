binary_search:
    test rsi, rsi
    jle .Lnot_found
    mov rcx, rsi
    dec rcx
    xor esi, esi
.Lloop:
    cmp rsi, rcx
    jg .Lnot_found
    mov rax, rcx
    sub rax, rsi
    sar rax, 1
    add rax, rsi
    mov r8, qword ptr [rdi + rax*8]
    cmp r8, rdx
    je .Lfound
    jl .Lgo_high
    lea rcx, [rax - 1]
    jmp .Lloop
.Lgo_high:
    lea rsi, [rax + 1]
    jmp .Lloop
.Lfound:
    ret
.Lnot_found:
    mov rax, -1
    ret
