export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %i =l copy 0
    %ptr =l copy %a
    jmp @loop
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @notfound
@body
    %val =l loadl %ptr
    %iszero =w ceql %val, 0
    jnz %iszero, @found, @cont
@cont
    %i =l add %i, 1
    %ptr =l add %ptr, 8
    jmp @loop
@found
    ret %i
@notfound
    ret %n
}
