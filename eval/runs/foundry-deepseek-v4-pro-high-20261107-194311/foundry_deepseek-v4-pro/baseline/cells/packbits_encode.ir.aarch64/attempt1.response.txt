export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %outpos =l copy 0
    jmp @outer_loop
@outer_loop
    %cmp_i_n =w csltl %i, %n
    jnz %cmp_i_n, @outer_body, @done
@outer_body
    %base_byte_i =w loadub %src, %i
    %runlen_i =l copy 1
    %run_idx_i =l add %i, 1
    jmp @run_loop_i
@run_loop_i
    %chk_idx_i =w csltl %run_idx_i, %n
    jnz %chk_idx_i, @run_cmp_byte_i, @end_run_i
@run_cmp_byte_i
    %cur_byte_i =w loadub %src, %run_idx_i
    %eq_i =w ceqw %base_byte_i, %cur_byte_i
    %run_len_ok_i =w csltl %runlen_i, 128
    %all_ok_i =w and %eq_i, %run_len_ok_i
    jnz %all_ok_i, @inc_run_i, @end_run_i
@inc_run_i
    %runlen_i =l add %runlen_i, 1
    %run_idx_i =l add %run_idx_i, 1
    jmp @run_loop_i
@end_run_i
    %r_i =l copy %runlen_i
    %repeat_cmp =w csgel %r_i, 3
    jnz %repeat_cmp, @repeat, @literal
@repeat
    %repeat_need =l add %outpos, 2
    %repeat_cap_ok =w cslel %repeat_need, %cap
    jnz %repeat_cap_ok, @do_repeat, @fail
@do_repeat
    %r_i_w =w copy %r_i
    %hdr_w =w sub 257, %r_i_w
    %out_ptr =l add %out, %outpos
    storeb %hdr_w, %out_ptr
    %b_w =w loadub %src, %i
    %out_ptr2 =l add %out_ptr, 1
    storeb %b_w, %out_ptr2
    %outpos =l add %outpos, 2
    %i =l add %i, %r_i
    jmp @outer_loop
@literal
    %j =l copy %i
    jmp @literal_loop
@literal_loop
    %lit_n_cmp =w csltl %j, %n
    jnz %lit_n_cmp, @check_len, @lit_end
@check_len
    %len_so_far =l sub %j, %i
    %len_ok =w csltl %len_so_far, 128
    jnz %len_ok, @check_run_at_j, @lit_end
@check_run_at_j
    %base_byte_j =w loadub %src, %j
    %runlen_j =l copy 1
    %run_idx_j =l add %j, 1
    jmp @run_loop_j
@run_loop_j
    %chk_idx_j =w csltl %run_idx_j, %n
    jnz %chk_idx_j, @run_cmp_byte_j, @end_run_j
@run_cmp_byte_j
    %cur_byte_j =w loadub %src, %run_idx_j
    %eq_j =w ceqw %base_byte_j, %cur_byte_j
    %run_len_ok_j =w csltl %runlen_j, 128
    %all_ok_j =w and %eq_j, %run_len_ok_j
    jnz %all_ok_j, @inc_run_j, @end_run_j
@inc_run_j
    %runlen_j =l add %runlen_j, 1
    %run_idx_j =l add %run_idx_j, 1
    jmp @run_loop_j
@end_run_j
    %r_j =l copy %runlen_j
    %r_j_w =w copy %r_j
    %is_run_start =w csltw %r_j_w, 3
    jnz %is_run_start, @inc_j, @lit_end
@inc_j
    %j =l add %j, 1
    jmp @literal_loop
@lit_end
    %L =l sub %j, %i
    %need_lit =l add %outpos, 1
    %need_lit =l add %need_lit, %L
    %lit_cap_ok =w cslel %need_lit, %cap
    jnz %lit_cap_ok, @emit_literal, @fail
@emit_literal
    %L_w =w copy %L
    %hdr_lit =w sub %L_w, 1
    %out_hdr =l add %out, %outpos
    storeb %hdr_lit, %out_hdr
    %lit_idx =l copy 0
    jmp @literal_copy_loop
@literal_copy_loop
    %cpy_cmp =w csltl %lit_idx, %L
    jnz %cpy_cmp, @literalcpy_body, @lit_copy_done
@literalcpy_body
    %src_off =l add %i, %lit_idx
    %b_cpy =w loadub %src, %src_off
    %out_off =l add %outpos, 1
    %out_off =l add %out_off, %lit_idx
    %out_ptr_cpy =l add %out, %out_off
    storeb %b_cpy, %out_ptr_cpy
    %lit_idx =l add %lit_idx, 1
    jmp @literal_copy_loop
@lit_copy_done
    %outpos =l add %outpos, %L
    %outpos =l add %outpos, 1
    %i =l add %i, %L
    jmp @outer_loop
@fail
    %ret =l copy -1
    ret %ret
@done
    ret %outpos
}
