export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	%max_ptr =l alloc8 1
	%i_ptr =l alloc8 1
	storel %first, %max_ptr
	%one =l copy 1
	storel %one, %i_ptr
	jmp @loop
@loop
	%i =l loadl %i_ptr
	%cond =w cultl %i, %n
	jnz %cond, @body, @end
@body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%val =l loadl %addr
	%cur_max =l loadl %max_ptr
	%gt =w csgtl %val, %cur_max
	jnz %gt, @update, @after
@update
	storel %val, %max_ptr
	jmp @after
@after
	%next_i =l add %i, 1
	storel %next_i, %i_ptr
	jmp @loop
@end
	%result =l loadl %max_ptr
	ret %result
}
