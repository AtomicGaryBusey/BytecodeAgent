sdivrem:
    push r12
    mov r12, rdx
    test rsi, rsi
    jz .L_b_zero
    mov rax, rdi
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne .L_normal
    cmp rsi, -1
    jne .L_normal
    mov qword ptr [r12], 0
    mov rax, rcx
    pop r12
    ret
.L_b_zero:
    mov qword ptr [r12], rdi
    xor eax, eax
    pop r12
    ret
.L_normal:
    cqo
    idiv rsi
    mov [r12], rdx
    pop r12
    ret
