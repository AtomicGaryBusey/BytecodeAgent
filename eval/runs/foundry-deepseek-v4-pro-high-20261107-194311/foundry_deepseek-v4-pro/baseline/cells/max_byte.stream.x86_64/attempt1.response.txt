max_byte:
   test rsi, rsi
   jz .Lzero
   pxor xmm0, xmm0
   mov rcx, rsi
.Lloop16:
   cmp rcx, 16
   jb .Lremainder
   movdqu xmm1, xmmword ptr [rdi]
   pmaxub xmm0, xmm1
   add rdi, 16
   sub rcx, 16
   jmp .Lloop16
.Lremainder:
   movdqa xmm1, xmm0
   psrldq xmm1, 8
   pmaxub xmm0, xmm1
   movdqa xmm1, xmm0
   psrldq xmm1, 4
   pmaxub xmm0, xmm1
   movdqa xmm1, xmm0
   psrldq xmm1, 2
   pmaxub xmm0, xmm1
   movdqa xmm1, xmm0
   psrldq xmm1, 1
   pmaxub xmm0, xmm1
   movd eax, xmm0
   and eax, 0xFF
   test rcx, rcx
   jz .Ldone
.Lscalar_loop:
   movzx edx, byte ptr [rdi]
   cmp dl, al
   jbe .Lnext
   movzx eax, dl
.Lnext:
   inc rdi
   dec rcx
   jnz .Lscalar_loop
.Ldone:
   ret
.Lzero:
   xor eax, eax
   ret
