export function l $count_desc(l %a, l %n) {
@start
    %cond =w csltl %n, 2
    jnz %cond, @ret0, @preloop
@ret0
    ret 0
@preloop
    %cnt =l copy 0
    %i =l copy 0
    %limit =l sub %n, 1
    jmp @loopcond
@loopcond
    %c =w csltl %i, %limit
    jnz %c, @loopbody, @return
@loopbody
    %off_i =l mul %i, 8
    %addr0 =l add %a, %off_i
    %val0 =l loadl %addr0
    %next =l add %i, 1
    %off_n =l mul %next, 8
    %addr1 =l add %a, %off_n
    %val1 =l loadl %addr1
    %gt =w csgtl %val0, %val1
    %inc =l extuw %gt
    %cnt =l add %cnt, %inc
    %i =l add %i, 1
    jmp @loopcond
@return
    ret %cnt
}
