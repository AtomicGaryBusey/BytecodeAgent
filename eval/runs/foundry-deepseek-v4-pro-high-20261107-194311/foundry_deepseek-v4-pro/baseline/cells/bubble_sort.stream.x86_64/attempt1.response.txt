bubble_sort:
    cmp rsi, 1
    jle .Lreturn_zero
    lea rcx, [rsi - 1]
.Louter:
    xor r8d, r8d
    xor edx, edx
.Linner:
    cmp rdx, rcx
    jge .Lcheck_swapped
    mov rax, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r9
    jle .Lno_swap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], rax
    mov r8d, 1
.Lno_swap:
    inc rdx
    jmp .Linner
.Lcheck_swapped:
    test r8d, r8d
    jz .Lreturn_zero
    dec rcx
    jnz .Louter
.Lreturn_zero:
    xor eax, eax
    ret
