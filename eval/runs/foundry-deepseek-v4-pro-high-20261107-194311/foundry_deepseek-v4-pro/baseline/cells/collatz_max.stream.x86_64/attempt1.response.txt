collatz_max:
    push r12
    push r13
    push r14
    mov r12, 0
    mov r13, 1
.Lloop:
    cmp r13, rdi
    jg .Ldone
    mov rcx, r13
    xor r14, r14
    cmp rcx, 1
    je .Lnext_k
.Linner:
    test cl, 1
    jz .Leven
    lea rcx, [rcx + rcx*2 + 1]
    jmp .Lstep
.Leven:
    shr rcx, 1
.Lstep:
    inc r14
    cmp rcx, 1
    jne .Linner
    cmp r14, r12
    cmovg r12, r14
.Lnext_k:
    inc r13
    jmp .Lloop
.Ldone:
    mov rax, r12
    pop r14
    pop r13
    pop r12
    ret
