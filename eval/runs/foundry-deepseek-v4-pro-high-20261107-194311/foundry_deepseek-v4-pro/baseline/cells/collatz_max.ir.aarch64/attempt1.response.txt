export function l $collatz_max(l %n) {
@entry
	%max_start =l copy 0
	%k_start =l copy 1
	jmp @kloop
@kloop
	%max =l phi @entry %max_start, @update_max_merge %max_new
	%k =l phi @entry %k_start, @update_max_merge %k_next
	%k_cond =w cslel %k, %n
	jnz %k_cond, @body, @done
@body
	%x_init =l copy %k
	%steps_init =l copy 0
	jmp @collatz_loop
@collatz_loop
	%x =l phi @body %x_init, @even %x_even, @odd %x_odd
	%steps =l phi @body %steps_init, @even %steps_even, @odd %steps_odd
	%is_one =w ceql %x, 1
	jnz %is_one, @update_max, @collatz_step
@collatz_step
	%rem =l rem %x, 2
	%even_cond =w ceql %rem, 0
	jnz %even_cond, @even, @odd
@even
	%x_even =l div %x, 2
	%steps_even =l add %steps, 1
	jmp @collatz_loop
@odd
	%three_x =l mul %x, 3
	%x_odd =l add %three_x, 1
	%steps_odd =l add %steps, 1
	jmp @collatz_loop
@update_max
	%gt =w csltl %max, %steps
	jnz %gt, @max_from_steps, @max_keep
@max_from_steps
	jmp @update_max_merge
@max_keep
	jmp @update_max_merge
@update_max_merge
	%max_new =l phi @max_from_steps %steps, @max_keep %max
	%k_next =l add %k, 1
	jmp @kloop
@done
	ret %max
}
