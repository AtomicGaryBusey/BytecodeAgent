packbits_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r15, rdx
    mov r14, rcx
    xor edi, edi
    mov r8, r15
    lea r9, [r15 + r14]
.Lloop:
    cmp rdi, r13
    jge .Ldone
    lea rdx, [r12 + rdi]
    movzx ecx, byte ptr [rdx]
    mov r10, r13
    sub r10, rdi
    mov eax, 128
    cmp r10, rax
    cmovg r10, rax
    mov r11, 1
.Lrun_loop:
    cmp r11, r10
    je .Lrun_done
    cmp byte ptr [rdx + r11], cl
    jne .Lrun_done
    inc r11
    jmp .Lrun_loop
.Lrun_done:
    cmp r11, 3
    jl .Lliteral
    lea rax, [r8 + 2]
    cmp rax, r9
    ja .Loverflow
    lea eax, [r11 - 1]
    neg al
    mov byte ptr [r8], al
    mov byte ptr [r8 + 1], cl
    add r8, 2
    add rdi, r11
    jmp .Lloop
.Lliteral:
    mov rcx, rdi
    lea rsi, [rdi + 128]
    cmp rsi, r13
    cmovg rsi, r13
    jmp .Lfind_lit_check
.Lfind_lit_check:
    cmp rcx, rsi
    jge .Llit_done_search
    lea r10, [r12 + rcx]
    movzx r11d, byte ptr [r10]
    mov rdx, r13
    sub rdx, rcx
    mov eax, 128
    cmp rdx, rax
    cmovg rdx, rax
    mov rax, 1
.Lrun_lit_loop_inner:
    cmp rax, rdx
    je .Lrun_lit_done_inner
    cmp byte ptr [r10 + rax], r11b
    jne .Lrun_lit_done_inner
    inc rax
    jmp .Lrun_lit_loop_inner
.Lrun_lit_done_inner:
    cmp rax, 3
    jge .Llit_done_search
    inc rcx
    jmp .Lfind_lit_check
.Llit_done_search:
    mov rax, rcx
    sub rax, rdi
    lea r11, [r8 + rax + 1]
    cmp r11, r9
    ja .Loverflow
    lea edx, [rax - 1]
    mov byte ptr [r8], dl
    mov rdx, rcx
    push rdi
    push rsi
    cld
    lea rsi, [r12 + rdi]
    lea rdi, [r8 + 1]
    mov rcx, rax
    rep movsb
    pop rsi
    pop rdi
    add r8, rax
    inc r8
    mov rdi, rdx
    jmp .Lloop
.Loverflow:
    mov rax, -1
    jmp .Lreturn
.Ldone:
    mov rax, r8
    sub rax, r15
.Lreturn:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
