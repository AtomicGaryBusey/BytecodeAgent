export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %written =l copy 0
    %out_ptr =l copy %out
    %one =l copy 1
    %two =l copy 2
    %c127 =l copy 127
    %c128 =l copy 128
    %c257 =l copy 257
    jmp @loop

@loop
    %cond_i =l cslt %i, %n
    jnz %cond_i, @process, @done

@process
    %src_i =l add %src, %i
    %b0 =w loadub %src_i
    %k =l copy 0
    jmp @run_loop

@run_loop
    %k_plus1 =l add %k, %one
    %cond_k =l cslt %k, %c127
    %i_k1 =l add %i, %k_plus1
    %cond_in =l cslt %i_k1, %n
    %both =l and %cond_k, %cond_in
    jnz %both, @run_check, @run_done

@run_check
    %addr =l add %src, %i_k1
    %bnext =w loadub %addr
    %eq =w ceqw %bnext, %b0
    jnz %eq, @run_inc, @run_done

@run_inc
    %k =l add %k, %one
    jmp @run_loop

@run_done
    %run_len =l add %k, %one
    %run_len_ge3 =l csge %run_len, 3
    jnz %run_len_ge3, @emit_repeat, @literal_start

@emit_repeat
    %new_written_r =l add %written, %two
    %cap_exceed_r =l csgt %new_written_r, %cap
    jnz %cap_exceed_r, @return_error, @write_repeat

@write_repeat
    %hdr_temp =l copy %c257
    %hdr_val =l sub %hdr_temp, %run_len
    %hdr_w =w copy %hdr_val
    storeb %hdr_w, %out_ptr
    %out_ptr2 =l add %out_ptr, %one
    storeb %b0, %out_ptr2
    %i_next =l add %i, %run_len
    %out_ptr_next =l add %out_ptr, %two
    %written_next =l add %written, %two
    jmp @update_vars

@literal_start
    %j =l copy %i
    jmp @literal_loop

@literal_loop
    %cond_j_n =l cslt %j, %n
    jnz %cond_j_n, @check_len_128, @literal_end

@check_len_128
    %len_so_far =l sub %j, %i
    %len_lt128 =l cslt %len_so_far, %c128
    jnz %len_lt128, @check_run_j, @literal_end

@check_run_j
    %j_plus2 =l add %j, %two
    %cond_j2_n =l cslt %j_plus2, %n
    jnz %cond_j2_n, @run3_check_bytes, @run3_false

@run3_check_bytes
    %addr_j0 =l add %src, %j
    %b_j0 =w loadub %addr_j0
    %addr_j1 =l add %src, %one
    %addr_j1_full =l add %addr_j1, %j
    %b_j1 =w loadub %addr_j1_full
    %eq01 =w ceqw %b_j0, %b_j1
    jnz %eq01, @run3_check23, @run3_false

@run3_check23
    %addr_j2_full =l add %addr_j1_full, %one
    %b_j2 =w loadub %addr_j2_full
    %eq12 =w ceqw %b_j1, %b_j2
    jnz %eq12, @run3_true, @run3_false

@run3_true
    jmp @literal_end

@run3_false
    %j =l add %j, %one
    jmp @literal_loop

@literal_end
    %L =l sub %j, %i
    %pack_size_lit =l add %L, %one
    %new_written_l =l add %written, %pack_size_lit
    %cap_exceed_l =l csgt %new_written_l, %cap
    jnz %cap_exceed_l, @return_error, @write_literal

@write_literal
    %lit_hdr_val =l sub %L, %one
    %lit_hdr_w =w copy %lit_hdr_val
    storeb %lit_hdr_w, %out_ptr
    %c =l copy 0
    jmp @copy_loop

@copy_loop
    %c_lt_L =l cslt %c, %L
    jnz %c_lt_L, @copy_one, @copy_done

@copy_one
    %src_off =l add %i, %c
    %src_addr =l add %src, %src_off
    %byte =w loadub %src_addr
    %dst_base =l add %out_ptr, %one
    %dst_addr =l add %dst_base, %c
    storeb %byte, %dst_addr
    %c =l add %c, %one
    jmp @copy_loop

@copy_done
    %i_next =l add %i, %L
    %out_ptr_next =l add %out_ptr, %pack_size_lit
    %written_next =l add %written, %pack_size_lit
    jmp @update_vars

@update_vars
    %i =l copy %i_next
    %out_ptr =l copy %out_ptr_next
    %written =l copy %written_next
    jmp @loop

@return_error
    %neg1 =l copy -1
    ret %neg1

@done
    ret %written
}
