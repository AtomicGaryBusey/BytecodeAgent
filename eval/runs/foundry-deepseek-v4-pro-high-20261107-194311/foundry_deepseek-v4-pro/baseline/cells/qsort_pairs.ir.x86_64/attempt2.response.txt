export function l $qsort_pairs(l %a0, l %n0) {
@start
    %c =w csgtl %n0, 1
    jnz %c, @loop, @done
@loop
    %a_curr =l phi @start %a0, @cont_left %a_for_loop, @cont_right %a_for_loop2
    %n_curr =l phi @start %n0, @cont_left %n_for_loop, @cont_right %n_for_loop2
    %nc =w csgtl %n_curr, 1
    jnz %nc, @partition, @done
@partition
    %pivot_idx =l div %n_curr, 2
    %piv_offset =l mul %pivot_idx, 16
    %piv_ptr =l add %a_curr, %piv_offset
    %piv_key_tmp =l loadl %piv_ptr
    %piv_val_tmp =l loadl %piv_ptr, 8
    %last_idx =l sub %n_curr, 1
    %last_offset =l mul %last_idx, 16
    %last_ptr =l add %a_curr, %last_offset
    %last_key =l loadl %last_ptr
    %last_val =l loadl %last_ptr, 8
    storel %piv_key_tmp, %last_ptr
    storel %piv_val_tmp, %last_ptr, 8
    storel %last_key, %piv_ptr
    storel %last_val, %piv_ptr, 8
    %piv_key =l loadl %last_ptr
    %piv_val =l loadl %last_ptr, 8
    %init_i =l sub %a_curr, 16
    %init_j =l copy %a_curr
    jmp @loop_check
@loop_check
    %i =l phi @partition %init_i, @merge_ij %i_next2
    %j =l phi @partition %init_j, @merge_ij %j_next2
    %j_lt =w csltl %j, %last_ptr
    jnz %j_lt, @loop_body, @part_done
@loop_body
    %cur_key =l loadl %j
    %cur_val =l loadl %j, 8
    %key_lt =w csltl %cur_key, %piv_key
    %key_eq =w ceql %cur_key, %piv_key
    %val_le =w cslel %cur_val, %piv_val
    %tmp =w and %key_eq, %val_le
    %swap_cond =w or %key_lt, %tmp
    jnz %swap_cond, @do_swap, @no_swap
@do_swap
    %i_next =l add %i, 16
    %i_key =l loadl %i_next
    %i_val =l loadl %i_next, 8
    storel %cur_key, %i_next
    storel %cur_val, %i_next, 8
    storel %i_key, %j
    storel %i_val, %j, 8
    %j_next_sw =l add %j, 16
    jmp @merge_ij
@no_swap
    %i_nsw =l copy %i
    %j_next_nsw =l add %j, 16
    jmp @merge_ij
@merge_ij
    %i_next2 =l phi @do_swap %i_next, @no_swap %i_nsw
    %j_next2 =l phi @do_swap %j_next_sw, @no_swap %j_next_nsw
    jmp @loop_check
@part_done
    %i_next_final =l add %i, 16
    %piv_key2 =l loadl %last_ptr
    %piv_val2 =l loadl %last_ptr, 8
    %i_key2 =l loadl %i_next_final
    %i_val2 =l loadl %i_next_final, 8
    storel %piv_key2, %i_next_final
    storel %piv_val2, %i_next_final, 8
    storel %i_key2, %last_ptr
    storel %i_val2, %last_ptr, 8
    %byte_diff =l sub %i_next_final, %a_curr
    %left_n =l shr %byte_diff, 4
    %right_start =l add %i_next_final, 16
    %right_n_tmp =l sub %n_curr, %left_n
    %right_n =l sub %right_n_tmp, 1
    %cmp_lr =w csltl %left_n, %right_n
    jnz %cmp_lr, @do_left_recurse, @do_right_recurse
@do_left_recurse
    %call_left =l call $qsort_pairs(l %a_curr, l %left_n)
    jmp @cont_left
@cont_left
    %a_for_loop =l copy %right_start
    %n_for_loop =l copy %right_n
    jmp @loop
@do_right_recurse
    %call_right =l call $qsort_pairs(l %right_start, l %right_n)
    jmp @cont_right
@cont_right
    %a_for_loop2 =l copy %a_curr
    %n_for_loop2 =l copy %left_n
    jmp @loop
@done
    ret 0
}
