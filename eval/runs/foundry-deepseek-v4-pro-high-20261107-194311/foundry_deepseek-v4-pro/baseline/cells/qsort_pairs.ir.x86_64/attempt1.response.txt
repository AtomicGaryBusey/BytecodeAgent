export function l $qsort_pairs(l %a, l %n) {
@start
    %c =w csgtl %n, 1
    jnz %c, @loop_entry, @done
@loop_entry
    jmp @loop(%a, %n)
@done
    ret 0
@loop( %a_loop, %n_loop )
    %n_gt1 =w csgtl %n_loop, 1
    jnz %n_gt1, @loop_continue, @done
@loop_continue
    %pivot_idx =l div %n_loop, 2
    %piv_offset =l mul %pivot_idx, 16
    %piv_ptr =l add %a_loop, %piv_offset
    %piv_key_tmp =l loadl %piv_ptr
    %piv_val_tmp =l loadl %piv_ptr, 8
    %last_idx =l sub %n_loop, 1
    %last_offset =l mul %last_idx, 16
    %last_ptr =l add %a_loop, %last_offset
    %last_key =l loadl %last_ptr
    %last_val =l loadl %last_ptr, 8
    storel %piv_key_tmp, %last_ptr
    storel %piv_val_tmp, %last_ptr, 8
    storel %last_key, %piv_ptr
    storel %last_val, %piv_ptr, 8
    %piv_key =l loadl %last_ptr
    %piv_val =l loadl %last_ptr, 8
    %init_i =l sub %a_loop, 16
    %init_j =l copy %a_loop
    jmp @partition_loop(%init_i, %init_j)
@partition_loop( %i_ptr, %j_ptr )
    %j_lt_last =w csltl %j_ptr, %last_ptr
    jnz %j_lt_last, @part_body, @part_done
@part_body
    %cur_key =l loadl %j_ptr
    %cur_val =l loadl %j_ptr, 8
    %key_lt =w csltl %cur_key, %piv_key
    %key_eq =w ceql %cur_key, %piv_key
    %val_le =w cslel %cur_val, %piv_val
    %and_tmp =w and %key_eq, %val_le
    %swap_cond =w or %key_lt, %and_tmp
    jnz %swap_cond, @do_swap, @no_swap
@do_swap
    %i_next =l add %i_ptr, 16
    %i_key =l loadl %i_next
    %i_val =l loadl %i_next, 8
    storel %cur_key, %i_next
    storel %cur_val, %i_next, 8
    storel %i_key, %j_ptr
    storel %i_val, %j_ptr, 8
    jmp @next_j(%i_next, %j_ptr)
@no_swap
    jmp @next_j(%i_ptr, %j_ptr)
@next_j( %i_in, %j_in )
    %j_new =l add %j_in, 16
    jmp @partition_loop(%i_in, %j_new)
@part_done
    %i_next_final =l add %i_ptr, 16
    %piv_key2 =l loadl %last_ptr
    %piv_val2 =l loadl %last_ptr, 8
    %i_key2 =l loadl %i_next_final
    %i_val2 =l loadl %i_next_final, 8
    storel %piv_key2, %i_next_final
    storel %piv_val2, %i_next_final, 8
    storel %i_key2, %last_ptr
    storel %i_val2, %last_ptr, 8
    %byte_diff =l sub %i_next_final, %a_loop
    %left_n =l shr %byte_diff, 4
    %right_start =l add %i_next_final, 16
    %left_plus_one =l add %left_n, 1
    %right_n =l sub %n_loop, %left_plus_one
    %cmp_lr =w csltl %left_n, %right_n
    jnz %cmp_lr, @recurse_left, @recurse_right
@recurse_left
    call $qsort_pairs(l %a_loop, l %left_n)
    jmp @loop(%right_start, %right_n)
@recurse_right
    call $qsort_pairs(l %right_start, l %right_n)
    jmp @loop(%a_loop, %left_n)
}
