bubble_sort:
    mov x9, x1
    mov x10, x0
    cmp x9, #2
    b.lt .Ldone
    mov x11, #0
.Louter_loop:
    sub x13, x9, x11
    sub x13, x13, #1
    cbz x13, .Louter_next
    mov x12, #0
.Linner_loop:
    ldr x14, [x10, x12, lsl #3]
    add x15, x12, #1
    ldr x15, [x10, x15, lsl #3]
    cmp x14, x15
    b.le .Lno_swap
    str x15, [x10, x12, lsl #3]
    add x16, x12, #1
    str x14, [x10, x16, lsl #3]
.Lno_swap:
    add x12, x12, #1
    cmp x12, x13
    b.lt .Linner_loop
.Louter_next:
    add x11, x11, #1
    cmp x11, x9
    b.lt .Louter_loop
.Ldone:
    mov x0, #0
    ret
