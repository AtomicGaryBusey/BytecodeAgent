export function l $binary_search(l %a, l %n, l %key) {
@start
	%nle0 =w cslel %n, 0
	jnz %nle0, @ret_neg, @init
@ret_neg
	%m1 =l copy -1
	ret %m1
@init
	%zero =l copy 0
	%high_init =l sub %n, 1
	jmp @loop(%zero, %high_init)
@loop(l %low, l %high)
	%cond =w cslel %low, %high
	jnz %cond, @body, @not_found
@not_found
	%m2 =l copy -1
	ret %m2
@body
	%diff =l sub %high, %low
	%diff1 =l shr %diff, 1
	%mid =l add %low, %diff1
	%offset =l shl %mid, 3
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%lt =w csltl %val, %key
	jnz %lt, @less, @ge
@less
	%low_new =l add %mid, 1
	jmp @loop(%low_new, %high)
@ge
	%eq =w ceql %val, %key
	jnz %eq, @found, @greater
@greater
	%high_new =l sub %mid, 1
	jmp @loop(%low, %high_new)
@found
	ret %mid
}
