export function l $binary_search(l %a, l %n, l %key) {
@start
	%nle0 = cslel %n, 0
	jnz %nle0, @ret_neg, @init
@ret_neg
	%m1 = copy -1
	ret %m1
@init
	%zero = copy 0
	%high_init = sub %n, 1
	jmp @loop(%zero, %high_init)
@loop(l %low, l %high)
	%cond = cslel %low, %high
	jnz %cond, @body, @not_found
@not_found
	%m2 = copy -1
	ret %m2
@body
	%diff = sub %high, %low
	%diff1 = shr %diff, 1
	%mid = add %low, %diff1
	%offset = shl %mid, 3
	%ptr = add %a, %offset
	%val = loadl %ptr
	%lt = csltl %val, %key
	jnz %lt, @less, @ge
@less
	%low_new = add %mid, 1
	jmp @loop(%low_new, %high)
@ge
	%eq = ceql %val, %key
	jnz %eq, @found, @greater
@greater
	%high_new = sub %mid, 1
	jmp @loop(%low, %high_new)
@found
	ret %mid
}
