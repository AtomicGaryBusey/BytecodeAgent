export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i0 =l copy 0
    %c0 =l copy 0
    jmp @loop(%i0, %c0)

@loop(%i, %count)
    %cond =w csltl %i, %n
    jnz %cond, @body(%i, %count), @end(%count)

@body(%i, %count)
    %offset =l mul %i, 8
    %src_addr =l add %src, %offset
    %val =l loadl %src_addr
    %res =l call %f(l %val)
    %dst_addr =l add %dst, %offset
    storel %res, %dst_addr
    %lt =w csltl %res, %val
    %inc =l extsw %lt
    %new_count =l add %count, %inc
    %new_i =l add %i, 1
    jmp @loop(%new_i, %new_count)

@end(%count)
    ret %count
}
