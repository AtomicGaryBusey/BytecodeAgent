memmove_i64:
    test rdx, rdx
    je .Lreturn
    cmp rdi, rsi
    je .Lreturn
    jb .Lforward
    std
    lea rsi, [rsi + rdx*8 - 8]
    lea rdi, [rdi + rdx*8 - 8]
    mov rcx, rdx
    rep movsq
    cld
    xor eax, eax
    ret
.Lforward:
    cld
    mov rcx, rdx
    rep movsq
    xor eax, eax
    ret
.Lreturn:
    xor eax, eax
    ret
