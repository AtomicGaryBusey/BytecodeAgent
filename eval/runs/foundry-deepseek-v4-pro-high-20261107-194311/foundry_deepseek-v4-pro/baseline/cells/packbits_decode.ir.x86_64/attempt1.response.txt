export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %si0 =l copy 0
    %oi0 =l copy 0
    jmp @loophead
@loophead
    %si =l phi @start %si0, @literal_done %si_lit, @repeat_done %si_rep, @nop %si1
    %oi =l phi @start %oi0, @literal_done %oi_lit, @repeat_done %oi_rep, @nop %oi
    %cond =w csltl %si, %n
    jnz %cond, @process_header, @success
@process_header
    %h_ptr =l add %src, %si
    %h =w loadsb %h_ptr
    %si1 =l add %si, 1
    %is_neg =w csltw %h, 0
    jnz %is_neg, @check_nop, @literal
@check_nop
    %is_nop =w ceqw %h, -128
    jnz %is_nop, @nop, @repeat
@nop
    jmp @loophead ( %si1, %oi )
@literal
    %h_l =l extsw %h
    %count =l add %h_l, 1
    %needed =l copy %count
    %remaining_lit =l sub %n, %si1
    %enough_lit =w cslel %needed, %remaining_lit
    jnz %enough_lit, @literal_copy, @src_error
@literal_copy
    %cnt_lit_start =l copy %count
    %si_lit_start =l copy %si1
    %oi_lit_start =l copy %oi
    jmp @literal_loop
@literal_loop
    %cnt_lit =l phi @literal_copy %cnt_lit_start, @literal_write %cnt_lit_next
    %si_lit =l phi @literal_copy %si_lit_start, @literal_write %si_lit_next
    %oi_lit =l phi @literal_copy %oi_lit_start, @literal_write %oi_lit_next
    %cnt_lit_zero =w ceql %cnt_lit, 0
    jnz %cnt_lit_zero, @literal_done, @literal_body
@literal_body
    %oi_full_lit =w ceql %oi_lit, %cap
    jnz %oi_full_lit, @cap_error, @literal_write
@literal_write
    %src_lit_ptr =l add %src, %si_lit
    %byte_lit =w loadub %src_lit_ptr
    %dst_lit_ptr =l add %out, %oi_lit
    storeb %byte_lit, %dst_lit_ptr
    %oi_lit_next =l add %oi_lit, 1
    %si_lit_next =l add %si_lit, 1
    %cnt_lit_next =l sub %cnt_lit, 1
    jmp @literal_loop
@literal_done
    jmp @loophead ( %si_lit, %oi_lit )
@repeat
    %count_w =w sub 1, %h
    %remaining_rep =l sub %n, %si1
    %need_one =l copy 1
    %enough_rep =w cslel %need_one, %remaining_rep
    jnz %enough_rep, @repeat_read, @src_error
@repeat_read
    %rep_byte_ptr =l add %src, %si1
    %rep_byte =w loadub %rep_byte_ptr
    %si_rep_start =l add %si1, 1
    %count_rep =l extsw %count_w
    %cnt_rep_start =l copy %count_rep
    %oi_rep_start =l copy %oi
    jmp @repeat_loop
@repeat_loop
    %cnt_rep =l phi @repeat_read %cnt_rep_start, @repeat_write %cnt_rep_next
    %si_rep =l phi @repeat_read %si_rep_start, @repeat_write %si_rep_next
    %oi_rep =l phi @repeat_read %oi_rep_start, @repeat_write %oi_rep_next
    %cnt_rep_zero =w ceql %cnt_rep, 0
    jnz %cnt_rep_zero, @repeat_done, @repeat_body
@repeat_body
    %oi_full_rep =w ceql %oi_rep, %cap
    jnz %oi_full_rep, @cap_error, @repeat_write
@repeat_write
    %dst_rep_ptr =l add %out, %oi_rep
    storeb %rep_byte, %dst_rep_ptr
    %oi_rep_next =l add %oi_rep, 1
    %cnt_rep_next =l sub %cnt_rep, 1
    %si_rep_next =l copy %si_rep
    jmp @repeat_loop
@repeat_done
    jmp @loophead ( %si_rep, %oi_rep )
@success
    ret %oi
@src_error
    %err_src =l copy -2
    ret %err_src
@cap_error
    %err_cap =l copy -1
    ret %err_cap
}
