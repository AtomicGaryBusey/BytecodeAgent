export function l $count_desc(l %a, l %n) {
@start
	%n2 =l copy 2
	%cond =w csltl %n, %n2
	%nminus1 =l sub %n, 1
	jnz %cond, @ret0, @loopinit
@ret0
	ret 0
@loopinit
	%i.0 =l copy 0
	%res.0 =l copy 0
	jmp @loopcond
@loopcond
	%i =l phi @loopinit %i.0, @loopbody %i_next
	%res =l phi @loopinit %res.0, @loopbody %res_next
	%cond2 =w csltl %i, %nminus1
	jnz %cond2, @loopbody, @end
@loopbody
	%ptr_i =l mul %i, 8
	%addr_i =l add %a, %ptr_i
	%val_i =l loadl %addr_i
	%i1 =l add %i, 1
	%ptr_i1 =l mul %i1, 8
	%addr_i1 =l add %a, %ptr_i1
	%val_i1 =l loadl %addr_i1
	%gt =w csgtl %val_i, %val_i1
	%inc =l extuw %gt
	%res_next =l add %res, %inc
	%i_next =l add %i, 1
	jmp @loopcond
@end
	ret %res
}
