qsort_pairs:
    push rbp
    push r15
    push r14
    push r12
    push r13
    sub rsp, 16
    mov r15, rdi
    mov r14, rsi
    cmp r14, 1
    jle .Lreturn0
    mov rax, r14
    dec rax
    mov rdx, rax
    shr rdx, 1
    lea r12, [r15]
    mov rcx, rdx
    shl rcx, 4
    lea r13, [r15 + rcx]
    mov rcx, rax
    shl rcx, 4
    lea rbp, [r15 + rcx]
    mov rdi, r13
    mov rsi, r12
    call .Lrecord_less
    test eax, eax
    jz .Lskip_swap_low_mid
    mov rax, [r12]
    mov rdx, [r12+8]
    mov rcx, [r13]
    mov r8, [r13+8]
    mov [r12], rcx
    mov [r12+8], r8
    mov [r13], rax
    mov [r13+8], rdx
.Lskip_swap_low_mid:
    mov rdi, rbp
    mov rsi, r12
    call .Lrecord_less
    test eax, eax
    jz .Lskip_swap_low_high
    mov rax, [r12]
    mov rdx, [r12+8]
    mov rcx, [rbp]
    mov r8, [rbp+8]
    mov [r12], rcx
    mov [r12+8], r8
    mov [rbp], rax
    mov [rbp+8], rdx
.Lskip_swap_low_high:
    mov rdi, rbp
    mov rsi, r13
    call .Lrecord_less
    test eax, eax
    jz .Lskip_swap_mid_high
    mov rax, [r13]
    mov rdx, [r13+8]
    mov rcx, [rbp]
    mov r8, [rbp+8]
    mov [r13], rcx
    mov [r13+8], r8
    mov [rbp], rax
    mov [rbp+8], rdx
.Lskip_swap_mid_high:
    mov rax, [r13]
    mov rdx, [r13+8]
    mov rcx, [rbp]
    mov r8, [rbp+8]
    mov [r13], rcx
    mov [r13+8], r8
    mov [rbp], rax
    mov [rbp+8], rdx
    mov rax, r14
    dec rax
    mov [rsp], rax
    mov r12, -1
    xor r13d, r13d
.Lpart_loop:
    mov rdx, [rsp]
    cmp r13, rdx
    jge .Lpart_end
    mov rcx, r13
    shl rcx, 4
    lea rdi, [r15 + rcx]
    mov rsi, rbp
    call .Lrecord_less
    test eax, eax
    jz .Lpart_no_swap
    inc r12
    mov rcx, r12
    shl rcx, 4
    lea rsi, [r15 + rcx]
    mov rcx, r13
    shl rcx, 4
    lea rdi, [r15 + rcx]
    mov rax, [rsi]
    mov rdx, [rsi+8]
    mov rcx, [rdi]
    mov r8, [rdi+8]
    mov [rsi], rcx
    mov [rsi+8], r8
    mov [rdi], rax
    mov [rdi+8], rdx
.Lpart_no_swap:
    inc r13
    jmp .Lpart_loop
.Lpart_end:
    mov rcx, r12
    inc rcx
    mov rdx, rcx
    shl rcx, 4
    lea rdi, [r15 + rcx]
    mov rax, [rdi]
    mov rcx, [rdi+8]
    mov rsi, [rbp]
    mov r8, [rbp+8]
    mov [rdi], rsi
    mov [rdi+8], r8
    mov [rbp], rax
    mov [rbp+8], rcx
    mov r12, rdx
    mov r13, r14
    sub r13, rdx
    dec r13
    lea rcx, [rdx+1]
    shl rcx, 4
    lea rbp, [r15 + rcx]
    mov rdi, r15
    mov rsi, r12
    call qsort_pairs
    mov rdi, rbp
    mov rsi, r13
    call qsort_pairs
    jmp .Lreturn0
.Lrecord_less:
    mov rax, [rdi]
    cmp rax, [rsi]
    jl .Lless_true
    jg .Lless_false
    mov rax, [rdi+8]
    cmp rax, [rsi+8]
    jl .Lless_true
.Lless_false:
    xor eax, eax
    ret
.Lless_true:
    mov eax, 1
    ret
.Lreturn0:
    xor eax, eax
    add rsp, 16
    pop r13
    pop r12
    pop r14
    pop r15
    pop rbp
    ret
