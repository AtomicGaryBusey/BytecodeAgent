export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
	%p_src =l copy %src
	%p_dst =l copy %dst
	jmp @check
@check
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%x =l loadl %p_src
	%y =l call %f(l %x)
	storel %y, %p_dst
	%cmp =w csltl %y, %x
	%cmp_l =l extuw %cmp
	%count =l add %count, %cmp_l
	%p_src =l add %p_src, 8
	%p_dst =l add %p_dst, 8
	%i =l add %i, 1
	jmp @check
@end
	ret %count
}
