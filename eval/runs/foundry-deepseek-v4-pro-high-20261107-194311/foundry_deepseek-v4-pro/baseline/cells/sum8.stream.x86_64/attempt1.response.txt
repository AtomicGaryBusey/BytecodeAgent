sum8:
    mov rax, rdi                     # a
    lea r11, [rsi + rsi]             # 2*b
    sub rax, r11                     # a - 2*b
    lea r11, [rdx + rdx*2]           # 3*c
    add rax, r11                     # a - 2*b + 3*c
    lea r11, [rcx*4]                 # 4*d
    sub rax, r11                     # a - 2*b + 3*c - 4*d
    lea r11, [r8 + r8*4]             # 5*e
    add rax, r11                     # ... + 5*e
    imul r11, r9, 6                  # 6*f
    sub rax, r11                     # ... - 6*f
    mov r10, qword ptr [rsp+8]       # g
    imul r11, r10, 7                 # 7*g
    add rax, r11                     # ... + 7*g
    mov r10, qword ptr [rsp+16]      # h
    lea r11, [r10*8]                 # 8*h
    sub rax, r11                     # ... - 8*h
    ret
