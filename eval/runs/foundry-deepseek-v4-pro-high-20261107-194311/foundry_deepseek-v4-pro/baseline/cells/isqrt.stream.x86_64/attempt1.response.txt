isqrt:
    test rdi, rdi
    jz .Lzero
    bsr rcx, rdi
    shr rcx, 1
    inc rcx
    mov eax, 1
    shl rax, cl
.Lloop:
    mov rcx, rax
    mov rax, rdi
    xor edx, edx
    div rcx
    add rax, rcx
    shr rax, 1
    cmp rax, rcx
    jb .Lloop
    mov rax, rcx
    ret
.Lzero:
    xor eax, eax
    ret
