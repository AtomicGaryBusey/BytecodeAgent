export function l $map_sum(l %a, l %n, l %f) {
@entry
	%sum =l copy 0
	%i   =l copy 0
	jmp @loop

@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%ret =l call %f(l %val)
	%sum =l add %sum, %ret
	%i   =l add %i, 1
	jmp @loop

@done
	ret %sum
}
