export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
        %j =l copy 0
        jmp @zero_loop

@zero_loop
        %zero_cond =w cslel %j, %n
        jnz %zero_cond, @zero_body, @outer_init

@zero_body
        %off =l mul %j, 8
        %ptr =l add %row, %off
        storel 0, %ptr
        %j =l add %j, 1
        jmp @zero_loop

@outer_init
        %i =l copy 0
        jmp @outer_loop

@outer_loop
        %outer_cond =w csltl %i, %m
        jnz %outer_cond, @outer_body, @load_result

@outer_body
        %diag =l copy 0
        %inner_j =l copy 1
        jmp @inner_loop

@inner_loop
        %inner_cond =w cslel %inner_j, %n
        jnz %inner_cond, @inner_body, @inner_next_i

@inner_body
        %off_j =l mul %inner_j, 8
        %rowj_ptr =l add %row, %off_j
        %up =l loadl %rowj_ptr
        %j_minus_1 =l sub %inner_j, 1
        %off_jminus_1 =l mul %j_minus_1, 8
        %rowj_1_ptr =l add %row, %off_jminus_1
        %left_val =l loadl %rowj_1_ptr
        %x_ptr =l add %x, %i
        %xi =w loadub %x_ptr
        %y_ptr =l add %y, %j_minus_1
        %yj =w loadub %y_ptr
        %eq =w ceqw %xi, %yj
        %cand =l add %diag, 1
        jnz %eq, @match, @mismatch

@match
        %new_val =l copy %cand
        jmp @inner_store

@mismatch
        %up_gt =w csgtl %up, %left_val
        jnz %up_gt, @max_up, @max_left

@max_up
        %new_val =l copy %up
        jmp @inner_store

@max_left
        %new_val =l copy %left_val
        jmp @inner_store

@inner_store
        storel %new_val, %rowj_ptr
        %diag =l copy %up
        %inner_j =l add %inner_j, 1
        jmp @inner_loop

@inner_next_i
        %i =l add %i, 1
        jmp @outer_loop

@load_result
        %off_res =l mul %n, 8
        %res_ptr =l add %row, %off_res
        %res =l loadl %res_ptr
        ret %res
}
