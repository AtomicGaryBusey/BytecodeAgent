export function l $binary_search(l %a, l %n, l %key) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @notfound, @init
@init
	%lo_init =l 0
	%hi_init =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @init %lo_init, @less %lo_next, @greater %lo
	%hi =l phi @init %hi_init, @less %hi, @greater %hi_next
	%cond =w cslel %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l sar %diff, 1
	%mid =l add %lo, %half
	%offset =l shl %mid, 3
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%is_less =w csltl %val, %key
	jnz %is_less, @less, @ge
@ge
	%is_greater =w csgtl %val, %key
	jnz %is_greater, @greater, @equal
@equal
	ret %mid
@less
	%lo_next =l add %mid, 1
	jmp @loop
@greater
	%hi_next =l sub %mid, 1
	jmp @loop
@notfound
	ret -1
}
