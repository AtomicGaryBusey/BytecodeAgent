export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %total = mull %n, %n
    %total8 = mull %total, 8
    %base = copy %tmp
    %res = add %tmp, %total8

    %i_ptr = alloc8 1
    %j_ptr = alloc8 1
    %t_ptr = alloc8 1
    %sum_ptr = alloc8 1
    %k_ptr = alloc8 1

    storel 0, %i_ptr
@copy_a_loop
    %idx = loadl %i_ptr
    %cond_a = csltl %idx, %total
    jnz %cond_a, @copy_a_body, @copy_a_done
@copy_a_body
    %a_off = mull %idx, 8
    %a_ptr = add %a, %a_off
    %val = loadl %a_ptr
    %b_off = mull %idx, 8
    %b_ptr = add %base, %b_off
    storel %val, %b_ptr
    %idx_next = add %idx, 1
    storel %idx_next, %i_ptr
    jmp @copy_a_loop
@copy_a_done

    storel 0, %i_ptr
@zero_loop
    %idx2 = loadl %i_ptr
    %cond_z = csltl %idx2, %total
    jnz %cond_z, @zero_body, @zero_done
@zero_body
    %res_off = mull %idx2, 8
    %res_ptr = add %res, %res_off
    storel 0, %res_ptr
    %idx2_next = add %idx2, 1
    storel %idx2_next, %i_ptr
    jmp @zero_loop
@zero_done

    storel 0, %i_ptr
@diag_loop
    %i_d = loadl %i_ptr
    %cond_d = csltl %i_d, %n
    jnz %cond_d, @diag_body, @diag_done
@diag_body
    %diag_idx = mull %i_d, %n
    %diag_idx2 = add %diag_idx, %i_d
    %diag_off = mull %diag_idx2, 8
    %diag_ptr = add %res, %diag_off
    storel 1, %diag_ptr
    %i_d_next = add %i_d, 1
    storel %i_d_next, %i_ptr
    jmp @diag_loop
@diag_done

    storel %k, %k_ptr
    %k_init = loadl %k_ptr
    %k_zero = ceql %k_init, 0
    jnz %k_zero, @done_bits, @exp_wloop

@exp_wloop
    %k_cur = loadl %k_ptr
    %k_bit_val = and %k_cur, 1
    %k_bit_test = cnel %k_bit_val, 0
    jnz %k_bit_test, @do_mul, @after_mul

@after_mul
    jmp @do_square

@after_square
    %k_cur2 = loadl %k_ptr
    %k_shr = shr %k_cur2, 1
    storel %k_shr, %k_ptr
    %k_cur3 = loadl %k_ptr
    %k_ne = cnel %k_cur3, 0
    jnz %k_ne, @exp_wloop, @done_bits

@do_mul
    storel 0, %i_ptr
@mul_i_loop
    %i_m = loadl %i_ptr
    %i_cond_m = csltl %i_m, %n
    jnz %i_cond_m, @mul_i_body, @mul_i_done
@mul_i_body
    storel 0, %j_ptr
@mul_j_loop
    %j_m = loadl %j_ptr
    %j_cond_m = csltl %j_m, %n
    jnz %j_cond_m, @mul_j_body, @mul_j_done
@mul_j_body
    storel 0, %sum_ptr
    storel 0, %t_ptr
@mul_t_loop
    %t_m = loadl %t_ptr
    %t_cond_m = csltl %t_m, %n
    jnz %t_cond_m, @mul_t_body, @mul_t_done
@mul_t_body
    %i_n_m = mull %i_m, %n
    %a_off_m = add %i_n_m, %t_m
    %a_off8_m = mull %a_off_m, 8
    %a_ptr_m = add %res, %a_off8_m
    %a_val_m = loadl %a_ptr_m
    %t_n_m = mull %t_m, %n
    %b_off_m = add %t_n_m, %j_m
    %b_off8_m = mull %b_off_m, 8
    %b_ptr_m = add %base, %b_off8_m
    %b_val_m = loadl %b_ptr_m
    %prod_m = mull %a_val_m, %b_val_m
    %sum_old_m = loadl %sum_ptr
    %sum_new_m = add %sum_old_m, %prod_m
    storel %sum_new_m, %sum_ptr
    %t_next_m = add %t_m, 1
    storel %t_next_m, %t_ptr
    jmp @mul_t_loop
@mul_t_done
    %i_n_m2 = mull %i_m, %n
    %c_off_m = add %i_n_m2, %j_m
    %c_off8_m = mull %c_off_m, 8
    %c_ptr_m = add %out, %c_off8_m
    %final_sum_m = loadl %sum_ptr
    storel %final_sum_m, %c_ptr_m
    %j_next_m = add %j_m, 1
    storel %j_next_m, %j_ptr
    jmp @mul_j_loop
@mul_j_done
    %i_next_m = add %i_m, 1
    storel %i_next_m, %i_ptr
    jmp @mul_i_loop
@mul_i_done
    jmp @copy_out_to_res

@copy_out_to_res
    storel 0, %i_ptr
@cr_loop
    %idx_cr = loadl %i_ptr
    %cr_cond = csltl %idx_cr, %total
    jnz %cr_cond, @cr_body, @cr_done
@cr_body
    %out_off = mull %idx_cr, 8
    %out_ptr = add %out, %out_off
    %val_cr = loadl %out_ptr
    %res_off = mull %idx_cr, 8
    %res_ptr_cr = add %res, %res_off
    storel %val_cr, %res_ptr_cr
    %idx_cr_next = add %idx_cr, 1
    storel %idx_cr_next, %i_ptr
    jmp @cr_loop
@cr_done
    jmp @after_mul

@do_square
    storel 0, %i_ptr
@sq_i_loop
    %i_s = loadl %i_ptr
    %i_cond_s = csltl %i_s, %n
    jnz %i_cond_s, @sq_i_body, @sq_i_done
@sq_i_body
    storel 0, %j_ptr
@sq_j_loop
    %j_s = loadl %j_ptr
    %j_cond_s = csltl %j_s, %n
    jnz %j_cond_s, @sq_j_body, @sq_j_done
@sq_j_body
    storel 0, %sum_ptr
    storel 0, %t_ptr
@sq_t_loop
    %t_s = loadl %t_ptr
    %t_cond_s = csltl %t_s, %n
    jnz %t_cond_s, @sq_t_body, @sq_t_done
@sq_t_body
    %i_n_s = mull %i_s, %n
    %a_off_s = add %i_n_s, %t_s
    %a_off8_s = mull %a_off_s, 8
    %a_ptr_s = add %base, %a_off8_s
    %a_val_s = loadl %a_ptr_s
    %t_n_s = mull %t_s, %n
    %b_off_s = add %t_n_s, %j_s
    %b_off8_s = mull %b_off_s, 8
    %b_ptr_s = add %base, %b_off8_s
    %b_val_s = loadl %b_ptr_s
    %prod_s = mull %a_val_s, %b_val_s
    %sum_old_s = loadl %sum_ptr
    %sum_new_s = add %sum_old_s, %prod_s
    storel %sum_new_s, %sum_ptr
    %t_next_s = add %t_s, 1
    storel %t_next_s, %t_ptr
    jmp @sq_t_loop
@sq_t_done
    %i_n_s2 = mull %i_s, %n
    %c_off_s = add %i_n_s2, %j_s
    %c_off8_s = mull %c_off_s, 8
    %c_ptr_s = add %out, %c_off8_s
    %final_sum_s = loadl %sum_ptr
    storel %final_sum_s, %c_ptr_s
    %j_next_s = add %j_s, 1
    storel %j_next_s, %j_ptr
    jmp @sq_j_loop
@sq_j_done
    %i_next_s = add %i_s, 1
    storel %i_next_s, %i_ptr
    jmp @sq_i_loop
@sq_i_done
    jmp @copy_out_to_base

@copy_out_to_base
    storel 0, %i_ptr
@cb_loop
    %idx_cb = loadl %i_ptr
    %cb_cond = csltl %idx_cb, %total
    jnz %cb_cond, @cb_body, @cb_done
@cb_body
    %out_off_cb = mull %idx_cb, 8
    %out_ptr_cb = add %out, %out_off_cb
    %val_cb = loadl %out_ptr_cb
    %base_off = mull %idx_cb, 8
    %base_ptr_cb = add %base, %base_off
    storel %val_cb, %base_ptr_cb
    %idx_cb_next = add %idx_cb, 1
    storel %idx_cb_next, %i_ptr
    jmp @cb_loop
@cb_done
    jmp @after_square

@done_bits
    storel 0, %i_ptr
@final_loop
    %idx_f = loadl %i_ptr
    %f_cond = csltl %idx_f, %total
    jnz %f_cond, @final_body, @final_done
@final_body
    %res_off_f = mull %idx_f, 8
    %res_ptr_f = add %res, %res_off_f
    %val_f = loadl %res_ptr_f
    %out_off_f = mull %idx_f, 8
    %out_ptr_f = add %out, %out_off_f
    storel %val_f, %out_ptr_f
    %idx_f_next = add %idx_f, 1
    storel %idx_f_next, %i_ptr
    jmp @final_loop
@final_done

    %total_tmp = add %total, %total
    storel 0, %i_ptr
@zero_tmp_loop
    %idx_z = loadl %i_ptr
    %z_cond = csltl %idx_z, %total_tmp
    jnz %z_cond, @zero_tmp_body, @zero_tmp_done
@zero_tmp_body
    %tmp_off = mull %idx_z, 8
    %tmp_ptr = add %tmp, %tmp_off
    storel 0, %tmp_ptr
    %idx_z_next = add %idx_z, 1
    storel %idx_z_next, %i_ptr
    jmp @zero_tmp_loop
@zero_tmp_done

    ret 0
}
