export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %total =l mul %n, %n
    %total8 =l shl %total, 3
    %base =l copy %tmp
    %res =l add %tmp, %total8

    %i_ptr =l alloc8 1
    %j_ptr =l alloc8 1
    %t_ptr =l alloc8 1
    %sum_ptr =l alloc8 1
    %k_ptr =l alloc8 1

    storel 0, %i_ptr
@copy_a_loop
    %idx =l loadl %i_ptr
    %cond_a =w csltl %idx, %total
    jnz %cond_a, @copy_a_body, @copy_a_done
@copy_a_body
    %a_off =l shl %idx, 3
    %a_ptr =l add %a, %a_off
    %val =l loadl %a_ptr
    %b_off =l shl %idx, 3
    %b_ptr =l add %base, %b_off
    storel %val, %b_ptr
    %idx_next =l add %idx, 1
    storel %idx_next, %i_ptr
    jmp @copy_a_loop
@copy_a_done

    storel 0, %i_ptr
@zero_loop
    %idx2 =l loadl %i_ptr
    %cond_z =w csltl %idx2, %total
    jnz %cond_z, @zero_body, @zero_done
@zero_body
    %res_off =l shl %idx2, 3
    %res_ptr =l add %res, %res_off
    storel 0, %res_ptr
    %idx2_next =l add %idx2, 1
    storel %idx2_next, %i_ptr
    jmp @zero_loop
@zero_done

    storel 0, %i_ptr
@diag_loop
    %i_d =l loadl %i_ptr
    %cond_d =w csltl %i_d, %n
    jnz %cond_d, @diag_body, @diag_done
@diag_body
    %diag_idx =l mul %i_d, %n
    %diag_idx2 =l add %diag_idx, %i_d
    %diag_off =l shl %diag_idx2, 3
    %diag_ptr =l add %res, %diag_off
    storel 1, %diag_ptr
    %i_d_next =l add %i_d, 1
    storel %i_d_next, %i_ptr
    jmp @diag_loop
@diag_done

    storel %k, %k_ptr
    %k_init =l loadl %k_ptr
    %k_zero =w ceql %k_init, 0
    jnz %k_zero, @done_bits, @exp_wloop

@exp_wloop
    %k_cur =l loadl %k_ptr
    %k_bit_val =l and %k_cur, 1
    %k_bit_test =w cnel %k_bit_val, 0
    jnz %k_bit_test, @do_mul, @after_mul

@after_mul
    jmp @do_square

@after_square
    %k_cur2 =l loadl %k_ptr
    %k_shr =l shr %k_cur2, 1
    storel %k_shr, %k_ptr
    %k_cur3 =l loadl %k_ptr
    %k_ne =w cnel %k_cur3, 0
    jnz %k_ne, @exp_wloop, @done_bits

@do_mul
    storel 0, %i_ptr
@mul_i_loop
    %i_m =l loadl %i_ptr
    %i_cond_m =w csltl %i_m, %n
    jnz %i_cond_m, @mul_i_body, @mul_i_done
@mul_i_body
    storel 0, %j_ptr
@mul_j_loop
    %j_m =l loadl %j_ptr
    %j_cond_m =w csltl %j_m, %n
    jnz %j_cond_m, @mul_j_body, @mul_j_done
@mul_j_body
    storel 0, %sum_ptr
    storel 0, %t_ptr
@mul_t_loop
    %t_m =l loadl %t_ptr
    %t_cond_m =w csltl %t_m, %n
    jnz %t_cond_m, @mul_t_body, @mul_t_done
@mul_t_body
    %i_n_m =l mul %i_m, %n
    %a_off_m =l add %i_n_m, %t_m
    %a_off8_m =l shl %a_off_m, 3
    %a_ptr_m =l add %res, %a_off8_m
    %a_val_m =l loadl %a_ptr_m
    %t_n_m =l mul %t_m, %n
    %b_off_m =l add %t_n_m, %j_m
    %b_off8_m =l shl %b_off_m, 3
    %b_ptr_m =l add %base, %b_off8_m
    %b_val_m =l loadl %b_ptr_m
    %prod_m =l mul %a_val_m, %b_val_m
    %sum_old_m =l loadl %sum_ptr
    %sum_new_m =l add %sum_old_m, %prod_m
    storel %sum_new_m, %sum_ptr
    %t_next_m =l add %t_m, 1
    storel %t_next_m, %t_ptr
    jmp @mul_t_loop
@mul_t_done
    %i_n_m2 =l mul %i_m, %n
    %c_off_m =l add %i_n_m2, %j_m
    %c_off8_m =l shl %c_off_m, 3
    %c_ptr_m =l add %out, %c_off8_m
    %final_sum_m =l loadl %sum_ptr
    storel %final_sum_m, %c_ptr_m
    %j_next_m =l add %j_m, 1
    storel %j_next_m, %j_ptr
    jmp @mul_j_loop
@mul_j_done
    %i_next_m =l add %i_m, 1
    storel %i_next_m, %i_ptr
    jmp @mul_i_loop
@mul_i_done
    jmp @copy_out_to_res

@copy_out_to_res
    storel 0, %i_ptr
@cr_loop
    %idx_cr =l loadl %i_ptr
    %cr_cond =w csltl %idx_cr, %total
    jnz %cr_cond, @cr_body, @cr_done
@cr_body
    %out_off =l shl %idx_cr, 3
    %out_ptr =l add %out, %out_off
    %val_cr =l loadl %out_ptr
    %res_off =l shl %idx_cr, 3
    %res_ptr_cr =l add %res, %res_off
    storel %val_cr, %res_ptr_cr
    %idx_cr_next =l add %idx_cr, 1
    storel %idx_cr_next, %i_ptr
    jmp @cr_loop
@cr_done
    jmp @after_mul

@do_square
    storel 0, %i_ptr
@sq_i_loop
    %i_s =l loadl %i_ptr
    %i_cond_s =w csltl %i_s, %n
    jnz %i_cond_s, @sq_i_body, @sq_i_done
@sq_i_body
    storel 0, %j_ptr
@sq_j_loop
    %j_s =l loadl %j_ptr
    %j_cond_s =w csltl %j_s, %n
    jnz %j_cond_s, @sq_j_body, @sq_j_done
@sq_j_body
    storel 0, %sum_ptr
    storel 0, %t_ptr
@sq_t_loop
    %t_s =l loadl %t_ptr
    %t_cond_s =w csltl %t_s, %n
    jnz %t_cond_s, @sq_t_body, @sq_t_done
@sq_t_body
    %i_n_s =l mul %i_s, %n
    %a_off_s =l add %i_n_s, %t_s
    %a_off8_s =l shl %a_off_s, 3
    %a_ptr_s =l add %base, %a_off8_s
    %a_val_s =l loadl %a_ptr_s
    %t_n_s =l mul %t_s, %n
    %b_off_s =l add %t_n_s, %j_s
    %b_off8_s =l shl %b_off_s, 3
    %b_ptr_s =l add %base, %b_off8_s
    %b_val_s =l loadl %b_ptr_s
    %prod_s =l mul %a_val_s, %b_val_s
    %sum_old_s =l loadl %sum_ptr
    %sum_new_s =l add %sum_old_s, %prod_s
    storel %sum_new_s, %sum_ptr
    %t_next_s =l add %t_s, 1
    storel %t_next_s, %t_ptr
    jmp @sq_t_loop
@sq_t_done
    %i_n_s2 =l mul %i_s, %n
    %c_off_s =l add %i_n_s2, %j_s
    %c_off8_s =l shl %c_off_s, 3
    %c_ptr_s =l add %out, %c_off8_s
    %final_sum_s =l loadl %sum_ptr
    storel %final_sum_s, %c_ptr_s
    %j_next_s =l add %j_s, 1
    storel %j_next_s, %j_ptr
    jmp @sq_j_loop
@sq_j_done
    %i_next_s =l add %i_s, 1
    storel %i_next_s, %i_ptr
    jmp @sq_i_loop
@sq_i_done
    jmp @copy_out_to_base

@copy_out_to_base
    storel 0, %i_ptr
@cb_loop
    %idx_cb =l loadl %i_ptr
    %cb_cond =w csltl %idx_cb, %total
    jnz %cb_cond, @cb_body, @cb_done
@cb_body
    %out_off_cb =l shl %idx_cb, 3
    %out_ptr_cb =l add %out, %out_off_cb
    %val_cb =l loadl %out_ptr_cb
    %base_off =l shl %idx_cb, 3
    %base_ptr_cb =l add %base, %base_off
    storel %val_cb, %base_ptr_cb
    %idx_cb_next =l add %idx_cb, 1
    storel %idx_cb_next, %i_ptr
    jmp @cb_loop
@cb_done
    jmp @after_square

@done_bits
    storel 0, %i_ptr
@final_loop
    %idx_f =l loadl %i_ptr
    %f_cond =w csltl %idx_f, %total
    jnz %f_cond, @final_body, @final_done
@final_body
    %res_off_f =l shl %idx_f, 3
    %res_ptr_f =l add %res, %res_off_f
    %val_f =l loadl %res_ptr_f
    %out_off_f =l shl %idx_f, 3
    %out_ptr_f =l add %out, %out_off_f
    storel %val_f, %out_ptr_f
    %idx_f_next =l add %idx_f, 1
    storel %idx_f_next, %i_ptr
    jmp @final_loop
@final_done

    %total_tmp =l add %total, %total
    storel 0, %i_ptr
@zero_tmp_loop
    %idx_z =l loadl %i_ptr
    %z_cond =w csltl %idx_z, %total_tmp
    jnz %z_cond, @zero_tmp_body, @zero_tmp_done
@zero_tmp_body
    %tmp_off =l shl %idx_z, 3
    %tmp_ptr =l add %tmp, %tmp_off
    storel 0, %tmp_ptr
    %idx_z_next =l add %idx_z, 1
    storel %idx_z_next, %i_ptr
    jmp @zero_tmp_loop
@zero_tmp_done

    ret 0
}
