export function l $fib(l %n) {
@start
    %is_lt2 =w cslel %n, 1
    jnz %is_lt2, @ret_n, @loop_init
@ret_n
    ret %n
@loop_init
    %a =l copy 0
    %b =l copy 1
    %i =l copy 2
@loop
    %cond =w cslel %i, %n
    jnz %cond, @body, @done
@body
    %c =l add %a, %b
    %a =l copy %b
    %b =l copy %c
    %i =l add %i, 1
    jmp @loop
@done
    ret %b
}
