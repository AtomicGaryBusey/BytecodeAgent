.globl _start
.text
_start:
    lea rbx, [rip + input_buf]
    lea r12, [rip + input_buf]
    add r12, 32768
read_loop:
    mov rdx, r12
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    mov eax, 0
    mov edi, 0
    mov rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rax, [rip + input_buf]
    mov r13, rbx
    sub r13, rax
    lea rbx, [rip + input_buf]
    mov r12, rbx
    add r12, r13
    mov r15, rbx
    xor r14d, r14d
parse_loop:
    cmp rbx, r12
    jae parse_end
    cmp byte ptr [rbx], 10
    jne parse_next
    lea rax, [rip + line_ptr]
    mov qword ptr [rax + r14*8], r15
    mov rax, rbx
    sub rax, r15
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + r14*8], rax
    inc r14
    mov r15, rbx
    inc r15
parse_next:
    inc rbx
    jmp parse_loop
parse_end:
    cmp r15, r12
    jae sort_start
    lea rax, [rip + line_ptr]
    mov qword ptr [rax + r14*8], r15
    mov rax, r12
    sub rax, r15
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + r14*8], rax
    inc r14
sort_start:
    mov ebx, 1
sort_outer:
    cmp rbx, r14
    jae sort_done
    lea rax, [rip + line_ptr]
    mov r12, qword ptr [rax + rbx*8]
    lea rax, [rip + line_len]
    mov r13, qword ptr [rax + rbx*8]
    mov r15, rbx
sort_inner:
    test r15, r15
    jz sort_insert
    lea rax, [rip + line_ptr]
    mov rdi, qword ptr [rax + r15*8 - 8]
    lea rax, [rip + line_len]
    mov rsi, qword ptr [rax + r15*8 - 8]
    mov rdx, r12
    mov rcx, r13
    call compare
    test rax, rax
    jle sort_insert
    lea rax, [rip + line_ptr]
    mov rdi, qword ptr [rax + r15*8 - 8]
    mov qword ptr [rax + r15*8], rdi
    lea rax, [rip + line_len]
    mov rsi, qword ptr [rax + r15*8 - 8]
    mov qword ptr [rax + r15*8], rsi
    dec r15
    jmp sort_inner
sort_insert:
    lea rax, [rip + line_ptr]
    mov qword ptr [rax + r15*8], r12
    lea rax, [rip + line_len]
    mov qword ptr [rax + r15*8], r13
    inc rbx
    jmp sort_outer
sort_done:
    xor ebx, ebx
output_loop:
    cmp rbx, r14
    jae exit_prog
    lea rax, [rip + line_ptr]
    mov r12, qword ptr [rax + rbx*8]
    lea rax, [rip + line_len]
    mov r13, qword ptr [rax + rbx*8]
    test r13, r13
    jz write_nl
write_line_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle write_nl
    add r12, rax
    sub r13, rax
    test r13, r13
    jnz write_line_loop
write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall
    inc rbx
    jmp output_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
compare:
    xor r9d, r9d
    mov r10, rsi
    cmp rcx, rsi
    jae 1f
    mov r10, rcx
1:
    cmp r9, r10
    jae 2f
    movzx eax, byte ptr [rdi + r9]
    movzx r8d, byte ptr [rdx + r9]
    cmp eax, r8d
    jb compare_less
    ja compare_greater
    inc r9
    jmp 1b
2:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov rax, -1
    ret
compare_greater:
    mov rax, 1
    ret
.section .rodata
newline_byte: .byte 10
.bss
.balign 16
input_buf: .zero 32768
.balign 8
line_ptr: .zero 8000
line_len: .zero 8000
