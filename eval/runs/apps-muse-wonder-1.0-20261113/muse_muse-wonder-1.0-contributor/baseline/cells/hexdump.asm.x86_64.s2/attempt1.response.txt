.globl _start
.text
_start:
	xor r12d, r12d
read_loop:
	cmp r12, 65536
	jae read_done
	mov eax, 0
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov edx, 65536
	sub rdx, r12
	syscall
	test rax, rax
	jle read_done
	add r12, rax
	jmp read_loop
read_done:
	xor r13d, r13d
row_loop:
	cmp r13, r12
	jae final_line
	mov rax, r12
	sub rax, r13
	mov r14, rax
	cmp r14, 16
	jbe n_ready
	mov r14d, 16
n_ready:
	lea r15, [rip + inbuf]
	add r15, r13
	lea rdi, [rip + outbuf]
	mov ebx, 28
offset_loop:
	mov rax, r13
	mov ecx, ebx
	shr rax, cl
	and eax, 15
	lea rsi, [rip + hexdigits]
	mov al, byte ptr [rsi + rax]
	mov byte ptr [rdi], al
	inc rdi
	sub ebx, 4
	jns offset_loop
	mov byte ptr [rdi], 32
	mov byte ptr [rdi + 1], 32
	add rdi, 2
	xor ebx, ebx
slot_loop:
	cmp rbx, r14
	jae slot_pad
	movzx eax, byte ptr [r15 + rbx]
	mov edx, eax
	shr edx, 4
	and eax, 15
	lea rcx, [rip + hexdigits]
	mov dl, byte ptr [rcx + rdx]
	mov byte ptr [rdi], dl
	inc rdi
	mov al, byte ptr [rcx + rax]
	mov byte ptr [rdi], al
	inc rdi
	mov byte ptr [rdi], 32
	inc rdi
	jmp slot_done
slot_pad:
	mov byte ptr [rdi], 32
	mov byte ptr [rdi + 1], 32
	mov byte ptr [rdi + 2], 32
	add rdi, 3
slot_done:
	cmp rbx, 7
	jne slot_noextra
	mov byte ptr [rdi], 32
	inc rdi
slot_noextra:
	inc rbx
	cmp rbx, 16
	jb slot_loop
	mov byte ptr [rdi], 32
	inc rdi
	mov byte ptr [rdi], 124
	inc rdi
	xor ebx, ebx
ascii_loop:
	cmp rbx, r14
	jae ascii_done
	movzx eax, byte ptr [r15 + rbx]
	cmp eax, 32
	jb ascii_dot
	cmp eax, 126
	jbe ascii_store
ascii_dot:
	mov eax, 46
ascii_store:
	mov byte ptr [rdi], al
	inc rdi
	inc rbx
	jmp ascii_loop
ascii_done:
	mov byte ptr [rdi], 124
	inc rdi
	mov byte ptr [rdi], 10
	inc rdi
	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
write_row_loop:
	test rdx, rdx
	jz write_row_done
	mov eax, 1
	mov edi, 1
	syscall
	test rax, rax
	jle write_row_done
	add rsi, rax
	sub rdx, rax
	jmp write_row_loop
write_row_done:
	add r13, r14
	jmp row_loop
final_line:
	lea rdi, [rip + outbuf]
	mov ebx, 28
final_loop:
	mov rax, r12
	mov ecx, ebx
	shr rax, cl
	and eax, 15
	lea rsi, [rip + hexdigits]
	mov al, byte ptr [rsi + rax]
	mov byte ptr [rdi], al
	inc rdi
	sub ebx, 4
	jns final_loop
	mov byte ptr [rdi], 10
	inc rdi
	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
write_final_loop:
	test rdx, rdx
	jz exit_prog
	mov eax, 1
	mov edi, 1
	syscall
	test rax, rax
	jle exit_prog
	add rsi, rax
	sub rdx, rax
	jmp write_final_loop
exit_prog:
	mov eax, 60
	xor edi, edi
	syscall
.section .rodata
hexdigits: .ascii "0123456789abcdef"
.bss
.balign 16
inbuf: .zero 65536
outbuf: .zero 128
