.globl _start
.text
_start:
	lea rbx, [rip + input]
	xor r12d, r12d
read_loop:
	cmp r12, 32768
	jae parse_start
	mov eax, 0
	xor edi, edi
	lea rsi, [rip + input]
	add rsi, r12
	mov rdx, 32768
	sub rdx, r12
	syscall
	cmp rax, 0
	jle parse_start
	add r12, rax
	jmp read_loop
parse_start:
	lea rbx, [rip + input]
	lea r8, [rip + offsets]
	lea r9, [rip + lengths]
	xor r13d, r13d
	xor esi, esi
	xor edi, edi
parse_loop:
	cmp rsi, r12
	jae parse_tail
	movzx eax, byte ptr [rbx + rsi]
	cmp al, 10
	jne parse_next
	mov rax, rbx
	add rax, rdi
	mov qword ptr [r8 + r13*8], rax
	mov rax, rsi
	sub rax, rdi
	mov qword ptr [r9 + r13*8], rax
	inc r13
	lea rdi, [rsi + 1]
parse_next:
	inc rsi
	jmp parse_loop
parse_tail:
	cmp rdi, r12
	jae sort_start
	mov rax, rbx
	add rax, rdi
	mov qword ptr [r8 + r13*8], rax
	mov rax, r12
	sub rax, rdi
	mov qword ptr [r9 + r13*8], rax
	inc r13
sort_start:
	cmp r13, 2
	jb output_start
	xor r14d, r14d
sort_outer:
	mov rax, r13
	dec rax
	cmp r14, rax
	jae output_start
	xor r15d, r15d
sort_inner:
	mov rax, r13
	sub rax, r14
	dec rax
	cmp r15, rax
	jae sort_next_outer
	lea r8, [rip + offsets]
	lea r9, [rip + lengths]
	mov rdi, qword ptr [r8 + r15*8]
	mov rsi, qword ptr [r9 + r15*8]
	mov rdx, qword ptr [r8 + r15*8 + 8]
	mov rcx, qword ptr [r9 + r15*8 + 8]
	call compare
	test eax, eax
	jle sort_no_swap
	lea r8, [rip + offsets]
	lea r9, [rip + lengths]
	mov rax, qword ptr [r8 + r15*8]
	mov rcx, qword ptr [r8 + r15*8 + 8]
	mov qword ptr [r8 + r15*8], rcx
	mov qword ptr [r8 + r15*8 + 8], rax
	mov rax, qword ptr [r9 + r15*8]
	mov rcx, qword ptr [r9 + r15*8 + 8]
	mov qword ptr [r9 + r15*8], rcx
	mov qword ptr [r9 + r15*8 + 8], rax
sort_no_swap:
	inc r15
	jmp sort_inner
sort_next_outer:
	inc r14
	jmp sort_outer
output_start:
	xor r14d, r14d
output_loop:
	cmp r14, r13
	jae exit_program
	lea r8, [rip + offsets]
	lea r9, [rip + lengths]
	mov rsi, qword ptr [r8 + r14*8]
	mov rdx, qword ptr [r9 + r14*8]
	call write_all
	lea rsi, [rip + newline_byte]
	mov edx, 1
	call write_all
	inc r14
	jmp output_loop
exit_program:
	mov eax, 60
	xor edi, edi
	syscall
compare:
	mov r8, rsi
	mov r9, rcx
	mov r11, r8
	cmp r11, r9
	jb compare_min_done
	mov r11, r9
compare_min_done:
	xor r10d, r10d
compare_loop:
	mov rax, r11
	sub rax, r10
	cmp rax, 8
	jb compare_byte_loop
	mov rax, qword ptr [rdi + r10]
	mov rcx, qword ptr [rdx + r10]
	cmp rax, rcx
	jne compare_chunk_diff
	add r10, 8
	jmp compare_loop
compare_chunk_diff:
	bswap rax
	bswap rcx
	cmp rax, rcx
	jb compare_less
	ja compare_greater
compare_byte_loop:
	cmp r10, r11
	jae compare_equal_prefix
	movzx eax, byte ptr [rdi + r10]
	movzx ecx, byte ptr [rdx + r10]
	cmp al, cl
	jb compare_less
	ja compare_greater
	inc r10
	jmp compare_byte_loop
compare_equal_prefix:
	cmp r8, r9
	jb compare_less
	ja compare_greater
	xor eax, eax
	ret
compare_less:
	mov eax, -1
	ret
compare_greater:
	mov eax, 1
	ret
write_all:
	push rbx
	push r12
	mov rbx, rsi
	mov r12, rdx
write_all_loop:
	test r12, r12
	jz write_all_done
	mov eax, 1
	mov edi, 1
	mov rsi, rbx
	mov rdx, r12
	syscall
	cmp rax, 0
	jle write_all_done
	add rbx, rax
	sub r12, rax
	jmp write_all_loop
write_all_done:
	pop r12
	pop rbx
	ret
.section .rodata
newline_byte:
	.byte 10
.bss
.balign 8
input:
	.zero 32768
.balign 8
offsets:
	.zero 8000
.balign 8
lengths:
	.zero 8000
