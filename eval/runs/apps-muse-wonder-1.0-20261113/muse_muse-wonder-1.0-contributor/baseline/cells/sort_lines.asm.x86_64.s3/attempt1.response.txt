.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 40000
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buffer]
    lea rdx, [rip + input_buffer]
    add rdx, rbx
    mov rcx, rsi
    xor r8d, r8d
parse_loop:
    cmp rcx, rdx
    jae parse_tail
    cmp byte ptr [rcx], 10
    jne parse_next
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r8*8], rsi
    mov rax, rcx
    sub rax, rsi
    lea r9, [rip + line_lens]
    mov qword ptr [r9 + r8*8], rax
    inc r8
    inc rcx
    mov rsi, rcx
    jmp parse_loop
parse_next:
    inc rcx
    jmp parse_loop
parse_tail:
    cmp rsi, rdx
    jae parse_done
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r8*8], rsi
    mov rax, rdx
    sub rax, rsi
    lea r9, [rip + line_lens]
    mov qword ptr [r9 + r8*8], rax
    inc r8
parse_done:
    mov rbx, r8
    cmp rbx, 2
    jb sort_done
    mov r12, 1
sort_outer:
    cmp r12, rbx
    jae sort_done
    lea rax, [rip + line_ptrs]
    mov r13, qword ptr [rax + r12*8]
    lea rax, [rip + line_lens]
    mov r14, qword ptr [rax + r12*8]
    lea r15, [r12 - 1]
sort_inner:
    test r15, r15
    js sort_insert
    lea rax, [rip + line_ptrs]
    mov rdi, qword ptr [rax + r15*8]
    lea rax, [rip + line_lens]
    mov rsi, qword ptr [rax + r15*8]
    mov rdx, r13
    mov rcx, r14
    call compare_lines
    test eax, eax
    jle sort_insert
    lea rax, [rip + line_ptrs]
    mov rcx, qword ptr [rax + r15*8]
    mov qword ptr [rax + r15*8 + 8], rcx
    lea rax, [rip + line_lens]
    mov rcx, qword ptr [rax + r15*8]
    mov qword ptr [rax + r15*8 + 8], rcx
    dec r15
    jmp sort_inner
sort_insert:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r15*8 + 8], r13
    lea rax, [rip + line_lens]
    mov qword ptr [rax + r15*8 + 8], r14
    inc r12
    jmp sort_outer
sort_done:
    xor r12d, r12d
output_loop:
    cmp r12, rbx
    jae output_done
    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + r12*8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + r12*8]
    call write_all
    lea rsi, [rip + newline_char]
    mov edx, 1
    call write_all
    inc r12
    jmp output_loop
output_done:
    mov eax, 60
    xor edi, edi
    syscall
compare_lines:
    xor r8d, r8d
    mov r9, rsi
    cmp r9, rcx
    jbe compare_loop
    mov r9, rcx
compare_loop:
    cmp r8, r9
    jae compare_prefix
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp eax, r10d
    jb compare_less
    ja compare_greater
    inc r8
    jmp compare_loop
compare_prefix:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret
write_all:
    test rdx, rdx
    jz write_all_done
write_all_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_all_done
    add rsi, rax
    sub rdx, rax
    jnz write_all_loop
write_all_done:
    ret
.section .rodata
newline_char:
    .byte 10
.bss
    .balign 8
input_buffer:
    .zero 40000
    .balign 8
line_ptrs:
    .zero 8000
    .balign 8
line_lens:
    .zero 8000
