.text
.globl _start
_start:
    xor ebx, ebx
    lea r12, [rip + inbuf]
read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    add r12, rbx
    lea rbx, [rip + inbuf]
process_loop:
    cmp rbx, r12
    jae exit_prog
    mov rsi, rbx
scan_loop:
    cmp rsi, r12
    jae scan_end
    cmp byte ptr [rsi], 10
    je scan_nl
    inc rsi
    jmp scan_loop
scan_nl:
    lea r13, [rsi + 1]
    jmp line_common
scan_end:
    mov rsi, r12
    mov r13, r12
line_common:
    mov r14, rsi
    sub r14, rbx
    mov rdi, rbx
    mov rsi, r14
    call reverse
    mov rdi, rbx
    mov rsi, r14
    call write_all
    lea rdi, [rip + nl]
    mov esi, 1
    call write_all
    mov rbx, r13
    jmp process_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
reverse:
    mov rax, rdi
    lea rcx, [rdi + rsi]
    dec rcx
1:
    cmp rax, rcx
    jae 2f
    mov dl, byte ptr [rax]
    mov r8b, byte ptr [rcx]
    mov byte ptr [rax], r8b
    mov byte ptr [rcx], dl
    inc rax
    dec rcx
    jmp 1b
2:
    ret
write_all:
    mov r8, rdi
    mov r9, rsi
1:
    test r9, r9
    jz 2f
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jle 2f
    add r8, rax
    sub r9, rax
    jmp 1b
2:
    ret
.section .rodata
nl:
    .byte 10
.bss
.align 16
inbuf:
    .zero 65536
