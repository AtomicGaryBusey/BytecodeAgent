.text
.globl _start
_start:
    lea r13, [rip + inbuf]
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r13 + rbx]
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    xor ebx, ebx
    mov rsi, r13
    lea rbp, [rip + valstack]
line_loop:
    cmp rbx, r12
    jae exit_prog
    mov r13, rbx
find_eol:
    cmp r13, r12
    jae line_ready
    cmp byte ptr [rsi + r13], 10
    je line_ready
    inc r13
    jmp find_eol
line_ready:
    mov rdi, rbx
    xor r14d, r14d
token_loop:
    cmp rdi, r13
    jae tokens_done
    movzx eax, byte ptr [rsi + rdi]
    cmp al, 32
    je sep_inc
    cmp al, 9
    jne token_found
sep_inc:
    inc rdi
    jmp token_loop
token_found:
    mov r8, rdi
    mov r9, rdi
scan_token:
    cmp r9, r13
    jae token_scanned
    movzx eax, byte ptr [rsi + r9]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc r9
    jmp scan_token
token_scanned:
    mov rdi, r9
    mov r10, r9
    sub r10, r8
    cmp r10, 1
    jne parse_literal
    movzx eax, byte ptr [rsi + r8]
    cmp al, 43
    je do_operator
    cmp al, 45
    je do_operator
    cmp al, 42
    je do_operator
    cmp al, 47
    je do_operator
    cmp al, 37
    je do_operator
    jmp parse_literal
do_operator:
    mov r10d, eax
    cmp r14, 2
    jb line_error
    dec r14
    mov r8, qword ptr [rbp + r14*8]
    dec r14
    mov rax, qword ptr [rbp + r14*8]
    cmp r10b, 43
    je op_add
    cmp r10b, 45
    je op_sub
    cmp r10b, 42
    je op_mul
    cmp r10b, 47
    je op_div
    jmp op_mod
op_add:
    add rax, r8
    jmp push_result
op_sub:
    sub rax, r8
    jmp push_result
op_mul:
    imul rax, r8
    jmp push_result
op_div:
    test r8, r8
    jz line_error
    cmp r8, -1
    jne div_ok
    mov rcx, qword ptr [rip + intmin_const]
    cmp rax, rcx
    je line_error
div_ok:
    cqo
    idiv r8
    jmp push_result
op_mod:
    test r8, r8
    jz line_error
    cmp r8, -1
    jne mod_ok
    mov rcx, qword ptr [rip + intmin_const]
    cmp rax, rcx
    je line_error
mod_ok:
    cqo
    idiv r8
    mov rax, rdx
    jmp push_result
push_result:
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp token_loop
parse_literal:
    mov rcx, r8
    xor eax, eax
    xor edx, edx
    cmp byte ptr [rsi + rcx], 45
    jne literal_digits_init
    mov edx, 1
    inc rcx
literal_digits_init:
    mov r11, r9
    sub r11, rcx
    cmp r11, 1
    jb line_error
    cmp r11, 18
    ja line_error
literal_loop:
    cmp rcx, r9
    jae literal_done
    movzx r10d, byte ptr [rsi + rcx]
    sub r10d, 48
    cmp r10d, 9
    ja line_error
    imul rax, rax, 10
    add rax, r10
    inc rcx
    jmp literal_loop
literal_done:
    test edx, edx
    jz push_result
    neg rax
    jmp push_result
tokens_done:
    cmp r14, 1
    jne line_error
    mov rax, qword ptr [rbp]
    mov r10, rax
    lea r9, [rip + outbuf]
    add r9, 32
    dec r9
    mov byte ptr [r9], 10
    test rax, rax
    jns mag_ready
    neg rax
mag_ready:
digit_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz digit_loop
    test r10, r10
    jns build_done
    dec r9
    mov byte ptr [r9], 45
build_done:
    lea rax, [rip + outbuf]
    add rax, 32
    sub rax, r9
    mov rdx, rax
    mov rsi, r9
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line
line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 6
    syscall
next_line:
    lea rsi, [rip + inbuf]
    lea rbp, [rip + valstack]
    mov rbx, r13
    cmp rbx, r12
    jae line_loop
    inc rbx
    jmp line_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata
err_msg:
    .ascii "error\n"
.balign 8
intmin_const:
    .quad 0x8000000000000000
.bss
.balign 16
inbuf:
    .zero 65537
.balign 8
valstack:
    .zero 8192
.balign 16
outbuf:
    .zero 32
