.globl _start
.text
_start:
    xor r12, r12
    lea rbx, [rip + buffer]
read_loop:
    mov rdx, 65536
    sub rdx, r12
    cmp rdx, 0
    je find_bang
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r12]
    syscall
    cmp rax, 0
    jle find_bang
    add r12, rax
    jmp read_loop
find_bang:
    xor rcx, rcx
find_loop:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je bang_found
    inc rcx
    jmp find_loop
bang_found:
    mov r13, rcx
    lea r15, [rcx + 1]
    jmp validate_init
no_bang:
    mov r13, r12
    mov r15, r12
validate_init:
    xor rcx, rcx
    xor r14, r14
    lea rsi, [rip + bstack]
    lea rdi, [rip + match_table]
validate_loop:
    cmp rcx, r13
    jae validate_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x5b
    je val_open
    cmp al, 0x5d
    je val_close
    inc rcx
    jmp validate_loop
val_open:
    mov dword ptr [rsi + r14*4], ecx
    inc r14
    inc rcx
    jmp validate_loop
val_close:
    cmp r14, 0
    je unbalanced
    dec r14
    mov edx, dword ptr [rsi + r14*4]
    mov dword ptr [rdi + rcx*4], edx
    mov dword ptr [rdi + rdx*4], ecx
    inc rcx
    jmp validate_loop
validate_done:
    cmp r14, 0
    jne unbalanced
    lea r8, [rip + tape]
    mov r9, rdi
    xor r14, r14
    xor rbp, rbp
exec_loop:
    cmp rbp, r13
    jae success
    movzx eax, byte ptr [rbx + rbp]
    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_plus
    cmp al, 0x2d
    je cmd_minus
    cmp al, 0x2e
    je cmd_dot
    cmp al, 0x2c
    je cmd_comma
    cmp al, 0x5b
    je cmd_open
    cmp al, 0x5d
    je cmd_close
    inc rbp
    jmp exec_loop
cmd_right:
    inc r14
    inc rbp
    jmp exec_loop
cmd_left:
    dec r14
    inc rbp
    jmp exec_loop
cmd_plus:
    inc byte ptr [r8 + r14]
    inc rbp
    jmp exec_loop
cmd_minus:
    dec byte ptr [r8 + r14]
    inc rbp
    jmp exec_loop
cmd_dot:
    mov rax, 1
    mov rdi, 1
    lea rsi, [r8 + r14]
    mov rdx, 1
    syscall
    inc rbp
    jmp exec_loop
cmd_comma:
    cmp r15, r12
    jae comma_zero
    movzx eax, byte ptr [rbx + r15]
    mov byte ptr [r8 + r14], al
    inc r15
    inc rbp
    jmp exec_loop
comma_zero:
    mov byte ptr [r8 + r14], 0
    inc rbp
    jmp exec_loop
cmd_open:
    cmp byte ptr [r8 + r14], 0
    jne open_next
    mov ebp, dword ptr [r9 + rbp*4]
    inc rbp
    jmp exec_loop
open_next:
    inc rbp
    jmp exec_loop
cmd_close:
    cmp byte ptr [r8 + r14], 0
    je close_next
    mov ebp, dword ptr [r9 + rbp*4]
    inc rbp
    jmp exec_loop
close_next:
    inc rbp
    jmp exec_loop
success:
    mov rax, 60
    xor edi, edi
    syscall
unbalanced:
    mov rax, 60
    mov edi, 1
    syscall
.bss
.balign 16
buffer:
    .zero 65536
.balign 16
tape:
    .zero 30000
.balign 16
match_table:
    .zero 262144
.balign 16
bstack:
    .zero 4096
