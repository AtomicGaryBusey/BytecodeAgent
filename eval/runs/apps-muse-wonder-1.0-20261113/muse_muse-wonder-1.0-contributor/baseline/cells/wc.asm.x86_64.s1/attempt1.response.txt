.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rdx, rax
    xor ecx, ecx
count_loop:
    cmp rcx, rdx
    jae read_loop
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 10
    jne check_ws
    inc r12
check_ws:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    ja is_word
is_ws:
    xor ebx, ebx
    inc rcx
    jmp count_loop
is_word:
    test ebx, ebx
    jnz word_cont
    inc r13
    mov ebx, 1
word_cont:
    inc rcx
    jmp count_loop
done:
    lea rdi, [rip + outbuf + 64]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call convert_num
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call convert_num
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call convert_num
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
convert_num:
    test rax, rax
    jnz 1f
    dec rdi
    mov byte ptr [rdi], 48
    ret
1:
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    ret
.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 64
