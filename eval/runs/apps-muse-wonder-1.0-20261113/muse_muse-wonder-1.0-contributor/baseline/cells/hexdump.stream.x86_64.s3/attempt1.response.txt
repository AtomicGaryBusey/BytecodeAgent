.globl _start
.text
_start:
    lea r13, [rip + input_buf]
    mov r14, r13
read_loop:
    mov rdx, r13
    add rdx, 65536
    sub rdx, r14
    test rdx, rdx
    jz read_done
    mov eax, 0
    mov edi, 0
    mov rsi, r14
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r14, rax
    jmp read_loop
read_done:
    mov r12, r14
    sub r12, r13
    xor ebx, ebx
row_loop:
    cmp rbx, r12
    jae final_line
    mov rax, r12
    sub rax, rbx
    cmp rax, 16
    jbe 1f
    mov eax, 16
1:
    mov r14, rax
    lea rdi, [rip + line_buf]
    mov eax, ebx
    call append_hex8
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    xor r15d, r15d
hex_slot_loop:
    cmp r15, r14
    jae slot_blank
    lea rax, [r13 + rbx]
    movzx eax, byte ptr [rax + r15]
    call append_hex2
    mov byte ptr [rdi], 32
    inc rdi
    jmp slot_after
slot_blank:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
slot_after:
    cmp r15, 7
    jne slot_noextra
    mov byte ptr [rdi], 32
    inc rdi
slot_noextra:
    inc r15
    cmp r15, 16
    jb hex_slot_loop
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    xor r15d, r15d
ascii_loop:
    cmp r15, r14
    jae ascii_done
    lea rax, [r13 + rbx]
    movzx eax, byte ptr [rax + r15]
    cmp eax, 32
    jb ascii_dot
    cmp eax, 126
    jbe ascii_store
ascii_dot:
    mov eax, 46
ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc r15
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add rbx, r14
    jmp row_loop
final_line:
    lea rdi, [rip + line_buf]
    mov eax, r12d
    call append_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
append_hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    lea rsi, [rip + hexdigits]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret
append_hex2:
    mov edx, eax
    shr edx, 4
    and edx, 15
    lea rsi, [rip + hexdigits]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    ret
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 128
