.globl _start
.text
_start:
lea rbx, [rip + buffer]
mov r12, rbx
read_loop:
lea rax, [rbx + 65536]
sub rax, r12
jz read_done
mov rdx, rax
mov rax, 0
mov rdi, 0
mov rsi, r12
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
mov rsi, rbx
process_loop:
cmp rsi, r12
jae exit_program
mov rdi, rsi
scan_loop:
cmp rdi, r12
jae scan_at_end
cmp byte ptr [rdi], 10
je scan_newline
inc rdi
jmp scan_loop
scan_newline:
mov r14, rdi
lea r15, [rdi + 1]
jmp reverse_line
scan_at_end:
mov r14, r12
mov r15, r12
reverse_line:
mov r8, rsi
mov r9, r14
dec r9
reverse_loop:
cmp r8, r9
jae reverse_done
mov al, byte ptr [r8]
mov cl, byte ptr [r9]
mov byte ptr [r8], cl
mov byte ptr [r9], al
inc r8
dec r9
jmp reverse_loop
reverse_done:
mov r8, rsi
mov r9, r14
sub r9, rsi
write_line_loop:
test r9, r9
jz write_newline
mov rax, 1
mov rdi, 1
mov rsi, r8
mov rdx, r9
syscall
add r8, rax
sub r9, rax
jmp write_line_loop
write_newline:
mov rax, 1
mov rdi, 1
lea rsi, [rip + newline_byte]
mov rdx, 1
syscall
mov rsi, r15
jmp process_loop
exit_program:
mov rax, 60
xor rdi, rdi
syscall
.section .rodata
newline_byte:
.byte 10
.bss
.balign 16
buffer:
.zero 65536
