.globl _start
.text
_start:
lea rbx, [rip + buf]
mov r12, rbx
read_loop:
lea rax, [rip + buf]
add rax, 65536
sub rax, r12
jz read_done
mov rdx, rax
xor eax, eax
xor edi, edi
mov rsi, r12
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
mov r13, rbx
line_loop:
cmp r13, r12
jae done
mov r14, r13
scan_loop:
cmp r14, r12
jae scan_done
cmp byte ptr [r14], 10
je scan_done
inc r14
jmp scan_loop
scan_done:
mov rsi, r13
mov rdi, r14
cmp rsi, rdi
jae rev_done
dec rdi
rev_loop:
cmp rsi, rdi
jae rev_done
mov al, byte ptr [rsi]
mov cl, byte ptr [rdi]
mov byte ptr [rsi], cl
mov byte ptr [rdi], al
inc rsi
dec rdi
jmp rev_loop
rev_done:
mov rsi, r13
mov rdx, r14
sub rdx, r13
call write_all
lea rsi, [rip + nl]
mov edx, 1
call write_all
lea r13, [r14 + 1]
jmp line_loop
done:
mov eax, 60
xor edi, edi
syscall
write_all:
test rdx, rdx
jz wa_ret
wa_loop:
mov eax, 1
mov edi, 1
syscall
test rax, rax
jle wa_ret
add rsi, rax
sub rdx, rax
jnz wa_loop
wa_ret:
ret
.section .rodata
nl:
.byte 10
.bss
.balign 16
buf:
.zero 65536
