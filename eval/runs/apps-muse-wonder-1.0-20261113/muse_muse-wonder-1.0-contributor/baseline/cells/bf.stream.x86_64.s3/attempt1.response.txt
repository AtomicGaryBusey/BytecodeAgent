.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], r12
    xor ecx, ecx
find_loop:
    cmp rcx, r12
    jae find_notfound
    cmp byte ptr [rbx + rcx], 33
    je find_found
    inc rcx
    jmp find_loop
find_notfound:
    mov qword ptr [rip + prog_len], r12
    mov qword ptr [rip + inp_pos], r12
    jmp validate_init
find_found:
    mov qword ptr [rip + prog_len], rcx
    lea rax, [rcx + 1]
    mov qword ptr [rip + inp_pos], rax
validate_init:
    xor ecx, ecx
    xor r13d, r13d
validate_loop:
    cmp rcx, qword ptr [rip + prog_len]
    jae validate_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    inc rcx
    jmp validate_loop
bracket_open:
    push rcx
    inc r13
    inc rcx
    jmp validate_loop
bracket_close:
    test r13, r13
    jz fail_exit
    pop rax
    dec r13
    lea rdx, [rip + match_table]
    mov dword ptr [rdx + rcx*4], eax
    mov dword ptr [rdx + rax*4], ecx
    inc rcx
    jmp validate_loop
validate_done:
    test r13, r13
    jnz fail_exit
    lea r14, [rip + tape]
    xor r15d, r15d
    xor r12d, r12d
    mov r13, qword ptr [rip + inp_pos]
exec_loop:
    cmp r12, qword ptr [rip + prog_len]
    jae success_exit
    movzx eax, byte ptr [rbx + r12]
    cmp al, 62
    je cmd_gt
    cmp al, 60
    je cmd_lt
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comma
    cmp al, 91
    je cmd_lb
    cmp al, 93
    je cmd_rb
    inc r12
    jmp exec_loop
cmd_gt:
    inc r15
    inc r12
    jmp exec_loop
cmd_lt:
    dec r15
    inc r12
    jmp exec_loop
cmd_plus:
    inc byte ptr [r14 + r15]
    inc r12
    jmp exec_loop
cmd_minus:
    dec byte ptr [r14 + r15]
    inc r12
    jmp exec_loop
cmd_dot:
    lea rsi, [r14 + r15]
    mov eax, 1
    mov edi, 1
    mov edx, 1
    syscall
    inc r12
    jmp exec_loop
cmd_comma:
    cmp r13, qword ptr [rip + input_len]
    jae cmd_comma_zero
    mov al, byte ptr [rbx + r13]
    mov byte ptr [r14 + r15], al
    inc r13
    inc r12
    jmp exec_loop
cmd_comma_zero:
    mov byte ptr [r14 + r15], 0
    inc r12
    jmp exec_loop
cmd_lb:
    cmp byte ptr [r14 + r15], 0
    jne cmd_inc_pc
    lea rdx, [rip + match_table]
    mov eax, dword ptr [rdx + r12*4]
    lea r12, [rax + 1]
    jmp exec_loop
cmd_rb:
    cmp byte ptr [r14 + r15], 0
    je cmd_inc_pc
    lea rdx, [rip + match_table]
    mov eax, dword ptr [rdx + r12*4]
    lea r12, [rax + 1]
    jmp exec_loop
cmd_inc_pc:
    inc r12
    jmp exec_loop
success_exit:
    mov eax, 60
    xor edi, edi
    syscall
fail_exit:
    mov eax, 60
    mov edi, 1
    syscall
.bss
.balign 8

prog_len: .zero 8
inp_pos: .zero 8
.balign 8

.balign 8
