.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jle done_read
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buffer]
count_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je is_space
    mov r8d, eax
    sub r8d, 9
    cmp r8d, 4
    jbe is_space
    test r15, r15
    jne next_byte
    inc r13
    mov r15, 1
    jmp next_byte
is_space:
    xor r15d, r15d
next_byte:
    inc rsi
    dec rcx
    jmp count_loop
done_read:
    lea rbx, [rip + outbuf]
    mov rax, r12
    call append_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r13
    call append_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r14
    call append_num
    mov byte ptr [rbx], 10
    inc rbx
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
append_num:
    lea rsi, [rip + numbuf]
    add rsi, 32
    mov r9, rsi
    mov r8, 10
1:
    xor edx, edx
    div r8
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jne 1b
    mov rdi, rbx
2:
    cmp rsi, r9
    jae 3f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp 2b
3:
    mov rbx, rdi
    ret
.bss
buffer:
    .zero 8192
outbuf:
    .zero 64
numbuf:
    .zero 32
