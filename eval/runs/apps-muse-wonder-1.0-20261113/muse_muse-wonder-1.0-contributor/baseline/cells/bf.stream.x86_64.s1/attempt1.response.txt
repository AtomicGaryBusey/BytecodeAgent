.globl _start
.text
_start:
    mov rbp, rsp
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbx, [rip + input_buf]
    xor ecx, ecx
find_loop:
    cmp rcx, r12
    jae not_found
    cmp byte ptr [rbx + rcx], 33
    je found_bang
    inc rcx
    jmp find_loop
not_found:
    mov r13, r12
    mov r14, r12
    jmp preprocess_start
found_bang:
    mov r13, rcx
    inc r13
    mov r14, r12
    mov r12, rcx
preprocess_start:
    xor r8d, r8d
    xor r9d, r9d
preprocess_loop:
    cmp r8, r12
    jae preprocess_done
    movzx eax, byte ptr [rbx + r8]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je append_command
    cmp al, 60
    je append_command
    cmp al, 43
    je append_command
    cmp al, 45
    je append_command
    cmp al, 46
    je append_command
    cmp al, 44
    je append_command
    inc r8
    jmp preprocess_loop
append_command:
    lea rdx, [rip + cmd_buf]
    mov byte ptr [rdx + r9], al
    inc r9
    inc r8
    jmp preprocess_loop
open_bracket:
    lea rdx, [rip + cmd_buf]
    mov byte ptr [rdx + r9], al
    push r9
    inc r9
    inc r8
    jmp preprocess_loop
close_bracket:
    cmp rsp, rbp
    je fail_exit
    pop rcx
    lea rdx, [rip + cmd_buf]
    mov byte ptr [rdx + r9], al
    lea rdx, [rip + match_table]
    mov dword ptr [rdx + r9*4], ecx
    mov dword ptr [rdx + rcx*4], r9d
    inc r9
    inc r8
    jmp preprocess_loop
preprocess_done:
    cmp rsp, rbp
    jne fail_exit
    mov r12, r9
    lea rbp, [rip + cmd_buf]
    lea r15, [rip + tape]
    xor r8d, r8d
    xor r10d, r10d
run_loop:
    cmp r8, r12
    jae success_exit
    movzx eax, byte ptr [rbp + r8]
    cmp al, 43
    je do_plus
    cmp al, 45
    je do_minus
    cmp al, 62
    je do_right
    cmp al, 60
    je do_left
    cmp al, 91
    je do_open
    cmp al, 93
    je do_close
    cmp al, 46
    je do_dot
    cmp al, 44
    je do_comma
    inc r8
    jmp run_loop
do_plus:
    inc byte ptr [r15 + r10]
    inc r8
    jmp run_loop
do_minus:
    dec byte ptr [r15 + r10]
    inc r8
    jmp run_loop
do_right:
    inc r10
    inc r8
    jmp run_loop
do_left:
    dec r10
    inc r8
    jmp run_loop
do_open:
    cmp byte ptr [r15 + r10], 0
    jne next_instruction
    lea rax, [rip + match_table]
    mov r8d, dword ptr [rax + r8*4]
    inc r8
    jmp run_loop
next_instruction:
    inc r8
    jmp run_loop
do_close:
    cmp byte ptr [r15 + r10], 0
    je next_instruction
    lea rax, [rip + match_table]
    mov r8d, dword ptr [rax + r8*4]
    inc r8
    jmp run_loop
do_dot:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + r10]
    mov edx, 1
    syscall
    inc r8
    jmp run_loop
do_comma:
    cmp r13, r14
    jae comma_eof
    movzx eax, byte ptr [rbx + r13]
    mov byte ptr [r15 + r10], al
    inc r13
    inc r8
    jmp run_loop
comma_eof:
    mov byte ptr [r15 + r10], 0
    inc r8
    jmp run_loop
success_exit:
    mov eax, 60
    xor edi, edi
    syscall
fail_exit:
    mov eax, 60
    mov edi, 1
    syscall
.bss
.balign 8
input_buf:
    .zero 65536
cmd_buf:
    .zero 65536
tape:
    .zero 30000
.balign 8
match_table:
    .zero 262144
