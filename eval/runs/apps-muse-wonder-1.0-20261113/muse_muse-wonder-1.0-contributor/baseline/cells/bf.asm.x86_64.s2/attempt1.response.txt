.text
.globl _start
_start:
    lea r8, [rip + input_buf]
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    mov rsi, r8
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_loop:
    cmp rcx, r12
    jae find_notfound
    movzx eax, byte ptr [r8 + rcx]
    cmp al, 33
    je find_found
    inc rcx
    jmp find_loop
find_found:
    mov r13, rcx
    lea r14, [rcx + 1]
    jmp bracket_init
find_notfound:
    mov r13, r12
    mov r14, r12
bracket_init:
    lea r10, [rip + jump_table]
    lea r11, [rip + bracket_stack]
    xor ebx, ebx
    xor ebp, ebp
bracket_loop:
    cmp rbx, r13
    jae bracket_check
    movzx eax, byte ptr [r8 + rbx]
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    jmp bracket_next
bracket_open:
    mov dword ptr [r11 + rbp*4], ebx
    inc rbp
    jmp bracket_next
bracket_close:
    test rbp, rbp
    jz unbalanced
    dec rbp
    mov ecx, dword ptr [r11 + rbp*4]
    mov dword ptr [r10 + rbx*4], ecx
    mov dword ptr [r10 + rcx*4], ebx
bracket_next:
    inc rbx
    jmp bracket_loop
bracket_check:
    test rbp, rbp
    jnz unbalanced
    lea r9, [rip + tape]
    xor ebx, ebx
    xor ebp, ebp
exec_loop:
    cmp rbx, r13
    jae exec_done
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp exec_loop
cmd_plus:
    inc byte ptr [r9 + rbp]
    jmp exec_loop
cmd_minus:
    dec byte ptr [r9 + rbp]
    jmp exec_loop
cmd_right:
    inc rbp
    jmp exec_loop
cmd_left:
    dec rbp
    jmp exec_loop
cmd_output:
    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + rbp]
    mov edx, 1
    syscall
    jmp exec_loop
cmd_input:
    cmp r14, r12
    jae cmd_input_eof
    movzx eax, byte ptr [r8 + r14]
    inc r14
    mov byte ptr [r9 + rbp], al
    jmp exec_loop
cmd_input_eof:
    mov byte ptr [r9 + rbp], 0
    jmp exec_loop
cmd_open:
    cmp byte ptr [r9 + rbp], 0
    jne exec_loop
    mov ecx, ebx
    dec ecx
    mov ebx, dword ptr [r10 + rcx*4]
    inc rbx
    jmp exec_loop
cmd_close:
    cmp byte ptr [r9 + rbp], 0
    je exec_loop
    mov ecx, ebx
    dec ecx
    mov ebx, dword ptr [r10 + rcx*4]
    inc rbx
    jmp exec_loop
exec_done:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
.bss
.balign 8
input_buf:
    .zero 65536
tape:
    .zero 30000
jump_table:
    .zero 262144
bracket_stack:
    .zero 262144
