xor edi, edi

jae line_finish

cmp al, 32

cmp al, 9

jae token_ready
movzx eax, byte ptr [rbp]
cmp al, 10
je token_ready
cmp al, 32
je token_ready
cmp al, 9
je token_ready
inc rbp

skip_sep:
inc rbp
jmp token_loop
newline_line:
inc rbp
jmp line_finish
token_ready:
mov r9, rbp
sub r9, r8
cmp r15, 0
jne token_loop
cmp r9, 1
jne parse_literal
movzx eax, byte ptr [r8]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47

cmp al, 37

xor r11d, r11d
xor ecx, ecx

cmp al, 45

mov r11d, 1
mov r10d, 1

sub rax, r10
cmp rax, 1
jb set_error
cmp rax, 18
ja set_error

jae literal_done

cmp al, 48
jb set_error
cmp al, 57
ja set_error
sub eax, 48
imul rcx, rcx, 10
add rcx, rax
inc r10
jmp digit_loop
literal_done:
test r11, r11
jz push_val
neg rcx

mov qword ptr [rbx + r12*8], rcx
inc r12
jmp token_loop
set_error:

jmp token_loop
do_operator:
mov r10d, eax
cmp r12, 2
jb set_error
dec r12
mov rcx, qword ptr [rbx + r12*8]

cmp r10b, 45
je op_sub
cmp r10b, 42
je op_mul

idiv rcx
cmp r10b, 37

jmp store_result
op_add:
add rax, rcx
jmp store_result
op_sub:
sub rax, rcx
jmp store_result
op_mul:

store_result:
mov qword ptr [rbx + r12*8 - 8], rax
jmp token_loop

jne print_error
cmp r12, 1
jne print_error

jz write_done

mov edi, 1
syscall

jle write_done
add rsi, rax
sub rdx, rax
jnz write_loop
write_done:
