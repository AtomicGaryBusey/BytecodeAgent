.text
.globl _start
_start:
    xor ebx, ebx
    lea rbp, [rip + buffer]
read_loop:
    cmp rbx, 32768
    jae read_done
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov edx, 32768
    sub edx, ebx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
parse_loop:
    cmp r8, rbx
    jae parse_final
    movzx eax, byte ptr [rbp + r8]
    cmp al, 10
    je parse_newline
    inc r8
    jmp parse_loop
parse_newline:
    lea rax, [rip + line_off]
    mov dword ptr [rax + r10*4], r9d
    mov rdx, r8
    sub rdx, r9
    lea rax, [rip + line_len]
    mov dword ptr [rax + r10*4], edx
    inc r10
    inc r8
    mov r9, r8
    jmp parse_loop
parse_final:
    cmp r9, rbx
    jae parse_done
    lea rax, [rip + line_off]
    mov dword ptr [rax + r10*4], r9d
    mov rdx, rbx
    sub rdx, r9
    lea rax, [rip + line_len]
    mov dword ptr [rax + r10*4], edx
    inc r10
parse_done:
    mov rbx, r10
    mov r12d, 1
sort_outer:
    cmp r12, rbx
    jae sort_done
    lea rax, [rip + line_off]
    mov r14d, dword ptr [rax + r12*4]
    lea rax, [rip + line_len]
    mov r15d, dword ptr [rax + r12*4]
    lea r13, [r12 - 1]
sort_inner:
    cmp r13, 0
    jl sort_insert
    lea rax, [rip + line_off]
    mov edi, dword ptr [rax + r13*4]
    lea rax, [rip + line_len]
    mov esi, dword ptr [rax + r13*4]
    mov rdx, r14
    mov rcx, r15
    call compare_lines
    test eax, eax
    jle sort_insert
    lea rax, [rip + line_off]
    mov edi, dword ptr [rax + r13*4]
    mov dword ptr [rax + r13*4 + 4], edi
    lea rax, [rip + line_len]
    mov esi, dword ptr [rax + r13*4]
    mov dword ptr [rax + r13*4 + 4], esi
    dec r13
    jmp sort_inner
sort_insert:
    lea rax, [rip + line_off]
    mov dword ptr [rax + r13*4 + 4], r14d
    lea rax, [rip + line_len]
    mov dword ptr [rax + r13*4 + 4], r15d
    inc r12
    jmp sort_outer
sort_done:
    xor r12d, r12d
output_loop:
    cmp r12, rbx
    jae program_exit
    lea rax, [rip + line_off]
    mov r13d, dword ptr [rax + r12*4]
    lea rax, [rip + line_len]
    mov r14d, dword ptr [rax + r12*4]
    lea rsi, [rip + buffer]
    add rsi, r13
    mov rdx, r14
write_line_loop:
    test rdx, rdx
    jz write_newline
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jz write_newline
    js write_newline
    add rsi, rax
    sub rdx, rax
    jmp write_line_loop
write_newline:
    lea rsi, [rip + newline_byte]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    inc r12
    jmp output_loop
program_exit:
    xor edi, edi
    mov eax, 60
    syscall
compare_lines:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    lea r9, [rip + buffer]
    add r9, rdi
    lea r10, [rip + buffer]
    add r10, rdx
    xor r11d, r11d
compare_loop:
    cmp r11, r8
    jae compare_lengths
    movzx eax, byte ptr [r9 + r11]
    movzx edx, byte ptr [r10 + r11]
    cmp al, dl
    jb compare_less
    ja compare_greater
    inc r11
    jmp compare_loop
compare_lengths:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret
.section .rodata
newline_byte:
    .byte 10
.bss
.balign 8
buffer:
    .zero 32768
.balign 4
line_off:
    .zero 4000
line_len:
    .zero 4000
