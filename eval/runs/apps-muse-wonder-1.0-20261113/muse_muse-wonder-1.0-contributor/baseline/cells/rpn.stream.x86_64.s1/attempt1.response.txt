.globl _start
.text
_start:
lea rbx, [rip + input_buf]
xor r12d, r12d
read_loop:
mov rax, 0
mov rdi, 0
lea rsi, [rbx + r12]
mov rdx, 70000
sub rdx, r12
jz read_done
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
lea r14, [rip + out_buf]
xor r15d, r15d
xor r13d, r13d
line_loop:
cmp r13, r12
jae finish
mov rbp, r13
find_end:
cmp rbp, r12
jae found_end
movzx eax, byte ptr [rbx + rbp]
cmp al, 10
je found_end
inc rbp
jmp find_end
found_end:
mov rsi, r13
xor r8d, r8d
xor r9d, r9d
token_loop:
cmp rsi, rbp
jae after_tokens
movzx eax, byte ptr [rbx + rsi]
cmp al, 32
je sep_inc
cmp al, 9
je sep_inc
jmp token_found
sep_inc:
inc rsi
jmp token_loop
token_found:
mov rdx, rsi
scan_loop:
cmp rsi, rbp
jae token_scanned
movzx eax, byte ptr [rbx + rsi]
cmp al, 32
je token_scanned
cmp al, 9
je token_scanned
inc rsi
jmp scan_loop
token_scanned:
mov rcx, rsi
sub rcx, rdx
cmp rcx, 1
jne parse_literal
movzx eax, byte ptr [rbx + rdx]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator
parse_literal:
mov r10, rdx
xor r11d, r11d
movzx eax, byte ptr [rbx + r10]
cmp al, 45
jne digits_start
mov r11d, 1
inc r10
digits_start:
mov rcx, rsi
sub rcx, r10
test rcx, rcx
jz invalid_token
cmp rcx, 18
ja invalid_token
xor eax, eax
digit_loop:
cmp r10, rsi
jae digits_done
movzx edi, byte ptr [rbx + r10]
sub edi, 48
cmp edi, 9
ja invalid_token
imul rax, rax, 10
add rax, rdi
inc r10
jmp digit_loop
digits_done:
test r11, r11
jz push_value
neg rax
push_value:
lea rcx, [rip + val_stack]
mov qword ptr [rcx + r8*8], rax
inc r8
jmp token_loop
do_operator:
movzx r10d, al
cmp r8, 2
jb invalid_token
lea rcx, [rip + val_stack]
mov rax, qword ptr [rcx + r8*8 - 16]
mov rdx, qword ptr [rcx + r8*8 - 8]
cmp r10b, 43
je op_add
cmp r10b, 45
je op_sub
cmp r10b, 42
je op_mul
cmp r10b, 47
je op_divmod
cmp r10b, 37
je op_divmod
jmp invalid_token
op_add:
add rax, rdx
jmp op_store
op_sub:
sub rax, rdx
jmp op_store
op_mul:
imul rax, rdx
jmp op_store
op_divmod:
test rdx, rdx
jz invalid_token
mov rcx, 1
shl rcx, 63
cmp rax, rcx
jne div_ok
cmp rdx, -1
je invalid_token
div_ok:
mov rcx, rdx
cqo
idiv rcx
cmp r10b, 37
jne op_store
mov rax, rdx
jmp op_store
op_store:
dec r8
lea rcx, [rip + val_stack]
mov qword ptr [rcx + r8*8 - 8], rax
jmp token_loop
invalid_token:
mov r9d, 1
jmp emit_error
after_tokens:
cmp r9, 0
jne emit_error
cmp r8, 1
jne emit_error
lea rcx, [rip + val_stack]
mov rax, qword ptr [rcx]
test rax, rax
jns number_positive
mov byte ptr [r14 + r15], 45
inc r15
neg rax
number_positive:
lea r10, [rip + num_buf]
add r10, 32
mov rsi, r10
mov rcx, 10
num_digit_loop:
xor edx, edx
div rcx
add dl, 48
dec rsi
mov byte ptr [rsi], dl
test rax, rax
jnz num_digit_loop
copy_digits:
cmp rsi, r10
jae digits_copied
movzx eax, byte ptr [rsi]
mov byte ptr [r14 + r15], al
inc r15
inc rsi
jmp copy_digits
digits_copied:
mov byte ptr [r14 + r15], 10
inc r15
jmp next_line
emit_error:
lea rsi, [rip + error_msg]
xor edx, edx
error_copy:
cmp rdx, 6
jae next_line
movzx eax, byte ptr [rsi + rdx]
mov byte ptr [r14 + r15], al
inc r15
inc rdx
jmp error_copy
next_line:
mov r13, rbp
cmp rbp, r12
jae line_loop
inc r13
jmp line_loop
finish:
xor r13d, r13d
write_loop:
cmp r13, r15
jae exit_program
mov rax, 1
mov rdi, 1
lea rsi, [r14 + r13]
mov rdx, r15
sub rdx, r13
syscall
cmp rax, 0
jle exit_program
add r13, rax
jmp write_loop
exit_program:
mov rax, 60
xor edi, edi
syscall
.section .rodata
error_msg:
.ascii "error\n"
.bss
.align 16
input_buf:
.zero 70000
.align 16
out_buf:
.zero 524288
.align 16
val_stack:
.zero 8192
.align 16
num_buf:
.zero 32
