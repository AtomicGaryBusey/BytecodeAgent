_start:
    xor ebx, ebx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], rbx
    call skip_ws
    call parse_value
    cmp byte ptr [rip + err_flag], 0
    jne print_error
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne top_extra_error
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, 3
    syscall
    mov rax, 60
    xor edi, edi
    syscall
top_extra_error:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + err_pos], rax
    mov byte ptr [rip + err_flag], 1
    jmp print_error
print_error:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + str_error_at]
    mov ecx, 9
print_error_copy:
    test rcx, rcx
    jz print_error_copied
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp print_error_copy
print_error_copied:
    mov rax, qword ptr [rip + err_pos]
    lea rsi, [rip + digits_rev]
    xor r8, r8
    cmp rax, 0
    jne print_error_convert
    mov byte ptr [rsi], 48
    mov r8, 1
    jmp print_error_emit
print_error_convert:
    test rax, rax
    jz print_error_emit
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, 48
    mov byte ptr [rsi + r8], dl
    inc r8
    jmp print_error_convert
print_error_emit:
    test r8, r8
    jz print_error_newline
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    jmp print_error_emit
print_error_newline:
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
skip_ws:
skip_ws_loop:
    mov rcx, qword ptr [rip + pos]
    cmp rcx, qword ptr [rip + input_len]
    jae skip_ws_done
    lea rdx, [rip + input]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 32
    je skip_ws_inc
    cmp al, 9
    je skip_ws_inc
    cmp al, 10
    je skip_ws_inc
    cmp al, 13
    je skip_ws_inc
    ret
skip_ws_inc:
    inc rcx
    mov qword ptr [rip + pos], rcx
    jmp skip_ws_loop
skip_ws_done:
    ret
parse_value:
    cmp byte ptr [rip + err_flag], 0
    jne parse_value_ret
    mov rcx, qword ptr [rip + pos]
    cmp rcx, qword ptr [rip + input_len]
    jae parse_value_error
    lea rdx, [rip + input]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je parse_value_true
    cmp al, 102
    je parse_value_false
    cmp al, 110
    je parse_value_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb parse_value_error
    cmp al, 57
    ja parse_value_error
    jmp parse_number
parse_value_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp expect_literal
parse_value_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp expect_literal
parse_value_null:

    mov ecx, 4
    jmp expect_literal
parse_value_error:

    mov qword ptr [rip + err_pos], rax
    mov byte ptr [rip + err_flag], 1
parse_value_ret:
    ret

expect_literal_loop:
    cmp r8, rcx
    jae expect_literal_done
    mov r9, qword ptr [rip + pos]
    cmp r9, qword ptr [rip + input_len]
    jae expect_literal_error
    lea rdx, [rip + input]
    movzx eax, byte ptr [rdx + r9]
    movzx edx, byte ptr [rsi + r8]
    cmp al, dl
    jne expect_literal_error
    inc r9
    mov qword ptr [rip + pos], r9
    inc r8
    jmp expect_literal_loop
expect_literal_done:
    ret
expect_literal_error:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + err_pos], rax
    mov byte ptr [rip + err_flag], 1
    ret
