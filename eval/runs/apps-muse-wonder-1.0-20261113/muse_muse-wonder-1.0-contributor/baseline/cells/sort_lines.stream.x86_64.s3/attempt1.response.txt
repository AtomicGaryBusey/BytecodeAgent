.text
.globl _start
_start:
    xor r12, r12
read_loop:
    cmp r12, 32768
    jae parse_init
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_init
    js parse_init
    add r12, rax
    jmp read_loop
parse_init:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    test r12, r12
    jz sort_init
parse_loop:
    cmp r15, r12
    jae parse_tail
    lea rcx, [rip + input_buf]
    movzx eax, byte ptr [rcx + r15]
    cmp al, 10
    jne parse_next
    lea rcx, [rip + line_off]
    mov dword ptr [rcx + r13*4], r14d
    mov rax, r15
    sub rax, r14
    lea rcx, [rip + line_len]
    mov dword ptr [rcx + r13*4], eax
    inc r13
    lea r14, [r15 + 1]
parse_next:
    inc r15
    jmp parse_loop
parse_tail:
    cmp r14, r12
    jae sort_init
    lea rcx, [rip + line_off]
    mov dword ptr [rcx + r13*4], r14d
    mov rax, r12
    sub rax, r14
    lea rcx, [rip + line_len]
    mov dword ptr [rcx + r13*4], eax
    inc r13
sort_init:
    cmp r13, 2
    jl output_init
    mov rbx, 1
sort_outer:
    cmp rbx, r13
    jae output_init
    lea rcx, [rip + line_off]
    mov r14d, dword ptr [rcx + rbx*4]
    lea rcx, [rip + line_len]
    mov r15d, dword ptr [rcx + rbx*4]
    lea rbp, [rbx - 1]
sort_inner:
    cmp rbp, 0
    jl sort_insert
    lea rax, [rip + line_off]
    mov eax, dword ptr [rax + rbp*4]
    lea rdi, [rip + input_buf]
    add rdi, rax
    lea rax, [rip + line_len]
    mov esi, dword ptr [rax + rbp*4]
    lea rdx, [rip + input_buf]
    mov eax, r14d
    add rdx, rax
    mov ecx, r15d
    call compare_lines
    cmp eax, 0
    jle sort_insert
    lea rax, [rip + line_off]
    mov ecx, dword ptr [rax + rbp*4]
    mov dword ptr [rax + rbp*4 + 4], ecx
    lea rax, [rip + line_len]
    mov ecx, dword ptr [rax + rbp*4]
    mov dword ptr [rax + rbp*4 + 4], ecx
    dec rbp
    jmp sort_inner
sort_insert:
    lea rax, [rip + line_off]
    mov dword ptr [rax + rbp*4 + 4], r14d
    lea rax, [rip + line_len]
    mov dword ptr [rax + rbp*4 + 4], r15d
    inc rbx
    jmp sort_outer
output_init:
    xor rbx, rbx
output_loop:
    cmp rbx, r13
    jae program_exit
    lea rax, [rip + line_len]
    mov r15d, dword ptr [rax + rbx*4]
    lea rax, [rip + line_off]
    mov eax, dword ptr [rax + rbx*4]
    lea rsi, [rip + input_buf]
    add rsi, rax
    test r15d, r15d
    jz output_newline
    mov eax, 1
    mov edi, 1
    mov edx, r15d
    syscall
output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall
    inc rbx
    jmp output_loop
program_exit:
    mov eax, 60
    xor edi, edi
    syscall
compare_lines:
    xor r8, r8
    mov r9, rsi
    cmp r9, rcx
    cmova r9, rcx
cmp_loop:
    cmp r8, r9
    jae cmp_prefix
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp eax, r10d
    jb cmp_less
    ja cmp_greater
    inc r8
    jmp cmp_loop
cmp_prefix:
    cmp rsi, rcx
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret
.section .rodata
newline_byte:
    .byte 10
.bss
.balign 16
input_buf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000
