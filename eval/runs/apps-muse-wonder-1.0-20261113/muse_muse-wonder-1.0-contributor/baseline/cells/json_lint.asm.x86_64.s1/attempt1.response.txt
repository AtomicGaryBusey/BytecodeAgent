.globl _start
.text
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
    xor ebx, ebx
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    mov eax, 0
    mov edi, 0
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
skip_ws:
skip_ws_loop:
    cmp rbx, r13
    jae skip_ws_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x20
    je skip_ws_next
    cmp al, 0x09
    je skip_ws_next
    cmp al, 0x0A
    je skip_ws_next
    cmp al, 0x0D
    je skip_ws_next
    ret
skip_ws_next:
    inc rbx
    jmp skip_ws_loop
skip_ws_done:
    ret
parse_value:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    jmp parse_number
parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x7D
    jne object_member
    inc rbx
    ret
object_member:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x3A
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x2C
    je object_comma
    cmp al, 0x7D
    jne fail
    inc rbx
    ret
object_comma:
    inc rbx
    call skip_ws
    jmp object_member
parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x5D
    jne array_element
    inc rbx
    ret
array_element:
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x2C
    je array_comma
    cmp al, 0x5D
    jne fail
    inc rbx
    ret
array_comma:
    inc rbx
    call skip_ws
    jmp array_element
parse_string:
    inc rbx
string_loop:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x22
    je string_done
    cmp al, 0x5C
    je string_escape
    cmp al, 0x20
    jb fail
    inc rbx
    jmp string_loop
string_done:
    inc rbx
    ret
string_escape:
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x22
    je string_escape_simple
    cmp al, 0x5C
    je string_escape_simple
    cmp al, 0x2F
    je string_escape_simple
    cmp al, 0x62
    je string_escape_simple
    cmp al, 0x66
    je string_escape_simple
    cmp al, 0x6E
    je string_escape_simple
    cmp al, 0x72
    je string_escape_simple
    cmp al, 0x74
    je string_escape_simple
    cmp al, 0x75
    je string_unicode
    jmp fail
string_escape_simple:
    inc rbx
    jmp string_loop
string_unicode:
    inc rbx
    mov r8, 4
string_unicode_loop:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe string_hex_ok
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    jbe string_hex_ok
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
string_hex_ok:
    inc rbx
    dec r8
    jnz string_unicode_loop
    jmp string_loop
parse_number:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x2D
    jne number_int
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
number_int:
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
number_digits:
    inc rbx
    cmp rbx, r13
    jae number_frac
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb number_frac
    cmp al, 0x39
    ja number_frac
    jmp number_digits
number_zero:
    inc rbx
number_frac:
    cmp rbx, r13
    jae number_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x2E
    jne number_exp
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
number_frac_digits:
    inc rbx
    cmp rbx, r13
    jae number_exp
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb number_exp
    cmp al, 0x39
    ja number_exp
    jmp number_frac_digits
number_exp:
    cmp rbx, r13
    jae number_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x65
    je number_exp_start
    cmp al, 0x45
    jne number_done
number_exp_start:
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x2B
    je number_exp_sign
    cmp al, 0x2D
    jne number_exp_digit
number_exp_sign:
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
number_exp_digit:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
number_exp_digits:
    inc rbx
    cmp rbx, r13
    jae number_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb number_done
    cmp al, 0x39
    ja number_done
    jmp number_exp_digits
number_done:
    ret
parse_true:
    lea rsi, [rip + lit_true]
    mov r8, 4
    jmp parse_literal
parse_false:
    lea rsi, [rip + lit_false]
    mov r8, 5
    jmp parse_literal
parse_null:
    lea rsi, [rip + lit_null]
    mov r8, 4
    jmp parse_literal
parse_literal:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r12 + rbx]
    mov cl, byte ptr [rsi]
    cmp al, cl
    jne fail
    inc rbx
    inc rsi
    dec r8
    jnz parse_literal
    ret
fail:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + err_prefix]
    mov ecx, 9
fail_copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz fail_copy_loop
    mov rax, rbx
    lea rsi, [rip + numbuf]
    xor r8d, r8d
    mov rcx, 10
fail_num_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    mov byte ptr [rsi + r8], dl
    inc r8
    test rax, rax
    jnz fail_num_loop
fail_emit_loop:
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    test r8, r8
    jnz fail_emit_loop
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
.bss
    .balign 16
buffer:
    .zero 65536
outbuf:
    .zero 64
numbuf:
    .zero 32
