.equ BUFFER_CAP, 65552

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov edx, BUFFER_CAP
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rcx, [rip + input_len]
    mov qword ptr [rcx], rbx
    xor eax, eax
    lea rcx, [rip + parse_pos]
    mov qword ptr [rcx], rax
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + parse_pos]
    mov rcx, qword ptr [rip + input_len]
    cmp rax, rcx
    jne fail_current
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    mov rax, qword ptr [rip + parse_pos]
    cmp rax, qword ptr [rip + input_len]
    jae skip_ws_done
    lea rcx, [rip + buffer]
    movzx edx, byte ptr [rcx + rax]
    cmp edx, 32
    je skip_ws_advance
    cmp edx, 9
    je skip_ws_advance
    cmp edx, 10
    je skip_ws_advance
    cmp edx, 13
    jne skip_ws_done
skip_ws_advance:
    inc rax
    mov qword ptr [rip + parse_pos], rax
    jmp skip_ws
skip_ws_done:
    ret

peek_byte:
    mov rax, qword ptr [rip + parse_pos]
    cmp rax, qword ptr [rip + input_len]
    jae peek_eof
    lea rcx, [rip + buffer]
    movzx eax, byte ptr [rcx + rax]
    ret
peek_eof:
    mov eax, 256
    ret

advance_pos:
    mov rax, qword ptr [rip + parse_pos]
    inc rax
    mov qword ptr [rip + parse_pos], rax
    ret

parse_value:
    call peek_byte
    cmp eax, 123
    je parse_object
    cmp eax, 91
    je parse_array
    cmp eax, 34
    je parse_string
    cmp eax, 116
    je parse_true
    cmp eax, 102
    je parse_false
    cmp eax, 110
    je parse_null
    cmp eax, 45
    je parse_number
    cmp eax, 48
    jb fail_current
    cmp eax, 57
    ja fail_current
    jmp parse_number
parse_true:
    lea rbx, [rip + lit_true]
    mov r12d, 4
    jmp expect_literal
parse_false:
    lea rbx, [rip + lit_false]
    mov r12d, 5
    jmp expect_literal
parse_null:
    lea rbx, [rip + lit_null]
    mov r12d, 4
    jmp expect_literal

expect_literal:
    test r12, r12
    jz expect_literal_done
    call peek_byte
    cmp al, byte ptr [rbx]
    jne fail_current
    call advance_pos
    inc rbx
    dec r12
    jmp expect_literal
expect_literal_done:
    ret

parse_string:
    call advance_pos
parse_string_loop:
    call peek_byte
    cmp eax, 256
    je fail_current
    cmp eax, 34
    je parse_string_end
    cmp eax, 92
    je parse_string_escape
    cmp eax, 32
    jb fail_current
    call advance_pos
    jmp parse_string_loop
parse_string_end:
    call advance_pos
    ret
parse_string_escape:
    call advance_pos
    call peek_byte
    cmp eax, 34
    je parse_string_escape_simple
    cmp eax, 92
    je parse_string_escape_simple
    cmp eax, 47
    je parse_string_escape_simple
    cmp eax, 98
    je parse_string_escape_simple
    cmp eax, 102
    je parse_string_escape_simple
    cmp eax, 110
    je parse_string_escape_simple
    cmp eax, 114
    je parse_string_escape_simple
    cmp eax, 116
    je parse_string_escape_simple
    cmp eax, 117
    je parse_string_unicode
    jmp fail_current
parse_string_escape_simple:
    call advance_pos
    jmp parse_string_loop
parse_string_unicode:
    call advance_pos
    mov ebx, 4
parse_string_hex_loop:
    call peek_byte
    cmp eax, 48
    jb fail_current
    cmp eax, 57
    jbe parse_string_hex_ok
    cmp eax, 65
    jb fail_current
    cmp eax, 70
    jbe parse_string_hex_ok
    cmp eax, 97
    jb fail_current
    cmp eax, 102
    ja fail_current
parse_string_hex_ok:
    call advance_pos
    dec ebx
    jnz parse_string_hex_loop
    jmp parse_string_loop

parse_number:
    call peek_byte
    cmp eax, 45
    jne parse_number_int
    call advance_pos
    call peek_byte
    cmp eax, 48
    jb fail_current
    cmp eax, 57
    ja fail_current
parse_number_int:
    call peek_byte
    cmp eax, 48
    jne parse_number_int_loop
    call advance_pos
    jmp parse_number_frac
parse_number_int_loop:
    call peek_byte
    cmp eax, 48
    jb parse_number_frac
    cmp eax, 57
    ja parse_number_frac
    call advance_pos
    jmp parse_number_int_loop
parse_number_frac:
    call peek_byte
    cmp eax, 46
    jne parse_number_exp
    call advance_pos
    call peek_byte
    cmp eax, 48
    jb fail_current
    cmp eax, 57
    ja fail_current
parse_number_frac_loop:
    call peek_byte
    cmp eax, 48
    jb parse_number_exp
    cmp eax, 57
    ja parse_number_exp
    call advance_pos
    jmp parse_number_frac_loop
parse_number_exp:
    call peek_byte
    cmp eax, 101
    je parse_number_exp_start
    cmp eax, 69
    jne parse_number_done
parse_number_exp_start:
    call advance_pos
    call peek_byte
    cmp eax, 43
    je parse_number_exp_sign
    cmp eax, 45
    je parse_number_exp_sign
    jmp parse_number_exp_digit_check
parse_number_exp_sign:
    call advance_pos
parse_number_exp_digit_check:
    call peek_byte
    cmp eax, 48
    jb fail_current
    cmp eax, 57
    ja fail_current
parse_number_exp_loop:
    call peek_byte
    cmp eax, 48
    jb parse_number_done
    cmp eax, 57
    ja parse_number_done
    call advance_pos
    jmp parse_number_exp_loop
parse_number_done:
    ret

parse_object:
    call advance_pos
    call skip_ws
    call peek_byte
    cmp eax, 125
    je parse_object_close
parse_object_member:
    call peek_byte
    cmp eax, 34
    jne fail_current
    call parse_string
    call skip_ws
    call peek_byte
    cmp eax, 58
    jne fail_current
    call advance_pos
    call skip_ws
    call parse_value
    call skip_ws
    call peek_byte
    cmp eax, 44
    je parse_object_comma
    cmp eax, 125
    je parse_object_close
    jmp fail_current
parse_object_comma:
    call advance_pos
    call skip_ws
    jmp parse_object_member
parse_object_close:
    call advance_pos
    ret

parse_array:
    call advance_pos
    call skip_ws
    call peek_byte
    cmp eax, 93
    je parse_array_close
parse_array_loop:
    call parse_value
    call skip_ws
    call peek_byte
    cmp eax, 44
    je parse_array_comma
    cmp eax, 93
    je parse_array_close
    jmp fail_current
parse_array_comma:
    call advance_pos
    call skip_ws
    jmp parse_array_loop
parse_array_close:
    call advance_pos
    ret

fail_current:
    mov rax, qword ptr [rip + parse_pos]
    jmp print_error

print_error:
    sub rsp, 80
    mov r8, rax
    lea rsi, [rip + err_prefix]
    mov rdi, rsp
    mov ecx, 9
    rep movsb
    mov rbx, rdi
    mov rax, r8
    test rax, rax
    jnz print_error_convert
    mov byte ptr [rbx], 48
    inc rbx
    jmp print_error_newline
print_error_convert:
    lea r8, [rsp + 80]
print_error_convert_loop:
    test rax, rax
    jz print_error_copy_digits
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    jmp print_error_convert_loop
print_error_copy_digits:
    lea rcx, [rsp + 80]
print_error_copy_loop:
    cmp r8, rcx
    jae print_error_newline
    mov al, byte ptr [r8]
    mov byte ptr [rbx], al
    inc r8
    inc rbx
    jmp print_error_copy_loop
print_error_newline:
    mov byte ptr [rbx], 10
    inc rbx
    mov rdx, rbx
    sub rdx, rsp
    mov rsi, rsp
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.align 8
buffer:
    .zero BUFFER_CAP
.align 8
input_len:
    .zero 8
parse_pos:
    .zero 8
