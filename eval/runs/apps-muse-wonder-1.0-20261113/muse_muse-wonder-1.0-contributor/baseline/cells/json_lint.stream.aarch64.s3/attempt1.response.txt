.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0
read_loop:
    mov x21, #65536
    sub x2, x21, x19
    cmp x2, #0
    b.eq read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt read_done
    add x19, x19, x0
    b read_loop
read_done:
    adrp x0, input_len
    str x19, [x0, :lo12:input_len]
    adrp x0, input_pos
    str xzr, [x0, :lo12:input_pos]
    bl skip_ws
    bl parse_value
    bl skip_ws
    adrp x0, input_pos
    ldr x1, [x0, :lo12:input_pos]
    adrp x0, input_len
    ldr x2, [x0, :lo12:input_len]
    cmp x1, x2
    b.ne error_exit
    b print_ok

print_ok:
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error_exit:
    adrp x0, input_pos
    ldr x19, [x0, :lo12:input_pos]
    mov x0, #1
    adrp x1, err_prefix
    add x1, x1, :lo12:err_prefix
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x10, digit_buf
    add x10, x10, :lo12:digit_buf
    add x10, x10, #32
    mov x11, x10
    cmp x19, #0
    b.ne error_convert_loop
    sub x10, x10, #1
    mov w12, #0x30
    strb w12, [x10]
    b error_write_digits
error_convert_loop:
    cmp x19, #0
    b.eq error_write_digits
    mov x12, #10
    udiv x13, x19, x12
    msub x14, x13, x12, x19
    add w14, w14, #0x30
    sub x10, x10, #1
    strb w14, [x10]
    mov x19, x13
    b error_convert_loop
error_write_digits:
    mov x0, #1
    mov x1, x10
    sub x2, x11, x10
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, newline_msg
    add x1, x1, :lo12:newline_msg
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b error_exit

peek_byte:
    adrp x1, input_pos
    ldr x2, [x1, :lo12:input_pos]
    adrp x1, input_len
    ldr x3, [x1, :lo12:input_len]
    cmp x2, x3
    b.hs peek_eof
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    ldrb w0, [x1, x2]
    ret
peek_eof:
    mov w0, #-1
    ret

inc_pos:
    adrp x1, input_pos
    ldr x2, [x1, :lo12:input_pos]
    add x2, x2, #1
    str x2, [x1, :lo12:input_pos]
    ret

skip_ws:
sw_loop:
    adrp x0, input_pos
    ldr x1, [x0, :lo12:input_pos]
    adrp x0, input_len
    ldr x2, [x0, :lo12:input_len]
    cmp x1, x2
    b.hs sw_done
    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    ldrb w3, [x0, x1]
    cmp w3, #0x20
    b.eq sw_advance
    cmp w3, #0x09
    b.eq sw_advance
    cmp w3, #0x0a
    b.eq sw_advance
    cmp w3, #0x0d
    b.eq sw_advance
    b sw_done
sw_advance:
    add x1, x1, #1
    adrp x0, input_pos
    str x1, [x0, :lo12:input_pos]
    b sw_loop
sw_done:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x7b
    b.eq pv_object
    cmp w0, #0x5b
    b.eq pv_array
    cmp w0, #0x22
    b.eq pv_string
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6e
    b.eq pv_null
    cmp w0, #0x2d
    b.eq pv_number
    cmp w0, #0x30
    b.lo error_exit
    cmp w0, #0x39
    b.hi error_exit
    b pv_number
pv_object:
    bl parse_object
    b pv_done
pv_array:
    bl parse_array
    b pv_done
pv_string:
    bl parse_string
    b pv_done
pv_true:
    adrp x4, str_true
    add x4, x4, :lo12:str_true
    mov w5, #4
    bl parse_literal
    b pv_done
pv_false:
    adrp x4, str_false
    add x4, x4, :lo12:str_false
    mov w5, #5
    bl parse_literal
    b pv_done
pv_null:
    adrp x4, str_null
    add x4, x4, :lo12:str_null
    mov w5, #4
    bl parse_literal
    b pv_done
pv_number:
    bl parse_number
    b pv_done
pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
literal_loop:
    cmp w5, #0
    b.eq literal_done
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    ldrb w6, [x4], #1
    cmp w0, w6
    b.ne error_exit
    bl inc_pos
    sub w5, w5, #1
    b literal_loop
literal_done:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl inc_pos
    bl skip_ws
    bl peek_byte
    cmp w0, #0x7d
    b.ne po_member
    bl inc_pos
    b po_done
po_member:
    bl peek_byte
    cmp w0, #0x22
    b.ne error_exit
    bl parse_string
    bl skip_ws
    bl peek_byte
    cmp w0, #0x3a
    b.ne error_exit
    bl inc_pos
    bl skip_ws
    bl parse_value
    bl skip_ws
    bl peek_byte
    cmp w0, #0x2c
    b.eq po_comma
    cmp w0, #0x7d
    b.ne error_exit
    bl inc_pos
    b po_done
po_comma:
    bl inc_pos
    bl skip_ws
    b po_member
po_done:
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl inc_pos
    bl skip_ws
    bl peek_byte
    cmp w0, #0x5d
    b.ne pa_loop
    bl inc_pos
    b pa_done
pa_loop:
    bl parse_value
    bl skip_ws
    bl peek_byte
    cmp w0, #0x2c
    b.eq pa_comma
    cmp w0, #0x5d
    b.ne error_exit
    bl inc_pos
    b pa_done
pa_comma:
    bl inc_pos
    bl skip_ws
    b pa_loop
pa_done:
    ldp x29, x30, [sp], #16
    ret

parse_string:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl inc_pos
ps_loop:
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x22
    b.eq ps_close
    cmp w0, #0x5c
    b.eq ps_escape
    cmp w0, #0x20
    b.lo error_exit
    bl inc_pos
    b ps_loop
ps_close:
    bl inc_pos
    ldp x29, x30, [sp], #16
    ret
ps_escape:
    bl inc_pos
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x22
    b.eq ps_escape_ok
    cmp w0, #0x5c
    b.eq ps_escape_ok
    cmp w0, #0x2f
    b.eq ps_escape_ok
    cmp w0, #0x62
    b.eq ps_escape_ok
    cmp w0, #0x66
    b.eq ps_escape_ok
    cmp w0, #0x6e
    b.eq ps_escape_ok
    cmp w0, #0x72
    b.eq ps_escape_ok
    cmp w0, #0x74
    b.eq ps_escape_ok
    cmp w0, #0x75
    b.eq ps_unicode
    b error_exit
ps_escape_ok:
    bl inc_pos
    b ps_loop
ps_unicode:
    bl inc_pos
    mov w4, #4
ps_hex_loop:
    cmp w4, #0
    b.eq ps_loop
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x30
    b.lo error_exit
    cmp w0, #0x39
    b.ls ps_hex_ok
    cmp w0, #0x41
    b.lo error_exit
    cmp w0, #0x46
    b.ls ps_hex_ok
    cmp w0, #0x61
    b.lo error_exit
    cmp w0, #0x66
    b.hi error_exit
ps_hex_ok:
    bl inc_pos
    sub w4, w4, #1
    b ps_hex_loop

parse_number:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl peek_byte
    cmp w0, #0x2d
    b.ne pn_int_check
    bl inc_pos
pn_int_check:
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x30
    b.eq pn_zero
    cmp w0, #0x31
    b.lo error_exit
    cmp w0, #0x39
    b.hi error_exit
pn_int_loop:
    bl peek_byte
    cmp w0, #0x30
    b.lo pn_frac
    cmp w0, #0x39
    b.hi pn_frac
    bl inc_pos
    b pn_int_loop
pn_zero:
    bl inc_pos
pn_frac:
    bl peek_byte
    cmp w0, #0x2e
    b.ne pn_exp
    bl inc_pos
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x30
    b.lo error_exit
    cmp w0, #0x39
    b.hi error_exit
pn_frac_loop:
    bl peek_byte
    cmp w0, #0x30
    b.lo pn_exp
    cmp w0, #0x39
    b.hi pn_exp
    bl inc_pos
    b pn_frac_loop
pn_exp:
    bl peek_byte
    cmp w0, #0x65
    b.eq pn_exp_body
    cmp w0, #0x45
    b.ne pn_done
pn_exp_body:
    bl inc_pos
    bl peek_byte
    cmp w0, #0x2b
    b.eq pn_exp_sign
    cmp w0, #0x2d
    b.ne pn_exp_digit_check
pn_exp_sign:
    bl inc_pos
pn_exp_digit_check:
    bl peek_byte
    cmn w0, #1
    b.eq error_exit
    cmp w0, #0x30
    b.lo error_exit
    cmp w0, #0x39
    b.hi error_exit
pn_exp_loop:
    bl peek_byte
    cmp w0, #0x30
    b.lo pn_done
    cmp w0, #0x39
    b.hi pn_done
    bl inc_pos
    b pn_exp_loop
pn_done:
    ldp x29, x30, [sp], #16
    ret

.section .rodata

    .ascii "true"

    .ascii "false"

    .ascii "null"
