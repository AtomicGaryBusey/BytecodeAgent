.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, 65536
    jae parse_start
    lea rsi, [rip + buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_start
    js parse_start
    add rbx, rax
    jmp read_loop
parse_start:
    mov qword ptr [rip + buflen], rbx
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jne fail
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
skip_ws:
sw_loop:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae sw_done
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x20
    je sw_inc
    cmp dl, 0x09
    je sw_inc
    cmp dl, 0x0a
    je sw_inc
    cmp dl, 0x0d
    je sw_inc
    ret
sw_inc:
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp sw_loop
sw_done:
    ret
parse_value:
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x7b
    je pv_object
    cmp dl, 0x5b
    je pv_array
    cmp dl, 0x22
    je pv_string
    cmp dl, 0x2d
    je pv_number
    cmp dl, 0x30
    jb pv_letter
    cmp dl, 0x39
    jbe pv_number
pv_letter:
    cmp dl, 0x74
    je pv_true
    cmp dl, 0x66
    je pv_false
    cmp dl, 0x6e
    je pv_null
    jmp fail
pv_object:
    jmp parse_object
pv_array:
    jmp parse_array
pv_string:
    jmp parse_string
pv_number:
    jmp parse_number
pv_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp parse_literal
pv_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp parse_literal
pv_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp parse_literal
parse_literal:
    xor r8d, r8d
pl_loop:
    cmp r8, rcx
    jae pl_done
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea r9, [rip + buf]
    movzx edx, byte ptr [r9 + rax]
    movzx eax, byte ptr [rsi + r8]
    cmp dl, al
    jne fail
    mov rax, qword ptr [rip + curpos]
    inc rax
    mov qword ptr [rip + curpos], rax
    inc r8
    jmp pl_loop
pl_done:
    ret
parse_string:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x22
    jne fail
    inc rax
    mov qword ptr [rip + curpos], rax
ps_loop:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x22
    je ps_close
    cmp dl, 0x5c
    je ps_escape
    cmp dl, 0x20
    jb fail
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp ps_loop
ps_close:
    inc rax
    mov qword ptr [rip + curpos], rax
    ret
ps_escape:
    inc rax
    mov qword ptr [rip + curpos], rax
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x75
    je ps_unicode
    cmp dl, 0x22
    je ps_escape_ok
    cmp dl, 0x5c
    je ps_escape_ok
    cmp dl, 0x2f
    je ps_escape_ok
    cmp dl, 0x62
    je ps_escape_ok
    cmp dl, 0x66
    je ps_escape_ok
    cmp dl, 0x6e
    je ps_escape_ok
    cmp dl, 0x72
    je ps_escape_ok
    cmp dl, 0x74
    je ps_escape_ok
    jmp fail
ps_escape_ok:
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp ps_loop
ps_unicode:
    inc rax
    mov qword ptr [rip + curpos], rax
    xor r8d, r8d
ps_hex_loop:
    cmp r8, 4
    jae ps_loop
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb fail
    cmp dl, 0x39
    jbe ps_hex_ok
    cmp dl, 0x41
    jb fail
    cmp dl, 0x46
    jbe ps_hex_ok
    cmp dl, 0x61
    jb fail
    cmp dl, 0x66
    jbe ps_hex_ok
    jmp fail
ps_hex_ok:
    inc rax
    mov qword ptr [rip + curpos], rax
    inc r8
    jmp ps_hex_loop
parse_number:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x2d
    jne pn_int_check
    inc rax
    mov qword ptr [rip + curpos], rax
pn_int_check:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    je pn_zero
    cmp dl, 0x31
    jb fail
    cmp dl, 0x39
    ja fail
pn_digit_loop:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_frac
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb pn_frac
    cmp dl, 0x39
    ja pn_frac
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp pn_digit_loop
pn_zero:
    inc rax
    mov qword ptr [rip + curpos], rax
pn_frac:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_ret
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x2e
    jne pn_exp
    inc rax
    mov qword ptr [rip + curpos], rax
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb fail
    cmp dl, 0x39
    ja fail
pn_frac_digit_loop:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_exp
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb pn_exp
    cmp dl, 0x39
    ja pn_exp
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp pn_frac_digit_loop
pn_exp:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_ret
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x65
    je pn_exp_have
    cmp dl, 0x45
    jne pn_ret
pn_exp_have:
    inc rax
    mov qword ptr [rip + curpos], rax
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_exp_req
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x2b
    je pn_exp_sign
    cmp dl, 0x2d
    jne pn_exp_req
pn_exp_sign:
    inc rax
    mov qword ptr [rip + curpos], rax
pn_exp_req:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb fail
    cmp dl, 0x39
    ja fail
pn_exp_digit_loop:
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae pn_ret
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x30
    jb pn_ret
    cmp dl, 0x39
    ja pn_ret
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp pn_exp_digit_loop
pn_ret:
    ret
parse_object:
    mov rax, qword ptr [rip + curpos]
    inc rax
    mov qword ptr [rip + curpos], rax
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x7d
    je po_close
po_loop:
    call parse_string
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x3a
    jne fail
    inc rax
    mov qword ptr [rip + curpos], rax
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x2c
    je po_comma
    cmp dl, 0x7d
    je po_close
    jmp fail
po_comma:
    inc rax
    mov qword ptr [rip + curpos], rax
    call skip_ws
    jmp po_loop
po_close:
    inc rax
    mov qword ptr [rip + curpos], rax
    ret
parse_array:
    mov rax, qword ptr [rip + curpos]
    inc rax
    mov qword ptr [rip + curpos], rax
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x5d
    je pa_close
pa_loop:
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + curpos]
    cmp rax, qword ptr [rip + buflen]
    jae fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x2c
    je pa_comma
    cmp dl, 0x5d
    je pa_close
    jmp fail
pa_comma:
    inc rax
    mov qword ptr [rip + curpos], rax
    jmp pa_loop
pa_close:
    inc rax
    mov qword ptr [rip + curpos], rax
    ret
fail:
    mov rbx, qword ptr [rip + curpos]
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    mov eax, 1
    syscall
    lea rcx, [rip + numbuf + 32]
conv_loop:
    mov rax, rbx
    xor edx, edx
    mov r8, 10
    div r8
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    mov rbx, rax
    test rax, rax
    jnz conv_loop
    lea rax, [rip + numbuf + 32]
    sub rax, rcx
    mov rdx, rax
    mov rsi, rcx
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    lea rsi, [rip + newline_msg]
    mov edx, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

    .ascii "\n"

    .ascii "true"
str_false:
    .ascii "false"

    .ascii "null"
