.text
.globl _start
_start:
    xor ebx, ebx
    xor ebp, ebp
    xor r12, r12
    xor r13, r13
read_next:
    lea rsi, [rip + buffer]
    mov edx, 8192
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle output
    mov r14, rax
    lea rsi, [rip + buffer]
process_loop:
    test r14, r14
    jz read_next
    movzx eax, byte ptr [rsi]
    inc rsi
    dec r14
    inc r12
    cmp al, 10
    jne 1f
    inc rbx
1:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    ja word_byte
whitespace:
    xor r13d, r13d
    jmp process_loop
word_byte:
    test r13, r13
    jnz process_loop
    inc rbp
    mov r13d, 1
    jmp process_loop
output:
    lea rdi, [rip + outbuf]
    mov rax, rbx
    call convert
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, rbp
    call convert
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call convert
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
convert:
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 48
    inc rdi
    ret
1:
    sub rsp, 40
    lea rsi, [rsp + 40]
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 2b
    lea rcx, [rsp + 40]
3:
    cmp rsi, rcx
    jae 4f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp 3b
4:
    add rsp, 40
    ret
.bss
.balign 16
buffer:
    .zero 8192
.balign 16
outbuf:
    .zero 64
