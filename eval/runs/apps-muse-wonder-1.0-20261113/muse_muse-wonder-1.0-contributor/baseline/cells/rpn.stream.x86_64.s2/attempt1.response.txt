.globl _start
_start:
xor r12d, r12d
read_loop:
cmp r12, 65536
jae read_done
mov eax, 0
xor edi, edi

add r12, rax

add r15, r12
lea rbp, [rip + input_buf]
outer_loop:
cmp rbp, r15
jae prog_done
mov rax, rbp
find_line_end:
cmp rax, r15
je line_end_found
cmp byte ptr [rax], 0x0A

line_end_found:
mov rbx, rax
xor r12d, r12d
xor r14d, r14d
mov rsi, rbp

token_loop:
cmp rsi, rbx
jae tokens_done
movzx eax, byte ptr [rsi]
cmp al, 0x20
je sep_skip
cmp al, 0x09
je sep_skip
jmp token_start
sep_skip:
inc rsi
jmp token_loop
token_start:
mov rdi, rsi
scan_token_end:
cmp rdi, rbx
jae token_end_found
movzx eax, byte ptr [rdi]
cmp al, 0x20
je token_end_found
cmp al, 0x09
je token_end_found
inc rdi
jmp scan_token_end
token_end_found:
mov rcx, rdi
sub rcx, rsi
cmp rcx, 1
jne parse_literal
movzx eax, byte ptr [rsi]
cmp al, 0x2B
je do_operator
cmp al, 0x2D
je do_operator
cmp al, 0x2A
je do_operator
cmp al, 0x2F
je do_operator
cmp al, 0x25
je do_operator
jmp parse_literal
do_operator:
mov r8d, eax
cmp r12, 2
jb error_line
mov rcx, r12
sub rcx, 1
mov rdx, qword ptr [r13 + rcx*8]
dec rcx
mov rax, qword ptr [r13 + rcx*8]
cmp r8b, 0x2B
je op_add
cmp r8b, 0x2D
je op_sub
cmp r8b, 0x2A
je op_mul
cmp r8b, 0x2F
je op_div
jmp op_mod
op_add:
add rax, rdx
jmp op_store
op_sub:
sub rax, rdx
jmp op_store
op_mul:
imul rax, rdx
jmp op_store
op_div:
cmp rdx, 0
je error_line
cmp rdx, -1
jne op_div_do
mov r9, 1
shl r9, 63
cmp rax, r9
je error_line
op_div_do:
mov r9, rdx
cqo
idiv r9
jmp op_store
op_mod:
cmp rdx, 0
je error_line
cmp rdx, -1
jne op_mod_do
mov r9, 1
shl r9, 63
cmp rax, r9
je error_line
op_mod_do:
mov r9, rdx
cqo
idiv r9
mov rax, rdx
jmp op_store
op_store:
mov qword ptr [r13 + rcx*8], rax
lea r12, [rcx + 1]
mov rsi, rdi
jmp token_loop
parse_literal:
xor r10d, r10d
xor r9d, r9d
movzx eax, byte ptr [rsi]
cmp al, 0x2D
jne lit_check_len
mov r10d, 1
mov r9d, 1
lit_check_len:
mov r8, rcx
sub r8, r9
cmp r8, 1
jb error_line
cmp r8, 18
ja error_line
xor eax, eax
lit_loop:
cmp r9, rcx
jae lit_done
movzx edx, byte ptr [rsi + r9]
cmp edx, 0x30
jb error_line
cmp edx, 0x39
ja error_line
sub edx, 0x30
imul rax, rax, 10
add rax, rdx
inc r9
jmp lit_loop
lit_done:
cmp r10d, 0
je lit_push
neg rax
lit_push:
mov qword ptr [r13 + r12*8], rax
inc r12
mov rsi, rdi
jmp token_loop
tokens_done:
cmp r12, 1
jne error_line
mov rax, qword ptr [r13]
jmp print_number
error_line:

mov edx, 6
call write_all
jmp advance_line
print_number:

mov r8, rax
cmp r8, 0

mov r10d, 1

num_positive:
lea rdi, [rip + num_buf]
add rdi, 32
num_convert:
mov rax, r8
xor edx, edx
mov ecx, 10
div rcx

dec rdi

lea rax, [rip + num_buf]
add rax, 32
mov byte ptr [rax], 0x0A
mov rdx, rax
sub rdx, rdi
add rdx, 1
mov rsi, rdi
call write_all
jmp advance_line
advance_line:
mov rbp, rbx
cmp rbp, r15
jae outer_loop
inc rbp
jmp outer_loop
prog_done:

syscall

jle write_ret
add rsi, rax
sub rdx, rax
cmp rdx, 0
jne write_loop

ret
