.text
.globl _start
_start:
xor r12d, r12d
xor r13d, r13d
xor r14d, r14d
xor r15d, r15d
read_loop:
mov eax, 0
xor edi, edi
lea rsi, [rip + buffer]
mov edx, 8192
syscall
test rax, rax
jle read_done
mov rcx, rax
add r14, rax
lea rsi, [rip + buffer]
process_loop:
test rcx, rcx
jz read_loop
movzx eax, byte ptr [rsi]
inc rsi
dec rcx
cmp al, 10
jne not_newline
inc r12
not_newline:
cmp al, 32
je is_space
cmp al, 9
jb is_word
cmp al, 13
ja is_word
is_space:
xor r15d, r15d
jmp process_loop
is_word:
test r15d, r15d
jnz process_loop
inc r13
mov r15d, 1
jmp process_loop
read_done:
lea rdi, [rip + outbuf]
mov rax, r12
call append_uint
mov byte ptr [rdi], 32
inc rdi
mov rax, r13
call append_uint
mov byte ptr [rdi], 32
inc rdi
mov rax, r14
call append_uint
mov byte ptr [rdi], 10
inc rdi
lea rsi, [rip + outbuf]
mov rdx, rdi
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
mov eax, 60
xor edi, edi
syscall
append_uint:
sub rsp, 24
mov r8, rdi
mov r9, 10
xor ecx, ecx
test rax, rax
jnz au_div_loop
mov byte ptr [rsp + rcx], 48
inc rcx
jmp au_copy_loop
au_div_loop:
test rax, rax
jz au_copy_loop
xor edx, edx
div r9
add dl, 48
mov byte ptr [rsp + rcx], dl
inc rcx
jmp au_div_loop
au_copy_loop:
test rcx, rcx
jz au_done
dec rcx
mov al, byte ptr [rsp + rcx]
mov byte ptr [r8], al
inc r8
jmp au_copy_loop
au_done:
mov rdi, r8
add rsp, 24
ret
.bss
.align 16
buffer:
.zero 8192
outbuf:
.zero 64
