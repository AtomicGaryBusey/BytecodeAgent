.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ecx, ecx
    mov r13, r12
    mov r14, r12
find_loop:
    cmp rcx, r12
    jae find_done
    cmp byte ptr [rbx + rcx], 33
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r13, rcx
    lea r14, [rcx + 1]
find_done:
    xor ecx, ecx
    xor r15d, r15d
    lea r8, [rip + stack_arr]
    lea r9, [rip + jump_table]
validate_loop:
    cmp rcx, r13
    jae validate_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 91
    je bracket_push
    cmp al, 93
    je bracket_pop
    inc rcx
    jmp validate_loop
bracket_push:
    mov dword ptr [r8 + r15*4], ecx
    inc r15
    inc rcx
    jmp validate_loop
bracket_pop:
    cmp r15, 0
    je unbalanced
    dec r15
    mov edx, dword ptr [r8 + r15*4]
    mov dword ptr [r9 + rcx*4], edx
    mov dword ptr [r9 + rdx*4], ecx
    inc rcx
    jmp validate_loop
validate_done:
    cmp r15, 0
    jne unbalanced
    lea r10, [rip + tape]
    xor ebp, ebp
    xor r15d, r15d
exec_loop:
    cmp rbp, r13
    jae exit_success
    movzx eax, byte ptr [rbx + rbp]
    cmp al, 62
    je cmd_inc_ptr
    cmp al, 60
    je cmd_dec_ptr
    cmp al, 43
    je cmd_inc_cell
    cmp al, 45
    je cmd_dec_cell
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    inc rbp
    jmp exec_loop
cmd_inc_ptr:
    inc r15
    inc rbp
    jmp exec_loop
cmd_dec_ptr:
    dec r15
    inc rbp
    jmp exec_loop
cmd_inc_cell:
    inc byte ptr [r10 + r15]
    inc rbp
    jmp exec_loop
cmd_dec_cell:
    dec byte ptr [r10 + r15]
    inc rbp
    jmp exec_loop
cmd_output:
    mov eax, 1
    mov edi, 1
    lea rsi, [r10 + r15]
    mov edx, 1
    syscall
    inc rbp
    jmp exec_loop
cmd_input:
    cmp r14, r12
    jae input_eof
    mov dl, byte ptr [rbx + r14]
    mov byte ptr [r10 + r15], dl
    inc r14
    inc rbp
    jmp exec_loop
input_eof:
    mov byte ptr [r10 + r15], 0
    inc rbp
    jmp exec_loop
cmd_open:
    cmp byte ptr [r10 + r15], 0
    je open_jump
    inc rbp
    jmp exec_loop
open_jump:
    mov eax, dword ptr [r9 + rbp*4]
    mov rbp, rax
    inc rbp
    jmp exec_loop
cmd_close:
    cmp byte ptr [r10 + r15], 0
    je close_next
    mov eax, dword ptr [r9 + rbp*4]
    mov rbp, rax
    inc rbp
    jmp exec_loop
close_next:
    inc rbp
    jmp exec_loop
exit_success:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
tape:
    .zero 30000
.balign 16
jump_table:
    .zero 262144
.balign 16
stack_arr:
    .zero 262144
