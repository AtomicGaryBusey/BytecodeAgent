_start:
    xor ebx, ebx
read_loop:
    cmp rbx, 49152
    jae read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 49152
    sub rdx, rbx
    xor edi, edi

    jle read_done
    add rbx, rax
    jmp read_loop
read_done:

    lea r15, [rip + outbuf]
    lea rsi, [rip + alphabet]

    xor r13d, r13d

encode_loop:

    xor r10d, r10d
    xor r11d, r11d
    cmp r8, 2
    jb have_bytes
    movzx r10d, byte ptr [rbp + r12 + 1]
    cmp r8, 3
    jb have_bytes
    movzx r11d, byte ptr [rbp + r12 + 2]
have_bytes:
    mov eax, 3
    cmp r8, 3
    jae have_step
    mov rax, r8
have_step:
    add r12, rax
    mov eax, r9d
    shr eax, 2

    call emit_char
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov ecx, r10d
    shr ecx, 4
    or eax, ecx
    movzx eax, byte ptr [rsi + rax]
    call emit_char
    cmp r8, 2
    jb emit_pad2
    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov ecx, r11d
    shr ecx, 6
    or eax, ecx

emit_pad2:

third_done:

    mov eax, r11d
    and eax, 63

    call emit_char
    jmp encode_loop
emit_pad3:
    mov eax, 61
    call emit_char

encode_done:
    test r14, r14
    je write_out
    mov byte ptr [r15 + r13], 10

write_out:
    xor ebx, ebx
write_loop:

    mov rdx, r13
    sub rdx, rbx

    test rax, rax
    jle exit_prog
    add rbx, rax
    jmp write_loop
exit_prog:
    xor edi, edi
    mov eax, 60
    syscall
emit_char:
    mov byte ptr [r15 + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne emit_ret
    mov byte ptr [r15 + r13], 10
    inc r13
    xor r14d, r14d
emit_ret:
