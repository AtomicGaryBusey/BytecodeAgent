.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], r12
    mov qword ptr [rip + input_pos], 0
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
1:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae 2f
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 32
    je 3f
    cmp al, 9
    je 3f
    cmp al, 10
    je 3f
    cmp al, 13
    je 3f
    ret
3:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    jmp 1b
2:
    ret

parse_string:
    mov rcx, qword ptr [rip + input_pos]
    inc rcx
    mov qword ptr [rip + input_pos], rcx
string_loop:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 34
    je string_close
    cmp al, 92
    je string_escape
    cmp al, 32
    jb fail
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    jmp string_loop
string_close:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    ret
string_escape:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 117
    je string_unicode
    cmp al, 34
    je string_escape_done
    cmp al, 92
    je string_escape_done
    cmp al, 47
    je string_escape_done
    cmp al, 98
    je string_escape_done
    cmp al, 102
    je string_escape_done
    cmp al, 110
    je string_escape_done
    cmp al, 114
    je string_escape_done
    cmp al, 116
    je string_escape_done
    jmp fail
string_escape_done:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    jmp string_loop
string_unicode:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov r8d, 4
string_unicode_loop:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb fail
    cmp al, 57
    jbe string_hex_ok
    cmp al, 65
    jb fail
    cmp al, 70
    jbe string_hex_ok
    cmp al, 97
    jb fail
    cmp al, 102
    jbe string_hex_ok
    jmp fail
string_hex_ok:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    dec r8d
    jnz string_unicode_loop
    jmp string_loop

parse_number:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 45
    jne number_first_digit
    inc rcx
    mov qword ptr [rip + input_pos], rcx
number_first_digit:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
    cmp al, 48
    je number_zero
number_digit_loop:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_frac
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb number_frac
    cmp al, 57
    ja number_frac
    jmp number_digit_loop
number_zero:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
number_frac:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_done
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 46
    jne number_exp
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
number_frac_digit_loop:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_exp
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb number_exp
    cmp al, 57
    ja number_exp
    jmp number_frac_digit_loop
number_exp:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_done
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 101
    je number_exp_consume
    cmp al, 69
    jne number_done
number_exp_consume:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_exp_require
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 43
    je number_exp_sign
    cmp al, 45
    jne number_exp_require
number_exp_sign:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
number_exp_require:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
number_exp_digit_loop:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae number_done
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 48
    jb number_done
    cmp al, 57
    ja number_done
    jmp number_exp_digit_loop
number_done:
    ret

parse_literal:
    xor r10d, r10d
literal_loop:
    cmp r10, r9
    jae literal_done
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    movzx edx, byte ptr [r8 + r10]
    cmp al, dl
    jne fail
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    inc r10
    jmp literal_loop
literal_done:
    ret

parse_value:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 34
    je parse_string
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 116
    je value_true
    cmp al, 102
    je value_false
    cmp al, 110
    je value_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
    jmp parse_number
value_true:
    lea r8, [rip + lit_true]
    mov r9d, 4
    jmp parse_literal
value_false:
    lea r8, [rip + lit_false]
    mov r9d, 5
    jmp parse_literal
value_null:
    lea r8, [rip + lit_null]
    mov r9d, 4
    jmp parse_literal

parse_object:
    mov rcx, qword ptr [rip + input_pos]
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    call skip_ws
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 125
    je object_close
object_member_loop:
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 34
    jne fail
    call parse_string
    call skip_ws
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 58
    jne fail
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 44
    je object_comma
    cmp al, 125
    je object_close
    jmp fail
object_comma:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    call skip_ws
    jmp object_member_loop
object_close:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    ret

parse_array:
    mov rcx, qword ptr [rip + input_pos]
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    call skip_ws
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 93
    je array_close
array_elem_loop:
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + input_len]
    mov rcx, qword ptr [rip + input_pos]
    cmp rcx, rax
    jae fail
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 44
    je array_comma
    cmp al, 93
    je array_close
    jmp fail
array_comma:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    call skip_ws
    jmp array_elem_loop
array_close:
    inc rcx
    mov qword ptr [rip + input_pos], rcx
    ret

fail:
    mov rax, qword ptr [rip + input_pos]
    lea r8, [rip + num_buf]
    add r8, 16
    mov r9, r8
    mov r10, 10
error_digit_loop:
    xor edx, edx
    div r10
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz error_digit_loop
    mov r12, r8
    mov r13, r9
    sub r13, r8
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_str]
    mov edx, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
