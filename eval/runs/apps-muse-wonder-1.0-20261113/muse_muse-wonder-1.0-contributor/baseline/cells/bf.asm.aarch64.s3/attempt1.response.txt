.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ TABLE_SIZE, 262144

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x0, #0
find_loop:
    cmp x0, x20
    b.hs find_none
    add x1, x19, x0
    ldrb w2, [x1]
    cmp w2, #0x21
    b.eq find_found
    add x0, x0, #1
    b find_loop
find_found:
    mov x21, x0
    add x28, x0, #1
    b bracket_setup
find_none:
    mov x21, x20
    mov x28, x20
bracket_setup:
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x27, bracket_stack
    add x27, x27, :lo12:bracket_stack
    mov x22, #0
    mov x24, #0
bracket_loop:
    cmp x24, x21
    b.hs bracket_done
    add x9, x19, x24
    ldrb w10, [x9]
    cmp w10, #0x5b
    b.eq bracket_open
    cmp w10, #0x5d
    b.eq bracket_close
    add x24, x24, #1
    b bracket_loop
bracket_open:
    add x13, x27, x22, lsl #2
    str w24, [x13]
    add x22, x22, #1
    add x24, x24, #1
    b bracket_loop
bracket_close:
    cbz x22, fail
    sub x22, x22, #1
    add x13, x27, x22, lsl #2
    ldr w14, [x13]
    add x13, x26, x24, lsl #2
    str w14, [x13]
    add x13, x26, x14, lsl #2
    str w24, [x13]
    add x24, x24, #1
    b bracket_loop
bracket_done:
    cbnz x22, fail
    adrp x25, tape
    add x25, x25, :lo12:tape
    mov x23, #0
    mov x24, #0
exec_loop:
    cmp x24, x21
    b.hs success
    add x9, x19, x24
    ldrb w10, [x9]
    add x11, x25, x23
    cmp w10, #0x3e
    b.eq exec_gt
    cmp w10, #0x3c
    b.eq exec_lt
    cmp w10, #0x2b
    b.eq exec_plus
    cmp w10, #0x2d
    b.eq exec_minus
    cmp w10, #0x2e
    b.eq exec_dot
    cmp w10, #0x2c
    b.eq exec_comma
    cmp w10, #0x5b
    b.eq exec_open
    cmp w10, #0x5d
    b.eq exec_close
    add x24, x24, #1
    b exec_loop
exec_gt:
    add x23, x23, #1
    add x24, x24, #1
    b exec_loop
exec_lt:
    sub x23, x23, #1
    add x24, x24, #1
    b exec_loop
exec_plus:
    ldrb w12, [x11]
    add w12, w12, #1
    strb w12, [x11]
    add x24, x24, #1
    b exec_loop
exec_minus:
    ldrb w12, [x11]
    sub w12, w12, #1
    strb w12, [x11]
    add x24, x24, #1
    b exec_loop
exec_dot:
    mov x0, #1
    mov x1, x11
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x24, x24, #1
    b exec_loop
exec_comma:
    cmp x28, x20
    b.hs exec_comma_eof
    add x9, x19, x28
    ldrb w12, [x9]
    add x28, x28, #1
    strb w12, [x11]
    add x24, x24, #1
    b exec_loop
exec_comma_eof:
    strb wzr, [x11]
    add x24, x24, #1
    b exec_loop
exec_open:
    ldrb w12, [x11]
    cbnz w12, exec_next
    add x13, x26, x24, lsl #2
    ldr w13, [x13]
    add x24, x13, #1
    b exec_loop
exec_close:
    ldrb w12, [x11]
    cbz w12, exec_next
    add x13, x26, x24, lsl #2
    ldr w13, [x13]
    add x24, x13, #1
    b exec_loop
exec_next:
    add x24, x24, #1
    b exec_loop
fail:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
    b fail
success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b success

.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
tape:
    .zero 30000
.balign 8
jump_table:
    .zero 262144
.balign 8
bracket_stack:
    .zero 262144
