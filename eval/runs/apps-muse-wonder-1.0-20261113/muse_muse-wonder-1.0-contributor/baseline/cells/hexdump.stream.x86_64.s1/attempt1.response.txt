.text
.globl _start
_start:
    xor r12d, r12d
    lea rbx, [rip + inbuf]
read_loop:
    mov edx, 65536
    sub edx, r12d
    test edx, edx
    jz dump_start
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle dump_start
    add r12, rax
    jmp read_loop
dump_start:
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae final_start
    mov rax, r12
    sub rax, r13
    mov r14d, 16
    cmp rax, 16
    jae n_done
    mov r14, rax
n_done:
    lea r15, [rip + linebuf]
    xor ebx, ebx
off_loop:
    mov eax, r13d
    mov ecx, 7
    sub ecx, ebx
    shl ecx, 2
    shr eax, cl
    and eax, 15
    lea rdx, [rip + hextab]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [r15], al
    inc r15
    inc ebx
    cmp ebx, 8
    jb off_loop
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15
    xor ebx, ebx
pos_loop:
    cmp rbx, r14
    jae pos_spaces
    lea rsi, [rip + inbuf]
    mov rax, r13
    add rax, rbx
    movzx eax, byte ptr [rsi + rax]
    lea rcx, [rip + hextab]
    mov r8d, eax
    shr r8d, 4
    and r8d, 15
    mov r8b, byte ptr [rcx + r8]
    mov byte ptr [r15], r8b
    inc r15
    and eax, 15
    mov al, byte ptr [rcx + rax]
    mov byte ptr [r15], al
    inc r15
    mov byte ptr [r15], 32
    inc r15
    jmp pos_done
pos_spaces:
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15
pos_done:
    cmp ebx, 7
    jne pos_next
    mov byte ptr [r15], 32
    inc r15
pos_next:
    inc ebx
    cmp ebx, 16
    jb pos_loop
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 124
    inc r15
    xor ebx, ebx
char_loop:
    cmp rbx, r14
    jae char_end
    lea rsi, [rip + inbuf]
    mov rax, r13
    add rax, rbx
    movzx eax, byte ptr [rsi + rax]
    cmp al, 32
    jb char_dot
    cmp al, 126
    ja char_dot
    jmp char_store
char_dot:
    mov al, 46
char_store:
    mov byte ptr [r15], al
    inc r15
    inc ebx
    jmp char_loop
char_end:
    mov byte ptr [r15], 124
    inc r15
    mov byte ptr [r15], 10
    inc r15
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov rdx, r15
    sub rdx, rsi
    syscall
    add r13, r14
    jmp row_loop
final_start:
    lea r15, [rip + linebuf]
    xor ebx, ebx
fin_loop:
    mov eax, r12d
    mov ecx, 7
    sub ecx, ebx
    shl ecx, 2
    shr eax, cl
    and eax, 15
    lea rdx, [rip + hextab]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [r15], al
    inc r15
    inc ebx
    cmp ebx, 8
    jb fin_loop
    mov byte ptr [r15], 10
    inc r15
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov rdx, r15
    sub rdx, rsi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata

.bss

inbuf:

linebuf:
