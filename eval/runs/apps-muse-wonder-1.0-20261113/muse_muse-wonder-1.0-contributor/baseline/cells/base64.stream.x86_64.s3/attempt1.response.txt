.equ INBUF_SIZE, 49152
.equ OUTBUF_SIZE, 66560

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov rdx, INBUF_SIZE
    sub rdx, r12
    test rdx, rdx
    jz read_done
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov r15, rsi
    add r15, r12
    lea rdi, [rip + outbuf]
    xor r13d, r13d
encode_loop:
    mov rax, r15
    sub rax, rsi
    cmp rax, 3
    jb encode_tail
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    add rsi, 3
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    mov r11d, edx
    shr r11d, 6
    or r10d, r11d
    and edx, 63
    mov eax, r8d
    call emit_one
    mov eax, r9d
    call emit_one
    mov eax, r10d
    call emit_one
    mov eax, edx
    call emit_one
    jmp encode_loop
encode_tail:
    test rax, rax
    jz encode_done
    cmp rax, 1
    je tail_one
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    mov eax, r8d
    call emit_one
    mov eax, r9d
    call emit_one
    mov eax, r10d
    call emit_one
    mov eax, 64
    call emit_one
    jmp encode_done
tail_one:
    movzx eax, byte ptr [rsi]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov eax, r8d
    call emit_one
    mov eax, r9d
    call emit_one
    mov eax, 64
    call emit_one
    mov eax, 64
    call emit_one
    jmp encode_done
encode_done:
    test r13, r13
    jz write_start
    mov byte ptr [rdi], 10
    inc rdi
write_start:
    lea rbx, [rip + outbuf]
    mov r12, rdi
    sub r12, rbx
write_loop:
    test r12, r12
    jz exit_prog
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    cmp rax, 0
    jle exit_prog
    add rbx, rax
    sub r12, rax
    jmp write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
emit_one:
    lea rcx, [rip + alphabet]
    mov al, byte ptr [rcx + rax]
    mov byte ptr [rdi], al
    inc rdi
    inc r13
    cmp r13, 76
    jne emit_ret
    mov byte ptr [rdi], 10
    inc rdi
    xor r13d, r13d
emit_ret:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="

.bss
.balign 16
inbuf:
    .zero INBUF_SIZE
.balign 16
outbuf:
    .zero OUTBUF_SIZE
