.equ BUF_SIZE, 131072
.globl _start
.text
_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov edx, BUF_SIZE
    sub rdx, r12
    mov rsi, rbx
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r14d, r14d
line_loop:
    cmp r14, r12
    jae exit_prog
    mov r15, r14
scan_loop:
    cmp r15, r12
    jae scan_done
    mov al, byte ptr [rbx + r15]
    cmp al, 10
    je scan_done
    inc r15
    jmp scan_loop
scan_done:
    cmp r14, r15
    jae reversed
    mov rsi, rbx
    add rsi, r14
    mov rdi, rbx
    add rdi, r15
    dec rdi
rev_loop:
    cmp rsi, rdi
    jae reversed
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev_loop
reversed:
    mov rsi, rbx
    add rsi, r14
    mov rdx, r15
    sub rdx, r14
    call write_all
    lea rsi, [rip + nl_byte]
    mov edx, 1
    call write_all
    mov r14, r15
    cmp r15, r12
    jae line_loop
    inc r14
    jmp line_loop
exit_prog:
    xor edi, edi
    mov eax, 60
    syscall
write_all:
    test rdx, rdx
    jz write_ret
write_loop:
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle write_ret
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_ret:
    ret
.section .rodata
nl_byte:
    .byte 10
.section .bss
.balign 16
inbuf:
    .zero BUF_SIZE
