.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r13d, r13d
row_loop:
    cmp r13, rbx
    jae final_line
    mov r14, rbx
    sub r14, r13
    cmp r14, 16
    jbe row_n_ok
    mov r14, 16
row_n_ok:
    lea r8, [rip + linebuf]
    mov r9d, 28
off_loop:
    mov eax, r13d
    mov ecx, r9d
    shr eax, cl
    and eax, 15
    lea r10, [rip + hexdig]
    mov al, byte ptr [r10 + rax]
    mov byte ptr [r8], al
    inc r8
    sub r9d, 4
    jns off_loop
    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 32
    add r8, 2
    xor r9d, r9d
pos_loop:
    cmp r9, r14
    jae pos_pad
    mov rax, r13
    add rax, r9
    movzx eax, byte ptr [r12 + rax]
    mov edx, eax
    shr edx, 4
    and eax, 15
    lea r10, [rip + hexdig]
    mov cl, byte ptr [r10 + rdx]
    mov byte ptr [r8], cl
    mov cl, byte ptr [r10 + rax]
    mov byte ptr [r8 + 1], cl
    mov byte ptr [r8 + 2], 32
    add r8, 3
    jmp pos_extra
pos_pad:
    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 32
    mov byte ptr [r8 + 2], 32
    add r8, 3
pos_extra:
    cmp r9, 7
    jne pos_next
    mov byte ptr [r8], 32
    inc r8
pos_next:
    inc r9
    cmp r9, 16
    jne pos_loop
    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 124
    add r8, 2
    xor r9d, r9d
ascii_loop:
    cmp r9, r14
    jae ascii_done
    mov rax, r13
    add rax, r9
    movzx eax, byte ptr [r12 + rax]
    cmp eax, 32
    jb ascii_dot
    cmp eax, 126
    ja ascii_dot
    mov byte ptr [r8], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [r8], 46
ascii_next:
    inc r8
    inc r9
    jmp ascii_loop
ascii_done:
    mov byte ptr [r8], 124
    mov byte ptr [r8 + 1], 10
    add r8, 2
    lea rsi, [rip + linebuf]
    mov rdx, r8
    sub rdx, rsi
    call write_all
    add r13, r14
    jmp row_loop
final_line:
    lea r8, [rip + linebuf]
    mov r9d, 28
final_loop:
    mov eax, ebx
    mov ecx, r9d
    shr eax, cl
    and eax, 15
    lea r10, [rip + hexdig]
    mov al, byte ptr [r10 + rax]
    mov byte ptr [r8], al
    inc r8
    sub r9d, 4
    jns final_loop
    mov byte ptr [r8], 10
    inc r8
    lea rsi, [rip + linebuf]
    mov rdx, r8
    sub rdx, rsi
    call write_all
    mov eax, 60
    xor edi, edi
    syscall
write_all:
    test rdx, rdx
    jz write_done
write_again:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jnz write_again
write_done:
    ret
.section .rodata
hexdig:
    .ascii "0123456789abcdef"
.bss
    .align 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
