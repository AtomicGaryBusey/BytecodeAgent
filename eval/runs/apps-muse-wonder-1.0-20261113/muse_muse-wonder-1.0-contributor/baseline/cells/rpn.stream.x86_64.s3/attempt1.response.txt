.text
.globl _start
_start:
lea rbx, [rip + input_buf]
lea r12, [rip + input_buf]
add r12, 65536
read_loop:
cmp rbx, r12
jae read_done
mov eax, 0
xor edi, edi
mov rsi, rbx
mov rdx, r12
sub rdx, rbx
syscall
test rax, rax
jz read_done
js read_done
add rbx, rax
jmp read_loop
read_done:
mov r12, rbx
lea rbx, [rip + input_buf]
lea r15, [rip + output_buf]
line_loop:
cmp rbx, r12
jae all_done
xor r13d, r13d
xor r14d, r14d
jmp token_seek
token_seek:
cmp rbx, r12
jae end_line
movzx eax, byte ptr [rbx]
cmp al, 10
je end_line
cmp al, 32
je skip_sep
cmp al, 9
je skip_sep
jmp token_start
skip_sep:
inc rbx
jmp token_seek
token_start:
test r14d, r14d
jnz scan_to_eol
mov r8, rbx
scan_token_loop:
cmp rbx, r12
jae token_ready
movzx eax, byte ptr [rbx]
cmp al, 10
je token_ready
cmp al, 32
je token_ready
cmp al, 9
je token_ready
inc rbx
jmp scan_token_loop
token_ready:
mov rdx, rbx
sub rdx, r8
cmp rdx, 1
jne try_literal
movzx eax, byte ptr [r8]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator
jmp try_literal
do_operator:
mov r9d, eax
cmp r13, 2
jb set_error
lea rcx, [rip + val_stack]
mov r10, qword ptr [rcx + r13*8 - 8]
mov rax, qword ptr [rcx + r13*8 - 16]
dec r13
cmp r9b, 43
je op_add
cmp r9b, 45
je op_sub
cmp r9b, 42
je op_mul
cmp r9b, 47
je op_div
jmp op_mod
op_add:
add rax, r10
jmp store_result
op_sub:
sub rax, r10
jmp store_result
op_mul:
imul rax, r10
jmp store_result
op_div:
test r10, r10
jz set_error
cmp r10, -1
jne do_div
mov r11, 0x8000000000000000
cmp rax, r11
je set_error
do_div:
cqo
idiv r10
jmp store_result
op_mod:
test r10, r10
jz set_error
cmp r10, -1
jne do_mod
mov r11, 0x8000000000000000
cmp rax, r11
je set_error
do_mod:
cqo
idiv r10
mov rax, rdx
jmp store_result
store_result:
lea rcx, [rip + val_stack]
mov qword ptr [rcx + r13*8 - 8], rax
jmp token_seek
try_literal:
mov rsi, r8
mov rcx, rdx
xor eax, eax
xor r11d, r11d
xor r10d, r10d
cmp byte ptr [rsi], 45
jne lit_digit_loop
mov r11d, 1
inc rsi
dec rcx
lit_digit_loop:
test rcx, rcx
jz lit_done
movzx r9, byte ptr [rsi]
sub r9d, 48
cmp r9d, 9
ja invalid_token
inc r10d
cmp r10d, 18
ja invalid_token
imul rax, rax, 10
add rax, r9
inc rsi
dec rcx
jmp lit_digit_loop
lit_done:
test r10d, r10d
jz invalid_token
test r11d, r11d
jz lit_push
neg rax
lit_push:
lea rcx, [rip + val_stack]
mov qword ptr [rcx + r13*8], rax
inc r13
jmp token_seek
invalid_token:
mov r14d, 1
jmp token_seek
set_error:
mov r14d, 1
jmp token_seek
scan_to_eol:
cmp rbx, r12
jae end_line
cmp byte ptr [rbx], 10
je end_line
inc rbx
jmp scan_to_eol
end_line:
cmp rbx, r12
jae end_line_output
cmp byte ptr [rbx], 10
jne end_line_output
inc rbx
end_line_output:
test r14d, r14d
jnz print_error
cmp r13, 1
jne print_error
lea rcx, [rip + val_stack]
mov rax, qword ptr [rcx]
xor r8d, r8d
test rax, rax
jns format_setup
mov r8d, 1
neg rax
format_setup:
lea r9, [rip + num_buf]
add r9, 32
mov rsi, r9
format_loop:
xor edx, edx
mov rcx, 10
div rcx
dec rsi
add dl, 48
mov byte ptr [rsi], dl
test rax, rax
jnz format_loop
test r8d, r8d
jz copy_number
dec rsi
mov byte ptr [rsi], 45
copy_number:
cmp rsi, r9
jae copy_done
movzx eax, byte ptr [rsi]
mov byte ptr [r15], al
inc rsi
inc r15
jmp copy_number
copy_done:
mov byte ptr [r15], 10
inc r15
jmp line_loop
print_error:
mov byte ptr [r15], 101
mov byte ptr [r15 + 1], 114
mov byte ptr [r15 + 2], 114
mov byte ptr [r15 + 3], 111
mov byte ptr [r15 + 4], 114

write_loop:
cmp rsi, r15
jae do_exit
mov eax, 1
mov edi, 1
mov rdx, r15
sub rdx, rsi
syscall
test rax, rax
jz do_exit
js do_exit
add rsi, rax
jmp write_loop

mov eax, 60
xor edi, edi
syscall
