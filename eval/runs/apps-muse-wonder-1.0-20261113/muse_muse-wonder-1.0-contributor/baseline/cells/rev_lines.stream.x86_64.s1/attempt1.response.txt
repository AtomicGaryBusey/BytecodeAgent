.globl _start
.text
_start:
    xor ebx, ebx
read_loop:
    cmp ebx, 65536
    jae read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub edx, ebx
    xor edi, edi
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r8d, r8d
    xor r9d, r9d
    lea r10, [rip + inbuf]
    lea r11, [rip + outbuf]
line_loop:
    cmp r8, rbx
    jae write_output
    mov r12, r8
    mov r13, r8
scan_loop:
    cmp r13, rbx
    jae scan_done
    mov al, byte ptr [r10 + r13]
    cmp al, 10
    je scan_done
    inc r13
    jmp scan_loop
scan_done:
    mov rcx, r13
rev_loop:
    cmp rcx, r12
    jbe rev_done
    dec rcx
    mov al, byte ptr [r10 + rcx]
    mov byte ptr [r11 + r9], al
    inc r9
    jmp rev_loop
rev_done:
    mov byte ptr [r11 + r9], 10
    inc r9
    mov r8, r13
    cmp r13, rbx
    jae line_loop
    inc r8
    jmp line_loop
write_output:
    xor r12d, r12d
write_loop:
    cmp r12, r9
    jae exit_program
    lea rsi, [rip + outbuf]
    add rsi, r12
    mov rdx, r9
    sub rdx, r12
    mov edi, 1
    mov eax, 1
    syscall
    cmp rax, 0
    jle exit_program
    add r12, rax
    jmp write_loop
exit_program:
    xor edi, edi
    mov eax, 60
    syscall
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
