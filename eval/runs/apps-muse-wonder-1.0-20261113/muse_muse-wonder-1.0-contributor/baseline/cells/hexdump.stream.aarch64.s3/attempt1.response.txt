.globl _start

.text
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #65536
    cmp x21, x0
    b.hs read_done
    mov x0, #0
    add x1, x20, x21
    mov x2, #65536
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

row_loop:
    cmp x22, x21
    b.hs final_line
    mov x23, x21
    sub x23, x23, x22
    cmp x23, #16
    b.lo row_size_done
    mov x23, #16
row_size_done:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x25, #0
    mov x26, #28

offset_loop:
    lsr x0, x22, x26
    and x0, x0, #15
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits
    ldrb w0, [x1, x0]
    strb w0, [x24, x25]
    add x25, x25, #1
    subs x26, x26, #4
    b.ge offset_loop

    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1

    mov x27, #0

slot_loop:
    cmp x27, #16
    b.hs slot_done
    cmp x27, x23
    b.hs pad_slot
    add x0, x22, x27
    ldrb w28, [x20, x0]
    mov w0, w28
    lsr w0, w0, #4
    and w0, w0, #15
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits
    ldrb w0, [x1, x0]
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, w28
    and w0, w0, #15
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits
    ldrb w0, [x1, x0]
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
    b after_slot

pad_slot:
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1

after_slot:
    cmp x27, #7
    b.ne no_extra_space
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
no_extra_space:
    add x27, x27, #1
    b slot_loop

slot_done:
    mov w0, #32
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #124
    strb w0, [x24, x25]
    add x25, x25, #1

    mov x27, #0

ascii_loop:
    cmp x27, x23
    b.hs ascii_done
    add x0, x22, x27
    ldrb w28, [x20, x0]
    cmp w28, #32
    b.lo ascii_dot
    cmp w28, #126
    b.hi ascii_dot
    mov w0, w28
    b ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    strb w0, [x24, x25]
    add x25, x25, #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24, x25]
    add x25, x25, #1
    mov w0, #10
    strb w0, [x24, x25]
    add x25, x25, #1

    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0

    add x22, x22, x23
    b row_loop

final_line:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x25, #0
    mov x26, #28

final_loop:
    lsr x0, x21, x26
    and x0, x0, #15
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits
    ldrb w0, [x1, x0]
    strb w0, [x24, x25]
    add x25, x25, #1
    subs x26, x26, #4
    b.ge final_loop

    mov w0, #10
    strb w0, [x24, x25]
    add x25, x25, #1

    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

exit_hang:
    b exit_hang

.section .rodata
.balign 16
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
out_buf:
    .zero 128
