.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jle print_result
    add r14, rax
    mov rcx, rax
    lea rbx, [rip + buffer]
byte_loop:
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jb 3f
    cmp al, 13
    ja 3f
2:
    xor r15, r15
    jmp 4f
3:
    test r15, r15
    jnz 4f
    inc r13
    mov r15, 1
4:
    dec rcx
    jnz byte_loop
    jmp read_loop
print_result:
    lea r8, [rip + outbuf_end]
    dec r8
    mov byte ptr [r8], 10
    mov rax, r14
    call convert_number
    dec r8
    mov byte ptr [r8], 32
    mov rax, r13
    call convert_number
    dec r8
    mov byte ptr [r8], 32
    mov rax, r12
    call convert_number
    mov rsi, r8
    lea rdx, [rip + outbuf_end]
    sub rdx, r8
write_loop:
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
convert_number:
    mov r9, 10
1:
    xor edx, edx
    div r9
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    ret
.bss
.balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
outbuf_end:
