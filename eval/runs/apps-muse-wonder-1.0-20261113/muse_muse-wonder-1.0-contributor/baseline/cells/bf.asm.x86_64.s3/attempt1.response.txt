.text
.globl _start
_start:
    xor ebx, ebx
    lea r12, [rip + buffer]
read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [r12 + rcx], 33
    je bang_found
    inc rcx
    jmp find_loop
bang_found:
    mov r13, rcx
    lea r14, [rcx + 1]
    jmp split_done
no_bang:
    mov r13, rbx
    mov r14, rbx
split_done:
    lea r8, [rip + cmds]
    xor ecx, ecx
    xor edx, edx
filter_loop:
    cmp rcx, r13
    jae filter_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 62
    je keep_cmd
    cmp al, 60
    je keep_cmd
    cmp al, 43
    je keep_cmd
    cmp al, 45
    je keep_cmd
    cmp al, 46
    je keep_cmd
    cmp al, 44
    je keep_cmd
    cmp al, 91
    je keep_cmd
    cmp al, 93
    je keep_cmd
    inc rcx
    jmp filter_loop
keep_cmd:
    mov byte ptr [r8 + rdx], al
    inc rdx
    inc rcx
    jmp filter_loop
filter_done:
    mov r15, rdx
    lea r9, [rip + match]
    mov r10, rsp
    xor ecx, ecx
build_loop:
    cmp rcx, r15
    jae build_done
    mov al, byte ptr [r8 + rcx]
    cmp al, 91
    je build_open
    cmp al, 93
    je build_close
    inc rcx
    jmp build_loop
build_open:
    push rcx
    inc rcx
    jmp build_loop
build_close:
    cmp rsp, r10
    je unbalanced
    pop rax
    mov dword ptr [r9 + rcx*4], eax
    mov dword ptr [r9 + rax*4], ecx
    inc rcx
    jmp build_loop
build_done:
    cmp rsp, r10
    jne unbalanced
    jmp exec_init
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
exec_init:
    lea r10, [rip + tape]
    xor r13d, r13d
    xor ebp, ebp
exec_loop:
    cmp r13, r15
    jae exit_success
    mov al, byte ptr [r8 + r13]
    cmp al, 62
    je do_inc_ptr
    cmp al, 60
    je do_dec_ptr
    cmp al, 43
    je do_inc_cell
    cmp al, 45
    je do_dec_cell
    cmp al, 46
    je do_output
    cmp al, 44
    je do_input
    cmp al, 91
    je do_open
    cmp byte ptr [r10 + rbp], 0
    je next_ip
    mov r13d, dword ptr [r9 + r13*4]
    jmp next_ip
do_inc_ptr:
    inc rbp
    jmp next_ip
do_dec_ptr:
    dec rbp
    jmp next_ip
do_inc_cell:
    inc byte ptr [r10 + rbp]
    jmp next_ip
do_dec_cell:
    dec byte ptr [r10 + rbp]
    jmp next_ip
do_output:
    mov eax, 1
    mov edi, 1
    lea rsi, [r10 + rbp]
    mov edx, 1
    syscall
    jmp next_ip
do_input:
    cmp r14, rbx
    jae do_input_eof
    mov al, byte ptr [r12 + r14]
    mov byte ptr [r10 + rbp], al
    inc r14
    jmp next_ip
do_input_eof:
    mov byte ptr [r10 + rbp], 0
    jmp next_ip
do_open:
    cmp byte ptr [r10 + rbp], 0
    jne next_ip
    mov r13d, dword ptr [r9 + r13*4]
    jmp next_ip
next_ip:
    inc r13
    jmp exec_loop
exit_success:
    mov eax, 60
    xor edi, edi
    syscall
.bss
.balign 8
buffer:
    .zero 65536
cmds:
    .zero 65536
match:
    .zero 262144
tape:
    .zero 30000
