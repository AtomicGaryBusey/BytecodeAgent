.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    xor ebx, ebx
    lea rbp, [rip + input_buf]
next_line:
    cmp rbx, r12
    jae exit_program
    xor r13d, r13d
token_loop:
skip_separators:
    cmp rbx, r12
    jae finish_line
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 10
    je consume_newline
    cmp al, 32
    je skip_one_separator
    cmp al, 9
    je skip_one_separator
    jmp token_start
skip_one_separator:
    inc rbx
    jmp skip_separators
consume_newline:
    inc rbx
    jmp finish_line
token_start:
    mov r8, rbx
token_scan:
    cmp rbx, r12
    jae token_ready
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc rbx
    jmp token_scan
token_ready:
    mov r9, rbx
    sub r9, r8
    cmp r9, 1
    jne literal_check
    movzx eax, byte ptr [rbp + r8]
    cmp al, 43
    je operator
    cmp al, 45
    je operator
    cmp al, 42
    je operator
    cmp al, 47
    je operator
    cmp al, 37
    je operator
literal_check:
    xor r10d, r10d
    xor r11d, r11d
    lea rsi, [rbp + r8]
    movzx eax, byte ptr [rsi]
    cmp al, 45
    jne digit_count
    mov r11d, 1
    mov r10d, 1
digit_count:
    mov rcx, r9
    sub rcx, r10
    cmp rcx, 1
    jb line_error_skip
    cmp rcx, 18
    ja line_error_skip
    xor eax, eax
parse_digit_loop:
    cmp r10, r9
    jae parse_done
    movzx edx, byte ptr [rsi + r10]
    cmp dl, 48
    jb line_error_skip
    cmp dl, 57
    ja line_error_skip
    sub edx, 48
    imul rax, rax, 10
    add rax, rdx
    inc r10
    jmp parse_digit_loop
parse_done:
    test r11d, r11d
    jz push_value
    neg rax
push_value:
    lea rcx, [rip + val_stack]
    mov qword ptr [rcx + r13*8], rax
    inc r13
    jmp token_loop
operator:
    mov r8d, eax
    cmp r13, 2
    jb line_error_skip
    lea rcx, [rip + val_stack]
    mov rax, qword ptr [rcx + r13*8 - 16]
    mov rdx, qword ptr [rcx + r13*8 - 8]
    dec r13
    cmp r8b, 43
    je op_add
    cmp r8b, 45
    je op_sub
    cmp r8b, 42
    je op_mul
    test rdx, rdx
    jz line_error_skip
    cmp rdx, -1
    jne do_division
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    je line_error_skip
do_division:
    mov rsi, rdx
    cqo
    idiv rsi
    cmp r8b, 37
    jne store_result
    mov rax, rdx
    jmp store_result
op_add:
    add rax, rdx
    jmp store_result
op_sub:
    sub rax, rdx
    jmp store_result
op_mul:
    imul rax, rdx
store_result:
    lea rcx, [rip + val_stack]
    mov qword ptr [rcx + r13*8 - 8], rax
    jmp token_loop
line_error_skip:
error_skip_loop:
    cmp rbx, r12
    jae print_error
    movzx eax, byte ptr [rbp + rbx]
    inc rbx
    cmp al, 10
    jne error_skip_loop
    jmp print_error
finish_line:
    cmp r13, 1
    jne print_error
    lea rcx, [rip + val_stack]
    mov rax, qword ptr [rcx]
    lea rsi, [rip + out_buf]
    add rsi, 64
    mov r9, rsi
    dec rsi
    mov byte ptr [rsi], 10
    xor r8d, r8d
    cmp rax, 0
    jge print_digit_loop
    mov r8d, 1
    neg rax
print_digit_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne print_digit_loop
    test r8d, r8d
    jz print_write
    dec rsi
    mov byte ptr [rsi], 45
print_write:
    mov rdx, r9
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    jmp next_line
print_error:
    lea rsi, [rip + err_msg]
    mov edx, 6
    mov edi, 1
    mov eax, 1
    syscall
    jmp next_line
exit_program:
    xor edi, edi
    mov eax, 60
    syscall
    jmp exit_program
.section .rodata
err_msg:
    .ascii "error\n"
.bss
.balign 16
input_buf:
    .zero 65536
.balign 8
val_stack:
    .zero 8192
.balign 16
out_buf:
    .zero 64
