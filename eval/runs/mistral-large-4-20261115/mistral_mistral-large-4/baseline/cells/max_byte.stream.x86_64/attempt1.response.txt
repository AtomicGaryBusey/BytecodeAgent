max_byte:
    xor eax, eax
    pxor xmm0, xmm0
    test rsi, rsi
    jz .Ldone
    mov rcx, rsi
    shr rcx, 4
    jz .Lreduce
.Lloop:
    movdqu xmm1, [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    dec rcx
    jnz .Lloop
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    and eax, 0xff
    and rsi, 15
    jz .Ldone
.Ltail:
    movzx edx, byte ptr [rdi]
    cmp edx, eax
    cmova eax, edx
    inc rdi
    dec rsi
    jnz .Ltail
.Ldone:
    ret
