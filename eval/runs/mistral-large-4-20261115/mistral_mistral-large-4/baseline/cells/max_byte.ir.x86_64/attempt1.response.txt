export function l $max_byte(l %p, l %n) {
@start
    %i =l copy 0
    %max =l copy 0
    j @next
@next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csltl %max, %b
    jnz %gt, @update, @inc
@update
    %max =l copy %b
    j @inc
@inc
    %i =l add %i, 1
    j @next
@done
    ret %max
}
