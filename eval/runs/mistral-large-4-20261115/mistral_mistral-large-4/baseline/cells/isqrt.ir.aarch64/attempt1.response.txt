export function l $isqrt(l %n) {
@start
    %lo0 =l copy 0
    %hi0 =l copy 4294967295
    jmp @loop
@loop
    %lo =l phi @start %lo0, @then %lo_then, @else %lo_else
    %hi =l phi @start %hi0, @then %hi_then, @else %hi_else
    %lt =w cultl %lo, %hi
    jnz %lt, @body, @done
@body
    %sum =l add %lo, %hi
    %sum1 =l add %sum, 1
    %two =l copy 2
    %mid =l div %sum1, %two
    %sq =l mul %mid, %mid
    %le =w culel %sq, %n
    jnz %le, @then, @else
@then
    %lo_then =l copy %mid
    %hi_then =l copy %hi
    jmp @loop
@else
    %lo_else =l copy %lo
    %one =l copy 1
    %hi_else =l sub %mid, %one
    jmp @loop
@done
    ret %lo
}
