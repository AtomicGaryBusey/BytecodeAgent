export function l $crc32(l %p, l %n) {
@start
	%init =w copy 0xffffffff
	%crc =w copy 0xffffffff
	%poly =w copy 0xedb88320
	%onew =w copy 1
	%zerow =w copy 0
	%onel =l copy 1
	%zerol =l copy 0
	%eightl =l copy 8
	%rem =l copy %n
@outer_cond
	%done =w ceql %rem, %zerol
	jnz %done, @finish, @outer_body
@outer_body
	%b =w loadub %p
	%crc =w xor %crc, %b
	%i =l copy %zerol
@inner_cond
	%idone =w ceql %i, %eightl
	jnz %idone, @inner_end, @inner_body
@inner_body
	%lsb =w and %crc, %onew
	%neg =w sub %zerow, %lsb
	%masked =w and %poly, %neg
	%shr =w shr %crc, %onew
	%crc =w xor %shr, %masked
	%i =l add %i, %onel
	jmp @inner_cond
@inner_end
	%p =l add %p, %onel
	%rem =l sub %rem, %onel
	jmp @outer_cond
@finish
	%resw =w xor %crc, %init
	%res =l extuw %resw
	ret %res
}
