mat_pow:
  cld
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 8
  mov rbx, rsi
  mov rbp, rdx
  mov r12, rdi
  mov r13, rcx
  mov r14, r8
  mov rax, rbx
  imul rax, rbx
  lea r15, [r14 + rax*8]
  mov rdi, r13
  mov rcx, rax
  xor eax, eax
  rep stosq
  xor r8, r8
.Ldiag:
  cmp r8, rbx
  jae .Ldiag_done
  mov rax, r8
  imul rax, rbx
  add rax, r8
  mov qword ptr [r13 + rax*8], 1
  inc r8
  jmp .Ldiag
.Ldiag_done:
.Lloop:
  test rbp, rbp
  jz .Lzero_tmp
  xor r8, r8
.Li:
  cmp r8, rbx
  jae .Li_done
  mov rax, r8
  imul rax, rbx
  lea rcx, [r13 + rax*8]
  lea rsi, [r15 + rax*8]
  xor r9, r9
.Lj:
  cmp r9, rbx
  jae .Lj_done
  xor rax, rax
  xor r10, r10
.Lt:
  cmp r10, rbx
  jae .Lt_done
  mov rdx, r10
  imul rdx, rbx
  add rdx, r9
  mov rdx, qword ptr [r12 + rdx*8]
  imul rdx, qword ptr [rcx + r10*8]
  add rax, rdx
  inc r10
  jmp .Lt
.Lt_done:
  mov qword ptr [rsi + r9*8], rax
  inc r9
  jmp .Lj
.Lj_done:
  inc r8
  jmp .Li
.Li_done:
  mov rsi, r15
  mov rdi, r13
  mov rcx, rbx
  imul rcx, rbx
  rep movsq
  dec rbp
  jmp .Lloop
.Lzero_tmp:
  mov rdi, r14
  mov rcx, rbx
  imul rcx, rbx
  shl rcx, 1
  xor eax, eax
  rep stosq
  xor eax, eax
  add rsp, 8
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
