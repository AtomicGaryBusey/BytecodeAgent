export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l copy 0
	%m1 =l copy -1
	%m2 =l copy -2
	jmp @loop

@loop
	%pos =l phi @start %zero, @lit_full_done %pos_lit, @rep_full_done %pos_rep, @nop %pos_nop
	%written =l phi @start %zero, @lit_full_done %written_lit, @rep_full_done %written_rep, @nop %written_nop
	%more =w csltl %pos, %n
	jnz %more, @body, @done

@done
	ret %written

@body
	%sptr =l add %src, %pos
	%hb =w loadsb %sptr
	%h =l extsw %hb
	%is_neg =w csltl %h, 0
	jnz %is_neg, @neg, @lit

@lit
	%cnt_lit =l add %h, 1
	%rem_lit =l sub %n, %pos
	%need_lit =l add %cnt_lit, 1
	%bad_lit =w csltl %rem_lit, %need_lit
	jnz %bad_lit, @ret_m2, @lit_check_cap

@lit_check_cap
	%space_lit =l sub %cap, %written
	%over_lit =w csgtl %cnt_lit, %space_lit
	jnz %over_lit, @lit_cap, @lit_full_prep

@lit_cap
	%dptr_lc =l add %sptr, 1
	%optr_lc =l add %out, %written
	jmp @lit_cap_loop

@lit_cap_loop
	%i_lc =l phi @lit_cap %zero, @lit_cap_body %i_lc_next
	%more_lc =w csltl %i_lc, %space_lit
	jnz %more_lc, @lit_cap_body, @lit_cap_done

@lit_cap_body
	%sp_lc =l add %dptr_lc, %i_lc
	%dp_lc =l add %optr_lc, %i_lc
	%b_lc =w loadub %sp_lc
	storeb %b_lc, %dp_lc
	%i_lc_next =l add %i_lc, 1
	jmp @lit_cap_loop

@lit_cap_done
	ret %m1

@lit_full_prep
	%dptr_lf =l add %sptr, 1
	%optr_lf =l add %out, %written
	jmp @lit_full_loop

@lit_full_loop
	%i_lf =l phi @lit_full_prep %zero, @lit_full_body %i_lf_next
	%more_lf =w csltl %i_lf, %cnt_lit
	jnz %more_lf, @lit_full_body, @lit_full_done

@lit_full_body
	%sp_lf =l add %dptr_lf, %i_lf
	%dp_lf =l add %optr_lf, %i_lf
	%b_lf =w loadub %sp_lf
	storeb %b_lf, %dp_lf
	%i_lf_next =l add %i_lf, 1
	jmp @lit_full_loop

@lit_full_done
	%written_lit =l add %written, %cnt_lit
	%pos_lit =l add %pos, %need_lit
	jmp @loop

@neg
	%is_m128 =w ceql %h, -128
	jnz %is_m128, @nop, @rep

@nop
	%pos_nop =l add %pos, 1
	%written_nop =l copy %written
	jmp @loop

@rep
	%rem_rep =l sub %n, %pos
	%bad_rep =w csltl %rem_rep, 2
	jnz %bad_rep, @ret_m2, @rep_check_cap

@rep_check_cap
	%cnt_rep =l sub 1, %h
	%space_rep =l sub %cap, %written
	%over_rep =w csgtl %cnt_rep, %space_rep
	jnz %over_rep, @rep_cap, @rep_full_prep

@rep_cap
	%dptr_rc =l add %sptr, 1
	%byte_rc =w loadub %dptr_rc
	%optr_rc =l add %out, %written
	jmp @rep_cap_loop

@rep_cap_loop
	%i_rc =l phi @rep_cap %zero, @rep_cap_body %i_rc_next
	%more_rc =w csltl %i_rc, %space_rep
	jnz %more_rc, @rep_cap_body, @rep_cap_done

@rep_cap_body
	%dp_rc =l add %optr_rc, %i_rc
	storeb %byte_rc, %dp_rc
	%i_rc_next =l add %i_rc, 1
	jmp @rep_cap_loop

@rep_cap_done
	ret %m1

@rep_full_prep
	%dptr_rf =l add %sptr, 1
	%byte_rf =w loadub %dptr_rf
	%optr_rf =l add %out, %written
	jmp @rep_full_loop

@rep_full_loop
	%i_rf =l phi @rep_full_prep %zero, @rep_full_body %i_rf_next
	%more_rf =w csltl %i_rf, %cnt_rep
	jnz %more_rf, @rep_full_body, @rep_full_done

@rep_full_body
	%dp_rf =l add %optr_rf, %i_rf
	storeb %byte_rf, %dp_rf
	%i_rf_next =l add %i_rf, 1
	jmp @rep_full_loop

@rep_full_done
	%written_rep =l add %written, %cnt_rep
	%pos_rep =l add %pos, 2
	jmp @loop

@ret_m2
	ret %m2
}
