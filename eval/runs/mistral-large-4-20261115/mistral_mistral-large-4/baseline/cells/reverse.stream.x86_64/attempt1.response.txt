reverse:
    test rsi, rsi
    je .Ldone
    lea rdx, [rsi - 1]
    lea r8, [rdi + rdx*8]
.Lloop:
    cmp rdi, r8
    jae .Ldone
    mov rax, qword ptr [rdi]
    mov rcx, qword ptr [r8]
    mov qword ptr [rdi], rcx
    mov qword ptr [r8], rax
    add rdi, 8
    sub r8, 8
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
