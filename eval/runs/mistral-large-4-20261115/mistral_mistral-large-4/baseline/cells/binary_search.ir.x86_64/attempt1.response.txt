export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @done
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%lt =w csltl %key, %val
	jnz %lt, @left, @checkeq
@checkeq
	%eq =w ceql %key, %val
	jnz %eq, @found, @right
@left
	%hi =l copy %mid
	jmp @loop
@right
	%midp1 =l add %mid, 1
	%lo =l copy %midp1
	jmp @loop
@found
	ret %mid
@done
	%minus =l copy -1
	ret %minus
}
