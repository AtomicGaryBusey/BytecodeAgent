export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %zero =l copy 0
    %n =l add %na, %nb
    jmp @zero_check
@zero_check
    %zi =l phi @start %zero, @zero_body %zi_next
    %zc =w cultl %zi, %n
    jnz %zc, @zero_body, @zero_done
@zero_body
    %zi8 =l shl %zi, 3
    %zptr =l add %r, %zi8
    storel %zero, %zptr
    %zi_next =l add %zi, 1
    jmp @zero_check
@zero_done
    %zc1 =w ceql %na, %zero
    %zc2 =w ceql %nb, %zero
    %zcz =w or %zc1, %zc2
    jnz %zcz, @return_zero, @outer_check
@return_zero
    ret %zero
@outer_check
    %i =l phi @zero_done %zero, @outer_next %i_next
    %oc =w cultl %i, %na
    jnz %oc, @inner_check, @done
@inner_check
    %j =l phi @outer_check %zero, @inner_body %j_next
    %carry =l phi @outer_check %zero, @inner_body %carry_next
    %ic =w cultl %j, %nb
    jnz %ic, @inner_body, @inner_done
@inner_body
    %ai8 =l shl %i, 3
    %aptr =l add %a, %ai8
    %av =l loadl %aptr
    %bj8 =l shl %j, 3
    %bptr =l add %b, %bj8
    %bv =l loadl %bptr
    %x0 =l and %av, 4294967295
    %x1 =l shr %av, 32
    %y0 =l and %bv, 4294967295
    %y1 =l shr %bv, 32
    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1
    %mid =l add %p01, %p10
    %mid_c =w cultl %mid, %p01
    %mid_shift =l shl %mid, 32
    %lo =l add %p00, %mid_shift
    %lo_c =w cultl %lo, %p00
    %mid_hi =l shr %mid, 32
    %mid_c_l =l extuw %mid_c
    %mid_c_shift =l shl %mid_c_l, 32
    %extra =l add %mid_hi, %mid_c_shift
    %lo_c_l =l extuw %lo_c
    %extra2 =l add %extra, %lo_c_l
    %hi =l add %p11, %extra2
    %k =l add %i, %j
    %k8 =l shl %k, 3
    %rptr =l add %r, %k8
    %rv =l loadl %rptr
    %s1 =l add %rv, %lo
    %lc1 =w cultl %s1, %rv
    %s2 =l add %s1, %carry
    %lc2 =w cultl %s2, %s1
    %lc1l =l extuw %lc1
    %lc2l =l extuw %lc2
    %lcsum =l add %lc1l, %lc2l
    storel %s2, %rptr
    %carry_next =l add %hi, %lcsum
    %j_next =l add %j, 1
    jmp @inner_check
@inner_done
    %k2 =l add %i, %nb
    %carry_init =l copy %carry
    %cz2 =w ceql %carry_init, %zero
    jnz %cz2, @outer_next, @prop_check
@prop_check
    %pk =l phi @inner_done %k2, @prop_body %pk_next
    %pc =l phi @inner_done %carry_init, @prop_body %pc_next
    %pc_zero =w ceql %pc, %zero
    %pc_nonzero =w sub 1, %pc_zero
    %pk_lt_n =w cultl %pk, %n
    %pcond =w and %pc_nonzero, %pk_lt_n
    jnz %pcond, @prop_body, @outer_next
@prop_body
    %pk8 =l shl %pk, 3
    %pptr =l add %r, %pk8
    %pv =l loadl %pptr
    %psum =l add %pv, %pc
    %pc_carry =w cultl %psum, %pv
    storel %psum, %pptr
    %pc_next =l extuw %pc_carry
    %pk_next =l add %pk, 1
    jmp @prop_check
@outer_next
    %i_next =l add %i, 1
    jmp @outer_check
@done
    jmp @sig_check
@sig_check
    %idx =l phi @done %zero, @sig_body %idx_next
    %sig =l phi @done %zero, @sig_body %sig_next
    %sc =w cultl %idx, %n
    jnz %sc, @sig_body, @sig_ret
@sig_body
    %idx8 =l shl %idx, 3
    %sptr =l add %r, %idx8
    %sv =l loadl %sptr
    %sv_zero =w ceql %sv, %zero
    %isnz =w sub 1, %sv_zero
    %idxp1 =l add %idx, 1
    %diff =l sub %idxp1, %sig
    %isnzl =l extuw %isnz
    %sig_add =l mul %diff, %isnzl
    %sig_next =l add %sig, %sig_add
    %idx_next =l add %idx, 1
    jmp @sig_check
@sig_ret
    ret %sig
}
