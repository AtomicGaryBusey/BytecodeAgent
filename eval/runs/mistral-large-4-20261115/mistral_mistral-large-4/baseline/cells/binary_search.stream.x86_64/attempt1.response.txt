binary_search:
    test rsi, rsi
    jz .Lnot_found
    xor r8, r8
    lea r9, [rsi - 1]
.Lloop:
    cmp r8, r9
    jg .Lnot_found
    mov r10, r9
    sub r10, r8
    sar r10, 1
    add r10, r8
    mov rax, qword ptr [rdi + r10*8]
    cmp rax, rdx
    je .Lfound
    jl .Llower
    lea r9, [r10 - 1]
    jmp .Lloop
.Llower:
    lea r8, [r10 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r10
    ret
.Lnot_found:
    mov rax, -1
    ret
