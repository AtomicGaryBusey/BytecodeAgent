export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @inc %inext
	%p =l phi @start %a, @inc %pnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%val =l loadl %p
	%zero =w ceql %val, 0
	jnz %zero, @done, @inc
@inc
	%pnext =l add %p, 8
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
