export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %c0 =l copy 0
  %c1 =l copy 1
  %c8 =l copy 8
  %n1 =l add %n, %c1
  %j0 =l copy %c0
  jmp @zero_loop
@zero_loop
  %j =l phi @start %j0, @zero_body %j_next
  %zc =w csltl %j, %n1
  jnz %zc, @zero_body, @zero_done
@zero_body
  %zoff =l mul %j, %c8
  %zaddr =l add %row, %zoff
  storel %c0, %zaddr
  %j_next =l add %j, %c1
  jmp @zero_loop
@zero_done
  %i0 =l copy %c0
  jmp @outer
@outer
  %i =l phi @zero_done %i0, @outer_next %i_next
  %oc =w csltl %i, %m
  jnz %oc, @outer_body, @done
@outer_body
  %xaddr =l add %x, %i
  %xb =w loadub %xaddr
  %diag0 =l copy %c0
  %jstart =l copy %c1
  jmp @inner
@inner
  %j =l phi @outer_body %jstart, @store %j_next
  %diag =l phi @outer_body %diag0, @store %diag_next
  %ic =w csltl %j, %n1
  jnz %ic, @inner_body, @outer_next
@inner_body
  %joff =l mul %j, %c8
  %jaddr =l add %row, %joff
  %up =l loadl %jaddr
  %jm1 =l sub %j, %c1
  %yaddr =l add %y, %jm1
  %yb =w loadub %yaddr
  %eq =w ceqw %xb, %yb
  jnz %eq, @match, @nomatch
@match
  %new_match =l add %diag, %c1
  jmp @store
@nomatch
  %ljoff =l mul %jm1, %c8
  %ljaddr =l add %row, %ljoff
  %left =l loadl %ljaddr
  %lt =w csltl %up, %left
  jnz %lt, @useleft, @useup
@useleft
  %new_left =l copy %left
  jmp @store_nm
@useup
  %new_up =l copy %up
  jmp @store_nm
@store_nm
  %new_nm =l phi @useleft %new_left, @useup %new_up
  jmp @store
@store
  %newv =l phi @match %new_match, @store_nm %new_nm
  storel %newv, %jaddr
  %diag_next =l copy %up
  %j_next =l add %j, %c1
  jmp @inner
@outer_next
  %i_next =l add %i, %c1
  jmp @outer
@done
  %noff =l mul %n, %c8
  %naddr =l add %row, %noff
  %ans =l loadl %naddr
  ret %ans
}
