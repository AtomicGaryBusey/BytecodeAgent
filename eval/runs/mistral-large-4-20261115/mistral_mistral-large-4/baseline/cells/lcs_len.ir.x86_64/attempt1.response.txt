export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%n1 =l add %n, 1
	%rowbytes1 =l mul %n1, 8
	%rowend =l add %row, %rowbytes1
	%p =l copy %row
@zero_loop
	%zcont =w csltl %p, %rowend
	jnz %zcont, @zero_body, @zero_done
@zero_body
	storel %p, 0
	%p =l add %p, 8
	br @zero_loop
@zero_done
	%i =l copy 0
@outer
	%ocont =w csltl %i, %m
	jnz %ocont, @outer_body, @outer_done
@outer_body
	%xi_addr =l add %x, %i
	%xi =w loadub %xi_addr
	%diag =l copy 0
	%rowp =l add %row, 8
	%yp =l copy %y
@inner
	%icont =w csltl %rowp, %rowend
	jnz %icont, @inner_body, @inner_done
@inner_body
	%up =l loadl %rowp
	%yb =w loadub %yp
	%leftp =l sub %rowp, 8
	%left =l loadl %leftp
	%gt =w csgtl %up, %left
	%maxv =l select %gt, %up, %left
	%eq =w ceqw %xi, %yb
	%diagp1 =l add %diag, 1
	%new =l select %eq, %diagp1, %maxv
	storel %rowp, %new
	%diag =l copy %up
	%rowp =l add %rowp, 8
	%yp =l add %yp, 1
	br @inner
@inner_done
	%i =l add %i, 1
	br @outer
@outer_done
	%nbytes =l mul %n, 8
	%resaddr =l add %row, %nbytes
	%res =l loadl %resaddr
	ret %res
}
