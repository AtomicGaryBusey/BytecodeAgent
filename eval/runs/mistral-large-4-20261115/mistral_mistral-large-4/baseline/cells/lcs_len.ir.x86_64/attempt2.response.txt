export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%n1 =l add %n, 1
	%rowbytes1 =l mul %n1, 8
	%rowend =l add %row, %rowbytes1
	jmp @zero_loop
@zero_loop
	%p =l phi @start %row, @zero_body %pnext
	%zcont =w csltl %p, %rowend
	jnz %zcont, @zero_body, @zero_done
@zero_body
	storel %p, %zero
	%pnext =l add %p, 8
	jmp @zero_loop
@zero_done
	jmp @outer
@outer
	%i =l phi @zero_done %zero, @inner_done %inext
	%ocont =w csltl %i, %m
	jnz %ocont, @outer_body, @outer_done
@outer_body
	%xi_addr =l add %x, %i
	%xi =w loadub %xi_addr
	%rowplus8 =l add %row, 8
	jmp @inner
@inner
	%rowp =l phi @outer_body %rowplus8, @inner_body %rowpnext
	%yp =l phi @outer_body %y, @inner_body %ypnext
	%diag =l phi @outer_body %zero, @inner_body %diagnext
	%icont =w csltl %rowp, %rowend
	jnz %icont, @inner_body, @inner_done
@inner_body
	%up =l loadl %rowp
	%yb =w loadub %yp
	%leftp =l sub %rowp, 8
	%left =l loadl %leftp
	%gt =w csgtl %up, %left
	%maxv =l select %gt, %up, %left
	%eq =w ceqw %xi, %yb
	%diagp1 =l add %diag, 1
	%new =l select %eq, %diagp1, %maxv
	storel %rowp, %new
	%diagnext =l copy %up
	%rowpnext =l add %rowp, 8
	%ypnext =l add %yp, 1
	jmp @inner
@inner_done
	%inext =l add %i, 1
	jmp @outer
@outer_done
	%nbytes =l mul %n, 8
	%resaddr =l add %row, %nbytes
	%res =l loadl %resaddr
	ret %res
}
