export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%n2 =l mul %n, %n
	%tmp_elems =l add %n2, %n2
	br @zero_out
@zero_out
	%zi =l phi @start %zero, @zero_out_body %zi_next
	%zc =w csltl %zi, %n2
	jnz %zc, @zero_out_body, @zero_out_done
@zero_out_body
	%zoff =l mul %zi, %eight
	%zptr =l add %out, %zoff
	storel %zero, %zptr
	%zi_next =l add %zi, %one
	br @zero_out
@zero_out_done
	br @set_diag
@set_diag
	%di =l phi @zero_out_done %zero, @set_diag_body %di_next
	%dc =w csltl %di, %n
	jnz %dc, @set_diag_body, @set_diag_done
@set_diag_body
	%drow =l mul %di, %n
	%didx =l add %drow, %di
	%doff =l mul %didx, %eight
	%dptr =l add %out, %doff
	storel %one, %dptr
	%di_next =l add %di, %one
	br @set_diag
@set_diag_done
	br @exp_loop
@exp_loop
	%e =l phi @set_diag_done %zero, @copy_done %e_next
	%ec =w csltl %e, %k
	jnz %ec, @prod_i, @exp_done
@prod_i
	%pi =l phi @exp_loop %zero, @prod_j_done %pi_next
	%pic =w csltl %pi, %n
	jnz %pic, @prod_j, @prod_i_done
@prod_j
	%pj =l phi @prod_i %zero, @prod_t_done %pj_next
	%pjc =w csltl %pj, %n
	jnz %pjc, @prod_t, @prod_j_done
@prod_t
	%pt =l phi @prod_j %zero, @prod_t_body %pt_next
	%sum =l phi @prod_j %zero, @prod_t_body %sum_new
	%ptc =w csltl %pt, %n
	jnz %ptc, @prod_t_body, @prod_t_done
@prod_t_body
	%out_row =l mul %pi, %n
	%out_idx =l add %out_row, %pt
	%out_off =l mul %out_idx, %eight
	%out_ptr =l add %out, %out_off
	%x =l loadl %out_ptr
	%a_row =l mul %pt, %n
	%a_idx =l add %a_row, %pj
	%a_off =l mul %a_idx, %eight
	%a_ptr =l add %a, %a_off
	%y =l loadl %a_ptr
	%xy =l mul %x, %y
	%sum_new =l add %sum, %xy
	%pt_next =l add %pt, %one
	br @prod_t
@prod_t_done
	%tmp_row =l mul %pi, %n
	%tmp_idx =l add %tmp_row, %pj
	%tmp_off =l mul %tmp_idx, %eight
	%tmp_ptr =l add %tmp, %tmp_off
	storel %sum, %tmp_ptr
	%pj_next =l add %pj, %one
	br @prod_j
@prod_j_done
	%pi_next =l add %pi, %one
	br @prod_i
@prod_i_done
	br @copy
@copy
	%ci =l phi @prod_i_done %zero, @copy_body %ci_next
	%cc =w csltl %ci, %n2
	jnz %cc, @copy_body, @copy_done
@copy_body
	%coff =l mul %ci, %eight
	%src =l add %tmp, %coff
	%dst =l add %out, %coff
	%v =l loadl %src
	storel %v, %dst
	%ci_next =l add %ci, %one
	br @copy
@copy_done
	%e_next =l add %e, %one
	br @exp_loop
@exp_done
	br @zero_tmp
@zero_tmp
	%zt =l phi @exp_done %zero, @zero_tmp_body %zt_next
	%ztc =w csltl %zt, %tmp_elems
	jnz %ztc, @zero_tmp_body, @zero_tmp_done
@zero_tmp_body
	%zoff2 =l mul %zt, %eight
	%zptr2 =l add %tmp, %zoff2
	storel %zero, %zptr2
	%zt_next =l add %zt, %one
	br @zero_tmp
@zero_tmp_done
	ret %zero
}
