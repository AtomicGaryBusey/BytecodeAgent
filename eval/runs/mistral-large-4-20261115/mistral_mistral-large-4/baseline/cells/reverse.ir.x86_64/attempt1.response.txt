export function l $reverse(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	storel 0, %ip
	%j0 =l sub %n, 1
	storel %j0, %jp
	jmp @loop
@loop
	%i =l loadl %ip
	%j =l loadl %jp
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%offi =l mul %i, 8
	%pi =l add %a, %offi
	%offj =l mul %j, 8
	%pj =l add %a, %offj
	%x =l loadl %pi
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%inext =l add %i, 1
	storel %inext, %ip
	%jnext =l sub %j, 1
	storel %jnext, %jp
	jmp @loop
@done
	ret 0
}
