export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%sum =l phi @start 0, @body %sum_next
	%p =l phi @start %a, @body %p_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%x =l loadl %p
	%fx =l call %f(l %x)
	%sum_next =l add %sum, %fx
	%p_next =l add %p, 8
	%i_next =l add %i, 1
	jmp @loop
@end
	ret %sum
}
