export function l $bubble_sort(l %a, l %n) {
@entry
	%zero =l sub %n, %n
	%nm1 =l sub %n, 1
	jmp @outer
@outer
	%i =l phi @entry %zero, @outer_end %i1
	%cont =w csltl %i, %nm1
	jnz %cont, @outer_body, @done
@outer_body
	%limit =l sub %n, %i
	%lim =l sub %limit, 1
	jmp @inner
@inner
	%j =l phi @outer_body %zero, @inner_end %j1
	%contj =w csltl %j, %lim
	jnz %contj, @compare, @inner_done
@compare
	%off =l mul %j, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @noswap
@swap
	storel %y, %p
	storel %x, %p2
	jmp @inner_end
@noswap
	jmp @inner_end
@inner_end
	%j1 =l add %j, 1
	jmp @inner
@inner_done
	jmp @outer_end
@outer_end
	%i1 =l add %i, 1
	jmp @outer
@done
	ret %zero
}
