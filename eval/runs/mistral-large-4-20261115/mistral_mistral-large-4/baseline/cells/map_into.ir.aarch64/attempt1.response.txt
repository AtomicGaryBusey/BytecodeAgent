export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l alloc8 8
	%cnt =l alloc8 8
	%sv =l alloc8 8
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	storel %zero, %i
	storel %zero, %cnt
	jmp @loop
@loop
	%iv =l loadl %i
	%cont =w csltl %iv, %n
	jnz %cont, @body, @end
@body
	%off =l mul %iv, %eight
	%psrc =l add %src, %off
	%sv0 =l loadl %psrc
	storel %sv0, %sv
	%res =l call %f(l %sv0)
	%iv2 =l loadl %i
	%off2 =l mul %iv2, %eight
	%pdst =l add %dst, %off2
	storel %res, %pdst
	%sv1 =l loadl %sv
	%less =w csltl %res, %sv1
	jnz %less, @inc, @next
@inc
	%cv =l loadl %cnt
	%c1 =l add %cv, %one
	storel %c1, %cnt
	jmp @next
@next
	%iv3 =l loadl %i
	%i1 =l add %iv3, %one
	storel %i1, %i
	jmp @loop
@end
	%ans =l loadl %cnt
	ret %ans
}
