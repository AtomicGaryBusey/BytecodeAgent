packbits_decode:
    xor r8, r8
.Lloop:
    test rsi, rsi
    jz .Ldone
    movsx r9, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp r9, -128
    je .Lloop
    test r9, r9
    js .Lrepeat
    lea r10, [r9 + 1]
    cmp rsi, r10
    jb .Lerr_src
    mov rax, rcx
    sub rax, r8
    cmp r10, rax
    jbe .Llit_fit
    mov r11, rax
    mov r9, 1
    jmp .Llit_copy
.Llit_fit:
    mov r11, r10
    xor r9, r9
.Llit_copy:
    test r11, r11
    jz .Llit_after
.Llit_copy_loop:
    movzx rax, byte ptr [rdi]
    mov [rdx], al
    inc rdi
    inc rdx
    dec r11
    jnz .Llit_copy_loop
.Llit_after:
    test r9, r9
    jnz .Lret_m1
    sub rsi, r10
    add r8, r10
    jmp .Lloop
.Lrepeat:
    mov r10, 1
    sub r10, r9
    cmp rsi, 1
    jb .Lerr_src
    movzx r9, byte ptr [rdi]
    mov rax, rcx
    sub rax, r8
    cmp r10, rax
    jbe .Lrep_fit
    mov r11, rax
    mov rax, 1
    jmp .Lrep_fill
.Lrep_fit:
    mov r11, r10
    xor rax, rax
.Lrep_fill:
    test r11, r11
    jz .Lrep_after
.Lrep_fill_loop:
    mov [rdx], r9b
    inc rdx
    dec r11
    jnz .Lrep_fill_loop
.Lrep_after:
    test rax, rax
    jnz .Lret_m1
    inc rdi
    dec rsi
    add r8, r10
    jmp .Lloop
.Ldone:
    mov rax, r8
    ret
.Lerr_src:
    mov rax, -2
    ret
.Lret_m1:
    mov rax, -1
    ret
