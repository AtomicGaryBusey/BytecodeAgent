export function l $sum_to(l %n) {
@start
	%one =l const 1
	%i0 =l const 1
	%acc0 =l const 0
	jmp @loop
@loop
	%iloop =l phi @start %i0, @body %inext
	%accloop =l phi @start %acc0, @body %accnext
	%cond =w cslel %iloop, %n
	jnz %cond, @body, @done
@body
	%accnext =l add %accloop, %iloop
	%inext =l add %iloop, %one
	jmp @loop
@done
	ret %accloop
}
