export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%loc =l alloc8 32
	%dpslot =l add %loc, 8
	%zpslot =l add %loc, 16
	storel 0, %loc
	storel 0, %dpslot
	jmp @loop

@loop
	%pc =l loadl %loc
	%d =l loadl %dpslot
	%end =w ceql %pc, %len
	jnz %end, @done, @fetch

@fetch
	%pc8 =l mul %pc, 8
	%caddr =l add %code, %pc8
	%op =l loadl %caddr
	%c0 =w ceql %op, 0
	jnz %c0, @done, @chk1
@chk1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @chk2
@chk2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @chk3
@chk3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @chk4
@chk4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @chk5
@chk5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @chk6
@chk6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @chk7
@chk7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @chk8
@chk8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @chk9
@chk9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @chk10
@chk10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @chk11
@chk11
	%c11 =w ceql %op, 11
	jnz %c11, @jmp, @default

@default
	%pc_def =l add %pc, 1
	storel %pc_def, %loc
	storel %d, %dpslot
	jmp @loop

@push
	%pc1_push =l add %pc, 1
	%off_push =l mul %pc1_push, 8
	%opaddr_push =l add %code, %off_push
	%val_push =l loadl %opaddr_push
	%d8_push =l mul %d, 8
	%saddr_push =l add %stack, %d8_push
	storel %val_push, %saddr_push
	%dnew_push =l add %d, 1
	%pcnew_push =l add %pc, 2
	storel %pcnew_push, %loc
	storel %dnew_push, %dpslot
	jmp @loop

@add
	%dm1_add =l sub %d, 1
	%dm2_add =l sub %d, 2
	%offb_add =l mul %dm1_add, 8
	%addrb_add =l add %stack, %offb_add
	%b_add =l loadl %addrb_add
	%offa_add =l mul %dm2_add, 8
	%addra_add =l add %stack, %offa_add
	%a_add =l loadl %addra_add
	%res_add =l add %a_add, %b_add
	storel %res_add, %addra_add
	%dnew_add =l sub %d, 1
	%pcnew_add =l add %pc, 1
	storel %pcnew_add, %loc
	storel %dnew_add, %dpslot
	jmp @loop

@sub
	%dm1_sub =l sub %d, 1
	%dm2_sub =l sub %d, 2
	%offb_sub =l mul %dm1_sub, 8
	%addrb_sub =l add %stack, %offb_sub
	%b_sub =l loadl %addrb_sub
	%offa_sub =l mul %dm2_sub, 8
	%addra_sub =l add %stack, %offa_sub
	%a_sub =l loadl %addra_sub
	%res_sub =l sub %a_sub, %b_sub
	storel %res_sub, %addra_sub
	%dnew_sub =l sub %d, 1
	%pcnew_sub =l add %pc, 1
	storel %pcnew_sub, %loc
	storel %dnew_sub, %dpslot
	jmp @loop

@mul
	%dm1_mul =l sub %d, 1
	%dm2_mul =l sub %d, 2
	%offb_mul =l mul %dm1_mul, 8
	%addrb_mul =l add %stack, %offb_mul
	%b_mul =l loadl %addrb_mul
	%offa_mul =l mul %dm2_mul, 8
	%addra_mul =l add %stack, %offa_mul
	%a_mul =l loadl %addra_mul
	%res_mul =l mul %a_mul, %b_mul
	storel %res_mul, %addra_mul
	%dnew_mul =l sub %d, 1
	%pcnew_mul =l add %pc, 1
	storel %pcnew_mul, %loc
	storel %dnew_mul, %dpslot
	jmp @loop

@dup
	%dm1_dup =l sub %d, 1
	%off_dup =l mul %dm1_dup, 8
	%addr_dup =l add %stack, %off_dup
	%top_dup =l loadl %addr_dup
	%d8_dup =l mul %d, 8
	%saddr_dup =l add %stack, %d8_dup
	storel %top_dup, %saddr_dup
	%dnew_dup =l add %d, 1
	%pcnew_dup =l add %pc, 1
	storel %pcnew_dup, %loc
	storel %dnew_dup, %dpslot
	jmp @loop

@swap
	%dm1_sw =l sub %d, 1
	%dm2_sw =l sub %d, 2
	%offb_sw =l mul %dm1_sw, 8
	%addrb_sw =l add %stack, %offb_sw
	%b_sw =l loadl %addrb_sw
	%offa_sw =l mul %dm2_sw, 8
	%addra_sw =l add %stack, %offa_sw
	%a_sw =l loadl %addra_sw
	storel %b_sw, %addra_sw
	storel %a_sw, %addrb_sw
	%pcnew_sw =l add %pc, 1
	storel %pcnew_sw, %loc
	storel %d, %dpslot
	jmp @loop

@drop
	%dnew_drop =l sub %d, 1
	%pcnew_drop =l add %pc, 1
	storel %pcnew_drop, %loc
	storel %dnew_drop, %dpslot
	jmp @loop

@over
	%dm2_ov =l sub %d, 2
	%off_ov =l mul %dm2_ov, 8
	%addr_ov =l add %stack, %off_ov
	%a_ov =l loadl %addr_ov
	%d8_ov =l mul %d, 8
	%saddr_ov =l add %stack, %d8_ov
	storel %a_ov, %saddr_ov
	%dnew_ov =l add %d, 1
	%pcnew_ov =l add %pc, 1
	storel %pcnew_ov, %loc
	storel %dnew_ov, %dpslot
	jmp @loop

@lt
	%dm1_lt =l sub %d, 1
	%dm2_lt =l sub %d, 2
	%offb_lt =l mul %dm1_lt, 8
	%addrb_lt =l add %stack, %offb_lt
	%b_lt =l loadl %addrb_lt
	%offa_lt =l mul %dm2_lt, 8
	%addra_lt =l add %stack, %offa_lt
	%a_lt =l loadl %addra_lt
	%cmp_lt =w csltl %a_lt, %b_lt
	%res_lt =l extsw %cmp_lt
	storel %res_lt, %addra_lt
	%dnew_lt =l sub %d, 1
	%pcnew_lt =l add %pc, 1
	storel %pcnew_lt, %loc
	storel %dnew_lt, %dpslot
	jmp @loop

@jz
	%pc1_jz =l add %pc, 1
	%off_jz =l mul %pc1_jz, 8
	%opaddr_jz =l add %code, %off_jz
	%t_jz =l loadl %opaddr_jz
	%dm1_jz =l sub %d, 1
	%offv_jz =l mul %dm1_jz, 8
	%vaddr_jz =l add %stack, %offv_jz
	%v_jz =l loadl %vaddr_jz
	%iszero_jz =w ceql %v_jz, 0
	jnz %iszero_jz, @jz_zero, @jz_nz
@jz_zero
	%pcnew_jz0 =l add %t_jz, 0
	%dnew_jz0 =l sub %d, 1
	storel %pcnew_jz0, %loc
	storel %dnew_jz0, %dpslot
	jmp @loop
@jz_nz
	%pcnew_jznz =l add %pc, 2
	%dnew_jznz =l sub %d, 1
	storel %pcnew_jznz, %loc
	storel %dnew_jznz, %dpslot
	jmp @loop

@jmp
	%pc1_jmp =l add %pc, 1
	%off_jmp =l mul %pc1_jmp, 8
	%opaddr_jmp =l add %code, %off_jmp
	%t_jmp =l loadl %opaddr_jmp
	storel %t_jmp, %loc
	storel %d, %dpslot
	jmp @loop

@done
	storel %d, %zpslot
	jmp @zero

@zero
	%z =l loadl %zpslot
	%zc =w csltl %z, %cap
	jnz %zc, @zero_body, @after_zero
@zero_body
	%zoff =l mul %z, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%znext =l add %z, 1
	storel %znext, %zpslot
	jmp @zero

@after_zero
	%has =w csltl 0, %d
	jnz %has, @ret_top, @ret_zero
@ret_top
	%dm1_ret =l sub %d, 1
	%off_ret =l mul %dm1_ret, 8
	%addr_ret =l add %stack, %off_ret
	%res =l loadl %addr_ret
	ret %res
@ret_zero
	ret 0
}
