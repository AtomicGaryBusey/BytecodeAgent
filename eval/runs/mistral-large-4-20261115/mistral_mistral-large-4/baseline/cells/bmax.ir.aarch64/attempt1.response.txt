export function l $bmax(l %a, l %n) {
@start
    %m =l loadl %a
    %i =l copy 1
    j @loop
@loop
    %idx =l phi @start %i, @next %i2
    %max =l phi @start %m, @next %m2
    %cont =w csltl %idx, %n
    jnz %cont, @body, @done
@body
    %off =l mul %idx, 8
    %addr =l add %a, %off
    %v =l loadl %addr
    %gt =w csgtl %v, %max
    jnz %gt, @update, @skip
@update
    %m_upd =l copy %v
    j @next
@skip
    %m_skip =l copy %max
    j @next
@next
    %m2 =l phi @update %m_upd, @skip %m_skip
    %i2 =l add %idx, 1
    j @loop
@done
    ret %max
}
