export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%iszero =w ceql %v, 0
	jnz %iszero, @done, @next
@next
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %i
}
