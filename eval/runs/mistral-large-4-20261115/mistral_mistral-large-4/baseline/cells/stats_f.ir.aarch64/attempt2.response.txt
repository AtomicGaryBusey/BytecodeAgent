export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %p =l copy %a
  %x =l loadl %p
  %y =l call %f(l %x)
  %min =l copy %y
  %max =l copy %y
  %sum =l copy %y
  %cnt =l and %y, 1
  %rem =l sub %n, 1
  %p =l add %p, 8
  %cond =w csgtl %rem, 0
  jnz %cond, @loop, @done
@loop
  %x =l loadl %p
  %y =l call %f(l %x)
  %sum =l add %sum, %y
  %odd =l and %y, 1
  %cnt =l add %cnt, %odd
  %lt =w csltl %y, %min
  %min =l select %lt, %y, %min
  %gt =w csgtl %y, %max
  %max =l select %gt, %y, %max
  %p =l add %p, 8
  %rem =l sub %rem, 1
  %cond =w csgtl %rem, 0
  jnz %cond, @loop, @done
@done
  storel %min, %out
  %out1 =l add %out, 8
  storel %max, %out1
  %out2 =l add %out, 16
  storel %sum, %out2
  ret %cnt
}
