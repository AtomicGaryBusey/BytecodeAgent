bubble_sort:
  cbz x1, .Ldone
  sub x10, x1, #1
.Louter:
  cbz x10, .Ldone
  mov x11, xzr
.Linner:
  cmp x11, x10
  b.ge .Lnext
  ldr x12, [x0, x11, lsl #3]
  add x13, x11, #1
  ldr x14, [x0, x13, lsl #3]
  cmp x12, x14
  b.le .Lnoswap
  str x14, [x0, x11, lsl #3]
  str x12, [x0, x13, lsl #3]
.Lnoswap:
  add x11, x11, #1
  b .Linner
.Lnext:
  sub x10, x10, #1
  b .Louter
.Ldone:
  mov x0, xzr
  ret
