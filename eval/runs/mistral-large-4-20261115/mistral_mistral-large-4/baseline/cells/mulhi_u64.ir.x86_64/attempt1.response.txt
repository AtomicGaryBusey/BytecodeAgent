export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =l and %a, 4294967295
    %a1 =l shr %a, 32
    %b0 =l and %b, 4294967295
    %b1 =l shr %b, 32
    %p0 =l mul %a0, %b0
    %carry0 =l shr %p0, 32
    %m0 =l mul %a1, %b0
    %m1 =l mul %a0, %b1
    %t =l add %m0, %m1
    %c1w =w cultl %t, %m0
    %c1 =l extuw %c1w
    %s =l add %t, %carry0
    %c2w =w cultl %s, %t
    %c2 =l extuw %c2w
    %s_hi =l shr %s, 32
    %csum =l add %c1, %c2
    %csum_shift =l shl %csum, 32
    %carry_mid =l add %csum_shift, %s_hi
    %a1b1 =l mul %a1, %b1
    %hi =l add %a1b1, %carry_mid
    ret %hi
}
