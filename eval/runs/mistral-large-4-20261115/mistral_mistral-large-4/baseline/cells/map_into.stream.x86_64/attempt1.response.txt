map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rcx
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    xor r15, r15
.Lloop:
    test r14, r14
    jz .Ldone
    mov rdi, qword ptr [r13]
    call rbx
    mov qword ptr [r12], rax
    cmp rax, qword ptr [r13]
    setl al
    movzx eax, al
    add r15, rax
    add r12, 8
    add r13, 8
    dec r14
    jmp .Lloop
.Ldone:
    mov rax, r15
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
