lcs_len:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, rsi
  mov rbp, rcx
  mov r12, rdi
  mov r13, rdx
  mov r14, r8
  mov rdi, r14
  lea rcx, [rbp + 1]
  xor eax, eax
  cld
  rep stosq
  xor r15d, r15d
.Louter:
  cmp r15, rbx
  jge .Ldone
  movzx eax, byte ptr [r12 + r15]
  xor edx, edx
  mov rcx, 1
.Linner:
  cmp rcx, rbp
  jg .Lend_inner
  mov rsi, qword ptr [r14 + rcx*8]
  mov rdi, rsi
  lea r10, [rcx - 1]
  movzx r8d, byte ptr [r13 + r10]
  cmp al, r8b
  jne .Lno_match
  lea r9, [rdx + 1]
  jmp .Lstore
.Lno_match:
  mov r9, qword ptr [r14 + r10*8]
  cmp rsi, r9
  cmovl rsi, r9
  mov r9, rsi
.Lstore:
  mov qword ptr [r14 + rcx*8], r9
  mov rdx, rdi
  inc rcx
  jmp .Linner
.Lend_inner:
  inc r15
  jmp .Louter
.Ldone:
  mov rax, qword ptr [r14 + rbp*8]
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
