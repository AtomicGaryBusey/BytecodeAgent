export function l $fill_down(l %a, l %n, l %start) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %nexti
	%p =l phi @start %a, @body %nextp
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%val =l sub %start, %i
	storel %val, %p
	%nexti =l add %i, 1
	%nextp =l add %p, 8
	jmp @loop
@done
	ret %n
}
