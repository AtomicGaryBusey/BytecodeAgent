export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%c128 =l copy 128
	%neg1 =l sub %zero, %one
	%neg2 =l sub %zero, %two
	%neg128 =l sub %zero, %c128
	%pos =l copy 0
	%written =l copy 0
	br @packet

@packet
	%rem0 =l sub %n, %pos
	%nohdr =w ceql %rem0, %zero
	jnz %nohdr, @done, @hdr

@hdr
	%hp =l add %src, %pos
	%hb =w loadsb %hp
	%h =l extsw %hb
	%pos =l add %pos, %one
	%nop =w ceql %h, %neg128
	jnz %nop, @packet, @notnop

@notnop
	%islit =w csgel %h, %zero
	jnz %islit, @literal, @repeat

@literal
	%lcount =l add %h, %one
	%lrem =l sub %n, %pos
	%lok =w csgel %lrem, %lcount
	jnz %lok, @lit_src_ok, @src_err

@lit_src_ok
	%lroom =l sub %cap, %written
	%lover =w csltl %lroom, %lcount
	jnz %lover, @lit_over, @lit_copy_start

@lit_copy_start
	%li =l copy 0
	br @lit_copy

@lit_copy
	%ldone =w ceql %li, %lcount
	jnz %ldone, @lit_finish, @lit_body

@lit_body
	%ls =l add %src, %pos
	%lsi =l add %ls, %li
	%lb =w loadub %lsi
	%ld =l add %out, %written
	%ldi =l add %ld, %li
	storeb %lb, %ldi
	%li =l add %li, %one
	br @lit_copy

@lit_finish
	%written =l add %written, %lcount
	%pos =l add %pos, %lcount
	br @packet

@lit_over
	%lroom2 =l sub %cap, %written
	%li2 =l copy 0
	br @lit_over_copy

@lit_over_copy
	%lol =w ceql %li2, %lroom2
	jnz %lol, @lit_over_done, @lit_over_body

@lit_over_body
	%ls2 =l add %src, %pos
	%lsi2 =l add %ls2, %li2
	%lb2 =w loadub %lsi2
	%ld2 =l add %out, %written
	%ldi2 =l add %ld2, %li2
	storeb %lb2, %ldi2
	%li2 =l add %li2, %one
	br @lit_over_copy

@lit_over_done
	ret %neg1

@src_err
	ret %neg2

@repeat
	%rrem =l sub %n, %pos
	%rok =w csgel %rrem, %one
	jnz %rok, @rep_src_ok, @src_err

@rep_src_ok
	%rp =l add %src, %pos
	%rb =w loadub %rp
	%pos =l add %pos, %one
	%rcount =l sub %one, %h
	%rroom =l sub %cap, %written
	%rover =w csltl %rroom, %rcount
	jnz %rover, @rep_over, @rep_copy_start

@rep_copy_start
	%ri =l copy 0
	br @rep_copy

@rep_copy
	%rdone =w ceql %ri, %rcount
	jnz %rdone, @rep_finish, @rep_body

@rep_body
	%rd =l add %out, %written
	%rdi =l add %rd, %ri
	storeb %rb, %rdi
	%ri =l add %ri, %one
	br @rep_copy

@rep_finish
	%written =l add %written, %rcount
	br @packet

@rep_over
	%rroom2 =l sub %cap, %written
	%ri2 =l copy 0
	br @rep_over_copy

@rep_over_copy
	%rol =w ceql %ri2, %rroom2
	jnz %rol, @rep_over_done, @rep_over_body

@rep_over_body
	%rd2 =l add %out, %written
	%rdi2 =l add %rd2, %ri2
	storeb %rb, %rdi2
	%ri2 =l add %ri2, %one
	br @rep_over_copy

@rep_over_done
	ret %neg1

@done
	ret %written
}
