collatz_max:
    mov x1, x0
    mov x3, #0
    mov x2, #1
.Louter:
    cmp x2, x1
    b.gt .Ldone
    mov x4, x2
    mov x5, #0
.Linner:
    cmp x4, #1
    b.eq .Linner_done
    tst x4, #1
    b.ne .Lodd
    lsr x4, x4, #1
    b .Linc
.Lodd:
    add x4, x4, x4, lsl #1
    add x4, x4, #1
.Linc:
    add x5, x5, #1
    b .Linner
.Linner_done:
    cmp x5, x3
    b.le .Lskip
    mov x3, x5
.Lskip:
    add x2, x2, #1
    b .Louter
.Ldone:
    mov x0, x3
    ret
