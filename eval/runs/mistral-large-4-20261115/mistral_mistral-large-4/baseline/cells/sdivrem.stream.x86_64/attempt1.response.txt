sdivrem:
    mov rcx, rdx
    test rsi, rsi
    je .Lzero
    cmp rsi, -1
    jne .Lnormal
    mov r8, 0x8000000000000000
    cmp rdi, r8
    je .Loverflow
.Lnormal:
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Lzero:
    xor eax, eax
    mov qword ptr [rcx], rdi
    ret
.Loverflow:
    mov rax, rdi
    mov qword ptr [rcx], 0
    ret
