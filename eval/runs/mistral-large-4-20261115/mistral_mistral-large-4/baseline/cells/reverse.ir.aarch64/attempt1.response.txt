export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l copy %n
	%j =l sub %j, 1
@loop
	%lt =w csltl %i, %j
	jnz %lt, @body, @done
@body
	%offi =l mul %i, 8
	%pi =l add %a, %offi
	%x =l loadl %pi
	%offj =l mul %j, 8
	%pj =l add %a, %offj
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	%res =l copy 0
	ret %res
}
