mp_mul:
    push rbx
    push r12
    push r13
    push r14
    mov rbx, rsi
    mov r12, rcx
    mov r13, rdx
    mov r14, r8
    lea r11, [rbx + r12]
    mov r9, r14
    xor eax, eax
.Lzero:
    test r11, r11
    jz .Lzero_done
    mov qword ptr [r9], rax
    add r9, 8
    dec r11
    jmp .Lzero
.Lzero_done:
    xor esi, esi
.Louter:
    cmp rsi, rbx
    jae .Ldone
    lea r11, [r14 + rsi*8]
    xor r10d, r10d
    xor ecx, ecx
.Linner:
    cmp rcx, r12
    jae .Linner_done
    mov rdx, qword ptr [r13 + rcx*8]
    mov rax, qword ptr [rdi + rsi*8]
    mul rdx
    add rax, qword ptr [r11]
    adc rdx, 0
    add rax, r10
    adc rdx, 0
    mov qword ptr [r11], rax
    mov r10, rdx
    add r11, 8
    inc rcx
    jmp .Linner
.Linner_done:
    mov qword ptr [r11], r10
    inc rsi
    jmp .Louter
.Ldone:
    lea rcx, [rbx + r12]
.Lscan:
    test rcx, rcx
    jz .Lzero_ret
    dec rcx
    cmp qword ptr [r14 + rcx*8], 0
    jne .Lfound
    jmp .Lscan
.Lfound:
    lea rax, [rcx + 1]
    jmp .Lret
.Lzero_ret:
    xor eax, eax
.Lret:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
