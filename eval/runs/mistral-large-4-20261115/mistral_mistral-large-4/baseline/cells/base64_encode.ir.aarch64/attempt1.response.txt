export function l $base64_encode(l %src, l %n, l %out) {
@start
  %np2 =l add %n, 2
  %groups =l div %np2, 3
  %outlen =l mul %groups, 4
  jmp @loop
@loop
  %i =l phi @start 0, @group_done %i_next
  %op =l phi @start %out, @group_done %op_next
  %more =w csltl %i, %groups
  jnz %more, @body, @done
@body
  %off =l mul %i, 3
  %rem =l sub %n, %off
  %islt3 =w csltl %rem, 3
  jnz %islt3, @check2, @full
@check2
  %is2 =w ceql %rem, 2
  jnz %is2, @rem2, @rem1
@full
  %p0_f =l add %src, %off
  %p1_f =l add %p0_f, 1
  %p2_f =l add %p0_f, 2
  %b0_f =l loadub %p0_f
  %b1_f =l loadub %p1_f
  %b2_f =l loadub %p2_f
  %t0_f =l shl %b0_f, 16
  %t1_f =l shl %b1_f, 8
  %v01_f =l or %t0_f, %t1_f
  %val_f =l or %v01_f, %b2_f
  %nc_f =l add 0, 4
  jmp @group_ready
@rem2
  %p0_r2 =l add %src, %off
  %p1_r2 =l add %p0_r2, 1
  %b0_r2 =l loadub %p0_r2
  %b1_r2 =l loadub %p1_r2
  %t0_r2 =l shl %b0_r2, 16
  %t1_r2 =l shl %b1_r2, 8
  %val_r2 =l or %t0_r2, %t1_r2
  %nc_r2 =l add 0, 3
  jmp @group_ready
@rem1
  %p0_r1 =l add %src, %off
  %b0_r1 =l loadub %p0_r1
  %val_r1 =l shl %b0_r1, 16
  %nc_r1 =l add 0, 2
  jmp @group_ready
@group_ready
  %val =l phi @full %val_f, @rem2 %val_r2, @rem1 %val_r1
  %nchars =l phi @full %nc_f, @rem2 %nc_r2, @rem1 %nc_r1
  %op0 =l add %op, 0
  jmp @inner
@inner
  %j =l phi @group_ready 0, @inner_next %j_next
  %q =l phi @group_ready %op0, @inner_next %q_next
  %jcond =w csltl %j, 4
  jnz %jcond, @inner_body, @group_done
@inner_body
  %shamt =l mul %j, 6
  %shift =l sub 18, %shamt
  %idx0 =l shr %val, %shift
  %idx =l and %idx0, 63
  %isalpha =w csltl %j, %nchars
  jnz %isalpha, @emit_alpha, @emit_pad
@emit_alpha
  %c26 =w csltl %idx, 26
  jnz %c26, @m_a, @m_not_a
@m_a
  %ch_a =l add %idx, 65
  jmp @m_done
@m_not_a
  %c52 =w csltl %idx, 52
  jnz %c52, @m_az, @m_not_az
@m_az
  %ch_az =l add %idx, 71
  jmp @m_done
@m_not_az
  %c62 =w csltl %idx, 62
  jnz %c62, @m_09, @m_special
@m_09
  %ch_09 =l sub %idx, 4
  jmp @m_done
@m_special
  %isplus =w ceql %idx, 62
  jnz %isplus, @m_plus, @m_slash
@m_plus
  %ch_plus =l add 0, 43
  jmp @m_done
@m_slash
  %ch_slash =l add 0, 47
  jmp @m_done
@m_done
  %ch =l phi @m_a %ch_a, @m_az %ch_az, @m_09 %ch_09, @m_plus %ch_plus, @m_slash %ch_slash
  storeb %ch, %q
  jmp @inner_next
@emit_pad
  %pad =l add 0, 61
  storeb %pad, %q
  jmp @inner_next
@inner_next
  %j_next =l add %j, 1
  %q_next =l add %q, 1
  jmp @inner
@group_done
  %i_next =l add %i, 1
  %op_next =l add %op, 4
  jmp @loop
@done
  ret %outlen
}
