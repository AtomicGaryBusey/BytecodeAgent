export function l $max_byte(l %p, l %n) {
@start
	%zero =l copy 0
	%c =w csltl %zero, %n
	jnz %c, @loop, @done
@loop
	%i =l phi @start %zero, @body %inext
	%max =l phi @start %zero, @body %maxnext
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%gt =w csgtl %b, %max
	jnz %gt, @set, @body
@set
	%maxnew =l copy %b
	jmp @body
@body
	%maxnext =l phi @loop %max, @set %maxnew
	%inext =l add %i, 1
	%cond =w csltl %inext, %n
	jnz %cond, @loop, @done
@done
	%res =l phi @start %zero, @body %maxnext
	ret %res
}
