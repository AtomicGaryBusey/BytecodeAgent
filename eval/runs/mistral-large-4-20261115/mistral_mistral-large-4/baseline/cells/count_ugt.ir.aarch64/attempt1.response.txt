export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%gt =w cugtl %val, %t
	%gtl =l extsw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
