export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %i =l copy 0
  %written =l copy 0
  %c0 =w csltl %i, %n
  jnz %c0, @loop, @done

@loop
  %run =l copy 1
  jmp @run_loop

@run_loop
  %r_lt =w csltl %run, 128
  %idx =l add %i, %run
  %idx_lt =w csltl %idx, %n
  %both =w and %r_lt, %idx_lt
  jnz %both, @run_check, @run_done

@run_check
  %p1 =l add %src, %idx
  %b1 =w loadub %p1
  %p0 =l add %src, %i
  %b0 =w loadub %p0
  %eq =w ceqw %b1, %b0
  jnz %eq, @run_inc, @run_done

@run_inc
  %run =l add %run, 1
  jmp @run_loop

@run_done
  %lt3 =w csltl %run, 3
  jnz %lt3, @literal, @repeat

@repeat
  %need =l add %written, 2
  %bad =w csltl %cap, %need
  jnz %bad, @fail, @rep_write

@rep_write
  %outp =l add %out, %written
  %runw =w copy %run
  %hdr =w sub 257, %runw
  storeb %hdr, %outp
  %srcp =l add %src, %i
  %b =w loadub %srcp
  %outp2 =l add %outp, 1
  storeb %b, %outp2
  %written =l add %written, 2
  %i =l add %i, %run
  %c1 =w csltl %i, %n
  jnz %c1, @loop, @done

@literal
  %j =l copy %i
  jmp @lit_loop

@lit_loop
  %j_lt =w csltl %j, %n
  %len =l sub %j, %i
  %len_lt =w csltl %len, 128
  %c2 =w and %j_lt, %len_lt
  jnz %c2, @lit_run, @lit_end

@lit_run
  %run =l copy 1
  jmp @run_loop2

@run_loop2
  %r_lt2 =w csltl %run, 128
  %idx2 =l add %j, %run
  %idx_lt2 =w csltl %idx2, %n
  %both2 =w and %r_lt2, %idx_lt2
  jnz %both2, @run_check2, @run_done2

@run_check2
  %p12 =l add %src, %idx2
  %b12 =w loadub %p12
  %p02 =l add %src, %j
  %b02 =w loadub %p02
  %eq2 =w ceqw %b12, %b02
  jnz %eq2, @run_inc2, @run_done2

@run_inc2
  %run =l add %run, 1
  jmp @run_loop2

@run_done2
  %lt3b =w csltl %run, 3
  jnz %lt3b, @lit_inc, @lit_end

@lit_inc
  %j =l add %j, 1
  jmp @lit_loop

@lit_end
  %L =l sub %j, %i
  %need =l add %L, 1
  %total =l add %written, %need
  %bad =w csltl %cap, %total
  jnz %bad, @fail, @lit_write

@lit_write
  %outp =l add %out, %written
  %Lw =w copy %L
  %hdr =w sub %Lw, 1
  storeb %hdr, %outp
  %k =l copy 0
  jmp @copy_loop

@copy_loop
  %k_lt =w csltl %k, %L
  jnz %k_lt, @copy_body, @copy_done

@copy_body
  %sp =l add %src, %i
  %spk =l add %sp, %k
  %b =w loadub %spk
  %op =l add %out, %written
  %opk =l add %op, %k
  %opk1 =l add %opk, 1
  storeb %b, %opk1
  %k =l add %k, 1
  jmp @copy_loop

@copy_done
  %written =l add %written, %need
  %i =l copy %j
  %c3 =w csltl %i, %n
  jnz %c3, @loop, @done

@fail
  %minus1 =l copy -1
  ret %minus1

@done
  ret %written
}
