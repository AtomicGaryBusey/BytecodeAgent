collatz_max:
    xor r9d, r9d
    mov r8, 1
.Louter:
    cmp r8, rdi
    jg .Ldone
    mov rax, r8
    xor ecx, ecx
.Lstep:
    cmp rax, 1
    je .Lafter
    test rax, 1
    jz .Leven
    lea rax, [rax + rax*2 + 1]
    jmp .Linc
.Leven:
    shr rax, 1
.Linc:
    inc rcx
    jmp .Lstep
.Lafter:
    cmp rcx, r9
    jle .Lnext
    mov r9, rcx
.Lnext:
    inc r8
    jmp .Louter
.Ldone:
    mov rax, r9
    ret
