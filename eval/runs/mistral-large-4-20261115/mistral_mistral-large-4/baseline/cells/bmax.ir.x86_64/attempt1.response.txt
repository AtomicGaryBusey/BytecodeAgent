export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%p0 =l add %a, 8
	%i0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %i0, @update %i_u, @keep %i_k
	%p =l phi @start %p0, @update %p_u, @keep %p_k
	%m =l phi @start %max0, @update %m_u, @keep %m_k
	%cont =w ceql %i, 0
	jnz %cont, @done, @body
@body
	%v =l loadl %p
	%lt =w csltl %m, %v
	jnz %lt, @update, @keep
@update
	%m_u =l copy %v
	%p_u =l add %p, 8
	%i_u =l sub %i, 1
	jmp @loop
@keep
	%m_k =l copy %m
	%p_k =l add %p, 8
	%i_k =l sub %i, 1
	jmp @loop
@done
	ret %m
}
