export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%eq =w ceql %dst, %src
	jnz %eq, @done, @cmp
@cmp
	%lt =w csltl %dst, %src
	jnz %lt, @forward_init, @backward_init
@forward_init
	%zero_f =l copy 0
	%limit =l mul %n, 8
	jmp @fcond
@fcond
	%off =l phi @forward_init %zero_f, @fbody %off_next
	%cond =w csltl %off, %limit
	jnz %cond, @fbody, @done
@fbody
	%saddr =l add %src, %off
	%val =l loadl %saddr
	%daddr =l add %dst, %off
	storel %val, %daddr
	%off_next =l add %off, 8
	jmp @fcond
@backward_init
	%limitb =l mul %n, 8
	%off_start =l sub %limitb, 8
	jmp @bcond
@bcond
	%offb =l phi @backward_init %off_start, @bbody %offb_next
	%condb =w csgel %offb, %zero
	jnz %condb, @bbody, @done
@bbody
	%saddrb =l add %src, %offb
	%valb =l loadl %saddrb
	%daddrb =l add %dst, %offb
	storel %valb, %daddrb
	%offb_next =l sub %offb, 8
	jmp @bcond
@done
	ret %zero
}
