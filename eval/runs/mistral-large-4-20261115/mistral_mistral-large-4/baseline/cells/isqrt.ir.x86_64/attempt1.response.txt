export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967296
	%one =l copy 1
	%two =l copy 2
@loop
	%diff =l subl %hi, %lo
	%cont =w cugtl %diff, %one
	jnz %cont, @body, @done
@body
	%sum =l addl %lo, %hi
	%mid =l divl %sum, %two
	%sq =l mull %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	%lo =l copy %mid
	jmp @loop
@sethi
	%hi =l copy %mid
	jmp @loop
@done
	ret %lo
}
