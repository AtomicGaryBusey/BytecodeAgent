memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.lo .Lforward
.Lbackward:
    sub x3, x2, #1
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    subs x2, x2, #1
    b.ne .Lbackward
    b .Ldone
.Lforward:
    mov x3, #0
.Lforward_loop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x3, x3, #1
    cmp x3, x2
    b.ne .Lforward_loop
.Ldone:
    mov x0, #0
    ret
