export function l $qsort_pairs(l %a, l %n) {
@start
	%zero =l copy 0
	%small =w csltl %n, 2
	jnz %small, @ret, @partition
@ret
	ret %zero
@partition
	%mid =l shr %n, 1
	%last_idx =l sub %n, 1
	%off_mid =l mul %mid, 16
	%p_mid =l add %a, %off_mid
	%off_last =l mul %last_idx, 16
	%p_last =l add %a, %off_last
	%km =l loadl %p_mid
	%vptr_mid =l add %p_mid, 8
	%vm =l loadl %vptr_mid
	%kl =l loadl %p_last
	%vptr_last =l add %p_last, 8
	%vl =l loadl %vptr_last
	storel %kl, %p_mid
	storel %vl, %vptr_mid
	storel %km, %p_last
	storel %vm, %vptr_last
	%i =l copy 0
	%j =l copy 0
	%limit =l sub %n, 1
	jmp @loop
@loop
	%cont =w csltl %j, %limit
	jnz %cont, @body, @after_loop
@body
	%off_j =l mul %j, 16
	%p_j =l add %a, %off_j
	%kj =l loadl %p_j
	%vptr_j =l add %p_j, 8
	%vj =l loadl %vptr_j
	%less =w csltl %kj, %km
	%eq =w ceql %kj, %km
	%vle =w cslel %vj, %vm
	%eqv =w and %eq, %vle
	%do_swap =w or %less, %eqv
	jnz %do_swap, @swap_ij, @after_if
@swap_ij
	%off_i =l mul %i, 16
	%p_i =l add %a, %off_i
	%ki =l loadl %p_i
	%vptr_i =l add %p_i, 8
	%vi =l loadl %vptr_i
	storel %kj, %p_i
	storel %vj, %vptr_i
	storel %ki, %p_j
	storel %vi, %vptr_j
	%i =l add %i, 1
	jmp @inc_j
@after_if
	jmp @inc_j
@inc_j
	%j =l add %j, 1
	jmp @loop
@after_loop
	%off_i2 =l mul %i, 16
	%p_i2 =l add %a, %off_i2
	%ki2 =l loadl %p_i2
	%vptr_i2 =l add %p_i2, 8
	%vi2 =l loadl %vptr_i2
	%kl2 =l loadl %p_last
	%vptr_last2 =l add %p_last, 8
	%vl2 =l loadl %vptr_last2
	storel %kl2, %p_i2
	storel %vl2, %vptr_i2
	storel %ki2, %p_last
	storel %vi2, %vptr_last2
	%left_n =l copy %i
	%right_start =l add %i, 1
	%right_n =l sub %n, %right_start
	%c_left =w csgtl %left_n, 0
	jnz %c_left, @call_left, @check_right
@call_left
	%rl =l call $qsort_pairs(l %a, l %left_n)
	jmp @check_right
@check_right
	%c_right =w csgtl %right_n, 0
	jnz %c_right, @call_right, @ret0
@call_right
	%off_right =l mul %right_start, 16
	%a_right =l add %a, %off_right
	%rr =l call $qsort_pairs(l %a_right, l %right_n)
	jmp @ret0
@ret0
	ret %zero
}
