export function l $collatz_max(l %n) {
@start
    %max =l copy 0
    %i =l copy 1
    jmp @outer_cond
@outer_cond
    %gt =w csgtl %i, %n
    jnz %gt, @done, @outer_body
@outer_body
    %x =l copy %i
    %steps =l copy 0
    jmp @inner_cond
@inner_cond
    %is_one =w ceql %x, 1
    jnz %is_one, @inner_done, @inner_body
@inner_body
    %m =l rem %x, 2
    %is_even =w ceql %m, 0
    jnz %is_even, @even, @odd
@even
    %x =l div %x, 2
    %steps =l add %steps, 1
    jmp @inner_cond
@odd
    %t =l mul %x, 3
    %x =l add %t, 1
    %steps =l add %steps, 1
    jmp @inner_cond
@inner_done
    %gtm =w csgtl %steps, %max
    jnz %gtm, @update, @outer_next
@update
    %max =l copy %steps
    jmp @outer_next
@outer_next
    %i =l add %i, 1
    jmp @outer_cond
@done
    ret %max
}
