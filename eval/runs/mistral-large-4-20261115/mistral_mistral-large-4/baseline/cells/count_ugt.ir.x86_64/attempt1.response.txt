export function l $count_ugt(l %a, l %n, l %t) {
@start
    %zero =l copy 0
    %nz =w cnel %n, %zero
    jnz %nz, @loop, @done
@loop
    %r =l phi @start %n, @next %r1
    %p =l phi @start %a, @next %p1
    %c =l phi @start %zero, @next %c1
    %x =l loadl %p
    %gt =w cugtl %x, %t
    jnz %gt, @inc, @next
@inc
    %cplus =l add %c, 1
    jmp @next
@next
    %c1 =l phi @loop %c, @inc %cplus
    %p1 =l add %p, 8
    %r1 =l sub %r, 1
    %nz2 =w cnel %r1, %zero
    jnz %nz2, @loop, @done
@done
    %res =l phi @start %zero, @next %c1
    ret %res
}
