export function l $count_desc(l %a, l %n) {
@start
	%two =l copy 2
	%small =w csltl %n, %two
	jnz %small, @zero, @init
@zero
	%ret0 =l copy 0
	ret %ret0
@init
	%limit =l sub %n, 1
	br @loop
@loop
	%i =l phi @init 0, @body %inext
	%cnt =l phi @init 0, @body %cntnext
	%cond =w csltl %i, %limit
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%inext =l add %i, 1
	%off1 =l mul %inext, 8
	%p1 =l add %a, %off1
	%y =l loadl %p1
	%desc =w csltl %y, %x
	%inc =l extsw %desc
	%cntnext =l add %cnt, %inc
	br @loop
@done
	ret %cnt
}
