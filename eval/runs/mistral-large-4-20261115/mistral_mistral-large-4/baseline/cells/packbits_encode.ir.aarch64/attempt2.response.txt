export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%r =w copy 1
	%r_l =l extsw %r
	%idx =l add %i, %r_l
	%src_i_ptr =l add %src, %i
	%src_i =w loadub %src_i_ptr
	jmp @run_loop
@run_loop
	%cont1 =w csltw %r, 128
	%idx_lt_n =w csltl %idx, %n
	%both =w and %cont1, %idx_lt_n
	jnz %both, @run_check, @run_done
@run_check
	%src_idx_ptr =l add %src, %idx
	%src_idx =w loadub %src_idx_ptr
	%eq =w ceqw %src_idx, %src_i
	jnz %eq, @run_inc, @run_done
@run_inc
	%r =w add %r, 1
	%r_l =l extsw %r
	%idx =l add %i, %r_l
	jmp @run_loop
@run_done
	%r_ge_3 =w csgew %r, 3
	jnz %r_ge_3, @repeat, @literal
@repeat
	%o_plus_2 =l add %o, 2
	%fits =w cslel %o_plus_2, %cap
	jnz %fits, @rep_write, @fail
@rep_write
	%hdr =w sub 257, %r
	%out_hdr_ptr =l add %out, %o
	storeb %hdr, %out_hdr_ptr
	%src_ptr =l add %src, %i
	%b =w loadub %src_ptr
	%o1 =l add %o, 1
	%out_b_ptr =l add %out, %o1
	storeb %b, %out_b_ptr
	%o =l add %o, 2
	%r_l =l extsw %r
	%i =l add %i, %r_l
	jmp @loop
@literal
	%j =l copy %i
	%len =w copy 0
	jmp @lit_loop
@lit_loop
	%j_lt_n =w csltl %j, %n
	%diff =l sub %j, %i
	%diff_lt_128 =w csltl %diff, 128
	%both =w and %j_lt_n, %diff_lt_128
	jnz %both, @lit_check_run, @lit_done
@lit_check_run
	%rj =w copy 1
	%rj_l =l extsw %rj
	%jidx =l add %j, %rj_l
	%src_j_ptr =l add %src, %j
	%src_j =w loadub %src_j_ptr
	jmp @lit_run_loop
@lit_run_loop
	%cont1 =w csltw %rj, 128
	%jidx_lt_n =w csltl %jidx, %n
	%both =w and %cont1, %jidx_lt_n
	jnz %both, @lit_run_check, @lit_run_done
@lit_run_check
	%src_jidx_ptr =l add %src, %jidx
	%src_jidx =w loadub %src_jidx_ptr
	%eq =w ceqw %src_jidx, %src_j
	jnz %eq, @lit_run_inc, @lit_run_done
@lit_run_inc
	%rj =w add %rj, 1
	%rj_l =l extsw %rj
	%jidx =l add %j, %rj_l
	jmp @lit_run_loop
@lit_run_done
	%run_lt_3 =w csltw %rj, 3
	jnz %run_lt_3, @lit_include, @lit_done
@lit_include
	%j =l add %j, 1
	%len =w add %len, 1
	jmp @lit_loop
@lit_done
	%L =l extsw %len
	%pkt_size =l add %L, 1
	%o_plus_pkt =l add %o, %pkt_size
	%fits =w cslel %o_plus_pkt, %cap
	jnz %fits, @lit_write, @fail
@lit_write
	%hdr =w sub %len, 1
	%out_hdr_ptr =l add %out, %o
	storeb %hdr, %out_hdr_ptr
	%k =l copy 0
	jmp @copy_loop
@copy_loop
	%k_lt_len =w csltl %k, %L
	jnz %k_lt_len, @copy_body, @copy_done
@copy_body
	%src_off =l add %i, %k
	%src_ptr =l add %src, %src_off
	%byte =w loadub %src_ptr
	%dst_off =l add %o, 1
	%dst_off_k =l add %dst_off, %k
	%dst_ptr =l add %out, %dst_off_k
	storeb %byte, %dst_ptr
	%k =l add %k, 1
	jmp @copy_loop
@copy_done
	%o =l add %o, %pkt_size
	%i =l copy %j
	jmp @loop
@fail
	%zero =l copy 0
	%one =l copy 1
	%m1 =l sub %zero, %one
	ret %m1
@done
	ret %o
}
