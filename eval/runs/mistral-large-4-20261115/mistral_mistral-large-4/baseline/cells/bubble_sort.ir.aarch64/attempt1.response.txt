export function l $bubble_sort(l %a, l %n) {
@start
    %nm1 =l sub %n, 1
    jmp @outer_check
@outer_check
    %i =l phi @start 0, @outer_inc %i_next
    %outer_cond =w csltl %i, %nm1
    jnz %outer_cond, @outer_body, @end
@outer_body
    %remaining =l sub %nm1, %i
    jmp @inner_check
@inner_check
    %j =l phi @outer_body 0, @inner_inc %j_next
    %inner_cond =w csltl %j, %remaining
    jnz %inner_cond, @inner_body, @outer_inc
@inner_body
    %off_j =l mul %j, 8
    %p_j =l add %a, %off_j
    %off_j1 =l add %off_j, 8
    %p_j1 =l add %a, %off_j1
    %v_j =l loadl %p_j
    %v_j1 =l loadl %p_j1
    %gt =w csgtl %v_j, %v_j1
    jnz %gt, @swap, @no_swap
@swap
    storel %v_j1, %p_j
    storel %v_j, %p_j1
    jmp @inner_inc
@no_swap
    jmp @inner_inc
@inner_inc
    %j_next =l add %j, 1
    jmp @inner_check
@outer_inc
    %i_next =l add %i, 1
    jmp @outer_check
@end
    ret 0
}
