bubble_sort:
    mov r10, rsi
    dec r10
.Louter:
    test r10, r10
    jle .Ldone
    mov rcx, r10
    mov r9, rdi
.Linner:
    mov rax, qword ptr [r9]
    cmp rax, qword ptr [r9 + 8]
    jle .Lnoswap
    mov rdx, qword ptr [r9 + 8]
    mov qword ptr [r9 + 8], rax
    mov qword ptr [r9], rdx
.Lnoswap:
    add r9, 8
    dec rcx
    jnz .Linner
    dec r10
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
