export function l $base64_encode(l %src, l %n, l %out) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%three =l copy 3
	%four =l copy 4
	%twentysix =w copy 26
	%fiftytwo =w copy 52
	%sixtytwo =w copy 62
	%sixtythree =w copy 63

	%c47 =w copy 47

	%groups =l divl %np2, %three
	%outlen =l mull %groups, %four
	jmp @loop
@loop
	%i =l phi @start %zero, @body %inext
	%o =l phi @start %zero, @body %onext

	%has3 =w cslel %three, %remaining
	jnz %has3, @body, @after
@body
	%f_p0 =l addl %src, %i
	%f_b0 =w loadub %f_p0
	%f_p1 =l addl %f_p0, 1
	%f_b1 =w loadub %f_p1

	%f_b2 =w loadub %f_p2

	%f_s1 =w shlw %f_b1, 8

	%f0_b =w addw %f_idx0, 71
	%f0_c =w subw %f_idx0, 4
	%f0_sel4 =w select %f0_eq62, %c43, %c47
	%f0_sel3 =w select %f0_lt62, %f0_c, %f0_sel4
	%f0_sel2 =w select %f0_lt52, %f0_b, %f0_sel3
	%f0_ch =w select %f0_lt26, %f0_a, %f0_sel2

	%f1_lt62 =w csltw %f_idx1, %sixtytwo

	%f1_b =w addw %f_idx1, 71

	%f1_sel2 =w select %f1_lt52, %f1_b, %f1_sel3
	%f1_ch =w select %f1_lt26, %f1_a, %f1_sel2
	%f2_lt26 =w csltw %f_idx2, %twentysix
	%f2_lt52 =w csltw %f_idx2, %fiftytwo

	%f2_sel2 =w select %f2_lt52, %f2_b, %f2_sel3
	%f2_ch =w select %f2_lt26, %f2_a, %f2_sel2
	%f3_lt26 =w csltw %f_idx3, %twentysix
	%f3_lt52 =w csltw %f_idx3, %fiftytwo
	%f3_lt62 =w csltw %f_idx3, %sixtytwo
	%f3_eq62 =w ceqw %f_idx3, %sixtytwo
	%f3_a =w addw %f_idx3, 65
	%f3_b =w addw %f_idx3, 71
	%f3_c =w subw %f_idx3, 4
	%f3_sel4 =w select %f3_eq62, %c43, %c47
	%f3_sel3 =w select %f3_lt62, %f3_c, %f3_sel4
	%f3_sel2 =w select %f3_lt52, %f3_b, %f3_sel3
	%f3_ch =w select %f3_lt26, %f3_a, %f3_sel2
	%f_op0 =l addl %out, %o
	storeb %f0_ch, %f_op0
	%f_op1 =l addl %f_op0, 1
	storeb %f1_ch, %f_op1
	%f_op2 =l addl %f_op0, 2
	storeb %f2_ch, %f_op2
	%f_op3 =l addl %f_op0, 3
	storeb %f3_ch, %f_op3
	%inext =l addl %i, %three
	%onext =l addl %o, %four
	jmp @loop
@after
	%rem =l subl %n, %i
	%is1 =w ceql %rem, %one
	jnz %is1, @rem1, @check2
@check2
	%is2 =w ceql %rem, %two
	jnz %is2, @rem2, @done
@rem1

	%p10_sel3 =w select %p10_lt62, %p10_c, %p10_sel4
	%p10_sel2 =w select %p10_lt52, %p10_b, %p10_sel3
	%p10_ch =w select %p10_lt26, %p10_a, %p10_sel2

	%p2_sh1 =w shrw %p2_v24, 12

	%p2_sh2 =w shrw %p2_v24, 6
	%p2_idx2 =w andw %p2_sh2, %sixtythree
