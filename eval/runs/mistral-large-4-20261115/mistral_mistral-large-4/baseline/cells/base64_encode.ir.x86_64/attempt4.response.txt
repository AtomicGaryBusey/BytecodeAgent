export function l $base64_encode(l %src, l %n, l %out) {
@start
	%p0 =l copy %src
	%o0 =l copy %out
	%r0 =l copy %n
	jmp @loop
@loop
	%p =l phi @start %p0, @edone %p_next
	%o =l phi @start %o0, @edone %o_next
	%rem =l phi @start %r0, @edone %r_next
	%isdone =w ceql %rem, 0
	jnz %isdone, @end, @body
@body
	%full =w csgel %rem, 3
	jnz %full, @full3, @partial
@full3
	%f_b0 =l loadub %p
	%f_p1 =l add %p, 1
	%f_b1 =l loadub %f_p1
	%f_p2 =l add %p, 2
	%f_b2 =l loadub %f_p2
	%f_v1 =l shl %f_b0, 16
	%f_v2 =l shl %f_b1, 8
	%f_v3 =l or %f_v1, %f_v2
	%f_v =l or %f_v3, %f_b2
	%f_nc =l copy 4
	%f_adv =l copy 3
	jmp @emit
@partial
	%rem_is_1 =w ceql %rem, 1
	jnz %rem_is_1, @one, @two
@one
	%ob0 =l loadub %p
	%ov1 =l shl %ob0, 16
	%ov =l copy %ov1
	%onc =l copy 2
	%oadv =l copy 1
	jmp @emit
@two
	%tb0 =l loadub %p
	%tp1 =l add %p, 1
	%tb1 =l loadub %tp1
	%tv1 =l shl %tb0, 16
	%tv2 =l shl %tb1, 8
	%tv =l or %tv1, %tv2
	%tnc =l copy 3
	%tadv =l copy 2
	jmp @emit
@emit
	%v =l phi @full3 %f_v, @one %ov, @two %tv
	%nchars =l phi @full3 %f_nc, @one %onc, @two %tnc
	%adv =l phi @full3 %f_adv, @one %oadv, @two %tadv
	%oc0 =l copy %o
	%j0 =l copy 0
	jmp @eloop
@eloop
	%j =l phi @emit %j0, @ebody %j_next
	%oc =l phi @emit %oc0, @ebody %oc_next
	%jdone =w ceql %j, 4
	jnz %jdone, @edone, @ebody
@ebody
	%negj =l mul %j, -6
	%shift =l add %negj, 18
	%idx_raw =l shr %v, %shift
	%idx =l and %idx_raw, 63
	%ge26 =w csgel %idx, 26
	%ge26l =l extsw %ge26
	%ge52 =w csgel %idx, 52
	%ge52l =l extsw %ge52
	%ge62 =w csgel %idx, 62
	%ge62l =l extsw %ge62
	%eq63 =w ceql %idx, 63
	%eq63l =l extsw %eq63
	%ch1 =l add %idx, 65
	%adj1 =l mul %ge26l, 6
	%ch2 =l add %ch1, %adj1
	%adj2 =l mul %ge52l, -75
	%ch3 =l add %ch2, %adj2
	%adj3 =l mul %ge62l, -15
	%ch4 =l add %ch3, %adj3
	%adj4 =l mul %eq63l, 3
	%ch =l add %ch4, %adj4
	%padw =w csgel %j, %nchars
	%padl =l extsw %padw
	%negch =l mul %ch, -1
	%toeq =l add %negch, 61
	%padadj =l mul %padl, %toeq
	%chout =l add %ch, %padadj
	storeb %chout, %oc
	%oc_next =l add %oc, 1
	%j_next =l add %j, 1
	jmp @eloop
@edone
	%o_next =l copy %oc
	%p_next =l add %p, %adv
	%r_next =l sub %rem, %adv
	jmp @loop
@end
	%len =l sub %o, %out
	ret %len
}
