export function l $base64_encode(l %src, l %n, l %out) {
@start
	%p0 =l copy %src
	%o0 =l copy %out
	%rem0 =l copy %n
	jmp @loop
@loop
	%p =l phi @start %p0, @edone %p_next
	%o =l phi @start %o0, @edone %o_next
	%rem =l phi @start %rem0, @edone %rem_next
	%done =w ceql %rem, 0
	jnz %done, @end, @body
@body
	%full =w csgel %rem, 3
	jnz %full, @load3, @loadpartial
@load3
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%v3 =l shl %b0, 16
	%t =l shl %b1, 8
	%v3 =l or %v3, %t
	%v3 =l or %v3, %b2
	%nc3 =l copy 4
	%adv3 =l copy 3
	jmp @emit
@loadpartial
	%one =w ceql %rem, 1
	jnz %one, @load1, @load2
@load1
	%b0 =l loadub %p
	%v1 =l shl %b0, 16
	%nc1 =l copy 2
	%adv1 =l copy 1
	jmp @emit
@load2
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%v2 =l shl %b0, 16
	%t =l shl %b1, 8
	%v2 =l or %v2, %t
	%nc2 =l copy 3
	%adv2 =l copy 2
	jmp @emit
@emit
	%v =l phi @load3 %v3, @load1 %v1, @load2 %v2
	%nchars =l phi @load3 %nc3, @load1 %nc1, @load2 %nc2
	%adv =l phi @load3 %adv3, @load1 %adv1, @load2 %adv2
	%oc0 =l copy %o
	%j0 =l copy 0
	jmp @eloop
@eloop
	%j =l phi @emit %j0, @ebody %j1
	%oc =l phi @emit %oc0, @ebody %oc1
	%jdone =w ceql %j, 4
	jnz %jdone, @edone, @ebody
@ebody
	%negj =l mul %j, -6
	%shift =l add %negj, 18
	%idx =l shr %v, %shift
	%idx =l and %idx, 63
	%c26 =w csgel %idx, 26
	%c26l =l extsw %c26
	%c52 =w csgel %idx, 52
	%c52l =l extsw %c52
	%c62 =w csgel %idx, 62
	%c62l =l extsw %c62
	%c63 =w ceql %idx, 63
	%c63l =l extsw %c63
	%ch =l add %idx, 65
	%t1 =l mul %c26l, 6
	%ch =l add %ch, %t1
	%t2 =l mul %c52l, -75
	%ch =l add %ch, %t2
	%t3 =l mul %c62l, -15
	%ch =l add %ch, %t3
	%t4 =l mul %c63l, 3
	%ch =l add %ch, %t4
	%padw =w csgel %j, %nchars
	%padl =l extsw %padw
	%neg_ch =l mul %ch, -1
	%diff =l add %neg_ch, 61
	%t5 =l mul %padl, %diff
	%ch =l add %ch, %t5
	storeb %ch, %oc
	%oc1 =l add %oc, 1
	%j1 =l add %j, 1
	jmp @eloop
@edone
	%o_next =l copy %oc
	%p_next =l add %p, %adv
	%rem_next =l sub %rem, %adv
	jmp @loop
@end
	%len =l sub %o, %out
	ret %len
}
