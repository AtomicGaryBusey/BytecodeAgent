export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy %n
	%done =w ceql %i, 0
	jnz %done, @finish, @byte
@byte
	%b =w loadub %p
	%crc =w xor %crc, %b
	%k =w copy 8
	jmp @bits
@bits
	%odd =w and %crc, 1
	jnz %odd, @poly, @shift
@poly
	%t =w shr %crc, 1
	%crc =w xor %t, 3988292384
	%k =w sub %k, 1
	jnz %k, @bits, @next
@shift
	%crc =w shr %crc, 1
	%k =w sub %k, 1
	jnz %k, @bits, @next
@next
	%p =l add %p, 1
	%i =l sub %i, 1
	%done =w ceql %i, 0
	jnz %done, @finish, @byte
@finish
	%res =w xor %crc, 4294967295
	%out =l extuw %res
	ret %out
}
