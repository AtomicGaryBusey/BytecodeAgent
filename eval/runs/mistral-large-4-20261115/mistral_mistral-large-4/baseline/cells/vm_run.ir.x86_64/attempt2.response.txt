export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pcp =l alloc8 8
    %spp =l alloc8 8
    %jp =l alloc8 8
    storel 0, %pcp
    storel 0, %spp
    jmp @loop
@loop
    %pc =l loadl %pcp
    %sp =l loadl %spp
    %end =w ceql %pc, %len
    jnz %end, @done, @fetch
@fetch
    %poff =l mul %pc, 8
    %paddr =l add %code, %poff
    %op =l loadl %paddr
    %c0 =w ceql %op, 0
    jnz %c0, @halt, @t1
@t1
    %c1 =w ceql %op, 1
    jnz %c1, @push, @t2
@t2
    %c2 =w ceql %op, 2
    jnz %c2, @add, @t3
@t3
    %c3 =w ceql %op, 3
    jnz %c3, @sub, @t4
@t4
    %c4 =w ceql %op, 4
    jnz %c4, @mul, @t5
@t5
    %c5 =w ceql %op, 5
    jnz %c5, @dup, @t6
@t6
    %c6 =w ceql %op, 6
    jnz %c6, @swap, @t7
@t7
    %c7 =w ceql %op, 7
    jnz %c7, @drop, @t8
@t8
    %c8 =w ceql %op, 8
    jnz %c8, @over, @t9
@t9
    %c9 =w ceql %op, 9
    jnz %c9, @lt, @t10
@t10
    %c10 =w ceql %op, 10
    jnz %c10, @jz, @t11
@t11
    %c11 =w ceql %op, 11
    jnz %c11, @jmp, @default
@halt
    jmp @done
@push
    %pc1 =l add %pc, 1
    %ooff =l mul %pc1, 8
    %oaddr =l add %code, %ooff
    %val =l loadl %oaddr
    %soff =l mul %sp, 8
    %saddr =l add %stack, %soff
    storel %val, %saddr
    %sp1 =l add %sp, 1
    storel %sp1, %spp
    %pc2 =l add %pc, 2
    storel %pc2, %pcp
    jmp @loop
@add
    %spm1 =l sub %sp, 1
    %boff =l mul %spm1, 8
    %baddr =l add %stack, %boff
    %b =l loadl %baddr
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    %sum =l add %a, %b
    storel %sum, %aaddr
    %spnew =l sub %sp, 1
    storel %spnew, %spp
    %pcnew =l add %pc, 1
    storel %pcnew, %pcp
    jmp @loop
@sub
    %spm1 =l sub %sp, 1
    %boff =l mul %spm1, 8
    %baddr =l add %stack, %boff
    %b =l loadl %baddr
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    %diff =l sub %a, %b
    storel %diff, %aaddr
    %spnew =l sub %sp, 1
    storel %spnew, %spp
    %pcnew =l add %pc, 1
    storel %pcnew, %pcp
    jmp @loop
@mul
    %spm1 =l sub %sp, 1
    %boff =l mul %spm1, 8
    %baddr =l add %stack, %boff
    %b =l loadl %baddr
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    %prod =l mul %a, %b
    storel %prod, %aaddr
    %spnew =l sub %sp, 1
    storel %spnew, %spp
    %pcnew =l add %pc, 1
    storel %pcnew, %pcp
    jmp @loop
@dup
    %spm1 =l sub %sp, 1
    %toff =l mul %spm1, 8
    %taddr =l add %stack, %toff
    %top =l loadl %taddr
    %soff =l mul %sp, 8
    %saddr =l add %stack, %soff
    storel %top, %saddr
    %sp1 =l add %sp, 1
    storel %sp1, %spp
    %pc1 =l add %pc, 1
    storel %pc1, %pcp
    jmp @loop
@swap
    %spm1 =l sub %sp, 1
    %boff =l mul %spm1, 8
    %baddr =l add %stack, %boff
    %b =l loadl %baddr
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    storel %b, %aaddr
    storel %a, %baddr
    %pc1 =l add %pc, 1
    storel %pc1, %pcp
    jmp @loop
@drop
    %sp1 =l sub %sp, 1
    storel %sp1, %spp
    %pc1 =l add %pc, 1
    storel %pc1, %pcp
    jmp @loop
@over
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    %soff =l mul %sp, 8
    %saddr =l add %stack, %soff
    storel %a, %saddr
    %sp1 =l add %sp, 1
    storel %sp1, %spp
    %pc1 =l add %pc, 1
    storel %pc1, %pcp
    jmp @loop
@lt
    %spm1 =l sub %sp, 1
    %boff =l mul %spm1, 8
    %baddr =l add %stack, %boff
    %b =l loadl %baddr
    %spm2 =l sub %sp, 2
    %aoff =l mul %spm2, 8
    %aaddr =l add %stack, %aoff
    %a =l loadl %aaddr
    %cmp =w csltl %a, %b
    jnz %cmp, @lt_true, @lt_false
@lt_true
    storel 1, %aaddr
    jmp @lt_common
@lt_false
    storel 0, %aaddr
    jmp @lt_common
@lt_common
    %spnew =l sub %sp, 1
    storel %spnew, %spp
    %pcnew =l add %pc, 1
    storel %pcnew, %pcp
    jmp @loop
@jz
    %pc1 =l add %pc, 1
    %ooff =l mul %pc1, 8
    %oaddr =l add %code, %ooff
    %target =l loadl %oaddr
    %spm1 =l sub %sp, 1
    %voff =l mul %spm1, 8
    %vaddr =l add %stack, %voff
    %v =l loadl %vaddr
    %iszero =w ceql %v, 0
    jnz %iszero, @jz_zero, @jz_nonzero
@jz_zero
    storel %target, %pcp
    storel %spm1, %spp
    jmp @loop
@jz_nonzero
    %pc2 =l add %pc, 2
    storel %pc2, %pcp
    storel %spm1, %spp
    jmp @loop
@jmp
    %pc1 =l add %pc, 1
    %ooff =l mul %pc1, 8
    %oaddr =l add %code, %ooff
    %target =l loadl %oaddr
    storel %target, %pcp
    jmp @loop
@default
    %pc1 =l add %pc, 1
    storel %pc1, %pcp
    jmp @loop
@done
    %spd =l loadl %spp
    storel %spd, %jp
    jmp @zero_tail
@zero_tail
    %j =l loadl %jp
    %zc =w csltl %j, %cap
    jnz %zc, @zero_tail_body, @zero_tail_done
@zero_tail_body
    %joff =l mul %j, 8
    %jaddr =l add %stack, %joff
    storel 0, %jaddr
    %j1 =l add %j, 1
    storel %j1, %jp
    jmp @zero_tail
@zero_tail_done
    %has =w csgtl %spd, 0
    jnz %has, @ret_top, @ret_zero
@ret_top
    %spm1 =l sub %spd, 1
    %toff =l mul %spm1, 8
    %taddr =l add %stack, %toff
    %top =l loadl %taddr
    ret %top
@ret_zero
    ret 0
}
