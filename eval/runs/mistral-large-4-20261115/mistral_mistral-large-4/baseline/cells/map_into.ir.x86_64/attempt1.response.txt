export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%v =l loadl %srcp
	%r =l call %f(l %v)
	%dstp =l add %dst, %off
	storel %r, %dstp
	%less =w csltl %r, %v
	jnz %less, @inc, @next
@inc
	%count =l add %count, 1
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %count
}
