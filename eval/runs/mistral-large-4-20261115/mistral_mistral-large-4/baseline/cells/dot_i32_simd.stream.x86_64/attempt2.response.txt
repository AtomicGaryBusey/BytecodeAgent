dot_i32_simd:
    pxor xmm5, xmm5
.Lloop:
    cmp rdx, 4
    jl .Lhoriz
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqa xmm2, xmm0
    pmuldq xmm2, xmm1
    pshufd xmm3, xmm0, 0xf5
    pshufd xmm4, xmm1, 0xf5
    pmuldq xmm3, xmm4
    paddq xmm5, xmm2
    paddq xmm5, xmm3
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    jmp .Lloop
.Lhoriz:
    movq rax, xmm5
    psrldq xmm5, 8
    movq rcx, xmm5
    add rax, rcx
.Ltail:
    test rdx, rdx
    jle .Lret
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jmp .Ltail
.Lret:
    ret
