export function l $count_desc(l %a, l %n) {
@start
	%zero =l copy 0
	%small =w csltl %n, 2
	jnz %small, @retzero, @pre
@retzero
	ret %zero
@pre
	%i0 =l copy 0
	%cnt0 =l copy 0
	jmp @loop
@loop
	%i =l phi @pre %i0, @back %inext
	%cnt =l phi @pre %cnt0, @back %cntnext
	%limit =l sub %n, 1
	%cond =w csltl %i, %limit
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v0 =l loadl %p
	%p1 =l add %p, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @inc, @noinc
@inc
	%cnt1 =l add %cnt, 1
	jmp @back
@noinc
	jmp @back
@back
	%cntnext =l phi @inc %cnt1, @noinc %cnt
	%inext =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
