export function l $fib(l %n) {
@start
	%one =l copy 1
	%small =w cslel %n, %one
	jnz %small, @ret_n, @init
@ret_n
	ret %n
@init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 2
	jmp @loop
@loop
	%cond =w cslel %i, %n
	jnz %cond, @body, @done
@body
	%t =l add %a, %b
	%a =l copy %b
	%b =l copy %t
	%i =l add %i, %one
	jmp @loop
@done
	ret %b
}
