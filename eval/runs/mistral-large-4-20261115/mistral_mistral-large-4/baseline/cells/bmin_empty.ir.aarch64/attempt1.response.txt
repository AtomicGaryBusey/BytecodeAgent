export function l $bmin_empty(l %a, l %n) {
@start
    %zero =l copy 0
    %max =l copy 9223372036854775807
    jmp @loop
@loop
    %i =l phi @start %zero, @body %i_next
    %min =l phi @start %max, @body %min_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %p =l add %a, %off
    %v =l loadl %p
    %lt =w csltl %v, %min
    %min_next =l select %lt, %v, %min
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %min
}
