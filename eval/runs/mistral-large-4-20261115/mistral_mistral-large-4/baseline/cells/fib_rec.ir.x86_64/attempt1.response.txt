export function l $fib_rec(l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @rec, @small
@small
	%one =w ceql %n, 1
	jnz %one, @ret1, @ret0
@ret1
	%v1 =l copy 1
	ret %v1
@ret0
	%v0 =l copy 0
	ret %v0
@rec
	%n1 =l sub %n, 1
	%a =l call $fib_rec(l %n1)
	%n2 =l sub %n, 2
	%b =l call $fib_rec(l %n2)
	%s =l add %a, %b
	ret %s
}
