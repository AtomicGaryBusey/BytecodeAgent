export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nz =w ceql %n, 0
	jnz %nz, @ret_zero, @cmp
@cmp
	%lt =w csltl %dst, %src
	jnz %lt, @forward, @backward
@forward
	%i =l copy 0
	jmp @forward_loop
@forward_loop
	%cont =w csltl %i, %n
	jnz %cont, @forward_body, @ret_zero
@forward_body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i =l add %i, 1
	jmp @forward_loop
@backward
	%k =l copy %n
	jmp @back_loop
@back_loop
	%kz =w ceql %k, 0
	jnz %kz, @ret_zero, @back_body
@back_body
	%k =l sub %k, 1
	%offb =l mul %k, 8
	%spb =l add %src, %offb
	%dpb =l add %dst, %offb
	%vb =l loadl %spb
	storel %vb, %dpb
	jmp @back_loop
@ret_zero
	%zero =l copy 0
	ret %zero
}
