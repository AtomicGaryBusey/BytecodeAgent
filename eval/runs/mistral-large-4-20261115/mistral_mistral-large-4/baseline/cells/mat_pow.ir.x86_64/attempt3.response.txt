export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%base =l copy %tmp
	%scratch =l add %tmp, %nn8
	%zero =l copy 0
	jmp @init_out_pre

@init_out_pre
	jmp @init_out_i

@init_out_i
	%oi =l phi @init_out_pre 0, @init_out_i_end %oi_next
	%oi_cond =w csltl %oi, %n
	jnz %oi_cond, @init_out_j_pre, @init_out_done

@init_out_j_pre
	jmp @init_out_j

@init_out_j
	%oj =l phi @init_out_j_pre 0, @init_out_body %oj_next
	%oj_cond =w csltl %oj, %n
	jnz %oj_cond, @init_out_body, @init_out_i_end

@init_out_body
	%oin =l mul %oi, %n
	%oidx =l add %oin, %oj
	%ooff =l mul %oidx, 8
	%oaddr =l add %out, %ooff
	%eq =w ceql %oi, %oj
	%val =l extsw %eq
	storel %val, %oaddr
	%oj_next =l add %oj, 1
	jmp @init_out_j

@init_out_i_end
	%oi_next =l add %oi, 1
	jmp @init_out_i

@init_out_done
	jmp @copy_base_pre

@copy_base_pre
	jmp @copy_base_i

@copy_base_i
	%ci =l phi @copy_base_pre 0, @copy_base_body %ci_next
	%ci_cond =w csltl %ci, %nn
	jnz %ci_cond, @copy_base_body, @copy_base_done

@copy_base_body
	%coff =l mul %ci, 8
	%csrc =l add %a, %coff
	%cdst =l add %base, %coff
	%cv =l loadl %csrc
	storel %cv, %cdst
	%ci_next =l add %ci, 1
	jmp @copy_base_i

@copy_base_done
	jmp @exp_pre

@exp_pre
	jmp @exp_loop

@exp_loop
	%kcur =l phi @exp_pre %k, @exp_body_end %kcur_next
	%kzero =w ceql %kcur, 0
	jnz %kzero, @zero_pre, @exp_body

@exp_body
	jmp @mul_i_pre

@mul_i_pre
	jmp @mul_i

@mul_i
	%mi =l phi @mul_i_pre 0, @mul_i_end %mi_next
	%mi_cond =w csltl %mi, %n
	jnz %mi_cond, @mul_j_pre, @mul_done

@mul_j_pre
	jmp @mul_j

@mul_j
	%mj =l phi @mul_j_pre 0, @mul_store %mj_next
	%mj_cond =w csltl %mj, %n
	jnz %mj_cond, @mul_t_pre, @mul_i_end

@mul_t_pre
	jmp @mul_t

@mul_t
	%mt =l phi @mul_t_pre 0, @mul_t_body %mt_next
	%msum =l phi @mul_t_pre 0, @mul_t_body %msum_next
	%mt_cond =w csltl %mt, %n
	jnz %mt_cond, @mul_t_body, @mul_store

@mul_t_body
	%ain =l mul %mi, %n
	%aidx =l add %ain, %mt
	%aoff =l mul %aidx, 8
	%aaddr =l add %out, %aoff
	%bt =l mul %mt, %n
	%bidx =l add %bt, %mj
	%boff =l mul %bidx, 8
	%baddr =l add %base, %boff
	%va =l loadl %aaddr
	%vb =l loadl %baddr
	%prod =l mul %va, %vb
	%msum_next =l add %msum, %prod
	%mt_next =l add %mt, 1
	jmp @mul_t

@mul_store
	%cin =l mul %mi, %n
	%cidx =l add %cin, %mj
	%coff =l mul %cidx, 8
	%caddr =l add %scratch, %coff
	storel %msum, %caddr
	%mj_next =l add %mj, 1
	jmp @mul_j

@mul_i_end
	%mi_next =l add %mi, 1
	jmp @mul_i

@mul_done
	jmp @copy_pre

@copy_pre
	jmp @copy_i

@copy_i
	%cpyi =l phi @copy_pre 0, @copy_body %cpyi_next
	%cpy_cond =w csltl %cpyi, %nn
	jnz %cpy_cond, @copy_body, @copy_done

@copy_body
	%cpoff =l mul %cpyi, 8
	%cpsrc =l add %scratch, %cpoff
	%cpdst =l add %out, %cpoff
	%cpv =l loadl %cpsrc
	storel %cpv, %cpdst
	%cpyi_next =l add %cpyi, 1
	jmp @copy_i

@copy_done
	jmp @exp_body_end

@exp_body_end
	%kcur_next =l sub %kcur, 1
	jmp @exp_loop

@zero_pre
	%ztotal =l add %nn, %nn
	jmp @zero_i

@zero_i
	%zi =l phi @zero_pre 0, @zero_body %zi_next
	%zi_cond =w csltl %zi, %ztotal
	jnz %zi_cond, @zero_body, @zero_done

@zero_body
	%zoff =l mul %zi, 8
	%zaddr =l add %tmp, %zoff
	storel %zero, %zaddr
	%zi_next =l add %zi, 1
	jmp @zero_i

@zero_done
	%ret =l copy 0
	ret %ret
}
