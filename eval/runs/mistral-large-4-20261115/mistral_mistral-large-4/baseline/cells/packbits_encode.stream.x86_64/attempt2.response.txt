packbits_encode:
    push rbx
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    mov r11, rcx
    xor rbx, rbx
    xor rdx, rdx
.Lloop:
    cmp rbx, r9
    jae .Ldone
    movzx ecx, byte ptr [r8 + rbx]
    lea rdi, [rbx + 1]
    mov rax, 1
.Lrun_loop:
    cmp rax, 128
    jae .Lrun_done
    cmp rdi, r9
    jae .Lrun_done
    cmp byte ptr [r8 + rdi], cl
    jne .Lrun_done
    inc rax
    inc rdi
    jmp .Lrun_loop
.Lrun_done:
    cmp rax, 3
    jae .Lrepeat
    mov rsi, rbx
.Llit_scan:
    cmp rsi, r9
    jae .Llit_end
    mov rdi, rsi
    sub rdi, rbx
    cmp rdi, 128
    jae .Llit_end
    movzx ecx, byte ptr [r8 + rsi]
    lea rdi, [rsi + 1]
    mov rax, 1
.Lrun_j:
    cmp rax, 128
    jae .Lrun_j_done
    cmp rdi, r9
    jae .Lrun_j_done
    cmp byte ptr [r8 + rdi], cl
    jne .Lrun_j_done
    inc rax
    inc rdi
    jmp .Lrun_j
.Lrun_j_done:
    cmp rax, 3
    jae .Llit_end
    inc rsi
    jmp .Llit_scan
.Llit_end:
    mov rdi, rsi
    sub rdi, rbx
    lea rax, [rdi + 1]
    mov rcx, r11
    sub rcx, rdx
    cmp rcx, rax
    jb .Lfail
    lea rax, [rdi - 1]
    mov [r10], al
    mov rcx, rbx
    lea rdi, [r10 + 1]
.Lcopy_loop:
    cmp rcx, rsi
    jae .Lcopy_done
    movzx eax, byte ptr [r8 + rcx]
    mov [rdi], al
    inc rcx
    inc rdi
    jmp .Lcopy_loop
.Lcopy_done:
    mov rax, rsi
    sub rax, rbx
    add r10, rax
    inc r10
    add rdx, rax
    inc rdx
    mov rbx, rsi
    jmp .Lloop
.Lrepeat:
    mov rcx, r11
    sub rcx, rdx
    cmp rcx, 2
    jb .Lfail
    mov rcx, 257
    sub rcx, rax
    mov [r10], cl
    movzx edi, byte ptr [r8 + rbx]
    mov [r10 + 1], dil
    add r10, 2
    add rdx, 2
    add rbx, rax
    jmp .Lloop
.Ldone:
    mov rax, rdx
    pop rbx
    ret
.Lfail:
    mov rax, -1
    pop rbx
    ret
