export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy %n
	j @outer
@outer
	%pos =w csltl 0, %k
	jnz %pos, @outer_body, @done
@outer_body
	%x =l copy %k
	%steps =l copy 0
	j @inner
@inner
	%is1 =w ceql %x, 1
	jnz %is1, @after_inner, @inner_body
@inner_body
	%par =l and %x, 1
	%odd =w ceql %par, 1
	jnz %odd, @odd, @even
@odd
	%triple =l mul %x, 3
	%x =l add %triple, 1
	j @inc
@even
	%x =l div %x, 2
	j @inc
@inc
	%steps =l add %steps, 1
	j @inner
@after_inner
	%gt =w csltl %best, %steps
	jnz %gt, @update, @next
@update
	%best =l copy %steps
	j @next
@next
	%k =l sub %k, 1
	j @outer
@done
	ret %best
}
