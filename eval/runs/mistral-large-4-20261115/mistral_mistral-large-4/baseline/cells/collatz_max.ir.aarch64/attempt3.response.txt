export function l $collatz_max(l %n) {
@start
	%bestp =l alloc8 8
	%kp =l alloc8 8
	%xp =l alloc8 8
	%stepsp =l alloc8 8
	storel 0, %bestp
	storel %n, %kp
	jmp @outer
@outer
	%k =l loadl %kp
	%cont =w csgtl %k, 0
	jnz %cont, @outer_body, @done
@outer_body
	storel %k, %xp
	storel 0, %stepsp
	jmp @inner
@inner
	%x =l loadl %xp
	%is1 =w ceql %x, 1
	jnz %is1, @after_inner, @inner_body
@inner_body
	%par =l and %x, 1
	%odd =w ceql %par, 1
	jnz %odd, @odd, @even
@odd
	%triple =l mul %x, 3
	%nx =l add %triple, 1
	storel %nx, %xp
	jmp @inc
@even
	%nx =l shr %x, 1
	storel %nx, %xp
	jmp @inc
@inc
	%s =l loadl %stepsp
	%s1 =l add %s, 1
	storel %s1, %stepsp
	jmp @inner
@after_inner
	%s2 =l loadl %stepsp
	%b =l loadl %bestp
	%gt =w csgtl %s2, %b
	jnz %gt, @update, @next
@update
	storel %s2, %bestp
	jmp @next
@next
	%k2 =l loadl %kp
	%k1 =l sub %k2, 1
	storel %k1, %kp
	jmp @outer
@done
	%res =l loadl %bestp
	ret %res
}
