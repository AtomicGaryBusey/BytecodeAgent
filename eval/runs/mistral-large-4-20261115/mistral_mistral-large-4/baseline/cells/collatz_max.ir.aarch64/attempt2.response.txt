export function l $collatz_max(l %n) {
@start
	%bestp =l alloc8 $8
	%kp =l alloc8 $8
	%xp =l alloc8 $8
	%stepsp =l alloc8 $8
	storel $0, %bestp
	storel %n, %kp
	jmp @outer
@outer
	%k_outer =l loadl %kp
	%pos =w csgtl %k_outer, $0
	jnz %pos, @outer_body, @done
@outer_body
	storel %k_outer, %xp
	storel $0, %stepsp
	jmp @inner
@inner
	%x =l loadl %xp
	%is1 =w ceql %x, $1
	jnz %is1, @after_inner, @inner_body
@inner_body
	%par =l and %x, $1
	%odd =w ceql %par, $1
	jnz %odd, @odd, @even
@odd
	%triple =l mul %x, $3
	%x_odd =l add %triple, $1
	storel %x_odd, %xp
	jmp @inc
@even
	%x_even =l shr %x, $1
	storel %x_even, %xp
	jmp @inc
@inc
	%steps_inc =l loadl %stepsp
	%steps_inc1 =l add %steps_inc, $1
	storel %steps_inc1, %stepsp
	jmp @inner
@after_inner
	%steps_after =l loadl %stepsp
	%best_after =l loadl %bestp
	%gt =w csltl %best_after, %steps_after
	jnz %gt, @update, @next
@update
	storel %steps_after, %bestp
	jmp @next
@next
	%k_next =l loadl %kp
	%k_next1 =l sub %k_next, $1
	storel %k_next1, %kp
	jmp @outer
@done
	%result =l loadl %bestp
	ret %result
}
