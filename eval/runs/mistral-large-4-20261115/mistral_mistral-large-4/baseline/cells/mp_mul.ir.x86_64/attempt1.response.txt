export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l copy 0
	%n =l add %na, %nb
	%i0 =l copy %zero
	jmp @zero_loop
@zero_loop
	%zc =w cultl %i0, %n
	jnz %zc, @zero_body, @zero_done
@zero_body
	%zoff =l mul %i0, 8
	%zptr =l add %r, %zoff
	storel %zero, %zptr
	%i0n =l add %i0, 1
	%i0 =l copy %i0n
	jmp @zero_loop
@zero_done
	%i =l copy %zero
	jmp @outer_cond
@outer_cond
	%oc =w cultl %i, %na
	jnz %oc, @outer_body, @outer_done
@outer_body
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%ai =l loadl %aptr
	%carry =l copy %zero
	%j =l copy %zero
	jmp @inner_cond
@inner_cond
	%ic =w cultl %j, %nb
	jnz %ic, @inner_body, @inner_done
@inner_body
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bj =l loadl %bptr
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rptr =l add %r, %roff
	%rij =l loadl %rptr
	%ai0 =l and %ai, 4294967295
	%ai1 =l shr %ai, 32
	%bj0 =l and %bj, 4294967295
	%bj1 =l shr %bj, 32
	%p00 =l mul %ai0, %bj0
	%p01 =l mul %ai0, %bj1
	%p10 =l mul %ai1, %bj0
	%p11 =l mul %ai1, %bj1
	%s =l add %p01, %p10
	%midc =w cultl %s, %p01
	%shifted =l shl %s, 32
	%losum =l add %p00, %shifted
	%loc =w cultl %losum, %p00
	%plo =l copy %losum
	%midhi =l shr %s, 32
	%midcl =l extuw %midc
	%midcshift =l shl %midcl, 32
	%hi1 =l add %p11, %midhi
	%hi2 =l add %hi1, %midcshift
	%locl =l extuw %loc
	%phi =l add %hi2, %locl
	%s1 =l add %plo, %rij
	%c1 =w cultl %s1, %plo
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%ctot =w add %c1, %c2
	%ctotl =l extuw %ctot
	%newcarry =l add %phi, %ctotl
	storel %s2, %rptr
	%carry =l copy %newcarry
	%jn =l add %j, 1
	%j =l copy %jn
	jmp @inner_cond
@inner_done
	%endidx =l add %i, %nb
	%endoff =l mul %endidx, 8
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%in =l add %i, 1
	%i =l copy %in
	jmp @outer_cond
@outer_done
	%scan =l sub %n, 1
	jmp @scan_loop
@scan_loop
	%neg =w csltl %scan, %zero
	jnz %neg, @scan_done, @scan_body
@scan_body
	%soff =l mul %scan, 8
	%sptr =l add %r, %soff
	%val =l loadl %sptr
	%iszero =w ceql %val, %zero
	jnz %iszero, @scan_next, @found
@scan_next
	%scanp =l sub %scan, 1
	%scan =l copy %scanp
	jmp @scan_loop
@found
	%res =l add %scan, 1
	ret %res
@scan_done
	ret %zero
}
