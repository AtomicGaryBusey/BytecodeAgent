export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00h =l shr %p00, 32
	%p01l =l and %p01, 4294967295
	%p10l =l and %p10, 4294967295
	%t =l add %p01l, %p10l
	%t2 =l add %t, %p00h
	%carry =l shr %t2, 32
	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%h1 =l add %p11, %p01h
	%h2 =l add %h1, %p10h
	%h3 =l add %h2, %carry
	ret %h3
}
