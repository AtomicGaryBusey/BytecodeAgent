export function l $qsort_pairs(l %a, l %n) {
@start
  %zero =l sub %n, %n
  %small =w csltl %n, 2
  jnz %small, @ret0, @body
@ret0
  ret %zero
@body
  %nm1 =l sub %n, 1
  %mid =l shr %n, 1
  %off_mid =l mul %mid, 16
  %pmid =l add %a, %off_mid
  %pmid1 =l add %pmid, 8
  %off_last =l mul %nm1, 16
  %plast =l add %a, %off_last
  %plast1 =l add %plast, 8
  %kmid =l loadl %pmid
  %vmid =l loadl %pmid1
  %klast =l loadl %plast
  %vlast =l loadl %plast1
  storel %klast, %pmid
  storel %vlast, %pmid1
  storel %kmid, %plast
  storel %vmid, %plast1
  %pkey =l loadl %plast
  %pval =l loadl %plast1
  jmp @loop
@loop
  %i =l phi @body %zero, @cont %i_next
  %j =l phi @body %zero, @cont %j_next
  %cond =w csltl %j, %nm1
  jnz %cond, @check, @after
@check
  %offj =l mul %j, 16
  %pj =l add %a, %offj
  %pj1 =l add %pj, 8
  %key =l loadl %pj
  %val =l loadl %pj1
  %lt_key =w csltl %key, %pkey
  %eq_key =w ceql %key, %pkey
  %lt_val =w csltl %val, %pval
  %t =w and %eq_key, %lt_val
  %less =w or %lt_key, %t
  jnz %less, @swap, @noswap
@swap
  %offi =l mul %i, 16
  %pi =l add %a, %offi
  %pi1 =l add %pi, 8
  %ki =l loadl %pi
  %vi =l loadl %pi1
  storel %key, %pi
  storel %val, %pi1
  storel %ki, %pj
  storel %vi, %pj1
  %i_inc_swap =l add %i, 1
  %j_inc_swap =l add %j, 1
  jmp @cont
@noswap
  %i_inc_noswap =l add %i, %zero
  %j_inc_noswap =l add %j, 1
  jmp @cont
@cont
  %i_next =l phi @swap %i_inc_swap, @noswap %i_inc_noswap
  %j_next =l phi @swap %j_inc_swap, @noswap %j_inc_noswap
  jmp @loop
@after
  %offi2 =l mul %i, 16
  %pi2 =l add %a, %offi2
  %pi12 =l add %pi2, 8
  %ki2 =l loadl %pi2
  %vi2 =l loadl %pi12
  %klast2 =l loadl %plast
  %vlast2 =l loadl %plast1
  storel %klast2, %pi2
  storel %vlast2, %pi12
  storel %ki2, %plast
  storel %vi2, %plast1
  %dummy1 =l call $qsort_pairs(l %a, l %i)
  %i_plus =l add %i, 1
  %off_right =l mul %i_plus, 16
  %pright =l add %a, %off_right
  %nright =l sub %n, %i_plus
  %dummy2 =l call $qsort_pairs(l %pright, l %nright)
  ret %zero
}
