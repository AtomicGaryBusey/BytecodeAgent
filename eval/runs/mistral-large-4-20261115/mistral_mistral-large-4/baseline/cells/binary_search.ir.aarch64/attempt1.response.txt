export function l $binary_search(l %a, l %n, l %key) {
@start
    %lo =l copy 0
    %hi =l copy %n
    jmp @loop
@loop
    %cond =w csltl %lo, %hi
    jnz %cond, @body, @notfound
@body
    %diff =l sub %hi, %lo
    %half =l div %diff, 2
    %mid =l add %lo, %half
    %off =l mul %mid, 8
    %p =l add %a, %off
    %v =l loadl %p
    %lt =w csltl %key, %v
    jnz %lt, @less, @checkgt
@less
    %hi =l copy %mid
    jmp @loop
@checkgt
    %gt =w csgtl %key, %v
    jnz %gt, @greater, @found
@greater
    %lo =l add %mid, 1
    jmp @loop
@found
    ret %mid
@notfound
    %neg =l copy -1
    ret %neg
}
