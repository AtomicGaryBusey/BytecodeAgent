isqrt:
    movz x2, #0xffff, lsl #16
    movk x2, #0xffff
    mov x1, #0
.Lloop:
    cmp x1, x2
    b.eq .Ldone
    add x3, x1, x2
    add x3, x3, #1
    lsr x3, x3, #1
    mul x4, x3, x3
    cmp x4, x0
    b.ls .Llo
    sub x2, x3, #1
    b .Lloop
.Llo:
    mov x1, x3
    b .Lloop
.Ldone:
    mov x0, x1
    ret
