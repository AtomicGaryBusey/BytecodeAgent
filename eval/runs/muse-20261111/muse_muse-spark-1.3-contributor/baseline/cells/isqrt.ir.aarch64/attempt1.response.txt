export function l $isqrt(l %n0) {
@start
  %one1 =l copy 1
  %one0 =l shl %one1, 62
  %res0 =l copy 0
  %op0 =l copy %n0
  jmp @shrink_check
@shrink_check
  %one_s =l phi @start %one0, @shrink_body %one_sb
  %c1 =w cugtl %one_s, %op0
  jnz %c1, @shrink_body, @main_check
@shrink_body
  %one_sb =l shr %one_s, 2
  jmp @shrink_check
@main_check
  %op_m =l phi @shrink_check %op0, @main_latch %op_next
  %res_m =l phi @shrink_check %res0, @main_latch %res_next
  %one_m =l phi @shrink_check %one_s, @main_latch %one_next
  %c2 =w cnel %one_m, 0
  jnz %c2, @main_body, @end
@main_body
  %tmp =l add %res_m, %one_m
  %c3 =w cugel %op_m, %tmp
  jnz %c3, @then, @else
@then
  %op_then =l sub %op_m, %tmp
  %sh =l shl %one_m, 1
  %res_then =l add %res_m, %sh
  jmp @main_latch
@else
  jmp @main_latch
@main_latch
  %op_lat =l phi @then %op_then, @else %op_m
  %res_lat =l phi @then %res_then, @else %res_m
  %res_next =l shr %res_lat, 1
  %one_next =l shr %one_m, 2
  %op_next =l copy %op_lat
  jmp @main_check
@end
  ret %res_m
}
