export function l $lcs_len(l %x0, l %m0, l %y0, l %n0, l %row0) {
@start
  %zero =l copy 0
  jmp @zero_cond
@zero_cond
  %k =l phi @start 0, @zero_body %k_next
  %zc =w csgtl %k, %n0
  jnz %zc, @outer_cond, @zero_body
@zero_body
  %zoff =l mul %k, 8
  %zptr =l add %row0, %zoff
  storel %zero, %zptr
  %k_next =l add %k, 1
  jmp @zero_cond
@outer_cond
  %i =l phi @zero_cond 0, @outer_latch %i_next
  %oc =w csltl %i, %m0
  jnz %oc, @inner_cond, @outer_done
@inner_cond
  %j =l phi @outer_cond 1, @inner_latch %j_next
  %diag =l phi @outer_cond 0, @inner_latch %diag_next
  %ic =w csgtl %j, %n0
  jnz %ic, @outer_latch, @inner_body
@inner_body
  %joff =l mul %j, 8
  %row_j_ptr =l add %row0, %joff
  %up =l loadl %row_j_ptr
  %xptr =l add %x0, %i
  %xb =w loadub %xptr
  %jm1 =l sub %j, 1
  %yptr =l add %y0, %jm1
  %yb =w loadub %yptr
  %eq =w ceqw %xb, %yb
  jnz %eq, @match, @nomatch
@match
  %dinc =l add %diag, 1
  storel %dinc, %row_j_ptr
  jmp @inner_latch
@nomatch
  %jm1_off =l mul %jm1, 8
  %row_jm1_ptr =l add %row0, %jm1_off
  %left =l loadl %row_jm1_ptr
  %mc =w csltl %up, %left
  jnz %mc, @take_left, @take_up
@take_left
  storel %left, %row_j_ptr
  jmp @inner_latch
@take_up
  storel %up, %row_j_ptr
  jmp @inner_latch
@inner_latch
  %j_next =l add %j, 1
  %diag_next =l copy %up
  jmp @inner_cond
@outer_latch
  %i_next =l add %i, 1
  jmp @outer_cond
@outer_done
  %noff =l mul %n0, 8
  %nptr =l add %row0, %noff
  %res =l loadl %nptr
  ret %res
}
