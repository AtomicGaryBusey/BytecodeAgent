bubble_sort:

mov x10, x1
cmp x10, #1
b.le .Ldone
mov x11, #0
.Louter:
sub x12, x10, #1
cmp x11, x12
b.ge .Ldone
sub x12, x12, x11
mov x13, #0
.Linner:
cmp x13, x12
b.ge .Lnext_outer
ldr x14, [x9, x13, lsl #3]
add x0, x13, #1
ldr x15, [x9, x0, lsl #3]
cmp x14, x15
b.le .Lnoswap
str x15, [x9, x13, lsl #3]
str x14, [x9, x0, lsl #3]
.Lnoswap:
add x13, x13, #1
b .Linner
.Lnext_outer:
add x11, x11, #1
b .Louter
.Ldone:
mov x0, #0
ret
