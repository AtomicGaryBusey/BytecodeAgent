export function l $map_sum(l %a, l %n, l %f) {
@start
	%z =l copy 0
	jmp @loop
@loop
	%i =l phi @start %z, @body %i2
	%s =l phi @start %z, @body %s2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%t =l mul %i, 8
	%p =l add %a, %t
	%x =l loadl %p
	%r =l call %f(l %x)
	%s2 =l add %s, %r
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %s
}
