export function l $fib_rec(l %n) {
@start
	%c0 =w ceql %n, 0
	jnz %c0, @ret0, @chk1
@ret0
	ret 0
@chk1
	%c1 =w ceql %n, 1
	jnz %c1, @ret1, @recurse
@ret1
	ret 1
@recurse
	%nm1 =l sub %n, 1
	%r1 =l call $fib_rec(l %nm1)
	%nm2 =l sub %n, 2
	%r2 =l call $fib_rec(l %nm2)
	%sum =l add %r1, %r2
	ret %sum
}
