packbits_encode:
mov r8, rdi
mov r10, rsi
mov r9, rdx
mov r11, rcx
.Louter:
test r10, r10
jz .Ldone
mov rsi, r10
cmp rsi, 128
jbe .LrunLimOk
mov rsi, 128
.LrunLimOk:
mov al, byte ptr [r8]
mov rcx, 1
.LrunLoop:
cmp rcx, rsi
jae .LrunDone
cmp byte ptr [r8 + rcx], al
jne .LrunDone
inc rcx
jmp .LrunLoop
.LrunDone:
cmp rcx, 3
jae .Lrepeat
mov rdi, 0
.LlitScan:
cmp rdi, r10
jae .LlitDone
cmp rdi, 128
jae .LlitDone
mov rax, r10
sub rax, rdi
cmp rax, 3
jb .LlitNext
mov al, byte ptr [r8 + rdi]
cmp byte ptr [r8 + rdi + 1], al
jne .LlitNext
cmp byte ptr [r8 + rdi + 2], al
jne .LlitNext
jmp .LlitDone
.LlitNext:
inc rdi
jmp .LlitScan
.LlitDone:
mov rax, rdi
inc rax
cmp rax, r11
ja .Lfail
mov rax, rdi
dec rax
mov byte ptr [r9], al
mov rcx, 0
.LcopyLoop:
cmp rcx, rdi
jae .LcopyDone
mov al, byte ptr [r8 + rcx]
mov byte ptr [r9 + rcx + 1], al
inc rcx
jmp .LcopyLoop
.LcopyDone:
add r8, rdi
sub r10, rdi
mov rax, rdi
inc rax
add r9, rax
sub r11, rax
jmp .Louter
.Lrepeat:
cmp r11, 2
jb .Lfail
mov rax, 257
sub rax, rcx
mov byte ptr [r9], al
mov al, byte ptr [r8]
mov byte ptr [r9 + 1], al
add r8, rcx
sub r10, rcx
add r9, 2
sub r11, 2
jmp .Louter
.Ldone:
mov rax, r9
sub rax, rdx
ret
.Lfail:
mov rax, -1
ret
