export function l $count_nonzero_prefix(l %a, l %n) {
@start
%zero =l copy 0
jmp @loop
@loop
%i =l phi @start %zero, @cont %inext
%c =w csltl %i, %n
jnz %c, @body, @done
@body
%off =l mul %i, 8
%addr =l add %a, %off
%v =l loadl %addr
%z =w ceql %v, %zero
jnz %z, @found, @cont
@cont
%inext =l add %i, 1
jmp @loop
@found
ret %i
@done
ret %i
}
