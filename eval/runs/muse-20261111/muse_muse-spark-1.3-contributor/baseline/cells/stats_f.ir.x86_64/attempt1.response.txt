export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %i0 =l copy 0
  %sum0 =l copy 0
  %cnt0 =l copy 0
  %min0 =l copy 0
  %max0 =l copy 0
  jmp @loop
@loop
  %i =l phi @start %i0, @maxjoin %inext
  %sum =l phi @start %sum0, @maxjoin %sum_back
  %cnt =l phi @start %cnt0, @maxjoin %cnt_back
  %min =l phi @start %min0, @maxjoin %min_back
  %max =l phi @start %max0, @maxjoin %max2
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %off =l mul %i, 8
  %addr =l add %a, %off
  %x =l loadl %addr
  %y =l call %f(l %x)
  %sumnext =l add %sum, %y
  %low =l and %y, 1
  %isodd =w cnel %low, 0
  jnz %isodd, @odd, @even
@odd
  %cnt_odd =l add %cnt, 1
  jmp @cntjoin
@even
  %cnt_even =l copy %cnt
  jmp @cntjoin
@cntjoin
  %cnt2 =l phi @odd %cnt_odd, @even %cnt_even
  %is_first =w ceql %i, 0
  %isless0 =w csltl %y, %min
  %isless =w or %is_first, %isless0
  jnz %isless, @newmin, @oldmin
@newmin
  %min_new =l copy %y
  jmp @minjoin
@oldmin
  %min_old =l copy %min
  jmp @minjoin
@minjoin
  %min2 =l phi @newmin %min_new, @oldmin %min_old
  %isgreater0 =w csgtl %y, %max
  %isgreater =w or %is_first, %isgreater0
  jnz %isgreater, @newmax, @oldmax
@newmax
  %max_new =l copy %y
  jmp @maxjoin
@oldmax
  %max_old =l copy %max
  jmp @maxjoin
@maxjoin
  %max2 =l phi @newmax %max_new, @oldmax %max_old
  %sum_back =l copy %sumnext
  %cnt_back =l copy %cnt2
  %min_back =l copy %min2
  %inext =l add %i, 1
  jmp @loop
@done
  storel %min, %out
  %out1 =l add %out, 8
  storel %max, %out1
  %out2 =l add %out, 16
  storel %sum, %out2
  ret %cnt
}
