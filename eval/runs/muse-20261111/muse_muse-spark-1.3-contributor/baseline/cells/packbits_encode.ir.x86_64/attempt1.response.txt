export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %i0 =w copy 0
  %o0 =w copy 0
  jmp @outer_cond
@outer_cond
  %i =w phi @start %i0, @repeat_write %i_rep_next, @lit_copy_done %i_lit_next
  %o =w phi @start %o0, @repeat_write %o_rep_next, @lit_copy_done %o_lit_next
  %i_ext =l extuw %i
  %c_outer =w csltl %i_ext, %n
  jnz %c_outer, @run_init, @done
@run_init
  %src_i_addr =l add %src, %i_ext
  %b =w loadub %src_i_addr
  %r0 =w copy 1
  jmp @run_cond
@run_cond
  %r =w phi @run_init %r0, @run_inc %r1
  %c_r128 =w csgew %r, 128
  jnz %c_r128, @run_done, @run_bounds
@run_bounds
  %ir_w =w add %i, %r
  %ir_ext =l extuw %ir_w
  %c_irn =w csltl %ir_ext, %n
  jnz %c_irn, @run_byte, @run_done
@run_byte
  %src_ir_addr =l add %src, %ir_ext
  %b2 =w loadub %src_ir_addr
  %c_eq =w ceqw %b2, %b
  jnz %c_eq, @run_inc, @run_done
@run_inc
  %r1 =w add %r, 1
  jmp @run_cond
@run_done
  %c_r3 =w csgew %r, 3
  jnz %c_r3, @repeat, @lit_init
@repeat
  %need_rep_w =w add %o, 2
  %need_rep =l extuw %need_rep_w
  %c_cap_rep =w csgtl %need_rep, %cap
  jnz %c_cap_rep, @ret_neg, @repeat_write
@repeat_write
  %k257 =w copy 257
  %hdr_rep =w sub %k257, %r
  %o_ext_rep =l extuw %o
  %out_o_addr =l add %out, %o_ext_rep
  storeb %hdr_rep, %out_o_addr
  %o1_w =w add %o, 1
  %o1_ext =l extuw %o1_w
  %out_o1_addr =l add %out, %o1_ext
  storeb %b, %out_o1_addr
  %o_rep_next =w add %o, 2
  %i_rep_next =w add %i, %r
  jmp @outer_cond
@lit_init
  jmp @lit_cond
@lit_cond
  %j =w phi @lit_init %i, @lit_inc %j1
  %j_ext =l extuw %j
  %c_jn =w csltl %j_ext, %n
  jnz %c_jn, @lit_len, @lit_emit
@lit_len
  %diff =w sub %j, %i
  %c_diff128 =w csgew %diff, 128
  jnz %c_diff128, @lit_emit, @lit_run_init
@lit_run_init
  %src_j_addr =l add %src, %j_ext
  %bj =w loadub %src_j_addr
  %lr0 =w copy 1
  jmp @lit_run_cond
@lit_run_cond
  %lr =w phi @lit_run_init %lr0, @lit_run_inc %lr1
  %c_lr128 =w csgew %lr, 128
  jnz %c_lr128, @lit_run_done, @lit_run_bounds
@lit_run_bounds
  %jr_w =w add %j, %lr
  %jr_ext =l extuw %jr_w
  %c_jrn =w csltl %jr_ext, %n
  jnz %c_jrn, @lit_run_byte, @lit_run_done
@lit_run_byte
  %src_jr_addr =l add %src, %jr_ext
  %btest =w loadub %src_jr_addr
  %c_jeq =w ceqw %btest, %bj
  jnz %c_jeq, @lit_run_inc, @lit_run_done
@lit_run_inc
  %lr1 =w add %lr, 1
  jmp @lit_run_cond
@lit_run_done
  %c_lr3 =w csgew %lr, 3
  jnz %c_lr3, @lit_emit, @lit_inc
@lit_inc
  %j1 =w add %j, 1
  jmp @lit_cond
@lit_emit
  %litlen =w sub %j, %i
  %tmp1 =w add %o, 1
  %need_lit_w =w add %tmp1, %litlen
  %need_lit =l extuw %need_lit_w
  %c_cap_lit =w csgtl %need_lit, %cap
  jnz %c_cap_lit, @ret_neg, @lit_write
@lit_write
  %hdr_lit =w sub %litlen, 1
  %o_ext_lit =l extuw %o
  %out_o_addr2 =l add %out, %o_ext_lit
  storeb %hdr_lit, %out_o_addr2
  %k0 =w copy 0
  jmp @lit_copy_cond
@lit_copy_cond
  %k =w phi @lit_write %k0, @lit_copy_body %k1
  %c_klen =w csltw %k, %litlen
  jnz %c_klen, @lit_copy_body, @lit_copy_done
@lit_copy_body
  %ik_w =w add %i, %k
  %ik_ext =l extuw %ik_w
  %src_ik_addr =l add %src, %ik_ext
  %byte =w loadub %src_ik_addr
  %o1k_w =w add %tmp1, %k
  %o1k_ext =l extuw %o1k_w
  %out_o1k_addr =l add %out, %o1k_ext
  storeb %byte, %out_o1k_addr
  %k1 =w add %k, 1
  jmp @lit_copy_cond
@lit_copy_done
  %o_lit_next =w copy %need_lit_w
  %i_lit_next =w copy %j
  jmp @outer_cond
@ret_neg
  %neg1 =l copy -1
  ret %neg1
@done
  %o_ext_done =l extuw %o
  ret %o_ext_done
}
