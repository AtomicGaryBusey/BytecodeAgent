export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %is_empty =w cslel %n, 0
  jnz %is_empty, @ret, @cmp
@ret
  ret 0
@cmp
  %nbytes =l mul %n, 8
  %diff =l sub %dst, %src
  %need_bwd =w cultl %diff, %nbytes
  jnz %need_bwd, @bwd_entry, @fwd_entry
@fwd_entry
  jmp @fwd_loop
@fwd_loop
  %fi =l phi @fwd_entry 0, @fwd_body %fi_next
  %fwd_cond =w csltl %fi, %n
  jnz %fwd_cond, @fwd_body, @ret
@fwd_body
  %fwd_off =l mul %fi, 8
  %fwd_src =l add %src, %fwd_off
  %fwd_dst =l add %dst, %fwd_off
  %fwd_val =l loadl %fwd_src
  storel %fwd_val, %fwd_dst
  %fi_next =l add %fi, 1
  jmp @fwd_loop
@bwd_entry
  %bi_init =l sub %n, 1
  jmp @bwd_loop
@bwd_loop
  %bi =l phi @bwd_entry %bi_init, @bwd_body %bi_next
  %bwd_cond =w csgel %bi, 0
  jnz %bwd_cond, @bwd_body, @ret
@bwd_body
  %bwd_off =l mul %bi, 8
  %bwd_src =l add %src, %bwd_off
  %bwd_dst =l add %dst, %bwd_off
  %bwd_val =l loadl %bwd_src
  storel %bwd_val, %bwd_dst
  %bi_next =l sub %bi, 1
  jmp @bwd_loop
}
