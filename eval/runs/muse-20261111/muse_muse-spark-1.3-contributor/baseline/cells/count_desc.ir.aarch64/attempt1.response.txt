export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @ret0, @looppre
@ret0
	ret 0
@looppre
	%limit =l sub %n, 1
	%i0 =l copy 0
	%cnt0 =l copy 0
	jmp @loop
@loop
	%i =l phi @looppre %i0, @loop %inext
	%cnt =l phi @looppre %cnt0, @loop %cntnext
	%off0 =l mul %i, 8
	%p0 =l add %a, %off0
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	%inc =l extuw %gt
	%cntnext =l add %cnt, %inc
	%inext =l add %i, 1
	%cont =w csltl %inext, %limit
	jnz %cont, @loop, @done
@done
	ret %cntnext
}
