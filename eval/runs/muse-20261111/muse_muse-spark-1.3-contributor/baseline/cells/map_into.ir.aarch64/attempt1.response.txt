export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i0 =l copy 0
    %c0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @body %inext
    %c =l phi @start %c0, @body %cnext
    %t =w csltl %i, %n
    jnz %t, @body, @end
@body
    %off =l mul %i, 8
    %saddr =l add %src, %off
    %sval =l loadl %saddr
    %rval =l call %f(l %sval)
    %daddr =l add %dst, %off
    storel %rval, %daddr
    %lt =w csltl %rval, %sval
    %ltl =l extuw %lt
    %cnext =l add %c, %ltl
    %inext =l add %i, 1
    jmp @loop
@end
    ret %c
}
