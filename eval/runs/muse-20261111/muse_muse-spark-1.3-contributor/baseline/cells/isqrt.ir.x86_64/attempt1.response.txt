export function l $isqrt(l %n) {
@start
  %zero =l copy 0
  %max =l copy 4294967295
  %one =l copy 1
  %two =l copy 2
  jmp @loop
@loop
  %lo =l phi @start %zero, @then %lo_next_t, @else %lo_next_e
  %hi =l phi @start %max, @then %hi_next_t, @else %hi_next_e
  %ans =l phi @start %zero, @then %ans_next_t, @else %ans_next_e
  %cond =w culel %lo, %hi
  jnz %cond, @body, @done
@body
  %sum =l add %lo, %hi
  %mid =l udiv %sum, %two
  %sq =l mul %mid, %mid
  %cmp =w culel %sq, %n
  jnz %cmp, @then, @else
@then
  %ans_next_t =l copy %mid
  %lo_next_t =l add %mid, %one
  %hi_next_t =l copy %hi
  jmp @loop
@else
  %midm1 =l sub %mid, %one
  %hi_next_e =l copy %midm1
  %lo_next_e =l copy %lo
  %ans_next_e =l copy %ans
  jmp @loop
@done
  ret %ans
}
