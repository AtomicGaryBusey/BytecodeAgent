export function l $mulhi_u64(l %a, l %b) {
@start
  %alo =l and %a, 4294967295
  %ahi =l shr %a, 32
  %blo =l and %b, 4294967295
  %bhi =l shr %b, 32
  %p0 =l mul %alo, %blo
  %p1 =l mul %alo, %bhi
  %p2 =l mul %ahi, %blo
  %p3 =l mul %ahi, %bhi
  %p0hi =l shr %p0, 32
  %p1lo =l and %p1, 4294967295
  %p1hi =l shr %p1, 32
  %p2lo =l and %p2, 4294967295
  %p2hi =l shr %p2, 32
  %p3lo =l and %p3, 4294967295
  %p3hi =l shr %p3, 32
  %s1a =l add %p0hi, %p1lo
  %s1 =l add %s1a, %p2lo
  %c1 =l shr %s1, 32
  %s2a =l add %p1hi, %p2hi
  %s2b =l add %s2a, %p3lo
  %s2 =l add %s2b, %c1
  %c2 =l shr %s2, 32
  %d2 =l and %s2, 4294967295
  %s3 =l add %p3hi, %c2
  %hhi =l shl %s3, 32
  %hi =l or %hhi, %d2
  ret %hi
}
