export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %z =l copy 0
  %one =l copy 1
  %mx0 =l shl %one, 63
  %mn0 =l sub %mx0, 1
  jmp @loop
@loop
  %i =l phi @start %z, @body %i_next
  %sum =l phi @start %z, @body %sum_next
  %cnt =l phi @start %z, @body %cnt_next
  %mn =l phi @start %mn0, @body %mn_next
  %mx =l phi @start %mx0, @body %mx_next
  %cc =w csltl %i, %n
  jnz %cc, @body, @done
@body
  %off =l shl %i, 3
  %pa =l add %a, %off
  %av =l loadl %pa
  %y =l call %f(l %av)
  %sum_next =l add %sum, %y
  %ob =l and %y, 1
  %oc =w cnel %ob, %z
  %oe =l extuw %oc
  %cnt_next =l add %cnt, %oe
  %lt =w csltl %y, %mn
  %ltl =l extuw %lt
  %ltm =l sub %z, %ltl
  %mnd =l sub %y, %mn
  %mnm =l and %mnd, %ltm
  %mn_next =l add %mn, %mnm
  %gt =w csgtl %y, %mx
  %gtl =l extuw %gt
  %gtm =l sub %z, %gtl
  %mxd =l sub %y, %mx
  %mxm =l and %mxd, %gtm
  %mx_next =l add %mx, %mxm
  %i_next =l add %i, 1
  jmp @loop
@done
  storel %mn, %out
  %o1 =l add %out, 8
  storel %mx, %o1
  %o2 =l add %out, 16
  storel %sum, %o2
  ret %cnt
}
