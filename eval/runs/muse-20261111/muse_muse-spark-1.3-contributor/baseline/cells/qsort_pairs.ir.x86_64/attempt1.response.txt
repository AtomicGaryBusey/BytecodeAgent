export function l $qsort_pairs(l %a, l %n) {
@start
%p_i =l alloc8 8
%p_j =l alloc8 8
%c0 =w cslel %n, 1
jnz %c0, @ret, @init
@ret
ret 0
@init
%half =l udiv %n, 2
%poff =l mul %half, 16
%paddr =l add %a, %poff
%pkey =l loadl %paddr
%paddr8 =l add %paddr, 8
%pval =l loadl %paddr8
%zero =l copy 0
storel %zero, %p_i
%nm1 =l sub %n, 1
storel %nm1, %p_j
jmp @outer
@outer
%i0 =l loadl %p_i
%j0 =l loadl %p_j
%c_outer =w cslel %i0, %j0
jnz %c_outer, @i1, @done
@i1
%i1 =l loadl %p_i
%j1 =l loadl %p_j
%c_i1 =w cslel %i1, %j1
jnz %c_i1, @i1cmp, @i2
@i1cmp
%i1off =l mul %i1, 16
%i1addr =l add %a, %i1off
%i1key =l loadl %i1addr
%i1addr8 =l add %i1addr, 8
%i1val =l loadl %i1addr8
%c_klt =w csltl %i1key, %pkey
jnz %c_klt, @i1inc, @i1eq
@i1eq
%c_keq =w ceql %i1key, %pkey
jnz %c_keq, @i1v, @i2
@i1v
%c_vlt =w csltl %i1val, %pval
jnz %c_vlt, @i1inc, @i2
@i1inc
%i_old =l loadl %p_i
%i_new =l add %i_old, 1
storel %i_new, %p_i
jmp @i1
@i2
%i2 =l loadl %p_i
%j2 =l loadl %p_j
%c_i2 =w cslel %i2, %j2
jnz %c_i2, @i2cmp, @swchk
@i2cmp
%j2off =l mul %j2, 16
%j2addr =l add %a, %j2off
%j2key =l loadl %j2addr
%j2addr8 =l add %j2addr, 8
%j2val =l loadl %j2addr8
%c_kgt =w csgtl %j2key, %pkey
jnz %c_kgt, @i2dec, @i2eq
@i2eq
%c_keq2 =w ceql %j2key, %pkey
jnz %c_keq2, @i2v, @swchk
@i2v
%c_vgt =w csgtl %j2val, %pval
jnz %c_vgt, @i2dec, @swchk
@i2dec
%j_old =l loadl %p_j
%j_new =l sub %j_old, 1
storel %j_new, %p_j
jmp @i2
@swchk
%is =l loadl %p_i
%js =l loadl %p_j
%c_s =w cslel %is, %js
jnz %c_s, @swap, @outer
@swap
%isoff =l mul %is, 16
%isaddr =l add %a, %isoff
%jsoff =l mul %js, 16
%jsaddr =l add %a, %jsoff
%ik =l loadl %isaddr
%isaddr8 =l add %isaddr, 8
%iv =l loadl %isaddr8
%jk =l loadl %jsaddr
%jsaddr8 =l add %jsaddr, 8
%jv =l loadl %jsaddr8
storel %jk, %isaddr
storel %jv, %isaddr8
storel %ik, %jsaddr
storel %iv, %jsaddr8
%isnew =l add %is, 1
storel %isnew, %p_i
%jsnew =l sub %js, 1
storel %jsnew, %p_j
jmp @outer
@done
%if =l loadl %p_i
%jf =l loadl %p_j
%leftn =l add %jf, 1
%r1 =l call $qsort_pairs(l %a, l %leftn)
%ifoff =l mul %if, 16
%rightp =l add %a, %ifoff
%rightn =l sub %n, %if
%r2 =l call $qsort_pairs(l %rightp, l %rightn)
ret 0
}
