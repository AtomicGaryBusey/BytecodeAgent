map_into:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
.Lloop:
cmp rbx, r14
jge .Ldone
mov rdi, qword ptr [r13 + rbx*8]
mov qword ptr [rsp], rdi
call r15
mov qword ptr [r12 + rbx*8], rax
cmp rax, qword ptr [rsp]
jge .Lskip
inc rbp
.Lskip:
inc rbx
jmp .Lloop
.Ldone:
mov rax, rbp
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
