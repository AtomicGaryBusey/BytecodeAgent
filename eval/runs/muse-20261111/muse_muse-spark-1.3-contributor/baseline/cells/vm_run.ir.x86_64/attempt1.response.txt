export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
%pcaddr =l alloc8 8
%spaddr =l alloc8 8
%idxaddr =l alloc8 8
%startzero =l copy 0
storel %startzero, %pcaddr
storel %startzero, %spaddr
jmp @loop
@loop
%looppc =l loadl %pcaddr
%loopc =w csgel %looppc, %len
jnz %loopc, @done, @chk0
@chk0
%chk0pc =l loadl %pcaddr
%chk0off =l mul %chk0pc, 8
%chk0addr =l add %code, %chk0off
%chk0op =l loadl %chk0addr
%chk0c =w ceql %chk0op, 0
jnz %chk0c, @do_halt, @chk1
@chk1
%chk1pc =l loadl %pcaddr
%chk1off =l mul %chk1pc, 8
%chk1addr =l add %code, %chk1off
%chk1op =l loadl %chk1addr
%chk1c =w ceql %chk1op, 1
jnz %chk1c, @do_push, @chk2
@chk2
%chk2pc =l loadl %pcaddr
%chk2off =l mul %chk2pc, 8
%chk2addr =l add %code, %chk2off
%chk2op =l loadl %chk2addr
%chk2c =w ceql %chk2op, 2
jnz %chk2c, @do_add, @chk3
@chk3
%chk3pc =l loadl %pcaddr
%chk3off =l mul %chk3pc, 8
%chk3addr =l add %code, %chk3off
%chk3op =l loadl %chk3addr
%chk3c =w ceql %chk3op, 3
jnz %chk3c, @do_sub, @chk4
@chk4
%chk4pc =l loadl %pcaddr
%chk4off =l mul %chk4pc, 8
%chk4addr =l add %code, %chk4off
%chk4op =l loadl %chk4addr
%chk4c =w ceql %chk4op, 4
jnz %chk4c, @do_mul, @chk5
@chk5
%chk5pc =l loadl %pcaddr
%chk5off =l mul %chk5pc, 8
%chk5addr =l add %code, %chk5off
%chk5op =l loadl %chk5addr
%chk5c =w ceql %chk5op, 5
jnz %chk5c, @do_dup, @chk6
@chk6
%chk6pc =l loadl %pcaddr
%chk6off =l mul %chk6pc, 8
%chk6addr =l add %code, %chk6off
%chk6op =l loadl %chk6addr
%chk6c =w ceql %chk6op, 6
jnz %chk6c, @do_swap, @chk7
@chk7
%chk7pc =l loadl %pcaddr
%chk7off =l mul %chk7pc, 8
%chk7addr =l add %code, %chk7off
%chk7op =l loadl %chk7addr
%chk7c =w ceql %chk7op, 7
jnz %chk7c, @do_drop, @chk8
@chk8
%chk8pc =l loadl %pcaddr
%chk8off =l mul %chk8pc, 8
%chk8addr =l add %code, %chk8off
%chk8op =l loadl %chk8addr
%chk8c =w ceql %chk8op, 8
jnz %chk8c, @do_over, @chk9
@chk9
%chk9pc =l loadl %pcaddr
%chk9off =l mul %chk9pc, 8
%chk9addr =l add %code, %chk9off
%chk9op =l loadl %chk9addr
%chk9c =w ceql %chk9op, 9
jnz %chk9c, @do_lt, @chk10
@chk10
%chk10pc =l loadl %pcaddr
%chk10off =l mul %chk10pc, 8
%chk10addr =l add %code, %chk10off
%chk10op =l loadl %chk10addr
%chk10c =w ceql %chk10op, 10
jnz %chk10c, @do_jz, @chk11
@chk11
%chk11pc =l loadl %pcaddr
%chk11off =l mul %chk11pc, 8
%chk11addr =l add %code, %chk11off
%chk11op =l loadl %chk11addr
%chk11c =w ceql %chk11op, 11
jnz %chk11c, @do_jmp, @do_default
@do_halt
jmp @done
@do_push
%pushpc =l loadl %pcaddr
%pushpc1 =l add %pushpc, 1
%pushoff =l mul %pushpc1, 8
%pushopaddr =l add %code, %pushoff
%pushval =l loadl %pushopaddr
%pushsp =l loadl %spaddr
%pushstoff =l mul %pushsp, 8
%pushstaddr =l add %stack, %pushstoff
storel %pushval, %pushstaddr
%pushsp1 =l add %pushsp, 1
storel %pushsp1, %spaddr
%pushpc2 =l add %pushpc, 2
storel %pushpc2, %pcaddr
jmp @loop
@do_add
%addsp =l loadl %spaddr
%addpc =l loadl %pcaddr
%addspm1 =l sub %addsp, 1
%addoffb =l mul %addspm1, 8
%addaddrb =l add %stack, %addoffb
%addb =l loadl %addaddrb
%addspm2 =l sub %addsp, 2
%addoffa =l mul %addspm2, 8
%addaddra =l add %stack, %addoffa
%adda =l loadl %addaddra
%addc =l add %adda, %addb
storel %addc, %addaddra
storel %addspm1, %spaddr
%addpc1 =l add %addpc, 1
storel %addpc1, %pcaddr
jmp @loop
@do_sub
%subsp =l loadl %spaddr
%subpc =l loadl %pcaddr
%subspm1 =l sub %subsp, 1
%suboffb =l mul %subspm1, 8
%subaddrb =l add %stack, %suboffb
%subb =l loadl %subaddrb
%subspm2 =l sub %subsp, 2
%suboffa =l mul %subspm2, 8
%subaddra =l add %stack, %suboffa
%suba =l loadl %subaddra
%subc =l sub %suba, %subb
storel %subc, %subaddra
storel %subspm1, %spaddr
%subpc1 =l add %subpc, 1
storel %subpc1, %pcaddr
jmp @loop
@do_mul
%mulsp =l loadl %spaddr
%mulpc =l loadl %pcaddr
%mulspm1 =l sub %mulsp, 1
%muloffb =l mul %mulspm1, 8
%muladdrb =l add %stack, %muloffb
%mulb =l loadl %muladdrb
%mulspm2 =l sub %mulsp, 2
%muloffa =l mul %mulspm2, 8
%muladdra =l add %stack, %muloffa
%mula =l loadl %muladdra
%mulc =l mul %mula, %mulb
storel %mulc, %muladdra
storel %mulspm1, %spaddr
%mulpc1 =l add %mulpc, 1
storel %mulpc1, %pcaddr
jmp @loop
@do_dup
%dupsp =l loadl %spaddr
%duppc =l loadl %pcaddr
%dupspm1 =l sub %dupsp, 1
%dupofft =l mul %dupspm1, 8
%dupaddrt =l add %stack, %dupofft
%dupt =l loadl %dupaddrt
%dupoffs =l mul %dupsp, 8
%dupaddrs =l add %stack, %dupoffs
storel %dupt, %dupaddrs
%dupsp1 =l add %dupsp, 1
storel %dupsp1, %spaddr
%duppc1 =l add %duppc, 1
storel %duppc1, %pcaddr
jmp @loop
@do_swap
%swsp =l loadl %spaddr
%swpc =l loadl %pcaddr
%swspm1 =l sub %swsp, 1
%swoffb =l mul %swspm1, 8
%swaddrb =l add %stack, %swoffb
%swb =l loadl %swaddrb
%swspm2 =l sub %swsp, 2
%swoffa =l mul %swspm2, 8
%swaddra =l add %stack, %swoffa
%swa =l loadl %swaddra
storel %swb, %swaddra
storel %swa, %swaddrb
%swpc1 =l add %swpc, 1
storel %swpc1, %pcaddr
jmp @loop
@do_drop
%dropsp =l loadl %spaddr
%droppc =l loadl %pcaddr
%dropspm1 =l sub %dropsp, 1
storel %dropspm1, %spaddr
%droppc1 =l add %droppc, 1
storel %droppc1, %pcaddr
jmp @loop
@do_over
%ovsp =l loadl %spaddr
%ovpc =l loadl %pcaddr
%ovspm2 =l sub %ovsp, 2
%ovoffa =l mul %ovspm2, 8
%ovaddra =l add %stack, %ovoffa
%ova =l loadl %ovaddra
%ovoffs =l mul %ovsp, 8
%ovaddrs =l add %stack, %ovoffs
storel %ova, %ovaddrs
%ovsp1 =l add %ovsp, 1
storel %ovsp1, %spaddr
%ovpc1 =l add %ovpc, 1
storel %ovpc1, %pcaddr
jmp @loop
@do_lt
%ltsp =l loadl %spaddr
%ltpc =l loadl %pcaddr
%ltspm1 =l sub %ltsp, 1
%ltoffb =l mul %ltspm1, 8
%ltaddrb =l add %stack, %ltoffb
%ltb =l loadl %ltaddrb
%ltspm2 =l sub %ltsp, 2
%ltoffa =l mul %ltspm2, 8
%ltaddra =l add %stack, %ltoffa
%lta =l loadl %ltaddra
%ltcnd =w csltl %lta, %ltb
%ltc =l extuw %ltcnd
storel %ltc, %ltaddra
storel %ltspm1, %spaddr
%ltpc1 =l add %ltpc, 1
storel %ltpc1, %pcaddr
jmp @loop
@do_jz
%jzsp =l loadl %spaddr
%jzspm1 =l sub %jzsp, 1
%jzoff =l mul %jzspm1, 8
%jzaddr =l add %stack, %jzoff
%jzv =l loadl %jzaddr
storel %jzspm1, %spaddr
%jzc =w ceql %jzv, 0
jnz %jzc, @jz_zero, @jz_nzero
@jz_zero
%jzzpc =l loadl %pcaddr
%jzzpc1 =l add %jzzpc, 1
%jzzoff =l mul %jzzpc1, 8
%jzzaddr =l add %code, %jzzoff
%jzzt =l loadl %jzzaddr
storel %jzzt, %pcaddr
jmp @loop
@jz_nzero
%jznpc =l loadl %pcaddr
%jznpc2 =l add %jznpc, 2
storel %jznpc2, %pcaddr
jmp @loop
@do_jmp
%jmppc =l loadl %pcaddr
%jmppc1 =l add %jmppc, 1
%jmpoff =l mul %jmppc1, 8
%jmpaddr =l add %code, %jmpoff
%jmpt =l loadl %jmpaddr
storel %jmpt, %pcaddr
jmp @loop
@do_default
%defpc =l loadl %pcaddr
%defpc1 =l add %defpc, 1
storel %defpc1, %pcaddr
jmp @loop
@done
%donesp =l loadl %spaddr
storel %donesp, %idxaddr
jmp @zloop
@zloop
%zloopidx =l loadl %idxaddr
%zloopc =w csltl %zloopidx, %cap
jnz %zloopc, @zbody, @zret
@zbody
%zbodyidx =l loadl %idxaddr
%zbodyoff =l mul %zbodyidx, 8
%zbodyaddr =l add %stack, %zbodyoff
%zbodyzero =l copy 0
storel %zbodyzero, %zbodyaddr
%zbodyidx1 =l add %zbodyidx, 1
storel %zbodyidx1, %idxaddr
jmp @zloop
@zret
%zretsp =l loadl %spaddr
%zretc =w ceql %zretsp, 0
jnz %zretc, @ret0, @retn
@ret0
%ret0v =l copy 0
ret %ret0v
@retn
%retnsp =l loadl %spaddr
%retnspm1 =l sub %retnsp, 1
%retnoff =l mul %retnspm1, 8
%retnaddr =l add %stack, %retnoff
%retnv =l loadl %retnaddr
ret %retnv
}
