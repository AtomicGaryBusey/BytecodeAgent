export function l $map_sum(l %a, l %n, l %f) {
@start
  %sumslot =l alloc8 8
  %islot =l alloc8 8
  %zero =l copy 0
  storel %zero, %sumslot
  storel %zero, %islot
  jmp @loop
@loop
  %i =l loadl %islot
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %i2 =l loadl %islot
  %eight =l copy 8
  %off =l mul %i2, %eight
  %addr =l add %a, %off
  %val =l loadl %addr
  %r =l call %f(l %val)
  %sum =l loadl %sumslot
  %nsum =l add %sum, %r
  storel %nsum, %sumslot
  %one =l copy 1
  %ni =l add %i2, %one
  storel %ni, %islot
  jmp @loop
@done
  %res =l loadl %sumslot
  ret %res
}
