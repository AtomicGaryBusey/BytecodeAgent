export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %zero =l copy 0
  %n =l add %na, %nb
  jmp @zero_cond
@zero_cond
  %zi =l phi @start %zero, @zero_body %zi_next
  %zcond =w csltl %zi, %n
  jnz %zcond, @zero_body, @outer_cond
@zero_body
  %zoff =l mul %zi, 8
  %zaddr =l add %r, %zoff
  storel %zero, %zaddr
  %zi_next =l add %zi, 1
  jmp @zero_cond
@outer_cond
  %i =l phi @zero_cond %zero, @outer_post %i_next
  %ocond =w csltl %i, %na
  jnz %ocond, @outer_body, @scan_cond
@outer_body
  %ioff =l mul %i, 8
  %aaddr =l add %a, %ioff
  %ai =l loadl %aaddr
  jmp @inner_cond
@inner_cond
  %j =l phi @outer_body %zero, @inner_body %j_next
  %carry =l phi @outer_body %zero, @inner_body %carry_next
  %icond =w csltl %j, %nb
  jnz %icond, @inner_body, @outer_post
@inner_body
  %joff =l mul %j, 8
  %baddr =l add %b, %joff
  %bj =l loadl %baddr
  %ij =l add %i, %j
  %ijoff =l mul %ij, 8
  %ijaddr =l add %r, %ijoff
  %r_old =l loadl %ijaddr
  %alo_sh =l shl %ai, 32
  %alo =l shr %alo_sh, 32
  %ahi =l shr %ai, 32
  %blo_sh =l shl %bj, 32
  %blo =l shr %blo_sh, 32
  %bhi =l shr %bj, 32
  %p0 =l mul %alo, %blo
  %p1 =l mul %alo, %bhi
  %p2 =l mul %ahi, %blo
  %p3 =l mul %ahi, %bhi
  %mid =l add %p1, %p2
  %cm =w cultl %mid, %p1
  %shifted =l shl %mid, 32
  %plo =l add %p0, %shifted
  %c_lo =w cultl %plo, %p0
  %mid_hi =l shr %mid, 32
  %cm_l =l extuw %cm
  %cm_sh =l shl %cm_l, 32
  %c_lo_l =l extuw %c_lo
  %t_hi1 =l add %p3, %mid_hi
  %t_hi2 =l add %t_hi1, %cm_sh
  %prodh =l add %t_hi2, %c_lo_l
  %lo1 =l add %plo, %r_old
  %c1 =w cultl %lo1, %plo
  %lo2 =l add %lo1, %carry
  %c2 =w cultl %lo2, %lo1
  %c1_l =l extuw %c1
  %c2_l =l extuw %c2
  %carry_tmp =l add %prodh, %c1_l
  %carry_next =l add %carry_tmp, %c2_l
  storel %lo2, %ijaddr
  %j_next =l add %j, 1
  jmp @inner_cond
@outer_post
  %inb =l add %i, %nb
  %inboff =l mul %inb, 8
  %inbaddr =l add %r, %inboff
  storel %carry, %inbaddr
  %i_next =l add %i, 1
  jmp @outer_cond
@scan_cond
  %k =l phi @outer_cond %n, @scan_next %k_dec
  %kpos =w csgtl %k, %zero
  jnz %kpos, @scan_check, @done
@scan_check
  %km1 =l sub %k, 1
  %koff =l mul %km1, 8
  %kaddr =l add %r, %koff
  %kval =l loadl %kaddr
  %iszero =w ceql %kval, %zero
  jnz %iszero, @scan_next, @done
@scan_next
  %k_dec =l sub %k, 1
  jmp @scan_cond
@done
  ret %k
}
