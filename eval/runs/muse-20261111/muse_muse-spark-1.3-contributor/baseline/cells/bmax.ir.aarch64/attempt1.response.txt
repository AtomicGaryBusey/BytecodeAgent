export function l $bmax(l %a, l %n) {
@start
  %m0 =l loadl %a
  %c1 =l copy 1
  jmp @loop
@loop
  %i =l phi @start %c1, @next %i2
  %max =l phi @start %m0, @next %max2
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %t =l mul %i, 8
  %p =l add %a, %t
  %v =l loadl %p
  %g =w csltl %max, %v
  jnz %g, @update, @keep
@update
  jmp @next
@keep
  jmp @next
@next
  %max2 =l phi @update %v, @keep %max
  %i2 =l add %i, 1
  jmp @loop
@done
  ret %max
}
