export function l $fib(l %n) {
@start
  %zero =l copy 0
  %one =l copy 1
  %pi =l alloc8 8
  %pa =l alloc8 8
  %pb =l alloc8 8
  storel %zero, %pi
  storel %zero, %pa
  storel %one, %pb
  jmp @loop
@loop
  %i =l loadl %pi
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %a =l loadl %pa
  %b =l loadl %pb
  %next =l add %a, %b
  storel %b, %pa
  storel %next, %pb
  %i2 =l add %i, %one
  storel %i2, %pi
  jmp @loop
@done
  %res =l loadl %pa
  ret %res
}
