export function l $binary_search(l %a, l %n, l %key) {
@start
  jmp @loop
@loop
  %lo =l phi @start 0, @less %lo_next, @greater %lo_keep
  %hi =l phi @start %n, @less %hi_keep, @greater %hi_next
  %cond =w csltl %lo, %hi
  jnz %cond, @body, @notfound
@body
  %diff =l sub %hi, %lo
  %half =l shr %diff, 1
  %mid =l add %lo, %half
  %off =l mul %mid, 8
  %addr =l add %a, %off
  %val =l loadl %addr
  %eq =w ceql %val, %key
  jnz %eq, @found, @check
@check
  %lt =w csltl %val, %key
  jnz %lt, @less, @greater
@less
  %lo_next =l add %mid, 1
  %hi_keep =l copy %hi
  jmp @loop
@greater
  %hi_next =l copy %mid
  %lo_keep =l copy %lo
  jmp @loop
@found
  ret %mid
@notfound
  %neg =l copy -1
  ret %neg
}
