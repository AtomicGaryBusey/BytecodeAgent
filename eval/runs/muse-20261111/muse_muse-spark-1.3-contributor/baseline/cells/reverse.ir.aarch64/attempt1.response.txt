export function l $reverse(l %a, l %n) {
@start
	%i0 =l copy 0
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i_next
	%j =l phi @start %j0, @body %j_next
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%off_i =l mul %i, 8
	%addr_i =l add %a, %off_i
	%off_j =l mul %j, 8
	%addr_j =l add %a, %off_j
	%vi =l loadl %addr_i
	%vj =l loadl %addr_j
	storel %vj, %addr_i
	storel %vi, %addr_j
	%i_next =l add %i, 1
	%j_next =l sub %j, 1
	jmp @loop
@done
	%r =l copy 0
	ret %r
}
