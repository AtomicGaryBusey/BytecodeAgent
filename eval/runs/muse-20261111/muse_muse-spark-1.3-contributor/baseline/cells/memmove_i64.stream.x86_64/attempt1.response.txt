memmove_i64:
test rdx, rdx
jle .Ldone
cmp rdi, rsi
jb .Lfwd
lea rax, [rsi + rdx*8]
cmp rdi, rax
jae .Lfwd
mov rcx, rdx
.Lbwd_loop:
dec rcx
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
test rcx, rcx
jnz .Lbwd_loop
jmp .Ldone
.Lfwd:
xor ecx, ecx
.Lfwd_loop:
cmp rcx, rdx
jge .Ldone
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
inc rcx
jmp .Lfwd_loop
.Ldone:
xor eax, eax
ret
