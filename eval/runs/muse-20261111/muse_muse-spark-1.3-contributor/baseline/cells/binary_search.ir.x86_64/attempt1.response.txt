export function l $binary_search(l %a, l %n, l %key) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%lo =l phi @start %zero, @less %lo_next, @greater %lo_keep
	%hi =l phi @start %n, @less %hi_keep, @greater %hi_next
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l udiv %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @less, @greater
@less
	%lo_next =l add %mid, 1
	%hi_keep =l copy %hi
	jmp @loop
@greater
	%hi_next =l copy %mid
	%lo_keep =l copy %lo
	jmp @loop
@found
	ret %mid
@notfound
	%m1 =l copy -1
	ret %m1
}
