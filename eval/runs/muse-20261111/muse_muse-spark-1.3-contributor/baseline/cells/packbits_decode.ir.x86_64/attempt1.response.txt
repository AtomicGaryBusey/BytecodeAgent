export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %i_p =l alloc8 8
  %o_p =l alloc8 8
  %k_p =l alloc8 8
  %l0 =l copy 0
  %l1 =l copy 1
  %l2 =l copy 2
  %w_neg1 =w copy -1
  %w_neg128 =w copy -128
  storel %l0, %i_p
  storel %l0, %o_p
  jmp @loop
@loop
  %i_loop =l loadl %i_p
  %o_loop =l loadl %o_p
  %done_c =w cslel %n, %i_loop
  jnz %done_c, @ret_ok, @have_hdr
@ret_ok
  %o_ret =l loadl %o_p
  ret %o_ret
@have_hdr
  %i_hdr =l loadl %i_p
  %ph_hdr =l add %src, %i_hdr
  %h_hdr =w loadsb %ph_hdr
  %is_nop_c =w ceqw %h_hdr, %w_neg128
  jnz %is_nop_c, @nop, @not_nop
@nop
  %i_nop =l loadl %i_p
  %i_nop1 =l add %i_nop, %l1
  storel %i_nop1, %i_p
  jmp @loop
@not_nop
  %is_lit_c =w csgtw %h_hdr, %w_neg1
  jnz %is_lit_c, @lit_hdr, @rep_hdr
@lit_hdr
  %hl_lit =l extsw %h_hdr
  %lit_len =l add %hl_lit, %l1
  %lit_need =l add %hl_lit, %l2
  %i_lithdr =l loadl %i_p
  %lit_end =l add %i_lithdr, %lit_need
  %lit_trunc_c =w csgtl %lit_end, %n
  jnz %lit_trunc_c, @ret_trunc, @lit_check_cap
@ret_trunc
  %neg2 =l copy -2
  ret %neg2
@lit_check_cap
  %o_litcheck =l loadl %o_p
  %lit_out_end =l add %o_litcheck, %lit_len
  %lit_over_c =w csgtl %lit_out_end, %cap
  jnz %lit_over_c, @lit_over_entry, @lit_full_entry
@lit_over_entry
  %o_litover =l loadl %o_p
  %lit_rem =l sub %cap, %o_litover
  %i_litover =l loadl %i_p
  %i_litover1 =l add %i_litover, %l1
  %lit_src_base_over =l add %src, %i_litover1
  %lit_dst_base_over =l add %out, %o_litover
  storel %l0, %k_p
  jmp @lit_over_check
@lit_over_check
  %k_litover_check =l loadl %k_p
  %lit_over_done_c =w csgel %k_litover_check, %lit_rem
  jnz %lit_over_done_c, @lit_over_done, @lit_over_body
@lit_over_body
  %k_litover_body =l loadl %k_p
  %lit_over_src_ptr =l add %lit_src_base_over, %k_litover_body
  %lit_over_byte =w loadub %lit_over_src_ptr
  %lit_over_dst_ptr =l add %lit_dst_base_over, %k_litover_body
  storeb %lit_over_byte, %lit_over_dst_ptr
  %k_litover_next =l add %k_litover_body, %l1
  storel %k_litover_next, %k_p
  jmp @lit_over_check
@lit_over_done
  jmp @ret_cap
@ret_cap
  %neg1 =l copy -1
  ret %neg1
@lit_full_entry
  %o_litfull =l loadl %o_p
  %i_litfull =l loadl %i_p
  %i_litfull1 =l add %i_litfull, %l1
  %lit_src_base_full =l add %src, %i_litfull1
  %lit_dst_base_full =l add %out, %o_litfull
  storel %l0, %k_p
  jmp @lit_full_check
@lit_full_check
  %k_litfull_check =l loadl %k_p
  %lit_full_done_c =w csgel %k_litfull_check, %lit_len
  jnz %lit_full_done_c, @lit_full_done, @lit_full_body
@lit_full_body
  %k_litfull_body =l loadl %k_p
  %lit_full_src_ptr =l add %lit_src_base_full, %k_litfull_body
  %lit_full_byte =w loadub %lit_full_src_ptr
  %lit_full_dst_ptr =l add %lit_dst_base_full, %k_litfull_body
  storeb %lit_full_byte, %lit_full_dst_ptr
  %k_litfull_next =l add %k_litfull_body, %l1
  storel %k_litfull_next, %k_p
  jmp @lit_full_check
@lit_full_done
  %i_litdone =l loadl %i_p
  %o_litdone =l loadl %o_p
  %i_litdone_new =l add %i_litdone, %lit_need
  %o_litdone_new =l add %o_litdone, %lit_len
  storel %i_litdone_new, %i_p
  storel %o_litdone_new, %o_p
  jmp @loop
@rep_hdr
  %i_rephdr =l loadl %i_p
  %rep_end =l add %i_rephdr, %l2
  %rep_trunc_c =w csgtl %rep_end, %n
  jnz %rep_trunc_c, @ret_trunc, @rep_check_cap
@rep_check_cap
  %hl_rep =l extsw %h_hdr
  %rep_len =l sub %l1, %hl_rep
  %i_repcheck =l loadl %i_p
  %i_repcheck1 =l add %i_repcheck, %l1
  %rep_val_ptr =l add %src, %i_repcheck1
  %rep_val =w loadub %rep_val_ptr
  %o_repcheck =l loadl %o_p
  %rep_out_end =l add %o_repcheck, %rep_len
  %rep_over_c =w csgtl %rep_out_end, %cap
  jnz %rep_over_c, @rep_over_entry, @rep_full_entry
@rep_over_entry
  %o_repover =l loadl %o_p
  %rep_rem =l sub %cap, %o_repover
  %rep_dst_base_over =l add %out, %o_repover
  storel %l0, %k_p
  jmp @rep_over_check
@rep_over_check
  %k_repover_check =l loadl %k_p
  %rep_over_done_c =w csgel %k_repover_check, %rep_rem
  jnz %rep_over_done_c, @rep_over_done, @rep_over_body
@rep_over_body
  %k_repover_body =l loadl %k_p
  %rep_over_dst_ptr =l add %rep_dst_base_over, %k_repover_body
  storeb %rep_val, %rep_over_dst_ptr
  %k_repover_next =l add %k_repover_body, %l1
  storel %k_repover_next, %k_p
  jmp @rep_over_check
@rep_over_done
  jmp @ret_cap
@rep_full_entry
  %o_repfull =l loadl %o_p
  %rep_dst_base_full =l add %out, %o_repfull
  storel %l0, %k_p
  jmp @rep_full_check
@rep_full_check
  %k_repfull_check =l loadl %k_p
  %rep_full_done_c =w csgel %k_repfull_check, %rep_len
  jnz %rep_full_done_c, @rep_full_done, @rep_full_body
@rep_full_body
  %k_repfull_body =l loadl %k_p
  %rep_full_dst_ptr =l add %rep_dst_base_full, %k_repfull_body
  storeb %rep_val, %rep_full_dst_ptr
  %k_repfull_next =l add %k_repfull_body, %l1
  storel %k_repfull_next, %k_p
  jmp @rep_full_check
@rep_full_done
  %i_repdone =l loadl %i_p
  %o_repdone =l loadl %o_p
  %i_repdone_new =l add %i_repdone, %l2
  %o_repdone_new =l add %o_repdone, %rep_len
  storel %i_repdone_new, %i_p
  storel %o_repdone_new, %o_p
  jmp @loop
}
