export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %n =l add %na, %nb
  %zero =l copy 0
  jmp @zero_cond
@zero_cond
  %zi =l phi @start 0, @zero_body %zi_next
  %zc =w cultl %zi, %n
  jnz %zc, @zero_body, @outer_cond
@zero_body
  %zoff =l mul %zi, 8
  %zaddr =l add %r, %zoff
  storel %zero, %zaddr
  %zi_next =l add %zi, 1
  jmp @zero_cond
@outer_cond
  %i =l phi @zero_cond 0, @outer_inc %i_next
  %ic =w csltl %i, %na
  jnz %ic, @inner_cond, @scan_init
@inner_cond
  %j =l phi @outer_cond 0, @carry_done %j_next
  %jc =w csltl %j, %nb
  jnz %jc, @inner_body, @outer_inc
@outer_inc
  %i_next =l add %i, 1
  jmp @outer_cond
@inner_body
  %ioff =l mul %i, 8
  %aaddr =l add %a, %ioff
  %aval =l loadl %aaddr
  %joff =l mul %j, 8
  %baddr =l add %b, %joff
  %bval =l loadl %baddr
  %xlo =l and %aval, 4294967295
  %xhi =l shr %aval, 32
  %ylo =l and %bval, 4294967295
  %yhi =l shr %bval, 32
  %p0 =l mul %xlo, %ylo
  %p1 =l mul %xlo, %yhi
  %p2 =l mul %xhi, %ylo
  %p3 =l mul %xhi, %yhi
  %mid =l add %p1, %p2
  %c_mid_w =w cultl %mid, %p1
  %c_mid =l extuw %c_mid_w
  %mid_shl =l shl %mid, 32
  %lo =l add %p0, %mid_shl
  %c_lo_w =w cultl %lo, %p0
  %c_lo =l extuw %c_lo_w
  %mid_shr =l shr %mid, 32
  %c_mid_shl =l shl %c_mid, 32
  %hi_t1 =l add %p3, %mid_shr
  %hi_t2 =l add %hi_t1, %c_mid_shl
  %hi =l add %hi_t2, %c_lo
  %pos =l add %i, %j
  %pos_off =l mul %pos, 8
  %pos_addr =l add %r, %pos_off
  %r_old =l loadl %pos_addr
  %r_new =l add %r_old, %lo
  %c1_w =w cultl %r_new, %r_old
  %carry1 =l extuw %c1_w
  storel %r_new, %pos_addr
  %pos1 =l add %pos, 1
  %pos1_off =l mul %pos1, 8
  %pos1_addr =l add %r, %pos1_off
  %r1_old =l loadl %pos1_addr
  %r1_tmp =l add %r1_old, %carry1
  %c2a_w =w cultl %r1_tmp, %r1_old
  %c2a =l extuw %c2a_w
  %r1_new =l add %r1_tmp, %hi
  %c2b_w =w cultl %r1_new, %r1_tmp
  %c2b =l extuw %c2b_w
  %carry2 =l add %c2a, %c2b
  storel %r1_new, %pos1_addr
  %prop_start =l add %pos, 2
  jmp @prop_cond
@prop_cond
  %k =l phi @inner_body %prop_start, @prop_body %k_next
  %pcarry =l phi @inner_body %carry2, @prop_body %pcarry_next
  %need_w =w cnel %pcarry, 0
  %inrng_w =w cultl %k, %n
  %doprop_w =w and %need_w, %inrng_w
  jnz %doprop_w, @prop_body, @carry_done
@prop_body
  %k_off =l mul %k, 8
  %k_addr =l add %r, %k_off
  %rk_old =l loadl %k_addr
  %rk_new =l add %rk_old, %pcarry
  %cp_w =w cultl %rk_new, %rk_old
  %pcarry_next =l extuw %cp_w
  storel %rk_new, %k_addr
  %k_next =l add %k, 1
  jmp @prop_cond
@carry_done
  %j_next =l add %j, 1
  jmp @inner_cond
@scan_init
  jmp @scan_cond
@scan_cond
  %sk =l phi @scan_init %n, @scan_dec %sk_next
  %skz =w ceql %sk, 0
  jnz %skz, @scan_done, @scan_check
@scan_check
  %skm1 =l sub %sk, 1
  %skm1_off =l mul %skm1, 8
  %skm1_addr =l add %r, %skm1_off
  %sk_val =l loadl %skm1_addr
  %iszw =w ceql %sk_val, 0
  jnz %iszw, @scan_dec, @scan_done
@scan_dec
  %sk_next =l sub %sk, 1
  jmp @scan_cond
@scan_done
  ret %sk
}
