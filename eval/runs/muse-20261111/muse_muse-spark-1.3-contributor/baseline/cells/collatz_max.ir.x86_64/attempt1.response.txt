export function l $collatz_max(l %n) {
@start
	jmp @outer_head
@outer_head
	%k =l phi @start 1, @outer_next %k_next
	%max =l phi @start 0, @outer_next %max_next
	%cond_outer =w cslel %k, %n
	jnz %cond_outer, @outer_body, @outer_done
@outer_body
	jmp @inner_head
@inner_head
	%x =l phi @outer_body %k, @inner_merge %x_next
	%steps =l phi @outer_body 0, @inner_merge %steps_next
	%cond_inner =w cnel %x, 1
	jnz %cond_inner, @inner_body, @inner_done
@inner_body
	%tmp =l and %x, 1
	%isodd =w cnel %tmp, 0
	jnz %isodd, @odd, @even
@even
	%x_even =l div %x, 2
	jmp @inner_merge
@odd
	%t1 =l mul %x, 3
	%x_odd =l add %t1, 1
	jmp @inner_merge
@inner_merge
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner_head
@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @update_max, @keep_max
@update_max
	jmp @outer_next
@keep_max
	jmp @outer_next
@outer_next
	%max_next =l phi @update_max %steps, @keep_max %max
	%k_next =l add %k, 1
	jmp @outer_head
@outer_done
	ret %max
}
