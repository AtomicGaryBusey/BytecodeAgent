export function l $mulhi_u64(l %a, l %b) {
@start
    %alo =l and %a, 4294967295
    %ahi =l shr %a, 32
    %blo =l and %b, 4294967295
    %bhi =l shr %b, 32
    %p0 =l mul %alo, %blo
    %p1 =l mul %alo, %bhi
    %p2 =l mul %ahi, %blo
    %p3 =l mul %ahi, %bhi
    %p0hi =l shr %p0, 32
    %mid =l add %p1, %p2
    %carry =w cultl %mid, %p1
    %carryl =l extuw %carry
    %midlo =l and %mid, 4294967295
    %midhi =l shr %mid, 32
    %s1 =l add %p0hi, %midlo
    %c1 =l shr %s1, 32
    %t1 =l add %p3, %midhi
    %t2 =l add %t1, %c1
    %cs =l shl %carryl, 32
    %hi =l add %t2, %cs
    ret %hi
}
