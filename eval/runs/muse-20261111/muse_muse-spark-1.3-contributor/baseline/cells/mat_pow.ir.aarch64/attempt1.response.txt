export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %nn =l mul %n, %n
  %nn8 =l mul %nn, 8
  %tmp1 =l add %tmp, %nn8
  %nn2 =l mul %nn, 2
  jmp @ident_header
@ident_header
  %p =l phi @start 0, @ident_body %p_next
  %c_ident =w csltl %p, %nn
  jnz %c_ident, @ident_body, @ident_done
@ident_body
  %idi =l udiv %p, %n
  %idj =l urem %p, %n
  %ideq =w ceql %idi, %idj
  %idval =l extuw %ideq
  %idoff =l mul %p, 8
  %idaddr =l add %out, %idoff
  storel %idval, %idaddr
  %p_next =l add %p, 1
  jmp @ident_header
@ident_done
  %k_is0 =w ceql %k, 0
  jnz %k_is0, @z2_header, @z1_header
@z2_header
  %q2 =l phi @ident_done 0, @z2_body %q2_next
  %c2 =w csltl %q2, %nn2
  jnz %c2, @z2_body, @ret0
@z2_body
  %off2 =l mul %q2, 8
  %addr2 =l add %tmp, %off2
  storel 0, %addr2
  %q2_next =l add %q2, 1
  jmp @z2_header
@z1_header
  %q1 =l phi @ident_done 0, @z1_body %q1_next
  %c1 =w csltl %q1, %nn
  jnz %c1, @z1_body, @iter_header
@z1_body
  %off1 =l mul %q1, 8
  %addr1 =l add %tmp, %off1
  storel 0, %addr1
  %q1_next =l add %q1, 1
  jmp @z1_header
@iter_header
  %it =l phi @z1_header 0, @copy_done %it_next
  %cit =w csltl %it, %k
  jnz %cit, @i_header, @ret1
@i_header
  %i =l phi @iter_header 0, @j_done %i_next
  %ci =w csltl %i, %n
  jnz %ci, @j_header, @copy_header
@j_header
  %j =l phi @i_header 0, @t_done %j_next
  %cj =w csltl %j, %n
  jnz %cj, @t_header, @j_done
@t_header
  %t =l phi @j_header 0, @t_body %t_next
  %sum =l phi @j_header 0, @t_body %sum_next
  %ct =w csltl %t, %n
  jnz %ct, @t_body, @t_done
@t_body
  %t_in =l mul %i, %n
  %t_xidx =l add %t_in, %t
  %t_xoff =l mul %t_xidx, 8
  %t_xaddr =l add %out, %t_xoff
  %t_x =l loadl %t_xaddr
  %t_tn =l mul %t, %n
  %t_yidx =l add %t_tn, %j
  %t_yoff =l mul %t_yidx, 8
  %t_yaddr =l add %a, %t_yoff
  %t_y =l loadl %t_yaddr
  %t_prod =l mul %t_x, %t_y
  %sum_next =l add %sum, %t_prod
  %t_next =l add %t, 1
  jmp @t_header
@t_done
  %d_in =l mul %i, %n
  %d_cidx =l add %d_in, %j
  %d_coff =l mul %d_cidx, 8
  %d_caddr =l add %tmp1, %d_coff
  storel %sum, %d_caddr
  %j_next =l add %j, 1
  jmp @j_header
@j_done
  %i_next =l add %i, 1
  jmp @i_header
@copy_header
  %cp =l phi @i_header 0, @copy_body %cp_next
  %ccp =w csltl %cp, %nn
  jnz %ccp, @copy_body, @copy_done
@copy_body
  %cp_off =l mul %cp, 8
  %cp_src =l add %tmp1, %cp_off
  %cp_val =l loadl %cp_src
  %cp_dst =l add %out, %cp_off
  storel %cp_val, %cp_dst
  storel 0, %cp_src
  %cp_next =l add %cp, 1
  jmp @copy_header
@copy_done
  %it_next =l add %it, 1
  jmp @iter_header
@ret0
  ret 0
@ret1
  ret 0
}
