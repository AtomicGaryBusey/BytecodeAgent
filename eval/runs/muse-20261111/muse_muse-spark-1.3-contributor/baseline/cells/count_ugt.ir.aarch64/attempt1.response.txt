export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %inext
	%c =l phi @start %c0, @body %cnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%gt =w cugtl %v, %t
	%gte =l extuw %gt
	%cnext =l add %c, %gte
	%inext =l add %i, 1
	jmp @loop
@end
	ret %c
}
