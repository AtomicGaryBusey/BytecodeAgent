collatz_max:
xor eax, eax
mov rsi, 1
cmp rdi, 1
jl .Ldone
.Louter:
mov rdx, rsi
xor ecx, ecx
.Linner:
cmp rdx, 1
je .Lcheck
test rdx, 1
jz .Leven
lea rdx, [rdx + rdx*2 + 1]
inc rcx
jmp .Linner
.Leven:
shr rdx, 1
inc rcx
jmp .Linner
.Lcheck:
cmp rcx, rax
cmova rax, rcx
inc rsi
cmp rsi, rdi
jle .Louter
.Ldone:
ret
