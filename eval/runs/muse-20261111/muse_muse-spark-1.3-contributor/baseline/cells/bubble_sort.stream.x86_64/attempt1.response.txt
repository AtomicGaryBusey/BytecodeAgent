bubble_sort:
cmp rsi, 2
jl .Ldone
mov rcx, rsi
dec rcx
.Louter:
xor rdx, rdx
.Linner:
cmp rdx, rcx
jge .Lnext
mov rax, qword ptr [rdi + rdx*8]
mov r8, qword ptr [rdi + rdx*8 + 8]
cmp rax, r8
jle .Lnoswap
mov qword ptr [rdi + rdx*8], r8
mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
inc rdx
jmp .Linner
.Lnext:
dec rcx
jnz .Louter
.Ldone:
mov rax, 0
ret
