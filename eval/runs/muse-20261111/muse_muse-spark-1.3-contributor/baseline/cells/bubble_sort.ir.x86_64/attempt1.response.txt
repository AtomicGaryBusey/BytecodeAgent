export function l $bubble_sort(l %a, l %n) {
@start
  %n1 =l sub %n, 1
  %c0 =w cslel %n, 1
  jnz %c0, @done, @outer
@done
  %r =l copy 0
  ret %r
@outer
  %i =l phi @start 0, @outer_inc %i_next
  %c_outer =w csltl %i, %n1
  jnz %c_outer, @inner_entry, @done
@inner_entry
  jmp @inner
@inner
  %j =l phi @inner_entry 0, @inner_next %j_next
  %bound =l sub %n1, %i
  %c_inner =w csltl %j, %bound
  jnz %c_inner, @body, @outer_inc
@body
  %j8 =l mul %j, 8
  %addr_j =l add %a, %j8
  %v_j =l loadl %addr_j
  %j1 =l add %j, 1
  %j1_8 =l mul %j1, 8
  %addr_j1 =l add %a, %j1_8
  %v_j1 =l loadl %addr_j1
  %c_swap =w csgtl %v_j, %v_j1
  jnz %c_swap, @swap, @noswap
@swap
  storel %v_j1, %addr_j
  storel %v_j, %addr_j1
  jmp @inner_next
@noswap
  jmp @inner_next
@inner_next
  %j_next =l add %j, 1
  jmp @inner
@outer_inc
  %i_next =l add %i, 1
  jmp @outer
}
