export function l $count_ugt(l %a, l %n, l %t) {
@start
  %zero =l copy 0
  jmp @loop
@loop
  %i =l phi @start %zero, @body %inext
  %cnt =l phi @start %zero, @body %cntnext
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %gt =w cugtl %val, %t
  %gtl =l extuw %gt
  %cntnext =l add %cnt, %gtl
  %inext =l add %i, 1
  jmp @loop
@done
  ret %cnt
}
