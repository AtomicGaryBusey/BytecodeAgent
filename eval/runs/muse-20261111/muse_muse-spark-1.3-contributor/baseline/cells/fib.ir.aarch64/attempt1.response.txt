export function l $fib(l %n) {
@start
	%is0 =w ceql %n, 0
	jnz %is0, @ret0, @loop
@ret0
	ret 0
@loop
	%a =l phi @start 0, @body %a_next
	%b =l phi @start 1, @body %b_next
	%i =l phi @start 1, @body %i_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%c =l add %a, %b
	%a_next =l copy %b
	%b_next =l copy %c
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %b
}
