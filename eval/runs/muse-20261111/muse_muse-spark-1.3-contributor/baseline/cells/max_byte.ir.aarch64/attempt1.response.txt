export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @upd %inext_u, @keep %inext_k
	%max =l phi @start 0, @upd %max_u, @keep %max_k
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @upd, @keep
@upd
	%max_u =l copy %b
	%inext_u =l add %i, 1
	jmp @loop
@keep
	%max_k =l copy %max
	%inext_k =l add %i, 1
	jmp @loop
@done
	ret %max
}
