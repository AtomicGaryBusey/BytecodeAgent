export function l $collatz_max(l %n) {
@start
	jmp @outer_cond
@outer_cond
	%i =l phi @start 1, @outer_next %i_next
	%max =l phi @start 0, @outer_next %max_next
	%oc =w cslel %i, %n
	jnz %oc, @outer_body, @outer_done
@outer_body
	jmp @inner_cond
@inner_cond
	%x =l phi @outer_body %i, @merge %x_next
	%cnt =l phi @outer_body 0, @merge %cnt_next
	%done =w ceql %x, 1
	jnz %done, @inner_done, @inner_step
@inner_step
	%oddbit =l and %x, 1
	%isodd =w cnel %oddbit, 0
	jnz %isodd, @odd, @even
@odd
	%t1 =l mul %x, 3
	%t1p =l add %t1, 1
	jmp @merge
@even
	%t2 =l div %x, 2
	jmp @merge
@merge
	%x_next =l phi @odd %t1p, @even %t2
	%cnt_next =l add %cnt, 1
	jmp @inner_cond
@inner_done
	%greater =w csgtl %cnt, %max
	jnz %greater, @take_cnt, @keep_max
@take_cnt
	jmp @outer_next
@keep_max
	jmp @outer_next
@outer_next
	%max_next =l phi @take_cnt %cnt, @keep_max %max
	%i_next =l add %i, 1
	jmp @outer_cond
@outer_done
	ret %max
}
