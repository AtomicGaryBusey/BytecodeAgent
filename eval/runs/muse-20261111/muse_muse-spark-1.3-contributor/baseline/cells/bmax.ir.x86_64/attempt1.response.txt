export function l $bmax(l %a, l %n) {
@start
  %p0 =l loadl %a
  %i0 =l copy 1
  jmp @loop
@loop
  %i =l phi @start %i0, @cont %i2
  %m =l phi @start %p0, @cont %m1
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %off =l mul %i, 8
  %addr =l add %a, %off
  %v =l loadl %addr
  %gt =w csgtl %v, %m
  jnz %gt, @upd, @keep
@upd
  %m_upd =l copy %v
  jmp @cont
@keep
  %m_keep =l copy %m
  jmp @cont
@cont
  %m1 =l phi @upd %m_upd, @keep %m_keep
  %i2 =l add %i, 1
  jmp @loop
@done
  ret %m
}
