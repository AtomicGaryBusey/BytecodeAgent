export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%small =w cslel %n, 1
	jnz %small, @done, @outer_entry
@outer_entry
	jmp @outer_head
@outer_head
	%i =l phi @outer_entry 0, @outer_latch %inext
	%lim =l sub %nm1, %i
	%ocond =w csltl %i, %nm1
	jnz %ocond, @inner_entry, @done
@inner_entry
	jmp @inner_head
@inner_head
	%j =l phi @inner_entry 0, @inner_latch %jnext
	%icond =w csltl %j, %lim
	jnz %icond, @inner_body, @outer_latch
@inner_body
	%off =l mul %j, 8
	%pj =l add %a, %off
	%vj =l loadl %pj
	%pj1 =l add %pj, 8
	%vj1 =l loadl %pj1
	%swap =w csgtl %vj, %vj1
	jnz %swap, @swap, @noswap
@swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @inner_latch
@noswap
	jmp @inner_latch
@inner_latch
	%jnext =l add %j, 1
	jmp @inner_head
@outer_latch
	%inext =l add %i, 1
	jmp @outer_head
@done
	ret 0
}
