export function l $crc32(l %p, l %n) {
@start
  %crc0 =w copy -1
  %i0 =l copy 0
  jmp @cond
@cond
  %i =l phi @start %i0, @next %i_next
  %crc_in =w phi @start %crc0, @next %crc_next
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %addr =l add %p, %i
  %b =w loadub %addr
  %crc_xor =w xor %crc_in, %b
  %bits0 =w copy 8
  jmp @bitcond
@bitcond
  %crc_b =w phi @body %crc_xor, @odd %crc_odd, @even %crc_even
  %bits =w phi @body %bits0, @odd %bits_odd, @even %bits_even
  %has =w cnew %bits, 0
  jnz %has, @bitbody, @bitdone
@bitbody
  %lsb =w and %crc_b, 1
  jnz %lsb, @odd, @even
@odd
  %shr_o =w shr %crc_b, 1
  %crc_odd =w xor %shr_o, -306674912
  %bits_odd =w sub %bits, 1
  jmp @bitcond
@even
  %crc_even =w shr %crc_b, 1
  %bits_even =w sub %bits, 1
  jmp @bitcond
@bitdone
  jmp @next
@next
  %crc_next =w copy %crc_b
  %i_next =l add %i, 1
  jmp @cond
@done
  %crc_f =w xor %crc_in, -1
  %res =l extuw %crc_f
  ret %res
}
