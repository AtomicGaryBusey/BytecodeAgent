export function l $count_nonzero_prefix(l %a, l %n) {
@start
  %z =l copy 0
  jmp @loop
@loop
  %i =l phi @start %z, @next %inext
  %c1 =w csltl %i, %n
  jnz %c1, @check, @done1
@check
  %off =l mul %i, 8
  %p =l add %a, %off
  %v =l loadl %p
  %c2 =w cnel %v, %z
  jnz %c2, @next, @done2
@next
  %inext =l add %i, 1
  jmp @loop
@done1
  ret %i
@done2
  ret %i
}
