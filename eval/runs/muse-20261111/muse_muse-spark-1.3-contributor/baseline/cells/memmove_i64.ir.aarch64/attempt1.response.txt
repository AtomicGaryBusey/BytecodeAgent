export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @done, @check1
@check1
	%dst_lt_src =w cultl %dst, %src
	jnz %dst_lt_src, @fwd_entry, @check2
@check2
	%nbytes =l mul %n, 8
	%src_end =l add %src, %nbytes
	%no_ov =w culel %src_end, %dst
	jnz %no_ov, @fwd_entry, @bwd_entry
@fwd_entry
	%nbytes_f =l mul %n, 8
	jmp @fwd_loop
@fwd_loop
	%off_f =l phi @fwd_entry 0, @fwd_body %off_f_next
	%cond_f =w cultl %off_f, %nbytes_f
	jnz %cond_f, @fwd_body, @done
@fwd_body
	%src_a_f =l add %src, %off_f
	%dst_a_f =l add %dst, %off_f
	%val_f =l loadl %src_a_f
	storel %val_f, %dst_a_f
	%off_f_next =l add %off_f, 8
	jmp @fwd_loop
@bwd_entry
	%nbytes_b =l mul %n, 8
	jmp @bwd_loop
@bwd_loop
	%off_b =l phi @bwd_entry %nbytes_b, @bwd_body %off_b_next
	%cond_b =w cnel %off_b, 0
	jnz %cond_b, @bwd_body, @done
@bwd_body
	%off_b_next =l sub %off_b, 8
	%src_a_b =l add %src, %off_b_next
	%dst_a_b =l add %dst, %off_b_next
	%val_b =l loadl %src_a_b
	storel %val_b, %dst_a_b
	jmp @bwd_loop
@done
	ret 0
}
