export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bzero =w ceql %b, 0
    jnz %bzero, @bzero, @check
@bzero
    %z1 =l copy 0
    storel %a, %rem
    ret %z1
@check
    %nega =l sub 0, %a
    %eqneg =w ceql %a, %nega
    %nonzero =w cnel %a, 0
    %ismin =w and %eqneg, %nonzero
    %isneg1 =w ceql %b, -1
    %ovf =w and %ismin, %isneg1
    jnz %ovf, @ovf, @div
@ovf
    %z2 =l copy 0
    storel %z2, %rem
    ret %a
@div
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
