export function l $qsort_pairs(l %a, l %n)
{
@start
%zero =l copy 0
%c0 =w csgtl %n, 1
jnz %c0, @setup, @ret
@ret
ret %zero
@setup
%nm1 =l sub %n, 1
%off_p =l mul %nm1, 16
%pp =l add %a, %off_p
%pk =l loadl %pp
%pp1 =l add %pp, 8
%pv =l loadl %pp1
jmp @loop
@loop
%i =l phi @setup %zero, @do_swap %i2, @no_swap %i1
%j =l phi @setup %zero, @do_swap %j2, @no_swap %j1
%lc =w csltl %j, %nm1
jnz %lc, @body, @part
@body
%off_j =l mul %j, 16
%addr_j =l add %a, %off_j
%k =l loadl %addr_j
%addr_j1 =l add %addr_j, 8
%v =l loadl %addr_j1
%lt =w csltl %k, %pk
jnz %lt, @do_swap, @cmp2
@cmp2
%gt =w csgtl %k, %pk
jnz %gt, @no_swap, @cmp3
@cmp3
%gtv =w csgtl %v, %pv
jnz %gtv, @no_swap, @do_swap
@do_swap
%off_i_s =l mul %i, 16
%addr_i_s =l add %a, %off_i_s
%addr_i1_s =l add %addr_i_s, 8
%off_j_s =l mul %j, 16
%addr_j_s =l add %a, %off_j_s
%addr_j1_s =l add %addr_j_s, 8
%ik_s =l loadl %addr_i_s
%iv_s =l loadl %addr_i1_s
%jk_s =l loadl %addr_j_s
%jv_s =l loadl %addr_j1_s
storel %jk_s, %addr_i_s
storel %jv_s, %addr_i1_s
storel %ik_s, %addr_j_s
storel %iv_s, %addr_j1_s
%i2 =l add %i, 1
%j2 =l add %j, 1
jmp @loop
@no_swap
%j1 =l add %j, 1
%i1 =l copy %i
jmp @loop
@part
%off_i_p =l mul %i, 16
%addr_i_p =l add %a, %off_i_p
%addr_i1_p =l add %addr_i_p, 8
%off_p_p =l mul %nm1, 16
%addr_p_p =l add %a, %off_p_p
%addr_p1_p =l add %addr_p_p, 8
%ik_p =l loadl %addr_i_p
%iv_p =l loadl %addr_i1_p
%pk_p =l loadl %addr_p_p
%pv_p =l loadl %addr_p1_p
storel %pk_p, %addr_i_p
storel %pv_p, %addr_i1_p
storel %ik_p, %addr_p_p
storel %iv_p, %addr_p1_p
%left_n =l copy %i
%ip1 =l add %i, 1
%right_n =l sub %nm1, %i
%off_r =l mul %ip1, 16
%right_p =l add %a, %off_r
%lc2 =w csgtl %left_n, 1
jnz %lc2, @call_left, @skip_left
@call_left
%r1 =l call $qsort_pairs(l %a, l %left_n)
jmp @after_left
@skip_left
jmp @after_left
@after_left
%rc2 =w csgtl %right_n, 1
jnz %rc2, @call_right, @skip_right
@call_right
%r2 =l call $qsort_pairs(l %right_p, l %right_n)
jmp @after_right
@skip_right
jmp @after_right
@after_right
ret %zero
}
