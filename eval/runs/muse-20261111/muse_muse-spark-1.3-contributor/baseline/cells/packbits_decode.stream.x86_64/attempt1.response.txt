packbits_decode:
xor r8d, r8d
xor r9d, r9d
.Lloop:
cmp r8, rsi
jge .Ldone
mov r10b, byte ptr [rdi + r8]
movsx r10, r10b
cmp r10, -128
je .Lnop
cmp r10, 0
jge .Lliteral
mov rax, rsi
sub rax, r8
cmp rax, 2
jl .Lsrcerr
mov r11, 1
sub r11, r10
mov r10b, byte ptr [rdi + r8 + 1]
mov rax, rcx
sub rax, r9
cmp rax, r11
jl .Lrepeat_trunc
.Lrepeat_loop:
mov byte ptr [rdx + r9], r10b
add r9, 1
sub r11, 1
jne .Lrepeat_loop
add r8, 2
jmp .Lloop
.Lrepeat_trunc:
test rax, rax
jz .Lcaperr
.Lrepeat_trunc_loop:
mov byte ptr [rdx + r9], r10b
add r9, 1
sub rax, 1
jne .Lrepeat_trunc_loop
jmp .Lcaperr
.Lliteral:
mov r11, r10
add r11, 2
mov rax, rsi
sub rax, r8
cmp rax, r11
jl .Lsrcerr
sub r11, 1
mov rax, rcx
sub rax, r9
cmp rax, r11
jl .Lliteral_trunc
add r8, 1
.Llit_loop:
mov r10b, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], r10b
add r8, 1
add r9, 1
sub r11, 1
jne .Llit_loop
jmp .Lloop
.Lliteral_trunc:
test rax, rax
jz .Lcaperr
add r8, 1
.Llit_trunc_loop:
mov r10b, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], r10b
add r8, 1
add r9, 1
sub rax, 1
jne .Llit_trunc_loop
jmp .Lcaperr
.Lnop:
add r8, 1
jmp .Lloop
.Ldone:
mov rax, r9
ret
.Lsrcerr:
mov rax, -2
ret
.Lcaperr:
mov rax, -1
ret
