dot_i32_simd:
xor eax, eax
test rdx, rdx
jle .Ldone
pxor xmm7, xmm7
mov rcx, rdx
shr rcx, 2
jz .Lscal_prep
.Lvloop:
movdqu xmm0, xmmword ptr [rdi]
movdqu xmm1, xmmword ptr [rsi]
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm7, xmm2
pshufd xmm0, xmm0, 238
pshufd xmm1, xmm1, 238
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm7, xmm2
add rdi, 16
add rsi, 16
dec rcx
jne .Lvloop
movdqa xmm0, xmm7
psrldq xmm0, 8
paddq xmm7, xmm0
movq rax, xmm7
.Lscal_prep:
mov rcx, rdx
and ecx, 3
jz .Ldone
.Ltail:
movsxd r8, dword ptr [rdi]
movsxd r9, dword ptr [rsi]
imul r8, r9
add rax, r8
add rdi, 4
add rsi, 4
dec rcx
jne .Ltail
.Ldone:
ret
