export function l $max_byte(l %p, l %n) {
@start
  %z0 =l copy 0
  %is0 =w ceql %n, %z0
  jnz %is0, @ret0, @head
@ret0
  %r0 =l copy 0
  ret %r0
@head
  %i =l phi @start %z0, @merge %i_next
  %m =l phi @start %z0, @merge %m_next
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %addr =l add %p, %i
  %b =l loadub %addr
  %gt =w cugtl %b, %m
  jnz %gt, @take, @keep
@take
  jmp @merge
@keep
  jmp @merge
@merge
  %m_next =l phi @take %b, @keep %m
  %i_next =l add %i, 1
  jmp @head
@done
  ret %m
}
