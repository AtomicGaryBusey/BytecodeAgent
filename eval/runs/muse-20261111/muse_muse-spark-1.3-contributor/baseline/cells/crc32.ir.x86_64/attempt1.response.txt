export function l $crc32(l %p, l %n) {
@start
  jmp @outer_check
@outer_check
  %i =l phi @start 0, @outer_next %i_next
  %crc =w phi @start -1, @outer_next %crc_after
  %cond =w csltl %i, %n
  jnz %cond, @outer_body, @done
@outer_body
  %addr =l add %p, %i
  %b =w loadub %addr
  %crc1 =w xor %crc, %b
  jmp @inner_check
@inner_check
  %k =w phi @outer_body 0, @inner_join %k_next
  %crc_inner =w phi @outer_body %crc1, @inner_join %crc_joined
  %inner_cond =w csltw %k, 8
  jnz %inner_cond, @inner_body, @outer_next
@inner_body
  %bit =w and %crc_inner, 1
  %isodd =w cnew %bit, 0
  jnz %isodd, @odd, @even
@odd
  %shr1 =w shr %crc_inner, 1
  %crc_odd =w xor %shr1, -306674912
  jmp @inner_join
@even
  %crc_even =w shr %crc_inner, 1
  jmp @inner_join
@inner_join
  %crc_joined =w phi @odd %crc_odd, @even %crc_even
  %k_next =w add %k, 1
  jmp @inner_check
@outer_next
  %i_next =l add %i, 1
  %crc_after =w copy %crc_inner
  jmp @outer_check
@done
  %final =w xor %crc, -1
  %res =l extuw %final
  ret %res
}
