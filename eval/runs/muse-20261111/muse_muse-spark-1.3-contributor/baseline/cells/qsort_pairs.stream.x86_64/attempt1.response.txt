qsort_pairs:
cmp rsi, 1
jle .Lbase
sub rsp, 56
mov qword ptr [rsp], rdi
mov qword ptr [rsp + 8], rsi
mov rax, rsi
shr rax, 1
shl rax, 4
mov rcx, rsi
dec rcx
shl rcx, 4
mov r8, qword ptr [rdi + rax]
mov r9, qword ptr [rdi + rax + 8]
lea r10, [rdi + rax]
lea r11, [rdi + rcx]
mov rdx, qword ptr [r10]
xchg rdx, qword ptr [r11]
mov qword ptr [r10], rdx
mov rdx, qword ptr [r10 + 8]
xchg rdx, qword ptr [r11 + 8]
mov qword ptr [r10 + 8], rdx
xor r10, r10
xor r11, r11
.Lloop:
mov rcx, rsi
dec rcx
cmp r11, rcx
jge .Lpartdone
mov rax, r11
shl rax, 4
mov rdx, qword ptr [rdi + rax]
cmp rdx, r8
jl .Ldoswap
jg .Lnext
mov rdx, qword ptr [rdi + rax + 8]
cmp rdx, r9
jge .Lnext
.Ldoswap:
mov rax, r10
shl rax, 4
add rax, rdi
mov rcx, r11
shl rcx, 4
add rcx, rdi
cmp rax, rcx
je .Linc_i
mov rdx, qword ptr [rax]
xchg rdx, qword ptr [rcx]
mov qword ptr [rax], rdx
mov rdx, qword ptr [rax + 8]
xchg rdx, qword ptr [rcx + 8]
mov qword ptr [rax + 8], rdx
.Linc_i:
inc r10
.Lnext:
inc r11
jmp .Lloop
.Lpartdone:
mov rax, r10
shl rax, 4
add rax, rdi
mov rcx, rsi
dec rcx
shl rcx, 4
add rcx, rdi
cmp rax, rcx
je .Lafter_swap
mov rdx, qword ptr [rax]
xchg rdx, qword ptr [rcx]
mov qword ptr [rax], rdx
mov rdx, qword ptr [rax + 8]
xchg rdx, qword ptr [rcx + 8]
mov qword ptr [rax + 8], rdx
.Lafter_swap:
mov qword ptr [rsp + 16], r10
mov rdi, qword ptr [rsp]
mov rsi, r10
call qsort_pairs
mov rax, qword ptr [rsp + 16]
mov rcx, qword ptr [rsp + 8]
mov rdi, qword ptr [rsp]
add rax, 1
mov rsi, rcx
sub rsi, rax
shl rax, 4
add rdi, rax
call qsort_pairs
add rsp, 56
xor eax, eax
ret
.Lbase:
xor eax, eax
ret
