export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  %zero =l copy 0
  jmp @loop
@loop
  %i =l phi @start %zero, @body %inext
  %cnt =l phi @start %zero, @body %cntnext
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %off =l mul %i, 8
  %saddr =l add %src, %off
  %sval =l loadl %saddr
  %rval =l call %f(l %sval)
  %daddr =l add %dst, %off
  storel %rval, %daddr
  %isless =w csltl %rval, %sval
  %islessl =l extuw %isless
  %cntnext =l add %cnt, %islessl
  %inext =l add %i, 1
  jmp @loop
@done
  ret %cnt
}
