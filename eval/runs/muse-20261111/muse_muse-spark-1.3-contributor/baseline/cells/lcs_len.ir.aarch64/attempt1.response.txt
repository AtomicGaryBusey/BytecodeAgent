export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	jmp @zero_cond
@zero_cond
	%k =l phi @start %zero, @zero_body %k_next
	%zc =w cslel %k, %n
	jnz %zc, @zero_body, @outer_init
@zero_body
	%zp =l mul %k, 8
	%za =l add %row, %zp
	storel %zero, %za
	%k_next =l add %k, 1
	jmp @zero_cond
@outer_init
	jmp @outer_cond
@outer_cond
	%i =l phi @outer_init %zero, @outer_latch %i_next
	%oc =w csltl %i, %m
	jnz %oc, @outer_body, @done
@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body %one, @inner_latch %j_next
	%diag =l phi @outer_body %zero, @inner_latch %diag_next
	%ic =w cslel %j, %n
	jnz %ic, @inner_body, @outer_latch
@inner_body
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%rj_off =l mul %j, 8
	%rj_addr =l add %row, %rj_off
	%up =l loadl %rj_addr
	%eq =w ceqw %xb, %yb
	jnz %eq, @then, @else
@then
	%dp1 =l add %diag, 1
	storel %dp1, %rj_addr
	jmp @inner_latch
@else
	%rjm1_off =l mul %jm1, 8
	%rjm1_addr =l add %row, %rjm1_off
	%left =l loadl %rjm1_addr
	%lt =w csltl %up, %left
	jnz %lt, @max_left, @max_up
@max_left
	storel %left, %rj_addr
	jmp @inner_latch
@max_up
	storel %up, %rj_addr
	jmp @inner_latch
@inner_latch
	%j_next =l add %j, 1
	%diag_next =l copy %up
	jmp @inner_cond
@outer_latch
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	%ro =l mul %n, 8
	%ra =l add %row, %ro
	%res =l loadl %ra
	ret %res
}
