export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %zero =l copy 0
  %one =l copy 1
  %eight =l copy 8
  %n2 =l mul %n, %n
  %tmp2n =l add %n2, %n2
  %n2_8 =l mul %n2, %eight
  %prod_base =l add %tmp, %n2_8
  %k_is_zero =w ceql %k, %zero
  jnz %k_is_zero, @k0_out_head, @copy_head
@k0_out_head
  %k0_idx =l phi @start %zero, @k0_out_latch %k0_idx_next
  %k0_i =l udiv %k0_idx, %n
  %k0_j =l urem %k0_idx, %n
  %k0_eq =w ceql %k0_i, %k0_j
  %k0_v =l extuw %k0_eq
  %k0_off8 =l mul %k0_idx, %eight
  %k0_addr =l add %out, %k0_off8
  storel %k0_v, %k0_addr
  jmp @k0_out_latch
@k0_out_latch
  %k0_idx_next =l add %k0_idx, %one
  %k0_out_cont =w csltl %k0_idx_next, %n2
  jnz %k0_out_cont, @k0_out_head, @k0_tmp_head
@k0_tmp_head
  %k0_tidx =l phi @k0_out_latch %zero, @k0_tmp_latch %k0_tidx_next
  %k0_toff8 =l mul %k0_tidx, %eight
  %k0_taddr =l add %tmp, %k0_toff8
  storel %zero, %k0_taddr
  jmp @k0_tmp_latch
@k0_tmp_latch
  %k0_tidx_next =l add %k0_tidx, %one
  %k0_tmp_cont =w csltl %k0_tidx_next, %tmp2n
  jnz %k0_tmp_cont, @k0_tmp_head, @done
@copy_head
  %cpy_idx =l phi @start %zero, @copy_latch %cpy_idx_next
  %cpy_off8 =l mul %cpy_idx, %eight
  %cpy_src =l add %a, %cpy_off8
  %cpy_val =l loadl %cpy_src
  %cpy_dst1 =l add %tmp, %cpy_off8
  storel %cpy_val, %cpy_dst1
  %cpy_dst2 =l add %out, %cpy_off8
  storel %cpy_val, %cpy_dst2
  jmp @copy_latch
@copy_latch
  %cpy_idx_next =l add %cpy_idx, %one
  %cpy_cont =w csltl %cpy_idx_next, %n2
  jnz %cpy_cont, @copy_head, @iter_check
@iter_check
  %iter =l phi @copy_latch %one, @iter_latch %iter_next
  %iter_cond =w csltl %iter, %k
  jnz %iter_cond, @prod_head, @final_zero_head
@prod_head
  %pidx =l phi @iter_check %zero, @prod_latch %pidx_next
  %pi =l udiv %pidx, %n
  %pj =l urem %pidx, %n
  jmp @inner_head
@inner_head
  %t =l phi @prod_head %zero, @inner_latch %t_next
  %sum =l phi @prod_head %zero, @inner_latch %sum_next
  %out_row =l mul %pi, %n
  %out_idx =l add %out_row, %t
  %out_off8 =l mul %out_idx, %eight
  %out_addr =l add %out, %out_off8
  %out_val =l loadl %out_addr
  %base_row =l mul %t, %n
  %base_idx =l add %base_row, %pj
  %base_off8 =l mul %base_idx, %eight
  %base_addr =l add %tmp, %base_off8
  %base_val =l loadl %base_addr
  %prod =l mul %out_val, %base_val
  jmp @inner_latch
@inner_latch
  %sum_next =l add %sum, %prod
  %t_next =l add %t, %one
  %t_cont =w csltl %t_next, %n
  jnz %t_cont, @inner_head, @prod_store
@prod_store
  %p_off8 =l mul %pidx, %eight
  %prod_addr =l add %prod_base, %p_off8
  storel %sum_next, %prod_addr
  jmp @prod_latch
@prod_latch
  %pidx_next =l add %pidx, %one
  %p_cont =w csltl %pidx_next, %n2
  jnz %p_cont, @prod_head, @copyback_head
@copyback_head
  %cb_idx =l phi @prod_latch %zero, @copyback_latch %cb_idx_next
  %cb_off8 =l mul %cb_idx, %eight
  %cb_src =l add %prod_base, %cb_off8
  %cb_val =l loadl %cb_src
  %cb_dst =l add %out, %cb_off8
  storel %cb_val, %cb_dst
  jmp @copyback_latch
@copyback_latch
  %cb_idx_next =l add %cb_idx, %one
  %cb_cont =w csltl %cb_idx_next, %n2
  jnz %cb_cont, @copyback_head, @iter_latch
@iter_latch
  %iter_next =l add %iter, %one
  jmp @iter_check
@final_zero_head
  %fz_idx =l phi @iter_check %zero, @final_zero_latch %fz_idx_next
  %fz_off8 =l mul %fz_idx, %eight
  %fz_addr =l add %tmp, %fz_off8
  storel %zero, %fz_addr
  jmp @final_zero_latch
@final_zero_latch
  %fz_idx_next =l add %fz_idx, %one
  %fz_cont =w csltl %fz_idx_next, %tmp2n
  jnz %fz_cont, @final_zero_head, @done
@done
  ret %zero
}
