export function l $reverse(l %a, l %n) {
@start
  %zero =l copy 0
  %half =l shr %n, 1
  jmp @loop
@loop
  %i =l phi @start %zero, @body %inext
  %cond =w csltl %i, %half
  jnz %cond, @body, @done
@body
  %off_i =l mul %i, 8
  %p_i =l add %a, %off_i
  %v_i =l loadl %p_i
  %nm1 =l sub %n, 1
  %j =l sub %nm1, %i
  %off_j =l mul %j, 8
  %p_j =l add %a, %off_j
  %v_j =l loadl %p_j
  storel %v_j, %p_i
  storel %v_i, %p_j
  %inext =l add %i, 1
  jmp @loop
@done
  ret 0
}
