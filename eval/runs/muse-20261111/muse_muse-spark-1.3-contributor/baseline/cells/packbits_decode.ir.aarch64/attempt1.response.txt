export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
%si0 =l copy 0
%di0 =l copy 0
jmp @loop
@loop
%si =l phi @start %si0, @nop %si_nop, @lit_done %si_lit, @rep_done %si_rep
%di =l phi @start %di0, @nop %di_nop, @lit_done %di_lit, @rep_done %di_rep
%has =w csltl %si, %n
jnz %has, @hdr, @done
@done
ret %di
@hdr
%hp =l add %src, %si
%h =w loadsb %hp
%isnop =w ceqw %h, -128
jnz %isnop, @nop, @chkpos
@nop
%si_nop =l add %si, 1
%di_nop =l copy %di
jmp @loop
@chkpos
%ispos =w csgew %h, 0
jnz %ispos, @lit_chk, @rep_chk
@lit_chk
%hl =l extsw %h
%need =l add %hl, 2
%rem =l sub %n, %si
%enough =w csgel %rem, %need
jnz %enough, @lit_setup, @src_err
@src_err
%r2 =l copy -2
ret %r2
@lit_setup
%llen =l add %hl, 1
%li0 =l copy 0
%ld0 =l copy %di
jmp @lit_loop
@lit_loop
%li =l phi @lit_setup %li0, @lit_body %li_next
%ld =l phi @lit_setup %ld0, @lit_body %ld_next
%more =w csltl %li, %llen
jnz %more, @lit_capchk, @lit_done
@lit_capchk
%hascap =w csltl %ld, %cap
jnz %hascap, @lit_body, @cap_err
@lit_body
%s_tmp =l add %si, 1
%s_off =l add %s_tmp, %li
%saddr =l add %src, %s_off
%b =w loadub %saddr
%daddr =l add %out, %ld
storeb %b, %daddr
%li_next =l add %li, 1
%ld_next =l add %ld, 1
jmp @lit_loop
@lit_done
%si_lit =l add %si, %need
%di_lit =l copy %ld
jmp @loop
@cap_err
%r1 =l copy -1
ret %r1
@rep_chk
%rem2 =l sub %n, %si
%enough2 =w csgel %rem2, 2
jnz %enough2, @rep_setup, @src_err
@rep_setup
%hl_r =l extsw %h
%one =l copy 1
%cnt =l sub %one, %hl_r
%s1 =l add %si, 1
%vaddr =l add %src, %s1
%v =w loadub %vaddr
%ri0 =l copy 0
%rd0 =l copy %di
jmp @rep_loop
@rep_loop
%ri =l phi @rep_setup %ri0, @rep_body %ri_next
%rd =l phi @rep_setup %rd0, @rep_body %rd_next
%rmore =w csltl %ri, %cnt
jnz %rmore, @rep_capchk, @rep_done
@rep_capchk
%rhascap =w csltl %rd, %cap
jnz %rhascap, @rep_body, @cap_err
@rep_body
%daddr2 =l add %out, %rd
storeb %v, %daddr2
%ri_next =l add %ri, 1
%rd_next =l add %rd, 1
jmp @rep_loop
@rep_done
%si_rep =l add %si, 2
%di_rep =l copy %rd
jmp @loop
}
