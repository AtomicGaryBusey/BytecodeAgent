push r13

push r15

mov r13, rsi

mov r14, rcx
mov r15, r8

imul rax, r13
mov rbp, rax
xor r9d, r9d
.Lid_i:
xor r10d, r10d
.Lid_j:
xor eax, eax
cmp r9, r10
sete al
mov r11, r9
imul r11, r13
add r11, r10
mov qword ptr [r14 + r11*8], rax
inc r10
cmp r10, r13
jl .Lid_j
inc r9
cmp r9, r13
jl .Lid_i
test rbx, rbx
jz .Lzero
.Lk:
xor r9d, r9d
.Lmul_i:
xor r10d, r10d
.Lmul_j:
xor r11d, r11d
xor r8d, r8d
.Lmul_t:
mov rax, r9
imul rax, r13
add rax, r8
mov rax, qword ptr [r14 + rax*8]
mov rcx, r8
imul rcx, r13
add rcx, r10
mov rcx, qword ptr [r12 + rcx*8]

add r11, rax

mov rax, r9

jl .Lmul_j
inc r9
cmp r9, r13
jl .Lmul_i
xor eax, eax
.Lcopy:

jge .Lcopy_done
mov rdx, qword ptr [r15 + rax*8]
mov qword ptr [r14 + rax*8], rdx
inc rax
jmp .Lcopy
.Lcopy_done:
dec rbx
jnz .Lk
.Lzero:

add rdx, rbp
xor eax, eax
.Lzero_loop:

jge .Ldone
mov qword ptr [r15 + rax*8], 0
inc rax
jmp .Lzero_loop
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
