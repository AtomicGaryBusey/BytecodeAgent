export function l $isqrt(l %n) {
@start
  %cmax =l copy 4294967295
  %cone =l copy 1
  %ctwo =l copy 2
  %is_small_n =w cultl %n, %ctwo
  jnz %is_small_n, @small, @setup
@small
  ret %n
@setup
  %is_small_hi =w cultl %n, %cmax
  jnz %is_small_hi, @his, @hib
@his
  %hi_his =l copy %n
  jmp @enter
@hib
  %hi_hib =l copy %cmax
  jmp @enter
@enter
  %hi0 =l phi @his %hi_his, @hib %hi_hib
  %lo_enter =l copy %cone
  %hi_enter =l copy %hi0
  %ans_enter =l copy %cone
  jmp @loop
@loop
  %lo =l phi @enter %lo_enter, @yes %lo_next, @no %lo_fwd
  %hi =l phi @enter %hi_enter, @yes %hi_fwd, @no %hi_next
  %ans =l phi @enter %ans_enter, @yes %ans_next, @no %ans_fwd
  %cond =w culel %lo, %hi
  jnz %cond, @body, @done
@body
  %diff =l sub %hi, %lo
  %half =l udiv %diff, %ctwo
  %mid =l add %lo, %half
  %q =l udiv %n, %mid
  %le =w culel %mid, %q
  jnz %le, @yes, @no
@yes
  %lo_next =l add %mid, %cone
  %hi_fwd =l copy %hi
  %ans_next =l copy %mid
  jmp @loop
@no
  %hi_next =l sub %mid, %cone
  %lo_fwd =l copy %lo
  %ans_fwd =l copy %ans
  jmp @loop
@done
  ret %ans
}
export function l $isqrt(l %n0) {
@start
	%res_init =l copy 0
	%bit_init =l copy 4611686018427387904
	jmp @test
@test
	%n =l phi @start %n0, @cont %n_next
	%res =l phi @start %res_init, @cont %res_next
	%bit =l phi @start %bit_init, @cont %bit_next
	%cond =w cnel %bit, 0
	jnz %cond, @body, @done
@body
	%sum =l add %res, %bit
	%c =w culel %sum, %n
	jnz %c, @then, @else
@then
	%n_sub =l sub %n, %sum
	%res_shr1 =l shr %res, 1
	%res_then =l add %res_shr1, %bit
	%bit_then =l shr %bit, 2
	jmp @cont
@else
	%n_else =l copy %n
	%res_else =l shr %res, 1
	%bit_else =l shr %bit, 2
	jmp @cont
@cont
	%n_next =l phi @then %n_sub, @else %n_else
	%res_next =l phi @then %res_then, @else %res_else
	%bit_next =l phi @then %bit_then, @else %bit_else
	jmp @test
@done
	ret %res
}
