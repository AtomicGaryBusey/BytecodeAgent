export function l $reverse(l %a, l %n) {
@start
	%p =l copy %a
	%t =l mul %n, 8
	%e =l add %a, %t
	%q =l sub %e, 8
	jmp @loop
@loop
	%c =w csltl %p, %q
	jnz %c, @body, @done
@body
	%x =l loadl %p
	%y =l loadl %q
	storel %y, %p
	storel %x, %q
	%p =l add %p, 8
	%q =l sub %q, 8
	jmp @loop
@done
	ret 0
}
