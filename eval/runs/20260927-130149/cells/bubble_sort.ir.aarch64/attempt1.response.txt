export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%m =l sub %n, 1
	%oc =w csltl %i, %m
	jnz %oc, @outer_body, @done
@outer_body
	%lim =l sub %m, %i
	%j =l copy 0
@inner
	%ic =w csltl %j, %lim
	jnz %ic, @inner_body, @outer_next
@inner_body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%sw =w csgtl %x, %y
	jnz %sw, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %q
@inner_next
	%j =l add %j, 1
	jmp @inner
@outer_next
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
