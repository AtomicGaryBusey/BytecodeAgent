bubble_sort:
    cmp x1, #2
    b.lt .Ldone
    sub x9, x1, #1          // x9 = number of comparisons in this pass
.Louter:
    mov x10, #0             // j
    mov x11, x0             // pointer
    mov x14, #0             // swapped flag
.Linner:
    ldp x12, x13, [x11]
    cmp x12, x13
    b.le .Lnoswap
    stp x13, x12, [x11]
    mov x14, #1
.Lnoswap:
    add x11, x11, #8
    add x10, x10, #1
    cmp x10, x9
    b.lt .Linner
    cbz x14, .Ldone
    subs x9, x9, #1
    b.gt .Louter
.Ldone:
    mov x0, #0
    ret
