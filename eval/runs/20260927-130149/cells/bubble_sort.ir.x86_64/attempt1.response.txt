export function l $bubble_sort(l %a, l %n) {
@start
	%i =l sub %n, 1
@outer
	%c =w csgtl %i, 0
	jnz %c, @oinit, @done
@oinit
	%j =l copy 0
@inner
	%c2 =w csltl %j, %i
	jnz %c2, @body, @onext
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%s =w csgtl %x, %y
	jnz %s, @swap, @inext
@swap
	storel %y, %p
	storel %x, %q
@inext
	%j =l add %j, 1
	jmp @inner
@onext
	%i =l sub %i, 1
	jmp @outer
@done
	ret 0
}
