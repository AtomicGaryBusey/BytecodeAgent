bubble_sort:
    cmp rsi, 1
    jle .Ldone
    mov r8, rsi
    dec r8
.Louter:
    xor rcx, rcx
    xor r9, r9
.Linner:
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rax, rdx
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rcx*8 + 8], rax
    mov r9, 1
.Lnoswap:
    inc rcx
    cmp rcx, r8
    jl .Linner
    test r9, r9
    jz .Ldone
    dec r8
    jg .Louter
.Ldone:
    xor eax, eax
    ret
