packbits_encode:
push rbx
cld
xor r8d, r8d
xor r9d, r9d
.Lmain:
cmp r8, rsi
jae .Lsuccess
movzx r10d, byte ptr [rdi + r8]
lea r11, [r8 + 1]
mov eax, 1
.Lrun_scan:
cmp r11, rsi
jae .Lrun_done
cmp rax, 128
jae .Lrun_done
cmp r10b, byte ptr [rdi + r11]
jne .Lrun_done
inc r11
inc rax
jmp .Lrun_scan
.Lrun_done:
cmp rax, 3
jb .Lliteral_begin
lea r11, [r9 + 2]
cmp r11, rcx
ja .Lfailure
mov r11d, 1
sub r11d, eax
mov byte ptr [rdx + r9], r11b
mov byte ptr [rdx + r9 + 1], r10b
add r9, 2
add r8, rax
jmp .Lmain
.Lliteral_begin:
mov r10, r8
mov r11, r8
.Lliteral_scan:
cmp r11, rsi
jae .Lliteral_done
mov rax, r11
sub rax, r10
cmp rax, 128
jae .Lliteral_done
lea rax, [r11 + 2]
cmp rax, rsi
jae .Lliteral_include
mov bl, byte ptr [rdi + r11]
cmp bl, byte ptr [rdi + r11 + 1]
jne .Lliteral_include
cmp bl, byte ptr [rdi + r11 + 2]
je .Lliteral_done
.Lliteral_include:
inc r11
jmp .Lliteral_scan
.Lliteral_done:
mov rax, r11
sub rax, r10
lea rbx, [r9 + rax + 1]
cmp rbx, rcx
ja .Lfailure
mov r8, r11
mov r11d, eax
dec r11d
mov byte ptr [rdx + r9], r11b
lea r11, [rdx + r9 + 1]
mov r9, rbx
.Lliteral_copy:
test rax, rax
jz .Lmain
mov bl, byte ptr [rdi + r10]
mov byte ptr [r11], bl
inc r10
inc r11
dec rax
jmp .Lliteral_copy
.Lsuccess:
mov rax, r9
pop rbx
ret
.Lfailure:
mov rax, -1
pop rbx
ret
