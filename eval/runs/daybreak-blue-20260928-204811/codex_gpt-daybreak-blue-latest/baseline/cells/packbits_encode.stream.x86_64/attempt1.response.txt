packbits_encode:
push rbx
cld
xor r8d, r8d
xor r9d, r9d
.Lmain:
cmp r8, rsi
jae .Lsuccess
movzx ebx, byte ptr [rdi + r8]
lea r11, [r8 + 1]
mov eax, 1
.Lrun_loop:
cmp r11, rsi
jae .Lrun_done
cmp rax, 128
jae .Lrun_done
cmp bl, byte ptr [rdi + r11]
jne .Lrun_done
inc r11
inc rax
jmp .Lrun_loop
.Lrun_done:
cmp rax, 3
jb .Lliteral_begin
lea r11, [r9 + 2]
cmp r11, rcx
ja .Lfailure
mov r10d, 257
sub r10d, eax
mov byte ptr [rdx + r9], r10b
mov byte ptr [rdx + r9 + 1], bl
add r8, rax
mov r9, r11
jmp .Lmain
.Lliteral_begin:
mov r10, r8
mov r11, r8
.Lliteral_scan:
cmp r11, rsi
jae .Lliteral_done
mov rax, r11
sub rax, r10
cmp rax, 128
jae .Lliteral_done
lea rax, [r11 + 2]
cmp rax, rsi
jae .Lliteral_include
mov bl, byte ptr [rdi + r11]
cmp bl, byte ptr [rdi + r11 + 1]
jne .Lliteral_include
cmp bl, byte ptr [rdi + r11 + 2]
je .Lliteral_done
.Lliteral_include:
inc r11
jmp .Lliteral_scan
.Lliteral_done:
mov rax, r11
sub rax, r10
lea rbx, [r9 + rax + 1]
cmp rbx, rcx
ja .Lfailure
mov r8, r11
lea r11, [rdx + r9]
mov ecx, eax
dec ecx
mov byte ptr [r11], cl
inc r11
mov r9, rbx
add r10, rdi
xor ecx, ecx
.Lliteral_copy:
cmp rcx, rax
jae .Lmain
mov bl, byte ptr [r10 + rcx]
mov byte ptr [r11 + rcx], bl
inc rcx
jmp .Lliteral_copy
.Lsuccess:
mov rax, r9
pop rbx
ret
.Lfailure:
mov rax, -1
pop rbx
ret
