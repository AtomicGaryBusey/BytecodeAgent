export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @check

@check
	%same =w ceql %dst, %src
	jnz %same, @done, @direction

@direction
	%forward =w cultl %dst, %src
	jnz %forward, @fhead, @bhead

@fhead
	%fi =l phi @direction 0, @fbody %fnext
	%fmore =w csltl %fi, %n
	jnz %fmore, @fbody, @done

@fbody
	%foff =l shl %fi, 3
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fval =l loadl %fsrc
	storel %fval, %fdst
	%fnext =l add %fi, 1
	jmp @fhead

@bhead
	%bi =l phi @direction %n, @bbody %bnext
	%bmore =w cnel %bi, 0
	jnz %bmore, @bbody, @done

@bbody
	%bnext =l sub %bi, 1
	%boff =l shl %bnext, 3
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	jmp @bhead

@done
	ret 0
}
