export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%islot =l alloc8 8
	%jslot =l alloc8 8
	%cslot =l alloc8 8
	%zslot =l alloc8 8
	storel 0, %zslot
	jmp @zero_test

@zero_test
	%zi =l loadl %zslot
	%zmore =w csltl %zi, %n
	jnz %zmore, @zero_body, @outer_init

@zero_body
	%zo =l shl %zi, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%zin =l add %zi, 1
	storel %zin, %zslot
	jmp @zero_test

@outer_init
	storel 0, %islot
	jmp @outer_test

@outer_test
	%i =l loadl %islot
	%imore =w csltl %i, %na
	jnz %imore, @outer_body, @scan_init

@outer_body
	%iao =l shl %i, 3
	%iap =l add %a, %iao
	%av =l loadl %iap
	storel 0, %jslot
	storel 0, %cslot
	jmp @inner_test

@inner_test
	%j =l loadl %jslot
	%jmore =w csltl %j, %nb
	jnz %jmore, @inner_body, @inner_done

@inner_body
	%jbo =l shl %j, 3
	%jbp =l add %b, %jbo
	%bv =l loadl %jbp

	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%p01lo =l shl %p01, 32
	%t0 =l add %p00, %p01lo
	%pc0w =w cultl %t0, %p00
	%pc0 =l extuw %pc0w

	%p10lo =l shl %p10, 32
	%plo =l add %t0, %p10lo
	%pc1w =w cultl %plo, %t0
	%pc1 =l extuw %pc1w

	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%ph0 =l add %p11, %p01hi
	%ph1 =l add %ph0, %p10hi
	%ph2 =l add %ph1, %pc0
	%phi =l add %ph2, %pc1

	%ij =l add %i, %j
	%rjo =l shl %ij, 3
	%rjp =l add %r, %rjo
	%rv =l loadl %rjp
	%carry =l loadl %cslot

	%s0 =l add %plo, %rv
	%sc0w =w cultl %s0, %plo
	%sc0 =l extuw %sc0w
	%s1 =l add %s0, %carry
	%sc1w =w cultl %s1, %s0
	%sc1 =l extuw %sc1w
	storel %s1, %rjp

	%nc0 =l add %phi, %sc0
	%nc =l add %nc0, %sc1
	storel %nc, %cslot

	%jn =l add %j, 1
	storel %jn, %jslot
	jmp @inner_test

@inner_done
	%oi =l loadl %islot
	%endidx =l add %oi, %nb
	%endo =l shl %endidx, 3
	%endp =l add %r, %endo
	%endcarry =l loadl %cslot
	storel %endcarry, %endp
	%in =l add %oi, 1
	storel %in, %islot
	jmp @outer_test

@scan_init
	storel %n, %zslot
	jmp @scan_test

@scan_test
	%k =l loadl %zslot
	%kpos =w csltl 0, %k
	jnz %kpos, @scan_body, @return_zero

@scan_body
	%km1 =l sub %k, 1
	%koff =l shl %km1, 3
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%knz =w cnel %kv, 0
	jnz %knz, @return_k, @scan_dec

@scan_dec
	storel %km1, %zslot
	jmp @scan_test

@return_k
	ret %k

@return_zero
	ret 0
}
