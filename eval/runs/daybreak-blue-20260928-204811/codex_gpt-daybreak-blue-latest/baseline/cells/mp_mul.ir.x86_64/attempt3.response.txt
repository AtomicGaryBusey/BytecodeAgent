export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%z0 =l copy 0
	jmp @ztest

@ztest
	%zi =l phi @start %z0, @zbody %znext
	%zmore =w csltl %zi, %n
	jnz %zmore, @zbody, @zdone

@zbody
	%zoff =l mul %zi, 8
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%znext =l add %zi, 1
	jmp @ztest

@zdone
	%i0 =l copy 0
	jmp @otest

@otest
	%i =l phi @zdone %i0, @iend %inext
	%omore =w csltl %i, %na
	jnz %omore, @obody, @odone

@obody
	%aioff =l mul %i, 8
	%aiptr =l add %a, %aioff
	%av =l loadl %aiptr
	%j0 =l copy 0
	%carry0 =l copy 0
	jmp @itest

@itest
	%j =l phi @obody %j0, @ibody %jnext
	%carry =l phi @obody %carry0, @ibody %carrynext
	%imore =w csltl %j, %nb
	jnz %imore, @ibody, @iend

@ibody
	%bjoff =l mul %j, 8
	%bjptr =l add %b, %bjoff
	%bv =l loadl %bjptr

	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%p00h =l shr %p00, 32
	%p01l =l and %p01, 4294967295
	%p10l =l and %p10, 4294967295
	%m0 =l add %p00h, %p01l
	%mid =l add %m0, %p10l

	%p00l =l and %p00, 4294967295
	%midlo =l shl %mid, 32
	%plo =l or %p00l, %midlo

	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%midh =l shr %mid, 32
	%h0 =l add %p11, %p01h
	%h1 =l add %h0, %p10h
	%phi =l add %h1, %midh

	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rptr =l add %r, %roff
	%old =l loadl %rptr

	%s0 =l add %old, %plo
	%c0w =w cultl %s0, %old
	%s1 =l add %s0, %carry
	%c1w =w cultl %s1, %s0
	storel %s1, %rptr

	%c0 =l extuw %c0w
	%c1 =l extuw %c1w
	%ch0 =l add %phi, %c0
	%carrynext =l add %ch0, %c1
	%jnext =l add %j, 1
	jmp @itest

@iend
	%endidx =l add %i, %nb
	%endoff =l mul %endidx, 8
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%inext =l add %i, 1
	jmp @otest

@odone
	jmp @strim

@strim
	%k =l phi @odone %n, @szero %kprev
	%kpos =w csltl 0, %k
	jnz %kpos, @sload, @sret

@sload
	%kprev =l sub %k, 1
	%koff =l mul %kprev, 8
	%kptr =l add %r, %koff
	%kv =l loadl %kptr
	%kiszero =w ceql %kv, 0
	jnz %kiszero, @szero, @sret

@szero
	jmp @strim

@sret
	ret %k
}
