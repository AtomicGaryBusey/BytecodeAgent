export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%less =w csltl %r, %x
	%inc =l extuw %less
	%countnext =l add %count, %inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
