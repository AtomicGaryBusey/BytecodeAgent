export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%back =w cultl %src, %dst
	jnz %back, @backward, @forward

@forward
	%i =l phi @start 0, @fcopy %inext
	%fmore =w csltl %i, %n
	jnz %fmore, @fcopy, @done

@fcopy
	%foff =l shl %i, 3
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fval =l loadl %fsrc
	storel %fval, %fdst
	%inext =l add %i, 1
	jmp @forward

@backward
	%j =l phi @start %n, @bcopy %jnext
	%bmore =w csgtl %j, 0
	jnz %bmore, @bcopy, @done

@bcopy
	%jnext =l sub %j, 1
	%boff =l shl %jnext, 3
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	jmp @backward

@done
	ret 0
}
