packbits_decode:
    xor eax, eax
.Lnext:
    test rsi, rsi
    jz .Ldone
    movsx r8d, byte ptr [rdi]
    inc rdi
    dec rsi
    test r8d, r8d
    jns .Lliteral
    cmp r8d, -128
    je .Lnext
    mov r9d, 1
    sub r9d, r8d
    test rsi, rsi
    jz .Lsource_error
    mov r8b, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp rcx, r9
    jb .Lrepeat_truncated
    sub rcx, r9
    add rax, r9
    mov r10, r9
.Lrepeat_loop:
    mov byte ptr [rdx], r8b
    inc rdx
    dec r10
    jnz .Lrepeat_loop
    jmp .Lnext
.Lrepeat_truncated:
    mov r10, rcx
    test r10, r10
    jz .Lcapacity_error
.Lrepeat_partial_loop:
    mov byte ptr [rdx], r8b
    inc rdx
    dec r10
    jnz .Lrepeat_partial_loop
    jmp .Lcapacity_error
.Lliteral:
    lea r9d, [r8d + 1]
    cmp rsi, r9
    jb .Lsource_error
    cmp rcx, r9
    jb .Lliteral_truncated
    sub rsi, r9
    sub rcx, r9
    add rax, r9
    mov r10, r9
.Lliteral_loop:
    mov r11b, byte ptr [rdi]
    mov byte ptr [rdx], r11b
    inc rdi
    inc rdx
    dec r10
    jnz .Lliteral_loop
    jmp .Lnext
.Lliteral_truncated:
    mov r10, rcx
    test r10, r10
    jz .Lcapacity_error
.Lliteral_partial_loop:
    mov r11b, byte ptr [rdi]
    mov byte ptr [rdx], r11b
    inc rdi
    inc rdx
    dec r10
    jnz .Lliteral_partial_loop
.Lcapacity_error:
    mov rax, -1
    ret
.Lsource_error:
    mov rax, -2
    ret
.Ldone:
    ret
