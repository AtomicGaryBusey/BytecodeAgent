export function l $bubble_sort(l %a, l %n) {
@start
	%end0 =l sub %n, 1
	jmp @outer
@outer
	%end =l phi @start %end0, @outer_next %end1
	%more =w csltl 0, %end
	jnz %more, @inner_init, @done
@inner_init
	jmp @inner
@inner
	%i =l phi @inner_init 0, @next %inext
	%within =w csltl %i, %end
	jnz %within, @compare, @outer_next
@compare
	%off =l shl %i, 3
	%addr0 =l add %a, %off
	%addr1 =l add %addr0, 8
	%v0 =l loadl %addr0
	%v1 =l loadl %addr1
	%swap =w csltl %v1, %v0
	jnz %swap, @do_swap, @next
@do_swap
	storel %v1, %addr0
	storel %v0, %addr1
	jmp @next
@next
	%inext =l add %i, 1
	jmp @inner
@outer_next
	%end1 =l sub %end, 1
	jmp @outer
@done
	ret 0
}
