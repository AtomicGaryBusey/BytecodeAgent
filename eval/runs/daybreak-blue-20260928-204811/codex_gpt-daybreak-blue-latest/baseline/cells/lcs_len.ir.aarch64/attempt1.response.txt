export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%lim =l add %n, 1
	jmp @init
@init
	%zj =l phi @start 0, @init_store %zjn
	%init_done =w ceql %zj, %lim
	jnz %init_done, @outer_setup, @init_store
@init_store
	%zoff =l shl %zj, 3
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%zjn =l add %zj, 1
	jmp @init
@outer_setup
	jmp @outer
@outer
	%i =l phi @outer_setup 0, @after_inner %inext
	%outer_done =w ceql %i, %m
	jnz %outer_done, @return, @inner_setup
@inner_setup
	%xptr =l add %x, %i
	%xb =w loadub %xptr
	jmp @inner
@inner
	%j =l phi @inner_setup 1, @store %jnext
	%diag =l phi @inner_setup 0, @store %up
	%inner_done =w ceql %j, %lim
	jnz %inner_done, @after_inner, @inner_body
@inner_body
	%joff =l shl %j, 3
	%rptr =l add %row, %joff
	%up =l loadl %rptr
	%jm1 =l sub %j, 1
	%yptr =l add %y, %jm1
	%yb =w loadub %yptr
	%bytes_eq =w ceqw %xb, %yb
	jnz %bytes_eq, @match, @nomatch
@match
	%matchv =l add %diag, 1
	jmp @store
@nomatch
	%loff =l shl %jm1, 3
	%lptr =l add %row, %loff
	%left =l loadl %lptr
	%up_gt =w csgtl %up, %left
	jnz %up_gt, @use_up, @use_left
@use_up
	jmp @store
@use_left
	jmp @store
@store
	%newv =l phi @match %matchv, @use_up %up, @use_left %left
	storel %newv, %rptr
	%jnext =l add %j, 1
	jmp @inner
@after_inner
	%inext =l add %i, 1
	jmp @outer
@return
	%noff =l shl %n, 3
	%nptr =l add %row, %noff
	%result =l loadl %nptr
	ret %result
}
