export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet

@packet
	%si =l phi @start 0, @literal_continue %literal_end, @repeat_continue %repeat_end, @noop %si_next
	%oi =l phi @start 0, @literal_continue %literal_out_end, @repeat_continue %repeat_out_end, @noop %oi
	%has_input =w csltl %si, %n
	jnz %has_input, @header, @success

@header
	%header_ptr =l add %src, %si
	%h =w loadsb %header_ptr
	%h64 =l extsw %h
	%si_next =l add %si, 1
	%is_noop =w ceqw %h, -128
	jnz %is_noop, @noop, @classify

@classify
	%is_repeat =w csltw %h, 0
	jnz %is_repeat, @repeat_check, @literal_check

@noop
	jmp @packet

@literal_check
	%literal_len =l add %h64, 1
	%literal_end =l add %si_next, %literal_len
	%literal_truncated =w csltl %n, %literal_end
	jnz %literal_truncated, @source_error, @literal_capacity

@literal_capacity
	%literal_out_end =l add %oi, %literal_len
	%literal_overflow =w csltl %cap, %literal_out_end
	jnz %literal_overflow, @literal_over, @literal_fit

@literal_over
	%literal_available =l sub %cap, %oi
	jmp @literal_setup

@literal_fit
	jmp @literal_setup

@literal_setup
	%literal_count =l phi @literal_over %literal_available, @literal_fit %literal_len
	%literal_failed =w phi @literal_over 1, @literal_fit 0
	jmp @literal_loop

@literal_loop
	%literal_i =l phi @literal_setup 0, @literal_store %literal_i_next
	%literal_more =w csltl %literal_i, %literal_count
	jnz %literal_more, @literal_store, @literal_done

@literal_store
	%literal_src_off =l add %si_next, %literal_i
	%literal_src_ptr =l add %src, %literal_src_off
	%literal_byte =w loadub %literal_src_ptr
	%literal_out_off =l add %oi, %literal_i
	%literal_out_ptr =l add %out, %literal_out_off
	storeb %literal_byte, %literal_out_ptr
	%literal_i_next =l add %literal_i, 1
	jmp @literal_loop

@literal_done
	jnz %literal_failed, @capacity_error, @literal_continue

@literal_continue
	jmp @packet

@repeat_check
	%repeat_has_byte =w csltl %si_next, %n
	jnz %repeat_has_byte, @repeat_prepare, @source_error

@repeat_prepare
	%repeat_ptr =l add %src, %si_next
	%repeat_byte =w loadub %repeat_ptr
	%repeat_end =l add %si_next, 1
	%repeat_len =l sub 1, %h64
	%repeat_out_end =l add %oi, %repeat_len
	%repeat_overflow =w csltl %cap, %repeat_out_end
	jnz %repeat_overflow, @repeat_over, @repeat_fit

@repeat_over
	%repeat_available =l sub %cap, %oi
	jmp @repeat_setup

@repeat_fit
	jmp @repeat_setup

@repeat_setup
	%repeat_count =l phi @repeat_over %repeat_available, @repeat_fit %repeat_len
	%repeat_failed =w phi @repeat_over 1, @repeat_fit 0
	jmp @repeat_loop

@repeat_loop
	%repeat_i =l phi @repeat_setup 0, @repeat_store %repeat_i_next
	%repeat_more =w csltl %repeat_i, %repeat_count
	jnz %repeat_more, @repeat_store, @repeat_done

@repeat_store
	%repeat_out_off =l add %oi, %repeat_i
	%repeat_out_ptr =l add %out, %repeat_out_off
	storeb %repeat_byte, %repeat_out_ptr
	%repeat_i_next =l add %repeat_i, 1
	jmp @repeat_loop

@repeat_done
	jnz %repeat_failed, @capacity_error, @repeat_continue

@repeat_continue
	jmp @packet

@success
	ret %oi

@source_error
	ret -2

@capacity_error
	ret -1
}
