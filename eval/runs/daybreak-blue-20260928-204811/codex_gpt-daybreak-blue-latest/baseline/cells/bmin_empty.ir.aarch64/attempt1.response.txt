	jnz %empty, @empty, @init
@empty

@init

	jmp @loop
@loop
	%i =l phi @init 1, @take %next, @keep %next
	%min =l phi @init %first, @take %value, @keep %min
	%done =w ceql %i, %n
	jnz %done, @done, @body
@body

@keep
