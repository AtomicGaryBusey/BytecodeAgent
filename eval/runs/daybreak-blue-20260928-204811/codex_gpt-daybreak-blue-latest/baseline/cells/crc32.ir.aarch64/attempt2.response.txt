export function l $crc32(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @byte_done %i_next
	%crc =w phi @start -1, @byte_done %crc_next
	%more =w cultl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%initial =w xor %crc, %b
	jmp @bit_loop
@bit_loop
	%k =w phi @byte 0, @bit_next %k_next
	%c =w phi @byte %initial, @bit_next %c_next
	%bits_left =w cultw %k, 8
	jnz %bits_left, @bit, @byte_done
@bit
	%lsb =w and %c, 1
	jnz %lsb, @odd, @even
@odd
	%shifted_odd =w shr %c, 1
	%odd_crc =w xor %shifted_odd, -306674912
	jmp @bit_next
@even
	%even_crc =w shr %c, 1
	jmp @bit_next
@bit_next
	%c_next =w phi @odd %odd_crc, @even %even_crc
	%k_next =w add %k, 1
	jmp @bit_loop
@byte_done
	%crc_next =w copy %c
	%i_next =l add %i, 1
	jmp @loop
@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
