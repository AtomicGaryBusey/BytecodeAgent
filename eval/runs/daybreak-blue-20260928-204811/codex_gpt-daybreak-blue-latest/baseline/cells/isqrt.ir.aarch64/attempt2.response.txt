export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%rem =l phi @start %n, @next %rem2
	%res =l phi @start 0, @next %res2
	%bit =l phi @start 4611686018427387904, @next %bit2
	%done =w ceql %bit, 0
	jnz %done, @return, @test
@test
	%trial =l add %res, %bit
	%less =w cultl %rem, %trial
	jnz %less, @skip, @take
@take
	%newrem =l sub %rem, %trial
	%half1 =l shr %res, 1
	%newres =l add %half1, %bit
	jmp @next
@skip
	%half2 =l shr %res, 1
	jmp @next
@next
	%rem2 =l phi @take %newrem, @skip %rem
	%res2 =l phi @take %newres, @skip %half2
	%bit2 =l shr %bit, 2
	jmp @loop
@return
	ret %res
}
