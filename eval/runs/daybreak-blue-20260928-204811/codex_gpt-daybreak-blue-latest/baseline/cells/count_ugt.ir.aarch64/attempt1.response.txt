export function l $count_ugt(l %a, l %n, l %t) {
@start
	%zero =l copy 0
	jnz %n, @loop, @done

@loop
	%p =l phi @start %a, @loop %nextp
	%i =l phi @start %zero, @loop %nexti
	%count =l phi @start %zero, @loop %nextcount
	%value =l loadl %p
	%greater =w cugtl %value, %t
	%increment =l extuw %greater
	%nextcount =l add %count, %increment
	%nextp =l add %p, 8
	%nexti =l add %i, 1
	%more =w csltl %nexti, %n
	jnz %more, @loop, @done_loop

@done_loop
	ret %nextcount

@done
	ret %zero
}
