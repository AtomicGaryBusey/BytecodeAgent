export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%is =l alloc8 8
	%os =l alloc8 8
	%js =l alloc8 8
	storel 0, %is
	storel 0, %os
	jmp @main

@main
	%i =l loadl %is
	%atend =w cslel %n, %i
	jnz %atend, @success, @run_i_init

@run_i_init
	%ip =l add %src, %i
	%ib =w loadub %ip
	%rik0 =l add %i, 1
	jmp @run_i_loop

@run_i_loop
	%rik =l phi @run_i_init %rik0, @run_i_inc %rik1
	%rir =l phi @run_i_init 1, @run_i_inc %rir1
	%ri_n =w csltl %rik, %n
	jnz %ri_n, @run_i_cap, @run_i_done

@run_i_cap
	%ri_c =w csltl %rir, 128
	jnz %ri_c, @run_i_cmp, @run_i_done

@run_i_cmp
	%rikp =l add %src, %rik
	%rikb =w loadub %rikp
	%ri_eq =w ceqw %rikb, %ib
	jnz %ri_eq, @run_i_inc, @run_i_done

@run_i_inc
	%rik1 =l add %rik, 1
	%rir1 =l add %rir, 1
	jmp @run_i_loop

@run_i_done
	%isrep =w csgel %rir, 3
	jnz %isrep, @repeat_check, @literal_init

@repeat_check
	%o_rep =l loadl %os
	%rep_end =l add %o_rep, 2
	%rep_over =w csgtl %rep_end, %cap
	jnz %rep_over, @failure, @repeat_write

@repeat_write
	%rep_hdr =l sub 257, %rir
	%rep_hp =l add %out, %o_rep
	storeb %rep_hdr, %rep_hp
	%rep_bp =l add %rep_hp, 1
	storeb %ib, %rep_bp
	%rep_i1 =l add %i, %rir
	storel %rep_i1, %is
	storel %rep_end, %os
	jmp @main

@literal_init
	storel %i, %js
	jmp @literal_scan

@literal_scan
	%lj =l loadl %js
	%lj_n =w csltl %lj, %n
	jnz %lj_n, @literal_len_check, @literal_ready

@literal_len_check
	%llen0 =l sub %lj, %i
	%llen_ok =w csltl %llen0, 128
	jnz %llen_ok, @literal_run_init, @literal_ready

@literal_run_init
	%ljp =l add %src, %lj
	%ljb =w loadub %ljp
	%lrk0 =l add %lj, 1
	jmp @literal_run_loop

@literal_run_loop
	%lrk =l phi @literal_run_init %lrk0, @literal_run_inc %lrk1
	%lrr =l phi @literal_run_init 1, @literal_run_inc %lrr1
	%lr_n =w csltl %lrk, %n
	jnz %lr_n, @literal_run_cap, @literal_run_done

@literal_run_cap
	%lr_c =w csltl %lrr, 128
	jnz %lr_c, @literal_run_cmp, @literal_run_done

@literal_run_cmp
	%lrkp =l add %src, %lrk
	%lrkb =w loadub %lrkp
	%lr_eq =w ceqw %lrkb, %ljb
	jnz %lr_eq, @literal_run_inc, @literal_run_done

@literal_run_inc
	%lrk1 =l add %lrk, 1
	%lrr1 =l add %lrr, 1
	jmp @literal_run_loop

@literal_run_done
	%lr_rep =w csgel %lrr, 3
	jnz %lr_rep, @literal_ready, @literal_advance

@literal_advance
	%lj1 =l add %lj, 1
	storel %lj1, %js
	jmp @literal_scan

@literal_ready
	%lend_i =l loadl %js
	%llen =l sub %lend_i, %i
	%o_lit =l loadl %os
	%packet_size =l add %llen, 1
	%lit_end =l add %o_lit, %packet_size
	%lit_over =w csgtl %lit_end, %cap
	jnz %lit_over, @failure, @literal_header

@literal_header
	%lit_hdr =l sub %llen, 1
	%lit_hp =l add %out, %o_lit
	storeb %lit_hdr, %lit_hp
	jmp @copy_loop

@copy_loop
	%q =l phi @literal_header 0, @copy_inc %q1
	%copy_more =w csltl %q, %llen
	jnz %copy_more, @copy_byte, @copy_done

@copy_byte
	%copy_si =l add %i, %q
	%copy_sp =l add %src, %copy_si
	%copy_b =w loadub %copy_sp
	%copy_oq =l add %o_lit, %q
	%copy_di =l add %copy_oq, 1
	%copy_dp =l add %out, %copy_di
	storeb %copy_b, %copy_dp
	jmp @copy_inc

@copy_inc
	%q1 =l add %q, 1
	jmp @copy_loop

@copy_done
	storel %lend_i, %is
	storel %lit_end, %os
	jmp @main

@success
	%result =l loadl %os
	ret %result

@failure
	ret -1
}
