reverse:
    xor eax, eax
    test rsi, rsi
    je .Ldone
    lea rdx, [rdi + rsi*8 - 8]
.Lloop:
    cmp rdi, rdx
    jae .Ldone
    mov rcx, qword ptr [rdi]
    mov r8, qword ptr [rdx]
    mov qword ptr [rdi], r8
    mov qword ptr [rdx], rcx
    add rdi, 8
    sub rdx, 8
    jmp .Lloop
.Ldone:
    ret
