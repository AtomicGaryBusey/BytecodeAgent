export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%rp =l alloc8 8
	%jp =l alloc8 8
	%kp =l alloc8 8
	storel 0, %ip
	storel 0, %op
	jmp @outer

@outer
	%i =l loadl %ip
	%more =w csltl %i, %n
	jnz %more, @measure_init, @done

@measure_init
	%src_i =l add %src, %i
	%b =w loadub %src_i
	storel 1, %rp
	jmp @measure_test

@measure_test
	%r =l loadl %rp
	%under128 =w csltl %r, 128
	jnz %under128, @measure_bound, @measure_done

@measure_bound
	%ir =l add %i, %r
	%inside =w csltl %ir, %n
	jnz %inside, @measure_byte, @measure_done

@measure_byte
	%src_ir =l add %src, %ir
	%br =w loadub %src_ir
	%same =w ceqw %br, %b
	jnz %same, @measure_inc, @measure_done

@measure_inc
	%r_next =l add %r, 1
	storel %r_next, %rp
	jmp @measure_test

@measure_done
	%run =l loadl %rp
	%is_repeat =w csgel %run, 3
	jnz %is_repeat, @repeat_check, @literal_init

@repeat_check
	%o_rep =l loadl %op
	%rep_end =l add %o_rep, 2
	%rep_fits =w cslel %rep_end, %cap
	jnz %rep_fits, @repeat_write, @fail

@repeat_write
	%rep_header_l =l sub 257, %run
	%rep_header =w copy %rep_header_l
	%out_rep_h =l add %out, %o_rep
	storeb %rep_header, %out_rep_h
	%out_rep_b =l add %out_rep_h, 1
	storeb %b, %out_rep_b
	storel %rep_end, %op
	%i_rep_next =l add %i, %run
	storel %i_rep_next, %ip
	jmp @outer

@literal_init
	storel %i, %jp
	jmp @literal_test

@literal_test
	%j =l loadl %jp
	%lit_len_now =l sub %j, %i
	%lit_room =w csltl %lit_len_now, 128
	jnz %lit_room, @literal_bound, @literal_ready

@literal_bound
	%j_inside =w csltl %j, %n
	jnz %j_inside, @literal_run_init, @literal_ready

@literal_run_init
	%src_j =l add %src, %j
	%bj =w loadub %src_j
	storel 1, %rp
	jmp @literal_run_test

@literal_run_test
	%lr =l loadl %rp
	%lr_under128 =w csltl %lr, 128
	jnz %lr_under128, @literal_run_bound, @literal_run_done

@literal_run_bound
	%jr =l add %j, %lr
	%jr_inside =w csltl %jr, %n
	jnz %jr_inside, @literal_run_byte, @literal_run_done

@literal_run_byte
	%src_jr =l add %src, %jr
	%bjr =w loadub %src_jr
	%lr_same =w ceqw %bjr, %bj
	jnz %lr_same, @literal_run_inc, @literal_run_done

@literal_run_inc
	%lr_next =l add %lr, 1
	storel %lr_next, %rp
	jmp @literal_run_test

@literal_run_done
	%jrun =l loadl %rp
	%j_repeat =w csgel %jrun, 3
	jnz %j_repeat, @literal_ready, @literal_advance

@literal_advance
	%j_next =l add %j, 1
	storel %j_next, %jp
	jmp @literal_test

@literal_ready
	%j_final =l loadl %jp
	%lit_len =l sub %j_final, %i
	%o_lit =l loadl %op
	%lit_payload_end =l add %o_lit, %lit_len
	%lit_end =l add %lit_payload_end, 1
	%lit_fits =w cslel %lit_end, %cap
	jnz %lit_fits, @literal_header, @fail

@literal_header
	%lit_header_l =l sub %lit_len, 1
	%lit_header =w copy %lit_header_l
	%out_lit_h =l add %out, %o_lit
	storeb %lit_header, %out_lit_h
	storel 0, %kp
	jmp @literal_copy_test

@literal_copy_test
	%k =l loadl %kp
	%copy_more =w csltl %k, %lit_len
	jnz %copy_more, @literal_copy, @literal_finish

@literal_copy
	%src_off =l add %i, %k
	%src_copy =l add %src, %src_off
	%copy_byte =w loadub %src_copy
	%out_base =l add %o_lit, 1
	%out_off =l add %out_base, %k
	%out_copy =l add %out, %out_off
	storeb %copy_byte, %out_copy
	%k_next =l add %k, 1
	storel %k_next, %kp
	jmp @literal_copy_test

@literal_finish
	storel %lit_end, %op
	storel %j_final, %ip
	jmp @outer

@fail
	ret -1

@done
	%result =l loadl %op
	ret %result
}
