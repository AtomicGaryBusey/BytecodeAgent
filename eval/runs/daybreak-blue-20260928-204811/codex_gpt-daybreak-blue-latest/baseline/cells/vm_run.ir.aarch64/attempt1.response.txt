export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	%rp =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop

@loop
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @stop, @fetch

@fetch
	%coff =l shl %pc, 3
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%is0 =w ceql %op, 0
	jnz %is0, @stop, @test1

@test1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @test2

@test2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @test3

@test3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @test4

@test4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @test5

@test5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @test6

@test6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @test7

@test7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @test8

@test8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @test9

@test9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @test10

@test10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @test11

@test11
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @other

@push
	%ppc =l loadl %pcp
	%ppc1 =l add %ppc, 1
	%pcoff =l shl %ppc1, 3
	%pvaddr =l add %code, %pcoff
	%pv =l loadl %pvaddr
	%pd =l loadl %dp
	%pdoff =l shl %pd, 3
	%psaddr =l add %stack, %pdoff
	storel %pv, %psaddr
	%pd1 =l add %pd, 1
	storel %pd1, %dp
	%ppc2 =l add %ppc, 2
	storel %ppc2, %pcp
	jmp @loop

@add
	%ad =l loadl %dp
	%adi1 =l sub %ad, 1
	%adi2 =l sub %ad, 2
	%ado1 =l shl %adi1, 3
	%ado2 =l shl %adi2, 3
	%ada1 =l add %stack, %ado1
	%ada2 =l add %stack, %ado2
	%adb =l loadl %ada1
	%ada =l loadl %ada2
	%adr =l add %ada, %adb
	storel %adr, %ada2
	storel %adi1, %dp
	%adpc =l loadl %pcp
	%adpc1 =l add %adpc, 1
	storel %adpc1, %pcp
	jmp @loop

@sub
	%sd =l loadl %dp
	%sdi1 =l sub %sd, 1
	%sdi2 =l sub %sd, 2
	%sdo1 =l shl %sdi1, 3
	%sdo2 =l shl %sdi2, 3
	%sda1 =l add %stack, %sdo1
	%sda2 =l add %stack, %sdo2
	%sdb =l loadl %sda1
	%sda =l loadl %sda2
	%sdr =l sub %sda, %sdb
	storel %sdr, %sda2
	storel %sdi1, %dp
	%sdpc =l loadl %pcp
	%sdpc1 =l add %sdpc, 1
	storel %sdpc1, %pcp
	jmp @loop

@mul
	%md =l loadl %dp
	%mdi1 =l sub %md, 1
	%mdi2 =l sub %md, 2
	%mdo1 =l shl %mdi1, 3
	%mdo2 =l shl %mdi2, 3
	%mda1 =l add %stack, %mdo1
	%mda2 =l add %stack, %mdo2
	%mdb =l loadl %mda1
	%mda =l loadl %mda2
	%mdr =l mul %mda, %mdb
	storel %mdr, %mda2
	storel %mdi1, %dp
	%mdpc =l loadl %pcp
	%mdpc1 =l add %mdpc, 1
	storel %mdpc1, %pcp
	jmp @loop

@dup
	%dd =l loadl %dp
	%ddi =l sub %dd, 1
	%ddio =l shl %ddi, 3
	%ddia =l add %stack, %ddio
	%ddv =l loadl %ddia
	%ddo =l shl %dd, 3
	%dda =l add %stack, %ddo
	storel %ddv, %dda
	%dd1 =l add %dd, 1
	storel %dd1, %dp
	%ddpc =l loadl %pcp
	%ddpc1 =l add %ddpc, 1
	storel %ddpc1, %pcp
	jmp @loop

@swap
	%swd =l loadl %dp
	%swi1 =l sub %swd, 1
	%swi2 =l sub %swd, 2
	%swo1 =l shl %swi1, 3
	%swo2 =l shl %swi2, 3
	%swa1 =l add %stack, %swo1
	%swa2 =l add %stack, %swo2
	%swv1 =l loadl %swa1
	%swv2 =l loadl %swa2
	storel %swv2, %swa1
	storel %swv1, %swa2
	%swpc =l loadl %pcp
	%swpc1 =l add %swpc, 1
	storel %swpc1, %pcp
	jmp @loop

@drop
	%drd =l loadl %dp
	%drd1 =l sub %drd, 1
	storel %drd1, %dp
	%drpc =l loadl %pcp
	%drpc1 =l add %drpc, 1
	storel %drpc1, %pcp
	jmp @loop

@over
	%ovd =l loadl %dp
	%ovi =l sub %ovd, 2
	%ovio =l shl %ovi, 3
	%ovia =l add %stack, %ovio
	%ovv =l loadl %ovia
	%ovdo =l shl %ovd, 3
	%ovda =l add %stack, %ovdo
	storel %ovv, %ovda
	%ovd1 =l add %ovd, 1
	storel %ovd1, %dp
	%ovpc =l loadl %pcp
	%ovpc1 =l add %ovpc, 1
	storel %ovpc1, %pcp
	jmp @loop

@lt
	%ltd =l loadl %dp
	%lti1 =l sub %ltd, 1
	%lti2 =l sub %ltd, 2
	%lto1 =l shl %lti1, 3
	%lto2 =l shl %lti2, 3
	%lta1 =l add %stack, %lto1
	%lta2 =l add %stack, %lto2
	%ltb =l loadl %lta1
	%lta =l loadl %lta2
	%ltw =w csltl %lta, %ltb
	%ltr =l extsw %ltw
	storel %ltr, %lta2
	storel %lti1, %dp
	%ltpc =l loadl %pcp
	%ltpc1 =l add %ltpc, 1
	storel %ltpc1, %pcp
	jmp @loop

@jz
	%jzd =l loadl %dp
	%jzd1 =l sub %jzd, 1
	%jzo =l shl %jzd1, 3
	%jza =l add %stack, %jzo
	%jzv =l loadl %jza
	storel %jzd1, %dp
	%jzpc =l loadl %pcp
	%jzpc1 =l add %jzpc, 1
	%jzco =l shl %jzpc1, 3
	%jzca =l add %code, %jzco
	%jzt =l loadl %jzca
	%jzz =w ceql %jzv, 0
	jnz %jzz, @jz_taken, @jz_next

@jz_taken
	storel %jzt, %pcp
	jmp @loop

@jz_next
	%jzpc2 =l add %jzpc, 2
	storel %jzpc2, %pcp
	jmp @loop

@jmp
	%jmpc =l loadl %pcp
	%jmpc1 =l add %jmpc, 1
	%jmco =l shl %jmpc1, 3
	%jmca =l add %code, %jmco
	%jmt =l loadl %jmca
	storel %jmt, %pcp
	jmp @loop

@other
	%opc =l loadl %pcp
	%opc1 =l add %opc, 1
	storel %opc1, %pcp
	jmp @loop

@stop
	%std =l loadl %dp
	%empty =w ceql %std, 0
	jnz %empty, @savezero, @savetop

@savezero
	storel 0, %rp
	jmp @clear_init

@savetop
	%sti =l sub %std, 1
	%sto =l shl %sti, 3
	%sta =l add %stack, %sto
	%stv =l loadl %sta
	storel %stv, %rp
	jmp @clear_init

@clear_init
	storel %std, %ip
	jmp @clear_test

@clear_test
	%ci =l loadl %ip
	%cmore =w csltl %ci, %cap
	jnz %cmore, @clear_body, @done

@clear_body
	%cio =l shl %ci, 3
	%cia =l add %stack, %cio
	storel 0, %cia
	%ci1 =l add %ci, 1
	storel %ci1, %ip
	jmp @clear_test

@done
	%result =l loadl %rp
	ret %result
}
