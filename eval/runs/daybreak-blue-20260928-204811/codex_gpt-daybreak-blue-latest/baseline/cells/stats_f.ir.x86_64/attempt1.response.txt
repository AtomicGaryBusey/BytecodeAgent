export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%p0 =l add %a, 8
	jmp @loop

@loop
	%i =l phi @start 1, @step %inext
	%p =l phi @start %p0, @step %pnext
	%min =l phi @start %y0, @step %newmin
	%max =l phi @start %y0, @step %newmax
	%sum =l phi @start %y0, @step %newsum
	%count =l phi @start %odd0, @step %newcount
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	jnz %ismin, @minyes, @minno

@minyes
	jmp @minmerge

@minno
	jmp @minmerge

@minmerge
	%newmin =l phi @minyes %y, @minno %min
	%ismax =w csgtl %y, %max
	jnz %ismax, @maxyes, @maxno

@maxyes
	jmp @maxmerge

@maxno
	jmp @maxmerge

@maxmerge
	%newmax =l phi @maxyes %y, @maxno %max
	%newsum =l add %sum, %y
	%odd =l and %y, 1
	%newcount =l add %count, %odd
	%pnext =l add %p, 8
	%inext =l add %i, 1
	jmp @step

@step
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
