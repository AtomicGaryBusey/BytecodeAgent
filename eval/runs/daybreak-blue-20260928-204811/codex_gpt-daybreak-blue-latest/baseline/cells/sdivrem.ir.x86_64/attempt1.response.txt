export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%iszero =w ceql %b, 0
	jnz %iszero, @divzero, @checkoverflow

@checkoverflow
	%ismin =w ceql %a, -9223372036854775808
	%isnegone =w ceql %b, -1
	%overflow =w and %ismin, %isnegone
	jnz %overflow, @overflow, @divide

@divzero
	storel %a, %rem
	ret 0

@overflow
	storel 0, %rem
	ret -9223372036854775808

@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
