export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%n2 =l mul %n, %n
	%i =l copy 0
	jmp @zero_out_test

@zero_out_test
	%c0 =w csltl %i, %n2
	jnz %c0, @zero_out_body, @diag_init

@zero_out_body
	%off0 =l mul %i, 8
	%op0 =l add %out, %off0
	storel 0, %op0
	%i =l add %i, 1
	jmp @zero_out_test

@diag_init
	%d =l copy 0
	jmp @diag_test

@diag_test
	%c1 =w csltl %d, %n
	jnz %c1, @diag_body, @power_init

@diag_body
	%dn =l mul %d, %n
	%didx =l add %dn, %d
	%doff =l mul %didx, 8
	%dptr =l add %out, %doff
	storel 1, %dptr
	%d =l add %d, 1
	jmp @diag_test

@power_init
	%e =l copy 0
	jmp @power_test

@power_test
	%c2 =w csltl %e, %k
	jnz %c2, @mul_init, @clear_tmp_init

@mul_init
	%r =l copy 0
	jmp @row_test

@row_test
	%c3 =w csltl %r, %n
	jnz %c3, @row_body, @copy_init

@row_body
	%col =l copy 0
	jmp @col_test

@col_test
	%c4 =w csltl %col, %n
	jnz %c4, @col_body, @next_row

@col_body
	%sum =l copy 0
	%t =l copy 0
	jmp @dot_test

@dot_test
	%c5 =w csltl %t, %n
	jnz %c5, @dot_body, @store_product

@dot_body
	%rn =l mul %r, %n
	%oidx =l add %rn, %t
	%ooff =l mul %oidx, 8
	%optr =l add %out, %ooff
	%ov =l loadl %optr
	%tn =l mul %t, %n
	%aidx =l add %tn, %col
	%aoff =l mul %aidx, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	%prod =l mul %ov, %av
	%sum =l add %sum, %prod
	%t =l add %t, 1
	jmp @dot_test

@store_product
	%srn =l mul %r, %n
	%sidx =l add %srn, %col
	%soff =l mul %sidx, 8
	%sptr =l add %tmp, %soff
	storel %sum, %sptr
	%col =l add %col, 1
	jmp @col_test

@next_row
	%r =l add %r, 1
	jmp @row_test

@copy_init
	%q =l copy 0
	jmp @copy_test

@copy_test
	%c6 =w csltl %q, %n2
	jnz %c6, @copy_body, @next_power

@copy_body
	%qoff =l mul %q, 8
	%qtmp =l add %tmp, %qoff
	%qval =l loadl %qtmp
	%qout =l add %out, %qoff
	storel %qval, %qout
	storel 0, %qtmp
	%q =l add %q, 1
	jmp @copy_test

@next_power
	%e =l add %e, 1
	jmp @power_test

@clear_tmp_init
	%twon2 =l mul %n2, 2
	%z =l copy 0
	jmp @clear_tmp_test

@clear_tmp_test
	%c7 =w csltl %z, %twon2
	jnz %c7, @clear_tmp_body, @done

@clear_tmp_body
	%zoff =l mul %z, 8
	%zptr =l add %tmp, %zoff
	storel 0, %zptr
	%z =l add %z, 1
	jmp @clear_tmp_test

@done
	ret 0
}
