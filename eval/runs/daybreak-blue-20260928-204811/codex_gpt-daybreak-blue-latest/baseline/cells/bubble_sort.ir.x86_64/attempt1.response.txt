export function l $bubble_sort(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @done, @outer_init

@outer_init
	%initial =l sub %n, 1
	jmp @outer

@outer
	%pass =l phi @outer_init %initial, @outer_next %passnext
	jmp @inner_test

@inner_test
	%j =l phi @outer 0, @advance %jnext
	%p =l phi @outer %a, @advance %pnext
	%inside =w csltl %j, %pass
	jnz %inside, @compare, @outer_next

@compare
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%reverse =w csltl %y, %x
	jnz %reverse, @swap, @advance

@swap
	storel %y, %p
	storel %x, %q
	jmp @advance

@advance
	%jnext =l add %j, 1
	%pnext =l add %p, 8
	jmp @inner_test

@outer_next
	%passnext =l sub %pass, 1
	%more =w csltl 0, %passnext
	jnz %more, @outer, @done

@done
	ret 0
}
