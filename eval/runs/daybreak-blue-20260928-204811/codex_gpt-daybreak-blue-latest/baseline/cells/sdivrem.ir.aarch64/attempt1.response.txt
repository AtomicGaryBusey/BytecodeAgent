export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @checknegone

@divzero
	storel %a, %rem
	ret 0

@checknegone
	%bn1 =w ceql %b, -1
	jnz %bn1, @checkmin, @normal

@checkmin
	%az =w ceql %a, 0
	jnz %az, @normal, @checkdouble

@checkdouble
	%twice =l add %a, %a
	%ismin =w ceql %twice, 0
	jnz %ismin, @overflow, @normal

@overflow
	storel 0, %rem
	ret %a

@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
