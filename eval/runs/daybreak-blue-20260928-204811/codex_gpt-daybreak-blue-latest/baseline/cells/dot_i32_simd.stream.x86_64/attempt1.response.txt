dot_i32_simd:
    pxor xmm4, xmm4
.Lvector:
    cmp rdx, 4
    jb .Lreduce
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pmuldq xmm0, xmm1
    psrldq xmm2, 4
    psrldq xmm3, 4
    pmuldq xmm2, xmm3
    paddq xmm4, xmm0
    paddq xmm4, xmm2
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    jmp .Lvector
.Lreduce:
    movdqa xmm0, xmm4
    psrldq xmm0, 8
    paddq xmm4, xmm0
    movq rax, xmm4
.Lscalar:
    test rdx, rdx
    jz .Ldone
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rdx
    jmp .Lscalar
.Ldone:
    ret
