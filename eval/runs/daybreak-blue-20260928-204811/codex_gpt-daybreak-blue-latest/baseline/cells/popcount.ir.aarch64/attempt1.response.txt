export function l $popcount(l %x) {
@start
	%empty =w ceql %x, 0
	jnz %empty, @zero, @loop
@loop
	%v =l phi @start %x, @loop %next
	%n =l phi @start 0, @loop %inc
	%dec =l sub %v, 1
	%next =l and %v, %dec
	%inc =l add %n, 1
	%done =w ceql %next, 0
	jnz %done, @result, @loop
@result
	ret %inc
@zero
	ret 0
}
