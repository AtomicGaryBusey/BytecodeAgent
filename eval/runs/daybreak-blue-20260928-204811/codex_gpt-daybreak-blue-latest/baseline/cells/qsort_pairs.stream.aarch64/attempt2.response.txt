qsort_pairs:
    cmp x1, #1
    b.le .Lreturn_zero
    stp x29, x30, [sp, #-48]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    mov x19, x0
    mov x20, x1
    mov x21, #0
    mov x22, #0
.Lpartition_loop:
    sub x9, x20, #1
    cmp x22, x9
    b.ge .Lplace_pivot
    add x10, x19, x22, lsl #4
    ldp x11, x12, [x10]
    add x13, x19, x9, lsl #4
    ldp x14, x15, [x13]
    cmp x11, x14
    b.lt .Lmove_left
    b.gt .Lnext_record
    cmp x12, x15
    b.lt .Lmove_left
    b .Lnext_record
.Lmove_left:
    add x13, x19, x21, lsl #4
    ldp x14, x15, [x13]
    stp x11, x12, [x13]
    stp x14, x15, [x10]
    add x21, x21, #1
.Lnext_record:
    add x22, x22, #1
    b .Lpartition_loop
.Lplace_pivot:
    sub x9, x20, #1
    add x10, x19, x9, lsl #4
    add x11, x19, x21, lsl #4
    ldp x12, x13, [x10]
    ldp x14, x15, [x11]
    stp x12, x13, [x11]
    stp x14, x15, [x10]
    cmp x21, #1
    b.le .Lskip_left
    mov x0, x19
    mov x1, x21
    bl qsort_pairs
.Lskip_left:
    sub x1, x20, x21
    sub x1, x1, #1
    cmp x1, #1
    b.le .Lskip_right
    add x9, x21, #1
    add x0, x19, x9, lsl #4
    bl qsort_pairs
.Lskip_right:
    mov x0, #0
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #48
    ret
.Lreturn_zero:
    mov x0, #0
    ret
