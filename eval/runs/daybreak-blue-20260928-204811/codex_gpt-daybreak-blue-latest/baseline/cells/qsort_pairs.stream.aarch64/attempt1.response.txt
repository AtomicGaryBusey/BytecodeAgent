qsort_pairs:
    cmp x1, #1
    b.le .Lreturn_zero
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    mov x19, x0
    mov x20, x1
    lsr x9, x20, #1
    add x10, x19, x9, lsl #4
    ldp x21, x22, [x10]
    mov x23, #0
    sub x24, x20, #1
.Lpartition:
.Lscan_left:
    add x9, x19, x23, lsl #4
    ldp x10, x11, [x9]
    cmp x10, x21
    b.lt .Lleft_advance
    b.gt .Lscan_right
    cmp x11, x22
    b.lt .Lleft_advance
    b .Lscan_right
.Lleft_advance:
    add x23, x23, #1
    b .Lscan_left
.Lscan_right:
    add x9, x19, x24, lsl #4
    ldp x10, x11, [x9]
    cmp x10, x21
    b.gt .Lright_retreat
    b.lt .Lcheck_cross
    cmp x11, x22
    b.gt .Lright_retreat
    b .Lcheck_cross
.Lright_retreat:
    sub x24, x24, #1
    b .Lscan_right
.Lcheck_cross:
    cmp x23, x24
    b.gt .Lpartition_done
    add x9, x19, x23, lsl #4
    add x10, x19, x24, lsl #4
    ldp x11, x12, [x9]
    ldp x13, x14, [x10]
    stp x13, x14, [x9]
    stp x11, x12, [x10]
    add x23, x23, #1
    sub x24, x24, #1
    b .Lpartition
.Lpartition_done:
    add x1, x24, #1
    cmp x1, #1
    b.le .Lskip_left
    mov x0, x19
    bl qsort_pairs
.Lskip_left:
    sub x1, x20, x23
    cmp x1, #1
    b.le .Lskip_right
    add x0, x19, x23, lsl #4
    bl qsort_pairs
.Lskip_right:
    mov x0, #0
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64
    ret
.Lreturn_zero:
    mov x0, #0
    ret
