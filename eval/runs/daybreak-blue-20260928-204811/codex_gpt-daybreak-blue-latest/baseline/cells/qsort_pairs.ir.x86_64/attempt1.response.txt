export function l $qsort_pairs(l %a, l %n) {
@entry
	%is =l alloc8 8
	%js =l alloc8 8
	%small =w cslel %n, 1
	jnz %small, @ret, @setup

@setup
	%nm1 =l sub %n, 1
	%lastoff =l mul %nm1, 16
	%lastp =l add %a, %lastoff
	%pivotk =l loadl %lastp
	%lastvptr =l add %lastp, 8
	%pivotv =l loadl %lastvptr
	storel 0, %is
	storel 0, %js
	jmp @loop

@loop
	%j =l loadl %js
	%more =w csltl %j, %nm1
	jnz %more, @inspect, @partition_done

@inspect
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%keyless =w csltl %jk, %pivotk
	jnz %keyless, @swap, @check_equal

@check_equal
	%keyeq =w ceql %jk, %pivotk
	jnz %keyeq, @check_value, @increment_j

@check_value
	%valless =w csltl %jv, %pivotv
	jnz %valless, @swap, @increment_j

@swap
	%i =l loadl %is
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%inext =l add %i, 1
	storel %inext, %is
	jmp @increment_j

@increment_j
	%jnext =l add %j, 1
	storel %jnext, %js
	jmp @loop

@partition_done
	%pi =l loadl %is
	%pioff =l mul %pi, 16
	%pip =l add %a, %pioff
	%oldk =l loadl %pip
	%pivp =l add %pip, 8
	%oldv =l loadl %pivp
	storel %pivotk, %pip
	storel %pivotv, %pivp
	storel %oldk, %lastp
	storel %oldv, %lastvptr
	%leftbig =w csgtl %pi, 1
	jnz %leftbig, @sort_left, @prepare_right

@sort_left
	%lr =l call $qsort_pairs(l %a, l %pi)
	jmp @prepare_right

@prepare_right
	%rightn0 =l sub %n, %pi
	%rightn =l sub %rightn0, 1
	%rightbig =w csgtl %rightn, 1
	jnz %rightbig, @sort_right, @ret

@sort_right
	%righti =l add %pi, 1
	%rightoff =l mul %righti, 16
	%rightp =l add %a, %rightoff
	%rr =l call $qsort_pairs(l %rightp, l %rightn)
	jmp @ret

@ret
	ret 0
}
