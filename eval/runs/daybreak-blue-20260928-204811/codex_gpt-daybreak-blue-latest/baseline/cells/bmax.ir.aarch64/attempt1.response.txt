export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
	jmp @loop
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%greater =w csgtl %val, %max
	jnz %greater, @update, @next
@update
	%max =l copy %val
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
