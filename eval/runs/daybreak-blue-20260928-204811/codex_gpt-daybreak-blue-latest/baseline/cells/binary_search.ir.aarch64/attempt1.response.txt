export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %next, @greater %mid
	%hi =l phi @start %n, @less %hi, @greater %mid
	%more =w csltl %lo, %hi
	jnz %more, @body, @notfound
@body
	%span =l sub %hi, %lo
	%half =l shrl %span, 1
	%mid =l add %lo, %half
	%offset =l shll %mid, 3
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%lower =w csltl %value, %key
	jnz %lower, @less, @check
@check
	%equal =w ceql %value, %key
	jnz %equal, @found, @greater
@less
	%next =l add %mid, 1
	jmp @loop
@greater
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
