export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%sum =l phi @start 0, @body %snext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%r =l call %f(l %x)
	%snext =l add %sum, %r
	%inext =l add %i, 1
	jmp @loop
@done
	ret %sum
}
