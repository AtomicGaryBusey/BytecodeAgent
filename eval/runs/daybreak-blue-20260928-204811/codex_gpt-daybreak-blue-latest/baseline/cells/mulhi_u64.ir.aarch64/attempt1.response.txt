export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l shr -1, 32
	%alo =l and %a, %mask
	%ahi =l shr %a, 32
	%blo =l and %b, %mask
	%bhi =l shr %b, 32

	%p00 =l mul %alo, %blo
	%p01 =l mul %alo, %bhi
	%p10 =l mul %ahi, %blo
	%p11 =l mul %ahi, %bhi

	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, %mask
	%p10lo =l and %p10, %mask
	%mid0 =l add %p00hi, %p01lo
	%mid1 =l add %mid0, %p10lo
	%carry =l shr %mid1, 32

	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%high0 =l add %p11, %p01hi
	%high1 =l add %high0, %p10hi
	%result =l add %high1, %carry
	ret %result
}
