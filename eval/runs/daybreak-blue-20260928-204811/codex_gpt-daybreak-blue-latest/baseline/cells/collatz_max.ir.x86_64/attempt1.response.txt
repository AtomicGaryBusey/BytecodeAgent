export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @after %knext
	%max =l phi @start 0, @after %maxnext
	%inrange =w cslel %k, %n
	jnz %inrange, @inner, @return
@inner
	%x =l phi @outer %k, @even %xeven, @odd %xodd
	%steps =l phi @outer 0, @even %stepseven, @odd %stepsodd
	%isone =w ceql %x, 1
	jnz %isone, @done, @transform
@transform
	%bit =l and %x, 1
	%iseven =w ceql %bit, 0
	jnz %iseven, @even, @odd
@even
	%xeven =l shrl %x, 1
	%stepseven =l add %steps, 1
	jmp @inner
@odd
	%triple =l mul %x, 3
	%xodd =l add %triple, 1
	%stepsodd =l add %steps, 1
	jmp @inner
@done
	%isgreater =w csgtl %steps, %max
	jnz %isgreater, @update, @keep
@update
	jmp @after
@keep
	jmp @after
@after
	%maxnext =l phi @update %steps, @keep %max
	%knext =l add %k, 1
	jmp @outer
@return
	ret %max
}
