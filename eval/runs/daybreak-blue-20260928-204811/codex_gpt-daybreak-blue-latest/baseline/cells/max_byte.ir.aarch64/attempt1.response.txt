export function l $max_byte(l %p, l %n) {
@entry
	%empty =w ceql %n, 0
	jnz %empty, @exit, @loop
@loop
	%i =l phi @entry 0, @join %ni
	%max =l phi @entry 0, @join %nmax
	%addr =l add %p, %i
	%byte =l loadub %addr
	%greater =w csltl %max, %byte
	jnz %greater, @take, @keep
@take
	jmp @join
@keep
	jmp @join
@join
	%nmax =l phi @take %byte, @keep %max
	%ni =l add %i, 1
	%more =w csltl %ni, %n
	jnz %more, @loop, @exit
@exit
	%result =l phi @entry 0, @join %nmax
	ret %result
}
