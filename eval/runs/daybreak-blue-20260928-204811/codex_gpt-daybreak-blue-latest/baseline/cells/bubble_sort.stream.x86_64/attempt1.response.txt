bubble_sort:
    cmp rsi, 2
    jb .Ldone
    lea r8, [rsi - 1]
.Louter:
    xor ecx, ecx
    xor r9d, r9d
.Linner:
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rax, rdx
    jle .Lnext
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rcx*8 + 8], rax
    mov r9d, 1
.Lnext:
    inc rcx
    cmp rcx, r8
    jb .Linner
    test r9d, r9d
    jz .Ldone
    dec r8
    jnz .Louter
.Ldone:
    xor eax, eax
    ret
