export function l $reverse(l %a, l %n) {
@start
	%half =l shr %n, 1
	%go =w cnel %half, 0
	jnz %go, @loop, @done
@loop
	%i =l phi @start 0, @loop %next
	%leftoff =l shl %i, 3
	%leftptr =l add %a, %leftoff
	%last =l sub %n, 1
	%rightidx =l sub %last, %i
	%rightoff =l shl %rightidx, 3
	%rightptr =l add %a, %rightoff
	%leftval =l loadl %leftptr
	%rightval =l loadl %rightptr
	storel %rightval, %leftptr
	storel %leftval, %rightptr
	%next =l add %i, 1
	%again =w csltl %next, %half
	jnz %again, @loop, @done
@done
	ret 0
}
