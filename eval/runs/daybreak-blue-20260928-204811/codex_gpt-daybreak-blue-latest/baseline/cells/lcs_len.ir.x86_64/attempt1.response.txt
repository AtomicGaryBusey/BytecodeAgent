export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	jmp @init_head

@init_head
	%zj =l phi @entry 0, @init_body %znext
	%zcond =w cslel %zj, %n
	jnz %zcond, @init_body, @init_done

@init_body
	%zoff =l mul %zj, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%znext =l add %zj, 1
	jmp @init_head

@init_done
	jmp @outer_head

@outer_head
	%i =l phi @init_done 0, @outer_latch %inext
	%icond =w csltl %i, %m
	jnz %icond, @outer_body, @done

@outer_body
	%xptr =l add %x, %i
	%xb =w loadub %xptr
	jmp @inner_head

@inner_head
	%j =l phi @outer_body 1, @inner_latch %jnext
	%diag =l phi @outer_body 0, @inner_latch %up
	%jcond =w cslel %j, %n
	jnz %jcond, @inner_body, @outer_latch

@inner_body
	%joff =l mul %j, 8
	%rptr =l add %row, %joff
	%up =l loadl %rptr
	%ym1 =l sub %j, 1
	%yptr =l add %y, %ym1
	%yb =w loadub %yptr
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch

@match
	%mval =l add %diag, 1
	jmp @store

@mismatch
	%loff =l mul %ym1, 8
	%lptr =l add %row, %loff
	%left =l loadl %lptr
	%takeleft =w csltl %up, %left
	jnz %takeleft, @use_left, @use_up

@use_left
	jmp @store

@use_up
	jmp @store

@store
	%val =l phi @match %mval, @use_left %left, @use_up %up
	storel %val, %rptr
	jmp @inner_latch

@inner_latch
	%jnext =l add %j, 1
	jmp @inner_head

@outer_latch
	%inext =l add %i, 1
	jmp @outer_head

@done
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%result =l loadl %nptr
	ret %result
}
