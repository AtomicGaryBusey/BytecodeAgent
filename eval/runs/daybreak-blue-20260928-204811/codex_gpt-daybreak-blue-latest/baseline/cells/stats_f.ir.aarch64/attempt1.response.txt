export function l $stats_f(l %a, l %n, l %f, l %out) {
@first
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%count0 =l and %y0, 1
	%more =w csltl 1, %n
	jnz %more, @loop, @done

@loop
	%i =l phi @first 1, @loop %inext
	%min =l phi @first %y0, @loop %minnew
	%max =l phi @first %y0, @loop %maxnew
	%sum =l phi @first %y0, @loop %sumnew
	%count =l phi @first %count0, @loop %countnew

	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)

	%lt =w csltl %y, %min
	%lt1 =l extuw %lt
	%ltmask =l sub 0, %lt1
	%mindiff =l xor %min, %y
	%minsel =l and %mindiff, %ltmask
	%minnew =l xor %min, %minsel

	%gt =w csltl %max, %y
	%gt1 =l extuw %gt
	%gtmask =l sub 0, %gt1
	%maxdiff =l xor %max, %y
	%maxsel =l and %maxdiff, %gtmask
	%maxnew =l xor %max, %maxsel

	%sumnew =l add %sum, %y
	%oddbit =l and %y, 1
	%countnew =l add %count, %oddbit
	%inext =l add %i, 1
	%again =w csltl %inext, %n
	jnz %again, @loop, @done

@done
	%resultmin =l phi @first %y0, @loop %minnew
	%resultmax =l phi @first %y0, @loop %maxnew
	%resultsum =l phi @first %y0, @loop %sumnew
	%resultcount =l phi @first %count0, @loop %countnew
	%out1 =l add %out, 8
	%out2 =l add %out, 16
	storel %resultmin, %out
	storel %resultmax, %out1
	storel %resultsum, %out2
	ret %resultcount
}
