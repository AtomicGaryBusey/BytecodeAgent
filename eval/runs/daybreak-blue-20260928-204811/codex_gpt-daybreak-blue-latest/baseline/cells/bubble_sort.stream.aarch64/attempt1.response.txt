bubble_sort:
    cmp x1, #2
    b.lt .Ldone
    sub x2, x1, #1
.Louter:
    mov x3, x0
    mov x4, x2
    mov x9, #0
.Linner:
    ldr x5, [x3]
    ldr x6, [x3, #8]
    cmp x5, x6
    b.le .Lnext
    str x6, [x3]
    str x5, [x3, #8]
    mov x9, #1
.Lnext:
    add x3, x3, #8
    subs x4, x4, #1
    b.ne .Linner
    cbz x9, .Ldone
    subs x2, x2, #1
    b.ne .Louter
.Ldone:
    mov x0, #0
    ret
