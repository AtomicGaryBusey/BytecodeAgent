export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32

	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1

	%x =l shl %p1, 32
	%lo1 =l add %p0, %x
	%c1w =w cultl %lo1, %p0
	%c1 =l extuw %c1w

	%y =l shl %p2, 32
	%lo2 =l add %lo1, %y
	%c2w =w cultl %lo2, %lo1
	%c2 =l extuw %c2w

	%h1 =l shr %p1, 32
	%h2 =l shr %p2, 32
	%h3 =l add %p3, %h1
	%h4 =l add %h3, %h2
	%h5 =l add %h4, %c1
	%hi =l add %h5, %c2
	ret %hi
}
