qsort_pairs:
cld
cmp rsi, 1
jle .Lbase
push rbp
push rbx
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov rax, rsi
shr rax, 1
shl rax, 4
mov r13, qword ptr [rbx + rax]
mov r14, qword ptr [rbx + rax + 8]
xor r15d, r15d
lea rbp, [r12 - 1]
.Lpartition:
.Lscan_left:
mov rax, r15
shl rax, 4
mov rdx, qword ptr [rbx + rax]
cmp rdx, r13
jl .Lleft_advance
jg .Lscan_right
mov rcx, qword ptr [rbx + rax + 8]
cmp rcx, r14
jl .Lleft_advance
jmp .Lscan_right
.Lleft_advance:
inc r15
jmp .Lscan_left
.Lscan_right:
mov rax, rbp
shl rax, 4
mov rdx, qword ptr [rbx + rax]
cmp rdx, r13
jg .Lright_retreat
jl .Lcheck_cross
mov rcx, qword ptr [rbx + rax + 8]
cmp rcx, r14
jg .Lright_retreat
jmp .Lcheck_cross
.Lright_retreat:
dec rbp
jmp .Lscan_right
.Lcheck_cross:
cmp r15, rbp
jg .Lpartition_done
mov r8, r15
shl r8, 4
add r8, rbx
mov r9, rbp
shl r9, 4
add r9, rbx
mov rax, qword ptr [r8]
mov rcx, qword ptr [r8 + 8]
mov rdx, qword ptr [r9]
mov rsi, qword ptr [r9 + 8]
mov qword ptr [r8], rdx
mov qword ptr [r8 + 8], rsi
mov qword ptr [r9], rax
mov qword ptr [r9 + 8], rcx
inc r15
dec rbp
jmp .Lpartition
.Lpartition_done:
mov rdi, rbx
lea rsi, [rbp + 1]
call qsort_pairs
mov rax, r15
shl rax, 4
lea rdi, [rbx + rax]
mov rsi, r12
sub rsi, r15
call qsort_pairs
xor eax, eax
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
cld
ret
.Lbase:
xor eax, eax
cld
