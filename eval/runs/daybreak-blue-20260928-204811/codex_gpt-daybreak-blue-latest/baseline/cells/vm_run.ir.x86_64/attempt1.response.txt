export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcslot =l alloc8 8
	%dslot =l alloc8 8
	%islot =l alloc8 8
	storel 0, %pcslot
	storel 0, %dslot
	jmp @loop

@loop
	%pc =l loadl %pcslot
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch

@fetch
	%pcoff =l shl %pc, 3
	%pcaddr =l add %code, %pcoff
	%op =l loadl %pcaddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @test1

@test1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @test2

@test2
	%is2 =w ceql %op, 2
	jnz %is2, @addop, @test3

@test3
	%is3 =w ceql %op, 3
	jnz %is3, @subop, @test4

@test4
	%is4 =w ceql %op, 4
	jnz %is4, @mulop, @test5

@test5
	%is5 =w ceql %op, 5
	jnz %is5, @dupop, @test6

@test6
	%is6 =w ceql %op, 6
	jnz %is6, @swapop, @test7

@test7
	%is7 =w ceql %op, 7
	jnz %is7, @dropop, @test8

@test8
	%is8 =w ceql %op, 8
	jnz %is8, @overop, @test9

@test9
	%is9 =w ceql %op, 9
	jnz %is9, @ltop, @test10

@test10
	%is10 =w ceql %op, 10
	jnz %is10, @jzop, @test11

@test11
	%is11 =w ceql %op, 11
	jnz %is11, @jmpop, @other

@push
	%ppc =l loadl %pcslot
	%operandpc =l add %ppc, 1
	%operandoff =l shl %operandpc, 3
	%operandaddr =l add %code, %operandoff
	%operand =l loadl %operandaddr
	%pd =l loadl %dslot
	%pdoff =l shl %pd, 3
	%pdaddr =l add %stack, %pdoff
	storel %operand, %pdaddr
	%pdnext =l add %pd, 1
	storel %pdnext, %dslot
	%ppcnext =l add %ppc, 2
	storel %ppcnext, %pcslot
	jmp @loop

@addop
	%ad =l loadl %dslot
	%ai =l sub %ad, 2
	%aj =l sub %ad, 1
	%aioff =l shl %ai, 3
	%ajoff =l shl %aj, 3
	%aiaddr =l add %stack, %aioff
	%ajaddr =l add %stack, %ajoff
	%av =l loadl %aiaddr
	%bv =l loadl %ajaddr
	%sum =l add %av, %bv
	storel %sum, %aiaddr
	%adnext =l sub %ad, 1
	storel %adnext, %dslot
	%apc =l loadl %pcslot
	%apcnext =l add %apc, 1
	storel %apcnext, %pcslot
	jmp @loop

@subop
	%sd =l loadl %dslot
	%si =l sub %sd, 2
	%sj =l sub %sd, 1
	%sioff =l shl %si, 3
	%sjoff =l shl %sj, 3
	%siaddr =l add %stack, %sioff
	%sjaddr =l add %stack, %sjoff
	%sav =l loadl %siaddr
	%sbv =l loadl %sjaddr
	%diff =l sub %sav, %sbv
	storel %diff, %siaddr
	%sdnext =l sub %sd, 1
	storel %sdnext, %dslot
	%spc =l loadl %pcslot
	%spcnext =l add %spc, 1
	storel %spcnext, %pcslot
	jmp @loop

@mulop
	%md =l loadl %dslot
	%mi =l sub %md, 2
	%mj =l sub %md, 1
	%mioff =l shl %mi, 3
	%mjoff =l shl %mj, 3
	%miaddr =l add %stack, %mioff
	%mjaddr =l add %stack, %mjoff
	%mav =l loadl %miaddr
	%mbv =l loadl %mjaddr
	%prod =l mul %mav, %mbv
	storel %prod, %miaddr
	%mdnext =l sub %md, 1
	storel %mdnext, %dslot
	%mpc =l loadl %pcslot
	%mpcnext =l add %mpc, 1
	storel %mpcnext, %pcslot
	jmp @loop

@dupop
	%dd =l loadl %dslot
	%dsrcidx =l sub %dd, 1
	%dsrcoff =l shl %dsrcidx, 3
	%ddstoff =l shl %dd, 3
	%dsrcaddr =l add %stack, %dsrcoff
	%ddstaddr =l add %stack, %ddstoff
	%dval =l loadl %dsrcaddr
	storel %dval, %ddstaddr
	%ddnext =l add %dd, 1
	storel %ddnext, %dslot
	%dpc =l loadl %pcslot
	%dpcnext =l add %dpc, 1
	storel %dpcnext, %pcslot
	jmp @loop

@swapop
	%swd =l loadl %dslot
	%swi =l sub %swd, 2
	%swj =l sub %swd, 1
	%swioff =l shl %swi, 3
	%swjoff =l shl %swj, 3
	%swiaddr =l add %stack, %swioff
	%swjaddr =l add %stack, %swjoff
	%swav =l loadl %swiaddr
	%swbv =l loadl %swjaddr
	storel %swbv, %swiaddr
	storel %swav, %swjaddr
	%swpc =l loadl %pcslot
	%swpcnext =l add %swpc, 1
	storel %swpcnext, %pcslot
	jmp @loop

@dropop
	%drd =l loadl %dslot
	%drdnext =l sub %drd, 1
	storel %drdnext, %dslot
	%drpc =l loadl %pcslot
	%drpcnext =l add %drpc, 1
	storel %drpcnext, %pcslot
	jmp @loop

@overop
	%od =l loadl %dslot
	%osrcidx =l sub %od, 2
	%osrcoff =l shl %osrcidx, 3
	%odstoff =l shl %od, 3
	%osrcaddr =l add %stack, %osrcoff
	%odstaddr =l add %stack, %odstoff
	%oval =l loadl %osrcaddr
	storel %oval, %odstaddr
	%odnext =l add %od, 1
	storel %odnext, %dslot
	%opc =l loadl %pcslot
	%opcnext =l add %opc, 1
	storel %opcnext, %pcslot
	jmp @loop

@ltop
	%ld =l loadl %dslot
	%li =l sub %ld, 2
	%lj =l sub %ld, 1
	%lioff =l shl %li, 3
	%ljoff =l shl %lj, 3
	%liaddr =l add %stack, %lioff
	%ljaddr =l add %stack, %ljoff
	%lav =l loadl %liaddr
	%lbv =l loadl %ljaddr
	%lcmp =w csltl %lav, %lbv
	%lval =l extuw %lcmp
	storel %lval, %liaddr
	%ldnext =l sub %ld, 1
	storel %ldnext, %dslot
	%lpc =l loadl %pcslot
	%lpcnext =l add %lpc, 1
	storel %lpcnext, %pcslot
	jmp @loop

@jzop
	%jzd =l loadl %dslot
	%jzdnext =l sub %jzd, 1
	%jzoff =l shl %jzdnext, 3
	%jzaddr =l add %stack, %jzoff
	%jzval =l loadl %jzaddr
	storel %jzdnext, %dslot
	%jzpc =l loadl %pcslot
	%jzzero =w ceql %jzval, 0
	jnz %jzzero, @jztaken, @jzfall

@jztaken
	%jztpc =l loadl %pcslot
	%jztargpc =l add %jztpc, 1
	%jztargoff =l shl %jztargpc, 3
	%jztargaddr =l add %code, %jztargoff
	%jztarget =l loadl %jztargaddr
	storel %jztarget, %pcslot
	jmp @loop

@jzfall
	%jzfpc =l loadl %pcslot
	%jzfnext =l add %jzfpc, 2
	storel %jzfnext, %pcslot
	jmp @loop

@jmpop
	%jmpc =l loadl %pcslot
	%jmtargpc =l add %jmpc, 1
	%jmtargoff =l shl %jmtargpc, 3
	%jmtargaddr =l add %code, %jmtargoff
	%jmtarget =l loadl %jmtargaddr
	storel %jmtarget, %pcslot
	jmp @loop

@other
	%otpc =l loadl %pcslot
	%otnext =l add %otpc, 1
	storel %otnext, %pcslot
	jmp @loop

@finish
	%fd =l loadl %dslot
	storel %fd, %islot
	jmp @clearcheck

@clearcheck
	%ci =l loadl %islot
	%cdone =w ceql %ci, %cap
	jnz %cdone, @returncheck, @clearone

@clearone
	%cioff =l shl %ci, 3
	%ciaddr =l add %stack, %cioff
	storel 0, %ciaddr
	%cinext =l add %ci, 1
	storel %cinext, %islot
	jmp @clearcheck

@returncheck
	%rd =l loadl %dslot
	%rempty =w ceql %rd, 0
	jnz %rempty, @returnzero, @returntop

@returnzero
	ret 0

@returntop
	%rtidx =l sub %rd, 1
	%rtoff =l shl %rtidx, 3
	%rtaddr =l add %stack, %rtoff
	%result =l loadl %rtaddr
	ret %result
}
