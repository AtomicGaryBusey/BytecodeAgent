export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %newlo, @upper %keeplo
	%hi =l phi @start %n, @lower %keephi, @upper %newhi
	%continue =w csltl %lo, %hi
	jnz %continue, @probe, @notfound
@probe
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @lower, @upper
@lower
	%newlo =l add %mid, 1
	%keephi =l copy %hi
	jmp @loop
@upper
	%keeplo =l copy %lo
	%newhi =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
