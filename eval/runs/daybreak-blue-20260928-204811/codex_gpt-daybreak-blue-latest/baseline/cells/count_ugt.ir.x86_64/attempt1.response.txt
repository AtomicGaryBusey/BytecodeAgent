export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %next_i
	%count =l phi @start 0, @body %next_count
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%value =l loadl %addr
	%greater =w cugtl %value, %t
	%increment =l extuw %greater
	%next_count =l add %count, %increment
	%next_i =l add %i, 1
	jmp @loop
@done
	ret %count
}
