export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @next %i2
	%crc =w phi @start -1, @next %crcnext
	%more =w csltl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%initial =w xor %crc, %b
	jmp @inner
@inner
	%j =w phi @byte 0, @step %j2
	%cur =w phi @byte %initial, @step %updated
	%bitsleft =w csltw %j, 8
	jnz %bitsleft, @bit, @next
@bit
	%low =w and %cur, 1
	%shifted =w shr %cur, 1
	jnz %low, @poly, @plain
@poly
	%withpoly =w xor %shifted, -306674912
	jmp @step
@plain
	jmp @step
@step
	%updated =w phi @poly %withpoly, @plain %shifted
	%j2 =w add %j, 1
	jmp @inner
@next
	%crcnext =w copy %cur
	%i2 =l add %i, 1
	jmp @outer
@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
