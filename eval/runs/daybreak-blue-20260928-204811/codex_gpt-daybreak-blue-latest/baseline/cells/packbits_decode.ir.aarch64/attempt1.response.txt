export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop

@loop
	%p =l phi @start %src, @lit_done %lit_pnext, @rep_done %rep_pnext
	%rem =l phi @start %n, @lit_done %lit_remnext, @rep_done %rep_remnext
	%op =l phi @start %out, @lit_done %lit_opnext, @rep_done %rep_opnext
	%written =l phi @start 0, @lit_done %lit_writtennext, @rep_done %rep_writtennext
	%empty =w ceql %rem, 0
	jnz %empty, @success, @header

@header
	%h =w loadsb %p
	%p1 =l add %p, 1
	%rem1 =l sub %rem, 1
	%nop =w ceqw %h, -128
	jnz %nop, @nop, @classify

@nop
	jmp @loop

@classify
	%negative =w csltw %h, 0
	jnz %negative, @repeat, @literal

@literal
	%hl =l extsw %h
	%lcnt =l add %hl, 1
	%ltrunc =w csltl %rem1, %lcnt
	jnz %ltrunc, @source_error, @literal_capacity

@literal_capacity
	%lroom =l sub %cap, %written
	%lover =w csltl %lroom, %lcnt
	jnz %lover, @lit_over_init, @lit_full_init

@lit_full_init
	jmp @lit_copy

@lit_copy
	%li =l phi @lit_full_init 0, @lit_copy_body %li1
	%lmore =w csltl %li, %lcnt
	jnz %lmore, @lit_copy_body, @lit_done

@lit_copy_body
	%lsp =l add %p1, %li
	%lb =w loadub %lsp
	%ldp =l add %op, %li
	storeb %lb, %ldp
	%li1 =l add %li, 1
	jmp @lit_copy

@lit_done
	%lit_pnext =l add %p1, %lcnt
	%lit_remnext =l sub %rem1, %lcnt
	%lit_opnext =l add %op, %lcnt
	%lit_writtennext =l add %written, %lcnt
	jmp @loop

@lit_over_init
	jmp @lit_over_copy

@lit_over_copy
	%loi =l phi @lit_over_init 0, @lit_over_body %loi1
	%lomore =w csltl %loi, %lroom
	jnz %lomore, @lit_over_body, @output_error

@lit_over_body
	%losp =l add %p1, %loi
	%lob =w loadub %losp
	%lodp =l add %op, %loi
	storeb %lob, %lodp
	%loi1 =l add %loi, 1
	jmp @lit_over_copy

@repeat
	%rtrunc =w ceql %rem1, 0
	jnz %rtrunc, @source_error, @repeat_valid

@repeat_valid
	%rb =w loadub %p1
	%p2 =l add %p1, 1
	%rem2 =l sub %rem1, 1
	%rhl =l extsw %h
	%rcnt =l sub 1, %rhl
	%rroom =l sub %cap, %written
	%rover =w csltl %rroom, %rcnt
	jnz %rover, @rep_over_init, @rep_full_init

@rep_full_init
	jmp @rep_copy

@rep_copy
	%ri =l phi @rep_full_init 0, @rep_copy_body %ri1
	%rmore =w csltl %ri, %rcnt
	jnz %rmore, @rep_copy_body, @rep_done

@rep_copy_body
	%rdp =l add %op, %ri
	storeb %rb, %rdp
	%ri1 =l add %ri, 1
	jmp @rep_copy

@rep_done
	%rep_pnext =l copy %p2
	%rep_remnext =l copy %rem2
	%rep_opnext =l add %op, %rcnt
	%rep_writtennext =l add %written, %rcnt
	jmp @loop

@rep_over_init
	jmp @rep_over_copy

@rep_over_copy
	%roi =l phi @rep_over_init 0, @rep_over_body %roi1
	%romore =w csltl %roi, %rroom
	jnz %romore, @rep_over_body, @output_error

@rep_over_body
	%rodp =l add %op, %roi
	storeb %rb, %rodp
	%roi1 =l add %roi, 1
	jmp @rep_over_copy

@success
	ret %written

@source_error
	ret -2

@output_error
	ret -1
}
