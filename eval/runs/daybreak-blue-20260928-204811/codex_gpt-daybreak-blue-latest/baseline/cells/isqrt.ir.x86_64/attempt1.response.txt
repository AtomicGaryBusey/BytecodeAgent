export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @setlo %mid, @sethi %lo
	%hi =l phi @start 4294967296, @setlo %hi, @sethi %mid
	%next =l add %lo, 1
	%cont =w cultl %next, %hi
	jnz %cont, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%quot =l udiv %n, %mid
	%fits =w culel %mid, %quot
	jnz %fits, @setlo, @sethi
@setlo
	jmp @loop
@sethi
	jmp @loop
@done
	ret %lo
}
