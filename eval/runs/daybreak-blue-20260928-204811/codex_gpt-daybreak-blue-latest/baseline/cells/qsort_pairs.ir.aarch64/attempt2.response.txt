export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @return, @setup

@setup
	%islot =l alloc8 8
	%jslot =l alloc8 8
	storel 0, %islot
	%jinit =l sub %n, 1
	storel %jinit, %jslot
	%mid =l div %n, 2
	%moff =l mul %mid, 16
	%paddr =l add %a, %moff
	%pkey =l loadl %paddr
	%pvaddr =l add %paddr, 8
	%pval =l loadl %pvaddr
	jmp @scan_i

@scan_i
	%i0 =l loadl %islot
	%iin =w csltl %i0, %n
	jnz %iin, @scan_i_load, @scan_j

@scan_i_load
	%ioff0 =l mul %i0, 16
	%iaddr0 =l add %a, %ioff0
	%ikey0 =l loadl %iaddr0
	%iklt =w csltl %ikey0, %pkey
	jnz %iklt, @inc_i, @scan_i_key_ge

@scan_i_key_ge
	%ikgt =w csgtl %ikey0, %pkey
	jnz %ikgt, @scan_j, @scan_i_value

@scan_i_value
	%ivaddr0 =l add %iaddr0, 8
	%ival0 =l loadl %ivaddr0
	%ivlt =w csltl %ival0, %pval
	jnz %ivlt, @inc_i, @scan_j

@inc_i
	%i1 =l add %i0, 1
	storel %i1, %islot
	jmp @scan_i

@scan_j
	%j0 =l loadl %jslot
	%jneg =w csltl %j0, 0
	jnz %jneg, @check, @scan_j_load

@scan_j_load
	%joff0 =l mul %j0, 16
	%jaddr0 =l add %a, %joff0
	%jkey0 =l loadl %jaddr0
	%jkgt =w csgtl %jkey0, %pkey
	jnz %jkgt, @dec_j, @scan_j_key_le

@scan_j_key_le
	%jklt =w csltl %jkey0, %pkey
	jnz %jklt, @check, @scan_j_value

@scan_j_value
	%jvaddr0 =l add %jaddr0, 8
	%jval0 =l loadl %jvaddr0
	%jvgt =w csgtl %jval0, %pval
	jnz %jvgt, @dec_j, @check

@dec_j
	%j1 =l sub %j0, 1
	storel %j1, %jslot
	jmp @scan_j

@check
	%ic =l loadl %islot
	%jc =l loadl %jslot
	%ordered =w cslel %ic, %jc
	jnz %ordered, @swap, @recurse

@swap
	%ioff1 =l mul %ic, 16
	%ia =l add %a, %ioff1
	%iva =l add %ia, 8
	%joff1 =l mul %jc, 16
	%ja =l add %a, %joff1
	%jva =l add %ja, 8
	%ik =l loadl %ia
	%iv =l loadl %iva
	%jk =l loadl %ja
	%jv =l loadl %jva
	storel %jk, %ia
	storel %jv, %iva
	storel %ik, %ja
	storel %iv, %jva
	%inext =l add %ic, 1
	%jnext =l sub %jc, 1
	storel %inext, %islot
	storel %jnext, %jslot
	jmp @scan_i

@recurse
	%jfinal =l loadl %jslot
	%leftn =l add %jfinal, 1
	%rl =l call $qsort_pairs(l %a, l %leftn)
	%ifinal =l loadl %islot
	%rightn =l sub %n, %ifinal
	%rightoff =l mul %ifinal, 16
	%righta =l add %a, %rightoff
	%rr =l call $qsort_pairs(l %righta, l %rightn)
	jmp @return

@return
	ret 0
}
