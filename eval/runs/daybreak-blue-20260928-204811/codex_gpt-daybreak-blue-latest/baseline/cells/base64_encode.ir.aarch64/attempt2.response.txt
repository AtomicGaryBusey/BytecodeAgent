export function l $base64_encode(l %src, l %n, l %out) {
@start
	%rounded =l add %n, 2
	%groups =l div %rounded, 3
	%result =l mul %groups, 4
	jmp @outer

@outer
	%i =l phi @start 0, @groupdone %inext
	%o =l phi @start 0, @groupdone %onext
	%more =w csltl %i, %n
	jnz %more, @load0, @done

@load0
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%has1 =w csltl 1, %rem
	jnz %has1, @load1, @one

@load1
	%i1 =l add %i, 1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	%b1s =l shl %b1, 8
	%v1 =l or %v0, %b1s
	%has2 =w csltl 2, %rem
	jnz %has2, @load2, @two

@load2
	%i2 =l add %i, 2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @three

@one
	jmp @ready

@two
	jmp @ready

@three
	jmp @ready

@ready
	%val =l phi @one %v0, @two %v1, @three %v2
	%valid =l phi @one 2, @two 3, @three 4
	jmp @chars

@chars
	%j =l phi @ready 0, @next %jnext
	%char_more =w csltl %j, 4
	jnz %char_more, @charcheck, @groupdone

@charcheck
	%opos =l add %o, %j
	%dst =l add %out, %opos
	%isvalid =w csltl %j, %valid
	jnz %isvalid, @index, @padding

@padding
	storeb 61, %dst
	jmp @next

@index
	%j6 =l mul %j, 6
	%shift =l sub 18, %j6
	%shifted =l shr %val, %shift
	%idx =l and %shifted, 63
	%lt26 =w csltl %idx, 26
	jnz %lt26, @mapupper, @check52

@mapupper
	%cu =l add %idx, 65
	storeb %cu, %dst
	jmp @next

@check52
	%lt52 =w csltl %idx, 52
	jnz %lt52, @maplower, @check62

@maplower
	%cl0 =l sub %idx, 26
	%cl =l add %cl0, 97
	storeb %cl, %dst
	jmp @next

@check62
	%lt62 =w csltl %idx, 62
	jnz %lt62, @mapdigit, @check63

@mapdigit
	%cd0 =l sub %idx, 52
	%cd =l add %cd0, 48
	storeb %cd, %dst
	jmp @next

@check63
	%is62 =w ceql %idx, 62
	jnz %is62, @mapplus, @mapslash

@mapplus
	storeb 43, %dst
	jmp @next

@mapslash
	storeb 47, %dst
	jmp @next

@next
	%jnext =l add %j, 1
	jmp @chars

@groupdone
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @outer

@done
	ret %result
}
