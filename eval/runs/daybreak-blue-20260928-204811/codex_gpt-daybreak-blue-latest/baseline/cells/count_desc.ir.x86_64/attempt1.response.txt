export function l $count_desc(l %a, l %n) {
@start
	%short =w csltl %n, 2
	jnz %short, @zero, @init
@zero
	ret 0
@init
	%first =l loadl %a
	%nextp =l add %a, 8
	%remaining =l sub %n, 1
	jmp @loop
@loop
	%p =l phi @init %nextp, @loop %p2
	%prev =l phi @init %first, @loop %curr
	%rem =l phi @init %remaining, @loop %rem2
	%count =l phi @init 0, @loop %count2
	%curr =l loadl %p
	%desc =w csltl %curr, %prev
	%inc =l extuw %desc
	%count2 =l add %count, %inc
	%p2 =l add %p, 8
	%rem2 =l sub %rem, 1
	%more =w cnel %rem2, 0
	jnz %more, @loop, @done
@done
	ret %count2
}
