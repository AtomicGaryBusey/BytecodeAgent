export function l $bmin_empty(l %a, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @empty, @init
@empty
	ret 9223372036854775807
@init
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @init 1, @latch %nexti
	%min =l phi @init %first, @latch %nextmin
	%done =w ceql %i, %n
	jnz %done, @return, @body
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%less =w csltl %value, %min
	jnz %less, @replace, @keep
@replace
	jmp @latch
@keep
	jmp @latch
@latch
	%nextmin =l phi @replace %value, @keep %min
	%nexti =l add %i, 1
	jmp @loop
@return
	ret %min
}
