export function l $collatz_max(l %n) {
@start
	%pk =l alloc8 8
	%px =l alloc8 8
	%psteps =l alloc8 8
	%pmax =l alloc8 8
	storel 1, %pk
	storel 0, %pmax
	jmp @outer

@outer
	%k =l loadl %pk
	%past =w csltl %n, %k
	jnz %past, @done, @begin

@begin
	storel %k, %px
	storel 0, %psteps
	jmp @inner

@inner
	%x =l loadl %px
	%isone =w ceql %x, 1
	jnz %isone, @finish, @step

@step
	%bit =l and %x, 1
	%iseven =w ceql %bit, 0
	jnz %iseven, @even, @odd

@even
	%half =l shr %x, 1
	storel %half, %px
	jmp @count

@odd
	%triple =l mul %x, 3
	%nextx =l add %triple, 1
	storel %nextx, %px
	jmp @count

@count
	%steps =l loadl %psteps
	%steps1 =l add %steps, 1
	storel %steps1, %psteps
	jmp @inner

@finish
	%finalsteps =l loadl %psteps
	%max =l loadl %pmax
	%greater =w csltl %max, %finalsteps
	jnz %greater, @update, @advance

@update
	storel %finalsteps, %pmax
	jmp @advance

@advance
	%oldk =l loadl %pk
	%newk =l add %oldk, 1
	storel %newk, %pk
	jmp @outer

@done
	%result =l loadl %pmax
	ret %result
}
