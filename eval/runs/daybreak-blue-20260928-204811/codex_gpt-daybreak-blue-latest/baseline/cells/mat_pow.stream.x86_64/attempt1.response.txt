mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
cld
mov rbx, rdi
mov rbp, rsi
mov r12, rdx
mov r13, rcx
mov r14, r8
mov rax, rbp
imul rax, rbp
lea r15, [r14 + rax*8]
xor r9, r9
.Linit:
cmp r9, rax
jge .Lidentity
mov r10, qword ptr [rbx + r9*8]
mov qword ptr [r14 + r9*8], r10
mov qword ptr [r15 + r9*8], 0
mov qword ptr [r13 + r9*8], 0
inc r9
jmp .Linit
.Lidentity:
xor r9, r9
.Lidentity_loop:
cmp r9, rbp
jge .Lpower_test
mov rcx, r9
imul rcx, rbp
add rcx, r9
mov qword ptr [r13 + rcx*8], 1
inc r9
jmp .Lidentity_loop
.Lpower_test:
test r12, r12
jz .Lcleanup
xor rdi, rdi
.Lrow:
cmp rdi, rbp
jge .Lcopy_product
mov r9, rdi
imul r9, rbp
xor rsi, rsi
.Lcolumn:
cmp rsi, rbp
jge .Lnext_row
xor r10, r10
xor rdx, rdx
.Ldot:
cmp rdx, rbp
jge .Lstore_product
lea r11, [r9 + rdx]
mov rax, qword ptr [r13 + r11*8]
mov rcx, rdx
imul rcx, rbp
add rcx, rsi
imul rax, qword ptr [r14 + rcx*8]
add r10, rax
inc rdx
jmp .Ldot
.Lstore_product:
lea rcx, [r9 + rsi]
mov qword ptr [r15 + rcx*8], r10
inc rsi
jmp .Lcolumn
.Lnext_row:
inc rdi
jmp .Lrow
.Lcopy_product:
mov rax, rbp
imul rax, rbp
xor r9, r9
.Lcopy_loop:
cmp r9, rax
jge .Lnext_power
mov r10, qword ptr [r15 + r9*8]
mov qword ptr [r13 + r9*8], r10
inc r9
jmp .Lcopy_loop
.Lnext_power:
dec r12
jmp .Lpower_test
.Lcleanup:
mov rax, rbp
imul rax, rbp
add rax, rax
xor r9, r9
.Lzero_tmp:
cmp r9, rax
jge .Ldone
mov qword ptr [r14 + r9*8], 0
inc r9
jmp .Lzero_tmp
.Ldone:
xor eax, eax
cld
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
