export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%n2 =l mul %n, %n
	%tmpn =l mul %n2, 2
	jmp @zcheck

@zcheck
	%z =l phi @entry 0, @zbody %znext
	%zmore =w csltl %z, %tmpn
	jnz %zmore, @zbody, @initcheck

@zbody
	%zoff =l shl %z, 3
	%zaddr =l add %tmp, %zoff
	storel 0, %zaddr
	%znext =l add %z, 1
	jmp @zcheck

@initcheck
	%p =l phi @zcheck 0, @initnext %pnext
	%pmore =w csltl %p, %n2
	jnz %pmore, @initbody, @itercheck

@initbody
	%prow =l div %p, %n
	%pcol =l rem %p, %n
	%peq =w ceql %prow, %pcol
	%poff =l shl %p, 3
	%paddr =l add %out, %poff
	jnz %peq, @initone, @initzero

@initone
	storel 1, %paddr
	jmp @initnext

@initzero
	storel 0, %paddr
	jmp @initnext

@initnext
	%pnext =l add %p, 1
	jmp @initcheck

@itercheck
	%e =l phi @initcheck 0, @iteradvance %enext
	%emore =w csltl %e, %k
	jnz %emore, @icheck, @done

@icheck
	%i =l phi @itercheck 0, @iend %inext
	%imore =w csltl %i, %n
	jnz %imore, @jcheck, @copycheck

@jcheck
	%j =l phi @icheck 0, @jfinish %jnext
	%jmore =w csltl %j, %n
	jnz %jmore, @tcheck, @iend

@tcheck
	%t =l phi @jcheck 0, @tbody %tnext
	%sum =l phi @jcheck 0, @tbody %sumnext
	%tmore =w csltl %t, %n
	jnz %tmore, @tbody, @jfinish

@tbody
	%ait =l mul %i, %n
	%aitidx =l add %ait, %t
	%aitoff =l shl %aitidx, 3
	%aitaddr =l add %out, %aitoff
	%x =l loadl %aitaddr
	%btj =l mul %t, %n
	%btjidx =l add %btj, %j
	%btjoff =l shl %btjidx, 3
	%btjaddr =l add %a, %btjoff
	%y =l loadl %btjaddr
	%prod =l mul %x, %y
	%sumnext =l add %sum, %prod
	%tnext =l add %t, 1
	jmp @tcheck

@jfinish
	%ijbase =l mul %i, %n
	%ijidx =l add %ijbase, %j
	%ijoff =l shl %ijidx, 3
	%ijaddr =l add %tmp, %ijoff
	storel %sum, %ijaddr
	%jnext =l add %j, 1
	jmp @jcheck

@iend
	%inext =l add %i, 1
	jmp @icheck

@copycheck
	%q =l phi @icheck 0, @copybody %qnext
	%qmore =w csltl %q, %n2
	jnz %qmore, @copybody, @iteradvance

@copybody
	%qoff =l shl %q, 3
	%qtaddr =l add %tmp, %qoff
	%qval =l loadl %qtaddr
	%qoaddr =l add %out, %qoff
	storel %qval, %qoaddr
	storel 0, %qtaddr
	%qnext =l add %q, 1
	jmp @copycheck

@iteradvance
	%enext =l add %e, 1
	jmp @itercheck

@done
	ret 0
}
