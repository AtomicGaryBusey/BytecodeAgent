export function l $max_byte(l %p, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @next %inext
	%max =l phi @entry 0, @next %maxnext
	%more =w csltl %i, %n
	jnz %more, @read, @done
@read
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w csgtl %b, %max
	jnz %greater, @useb, @keep
@useb
	jmp @next
@keep
	jmp @next
@next
	%maxnext =l phi @useb %b, @keep %max
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
