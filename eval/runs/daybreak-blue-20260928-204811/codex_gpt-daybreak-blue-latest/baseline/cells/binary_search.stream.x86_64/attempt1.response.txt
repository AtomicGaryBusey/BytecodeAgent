binary_search:
    xor ecx, ecx
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jge .Lnot_found
    mov rax, r8
    sub rax, rcx
    shr rax, 1
    add rax, rcx
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    jl .Lless
    jg .Lgreater
    ret
.Lless:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lgreater:
    mov r8, rax
    jmp .Lloop
.Lnot_found:
    mov rax, -1
    ret
