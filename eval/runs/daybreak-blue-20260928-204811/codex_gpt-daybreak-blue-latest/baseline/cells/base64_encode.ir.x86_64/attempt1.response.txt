export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%zero =l copy 0
	jmp @outer

@outer
	%pos =l phi @entry %zero, @done %npos
	%written =l phi @entry %zero, @done %nwritten
	%more =w csltl %pos, %n
	jnz %more, @group, @return

@group
	%p0 =l add %src, %pos
	%b0 =l loadub %p0
	%rem =l sub %n, %pos
	%has2 =w csltl 1, %rem
	jnz %has2, @load2, @one

@one
	%b1a =l copy 0
	%b2a =l copy 0
	%valida =l copy 2
	jmp @build

@load2
	%p1 =l add %p0, 1
	%b1b =l loadub %p1
	%has3 =w csltl 2, %rem
	jnz %has3, @load3, @two

@two
	%b2b =l copy 0
	%validb =l copy 3
	jmp @build

@load3
	%p2 =l add %p0, 2
	%b2c =l loadub %p2
	%validc =l copy 4
	jmp @build

@build
	%b1 =l phi @one %b1a, @two %b1b, @load3 %b1b
	%b2 =l phi @one %b2a, @two %b2b, @load3 %b2c
	%valid =l phi @one %valida, @two %validb, @load3 %validc
	%v0 =l shl %b0, 16
	%v1 =l shl %b1, 8
	%v01 =l or %v0, %v1
	%value =l or %v01, %b2
	jmp @emit

@emit
	%j =l phi @build %zero, @mapped %jnext
	%real =w csltl %j, %valid
	jnz %real, @index, @padding

@padding
	%pad =l copy 61
	jmp @mapped

@index
	%j6 =l mul %j, 6
	%shift =l sub 18, %j6
	%shifted =l shr %value, %shift
	%idx =l and %shifted, 63
	%lt26 =w csltl %idx, 26
	jnz %lt26, @upper, @test52

@upper
	%cu =l add %idx, 65
	jmp @mapped

@test52
	%lt52 =w csltl %idx, 52
	jnz %lt52, @lower, @test62

@lower
	%cl =l add %idx, 71
	jmp @mapped

@test62
	%lt62 =w csltl %idx, 62
	jnz %lt62, @digit, @testplus

@digit
	%cd =l sub %idx, 4
	jmp @mapped

@testplus
	%is62 =w ceql %idx, 62
	jnz %is62, @plus, @slash

@plus
	%cp =l copy 43
	jmp @mapped

@slash
	%cs =l copy 47
	jmp @mapped

@mapped
	%ch =l phi @padding %pad, @upper %cu, @lower %cl, @digit %cd, @plus %cp, @slash %cs
	%off =l add %written, %j
	%op =l add %out, %off
	storeb %ch, %op
	%jnext =l add %j, 1
	%again =w csltl %jnext, 4
	jnz %again, @emit, @done

@done
	%npos =l add %pos, 3
	%nwritten =l add %written, 4
	jmp @outer

@return
	ret %written
}
