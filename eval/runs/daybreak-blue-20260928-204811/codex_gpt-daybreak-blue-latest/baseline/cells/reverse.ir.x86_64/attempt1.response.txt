export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	%more =w csltl %i, %j
	jnz %more, @swap, @done
@swap
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%ip =l add %a, %ioff
	%jp =l add %a, %joff
	%x =l loadl %ip
	%y =l loadl %jp
	storel %y, %ip
	storel %x, %jp
	%i =l add %i, 1
	%j =l sub %j, 1
	%more2 =w csltl %i, %j
	jnz %more2, @swap, @done
@done
	ret 0
}
