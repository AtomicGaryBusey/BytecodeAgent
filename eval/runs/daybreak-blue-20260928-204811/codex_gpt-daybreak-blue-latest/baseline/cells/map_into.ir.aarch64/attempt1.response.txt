export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%has =w csltl 0, %n
	jnz %has, @loop, @done
@loop
	%i =l phi @start 0, @next %i1
	%count =l phi @start 0, @next %count1
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%less =w csltl %r, %x
	%inc =l extuw %less
	%count1 =l add %count, %inc
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @next, @done
@next
	jmp @loop
@done
	%result =l phi @start 0, @loop %count1
	ret %result
}
