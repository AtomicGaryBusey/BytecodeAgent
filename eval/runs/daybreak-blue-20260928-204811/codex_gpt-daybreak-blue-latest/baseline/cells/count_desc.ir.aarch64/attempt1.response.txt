export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %next_i
	%count =l phi @init 0, @body %next_count
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%left =l loadl %ptr
	%next_ptr =l add %ptr, 8
	%right =l loadl %next_ptr
	%descending =w csltl %right, %left
	%increment =l extuw %descending
	%next_count =l add %count, %increment
	%next_i =l add %i, 1
	jmp @loop
@done
	ret %count
}
