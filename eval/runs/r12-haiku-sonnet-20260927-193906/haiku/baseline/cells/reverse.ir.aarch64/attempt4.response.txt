export function l $reverse(l %a, l %n) {
@start
    %n_minus_1 = subl %n, 1
    jmp @loop
    
@loop
    %left = phi l 0 @start, %left_inc @body
    %right = phi l %n_minus_1 @start, %right_dec @body
    %cond = csltl %left, %right
    jnz %cond @body @done
    
@body
    %left_offset = mull %left, 8
    %left_addr = addl %a, %left_offset
    %right_offset = mull %right, 8
    %right_addr = addl %a, %right_offset
    
    %lval = loadl %left_addr
    %rval = loadl %right_addr
    
    storel %rval, %left_addr
    storel %lval, %right_addr
    
    %left_inc = addl %left, 1
    %right_dec = subl %right, 1
    jmp @loop
    
@done
    ret 0
}
