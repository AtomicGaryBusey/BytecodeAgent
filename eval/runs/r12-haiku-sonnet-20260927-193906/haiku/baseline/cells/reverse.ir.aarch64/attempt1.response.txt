export function l $reverse(l %a, l %n) {
@start
    %left = copy 0
    %right = sub %n, 1
    jmp @check
    
@check
    %cond = csltl %left, %right
    jnz %cond, @body, @done
    
@body
    %left_offset = mul %left, 8
    %left_addr = add %a, %left_offset
    %right_offset = mul %right, 8
    %right_addr = add %a, %right_offset
    
    %lval = loadl %left_addr
    %rval = loadl %right_addr
    
    storel %rval, %left_addr
    storel %lval, %right_addr
    
    %left = add %left, 1
    %right = sub %right, 1
    jmp @check
    
@done
    ret 0
}
