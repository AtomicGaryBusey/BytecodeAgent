export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
	jmp @check

@check
	%continue =w csltl %i, %n
	jnz %continue @load
	jmp @exit

@load
	%offset =l mull %i, 8
	%addr =l addl %a, %offset
	%val =l loadl %addr
	%is_zero =w ceql %val, 0
	jnz %is_zero @exit
	%i =l addl %i, 1
	jmp @check

@exit
	ret %i
}
