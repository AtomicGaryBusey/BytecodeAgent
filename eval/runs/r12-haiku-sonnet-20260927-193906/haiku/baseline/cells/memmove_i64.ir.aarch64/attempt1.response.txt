@entry
    %n_is_zero = ceql %n, 0
    jnz %n_is_zero, @return_zero

    jmp @copy_backward

@copy_backward

    jmp @back_loop

@back_loop

@back_continue
    jmp @back_loop

@copy_forward
    %fwd_start = 0
    jmp @fwd_loop

@fwd_loop
    %i_f = phi @copy_forward %fwd_start, @fwd_continue %i_f_next

@fwd_continue

@return_zero
    ret 0
}
