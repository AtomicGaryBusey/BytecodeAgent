export function l $binary_search(l %a, l %n, l %key) {
@start
    %zero = copy 0
    %neg_one = copy -1
    %one = copy 1
    %two = copy 2
    %eight = copy 8
    
    %n_is_zero = ceql %n, %zero
    jnz %n_is_zero, @return_not_found
    
    %left_init = copy %zero
    %right_init = sub %n, %one
    jmp @loop_header
    
@loop_header
    %left = phi @start %left_init, @search_left %left_left, @search_right %left_right
    %right = phi @start %right_init, @search_left %right_left, @search_right %right_right
    
    %left_le_right = csled %left, %right
    jnz %left_le_right, @loop_body
    jmp @return_not_found
    
@loop_body
    %sum = add %left, %right
    %mid = sdivl %sum, %two
    
    %mid_offset = mul %mid, %eight
    %mid_ptr = add %a, %mid_offset
    %mid_val = loadl %mid_ptr
    
    %is_equal = ceql %mid_val, %key
    jnz %is_equal, @return_mid
    
    %is_less = csltl %mid_val, %key
    jnz %is_less, @search_right
    jmp @search_left
    
@search_left
    %right_left = sub %mid, %one
    %left_left = copy %left
    jmp @loop_header
    
@search_right
    %left_right = add %mid, %one
    %right_right = copy %right
    jmp @loop_header
    
@return_mid
    ret %mid
    
@return_not_found
    ret %neg_one
}
