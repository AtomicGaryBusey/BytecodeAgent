export function l $binary_search(l a, l n, l key) {
@entry
    %left0 = copy 0
    %right0 = subl %n, 1
    jmp @loop

@loop
    %left = phi @entry %left0, @go_left %left1, @go_right %left2
    %right = phi @entry %right0, @go_left %right1, @go_right %right2
    %cmp = cslel %left, %right
    jnz %cmp, @body
    ret -1

@body
    %sum = addl %left, %right
    %mid = sdivl %sum, 2
    %offset = mull %mid, 8
    %ptr = addl %a, %offset
    %val = loadl %ptr
    %eq = ceql %val, %key
    jnz %eq, @found
    %lt = csltl %val, %key
    jnz %lt, @go_right
    jmp @go_left

@go_left
    %right1 = subl %mid, 1
    %left1 = copy %left
    jmp @loop

@go_right
    %left2 = addl %mid, 1
    %right2 = copy %right
    jmp @loop

@found
    ret %mid
}
