export function l $gcd(l %a, l %b) {
@entry
    jmp @loop

@loop
    %a_loop = phi %a, %a_new
    %b_loop = phi %b, %b_new
    %b_zero = ceql %b_loop, 0
    jnz %b_zero, @done, @loop_body

@loop_body
    %rem = rem %a_loop, %b_loop
    %a_new = copy %b_loop
    %b_new = copy %rem
    jmp @loop

@done
    ret %a_loop
}
