memmove_i64:
    xor rax, rax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    je .Ldone
    jl .Lforward
.Lbackward:
    mov rcx, rdx
.Lbackward_loop:
    test rcx, rcx
    jz .Ldone
    dec rcx
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    jmp .Lbackward_loop
.Lforward:
    xor rcx, rcx
.Lforward_loop:
    cmp rcx, rdx
    jge .Ldone
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    jmp .Lforward_loop
.Ldone:
    ret
