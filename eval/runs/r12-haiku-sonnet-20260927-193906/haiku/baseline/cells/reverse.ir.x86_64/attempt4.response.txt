export function l $reverse(l %a, l %n) {
@start
    %right0 = sub %n, 1
    jmp @loop

@loop
    %left = phi @start 0, @body %left_next
    %right = phi @start %right0, @body %right_next
    
    %cond = cslt %left, %right
    jnz %cond @body, @end

@body
    %left_offset = mul %left, 8
    %left_addr = add %a, %left_offset
    
    %right_offset = mul %right, 8
    %right_addr = add %a, %right_offset
    
    %left_val = loadl %left_addr
    %right_val = loadl %right_addr
    
    storel %right_val, %left_addr
    storel %left_val, %right_addr
    
    %left_next = add %left, 1
    %right_next = sub %right, 1
    
    jmp @loop

@end
    ret 0
}
