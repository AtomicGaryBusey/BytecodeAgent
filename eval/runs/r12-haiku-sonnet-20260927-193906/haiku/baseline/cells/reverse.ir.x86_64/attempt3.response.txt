export function l $reverse(l %a, l %n) {
@start
    %right0 = subl %n, 1
    jmp @loop

@loop
    %left = phi @start 0, @body %left_next
    %right = phi @start %right0, @body %right_next
    
    %cond = csltl %left, %right
    jnz %cond @body, @end

@body
    %left_offset = mull %left, 8
    %left_addr = addl %a, %left_offset
    
    %right_offset = mull %right, 8
    %right_addr = addl %a, %right_offset
    
    %left_val = loadl %left_addr
    %right_val = loadl %right_addr
    
    storel %right_val, %left_addr
    storel %left_val, %right_addr
    
    %left_next = addl %left, 1
    %right_next = subl %right, 1
    
    jmp @loop

@end
    ret 0
}
