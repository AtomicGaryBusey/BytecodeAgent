export function l $collatz_max(l %n) {
@entry
    %m0 = copy 0
    %k0 = copy 1
    jmp @for_check

@for_check
    %max = phi @entry %m0, @for_iter %max1
    %k = phi @entry %k0, @for_iter %k1
    %t0 = csle %k, %n
    jnz %t0, @collatz_begin, @exit

@collatz_begin
    %k_init = copy %k
    %s_init = copy 0
    jmp @collatz_loop

@collatz_loop
    %x = phi @collatz_begin %k_init, @collatz_even %x_even, @collatz_odd %x_odd
    %steps = phi @collatz_begin %s_init, @collatz_even %steps_inc, @collatz_odd %steps_inc
    %t1 = ceq %x, 1
    jnz %t1, @collatz_exit, @collatz_body

@collatz_body
    %t2 = and %x, 1
    %t3 = ceq %t2, 1
    jnz %t3, @collatz_odd, @collatz_even

@collatz_even
    %x_even = shr %x, 1
    %steps_inc = add %steps, 1
    jmp @collatz_loop

@collatz_odd
    %t4 = mul %x, 3
    %x_odd = add %t4, 1
    %steps_inc = add %steps, 1
    jmp @collatz_loop

@collatz_exit
    %t5 = csgt %steps, %max
    jnz %t5, @update_max, @no_update

@update_max
    %max1 = copy %steps
    jmp @for_iter

@no_update
    %max1 = copy %max
    jmp @for_iter

@for_iter
    %k1 = add %k, 1
    jmp @for_check

@exit
    ret %max
}
