export function l $collatz_max(l %n) {
@entry
    %max_steps = copy 0
    %k = copy 1

@main_loop_cond
    %k_le_n = cslel %k, %n
    jnz %k_le_n, @collatz_start, @return_result

@collatz_start
    %x = copy %k
    %steps = copy 0

@collatz_cond
    %x_eq_1 = ceql %x, 1
    jnz %x_eq_1, @collatz_end, @collatz_step

@collatz_step
    %x_and_1 = and %x, 1
    %is_even = ceql %x_and_1, 0
    jnz %is_even, @collatz_even, @collatz_odd

@collatz_even
    %x = shr %x, 1
    %steps = add %steps, 1
    %x_eq_1_even = ceql %x, 1
    jnz %x_eq_1_even, @collatz_end, @collatz_step

@collatz_odd
    %x = mul %x, 3
    %x = add %x, 1
    %steps = add %steps, 1
    %x_eq_1_odd = ceql %x, 1
    jnz %x_eq_1_odd, @collatz_end, @collatz_step

@collatz_end
    %steps_gt = csgtl %steps, %max_steps
    jnz %steps_gt, @update_max, @next_k

@update_max
    %max_steps = copy %steps
    %one = copy 1
    jnz %one, @next_k, @next_k

@next_k
    %k = add %k, 1
    %one2 = copy 1
    jnz %one2, @main_loop_cond, @main_loop_cond

@return_result
    ret %max_steps
}
