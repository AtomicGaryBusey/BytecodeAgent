export function l $collatz_max(l %n) {
@entry
    jmp @for_init

@for_init
    %m0 = add 0, 0
    %k0 = add 0, 1
    jmp @for_check

@for_check
    %max = phi @for_init %m0, @for_iter %max1
    %k = phi @for_init %k0, @for_iter %k1
    %t0 = csle %k, %n
    jnz %t0, @collatz_begin, @exit

@collatz_begin
    %x_init = copy %k
    %s_init = add 0, 0
    jmp @collatz_loop

@collatz_loop
    %x = phi @collatz_begin %x_init, @collatz_even %x_even, @collatz_odd %x_odd
    %steps = phi @collatz_begin %s_init, @collatz_even %steps_even, @collatz_odd %steps_odd
    %t1 = ceq %x, 1
    jnz %t1, @collatz_exit, @collatz_body

@collatz_body
    %t2 = and %x, 1
    %t3 = ceq %t2, 1
    jnz %t3, @collatz_odd, @collatz_even

@collatz_even
    %x_even = shr %x, 1
    %steps_even = add %steps, 1
    jmp @collatz_loop

@collatz_odd
    %t4 = mul %x, 3
    %x_odd = add %t4, 1
    %steps_odd = add %steps, 1
    jmp @collatz_loop

@collatz_exit
    %t5 = csgt %steps, %max
    jnz %t5, @update_max, @no_update

@update_max
    %max1 = copy %steps
    jmp @for_iter

@no_update
    %max1 = copy %max
    jmp @for_iter

@for_iter
    %k1 = add %k, 1
    jmp @for_check

@exit
    ret %max
}
