export function l $fib(l %n) {
@entry
    %z =w ceql %n, 0
    jnz %z, @z_ret
    jmp @check_one
@check_one
    %o =w ceql %n, 1
    jnz %o, @o_ret
    jmp @loop
@z_ret
    ret 0
@o_ret
    ret 1
@loop
    %a =l phi @check_one 0, @next %na
    %b =l phi @check_one 1, @next %nb
    %i =l phi @check_one 2, @next %ni
    %c =w csltl %i, %n
    jnz %c, @next
    jmp @exit
@next
    %s =l addl %a, %b
    %na =l copy %b
    %nb =l copy %s
    %ni =l addl %i, 1
    jmp @loop
@exit
    ret %b
}
