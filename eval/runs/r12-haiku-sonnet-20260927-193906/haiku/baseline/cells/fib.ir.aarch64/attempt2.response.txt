export function l $fib(l %n) {
@start
    %is_zero = ceql %n, 0
    jnz %is_zero, @ret_zero
    %is_one = ceql %n, 1
    jnz %is_one, @ret_one
    jmp @loop
    
@ret_zero
    ret 0
    
@ret_one
    ret 1
    
@loop
    %a = phi @start 0, @loop_body %b
    %b = phi @start 1, @loop_body %c
    %i = phi @start 2, @loop_body %i_next
    %cond = cslel %i, %n
    jnz %cond, @loop_body
    jmp @done
    
@loop_body
    %c = addl %a, %b
    %i_next = addl %i, 1
    jmp @loop
    
@done
    ret %b
}
