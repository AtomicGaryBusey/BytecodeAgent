export function l $fib(l %n) {
@start
    %zero_check =w ceql %n, 0
    jnz %zero_check, @ret_zero
    %one_check =w ceql %n, 1
    jnz %one_check, @ret_one
    jmp @loop
    
@ret_zero
    ret 0
    
@ret_one
    ret 1
    
@loop
    %a =l phi @start 0, @loop_body %a_next
    %b =l phi @start 1, @loop_body %b_next
    %i =l phi @start 2, @loop_body %i_next
    %should_continue =w cslel %i, %n
    jnz %should_continue, @loop_body
    jmp @done
    
@loop_body
    %sum =l addl %a, %b
    %i_next =l addl %i, 1
    %a_next =l copy %b
    %b_next =l copy %sum
    jmp @loop
    
@done
    ret %b
}
