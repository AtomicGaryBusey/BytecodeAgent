export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	%i = const l 0
	jmp @loop

@loop
	%cond = cultl %i, %n
	jnz %cond @check_zero @return_n

@check_zero
	%offset = mul l %i, 8
	%addr = add l %a, %offset
	%elem = loadl %addr
	%is_zero = ceql %elem, 0
	jnz %is_zero @return_i @increment

@increment
	%i = add l %i, 1
	jmp @loop

@return_i
	ret %i

@return_n
	ret %n
}
