export function l $binary_search(l %a, l %n, l %key) {
@start
    %is_zero = ceql %n, constl 0
    jnz %is_zero, @notfound
    
    %zero = constl 0
    %r = sub %n, constl 1
    jmp @loop
    
@loop
    %l = phi @start %zero, @loop_left %l_next, @loop_right %l_next2
    %r_val = phi @start %r, @loop_left %r_next, @loop_right %r_next2
    
    %gt = cgtl %l, %r_val
    jnz %gt, @notfound
    
    %sum = add %l, %r_val
    %m = divl %sum, constl 2
    %addr_offset = mul %m, constl 8
    %addr = add %a, %addr_offset
    %elem = loadl %addr
    
    %found_eq = ceql %elem, %key
    jnz %found_eq, @found
    
    %is_less = cltl %elem, %key
    jnz %is_less, @loop_left
    jmp @loop_right
    
@loop_left
    %l_next = add %m, constl 1
    %r_next = copy %r_val
    jmp @loop
    
@loop_right
    %l_next2 = copy %l
    %r_next2 = sub %m, constl 1
    jmp @loop
    
@found
    ret %m
    
@notfound
    %neg_one = constl -1
    ret %neg_one
}
