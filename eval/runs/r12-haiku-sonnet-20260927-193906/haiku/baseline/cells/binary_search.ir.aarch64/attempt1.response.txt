export function l $binary_search(l %a, l %n, l %key) {
@start
    %cond_n = clel %n, constl 0
    jnz %cond_n, @notfound
    
    %left_init = constl 0
    %right_init = sub %n, constl 1
    jmp @loop_header
    
@loop_header
    %left = phi @start %left_init, @merge_next %left_next
    %right = phi @start %right_init, @merge_next %right_next
    
    %cond_loop = clel %left, %right
    jnz %cond_loop, @loop_body
    jmp @notfound
    
@loop_body
    %sum = add %left, %right
    %mid = divl %sum, constl 2
    %offset = mul %mid, constl 8
    %ptr = add %a, %offset
    %val = loadl %ptr
    
    %cond_eq = ceql %val, %key
    jnz %cond_eq, @found
    
    %cond_lt = cltl %val, %key
    jnz %cond_lt, @branch_lt
    jmp @branch_ge
    
@branch_lt
    %left_next_lt = add %mid, constl 1
    %right_next_lt = copy %right
    jmp @merge_next
    
@branch_ge
    %left_next_ge = copy %left
    %right_next_ge = sub %mid, constl 1
    jmp @merge_next
    
@merge_next
    %left_next = phi @branch_lt %left_next_lt, @branch_ge %left_next_ge
    %right_next = phi @branch_lt %right_next_lt, @branch_ge %right_next_ge
    jmp @loop_header
    
@found
    ret %mid
    
@notfound
    ret constl -1
}
