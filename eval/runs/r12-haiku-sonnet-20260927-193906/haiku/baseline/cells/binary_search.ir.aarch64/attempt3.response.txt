export function l $binary_search(l a, l n, l key) {
@start
    %cmp = ceql %n, constl 0
    jnz %cmp, @notfound
    
    %init_left = constl 0
    %init_right = sub %n, constl 1
    jmp @loop_check
    
@loop_check
    %left = phi @start %init_left, @right_search %right_left, @left_search %left_left
    %right = phi @start %init_right, @right_search %right_right, @left_search %left_right
    
    %cond = cgtl %left, %right
    jnz %cond, @notfound
    jmp @loop_body
    
@loop_body
    %sum = add %left, %right
    %mid = divl %sum, constl 2
    
    %addr_off = mul %mid, constl 8
    %addr = add %a, %addr_off
    %elem = loadl %addr
    
    %eq = ceql %elem, %key
    jnz %eq, @found
    
    %lt = cltl %elem, %key
    jnz %lt, @right_search
    jmp @left_search
    
@right_search
    %right_left = add %mid, constl 1
    %right_right = %right
    jmp @loop_check
    
@left_search
    %left_left = %left
    %left_right = sub %mid, constl 1
    jmp @loop_check
    
@found
    ret %mid
    
@notfound
    ret constl -1
}
