export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %cmp_n = csgtl %n, $0
  jnz %cmp_n, @check_direction, @done
  
@check_direction
  %need_backward = csgtl %dst, %src
  jnz %need_backward, @backward, @forward
  
@forward
  jmp @forward_loop
  
@forward_loop
  %i = phi $0, @forward, %i_next, @forward_body
  %continue = csltl %i, %n
  jnz %continue, @forward_body, @done
  
@forward_body
  %offset = mull %i, $8
  %src_addr = addl %src, %offset
  %dst_addr = addl %dst, %offset
  %val = loadl %src_addr
  storel %val, %dst_addr
  %i_next = addl %i, $1
  jmp @forward_loop
  
@backward
  %i_init = subl %n, $1
  jmp @backward_loop
  
@backward_loop
  %i = phi %i_init, @backward, %i_next, @backward_body
  %continue = csgtl %i, $-1
  jnz %continue, @backward_body, @done
  
@backward_body
  %offset = mull %i, $8
  %src_addr = addl %src, %offset
  %dst_addr = addl %dst, %offset
  %val = loadl %src_addr
  storel %val, %dst_addr
  %i_next = subl %i, $1
  jmp @backward_loop
  
@done
  ret $0
}
