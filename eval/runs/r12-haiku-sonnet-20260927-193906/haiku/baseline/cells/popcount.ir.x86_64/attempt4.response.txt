export function l $popcount(l %x) {
@start
    %count =l copy 0
    %cond =w cnel %x, 0
    jnz %cond, @loop, @return

@loop
    %x_loop =l phi @start %x, @loop %x_new
    %count_loop =l phi @start %count, @loop %count_new
    %x_minus_1 =l sub l %x_loop, 1
    %x_new =l and l %x_loop, %x_minus_1
    %count_new =l add l %count_loop, 1
    %cond_new =w cnel %x_new, 0
    jnz %cond_new, @loop, @return

@return
    %result =l phi @start %count, @loop %count_new
    ret l %result
}
