export function l $isqrt(l %n) {
@entry
    jmp @loop_init

@loop_init
    %low = copy l 0
    %high = copy l 4294967295
    jmp @loop_cond

@loop_cond
    %cond = culel %low %high
    jnz %cond @loop_body @loop_end

@loop_body
    %diff = subl %high %low
    %half = divl %diff l 2
    %mid = addl %low %half
    %sq = mull %mid %mid
    %cmp = culel %sq %n
    jnz %cmp @loop_update_low @loop_update_high

@loop_update_low
    %new_low = addl %mid l 1
    %new_high = copy l %high
    jmp @loop_update_end

@loop_update_high
    %new_low = copy l %low
    %new_high = subl %mid l 1
    jmp @loop_update_end

@loop_update_end
    %low = copy l %new_low
    %high = copy l %new_high
    jmp @loop_cond

@loop_end
    ret l %high
}
