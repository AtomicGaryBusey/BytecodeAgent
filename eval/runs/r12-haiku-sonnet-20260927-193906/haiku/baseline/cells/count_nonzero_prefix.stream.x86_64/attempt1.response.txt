count_nonzero_prefix:
    xor rax, rax           # rax = 0 (index counter)
    test rsi, rsi          # if n == 0
    jz .Lreturn            # return 0
.Lloop:
    cmp qword ptr [rdi + rax*8], 0  # compare a[rax] with 0
    je .Lreturn            # if equal to 0, return current index
    inc rax                # rax++
    cmp rax, rsi           # compare rax with n
    jl .Lloop              # if rax < n, continue
.Lreturn:
    ret
