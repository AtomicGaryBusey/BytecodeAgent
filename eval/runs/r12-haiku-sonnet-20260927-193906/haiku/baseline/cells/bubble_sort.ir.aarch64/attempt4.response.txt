export function l $bubble_sort(l %a, l %n) {
@entry
  jmp @loop_i_check

@loop_i_check
  %i =l phi 0 %i_next
  %i_lt_n =w csltl %i, %n
  jnz %i_lt_n, @loop_j_init
  ret 0

@loop_j_init
  %n_minus_1 =l subl %n, 1
  %limit =l subl %n_minus_1, %i
  jmp @loop_j_check

@loop_j_check
  %j =l phi 0 %j_next
  %j_lt_limit =w csltl %j, %limit
  jnz %j_lt_limit, @compare
  jmp @loop_i_update

@compare
  %j_offset =l mull %j, 8
  %a_j_addr =l addl %a, %j_offset
  %a_j =l loadl %a_j_addr
  %j_plus_1 =l addl %j, 1
  %j_plus_1_offset =l mull %j_plus_1, 8
  %a_j_plus_1_addr =l addl %a, %j_plus_1_offset
  %a_j_plus_1 =l loadl %a_j_plus_1_addr
  %a_j_gt_a_j_plus_1 =w csgtl %a_j, %a_j_plus_1
  jnz %a_j_gt_a_j_plus_1, @swap
  jmp @loop_j_update

@swap
  storel %a_j_plus_1, %a_j_addr
  storel %a_j, %a_j_plus_1_addr
  jmp @loop_j_update

@loop_j_update
  %j_next =l addl %j, 1
  jmp @loop_j_check

@loop_i_update
  %i_next =l addl %i, 1
  jmp @loop_i_check
}
