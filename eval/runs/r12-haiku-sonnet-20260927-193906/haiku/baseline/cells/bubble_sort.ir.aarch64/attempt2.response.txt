export function l $bubble_sort(l %a, l %n) {
@entry
  %zero = constl 0
  %one = constl 1
  %eight = constl 8
  jmp @loop_i(%zero)

@loop_i(%i)
  %i_cmp = csltl %i, %n
  jnz %i_cmp, @loop_j_start(%i)
  ret 0

@loop_j_start(%i)
  %n_minus_1 = subl %n, %one
  %limit = subl %n_minus_1, %i
  jmp @loop_j(%i, %zero, %limit)

@loop_j(%i, %j, %limit)
  %j_cmp = csltl %j, %limit
  jnz %j_cmp, @compare(%i, %j)
  jmp @next_i(%i)

@compare(%i, %j)
  %j_offset = mull %j, %eight
  %a_j_addr = addl %a, %j_offset
  %a_j = loadl %a_j_addr
  %j_plus_1 = addl %j, %one
  %j_plus_1_offset = mull %j_plus_1, %eight
  %a_j_plus_1_addr = addl %a, %j_plus_1_offset
  %a_j_plus_1 = loadl %a_j_plus_1_addr
  %cmp = csgtl %a_j, %a_j_plus_1
  jnz %cmp, @swap(%i, %j, %limit, %a_j, %a_j_plus_1, %a_j_addr, %a_j_plus_1_addr)
  jmp @next_j(%i, %j, %limit)

@swap(%i, %j, %limit, %a_j, %a_j_plus_1, %a_j_addr, %a_j_plus_1_addr)
  storel %a_j, %a_j_plus_1_addr
  storel %a_j_plus_1, %a_j_addr
  jmp @next_j(%i, %j, %limit)

@next_j(%i, %j, %limit)
  %j_next = addl %j, %one
  jmp @loop_j(%i, %j_next, %limit)

@next_i(%i)
  %i_next = addl %i, %one
  jmp @loop_i(%i_next)
}
