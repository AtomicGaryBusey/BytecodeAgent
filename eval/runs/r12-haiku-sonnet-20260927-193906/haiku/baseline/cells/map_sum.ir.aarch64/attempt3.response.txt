export function l $map_sum(l %a, l %n, l %f) {
@entry
    %sum =l copy 0
    %i =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond @body, @end

@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %val =l loadl %addr
    %result =l call %f(l %val)
    %sum =l add %sum, %result
    %i =l add %i, 1
    jmp @loop

@end
    ret %sum
}
