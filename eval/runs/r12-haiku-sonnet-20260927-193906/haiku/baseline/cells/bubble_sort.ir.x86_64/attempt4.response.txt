export function l $bubble_sort(l %a, l %n) {
@start
    jmp @outer_check

@outer_check
    %i =l phi @start 0, @outer_next %i_next
    %outer_cond =w csltl %i, %n
    jnz %outer_cond, @inner_init, @done

@inner_init
    %n_minus_i =l subl %n, %i
    %limit =l subl %n_minus_i, 1
    jmp @inner_check

@inner_check
    %j =l phi @inner_init 0, @inner_next %j_next
    %inner_cond =w csltl %j, %limit
    jnz %inner_cond, @body, @outer_next

@body
    %j_offset =l mull %j, 8
    %j_addr =l addl %a, %j_offset
    %j_val =l loadl %j_addr
    
    %j_plus_one =l addl %j, 1
    %j1_offset =l mull %j_plus_one, 8
    %j1_addr =l addl %a, %j1_offset
    %j1_val =l loadl %j1_addr
    
    %swap_cond =w csgtl %j_val, %j1_val
    jnz %swap_cond, @swap, @inner_next

@swap
    storel %j1_val, %j_addr
    storel %j_val, %j1_addr
    jmp @inner_next

@inner_next
    %j_next =l addl %j, 1
    jmp @inner_check

@outer_next
    %i_next =l addl %i, 1
    jmp @outer_check

@done
    ret 0
}
