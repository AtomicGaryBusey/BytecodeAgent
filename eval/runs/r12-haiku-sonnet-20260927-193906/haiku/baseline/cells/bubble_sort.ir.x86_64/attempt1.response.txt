export function l $bubble_sort(l %a, l %n) {
@start
    jmp @outer_loop(0)

@outer_loop(%i)
    %exit = csltl %i, %n
    jnz %exit, @inner_loop(%i, 0), @ret

@inner_loop(%i, %j)
    %n_minus_i = subl %n, %i
    %limit = subl %n_minus_i, 1
    %continue = csltl %j, %limit
    jnz %continue, @compare(%i, %j), @inner_end(%i)

@compare(%i, %j)
    %offset_j = mull %j, 8
    %addr_j = addl %a, %offset_j
    %val_j = loadl %addr_j
    
    %j_plus_1 = addl %j, 1
    %offset_j1 = mull %j_plus_1, 8
    %addr_j1 = addl %a, %offset_j1
    %val_j1 = loadl %addr_j1
    
    %should_swap = csgtl %val_j, %val_j1
    jnz %should_swap, @do_swap(%i, %j, %addr_j, %addr_j1, %val_j, %val_j1), @no_swap(%i, %j)

@do_swap(%i, %j, %addr_j, %addr_j1, %val_j, %val_j1)
    storel %val_j1, %addr_j
    storel %val_j, %addr_j1
    jmp @no_swap(%i, %j)

@no_swap(%i, %j)
    %j_next = addl %j, 1
    jmp @inner_loop(%i, %j_next)

@inner_end(%i)
    %i_next = addl %i, 1
    jmp @outer_loop(%i_next)

@ret
    ret 0
}
