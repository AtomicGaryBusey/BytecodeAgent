export function l $add_overflow(l %a, l %b, l %out)
{
@start
    %sum = add %a, %b
    storel %sum, %out
    
    %a_sign = asrl %a, 63
    %b_sign = asrl %b, 63
    %sum_sign = asrl %sum, 63
    
    %same_sign = ceql %a_sign, %b_sign
    %diff_sign = cnel %sum_sign, %a_sign
    
    %overflow = and %same_sign, %diff_sign
    
    %result = extuw %overflow
    
    ret %result
}
