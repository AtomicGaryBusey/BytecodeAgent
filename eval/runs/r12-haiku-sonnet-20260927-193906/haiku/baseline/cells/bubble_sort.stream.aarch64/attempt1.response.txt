bubble_sort:
    cmp x1, #1
    b.le .Lreturn_zero
    mov x2, #0
.Louter_loop:
    cmp x2, x1
    b.ge .Lreturn_zero
    mov x3, #0
    sub x4, x1, x2
    sub x4, x4, #1
.Linner_loop:
    cmp x3, x4
    b.ge .Linner_done
    add x5, x0, x3, lsl #3
    ldr x6, [x5]
    ldr x7, [x5, #8]
    cmp x6, x7
    b.le .Lno_swap
    str x7, [x5]
    str x6, [x5, #8]
.Lno_swap:
    add x3, x3, #1
    b .Linner_loop
.Linner_done:
    add x2, x2, #1
    b .Louter_loop
.Lreturn_zero:
    mov x0, #0
    ret
