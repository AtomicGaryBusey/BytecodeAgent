export function l $fib_rec(l %n) {
@entry
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @return_0
    
    %cmp1 =w ceql %n, 1
    jnz %cmp1, @return_1
    
    %n_minus_1 =l subl %n, 1
    %res1 =l call $fib_rec(l %n_minus_1)
    
    %n_minus_2 =l subl %n, 2
    %res2 =l call $fib_rec(l %n_minus_2)
    
    %result =l addl %res1, %res2
    ret %result

@return_0
    ret 0

@return_1
    ret 1
}
