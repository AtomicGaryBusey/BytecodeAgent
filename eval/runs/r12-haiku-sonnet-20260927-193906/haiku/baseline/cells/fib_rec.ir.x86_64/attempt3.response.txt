export function l $fib_rec(l %n) {
@entry
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @return_0, @check1
@check1
    %cmp1 =w ceql %n, 1
    jnz %cmp1, @return_1, @recurse
@recurse
    %n_minus_1 =l sub %n, 1
    %res1 =l call $fib_rec(l %n_minus_1)
    %n_minus_2 =l sub %n, 2
    %res2 =l call $fib_rec(l %n_minus_2)
    %result =l add %res1, %res2
    ret %result
@return_0
    ret 0
@return_1
    ret 1
}
