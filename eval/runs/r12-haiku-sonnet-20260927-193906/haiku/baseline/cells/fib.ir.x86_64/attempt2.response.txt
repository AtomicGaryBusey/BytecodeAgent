export function l $fib(l %n) {
@entry
	%is_zero = ceq l %n, 0
	jnz %is_zero, @zero
	
	%is_one = ceq l %n, 1
	jnz %is_one, @one
	
	jmp @loop
	
@loop
	%a = phi l @entry 0, @loopupdate %a_new
	%b = phi l @entry 1, @loopupdate %b_new
	%i = phi l @entry 2, @loopupdate %i_new
	
	%cond = csle l %i, %n
	jnz %cond, @loopupdate
	ret %b
	
@loopupdate
	%sum = add l %a, %b
	%a_new = %b
	%b_new = %sum
	%i_new = add l %i, 1
	jmp @loop
	
@zero
	ret 0
	
@one
	ret 1
}
