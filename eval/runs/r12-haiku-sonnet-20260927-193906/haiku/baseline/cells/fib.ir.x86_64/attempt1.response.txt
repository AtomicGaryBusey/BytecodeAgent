export function l $fib(l %n) {
@entry
	%is_zero = ceql %n, 0
	jnz %is_zero, @zero
	
	%is_one = ceql %n, 1
	jnz %is_one, @one
	
	jmp @loop
	
@loop
	%a = phi @entry 0, @loopupdate %a_new
	%b = phi @entry 1, @loopupdate %b_new
	%i = phi @entry 2, @loopupdate %i_new
	
	%cond = cslel %i, %n
	jnz %cond, @loopupdate
	ret %b
	
@loopupdate
	%sum = add %a, %b
	%a_new = copy %b
	%b_new = copy %sum
	%i_new = add %i, 1
	jmp @loop
	
@zero
	ret 0
	
@one
	ret 1
}
