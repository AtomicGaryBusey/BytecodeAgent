export function l $collatz_max(l %n) {
@entry
    jmp @loop_k_cond

@loop_k_cond
    %k = phi l $1, @entry, %k_next, @loop_end
    %max = phi l $0, @entry, %max_next, @loop_end
    %n_plus_1 = addl %n, $1
    %k_le_n = csltl %k, %n_plus_1
    jnz %k_le_n @collatz_start
    jmp @done

@collatz_start
    jmp @collatz_cond

@collatz_cond
    %x = phi l %k, @collatz_start, %x_next, @collatz_merge
    %count = phi l $0, @collatz_start, %count_next, @collatz_merge
    %x_eq_1 = ceql %x, $1
    jnz %x_eq_1 @collatz_end
    jmp @collatz_body

@collatz_body
    %is_odd = andl %x, $1
    jnz %is_odd @collatz_odd
    jmp @collatz_even

@collatz_even
    %x_even = divl %x, $2
    %count_even = addl %count, $1
    jmp @collatz_merge

@collatz_odd
    %temp = mull %x, $3
    %x_odd = addl %temp, $1
    %count_odd = addl %count, $1
    jmp @collatz_merge

@collatz_merge
    %x_next = phi l %x_even, @collatz_even, %x_odd, @collatz_odd
    %count_next = phi l %count_even, @collatz_even, %count_odd, @collatz_odd
    jmp @collatz_cond

@collatz_end
    %max_greater = csltl %max, %count
    jnz %max_greater @update_max
    jmp @no_update

@update_max
    %new_max = addl %count, $0
    jmp @loop_end

@no_update
    %new_max = addl %max, $0
    jmp @loop_end

@loop_end
    %max_next = phi l %new_max, @update_max, %new_max, @no_update
    %k_next = addl %k, $1
    jmp @loop_k_cond

@done
    ret %max
}
