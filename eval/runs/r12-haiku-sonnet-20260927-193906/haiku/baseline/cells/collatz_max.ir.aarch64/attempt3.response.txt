export function l $collatz_max(l %n) {
@entry
    %zero = sub l %n, %n
    %one = add l %zero, 1
    jmp @loop_k_cond

@loop_k_cond
    %k = phi l %one, @entry, %k_next, @loop_end
    %max = phi l %zero, @entry, %max_next, @loop_end
    %n_plus_1 = add l %n, 1
    %k_le_n = cslt l %k, %n_plus_1
    jnz %k_le_n @collatz_start
    jmp @done

@collatz_start
    jmp @collatz_cond

@collatz_cond
    %x = phi l %k, @collatz_start, %x_next, @collatz_merge
    %count = phi l %zero, @collatz_start, %count_next, @collatz_merge
    %x_eq_1 = ceq l %x, 1
    jnz %x_eq_1 @collatz_end
    jmp @collatz_body

@collatz_body
    %is_odd = and l %x, 1
    jnz %is_odd @collatz_odd
    jmp @collatz_even

@collatz_even
    %x_even = div l %x, 2
    %count_even = add l %count, 1
    jmp @collatz_merge

@collatz_odd
    %temp = mul l %x, 3
    %x_odd = add l %temp, 1
    %count_odd = add l %count, 1
    jmp @collatz_merge

@collatz_merge
    %x_next = phi l %x_even, @collatz_even, %x_odd, @collatz_odd
    %count_next = phi l %count_even, @collatz_even, %count_odd, @collatz_odd
    jmp @collatz_cond

@collatz_end
    %max_greater = cslt l %max, %count
    jnz %max_greater @update_max
    jmp @no_update

@update_max
    %new_max_u = add l %count, %zero
    jmp @loop_end

@no_update
    %new_max_n = add l %max, %zero
    jmp @loop_end

@loop_end
    %max_next = phi l %new_max_u, @update_max, %new_max_n, @no_update
    %k_next = add l %k, 1
    jmp @loop_k_cond

@done
    ret %max
}
