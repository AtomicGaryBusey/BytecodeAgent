export function l $collatz_max(l %n) {
@entry
    jmp @loop_k_start

@loop_k_start
    %max_start = copy l 0
    %k_start = copy l 1
    jmp @loop_k_cond

@loop_k_cond
    %max = phi l %max_start, @loop_k_start, %max_new, @join_branches
    %k = phi l %k_start, @loop_k_start, %k_next, @join_branches
    %n_plus_1 = add l %n, 1
    %cond = csltl %k, %n_plus_1
    jnz %cond @collatz_count
    ret %max

@collatz_count
    %x = copy l %k
    %count = copy l 0
    jmp @collatz_loop_cond

@collatz_loop_cond
    %x_curr = phi l %x, @collatz_count, %x_next, @collatz_loop_body_end
    %count_curr = phi l %count, @collatz_count, %count_next, @collatz_loop_body_end
    %cond = ceql %x_curr, 1
    jnz %cond @collatz_done
    jmp @collatz_loop_body

@collatz_loop_body
    %bit = and l %x_curr, 1
    %is_odd = ceql %bit, 1
    jnz %is_odd @collatz_odd
    jmp @collatz_even

@collatz_even
    %x_next_even = div l %x_curr, 2
    jmp @collatz_loop_body_end

@collatz_odd
    %tmp = mul l %x_curr, 3
    %x_next_odd = add l %tmp, 1
    jmp @collatz_loop_body_end

@collatz_loop_body_end
    %x_next = phi l %x_next_even, @collatz_even, %x_next_odd, @collatz_odd
    %count_next = add l %count_curr, 1
    jmp @collatz_loop_cond

@collatz_done
    %is_greater = csltl %max, %count_curr
    jnz %is_greater @update_max_branch
    jmp @no_update_branch

@update_max_branch
    jmp @join_branches

@no_update_branch
    jmp @join_branches

@join_branches
    %max_new = phi l %count_curr, @update_max_branch, %max, @no_update_branch
    %k_next = add l %k, 1
    jmp @loop_k_cond
}
