export function l $bmax(l %a, l %n) {
@entry
    %first = load l %a
    %one = const l 1
    %eight = const l 8
    jmp @loop

@loop
    %i = phi @entry %one @next_iter %next_i
    %max = phi @entry %first @next_iter %new_max
    %cmp = csltl %i %n
    jnz %cmp @body @done

@body
    %offset = mull %i %eight
    %ptr = addl %a %offset
    %elem = load l %ptr
    %gt = csgtl %elem %max
    jnz %gt @take_elem @keep_max

@take_elem
    jmp @next_iter

@keep_max
    jmp @next_iter

@next_iter
    %new_max = phi @take_elem %elem @keep_max %max
    %next_i = addl %i %one
    jmp @loop

@done
    ret %max
}
