export function l $bmax(l %a, l %n) {
@start
    %first = loadl %a
    %one = const l 1
    %eight = const l 8
    jmp @loop_cond
    
@loop_cond
    %idx = phi @start %one, @after_update %next_idx
    %max = phi @start %first, @after_update %new_max
    %cond = csltl %idx, %n
    jnz %cond @loop_body
    ret %max
    
@loop_body
    %offset = mul l %idx, %eight
    %ptr = add l %a, %offset
    %val = loadl %ptr
    %is_smaller = csltl %val, %max
    jnz %is_smaller @skip_update
    jmp @do_update
    
@do_update
    %new_max = add l %val, 0
    jmp @after_update
    
@skip_update
    %new_max = add l %max, 0
    jmp @after_update
    
@after_update
    %next_idx = add l %idx, %one
    jmp @loop_cond
}
