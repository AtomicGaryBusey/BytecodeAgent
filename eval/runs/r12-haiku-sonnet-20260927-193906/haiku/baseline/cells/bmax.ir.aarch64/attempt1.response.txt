export function l $bmax(l %a, l %n) {
@start
    %first = loadl %a
    %one = const l 1
    %eight = const l 8
    jmp @loop_cond
    
@loop_cond
    %idx = phi @start %one, @loop_body %next_idx
    %max = phi @start %first, @loop_body %new_max
    %cond = csltl %idx, %n
    jnz %cond @loop_body
    ret %max
    
@loop_body
    %offset = mul l %idx, %eight
    %ptr = add l %a, %offset
    %val = loadl %ptr
    %is_smaller = csltl %val, %max
    %new_max = select %is_smaller, %max, %val
    %next_idx = add l %idx, %one
    jmp @loop_cond
}
