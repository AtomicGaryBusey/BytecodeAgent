export function l $bmax(l %a, l %n) {
@entry
    %val0 = loadl %a
    %const1 = const l 1
    %const8 = const l 8
    jmp @loop

@loop
    %idx = phi @entry %const1 @end %next_idx
    %max = phi @entry %val0 @end %new_max
    %cmp = csltl %idx, %n
    jnz %cmp @body
    ret %max

@body
    %offset = mull %idx, %const8
    %ptr = addl %a, %offset
    %elem = loadl %ptr
    %gt = csgtl %elem, %max
    jnz %gt @use_elem
    jmp @use_max

@use_elem
    jmp @end

@use_max
    jmp @end

@end
    %new_max = phi @use_elem %elem @use_max %max
    %next_idx = addl %idx, %const1
    jmp @loop
}
