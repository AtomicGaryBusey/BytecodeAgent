bubble_sort:
	cmp rsi, 1
	jle .Ldone
	xor r8, r8
.Louter:
	cmp r8, rsi
	jge .Ldone
	xor r9, r9
.Linner:
	mov rax, rsi
	dec rax
	sub rax, r8
	cmp r9, rax
	jge .Linner_end
	mov r10, qword ptr [rdi + r9*8]
	mov r11, qword ptr [rdi + r9*8 + 8]
	cmp r10, r11
	jle .Lno_swap
	mov qword ptr [rdi + r9*8], r11
	mov qword ptr [rdi + r9*8 + 8], r10
.Lno_swap:
	inc r9
	jmp .Linner
.Linner_end:
	inc r8
	jmp .Louter
.Ldone:
	xor rax, rax
	ret
