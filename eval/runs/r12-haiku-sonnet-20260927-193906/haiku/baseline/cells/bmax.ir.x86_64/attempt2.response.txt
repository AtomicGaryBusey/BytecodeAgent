export function l $bmax(l %a, l %n) {
@entry
    %m0 = loadl %a
    %i0 = copy 1
    jmp @loop

@loop
    %m = phi @entry %m0, @inc %m1
    %i = phi @entry %i0, @inc %i1
    %lt = cltl %i %n
    jnz %lt @cmp
    ret %m

@cmp
    %off = mull %i 8
    %p = addl %a %off
    %e = loadl %p
    %gt = cgtl %e %m
    jnz %gt @set
    %m1 = copy %m
    jmp @inc

@set
    %m1 = copy %e
    jmp @inc

@inc
    %i1 = addl %i 1
    jmp @loop
}
