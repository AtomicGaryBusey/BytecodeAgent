export function l $bmax(l %a, l %n) {
@entry
    %max_init = loadl %a
    %i_init = copy 1
    jmp @loop

@loop
    %max = phi @entry %max_init, @increment %max_upd
    %i = phi @entry %i_init, @increment %i_new
    %cond = csltl %i %n
    jnz %cond @body
    ret %max

@body
    %offset = mull %i 8
    %ptr = addl %a %offset
    %elem = loadl %ptr
    %is_greater = csgtl %max %elem
    jnz %is_greater @skip_update
    jmp @do_update

@do_update
    %max_upd = copy %elem
    jmp @increment

@skip_update
    %max_upd = copy %max
    jmp @increment

@increment
    %i_new = addl %i 1
    jmp @loop
}
