@start
    %a0 = l and %a, 0xffffffff
    %a1 = l shr %a, 32
    %b0 = l and %b, 0xffffffff
    %b1 = l shr %b, 32

    %p10 = l mul %a1, %b0

    %p00_hi = l shr %p00, 32
    %p01_lo = l and %p01, 0xffffffff
    %p10_lo = l and %p10, 0xffffffff
    
    %mid1 = l add %p00_hi, %p01_lo
    %mid2 = l add %mid1, %p10_lo
    
    %p01_hi = l shr %p01, 32
    %p10_hi = l shr %p10, 32
    %mid_hi = l shr %mid2, 32

    ret %result
}
