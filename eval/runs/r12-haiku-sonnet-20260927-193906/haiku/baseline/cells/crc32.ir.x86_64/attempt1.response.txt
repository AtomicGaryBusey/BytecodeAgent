export function l $crc32(l %p, l %n) {
@entry
    %crc =w copy 0xffffffff
    %i =l copy 0
    jmp @loop_start
    
@loop_start
    %cond =w csltl %i, %n
    jnz %cond, @process_byte, @finish
    
@process_byte
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    
    %j =w copy 8
    jmp @bit_loop_start
    
@bit_loop_start
    %bcond =w csgtw %j, 0
    jnz %bcond, @bit_process, @next_byte
    
@bit_process
    %mask =w and %crc, 1
    %is_one =w cnew %mask, 0
    jnz %is_one, @if_one, @if_zero
    
@if_one
    %temp =w shr %crc, 1
    %crc =w xor %temp, 0xedb88320
    jmp @bit_decrement
    
@if_zero
    %crc =w shr %crc, 1
    jmp @bit_decrement
    
@bit_decrement
    %j =w sub %j, 1
    jmp @bit_loop_start
    
@next_byte
    %i =l add %i, 1
    jmp @loop_start
    
@finish
    %final =w xor %crc, 0xffffffff
    %result =l extuw %final
    ret %result
}
