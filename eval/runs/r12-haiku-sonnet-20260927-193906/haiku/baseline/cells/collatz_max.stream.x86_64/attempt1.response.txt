collatz_max:
    xor rax, rax        # max_steps = 0
    mov rcx, 1          # k = 1
.Lk_loop:
    cmp rcx, rdi        # k > n?
    jg .Ldone
    mov rdx, rcx        # x = k
    xor r8, r8          # steps = 0
.Lx_loop:
    cmp rdx, 1          # x == 1?
    je .Lx_done
    test rdx, 1         # odd?
    jz .Lx_even
    imul rdx, rdx, 3    # x = 3*x + 1
    add rdx, 1
    jmp .Lx_step
.Lx_even:
    shr rdx, 1          # x = x / 2
.Lx_step:
    inc r8              # steps++
    jmp .Lx_loop
.Lx_done:
    cmp rax, r8         # update max
    cmovl rax, r8
    inc rcx              # next k
    jmp .Lk_loop
.Ldone:
    ret
