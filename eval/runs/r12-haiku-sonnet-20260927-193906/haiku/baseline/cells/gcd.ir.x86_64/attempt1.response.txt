export function l $gcd(l %a, l %b) {
@entry
    jnz %b, @loop, @done

@loop
    %a.loop = phi @entry %a, @loop %a.next
    %b.loop = phi @entry %b, @loop %b.next
    %a.next = copy %b.loop
    %b.next = srem %a.loop, %b.loop
    jnz %b.next, @loop, @done

@done
    %a.final = phi @entry %a, @loop %a.next
    ret %a.final
}
