export function l $gcd(l %a, l %b) {
@entry
    %is_zero = ceql %b, 0
    jnz %is_zero, @done, @loop

@loop
    %a_cur = phi @entry %a, @loop %a_next
    %b_cur = phi @entry %b, @loop %b_next
    %rem = srem %a_cur, %b_cur
    %a_next = copy %b_cur
    %b_next = copy %rem
    %rem_zero = ceql %b_next, 0
    jnz %rem_zero, @done, @loop

@done
    %res = phi @entry %a, @loop %a_next
    ret %res
}
