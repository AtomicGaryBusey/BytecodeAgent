export function l $isqrt(l %n) {
@entry
    %is_zero = ceql %n 0
    jnz %is_zero @return_zero
    br @loop

@loop
    %x = phi @entry 4294967296 @continue %x_next
    %div = udivl %n %x
    %sum = addl %x %div
    %x_next = udivl %sum 2
    
    %cmp = csltl %x_next %x
    jnz %cmp @continue
    ret %x

@continue
    br @loop

@return_zero
    ret 0
}
