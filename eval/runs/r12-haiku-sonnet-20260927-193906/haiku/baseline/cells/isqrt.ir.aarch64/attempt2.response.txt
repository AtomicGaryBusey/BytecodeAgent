export function l $isqrt(l %n) {
@entry
    %is_zero = ceql %n 0
    jnz %is_zero @return_zero
    
    %x = addl %n 0
    br @loop

@loop
    %div = udivl %n %x
    %sum = addl %x %div
    %x_next = udivl %sum 2
    
    %cmp = csltl %x_next %x
    jnz %cmp @continue
    
    ret %x

@continue
    %x = addl %x_next 0
    br @loop

@return_zero
    ret 0
}
