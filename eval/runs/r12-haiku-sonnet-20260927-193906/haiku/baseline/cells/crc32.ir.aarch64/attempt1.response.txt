export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 0xffffffff
	%i =l copy 0
	jmp @looptest
	
@looptest
	%cond =w csltl %i, %n
	jnz %cond, @loopbody, @endloop
	
@loopbody
	%addr =l add %p, %i
	%byte =w loadub %addr
	%crc =w xor %crc, %byte
	%bit =w copy 0
	jmp @bitloop
	
@bitloop
	%lsb =w and %crc, 1
	jnz %lsb, @ifbit, @elsebit
	
@ifbit
	%shifted =w shr %crc, 1
	%crc =w xor %shifted, 0xedb88320
	jmp @bitend
	
@elsebit
	%crc =w shr %crc, 1
	jmp @bitend
	
@bitend
	%bit =w add %bit, 1
	%bitdone =w ceql %bit, 8
	jnz %bitdone, @nextbyte, @bitloop
	
@nextbyte
	%i =l add %i, 1
	jmp @looptest
	
@endloop
	%result =w xor %crc, 0xffffffff
	%resultl =l extuw %result
	ret %resultl
}
