export function l $crc32(l %p, l %n) {
@start
	%crc =l copy 4294967295
	%i =l copy 0
	jmp @looptest
	
@looptest
	%cond =w csltl %i, %n
	jnz %cond, @loopbody, @endloop
	
@loopbody
	%addr =l add %p, %i
	%byte =w loadub %addr
	%byte_l =l extuw %byte
	%crc =l xor %crc, %byte_l
	%j =l copy 0
	jmp @bitloop
	
@bitloop
	%lsb =l and %crc, 1
	%is_one =w cnel %lsb, 0
	jnz %is_one, @ifbit, @elsebit
	
@ifbit
	%shifted =l shr %crc, 1
	%crc =l xor %shifted, 3735928320
	jmp @bitend
	
@elsebit
	%crc =l shr %crc, 1
	jmp @bitend
	
@bitend
	%j =l add %j, 1
	%j_done =w ceql %j, 8
	jnz %j_done, @nextbyte, @bitloop
	
@nextbyte
	%i =l add %i, 1
	jmp @looptest
	
@endloop
	%result =l xor %crc, 4294967295
	ret %result
}
