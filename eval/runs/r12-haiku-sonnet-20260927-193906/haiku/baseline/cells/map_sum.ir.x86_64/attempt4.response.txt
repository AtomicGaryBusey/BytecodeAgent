export function l $map_sum(l %a, l %n, l %f) {
@entry
	jmp @loop_header

@loop_header
	%sum =l phi l 0 @entry l %new_sum @loop_body
	%i =l phi l 0 @entry l %new_i @loop_body
	%cond =w csltl %i %n
	jnz %cond @loop_body @done

@loop_body
	%offset =l mul %i 8
	%ptr =l add %a %offset
	%val =l loadl %ptr
	%res =l call %f(l %val)
	%new_sum =l add %sum %res
	%new_i =l add %i 1
	jmp @loop_header

@done
	ret %sum
}
