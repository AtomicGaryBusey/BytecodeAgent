export function l $map_sum(l %a, l %n, l %f) {
@entry
	%init_sum =l copy 0
	%init_i =l copy 0
	jmp @loop

@loop
	%sum =l phi %init_sum %new_sum
	%i =l phi %init_i %new_i
	%a_curr =l phi %a %a_next
	
	%cond =w csltl %i %n
	jnz %cond @loop_body @done

@loop_body
	%val =l loadl %a_curr
	%res =l call %f(l %val)
	%new_sum =l add %sum %res
	%a_next =l add %a_curr 8
	%new_i =l add %i 1
	jmp @loop

@done
	ret %sum
}
