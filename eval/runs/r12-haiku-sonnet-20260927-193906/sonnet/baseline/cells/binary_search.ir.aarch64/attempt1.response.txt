export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @lower, @upper
@lower
	%mid1 =l add %mid, 1
	%lo =l copy %mid1
	jmp @loop
@upper
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
