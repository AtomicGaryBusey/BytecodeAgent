export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%c =w csltl %i, %j
	jnz %c, %body, @end
@body
	%pi =l mul %i, 8
	%pj =l mul %j, 8
	%ai =l add %a, %pi
	%aj =l add %a, %pj
	%vi =l loadl %ai
	%vj =l loadl %aj
	storel %vj, %ai
	storel %vi, %aj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@end
	ret 0
}
