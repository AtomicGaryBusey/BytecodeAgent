export function l $bubble_sort(l %a, l %n) {
@start
    %pi =l alloc8 8
    %pj =l alloc8 8
    storel 0, %pi
@outer
    %i =l loadl %pi
    %c1 =w csltl %i, %n
    jnz %c1, @outer_body, @done
@outer_body
    %lim1 =l sub %n, 1
    %lim =l sub %lim1, %i
    storel 0, %pj
@inner
    %j =l loadl %pj
    %c2 =w csltl %j, %lim
    jnz %c2, @inner_body, @outer_end
@inner_body
    %off1 =l mul %j, 8
    %addr1 =l add %a, %off1
    %j1 =l add %j, 1
    %off2 =l mul %j1, 8
    %addr2 =l add %a, %off2
    %v1 =l loadl %addr1
    %v2 =l loadl %addr2
    %cmp =w csgtl %v1, %v2
    jnz %cmp, @swap, @noswap
@swap
    storel %v2, %addr1
    storel %v1, %addr2
    jmp @noswap
@noswap
    storel %j1, %pj
    jmp @inner
@outer_end
    %i1 =l add %i, 1
    storel %i1, %pi
    jmp @outer
@done
    ret 0
}
