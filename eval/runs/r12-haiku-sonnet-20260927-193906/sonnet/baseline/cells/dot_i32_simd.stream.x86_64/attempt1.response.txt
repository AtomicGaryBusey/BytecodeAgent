dot_i32_simd:
    pxor xmm0, xmm0
.Lvec_loop:
    cmp rdx, 4
    jl .Lreduce
    movdqu xmm1, xmmword ptr [rdi]
    movdqu xmm2, xmmword ptr [rsi]
    movdqa xmm3, xmm1
    movdqa xmm4, xmm2
    psrlq xmm3, 32
    psrlq xmm4, 32
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm0, xmm1
    paddq xmm0, xmm3
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    jmp .Lvec_loop
.Lreduce:
    pshufd xmm1, xmm0, 238
    paddq xmm0, xmm1
    movq rax, xmm0
    test rdx, rdx
    jz .Ldone
.Lscalar_loop:
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Lscalar_loop
.Ldone:
    ret
