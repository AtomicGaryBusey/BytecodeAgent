reverse:
    xor rax, rax
    cmp rsi, 1
    jle .Ldone
    lea rcx, [rsi - 1]
.Lloop:
    cmp rax, rcx
    jge .Ldone
    mov rdx, qword ptr [rdi + rax*8]
    mov r8, qword ptr [rdi + rcx*8]
    mov qword ptr [rdi + rax*8], r8
    mov qword ptr [rdi + rcx*8], rdx
    inc rax
    dec rcx
    jmp .Lloop
.Ldone:
    xor rax, rax
    ret
