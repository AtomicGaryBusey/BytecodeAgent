bubble_sort:
    cmp rsi, 2
    jl .Ldone
    xor rcx, rcx        # i = 0
.Louter:
    mov rax, rsi
    dec rax
    cmp rcx, rax
    jge .Lend_outer
    xor rdx, rdx        # j = 0
    mov r8, rsi
    dec r8
    sub r8, rcx         # r8 = n-1-i
.Linner:
    cmp rdx, r8
    jge .Lend_inner
    mov r9, qword ptr [rdi + rdx*8]
    mov r10, qword ptr [rdi + rdx*8 + 8]
    cmp r9, r10
    jle .Lno_swap
    mov qword ptr [rdi + rdx*8], r10
    mov qword ptr [rdi + rdx*8 + 8], r9
.Lno_swap:
    inc rdx
    jmp .Linner
.Lend_inner:
    inc rcx
    jmp .Louter
.Lend_outer:
.Ldone:
    xor rax, rax
    ret
