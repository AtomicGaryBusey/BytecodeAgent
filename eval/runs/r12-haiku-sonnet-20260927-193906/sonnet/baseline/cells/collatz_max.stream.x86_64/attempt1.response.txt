collatz_max:
    push rbx
    push r12
    xor rbx, rbx
    mov r12, 1
.Louter:
    cmp r12, rdi
    jg .Ldone
    mov rax, r12
    xor rcx, rcx
.Linner:
    cmp rax, 1
    je .Linner_done
    test rax, 1
    jnz .Lodd
    shr rax, 1
    jmp .Lcont
.Lodd:
    lea rax, [rax*2 + rax + 1]
.Lcont:
    inc rcx
    jmp .Linner
.Linner_done:
    cmp rcx, rbx
    jle .Lskip
    mov rbx, rcx
.Lskip:
    inc r12
    jmp .Louter
.Ldone:
    mov rax, rbx
    pop r12
    pop rbx
    ret
