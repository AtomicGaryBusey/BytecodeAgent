bubble_sort:
    cmp x1, #2
    b.lt .Ldone
    sub x11, x1, #1
    mov x9, #0
.Louter:
    cmp x9, x11
    b.ge .Ldone
    mov x10, #0
    sub x12, x11, x9
.Linner:
    cmp x10, x12
    b.ge .Lnext_outer
    lsl x2, x10, #3
    add x3, x0, x2
    ldr x4, [x3]
    ldr x5, [x3, #8]
    cmp x4, x5
    b.le .Lnoswap
    str x5, [x3]
    str x4, [x3, #8]
.Lnoswap:
    add x10, x10, #1
    b .Linner
.Lnext_outer:
    add x9, x9, #1
    b .Louter
.Ldone:
    mov x0, #0
    ret
