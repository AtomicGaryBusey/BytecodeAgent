export function l $crc32(l %p, l %n) {
@start
	jmp @loopcond
@loopcond
	%i =l phi @start 0, @bitdone %inext
	%crc =w phi @start 4294967295, @bitdone %crcafter
	%cond =w csltl %i, %n
	jnz %cond, @loopbody, @done
@loopbody
	%pi =l add %p, %i
	%b =w loadub %pi
	%crc1 =w xor %crc, %b
	jmp @bitcond
@bitcond
	%j =w phi @loopbody 0, @bitnext %jnext
	%bc =w phi @loopbody %crc1, @bitnext %bcnext
	%jlt =w csltw %j, 8
	jnz %jlt, @bitbody, @bitdone
@bitbody
	%lsb =w and %bc, 1
	%iszero =w ceqw %lsb, 0
	jnz %iszero, @even, @odd
@even
	%e =w shr %bc, 1
	jmp @bitnext
@odd
	%t =w shr %bc, 1
	%o =w xor %t, 3988292384
	jmp @bitnext
@bitnext
	%bcnext =w phi @even %e, @odd %o
	%jnext =w add %j, 1
	jmp @bitcond
@bitdone
	%crcafter =w copy %bc
	%inext =l add %i, 1
	jmp @loopcond
@done
	%result =w xor %crc, 4294967295
	%r64 =l extuw %result
	ret %r64
}
