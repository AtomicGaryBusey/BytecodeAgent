memmove_i64:
    cld
    xor rax, rax
    cmp rdx, 0
    je .Ldone
    cmp rdi, rsi
    jbe .Lforward
    lea r8, [rsi + rdx*8]
    cmp rdi, r8
    jae .Lforward
    lea rdi, [rdi + rdx*8 - 8]
    lea rsi, [rsi + rdx*8 - 8]
.Lback_loop:
    mov rcx, qword ptr [rsi]
    mov qword ptr [rdi], rcx
    sub rdi, 8
    sub rsi, 8
    dec rdx
    jnz .Lback_loop
    jmp .Ldone
.Lforward:
    mov rcx, qword ptr [rsi]
    mov qword ptr [rdi], rcx
    add rdi, 8
    add rsi, 8
    dec rdx
    jnz .Lforward
.Ldone:
    xor rax, rax
    ret
