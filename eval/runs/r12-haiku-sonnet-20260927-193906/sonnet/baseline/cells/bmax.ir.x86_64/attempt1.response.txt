export function l $bmax(l %a, l %n) {
@start
	%i =l copy 1
	%p0 =l loadl %a
	%max =l copy %p0
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%gt =w csgtl %v, %max
	jnz %gt, @update, @next
@update
	%max =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
