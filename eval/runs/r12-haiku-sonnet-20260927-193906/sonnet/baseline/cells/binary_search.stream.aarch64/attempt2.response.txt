binary_search:
    mov x9, #0
    sub x10, x1, #1
.Lloop:
    cmp x9, x10
    b.gt .Lnotfound
    add x11, x9, x10
    asr x11, x11, #1
    ldr x12, [x0, x11, lsl #3]
    cmp x12, x2
    b.eq .Lfound
    b.lt .Lgreater
    sub x10, x11, #1
    b .Lloop
.Lgreater:
    add x9, x11, #1
    b .Lloop
.Lfound:
    mov x0, x11
    ret
.Lnotfound:
    mov x0, #-1
    ret
