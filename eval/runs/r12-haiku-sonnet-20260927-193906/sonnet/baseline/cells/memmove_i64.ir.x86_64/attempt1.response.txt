export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp = csltl %dst, %src
	jnz %cmp, @fwd, @bwd
@fwd
	%i =l copy 0
@fwd_loop
	%c1 =w csltl %i, %n
	jnz %c1, @fwd_body, @done
@fwd_body
	%o1 =l mul %i, 8
	%sp1 =l add %src, %o1
	%dp1 =l add %dst, %o1
	%v1 =l loadl %sp1
	storel %v1, %dp1
	%i1 =l add %i, 1
	%i =l copy %i1
	jmp @fwd_loop
@bwd
	%j =l sub %n, 1
@bwd_loop
	%c2 =w csgel %j, 0
	jnz %c2, @bwd_body, @done
@bwd_body
	%o2 =l mul %j, 8
	%sp2 =l add %src, %o2
	%dp2 =l add %dst, %o2
	%v2 =l loadl %sp2
	storel %v2, %dp2
	%j1 =l sub %j, 1
	%j =l copy %j1
	jmp @bwd_loop
@done
	ret 0
}
