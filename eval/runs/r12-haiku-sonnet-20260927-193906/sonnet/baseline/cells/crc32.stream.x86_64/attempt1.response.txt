crc32:
    xor eax, eax
    not eax
    xor ecx, ecx
.Lbyteloop:
    cmp rcx, rsi
    jge .Ldone
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbitloop:
    test eax, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lnext
.Lshift:
    shr eax, 1
.Lnext:
    dec r8d
    jnz .Lbitloop
    inc rcx
    jmp .Lbyteloop
.Ldone:
    not eax
    ret
