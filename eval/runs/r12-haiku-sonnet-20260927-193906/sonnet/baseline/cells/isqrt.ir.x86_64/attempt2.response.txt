export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%r =l phi @start 0, @skip %rnext
	%bit =l phi @start 31, @skip %bitnext
	%mask =l shl 1, %bit
	%cand =l add %r, %mask
	%sq =l mul %cand, %cand
	%le =w culel %sq, %n
	jnz %le, @take, @skip
@take
	jmp @skip
@skip
	%rnext =l phi @loop %r, @take %cand
	%bitnext =l sub %bit, 1
	%cont =w csgel %bitnext, 0
	jnz %cont, @loop, @done
@done
	ret %rnext
}
