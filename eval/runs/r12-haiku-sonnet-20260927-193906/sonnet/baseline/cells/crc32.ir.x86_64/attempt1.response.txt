export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%pi =l add %p, %i
	%byte =w loadub %pi
	%c0 =w xor %crc, %byte

	%bit0 =w and %c0, 1
	%s0 =w shr %c0, 1
	%m0 =w sub 0, %bit0
	%x0 =w and %m0, -306674912
	%c1 =w xor %s0, %x0

	%bit1 =w and %c1, 1
	%s1 =w shr %c1, 1
	%m1 =w sub 0, %bit1
	%x1 =w and %m1, -306674912
	%c2 =w xor %s1, %x1

	%bit2 =w and %c2, 1
	%s2 =w shr %c2, 1
	%m2 =w sub 0, %bit2
	%x2 =w and %m2, -306674912
	%c3 =w xor %s2, %x2

	%bit3 =w and %c3, 1
	%s3 =w shr %c3, 1
	%m3 =w sub 0, %bit3
	%x3 =w and %m3, -306674912
	%c4 =w xor %s3, %x3

	%bit4 =w and %c4, 1
	%s4 =w shr %c4, 1
	%m4 =w sub 0, %bit4
	%x4 =w and %m4, -306674912
	%c5 =w xor %s4, %x4

	%bit5 =w and %c5, 1
	%s5 =w shr %c5, 1
	%m5 =w sub 0, %bit5
	%x5 =w and %m5, -306674912
	%c6 =w xor %s5, %x5

	%bit6 =w and %c6, 1
	%s6 =w shr %c6, 1
	%m6 =w sub 0, %bit6
	%x6 =w and %m6, -306674912
	%c7 =w xor %s6, %x6

	%bit7 =w and %c7, 1
	%s7 =w shr %c7, 1
	%m7 =w sub 0, %bit7
	%x7 =w and %m7, -306674912
	%c8 =w xor %s7, %x7

	%crc =w copy %c8
	%i1 =l add %i, 1
	%i =l copy %i1
	jmp @loop
@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
