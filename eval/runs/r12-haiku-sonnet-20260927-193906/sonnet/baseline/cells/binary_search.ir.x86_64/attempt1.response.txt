export function l $binary_search(l %a, l %n, l %key) {
@start
    %lo0 =l copy 0
    %hi0 =l sub %n, 1
@loop
    %lo =l phi @start %lo0, @gohigh %lonew, @golow %lo
    %hi =l phi @start %hi0, @gohigh %hi, @golow %hinew
    %c1 =w csgtl %lo, %hi
    jnz %c1, @notfound, @body
@body
    %sum =l add %lo, %hi
    %mid =l div %sum, 2
    %off =l mul %mid, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %eq =w ceql %val, %key
    jnz %eq, @found, @check
@check
    %lt =w csltl %val, %key
    jnz %lt, @gohigh, @golow
@gohigh
    %lonew =l add %mid, 1
    jmp @loop
@golow
    %hinew =l sub %mid, 1
    jmp @loop
@found
    ret %mid
@notfound
    %neg1 =l copy -1
    ret %neg1
}
