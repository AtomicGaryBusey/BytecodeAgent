map_sum:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    xor r13, r13
    xor r14, r14
.Lloop:
    cmp r14, rbp
    jge .Ldone
    mov rdi, qword ptr [rbx + r14*8]
    call r12
    add r13, rax
    inc r14
    jmp .Lloop
.Ldone:
    mov rax, r13
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
