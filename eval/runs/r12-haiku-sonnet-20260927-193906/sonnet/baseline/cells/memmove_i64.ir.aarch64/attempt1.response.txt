export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%islt =w csltl %dst, %src
	jnz %islt, @fwd, @checkeq
@checkeq
	%iseq =w ceql %dst, %src
	jnz %iseq, @done, @bwd
@fwd
	jmp @fwdloop
@fwdloop
	%i =l phi @fwd 0, @fwdbody %i2
	%fcond =w csltl %i, %n
	jnz %fcond, @fwdbody, @done
@fwdbody
	%foff =l mul %i, 8
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fv =l loadl %fsp
	storel %fv, %fdp
	%i2 =l add %i, 1
	jmp @fwdloop
@bwd
	%j0 =l sub %n, 1
	jmp @bwdloop
@bwdloop
	%j =l phi @bwd %j0, @bwdbody %j2
	%bcond =w csgel %j, 0
	jnz %bcond, @bwdbody, @done
@bwdbody
	%boff =l mul %j, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	%j2 =l sub %j, 1
	jmp @bwdloop
@done
	ret 0
}
