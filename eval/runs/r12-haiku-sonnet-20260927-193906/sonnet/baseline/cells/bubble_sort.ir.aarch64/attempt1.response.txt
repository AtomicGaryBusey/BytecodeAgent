export function l $bubble_sort(l %a, l %n) {
@start
    jmp @outer
@outer
    %i =l phi @start 0, @outerdone %i1
    %c1 =w csltl %i, %n
    jnz %c1, @outerchk, @done
@outerchk
    %nm1 =l sub %n, 1
    %c2 =w csltl %i, %nm1
    jnz %c2, @innerinit, @done
@innerinit
    %lim =l sub %nm1, %i
    jmp @inner
@inner
    %j =l phi @innerinit 0, @innerstep %j1
    %c3 =w csltl %j, %lim
    jnz %c3, @innerbody, @innerdone
@innerbody
    %off1 =l mul %j, 8
    %p1 =l add %a, %off1
    %v1 =l loadl %p1
    %jp1 =l add %j, 1
    %off2 =l mul %jp1, 8
    %p2 =l add %a, %off2
    %v2 =l loadl %p2
    %swap =w csgtl %v1, %v2
    jnz %swap, @doswap, @innerstep
@doswap
    storel %v2, %p1
    storel %v1, %p2
    jmp @innerstep
@innerstep
    %j1 =l add %j, 1
    jmp @inner
@innerdone
    %i1 =l add %i, 1
    jmp @outerdone
@outerdone
    jmp @outer
@done
    ret 0
}
