map_sum:
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    str x23, [sp, #48]
    mov x19, x0        // a
    mov x20, x1        // n
    mov x21, x2        // f
    mov x22, #0        // sum
    mov x23, #0        // i
.Lloop:
    cmp x23, x20
    b.ge .Ldone
    ldr x0, [x19, x23, lsl #3]
    blr x21
    add x22, x22, x0
    add x23, x23, #1
    b .Lloop
.Ldone:
    mov x0, x22
    ldr x23, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64
    ret
