export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	%p1 =l mul %i, 8
	%p1 =l add %a, %p1
	jmp @loop
@loop
	%c =w csltl %i, %j
	jnz %c, @body, @end
@body
	%oi =l mul %i, 8
	%pi =l add %a, %oi
	%oj =l mul %j, 8
	%pj =l add %a, %oj
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@end
	ret 0
}
