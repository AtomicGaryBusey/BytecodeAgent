export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%isg =w csgtl %v, %max
	jnz %isg, @take, @next
@take
	%max =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
