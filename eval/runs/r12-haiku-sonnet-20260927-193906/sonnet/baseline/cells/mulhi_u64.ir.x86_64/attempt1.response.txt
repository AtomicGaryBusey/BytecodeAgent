export function l $mulhi_u64(l %a, l %b) {
@start
	%amask =l and %a, 4294967295
	%ahi =l shr %a, 32
	%bmask =l and %b, 4294967295
	%bhi =l shr %b, 32

	%p0 =l mul %amask, %bmask
	%p1 =l mul %amask, %bhi
	%p2 =l mul %ahi, %bmask
	%p3 =l mul %ahi, %bhi

	%p1lo =l and %p1, 4294967295
	%p2lo =l and %p2, 4294967295
	%p0hi =l shr %p0, 32

	%t1 =l add %p1lo, %p2lo
	%carry =l add %t1, %p0hi

	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%carryhi =l shr %carry, 32

	%s1 =l add %p3, %p1hi
	%s2 =l add %s1, %p2hi
	%res =l add %s2, %carryhi

	ret %res
}
