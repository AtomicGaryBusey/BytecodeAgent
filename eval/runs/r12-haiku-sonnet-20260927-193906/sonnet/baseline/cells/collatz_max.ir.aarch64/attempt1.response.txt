export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%maxv =l copy 0
	jmp @outercheck
@outercheck
	%cont =w csgtl %k, %n
	jnz %cont, @done, @outerbody
@outerbody
	%x =l copy %k
	%steps =l copy 0
	jmp @innercheck
@innercheck
	%iseq1 =w ceql %x, 1
	jnz %iseq1, @innerdone, @innerbody
@innerbody
	%r =l and %x, 1
	%iseven =w ceql %r, 0
	jnz %iseven, @evencase, @oddcase
@evencase
	%x2 =l div %x, 2
	%x =l copy %x2
	jmp @stepinc
@oddcase
	%x3 =l mul %x, 3
	%x4 =l add %x3, 1
	%x =l copy %x4
	jmp @stepinc
@stepinc
	%steps1 =l add %steps, 1
	%steps =l copy %steps1
	jmp @innercheck
@innerdone
	%isgreater =w csgtl %steps, %maxv
	jnz %isgreater, @updatemax, @knext
@updatemax
	%maxv =l copy %steps
	jmp @knext
@knext
	%k1 =l add %k, 1
	%k =l copy %k1
	jmp @outercheck
@done
	ret %maxv
}
