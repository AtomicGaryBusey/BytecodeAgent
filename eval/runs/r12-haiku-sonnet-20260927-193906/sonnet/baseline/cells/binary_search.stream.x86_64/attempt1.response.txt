binary_search:
    xor rax, rax
    sub rax, 1
    xor r8, r8
    mov r9, rsi
.Lloop:
    cmp r8, r9
    jge .Lnotfound
    mov rcx, r9
    sub rcx, r8
    sar rcx, 1
    add rcx, r8
    cmp qword ptr [rdi + rcx*8], rdx
    je .Lfound
    jl .Lgreater
    mov r9, rcx
    jmp .Lloop
.Lgreater:
    mov r8, rcx
    add r8, 1
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lnotfound:
    ret
