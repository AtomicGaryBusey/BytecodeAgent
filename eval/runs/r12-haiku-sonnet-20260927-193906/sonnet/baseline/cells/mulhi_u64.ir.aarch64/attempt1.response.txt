export function l $mulhi_u64(l %a, l %b) {
@start
    %a_lo =l and %a, 4294967295
    %a_hi =l shr %a, 32
    %b_lo =l and %b, 4294967295
    %b_hi =l shr %b, 32
    %t0 =l mul %a_lo, %b_lo
    %t1 =l mul %a_lo, %b_hi
    %t2 =l mul %a_hi, %b_lo
    %t3 =l mul %a_hi, %b_hi
    %t0_hi =l shr %t0, 32
    %t1_lo =l and %t1, 4294967295
    %t2_lo =l and %t2, 4294967295
    %mid =l add %t0_hi, %t1_lo
    %mid2 =l add %mid, %t2_lo
    %carry =l shr %mid2, 32
    %t1_hi =l shr %t1, 32
    %t2_hi =l shr %t2, 32
    %hi1 =l add %t3, %t1_hi
    %hi2 =l add %hi1, %t2_hi
    %hi3 =l add %hi2, %carry
    ret %hi3
}
