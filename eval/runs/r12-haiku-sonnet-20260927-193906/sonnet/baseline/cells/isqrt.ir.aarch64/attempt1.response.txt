export function l $isqrt(l %n0) {
@start
	%bit0 =l copy 4611686018427387904
	jmp @adjust
@adjust
	%bit1 =l phi @start %bit0, @adjust_body %bit2
	%c1 =w cugtl %bit1, %n0
	jnz %c1, @adjust_body, @loop_init
@adjust_body
	%bit2 =l shr %bit1, 2
	jmp @adjust
@loop_init
	jmp @loop
@loop
	%n1 =l phi @loop_init %n0, @loop_then %n2, @loop_else %n1e
	%res1 =l phi @loop_init 0, @loop_then %res2, @loop_else %res3
	%bit3 =l phi @loop_init %bit1, @loop_then %bit4, @loop_else %bit4e
	%c2 =w ceql %bit3, 0
	jnz %c2, @done, @loop_body
@loop_body
	%sum =l add %res1, %bit3
	%c3 =w cugel %n1, %sum
	jnz %c3, @loop_then, @loop_else
@loop_then
	%n2 =l sub %n1, %sum
	%resshr =l shr %res1, 1
	%res2 =l add %resshr, %bit3
	%bit4 =l shr %bit3, 2
	jmp @loop
@loop_else
	%res3 =l shr %res1, 1
	%n1e =l copy %n1
	%bit4e =l shr %bit3, 2
	jmp @loop
@done
	ret %res1
}
