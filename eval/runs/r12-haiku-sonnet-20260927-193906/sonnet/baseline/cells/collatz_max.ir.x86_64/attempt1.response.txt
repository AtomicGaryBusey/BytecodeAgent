export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@outer
	%cmp1 =w csgtl %k, %n
	jnz %cmp1, @done, @body
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%cmp2 =w ceql %x, 1
	jnz %cmp2, @innerdone, @step
@step
	%r =l rem %x, 2
	%iseven =w ceql %r, 0
	jnz %iseven, @even, @odd
@even
	%x2 =l div %x, 2
	%x =l copy %x2
	jmp @stepend
@odd
	%x3 =l mul %x, 3
	%x4 =l add %x3, 1
	%x =l copy %x4
	jmp @stepend
@stepend
	%steps1 =l add %steps, 1
	%steps =l copy %steps1
	jmp @inner
@innerdone
	%isgreater =w csgtl %steps, %max
	jnz %isgreater, @updatemax, @next
@updatemax
	%max =l copy %steps
	jmp @next
@next
	%k1 =l add %k, 1
	%k =l copy %k1
	jmp @outer
@done
	ret %max
}
