export function l $map_sum(l %a, l %n, l %f) {
@start
	%i =l copy 0
	%s =l copy 0
	%p =l copy %a
	jmp @cond
@cond
	%i1 =l phi @start %i, @body %i2
	%s1 =l phi @start %s, @body %s2
	%p1 =l phi @start %p, @body %p2
	%c =w csltl %i1, %n
	jnz %c, @body, @end
@body
	%x =l loadl %p1
	%r =l call %f(l %x)
	%s2 =l add %s1, %r
	%p2 =l add %p1, 8
	%i2 =l add %i1, 1
	jmp @cond
@end
	ret %s1
}
