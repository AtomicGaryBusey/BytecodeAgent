binary_search:
    xor ecx, ecx
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jge .Lnotfound
    lea rax, [rcx + r8]
    shr rax, 1
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    je .Ldone
    jl .Lright
    mov r8, rax
    jmp .Lloop
.Lright:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lnotfound:
    mov rax, -1
.Ldone:
    ret
