export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
@l1
	%c =w cugtl %bit, %x
	jnz %c, @shr, @l2
@shr
	%bit =l shr %bit, 2
	jmp @l1
@l2
	%z =w ceql %bit, 0
	jnz %z, @done, @body
@body
	%t =l add %res, %bit
	%ge =w cugel %x, %t
	%res =l shr %res, 1
	jnz %ge, @sub, @next
@sub
	%x =l sub %x, %t
	%res =l add %res, %bit
	jmp @next
@next
	%bit =l shr %bit, 2
	jmp @l2
@done
	ret %res
}
