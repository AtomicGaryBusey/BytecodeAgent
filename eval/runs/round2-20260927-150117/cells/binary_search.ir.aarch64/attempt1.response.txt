export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @notfound
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @goright, @goleft
@goright
	%lo =l add %mid, 1
	jmp @loop
@goleft
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
