export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%ll =l mul %alo, %blo
	%lh =l mul %alo, %bhi
	%hl =l mul %ahi, %blo
	%hh =l mul %ahi, %bhi
	%llh =l shr %ll, 32
	%t =l add %hl, %llh
	%w1a =l and %t, 4294967295
	%w2 =l shr %t, 32
	%w1 =l add %w1a, %lh
	%w1h =l shr %w1, 32
	%r1 =l add %hh, %w2
	%r =l add %r1, %w1h
	ret %r
}
