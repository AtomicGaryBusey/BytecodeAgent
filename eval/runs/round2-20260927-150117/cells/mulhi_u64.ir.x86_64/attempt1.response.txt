export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%ll =l mul %alo, %blo
	%lh =l mul %alo, %bhi
	%hl =l mul %ahi, %blo
	%hh =l mul %ahi, %bhi
	%llh =l shr %ll, 32
	%lhl =l and %lh, 4294967295
	%hll =l and %hl, 4294967295
	%m1 =l add %llh, %lhl
	%mid =l add %m1, %hll
	%midh =l shr %mid, 32
	%lhh =l shr %lh, 32
	%hlh =l shr %hl, 32
	%r1 =l add %hh, %lhh
	%r2 =l add %r1, %hlh
	%r =l add %r2, %midh
	ret %r
}
