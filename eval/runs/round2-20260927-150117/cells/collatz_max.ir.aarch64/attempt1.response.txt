export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy 1
@outer
	%c =w cslel %k, %n
	jnz %c, @body, @done
@body
	%x =l copy %k
	%s =l copy 0
@inner
	%one =w ceql %x, 1
	jnz %one, @endinner, @step
@step
	%b =l and %x, 1
	%odd =w cnel %b, 0
	jnz %odd, @isodd, @iseven
@isodd
	%t =l mul %x, 3
	%x =l add %t, 1
	jmp @next
@iseven
	%x =l shr %x, 1
@next
	%s =l add %s, 1
	jmp @inner
@endinner
	%g =w csgtl %s, %best
	jnz %g, @upd, @cont
@upd
	%best =l copy %s
@cont
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
