export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%a =l add %p, %i
	%b =w loadub %a
	%crc =w xor %crc, %b
	%k =w copy 0
	jmp @bitloop
@bitloop
	%bc =w csltw %k, 8
	jnz %bc, @bitbody, @next
@bitbody
	%lsb =w and %crc, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, 3988292384
	%sh =w shr %crc, 1
	%crc =w xor %sh, %poly
	%k =w add %k, 1
	jmp @bitloop
@next
	%i =l add %i, 1
	jmp @loop
@done
	%r =w xor %crc, -1
	%res =l extuw %r
	ret %res
}
