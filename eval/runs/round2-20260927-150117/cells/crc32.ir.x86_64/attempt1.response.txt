export function l $crc32(l %p, l %n) {
@start
	%c =w copy 4294967295
	%i =l copy 0
@outer
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%c =w xor %c, %b
	%k =w copy 8
@inner
	%bit =w and %c, 1
	%mask =w sub 0, %bit
	%poly =w and %mask, 3988292384
	%c =w shr %c, 1
	%c =w xor %c, %poly
	%k =w sub %k, 1
	jnz %k, @inner, @next
@next
	%i =l add %i, 1
	jmp @outer
@done
	%r =w xor %c, 4294967295
	%rl =l extuw %r
	ret %rl
}
