dot_i32_simd:
    pxor xmm0, xmm0
    xor ecx, ecx
    mov r10, rdx
    and r10, -4
    test r10, r10
    jz .Lreduce
.Lvec:
    pmovsxdq xmm1, qword ptr [rdi + rcx*4]
    pmovsxdq xmm2, qword ptr [rsi + rcx*4]
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    pmovsxdq xmm3, qword ptr [rdi + rcx*4 + 8]
    pmovsxdq xmm4, qword ptr [rsi + rcx*4 + 8]
    pmuldq xmm3, xmm4
    paddq xmm0, xmm3
    add rcx, 4
    cmp rcx, r10
    jb .Lvec
.Lreduce:
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq rax, xmm0
    cmp rcx, rdx
    jge .Ldone
.Lscalar:
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    cmp rcx, rdx
    jl .Lscalar
.Ldone:
    ret
