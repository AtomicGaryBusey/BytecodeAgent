export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shrink
	%c1 =w cugtl %bit, %x
	jnz %c1, @doshr, @loop
@doshr
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%c2 =w cnel %bit, 0
	jnz %c2, @body, @done
@body
	%t =l add %res, %bit
	%c3 =w cugel %x, %t
	jnz %c3, @take, @skip
@take
	%x =l sub %x, %t
	%h =l shr %res, 1
	%res =l add %h, %bit
	jmp @next
@skip
	%res =l shr %res, 1
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
