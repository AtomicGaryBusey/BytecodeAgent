memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    ja .Lbackward
    xor ecx, ecx
.Lfwd:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jl .Lfwd
    ret
.Lbackward:
    mov rcx, rdx
.Lbwd:
    dec rcx
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    test rcx, rcx
    jg .Lbwd
.Ldone:
    ret
