isqrt:
    xor eax, eax
    movabs rcx, 0x4000000000000000
.Lalign:
    cmp rcx, rdi
    jbe .Lloop
    shr rcx, 2
    jnz .Lalign
.Lloop:
    test rcx, rcx
    jz .Ldone
    lea rdx, [rax + rcx]
    shr rax, 1
    cmp rdi, rdx
    jb .Lskip
    sub rdi, rdx
    add rax, rcx
.Lskip:
    shr rcx, 2
    jmp .Lloop
.Ldone:
    ret
