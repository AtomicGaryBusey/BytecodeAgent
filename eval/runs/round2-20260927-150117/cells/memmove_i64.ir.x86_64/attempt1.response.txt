export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c =w cultl %dst, %src
	jnz %c, @fwd_init, @bwd_init
@fwd_init
	%i =l copy 0
@fwd_cond
	%fc =w csltl %i, %n
	jnz %fc, @fwd_body, @done
@fwd_body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i =l add %i, 1
	jmp @fwd_cond
@bwd_init
	%j =l copy %n
@bwd_cond
	%bc =w csgtl %j, 0
	jnz %bc, @bwd_body, @done
@bwd_body
	%j =l sub %j, 1
	%off2 =l mul %j, 8
	%sp2 =l add %src, %off2
	%dp2 =l add %dst, %off2
	%v2 =l loadl %sp2
	storel %v2, %dp2
	jmp @bwd_cond
@done
	ret 0
}
