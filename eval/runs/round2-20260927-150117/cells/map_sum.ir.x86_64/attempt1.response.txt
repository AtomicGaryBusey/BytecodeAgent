export function l $map_sum(l %a, l %n, l %f) {
@start
	%i0 =l copy 0
	%s0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%s =l phi @start %s0, @body %s1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%s1 =l add %s, %r
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %s
}
