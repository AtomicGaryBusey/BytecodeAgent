binary_search:
    mov x9, #0              // lo
    mov x10, x1             // hi (exclusive)
.Lloop:
    cmp x9, x10
    b.ge .Lnotfound
    sub x11, x10, x9
    add x11, x9, x11, lsr #1   // mid
    ldr x12, [x0, x11, lsl #3]
    cmp x12, x2
    b.eq .Lfound
    b.lt .Lright
    mov x10, x11
    b .Lloop
.Lright:
    add x9, x11, #1
    b .Lloop
.Lfound:
    mov x0, x11
    ret
.Lnotfound:
    mov x0, #-1
    ret
