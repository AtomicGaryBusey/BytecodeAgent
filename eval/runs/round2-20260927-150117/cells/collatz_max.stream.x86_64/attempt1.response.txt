collatz_max:
    xor eax, eax
    mov rsi, 1
.Louter:
    cmp rsi, rdi
    jg .Ldone
    mov rdx, rsi
    xor ecx, ecx
.Linner:
    cmp rdx, 1
    je .Lend
    test rdx, 1
    jnz .Lodd
    shr rdx, 1
    inc rcx
    jmp .Linner
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
    inc rcx
    jmp .Linner
.Lend:
    cmp rcx, rax
    cmovg rax, rcx
    inc rsi
    jmp .Louter
.Ldone:
    ret
