crc32:
    mov eax, -1
    test rsi, rsi
    jle .Ldone
    xor ecx, ecx
.Lbyte:
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit:
    mov edx, eax
    and edx, 1
    neg edx
    and edx, 0xEDB88320
    shr eax, 1
    xor eax, edx
    dec r8d
    jnz .Lbit
    inc rcx
    cmp rcx, rsi
    jl .Lbyte
.Ldone:
    not eax
    ret
