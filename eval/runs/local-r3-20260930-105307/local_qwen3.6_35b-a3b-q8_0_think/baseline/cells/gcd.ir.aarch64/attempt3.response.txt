export function l $gcd(l %a, l %b) {
@start
    set $is_zero, ceql %b, 0
    jnz $is_zero, @done, @loop

@loop
    set $quot, divl %a, %b
    set $prod, mull $quot, %b
    set $rem, subl %a, $prod
    
    set $new_a, %b
    set $new_b, $rem
    
    set %a, $new_a
    set %b, $new_b
    j @start

@done
    ret %a
}
