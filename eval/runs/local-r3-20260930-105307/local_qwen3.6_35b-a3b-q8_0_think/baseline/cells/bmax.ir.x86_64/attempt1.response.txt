export function l $bmax(l %a, l %n) {
    @entry
        %max = loadl *%a
        %i = 0
    @loop
        %cond = csgtl %n, %i
        jnz %cond, @body
        ret %max
    @body
        %off = shll 3, %i
        %ptr = addl %a, %off
        %val = loadl *%ptr
        %gt = csgtl %val, %max
        jnz %gt, @update
    @skip
        %i = addl %i, 1
        jmp @loop
    @update
        %max = %val
        %i = addl %i, 1
        jmp @loop
}
