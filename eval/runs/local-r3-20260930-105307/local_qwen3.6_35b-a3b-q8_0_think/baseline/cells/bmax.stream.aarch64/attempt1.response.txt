bmax:
    ldr     x9, [x0], #8      // load first element and advance pointer
    sub     x1, x1, #1        // decrement count
.Lloop:
    cbz     x1, .Lend         // if count == 0, done
    ldr     x10, [x0], #8     // load next element and advance
    cmp     x10, x9           // compare with current max
    csel    x9, x10, x9, gt   // update max if new element is greater
    sub     x1, x1, #1        // decrement count
    b       .Lloop            // continue loop
.Lend:
    mov     x0, x9            // move result to return register
    ret                       // return
