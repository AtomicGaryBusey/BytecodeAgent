export function l $fib_rec(l %n) {
@entry
    %cmp0 = ceql %n, 0
    jnz @base_0

    %cmp1 = ceql %n, 1
    jnz @base_1

@recurse
    %n1 = addl %n, -1
    %r1 =l call $fib_rec(l %n1)

    %n2 = addl %n, -2
    %r2 =l call $fib_rec(l %n2)

    %res = addl %r1, %r2
    ret %res

@base_0
    ret 0

@base_1
    ret 1
}
