export function l $map_sum(l %a, l %n, l %f) {
@head
  %cmp = csltl %n, 0
  jnz @empty
  %sum = addl 0, 0
  %i = addl 0, 0
@loop
  %done = cselt %i, %n
  jnz @return_sum
  %idx = shl %i, 3
  %addr = addl %a, %idx
  %val = loadl %addr
  %res = call %f(l %val)
  %sum = addl %sum, %res
  %i = addl %i, 1
  jmp @loop
@empty
  ret 0
@return_sum
  ret %sum
}
