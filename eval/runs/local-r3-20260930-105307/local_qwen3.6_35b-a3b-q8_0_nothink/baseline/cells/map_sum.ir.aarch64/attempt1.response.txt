export function l $map_sum(l %a, l %n, l %f) {
@loop
  %cmp = csltl %n, 0
  jnz @return_zero
@body
  %sum = addl 0, 0
  %i = addl 0, 0
@head
  %done = cselt %i, %n
  jnz @end_body
  %idx = shl %i, 3
  %val_addr = addl %a, %idx
  %val = loadl %val_addr
  %res = call %f(l %val)
  %sum = addl %sum, %res
  %i = addl %i, 1
  jmp @head
@end_body
  ret %sum
@return_zero
  ret 0
}
