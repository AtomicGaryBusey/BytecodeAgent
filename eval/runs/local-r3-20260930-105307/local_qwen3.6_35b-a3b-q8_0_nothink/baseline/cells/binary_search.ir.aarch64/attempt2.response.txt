export function l $binary_search(l %a, l %n, l %key) {
@entry
	%cset = csetl sltl (const w 0), (trunc l to w %n)
	jnz %cset, @search, @not_found

@search
	%lo = const l 0
	%hi = sub l %n, 1

@loop
	%csl = csltl %lo, %hi
	jnz %csl, @mid_calc, @done

@mid_calc
	%sum = add l %lo, %hi
	%mid = asr_l %sum, 1
	%val = loadl (%a + %mid * 8)
	%eq = ceql %val, %key
	jnz %eq, @found

	%cgt = csgtl %val, %key
	jnz %cgt, @left, @right

@right
	%new_lo = add l %mid, 1
	brt %new_lo, %hi, @loop
	br %done

@left
	cslt %lo, %mid, %new_hi
	brt %lo, %new_hi, @loop
	br %done

@found
	ret l %mid

@done
	ret w -1

@not_found
	ret w -1
}
