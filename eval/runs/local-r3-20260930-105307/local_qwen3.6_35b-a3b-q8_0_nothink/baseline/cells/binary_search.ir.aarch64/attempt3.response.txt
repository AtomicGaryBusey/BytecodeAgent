export function l $binary_search(l %a, l %n, l %key) {
@entry
	csltl (const w 0), (trunc l to w %n)
	jnz @empty, @search

@empty
	ret w -1

@search
	%lo = const l 0
	%hi = sub l %n, 1

@loop
	%csl = csltl %lo, %hi
	jnz @mid_calc, @done

@mid_calc
	%sum = add l %lo, %hi
	%mid = asr_l %sum, 1
	%val = loadl (%a + %mid * 8)
	%eq = ceql %val, %key
	jnz @found

	%cgt = csgtl %val, %key
	jnz @left, @right

@right
	%new_lo = add l %mid, 1
	cslt %new_lo, %hi, %ok
	jnz @loop2, @done

@loop2
	br %loop

@left
	%new_hi = sub l %mid, 1
	cslt %lo, %new_hi, %ok2
	jnz @loop3, @done

@loop3
	br %loop

@found
	ret l %mid

@done
	ret w -1
}
