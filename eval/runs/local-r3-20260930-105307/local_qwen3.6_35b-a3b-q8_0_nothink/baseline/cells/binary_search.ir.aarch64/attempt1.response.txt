export function l $binary_search(l %a, l %n, l %key) {
@entry
	%zero = const w 0
	%nw = trunc l to w %n
	csetl %eq_zero = eqw %nw, %zero
	jnz %eq_zero, @not_found

	%lo = const l 0
	%hi = sub l %n, 1
	%result = const l -1

@loop
	%cslt_lo_hi = csltl %lo, %hi
	jnz %cslt_lo_hi, @body, @check_result

@body
	%mid = add l %lo, %hi
	%mid = asr_l %mid, 1
	%val = loadl (%a + %mid * 8)
	%eq_val_key = ceql %val, %key
	jnz %eq_val_key, @found

	%cgt_val_key = csgtl %val, %key
	jnz %cgt_val_key, @go_left, @go_right

@go_right
	%new_lo = add l %mid, 1
	brt %new_lo, %hi, %loop
	jmp @check_result

@go_left
	cslt %lo, %mid, %new_hi
	brt %lo, %new_hi, %loop
	jmp @check_result

@found
	%result = movl %mid
	ret l %result

@check_result
	csetl %has_found = csltl (const w 0), (const w -1) ; This logic is flawed if result was never updated properly, but we init to -1. Actually, if loop exits without finding, result remains -1.
	; Wait, if n > 0 and not found, lo might become > hi. We need to ensure result stays -1.
	; The initialization %result = const l -1 handles the "not found" case naturally.
	ret l %result

@not_found
	ret w (const l -1)
}
