export function l $binary_search(l %a, l %n, l %key) {
@entry
	csltl (const w 0), (trunc l to w %n)
	jnz @not_found, @loop

@loop
	%lo = const l 0
	%hi = sub l %n, 1

@check
	cslt %lo, %hi, %cmp
	jz @not_found, @mid_calc

@mid_calc
	%sum = add l %lo, %hi
	%mid = asr_l %sum, 1
	%val = loadl (%a + %mid * 8)

	%eq = ceql %val, %key
	jnz @found, @compare

@less_than
	%new_hi = sub l %mid, 1
	cslt %lo, %new_hi, %cont_left
	jz @not_found, @loop
	%hi = %new_hi
	br @check

@greater_than
	%new_lo = add l %mid, 1
	cslt %new_lo, %hi, %cont_right
	jz @not_found, @loop
	%lo = %new_lo
	br @check

@compare
	csltl %val, %key, %is_less
	jnz @less_than, @greater_than

@found
	ret l %mid

@not_found
	ret w -1
}
