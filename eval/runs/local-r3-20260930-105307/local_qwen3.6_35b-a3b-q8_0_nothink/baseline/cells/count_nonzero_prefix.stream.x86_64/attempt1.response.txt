count_nonzero_prefix:
	test rsi, rsi
	je .L_done
	xor rax, rax
	mov r8, qword ptr [rdi]
	test r8, r8
	jz .L_return_rax
	inc rsi
	add rdi, 8
.L_loop:
	cmp rax, rsi
	jge .L_done
	mov r8, qword ptr [rdi + rax*8]
	test r8, r8
	jz .L_return_rax
	inc rax
	cmp rax, rsi
	jl .L_loop
.L_done:
	ret
.L_return_rax:
	ret
