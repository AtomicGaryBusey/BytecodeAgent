@one = const i64 1
@two = const i64 2

export function l $isqrt(l n) {
%start:
    %cmp_zero = ceql l n, @one
    jnz %check_gt_one
    
    ; If n is 0 or 1, sqrt(n) is n.
    ret l n

%check_gt_one:
    cmpgtl l n, @one
    jz %ret_n
    
    ; Newton's method.
    ; Start guess x = n / 2.
    %x = divll n, @two
    
@newton_loop:
    ; q = n / x
    %q = divll n, x
    
    ; next_x = (x + q) / 2
    %sum = addll x, q
    %next_x = ashrll sum, @1
    
    ; If next_x >= x, we have reached the minimum.
    cmpgeal l next_x, x
    jnz %found
    
    ; Else, continue iteration.
    %x = next_x
    jmp @newton_loop

%found:
    ret l x

%ret_n:
    ret l n
}
