@0 = const i64 0
@1 = const i64 1
@3 = const i64 3
@2 = const i64 2

export function l $isqrt(l n) {
%start:
    %cmp = ceql l @0, n
    jnz %is_zero
    
    ; Newton's method for integer square root.
    ; Start with a good estimate. For u64, max sqrt is 2^32 - 1.
    ; We can use binary search or a shifted starting point.
    ; Let's use an initial guess: if n >= 2^32, guess is high bits, else low.
    ; Simpler: start with 2^31 as a safe upper bound for all inputs except overflow cases?
    ; Actually, let's use a simple binary search range [0, 2^32 - 1].
    
    %low = addl l @0, n  ; Wait, n can be large. low=0.
    %low = addl l @0, n
    %low = subll l n, n   ; 0
    
    ; Upper bound: 2^32 is sufficient since (2^32)^2 > 2^64 - 1 is false?
    ; (2^32)^2 = 2^64. So sqrt(2^64-1) < 2^32.
    ; Max possible root is 4294967295 (0xFFFFFFFF).
    %high = addll n, n   ; This might overflow if n is huge? No, we want a fixed bound.
    ; Let's just use a large constant for high.
    
    ; Actually, Newton's method converges quadratically and handles 64-bit easily.
    ; Initial guess x0: can be obtained from floating point conversion or bit manipulation.
    ; Since no FP allowed, let's do binary search. It's robust and easy to implement in IL.
    
    ; Binary Search for floor(sqrt(n))
    ; Range [low, high] = [0, 2^32 - 1]
    
    %low = subll n, n      ; 0
    %high = subll n, n     ; Clear high
    
    ; Set high to 0x100000000 (2^32)
    %high_hi = const i64 1
    %high_lo = const i64 0
    ; QBE doesn't support immediate 64-bit constants easily split? 
    ; Actually `const i64` works for literal values.
    %high = addll n, n      ; Let's use a simpler way to get 2^32
    
    ; Re-eval: Just use a simple loop with shifting.
    ; Or Newton's method with integer arithmetic only.
    
    ; Newton's Method: x_{k+1} = (x_k + n / x_k) / 2
    ; We need a function for division or implement it? 
    ; QBE has `/` operator. But we need to avoid division by zero and ensure convergence.
    
    %x = subll n, n        ; Initial guess: let's pick something big enough but not zero.
    
    ; Better initial guess:
    ; If n is large, sqrt(n) <= 2^32 - 1.
    ; Let's start with x = 2^31 (0x8000000000000000? No, 2^31 is small for u64 range).
    ; Max n ~ 1.8e19. sqrt(n) ~ 4.2e9. 
    ; Let's start with x = (n >> 1) + 1 ? That might be too big initially but will converge.
    
    %shift_amount = const i64 63
    %max_bit = ashl n, @shift_amount ; Get the MSB position effect? No.
    
    ; Let's just use x = n if n < 2^32 else something larger.
    ; Actually, let's use the property that for n > 0, sqrt(n) is at most n/2 + 1.
    
    %x = divll n, @2       ; Initial guess x0 = n/2
    %one = const i64 1
    %x = addll x, @1       ; Ensure positive
    
@loop:
    %x_new = divll n, x
    %sum = addll x, x_new
    %half = ashl @1, @31   ; This is not correct shift. /2 for signed? 
    ; Integer division by 2 is right shift or shift left with carry?
    ; QBE `ashl` shifts logical. For dividing sum by 2: `ashrl` (arithmetic right) or just `(x+x_new)>>1`.
    ; Since x and x_new are positive, we can use arithmetic right shift or logic.
    
    ; We need to compute floor((x + n/x)/2).
    ; Let's use right shift by 1. QBE has `ashrl` (arithmetic shift right) and `lsr`? 
    ; Standard QBE: `ashrl` is arithmetic. `ushrl` might not be standard? 
    ; Usually `ashrl` for signed, but here unsigned. 
    ; Let's check QBE ops. It has `ashrl` (arithmetic shift right long).
    ; Since our numbers are positive, arithmetic and logical RHS are same.
    
    %shift_one = const i64 1
    %x_new = divll n, x
    %sum = addll x, x_new
    
    ; Check for convergence or overflow in sum?
    ; If x_new >= x, we have converged (for integer Newton).
    ; Because x_{k+1} >= sqrt(n) >= x_k when approaching from above.
    ; Once x_{k+1} <= x_k, we are at the minimum.
    
    %cmp = csgtl x_new, x   ; if x_new > x
    jnz %next_iter
    
    %result = x
    ret l %result

@next_iter:
    %x = ashrll sum, @1      ; Divide sum by 2 using arithmetic right shift
    jz %loop                 ; Infinite loop guard? No, Newton converges.
                             ; But QBE `jnz` is needed. Let's restructure.
    
    ; Restructure loop for QBE which lacks unconditional jump to label inside?
    ; It has `jmp`.
    jmp @loop

@is_zero:
    ret l @0
}
