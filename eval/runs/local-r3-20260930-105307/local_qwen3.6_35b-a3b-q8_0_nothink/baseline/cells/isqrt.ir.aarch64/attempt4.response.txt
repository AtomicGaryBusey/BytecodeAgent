export function l $isqrt(l n) {
@entry:
    %zero = const i64 0
    %one = const i64 1
    
    ; If n <= 1, return n
    %cmp_zero = ceql l n, @zero
    jnz @ret_one_or_zero
    
    %cmp_gt_one = csgtl l n, @one
    jz @ret_one
    
    ; Binary search for floor(sqrt(n))
    ; Range [low, high]
    ; low = 1
    ; high = n / 2 + 1 (since sqrt(n) <= n/2 for n >= 4, and we handled small n)
    ; Actually, for n=2,3 sqrt is 1. n/2 + 1 covers it.
    
    %two = const i64 2
    %high = addll divll n, @two, @one
    
    ; But wait, if n is very large (e.g. 2^64-1), high might overflow if we are not careful?
    ; n/2 + 1 fits in 64 bits for any unsigned 64-bit n except 0 (handled) and maybe max uint64_t?
    ; Max uint64_t is 2^64-1. (2^64-1)/2 + 1 = 2^63 - 1 + 1 = 2^63. Fits in unsigned long (signed positive).
    ; QBE `l` is signed? Usually in QBE `l` is a native machine word, treated as two's complement for arithmetic.
    ; Since we are doing unsigned logic, we need to be careful with comparisons.
    ; csgtl is signed greater than. cugtl is unsigned greater than (if available) or use specific ops.
    ; QBE has: `csltl` (signed less), `csrtl` (signed less or equal?), `ceql`, etc.
    ; For unsigned, we typically rely on the fact that large signed numbers are negative? 
    ; No, we should stick to operations that work correctly for unsigned values or map them.
    ; QBE instructions:
    ; addl, subl, mulll (signed multiply low?), divll (signed div?)
    ; The prompt says "Integer/pointer arguments...". QBE usually treats `l` as signed integers for arithmetic unless specified.
    ; However, sqrt of unsigned max requires unsigned logic or careful handling.
    ; Let's use binary search with unsigned semantics mapped to signed if possible, or just assume `divll` etc work on the bit pattern as we want.
    ; Actually, `divll` in QBE is signed division? If so, -1 / 2 = 0 (trunc towards zero). 
    ; Unsigned n/2 should be floor(n/2). For positive numbers, signed and unsigned div are same.
    ; n is always >= 0. So signed div works fine for n/2.
    
    %low = const i64 1
    
@bs_loop:
    ; Check if low > high
    %cmp_gt = csgtl l high, @low 
    ; Wait, csgtl is "signed greater than". If high > low, this is true?
    ; We want to loop while low <= high. So we exit when low > high.
    ; Let's use `csrtl` (signed less or equal)? QBE spec:
    ; `ceql` eq, `csltl` <, `csrtl` <=, `csgtl` >, `csgtl` >= ?
    ; Actually standard QBE ops are: ceql, csltl, csrtl, csgtl, cugtl, etc?
    ; The prompt doesn't list all ops. Assuming standard ones: 
    ; If I don't know unsigned compare, I can implement unsigned > using signed if numbers are positive (which they are).
    ; Since low and high are positive (low=1, high <= 2^63), signed comparison is identical to unsigned.
    
    %exit = csgtl l low, high 
    ; If low > high, we exit? No, if low > high, loop ends.
    ; Condition: while (low <= high) { ... }
    ; So we continue if NOT (low > high).
    ; Let's check `csltl` (less than). If low < high or low == high, we continue.
    
    %continue = csrtl l low, high
    jz @bs_done
    
    %mid = addll divll addll low, high, @one, @two
    
    ; mid * mid <= n ?
    ; Check for overflow before multiplying? 
    ; If mid > sqrt(2^64-1) ~ 4.29e9, mid*mid might overflow unsigned 64-bit if we are not careful?
    ; Wait, max result is 2^32 - 1 approx. 
    ; If mid is large (close to 2^63), mid*mid will definitely overflow 64 bits.
    ; But our search range high is at most ~2^63. 
    ; We need to check if mid * mid <= n without full 128-bit arithmetic if possible, or handle the overflow case.
    
    ; Safe check: if mid > n, then mid*mid > n (since n>=1).
    ; More precisely: if mid > 0 and mid > n / mid, then mid*mid > n.
    
    %is_mid_gt_n = csgtl l mid, n
    
    %q = divll n, mid
    %mult_overflows_or_exceeds = orl is_mid_gt_n, csgtl l mid, q
    
    ; If mult_overflows_or_exceeds is true (mid*mid > n), then high = mid - 1
    jnz @bs_low_set_high
    
    ; Else low = mid + 1? No.
    ; If mid*mid <= n, candidate answer is at least mid. So low = mid + 1 to search higher?
    ; Standard binary search for largest x such that x*x <= n:
    ; if mid*mid <= n: ans = mid, low = mid + 1
    ; else: high = mid - 1
    
    %ans = const i64 -1 
    ; We need to return the answer. Let's keep track of `ans`.
    
    ; Store candidate in a temp var? QBE doesn't have memory ops inside this simple function easily without alloc?
    ; Ah, I can use stack? No, "No data definitions...".
    ; But I can spill to stack if needed. Or just maintain `ans` in a register.
    
    ; Let's refine the loop variables: low, high, ans.
    ; Initially ans = 0 (or 1 since n>=2).
    %ans = const i64 1

@bs_low_set_high_false: ; Fallthrough from no overflow check
    
    ; If we didn't jump to bs_low_set_high, it means mid*mid <= n.
    %ans = mid
    %new_low = addll mid, @one
    %low = new_low
    jmp @bs_loop

@bs_low_set_high:
    ; mid*mid > n (or overflow)
    %high = subll mid, @one
    jmp @bs_loop

@bs_done:
    ret l ans

@ret_one_or_zero:
    ret l n
    
@ret_one:
    ret l one
}
