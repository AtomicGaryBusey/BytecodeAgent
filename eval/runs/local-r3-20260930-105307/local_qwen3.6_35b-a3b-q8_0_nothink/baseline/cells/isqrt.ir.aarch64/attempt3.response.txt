export function l $isqrt(l n) {
@entry:
    %zero = const i64 0
    %one = const i64 1
    
    ; If n <= 1, return n
    %cmp = ceql l n, @zero
    jnz @ret_n
    %cmp_gt_one = csgtl l n, @one
    jz @ret_n
    
    ; Initial guess: x = n / 2
    %two = const i64 2
    %x = divll n, @two
    %prev_x = const i64 0

@newton_loop:
    ; q = n / x (safe since x >= 1)
    %q = divll n, x
    
    ; next_x = (x + q) / 2
    %sum = addll x, q
    %next_x = ashrll sum, @one
    
    ; If next_x >= x, we are at or below the root.
    ; Actually, Newton's method for integer sqrt converges such that
    ; it oscillates or reaches the floor. We want the largest r where r*r <= n.
    ; The standard integer Newton loop:
    ; while (x > 0 && x > n/x) x = (x + n/x)/2;
    ; But simpler: just iterate until convergence.
    
    %cmp_next_ge_x = csgtl l next_x, x
    jz @converged
    
    %prev_x = x
    %x = next_x
    jmp @newton_loop

@converged:
    ; Check if next_x is too small (next_x^2 <= n) vs current x?
    ; Actually, with integer division, Newton's method may overshoot slightly below the true root 
    ; or land on it. The sequence decreases monotonically until it might jump back up by 1 if we are not careful,
    ; but typically for sqrt, it converges to floor(sqrt(n)).
    ; Let's verify: is q*q <= n?
    ; If next_x * next_x > n, then x = next_x - 1.
    ; However, a robust integer sqrt using Newton usually ends with x such that x*x <= n < (x+1)*(x+1).
    
    ; Let's check if next_x is actually better or if we should stick with x.
    ; In many implementations, the final answer is min(x, next_x) after convergence check fails.
    ; But here jz @converged means next_x <= x. Since it was strictly decreasing before, 
    ; next_x == x usually means we found it. If next_x < x, it might be an oscillation if not handled.
    ; Let's use a safer integer Newton approach:
    
    ; Re-evaluate strategy: Standard binary search is safer against overflow issues and simpler to reason about.
    ; But let's stick to the Newton result if possible. 
    ; Actually, the loop condition `next_x >= x` (jnz) means we continue while strictly decreasing? No.
    ; If next_x < x, we converged.
    
    ; Wait, my jz @converged branches when next_x <= x.
    ; If next_x == x, we are done.
    ; If next_x < x, it might be that we overshot below? 
    ; For sqrt(n), f(x) = (x + n/x)/2. If x > sqrt(n), then next_x < x and next_x >= sqrt(n).
    ; So it converges from above. Thus next_x == x is the fixed point.
    
    ret l x

@ret_n:
    ret l n
}
