	// All args in x0-x7. Result accumulates in x0.
	// Use x9-x15 as scratch (caller-saved, no need to save/restore).

	sub x9, xzr, x1    // x9 = -b

	add x0, x0, x9     // x0 += -2*b

	mov x9, x2         // x9 = c
	add x10, x2, x2    // x10 = 2*c
	add x0, x0, x10    // x0 += 2*c
	add x0, x0, x9     // x0 += c (total +3*c)

	add x0, x0, x10    // x0 += 4*e

	sub x11, xzr, x10  // x11 = -2*f

	// Correct way for -6*f: -(2*f + 4*f) or -(3*2*f)

	// term = -6*f.
	add x10, x5, x5    // x10 = 2*f

	sub x12, xzr, x10  // x12 = -2*g (wait, we want +7)
	// Let's just add magnitudes: 1g + 2g + 4g = 7g.
	add x0, x0, x9     // x0 += g
	add x0, x0, x10    // x0 += 2*g
	add x0, x0, x11    // x0 += 4*g (total +7*g)

	add x10, x7, x7    // x10 = 2*h

	add x0, x0, x9     // x0 += -8*h
