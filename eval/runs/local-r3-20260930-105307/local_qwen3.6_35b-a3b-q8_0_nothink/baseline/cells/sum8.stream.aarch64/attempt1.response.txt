sum8:
	// a is in x0, b in x1, c in x2, d in x3
	// e in x4, f in x5, g in x6, h in x7
	// Result = a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h

	// Compute -2*b: negate b, then add b again? No.
	// Let's do it term by term accumulating in x0.
	// Current x0 = a
	
	// Term -2*b: x1 -> x0 - 2*x1
	sub x18, x1, x1      // x18 = 2*b (actually we want -2b, so let's keep sign separate or adjust)
	add x18, x18, x1     // x18 = 3*b ... wait.
	// Easier: negate b first to get -b, then add b to get -0? No.
	// To get -2*b: sub x19, xzr, x1 (x19 = -b). Then sub x19, x19, x1 (x19 = -2*b).
	sub x19, xzr, x1     // x19 = -b
	sub x19, x19, x1     // x19 = -2*b
	add x0, x0, x19      // x0 = a - 2*b

	// Term +3*c: c + 2*c or c*3.
	// add x20, x2, x2 (x20 = 2c), then add x20, x20, x2 (x20 = 3c).
	add x20, x2, x2      // x20 = 2*c
	add x20, x20, x2     // x20 = 3*c
	add x0, x0, x20      // x0 = a - 2*b + 3*c

	// Term -4*d: -(4*d).
	// neg d -> -d. add d -> 0? No.
	// sub xzr, x3 (x21 = -d). sub x21, x21, x3 (x21 = -2d). sub x21, x21, x3 (x21 = -3d). sub x21, x21, x3 (x21 = -4d).
	// Or: mul w22, wzr, x3 ? No immediate 4 is small but MUL uses registers.
	// Let's use repeated add/sub.
	neg x21, x3          // x21 = -d
	add x21, x21, x3     // x21 = 0? No neg(d) + d = 0. This is wrong.
	
	// Correct logic for -4*d:
	// temp = d + d = 2d. temp2 = 2d + 2d = 4d. result = -4d = -temp2.
	add x21, x3, x3      // x21 = 2*d
	sub x21, xzr, x21    // x21 = -2*d? No, we want -4d.
	// Let's just do: temp = 4*d then neg it.
	add x21, x3, x3      // x21 = 2*d
	add x21, x21, x21    // x21 = 4*d
	neg x21, x21         // x21 = -4*d
	add x0, x0, x21      // x0 = a - 2*b + 3*c - 4*d

	// Term +5*e: e*5.
	// temp = 4*e (from above method on x4). add e.
	add x22, x4, x4      // x22 = 2*e
	add x22, x22, x22    // x22 = 4*e
	add x22, x22, x4     // x22 = 5*e
	add x0, x0, x22      // x0 = ... + 5*e

	// Term -6*f: -(6*f).
	// temp = 2*f. temp2 = 3*f (temp+temp2? no temp*1.5).
	// f*6 = f * 2 * 3.
	add x23, x5, x5      // x23 = 2*f
	add x24, x23, x23    // x24 = 4*f? No.
	// Let's do 2f + 4f = 6f.
	add x23, x5, x5      // x23 = 2*f
	sub x23, xzr, x23    // x23 = -2*f ... wait sign is tricky with accumulation.
	// Better: calculate positive magnitude then negate if needed at end? 
	// Or just accumulate signed terms.
	
	// -6*f:
	add x23, x5, x5      // x23 = 2*f
	add x24, x23, x23    // x24 = 4*f
	add x23, x23, x24    // x23 = 6*f
	neg x23, x23         // x23 = -6*f
	add x0, x0, x23      // x0 = ... - 6*f

	// Term +7*g: g*7.
	// 4g + 2g + g = 7g.
	add x24, x6, x6      // x24 = 2*g
	add x25, x24, x24    // x25 = 4*g
	add x24, x24, x25    // x24 = 6*g
	add x24, x24, x6     // x24 = 7*g
	add x0, x0, x24      // x0 = ... + 7*g

	// Term -8*h: -(8*h).
	// h*8 is easy via shift or repeats.
	add x25, x7, x7      // x25 = 2*h
	add x25, x25, x25    // x25 = 4*h
	add x25, x25, x25    // x25 = 8*h
	neg x25, x25         // x25 = -8*h
	add x0, x0, x25      // x0 = final result

	ret
