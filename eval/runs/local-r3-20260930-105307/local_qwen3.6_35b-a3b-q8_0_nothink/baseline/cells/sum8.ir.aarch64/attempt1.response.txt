export function l $sum8(l a, l b, l c, l d, l e, l f, l g, l h) {
@entry
	%neg_b = subw l -2, b @ We need to be careful with immediate handling in QBE
	%term1 = mul l a, %neg_b
	%term2_3 = sub l 0, c
	%add_c = addl %term1, %term2_3
	
	@ Actually, let's do the math directly with available ops.
	@ Result = a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h
	
	%double_b = shl b, 1
	%sub_b = subl a, %double_b
	
	%triple_c = addl c, c
	%add_c = addl %sub_b, c
	
	%quad_d = shl d, 2
	%sub_d = subl %add_c, %quad_d
	
	%five_e = addl e, e
	%add_e = addl %five_e, e
	%add_e2 = addl %sub_d, %add_e
	
	%six_f = shl f, 1
	%double_f = shl %six_f, 1 @ This is wrong. 6*f = 4*f + 2*f or just sub/fadd loops.
                           @ QBE doesn't have imul immediate for non-power-of-2 generally without registers?
                           @ Let's use register multiplication to be safe and clear.

	%term_a = lmul l a, l 1
	%term_b = lmul l b, l -2
	%term_c = lmul l c, l 3
	%term_d = lmul l d, l -4
	%term_e = lmul l e, l 5
	%term_f = lmul l f, l -6
	%term_g = lmul l g, l 7
	%term_h = lmul l h, l -8
	
	%sum1 = addl %term_a, %term_b
	%sum2 = addl %sum1, %term_c
	%sum3 = addl %sum2, %term_d
	%sum4 = addl %sum3, %term_e
	%sum5 = addl %sum4, %term_f
	%sum6 = addl %sum5, %term_g
	%result = subl %sum6, %term_h
	
	ret l %result
}
