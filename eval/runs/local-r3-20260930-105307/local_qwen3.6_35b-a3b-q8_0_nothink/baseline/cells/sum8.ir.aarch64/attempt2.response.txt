export function l $sum8(l a, l b, l c, l d, l e, l f, l g, l h) {
@entry
	%tb = shl b, 1
	%su1 = sub l a, %tb
	
	%tc = addl c, c
	%sc = addl %su1, %tc
	
	%td = shl d, 2
	%sd = subl %sc, %td
	
	%te0 = addl e, e
	%te = addl %te0, %te0
	%se = addl %sd, %te
	
	%tf0 = shl f, 1
	%tf1 = shl %tf0, 1
	%tf = addl %tf0, %tf1
	%sf = subl %se, %tf
	
	%tg0 = addl g, g
	%tg = addl %tg0, %tg0
	%sg = addl %sf, %tg
	
	%th = shl h, 3
	%result = subl %sg, %th
	
	ret l %result
}
