export function l $bmax(l %a, l %n) {
@start
	%ptr = loadl (%a)
	%max = loadl [%ptr]
	%n1 = subll %n, i 1
	csetle @done
	jnz @return_max
@loop
	%n1 = subll %n1, i 1
	csetge @cont
	jz @return_max
	%idx = addll %n1, i 1
	%ptr2 = getelementptr l %a, l %idx
	%val = loadl [%ptr2]
	csltl %max, %val
	jnz @next
@store:
	storel %val, @max
@next:
	jmp @loop
@return_max:
	ret l %max
}
