export function l $crc32(l %p, l %n) {
@entry
  storel 8(%sp), %p
  storel 0(%sp), %n
  storew -4(%sp), -1w
  storel -12(%sp), 7l
  jmp @check_byte

@check_byte
  %cur_n = loadl 0(%sp)
  jz @calc_final
  jmp @process_byte

@process_byte
  ; Load byte and zero extend to w
  %b = extuw (loadub (%p))
  
  ; XOR into CRC
  %crc_w = loadw -4(%sp)
  %crc_w = xor w, %crc_w, %b
  
  ; Reset bit counter
  storel -12(%sp), 7l
  jmp @check_bit

@check_bit
  %bit_cnt = loadl -12(%sp)
  jz @next_byte
  jmp @process_bit

@process_bit
  %crc_l = extuw %crc_w
  %lsb = and l, %crc_l, 1l
  
  ; If LSB is 0, just shift logical right (QBE `>>` on unsigned long is logical)
  ; If LSB is 1, shift and XOR with polynomial 0xEDB88320
  
  jnz @bit_is_one, @bit_is_zero

@bit_is_one
  %crc_l >>= 1
  %poly = -333923440l ; 0xEDB88320 interpreted as positive signed 64-bit int (high bit 0)
  %crc_l = xor l, %crc_l, %poly
  jmp @save_bit_result

@bit_is_zero
  %crc_l >>= 1
  jmp @save_bit_result

@save_bit_result
  %new_crc_w = extuw %crc_l
  storew -4(%sp), %new_crc_w
  
  ; Decrement bit counter
  %new_cnt = subu l, %bit_cnt, 1l
  storel -12(%sp), %new_cnt
  jmp @check_bit

@next_byte
  ; Decrement byte counter
  %cur_n = loadl 0(%sp)
  %next_n = subu l, %cur_n, 1l
  storel 0(%sp), %next_n
  
  ; Increment pointer
  %cur_p = loadl 8(%sp)
  %next_p = add l, %cur_p, 1l
  storel 8(%sp), %next_p
  
  jmp @check_byte

@calc_final
  %crc_w = loadw -4(%sp)
  %result_w = xor w, %crc_w, -1w
  ret l, %result_w
}
