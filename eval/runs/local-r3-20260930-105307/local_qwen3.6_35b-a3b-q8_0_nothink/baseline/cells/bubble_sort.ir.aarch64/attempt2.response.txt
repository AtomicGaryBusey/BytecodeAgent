export function l $bubble_sort(l %a, l %n) {
@start
	%limit = sub l, %n, i64 1
	brslt @done, %limit
@outer_loop
	%j = alloca l
	storel i64 -1, %j
@inner_start
	%jj = loadl %j
	%inc_j = add l, %jj, i64 1
	storel %inc_j, %j
	cgell @outer_end, %limit, %inc_j
@inner_loop
	%idx1 = loadl %j
	%p1 = getelementptr l, l %a, l %idx1
	%s1 = loadl %p1
	%idx2 = add l, %idx1, i64 1
	%p2 = getelementptr l, l %a, l %idx2
	%s2 = loadl %p2
	csltl @swap_needed, %s1, %s2
	br @inner_done
@swap_needed
	storel %s2, %p1
	storel %s1, %p2
@inner_done
	br @inner_start
@outer_end
	%new_limit = sub l, %limit, i64 1
	cgell @outer_loop, i64 -1, %new_limit
@done
	ret i64 0
}
