export function l $bubble_sort(l %a, l %n) {
@start
	brslt @done, %n
	%limit = sub l, %n, i64 1
	csltl @skip_init, i64 -1, %limit
	storel i64 -1, %limit
@skip_init
@outer_loop
	%j = alloca l
	%jj = loadl %limit
	storel %jj, %j
	br @inner_start
@inner_done
	%new_limit = sub l, %limit, i64 1
	storel %new_limit, %limit
	cgell @outer_loop, i64 -1, %new_limit
	ret i64 0
@swap_needed
	%idx1 = loadl %j
	%p1 = getelementptr l, l %a, l %idx1
	%s1 = loadl %p1
	%idx2 = add l, %idx1, i64 1
	%p2 = getelementptr l, l %a, l %idx2
	%s2 = loadl %p2
	storel %s2, %p1
	storel %s1, %p2
@inner_start
	%jj = loadl %j
	cgell @outer_done, %limit, %jj
@inner_loop
	%idx1 = loadl %j
	%p1 = getelementptr l, l %a, l %idx1
	%s1 = loadl %p1
	%idx2 = add l, %idx1, i64 1
	%p2 = getelementptr l, l %a, l %idx2
	%s2 = loadl %p2
	csltl @swap_needed, %s1, %s2
@inner_next
	%jj = loadl %j
	%inc_j = add l, %jj, i64 1
	storel %inc_j, %j
	br @inner_done
@outer_done
	ret i64 0
@done
	ret i64 0
}
