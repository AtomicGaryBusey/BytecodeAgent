export function l $bubble_sort(l %a, l %n) {
@start
	brz @ret_zero, %n
	%limit = sub l, %n, i64 1
	cgell @outer_loop, %limit, i64 0
	br @done_limit
@outer_loop
	%j = alloca l
	storel i64 0, %j
@inner_start
	%idx = loadl %j
	%cidx = add l, %idx, i64 1
	cgell @inner_exit, %cidx, %limit
	%p1 = getelementptr l, l %a, l %idx
	%s1 = loadl %p1
	%p2 = getelementptr l, l %a, l %cidx
	%s2 = loadl %p2
	csltl @do_swap, %s1, %s2
	br @inner_next
@do_swap
	storel %s2, %p1
	storel %s1, %p2
@inner_next
	storel %cidx, %j
	br @inner_start
@inner_exit
	%new_limit = sub l, %limit, i64 1
	storel %new_limit, %limit
	cgell @outer_loop, %new_limit, i64 0
	ret i64 0
@done_limit
	storel i64 0, %limit
	br @outer_loop
@ret_zero
	ret i64 0
}
