isqrt:
	// Calculate sqrt(n) for 64-bit unsigned integer using binary search.
	// We maintain [low, high] such that low*low <= n < high*high.
	// To avoid overflow when computing mid*mid, we can work with division or use carry-aware multiplication.
	// A safer method for full 64-bit range is to determine bits from MSB down.
	
	// Handle edge cases
	mov	x19, x0		// n = x0
	cbnz	x19, .Lnon_zero
	ret			// sqrt(0) = 0

.Lnon_zero:
	// Find the most significant bit position to initialize high.
	// Alternatively, use Newton's method with overflow protection or binary search on bits.
	// Binary search approach:
	// low = 1
	// high = n (but capped to avoid excessive iterations? No, log2(2^64)=64 iters max)
	// But mid*mid overflows.
	
	// Let's use a bit-by-bit construction method which avoids large intermediate squares.
	// We build the result bit by bit from MSB to LSB.
	
	mov	x19, x0		// n
	mov	x20, 0		// result (ans)
	mov	x21, 1UL << 62	// trial_square (initially 2^62)
	
.Lloop:
	cbnz	x21, .Lcheck_bit
	ret			// Done

.Lcheck_bit:
	// We want to see if ans + trial_square gives a square <= n.
	// New candidate: cand = ans + trial_square
	// Check: cand * cand <= n
	// (ans + ts)^2 = ans^2 + 2*ans*ts + ts^2
	// Since we process bits from high to low, we can maintain the remainder or just compute.
	// Computing (ans+ts)*(ans+ts) might overflow 64-bit unsigned if not careful.
	// However, since n < 2^64, and the answer is < 2^32, actually ans < 2^32 at this stage?
	// No, final ans < 2^32. During build, ans can be up to ~2^31 + ...
    // Max value of ans is roughly sqrt(n). ts starts at 2^62.
    // If we try bit 62, ans=0, ts=2^62, cand=2^62, cand^2 = 2^124 (overflow).
    
    // Better approach: Standard bitwise integer sqrt algorithm.
    // For each bit position i from 31 down to 0:
    //   candidate = ans | (1 << i)
    //   if candidate * candidate <= n:
    //     ans = candidate
    
    // Since max result is 2^32 - 1, we only need to check bits 31..0.
    
	mov	x20, xzr	// ans = 0
	mov	x21, 31		// bit index
    
.L_bit_loop:
	cbnz	x21, .Lnext_bit
	
.L_return_ans:
	movn	x0, x20, lsl #0 // Copy ans to return
	ret
