isqrt:
	// Integer square root using binary search.
	// Result is in [0, 2^32-1] since (2^32)^2 > 2^64-1.
	// We use 128-bit arithmetic for the mid*mid check to avoid overflow.
	
	mov	x19, xzr		// low = 0
	mov	x20, 1UL << 32	// high = 2^32 (exclusive upper bound)
	
.Lloop:
	cmp	x19, x20
	b.hs	.Ldone
	// mid = (low + high) >> 1
	add	x21, x19, x20
	lsr	x21, x21, #1
	
	// Check if mid * mid <= n
	// Use umull/u64mul to get high part of product.
	// If high part > 0, then product > 2^64-1 (assuming mid < 2^32).
	// If mid < 2^32, mid*mid fits in 64 bits only if mid < 2^32.
	// Wait, max mid is 2^31 roughly? No, mid can be up to 2^32-1.
	// (2^32-1)^2 = 2^64 - 2^33 + 1 which overflows 64 bits.
	// So we must use 128-bit comparison.
	
	umull	x22, x23, x21, x21 // x23:x22 = mid * mid
	
	c
