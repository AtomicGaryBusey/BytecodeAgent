isqrt:
	// Integer square root using binary search.
	// Result is in [0, 2^32-1] since (2^32)^2 > 2^64-1.
	// We use 128-bit arithmetic for the mid*mid check to avoid overflow.
	
	mov	x19, xzr		// low = 0
	mov	x20, 1UL << 32	// high = 2^32 (exclusive upper bound)
	
.Lloop:
	cmp	x19, x20
	b.hs	.Ldone
	// mid = (low + high) >> 1
	add	x21, x19, x20
	lsr	x21, x21, #1
	
	// Check if mid * mid <= n
	// Use umull to get low and high parts of 64x64->128 product.
	// x23:x22 = mid * mid
	umull	x22, x23, x21, x21
