isqrt:
	// Integer square root using binary search on bits.
	// Since sqrt(2^64-1) < 2^32, the result fits in 32 bits.
	// We can construct the answer bit by bit from MSB (bit 31) to LSB (bit 0).
	
	mov	x19, xzr	// ans = 0
	mov	x20, 1UL << 31 // trial bit
	
.Lloop:
	cbnz	x20, .Lnext_bit
	ret			// return ans in x0

.Lnext_bit:
	// Check if (ans | trial_bit)^2 <= n
	// Let cand = ans | trial_bit
	// We need to check if cand * cand <= n
	// To avoid overflow of 128-bit intermediate result, we can use a helper 
	// or simply notice that since we are building from high bits, 
	// we can maintain the remainder.
	
	// However, standard bitwise sqrt algorithm:
	// remainder = n
	// For bit = 31 downto 0:
	//   subtrahend = (ans << 1) | trial_bit  <-- this is actually for a different variant
	//   
	// Let's use the property: 
	// We want to find largest 'a' such that a*a <= n.
	// Try setting bit b in a. New_a = a + 2^b.
	// Check if (New_a)^2 <= n.
	// (a + 2^b)^2 = a^2 + 2*a*2^b + 2^(2b)
	// 
	// Alternative: Use 128-bit arithmetic emulation using two 64-bit registers if needed, 
	// but AArch64 doesn't have native UMOVD/NMULL that outputs full 128 bit easily in scalar context without SIMD.
	// 
	// Simpler approach for full range: Newton's method with overflow check or just binary search 
	// with safe multiplication.
	// Given the constraints, let's use a safe binary search over [0, 2^32-1].
	// low=0, high=2^32 (exclusive)
	// while low < high:
	//   mid = (low + high) / 2
	//   if mid*mid > n: high = mid
	//   else: low = mid + 1
	// return low - 1
	
	mov	x0, xzr		// n (argument) is already in x0, save it
	mov	x19, x20	// wait, we destroyed x20? No, let's restart cleanly.
