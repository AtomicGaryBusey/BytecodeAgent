export function l $fib_rec(l) {
@start
    %c = csltl l $0, l %n
    jnz %c @base_case
@recursive
    %sub1 = subl l %n, l $1
    %r1 = call $fib_rec(l %sub1)
    %sub2 = subl l %n, l $2
    %r2 = call $fib_rec(l %sub2)
    %result = addl l %r1, l %r2
    ret l %result
@base_case
    %c0 = ceql l %n, l $0
    jnz %c0 @ret_zero
@ret_one
    ret l $1
@ret_zero
    ret l $0
}
