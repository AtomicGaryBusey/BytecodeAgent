bmax:
	// x0 = pointer to array a
	// x1 = n (count), n >= 1
	cmp	x1, #1
	b.le	.Lload_first
	ldr	x2, [x0], #8 // max = a[0], advance ptr
	sub	x1, x1, #1   // decrement count
.Lloop:
	cbz	x1, .Ldone
	ldr	x3, [x0], #8 // load next element
	cmp	x3, x2
	csel	x2, x3, x2, hi // max = (new > max) ? new : max
	sub	x1, x1, #1
	b	.Lloop
.Lload_first:
	cbnz	x1, .Lcont
	ldr	x0, [x0]
	ret
.Lcont:
	ldr	x2, [x0], #8 // max = a[0], advance ptr
	sub	x1, x1, #1   // decrement count
	b	.Lloop
.Ldone:
	mov	x0, x2
	ret
