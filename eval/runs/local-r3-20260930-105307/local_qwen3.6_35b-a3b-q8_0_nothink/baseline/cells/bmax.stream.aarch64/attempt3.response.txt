bmax:
	// x0 = pointer to array a
	// x1 = n (count), n >= 1
	cbz	x1, .Lerror_n_zero
	cmp	x1, #1
	b.eq	.Lload_first
	ldr	x2, [x0]       // max = a[0]
	sub	x1, x1, #1     // remaining count
	add	x0, x0, #8     // advance ptr to a[1]
.Lloop:
	cbz	x1, .Ldone
	ldr	x3, [x0], #8   // load next element
	cmp	x3, x2
	csel	x2, x3, x2, gt // max = (new > max) ? new : max
	sub	x1, x1, #1
	b	.Lloop
.Lload_first:
	ldr	x0, [x0]
	ret
.Ldone:
	mov	x0, x2
	ret
.Lerror_n_zero:
	// n >= 1 per spec, but handle gracefully by returning 0 or undefined
	mov	x0, #0
	ret
