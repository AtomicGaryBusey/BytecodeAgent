bmax:
	// x0 = pointer to array a
	// x1 = n (count), n >= 1
	mov	x2, #0		// i = 0
	ldr	x3, [x0]	// max = a[0]
	cbnz	x1, .Lcheck_more
	b	.Ldone

.Lcheck_more:
	sub	x4, x1, #1	// remaining = n - 1
	cbz	x4, .Ldone	// if only one element, done

.Lloop:
	add	x2, x2, #1	// i++
	cmp	x2, x1
	b.hs	.Ldone
	ldr	x5, [x0, x2, lsl #3] // load a[i]
	cmp	x5, x3
	stp	x4, x6, [sp, #-16]! // save scratch and adjust sp
	mov	x4, x3
	mov	x5, x5
	sub	x3, x2, #1
	add	x2, x6, #0
	ldp	x4, x6, [sp], #16 // restore
	csel	x3, x3, x5, hi   // max = (a[i] > max) ? a[i] : max
	b	.Lloop

.Ldone:
	ret
