export function l $reverse(l %a, l %n) {
@start:
    ; Calculate pointer to the last element: a + (n - 1)
    ; If n <= 1, no swapping needed, just return 0
    cmp le %n, 2
    jnz @return_zero

    ; Initialize pointers
    ; rsi = &a[n-1] = a + n - 1
    sub le %n, 1
    addl %a, %n
    lptr %last_ptr, %n
    
    ; rdi = &a[0] (already in %a)

@loop:
    ; Calculate number of swaps needed: n / 2
    ; We can use the original n or compute remaining steps.
    ; Let's keep track of how many elements are left to process.
    ; Actually, it's easier to just iterate until ptr_first >= ptr_last
    
    ; Load first element
    loadl %val1, %a
    
    ; Load last element
    loadl %val2, %last_ptr
    
    ; Compare pointers: if a > last_ptr, we are done (or equal if odd length)
    cmp gt %a, %last_ptr
    jnz @done
    
    ; Store val2 into first position
    storel %val2, %a
    
    ; Store val1 into last position
    storel %val1, %last_ptr
    
    ; Increment first pointer
    addq $8, %a
    
    ; Decrement last pointer
    subq $8, %last_ptr
    
    jmp @loop

@return_zero:
    movl $0, %result
    ret

@done:
    movl $0, %result
    ret
}
