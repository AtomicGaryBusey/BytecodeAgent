export function l $reverse(l %a, l %n) {
@start:
    cmp le %n, 1
    jnz @process

@return_zero:
    movl $0, %result
    ret

@process:
    ; n is the number of elements.
    ; We need to swap a[i] with a[n-1-i].
    ; Number of swaps is n / 2.
    ; We can loop i from 0 to n/2 - 1.
    
    ; Calculate limit = n / 2 (integer division)
    movl %n, %i
    sarl $1, %i
    
    cmp le %i, 0
    jnz @swap_loop

@return_zero:
    movl $0, %result
    ret

@swap_loop:
    ; Calculate pointer to current first element: a + i
    shlq $3, %i
    addl %a, %temp_a
    movl %temp_a, %ptr_first
    
    ; Calculate pointer to current last element: a + n - 1 - i
    ; = (a + n - 1) - i
    movl %n, %temp_n
    subq $1, %temp_n
    shlq $3, %temp_n
    addl %a, %ptr_last
    movl %ptr_last, %orig_ptr_last
    
    ; Subtract offset for i from last pointer
    movl %i, %offset
    shlq $3, %offset
    subq %offset, %ptr_last

    ; Load values
    loadl %val_first, %ptr_first
    loadl %val_last, %ptr_last

    ; Store swapped values
    storel %val_last, %ptr_first
    storel %val_first, %ptr_last

    ; Decrement swap counter
    subq $1, %i
    
    cmp gt %i, 0
    jnz @swap_loop

@return_zero_final:
    movl $0, %result
    ret
}
