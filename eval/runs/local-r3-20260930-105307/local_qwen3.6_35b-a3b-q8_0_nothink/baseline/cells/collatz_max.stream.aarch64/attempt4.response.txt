collatz_max:
	cbnz	x0, .Lstart
	mov	x0, #0
	ret
.Lstart:
	mov	x19, #1			// k = 1
	mov	x20, #0			// max_steps = 0
.c_kloop:
	cmp	x19, x0
	b.gt	.L_return_max
	mov	x21, x19			// current value for Collatz sequence
	mov	x22, #0			// steps for current k
.L_collatz_loop:
	cbnz	x21, .L_not_one
	b	.L_done_k
.L_not_one:
	tbnz	x21, #0, .L_odd
	lsr	x21, x21, #1		// x / 2
	b	.L_after_step
.L_odd:
	add	x23, x21, x21		// x * 2
	add	x21, x23, x21		// 3 * x + 1
.L_after_step:
	add	x22, x22, #1
	b	.L_collatz_loop
.L_done_k:
	cmp	x20, x22
	b.hs	.L_next_k
	mov	x20, x22			// update max_steps
.L_next_k:
	add	x19, x19, #1
	b	.c_kloop
.L_return_max:
	mov	x0, x20
	ret
