collatz_max:
	// x0 = n
	cbnz	x0, .Lstart_check
	mov	x0, #0
	ret
.Lstart_check:
	mov	x20, #1			// k = 1
	cmp	x20, x0
	b.gt	.Lreturn_max
	mov	x19, #0			// max_steps = 0
.b_kloop:
	mov	x6, x20			// x = k
	mov	x5, #0			// steps = 0
.L_collatz:
	cmp	x6, #1
	beq	.L_done_collatz
	tbnz	x6, #0, .L_odd_step
.as_even_step:
	lsr	x6, x6, #1		// x / 2
	b	.L_collatz
.L_odd_step:
	umaddl	x11, x6, #3, x6	// x = 3*x + 1 (x6 is odd, so this works as 3*x+x)
	add
