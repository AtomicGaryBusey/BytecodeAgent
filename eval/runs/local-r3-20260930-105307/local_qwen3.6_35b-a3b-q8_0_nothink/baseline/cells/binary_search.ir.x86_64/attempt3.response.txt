export function l $binary_search(l %a, l %n, l %key) {
@start
    %empty = csetl sle %n, 0
    br nz %empty, @not_found_case, @init

@not_found_case
    ret l -1

@init
    %low = add l 0, 0
    %high = sub l %n, 1
    br @loop

@loop
    %cond = cmp sle %low, %high
    br nz %cond, @search, @not_found

@search
    %sum = add l %low, %high
    %mid = ashr l %sum, 1
    
    %offset = mul l %mid, 8
    %addr = add l %a, %offset
    %val = loadl %addr
    
    %eq = cmp eq %val, %key
    br nz %eq, @found_it, @go_right

@go_right
    %low_plus_1 = add l %mid, 1
    %low = low_plus_1
    br @loop

@go_left
    %high_minus_1 = sub l %mid, 1
    %high = high_minus_1
    br @loop

@found_it
    ret l %mid

@not_found
    ret l -1
}
