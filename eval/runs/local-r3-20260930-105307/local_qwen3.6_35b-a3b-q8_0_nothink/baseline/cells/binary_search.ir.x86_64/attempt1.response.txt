export function l $binary_search(l %a, l %n, l %key) {
@start
    %ret = sub l -1, 0
    br i32 %n eq 0, @empty, @init

@empty
    ret l %ret

@init
    %low = add l 0, 0
    %high = sub l %n, 1
    br @loop

@loop
    cmp bgt %low, %high, @not_found
    
    %mid_sum = add l %low, %high
    %shifted = ashr l %mid_sum, 1
    %mid = shiftl %shifted, 0
    
    %elem_addr = mul l %mid, 8
    %val_addr = add l %a, %elem_addr
    %val = load l %val_addr
    
    cmp slt %val, %key, @go_right
    cmp sgt %val, %key, @go_left
    
@found
    %result = shiftl %mid, 0
    ret l %result

@go_left
    %high_minus_1 = sub l %mid, 1
    %new_high = shiftl %high_minus_1, 0
    br @loop

@go_right
    %low_plus_1 = add l %mid, 1
    %new_low = shiftl %low_plus_1, 0
    br @loop

@not_found
    ret l %ret
}
