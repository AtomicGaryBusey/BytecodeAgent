export function l $bmax(l %a, l %n) {
%entry:
    %base = loadl (%a)
    %end_ptr = addl %a, shlwi(3, %n)
    %ptr = addl %a, 8
    br i1 csltl(%ptr, %end_ptr), @loop, @done

@loop:
    %val = loadl (%ptr)
    br i1 csltl(%base, %val), @update, @next

@update:
    %base = %val

@next:
    %ptr = addl %ptr, 8
    br i1 csltl(%ptr, %end_ptr), @loop, @done

@done:
    ret l %base
}
