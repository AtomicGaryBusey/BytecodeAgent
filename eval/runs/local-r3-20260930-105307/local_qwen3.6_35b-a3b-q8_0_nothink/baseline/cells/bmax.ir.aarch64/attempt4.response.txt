export function l $bmax(l %a, l %n) {
%entry:
    %val = loadl (%a)
    %i = addl %a, 8
    %limit = addl %a, shlwi(3, %n)
    br i1 ceql(%i, %limit), @done, @loop

@loop:
    %curr = loadl (%i)
    br i1 csgtl(%curr, %val), @update, @next

@update:
    %val = %curr

@next:
    %i = addl %i, 8
    br i1 csltl(%i, %limit), @loop, @done

@done:
    ret l %val
}
