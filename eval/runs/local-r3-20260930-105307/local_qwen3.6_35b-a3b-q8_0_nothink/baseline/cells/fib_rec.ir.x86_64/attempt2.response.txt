export function l $fib_rec(l %n) {
@start
    %base0 = slt %n, 0
    jnz @return_zero
    %base1 = eq %n, 1
    jnz @return_one

@recurse
    %n_minus_2 = sub %n, 2
    %r_sub = call $fib_rec(l %n_minus_2)

    %n_minus_1 = sub %n, 1
    %r_main = call $fib_rec(l %n_minus_1)

    %result = add %r_main, %r_sub
    ret %result

@return_zero
    ret 0

@return_one
    ret 1
}
