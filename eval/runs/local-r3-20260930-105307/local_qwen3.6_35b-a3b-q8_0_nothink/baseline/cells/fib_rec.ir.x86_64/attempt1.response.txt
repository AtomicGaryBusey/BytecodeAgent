export function l $fib_rec(l %n) {
@start
    %base0 = slel %n, 0
    jnz @return_zero
    %base1 = ceql %n, 1
    jnz @return_one

@recurse
    %n_minus_2 = addl %n, -2
    %r_sub = l call $fib_rec(l %n_minus_2)

    %n_minus_1 = addl %n, -1
    %r_main = l call $fib_rec(l %n_minus_1)

    %result = addl %r_main, %r_sub
    ret %result

@return_zero
    ret 0

@return_one
    ret 1
}
