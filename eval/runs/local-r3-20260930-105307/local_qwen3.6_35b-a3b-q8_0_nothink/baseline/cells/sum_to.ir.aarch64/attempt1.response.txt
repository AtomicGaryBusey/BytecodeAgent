export function l $sum_to(l %n) {
@start:
    ; If n <= 0, return 0
    %cmp = csltl %n, 0
    jnz %zero
    
@loop:
    ; Initialize sum = 0, i = 1
    %sum = alloc l
    storel 0, %sum
    %i = alloc l
    storel 1, %i

@check_loop:
    %ci = loadl %i
    %cmp2 = csltl %n, %ci
    jnz @done
    
@body:
    ; sum += i
    %s = loadl %sum
    %c = addl %s, %ci
    storel %c, %sum
    
    ; i++
    %ii = loadl %i
    %ni = addl %ii, 1
    storel %ni, %i
    
    jmp @check_loop

@done:
    %result = loadl %sum
    ret %result

@zero:
    ret 0
}
