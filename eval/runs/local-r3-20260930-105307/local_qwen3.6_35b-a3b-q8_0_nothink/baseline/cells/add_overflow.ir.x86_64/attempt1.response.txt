export function l $add_overflow(l %a, l %b, l *%out) {
    @start
        %sum = addl %a, %b
        storel %sum, %out
        %of1 = csltl %a, #0x8000000000000000
        %of2 = csltl %b, #0x8000000000000000
        %res1 = addl %of1, %of2
        
        @check_neg_pos
            %neg_a = slt l %a, #0
            %pos_b = sle l %b, #0
            %mismatch1 = xorw %neg_a, %pos_b
            
        @check_pos_neg
            %pos_a = sge l %a, #0
            %neg_b = slt l %b, #0
            %mismatch2 = xorw %pos_a, %neg_b

        @combine_flags
        ; We can detect overflow by: 
        ; If signs are same (both pos or both neg), overflow if sign of result differs from operands.
        ; If signs are different, no overflow possible for addition.
        
        %sign_sum = slt l %sum, #0
        %sign_a = slt l %a, #0
        %mismatch_sign = xorw %sign_a, %sign_sum
        
        ; Overflow occurs if:
        ; (a > 0 && b > 0 && sum < 0) OR (a < 0 && b < 0 && sum >= 0)
        
        @case1_pos_pos_neg
            %is_pos = andw %pos_a, %neg_b  ; wait neg_b is negative? no.
            ; Let's use: sign_a != 0 if a < 0
            
        ; Correct logic:
        ; Overflow = ((a ^ b) >= 0) & ((a ^ sum) < 0)
        ; XORing two numbers with same sign gives non-negative. 
        ; If result has different sign, overflow occurred.
        
        %xor_ab = xorl %a, %b
        %xor_sum = xorl %a, %sum
        
        @same_sign
            %no_overflow_flag = sge l %xor_ab, #0 ; Same sign implies >= 0
            
        @different_sign_from_result
            %overflow_from_xor = slt l %xor_sum, #0 ; Result different sign than operands
            
        @combine
            %is_overflow = andw %no_overflow_flag, %overflow_from_xor
        
        ret %is_overflow
}
