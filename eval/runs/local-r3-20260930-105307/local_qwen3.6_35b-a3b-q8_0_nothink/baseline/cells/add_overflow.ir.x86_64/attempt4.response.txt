export function l $add_overflow(l %a, l %b, l *%out) {

        %sum = addl %a, %b
        storel %sum, %out
        %xor_ab = xorl %a, %b
        %xor_sum = xorl %sum, %a
        %sign_a = csgtl #0, %a
        %sign_b = csgtl #0, %b
        %sign_sum = csgtl #0, %sum

}
