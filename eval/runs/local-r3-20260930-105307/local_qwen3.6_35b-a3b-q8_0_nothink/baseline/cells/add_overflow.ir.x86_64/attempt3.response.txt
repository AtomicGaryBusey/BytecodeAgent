export function l $add_overflow(l %a, l %b, l *%out) {
    @start
        %sum = addl %a, %b
        storel %sum, %out
        %xor_ab = xorl %a, %b
        %xor_sum = xorl %a, %sum
        %same_sign = csgtl %xor_ab, #0
        %diff_sign = csgtl #0, %xor_sum
        %is_overflow = andw %same_sign, %diff_sign
        ret %is_overflow
}
