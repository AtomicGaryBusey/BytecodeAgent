@start
    %is_zero = ceql n, #0
    jnz @ret_zero
    
    %is_one = ceql n, #1
    jnz @ret_one
    
    ; Save a=0, b=1, i=2 to stack
    storel [%sp], #0
    storel [%sp + 8], #1
    storel [%sp + 16], #2

@loop
    %i = loadl [%sp + 16]
    %cond = csltl %n, %i
    jnz @ret_b
    
    %a = loadl [%sp]
    %b = loadl [%sp + 8]
    
    %sum = addl %a, %b
    
    ; a becomes old b
    storel [%sp], %b
    
    ; b becomes sum
    storel [%sp + 8], %sum
    
    ; i increments
    %next_i = addl %i, #1
    storel [%sp + 16], %next_i
    
    j @loop

@ret_zero
    ret #0

@ret_one
    ret #1

@ret_b
    %res = loadl [%sp + 8]
    ret %res
}
