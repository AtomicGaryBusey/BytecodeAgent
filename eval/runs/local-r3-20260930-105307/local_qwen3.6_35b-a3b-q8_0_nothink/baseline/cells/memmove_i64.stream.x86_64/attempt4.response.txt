memmove_i64:
	test rdx, rdx
	je .L_zero
	cmp rdi, rsi
	jb .L_overlap_backward
.L_forward_loop:
	mov rax, qword ptr [rsi]
	mov qword ptr [rdi], rax
	add rsi, 8
	add rdi, 8
	sub rdx, 1
	jae .L_forward_loop
	xor eax, eax
	ret
.L_overlap_backward:
	sub rdx, 1
	jl .L_zero_done
	lea rax, [rdi + rdx*8]
	lea rcx, [rsi + rdx*8]
	mov rax, qword ptr [rcx]
	mov qword ptr [rax], rax
	sub rax, 8
	sub rcx, 8
	add rdx, 1
	cmp rdx, 0
	jne .L_overlap_backward
.L_zero_done:
	xor eax, eax
	ret
.L_zero:
	xor eax, eax
	ret
