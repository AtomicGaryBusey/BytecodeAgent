memmove_i64:
	test rdx, rdx
	je .L_zero
	cmp rdi, rsi
	jb .L_overlap_backward
	sub rcx, 1
	lea rax, [rdi + rcx*8]
	lea r9, [rsi + rcx*8]
.L_forward:
	mov rax, qword ptr [rsi]
	mov qword ptr [rdi], rax
	add rsi, 8
	add rdi, 8
	sub rcx, 1
	jae .L_forward
	xor eax, eax
	ret
.L_overlap_backward:
	cmp rdi, rsi
	ja .L_overlap_same
	sub rcx, 1
	lea rax, [rdi + rcx*8]
	lea r9, [rsi + rcx*8]
.L_backward:
	mov rax, qword ptr [r9]
	mov qword ptr [rax], rax
	sub r9, 8
	sub rax, 8
	sub rcx, 1
	jae .L_backward
	xor eax, eax
	ret
.L_overlap_same:
	cmp rdi, rsi
	je .L_zero
	sub rcx, 1
	sub rdx, 1
	add rsi, rcx*8
	add
