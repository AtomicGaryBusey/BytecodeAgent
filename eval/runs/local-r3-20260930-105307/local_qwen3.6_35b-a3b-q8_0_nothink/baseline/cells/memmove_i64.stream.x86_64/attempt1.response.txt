memmove_i64:
	test rdx, rdx
	je .L_done
	cmp rdi, rsi
	jae .L_fallback
	rep movsq
	xor eax, eax
	ret
.L_fallback:
	cmp rdi, rsi
	jb .L_forward
	sub rdi, 8
	sub rsi, 8
	mov rcx, rdx
	test rcx, rcx
	je .L_done
.L_backward:
	mov rax, qword ptr [rsi]
	add rsi, 8
	mov qword ptr [rdi], rax
	add rdi, 8
	dec rcx
	jnz .L_backward
	xor eax, eax
	ret
.L_forward:
	mov rcx, rdx
	test rcx, rcx
	je .L_done
.L_loop_fwd:
	mov rax, qword ptr [rsi]
	add rsi, 8
	mov qword ptr [rdi], rax
	add rdi, 8
	dec rcx
	jnz .L_forward
	xor eax, eax
	ret
.L_done:
	xor eax, eax
	ret
