memmove_i64:
	test rdx, rdx
	je .L_done
	cmp rdi, rsi
	jb .L_overlap
rep movsq
	xor eax, eax
	ret
.L_overlap:
	cmp rdi, rsi
	ja .L_backward
	sub rsi, 8
	sub rdi, 8
	mov rcx, rdx
	test rcx, rcx
	je .L_done
.L_backward:
	mov rax, qword ptr [rsi + 8]
	mov qword ptr [rdi + 8], rax
	add rsi, 8
	add rdi, 8
	dec rcx
	jnz .L_backward
	xor eax, eax
	ret
.L_forward:
	sub rsi, 8
	sub rdi, 8
	mov rcx, rdx
	test rcx, rcx
	je .L_done
.L_loop_fwd:
	add rsi, 8
	add rdi, 8
	mov rax, qword ptr [rsi]
	mov qword ptr [rdi], rax
	dec rcx
	jnz .L_loop_fwd
	xor eax, eax
	ret
.L_done:
	xor eax, eax
	ret
