export function l $isqrt(l n) {
    @entry
        ; Handle n=0 quickly
        cmpu $n, const 0
        jz @return_zero

        ; We use the bit-by-bit construction algorithm from MSB to LSB.
        ; The maximum possible root is 2^32 - 1.
        ; So we only need to check bits 0 through 31.
        ; Start with bit = 31 (value 2^31).
        
        %res = const 0
        %bit_val = const 2147483648       ; 1 << 31
        
        @loop
            cmpu $n, %res
            jz @done                      ; if n == res*res exactly? No, res is just the current candidate without bit added.
                                        ; Actually, standard algo:
                                        ; try_add = res | bit_val
                                        ; sq = try_add * try_add
                                        ; if sq <= n: res = try_add
        
            %try_add = or $res, %bit_val
            
            ; Compute square of %try_add. 
            ; Since max try_add is (2^31 + (2^31-1))^2? No.
            ; Max res is ~2^32. Max bit_val starts at 2^31.
            ; At any step, res < 2^(k+1) and bit_val = 2^k.
            ; try_add < 2^(k+1).
            ; Square can be up to 2^(64). 
            ; We need to compute (try_add * try_add) <= n correctly for 64-bit unsigned arithmetic.
            ; QBE 'mul' on 64-bit integers is modular.
            ; However, we only care if the result is <= n.
            ; Since n < 2^64 and max square of a 32-bit number is < 2^64, 
            ; direct multiplication works for unsigned comparison IF the true mathematical square < 2^64.
            ; The maximum root we test is 2^32 - 1 (when bit_val is 0 eventually, but we stop before).
            ; Wait, if res has bit 31 set, and we try to add lower bits...
            ; The largest value for `try_add` we ever compute the square for is roughly sqrt(max_uint64) ~ 2^32.
            ; Specifically, if n = 2^64 - 1, root is 2^32 - 1.
            ; (2^32 - 1)^2 = 2^64 - 2^33 + 1. This fits in uint64_t.
            ; So direct 64-bit multiplication and comparison is safe for unsigned types.

            %sq = mul $try_add, $try_add
            
            cmpu $n, %sq
            jnge @no_set_bit              ; if sq > n, don't set bit
            
            ; Otherwise, accept the bit
            %res = $try_add
            
        @no_set_bit
            shr $bit_val, const 1         ; shift bit_val right by 1
            cmpu $bit_val, const 0
            jnz @loop                     ; if bit_val is still non-zero, continue
        
        @done
            ret %res

    @return_zero
        ret const 0
}
