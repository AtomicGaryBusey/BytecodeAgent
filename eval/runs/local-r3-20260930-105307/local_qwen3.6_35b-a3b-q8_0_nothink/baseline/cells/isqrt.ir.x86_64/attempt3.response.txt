export function l $isqrt(l n) {
    @entry
        cmpu $n, const 0
        jz @ret_zero
        
        ; Initialize result and trial bit.
        ; We want floor(sqrt(n)). Max root is 2^32 - 1.
        ; Start with bit = 2^31 (highest possible bit of the root).
        %res = const 0
        %bit = const 2147483648   ; 1 << 31
        
    @loop
        cmpu $n, %res
        jz @done                  ; If n is exactly equal to current square? 
                                  ; Wait, %res is the accumulated root so far.
                                  ; The algorithm maintains: res*res <= original_n_remaining? No.
                                  ; Standard algorithm:
                                  ; for i from 31 down to 0:
                                  ;   b = 1 << i
                                  ;   if (res | b)^2 <= n:
                                  ;     res |= b
        
        %try_res = or $res, %bit
        
        ; Calculate square of try_res. 
        ; Since max value is < 2^32, square is < 2^64. Unsigned multiplication is safe.
        %sq = mul $try_res, $try_res
        
        cmpu $n, %sq
        jnge @skip_set            ; if sq > n, skip setting bit
        
        %res = $try_res
        
    @skip_set
        shr $bit, const 1         ; shift bit right by 1
        cmpu $bit, const 0
        jnz @loop                 ; while bit != 0
    
    @done
        ret $res

    @ret_zero
        ret const 0
}
