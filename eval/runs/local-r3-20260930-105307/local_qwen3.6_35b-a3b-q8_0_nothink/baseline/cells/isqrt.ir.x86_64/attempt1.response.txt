export function l $isqrt(l n) {
    @entry
        %one = const 1
        %shift = sub %one, %one     ; 0, will count shifts needed
        
        ; Find the highest bit of n to determine shift amount
        ; If n is 0, return 0
        jz @return_zero
        
        ; We can use a binary search / bit-by-bit construction method.
        ; Algorithm: 
        ; result = 0
        ; for i from 63 down to 0:
        ;   result |= (1 << i)
        ;   if (result * result > n):
        ;     result &= ~(1 << i)
        ; This works because we are building the root bit by bit from MSB.
        
        ; However, multiplying `result * result` can overflow 64 bits.
        ; But note: result starts small. When result has k bits set or 
        ; effectively value around 2^k, result*result is 2^(2k).
        ; Max n is 2^64-1. Max root is 2^32-1. 
        ; We only need to check up to bit 31 essentially (since 2^32 squared is 2^64 which overflows, but we handle it carefully).
        ; Actually, 4294967295 * 4294967295 = 2^64 - 2^33 + 1? 
        ; (2^32 - 1)^2 = 2^64 - 2^33 + 1. This fits in 64-bit unsigned IF we interpret it as unsigned, but actually it wraps modulo 2^64.
        ; Wait: 2^64 - 2^33 + 1 is less than 2^64, so it fits in a uint64_t.
        ; 4294967295^2 = 18446744065119617025.
        ; Max uint64 is 18446744073709551615.
        ; So it DOES fit in 64-bit unsigned integer arithmetic without overflow/mask issues for the final answer.
        
        ; Let's use a simpler iterative approach:
        ; res = 0;
        ; bit = highest one bit of n (or 2^32 if we want to be safe and just do 63..0 shifts, but we can optimize).
        
        ; To find the highest bit efficiently or just loop from 31 down to 0 since max root is < 2^32.
        ; Let's just loop i from 31 down to 0.
        
        %res = const 0
        %i = sub %one, %one          ; 0
        %loop_limit = sub %one, %one ; will be set to 31
        
        @start_loop
            %i_add = add %i, %one    ; i+1
        
        @check_bit
            ; We want to try setting bit `k` where k goes from 31 down to 0.
            ; Let's maintain a separate counter or just use subtraction from 32.
            
        jnz @done
            
    }
}
