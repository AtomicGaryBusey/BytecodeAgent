export function l $bubble_sort(l %a, l %n) {
@entry:
    cmp n, 0w
    jnz @base_case

@main_loop:
    movi 0w, %outer_i
    cmp %outer_i, n
    jge @done

@inner_start:
    movi 0w, %inner_j
    mov %a, %a_start
    movn %n, %n_minus_1
    cmp %inner_j, %n_minus_1
    jge @inc_outer

@inner_body:
    lea %inner_j, %idx1, %a_start
    loadl [%idx1], %val1
    lea %inner_j, %idx2, %a_start
    addi 8w, %inner_j, %idx2
    loadl [%idx2], %val2
    csltl %val1, %val2, %swap_needed
    jz @no_swap

@do_swap:
    lea %inner_j, %idx1, %a_start
    storel %val2, [%idx1]
    lea %inner_j, %idx2, %a_start
    addi 8w, %inner_j, %idx2
    storel %val1, [%idx2]

@no_swap:
    addi 1w, %inner_j, %inner_j
    cmp %inner_j, %n_minus_1
    jlt @inner_body
    j @inc_outer

@inc_outer:
    addi 1w, %outer_i, %outer_i
    cmp %outer_i, n
    jlt @inner_start
    j @main_loop

@base_case:
    movi 0w, %result
    retl %result

@done:
    movi 0w, %result
    retl %result
}
