export function l $bubble_sort(l %a, l %n) {
@entry:
    cmp %n, 0w
    jz @done

@outer_loop:
    movi -1w, %i
@outer_check:
    addi 1w, %i, %next_i
    cmp %next_i, %n
    jge @done

    movi 0w, %j
@inner_check:
    addi 1w, %j, %next_j
    cmp %next_j, %i
    jge @inc_outer

@inner_body:
    shl 3w, %j, %off1
    lea %a, %ptr1, %off1
    loadl [%ptr1], %v1
    lea %a, %ptr2, %next_j
    addi 0w, %ptr2, %off2
    add %off1, %off2, %off2_2
    loadl [%off2_2], %v2

    csltl %v2, %v1, %do_swap
    jz @no_swap

@do_swap:
    lea %a, %ptr3, %off1
    storel %v2, [%ptr3]
    storel %v1, [%off2_2]

@no_swap:
    movi %next_j, %j
    jmp @inner_check

@inc_outer:
    movi %next_i, %i
    jmp @outer_loop

@done:
    movi 0w, %ret_val
    retl %ret_val
}
