export function l $bubble_sort(l %a, l %n) {
@entry:
	cmp %n, 0w
	jz @done

	movi -1w, %i

@outer_check:
	addi 1w, %i, %next_i
	cmp %next_i, %n
	jge @done

	movi 0w, %j

@inner_check:
	cmp %j, %next_i
	jl @inner_body
	jmp @inc_outer

@inner_body:
	shl 3w, %j, %off1
	lea %a, %ptr1, %off1
	loadl [%ptr1], %v1
	addi 1w, %j, %next_j
	shl 3w, %next_j, %off2
	lea %a, %ptr2, %off2
	loadl [%ptr2], %v2

	csltl %v2, %v1, %do_swap
	jz @no_swap

@do_swap:
	storel %v2, [%ptr1]
	storel %v1, [%ptr2]

@no_swap:
	movi %next_j, %j
	jmp @inner_check

@inc_outer:
	movi %next_i, %i
	jmp @outer_check

@done:
	movi 0w, %ret_val
	retl %ret_val
}
