export function l $bubble_sort(l %a, l %n) {
@entry:
    cmp %n, 0w
    jz @done

@outer_start:
    movi 1w, %i
@outer_check:
    cmp %i, %n
    jge @done

@inner_start:
    movi 0w, %j
@inner_bound_check:
    cmp %j, %i
    jge @inc_outer

@inner_body:
    lea %a, %ptr1, %j
    loadl [%ptr1], %val1
    lea %a, %ptr2, %j
    addi 8w, %j, %tmp_j
    loadl [%ptr2], %val2
    csltl %val1, %val2, %swap
    jz @no_swap

@do_swap:
    lea %a, %ptr3, %j
    storel %val2, [%ptr3]
    lea %a, %ptr4, %tmp_j
    storel %val1, [%ptr4]
    j @inc_inner

@no_swap:
@inc_inner:
    addi 1w, %j, %j
    jmp @inner_bound_check

@inc_outer:
    addi 1w, %i, %i
    jmp @outer_check

@done:
    movi 0w, %result
    retl %result
}
