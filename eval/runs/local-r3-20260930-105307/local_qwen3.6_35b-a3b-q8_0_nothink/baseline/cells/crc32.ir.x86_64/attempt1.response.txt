export function l $crc32(l %p, l %n) {
@start
	%crc = const w 0xFFFFFFFF
	%mask = const w 0x80000000
	%xor_val = const w 0xEDB88320

@loop
	jl %n, 0, @done
	%byte =w loadub %p
	%crc =w xor %crc, %byte

	%i = l 0
@bit_loop
	csltl %i, 8, @bit_continue
	jmp @end_bit
@bit_continue
	%lsb =w and %crc, 1
	csetz %is_zero, %lsb, %lsb
	jnz %is_zero, @shift_only

	%crc =w xor %crc, %xor_val
	jmp @right_shift
@shift_only
@end_bit

	%hi = l (0xFFFFFFFF << 32)
	%low_crc =l extuw %crc
	%temp =l or %hi, %low_crc
	%crc_l =l shr %temp, 1
	%crc =w truncl %crc_l

	%i = addl %i, 1
	jmp @bit_loop

@done
	%xor_final = const w 0xFFFFFFFF
	%crc =w xor %crc, %xor_final
	%hi = l (0xFFFFFFFF << 32)
	%low_crc =l extuw %crc
	%result =l or %hi, %low_crc
	ret %result
}
