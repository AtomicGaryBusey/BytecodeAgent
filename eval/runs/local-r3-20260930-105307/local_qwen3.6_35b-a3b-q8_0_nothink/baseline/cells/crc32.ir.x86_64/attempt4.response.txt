export function l $crc32(l %p, l %n) {
@start
	testw n:%n, n:%n
	jz @done
	
	%crc =constw 0xFFFFFFFF
	
@loop_byte
	loadub %b, %p
	xorw %crc, %b, %crc
	%i = constl 0
	
@bit_loop
	csltl %i, 8, @continue_bit
	jmp @end_bits
@continue_bit
	andwl %crc, 1, %lsb
	jnz %lsb, @flip
	
	shrw %crc, 1, %crc
	jmp @next_bit
@flip
	shrw %crc, 1, %crc
	xorw %crc, 0xEDB88320, %crc
@next_bit
	%i = addl %i, 1
	jmp @bit_loop
@end_bits

	addl %p, 1, %p
	subln %n, 1, %n
	jnz %n, @loop_byte

@done
	xorw %crc, 0xFFFFFFFF, %crc
	
	extuw %result_low, %crc
	ret %result_low
}
