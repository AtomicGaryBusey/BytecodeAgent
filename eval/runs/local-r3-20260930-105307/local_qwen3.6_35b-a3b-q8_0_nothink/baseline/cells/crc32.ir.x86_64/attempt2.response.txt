export function l $crc32(l %p, l %n) {
@start
	%crc = const w 0xFFFFFFFF
	%xor_val = const w 0xEDB88320

	testl %n, %n
	jz @done

@loop_byte
	%byte =w loadub %p
	%crc =w xor %crc, %byte
	
	%i = l 0
@bit_loop
	csltl %i, 8, @continue_bit
	jmp @end_bits
@continue_bit
	%lsb =w and %crc, 1
	csetz %is_zero, %lsb, %lsb
	jnz %is_zero, @shift_only

	%crc =w xor %crc, %xor_val
@shift_only
	%crc =w shr %crc, 1

	%i = addl %i, 1
	jmp @bit_loop
@end_bits

	%p = addl %p, 1
	%n = subln %n, 1
	jmp @loop_byte

@done
	%xor_final = const w 0xFFFFFFFF
	%crc =w xor %crc, %xor_final
	
	%hi = l (0xFFFFFFFF << 32)
	%low_crc =l extuw %crc
	%result =l or %hi, %low_crc
	ret %result
}
