@start_loop
	csltl %n, 0, %cond
	jnz %done
	@dst = dst + (n-1)*8
	subl %tmp_n_1, %n, 1
	sall $3, %tmp_n_1, %offset
	addl %src_offset, %offset, %loop_dst
	addl %src, %offset, %loop_src
	@dst = dst + (n-1)*8
	subl %tmp_n_1, %n, 1
	sall $3, %tmp_n_1, %dst_offset
	addl %dst, %dst_offset, %loop_dst
	subl %tmp_n_1, %n, 1
	sall $3, %src_offset2, %tmp_n_1
	addl %src, %src_offset2, %loop_src
	jne %after_cmp, %after_loop_setup
	@ Copy backwards if dst < src is false (dst >= src), actually for overlap dst>=src we need to copy from end to start to avoid overwriting. If dst < src, copying forward is safe. So: if dst < src, copy forward. Else copy backward.
	cslt %loop_dst, %loop_src, %forward
	jnz %copy_forward
	@ Copy backwards
	subl %n, 1, %i
	sall $3, %idx_offset, %i
	addl %src, %idx_offset, %loop_src
	subl %n, 1, %i
	sall $3, %dst_idx_offset, %i
	addl %dst, %dst_idx_offset, %loop_dst
	@ Loop backward
	jge %after_loop_cond, %back_end_cond
	back_iter:
	loadl %val, [%loop_src]
	storel %val, [%loop_dst]
	subl $8, %loop_src, %loop_src
	subl $8, %loop_dst, %loop_dst
	subl $1, %i, %i
	jge %back_iter, %back_loop_cond
	@ Loop backward
	back_loop_cond:
	cmpl 0, %i
	jeq %after_back_loop, %back_iter
	back_end_cond:
	jne %done, %after_back_loop
	@ Copy forward
	subl $1, %n, %i
	forward_iter:
	loadl %val, [%src]
	storel %val, [%dst]
	addl $8, %src, %src
	addl $8, %dst, %dst
	subl $1, %i, %i
	cmpl 0, %i
	jge %forward_iter, %forward_loop_cond
	forward_loop_cond:
	jne %done, %after_forward_loop
	@ Return 0
	movl $0, %res
	storel %res, [%rsp]
	ret
	@ Labels and blocks
@copy_forward
	csltl %loop_dst, %loop_src, %need_back
	jnz @back_start
	subl $1, %n, %i
	forward_iter:
	loadl %val, [%src]
	storel %val, [%dst]
	addl $8, %src, %src
	addl $8, %dst, %dst
	subl $1, %i, %i
	cmpl 0, %i
	jge %forward_iter, @fwd_loop_end
	@store_result
	movl $0, %res
	ret
@back_start
	subl %n, 1, %back_i
	sall $3, %back_off, %back_i
	addl %src, %back_off, %back_src
	subl %n, 1, %back_dst_i
	sall $3, %back_dst_off, %back_dst_i
	addl %dst, %back_dst_off, %back_dst
	@ Loop backward
	back_iter:
	loadl %val, [%back_src]
	storel %val, [%back_dst]
	subl $8, %back_src, %back_src
	subl $8, %back_dst, %back_dst
	subl $1, %back_i, %back_i
	cmpl 0, %back_i
	jeq @store_result
	jne %back_iter, @b_loop_cond
@b_loop_cond
	jne %done, @store_result
