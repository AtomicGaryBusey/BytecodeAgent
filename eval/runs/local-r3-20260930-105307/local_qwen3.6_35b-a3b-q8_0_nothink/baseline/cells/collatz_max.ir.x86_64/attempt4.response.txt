export function l $collatz_max(l %n) {
  @loop
  %result = addl %n, 0
  %k = addl %n, -1
@skip_check
  cmplt %k, 2
  jnz @done_outer
  %start = %k
  %max_steps = 0
@k_loop
  cmpneq %start, 1
  je @update_max_k
  %x = %start
  %steps = 0
@inner_loop
  cmpneq %x, 1
  jz @inner_done
  %is_even = andl %x, 1
  cmpeq %is_even, 1
  jnz @do_odd
  %x = srair %x, 1
  jmp @next_step_inner
@do_odd
  %tmp = addl %x, %x
  %x = addl %tmp, %x
  %x = addl %x, 1
@next_step_inner
  %steps = addl %steps, 1
  jmp @inner_loop
@inner_done
@update_max_k
  cmplt %max_steps, %steps
  jnc @k_cont
  %max_steps = %steps
@k_cont
  %start = sub l %start, 1
  cmpneq %start, 1
  je @k_update
  jmp @inner_loop
@k_update
  cmpneq %start, 2
  jz @check_done_outer
  jmp @k_loop
@check_done_outer
  cmplt %result, %max_steps
  jnc @end_max
  %result = %max_steps
@end_max
  ret %result
@done_outer
  cmpneq %n, 0
  jnz @min_one
  ret 0
@min_one
  cmpneq %n, 1
  jnz @return_one
  %tmp = addl %n, -1
  %x = %tmp
  %steps = 0
@odd_loop_check
  cmpneq %x, 1
  je @odd_done
  %is_even = andl %x, 1
  cmpeq %is_even, 1
  jnz @odd_do_even
  %x = srair %x, 1
  jmp @odd_next
@odd_do_even
  %tmp2 = addl %x, %x
  %x = addl %tmp2, %x
  %x = addl %x, 1
@odd_next
  %steps = addl %steps, 1
  jmp @odd_loop_check
@odd_done
  ret %steps
@return_one
  ret 1
}
