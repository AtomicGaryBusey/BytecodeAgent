export function l $collatz_max(l %n) {
@start:
	%max_steps = add l 0, l 0
	%k = add l 1, l 0
@loop_k:
	cmpl %k, %n
	jg @return_max
	%steps = call i64 @collatz_steps(i64 %k)
	csltl %max_steps, %steps
	movl %steps, @c_true, @c_false
	cbranch @c_true %steps, %max_steps, @set_new_max
@next_k:
	kaddl %k, 1
	jmp @loop_k
@set_new_max:
	%max_steps = %steps
	jmp @next_k
@return_max:
	movl %max_steps, %rax
	ret
}

function i64 @collatz_steps(i64 %start) {
@cs_start:
	%x = add l 0, l %start
	%count = add l 0, l 0
@cs_loop:
	cmpeq %x, 1
	jnz @cs_done
	%is_even = andl %x, 1
	csltl %is_even, 1
	movl 1, @c_true2, @c_false2
	jmpz @c_false2 @cs_odd
@cs_even:
	%x = srair %x, 1
	jmp @cs_next_step
@cs_odd:
	%x = addl %x, %x
	%x = addl %x, %x
	%x = addl %x, 1
@cs_next_step:
	%count = addl %count, 1
	jmp @cs_loop
@cs_done:
	movl %count, %rax
	ret
}
