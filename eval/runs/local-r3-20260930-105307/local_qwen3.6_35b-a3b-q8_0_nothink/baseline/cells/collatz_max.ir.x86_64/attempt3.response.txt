export function l $collatz_max(l %n) {
  %result = add l 0, l 0
  %k = add l 1, l 0
@loop:
  cmpneq %k, %n
  jz @check_max
  
  %val = %k
  %steps = add l 0, l 0
@inner_loop:
  cmpneq %val, 1
  jz @inner_done
  
  %is_even = andl %val, 1
  cmpneq %is_even, 0
  jz @even_case
  
  %tmp = addl %val, %val
  %val = addl %tmp, %val
  %val = addl %val, 1
  jmp @next_step
@even_case:
  %val = srair %val, 1
@next_step:
  %steps = addl %steps, 1
  jmp @inner_loop
@inner_done:
  
  csltl %result, %steps
  jnz @update_max
  
  jmp @next_k
@update_max:
  %result = %steps
@check_max:
  cmpneq %k, %n
  je @end_check
@next_k:
  %k = addl %k, 1
  jmp @loop
@end_check:
  ret %result
}
