export function l $reverse(l %a, l %n) {
@start:
    cmpslt %n, 0w
    jnz @done

    %half_n = asrwi %n, 1
    cmpslti %half_n, 1
    jz @done

    %left = %a
    %right_idx = sub %n, 1
    %right = getelement l[0] %a, %right_idx

@loop:
    cmpge %left, %right
    jnz @done_loop

    %val_left = loadl %left
    %val_right = loadl %right

    storel %val_right, %left
    storel %val_left, %right

    %left_inc = addi %left, 8
    %right_dec = subi %right, 8

    %left = %left_inc
    %right = %right_dec

    jmp @loop

@done_loop:
@done:
    ret 0w
}
