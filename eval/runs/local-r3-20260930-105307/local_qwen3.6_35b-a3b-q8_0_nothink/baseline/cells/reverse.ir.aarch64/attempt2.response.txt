export function l $reverse(l %a, l %n) {
@entry:
    %n_neg = cmpslt %n, 0w
    jnz @done

    %half_n = asrwi %n, 1
    %i = const i64 0
    %end_ptr = add %a, %n
    %start_ptr = %a
    %mid_ptr = sub %end_ptr, %start_ptr
    %half_bytes = asrwi %n, 1
    %end_ptr_dec = sub %end_ptr, 8

@loop:
    cmpslt %i, %half_bytes
    jz @done_loop

    %val_i = loadl %start_ptr
    %val_j = loadl %end_ptr_dec

    storel %val_i, %end_ptr_dec
    storel %val_j, %start_ptr

    %start_ptr_inc = add %start_ptr, 8
    %end_ptr_dec_dec = sub %end_ptr_dec, 8
    %i_inc = add %i, 1

    start_ptr = %start_ptr_inc
    end_ptr_dec = %end_ptr_dec_dec
    i = %i_inc

    jz @loop

@done_loop:
@done:
    ret 0w
}
