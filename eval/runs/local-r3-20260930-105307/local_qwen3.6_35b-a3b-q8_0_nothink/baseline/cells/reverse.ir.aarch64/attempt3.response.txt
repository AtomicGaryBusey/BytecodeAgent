export function l $reverse(l %a, l %n) {
@entry:
    cmpslt %n, 0w
    jnz @done

    %half_n = asrwi %n, 1
    cmpslti %half_n, 1
    jz @loop
    ret 0w

@loop:
    %start_idx = sub %i_ptr, %a

    cmpge %start_idx, %half_n
    jnz @done_loop

    %val_start = loadl %a, %start_idx
    %val_end = loadl %a, %end_idx

    storel %val_start, %end_idx
    storel %val_end, %start_idx

    %i_ptr_inc = addi %i_ptr, 8
    %end_idx_dec = sub %end_idx, 8

    i_ptr = %i_ptr_inc
    end_idx = %end_idx_dec

    jmp @loop

@done_loop:
@done:
    ret 0w
}
