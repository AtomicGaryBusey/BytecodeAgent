export function l $reverse(l %a, l %n) {
@entry:
    %n_neg = cmpslt %n, 0w
    jnz @done

    %half_n = asrwi %n, 1
    %i = const i64 0
    %j = sub %n, 1
    jz @done_loop

@loop:
    %idx_i = shlwi %i, 3
    %ptr_i = add %a, %idx_i
    %val_i = loadl %ptr_i

    %idx_j = shlwi %j, 3
    %ptr_j = add %a, %idx_j
    %val_j = loadl %ptr_j

    storel %val_i, %ptr_j
    storel %val_j, %ptr_i

    %i_inc = add %i, 1
    %j_dec = sub %j, 1
    %i_cmp = cmpslt %i, %half_n
    jnz @loop

@done_loop:
@done:
    ret 0w
}
