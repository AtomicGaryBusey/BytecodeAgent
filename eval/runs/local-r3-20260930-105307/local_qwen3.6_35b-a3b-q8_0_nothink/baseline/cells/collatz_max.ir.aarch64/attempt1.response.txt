export function l $collatz_max(l $n) {
@entry:
  %max_steps = constant i64 0
  %k = loadl 0x0
  storel %k, @k
  loop_k:
    %is_done = csltl %k, %n
    jnz @loop_k_end
    
    %current = loadl @k
    %steps = constant i64 0
    storel %steps, @steps
    step_loop:
      %is_one = ceql %current, constant i64 1
      jnz @step_loop_done
      
      %is_even = bandw %current, constant i32 1
      jnz @odd_case
      
    @even_case:
      %current = sarl %current, 1
      storel %current, @current
      goto @update_steps
      
    @odd_case:
      %three_x_plus_1 = addl %current, %current
      %three_x_plus_1 = addl %three_x_plus_1, %current
      %three_x_plus_1 = addl %three_x_plus_1, constant i64 1
      storel %three_x_plus_1, @current
      goto @update_steps
      
    @update_steps:
      %old_steps = loadl @steps
      %new_steps = addl %old_steps, constant i64 1
      storel %new_steps, @steps
      goto @step_loop
      
    @step_loop_done:
      %final_steps = loadl @steps
      
      %is_greater = csgtl %final_steps, %max_steps
      jnz @update_max
      
      %updated_max = constant i64 0
      storel %updated_max, @max_steps
      goto @increment_k
      
    @update_max:
      storel %final_steps, @max_steps
      
    @increment_k:
      %old_k = loadl @k
      %new_k = addl %old_k, constant i64 1
      storel %new_k, @k
      goto @loop_k
      
@loop_k_end:
  %result = loadl @max_steps
  ret %result
}
