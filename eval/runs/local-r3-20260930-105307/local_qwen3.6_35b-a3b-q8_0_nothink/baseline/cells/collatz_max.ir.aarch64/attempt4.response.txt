  %max = constant i64 0
  br @loop_k

@loop_k:
  %k_is_gt_n = csgtl %n, %k_is_zero_not_used
  %k = loadl [%sp] ;; Wait, we can't use stack easily without setup. Let's just use a loop counter.
  ;; Actually, let's restructure to avoid complexity. n is in argument $n.
  ;; We need to iterate k from 1 to n.
  ;; Since QBE doesn't have explicit locals, we use temporaries.
  ;; But wait, the previous error was "invalid function parameter". 
  ;; The signature `export function l $collatz_max(l $n)` is correct for QBE.
  ;; The failure might be due to something else or a typo in my thought process simulation?
  ;; No, the prompt says the PREVIOUS attempt failed. Let's look at the previous code again.
  ;; Ah, `loadl [%sp]` is wrong because %sp isn't defined as a label or register in QBE IL directly like that for stack access in this context without proper alignment/stack frame management which QBE handles automatically if you use alloc. But we are writing pure IL. 
  ;; Let's write a clean loop.

  ; Loop k from 1 to n
  %k = constant i64 1
  br @check_k

@check_k:
  %k_done = csgtl %k, $n
  jnz @return_max

  ; Compute collatz steps for %k
  %cur = copy %k
  %steps = constant i64 0
  br @collatz_loop

@collatz_loop:
  %cur_is_one = ceql %cur, constant i64 1
  jnz @collatz_done

  ; If odd: 3x + 1
  ; If even: x / 2
  %odd_bit = bandw %cur, constant i32 1
  %is_odd = cselw %odd_bit, constant i32 1, %odd_bit ;; if bit0 is 1, it's odd. 
  
  %test_odd = ceqw %is_odd, constant i32 1
  jnz @do_odd

@do_even:
  %cur = srll %cur, constant i64 1
  br @update_steps_collatz

@do_odd:
  ; 3*cur + 1
  %triple = addl %cur, %cur
  %cur = addl %triple, %cur
  %cur = addl %cur, constant i64 1
  br @update_steps_collatz

@update_steps_collatz:
  %steps = addl %steps, constant i64 1
  br @collatz_loop

@collatz_done:
  ; Update max
  %is_new_max = csgtl %steps, %max
  %take_val = zextw %is_new_max
  %new_max = cseln %max, %steps, %take_val, %max ;; Wait, cseln syntax is cseln dest, test, if_true, if_false? 
  ;; QBE doc: cseln %dest, %test, %if_true, %if_false.
  %new_max = cseln %new_temp, %is_new_max, %steps, %max
  %max = copy %new_temp
  
  ; k++
  %k = addl %k, constant i64 1
  br @check_k

@return_max:
  ret %max
}
