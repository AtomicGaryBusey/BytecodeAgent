export function l $collatz_max(l $n) {
@entry:
  %max_steps = constant i64 0
  %k = copy $n
  br @next_k

@next_k:
  %k_is_zero = ceql %k, constant i64 0
  jnz @return_max

  %current = copy %k
  %steps = constant i64 0
  br @inner_loop

@inner_loop:
  %is_one = ceql %current, constant i64 1
  jnz @update_max

  %bit0 = bandw %current, constant i32 1
  cselw %is_even, constant i32 0, %bit0

  br @choose_op

@choose_op:
  %if_not_even = ceqw %is_even, constant i32 0
  jnz @op_odd

@op_even:
  %current = srll %current, constant i64 1
  br @update_steps

@op_odd:
  %tmp = addl %current, %current
  %current = addl %tmp, constant i64 1
  br @update_steps

@update_steps:
  %steps = addl %steps, constant i64 1
  br @inner_loop

@update_max:
  %is_greater = csgtl %steps, %max_steps
  %take_new = zextw %is_greater
  %new_max = cseln %max_steps, %take_new, %steps, %max_steps
  %max_steps = copy %new_max
  
  %k = addl %k, constant i64 1
  br @next_k

@return_max:
  ret %max_steps
}
