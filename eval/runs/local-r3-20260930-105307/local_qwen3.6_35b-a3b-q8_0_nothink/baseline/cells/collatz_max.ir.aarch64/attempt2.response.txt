export function l $collatz_max(l $n) {
@entry:
  %max_steps = constant i64 0
  k_loop:
    %k_is_zero = ceql $n, constant i64 0
    jnz @end

    %current = copy $n
    %steps = constant i64 0
    
  inner_loop:
    %is_one = ceql %current, constant i64 1
    jnz @found_max
    
    %bit0 = bandw %current, constant i32 1
    cselw %is_even, constant i32 0, %bit0

    csell %current, %is_even, 
      srl %current, constant i64 1,  %current, 
      addl addl %current, %current, constant i64 1, %current, constant i64 1
    
    %steps = addl %steps, constant i64 1
    goto @inner_loop

  @found_max:
    %is_new_max = csgtl %steps, %max_steps
    cselw %take_new, %is_new_max, constant i32 1, constant i32 0
    cseln %max_steps, %take_new, %steps, %max_steps
    
    $n = addl $n, constant i64 1
    goto @k_loop

@end:
  ret %max_steps
}
