sum_to:
	// Compute sum = n * (n + 1) / 2
	// if n <= 0 return 0
	cmp x0, #0
	ble .Lzero_result

	// n > 0, compute result = n * (n+1) / 2
	add x1, x0, #1  // x1 = n + 1
	mul x1, x0, x1  // x1 = n * (n + 1)

	// Check if n*(n+1) is odd or even to handle division by 2
	// Actually n*(n+1) is always even, so we can just shift right
	srsl x0, x1, #1
	ret
