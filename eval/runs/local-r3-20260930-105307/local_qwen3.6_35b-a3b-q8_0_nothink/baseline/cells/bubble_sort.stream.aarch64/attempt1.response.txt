bubble_sort:
	// int64_t bubble_sort(int64_t *a, int64_t n);
	// a in x0, n in x1
	// return 0 in x0
	
	cmp	x1, #0
	b.le	.done
	
	// We need to preserve callee-saved registers if used.
	// We will use x19-x28 and d8-d15 only if necessary.
	// To minimize register pressure and complexity, let's try to use scratch registers x9-x15.
	// Since n <= 64, we don't need huge stack frames, but must keep sp 16-byte aligned.
	// We'll save x29/x30 (lr) if we call any function, but we won't.
	// Actually, standard AAPCS64 requires saving callee-saved registers in the prologue if used.
	// Let's stick to x9-x15 which are caller-saved (scratch). No need to save them.
	
	// Outer loop: i from n-1 down to 1
	// Inner loop: j from 0 to i-1
	
	mov	x2, x1		// i = n (will decrement to become count)
	sub	x2, x2, #1	// i = n - 1
	
.outer_loop:
	cbnz	x2, .run_outer
	b	.done

.run_outer:
	mov	x3, #0		// j = 0
	
.inner_loop:
	// Check if j < i
	cmp	x3, x2
	bge	.end_inner
	
	// Load a[j] and a[j+1]
	ldr	x4, [x0, x3, uxtl #3]  // x4 = a[j]
	ldr
