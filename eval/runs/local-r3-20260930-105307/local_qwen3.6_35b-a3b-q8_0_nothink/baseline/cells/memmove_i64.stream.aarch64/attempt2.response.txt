memmove_i64:
// Arguments: x0=dst, x1=src, x2=n
// Return 0 in x0 at exit
cbz x2, .L_done
cmp x2, #10 // Small N threshold for unrolled copy
bhi .L_large_block
.L_small_copy:
// Use a simple loop for small N
// We can safely use overlapping logic by copying forward if dst <= src,
// else backward. However, since n is small, we can just use a temp buffer in registers?
// No, we cannot use stack. Registers are limited.
// Actually, memmove with no extra memory must handle overlap.
// If dst < src: copy forward. Safe because we read src[i] before writing dst[i+diff].
// If dst > src: copy backward. Safe because we read src[n-1-i] before writing dst[n-1-i+diff].
// Let's implement this logic.

cmp x0, x1
bhs .L_backward_copy_forward
L_forward_loop:
// Forward copy from start
cbz x2, .L_done
ldp x3, x4, [x1], #16
stp x3, x4, [x0], #16
subs x2, x2, #2
bhi L_forward_loop
cbnz x2, L_forward_single
b .L_done
L_forward_single:
ldr x3, [x1]
str x3, [x0]
b .L_done

L_backward_copy_forward:
// Backward copy from end
// src and dst are pointers to the last element we need to copy?
// No, let's adjust pointers. We want to copy n elements.
// End pointer for src: x1 + (n-1)*8
// End pointer for dst: x0 + (n-1)*8
// But calculating end pointers requires arithmetic.
// Let's use x5/x6 for temp calculation if needed, but we have limited regs.
// Actually, simpler to just do backward loop using negative offsets or adjusting pointers.

// Calculate end pointers
// We need to preserve original src/dst? No, only x0 is result.
// Save n? x2 is about to be used.
mov x3, x1 // temp src ptr
mov x4, x0 // temp dst ptr
add x5, x2, #1 // count for loop control
sub x2, x2, #1

backward_loop:
cbz x2, .L_back_done
ldp x6, x7, [x3, x2, lsl #3]
stp
