isqrt:
# Calculate floor(sqrt(n)) for n in rdi using integer Newton's method.
# Result returned in rax.
# Uses rbx as the current estimate (callee-saved).
# Clobbers: rcx, rdx, rsi, r10-r11 (caller-saved), flags.

push rbx
mov rbx, rdi

test rbx, rbx
je .L_done
cmp rbx, 1
jne .L_newton_start
mov rax, 1
pop rbx
ret

.L_newton_start:
# Initial guess: 2^(ceil(bit_width(n)/2))
bsr rcx, rbx
add rcx, 1
shr rcx, 1
mov rax, 1
shl rax, cl
mov rbx, rax

.L_newton_loop:
# Newton step: x_new = (x_old + n / x_old) / 2
mov rax, rbx
mul rbx # Check for zero divisor or if square > n to avoid div by 0 issues? No, guess is >= sqrt.
# Actually, we need n / rbx.
xor rdx, rdx
mov rax, rbx # dividend low
# We need to divide rdi by rbx. rdi is in rdi.
# div r64 divides rdx:rax by operand.
mov rax, rdi
div rbx # quotient in rax (n/x), remainder in rdx

# x_new = (rbx + (n/rbx)) / 2
add rax, rbx
shr rax, 1

cmp rax, rbx
jbe .L_converged

mov rbx, rax
jmp .L_newton_loop

.L_converged:
# Check if rbx * rbx > n. If so, rbx is too big.
mov rax, rbx
mul rbx # Result in rdx:rax = rbx * rbx

cmp rdx, 0
ja .L_too_big

cmp rax, rdi
jbe .L_correct_result
jmp .L_too_big

.L_correct_result:
mov rax, rbx
pop rbx
ret

.L_too_big:
dec rbx
mov rax, rbx
pop rbx
ret

.L_done:
xor rax, rax
pop rbx
ret
