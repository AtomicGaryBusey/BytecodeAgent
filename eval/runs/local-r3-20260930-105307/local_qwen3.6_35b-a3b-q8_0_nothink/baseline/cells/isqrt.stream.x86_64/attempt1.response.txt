isqrt:
# Handle small cases quickly to save time
test rdi, rdi
je .L_done_zero
cmp rdi, 1
je .L_done_one

# Use Newton's method for integer square root.
# Start with an initial guess x0 = 2^32 (approx sqrt(2^64)).
# Or better: find the highest set bit of n to get a closer start.
# We'll use a binary search approach as it's robust and avoids division issues in the loop condition near overflow,
# but Newton is faster O(log bits). Let's use Newton with careful 128-bit intermediate math using shifts.

# Initial guess: 2^32
mov rax, 1
shl rax, 32

.L_newton_loop:
# next = (x + n/x) / 2
# We need to compute n/x. Since x >= sqrt(n), n/x <= x.
# Division is expensive but necessary for Newton.
# Use div instruction: rdx:rax / rcx -> rax quotient, rdx remainder.
# Note: n is in rdi.
mov rdx, 0
mov rax, rdi # numerator low
div rax      # Wait, I need to divide by current guess 'x' which is in rax? No, divisor is separate.

# Let's restart the logic with registers clearly defined.
# n = rdi (input, preserved)
# x = current guess (rax)
# temp = for calculation

.L_newton_step:
# Calculate n / x
mov rdx, 0
mov rax, rdi # Load n into rdx:rax implicitly by clearing rdx and moving n to rax? No.
             # div rcx divides rdx:rax by rcx.
             # We want to divide n (rdi) by x (rax).
             # So put n in rdx:rax, divisor in rcx.

mov rax, rdi
xor rdx, rdx
mov rcx, rax # This is wrong, I clobbered x.

# Correct approach:
# Save n to a callee saved register or stack if needed? No, rdi is safe if we don't call.
# But div uses rdx:rax. We need x in a register.
# Let's use rbx for x (callee-saved).

# Re-architecture:
# Input n in rdi.
# Use rbx as the iterate x.

mov rbx, 1
shl rbx, 32 # Initial guess 2^32

.L_newton:
# Check if x*x <= n < (x+1)*(x+1) roughly by checking convergence.
# Next estimate y = (x + n/x) / 2

# Compute q = n / x
mov rax, rdi
xor rdx, rdx
div rbx # quotient in rax (q), remainder in rdx (unused)
        # Divisor was rbx (current guess x)

# Compute y = (x + q) / 2
add rax, rbx
shr rax, 1

# If y >= x, then x is the floor sqrt or close.
# For integer square root, if next <= current, we might be oscillating or converged.
# Standard check: if y < x, update x = y and continue.
# If y >= x, stop? Not exactly for all cases, but usually for positive n it converges monotonically from above?
# Actually Newton's method for sqrt(1) starts at large numbers and decreases.
# It can overshoot below if not careful, but with integer division truncation, it tends to settle on k or k+1.
# We want the largest r such that r*r <= n.

cmp rax, rbx
jb .L_update_x

# If y >= x, we are likely at the minimum or oscillating between floor(sqrt(n)) and ceil(sqrt(n)).
# Since we started high, we approach from above. The sequence is non-increasing until it hits the root region.
# If next >= current, we have converged to the bottom of the valley.
# However, due to integer truncation, it might jump between k and k+1.
# We check if x*x <= n.
jmp .L_check_convergence

.L_update_x:
mov rbx, rax
jmp .L_newton

.L_check_convergence:
# Check if rbx * rbx > n. If so, we overshot below the root? 
# Actually, if Newton steps up (y >= x), it means we are at the minimum of the discrete function f(x) = |x^2 - n| roughly.
# We need to ensure rbx is the floor sqrt.
# Test rbx * rbx <= n.
# Use mul: rdx:rax = rbx * rbx.
mov rax, rbx
mul rbx # Result in rdx:rax. Since n < 2^64, if result > n, either high part non-zero or low part > n.

cmp rdx, 0
ja .L_too_big
jb .L_done
# High bits are 0, check low bits
cmp rax, rdi
jbe .L_done
.too_big_check:
.L_too_big:
# rbx * rbx > n, so rbx is too big. Result is rbx - 1.
dec rbx

.done_label:
.L_done:
.L_done_zero:
mov rax, rbx
ret
