isqrt:
# Calculate floor(sqrt(n)) for n in rdi using integer Newton's method.
# Result returned in rax.
# Uses rbx as the current estimate (callee-saved).
# Clobbers: rcx, rdx, rsi, r10-r11 (caller-saved), flags.

# Handle n=0 and n=1 quickly
test rdi, rdi
je .L_done
cmp rdi, 1
jne .L_newton_start
mov rax, 1
ret

.L_newton_start:
# Initial guess: 2^(ceil(bit_width(n)/2))
# Find highest set bit of n to determine shift amount.
# We can use bsr for this. If n=0, handled above.
bsr rcx, rdi
# Shift count for initial guess: (rcx + 1) / 2
# Actually, sqrt(2^k) = 2^(k/2). So if highest bit is at position k (0-indexed),
# the root is roughly at k/2. Let's start with 2^((k/2)+1) to be safe from above.
add rcx, 1
shr rcx, 1
mov rax, 1
shl rax, cl
mov rbx, rax

.L_newton_loop:
# Newton step: x_new = (x_old + n / x_old) / 2
# We need to compute n / rbx.
# Divisor is in rbx. Numerator is n (rdi).
# div rcx divides rdx:rax by rcx.
# So we put n in rdx:rax (clearing rdx) and rbx in a register for divisor.
mov rax, rdi
xor rdx, rdx
div rbx # quotient in rax (n/x), remainder in rdx

# x_new = (rbx + (n/rbx)) / 2
add rax, rbx
shr rax, 1

# Check for convergence or oscillation.
# Newton's method for integer sqrt converges to k or k+1 where k^2 <= n < (k+1)^2.
# The sequence is non-increasing after the first step if we start above sqrt(n).
# If rax >= rbx, we have reached the minimum of the discrete parabola, so rbx is likely floor(sqrt(n)) or close.
cmp rax, rbx
ja .L_update

# We converged or overshot slightly below?
# If rax < rbx, we can update to a better estimate from above.
.L_update:
mov rbx, rax
jmp .L_check_convergence

.L_check_convergence:
# Check if rbx * rbx > n. If so, rbx is too big (e.g., it's k+1).
# We want the largest r such that r*r <= n.
mov rax, rbx
mul rbx # Result in rdx:rax = rbx * rbx

# If high bits are non-zero, product > 2^64 - 1 >= n (since n < 2^64).
cmp rdx, 0
ja .L_too_big

# Check if low bits product > n
cmp rax, rdi
jbe .L_correct_result
jmp .L_too_big

.L_correct_result:
mov rax, rbx
ret

.L_too_big:
# rbx * rbx > n, so rbx is too large. Result is rbx - 1.
dec rbx
mov rax, rbx
ret
