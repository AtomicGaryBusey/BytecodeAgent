fib_rec:
	// n in x0. Returns fib(n) in x0.
	// Base cases: fib(0)=0, fib(1)=1
	cmp	x0, #0
	b.eq	.L_done_zero
	cmp	x0, #1
	b.eq	.L_done_one

	// For n >= 2, fib(n) = fib(n-1) + fib(n-2).
	// We need to make two recursive calls.
	// x0-x18 and x30 are caller-saved (clobbered by bl).
	// sp must be 16-byte aligned at each call site.
	// Entry: sp is 16-byte aligned.

	// Save return address and frame pointer if we treat this as a non-leaf.
	// Even though we don't use x29, saving x30 is mandatory because bl overwrites it.
	stp	x29, x30, [sp, #-16]!
	mov	x29, sp

	// Save n (current x0) on stack because it will be clobbered by recursive calls.
	// We need n to compute n-2 later, or we can just use the saved result of fib(n-1).
	// Let's save n.
	stp	x0, x29, [sp, #-16]! // sp -= 32 relative to original entry sp? No, sp was at -16 already.
	// Current sp offset: -16 (stp fp, lr) -> -32 (stp n, fp). sp is aligned (-48 mod 16 = 0).

	// Prepare argument for first call: fib(n-1)
	sub	x0, x0, #1
	bl	fib_rec

	// Now x0 holds fib(n-1). We need to save this result because the next bl will clobber x0.
	// Also need to prepare n-2 for the second call.
	// Let's restore n from stack.
	// Current sp is at -32 relative to entry.
	// Load original n.
	ldr	x1, [sp, #16] // Load n (stored earlier). Note: fp was stored below n?
	// Let's trace stack layout carefully.
	// Entry sp: S0
	// stp x29, x30, [sp, #-16]! -> S-1: x29, x30. sp is S-16.
	// stp x0, x29, [sp, #-16]! -> S-2: n, x29. sp is S-32.
	// Call fib_rec(n-1). Returns in x0.
	// We need to save x0 (result of fib(n-1)).
	// And we need original n to compute n-2.
	// Original n was stored at [sp, #16] from current sp position?
	// Current sp is S-32. 
	// [S-32+16] = [S-16] contains x29 (saved fp).
	// [S-32+8] = [S-24] contains n. Wait, stp stores two words.
	// Layout at S-32:
	// [S-32]: low word of original n? No, stp x0, x29 stores x0 then x29.
	// So [S-32] = x0 (n), [S-24] = x29.
	// Wait, sp decreased by 16 for stp.
	// Let's re-verify stack operations.
	// Start: sp = P.
	// 1. stp x29, x30, [sp, #-16]! : sp = P-16. Mem[P-16]=x29, Mem[P-8]=x30.
	// 2. stp x0, x29, [sp, #-16]! : sp = P-32. Mem[P-32]=x0 (n), Mem[P-24]=x29.
	
	// After bl fib_rec(x0-1):
	// x0 has fib(n-1).
	// We need to preserve x0 for the addition later.
	// Let's save x0 (fib(n-1)) to stack as well.
	stp	x0, x29, [sp, #-16]! // sp = P-48. Mem[P-48]=x0(fib n-1), Mem[P-40]=x29.
	
	// Now restore original n to compute n-2.
	ldr	x0, [sp, #32] // x0 = Mem[P-32] which is original n. sp is P-48. 48+32=80? No.
	// sp is P-48. We want Mem[P-32]. Offset from sp is +16.
	ldr	x0, [sp, #16] // x0 = original n.
	
	// Prepare argument for second call: fib(n-2)
	sub	x0, x0, #2
	bl	fib_rec
	
	// Now x0 holds fib(n-2).
	// We need to add it to fib(n-1) which is saved on stack.
	ldr	x1, [sp, #16] // x1 = Mem[P-48+16] = Mem[P-32]? No.
	// sp is P-48. Saved fib(n-1) was at Mem[P-48]. So it's at [sp, #0].
	ldr	x1, [sp, #0]  // x1 = fib(n-1).
	
	// Add results
	add	x0, x1, x0
	
	// Cleanup stack and restore frame
	// Restore x29 from the first saved slot (at sp+16 relative to current sp P-48 is P-32, no wait)
	// We need to remove 48 bytes from sp.
	add	sp, sp, #48
	
	// Pop frame pointer and return address
	ldp	x29, x30, [sp], #16 // Wait, we did stp with decrement first. So restore is ldp from sp+16?
	// No, standard epilogue:
	// We saved fp, lr at P-16. sp was P-32 after first stp, then P-48 after second.
	// To restore correctly:
	// First, just restore sp to P-16.
	add	sp, sp, #32 // Now sp is P-16.
	
	// Now pop fp and lr
	ldp	x29, x30, [sp], #16 // sp becomes P. x29=Mem[P-16], x30=Mem[P-8].

	// Return with result in x0
	ret

.L_done_zero:
	mov	x0, #0
	ret

.L_done_one:
	mov	x0, #1
	ret
