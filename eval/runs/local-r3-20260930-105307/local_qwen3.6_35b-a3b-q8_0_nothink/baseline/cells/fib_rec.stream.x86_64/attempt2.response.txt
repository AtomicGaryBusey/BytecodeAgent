fib_rec:
	cmp rdi, 1
	jg .Lrecurse
	je .Lone
	xor eax, eax
	ret
.Lone:
	mov eax, 1
	ret
.Lrecurse:
	sub rsp, 40
	push rbx
	push r12
	push r13
	mov qword ptr [rsp + 40], rdi
	call fib_rec
	mov r12, rax
	mov rdi, qword ptr [rsp + 40]
	dec rdi
	sub rdi, 1
	call fib_rec
	add rax, r12
	pop r13
	pop r12
	pop rbx
	add rsp, 40
	ret
