export function l $count_nonzero_prefix(l %a, l %n) {
    @entry
        %idx = alloc l
        *%idx = 0
    
    @loop
        %i = loadl [%idx]
        %cond = csltl %i, %n
        jz %cond, @done
        
        %off = mull %i, 8
        %addr = addl %a, %off
        %val = loadl [%addr]
        
        %is_zero = ceql %val, 0
        jnz %is_zero, @found
        
        %next_i = addl %i, 1
        storel [%idx], %next_i
        jump @loop
    
    @found
        jump @done
    
    @done
        %result = loadl [%idx]
        ret %result l
}
