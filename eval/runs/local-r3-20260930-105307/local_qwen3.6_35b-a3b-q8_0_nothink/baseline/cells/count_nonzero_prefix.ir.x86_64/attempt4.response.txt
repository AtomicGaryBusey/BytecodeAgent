export function l $count_nonzero_prefix(l %a, l %n) {
    @entry
        %idx = 0 l

    @loop
        %cond = csltl %idx, %n
        jz %cond, @done
        
        %off = mull %idx, 8 l
        %addr = addl %a, %off
        %val = loadl [%addr]
        
        %is_zero = ceql %val, 0 l
        jnz %is_zero, @done
        
        %next_idx = addl %idx, 1 l
        jump @loop

    @done
        ret %idx l
}
