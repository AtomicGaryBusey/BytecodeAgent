count_nonzero_prefix:
	cbz x1, .Lreturn_n
	sub x5, x0, #8
.Lloop_start:
	ldp x2, x3, [x0], #16
	cbnz x2, .Lnext_x3
.Lfound_zero_x2:
	add x0, x0, #8
	sub x0, x0, x5
	ret
.Lnext_x3:
	cbnz x3, .Lcontinue_loop
.Lfound_zero_x3:
	add x0, x0, #8
	sub x0, x0, x5
	subs x1, x1, #2
	b.gt .Lloop_start
	// Handle odd element if we stopped here with n>=1 but only 1 left was x3 which was zero
	// Actually if we are here, x3 was zero. The index is current_x0 - base_x0
	// Since x3 is at offset 8 from start of pair, and we advanced x0 by 16,
	// real address of x3 was old_x0 + 8.
	// Current x0 is old_x0 + 16.
	// We want to return index = (old_x0 + 8 - base) / 8 = (x0 - 8 - base)/8 = (x0 - base)/8 - 1.
	// But simpler: just calculate based on original pointers.
	// Let's use a different approach for the tail.
.Lloop_end_check:
	cbz x1, .Lreturn_n
	ldr x2, [x0], #8
	sub x1, x1, #1
	cbnz x2, .Lloop_end_check
.Lfound_zero_tail:
	add x0, x0, #-8
	sub x0, x0, x5
	ret
.Lcontinue_loop:
	subs x1, x1, #2
	b.gt .Lloop_start
	b .Lloop_end_check
.Lreturn_n:
	mov x0, x1
	ret
