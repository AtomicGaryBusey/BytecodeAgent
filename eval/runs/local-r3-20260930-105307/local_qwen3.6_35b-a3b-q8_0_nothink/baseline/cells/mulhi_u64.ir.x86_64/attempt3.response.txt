export function l $mulhi_u64(l %a, l %b) {
@start
	%ah = shr.u.l %a, 32
	%al = and.l %a, 4294967295
	%bh = shr.u.l %b, 32
	%bl = and.l %b, 4294967295
	%p0 = mul.u.l %al, %bl
	%t1_hi = mul.u.l %ah, %bl
	%t2_hi = mul.u.l %al, %bh
	
	%carry1_shr = shr.u.l %p0, 32
	%low16 = and.l %p0, 4294967295
	
	%mid1_add = add.u.l %carry1_shr, %t1_hi
	%mid1_lo = and.l %mid1_add, 4294967295
	%carry2_shr = shr.u.l %mid1_add, 32
	
	%mid2_add = add.u.l %low16, %t2_hi
	%carry3_shr = shr.u.l %mid2_add, 32
	
	%high_part1 = add.u.l %carry2_shr, %carry3_shr
	
	%t3_hi = mul.u.l %ah, %bh
	
	%final_high = add.u.l %high_part1, %t3_hi
	ret.l %final_high
}
