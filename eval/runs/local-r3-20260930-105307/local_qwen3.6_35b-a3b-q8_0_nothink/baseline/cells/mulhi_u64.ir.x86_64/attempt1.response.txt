export function l $mulhi_u64(l %a, l %b) {
@start
	%ah = shr.l %a, 32
	%al = and.l %a, 4294967295
	%bh = shr.l %b, 32
	%bl = and.l %b, 4294967295
	%p0 = mul.u.l %al, %bl
	%t1_hi = mul.u.l %ah, %bl
	%t2_hi = mul.u.l %al, %bh
	%carry1 = shr.u.w %p0, 32
	%low16 = and.l %p0, 4294967295
	%mid1 = add.l %carry1, %t1_hi
	%mid2 = add.l %low16, %t2_hi
	%carry2 = shr.u.w %mid1, 32
	%sum_low = and.l %mid1, 4294967295
	%carry3 = shr.u.w %mid2, 32
	%high_part1 = add.l %carry2, %carry3
	%t3_hi = mul.u.l %ah, %bh
	%final_high = add.l %high_part1, %t3_hi
	ret.l %final_high
}
