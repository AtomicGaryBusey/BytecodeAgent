export function l $mulhi_u64(l %a, l %b) {
@start
	%ah = shr.u.l %a, 32
	%al = and.l %a, 4294967295
	%bh = shr.u.l %b, 32
	%bl = and.l %b, 4294967295
	%p0 = mul.u.l %al, %bl
	%t1_hi = mul.u.l %ah, %bl
	%t2_hi = mul.u.l %al, %bh
	%carry1_shr = shr.u.w %p0, 32
	%carry1 = extu.l %carry1_shr
	%low16 = and.l %p0, 4294967295
	%mid1_hi = add.u.l %carry1, %t1_hi
	%mid1_lo = and.l %mid1_hi, 4294967295
	%carry2_shr = shr.u.w %mid1_hi, 32
	%carry2 = extu.l %carry2_shr
	%mid2_add = add.u.l %low16, %t2_hi
	%carry3_shr = shr.u.w %mid2_add, 32
	%carry3 = extu.l %carry3_shr
	%mid2_lo = and.l %mid2_add, 4294967295
	%high_part1 = add.u.l %carry2, %carry3
	%t3_hi = mul.u.l %ah, %bh
	%final_high = add.u.l %high_part1, %t3_hi
	ret.l %final_high
}
