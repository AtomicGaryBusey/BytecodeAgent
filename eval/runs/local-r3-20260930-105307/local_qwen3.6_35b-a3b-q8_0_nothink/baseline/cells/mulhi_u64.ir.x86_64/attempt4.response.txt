export function l $mulhi_u64(l %a, l %b) {
@start
	%ah = shr.l %a, 32
	%al = and.l %a, 4294967295
	%bh = shr.l %b, 32
	%bl = and.l %b, 4294967295

	%p0 = mul.u.l %al, %bl
	%carry_p0 = shr.l %p0, 32
	%low16 = and.l %p0, 4294967295

	%m1 = mul.u.l %ah, %bl
	%m1_add = add.u.l %carry_p0, %m1
	%m1_carry = shr.l %m1_add, 32
	%m1_low = and.l %m1_add, 4294967295

	%m2 = mul.u.l %al, %bh
	%m2_add = add.u.l %low16, %m2
	%m2_carry = shr.l %m2_add, 32
	%m2_low = and.l %m2_add, 4294967295

	$mid_sum = add.u.l $m1_carry, $m2_carry

	$m3 = mul.u.l %ah, %bh
	$final_high = add.u.l $m3, $mid_sum
	ret.l $final_high
}
