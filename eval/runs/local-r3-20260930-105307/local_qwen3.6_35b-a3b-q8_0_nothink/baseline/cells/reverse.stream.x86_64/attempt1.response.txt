reverse:
    test rsi, rsi
    jle .Ldone
    lea rcx, [rsi - 1]
    mov r8, rdi
    mov r9, rdi
    add r9, rcx
    shl rcx, 3
    sub r9, rcx
.Lloop:
    cmp r8, r9
    jge .Ldone
    mov rax, qword ptr [r8]
    mov rdx, qword ptr [r9]
    mov qword ptr [r8], rdx
    mov qword ptr [r9], rax
    add r8, 8
    sub r9, 8
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
