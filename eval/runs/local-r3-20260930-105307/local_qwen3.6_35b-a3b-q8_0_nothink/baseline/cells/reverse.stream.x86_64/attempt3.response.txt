reverse:
    test rsi, rsi
    jle .Ldone
    mov r10, rdi
    lea r11, [rsi - 1]
    imul r11, r11, 8
    add r11, rdi
    shr ecx, 1
.Lloop:
    test rcx, rcx
    jle .Ldone
    mov rax, qword ptr [r10]
    mov rdx, qword ptr [r11]
    mov qword ptr [r10], rdx
    mov qword ptr [r11], rax
    add r10, 8
    sub r11, 8
    dec ecx
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
