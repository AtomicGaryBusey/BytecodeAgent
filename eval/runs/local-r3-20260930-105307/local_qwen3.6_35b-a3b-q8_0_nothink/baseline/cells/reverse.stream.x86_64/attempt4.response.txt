    jle .Ldone
    lea r10, [rdi + rsi*8 - 8]
    shr rsi, 1
.Lloop:
    mov rax, qword ptr [rdi]
    mov rcx, qword ptr [r10]
    mov qword ptr [rdi], rcx
    mov qword ptr [r10], rax
    add rdi, 8
    sub r10, 8
    dec rsi
    jnz .Lloop
.Ldone:
    xor eax, eax
    ret
