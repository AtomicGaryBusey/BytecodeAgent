export function l $map_sum(l *a, l n, l f) {
    %result = consti 0l

    @loop:
    %cmp = csltl n, consti 1l
    jnz %done

    %idx = sub n, consti 1l
    %addr = ptr_add a, idx
    %val = loadl *%addr

    %res = call f(val)
    %sum = ladd result, res

    %new_n = sub n, consti 1l
    result = sum
    n = new_n
    jump @loop

    @done:
    return result
}
