@map_sum: export function l (l *a, l n, l f) {
    %zero = consti 0
    %one = consti 1
    %result = ladd %zero, %zero
    
    @loop_start:
    %cmp = csltl %n, %one
    jnz %done
    
    %idx = sub %n, %one
    %addr = ptr_add l *a, l %idx
    %val = loadl l *%addr
    
    %res = call l %f(l %val)
    %sum = ladd %result, %res
    
    %new_n = sub %n, %one
    %result = %sum
    %n = %new_n
    jump @loop_start
    
    @done:
    return l %result
}
