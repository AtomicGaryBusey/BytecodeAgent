export function l $map_sum(l %a, l %n, l %f) {
    %result = consti 0l
    %i = consti 0l

@loop:
    %done = cselge %i, %n
    jnz %end

    %addr = ptr_add %a, %i
    %val = loadl *%addr
    %res = call %f(l %val)
    %sum = ladd %result, %res
    %next_i = add %i, consti 1l
    %result = %sum
    %i = %next_i
    jump @loop

@end:
    return %result
}
