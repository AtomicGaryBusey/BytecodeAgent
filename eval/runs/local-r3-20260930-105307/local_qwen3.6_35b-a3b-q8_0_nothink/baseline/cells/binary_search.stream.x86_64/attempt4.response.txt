binary_search:
    cmp rsi, 0
    jle .L_not_found
    mov r12, rdi          # r12 = base pointer a
    xor r8, r8            # r8 = low = 0
    lea r9, [rsi - 1]     # r9 = high = n - 1
.L_loop:
    cmp r8, r9
    jg .L_not_found
    mov rax, r8
    add rax, r9
    shr rax, 1            # rax = mid index
    mov rcx, qword ptr [r12 + rax*8] # a[mid]
    cmp rcx, rdx          # compare a[mid] with key (rdx)
    je .L_found
    jl .L_go_right
.L_go_left:
    dec r9
    jmp .L_loop
.L_go_right:
    inc r8
    jmp .L_loop
.L_found:
    mov rax, rax          # mid index is in rax
    ret
.L_not_found:
    mov rax, -1
    ret
