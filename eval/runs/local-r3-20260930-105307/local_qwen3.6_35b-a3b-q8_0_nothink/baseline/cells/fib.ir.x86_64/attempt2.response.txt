export function l $fib(l %n) {
@entry:
    cmpslt i64 %n, 1
    jz @loop_start
    
    ret i64 0

@loop_start:
    cmpslt i64 %n, 2
    jnz @done_1
    
    %a = add i64 0, 1
    %b = i64 1
    %i = sub i64 %n, 2
    
@iter:
    cmpslt i64 %i, 1
    jz @done_final
    
    %sum = add i64 %a, %b
    %a = %b
    %b = %sum
    %next_i = sub i64 %i, 1
    jmp @iter

@done_1:
    ret i64 1

@done_final:
    ret i64 %b
}
