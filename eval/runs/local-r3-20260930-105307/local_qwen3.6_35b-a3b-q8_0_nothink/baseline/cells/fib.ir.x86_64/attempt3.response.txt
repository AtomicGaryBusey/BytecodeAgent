export function l $fib(l %n) {
@entry:
    cmpslt i64 %n, 0
    jnz @error

    cmpslt i64 %n, 2
    jnz @loop_start

    ret i64 %n

@loop_start:
    %a = i64 0
    %b = i64 1
    %i = sub i64 %n, 1

@iter:
    cmpslt i64 %i, 2
    jnz @done

    %sum = add i64 %a, %b
    %new_a = %b
    %new_b = %sum
    %next_i = sub i64 %i, 1

    %a = %new_a
    %b = %new_b
    %i = %next_i

    jmp @iter

@done:
    ret i64 %b

@error:
    ret i64 -1
}
