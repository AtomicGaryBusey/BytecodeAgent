export function l $fib(l %n) {
@entry:
    %n_val = phi i64 [%n]
    %c1 = phi i64 [0]
    %c2 = phi i64 [1]
    
@loop:
    %is_zero = icmp slt, %n_val, 1
    jnz @base_case_small
    
    %next_n = sub i64, %n_val, 1
    %sum = add i64, %c1, %c2
    %new_c1 = %c2
    %new_c2 = %sum
    
@loop_again:
    %is_done = icmp slt, %next_n, 1
    jnz @done
    
    %next_next_n = sub i64, %next_n, 1
    %sum2 = add i64, %new_c1, %new_c2
    %newer_c1 = %new_c2
    %newer_c2 = %sum2
    
    jmp @loop_again
    
@base_case_small:
    %result = phi i64 [0, @entry]
    
@done:
    %result_final = phi i64 [%next_n, @loop], [%sum2, @loop_again]
    ret i64 %result_final
}
