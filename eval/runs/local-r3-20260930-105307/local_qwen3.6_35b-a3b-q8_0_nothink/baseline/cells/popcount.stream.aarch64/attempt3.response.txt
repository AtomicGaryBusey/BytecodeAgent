popcount:
	// Step 1: Count bits in each 2-bit group
	mov x1, #0x5555555555555555
	and x2, x0, x1
	lsr x3, x0, #1
	sub x4, x3, x2

	// Step 2: Count bits in each 4-bit group
	mov x1, #0x3333333333333333
	and x5, x4, x1
	lsr x6, x4, #2
	add x7, x5, x6

	// Step 3: Count bits in each 8-bit group
	mov x1, #0x0f0f0f0f0f0f0f0f
	and x5, x7, x1
	lsr x6, x7, #4
	add x7, x5, x6

	// Step 4: Sum all bytes using shifts and adds
	orr x5, x7, x7, lsl #32
	adds x6, x5, x5, lsr #32
	and x5, x6, #0xffff
	orr x6, x5, x5, lsl #16
	adds x0, x6, x6, lsr #16
	and x0, x0, #0xff
	ret
