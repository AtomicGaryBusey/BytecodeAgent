popcount:
	// Use bit manipulation to count bits in O(1) without tables or loops.
	// This is the classic parallel bit count algorithm.
	
	// Step 1: Count bits in each 2-bit group
	mov x1, #0x5555555555555555
	and x2, x0, x1
	lsr x3, x0, #1
	sub x4, x3, x2
	
	// Step 2: Count bits in each 4-bit group
	mov x5, #0x3333333333333333
	and x6, x4, x5
	lsr x7, x4, #2
	sub x4, x6, x7
	
	// Step 3: Count bits in each 8-bit group
	mov x1, #0x0f0f0f0f0f0f0f0f
	and x2, x4, x1
	lsr x3, x4, #4
	add x4, x2, x3
	
	// Step 4: Count bits in each 16-bit group (multiply and shift)
	mov x5, #0x0101010101010101
	and x6, x4, x5
	mul x7, x4, x5
	sub x0, x7, x6
	
	// Step 5: Horizontal sum across the byte sums
	movk x1, #0xffff, lsl #32
	and x2, x4, x1 // Keep only lower part for final add? No, we need to shift.
	
	// Actually, let's do horizontal add using shifts and adds for clarity and correctness on 64-bit result
	
	// Re-do Step 5: Sum up the bytes in x4 into a single 64-bit count.
	// x4 currently has sums of 8-bit chunks in each byte position? 
	// No, after step 3, x4 has 8-bit sums. 
	// Let's restart the final accumulation from Step 3 result (x4 holds 8-bit counts).
	
	// Sum adjacent bytes: x4 = b7..b0. We want sum(b7..b0) -> total count.
	
	// Move to scratch registers for clarity
	mov x5, x4
	
	// Sum pairs of bytes into words (16 bits each), then sum words, etc.
	// This is getting complex. Let's use the multiplication trick which is standard.
	
	// Reset x0 from input? No, we modified x4. Let's keep x4 as the accumulator.
	// x4 holds 8 counts of up to 8 bits each.
	
	// Multiply by magic number to sum bytes into top byte
	mov x1, #0x0101010101010101
	and x2, x4, x1
	mul x3, x4, x1
	sub x0, x3, x2
	
	// Now x0 contains the sum of bytes in the most significant byte?
	// Actually, `x3 - x2` is a bit tricky. Let's use shifts.
	
	// Alternative Step 5: Shift and add to collapse into one register.
	mov x0, x4
	lsr x1, x0, #8
	add x0, x0, x1
	lsr x2, x0, #16
	add x0, x0, x2
	lsr x3, x0, #32
	add x0, x0, x3
