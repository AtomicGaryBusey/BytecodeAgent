popcount:
	// Step 1: Count bits in each 2-bit group
	mov x1, #0x5555555555555555
	and x2, x0, x1
	lsr x3, x0, #1
	sub x4, x3, x2

	// Step 2: Count bits in each 4-bit group
	mov x1, #0x3333333333333333
	and x2, x4, x1
	lsr x3, x4, #2
	add x4, x2, x3

	// Step 3: Count bits in each 8-bit group
	mov x1, #0x0f0f0f0f0f0f0f0f
	and x2, x4, x1
	lsr x3, x4, #4
	add x4, x2, x3

	// Step 4: Sum bytes into the top byte using multiplication
	mov x1, #0x0101010101010101
	mul x0, x4, x1

	// Step 5: Extract the count from the most significant byte
	lsr x0, x0, #56
	ret
