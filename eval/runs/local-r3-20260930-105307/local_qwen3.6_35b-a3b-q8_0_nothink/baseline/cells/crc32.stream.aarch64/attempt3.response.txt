crc32:
	// x0 = p, x1 = n
	cbnz	x1, .Lprocess_bytes
	mov	w0, #0xFFFFFFFF
	ret
.Lprocess_bytes:
	sub	sp, sp, #16
	stp	x29, x30, [sp]
	add	x29, sp, #0
	mov	x19, x0 // save p
	mov	x20, x1 // save n
	adr	x21, .Lcrc_table_base
.Lmain_loop:
	cbnz	x20, .Lhas_bytes
	b	.Lfinalize
.Lhas_bytes:
	sub	x22, x20, #1
	str	w22, [sp, #4] // save remaining count (partial)
	ldrb	w8, [x19], #1 // load byte
	mov	w9, #0xFFFFFFFF // initial crc
.Lbyte_loop:
	eor	w9, w9, w8
	mov	w10, #32 // 32 bits per byte
.Lbit_loop_start:
	and	x11, w9, #1
	c
