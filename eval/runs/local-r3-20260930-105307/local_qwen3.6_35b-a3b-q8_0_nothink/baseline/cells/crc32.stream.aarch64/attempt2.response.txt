crc32:
	// x0 = p, x1 = n
	cbnz	x1, .Lprocess_bytes
	mov	w0, #0xFFFFFFFF
	ret
.Lprocess_bytes:
	and	x8, x1, #7 // remainder
	cbnz	x8, .Lload_remaining
	sub	x2, x1, x8 // aligned count
	and	x9, x1, #7 // actual remainder if any
	mov	w3, #0xFFFFFFFF // crc in w3 (lower 32)
.Lmain_loop:
	ldrb	w4, [x0], #1
	eor	w3, w3, w4
	mov	w5, #8
.Lbit_loop:
	and	x6, w3, #1
	c
