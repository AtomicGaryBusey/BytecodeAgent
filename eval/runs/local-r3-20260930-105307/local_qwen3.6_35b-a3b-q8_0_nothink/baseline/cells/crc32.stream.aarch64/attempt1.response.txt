crc32:
	// x0 = p, x1 = n
	cbnz	x1, .Lprocess_bytes
	mov	x0, #0xFFFFFFFF
	b	.Lreturn
.Lprocess_bytes:
	sub	x2, x1, #1
	mov	x3, #0xFFFFFFFF // crc accumulator (lower 32 bits of x3)
	and	x4, x2, #7
	cbnz	x4, .Lhandle_tail
	add	x5, x0, x1 // end pointer
.Lloop:
	ldrb	w6, [x0], #1
	eor	x3, x3, x6
	mov	x7, #8
.Lbit_loop:
	and	x8, x3, #1
	cbnz	x8, .Lapply_poly
	asr	x3, x3, #1
	b	.Ldec_bit_count
.Lapply_poly:
	asr	x9, x3, #1
	eor	x3, x9, #0xEDB88320
.L
