export function l $mulhi_u64(l %a, l %b) {
    @start
        ; Split a and b into low and high 32-bit halves
        ; a_lo = a & 0xFFFFFFFF, a_hi = a >> 32
        ; b_lo = b & 0xFFFFFFFF, b_hi = b >> 32
        
        %a_lo = andl %a, 4294967295      ; a & 0xFFFFFFFF
        %a_hi = shrli %a, 32             ; a >> 32
        %b_lo = andl %b, 4294967295      ; b & 0xFFFFFFFF
        %b_hi = shrli %b, 32             ; b >> 32
        
        ; Compute the four 32x32 -> 64-bit products:
        ; p0 = a_lo * b_lo (lowest 32 bits of result, ignored for high part)
        ; p1 = a_lo * b_hi
        ; p2 = a_hi * b_lo
        ; p3 = a_hi * b_hi (highest 32 bits contribute to high 64 bits directly)
        
        %p0 = mull %a_lo, %b_lo
        %p1 = mull %a_lo, %b_hi
        %p2 = mull %a_hi, %b_lo
        %p3 = mull %a_hi, %b_hi
        
        ; The high 64 bits of the 128-bit product consist of:
        ; upper_64 = p3 + (upper_32(p1) + upper_32(p2) + carry_from_low_parts)
        
        ; Extract upper 32 bits of p1 and p2
        %p1_hi = shrli %p1, 32
        %p2_hi = shrli %p2, 32
        
        ; Lower 32 bits of p1 and p2 contribute to the middle 64 bits, but we need
        ; their carry into the high part. The sum of lower_32(p1) + lower_32(p2) + 
        ; upper_32(p0) produces a carry. However, a simpler approach:
        
        ; Full product = p3 << 64 | (p2_hi << 32 | p2_lo) | (p1_hi << 32 | p1_lo) | p0
        ; High 64 bits = p3 + (p2_hi << 32) + (p1_hi << 32) + carry_from_low_half
        
        ; The low 64 bits of the product are: p1_lo + (p2_lo << 32) + p0
        ; We need the carry from adding these three values into the high part.
        
        %p0_lo = andl %p0, 4294967295    ; lower 32 bits of p0
        %p1_lo = andl %p1, 4294967295    ; lower 32 bits of p1
        %p2_lo = andl %p2, 4294967295    ; lower 32 bits of p2
        
        ; Sum low parts: temp = p0 + (p1_lo << 32) + (p2_lo << 32)
        ; But it's easier to compute: mid = p0 + (%p1_lo << 32) + (%p2_lo << 32)
        ; and track carry.
        
        ; Let's compute the middle 32 bits and the carry into the high part.
        %shifted_p1_lo = slli %p1_lo, 32
        %shifted_p2_lo = slli %p2_lo, 32
        
        %low_sum = addl %p0, %shifted_p1_lo
        ; Check for overflow in low_sum + shifted_p2_lo
        %mid_sum = addl %low_sum, %shifted_p2_lo
        
        ; The carry into the high part is determined by whether low_sum + shifted_p2_lo 
        ; overflowed. But we also need to account for the fact that p1_hi and p2_hi 
        ; are added at bit 32 position.
        
        ; Actually, the standard decomposition:
        ; Result[64:127] = a_hi*b_hi + (a_hi*b_lo)>>32 + (a_lo*b_hi)>>32 + carry
        
        %carry_mid = addl %p1_hi, %p2_hi   ; Sum of the upper halves of the cross terms
        %high_base = addl %p3, %carry_mid  ; a_hi*b_hi + top32(p1) + top32(p2)
        
        ; Now we need to add the carry from the lower part addition.
        ; The lower 64 bits of the product are: p0 + (p1_lo << 32) + (p2_lo << 32)
        %low_part_sum = addl %shifted_p1_lo, %shifted_p2_lo
        %low_total = addl %p0, %low_part_sum
        
        ; The carry from low_total is 1 if it overflowed past 64 bits.
        %cmp_overflow = csltl %low_total, %p0      ; Check if p0 + low_part_sum overflows (unsigned)
        ; Actually, for unsigned add, carry out is detected by result < operand that was added to the smaller value.
        ; Better: carry = (low_total < p0) || ((low_total == p0 && low_part_sum != 0)) but simpler:
        ; For a + b, if (a + b) < a, there was a carry. But we did p0 + (shifted_p1_lo + shifted_p2_lo).
        
        %temp = addl %low_part_sum, %p0
        %has_carry = csltl %temp, %p0
        
        ; Add the carry to high_base
        %result = adcl %high_base, %has_carry  ; adcl adds with carry flag set by previous comparison? 
        ; QBE doesn't have adcl. We need to simulate add-with-carry.
        
        ; Let's use a different approach: compute everything explicitly.
        @start2
        ; Re-compute cleanly
        
        %a_lo = andl %a, 4294967295
        %a_hi = shrli %a, 32
        %b_lo = andl %b, 4294967295
        %b_hi = shrli %b, 32
        
        %p0 = mull %a_lo, %b_lo
        %p1 = mull %a_lo, %b_hi
        %p2 = mull %a_hi, %b_lo
        %p3 = mull %a_hi, %b_hi
        
        ; p0: 64 bits, contributes to result[0:63]
        ; p1: a_lo * b_hi, shifted by 32 in effective position. Contributes to result[32:95]
        ; p2: a_hi * b_lo, shifted by 32. Contributes to result[32:95]
        ; p3: a_hi * b_hi, shifted by 64. Contributes to result[64:127]
        
        ; High 64 bits = p3 + (p1 >> 32) + (p2 >> 32) + carry_from(p0 + (p1 & 0xFFFFFFFFFFFFFFFF)<<? ... 
        
        ; Let's build the high 64 bits step by step.
        ; The contribution to bit positions 64+ comes from:
        ; - p3 (bits 64-127 of product, stored in 64-bit register)
        ; - upper 32 bits of p1 (at bits 64-95)
        ; - upper 32 bits of p2 (at bits 64-95)
        ; - carry from the addition of lower parts into bit 64.
        
        %p1_high32 = shrli %p1, 32
        %p2_high32 = shrli %p2, 32
        
        ; Sum these three: p3 + p1_high32 + p2_high32
        %sum_high = addl %p3, %p1_high32
        %carry1_w = csltl %sum_high, %p3  ; Carry from p3 + p1_high32
        
        %sum_high_with_p2 = addl %sum_high, %p2_high32
        %carry2_w = csltl %sum_high_with_p2, %sum_high
        
        ; Now compute carry into bit 64 from the lower parts.
        ; Lower 64 bits sum = p0 + (p1_low << 32) + (p2_low << 32)
        %p1_low = andl %p1, 4294967295
        %p2_low = andl %p2, 4294967295
        
        %shift_p1_low = slli %p1_low, 32
        %shift_p2_low = slli %p2_low, 32
        
        %sum_shifted = addl %shift_p1_low, %shift_p2_low
        %carry_shift_w = csltl %sum_shifted, %shift_p1_low
        
        %low_total = addl %p0, %sum_shifted
        %carry_low_w = csltl %low_total, %p0
        
        ; Final carry into high part is 1 if (carry_shift OR carry_low) or if the actual addition caused overflow.
        ; Actually, the carry into bit 64 is 1 if p0 + shift_p1_low + shift_p2_low >= 2^64.
        %carry_to_high_w = orw %carry_shift_w, %carry_low_w
        
        ; Add this carry to our high sum.
        %result = addl %sum_high_with_p2, %carry_to_high_w
        
        ret %result
    }
}
