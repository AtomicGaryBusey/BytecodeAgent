export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %cmp = ceql %n, 0i64
    jnz @done_zero
@loop
    %idx = loadl %a
    %neq = csetne %idx, 0i64
    %n_dec = addl %n, -1i64
    %n_zero = ceql %n_dec, 0i64
    jnz @done_n_minus_1
    jz @loop_end
    %a_inc = addl %a, 8i64
    %n_loop = %n_dec
    ret i64 %n_loop
@loop_end
    %ptr_add = addl %a, 8i64
    %n_next = %n_dec
    %a_new = %ptr_add
    %n_curr = %n_next
    jnz @start_check_again
@done_n_minus_1
    ret i64 %n_dec
@done_zero
    ret i64 0i64
@start_check_again
    %cmp2 = ceql %n_curr, 0i64
    jnz @final_zero
    %val = loadl %a_new
    %is_zero = ceql %val, 0i64
    %if_z = zextw %is_zero
    jz @recurse_not_zero
@recurse_done
    ret i64 0i64
@recurse_not_zero
    %ptr2 = addl %a_new, 8i64
    %n2 = sub l %n_curr, 1i64
    %res = call l $count_nonzero_prefix(l %ptr2, l %n2)
    ret i64 %res
@final_zero
    ret i64 0i64
}
