@start
    %is_zero_n = ceql %n, 0i64
    jnz @return_zero

@loop_top
    %val = loadl [%a]
    %is_zero_val = ceql %val, 0i64
    jnz @return_result

    %a_inc = addl %a, 8i64
    %n_dec = subl %n, 1i64
    %done_n = ceql %n_dec, 0i64
    jnz @return_one_less

    %a = %a_inc

    jz @loop_top

@return_result

@return_one_less

}
