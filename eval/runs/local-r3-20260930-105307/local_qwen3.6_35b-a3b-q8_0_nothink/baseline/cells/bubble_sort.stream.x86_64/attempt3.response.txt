bubble_sort:
    test rsi, rsi
    je .L_done
    mov r12, rsi
    sub r12, 1
.L_outer_loop:
    mov r13, rdi
    mov r14, r12
.L_inner_loop:
    mov rax, qword ptr [r13]
    cmp qword ptr [r13 + 8], rax
    jle .L_no_swap
    mov rbx, qword ptr [r13 + 8]
    mov qword ptr [r13], rbx
    mov qword ptr [r13 + 8], rax
.L_no_swap:
    add r13, 8
    dec r14
    jne .L_inner_loop
    dec r12
    jg .L_outer_loop
.L_done:
    xor eax, eax
    ret
