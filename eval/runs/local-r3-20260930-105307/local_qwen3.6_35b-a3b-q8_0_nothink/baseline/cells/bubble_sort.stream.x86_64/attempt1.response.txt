bubble_sort:
    test rsi, rsi
    je .L_done
    mov r9, rsi
    sub r9, 1
.L_outer_loop:
    test r9, r9
    jle .L_done
    xor rbx, rbx
    mov r8, rsi
    sub r8, 1
    mov rcx, rsi
    dec rcx
    lea rdi, [rdi + 8]
.L_inner_loop:
    cmp rbx, rcx
    jge .L_outer_cont
    mov rax, qword ptr [rdi + rbx*8]
    mov rdx, qword ptr [rdi + (rbx+1)*8]
    cmp rax, rdx
    cmovle rax, rdx
    cmova rdx, rax
    mov qword ptr [rdi + rbx*8], rdx
    mov qword ptr [rdi + (rbx+1)*8], rax
    inc rbx
    jmp .L_inner_loop
.L_outer_cont:
    dec r9
    jmp .L_outer_loop
.L_done:
    xor eax, eax
    ret
