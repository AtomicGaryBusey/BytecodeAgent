bubble_sort:
    test rsi, rsi
    je .L_done
    mov r9, rsi
    sub r9, 1
.L_outer_loop:
    mov r10, r9
    mov rcx, rdi
    lea rdx, [rcx + 8]
.L_inner_loop:
    cmp r10, 0
    jle .L_outer_done
    mov rax, qword ptr [rcx]
    cmp qword ptr [rdx], rax
    jle .L_no_swap
    mov rbx, qword ptr [rdx]
    mov qword ptr [rcx], rbx
    mov qword ptr [rdx], rax
.L_no_swap:
    add rcx, 8
    add rdx, 8
    dec r10
    jmp .L_inner_loop
.L_outer_done:
    dec r9
    jg .L_outer_loop
.L_done:
    xor eax, eax
    ret
