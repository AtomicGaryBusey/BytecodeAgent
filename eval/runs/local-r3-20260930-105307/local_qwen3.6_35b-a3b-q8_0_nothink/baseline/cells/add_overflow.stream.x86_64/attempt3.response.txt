add_overflow:
    mov rax, rdi
    add rax, rsi
    test rdx, rdx
    jne .store_out
.ret_zero:
    xor eax, eax
    ret
.store_out:
    mov qword ptr [rdx], rax
    cmp rdi, 0
    js .neg_a
.pos_a:
    cmp rsi, 0
    jns .no_overflow
    mov eax, 1
    ret
.neg_a:
    cmp rsi, 0
    jz .no_overflow
    cmp rsi, -9223372036854775807
    ja
