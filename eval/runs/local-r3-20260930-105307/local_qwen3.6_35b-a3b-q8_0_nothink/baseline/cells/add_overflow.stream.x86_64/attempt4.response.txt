add_overflow:
    mov rax, rdi
    add rax, rsi
    mov qword ptr [rdx], rax
    # Check for signed overflow
    # Overflow occurs if operands have same sign and result has different sign
    xor rcx, rcx
    cmp rdi, 0
    js .neg_a
.pos_a:
    # a >= 0
    test rsi, rsi
    jns .no_overflow
    # a >= 0, b < 0. Sum can only overflow if result > a (wrap around positive)
    # But since b is negative, sum <= a, so no overflow from positive to negative wrap
    # Actually: if a>=0 and b<0, result is between min(a+b, a). Max result is a. No overflow.
    # Wait, let's use the standard flag check.
    jc .overflow_pos_to_neg # Not possible for a>=0, b<0 in 64-bit without carry out? 
    # Actually, unsigned overflow (carry) happens if a > result when adding positive numbers.
    # Let's just use ADC approach or test the sign bits directly.
    jnz .no_overflow
.no_overflow:
    xor eax, eax
    ret
.overflow_pos_to_neg:
    mov eax, 1
    ret
.neg_a:
    # a < 0
    test rsi, rsi
    js .neg_b
.pos_b:
    # a < 0, b >= 0. Sum can only overflow if result < a (wrap around negative)
    # Since b>=0, sum >= a. No overflow possible into lower half of range unless it wrapped through zero?
    # If a is very negative and b is large positive, result could be positive.
    # This is underflow for signed: min_int64 - (negative) -> positive? No, adding positive to negative moves up.
    # So no overflow in this case.
    jnz .no_overflow
.neg_b:
    # a < 0, b < 0
    # Sum can only be negative or wrap to positive (overflow)
    # If result >= 0, we have signed overflow
    test rax, rax
    js .no_overflow
    mov eax, 1
    ret
