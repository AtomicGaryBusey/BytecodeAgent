sum8:
    movsxd rax, edx # c is in rdx (int64_t passed as 3rd arg)
    shl rax, 1
    add rax, rcx # 2*c + c = 3*c
    shl rax, 1
    sub rax, rsi # a + 3*c - 2*b
    movsxd rdx, ecx # d is in rcx (4th arg)
    shl rdx, 2
    sub rax, rdx # a - 2*b + 3*c - 4*d
    movsxd rdx, r8d # e is in r8 (5th arg)
    shl rdx, 2
    add rdx, r8 # 5*e
    add rax, rdx # ... + 5*e
    movsxd rdx, r9d # f is in r9 (6th arg)
    shl rdx, 1
    shl rdx, 1
    sub rax, rdx # ... - 6*f
    mov r10, [rsp + 8] # g is at [rsp+8]
    shl r10, 2
    add r10, r10 # 7*g
    add rax, r10 # ... + 7*g
    mov r10, [rsp + 16] # h is at [rsp+16]
    shl r10, 3
    sub rax, r10 # ... - 8*h
    ret
