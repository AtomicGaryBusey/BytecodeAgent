sum8:
    # Arguments: a(rdi), b(rsi), c(rdx), d(rcx), e(r8), f(r9)
    # Stack: [rsp]=ret, [rsp+8]=g, [rsp+16]=h

    # Compute -2*b
    mov rax, rsi
    neg rax
    add rax, rax          # rax = -2*b

    # Add 3*c
    mov rdx, rdx          # c is already in rdx (3rd arg)
    lea rax, [rax + rdx*8] # rax += 8*c; then we need +3*c. 
                           # Wait, let's do it properly.
                           # rax = -2*b + 3*c
    # Let's restart accumulation cleanly.

    # a(rdi) - 2*b(rsi) + 3*c(rdx) - 4*d(rcx) + 5*e(r8) - 6*f(r9) + 7*g([rsp+8]) - 8*h([rsp+16])

    mov rax, rdi          # a
    add rax, rsi          # a + b (temporary, will subtract later) -- no, let's use mul or shifts.
    
    # Strategy: accumulate in rax, rbx, rcx, rdx using lea/add/sub/shl
    
    # Start with a
    mov rax, rdi
    
    # -2*b
    lea rbx, [rsi + rsi]
    neg rbx
    add rax, rbx          # a - 2b
    
    # +3*c
    lea rbx, [rdx + rdx*2]
    add rax, rbx          # a - 2b + 3c
    
    # -4*d
    shl rcx, 2
    neg rcx               # -4d
    add rax, rcx          # ... - 4d
    
    # +5*e
    lea rbx, [r8 + r8*4]
    add rax, rbx          # ... + 5e
    
    # -6*f
    lea rbx, [r9 + r9*2]  # 3f
    shl rbx, 1            # 6f
    neg rbx               # -6f
    add rax, rbx          # ... - 6f
    
    # +7*g (from stack)
    mov rbx, qword ptr [rsp+8]
    lea rbx, [rbx*2 + rbx*4 + rbx] # 7g is tricky with lea? lea can do x+x*2+x*4 but regs are limited.
