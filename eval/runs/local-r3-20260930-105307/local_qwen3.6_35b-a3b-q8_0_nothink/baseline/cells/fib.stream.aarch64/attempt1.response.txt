fib:
// Compute fib(n) iteratively using 64-bit integer arithmetic
// n is in x0, result in x0 on return
// Uses x1 for current fib, x2 for previous fib, x3 as loop counter
// Callee-saved registers not used, so no save needed

	cmp	x0, #1
	b.le	.Ldone_zero
	mov	x1, #0		// prev = 0 (fib(0))
	mov	x2, #1		// curr = 1 (fib(1))
	sub	x3, x0, #1	// loop count = n - 1
.Lloop:
	add	x4, x1, x2	// next = prev + curr
	mov	x1, x2		// prev = curr
	mov	x2, x4		// curr = next
	sub	x3, x3, #1
	cbnz	x3, .Lloop
.Ldone_zero:
	cmp	x0, #0
	b.eq	.Lreturn_0
.Lreturn_1:
	mov	x0, x2
	ret
.Lreturn_0:
	mov	x0, #0
	ret
