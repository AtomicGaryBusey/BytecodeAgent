dot_i32_simd:
	// Prologue: save frame and link register. sp is 16-byte aligned on entry.
	stp	x29, x30, [sp, #-16]!
	mov	x29, sp

	// Save callee-saved registers if needed. We will use x19-x28 for pointers/counters.
	stp	x19, x20, [sp, #-16]!
	stp	x21, x22, [sp, #-16]!
	stp	x23, x24, [sp, #-16]!

	// Initialize loop state.
	// x19 = a pointer, x20 = b pointer, x21 = remaining elements (n).
	mov	x19, x0
	mov	x20, x1
	mov	x21, x2

	// Handle small n where scalar is faster or simpler to avoid branch overhead.
	cmp	x21, #8
	b.lt	.L_scalar_only

	// Main SIMD loop. Process 8 elements per iteration (2x smull pairs).
	// We accumulate into v0-v1 and v2-v3 as int64x2 vectors.
	// Clear accumulators.
	movi	v0, #0
	mov
