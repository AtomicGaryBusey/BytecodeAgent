dot_i32_simd:
	cmp x2, #4
	b.lt .L_scalar_tail
	// Process in blocks of 8 elements (two 128-bit NEON vectors loaded per pair)
	// We use smulh/smlal or similar. AArch64 has SMLAL for signed multiply accumulate to double width.
	// v0, v1 will hold accumulators (int64x2_t).
	// Load 4 int32s from a into v4, 4 from b into v5 -> vmull_s32 -> add to acc.
	// Repeat for next 4. Total 8 per loop iteration.
	mov x19, x0 // a pointer
	mov x20, x1 // b pointer
	sub x21, x2, #7 // remaining count if not divisible by 8, or handle remainder later
	and x23, x2, #7 // remainder elements

	// Prologue for alignment: sp must be 16-byte aligned.
	stp x29, x30, [sp, #-32]!
	add x29, sp, #16

	movi v0, #0 // acc_a (low part if we used q registers, but let's use d pairs or q)
	mov
