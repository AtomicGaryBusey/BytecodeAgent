collatz_max:
    push rbp
    mov rbp, rsp
    sub rsp, 8

    xor rax, rax          # max_steps = 0
    xor rcx, rcx          # k = 1

.Louter_loop:
    cmp rcx, rdi
    jg .Lreturn           # if k > n, return

    mov rdx, rcx          # x = k
    xor r8, r8            # steps = 0

    # Handle k=1 case separately to avoid infinite loop logic
    test rdx, rdx
    je .Lcheck_max        # should not happen given constraint k>=1
    
    # Special case: if input is 1, collatz sequence is empty (0 steps)
    cmp rcx, 1
    je .Lcheck_max

.Linner_loop:
    test rdx, 1           # check if x is even
    jz .Leven_case

.Lodd_case:
    lea rax, [rdx + rdx*2] # rax = 3*x
    add rax, 1            # rax = 3*x + 1
    mov rdx, rax          # x = 3*x + 1
    inc r8                # steps++
    cmp rdx, 1
    jne .Linner_loop
    jmp .Lcheck_max

.Leven_case:
    shr rdx, 1            # x = x / 2
    inc r8                # steps++
    cmp rdx, 1
    jne .Linner_loop

.Lcheck_max:
    cmp rax, r8           # compare current max with new steps
    cmovl rax, r8         # if new steps > max, update max
    
    inc rcx               # k++
    jmp .Louter_loop

.Lreturn:
    mov rsp, rbp
    pop rbp
    ret
