collatz_max:
    push rbx
    mov rbx, rdi
    xor rax, rax          # max_steps = 0
    test rbx, rbx
    jle .Lreturn

    xor rcx, rcx          # k = 0

.Louter_loop:
    inc rcx               # k++
    cmp rcx, rbx
    jg .Lreturn           # if k > n, return

    mov rdx, rcx          # x = k
    xor r8, r8            # steps = 0

    test rdx, rdx
    je .Lcheck_max        # k=1 -> 0 steps, but handled by logic below? No, k>=1. If k=1, loop condition x!=1 fails immediately.

.Linner_loop:
    cmp rdx, 1
    je .Lsteps_done       # if x == 1, done

    test rdx, 1           # check if odd
    jnz .Lodd_case

    shr rdx, 1            # x = x / 2
    inc r8                # steps++
    jmp .Linner_loop

.Lodd_case:
    lea rax, [rdx + rdx*2] # rax = 3*x
    add rax, 1            # rax = 3*x + 1
    mov rdx, rax          # x = 3*x + 1
    inc r8                # steps++
    jmp .Linner_loop

.Lsteps_done:
    cmp rax, r8           # compare current max (in rax) with new steps (in r8)
    cmovl rax, r8         # if new steps > max, update max
    
    jmp .Louter_loop

.Lreturn:
    pop rbx
    ret
