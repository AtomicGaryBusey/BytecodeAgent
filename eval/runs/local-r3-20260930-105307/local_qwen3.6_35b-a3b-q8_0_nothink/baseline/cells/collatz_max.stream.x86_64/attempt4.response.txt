collatz_max:
    push rbx
    sub rsp, 8
    mov rbx, rdi          # n in rbx
    xor rax, rax          # max_steps = 0
    cmp rbx, 1
    jle .Lreturn
    
    xor rcx, rcx          # k = 0

.Louter_loop:
    inc rcx               # k++
    cmp rcx, rbx
    ja .Lreturn           # if k > n, return

    mov rdx, rcx          # x = k
    xor r8, r8            # steps = 0

.Linner_loop:
    cmp rdx, 1
    je .Lsteps_done       # if x == 1, done

    test rdx, 1           # check if odd
    jnz .Lodd_case

    shr rdx, 1            # x = x / 2
    inc r8                # steps++
    jmp .Linner_loop

.Lodd_case:
    lea r9, [rdx + rdx*2] # r9 = 3*x
    add r9, 1             # r9 = 3*x + 1
    mov rdx, r9           # x = 3*x + 1
    inc r8                # steps++
    jmp .Linner_loop

.Lsteps_done:
    cmp rax, r8           # compare current max (in rax) with new steps (in r8)
    cmovl rax, r8         # if new steps > max, update max
    
    jmp .Louter_loop

.Lreturn:
    add rsp, 8
    pop rbx
    ret
