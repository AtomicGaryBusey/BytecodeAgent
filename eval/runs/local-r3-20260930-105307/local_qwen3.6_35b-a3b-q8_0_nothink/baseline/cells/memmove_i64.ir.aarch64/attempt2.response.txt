export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero = const l 0
	brz %n, @done

	%cmp = csltl %dst, %src
	jnz %cmp, @forward
	br @backward

@forward
	%i = const l 0
	br @check_fwd

@inc_fwd
	%i = phi %i_0, %i_next
	%i_0 = const l 0
	%i_next = addl %i, 1
	check_fwd:
	%cond = csltl %i, %n
	brz %cond, @done

	%off = mull %i, 8
	%src_p = addl %src, %off
	%val = loadl [%src_p]

	%dst_p = addl %dst, %off
	storel %val, [%dst_p]

	br @inc_fwd

@backward
	%i = subli %n, 1
	br @check_bwd

@dec_bwd
	%i = phi %i_0, %i_next
	%i_0 = const l 0 ; Dummy initial value for phi, overwritten by control flow before first use check effectively via structure, but QBE requires valid initial.
	                 ; Correct logic: Start with n-1. Loop condition checks >= 0. Decrement at end.
	                 ; Let's restructure to avoid complex phi initialization issues if possible, 
	                 ; or just set dummy and rely on block entry values being undefined? No, QBE is strict.
	                 ; We will use a phi node with two predecessors: @backward (entry) and @dec_bwd (loop back).

	%i_0 = subli %n, 1
	%i_next = subli %i, 1

	check_bwd:
	%cond = csgei %i, 0
	brz %cond, @done

	%off = mull %i, 8
	%src_p = addl %src, %off
	%val = loadl [%src_p]

	%dst_p = addl %dst, %off
	storel %val, [%dst_p]

	br @dec_bwd

@done
	ret 0
}
