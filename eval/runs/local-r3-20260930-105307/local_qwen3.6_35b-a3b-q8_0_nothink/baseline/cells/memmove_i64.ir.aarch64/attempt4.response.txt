export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	brz %n, @done

	%cmp = csltl %dst, %src
	jnz %cmp, @forward
	br @backward

@forward
	%i = phi 0, %i_next
	%i_0 = 0
	%i_next = addl %i, 1
check_fwd:
	%cond = csltl %i, %n
	brz %cond, @done

	%off = mull %i, 8
	%src_p = addl %src, %off
	%val = loadl [%src_p]

	%dst_p = addl %dst, %off
	storel %val, [%dst_p]

	br @check_fwd

@backward
	%i = phi -1, %i_next
	%i_0 = subli %n, 1
	%i_next = subli %i, 1
check_bwd:
	%cond = csgei %i, 0
	brz %cond, @done

	%off = mull %i, 8
	%src_p = addl %src, %off
	%val = loadl [%src_p]

	%dst_p = addl %dst, %off
	storel %val, [%dst_p]

	br @check_bwd

@done
	ret 0
}
