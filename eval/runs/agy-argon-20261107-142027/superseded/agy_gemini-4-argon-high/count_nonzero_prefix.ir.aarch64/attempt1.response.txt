export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%pos =w csgtl %n, 0
	jnz %pos, @loop, @ret0
@ret0
	ret 0
@loop
	%i =l phi @start 0, @next %i1
	%p =l phi @start %a, @next %p1
	%v =l loadl %p
	%isz =w ceql %v, 0
	jnz %isz, @done, @next
@next
	%i1 =l add %i, 1
	%p1 =l add %p, 8
	%more =w csltl %i1, %n
	jnz %more, @loop, @done_n
@done
	ret %i
@done_n
	ret %i1
}
