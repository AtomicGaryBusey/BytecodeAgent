export function l $bubble_sort(l %a, l %n) {
@start
	%trivial =w cslel %n, 1
	jnz %trivial, @done, @init
@init
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 8
	%end0 =l add %a, %off
	jmp @outer
@outer
	%end =l phi @init %end0, @outer_latch %end_next
	jmp @inner
@inner
	%p =l phi @outer %a, @inner_latch %p_next
	%swapped =w phi @outer 0, @inner_latch %swapped_next
	%p_next =l add %p, 8
	%x =l loadl %p
	%y =l loadl %p_next
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @inner_latch
@swap
	storel %y, %p
	storel %x, %p_next
	jmp @inner_latch
@inner_latch
	%swapped_next =w phi @inner %swapped, @swap 1
	%more_inner =w cnel %p_next, %end
	jnz %more_inner, @inner, @outer_latch
@outer_latch
	%end_next =l sub %end, 8
	%more_outer =w cnel %end_next, %a
	%cont =w and %swapped_next, %more_outer
	jnz %cont, @outer, @done
@done
	ret 0
}
