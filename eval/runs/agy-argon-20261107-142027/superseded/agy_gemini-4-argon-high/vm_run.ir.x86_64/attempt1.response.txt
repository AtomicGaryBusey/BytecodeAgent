export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	%clr_i =l copy 0
	jmp @loop

@loop
	%at_end =w csgel %pc, %len
	jnz %at_end, @halt, @fetch

@fetch
	%pc_off =l shl %pc, 3
	%op_ptr =l add %code, %pc_off
	%op =l loadl %op_ptr
	%lt6 =w csltl %op, 6
	jnz %lt6, @op_0_5, @op_6_11

@op_0_5
	%lt3 =w csltl %op, 3
	jnz %lt3, @op_0_2, @op_3_5

@op_0_2
	%is0 =w ceql %op, 0
	jnz %is0, @halt, @op_1_2

@op_1_2
	%is1 =w ceql %op, 1
	jnz %is1, @do_push, @do_add

@op_3_5
	%is3 =w ceql %op, 3
	jnz %is3, @do_sub, @op_4_5

@op_4_5
	%is4 =w ceql %op, 4
	jnz %is4, @do_mul, @do_dup

@op_6_11
	%lt9 =w csltl %op, 9
	jnz %lt9, @op_6_8, @op_9_11

@op_6_8
	%is6 =w ceql %op, 6
	jnz %is6, @do_swap, @op_7_8

@op_7_8
	%is7 =w ceql %op, 7
	jnz %is7, @do_drop, @do_over

@op_9_11
	%is9 =w ceql %op, 9
	jnz %is9, @do_lt, @op_10_11

@op_10_11
	%is10 =w ceql %op, 10
	jnz %is10, @do_jz, @do_jmp

@do_push
	%push_pc1 =l add %pc, 1
	%push_coff =l shl %push_pc1, 3
	%push_cptr =l add %code, %push_coff
	%push_val =l loadl %push_cptr
	%push_soff =l shl %sp, 3
	%push_sptr =l add %stack, %push_soff
	storel %push_val, %push_sptr
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop

@do_add
	%add_sp1 =l sub %sp, 1
	%add_sp2 =l sub %sp, 2
	%add_boff =l shl %add_sp1, 3
	%add_bptr =l add %stack, %add_boff
	%add_b =l loadl %add_bptr
	%add_aoff =l shl %add_sp2, 3
	%add_aptr =l add %stack, %add_aoff
	%add_a =l loadl %add_aptr
	%add_res =l add %add_a, %add_b
	storel %add_res, %add_aptr
	%sp =l copy %add_sp1
	%pc =l add %pc, 1
	jmp @loop

@do_sub
	%sub_sp1 =l sub %sp, 1
	%sub_sp2 =l sub %sp, 2
	%sub_boff =l shl %sub_sp1, 3
	%sub_bptr =l add %stack, %sub_boff
	%sub_b =l loadl %sub_bptr
	%sub_aoff =l shl %sub_sp2, 3
	%sub_aptr =l add %stack, %sub_aoff
	%sub_a =l loadl %sub_aptr
	%sub_res =l sub %sub_a, %sub_b
	storel %sub_res, %sub_aptr
	%sp =l copy %sub_sp1
	%pc =l add %pc, 1
	jmp @loop

@do_mul
	%mul_sp1 =l sub %sp, 1
	%mul_sp2 =l sub %sp, 2
	%mul_boff =l shl %mul_sp1, 3
	%mul_bptr =l add %stack, %mul_boff
	%mul_b =l loadl %mul_bptr
	%mul_aoff =l shl %mul_sp2, 3
	%mul_aptr =l add %stack, %mul_aoff
	%mul_a =l loadl %mul_aptr
	%mul_res =l mul %mul_a, %mul_b
	storel %mul_res, %mul_aptr
	%sp =l copy %mul_sp1
	%pc =l add %pc, 1
	jmp @loop

@do_dup
	%dup_sp1 =l sub %sp, 1
	%dup_toff =l shl %dup_sp1, 3
	%dup_tptr =l add %stack, %dup_toff
	%dup_val =l loadl %dup_tptr
	%dup_noff =l shl %sp, 3
	%dup_nptr =l add %stack, %dup_noff
	storel %dup_val, %dup_nptr
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop

@do_swap
	%swp_sp1 =l sub %sp, 1
	%swp_sp2 =l sub %sp, 2
	%swp_boff =l shl %swp_sp1, 3
	%swp_bptr =l add %stack, %swp_boff
	%swp_b =l loadl %swp_bptr
	%swp_aoff =l shl %swp_sp2, 3
	%swp_aptr =l add %stack, %swp_aoff
	%swp_a =l loadl %swp_aptr
	storel %swp_b, %swp_aptr
	storel %swp_a, %swp_bptr
	%pc =l add %pc, 1
	jmp @loop

@do_drop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop

@do_over
	%ovr_sp2 =l sub %sp, 2
	%ovr_aoff =l shl %ovr_sp2, 3
	%ovr_aptr =l add %stack, %ovr_aoff
	%ovr_a =l loadl %ovr_aptr
	%ovr_noff =l shl %sp, 3
	%ovr_nptr =l add %stack, %ovr_noff
	storel %ovr_a, %ovr_nptr
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop

@do_lt
	%lt_sp1 =l sub %sp, 1
	%lt_sp2 =l sub %sp, 2
	%lt_boff =l shl %lt_sp1, 3
	%lt_bptr =l add %stack, %lt_boff
	%lt_b =l loadl %lt_bptr
	%lt_aoff =l shl %lt_sp2, 3
	%lt_aptr =l add %stack, %lt_aoff
	%lt_a =l loadl %lt_aptr
	%lt_cmp =w csltl %lt_a, %lt_b
	%lt_res =l extuw %lt_cmp
	storel %lt_res, %lt_aptr
	%sp =l copy %lt_sp1
	%pc =l add %pc, 1
	jmp @loop

@do_jz
	%jz_sp1 =l sub %sp, 1
	%jz_voff =l shl %jz_sp1, 3
	%jz_vptr =l add %stack, %jz_voff
	%jz_v =l loadl %jz_vptr
	%sp =l copy %jz_sp1
	%jz_cond =w ceql %jz_v, 0
	jnz %jz_cond, @jz_taken, @jz_ntaken

@jz_taken
	%jz_pc1 =l add %pc, 1
	%jz_toff =l shl %jz_pc1, 3
	%jz_tptr =l add %code, %jz_toff
	%pc =l loadl %jz_tptr
	jmp @loop

@jz_ntaken
	%pc =l add %pc, 2
	jmp @loop

@do_jmp
	%jmp_pc1 =l add %pc, 1
	%jmp_toff =l shl %jmp_pc1, 3
	%jmp_tptr =l add %code, %jmp_toff
	%pc =l loadl %jmp_tptr
	jmp @loop

@halt
	%clr_i =l copy %sp
	jmp @clr_check

@clr_check
	%clr_more =w csltl %clr_i, %cap
	jnz %clr_more, @clr_body, @done

@clr_body
	%clr_off =l shl %clr_i, 3
	%clr_ptr =l add %stack, %clr_off
	%clr_zero =l copy 0
	storel %clr_zero, %clr_ptr
	%clr_i =l add %clr_i, 1
	jmp @clr_check

@done
	%empty =w ceql %sp, 0
	jnz %empty, @ret_zero, @ret_top

@ret_zero
	%ret_z =l copy 0
	ret %ret_z

@ret_top
	%top_idx =l sub %sp, 1
	%top_off =l shl %top_idx, 3
	%top_ptr =l add %stack, %top_off
	%top_val =l loadl %top_ptr
	ret %top_val
}
