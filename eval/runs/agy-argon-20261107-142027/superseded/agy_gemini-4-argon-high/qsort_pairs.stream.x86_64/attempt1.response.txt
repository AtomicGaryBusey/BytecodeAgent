qsort_pairs:
    cmp rsi, 1
    jle .Lret_fast
    push rbp
    push rbx
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
.Lloop:
    lea r10, [rbp - 1]
    mov rax, rbx
    mov rcx, r10
    shr rcx, 1
    shl rcx, 4
    add rcx, rbx
    mov rdx, r10
    shl rdx, 4
    add rdx, rbx
    mov r8, qword ptr [rax]
    cmp r8, qword ptr [rcx]
    jg .Lswap1
    jl .Ldone1
    mov r8, qword ptr [rax + 8]
    cmp r8, qword ptr [rcx + 8]
    jle .Ldone1
.Lswap1:
    xchg rax, rcx
.Ldone1:
    mov r8, qword ptr [rcx]
    cmp r8, qword ptr [rdx]
    jg .Lswap2
    jl .Lmed_done
    mov r8, qword ptr [rcx + 8]
    cmp r8, qword ptr [rdx + 8]
    jle .Lmed_done
.Lswap2:
    xchg rcx, rdx
    mov r8, qword ptr [rax]
    cmp r8, qword ptr [rcx]
    jg .Luse_rax
    jl .Lmed_done
    mov r8, qword ptr [rax + 8]
    cmp r8, qword ptr [rcx + 8]
    jle .Lmed_done
.Luse_rax:
    mov rcx, rax
.Lmed_done:
    mov r8, qword ptr [rcx]
    mov r9, qword ptr [rcx + 8]
    lea rcx, [rbx - 16]
    mov rdx, rbp
    shl rdx, 4
    add rdx, rbx
.Lscan_i:
    add rcx, 16
    cmp qword ptr [rcx], r8
    jl .Lscan_i
    jg .Lscan_j
    cmp qword ptr [rcx + 8], r9
    jl .Lscan_i
.Lscan_j:
    sub rdx, 16
    cmp qword ptr [rdx], r8
    jg .Lscan_j
    jl .Lcheck_cross
    cmp qword ptr [rdx + 8], r9
    jg .Lscan_j
.Lcheck_cross:
    cmp rcx, rdx
    jae .Lpart_done
    mov rax, qword ptr [rcx]
    mov rsi, qword ptr [rcx + 8]
    mov rdi, qword ptr [rdx]
    mov r10, qword ptr [rdx + 8]
    mov qword ptr [rcx], rdi
    mov qword ptr [rcx + 8], r10
    mov qword ptr [rdx], rax
    mov qword ptr [rdx + 8], rsi
    jmp .Lscan_i
.Lpart_done:
    lea r10, [rdx + 16]
    mov rax, r10
    sub rax, rbx
    shr rax, 4
    mov rcx, rbp
    sub rcx, rax
    cmp rax, rcx
    jge .Lrecurse_right
    mov rdi, rbx
    mov rsi, rax
    mov rbx, r10
    mov rbp, rcx
    jmp .Ldo_call
.Lrecurse_right:
    mov rdi, r10
    mov rsi, rcx
    mov rbp, rax
.Ldo_call:
    call qsort_pairs
    cmp rbp, 1
    jg .Lloop
    add rsp, 8
    pop rbx
    pop rbp
.Lret_fast:
    xor eax, eax
    ret
