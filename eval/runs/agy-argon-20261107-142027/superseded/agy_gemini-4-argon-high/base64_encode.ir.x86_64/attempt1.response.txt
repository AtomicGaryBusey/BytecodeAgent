export function l $base64_encode(l %src_arg, l %n_arg, l %out_arg) {
@start
	jmp @loop

@loop
	%src =l phi @start %src_arg, @group_done %src_next
	%n =l phi @start %n_arg, @group_done %n_next
	%out =l phi @start %out_arg, @group_done %out_next
	%done =w cslel %n, 0
	jnz %done, @exit, @body

@body
	%b0 =l loadub %src
	%acc0 =l shl %b0, 16
	%gt1 =w csgtl %n, 1
	jnz %gt1, @has2, @encode

@has2
	%p1 =l add %src, 1
	%b1 =l loadub %p1
	%sh1 =l shl %b1, 8
	%acc1 =l or %acc0, %sh1
	%gt2 =w csgtl %n, 2
	jnz %gt2, @has3, @encode

@has3
	%p2 =l add %src, 2
	%b2 =l loadub %p2
	%acc2 =l or %acc1, %b2
	jmp @encode

@encode
	%acc =l phi @body %acc0, @has2 %acc1, @has3 %acc2
	jmp @char_loop

@char_loop
	%i =l phi @encode 0, @char_next %i_next
	%sh =w phi @encode 18, @char_next %sh_next
	%is_pad =w csgtl %i, %n
	jnz %is_pad, @write_pad, @write_b64

@write_pad
	%ch_pad =l copy 61
	jmp @char_next

@write_b64
	%tmp =l shr %acc, %sh
	%v =l and %tmp, 63
	%t1 =w csgtl %v, 25
	%b1_ext =l extuw %t1
	%o1 =l mul %b1_ext, 6
	%t2 =w csgtl %v, 51
	%b2_ext =l extuw %t2
	%o2 =l mul %b2_ext, 75
	%t3 =w csgtl %v, 61
	%b3_ext =l extuw %t3
	%o3 =l mul %b3_ext, 15
	%t4 =w csgtl %v, 62
	%b4_ext =l extuw %t4
	%o4 =l mul %b4_ext, 3
	%c0 =l add %v, 65
	%c1 =l add %c0, %o1
	%c2 =l sub %c1, %o2
	%c3 =l sub %c2, %o3
	%ch_b64 =l add %c3, %o4
	jmp @char_next

@char_next
	%ch =l phi @write_pad %ch_pad, @write_b64 %ch_b64
	%dst =l add %out, %i
	storeb %ch, %dst
	%i_next =l add %i, 1
	%sh_next =w sub %sh, 6
	%more_chars =w csltl %i_next, 4
	jnz %more_chars, @char_loop, @group_done

@group_done
	%src_next =l add %src, 3
	%n_next =l sub %n, 3
	%out_next =l add %out, 4
	jmp @loop

@exit
	%ret =l sub %out, %out_arg
	ret %ret
}
