export function l $fib(l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @base, @loop
@base
	%le0 =w cslel %n, 0
	jnz %le0, @zero, @one
@zero
	ret 0
@one
	ret 1
@loop
	%k =l phi @start %n, @loop %k1
	%a =l phi @start 0, @loop %b
	%b =l phi @start 1, @loop %next
	%next =l add %a, %b
	%k1 =l sub %k, 1
	%cond =w csgtl %k1, 1
	jnz %cond, @loop, @done
@done
	ret %next
}
