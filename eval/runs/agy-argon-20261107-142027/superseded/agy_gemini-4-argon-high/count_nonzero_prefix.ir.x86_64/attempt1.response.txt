export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%z0 =l copy 0
	%has =w csltl %z0, %n
	jnz %has, @loop, @done
@loop
	%i =l phi @start %z0, @step %i1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%z =w ceql %val, 0
	jnz %z, @done, @step
@step
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @done
@done
	%res =l phi @start %z0, @loop %i, @step %i1
	ret %res
}
