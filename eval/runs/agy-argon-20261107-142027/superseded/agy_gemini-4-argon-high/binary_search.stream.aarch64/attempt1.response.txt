binary_search:
    cmp     x1, #0
    b.le    .Lnot_found
    mov     x3, #0
.Lloop:
    sub     x4, x1, x3
    add     x4, x3, x4, lsr #1
    ldr     x5, [x0, x4, lsl #3]
    cmp     x5, x2
    b.eq    .Lfound
    csinc   x3, x3, x4, gt
    csel    x1, x4, x1, gt
    cmp     x3, x1
    b.lt    .Lloop
.Lnot_found:
    mov     x0, #-1
    ret
.Lfound:
    mov     x0, x4
    ret
