export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
	%more0 =w csltl %i0, %n
	jnz %more0, @loop, @done
@loop
	%m =l phi @start %m0, @next %m1
	%i =l phi @start %i0, @next %i1
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%lt =w csltl %m, %val
	jnz %lt, @update, @next
@update
	jmp @next
@next
	%m1 =l phi @update %val, @loop %m
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @done
@done
	%res =l phi @start %m0, @next %m1
	ret %res
}
