export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%res =l phi @start 0, @loop %res1
	%bit =l phi @start 2147483648, @loop %bit1
	%cand =l or %res, %bit
	%sq =l mul %cand, %cand
	%le =w culel %sq, %n
	%le64 =l extuw %le
	%mask =l sub 0, %le64
	%pick =l and %bit, %mask
	%res1 =l or %res, %pick
	%bit1 =l shr %bit, 1
	%cont =w cnel %bit1, 0
	jnz %cont, @loop, @done
@done
	ret %res1
}
