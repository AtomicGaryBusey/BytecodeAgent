export function l $reverse(l %a, l %n) {
@start
	%skip =w csltl %n, 2
	jnz %skip, @done, @init
@init
	%nm1 =l sub %n, 1
	%off =l shl %nm1, 3
	%hi0 =l add %a, %off
	jmp @loop
@loop
	%lo =l phi @init %a, @loop %lo1
	%hi =l phi @init %hi0, @loop %hi1
	%x =l loadl %lo
	%y =l loadl %hi
	storel %y, %lo
	storel %x, %hi
	%lo1 =l add %lo, 8
	%hi1 =l sub %hi, 8
	%more =w cultl %lo1, %hi1
	jnz %more, @loop, @done
@done
	ret 0
}
