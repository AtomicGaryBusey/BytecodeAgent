export function l $crc32(l %p, l %n) {
@start
	jmp @byte_loop
@byte_loop
	%i =l phi @start 0, @byte_next %i_next
	%crc =w phi @start -1, @byte_next %crc_step
	%cond =w csltl %i, %n
	jnz %cond, @byte_body, @done
@byte_body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc0 =w xor %crc, %b
	jmp @bit_loop
@bit_loop
	%rem =w phi @byte_body 8, @bit_loop %rem_next
	%crc_cur =w phi @byte_body %crc0, @bit_loop %crc_step
	%lsb =w and %crc_cur, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -306674912
	%shifted =w shr %crc_cur, 1
	%crc_step =w xor %shifted, %poly
	%rem_next =w sub %rem, 1
	jnz %rem_next, @bit_loop, @byte_next
@byte_next
	%i_next =l add %i, 1
	jmp @byte_loop
@done
	%res32 =w xor %crc, -1
	%res =l extuw %res32
	ret %res
}
