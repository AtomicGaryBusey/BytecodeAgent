isqrt:
    xor eax, eax
    test rdi, rdi
    jz .Ldone
    bsr rcx, rdi
    and ecx, -2
    mov esi, 1
    shl rsi, cl
.Lloop:
    lea rdx, [rax + rsi]
    shr rax, 1
    mov rcx, rdi
    sub rcx, rdx
    lea rdx, [rax + rsi]
    cmovae rdi, rcx
    cmovae rax, rdx
    shr rsi, 2
    jnz .Lloop
.Ldone:
    ret
