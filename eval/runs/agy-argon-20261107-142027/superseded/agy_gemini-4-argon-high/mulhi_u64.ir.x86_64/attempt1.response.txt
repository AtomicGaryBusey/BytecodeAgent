export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%p0 =l mul %al, %bl
	%p1 =l mul %ah, %bl
	%p2 =l mul %al, %bh
	%p3 =l mul %ah, %bh
	%c0 =l shr %p0, 32
	%t =l add %p1, %c0
	%tl =l and %t, 4294967295
	%th =l shr %t, 32
	%u =l add %p2, %tl
	%uh =l shr %u, 32
	%hi0 =l add %p3, %th
	%hi =l add %hi0, %uh
	ret %hi
}
