export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %cbytes =l shl %len, 3
    %cend =l add %code, %cbytes
    %pc =l copy %code
    %sp =l copy %stack
    jmp @loop

@loop
    %at_end =w cugel %pc, %cend
    jnz %at_end, @done, @disp

@disp
    %op =l loadl %pc
    %clt6 =w csltl %op, 6
    jnz %clt6, @op_0_5, @op_6_11

@op_0_5
    %clt3 =w csltl %op, 3
    jnz %clt3, @op_0_2, @op_3_5

@op_0_2
    %ceq1 =w ceql %op, 1
    jnz %ceq1, @op_push, @op_0_or_2

@op_0_or_2
    %ceq0 =w ceql %op, 0
    jnz %ceq0, @done, @op_add

@op_3_5
    %ceq4 =w ceql %op, 4
    jnz %ceq4, @op_mul, @op_3_or_5

@op_3_or_5
    %ceq3 =w ceql %op, 3
    jnz %ceq3, @op_sub, @op_dup

@op_6_11
    %clt9 =w csltl %op, 9
    jnz %clt9, @op_6_8, @op_9_11

@op_6_8
    %ceq7 =w ceql %op, 7
    jnz %ceq7, @op_drop, @op_6_or_8

@op_6_or_8
    %ceq6 =w ceql %op, 6
    jnz %ceq6, @op_swap, @op_over

@op_9_11
    %ceq10 =w ceql %op, 10
    jnz %ceq10, @op_jz, @op_9_or_11

@op_9_or_11
    %ceq9 =w ceql %op, 9
    jnz %ceq9, @op_lt, @op_jmp

@op_push
    %psh_ap =l add %pc, 8
    %psh_v =l loadl %psh_ap
    storel %psh_v, %sp
    %sp =l add %sp, 8
    %pc =l add %pc, 16
    jmp @loop

@op_add
    %add_p1 =l sub %sp, 8
    %add_p2 =l sub %sp, 16
    %add_b =l loadl %add_p1
    %add_a =l loadl %add_p2
    %add_r =l add %add_a, %add_b
    storel %add_r, %add_p2
    %sp =l copy %add_p1
    %pc =l add %pc, 8
    jmp @loop

@op_sub
    %sub_p1 =l sub %sp, 8
    %sub_p2 =l sub %sp, 16
    %sub_b =l loadl %sub_p1
    %sub_a =l loadl %sub_p2
    %sub_r =l sub %sub_a, %sub_b
    storel %sub_r, %sub_p2
    %sp =l copy %sub_p1
    %pc =l add %pc, 8
    jmp @loop

@op_mul
    %mul_p1 =l sub %sp, 8
    %mul_p2 =l sub %sp, 16
    %mul_b =l loadl %mul_p1
    %mul_a =l loadl %mul_p2
    %mul_r =l mul %mul_a, %mul_b
    storel %mul_r, %mul_p2
    %sp =l copy %mul_p1
    %pc =l add %pc, 8
    jmp @loop

@op_dup
    %dup_p1 =l sub %sp, 8
    %dup_v =l loadl %dup_p1
    storel %dup_v, %sp
    %sp =l add %sp, 8
    %pc =l add %pc, 8
    jmp @loop

@op_swap
    %swp_p1 =l sub %sp, 8
    %swp_p2 =l sub %sp, 16
    %swp_b =l loadl %swp_p1
    %swp_a =l loadl %swp_p2
    storel %swp_a, %swp_p1
    storel %swp_b, %swp_p2
    %pc =l add %pc, 8
    jmp @loop

@op_drop
    %sp =l sub %sp, 8
    %pc =l add %pc, 8
    jmp @loop

@op_over
    %ovr_p2 =l sub %sp, 16
    %ovr_a =l loadl %ovr_p2
    storel %ovr_a, %sp
    %sp =l add %sp, 8
    %pc =l add %pc, 8
    jmp @loop

@op_lt
    %lt_p1 =l sub %sp, 8
    %lt_p2 =l sub %sp, 16
    %lt_b =l loadl %lt_p1
    %lt_a =l loadl %lt_p2
    %lt_cmp =w csltl %lt_a, %lt_b
    %lt_r =l extuw %lt_cmp
    storel %lt_r, %lt_p2
    %sp =l copy %lt_p1
    %pc =l add %pc, 8
    jmp @loop

@op_jz
    %jz_ap =l add %pc, 8
    %jz_t =l loadl %jz_ap
    %sp =l sub %sp, 8
    %jz_v =l loadl %sp
    %jz_z =w ceql %jz_v, 0
    jnz %jz_z, @jz_taken, @jz_ntaken

@jz_taken
    %jz_off =l shl %jz_t, 3
    %pc =l add %code, %jz_off
    jmp @loop

@jz_ntaken
    %pc =l add %pc, 16
    jmp @loop

@op_jmp
    %jmp_ap =l add %pc, 8
    %jmp_t =l loadl %jmp_ap
    %jmp_off =l shl %jmp_t, 3
    %pc =l add %code, %jmp_off
    jmp @loop

@done
    %retval =l copy 0
    %empty =w ceql %sp, %stack
    jnz %empty, @clr_init, @nonempty

@nonempty
    %top_p =l sub %sp, 8
    %retval =l loadl %top_p
    jmp @clr_init

@clr_init
    %sbytes =l shl %cap, 3
    %send =l add %stack, %sbytes
    %clr =l copy %sp
    jmp @clr_cond

@clr_cond
    %more =w cultl %clr, %send
    jnz %more, @clr_body, @exit

@clr_body
    storel 0, %clr
    %clr =l add %clr, 8
    jmp @clr_cond

@exit
    ret %retval
}
