export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%le0 =w cslel %n, 0
	jnz %le0, @done, @check_same
@check_same
	%same =w ceql %dst, %src
	jnz %same, @done, @check_dir
@check_dir
	%is_fwd =w cultl %dst, %src
	jnz %is_fwd, @fwd_init, @bwd_init
@fwd_init
	jmp @fwd_loop
@fwd_loop
	%fi =l phi @fwd_init 0, @fwd_loop %fi_next
	%foff =l shl %fi, 3
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fval =l loadl %fsp
	storel %fval, %fdp
	%fi_next =l add %fi, 1
	%fcont =w csltl %fi_next, %n
	jnz %fcont, @fwd_loop, @done
@bwd_init
	%bi0 =l sub %n, 1
	jmp @bwd_loop
@bwd_loop
	%bi =l phi @bwd_init %bi0, @bwd_loop %bi_next
	%boff =l shl %bi, 3
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bval =l loadl %bsp
	storel %bval, %bdp
	%bi_next =l sub %bi, 1
	%bcont =w csgel %bi_next, 0
	jnz %bcont, @bwd_loop, @done
@done
	ret 0
}
