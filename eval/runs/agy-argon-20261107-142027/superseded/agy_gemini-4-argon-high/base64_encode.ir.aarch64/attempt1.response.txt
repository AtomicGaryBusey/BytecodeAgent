export function l $base64_encode(l %src, l %n, l %out) {
@entry
	jmp @outer_loop

@outer_loop
	%i =l phi @entry 0, @inner_done %i_next
	%out_idx =l phi @entry 0, @inner_done %out_idx_next
	%cond_outer =w csltl %i, %n
	jnz %cond_outer, @outer_body, @done

@outer_body
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%cond_b1 =w csgtl %rem, 1
	jnz %cond_b1, @load_b1, @after_b1

@load_b1
	%p1 =l add %p0, 1
	%b1_loaded =l loadub %p1
	jmp @after_b1

@after_b1
	%b1 =l phi @outer_body 0, @load_b1 %b1_loaded
	%cond_b2 =w csgtl %rem, 2
	jnz %cond_b2, @load_b2, @after_b2

@load_b2
	%p2 =l add %p0, 2
	%b2_loaded =l loadub %p2
	jmp @after_b2

@after_b2
	%b2 =l phi @after_b1 0, @load_b2 %b2_loaded
	%t0 =l shl %b0, 16
	%t1 =l shl %b1, 8
	%t01 =l or %t0, %t1
	%val =l or %t01, %b2
	jmp @inner_loop

@inner_loop
	%k =l phi @after_b2 0, @inner_next %k_next
	%cur_val =l phi @after_b2 %val, @inner_next %next_val
	%cond_pad =w csgtl %k, %rem
	jnz %cond_pad, @emit_pad, @emit_b64

@emit_pad
	jmp @inner_next

@emit_b64
	%shifted =l shr %cur_val, 18
	%v =l and %shifted, 63
	%c0 =l add %v, 65
	%ge26_w =w csgtl %v, 25
	%ge26 =l extuw %ge26_w
	%d26 =l mul %ge26, 6
	%c1 =l add %c0, %d26
	%ge52_w =w csgtl %v, 51
	%ge52 =l extuw %ge52_w
	%d52 =l mul %ge52, 75
	%c2 =l sub %c1, %d52
	%ge62_w =w csgtl %v, 61
	%ge62 =l extuw %ge62_w
	%d62 =l mul %ge62, 15
	%c3 =l sub %c2, %d62
	%ge63_w =w csgtl %v, 62
	%ge63 =l extuw %ge63_w
	%d63 =l mul %ge63, 3
	%c_b64 =l add %c3, %d63
	jmp @inner_next

@inner_next
	%c =l phi @emit_pad 61, @emit_b64 %c_b64
	%out_pos =l add %out_idx, %k
	%out_ptr =l add %out, %out_pos
	storeb %c, %out_ptr
	%next_val =l shl %cur_val, 6
	%k_next =l add %k, 1
	%cond_inner =w csltl %k_next, 4
	jnz %cond_inner, @inner_loop, @inner_done

@inner_done
	%i_next =l add %i, 3
	%out_idx_next =l add %out_idx, 4
	jmp @outer_loop

@done
	ret %out_idx
}
