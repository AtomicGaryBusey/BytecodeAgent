export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @go_left %lo_left, @go_right %lo_next
	%hi =l phi @start %n, @go_left %mid, @go_right %hi_right
	%cont =w csltl %lo, %hi
	jnz %cont, @body, @not_found
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check_lt
@check_lt
	%lt =w csltl %val, %key
	jnz %lt, @go_right, @go_left
@go_right
	%lo_next =l add %mid, 1
	%hi_right =l copy %hi
	jmp @loop
@go_left
	%lo_left =l copy %lo
	jmp @loop
@found
	ret %mid
@not_found
	ret -1
}
