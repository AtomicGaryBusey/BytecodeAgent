export function l $isqrt(l %n) {
@start
	%hi0 =l shl 1, 32
	jmp @loop
@loop
	%lo =l phi @start 0, @next %nlo
	%hi =l phi @start %hi0, @next %nhi
	%lo1 =l add %lo, 1
	%cont =w cultl %lo1, %hi
	jnz %cont, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @take_lo, @take_hi
@take_lo
	%lo_a =l copy %mid
	%hi_a =l copy %hi
	jmp @next
@take_hi
	%lo_b =l copy %lo
	%hi_b =l copy %mid
	jmp @next
@next
	%nlo =l phi @take_lo %lo_a, @take_hi %lo_b
	%nhi =l phi @take_lo %hi_a, @take_hi %hi_b
	jmp @loop
@done
	ret %lo
}
