export function l $bubble_sort(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @done, @init
@init
	%nm1 =l sub %n, 1
	jmp @outer_hdr
@outer_hdr
	%i =l phi @init 0, @outer_inc %i_next
	%cond_i =w csltl %i, %nm1
	jnz %cond_i, @outer_body, @done
@outer_body
	%limit =l sub %nm1, %i
	jmp @inner_hdr
@inner_hdr
	%j =l phi @outer_body 0, @inner_inc %j_next
	%cond_j =w csltl %j, %limit
	jnz %cond_j, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%do_swap =w csltl %v1, %v0
	jnz %do_swap, @swap, @inner_inc
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner_hdr
@outer_inc
	%i_next =l add %i, 1
	jmp @outer_hdr
@done
	ret 0
}
