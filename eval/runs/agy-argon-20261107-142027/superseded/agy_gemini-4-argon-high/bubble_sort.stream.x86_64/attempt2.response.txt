bubble_sort:
    cmp rsi, 1
    jle .Ldone
.Louter:
    dec rsi
    xor ecx, ecx
    xor r8d, r8d
.Linner:
    mov rdx, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp rdx, r9
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], rdx
    mov r8d, 1
.Lnoswap:
    inc rcx
    cmp rcx, rsi
    jl .Linner
    test r8d, r8d
    jz .Ldone
    cmp rsi, 1
    jg .Louter
.Ldone:
    xor eax, eax
    ret
