bmax:
    mov rax, qword ptr [rdi]
    sub rsi, 1
    jle .Ldone
    mov rcx, rax
    cmp rsi, 4
    jl .Ltail
.Lloop4:
    mov rdx, qword ptr [rdi + 8]
    mov r8, qword ptr [rdi + 16]
    cmp rax, rdx
    cmovl rax, rdx
    cmp rcx, r8
    cmovl rcx, r8
    mov rdx, qword ptr [rdi + 24]
    mov r8, qword ptr [rdi + 32]
    cmp rax, rdx
    cmovl rax, rdx
    cmp rcx, r8
    cmovl rcx, r8
    add rdi, 32
    sub rsi, 4
    cmp rsi, 4
    jge .Lloop4
    cmp rax, rcx
    cmovl rax, rcx
    test rsi, rsi
    jz .Ldone
.Ltail:
    mov rdx, qword ptr [rdi + rsi*8]
    cmp rax, rdx
    cmovl rax, rdx
    sub rsi, 1
    jg .Ltail
.Ldone:
    ret
