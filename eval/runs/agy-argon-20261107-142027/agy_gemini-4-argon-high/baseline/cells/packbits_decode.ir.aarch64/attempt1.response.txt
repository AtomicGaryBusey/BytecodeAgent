export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @main_loop

@main_loop
	%i =l phi @entry 0, @nop_next %i1, @lit_next %next_i, @rep_next %next_i_rep
	%written =l phi @entry 0, @nop_next %written_nop, @lit_next %new_written, @rep_next %new_written_rep
	%at_end =w csgel %i, %n
	jnz %at_end, @ret_ok, @read_hdr

@read_hdr
	%hdr_ptr =l add %src, %i
	%h_w =w loadsb %hdr_ptr
	%h =l extsw %h_w
	%i1 =l add %i, 1
	%is_nop =w ceql %h, -128
	jnz %is_nop, @nop_next, @check_type

@nop_next
	%written_nop =l copy %written
	jmp @main_loop

@check_type
	%is_neg =w csltl %h, 0
	jnz %is_neg, @repeat_packet, @literal_packet

@literal_packet
	%cnt =l add %h, 1
	%next_i =l add %i1, %cnt
	%src_err =w csgtl %next_i, %n
	jnz %src_err, @ret_err2, @lit_ok

@lit_ok
	%new_written =l add %written, %cnt
	%ovf =w csgtl %new_written, %cap
	jnz %ovf, @lit_ovf, @lit_no_ovf

@lit_ovf
	%k_ovf =l sub %cap, %written
	jmp @lit_copy_init

@lit_no_ovf
	%k_no_ovf =l copy %cnt
	jmp @lit_copy_init

@lit_copy_init
	%k =l phi @lit_ovf %k_ovf, @lit_no_ovf %k_no_ovf
	%src_p0 =l add %src, %i1
	%out_p0 =l add %out, %written
	jmp @lit_copy_cond

@lit_copy_cond
	%rem =l phi @lit_copy_init %k, @lit_copy_body %rem_next
	%sp =l phi @lit_copy_init %src_p0, @lit_copy_body %sp_next
	%op =l phi @lit_copy_init %out_p0, @lit_copy_body %op_next
	%done =w ceql %rem, 0
	jnz %done, @lit_copy_done, @lit_copy_body

@lit_copy_body
	%byte =w loadub %sp
	storeb %byte, %op
	%sp_next =l add %sp, 1
	%op_next =l add %op, 1
	%rem_next =l sub %rem, 1
	jmp @lit_copy_cond

@lit_copy_done
	jnz %ovf, @ret_err1, @lit_next

@lit_next
	jmp @main_loop

@repeat_packet
	%rep_src_err =w csgel %i1, %n
	jnz %rep_src_err, @ret_err2, @rep_ok

@rep_ok
	%rep_sp =l add %src, %i1
	%rbyte =w loadub %rep_sp
	%next_i_rep =l add %i1, 1
	%rcnt =l sub 1, %h
	%new_written_rep =l add %written, %rcnt
	%rovf =w csgtl %new_written_rep, %cap
	jnz %rovf, @rep_ovf, @rep_no_ovf

@rep_ovf
	%rk_ovf =l sub %cap, %written
	jmp @rep_fill_init

@rep_no_ovf
	%rk_no_ovf =l copy %rcnt
	jmp @rep_fill_init

@rep_fill_init
	%rk =l phi @rep_ovf %rk_ovf, @rep_no_ovf %rk_no_ovf
	%rout_p0 =l add %out, %written
	jmp @rep_fill_cond

@rep_fill_cond
	%rrem =l phi @rep_fill_init %rk, @rep_fill_body %rrem_next
	%rop =l phi @rep_fill_init %rout_p0, @rep_fill_body %rop_next
	%rdone =w ceql %rrem, 0
	jnz %rdone, @rep_fill_done, @rep_fill_body

@rep_fill_body
	storeb %rbyte, %rop
	%rop_next =l add %rop, 1
	%rrem_next =l sub %rrem, 1
	jmp @rep_fill_cond

@rep_fill_done
	jnz %rovf, @ret_err1, @rep_next

@rep_next
	jmp @main_loop

@ret_ok
	ret %written

@ret_err1
	ret -1

@ret_err2
	ret -2
}
