memmove_i64:
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    ja .Lbackward
    xor ecx, ecx
.Lforward:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lforward
    xor eax, eax
    ret
.Lbackward:
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov qword ptr [rdi + rdx*8], rax
    test rdx, rdx
    jg .Lbackward
.Ldone:
    xor eax, eax
    ret
