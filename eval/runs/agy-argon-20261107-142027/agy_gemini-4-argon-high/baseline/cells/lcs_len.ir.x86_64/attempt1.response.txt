export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %j0 =l copy 0
@init_loop
    %j =l phi @start %j0, @init_body %j_next
    %cmp_init =w cslel %j, %n
    jnz %cmp_init, @init_body, @outer_pre
@init_body
    %off_init =l mul %j, 8
    %ptr_init =l add %row, %off_init
    storel 0, %ptr_init
    %j_next =l add %j, 1
    jmp @init_loop
@outer_pre
    %i0 =l copy 0
@outer_loop
    %i =l phi @outer_pre %i0, @outer_next %i_next
    %cmp_outer =w csltl %i, %m
    jnz %cmp_outer, @outer_body, @done
@outer_body
    %xi_ptr =l add %x, %i
    %xi =w loadub %xi_ptr
    %k0 =l copy 1
    %diag0 =l copy 0
    %prev0 =l copy 0
@inner_loop
    %k =l phi @outer_body %k0, @inner_next %k_next
    %diag =l phi @outer_body %diag0, @inner_next %diag_next
    %prev =l phi @outer_body %prev0, @inner_next %prev_next
    %cmp_inner =w cslel %k, %n
    jnz %cmp_inner, @inner_body, @outer_next
@inner_body
    %km1 =l sub %k, 1
    %yk_ptr =l add %y, %km1
    %yk =w loadub %yk_ptr
    %k_off =l mul %k, 8
    %rk_ptr =l add %row, %k_off
    %up =l loadl %rk_ptr
    %eq =w ceqw %xi, %yk
    jnz %eq, @match, @nomatch
@match
    %cur_match =l add %diag, 1
    jmp @inner_next
@nomatch
    %cmp_max =w csltl %up, %prev
    jnz %cmp_max, @use_prev, @use_up
@use_prev
    %cur_prev =l copy %prev
    jmp @inner_next
@use_up
    %cur_up =l copy %up
    jmp @inner_next
@inner_next
    %cur =l phi @match %cur_match, @use_prev %cur_prev, @use_up %cur_up
    storel %cur, %rk_ptr
    %diag_next =l copy %up
    %prev_next =l copy %cur
    %k_next =l add %k, 1
    jmp @inner_loop
@outer_next
    %i_next =l add %i, 1
    jmp @outer_loop
@done
    %n_off =l mul %n, 8
    %rn_ptr =l add %row, %n_off
    %res =l loadl %rn_ptr
    ret %res
}
