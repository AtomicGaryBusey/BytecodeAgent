count_ugt:
    xor eax, eax
    cmp rsi, 4
    jl .Ltail_check
    pcmpeqd xmm2, xmm2
    psllq xmm2, 63
    movq xmm1, rdx
    punpcklqdq xmm1, xmm1
    pxor xmm1, xmm2
    pxor xmm3, xmm3
    pxor xmm4, xmm4
.Lloop4:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm5, xmmword ptr [rdi + 16]
    pxor xmm0, xmm2
    pxor xmm5, xmm2
    pcmpgtq xmm0, xmm1
    pcmpgtq xmm5, xmm1
    psubq xmm3, xmm0
    psubq xmm4, xmm5
    add rdi, 32
    sub rsi, 4
    cmp rsi, 4
    jge .Lloop4
    paddq xmm3, xmm4
    pshufd xmm0, xmm3, 78
    paddq xmm3, xmm0
    movq rax, xmm3
.Ltail_check:
    test rsi, rsi
    jz .Ldone
.Ltail:
    cmp rdx, qword ptr [rdi]
    adc rax, 0
    add rdi, 8
    dec rsi
    jnz .Ltail
.Ldone:
    ret
