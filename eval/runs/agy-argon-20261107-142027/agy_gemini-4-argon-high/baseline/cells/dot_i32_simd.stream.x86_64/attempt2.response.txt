dot_i32_simd:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    xor ecx, ecx
    cmp rdx, 4
    jl .Lscalar_loop
    mov r10, rdx
    and r10, -4
    pxor xmm0, xmm0
    pxor xmm5, xmm5
.Lsimd_loop:
    movdqu xmm1, xmmword ptr [rdi + rcx*4]
    movdqu xmm2, xmmword ptr [rsi + rcx*4]
    pshufd xmm3, xmm1, 0xf5
    pshufd xmm4, xmm2, 0xf5
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm0, xmm1
    paddq xmm5, xmm3
    add rcx, 4
    cmp rcx, r10
    jl .Lsimd_loop
    paddq xmm0, xmm5
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq rax, xmm0
    cmp rcx, rdx
    jge .Ldone
.Lscalar_loop:
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    cmp rcx, rdx
    jl .Lscalar_loop
.Ldone:
    ret
