dot_i32_simd:
    xor eax, eax
    cmp rdx, 4
    jl .Ltail
    pxor xmm0, xmm0
.Lvec_loop:
    pmovsxdq xmm1, qword ptr [rdi]
    pmovsxdq xmm2, qword ptr [rsi]
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    pmovsxdq xmm3, qword ptr [rdi + 8]
    pmovsxdq xmm4, qword ptr [rsi + 8]
    pmuldq xmm3, xmm4
    paddq xmm0, xmm3
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jge .Lvec_loop
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq rax, xmm0
.Ltail:
    test rdx, rdx
    jle .Ldone
.Lscalar_loop:
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Lscalar_loop
.Ldone:
    ret
