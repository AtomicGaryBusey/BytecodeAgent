bubble_sort:
    xor eax, eax
    dec rsi
    jle .Ldone
.Louter:
    xor ecx, ecx
.Linner:
    mov rdx, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rcx*8 + 8]
    cmp rdx, r8
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rcx*8 + 8], rdx
.Lnoswap:
    inc rcx
    cmp rcx, rsi
    jl .Linner
    dec rsi
    jg .Louter
.Ldone:
    ret
