    mvn w0, wzr
    cbz x1, .Ldone
    movz w3, #0x8320
    movk w3, #0xedb8, lsl #16
.Lbyte_loop:
    ldrb w4, [x2], #1
    eor w0, w0, w4
    mov w5, #8
.Lbit_loop:
    tst w0, #1
    lsr w0, w0, #1
    eor w4, w0, w3
    csel w0, w4, w0, ne
    subs w5, w5, #1
    b.ne .Lbit_loop
    subs x1, x1, #1
    b.ne .Lbyte_loop
.Ldone:
    mvn w0, w0
    ret
