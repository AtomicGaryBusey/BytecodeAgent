collatz_max:
    xor eax, eax
    mov r8, rdi
    shr r8, 1
    inc r8
.Louter:
    mov rdx, r8
    bsf rcx, rdx
    shr rdx, cl
    mov rsi, rcx
    cmp rdx, 1
    je .Lcheck_max
.Linner:
    lea rdx, [rdx + rdx*2 + 1]
    bsf rcx, rdx
    shr rdx, cl
    lea rsi, [rsi + rcx + 1]
    cmp rdx, 1
    jne .Linner
.Lcheck_max:
    cmp rsi, rax
    cmova rax, rsi
    inc r8
    cmp r8, rdi
    jle .Louter
    ret
