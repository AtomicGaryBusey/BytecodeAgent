export function l $collatz_max(l %n) {
@start
	%le1 =w cslel %n, 1
	jnz %le1, @ret_zero, @outer_init
@ret_zero
	ret 0
@outer_init
	jmp @outer_loop
@outer_loop
	%k =l phi @outer_init 2, @after_max %k_next
	%max =l phi @outer_init 0, @after_max %max_next
	jmp @inner_loop
@inner_loop
	%x =l phi @outer_loop %k, @inner_loop %x_next
	%steps =l phi @outer_loop 0, @inner_loop %steps_next
	%p =l and %x, 1
	%mask =l sub 0, %p
	%even =l shr %x, 1
	%x3 =l mul %x, 3
	%odd =l add %x3, 1
	%diff =l sub %odd, %even
	%masked =l and %diff, %mask
	%x_next =l add %even, %masked
	%steps_next =l add %steps, 1
	%more =w cnel %x_next, 1
	jnz %more, @inner_loop, @outer_next
@outer_next
	%gt =w csgtl %steps_next, %max
	jnz %gt, @update_max, @keep_max
@update_max
	jmp @after_max
@keep_max
	jmp @after_max
@after_max
	%max_next =l phi @update_max %steps_next, @keep_max %max
	%k_next =l add %k, 1
	%cont =w cslel %k_next, %n
	jnz %cont, @outer_loop, @exit
@exit
	ret %max_next
}
