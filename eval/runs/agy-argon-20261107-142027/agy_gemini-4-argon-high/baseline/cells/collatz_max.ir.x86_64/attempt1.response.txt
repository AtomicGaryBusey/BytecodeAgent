export function l $collatz_max(l %n) {
@start
	jmp @outer_header
@outer_header
	%k =l phi @start 1, @outer_latch %k_next
	%max =l phi @start 0, @outer_latch %max_next
	%cmp_k =w cslel %k, %n
	jnz %cmp_k, @inner_header, @done
@inner_header
	%x =l phi @outer_header %k, @inner_latch %x_next
	%steps =l phi @outer_header 0, @inner_latch %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @outer_check, @inner_body
@inner_body
	%bit =l and %x, 1
	%is_odd =w cnel %bit, 0
	jnz %is_odd, @odd, @even
@even
	%x_even =l shr %x, 1
	jmp @inner_latch
@odd
	%x3 =l mul %x, 3
	%x_odd =l add %x3, 1
	jmp @inner_latch
@inner_latch
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner_header
@outer_check
	%cmp_max =w csgtl %steps, %max
	jnz %cmp_max, @new_max, @outer_latch
@new_max
	jmp @outer_latch
@outer_latch
	%max_next =l phi @outer_check %max, @new_max %steps
	%k_next =l add %k, 1
	jmp @outer_header
@done
	ret %max
}
