max_byte:
    cmp rsi, 16
    jb .Llt16
    lea rdx, [rdi + rsi - 16]
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rdx]
    pmaxub xmm0, xmm1
    cmp rsi, 32
    jbe .Lreduce16
    movdqu xmm1, xmmword ptr [rdi + 16]
    movdqu xmm2, xmmword ptr [rdx - 16]
    pmaxub xmm1, xmm2
    pmaxub xmm0, xmm1
    cmp rsi, 64
    jbe .Lreduce16
    movdqu xmm1, xmmword ptr [rdi + 32]
    movdqu xmm2, xmmword ptr [rdi + 48]
    pmaxub xmm1, xmm2
    movdqu xmm2, xmmword ptr [rdx - 48]
    movdqu xmm3, xmmword ptr [rdx - 32]
    pmaxub xmm2, xmm3
    pmaxub xmm0, xmm1
    pmaxub xmm0, xmm2
    cmp rsi, 128
    jbe .Lreduce16
    movdqu xmm1, xmmword ptr [rdi + 64]
    movdqu xmm2, xmmword ptr [rdi + 80]
    pmaxub xmm1, xmm2
    movdqu xmm2, xmmword ptr [rdi + 96]
    movdqu xmm3, xmmword ptr [rdx - 80]
    pmaxub xmm2, xmm3
    movdqu xmm3, xmmword ptr [rdx - 64]
    pmaxub xmm1, xmm2
    pmaxub xmm0, xmm3
    pmaxub xmm0, xmm1
.Lreduce16:
    pshufd xmm1, xmm0, 0x4e
    pmaxub xmm0, xmm1
.Lreduce8:
    pshuflw xmm1, xmm0, 0x4e
    pmaxub xmm0, xmm1
.Lreduce4:
    pshuflw xmm1, xmm0, 0xb1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx edx, ah
    movzx eax, al
    cmp eax, edx
    cmovb eax, edx
    ret
.Llt16:
    cmp rsi, 8
    jae .L8to15
    cmp rsi, 4
    jae .L4to7
    test rsi, rsi
    jz .Lzero
    movzx eax, byte ptr [rdi]
    movzx edx, byte ptr [rdi + rsi - 1]
    shr rsi, 1
    movzx ecx, byte ptr [rdi + rsi]
    cmp eax, edx
    cmovb eax, edx
    cmp eax, ecx
    cmovb eax, ecx
    ret
.L8to15:
    movq xmm0, qword ptr [rdi]
    movq xmm1, qword ptr [rdi + rsi - 8]
    pmaxub xmm0, xmm1
    jmp .Lreduce8
.L4to7:
    movd xmm0, dword ptr [rdi]
    movd xmm1, dword ptr [rdi + rsi - 4]
    pmaxub xmm0, xmm1
    jmp .Lreduce4
.Lzero:
    xor eax, eax
    ret
