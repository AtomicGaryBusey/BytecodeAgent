export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret_zero, @init
@init
	%first =l loadl %a
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %i_next
	%cnt =l phi @init 0, @body %cnt_next
	%prev =l phi @init %first, @body %val
	%cond =w csltl %i, %limit
	jnz %cond, @body, @done
@body
	%i_next =l add %i, 1
	%off =l mul %i_next, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %prev, %val
	%inc =l extuw %gt
	%cnt_next =l add %cnt, %inc
	jmp @loop
@done
	ret %cnt
@ret_zero
	ret 0
}
