export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %nn2 =l mul %nn, 2
    jmp @init_i_loop

@init_i_loop
    %init_i =l phi @start 0, @init_i_next %init_i_inc
    %init_i_cond =w csltl %init_i, %n
    jnz %init_i_cond, @init_j_pre, @step_pre

@init_j_pre
    jmp @init_j_loop

@init_j_loop
    %init_j =l phi @init_j_pre 0, @init_body %init_j_inc
    %init_j_cond =w csltl %init_j, %n
    jnz %init_j_cond, @init_body, @init_i_next

@init_body
    %init_row_off =l mul %init_i, %n
    %init_idx =l add %init_row_off, %init_j
    %init_byte_off =l mul %init_idx, 8
    %init_ptr =l add %out, %init_byte_off
    %is_diag =w ceql %init_i, %init_j
    %diag_val =l extuw %is_diag
    storel %diag_val, %init_ptr
    %init_j_inc =l add %init_j, 1
    jmp @init_j_loop

@init_i_next
    %init_i_inc =l add %init_i, 1
    jmp @init_i_loop

@step_pre
    jmp @step_loop

@step_loop
    %step =l phi @step_pre 0, @step_next %step_inc
    %step_cond =w csltl %step, %k
    jnz %step_cond, @mul_i_pre, @clear_pre

@mul_i_pre
    jmp @mul_i_loop

@mul_i_loop
    %mul_i =l phi @mul_i_pre 0, @mul_i_next %mul_i_inc
    %mul_i_cond =w csltl %mul_i, %n
    jnz %mul_i_cond, @mul_j_pre, @copy_pre

@mul_j_pre
    jmp @mul_j_loop

@mul_j_loop
    %mul_j =l phi @mul_j_pre 0, @mul_j_next %mul_j_inc
    %mul_j_cond =w csltl %mul_j, %n
    jnz %mul_j_cond, @mul_t_pre, @mul_i_next

@mul_t_pre
    jmp @mul_t_loop

@mul_t_loop
    %mul_t =l phi @mul_t_pre 0, @mul_t_body %mul_t_inc
    %sum =l phi @mul_t_pre 0, @mul_t_body %sum_next
    %mul_t_cond =w csltl %mul_t, %n
    jnz %mul_t_cond, @mul_t_body, @mul_j_next

@mul_t_body
    %out_row_off =l mul %mul_i, %n
    %out_idx =l add %out_row_off, %mul_t
    %out_byte_off =l mul %out_idx, 8
    %out_ptr =l add %out, %out_byte_off
    %out_val =l loadl %out_ptr
    %a_row_off =l mul %mul_t, %n
    %a_idx =l add %a_row_off, %mul_j
    %a_byte_off =l mul %a_idx, 8
    %a_ptr =l add %a, %a_byte_off
    %a_val =l loadl %a_ptr
    %prod =l mul %out_val, %a_val
    %sum_next =l add %sum, %prod
    %mul_t_inc =l add %mul_t, 1
    jmp @mul_t_loop

@mul_j_next
    %tmp_row_off =l mul %mul_i, %n
    %tmp_idx =l add %tmp_row_off, %mul_j
    %tmp_byte_off =l mul %tmp_idx, 8
    %tmp_ptr =l add %tmp, %tmp_byte_off
    storel %sum, %tmp_ptr
    %mul_j_inc =l add %mul_j, 1
    jmp @mul_j_loop

@mul_i_next
    %mul_i_inc =l add %mul_i, 1
    jmp @mul_i_loop

@copy_pre
    jmp @copy_loop

@copy_loop
    %copy_idx =l phi @copy_pre 0, @copy_body %copy_inc
    %copy_cond =w csltl %copy_idx, %nn
    jnz %copy_cond, @copy_body, @step_next

@copy_body
    %copy_byte_off =l mul %copy_idx, 8
    %copy_src =l add %tmp, %copy_byte_off
    %copy_dst =l add %out, %copy_byte_off
    %copy_val =l loadl %copy_src
    storel %copy_val, %copy_dst
    %copy_inc =l add %copy_idx, 1
    jmp @copy_loop

@step_next
    %step_inc =l add %step, 1
    jmp @step_loop

@clear_pre
    jmp @clear_loop

@clear_loop
    %clear_idx =l phi @clear_pre 0, @clear_body %clear_inc
    %clear_cond =w csltl %clear_idx, %nn2
    jnz %clear_cond, @clear_body, @done

@clear_body
    %clear_byte_off =l mul %clear_idx, 8
    %clear_ptr =l add %tmp, %clear_byte_off
    storel 0, %clear_ptr
    %clear_inc =l add %clear_idx, 1
    jmp @clear_loop

@done
    ret 0
}
