export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @outer_cond

@outer_cond
	%i =l phi @start 0, @outer_next %i_next
	%out_len =l phi @start 0, @outer_next %out_len_next
	%cond =w csltl %i, %n
	jnz %cond, @outer_body, @done

@outer_body
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%has1 =w csgel %rem, 2
	jnz %has1, @load1, @merge1

@load1
	%i1 =l add %i, 1
	%p1 =l add %src, %i1
	%b1_loaded =l loadub %p1
	jmp @merge1

@merge1
	%b1 =l phi @outer_body 0, @load1 %b1_loaded
	%has2 =w csgel %rem, 3
	jnz %has2, @load2, @merge2

@load2
	%i2 =l add %i, 2
	%p2 =l add %src, %i2
	%b2_loaded =l loadub %p2
	jmp @merge2

@merge2
	%b2 =l phi @merge1 0, @load2 %b2_loaded
	%t0 =l shl %b0, 16
	%t1 =l shl %b1, 8
	%t01 =l or %t0, %t1
	%triple =l or %t01, %b2
	jmp @inner_body

@inner_body
	%k =l phi @merge2 0, @inner_next %k_next
	%sh =w phi @merge2 18, @inner_next %sh_next
	%is_pad =w csgtl %k, %rem
	jnz %is_pad, @store_char, @map_char

@map_char
	%shifted =l shr %triple, %sh
	%v =l and %shifted, 63
	%c0 =l add %v, 65
	%ge26 =w csgel %v, 26
	%ge26_l =l extuw %ge26
	%d1 =l mul %ge26_l, 6
	%c1 =l add %c0, %d1
	%ge52 =w csgel %v, 52
	%ge52_l =l extuw %ge52
	%d2 =l mul %ge52_l, 75
	%c2 =l sub %c1, %d2
	%ge62 =w csgel %v, 62
	%ge62_l =l extuw %ge62
	%d3 =l mul %ge62_l, 15
	%c3 =l sub %c2, %d3
	%ge63 =w csgel %v, 63
	%ge63_l =l extuw %ge63
	%d4 =l mul %ge63_l, 3
	%c_mapped =l add %c3, %d4
	jmp @store_char

@store_char
	%c =l phi @inner_body 61, @map_char %c_mapped
	%out_idx =l add %out_len, %k
	%out_ptr =l add %out, %out_idx
	storeb %c, %out_ptr
	jmp @inner_next

@inner_next
	%k_next =l add %k, 1
	%sh_next =w sub %sh, 6
	%inner_cond =w csltl %k_next, 4
	jnz %inner_cond, @inner_body, @outer_next

@outer_next
	%i_next =l add %i, 3
	%out_len_next =l add %out_len, 4
	jmp @outer_cond

@done
	ret %out_len
}
