    test rsi, rsi
    jle .Ldone
    mov r8d, 0xedb88320
.Lbyte:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit:

    shr eax, 1
    cmovc edx, r8d

    jnz .Lbit
    inc rdi
    dec rsi

    ret
