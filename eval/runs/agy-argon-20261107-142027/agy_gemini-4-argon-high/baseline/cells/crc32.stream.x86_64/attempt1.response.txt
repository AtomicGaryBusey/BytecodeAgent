crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
    mov r8d, 0xEDB88320
.Lbyte_loop:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit_loop:
    shr eax, 1
    sbb edx, edx
    and edx, r8d
    xor eax, edx
    dec ecx
    jnz .Lbit_loop
    inc rdi
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    not eax
    ret
