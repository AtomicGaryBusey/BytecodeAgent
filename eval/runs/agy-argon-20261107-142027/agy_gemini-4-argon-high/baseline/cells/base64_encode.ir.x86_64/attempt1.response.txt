export function l $base64_encode(l %src, l %n, l %out) {
@start
    %i =l copy 0
    %out_ptr =l copy %out
@outer_check
    %more =w csltl %i, %n
    jnz %more, @outer_body, @done
@outer_body
    %p0 =l add %src, %i
    %b0 =l loadub %p0
    %i1 =l add %i, 1
    %has1 =w csltl %i1, %n
    jnz %has1, @load_b1, @zero_b1
@load_b1
    %p1 =l add %src, %i1
    %b1 =l loadub %p1
    jmp @check_b2
@zero_b1
    %b1 =l copy 0
@check_b2
    %i2 =l add %i, 2
    %has2 =w csltl %i2, %n
    jnz %has2, @load_b2, @zero_b2
@load_b2
    %p2 =l add %src, %i2
    %b2 =l loadub %p2
    jmp @pack
@zero_b2
    %b2 =l copy 0
@pack
    %s0 =l shl %b0, 16
    %s1 =l shl %b1, 8
    %t0 =l or %s0, %s1
    %val =l or %t0, %b2
    %k =l copy 0
@inner_body
    %v0 =l shr %val, 18
    %v =l and %v0, 63
    %lt26 =w csltl %v, 26
    jnz %lt26, @map_az_upper, @check_52
@map_az_upper
    %c =l add %v, 65
    jmp @store_char
@check_52
    %lt52 =w csltl %v, 52
    jnz %lt52, @map_az_lower, @check_62
@map_az_lower
    %c =l add %v, 71
    jmp @store_char
@check_62
    %lt62 =w csltl %v, 62
    jnz %lt62, @map_09, @check_plus
@map_09
    %c =l sub %v, 4
    jmp @store_char
@check_plus
    %eq62 =w ceql %v, 62
    jnz %eq62, @map_plus, @map_slash
@map_plus
    %c =l copy 43
    jmp @store_char
@map_slash
    %c =l copy 47
@store_char
    %dst =l add %out_ptr, %k
    storeb %c, %dst
    %val =l shl %val, 6
    %k =l add %k, 1
    %kmore =w csltl %k, 4
    jnz %kmore, @inner_body, @pad_check
@pad_check
    jnz %has2, @outer_next, @pad_3
@pad_3
    %dst3 =l add %out_ptr, 3
    storeb 61, %dst3
    jnz %has1, @outer_next, @pad_2
@pad_2
    %dst2 =l add %out_ptr, 2
    storeb 61, %dst2
@outer_next
    %i =l add %i, 3
    %out_ptr =l add %out_ptr, 4
    jmp @outer_check
@done
    %written =l sub %out_ptr, %out
    ret %written
}
