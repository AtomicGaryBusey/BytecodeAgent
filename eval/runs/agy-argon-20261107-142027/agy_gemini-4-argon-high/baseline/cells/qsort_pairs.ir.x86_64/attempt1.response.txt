export function l $qsort_pairs(l %a, l %n) {
@start
	%base_cmp =w csltl %n, 2
	jnz %base_cmp, @ret_zero, @partition

@ret_zero
	ret 0

@partition
	%a0_p8 =l add %a, 8
	%ak =l loadl %a
	%av =l loadl %a0_p8

	%nm1 =l sub %n, 1
	%mid =l div %nm1, 2
	%mid_off =l mul %mid, 16
	%bm_p =l add %a, %mid_off
	%bm_p8 =l add %bm_p, 8
	%bk =l loadl %bm_p
	%bv =l loadl %bm_p8

	%last_off =l mul %nm1, 16
	%cl_p =l add %a, %last_off
	%cl_p8 =l add %cl_p, 8
	%ck =l loadl %cl_p
	%cv =l loadl %cl_p8

	%ab_klt =w csltl %ak, %bk
	%ab_keq =w ceql %ak, %bk
	%ab_vlt =w csltl %av, %bv
	%ab_ev =w and %ab_keq, %ab_vlt
	%a_lt_b =w or %ab_klt, %ab_ev
	jnz %a_lt_b, @med_ab, @med_ba

@med_ab
	%bc1_klt =w csltl %bk, %ck
	%bc1_keq =w ceql %bk, %ck
	%bc1_vlt =w csltl %bv, %cv
	%bc1_ev =w and %bc1_keq, %bc1_vlt
	%b_lt_c1 =w or %bc1_klt, %bc1_ev
	jnz %b_lt_c1, @piv_b, @med_ab_not_bc

@med_ab_not_bc
	%ac1_klt =w csltl %ak, %ck
	%ac1_keq =w ceql %ak, %ck
	%ac1_vlt =w csltl %av, %cv
	%ac1_ev =w and %ac1_keq, %ac1_vlt
	%a_lt_c1 =w or %ac1_klt, %ac1_ev
	jnz %a_lt_c1, @piv_c, @piv_a

@med_ba
	%ac2_klt =w csltl %ak, %ck
	%ac2_keq =w ceql %ak, %ck
	%ac2_vlt =w csltl %av, %cv
	%ac2_ev =w and %ac2_keq, %ac2_vlt
	%a_lt_c2 =w or %ac2_klt, %ac2_ev
	jnz %a_lt_c2, @piv_a, @med_ba_not_ac

@med_ba_not_ac
	%bc2_klt =w csltl %bk, %ck
	%bc2_keq =w ceql %bk, %ck
	%bc2_vlt =w csltl %bv, %cv
	%bc2_ev =w and %bc2_keq, %bc2_vlt
	%b_lt_c2 =w or %bc2_klt, %bc2_ev
	jnz %b_lt_c2, @piv_c, @piv_b

@piv_a
	jmp @piv_done

@piv_b
	jmp @piv_done

@piv_c
	jmp @piv_done

@piv_done
	%pk =l phi @piv_a %ak, @piv_b %bk, @piv_c %ck
	%pv =l phi @piv_a %av, @piv_b %bv, @piv_c %cv
	%i_init =l sub 0, 1
	%j_init =l copy %n
	jmp @outer_loop

@outer_loop
	%i_start =l phi @piv_done %i_init, @do_swap %i
	%j_start =l phi @piv_done %j_init, @do_swap %j
	jmp @i_loop

@i_loop
	%i_prev =l phi @outer_loop %i_start, @i_loop %i
	%i =l add %i_prev, 1
	%i_off =l mul %i, 16
	%ip =l add %a, %i_off
	%ip8 =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ip8
	%ik_lt =w csltl %ik, %pk
	%ik_eq =w ceql %ik, %pk
	%iv_lt =w csltl %iv, %pv
	%i_eq_vlt =w and %ik_eq, %iv_lt
	%i_lt_p =w or %ik_lt, %i_eq_vlt
	jnz %i_lt_p, @i_loop, @j_loop

@j_loop
	%j_prev =l phi @i_loop %j_start, @j_loop %j
	%j =l sub %j_prev, 1
	%j_off =l mul %j, 16
	%jp =l add %a, %j_off
	%jp8 =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jp8
	%pk_lt =w csltl %pk, %jk
	%pk_eq =w ceql %pk, %jk
	%pv_lt =w csltl %pv, %jv
	%p_eq_vlt =w and %pk_eq, %pv_lt
	%j_gt_p =w or %pk_lt, %p_eq_vlt
	jnz %j_gt_p, @j_loop, @check_cross

@check_cross
	%i_lt_j =w csltl %i, %j
	jnz %i_lt_j, @do_swap, @part_done

@do_swap
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	jmp @outer_loop

@part_done
	%left_n =l add %j, 1
	%right_n =l sub %n, %left_n
	%right_off =l mul %left_n, 16
	%right_p =l add %a, %right_off
	%r1 =l call $qsort_pairs(l %a, l %left_n)
	%r2 =l call $qsort_pairs(l %right_p, l %right_n)
	ret 0
}
