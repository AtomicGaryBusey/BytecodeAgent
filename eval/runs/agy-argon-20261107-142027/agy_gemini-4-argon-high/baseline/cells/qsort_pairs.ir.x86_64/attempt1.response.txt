export function l $qsort_pairs(l %a, l %n) {
@start
	%cmp_n =w cslel %n, 1
	jnz %cmp_n, @ret, @init

@init
	%mid =l shr %n, 1
	%mid_off =l shl %mid, 4
	%mid_ptr =l add %a, %mid_off
	%pk =l loadl %mid_ptr
	%mid_ptr_v =l add %mid_ptr, 8
	%pv =l loadl %mid_ptr_v
	%i =l copy 0
	%j =l sub %n, 1
	jmp @outer_check

@outer_check
	%cmp_ij =w cslel %i, %j
	jnz %cmp_ij, @scan_i, @recurse

@scan_i
	%i_off =l shl %i, 4
	%i_ptr =l add %a, %i_off
	%ik =l loadl %i_ptr
	%i_ptr_v =l add %i_ptr, 8
	%iv =l loadl %i_ptr_v
	%lt_k =w csltl %ik, %pk
	%eq_k =w ceql %ik, %pk
	%lt_v =w csltl %iv, %pv
	%eq_lt =w and %eq_k, %lt_v
	%is_lt =w or %lt_k, %eq_lt
	jnz %is_lt, @inc_i, @scan_j

@inc_i
	%i =l add %i, 1
	jmp @scan_i

@scan_j
	%j_off =l shl %j, 4
	%j_ptr =l add %a, %j_off
	%jk =l loadl %j_ptr
	%j_ptr_v =l add %j_ptr, 8
	%jv =l loadl %j_ptr_v
	%gt_k =w csgtl %jk, %pk
	%eq_k2 =w ceql %jk, %pk
	%gt_v =w csgtl %jv, %pv
	%eq_gt =w and %eq_k2, %gt_v
	%is_gt =w or %gt_k, %eq_gt
	jnz %is_gt, @dec_j, @check_swap

@dec_j
	%j =l sub %j, 1
	jmp @scan_j

@check_swap
	%do_swap =w cslel %i, %j
	jnz %do_swap, @swap, @recurse

@swap
	%need_write =w csltl %i, %j
	jnz %need_write, @do_write, @after_write

@do_write
	storel %jk, %i_ptr
	storel %jv, %i_ptr_v
	storel %ik, %j_ptr
	storel %iv, %j_ptr_v
	jmp @after_write

@after_write
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @outer_check

@recurse
	%left_n =l add %j, 1
	%r1 =l call $qsort_pairs(l %a, l %left_n)
	%right_off =l shl %i, 4
	%right_a =l add %a, %right_off
	%right_n =l sub %n, %i
	%r2 =l call $qsort_pairs(l %right_a, l %right_n)
	jmp @ret

@ret
	ret 0
}
