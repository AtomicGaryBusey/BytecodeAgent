export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @header_loop

@header_loop
    %src_pos =l phi @start 0, @nop_cont %src_pos_1, @lit_done %lit_src_next, @rep_done %rep_src_next
    %out_pos =l phi @start 0, @nop_cont %out_pos, @lit_done %lit_out_next, @rep_done %rep_out_next
    %done_cond =w csgel %src_pos, %n
    jnz %done_cond, @ret_done, @read_hdr

@read_hdr
    %hdr_ptr =l add %src, %src_pos
    %h_w =w loadsb %hdr_ptr
    %h =l extsw %h_w
    %src_pos_1 =l add %src_pos, 1
    %is_nop =w ceql %h, -128
    jnz %is_nop, @nop_cont, @check_type

@nop_cont
    jmp @header_loop

@check_type
    %is_lit =w csgel %h, 0
    jnz %is_lit, @literal, @repeat

@literal
    %lit_cnt =l add %h, 1
    %rem_src =l sub %n, %src_pos_1
    %lit_src_err =w csgtl %lit_cnt, %rem_src
    jnz %lit_src_err, @ret_err2, @lit_check_out

@lit_check_out
    %lit_rem_out =l sub %cap, %out_pos
    %lit_out_err =w csgtl %lit_cnt, %lit_rem_out
    jnz %lit_out_err, @lit_overflow_init, @lit_ok_init

@lit_overflow_init
    jmp @lit_loop_chk

@lit_ok_init
    jmp @lit_loop_chk

@lit_loop_chk
    %lit_i =l phi @lit_overflow_init 0, @lit_ok_init 0, @lit_loop_body %lit_i_next
    %lit_lim =l phi @lit_overflow_init %lit_rem_out, @lit_ok_init %lit_cnt, @lit_loop_body %lit_lim
    %lit_cmp =w csltl %lit_i, %lit_lim
    jnz %lit_cmp, @lit_loop_body, @lit_loop_end

@lit_loop_body
    %s_off =l add %src_pos_1, %lit_i
    %s_ptr =l add %src, %s_off
    %b =w loadub %s_ptr
    %d_off =l add %out_pos, %lit_i
    %d_ptr =l add %out, %d_off
    storeb %b, %d_ptr
    %lit_i_next =l add %lit_i, 1
    jmp @lit_loop_chk

@lit_loop_end
    jnz %lit_out_err, @ret_err1, @lit_done

@lit_done
    %lit_src_next =l add %src_pos_1, %lit_cnt
    %lit_out_next =l add %out_pos, %lit_cnt
    jmp @header_loop

@repeat
    %rep_src_err =w csgel %src_pos_1, %n
    jnz %rep_src_err, @ret_err2, @rep_check_out

@rep_check_out
    %rep_s_ptr =l add %src, %src_pos_1
    %rep_val =w loadub %rep_s_ptr
    %rep_src_next =l add %src_pos_1, 1
    %rep_cnt =l sub 1, %h
    %rep_rem_out =l sub %cap, %out_pos
    %rep_out_err =w csgtl %rep_cnt, %rep_rem_out
    jnz %rep_out_err, @rep_overflow_init, @rep_ok_init

@rep_overflow_init
    jmp @rep_loop_chk

@rep_ok_init
    jmp @rep_loop_chk

@rep_loop_chk
    %rep_i =l phi @rep_overflow_init 0, @rep_ok_init 0, @rep_loop_body %rep_i_next
    %rep_lim =l phi @rep_overflow_init %rep_rem_out, @rep_ok_init %rep_cnt, @rep_loop_body %rep_lim
    %rep_cmp =w csltl %rep_i, %rep_lim
    jnz %rep_cmp, @rep_loop_body, @rep_loop_end

@rep_loop_body
    %rep_d_off =l add %out_pos, %rep_i
    %rep_d_ptr =l add %out, %rep_d_off
    storeb %rep_val, %rep_d_ptr
    %rep_i_next =l add %rep_i, 1
    jmp @rep_loop_chk

@rep_loop_end
    jnz %rep_out_err, @ret_err1, @rep_done

@rep_done
    %rep_out_next =l add %out_pos, %rep_cnt
    jmp @header_loop

@ret_done
    ret %out_pos

@ret_err1
    ret -1

@ret_err2
    ret -2
}
