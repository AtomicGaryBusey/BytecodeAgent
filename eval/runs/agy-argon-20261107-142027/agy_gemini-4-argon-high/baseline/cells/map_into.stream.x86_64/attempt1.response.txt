map_into:
    test rdx, rdx
    jle .Lzero
    push r14
    push r13
    push r12
    push rbp
    push rbx
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    mov r13, rcx
    xor r14d, r14d
.Lloop:
    mov rdi, qword ptr [rbp]
    call r13
    mov qword ptr [rbx], rax
    xor edx, edx
    cmp rax, qword ptr [rbp]
    setl dl
    add r14, rdx
    add rbp, 8
    add rbx, 8
    dec r12
    jnz .Lloop
    mov rax, r14
    pop rbx
    pop rbp
    pop r12
    pop r13
    pop r14
    ret
.Lzero:
    xor eax, eax
    ret
