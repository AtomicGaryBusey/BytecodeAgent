export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%tot =l add %na, %nb
	jmp @zloop

@zloop
	%zk =l phi @start 0, @zbody %zk1
	%zcmp =w csltl %zk, %tot
	jnz %zcmp, @zbody, @zdone

@zbody
	%zoff =l shl %zk, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zk1 =l add %zk, 1
	jmp @zloop

@zdone
	%na0 =w ceql %na, 0
	jnz %na0, @ret0, @check_nb

@check_nb
	%nb0 =w ceql %nb, 0
	jnz %nb0, @ret0, @iloop

@ret0
	ret 0

@iloop
	%i =l phi @check_nb 0, @idone %i1
	%icmp =w csltl %i, %na
	jnz %icmp, @ibody, @norm

@ibody
	%ioff =l shl %i, 3
	%aptr =l add %a, %ioff
	%ai =l loadl %aptr
	%ai0 =l and %ai, 4294967295
	%ai1 =l shr %ai, 32
	jmp @jloop

@jloop
	%j =l phi @ibody 0, @jbody %j1
	%carry =l phi @ibody 0, @jbody %carry_next
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @jbody, @idone

@jbody
	%joff =l shl %j, 3
	%bptr =l add %b, %joff
	%bj =l loadl %bptr
	%bj0 =l and %bj, 4294967295
	%bj1 =l shr %bj, 32

	%p00 =l mul %ai0, %bj0
	%p01 =l mul %ai0, %bj1
	%p10 =l mul %ai1, %bj0
	%p11 =l mul %ai1, %bj1

	%p00_lo =l and %p00, 4294967295
	%p00_hi =l shr %p00, 32
	%p01_lo =l and %p01, 4294967295
	%p01_hi =l shr %p01, 32
	%p10_lo =l and %p10, 4294967295
	%p10_hi =l shr %p10, 32

	%mid1 =l add %p00_hi, %p01_lo
	%mid =l add %mid1, %p10_lo
	%mid_shl =l shl %mid, 32
	%lo =l add %p00_lo, %mid_shl

	%mid_hi =l shr %mid, 32
	%hi_a =l add %p11, %p01_hi
	%hi_b =l add %hi_a, %p10_hi
	%hi =l add %hi_b, %mid_hi

	%ij =l add %i, %j
	%ijoff =l shl %ij, 3
	%rptr =l add %r, %ijoff
	%rij =l loadl %rptr

	%s1 =l add %lo, %rij
	%c1w =w cultl %s1, %lo
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rptr

	%hi1 =l add %hi, %c1
	%carry_next =l add %hi1, %c2
	%j1 =l add %j, 1
	jmp @jloop

@idone
	%inb =l add %i, %nb
	%inboff =l shl %inb, 3
	%rcptr =l add %r, %inboff
	storel %carry, %rcptr
	%i1 =l add %i, 1
	jmp @iloop

@norm
	%k =l phi @iloop %tot, @ncheck %k1
	%kgt0 =w csgtl %k, 0
	jnz %kgt0, @ncheck, @ret0

@ncheck
	%k1 =l sub %k, 1
	%koff =l shl %k1, 3
	%rkptr =l add %r, %koff
	%rk =l loadl %rkptr
	%rkz =w ceql %rk, 0
	jnz %rkz, @norm, @retk

@retk
	ret %k
}
