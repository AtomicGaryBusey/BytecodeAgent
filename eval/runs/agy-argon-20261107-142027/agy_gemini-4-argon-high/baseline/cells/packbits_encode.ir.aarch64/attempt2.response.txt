export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%out_idx =l copy 0

@outer_loop
	%cond_outer =w csltl %i, %n
	jnz %cond_outer, @check_run, @done

@check_run
	%i_plus_2 =l add %i, 2
	%has_3 =w csltl %i_plus_2, %n
	jnz %has_3, @check_run_bytes, @start_literal

@check_run_bytes
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%p1 =l add %p0, 1
	%b1 =w loadub %p1
	%eq01 =w ceqw %b0, %b1
	jnz %eq01, @check_run_byte2, @start_literal

@check_run_byte2
	%p2 =l add %p0, 2
	%b2 =w loadub %p2
	%eq02 =w ceqw %b0, %b2
	jnz %eq02, @do_repeat, @start_literal

@do_repeat
	%r =l copy 3
	%rhdr =w copy 254

@repeat_len_loop
	%r_lt_128 =w csltl %r, 128
	jnz %r_lt_128, @repeat_check_bounds, @emit_repeat

@repeat_check_bounds
	%ir =l add %i, %r
	%ir_lt_n =w csltl %ir, %n
	jnz %ir_lt_n, @repeat_check_byte, @emit_repeat

@repeat_check_byte
	%pr =l add %src, %ir
	%br =w loadub %pr
	%eq_r =w ceqw %b0, %br
	jnz %eq_r, @repeat_inc, @emit_repeat

@repeat_inc
	%r =l add %r, 1
	%rhdr =w sub %rhdr, 1
	jmp @repeat_len_loop

@emit_repeat
	%next_out_r =l add %out_idx, 2
	%fits_r =w cslel %next_out_r, %cap
	jnz %fits_r, @write_repeat, @fail

@write_repeat
	%out_p0 =l add %out, %out_idx
	storeb %rhdr, %out_p0
	%out_p1 =l add %out_p0, 1
	storeb %b0, %out_p1
	%out_idx =l copy %next_out_r
	%i =l add %i, %r
	jmp @outer_loop

@start_literal
	%j =l add %i, 1
	%lhdr =w copy 0

@literal_scan_loop
	%j_lt_n =w csltl %j, %n
	jnz %j_lt_n, @literal_check_128, @emit_literal

@literal_check_128
	%cur_len =l sub %j, %i
	%len_lt_128 =w csltl %cur_len, 128
	jnz %len_lt_128, @literal_check_run3, @emit_literal

@literal_check_run3
	%j_plus_2 =l add %j, 2
	%j_has_3 =w csltl %j_plus_2, %n
	jnz %j_has_3, @literal_check_b01, @literal_inc

@literal_check_b01
	%pj0 =l add %src, %j
	%jb0 =w loadub %pj0
	%pj1 =l add %pj0, 1
	%jb1 =w loadub %pj1
	%jeq01 =w ceqw %jb0, %jb1
	jnz %jeq01, @literal_check_b02, @literal_inc

@literal_check_b02
	%pj2 =l add %pj0, 2
	%jb2 =w loadub %pj2
	%jeq02 =w ceqw %jb0, %jb2
	jnz %jeq02, @emit_literal, @literal_inc

@literal_inc
	%j =l add %j, 1
	%lhdr =w add %lhdr, 1
	jmp @literal_scan_loop

@emit_literal
	%L =l sub %j, %i
	%pkt_len =l add %L, 1
	%next_out_l =l add %out_idx, %pkt_len
	%fits_l =w cslel %next_out_l, %cap
	jnz %fits_l, @write_literal_hdr, @fail

@write_literal_hdr
	%out_lhdr_p =l add %out, %out_idx
	storeb %lhdr, %out_lhdr_p
	%out_idx =l add %out_idx, 1

@write_literal_loop
	%copy_more =w csltl %i, %j
	jnz %copy_more, @write_literal_byte, @outer_loop

@write_literal_byte
	%src_p =l add %src, %i
	%val =w loadub %src_p
	%dst_p =l add %out, %out_idx
	storeb %val, %dst_p
	%i =l add %i, 1
	%out_idx =l add %out_idx, 1
	jmp @write_literal_loop

@done
	ret %out_idx

@fail
	ret -1
}
