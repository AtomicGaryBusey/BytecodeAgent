export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%src_p =l add %src, %off
	%x =l loadl %src_p
	%y =l call %f(l %x)
	%dst_p =l add %dst, %off
	storel %y, %dst_p
	%lt =w csltl %y, %x
	%inc =l extuw %lt
	%cnt =l add %cnt, %inc
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
