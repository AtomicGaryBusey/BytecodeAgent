export function l $isqrt(l %n) {
@start
@loop
	%lo =l phi @start 0, @then %mid, @else %lo
	%hi =l phi @start 4294967295, @then %hi, @else %nhi
	%cond =w cultl %lo, %hi
	jnz %cond, @body, @done
@body
	%s1 =l add %lo, %hi
	%s2 =l add %s1, 1
	%mid =l shr %s2, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @then, @else
@then
	jmp @loop
@else
	%nhi =l sub %mid, 1
	jmp @loop
@done
	ret %lo
}
