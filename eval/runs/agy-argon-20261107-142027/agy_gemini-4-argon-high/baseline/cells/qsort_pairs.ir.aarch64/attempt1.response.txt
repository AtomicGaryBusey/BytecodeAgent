export function l $qsort_pairs(l %a0, l %n0) {
@start
	%done0 =w cslel %n0, 1
	jnz %done0, @ret, @part_init

@part_init
	%a =l phi @start %a0, @cont %next_a
	%n =l phi @start %n0, @cont %next_n
	%nm1 =l sub %n, 1
	%mid =l sar %nm1, 1
	%mid_off =l shl %mid, 4
	%mid_p =l add %a, %mid_off
	%mid_p8 =l add %mid_p, 8
	%pk =l loadl %mid_p
	%pv =l loadl %mid_p8
	jmp @outer

@outer
	%i_start =l phi @part_init 0, @swap %i_next
	%j_start =l phi @part_init %nm1, @swap %j_next
	jmp @iloop

@iloop
	%i =l phi @outer %i_start, @inext %i_inc
	%ioff =l shl %i, 4
	%pi =l add %a, %ioff
	%pi8 =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %pi8
	%klt =w csltl %ki, %pk
	%keq =w ceql %ki, %pk
	%vlt =w csltl %vi, %pv
	%eq_vlt =w and %keq, %vlt
	%ilt =w or %klt, %eq_vlt
	jnz %ilt, @inext, @jloop

@inext
	%i_inc =l add %i, 1
	jmp @iloop

@jloop
	%j =l phi @iloop %j_start, @jnext %j_dec
	%joff =l shl %j, 4
	%pj =l add %a, %joff
	%pj8 =l add %pj, 8
	%kj =l loadl %pj
	%vj =l loadl %pj8
	%kgt =w csgtl %kj, %pk
	%keq2 =w ceql %kj, %pk
	%vgt =w csgtl %vj, %pv
	%eq_vgt =w and %keq2, %vgt
	%jgt =w or %kgt, %eq_vgt
	jnz %jgt, @jnext, @check_cross

@jnext
	%j_dec =l sub %j, 1
	jmp @jloop

@check_cross
	%crossed =w csgel %i, %j
	jnz %crossed, @part_done, @swap

@swap
	%pi8_s =l add %pi, 8
	%pj8_s =l add %pj, 8
	storel %kj, %pi
	storel %vj, %pi8_s
	storel %ki, %pj
	storel %vi, %pj8_s
	%i_next =l add %i, 1
	%j_next =l sub %j, 1
	jmp @outer

@part_done
	%left_n =l add %j, 1
	%right_off =l shl %left_n, 4
	%right_a =l add %a, %right_off
	%right_n =l sub %n, %left_n
	%left_smaller =w csltl %left_n, %right_n
	jnz %left_smaller, @rec_left, @rec_right

@rec_left
	%r1 =l call $qsort_pairs(l %a, l %left_n)
	jmp @cont

@rec_right
	%r2 =l call $qsort_pairs(l %right_a, l %right_n)
	jmp @cont

@cont
	%next_a =l phi @rec_left %right_a, @rec_right %a
	%next_n =l phi @rec_left %right_n, @rec_right %left_n
	%done1 =w cslel %next_n, 1
	jnz %done1, @ret, @part_init

@ret
	ret 0
}
