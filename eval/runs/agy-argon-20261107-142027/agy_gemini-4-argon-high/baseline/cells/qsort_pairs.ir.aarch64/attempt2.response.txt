export function l $qsort_pairs(l %a, l %n) {
@start
	%cmp_n =w cslel %n, 1
	jnz %cmp_n, @done, @init
@init
	%mid =l sar %n, 1
	%mid_off =l shl %mid, 4
	%mid_ptr =l add %a, %mid_off
	%mid_ptr8 =l add %mid_ptr, 8
	%pk =l loadl %mid_ptr
	%pv =l loadl %mid_ptr8
	jmp @loop
@loop
	%lt =l phi @init 0, @do_lt %lt_next, @do_gt %lt, @do_eq %lt
	%i =l phi @init 0, @do_lt %i_next_lt, @do_gt %i, @do_eq %i_next_eq
	%gt =l phi @init %n, @do_lt %gt, @do_gt %gt_next, @do_eq %gt
	%cond =w csltl %i, %gt
	jnz %cond, @body, @recurse
@body
	%i_off =l shl %i, 4
	%i_ptr =l add %a, %i_off
	%i_ptr8 =l add %i_ptr, 8
	%ik =l loadl %i_ptr
	%iv =l loadl %i_ptr8
	%k_lt =w csltl %ik, %pk
	jnz %k_lt, @do_lt, @check_k_gt
@check_k_gt
	%k_gt =w csgtl %ik, %pk
	jnz %k_gt, @do_gt, @check_v_lt
@check_v_lt
	%v_lt =w csltl %iv, %pv
	jnz %v_lt, @do_lt, @check_v_gt
@check_v_gt
	%v_gt =w csgtl %iv, %pv
	jnz %v_gt, @do_gt, @do_eq
@do_lt
	%lt_off =l shl %lt, 4
	%lt_ptr =l add %a, %lt_off
	%lt_ptr8 =l add %lt_ptr, 8
	%ltk =l loadl %lt_ptr
	%ltv =l loadl %lt_ptr8
	storel %ik, %lt_ptr
	storel %iv, %lt_ptr8
	storel %ltk, %i_ptr
	storel %ltv, %i_ptr8
	%lt_next =l add %lt, 1
	%i_next_lt =l add %i, 1
	jmp @loop
@do_gt
	%gt_next =l sub %gt, 1
	%gt_off =l shl %gt_next, 4
	%gt_ptr =l add %a, %gt_off
	%gt_ptr8 =l add %gt_ptr, 8
	%gtk =l loadl %gt_ptr
	%gtv =l loadl %gt_ptr8
	storel %ik, %gt_ptr
	storel %iv, %gt_ptr8
	storel %gtk, %i_ptr
	storel %gtv, %i_ptr8
	jmp @loop
@do_eq
	%i_next_eq =l add %i, 1
	jmp @loop
@recurse
	%r1 =l call $qsort_pairs(l %a, l %lt)
	%r_off =l shl %gt, 4
	%r_ptr =l add %a, %r_off
	%r_len =l sub %n, %gt
	%r2 =l call $qsort_pairs(l %r_ptr, l %r_len)
@done
	ret 0
}
