export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%q0 =l shr %p00, 32
	%p10 =l mul %a1, %b0
	%t =l add %p10, %q0
	%t0 =l and %t, 4294967295
	%t1 =l shr %t, 32
	%p01 =l mul %a0, %b1
	%u =l add %p01, %t0
	%u1 =l shr %u, 32
	%p11 =l mul %a1, %b1
	%s =l add %p11, %t1
	%res =l add %s, %u1
	ret %res
}
