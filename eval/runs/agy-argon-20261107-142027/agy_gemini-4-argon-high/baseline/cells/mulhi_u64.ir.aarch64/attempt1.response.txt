export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a_lo =l and %a, %mask
	%a_hi =l shr %a, 32
	%b_lo =l and %b, %mask
	%b_hi =l shr %b, 32
	%p0 =l mul %a_lo, %b_lo
	%p1 =l mul %a_lo, %b_hi
	%p2 =l mul %a_hi, %b_lo
	%p3 =l mul %a_hi, %b_hi
	%p0_hi =l shr %p0, 32
	%p1_lo =l and %p1, %mask
	%p2_lo =l and %p2, %mask
	%t0 =l add %p0_hi, %p1_lo
	%mid =l add %t0, %p2_lo
	%mid_hi =l shr %mid, 32
	%p1_hi =l shr %p1, 32
	%p2_hi =l shr %p2, 32
	%h0 =l add %p3, %p1_hi
	%h1 =l add %h0, %p2_hi
	%res =l add %h1, %mid_hi
	ret %res
}
