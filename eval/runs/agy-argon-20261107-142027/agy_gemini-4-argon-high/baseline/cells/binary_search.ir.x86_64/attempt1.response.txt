export function l $binary_search(l %a, l %n, l %key) {
@start
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start 0, @left %lo2, @right %lo1
	%hi =l phi @start %hi0, @left %hi1, @right %hi2
	%cmp =w cslel %lo, %hi
	jnz %cmp, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l sar %sum, 1
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check_lt
@check_lt
	%lt =w csltl %val, %key
	jnz %lt, @right, @left
@right
	%lo1 =l add %mid, 1
	%hi2 =l copy %hi
	jmp @loop
@left
	%lo2 =l copy %lo
	%hi1 =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
