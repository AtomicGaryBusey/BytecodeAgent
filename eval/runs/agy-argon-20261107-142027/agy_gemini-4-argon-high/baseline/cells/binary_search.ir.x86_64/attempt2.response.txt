export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	jmp @loop
@loop
	%lo =l phi @start %lo0, @left %lo_l, @right %lo_r
	%hi =l phi @start %n, @left %hi_l, @right %hi_r
	%cont =w csltl %lo, %hi
	jnz %cont, @body, @not_found
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp_lt
@cmp_lt
	%lt =w csltl %val, %key
	jnz %lt, @right, @left
@left
	%lo_l =l copy %lo
	%hi_l =l copy %mid
	jmp @loop
@right
	%lo_r =l add %mid, 1
	%hi_r =l copy %hi
	jmp @loop
@found
	ret %mid
@not_found
	ret -1
}
