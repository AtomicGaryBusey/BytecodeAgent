dot_i32_simd:
    movi v2.2d, #0
    cmp x2, #4
    b.lt .Lvec_done
.Lvec_loop:
    ldr q0, [x0], #16
    ldr q1, [x1], #16
    smlal v2.2d, v0.2s, v1.2s
    smlal2 v2.2d, v0.4s, v1.4s
    sub x2, x2, #4
    cmp x2, #4
    b.ge .Lvec_loop
.Lvec_done:
    addp d2, v2.2d
    fmov x3, d2
    cmp x2, #0
    b.le .Ldone
.Lscalar_loop:
    ldr w4, [x0], #4
    ldr w5, [x1], #4
    smaddl x3, w4, w5, x3
    subs x2, x2, #1
    b.gt .Lscalar_loop
.Ldone:
    mov x0, x3
    ret
