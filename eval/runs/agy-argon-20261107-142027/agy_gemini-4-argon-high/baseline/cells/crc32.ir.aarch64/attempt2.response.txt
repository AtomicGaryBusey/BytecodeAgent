export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
@outer_cond
	%crc =w phi @start %crc0, @outer_next %crc_next
	%i =l phi @start %i0, @outer_next %i_next
	%cond =w csltl %i, %n
	jnz %cond, @outer_body, @done
@outer_body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc, %b
	%k0 =w copy 0
@inner_body
	%c =w phi @outer_body %crc1, @inner_next %c_next
	%k =w phi @outer_body %k0, @inner_next %k_next
	%bit =w and %c, 1
	%c_shr =w shr %c, 1
	jnz %bit, @is_odd, @is_even
@is_odd
	%c_odd =w xor %c_shr, 3988292384
	jmp @inner_next
@is_even
	%c_even =w copy %c_shr
	jmp @inner_next
@inner_next
	%c_next =w phi @is_odd %c_odd, @is_even %c_even
	%k_next =w add %k, 1
	%k_cond =w csltw %k_next, 8
	jnz %k_cond, @inner_body, @outer_next
@outer_next
	%crc_next =w copy %c_next
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	%res_w =w xor %crc, 4294967295
	%res =l extuw %res_w
	ret %res
}
