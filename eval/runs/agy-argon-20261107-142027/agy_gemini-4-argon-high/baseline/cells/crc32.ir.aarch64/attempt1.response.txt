export function l $crc32(l %p, l %n) {
@start
	%c =w copy 4294967295
	%i =l copy 0
	jmp @byteloop
@byteloop
	%more =w csltl %i, %n
	jnz %more, @bytebody, @done
@bytebody
	%addr =l add %p, %i
	%b =w loadub %addr
	%c =w xor %c, %b
	%bit =w copy 0
	jmp @bitloop
@bitloop
	%lsb =w and %c, 1
	%mask =w sub 0, %lsb
	%pmsk =w and %mask, 3988292384
	%sh =w shr %c, 1
	%c =w xor %sh, %pmsk
	%bit =w add %bit, 1
	%bitmore =w csltw %bit, 8
	jnz %bitmore, @bitloop, @nextbyte
@nextbyte
	%i =l add %i, 1
	jmp @byteloop
@done
	%c =w xor %c, 4294967295
	%res =l extuw %c
	ret %res
}
