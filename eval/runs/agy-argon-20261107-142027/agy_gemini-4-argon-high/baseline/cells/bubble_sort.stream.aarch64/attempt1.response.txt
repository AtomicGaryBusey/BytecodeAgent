bubble_sort:
    subs x1, x1, #1
    b.le .Ldone
.Louter:
    mov x2, x0
    mov x3, x1
.Linner:
    ldp x4, x5, [x2]
    cmp x4, x5
    b.le .Lnoswap
    stp x5, x4, [x2]
.Lnoswap:
    add x2, x2, #8
    subs x3, x3, #1
    b.gt .Linner
    subs x1, x1, #1
    b.gt .Louter
.Ldone:
    mov x0, #0
    ret
