bubble_sort:
    subs x1, x1, #1
    b.le .Ldone
.Louter:
    mov x2, #0
    mov x3, #0
    mov x4, x0
.Linner:
    ldp x5, x6, [x4]
    cmp x5, x6
    b.le .Lnoswap
    stp x6, x5, [x4]
    mov x2, x3
.Lnoswap:
    add x4, x4, #8
    add x3, x3, #1
    cmp x3, x1
    b.lt .Linner
    mov x1, x2
    cbnz x1, .Louter
.Ldone:
    mov x0, #0
    ret
