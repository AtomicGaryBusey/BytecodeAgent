export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%i =l copy 0
@outer_cond
	%c_out =w csltl %i, %nm1
	jnz %c_out, @outer_body, @done
@outer_body
	%j =l copy 0
	%limit =l sub %nm1, %i
@inner_cond
	%c_in =w csltl %j, %limit
	jnz %c_in, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%c_swap =w csltl %v1, %v0
	jnz %c_swap, @do_swap, @inner_inc
@do_swap
	storel %v1, %p0
	storel %v0, %p1
@inner_inc
	%j =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
