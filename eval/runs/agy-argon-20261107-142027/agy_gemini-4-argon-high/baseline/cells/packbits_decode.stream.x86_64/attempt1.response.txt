packbits_decode:
    mov r8, rsi
    mov rsi, rdi
    mov rdi, rdx
    mov r9, rdx
    mov r10, rcx

.Lloop:
    test r8, r8
    jz .Ldone
    movsx ecx, byte ptr [rsi]
    inc rsi
    dec r8
    test ecx, ecx
    js .Lnegative
    inc ecx
    cmp r8, rcx
    jl .Lerr_src
    sub r8, rcx
    cmp r10, rcx
    jl .Llit_overflow
    sub r10, rcx
    rep movsb
    jmp .Lloop

.Lnegative:
    cmp ecx, -128
    je .Lloop
    test r8, r8
    jz .Lerr_src
    movzx eax, byte ptr [rsi]
    inc rsi
    dec r8
    neg ecx
    inc ecx
    cmp r10, rcx
    jl .Lrep_overflow
    sub r10, rcx
    rep stosb
    jmp .Lloop

.Llit_overflow:
    mov rcx, r10
    rep movsb
    mov rax, -1
    ret

.Lrep_overflow:
    mov rcx, r10
    rep stosb
    mov rax, -1
    ret

.Ldone:
    mov rax, rdi
    sub rax, r9
    ret

.Lerr_src:
    mov rax, -2
    ret
