export function l $xxh64(l %p0, l %n, l %seed) {
@entry
	%P1 =l copy -7046029288634856825
	%P2 =l copy -4417276706812531889
	%P3 =l copy 1609587929392839161
	%P4 =l copy -8796714831421723037
	%P5 =l copy 2870177450012600261
	%q =l copy %p0
	%rem =l copy %n
	%cmp32 =w csltl %n, 32
	jnz %cmp32, @small, @large_init

@large_init

	%v1 =l add %t0, %P2
	%v2 =l add %seed, %P2
	%v3 =l copy %seed
	%v4 =l sub %seed, %P1
	jmp @stripe_loop

@stripe_loop
	%x1 =l loadl %q
	%m1 =l mul %x1, %P2
	%a1 =l add %v1, %m1
	%sl1 =l shl %a1, 31
	%sr1 =l shr %a1, 33
	%r1 =l or %sl1, %sr1
	%v1 =l mul %r1, %P1

	%q2 =l add %q, 8
	%x2 =l loadl %q2
	%m2 =l mul %x2, %P2
	%a2 =l add %v2, %m2
	%sl2 =l shl %a2, 31
	%sr2 =l shr %a2, 33
	%r2 =l or %sl2, %sr2
	%v2 =l mul %r2, %P1

	%q3 =l add %q, 16
	%x3 =l loadl %q3
	%m3 =l mul %x3, %P2

	%sl3 =l shl %a3, 31
	%sr3 =l shr %a3, 33
	%r3 =l or %sl3, %sr3
	%v3 =l mul %r3, %P1

	%q4 =l add %q, 24
	%x4 =l loadl %q4
	%m4 =l mul %x4, %P2
	%a4 =l add %v4, %m4
	%sl4 =l shl %a4, 31
	%sr4 =l shr %a4, 33
	%r4 =l or %sl4, %sr4
	%v4 =l mul %r4, %P1

	%q =l add %q, 32
	%rem =l sub %rem, 32
	%more32 =w csgel %rem, 32
	jnz %more32, @stripe_loop, @stripe_done

@stripe_done
	%v1sl =l shl %v1, 1
	%v1sr =l shr %v1, 63
	%v1r =l or %v1sl, %v1sr

	%v2sl =l shl %v2, 7
	%v2sr =l shr %v2, 57
	%v2r =l or %v2sl, %v2sr

	%v3sl =l shl %v3, 12
	%v3sr =l shr %v3, 52
	%v3r =l or %v3sl, %v3sr

	%v4sl =l shl %v4, 18
	%v4sr =l shr %v4, 46
	%v4r =l or %v4sl, %v4sr

	%s12 =l add %v1r, %v2r
	%s34 =l add %v3r, %v4r
	%h =l add %s12, %s34

	%mv1m =l mul %v1, %P2
	%mv1sl =l shl %mv1m, 31
	%mv1sr =l shr %mv1m, 33
	%mv1r =l or %mv1sl, %mv1sr
	%mv1p =l mul %mv1r, %P1
	%hx1 =l xor %h, %mv1p
	%hm1 =l mul %hx1, %P1
	%h =l add %hm1, %P4

	%mv2m =l mul %v2, %P2
	%mv2sl =l shl %mv2m, 31
	%mv2sr =l shr %mv2m, 33
	%mv2r =l or %mv2sl, %mv2sr
	%mv2p =l mul %mv2r, %P1
	%hx2 =l xor %h, %mv2p
	%hm2 =l mul %hx2, %P1
	%h =l add %hm2, %P4

	%mv3m =l mul %v3, %P2
	%mv3sl =l shl %mv3m, 31
	%mv3sr =l shr %mv3m, 33
	%mv3r =l or %mv3sl, %mv3sr
	%mv3p =l mul %mv3r, %P1
	%hx3 =l xor %h, %mv3p
	%hm3 =l mul %hx3, %P1
	%h =l add %hm3, %P4

	%mv4m =l mul %v4, %P2
	%mv4sl =l shl %mv4m, 31
	%mv4sr =l shr %mv4m, 33
	%mv4r =l or %mv4sl, %mv4sr
	%mv4p =l mul %mv4r, %P1
	%hx4 =l xor %h, %mv4p
	%hm4 =l mul %hx4, %P1
	%h =l add %hm4, %P4

	jmp @after_stripes

@small
	%h =l add %seed, %P5
	jmp @after_stripes

@after_stripes
	%h =l add %h, %n
	jmp @loop8_check

@loop8_check
	%c8 =w csgel %rem, 8
	jnz %c8, @loop8_body, @check4

@loop8_body
	%k8 =l loadl %q
	%k8m =l mul %k8, %P2
	%k8sl =l shl %k8m, 31
	%k8sr =l shr %k8m, 33
	%k8r =l or %k8sl, %k8sr
	%k8p =l mul %k8r, %P1

	%h8m =l mul %h8r, %P1
	%h =l add %h8m, %P4
	%q =l add %q, 8
	%rem =l sub %rem, 8
	jmp @loop8_check

@check4

	jnz %c4, @body4, @loop1_check

	%rem =l sub %rem, 4
	jmp @loop1_check

@loop1_check

	jnz %c1, @loop1_body, @finalize

@loop1_body

	%k1m =l mul %k1, %P5
	%h =l xor %h, %k1m
	%h1sl =l shl %h, 11
	%h1sr =l shr %h, 53
	%h1r =l or %h1sl, %h1sr
	%h =l mul %h1r, %P1
	%q =l add %q, 1
	%rem =l sub %rem, 1
	jmp @loop1_check

@finalize

	%h =l xor %h, %f1

}
