binary_search:
    mov x3, #0
    subs x4, x1, #1
    b.lt .Lnot_found
.Lloop:
    add x5, x3, x4
    lsr x5, x5, #1
    ldr x6, [x0, x5, lsl #3]
    cmp x6, x2
    b.eq .Lfound
    add x7, x5, #1
    sub x8, x5, #1
    csel x3, x7, x3, lt
    csel x4, x4, x8, lt
    cmp x3, x4
    b.le .Lloop
.Lnot_found:
    mov x0, #-1
    ret
.Lfound:
    mov x0, x5
    ret
