export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%sum =l phi @start 0, @body %sum1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%res =l call %f(l %val)
	%sum1 =l add %sum, %res
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %sum
}
