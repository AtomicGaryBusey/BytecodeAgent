export function l $bubble_sort(l %a, l %n) {
@start
	%ni =l sub %n, 1
	jmp @outer_cond

@outer_cond
	%i =l phi @start 0, @inner_done %i_next
	%cond_i =w csltl %i, %ni
	jnz %cond_i, @outer_body, @done

@outer_body
	%limit =l sub %ni, %i
	jmp @inner_cond

@inner_cond
	%j =l phi @outer_body 0, @noswap %j_next
	%cond_j =w csltl %j, %limit
	jnz %cond_j, @inner_body, @inner_done

@inner_body
	%offset =l mul %j, 8
	%ptr0 =l add %a, %offset
	%ptr1 =l add %ptr0, 8
	%val0 =l loadl %ptr0
	%val1 =l loadl %ptr1
	%cmp =w csgtl %val0, %val1
	jnz %cmp, @swap, @noswap

@swap
	storel %val1, %ptr0
	storel %val0, %ptr1
	jmp @noswap

@noswap
	%j_next =l add %j, 1
	jmp @inner_cond

@inner_done
	%i_next =l add %i, 1
	jmp @outer_cond

@done
	ret 0
}
