export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
@main_loop
    %i =l phi @start 0, @main_latch %i_next
    %out_pos =l phi @start 0, @main_latch %out_pos_next
    %cmp_in =w csltl %i, %n
    jnz %cmp_in, @check_run, @done

@check_run
    %i_plus_2 =l add %i, 2
    %can_run3 =w csltl %i_plus_2, %n
    jnz %can_run3, @check_run_bytes, @start_literal

@check_run_bytes
    %p0 =l add %src, %i
    %b0 =w loadub %p0
    %p1 =l add %p0, 1
    %b1 =w loadub %p1
    %eq01 =w ceqw %b0, %b1
    jnz %eq01, @check_run_byte2, @start_literal

@check_run_byte2
    %p2 =l add %p0, 2
    %b2 =w loadub %p2
    %eq02 =w ceqw %b0, %b2
    jnz %eq02, @do_repeat, @start_literal

@do_repeat
@repeat_loop
    %r =l phi @do_repeat 3, @repeat_latch %r_next
    %r_lt_128 =w csltl %r, 128
    jnz %r_lt_128, @repeat_check_n, @repeat_emit

@repeat_check_n
    %ir =l add %i, %r
    %ir_lt_n =w csltl %ir, %n
    jnz %ir_lt_n, @repeat_check_byte, @repeat_emit

@repeat_check_byte
    %pr =l add %src, %ir
    %br =w loadub %pr
    %eq_r =w ceqw %b0, %br
    jnz %eq_r, @repeat_latch, @repeat_emit

@repeat_latch
    %r_next =l add %r, 1
    jmp @repeat_loop

@repeat_emit
    %out_after_rep =l add %out_pos, 2
    %rep_fits =w cslel %out_after_rep, %cap
    jnz %rep_fits, @repeat_write, @fail

@repeat_write
    %hdr_rep =w sub 257, %r
    %out_ptr0 =l add %out, %out_pos
    storeb %hdr_rep, %out_ptr0
    %out_ptr1 =l add %out_ptr0, 1
    storeb %b0, %out_ptr1
    %i_after_rep =l add %i, %r
    jmp @main_latch

@start_literal
    %j_init =l add %i, 1
@lit_loop
    %j =l phi @start_literal %j_init, @lit_latch %j_next
    %j_lt_n =w csltl %j, %n
    jnz %j_lt_n, @lit_check_len, @lit_emit

@lit_check_len
    %cur_len =l sub %j, %i
    %len_lt_128 =w csltl %cur_len, 128
    jnz %len_lt_128, @lit_check_run, @lit_emit

@lit_check_run
    %j_plus_2 =l add %j, 2
    %j_can_run3 =w csltl %j_plus_2, %n
    jnz %j_can_run3, @lit_check_run_bytes, @lit_latch

@lit_check_run_bytes
    %pj0 =l add %src, %j
    %bj0 =w loadub %pj0
    %pj1 =l add %pj0, 1
    %bj1 =w loadub %pj1
    %jeq01 =w ceqw %bj0, %bj1
    jnz %jeq01, @lit_check_run_byte2, @lit_latch

@lit_check_run_byte2
    %pj2 =l add %pj0, 2
    %bj2 =w loadub %pj2
    %jeq02 =w ceqw %bj0, %bj2
    jnz %jeq02, @lit_emit, @lit_latch

@lit_latch
    %j_next =l add %j, 1
    jmp @lit_loop

@lit_emit
    %L =l sub %j, %i
    %pkt_len =l add %L, 1
    %out_after_lit =l add %out_pos, %pkt_len
    %lit_fits =w cslel %out_after_lit, %cap
    jnz %lit_fits, @lit_write_hdr, @fail

@lit_write_hdr
    %hdr_lit =w sub %L, 1
    %out_lit_hdr_ptr =l add %out, %out_pos
    storeb %hdr_lit, %out_lit_hdr_ptr
    %out_lit_data_base =l add %out_lit_hdr_ptr, 1

@lit_copy_loop
    %k =l phi @lit_write_hdr 0, @lit_copy_latch %k_next
    %src_k_idx =l add %i, %k
    %src_k_ptr =l add %src, %src_k_idx
    %byte_k =w loadub %src_k_ptr
    %out_k_ptr =l add %out_lit_data_base, %k
    storeb %byte_k, %out_k_ptr
    %k_next =l add %k, 1
    %more_copy =w csltl %k_next, %L
    jnz %more_copy, @lit_copy_latch, @lit_done

@lit_copy_latch
    jmp @lit_copy_loop

@lit_done
    jmp @main_latch

@main_latch
    %i_next =l phi @repeat_write %i_after_rep, @lit_done %j
    %out_pos_next =l phi @repeat_write %out_after_rep, @lit_done %out_after_lit
    jmp @main_loop

@done
    ret %out_pos

@fail
    ret -1
}
