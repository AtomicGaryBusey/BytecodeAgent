export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%max0 =l shl 1, 63
	%min0 =l sub %max0, 1
@loop
	%i =l phi @start 0, @next %i1
	%ptr =l phi @start %a, @next %ptr1
	%min =l phi @start %min0, @next %min2
	%max =l phi @start %max0, @next %max2
	%sum =l phi @start 0, @next %sum1
	%odd =l phi @start 0, @next %odd1
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%sum1 =l add %sum, %y
	%is_odd =l and %y, 1
	%odd1 =l add %odd, %is_odd
	%lt =w csltl %y, %min
	jnz %lt, @new_min, @keep_min
@new_min
	jmp @after_min
@keep_min
	jmp @after_min
@after_min
	%min2 =l phi @new_min %y, @keep_min %min
	%gt =w csltl %max, %y
	jnz %gt, @new_max, @keep_max
@new_max
	jmp @next
@keep_max
	jmp @next
@next
	%max2 =l phi @new_max %y, @keep_max %max
	%ptr1 =l add %ptr, 8
	%i1 =l add %i, 1
	%cont =w csltl %i1, %n
	jnz %cont, @loop, @done
@done
	storel %min2, %out
	%out1 =l add %out, 8
	storel %max2, %out1
	%out2 =l add %out, 16
	storel %sum1, %out2
	ret %odd1
}
