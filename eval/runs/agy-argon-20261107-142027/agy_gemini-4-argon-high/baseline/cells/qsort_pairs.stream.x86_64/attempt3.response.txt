qsort_pairs:
    cmp rsi, 1
    jle .Lret
    push rbp
    push rbx
    sub rsp, 8
    lea rax, [rsi - 1]
    and rax, -2
    mov rcx, qword ptr [rdi + rax*8]
    mov rdx, qword ptr [rdi + rax*8 + 8]
    mov r8, -2
    lea r9, [rsi + rsi]
.Li_loop:
    add r8, 2
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, rcx
    jl .Li_loop
    jg .Lj_loop
    mov rax, qword ptr [rdi + r8*8 + 8]
    cmp rax, rdx
    jl .Li_loop
.Lj_loop:
    sub r9, 2
    mov rax, qword ptr [rdi + r9*8]
    cmp rax, rcx
    jg .Lj_loop
    jl .Lcheck
    mov rax, qword ptr [rdi + r9*8 + 8]
    cmp rax, rdx
    jg .Lj_loop
.Lcheck:
    cmp r8, r9
    jge .Ldone_part
    mov rax, qword ptr [rdi + r8*8]
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov r11, qword ptr [rdi + r9*8]
    mov qword ptr [rdi + r8*8], r11
    mov r11, qword ptr [rdi + r9*8 + 8]
    mov qword ptr [rdi + r8*8 + 8], r11
    mov qword ptr [rdi + r9*8], rax
    mov qword ptr [rdi + r9*8 + 8], r10
    jmp .Li_loop
.Ldone_part:
    lea rbp, [rdi + r9*8 + 16]
    sar r9, 1
    inc r9
    mov rbx, rsi
    sub rbx, r9
    mov rsi, r9
    call qsort_pairs
    mov rdi, rbp
    mov rsi, rbx
    call qsort_pairs
    add rsp, 8
    pop rbx
    pop rbp
.Lret:
    xor eax, eax
    ret
