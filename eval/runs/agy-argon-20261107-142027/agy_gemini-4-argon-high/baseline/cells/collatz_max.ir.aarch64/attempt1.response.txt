export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@loop_k
	%cond_k =w cslel %k, %n
	jnz %cond_k, @init_x, @done
@init_x
	%x =l copy %k
	%steps =l copy 0
@loop_x
	%done_x =w ceql %x, 1
	jnz %done_x, @check_max, @step_x
@step_x
	%odd =l and %x, 1
	%is_even =w ceql %odd, 0
	jnz %is_even, @even_step, @odd_step
@even_step
	%x =l shr %x, 1
	jmp @inc_steps
@odd_step
	%t =l mul %x, 3
	%x =l add %t, 1
@inc_steps
	%steps =l add %steps, 1
	jmp @loop_x
@check_max
	%gt =w csgtl %steps, %max
	jnz %gt, @update_max, @next_k
@update_max
	%max =l copy %steps
@next_k
	%k =l add %k, 1
	jmp @loop_k
@done
	ret %max
}
