export function l $collatz_max(l %n) {
@start
	jmp @outer_hdr
@outer_hdr
	%k =l phi @start 1, @outer_latch %k_next
	%max =l phi @start 0, @outer_latch %max_next
	%done_cond =w csltl %n, %k
	jnz %done_cond, @done, @inner_hdr
@inner_hdr
	%x =l phi @outer_hdr %k, @inner_latch %x_next
	%steps =l phi @outer_hdr 0, @inner_latch %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @inner_done, @inner_body
@inner_body
	%bit =l and %x, 1
	%is_even =w ceql %bit, 0
	jnz %is_even, @even, @odd
@even
	%x_even =l shr %x, 1
	jmp @inner_latch
@odd
	%t =l mul %x, 3
	%x_odd =l add %t, 1
	jmp @inner_latch
@inner_latch
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner_hdr
@inner_done
	%is_new_max =w csltl %max, %steps
	jnz %is_new_max, @new_max, @outer_latch
@new_max
	%m1 =l copy %steps
	jmp @outer_latch
@outer_latch
	%max_next =l phi @new_max %m1, @inner_done %max
	%k_next =l add %k, 1
	jmp @outer_hdr
@done
	ret %max
}
