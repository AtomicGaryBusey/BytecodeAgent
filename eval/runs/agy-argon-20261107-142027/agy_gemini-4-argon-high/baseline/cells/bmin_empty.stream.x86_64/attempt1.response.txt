bmin_empty:
    cmp rsi, 4
    jb .Llt4
    mov rax, qword ptr [rdi]
    mov rdx, qword ptr [rdi + 8]
    mov r8, qword ptr [rdi + 16]
    mov r9, qword ptr [rdi + 24]
    je .Lreduce4
    lea rcx, [rdi + rsi*8 - 32]
    add rdi, 32
    cmp rdi, rcx
    jae .Ltail4
.Lloop4:
    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]
    cmp r10, rax
    cmovl rax, r10
    cmp r11, rdx
    cmovl rdx, r11
    mov r10, qword ptr [rdi + 16]
    mov r11, qword ptr [rdi + 24]
    cmp r10, r8
    cmovl r8, r10
    cmp r11, r9
    cmovl r9, r11
    add rdi, 32
    cmp rdi, rcx
    jb .Lloop4
.Ltail4:
    mov r10, qword ptr [rcx]
    mov r11, qword ptr [rcx + 8]
    cmp r10, rax
    cmovl rax, r10
    cmp r11, rdx
    cmovl rdx, r11
    mov r10, qword ptr [rcx + 16]
    mov r11, qword ptr [rcx + 24]
    cmp r10, r8
    cmovl r8, r10
    cmp r11, r9
    cmovl r9, r11
.Lreduce4:
    cmp rdx, rax
    cmovl rax, rdx
    cmp r9, r8
    cmovl r8, r9
    cmp r8, rax
    cmovl rax, r8
    ret
.Llt4:
    cmp rsi, 1
    jb .Lempty
    mov rax, qword ptr [rdi]
    je .Ldone
    mov rdx, qword ptr [rdi + 8]
    mov rcx, qword ptr [rdi + rsi*8 - 8]
    cmp rdx, rax
    cmovl rax, rdx
    cmp rcx, rax
    cmovl rax, rcx
.Ldone:
    ret
.Lempty:
    movabs rax, 0x7fffffffffffffff
    ret
