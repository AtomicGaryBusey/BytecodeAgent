export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @next %nlo
	%hi =l phi @start 4294967295, @next %nhi
	%cmp =w cultl %lo, %hi
	jnz %cmp, @body, @done
@body
	%s1 =l add %lo, %hi
	%s2 =l add %s1, 1
	%mid =l shr %s2, 1
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @next, @gt
@gt
	%him1 =l sub %mid, 1
	jmp @next
@next
	%nlo =l phi @body %mid, @gt %lo
	%nhi =l phi @body %hi, @gt %him1
	jmp @loop
@done
	ret %
