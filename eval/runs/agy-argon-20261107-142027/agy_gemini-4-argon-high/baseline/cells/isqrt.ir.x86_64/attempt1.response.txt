export function l $isqrt(l %n0) {
@start
	%bit0 =l copy 4611686018427387904
	%res0 =l copy 0
	jmp @loop
@loop
	%n =l phi @start %n0, @step %n2
	%res =l phi @start %res0, @step %res2
	%bit =l phi @start %bit0, @step %bit2
	%cond =w cnel %bit, 0
	jnz %cond, @body, @done
@body
	%t =l add %res, %bit
	%res1 =l shr %res, 1
	%ge =w cugel %n, %t
	jnz %ge, @take, @step
@take
	%n1 =l sub %n, %t
	%res_take =l add %res1, %bit
	jmp @step
@step
	%n2 =l phi @body %n, @take %n1
	%res2 =l phi @body %res1, @take %res_take
	%bit2 =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
