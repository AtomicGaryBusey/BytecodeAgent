export function l $crc32(l %p, l %n) {
@start
	jmp @check
@check
	%i =l phi @start 0, @bitdone %i1
	%crc =w phi @start 4294967295, @bitdone %crc4
	%cond =w csltl %i, %n
	jnz %cond, @byteloop, @done
@byteloop
	%ptr =l add %p, %i
	%byte =w loadub %ptr
	%crc1 =w xor %crc, %byte
	jmp @bitloop
@bitloop
	%b_idx =w phi @byteloop 0, @bitloop %b_idx1
	%crc2 =w phi @byteloop %crc1, @bitloop %crc4
	%bit =w and %crc2, 1
	%mask =w sub 0, %bit
	%poly =w and %mask, 3988292384
	%shr =w shr %crc2, 1
	%crc4 =w xor %shr, %poly
	%b_idx1 =w add %b_idx, 1
	%b_cond =w csltw %b_idx1, 8
	jnz %b_cond, @bitloop, @bitdone
@bitdone
	%i1 =l add %i, 1
	jmp @check
@done
	%res_w =w xor %crc, 4294967295
	%res =l extuw %res_w
	ret %res
}
