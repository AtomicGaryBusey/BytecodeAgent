export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%c =l phi @start %c0, @body %c1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%dptr =l add %dst, %off
	%sv =l loadl %sptr
	%dv =l call %f(l %sv)
	storel %dv, %dptr
	%lt =w csltl %dv, %sv
	%inc =l extuw %lt
	%c1 =l add %c, %inc
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %c
}
