    cmp x1, #1
    b.le .Ldone
    sub x2, x1, #2
    add x2, x0, x2, lsl #3
    lsr x3, x1, #2
    cbz x3, .Ltail
.Lloop:
    ldp x4, x5, [x0]
    ldp x6, x7, [x2]
    stp x7, x6, [x0], #16
    stp x5, x4, [x2], #-16
    subs x3, x3, #1
    b.ne .Lloop
.Ltail:
    tbz x1, #1, .Ldone
    ldr x4, [x0]
    ldr x5, [x2, #8]
    str x5, [x0]
    str x4, [x2, #8]
.Ldone:
    mov x0, #0
    ret
