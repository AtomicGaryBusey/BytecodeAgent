export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i =l copy 0
	%sum =l copy 0
	%odds =l copy 0
	%min =l copy 0
	%max =l copy 0
@loop
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%sum =l add %sum, %y
	%is_odd =l and %y, 1
	%odds =l add %odds, %is_odd
	%is_first =w ceql %i, 0
	jnz %is_first, @first, @check_min
@first
	%min =l copy %y
	%max =l copy %y
	jmp @next
@check_min
	%lt =w csltl %y, %min
	jnz %lt, @upd_min, @check_max
@upd_min
	%min =l copy %y
	jmp @check_max
@check_max
	%gt =w csgtl %y, %max
	jnz %gt, @upd_max, @next
@upd_max
	%max =l copy %y
	jmp @next
@next
	%i =l add %i, 1
	%more =w csltl %i, %n
	jnz %more, @loop, @done
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odds
}
