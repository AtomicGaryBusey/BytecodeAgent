export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%max =l phi @start %max0, @next %max1
	%i =l phi @start %i0, @next %i1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %max
	jnz %gt, @upd, @next
@upd
	jmp @next
@next
	%max1 =l phi @upd %v, @body %max
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %max
}
