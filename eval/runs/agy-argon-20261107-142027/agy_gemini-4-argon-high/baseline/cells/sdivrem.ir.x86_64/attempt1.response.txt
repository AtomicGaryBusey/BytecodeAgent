export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%z =w ceql %b, 0
	jnz %z, @byzero, @checkneg1
@byzero
	storel %a, %rem
	ret 0
@checkneg1
	%m =w ceql %b, -1
	jnz %m, @byneg1, @normal
@byneg1
	%na =l sub 0, %a
	storel 0, %rem
	ret %na
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
