memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.hi .Lbackward
.Lforward:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    b.ne .Lforward
    mov x0, #0
    ret
.Lbackward:
    sub x2, x2, #1
    ldr x3, [x1, x2, lsl #3]
    str x3, [x0, x2, lsl #3]
    cbnz x2, .Lbackward
.Ldone:
    mov x0, #0
    ret
