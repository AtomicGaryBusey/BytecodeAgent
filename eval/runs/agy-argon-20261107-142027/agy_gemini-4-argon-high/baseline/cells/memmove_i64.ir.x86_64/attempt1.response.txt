export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w cslel %n, 0
    jnz %empty, @done, @check_dir
@check_dir
    %fwd =w culel %dst, %src
    jnz %fwd, @fwd_init, @bwd_init
@fwd_init
    %i0 =l copy 0
    jmp @fwd_loop
@fwd_loop
    %i =l phi @fwd_init %i0, @fwd_body %i_next
    %cond_f =w csltl %i, %n
    jnz %cond_f, @fwd_body, @done
@fwd_body
    %off_f =l mul %i, 8
    %sp_f =l add %src, %off_f
    %dp_f =l add %dst, %off_f
    %val_f =l loadl %sp_f
    storel %val_f, %dp_f
    %i_next =l add %i, 1
    jmp @fwd_loop
@bwd_init
    %j0 =l copy %n
    jmp @bwd_loop
@bwd_loop
    %j =l phi @bwd_init %j0, @bwd_body %j_prev
    %cond_b =w csgtl %j, 0
    jnz %cond_b, @bwd_body, @done
@bwd_body
    %j_prev =l sub %j, 1
    %off_b =l mul %j_prev, 8
    %sp_b =l add %src, %off_b
    %dp_b =l add %dst, %off_b
    %val_b =l loadl %sp_b
    storel %val_b, %dp_b
    jmp @bwd_loop
@done
    ret 0
}
