export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	jmp @zero_out_hdr

@zero_out_hdr
	%zo_i =l phi @start 0, @zero_out_body %zo_i_next
	%zo_cmp =w csltl %zo_i, %nn
	jnz %zo_cmp, @zero_out_body, @diag_hdr

@zero_out_body
	%zo_off =l mul %zo_i, 8
	%zo_ptr =l add %out, %zo_off
	storel 0, %zo_ptr
	%zo_i_next =l add %zo_i, 1
	jmp @zero_out_hdr

@diag_hdr
	%d_i =l phi @zero_out_hdr 0, @diag_body %d_i_next
	%d_cmp =w csltl %d_i, %n
	jnz %d_cmp, @diag_body, @pow_hdr

@diag_body
	%d_row =l mul %d_i, %n
	%d_idx =l add %d_row, %d_i
	%d_off =l mul %d_idx, 8
	%d_ptr =l add %out, %d_off
	storel 1, %d_ptr
	%d_i_next =l add %d_i, 1
	jmp @diag_hdr

@pow_hdr
	%rem =l phi @diag_hdr %k, @pow_next %rem_next
	%pow_cmp =w csgtl %rem, 0
	jnz %pow_cmp, @mm_i_hdr, @zero_tmp_hdr

@mm_i_hdr
	%mi =l phi @pow_hdr 0, @mm_i_next %mi_next
	%mi_cmp =w csltl %mi, %n
	jnz %mi_cmp, @mm_j_hdr, @copy_hdr

@mm_j_hdr
	%mj =l phi @mm_i_hdr 0, @mm_j_next %mj_next
	%mj_cmp =w csltl %mj, %n
	jnz %mj_cmp, @mm_t_hdr, @mm_i_next

@mm_t_hdr
	%mt =l phi @mm_j_hdr 0, @mm_t_body %mt_next
	%sum =l phi @mm_j_hdr 0, @mm_t_body %sum_next
	%mt_cmp =w csltl %mt, %n
	jnz %mt_cmp, @mm_t_body, @mm_j_next

@mm_t_body
	%o_row =l mul %mi, %n
	%o_idx =l add %o_row, %mt
	%o_off =l mul %o_idx, 8
	%o_ptr =l add %out, %o_off
	%o_val =l loadl %o_ptr
	%a_row =l mul %mt, %n
	%a_idx =l add %a_row, %mj
	%a_off =l mul %a_idx, 8
	%a_ptr =l add %a, %a_off
	%a_val =l loadl %a_ptr
	%prod =l mul %o_val, %a_val
	%sum_next =l add %sum, %prod
	%mt_next =l add %mt, 1
	jmp @mm_t_hdr

@mm_j_next
	%t_row =l mul %mi, %n
	%t_idx =l add %t_row, %mj
	%t_off =l mul %t_idx, 8
	%t_ptr =l add %tmp, %t_off
	storel %sum, %t_ptr
	%mj_next =l add %mj, 1
	jmp @mm_j_hdr

@mm_i_next
	%mi_next =l add %mi, 1
	jmp @mm_i_hdr

@copy_hdr
	%ci =l phi @mm_i_hdr 0, @copy_body %ci_next
	%c_cmp =w csltl %ci, %nn
	jnz %c_cmp, @copy_body, @pow_next

@copy_body
	%c_off =l mul %ci, 8
	%c_src =l add %tmp, %c_off
	%c_dst =l add %out, %c_off
	%c_val =l loadl %c_src
	storel %c_val, %c_dst
	%ci_next =l add %ci, 1
	jmp @copy_hdr

@pow_next
	%rem_next =l sub %rem, 1
	jmp @pow_hdr

@zero_tmp_hdr
	%zt_i =l phi @pow_hdr 0, @zero_tmp_body %zt_i_next
	%zt_cmp =w csltl %zt_i, %nn2
	jnz %zt_cmp, @zero_tmp_body, @done

@zero_tmp_body
	%zt_off =l mul %zt_i, 8
	%zt_ptr =l add %tmp, %zt_off
	storel 0, %zt_ptr
	%zt_i_next =l add %zt_i, 1
	jmp @zero_tmp_hdr

@done
	ret 0
}
