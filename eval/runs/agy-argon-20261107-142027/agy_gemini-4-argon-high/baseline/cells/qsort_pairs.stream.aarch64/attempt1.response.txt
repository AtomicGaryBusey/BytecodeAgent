qsort_pairs:
    cmp x1, #1
    b.le .Lret_zero
    stp x29, x30, [sp, #-32]!
    stp x19, x20, [sp, #16]
    mov x29, sp
    sub x10, x1, #1
    lsr x10, x10, #1
    add x10, x0, x10, lsl #4
    ldp x6, x7, [x10]
    sub x8, x0, #16
    add x9, x0, x1, lsl #4
.Lscan_i:
    ldp x2, x3, [x8, #16]!
    cmp x2, x6
    b.ne .Lcmp_i_done
    cmp x3, x7
.Lcmp_i_done:
    b.lt .Lscan_i
.Lscan_j:
    ldp x4, x5, [x9, #-16]!
    cmp x4, x6
    b.ne .Lcmp_j_done
    cmp x5, x7
.Lcmp_j_done:
    b.gt .Lscan_j
    cmp x8, x9
    b.hs .Lpartition_done
    stp x4, x5, [x8]
    stp x2, x3, [x9]
    b .Lscan_i
.Lpartition_done:
    add x19, x9, #16
    sub x10, x19, x0
    lsr x10, x10, #4
    sub x20, x1, x10
    mov x1, x10
    bl qsort_pairs
    mov x0, x19
    mov x1, x20
    bl qsort_pairs
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
.Lret_zero:
    mov x0, #0
    ret
