export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%p =l phi @start %a, @next %p1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @next, @done
@next
	%i1 =l add %i, 1
	%p1 =l add %p, 8
	jmp @loop
@done
	ret %i
}
