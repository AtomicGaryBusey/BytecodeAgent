export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0

@loop
	%end_c =w cslel %len, %pc
	jnz %end_c, @halt, @fetch

@fetch
	%pc_off =l shl %pc, 3
	%code_ptr =l add %code, %pc_off
	%op =l loadl %code_ptr

	%c0 =w ceql %op, 0
	jnz %c0, @halt, @check1
@check1
	%c1 =w ceql %op, 1
	jnz %c1, @op_push, @check2
@check2
	%c2 =w ceql %op, 2
	jnz %c2, @op_add, @check3
@check3
	%c3 =w ceql %op, 3
	jnz %c3, @op_sub, @check4
@check4
	%c4 =w ceql %op, 4
	jnz %c4, @op_mul, @check5
@check5
	%c5 =w ceql %op, 5
	jnz %c5, @op_dup, @check6
@check6
	%c6 =w ceql %op, 6
	jnz %c6, @op_swap, @check7
@check7
	%c7 =w ceql %op, 7
	jnz %c7, @op_drop, @check8
@check8
	%c8 =w ceql %op, 8
	jnz %c8, @op_over, @check9
@check9
	%c9 =w ceql %op, 9
	jnz %c9, @op_lt, @check10
@check10
	%c10 =w ceql %op, 10
	jnz %c10, @op_jz, @op_jmp

@op_push
	%v_ptr =l add %code_ptr, 8
	%v =l loadl %v_ptr
	%push_off =l shl %d, 3
	%push_ptr =l add %stack, %push_off
	storel %v, %push_ptr
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop

@op_add
	%add_off =l shl %d, 3
	%add_top =l add %stack, %add_off
	%add_bp =l sub %add_top, 8
	%add_ap =l sub %add_top, 16
	%add_b =l loadl %add_bp
	%add_a =l loadl %add_ap
	%add_r =l add %add_a, %add_b
	storel %add_r, %add_ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_sub
	%sub_off =l shl %d, 3
	%sub_top =l add %stack, %sub_off
	%sub_bp =l sub %sub_top, 8
	%sub_ap =l sub %sub_top, 16
	%sub_b =l loadl %sub_bp
	%sub_a =l loadl %sub_ap
	%sub_r =l sub %sub_a, %sub_b
	storel %sub_r, %sub_ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_mul
	%mul_off =l shl %d, 3
	%mul_top =l add %stack, %mul_off
	%mul_bp =l sub %mul_top, 8
	%mul_ap =l sub %mul_top, 16
	%mul_b =l loadl %mul_bp
	%mul_a =l loadl %mul_ap
	%mul_r =l mul %mul_a, %mul_b
	storel %mul_r, %mul_ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_dup
	%dup_off =l shl %d, 3
	%dup_top =l add %stack, %dup_off
	%dup_src =l sub %dup_top, 8
	%dup_v =l loadl %dup_src
	storel %dup_v, %dup_top
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_swap
	%swap_off =l shl %d, 3
	%swap_top =l add %stack, %swap_off
	%swap_bp =l sub %swap_top, 8
	%swap_ap =l sub %swap_top, 16
	%swap_b =l loadl %swap_bp
	%swap_a =l loadl %swap_ap
	storel %swap_b, %swap_ap
	storel %swap_a, %swap_bp
	%pc =l add %pc, 1
	jmp @loop

@op_drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_over
	%over_off =l shl %d, 3
	%over_top =l add %stack, %over_off
	%over_src =l sub %over_top, 16
	%over_v =l loadl %over_src
	storel %over_v, %over_top
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_lt
	%lt_off =l shl %d, 3
	%lt_top =l add %stack, %lt_off
	%lt_bp =l sub %lt_top, 8
	%lt_ap =l sub %lt_top, 16
	%lt_b =l loadl %lt_bp
	%lt_a =l loadl %lt_ap
	%lt_cw =w csltl %lt_a, %lt_b
	%lt_cl =l extuw %lt_cw
	storel %lt_cl, %lt_ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop

@op_jz
	%jz_tp =l add %code_ptr, 8
	%jz_t =l loadl %jz_tp
	%d =l sub %d, 1
	%jz_off =l shl %d, 3
	%jz_vp =l add %stack, %jz_off
	%jz_v =l loadl %jz_vp
	%jz_c =w ceql %jz_v, 0
	jnz %jz_c, @jz_take, @jz_skip
@jz_take
	%pc =l copy %jz_t
	jmp @loop
@jz_skip
	%pc =l add %pc, 2
	jmp @loop

@op_jmp
	%jmp_tp =l add %code_ptr, 8
	%pc =l loadl %jmp_tp
	jmp @loop

@halt
	%z =l copy %d
@zloop
	%zc =w cslel %cap, %z
	jnz %zc, @done, @zbody
@zbody
	%z_off =l shl %z, 3
	%z_ptr =l add %stack, %z_off
	storel 0, %z_ptr
	%z =l add %z, 1
	jmp @zloop

@done
	%empty_c =w ceql %d, 0
	jnz %empty_c, @ret0, @rettop
@ret0
	ret 0
@rettop
	%ret_off =l shl %d, 3
	%ret_p1 =l add %stack, %ret_off
	%ret_p =l sub %ret_p1, 8
	%ret_v =l loadl %ret_p
	ret %ret_v
}
