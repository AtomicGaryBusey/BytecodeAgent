export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    jmp @loop

@loop
    %pc =l phi @start 0, @step %pc_next
    %sp =l phi @start %stack, @step %sp_next
    %cmp_end =w csgel %pc, %len
    jnz %cmp_end, @done, @exec

@exec
    %off =l shl %pc, 3
    %cptr =l add %code, %off
    %op =l loadl %cptr
    %is0 =w ceql %op, 0
    jnz %is0, @done, @chk1
@chk1
    %is1 =w ceql %op, 1
    jnz %is1, @op_push, @chk2
@chk2
    %is2 =w ceql %op, 2
    jnz %is2, @op_add, @chk3
@chk3
    %is3 =w ceql %op, 3
    jnz %is3, @op_sub, @chk4
@chk4
    %is4 =w ceql %op, 4
    jnz %is4, @op_mul, @chk5
@chk5
    %is5 =w ceql %op, 5
    jnz %is5, @op_dup, @chk6
@chk6
    %is6 =w ceql %op, 6
    jnz %is6, @op_swap, @chk7
@chk7
    %is7 =w ceql %op, 7
    jnz %is7, @op_drop, @chk8
@chk8
    %is8 =w ceql %op, 8
    jnz %is8, @op_over, @chk9
@chk9
    %is9 =w ceql %op, 9
    jnz %is9, @op_lt, @chk10
@chk10
    %is10 =w ceql %op, 10
    jnz %is10, @op_jz, @op_jmp

@op_push
    %vptr =l add %cptr, 8
    %v =l loadl %vptr
    storel %v, %sp
    %sp_push =l add %sp, 8
    %pc_push =l add %pc, 2
    jmp @step

@op_add
    %sp_add =l sub %sp, 8
    %b_add =l loadl %sp_add
    %aptr_add =l sub %sp, 16
    %a_add =l loadl %aptr_add
    %res_add =l add %a_add, %b_add
    storel %res_add, %aptr_add
    jmp @step1

@op_sub
    %sp_sub =l sub %sp, 8
    %b_sub =l loadl %sp_sub
    %aptr_sub =l sub %sp, 16
    %a_sub =l loadl %aptr_sub
    %res_sub =l sub %a_sub, %b_sub
    storel %res_sub, %aptr_sub
    jmp @step1

@op_mul
    %sp_mul =l sub %sp, 8
    %b_mul =l loadl %sp_mul
    %aptr_mul =l sub %sp, 16
    %a_mul =l loadl %aptr_mul
    %res_mul =l mul %a_mul, %b_mul
    storel %res_mul, %aptr_mul
    jmp @step1

@op_dup
    %tptr_dup =l sub %sp, 8
    %v_dup =l loadl %tptr_dup
    storel %v_dup, %sp
    %sp_dup =l add %sp, 8
    jmp @step1

@op_swap
    %bptr_swap =l sub %sp, 8
    %b_swap =l loadl %bptr_swap
    %aptr_swap =l sub %sp, 16
    %a_swap =l loadl %aptr_swap
    storel %b_swap, %aptr_swap
    storel %a_swap, %bptr_swap
    jmp @step1

@op_drop
    %sp_drop =l sub %sp, 8
    jmp @step1

@op_over
    %aptr_over =l sub %sp, 16
    %v_over =l loadl %aptr_over
    storel %v_over, %sp
    %sp_over =l add %sp, 8
    jmp @step1

@op_lt
    %sp_lt =l sub %sp, 8
    %b_lt =l loadl %sp_lt
    %aptr_lt =l sub %sp, 16
    %a_lt =l loadl %aptr_lt
    %cmp_lt =w csltl %a_lt, %b_lt
    %res_lt =l extuw %cmp_lt
    storel %res_lt, %aptr_lt
    jmp @step1

@step1
    %sp_s1 =l phi @op_add %sp_add, @op_sub %sp_sub, @op_mul %sp_mul, @op_dup %sp_dup, @op_swap %sp, @op_drop %sp_drop, @op_over %sp_over, @op_lt %sp_lt
    %pc_s1 =l add %pc, 1
    jmp @step

@op_jz
    %sp_jz =l sub %sp, 8
    %v_jz =l loadl %sp_jz
    %z_jz =w ceql %v_jz, 0
    jnz %z_jz, @op_jz_taken, @op_jz_not_taken
@op_jz_taken
    %tptr_jz =l add %cptr, 8
    %pc_jz1 =l loadl %tptr_jz
    jmp @step
@op_jz_not_taken
    %pc_jz2 =l add %pc, 2
    jmp @step

@op_jmp
    %tptr_jmp =l add %cptr, 8
    %pc_jmp =l loadl %tptr_jmp
    jmp @step

@step
    %pc_next =l phi @op_push %pc_push, @step1 %pc_s1, @op_jz_taken %pc_jz1, @op_jz_not_taken %pc_jz2, @op_jmp %pc_jmp
    %sp_next =l phi @op_push %sp_push, @step1 %sp_s1, @op_jz_taken %sp_jz, @op_jz_not_taken %sp_jz, @op_jmp %sp
    jmp @loop

@done
    %empty =w ceql %sp, %stack
    jnz %empty, @ret_empty, @ret_nonempty
@ret_empty
    jmp @zloop_init
@ret_nonempty
    %top_ptr =l sub %sp, 8
    %top_val =l loadl %top_ptr
    jmp @zloop_init
@zloop_init
    %retval =l phi @ret_empty 0, @ret_nonempty %top_val
    %cap_bytes =l shl %cap, 3
    %stack_end =l add %stack, %cap_bytes
    jmp @zloop
@zloop
    %zptr =l phi @zloop_init %sp, @zbody %zptr_next
    %zd =w cugel %zptr, %stack_end
    jnz %zd, @exit, @zbody
@zbody
    storel 0, %zptr
    %zptr_next =l add %zptr, 8
    jmp @zloop
@exit
    ret %retval
}
