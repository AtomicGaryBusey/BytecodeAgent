export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%p =l phi @start %a, @body %p1
	%sum =l phi @start 0, @body %sum1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum1 =l add %sum, %r
	%p1 =l add %p, 8
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %sum
}
