export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%cmp =w csltl %max, %b
	jnz %cmp, @update, @next
@update
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
