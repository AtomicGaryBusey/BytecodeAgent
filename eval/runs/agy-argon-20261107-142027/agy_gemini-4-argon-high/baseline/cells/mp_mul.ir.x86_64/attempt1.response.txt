export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %ntot =l add %na, %nb
    jmp @zero_check

@zero_check
    %z =l phi @start 0, @zero_body %z_next
    %z_cond =w csltl %z, %ntot
    jnz %z_cond, @zero_body, @check_empty

@zero_body
    %z_off =l shl %z, 3
    %z_ptr =l add %r, %z_off
    storel 0, %z_ptr
    %z_next =l add %z, 1
    jmp @zero_check

@check_empty
    %na_zero =w cslel %na, 0
    jnz %na_zero, @ret_zero, @check_nb

@check_nb
    %nb_zero =w cslel %nb, 0
    jnz %nb_zero, @ret_zero, @outer_check

@ret_zero
    ret 0

@outer_check
    %i =l phi @check_nb 0, @outer_end %i_next
    %i_cond =w csltl %i, %na
    jnz %i_cond, @outer_body, @norm_check

@outer_body
    %i_off =l shl %i, 3
    %ai_ptr =l add %a, %i_off
    %ai =l loadl %ai_ptr
    %mask =l copy 4294967295
    %ai_lo =l and %ai, %mask
    %ai_hi =l shr %ai, 32
    jmp @inner_check

@inner_check
    %j =l phi @outer_body 0, @inner_body %j_next
    %carry =l phi @outer_body 0, @inner_body %carry_next
    %j_cond =w csltl %j, %nb
    jnz %j_cond, @inner_body, @outer_end

@inner_body
    %j_off =l shl %j, 3
    %bj_ptr =l add %b, %j_off
    %bj =l loadl %bj_ptr
    %bj_lo =l and %bj, %mask
    %bj_hi =l shr %bj, 32

    %p00 =l mul %ai_lo, %bj_lo
    %p01 =l mul %ai_lo, %bj_hi
    %p10 =l mul %ai_hi, %bj_lo
    %p11 =l mul %ai_hi, %bj_hi

    %p00_lo =l and %p00, %mask
    %p00_hi =l shr %p00, 32
    %p10_lo =l and %p10, %mask
    %p10_hi =l shr %p10, 32

    %mid1 =l add %p01, %p00_hi
    %mid =l add %mid1, %p10_lo
    %mid_shl =l shl %mid, 32
    %prod_lo =l or %p00_lo, %mid_shl

    %mid_hi =l shr %mid, 32
    %hi1 =l add %p11, %p10_hi
    %prod_hi =l add %hi1, %mid_hi

    %ij =l add %i, %j
    %ij_off =l shl %ij, 3
    %rij_ptr =l add %r, %ij_off
    %rij =l loadl %rij_ptr

    %s1 =l add %prod_lo, %carry
    %c1_w =w cultl %s1, %prod_lo
    %c1 =l extuw %c1_w

    %s2 =l add %s1, %rij
    %c2_w =w cultl %s2, %s1
    %c2 =l extuw %c2_w

    storel %s2, %rij_ptr

    %ch1 =l add %prod_hi, %c1
    %carry_next =l add %ch1, %c2

    %j_next =l add %j, 1
    jmp @inner_check

@outer_end
    %inb =l add %i, %nb
    %inb_off =l shl %inb, 3
    %rinb_ptr =l add %r, %inb_off
    storel %carry, %rinb_ptr
    %i_next =l add %i, 1
    jmp @outer_check

@norm_check
    %k =l phi @outer_check %ntot, @norm_body %k_prev
    %k_pos =w csgtl %k, 0
    jnz %k_pos, @norm_body, @norm_done

@norm_body
    %k_prev =l sub %k, 1
    %k_off =l shl %k_prev, 3
    %rk_ptr =l add %r, %k_off
    %rk =l loadl %rk_ptr
    %rk_nz =w cnel %rk, 0
    jnz %rk_nz, @norm_done, @norm_check

@norm_done
    ret %k
}
