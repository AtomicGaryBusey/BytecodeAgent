export function l $count_desc(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @zero, @loop_init
@zero
	ret 0
@loop_init
	%v0 =l loadl %a
	%lim =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @loop_init 0, @loop %i_next
	%cnt =l phi @loop_init 0, @loop %cnt_next
	%prev =l phi @loop_init %v0, @loop %cur
	%i_next =l add %i, 1
	%off =l mul %i_next, 8
	%ptr =l add %a, %off
	%cur =l loadl %ptr
	%is_gt =w csgtl %prev, %cur
	%inc =l extuw %is_gt
	%cnt_next =l add %cnt, %inc
	%cont =w csltl %i_next, %lim
	jnz %cont, @loop, @done
@done
	ret %cnt_next
}
