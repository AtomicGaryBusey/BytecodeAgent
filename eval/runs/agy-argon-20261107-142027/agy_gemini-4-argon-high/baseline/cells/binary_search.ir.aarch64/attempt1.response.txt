export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @left %lo1, @right %lo2
	%hi =l phi @start %n, @left %hi1, @right %hi2
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @right, @left
@left
	%lo1 =l copy %lo
	%hi1 =l copy %mid
	jmp @loop
@right
	%lo2 =l add %mid, 1
	%hi2 =l copy %hi
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
