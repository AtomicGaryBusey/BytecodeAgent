export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @not_found
@body
	%diff =l sub %hi, %lo
	%half =l sar %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check_lt
@check_lt
	%lt =w csltl %val, %key
	jnz %lt, @go_right, @go_left
@go_right
	%lo =l add %mid, 1
	jmp @loop
@go_left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@not_found
	ret -1
}
