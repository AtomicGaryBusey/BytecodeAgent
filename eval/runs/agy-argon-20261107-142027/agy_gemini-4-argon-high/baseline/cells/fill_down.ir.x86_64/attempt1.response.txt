export function l $fill_down(l %a, l %n, l %start) {
@start
@loop
    %i =l phi @start 0, @body %i_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l shl %i, 3
    %ptr =l add %a, %off
    %val =l sub %start, %i
    storel %val, %ptr
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %n
}
