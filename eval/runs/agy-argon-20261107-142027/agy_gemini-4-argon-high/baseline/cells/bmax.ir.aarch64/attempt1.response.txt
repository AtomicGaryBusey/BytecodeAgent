export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %i1
	%m =l phi @start %m0, @next %m2
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %m
	jnz %gt, @take, @skip
@take
	jmp @next
@skip
	jmp @next
@next
	%m2 =l phi @take %val, @skip %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
