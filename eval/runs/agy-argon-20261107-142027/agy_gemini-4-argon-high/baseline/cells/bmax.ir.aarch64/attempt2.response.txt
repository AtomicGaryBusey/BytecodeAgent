export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
	%c0 =w csltl %i0, %n
	jnz %c0, @loop, @done
@loop
	%m =l phi @start %m0, @next %m2
	%p =l phi @start %a, @next %p1
	%i =l phi @start %i0, @next %i1
	%p1 =l add %p, 8
	%val =l loadl %p1
	%i1 =l add %i, 1
	%lt =w csltl %m, %val
	jnz %lt, @take, @next
@take
	%m1 =l copy %val
	jmp @next
@next

	%c1 =w csltl %i1, %n
	jnz %c1, @loop, @done
@done

	ret %res
}
