lcs_len:
    push rbx
    xor eax, eax
.Lzero_loop:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Lzero_loop
    test rsi, rsi
    jle .Ldone
    test rcx, rcx
    jle .Ldone
.Louter_loop:
    movzx r10d, byte ptr [rdi]
    xor eax, eax
    xor r11d, r11d
    mov r9, 1
.Linner_loop:
    mov rbx, qword ptr [r8 + r9*8]
    cmp rax, rbx
    cmovl rax, rbx
    inc r11
    cmp r10b, byte ptr [rdx + r9 - 1]
    cmove rax, r11
    mov qword ptr [r8 + r9*8], rax
    mov r11, rbx
    inc r9
    cmp r9, rcx
    jle .Linner_loop
    inc rdi
    dec rsi
    jnz .Louter_loop
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    pop rbx
    ret
