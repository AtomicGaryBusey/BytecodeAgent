max_byte:
    cmp x1, #16
    b.lt .Lsmall
    ldr q0, [x0]
    add x2, x0, x1
    sub x1, x1, #16
    cbz x1, .Lreduce16
.Lloop16:
    add x0, x0, #16
    subs x1, x1, #16
    b.le .Ltail16
    ldr q1, [x0]
    umax v0.16b, v0.16b, v1.16b
    b .Lloop16
.Ltail16:
    ldr q1, [x2, #-16]
    umax v0.16b, v0.16b, v1.16b
.Lreduce16:
    umaxv b0, v0.16b
    umov w0, v0.b[0]
    ret
.Lsmall:
    cmp x1, #8
    b.lt .Ltiny
    ldr d0, [x0]
    add x2, x0, x1
    ldr d1, [x2, #-8]
    umax v0.8b, v0.8b, v1.8b
    umaxv b0, v0.8b
    umov w0, v0.b[0]
    ret
.Ltiny:
    mov x2, x0
    mov w0, #0
    cbz x1, .Ldone
.Lloop1:
    ldrb w3, [x2], #1
    cmp w0, w3
    csel w0, w0, w3, hi
    subs x1, x1, #1
    b.ne .Lloop1
.Ldone:
    ret
