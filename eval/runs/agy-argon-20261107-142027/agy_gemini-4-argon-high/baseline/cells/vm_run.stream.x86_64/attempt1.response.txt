    test rsi, rsi
    jz .Ldone_no_frame
    sub rsp, 104
    lea rax, [rip + .Ldone]
    lea r8, [rip + .L_push]
    mov qword ptr [rsp], rax
    mov qword ptr [rsp + 8], r8
    lea rax, [rip + .L_add]
    lea r8, [rip + .L_sub]
    mov qword ptr [rsp + 16], rax

    lea r8, [rip + .L_jmp]

    mov qword ptr [rsp + 88], r8

.L_push:

    mov qword ptr [r10], r9

    add qword ptr [r10 - 16], r9

    add r8, 8

    jmp qword ptr [rsp + rax*8]

.L_dup:

    add r8, 8

    je .Ldone

    jmp qword ptr [rsp + rax*8]

    cmp qword ptr [r10 - 16], r9
    setl r11b

    sub r10, 8
    add r8, 8

    je .Ldone
    mov rax, qword ptr [r8]
    jmp qword ptr [rsp + rax*8]

.L_jmp:

    je .Ldone
    mov rax, qword ptr [r8]
    jmp qword ptr [rsp + rax*8]

.Ldone_no_frame:

    cmp r10, rdx

.Lzero_tail:
    lea r8, [rdx + rcx*8]
    mov r9, r8
    sub r9, r10
    jz .Lret
    xorps xmm0, xmm0
    cmp r9, 32
    jb .Lzero_lt32
    lea r11, [r8 - 32]
    cmp r10, r11
    je .Lzero_tail32
.Lzero_loop32:

    movups xmmword ptr [r10 + 16], xmm0
    add r10, 32
    cmp r10, r11
    jb .Lzero_loop32
.Lzero_tail32:

    movups xmmword ptr [r11 + 16], xmm0
    ret
.Lzero_lt32:

    jb .Lzero_8
    movups xmmword ptr [r10], xmm0

.Lzero_8:

.Lret:
