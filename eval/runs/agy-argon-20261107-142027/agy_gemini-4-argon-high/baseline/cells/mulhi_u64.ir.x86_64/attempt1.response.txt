export function l $mulhi_u64(l %a, l %b) {
@start
    %mask =l copy 4294967295
    %a_lo =l and %a, %mask
    %a_hi =l shr %a, 32
    %b_lo =l and %b, %mask
    %b_hi =l shr %b, 32

    %p00 =l mul %a_lo, %b_lo
    %p01 =l mul %a_lo, %b_hi
    %p10 =l mul %a_hi, %b_lo
    %p11 =l mul %a_hi, %b_hi

    %p00_hi =l shr %p00, 32
    %p01_lo =l and %p01, %mask
    %p10_lo =l and %p10, %mask

    %mid1 =l add %p00_hi, %p01_lo
    %mid =l add %mid1, %p10_lo
    %carry =l shr %mid, 32

    %p01_hi =l shr %p01, 32
    %p10_hi =l shr %p10, 32

    %hi1 =l add %p11, %p01_hi
    %hi2 =l add %hi1, %p10_hi
    %hi =l add %hi2, %carry

    ret %hi
}
