export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%oa =l mul %i, 8
	%pa =l add %a, %oa
	%ob =l mul %j, 8
	%pb =l add %a, %ob
	%va =l loadl %pa
	%vb =l loadl %pb
	storel %vb, %pa
	storel %va, %pb
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
