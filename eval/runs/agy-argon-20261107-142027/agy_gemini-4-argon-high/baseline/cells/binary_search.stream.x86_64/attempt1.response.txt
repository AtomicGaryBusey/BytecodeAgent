binary_search:
    test rsi, rsi
    jle .Lnot_found
    xor ecx, ecx
.Lloop:
    lea rax, [rcx + rsi]
    shr rax, 1
    cmp rdx, qword ptr [rdi + rax*8]
    je .Lfound
    lea r8, [rax + 1]
    cmovl rsi, rax
    cmovg rcx, r8
    cmp rcx, rsi
    jl .Lloop
.Lnot_found:
    mov rax, -1
.Lfound:
    ret
