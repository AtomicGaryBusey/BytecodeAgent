binary_search:
    xor ecx, ecx
    sub rsi, 1
.Lloop:
    cmp rcx, rsi
    jg .Lnot_found
    lea rax, [rcx + rsi]
    sar rax, 1
    cmp qword ptr [rdi + rax*8], rdx
    je .Lfound
    jg .Lgreater
    lea rcx, [rax + 1]
    jmp .Lloop
.Lgreater:
    lea rsi, [rax - 1]
    jmp .Lloop
.Lnot_found:
    mov rax, -1
.Lfound:
    ret
