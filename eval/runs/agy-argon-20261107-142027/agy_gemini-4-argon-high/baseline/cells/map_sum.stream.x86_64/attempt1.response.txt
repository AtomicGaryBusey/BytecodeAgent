map_sum:
    test rsi, rsi
    jle .Lempty
    push r13
    push r12
    push rbp
    push rbx
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    xor r13d, r13d
.Lloop:
    mov rdi, qword ptr [rbx]
    call r12
    add r13, rax
    add rbx, 8
    sub rbp, 1
    jnz .Lloop
    mov rax, r13
    add rsp, 8
    pop rbx
    pop rbp
    pop r12
    pop r13
    ret
.Lempty:
    xor eax, eax
    ret
