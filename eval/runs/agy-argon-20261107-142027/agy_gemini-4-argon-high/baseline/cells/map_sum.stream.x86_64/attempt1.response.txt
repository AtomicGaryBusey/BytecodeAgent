map_sum:
    test rsi, rsi
    jle .Lzero
    push rbp
    push rbx
    push r12
    push r13
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    xor r13d, r13d
.Lloop:
    mov rdi, qword ptr [rbx]
    call r12
    add r13, rax
    add rbx, 8
    dec rbp
    jnz .Lloop
    mov rax, r13
    add rsp, 8
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
.Lzero:
    xor eax, eax
    ret
