export function l $count_ugt(l %a, l %n, l %t) {
@start
@loop
	%ptr =l phi @start %a, @body %ptr1
	%rem =l phi @start %n, @body %rem1
	%cnt =l phi @start 0, @body %cnt1
	%cond =w csgtl %rem, 0
	jnz %cond, @body, @done
@body
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%inc =l extuw %gt
	%cnt1 =l add %cnt, %inc
	%ptr1 =l add %ptr, 8
	%rem1 =l sub %rem, 1
	jmp @loop
@done
	ret %cnt
}
