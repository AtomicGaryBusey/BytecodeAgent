count_nonzero_prefix:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    mov rcx, rsi
    and rcx, -4
    je .Ltail
.Lloop4:
    cmp qword ptr [rdi + rax*8], 0
    je .Ldone
    cmp qword ptr [rdi + rax*8 + 8], 0
    je .Lfound1
    cmp qword ptr [rdi + rax*8 + 16], 0
    je .Lfound2
    cmp qword ptr [rdi + rax*8 + 24], 0
    je .Lfound3
    add rax, 4
    cmp rax, rcx
    jne .Lloop4
.Ltail:
    cmp rax, rsi
    je .Ldone
.Ltailloop:
    cmp qword ptr [rdi + rax*8], 0
    je .Ldone
    inc rax
    cmp rax, rsi
    jne .Ltailloop
.Ldone:
    ret
.Lfound1:
    inc rax
    ret
.Lfound2:
    add rax, 2
    ret
.Lfound3:
    add rax, 3
    ret
