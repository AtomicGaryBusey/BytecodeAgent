export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %j0 =l copy 0
@init_loop
    %j =l phi @start %j0, @init_body %j_next
    %cond_init =w cslel %j, %n
    jnz %cond_init, @init_body, @outer_check
@init_body
    %off_init =l mul %j, 8
    %ptr_init =l add %row, %off_init
    storel 0, %ptr_init
    %j_next =l add %j, 1
    jmp @init_loop
@outer_check
    %i0 =l copy 0
@outer_loop
    %i =l phi @outer_check %i0, @outer_next %i_next
    %cond_outer =w csltl %i, %m
    jnz %cond_outer, @outer_body, @done
@outer_body
    %xi_ptr =l add %x, %i
    %xi =w loadub %xi_ptr
    %diag0 =l copy 0
    %left0 =l copy 0
    %j1 =l copy 1
@inner_loop
    %j_in =l phi @outer_body %j1, @inner_next %j_in_next
    %diag =l phi @outer_body %diag0, @inner_next %up
    %left =l phi @outer_body %left0, @inner_next %new_val
    %cond_inner =w cslel %j_in, %n
    jnz %cond_inner, @inner_body, @outer_next
@inner_body
    %row_off =l mul %j_in, 8
    %row_ptr =l add %row, %row_off
    %up =l loadl %row_ptr
    %jm1 =l sub %j_in, 1
    %y_ptr =l add %y, %jm1
    %yj =w loadub %y_ptr
    %eq =w ceqw %xi, %yj
    jnz %eq, @match, @nomatch
@match
    %v_match =l add %diag, 1
    jmp @inner_next
@nomatch
    %gt =w csgtl %up, %left
    jnz %gt, @take_up, @take_left
@take_up
    %v_up =l copy %up
    jmp @inner_next
@take_left
    %v_left =l copy %left
    jmp @inner_next
@inner_next
    %new_val =l phi @match %v_match, @take_up %v_up, @take_left %v_left
    storel %new_val, %row_ptr
    %j_in_next =l add %j_in, 1
    jmp @inner_loop
@outer_next
    %i_next =l add %i, 1
    jmp @outer_loop
@done
    %end_off =l mul %n, 8
    %end_ptr =l add %row, %end_off
    %ans =l loadl %end_ptr
    ret %ans
}
