export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w cslel %n, 0
	jnz %empty, @done, @dir
@dir
	%is_fwd =w culel %dst, %src
	jnz %is_fwd, @fwd_loop, @bwd_init
@fwd_loop
	%fi =l phi @dir 0, @fwd_loop %fi_next
	%foff =l mul %fi, 8
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fval =l loadl %fsrc
	storel %fval, %fdst
	%fi_next =l add %fi, 1
	%fmore =w csltl %fi_next, %n
	jnz %fmore, @fwd_loop, @done
@bwd_init
	%bi0 =l sub %n, 1
	jmp @bwd_loop
@bwd_loop
	%bi =l phi @bwd_init %bi0, @bwd_loop %bi_next
	%boff =l mul %bi, 8
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	%bi_next =l sub %bi, 1
	%bmore =w csgel %bi_next, 0
	jnz %bmore, @bwd_loop, @done
@done
	ret 0
}
