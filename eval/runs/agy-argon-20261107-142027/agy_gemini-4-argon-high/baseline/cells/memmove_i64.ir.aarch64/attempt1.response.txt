export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%le0 =w cslel %n, 0
	jnz %le0, @done, @check_dir

@check_dir
	%fwd =w culel %dst, %src
	jnz %fwd, @fwd_init, @bwd_init

@fwd_init
	%i0 =l copy 0
	jmp @fwd_loop

@fwd_loop
	%i =l phi @fwd_init %i0, @fwd_body %i_next
	%cond_f =w csltl %i, %n
	jnz %cond_f, @fwd_body, @done

@fwd_body
	%off_f =l shl %i, 3
	%sp_f =l add %src, %off_f
	%dp_f =l add %dst, %off_f
	%val_f =l loadl %sp_f
	storel %val_f, %dp_f
	%i_next =l add %i, 1
	jmp @fwd_loop

@bwd_init
	%j0 =l sub %n, 1
	jmp @bwd_loop

@bwd_loop
	%j =l phi @bwd_init %j0, @bwd_body %j_next
	%cond_b =w csgel %j, 0
	jnz %cond_b, @bwd_body, @done

@bwd_body
	%off_b =l shl %j, 3
	%sp_b =l add %src, %off_b
	%dp_b =l add %dst, %off_b
	%val_b =l loadl %sp_b
	storel %val_b, %dp_b
	%j_next =l sub %j, 1
	jmp @bwd_loop

@done
	ret 0
}
