export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	%eight =l copy 8
	@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%b =w loadub %p
	%crc =w xor %crc, %b
	%j =l copy 0
	@inner
	%jmore =w csltl %j, %eight
	jnz %jmore, @bit, @next
@bit
	%odd =w and %crc, 1
	jnz %odd, @poly, @shift
@poly
	%s =w shr %crc, 1
	%crc =w xor %s, 3988292384
	jmp @inc
@shift
	%crc =w shr %crc, 1
	jmp @inc
@inc
	%j =l add %j, 1
	jmp @inner
@next
	%p =l add %p, 1
	%i =l add %i, 1
	jmp @loop
@done
	%res =w xor %crc, 4294967295
	%out =l extuw %res
	ret %out
}
