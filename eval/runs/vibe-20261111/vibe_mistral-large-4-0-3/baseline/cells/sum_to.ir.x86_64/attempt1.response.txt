export function l $sum_to(l %n) {
@start
	%one =l copy 1
	%nonpos =w csltl %n, %one
	jnz %nonpos, @zero, @init
@zero
	%z =l copy 0
	ret %z
@init
	%sum0 =l copy 0
	%i0 =l copy 1
	jmp @loop
@loop
	%sum =l phi @init %sum0, @body %sumnext
	%i =l phi @init %i0, @body %inext
	%cont =w cslel %i, %n
	jnz %cont, @body, @done
@body
	%sumnext =l add %sum, %i
	%inext =l add %i, %one
	jmp @loop
@done
	ret %sum
}
