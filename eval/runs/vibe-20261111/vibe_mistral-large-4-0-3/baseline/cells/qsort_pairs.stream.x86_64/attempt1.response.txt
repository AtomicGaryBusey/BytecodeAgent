qsort_pairs:
    cmp rsi, 2
    jl .Lret0
    push rbx
    push r12
    push r13
    mov rbx, rdi
    mov r12, rsi
.Lloop:
    cmp r12, 2
    jl .Ldone
    mov r8, r12
    dec r8
    shl r8, 4
    add r8, rbx
    mov rax, [r8]
    mov rdi, [r8+8]
    xor ecx, ecx
    mov r10, rbx
    xor edx, edx
    mov r9, rbx
.Lpartition:
    cmp rdx, r12
    jge .Lpartition_done
    mov r11, [r9]
    cmp r11, rax
    jl .Lless
    jne .Lnext
    mov r11, [r9+8]
    cmp r11, rdi
    jl .Lless
    jmp .Lnext
.Lless:
    mov r11, [r10]
    mov rsi, [r9]
    mov [r9], r11
    mov [r10], rsi
    mov r11, [r10+8]
    mov rsi, [r9+8]
    mov [r9+8], r11
    mov [r10+8], rsi
    inc rcx
    add r10, 16
.Lnext:
    inc rdx
    add r9, 16
    jmp .Lpartition
.Lpartition_done:
    mov r11, [r10]
    mov rsi, [r8]
    mov [r8], r11
    mov [r10], rsi
    mov r11, [r10+8]
    mov rsi, [r8+8]
    mov [r8+8], r11
    mov [r10+8], rsi
    mov r13, rcx
    mov rax, r12
    sub rax, r13
    dec rax
    cmp r13, 2
    jge .Lsetup_left
    cmp rax, 2
    jl .Ldone
    mov rdi, r13
    shl rdi, 4
    add rdi, rbx
    add rdi, 16
    mov rsi, rax
    jmp .Ldo_call
.Lsetup_left:
    mov rdi, rbx
    mov rsi, r13
.Ldo_call:
    call qsort_pairs
    cmp r13, 2
    jge .Lloop_right
    mov r12, r13
    jmp .Lloop
.Lloop_right:
    mov rdi, r13
    shl rdi, 4
    add rdi, rbx
    add rdi, 16
    mov rsi, r12
    sub rsi, r13
    dec rsi
    mov rbx, rdi
    mov r12, rsi
    jmp .Lloop
.Ldone:
    xor eax, eax
    pop r13
    pop r12
    pop rbx
    ret
.Lret0:
    xor eax, eax
    ret
