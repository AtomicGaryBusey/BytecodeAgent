export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%count =l phi @start 0, @body %count_next
	%done =w ceql %i, %n
	jnz %done, @return, @body
@body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%x =l loadl %srcp
	%y =l call %f(l %x)
	storel %y, %dstp
	%lt =w csltl %y, %x
	%ltl =l extsw %lt
	%count_next =l add %count, %ltl
	%i_next =l add %i, 1
	jmp @loop
@return
	ret %count
}
