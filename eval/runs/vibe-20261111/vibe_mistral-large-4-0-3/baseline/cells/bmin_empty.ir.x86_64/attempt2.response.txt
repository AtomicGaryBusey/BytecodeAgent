export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @join %inext
	%m =l phi @start 9223372036854775807, @join %mnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %m
	jnz %lt, @take, @skip
@take
	%mtake =l copy %v
	jmp @join
@skip
	jmp @join
@join
	%mnext =l phi @take %mtake, @skip %m
	%inext =l add %i, 1
	jmp @loop
@done
	ret %m
}
