export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%m =l phi @start 9223372036854775807, @body %mnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %m
	%mnext =l select %lt, %v, %m
	%inext =l add %i, 1
	jmp @loop
@done
	ret %m
}
