export function l $bubble_sort(l %a, l %n) {
@start
  %i =l sub %n, 1
  jmp @outer
@outer
  %cont =w csltl 0, %i
  jnz %cont, @outer_body, @ret
@outer_body
  %j =l copy 0
  jmp @inner
@inner
  %go =w csltl %j, %i
  jnz %go, @cmp, @outer_dec
@cmp
  %off =l mul %j, 8
  %pj =l add %a, %off
  %off2 =l add %off, 8
  %pj2 =l add %a, %off2
  %x =l loadl %pj
  %y =l loadl %pj2
  %need =w csltl %y, %x
  jnz %need, @swap, @inc
@swap
  storel %y, %pj
  storel %x, %pj2
  jmp @inc
@inc
  %jnext =l add %j, 1
  %j =l copy %jnext
  jmp @inner
@outer_dec
  %inext =l sub %i, 1
  %i =l copy %inext
  jmp @outer
@ret
  %zero =l copy 0
  ret %zero
}
