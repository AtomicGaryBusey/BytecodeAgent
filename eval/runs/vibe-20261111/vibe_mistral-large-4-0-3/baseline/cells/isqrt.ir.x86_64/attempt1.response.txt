export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
	jmp @skip
@skip
	%c =w cultl %x, %bit
	jnz %c, @shift, @main
@shift
	%bit =l shr %bit, 2
	jmp @skip
@main
	%z =w ceql %bit, 0
	jnz %z, @done, @body
@body
	%sum =l add %res, %bit
	%t =w cugel %x, %sum
	jnz %t, @take, @untake
@take
	%x =l sub %x, %sum
	%half =l shr %res, 1
	%res =l add %half, %bit
	jmp @next
@untake
	%res =l shr %res, 1
	jmp @next
@next
	%bit =l shr %bit, 2
	jmp @main
@done
	ret %res
}
