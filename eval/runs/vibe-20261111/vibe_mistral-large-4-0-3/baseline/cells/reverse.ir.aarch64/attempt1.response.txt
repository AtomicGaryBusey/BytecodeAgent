export function l $reverse(l %a, l %n) {
@start
  %i0 =l copy 0
  %j0 =l copy %n
  %one =l copy 1
  %eight =l copy 8
  j @loop
@loop
  %i =l phi @start %i0, @body %inext
  %j =l phi @start %j0, @body %jnext
  %c =w csltl %i, %j
  jnz %c, @body, @done
@body
  %io =l mul %i, %eight
  %p =l add %a, %io
  %jm =l sub %j, %one
  %jo =l mul %jm, %eight
  %q =l add %a, %jo
  %x =l loadl %p
  %y =l loadl %q
  storel %y, %p
  storel %x, %q
  %inext =l add %i, %one
  %jnext =l sub %j, %one
  j @loop
@done
  %r =l copy 0
  ret %r
}
