export function l $count_desc(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%limit =l sub %n, %one
	jmp @loop
@loop
	%p =l phi @start %a, @body %pnext
	%i =l phi @start %zero, @body %inext
	%count =l phi @start %zero, @body %countnext
	%cond =w csltl %i, %limit
	jnz %cond, @body, @done
@body
	%x =l loadl %p
	%pnext =l add %p, %eight
	%y =l loadl %pnext
	%isdesc =w csgtl %x, %y
	%inc =l select %isdesc, %one, %zero
	%countnext =l add %count, %inc
	%inext =l add %i, %one
	jmp @loop
@done
	ret %count
}
