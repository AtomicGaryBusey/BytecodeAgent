export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  %i =l copy 0
  %count =l copy 0
  %ps =l copy %src
  %pd =l copy %dst
  jmp @loop
@loop
  %cont =w csltl %i, %n
  jnz %cont, @body, @done
@body
  %x =l loadl %ps
  %y =l call %f(l %x)
  storel %y, %pd
  %lt =w csltl %y, %x
  %ltl =l extsw %lt
  %count =l add %count, %ltl
  %ps =l add %ps, 8
  %pd =l add %pd, 8
  %i =l add %i, 1
  jmp @loop
@done
  ret %count
}
