export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%b0 =w ceql %b, 0
	jnz %b0, @zero, @bnonzero
@zero
	storel %a, %rem
	ret 0
@bnonzero
	%bm1 =w ceql %b, -1
	jnz %bm1, @checkneg, @normal
@checkneg
	%aneg =w csltl %a, 0
	jnz %aneg, @negcheck, @normal
@negcheck
	%neg =l sub 0, %a
	%ismin =w ceql %neg, %a
	jnz %ismin, @overflow, @normal
@overflow
	storel 0, %rem
	ret %a
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
