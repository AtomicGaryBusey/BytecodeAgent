binary_search:
mov r8, 0
mov r9, rsi
.Lloop:
cmp r8, r9
jge .Lnot_found
mov r10, r9
sub r10, r8
shr r10, 1
add r10, r8
cmp qword ptr [rdi + r10*8], rdx
je .Lfound
jl .Lkey_greater
mov r9, r10
jmp .Lloop
.Lkey_greater:
lea r8, [r10 + 1]
jmp .Lloop
.Lfound:
mov rax, r10
ret
.Lnot_found:
mov rax, -1
ret
