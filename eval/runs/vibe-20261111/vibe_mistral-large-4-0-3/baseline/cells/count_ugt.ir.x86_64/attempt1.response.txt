export function l $count_ugt(l %a, l %n, l %t) {
@start
	%p0 =l copy %a
	%c0 =l copy 0
	%n8 =l mul %n, 8
	%end =l add %a, %n8
	jmp @loop
@loop
	%p =l phi @start %p0, @body %pnext
	%c =l phi @start %c0, @body %cnext
	%done =w ceql %p, %end
	jnz %done, @exit, @body
@body
	%x =l loadl %p
	%gt =w cugtl %x, %t
	%one =l extsw %gt
	%cnext =l add %c, %one
	%pnext =l add %p, 8
	jmp @loop
@exit
	ret %c
}
