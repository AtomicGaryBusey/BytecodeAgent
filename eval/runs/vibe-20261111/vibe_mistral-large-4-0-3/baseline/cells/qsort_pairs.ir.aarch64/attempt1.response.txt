export function l $qsort_pairs(l %a, l %n) {
@start
	%aslot =l alloc8 8
	%nslot =l alloc8 8
	%islot =l alloc8 8
	%jslot =l alloc8 8
	%rpslot =l alloc8 8
	%rnslot =l alloc8 8
	storel %a, %aslot
	storel %n, %nslot
	%n0 =l loadl %nslot
	%small =w csltl %n0, 2
	jnz %small, @ret, @init
@ret
	ret 0
@init
	%a1 =l loadl %aslot
	%n1 =l loadl %nslot
	%nm1a =l sub %n1, 1
	%poffa =l mul %nm1a, 16
	%pa =l add %a1, %poffa
	%pvapa =l add %pa, 8
	%pkey =l loadl %pa
	%pval =l loadl %pvapa
	storel 0, %islot
	storel 0, %jslot
	jmp @loop
@loop
	%jl =l loadl %jslot
	%nl =l loadl %nslot
	%nm1l =l sub %nl, 1
	%contl =w csltl %jl, %nm1l
	jnz %contl, @body, @fin
@body
	%ab =l loadl %aslot
	%ib =l loadl %islot
	%jb =l loadl %jslot
	%ioffb =l mul %ib, 16
	%joffb =l mul %jb, 16
	%pib =l add %ab, %ioffb
	%pjb =l add %ab, %joffb
	%pibv =l add %pib, 8
	%pjbv =l add %pjb, 8
	%kib =l loadl %pib
	%vib =l loadl %pibv
	%kjb =l loadl %pjb
	%vjb =l loadl %pjbv
	%lessb =w csltl %kjb, %pkey
	jnz %lessb, @swap, @checkeq
@checkeq
	%eqb =w ceql %kjb, %pkey
	jnz %eqb, @cmpval, @noswap
@cmpval
	%lessvb =w csltl %vjb, %pval
	jnz %lessvb, @swap, @noswap
@swap
	storel %kjb, %pib
	storel %vjb, %pibv
	storel %kib, %pjb
	storel %vib, %pjbv
	%inext =l add %ib, 1
	storel %inext, %islot
	jmp @incj
@noswap
	jmp @incj
@incj
	%jinc =l loadl %jslot
	%jnext =l add %jinc, 1
	storel %jnext, %jslot
	jmp @loop
@fin
	%af =l loadl %aslot
	%nf =l loadl %nslot
	%if =l loadl %islot
	%nm1f =l sub %nf, 1
	%iofff =l mul %if, 16
	%pofff =l mul %nm1f, 16
	%pif =l add %af, %iofff
	%pf =l add %af, %pofff
	%pifv =l add %pif, 8
	%pfv =l add %pf, 8
	%kif =l loadl %pif
	%vif =l loadl %pifv
	%kpf =l loadl %pf
	%vpf =l loadl %pfv
	storel %kpf, %pif
	storel %vpf, %pifv
	storel %kif, %pf
	storel %vif, %pfv
	%rightp =l add %pif, 16
	%rightn =l sub %nm1f, %if
	storel %rightp, %rpslot
	storel %rightn, %rnslot
	%leftp =l loadl %aslot
	%leftn =l loadl %islot
	%r1 =l call $qsort_pairs(l %leftp, l %leftn)
	%rp2 =l loadl %rpslot
	%rn2 =l loadl %rnslot
	%r2 =l call $qsort_pairs(l %rp2, l %rn2)
	ret 0
}
