reverse:
    mov rax, rsi
    shr rax, 1
    test rax, rax
    jz .Ldone
    lea rdx, [rdi + rsi*8 - 8]
    mov rcx, rdi
.Lloop:
    mov r8, qword ptr [rcx]
    mov r9, qword ptr [rdx]
    mov qword ptr [rcx], r9
    mov qword ptr [rdx], r8
    add rcx, 8
    sub rdx, 8
    dec rax
    jnz .Lloop
.Ldone:
    xor eax, eax
    ret
