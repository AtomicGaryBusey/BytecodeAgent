export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	jmp @loop
@loop
	%done =w ceql %pc, %len
	jnz %done, @finish, @fetch
@fetch
	%pc8 =l mul %pc, 8
	%pcaddr =l add %code, %pc8
	%op =l loadl %pcaddr
	%c0 =w ceql %op, 0
	jnz %c0, @op_halt, @chk1
@chk1
	%c1 =w ceql %op, 1
	jnz %c1, @op_push, @chk2
@chk2
	%c2 =w ceql %op, 2
	jnz %c2, @op_add, @chk3
@chk3
	%c3 =w ceql %op, 3
	jnz %c3, @op_sub, @chk4
@chk4
	%c4 =w ceql %op, 4
	jnz %c4, @op_mul, @chk5
@chk5
	%c5 =w ceql %op, 5
	jnz %c5, @op_dup, @chk6
@chk6
	%c6 =w ceql %op, 6
	jnz %c6, @op_swap, @chk7
@chk7
	%c7 =w ceql %op, 7
	jnz %c7, @op_drop, @chk8
@chk8
	%c8 =w ceql %op, 8
	jnz %c8, @op_over, @chk9
@chk9
	%c9 =w ceql %op, 9
	jnz %c9, @op_lt, @chk10
@chk10
	%c10 =w ceql %op, 10
	jnz %c10, @op_jz, @chk11
@chk11
	%c11 =w ceql %op, 11
	jnz %c11, @op_jmp, @op_other
@op_other
	%pc =l add %pc, 1
	jmp @loop
@op_halt
	jmp @finish
@op_push
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%oaddr =l add %code, %pc18
	%v =l loadl %oaddr
	%soff =l mul %sp, 8
	%saddr =l add %stack, %soff
	storel %v, %saddr
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop
@op_add
	%bm1 =l sub %sp, 1
	%boff =l mul %bm1, 8
	%baddr =l add %stack, %boff
	%b =l loadl %baddr
	%am2 =l sub %sp, 2
	%aoff =l mul %am2, 8
	%aaddr =l add %stack, %aoff
	%a =l loadl %aaddr
	%r =l add %a, %b
	storel %r, %aaddr
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_sub
	%bm1 =l sub %sp, 1
	%boff =l mul %bm1, 8
	%baddr =l add %stack, %boff
	%b =l loadl %baddr
	%am2 =l sub %sp, 2
	%aoff =l mul %am2, 8
	%aaddr =l add %stack, %aoff
	%a =l loadl %aaddr
	%r =l sub %a, %b
	storel %r, %aaddr
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_mul
	%bm1 =l sub %sp, 1
	%boff =l mul %bm1, 8
	%baddr =l add %stack, %boff
	%b =l loadl %baddr
	%am2 =l sub %sp, 2
	%aoff =l mul %am2, 8
	%aaddr =l add %stack, %aoff
	%a =l loadl %aaddr
	%r =l mul %a, %b
	storel %r, %aaddr
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_dup
	%im1 =l sub %sp, 1
	%ioff =l mul %im1, 8
	%iaddr =l add %stack, %ioff
	%v =l loadl %iaddr
	%soff =l mul %sp, 8
	%saddr =l add %stack, %soff
	storel %v, %saddr
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_swap
	%im1 =l sub %sp, 1
	%off1 =l mul %im1, 8
	%addr1 =l add %stack, %off1
	%im2 =l sub %sp, 2
	%off2 =l mul %im2, 8
	%addr2 =l add %stack, %off2
	%v1 =l loadl %addr1
	%v2 =l loadl %addr2
	storel %v1, %addr2
	storel %v2, %addr1
	%pc =l add %pc, 1
	jmp @loop
@op_drop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_over
	%im2 =l sub %sp, 2
	%ioff =l mul %im2, 8
	%iaddr =l add %stack, %ioff
	%v =l loadl %iaddr
	%soff =l mul %sp, 8
	%saddr =l add %stack, %soff
	storel %v, %saddr
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_lt
	%bm1 =l sub %sp, 1
	%boff =l mul %bm1, 8
	%baddr =l add %stack, %boff
	%b =l loadl %baddr
	%am2 =l sub %sp, 2
	%aoff =l mul %am2, 8
	%aaddr =l add %stack, %aoff
	%a =l loadl %aaddr
	%less =w csltl %a, %b
	jnz %less, @lt_one, @lt_zero
@lt_one
	storel 1, %aaddr
	jmp @lt_done
@lt_zero
	storel 0, %aaddr
	jmp @lt_done
@lt_done
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@op_jz
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%oaddr =l add %code, %pc18
	%t =l loadl %oaddr
	%vm1 =l sub %sp, 1
	%voff =l mul %vm1, 8
	%vaddr =l add %stack, %voff
	%v =l loadl %vaddr
	%sp =l sub %sp, 1
	%z =w ceql %v, 0
	jnz %z, @jz_taken, @jz_fall
@jz_taken
	%pc =l copy %t
	jmp @loop
@jz_fall
	%pc =l add %pc, 2
	jmp @loop
@op_jmp
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%oaddr =l add %code, %pc18
	%t =l loadl %oaddr
	%pc =l copy %t
	jmp @loop
@finish
	%i =l copy %sp
	jmp @zloop
@zloop
	%inrange =w csltl %i, %cap
	jnz %inrange, @zbody, @zret
@zbody
	%zoff =l mul %i, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%i =l add %i, 1
	jmp @zloop
@zret
	%zero =l copy 0
	%nonempty =w csltl %zero, %sp
	jnz %nonempty, @top, @empty
@top
	%tm1 =l sub %sp, 1
	%toff =l mul %tm1, 8
	%taddr =l add %stack, %toff
	%res =l loadl %taddr
	ret %res
@empty
	%res0 =l copy 0
	ret %res0
}
