export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%cnt0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @body %inext
	%mn =l phi @start %y0, @body %mnnext
	%mx =l phi @start %y0, @body %mxnext
	%sum =l phi @start %y0, @body %sumnext
	%cnt =l phi @start %cnt0, @body %cntnext
	%ge =w csgel %i, %n
	jnz %ge, @done, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%ai =l loadl %p
	%y =l call %f(l %ai)
	%sumnext =l add %sum, %y
	%andy =l and %y, 1
	%cntnext =l add %cnt, %andy
	%ltw =w csltl %y, %mn
	%lt =l extsw %ltw
	%mlt =l sub 0, %lt
	%x1 =l xor %mn, %y
	%a1 =l and %x1, %mlt
	%mnnext =l xor %mn, %a1
	%gtw =w csgtl %y, %mx
	%gt =l extsw %gtw
	%mgt =l sub 0, %gt
	%x2 =l xor %mx, %y
	%a2 =l and %x2, %mgt
	%mxnext =l xor %mx, %a2
	%inext =l add %i, 1
	jmp @loop
@done
	%o1 =l add %out, 8
	%o2 =l add %out, 16
	storel %mn, %out
	storel %mx, %o1
	storel %sum, %o2
	ret %cnt
}
