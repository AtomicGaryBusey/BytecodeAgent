export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
  %zero =l 0
  %one =l 1
  %nn =l mul %n, %n
  %two_nn =l mul %nn, 2
  jmp @init_zero

@init_zero
  %zidx =l phi @entry 0, @init_zero_body %zidx_next
  %zcond =w csltl %zidx, %nn
  jnz %zcond, @init_zero_body, @init_diag

@init_zero_body
  %zoff0 =l mul %zidx, 8
  %zp0 =l add %out, %zoff0
  storel %zero, %zp0
  %zidx_next =l add %zidx, 1
  jmp @init_zero

@init_diag
  %didx =l phi @init_zero 0, @init_diag_body %didx_next
  %dcond =w csltl %didx, %n
  jnz %dcond, @init_diag_body, @main

@init_diag_body
  %drow =l mul %didx, %n
  %del =l add %drow, %didx
  %doff =l mul %del, 8
  %dp =l add %out, %doff
  storel %one, %dp
  %didx_next =l add %didx, 1
  jmp @init_diag

@main
  %cnt =l phi @init_diag 0, @mul_end %cnt_next
  %kcond =w csltl %cnt, %k
  jnz %kcond, @mul, @zero_tmp

@mul
  %i =l phi @main 0, @mul_i_end %i_next
  %icond =w csltl %i, %n
  jnz %icond, @mul_i_body, @copy

@mul_i_body
  %irow =l mul %i, %n
  jmp @mul_j

@mul_j
  %j =l phi @mul_i_body 0, @mul_j_end %j_next
  %jcond =w csltl %j, %n
  jnz %jcond, @mul_t, @mul_i_end

@mul_t
  %t =l phi @mul_j 0, @mul_t_body %t_next
  %sum =l phi @mul_j 0, @mul_t_body %sum_next
  %tcond =w csltl %t, %n
  jnz %tcond, @mul_t_body, @mul_j_end

@mul_t_body
  %out_el =l add %irow, %t
  %out_off =l mul %out_el, 8
  %out_p =l add %out, %out_off
  %xv =l loadl %out_p
  %trow =l mul %t, %n
  %a_el =l add %trow, %j
  %a_off =l mul %a_el, 8
  %a_p =l add %a, %a_off
  %yv =l loadl %a_p
  %prod =l mul %xv, %yv
  %sum_next =l add %sum, %prod
  %t_next =l add %t, 1
  jmp @mul_t

@mul_j_end
  %s_el =l add %irow, %j
  %s_off =l mul %s_el, 8
  %s_p =l add %tmp, %s_off
  storel %sum, %s_p
  %j_next =l add %j, 1
  jmp @mul_j

@mul_i_end
  %i_next =l add %i, 1
  jmp @mul

@copy
  %c =l phi @mul 0, @copy_body %c_next
  %ccond =w csltl %c, %nn
  jnz %ccond, @copy_body, @mul_end

@copy_body
  %coff =l mul %c, 8
  %src_p =l add %tmp, %coff
  %v =l loadl %src_p
  %dst_p =l add %out, %coff
  storel %v, %dst_p
  %c_next =l add %c, 1
  jmp @copy

@mul_end
  %cnt_next =l add %cnt, 1
  jmp @main

@zero_tmp
  %z =l phi @main 0, @zero_tmp_body %z_next
  %zcond2 =w csltl %z, %two_nn
  jnz %zcond2, @zero_tmp_body, @ret

@zero_tmp_body
  %zoff =l mul %z, 8
  %zp =l add %tmp, %zoff
  storel %zero, %zp
  %z_next =l add %z, 1
  jmp @zero_tmp

@ret
  ret 0
}
