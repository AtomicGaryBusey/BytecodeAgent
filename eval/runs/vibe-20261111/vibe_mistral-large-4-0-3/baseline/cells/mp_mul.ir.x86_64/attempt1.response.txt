export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
  %zero =l copy 0
  %one =l copy 1
  %eight =l copy 8
  %mask =l copy 4294967295
  %n =l add %na, %nb
  jmp @zero

@zero
  %zi =l phi @entry %zero, @zerobody %zi_next
  %zp =l phi @entry %r, @zerobody %zp_next
  %zcond =w cultl %zi, %n
  jnz %zcond, @zerobody, @outer

@zerobody
  storel %zero, %zp
  %zi_next =l add %zi, %one
  %zp_next =l add %zp, %eight
  jmp @zero

@outer
  %i =l phi @zero %zero, @inner_done %i_next
  %ocond =w cultl %i, %na
  jnz %ocond, @outer_body, @scan

@outer_body
  %i8 =l shl %i, 3
  %aip =l add %a, %i8
  %ai =l loadl %aip
  %r_plus_i8 =l add %r, %i8
  jmp @inner

@inner
  %j =l phi @outer_body %zero, @inner_body %j_next
  %carry =l phi @outer_body %zero, @inner_body %carry_next2
  %kp =l phi @outer_body %r_plus_i8, @inner_body %kp_next
  %jcond =w cultl %j, %nb
  jnz %jcond, @inner_body, @inner_done

@inner_body
  %j8 =l shl %j, 3
  %bjp =l add %b, %j8
  %bj =l loadl %bjp
  %rk =l loadl %kp
  %x0 =l and %ai, %mask
  %x1 =l shr %ai, 32
  %y0 =l and %bj, %mask
  %y1 =l shr %bj, 32
  %p00 =l mul %x0, %y0
  %p01 =l mul %x0, %y1
  %p10 =l mul %x1, %y0
  %p11 =l mul %x1, %y1
  %p01lo =l and %p01, %mask
  %p01los =l shl %p01lo, 32
  %t =l add %p00, %p01los
  %c1 =w cultl %t, %p00
  %c1l =l extuw %c1
  %p10lo =l and %p10, %mask
  %p10los =l shl %p10lo, 32
  %lo =l add %t, %p10los
  %c2 =w cultl %lo, %t
  %c2l =l extuw %c2
  %p01hi =l shr %p01, 32
  %hi1 =l add %p11, %p01hi
  %p10hi =l shr %p10, 32
  %hi2 =l add %hi1, %p10hi
  %hi3 =l add %hi2, %c1l
  %hi4 =l add %hi3, %c2l
  %s1 =l add %lo, %rk
  %d1 =w cultl %s1, %lo
  %d1l =l extuw %d1
  %s2 =l add %s1, %carry
  %d2 =w cultl %s2, %s1
  %d2l =l extuw %d2
  storel %s2, %kp
  %carry_next =l add %hi4, %d1l
  %carry_next2 =l add %carry_next, %d2l
  %j_next =l add %j, %one
  %kp_next =l add %kp, %eight
  jmp @inner

@inner_done
  storel %carry, %kp
  %i_next =l add %i, %one
  jmp @outer

@scan
  %idx =l phi @outer %n, @scan_next %idx_next
  %pos =w csltl %zero, %idx
  jnz %pos, @scan_body, @scan_done

@scan_body
  %idxm1 =l sub %idx, %one
  %idxm8 =l shl %idxm1, 3
  %ptr =l add %r, %idxm8
  %limb =l loadl %ptr
  %nz =w cnel %limb, %zero
  jnz %nz, @scan_found, @scan_next

@scan_next
  %idx_next =l sub %idx, %one
  jmp @scan

@scan_found
  ret %idx

@scan_done
  ret %zero
}
