export function l $shift_sat(l %x, l %n) {
@start
  %c64 =l copy 64
  %ge =w csgel %n, %c64
  jnz %ge, @big, @small
@big
  %zero =l copy 0
  %isneg =w csltl %x, %zero
  jnz %isneg, @neg, @nonneg
@nonneg
  ret %zero
@neg
  %m1 =l copy -1
  ret %m1
@small
  %l =l shl %x, %n
  %r =l sar %x, %n
  %sum =l add %l, %r
  ret %sum
}
