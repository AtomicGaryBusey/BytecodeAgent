export function l $gcd(l %a, l %b) {
@start
	%pa =l alloc8 8
	%pb =l alloc8 8
	storel %a, %pa
	storel %b, %pb
	jmp @loop
@loop
	%bcur =l loadl %pb
	%z =w ceql %bcur, 0
	jnz %z, @end, @body
@body
	%acur =l loadl %pa
	%t =l rem %acur, %bcur
	storel %bcur, %pa
	storel %t, %pb
	jmp @loop
@end
	%res =l loadl %pa
	ret %res
}
