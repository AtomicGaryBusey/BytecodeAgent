export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bzero =w ceql %b, 0
	jnz %bzero, @zero, @nonzero
@zero
	storel %a, %rem
	ret 0
@nonzero
	%bneg1 =w ceql %b, -1
	jnz %bneg1, @checkmin, @normal
@checkmin
	%min =l sub -9223372036854775807, 1
	%amin =w ceql %a, %min
	jnz %amin, @overflow, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@overflow
	storel 0, %rem
	ret %min
}
