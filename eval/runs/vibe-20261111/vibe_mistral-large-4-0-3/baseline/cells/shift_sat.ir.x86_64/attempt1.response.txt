export function l $shift_sat(l %x, l %n) {
@start
  %px =l alloc8 8
  %pn =l alloc8 8
  %pl =l alloc8 8
  %pr =l alloc8 8
  storel %x, %px
  storel %n, %pn
  storel %x, %pl
  storel %x, %pr
  jmp @loop
@loop
  %i =l loadl %pn
  %done =w ceql %i, 0
  jnz %done, @end, @body
@body
  %l0 =l loadl %pl
  %l1 =l shl %l0, 1
  storel %l1, %pl
  %r0 =l loadl %pr
  %r1 =l sar %r0, 1
  storel %r1, %pr
  %i1 =l sub %i, 1
  storel %i1, %pn
  jmp @loop
@end
  %l =l loadl %pl
  %r =l loadl %pr
  %res =l add %l, %r
  ret %res
}
