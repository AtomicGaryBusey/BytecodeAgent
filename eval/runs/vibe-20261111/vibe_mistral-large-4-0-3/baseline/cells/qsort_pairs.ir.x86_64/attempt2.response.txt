export function l $qsort_pairs(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret, @sort
@ret
	ret 0
@sort
	%nm1 =l sub %n, 1
	%pvb =l mul %nm1, 16
	%pkp =l add %a, %pvb
	%pvp =l add %pkp, 8
	%pkey =l loadl %pkp
	%pval =l loadl %pvp
	%zero =l sub %n, %n
	%j0 =l add 0, 0
	%i0 =l sub 0, 1
	jmp @loop
@loop
	%j =l phi @sort %j0, @swap %jnext, @noswap %jnext2
	%i =l phi @sort %i0, @swap %iinc, @noswap %ikeep
	%more =w csltl %j, %nm1
	jnz %more, @body, @place
@body
	%jb =l mul %j, 16
	%ajp =l add %a, %jb
	%ajv =l add %ajp, 8
	%kj =l loadl %ajp
	%vj =l loadl %ajv
	%ltkey =w csltl %kj, %pkey
	%eqkey =w ceql %kj, %pkey
	%ltval =w csltl %vj, %pval
	%ltboth =w and %eqkey, %ltval
	%less =w or %ltkey, %ltboth
	jnz %less, @swap, @noswap
@swap
	%iinc =l add %i, 1
	%ib =l mul %iinc, 16
	%aip =l add %a, %ib
	%aiv =l add %aip, 8
	%ki =l loadl %aip
	%vi =l loadl %aiv
	storel %kj, %aip
	storel %vj, %aiv
	storel %ki, %ajp
	storel %vi, %ajv
	%jnext =l add %j, 1
	jmp @loop
@noswap
	%ikeep =l add %i, 0
	%jnext2 =l add %j, 1
	jmp @loop
@place
	%ip1 =l add %i, 1
	%ipb =l mul %ip1, 16
	%aipp =l add %a, %ipb
	%aipv =l add %aipp, 8
	%kt =l loadl %aipp
	%vt =l loadl %aipv
	storel %pkey, %aipp
	storel %pval, %aipv
	%nb =l mul %nm1, 16
	%anp =l add %a, %nb
	%anv =l add %anp, 8
	storel %kt, %anp
	storel %vt, %anv
	%hasleft =w csltl %zero, %ip1
	jnz %hasleft, @left, @noleft
@left
	%rl =l call $qsort_pairs(l %a, l %ip1)
	jmp @right
@noleft
	jmp @right
@right
	%rs =l add %ip1, 1
	%rc =l sub %nm1, %ip1
	%hasright =w csltl %zero, %rc
	jnz %hasright, @doright, @done
@doright
	%rb =l mul %rs, 16
	%rap =l add %a, %rb
	%rr =l call $qsort_pairs(l %rap, l %rc)
	jmp @done
@done
	ret 0
}
