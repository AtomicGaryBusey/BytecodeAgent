export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %k128 =l copy 128
    %neg128 =l sub %zero, %k128
    %m1 =l sub %zero, %one
    %m2 =l sub %zero, %two
    jmp @loop

@loop
    %i =l phi @start %zero, @cont %i_next
    %o =l phi @start %zero, @cont %o_next
    %more =w csltl %i, %n
    jnz %more, @hdr, @done

@done
    ret %o

@hdr
    %hp =l add %src, %i
    %hwb =w loadsb %hp
    %h =l extsw %hwb
    %hneg =w csltl %h, %zero
    jnz %hneg, @neg, @lit

@lit
    %lcnt =l add %h, %one
    %li1 =l add %i, %one
    %lneed =l add %li1, %lcnt
    %lsrcok =w cslel %lneed, %n
    jnz %lsrcok, @lit_ok, @err2

@err2
    ret %m2

@lit_ok
    %lrem =l sub %cap, %o
    %lfits =w cslel %lcnt, %lrem
    jnz %lfits, @lit_all, @lit_over

@lit_all
    jmp @lit_copy

@lit_copy
    %lk =l phi @lit_all %zero, @lit_body %lk_next
    %lkdone =w csltl %lk, %lcnt
    jnz %lkdone, @lit_body, @lit_done

@lit_body
    %lsrc =l add %i, %one
    %lsrck =l add %lsrc, %lk
    %lb =w loadub %lsrck
    %ldst =l add %out, %o
    %ldstk =l add %ldst, %lk
    storeb %lb, %ldstk
    %lk_next =l add %lk, %one
    jmp @lit_copy

@lit_done
    %li1b =l add %i, %one
    %i_lit_next =l add %li1b, %lcnt
    %o_lit_next =l add %o, %lcnt
    jmp @cont

@lit_over
    jmp @lit_ocopy

@lit_ocopy
    %lok =l phi @lit_over %zero, @lit_obody %lok_next
    %lodone =w csltl %lok, %lrem
    jnz %lodone, @lit_obody, @lit_oret

@lit_obody
    %losrc =l add %i, %one
    %losrck =l add %losrc, %lok
    %lob =w loadub %losrck
    %lodst =l add %out, %o
    %lodstk =l add %lodst, %lok
    storeb %lob, %lodstk
    %lok_next =l add %lok, %one
    jmp @lit_ocopy

@lit_oret
    ret %m1

@neg
    %is128 =w ceql %h, %neg128
    jnz %is128, @nop_done, @rep

@nop_done
    %i_nop_next =l add %i, %one
    %o_nop_next =l copy %o
    jmp @cont

@rep
    %rneed =l add %i, %two
    %rsrcok =w cslel %rneed, %n
    jnz %rsrcok, @rep_ok, @err2

@rep_ok
    %rsrc =l add %src, %i
    %rsrc1 =l add %rsrc, %one
    %rb =w loadub %rsrc1
    %rcnt =l sub %one, %h
    %rrem =l sub %cap, %o
    %rfit =w cslel %rcnt, %rrem
    jnz %rfit, @rep_all, @rep_over

@rep_all
    jmp @rep_copy

@rep_copy
    %rk =l phi @rep_all %zero, @rep_body %rk_next
    %rkdone =w csltl %rk, %rcnt
    jnz %rkdone, @rep_body, @rep_done

@rep_body
    %rdst =l add %out, %o
    %rdstk =l add %rdst, %rk
    storeb %rb, %rdstk
    %rk_next =l add %rk, %one
    jmp @rep_copy

@rep_done
    %i_rep_next =l add %i, %two
    %o_rep_next =l add %o, %rcnt
    jmp @cont

@rep_over
    jmp @rep_ocopy

@rep_ocopy
    %rok =l phi @rep_over %zero, @rep_obody %rok_next
    %rodone =w csltl %rok, %rrem
    jnz %rodone, @rep_obody, @rep_oret

@rep_obody
    %rodst =l add %out, %o
    %rodstk =l add %rodst, %rok
    storeb %rb, %rodstk
    %rok_next =l add %rok, %one
    jmp @rep_ocopy

@rep_oret
    ret %m1

@cont
    %i_next =l phi @lit_done %i_lit_next, @rep_done %i_rep_next, @nop_done %i_nop_next
    %o_next =l phi @lit_done %o_lit_next, @rep_done %o_rep_next, @nop_done %o_nop_next
    jmp @loop
}
