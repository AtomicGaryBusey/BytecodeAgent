export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%fr =l alloc16 64
	%pi =l add %fr, 0
	%pw =l add %fr, 8
	%pneed =l add %fr, 16
	%pcnt =l add %fr, 24
	%pkind =l add %fr, 32
	%pk =l add %fr, 40
	%povf =l add %fr, 48
	%pj =l add %fr, 56
	%z0 =l copy 0
	storel %z0, %pi
	storel %z0, %pw
	jmp @loop

@loop
	%i_l =l loadl %pi
	%w_l =l loadl %pw
	%more =w csltl %i_l, %n
	jnz %more, @packet, @done

@done
	%res_done =l loadl %pw
	ret %res_done

@packet
	%i_p =l loadl %pi
	%srcp =l add %src, %i_p
	%h =w loadsb %srcp
	%is_lit =w csgew %h, 0
	jnz %is_lit, @lit, @negs

@negs
	%is_nop =w ceqw %h, -128
	jnz %is_nop, @nop, @rep

@lit
	%hl_lit =l extsw %h
	%need_lit =l add %hl_lit, 1
	%kind_lit =l copy 1
	storel %need_lit, %pneed
	storel %need_lit, %pcnt
	storel %kind_lit, %pkind
	jmp @srccheck

@rep
	%hl_rep =l extsw %h
	%cnt_rep =l sub 1, %hl_rep
	%need_rep =l copy 1
	%kind_rep =l copy 2
	storel %need_rep, %pneed
	storel %cnt_rep, %pcnt
	storel %kind_rep, %pkind
	jmp @srccheck

@nop
	%need_nop =l copy 0
	%cnt_nop =l copy 0
	%kind_nop =l copy 0
	storel %need_nop, %pneed
	storel %cnt_nop, %pcnt
	storel %kind_nop, %pkind
	jmp @srccheck

@srccheck
	%i_s =l loadl %pi
	%need_s =l loadl %pneed
	%tot_s =l add %i_s, 1
	%tot2_s =l add %tot_s, %need_s
	%bad =w csgtl %tot2_s, %n
	jnz %bad, @err2, @outcheck

@err2
	%m2 =l copy -2
	ret %m2

@outcheck
	%w_o =l loadl %pw
	%cnt_o =l loadl %pcnt
	%rem_o =l sub %cap, %w_o
	%ov =w csgtl %cnt_o, %rem_o
	jnz %ov, @setovf, @setnorm

@setnorm
	storel %cnt_o, %pk
	%z_norm =l copy 0
	storel %z_norm, %povf
	jmp @dowrite

@setovf
	storel %rem_o, %pk
	%o_ovf =l copy 1
	storel %o_ovf, %povf
	jmp @dowrite

@dowrite
	%kind_d =l loadl %pkind
	%is_nopd =w ceql %kind_d, 0
	jnz %is_nopd, @finish, @notnop

@notnop
	%is_litd =w ceql %kind_d, 1
	jnz %is_litd, @litcopy, @repcopy

@litcopy
	%z_lc =l copy 0
	storel %z_lc, %pj
	jmp @litloop

@litloop
	%j_l =l loadl %pj
	%k_l =l loadl %pk
	%cont_l =w csltl %j_l, %k_l
	jnz %cont_l, @litbody, @finish

@litbody
	%i_b =l loadl %pi
	%w_b =l loadl %pw
	%srcd_b =l add %src, %i_b
	%srcd2_b =l add %srcd_b, 1
	%srcj_b =l add %srcd2_b, %j_l
	%v_b =w loadub %srcj_b
	%dst_b =l add %out, %w_b
	%dstj_b =l add %dst_b, %j_l
	storeb %v_b, %dstj_b
	%j1_b =l add %j_l, 1
	storel %j1_b, %pj
	jmp @litloop

@repcopy
	%z_rc =l copy 0
	storel %z_rc, %pj
	jmp @reploop

@reploop
	%j_r =l loadl %pj
	%k_r =l loadl %pk
	%cont_r =w csltl %j_r, %k_r
	jnz %cont_r, @repbody, @finish

@repbody
	%i_r =l loadl %pi
	%w_r =l loadl %pw
	%srcd_r =l add %src, %i_r
	%srcd2_r =l add %srcd_r, 1
	%v_r =w loadub %srcd2_r
	%dst_r =l add %out, %w_r
	%dstj_r =l add %dst_r, %j_r
	storeb %v_r, %dstj_r
	%j1_r =l add %j_r, 1
	storel %j1_r, %pj
	jmp @reploop

@finish
	%i_f =l loadl %pi
	%need_f =l loadl %pneed
	%w_f =l loadl %pw
	%k_f =l loadl %pk
	%adv_f =l add %need_f, 1
	%inew_f =l add %i_f, %adv_f
	%wnew_f =l add %w_f, %k_f
	storel %inew_f, %pi
	storel %wnew_f, %pw
	%ovf_f =l loadl %povf
	%is_ovf =w ceql %ovf_f, 1
	jnz %is_ovf, @err1, @loop

@err1
	%m1 =l copy -1
	ret %m1
}
