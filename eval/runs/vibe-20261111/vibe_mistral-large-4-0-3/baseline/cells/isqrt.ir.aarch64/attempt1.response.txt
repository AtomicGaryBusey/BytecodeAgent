export function l $isqrt(l %n) {
@start
  %lo =l copy 0
  %hi =l copy 4294967296
@loop
  %c =w cultl %lo, %hi
  jnz %c, @body, @done
@body
  %diff =l sub %hi, %lo
  %half =l shrl %diff, 1
  %mid =l add %lo, %half
  %sq =l mul %mid, %mid
  %ok =w culel %sq, %n
  jnz %ok, @then, @else
@then
  %lo =l add %mid, 1
  jmp @loop
@else
  %hi =l copy %mid
  jmp @loop
@done
  %res =l sub %lo, 1
  ret %res
}
