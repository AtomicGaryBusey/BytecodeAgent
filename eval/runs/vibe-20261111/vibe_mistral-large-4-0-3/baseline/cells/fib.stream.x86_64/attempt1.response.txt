fib:
    test rdi, rdi
    jnz .Lnonzero
    xor eax, eax
    ret
.Lnonzero:
    cmp rdi, 1
    jne .Literate
    mov eax, 1
    ret
.Literate:
    xor eax, eax
    mov rdx, 1
    mov rcx, rdi
    dec rcx
.Lloop:
    mov r8, rdx
    add r8, rax
    mov rax, rdx
    mov rdx, r8
    dec rcx
    jnz .Lloop
    mov rax, rdx
    ret
