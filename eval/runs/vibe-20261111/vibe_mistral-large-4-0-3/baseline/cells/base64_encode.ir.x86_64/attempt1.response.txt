export function l $base64_encode(l %src, l %n, l %out) {
@start
  %k0 =l copy 0
  %k1 =l copy 1
  %k2 =l copy 2
  %k3 =l copy 3
  %k4 =l copy 4
  %k26 =l copy 26
  %k52 =l copy 52
  %k62 =l copy 62
  %sbase =l alloc8 8
  %so =l alloc8 8
  %sr =l alloc8 8
  %sb0 =l alloc8 8
  %sb1 =l alloc8 8
  %sb2 =l alloc8 8
  %sv =l alloc8 8
  %sc =l alloc8 8
  storel %k0, %sbase
  storel %k0, %so
  storel %k0, %sr
  %np2 =l add %n, 2
  %groups =l div %np2, %k3
  %outlen =l mul %groups, %k4
  jmp @outer
@outer
  %base_o =l loadl %sbase
  %cont =w csltl %base_o, %n
  jnz %cont, @body, @done
@body
  %base_b =l loadl %sbase
  %a0 =l add %src, %base_b
  %b0 =l loadub %a0
  storel %b0, %sb0
  %off1 =l add %base_b, %k1
  %has1 =w csltl %off1, %n
  jnz %has1, @load1, @skip1
@load1
  %a1 =l add %a0, %k1
  %t1 =l loadub %a1
  storel %t1, %sb1
  jmp @after1
@skip1
  storel %k0, %sb1
  jmp @after1
@after1
  %off2 =l add %base_b, %k2
  %has2 =w csltl %off2, %n
  jnz %has2, @load2, @skip2
@load2
  %a2 =l add %a0, %k2
  %t2 =l loadub %a2
  storel %t2, %sb2
  jmp @after2
@skip2
  storel %k0, %sb2
  jmp @after2
@after2
  storel %k0, %sr
  jmp @rloop
@rloop
  %r_rl =l loadl %sr
  %rcont =w csltl %r_rl, %k4
  jnz %rcont, @rbody, @rend
@rbody
  %base_r =l loadl %sbase
  %rem =l sub %n, %base_r
  %r =l loadl %sr
  %rem1 =w ceql %rem, %k1
  %rge2 =w cslel %k2, %r
  %padA =w and %rem1, %rge2
  %rem2 =w ceql %rem, %k2
  %req3 =w ceql %r, %k3
  %padB =w and %rem2, %req3
  %ispad =w or %padA, %padB
  %b0r =l loadl %sb0
  %b1r =l loadl %sb1
  %b2r =l loadl %sb2
  %ris0 =w ceql %r, %k0
  jnz %ris0, @v0, @chk1
@chk1
  %ris1 =w ceql %r, %k1
  jnz %ris1, @v1, @chk2
@chk2
  %ris2 =w ceql %r, %k2
  jnz %ris2, @v2, @v3
@v0
  %v0 =l shr %b0r, 2
  storel %v0, %sv
  jmp @decide
@v1
  %lo1 =l and %b0r, 3
  %sh1 =l shl %lo1, 4
  %hi1 =l shr %b1r, 4
  %v1 =l or %sh1, %hi1
  storel %v1, %sv
  jmp @decide
@v2
  %lo2 =l and %b1r, 15
  %sh2 =l shl %lo2, 2
  %hi2 =l shr %b2r, 6
  %v2 =l or %sh2, %hi2
  storel %v2, %sv
  jmp @decide
@v3
  %v3 =l and %b2r, 63
  storel %v3, %sv
  jmp @decide
@decide
  jnz %ispad, @pad, @mapblock
@pad
  %eqc =l copy 61
  storel %eqc, %sc
  jmp @store
@mapblock
  %v =l loadl %sv
  %lt26 =w csltl %v, %k26
  %m1 =l extsw %lt26
  %ge26 =w cslel %k26, %v
  %lt52 =w csltl %v, %k52
  %b2m =w and %ge26, %lt52
  %m2 =l extsw %b2m
  %ge52 =w cslel %k52, %v
  %lt62 =w csltl %v, %k62
  %b3m =w and %ge52, %lt62
  %m3 =l extsw %b3m
  %ge62 =w cslel %k62, %v
  %m4 =l extsw %ge62
  %t1 =l add %v, 65
  %p1 =l mul %t1, %m1
  %t2 =l add %v, 71
  %p2 =l mul %t2, %m2
  %t3 =l sub %v, 4
  %p3 =l mul %t3, %m3
  %d =l sub %v, %k62
  %d4 =l mul %d, 4
  %t4 =l add %d4, 43
  %p4 =l mul %t4, %m4
  %s1 =l add %p1, %p2
  %s2 =l add %s1, %p3
  %ch =l add %s2, %p4
  storel %ch, %sc
  jmp @store
@store
  %ch_s =l loadl %sc
  %op_s =l loadl %so
  %r_s =l loadl %sr
  %addr =l add %out, %op_s
  %addr_r =l add %addr, %r_s
  storeb %ch_s, %addr_r
  %rnext =l add %r_s, %k1
  storel %rnext, %sr
  jmp @rloop
@rend
  %op_r =l loadl %so
  %opn =l add %op_r, %k4
  storel %opn, %so
  %base_r2 =l loadl %sbase
  %basen =l add %base_r2, %k3
  storel %basen, %sbase
  jmp @outer
@done
  ret %outlen
}
