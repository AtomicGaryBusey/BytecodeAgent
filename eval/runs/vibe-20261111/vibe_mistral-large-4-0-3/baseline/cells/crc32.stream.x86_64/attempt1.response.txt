crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
    xor rcx, rcx
.Louter:
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit:
    mov r9d, eax
    and r9d, 1
    shr eax, 1
    test r9d, r9d
    jz .Lnoxor
    xor eax, 0xEDB88320
.Lnoxor:
    dec r8d
    jnz .Lbit
    inc rcx
    cmp rcx, rsi
    jb .Louter
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
