export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%lt =w cultl %dst, %src
	jnz %lt, @fwd, @bwd
@fwd
	%fi =l copy %zero
	jmp @floop
@floop
	%fcond =w csltl %fi, %n
	jnz %fcond, @fbody, @done
@fbody
	%foff =l mul %fi, 8
	%fps =l add %src, %foff
	%fpd =l add %dst, %foff
	%fv =l loadl %fps
	storel %fv, %fpd
	%fi =l add %fi, %one
	jmp @floop
@bwd
	%bi =l copy %n
	jmp @bloop
@bloop
	%bpos =w csgtl %bi, %zero
	jnz %bpos, @bbody, @done
@bbody
	%bim1 =l sub %bi, %one
	%boff =l mul %bim1, 8
	%bps =l add %src, %boff
	%bpd =l add %dst, %boff
	%bv =l loadl %bps
	storel %bv, %bpd
	%bi =l copy %bim1
	jmp @bloop
@done
	ret %zero
}
