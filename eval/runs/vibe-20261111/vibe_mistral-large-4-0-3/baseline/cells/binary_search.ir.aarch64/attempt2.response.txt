export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	%hi0 =l copy %n
	jmp @loop
@loop
	%lo =l phi @start %lo0, @lower %next_lo, @upper %upper_lo
	%hi =l phi @start %hi0, @lower %lower_hi, @upper %upper_hi
	%more =w csltl %lo, %hi
	jnz %more, @probe, @not_found
@probe
	%sum =l add %lo, %hi
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @compare
@compare
	%less =w csltl %v, %key
	jnz %less, @lower, @upper
@lower
	%next_lo =l add %mid, 1
	%lower_hi =l copy %hi
	jmp @loop
@upper
	%upper_lo =l copy %lo
	%upper_hi =l copy %mid
	jmp @loop
@found
	ret %mid
@not_found
	%neg1 =l copy -1
	ret %neg1
}
