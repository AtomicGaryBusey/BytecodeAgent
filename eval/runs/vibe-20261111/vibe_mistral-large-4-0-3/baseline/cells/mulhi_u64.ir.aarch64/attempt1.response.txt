export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p =l mul %alo, %bhi
	%q =l mul %ahi, %blo
	%d =l mul %alo, %blo
	%r =l mul %ahi, %bhi
	%pl =l and %p, 4294967295
	%ql =l and %q, 4294967295
	%lsum =l add %pl, %ql
	%carrylow =l shr %lsum, 32
	%clow =l and %lsum, 4294967295
	%phigh =l shr %p, 32
	%qhigh =l shr %q, 32
	%chigh =l add %phigh, %qhigh
	%chigh2 =l add %chigh, %carrylow
	%dhigh =l shr %d, 32
	%sum =l add %clow, %dhigh
	%carry =l shr %sum, 32
	%tmp =l add %r, %chigh2
	%res =l add %tmp, %carry
	ret %res
}
