fib_rec:
    stp x29, x30, [sp, #-32]!
    stp x19, x20, [sp, #16]
    mov x29, sp
    mov x19, x0
    cbz x19, .Lfib0
    cmp x19, #1
    b.eq .Lfib1
    sub x0, x19, #1
    bl fib_rec
    mov x20, x0
    sub x0, x19, #2
    bl fib_rec
    add x0, x20, x0
    b .Lfib_done
.Lfib0:
    mov x0, #0
    b .Lfib_done
.Lfib1:
    mov x0, #1
.Lfib_done:
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret
