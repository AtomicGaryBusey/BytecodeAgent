export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @done
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %key
	jnz %lt, @less, @ge
@less
	%lo =l add %mid, 1
	jmp @loop
@ge
	%eq =w ceql %v, %key
	jnz %eq, @found, @greater
@greater
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@done
	%neg =l copy -1
	ret %neg
}
