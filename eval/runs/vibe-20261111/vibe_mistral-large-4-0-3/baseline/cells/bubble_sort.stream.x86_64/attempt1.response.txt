bubble_sort:
    cmp rsi, 1
    jle .Ldone
    mov r8, rsi
    dec r8
.Louter:
    xor ecx, ecx
.Linner:
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rax, rdx
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rcx*8 + 8], rax
.Lnoswap:
    inc rcx
    cmp rcx, r8
    jl .Linner
    dec r8
    jne .Louter
.Ldone:
    xor eax, eax
    ret
