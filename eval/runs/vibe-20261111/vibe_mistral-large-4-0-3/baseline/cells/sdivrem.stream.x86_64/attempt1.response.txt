sdivrem:
  mov r10, rdx
  mov rax, rdi
  test rsi, rsi
  je .Lzero
  cmp rsi, -1
  jne .Ldivide
  mov r11, 0x8000000000000000
  cmp rax, r11
  je .Loverflow
.Ldivide:
  cqo
  idiv rsi
  mov qword ptr [r10], rdx
  ret
.Lzero:
  mov qword ptr [r10], rax
  xor eax, eax
  ret
.Loverflow:
  mov qword ptr [r10], 0
  mov rax, 0x8000000000000000
  ret
