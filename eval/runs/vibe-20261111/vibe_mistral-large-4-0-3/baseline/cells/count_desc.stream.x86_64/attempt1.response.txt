count_desc:
    cmp rsi, 2
    jb .Lzero
    xor rax, rax
    xor rcx, rcx
    mov rdx, rsi
    dec rdx
.Lloop:
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lnext
    inc rax
.Lnext:
    inc rcx
    cmp rcx, rdx
    jb .Lloop
    ret
.Lzero:
    xor rax, rax
    ret
