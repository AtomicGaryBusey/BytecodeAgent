export function l $fill_down(l %a, l %n, l %start) {
@start
  jmp @loop
@loop
  %p =l phi @start %a, @body %pnext
  %v =l phi @start %start, @body %vnext
  %i =l phi @start %n, @body %inext
  %z =l const 0
  %done =w ceql %i, %z
  jnz %done, @end, @body
@body
  storel %v, %p
  %pnext =l add %p, 8
  %vnext =l sub %v, 1
  %inext =l sub %i, 1
  jmp @loop
@end
  ret %n
}
