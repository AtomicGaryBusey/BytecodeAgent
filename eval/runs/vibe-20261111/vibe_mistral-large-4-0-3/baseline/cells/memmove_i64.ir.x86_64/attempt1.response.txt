export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %zero =l copy 0
  %iszero =w ceql %n, 0
  jnz %iszero, @end, @checkdir
@checkdir
  %back =w csltl %src, %dst
  %starti =l sub %n, 1
  jnz %back, @backloop, @forward
@forward
  %i =l phi @checkdir 0, @fbody %inext
  %done =w ceql %i, %n
  jnz %done, @end, @fbody
@fbody
  %off =l mul %i, 8
  %sp =l add %src, %off
  %v =l loadl %sp
  %dp =l add %dst, %off
  storel %v, %dp
  %inext =l add %i, 1
  jmp @forward
@backloop
  %j =l phi @checkdir %starti, @bbody %jnext
  %neg =w csltl %j, 0
  jnz %neg, @end, @bbody
@bbody
  %off =l mul %j, 8
  %sp =l add %src, %off
  %v =l loadl %sp
  %dp =l add %dst, %off
  storel %v, %dp
  %jnext =l sub %j, 1
  jmp @backloop
@end
  ret %zero
}
