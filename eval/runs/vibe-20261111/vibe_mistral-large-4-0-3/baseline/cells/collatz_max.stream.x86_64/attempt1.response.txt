collatz_max:
xor rax, rax
mov r8, 1
.Louter:
cmp r8, rdi
ja .Ldone
mov rcx, r8
xor r9, r9
.Lstep:
cmp rcx, 1
je .Lhave
test rcx, 1
jz .Leven
lea rcx, [rcx + rcx*2 + 1]
jmp .Linc
.Leven:
shr rcx, 1
.Linc:
inc r9
jmp .Lstep
.Lhave:
cmp r9, rax
jbe .Lnext
mov rax, r9
.Lnext:
inc r8
jmp .Louter
.Ldone:
ret
