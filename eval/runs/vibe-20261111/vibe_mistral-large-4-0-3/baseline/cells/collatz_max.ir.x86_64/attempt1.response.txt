export function l $collatz_max(l %n) {
@start
    %best =l copy 0
    %k =l copy 1
    jmp @outer
@outer
    %done =w csgtl %k, %n
    jnz %done, @end, @body
@body
    %x =l copy %k
    %steps =l copy 0
    jmp @inner
@inner
    %one =w ceql %x, 1
    jnz %one, @after_inner, @step
@step
    %r =l rem %x, 2
    %even =w ceql %r, 0
    jnz %even, @even, @odd
@even
    %x =l div %x, 2
    %steps =l add %steps, 1
    jmp @inner
@odd
    %t =l mul %x, 3
    %x =l add %t, 1
    %steps =l add %steps, 1
    jmp @inner
@after_inner
    %better =w csgtl %steps, %best
    jnz %better, @update, @next
@update
    %best =l copy %steps
    jmp @next
@next
    %k =l add %k, 1
    jmp @outer
@end
    ret %best
}
