export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l add 0, 0
	%n =l add %na, %nb
	jmp @zero_init
@zero_init
	%zk =l add %zero, %zero
	jmp @zero_cond
@zero_cond
	%zk_lt =w csltl %zk, %n
	jnz %zk_lt, @zero_body, @zero_done
@zero_body
	%zoff =l mul %zk, 8
	%zp =l add %r, %zoff
	storel %zero, %zp
	%zk =l add %zk, 1
	jmp @zero_cond
@zero_done
	%na_zero =w ceql %na, %zero
	jnz %na_zero, @ret_zero, @check_nb
@check_nb
	%nb_zero =w ceql %nb, %zero
	jnz %nb_zero, @ret_zero, @outer_init
@ret_zero
	ret %zero
@outer_init
	%i =l add %zero, %zero
	jmp @outer_cond
@outer_cond
	%i_lt =w csltl %i, %na
	jnz %i_lt, @outer_body, @sig_init
@outer_body
	%carry =l add %zero, %zero
	%j =l add %zero, %zero
	jmp @inner_cond
@inner_cond
	%j_lt =w csltl %j, %nb
	jnz %j_lt, @inner_body, @outer_next
@inner_body
	%k =l add %i, %j
	%aoff =l mul %i, 8
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	%boff =l mul %j, 8
	%bp =l add %b, %boff
	%bj =l loadl %bp
	%roff =l mul %k, 8
	%rp =l add %r, %roff
	%cur =l loadl %rp
	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%mid =l add %p01, %p10
	%mid_c =w cultl %mid, %p01
	%mid_cl =l extuw %mid_c
	%mid_low =l and %mid, 4294967295
	%mid_shift =l shl %mid_low, 32
	%mid_hi =l shr %mid, 32
	%mid_ch =l shl %mid_cl, 32
	%high_mid =l add %mid_hi, %mid_ch
	%t1 =l add %cur, %p00
	%c1 =w cultl %t1, %cur
	%c1l =l extuw %c1
	%t2 =l add %t1, %mid_shift
	%c2 =w cultl %t2, %t1
	%c2l =l extuw %c2
	%t3 =l add %t2, %carry
	%c3 =w cultl %t3, %t2
	%c3l =l extuw %c3
	storel %t3, %rp
	%carry_a =l add %p11, %high_mid
	%carry_b =l add %carry_a, %c1l
	%carry_c =l add %carry_b, %c2l
	%carry =l add %carry_c, %c3l
	%j =l add %j, 1
	jmp @inner_cond
@outer_next
	%fin =l add %i, %nb
	%foff =l mul %fin, 8
	%fp =l add %r, %foff
	storel %carry, %fp
	%i =l add %i, 1
	jmp @outer_cond
@sig_init
	%sig =l add %n, %zero
	jmp @sig_cond
@sig_cond
	%sig_gt =w csltl %zero, %sig
	jnz %sig_gt, @sig_check, @sig_done
@sig_check
	%idx =l sub %sig, 1
	%ioff =l mul %idx, 8
	%ip =l add %r, %ioff
	%ival =l loadl %ip
	%ival_zero =w ceql %ival, %zero
	jnz %ival_zero, @sig_dec, @sig_done
@sig_dec
	%sig =l sub %sig, 1
	jmp @sig_cond
@sig_done
	ret %sig
}
