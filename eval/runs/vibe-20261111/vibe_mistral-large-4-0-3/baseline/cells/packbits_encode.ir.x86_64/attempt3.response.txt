export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @after %inext
	%pos =l phi @start 0, @after %posnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@done
	ret %pos
@body
	%i2 =l add %i, 2
	%has3 =w csltl %i2, %n
	jnz %has3, @check3, @literal_init
@check3
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%p1 =l add %p0, 1
	%b1 =w loadub %p1
	%p2 =l add %p0, 2
	%b2 =w loadub %p2
	%e01 =w ceqw %b0, %b1
	%e12 =w ceqw %b1, %b2
	%triple =w and %e01, %e12
	jnz %triple, @measure, @literal_init
@measure
	jmp @run_loop
@run_loop
	%k =w phi @measure 1, @run_body %knext
	%kl =l phi @measure 1, @run_body %klnext
	%k_lt =w csltl %kl, 128
	%idx =l add %i, %kl
	%idx_lt =w csltl %idx, %n
	%go =w and %k_lt, %idx_lt
	jnz %go, @run_cmp, @run_done
@run_cmp
	%pi =l add %src, %i
	%b0r =w loadub %pi
	%pj =l add %pi, %kl
	%bj =w loadub %pj
	%eq =w ceqw %b0r, %bj
	jnz %eq, @run_body, @run_done
@run_body
	%knext =w add %k, 1
	%klnext =l add %kl, 1
	jmp @run_loop
@run_done
	%endp =l add %pos, 2
	%lt =w csltl %cap, %endp
	jnz %lt, @fail, @rep_write
@rep_write
	%hdr =w sub 257, %k
	%op =l add %out, %pos
	storeb %hdr, %op
	%pi2 =l add %src, %i
	%bval =w loadub %pi2
	%op1 =l add %op, 1
	storeb %bval, %op1
	%posnext_rep =l add %pos, 2
	%inext_rep =l add %i, %kl
	jmp @after
@literal_init
	jmp @lit_loop
@lit_loop
	%j =l phi @literal_init %i, @lit_include %jnext
	%len =w phi @literal_init 0, @lit_include %lennext
	%lenl =l phi @literal_init 0, @lit_include %lenlnext
	%j_lt =w csltl %j, %n
	%len_lt =w csltl %lenl, 128
	%scan =w and %j_lt, %len_lt
	jnz %scan, @lit_check3, @lit_emit
@lit_check3
	%j2 =l add %j, 2
	%has3l =w csltl %j2, %n
	jnz %has3l, @lit_triple, @lit_include
@lit_triple
	%q0 =l add %src, %j
	%c0 =w loadub %q0
	%q1 =l add %q0, 1
	%c1 =w loadub %q1
	%q2 =l add %q0, 2
	%c2 =w loadub %q2
	%f01 =w ceqw %c0, %c1
	%f12 =w ceqw %c1, %c2
	%tripl =w and %f01, %f12
	jnz %tripl, @lit_emit, @lit_include
@lit_include
	%jnext =l add %j, 1
	%lennext =w add %len, 1
	%lenlnext =l add %lenl, 1
	jmp @lit_loop
@lit_emit
	%size =l add %lenl, 1
	%rem =l sub %cap, %pos
	%lt2 =w csltl %rem, %size
	jnz %lt2, @fail, @lit_write
@lit_write
	%hdr =w sub %len, 1
	%op =l add %out, %pos
	storeb %hdr, %op
	%srcbase =l add %src, %i
	%dstbase =l add %op, 1
	jmp @lit_copy
@lit_copy
	%k2 =l phi @lit_write 0, @lit_copy_body %k2next
	%k2_lt =w csltl %k2, %lenl
	jnz %k2_lt, @lit_copy_body, @lit_done
@lit_copy_body
	%ps =l add %srcbase, %k2
	%v =w loadub %ps
	%pd =l add %dstbase, %k2
	storeb %v, %pd
	%k2next =l add %k2, 1
	jmp @lit_copy
@lit_done
	%newpos =l add %pos, %size
	%inext_lit =l add %i, %lenl
	jmp @after
@after
	%posnext =l phi @lit_done %newpos, @rep_write %posnext_rep
	%inext =l phi @lit_done %inext_lit, @rep_write %inext_rep
	jmp @main
@fail
	ret -1
}
