export function l $base64_encode(l %src, l %n, l %out) {
@start
	%np2 =l add %n, 2
	%ngroups =l div %np2, 3
	%off =l copy 0
	%p =l copy %out
	jmp @cond
@cond
	%cnd =w csltl %off, %n
	jnz %cnd, @body, @done
@body
	%si =l add %src, %off
	%rem =l sub %n, %off
	%b0 =l loadub %si
	%has1 =w csgtl %rem, 1
	%has2 =w csgtl %rem, 2
	%d1 =l select %has1, 1, 0
	%pi1 =l add %si, %d1
	%raw1 =l loadub %pi1
	%b1 =l select %has1, %raw1, 0
	%d2 =l select %has2, 2, 0
	%pi2 =l add %si, %d2
	%raw2 =l loadub %pi2
	%b2 =l select %has2, %raw2, 0
	%v1 =l shl %b0, 16
	%t1 =l shl %b1, 8
	%v2 =l or %v1, %t1
	%v3 =l or %v2, %b2
	%i0 =l shr %v3, 18
	%j1 =l shr %v3, 12
	%i1 =l and %j1, 63
	%j2 =l shr %v3, 6
	%i2 =l and %j2, 63
	%i3 =l and %v3, 63
	%lt260 =w csltl %i0, 26
	%lt520 =w csltl %i0, 52
	%lt620 =w csltl %i0, 62
	%eq620 =w ceql %i0, 62
	%ca0 =l add 65, %i0
	%cb0 =l add %i0, 71
	%cc0 =l sub %i0, 4
	%last0 =l select %eq620, 43, 47
	%mid0 =l select %lt620, %cc0, %last0
	%mid20 =l select %lt520, %cb0, %mid0
	%m0 =l select %lt260, %ca0, %mid20
	%lt261 =w csltl %i1, 26
	%lt521 =w csltl %i1, 52
	%lt621 =w csltl %i1, 62
	%eq621 =w ceql %i1, 62
	%ca1 =l add 65, %i1
	%cb1 =l add %i1, 71
	%cc1 =l sub %i1, 4
	%last1 =l select %eq621, 43, 47
	%mid1 =l select %lt621, %cc1, %last1
	%mid21 =l select %lt521, %cb1, %mid1
	%m1 =l select %lt261, %ca1, %mid21
	%lt262 =w csltl %i2, 26
	%lt522 =w csltl %i2, 52
	%lt622 =w csltl %i2, 62
	%eq622 =w ceql %i2, 62
	%ca2 =l add 65, %i2
	%cb2 =l add %i2, 71
	%cc2 =l sub %i2, 4
	%last2 =l select %eq622, 43, 47
	%mid2 =l select %lt622, %cc2, %last2
	%mid22 =l select %lt522, %cb2, %mid2
	%m2 =l select %lt262, %ca2, %mid22
	%lt263 =w csltl %i3, 26
	%lt523 =w csltl %i3, 52
	%lt623 =w csltl %i3, 62
	%eq623 =w ceql %i3, 62
	%ca3 =l add 65, %i3
	%cb3 =l add %i3, 71
	%cc3 =l sub %i3, 4
	%last3 =l select %eq623, 43, 47
	%mid3 =l select %lt623, %cc3, %last3
	%mid23 =l select %lt523, %cb3, %mid3
	%m3 =l select %lt263, %ca3, %mid23
	%o1 =l select %has1, %m1, 61
	%o2 =l select %has1, %m2, 61
	%o3 =l select %has2, %m3, 61
	%q0 =l add %p, 0
	storeb %m0, %q0
	%q1 =l add %p, 1
	storeb %o1, %q1
	%q2 =l add %p, 2
	storeb %o2, %q2
	%q3 =l add %p, 3
	storeb %o3, %q3
	%p =l add %p, 4
	%off =l add %off, 3
	jmp @cond
@done
	%res =l mul %ngroups, 4
	ret %res
}
