memmove_i64:
cld
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jbe .Lfwd
mov r8, rdx
dec r8
.Lbwd:
mov r9, qword ptr [rsi + r8*8]
mov qword ptr [rdi + r8*8], r9
dec r8
jns .Lbwd
jmp .Ldone
.Lfwd:
xor rcx, rcx
.Lfwdloop:
mov r9, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r9
inc rcx
cmp rcx, rdx
jb .Lfwdloop
.Ldone:
xor eax, eax
ret
