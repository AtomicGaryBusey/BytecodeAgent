export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy 1
@outer
	%done =w csgtl %k, %n
	jnz %done, @end, @body
@body
	%x =l copy %k
	%steps =l copy 0
	jmp @inner
@inner
	%at1 =w ceql %x, 1
	jnz %at1, @after, @step
@step
	%oddbit =l and %x, 1
	%odd =w cnel %oddbit, 0
	jnz %odd, @odd, @even
@odd
	%triple =l mul %x, 3
	%x =l add %triple, 1
	%steps =l add %steps, 1
	jmp @inner
@even
	%x =l div %x, 2
	%steps =l add %steps, 1
	jmp @inner
@after
	%greater =w csgtl %steps, %best
	jnz %greater, @update, @advance
@update
	%best =l copy %steps
	jmp @advance
@advance
	%k =l add %k, 1
	jmp @outer
@end
	ret %best
}
