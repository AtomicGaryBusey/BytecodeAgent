export function l $collatz_max(l %n) {
@start
	%zero = mul %n, 0
	%best = copy %zero
	%k = add %zero, 1
@outer
	%done = csgtl %k, %n
	jnz %done, @end, @body
@body
	%x = copy %k
	%steps = copy %zero
	jmp @inner
@inner
	%at1 = ceql %x, 1
	jnz %at1, @after, @step
@step
	%oddbit = and %x, 1
	%odd = cneql %oddbit, 0
	jnz %odd, @odd, @even
@odd
	%triple = mul %x, 3
	%xnext = add %triple, 1
	%x = copy %xnext
	%nextsteps = add %steps, 1
	%steps = copy %nextsteps
	jmp @inner
@even
	%xnext = div %x, 2
	%x = copy %xnext
	%nextsteps = add %steps, 1
	%steps = copy %nextsteps
	jmp @inner
@after
	%greater = csgtl %steps, %best
	jnz %greater, @update, @advance
@update
	%best = copy %steps
	jmp @advance
@advance
	%knext = add %k, 1
	%k = copy %knext
	jmp @outer
@end
	ret %best
}
