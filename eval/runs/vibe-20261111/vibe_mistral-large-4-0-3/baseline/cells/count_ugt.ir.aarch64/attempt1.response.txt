export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%cnt =l copy 0
	%p =l copy %a
	jmp @cond
@cond
	%more =w cultl %i, %n
	jnz %more, @body, @done
@body
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, 1
	%p =l add %p, 8
	jmp @cond
@done
	ret %cnt
}
