export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	%p0 =l add %a, 8
	jmp @loop
@loop
	%i =l phi @start %i0, @take %inext, @skip %inext
	%p =l phi @start %p0, @take %pnext, @skip %pnext
	%max =l phi @start %max0, @take %maxnextt, @skip %maxnexts
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%v =l loadl %p
	%inext =l add %i, 1
	%pnext =l add %p, 8
	%gt =w csgtl %v, %max
	jnz %gt, @take, @skip
@take
	%maxnextt =l copy %v
	jmp @loop
@skip
	%maxnexts =l copy %max
	jmp @loop
@done
	ret %max
}
