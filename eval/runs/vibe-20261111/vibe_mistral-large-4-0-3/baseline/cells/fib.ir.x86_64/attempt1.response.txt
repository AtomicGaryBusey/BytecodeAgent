export function l $fib(l %n) {
@start
  %two =l copy 2
  %small =w csltl %n, %two
  jnz %small, @ret_n, @init
@ret_n
  ret %n
@init
  %k =l sub %n, 1
  %a =l copy 0
  %b =l copy 1
  jmp @loop
@loop
  %zero =l copy 0
  %pos =w csltl %zero, %k
  jnz %pos, @body, @done
@body
  %t =l add %a, %b
  %a =l copy %b
  %b =l copy %t
  %k =l sub %k, 1
  jmp @loop
@done
  ret %b
}
