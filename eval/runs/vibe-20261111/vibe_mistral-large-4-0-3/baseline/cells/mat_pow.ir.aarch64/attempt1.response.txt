export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%zero =l mul %n, 0
	%one =l add %zero, 1
	%nn =l mul %n, %n
	%two_nn =l mul %nn, 2
	jmp @init_out_zero
@init_out_zero
	%zidx =l phi @start %zero, @init_out_zero_body %zidx_next
	%zc =w csltl %zidx, %nn
	jnz %zc, @init_out_zero_body, @init_diag
@init_out_zero_body
	%zoff =l mul %zidx, 8
	%zp =l add %out, %zoff
	storel %zero, %zp
	%zidx_next =l add %zidx, %one
	jmp @init_out_zero
@init_diag
	%di =l phi @init_out_zero %zero, @init_diag_body %di_next
	%dc =w csltl %di, %n
	jnz %dc, @init_diag_body, @exp_loop
@init_diag_body
	%drow =l mul %di, %n
	%didx =l add %drow, %di
	%doff =l mul %didx, 8
	%dp =l add %out, %doff
	storel %one, %dp
	%di_next =l add %di, %one
	jmp @init_diag
@exp_loop
	%e =l phi @init_diag %zero, @exp_next %e_next
	%ec =w csltl %e, %k
	jnz %ec, @exp_body, @zero_tmp
@exp_body
	jmp @mul_i
@mul_i
	%mi =l phi @exp_body %zero, @mul_i_next %mi_next
	%mic =w csltl %mi, %n
	jnz %mic, @mul_i_body, @mul_done
@mul_i_body
	jmp @mul_j
@mul_j
	%mj =l phi @mul_i_body %zero, @mul_j_next %mj_next
	%mjc =w csltl %mj, %n
	jnz %mjc, @mul_j_body, @mul_i_next
@mul_j_body
	jmp @mul_t
@mul_t
	%mt =l phi @mul_j_body %zero, @mul_t_body %mt_next
	%sum =l phi @mul_j_body %zero, @mul_t_body %sum_next
	%mtc =w csltl %mt, %n
	jnz %mtc, @mul_t_body, @mul_store
@mul_t_body
	%aidx =l mul %mi, %n
	%aidx2 =l add %aidx, %mt
	%aoff =l mul %aidx2, 8
	%ap =l add %out, %aoff
	%av =l loadl %ap
	%bidx =l mul %mt, %n
	%bidx2 =l add %bidx, %mj
	%boff =l mul %bidx2, 8
	%bp =l add %a, %boff
	%bv =l loadl %bp
	%prod =l mul %av, %bv
	%sum_next =l add %sum, %prod
	%mt_next =l add %mt, %one
	jmp @mul_t
@mul_store
	%cidx =l mul %mi, %n
	%cidx2 =l add %cidx, %mj
	%coff =l mul %cidx2, 8
	%cp =l add %tmp, %coff
	storel %sum, %cp
	jmp @mul_j_next
@mul_j_next
	%mj_next =l add %mj, %one
	jmp @mul_j
@mul_i_next
	%mi_next =l add %mi, %one
	jmp @mul_i
@mul_done
	jmp @copy_loop
@copy_loop
	%c =l phi @mul_done %zero, @copy_body %c_next
	%cc =w csltl %c, %nn
	jnz %cc, @copy_body, @exp_next
@copy_body
	%copyoff =l mul %c, 8
	%tp =l add %tmp, %copyoff
	%tv =l loadl %tp
	%op =l add %out, %copyoff
	storel %tv, %op
	%c_next =l add %c, %one
	jmp @copy_loop
@exp_next
	%e_next =l add %e, %one
	jmp @exp_loop
@zero_tmp
	%t =l phi @exp_loop %zero, @zero_tmp_body %t_next
	%tc =w csltl %t, %two_nn
	jnz %tc, @zero_tmp_body, @return
@zero_tmp_body
	%toff =l mul %t, 8
	%tp2 =l add %tmp, %toff
	storel %zero, %tp2
	%t_next =l add %t, %one
	jmp @zero_tmp
@return
	ret %zero
}
