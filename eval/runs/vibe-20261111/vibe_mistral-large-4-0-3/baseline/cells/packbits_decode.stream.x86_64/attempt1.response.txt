packbits_decode:
    cld
    xor eax, eax
.Lloop:
    test rsi, rsi
    jz .Ldone
    movsx r9, byte ptr [rdi]
    inc rdi
    dec rsi
    test r9, r9
    js .Lneg
    lea r8, [r9 + 1]
    cmp rsi, r8
    jb .Lsrc_err
    cmp rcx, r8
    jae .Llit_full
    mov rsi, rdi
    mov rdi, rdx
    rep movsb
    mov rax, -1
    ret
.Llit_full:
    mov r10, rdi
    mov r11, rsi
    mov r9, rcx
    mov rsi, r10
    mov rdi, rdx
    mov rcx, r8
    rep movsb
    mov rdx, rdi
    mov rdi, rsi
    mov rsi, r11
    sub rsi, r8
    mov rcx, r9
    sub rcx, r8
    add rax, r8
    jmp .Lloop
.Lneg:
    cmp r9, -128
    je .Lloop
    mov r8, 1
    sub r8, r9
    test rsi, rsi
    jz .Lsrc_err
    movzx r10d, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp rcx, r8
    jae .Lrep_full
    mov rdi, rdx
    mov al, r10b
    rep stosb
    mov rax, -1
    ret
.Lrep_full:
    mov r11, r8
.Lrep_fill:
    test r11, r11
    jz .Lrep_done
    mov [rdx], r10b
    inc rdx
    dec r11
    jmp .Lrep_fill
.Lrep_done:
    add rax, r8
    sub rcx, r8
    jmp .Lloop
.Lsrc_err:
    mov rax, -2
    ret
.Ldone:
    ret
