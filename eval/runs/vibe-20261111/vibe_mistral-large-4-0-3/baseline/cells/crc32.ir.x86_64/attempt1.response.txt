export function l $crc32(l %p, l %n) {
@start
  %crc =w copy 4294967295
  %i =l copy 0
  j @outer_cond
@outer_cond
  %cont =w csltl %i, %n
  jnz %cont, @outer_body, @finish
@outer_body
  %ptr =l add %p, %i
  %b =w loadub %ptr
  %crc =w xor %crc, %b
  %j =w copy 0
  j @inner_cond
@inner_cond
  %jcont =w csltw %j, 8
  jnz %jcont, @inner_body, @outer_next
@inner_body
  %bit =w and %crc, 1
  jnz %bit, @odd, @even
@odd
  %shr =w shr %crc, 1
  %crc =w xor %shr, 3988292384
  j @inner_next
@even
  %crc =w shr %crc, 1
  j @inner_next
@inner_next
  %j =w add %j, 1
  j @inner_cond
@outer_next
  %i =l add %i, 1
  j @outer_cond
@finish
  %final =w xor %crc, 4294967295
  %result =l extuw %final
  ret %result
}
