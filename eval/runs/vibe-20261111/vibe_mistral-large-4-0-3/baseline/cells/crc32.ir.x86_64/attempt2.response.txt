export function l $crc32(l %p, l %n) {
@start
  %crc =w copy 4294967295
  %i =l copy 0
  jmp @outer_cond
@outer_cond
  %cont =w csltl %i, %n
  jnz %cont, @outer_body, @finish
@outer_body
  %ptr =l add %p, %i
  %b =w loadub %ptr
  %crc =w xor %crc, %b
  %k =w copy 0
  jmp @inner_cond
@inner_cond
  %kcont =w csltw %k, 8
  jnz %kcont, @inner_body, @outer_next
@inner_body
  %bit =w and %crc, 1
  jnz %bit, @odd, @even
@odd
  %shr =w shr %crc, 1
  %crc =w xor %shr, 3988292384
  jmp @inner_next
@even
  %crc =w shr %crc, 1
  jmp @inner_next
@inner_next
  %k =w add %k, 1
  jmp @inner_cond
@outer_next
  %i =l add %i, 1
  jmp @outer_cond
@finish
  %final =w xor %crc, 4294967295
  %result =l extuw %final
  ret %result
}
