bmin_empty:
  test rsi, rsi
  jne .Lnonempty
  mov rax, 0x7fffffffffffffff
  ret
.Lnonempty:
  mov rax, qword ptr [rdi]
  cmp rsi, 1
  je .Ldone
  mov rcx, rsi
  dec rcx
.Lloop:
  mov rdx, qword ptr [rdi + rcx*8]
  cmp rdx, rax
  jge .Lnext
  mov rax, rdx
.Lnext:
  dec rcx
  jnz .Lloop
.Ldone:
  ret
