export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%p =l copy %a
	%rem =l copy %n
	%one =l copy 1
@loop
	%more =w csltl %one, %rem
	jnz %more, @body, @done
@body
	%np =l add %p, 8
	%v =l loadl %np
	%lt =w csltl %max, %v
	jnz %lt, @update, @next
@update
	%max =l copy %v
@next
	%p =l copy %np
	%rem =l sub %rem, %one
	jmp @loop
@done
	ret %max
}
