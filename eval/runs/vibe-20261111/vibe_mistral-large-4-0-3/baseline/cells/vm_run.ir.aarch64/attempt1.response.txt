export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
  %s_ip =l alloc8
  %s_sp =l alloc8
  storel %code, %s_ip
  storel %stack, %s_sp
  %len8 =l mul %len, 8
  %end =l add %code, %len8
  %cap8 =l mul %cap, 8
  %capend =l add %stack, %cap8
  jmp @loop
@loop
  %ip0 =l loadl %s_ip
  %done0 =w ceql %ip0, %end
  jnz %done0, @finish, @fetch
@fetch
  %op0 =l loadl %ip0
  %c0 =w ceql %op0, 0
  jnz %c0, @halt, @t1
@t1
  %c1 =w ceql %op0, 1
  jnz %c1, @push, @t2
@t2
  %c2 =w ceql %op0, 2
  jnz %c2, @add, @t3
@t3
  %c3 =w ceql %op0, 3
  jnz %c3, @sub, @t4
@t4
  %c4 =w ceql %op0, 4
  jnz %c4, @mul, @t5
@t5
  %c5 =w ceql %op0, 5
  jnz %c5, @dup, @t6
@t6
  %c6 =w ceql %op0, 6
  jnz %c6, @swap, @t7
@t7
  %c7 =w ceql %op0, 7
  jnz %c7, @drop, @t8
@t8
  %c8 =w ceql %op0, 8
  jnz %c8, @over, @t9
@t9
  %c9 =w ceql %op0, 9
  jnz %c9, @lt, @t10
@t10
  %c10 =w ceql %op0, 10
  jnz %c10, @jz, @t11
@t11
  %c11 =w ceql %op0, 11
  jnz %c11, @jmp, @other
@other
  %ip_other =l add %ip0, 8
  storel %ip_other, %s_ip
  jmp @loop
@halt
  jmp @finish
@push
  %ip_p1 =l add %ip0, 8
  %val_p =l loadl %ip_p1
  %sp_p1 =l loadl %s_sp
  storel %val_p, %sp_p1
  %sp_p2 =l add %sp_p1, 8
  storel %sp_p2, %s_sp
  %ip_p2 =l add %ip0, 16
  storel %ip_p2, %s_ip
  jmp @loop
@add
  %sp_a0 =l loadl %s_sp
  %sp_a1 =l sub %sp_a0, 8
  %b_a =l loadl %sp_a1
  %sp_a2 =l sub %sp_a1, 8
  %a_a =l loadl %sp_a2
  %r_a =l add %a_a, %b_a
  storel %r_a, %sp_a2
  %sp_a3 =l add %sp_a2, 8
  storel %sp_a3, %s_sp
  %ip_a =l add %ip0, 8
  storel %ip_a, %s_ip
  jmp @loop
@sub
  %sp_s0 =l loadl %s_sp
  %sp_s1 =l sub %sp_s0, 8
  %b_s =l loadl %sp_s1
  %sp_s2 =l sub %sp_s1, 8
  %a_s =l loadl %sp_s2
  %r_s =l sub %a_s, %b_s
  storel %r_s, %sp_s2
  %sp_s3 =l add %sp_s2, 8
  storel %sp_s3, %s_sp
  %ip_s =l add %ip0, 8
  storel %ip_s, %s_ip
  jmp @loop
@mul
  %sp_m0 =l loadl %s_sp
  %sp_m1 =l sub %sp_m0, 8
  %b_m =l loadl %sp_m1
  %sp_m2 =l sub %sp_m1, 8
  %a_m =l loadl %sp_m2
  %r_m =l mul %a_m, %b_m
  storel %r_m, %sp_m2
  %sp_m3 =l add %sp_m2, 8
  storel %sp_m3, %s_sp
  %ip_m =l add %ip0, 8
  storel %ip_m, %s_ip
  jmp @loop
@dup
  %sp_d0 =l loadl %s_sp
  %addr_d =l sub %sp_d0, 8
  %v_d =l loadl %addr_d
  storel %v_d, %sp_d0
  %sp_d1 =l add %sp_d0, 8
  storel %sp_d1, %s_sp
  %ip_d =l add %ip0, 8
  storel %ip_d, %s_ip
  jmp @loop
@swap
  %sp_w0 =l loadl %s_sp
  %addrstop_w =l sub %sp_w0, 8
  %b_w =l loadl %addrstop_w
  %addrsbelow_w =l sub %sp_w0, 16
  %a_w =l loadl %addrsbelow_w
  storel %b_w, %addrsbelow_w
  storel %a_w, %addrstop_w
  %ip_w =l add %ip0, 8
  storel %ip_w, %s_ip
  jmp @loop
@drop
  %sp_r0 =l loadl %s_sp
  %sp_r1 =l sub %sp_r0, 8
  storel %sp_r1, %s_sp
  %ip_r =l add %ip0, 8
  storel %ip_r, %s_ip
  jmp @loop
@over
  %sp_o0 =l loadl %s_sp
  %addr_o =l sub %sp_o0, 16
  %v_o =l loadl %addr_o
  storel %v_o, %sp_o0
  %sp_o1 =l add %sp_o0, 8
  storel %sp_o1, %s_sp
  %ip_o =l add %ip0, 8
  storel %ip_o, %s_ip
  jmp @loop
@lt
  %sp_l0 =l loadl %s_sp
  %sp_l1 =l sub %sp_l0, 8
  %b_l =l loadl %sp_l1
  %sp_l2 =l sub %sp_l1, 8
  %a_l =l loadl %sp_l2
  %cmp_l =w csltl %a_l, %b_l
  %r_l =l extsw %cmp_l
  storel %r_l, %sp_l2
  %sp_l3 =l add %sp_l2, 8
  storel %sp_l3, %s_sp
  %ip_l =l add %ip0, 8
  storel %ip_l, %s_ip
  jmp @loop
@jz
  %ip_j0 =l add %ip0, 8
  %t_j =l loadl %ip_j0
  %sp_j0 =l loadl %s_sp
  %sp_j1 =l sub %sp_j0, 8
  %v_j =l loadl %sp_j1
  storel %sp_j1, %s_sp
  %iszero_j =w ceql %v_j, 0
  jnz %iszero_j, @jz_taken, @jz_next
@jz_taken
  %t8_j =l mul %t_j, 8
  %ip_jt =l add %code, %t8_j
  storel %ip_jt, %s_ip
  jmp @loop
@jz_next
  %ip_jn =l add %ip0, 16
  storel %ip_jn, %s_ip
  jmp @loop
@jmp
  %ip_m0 =l add %ip0, 8
  %t_m =l loadl %ip_m0
  %t8_m =l mul %t_m, 8
  %ip_m1 =l add %code, %t8_m
  storel %ip_m1, %s_ip
  jmp @loop
@finish
  %sp_f =l loadl %s_sp
  %less_f =w csltl %sp_f, %capend
  jnz %less_f, @zero_body, @ret
@zero_body
  storel 0, %sp_f
  %sp_z =l add %sp_f, 8
  storel %sp_z, %s_sp
  jmp @finish
@ret
  %empty_r =w ceql %sp_f, %stack
  jnz %empty_r, @ret0, @rettop
@ret0
  ret 0
@rettop
  %addr_rt =l sub %sp_f, 8
  %top_r =l loadl %addr_rt
  ret %top_r
}
