base64_encode:
 xor r11d, r11d
.Lloop:
 test rsi, rsi
 jz .Ldone
 cmp rsi, 3
 jae .Lfull
 cmp rsi, 2
 je .Ltwo
 movzx eax, byte ptr [rdi]
 shl eax, 16
 mov r10d, eax
 mov r9d, 2
 jmp .Lchars
.Lfull:
 movzx eax, byte ptr [rdi]
 shl eax, 8
 movzx r8d, byte ptr [rdi + 1]
 or eax, r8d
 shl eax, 8
 movzx r8d, byte ptr [rdi + 2]
 or eax, r8d
 mov r10d, eax
 xor r9d, r9d
 jmp .Lchars
.Ltwo:
 movzx eax, byte ptr [rdi]
 shl eax, 8
 movzx r8d, byte ptr [rdi + 1]
 or eax, r8d
 shl eax, 8
 mov r10d, eax
 mov r9d, 1
.Lchars:
 xor r8d, r8d
.Lchar_loop:
 cmp r8d, 4
 jge .Lgroup_done
 mov eax, 4
 sub eax, r9d
 cmp r8d, eax
 jae .Lpad_char
 imul eax, r8d, 6
 mov ecx, 18
 sub ecx, eax
 mov eax, r10d
 shr eax, cl
 and eax, 63
 cmp eax, 25
 jbe .Lmap_upper
 cmp eax, 51
 jbe .Lmap_lower
 cmp eax, 61
 jbe .Lmap_digit
 cmp eax, 62
 je .Lmap_plus
 mov eax, 47
 jmp .Lmap_done
.Lmap_upper:
 add eax, 65
 jmp .Lmap_done
.Lmap_lower:
 add eax, 71
 jmp .Lmap_done
.Lmap_digit:
 sub eax, 4
 jmp .Lmap_done
.Lmap_plus:
 mov eax, 43
.Lmap_done:
 mov byte ptr [rdx + r8], al
 jmp .Lnext_char
.Lpad_char:
 mov byte ptr [rdx + r8], 43
.Lnext_char:
 inc r8
 jmp .Lchar_loop
.Lgroup_done:
 cmp rsi, 3
 jae .Ladv3
 add rdi, rsi
 xor rsi, rsi
 jmp .Ladv_out
.Ladv3:
 add rdi, 3
 sub rsi, 3
.Ladv_out:
 add rdx, 4
 add r11, 4
 jmp .Lloop
.Ldone:
 mov rax, r11
 ret
