memmove_i64:

    cmp x0, x1
    b.eq .Lret
    b.lo .Lfwd
.Lback:
    sub x2, x2, #1
    ldr x9, [x1, x2, lsl #3]
    str x9, [x0, x2, lsl #3]
    cbnz x2, .Lback
    b .Lret
.Lfwd:
    ldr x9, [x1], #8
    str x9, [x0], #8
    subs x2, x2, #1
    b.ne .Lfwd
.Lret:
    mov x0, #0
    ret
