dot_i32_simd:
    mov r8, rdx
    pxor xmm0, xmm0
    pxor xmm1, xmm1
.Lloop:
    cmp r8, 4
    jb .Ltail
    movdqu xmm2, xmmword ptr [rdi]
    movdqu xmm3, xmmword ptr [rsi]
    pmovsxdq xmm4, xmm2
    psrldq xmm2, 8
    pmovsxdq xmm5, xmm2
    pmovsxdq xmm6, xmm3
    psrldq xmm3, 8
    pmovsxdq xmm7, xmm3
    pmuldq xmm4, xmm6
    pmuldq xmm5, xmm7
    paddq xmm0, xmm4
    paddq xmm1, xmm5
    add rdi, 16
    add rsi, 16
    sub r8, 4
    jmp .Lloop
.Ltail:
    paddq xmm0, xmm1
    movq rax, xmm0
    psrldq xmm0, 8
    movq rcx, xmm0
    add rax, rcx
.Lscalar:
    test r8, r8
    jz .Ldone
    movsxd r9, dword ptr [rdi]
    movsxd r10, dword ptr [rsi]
    imul r9, r10
    add rax, r9
    add rdi, 4
    add rsi, 4
    sub r8, 1
    jmp .Lscalar
.Ldone:
    ret
