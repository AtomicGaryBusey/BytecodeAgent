export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =w copy 0
    %k =w copy 0
    jmp @outer_cond
@outer_cond
    %il =l extuw %i
    %more =w csltl %il, %n
    jnz %more, @have_run, @finish
@have_run
    %r =w copy 1
    jmp @run_i_cond
@run_i_cond
    %tr =w add %i, %r
    %trl =l extuw %tr
    %inb =w csltl %trl, %n
    jnz %inb, @run_i_rlt, @run_i_done
@run_i_rlt
    %rlt =w csltw %r, 128
    jnz %rlt, @run_i_eq, @run_i_done
@run_i_eq
    %pa =l add %src, %trl
    %ba =w loadub %pa
    %pb =l add %src, %il
    %bb =w loadub %pb
    %eq =w ceqw %ba, %bb
    jnz %eq, @run_i_inc, @run_i_done
@run_i_inc
    %r =w add %r, 1
    jmp @run_i_cond
@run_i_done
    %isrep =w csltw 2, %r
    jnz %isrep, @repeat, @literal
@repeat
    %kl =l extuw %k
    %need =l add %kl, 2
    %nofit =w csltl %cap, %need
    jnz %nofit, @retm1, @rep_write
@rep_write
    %outp =l add %out, %kl
    %h =w sub 257, %r
    storeb %h, %outp
    %outp2 =l add %outp, 1
    %isrc =l extuw %i
    %srcp =l add %src, %isrc
    %b =w loadub %srcp
    storeb %b, %outp2
    %k =w add %k, 2
    %i =w add %i, %r
    jmp @outer_cond
@literal
    %j =w copy %i
    jmp @lit_cond
@lit_cond
    %jl =l extuw %j
    %jlt =w csltl %jl, %n
    jnz %jlt, @lit_len, @lit_emit
@lit_len
    %len0 =w sub %j, %i
    %lenlt =w csltw %len0, 128
    jnz %lenlt, @lit_run, @lit_emit
@lit_run
    %rr =w copy 1
    jmp @lit_run_cond
@lit_run_cond
    %trr =w add %j, %rr
    %trrl =l extuw %trr
    %inbr =w csltl %trrl, %n
    jnz %inbr, @lit_run_rlt, @lit_run_done
@lit_run_rlt
    %rrlt =w csltw %rr, 128
    jnz %rrlt, @lit_run_eq, @lit_run_done
@lit_run_eq
    %pa2 =l add %src, %trrl
    %ba2 =w loadub %pa2
    %pb2 =l add %src, %jl
    %bb2 =w loadub %pb2
    %eq2 =w ceqw %ba2, %bb2
    jnz %eq2, @lit_run_inc, @lit_run_done
@lit_run_inc
    %rr =w add %rr, 1
    jmp @lit_run_cond
@lit_run_done
    %runlt3 =w csltw %rr, 3
    jnz %runlt3, @lit_inc, @lit_emit
@lit_inc
    %j =w add %j, 1
    jmp @lit_cond
@lit_emit
    %len =w sub %j, %i
    %kl2 =l extuw %k
    %lenl =l extuw %len
    %need2 =l add %kl2, %lenl
    %needlit =l add %need2, 1
    %nofit2 =w csltl %cap, %needlit
    jnz %nofit2, @retm1, @lit_write
@lit_write
    %outk =l add %out, %kl2
    %hdr =w sub %len, 1
    storeb %hdr, %outk
    %dstbase =l add %outk, 1
    %ibase =l extuw %i
    %t =w copy 0
    jmp @copy_cond
@copy_cond
    %tlt =w csltw %t, %len
    jnz %tlt, @copy_body, @lit_done
@copy_body
    %tl =l extuw %t
    %toff =l add %dstbase, %tl
    %sfrom =l add %ibase, %tl
    %v =w loadub %sfrom
    storeb %v, %toff
    %t =w add %t, 1
    jmp @copy_cond
@lit_done
    %k1 =w add %k, %len
    %k =w add %k1, 1
    %i =w copy %j
    jmp @outer_cond
@finish
    %res =l extuw %k
    ret %res
@retm1
    %neg =l copy -1
    ret %neg
}
