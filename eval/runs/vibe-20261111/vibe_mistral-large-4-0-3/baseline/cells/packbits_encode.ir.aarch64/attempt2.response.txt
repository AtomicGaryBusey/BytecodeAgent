export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %end =l add %src, %n
    %ip =l copy %src
    %k =l copy 0
    jmp @outer_cond
@outer_cond
    %more =w csltl %ip, %end
    jnz %more, @have_run, @finish
@have_run
    %r =w copy 1
    %q =l add %ip, 1
    jmp @run_cond
@run_cond
    %qin =w csltl %q, %end
    jnz %qin, @run_rcheck, @run_done
@run_rcheck
    %rlt =w csltw %r, 128
    jnz %rlt, @run_eq, @run_done
@run_eq
    %bq =w loadub %q
    %bi =w loadub %ip
    %eq =w ceqw %bq, %bi
    jnz %eq, @run_inc, @run_done
@run_inc
    %q =l add %q, 1
    %r =w add %r, 1
    jmp @run_cond
@run_done
    %isrep =w csltw 2, %r
    jnz %isrep, @repeat, @literal
@repeat
    %needr =l add %k, 2
    %nofitr =w csltl %cap, %needr
    jnz %nofitr, @retm1, @rep_write
@rep_write
    %op =l add %out, %k
    %h =w sub 257, %r
    storeb %h, %op
    %op2 =l add %op, 1
    %bv =w loadub %ip
    storeb %bv, %op2
    %k =l add %k, 2
    %ip =l copy %q
    jmp @outer_cond
@literal
    %jp =l copy %ip
    %lenw =w copy 0
    %lenl =l copy 0
    jmp @lit_cond
@lit_cond
    %jlt =w csltl %jp, %end
    jnz %jlt, @lit_len, @lit_emit
@lit_len
    %lenlt =w csltw %lenw, 128
    jnz %lenlt, @lit_run, @lit_emit
@lit_run
    %rr =w copy 1
    %lq =l add %jp, 1
    jmp @lit_run_cond
@lit_run_cond
    %lqin =w csltl %lq, %end
    jnz %lqin, @lit_run_rcheck, @lit_run_done
@lit_run_rcheck
    %rrlt =w csltw %rr, 128
    jnz %rrlt, @lit_run_eq, @lit_run_done
@lit_run_eq
    %lbq =w loadub %lq
    %lbj =w loadub %jp
    %leq =w ceqw %lbq, %lbj
    jnz %leq, @lit_run_inc, @lit_run_done
@lit_run_inc
    %lq =l add %lq, 1
    %rr =w add %rr, 1
    jmp @lit_run_cond
@lit_run_done
    %runlt3 =w csltw %rr, 3
    jnz %runlt3, @lit_inc, @lit_emit
@lit_inc
    %jp =l add %jp, 1
    %lenw =w add %lenw, 1
    %lenl =l add %lenl, 1
    jmp @lit_cond
@lit_emit
    %needl =l add %k, %lenl
    %needl1 =l add %needl, 1
    %nofitl =w csltl %cap, %needl1
    jnz %nofitl, @retm1, @lit_write
@lit_write
    %lop =l add %out, %k
    %hdr =w sub %lenw, 1
    storeb %hdr, %lop
    %dst =l add %lop, 1
    %sp =l copy %ip
    %t =l copy 0
    jmp @copy_cond
@copy_cond
    %tlt =w csltl %t, %lenl
    jnz %tlt, @copy_body, @lit_done
@copy_body
    %v =w loadub %sp
    storeb %v, %dst
    %sp =l add %sp, 1
    %dst =l add %dst, 1
    %t =l add %t, 1
    jmp @copy_cond
@lit_done
    %k1 =l add %k, %lenl
    %k =l add %k1, 1
    %ip =l copy %jp
    jmp @outer_cond
@finish
    ret %k
@retm1
    %neg =l copy -1
    ret %neg
}
