export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%min0 =l copy %y0
	%max0 =l copy %y0
	%sum0 =l copy %y0
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @max_join %inext
	%min =l phi @start %min0, @max_join %minnext
	%max =l phi @start %max0, @max_join %maxnext
	%sum =l phi @start %sum0, @max_join %sumnext
	%odd =l phi @start %odd0, @max_join %oddnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%ai =l loadl %p
	%y =l call %f(l %ai)
	%ltmin =w csltl %y, %min
	jnz %ltmin, @min_y, @min_old
@min_y
	%min_yv =l copy %y
	jmp @min_join
@min_old
	%min_oldv =l copy %min
	jmp @min_join
@min_join
	%minnext =l phi @min_y %min_yv, @min_old %min_oldv
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @max_y, @max_old
@max_y
	%max_yv =l copy %y
	jmp @max_join
@max_old
	%max_oldv =l copy %max
	jmp @max_join
@max_join
	%maxnext =l phi @max_y %max_yv, @max_old %max_oldv
	%sumnext =l add %sum, %y
	%bit =l and %y, 1
	%oddnext =l add %odd, %bit
	%inext =l add %i, 1
	jmp @loop
@done
	%o1 =l add %out, 8
	%o2 =l add %out, 16
	storel %min, %out
	storel %max, %o1
	storel %sum, %o2
	ret %odd
}
