bubble_sort:
  cmp x1, #1
  b.le .Ldone
  sub x2, x1, #1
.Louter:
  cmp x2, #0
  b.le .Ldone
  mov x3, x0
  mov x4, x2
.Linner:
  ldr x5, [x3]
  ldr x6, [x3, #8]
  cmp x5, x6
  b.le .Lnoswap
  str x6, [x3]
  str x5, [x3, #8]
.Lnoswap:
  add x3, x3, #8
  subs x4, x4, #1
  b.ne .Linner
  sub x2, x2, #1
  b .Louter
.Ldone:
  mov x0, #0
  ret
