export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%outer_lim =l sub %n, %one
	%i =l copy 0
	jmp @outer_check
@outer_check
	%oc =w csltl %i, %outer_lim
	jnz %oc, @outer_body, @done
@outer_body
	%inner_lim =l sub %outer_lim, %i
	%j =l copy 0
	jmp @inner_check
@inner_check
	%ic =w csltl %j, %inner_lim
	jnz %ic, @inner_body, @outer_next
@inner_body
	%j8 =l mul %j, %eight
	%p =l add %a, %j8
	%q =l add %p, %eight
	%x =l loadl %p
	%y =l loadl %q
	%gt =w csltl %y, %x
	jnz %gt, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner_next
@inner_next
	%j =l add %j, %one
	jmp @inner_check
@outer_next
	%i =l add %i, %one
	jmp @outer_check
@done
	ret %zero
}
