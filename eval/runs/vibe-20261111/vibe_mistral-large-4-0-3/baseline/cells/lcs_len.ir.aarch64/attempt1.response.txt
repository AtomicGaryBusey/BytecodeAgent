export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@init
  %zero =l mul %n, 0
  %one =l add %zero, 1
  %eight =l shl %one, 3
  %nplus =l add %n, %one
  %rpz0 =l add %row, %zero
  jmp @zloop
@zloop
  %zj =l phi @init %zero, @zbody %zj1
  %zp =l phi @init %rpz0, @zbody %zp1
  %zc =w csltl %zj, %nplus
  jnz %zc, @zbody
  jmp @afterzero
@zbody
  storel %zero, %zp
  %zj1 =l add %zj, %one
  %zp1 =l add %zp, %eight
  jmp @zloop
@afterzero
  %i0 =l add %zero, %zero
  %xp0 =l add %x, %zero
  jmp @outer
@outer
  %i =l phi @afterzero %i0, @outerend %i1
  %xp =l phi @afterzero %xp0, @outerend %xp1
  %oc =w csltl %i, %m
  jnz %oc, @outerbody
  jmp @done
@outerbody
  %jstart =l add %one, %zero
  %diagstart =l add %zero, %zero
  %rpstart =l add %row, %eight
  jmp @inner
@inner
  %j =l phi @outerbody %jstart, @innerbody %j1
  %diag =l phi @outerbody %diagstart, @innerbody %diag1
  %rp =l phi @outerbody %rpstart, @innerbody %rp1
  %ic =w csltl %j, %nplus
  jnz %ic, @innerbody
  jmp @outerend
@innerbody
  %up =l loadl %rp
  %rprev =l sub %rp, %eight
  %left =l loadl %rprev
  %jm1 =l sub %j, %one
  %yoff =l mul %jm1, %eight
  %yp =l add %y, %yoff
  %yb =w loadub %yp
  %xb =w loadub %xp
  %eq =w ceqw %xb, %yb
  %eql =l extsw %eq
  %lt =w csltl %up, %left
  %ltl =l extsw %lt
  %maxdiff =l sub %left, %up
  %maxinc =l mul %ltl, %maxdiff
  %maxv =l add %up, %maxinc
  %diagp1 =l add %diag, %one
  %canddiff =l sub %diagp1, %maxv
  %candinc =l mul %eql, %canddiff
  %new =l add %maxv, %candinc
  storel %new, %rp
  %j1 =l add %j, %one
  %rp1 =l add %rp, %eight
  %diag1 =l add %up, %zero
  jmp @inner
@outerend
  %i1 =l add %i, %one
  %xp1 =l add %xp, %one
  jmp @outer
@done
  %noff =l mul %n, %eight
  %rpn =l add %row, %noff
  %res =l loadl %rpn
  ret %res
}
