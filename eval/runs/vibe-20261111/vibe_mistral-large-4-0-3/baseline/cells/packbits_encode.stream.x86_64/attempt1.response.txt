packbits_encode:
    push rbx
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movzx r11d, byte ptr [rdi + r8]
    mov r10, 1
.Lrun_scan:
    cmp r10, 128
    jge .Lrun_done
    lea rax, [r8 + r10]
    cmp rax, rsi
    jge .Lrun_done
    cmp byte ptr [rdi + rax], r11b
    jne .Lrun_done
    inc r10
    jmp .Lrun_scan
.Lrun_done:
    cmp r10, 3
    jl .Lliteral
    lea rax, [r9 + 2]
    cmp rax, rcx
    ja .Lfail
    mov eax, 257
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r8, r10
    add r9, 2
    jmp .Lloop
.Lliteral:
    mov r10, r8
.Llit_scan:
    cmp r10, rsi
    jge .Llit_done
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jge .Llit_done
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Llit_include
    movzx r11d, byte ptr [rdi + r10]
    cmp byte ptr [rdi + r10 + 1], r11b
    jne .Llit_include
    cmp byte ptr [rdi + r10 + 2], r11b
    je .Llit_done
.Llit_include:
    inc r10
    jmp .Llit_scan
.Llit_done:
    mov rax, r10
    sub rax, r8
    lea r11, [r9 + rax + 1]
    cmp r11, rcx
    ja .Lfail
    mov rbx, rax
    dec rbx
    mov byte ptr [rdx + r9], bl
    mov r10, r8
    mov r11, rax
    lea rbx, [r9 + 1]
.Lcopy:
    test r11, r11
    jz .Lcopy_done
    movzx eax, byte ptr [rdi + r10]
    mov byte ptr [rdx + rbx], al
    inc r10
    inc rbx
    dec r11
    jmp .Lcopy
.Lcopy_done:
    mov r8, r10
    mov r9, rbx
    jmp .Lloop
.Lfail:
    mov rax, -1
    jmp .Lepilogue
.Ldone:
    mov rax, r9
.Lepilogue:
    pop rbx
    ret
