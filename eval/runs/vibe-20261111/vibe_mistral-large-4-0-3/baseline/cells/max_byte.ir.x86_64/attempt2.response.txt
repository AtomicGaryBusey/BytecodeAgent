export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%ptr =l phi @start %p, @next %pnext
	%max =l phi @start 0, @next %maxnext
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%b =l loadub %ptr
	%gt =w csgtl %b, %max
	jnz %gt, @update, @skip
@update
	jmp @next
@skip
	jmp @next
@next
	%maxnext =l phi @update %b, @skip %max
	%inext =l add %i, 1
	%pnext =l add %ptr, 1
	jmp @loop
@done
	ret %max
}
