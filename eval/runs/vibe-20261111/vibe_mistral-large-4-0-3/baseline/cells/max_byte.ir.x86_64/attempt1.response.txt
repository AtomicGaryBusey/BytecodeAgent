export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @merge %inext
	%ptr =l phi @start %p, @merge %pnext
	%max =l phi @start 0, @merge %maxnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%b =l loadub %ptr
	%gt =w csgtl %b, %max
	jnz %gt, @update, @skip
@update
	%max2 =l copy %b
	jmp @merge
@skip
	%max2 =l copy %max
	jmp @merge
@merge
	%maxnext =l phi @update %max2, @skip %max2
	%inext =l add %i, 1
	%pnext =l add %ptr, 1
	jmp @loop
@done
	ret %max
}
