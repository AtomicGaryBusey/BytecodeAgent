export function l $reverse(l %a, l %n) {
@start
  %i0 =l copy 0
  %j0 =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @start %i0, @body %i1
  %j =l phi @start %j0, @body %j1
  %c =w csltl %i, %j
  jnz %c, @body, @done
@body
  %oi =l mul %i, 8
  %oj =l mul %j, 8
  %pa =l add %a, %oi
  %pb =l add %a, %oj
  %x =l loadl %pa
  %y =l loadl %pb
  storel %y, %pa
  storel %x, %pb
  %i1 =l add %i, 1
  %j1 =l sub %j, 1
  jmp @loop
@done
  %r =l copy 0
  ret %r
}
