lcs_len:
push rbx
push rbp
push r12
push r13
push r14
mov rbx, rdi
mov rbp, rsi
mov r12, rdx
mov r13, rcx
mov r14, r8
xor rcx, rcx
.Lzero:
cmp rcx, r13
jg .Lzero_done
mov qword ptr [r14 + rcx*8], 0
inc rcx
jmp .Lzero
.Lzero_done:
xor rcx, rcx
.Louter:
cmp rcx, rbp
jge .Ldone
movzx r8d, byte ptr [rbx + rcx]
xor rsi, rsi
mov rdx, 1
.Linner:
cmp rdx, r13
jg .Linner_done
mov rdi, qword ptr [r14 + rdx*8]
movzx r9d, byte ptr [r12 + rdx - 1]
cmp r8, r9
jne .Lno_match
lea rax, [rsi + 1]
jmp .Lstore
.Lno_match:
mov rax, rdi
mov r9, qword ptr [r14 + rdx*8 - 8]
cmp rax, r9
jge .Lstore
mov rax, r9
.Lstore:
mov qword ptr [r14 + rdx*8], rax
mov rsi, rdi
inc rdx
jmp .Linner
.Linner_done:
inc rcx
jmp .Louter
.Ldone:
mov rax, qword ptr [r14 + r13*8]
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
