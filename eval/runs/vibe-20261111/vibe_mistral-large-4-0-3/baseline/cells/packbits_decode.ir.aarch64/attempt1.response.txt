export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%retm1 =l sub 0, 1
	%retm2 =l sub 0, 2
	%m128 =l sub 0, 128
	jmp @loop
@loop
	%i =l phi @entry 0, @lit_done %i_lit_done, @rep_done %i_rep_done, @nop %i_nop
	%o =l phi @entry 0, @lit_done %o_lit_done, @rep_done %o_rep_done, @nop %o_nop
	%more =w csltl %i, %n
	jnz %more, @header, @done
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%i1 =l add %i, 1
	%isneg =w csltl %h, 0
	jnz %isneg, @neg, @lit
@lit
	%cnt_lit =l add %h, 1
	%need_lit =l add %cnt_lit, 1
	%rem_lit =l sub %n, %i
	%lack_lit =w csltl %rem_lit, %need_lit
	jnz %lack_lit, @srcerr, @lit_fits
@lit_fits
	%spayload =l add %src, %i1
	%room_lit =l sub %cap, %o
	%fits_lit =w cslel %cnt_lit, %room_lit
	jnz %fits_lit, @lit_copy, @lit_cap
@lit_copy
	%lit_k =l phi @lit_fits 0, @lit_copy_body %lit_k_next
	%lit_k_eq =w ceql %lit_k, %cnt_lit
	jnz %lit_k_eq, @lit_done, @lit_copy_body
@lit_copy_body
	%lit_src_addr =l add %spayload, %lit_k
	%lit_b =w loadub %lit_src_addr
	%lit_out_base =l add %out, %o
	%lit_out_addr =l add %lit_out_base, %lit_k
	storeb %lit_b, %lit_out_addr
	%lit_k_next =l add %lit_k, 1
	jmp @lit_copy
@lit_done
	%i_lit_done =l add %i, %need_lit
	%o_lit_done =l add %o, %cnt_lit
	jmp @loop
@lit_cap
	jmp @lit_cap_copy
@lit_cap_copy
	%lit_cap_k =l phi @lit_cap 0, @lit_cap_body %lit_cap_k_next
	%lit_cap_k_eq =w ceql %lit_cap_k, %room_lit
	jnz %lit_cap_k_eq, @lit_cap_done, @lit_cap_body
@lit_cap_body
	%lit_cap_src_addr =l add %spayload, %lit_cap_k
	%lit_cap_b =w loadub %lit_cap_src_addr
	%lit_cap_out_base =l add %out, %o
	%lit_cap_out_addr =l add %lit_cap_out_base, %lit_cap_k
	storeb %lit_cap_b, %lit_cap_out_addr
	%lit_cap_k_next =l add %lit_cap_k, 1
	jmp @lit_cap_copy
@lit_cap_done
	ret %retm1
@neg
	%isnop =w ceql %h, %m128
	jnz %isnop, @nop, @repeat
@nop
	%i_nop =l add %i, 1
	%o_nop =l add %o, 0
	jmp @loop
@repeat
	%cnt_rep =l sub 1, %h
	%rem_rep =l sub %n, %i
	%lack_rep =w csltl %rem_rep, 2
	jnz %lack_rep, @srcerr, @rep_fits
@rep_fits
	%rb_ptr =l add %src, %i1
	%rb =w loadub %rb_ptr
	%room_rep =l sub %cap, %o
	%fits_rep =w cslel %cnt_rep, %room_rep
	jnz %fits_rep, @rep_fill, @rep_cap
@rep_fill
	%rep_k =l phi @rep_fits 0, @rep_fill_body %rep_k_next
	%rep_k_eq =w ceql %rep_k, %cnt_rep
	jnz %rep_k_eq, @rep_done, @rep_fill_body
@rep_fill_body
	%rep_out_base =l add %out, %o
	%rep_out_addr =l add %rep_out_base, %rep_k
	storeb %rb, %rep_out_addr
	%rep_k_next =l add %rep_k, 1
	jmp @rep_fill
@rep_done
	%i_rep_done =l add %i, 2
	%o_rep_done =l add %o, %cnt_rep
	jmp @loop
@rep_cap
	jmp @rep_cap_fill
@rep_cap_fill
	%rep_cap_k =l phi @rep_cap 0, @rep_cap_body %rep_cap_k_next
	%rep_cap_k_eq =w ceql %rep_cap_k, %room_rep
	jnz %rep_cap_k_eq, @rep_cap_done, @rep_cap_body
@rep_cap_body
	%rep_cap_out_base =l add %out, %o
	%rep_cap_out_addr =l add %rep_cap_out_base, %rep_cap_k
	storeb %rb, %rep_cap_out_addr
	%rep_cap_k_next =l add %rep_cap_k, 1
	jmp @rep_cap_fill
@rep_cap_done
	ret %retm1
@srcerr
	ret %retm2
@done
	ret %o
}
