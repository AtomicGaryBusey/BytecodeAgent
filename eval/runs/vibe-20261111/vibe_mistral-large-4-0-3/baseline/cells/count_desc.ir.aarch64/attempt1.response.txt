export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret0, @init
@ret0
	%zero =l copy 0
	ret %zero
@init
	%i =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%ip1 =l add %i, 1
	%cond =w csltl %ip1, %n
	jnz %cond, @body, @done
@body
	%off =l shl %i, 3
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	%gtl =l extsw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
