export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %zero =l copy 0
  %one =l copy 1
  %eight =l copy 8
  %np1 =l add %n, %one
  %zp =l alloc8 8
  %ip =l alloc8 8
  %jp =l alloc8 8
  %dp =l alloc8 8
  storel %zero, %zp
  storel %zero, %ip
  jmp @zloop
@zloop
  %j0 =l loadl %zp
  %zc =w csltl %j0, %np1
  jnz %zc, @zbody, @zdone
@zbody
  %off0 =l mul %j0, %eight
  %addr0 =l add %row, %off0
  storel %zero, %addr0
  %j0n =l add %j0, %one
  storel %j0n, %zp
  jmp @zloop
@zdone
  jmp @oloop
@oloop
  %i =l loadl %ip
  %oc =w csltl %i, %m
  jnz %oc, @obody, @odone
@obody
  storel %zero, %dp
  %xaddr =l add %x, %i
  %xi =w loadub %xaddr
  storel %one, %jp
  jmp @iloop
@iloop
  %j =l loadl %jp
  %ic =w csltl %j, %np1
  jnz %ic, @ibody, @idone
@ibody
  %offj =l mul %j, %eight
  %addrj =l add %row, %offj
  %up =l loadl %addrj
  %jm1 =l sub %j, %one
  %yaddr =l add %y, %jm1
  %yj =w loadub %yaddr
  %eq =w ceqw %xi, %yj
  jnz %eq, @match, @nomatch
@match
  %diag =l loadl %dp
  %valm =l add %diag, %one
  storel %valm, %addrj
  jmp @after
@nomatch
  %offm =l mul %jm1, %eight
  %addrm =l add %row, %offm
  %left =l loadl %addrm
  %lt =w csltl %up, %left
  jnz %lt, @useleft, @useup
@useleft
  storel %left, %addrj
  jmp @after
@useup
  storel %up, %addrj
  jmp @after
@after
  storel %up, %dp
  %jn =l add %j, %one
  storel %jn, %jp
  jmp @iloop
@idone
  %in =l add %i, %one
  storel %in, %ip
  jmp @oloop
@odone
  %offn =l mul %n, %eight
  %resaddr =l add %row, %offn
  %res =l loadl %resaddr
  ret %res
}
