export function l $bmin_empty(l %a, l %n) {
@start
    %result =l 0x7fffffffffffffff
    %i =l 0

@loop
    %done =w cgeql %i, %n
    jnz %done, @end
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %is_less =w csltl %val, %result
    jnz %is_less, @update
    jmp @next

@update
    %result = %val

@next
    %i =l add %i, 1
    jmp @loop

@end
    ret %result
}
