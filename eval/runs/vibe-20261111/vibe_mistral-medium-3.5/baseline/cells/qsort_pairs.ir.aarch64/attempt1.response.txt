export function l $qsort_pairs(l %a, l %n) {
@start
    %n_le_1 = csle l %n, 1
    jnz %n_le_1, @base
    %pivot_idx = sar %n, 1
    %pivot_off = shl %pivot_idx, 1
    %last_off = subl shl %n, 1, 2
    %pivot_key = loadl addl %a, %pivot_off
    %pivot_val = loadl addl %a, %pivot_off, 1
    %last_key = loadl addl %a, %last_off
    %last_val = loadl addl %a, %last_off, 1
    storel %pivot_key, addl %a, %last_off
    storel %pivot_val, addl %a, %last_off, 1
    storel %last_key, addl %a, %pivot_off
    storel %last_val, addl %a, %pivot_off, 1
    %i = subl 0, 1
    %j = 0
@loop
    %j_lt_nm1 = csltl %j, subl %n, 1
    jz %j_lt_nm1, @end_loop
    %j_key = loadl addl %a, shl %j, 1
    %j_val = loadl addl %a, shl %j, 1, 1
    %lt_key = csltl %j_key, %pivot_key
    %eq_key = ceql %j_key, %pivot_key
    %lt_val = csltl %j_val, %pivot_val
    %and_val = andw %eq_key, %lt_val
    %less = orw %lt_key, %and_val
    jz %less, @next
    %i = addl %i, 1
    %i_key = loadl addl %a, shl %i, 1
    %i_val = loadl addl %a, shl %i, 1, 1
    storel %j_key, addl %a, shl %i, 1
    storel %j_val, addl %a, shl %i, 1, 1
    storel %i_key, addl %a, shl %j, 1
    storel %i_val, addl %a, shl %j, 1, 1
@next
    %j = addl %j, 1
    jump @loop
@end_loop
    %i1 = addl %i, 1
    %i1_key = loadl addl %a, shl %i1, 1
    %i1_val = loadl addl %a, shl %i1, 1, 1
    storel %pivot_key, addl %a, shl %i1, 1
    storel %pivot_val, addl %a, shl %i1, 1, 1
    storel %i1_key, addl %a, %last_off
    storel %i1_val, addl %a, %last_off, 1
    %r1 =l call $qsort_pairs(l %a, l %i1)
    %right_ptr = addl %a, shl %i1, 1, 2
    %n_right = subl %n, %i1, 1
    %r2 =l call $qsort_pairs(l %right_ptr, l %n_right)
    ret 0
@base
    ret 0
}
