export function l $qsort_pairs(l %a, l %n) {
@start
    %n_lt_2 = cltl %n, 2
    jnz %n_lt_2, @base
    %pivot_idx = sar %n, 1
    %pivot_byte_off = shl %pivot_idx, 4
    %last_idx = subl %n, 1
    %last_byte_off = shl %last_idx, 4
    %pivot_key = loadl addl %a, %pivot_byte_off
    %pivot_val = loadl addl %a, %pivot_byte_off, 8
    %last_key = loadl addl %a, %last_byte_off
    %last_val = loadl addl %a, %last_byte_off, 8
    storel %pivot_key, addl %a, %last_byte_off
    storel %pivot_val, addl %a, %last_byte_off, 8
    storel %last_key, addl %a, %pivot_byte_off
    storel %last_val, addl %a, %pivot_byte_off, 8
    %i = subl 0, 1
    %j = 0
@loop
    %nm1 = subl %n, 1
    %j_lt_nm1 = cltl %j, %nm1
    jz %j_lt_nm1, @end_loop
    %j_off = shl %j, 4
    %j_key = loadl addl %a, %j_off
    %j_val = loadl addl %a, %j_off, 8
    %lt_key = cltl %j_key, %pivot_key
    %eq_key = ceql %j_key, %pivot_key
    %lt_val = cltl %j_val, %pivot_val
    %and_val = andw %eq_key, %lt_val
    %less = orw %lt_key, %and_val
    jz %less, @next
    %i = addl %i, 1
    %i_off = shl %i, 4
    %i_key = loadl addl %a, %i_off
    %i_val = loadl addl %a, %i_off, 8
    storel %j_key, addl %a, %i_off
    storel %j_val, addl %a, %i_off, 8
    storel %i_key, addl %a, %j_off
    storel %i_val, addl %a, %j_off, 8
@next
    %j = addl %j, 1
    jump @loop
@end_loop
    %i1 = addl %i, 1
    %i1_off = shl %i1, 4
    %i1_key = loadl addl %a, %i1_off
    %i1_val = loadl addl %a, %i1_off, 8
    storel %pivot_key, addl %a, %i1_off
    storel %pivot_val, addl %a, %i1_off, 8
    storel %i1_key, addl %a, %last_byte_off
    storel %i1_val, addl %a, %last_byte_off, 8
    %r1 =l call $qsort_pairs(l %a, l %i1)
    %i1_plus_1 = addl %i1, 1
    %right_ptr = addl %a, shl %i1_plus_1, 4
    %n_right = subl %n, %i1_plus_1
    %r2 =l call $qsort_pairs(l %right_ptr, l %n_right)
    ret 0
@base
    ret 0
}
