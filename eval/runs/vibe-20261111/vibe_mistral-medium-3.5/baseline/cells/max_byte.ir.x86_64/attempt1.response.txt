export function l $max_byte(l %p, l %n) {
@start
	%zero =l copy 0
	%max =l copy %zero
	%i =l copy %zero
	%limit =l copy %n
	jmp @loop_test
@loop_test
	%done =w cuge %i, %limit
	jnz %done, @return, @loop_body
@loop_body
	%addr =l add %p, %i
	%val =l loadub %addr
	%is_greater =w cugt %val, %max
	jnz %is_greater, @update, @loop_inc
@update
	%max =l copy %val
@loop_inc
	%i =l add %i, 1
	jmp @loop_test
@return
	ret %max
}
