export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @loop_test
@loop_test
	%more =w cultl %i, %n
	jnz %more, @loop_body, @return
@loop_body
	%addr =l add %p, %i
	%val =l loadub %addr
	%is_greater =w cultl %max, %val
	jnz %is_greater, @update, @loop_inc
@update
	%max =l copy %val
@loop_inc
	%i =l add %i, 1
	jmp @loop_test
@return
	ret %max
}
