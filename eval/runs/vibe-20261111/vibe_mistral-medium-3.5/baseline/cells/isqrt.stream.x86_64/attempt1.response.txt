isqrt:
  test rdi, rdi
  jz .zero
  xor esi, esi
  mov edx, 1
  shl rdx, 32
.loop:
  cmp rsi, rdx
  jge .done
  lea rax, [rsi + rdx + 1]
  shr rax, 1
  mov r8, rax
  mov r9, rdx
  mul r8
  test rdx, rdx
  jnz .toohigh
  cmp rax, rdi
  ja .toohigh
  mov rsi, r8
  mov rdx, r9
  jmp .loop
.toohigh:
  dec r8
  mov rdx, r8
  jmp .loop
.done:
  mov rax, rsi
  ret
.zero:
  xor eax, eax
  ret
