export function l $max_byte(l %p, l %n) {
@start
    %max =l copy 0
    %i =l copy 0
    jmp @test
@test
    %not_done =w csltl %i, %n
    jnz %not_done, @body, @return
@body
    %addr =l add %p, %i
    %byte =l loadub %addr
    %gt =w csltl %max, %byte
    jnz %gt, @update, @next
@update
    %max =l copy %byte
@next
    %i =l add %i, 1
    jmp @test
@return
    ret %max
}
