export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  %offset =l mul %n, 8
  %src_end =l add %src, %offset
  %src_ptr =l copy %src
  %dst_ptr =l copy %dst
  %count =l copy 0
  jump @check

@loop
  %val =l loadl %src_ptr
  %result =l call %f(l %val)
  storel %result, %dst_ptr
  %cmp =w csltl %result, %val
  %cmp_l =l zx %cmp
  %count =l add %count, %cmp_l
  %src_ptr =l add %src_ptr, 8
  %dst_ptr =l add %dst_ptr, 8

@check
  %cond =w csltl %src_ptr, %src_end
  jnz %cond, @loop, @done

@done
  retl %count
}
