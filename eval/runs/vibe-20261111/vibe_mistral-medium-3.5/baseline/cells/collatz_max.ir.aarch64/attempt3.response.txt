export function l $collatz_max(l %n) {
@start
%max =l 0
%k =l 1
%n_plus_1 =l add %n 1
jmp @loop_k
@loop_k
%steps =l 0
%x =l %k
jmp @loop_collatz
@loop_collatz
%is_one =w ceq %x 1
jnz %is_one @end_collatz
%parity =l and %x 1
%is_even =w ceq %parity 0
jnz %is_even @even
jmp @odd
@odd
%tmp =l mul %x 3
%x =l add %tmp 1
jmp @step
@even
%x =l srl %x 1
jmp @step
@step
%steps =l add %steps 1
jmp @loop_collatz
@end_collatz
%is_max =w cslt %max %steps
jnz %is_max @update_max
jmp @next_k
@update_max
%max =l %steps
jmp @next_k
@next_k
%k =l add %k 1
%cond =w cslt %k %n_plus_1
jnz %cond @loop_k
retl %max
}
