export function l $collatz_max(l %n) {
@start
%one =l copy 1
%three =l copy 3
%max =l copy 0
%k =l copy %one
%n_plus_1 =l add %n %one
jmp @loop_k
@loop_k
%steps =l copy 0
%x =l copy %k
jmp @loop_collatz
@loop_collatz
%is_one =w ceql %x %one
jnz %is_one @end_collatz
%parity =l and %x %one
%is_even =w ceql %parity 0
jnz %is_even @even
jmp @odd
@odd
%tmp =l mul %x %three
%x =l add %tmp %one
jmp @step
@even
%x =l srl %x %one
jmp @step
@step
%steps =l add %steps %one
jmp @loop_collatz
@end_collatz
%is_max =w csltl %max %steps
jnz %is_max @update_max
jmp @next_k
@update_max
%max =l copy %steps
jmp @next_k
@next_k
%k =l add %k %one
%cond =w csltl %k %n_plus_1
jnz %cond @loop_k
retl %max
}
