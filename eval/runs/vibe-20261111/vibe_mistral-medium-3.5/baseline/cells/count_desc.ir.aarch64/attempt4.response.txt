export function l $count_desc(l %a, l %n) {
@start
    %n8 =l mul %n, 8
    %end =l add %a, %n8
    %end_prev =l sub %end, 8
    %p =l %a
    %count =l iconst 0
    jmp @loop_cond

@loop_cond
    %cond =w cslt %p, %end_prev
    jnz %cond, @loop_body, @end

@loop_body
    %x =l loadl %p
    %y =l loadl %p, 8
    %is_desc =w cgt %x, %y
    %is_desc_l =l zxtw %is_desc
    %count =l add %count, %is_desc_l
    %p =l add %p, 8
    jmp @loop_cond

@end
    ret %count
}
