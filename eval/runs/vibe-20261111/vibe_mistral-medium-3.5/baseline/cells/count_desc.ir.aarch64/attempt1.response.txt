export function l $count_desc(l %a, l %n) {
@start
    %end =l add %a, mul %n, 8
    %end_prev =l sub %end, 8
    %p =l %a
    %count =l 0
    jmp @loop_cond

@loop_cond
    %cond =w csltl %p, %end_prev
    jnz %cond, @loop_body, @end

@loop_body
    %x =l loadl %p
    %y =l loadl add %p, 8
    %is_desc =w cgtl %x, %y
    %count =l add %count, %is_desc
    %p =l add %p, 8
    jmp @loop_cond

@end
    ret %count
}
