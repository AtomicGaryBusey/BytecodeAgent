memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.le .Lforward
    lsl x3, x2, #3
    add x4, x0, x3
    add x5, x1, x3
    subs x2, x2, #1
.Lbackward_loop:
    ldr x6, [x5, #-8]!
    str x6, [x4, #-8]!
    subs x2, x2, #1
    b.ge .Lbackward_loop
    b .Ldone
.Lforward:
    subs x2, x2, #1
.Lforward_loop:
    ldr x6, [x1], #8
    str x6, [x0], #8
    subs x2, x2, #1
    b.ge .Lforward_loop
.Ldone:
    mov x0, #0
    ret
