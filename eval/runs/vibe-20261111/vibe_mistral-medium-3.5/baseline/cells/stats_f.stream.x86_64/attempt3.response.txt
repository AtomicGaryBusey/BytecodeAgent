stats_f:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 24
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  xor rbx, rbx
  xor rbp, rbp
  mov rax, 0x7fffffffffffffff
  mov qword ptr [rsp+0], rax
  mov rax, 0x8000000000000000
  mov qword ptr [rsp+8], rax
  mov qword ptr [rsp+16], 0
.Lloop:
  cmp rbx, r13
  jge .Ldone
  mov rdi, qword ptr [r12 + rbx*8]
  call r14
  cmp rax, qword ptr [rsp+0]
  jge .Lnot_min
  mov qword ptr [rsp+0], rax
.Lnot_min:
  cmp rax, qword ptr [rsp+8]
  jle .Lnot_max
  mov qword ptr [rsp+8], rax
.Lnot_max:
  add rbp, rax
  test rax, 1
  jz .Leven
  inc qword ptr [rsp+16]
.Leven:
  inc rbx
  jmp .Lloop
.Ldone:
  mov rax, qword ptr [rsp+0]
  mov qword ptr [r15+0], rax
  mov rax, qword ptr [rsp+8]
  mov qword ptr [r15+8], rax
  mov qword ptr [r15+16], rbp
  mov rax, qword ptr [rsp+16]
  add rsp, 24
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
