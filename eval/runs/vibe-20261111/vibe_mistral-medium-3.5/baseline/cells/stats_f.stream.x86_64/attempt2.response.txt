stats_f:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 8
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  mov rbx, 0
  mov rbp, 0
  mov rax, 0x7fffffffffffffff
  mov qword ptr [rsp+0], rax
  mov rax, 0x8000000000000000
  mov qword ptr [rsp+8], rax
  mov qword ptr [rsp+16], 0
.Lloop:
  cmp rbx, r13
  jge .Ldone
  mov rdi, qword ptr [r12 + rbx*8]
  call r14
  mov rcx, rax
  cmp rcx, qword ptr [rsp+0]
  jl .Lupdate_min
.Lnot_min:
  cmp rcx, qword ptr [rsp+8]
  jg .Lupdate_max
.Lnot_max:
  add rbp, rcx
  test rcx, 1
  jz .Leven
  add qword ptr [rsp+16], 1
.Leven:
  add rbx, 1
  jmp .Lloop
.Lupdate_min:
  mov qword ptr [rsp+0], rcx
  jmp .Lnot_min
.Lupdate_max:
  mov qword ptr [rsp+8], rcx
  jmp .Lnot_max
.Ldone:
  mov rax, qword ptr [rsp+16]
  mov rcx, qword ptr [rsp+0]
  mov qword ptr [r15+0], rcx
  mov rcx, qword ptr [rsp+8]
  mov qword ptr [r15+8], rcx
  mov qword ptr [r15+16], rbp
  add rsp, 8
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
