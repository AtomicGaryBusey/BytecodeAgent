export function l $count_nonzero_prefix(l %a, l %n) {
@start
  %p =l copy %a
  %i =l copy 0
@loop
  %cond =w csltl %i, %n
  jnz %cond, @check, @end
@check
  %val =l loadl %p
  %is_zero =w ceql %val, 0
  jnz %is_zero, @end, @increment
@increment
  %i =l addl %i, 1
  %p =l addl %p, 8
  jump @loop
@end
  return %i
}
