export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%count =l copy 0
	%i =l copy 0
	jmp @condition

@condition
	%cond =w csltl %i, %n
	jnz %cond, @body, @end

@body
	%offset =l mul %i, 8
	%src_ptr =l add %src, %offset
	%src_val =l loadl %src_ptr
	%res =l call %f(l %src_val)
	%dst_ptr =l add %dst, %offset
	storel %res, %dst_ptr
	%is_less =w csltl %res, %src_val
	%count =l add %count, %is_less
	%i =l add %i, 1
	jmp @condition

@end
	ret %count
}
