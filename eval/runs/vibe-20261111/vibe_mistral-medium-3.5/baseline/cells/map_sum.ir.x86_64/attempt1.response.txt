export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return, @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @return
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%res =l call %f(l %val)
	%sum =l add %sum, %res
	%i =l add %i, 1
	jmp @loop
@return
	ret %sum
}
