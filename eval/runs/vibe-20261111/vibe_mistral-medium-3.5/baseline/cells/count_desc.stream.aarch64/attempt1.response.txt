count_desc:
    cmp x1, #2
    b.lt .Lreturn_zero
    mov x2, x0
    mov x3, #0
    sub x4, x1, #1
.Lloop:
    ldr x5, [x2]
    ldr x6, [x2, #8]
    cmp x5, x6
    b.le .Lno_count
    add x3, x3, #1
.Lno_count:
    add x2, x2, #8
    subs x4, x4, #1
    b.gt .Lloop
    mov x0, x3
    ret
.Lreturn_zero:
    mov x0, #0
    ret
