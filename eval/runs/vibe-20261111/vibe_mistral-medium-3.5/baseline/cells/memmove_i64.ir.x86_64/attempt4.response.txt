export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %zero =l 0
  %n_zero = ceq %n, %zero
  jnz %n_zero, @return_zero, @check_direction
@check_direction
  %dst_le_src = csle %dst, %src
  jnz %dst_le_src, @copy_forward, @copy_backward
@copy_forward
  %i =l 0
@loop_forward
  %cond = cslt %i, %n
  jnz %cond, @body_forward, @return_zero
@body_forward
  %val = loadl %src[%i]
  storel %val, %dst[%i]
  %i = addl %i, 1
  jmp @loop_forward
@copy_backward
  %i = subl %n, 1
@loop_backward
  %cond = csge %i, %zero
  jnz %cond, @body_backward, @return_zero
@body_backward
  %val = loadl %src[%i]
  storel %val, %dst[%i]
  %i = subl %i, 1
  jmp @loop_backward
@return_zero
  ret 0
}
