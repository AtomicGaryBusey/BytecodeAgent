export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %n_zero = ceql %n, 0
  jnz %n_zero, @return_zero, @check_overlap
@check_overlap
  %dst_lt_src = csltl %dst, %src
  jnz %dst_lt_src, @copy_forward, @check_dst_gt_src
@check_dst_gt_src
  %dst_gt_src = csgtl %dst, %src
  jnz %dst_gt_src, @copy_backward, @return_zero
@copy_forward
  %i = 0
@loop_forward
  %cond = csltl %i, %n
  jnz %cond, @body_forward, @return_zero
@body_forward
  %val = loadl %src[%i]
  storel %val, %dst[%i]
  %i = addl %i, 1
  jmp @loop_forward
@copy_backward
  %i = subl %n, 1
@loop_backward
  %cond = csgel %i, 0
  jnz %cond, @body_backward, @return_zero
@body_backward
  %val = loadl %src[%i]
  storel %val, %dst[%i]
  %i = subl %i, 1
  jmp @loop_backward
@return_zero
  ret 0
}
