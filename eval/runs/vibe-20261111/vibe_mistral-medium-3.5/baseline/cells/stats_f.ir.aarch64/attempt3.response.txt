export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %y0 =l call %f(l loadl %a)
    %min = %y0
    %max = %y0
    %sum = %y0
    %odd_count = and %y0, 1
    %i = 1
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body
    jump @done
@body
    %a_val =l loadl add %a, mul %i, 8
    %y =l call %f(l %a_val)
    %cmp_min =w csltl %y, %min
    jnz %cmp_min, @update_min
    jump @check_max
@update_min
    %min = %y
@check_max
    %cmp_max =w csgtl %y, %max
    jnz %cmp_max, @update_max
    jump @update_sum
@update_max
    %max = %y
@update_sum
    %sum = add %sum, %y
    %odd_count = add %odd_count, and %y, 1
    %i = add %i, 1
    jump @loop
@done
    storel %min, %out
    storel %max, add %out, 8
    storel %sum, add %out, 16
    ret %odd_count
}
