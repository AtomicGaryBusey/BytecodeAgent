export function l $stats_f(l %a, l %n, l %f, l %out) {
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %min =l %y0
    %max =l %y0
    %sum =l %y0
    %odd_count =l and %y0, 1
    %i =l 1
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body
    jump @done
@body
    %offset =l mul %i, 8
    %a_ptr =l add %a, %offset
    %a_val =l loadl %a_ptr
    %y =l call %f(l %a_val)
    %cmp_min =w csltl %y, %min
    jnz %cmp_min, @update_min
    jump @check_max
@update_min
    %min =l %y
@check_max
    %cmp_max =w csgtl %y, %max
    jnz %cmp_max, @update_max
    jump @update_sum
@update_max
    %max =l %y
@update_sum
    %sum =l add %sum, %y
    %is_odd =l and %y, 1
    %odd_count =l add %odd_count, %is_odd
    %i =l add %i, 1
    jump @loop
@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd_count
}
