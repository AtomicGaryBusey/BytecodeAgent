export function l $count_ugt(l %a, l %n, l %t) {
@start
	%count =l 0
	%i =l 0
@loop
	%cond =w cslt %i, %n
	jnz %cond, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%is_gt =w cugt %val, %t
	jnz %is_gt, @incr, @next
@incr
	%count =l add %count, 1
@next
	%i =l add %i, 1
	jump @loop
@done
	ret %count
}
