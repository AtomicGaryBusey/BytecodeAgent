export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %val0 =l loadl %a
    %y0 =l call %f(l %val0)
    %min =l copy %y0
    %max =l copy %y0
    %sum =l copy %y0
    %odd0 =l and %y0, 1
    %odd_count =l copy %odd0
    %i =l copy 1
    jmp @loop %i %min %max %sum %odd_count

@loop
    %i =l copy %0
    %min =l copy %1
    %max =l copy %2
    %sum =l copy %3
    %odd_count =l copy %4
    %lt =w csltl %i, %n
    jnz %lt, @body, @done

@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %val =l loadl %addr
    %y =l call %f(l %val)
    %sum_next =l add %sum, %y
    %odd =l and %y, 1
    %odd_count_next =l add %odd_count, %odd
    %ylt =w csltl %y, %min
    %min_next =l sel %ylt, %y, %min
    %ygt =w csgtl %y, %max
    %max_next =l sel %ygt, %y, %max
    %i_next =l add %i, 1
    jmp @loop %i_next %min_next %max_next %sum_next %odd_count_next

@done
    storel %min, %out
    storel %max, %out + 8
    storel %sum, %out + 16
    ret %odd_count
}
