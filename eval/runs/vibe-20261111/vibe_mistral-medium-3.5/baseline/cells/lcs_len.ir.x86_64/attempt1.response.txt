export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start:
    %zero =l copy 0
    %j =l copy 0
    %n1 =l add %n, 1
@init_loop:
    %cond =l csltl %j, %n1
    jnz %cond, @init_body, @init_done
@init_body:
    %byte_offset =l shl %j, 3
    storel %zero, %row, %byte_offset
    %j =l add %j, 1
    jmp @init_loop
@init_done:
    %i =l copy 0
@outer_loop:
    %cond_i =l csltl %i, %m
    jnz %cond_i, @outer_body, @done
@outer_body:
    %diag =l copy 0
    %j =l copy 1
@inner_loop:
    %cond_j =l csltl %j, %n1
    jnz %cond_j, @inner_body, @next_i
@inner_body:
    %j_offset =l shl %j, 3
    %up =l loadl %row, %j_offset
    %jm1_offset =l add %j_offset, -8
    %row_jm1 =l loadl %row, %jm1_offset
    %y_offset =l add %j, -1
    %y_byte =w loadub %y, %y_offset
    %x_byte =w loadub %x, %i
    %eq =w ceqb %x_byte, %y_byte
    %then_val =l add %diag, 1
    %else_val =l smax %up, %row_jm1
    %new_val =l select %eq, %then_val, %else_val
    storel %new_val, %row, %j_offset
    %diag =l copy %up
    %j =l add %j, 1
    jmp @inner_loop
@next_i:
    %i =l add %i, 1
    jmp @outer_loop
@done:
    %n_offset =l shl %n, 3
    %result =l loadl %row, %n_offset
    ret %result
}
