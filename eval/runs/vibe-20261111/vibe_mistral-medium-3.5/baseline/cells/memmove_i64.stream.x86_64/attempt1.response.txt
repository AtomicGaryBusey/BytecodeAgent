memmove_i64:
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    jl .Lcopy_forward
    dec rdx
.Lbackward_loop:
    mov rax, [rsi + rdx*8]
    mov [rdi + rdx*8], rax
    dec rdx
    jns .Lbackward_loop
    jmp .Ldone
.Lcopy_forward:
    xor rcx, rcx
.Lforward_loop:
    mov rax, [rsi + rcx*8]
    mov [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lforward_loop
.Ldone:
    xor rax, rax
    ret
