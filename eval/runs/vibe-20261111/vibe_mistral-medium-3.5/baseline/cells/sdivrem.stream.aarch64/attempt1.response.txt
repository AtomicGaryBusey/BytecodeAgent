sdivrem:
    cbz x1, b_zero
    movz x3, #0x8000, lsl #48
    cmp x0, x3
    b.ne do_div
    cmn x1, #1
    b.eq special
do_div:
    sdiv x3, x0, x1
    msub x4, x3, x1, x0
    str x4, [x2]
    mov x0, x3
    ret
special:
    str xzr, [x2]
    mov x0, x3
    ret
b_zero:
    str x0, [x2]
    mov x0, xzr
    ret
