export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
%written =l 0
%i =l 0
@loop
%done =w cugel %i, %n
jnz %done, @end, @process
@process
%b =w loadub %src, %i
%run_len =l 1
%j =l %i
@run_loop
%j_next =l add %j, 1
%j_ok =w cultl %j_next, %n
jnz %j_ok, @run_check, @run_done
@run_check
%b_next =w loadub %src, %j_next
%same =w ceql %b_next, %b
%run_lt_128 =w cultl %run_len, 128
%both =w and %same, %run_lt_128
jnz %both, @run_extend, @run_done
@run_extend
%run_len =l add %run_len, 1
%j =l %j_next
jmp @run_loop
@run_done
%is_repeat =w cugel %run_len, 3
jnz %is_repeat, @emit_repeat, @emit_literal
@emit_repeat
%remaining =l sub %cap, %written
%space_ok =w cugel %remaining, 2
jnz %space_ok, @do_repeat, @return_error
@do_repeat
%header =w sub 257, %run_len
%out_ptr =l add %out, %written
storeb %header, %out_ptr
storeb %b, %out_ptr, 1
%written =l add %written, 2
%i =l add %i, %run_len
jmp @loop
@emit_literal
%literal_len =l 0
%j =l %i
@literal_scan
%j_at_end =w cugel %j, %n
%len_full =w ceql %literal_len, 128
%stop_len =w or %j_at_end, %len_full
jnz %stop_len, @literal_stop, @check_run3
@check_run3
%j_plus2 =l add %j, 2
%j_plus2_ok =w cultl %j_plus2, %n
jnz %j_plus2_ok, @check_three_bytes, @literal_include
@check_three_bytes
%b0 =w loadub %src, %j
%b1 =w loadub %src, %j, 1
%b2 =w loadub %src, %j, 2
%eq01 =w ceql %b0, %b1
%eq12 =w ceql %b1, %b2
%is_run3 =w and %eq01, %eq12
jnz %is_run3, @literal_stop, @literal_include
@literal_include
%literal_len =l add %literal_len, 1
%j =l add %j, 1
jmp @literal_scan
@literal_stop
%space_needed =l add %literal_len, 1
%remaining =l sub %cap, %written
%space_ok =w cugel %remaining, %space_needed
jnz %space_ok, @do_literal, @return_error
@do_literal
%header =w sub %literal_len, 1
%out_ptr =l add %out, %written
storeb %header, %out_ptr
%k =l 0
%src_ptr =l add %src, %i
%dst_ptr =l add %out_ptr, 1
jmp @copy_loop
@copy_loop
%copy_done_flag =w ceql %k, %literal_len
jnz %copy_done_flag, @copy_done, @do_copy
@do_copy
%byte =w loadub %src_ptr, %k
storeb %byte, %dst_ptr, %k
%k =l add %k, 1
jmp @copy_loop
@copy_done
%written =l add %written, %space_needed
%i =l add %i, %literal_len
jmp @loop
@return_error
ret -1
@end
ret %written
}
