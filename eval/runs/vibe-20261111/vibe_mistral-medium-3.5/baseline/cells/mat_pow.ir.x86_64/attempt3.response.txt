export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %total = mul %n %n
  %scratch = add %tmp %total
  %k_eq_0 = ceq %k 0
  jnz %k_eq_0 @k_zero
  %i = 0
@copy_a
  %i_lt_total = cslt %i %total
  jnz %i_lt_total @copy_a_body
  jmp @init_out
@copy_a_body
  %val = loadl %a + %i
  storel %val %tmp + %i
  %i = add %i 1
  jmp @copy_a
@init_out
  %i = 0
@init_i
  %i_lt_n = cslt %i %n
  jnz %i_lt_n @init_i_body
  jmp @loop
@init_i_body
  %j = 0
@init_j
  %j_lt_n = cslt %j %n
  jnz %j_lt_n @init_ij
  jmp @init_i_next
@init_ij
  %mul1 = mul %i %n
  %idx = add %mul1 %j
  %val = ceq %i %j
  %val = zextw %val
  storel %val %out + %idx
  %j = add %j 1
  jmp @init_j
@init_i_next
  %i = add %i 1
  jmp @init_i
@loop
  %k_gt_0 = cgt %k 0
  jnz %k_gt_0 @loop_body
  jmp @zero_tmp
@loop_body
  %is_odd = and %k 1
  %is_odd = cneq %is_odd 0
  jnz %is_odd @multiply_result
  jmp @square_base
@multiply_result
  %i = 0
@copy_result
  %i_lt_total = cslt %i %total
  jnz %i_lt_total @copy_result_body
  jmp @do_multiply
@copy_result_body
  %val = loadl %out + %i
  storel %val %scratch + %i
  %i = add %i 1
  jmp @copy_result
@do_multiply
  %i = 0
@mul_i
  %i_lt_n = cslt %i %n
  jnz %i_lt_n @mul_i_body
  jmp @square_base
@mul_i_body
  %j = 0
@mul_j
  %j_lt_n = cslt %j %n
  jnz %j_lt_n @mul_j_body
  jmp @mul_i_next
@mul_j_body
  %sum = 0
  %t = 0
@mul_t
  %t_lt_n = cslt %t %n
  jnz %t_lt_n @mul_t_body
  jmp @mul_t_done
@mul_t_body
  %s_mul = mul %i %n
  %s_idx = add %s_mul %t
  %s_val = loadl %scratch + %s_idx
  %b_mul = mul %t %n
  %b_idx = add %b_mul %j
  %b_val = loadl %tmp + %b_idx
  %product = mul %s_val %b_val
  %sum = add %sum %product
  %t = add %t 1
  jmp @mul_t
@mul_t_done
  %o_mul = mul %i %n
  %o_idx = add %o_mul %j
  storel %sum %out + %o_idx
  %j = add %j 1
  jmp @mul_j
@mul_i_next
  %i = add %i 1
  jmp @mul_i
@square_base
  %i = 0
@copy_base
  %i_lt_total = cslt %i %total
  jnz %i_lt_total @copy_base_body
  jmp @do_square
@copy_base_body
  %val = loadl %tmp + %i
  storel %val %scratch + %i
  %i = add %i 1
  jmp @copy_base
@do_square
  %i = 0
@sq_i
  %i_lt_n = cslt %i %n
  jnz %i_lt_n @sq_i_body
  jmp @square_done
@sq_i_body
  %j = 0
@sq_j
  %j_lt_n = cslt %j %n
  jnz %j_lt_n @sq_j_body
  jmp @sq_i_next
@sq_j_body
  %sum = 0
  %t = 0
@sq_t
  %t_lt_n = cslt %t %n
  jnz %t_lt_n @sq_t_body
  jmp @sq_t_done
@sq_t_body
  %s_mul = mul %i %n
  %s_idx = add %s_mul %t
  %s_val = loadl %scratch + %s_idx
  %b_mul = mul %t %n
  %b_idx = add %b_mul %j
  %b_val = loadl %tmp + %b_idx
  %product = mul %s_val %b_val
  %sum = add %sum %product
  %t = add %t 1
  jmp @sq_t
@sq_t_done
  %t_mul = mul %i %n
  %t_idx = add %t_mul %j
  storel %sum %tmp + %t_idx
  %j = add %j 1
  jmp @sq_j
@sq_i_next
  %i = add %i 1
  jmp @sq_i
@square_done
  %k = shr %k 1
  jmp @loop
@zero_tmp
  %i = 0
  %tmp_size = mul %total 2
@zero_loop
  %i_lt_size = cslt %i %tmp_size
  jnz %i_lt_size @zero_body
  jmp @return
@zero_body
  storel 0 %tmp + %i
  %i = add %i 1
  jmp @zero_loop
@return
  ret 0
@k_zero
  %i = 0
@kz_i
  %i_lt_n = cslt %i %n
  jnz %i_lt_n @kz_i_body
  jmp @kz_init_done
@kz_i_body
  %j = 0
@kz_j
  %j_lt_n = cslt %j %n
  jnz %j_lt_n @kz_ij
  jmp @kz_i_next
@kz_ij
  %mul1 = mul %i %n
  %idx = add %mul1 %j
  %val = ceq %i %j
  %val = zextw %val
  storel %val %out + %idx
  %j = add %j 1
  jmp @kz_j
@kz_i_next
  %i = add %i 1
  jmp @kz_i
@kz_init_done
  %i = 0
  %tmp_size = mul %total 2
@kz_zero
  %i_lt_size = cslt %i %tmp_size
  jnz %i_lt_size @kz_zero_body
  jmp @return
@kz_zero_body
  storel 0 %tmp + %i
  %i = add %i 1
  jmp @kz_zero
}
