dot_i32_simd:
  xor rax, rax
  test rdx, rdx
  jz .Ldone
.Lvec_loop:
  cmp rdx, 4
  jb .Lscalar
  movdqu xmm0, [rdi]
  movdqu xmm1, [rsi]
  pmovsxdq xmm2, xmm0
  pmovsxdq xmm3, xmm1
  pmuldq xmm2, xmm3
  pslldq xmm0, 8
  pslldq xmm1, 8
  pmovsxdq xmm4, xmm0
  pmovsxdq xmm5, xmm1
  pmuldq xmm4, xmm5
  paddq xmm2, xmm4
  movq rcx, xmm2
  psrldq xmm2, 8
  movq r8, xmm2
  add rax, rcx
  add rax, r8
  add rdi, 16
  add rsi, 16
  sub rdx, 4
  jmp .Lvec_loop
.Lscalar:
  test rdx, rdx
  jz .Ldone
.Lscalar_loop:
  movsxd rcx, dword ptr [rdi]
  movsxd r8, dword ptr [rsi]
  imul rcx, r8
  add rax, rcx
  add rdi, 4
  add rsi, 4
  dec rdx
  jnz .Lscalar_loop
.Ldone:
  ret
