export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%s =l %src
	%se =l add %s %n
	%o =l %out
	%oe =l add %o %cap
	jmp @loop
@loop
	%has_header =w cultl %s %se
	jz %has_header @done
	%h =w loadsb %s
	%h_l =l extsw %h
	%s =l add %s 1
	%is_noop =w ceql %h_l -128
	jnz %is_noop @loop
	%is_literal =w csleql 0 %h_l
	jnz %is_literal @literal
	jmp @repeat
@literal
	%literal_count =l add %h_l 1
	%src_after =l add %s %literal_count
	%src_fail =w cugtl %src_after %se
	jnz %src_fail @src_error
	%out_after =l add %o %literal_count
	%out_ok =w culel %out_after %oe
	jnz %out_ok @copy_full_literal
	%copy_count =l sub %oe %o
	jmp @copy_literal
@copy_full_literal
	%copy_count =l %literal_count
@copy_literal
	%c =l %copy_count
	%ps =l %s
	%po =l %o
@copy_literal_loop
	jz %c @copy_literal_done
	%b =w loadub %ps
	storeb %b %po
	%ps =l add %ps 1
	%po =l add %po 1
	%c =l sub %c 1
	jmp @copy_literal_loop
@copy_literal_done
	%s =l %ps
	%o =l %po
	%overflow =w cnel %copy_count %literal_count
	jnz %overflow @cap_error
	jmp @loop
@repeat
	%repeat_count =l sub 1 %h_l
	%src_fail =w cugel %s %se
	jnz %src_fail @src_error
	%byte =w loadub %s
	%s =l add %s 1
	%out_after =l add %o %repeat_count
	%out_ok =w culel %out_after %oe
	jnz %out_ok @fill_full_repeat
	%fill_count =l sub %oe %o
	jmp @fill_repeat
@fill_full_repeat
	%fill_count =l %repeat_count
@fill_repeat
	%c =l %fill_count
	%p =l %o
@fill_repeat_loop
	jz %c @fill_repeat_done
	storeb %byte %p
	%p =l add %p 1
	%c =l sub %c 1
	jmp @fill_repeat_loop
@fill_repeat_done
	%o =l %p
	%overflow =w cnel %fill_count %repeat_count
	jnz %overflow @cap_error
	jmp @loop
@src_error
	return -2
@cap_error
	return -1
@done
	%written =l sub %o %out
	return %written
}
