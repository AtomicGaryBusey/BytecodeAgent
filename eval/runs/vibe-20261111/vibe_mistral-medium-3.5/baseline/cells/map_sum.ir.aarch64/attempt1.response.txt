export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
	%limit =l copy %n
@loop_check
	%cond =w csltl %i, %limit
	jnz %cond, @loop_body, @done
@loop_body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%res =l call %f(l %val)
	%sum =l add %sum, %res
	%i =l add %i, 1
	jmp @loop_check
@done
	retl %sum
}
