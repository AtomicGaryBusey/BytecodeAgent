  stp x29, x30, [sp, #-16]!
  mov x29, sp
  stp x19, x20, [sp, #-16]!
  stp x21, x22, [sp, #-16]!
  stp x23, x24, [sp, #-16]!

  mov x20, x1

  lsr x2, x20, #1
  lsl x3, x2, #4
  ldr x21, [x19, x3]
  ldr x22, [x19, x3, #8]
  sub x4, x20, #1
  lsl x4, x4, #4
  ldr x5, [x19, x4]
  ldr x6, [x19, x4, #8]

  str x22, [x19, x4, #8]
  str x5, [x19, x3]

  mov x7, #0

.Lloop:
  cmp x7, x24
  b.hs .Lpartition_done
  lsl x8, x7, #4
  ldr x9, [x19, x8]
  ldr x10, [x19, x8, #8]
  cmp x9, x21
  b.lt .Lswap
  b.gt .Lnext
  cmp x10, x22
  b.lt .Lswap
  b .Lnext

  ldr x11, [x19, x8]
  ldr x12, [x19, x8, #8]
  lsl x8, x7, #4
  str x11, [x19, x8]
  str x12, [x19, x8, #8]
  lsl x8, x23, #4
  str x9, [x19, x8]
  str x10, [x19, x8, #8]

.Lnext:
  add x7, x7, #1
  b .Lloop
.Lpartition_done:

  lsl x4, x4, #4
  lsl x8, x23, #4
  ldr x5, [x19, x4]
  ldr x6, [x19, x4, #8]
  ldr x9, [x19, x8]
  ldr x10, [x19, x8, #8]
  str x9, [x19, x4]
  str x10, [x19, x4, #8]
  str x5, [x19, x8]
  str x6, [x19, x8, #8]
  mov x0, x19
  mov x1, x23
  bl qsort_pairs
  add x0, x19, x23, lsl #4
  add x0, x0, #16
  sub x1, x20, x23
  sub x1, x1, #1
  bl qsort_pairs
.Lreturn:

  ldp x23, x24, [sp], #16
  ldp x21, x22, [sp], #16
