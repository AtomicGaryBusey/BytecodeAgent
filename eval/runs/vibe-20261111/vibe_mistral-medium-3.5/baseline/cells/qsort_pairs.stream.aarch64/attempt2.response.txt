qsort_pairs:
  stp x29, x30, [sp, #-16]!
  mov x29, sp
  stp x19, x20, [sp, #-16]!
  stp x21, x22, [sp, #-16]!
  stp x23, x24, [sp, #-16]!
  mov x19, x0
  mov x20, x1
  cmp x20, #1
  ble .Lreturn
  lsr x2, x20, #1
  lsl x3, x2, #4
  add x4, x19, x3
  ldp x5, x6, [x19]
  ldp x7, x8, [x4]
  stp x7, x8, [x19]
  stp x5, x6, [x4]
  ldp x21, x22, [x19]
  mov x23, #-1
  mov x24, x20
.Lpartition:
.Li_loop:
  add x23, x23, #1
  cmp x23, x24
  bge .Lpartition_done
  lsl x4, x23, #4
  add x4, x19, x4
  ldp x5, x6, [x4]
  cmp x5, x21
  b.lt .Li_loop
  b.gt .Lj_loop
  cmp x6, x22
  b.lt .Li_loop
.Lj_loop:
  sub x24, x24, #1
  cmp x23, x24
  bge .Lpartition_done
  lsl x4, x24, #4
  add x4, x19, x4
  ldp x5, x6, [x4]
  cmp x5, x21
  b.gt .Lj_loop
  b.lt .Lcheck_swap
  cmp x6, x22
  b.gt .Lj_loop
.Lcheck_swap:
  cmp x23, x24
  bge .Lpartition_done
  lsl x4, x23, #4
  add x4, x19, x4
  lsl x5, x24, #4
  add x5, x19, x5
  ldp x6, x7, [x4]
  ldp x8, x9, [x5]
  stp x8, x9, [x4]
  stp x6, x7, [x5]
  b .Lpartition
.Lpartition_done:
  mov x0, x19
  add x1, x24, #1
  cmp x1, #1
  bgt .Lsort_left
.Lcheck_right:
  add x2, x24, #1
  lsl x2, x2, #4
  add x0, x19, x2
  sub x1, x20, x24
  sub x1, x1, #1
  cmp x1, #1
  bgt .Lsort_right
  b .Lreturn
.Lsort_left:
  bl qsort_pairs
  b .Lcheck_right
.Lsort_right:
  bl qsort_pairs
.Lreturn:
  mov x0, #0
  ldp x23, x24, [sp], #16
  ldp x21, x22, [sp], #16
  ldp x19, x20, [sp], #16
  ldp x29, x30, [sp], #16
  ret
