qsort_pairs:
  cmp x1, #1
  ble .Lreturn_zero
  sub sp, sp, #64
  stp x19, x20, [sp, #48]
  stp x21, x22, [sp, #32]
  stp x23, x24, [sp, #16]
  stp x29, x30, [sp]
  mov x19, x0
  mov x20, x1
  ldp x21, x22, [x19]
  mov x23, #0
  sub x24, x20, #1
.Lpartition:
.Li_loop:
  cmp x23, x20
  b.ge .Lj_loop
  lsl x4, x23, #4
  add x4, x19, x4
  ldp x5, x6, [x4]
  cmp x5, x21
  b.lt .Li_inc
  b.gt .Lj_loop
  cmp x6, x22
  b.lt .Li_inc
.Lj_loop:
  cmp x24, #0
  blt .Lcheck
  lsl x4, x24, #4
  add x4, x19, x4
  ldp x5, x6, [x4]
  cmp x5, x21
  b.gt .Lj_dec
  b.lt .Lcheck
  cmp x6, x22
  b.gt .Lj_dec
.Lcheck:
  cmp x23, x24
  b.ge .Lpartition_done
  lsl x4, x23, #4
  add x4, x19, x4
  lsl x5, x24, #4
  add x5, x19, x5
  ldp x6, x7, [x4]
  ldp x8, x9, [x5]
  stp x8, x9, [x4]
  stp x6, x7, [x5]
  add x23, x23, #1
  sub x24, x24, #1
  b .Lpartition
.Li_inc:
  add x23, x23, #1
  b .Li_loop
.Lj_dec:
  sub x24, x24, #1
  b .Lj_loop
.Lpartition_done:
  mov x0, x19
  add x1, x24, #1
  cmp x1, #1
  ble .Lsort_right
  bl qsort_pairs
.Lsort_right:
  add x2, x24, #1
  lsl x2, x2, #4
  add x0, x19, x2
  sub x1, x20, x24
  sub x1, x1, #1
  cmp x1, #1
  ble .Lreturn
  bl qsort_pairs
.Lreturn:
  ldp x29, x30, [sp]
  ldp x23, x24, [sp, #16]
  ldp x21, x22, [sp, #32]
  ldp x19, x20, [sp, #48]
  add sp, sp, #64
.Lreturn_zero:
  mov x0, #0
  ret
