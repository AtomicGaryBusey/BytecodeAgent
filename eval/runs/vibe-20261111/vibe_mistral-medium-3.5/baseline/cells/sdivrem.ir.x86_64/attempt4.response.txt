export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%neg_one =l copy -1
	%s63 =l copy 63
	%int64_min =l shl %neg_one, %s63
	%b_zero =w ceql %b, 0
	jnz %b_zero, @b_is_zero
	%a_is_min =w ceql %a, %int64_min
	%b_is_neg_one =w ceql %b, -1
	%both =w and %a_is_min, %b_is_neg_one
	jnz %both, @overflow_case
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@b_is_zero
	storel %a, %rem
	ret 0
@overflow_case
	storel 0, %rem
	ret %int64_min
}
