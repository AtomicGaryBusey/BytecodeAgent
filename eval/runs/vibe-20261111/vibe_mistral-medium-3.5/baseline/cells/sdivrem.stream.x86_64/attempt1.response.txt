sdivrem:
    mov r8, rdx
    test rsi, rsi
    jz .b_zero
    cmp rsi, -1
    jne .do_div
    mov rax, rdi
    neg rax
    jo .special
.do_div:
    mov rax, rdi
    cqo
    mov rcx, rsi
    idiv rcx
    mov [r8], rdx
    ret
.special:
    xor edx, edx
    mov [r8], rdx
    ret
.b_zero:
    xor rax, rax
    mov [r8], rdi
    ret
