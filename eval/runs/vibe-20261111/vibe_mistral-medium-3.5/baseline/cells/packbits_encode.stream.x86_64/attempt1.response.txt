packbits_encode:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jge .Ldone
    mov al, byte ptr [rdi + r8]
    mov r10, 1
    mov r11, r8
    inc r11
.Lrun_loop:
    cmp r11, rsi
    jge .Lrun_done
    cmp r10, 128
    jge .Lrun_done
    cmp byte ptr [rdi + r11], al
    jne .Lrun_done
    inc r10
    inc r11
    jmp .Lrun_loop
.Lrun_done:
    cmp r10, 3
    jge .Lrepeat
    mov r11, r8
.Lliteral_loop:
    cmp r11, rsi
    jge .Lliteral_done
    mov rax, r11
    sub rax, r8
    cmp rax, 128
    jge .Lliteral_done
    mov rax, rsi
    sub rax, 2
    cmp r11, rax
    jge .Lrun_less_than_3
    mov bl, byte ptr [rdi + r11]
    cmp byte ptr [rdi + r11 + 1], bl
    jne .Lrun_less_than_3
    cmp byte ptr [rdi + r11 + 2], bl
    jne .Lrun_less_than_3
    jmp .Lliteral_done
.Lrun_less_than_3:
    inc r11
    jmp .Lliteral_loop
.Lliteral_done:
    mov r10, r11
    sub r10, r8
    lea rax, [r9 + r10 + 1]
    cmp rax, rcx
    jg .Lerror
    dec r10
    mov byte ptr [rdx + r9], r10b
    inc r10
    mov rcx, r10
    mov rax, r8
    lea r10, [r9 + 1]
.Lcopy_loop:
    mov al, byte ptr [rdi + rax]
    mov byte ptr [rdx + r10], al
    inc rax
    inc r10
    dec rcx
    jnz .Lcopy_loop
    mov r9, r10
    mov r8, r11
    jmp .Lloop
.Lrepeat:
    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lerror
    mov bl, 255
    sub bl, r10b
    add bl, 2
    mov byte ptr [rdx + r9], bl
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lloop
.Lerror:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
