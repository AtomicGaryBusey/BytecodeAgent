binary_search:
  cbz x1, .Lnot_found
  mov x9, #0
  sub x10, x1, #1
.Lloop:
  cmp x9, x10
  b.gt .Lnot_found
  sub x11, x10, x9
  lsr x11, x11, #1
  add x11, x9, x11
  ldr x12, [x0, x11, lsl #3]
  cmp x12, x2
  b.eq .Lfound
  b.lt .Lless
  sub x10, x11, #1
  b .Lloop
.Lless:
  add x9, x11, #1
  b .Lloop
.Lfound:
  mov x0, x11
  ret
.Lnot_found:
  mov x0, #-1
  ret
