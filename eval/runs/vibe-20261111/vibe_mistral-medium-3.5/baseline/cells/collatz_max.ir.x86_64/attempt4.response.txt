export function l $collatz_max(l %n) {
@start
  %max l 0
  %k l 1
  jmp @outer_loop_check

@outer_loop_check
  %cond w cslel %k %n
  jnz %cond, @outer_loop_body, @done

@outer_loop_body
  %steps l 0
  %x l %k
@collatz_loop
  %is_one w ceql %x 1
  jnz %is_one, @collatz_done, @collatz_continue
@collatz_continue
  %lsb l andl %x 1
  %is_even w ceql %lsb 0
  jnz %is_even, @even, @odd
@even
  %x l shrl %x 1
  jmp @step
@odd
  %temp l mull %x 3
  %x l addl %temp 1
@step
  %steps l addl %steps 1
  jmp @collatz_loop

@collatz_done
  %is_greater w csgtl %steps %max
  jnz %is_greater, @update_max, @next_k
@update_max
  %max l %steps
@next_k
  %k l addl %k 1
  jmp @outer_loop_check

@done
  retl %max
}
