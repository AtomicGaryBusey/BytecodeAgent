export function l $bmax(l a, l n) {
@start
	%max = loadl a
	%p = add a, 8
	%end = add a, mul n, 8

@loop
	%cond = cultl %p, %end
	jnz %cond, @body, @end

@body
	%elem = loadl %p
	%is_greater = csltl %max, %elem
	jnz %is_greater, @update, @next

@update
	%max = copy %elem

@next
	%p = add %p, 8
	jmp @loop

@end
	return %max
}
