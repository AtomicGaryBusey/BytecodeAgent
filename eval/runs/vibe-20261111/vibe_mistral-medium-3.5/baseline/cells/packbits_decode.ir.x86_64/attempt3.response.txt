export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%src_cur =l copy %src
	%src_end =l add %src, %n
	%out_cur =l copy %out
	%out_end =l add %out, %cap

@loop
	%src_done =w csgel %src_cur, %src_end
	jnz %src_done, @compute_result, @read_header

@read_header
	%h =w loadsb %src_cur
	%src_cur =l add %src_cur, 1

	%is_nop =w cseq %h, -128
	jnz %is_nop, @loop, @check_type

@check_type
	%is_literal =w csgel %h, 0
	jnz %is_literal, @literal, @repeat

@literal
	%h_long =l sxtw %h
	%len =l add %h_long, 1

	%rem_src =l sub %src_end, %src_cur
	%src_ok =w csgel %rem_src, %len
	jnz %src_ok, @lit_src_ok, @error2

@lit_src_ok
	%rem_out =l sub %out_end, %out_cur
	%out_ok =w csgel %rem_out, %len
	jnz %out_ok, @lit_copy_full, @lit_copy_partial

@lit_copy_full
	%k =l copy 0
@lit_copy_loop
	%cond =w csltl %k, %len
	jnz %cond, @lit_copy_body, @lit_done
@lit_copy_body
	%b =w loadub %src_cur
	storeb %b, %out_cur
	%src_cur =l add %src_cur, 1
	%out_cur =l add %out_cur, 1
	%k =l add %k, 1
	jmp @lit_copy_loop
@lit_done
	jmp @loop

@lit_copy_partial
	%copy =l copy %rem_out
	%k =l copy 0
@lit_partial_loop
	%cond =w csltl %k, %copy
	jnz %cond, @lit_partial_body, @lit_partial_done
@lit_partial_body
	%b =w loadub %src_cur
	storeb %b, %out_cur
	%src_cur =l add %src_cur, 1
	%out_cur =l add %out_cur, 1
	%k =l add %k, 1
	jmp @lit_partial_loop
@lit_partial_done
	%result =l copy -1
	retl %result

@repeat
	%h_long =l sxtw %h
	%count =l neg %h_long
	%count =l add %count, 1

	%rem_src =l sub %src_end, %src_cur
	%src_ok =w csgel %rem_src, 1
	jnz %src_ok, @rep_src_ok, @error2

@rep_src_ok
	%b =w loadub %src_cur
	%src_cur =l add %src_cur, 1

	%rem_out =l sub %out_end, %out_cur
	%out_ok =w csgel %rem_out, %count
	jnz %out_ok, @rep_copy_full, @rep_copy_partial

@rep_copy_full
	%k =l copy 0
@rep_copy_loop
	%cond =w csltl %k, %count
	jnz %cond, @rep_copy_body, @rep_done
@rep_copy_body
	storeb %b, %out_cur
	%out_cur =l add %out_cur, 1
	%k =l add %k, 1
	jmp @rep_copy_loop
@rep_done
	jmp @loop

@rep_copy_partial
	%copy =l copy %rem_out
	%k =l copy 0
@rep_partial_loop
	%cond =w csltl %k, %copy
	jnz %cond, @rep_partial_body, @rep_partial_done
@rep_partial_body
	storeb %b, %out_cur
	%out_cur =l add %out_cur, 1
	%k =l add %k, 1
	jmp @rep_partial_loop
@rep_partial_done
	%result =l copy -1
	retl %result

@error2
	%result =l copy -2
	retl %result

@compute_result
	%result =l sub %out_cur, %out
	retl %result
}
