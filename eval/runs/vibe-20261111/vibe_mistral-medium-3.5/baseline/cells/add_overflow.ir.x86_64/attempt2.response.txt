export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum = add %a, %b
    storel %sum, %out
    %a_neg = csltl %a, 0
    %b_neg = csltl %b, 0
    %sum_neg = csltl %sum, 0
    %same_sign = ceql %a_neg, %b_neg
    %diff_sign = cneq %a_neg, %sum_neg
    %overflow = andw %same_sign, %diff_sign
    retl %overflow
}
