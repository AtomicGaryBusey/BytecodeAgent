export function l $binary_search(l %a, l %n, l %key) {
@init
        %low =l copy 0
        %high =l sub %n, 1

@loop
        %low_le_high =w cslel %low, %high
        jnz %low_le_high, @body, @return_neg1

@body
        %sum =l add %low, %high
        %mid =l shr %sum, 1
        %offset =l mul %mid, 8
        %addr =l add %a, %offset
        %val =l loadl %addr
        %val_lt_key =w csltl %val, %key
        jnz %val_lt_key, @less, @check_equal

@less
        %low =l add %mid, 1
        jmp @loop

@check_equal
        %val_gt_key =w csgtl %val, %key
        jnz %val_gt_key, @greater, @found

@greater
        %high =l sub %mid, 1
        jmp @loop

@found
        ret %mid

@return_neg1
        ret -1
}
