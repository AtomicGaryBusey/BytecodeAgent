vm_run:
    xor r8, r8
    xor r9, r9
.main_loop:
    cmp r8, rsi
    je .done
    mov rax, [rdi + r8*8]
    cmp rax, 0
    je .halt
    cmp rax, 1
    je .push
    cmp rax, 2
    je .add
    cmp rax, 3
    je .sub
    cmp rax, 4
    je .mul
    cmp rax, 5
    je .dup
    cmp rax, 6
    je .swap
    cmp rax, 7
    je .drop
    cmp rax, 8
    je .over
    cmp rax, 9
    je .lt
    cmp rax, 10
    je .jz
    cmp rax, 11
    je .jmp
    inc r8
    jmp .main_loop
.halt:
    jmp .done
.push:
    mov rax, [rdi + r8*8 + 8]
    mov [rdx + r9*8], rax
    inc r9
    add r8, 2
    jmp .main_loop
.add:
    dec r9
    mov rax, [rdx + r9*8]
    dec r9
    add [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.sub:
    dec r9
    mov rax, [rdx + r9*8]
    dec r9
    sub [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.mul:
    dec r9
    mov r10, [rdx + r9*8]
    dec r9
    mov rax, [rdx + r9*8]
    mov r11, rdx
    mov rdx, r10
    mul rdx
    mov rdx, r11
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.dup:
    mov rax, [rdx + r9*8 - 8]
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.swap:
    mov rax, [rdx + r9*8 - 8]
    mov r10, [rdx + r9*8 - 16]
    mov [rdx + r9*8 - 8], r10
    mov [rdx + r9*8 - 16], rax
    inc r8
    jmp .main_loop
.drop:
    dec r9
    inc r8
    jmp .main_loop
.over:
    mov rax, [rdx + r9*8 - 16]
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.lt:
    dec r9
    mov rax, [rdx + r9*8]
    dec r9
    cmp [rdx + r9*8], rax
    setl al
    movzx rax, al
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .main_loop
.jz:
    dec r9
    mov rax, [rdx + r9*8]
    test rax, rax
    jnz .jz_not_zero
    mov r8, [rdi + r8*8 + 8]
    jmp .main_loop
.jz_not_zero:
    add r8, 2
    jmp .main_loop
.jmp:
    mov r8, [rdi + r8*8 + 8]
    jmp .main_loop
.done:
    test r9, r9
    jz .return_zero
    mov rax, [rdx + r9*8 - 8]
    jmp .return
.return_zero:
    xor rax, rax
.return:
    lea r10, [rdx + r9*8]
    lea r11, [rdx + rcx*8]
.zero_loop:
    cmp r10, r11
    jae .return_final
    mov qword ptr [r10], 0
    add r10, 8
    jmp .zero_loop
.return_final:
    ret
