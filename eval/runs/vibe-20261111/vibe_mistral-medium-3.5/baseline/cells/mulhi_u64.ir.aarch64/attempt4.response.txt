export function l $mulhi_u64(l %a, l %b) {
@start
	%a_lo = and %a, 4294967295l
	%a_hi = shr %a, 32
	%b_lo = and %b, 4294967295l
	%b_hi = shr %b, 32
	%p0 = mul %a_lo, %b_lo
	%p0_hi = shr %p0, 32
	%p1 = mul %a_lo, %b_hi
	%p1_lo = and %p1, 4294967295l
	%p1_hi = shr %p1, 32
	%p2 = mul %a_hi, %b_lo
	%p2_lo = and %p2, 4294967295l
	%p2_hi = shr %p2, 32
	%sum_lo = add %p1_lo, %p2_lo
	%sum_lo = add %sum_lo, %p0_hi
	%carry = shr %sum_lo, 32
	%sum_hi = add %p1_hi, %p2_hi
	%sum_hi = add %sum_hi, %carry
	%p3 = mul %a_hi, %b_hi
	%res = add %p3, %sum_hi
	ret %res
}
