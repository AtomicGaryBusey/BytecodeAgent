export function l $mulhi_u64(l %a, l %b) {
@start
%a_lo = and %a, 0xFFFFFFFF
%a_hi = shr %a, 32
%b_lo = and %b, 0xFFFFFFFF
%b_hi = shr %b, 32
%p0 = mul %a_lo, %b_lo
%p1 = mul %a_lo, %b_hi
%p2 = mul %a_hi, %b_lo
%p3 = mul %a_hi, %b_hi
%mid = add %p1, %p2
%c1 = add shr %p0, 32, and %mid, 0xFFFFFFFF
%c2 = add %c1, and %p0, 0xFFFFFFFF
%carry = shr %c2, 32
ret add %p3, add shr %mid, 32, %carry
}
