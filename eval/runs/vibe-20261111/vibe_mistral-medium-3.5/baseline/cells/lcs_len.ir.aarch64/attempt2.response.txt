export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start:
%count =l add %n, 1
%ptr =l copy %row
@init:
storel 0, %ptr
%count =l sub %count, 1
%ptr =l add %ptr, 8
jnz %count, @init
%i =l copy 0
%xi =l copy %x
%n1 =l add %n, 1
@outer:
%cond =l csltl %i, %m
jnz %cond, @outer_body
jmp @done
@outer_body:
%diag =l copy 0
%j =l copy 1
%yj =l copy %y
@inner:
%cond =l csltl %j, %n1
jnz %cond, @inner_body
jmp @outer_next
@inner_body:
%j8 =l shl %j, 3
%p =l add %row, %j8
%up =l loadl %p
%xb =w loadub %xi
%yb =w loadub %yj
%eq =w ceql %xb, %yb
%p1 =l sub %p, 8
%left =l loadl %p1
%v1 =l add %diag, 1
%lt =w csltl %up, %left
%v2 =l select %lt, %left, %up
%v =l select %eq, %v1, %v2
storel %v, %p
%diag =l copy %up
%j =l add %j, 1
%yj =l add %yj, 1
jmp @inner
@outer_next:
%i =l add %i, 1
%xi =l add %xi, 1
jmp @outer
@done:
%n8 =l shl %n, 3
%p =l add %row, %n8
%res =l loadl %p
ret %res
}
