map_sum:
    test rsi, rsi
    jz .Lreturn_zero
    push r15
    push rbx
    push r14
    push r13
    push r12
    mov r12, rdx
    mov r13, rdi
    xor r14, r14
    xor rbx, rbx
.Lloop:
    mov rdi, qword ptr [r13 + rbx*8]
    call r12
    add r14, rax
    inc rbx
    cmp rbx, rsi
    jl .Lloop
    mov rax, r14
    pop r12
    pop r13
    pop r14
    pop rbx
    pop r15
    ret
.Lreturn_zero:
    xor rax, rax
    ret
