export function l $binary_search(l a, l n, l key) {
@start
	%n_zero = ceql n 0
	jnz %n_zero @return_neg1
	%high = subl n 1
	%low = 0
	jmp @loop
@loop
	%low_gt_high = cgtl %low %high
	jnz %low_gt_high @return_neg1
	%diff = subl %high %low
	%half_diff = shrl %diff 1
	%mid = addl %low %half_diff
	%val = loadl a %mid 8
	%eq = ceql %val key
	jnz %eq @found
	%lt = cltl %val key
	jnz %lt @less
	%high = subl %mid 1
	jmp @loop
@less
	%low = addl %mid 1
	jmp @loop
@found
	return %mid
@return_neg1
	return -1
}
