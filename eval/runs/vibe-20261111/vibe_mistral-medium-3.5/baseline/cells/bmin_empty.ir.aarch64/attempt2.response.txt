export function l $bmin_empty(l %a, l %n) {
@start
	%max = 9223372036854775807
	%zero = 0
	%n_zero =w ceql %n %zero
	jnz %n_zero @return_max
	%min =l loadl %a
	%i = 1
@loop
	%cond =w csltl %i %n
	jnz %cond @body
	ret %min
@body
	%offset =l sll %i 3
	%addr =l addl %a %offset
	%val =l loadl %addr
	%update =w csltl %val %min
	%min =l select %update %val %min
	%i =l addl %i 1
	jump @loop
@return_max
	ret %max
}
