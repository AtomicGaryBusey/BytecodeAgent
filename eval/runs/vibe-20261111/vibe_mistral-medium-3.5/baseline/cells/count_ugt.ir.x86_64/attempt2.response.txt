export function l $count_ugt(l %a, l %n, l %t) {
@start
    %count =l copy 0
    jmp @loop_test
@loop_test
    %cond =w cne %n, 0
    jnz %cond, @loop_body, @done
@loop_body
    %elem =l loadl %a
    %ugt =w cugt %elem, %t
    %count =l add %count, %ugt
    %a =l add %a, 8
    %n =l sub %n, 1
    jmp @loop_test
@done
    ret %count
}
