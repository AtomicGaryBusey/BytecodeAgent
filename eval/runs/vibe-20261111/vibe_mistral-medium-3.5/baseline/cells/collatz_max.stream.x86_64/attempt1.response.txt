collatz_max:
    mov r9, rdi
    xor rdx, rdx
    mov rsi, 1
.lloop_k:
    cmp rsi, r9
    jg .ldone
    mov rcx, rsi
    xor r8, r8
.lloop_collatz:
    cmp rcx, 1
    je .lcollatz_done
    test rcx, 1
    jz .leven
    lea rcx, [rcx + rcx*2 + 1]
    jmp .lstep
.leven:
    shr rcx, 1
.lstep:
    inc r8
    jmp .lloop_collatz
.lcollatz_done:
    cmp r8, rdx
    jle .lnext_k
    mov rdx, r8
.lnext_k:
    inc rsi
    jmp .lloop_k
.ldone:
    mov rax, rdx
    ret
