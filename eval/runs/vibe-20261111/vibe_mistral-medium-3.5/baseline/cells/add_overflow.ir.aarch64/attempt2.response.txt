export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum = add %a, %b
	storel %sum, %out
	%xor_ab = xor %a, %b
	%sign_ab = shrx %xor_ab, 63
	%same_sign = sub -1, %sign_ab
	%xor_as = xor %a, %sum
	%sign_as = shrx %xor_as, 63
	%overflow = and %same_sign, %sign_as
	%result = neg %overflow
	ret %result
}
