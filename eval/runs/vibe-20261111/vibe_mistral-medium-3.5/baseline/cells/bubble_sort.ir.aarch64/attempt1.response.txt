export function l $bubble_sort(l %a, l %n) {
@start
%outer = 0
%cmp_outer = csltl %outer, %n
jnz %cmp_outer, @outer_loop, @done
@outer_loop
%inner_limit = subl %n, %outer
%inner_limit = subl %inner_limit, 1
%inner = 0
%cmp_inner = csltl %inner, %inner_limit
jnz %cmp_inner, @inner_loop, @outer_continue
@inner_loop
%j8 = mull %inner, 8
%addr1 = addl %a, %j8
%addr2 = addl %addr1, 8
%val1 = loadl %addr1
%val2 = loadl %addr2
%cmp = cgttl %val1, %val2
jnz %cmp, @swap, @no_swap
@swap
storel %val2, %addr1
storel %val1, %addr2
@no_swap
%inner = addl %inner, 1
%cmp_inner = csltl %inner, %inner_limit
jnz %cmp_inner, @inner_loop, @outer_continue
@outer_continue
%outer = addl %outer, 1
%cmp_outer = csltl %outer, %n
jnz %cmp_outer, @outer_loop, @done
@done
ret 0
}
