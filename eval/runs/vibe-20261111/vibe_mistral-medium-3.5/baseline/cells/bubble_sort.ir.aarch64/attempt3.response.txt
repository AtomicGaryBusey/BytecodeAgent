export function l $bubble_sort(l %a, l %n) {
@start
%outer =l 0
%cmp_outer =w csllt %outer, %n
jnz %cmp_outer, @outer_loop, @done
@outer_loop
%inner_limit =l subl %n, %outer
%inner_limit =l subl %inner_limit, 1
%inner =l 0
%cmp_inner =w csllt %inner, %inner_limit
jnz %cmp_inner, @inner_loop, @outer_continue
@inner_loop
%j8 =l mull %inner, 8
%addr1 =l addl %a, %j8
%addr2 =l addl %addr1, 8
%val1 =l loadl %addr1
%val2 =l loadl %addr2
%cmp =w csgtl %val1, %val2
jnz %cmp, @swap, @no_swap
@swap
storel %val2, %addr1
storel %val1, %addr2
@no_swap
%inner =l addl %inner, 1
%cmp_inner =w csllt %inner, %inner_limit
jnz %cmp_inner, @inner_loop, @outer_continue
@outer_continue
%outer =l addl %outer, 1
%cmp_outer =w csllt %outer, %n
jnz %cmp_outer, @outer_loop, @done
@done
retl 0
}
