crc32:
    mvn w2, wzr
    mov w3, #0xEDB8
    movk w3, #0x8320, lsl #16
    cbz x1, .Ldone
.Lbyte_loop:
    ldrb w4, [x0], #1
    eor w2, w2, w4
    mov w5, #8
.Lbit_loop:
    tst w2, #1
    lsr w6, w2, #1
    b.eq .Lno_xor
    eor w6, w6, w3
.Lno_xor:
    mov w2, w6
    subs w5, w5, #1
    b.ne .Lbit_loop
    subs x1, x1, #1
    b.ne .Lbyte_loop
.Ldone:
    mvn w2, w2
    uxtw x0, w2
    ret
