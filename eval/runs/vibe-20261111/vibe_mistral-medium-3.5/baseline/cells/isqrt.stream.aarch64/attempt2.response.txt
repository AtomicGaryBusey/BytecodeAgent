isqrt:
    mov x1, #0
    mov x2, #0xffffffff
    cmp x0, x2
    csel x2, x0, x2, ls
.Lloop:
    cmp x1, x2
    b.hi .Ldone
    sub x3, x2, x1
    lsr x3, x3, #1
    add x3, x1, x3
    mul x4, x3, x3
    cmp x4, x0
    b.ls .Lthen
    sub x2, x3, #1
    b .Lloop
.Lthen:
    add x1, x3, #1
    b .Lloop
.Ldone:
    mov x0, x2
    ret
