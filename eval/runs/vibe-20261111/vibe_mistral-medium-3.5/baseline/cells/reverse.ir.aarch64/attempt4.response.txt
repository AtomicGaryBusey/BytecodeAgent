export function l $reverse(l %a, l %n) {
@start
    %i =l %a
    %tmp1 =l subl %n, 1
    %tmp2 =l mull %tmp1, 8
    %j =l addl %a, %tmp2
    jmp @loop

@loop
    %phi_i =l phi %i, %i_next
    %phi_j =l phi %j, %j_next
    %cond =w csltl %phi_i, %phi_j
    jnz %cond, @swap, @done

@swap
    %val_i =l loadl %phi_i
    %val_j =l loadl %phi_j
    storel %val_j, %phi_i
    storel %val_i, %phi_j
    %i_next =l addl %phi_i, 8
    %j_next =l subl %phi_j, 8
    jmp @loop

@done
    ret 0
}
