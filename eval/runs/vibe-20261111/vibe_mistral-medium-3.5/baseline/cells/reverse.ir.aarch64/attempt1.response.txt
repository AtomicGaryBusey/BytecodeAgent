export function l $reverse(l %a, l %n) {
    %i =l copy %a
    %j =l sub %n, 1
    %j =l mul %j, 8
    %j =l add %a, %j

@loop
    %cond =w csltl %i, %j
    jnz %cond, @swap, @done

@swap
    %val_i =l loadl %i
    %val_j =l loadl %j
    storel %val_j, %i
    storel %val_i, %j
    %i =l add %i, 8
    %j =l sub %j, 8
    jmp @loop

@done
    ret 0
}
