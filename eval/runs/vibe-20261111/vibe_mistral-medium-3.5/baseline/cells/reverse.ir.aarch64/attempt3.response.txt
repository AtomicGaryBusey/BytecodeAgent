export function l $reverse(l %a, l %n) {
@start
    %i =l %a
    %j =l subl %n, 1
    %j =l mull %j, 8
    %j =l addl %a, %j
    jmp @loop

@loop
    %cond =w csltl %i, %j
    jnz %cond, @swap, @done

@swap
    %val_i =l loadl %i
    %val_j =l loadl %j
    storel %val_j, %i
    storel %val_i, %j
    %i =l addl %i, 8
    %j =l subl %j, 8
    jmp @loop

@done
    ret 0
}
