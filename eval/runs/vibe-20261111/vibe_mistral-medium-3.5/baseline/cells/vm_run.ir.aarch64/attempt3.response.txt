export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
 %pc =l copy 0
 %sp =l copy 0
 jmp @loop

@loop
 %ge =w csgel %pc, %len
 jnz %ge, @halt
 %addr =l addl %code, %pc
 %opcode =l loadl %addr
 %eq0 =w ceql %opcode, 0
 jnz %eq0, @op_halt
 %eq1 =w ceql %opcode, 1
 jnz %eq1, @op_push
 %eq2 =w ceql %opcode, 2
 jnz %eq2, @op_add
 %eq3 =w ceql %opcode, 3
 jnz %eq3, @op_sub
 %eq4 =w ceql %opcode, 4
 jnz %eq4, @op_mul
 %eq5 =w ceql %opcode, 5
 jnz %eq5, @op_dup
 %eq6 =w ceql %opcode, 6
 jnz %eq6, @op_swap
 %eq7 =w ceql %opcode, 7
 jnz %eq7, @op_drop
 %eq8 =w ceql %opcode, 8
 jnz %eq8, @op_over
 %eq9 =w ceql %opcode, 9
 jnz %eq9, @op_lt
 %eq10 =w ceql %opcode, 10
 jnz %eq10, @op_jz
 %eq11 =w ceql %opcode, 11
 jnz %eq11, @op_jmp
 %pc =l addl %pc, 1
 jmp @loop

@op_halt
 jmp @halt

@op_push
 %pc =l addl %pc, 1
 %addr =l addl %code, %pc
 %value =l loadl %addr
 %pc =l addl %pc, 1
 %addr_stack =l addl %stack, %sp
 storel %value, %addr_stack
 %sp =l addl %sp, 1
 jmp @loop

@op_add
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %b =l loadl %addr
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %a =l loadl %addr
 %result =l addl %a, %b
 storel %result, %addr
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_sub
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %b =l loadl %addr
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %a =l loadl %addr
 %result =l subl %a, %b
 storel %result, %addr
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_mul
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %b =l loadl %addr
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %a =l loadl %addr
 %result =l mull %a, %b
 storel %result, %addr
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_dup
 %sp_minus_1 =l subl %sp, 1
 %addr =l addl %stack, %sp_minus_1
 %top =l loadl %addr
 %addr_stack =l addl %stack, %sp
 storel %top, %addr_stack
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_swap
 %sp_minus_1 =l subl %sp, 1
 %addr_top =l addl %stack, %sp_minus_1
 %top =l loadl %addr_top
 %sp_minus_2 =l subl %sp, 2
 %addr_second =l addl %stack, %sp_minus_2
 %second =l loadl %addr_second
 storel %second, %addr_top
 storel %top, %addr_second
 %pc =l addl %pc, 1
 jmp @loop

@op_drop
 %sp =l subl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_over
 %sp_minus_2 =l subl %sp, 2
 %addr =l addl %stack, %sp_minus_2
 %second =l loadl %addr
 %addr_stack =l addl %stack, %sp
 storel %second, %addr_stack
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_lt
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %b =l loadl %addr
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %a =l loadl %addr
 %result =l sltl %a, %b
 storel %result, %addr
 %sp =l addl %sp, 1
 %pc =l addl %pc, 1
 jmp @loop

@op_jz
 %sp =l subl %sp, 1
 %addr =l addl %stack, %sp
 %value =l loadl %addr
 %pc =l addl %pc, 1
 %addr_code =l addl %code, %pc
 %target =l loadl %addr_code
 %pc =l addl %pc, 1
 %is_zero =w ceql %value, 0
 jnz %is_zero, @jz_jump
 jmp @loop
@jz_jump
 %pc =l copy %target
 jmp @loop

@op_jmp
 %pc =l addl %pc, 1
 %addr =l addl %code, %pc
 %target =l loadl %addr
 %pc =l copy %target
 jmp @loop

@halt
 %p =l copy %sp
 %end =l copy %cap
 jmp @zero_check
@zero_loop
 %addr =l addl %stack, %p
 storel 0, %addr
 %p =l addl %p, 1
@zero_check
 %not_done =w csltl %p, %end
 jnz %not_done, @zero_loop
 %empty =w ceql %sp, 0
 jnz %empty, @return_zero
 %sp_minus_1 =l subl %sp, 1
 %addr =l addl %stack, %sp_minus_1
 %result =l loadl %addr
 ret %result

@return_zero
 ret 0
}
