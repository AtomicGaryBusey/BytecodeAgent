export function l $vm_run(l %code l %len l %stack l %cap) {
@start
 %pc =l copy 0
 %sp =l copy 0
 jmp @loop

@loop
 %ge =w csge %pc %len
 jnz %ge @halt
 %addr =l add %code %pc
 %opcode =l loadl %addr
 %eq0 =w ceq %opcode 0
 jnz %eq0 @op_halt
 %eq1 =w ceq %opcode 1
 jnz %eq1 @op_push
 %eq2 =w ceq %opcode 2
 jnz %eq2 @op_add
 %eq3 =w ceq %opcode 3
 jnz %eq3 @op_sub
 %eq4 =w ceq %opcode 4
 jnz %eq4 @op_mul
 %eq5 =w ceq %opcode 5
 jnz %eq5 @op_dup
 %eq6 =w ceq %opcode 6
 jnz %eq6 @op_swap
 %eq7 =w ceq %opcode 7
 jnz %eq7 @op_drop
 %eq8 =w ceq %opcode 8
 jnz %eq8 @op_over
 %eq9 =w ceq %opcode 9
 jnz %eq9 @op_lt
 %eq10 =w ceq %opcode 10
 jnz %eq10 @op_jz
 %eq11 =w ceq %opcode 11
 jnz %eq11 @op_jmp
 %pc =l add %pc 1
 jmp @loop

@op_halt
 jmp @halt

@op_push
 %pc =l add %pc 1
 %addr =l add %code %pc
 %value =l loadl %addr
 %pc =l add %pc 1
 %addr_stack =l add %stack %sp
 storel %value %addr_stack
 %sp =l add %sp 1
 jmp @loop

@op_add
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %b =l loadl %addr
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %a =l loadl %addr
 %result =l add %a %b
 storel %result %addr
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_sub
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %b =l loadl %addr
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %a =l loadl %addr
 %result =l sub %a %b
 storel %result %addr
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_mul
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %b =l loadl %addr
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %a =l loadl %addr
 %result =l mul %a %b
 storel %result %addr
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_dup
 %sp_minus_1 =l sub %sp 1
 %addr =l add %stack %sp_minus_1
 %top =l loadl %addr
 %addr_stack =l add %stack %sp
 storel %top %addr_stack
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_swap
 %sp_minus_1 =l sub %sp 1
 %addr_top =l add %stack %sp_minus_1
 %top =l loadl %addr_top
 %sp_minus_2 =l sub %sp 2
 %addr_second =l add %stack %sp_minus_2
 %second =l loadl %addr_second
 storel %second %addr_top
 storel %top %addr_second
 %pc =l add %pc 1
 jmp @loop

@op_drop
 %sp =l sub %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_over
 %sp_minus_2 =l sub %sp 2
 %addr =l add %stack %sp_minus_2
 %second =l loadl %addr
 %addr_stack =l add %stack %sp
 storel %second %addr_stack
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_lt
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %b =l loadl %addr
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %a =l loadl %addr
 %result =l slt %a %b
 storel %result %addr
 %sp =l add %sp 1
 %pc =l add %pc 1
 jmp @loop

@op_jz
 %sp =l sub %sp 1
 %addr =l add %stack %sp
 %value =l loadl %addr
 %pc =l add %pc 1
 %addr_code =l add %code %pc
 %target =l loadl %addr_code
 %pc =l add %pc 1
 %is_zero =w ceq %value 0
 jnz %is_zero @jz_jump
 jmp @loop
@jz_jump
 %pc =l copy %target
 jmp @loop

@op_jmp
 %pc =l add %pc 1
 %addr =l add %code %pc
 %target =l loadl %addr
 %pc =l copy %target
 jmp @loop

@halt
 %p =l copy %sp
 %end =l copy %cap
 jmp @zero_check
@zero_loop
 %addr =l add %stack %p
 storel 0 %addr
 %p =l add %p 1
@zero_check
 %not_done =w cslt %p %end
 jnz %not_done @zero_loop
 %empty =w ceq %sp 0
 jnz %empty @return_zero
 %sp_minus_1 =l sub %sp 1
 %addr =l add %stack %sp_minus_1
 %result =l loadl %addr
 ret %result

@return_zero
 ret 0
}
