map_into:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 8
  mov rbx, rdi
  mov rbp, rsi
  mov r12, rdx
  mov r13, rcx
  xor r14, r14
  xor r15, r15
  test r12, r12
  jz .Ldone
.Lloop:
  mov rdi, [rbp + r14*8]
  call r13
  mov [rbx + r14*8], rax
  cmp rax, [rbp + r14*8]
  jge .Lno_increment
  inc r15
.Lno_increment:
  inc r14
  cmp r14, r12
  jl .Lloop
.Ldone:
  mov rax, r15
  add rsp, 8
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
