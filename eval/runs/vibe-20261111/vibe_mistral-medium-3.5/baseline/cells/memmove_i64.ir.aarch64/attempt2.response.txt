export function l $memmove_i64(l dst, l src, l n) {
@start
%is_zero = ceq %n, 0
jnz %is_zero, @return_zero
%is_forward = cle %dst, %src
jnz %is_forward, @copy_forward
jmp @copy_backward

@copy_forward
%i = 0
%src_ptr = %src
%dst_ptr = %dst
@loop_forward
%cond = clt %i, %n
jnz %cond, @forward_body
jmp @return_zero

@forward_body
%val = loadl %src_ptr
storel %val, %dst_ptr
%src_ptr = addl %src_ptr, 8
%dst_ptr = addl %dst_ptr, 8
%i = addl %i, 1
jmp @loop_forward

@copy_backward
%i = 0
%tmp = mul %n, 8
%src_ptr = addl %src, %tmp
%dst_ptr = addl %dst, %tmp
@loop_backward
%cond = clt %i, %n
jnz %cond, @backward_body
jmp @return_zero

@backward_body
%src_ptr = subl %src_ptr, 8
%dst_ptr = subl %dst_ptr, 8
%val = loadl %src_ptr
storel %val, %dst_ptr
%i = addl %i, 1
jmp @loop_backward

@return_zero
ret 0
}
