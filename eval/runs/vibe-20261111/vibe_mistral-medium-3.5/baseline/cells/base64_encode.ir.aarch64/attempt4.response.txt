  %e =l copy 47

  %lt52 =w csltl %idx1, 52

  %b =l add %idx1, 71

  %e =l copy 47

  %step2 =l sel %lt52, %step1, %c
  %step3 =l sel %lt62, %step2, %d
  %c1 =l sel %eq63, %e, %step3
  storeb %c1, %out, %outidx+1
  %lt26 =w csltl %idx2, 26
  %lt52 =w csltl %idx2, 52
  %lt62 =w csltl %idx2, 62
  %eq63 =w ceql %idx2, 63
  %a =l add %idx2, 65
  %b =l add %idx2, 71
  %c =l sub %idx2, 4
  %d =l copy 43
  %e =l copy 47

  %step2 =l sel %lt52, %step1, %c

  %b =l add %idx3, 71
  %c =l sub %idx3, 4

  %step2 =l sel %lt52, %step1, %c
  %step3 =l sel %lt62, %step2, %d
  %c3 =l sel %eq63, %e, %step3
  storeb %c3, %out, %outidx+3

@handleremainder:

  %lt52 =w csltl %idx0r1, 52
  %lt62 =w csltl %idx0r1, 62

  %b =l add %idx0r1, 71
  %c =l sub %idx0r1, 4

  %e =l copy 47

  %step3 =l sel %lt62, %step2, %d
  %c0r1 =l sel %eq63, %e, %step3
