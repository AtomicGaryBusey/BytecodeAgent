export function l $base64_encode(l %src, l %n, l %out) {
@start:
  %nplus2 =l add %n, 2
  %div =l udivl %nplus2, 3
  %outlen =l sll %div, 2
  %i =l copy 0
  %outidx =l copy 0
@loop:
  %remaining =l sub %n, %i
  %canread3 =w csgtl %remaining, 2
  jz %canread3, @handleremainder
  %b0 =l loadub %src, %i
  %b1 =l loadub %src, %i+1
  %b2 =l loadub %src, %i+2
  %val0 =l sll %b0, 16
  %val1 =l sll %b1, 8
  %val =l or %val0, %val1
  %val =l or %val, %b2
  %idx0 =l shr %val, 18
  %idx1a =l shr %val, 12
  %idx1 =l and %idx1a, 63
  %idx2a =l shr %val, 6
  %idx2 =l and %idx2a, 63
  %idx3 =l and %val, 63
  %c0 =l call $map64(l %idx0)
  storeb %c0, %out, %outidx
  %c1 =l call $map64(l %idx1)
  storeb %c1, %out, %outidx+1
  %c2 =l call $map64(l %idx2)
  storeb %c2, %out, %outidx+2
  %c3 =l call $map64(l %idx3)
  storeb %c3, %out, %outidx+3
  %i =l add %i, 3
  %outidx =l add %outidx, 4
  jmp @loop
@handleremainder:
  %remaining =l sub %n, %i
  %is1 =w ceql %remaining, 1
  jz %is1, @check2
  %b0r1 =l loadub %src, %i
  %idx0r1 =l shr %b0r1, 2
  %tmp1 =l and %b0r1, 3
  %idx1r1 =l sll %tmp1, 4
  %c0r1 =l call $map64(l %idx0r1)
  storeb %c0r1, %out, %outidx
  %c1r1 =l call $map64(l %idx1r1)
  storeb %c1r1, %out, %outidx+1
  storeb 61, %out, %outidx+2
  storeb 61, %out, %outidx+3
  jmp @done
@check2:
  %is2 =w ceql %remaining, 2
  jz %is2, @done
  %b0r2 =l loadub %src, %i
  %b1r2 =l loadub %src, %i+1
  %val0r2 =l sll %b0r2, 8
  %valr2 =l or %val0r2, %b1r2
  %idx0r2 =l shr %valr2, 10
  %idx1ar2 =l shr %valr2, 4
  %idx1r2 =l and %idx1ar2, 63
  %tmp2 =l and %valr2, 15
  %idx2r2 =l sll %tmp2, 2
  %c0r2 =l call $map64(l %idx0r2)
  storeb %c0r2, %out, %outidx
  %c1r2 =l call $map64(l %idx1r2)
  storeb %c1r2, %out, %outidx+1
  %c2r2 =l call $map64(l %idx2r2)
  storeb %c2r2, %out, %outidx+2
  storeb 61, %out, %outidx+3
@done:
  ret %outlen
}
export function l $map64(l %v) {
@start:
  %lt26 =w csltl %v, 26
  %lt52 =w csltl %v, 52
  %lt62 =w csltl %v, 62
  %eq62 =w ceql %v, 62
  %a =l add %v, 65
  %b =l add %v, 71
  %c =l sub %v, 4
  %d =l sel %eq62, 43, 47
  %e =l sel %lt26, %a, %b
  %f =l sel %lt52, %e, %c
  %g =l sel %lt62, %f, %d
  ret %g
}
