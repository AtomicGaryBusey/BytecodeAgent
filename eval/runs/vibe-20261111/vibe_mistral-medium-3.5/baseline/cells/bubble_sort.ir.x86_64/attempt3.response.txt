export function l $bubble_sort(l %a, l %n) {
@start
    %zero =l copy 0
    %one =l copy 1
    %i =l copy %zero
    jmp @outer_test
@outer_test
    %i_lt_n =w csltl %i, %n
    jnz %i_lt_n, @outer_body
    ret %zero
@outer_body
    %j =l copy %zero
    jmp @inner_test
@inner_test
    %n_minus_i =l subl %n, %i
    %bound =l subl %n_minus_i, %one
    %j_lt_bound =w csltl %j, %bound
    jnz %j_lt_bound, @inner_body
    %i =l addl %i, %one
    jmp @outer_test
@inner_body
    %offset =l mull %j, 8
    %ptr_j =l addl %a, %offset
    %val_j =l loadl %ptr_j
    %ptr_j1 =l addl %ptr_j, 8
    %val_j1 =l loadl %ptr_j1
    %do_swap =w csltl %val_j1, %val_j
    jnz %do_swap, @swap
    jmp @inner_next
@swap
    storel %val_j, %ptr_j1
    storel %val_j1, %ptr_j
    jmp @inner_next
@inner_next
    %j =l addl %j, %one
    jmp @inner_test
}
