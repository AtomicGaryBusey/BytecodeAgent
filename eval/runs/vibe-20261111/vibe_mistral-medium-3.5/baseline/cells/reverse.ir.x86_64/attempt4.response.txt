export function l $reverse(l %a, l %n)
@start
  %left =l copy 0l
  %right =l sub %n, 1l
  jmp @loop_test

@loop_test
  %cond =w csltl %left, %right
  jnz %cond, @loop_body, @done

@loop_body
  %left_offset =l shl %left, 3l
  %right_offset =l shl %right, 3l
  %left_addr =l add %a, %left_offset
  %right_addr =l add %a, %right_offset
  %left_val =l loadl %left_addr
  %right_val =l loadl %right_addr
  storel %right_val, %left_addr
  storel %left_val, %right_addr
  %left =l add %left, 1l
  %right =l sub %right, 1l
  jmp @loop_test

@done
  ret 0l
