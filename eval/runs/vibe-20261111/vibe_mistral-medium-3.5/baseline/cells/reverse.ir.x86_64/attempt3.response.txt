export function l $reverse(l %a, l %n) {
@start
  %left = copy 0l
  %right = sub %n, 1l
  jmp @loop_test

@loop_test
  %cond = csltl %left, %right
  jnz %cond, @loop_body, @done

@loop_body
  %left_offset = shl %left, 3l
  %right_offset = shl %right, 3l
  %left_addr = add %a, %left_offset
  %right_addr = add %a, %right_offset
  %left_val = loadl %left_addr
  %right_val = loadl %right_addr
  storel %right_val, %left_addr
  storel %left_val, %right_addr
  %left = add %left, 1l
  %right = sub %right, 1l
  jmp @loop_test

@done
  ret 0l
}
