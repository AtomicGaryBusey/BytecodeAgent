mp_mul:
    push rbx
    push r12
    mov rbx, rdx
    test rsi, rsi
    jz .Lreturn_zero
    test rcx, rcx
    jz .Lreturn_zero
    mov r9, r8
    mov r10, rsi
    add r10, rcx
    xor r11, r11
.Lzero:
    test r10, r10
    jz .Lzero_done
    mov qword ptr [r9], r11
    add r9, 8
    dec r10
    jmp .Lzero
.Lzero_done:
    xor r9, r9
.Li:
    cmp r9, rsi
    jge .Lmul_done
    xor r11, r11
    xor r10, r10
.Lj:
    cmp r10, rcx
    jge .Lj_done
    mov r12, r9
    add r12, r10
    mov rax, [rdi + r9*8]
    mul qword ptr [rbx + r10*8]
    add rax, r11
    adc rdx, 0
    add rax, [r8 + r12*8]
    adc rdx, 0
    mov [r8 + r12*8], rax
    mov r11, rdx
    inc r10
    jmp .Lj
.Lj_done:
    mov [r8 + r9*8 + rcx*8], r11
    inc r9
    jmp .Li
.Lmul_done:
    mov r9, rsi
    add r9, rcx
    dec r9
.Lcount:
    cmp qword ptr [r8 + r9*8], 0
    jne .Lcount_done
    dec r9
    jns .Lcount
.Lcount_done:
    inc r9
    mov rax, r9
    pop r12
    pop rbx
    ret
.Lreturn_zero:
    xor rax, rax
    pop r12
    pop rbx
    ret
