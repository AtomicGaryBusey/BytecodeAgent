export function l $crc32(l %p, l %n) {
@start
	%crc =w 0xffffffff
	%i =l 0
	jmp @loop_byte
@loop_byte
	%cond =w cultl %i, %n
	jnz %cond, @process_byte, @end
@process_byte
	%byte_ptr =l add %p, %i
	%byte =w loadub %byte_ptr
	%crc =w xor %crc, %byte
	%j =l 0
	jmp @bit_loop
@bit_loop
	%lsb =w and %crc, 1
	%crc =w shr %crc, 1
	%cond_lsb =w ceqw %lsb, 1
	jnz %cond_lsb, @do_xor, @next_bit
@do_xor
	%crc =w xor %crc, 0xedb88320
	jmp @next_bit
@next_bit
	%j =l add %j, 1
	%cond_bit =w cultl %j, 8
	jnz %cond_bit, @bit_loop, @byte_done
@byte_done
	%i =l add %i, 1
	jmp @loop_byte
@end
	%crc =w xor %crc, 0xffffffff
	%result =l extuw %crc
	ret %result
}
