export function l $mat_pow(l a, l n, l k, l out, l tmp) {
@start
	%is_zero =w ceql k, 0
	jnz %is_zero, @k_zero, @k_nonzero
@k_zero
	%i =l 0
	%n_val =l n
	%out_ptr =l out
	%tmp_ptr =l tmp
	%tmp_size =l mul n, n
	%tmp_total =l mul %tmp_size, 2
	jmp @write_identity
@write_identity
	%done_i =w csltl %i, %n_val
	jnz %done_i, @identity_i_loop, @zero_tmp_k0
@identity_i_loop
	%j =l 0
	jmp @identity_j_loop
@identity_j_loop
	%done_j =w csltl %j, %n_val
	jnz %done_j, @identity_set, @identity_next_i
@identity_set
	%is_diag =w ceql %i, %j
	jnz %is_diag, @identity_diag, @identity_off_diag
@identity_diag
	%row =l mul %i, %n_val
	%addr =l add %out_ptr, %row
	%addr =l add %addr, %j
	storel 1, %addr
	jmp @identity_next_j
@identity_off_diag
	%row =l mul %i, %n_val
	%addr =l add %out_ptr, %row
	%addr =l add %addr, %j
	storel 0, %addr
	jmp @identity_next_j
@identity_next_j
	%j =l add %j, 1
	jmp @identity_j_loop
@identity_next_i
	%i =l add %i, 1
	jmp @write_identity
@zero_tmp_k0
	%idx =l 0
	jmp @zero_tmp_k0_loop
@zero_tmp_k0_loop
	%done =w csltl %idx, %tmp_total
	jnz %done, @zero_tmp_k0_body, @return_zero
@zero_tmp_k0_body
	storel 0, add %tmp_ptr, %idx
	%idx =l add %idx, 1
	jmp @zero_tmp_k0_loop
@return_zero
	return 0
@k_nonzero
	%i =l 0
	%tmp_size =l mul n, n
	%tmp_total =l mul %tmp_size, 2
	%a_ptr =l a
	%out_ptr =l out
	%tmp_ptr =l tmp
	%count =l 1
	jmp @copy_initial
@copy_initial
	%done =w csltl %i, %tmp_size
	jnz %done, @copy_initial_body, @mult_loop_start
@copy_initial_body
	%offset =l add %a_ptr, %i
	%val =l loadl %offset
	%out_addr =l add %out_ptr, %i
	storel %val, %out_addr
	%i =l add %i, 1
	jmp @copy_initial
@mult_loop_start
	%done =w ceql %count, k
	jnz %done, @mult_loop_end, @mult_loop_body
@mult_loop_body
	%i =l 0
	jmp @mult_i
@mult_i
	%done_i =w csltl %i, n
	jnz %done_i, @mult_j, @mult_after
@mult_j
	%j =l 0
	jmp @mult_j2
@mult_j2
	%done_j =w csltl %j, n
	jnz %done_j, @mult_t, @mult_next_i
@mult_t
	%t =l 0
	%sum =l 0
	jmp @mult_t2
@mult_t2
	%done_t =w csltl %t, n
	jnz %done_t, @mult_accum, @mult_store
@mult_accum
	%a_row =l mul %i, n
	%a_offset =l add %a_row, %t
	%a_addr =l add %out_ptr, %a_offset
	%a_val =l loadl %a_addr
	%b_row =l mul %t, n
	%b_offset =l add %b_row, %j
	%b_addr =l add %a_ptr, %b_offset
	%b_val =l loadl %b_addr
	%product =l mul %a_val, %b_val
	%sum =l add %sum, %product
	%t =l add %t, 1
	jmp @mult_t2
@mult_store
	%c_row =l mul %i, n
	%c_offset =l add %c_row, %j
	%c_addr =l add %tmp_ptr, %c_offset
	storel %sum, %c_addr
	%j =l add %j, 1
	jmp @mult_j2
@mult_next_i
	%i =l add %i, 1
	jmp @mult_i
@mult_after
	%i =l 0
	jmp @copy_result_loop
@copy_result_loop
	%done =w csltl %i, %tmp_size
	jnz %done, @copy_result_body, @mult_loop_increment
@copy_result_body
	%src_addr =l add %tmp_ptr, %i
	%val =l loadl %src_addr
	%dst_addr =l add %out_ptr, %i
	storel %val, %dst_addr
	%i =l add %i, 1
	jmp @copy_result_loop
@mult_loop_increment
	%count =l add %count, 1
	jmp @mult_loop_start
@mult_loop_end
	%i =l 0
	jmp @zero_tmp_final
@zero_tmp_final
	%done =w csltl %i, %tmp_total
	jnz %done, @zero_tmp_body, @return_zero_final
@zero_tmp_body
	%tmp_addr =l add %tmp_ptr, %i
	storel 0, %tmp_addr
	%i =l add %i, 1
	jmp @zero_tmp_final
@return_zero_final
	return 0
}
