export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%mat_size =l mul %n, %n
	%k_is_zero =w ceql %k, 0
	jnz %k_is_zero, @handle_k0, @handle_k_pos
@handle_k0
	%i =l 0
@zero_out
	%done =w csltl %i, %mat_size
	jnz %done, @zero_out_body, @set_diag
@zero_out_body
	%addr =l add %out, %i
	storel 0, %addr
	%i =l add %i, 1
	jmp @zero_out
@set_diag
	%i =l 0
@set_diag_loop
	%done =w csltl %i, %n
	jnz %done, @set_diag_body, @zero_tmp
@set_diag_body
	%row_start =l mul %i, %n
	%offset =l add %row_start, %i
	%addr =l add %out, %offset
	storel 1, %addr
	%i =l add %i, 1
	jmp @set_diag_loop
@handle_k_pos
	%i =l 0
@copy_a_to_out
	%done =w csltl %i, %mat_size
	jnz %done, @copy_body, @multiply_setup
@copy_body
	%addr_a =l add %a, %i
	%addr_out =l add %out, %i
	%val =l loadl %addr_a
	storel %val, %addr_out
	%i =l add %i, 1
	jmp @copy_a_to_out
@multiply_setup
	%remaining =l sub %k, 1
@multiply_loop
	%done_loop =w ceql %remaining, 0
	jnz %done_loop, @zero_tmp, @do_multiply
@do_multiply
	%i =l 0
@mult_i
	%done_i =w csltl %i, %n
	jnz %done_i, @mult_j, @copy_tmp_to_out
@mult_j
	%j =l 0
@mult_j_loop
	%done_j =w csltl %j, %n
	jnz %done_j, @mult_t, @mult_next_i
@mult_t
	%t =l 0
	%sum =l 0
@mult_t_loop
	%done_t =w csltl %t, %n
	jnz %done_t, @mult_accum, @mult_store
@mult_accum
	%out_row =l mul %i, %n
	%out_offset =l add %out_row, %t
	%out_addr =l add %out, %out_offset
	%out_val =l loadl %out_addr
	%a_row =l mul %t, %n
	%a_offset =l add %a_row, %j
	%a_addr =l add %a, %a_offset
	%a_val =l loadl %a_addr
	%product =l mul %out_val, %a_val
	%sum =l add %sum, %product
	%t =l add %t, 1
	jmp @mult_t_loop
@mult_store
	%tmp_row =l mul %i, %n
	%tmp_offset =l add %tmp_row, %j
	%tmp_addr =l add %tmp, %tmp_offset
	storel %sum, %tmp_addr
	%j =l add %j, 1
	jmp @mult_j_loop
@mult_next_i
	%i =l add %i, 1
	jmp @mult_i
@copy_tmp_to_out
	%i =l 0
@copy_loop
	%done =w csltl %i, %mat_size
	jnz %done, @copy_body2, @update_remaining
@copy_body2
	%addr_tmp =l add %tmp, %i
	%addr_out =l add %out, %i
	%val =l loadl %addr_tmp
	storel %val, %addr_out
	%i =l add %i, 1
	jmp @copy_loop
@update_remaining
	%remaining =l sub %remaining, 1
	jmp @multiply_loop
@zero_tmp
	%i =l 0
	%tmp_total =l mul %mat_size, 2
@zero_tmp_loop
	%done =w csltl %i, %tmp_total
	jnz %done, @zero_tmp_body, @return
@zero_tmp_body
	%addr =l add %tmp, %i
	storel 0, %addr
	%i =l add %i, 1
	jmp @zero_tmp_loop
@return
	return 0
}
