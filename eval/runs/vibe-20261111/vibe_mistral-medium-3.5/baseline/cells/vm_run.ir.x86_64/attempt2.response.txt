export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %sp =l 0
    %pc =l 0
    jmp @loop

@loop
    %pc_lt_len =w slt %pc, %len
    jnz %pc_lt_len, @fetch, @done

@fetch
    %pc_bytes =l mul %pc, 8
    %opcode =l loadl %code, %pc_bytes
    %is_0 =w eq %opcode, 0
    jnz %is_0, @op_0_halt, @check_1
@check_1
    %is_1 =w eq %opcode, 1
    jnz %is_1, @op_1_push, @check_2
@check_2
    %is_2 =w eq %opcode, 2
    jnz %is_2, @op_2_add, @check_3
@check_3
    %is_3 =w eq %opcode, 3
    jnz %is_3, @op_3_sub, @check_4
@check_4
    %is_4 =w eq %opcode, 4
    jnz %is_4, @op_4_mul, @check_5
@check_5
    %is_5 =w eq %opcode, 5
    jnz %is_5, @op_5_dup, @check_6
@check_6
    %is_6 =w eq %opcode, 6
    jnz %is_6, @op_6_swap, @check_7
@check_7
    %is_7 =w eq %opcode, 7
    jnz %is_7, @op_7_drop, @check_8
@check_8
    %is_8 =w eq %opcode, 8
    jnz %is_8, @op_8_over, @check_9
@check_9
    %is_9 =w eq %opcode, 9
    jnz %is_9, @op_9_lt, @check_10
@check_10
    %is_10 =w eq %opcode, 10
    jnz %is_10, @op_10_jz, @check_11
@check_11
    %is_11 =w eq %opcode, 11
    jnz %is_11, @op_11_jmp, @op_default

@op_0_halt
    jmp @done

@op_1_push
    %pc_plus_1 =l add %pc, 1
    %pc_plus_1_bytes =l mul %pc_plus_1, 8
    %v =l loadl %code, %pc_plus_1_bytes
    %sp_bytes =l mul %sp, 8
    storel %v, %stack, %sp_bytes
    %sp =l add %sp, 1
    %pc =l add %pc, 2
    jmp @loop

@op_2_add
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %b =l loadl %stack, %sp_minus_1_bytes
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    %res =l add %a, %b
    storel %res, %stack, %sp_minus_2_bytes
    %sp =l sub %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_3_sub
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %b =l loadl %stack, %sp_minus_1_bytes
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    %res =l sub %a, %b
    storel %res, %stack, %sp_minus_2_bytes
    %sp =l sub %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_4_mul
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %b =l loadl %stack, %sp_minus_1_bytes
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    %res =l mul %a, %b
    storel %res, %stack, %sp_minus_2_bytes
    %sp =l sub %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_5_dup
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %v =l loadl %stack, %sp_minus_1_bytes
    %sp_bytes =l mul %sp, 8
    storel %v, %stack, %sp_bytes
    %sp =l add %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_6_swap
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %b =l loadl %stack, %sp_minus_1_bytes
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    storel %b, %stack, %sp_minus_2_bytes
    storel %a, %stack, %sp_minus_1_bytes
    %pc =l add %pc, 1
    jmp @loop

@op_7_drop
    %sp =l sub %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_8_over
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    %sp_bytes =l mul %sp, 8
    storel %a, %stack, %sp_bytes
    %sp =l add %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_9_lt
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %b =l loadl %stack, %sp_minus_1_bytes
    %sp_minus_2 =l sub %sp, 2
    %sp_minus_2_bytes =l mul %sp_minus_2, 8
    %a =l loadl %stack, %sp_minus_2_bytes
    %res_w =w slt %a, %b
    %res =l zxtw %res_w
    storel %res, %stack, %sp_minus_2_bytes
    %sp =l sub %sp, 1
    %pc =l add %pc, 1
    jmp @loop

@op_10_jz
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %v =l loadl %stack, %sp_minus_1_bytes
    %sp =l sub %sp, 1
    %pc_plus_1 =l add %pc, 1
    %pc_plus_1_bytes =l mul %pc_plus_1, 8
    %t =l loadl %code, %pc_plus_1_bytes
    %is_zero =w eq %v, 0
    jnz %is_zero, @jz_taken, @jz_not_taken
@jz_taken
    %pc =l %t
    jmp @loop
@jz_not_taken
    %pc =l add %pc, 2
    jmp @loop

@op_11_jmp
    %pc_plus_1 =l add %pc, 1
    %pc_plus_1_bytes =l mul %pc_plus_1, 8
    %t =l loadl %code, %pc_plus_1_bytes
    %pc =l %t
    jmp @loop

@op_default
    %pc =l add %pc, 1
    jmp @loop

@done
    %i =l %sp
    %limit =l %cap
@zero_loop
    %i_lt_limit =w slt %i, %limit
    jnz %i_lt_limit, @zero_body, @after_zero
@zero_body
    %i_bytes =l mul %i, 8
    storel 0, %stack, %i_bytes
    %i =l add %i, 1
    jmp @zero_loop
@after_zero
    %is_empty =w eq %sp, 0
    jnz %is_empty, @return_zero, @return_top
@return_zero
    ret 0
@return_top
    %sp_minus_1 =l sub %sp, 1
    %sp_minus_1_bytes =l mul %sp_minus_1, 8
    %result =l loadl %stack, %sp_minus_1_bytes
    ret %result
}
