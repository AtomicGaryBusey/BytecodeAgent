bmin_empty:
    test rsi, rsi
    jz .Lempty
    mov rax, qword ptr [rdi]
    lea rcx, [rdi + 8]
    mov rdx, rsi
    dec rdx
    jz .Ldone
.Lloop:
    mov r8, qword ptr [rcx]
    cmp r8, rax
    cmovl rax, r8
    add rcx, 8
    dec rdx
    jnz .Lloop
.Ldone:
    ret
.Lempty:
    mov rax, 0x7fffffffffffffff
    ret
