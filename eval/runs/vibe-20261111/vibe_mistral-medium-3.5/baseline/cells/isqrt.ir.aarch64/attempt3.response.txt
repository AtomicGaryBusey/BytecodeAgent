export function l $isqrt(l %n) {
@start
%cond =w ceql %n, 0
jnz %cond @zero @init

@zero
ret 0

@init
%low =l 0
%high =l 4294967295
jmp @loop

@loop
%cond =w csltu %low, %high
jnz %cond @body @done

@body
%mid =l add %low, %high
%mid =l add %mid, 1
%mid =l shr %mid, 1
%mid_sq =l mul %mid, %mid
%cond =w csleu %mid_sq, %n
jnz %cond @update_low @update_high

@update_low
%low =l %mid
jmp @loop

@update_high
%high =l sub %mid, 1
jmp @loop

@done
ret %low
}
