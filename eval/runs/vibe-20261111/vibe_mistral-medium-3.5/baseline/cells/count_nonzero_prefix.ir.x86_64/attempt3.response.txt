export function l $count_nonzero_prefix(l %a, l %n) {
@entry
    %i =l copy 0
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %val =l loadl %addr
    %is_zero =w ceql %val, 0
    jnz %is_zero, @found_zero, @inc
@inc
    %i =l add %i, 1
    jmp @loop
@found_zero
    ret %i
@end
    ret %n
}
