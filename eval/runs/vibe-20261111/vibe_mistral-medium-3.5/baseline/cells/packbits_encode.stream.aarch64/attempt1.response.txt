packbits_encode:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    add x19, x0, x1
    mov x21, x2
    add x20, x2, x3
    mov x22, x2
    cmp x0, x19
    b.eq .Ldone
    cmp x21, x20
    b.eq .Lcap_zero
.Lloop:
    cmp x0, x19
    b.eq .Ldone
    ldrb w4, [x0]
    mov x5, #1
    add x7, x0, #1
    cmp x7, x19
    b.ge .Lcheck_repeat
.Lrun_loop:
    ldrb w8, [x7]
    cmp w8, w4
    b.ne .Lcheck_repeat
    add x5, x5, #1
    cmp x5, #128
    b.ge .Lcheck_repeat
    add x7, x7, #1
    cmp x7, x19
    b.lt .Lrun_loop
.Lcheck_repeat:
    cmp x5, #3
    b.lt .Lliteral
    add x7, x22, #2
    cmp x7, x20
    b.hi .Lno_room
    sub w7, wzr, w5
    add w7, w7, #1
    strb w7, [x22]
    add x22, x22, #1
    strb w4, [x22]
    add x22, x22, #1
    add x0, x0, x5
    b .Lloop
.Lliteral:
    mov x6, x0
.Llit_loop:
    cmp x6, x19
    b.eq .Llit_done
    sub x8, x6, x0
    cmp x8, #128
    b.ge .Llit_done
    ldrb w4, [x6]
    add x9, x6, #1
    cmp x9, x19
    b.ge .Llit_advance
    ldrb w10, [x9]
    cmp w10, w4
    b.ne .Llit_advance
    add x9, x9, #1
    cmp x9, x19
    b.ge .Llit_advance
    ldrb w10, [x9]
    cmp w10, w4
    b.ne .Llit_advance
    b .Llit_done
.Llit_advance:
    add x6, x6, #1
    b .Llit_loop
.Llit_done:
    sub x7, x6, x0
    add x8, x22, x7
    add x8, x8, #1
    cmp x8, x20
    b.hi .Lno_room
    sub w9, w7, #1
    strb w9, [x22]
    add x22, x22, #1
    cmp x7, #0
    b.eq .Llit_done_copy
.Lcopy_loop:
    ldrb w8, [x0], #1
    strb w8, [x22], #1
    subs x7, x7, #1
    b.gt .Lcopy_loop
.Llit_done_copy:
    mov x0, x6
    b .Lloop
.Lno_room:
    mov x0, #-1
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
.Lcap_zero:
    cmp x0, x19
    b.eq .Lreturn_0
    mov x0, #-1
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
.Ldone:
    sub x0, x22, x21
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
.Lreturn_0:
    mov x0, #0
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
