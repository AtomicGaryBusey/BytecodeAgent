bubble_sort:
    cmp rsi, 1
    jle .Ldone
    xor r8, r8
.Louter:
    mov rcx, rsi
    sub rcx, r8
    dec rcx
    jle .Ldone
    xor r9, r9
.Linner:
    mov rax, qword ptr [rdi + r9*8]
    mov r10, qword ptr [rdi + r9*8 + 8]
    cmp rax, r10
    jle .Lno_swap
    mov qword ptr [rdi + r9*8], r10
    mov qword ptr [rdi + r9*8 + 8], rax
.Lno_swap:
    inc r9
    cmp r9, rcx
    jl .Linner
    inc r8
    cmp r8, rsi
    jl .Louter
.Ldone:
    xor rax, rax
    ret
