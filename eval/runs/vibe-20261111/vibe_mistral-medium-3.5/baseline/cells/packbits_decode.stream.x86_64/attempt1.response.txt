packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    lea r12, [rdi + rsi]
    mov rbp, rdx
    lea r13, [rdx + rcx]
.Lcheck:
    cmp rbx, r12
    jge .Ldone
    movsx rax, byte ptr [rbx]
    inc rbx
    cmp al, -128
    je .Lcheck
    cmp al, 0
    jl .Lrepeat
    inc rax
    lea r14, [rbx + rax]
    cmp r14, r12
    jg .Lerror_src
    lea r14, [rbp + rax]
    cmp r14, r13
    jg .Lerror_cap_literal
    mov r14, rax
.Lliteral_copy:
    mov al, byte ptr [rbx]
    mov byte ptr [rbp], al
    inc rbx
    inc rbp
    dec r14
    jnz .Lliteral_copy
    jmp .Lcheck
.Lrepeat:
    neg al
    inc al
    movzx rax, al
    cmp rbx, r12
    jge .Lerror_src
    lea r14, [rbp + rax]
    cmp r14, r13
    jg .Lerror_cap_repeat
    movzx r14, byte ptr [rbx]
    inc rbx
    mov r15, rax
.Lrepeat_write:
    mov byte ptr [rbp], r14b
    inc rbp
    dec r15
    jnz .Lrepeat_write
    jmp .Lcheck
.Lerror_src:
    mov rax, -2
    jmp .Lcleanup
.Lerror_cap_literal:
    mov r14, r13
    sub r14, rbp
    jz .Lerror_cap_done_literal
.Lliteral_copy_partial:
    mov al, byte ptr [rbx]
    mov byte ptr [rbp], al
    inc rbx
    inc rbp
    dec r14
    jnz .Lliteral_copy_partial
.Lerror_cap_done_literal:
    mov rax, -1
    jmp .Lcleanup
.Lerror_cap_repeat:
    mov r14, r13
    sub r14, rbp
    jz .Lerror_cap_done_repeat
    movzx r15, byte ptr [rbx]
    inc rbx
.Lrepeat_write_partial:
    mov byte ptr [rbp], r15b
    inc rbp
    dec r14
    jnz .Lrepeat_write_partial
.Lerror_cap_done_repeat:
    mov rax, -1
    jmp .Lcleanup
.Ldone:
    mov rax, rbp
    sub rax, rdx
.Lcleanup:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
