export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%b_zero =w ceql %b, 0
	jnz %b_zero, @b_zero, @check_overflow
@b_zero
	storel %a, %rem
	ret 0
@check_overflow
	%min =l copy -9223372036854775808
	%a_min =w ceql %a, %min
	%b_neg1 =w ceql %b, -1
	%overflow =w and %a_min, %b_neg1
	jnz %overflow, @overflow, @normal
@overflow
	storel 0, %rem
	%q =l copy -9223372036854775808
	ret %q
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
