export function l $isqrt(l %n) {
@entry
%low0 =l conl 0
%high0 =l conl 4294967296
jmp @loop

@loop
%low =l φ @entry %low0, @then %then_low, @else %low
%high =l φ @entry %high0, @then %high, @else %mid
%cond =w cultl %low, %high
jnz %cond, @body
jmp @end

@body
%sum =l addl %low, %high
%mid =l shrl %sum, 1
%mid_sq =l umul %mid, %mid
%le =w culel %mid_sq, %n
jnz %le, @then
jmp @else

@then
%then_low =l addl %mid, 1
jmp @loop

@else
jmp @loop

@end
%ans =l subl %low, 1
return %ans
}
