export function l $isqrt(l %n) {
@entry
%high = 4294967296
%low = 0
jmp @loop

@loop
%cond = cltl %low, %high
jnz %cond, @body
jmp @end

@body
%sum = addl %low, %high
%mid = shrl %sum, 1
%mid_sq = umul %mid, %mid
%le = clel %mid_sq, %n
jnz %le, @then
%high = %mid
jmp @loop

@then
%low = addl %mid, 1
jmp @loop

@end
%ans = subl %low, 1
return %ans
}
