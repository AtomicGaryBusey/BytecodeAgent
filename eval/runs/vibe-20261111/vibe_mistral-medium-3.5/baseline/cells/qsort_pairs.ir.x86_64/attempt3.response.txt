export function l $qsort_pairs(l %a, l %n) {
@start
    %is_le_one =w cslel %n, 1
    jnz %is_le_one, @end
    %sixteen_n =l mul %n, 16
    %pivot_key_offset =l sub %sixteen_n, 16
    %pivot_val_offset =l add %pivot_key_offset, 8
    %pivot_key =l loadl %a, %pivot_key_offset
    %pivot_val =l loadl %a, %pivot_val_offset
    %i =l copy -1
    %j =l copy 0
    %n_minus_one =l sub %n, 1
@loop_test
    %j_lt_n_minus_one =w csltl %j, %n_minus_one
    jnz %j_lt_n_minus_one, @loop_body
    jmp @after_loop
@loop_body
    %j_offset =l mul %j, 16
    %j_val_offset =l add %j_offset, 8
    %key_j =l loadl %a, %j_offset
    %val_j =l loadl %a, %j_val_offset
    %cmp_lt =w csltl %key_j, %pivot_key
    %key_eq =w ceql %key_j, %pivot_key
    %val_le =w cslel %val_j, %pivot_val
    %key_eq_and_val_le =w and %key_eq, %val_le
    %is_le =w or %cmp_lt, %key_eq_and_val_le
    jnz %is_le, @do_swap
    jmp @next_j
@do_swap
    %i =l add %i, 1
    %i_offset =l mul %i, 16
    %i_val_offset =l add %i_offset, 8
    %key_i =l loadl %a, %i_offset
    %val_i =l loadl %a, %i_val_offset
    storel %key_j, %a, %i_offset
    storel %val_j, %a, %i_val_offset
    storel %key_i, %a, %j_offset
    storel %val_i, %a, %j_val_offset
@next_j
    %j =l add %j, 1
    jmp @loop_test
@after_loop
    %i_plus_one =l add %i, 1
    %i_plus_one_offset =l mul %i_plus_one, 16
    %i_plus_one_val_offset =l add %i_plus_one_offset, 8
    %key_i_plus_one =l loadl %a, %i_plus_one_offset
    %val_i_plus_one =l loadl %a, %i_plus_one_val_offset
    storel %pivot_key, %a, %i_plus_one_offset
    storel %pivot_val, %a, %i_plus_one_val_offset
    storel %key_i_plus_one, %a, %pivot_key_offset
    storel %val_i_plus_one, %a, %pivot_val_offset
    %p =l copy %i_plus_one
    %p_offset =l mul %p, 16
    %dummy =l call $qsort_pairs(l %a, l %p)
    %a =l add %a, %p_offset
    %a =l add %a, 16
    %n =l sub %n, %p
    %n =l sub %n, 1
    jmp @start
@end
    ret 0
}
