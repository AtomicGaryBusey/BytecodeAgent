export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l 0
	%one =l 1
	%two =l 2
	%three =l 3
	%max_run =l 128
	%repeat_base =l 257
	%neg_one =l neg %one
	%i =l %zero
	%out_pos =l %zero
	%n_zero =w ceql %n, %zero
	jnz %n_zero, @done, @main_loop
@main_loop
	%i_ge_n =w cgel %i, %n
	jnz %i_ge_n, @done, @compute_run_i
@compute_run_i
	%r_i =l %one
@run_i_loop
	%r_i_ge_max =w cgel %r_i, %max_run
	jnz %r_i_ge_max, @run_i_done, @run_i_check_next
@run_i_check_next
	%next_k1 =l add %i, %r_i
	%next_k1_ge_n =w cgel %next_k1, %n
	jnz %next_k1_ge_n, @run_i_done, @run_i_load_bytes
@run_i_load_bytes
	%src_i_ptr1 =l add %src, %i
	%src_next_ptr1 =l add %src, %next_k1
	%b_i1 =w loadub %src_i_ptr1
	%b_next1 =w loadub %src_next_ptr1
	%bytes_eq =w ceq %b_i1, %b_next1
	jnz %bytes_eq, @run_i_inc, @run_i_done
@run_i_inc
	%r_i =l add %r_i, %one
	jmp @run_i_loop
@run_i_done
	%r_i_ge_3 =w cgel %r_i, %three
	jnz %r_i_ge_3, @handle_repeat, @handle_literal
@handle_repeat
	%src_i_ptr2 =l add %src, %i
	%b_rep =w loadub %src_i_ptr2
	%packet_len =l %two
	%new_out_pos =l add %out_pos, %packet_len
	%fits =w clel %new_out_pos, %cap
	jnz %fits, @write_repeat, @error
@write_repeat
	%neg_r_i =l neg %r_i
	%header_l =l add %repeat_base, %neg_r_i
	%header_w =w trunc %header_l
	%out_hdr =l add %out, %out_pos
	storeb %header_w, %out_hdr
	%out_byte =l add %out_hdr, %one
	storeb %b_rep, %out_byte
	%out_pos =l %new_out_pos
	%i =l add %i, %r_i
	jmp @main_loop
@handle_literal
	%L =l %zero
	%j =l %i
@literal_find_loop
	%L_ge_max =w cgel %L, %max_run
	jnz %L_ge_max, @literal_done, @literal_check_j
@literal_check_j
	%j_ge_n =w cgel %j, %n
	jnz %j_ge_n, @literal_done, @compute_run_j
@compute_run_j
	%r_j =l %one
@run_j_loop
	%r_j_ge_max =w cgel %r_j, %max_run
	jnz %r_j_ge_max, @run_j_done, @run_j_check_next
@run_j_check_next
	%next_k2 =l add %j, %r_j
	%next_k2_ge_n =w cgel %next_k2, %n
	jnz %next_k2_ge_n, @run_j_done, @run_j_load_bytes
@run_j_load_bytes
	%src_j_ptr =l add %src, %j
	%src_next_ptr2 =l add %src, %next_k2
	%b_j =w loadub %src_j_ptr
	%b_next2 =w loadub %src_next_ptr2
	%b_eq =w ceq %b_j, %b_next2
	jnz %b_eq, @run_j_inc, @run_j_done
@run_j_inc
	%r_j =l add %r_j, %one
	jmp @run_j_loop
@run_j_done
	%r_j_ge_3 =w cgel %r_j, %three
	jnz %r_j_ge_3, @literal_done, @literal_continue
@literal_continue
	%L =l add %L, %one
	%j =l add %j, %one
	jmp @literal_find_loop
@literal_done
	%packet_len2 =l add %L, %one
	%new_out_pos2 =l add %out_pos, %packet_len2
	%fits2 =w clel %new_out_pos2, %cap
	jnz %fits2, @write_literal, @error
@write_literal
	%header_l2 =l add %L, %neg_one
	%header_w2 =w trunc %header_l2
	%out_hdr2 =l add %out, %out_pos
	storeb %header_w2, %out_hdr2
	%L_temp =l %L
	%src_ptr =l add %src, %i
	%out_ptr =l add %out_hdr2, %one
@copy_loop
	%L_temp_zero =w ceql %L_temp, %zero
	jnz %L_temp_zero, @copy_done, @copy_byte
@copy_byte
	%b =w loadub %src_ptr
	storeb %b, %out_ptr
	%L_temp =l add %L_temp, %neg_one
	%src_ptr =l add %src_ptr, %one
	%out_ptr =l add %out_ptr, %one
	jmp @copy_loop
@copy_done
	%out_pos =l %new_out_pos2
	%i =l %j
	jmp @main_loop
@error
	ret %neg_one
@done
	ret %out_pos
}
