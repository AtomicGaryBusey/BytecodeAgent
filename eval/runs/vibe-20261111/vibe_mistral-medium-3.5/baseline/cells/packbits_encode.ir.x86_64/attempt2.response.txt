export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %zero =l 0
  %one =l 1
  %two =l 2
  %three =l 3
  %max_run =l 128
  %repeat_base =l 257
  %neg_one =l -1
  %i =l %zero
  %out_pos =l %zero
  %n_zero =w ceql %n, %zero
  jnz %n_zero, @done, @main_loop
@main_loop
  %i_ge_n =w cgel %i, %n
  jnz %i_ge_n, @done, @compute_run_i
@compute_run_i
  %r_i =l %one
@run_i_loop
  %r_i_ge_max =w cgel %r_i, %max_run
  jnz %r_i_ge_max, @run_i_done, @run_i_check_next
@run_i_check_next
  %next_k =l add %i, %r_i
  %next_k_ge_n =w cgel %next_k, %n
  jnz %next_k_ge_n, @run_i_done, @run_i_load_bytes
@run_i_load_bytes
  %src_i =l add %src, %i
  %src_next =l add %src, %next_k
  %b_i =w loadub %src_i
  %b_next =w loadub %src_next
  %bytes_eq =w ceq %b_i, %b_next
  jnz %bytes_eq, @run_i_inc, @run_i_done
@run_i_inc
  %r_i =l add %r_i, %one
  jmp @run_i_loop
@run_i_done
  %r_i_ge_3 =w cgel %r_i, %three
  jnz %r_i_ge_3, @handle_repeat, @handle_literal
@handle_repeat
  %src_i =l add %src, %i
  %b_rep =w loadub %src_i
  %packet_len =l %two
  %new_out_pos =l add %out_pos, %packet_len
  %fits =w clel %new_out_pos, %cap
  jnz %fits, @write_repeat, @error
@write_repeat
  %header_l =l sub %repeat_base, %r_i
  %header_w =w trunc %header_l
  %out_hdr =l add %out, %out_pos
  storeb %header_w, %out_hdr
  %out_byte =l add %out, %out_pos
  %out_byte =l add %out_byte, %one
  storeb %b_rep, %out_byte
  %out_pos =l %new_out_pos
  %i =l add %i, %r_i
  jmp @main_loop
@handle_literal
  %L =l %zero
  %j =l %i
@literal_find_loop
  %L_ge_max =w cgel %L, %max_run
  jnz %L_ge_max, @literal_done, @literal_check_j
@literal_check_j
  %j_ge_n =w cgel %j, %n
  jnz %j_ge_n, @literal_done, @compute_run_j
@compute_run_j
  %r_j =l %one
@run_j_loop
  %r_j_ge_max =w cgel %r_j, %max_run
  jnz %r_j_ge_max, @run_j_done, @run_j_check_next
@run_j_check_next
  %next_k =l add %j, %r_j
  %next_k_ge_n =w cgel %next_k, %n
  jnz %next_k_ge_n, @run_j_done, @run_j_load_bytes
@run_j_load_bytes
  %src_j =l add %src, %j
  %src_next =l add %src, %next_k
  %b_j =w loadub %src_j
  %b_next =w loadub %src_next
  %b_eq =w ceq %b_j, %b_next
  jnz %b_eq, @run_j_inc, @run_j_done
@run_j_inc
  %r_j =l add %r_j, %one
  jmp @run_j_loop
@run_j_done
  %r_j_ge_3 =w cgel %r_j, %three
  jnz %r_j_ge_3, @literal_done, @literal_continue
@literal_continue
  %L =l add %L, %one
  %j =l add %j, %one
  jmp @literal_find_loop
@literal_done
  %packet_len =l add %L, %one
  %new_out_pos =l add %out_pos, %packet_len
  %fits =w clel %new_out_pos, %cap
  jnz %fits, @write_literal, @error
@write_literal
  %header_l =l sub %L, %one
  %header_w =w trunc %header_l
  %out_hdr =l add %out, %out_pos
  storeb %header_w, %out_hdr
  %L_temp =l %L
  %src_ptr =l add %src, %i
  %out_ptr =l add %out, %out_pos
  %out_ptr =l add %out_ptr, %one
@copy_loop
  %L_temp_zero =w ceql %L_temp, %zero
  jnz %L_temp_zero, @copy_done, @copy_byte
@copy_byte
  %b =w loadub %src_ptr
  storeb %b, %out_ptr
  %L_temp =l sub %L_temp, %one
  %src_ptr =l add %src_ptr, %one
  %out_ptr =l add %out_ptr, %one
  jmp @copy_loop
@copy_done
  %out_pos =l %new_out_pos
  %i =l %j
  jmp @main_loop
@error
  ret %neg_one
@done
  ret %out_pos
}
