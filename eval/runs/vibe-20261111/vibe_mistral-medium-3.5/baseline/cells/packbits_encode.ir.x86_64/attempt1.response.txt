export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %zero =l copy 0
  %one =l copy 1
  %two =l copy 2
  %three =l copy 3
  %128 =l copy 128
  %257 =l copy 257
  %neg_one =l copy -1
  %i =l copy %zero
  %out_pos =l copy %zero
  %n_zero =w ceql %n, %zero
  jnz %n_zero, @done, @main_loop
@main_loop
  %i_ge_n =w cge %i, %n
  jnz %i_ge_n, @done, @compute_run_i
@compute_run_i
  %r_i =l copy %one
  %k_i =l copy %i
@run_i_loop
  %r_i_ge_128 =w cge %r_i, %128
  jnz %r_i_ge_128, @run_i_done, @run_i_check_next
@run_i_check_next
  %next_k_i =l add %i, %r_i
  %next_k_i_ge_n =w cge %next_k_i, %n
  jnz %next_k_i_ge_n, @run_i_done, @run_i_load_bytes
@run_i_load_bytes
  %src_i_ptr =l add %src, %i
  %src_next_i_ptr =l add %src, %next_k_i
  %b_i =w loadub %src_i_ptr
  %b_next_i =w loadub %src_next_i_ptr
  %bytes_eq =w ceq %b_i, %b_next_i
  jnz %bytes_eq, @run_i_inc, @run_i_done
@run_i_inc
  %r_i =l add %r_i, %one
  jmp @run_i_loop
@run_i_done
  %r_i_ge_3 =w cge %r_i, %three
  jnz %r_i_ge_3, @handle_repeat, @handle_literal
@handle_repeat
  %src_i_ptr =l add %src, %i
  %b_rep =w loadub %src_i_ptr
  %packet_len =l copy %two
  %new_out_pos =l add %out_pos, %packet_len
  %fits =w cle %new_out_pos, %cap
  jnz %fits, @write_repeat, @error
@write_repeat
  %header =l sub %257, %r_i
  %out_hdr_ptr =l add %out, %out_pos
  storeb %header, %out_hdr_ptr
  %out_byte_ptr =l add %out, %out_pos
  %out_byte_ptr =l add %out_byte_ptr, %one
  storeb %b_rep, %out_byte_ptr
  %out_pos =l copy %new_out_pos
  %i =l add %i, %r_i
  jmp @main_loop
@handle_literal
  %L =l copy %zero
  %j =l copy %i
@literal_find_loop
  %L_ge_128 =w cge %L, %128
  jnz %L_ge_128, @literal_done, @literal_check_j
@literal_check_j
  %j_ge_n =w cge %j, %n
  jnz %j_ge_n, @literal_done, @compute_run_j
@compute_run_j
  %r_j =l copy %one
  %k_j =l copy %j
@run_j_loop
  %r_j_ge_128 =w cge %r_j, %128
  jnz %r_j_ge_128, @run_j_done, @run_j_check_next
@run_j_check_next
  %next_k_j =l add %j, %r_j
  %next_k_j_ge_n =w cge %next_k_j, %n
  jnz %next_k_j_ge_n, @run_j_done, @run_j_load_bytes
@run_j_load_bytes
  %src_j_ptr =l add %src, %j
  %src_next_j_ptr =l add %src, %next_k_j
  %b_j =w loadub %src_j_ptr
  %b_next_j =w loadub %src_next_j_ptr
  %b_eq_j =w ceq %b_j, %b_next_j
  jnz %b_eq_j, @run_j_inc, @run_j_done
@run_j_inc
  %r_j =l add %r_j, %one
  jmp @run_j_loop
@run_j_done
  %r_j_ge_3 =w cge %r_j, %three
  jnz %r_j_ge_3, @literal_done, @literal_continue
@literal_continue
  %L =l add %L, %one
  %j =l add %j, %one
  jmp @literal_find_loop
@literal_done
  %packet_len =l add %one, %L
  %new_out_pos =l add %out_pos, %packet_len
  %fits =w cle %new_out_pos, %cap
  jnz %fits, @write_literal, @error
@write_literal
  %header =l sub %L, %one
  %out_hdr_ptr =l add %out, %out_pos
  storeb %header, %out_hdr_ptr
  %L_temp =l copy %L
  %src_ptr =l add %src, %i
  %out_ptr =l add %out, %out_pos
  %out_ptr =l add %out_ptr, %one
@copy_loop
  %L_temp_zero =w ceql %L_temp, %zero
  jnz %L_temp_zero, @copy_done, @copy_byte
@copy_byte
  %b =w loadub %src_ptr
  storeb %b, %out_ptr
  %L_temp =l sub %L_temp, %one
  %src_ptr =l add %src_ptr, %one
  %out_ptr =l add %out_ptr, %one
  jmp @copy_loop
@copy_done
  %out_pos =l copy %new_out_pos
  %i =l copy %j
  jmp @main_loop
@error
  ret %neg_one
@done
  ret %out_pos
}
