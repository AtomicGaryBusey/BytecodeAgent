export function l $fib_rec(l %n) {
@start
    %one =l copy 1
    %is_zero =w ceql %n, 0
    jnz %is_zero, @return_zero, @check_one
@check_one
    %is_one =w ceql %n, %one
    jnz %is_one, @return_one, @recursive
@return_zero
    ret 0
@return_one
    ret %one
@recursive
    %n1 =l sub %n, %one
    %n2 =l sub %n, 2
    %f1 =l call $fib_rec(l %n1)
    %f2 =l call $fib_rec(l %n2)
    %res =l add %f1, %f2
    ret %res
}
