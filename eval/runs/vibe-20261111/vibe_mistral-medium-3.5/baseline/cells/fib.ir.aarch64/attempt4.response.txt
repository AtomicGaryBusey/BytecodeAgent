export function l $fib(l %n) {
@start
    %cmp0 = ceql %n, 0l
    jnz %cmp0, @return0, @setup

@setup
    %a = copy 0l
    %b = copy 1l
    %i = copy 1l

@loop
    %cmp = csltl %i, %n
    jnz %cmp, @loop_body, @return_b

@loop_body
    %next = addl %a, %b
    %a = copy %b
    %b = copy %next
    %i = addl %i, 1l
    jmp @loop

@return0
    ret 0l

@return_b
    ret %b
}
