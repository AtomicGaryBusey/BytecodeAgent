export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
%na_zero =w ceql %na, 0
%nb_zero =w ceql %nb, 0
%either_zero =w or %na_zero, %nb_zero
jnz %either_zero, @zero_case, @main_case

@zero_case
%n =l add %na, %nb
%k =l copy 0
@zero_loop
%k_lt_n =w csltl %k, %n
jnz %k_lt_n, @zero_body, @zero_done
@zero_body
storel 0, %r, %k
%k =l add %k, 1
jmp @zero_loop
@zero_done
return 0

@main_case
%n =l add %na, %nb
%k =l copy 0
@zero_loop2
%k_lt_n =w csltl %k, %n
jnz %k_lt_n, @zero_body2, @after_zero
@zero_body2
storel 0, %r, %k
%k =l add %k, 1
jmp @zero_loop2
@after_zero

%i =l copy 0
@outer
%i_lt_na =w csltl %i, %na
jnz %i_lt_na, @outer_body, @after_outer
@outer_body
%carry =l copy 0
%j =l copy 0
@inner
%j_lt_nb =w csltl %j, %nb
jnz %j_lt_nb, @inner_body, @after_inner
@inner_body
%ai =l loadl %a, %i
%bj =l loadl %b, %j

%ai_lo =l and %ai, 0xFFFFFFFF
%ai_hi =l shr %ai, 32
%bj_lo =l and %bj, 0xFFFFFFFF
%bj_hi =l shr %bj, 32

%p0 =l mul %ai_lo, %bj_lo
%p1 =l mul %ai_lo, %bj_hi
%p2 =l mul %ai_hi, %bj_lo
%p3 =l mul %ai_hi, %bj_hi

%s =l add %p1, %p2
%s_lo =l and %s, 0xFFFFFFFF
%s_hi =l shr %s, 32

%s_lo_shifted =l shl %s_lo, 32
%lo =l add %p0, %s_lo_shifted
%carry_lo =w cultl %lo, %p0
%carry_lo_l =l zext %carry_lo

%hi =l add %p3, %s_hi
%hi =l add %hi, %carry_lo_l

%ij =l add %i, %j
%rij =l loadl %r, %ij

%temp =l add %lo, %rij
%c1 =w cultl %temp, %lo
%temp2 =l add %temp, %carry
%c2 =w cultl %temp2, %temp
%carry_new =w or %c1, %c2

storel %temp2, %r, %ij

%carry_new_l =l zext %carry_new
%carry =l add %hi, %carry_new_l

%j =l add %j, 1
jmp @inner
@after_inner
%i_nb =l add %i, %nb
storel %carry, %r, %i_nb

%i =l add %i, 1
jmp @outer
@after_outer

%k =l copy %n
@count
%k_gt_0 =w csgtl %k, 0
jnz %k_gt_0, @count_body, @count_done
@count_body
%k_minus_1 =l sub %k, 1
%rk =l loadl %r, %k_minus_1
%rk_zero =w ceql %rk, 0
jnz %rk_zero, @count_zero, @count_done
@count_zero
%k =l copy %k_minus_1
jmp @count
@count_done
return %k
}
