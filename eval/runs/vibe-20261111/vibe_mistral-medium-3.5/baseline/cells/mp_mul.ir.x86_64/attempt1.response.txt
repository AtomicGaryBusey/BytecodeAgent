export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
%n =l add %na, %nb
%size =l shl %n, 4
%tmp =l alloca %size
%k =l 0
%limit =l shl %n, 1
@zero_loop
%k_lt_limit =w cltl %k, %limit
jnz %k_lt_limit, @zero_body
jmp @zero_done
@zero_body
%k8 =l shl %k, 3
%tmp_k_ptr =l add %tmp, %k8
storel 0, %tmp_k_ptr
%k_next =l add %k, 1
%k =l %k_next
jmp @zero_loop
@zero_done
%i =l 0
@i_loop
%i_lt_na =w cltl %i, %na
jnz %i_lt_na, @i_body
jmp @i_done
@i_body
%j =l 0
@j_loop
%j_lt_nb =w cltl %j, %nb
jnz %j_lt_nb, @j_body
jmp @j_done
@j_body
%i8 =l shl %i, 3
%a_i_ptr =l add %a, %i8
%a_i_val =l loadl %a_i_ptr
%j8 =l shl %j, 3
%b_j_ptr =l add %b, %j8
%b_j_val =l loadl %b_j_ptr
%x =l %a_i_val
%y =l %b_j_val
%x_lo =l and %x, 4294967295
%x_hi =l shr %x, 32
%y_lo =l and %y, 4294967295
%y_hi =l shr %y, 32
%p0 =l mul %x_lo, %y_lo
%p1 =l mul %x_lo, %y_hi
%p2 =l mul %x_hi, %y_lo
%p3 =l mul %x_hi, %y_hi
%s1 =l add %p1, %p2
%s1_lo =l and %s1, 4294967295
%s1_hi =l shr %s1, 32
%s1_lo_shl32 =l shl %s1_lo, 32
%p_lo =l add %p0, %s1_lo_shl32
%carry_low =w cultl %p_lo, %p0
%carry_low_l =l zxtw %carry_low
%mid =l add %s1_hi, %carry_low_l
%p_hi =l add %p3, %mid
%k =l add %i, %j
%k16 =l shl %k, 4
%tmp_lo_ptr =l add %tmp, %k16
%tmp_hi_ptr =l add %tmp_lo_ptr, 8
%tmp_lo_val =l loadl %tmp_lo_ptr
%tmp_hi_val =l loadl %tmp_hi_ptr
%new_lo =l add %tmp_lo_val, %p_lo
%carry1 =w cultl %new_lo, %tmp_lo_val
%new_hi =l add %tmp_hi_val, %p_hi
%carry1_l =l zxtw %carry1
%new_hi =l add %new_hi, %carry1_l
storel %new_lo, %tmp_lo_ptr
storel %new_hi, %tmp_hi_ptr
%j_next =l add %j, 1
%j =l %j_next
jmp @j_loop
@j_done
%i_next =l add %i, 1
%i =l %i_next
jmp @i_loop
@i_done
%carry_lo =l 0
%carry_hi =l 0
%k =l 0
@carry_loop
%k_lt_n =w cltl %k, %n
jnz %k_lt_n, @carry_body
jmp @carry_done
@carry_body
%k16 =l shl %k, 4
%tmp_lo_ptr =l add %tmp, %k16
%tmp_hi_ptr =l add %tmp_lo_ptr, 8
%tmp_lo_val =l loadl %tmp_lo_ptr
%tmp_hi_val =l loadl %tmp_hi_ptr
%total_lo =l add %tmp_lo_val, %carry_lo
%carry1 =w cultl %total_lo, %tmp_lo_val
%total_hi =l add %tmp_hi_val, %carry_hi
%carry1_l =l zxtw %carry1
%total_hi =l add %total_hi, %carry1_l
%carry2 =w cultl %total_hi, %tmp_hi_val
%k8 =l shl %k, 3
%r_k_ptr =l add %r, %k8
storel %total_lo, %r_k_ptr
%new_carry_lo =l %total_hi
%new_carry_hi =l zxtw %carry2
%k_next =l add %k, 1
%k =l %k_next
%carry_lo =l %new_carry_lo
%carry_hi =l %new_carry_hi
jmp @carry_loop
@carry_done
%m =l sub %n, 1
@signif_loop
%m_ge_0 =w csgel %m, 0
jnz %m_ge_0, @signif_body
jmp @signif_done_zero
@signif_body
%m8 =l shl %m, 3
%r_m_ptr =l add %r, %m8
%r_m_val =l loadl %r_m_ptr
%nonzero =w cnel %r_m_val, 0
jnz %nonzero, @signif_found
%m_next =l sub %m, 1
%m =l %m_next
jmp @signif_loop
@signif_found
%result =l add %m, 1
ret %result
@signif_done_zero
ret 0
}
