export function l $mulhi_u64(l a, l b) {
@start
    %a_lo =l and a, 0xFFFFFFFF
    %a_hi =l shr a, 32
    %b_lo =l and b, 0xFFFFFFFF
    %b_hi =l shr b, 32
    %p0 =l mul %a_lo, %b_lo
    %p1 =l mul %a_lo, %b_hi
    %p2 =l mul %a_hi, %b_lo
    %p3 =l mul %a_hi, %b_hi
    %sum_p1p2 =l add %p1, %p2
    %carry_p1p2_w =w csltl %sum_p1p2, %p1
    %carry_p1p2 =l zextl %carry_p1p2_w
    %p0_hi =l shr %p0, 32
    %sum_mid =l add %sum_p1p2, %p0_hi
    %carry_mid_w =w csltl %sum_mid, %sum_p1p2
    %carry_mid =l zextl %carry_mid_w
    %sum_mid_hi =l shr %sum_mid, 32
    %tmp1 =l add %p3, %sum_mid_hi
    %tmp2 =l add %tmp1, %carry_p1p2
    %res =l add %tmp2, %carry_mid
    ret %res
}
