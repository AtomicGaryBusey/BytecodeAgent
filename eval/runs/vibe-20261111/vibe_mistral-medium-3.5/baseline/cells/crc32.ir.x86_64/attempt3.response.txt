export function l $crc32(l %p, l %n) {
@start
	%crc =w copy #FFFFFFFF
	%index =l copy 0
@loop
	%cond =w cultl %index, %n
	jnz %cond, @process_byte, @final_xor
@process_byte
	%byte =w loadub %p
	%crc =w xor %crc, %byte
	%p =l add %p, 1
	%index =l add %index, 1
	%bit_index =w copy 0
@bit_loop
	%bit_cond =w cultw %bit_index, 8
	jnz %bit_cond, @bit_process, @bit_done
@bit_process
	%lsb =w and %crc, 1
	%crc =w shr %crc, 1
	%lsb_cond =w cneqw %lsb, 0
	jnz %lsb_cond, @apply_xor, @skip_xor
@apply_xor
	%crc =w xor %crc, #EDB88320
@skip_xor
	%bit_index =w add %bit_index, 1
	jmp @bit_loop
@bit_done
	jmp @loop
@final_xor
	%crc =w xor %crc, #FFFFFFFF
	%result =l extuw %crc
	ret %result
}
