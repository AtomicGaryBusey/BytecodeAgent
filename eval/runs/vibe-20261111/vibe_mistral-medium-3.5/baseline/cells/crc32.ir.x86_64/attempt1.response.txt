export function l $crc32(l %p, l %n) {
@start:
	%crc =w copy 0xFFFFFFFF
	%index =l copy 0
@loop:
	%cond =w csltl %index, %n
	jnz %cond, @process_byte, @final_xor
@process_byte:
	%byte =w loadub %p
	%crc =w xor %crc, %byte
	%p_next =l add %p, 1
	%index_next =l add %index, 1
	%bit_index =l copy 0
	jmp @bit_loop
@bit_loop:
	%bit_cond =w csltl %bit_index, 8
	jnz %bit_cond, @bit_process, @bit_done
@bit_process:
	%lsb =w and %crc, 1
	%crc =w shr %crc, 1
	%mask =w neg %lsb
	%mask32 =w and %mask, 0xEDB88320
	%crc =w xor %crc, %mask32
	%bit_index =l add %bit_index, 1
	jmp @bit_loop
@bit_done:
	%p =l copy %p_next
	%index =l copy %index_next
	jmp @loop
@final_xor:
	%crc =w xor %crc, 0xFFFFFFFF
	%result =l extuw %crc
	ret %result
}
