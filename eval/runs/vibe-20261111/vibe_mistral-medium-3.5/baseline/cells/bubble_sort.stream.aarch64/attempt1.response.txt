bubble_sort:
    mov x2, x1
    mov x3, #0

.outer_loop:
    subs x5, x2, x3
    subs x5, x5, #1
    b.le .done
    mov x4, #0

.inner_loop:
    cmp x4, x5
    b.ge .next_outer
    ldr x6, [x0, x4, lsl #3]
    add x7, x4, #1
    ldr x8, [x0, x7, lsl #3]
    cmp x6, x8
    b.le .no_swap
    str x8, [x0, x4, lsl #3]
    str x6, [x0, x7, lsl #3]
.no_swap:
    add x4, x4, #1
    b .inner_loop

.next_outer:
    add x3, x3, #1
    b .outer_loop

.done:
    mov x0, #0
    ret
