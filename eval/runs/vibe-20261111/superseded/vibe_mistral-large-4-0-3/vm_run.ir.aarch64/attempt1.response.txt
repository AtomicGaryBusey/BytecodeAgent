export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
  %c0 =l const 0
  %c1 =l const 1
  %c2 =l const 2
  %c3 =l const 3
  %c4 =l const 4
  %c5 =l const 5
  %c6 =l const 6
  %c7 =l const 7
  %c8 =l const 8
  %c9 =l const 9
  %c10 =l const 10
  %c11 =l const 11
  jmp @loop
@loop
  %pc =l phi @start %c0, @op_push %push_pcnext, @op_add %add_pcnext, @op_sub %sub_pcnext, @op_mul %mul_pcnext, @op_dup %dup_pcnext, @op_swap %swap_pcnext, @op_drop %drop_pcnext, @op_over %over_pcnext, @op_lt %lt_pcnext, @jz_taken %jzt_pcnext, @jz_not %jzn_pcnext, @op_jmp %jmp_pcnext, @op_default %def_pcnext
  %depth =l phi @start %c0, @op_push %push_dnext, @op_add %add_dnext, @op_sub %sub_dnext, @op_mul %mul_dnext, @op_dup %dup_dnext, @op_swap %swap_dnext, @op_drop %drop_dnext, @op_over %over_dnext, @op_lt %lt_dnext, @jz_taken %jzt_dnext, @jz_not %jzn_dnext, @op_jmp %jmp_dnext, @op_default %def_dnext
  %isdone =w ceql %pc, %len
  jnz %isdone, @finish, @fetch
@fetch
  %fetch_off =l mul %pc, %c8
  %fetch_addr =l add %code, %fetch_off
  %op =l loadl %fetch_addr
  %eq0 =w ceql %op, %c0
  jnz %eq0, @op_halt, @chk1
@chk1
  %eq1 =w ceql %op, %c1
  jnz %eq1, @op_push, @chk2
@chk2
  %eq2 =w ceql %op, %c2
  jnz %eq2, @op_add, @chk3
@chk3
  %eq3 =w ceql %op, %c3
  jnz %eq3, @op_sub, @chk4
@chk4
  %eq4 =w ceql %op, %c4
  jnz %eq4, @op_mul, @chk5
@chk5
  %eq5 =w ceql %op, %c5
  jnz %eq5, @op_dup, @chk6
@chk6
  %eq6 =w ceql %op, %c6
  jnz %eq6, @op_swap, @chk7
@chk7
  %eq7 =w ceql %op, %c7
  jnz %eq7, @op_drop, @chk8
@chk8
  %eq8 =w ceql %op, %c8
  jnz %eq8, @op_over, @chk9
@chk9
  %eq9 =w ceql %op, %c9
  jnz %eq9, @op_lt, @chk10
@chk10
  %eq10 =w ceql %op, %c10
  jnz %eq10, @op_jz, @chk11
@chk11
  %eq11 =w ceql %op, %c11
  jnz %eq11, @op_jmp, @op_default
@op_halt
  jmp @finish
@op_push
  %push_pc1 =l add %pc, %c1
  %push_poff =l mul %push_pc1, %c8
  %push_paddr =l add %code, %push_poff
  %push_val =l loadl %push_paddr
  %push_doff =l mul %depth, %c8
  %push_daddr =l add %stack, %push_doff
  storel %push_val, %push_daddr
  %push_dnext =l add %depth, %c1
  %push_pcnext =l add %pc, %c2
  jmp @loop
@op_add
  %add_dm1 =l sub %depth, %c1
  %add_dm2 =l sub %depth, %c2
  %add_offb =l mul %add_dm1, %c8
  %add_addrb =l add %stack, %add_offb
  %add_vb =l loadl %add_addrb
  %add_offa =l mul %add_dm2, %c8
  %add_addra =l add %stack, %add_offa
  %add_va =l loadl %add_addra
  %add_sum =l add %add_va, %add_vb
  storel %add_sum, %add_addra
  %add_dnext =l sub %depth, %c1
  %add_pcnext =l add %pc, %c1
  jmp @loop
@op_sub
  %sub_dm1 =l sub %depth, %c1
  %sub_dm2 =l sub %depth, %c2
  %sub_offb =l mul %sub_dm1, %c8
  %sub_addrb =l add %stack, %sub_offb
  %sub_vb =l loadl %sub_addrb
  %sub_offa =l mul %sub_dm2, %c8
  %sub_addra =l add %stack, %sub_offa
  %sub_va =l loadl %sub_addra
  %sub_diff =l sub %sub_va, %sub_vb
  storel %sub_diff, %sub_addra
  %sub_dnext =l sub %depth, %c1
  %sub_pcnext =l add %pc, %c1
  jmp @loop
@op_mul
  %mul_dm1 =l sub %depth, %c1
  %mul_dm2 =l sub %depth, %c2
  %mul_offb =l mul %mul_dm1, %c8
  %mul_addrb =l add %stack, %mul_offb
  %mul_vb =l loadl %mul_addrb
  %mul_offa =l mul %mul_dm2, %c8
  %mul_addra =l add %stack, %mul_offa
  %mul_va =l loadl %mul_addra
  %mul_prod =l mul %mul_va, %mul_vb
  storel %mul_prod, %mul_addra
  %mul_dnext =l sub %depth, %c1
  %mul_pcnext =l add %pc, %c1
  jmp @loop
@op_dup
  %dup_dm1 =l sub %depth, %c1
  %dup_off =l mul %dup_dm1, %c8
  %dup_addr =l add %stack, %dup_off
  %dup_top =l loadl %dup_addr
  %dup_doff =l mul %depth, %c8
  %dup_daddr =l add %stack, %dup_doff
  storel %dup_top, %dup_daddr
  %dup_dnext =l add %depth, %c1
  %dup_pcnext =l add %pc, %c1
  jmp @loop
@op_swap
  %swap_dm1 =l sub %depth, %c1
  %swap_dm2 =l sub %depth, %c2
  %swap_offb =l mul %swap_dm1, %c8
  %swap_addrb =l add %stack, %swap_offb
  %swap_vb =l loadl %swap_addrb
  %swap_offa =l mul %swap_dm2, %c8
  %swap_addra =l add %stack, %swap_offa
  %swap_va =l loadl %swap_addra
  storel %swap_vb, %swap_addra
  storel %swap_va, %swap_addrb
  %swap_dnext =l add %depth, %c0
  %swap_pcnext =l add %pc, %c1
  jmp @loop
@op_drop
  %drop_dnext =l sub %depth, %c1
  %drop_pcnext =l add %pc, %c1
  jmp @loop
@op_over
  %over_dm2 =l sub %depth, %c2
  %over_off =l mul %over_dm2, %c8
  %over_addr =l add %stack, %over_off
  %over_below =l loadl %over_addr
  %over_doff =l mul %depth, %c8
  %over_daddr =l add %stack, %over_doff
  storel %over_below, %over_daddr
  %over_dnext =l add %depth, %c1
  %over_pcnext =l add %pc, %c1
  jmp @loop
@op_lt
  %lt_dm1 =l sub %depth, %c1
  %lt_dm2 =l sub %depth, %c2
  %lt_offb =l mul %lt_dm1, %c8
  %lt_addrb =l add %stack, %lt_offb
  %lt_vb =l loadl %lt_addrb
  %lt_offa =l mul %lt_dm2, %c8
  %lt_addra =l add %stack, %lt_offa
  %lt_va =l loadl %lt_addra
  %lt_cond =w csltl %lt_va, %lt_vb
  %lt_res =l extsw %lt_cond
  storel %lt_res, %lt_addra
  %lt_dnext =l sub %depth, %c1
  %lt_pcnext =l add %pc, %c1
  jmp @loop
@op_jz
  %jz_dm1 =l sub %depth, %c1
  %jz_off =l mul %jz_dm1, %c8
  %jz_addr =l add %stack, %jz_off
  %jz_v =l loadl %jz_addr
  %jz_pc1 =l add %pc, %c1
  %jz_poff =l mul %jz_pc1, %c8
  %jz_paddr =l add %code, %jz_poff
  %jz_target =l loadl %jz_paddr
  %jz_iszero =w ceql %jz_v, %c0
  jnz %jz_iszero, @jz_taken, @jz_not
@jz_taken
  %jzt_dnext =l sub %depth, %c1
  %jzt_pcnext =l add %jz_target, %c0
  jmp @loop
@jz_not
  %jzn_dnext =l sub %depth, %c1
  %jzn_pcnext =l add %pc, %c2
  jmp @loop
@op_jmp
  %jmp_pc1 =l add %pc, %c1
  %jmp_poff =l mul %jmp_pc1, %c8
  %jmp_paddr =l add %code, %jmp_poff
  %jmp_target =l loadl %jmp_paddr
  %jmp_dnext =l add %depth, %c0
  %jmp_pcnext =l add %jmp_target, %c0
  jmp @loop
@op_default
  %def_dnext =l add %depth, %c0
  %def_pcnext =l add %pc, %c1
  jmp @loop
@finish
  %fin_has =w csltl %c0, %depth
  jnz %fin_has, @ret_top, @ret_zero
@ret_top
  %rt_dm1 =l sub %depth, %c1
  %rt_off =l mul %rt_dm1, %c8
  %rt_addr =l add %stack, %rt_off
  %rt_val =l loadl %rt_addr
  jmp @zeroinit
@ret_zero
  %rz_val =l add %c0, %c0
  jmp @zeroinit
@zeroinit
  %res =l phi @ret_top %rt_val, @ret_zero %rz_val
  %i_init =l add %depth, %c0
  jmp @zloop
@zloop
  %i =l phi @zeroinit %i_init, @zbody %z_inext
  %zmore =w csltl %i, %cap
  jnz %zmore, @zbody, @done
@zbody
  %z_off =l mul %i, %c8
  %z_addr =l add %stack, %z_off
  storel %c0, %z_addr
  %z_inext =l add %i, %c1
  jmp @zloop
@done
  ret %res
}
