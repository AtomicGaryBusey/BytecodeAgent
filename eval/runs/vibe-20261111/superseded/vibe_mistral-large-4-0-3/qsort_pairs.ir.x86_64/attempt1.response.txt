export function l $qsort_pairs(l %a, l %n) {
@entry
	jmp @sort
@sort
	%ap =l phi @entry %a, @place %rptr
	%np =l phi @entry %n, @place %rn
	%zero =l const 0
	%one =l const 1
	%two =l const 2
	%eight =l const 8
	%sixteen =l const 16
	%small =w csltl %np, %two
	jnz %small, @ret, @partition
@ret
	ret %zero
@partition
	%nminus1 =l sub %np, %one
	%lastoff =l mul %nminus1, %sixteen
	%lastkeyaddr =l add %ap, %lastoff
	%lastvaladdr =l add %lastkeyaddr, %eight
	%pkey =l loadl %lastkeyaddr
	%pval =l loadl %lastvaladdr
	jmp @scan
@scan
	%i =l phi @partition %zero, @next %inext
	%j =l phi @partition %zero, @next %jnext
	%more =w csltl %i, %nminus1
	jnz %more, @cmp, @place
@cmp
	%ioff =l mul %i, %sixteen
	%iaddr =l add %ap, %ioff
	%ivaladdr =l add %iaddr, %eight
	%ikey =l loadl %iaddr
	%ival =l loadl %ivaladdr
	%klt =w csltl %ikey, %pkey
	jnz %klt, @swap, @keqcheck
@keqcheck
	%keq =w ceql %ikey, %pkey
	jnz %keq, @noswap, @vltcheck
@vltcheck
	%vlt =w csltl %ival, %pval
	jnz %vlt, @swap, @noswap
@swap
	%joff =l mul %j, %sixteen
	%jaddr =l add %ap, %joff
	%jvaladdr =l add %jaddr, %eight
	%jkey =l loadl %jaddr
	%jval =l loadl %jvaladdr
	storel %ikey, %jaddr
	storel %ival, %jvaladdr
	storel %jkey, %iaddr
	storel %jval, %ivaladdr
	%jinc =l add %j, %one
	jmp @next
@noswap
	jmp @next
@next
	%inext =l add %i, %one
	%jnext =l phi @swap %jinc, @noswap %j
	jmp @scan
@place
	%joff =l mul %j, %sixteen
	%jaddr =l add %ap, %joff
	%jvaladdr =l add %jaddr, %eight
	%jkey =l loadl %jaddr
	%jval =l loadl %jvaladdr
	storel %jkey, %lastkeyaddr
	storel %jval, %lastvaladdr
	storel %pkey, %jaddr
	storel %pval, %jvaladdr
	%leftret =l call $qsort_pairs(l %ap, l %j)
	%j1 =l add %j, %one
	%roff =l mul %j1, %sixteen
	%rptr =l add %ap, %roff
	%rn =l sub %np, %j1
	jmp @sort
}
