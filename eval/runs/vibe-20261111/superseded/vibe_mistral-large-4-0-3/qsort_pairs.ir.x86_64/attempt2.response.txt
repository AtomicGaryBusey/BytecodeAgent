export function l $qsort_pairs(l %a, l %n) {
@entry
	%zero =l sub %a, %a
	jmp @sort
@sort
	%ap =l phi @entry %a, @place %rptr
	%np =l phi @entry %n, @place %rn
	%small =w csltl %np, 2
	jnz %small, @done, @partition
@done
	ret %zero
@partition
	%nminus1 =l sub %np, 1
	%lastoff =l mul %nminus1, 16
	%lastkeyaddr =l add %ap, %lastoff
	%lastvaladdr =l add %lastkeyaddr, 8
	%pkey =l loadl %lastkeyaddr
	%pval =l loadl %lastvaladdr
	jmp @scan
@scan
	%i =l phi @partition %zero, @next %inext
	%j =l phi @partition %zero, @next %jnext
	%more =w csltl %i, %nminus1
	jnz %more, @cmp, @place
@cmp
	%ioff =l mul %i, 16
	%iaddr =l add %ap, %ioff
	%ivaladdr =l add %iaddr, 8
	%ikey =l loadl %iaddr
	%ival =l loadl %ivaladdr
	%klt =w csltl %ikey, %pkey
	jnz %klt, @swap, @keqcheck
@keqcheck
	%keq =w ceql %ikey, %pkey
	jnz %keq, @vltcheck, @noswap
@vltcheck
	%vlt =w csltl %ival, %pval
	jnz %vlt, @swap, @noswap
@swap
	%joff =l mul %j, 16
	%jaddr =l add %ap, %joff
	%jvaladdr =l add %jaddr, 8
	%jkey =l loadl %jaddr
	%jval =l loadl %jvaladdr
	storel %ikey, %jaddr
	storel %ival, %jvaladdr
	storel %jkey, %iaddr
	storel %jval, %ivaladdr
	%jinc =l add %j, 1
	jmp @next
@noswap
	jmp @next
@next
	%jnext =l phi @swap %jinc, @noswap %j
	%inext =l add %i, 1
	jmp @scan
@place
	%joff =l mul %j, 16
	%jaddr =l add %ap, %joff
	%jvaladdr =l add %jaddr, 8
	%jkey =l loadl %jaddr
	%jval =l loadl %jvaladdr
	storel %jkey, %lastkeyaddr
	storel %jval, %lastvaladdr
	storel %pkey, %jaddr
	storel %pval, %jvaladdr
	%leftret =l call $qsort_pairs(l %ap, l %j)
	%j1 =l add %j, 1
	%roff =l mul %j1, 16
	%rptr =l add %ap, %roff
	%rn =l sub %np, %j1
	jmp @sort
}
