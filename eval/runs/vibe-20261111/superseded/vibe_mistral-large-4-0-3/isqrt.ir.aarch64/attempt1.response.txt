export function l $isqrt(l %n) {
@start
    %res =l copy 0
    %bit =l copy 4611686018427387904
    jmp @initc
@initc
    %c =w cugtl %bit, %n
    jnz %c, @inits, @mainc
@inits
    %bit2 =l shr %bit, 2
    %bit =l copy %bit2
    jmp @initc
@mainc
    %z =w ceql %bit, 0
    jnz %z, @done, @mainb
@mainb
    %sum =l add %res, %bit
    %lt =w cultl %n, %sum
    jnz %lt, @else, @then
@then
    %n2 =l sub %n, %sum
    %half =l shr %res, 1
    %res2 =l add %half, %bit
    %n =l copy %n2
    %res =l copy %res2
    jmp @shift
@else
    %res3 =l shr %res, 1
    %res =l copy %res3
    jmp @shift
@shift
    %bit3 =l shr %bit, 2
    %bit =l copy %bit3
    jmp @mainc
@done
    ret %res
}
