export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%i =l sub %n, %one
	jmp @outer
@outer
	%outer_cond =w csltl %zero, %i
	jnz %outer_cond, @outer_body, @done
@outer_body
	%j =l copy %zero
	jmp @inner
@inner
	%inner_cond =w csltl %j, %i
	jnz %inner_cond, @compare, @outer_next
@compare
	%offj =l mul %j, %eight
	%pj =l add %a, %offj
	%x =l loadl %pj
	%jp1 =l add %j, %one
	%offjp1 =l mul %jp1, %eight
	%pjp1 =l add %a, %offjp1
	%y =l loadl %pjp1
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @inner_next
@swap
	storel %y, %pj
	storel %x, %pjp1
	jmp @inner_next
@inner_next
	%j =l add %j, %one
	jmp @inner
@outer_next
	%i =l sub %i, %one
	jmp @outer
@done
	ret %zero
}
