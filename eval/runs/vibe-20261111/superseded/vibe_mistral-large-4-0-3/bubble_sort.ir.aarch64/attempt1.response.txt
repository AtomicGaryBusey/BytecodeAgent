export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l const 0
	%one =l const 1
	%eight =l const 8
	%i0 =l sub %n, %one
	jmp @outer(%i0)
@outer %i:l
	%outer_cond =w csltl %zero, %i
	jnz %outer_cond, @outer_body(%i), @done
@outer_body %i:l
	jmp @inner(%i, %zero)
@inner %i:l, %j:l
	%inner_cond =w csltl %j, %i
	jnz %inner_cond, @compare(%i, %j), @outer_next(%i)
@compare %i:l, %j:l
	%offj =l mul %j, %eight
	%pj =l add %a, %offj
	%x =l loadl %pj
	%jp1 =l add %j, %one
	%offjp1 =l mul %jp1, %eight
	%pjp1 =l add %a, %offjp1
	%y =l loadl %pjp1
	%gt =w csgtl %x, %y
	jnz %gt, @swap(%i, %j), @noswap(%i, %j)
@swap %i:l, %j:l
	%offj_s =l mul %j, %eight
	%pj_s =l add %a, %offj_s
	%jp1_s =l add %j, %one
	%offjp1_s =l mul %jp1_s, %eight
	%pjp1_s =l add %a, %offjp1_s
	%x_s =l loadl %pj_s
	%y_s =l loadl %pjp1_s
	storel %y_s, %pj_s
	storel %x_s, %pjp1_s
	%jnext_s =l add %j, %one
	jmp @inner(%i, %jnext_s)
@noswap %i:l, %j:l
	%jnext_n =l add %j, %one
	jmp @inner(%i, %jnext_n)
@outer_next %i:l
	%inext =l sub %i, %one
	jmp @outer(%inext)
@done
	ret %zero
}
