export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	%zero =l copy 0
@loop
	%lt =w csltl %i, %j
	jnz %lt, @body, @done
@body
	%offi =l mul %i, 8
	%offj =l mul %j, 8
	%pi =l add %a, %offi
	%pj =l add %a, %offj
	%x =l loadl %pi
	%y =l loadl %pj
	storel %pi, %y
	storel %pj, %x
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	ret %zero
}
