export function l $reverse(l %a, l %n) {
@start
	%i0 =l copy 0
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %i0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%lt =w csltl %i, %j
	jnz %lt, @body, @done
@body
	%offi =l mul %i, 8
	%offj =l mul %j, 8
	%pi =l add %a, %offi
	%pj =l add %a, %offj
	%x =l loadl %pi
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	%zero =l copy 0
	ret %zero
}
