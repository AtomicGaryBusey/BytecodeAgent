.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0
    mov x10, #65536

read_loop:
    sub x2, x10, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    cmp x19, x10
    b.ne read_loop

read_done:
    mov x21, #0

find_delimiter:
    cmp x21, x19
    b.hs no_delimiter
    ldrb w9, [x20, x21]
    cmp w9, #0x21
    b.eq delimiter_found
    add x21, x21, #1
    b find_delimiter

delimiter_found:
    add x22, x21, #1
    b validate_setup

no_delimiter:
    mov x22, x19

validate_setup:
    adrp x25, bracket_stack
    add x25, x25, :lo12:bracket_stack
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    mov x23, #0
    mov x11, #0

validate_loop:
    cmp x23, x21
    b.hs validate_done
    ldrb w9, [x20, x23]
    cmp w9, #0x5b
    b.eq open_bracket
    cmp w9, #0x5d
    b.eq close_bracket
validate_next:
    add x23, x23, #1
    b validate_loop

open_bracket:
    str w23, [x25, x11, lsl #2]
    add x11, x11, #1
    b validate_next

close_bracket:
    cbz x11, invalid_program
    sub x11, x11, #1
    ldr w12, [x25, x11, lsl #2]
    str w23, [x26, x12, lsl #2]
    str w12, [x26, x23, lsl #2]
    b validate_next

validate_done:
    cbnz x11, invalid_program
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, #0
    mov x23, #0

execute_loop:
    cmp x23, x21
    b.hs program_done
    ldrb w9, [x20, x23]
    cmp w9, #0x3e
    b.eq move_right
    cmp w9, #0x3c
    b.eq move_left
    cmp w9, #0x2b
    b.eq increment_cell
    cmp w9, #0x2d
    b.eq decrement_cell
    cmp w9, #0x2e
    b.eq output_cell
    cmp w9, #0x2c
    b.eq input_cell
    cmp w9, #0x5b
    b.eq loop_open
    cmp w9, #0x5d
    b.eq loop_close
execute_next:
    add x23, x23, #1
    b execute_loop

move_right:
    add x25, x25, #1
    b execute_next

move_left:
    sub x25, x25, #1
    b execute_next

increment_cell:
    ldrb w9, [x25]
    add w9, w9, #1
    strb w9, [x25]
    b execute_next

decrement_cell:
    ldrb w9, [x25]
    sub w9, w9, #1
    strb w9, [x25]
    b execute_next

output_cell:
    ldrb w9, [x25]
    strb w9, [x27, x28]
    add x28, x28, #1
    b execute_next

input_cell:
    mov w9, #0
    cmp x22, x19
    b.hs input_store
    ldrb w9, [x20, x22]
    add x22, x22, #1
input_store:
    strb w9, [x25]
    b execute_next

loop_open:
    ldrb w9, [x25]
    cbnz w9, execute_next
    ldr w23, [x26, x23, lsl #2]
    b execute_next

loop_close:
    ldrb w9, [x25]
    cbz w9, execute_next
    ldr w23, [x26, x23, lsl #2]
    b execute_next

program_done:
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 8
input_buf:
    .zero 65536
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
output_buf:
    .zero 65536
