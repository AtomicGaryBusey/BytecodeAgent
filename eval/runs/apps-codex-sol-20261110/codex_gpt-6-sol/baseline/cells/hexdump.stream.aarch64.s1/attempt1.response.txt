.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 16
output_buf:
    .space 80

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    add x1, x19, x21
    mov x2, #16
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x21, x21, x0
    add x22, x22, x0
    cmp x21, #16
    b.ne read_loop
    bl print_row
    mov x21, #0
    b read_loop

end_input:
    cbz x21, print_total
    bl print_row

print_total:
    mov x9, x20
    mov x0, x22
    mov x11, #28
total_hex_loop:
    lsr x12, x0, x11
    and x12, x12, #15
    ldrb w12, [x10, x12]
    strb w12, [x9], #1
    subs x11, x11, #4
    b.pl total_hex_loop
    mov w12, #10
    strb w12, [x9], #1

    mov x0, #1
    mov x1, x20
    sub x2, x9, x20
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_row:
    mov x9, x20
    sub x0, x22, x21
    mov x11, #28
row_hex_loop:
    lsr x12, x0, x11
    and x12, x12, #15
    ldrb w12, [x10, x12]
    strb w12, [x9], #1
    subs x11, x11, #4
    b.pl row_hex_loop

    mov w12, #32
    strb w12, [x9], #1
    strb w12, [x9], #1

    mov x14, #0
slot_loop:
    cmp x14, #16
    b.eq slots_done
    cmp x14, x21
    b.hs blank_slot

    ldrb w13, [x19, x14]
    lsr w15, w13, #4
    ldrb w15, [x10, x15]
    strb w15, [x9], #1
    and w15, w13, #15
    ldrb w15, [x10, x15]
    strb w15, [x9], #1
    mov w15, #32
    strb w15, [x9], #1
    b slot_end

blank_slot:
    mov w15, #32
    strb w15, [x9], #1
    strb w15, [x9], #1
    strb w15, [x9], #1

slot_end:
    cmp x14, #7
    b.ne next_slot
    mov w15, #32
    strb w15, [x9], #1
next_slot:
    add x14, x14, #1
    b slot_loop

slots_done:
    mov w15, #32
    strb w15, [x9], #1
    mov w15, #124
    strb w15, [x9], #1

    mov x14, #0
ascii_loop:
    cmp x14, x21
    b.eq ascii_done
    ldrb w13, [x19, x14]
    cmp w13, #32
    b.lo nonprintable
    cmp w13, #126
    b.hi nonprintable
    b store_ascii
nonprintable:
    mov w13, #46
store_ascii:
    strb w13, [x9], #1
    add x14, x14, #1
    b ascii_loop

ascii_done:
    mov w15, #124
    strb w15, [x9], #1
    mov w15, #10
    strb w15, [x9], #1

    mov x0, #1
    mov x1, x20
    sub x2, x9, x20
    mov x8, #64
    svc #0
    ret
