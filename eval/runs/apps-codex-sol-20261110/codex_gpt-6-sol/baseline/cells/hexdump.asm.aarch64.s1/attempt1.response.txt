.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    adrp x21, output
    add x21, x21, :lo12:output
    adrp x24, hex_digits
    add x24, x24, :lo12:hex_digits

    mov x20, #0
    movz x26, #1
    movk x26, #1, lsl #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x26, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, x21
    mov x23, #0

row_loop:
    cmp x23, x20
    b.hs final_offset

    sub x25, x20, x23
    cmp x25, #16
    b.ls row_length_ready
    mov x25, #16

row_length_ready:
    mov w0, w23
    mov x1, x22
    bl print_hex8
    mov x22, x1

    mov w5, #32
    strb w5, [x22], #1
    strb w5, [x22], #1

    add x28, x19, x23
    mov x27, #0

hex_slots:
    cmp x27, x25
    b.hs empty_slot

    ldrb w5, [x28, x27]
    lsr w6, w5, #4
    ldrb w7, [x24, w6, uxtw]
    strb w7, [x22], #1
    and w6, w5, #15
    ldrb w7, [x24, w6, uxtw]
    strb w7, [x22], #1
    b slot_space

empty_slot:
    mov w5, #32
    strb w5, [x22], #1
    strb w5, [x22], #1

slot_space:
    mov w5, #32
    strb w5, [x22], #1
    add x27, x27, #1
    cmp x27, #8
    b.ne slot_next
    strb w5, [x22], #1

slot_next:
    cmp x27, #16
    b.lo hex_slots

    mov w5, #32
    strb w5, [x22], #1
    mov w5, #124
    strb w5, [x22], #1

    mov x27, #0
ascii_loop:
    cmp x27, x25
    b.hs ascii_done
    ldrb w5, [x28, x27]
    cmp w5, #32
    b.lo nonprintable
    cmp w5, #126
    b.ls printable

nonprintable:
    mov w5, #46

printable:
    strb w5, [x22], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w5, #124
    strb w5, [x22], #1
    mov w5, #10
    strb w5, [x22], #1
    add x23, x23, #16
    b row_loop

final_offset:
    mov w0, w20
    mov x1, x22
    bl print_hex8
    mov x22, x1
    mov w5, #10
    strb w5, [x22], #1

    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_hex8:
    mov w2, #8
1:
    lsr w3, w0, #28
    ldrb w4, [x24, w3, uxtw]
    strb w4, [x1], #1
    lsl w0, w0, #4
    subs w2, w2, #1
    b.ne 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 323600
