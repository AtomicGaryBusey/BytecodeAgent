.section .bss
.balign 16
input_buffer:
.zero 65537
output_buffer:
.zero 65537

.text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    xor r8d, r8d
    xor r10d, r10d

scan_loop:
    cmp rbx, r12
    jae scan_done
    cmp byte ptr [r13 + rbx], 10
    je line_done
    inc rbx
    jmp scan_loop

line_done:
    mov r9, rbx
    call reverse_line
    mov byte ptr [r14 + r10], 10
    inc r10
    lea r8, [rbx + 1]
    inc rbx
    jmp scan_loop

scan_done:
    cmp r8, r12
    jae write_output
    mov r9, r12
    call reverse_line
    mov byte ptr [r14 + r10], 10
    inc r10

write_output:
    test r10, r10
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

reverse_line:
    cmp r9, r8
    jbe reverse_done
    dec r9
    mov al, byte ptr [r13 + r9]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_line

reverse_done:
    ret
