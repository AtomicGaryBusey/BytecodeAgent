.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x22, x19
    b.eq end_input
    ldrb w9, [x20, x22]
    cmp w9, #10
    b.eq reverse_line
    add x22, x22, #1
    b scan_loop

end_input:
    cmp x21, x19
    b.eq write_output

reverse_line:
    mov x25, x22
reverse_loop:
    cmp x25, x21
    b.eq line_finished
    sub x25, x25, #1
    ldrb w9, [x20, x25]
    strb w9, [x24, x23]
    add x23, x23, #1
    b reverse_loop

line_finished:
    mov w9, #10
    strb w9, [x24, x23]
    add x23, x23, #1
    cmp x22, x19
    b.eq write_output
    add x22, x22, #1
    mov x21, x22
    b scan_loop

write_output:
    cbz x23, exit_program
    mov x0, #1
    mov x1, x24
    mov x2, x23
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero 65537
output_buf:
    .zero 65537
