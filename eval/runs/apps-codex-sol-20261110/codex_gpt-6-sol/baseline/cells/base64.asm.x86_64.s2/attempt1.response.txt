.section .bss
input_buffer:
    .zero 49153
output_buffer:
    .zero 66500

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rbx, r13
    mov r8, r12
    lea r10, [rip + alphabet]
    lea r14, [rip + output_buffer]
    mov r15, r14
    xor r9d, r9d

encode_loop:
    test r8, r8
    jz encode_done

    mov r11d, 3
    cmp r8, 3
    jae group_size_ready
    mov r11, r8

group_size_ready:
    movzx eax, byte ptr [rbx]
    shl eax, 16
    cmp r11, 2
    jb group_ready
    movzx edx, byte ptr [rbx + 1]
    shl edx, 8
    or eax, edx
    cmp r11, 3
    jb group_ready
    movzx edx, byte ptr [rbx + 2]
    or eax, edx

group_ready:
    add rbx, r11
    sub r8, r11

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r15], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r15 + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r15 + 2], dl

    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r15 + 3], dl

    cmp r11, 3
    je padding_done
    mov byte ptr [r15 + 3], '='
    cmp r11, 2
    je padding_done
    mov byte ptr [r15 + 2], '='

padding_done:
    add r15, 4
    add r9, 4
    cmp r9, 76
    jne encode_loop
    mov byte ptr [r15], 10
    inc r15
    xor r9d, r9d
    jmp encode_loop

encode_done:
    test r9, r9
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    mov rdx, r15
    sub rdx, r14
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
