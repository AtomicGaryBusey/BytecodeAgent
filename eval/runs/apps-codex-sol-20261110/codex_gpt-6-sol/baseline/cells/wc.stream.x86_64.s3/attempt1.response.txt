.section .bss
input_buf:
    .zero 8192
output_buf:
    .zero 64

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, 8192
    syscall
    test rax, rax
    jz output
    add r14, rax
    mov r9, rax
    xor ecx, ecx

scan_loop:
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    inc rcx
    cmp rcx, r9
    jb scan_loop
    jmp read_loop

output:
    lea rdi, [rip + output_buf + 64]
    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call prepend_decimal
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call prepend_decimal
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call prepend_decimal

    lea rdx, [rip + output_buf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

prepend_decimal:
    mov r10d, 10
digit_loop:
    xor edx, edx
    div r10
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz digit_loop
    ret
