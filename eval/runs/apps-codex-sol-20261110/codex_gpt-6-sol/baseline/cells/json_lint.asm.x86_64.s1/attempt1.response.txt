.globl _start

.section .bss
buffer:
    .space 65537
outbuf:
    .space 64

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
true_word:
    .ascii "true"
false_word:
    .ascii "false"
null_word:
    .ascii "null"

.section .text
_start:
    lea r14, [rip + buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

error:
    lea rbx, [rip + outbuf + 63]
    mov byte ptr [rbx], 10
    mov rax, r12
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    dec rbx
    add dl, '0'
    mov byte ptr [rbx], dl
    test rax, rax
    jne decimal_loop

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 9
    syscall

    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    lea rdx, [rip + outbuf + 64]
    sub rdx, rbx
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r12, r13
    jae ws_done
    mov al, byte ptr [r14 + r12]
    cmp al, ' '
    je ws_advance
    cmp al, 9
    je ws_advance
    cmp al, 10
    je ws_advance
    cmp al, 13
    jne ws_done
ws_advance:
    inc r12
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe parse_number
    jmp error

value_true:
    lea rsi, [rip + true_word]
    mov edx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + false_word]
    mov edx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + null_word]
    mov edx, 4

parse_literal:
    test edx, edx
    je literal_done
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, byte ptr [rsi]
    jne error
    inc r12
    inc rsi
    dec edx
    jmp parse_literal
literal_done:
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae error
    cmp byte ptr [r14 + r12], '}'
    je object_close
object_member:
    call parse_string
    call skip_ws
    cmp r12, r13
    jae error
    cmp byte ptr [r14 + r12], ':'
    jne error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '}'
    je object_close
    cmp al, ','
    jne error
    inc r12
    call skip_ws
    jmp object_member
object_close:
    inc r12
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae error
    cmp byte ptr [r14 + r12], ']'
    je array_close
array_element:
    call parse_value
    call skip_ws
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, ']'
    je array_close
    cmp al, ','
    jne error
    inc r12
    call skip_ws
    jmp array_element
array_close:
    inc r12
    ret

parse_string:
    cmp r12, r13
    jae error
    cmp byte ptr [r14 + r12], '"'
    jne error
    inc r12
string_loop:
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '"'
    je string_close
    cmp al, '\'
    je string_escape
    cmp al, 32
    jb error
    inc r12
    jmp string_loop
string_close:
    inc r12
    ret

string_escape:
    inc r12
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '"'
    je escape_done
    cmp al, '\'
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    je escape_done
    cmp al, 'u'
    jne error
    inc r12
    mov ecx, 4
unicode_loop:
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe unicode_digit
    cmp al, 'A'
    jb error
    cmp al, 'F'
    jbe unicode_digit
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
unicode_digit:
    inc r12
    dec ecx
    jne unicode_loop
    jmp string_loop
escape_done:
    inc r12
    jmp string_loop

parse_number:
    cmp byte ptr [r14 + r12], '-'
    jne number_int
    inc r12
    cmp r12, r13
    jae error
number_int:
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
number_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    jbe number_digits
    jmp number_suffix
number_zero:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
number_suffix:
    cmp al, '.'
    je number_fraction
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
number_done:
    ret

number_fraction:
    inc r12
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
fraction_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb fraction_end
    cmp al, '9'
    jbe fraction_digits
fraction_end:
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    ret

number_exponent:
    inc r12
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc r12
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
exponent_first:
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
exponent_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits
    ret
