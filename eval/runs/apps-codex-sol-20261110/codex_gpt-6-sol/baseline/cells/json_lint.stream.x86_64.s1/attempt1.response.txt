.globl _start

.section .bss
buffer:
    .zero 65537
states:
    .zero 1002
digits:
    .zero 32

.section .rodata
ok_message:
    .ascii "ok\n"
error_message:
    .ascii "error at "
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.section .text
_start:
    lea r12, [rip + buffer]
    lea rbx, [rip + states]
    xor r13d, r13d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    xor r14d, r14d
    xor r15d, r15d
    mov byte ptr [rbx], 0

main_loop:
    cmp r14, r13
    jae at_end
    movzx r9d, byte ptr [r12 + r14]
    cmp r9b, ' '
    je skip_space
    cmp r9b, 9
    je skip_space
    cmp r9b, 10
    je skip_space
    cmp r9b, 13
    je skip_space

    movzx eax, byte ptr [rbx + r15]
    cmp eax, 0
    je parse_value
    cmp eax, 1
    je bad
    cmp eax, 2
    je array_first
    cmp eax, 3
    je array_after
    cmp eax, 4
    je parse_value
    cmp eax, 5
    je object_first
    cmp eax, 6
    je object_colon
    cmp eax, 7
    je parse_value
    cmp eax, 8
    je object_after
    jmp object_key

skip_space:
    inc r14
    jmp main_loop

at_end:
    cmp r15, 0
    jne bad
    cmp byte ptr [rbx], 1
    jne bad
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

array_first:
    cmp r9b, ']'
    je close_container
    jmp parse_value

array_after:
    cmp r9b, ']'
    je close_container
    cmp r9b, ','
    jne bad
    mov byte ptr [rbx + r15], 4
    inc r14
    jmp main_loop

object_first:
    cmp r9b, '}'
    je close_container
    cmp r9b, '"'
    jne bad
    jmp parse_key

object_key:
    cmp r9b, '"'
    jne bad

parse_key:
    call parse_string
    mov byte ptr [rbx + r15], 6
    jmp main_loop

object_colon:
    cmp r9b, ':'
    jne bad
    mov byte ptr [rbx + r15], 7
    inc r14
    jmp main_loop

object_after:
    cmp r9b, '}'
    je close_container
    cmp r9b, ','
    jne bad
    mov byte ptr [rbx + r15], 9
    inc r14
    jmp main_loop

close_container:
    inc r14
    dec r15
    jmp finish_value

parse_value:
    cmp r9b, '{'
    je start_object
    cmp r9b, '['
    je start_array
    cmp r9b, '"'
    je value_string
    cmp r9b, 't'
    je value_true
    cmp r9b, 'f'
    je value_false
    cmp r9b, 'n'
    je value_null
    cmp r9b, '-'
    je value_number
    cmp r9b, '0'
    jb bad
    cmp r9b, '9'
    ja bad
    jmp value_number

start_object:
    inc r14
    inc r15
    mov byte ptr [rbx + r15], 5
    jmp main_loop

start_array:
    inc r14
    inc r15
    mov byte ptr [rbx + r15], 2
    jmp main_loop

value_string:
    call parse_string
    jmp finish_value

value_true:
    lea rsi, [rip + true_text]
    mov edx, 4
    call parse_literal
    jmp finish_value

value_false:
    lea rsi, [rip + false_text]
    mov edx, 5
    call parse_literal
    jmp finish_value

value_null:
    lea rsi, [rip + null_text]
    mov edx, 4
    call parse_literal
    jmp finish_value

value_number:
    call parse_number

finish_value:
    movzx eax, byte ptr [rbx + r15]
    cmp eax, 0
    je finish_root
    cmp eax, 2
    je finish_array
    cmp eax, 4
    je finish_array
    mov byte ptr [rbx + r15], 8
    jmp main_loop

finish_root:
    mov byte ptr [rbx], 1
    jmp main_loop

finish_array:
    mov byte ptr [rbx + r15], 3
    jmp main_loop

parse_literal:
    xor ecx, ecx
literal_loop:
    cmp rcx, rdx
    je literal_done
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
    cmp al, byte ptr [rsi + rcx]
    jne bad
    inc r14
    inc rcx
    jmp literal_loop
literal_done:
    ret

parse_string:
    inc r14
string_loop:
    cmp r14, r13
    jae bad
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je string_done
    cmp al, 0x20
    jb bad
    cmp al, 0x5c
    je string_escape
    inc r14
    jmp string_loop

string_done:
    inc r14
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
    cmp al, '"'
    je escape_done
    cmp al, 0x5c
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    je escape_done
    cmp al, 'u'
    jne bad
    mov ecx, 4
hex_loop:
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
    cmp al, '0'
    jb bad
    cmp al, '9'
    jbe hex_good
    cmp al, 'A'
    jb bad
    cmp al, 'F'
    jbe hex_good
    cmp al, 'a'
    jb bad
    cmp al, 'f'
    ja bad
hex_good:
    dec ecx
    jnz hex_loop
escape_done:
    inc r14
    jmp string_loop

parse_number:
    mov al, byte ptr [r12 + r14]
    cmp al, '-'
    jne number_first
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
number_first:
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb bad
    cmp al, '9'
    ja bad
number_integer_loop:
    inc r14
    cmp r14, r13
    jae number_done
    mov al, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_fraction
    cmp al, '9'
    jbe number_integer_loop
    jmp number_fraction

number_zero:
    inc r14
    cmp r14, r13
    jae number_done
    mov al, byte ptr [r12 + r14]

number_fraction:
    cmp al, '.'
    jne number_exponent
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
    cmp al, '0'
    jb bad
    cmp al, '9'
    ja bad
number_fraction_loop:
    inc r14
    cmp r14, r13
    jae number_done
    mov al, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    jbe number_fraction_loop

number_exponent:
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_done
exponent_start:
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc r14
    cmp r14, r13
    jae bad
    mov al, byte ptr [r12 + r14]
exponent_first:
    cmp al, '0'
    jb bad
    cmp al, '9'
    ja bad
exponent_loop:
    inc r14
    cmp r14, r13
    jae number_done
    mov al, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_loop
number_done:
    ret

bad:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 9
    syscall

    lea rsi, [rip + digits + 31]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop

    lea rdx, [rip + digits + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
