    adrp x20, tape
    add x20, x20, :lo12:tape

    cmp w9, #60

    cmp w9, #43
    b.eq increment_cell
    cmp w9, #45
    b.eq decrement_cell
    cmp w9, #46
    b.eq output_cell
    cmp w9, #44
    b.eq input_cell
    cmp w9, #91

    cmp w9, #93
