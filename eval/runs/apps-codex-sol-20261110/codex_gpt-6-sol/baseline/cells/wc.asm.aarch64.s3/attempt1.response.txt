.section .bss
.balign 16
input_buffer:
    .space 4096
output_buffer:
    .space 64

.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // inside a word
    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output_counts

    add x21, x21, x0
    mov x24, x23
    add x25, x23, x0

scan_loop:
    ldrb w3, [x24], #1
    cmp w3, #10
    b.eq newline
    cmp w3, #32
    b.eq whitespace
    cmp w3, #9
    b.eq whitespace
    sub w4, w3, #10
    cmp w4, #3
    b.ls whitespace

    cbnz x22, scan_next
    add x20, x20, #1
    mov x22, #1
    b scan_next

newline:
    add x19, x19, #1
whitespace:
    mov x22, #0

scan_next:
    cmp x24, x25
    b.ne scan_loop
    b read_loop

output_counts:
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x1, x26
    mov x0, x19
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1
    mov x0, x20
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1
    mov x0, x21
    bl append_decimal
    mov w3, #10
    strb w3, [x1], #1

    sub x2, x1, x26
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// Append the decimal representation of x0 at x1; return the new end in x1.
append_decimal:
    sub sp, sp, #32
    add x5, sp, #32
    mov x2, x5
    mov x7, #10
1:
    udiv x3, x0, x7
    msub x4, x3, x7, x0
    add w4, w4, #48
    strb w4, [x2, #-1]!
    mov x0, x3
    cbnz x0, 1b
2:
    ldrb w4, [x2], #1
    strb w4, [x1], #1
    cmp x2, x5
    b.ne 2b
    add sp, sp, #32
    ret
