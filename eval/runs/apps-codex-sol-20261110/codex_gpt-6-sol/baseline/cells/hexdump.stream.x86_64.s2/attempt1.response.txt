.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
.zero 65537
output_buffer:
.zero 80

.text
.globl _start
_start:
    lea r15, [rip + input_buffer]
    lea r9, [rip + hex_digits]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_length_ready
    mov r14d, 16

row_length_ready:
    lea r8, [rip + output_buffer]
    mov eax, r13d
    call emit_offset
    mov byte ptr [r8], ' '
    mov byte ptr [r8 + 1], ' '
    add r8, 2

    lea rsi, [r15 + r13]
    xor ebx, ebx

hex_slots:
    cmp rbx, r14
    jae empty_slot

    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r8], dl
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [r8 + 1], al
    mov byte ptr [r8 + 2], ' '
    jmp slot_done

empty_slot:
    mov byte ptr [r8], ' '
    mov byte ptr [r8 + 1], ' '
    mov byte ptr [r8 + 2], ' '

slot_done:
    add r8, 3
    cmp ebx, 7
    jne no_middle_space
    mov byte ptr [r8], ' '
    inc r8

no_middle_space:
    inc rbx
    cmp ebx, 16
    jb hex_slots

    mov byte ptr [r8], ' '
    mov byte ptr [r8 + 1], '|'
    add r8, 2
    xor ebx, ebx

ascii_loop:
    cmp rbx, r14
    jae ascii_done
    mov al, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable

nonprintable:
    mov al, '.'

printable:
    mov byte ptr [r8], al
    inc r8
    inc rbx
    jmp ascii_loop

ascii_done:
    mov byte ptr [r8], '|'
    mov byte ptr [r8 + 1], 10
    add r8, 2

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r8
    sub rdx, rsi
    syscall

    add r13, r14
    jmp row_loop

final_offset:
    lea r8, [rip + output_buffer]
    mov eax, r12d
    call emit_offset
    mov byte ptr [r8], 10

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov ecx, 28

offset_digit:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r8], dl
    inc r8
    sub ecx, 4
    jns offset_digit
    ret
