.globl _start

.section .text
_start:
    xor ebx, ebx

read_input:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + source]
    add rsi, rbx
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz find_delimiter
    add rbx, rax
    jmp read_input

find_delimiter:
    xor r8d, r8d
    mov r15, rbx
    mov r14, rbx
    lea rdx, [rip + source]

delimiter_loop:
    cmp r8, rbx
    jae validate
    cmp byte ptr [rdx + r8], 33
    je delimiter_found
    inc r8
    jmp delimiter_loop

delimiter_found:
    mov r15, r8
    lea r14, [r8 + 1]

validate:
    xor r8d, r8d
    xor r9d, r9d
    lea rdi, [rip + matches]
    lea rsi, [rip + bracket_stack]

validate_loop:
    cmp r8, r15
    jae validate_done
    mov al, byte ptr [rdx + r8]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket

validate_next:
    inc r8
    jmp validate_loop

open_bracket:
    mov dword ptr [rsi + r9*4], r8d
    inc r9
    jmp validate_next

close_bracket:
    test r9, r9
    jz invalid
    dec r9
    mov eax, dword ptr [rsi + r9*4]
    mov dword ptr [rdi + rax*4], r8d
    mov dword ptr [rdi + r8*4], eax
    jmp validate_next

validate_done:
    test r9, r9
    jnz invalid
    xor r12d, r12d
    xor r13d, r13d
    xor r10d, r10d
    lea r8, [rip + source]
    lea r9, [rip + tape]
    lea rsi, [rip + output]

execute:
    cmp r12, r15
    jae finish
    mov al, byte ptr [r8 + r12]
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 46
    je emit
    cmp al, 44
    je accept
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close

next_command:
    inc r12
    jmp execute

move_right:
    inc r13
    jmp next_command

move_left:
    dec r13
    jmp next_command

increment:
    inc byte ptr [r9 + r13]
    jmp next_command

decrement:
    dec byte ptr [r9 + r13]
    jmp next_command

emit:
    mov al, byte ptr [r9 + r13]
    mov byte ptr [rsi + r10], al
    inc r10
    jmp next_command

accept:
    xor eax, eax
    cmp r14, rbx
    jae accept_store
    mov al, byte ptr [r8 + r14]
    inc r14
accept_store:
    mov byte ptr [r9 + r13], al
    jmp next_command

loop_open:
    cmp byte ptr [r9 + r13], 0
    jne next_command
    mov r12d, dword ptr [rdi + r12*4]
    jmp next_command

loop_close:
    cmp byte ptr [r9 + r13], 0
    je next_command
    mov r12d, dword ptr [rdi + r12*4]
    jmp next_command

finish:
    test r10, r10
    jz success
    mov eax, 1
    mov edi, 1
    mov rdx, r10
    syscall

success:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.section .bss
.balign 8
source:
    .zero 65537
tape:
    .zero 30000
output:
    .zero 65536
.balign 4
matches:
    .zero 262144
bracket_stack:
    .zero 4000
