.section .rodata
digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
.zero 65536
output_buffer:
.zero 128

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    lea r15, [rip + output_buffer]
    xor r13d, r13d

read_loop:
    mov edx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14d, r14d

row_loop:
    cmp r14, r13
    jae final_offset

    mov r8, r13
    sub r8, r14
    cmp r8, 16
    jbe row_count_ready
    mov r8d, 16
row_count_ready:
    call emit_offset
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    add rbx, 2
    lea rsi, [r12 + r14]
    xor r9d, r9d

hex_loop:
    cmp r9, r8
    jae blank_slot

    movzx edx, byte ptr [rsi + r9]
    mov ecx, edx
    shr ecx, 4
    mov cl, byte ptr [rdi + rcx]
    mov byte ptr [rbx], cl
    and edx, 15
    mov dl, byte ptr [rdi + rdx]
    mov byte ptr [rbx + 1], dl
    mov byte ptr [rbx + 2], ' '
    jmp slot_done

blank_slot:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    mov byte ptr [rbx + 2], ' '

slot_done:
    add rbx, 3
    cmp r9, 7
    jne no_middle_space
    mov byte ptr [rbx], ' '
    inc rbx
no_middle_space:
    inc r9
    cmp r9, 16
    jb hex_loop

    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], '|'
    add rbx, 2
    xor r9d, r9d

ascii_loop:
    cmp r9, r8
    jae ascii_done
    mov al, byte ptr [rsi + r9]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [rbx], al
    inc rbx
    inc r9
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx], '|'
    mov byte ptr [rbx + 1], 10
    add rbx, 2
    mov rdx, rbx
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall
    add r14, r8
    jmp row_loop

final_offset:
    call emit_offset
    mov byte ptr [rbx], 10
    inc rbx
    mov rdx, rbx
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov rbx, r15
    mov eax, r14d
    mov ecx, 8
    lea rdi, [rip + digits]
offset_digit:
    mov edx, eax
    shr edx, 28
    mov dl, byte ptr [rdi + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    shl eax, 4
    dec ecx
    jnz offset_digit
    ret
