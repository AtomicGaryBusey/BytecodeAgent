.globl _start

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    add x1, x19, x20
    mov x0, #0
    movz x2, #1, lsl #16
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, state_stack
    add x23, x23, :lo12:state_stack
    mov x21, #0
    mov x22, #0
    mov x24, #0

parse_loop:
    cmp x21, x20
    b.eq at_end
    ldrb w9, [x19, x21]
    cmp w9, #32
    b.eq skip_ws
    cmp w9, #9
    b.eq skip_ws
    cmp w9, #10
    b.eq skip_ws
    cmp w9, #13
    b.eq skip_ws

    cmp x24, #0
    b.eq value
    cmp x24, #2
    b.eq array_first
    cmp x24, #4
    b.eq value
    cmp x24, #7
    b.eq value
    cmp x24, #5
    b.eq object_first
    cmp x24, #9
    b.eq object_key
    cmp x24, #6
    b.eq object_colon
    cmp x24, #3
    b.eq array_after
    cmp x24, #8
    b.eq object_after
    b error

skip_ws:
    add x21, x21, #1
    b parse_loop

at_end:
    cmp x24, #1
    b.eq success
    b error

array_first:
    cmp w9, #93
    b.eq close_container
    b value

object_first:
    cmp w9, #125
    b.eq close_container

object_key:
    cmp w9, #34
    b.ne error
    mov x24, #6
    bl parse_string
    b parse_loop

object_colon:
    cmp w9, #58
    b.ne error
    add x21, x21, #1
    mov x24, #7
    b parse_loop

array_after:
    cmp w9, #93
    b.eq close_container
    cmp w9, #44
    b.ne error
    add x21, x21, #1
    mov x24, #4
    b parse_loop

object_after:
    cmp w9, #125
    b.eq close_container
    cmp w9, #44
    b.ne error
    add x21, x21, #1
    mov x24, #9
    b parse_loop

close_container:
    add x21, x21, #1
    sub x22, x22, #1
    ldrb w24, [x23, x22]
    b parse_loop

value:
    cmp x24, #0
    b.ne value_nested
    mov x24, #1
    b value_token

value_nested:
    cmp x24, #7
    b.ne value_array
    mov x24, #8
    b value_token

value_array:
    mov x24, #3

value_token:
    cmp w9, #123
    b.eq open_object
    cmp w9, #91
    b.eq open_array
    cmp w9, #34
    b.eq string_value
    cmp w9, #45
    b.eq number_value
    cmp w9, #48
    b.lo value_other
    cmp w9, #57
    b.ls number_value

value_other:
    cmp w9, #116
    b.eq true_value
    cmp w9, #102
    b.eq false_value
    cmp w9, #110
    b.eq null_value
    b error

open_object:
    strb w24, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov x24, #5
    b parse_loop

open_array:
    strb w24, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov x24, #2
    b parse_loop

string_value:
    bl parse_string
    b parse_loop

number_value:
    bl parse_number
    b parse_loop

true_value:
    adrp x25, true_text
    add x25, x25, :lo12:true_text
    mov x26, #4
    bl parse_literal
    b parse_loop

false_value:
    adrp x25, false_text
    add x25, x25, :lo12:false_text
    mov x26, #5
    bl parse_literal
    b parse_loop

null_value:
    adrp x25, null_text
    add x25, x25, :lo12:null_text
    mov x26, #4
    bl parse_literal
    b parse_loop

parse_literal:
    cmp x26, #0
    b.eq literal_done
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    ldrb w11, [x25], #1
    cmp w10, w11
    b.ne error
    add x21, x21, #1
    sub x26, x26, #1
    b parse_literal
literal_done:
    ret

parse_string:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    cmp w10, #34
    b.eq string_done
    cmp w10, #92
    b.eq string_escape
    cmp w10, #32
    b.lo error
    add x21, x21, #1
    b string_loop

string_done:
    add x21, x21, #1
    ret

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    cmp w10, #117
    b.eq string_unicode
    cmp w10, #34
    b.eq escape_done
    cmp w10, #92
    b.eq escape_done
    cmp w10, #47
    b.eq escape_done
    cmp w10, #98
    b.eq escape_done
    cmp w10, #102
    b.eq escape_done
    cmp w10, #110
    b.eq escape_done
    cmp w10, #114
    b.eq escape_done
    cmp w10, #116
    b.ne error
escape_done:
    add x21, x21, #1
    b string_loop

string_unicode:
    add x21, x21, #1
    mov x11, #4
hex_loop:
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    cmp w10, #48
    b.lo error
    cmp w10, #57
    b.ls hex_valid
    cmp w10, #65
    b.lo error
    cmp w10, #70
    b.ls hex_valid
    cmp w10, #97
    b.lo error
    cmp w10, #102
    b.hi error
hex_valid:
    add x21, x21, #1
    subs x11, x11, #1
    b.ne hex_loop
    b string_loop

parse_number:
    ldrb w10, [x19, x21]
    cmp w10, #45
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]

number_integer:
    cmp w10, #48
    b.eq number_zero
    cmp w10, #49
    b.lo error
    cmp w10, #57
    b.hi error
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w10, [x19, x21]
    cmp w10, #48
    b.lo number_suffix
    cmp w10, #57
    b.ls number_digits
    b number_suffix

number_zero:
    add x21, x21, #1

number_suffix:
    cmp x21, x20
    b.eq number_done
    ldrb w10, [x19, x21]
    cmp w10, #46
    b.ne number_exponent
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    cmp w10, #48
    b.lo error
    cmp w10, #57
    b.hi error
fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w10, [x19, x21]
    cmp w10, #48
    b.lo number_exponent
    cmp w10, #57
    b.ls fraction_digits

number_exponent:
    cmp w10, #101
    b.eq exponent_start
    cmp w10, #69
    b.ne number_done
exponent_start:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
    cmp w10, #43
    b.eq exponent_sign
    cmp w10, #45
    b.ne exponent_first
exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w10, [x19, x21]
exponent_first:
    cmp w10, #48
    b.lo error
    cmp w10, #57
    b.hi error
exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w10, [x19, x21]
    cmp w10, #48
    b.lo number_done
    cmp w10, #57
    b.ls exponent_digits
number_done:
    ret

success:
    mov x0, #1
    adrp x1, ok_text
    add x1, x1, :lo12:ok_text
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x3, x1, #48
    mov w4, #10
    strb w4, [x3]
    mov x5, x21
    mov x6, #10
decimal_loop:
    udiv x7, x5, x6
    msub x4, x7, x6, x5
    add w4, w4, #48
    sub x3, x3, #1
    strb w4, [x3]
    mov x5, x7
    cbnz x5, decimal_loop

    mov x0, #1
    mov x1, x3
    adrp x2, output_buf
    add x2, x2, :lo12:output_buf
    add x2, x2, #49
    sub x2, x2, x3
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"
ok_text:
    .ascii "ok\n"
error_text:
    .ascii "error at "

.section .bss
.balign 16
input_buf:
    .zero 65537
state_stack:
    .zero 1008
output_buf:
    .zero 64
