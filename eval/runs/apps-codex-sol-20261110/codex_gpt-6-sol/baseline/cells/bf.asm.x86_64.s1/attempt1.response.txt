.section .bss
.balign 8
program:
.zero 65537
tape:
.zero 30000
output:
.zero 65536
.balign 4
bracket_stack:
.zero 4000
bracket_map:
.zero 262144

.section .text
.globl _start
_start:
    lea r12, [rip + program]
    xor r15d, r15d

read_input:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r15]
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    jz find_delimiter
    add r15, rax
    jmp read_input

find_delimiter:
    xor ecx, ecx
delimiter_loop:
    cmp rcx, r15
    jae no_delimiter
    cmp byte ptr [r12 + rcx], 33
    je delimiter_found
    inc rcx
    jmp delimiter_loop

delimiter_found:
    mov r13, rcx
    lea r14, [rcx + 1]
    jmp check_brackets

no_delimiter:
    mov r13, r15
    mov r14, r15

check_brackets:
    lea r10, [rip + bracket_stack]
    lea r11, [rip + bracket_map]
    xor ebx, ebx
    xor ecx, ecx

scan_loop:
    cmp rbx, r13
    jae scan_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 91
    je scan_open
    cmp al, 93
    je scan_close
scan_next:
    inc rbx
    jmp scan_loop

scan_open:
    mov dword ptr [r10 + rcx*4], ebx
    inc ecx
    jmp scan_next

scan_close:
    test ecx, ecx
    jz invalid_program
    dec ecx
    mov edx, dword ptr [r10 + rcx*4]
    lea eax, [rbx + 1]
    mov dword ptr [r11 + rdx*4], eax
    lea eax, [rdx + 1]
    mov dword ptr [r11 + rbx*4], eax
    jmp scan_next

scan_done:
    test ecx, ecx
    jnz invalid_program
    lea r8, [rip + tape]
    lea r10, [rip + output]
    xor ebx, ebx
    xor r9d, r9d

execute:
    cmp rbx, r13
    jae finish
    mov al, byte ptr [r12 + rbx]
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 46
    je emit
    cmp al, 44
    je accept
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
next_command:
    inc rbx
    jmp execute

move_right:
    inc r8
    jmp next_command

move_left:
    dec r8
    jmp next_command

increment:
    inc byte ptr [r8]
    jmp next_command

decrement:
    dec byte ptr [r8]
    jmp next_command

emit:
    mov al, byte ptr [r8]
    mov byte ptr [r10 + r9], al
    inc r9
    jmp next_command

accept:
    xor eax, eax
    cmp r14, r15
    jae accept_store
    mov al, byte ptr [r12 + r14]
    inc r14
accept_store:
    mov byte ptr [r8], al
    jmp next_command

loop_open:
    cmp byte ptr [r8], 0
    jne next_command
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

loop_close:
    cmp byte ptr [r8], 0
    je next_command
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

finish:
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r9
    syscall
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall
