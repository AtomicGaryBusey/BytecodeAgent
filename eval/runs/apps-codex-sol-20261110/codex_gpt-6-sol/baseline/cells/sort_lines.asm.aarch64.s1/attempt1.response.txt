.section .rodata
newline:
    .byte 10

.bss
.balign 16
input:
    .space 32769
.balign 16
records:
    .space 16000

.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    adrp x21, records
    add x21, x21, :lo12:records
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #32769
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, parse_start
    add x20, x20, x0
    b read_loop

parse_start:
    mov x22, #0
    mov x23, #0
    mov x24, #0

parse_loop:
    cmp x24, x20
    b.hs parse_end
    ldrb w9, [x19, x24]
    cmp w9, #10
    b.ne parse_next
    add x10, x21, x23, lsl #4
    sub x11, x24, x22
    stp x22, x11, [x10]
    add x23, x23, #1
    add x22, x24, #1
parse_next:
    add x24, x24, #1
    b parse_loop

parse_end:
    cmp x22, x20
    b.hs sort_start
    add x10, x21, x23, lsl #4
    sub x11, x20, x22
    stp x22, x11, [x10]
    add x23, x23, #1

sort_start:
    mov x9, #1

sort_outer:
    cmp x9, x23
    b.hs output_start
    add x18, x21, x9, lsl #4
    ldp x11, x12, [x18]
    sub x10, x9, #1

sort_inner:
    cmp x10, #0
    b.lt sort_insert
    add x18, x21, x10, lsl #4
    ldp x5, x6, [x18]
    mov x15, x5
    mov x16, x11
    cmp x6, x12
    csel x17, x6, x12, lo

compare_bytes:
    cbz x17, compare_lengths
    ldrb w7, [x19, x15]
    ldrb w8, [x19, x16]
    cmp w7, w8
    b.lo sort_insert
    b.hi sort_shift
    add x15, x15, #1
    add x16, x16, #1
    sub x17, x17, #1
    b compare_bytes

compare_lengths:
    cmp x6, x12
    b.ls sort_insert

sort_shift:
    stp x5, x6, [x18, #16]
    sub x10, x10, #1
    b sort_inner

sort_insert:
    add x10, x10, #1
    add x18, x21, x10, lsl #4
    stp x11, x12, [x18]
    add x9, x9, #1
    b sort_outer

output_start:
    mov x9, #0

output_loop:
    cmp x9, x23
    b.hs done
    add x10, x21, x9, lsl #4
    ldp x11, x12, [x10]
    cbz x12, output_newline
    mov x0, #1
    add x1, x19, x11
    mov x2, x12
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x9, #1
    b output_loop

done:
    mov x0, #0
    mov x8, #93
    svc #0
