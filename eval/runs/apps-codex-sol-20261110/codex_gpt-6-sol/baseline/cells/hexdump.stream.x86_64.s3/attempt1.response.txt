.section .bss
input_buf:
    .zero 16
output_buf:
    .zero 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    lea r14, [rip + input_buf]
    lea r15, [rip + hex_digits]

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 16
    sub rdx, r13
    syscall
    test rax, rax
    jz end_of_input
    add r13, rax
    cmp r13, 16
    jne read_more
    call print_row
    jmp read_more

end_of_input:
    test r13, r13
    jz print_total
    call print_row

print_total:
    lea rbx, [rip + output_buf]
    call print_offset
    mov byte ptr [rbx], 10
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

print_row:
    lea rbx, [rip + output_buf]
    call print_offset
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    add rbx, 2

    xor r8d, r8d
hex_slots:
    cmp r8, r13
    jae empty_slot
    movzx edx, byte ptr [r14 + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx], al
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rbx + 1], dl
    mov byte ptr [rbx + 2], ' '
    jmp slot_done

empty_slot:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    mov byte ptr [rbx + 2], ' '

slot_done:
    add rbx, 3
    cmp r8, 7
    jne no_middle_space
    mov byte ptr [rbx], ' '
    inc rbx
no_middle_space:
    inc r8
    cmp r8, 16
    jne hex_slots

    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], '|'
    add rbx, 2
    xor r8d, r8d
ascii_bytes:
    cmp r8, r13
    jae row_end
    mov al, byte ptr [r14 + r8]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [rbx], al
    inc rbx
    inc r8
    jmp ascii_bytes

row_end:
    mov byte ptr [rbx], '|'
    mov byte ptr [rbx + 1], 10
    add rbx, 2
    add r12, r13
    xor r13d, r13d

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, rbx
    sub rdx, rsi
    syscall
    ret

print_offset:
    mov rax, r12
    mov r8d, 8
offset_digit:
    mov rdx, rax
    shr rdx, 28
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    shl eax, 4
    dec r8
    jne offset_digit
    ret
