.text
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x22, #0

read_loop:
    movz x2, #1, lsl #16
    cmp x22, x2
    b.eq read_full
    sub x2, x2, x22
    mov x0, #0
    add x1, x21, x22
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x22, x22, x0
    b read_loop

read_full:
    mov x0, #0
    adrp x1, read_scratch
    add x1, x1, :lo12:read_scratch
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.gt read_full

read_done:
    mov x19, x21
    add x20, x21, x22
    adrp x21, value_stack
    add x21, x21, :lo12:value_stack

line_start:
    cmp x19, x20
    b.eq program_exit
    mov x22, #0

token_loop:
    cmp x19, x20
    b.eq line_done
    ldrb w9, [x19]
    cmp w9, #10
    b.eq line_done
    cmp w9, #32
    b.eq separator
    cmp w9, #9
    b.eq separator
    mov x24, x19

scan_token:
    cmp x19, x20
    b.eq token_end
    ldrb w9, [x19]
    cmp w9, #10
    b.eq token_end
    cmp w9, #32
    b.eq token_end
    cmp w9, #9
    b.eq token_end
    add x19, x19, #1
    b scan_token

separator:
    add x19, x19, #1
    b token_loop

token_end:
    mov x25, x19
    sub x9, x25, x24
    cmp x9, #1
    b.ne parse_literal
    ldrb w14, [x24]
    cmp w14, #'+'
    b.eq apply_operator
    cmp w14, #'-'
    b.eq apply_operator
    cmp w14, #'*'
    b.eq apply_operator
    cmp w14, #'/'
    b.eq apply_operator
    cmp w14, #'%'
    b.eq apply_operator

parse_literal:
    mov x26, #0
    mov x27, #0
    mov x9, x24
    ldrb w10, [x9]
    cmp w10, #'-'
    b.ne check_digits
    mov x27, #1
    add x9, x9, #1

check_digits:
    cmp x9, x25
    b.eq line_error
    sub x10, x25, x9
    cmp x10, #18
    b.hi line_error

digit_loop:
    ldrb w10, [x9], #1
    cmp w10, #'0'
    b.lo line_error
    cmp w10, #'9'
    b.hi line_error
    sub w10, w10, #'0'
    mov x11, #10
    madd x26, x26, x11, x10
    cmp x9, x25
    b.ne digit_loop
    cbz x27, push_literal
    neg x26, x26

push_literal:
    str x26, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop

apply_operator:
    cmp x22, #2
    b.lo line_error
    sub x9, x22, #1
    ldr x12, [x21, x9, lsl #3]
    sub x9, x22, #2
    ldr x11, [x21, x9, lsl #3]
    cmp w14, #'+'
    b.eq op_add
    cmp w14, #'-'
    b.eq op_sub
    cmp w14, #'*'
    b.eq op_mul
    cbz x12, line_error
    movz x10, #0x8000, lsl #48
    cmp x11, x10
    b.ne op_divide
    cmn x12, #1
    b.eq line_error

op_divide:
    sdiv x13, x11, x12
    cmp w14, #'/'
    b.eq store_result
    msub x13, x13, x12, x11
    b store_result

op_add:
    add x13, x11, x12
    b store_result

op_sub:
    sub x13, x11, x12
    b store_result

op_mul:
    mul x13, x11, x12

store_result:
    str x13, [x21, x9, lsl #3]
    sub x22, x22, #1
    b token_loop

line_done:
    cmp x22, #1
    b.ne line_error
    ldr x9, [x21]
    mov x27, #0
    cmp x9, #0
    b.ge format_result
    mov x27, #1
    neg x9, x9

format_result:
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    add x25, x25, #32
    sub x24, x25, #1
    mov w10, #10
    strb w10, [x24]

format_digit:
    mov x10, #10
    udiv x11, x9, x10
    msub x12, x11, x10, x9
    add w12, w12, #'0'
    strb w12, [x24, #-1]!
    mov x9, x11
    cbnz x9, format_digit
    cbz x27, write_result
    mov w12, #'-'
    strb w12, [x24, #-1]!

write_result:
    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0
    b line_advance

line_error:
    cmp x19, x20
    b.eq write_error
    ldrb w9, [x19]
    cmp w9, #10
    b.eq write_error
    add x19, x19, #1
    b line_error

write_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0

line_advance:
    cmp x19, x20
    b.eq line_start
    add x19, x19, #1
    b line_start

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_msg:
    .ascii "error\n"

.bss
.balign 8
input_buf:
    .zero 65536
read_scratch:
    .zero 1
.balign 8
value_stack:
    .zero 8008
output_buf:
    .zero 32
