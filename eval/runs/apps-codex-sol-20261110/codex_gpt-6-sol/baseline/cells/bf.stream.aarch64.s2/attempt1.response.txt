.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x28, x20
    mov x21, x20
    mov x9, #0

find_separator:
    cmp x9, x20
    b.hs separator_done
    ldrb w10, [x19, x9]
    cmp w10, #33
    b.eq separator_found
    add x9, x9, #1
    b find_separator

separator_found:
    add x21, x9, #1
    mov x20, x9

separator_done:
    adrp x25, jump_table
    add x25, x25, :lo12:jump_table
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x9, #0
    mov x10, #0

check_brackets:
    cmp x9, x20
    b.hs brackets_checked
    ldrb w12, [x19, x9]
    cmp w12, #91
    b.eq open_bracket
    cmp w12, #93
    b.eq close_bracket

next_bracket:
    add x9, x9, #1
    b check_brackets

open_bracket:
    str w9, [x11, x10, lsl #2]
    add x10, x10, #1
    b next_bracket

close_bracket:
    cbz x10, invalid_program
    sub x10, x10, #1
    ldr w12, [x11, x10, lsl #2]
    str w9, [x25, x12, lsl #2]
    str w12, [x25, x9, lsl #2]
    b next_bracket

brackets_checked:
    cbnz x10, invalid_program
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x23, x22
    mov x24, #0
    mov x27, #0

execute:
    cmp x24, x20
    b.hs write_output
    ldrb w9, [x19, x24]
    add x24, x24, #1
    cmp w9, #62
    b.eq move_right
    cmp w9, #60
    b.eq move_left
    cmp w9, #43
    b.eq increment_cell
    cmp w9, #45
    b.eq decrement_cell
    cmp w9, #46
    b.eq output_cell
    cmp w9, #44
    b.eq input_cell
    cmp w9, #91
    b.eq loop_start
    cmp w9, #93
    b.eq loop_end
    b execute

move_right:
    add x23, x23, #1
    b execute

move_left:
    sub x23, x23, #1
    b execute

increment_cell:
    ldrb w9, [x23]
    add w9, w9, #1
    strb w9, [x23]
    b execute

decrement_cell:
    ldrb w9, [x23]
    sub w9, w9, #1
    strb w9, [x23]
    b execute

output_cell:
    ldrb w9, [x23]
    strb w9, [x26, x27]
    add x27, x27, #1
    b execute

input_cell:
    cmp x21, x28
    b.hs input_exhausted
    ldrb w9, [x19, x21]
    add x21, x21, #1
    strb w9, [x23]
    b execute

    b execute

loop_start:
    ldrb w9, [x23]
    cbnz w9, execute
    sub x9, x24, #1
    ldr w24, [x25, x9, lsl #2]
    add x24, x24, #1
    b execute

loop_end:
    ldrb w9, [x23]
    cbz w9, execute
    sub x9, x24, #1
    ldr w24, [x25, x9, lsl #2]
    add x24, x24, #1
    b execute

write_output:

    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0

success:

    mov x8, #93
    svc #0

    mov x8, #93
