.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, x19
    mov x24, x19
    add x25, x19, x20

scan_loop:
    cmp x23, x25
    b.eq final_line
    ldrb w27, [x23]
    cmp w27, #10
    b.eq line_end
    add x23, x23, #1
    b scan_loop

line_end:
    mov x26, x23
reverse_line:
    cmp x26, x24
    b.eq append_newline
    ldrb w27, [x26, #-1]!
    strb w27, [x22], #1
    b reverse_line

append_newline:
    mov w27, #10
    strb w27, [x22], #1
    add x23, x23, #1
    mov x24, x23
    b scan_loop

final_line:
    cmp x24, x25
    b.eq write_output
    mov x26, x25
reverse_final:
    cmp x26, x24
    b.eq final_newline
    ldrb w27, [x26, #-1]!
    strb w27, [x22], #1
    b reverse_final

final_newline:
    mov w27, #10
    strb w27, [x22], #1

write_output:
    sub x2, x22, x21
    cbz x2, exit_success
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.balign 16
input_buf:
    .zero 65537
output_buf:
    .zero 65537
