.globl _start

.section .text
_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65537
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0

scan_loop:
    cmp x21, x20
    b.eq input_end
    ldrb w9, [x19, x21]
    add x21, x21, #1
    cmp w9, #10
    b.ne scan_loop
    b process_line

input_end:
    cmp x22, x20
    b.eq exit_ok
    mov w9, #10
    strb w9, [x19, x20]
    add x20, x20, #1
    mov x21, x20

process_line:
    mov x24, x22
    sub x25, x21, #1

reverse_loop:
    cmp x24, x25
    b.hs write_line
    sub x25, x25, #1
    cmp x24, x25
    b.hs write_line
    ldrb w9, [x19, x24]
    ldrb w10, [x19, x25]
    strb w10, [x19, x24]
    strb w9, [x19, x25]
    add x24, x24, #1
    b reverse_loop

write_line:
    mov x0, #1
    add x1, x19, x22
    sub x2, x21, x22
    mov x8, #64
    svc #0
    mov x22, x21
    b scan_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
    .balign 16
buffer:
    .zero 65537
