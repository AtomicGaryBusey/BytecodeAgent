.globl _start

.section .text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x24, x20
    b.eq final_line
    ldrb w9, [x19, x24]
    cmp w9, #10
    b.eq complete_line
    add x24, x24, #1
    b scan_loop

complete_line:
    mov x25, x24

reverse_loop:
    cmp x25, x23
    b.eq append_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x21, x22]
    add x22, x22, #1
    b reverse_loop

append_newline:
    mov w9, #10
    strb w9, [x21, x22]
    add x22, x22, #1
    add x24, x24, #1
    mov x23, x24
    b scan_loop

final_line:
    cmp x23, x20
    b.eq write_output
    mov x25, x20

reverse_final:
    cmp x25, x23
    b.eq append_final_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x21, x22]
    add x22, x22, #1
    b reverse_final

append_final_newline:
    mov w9, #10
    strb w9, [x21, x22]
    add x22, x22, #1

write_output:
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537
