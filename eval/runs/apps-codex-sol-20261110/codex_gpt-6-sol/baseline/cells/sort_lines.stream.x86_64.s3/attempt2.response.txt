.globl _start

.section .text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea rbx, [rip + lines]
    xor r14d, r14d
    mov r8, r12
    mov r9, r12
    lea r10, [r12 + r13]

scan_loop:
    cmp r8, r10
    je scan_done
    cmp byte ptr [r8], 10
    jne scan_next

    lea r11, [r14 + r14]
    mov qword ptr [rbx + r11*8], r9
    mov rax, r8
    sub rax, r9
    mov qword ptr [rbx + r11*8 + 8], rax
    inc r14
    lea r9, [r8 + 1]

scan_next:
    inc r8
    jmp scan_loop

scan_done:
    cmp r9, r10
    je sort_start
    lea r11, [r14 + r14]
    mov qword ptr [rbx + r11*8], r9
    mov rax, r10
    sub rax, r9
    mov qword ptr [rbx + r11*8 + 8], rax
    inc r14

sort_start:
    mov r10d, 1

sort_next:
    cmp r10, r14
    jae output_start
    lea rax, [r10 + r10]
    mov r12, qword ptr [rbx + rax*8]
    mov r13, qword ptr [rbx + rax*8 + 8]
    mov r15, r10

sort_shift:
    test r15, r15
    jz sort_insert
    mov rax, r15
    dec rax
    lea rax, [rax + rax]
    mov rsi, qword ptr [rbx + rax*8]
    mov rcx, qword ptr [rbx + rax*8 + 8]
    mov rdi, r12
    mov rdx, r13
    call compare_lines
    test eax, eax
    jns sort_insert

    lea rax, [r15 + r15]
    mov r8, qword ptr [rbx + rax*8 - 16]
    mov r9, qword ptr [rbx + rax*8 - 8]
    mov qword ptr [rbx + rax*8], r8
    mov qword ptr [rbx + rax*8 + 8], r9
    dec r15
    jmp sort_shift

sort_insert:
    lea rax, [r15 + r15]
    mov qword ptr [rbx + rax*8], r12
    mov qword ptr [rbx + rax*8 + 8], r13
    inc r10
    jmp sort_next

compare_lines:
    mov r8, rdx
    mov r9, rcx
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jne compare_different

compare_lengths:
    cmp r8, r9
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_different:
    movzx eax, byte ptr [rdi - 1]
    movzx edx, byte ptr [rsi - 1]
    cmp eax, edx
    jb compare_less

compare_greater:
    mov eax, 1
    ret

compare_less:
    mov eax, -1
    ret

output_start:
    xor r15d, r15d

output_loop:
    cmp r15, r14
    jae finish
    lea r8, [r15 + r15]
    mov rdx, qword ptr [rbx + r8*8 + 8]
    test rdx, rdx
    jz output_newline
    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [rbx + r8*8]
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r15
    jmp output_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.balign 16
input_buffer:
    .zero 32769
.balign 16
lines:
    .zero 16000
