.section .bss
.balign 8
input_buf:
.zero 65537
bracket_map:
.zero 524288
bracket_stack:
.zero 8000
tape:
.zero 30000
output_buf:
.zero 65536

.text
.globl _start
_start:
    lea rbp, [rip + input_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbp + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz find_separator
    add r12, rax
    jmp read_loop

find_separator:
    mov r13, r12
    mov r14, r12
    xor ebx, ebx

separator_loop:
    cmp rbx, r12
    jae check_brackets
    cmp byte ptr [rbp + rbx], 33
    je separator_found
    inc rbx
    jmp separator_loop

separator_found:
    mov r13, rbx
    lea r14, [rbx + 1]

check_brackets:
    lea r8, [rip + bracket_map]
    lea r9, [rip + bracket_stack]
    xor r15d, r15d
    xor ebx, ebx

bracket_loop:
    cmp rbx, r13
    jae brackets_done
    mov al, byte ptr [rbp + rbx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
bracket_next:
    inc rbx
    jmp bracket_loop

open_bracket:
    mov qword ptr [r9 + r15*8], rbx
    inc r15
    jmp bracket_next

close_bracket:
    test r15, r15
    jz invalid_brackets
    dec r15
    mov rax, qword ptr [r9 + r15*8]
    mov qword ptr [r8 + rax*8], rbx
    mov qword ptr [r8 + rbx*8], rax
    jmp bracket_next

brackets_done:
    test r15, r15
    jnz invalid_brackets
    lea r8, [rip + tape]
    lea rdi, [rip + output_buf]
    lea rdx, [rip + bracket_map]
    xor r9d, r9d
    xor ebx, ebx

execute:
    cmp rbx, r13
    jae write_output
    mov al, byte ptr [rbp + rbx]
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment_cell
    cmp al, 45
    je decrement_cell
    cmp al, 46
    je output_cell
    cmp al, 44
    je input_cell
    cmp al, 91
    je loop_start
    cmp al, 93
    je loop_end
next_command:
    inc rbx
    jmp execute

move_right:
    inc r8
    jmp next_command

move_left:
    dec r8
    jmp next_command

increment_cell:
    inc byte ptr [r8]
    jmp next_command

decrement_cell:
    dec byte ptr [r8]
    jmp next_command

output_cell:
    mov al, byte ptr [r8]
    mov byte ptr [rdi + r9], al
    inc r9
    jmp next_command

input_cell:
    xor eax, eax
    cmp r14, r12
    jae store_input
    mov al, byte ptr [rbp + r14]
    inc r14
store_input:
    mov byte ptr [r8], al
    jmp next_command

loop_start:
    cmp byte ptr [r8], 0
    jne next_command
    mov rbx, qword ptr [rdx + rbx*8]
    jmp next_command

loop_end:
    cmp byte ptr [r8], 0
    je next_command
    mov rbx, qword ptr [rdx + rbx*8]
    jmp next_command

write_output:
    test r9, r9
    jz success
    mov rsi, rdi
    mov edi, 1
    mov rdx, r9
    mov eax, 1
    syscall

success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_brackets:
    mov edi, 1
    mov eax, 60
    syscall
