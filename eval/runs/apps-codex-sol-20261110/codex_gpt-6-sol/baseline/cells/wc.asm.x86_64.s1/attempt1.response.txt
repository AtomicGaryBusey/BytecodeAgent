.globl _start

.section .bss
buffer:
    .zero 4096
outbuf:
    .zero 96

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jle print_counts

    add r14, rax
    mov r9, rax
    lea rsi, [rip + buffer]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jne next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec r9
    jne scan_loop
    jmp read_loop

print_counts:
    lea rdi, [rip + outbuf + 95]
    mov byte ptr [rdi], 10
    dec rdi

    mov rax, r14
    call emit_decimal
    mov byte ptr [rdi], 32
    dec rdi

    mov rax, r13
    call emit_decimal
    mov byte ptr [rdi], 32
    dec rdi

    mov rax, r12
    call emit_decimal

    lea rsi, [rdi + 1]
    lea rdx, [rip + outbuf + 96]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_decimal:
    mov ecx, 10
digit_loop:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jne digit_loop
    ret
