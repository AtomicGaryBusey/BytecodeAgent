.section .bss
.balign 16
input_buffer:
.zero 4096
output_buffer:
.zero 64

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, 4096
    syscall
    test rax, rax
    jz print_counts

    add r14, rax
    mov r9, rax
    xor r8d, r8d

count_byte:
    movzx eax, byte ptr [rbx + r8]
    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    inc r8
    cmp r8, r9
    jb count_byte
    jmp read_more

print_counts:
    mov byte ptr [rip + output_buffer + 63], 10
    lea rdi, [rip + output_buffer + 63]
    mov r10d, 10

    mov rax, r14
    call decimal
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call decimal
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call decimal

    lea rdx, [rip + output_buffer + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

decimal:
    xor edx, edx
    div r10
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal
    ret
