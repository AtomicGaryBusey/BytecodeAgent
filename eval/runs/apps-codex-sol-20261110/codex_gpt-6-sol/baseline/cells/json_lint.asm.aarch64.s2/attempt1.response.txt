.section .text
.globl _start
_start:
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf
    mov x23, #0

read_loop:
    mov x0, #0
    add x1, x22, x23
    mov x2, #65537
    sub x2, x2, x23
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x23, x23, x0
    b read_loop

read_done:
    mov x21, x22
    mov x19, x22
    add x20, x22, x23
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.ne fail

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

fail:
    mov x0, #1
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    mov x8, #64
    svc #0

    sub x25, x19, x21
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    add x26, x26, #32
    mov w0, #10
    strb w0, [x26]
    add x27, x26, #1
    mov x28, #10

decimal_loop:
    udiv x0, x25, x28
    msub x1, x0, x28, x25
    add w1, w1, #48
    strb w1, [x26, #-1]!
    mov x25, x0
    cbnz x25, decimal_loop

    mov x0, #1
    mov x1, x26
    sub x2, x27, x26
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x19, x20
    b.eq ws_done
    ldrb w0, [x19]
    cmp w0, #0x20
    b.eq ws_advance
    cmp w0, #0x09
    b.eq ws_advance
    cmp w0, #0x0a
    b.eq ws_advance
    cmp w0, #0x0d
    b.ne ws_done
ws_advance:
    add x19, x19, #1
    b skip_ws
ws_done:
    ret

parse_value:
    str x30, [sp, #-16]!
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x7b
    b.eq value_object
    cmp w0, #0x5b
    b.eq value_array
    cmp w0, #0x22
    b.eq value_string
    cmp w0, #0x74
    b.eq value_true
    cmp w0, #0x66
    b.eq value_false
    cmp w0, #0x6e
    b.eq value_null
    cmp w0, #0x2d
    b.eq value_number
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.hi fail
value_number:
    bl parse_number
    b value_done
value_object:
    bl parse_object
    b value_done
value_array:
    bl parse_array
    b value_done
value_string:
    bl parse_string
    b value_done
value_true:
    adrp x0, true_text
    add x0, x0, :lo12:true_text
    mov x1, #4
    bl match_literal
    b value_done
value_false:
    adrp x0, false_text
    add x0, x0, :lo12:false_text
    mov x1, #5
    bl match_literal
    b value_done
value_null:
    adrp x0, null_text
    add x0, x0, :lo12:null_text
    mov x1, #4
    bl match_literal
value_done:
    ldr x30, [sp], #16
    ret

match_literal:
    cbz x1, literal_done
literal_next:
    cmp x19, x20
    b.eq fail
    ldrb w2, [x19]
    ldrb w3, [x0], #1
    cmp w2, w3
    b.ne fail
    add x19, x19, #1
    subs x1, x1, #1
    b.ne literal_next
literal_done:
    ret

parse_object:
    str x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x7d
    b.eq object_close
object_member:
    bl parse_string
    bl skip_ws
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x3a
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x7d
    b.eq object_close
    cmp w0, #0x2c
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    b object_member
object_close:
    add x19, x19, #1
    ldr x30, [sp], #16
    ret

parse_array:
    str x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x5d
    b.eq array_close
array_element:
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x5d
    b.eq array_close
    cmp w0, #0x2c
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    b array_element
array_close:
    add x19, x19, #1
    ldr x30, [sp], #16
    ret

parse_string:
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x22
    b.ne fail
    add x19, x19, #1
string_next:
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x20
    b.lo fail
    cmp w0, #0x22
    b.eq string_close
    cmp w0, #0x5c
    b.eq string_escape
    add x19, x19, #1
    b string_next
string_close:
    add x19, x19, #1
    ret
string_escape:
    add x19, x19, #1
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x22
    b.eq escape_done
    cmp w0, #0x5c
    b.eq escape_done
    cmp w0, #0x2f
    b.eq escape_done
    cmp w0, #0x62
    b.eq escape_done
    cmp w0, #0x66
    b.eq escape_done
    cmp w0, #0x6e
    b.eq escape_done
    cmp w0, #0x72
    b.eq escape_done
    cmp w0, #0x74
    b.eq escape_done
    cmp w0, #0x75
    b.ne fail
    add x19, x19, #1
    mov x1, #4
hex_next:
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.ls hex_valid
    cmp w0, #0x41
    b.lo fail
    cmp w0, #0x46
    b.ls hex_valid
    cmp w0, #0x61
    b.lo fail
    cmp w0, #0x66
    b.hi fail
hex_valid:
    add x19, x19, #1
    subs x1, x1, #1
    b.ne hex_next
    b string_next
escape_done:
    add x19, x19, #1
    b string_next

parse_number:
    ldrb w0, [x19]
    cmp w0, #0x2d
    b.ne number_integer
    add x19, x19, #1
    cmp x19, x20
    b.eq fail
number_integer:
    ldrb w0, [x19]
    cmp w0, #0x30
    b.eq number_zero
    cmp w0, #0x31
    b.lo fail
    cmp w0, #0x39
    b.hi fail
number_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq number_done
    ldrb w0, [x19]
    cmp w0, #0x30
    b.lo number_suffix
    cmp w0, #0x39
    b.ls number_digits
    b number_suffix
number_zero:
    add x19, x19, #1
    cmp x19, x20
    b.eq number_done
    ldrb w0, [x19]
number_suffix:
    cmp w0, #0x2e
    b.eq number_fraction
    cmp w0, #0x65
    b.eq number_exponent
    cmp w0, #0x45
    b.eq number_exponent
number_done:
    ret

number_fraction:
    add x19, x19, #1
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.hi fail
fraction_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq number_done
    ldrb w0, [x19]
    cmp w0, #0x30
    b.lo fraction_end
    cmp w0, #0x39
    b.ls fraction_digits
fraction_end:
    cmp w0, #0x65
    b.eq number_exponent
    cmp w0, #0x45
    b.eq number_exponent
    ret

number_exponent:
    add x19, x19, #1
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
    cmp w0, #0x2b
    b.eq exponent_sign
    cmp w0, #0x2d
    b.ne exponent_first
exponent_sign:
    add x19, x19, #1
    cmp x19, x20
    b.eq fail
    ldrb w0, [x19]
exponent_first:
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.hi fail
exponent_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq number_done
    ldrb w0, [x19]
    cmp w0, #0x30
    b.lo number_done
    cmp w0, #0x39
    b.ls exponent_digits
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.section .bss
.balign 16
input_buf:
    .zero 65537
output_buf:
    .zero 40
