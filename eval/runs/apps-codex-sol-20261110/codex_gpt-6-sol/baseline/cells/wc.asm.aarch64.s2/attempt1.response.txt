.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x25, input_buf
    add x25, x25, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x25
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output

    add x21, x21, x0
    mov x9, x0
    mov x10, x25

scan_loop:
    ldrb w11, [x10], #1
    cmp w11, #10
    b.eq newline
    cmp w11, #32
    b.eq whitespace
    sub w12, w11, #9
    cmp w12, #4
    b.ls whitespace

    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

newline:
    add x19, x19, #1

whitespace:
    mov x22, #0

next_byte:
    subs x9, x9, #1
    b.ne scan_loop
    b read_loop

output:
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    add x23, x23, #64
    mov x24, x23

    mov w9, #10
    strb w9, [x23, #-1]!

    mov x0, x21
    bl prepend_decimal
    mov w9, #32
    strb w9, [x23, #-1]!

    mov x0, x20
    bl prepend_decimal
    mov w9, #32
    strb w9, [x23, #-1]!

    mov x0, x19
    bl prepend_decimal

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

prepend_decimal:
    mov x4, #10
1:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    strb w5, [x23, #-1]!
    mov x0, x3
    cbnz x0, 1b
    ret

.bss
.balign 16
input_buf:
    .space 4096
output_buf:
    .space 128
