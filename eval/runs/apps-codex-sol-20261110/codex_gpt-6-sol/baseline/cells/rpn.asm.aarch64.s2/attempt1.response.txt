.section .bss
.balign 16
input_buffer:
    .space 65552
value_stack:
    .space 8192
output_buffer:
    .space 32

.section .rodata
error_message:
    .ascii "error\n"

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x21, #0x10
    movk x21, #1, lsl #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x20, x19, x20
    mov x21, x19
    mov x22, x21

find_line:
    cmp x22, x20
    b.hs input_done
    ldrb w9, [x22]
    cmp w9, #10
    b.eq complete_line
    add x22, x22, #1
    b find_line

complete_line:
    mov x23, x22
    bl evaluate_line
    add x21, x22, #1
    mov x22, x21
    b find_line

input_done:
    cmp x21, x20
    b.hs exit_program
    mov x23, x20
    bl evaluate_line

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

evaluate_line:
    stp x29, x30, [sp, #-16]!
    mov x24, x21
    adrp x25, value_stack
    add x25, x25, :lo12:value_stack
    mov x26, #0

skip_separators:
    cmp x24, x23
    b.hs finish_line
    ldrb w9, [x24]
    cmp w9, #32
    b.eq advance_separator
    cmp w9, #9
    b.ne token_start

advance_separator:
    add x24, x24, #1
    b skip_separators

token_start:
    mov x27, x24
    mov x28, x24

find_token_end:
    cmp x28, x23
    b.hs classify_token
    ldrb w9, [x28]
    cmp w9, #32
    b.eq classify_token
    cmp w9, #9
    b.eq classify_token
    add x28, x28, #1
    b find_token_end

classify_token:
    sub x9, x28, x27
    cmp x9, #1
    b.ne parse_literal
    ldrb w14, [x27]
    cmp w14, #43
    b.eq apply_operator
    cmp w14, #45
    b.eq apply_operator
    cmp w14, #42
    b.eq apply_operator
    cmp w14, #47
    b.eq apply_operator
    cmp w14, #37
    b.eq apply_operator

parse_literal:
    mov x10, #0
    mov x11, #0
    mov x12, #0
    mov x9, x27
    ldrb w13, [x9]
    cmp w13, #45
    b.ne digit_loop
    mov x12, #1
    add x9, x9, #1
    cmp x9, x28
    b.hs line_error

digit_loop:
    cmp x9, x28
    b.hs literal_done
    ldrb w13, [x9], #1
    sub w13, w13, #48
    cmp w13, #9
    b.hi line_error
    add x11, x11, #1
    cmp x11, #18
    b.hi line_error
    mov x14, #10
    mul x10, x10, x14
    add x10, x10, x13
    b digit_loop

literal_done:
    cbz x11, line_error
    cbz x12, push_literal
    neg x10, x10

push_literal:
    str x10, [x25, x26, lsl #3]
    add x26, x26, #1
    mov x24, x28
    b skip_separators

apply_operator:
    cmp x26, #2
    b.lo line_error
    sub x26, x26, #1
    ldr x10, [x25, x26, lsl #3]
    sub x26, x26, #1
    ldr x11, [x25, x26, lsl #3]
    cmp w14, #43
    b.eq operator_add
    cmp w14, #45
    b.eq operator_subtract
    cmp w14, #42
    b.eq operator_multiply

    cbz x10, line_error
    mov x13, #1
    lsl x13, x13, #63
    cmp x11, x13
    b.ne operator_divide
    cmn x10, #1
    b.eq line_error

operator_divide:
    sdiv x12, x11, x10
    cmp w14, #47
    b.eq store_result
    msub x12, x12, x10, x11
    b store_result

operator_add:
    add x12, x11, x10
    b store_result

operator_subtract:
    sub x12, x11, x10
    b store_result

operator_multiply:
    mul x12, x11, x10

store_result:
    str x12, [x25, x26, lsl #3]
    add x26, x26, #1
    mov x24, x28
    b skip_separators

finish_line:
    cmp x26, #1
    b.ne line_error
    ldr x10, [x25]
    mov x11, #0
    cmp x10, #0
    b.ge format_result
    mov x11, #1
    neg x10, x10

format_result:
    adrp x16, output_buffer
    add x16, x16, :lo12:output_buffer
    add x16, x16, #32
    sub x15, x16, #1
    mov w9, #10
    strb w9, [x15]
    mov x13, #10

format_digit:
    udiv x12, x10, x13
    msub x14, x12, x13, x10
    add w14, w14, #48
    strb w14, [x15, #-1]!
    mov x10, x12
    cbnz x10, format_digit
    cbz x11, write_result
    mov w9, #45
    strb w9, [x15, #-1]!

write_result:
    mov x0, #1
    mov x1, x15
    sub x2, x16, x15
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret

line_error:
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #6
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret
