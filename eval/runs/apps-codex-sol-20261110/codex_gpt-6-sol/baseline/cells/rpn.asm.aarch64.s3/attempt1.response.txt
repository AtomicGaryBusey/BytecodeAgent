.section .text

    add x19, x19, :lo12:input_buf
    mov x21, x19
    movz x22, #1, lsl #16
    add x22, x22, #1

    mov x1, x21
    sub x2, x22, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0

    mov x20, x21
    mov x21, x19
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack

    cmp x21, x20
    b.eq program_exit
    mov x22, x21

    cmp x22, x20
    b.eq line_found
    ldrb w0, [x22]
    cmp w0, #10
    b.eq line_found
    add x22, x22, #1
    b find_line_end

line_found:
    mov x23, #0

token_start:
    cmp x21, x22
    b.eq line_done
    ldrb w0, [x21]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator
    mov x25, x21

    cmp x21, x22
    b.eq token_found
    ldrb w0, [x21]
    cmp w0, #32
    b.eq token_found
    cmp w0, #9
    b.eq token_found
    add x21, x21, #1
    b find_token_end

token_found:
    sub x26, x21, x25
    cmp x26, #1
    b.ne parse_literal
    ldrb w27, [x25]
    cmp w27, #43
    b.eq apply_operator
    cmp w27, #45
    b.eq apply_operator
    cmp w27, #42
    b.eq apply_operator
    cmp w27, #47
    b.eq apply_operator
    cmp w27, #37
    b.eq apply_operator

parse_literal:
    mov x5, x25
    mov x6, x26
    mov x14, #0
    ldrb w0, [x5]
    cmp w0, #45
    b.ne literal_length
    mov x14, #1
    add x5, x5, #1
    sub x6, x6, #1

literal_length:
    cbz x6, line_error
    cmp x6, #18
    b.hi line_error
    mov x7, #0

literal_digit:
    ldrb w8, [x5], #1
    sub w8, w8, #48
    cmp w8, #9
    b.hi line_error
    add x7, x7, x7, lsl #2
    add x7, x8, x7, lsl #1
    subs x6, x6, #1
    b.ne literal_digit
    cbz x14, push_literal
    neg x7, x7

push_literal:
    str x7, [x24, x23, lsl #3]
    add x23, x23, #1
    b token_start

apply_operator:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #1
    ldr x9, [x24, x23, lsl #3]
    sub x11, x23, #1
    ldr x10, [x24, x11, lsl #3]
    cmp w27, #43
    b.eq op_add
    cmp w27, #45
    b.eq op_sub
    cmp w27, #42
    b.eq op_mul
    cbz x9, line_error
    mov x12, #1
    lsl x12, x12, #63
    cmp x10, x12
    b.ne op_divide
    cmn x9, #1
    b.eq line_error

op_divide:
    sdiv x12, x10, x9
    cmp w27, #47
    b.eq op_quotient
    msub x10, x12, x9, x10
    b store_result

op_quotient:
    mov x10, x12
    b store_result

op_add:
    add x10, x10, x9
    b store_result

op_sub:
    sub x10, x10, x9
    b store_result

    mul x10, x10, x9

    b token_start

    b token_start

    b.ne line_error
    ldr x7, [x24]
    mov x14, #0
    cmp x7, #0
    b.ge magnitude_ready
    mov x14, #1
    neg x7, x7

    add x5, x5, :lo12:output_buf
    add x13, x5, #32
    sub x6, x13, #1
    mov w0, #10
    strb w0, [x6]
    mov x9, #10

    msub x11, x8, x9, x7
    add w11, w11, #48
    strb w11, [x6, #-1]!
    mov x7, x8
    cbnz x7, decimal_digit
    cbz x14, write_result
    mov w0, #45
    strb w0, [x6, #-1]!

    mov x0, #1
    mov x1, x6
    sub x2, x13, x6
    mov x8, #64
    svc #0

    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    mov x8, #64
    svc #0

    mov x8, #93
