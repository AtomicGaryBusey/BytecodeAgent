.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x20, x19, x20
    adrp x21, value_stack
    add x21, x21, :lo12:value_stack

line_loop:
    cmp x19, x20
    b.eq exit_program
    mov x22, #0

scan:
    cmp x19, x20
    b.eq finish_line
    ldrb w9, [x19]
    cmp w9, #10
    b.eq finish_newline
    cmp w9, #32
    b.eq skip_separator
    cmp w9, #9
    b.eq skip_separator
    mov x23, x19

scan_token:
    cmp x19, x20
    b.eq token_done
    ldrb w9, [x19]
    cmp w9, #10
    b.eq token_done
    cmp w9, #32
    b.eq token_done
    cmp w9, #9
    b.eq token_done
    add x19, x19, #1
    b scan_token

token_done:
    sub x24, x19, x23
    cmp x24, #1
    b.ne parse_literal
    ldrb w9, [x23]
    cmp w9, #43
    b.eq apply_operator
    cmp w9, #45
    b.eq apply_operator
    cmp w9, #42
    b.eq apply_operator
    cmp w9, #47
    b.eq apply_operator
    cmp w9, #37
    b.eq apply_operator

parse_literal:
    mov x25, x23
    mov x26, #0
    ldrb w9, [x25]
    cmp w9, #45
    b.ne check_digit_count
    mov x26, #1
    add x25, x25, #1
    sub x24, x24, #1

check_digit_count:
    cbz x24, error_skip
    cmp x24, #18
    b.hi error_skip
    mov x27, #0
    mov x10, #10

parse_digit:
    ldrb w9, [x25], #1
    sub w9, w9, #48
    cmp w9, #9
    b.hi error_skip
    mul x27, x27, x10
    add x27, x27, w9, uxtw
    subs x24, x24, #1
    b.ne parse_digit
    cbz x26, push_literal
    neg x27, x27

push_literal:
    str x27, [x21, x22, lsl #3]
    add x22, x22, #1
    b scan

apply_operator:
    cmp x22, #2
    b.lo error_skip
    sub x14, x22, #1
    ldr x12, [x21, x14, lsl #3]
    sub x14, x14, #1
    ldr x11, [x21, x14, lsl #3]
    cmp w9, #43
    b.eq op_add
    cmp w9, #45
    b.eq op_sub
    cmp w9, #42
    b.eq op_mul
    cbz x12, error_skip
    mov x15, #1
    lsl x15, x15, #63
    cmp x11, x15
    b.ne do_division
    cmn x12, #1
    b.eq error_skip

do_division:
    sdiv x13, x11, x12
    cmp w9, #37
    b.ne store_result
    msub x13, x13, x12, x11
    b store_result

op_add:
    add x13, x11, x12
    b store_result

op_sub:
    sub x13, x11, x12
    b store_result

op_mul:
    mul x13, x11, x12

store_result:
    str x13, [x21, x14, lsl #3]
    add x22, x14, #1
    b scan

skip_separator:
    add x19, x19, #1
    b scan

finish_newline:
    add x19, x19, #1

finish_line:
    cmp x22, #1
    b.ne error_skip
    ldr x27, [x21]
    bl print_number
    b line_loop

error_skip:
    cmp x19, x20
    b.eq report_error
    ldrb w9, [x19], #1
    cmp w9, #10
    b.ne error_skip

report_error:
    bl print_error
    b line_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

print_error:
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    mov x8, #64
    svc #0
    ret

print_number:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x3, x1, #32
    mov x4, x3
    mov w5, #10
    strb w5, [x1, #31]
    sub x1, x3, #1
    mov x5, #10
    mov x6, #0
    cmp x27, #0
    b.ge number_digits
    mov x6, #1
    neg x27, x27

number_digits:
    udiv x7, x27, x5
    msub x9, x7, x5, x27
    add w9, w9, #48
    strb w9, [x1, #-1]!
    mov x27, x7
    cbnz x27, number_digits
    cbz x6, number_write
    mov w9, #45
    strb w9, [x1, #-1]!

number_write:
    sub x2, x4, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
error_text:
    .ascii "error\n"

.section .bss
.balign 16
input_buf:
    .space 65537
.balign 16
value_stack:
    .space 8192
.balign 16
output_buf:
    .space 32
