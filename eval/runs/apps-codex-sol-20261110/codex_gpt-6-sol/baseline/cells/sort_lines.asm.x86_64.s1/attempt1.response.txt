.globl _start

.section .text
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea r13, [rbx]
    lea r8, [rbx]
    lea r10, [rbx + r12]
    lea r9, [rip + lines]
    xor r15d, r15d

parse_loop:
    cmp r8, r10
    je parse_tail
    cmp byte ptr [r8], 10
    jne parse_next
    mov qword ptr [r9], r13
    mov rax, r8
    sub rax, r13
    mov qword ptr [r9 + 8], rax
    add r9, 16
    inc r15
    inc r8
    mov r13, r8
    jmp parse_loop

parse_next:
    inc r8
    jmp parse_loop

parse_tail:
    cmp r13, r10
    je parsed
    mov qword ptr [r9], r13
    sub r10, r13
    mov qword ptr [r9 + 8], r10
    inc r15

parsed:
    test r15, r15
    jz exit_success
    lea r13, [rip + lines]
    lea r14, [rip + scratch]
    mov r12d, 1

sort_pass:
    cmp r12, r15
    jae output_lines
    xor ebx, ebx

merge_group:
    cmp rbx, r15
    jae pass_done

    mov rax, rbx
    shl rax, 4
    lea rdx, [r13 + rax]
    lea r11, [r14 + rax]

    lea rax, [rbx + r12]
    cmp rax, r15
    cmova rax, r15
    mov r9, rax
    shl rax, 4
    lea rcx, [r13 + rax]

    lea rax, [r9 + r12]
    cmp rax, r15
    cmova rax, r15
    mov r10, rax
    shl rax, 4
    lea r10, [r13 + rax]

merge_loop:
    cmp rdx, rcx
    je copy_right
    cmp rcx, r10
    je copy_left

    mov rdi, qword ptr [rdx]
    mov rsi, qword ptr [rcx]
    mov rax, qword ptr [rdx + 8]
    mov r8, qword ptr [rcx + 8]
    cmp rax, r8
    cmova rax, r8
    test rax, rax
    jz compare_lengths

    push rcx
    mov rcx, rax
    repe cmpsb
    pop rcx
    ja take_right
    jne take_left

compare_lengths:
    mov rax, qword ptr [rdx + 8]
    cmp rax, r8
    ja take_right

take_left:
    mov rax, qword ptr [rdx]
    mov qword ptr [r11], rax
    mov rax, qword ptr [rdx + 8]
    mov qword ptr [r11 + 8], rax
    add rdx, 16
    add r11, 16
    jmp merge_loop

take_right:
    mov rax, qword ptr [rcx]
    mov qword ptr [r11], rax
    mov rax, qword ptr [rcx + 8]
    mov qword ptr [r11 + 8], rax
    add rcx, 16
    add r11, 16
    jmp merge_loop

copy_left:
    cmp rdx, rcx
    je group_done
    mov rax, qword ptr [rdx]
    mov qword ptr [r11], rax
    mov rax, qword ptr [rdx + 8]
    mov qword ptr [r11 + 8], rax
    add rdx, 16
    add r11, 16
    jmp copy_left

copy_right:
    cmp rcx, r10
    je group_done
    mov rax, qword ptr [rcx]
    mov qword ptr [r11], rax
    mov rax, qword ptr [rcx + 8]
    mov qword ptr [r11 + 8], rax
    add rcx, 16
    add r11, 16
    jmp copy_right

group_done:
    mov rbx, r10
    sub rbx, r13
    shr rbx, 4
    jmp merge_group

pass_done:
    xchg r13, r14
    shl r12, 1
    jmp sort_pass

output_lines:
    mov rbx, r13
    mov rax, r15
    shl rax, 4
    lea r8, [r13 + rax]
    lea rdi, [rip + output_buf]

output_loop:
    cmp rbx, r8
    je write_output
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    jmp output_loop

write_output:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.balign 16
input_buf:
    .zero 32769
.balign 16
lines:
    .zero 16000
.balign 16
scratch:
    .zero 16000
.balign 16
output_buf:
    .zero 33768
