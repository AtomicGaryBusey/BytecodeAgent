.globl _start

.section .text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r10, [rip + source]
    xor ebx, ebx

find_separator:
    cmp rbx, r12
    jae no_separator
    cmp byte ptr [r10 + rbx], 33
    je separator_found
    inc rbx
    jmp find_separator

separator_found:
    mov r13, rbx
    lea r14, [rbx + 1]
    jmp validate

no_separator:
    mov r13, r12
    mov r14, r12

validate:
    lea r11, [rip + jumps]
    lea rsi, [rip + bracket_stack]
    xor ebx, ebx
    xor r9d, r9d

validate_next:
    cmp rbx, r13
    jae validate_done
    mov al, byte ptr [r10 + rbx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket

validate_advance:
    inc rbx
    jmp validate_next

open_bracket:
    mov dword ptr [rsi + r9*4], ebx
    inc r9
    jmp validate_advance

close_bracket:
    test r9, r9
    jz invalid
    dec r9
    mov edx, dword ptr [rsi + r9*4]
    lea eax, [rbx + 1]
    mov dword ptr [r11 + rdx*4], eax
    lea eax, [rdx + 1]
    mov dword ptr [r11 + rbx*4], eax
    jmp validate_advance

validate_done:
    test r9, r9
    jnz invalid
    lea rbp, [rip + tape]
    lea r8, [rip + output]
    xor r9d, r9d
    xor ebx, ebx

execute:
    cmp rbx, r13
    jae finish
    mov al, byte ptr [r10 + rbx]
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment_cell
    cmp al, 45
    je decrement_cell
    cmp al, 46
    je emit_cell
    cmp al, 44
    je input_cell
    cmp al, 91
    je execute_open
    cmp al, 93
    je execute_close

execute_advance:
    inc rbx
    jmp execute

move_right:
    inc rbp
    jmp execute_advance

move_left:
    dec rbp
    jmp execute_advance

increment_cell:
    inc byte ptr [rbp]
    jmp execute_advance

decrement_cell:
    dec byte ptr [rbp]
    jmp execute_advance

emit_cell:
    mov al, byte ptr [rbp]
    mov byte ptr [r8 + r9], al
    inc r9
    jmp execute_advance

input_cell:
    xor eax, eax
    cmp r14, r12
    jae input_store
    mov al, byte ptr [r10 + r14]
    inc r14

input_store:
    mov byte ptr [rbp], al
    jmp execute_advance

execute_open:
    cmp byte ptr [rbp], 0
    jne execute_advance
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

execute_close:
    cmp byte ptr [rbp], 0
    je execute_advance
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

finish:
    test r9, r9
    jz success
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall

success:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.section .bss
.balign 8
source:
    .zero 65537
.balign 8
jumps:
    .zero 262144
.balign 8
bracket_stack:
    .zero 4000
.balign 8
tape:
    .zero 30000
.balign 8
output:
    .zero 65536
