.section .bss
.balign 16
input_buffer:
.zero 32768
scratch:
.zero 1
.balign 16
descriptors:
.zero 16000
array_a:
.zero 8000
array_b:
.zero 8000
output_buffer:
.zero 33768

.section .text
.globl _start
_start:
    lea rbp, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, 32768
    je read_full
    xor eax, eax
    xor edi, edi
    lea rsi, [rbp + r12]
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

read_full:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + scratch]
    mov edx, 1
    syscall
    test rax, rax
    jnz read_full

parse_input:
    lea rbx, [rip + descriptors]
    lea r8, [rip + array_a]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

scan_loop:
    cmp r13, r12
    jae scan_done
    cmp byte ptr [rbp + r13], 10
    jne scan_next
    mov rax, r13
    sub rax, r14
    call add_line
    lea r14, [r13 + 1]
scan_next:
    inc r13
    jmp scan_loop

scan_done:
    cmp r14, r12
    jae lines_ready
    mov rax, r12
    sub rax, r14
    call add_line

lines_ready:
    test r15, r15
    jz exit_success
    lea r12, [rip + array_a]
    lea r13, [rip + array_b]
    mov r14d, 1

sort_pass:
    cmp r14, r15
    jae emit_lines
    xor ebp, ebp

merge_next:
    cmp rbp, r15
    jae pass_done
    lea rbx, [rbp + r14]
    cmp rbx, r15
    cmova rbx, r15
    lea r10, [rbx + r14]
    cmp r10, r15
    cmova r10, r15
    mov r8, rbp
    mov r9, rbx
    mov r11, rbp

merge_loop:
    cmp r8, rbx
    jae copy_right
    cmp r9, r10
    jae copy_left
    mov rdi, qword ptr [r12 + r8*8]
    mov rsi, qword ptr [r12 + r9*8]
    call compare_lines
    test eax, eax
    jg copy_right

copy_left:
    mov rax, qword ptr [r12 + r8*8]
    mov qword ptr [r13 + r11*8], rax
    inc r8
    inc r11
    jmp merge_loop

copy_right:
    cmp r9, r10
    jae merge_done
    mov rax, qword ptr [r12 + r9*8]
    mov qword ptr [r13 + r11*8], rax
    inc r9
    inc r11
    jmp merge_loop

merge_done:
    mov rbp, r10
    jmp merge_next

pass_done:
    xchg r12, r13
    shl r14, 1
    jmp sort_pass

emit_lines:
    lea rdi, [rip + output_buffer]
    mov rbp, rdi
    xor ebx, ebx

emit_next:
    cmp rbx, r15
    jae write_output
    mov rax, qword ptr [r12 + rbx*8]
    mov rsi, qword ptr [rax]
    mov rcx, qword ptr [rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp emit_next

write_output:
    mov rdx, rdi
    sub rdx, rbp
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

add_line:
    mov rdx, r15
    add rdx, rdx
    lea rcx, [rbx + rdx*8]
    lea rdx, [rbp + r14]
    mov qword ptr [rcx], rdx
    mov qword ptr [rcx + 8], rax
    mov qword ptr [r8 + r15*8], rcx
    inc r15
    ret

compare_lines:
    mov rax, qword ptr [rdi + 8]
    mov rdx, qword ptr [rsi + 8]
    mov rdi, qword ptr [rdi]
    mov rsi, qword ptr [rsi]
    mov rcx, rax
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jne bytes_differ

compare_lengths:
    cmp rax, rdx
    jb line_less
    ja line_greater
    xor eax, eax
    ret

bytes_differ:
    jb line_less

line_greater:
    mov eax, 1
    ret

line_less:
    mov eax, -1
    ret
