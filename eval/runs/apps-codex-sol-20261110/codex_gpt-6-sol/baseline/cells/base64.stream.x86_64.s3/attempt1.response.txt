.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
.zero 49153
output:
.zero 67000

.text
_start:
    xor ebx, ebx

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, rbx
    mov edx, 49153
    sub rdx, rbx
    syscall
    test rax, rax
    jz encode_start
    add rbx, rax
    jmp read_loop

encode_start:
    lea r8, [rip + input]
    lea r10, [rip + output]
    lea r13, [rip + alphabet]
    mov r14, rbx
    xor r12d, r12d

full_group:
    cmp r14, 3
    jb final_group

    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r8 + 1]
    movzx esi, byte ptr [r8 + 2]

    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10], cl

    and eax, 3
    shl eax, 4
    mov ecx, edx
    shr ecx, 4
    or ecx, eax
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10 + 1], cl

    and edx, 15
    shl edx, 2
    mov ecx, esi
    shr ecx, 6
    or ecx, edx
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10 + 2], cl

    and esi, 63
    mov cl, byte ptr [r13 + rsi]
    mov byte ptr [r10 + 3], cl

    add r8, 3
    sub r14, 3
    add r10, 4
    add r12d, 4
    cmp r12d, 76
    jne full_group
    mov byte ptr [r10], 10
    inc r10
    xor r12d, r12d
    jmp full_group

final_group:
    test r14, r14
    jz finish_lines

    movzx eax, byte ptr [r8]
    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10], cl

    and eax, 3
    shl eax, 4
    cmp r14, 2
    je two_bytes

    mov cl, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], cl
    mov byte ptr [r10 + 2], '='
    mov byte ptr [r10 + 3], '='
    jmp final_done

two_bytes:
    movzx edx, byte ptr [r8 + 1]
    mov ecx, edx
    shr ecx, 4
    or ecx, eax
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10 + 1], cl

    and edx, 15
    shl edx, 2
    mov cl, byte ptr [r13 + rdx]
    mov byte ptr [r10 + 2], cl
    mov byte ptr [r10 + 3], '='

final_done:
    add r10, 4
    add r12d, 4

finish_lines:
    test r12d, r12d
    jz write_output
    mov byte ptr [r10], 10
    inc r10

write_output:
    lea rsi, [rip + output]
    mov rdx, r10
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
