.section .bss
.balign 16
input_buffer:
.zero 32769
.balign 16
line_table:
.zero 16000
.balign 16
output_buffer:
.zero 33768

.section .text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    lea r12, [rip + line_table]
    xor ebx, ebx

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + rbx]
    mov edx, 32769
    sub rdx, rbx
    syscall
    test rax, rax
    jz parse_input
    add rbx, rax
    jmp read_loop

parse_input:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

parse_loop:
    cmp r9, rbx
    jae parse_remainder
    cmp byte ptr [r13 + r9], 10
    jne parse_next

    mov rax, r10
    shl rax, 4
    mov qword ptr [r12 + rax], r8
    mov rdx, r9
    sub rdx, r8
    mov qword ptr [r12 + rax + 8], rdx
    inc r10
    lea r8, [r9 + 1]

parse_next:
    inc r9
    jmp parse_loop

parse_remainder:
    cmp r8, rbx
    jae sort_init
    mov rax, r10
    shl rax, 4
    mov qword ptr [r12 + rax], r8
    sub rbx, r8
    mov qword ptr [r12 + rax + 8], rbx
    inc r10

sort_init:
    mov r15, r10
    shl r15, 4
    mov r14d, 16

sort_outer:
    cmp r14, r15
    jae output_init
    mov r8, qword ptr [r12 + r14]
    mov r9, qword ptr [r12 + r14 + 8]
    mov rbx, r14

sort_inner:
    test rbx, rbx
    jz sort_place

    mov rax, qword ptr [r12 + rbx - 16]
    mov rdx, qword ptr [r12 + rbx - 8]
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz sort_prefix

    lea rdi, [r13 + r8]
    lea rsi, [r13 + rax]
    repe cmpsb
    jne sort_different

sort_prefix:
    cmp r9, rdx
    jae sort_place
    jmp sort_shift

sort_different:
    movzx ecx, byte ptr [rdi - 1]
    movzx esi, byte ptr [rsi - 1]
    cmp ecx, esi
    jae sort_place

sort_shift:
    mov qword ptr [r12 + rbx], rax
    mov qword ptr [r12 + rbx + 8], rdx
    sub rbx, 16
    jmp sort_inner

sort_place:
    mov qword ptr [r12 + rbx], r8
    mov qword ptr [r12 + rbx + 8], r9
    add r14, 16
    jmp sort_outer

output_init:
    lea r14, [rip + output_buffer]
    mov rdi, r14
    xor ebx, ebx

output_loop:
    cmp rbx, r15
    jae write_output
    mov rax, qword ptr [r12 + rbx]
    lea rsi, [r13 + rax]
    mov rcx, qword ptr [r12 + rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    jmp output_loop

write_output:
    mov rdx, rdi
    sub rdx, r14
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

    mov eax, 60
    xor edi, edi
    syscall
