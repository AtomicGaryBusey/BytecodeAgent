.section .text
.globl _start

_start:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    xor ebx, ebx
    xor ebp, ebp

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz input_end
    mov r13, rax
    lea r12, [rip + input_buffer]

next_byte:
    movzx eax, byte ptr [r12]
    inc r12
    dec r13
    cmp al, 10
    je line_end

    mov ebp, 1
    test ebx, ebx
    jnz byte_done
    cmp al, 32
    je separator
    cmp al, 9
    je separator
    lea rdx, [rip + token_buffer]
    mov byte ptr [rdx + r14], al
    inc r14
    jmp byte_done

separator:
    call finish_token

byte_done:
    test r13, r13
    jnz next_byte
    jmp read_more

line_end:
    call finish_line
    test r13, r13
    jnz next_byte
    jmp read_more

input_end:
    test ebp, ebp
    jz exit_ok
    call finish_line

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

finish_token:
    test r14, r14
    jz token_return
    mov rcx, r14
    xor r14d, r14d
    lea rsi, [rip + token_buffer]
    cmp rcx, 1
    jne parse_literal

    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

parse_literal:
    xor r8d, r8d
    cmp byte ptr [rsi], '-'
    jne digits_ready
    mov r8d, 1
    inc rsi
    dec rcx
digits_ready:
    test rcx, rcx
    jz token_error
    cmp rcx, 18
    ja token_error
    xor eax, eax

digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja token_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digit_loop
    test r8d, r8d
    jz push_value
    neg rax

push_value:
    lea rdi, [rip + value_stack]
    mov qword ptr [rdi + r15*8], rax
    inc r15
token_return:
    ret

op_add:
    cmp r15, 2
    jb token_error
    dec r15
    lea rdi, [rip + value_stack]
    mov rax, qword ptr [rdi + r15*8 - 8]
    add rax, qword ptr [rdi + r15*8]
    jmp store_result

op_sub:
    cmp r15, 2
    jb token_error
    dec r15
    lea rdi, [rip + value_stack]
    mov rax, qword ptr [rdi + r15*8 - 8]
    sub rax, qword ptr [rdi + r15*8]
    jmp store_result

op_mul:
    cmp r15, 2
    jb token_error
    dec r15
    lea rdi, [rip + value_stack]
    mov rax, qword ptr [rdi + r15*8 - 8]
    imul rax, qword ptr [rdi + r15*8]
    jmp store_result

op_div:
    xor r10d, r10d
    jmp divide_result

op_mod:
    mov r10d, 1

divide_result:
    cmp r15, 2
    jb token_error
    dec r15
    lea rdi, [rip + value_stack]
    mov r9, qword ptr [rdi + r15*8]
    test r9, r9
    jz token_error
    mov rax, qword ptr [rdi + r15*8 - 8]
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne division_safe
    cmp r9, -1
    je token_error
division_safe:
    cqo
    idiv r9
    test r10d, r10d
    jz store_result
    mov rax, rdx

store_result:
    mov qword ptr [rdi + r15*8 - 8], rax
    ret

token_error:
    mov ebx, 1
    ret

finish_line:
    test ebx, ebx
    jnz print_error
    call finish_token
    test ebx, ebx
    jnz print_error
    cmp r15, 1
    jne print_error

    lea rdi, [rip + value_stack]
    mov rax, qword ptr [rdi]
    xor r8d, r8d
    test rax, rax
    jns number_positive
    mov r8d, 1
    neg rax

number_positive:
    lea rsi, [rip + output_end]
    dec rsi
    mov byte ptr [rsi], 10
    mov ecx, 10

number_digit:
    xor edx, edx
    div rcx
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    test rax, rax
    jnz number_digit
    test r8d, r8d
    jz write_number
    dec rsi
    mov byte ptr [rsi], '-'

write_number:
    lea rdx, [rip + output_end]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp reset_line

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall

reset_line:
    xor r14d, r14d
    xor r15d, r15d
    xor ebx, ebx
    xor ebp, ebp
    ret

.section .rodata
error_message:
    .ascii "error\n"

.section .bss
.balign 8
input_buffer:
    .zero 4096
token_buffer:
    .zero 1001
.balign 8
value_stack:
    .zero 8008
output_buffer:
    .zero 32
output_end:
