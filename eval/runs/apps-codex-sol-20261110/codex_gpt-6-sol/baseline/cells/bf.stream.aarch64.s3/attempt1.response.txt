.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, tape
    add x21, x21, :lo12:tape
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x23, jumps
    add x23, x23, :lo12:jumps
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack

    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cbnz x2, read_call
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, find_delimiter
    add x20, x20, x0
    b read_loop

find_delimiter:
    mov x25, x20
    mov x26, x20
    mov x9, #0
delimiter_loop:
    cmp x9, x20
    b.hs validate
    ldrb w10, [x19, x9]
    cmp w10, #33
    b.eq delimiter_found
    add x9, x9, #1
    b delimiter_loop
delimiter_found:
    mov x25, x9
    add x26, x9, #1

validate:
    mov x9, #0
    mov x13, #0
validate_loop:
    cmp x9, x25
    b.hs validate_done
    ldrb w10, [x19, x9]
    cmp w10, #91
    b.eq open_bracket
    cmp w10, #93
    b.eq close_bracket
validate_next:
    add x9, x9, #1
    b validate_loop
open_bracket:
    str w9, [x24, x13, lsl #2]
    add x13, x13, #1
    b validate_next
close_bracket:
    cbz x13, invalid
    sub x13, x13, #1
    ldr w12, [x24, x13, lsl #2]
    add w14, w9, #1
    str w14, [x23, x12, lsl #2]
    add w14, w12, #1
    str w14, [x23, x9, lsl #2]
    b validate_next
validate_done:
    cbnz x13, invalid

    mov x9, #0
    mov x27, x21
    mov x28, #0
execute:
    cmp x9, x25
    b.hs finish
    ldrb w10, [x19, x9]
    cmp w10, #62
    b.eq move_right
    cmp w10, #60
    b.eq move_left
    cmp w10, #43
    b.eq increment
    cmp w10, #45
    b.eq decrement
    cmp w10, #46
    b.eq output
    cmp w10, #44
    b.eq input
    cmp w10, #91
    b.eq loop_open
    cmp w10, #93
    b.eq loop_close
next_command:
    add x9, x9, #1
    b execute

move_right:
    add x27, x27, #1
    b next_command
move_left:
    sub x27, x27, #1
    b next_command
increment:
    ldrb w11, [x27]
    add w11, w11, #1
    strb w11, [x27]
    b next_command
decrement:
    ldrb w11, [x27]
    sub w11, w11, #1
    strb w11, [x27]
    b next_command
output:
    ldrb w11, [x27]
    strb w11, [x22, x28]
    add x28, x28, #1
    b next_command
input:
    mov w11, #0
    cmp x26, x20
    b.hs input_store
    ldrb w11, [x19, x26]
    add x26, x26, #1
input_store:
    strb w11, [x27]
    b next_command
loop_open:
    ldrb w11, [x27]
    cbnz w11, next_command
    ldr w9, [x23, x9, lsl #2]
    b execute
loop_close:
    ldrb w11, [x27]
    cbz w11, next_command
    ldr w9, [x23, x9, lsl #2]
    b execute

finish:
    mov x0, #1
    mov x1, x22
    mov x2, x28
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

invalid:
    mov x0, #1
    mov x8, #93
    svc #0

.section .bss
.balign 16
input_buf:
    .space 65537
.balign 16
tape:
    .space 30000
.balign 16
output_buf:
    .space 65536
.balign 16
jumps:
    .space 262144
.balign 16
bracket_stack:
    .space 4000
