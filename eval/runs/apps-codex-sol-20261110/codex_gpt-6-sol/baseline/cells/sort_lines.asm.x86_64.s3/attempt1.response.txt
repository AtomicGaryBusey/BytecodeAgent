.section .bss
.balign 8
records:
.zero 8000
input_buf:
.zero 32769

.text
.globl _start
_start:
    lea r12, [rip + input_buf]
    xor ebx, ebx

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 32768
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r14, [rip + records]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan_loop:
    cmp r8, rbx
    jae scan_done
    cmp byte ptr [r12 + r8], 10
    jne scan_next
    mov rax, r8
    sub rax, r9
    shl rax, 32
    or rax, r9
    mov qword ptr [r14 + r10*8], rax
    inc r10
    lea r9, [r8 + 1]

scan_next:
    inc r8
    jmp scan_loop

scan_done:
    cmp r9, rbx
    je sort_start
    mov byte ptr [r12 + rbx], 10
    mov rax, rbx
    sub rax, r9
    shl rax, 32
    or rax, r9
    mov qword ptr [r14 + r10*8], rax
    inc r10

sort_start:
    mov r15, r10
    mov ebx, 1

sort_outer:
    cmp rbx, r15
    jae output_start
    mov r12, qword ptr [r14 + rbx*8]
    mov r13, rbx

sort_inner:
    test r13, r13
    jz sort_insert
    mov rdi, r12
    mov rsi, qword ptr [r14 + r13*8 - 8]
    call compare_lines
    test eax, eax
    jns sort_insert
    mov rax, qword ptr [r14 + r13*8 - 8]
    mov qword ptr [r14 + r13*8], rax
    dec r13
    jmp sort_inner

sort_insert:
    mov qword ptr [r14 + r13*8], r12
    inc rbx
    jmp sort_outer

output_start:
    lea r8, [rip + input_buf]
    xor ebx, ebx

output_loop:
    cmp rbx, r15
    jae done
    mov rax, qword ptr [r14 + rbx*8]
    mov rdx, rax
    shr rdx, 32
    inc rdx
    mov esi, eax
    add rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    inc rbx
    jmp output_loop

done:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, rdi
    shr r8, 32
    mov r9, rsi
    shr r9, 32
    mov edi, edi
    mov esi, esi
    lea r10, [rip + input_buf]
    add rdi, r10
    add rsi, r10
    mov rcx, r8
    cmp rcx, r9
    cmova rcx, r9
    test rcx, rcx
    jz compare_lengths

compare_bytes:
    mov al, byte ptr [rdi]
    cmp al, byte ptr [rsi]
    jb compare_less
    ja compare_greater
    inc rdi
    inc rsi
    dec rcx
    jnz compare_bytes

compare_lengths:
    cmp r8, r9
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret
