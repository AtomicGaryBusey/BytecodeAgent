.section .text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // inside a word
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf

read_next:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output_counts

    add x21, x21, x0
    mov x3, x0
    mov x4, x23

scan_byte:
    ldrb w5, [x4], #1
    cmp w5, #10
    b.ne check_space
    add x19, x19, #1

check_space:
    cmp w5, #32
    b.eq whitespace
    sub w6, w5, #9
    cmp w6, #4
    b.ls whitespace

    cbnz x22, scan_next
    add x20, x20, #1
    mov x22, #1
    b scan_next

whitespace:
    mov x22, #0

scan_next:
    subs x3, x3, #1
    b.ne scan_byte
    b read_next

output_counts:
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x1, x24

    mov x0, x19
    bl decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl decimal
    mov w2, #10
    strb w2, [x1], #1

    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// Append the decimal representation of x0 at x1; return the next byte in x1.
decimal:
    mov x2, x1
    mov x4, #10
1:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    strb w5, [x1], #1
    mov x0, x3
    cbnz x0, 1b

    sub x6, x1, #1
2:
    cmp x2, x6
    b.hs 3f
    ldrb w3, [x2]
    ldrb w5, [x6]
    strb w5, [x2], #1
    strb w3, [x6]
    sub x6, x6, #1
    b 2b
3:
    ret

.section .bss
.balign 16
input_buf:
    .zero 4096
output_buf:
    .zero 64
