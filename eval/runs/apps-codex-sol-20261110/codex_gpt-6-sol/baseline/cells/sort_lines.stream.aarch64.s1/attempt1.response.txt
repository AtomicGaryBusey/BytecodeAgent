.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x20, line_records
    add x20, x20, :lo12:line_records
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    add x1, x19, x22
    mov x2, #32769
    sub x2, x2, x22
    mov x8, #63
    svc #0
    cbz x0, parse_input
    add x22, x22, x0
    b read_loop

parse_input:
    mov x9, #0
    mov x10, #0

parse_loop:
    cmp x9, x22
    b.hs parse_remainder
    ldrb w11, [x19, x9]
    cmp w11, #10
    b.ne parse_next
    add x12, x20, x21, lsl #4
    sub x13, x9, x10
    stp x10, x13, [x12]
    add x21, x21, #1
    add x10, x9, #1

parse_next:
    add x9, x9, #1
    b parse_loop

parse_remainder:
    cmp x10, x22
    b.hs sort_begin
    add x12, x20, x21, lsl #4
    sub x13, x22, x10
    stp x10, x13, [x12]
    add x21, x21, #1

sort_begin:
    mov x11, #1

sort_next:
    cmp x11, x21
    b.hs output_begin
    add x13, x20, x11, lsl #4
    ldp x23, x24, [x13]
    mov x12, x11

sort_compare:
    cbz x12, sort_insert
    sub x14, x12, #1
    add x13, x20, x14, lsl #4
    ldp x26, x27, [x13]
    add x14, x19, x26
    add x15, x19, x23
    mov x16, #0

compare_bytes:
    cmp x16, x27
    b.hs compare_lengths
    cmp x16, x24
    b.hs sort_shift
    ldrb w17, [x14, x16]
    ldrb w18, [x15, x16]
    cmp w17, w18
    b.lo sort_insert
    b.hi sort_shift
    add x16, x16, #1
    b compare_bytes

compare_lengths:
    cmp x27, x24
    b.ls sort_insert

sort_shift:
    stp x26, x27, [x13, #16]
    sub x12, x12, #1
    b sort_compare

sort_insert:
    add x13, x20, x12, lsl #4
    stp x23, x24, [x13]
    add x11, x11, #1
    b sort_next

output_begin:
    mov x11, #0

output_loop:
    cmp x11, x21
    b.hs finish
    add x13, x20, x11, lsl #4
    ldp x23, x24, [x13]
    cbz x24, write_newline
    mov x0, #1
    add x1, x19, x23
    mov x2, x24
    mov x8, #64
    svc #0

write_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x11, x11, #1
    b output_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata

    .byte 10

.bss

input_buffer:
