.globl _start

.section .bss
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537

.section .text
_start:
    lea r12, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    je read_done
    add r13, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    xor r8d, r8d
    xor r15d, r15d

scan_loop:
    cmp r8, r13
    je scan_done
    cmp byte ptr [r12 + r8], 10
    je line_end
    inc r8
    jmp scan_loop

line_end:
    call emit_line
    inc r8
    mov rbx, r8
    jmp scan_loop

scan_done:
    cmp rbx, r13
    je write_output
    mov r8, r13
    call emit_line

write_output:
    test r15, r15
    je exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    mov r9, r8
reverse_loop:
    cmp r9, rbx
    jbe append_newline
    dec r9
    mov al, byte ptr [r12 + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_loop

append_newline:
    mov byte ptr [r14 + r15], 10
    inc r15
    ret
