.section .bss
.balign 16
input_buffer:
.zero 65537
.balign 8
value_stack:
.zero 8008
output_buffer:
.zero 32

.section .rodata
error_message:
.ascii "error\n"

.text
.globl _start
_start:
    xor ebx, ebx

read_input:
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 65536
    sub edx, ebx
    jnz read_call
    mov edx, 1

read_call:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz input_done
    add rbx, rax
    jmp read_input

input_done:
    lea r12, [rip + input_buffer]
    lea r13, [r12 + rbx]
    mov r14, r12

scan_lines:
    cmp r12, r13
    je final_line
    cmp byte ptr [r12], 10
    jne scan_next
    mov rdi, r14
    mov rsi, r12
    call process_line
    inc r12
    mov r14, r12
    jmp scan_lines

scan_next:
    inc r12
    jmp scan_lines

final_line:
    cmp r14, r13
    je finish
    mov rdi, r14
    mov rsi, r13
    call process_line

finish:
    mov eax, 60
    xor edi, edi
    syscall

process_line:
    xor ebx, ebx
    mov r8, rdi
    mov r9, rsi

next_token:
    cmp r8, r9
    je line_complete
    mov al, byte ptr [r8]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov r10, r8
find_token_end:
    cmp r10, r9
    je token_found
    mov al, byte ptr [r10]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r10
    jmp find_token_end

token_found:
    mov rdx, r10
    sub rdx, r8
    cmp rdx, 1
    jne parse_literal
    mov r11b, byte ptr [r8]
    cmp r11b, '+'
    je apply_operator
    cmp r11b, '-'
    je apply_operator
    cmp r11b, '*'
    je apply_operator
    cmp r11b, '/'
    je apply_operator
    cmp r11b, '%'
    je apply_operator

parse_literal:
    mov rdi, r8
    xor esi, esi
    cmp byte ptr [rdi], '-'
    jne literal_digits
    mov esi, 1
    inc rdi
    dec rdx

literal_digits:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error
    mov rcx, rdx
    xor eax, eax

digit_loop:
    movzx edx, byte ptr [rdi]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rdi
    dec rcx
    jnz digit_loop

    test esi, esi
    jz push_literal
    neg rax

push_literal:
    lea rdi, [rip + value_stack]
    mov qword ptr [rdi + rbx*8], rax
    inc rbx
    mov r8, r10
    jmp next_token

apply_operator:
    cmp rbx, 2
    jb line_error
    lea rdi, [rip + value_stack]
    dec rbx
    mov rcx, qword ptr [rdi + rbx*8]
    dec rbx
    mov rax, qword ptr [rdi + rbx*8]

    cmp r11b, '+'
    je operation_add
    cmp r11b, '-'
    je operation_subtract
    cmp r11b, '*'
    je operation_multiply

    test rcx, rcx
    jz line_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne operation_divide
    cmp rcx, -1
    je line_error

operation_divide:
    cqo
    idiv rcx
    cmp r11b, '%'
    jne store_result
    mov rax, rdx
    jmp store_result

operation_add:
    add rax, rcx
    jmp store_result

operation_subtract:
    sub rax, rcx
    jmp store_result

operation_multiply:
    imul rax, rcx

store_result:
    mov qword ptr [rdi + rbx*8], rax
    inc rbx
    mov r8, r10
    jmp next_token

skip_separator:
    inc r8
    jmp next_token

line_complete:
    cmp rbx, 1
    jne line_error
    mov rax, qword ptr [rip + value_stack]
    lea rsi, [rip + output_buffer + 31]
    mov byte ptr [rsi], 10
    mov ecx, 1
    xor r9d, r9d
    test rax, rax
    jns format_digits
    mov r9d, 1
    neg rax

format_digits:
    mov r8d, 10

format_digit:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz format_digit

    test r9d, r9d
    jz write_result
    dec rsi
    mov byte ptr [rsi], '-'
    inc ecx

write_result:
    mov edx, ecx
    mov edi, 1
    mov eax, 1
    syscall
    ret

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall
    ret
