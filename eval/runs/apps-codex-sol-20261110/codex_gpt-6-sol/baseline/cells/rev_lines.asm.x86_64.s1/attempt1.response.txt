.section .bss
buffer:
    .zero 65537
output:
    .zero 65537

.section .text
.globl _start
_start:
    lea r8, [rip + buffer]
    lea r15, [rip + output]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d

scan_loop:
    cmp rbx, r12
    je scan_done
    cmp byte ptr [r8 + rbx], 10
    jne next_byte
    call emit_line
    lea r13, [rbx + 1]

next_byte:
    inc rbx
    jmp scan_loop

scan_done:
    cmp r13, r12
    je write_output
    call emit_line

write_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r14
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    mov r10, rbx

reverse_loop:
    cmp r10, r13
    je line_done
    dec r10
    mov al, byte ptr [r8 + r10]
    mov byte ptr [r15 + r14], al
    inc r14
    jmp reverse_loop

line_done:
    mov byte ptr [r15 + r14], 10
    inc r14
    ret
