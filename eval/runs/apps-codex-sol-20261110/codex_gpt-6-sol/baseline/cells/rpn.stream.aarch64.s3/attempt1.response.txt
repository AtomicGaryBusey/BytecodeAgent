.section .bss
.balign 16
input_buf:
.space 65537
.balign 16
value_stack:
.space 8016
output_buf:
.space 32

.section .rodata
error_msg:
.ascii "error\n"
error_len = . - error_msg

.section .text
.globl _start
_start:
adrp x19, input_buf
add x19, x19, :lo12:input_buf
mov x20, #0

read_more:

add x1, x19, x20
movz x2, #1
movk x2, #1, lsl #16
sub x2, x2, x20
mov x8, #63

cbz x0, read_done

read_done:
add x20, x19, x20
adrp x22, value_stack
add x22, x22, :lo12:value_stack
adrp x27, output_buf
add x27, x27, :lo12:output_buf

next_line:
cmp x19, x20
b.eq exit_program
mov x21, x19

find_line_end:
cmp x21, x20
b.eq line_found
ldrb w9, [x21]
cmp w9, #10
b.eq line_found
add x21, x21, #1
b find_line_end

line_found:
mov x23, #0
mov x24, x19

next_token:
cmp x24, x21
b.eq finish_line
ldrb w9, [x24]
cmp w9, #32
b.eq skip_separator
cmp w9, #9
b.eq skip_separator
mov x25, x24

find_token_end:
cmp x25, x21
b.eq token_found
ldrb w9, [x25]
cmp w9, #32
b.eq token_found
cmp w9, #9
b.eq token_found
add x25, x25, #1
b find_token_end

skip_separator:
add x24, x24, #1
b next_token

token_found:
sub x10, x25, x24
cmp x10, #1
b.ne parse_literal
ldrb w9, [x24]
cmp w9, #43
b.eq apply_operator
cmp w9, #45
b.eq apply_operator
cmp w9, #42
b.eq apply_operator
cmp w9, #47
b.eq apply_operator
cmp w9, #37
b.eq apply_operator

parse_literal:
mov x9, #0
mov x11, x24
ldrb w14, [x11]
cmp w14, #45
b.ne check_literal_length
mov x9, #1
add x11, x11, #1
sub x10, x10, #1

check_literal_length:
cbz x10, line_error
cmp x10, #18
b.hi line_error
mov x12, #0
mov x13, #10

literal_digit:
ldrb w14, [x11], #1
sub w14, w14, #48
cmp w14, #9
b.hi line_error
madd x12, x12, x13, x14
cmp x11, x25
b.ne literal_digit
cbz x9, push_literal
neg x12, x12

push_literal:
str x12, [x22, x23, lsl #3]
add x23, x23, #1
mov x24, x25
b next_token

apply_operator:
cmp x23, #2
b.lo line_error
sub x23, x23, #1
ldr x10, [x22, x23, lsl #3]
sub x23, x23, #1
ldr x11, [x22, x23, lsl #3]
cmp w9, #43
b.eq operator_add
cmp w9, #45
b.eq operator_sub
cmp w9, #42
b.eq operator_mul

cbz x10, line_error
cmn x10, #1
b.ne operator_divide
mov x13, #1
lsl x13, x13, #63
cmp x11, x13
b.eq line_error

operator_divide:

cmp w9, #37
b.ne push_result
msub x12, x12, x10, x11
b push_result

operator_add:
add x12, x11, x10
b push_result

operator_sub:
sub x12, x11, x10
b push_result

operator_mul:

push_result:
str x12, [x22, x23, lsl #3]
add x23, x23, #1
mov x24, x25
b next_token

finish_line:

b.ne line_error
ldr x9, [x22]
cmp x9, #0
cset w15, lt
b.ge magnitude_ready
neg x9, x9

mov w14, #10
strb w14, [x28, #-1]!

msub x13, x11, x10, x9
add w14, w13, #48
strb w14, [x28, #-1]!
mov x9, x11
cbnz x9, format_digit
cbz w15, write_result
mov w14, #45
strb w14, [x28, #-1]!

write_result:

mov x1, x28
add x2, x27, #32
sub x2, x2, x28
mov x8, #64

b advance_line

line_error:

adrp x1, error_msg
add x1, x1, :lo12:error_msg
mov x2, #error_len
mov x8, #64

advance_line:

cmp x19, x20
b.eq next_line
add x19, x19, #1
b next_line

exit_program:
mov x0, #0
mov x8, #93
svc #0
