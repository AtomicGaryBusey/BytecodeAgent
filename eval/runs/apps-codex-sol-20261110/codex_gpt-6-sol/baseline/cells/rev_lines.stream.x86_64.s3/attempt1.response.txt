.globl _start

.section .text
_start:
    lea r12, [rip + input_buffer]
    lea r13, [rip + output_buffer]
    xor ebx, ebx

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan_loop:
    cmp r10, rbx
    jae final_line
    cmp byte ptr [r12 + r10], 10
    je complete_line
    inc r10
    jmp scan_loop

complete_line:
    mov r11, r10
reverse_line:
    cmp r11, r8
    je line_newline
    dec r11
    mov al, byte ptr [r12 + r11]
    mov byte ptr [r13 + r9], al
    inc r9
    jmp reverse_line

line_newline:
    mov byte ptr [r13 + r9], 10
    inc r9
    inc r10
    mov r8, r10
    jmp scan_loop

final_line:
    cmp r8, rbx
    je write_output
    mov r11, rbx
reverse_final:
    cmp r11, r8
    je final_newline
    dec r11
    mov al, byte ptr [r12 + r11]
    mov byte ptr [r13 + r9], al
    inc r9
    jmp reverse_final

final_newline:
    mov byte ptr [r13 + r9], 10
    inc r9

write_output:
    xor ebx, ebx
write_loop:
    cmp rbx, r9
    jae exit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [r13 + rbx]
    mov rdx, r9
    sub rdx, rbx
    syscall
    add rbx, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537
