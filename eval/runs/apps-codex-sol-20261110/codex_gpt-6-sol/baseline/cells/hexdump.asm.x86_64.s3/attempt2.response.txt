.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
.zero 65536
line_buffer:
.zero 80
extra_byte:
.zero 1

.text
.globl _start
_start:
    lea r15, [rip + input_buffer]
    lea r14, [rip + line_buffer]
    lea r13, [rip + hex_digits]
    xor ebx, ebx

read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_extra
    lea rsi, [r15 + rbx]
    jmp do_read

read_extra:
    lea rsi, [rip + extra_byte]
    mov edx, 1

do_read:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r9d, ebx
    xor r8d, r8d

row_loop:
    cmp r8d, r9d
    jae final_offset

    mov ebx, r9d
    sub ebx, r8d
    cmp ebx, 16
    jbe row_size_ready
    mov ebx, 16

row_size_ready:
    lea rdi, [r14 + 8]
    mov ecx, 52
    mov al, ' '
    rep stosb
    mov byte ptr [r14 + 60], '|'

    mov rdi, r14
    call format_offset
    lea rdi, [r14 + 10]
    xor r10d, r10d

byte_loop:
    cmp r10d, ebx
    jae row_complete

    lea rsi, [r15 + r8]
    movzx edx, byte ptr [rsi + r10]

    mov eax, edx
    shr eax, 4
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi], al

    mov eax, edx
    and eax, 15
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, edx
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe ascii_ready

nonprintable:
    mov al, '.'

ascii_ready:
    mov byte ptr [r14 + r10 + 61], al
    add rdi, 3
    inc r10d
    cmp r10d, 8
    jne byte_loop
    inc rdi
    jmp byte_loop

row_complete:
    mov byte ptr [r14 + rbx + 61], '|'
    mov byte ptr [r14 + rbx + 62], 10

    mov eax, 1
    mov edi, 1
    mov rsi, r14
    lea edx, [rbx + 63]
    syscall

    add r8d, ebx
    jmp row_loop

final_offset:
    mov rdi, r14
    call format_offset
    mov byte ptr [r14 + 8], 10

    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

format_offset:
    mov r10d, r8d
    mov ecx, 8

format_digit:
    mov edx, r10d
    shr edx, 28
    mov al, byte ptr [r13 + rdx]
    mov byte ptr [rdi], al
    inc rdi
    shl r10d, 4
    dec ecx
    jne format_digit
    ret
