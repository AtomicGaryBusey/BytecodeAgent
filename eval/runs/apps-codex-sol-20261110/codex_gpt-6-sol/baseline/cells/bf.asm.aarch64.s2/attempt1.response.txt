.section .text
.globl _start
_start:
    adrp x19, source
    add x19, x19, :lo12:source
    adrp x25, matches
    add x25, x25, :lo12:matches
    adrp x26, tape
    add x26, x26, :lo12:tape
    adrp x27, output
    add x27, x27, :lo12:output
    adrp x29, bracket_stack
    add x29, x29, :lo12:bracket_stack
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x20
    mov x22, x20
    mov x9, #0

find_separator:
    cmp x9, x20
    b.hs validate_start
    ldrb w10, [x19, x9]
    cmp w10, #33
    b.eq separator_found
    add x9, x9, #1
    b find_separator

separator_found:
    mov x21, x9
    add x22, x9, #1

validate_start:
    mov x9, #0
    mov w10, #0

validate_loop:
    cmp x9, x21
    b.hs validate_done
    ldrb w11, [x19, x9]
    cmp w11, #91
    b.eq validate_open
    cmp w11, #93
    b.eq validate_close
validate_next:
    add x9, x9, #1
    b validate_loop

validate_open:
    str w9, [x29, w10, uxtw #2]
    add w10, w10, #1
    b validate_next

validate_close:
    cbz w10, invalid_program
    sub w10, w10, #1
    ldr w11, [x29, w10, uxtw #2]
    str w9, [x25, w11, uxtw #2]
    str w11, [x25, x9, lsl #2]
    b validate_next

validate_done:
    cbnz w10, invalid_program
    mov x23, #0
    mov x24, #0
    mov x28, #0

execute_loop:
    cmp x23, x21
    b.hs program_done
    ldrb w9, [x19, x23]
    cmp w9, #62
    b.eq command_right
    cmp w9, #60
    b.eq command_left
    cmp w9, #43
    b.eq command_increment
    cmp w9, #45
    b.eq command_decrement
    cmp w9, #46
    b.eq command_output
    cmp w9, #44
    b.eq command_input
    cmp w9, #91
    b.eq command_open
    cmp w9, #93
    b.eq command_close

execute_next:
    add x23, x23, #1
    b execute_loop

command_right:
    add x24, x24, #1
    b execute_next

command_left:
    sub x24, x24, #1
    b execute_next

command_increment:
    ldrb w10, [x26, x24]
    add w10, w10, #1
    strb w10, [x26, x24]
    b execute_next

command_decrement:
    ldrb w10, [x26, x24]
    sub w10, w10, #1
    strb w10, [x26, x24]
    b execute_next

command_output:
    ldrb w10, [x26, x24]
    strb w10, [x27, x28]
    add x28, x28, #1
    b execute_next

command_input:
    mov w10, #0
    cmp x22, x20
    b.hs input_store
    ldrb w10, [x19, x22]
    add x22, x22, #1
input_store:
    strb w10, [x26, x24]
    b execute_next

command_open:
    ldrb w10, [x26, x24]
    cbnz w10, execute_next
    ldr w23, [x25, x23, lsl #2]
    b execute_next

command_close:
    ldrb w10, [x26, x24]
    cbz w10, execute_next
    ldr w23, [x25, x23, lsl #2]
    b execute_next

program_done:
    cbz x28, successful_exit
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0

successful_exit:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.section .bss
.balign 16
source:
    .zero 65536
matches:
    .zero 262144
tape:
    .zero 30000
output:
    .zero 65536
bracket_stack:
    .zero 4000
