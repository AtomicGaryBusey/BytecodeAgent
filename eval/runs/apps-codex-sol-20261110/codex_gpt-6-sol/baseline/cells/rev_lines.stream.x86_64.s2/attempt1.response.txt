.globl _start

.section .bss
input:
    .zero 65537
output:
    .zero 65537

.text
_start:
    lea r8, [rip + input]
    lea r9, [rip + output]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz process
    add r12, rax
    jmp read_loop

process:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

scan:
    cmp r15, r12
    je finish_scan
    cmp byte ptr [r8 + r15], 10
    jne next_byte
    mov rbx, r15
    call emit_line
    inc r15
    mov r14, r15
    jmp scan

next_byte:
    inc r15
    jmp scan

finish_scan:
    cmp r14, r12
    je write_output
    mov rbx, r12
    call emit_line

write_output:
    test r13, r13
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r13
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    cmp rbx, r14
    je emit_newline
    dec rbx
    mov al, byte ptr [r8 + rbx]
    mov byte ptr [r9 + r13], al
    inc r13
    jmp emit_line

emit_newline:
    mov byte ptr [r9 + r13], 10
    inc r13
    ret
