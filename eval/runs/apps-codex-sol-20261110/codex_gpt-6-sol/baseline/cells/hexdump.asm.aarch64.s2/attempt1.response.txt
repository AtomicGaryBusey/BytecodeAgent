.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65537
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, #0
    adrp x24, hex_digits
    add x24, x24, :lo12:hex_digits

row_loop:
    cmp x23, x20
    b.hs final_offset

    sub x25, x20, x23
    cmp x25, #16
    mov x9, #16
    csel x25, x25, x9, lo
    add x26, x19, x23

    mov w0, w23
    mov x1, x22
    bl emit_hex8
    mov x22, x1

    mov w9, #32
    strb w9, [x22], #1
    strb w9, [x22], #1

    mov x10, #0
hex_slots:
    cmp x10, x25
    b.hs empty_slot

    ldrb w11, [x26, x10]
    lsr w12, w11, #4
    ldrb w12, [x24, x12]
    strb w12, [x22], #1
    and w12, w11, #15
    ldrb w12, [x24, x12]
    strb w12, [x22], #1
    mov w9, #32
    strb w9, [x22], #1
    b slot_done

empty_slot:
    mov w9, #32
    strb w9, [x22], #1
    strb w9, [x22], #1
    strb w9, [x22], #1

slot_done:
    cmp x10, #7
    b.ne no_group_space
    mov w9, #32
    strb w9, [x22], #1
no_group_space:
    add x10, x10, #1
    cmp x10, #16
    b.lo hex_slots

    mov w9, #32
    strb w9, [x22], #1
    mov w9, #124
    strb w9, [x22], #1

    mov x10, #0
ascii_loop:
    cmp x10, x25
    b.hs ascii_done
    ldrb w11, [x26, x10]
    cmp w11, #32
    b.lo ascii_dot
    cmp w11, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w11, #46
ascii_store:
    strb w11, [x22], #1
    add x10, x10, #1
    b ascii_loop

ascii_done:
    mov w9, #124
    strb w9, [x22], #1
    mov w9, #10
    strb w9, [x22], #1
    add x23, x23, x25
    b row_loop

final_offset:
    mov w0, w20
    mov x1, x22
    bl emit_hex8
    mov x22, x1
    mov w9, #10
    strb w9, [x22], #1

    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

emit_hex8:
    mov w2, #28
1:
    lsr w3, w0, w2
    and w3, w3, #15
    ldrb w3, [x24, x3]
    strb w3, [x1], #1
    subs w2, w2, #4
    b.ge 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buf:
    .space 65537
.balign 16
output_buf:
    .space 330000
