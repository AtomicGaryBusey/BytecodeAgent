.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 53248
output_buffer:
    .zero 67000

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, alphabet
    add x23, x23, :lo12:alphabet
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x21, x24
    mov x22, #0
    mov x25, x20

encode_loop:
    cmp x25, #3
    b.lo encode_tail

    ldrb w3, [x19], #1
    ldrb w5, [x19], #1
    ldrb w6, [x19], #1

    lsr w4, w3, #2
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    and w4, w3, #3
    lsl w4, w4, #4
    orr w4, w4, w5, lsr #4
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    and w4, w5, #15
    lsl w4, w4, #2
    orr w4, w4, w6, lsr #6
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    and w4, w6, #63
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    sub x25, x25, #3
    add x22, x22, #4
    cmp x22, #76
    b.ne encode_loop
    mov w4, #10
    strb w4, [x21], #1
    mov x22, #0
    b encode_loop

encode_tail:
    cbz x25, finish_lines

    ldrb w3, [x19], #1
    mov w5, #0
    cmp x25, #2
    b.ne tail_bytes_ready
    ldrb w5, [x19]

tail_bytes_ready:
    lsr w4, w3, #2
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    and w4, w3, #3
    lsl w4, w4, #4
    orr w4, w4, w5, lsr #4
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1

    cmp x25, #2
    b.ne one_byte_tail
    and w4, w5, #15
    lsl w4, w4, #2
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x21], #1
    b tail_padding

one_byte_tail:
    mov w4, #61
    strb w4, [x21], #1

tail_padding:
    mov w4, #61
    strb w4, [x21], #1
    add x22, x22, #4

finish_lines:
    cbz x22, write_output
    mov w4, #10
    strb w4, [x21], #1

write_output:
    mov x0, #1
    mov x1, x24
    sub x2, x21, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
