.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49153
input_end:
output:
.zero 66560

.text
.globl _start
_start:
    lea rbx, [rip + input]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    lea rdx, [rip + input_end]
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r12, rbx
    lea rbx, [rip + input]
    lea rdi, [rip + output]
    lea r15, [rip + output]
    lea r13, [rip + alphabet]
    xor r14d, r14d

encode_loop:
    mov rax, r12
    sub rax, rbx
    cmp rax, 3
    jb encode_tail

    movzx r9d, byte ptr [rbx]
    shl r9d, 16
    movzx eax, byte ptr [rbx + 1]
    shl eax, 8
    or r9d, eax
    movzx eax, byte ptr [rbx + 2]
    or r9d, eax
    add rbx, 3
    mov r10d, 3
    jmp emit_group

encode_tail:
    test rax, rax
    jz write_output
    mov r10d, eax
    movzx r9d, byte ptr [rbx]
    shl r9d, 16
    cmp r10d, 2
    jb emit_group
    movzx eax, byte ptr [rbx + 1]
    shl eax, 8
    or r9d, eax

emit_group:
    mov eax, r9d
    shr eax, 18
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi], al

    mov eax, r9d
    shr eax, 12
    and eax, 63
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, r9d
    shr eax, 6
    and eax, 63
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi + 2], al

    mov eax, r9d
    and eax, 63
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi + 3], al

    cmp r10d, 3
    je group_done
    mov byte ptr [rdi + 3], '='
    cmp r10d, 2
    je group_done
    mov byte ptr [rdi + 2], '='

group_done:
    add rdi, 4
    inc r14d
    cmp r14d, 19
    jne check_tail
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d

check_tail:
    cmp r10d, 3
    je encode_loop
    jmp finish_lines

write_output:
    test r14d, r14d
    jz flush
finish_lines:
    test r14d, r14d
    jz flush
    mov byte ptr [rdi], 10
    inc rdi

flush:
    mov rdx, rdi
    sub rdx, r15
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
