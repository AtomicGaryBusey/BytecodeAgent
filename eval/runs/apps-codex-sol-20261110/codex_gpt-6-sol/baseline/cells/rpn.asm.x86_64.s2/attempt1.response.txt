.section .bss
.balign 8
input:
.zero 65536
values:
.zero 8008
outbuf:
.zero 32
scratch:
.zero 1

.section .rodata
error_msg:
.ascii "error\n"

.text
.globl _start
_start:
    lea rbx, [rip + input]
    xor r12d, r12d

read_loop:
    mov edx, 65536
    sub rdx, r12
    jz read_full
    lea rsi, [rbx + r12]
    jmp do_read

read_full:
    lea rsi, [rip + scratch]
    mov edx, 1

do_read:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rbx + r12]
    mov r12, rbx
    lea rbx, [rip + values]

next_line:
    cmp r12, r13
    jae finish
    mov r15, r12

find_line_end:
    cmp r15, r13
    je line_ready
    cmp byte ptr [r15], 10
    je line_ready
    inc r15
    jmp find_line_end

line_ready:
    mov r14, r12
    xor r8d, r8d

next_token:
    cmp r14, r15
    jae line_complete
    mov al, byte ptr [r14]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov r9, r14
    mov r10, r14

find_token_end:
    cmp r10, r15
    jae token_ready
    mov al, byte ptr [r10]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc r10
    jmp find_token_end

token_ready:
    mov rdx, r10
    sub rdx, r9
    cmp rdx, 1
    jne parse_literal
    movzx edi, byte ptr [r9]
    cmp edi, '+'
    je apply_operator
    cmp edi, '-'
    je apply_operator
    cmp edi, '*'
    je apply_operator
    cmp edi, '/'
    je apply_operator
    cmp edi, '%'
    je apply_operator

parse_literal:
    mov rsi, r9
    xor edi, edi
    cmp byte ptr [rsi], '-'
    jne check_digits
    inc rsi
    mov edi, 1

check_digits:
    mov rcx, r10
    sub rcx, rsi
    cmp rcx, 1
    jb line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax

digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, r10
    jb digit_loop
    test edi, edi
    jz push_value
    neg rax

push_value:
    mov qword ptr [rbx + r8*8], rax
    inc r8
    mov r14, r10
    jmp next_token

apply_operator:
    cmp r8, 2
    jb line_error
    dec r8
    mov rdx, qword ptr [rbx + r8*8]
    dec r8
    mov rax, qword ptr [rbx + r8*8]
    cmp edi, '+'
    je op_add
    cmp edi, '-'
    je op_sub
    cmp edi, '*'
    je op_mul
    test rdx, rdx
    jz line_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne op_divide
    cmp rdx, -1
    je line_error

op_divide:
    mov rsi, rdx
    cqo
    idiv rsi
    cmp edi, '%'
    jne push_value
    mov rax, rdx
    jmp push_value

op_add:
    add rax, rdx
    jmp push_value

op_sub:
    sub rax, rdx
    jmp push_value

op_mul:
    imul rax, rdx
    jmp push_value

skip_separator:
    inc r14
    jmp next_token

line_complete:
    cmp r8, 1
    jne line_error
    mov rax, qword ptr [rbx]
    xor edi, edi
    test rax, rax
    jns convert_number
    mov edi, 1
    neg rax

convert_number:
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ecx, 10

convert_digit:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne convert_digit
    test edi, edi
    jz write_number
    dec rsi
    mov byte ptr [rsi], '-'

write_number:
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp advance_line

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall

advance_line:
    mov r12, r15
    cmp r12, r13
    je finish
    inc r12
    jmp next_line

finish:
    mov eax, 60
    xor edi, edi
    syscall
