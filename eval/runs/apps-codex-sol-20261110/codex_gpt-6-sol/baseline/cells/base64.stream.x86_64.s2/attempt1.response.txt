.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
inputbuf:
.zero 49153
outputbuf:
.zero 66500

.text
_start:
    xor ebx, ebx

read_loop:
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    mov edx, 49153
    sub rdx, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_success

    lea r8, [rip + inputbuf]
    lea r9, [r8 + rbx]
    lea r10, [rip + outputbuf]
    lea r13, [rip + alphabet]
    xor r12d, r12d

encode_loop:
    mov rax, r9
    sub rax, r8
    cmp rax, 3
    jb encode_tail

    movzx r14d, byte ptr [r8]
    movzx r15d, byte ptr [r8 + 1]
    movzx ebp, byte ptr [r8 + 2]

    mov eax, r14d
    shr eax, 2
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10], al

    mov eax, r14d
    and eax, 3
    shl eax, 4
    mov ecx, r15d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], al

    mov eax, r15d
    and eax, 15
    shl eax, 2
    mov ecx, ebp
    shr ecx, 6
    or eax, ecx
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 2], al

    mov eax, ebp
    and eax, 63
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 3], al

    add r8, 3
    add r10, 4
    add r12d, 4
    cmp r12d, 76
    jne encode_loop
    mov byte ptr [r10], 10
    inc r10
    xor r12d, r12d
    jmp encode_loop

encode_tail:
    test rax, rax
    jz encode_done

    movzx r14d, byte ptr [r8]
    mov eax, r14d
    shr eax, 2
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10], al

    mov eax, r14d
    and eax, 3
    shl eax, 4
    cmp r8, r9
    je encode_done
    mov rcx, r9
    sub rcx, r8
    cmp rcx, 2
    jb one_byte

    movzx r15d, byte ptr [r8 + 1]
    mov ecx, r15d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], al

    mov eax, r15d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 2], al
    mov byte ptr [r10 + 3], '='
    jmp tail_newline

one_byte:
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], al
    mov byte ptr [r10 + 2], '='
    mov byte ptr [r10 + 3], '='

tail_newline:
    mov byte ptr [r10 + 4], 10
    add r10, 5
    jmp write_output

encode_done:
    test r12d, r12d
    jz write_output
    mov byte ptr [r10], 10
    inc r10

write_output:
    lea rsi, [rip + outputbuf]
    mov rdx, r10
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
