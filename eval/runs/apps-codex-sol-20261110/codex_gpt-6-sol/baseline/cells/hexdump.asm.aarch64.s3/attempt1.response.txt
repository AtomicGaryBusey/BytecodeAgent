.text
.globl _start

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits
    mov x19, #0
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x21, x20
    mov x2, #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x20, x20, x0
    cmp x20, #16
    b.ne read_loop

    bl output_row
    add x19, x19, #16
    mov x20, #0
    b read_loop

end_input:
    cbz x20, final_offset
    bl output_row
    add x19, x19, x20

final_offset:
    mov x1, x22
    mov w0, w19
    bl hex8
    mov w4, #10
    strb w4, [x1]
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

output_row:
    stp x29, x30, [sp, #-16]!
    mov x1, x22
    mov w0, w19
    bl hex8
    mov x24, x1

    mov w6, #32
    mov w7, #124
    strb w6, [x24], #1
    strb w6, [x24], #1
    mov x27, #0

hex_slots:
    cmp x27, x20
    b.hs empty_slot

    ldrb w4, [x21, x27]
    lsr w5, w4, #4
    ldrb w5, [x23, w5, uxtw]
    strb w5, [x24], #1
    and w5, w4, #15
    ldrb w5, [x23, w5, uxtw]
    strb w5, [x24], #1
    strb w6, [x24], #1
    b slot_done

empty_slot:
    strb w6, [x24], #1
    strb w6, [x24], #1
    strb w6, [x24], #1

slot_done:
    add x27, x27, #1
    cmp x27, #8
    b.ne no_middle_space
    strb w6, [x24], #1

no_middle_space:
    cmp x27, #16
    b.ne hex_slots

    strb w6, [x24], #1
    strb w7, [x24], #1
    mov x27, #0

ascii_bytes:
    cmp x27, x20
    b.hs ascii_done
    ldrb w4, [x21, x27]
    cmp w4, #32
    b.lo ascii_dot
    cmp w4, #126
    b.ls ascii_store

ascii_dot:
    mov w4, #46

ascii_store:
    strb w4, [x24], #1
    add x27, x27, #1
    b ascii_bytes

ascii_done:
    strb w7, [x24], #1
    mov w4, #10
    strb w4, [x24], #1
    mov x0, #1
    mov x1, x22
    sub x2, x24, x22
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret

hex8:
    mov w3, #28

hex8_loop:
    lsr w4, w0, w3
    and w4, w4, #15
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x1], #1
    subs w3, w3, #4
    b.pl hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 16
output_buf:
    .space 80
