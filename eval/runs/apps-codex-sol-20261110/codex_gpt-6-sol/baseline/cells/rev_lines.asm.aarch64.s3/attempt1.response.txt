.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0

line_loop:
    cmp x22, x20
    b.hs write_output
    mov x24, x22
    mov x25, x22

find_end:
    cmp x25, x20
    b.hs reverse_line
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.eq reverse_line
    add x25, x25, #1
    b find_end

reverse_line:
    mov x26, x25

reverse_byte:
    cmp x26, x24
    b.ls line_newline
    sub x26, x26, #1
    ldrb w0, [x19, x26]
    strb w0, [x21, x23]
    add x23, x23, #1
    b reverse_byte

line_newline:
    mov w0, #10
    strb w0, [x21, x23]
    add x23, x23, #1
    cmp x25, x20
    b.eq line_at_eof
    add x22, x25, #1
    b line_loop

line_at_eof:
    mov x22, x25
    b line_loop

write_output:
    cbz x23, exit_success
    mov x0, #1
    mov x1, x21
    mov x2, x23
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero 65537
output_buf:
    .zero 65537
