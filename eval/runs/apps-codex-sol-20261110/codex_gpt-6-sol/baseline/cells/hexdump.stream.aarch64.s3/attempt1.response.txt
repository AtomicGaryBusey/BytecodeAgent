.text
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits

    mov x19, #0
    mov x20, #0
    mov x28, #0

read_loop:
    mov x0, #0
    add x1, x21, x20
    mov x2, #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x20, x20, x0
    cmp x20, #16
    b.ne read_loop
    b print_row

end_input:
    mov x28, #1
    cbz x20, print_final

print_row:
    mov x24, x22
    bl print_offset
    mov w9, #32
    strb w9, [x24], #1
    strb w9, [x24], #1

    mov x26, #0
hex_slots:
    cmp x26, x20
    b.hs empty_slot

    ldrb w27, [x21, x26]
    lsr w9, w27, #4
    ldrb w10, [x23, x9]
    strb w10, [x24], #1
    and w9, w27, #15
    ldrb w10, [x23, x9]
    strb w10, [x24], #1
    mov w10, #32
    strb w10, [x24], #1
    b slot_done

empty_slot:
    mov w10, #32
    strb w10, [x24], #1
    strb w10, [x24], #1
    strb w10, [x24], #1

slot_done:
    cmp x26, #7
    b.ne no_middle_space
    mov w10, #32
    strb w10, [x24], #1
no_middle_space:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_slots

    mov w10, #32
    strb w10, [x24], #1
    mov w10, #124
    strb w10, [x24], #1

    mov x26, #0
ascii_loop:
    cmp x26, x20
    b.hs ascii_done
    ldrb w9, [x21, x26]
    cmp w9, #32
    b.lo ascii_dot
    cmp w9, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w9, #46
ascii_store:
    strb w9, [x24], #1
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w10, #124
    strb w10, [x24], #1
    mov w10, #10
    strb w10, [x24], #1

    mov x0, #1
    mov x1, x22
    sub x2, x24, x22
    mov x8, #64
    svc #0

    add x19, x19, x20
    mov x20, #0
    cbz x28, read_loop

print_final:
    mov x24, x22
    bl print_offset
    mov w10, #10
    strb w10, [x24], #1

    mov x0, #1
    mov x1, x22
    sub x2, x24, x22
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_offset:
    mov x25, #28
offset_digit:
    lsr x9, x19, x25
    and x9, x9, #15
    ldrb w10, [x23, x9]
    strb w10, [x24], #1
    subs x25, x25, #4
    b.pl offset_digit
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 16
output_buf:
    .space 80
