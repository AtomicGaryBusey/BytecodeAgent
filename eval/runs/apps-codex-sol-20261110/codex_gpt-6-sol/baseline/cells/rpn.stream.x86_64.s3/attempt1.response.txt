.globl _start

.section .bss
.balign 8
input_buffer:
.space 65536
value_stack:
.space 8008
output_buffer:
.space 32

.section .rodata
error_message:
.ascii "error\n"

.section .text
_start:
    lea r8, [rip + input_buffer]
    lea r9, [rip + value_stack]
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz input_ready
    add r12, rax
    jmp read_more

input_ready:
    xor r13d, r13d

line_start:
    cmp r13, r12
    je exit_success
    xor r14d, r14d

skip_separators:
    cmp r13, r12
    je line_done
    movzx eax, byte ptr [r8 + r13]
    cmp al, 10
    je line_newline
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    mov rbx, r13

scan_token:
    inc r13
    cmp r13, r12
    je token_end
    movzx eax, byte ptr [r8 + r13]
    cmp al, 10
    je token_end
    cmp al, 32
    je token_end
    cmp al, 9
    jne scan_token

token_end:
    mov r10, r13
    sub r10, rbx
    call process_token
    jc line_error
    jmp skip_separators

skip_one:
    inc r13
    jmp skip_separators

line_newline:
    inc r13

line_done:
    cmp r14, 1
    jne write_error

    mov rax, qword ptr [r9]
    lea rsi, [rip + output_buffer + 31]
    mov byte ptr [rsi], 10
    mov r10d, 10
    xor r11d, r11d
    test rax, rax
    jns decimal_digits
    mov r11d, 1
    neg rax

decimal_digits:
    xor edx, edx
    div r10
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jne decimal_digits
    test r11d, r11d
    jz write_number
    dec rsi
    mov byte ptr [rsi], 45

write_number:
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp line_start

line_error:
    cmp r13, r12
    je write_error
    cmp byte ptr [r8 + r13], 10
    je error_newline
    inc r13
    jmp line_error

error_newline:
    inc r13

write_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall
    jmp line_start

process_token:
    lea rsi, [r8 + rbx]
    cmp r10, 1
    jne parse_literal

    movzx ecx, byte ptr [rsi]
    cmp cl, 43
    je operator_add
    cmp cl, 45
    je operator_sub
    cmp cl, 42
    je operator_mul
    cmp cl, 47
    je operator_div
    cmp cl, 37
    je operator_mod

parse_literal:
    xor edx, edx
    xor ecx, ecx
    cmp byte ptr [rsi], 45
    jne literal_length
    mov ecx, 1
    inc rsi
    dec r10

literal_length:
    test r10, r10
    jz invalid_token
    cmp r10, 18
    ja invalid_token
    xor eax, eax

literal_digit:
    movzx edi, byte ptr [rsi]
    sub edi, 48
    cmp edi, 9
    ja invalid_token
    imul rax, rax, 10
    add rax, rdi
    inc rsi
    dec r10
    jne literal_digit
    test ecx, ecx
    jz literal_push
    neg rax

literal_push:
    mov qword ptr [r9 + r14*8], rax
    inc r14
    clc
    ret

operator_add:
    cmp r14, 2
    jb invalid_token
    mov rax, qword ptr [r9 + r14*8 - 16]
    add rax, qword ptr [r9 + r14*8 - 8]
    jmp operator_push

operator_sub:
    cmp r14, 2
    jb invalid_token
    mov rax, qword ptr [r9 + r14*8 - 16]
    sub rax, qword ptr [r9 + r14*8 - 8]
    jmp operator_push

operator_mul:
    cmp r14, 2
    jb invalid_token
    mov rax, qword ptr [r9 + r14*8 - 16]
    imul rax, qword ptr [r9 + r14*8 - 8]
    jmp operator_push

operator_div:
    xor ecx, ecx
    jmp divide_common

operator_mod:
    mov ecx, 1

divide_common:
    cmp r14, 2
    jb invalid_token
    mov rbx, qword ptr [r9 + r14*8 - 8]
    test rbx, rbx
    jz invalid_token
    mov rax, qword ptr [r9 + r14*8 - 16]
    cmp rbx, -1
    jne divide_safe
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    je invalid_token

divide_safe:
    cqo
    idiv rbx
    test ecx, ecx
    jz operator_push
    mov rax, rdx

operator_push:
    dec r14
    mov qword ptr [r9 + r14*8 - 8], rax
    clc
    ret

invalid_token:
    stc
    ret

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
