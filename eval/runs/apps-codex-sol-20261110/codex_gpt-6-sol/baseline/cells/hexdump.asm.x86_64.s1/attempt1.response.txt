.globl _start

.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
input_buffer:
.zero 16
output_buffer:
.zero 80

.text
_start:
    lea rbx, [rip + hex_digits]
    xor r12d, r12d
    xor r13d, r13d

read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 16
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz end_of_input
    add r12, rax
    cmp r12, 16
    jne read_loop
    call print_row
    jmp read_loop

end_of_input:
    test r12, r12
    jz print_total
    call print_row

print_total:
    lea rdi, [rip + output_buffer]
    mov r8d, r13d
    call print_offset
    mov byte ptr [rdi], 10
    inc rdi
    call write_output
    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov ecx, 8
offset_loop:
    mov edx, r8d
    shr edx, 28
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl r8d, 4
    dec ecx
    jne offset_loop
    ret

print_row:
    lea rdi, [rip + output_buffer]
    mov r8d, r13d
    call print_offset
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2
    lea r9, [rip + input_buffer]
    xor ecx, ecx

hex_loop:
    cmp rcx, r12
    jae empty_slot
    movzx eax, byte ptr [r9 + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    jmp slot_done

empty_slot:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32

slot_done:
    add rdi, 3
    cmp ecx, 7
    jne no_middle_space
    mov byte ptr [rdi], 32
    inc rdi

no_middle_space:
    inc ecx
    cmp ecx, 16
    jne hex_loop
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2
    xor ecx, ecx

ascii_loop:
    cmp rcx, r12
    jae row_end
    mov al, byte ptr [r9 + rcx]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    jbe printable

nonprintable:
    mov al, 46

printable:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_loop

row_end:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    call write_output
    add r13, r12
    xor r12d, r12d
    ret

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    ret
