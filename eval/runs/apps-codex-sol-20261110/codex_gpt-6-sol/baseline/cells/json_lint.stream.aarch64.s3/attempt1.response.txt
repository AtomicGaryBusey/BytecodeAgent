.section .bss
.balign 16
input_buf:
.space 65536
output_buf:
.space 64

.section .rodata
ok_msg:
.ascii "ok\n"
error_prefix:
.ascii "error at "
true_text:
.ascii "true"
false_text:
.ascii "false"
null_text:
.ascii "null"

.text
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, x21
    mov x2, #65536
    add x22, x21, x2

read_loop:
    mov x0, #0
    mov x1, x20
    sub x2, x22, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x19, x21
    bl skip_ws
    bl parse_value
    cbnz w0, report_error
    bl skip_ws
    cmp x19, x20
    b.ne report_error

    mov x0, #1
    adr x1, ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

report_error:
    mov x0, #1
    adr x1, error_prefix
    mov x2, #9
    mov x8, #64
    svc #0

    sub x4, x19, x21
    mov x5, #10
    adrp x6, output_buf
    add x6, x6, :lo12:output_buf
    add x10, x6, #64
    mov x6, x10
    mov w7, #10
    strb w7, [x6, #-1]!

decimal_loop:
    udiv x7, x4, x5
    msub x9, x7, x5, x4
    add w9, w9, #48
    strb w9, [x6, #-1]!
    mov x4, x7
    cbnz x4, decimal_loop

    mov x0, #1
    mov x1, x6
    sub x2, x10, x6
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x19, x20
    b.hs skip_ws_done
    ldrb w9, [x19]
    cmp w9, #32
    b.eq skip_ws_advance
    cmp w9, #9
    b.eq skip_ws_advance
    cmp w9, #10
    b.eq skip_ws_advance
    cmp w9, #13
    b.ne skip_ws_done
skip_ws_advance:
    add x19, x19, #1
    b skip_ws
skip_ws_done:
    ret

parse_value:
    cmp x19, x20
    b.hs parse_value_error
    ldrb w9, [x19]
    cmp w9, #123
    b.eq parse_object
    cmp w9, #91
    b.eq parse_array
    cmp w9, #34
    b.eq parse_string
    cmp w9, #45
    b.eq parse_number
    cmp w9, #48
    b.lo parse_value_other
    cmp w9, #57
    b.ls parse_number
parse_value_other:
    cmp w9, #116
    b.eq parse_true
    cmp w9, #102
    b.eq parse_false
    cmp w9, #110
    b.eq parse_null
parse_value_error:
    mov w0, #1
    ret

parse_true:
    adr x0, true_text
    mov x2, #4
    b parse_literal
parse_false:
    adr x0, false_text
    mov x2, #5
    b parse_literal
parse_null:
    adr x0, null_text
    mov x2, #4

parse_literal:
    cbz x2, parse_literal_ok
    cmp x19, x20
    b.hs parse_literal_error
    ldrb w9, [x19]
    ldrb w10, [x0]
    cmp w9, w10
    b.ne parse_literal_error
    add x19, x19, #1
    add x0, x0, #1
    sub x2, x2, #1
    b parse_literal
parse_literal_ok:
    mov w0, #0
    ret
parse_literal_error:
    mov w0, #1
    ret

parse_string:
    add x19, x19, #1
string_loop:
    cmp x19, x20
    b.hs string_error
    ldrb w9, [x19]
    cmp w9, #32
    b.lo string_error
    cmp w9, #34
    b.eq string_close
    cmp w9, #92
    b.eq string_escape
    add x19, x19, #1
    b string_loop
string_close:
    add x19, x19, #1
    mov w0, #0
    ret

string_escape:
    add x19, x19, #1
    cmp x19, x20
    b.hs string_error
    ldrb w9, [x19]
    cmp w9, #34
    b.eq string_escape_done
    cmp w9, #92
    b.eq string_escape_done
    cmp w9, #47
    b.eq string_escape_done
    cmp w9, #98
    b.eq string_escape_done
    cmp w9, #102
    b.eq string_escape_done
    cmp w9, #110
    b.eq string_escape_done
    cmp w9, #114
    b.eq string_escape_done
    cmp w9, #116
    b.eq string_escape_done
    cmp w9, #117
    b.ne string_error
    add x19, x19, #1
    mov w10, #4
string_hex:
    cmp x19, x20
    b.hs string_error
    ldrb w9, [x19]
    cmp w9, #48
    b.lo string_error
    cmp w9, #57
    b.ls string_hex_done
    cmp w9, #65
    b.lo string_error
    cmp w9, #70
    b.ls string_hex_done
    cmp w9, #97
    b.lo string_error
    cmp w9, #102
    b.hi string_error
string_hex_done:
    add x19, x19, #1
    subs w10, w10, #1
    b.ne string_hex
    b string_loop
string_escape_done:
    add x19, x19, #1
    b string_loop
string_error:
    mov w0, #1
    ret

parse_number:
    ldrb w9, [x19]
    cmp w9, #45
    b.ne number_int
    add x19, x19, #1
    cmp x19, x20
    b.hs number_error
    ldrb w9, [x19]
number_int:
    cmp w9, #48
    b.eq number_zero
    cmp w9, #49
    b.lo number_error
    cmp w9, #57
    b.hi number_error
number_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_ok
    ldrb w9, [x19]
    cmp w9, #48
    b.lo number_after_int
    cmp w9, #57
    b.ls number_digits
    b number_after_int
number_zero:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_ok
    ldrb w9, [x19]
number_after_int:
    cmp w9, #46
    b.ne number_check_exp
    add x19, x19, #1
    cmp x19, x20
    b.hs number_error
    ldrb w9, [x19]
    cmp w9, #48
    b.lo number_error
    cmp w9, #57
    b.hi number_error
number_frac_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_ok
    ldrb w9, [x19]
    cmp w9, #48
    b.lo number_check_exp
    cmp w9, #57
    b.ls number_frac_digits
number_check_exp:
    cmp w9, #101
    b.eq number_exp
    cmp w9, #69
    b.ne number_ok
number_exp:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_error
    ldrb w9, [x19]
    cmp w9, #43
    b.eq number_exp_sign
    cmp w9, #45
    b.ne number_exp_first
number_exp_sign:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_error
    ldrb w9, [x19]
number_exp_first:
    cmp w9, #48
    b.lo number_error
    cmp w9, #57
    b.hi number_error
number_exp_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs number_ok
    ldrb w9, [x19]
    cmp w9, #48
    b.lo number_ok
    cmp w9, #57
    b.ls number_exp_digits
number_ok:
    mov w0, #0
    ret
number_error:
    mov w0, #1
    ret

parse_array:
    str x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs array_error
    ldrb w9, [x19]
    cmp w9, #93
    b.eq array_close
array_element:
    bl parse_value
    cbnz w0, array_return
    bl skip_ws
    cmp x19, x20
    b.hs array_error
    ldrb w9, [x19]
    cmp w9, #93
    b.eq array_close
    cmp w9, #44
    b.ne array_error
    add x19, x19, #1
    bl skip_ws
    b array_element
array_close:
    add x19, x19, #1
    mov w0, #0
    b array_return
array_error:
    mov w0, #1
array_return:
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs object_error
    ldrb w9, [x19]
    cmp w9, #125
    b.eq object_close
object_member:
    cmp x19, x20
    b.hs object_error
    ldrb w9, [x19]
    cmp w9, #34
    b.ne object_error
    bl parse_string
    cbnz w0, object_return
    bl skip_ws
    cmp x19, x20
    b.hs object_error
    ldrb w9, [x19]
    cmp w9, #58
    b.ne object_error
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    cbnz w0, object_return
    bl skip_ws
    cmp x19, x20
    b.hs object_error
    ldrb w9, [x19]
    cmp w9, #125
    b.eq object_close
    cmp w9, #44
    b.ne object_error
    add x19, x19, #1
    bl skip_ws
    b object_member
object_close:
    add x19, x19, #1
    mov w0, #0
    b object_return
object_error:
    mov w0, #1
object_return:
    ldr x30, [sp], #16
    ret
