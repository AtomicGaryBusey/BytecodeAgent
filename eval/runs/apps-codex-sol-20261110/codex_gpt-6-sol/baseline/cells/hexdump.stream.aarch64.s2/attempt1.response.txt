.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
row:
    .space 16
outbuf:
    .space 128

.text
_start:
    adrp x21, row
    add x21, x21, :lo12:row
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits
    mov x19, #0
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x21, x20
    mov x2, #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x20, x20, x0
    cmp x20, #16
    b.ne read_loop

    mov x0, x20
    bl print_row
    add x19, x19, #16
    mov x20, #0
    b read_loop

end_input:
    cbz x20, print_total
    mov x0, x20
    bl print_row
    add x19, x19, x20

print_total:
    mov x9, x22
    mov x11, #28
total_digit:
    lsr x10, x19, x11
    and x10, x10, #15
    ldrb w10, [x23, x10]
    strb w10, [x9], #1
    subs x11, x11, #4
    b.pl total_digit
    mov w10, #10
    strb w10, [x9], #1

    mov x0, #1
    mov x1, x22
    sub x2, x9, x22
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_row:
    mov x14, x0
    mov x9, x22
    mov x11, #28
row_offset_digit:
    lsr x10, x19, x11
    and x10, x10, #15
    ldrb w10, [x23, x10]
    strb w10, [x9], #1
    subs x11, x11, #4
    b.pl row_offset_digit

    mov w10, #32
    strb w10, [x9], #1
    strb w10, [x9], #1
    mov x12, #0

hex_slot:
    cmp x12, #8
    b.ne no_group_space
    mov w10, #32
    strb w10, [x9], #1
no_group_space:
    cmp x12, x14
    b.hs empty_slot

    ldrb w15, [x21, x12]
    lsr w16, w15, #4
    ldrb w16, [x23, w16, uxtw]
    strb w16, [x9], #1
    and w16, w15, #15
    ldrb w16, [x23, w16, uxtw]
    strb w16, [x9], #1
    b slot_space

empty_slot:
    mov w10, #32
    strb w10, [x9], #1
    strb w10, [x9], #1

slot_space:
    mov w10, #32
    strb w10, [x9], #1
    add x12, x12, #1
    cmp x12, #16
    b.lo hex_slot

    mov w10, #32
    strb w10, [x9], #1
    mov w10, #124
    strb w10, [x9], #1
    mov x12, #0

ascii_loop:
    cmp x12, x14
    b.hs ascii_done
    ldrb w15, [x21, x12]
    cmp w15, #32
    b.lo ascii_dot
    cmp w15, #126
    b.ls ascii_store
ascii_dot:
    mov w15, #46
ascii_store:
    strb w15, [x9], #1
    add x12, x12, #1
    b ascii_loop

ascii_done:
    mov w10, #124
    strb w10, [x9], #1
    mov w10, #10
    strb w10, [x9], #1

    mov x0, #1
    mov x1, x22
    sub x2, x9, x22
    mov x8, #64
    svc #0
    ret
