.section .bss
.balign 16
input_buf:
.space 65537
digits:
.space 32

.section .rodata
ok_msg:
.ascii "ok\n"
err_msg:
.ascii "error at "
true_text:
.ascii "true"
false_text:
.ascii "false"
null_text:
.ascii "null"

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65537
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    bl skip_ws
    bl parse_value
    cbnz x0, report_error
    bl skip_ws
    cmp x21, x20
    b.ne report_error

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

report_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x26, digits
    add x26, x26, :lo12:digits
    add x26, x26, #32
    sub x25, x26, #1
    mov w3, #10
    strb w3, [x25]
    mov x4, x21
    mov x5, #10
digit_loop:
    udiv x6, x4, x5
    msub x7, x6, x5, x4
    add w7, w7, #48
    strb w7, [x25, #-1]!
    mov x4, x6
    cbnz x4, digit_loop

    mov x0, #1
    mov x1, x25
    sub x2, x26, x25
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// Advances x21 over JSON whitespace.
skip_ws:
    cmp x21, x20
    b.hs 2f
1:
    ldrb w9, [x19, x21]
    cmp w9, #32
    b.eq 3f
    cmp w9, #9
    b.eq 3f
    cmp w9, #10
    b.eq 3f
    cmp w9, #13
    b.ne 2f
3:
    add x21, x21, #1
    cmp x21, x20
    b.lo 1b
2:
    ret

// Returns x0=0 for a value, x0=1 at the first invalid byte.
parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x21, x20
    b.hs value_bad
    ldrb w9, [x19, x21]
    cmp w9, #123
    b.eq value_object
    cmp w9, #91
    b.eq value_array
    cmp w9, #34
    b.eq value_string
    cmp w9, #116
    b.eq value_true
    cmp w9, #102
    b.eq value_false
    cmp w9, #110
    b.eq value_null
    cmp w9, #45
    b.eq value_number
    cmp w9, #48
    b.lo value_bad
    cmp w9, #57
    b.hi value_bad
value_number:
    bl parse_number
    b value_return
value_object:
    bl parse_object
    b value_return
value_array:
    bl parse_array
    b value_return
value_string:
    bl parse_string
    b value_return
value_true:
    adrp x1, true_text
    add x1, x1, :lo12:true_text
    mov x2, #4
    bl parse_literal
    b value_return
value_false:
    adrp x1, false_text
    add x1, x1, :lo12:false_text
    mov x2, #5
    bl parse_literal
    b value_return
value_null:
    adrp x1, null_text
    add x1, x1, :lo12:null_text
    mov x2, #4
    bl parse_literal
    b value_return
value_bad:
    mov x0, #1
value_return:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
1:
    cmp x21, x20
    b.hs 2f
    ldrb w9, [x19, x21]
    ldrb w10, [x1], #1
    cmp w9, w10
    b.ne 2f
    add x21, x21, #1
    subs x2, x2, #1
    b.ne 1b
    mov x0, #0
    ret
2:
    mov x0, #1
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs array_bad
    ldrb w9, [x19, x21]
    cmp w9, #93
    b.eq array_close
array_element:
    bl parse_value
    cbnz x0, array_return
    bl skip_ws
    cmp x21, x20
    b.hs array_bad
    ldrb w9, [x19, x21]
    cmp w9, #93
    b.eq array_close
    cmp w9, #44
    b.ne array_bad
    add x21, x21, #1
    bl skip_ws
    b array_element
array_close:
    add x21, x21, #1
    mov x0, #0
    b array_return
array_bad:
    mov x0, #1
array_return:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs object_bad
    ldrb w9, [x19, x21]
    cmp w9, #125
    b.eq object_close
object_member:
    cmp x21, x20
    b.hs object_bad
    ldrb w9, [x19, x21]
    cmp w9, #34
    b.ne object_bad
    bl parse_string
    cbnz x0, object_return
    bl skip_ws
    cmp x21, x20
    b.hs object_bad
    ldrb w9, [x19, x21]
    cmp w9, #58
    b.ne object_bad
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cbnz x0, object_return
    bl skip_ws
    cmp x21, x20
    b.hs object_bad
    ldrb w9, [x19, x21]
    cmp w9, #125
    b.eq object_close
    cmp w9, #44
    b.ne object_bad
    add x21, x21, #1
    bl skip_ws
    b object_member
object_close:
    add x21, x21, #1
    mov x0, #0
    b object_return
object_bad:
    mov x0, #1
object_return:
    ldp x29, x30, [sp], #16
    ret

parse_string:
    add x21, x21, #1
string_next:
    cmp x21, x20
    b.hs string_bad
    ldrb w9, [x19, x21]
    cmp w9, #34
    b.eq string_close
    cmp w9, #32
    b.lo string_bad
    cmp w9, #92
    b.eq string_escape
    add x21, x21, #1
    b string_next
string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.hs string_bad
    ldrb w9, [x19, x21]
    cmp w9, #117
    b.eq string_unicode
    cmp w9, #34
    b.eq string_escaped
    cmp w9, #92
    b.eq string_escaped
    cmp w9, #47
    b.eq string_escaped
    cmp w9, #98
    b.eq string_escaped
    cmp w9, #102
    b.eq string_escaped
    cmp w9, #110
    b.eq string_escaped
    cmp w9, #114
    b.eq string_escaped
    cmp w9, #116
    b.ne string_bad
string_escaped:
    add x21, x21, #1
    b string_next
string_unicode:
    add x21, x21, #1
    mov x10, #4
string_hex:
    cmp x21, x20
    b.hs string_bad
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo string_bad
    cmp w9, #57
    b.ls string_hex_ok
    cmp w9, #65
    b.lo string_bad
    cmp w9, #70
    b.ls string_hex_ok
    cmp w9, #97
    b.lo string_bad
    cmp w9, #102
    b.hi string_bad
string_hex_ok:
    add x21, x21, #1
    subs x10, x10, #1
    b.ne string_hex
    b string_next
string_close:
    add x21, x21, #1
    mov x0, #0
    ret
string_bad:
    mov x0, #1
    ret

parse_number:
    ldrb w9, [x19, x21]
    cmp w9, #45
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.hs number_bad
number_integer:
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.eq number_zero
    cmp w9, #49
    b.lo number_bad
    cmp w9, #57
    b.hi number_bad
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_good
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_suffix
    cmp w9, #57
    b.ls number_digits
    b number_suffix
number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_good
    ldrb w9, [x19, x21]
number_suffix:
    cmp w9, #46
    b.eq number_fraction
    cmp w9, #101
    b.eq number_exponent
    cmp w9, #69
    b.eq number_exponent
    b number_good
number_fraction:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_bad
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_bad
    cmp w9, #57
    b.hi number_bad
number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_good
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_fraction_end
    cmp w9, #57
    b.ls number_fraction_digits
number_fraction_end:
    cmp w9, #101
    b.eq number_exponent
    cmp w9, #69
    b.ne number_good
number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_bad
    ldrb w9, [x19, x21]
    cmp w9, #43
    b.eq number_exp_sign
    cmp w9, #45
    b.ne number_exp_first
number_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_bad
    ldrb w9, [x19, x21]
number_exp_first:
    cmp w9, #48
    b.lo number_bad
    cmp w9, #57
    b.hi number_bad
number_exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_good
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_good
    cmp w9, #57
    b.ls number_exp_digits
number_good:
    mov x0, #0
    ret
number_bad:
    mov x0, #1
    ret
