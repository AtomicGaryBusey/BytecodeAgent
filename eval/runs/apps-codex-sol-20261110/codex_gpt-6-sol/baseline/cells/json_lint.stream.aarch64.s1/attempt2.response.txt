.section .bss
.balign 16
input_buf:
.space 65537
digits:
.space 32

.section .rodata
ok_msg:
.ascii "ok\n"
err_msg:
.ascii "error at "
true_text:
.ascii "true"
false_text:
.ascii "false"
null_text:
.ascii "null"

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne error_output

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

skip_ws:
    cmp x21, x20
    b.hs skip_ws_done
    ldrb w9, [x19, x21]
    cmp w9, #32
    b.eq skip_ws_advance
    cmp w9, #9
    b.eq skip_ws_advance
    cmp w9, #10
    b.eq skip_ws_advance
    cmp w9, #13
    b.ne skip_ws_done
skip_ws_advance:
    add x21, x21, #1
    b skip_ws
skip_ws_done:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #34
    b.eq value_string
    cmp w9, #123
    b.eq value_object
    cmp w9, #91
    b.eq value_array
    cmp w9, #116
    b.eq value_true
    cmp w9, #102
    b.eq value_false
    cmp w9, #110
    b.eq value_null
    cmp w9, #45
    b.eq value_number
    cmp w9, #48
    b.lo error_output
    cmp w9, #57
    b.hi error_output
value_number:
    bl parse_number
    b value_done
value_string:
    bl parse_string
    b value_done
value_object:
    bl parse_object
    b value_done
value_array:
    bl parse_array
    b value_done
value_true:
    adrp x2, true_text
    add x2, x2, :lo12:true_text
    mov x3, #4
    bl parse_literal
    b value_done
value_false:
    adrp x2, false_text
    add x2, x2, :lo12:false_text
    mov x3, #5
    bl parse_literal
    b value_done
value_null:
    adrp x2, null_text
    add x2, x2, :lo12:null_text
    mov x3, #4
    bl parse_literal
value_done:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    ldrb w10, [x2], #1
    cmp w9, w10
    b.ne error_output
    add x21, x21, #1
    subs x3, x3, #1
    b.ne parse_literal
    ret

parse_string:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #34
    b.eq string_end
    cmp w9, #92
    b.eq string_escape
    cmp w9, #32
    b.lo error_output
    add x21, x21, #1
    b string_loop
string_end:
    add x21, x21, #1
    ret

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #117
    b.eq string_unicode
    cmp w9, #34
    b.eq string_escaped
    cmp w9, #92
    b.eq string_escaped
    cmp w9, #47
    b.eq string_escaped
    cmp w9, #98
    b.eq string_escaped
    cmp w9, #102
    b.eq string_escaped
    cmp w9, #110
    b.eq string_escaped
    cmp w9, #114
    b.eq string_escaped
    cmp w9, #116
    b.ne error_output
string_escaped:
    add x21, x21, #1
    b string_loop

string_unicode:
    add x21, x21, #1
    mov x10, #4
unicode_loop:
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo error_output
    cmp w9, #57
    b.ls unicode_digit
    cmp w9, #65
    b.lo error_output
    cmp w9, #70
    b.ls unicode_digit
    cmp w9, #97
    b.lo error_output
    cmp w9, #102
    b.hi error_output
unicode_digit:
    add x21, x21, #1
    subs x10, x10, #1
    b.ne unicode_loop
    b string_loop

parse_number:
    ldrb w9, [x19, x21]
    cmp w9, #45
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
number_integer:
    cmp w9, #48
    b.eq number_zero
    cmp w9, #49
    b.lo error_output
    cmp w9, #57
    b.hi error_output
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_fraction_check
    cmp w9, #57
    b.ls number_digits
    b number_fraction_check
number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w9, [x19, x21]
number_fraction_check:
    cmp w9, #46
    b.ne number_exponent_check
    add x21, x21, #1
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo error_output
    cmp w9, #57
    b.hi error_output
number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_exponent_check
    cmp w9, #57
    b.ls number_fraction_digits
number_exponent_check:
    cmp w9, #101
    b.eq number_exponent
    cmp w9, #69
    b.ne number_done
number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #43
    b.eq number_exponent_sign
    cmp w9, #45
    b.ne number_exponent_first
number_exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
number_exponent_first:
    cmp w9, #48
    b.lo error_output
    cmp w9, #57
    b.hi error_output
number_exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w9, [x19, x21]
    cmp w9, #48
    b.lo number_done
    cmp w9, #57
    b.ls number_exponent_digits
number_done:
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #93
    b.eq array_end
array_element:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #93
    b.eq array_end
    cmp w9, #44
    b.ne error_output
    add x21, x21, #1
    b array_element
array_end:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #125
    b.eq object_end
object_member:
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #34
    b.ne error_output
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #58
    b.ne error_output
    add x21, x21, #1
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs error_output
    ldrb w9, [x19, x21]
    cmp w9, #125
    b.eq object_end
    cmp w9, #44
    b.ne error_output
    add x21, x21, #1
    bl skip_ws
    b object_member
object_end:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

error_output:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x7, digits
    add x7, x7, :lo12:digits
    add x7, x7, #31
    mov w9, #10
    strb w9, [x7]
    mov x4, x21
    mov x6, #10
decimal_loop:
    udiv x5, x4, x6
    msub x9, x5, x6, x4
    add w9, w9, #48
    strb w9, [x7, #-1]!
    mov x4, x5
    cbnz x4, decimal_loop

    mov x0, #1
    mov x1, x7
    adrp x2, digits
    add x2, x2, :lo12:digits
    add x2, x2, #32
    sub x2, x2, x7
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
