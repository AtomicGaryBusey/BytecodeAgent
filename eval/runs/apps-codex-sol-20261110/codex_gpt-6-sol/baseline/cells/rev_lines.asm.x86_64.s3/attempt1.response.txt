.section .bss
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    lea r13, [rip + output_buffer]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz process_input
    add r12, rax
    jmp read_loop

process_input:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan_loop:
    cmp r8, r12
    jae final_line
    cmp byte ptr [rbx + r8], 10
    je newline
    inc r8
    jmp scan_loop

newline:
    mov rdx, r8
reverse_line:
    cmp rdx, r9
    jbe line_done
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [r13 + r10], al
    inc r10
    jmp reverse_line

line_done:
    mov byte ptr [r13 + r10], 10
    inc r10
    inc r8
    mov r9, r8
    jmp scan_loop

final_line:
    cmp r9, r12
    je write_output
    mov rdx, r12
reverse_final:
    cmp rdx, r9
    jbe final_done
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [r13 + r10], al
    inc r10
    jmp reverse_final

final_done:
    mov byte ptr [r13 + r10], 10
    inc r10

write_output:
    test r10, r10
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r10
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
