.text
.globl _start
_start:
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x25, x23
    mov x26, x0

scan_loop:
    ldrb w4, [x25], #1
    cmp w4, #10
    b.ne check_whitespace
    add x19, x19, #1

check_whitespace:
    cmp w4, #32
    b.eq whitespace
    cmp w4, #9
    b.lo word_byte
    cmp w4, #13
    b.ls whitespace

word_byte:
    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

whitespace:
    mov x22, #0

next_byte:
    subs x26, x26, #1
    b.ne scan_loop
    b read_loop

input_done:
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    add x24, x24, #128
    mov x1, x24

    mov w4, #10
    strb w4, [x1, #-1]!
    mov x0, x21
    bl prepend_decimal

    mov w4, #32
    strb w4, [x1, #-1]!
    mov x0, x20
    bl prepend_decimal

    mov w4, #32
    strb w4, [x1, #-1]!
    mov x0, x19
    bl prepend_decimal

    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

prepend_decimal:
    mov x3, #10
1:
    udiv x2, x0, x3
    msub x4, x2, x3, x0
    add w4, w4, #48
    strb w4, [x1, #-1]!
    mov x0, x2
    cbnz x0, 1b
    ret

.bss
.balign 16
input_buf:
    .space 4096
output_buf:
    .space 128
