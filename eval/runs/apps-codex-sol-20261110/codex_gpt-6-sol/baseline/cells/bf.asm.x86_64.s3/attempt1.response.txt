.section .bss
.balign 8
input_buffer:
.zero 65536
extra_byte:
.zero 1
tape:
.zero 30000
output_buffer:
.zero 65536
bracket_map:
.zero 262144
bracket_stack:
.zero 4000

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_input:
    mov edx, 65536
    sub rdx, r12
    jz check_eof
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz find_delimiter
    add r12, rax
    jmp read_input

check_eof:
    lea rsi, [rip + extra_byte]
    xor edi, edi
    mov edx, 1
    xor eax, eax
    syscall
    test rax, rax
    jnz check_eof

find_delimiter:
    mov r13, r12
    xor r12d, r12d
    mov r14, r13

delimiter_loop:
    cmp r12, r13
    jae validate
    cmp byte ptr [rbx + r12], 33
    je found_delimiter
    inc r12
    jmp delimiter_loop

found_delimiter:
    mov r14, r12

validate:
    lea r8, [rip + bracket_stack]
    lea r9, [rip + input_buffer]
    lea rbx, [rip + bracket_map]
    xor r12d, r12d
    xor r15d, r15d

validate_loop:
    cmp r12, r14
    jae validation_done
    movzx eax, byte ptr [r9 + r12]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket

validate_next:
    inc r12
    jmp validate_loop

open_bracket:
    mov dword ptr [r8 + r15*4], r12d
    inc r15
    jmp validate_next

close_bracket:
    test r15, r15
    jz invalid_program
    dec r15
    mov eax, dword ptr [r8 + r15*4]
    lea edx, [r12 + 1]
    mov dword ptr [rbx + rax*4], edx
    inc eax
    mov dword ptr [rbx + r12*4], eax
    jmp validate_next

validation_done:
    test r15, r15
    jnz invalid_program

    lea r8, [rip + input_buffer]
    lea r9, [rip + tape]
    lea r10, [rip + output_buffer]
    mov r11, rbx
    xor ebx, ebx
    xor r12d, r12d
    xor r13d, r13d
    lea r15, [r14 + 1]
    mov rdi, r14

execute:
    cmp r12, rdi
    jae finish
    movzx eax, byte ptr [r8 + r12]
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment_cell
    cmp al, 45
    je decrement_cell
    cmp al, 46
    je output_cell
    cmp al, 44
    je input_cell
    cmp al, 91
    je loop_start
    cmp al, 93
    je loop_end

next_command:
    inc r12
    jmp execute

move_right:
    inc rbx
    jmp next_command

move_left:
    dec rbx
    jmp next_command

increment_cell:
    inc byte ptr [r9 + rbx]
    jmp next_command

decrement_cell:
    dec byte ptr [r9 + rbx]
    jmp next_command

output_cell:
    mov al, byte ptr [r9 + rbx]
    mov byte ptr [r10 + r13], al
    inc r13
    jmp next_command

input_cell:
    xor eax, eax
    cmp r15, rdi
    jae store_input
    cmp r15, qword ptr [rip + input_length]
    jae store_input
    mov al, byte ptr [r8 + r15]
    inc r15

store_input:
    mov byte ptr [r9 + rbx], al
    jmp next_command

loop_start:
    cmp byte ptr [r9 + rbx], 0
    jne next_command
    mov r12d, dword ptr [r11 + r12*4]
    jmp execute

loop_end:
    cmp byte ptr [r9 + rbx], 0
    je next_command
    mov r12d, dword ptr [r11 + r12*4]
    jmp execute

finish:
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r13
    syscall
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.section .data
input_length:
.quad 0
