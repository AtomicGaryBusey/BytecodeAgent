.section .bss
.balign 16
buffer:
.space 8192
outbuf:
.space 64

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    mov r9, rax
    lea r8, [rip + buffer]

process_byte:
    movzx eax, byte ptr [r8]
    inc r8
    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jne next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec r9
    jne process_byte
    jmp read_loop

input_done:
    lea rdi, [rip + outbuf]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

append_decimal:
    sub rsp, 32
    xor ecx, ecx
    mov r8d, 10

decimal_digit:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rsp + rcx], dl
    inc ecx
    test rax, rax
    jne decimal_digit

copy_digit:
    dec ecx
    mov dl, byte ptr [rsp + rcx]
    mov byte ptr [rdi], dl
    inc rdi
    test ecx, ecx
    jne copy_digit

    add rsp, 32
    ret
