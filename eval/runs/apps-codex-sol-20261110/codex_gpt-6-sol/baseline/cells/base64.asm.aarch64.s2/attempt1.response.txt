.section .rodata

    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .space 49153

    .space 66400

.text

    add x19, x19, :lo12:input_buffer

    add x1, x19, x20

    mov x8, #63

    add x24, x24, :lo12:alphabet
    adrp x25, output_buffer
    add x25, x25, :lo12:output_buffer
    mov x21, x25
    mov x22, #0
    mov x23, x20

    b.lo encode_tail

    ldrb w9, [x19], #1
    ldrb w10, [x19], #1
    ldrb w11, [x19], #1
    sub x23, x23, #3

    lsr w12, w9, #2
    ldrb w12, [x24, w12, uxtw]

    and w12, w9, #3
    lsl w12, w12, #4
    orr w12, w12, w10, lsr #4
    ldrb w12, [x24, w12, uxtw]

    lsl w12, w12, #2
    orr w12, w12, w11, lsr #6
    ldrb w12, [x24, w12, uxtw]

    ldrb w12, [x24, w12, uxtw]
    strb w12, [x21], #1

    cmp x22, #76
    b.ne encode_loop
    mov w12, #10
    strb w12, [x21], #1
    mov x22, #0

    cbz x23, finish_lines

    ldrb w9, [x19], #1
    lsr w12, w9, #2
    ldrb w12, [x24, w12, uxtw]
    strb w12, [x21], #1

    and w12, w9, #3
    lsl w12, w12, #4
    cmp x23, #2
    b.ne one_byte
    ldrb w10, [x19], #1
    orr w12, w12, w10, lsr #4

    strb w12, [x21], #1

    cmp x23, #2
    b.ne pad_third
    and w12, w10, #15
    lsl w12, w12, #2
    ldrb w12, [x24, w12, uxtw]
    b store_third
pad_third:
    mov w12, #61
store_third:
    strb w12, [x21], #1

    mov w12, #61
    strb w12, [x21], #1
    add x22, x22, #4

finish_lines:
    cbz x22, write_output
    mov w12, #10
    strb w12, [x21], #1

    mov x1, x25
    sub x2, x21, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
