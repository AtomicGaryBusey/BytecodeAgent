.globl _start

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #32768
    sub x2, x2, x20
    cbz x2, parse_start
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, parse_start
    add x20, x20, x0
    b read_loop

parse_start:
    adrp x23, records
    add x23, x23, :lo12:records
    mov x21, #0
    mov x22, #0
    mov x24, #0

parse_loop:
    cmp x21, x20
    b.hs parse_final
    ldrb w9, [x19, x21]
    cmp w9, #10
    b.ne parse_next
    sub x10, x21, x22
    add x11, x23, x24, lsl #4
    stp x22, x10, [x11]
    add x24, x24, #1
    add x22, x21, #1

parse_next:
    add x21, x21, #1
    b parse_loop

parse_final:
    cmp x22, x20
    b.hs sort_start
    sub x10, x20, x22
    add x11, x23, x24, lsl #4
    stp x22, x10, [x11]
    add x24, x24, #1

sort_start:
    mov x21, #1

sort_loop:
    cmp x21, x24
    b.hs output_start
    add x9, x23, x21, lsl #4
    ldp x25, x26, [x9]
    mov x27, x21

sort_compare_previous:
    cbz x27, sort_insert
    sub x10, x27, #1
    add x11, x23, x10, lsl #4
    ldp x12, x13, [x11]
    mov x14, #0

sort_compare_bytes:
    cmp x14, x26
    b.hs sort_compare_lengths
    cmp x14, x13
    b.hs sort_insert
    add x15, x25, x14
    ldrb w16, [x19, x15]
    add x15, x12, x14
    ldrb w17, [x19, x15]
    cmp w16, w17
    b.lo sort_shift
    b.hi sort_insert
    add x14, x14, #1
    b sort_compare_bytes

sort_compare_lengths:
    cmp x26, x13
    b.hs sort_insert

sort_shift:
    stp x12, x13, [x9]
    mov x27, x10
    mov x9, x11
    b sort_compare_previous

sort_insert:
    stp x25, x26, [x9]
    add x21, x21, #1
    b sort_loop

output_start:
    adrp x22, newline
    add x22, x22, :lo12:newline
    mov x21, #0

output_loop:
    cmp x21, x24
    b.hs program_exit
    add x9, x23, x21, lsl #4
    ldp x25, x26, [x9]
    cbz x26, output_newline
    mov x0, #1
    add x1, x19, x25
    mov x2, x26
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    mov x1, x22
    mov x2, #1
    mov x8, #64
    svc #0
    add x21, x21, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.section .bss
    .balign 16
input_buf:
    .space 32768
    .balign 16
records:
    .space 16000
