.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input_buf:
.zero 49153
output_buf:
.zero 66400

.text
_start:
    xor ebx, ebx

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 49153
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r12, rbx
    test r12, r12
    jz exit
    lea rbx, [rip + input_buf]
    lea r13, [rip + output_buf]
    mov r15, r13
    lea r14, [rip + alphabet]
    xor r8d, r8d

encode_loop:
    movzx eax, byte ptr [rbx]
    xor ecx, ecx
    xor edx, edx
    cmp r12, 2
    jb encode_group
    movzx ecx, byte ptr [rbx + 1]
    cmp r12, 3
    jb encode_group
    movzx edx, byte ptr [rbx + 2]

encode_group:
    mov r9d, eax
    shr r9d, 2
    mov r10b, byte ptr [r14 + r9]
    mov byte ptr [r13], r10b

    and eax, 3
    shl eax, 4
    mov r9d, ecx
    shr r9d, 4
    or r9d, eax
    mov r10b, byte ptr [r14 + r9]
    mov byte ptr [r13 + 1], r10b

    mov r9d, ecx
    and r9d, 15
    shl r9d, 2
    mov eax, edx
    shr eax, 6
    or r9d, eax
    mov r10b, byte ptr [r14 + r9]
    cmp r12, 1
    jne third_ready
    mov r10b, '='
third_ready:
    mov byte ptr [r13 + 2], r10b

    and edx, 63
    mov r10b, byte ptr [r14 + rdx]
    cmp r12, 3
    jae fourth_ready
    mov r10b, '='
fourth_ready:
    mov byte ptr [r13 + 3], r10b

    add r13, 4
    add r8d, 4
    cmp r8d, 76
    jne line_done
    mov byte ptr [r13], 10
    inc r13
    xor r8d, r8d
line_done:
    cmp r12, 3
    jbe last_group
    sub r12, 3
    add rbx, 3
    jmp encode_loop

last_group:
    test r8d, r8d
    jz write_output
    mov byte ptr [r13], 10
    inc r13

write_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r13
    sub rdx, r15
    syscall

exit:
    mov eax, 60
    xor edi, edi
    syscall
