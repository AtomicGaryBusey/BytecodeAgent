.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49155
output:
.zero 67000

.text
_start:
    lea rbx, [rip + input]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r15, r12
    mov r8, rbx
    lea r14, [rip + output]
    mov r10, r14
    lea r13, [rip + alphabet]
    xor r12d, r12d
    xor r11d, r11d

encode_loop:
    test r15, r15
    jz encode_done
    cmp r15, 3
    jb short_group
    sub r15, 3
    jmp encode_group

short_group:
    mov r11d, 3
    sub r11, r15
    xor r15d, r15d

encode_group:
    movzx eax, byte ptr [r8]
    movzx ebx, byte ptr [r8 + 1]
    movzx ecx, byte ptr [r8 + 2]
    add r8, 3

    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10], dl

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov esi, ebx
    shr esi, 4
    or edx, esi
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10 + 1], dl

    mov edx, ebx
    and edx, 15
    shl edx, 2
    mov esi, ecx
    shr esi, 6
    or edx, esi
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10 + 2], dl

    and ecx, 63
    mov cl, byte ptr [r13 + rcx]
    mov byte ptr [r10 + 3], cl

    add r10, 4
    test r11, r11
    jz padding_done
    mov byte ptr [r10 - 1], '='
    cmp r11, 2
    jne padding_done
    mov byte ptr [r10 - 2], '='

padding_done:
    add r12d, 4
    cmp r12d, 76
    jne encode_loop
    mov byte ptr [r10], 10
    inc r10
    xor r12d, r12d
    jmp encode_loop

encode_done:
    test r12d, r12d
    jz write_output
    mov byte ptr [r10], 10
    inc r10

write_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    sub rdx, r14
    syscall

    mov eax, 60
    xor edi, edi
    syscall
