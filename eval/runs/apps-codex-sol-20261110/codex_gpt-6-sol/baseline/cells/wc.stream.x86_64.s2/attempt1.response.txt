.globl _start

.section .text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jz output
    add r14, rax
    mov rdx, rax
    lea rsi, [rip + input_buffer]

count_loop:
    mov al, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rdx
    jnz count_loop
    jmp read_loop

output:
    lea rbx, [rip + output_buffer]

    mov rax, r12
    mov rdi, rbx
    call append_decimal
    mov rbx, rdi
    mov byte ptr [rbx], 32
    inc rbx

    mov rax, r13
    mov rdi, rbx
    call append_decimal
    mov rbx, rdi
    mov byte ptr [rbx], 32
    inc rbx

    mov rax, r14
    mov rdi, rbx
    call append_decimal
    mov rbx, rdi
    mov byte ptr [rbx], 10
    inc rbx

    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    lea r8, [rip + digits + 32]
    mov r9, r8
    mov ecx, 10

decimal_loop:
    xor edx, edx
    div rcx
    dec r9
    add dl, 48
    mov byte ptr [r9], dl
    test rax, rax
    jnz decimal_loop

    mov rsi, r9
    mov rcx, r8
    sub rcx, r9
    rep movsb
    ret

.section .bss
input_buffer:
    .zero 8192
digits:
    .zero 32
output_buffer:
    .zero 64
