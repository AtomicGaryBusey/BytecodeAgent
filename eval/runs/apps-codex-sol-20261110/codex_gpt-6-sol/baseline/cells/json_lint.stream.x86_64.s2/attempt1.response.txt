.section .bss
.balign 16
input_buffer:
.zero 65537
digit_buffer:
.zero 32

.section .rodata
ok_message:
.ascii "ok\n"
error_prefix:
.ascii "error at "
newline:
.ascii "\n"
true_text:
.ascii "true"
false_text:
.ascii "false"
null_text:
.ascii "null"

.text
.globl _start
_start:
    lea r14, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

get_required:
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    ret

skip_ws:
    cmp r12, r13
    jae skip_ws_done
    mov al, byte ptr [r14 + r12]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_advance:
    inc r12
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    call get_required
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe parse_number
    jmp error

value_true:
    lea rsi, [rip + true_text]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + false_text]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + null_text]
    mov ecx, 4
parse_literal:
    call get_required
    cmp al, byte ptr [rsi]
    jne error
    inc r12
    inc rsi
    dec ecx
    jne parse_literal
    ret

parse_string:
    call get_required
    cmp al, '"'
    jne error
    inc r12
string_loop:
    call get_required
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb error
    cmp al, 0x5c
    je string_escape
    inc r12
    jmp string_loop
string_end:
    inc r12
    ret
string_escape:
    inc r12
    call get_required
    cmp al, '"'
    je escape_done
    cmp al, 0x5c
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    je escape_done
    cmp al, 'u'
    jne error
    inc r12
    mov ecx, 4
hex_loop:
    call get_required
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe hex_done
    cmp al, 'A'
    jb error
    cmp al, 'F'
    jbe hex_done
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
hex_done:
    inc r12
    dec ecx
    jne hex_loop
    jmp string_loop
escape_done:
    inc r12
    jmp string_loop

parse_number:
    call get_required
    cmp al, '-'
    jne number_first
    inc r12
    call get_required
number_first:
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc r12
number_digits:
    cmp r12, r13
    jae number_suffix
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    ja number_suffix
    inc r12
    jmp number_digits
number_zero:
    inc r12
number_suffix:
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '.'
    jne number_exponent_check
    inc r12
    call get_required
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r12
number_fraction_digits:
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_exponent_check
    cmp al, '9'
    ja number_exponent_check
    inc r12
    jmp number_fraction_digits
number_exponent_check:
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    jne number_done
number_exponent:
    inc r12
    call get_required
    cmp al, '+'
    je number_exponent_sign
    cmp al, '-'
    jne number_exponent_first
number_exponent_sign:
    inc r12
    call get_required
number_exponent_first:
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r12
number_exponent_digits:
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    ja number_done
    inc r12
    jmp number_exponent_digits
number_done:
    ret

parse_array:
    inc r12
    call skip_ws
    call get_required
    cmp al, ']'
    je array_end
array_element:
    call parse_value
    call skip_ws
    call get_required
    cmp al, ']'
    je array_end
    cmp al, ','
    jne error
    inc r12
    call skip_ws
    jmp array_element
array_end:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    call get_required
    cmp al, '}'
    je object_end
object_member:
    call parse_string
    call skip_ws
    call get_required
    cmp al, ':'
    jne error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    call get_required
    cmp al, '}'
    je object_end
    cmp al, ','
    jne error
    inc r12
    call skip_ws
    jmp object_member
object_end:
    inc r12
    ret

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, 9
    syscall

    mov rax, r12
    lea rsi, [rip + digit_buffer + 32]
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    test rax, rax
    jne decimal_loop

    lea rdx, [rip + digit_buffer + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov eax, 60
    mov edi, 1
    syscall
