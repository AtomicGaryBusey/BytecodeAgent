.globl _start

.section .bss
.balign 16
input_buffer:
.zero 32769
.balign 16
lines_a:
.zero 16000
lines_b:
.zero 16000

.section .rodata
newline:
.byte 10

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz parse_input
    add r13, rax
    jmp read_loop

parse_input:
    mov rbx, r12
    lea r12, [rip + lines_a]
    xor r8d, r8d
    xor r9d, r9d
    xor r14d, r14d

scan_loop:
    cmp r9, r13
    jae scan_done
    cmp byte ptr [rbx + r9], 10
    jne scan_next

    lea rax, [rbx + r8]
    mov qword ptr [r12 + r14], rax
    mov rax, r9
    sub rax, r8
    mov qword ptr [r12 + r14 + 8], rax
    add r14, 16
    lea r8, [r9 + 1]

scan_next:
    inc r9
    jmp scan_loop

scan_done:
    cmp r8, r13
    jae sort_init
    lea rax, [rbx + r8]
    mov qword ptr [r12 + r14], rax
    mov rax, r13
    sub rax, r8
    mov qword ptr [r12 + r14 + 8], rax
    add r14, 16

sort_init:
    lea r13, [rip + lines_b]
    mov r15d, 16

sort_pass:
    cmp r15, r14
    jae output_init
    xor ebx, ebx
    xor ebp, ebp

run_loop:
    cmp rbx, r14
    jae pass_done

    mov r10, rbx
    add r10, r15
    cmp r10, r14
    cmova r10, r14

    mov r11, r10
    add r11, r15
    cmp r11, r14
    cmova r11, r14

    mov r8, rbx
    mov r9, r10

merge_loop:
    cmp r8, r10
    jae copy_right
    cmp r9, r11
    jae copy_left

    mov rdi, qword ptr [r12 + r8]
    mov rsi, qword ptr [r12 + r9]
    mov rcx, qword ptr [r12 + r8 + 8]
    mov rax, qword ptr [r12 + r9 + 8]
    cmp rcx, rax
    cmova rcx, rax
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jne compare_bytes

compare_lengths:
    mov rax, qword ptr [r12 + r8 + 8]
    cmp rax, qword ptr [r12 + r9 + 8]
    jbe take_left
    jmp take_right

compare_bytes:
    mov al, byte ptr [rdi - 1]
    cmp al, byte ptr [rsi - 1]
    jbe take_left

take_right:
    mov rax, qword ptr [r12 + r9]
    mov qword ptr [r13 + rbp], rax
    mov rax, qword ptr [r12 + r9 + 8]
    mov qword ptr [r13 + rbp + 8], rax
    add r9, 16
    add rbp, 16
    jmp merge_loop

take_left:
    mov rax, qword ptr [r12 + r8]
    mov qword ptr [r13 + rbp], rax
    mov rax, qword ptr [r12 + r8 + 8]
    mov qword ptr [r13 + rbp + 8], rax
    add r8, 16
    add rbp, 16
    jmp merge_loop

copy_left:
    cmp r8, r10
    jae run_done
    mov rax, qword ptr [r12 + r8]
    mov qword ptr [r13 + rbp], rax
    mov rax, qword ptr [r12 + r8 + 8]
    mov qword ptr [r13 + rbp + 8], rax
    add r8, 16
    add rbp, 16
    jmp copy_left

copy_right:
    cmp r9, r11
    jae run_done
    mov rax, qword ptr [r12 + r9]
    mov qword ptr [r13 + rbp], rax
    mov rax, qword ptr [r12 + r9 + 8]
    mov qword ptr [r13 + rbp + 8], rax
    add r9, 16
    add rbp, 16
    jmp copy_right

run_done:
    mov rbx, r11
    jmp run_loop

pass_done:
    xchg r12, r13
    shl r15, 1
    jmp sort_pass

output_init:
    xor ebx, ebx

output_loop:
    cmp rbx, r14
    jae exit_success
    mov rdx, qword ptr [r12 + rbx + 8]
    test rdx, rdx
    jz write_newline
    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [r12 + rbx]
    syscall

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    add rbx, 16
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
