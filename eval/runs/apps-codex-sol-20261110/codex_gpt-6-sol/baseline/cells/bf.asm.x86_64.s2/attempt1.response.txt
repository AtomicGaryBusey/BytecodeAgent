.section .bss
.balign 16
input_buffer:
.zero 65537
.balign 16
bracket_map:
.zero 262144
.balign 16
bracket_stack:
.zero 4000
.balign 16
tape:
.zero 30000
.balign 16
output_buffer:
.zero 65536

.text
.globl _start
_start:
    lea r8, [rip + input_buffer]
    xor ebx, ebx

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + rbx]
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz input_done
    add rbx, rax
    jmp read_more

input_done:
    mov r11, rbx
    mov r13, rbx
    mov r14, rbx
    xor ebx, ebx

find_delimiter:
    cmp rbx, r11
    jae delimiter_done
    cmp byte ptr [r8 + rbx], 33
    je delimiter_found
    inc rbx
    jmp find_delimiter

delimiter_found:
    mov r13, rbx
    lea r14, [rbx + 1]

delimiter_done:
    lea r9, [rip + bracket_map]
    lea rdx, [rip + bracket_stack]
    xor ebx, ebx
    xor ebp, ebp

check_brackets:
    cmp rbx, r13
    jae brackets_checked
    mov al, byte ptr [r8 + rbx]
    cmp al, 91
    je opening_bracket
    cmp al, 93
    je closing_bracket

next_bracket:
    inc rbx
    jmp check_brackets

opening_bracket:
    mov dword ptr [rdx + rbp*4], ebx
    inc rbp
    jmp next_bracket

closing_bracket:
    test rbp, rbp
    jz invalid_program
    dec rbp
    mov eax, dword ptr [rdx + rbp*4]
    lea esi, [rbx + 1]
    mov dword ptr [r9 + rax*4], esi
    lea esi, [rax + 1]
    mov dword ptr [r9 + rbx*4], esi
    jmp next_bracket

brackets_checked:
    test rbp, rbp
    jnz invalid_program

    lea r10, [rip + tape]
    lea r12, [rip + output_buffer]
    xor ebx, ebx
    xor ebp, ebp
    xor r15d, r15d

execute:
    cmp rbx, r13
    jae finish
    mov al, byte ptr [r8 + rbx]
    inc rbx
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment_cell
    cmp al, 45
    je decrement_cell
    cmp al, 46
    je output_cell
    cmp al, 44
    je input_cell
    cmp al, 91
    je loop_start
    cmp al, 93
    je loop_end
    jmp execute

move_right:
    inc rbp
    jmp execute

move_left:
    dec rbp
    jmp execute

increment_cell:
    inc byte ptr [r10 + rbp]
    jmp execute

decrement_cell:
    dec byte ptr [r10 + rbp]
    jmp execute

output_cell:
    mov al, byte ptr [r10 + rbp]
    mov byte ptr [r12 + r15], al
    inc r15
    jmp execute

input_cell:
    cmp r14, r11
    jae input_exhausted
    mov al, byte ptr [r8 + r14]
    inc r14
    mov byte ptr [r10 + rbp], al
    jmp execute

input_exhausted:
    mov byte ptr [r10 + rbp], 0
    jmp execute

loop_start:
    cmp byte ptr [r10 + rbp], 0
    jne execute
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    jmp execute

loop_end:
    cmp byte ptr [r10 + rbp], 0
    je execute
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    jmp execute

finish:
    test r15, r15
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r15
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall
