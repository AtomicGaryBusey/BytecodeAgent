.section .bss
.balign 8
input_buffer:
    .zero 65537
.balign 8
tape:
    .zero 30000
.balign 8
output_buffer:
    .zero 65536
.balign 8
bracket_stack:
    .zero 4000
.balign 8
bracket_matches:
    .zero 262144

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x12, #1
    movk x12, #1, lsl #16

read_more:
    mov x0, #0
    add x1, x19, x20
    sub x2, x12, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_more

read_done:
    mov x21, x20
    mov x9, #0

find_delimiter:
    cmp x9, x20
    b.hs delimiter_done
    ldrb w10, [x19, x9]
    cmp w10, #33
    b.eq delimiter_found
    add x9, x9, #1
    b find_delimiter

delimiter_found:
    mov x21, x9
    add x22, x9, #1
    b prepare_brackets

delimiter_done:
    mov x22, x20

prepare_brackets:
    adrp x16, bracket_stack
    add x16, x16, :lo12:bracket_stack
    adrp x25, bracket_matches
    add x25, x25, :lo12:bracket_matches
    mov x17, #0
    mov x9, #0

check_brackets:
    cmp x9, x21
    b.hs brackets_checked
    ldrb w10, [x19, x9]
    cmp w10, #91
    b.eq opening_bracket
    cmp w10, #93
    b.eq closing_bracket

next_checked_byte:
    add x9, x9, #1
    b check_brackets

opening_bracket:
    str w9, [x16, x17, lsl #2]
    add x17, x17, #1
    b next_checked_byte

closing_bracket:
    cbz x17, invalid_program
    sub x17, x17, #1
    ldr w11, [x16, x17, lsl #2]
    str w11, [x25, x9, lsl #2]
    str w9, [x25, x11, lsl #2]
    b next_checked_byte

brackets_checked:
    cbnz x17, invalid_program
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x27, #0
    mov x28, #0

execute:
    cmp x28, x21
    b.hs write_output
    ldrb w9, [x19, x28]
    cmp w9, #62
    b.eq move_right
    cmp w9, #60
    b.eq move_left
    cmp w9, #43
    b.eq increment_cell
    cmp w9, #45
    b.eq decrement_cell
    cmp w9, #46
    b.eq output_cell
    cmp w9, #44
    b.eq input_cell
    cmp w9, #91
    b.eq loop_start
    cmp w9, #93
    b.eq loop_end

next_command:
    add x28, x28, #1
    b execute

move_right:
    add x23, x23, #1
    b next_command

move_left:
    sub x23, x23, #1
    b next_command

increment_cell:
    ldrb w10, [x23]
    add w10, w10, #1
    strb w10, [x23]
    b next_command

decrement_cell:
    ldrb w10, [x23]
    sub w10, w10, #1
    strb w10, [x23]
    b next_command

output_cell:
    ldrb w10, [x23]
    strb w10, [x26, x27]
    add x27, x27, #1
    b next_command

input_cell:
    cmp x22, x20
    b.hs input_exhausted
    ldrb w10, [x19, x22]
    add x22, x22, #1
    strb w10, [x23]
    b next_command

input_exhausted:
    strb wzr, [x23]
    b next_command

loop_start:
    ldrb w10, [x23]
    cbnz w10, next_command
    ldr w28, [x25, x28, lsl #2]
    b next_command

loop_end:
    ldrb w10, [x23]
    cbz w10, next_command
    ldr w28, [x25, x28, lsl #2]
    b next_command

write_output:
    cbz x27, successful_exit
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0

successful_exit:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0
