.globl _start

.section .bss
.balign 8
input:      .zero 65537
values:     .zero 8008
outbuf:     .zero 32

.section .rodata
error_msg:  .ascii "error\n"

.section .text
_start:
    xor ebx, ebx
read_more:
    lea rsi, [rip + input]
    add rsi, rbx
    mov edx, 65537
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    je read_done
    add rbx, rax
    jmp read_more

read_done:
    lea r12, [rip + input]
    lea r13, [r12 + rbx]
    mov r10, r12
    xor r14d, r14d
    xor r15d, r15d

next_byte:
    cmp r12, r13
    je at_eof
    movzx eax, byte ptr [r12]
    cmp al, 10
    je at_newline
    test r15d, r15d
    jne advance
    cmp al, 32
    je advance
    cmp al, 9
    je advance

    mov rbx, r12
scan_token:
    cmp r12, r13
    je token_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_end
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    inc r12
    jmp scan_token

token_end:
    mov r8, r12
    sub r8, rbx
    cmp r8, 1
    jne parse_literal
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_literal:
    mov r9, rbx
    xor edi, edi
    cmp byte ptr [r9], '-'
    jne digits_ready
    mov edi, 1
    inc r9
    dec r8
digits_ready:
    test r8, r8
    je mark_error
    cmp r8, 18
    ja mark_error
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [r9]
    sub edx, '0'
    cmp edx, 9
    ja mark_error
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc r9
    dec r8
    jne digit_loop
    test edi, edi
    je push_literal
    neg rax
push_literal:
    lea r9, [rip + values]
    mov qword ptr [r9 + r14*8], rax
    inc r14
    jmp next_byte

operator:
    cmp r14, 2
    jb mark_error
    lea r9, [rip + values]
    dec r14
    mov rdi, qword ptr [r9 + r14*8]
    dec r14
    mov rax, qword ptr [r9 + r14*8]
    movzx ecx, byte ptr [rbx]
    cmp cl, '+'
    je do_add
    cmp cl, '-'
    je do_sub
    cmp cl, '*'
    je do_mul
    test rdi, rdi
    je mark_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_div
    cmp rdi, -1
    je mark_error
do_div:
    cqo
    idiv rdi
    cmp cl, '%'
    jne push_result
    mov rax, rdx
    jmp push_result
do_add:
    add rax, rdi
    jmp push_result
do_sub:
    sub rax, rdi
    jmp push_result
do_mul:
    imul rax, rdi
push_result:
    mov qword ptr [r9 + r14*8], rax
    inc r14
    jmp next_byte

mark_error:
    mov r15d, 1
    jmp next_byte

advance:
    inc r12
    jmp next_byte

at_newline:
    call print_result
    inc r12
    mov r10, r12
    xor r14d, r14d
    xor r15d, r15d
    jmp next_byte

at_eof:
    cmp r10, r13
    je exit_ok
    call print_result
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

print_result:
    test r15d, r15d
    jne print_error
    cmp r14, 1
    jne print_error
    lea r9, [rip + values]
    mov rax, qword ptr [r9]
    xor r8d, r8d
    test rax, rax
    jns magnitude_ready
    mov r8d, 1
    neg rax
magnitude_ready:
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ecx, 10
convert_digit:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne convert_digit
    test r8d, r8d
    je write_number
    dec rsi
    mov byte ptr [rsi], '-'
write_number:
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall
    ret
