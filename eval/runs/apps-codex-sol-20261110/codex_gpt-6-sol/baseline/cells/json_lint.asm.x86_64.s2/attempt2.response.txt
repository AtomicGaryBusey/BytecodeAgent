.globl _start

.section .bss
.align 16
input:
.zero 65536
num_buf:
.zero 32

.section .rodata
ok_msg:
.ascii "ok\n"
error_msg:
.ascii "error at "
lit_true:
.ascii "true"
lit_false:
.ascii "false"
lit_null:
.ascii "null"

.section .text
_start:
    lea r14, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne parse_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp r12, r13
    jae ws_done
    mov al, byte ptr [r14 + r12]
    cmp al, 0x20
    je ws_advance
    cmp al, 0x09
    je ws_advance
    cmp al, 0x0a
    je ws_advance
    cmp al, 0x0d
    jne ws_done
ws_advance:
    inc r12
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe parse_number
    jmp parse_error

value_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + lit_null]
    mov edx, 4

parse_literal:
    test edx, edx
    jz literal_done
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc r12
    inc rsi
    dec edx
    jmp parse_literal
literal_done:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae parse_error
    cmp byte ptr [r14 + r12], ']'
    je array_close
array_item:
    call parse_value
    call skip_ws
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, ']'
    je array_close
    cmp al, ','
    jne parse_error
    inc r12
    call skip_ws
    jmp array_item
array_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae parse_error
    cmp byte ptr [r14 + r12], '}'
    je object_close
object_member:
    cmp r12, r13
    jae parse_error
    cmp byte ptr [r14 + r12], '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp r12, r13
    jae parse_error
    cmp byte ptr [r14 + r12], ':'
    jne parse_error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '}'
    je object_close
    cmp al, ','
    jne parse_error
    inc r12
    call skip_ws
    jmp object_member
object_close:
    inc r12
    ret

parse_string:
    inc r12
string_next:
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '"'
    je string_close
    cmp al, 0x20
    jb parse_error
    cmp al, 0x5c
    je string_escape
    inc r12
    jmp string_next
string_close:
    inc r12
    ret

string_escape:
    inc r12
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '"'
    je escape_done
    cmp al, 0x5c
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    je escape_done
    cmp al, 'u'
    jne parse_error
    inc r12
    mov ebx, 4
hex_next:
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe hex_done
    cmp al, 'A'
    jb parse_error
    cmp al, 'F'
    jbe hex_done
    cmp al, 'a'
    jb parse_error
    cmp al, 'f'
    ja parse_error
hex_done:
    inc r12
    dec ebx
    jnz hex_next
    jmp string_next
escape_done:
    inc r12
    jmp string_next

parse_number:
    cmp byte ptr [r14 + r12], '-'
    jne number_int
    inc r12
    cmp r12, r13
    jae parse_error
number_int:
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error
number_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_after_int
    cmp al, '9'
    jbe number_digits
    jmp number_after_int
number_zero:
    inc r12
number_after_int:
    cmp r12, r13
    jae number_done
    cmp byte ptr [r14 + r12], '.'
    jne number_exp
    inc r12
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
frac_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_exp
    cmp al, '9'
    jbe frac_digits
number_exp:
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, 'e'
    je exp_start
    cmp al, 'E'
    jne number_done
exp_start:
    inc r12
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
    cmp al, '+'
    je exp_sign
    cmp al, '-'
    jne exp_first
exp_sign:
    inc r12
    cmp r12, r13
    jae parse_error
    mov al, byte ptr [r14 + r12]
exp_first:
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
exp_digits:
    inc r12
    cmp r12, r13
    jae number_done
    mov al, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exp_digits
number_done:
    ret

parse_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 9
    syscall

    lea rsi, [rip + num_buf + 31]
    mov byte ptr [rsi], 0x0a
    mov ecx, 1
    mov rax, r12
    mov ebx, 10
decimal_digit:
    xor edx, edx
    div rbx
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    inc rcx
    test rax, rax
    jnz decimal_digit

    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
