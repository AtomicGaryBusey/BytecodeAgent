.section .text
.globl _start
_start:
    adrp x22, line_buffer
    add x22, x22, :lo12:line_buffer
    adrp x23, read_buffer
    add x23, x23, :lo12:read_buffer
    mov x21, #0

read_more:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done
    mov x19, x23
    mov x20, x0

consume_byte:
    ldrb w9, [x19], #1
    cmp w9, #10
    b.eq line_complete
    strb w9, [x22, x21]
    add x21, x21, #1
    b next_byte

line_complete:
    mov x0, x22
    mov x1, x21
    bl evaluate
    mov x21, #0

next_byte:
    subs x20, x20, #1
    b.ne consume_byte
    b read_more

input_done:
    cbz x21, exit_success
    mov x0, x22
    mov x1, x21
    bl evaluate

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

evaluate:
    stp x29, x30, [sp, #-16]!
    mov x28, x0
    add x25, x0, x1
    adrp x26, value_stack
    add x26, x26, :lo12:value_stack
    mov x27, #0

skip_separators:
    cmp x28, x25
    b.eq tokens_done
    ldrb w9, [x28]
    cmp w9, #32
    b.eq skip_one
    cmp w9, #9
    b.ne token_start
skip_one:
    add x28, x28, #1
    b skip_separators

token_start:
    mov x11, x28
token_scan:
    cmp x28, x25
    b.eq token_end
    ldrb w10, [x28]
    cmp w10, #32
    b.eq token_end
    cmp w10, #9
    b.eq token_end
    add x28, x28, #1
    b token_scan

token_end:
    sub x12, x28, x11
    ldrb w9, [x11]
    cmp x12, #1
    b.ne parse_literal
    cmp w9, #43
    b.eq apply_operator
    cmp w9, #45
    b.eq apply_operator
    cmp w9, #42
    b.eq apply_operator
    cmp w9, #47
    b.eq apply_operator
    cmp w9, #37
    b.eq apply_operator

parse_literal:
    mov x13, x11
    mov x14, #0
    mov x16, #10
    cmp w9, #45
    b.ne literal_length
    add x13, x13, #1
    sub x12, x12, #1
literal_length:
    cbz x12, evaluation_error
    cmp x12, #18
    b.hi evaluation_error
    add x18, x13, x12

literal_digit:
    ldrb w15, [x13], #1
    sub w15, w15, #48
    cmp w15, #9
    b.hi evaluation_error
    madd x14, x14, x16, x15
    cmp x13, x18
    b.ne literal_digit
    cmp w9, #45
    b.ne literal_push
    neg x14, x14

literal_push:
    str x14, [x26, x27, lsl #3]
    add x27, x27, #1
    b skip_separators

apply_operator:
    cmp x27, #2
    b.lo evaluation_error
    sub x27, x27, #2
    ldr x13, [x26, x27, lsl #3]
    add x16, x27, #1
    ldr x14, [x26, x16, lsl #3]
    cmp w9, #43
    b.eq op_add
    cmp w9, #45
    b.eq op_sub
    cmp w9, #42
    b.eq op_mul
    cbz x14, evaluation_error
    movz x16, #0x8000, lsl #48
    cmp x13, x16
    b.ne op_divide
    cmn x14, #1
    b.eq evaluation_error

op_divide:
    sdiv x15, x13, x14
    cmp w9, #47
    b.eq operator_push
    msub x15, x15, x14, x13
    b operator_push

op_add:
    add x15, x13, x14
    b operator_push

op_sub:
    sub x15, x13, x14
    b operator_push

op_mul:
    mul x15, x13, x14

operator_push:
    str x15, [x26, x27, lsl #3]
    add x27, x27, #1
    b skip_separators

tokens_done:
    cmp x27, #1
    b.ne evaluation_error
    ldr x0, [x26]
    bl print_number
    ldp x29, x30, [sp], #16
    ret

evaluation_error:
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret

print_number:
    adrp x15, output_buffer
    add x15, x15, :lo12:output_buffer
    add x15, x15, #32
    mov x1, x15
    mov w10, #10
    strb w10, [x1, #-1]!
    cmp x0, #0
    cset w14, mi
    cneg x9, x0, mi
    mov x11, #10

number_digit:
    udiv x12, x9, x11
    msub x13, x12, x11, x9
    add w13, w13, #48
    strb w13, [x1, #-1]!
    mov x9, x12
    cbnz x9, number_digit
    cbz w14, number_write
    mov w13, #45
    strb w13, [x1, #-1]!

number_write:
    sub x2, x15, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
error_text:
    .ascii "error\n"

.section .bss
.balign 8
read_buffer:
    .space 4096
line_buffer:
    .space 1008
.balign 8
value_stack:
    .space 8008
output_buffer:
    .space 32
