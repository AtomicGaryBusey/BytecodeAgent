.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x19, x21
    movz x2, #1, lsl #16
    add x2, x2, #1
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x22, x21
    b.eq final_line
    ldrb w9, [x19, x22]
    cmp w9, #10
    b.eq complete_line
    add x22, x22, #1
    b scan_loop

complete_line:
    mov x25, x22

reverse_line:
    cmp x25, x23
    b.eq append_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x20, x24]
    add x24, x24, #1
    b reverse_line

append_newline:
    mov w9, #10
    strb w9, [x20, x24]
    add x24, x24, #1
    add x22, x22, #1
    mov x23, x22
    b scan_loop

final_line:
    cmp x23, x21
    b.eq write_output
    mov x25, x21

reverse_final:
    cmp x25, x23
    b.eq append_final_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x20, x24]
    add x24, x24, #1
    b reverse_final

append_final_newline:
    mov w9, #10
    strb w9, [x20, x24]
    add x24, x24, #1

write_output:
    cbz x24, exit_success
    mov x0, #1
    mov x1, x20
    mov x2, x24
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero 65537
output_buf:
    .zero 65537
