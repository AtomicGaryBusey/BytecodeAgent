.section .bss
.balign 16
input_buf:
    .space 4096
output_buf:
    .space 64

.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // inside a word

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, finish

    add x21, x21, x0
    mov x24, x0
    mov x25, x23

byte_loop:
    ldrb w3, [x25], #1
    cmp w3, #10
    b.ne check_space
    add x19, x19, #1

check_space:
    cmp w3, #32
    b.eq whitespace
    cmp w3, #9
    b.lo word_byte
    cmp w3, #13
    b.ls whitespace

word_byte:
    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

whitespace:
    mov x22, #0

next_byte:
    subs x24, x24, #1
    b.ne byte_loop
    b read_loop

finish:
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    add x26, x26, #64
    mov x27, x26

    mov w5, #10
    strb w5, [x26, #-1]!

    mov x0, x21
    bl decimal_backward
    mov w5, #32
    strb w5, [x26, #-1]!

    mov x0, x20
    bl decimal_backward
    mov w5, #32
    strb w5, [x26, #-1]!

    mov x0, x19
    bl decimal_backward

    mov x0, #1
    mov x1, x26
    sub x2, x27, x26
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

decimal_backward:
    mov x4, #10
1:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    strb w5, [x26, #-1]!
    mov x0, x3
    cbnz x0, 1b
    ret
