.text
.globl _start

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x21, x20
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x19, #0
    bl parse_value
    cbnz x0, report_error
    bl skip_ws
    cmp x19, x20
    b.ne report_error

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

report_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x9, output_buf
    add x9, x9, :lo12:output_buf
    add x5, x9, #32
    mov w7, #10
    strb w7, [x5]
    mov x3, x19
    mov x4, #10

decimal_loop:
    udiv x6, x3, x4
    msub x7, x6, x4, x3
    sub x5, x5, #1
    add w7, w7, #48
    strb w7, [x5]
    mov x3, x6
    cbnz x3, decimal_loop

    mov x0, #1
    mov x1, x5
    add x2, x9, #33
    sub x2, x2, x5
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// get_byte returns 256 at end of input without advancing x19.
get_byte:
    cmp x19, x20
    b.hs get_eof
    ldrb w0, [x21, x19]
    ret
get_eof:
    mov w0, #256
    ret

skip_ws:
    stp x29, x30, [sp, #-16]!
skip_ws_loop:
    bl get_byte
    cmp w0, #32
    b.eq skip_ws_advance
    cmp w0, #9
    b.eq skip_ws_advance
    cmp w0, #10
    b.eq skip_ws_advance
    cmp w0, #13
    b.eq skip_ws_advance
    ldp x29, x30, [sp], #16
    ret
skip_ws_advance:
    add x19, x19, #1
    b skip_ws_loop

parse_value:
    stp x29, x30, [sp, #-16]!
    bl skip_ws
    bl get_byte
    cmp w0, #123
    b.eq value_object
    cmp w0, #91
    b.eq value_array
    cmp w0, #34
    b.eq value_string
    cmp w0, #116
    b.eq value_true
    cmp w0, #102
    b.eq value_false
    cmp w0, #110
    b.eq value_null
    cmp w0, #45
    b.eq value_number
    sub w1, w0, #48
    cmp w1, #9
    b.ls value_number
    mov x0, #1
    b value_return

value_object:
    bl parse_object
    b value_return
value_array:
    bl parse_array
    b value_return
value_string:
    bl parse_string
    b value_return
value_number:
    bl parse_number
    b value_return
value_true:
    adrp x1, true_word
    add x1, x1, :lo12:true_word
    mov x2, #4
    bl parse_literal
    b value_return
value_false:
    adrp x1, false_word
    add x1, x1, :lo12:false_word
    mov x2, #5
    bl parse_literal
    b value_return
value_null:
    adrp x1, null_word
    add x1, x1, :lo12:null_word
    mov x2, #4
    bl parse_literal
value_return:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    stp x29, x30, [sp, #-16]!
literal_loop:
    bl get_byte
    ldrb w3, [x1]
    cmp w0, w3
    b.ne literal_error
    add x19, x19, #1
    add x1, x1, #1
    subs x2, x2, #1
    b.ne literal_loop
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
literal_error:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    bl get_byte
    cmp w0, #125
    b.eq object_close

object_member:
    cmp w0, #34
    b.ne object_error
    bl parse_string
    cbnz x0, object_error
    bl skip_ws
    bl get_byte
    cmp w0, #58
    b.ne object_error
    add x19, x19, #1
    bl parse_value
    cbnz x0, object_error
    bl skip_ws
    bl get_byte
    cmp w0, #125
    b.eq object_close
    cmp w0, #44
    b.ne object_error
    add x19, x19, #1
    bl skip_ws
    bl get_byte
    b object_member

object_close:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
object_error:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    bl get_byte
    cmp w0, #93
    b.eq array_close

array_element:
    bl parse_value
    cbnz x0, array_error
    bl skip_ws
    bl get_byte
    cmp w0, #93
    b.eq array_close
    cmp w0, #44
    b.ne array_error
    add x19, x19, #1
    b array_element

array_close:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
array_error:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_string:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
string_loop:
    bl get_byte
    cmp w0, #34
    b.eq string_close
    cmp w0, #32
    b.lo string_error
    cmp w0, #256
    b.eq string_error
    cmp w0, #92
    b.eq string_escape
    add x19, x19, #1
    b string_loop

string_escape:
    add x19, x19, #1
    bl get_byte
    cmp w0, #117
    b.eq string_unicode
    cmp w0, #34
    b.eq string_escaped_char
    cmp w0, #92
    b.eq string_escaped_char
    cmp w0, #47
    b.eq string_escaped_char
    cmp w0, #98
    b.eq string_escaped_char
    cmp w0, #102
    b.eq string_escaped_char
    cmp w0, #110
    b.eq string_escaped_char
    cmp w0, #114
    b.eq string_escaped_char
    cmp w0, #116
    b.eq string_escaped_char
    b string_error

string_escaped_char:
    add x19, x19, #1
    b string_loop

string_unicode:
    add x19, x19, #1
    mov x10, #4
string_hex_loop:
    bl get_byte
    sub w11, w0, #48
    cmp w11, #9
    b.ls string_hex_valid
    sub w11, w0, #65
    cmp w11, #5
    b.ls string_hex_valid
    sub w11, w0, #97
    cmp w11, #5
    b.hi string_error
string_hex_valid:
    add x19, x19, #1
    subs x10, x10, #1
    b.ne string_hex_loop
    b string_loop

string_close:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
string_error:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_number:
    stp x29, x30, [sp, #-16]!
    bl get_byte
    cmp w0, #45
    b.ne number_first_digit
    add x19, x19, #1
    bl get_byte

number_first_digit:
    cmp w0, #48
    b.eq number_zero
    sub w1, w0, #48
    cmp w1, #1
    b.lo number_error
    cmp w1, #9
    b.hi number_error
    add x19, x19, #1

number_int_loop:
    bl get_byte
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_after_int
    add x19, x19, #1
    b number_int_loop

number_zero:
    add x19, x19, #1
    bl get_byte

number_after_int:
    cmp w0, #46
    b.ne number_check_exp
    add x19, x19, #1

number_frac_loop:
    bl get_byte
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_error
    add x19, x19, #1

number_frac_more:
    bl get_byte
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_check_exp
    add x19, x19, #1
    b number_frac_more

number_check_exp:
    cmp w0, #101
    b.eq number_exp
    cmp w0, #69
    b.ne number_success

number_exp:
    add x19, x19, #1
    bl get_byte
    cmp w0, #43
    b.eq number_exp_sign
    cmp w0, #45
    b.ne number_exp_digit
number_exp_sign:
    add x19, x19, #1
    bl get_byte

number_exp_digit:
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_error
    add x19, x19, #1

number_exp_more:
    bl get_byte
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_success
    add x19, x19, #1
    b number_exp_more

number_success:
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
number_error:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
true_word:
    .ascii "true"
false_word:
    .ascii "false"
null_word:
    .ascii "null"

.bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 48
