.section .bss
.balign 16
input_buffer:
    .space 4096
output_buffer:
    .space 64

.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // inside a word
    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output_counts

    add x21, x21, x0
    mov x3, x23
    add x4, x23, x0

scan_loop:
    ldrb w5, [x3], #1
    cmp w5, #10
    b.ne check_whitespace
    add x19, x19, #1

check_whitespace:
    cmp w5, #32
    b.eq whitespace
    sub w6, w5, #9
    cmp w6, #4
    b.ls whitespace
    cbnz x22, scan_next
    add x20, x20, #1
    mov x22, #1
    b scan_next

whitespace:
    mov x22, #0

scan_next:
    cmp x3, x4
    b.lo scan_loop
    b read_loop

output_counts:
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x24, x26

    mov x0, x19
    bl append_decimal
    mov w5, #32
    strb w5, [x24], #1

    mov x0, x20
    bl append_decimal
    mov w5, #32
    strb w5, [x24], #1

    mov x0, x21
    bl append_decimal
    mov w5, #10
    strb w5, [x24], #1

    mov x0, #1
    mov x1, x26
    sub x2, x24, x26
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// Append x0 in decimal to the buffer at x24.
append_decimal:
    mov x2, x24
    mov x3, #10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    strb w5, [x24], #1
    mov x0, x4
    cbnz x0, 1b

    sub x1, x24, #1
2:
    cmp x2, x1
    b.hs 3f
    ldrb w4, [x2]
    ldrb w5, [x1]
    strb w5, [x2], #1
    strb w4, [x1]
    sub x1, x1, #1
    b 2b
3:
    ret
