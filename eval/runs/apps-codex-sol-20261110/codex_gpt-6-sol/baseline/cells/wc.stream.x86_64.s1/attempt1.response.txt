.section .bss
buffer:
    .zero 8192
digits:
    .zero 32
output:
    .zero 128

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jz print_counts

    add r14, rax
    lea rbx, [rip + buffer]
    lea r8, [rbx + rax]

scan_loop:
    cmp rbx, r8
    je read_loop
    movzx eax, byte ptr [rbx]
    inc rbx

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz scan_loop
    inc r13
    mov r15d, 1
    jmp scan_loop

whitespace:
    xor r15d, r15d
    jmp scan_loop

print_counts:
    lea rdi, [rip + output]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

append_decimal:
    lea rsi, [rip + digits + 32]
    mov r9d, 10

digit_loop:
    xor edx, edx
    div r9
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jnz digit_loop

    lea rcx, [rip + digits + 32]
    sub rcx, rsi
    rep movsb
    ret
