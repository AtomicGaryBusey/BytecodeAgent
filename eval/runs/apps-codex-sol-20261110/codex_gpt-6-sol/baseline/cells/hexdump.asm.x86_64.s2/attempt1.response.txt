.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
.zero 65537
output_buffer:
.zero 324000

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + hex_digits]
    lea rdi, [rip + output_buffer]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16

row_size_ready:
    call emit_offset
    mov word ptr [rdi], 0x2020
    add rdi, 2

    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov r10, rsi
    xor r8d, r8d

hex_slot:
    cmp r8, r14
    jae empty_slot

    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], ' '
    inc rsi
    jmp slot_done

empty_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], ' '

slot_done:
    add rdi, 3
    inc r8
    cmp r8, 8
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi

no_middle_space:
    cmp r8, 16
    jb hex_slot

    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor r8d, r8d

ascii_loop:
    cmp r8, r14
    jae ascii_done
    mov al, byte ptr [r10 + r8]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable

nonprintable:
    mov al, '.'

printable:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r14
    jmp row_loop

final_offset:
    call emit_offset
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov ecx, 28

offset_digit:
    mov rax, r13
    shr rax, cl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub ecx, 4
    jns offset_digit
    ret
