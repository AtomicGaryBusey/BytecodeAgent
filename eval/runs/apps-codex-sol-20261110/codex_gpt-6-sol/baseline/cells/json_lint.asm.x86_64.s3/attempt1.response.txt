.globl _start

.section .bss
buffer:
    .zero 65537
outbuf:
    .zero 32

.section .rodata
ok_msg:
    .ascii "ok\n"
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"
error_text:
    .ascii "error at "

.section .text
_start:
    lea r14, [rip + buffer]
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop

read_done:
    mov r12, r14
    lea r13, [r14 + r15]
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne fail

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    cmp r12, r13
    je ws_done
    mov al, byte ptr [r12]
    cmp al, 0x20
    je ws_advance
    cmp al, 9
    je ws_advance
    cmp al, 10
    je ws_advance
    cmp al, 13
    jne ws_done
ws_advance:
    inc r12
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe parse_number
    jmp fail

value_true:
    lea rsi, [rip + true_text]
    mov ecx, 4
    jmp match_literal
value_false:
    lea rsi, [rip + false_text]
    mov ecx, 5
    jmp match_literal
value_null:
    lea rsi, [rip + null_text]
    mov ecx, 4

match_literal:
    test ecx, ecx
    je literal_done
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, byte ptr [rsi]
    jne fail
    inc r12
    inc rsi
    dec ecx
    jmp match_literal
literal_done:
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], '}'
    je object_close
object_member:
    cmp byte ptr [r12], '"'
    jne fail
    call parse_string
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], ':'
    jne fail
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, '}'
    je object_close
    cmp al, ','
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    jmp object_member
object_close:
    inc r12
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], ']'
    je array_close
array_element:
    call parse_value
    call skip_ws
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, ']'
    je array_close
    cmp al, ','
    jne fail
    inc r12
    call skip_ws
    jmp array_element
array_close:
    inc r12
    ret

parse_string:
    inc r12
string_next:
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, 0x20
    jb fail
    cmp al, '"'
    je string_close
    cmp al, 0x5c
    je string_escape
    inc r12
    jmp string_next
string_close:
    inc r12
    ret
string_escape:
    inc r12
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, 'u'
    je string_unicode
    cmp al, '"'
    je escape_done
    cmp al, 0x5c
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    jne fail
escape_done:
    inc r12
    jmp string_next
string_unicode:
    inc r12
    mov ecx, 4
unicode_digit:
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe unicode_advance
    cmp al, 'A'
    jb fail
    cmp al, 'F'
    jbe unicode_advance
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
unicode_advance:
    inc r12
    dec ecx
    jne unicode_digit
    jmp string_next

parse_number:
    cmp byte ptr [r12], '-'
    jne number_integer
    inc r12
    cmp r12, r13
    je fail
number_integer:
    mov al, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb fail
    cmp al, '9'
    ja fail
    inc r12
number_digits:
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    ja number_suffix
    inc r12
    jmp number_digits
number_zero:
    inc r12
number_suffix:
    cmp r12, r13
    je number_done
    cmp byte ptr [r12], '.'
    jne number_exponent
    inc r12
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
fraction_digits:
    inc r12
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    jbe fraction_digits
number_exponent:
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_done
exponent_start:
    inc r12
    cmp r12, r13
    je fail
    mov al, byte ptr [r12]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc r12
    cmp r12, r13
    je fail
exponent_first:
    mov al, byte ptr [r12]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
exponent_digits:
    inc r12
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits
number_done:
    ret

fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_text]
    mov edx, 9
    syscall

    mov rax, r12
    sub rax, r14
    lea r8, [rip + outbuf + 31]
    mov byte ptr [r8], 10
    mov r9, r8
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    dec r9
    add dl, '0'
    mov byte ptr [r9], dl
    test rax, rax
    jne decimal_loop

    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    sub rdx, r9
    inc rdx
    syscall

    mov eax, 60
    mov edi, 1
    syscall
