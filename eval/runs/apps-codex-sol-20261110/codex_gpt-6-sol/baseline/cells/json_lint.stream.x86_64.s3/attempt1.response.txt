.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
true_lit:
    .ascii "true"
false_lit:
    .ascii "false"
null_lit:
    .ascii "null"

.bss
buffer:
    .space 65537
err_digits:
    .space 32

.text
_start:
    xor r12d, r12d
    lea r13, [rip + buffer]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jne fail

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall

    mov rax, rbx
    mov r8d, 10
    lea rsi, [rip + err_digits + 20]
    mov byte ptr [rsi], 10
decimal_loop:
    xor edx, edx
    div r8
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    test rax, rax
    jne decimal_loop

    lea rdx, [rip + err_digits + 21]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ' '
    je ws_advance
    cmp al, 9
    je ws_advance
    cmp al, 10
    je ws_advance
    cmp al, 13
    je ws_advance
    ret
ws_advance:
    inc rbx
    jmp skip_ws

parse_value:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe parse_number
    jmp fail

value_true:
    lea rsi, [rip + true_lit]
    mov ecx, 4
    jmp expect_literal
value_false:
    lea rsi, [rip + false_lit]
    mov ecx, 5
    jmp expect_literal
value_null:
    lea rsi, [rip + null_lit]
    mov ecx, 4

expect_literal:
    mov al, byte ptr [r13 + rbx]
    cmp al, byte ptr [rsi]
    jne fail
    inc rbx
    inc rsi
    dec ecx
    jne expect_literal
    ret

parse_string:
    inc rbx
string_loop:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb fail
    cmp al, 0x5c
    je string_escape
    inc rbx
    jmp string_loop

string_end:
    inc rbx
    ret

string_escape:
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je escape_end
    cmp al, 0x5c
    je escape_end
    cmp al, '/'
    je escape_end
    cmp al, 'b'
    je escape_end
    cmp al, 'f'
    je escape_end
    cmp al, 'n'
    je escape_end
    cmp al, 'r'
    je escape_end
    cmp al, 't'
    je escape_end
    cmp al, 'u'
    jne fail
    inc rbx
    mov ecx, 4
hex_loop:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe hex_valid
    cmp al, 'A'
    jb fail
    cmp al, 'F'
    jbe hex_valid
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
hex_valid:
    inc rbx
    dec ecx
    jne hex_loop
    jmp string_loop

escape_end:
    inc rbx
    jmp string_loop

parse_number:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '-'
    jne number_integer
    inc rbx
number_integer:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb fail
    cmp al, '9'
    ja fail
number_digits:
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb number_fraction
    cmp al, '9'
    jbe number_digits
    jmp number_fraction

number_zero:
    inc rbx
number_fraction:
    cmp byte ptr [r13 + rbx], '.'
    jne number_exponent
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
fraction_digits:
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    jbe fraction_digits

number_exponent:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_end
exponent_start:
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc rbx
exponent_first:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
exponent_digits:
    inc rbx
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb number_end
    cmp al, '9'
    jbe exponent_digits
number_end:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp byte ptr [r13 + rbx], ']'
    je array_end
array_element:
    call parse_value
    call skip_ws
    mov al, byte ptr [r13 + rbx]
    cmp al, ']'
    je array_end
    cmp al, ','
    jne fail
    inc rbx
    call skip_ws
    jmp array_element
array_end:
    inc rbx
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp byte ptr [r13 + rbx], '}'
    je object_end
object_member:
    cmp byte ptr [r13 + rbx], '"'
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [r13 + rbx], ':'
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    mov al, byte ptr [r13 + rbx]
    cmp al, '}'
    je object_end
    cmp al, ','
    jne fail
    inc rbx
    call skip_ws
    jmp object_member
object_end:
    inc rbx
    ret
