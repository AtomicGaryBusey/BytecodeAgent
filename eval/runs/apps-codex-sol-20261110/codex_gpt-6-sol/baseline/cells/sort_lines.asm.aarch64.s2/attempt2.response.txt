.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x21, descriptors
    add x21, x21, :lo12:descriptors
    adrp x25, pointers
    add x25, x25, :lo12:pointers
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #32768
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x24, x19
    b.hs scan_done
    ldrb w9, [x20, x24]
    cmp w9, #10
    b.ne scan_next
    add x9, x21, x22, lsl #4
    add x10, x20, x23
    str x10, [x9]
    sub x10, x24, x23
    str x10, [x9, #8]
    add x22, x22, #1
    add x23, x24, #1

scan_next:
    add x24, x24, #1
    b scan_loop

scan_done:
    cmp x23, x19
    b.hs init_pointers
    add x9, x21, x22, lsl #4
    add x10, x20, x23
    str x10, [x9]
    sub x10, x19, x23
    str x10, [x9, #8]
    add x22, x22, #1

init_pointers:
    mov x26, #0

init_loop:
    cmp x26, x22
    b.hs sort_start
    add x9, x21, x26, lsl #4
    str x9, [x25, x26, lsl #3]
    add x26, x26, #1
    b init_loop

sort_start:
    mov x26, #1

sort_outer:
    cmp x26, x22
    b.hs output_start
    ldr x27, [x25, x26, lsl #3]
    mov x28, x26

sort_inner:
    cbz x28, sort_insert
    sub x9, x28, #1
    ldr x10, [x25, x9, lsl #3]
    mov x0, x10
    mov x1, x27
    bl greater
    cbz x0, sort_insert
    str x10, [x25, x28, lsl #3]
    sub x28, x28, #1
    b sort_inner

sort_insert:
    str x27, [x25, x28, lsl #3]
    add x26, x26, #1
    b sort_outer

// Return 1 if the line at x0 is greater than the line at x1.
greater:
    ldr x2, [x0]
    ldr x3, [x0, #8]
    ldr x4, [x1]
    ldr x5, [x1, #8]
    mov x6, #0

compare_loop:
    cmp x6, x3
    b.hs compare_lengths
    cmp x6, x5
    b.hs compare_lengths
    ldrb w7, [x2, x6]
    ldrb w8, [x4, x6]
    cmp w7, w8
    b.hi compare_greater
    b.lo compare_less
    add x6, x6, #1
    b compare_loop

compare_lengths:
    cmp x3, x5
    b.hi compare_greater

compare_less:
    mov x0, #0
    ret

compare_greater:
    mov x0, #1
    ret

output_start:
    mov x26, #0

output_loop:
    cmp x26, x22
    b.hs finish
    ldr x24, [x25, x26, lsl #3]
    ldr x2, [x24, #8]
    cbz x2, output_newline
    mov x0, #1
    ldr x1, [x24]
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x26, x26, #1
    b output_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buf:
    .space 32768
.balign 16
descriptors:
    .space 16000
.balign 16
pointers:
    .space 8000
