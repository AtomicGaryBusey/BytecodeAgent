.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 49153
.balign 16
output_buffer:
    .zero 66400

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #49153
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    mov x22, x21
    mov x23, #0
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet
    mov x25, x20

encode_loop:
    cmp x25, #3
    b.lo encode_tail

    ldrb w4, [x19], #1
    ldrb w5, [x19], #1
    ldrb w6, [x19], #1

    lsr w7, w4, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w4, #3
    lsl w7, w7, #4
    orr w7, w7, w5, lsr #4
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w5, #15
    lsl w7, w7, #2
    orr w7, w7, w6, lsr #6
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w6, #63
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    sub x25, x25, #3
    add x23, x23, #4
    cmp x23, #76
    b.ne encode_loop
    mov w9, #10
    strb w9, [x22], #1
    mov x23, #0
    b encode_loop

encode_tail:
    cbz x25, finish_lines

    ldrb w4, [x19], #1
    lsr w7, w4, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w4, #3
    lsl w7, w7, #4
    cmp x25, #1
    b.eq one_byte
    ldrb w5, [x19]
    orr w7, w7, w5, lsr #4

one_byte:
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1
    cmp x25, #1
    b.eq padding

    and w7, w5, #15
    lsl w7, w7, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1
    b last_padding

padding:
    mov w9, #61
    strb w9, [x22], #1

last_padding:
    mov w9, #61
    strb w9, [x22], #1
    add x23, x23, #4

finish_lines:
    cbz x23, write_output
    mov w9, #10
    strb w9, [x22], #1

write_output:
    sub x2, x22, x21
    cbz x23, write_if_nonempty
    add x2, x2, #1

write_if_nonempty:
    cbz x2, exit_success
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0
