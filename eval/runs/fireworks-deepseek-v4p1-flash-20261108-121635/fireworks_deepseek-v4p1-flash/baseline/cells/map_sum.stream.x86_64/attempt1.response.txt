map_sum:
    push rbx
    push rbp
    push r12
    push r13
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    xor r13d, r13d
    test rbp, rbp
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [rbx]
    call r12
    add r13, rax
    add rbx, 8
    dec rbp
    jne .Lloop
.Ldone:
    mov rax, r13
    add rsp, 8
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
