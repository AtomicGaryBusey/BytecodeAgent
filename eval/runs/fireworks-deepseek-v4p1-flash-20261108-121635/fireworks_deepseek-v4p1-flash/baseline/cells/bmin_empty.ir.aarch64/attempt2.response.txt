export function l $bmin_empty(l %a, l %n) {
@start
  %is_zero =w ceql %n, 0
  jnz %is_zero, @empty, @init

@empty
  %max =l copy 9223372036854775807
  ret %max

@init
  %first =l loadl %a
  %i0 =l copy 1
  jmp @loop

@loop
  %i =l phi @init %i0, @upd %inext_upd, @skip %inext_skip
  %min =l phi @init %first, @upd %min_upd, @skip %min_skip
  %cmp =w csltl %i, %n
  jnz %cmp, @body, @done

@body
  %offset =l mul %i, 8
  %addr =l add %a, %offset
  %val =l loadl %addr
  %less =w csltl %val, %min
  jnz %less, @upd, @skip

@upd
  %inext_upd =l add %i, 1
  %min_upd =l copy %val
  jmp @loop

@skip
  %inext_skip =l add %i, 1
  %min_skip =l copy %min
  jmp @loop

@done
  ret %min
}
