export function l $bmin_empty(l %a, l %n) {
@start
  %is_zero =w ceql %n, 0
  jnz %is_zero, @empty, @init

@empty
  %max =l copy 9223372036854775807
  ret %max

@init
  %first =l loadl %a
  %i0 =l copy 1
  jmp @loop

@loop
  %i =l phi @init %i0, @body %inext
  %min =l phi @init %first, @body %newmin
  %cmp =w csltl %i, %n
  jnz %cmp, @body, @done

@body
  %offset =l mul %i, 8
  %addr =l add %a, %offset
  %val =l loadl %addr
  %less =w csltl %val, %min
  %newmin =l select %less, %val, %min
  %inext =l add %i, 1
  jmp @loop

@done
  ret %min
}
