export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    %i0 =l copy 1
    jmp @loop
@loop
    %max =l phi @start %first, @cont %next_max
    %i =l phi @start %i0, @cont %next_i
    %c =w csltl %i, %n
    jnz %c, @body, @end
@body
    %off =l mul %i, 8
    %addr =l add %a, %off
    %cur =l loadl %addr
    %gt =w csgtl %cur, %max
    jnz %gt, @update, @cont
@update
    %new_max =l copy %cur
    jmp @cont
@cont
    %next_max =l phi @update %new_max, @body %max
    %next_i =l add %i, 1
    jmp @loop
@end
    ret %max
}
