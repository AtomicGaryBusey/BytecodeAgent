sdivrem:
    mov r8, rdx
    test rsi, rsi
    jz .Lzero_b
    cmp rsi, -1
    jne .Lnormal
    mov r9, 0x8000000000000000
    cmp rdi, r9
    je .Loverflow
.Lnormal:
    mov rax, rdi
    cqo
    idiv rsi
    mov [r8], rdx
    ret
.Lzero_b:
    xor eax, eax
    mov [r8], rdi
    ret
.Loverflow:
    mov rax, 0x8000000000000000
    mov qword ptr [r8], 0
    ret
