export function l $count_desc(l %a, l %n) {
@start
  %small =w csltl %n, 2
  jnz %small, @zero, @init
@zero
  ret 0
@init
  %limit =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @init 0, @body %i2
  %count =l phi @init 0, @body %count2
  %done =w csgel %i, %limit
  jnz %done, @end, @body
@body
  %off =l mul %i, 8
  %addr0 =l add %a, %off
  %v0 =l loadl %addr0
  %addr1 =l add %addr0, 8
  %v1 =l loadl %addr1
  %gt =w csgtl %v0, %v1
  %gtl =l extuw %gt
  %count2 =l add %count, %gtl
  %i2 =l add %i, 1
  jmp @loop
@end
  ret %count
}
