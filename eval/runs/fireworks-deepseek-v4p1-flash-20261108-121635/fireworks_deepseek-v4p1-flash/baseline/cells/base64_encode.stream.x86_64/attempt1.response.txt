base64_encode:
    push rbx
    push r12
    mov rbx, rdx
    lea rsi, [rdi + rsi]
    xor r8d, r8d
.Lloop:
    mov rax, rsi
    sub rax, rdi
    cmp rax, 3
    jb .Lremainder
    movzx eax, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    movzx r10d, byte ptr [rdi + 2]
    shl eax, 16
    shl r9d, 8
    or eax, r9d
    or eax, r10d
    add rdi, 3
    xor r8d, r8d
    jmp .Lencode4
.Lremainder:
    test rax, rax
    jz .Ldone
    cmp rax, 1
    je .Lrem1
    movzx eax, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    shl eax, 16
    shl r9d, 8
    or eax, r9d
    add rdi, 2
    mov r8d, 1
    jmp .Lencode4
.Lrem1:
    movzx eax, byte ptr [rdi]
    shl eax, 16
    inc rdi
    mov r8d, 2
    jmp .Lencode4
.Lencode4:
    mov ecx, 4
.Lgroup_loop:
    mov r9d, eax
    shr r9d, 18
    and r9d, 63
    mov r10d, 65
    xor r11d, r11d
    cmp r9d, 26
    mov r12d, 97
    cmovae r10d, r12d
    mov r12d, 26
    cmovae r11d, r12d
    cmp r9d, 52
    mov r12d, 48
    cmovae r10d, r12d
    mov r12d, 52
    cmovae r11d, r12d
    mov r12d, 43
    cmp r9d, 62
    cmove r10d, r12d
    mov r12d, 47
    cmp r9d, 63
    cmove r10d, r12d
    mov r12d, r9d
    cmp r9d, 62
    cmovae r11d, r12d
    sub r9d, r11d
    add r10d, r9d
    mov byte ptr [rdx], r10b
    inc rdx
    shl eax, 6
    dec ecx
    jnz .Lgroup_loop
    test r8d, r8d
    jz .Lloop
    cmp r8d, 2
    je .Lpad2
    mov byte ptr [rdx - 1], 61
    jmp .Ldone
.Lpad2:
    mov byte ptr [rdx - 2], 61
    mov byte ptr [rdx - 1], 61
    jmp .Ldone
.Ldone:
    mov rax, rdx
    sub rax, rbx
    pop r12
    pop rbx
    ret
