mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbx, r8
    mov r11, r13
    add r11, r15
    xor rcx, rcx
.Lzero:
    cmp rcx, r11
    jae .Lzero_done
    mov qword ptr [rbx + rcx*8], 0
    inc rcx
    jmp .Lzero
.Lzero_done:
    xor rbp, rbp
.Louter:
    cmp rbp, r13
    jae .Louter_done
    xor r10, r10
    xor r9, r9
.Linner:
    cmp r9, r15
    jae .Linner_done
    mov rcx, rbp
    add rcx, r9
    mov rax, qword ptr [r12 + rbp*8]
    mul qword ptr [r14 + r9*8]
    add rax, qword ptr [rbx + rcx*8]
    adc rdx, 0
    add rax, r10
    adc rdx, 0
    mov qword ptr [rbx + rcx*8], rax
    mov r10, rdx
    inc r9
    jmp .Linner
.Linner_done:
    mov rcx, rbp
    add rcx, r15
    mov qword ptr [rbx + rcx*8], r10
    inc rbp
    jmp .Louter
.Louter_done:
    mov rcx, r13
    add rcx, r15
    mov rax, rcx
.Lscan:
    test rax, rax
    jz .Lscan_done
    cmp qword ptr [rbx + rax*8 - 8], 0
    jne .Lscan_done
    dec rax
    jmp .Lscan
.Lscan_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
