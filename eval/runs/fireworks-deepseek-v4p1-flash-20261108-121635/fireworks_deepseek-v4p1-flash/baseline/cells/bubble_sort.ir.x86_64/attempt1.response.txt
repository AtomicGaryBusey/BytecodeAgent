export function l $bubble_sort(l %a, l %n) {
@start
  %cond0 =w csgtl %n, 0
  jnz %cond0, @outer_init, @ret0
@outer_init
  %i0 =l copy 0
  %n1 =l sub %n, 1
  jmp @outer_cond
@outer_cond
  %i =l phi @outer_init %i0, @outer_inc %i_next
  %c =w csltl %i, %n1
  jnz %c, @inner_init, @ret0
@inner_init
  %j0 =l copy 0
  jmp @inner_cond
@inner_cond
  %j =l phi @inner_init %j0, @inner_inc %j_next
  %limit =l sub %n1, %i
  %c2 =w csltl %j, %limit
  jnz %c2, @inner_body, @outer_inc
@inner_body
  %offj =l mul %j, 8
  %p1 =l add %a, %offj
  %v1 =l loadl %p1
  %offj1 =l add %offj, 8
  %p2 =l add %a, %offj1
  %v2 =l loadl %p2
  %gt =w csgtl %v1, %v2
  jnz %gt, @swap, @inner_inc
@swap
  storel %v2, %p1
  storel %v1, %p2
  jmp @inner_inc
@inner_inc
  %j_next =l add %j, 1
  jmp @inner_cond
@outer_inc
  %i_next =l add %i, 1
  jmp @outer_cond
@ret0
  ret 0
}
