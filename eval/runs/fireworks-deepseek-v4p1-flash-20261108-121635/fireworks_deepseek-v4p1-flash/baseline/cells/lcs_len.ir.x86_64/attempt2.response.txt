export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %np1 =l add %n, 1
  jmp @zero_test

@zero_test
  %jz =l phi @start 0, @zero_body %jz_next
  %cond_zero =w csltl %jz, %np1
  jnz %cond_zero, @zero_body, @zero_end

@zero_body
  %zoff =l mul %jz, 8
  %zaddr =l add %row, %zoff
  storel 0, %zaddr
  %jz_next =l add %jz, 1
  jmp @zero_test

@zero_end
  jmp @outer_test

@outer_test
  %i =l phi @zero_end 0, @inner_end %inext
  %cond_outer =w csltl %i, %m
  jnz %cond_outer, @outer_body, @outer_end

@outer_body
  %diag_init =l copy 0
  %j_init =l copy 1
  jmp @inner_test

@inner_test
  %j =l phi @outer_body %j_init, @inner_body %jnext
  %diag =l phi @outer_body %diag_init, @inner_body %up
  %cond_inner =w csltl %j, %np1
  jnz %cond_inner, @inner_body, @inner_end

@inner_body
  %joff =l mul %j, 8
  %jaddr =l add %row, %joff
  %up =l loadl %jaddr
  %jm1 =l sub %j, 1
  %jm1off =l mul %jm1, 8
  %jm1addr =l add %row, %jm1off
  %prev =l loadl %jm1addr
  %xaddr =l add %x, %i
  %xi =w loadub %xaddr
  %yaddr =l add %y, %jm1
  %yj =w loadub %yaddr
  %xi64 =l extuw %xi
  %yj64 =l extuw %yj
  %eq =w ceql %xi64, %yj64
  %match_val =l add %diag, 1
  %diff =l sub %up, %prev
  %mask =l sar %diff, 63
  %diff_and_mask =l and %diff, %mask
  %max_val =l sub %up, %diff_and_mask
  %diff2 =l sub %match_val, %max_val
  %eq_l =l extsw %eq
  %prod =l mul %eq_l, %diff2
  %newval =l add %max_val, %prod
  storel %newval, %jaddr
  %jnext =l add %j, 1
  jmp @inner_test

@inner_end
  %inext =l add %i, 1
  jmp @outer_test

@outer_end
  %noff =l mul %n, 8
  %naddr =l add %row, %noff
  %res =l loadl %naddr
  ret %res
}
