export function l $reverse(l %a, l %n) {
@start
    %half =l div %n, 2
    %i =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %half
    jnz %cond, @body, @end

@body
    %offi =l mul %i, 8
    %pi =l add %a, %offi
    %x =l loadl %pi

    %j =l sub %n, 1
    %j =l sub %j, %i
    %offj =l mul %j, 8
    %pj =l add %a, %offj
    %y =l loadl %pj

    storel %y, %pi
    storel %x, %pj

    %i =l add %i, 1
    jmp @loop

@end
    ret 0
}
