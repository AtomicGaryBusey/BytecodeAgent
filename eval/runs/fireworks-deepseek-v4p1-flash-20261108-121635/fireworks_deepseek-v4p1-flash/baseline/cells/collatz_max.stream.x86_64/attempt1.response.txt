collatz_max:
    xor r8d, r8d
    mov r9d, 1
.Louter:
    mov rax, r9
    xor ecx, ecx
.Linner:
    cmp rax, 1
    je .Linner_done
    test rax, 1
    jz .Leven
    lea rax, [rax + rax*2 + 1]
    jmp .Lcount
.Leven:
    shr rax, 1
.Lcount:
    inc ecx
    jmp .Linner
.Linner_done:
    cmp rcx, r8
    jle .Lnext
    mov r8, rcx
.Lnext:
    inc r9
    cmp r9, rdi
    jle .Louter
    mov rax, r8
    ret
