packbits_decode:
    xor eax, eax
.Lloop:
    test rsi, rsi
    jz .Ldone
    movsx r8d, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp r8d, 0
    jl .Lneg
    lea r9, [r8 + 1]
    cmp rsi, r9
    jb .Lerr_src
    cmp rcx, r9
    jb .Ltrunc_lit
    mov r10, r9
.Lcopy_lit:
    mov r11b, byte ptr [rdi]
    mov byte ptr [rdx], r11b
    inc rdi
    inc rdx
    dec r10
    jnz .Lcopy_lit
    add rax, r9
    sub rcx, r9
    sub rsi, r9
    jmp .Lloop
.Lneg:
    cmp r8d, -128
    je .Lloop
    mov r9d, 1
    sub r9d, r8d
    cmp rsi, 1
    jb .Lerr_src
    mov r10b, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp rcx, r9
    jb .Ltrunc_rep
    mov r11, r9
.Lcopy_rep:
    mov byte ptr [rdx], r10b
    inc rdx
    dec r11
    jnz .Lcopy_rep
    add rax, r9
    sub rcx, r9
    jmp .Lloop
.Ltrunc_lit:
    test rcx, rcx
    jz .Ltrunc_done
    mov r10, rcx
.Ltrunc_lit_loop:
    mov r11b, byte ptr [rdi]
    mov byte ptr [rdx], r11b
    inc rdi
    inc rdx
    dec r10
    jnz .Ltrunc_lit_loop
.Ltrunc_done:
    mov rax, -1
    ret
.Ltrunc_rep:
    test rcx, rcx
    jz .Ltrunc_rep_done
    mov r11, rcx
.Ltrunc_rep_loop:
    mov byte ptr [rdx], r10b
    inc rdx
    dec r11
    jnz .Ltrunc_rep_loop
.Ltrunc_rep_done:
    mov rax, -1
    ret
.Lerr_src:
    mov rax, -2
    ret
.Ldone:
    ret
