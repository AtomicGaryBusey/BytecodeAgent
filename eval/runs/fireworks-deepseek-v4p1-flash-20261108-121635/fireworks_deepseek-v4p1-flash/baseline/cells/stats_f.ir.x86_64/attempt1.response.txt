export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %x0 =l loadl %a
  %y0 =l call %f(l %x0)
  %min =l copy %y0
  %max =l copy %y0
  %sum =l copy %y0
  %cnt =l and %y0, 1
  %i =l copy 1
  jmp @loop

@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %off =l mul %i, 8
  %addr =l add %a, %off
  %x =l loadl %addr
  %y =l call %f(l %x)
  %sum =l add %sum, %y
  %odd =l and %y, 1
  %cnt =l add %cnt, %odd
  %lt =w csltl %y, %min
  jnz %lt, @set_min, @check_max

@set_min
  %min =l copy %y
  jmp @check_max

@check_max
  %gt =w csgtl %y, %max
  jnz %gt, @set_max, @next

@set_max
  %max =l copy %y
  jmp @next

@next
  %i =l add %i, 1
  jmp @loop

@done
  storel %min, %out
  %out1 =l add %out, 8
  storel %max, %out1
  %out2 =l add %out, 16
  storel %sum, %out2
  ret %cnt
}
