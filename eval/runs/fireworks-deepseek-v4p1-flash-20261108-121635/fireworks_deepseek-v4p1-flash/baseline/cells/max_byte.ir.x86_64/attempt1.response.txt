export function l $max_byte(l %p, l %n) {
@start
  %iszero =w ceql %n, 0
  jnz %iszero, @ret0, @loop
@loop
  %i =l phi @start 0, @next %i1
  %max =w phi @start 0, @next %max_new
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %addr =l add %p, %i
  %b =w loadub %addr
  %gt =w cugtw %b, %max
  jnz %gt, @upd, @next
@upd
  %max_upd =w copy %b
  jmp @next
@next
  %max_new =w phi @upd %max_upd, @body %max
  %i1 =l add %i, 1
  jmp @loop
@done
  %res =l extuw %max
  ret %res
@ret0
  %zero =l copy 0
  ret %zero
}
