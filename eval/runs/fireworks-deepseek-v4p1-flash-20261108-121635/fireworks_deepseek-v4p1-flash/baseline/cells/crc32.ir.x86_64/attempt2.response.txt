export function l $crc32(l %p, l %n) {
@start
	%one =w copy 1
	%sign =w shl %one, 31
	%all =w sar %sign, 31
	%a1 =w copy 237
	%a1s =w shl %a1, 24
	%a2 =w copy 184
	%a2s =w shl %a2, 16
	%a3 =w copy 131
	%a3s =w shl %a3, 8
	%a4 =w copy 32
	%ah =w or %a1s, %a2s
	%al =w or %a3s, %a4
	%poly =w or %ah, %al
	%i0 =l copy 0
	jmp @loop

@loop
	%crc =w phi @start %all, @body %c9
	%i =l phi @start %i0, @body %inext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%c1 =w xor %crc, %b
	%t1 =w shl %c1, 31
	%m1 =w sar %t1, 31
	%k1 =w and %m1, %poly
	%s1 =w shr %c1, 1
	%c2 =w xor %s1, %k1
	%t2 =w shl %c2, 31
	%m2 =w sar %t2, 31
	%k2 =w and %m2, %poly
	%s2 =w shr %c2, 1
	%c3 =w xor %s2, %k2
	%t3 =w shl %c3, 31
	%m3 =w sar %t3, 31
	%k3 =w and %m3, %poly
	%s3 =w shr %c3, 1
	%c4 =w xor %s3, %k3
	%t4 =w shl %c4, 31
	%m4 =w sar %t4, 31
	%k4 =w and %m4, %poly
	%s4 =w shr %c4, 1
	%c5 =w xor %s4, %k4
	%t5 =w shl %c5, 31
	%m5 =w sar %t5, 31
	%k5 =w and %m5, %poly
	%s5 =w shr %c5, 1
	%c6 =w xor %s5, %k5
	%t6 =w shl %c6, 31
	%m6 =w sar %t6, 31
	%k6 =w and %m6, %poly
	%s6 =w shr %c6, 1
	%c7 =w xor %s6, %k6
	%t7 =w shl %c7, 31
	%m7 =w sar %t7, 31
	%k7 =w and %m7, %poly
	%s7 =w shr %c7, 1
	%c8 =w xor %s7, %k7
	%t8 =w shl %c8, 31
	%m8 =w sar %t8, 31
	%k8 =w and %m8, %poly
	%s8 =w shr %c8, 1
	%c9 =w xor %s8, %k8
	%inext =l add %i, 1
	jmp @loop

@done
	%res =w xor %crc, %all
	%ret =l extuw %res
	ret %ret
}
