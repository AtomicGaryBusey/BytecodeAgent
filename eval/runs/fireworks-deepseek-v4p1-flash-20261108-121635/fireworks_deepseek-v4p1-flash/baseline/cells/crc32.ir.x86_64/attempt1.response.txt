export function l $crc32(l %p, l %n) {
@start
    %crc0 =w copy 0xFFFFFFFF
    %i0 =l copy 0
    jmp @loop

@loop
    %crc =w phi @start %crc0, @body %c9
    %i =l phi @start %i0, @body %i_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %ptr =l add %p, %i
    %b =w loadub %ptr
    %c1 =w xor %crc, %b
    %l1 =w and %c1, 1
    %m1 =w sub 0, %l1
    %k1 =w and %m1, 0xEDB88320
    %s1 =w shr %c1, 1
    %c2 =w xor %s1, %k1
    %l2 =w and %c2, 1
    %m2 =w sub 0, %l2
    %k2 =w and %m2, 0xEDB88320
    %s2 =w shr %c2, 1
    %c3 =w xor %s2, %k2
    %l3 =w and %c3, 1
    %m3 =w sub 0, %l3
    %k3 =w and %m3, 0xEDB88320
    %s3 =w shr %c3, 1
    %c4 =w xor %s3, %k3
    %l4 =w and %c4, 1
    %m4 =w sub 0, %l4
    %k4 =w and %m4, 0xEDB88320
    %s4 =w shr %c4, 1
    %c5 =w xor %s4, %k4
    %l5 =w and %c5, 1
    %m5 =w sub 0, %l5
    %k5 =w and %m5, 0xEDB88320
    %s5 =w shr %c5, 1
    %c6 =w xor %s5, %k5
    %l6 =w and %c6, 1
    %m6 =w sub 0, %l6
    %k6 =w and %m6, 0xEDB88320
    %s6 =w shr %c6, 1
    %c7 =w xor %s6, %k6
    %l7 =w and %c7, 1
    %m7 =w sub 0, %l7
    %k7 =w and %m7, 0xEDB88320
    %s7 =w shr %c7, 1
    %c8 =w xor %s7, %k7
    %l8 =w and %c8, 1
    %m8 =w sub 0, %l8
    %k8 =w and %m8, 0xEDB88320
    %s8 =w shr %c8, 1
    %c9 =w xor %s8, %k8
    %i_next =l add %i, 1
    jmp @loop

@done
    %res =w xor %crc, 0xFFFFFFFF
    %ret =l extuw %res
    ret %ret
}
