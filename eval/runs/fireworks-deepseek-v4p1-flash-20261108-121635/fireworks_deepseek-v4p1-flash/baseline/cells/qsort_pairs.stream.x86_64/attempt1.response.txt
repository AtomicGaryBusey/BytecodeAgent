qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    cmp r12, 1
    jle .Ldone
    mov rax, r12
    dec rax
    shl rax, 4
    mov r13, qword ptr [rbx + rax]
    mov r14, qword ptr [rbx + rax + 8]
    mov r11, r12
    dec r11
    xor r15, r15
    xor rcx, rcx
.Lloop:
    cmp rcx, r11
    jge .Lafter_loop
    mov rax, rcx
    shl rax, 4
    mov rdx, qword ptr [rbx + rax]
    mov rsi, qword ptr [rbx + rax + 8]
    cmp rdx, r13
    jl .Lless
    jg .Lnot_less
    cmp rsi, r14
    jge .Lnot_less
.Lless:
    mov rdi, r15
    shl rdi, 4
    mov r8, qword ptr [rbx + rdi]
    mov r9, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rdi], rdx
    mov qword ptr [rbx + rdi + 8], rsi
    mov qword ptr [rbx + rax], r8
    mov qword ptr [rbx + rax + 8], r9
    inc r15
.Lnot_less:
    inc rcx
    jmp .Lloop
.Lafter_loop:
    mov rax, r11
    shl rax, 4
    mov rdi, r15
    shl rdi, 4
    mov rdx, qword ptr [rbx + rdi]
    mov rsi, qword ptr [rbx + rdi + 8]
    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rdi], r8
    mov qword ptr [rbx + rdi + 8], r9
    mov qword ptr [rbx + rax], rdx
    mov qword ptr [rbx + rax + 8], rsi
    cmp r15, 1
    jle .Lskip_left
    mov rdi, rbx
    mov rsi, r15
    call qsort_pairs
.Lskip_left:
    mov rsi, r12
    sub rsi, r15
    dec rsi
    cmp rsi, 1
    jle .Lskip_right
    mov rdi, rbx
    mov rax, r15
    inc rax
    shl rax, 4
    add rdi, rax
    call qsort_pairs
.Lskip_right:
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
