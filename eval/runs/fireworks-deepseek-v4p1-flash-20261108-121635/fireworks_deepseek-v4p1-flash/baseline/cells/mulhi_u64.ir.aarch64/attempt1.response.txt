export function l $mulhi_u64(l %a, l %b) {
@start
  %al =l and %a, 4294967295
  %ah =l shr %a, 32
  %bl =l and %b, 4294967295
  %bh =l shr %b, 32
  %p0 =l mul %al, %bl
  %p1 =l mul %al, %bh
  %p2 =l mul %ah, %bl
  %p3 =l mul %ah, %bh
  %p0_hi =l shr %p0, 32
  %p1_lo =l and %p1, 4294967295
  %p2_lo =l and %p2, 4294967295
  %p1_hi =l shr %p1, 32
  %p2_hi =l shr %p2, 32
  %t0 =l add %p0_hi, %p1_lo
  %t1 =l add %t0, %p2_lo
  %carry =l shr %t1, 32
  %h0 =l add %p3, %p1_hi
  %h1 =l add %h0, %p2_hi
  %hi =l add %h1, %carry
  ret %hi
}
