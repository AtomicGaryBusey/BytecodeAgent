memmove_i64:
xor eax, eax
test rdx, rdx
jle .Ldone
cmp rdi, rsi
je .Ldone
jb .Lfwd
mov rcx, rdx
dec rcx
.Lback:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
dec rcx
jns .Lback
ret
.Lfwd:
xor ecx, ecx
.Lfwd_loop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
cmp rcx, rdx
jl .Lfwd_loop
.Ldone:
ret
