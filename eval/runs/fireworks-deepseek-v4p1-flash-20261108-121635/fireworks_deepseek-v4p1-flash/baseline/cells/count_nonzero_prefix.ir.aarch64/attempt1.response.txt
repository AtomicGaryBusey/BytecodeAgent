export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @ret_zero, @init
@ret_zero
    ret 0
@init
    jmp @loop
@loop
    %i =l phi @init 0, @next %i_next
    %cond =w csltl %i, %n
    jnz %cond, @check, @end
@check
    %off =l mul %i, 8
    %addr =l add %a, %off
    %val =l loadl %addr
    %is_zero_elem =w ceql %val, 0
    jnz %is_zero_elem, @found, @next
@found
    ret %i
@next
    %i_next =l add %i, 1
    jmp @loop
@end
    ret %n
}
