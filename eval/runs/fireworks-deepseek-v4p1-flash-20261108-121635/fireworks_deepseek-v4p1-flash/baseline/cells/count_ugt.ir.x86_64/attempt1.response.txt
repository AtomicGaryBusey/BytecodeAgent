export function l $count_ugt(l %a, l %n, l %t) {
@start
  %i =l copy 0
  %cnt =l copy 0
  jmp @check

@check
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %gt =w cugtl %val, %t
  %gt64 =l extuw %gt
  %cnt =l add %cnt, %gt64
  %i =l add %i, 1
  jmp @check

@done
  ret %cnt
}
