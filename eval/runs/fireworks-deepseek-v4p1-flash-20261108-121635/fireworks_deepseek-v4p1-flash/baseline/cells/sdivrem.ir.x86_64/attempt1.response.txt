export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bz =w ceql %b, 0
    jnz %bz, @zero, @check_ovf

@zero
    storel %a, %rem
    ret 0

@check_ovf
    %min =l shl 1, 63
    %a_is_min =w ceql %a, %min
    %b_is_m1 =w ceql %b, -1
    %both =w and %a_is_min, %b_is_m1
    jnz %both, @ovf, @normal

@ovf
    storel 0, %rem
    ret %min

@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
