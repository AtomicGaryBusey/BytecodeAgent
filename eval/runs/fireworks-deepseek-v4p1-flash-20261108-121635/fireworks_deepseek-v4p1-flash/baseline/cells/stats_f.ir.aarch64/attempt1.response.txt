export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%p0 =l copy %a
	%x0 =l loadl %p0
	%y0 =l call %f(l %x0)
	%t0 =l and %y0, 1
	%odd0 =w cnel %t0, 0
	%count0 =l extuw %odd0
	%i0 =l copy 1
	jmp @loop

@loop
	%i =l phi @start %i0, @merge_max %i_new
	%min =l phi @start %y0, @merge_max %min_new
	%max =l phi @start %y0, @merge_max %max_new
	%sum =l phi @start %y0, @merge_max %sum_new
	%count =l phi @start %count0, @merge_max %count_new
	%cond =w csltl %i, %n
	jnz %cond, @body, @done

@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%cmin =w csltl %y, %min
	jnz %cmin, @upd_min, @skip_min

@upd_min
	%min1 =l copy %y
	jmp @merge_min

@skip_min
	%min2 =l copy %min
	jmp @merge_min

@merge_min
	%min_new =l phi @upd_min %min1, @skip_min %min2
	%cmax =w csgtl %y, %max
	jnz %cmax, @upd_max, @skip_max

@upd_max
	%max1 =l copy %y
	jmp @merge_max

@skip_max
	%max2 =l copy %max
	jmp @merge_max

@merge_max
	%max_new =l phi @upd_max %max1, @skip_max %max2
	%sum_new =l add %sum, %y
	%t =l and %y, 1
	%odd =w cnel %t, 0
	%odd64 =l extuw %odd
	%count_new =l add %count, %odd64
	%i_new =l add %i, 1
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
