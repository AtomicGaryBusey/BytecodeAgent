export function l $fib(l %n) {
@start
    %is0 =w ceql %n, 0
    jnz %is0, @ret0, @check1
@check1
    %is1 =w ceql %n, 1
    jnz %is1, @ret1, @init
@ret0
    ret 0
@ret1
    ret 1
@init
    jmp @loop
@loop
    %a =l phi @init 0, @body %a_next
    %b =l phi @init 1, @body %b_next
    %i =l phi @init 2, @body %i_next
    %cond =w cslel %i, %n
    jnz %cond, @body, @done
@body
    %t =l add %a, %b
    %a_next =l copy %b
    %b_next =l copy %t
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %b
}
