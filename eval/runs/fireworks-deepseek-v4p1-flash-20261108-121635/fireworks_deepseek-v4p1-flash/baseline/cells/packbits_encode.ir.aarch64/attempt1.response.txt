export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %frame =l alloc8 64
    %i_ptr =l add %frame, 0
    %op_ptr =l add %frame, 8
    %j_ptr =l add %frame, 16
    %r_ptr =l add %frame, 24
    %c_ptr =l add %frame, 32
    %L_ptr =l add %frame, 40
    %k_ptr =l add %frame, 48
    storew 0, %i_ptr
    storew 0, %op_ptr
    jmp @loop

@loop
    %i =w loadw %i_ptr
    %i_l =l extuw %i
    %cond =w csltl %i_l, %n
    jnz %cond, @body, @done

@body
    %i =w loadw %i_ptr
    %i_l =l extuw %i
    %p =l add %src, %i_l
    %c =w loadub %p
    storew %c, %c_ptr
    storew 0, %r_ptr
    storew %i, %j_ptr
    jmp @run_loop

@run_loop
    %j =w loadw %j_ptr
    %j_l =l extuw %j
    %r =w loadw %r_ptr
    %c1 =w csltl %j_l, %n
    jnz %c1, @run_check_r, @run_done

@run_check_r
    %r =w loadw %r_ptr
    %c2 =w csltw %r, 128
    jnz %c2, @run_check_byte, @run_done

@run_check_byte
    %j =w loadw %j_ptr
    %j_l =l extuw %j
    %pj =l add %src, %j_l
    %cj =w loadub %pj
    %c_cur =w loadw %c_ptr
    %eq =w ceqw %cj, %c_cur
    jnz %eq, @run_inc, @run_done

@run_inc
    %r =w loadw %r_ptr
    %r_new =w add %r, 1
    storew %r_new, %r_ptr
    %j =w loadw %j_ptr
    %j_new =w add %j, 1
    storew %j_new, %j_ptr
    jmp @run_loop

@run_done
    %r_final =w loadw %r_ptr
    %is_run =w csgtw %r_final, 2
    jnz %is_run, @emit_repeat, @emit_literal

@emit_repeat
    %op =w loadw %op_ptr
    %op_l =l extuw %op
    %cap_minus =l sub %cap, 2
    %fit =w cslel %op_l, %cap_minus
    jnz %fit, @write_repeat, @fail

@write_repeat
    %r_final =w loadw %r_ptr
    %op =w loadw %op_ptr
    %op_l =l extuw %op
    %h =w sub 257, %r_final
    %outp =l add %out, %op_l
    storeb %h, %outp
    %outp1 =l add %outp, 1
    %c_val =w loadw %c_ptr
    storeb %c_val, %outp1
    %op_new =w add %op, 2
    storew %op_new, %op_ptr
    %i_cur =w loadw %i_ptr
    %i_new =w add %i_cur, %r_final
    storew %i_new, %i_ptr
    jmp @loop

@emit_literal
    %i_cur =w loadw %i_ptr
    storew %i_cur, %j_ptr
    storew 0, %L_ptr
    jmp @lit_loop

@lit_loop
    %j =w loadw %j_ptr
    %L =w loadw %L_ptr
    %j_l =l extuw %j
    %c1 =w csltl %j_l, %n
    jnz %c1, @lit_check_L, @lit_done

@lit_check_L
    %L =w loadw %L_ptr
    %c2 =w csltw %L, 128
    jnz %c2, @lit_check_three, @lit_done

@lit_check_three
    %j =w loadw %j_ptr
    %j2 =w add %j, 2
    %j2_l =l extuw %j2
    %c3 =w csltl %j2_l, %n
    jnz %c3, @lit_load_three, @lit_inc

@lit_load_three
    %j =w loadw %j_ptr
    %j_l =l extuw %j
    %pj =l add %src, %j_l
    %b0 =w loadub %pj
    %pj1 =l add %pj, 1
    %b1 =w loadub %pj1
    %eq01 =w ceqw %b0, %b1
    jnz %eq01, @lit_check_12, @lit_inc

@lit_check_12
    %j =w loadw %j_ptr
    %j_l =l extuw %j
    %pj =l add %src, %j_l
    %pj1 =l add %pj, 1
    %b1 =w loadub %pj1
    %pj2 =l add %pj, 2
    %b2 =w loadub %pj2
    %eq12 =w ceqw %b1, %b2
    jnz %eq12, @lit_done, @lit_inc

@lit_inc
    %j =w loadw %j_ptr
    %j_new =w add %j, 1
    storew %j_new, %j_ptr
    %L =w loadw %L_ptr
    %L_new =w add %L, 1
    storew %L_new, %L_ptr
    jmp @lit_loop

@lit_done
    %op =w loadw %op_ptr
    %op_l =l extuw %op
    %L_val =w loadw %L_ptr
    %L_l =l extuw %L_val
    %packet_len =l add 1, %L_l
    %new_op =l add %op_l, %packet_len
    %fit =w cslel %new_op, %cap
    jnz %fit, @write_literal, @fail

@write_literal
    %op =w loadw %op_ptr
    %op_l =l extuw %op
    %L_val =w loadw %L_ptr
    %h =w sub %L_val, 1
    %outp =l add %out, %op_l
    storeb %h, %outp
    storew 0, %k_ptr
    %i_cur =w loadw %i_ptr
    %i_l =l extuw %i_cur
    %srcp =l add %src, %i_l
    %destp =l add %outp, 1
    jmp @copy_loop

@copy_loop
    %k =w loadw %k_ptr
    %L_val =w loadw %L_ptr
    %c_k =w csltw %k, %L_val
    jnz %c_k, @copy_body, @copy_done

@copy_body
    %k =w loadw %k_ptr
    %k_l =l extuw %k
    %sp =l add %srcp, %k_l
    %byte =w loadub %sp
    %dp =l add %destp, %k_l
    storeb %byte, %dp
    %k_new =w add %k, 1
    storew %k_new, %k_ptr
    jmp @copy_loop

@copy_done
    %op =w loadw %op_ptr
    %L_val =w loadw %L_ptr
    %op_new =w add %op, 1
    %op_new2 =w add %op_new, %L_val
    storew %op_new2, %op_ptr
    %j_val =w loadw %j_ptr
    storew %j_val, %i_ptr
    jmp @loop

@fail
    ret -1

@done
    %op_final =w loadw %op_ptr
    %ret =l extuw %op_final
    ret %ret
}
