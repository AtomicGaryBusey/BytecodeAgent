export function l $collatz_max(l %n) {
@start
  %best =l copy 0
  %k =l copy 1
  jmp @loop_check

@loop_check
  %done =w csgtl %k, %n
  jnz %done, @end, @body

@body
  %x =l copy %k
  %steps =l copy 0
  jmp @inner_check

@inner_check
  %is_one =w ceql %x, 1
  jnz %is_one, @inner_done, @inner_body

@inner_body
  %odd =l and %x, 1
  %is_odd =w cnel %odd, 0
  jnz %is_odd, @odd_case, @even_case

@odd_case
  %t1 =l mul %x, 3
  %x =l add %t1, 1
  jmp @inc_steps

@even_case
  %x =l div %x, 2
  jmp @inc_steps

@inc_steps
  %steps =l add %steps, 1
  jmp @inner_check

@inner_done
  %better =w csgtl %steps, %best
  jnz %better, @update_best, @next_k

@update_best
  %best =l copy %steps
  jmp @next_k

@next_k
  %k =l add %k, 1
  jmp @loop_check

@end
  ret %best
}
