binary_search:
    mov r8, 0
    mov r9, rsi
    dec r9
.Lloop:
    cmp r8, r9
    jg .Lnot_found
    mov r10, r9
    sub r10, r8
    shr r10, 1
    add r10, r8
    mov r11, qword ptr [rdi + r10*8]
    cmp r11, rdx
    je .Lfound
    jl .Lless
    lea r9, [r10 - 1]
    jmp .Lloop
.Lless:
    lea r8, [r10 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r10
    ret
.Lnot_found:
    mov rax, -1
    ret
