export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  jmp @init_loop
@init_loop
  %j =l phi @start 0, @init_body %jnext
  %cond =w cslel %j, %n
  jnz %cond, @init_body, @init_end
@init_body
  %idx =l shl %j, 3
  %addr =l add %row, %idx
  storel 0, %addr
  %jnext =l add %j, 1
  jmp @init_loop
@init_end
  jmp @outer_loop
@outer_loop
  %i =l phi @init_end 0, @inner_end %inext
  %condi =w csltl %i, %m
  jnz %condi, @outer_body, @outer_end
@outer_body
  %xi_addr =l add %x, %i
  %xi =w loadub %xi_addr
  jmp @inner_loop
@inner_loop
  %j2 =l phi @outer_body 1, @inner_body %jnext2
  %left =l phi @outer_body 0, @inner_body %cur
  %diag =l phi @outer_body 0, @inner_body %up
  %condj =w cslel %j2, %n
  jnz %condj, @inner_body, @inner_end
@inner_body
  %idx2 =l shl %j2, 3
  %addr2 =l add %row, %idx2
  %up =l loadl %addr2
  %jm1 =l sub %j2, 1
  %yaddr =l add %y, %jm1
  %yj =w loadub %yaddr
  %eq =w ceqw %xi, %yj
  %eql =l extuw %eq
  %diag_plus1 =l add %diag, 1
  %diff =l sub %up, %left
  %mask =l sar %diff, 63
  %tmp =l and %mask, %diff
  %max =l sub %up, %tmp
  %diff2 =l sub %diag_plus1, %max
  %tmp2 =l mul %eql, %diff2
  %cur =l add %max, %tmp2
  storel %cur, %addr2
  %jnext2 =l add %j2, 1
  jmp @inner_loop
@inner_end
  %inext =l add %i, 1
  jmp @outer_loop
@outer_end
  %nidx =l shl %n, 3
  %naddr =l add %row, %nidx
  %res =l loadl %naddr
  ret %res
}
