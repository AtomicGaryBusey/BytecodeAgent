map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rcx
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
    xor ebx, ebx
    xor ebp, ebp
.Lloop:
    cmp rbp, r15
    jge .Ldone
    mov rdi, qword ptr [r14 + rbp*8]
    mov qword ptr [rsp], rdi
    call r12
    mov qword ptr [r13 + rbp*8], rax
    mov rdx, qword ptr [rsp]
    cmp rax, rdx
    setl al
    movzx eax, al
    add rbx, rax
    inc rbp
    jmp .Lloop
.Ldone:
    mov rax, rbx
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
