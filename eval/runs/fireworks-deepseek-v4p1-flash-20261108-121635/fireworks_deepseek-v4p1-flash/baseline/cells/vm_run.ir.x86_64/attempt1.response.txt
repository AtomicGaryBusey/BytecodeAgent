export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pcp =l alloc8 8
    %spp =l alloc8 8
    %ip =l alloc8 8
    storel 0, %pcp
    storel 0, %spp
    jmp @loop

@loop
    %pcv =l loadl %pcp
    %spv =l loadl %spp
    %loop_cond =w csltl %pcv, %len
    jnz %loop_cond, @body, @finish

@body
    %body_off =l mul %pcv, 8
    %body_addr =l add %code, %body_off
    %op =l loadl %body_addr
    %is0 =w ceql %op, 0
    jnz %is0, @halt, @check1

@check1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @check2

@check2
    %is2 =w ceql %op, 2
    jnz %is2, @add, @check3

@check3
    %is3 =w ceql %op, 3
    jnz %is3, @sub, @check4

@check4
    %is4 =w ceql %op, 4
    jnz %is4, @mul, @check5

@check5
    %is5 =w ceql %op, 5
    jnz %is5, @dup, @check6

@check6
    %is6 =w ceql %op, 6
    jnz %is6, @swap, @check7

@check7
    %is7 =w ceql %op, 7
    jnz %is7, @drop, @check8

@check8
    %is8 =w ceql %op, 8
    jnz %is8, @over, @check9

@check9
    %is9 =w ceql %op, 9
    jnz %is9, @lt, @check10

@check10
    %is10 =w ceql %op, 10
    jnz %is10, @jz, @check11

@check11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp, @default

@halt
    jmp @finish

@push
    %push_pc1 =l add %pcv, 1
    %push_off1 =l mul %push_pc1, 8
    %push_addr1 =l add %code, %push_off1
    %push_v =l loadl %push_addr1
    %push_soff =l mul %spv, 8
    %push_saddr =l add %stack, %push_soff
    storel %push_v, %push_saddr
    %push_sp1 =l add %spv, 1
    storel %push_sp1, %spp
    %push_pc2 =l add %pcv, 2
    storel %push_pc2, %pcp
    jmp @loop

@add
    %add_spm1 =l sub %spv, 1
    %add_spm2 =l sub %spv, 2
    %add_offa =l mul %add_spm2, 8
    %add_addra =l add %stack, %add_offa
    %add_a =l loadl %add_addra
    %add_offb =l mul %add_spm1, 8
    %add_addrb =l add %stack, %add_offb
    %add_b =l loadl %add_addrb
    %add_res =l add %add_a, %add_b
    storel %add_res, %add_addra
    storel %add_spm1, %spp
    %add_pc1 =l add %pcv, 1
    storel %add_pc1, %pcp
    jmp @loop

@sub
    %sub_spm1 =l sub %spv, 1
    %sub_spm2 =l sub %spv, 2
    %sub_offa =l mul %sub_spm2, 8
    %sub_addra =l add %stack, %sub_offa
    %sub_a =l loadl %sub_addra
    %sub_offb =l mul %sub_spm1, 8
    %sub_addrb =l add %stack, %sub_offb
    %sub_b =l loadl %sub_addrb
    %sub_res =l sub %sub_a, %sub_b
    storel %sub_res, %sub_addra
    storel %sub_spm1, %spp
    %sub_pc1 =l add %pcv, 1
    storel %sub_pc1, %pcp
    jmp @loop

@mul
    %mul_spm1 =l sub %spv, 1
    %mul_spm2 =l sub %spv, 2
    %mul_offa =l mul %mul_spm2, 8
    %mul_addra =l add %stack, %mul_offa
    %mul_a =l loadl %mul_addra
    %mul_offb =l mul %mul_spm1, 8
    %mul_addrb =l add %stack, %mul_offb
    %mul_b =l loadl %mul_addrb
    %mul_res =l mul %mul_a, %mul_b
    storel %mul_res, %mul_addra
    storel %mul_spm1, %spp
    %mul_pc1 =l add %pcv, 1
    storel %mul_pc1, %pcp
    jmp @loop

@dup
    %dup_spm1 =l sub %spv, 1
    %dup_offtop =l mul %dup_spm1, 8
    %dup_addrtop =l add %stack, %dup_offtop
    %dup_top =l loadl %dup_addrtop
    %dup_soff =l mul %spv, 8
    %dup_saddr =l add %stack, %dup_soff
    storel %dup_top, %dup_saddr
    %dup_sp1 =l add %spv, 1
    storel %dup_sp1, %spp
    %dup_pc1 =l add %pcv, 1
    storel %dup_pc1, %pcp
    jmp @loop

@swap
    %swap_spm1 =l sub %spv, 1
    %swap_spm2 =l sub %spv, 2
    %swap_off1 =l mul %swap_spm1, 8
    %swap_addr1 =l add %stack, %swap_off1
    %swap_v1 =l loadl %swap_addr1
    %swap_off2 =l mul %swap_spm2, 8
    %swap_addr2 =l add %stack, %swap_off2
    %swap_v2 =l loadl %swap_addr2
    storel %swap_v2, %swap_addr1
    storel %swap_v1, %swap_addr2
    %swap_pc1 =l add %pcv, 1
    storel %swap_pc1, %pcp
    jmp @loop

@drop
    %drop_spm1 =l sub %spv, 1
    storel %drop_spm1, %spp
    %drop_pc1 =l add %pcv, 1
    storel %drop_pc1, %pcp
    jmp @loop

@over
    %over_spm2 =l sub %spv, 2
    %over_offa =l mul %over_spm2, 8
    %over_addra =l add %stack, %over_offa
    %over_a =l loadl %over_addra
    %over_soff =l mul %spv, 8
    %over_saddr =l add %stack, %over_soff
    storel %over_a, %over_saddr
    %over_sp1 =l add %spv, 1
    storel %over_sp1, %spp
    %over_pc1 =l add %pcv, 1
    storel %over_pc1, %pcp
    jmp @loop

@lt
    %lt_spm1 =l sub %spv, 1
    %lt_spm2 =l sub %spv, 2
    %lt_offa =l mul %lt_spm2, 8
    %lt_addra =l add %stack, %lt_offa
    %lt_a =l loadl %lt_addra
    %lt_offb =l mul %lt_spm1, 8
    %lt_addrb =l add %stack, %lt_offb
    %lt_b =l loadl %lt_addrb
    %lt_cmp =w csltl %lt_a, %lt_b
    %lt_res =l extuw %lt_cmp
    storel %lt_res, %lt_addra
    storel %lt_spm1, %spp
    %lt_pc1 =l add %pcv, 1
    storel %lt_pc1, %pcp
    jmp @loop

@jz
    %jz_spm1 =l sub %spv, 1
    %jz_offv =l mul %jz_spm1, 8
    %jz_addrv =l add %stack, %jz_offv
    %jz_v =l loadl %jz_addrv
    storel %jz_spm1, %spp
    %jz_iszero =w ceql %jz_v, 0
    jnz %jz_iszero, @jz_taken, @jz_not

@jz_taken
    %jz_pc1 =l add %pcv, 1
    %jz_off1 =l mul %jz_pc1, 8
    %jz_addr1 =l add %code, %jz_off1
    %jz_target =l loadl %jz_addr1
    storel %jz_target, %pcp
    jmp @loop

@jz_not
    %jz_pc2 =l add %pcv, 2
    storel %jz_pc2, %pcp
    jmp @loop

@jmp
    %jmp_pc1 =l add %pcv, 1
    %jmp_off1 =l mul %jmp_pc1, 8
    %jmp_addr1 =l add %code, %jmp_off1
    %jmp_target =l loadl %jmp_addr1
    storel %jmp_target, %pcp
    jmp @loop

@default
    %def_pc1 =l add %pcv, 1
    storel %def_pc1, %pcp
    jmp @loop

@finish
    %fin_spv =l loadl %spp
    storel %fin_spv, %ip
    jmp @zero_loop

@zero_loop
    %zero_i =l loadl %ip
    %zero_cond =w csltl %zero_i, %cap
    jnz %zero_cond, @zero_body, @check_empty

@zero_body
    %zero_off =l mul %zero_i, 8
    %zero_addr =l add %stack, %zero_off
    storel 0, %zero_addr
    %zero_i1 =l add %zero_i, 1
    storel %zero_i1, %ip
    jmp @zero_loop

@check_empty
    %chk_spv =l loadl %spp
    %chk_isempty =w ceql %chk_spv, 0
    jnz %chk_isempty, @ret_zero, @ret_top

@ret_zero
    ret 0

@ret_top
    %ret_spv =l loadl %spp
    %ret_spm1 =l sub %ret_spv, 1
    %ret_off =l mul %ret_spm1, 8
    %ret_addr =l add %stack, %ret_off
    %ret_topval =l loadl %ret_addr
    ret %ret_topval
}
