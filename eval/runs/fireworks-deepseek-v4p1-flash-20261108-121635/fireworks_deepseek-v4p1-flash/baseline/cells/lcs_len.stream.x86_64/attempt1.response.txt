lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    mov r13, rcx
    mov r14, r8
    mov qword ptr [r14], 0
    mov rcx, r13
    test rcx, rcx
    jz .Lzero_done
    lea rdi, [r14+8]
    xor eax, eax
.Lzero_loop:
    mov qword ptr [rdi], rax
    add rdi, 8
    dec rcx
    jnz .Lzero_loop
.Lzero_done:
    test rbp, rbp
    jz .Louter_done
.Louter:
    movzx r8d, byte ptr [rbx]
    xor r15d, r15d
    lea r9, [r14+8]
    mov r10, r14
    mov r11, r12
    mov rcx, r13
    test rcx, rcx
    jz .Linner_done
.Linner:
    mov rsi, qword ptr [r9]
    mov al, byte ptr [r11]
    cmp r8b, al
    je .Lmatch
    mov rdx, rsi
    mov rax, qword ptr [r10]
    cmp rdx, rax
    cmovl rdx, rax
    jmp .Lstore
.Lmatch:
    lea rdx, [r15+1]
.Lstore:
    mov qword ptr [r9], rdx
    mov r15, rsi
    add r9, 8
    add r10, 8
    inc r11
    dec rcx
    jnz .Linner
.Linner_done:
    add rbx, 1
    dec rbp
    jnz .Louter
.Louter_done:
    mov rax, qword ptr [r14 + r13*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
