export function l $gcd(l %a, l %b) {
@start
	%c0 =w ceql %a, 0
	jnz %c0, @retb
	%c1 =w ceql %b, 0
	jnz %c1, @reta
	jmp @loop
@retb
	ret %b
@reta
	ret %a
@loop
	%x =l phi @start %a, @body %nx
	%y =l phi @start %b, @body %ny
	%r =l rem %x, %y
	%z =w ceql %r, 0
	jnz %z, @done
	jmp @body
@body
	%nx =l copy %y
	%ny =l copy %r
	jmp @loop
@done
	ret %y
}
