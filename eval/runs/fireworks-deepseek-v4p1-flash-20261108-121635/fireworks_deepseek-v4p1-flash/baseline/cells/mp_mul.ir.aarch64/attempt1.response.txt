export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %total =l add %na, %nb
  %i0 =l copy 0
  jmp @zero_test
@zero_test
  %c =w cultl %i0, %total
  jnz %c, @zero_body, @zero_end
@zero_body
  %off =l mul %i0, 8
  %addr =l add %r, %off
  storel 0, %addr
  %i0 =l add %i0, 1
  jmp @zero_test
@zero_end
  %i =l copy 0
  jmp @outer_test
@outer_test
  %c =w cultl %i, %na
  jnz %c, @outer_body, @outer_end
@outer_body
  %carry =l copy 0
  %j =l copy 0
  jmp @inner_test
@inner_test
  %c =w cultl %j, %nb
  jnz %c, @inner_body, @inner_end
@inner_body
  %off_a =l mul %i, 8
  %addr_a =l add %a, %off_a
  %x =l loadl %addr_a
  %off_b =l mul %j, 8
  %addr_b =l add %b, %off_b
  %y =l loadl %addr_b
  %idx =l add %i, %j
  %off_r =l mul %idx, 8
  %addr_r =l add %r, %off_r
  %z =l loadl %addr_r
  %x0 =l and %x, 0xffffffff
  %x1 =l shr %x, 32
  %y0 =l and %y, 0xffffffff
  %y1 =l shr %y, 32
  %p00 =l mul %x0, %y0
  %p01 =l mul %x0, %y1
  %p10 =l mul %x1, %y0
  %p11 =l mul %x1, %y1
  %sum_mid =l add %p01, %p10
  %carry_mid =w cultl %sum_mid, %p01
  %carry_mid_l =l extuw %carry_mid
  %carry_mid_shifted =l shl %carry_mid_l, 32
  %sum_mid_low =l and %sum_mid, 0xffffffff
  %sum_mid_high =l shr %sum_mid, 32
  %mid_shifted =l shl %sum_mid_low, 32
  %low_part =l add %p00, %mid_shifted
  %carry_low =w cultl %low_part, %p00
  %carry_low_l =l extuw %carry_low
  %high_part =l add %p11, %sum_mid_high
  %high_part =l add %high_part, %carry_low_l
  %high_part =l add %high_part, %carry_mid_shifted
  %sum1 =l add %low_part, %z
  %carry1 =w cultl %sum1, %low_part
  %carry1_l =l extuw %carry1
  %sum2 =l add %sum1, %carry
  %carry2 =w cultl %sum2, %sum1
  %carry2_l =l extuw %carry2
  %new_low =l copy %sum2
  %new_high =l add %high_part, %carry1_l
  %new_high =l add %new_high, %carry2_l
  storel %new_low, %addr_r
  %carry =l copy %new_high
  %j =l add %j, 1
  jmp @inner_test
@inner_end
  %idx2 =l add %i, %nb
  %off_r2 =l mul %idx2, 8
  %addr_r2 =l add %r, %off_r2
  storel %carry, %addr_r2
  %i =l add %i, 1
  jmp @outer_test
@outer_end
  %k =l sub %total, 1
  jmp @scan_test
@scan_test
  %c =w csltl %k, 0
  jnz %c, @return_zero, @scan_body
@scan_body
  %off_k =l mul %k, 8
  %addr_k =l add %r, %off_k
  %v =l loadl %addr_k
  %c =w ceql %v, 0
  jnz %c, @scan_next, @scan_found
@scan_next
  %k =l sub %k, 1
  jmp @scan_test
@scan_found
  %res =l add %k, 1
  ret %res
@return_zero
  ret 0
}
