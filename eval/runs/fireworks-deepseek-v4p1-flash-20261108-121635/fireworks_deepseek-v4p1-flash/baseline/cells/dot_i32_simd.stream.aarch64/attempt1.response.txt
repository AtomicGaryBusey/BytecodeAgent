dot_i32_simd:
    movi v2.2d, #0
    movi v3.2d, #0
    cmp x2, #4
    b.lt .Ltail
.Lloop:
    ld1 {v0.4s}, [x0], #16
    ld1 {v1.4s}, [x1], #16
    smlal v2.2d, v0.2s, v1.2s
    smlal2 v3.2d, v0.4s, v1.4s
    sub x2, x2, #4
    cmp x2, #4
    b.ge .Lloop
.Ltail:
    add v2.2d, v2.2d, v3.2d
    addp v2.2d, v2.2d, v2.2d
    umov x3, v2.d[0]
    cbz x2, .Ldone
.Lscalar:

    ldr w5, [x1], #4
    smaddl x3, w4, w5, x3
    subs x2, x2, #1
    b.ne .Lscalar
.Ldone:
    mov x0, x3
    ret
