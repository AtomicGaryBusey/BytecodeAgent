crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Lfinal
    lea r8, [rdi + rsi]
.Lbyte:
    movzx ecx, byte ptr [rdi]
    xor eax, ecx
    mov ecx, 8
.Lbit:
    shr eax, 1
    jnc .Lno_xor
    xor eax, 0xEDB88320
.Lno_xor:
    dec ecx
    jnz .Lbit
    inc rdi
    cmp rdi, r8
    jb .Lbyte
.Lfinal:
    xor eax, 0xFFFFFFFF
    ret
