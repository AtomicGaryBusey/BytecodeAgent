export function l $binary_search(l %a, l %n, l %key) {
@start
    %n_le_zero =w cslel %n, 0
    jnz %n_le_zero, @ret_minus1, @init

@init
    %low0 =l copy 0
    %high0 =l sub %n, 1
    jmp @loop

@loop
    %low =l phi @init %low0, @less %low_less, @greater %low_greater
    %high =l phi @init %high0, @less %high_less, @greater %high_greater
    %cond =w cslel %low, %high
    jnz %cond, @body, @ret_minus1

@body
    %diff =l sub %high, %low
    %half =l div %diff, 2
    %mid =l add %low, %half
    %off =l mul %mid, 8
    %addr =l add %a, %off
    %val =l loadl %addr
    %eq =w ceql %val, %key
    jnz %eq, @found, @cont

@cont
    %lt =w csltl %val, %key
    jnz %lt, @less, @greater

@found
    ret %mid

@less
    %mid_plus_1 =l add %mid, 1
    %low_less =l copy %mid_plus_1
    %high_less =l copy %high
    jmp @loop

@greater
    %mid_minus_1 =l sub %mid, 1
    %low_greater =l copy %low
    %high_greater =l copy %mid_minus_1
    jmp @loop

@ret_minus1
    %minus1 =l copy -1
    ret %minus1
}
