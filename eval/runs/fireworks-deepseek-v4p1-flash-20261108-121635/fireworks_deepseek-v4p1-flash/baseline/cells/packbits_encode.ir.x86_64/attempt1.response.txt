export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %o =l copy 0
    jmp @loop

@loop
    %c =w csltl %i, %n
    jnz %c, @body, @done

@body
    %bip =l add %src, %i
    %b0 =w loadub %bip
    %r =w copy 1
    %k =l add %i, 1
    jmp @rloop

@rloop
    %rge =w csgew %r, 128
    jnz %rge, @rdone, @rchk
@rchk
    %kge =w csgel %k, %n
    jnz %kge, @rdone, @rstep
@rstep
    %bkp =l add %src, %k
    %b1 =w loadub %bkp
    %eq =w ceqw %b0, %b1
    jnz %eq, @rinc, @rdone
@rinc
    %r2 =w add %r, 1
    %k2 =l add %k, 1
    %r =w copy %r2
    %k =l copy %k2
    jmp @rloop

@rdone
    %isrun =w csgew %r, 3
    jnz %isrun, @repeat, @literal

@repeat
    %o2 =l add %o, 2
    %fit =w cslel %o2, %cap
    jnz %fit, @repeat_ok, @fail
@repeat_ok
    %h =w sub 1, %r
    %op =l add %out, %o
    storeb %h, %op
    %op1 =l add %op, 1
    storeb %b0, %op1
    %rl =l extuw %r
    %i2 =l add %i, %rl
    %o3 =l add %o, 2
    %i =l copy %i2
    %o =l copy %o3
    jmp @loop

@literal
    %j =l copy %i
    %cnt =w copy 0
    jmp @lloop

@lloop
    %cntlt =w csltw %cnt, 128
    jnz %cntlt, @lloop2, @ldone
@lloop2
    %jlt =w csltl %j, %n
    jnz %jlt, @lloop3, @ldone
@lloop3
    %bjp =l add %src, %j
    %bj =w loadub %bjp
    %rj =w copy 1
    %kk =l add %j, 1
    jmp @rl2

@rl2
    %rjge =w csgew %rj, 128
    jnz %rjge, @rldone, @rl2b
@rl2b
    %kkge =w csgel %kk, %n
    jnz %kkge, @rldone, @rl2c
@rl2c
    %bkp2 =l add %src, %kk
    %bkk =w loadub %bkp2
    %eq2 =w ceqw %bj, %bkk
    jnz %eq2, @rl2inc, @rldone
@rl2inc
    %rj2 =w add %rj, 1
    %kk2 =l add %kk, 1
    %rj =w copy %rj2
    %kk =l copy %kk2
    jmp @rl2

@rldone
    %rjlt3 =w csltw %rj, 3
    jnz %rjlt3, @ladv, @ldone

@ladv
    %j2 =l add %j, 1
    %cnt2 =w add %cnt, 1
    %j =l copy %j2
    %cnt =w copy %cnt2
    jmp @lloop

@ldone
    %cntl =l extuw %cnt
    %need =l add %o, %cntl
    %need1 =l add %need, 1
    %fit2 =w cslel %need1, %cap
    jnz %fit2, @lit_ok, @fail

@lit_ok
    %op2 =l add %out, %o
    %Lm1 =w sub %cnt, 1
    storeb %Lm1, %op2
    %cc =w copy 0
    jmp @cloop

@cloop
    %cclt =w csltw %cc, %cnt
    jnz %cclt, @cbody, @cdone

@cbody
    %ccl =l extuw %cc
    %sidx =l add %i, %ccl
    %sp =l add %src, %sidx
    %cv =w loadub %sp
    %didx =l add %o, 1
    %didx2 =l add %didx, %ccl
    %dp =l add %out, %didx2
    storeb %cv, %dp
    %cc2 =w add %cc, 1
    %cc =w copy %cc2
    jmp @cloop

@cdone
    %o4 =l add %o, 1
    %o5 =l add %o4, %cntl
    %o =l copy %o5
    %i =l copy %j
    jmp @loop

@fail
    %m1 =l copy -1
    ret %m1

@done
    ret %o
}
