export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %inext
    %sum =l phi @start 0, @body %sumnext
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %addr =l add %a, %off
    %val =l loadl %addr
    %res =l call %f(l %val)
    %sumnext =l add %sum, %res
    %inext =l add %i, 1
    jmp @loop
@done
    ret %sum
}
