mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r14, rdi
    mov r13, rsi
    mov r12, rdx
    mov r15, rcx
    mov rbx, r8
    mov rbp, r13
    shl rbp, 3
    xor rdi, rdi
.Lid_i:
    cmp rdi, r13
    jge .Lid_done
    xor rsi, rsi
.Lid_j:
    cmp rsi, r13
    jge .Lid_next_i
    mov rax, rdi
    imul rax, r13
    add rax, rsi
    shl rax, 3
    xor edx, edx
    cmp rdi, rsi
    sete dl
    mov qword ptr [r15 + rax], rdx
    inc rsi
    jmp .Lid_j
.Lid_next_i:
    inc rdi
    jmp .Lid_i
.Lid_done:
    test r12, r12
    jz .Lzero_tmp_start
.Lmult_loop:
    xor rdi, rdi
.Li_loop:
    cmp rdi, r13
    jge .Lmatmul_done
    mov r8, rdi
    imul r8, rbp
    add r8, r15
    mov r9, rdi
    imul r9, rbp
    add r9, rbx
    xor rsi, rsi
.Lj_loop:
    cmp rsi, r13
    jge .Lnext_i
    xor edx, edx
    mov r10, r8
    mov r11, rsi
    shl r11, 3
    add r11, r14
    xor ecx, ecx
.Lt_loop:
    cmp rcx, r13
    jge .Lstore
    mov rax, qword ptr [r10]
    imul rax, qword ptr [r11]
    add rdx, rax
    add r10, 8
    add r11, rbp
    inc rcx
    jmp .Lt_loop
.Lstore:
    mov rax, rsi
    shl rax, 3
    add rax, r9
    mov qword ptr [rax], rdx
    inc rsi
    jmp .Lj_loop
.Lnext_i:
    inc rdi
    jmp .Li_loop
.Lmatmul_done:
    mov rdi, rbx
    mov rsi, r15
    mov rcx, r13
    imul rcx, r13
.Lcopy_loop:
    test rcx, rcx
    jz .Lcopy_done
    mov rax, qword ptr [rdi]
    mov qword ptr [rsi], rax
    add rdi, 8
    add rsi, 8
    dec rcx
    jmp .Lcopy_loop
.Lcopy_done:
    dec r12
    jnz .Lmult_loop
.Lzero_tmp_start:
    mov rdi, rbx
    mov rcx, r13
    imul rcx, r13
    shl rcx, 1
    xor eax, eax
.Lzero_loop:
    test rcx, rcx
    jz .Lzero_done
    mov qword ptr [rdi], rax
    add rdi, 8
    dec rcx
    jmp .Lzero_loop
.Lzero_done:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
