export function l $binary_search(l %a, l %n, l %key) {
@start
    jmp @loop

@loop
    %lo =l phi @start 0, @update %lo_next
    %hi =l phi @start %n, @update %hi_next
    %c =w csltl %lo, %hi
    jnz %c, @body, @notfound

@body
    %sum =l add %lo, %hi
    %mid =l div %sum, 2
    %off =l mul %mid, 8
    %p =l add %a, %off
    %v =l loadl %p
    %eq =w ceql %v, %key
    jnz %eq, @found, @not_equal

@found
    ret %mid

@not_equal
    %lt =w csltl %v, %key
    jnz %lt, @go_right, @go_left

@go_right
    %lo_r =l add %mid, 1
    %hi_r =l copy %hi
    jmp @update

@go_left
    %lo_l =l copy %lo
    %hi_l =l copy %mid
    jmp @update

@update
    %lo_next =l phi @go_right %lo_r, @go_left %lo_l
    %hi_next =l phi @go_right %hi_r, @go_left %hi_l
    jmp @loop

@notfound
    %r =l copy -1
    ret %r
}
