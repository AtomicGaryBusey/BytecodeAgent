export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %iszero =w ceql %n, 0
    jnz %iszero, @done, @check

@check
    %n8 =l mul %n, 8
    %end_src =l add %src, %n8
    %dst_gt_src =w cugtl %dst, %src
    %dst_lt_end =w cultl %dst, %end_src
    %overlap =w and %dst_gt_src, %dst_lt_end
    jnz %overlap, @backward, @forward

@forward
    jmp @f_loop

@f_loop
    %i =l phi @forward 0, @f_body %inext
    %cond =w csgel %i, %n
    jnz %cond, @done, @f_body

@f_body
    %offset =l mul %i, 8
    %asrc =l add %src, %offset
    %adst =l add %dst, %offset
    %val =l loadl %asrc
    storel %val, %adst
    %inext =l add %i, 1
    jmp @f_loop

@backward
    %n1 =l sub %n, 1
    jmp @b_loop

@b_loop
    %j =l phi @backward %n1, @b_body %jnext
    %condb =w csltl %j, 0
    jnz %condb, @done, @b_body

@b_body
    %offsetb =l mul %j, 8
    %bsrc =l add %src, %offsetb
    %bdst =l add %dst, %offsetb
    %valb =l loadl %bsrc
    storel %valb, %bdst
    %jnext =l sub %j, 1
    jmp @b_loop

@done
    ret 0
}
