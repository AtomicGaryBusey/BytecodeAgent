export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %is_eq =w ceql %dst, %src
    jnz %is_eq, @done
    %is_forward =w cultl %dst, %src
    jnz %is_forward, @forward_init
    jmp @backward_init

@forward_init
    %i =l copy 0
    jmp @forward_cond

@forward_cond
    %cond =w csltl %i, %n
    jnz %cond, @forward_body
    jmp @done

@forward_body
    %off =l mul %i, 8
    %s =l add %src, %off
    %d =l add %dst, %off
    %val =l loadl %s
    storel %val, %d
    %i =l add %i, 1
    jmp @forward_cond

@backward_init
    %j =l copy %n
    jmp @backward_cond

@backward_cond
    %cond2 =w csgtl %j, 0
    jnz %cond2, @backward_body
    jmp @done

@backward_body
    %j =l sub %j, 1
    %off2 =l mul %j, 8
    %s2 =l add %src, %off2
    %d2 =l add %dst, %off2
    %val2 =l loadl %s2
    storel %val2, %d2
    jmp @backward_cond

@done
    %zero =l copy 0
    ret %zero
}
