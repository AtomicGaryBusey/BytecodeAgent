fib_rec:
    cmp x0, #1
    b.hi .Lrec
    ret
.Lrec:
    stp x19, x20, [sp, #-32]!
    stp x29, x30, [sp, #16]
    mov x29, sp
    mov x19, x0
    sub x0, x19, #1
    bl fib_rec
    mov x20, x0
    sub x0, x19, #2
    bl fib_rec
    add x0, x0, x20
    ldp x29, x30, [sp, #16]
    ldp x19, x20, [sp], #32
    ret
