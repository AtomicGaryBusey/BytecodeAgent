export function l $count_desc(l %a, l %n) {
@start
  %c =w csltl %n, 2
  jnz %c, @ret_zero, @init

@ret_zero
  ret 0

@init
  %count =l copy 0
  %i =l copy 0
  %limit =l sub %n, 1
  jmp @loop

@loop
  %cmp =w csltl %i, %limit
  jnz %cmp, @body, @done

@body
  %off =l mul %i, 8
  %addr1 =l add %a, %off
  %val1 =l loadl %addr1
  %off2 =l add %off, 8
  %addr2 =l add %a, %off2
  %val2 =l loadl %addr2
  %gt =w csgtl %val1, %val2
  jnz %gt, @inc, @next

@inc
  %count =l add %count, 1
  jmp @next

@next
  %i =l add %i, 1
  jmp @loop

@done
  ret %count
}
