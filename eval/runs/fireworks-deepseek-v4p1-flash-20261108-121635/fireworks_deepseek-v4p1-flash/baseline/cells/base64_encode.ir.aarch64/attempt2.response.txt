export function l $base64_encode(l %src, l %n, l %out) {
@start
	%buf =l alloc8 32
	%scr =l alloc8 8
	%outp =l copy %out
	%i =l copy 0
	%count =l copy 0
	jmp @loop

@loop
	%more =w csltl %i, %n
	jnz %more, @group, @done

@group
	%rem =l sub %n, %i
	%ge3 =w csgel %rem, 3
	jnz %ge3, @g3, @chk2

@chk2
	%eq2 =w ceql %rem, 2
	jnz %eq2, @g2, @g1

@g3
	%pa3 =l add %src, %i
	%ba3 =l loadub %pa3
	%pb3 =l add %pa3, 1
	%bb3 =l loadub %pb3
	%pc3 =l add %pa3, 2
	%bc3 =l loadub %pc3
	%ta3 =l shl %ba3, 16
	%tb3 =l shl %bb3, 8
	%ua3 =l or %ta3, %tb3
	%v3 =l or %ua3, %bc3
	%w3a =l shr %v3, 18
	%x3a =l and %w3a, 63
	%w3b =l shr %v3, 12
	%x3b =l and %w3b, 63
	%w3c =l shr %v3, 6
	%x3c =l and %w3c, 63
	%x3d =l and %v3, 63
	storel %x3a, %buf
	%q3a =l add %buf, 8
	storel %x3b, %q3a
	%q3b =l add %buf, 16
	storel %x3c, %q3b
	%q3c =l add %buf, 24
	storel %x3d, %q3c
	%i =l add %i, 3
	%k =l copy 4
	jmp @emit

@g2
	%pa2 =l add %src, %i
	%ba2 =l loadub %pa2
	%pb2 =l add %pa2, 1
	%bb2 =l loadub %pb2
	%ta2 =l shl %ba2, 16
	%tb2 =l shl %bb2, 8
	%v2 =l or %ta2, %tb2
	%w2a =l shr %v2, 18
	%x2a =l and %w2a, 63
	%w2b =l shr %v2, 12
	%x2b =l and %w2b, 63
	%w2c =l shr %v2, 6
	%x2c =l and %w2c, 63
	storel %x2a, %buf
	%q2a =l add %buf, 8
	storel %x2b, %q2a
	%q2b =l add %buf, 16
	storel %x2c, %q2b
	%i =l add %i, 2
	%k =l copy 3
	jmp @emit

@g1
	%pa1 =l add %src, %i
	%ba1 =l loadub %pa1
	%v1 =l shl %ba1, 16
	%w1a =l shr %v1, 18
	%x1a =l and %w1a, 63
	%w1b =l shr %v1, 12
	%x1b =l and %w1b, 63
	storel %x1a, %buf
	%q1a =l add %buf, 8
	storel %x1b, %q1a
	%i =l add %i, 1
	%k =l copy 2
	jmp @emit

@emit
	%j =l copy 0
	jmp @emit_loop

@emit_loop
	%jl =w csltl %j, %k
	jnz %jl, @emit_body, @pad_loop

@emit_body
	%jo =l shl %j, 3
	%bp =l add %buf, %jo
	%il =l loadub %bp
	storel %il, %scr
	%idx =w loadw %scr
	%lt26 =w csltw %idx, 26
	jnz %lt26, @m_up, @m_chk2

@m_up
	%cu =w add %idx, 65
	storeb %cu, %outp
	jmp @emit_next

@m_chk2
	%lt52 =w csltw %idx, 52
	jnz %lt52, @m_lo, @m_chk3

@m_lo
	%tl =w sub %idx, 26
	%cl =w add %tl, 97
	storeb %cl, %outp
	jmp @emit_next

@m_chk3
	%lt62 =w csltw %idx, 62
	jnz %lt62, @m_di, @m_chk4

@m_di
	%td =w sub %idx, 52
	%cd =w add %td, 48
	storeb %cd, %outp
	jmp @emit_next

@m_chk4
	%eq62 =w ceqw %idx, 62
	jnz %eq62, @m_pl, @m_sl

@m_pl
	storeb 43, %outp
	jmp @emit_next

@m_sl
	storeb 47, %outp
	jmp @emit_next

@emit_next
	%outp =l add %outp, 1
	%j =l add %j, 1
	jmp @emit_loop

@pad_loop
	%pl =w csltl %k, 4
	jnz %pl, @pad_body, @group_done

@pad_body
	storeb 61, %outp
	%outp =l add %outp, 1
	%k =l add %k, 1
	jmp @pad_loop

@group_done
	%count =l add %count, 4
	jmp @loop

@done
	ret %count
}
