export function l $base64_encode(l %src, l %n, l %out) {
@start
  %buf =l alloc4 4
  %count =l copy 0
  %i =l copy 0
  %outptr =l copy %out
  jmp @loop

@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @end

@body
  %rem =l sub %n, %i
  %c3 =w csgel %rem, 3
  jnz %c3, @do3, @check2

@check2
  %c2 =w ceql %rem, 2
  jnz %c2, @do2, @do1

@do3
  %p0 =l add %src, %i
  %b0 =l loadub %p0
  %p1 =l add %p0, 1
  %b1 =l loadub %p1
  %p2 =l add %p0, 2
  %b2 =l loadub %p2

  %t0 =l shl %b0, 16
  %t1 =l shl %b1, 8
  %v =l or %t0, %t1
  %v =l or %v, %b2

  %i0 =l shr %v, 18
  %i0 =l and %i0, 63
  %i1 =l shr %v, 12
  %i1 =l and %i1, 63
  %i2 =l shr %v, 6
  %i2 =l and %i2, 63
  %i3 =l and %v, 63

  %i0w =w truncl %i0
  storeb %i0w, %buf
  %buf1 =l add %buf, 1
  %i1w =w truncl %i1
  storeb %i1w, %buf1
  %buf2 =l add %buf, 2
  %i2w =w truncl %i2
  storeb %i2w, %buf2
  %buf3 =l add %buf, 3
  %i3w =w truncl %i3
  storeb %i3w, %buf3

  %k =l copy 4
  %i =l add %i, 3
  jmp @emit

@do2
  %p0 =l add %src, %i
  %b0 =l loadub %p0
  %p1 =l add %p0, 1
  %b1 =l loadub %p1

  %t0 =l shl %b0, 16
  %t1 =l shl %b1, 8
  %v =l or %t0, %t1

  %i0 =l shr %v, 18
  %i0 =l and %i0, 63
  %i1 =l shr %v, 12
  %i1 =l and %i1, 63
  %i2 =l shr %v, 6
  %i2 =l and %i2, 63

  %i0w =w truncl %i0
  storeb %i0w, %buf
  %buf1 =l add %buf, 1
  %i1w =w truncl %i1
  storeb %i1w, %buf1
  %buf2 =l add %buf, 2
  %i2w =w truncl %i2
  storeb %i2w, %buf2

  %k =l copy 3
  %i =l add %i, 2
  jmp @emit

@do1
  %p0 =l add %src, %i
  %b0 =l loadub %p0
  %v =l shl %b0, 16

  %i0 =l shr %v, 18
  %i0 =l and %i0, 63
  %i1 =l shr %v, 12
  %i1 =l and %i1, 63

  %i0w =w truncl %i0
  storeb %i0w, %buf
  %buf1 =l add %buf, 1
  %i1w =w truncl %i1
  storeb %i1w, %buf1

  %k =l copy 2
  %i =l add %i, 1
  jmp @emit

@emit
  %j =l copy 0
  jmp @emit_loop

@emit_loop
  %cond_j =w csltl %j, %k
  jnz %cond_j, @emit_body, @emit_pad

@emit_body
  %bp =l add %buf, %j
  %idx =l loadub %bp
  %lt26 =w csltl %idx, 26
  jnz %lt26, @map_upper, @check_lower

@map_upper
  %ch =l add %idx, 65
  jmp @map_done

@check_lower
  %lt52 =w csltl %idx, 52
  jnz %lt52, @map_lower, @check_digit

@map_lower
  %t =l sub %idx, 26
  %ch =l add %t, 97
  jmp @map_done

@check_digit
  %lt62 =w csltl %idx, 62
  jnz %lt62, @map_digit, @check_special

@map_digit
  %t =l sub %idx, 52
  %ch =l add %t, 48
  jmp @map_done

@check_special
  %is62 =w ceql %idx, 62
  jnz %is62, @map_plus, @map_slash

@map_plus
  %ch =l copy 43
  jmp @map_done

@map_slash
  %ch =l copy 47
  jmp @map_done

@map_done
  %chw =w truncl %ch
  storeb %chw, %outptr
  %outptr =l add %outptr, 1
  %j =l add %j, 1
  jmp @emit_loop

@emit_pad
  %need_pad =w csltl %k, 4
  jnz %need_pad, @pad_loop, @emit_done

@pad_loop
  %cond_pad =w csltl %k, 4
  jnz %cond_pad, @pad_body, @emit_done

@pad_body
  %eq =l copy 61
  %eqw =w truncl %eq
  storeb %eqw, %outptr
  %outptr =l add %outptr, 1
  %k =l add %k, 1
  jmp @pad_loop

@emit_done
  %count =l add %count, 4
  jmp @loop

@end
  ret %count
}
