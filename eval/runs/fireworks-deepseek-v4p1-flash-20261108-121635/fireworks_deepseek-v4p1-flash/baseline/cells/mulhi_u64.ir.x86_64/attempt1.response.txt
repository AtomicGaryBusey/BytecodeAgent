export function l $mulhi_u64(l %a, l %b) {
@start
  %a0 =l and %a, 4294967295
  %a1 =l shr %a, 32
  %b0 =l and %b, 4294967295
  %b1 =l shr %b, 32
  %p00 =l mul %a0, %b0
  %p01 =l mul %a0, %b1
  %p10 =l mul %a1, %b0
  %p11 =l mul %a1, %b1
  %mid =l add %p01, %p10
  %c1w =w cultl %mid, %p01
  %c1 =l extuw %c1w
  %p00hi =l shr %p00, 32
  %mid2 =l add %mid, %p00hi
  %c2w =w cultl %mid2, %mid
  %c2 =l extuw %c2w
  %carry =l add %c1, %c2
  %mid2hi =l shr %mid2, 32
  %carry_sh =l shl %carry, 32
  %tmp =l add %p11, %mid2hi
  %high =l add %tmp, %carry_sh
  ret %high
}
