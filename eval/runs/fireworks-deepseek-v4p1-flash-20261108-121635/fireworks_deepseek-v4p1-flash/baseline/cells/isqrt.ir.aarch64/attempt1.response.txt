export function l $isqrt(l %n) {
@start
  %lo0 =l copy 0
  %hi0 =l copy 4294967295
  %ans0 =l copy 0
  jmp @loop

@loop
  %lo =l phi @start %lo0, @update %lo_new, @reduce %lo_r
  %hi =l phi @start %hi0, @update %hi_u, @reduce %hi_new
  %ans =l phi @start %ans0, @update %ans_new, @reduce %ans_r
  %cmp =w culel %lo, %hi
  jnz %cmp, @body, @done

@body
  %sum =l add %lo, %hi
  %sum1 =l add %sum, 1
  %mid =l udiv %sum1, 2
  %sq =l mul %mid, %mid
  %le =w culel %sq, %n
  jnz %le, @update, @reduce

@update
  %lo_new =l add %mid, 1
  %ans_new =l copy %mid
  %hi_u =l copy %hi
  jmp @loop

@reduce
  %hi_new =l sub %mid, 1
  %lo_r =l copy %lo
  %ans_r =l copy %ans
  jmp @loop

@done
  ret %ans
}
