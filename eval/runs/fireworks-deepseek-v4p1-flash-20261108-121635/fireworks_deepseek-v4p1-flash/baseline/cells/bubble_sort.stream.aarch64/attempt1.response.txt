bubble_sort:
    cmp x1, #2
    b.lt .Lret
    mov x9, x0
    mov x10, x1
    sub x10, x10, #1
    mov x2, #0
.Louter:
    cmp x2, x10
    b.ge .Lret
    mov x3, #0
    sub x11, x10, x2
.Linner:
    cmp x3, x11
    b.ge .Lnext_outer
    ldr x4, [x9, x3, lsl #3]
    add x5, x3, #1
    ldr x5, [x9, x5, lsl #3]
    cmp x4, x5
    b.le .Lno_swap
    str x5, [x9, x3, lsl #3]
    add x6, x3, #1
    str x4, [x9, x6, lsl #3]
.Lno_swap:
    add x3, x3, #1
    b .Linner
.Lnext_outer:
    add x2, x2, #1
    b .Louter
.Lret:
    mov x0, #0
    ret
