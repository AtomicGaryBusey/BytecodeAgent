    add x1, x0, x1
    add x3, x2, x3
.Lloop:
    cmp x0, x1
    b.hs .Ldone
    ldrsb w5, [x0], #1
    cmp w5, #0
    b.lt .Lneg
    add w6, w5, #1
    add x7, x0, x6
    cmp x7, x1
    b.hi .Lsrc_err
    add x8, x2, x6
    cmp x8, x3
    b.hi .Lout_lit
.Lcopy_lit:
    ldrb w9, [x0], #1
    strb w9, [x2], #1
    subs w6, w6, #1
    b.ne .Lcopy_lit
    b .Lloop
.Lneg:
    cmn w5, #128
    b.eq .Lloop
    rsb w6, w5, #1
