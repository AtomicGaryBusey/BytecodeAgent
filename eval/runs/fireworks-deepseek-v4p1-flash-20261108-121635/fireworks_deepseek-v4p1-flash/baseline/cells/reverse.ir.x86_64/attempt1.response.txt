export function l $reverse(l %a, l %n) {
@start
	%half =l div %n, 2
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cond =w csltl %i, %half
	jnz %cond, @body, @end
@body
	%j =l sub %n, 1
	%j2 =l sub %j, %i
	%off_i =l mul %i, 8
	%p =l add %a, %off_i
	%off_j =l mul %j2, 8
	%q =l add %a, %off_j
	%x =l loadl %p
	%y =l loadl %q
	storel %y, %p
	storel %x, %q
	%i1 =l add %i, 1
	jmp @loop
@end
	ret 0
}
