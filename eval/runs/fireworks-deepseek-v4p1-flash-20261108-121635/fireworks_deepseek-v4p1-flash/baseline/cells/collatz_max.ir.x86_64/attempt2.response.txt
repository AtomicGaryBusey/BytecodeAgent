export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%i0 =l copy 1
	jmp @outer
@outer
	%i =l phi @start %i0, @next %i_next
	%max =l phi @start %max0, @next %max_next
	%cond =w csgtl %i, %n
	jnz %cond, @end, @body
@body
	%x0 =l copy %i
	%cnt0 =l copy 0
	jmp @inner
@inner
	%x =l phi @body %x0, @odd %xo, @even %xe
	%cnt =l phi @body %cnt0, @odd %co, @even %ce
	%done =w ceql %x, 1
	jnz %done, @idone, @cont
@cont
	%oddbit =l and %x, 1
	%isodd =w ceql %oddbit, 1
	jnz %isodd, @odd, @even
@odd
	%t3 =l mul %x, 3
	%xo =l add %t3, 1
	%co =l add %cnt, 1
	jmp @inner
@even
	%xe =l shr %x, 1
	%ce =l add %cnt, 1
	jmp @inner
@idone
	%gt =w csgtl %cnt, %max
	jnz %gt, @upd, @next
@upd
	%max_new =l copy %cnt
	jmp @next
@next
	%max_next =l phi @idone %max, @upd %max_new
	%i_next =l add %i, 1
	jmp @outer
@end
	ret %max
}
