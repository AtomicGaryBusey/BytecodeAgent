export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%i0 =l copy 1
	jmp @outer
@outer
	%i =l phi @start %i0, @outer_next %i_next
	%max =l phi @start %max0, @outer_next %max_next
	%cond =w csgtl %i, %n
	jnz %cond, @end
	%x0 =l copy %i
	%cnt0 =l copy 0
	jmp @inner
@inner
	%x =l phi @outer %x0, @inner_odd %x_odd, @inner_even %x_even
	%cnt =l phi @outer %cnt0, @inner_odd %cnt_odd, @inner_even %cnt_even
	%done =w ceql %x, 1
	jnz %done, @inner_done
	%isodd =l and %x, 1
	%isodd_w =w ceql %isodd, 1
	jnz %isodd_w, @inner_odd
	jmp @inner_even
@inner_odd
	%x3 =l mul %x, 3
	%x_odd =l add %x3, 1
	%cnt_odd =l add %cnt, 1
	jmp @inner
@inner_even
	%x_even =l shr %x, 1
	%cnt_even =l add %cnt, 1
	jmp @inner
@inner_done
	%gt =w csgtl %cnt, %max
	jnz %gt, @update_max
	jmp @outer_next
@update_max
	%max_new =l copy %cnt
	jmp @outer_next
@outer_next
	%max_next =l phi @inner_done %max, @update_max %max_new
	%i_next =l add %i, 1
	jmp @outer
@end
	ret %max
}
