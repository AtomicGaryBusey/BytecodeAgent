export function l $isqrt(l %n) {
@start
    jmp @loop

@loop
    %lo =l phi @start 0, @take %lo_take, @drop %lo_drop
    %hi =l phi @start 4294967295, @take %hi_take, @drop %hi_drop
    %cond =w cultl %lo, %hi
    jnz %cond, @body, @done

@body
    %sum =l add %lo, %hi
    %sum1 =l add %sum, 1
    %mid =l shr %sum1, 1
    %sq =l mul %mid, %mid
    %le =w culel %sq, %n
    jnz %le, @take, @drop

@take
    %lo_take =l copy %mid
    %hi_take =l copy %hi
    jmp @loop

@drop
    %lo_drop =l copy %lo
    %hi_drop =l sub %mid, 1
    jmp @loop

@done
    ret %lo
}
