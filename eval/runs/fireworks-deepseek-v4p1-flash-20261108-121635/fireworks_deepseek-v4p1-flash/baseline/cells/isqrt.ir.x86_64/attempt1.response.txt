export function l $isqrt(l %n) {
@start
    jmp @loop

@loop
    %lo =l phi @start 0, @take %mid, @drop %lo_drop
    %hi =l phi @start 4294967295, @take %hi, @drop %hi_drop
    %cond =w cult %lo, %hi
    jnz %cond, @body, @done

@body
    %sum =l add %lo, %hi
    %sum1 =l add %sum, 1
    %mid =l shr %sum1, 1
    %sq =l mul %mid, %mid
    %le =w cule %sq, %n
    jnz %le, @take, @drop

@take
    jmp @loop

@drop
    %lo_drop =l copy %lo
    %hi_drop =l sub %hi, 1
    jmp @loop

@done
    ret %lo
}
