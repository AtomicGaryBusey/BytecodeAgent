crc32:

    cbz x1, .Ldone
    mov w4, #0x8320
    movk w4, #0xEDB8, lsl #16
.Lbyteloop:
    ldrb w3, [x0], #1
    eor w2, w2, w3
    mov w5, #8
.Lbitloop:
    tbz w2, #0, .Lshift
    lsr w2, w2, #1
    eor w2, w2, w4
    b .Lnext
.Lshift:
    lsr w2, w2, #1
.Lnext:
    subs w5, w5, #1
    b.ne .Lbitloop
    subs x1, x1, #1
    b.ne .Lbyteloop
.Ldone:
    mvn w2, w2
    mov x0, w2
