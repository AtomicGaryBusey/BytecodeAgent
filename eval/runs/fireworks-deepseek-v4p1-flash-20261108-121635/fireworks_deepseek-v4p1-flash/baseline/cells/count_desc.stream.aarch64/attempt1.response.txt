count_desc:
    cmp x1, #2
    b.lt .Lzero
    mov x2, x1
    mov x3, x0
    mov x0, #0
    sub x2, x2, #1
    mov x4, #0
.Lloop:
    add x7, x3, x4
    ldr x5, [x7]
    ldr x6, [x7, #8]
    cmp x5, x6
    b.le .Lnext
    add x0, x0, #1
.Lnext:
    add x4, x4, #8
    subs x2, x2, #1
    b.ne .Lloop
    ret
.Lzero:

    ret
