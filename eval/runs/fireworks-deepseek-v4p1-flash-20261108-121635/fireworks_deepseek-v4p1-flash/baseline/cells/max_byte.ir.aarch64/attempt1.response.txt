export function l $max_byte(l %p, l %n) {
@start
	%z =w ceql %n, 0
	jnz %z, @zero, @init
@zero
	ret 0
@init
	jmp @loop
@loop
	%i =l phi @init 0, @cont %next
	%m =l phi @init 0, @cont %newm
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l add %p, %i
	%b =l loadub %off
	%gt =w csgtl %b, %m
	jnz %gt, @update, @skip
@update
	jmp @cont
@skip
	jmp @cont
@cont
	%newm =l phi @update %b, @skip %m
	%next =l add %i, 1
	jmp @loop
@done
	ret %m
}
