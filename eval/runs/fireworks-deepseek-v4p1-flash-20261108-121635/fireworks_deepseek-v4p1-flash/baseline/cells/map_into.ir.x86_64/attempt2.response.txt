export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  %zero =l copy 0
  %cond =w ceql %n, %zero
  jnz %cond, @end, @init

@init
  %src0 =l copy %src
  %dst0 =l copy %dst
  %n0 =l copy %n
  %zero0 =l copy 0
  jmp @loop

@loop
  %srcp =l phi @init %src0, @cont %srcp_next
  %dstp =l phi @init %dst0, @cont %dstp_next
  %count =l phi @init %zero0, @cont %count_next
  %rem =l phi @init %n0, @cont %rem_next
  %remzero =w ceql %rem, 0
  jnz %remzero, @end_loop, @body

@body
  %x =l loadl %srcp
  %y =l call %f(l %x)
  storel %y, %dstp
  %less =w csltl %y, %x
  jnz %less, @inc, @noinc

@inc
  %count_inc =l add %count, 1
  jmp @cont

@noinc
  %count_inc2 =l copy %count
  jmp @cont

@cont
  %count_next =l phi @inc %count_inc, @noinc %count_inc2
  %srcp_next =l add %srcp, 8
  %dstp_next =l add %dstp, 8
  %rem_next =l sub %rem, 1
  jmp @loop

@end_loop
  ret %count

@end
  ret %zero
}
