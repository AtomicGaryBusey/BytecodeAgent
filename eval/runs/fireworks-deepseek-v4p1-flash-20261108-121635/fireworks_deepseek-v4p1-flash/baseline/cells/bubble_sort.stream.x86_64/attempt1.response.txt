bubble_sort:
    xor eax, eax
    cmp rsi, 1
    jle .done
    mov rcx, rsi
.outer:
    xor edx, edx
    mov r10, rcx
    dec r10
.inner:
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r8, r9
    jle .no_swap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], r8
.no_swap:
    inc rdx
    cmp rdx, r10
    jl .inner
    dec rcx
    cmp rcx, 1
    jg .outer
.done:
    ret
