export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %zero =l copy 0
    jmp @loop

@loop
    %pc =l phi @start %zero, @op1 %pc1, @op2 %pc2, @op3 %pc3, @op4 %pc4, @op5 %pc5, @op6 %pc6, @op7 %pc7, @op8 %pc8, @op9 %pc9, @op10 %pc10, @op11 %pc11, @default %pc_def
    %sp =l phi @start %zero, @op1 %sp1, @op2 %sp2, @op3 %sp3, @op4 %sp4, @op5 %sp5, @op6 %sp6, @op7 %sp7, @op8 %sp8, @op9 %sp9, @op10 %sp10, @op11 %sp11, @default %sp_def
    %cond_end =w ceql %pc, %len
    jnz %cond_end, @end, @body

@body
    %off =l mul %pc, 8
    %addr =l add %code, %off
    %op =l loadl %addr
    %is0 =w ceql %op, 0
    jnz %is0, @op0, @check1

@check1
    %is1 =w ceql %op, 1
    jnz %is1, @op1, @check2

@check2
    %is2 =w ceql %op, 2
    jnz %is2, @op2, @check3

@check3
    %is3 =w ceql %op, 3
    jnz %is3, @op3, @check4

@check4
    %is4 =w ceql %op, 4
    jnz %is4, @op4, @check5

@check5
    %is5 =w ceql %op, 5
    jnz %is5, @op5, @check6

@check6
    %is6 =w ceql %op, 6
    jnz %is6, @op6, @check7

@check7
    %is7 =w ceql %op, 7
    jnz %is7, @op7, @check8

@check8
    %is8 =w ceql %op, 8
    jnz %is8, @op8, @check9

@check9
    %is9 =w ceql %op, 9
    jnz %is9, @op9, @check10

@check10
    %is10 =w ceql %op, 10
    jnz %is10, @op10, @check11

@check11
    %is11 =w ceql %op, 11
    jnz %is11, @op11, @default

@op0
    jmp @end

@op1
    %off1 =l add %pc, 1
    %off1b =l mul %off1, 8
    %addr1 =l add %code, %off1b
    %v1 =l loadl %addr1
    %spoff1 =l mul %sp, 8
    %staddr1 =l add %stack, %spoff1
    storel %v1, %staddr1
    %sp1 =l add %sp, 1
    %pc1 =l add %pc, 2
    jmp @loop

@op2
    %spm1_2 =l sub %sp, 1
    %offb2 =l mul %spm1_2, 8
    %addrb2 =l add %stack, %offb2
    %b2 =l loadl %addrb2
    %spm2_2 =l sub %sp, 2
    %offa2 =l mul %spm2_2, 8
    %addra2 =l add %stack, %offa2
    %a2 =l loadl %addra2
    %r2 =l add %a2, %b2
    storel %r2, %addra2
    %sp2 =l sub %sp, 1
    %pc2 =l add %pc, 1
    jmp @loop

@op3
    %spm1_3 =l sub %sp, 1
    %offb3 =l mul %spm1_3, 8
    %addrb3 =l add %stack, %offb3
    %b3 =l loadl %addrb3
    %spm2_3 =l sub %sp, 2
    %offa3 =l mul %spm2_3, 8
    %addra3 =l add %stack, %offa3
    %a3 =l loadl %addra3
    %r3 =l sub %a3, %b3
    storel %r3, %addra3
    %sp3 =l sub %sp, 1
    %pc3 =l add %pc, 1
    jmp @loop

@op4
    %spm1_4 =l sub %sp, 1
    %offb4 =l mul %spm1_4, 8
    %addrb4 =l add %stack, %offb4
    %b4 =l loadl %addrb4
    %spm2_4 =l sub %sp, 2
    %offa4 =l mul %spm2_4, 8
    %addra4 =l add %stack, %offa4
    %a4 =l loadl %addra4
    %r4 =l mul %a4, %b4
    storel %r4, %addra4
    %sp4 =l sub %sp, 1
    %pc4 =l add %pc, 1
    jmp @loop

@op5
    %spm1_5 =l sub %sp, 1
    %off5 =l mul %spm1_5, 8
    %addr5 =l add %stack, %off5
    %v5 =l loadl %addr5
    %off5b =l mul %sp, 8
    %addr5b =l add %stack, %off5b
    storel %v5, %addr5b
    %sp5 =l add %sp, 1
    %pc5 =l add %pc, 1
    jmp @loop

@op6
    %spm1_6 =l sub %sp, 1
    %offb6 =l mul %spm1_6, 8
    %addrb6 =l add %stack, %offb6
    %b6 =l loadl %addrb6
    %spm2_6 =l sub %sp, 2
    %offa6 =l mul %spm2_6, 8
    %addra6 =l add %stack, %offa6
    %a6 =l loadl %addra6
    storel %a6, %addrb6
    storel %b6, %addra6
    %sp6 =l copy %sp
    %pc6 =l add %pc, 1
    jmp @loop

@op7
    %sp7 =l sub %sp, 1
    %pc7 =l add %pc, 1
    jmp @loop

@op8
    %spm2_8 =l sub %sp, 2
    %off8 =l mul %spm2_8, 8
    %addr8 =l add %stack, %off8
    %v8 =l loadl %addr8
    %off8b =l mul %sp, 8
    %addr8b =l add %stack, %off8b
    storel %v8, %addr8b
    %sp8 =l add %sp, 1
    %pc8 =l add %pc, 1
    jmp @loop

@op9
    %spm1_9 =l sub %sp, 1
    %offb9 =l mul %spm1_9, 8
    %addrb9 =l add %stack, %offb9
    %b9 =l loadl %addrb9
    %spm2_9 =l sub %sp, 2
    %offa9 =l mul %spm2_9, 8
    %addra9 =l add %stack, %offa9
    %a9 =l loadl %addra9
    %c9 =w csltl %a9, %b9
    %r9 =l extuw %c9
    storel %r9, %addra9
    %sp9 =l sub %sp, 1
    %pc9 =l add %pc, 1
    jmp @loop

@op10
    %spm1_10 =l sub %sp, 1
    %off10 =l mul %spm1_10, 8
    %addr10 =l add %stack, %off10
    %v10 =l loadl %addr10
    %sp10 =l sub %sp, 1
    %offt10 =l add %pc, 1
    %offt10b =l mul %offt10, 8
    %addrt10 =l add %code, %offt10b
    %t10 =l loadl %addrt10
    %cond10 =w ceql %v10, 0
    %cond10l =l extuw %cond10
    %pc_plus_2_10 =l add %pc, 2
    %diff10 =l sub %t10, %pc_plus_2_10
    %sel10 =l mul %cond10l, %diff10
    %pc10 =l add %pc_plus_2_10, %sel10
    jmp @loop

@op11
    %offt11 =l add %pc, 1
    %offt11b =l mul %offt11, 8
    %addrt11 =l add %code, %offt11b
    %t11 =l loadl %addrt11
    %pc11 =l copy %t11
    %sp11 =l copy %sp
    jmp @loop

@default
    %pc_def =l add %pc, 1
    %sp_def =l copy %sp
    jmp @loop

@end
    %sp_end =l copy %sp
    jmp @clear_init

@clear_init
    %i =l phi @end %sp_end, @clear_body %i_next
    %cond_clear =w csltl %i, %cap
    jnz %cond_clear, @clear_body, @clear_done

@clear_body
    %offc =l mul %i, 8
    %addrc =l add %stack, %offc
    storel %zero, %addrc
    %i_next =l add %i, 1
    jmp @clear_init

@clear_done
    %cond_empty =w ceql %sp, 0
    jnz %cond_empty, @ret_zero, @ret_top

@ret_zero
    ret 0

@ret_top
    %spm1_end =l sub %sp, 1
    %offt_end =l mul %spm1_end, 8
    %addrt_end =l add %stack, %offt_end
    %top =l loadl %addrt_end
    ret %top
}
