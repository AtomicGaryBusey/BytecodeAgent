export function l $qsort_pairs(l %a, l %n) {
@start
  %c =w csltl %n, 2
  jnz %c, @ret0, @body

@ret0
  ret 0

@body
  %pk =l loadl %a
  %a8 =l add %a, 8
  %pv =l loadl %a8
  jmp @loop_check

@loop_check
  %i =l phi @body 1, @inc_j %i_next
  %j =l phi @body 1, @inc_j %j_next
  %c_loop =w csltl %j, %n
  jnz %c_loop, @loop_body, @loop_end

@loop_body
  %j16 =l mul %j, 16
  %jaddr =l add %a, %j16
  %jk =l loadl %jaddr
  %jv_addr =l add %jaddr, 8
  %jv =l loadl %jv_addr
  %c1 =w csltl %jk, %pk
  %c2 =w ceql %jk, %pk
  %c3 =w csltl %jv, %pv
  %c4 =w and %c2, %c3
  %less =w or %c1, %c4
  jnz %less, @swap, @no_swap

@swap
  %i16 =l mul %i, 16
  %iaddr =l add %a, %i16
  %ik =l loadl %iaddr
  %iv_addr =l add %iaddr, 8
  %iv =l loadl %iv_addr
  storel %jk, %iaddr
  storel %jv, %iv_addr
  storel %ik, %jaddr
  storel %iv, %jv_addr
  %i_next_swap =l add %i, 1
  jmp @inc_j

@no_swap
  %i_next_no =l copy %i
  jmp @inc_j

@inc_j
  %i_next =l phi @swap %i_next_swap, @no_swap %i_next_no
  %j_next =l add %j, 1
  jmp @loop_check

@loop_end
  %i_minus_1 =l sub %i, 1
  %a0k =l loadl %a
  %a0v_addr =l add %a, 8
  %a0v =l loadl %a0v_addr
  %im1_16 =l mul %i_minus_1, 16
  %im1_addr =l add %a, %im1_16
  %im1k =l loadl %im1_addr
  %im1v_addr =l add %im1_addr, 8
  %im1v =l loadl %im1v_addr
  storel %im1k, %a
  storel %im1v, %a0v_addr
  storel %a0k, %im1_addr
  storel %a0v, %im1v_addr
  %left_n =l sub %i, 1
  %r1 =l call $qsort_pairs(l %a, l %left_n)
  %i16_end =l mul %i, 16
  %right_a =l add %a, %i16_end
  %right_n =l sub %n, %i
  %r2 =l call $qsort_pairs(l %right_a, l %right_n)
  ret 0
}
