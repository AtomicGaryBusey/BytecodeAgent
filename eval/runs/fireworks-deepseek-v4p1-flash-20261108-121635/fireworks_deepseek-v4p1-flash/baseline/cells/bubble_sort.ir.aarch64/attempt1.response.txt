export function l $bubble_sort(l %a, l %n) {
@start
  %cond0 =w cslel %n, 1
  jnz %cond0, @ret, @init
@init
  %n1 =l sub %n, 1
  %i0 =l copy 0
  jmp @outer_loop
@outer_loop
  %i =l phi @init %i0, @outer_inc %i_next
  %cond1 =w csltl %i, %n1
  jnz %cond1, @inner_init, @ret
@inner_init
  %j0 =l copy 0
  %tmp =l sub %n, %i
  %lim =l sub %tmp, 1
  jmp @inner_loop
@inner_loop
  %j =l phi @inner_init %j0, @inner_inc %j_next
  %cond2 =w csltl %j, %lim
  jnz %cond2, @inner_body, @outer_inc
@inner_body
  %off =l mul %j, 8
  %addr1 =l add %a, %off
  %val1 =l loadl %addr1
  %addr2 =l add %addr1, 8
  %val2 =l loadl %addr2
  %cond3 =w csgtl %val1, %val2
  jnz %cond3, @swap, @inner_inc
@swap
  storel %val2, %addr1
  storel %val1, %addr2
  jmp @inner_inc
@inner_inc
  %j_next =l add %j, 1
  jmp @inner_loop
@outer_inc
  %i_next =l add %i, 1
  jmp @outer_loop
@ret
  ret 0
}
