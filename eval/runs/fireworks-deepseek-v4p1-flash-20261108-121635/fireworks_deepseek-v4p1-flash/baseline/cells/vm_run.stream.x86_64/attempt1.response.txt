vm_run:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx
    xor r15d, r15d
    xor ebp, ebp
.Lloop:
    cmp rbp, r12
    jae .Lhalt
    mov rax, qword ptr [rbx + rbp*8]
    cmp rax, 0
    je .Lhalt
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbp
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rbx + rbp*8 + 8]
    mov qword ptr [r13 + r15*8], rax
    inc r15
    add rbp, 2
    jmp .Lloop
.Ladd:
    dec r15
    mov r8, qword ptr [r13 + r15*8]
    dec r15
    mov r9, qword ptr [r13 + r15*8]
    add r9, r8
    mov qword ptr [r13 + r15*8], r9
    inc r15
    inc rbp
    jmp .Lloop
.Lsub:
    dec r15
    mov r8, qword ptr [r13 + r15*8]
    dec r15
    mov r9, qword ptr [r13 + r15*8]
    sub r9, r8
    mov qword ptr [r13 + r15*8], r9
    inc r15
    inc rbp
    jmp .Lloop
.Lmul:
    dec r15
    mov r8, qword ptr [r13 + r15*8]
    dec r15
    mov r9, qword ptr [r13 + r15*8]
    imul r9, r8
    mov qword ptr [r13 + r15*8], r9
    inc r15
    inc rbp
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [r13 + r15*8 - 8]
    mov qword ptr [r13 + r15*8], rax
    inc r15
    inc rbp
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [r13 + r15*8 - 8]
    mov r8, qword ptr [r13 + r15*8 - 16]
    mov qword ptr [r13 + r15*8 - 8], r8
    mov qword ptr [r13 + r15*8 - 16], rax
    inc rbp
    jmp .Lloop
.Ldrop:
    dec r15
    inc rbp
    jmp .Lloop
.Lover:
    mov rax, qword ptr [r13 + r15*8 - 16]
    mov qword ptr [r13 + r15*8], rax
    inc r15
    inc rbp
    jmp .Lloop
.Llt:
    dec r15
    mov r8, qword ptr [r13 + r15*8]
    dec r15
    mov r9, qword ptr [r13 + r15*8]
    cmp r9, r8
    setl al
    movzx rax, al
    mov qword ptr [r13 + r15*8], rax
    inc r15
    inc rbp
    jmp .Lloop
.Ljz:
    dec r15
    mov rax, qword ptr [r13 + r15*8]
    test rax, rax
    jne .Ljz_not_zero
    mov rbp, qword ptr [rbx + rbp*8 + 8]
    jmp .Lloop
.Ljz_not_zero:
    add rbp, 2
    jmp .Lloop
.Ljmp:
    mov rbp, qword ptr [rbx + rbp*8 + 8]
    jmp .Lloop
.Lhalt:
    test r15, r15
    jz .Lempty
    mov rax, qword ptr [r13 + r15*8 - 8]
    jmp .Lzero
.Lempty:
    xor eax, eax
.Lzero:
    mov r8, r15
.Lzero_loop:
    cmp r8, r14
    jae .Ldone
    mov qword ptr [r13 + r8*8], 0
    inc r8
    jmp .Lzero_loop
.Ldone:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
