export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %cmp =w cultl %dst, %src
  jnz %cmp, @forward, @backward
@forward
  %zero =l copy 0
  jmp @forward_loop
@forward_loop
  %i =l phi @forward %zero, @forward_body %inext
  %cond =w csltl %i, %n
  jnz %cond, @forward_body, @done
@forward_body
  %off =l mul %i, 8
  %sptr =l add %src, %off
  %dptr =l add %dst, %off
  %val =l loadl %sptr
  storel %val, %dptr
  %inext =l add %i, 1
  jmp @forward_loop
@backward
  jmp @backward_loop
@backward_loop
  %j =l phi @backward %n, @backward_body %jdec
  %jz =w ceql %j, 0
  jnz %jz, @done, @backward_body
@backward_body
  %jdec =l sub %j, 1
  %off2 =l mul %jdec, 8
  %sptr2 =l add %src, %off2
  %dptr2 =l add %dst, %off2
  %val2 =l loadl %sptr2
  storel %val2, %dptr2
  jmp @backward_loop
@done
  %ret =l copy 0
  ret %ret
}
