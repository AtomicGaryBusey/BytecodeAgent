memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.lo .Lforward
    sub x3, x2, #1
.Lbackward_loop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    subs x3, x3, #1
    b.ge .Lbackward_loop
    b .Ldone
.Lforward:
    mov x3, #0
.Lforward_loop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x3, x3, #1
    cmp x3, x2
    b.lo .Lforward_loop
.Ldone:
    mov x0, #0
    ret
