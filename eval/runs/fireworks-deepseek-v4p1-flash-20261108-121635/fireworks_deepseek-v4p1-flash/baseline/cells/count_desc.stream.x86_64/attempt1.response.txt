    cmp rsi, 2
    jl .Ldone
    lea rcx, [rsi - 1]
    xor edx, edx
    mov r8, qword ptr [rdi]
.Lloop:
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r8, r9
    setg r10b
    movzx r10d, r10b

    cmp rdx, rcx
    jb .Lloop
.Ldone:
