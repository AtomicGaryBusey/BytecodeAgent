binary_search:
    mov x3, #0
    mov x4, x1
.Lloop:
    cmp x3, x4
    b.ge .Lnot_found
    sub x5, x4, x3
    lsr x5, x5, #1
    add x5, x5, x3
    ldr x6, [x0, x5, lsl #3]
    cmp x6, x2
    b.eq .Lfound
    b.lt .Lgo_right
    mov x4, x5
    b .Lloop
.Lgo_right:
    add x3, x5, #1
    b .Lloop
.Lfound:
    mov x0, x5
    ret
.Lnot_found:
    movn x0, #0
    ret
