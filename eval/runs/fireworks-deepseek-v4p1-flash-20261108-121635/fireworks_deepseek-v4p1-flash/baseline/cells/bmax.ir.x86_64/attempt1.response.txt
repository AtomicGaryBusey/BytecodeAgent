export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    %i0 =l copy 1
    jmp @loop
@loop
    %i =l phi @start %i0, @next %inext
    %max =l phi @start %first, @next %maxnext
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %p =l add %a, %off
    %v =l loadl %p
    %gt =w csgtl %v, %max
    jnz %gt, @update, @next
@update
    jmp @next
@next
    %maxnext =l phi @update %v, @body %max
    %inext =l add %i, 1
    jmp @loop
@done
    ret %max
}
