export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %min =l shl 1, 63
    %bzero =w ceql %b, 0
    jnz %bzero, @zero, @check_ovf
@zero
    storel %a, %rem
    ret 0
@check_ovf
    %a_min =w ceql %a, %min
    %b_neg1 =w ceql %b, -1
    %ovf =w and %a_min, %b_neg1
    jnz %ovf, @overflow, @normal
@overflow
    storel 0, %rem
    ret %min
@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
