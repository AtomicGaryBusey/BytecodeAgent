export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @next %i_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %iszero =w ceql %val, 0
    jnz %iszero, @found_zero, @next

@next
    %i_next =l add %i, 1
    jmp @loop

@found_zero
    ret %i

@done
    ret %n
}
