export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %si_init =l copy 0
    %di_init =l copy 0
    jmp @main_cond

@main_cond
    %si =l phi @start %si_init, @main_next %si_next
    %di =l phi @start %di_init, @main_next %di_next
    %cond =w csltl %si, %n
    jnz %cond, @main_body, @done

@main_body
    %src_ptr =l add %src, %si
    %h =l loadsb %src_ptr
    %si1 =l add %si, 1
    %is_noop =w ceql %h, -128
    jnz %is_noop, @noop, @check_literal

@noop
    %si_next_noop =l copy %si1
    %di_next_noop =l copy %di
    jmp @main_next

@check_literal
    %is_ge0 =w csgel %h, 0
    jnz %is_ge0, @literal, @repeat

@literal
    %count =l add %h, 1
    %need_end =l add %si1, %count
    %src_ok =w cslel %need_end, %n
    jnz %src_ok, @literal_src_ok, @error_src

@literal_src_ok
    %di_end =l add %di, %count
    %cap_ok =w cslel %di_end, %cap
    jnz %cap_ok, @literal_cap_ok, @literal_overflow

@literal_cap_ok
    %i =l copy 0
    jmp @lit_copy_cond

@lit_copy_cond
    %i_phi =l phi @literal_cap_ok %i, @lit_copy_body %i_next
    %cond_i =w csltl %i_phi, %count
    jnz %cond_i, @lit_copy_body, @lit_copy_done

@lit_copy_body
    %src_off =l add %si1, %i_phi
    %src_byte_ptr =l add %src, %src_off
    %byte =w loadub %src_byte_ptr
    %out_off =l add %di, %i_phi
    %out_byte_ptr =l add %out, %out_off
    storeb %byte, %out_byte_ptr
    %i_next =l add %i_phi, 1
    jmp @lit_copy_cond

@lit_copy_done
    %si_next_lit =l add %si1, %count
    %di_next_lit =l add %di, %count
    jmp @main_next

@literal_overflow
    %rem =l sub %cap, %di
    %i_ov =l copy 0
    jmp @lit_ov_cond

@lit_ov_cond
    %i_ov_phi =l phi @literal_overflow %i_ov, @lit_ov_body %i_ov_next
    %cond_ov =w csltl %i_ov_phi, %rem
    jnz %cond_ov, @lit_ov_body, @lit_ov_done

@lit_ov_body
    %src_off_ov =l add %si1, %i_ov_phi
    %src_byte_ptr_ov =l add %src, %src_off_ov
    %byte_ov =w loadub %src_byte_ptr_ov
    %out_off_ov =l add %di, %i_ov_phi
    %out_byte_ptr_ov =l add %out, %out_off_ov
    storeb %byte_ov, %out_byte_ptr_ov
    %i_ov_next =l add %i_ov_phi, 1
    jmp @lit_ov_cond

@lit_ov_done
    ret -1

@repeat
    %count_rep =l sub 1, %h
    %need_end_rep =l add %si1, 1
    %src_ok_rep =w cslel %need_end_rep, %n
    jnz %src_ok_rep, @repeat_src_ok, @error_src

@repeat_src_ok
    %val_ptr =l add %src, %si1
    %val =w loadub %val_ptr
    %si2 =l add %si1, 1
    %di_end_rep =l add %di, %count_rep
    %cap_ok_rep =w cslel %di_end_rep, %cap
    jnz %cap_ok_rep, @repeat_cap_ok, @repeat_overflow

@repeat_cap_ok
    %j =l copy 0
    jmp @rep_cond

@rep_cond
    %j_phi =l phi @repeat_cap_ok %j, @rep_body %j_next
    %cond_j =w csltl %j_phi, %count_rep
    jnz %cond_j, @rep_body, @rep_done

@rep_body
    %out_off_rep =l add %di, %j_phi
    %out_ptr_rep =l add %out, %out_off_rep
    storeb %val, %out_ptr_rep
    %j_next =l add %j_phi, 1
    jmp @rep_cond

@rep_done
    %si_next_rep =l copy %si2
    %di_next_rep =l add %di, %count_rep
    jmp @main_next

@repeat_overflow
    %rem_rep =l sub %cap, %di
    %k =l copy 0
    jmp @rep_ov_cond

@rep_ov_cond
    %k_phi =l phi @repeat_overflow %k, @rep_ov_body %k_next
    %cond_k =w csltl %k_phi, %rem_rep
    jnz %cond_k, @rep_ov_body, @rep_ov_done

@rep_ov_body
    %out_off_ov_rep =l add %di, %k_phi
    %out_ptr_ov_rep =l add %out, %out_off_ov_rep
    storeb %val, %out_ptr_ov_rep
    %k_next =l add %k_phi, 1
    jmp @rep_ov_cond

@rep_ov_done
    ret -1

@main_next
    %si_next =l phi @noop %si_next_noop, @lit_copy_done %si_next_lit, @rep_done %si_next_rep
    %di_next =l phi @noop %di_next_noop, @lit_copy_done %di_next_lit, @rep_done %di_next_rep
    jmp @main_cond

@error_src
    ret -2

@done
    ret %di
}
