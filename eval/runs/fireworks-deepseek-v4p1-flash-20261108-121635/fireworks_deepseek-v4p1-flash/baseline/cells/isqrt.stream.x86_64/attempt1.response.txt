isqrt:
    mov r8, rdi
    xor eax, eax
    xor r9d, r9d
    mov ecx, 0xffffffff
.Lloop:
    cmp rax, rcx
    ja .Ldone
    mov rdx, rax
    add rdx, rcx
    shr rdx, 1
    mov rsi, rdx
    imul rsi, rsi
    cmp rsi, r8
    ja .Lhigh
    mov r9, rdx
    lea rax, [rdx + 1]
    jmp .Lloop
.Lhigh:
    lea rcx, [rdx - 1]
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
