export function l $bmin_empty(l %a, l %n) {
@start
  %iszero =w ceql %n, 0
  jnz %iszero, @empty, @init

@empty
  ret 9223372036854775807

@init
  %first =l loadl %a
  jmp @loop

@loop
  %i =l phi @init 1, @update %next_i
  %min =l phi @init %first, @update %next_min
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %isless =w csltl %val, %min
  jnz %isless, @less, @notless

@less
  jmp @update

@notless
  jmp @update

@update
  %next_min =l phi @less %val, @notless %min
  %next_i =l add %i, 1
  jmp @loop

@done
  ret %min
}
