packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    xor r8d, r8d
    xor r9d, r9d
    test rsi, rsi
    jle .Ldone
.Lloop:
    cmp r9, rsi
    jge .Ldone
    movzx eax, byte ptr [rdi + r9]
    mov r10, r9
    xor r11d, r11d
.Lcount_run:
    cmp r10, rsi
    jge .Lrun_done
    cmp r11, 128
    jge .Lrun_done
    cmp byte ptr [rdi + r10], al
    jne .Lrun_done
    inc r11
    inc r10
    jmp .Lcount_run
.Lrun_done:
    cmp r11, 3
    jl .Lliteral
    lea r10, [r8 + 2]
    cmp r10, rcx
    jg .Lfail
    mov eax, 257
    sub eax, r11d
    mov byte ptr [rdx + r8], al
    movzx eax, byte ptr [rdi + r9]
    mov byte ptr [rdx + r8 + 1], al
    add r8, 2
    add r9, r11
    jmp .Lloop
.Lliteral:
    mov r10, r9
.Llit_loop:
    cmp r10, rsi
    jge .Llit_done
    mov rax, r10
    sub rax, r9
    cmp rax, 128
    jge .Llit_done
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Llit_ok
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llit_ok
    cmp al, byte ptr [rdi + r10 + 2]
    je .Llit_done
.Llit_ok:
    inc r10
    jmp .Llit_loop
.Llit_done:
    mov r11, r10
    sub r11, r9
    lea rax, [r8 + r11 + 1]
    cmp rax, rcx
    jg .Lfail
    mov eax, r11d
    sub eax, 1
    mov byte ptr [rdx + r8], al
    inc r8
    lea r12, [rdi + r9]
    lea r13, [rdx + r8]
    mov r14, r11
.Lcopy:
    test r14, r14
    jle .Lcopy_done
    mov al, byte ptr [r12]
    mov byte ptr [r13], al
    inc r12
    inc r13
    dec r14
    jmp .Lcopy
.Lcopy_done:
    add r8, r11
    mov r9, r10
    jmp .Lloop
.Lfail:
    mov rax, -1
    jmp .Lepilogue
.Ldone:
    mov rax, r8
.Lepilogue:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
