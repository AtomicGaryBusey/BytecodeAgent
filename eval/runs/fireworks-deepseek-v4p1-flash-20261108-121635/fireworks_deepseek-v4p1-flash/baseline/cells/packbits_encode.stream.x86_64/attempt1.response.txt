packbits_encode:
    push rbx
    xor eax, eax
    xor r8d, r8d
    xor r9d, r9d
    test rsi, rsi
    jle .Lreturn
.Lloop:
    cmp r9, rsi
    jge .Lreturn
    movzx eax, byte ptr [rdi + r9]
    lea r10, [r9 + 1]
    mov r11d, 1
.Lcount_run:
    cmp r10, rsi
    jge .Lrun_done
    mov rbx, r10
    sub rbx, r9
    cmp rbx, 128
    jge .Lrun_done
    cmp byte ptr [rdi + r10], al
    jne .Lrun_done
    inc r11
    inc r10
    jmp .Lcount_run
.Lrun_done:
    cmp r11, 3
    jl .Lliteral
    mov eax, 257
    sub eax, r11d
    lea r10, [r8 + 2]
    cmp r10, rcx
    jg .Lfail
    mov byte ptr [rdx + r8], al
    mov al, byte ptr [rdi + r9]
    mov byte ptr [rdx + r8 + 1], al
    add r8, 2
    add r9, r11
    jmp .Lloop
.Lliteral:
    mov r10, r9
.Llit_loop:
    cmp r10, rsi
    jge .Llit_done
    mov rbx, r10
    sub rbx, r9
    cmp rbx, 128
    jge .Llit_done
    lea rbx, [r10 + 2]
    cmp rbx, rsi
    jge .Llit_ok
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llit_ok
    cmp al, byte ptr [rdi + r10 + 2]
    je .Llit_done
.Llit_ok:
    inc r10
    jmp .Llit_loop
.Llit_done:
    mov r11, r10
    sub r11, r9
    lea rax, [r8 + r11 + 1]
    cmp rax, rcx
    jg .Lfail
    mov eax, r11d
    sub eax, 1
    mov byte ptr [rdx + r8], al
    inc r8
    xor eax, eax
.Lcopy:
    cmp rax, r11
    jge .Lcopy_done
    mov bl, byte ptr [rdi + r9 + rax]
