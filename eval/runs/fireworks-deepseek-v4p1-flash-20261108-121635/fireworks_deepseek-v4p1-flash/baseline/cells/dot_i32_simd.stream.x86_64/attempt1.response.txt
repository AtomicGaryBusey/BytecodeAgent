dot_i32_simd:
    xor eax, eax
    test rdx, rdx
    jle .Lret
    pxor xmm8, xmm8
    xor ecx, ecx
.Lloop8:
    lea r8, [rcx + 8]
    cmp r8, rdx
    jg .Lloop4
    pmovsxdq xmm0, qword ptr [rdi + rcx*4]
    pmovsxdq xmm1, qword ptr [rsi + rcx*4]
    pmuldq xmm0, xmm1
    pmovsxdq xmm2, qword ptr [rdi + rcx*4 + 8]
    pmovsxdq xmm3, qword ptr [rsi + rcx*4 + 8]
    pmuldq xmm2, xmm3
    paddq xmm0, xmm2
    pmovsxdq xmm4, qword ptr [rdi + rcx*4 + 16]
    pmovsxdq xmm5, qword ptr [rsi + rcx*4 + 16]
    pmuldq xmm4, xmm5
    pmovsxdq xmm6, qword ptr [rdi + rcx*4 + 24]
    pmovsxdq xmm7, qword ptr [rsi + rcx*4 + 24]
    pmuldq xmm6, xmm7
    paddq xmm4, xmm6
    paddq xmm0, xmm4
    paddq xmm8, xmm0
    add rcx, 8
    jmp .Lloop8
.Lloop4:
    lea r8, [rcx + 4]
    cmp r8, rdx
    jg .Lreduce
    pmovsxdq xmm0, qword ptr [rdi + rcx*4]
    pmovsxdq xmm1, qword ptr [rsi + rcx*4]
    pmuldq xmm0, xmm1
    pmovsxdq xmm2, qword ptr [rdi + rcx*4 + 8]
    pmovsxdq xmm3, qword ptr [rsi + rcx*4 + 8]
    pmuldq xmm2, xmm3
    paddq xmm0, xmm2
    paddq xmm8, xmm0
    add rcx, 4
    jmp .Lloop4
.Lreduce:
    pshufd xmm0, xmm8, 0xEE
    paddq xmm8, xmm0
    movq rax, xmm8
.Lscalar:
    cmp rcx, rdx
    jge .Lret
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    jmp .Lscalar
.Lret:
    ret
