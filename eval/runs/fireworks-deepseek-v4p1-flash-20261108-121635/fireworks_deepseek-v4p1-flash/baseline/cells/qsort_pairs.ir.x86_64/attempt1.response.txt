export function l $qsort_pairs(l %a, l %n) {
@start
  %c =w cslel %n, 1
  jnz %c, @base, @body

@base
  ret 0

@body
  %zero =l copy 0
  %pi =l sub %n, 1
  %poff =l mul %pi, 16
  %paddr =l add %a, %poff
  %paddr8 =l add %paddr, 8
  %pk =l loadl %paddr
  %pv =l loadl %paddr8
  jmp @loop

@loop
  %i =l phi @body %zero, @swap %i_next, @no_swap %i_copy
  %j =l phi @body %zero, @swap %j_next, @no_swap %j_next2
  %cj =w csltl %j, %pi
  jnz %cj, @body_loop, @done_loop

@body_loop
  %joff =l mul %j, 16
  %jaddr =l add %a, %joff
  %jaddr8 =l add %jaddr, 8
  %jk =l loadl %jaddr
  %jv =l loadl %jaddr8
  %c1 =w csltl %jk, %pk
  %c2 =w ceql %jk, %pk
  %c3 =w cslel %jv, %pv
  %c23 =w and %c2, %c3
  %c =w or %c1, %c23
  jnz %c, @swap, @no_swap

@swap
  %ioff =l mul %i, 16
  %iaddr =l add %a, %ioff
  %iaddr8 =l add %iaddr, 8
  %ik =l loadl %iaddr
  %iv =l loadl %iaddr8
  storel %jk, %iaddr
  storel %jv, %iaddr8
  storel %ik, %jaddr
  storel %iv, %jaddr8
  %i_next =l add %i, 1
  %j_next =l add %j, 1
  jmp @loop

@no_swap
  %i_copy =l copy %i
  %j_next2 =l add %j, 1
  jmp @loop

@done_loop
  %ioff2 =l mul %i, 16
  %iaddr2 =l add %a, %ioff2
  %iaddr8_2 =l add %iaddr2, 8
  %ik2 =l loadl %iaddr2
  %iv2 =l loadl %iaddr8_2
  storel %pk, %iaddr2
  storel %pv, %iaddr8_2
  storel %ik2, %paddr
  storel %iv2, %paddr8
  %r1 =l call $qsort_pairs(l %a, l %i)
  %i1 =l add %i, 1
  %roff =l mul %i1, 16
  %raddr =l add %a, %roff
  %rlen =l sub %n, %i1
  %r2 =l call $qsort_pairs(l %raddr, l %rlen)
  ret 0
}
