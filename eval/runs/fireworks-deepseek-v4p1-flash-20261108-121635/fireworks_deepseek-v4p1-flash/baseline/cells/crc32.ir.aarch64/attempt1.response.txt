export function l $crc32(l %p, l %n) {
@start
    %crc0 =w copy -1
    %i0 =l copy 0
    jmp @byte_check
@byte_check
    %i =l phi @start %i0, @byte_next %i_next
    %crc =w phi @start %crc0, @byte_next %crc_next
    %cond =l csltl %i, %n
    jnz %cond, @byte_body, @end
@byte_body
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc_x =w xor %crc, %b
    %j0 =w copy 0
    jmp @bit_check
@bit_check
    %j =w phi @byte_body %j0, @bit_merge %j_next
    %crc_b =w phi @byte_body %crc_x, @bit_merge %crc_next2
    %bitcond =w csltw %j, 8
    jnz %bitcond, @bit_body, @byte_next
@bit_body
    %lsb =w and %crc_b, 1
    %isone =w ceqw %lsb, 1
    jnz %isone, @bit_xor, @bit_shift
@bit_shift
    %crc_s =w shr %crc_b, 1
    jmp @bit_merge
@bit_xor
    %crc_t =w shr %crc_b, 1
    %crc_u =w xor %crc_t, -306674912
    jmp @bit_merge
@bit_merge
    %crc_m =w phi @bit_shift %crc_s, @bit_xor %crc_u
    %j_next =w add %j, 1
    %crc_next2 =w copy %crc_m
    jmp @bit_check
@byte_next
    %crc_next =w copy %crc_b
    %i_next =l add %i, 1
    jmp @byte_check
@end
    %res =w xor %crc, -1
    %resl =l extuw %res
    ret %resl
}
