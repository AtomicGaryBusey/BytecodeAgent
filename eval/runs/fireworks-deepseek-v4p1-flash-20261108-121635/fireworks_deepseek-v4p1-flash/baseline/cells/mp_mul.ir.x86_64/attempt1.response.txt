export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %n =l add %na, %nb
  %iszero =w ceql %n, 0
  jnz %iszero, @ret_zero, @zero_init

@ret_zero
  ret 0

@zero_init
  %k =l copy 0
  jmp @zero_loop

@zero_loop
  %cond =w csltl %k, %n
  jnz %cond, @zero_body, @outer_init

@zero_body
  %koff =l mul %k, 8
  %kaddr =l add %r, %koff
  storel 0, %kaddr
  %knext =l add %k, 1
  %k =l copy %knext
  jmp @zero_loop

@outer_init
  %i =l copy 0
  jmp @outer_loop

@outer_loop
  %ocond =w csltl %i, %na
  jnz %ocond, @outer_body, @sign_init

@outer_body
  %carry =l copy 0
  %j =l copy 0
  jmp @inner_loop

@inner_loop
  %icond =w csltl %j, %nb
  jnz %icond, @inner_body, @inner_done

@inner_body
  %aoff =l mul %i, 8
  %aaddr =l add %a, %aoff
  %x =l loadl %aaddr
  %boff =l mul %j, 8
  %baddr =l add %b, %boff
  %y =l loadl %baddr
  %x0 =l and %x, 4294967295
  %x1 =l shr %x, 32
  %y0 =l and %y, 4294967295
  %y1 =l shr %y, 32
  %p00 =l mul %x0, %y0
  %p01 =l mul %x0, %y1
  %p10 =l mul %x1, %y0
  %p11 =l mul %x1, %y1
  %S =l add %p01, %p10
  %cS =w cultl %S, %p01
  %Shi =l shr %S, 32
  %Sshl =l shl %S, 32
  %plo =l add %p00, %Sshl
  %cL =w cultl %plo, %p00
  %cSl =l extuw %cS
  %cSshl =l shl %cSl, 32
  %cLl =l extuw %cL
  %phi1 =l add %p11, %Shi
  %phi2 =l add %phi1, %cSshl
  %phi =l add %phi2, %cLl
  %idx =l add %i, %j
  %roff =l mul %idx, 8
  %raddr =l add %r, %roff
  %rval =l loadl %raddr
  %t =l add %rval, %plo
  %c1 =w cultl %t, %rval
  %sum =l add %t, %carry
  %c2 =w cultl %sum, %t
  storel %sum, %raddr
  %c1l =l extuw %c1
  %c2l =l extuw %c2
  %carry1 =l add %phi, %c1l
  %carry2 =l add %carry1, %c2l
  %carry =l copy %carry2
  %jnext =l add %j, 1
  %j =l copy %jnext
  jmp @inner_loop

@inner_done
  %inb =l add %i, %nb
  %coff =l mul %inb, 8
  %caddr =l add %r, %coff
  storel %carry, %caddr
  %inext =l add %i, 1
  %i =l copy %inext
  jmp @outer_loop

@sign_init
  %k2 =l copy %n
  jmp @sign_loop

@sign_loop
  %szero =w ceql %k2, 0
  jnz %szero, @sign_done, @sign_check

@sign_check
  %k2m1 =l sub %k2, 1
  %k2off =l mul %k2m1, 8
  %k2addr =l add %r, %k2off
  %last =l loadl %k2addr
  %islastzero =w ceql %last, 0
  jnz %islastzero, @sign_dec, @sign_done

@sign_dec
  %k2next =l sub %k2, 1
  %k2 =l copy %k2next
  jmp @sign_loop

@sign_done
  ret %k2
}
