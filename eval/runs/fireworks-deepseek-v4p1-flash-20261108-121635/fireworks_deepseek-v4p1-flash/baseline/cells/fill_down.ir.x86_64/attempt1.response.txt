export function l $fill_down(l %a, l %n, l %start) {
@start
    %ispos =w csgtl %n, 0
    jnz %ispos, @init, @done
@init
    %i0 =l copy 0
    %p0 =l copy %a
    %v0 =l copy %start
    jmp @loop
@loop
    %i =l phi @init %i0, @body %inext
    %p =l phi @init %p0, @body %pnext
    %v =l phi @init %v0, @body %vnext
    storel %v, %p
    %inext =l add %i, 1
    %pnext =l add %p, 8
    %vnext =l sub %v, 1
    %cmp =w csltl %inext, %n
    jnz %cmp, @body, @done
@body
    jmp @loop
@done
    ret %n
}
