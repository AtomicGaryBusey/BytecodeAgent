export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%cond =w cultl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
