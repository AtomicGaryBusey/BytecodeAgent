export function l $fib(l %n) {
@start
  %zero =w ceql %n, 0
  jnz %zero, @ret0, @init
@ret0
  ret 0
@init
  %a0 =l copy 0
  %b0 =l copy 1
  %i0 =l copy 1
  jmp @loop
@loop
  %a =l phi @init %a0, @body %a1
  %b =l phi @init %b0, @body %b1
  %i =l phi @init %i0, @body %i1
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %t =l add %a, %b
  %a1 =l copy %b
  %b1 =l copy %t
  %i1 =l add %i, 1
  jmp @loop
@done
  ret %b
}
