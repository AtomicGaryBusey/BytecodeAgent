export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %inext
    %count =l phi @start 0, @body %countnext
    %c =w csltl %i, %n
    jnz %c, @body, @end
@body
    %off =l mul %i, 8
    %sp =l add %src, %off
    %x =l loadl %sp
    %r =l call %f(l %x)
    %dp =l add %dst, %off
    storel %r, %dp
    %less =w csltl %r, %x
    %lessl =l extuw %less
    %countnext =l add %count, %lessl
    %inext =l add %i, 1
    jmp @loop
@end
    ret %count
}
