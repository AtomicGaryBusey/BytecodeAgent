export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %si =l copy 0
  %di =l copy 0
  jmp @loop

@loop
  %cond =w csltl %si, %n
  jnz %cond, @body, @done

@body
  %p =l add %src, %si
  %h =w loadsb %p
  %si =l add %si, 1
  %c =w csltw %h, 0
  jnz %c, @neg, @lit

@lit
  %hlen =l extsw %h
  %len =l add %hlen, 1
  %si_end =l add %si, %len
  %ok =w cslel %si_end, %n
  jnz %ok, @lit_ok, @err_src

@lit_ok
  %avail =l sub %cap, %di
  %ncopy =l copy %len
  %c =w csltl %avail, %len
  jnz %c, @lit_limit, @lit_copy_start

@lit_limit
  %ncopy =l copy %avail
  jmp @lit_copy_start

@lit_copy_start
  %i =l copy 0
  jmp @lit_copy_loop

@lit_copy_loop
  %c =w csltl %i, %ncopy
  jnz %c, @lit_copy_body, @lit_copy_end

@lit_copy_body
  %ps =l add %src, %si
  %ps =l add %ps, %i
  %b =w loadub %ps
  %pd =l add %out, %di
  %pd =l add %pd, %i
  storeb %b, %pd
  %i =l add %i, 1
  jmp @lit_copy_loop

@lit_copy_end
  %si =l add %si, %len
  %di =l add %di, %ncopy
  %c =w csltl %ncopy, %len
  jnz %c, @err_cap, @loop

@neg
  %c =w ceqw %h, -128
  jnz %c, @noop, @repeat

@noop
  jmp @loop

@repeat
  %h_l =l extsw %h
  %count =l sub 1, %h_l
  %ok =w csltl %si, %n
  jnz %ok, @repeat_ok, @err_src

@repeat_ok
  %p =l add %src, %si
  %b =w loadub %p
  %si =l add %si, 1
  %avail =l sub %cap, %di
  %ncopy =l copy %count
  %c =w csltl %avail, %count
  jnz %c, @repeat_limit, @repeat_copy_start

@repeat_limit
  %ncopy =l copy %avail
  jmp @repeat_copy_start

@repeat_copy_start
  %i =l copy 0
  jmp @repeat_copy_loop

@repeat_copy_loop
  %c =w csltl %i, %ncopy
  jnz %c, @repeat_copy_body, @repeat_copy_end

@repeat_copy_body
  %pd =l add %out, %di
  %pd =l add %pd, %i
  storeb %b, %pd
  %i =l add %i, 1
  jmp @repeat_copy_loop

@repeat_copy_end
  %di =l add %di, %ncopy
  %c =w csltl %ncopy, %count
  jnz %c, @err_cap, @loop

@err_src
  %ret =l copy -2
  ret %ret

@err_cap
  %ret =l copy -1
  ret %ret

@done
  ret %di
}
