export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop

@loop
    %sp =l phi @start 0, @noop %sp_noop, @lit_done %sp_lit, @rep_done %sp_rep
    %op =l phi @start 0, @noop %op_noop, @lit_done %op_lit, @rep_done %op_rep
    %has_more =w csltl %sp, %n
    jnz %has_more, @header, @done

@header
    %hptr =l add %src, %sp
    %h =w loadsb %hptr
    %sp1 =l add %sp, 1
    %is_noop =w ceqw %h, -128
    jnz %is_noop, @noop, @dispatch

@noop
    %sp_noop =l copy %sp1
    %op_noop =l copy %op
    jmp @loop

@dispatch
    %neg =w csltw %h, 0
    jnz %neg, @repeat, @literal

@literal
    %lcnt =w add %h, 1
    %lcnt64 =l extsw %lcnt
    %lsrc_end =l add %sp1, %lcnt64
    %lsrc_ok =w cslel %lsrc_end, %n
    jnz %lsrc_ok, @lit_out_check, @error_src

@lit_out_check
    %lout_end =l add %op, %lcnt64
    %lout_ok =w cslel %lout_end, %cap
    jnz %lout_ok, @lit_copy, @lit_ovf

@lit_copy
    %li =l phi @lit_out_check 0, @lit_copy_do %li_next
    %lit_more =w csltl %li, %lcnt64
    jnz %lit_more, @lit_copy_do, @lit_done

@lit_copy_do
    %lsidx =l add %sp1, %li
    %lsptr =l add %src, %lsidx
    %lbyte =w loadub %lsptr
    %loidx =l add %op, %li
    %loptr =l add %out, %loidx
    storeb %lbyte, %loptr
    %li_next =l add %li, 1
    jmp @lit_copy

@lit_done
    %sp_lit =l add %sp1, %lcnt64
    %op_lit =l add %op, %lcnt64
    jmp @loop

@lit_ovf
    %lio =l phi @lit_out_check 0, @lit_ovf_do %lio_next
    %lbytes =l sub %cap, %op
    %lit_ovf_more =w csltl %lio, %lbytes
    jnz %lit_ovf_more, @lit_ovf_do, @error_cap

@lit_ovf_do
    %losidx =l add %sp1, %lio
    %losptr =l add %src, %losidx
    %lobyte =w loadub %losptr
    %looidx =l add %op, %lio
    %looptr =l add %out, %looidx
    storeb %lobyte, %looptr
    %lio_next =l add %lio, 1
    jmp @lit_ovf

@repeat
    %rneg =w sub 0, %h
    %rcnt =w add %rneg, 1
    %rcnt64 =l extsw %rcnt
    %rsrc_end =l add %sp1, 1
    %rsrc_ok =w cslel %rsrc_end, %n
    jnz %rsrc_ok, @rep_read, @error_src

@rep_read
    %rbptr =l add %src, %sp1
    %rbyte =w loadub %rbptr
    %sp2 =l add %sp1, 1
    %rout_end =l add %op, %rcnt64
    %rout_ok =w cslel %rout_end, %cap
    jnz %rout_ok, @rep_write, @rep_ovf

@rep_write
    %ri =l phi @rep_read 0, @rep_write_do %ri_next
    %rep_more =w csltl %ri, %rcnt64
    jnz %rep_more, @rep_write_do, @rep_done

@rep_write_do
    %roidx =l add %op, %ri
    %roptr =l add %out, %roidx
    storeb %rbyte, %roptr
    %ri_next =l add %ri, 1
    jmp @rep_write

@rep_done
    %sp_rep =l copy %sp2
    %op_rep =l add %op, %rcnt64
    jmp @loop

@rep_ovf
    %rio =l phi @rep_read 0, @rep_ovf_do %rio_next
    %rbytes =l sub %cap, %op
    %rep_ovf_more =w csltl %rio, %rbytes
    jnz %rep_ovf_more, @rep_ovf_do, @error_cap

@rep_ovf_do
    %rooidx =l add %op, %rio
    %rooptr =l add %out, %rooidx
    storeb %rbyte, %rooptr
    %rio_next =l add %rio, 1
    jmp @rep_ovf

@error_src
    ret -2

@error_cap
    ret -1

@done
    ret %op
}
