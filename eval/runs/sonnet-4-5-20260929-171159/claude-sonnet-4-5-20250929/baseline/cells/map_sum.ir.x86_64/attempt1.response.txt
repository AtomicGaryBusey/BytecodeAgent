export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @loop

@loop
    %sum =l phi @start 0, @body %sum.next
    %i =l phi @start 0, @body %i.next
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %result =l call %f(l %val)
    %sum.next =l add %sum, %result
    %i.next =l add %i, 1
    jmp @loop

@end
    ret %sum
}
