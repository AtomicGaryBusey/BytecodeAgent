reverse:
    test rsi, rsi
    jz .Ldone
    lea rcx, [rdi + rsi*8 - 8]
.Lloop:
    cmp rdi, rcx
    jge .Ldone
    mov rax, qword ptr [rdi]
    mov rdx, qword ptr [rcx]
    mov qword ptr [rcx], rax
    mov qword ptr [rdi], rdx
    add rdi, 8
    sub rcx, 8
    jmp .Lloop
.Ldone:
    xor rax, rax
    ret
