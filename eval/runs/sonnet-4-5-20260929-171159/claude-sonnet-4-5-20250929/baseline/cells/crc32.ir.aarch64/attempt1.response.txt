export function l $crc32(l %p, l %n) {
@start
    jmp @outer_loop

@outer_loop
    %crc.outer =w phi @start 0xFFFFFFFF, @outer_next %crc.next
    %i =l phi @start 0, @outer_next %i.next
    %cond =w cultl %i, %n
    jnz %cond, @outer_body, @finish

@outer_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc.xor =w xor %crc.outer, %byte
    jmp @inner_loop

@inner_loop
    %crc.inner =w phi @outer_body %crc.xor, @inner_next %crc.updated
    %j =w phi @outer_body 0, @inner_next %j.next
    %jcond =w cultw %j, 8
    jnz %jcond, @inner_body, @outer_next

@inner_body
    %bit =w and %crc.inner, 1
    jnz %bit, @do_xor, @no_xor

@do_xor
    %shifted1 =w shr %crc.inner, 1
    %crc.xor2 =w xor %shifted1, 0xEDB88320
    jmp @inner_next

@no_xor
    %shifted2 =w shr %crc.inner, 1
    jmp @inner_next

@inner_next
    %crc.updated =w phi @do_xor %crc.xor2, @no_xor %shifted2
    %j.next =w add %j, 1
    jmp @inner_loop

@outer_next
    %crc.next =w copy %crc.inner
    %i.next =l add %i, 1
    jmp @outer_loop

@finish
    %result =w xor %crc.outer, 0xFFFFFFFF
    %ret =l extuw %result
    ret %ret
}
