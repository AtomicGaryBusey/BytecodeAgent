export function l $bubble_sort(l %a, l %n) {
@start
    %n_le_1 =w csltl %n, 2
    jnz %n_le_1, @return, @outer_start

@outer_start
    %n_minus_1 =l sub %n, 1
    jmp @outer_loop

@outer_loop
    %i =l phi @outer_start 0, @outer_next %i_next
    %outer_cont =w csltl %i, %n_minus_1
    jnz %outer_cont, @inner_start, @return

@inner_start
    %inner_limit =l sub %n_minus_1, %i
    jmp @inner_loop

@inner_loop
    %j =l phi @inner_start 0, @inner_next %j_next
    %inner_cont =w csltl %j, %inner_limit
    jnz %inner_cont, @compare, @outer_next

@compare
    %j_offset =l mul %j, 8
    %j_ptr =l add %a, %j_offset
    %j_val =l loadl %j_ptr
    %j_plus_1 =l add %j, 1
    %j_plus_1_offset =l mul %j_plus_1, 8
    %j_plus_1_ptr =l add %a, %j_plus_1_offset
    %j_plus_1_val =l loadl %j_plus_1_ptr
    %need_swap =w csgtl %j_val, %j_plus_1_val
    jnz %need_swap, @do_swap, @inner_next

@do_swap
    storel %j_plus_1_val, %j_ptr
    storel %j_val, %j_plus_1_ptr
    jmp @inner_next

@inner_next
    %j_next =l add %j, 1
    jmp @inner_loop

@outer_next
    %i_next =l add %i, 1
    jmp @outer_loop

@return
    ret 0
}
