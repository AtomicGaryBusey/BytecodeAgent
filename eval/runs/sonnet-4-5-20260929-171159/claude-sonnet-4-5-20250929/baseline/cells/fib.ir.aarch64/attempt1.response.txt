export function l $fib(l %n) {
@start
    %is_base =w csltl %n, 2
    jnz %is_base, @base, @init
@base
    ret %n
@init
    jmp @loop
@loop
    %a =l phi @init 0, @body %newa
    %b =l phi @init 1, @body %newb
    %i =l phi @init 1, @body %newi
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %temp =l add %a, %b
    %newa =l copy %b
    %newb =l copy %temp
    %newi =l add %i, 1
    jmp @loop
@done
    ret %b
}
