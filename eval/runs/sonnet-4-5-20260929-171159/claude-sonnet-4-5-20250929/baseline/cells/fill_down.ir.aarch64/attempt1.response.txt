export function l $fill_down(l %a, l %n, l %start) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @done, @loop_head

@loop_head
    %i =l phi @start 0, @loop_body %i_next
    %cmp =w csltl %i, %n
    jnz %cmp, @loop_body, @done

@loop_body
    %val =l sub %start, %i
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    storel %val, %ptr
    %i_next =l add %i, 1
    jmp @loop_head

@done
    ret %n
}
