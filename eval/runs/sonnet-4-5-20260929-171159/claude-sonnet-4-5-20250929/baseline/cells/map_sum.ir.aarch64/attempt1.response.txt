export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @check

@check
    %sum =l phi @start 0, @body %sum.1
    %i =l phi @start 0, @body %i.1
    %cond =w csltl %i, %n
    jnz %cond, @body, @ret

@body
    %off =l mul %i, 8
    %p =l add %a, %off
    %elem =l loadl %p
    %res =l call %f(l %elem)
    %sum.1 =l add %sum, %res
    %i.1 =l add %i, 1
    jmp @check

@ret
    ret %sum
}
