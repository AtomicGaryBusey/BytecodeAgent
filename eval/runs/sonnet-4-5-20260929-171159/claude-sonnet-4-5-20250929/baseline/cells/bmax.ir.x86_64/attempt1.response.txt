export function l $bmax(l %a, l %n) {
@start
    %max0 =l loadl %a
    %idx0 =l copy 1
    jmp @check

@check
    %max =l phi @start %max0, @next %maxn
    %idx =l phi @start %idx0, @next %idx1
    %cond =w csltl %idx, %n
    jnz %cond, @body, @end

@body
    %off =l mul %idx, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %cmp =w csgtl %val, %max
    jnz %cmp, @update, @keep

@update
    %maxu =l copy %val
    jmp @next

@keep
    %maxk =l copy %max
    jmp @next

@next
    %maxn =l phi @update %maxu, @keep %maxk
    %idx1 =l add %idx, 1
    jmp @check

@end
    ret %max
}
