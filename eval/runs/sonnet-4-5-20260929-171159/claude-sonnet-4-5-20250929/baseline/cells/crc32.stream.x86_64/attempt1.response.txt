    test rsi, rsi
    jz .Ldone
.Lbyte_loop:
    movzx ecx, byte ptr [rdi]
    xor eax, ecx
    mov ecx, 8
.Lbit_loop:
    shr eax, 1
    jnc .Lno_xor
    xor eax, 0xEDB88320
.Lno_xor:
    dec ecx
    jnz .Lbit_loop
    inc rdi
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
