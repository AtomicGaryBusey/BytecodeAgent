export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %is_zero =w ceql %b, 0
    jnz %is_zero, @div_by_zero, @check_overflow

@div_by_zero
    storel %a, %rem
    ret 0

@check_overflow
    %is_min =w ceql %a, -9223372036854775808
    jnz %is_min, @check_minus_one, @normal_div

@check_minus_one
    %is_neg_one =w ceql %b, -1
    jnz %is_neg_one, @overflow, @normal_div

@overflow
    storel 0, %rem
    ret -9223372036854775808

@normal_div
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
