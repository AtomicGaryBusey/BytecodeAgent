export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @iter %i_next
    %odd =l phi @entry 0, @iter %odd_next
    %sum =l phi @entry 0, @iter %sum_next
    %min =l phi @entry 0, @iter %min_next
    %max =l phi @entry 0, @iter %max_next
    %first =w phi @entry 1, @iter 0
    %cont =w csltl %i, %n
    jnz %cont, @body, @done

@body
    %offset =l mul %i, 8
    %aptr =l add %a, %offset
    %aval =l loadl %aptr
    %y =l call %f(l %aval)
    jnz %first, @init, @check_min

@init
    %min_i =l copy %y
    %max_i =l copy %y
    jmp @add_sum

@check_min
    %lt_min =w csltl %y, %min
    jnz %lt_min, @set_min, @keep_min

@set_min
    %min_s =l copy %y
    jmp @check_max

@keep_min
    %min_k =l copy %min
    jmp @check_max

@check_max
    %min_new =l phi @set_min %min_s, @keep_min %min_k
    %gt_max =w csgtl %y, %max
    jnz %gt_max, @set_max, @keep_max

@set_max
    %max_s =l copy %y
    jmp @merge

@keep_max
    %max_k =l copy %max
    jmp @merge

@merge
    %max_new =l phi @set_max %max_s, @keep_max %max_k
    jmp @add_sum

@add_sum
    %min_r =l phi @init %min_i, @merge %min_new
    %max_r =l phi @init %max_i, @merge %max_new
    %sum_r =l add %sum, %y
    %yand =l and %y, 1
    %yodd =w cnel %yand, 0
    jnz %yodd, @odd_inc, @odd_same

@odd_inc
    %odd_r1 =l add %odd, 1
    jmp @iter

@odd_same
    %odd_r2 =l copy %odd
    jmp @iter

@iter
    %odd_next =l phi @odd_inc %odd_r1, @odd_same %odd_r2
    %min_next =l copy %min_r
    %max_next =l copy %max_r
    %sum_next =l copy %sum_r
    %i_next =l add %i, 1
    jmp @loop

@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
