export function l $max_byte(l %p, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @return_zero, @loop_header
@return_zero
    ret 0
@loop_header
    %i =l phi @start 0, @loop_continue %next_i
    %max =l phi @start 0, @loop_continue %max_updated
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done
@loop_body
    %offset =l add %p, %i
    %byte =l loadub %offset
    %is_greater =w cugtl %byte, %max
    jnz %is_greater, @set_max, @keep_max
@set_max
    jmp @loop_continue
@keep_max
    jmp @loop_continue
@loop_continue
    %max_updated =l phi @set_max %byte, @keep_max %max
    %next_i =l add %i, 1
    jmp @loop_header
@done
    ret %max
}
