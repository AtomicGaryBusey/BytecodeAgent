export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %i0 =l copy 0
    jmp @loop

@loop
    %i =l phi @start %i0, @continue %next_i
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %is_zero =w ceql %val, 0
    jnz %is_zero, @found_zero, @continue

@continue
    %next_i =l add %i, 1
    jmp @loop

@found_zero
    ret %i

@done
    ret %n
}
