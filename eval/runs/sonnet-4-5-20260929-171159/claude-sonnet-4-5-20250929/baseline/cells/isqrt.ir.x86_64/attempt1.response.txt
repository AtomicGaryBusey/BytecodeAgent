export function l $isqrt(l %n) {
@start
    %low0 =l copy 0
    %high0 =l copy 4294967295
    %result0 =l copy 0
    jmp @loop

@loop
    %low =l phi @start %low0, @if_true %low1, @if_false %low2
    %high =l phi @start %high0, @if_true %high1, @if_false %high2
    %result =l phi @start %result0, @if_true %result1, @if_false %result2
    %cmp =w cultl %high, %low
    jnz %cmp, @end, @body

@body
    %sum =l add %low, %high
    %mid =l udiv %sum, 2
    %square =l mul %mid, %mid
    %cmp2 =w culel %square, %n
    jnz %cmp2, @if_true, @if_false

@if_true
    %result1 =l copy %mid
    %low1 =l add %mid, 1
    %high1 =l copy %high
    jmp @loop

@if_false
    %result2 =l copy %result
    %low2 =l copy %low
    %high2 =l sub %mid, 1
    jmp @loop

@end
    ret %result
}
