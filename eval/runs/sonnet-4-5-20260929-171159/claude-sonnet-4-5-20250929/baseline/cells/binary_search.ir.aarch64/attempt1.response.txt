export function l $binary_search(l %a, l %n, l %key) {
@start
    %init_left =l copy 0
    %init_right =l sub %n, 1
    jmp @loop

@loop
    %left =l phi @start %init_left, @go_right %newleft, @go_left %left_copy
    %right =l phi @start %init_right, @go_right %right_copy, @go_left %newright
    %cond =w csgtl %left, %right
    jnz %cond, @notfound, @body

@body
    %diff =l sub %right, %left
    %halfdiff =l div %diff, 2
    %mid =l add %left, %halfdiff
    %offset =l mul %mid, 8
    %addr =l add %a, %offset
    %midval =l loadl %addr
    %eq =w ceql %midval, %key
    jnz %eq, @found, @check_less

@check_less
    %lt =w csltl %midval, %key
    jnz %lt, @go_right, @go_left

@go_right
    %newleft =l add %mid, 1
    %right_copy =l copy %right
    jmp @loop

@go_left
    %left_copy =l copy %left
    %newright =l sub %mid, 1
    jmp @loop

@found
    ret %mid

@notfound
    %neg_one =l copy -1
    ret %neg_one
}
