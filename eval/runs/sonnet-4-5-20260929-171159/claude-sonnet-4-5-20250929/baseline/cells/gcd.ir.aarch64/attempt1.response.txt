export function l $gcd(l %a, l %b) {
@start
    jmp @loop
@loop
    %a.curr =l phi @start %a, @continue %a.next
    %b.curr =l phi @start %b, @continue %b.next
    %cond =w ceql %b.curr, 0
    jnz %cond, @done, @continue
@continue
    %a.next =l copy %b.curr
    %b.next =l rem %a.curr, %b.curr
    jmp @loop
@done
    ret %a.curr
}
