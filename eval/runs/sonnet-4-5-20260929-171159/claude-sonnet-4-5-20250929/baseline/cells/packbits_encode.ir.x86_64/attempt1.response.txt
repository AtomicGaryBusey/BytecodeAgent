export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @after_repeat %i_after_repeat, @after_literal %i_after_literal
    %outpos =l phi @start 0, @after_repeat %outpos_after_repeat, @after_literal %outpos_after_literal
    %cmp_i_n =w csltl %i, %n
    jnz %cmp_i_n, @calc_run, @done

@calc_run
    %src_i =l add %src, %i
    %byte_i =w loadub %src_i
    %run_pos_init =l add %i, 1
    jmp @run_loop

@run_loop
    %run_count =l phi @calc_run 1, @run_increment %run_count_next
    %run_pos =l phi @calc_run %run_pos_init, @run_increment %run_pos_next
    %cmp_run_pos_n =w csltl %run_pos, %n
    jnz %cmp_run_pos_n, @run_check_count, @run_done

@run_check_count
    %cmp_run_count_128 =w csltl %run_count, 128
    jnz %cmp_run_count_128, @run_check_byte, @run_done

@run_check_byte
    %src_run_pos =l add %src, %run_pos
    %byte_run_pos =w loadub %src_run_pos
    %cmp_byte =w ceqw %byte_run_pos, %byte_i
    jnz %cmp_byte, @run_increment, @run_done

@run_increment
    %run_count_next =l add %run_count, 1
    %run_pos_next =l add %run_pos, 1
    jmp @run_loop

@run_done
    %cmp_run_3 =w csltl %run_count, 3
    jnz %cmp_run_3, @literal_start, @repeat_start

@repeat_start
    %room_repeat =l sub %cap, %outpos
    %has_no_room_repeat =w csltl %room_repeat, 2
    jnz %has_no_room_repeat, @fail, @repeat_emit

@repeat_emit
    %run_count_m1 =l sub %run_count, 1
    %header_repeat =l sub 0, %run_count_m1
    %header_repeat_b =w copy %header_repeat
    %out_pos_repeat =l add %out, %outpos
    storeb %header_repeat_b, %out_pos_repeat
    %outpos_1_repeat =l add %outpos, 1
    %out_pos_1_repeat =l add %out, %outpos_1_repeat
    storeb %byte_i, %out_pos_1_repeat
    jmp @after_repeat

@after_repeat
    %outpos_after_repeat =l add %outpos, 2
    %i_after_repeat =l add %i, %run_count
    jmp @loop

@literal_start
    jmp @literal_loop

@literal_loop
    %j =l phi @literal_start %i, @literal_advance %j_next
    %cmp_j_n =w csltl %j, %n
    jnz %cmp_j_n, @literal_check1, @literal_emit_prep

@literal_check1
    %diff =l sub %j, %i
    %cmp_diff_128 =w csltl %diff, 128
    jnz %cmp_diff_128, @literal_calc_run_j, @literal_emit_prep

@literal_calc_run_j
    %src_j =l add %src, %j
    %byte_j =w loadub %src_j
    %run_j_pos_init =l add %j, 1
    jmp @run_j_loop

@run_j_loop
    %run_j_count =l phi @literal_calc_run_j 1, @run_j_increment %run_j_count_next
    %run_j_pos =l phi @literal_calc_run_j %run_j_pos_init, @run_j_increment %run_j_pos_next
    %cmp_run_j_pos_n =w csltl %run_j_pos, %n
    jnz %cmp_run_j_pos_n, @run_j_check_count, @run_j_done

@run_j_check_count
    %cmp_run_j_count_128 =w csltl %run_j_count, 128
    jnz %cmp_run_j_count_128, @run_j_check_byte, @run_j_done

@run_j_check_byte
    %src_run_j_pos =l add %src, %run_j_pos
    %byte_run_j_pos =w loadub %src_run_j_pos
    %cmp_byte_j =w ceqw %byte_run_j_pos, %byte_j
    jnz %cmp_byte_j, @run_j_increment, @run_j_done

@run_j_increment
    %run_j_count_next =l add %run_j_count, 1
    %run_j_pos_next =l add %run_j_pos, 1
    jmp @run_j_loop

@run_j_done
    %cmp_run_j_3 =w csltl %run_j_count, 3
    jnz %cmp_run_j_3, @literal_advance, @literal_emit_prep

@literal_advance
    %j_next =l add %j, 1
    jmp @literal_loop

@literal_emit_prep
    %len =l sub %j, %i
    %needed =l add %len, 1
    %room_literal =l sub %cap, %outpos
    %has_no_room_literal =w csltl %room_literal, %needed
    jnz %has_no_room_literal, @fail, @literal_emit

@literal_emit
    %header_literal =l sub %len, 1
    %header_literal_b =w copy %header_literal
    %out_pos_literal =l add %out, %outpos
    storeb %header_literal_b, %out_pos_literal
    %outpos_after_header =l add %outpos, 1
    jmp @copy_loop

@copy_loop
    %k =l phi @literal_emit %i, @copy_byte_done %k_next
    %outpos_copy =l phi @literal_emit %outpos_after_header, @copy_byte_done %outpos_copy_next
    %cmp_k_j =w csltl %k, %j
    jnz %cmp_k_j, @copy_byte, @after_literal

@copy_byte
    %src_k =l add %src, %k
    %byte_k =w loadub %src_k
    %out_pos_copy =l add %out, %outpos_copy
    storeb %byte_k, %out_pos_copy
    jmp @copy_byte_done

@copy_byte_done
    %outpos_copy_next =l add %outpos_copy, 1
    %k_next =l add %k, 1
    jmp @copy_loop

@after_literal
    %i_after_literal =l copy %j
    %outpos_after_literal =l copy %outpos_copy
    jmp @loop

@done
    ret %outpos

@fail
    ret -1
}
