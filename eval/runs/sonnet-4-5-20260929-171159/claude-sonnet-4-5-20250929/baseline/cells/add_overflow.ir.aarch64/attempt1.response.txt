export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum =l add %a, %b
    storel %sum, %out
    %ab_xor =l xor %a, %b
    %a_sum_xor =l xor %a, %sum
    %same_sign =w csgel %ab_xor, 0
    %diff_sign =w csltl %a_sum_xor, 0
    %ovf =w and %same_sign, %diff_sign
    %result =l extsw %ovf
    ret %result
}
