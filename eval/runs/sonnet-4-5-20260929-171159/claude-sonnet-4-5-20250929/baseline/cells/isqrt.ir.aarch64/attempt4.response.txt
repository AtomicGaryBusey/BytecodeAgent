export function l $isqrt(l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @ret_zero, @check_one

@ret_zero
    ret 0

@check_one
    %is_one =w ceql %n, 1
    jnz %is_one, @ret_one, @init

@ret_one
    ret 1

@init
    %x0 =l copy 4294967296
    jmp @loop

@loop
    %x =l phi @init %x0, @next %xnew
    %q =l udiv %n, %x
    %s =l add %x, %q
    %xnew =l shr %s, 1
    %cont =w cugtl %x, %xnew
    jnz %cont, @next, @done

@next
    jmp @loop

@done
    %qf =l udiv %n, %xnew
    %big =w cugtl %xnew, %qf
    jnz %big, @dec, @ret_result

@dec
    %r =l sub %xnew, 1
    ret %r

@ret_result
    ret %xnew
}
