export function l $qsort_pairs(l %a, l %n) {
@start
    %lt2 =w csltl %n, 2
    jnz %lt2, @ret0, @sort

@ret0
    ret 0

@sort
    %c2 =l copy 2
    %c8 =l copy 8
    %c1 =l copy 1
    %cm1 =l copy -1
    %c0 =l copy 0
    
    %pidx =l div %n, 2
    %proff =l mul %pidx, %c2
    %pboff =l mul %proff, %c8
    %pkp =l add %a, %pboff
    %pk =l loadl %pkp
    %pvbo =l add %pboff, %c8
    %pvp =l add %a, %pvbo
    %pv =l loadl %pvp
    
    %li =l sub %n, %c1
    %lro =l mul %li, %c2
    %lbo =l mul %lro, %c8
    %lkp =l add %a, %lbo
    %lk =l loadl %lkp
    %lvbo =l add %lbo, %c8
    %lvp =l add %a, %lvbo
    %lv =l loadl %lvp
    
    storel %lk, %pkp
    storel %lv, %pvp
    storel %pk, %lkp
    storel %pv, %lvp
    
    %nm2 =l sub %n, %c2
    jmp @lp

@lp
    %i =l phi @sort %cm1, @linc %inext
    %j =l phi @sort %c0, @linc %jnext
    %jdone =w csgtl %j, %nm2
    jnz %jdone, @dn, @lb

@lb
    %jro =l mul %j, %c2
    %jbo =l mul %jro, %c8
    %jkp =l add %a, %jbo
    %jk =l loadl %jkp
    %jvbo =l add %jbo, %c8
    %jvp =l add %a, %jvbo
    %jv =l loadl %jvp
    
    %kl =w csltl %jk, %pk
    jnz %kl, @sw, @ke

@ke
    %ke2 =w ceql %jk, %pk
    jnz %ke2, @vl, @linc

@vl
    %vl2 =w csltl %jv, %pv
    jnz %vl2, @sw, @linc

@sw
    %inew =l add %i, %c1
    %iro =l mul %inew, %c2
    %ibo =l mul %iro, %c8
    %ikp =l add %a, %ibo
    %ik =l loadl %ikp
    %ivbo =l add %ibo, %c8
    %ivp =l add %a, %ivbo
    %iv =l loadl %ivp
    
    storel %jk, %ikp
    storel %jv, %ivp
    storel %ik, %jkp
    storel %iv, %jvp
    
    jmp @linc

@linc
    %inext =l phi @sw %inew, @ke %i, @vl %i
    %jnext =l add %j, %c1
    jmp @lp

@dn
    %ifin =l add %i, %c1
    %ifro =l mul %ifin, %c2
    %ifbo =l mul %ifro, %c8
    %ifkp =l add %a, %ifbo
    %ifk =l loadl %ifkp
    %ifvbo =l add %ifbo, %c8
    %ifvp =l add %a, %ifvbo
    %ifv =l loadl %ifvp
    
    storel %pk, %ifkp
    storel %pv, %ifvp
    storel %ifk, %lkp
    storel %ifv, %lvp
    
    %r1 =l call $qsort_pairs(l %a, l %ifin)
    
    %ri =l add %ifin, %c1
    %rro =l mul %ri, %c2
    %rbo =l mul %rro, %c8
    %rp =l add %a, %rbo
    %rn =l sub %n, %ri
    %r2 =l call $qsort_pairs(l %rp, l %rn)
    
    ret 0
}
