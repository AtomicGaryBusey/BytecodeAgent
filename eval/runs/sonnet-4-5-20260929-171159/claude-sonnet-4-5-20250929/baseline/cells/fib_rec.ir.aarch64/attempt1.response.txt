export function l $fib_rec(l %n) {
@start
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @ret0, @check1
@ret0
    ret 0
@check1
    %cmp1 =w ceql %n, 1
    jnz %cmp1, @ret1, @recurse
@ret1
    ret 1
@recurse
    %n1 =l sub %n, 1
    %fib1 =l call $fib_rec(l %n1)
    %n2 =l sub %n, 2
    %fib2 =l call $fib_rec(l %n2)
    %result =l add %fib1, %fib2
    ret %result
}
