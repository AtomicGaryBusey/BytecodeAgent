export function l $popcount(l %x) {
@start
    jmp @loop
@loop
    %x_curr =l phi @start %x, @body %x_next
    %count_curr =l phi @start 0, @body %count_next
    %is_zero =w ceql %x_curr, 0
    jnz %is_zero, @end, @body
@body
    %x_minus_1 =l sub %x_curr, 1
    %x_next =l and %x_curr, %x_minus_1
    %count_next =l add %count_curr, 1
    jmp @loop
@end
    ret %count_curr
}
