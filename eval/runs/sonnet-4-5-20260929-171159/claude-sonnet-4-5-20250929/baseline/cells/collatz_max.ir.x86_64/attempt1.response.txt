export function l $collatz_max(l %n) {
@start
    jmp @loop_check

@loop_check
    %k =l phi @start 1, @loop_next %k_next
    %max =l phi @start 0, @loop_next %max_next
    %cond =w csgtl %k, %n
    jnz %cond, @end, @collatz_init

@collatz_init
    jmp @collatz_loop

@collatz_loop
    %x =l phi @collatz_init %k, @collatz_even %x_even, @collatz_odd %x_odd
    %steps =l phi @collatz_init 0, @collatz_even %steps_even, @collatz_odd %steps_odd
    %is_one =w ceql %x, 1
    jnz %is_one, @update_check, @collatz_body

@collatz_body
    %rem =l rem %x, 2
    %is_even =w ceql %rem, 0
    jnz %is_even, @collatz_even, @collatz_odd

@collatz_even
    %x_even =l div %x, 2
    %steps_even =l add %steps, 1
    jmp @collatz_loop

@collatz_odd
    %temp =l mul %x, 3
    %x_odd =l add %temp, 1
    %steps_odd =l add %steps, 1
    jmp @collatz_loop

@update_check
    %is_greater =w csgtl %steps, %max
    jnz %is_greater, @update_max, @loop_next

@update_max
    jmp @loop_next

@loop_next
    %max_next =l phi @update_check %max, @update_max %steps
    %k_next =l add %k, 1
    jmp @loop_check

@end
    ret %max
}
