export function l $bubble_sort(l %a, l %n) {
@start
    %n_minus_1 =l sub %n, 1
    jmp @outer_loop

@outer_loop
    %i =l phi @start 0, @outer_next %i_next
    %i_lt =w csltl %i, %n_minus_1
    jnz %i_lt, @inner_init, @done

@inner_init
    jmp @inner_loop

@inner_loop
    %j =l phi @inner_init 0, @inner_next %j_next
    %limit =l sub %n_minus_1, %i
    %j_lt =w csltl %j, %limit
    jnz %j_lt, @inner_body, @outer_next

@inner_body
    %j_offset =l mul %j, 8
    %aj_addr =l add %a, %j_offset
    %aj =l loadl %aj_addr
    %j_plus_1 =l add %j, 1
    %j1_offset =l mul %j_plus_1, 8
    %aj1_addr =l add %a, %j1_offset
    %aj1 =l loadl %aj1_addr
    %cmp =w csgtl %aj, %aj1
    jnz %cmp, @swap, @inner_next

@swap
    storel %aj1, %aj_addr
    storel %aj, %aj1_addr
    jmp @inner_next

@inner_next
    %j_next =l add %j, 1
    jmp @inner_loop

@outer_next
    %i_next =l add %i, 1
    jmp @outer_loop

@done
    ret 0
}
