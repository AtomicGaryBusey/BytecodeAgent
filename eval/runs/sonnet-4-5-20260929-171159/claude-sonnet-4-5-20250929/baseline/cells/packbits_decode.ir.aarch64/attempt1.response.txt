export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop_entry

@loop_entry
    %sp0 =l phi @start 0, @after_noop %sp_noop, @after_literal %sp_lit_end, @after_repeat %sp_rep_end
    %op0 =l phi @start 0, @after_noop %op_noop, @after_literal %op_lit_end, @after_repeat %op_rep_end
    %has =w csltl %sp0, %n
    jnz %has, @read_hdr, @success

@read_hdr
    %hp =l add %src, %sp0
    %h =w loadsb %hp
    %sp1 =l add %sp0, 1
    %m128 =w copy -128
    %nop =w ceqw %h, %m128
    jnz %nop, @do_noop, @check_type

@do_noop
    %sp_noop =l copy %sp1
    %op_noop =l copy %op0
    jmp @after_noop

@after_noop
    jmp @loop_entry

@check_type
    %zero =w copy 0
    %neg =w csltw %h, %zero
    jnz %neg, @do_repeat, @do_literal

@do_literal
    %one =w copy 1
    %lc_w =w add %h, %one
    %lc =l extsw %lc_w
    %sp_need =l add %sp1, %lc
    %ovf =w csltl %n, %sp_need
    jnz %ovf, @err_src, @lit_loop_entry

@lit_loop_entry
    %sp_lit =l phi @do_literal %sp1, @lit_body %sp_lit_next
    %op_lit =l phi @do_literal %op0, @lit_body %op_lit_next
    %li =l phi @do_literal 0, @lit_body %li_next
    %cont_lit =w csltl %li, %lc
    jnz %cont_lit, @lit_check_cap, @lit_done

@lit_check_cap
    %cap_ok =w csltl %op_lit, %cap
    jnz %cap_ok, @lit_body, @err_cap

@lit_body
    %rp =l add %src, %sp_lit
    %b =w loadub %rp
    %wp =l add %out, %op_lit
    storeb %b, %wp
    %sp_lit_next =l add %sp_lit, 1
    %op_lit_next =l add %op_lit, 1
    %li_next =l add %li, 1
    jmp @lit_loop_entry

@lit_done
    %sp_lit_end =l copy %sp_lit
    %op_lit_end =l copy %op_lit
    jmp @after_literal

@after_literal
    jmp @loop_entry

@do_repeat
    %one2 =w copy 1
    %rc_w =w sub %one2, %h
    %rc =l extsw %rc_w
    %sp_ok =w csltl %sp1, %n
    jnz %sp_ok, @rep_read, @err_src

@rep_read
    %rbp =l add %src, %sp1
    %rb =w loadub %rbp
    %sp2 =l add %sp1, 1
    jmp @rep_loop_entry

@rep_loop_entry
    %op_rep =l phi @rep_read %op0, @rep_body %op_rep_next
    %ri =l phi @rep_read 0, @rep_body %ri_next
    %cont_rep =w csltl %ri, %rc
    jnz %cont_rep, @rep_check_cap, @rep_done

@rep_check_cap
    %cap_ok2 =w csltl %op_rep, %cap
    jnz %cap_ok2, @rep_body, @err_cap

@rep_body
    %rwp =l add %out, %op_rep
    storeb %rb, %rwp
    %op_rep_next =l add %op_rep, 1
    %ri_next =l add %ri, 1
    jmp @rep_loop_entry

@rep_done
    %sp_rep_end =l copy %sp2
    %op_rep_end =l copy %op_rep
    jmp @after_repeat

@after_repeat
    jmp @loop_entry

@err_src
    ret -2

@err_cap
    ret -1

@success
    ret %op0
}
