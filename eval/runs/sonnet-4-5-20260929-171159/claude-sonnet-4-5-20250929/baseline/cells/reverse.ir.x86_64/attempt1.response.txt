export function l $reverse(l %a, l %n) {
@start
    %right_init =l sub %n, 1
    jmp @loop

@loop
    %left =l phi @start 0, @swap %left_new
    %right =l phi @start %right_init, @swap %right_new
    %cmp =w csltl %left, %right
    jnz %cmp, @swap, @end

@swap
    %left_offset =l mul %left, 8
    %right_offset =l mul %right, 8
    %left_addr =l add %a, %left_offset
    %right_addr =l add %a, %right_offset
    %left_val =l loadl %left_addr
    %right_val =l loadl %right_addr
    storel %right_val, %left_addr
    storel %left_val, %right_addr
    %left_new =l add %left, 1
    %right_new =l sub %right, 1
    jmp @loop

@end
    ret 0
}
