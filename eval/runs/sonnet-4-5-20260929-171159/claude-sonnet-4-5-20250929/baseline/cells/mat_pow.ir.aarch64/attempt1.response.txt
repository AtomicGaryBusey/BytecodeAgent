export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n_sq =l mul %n, %n
    %two_n_sq =l mul %n_sq, 2
    %i_init_0 =l copy 0
    jmp @init_loop

@init_loop
    %i_init =l phi @start %i_init_0, @init_next %i_init_next
    %cmp_init =w csltl %i_init, %n_sq
    jnz %cmp_init, @init_body, @init_done

@init_body
    %row_init =l div %i_init, %n
    %col_init =l rem %i_init, %n
    %is_diag =w ceql %row_init, %col_init
    jnz %is_diag, @init_one, @init_zero

@init_one
    %val_one =l copy 1
    jmp @init_store

@init_zero
    %val_zero =l copy 0
    jmp @init_store

@init_store
    %val_init =l phi @init_one %val_one, @init_zero %val_zero
    %off_init =l mul %i_init, 8
    %ptr_init =l add %out, %off_init
    storel %val_init, %ptr_init
    jmp @init_next

@init_next
    %i_init_next =l add %i_init, 1
    jmp @init_loop

@init_done
    %i_copy_0 =l copy 0
    jmp @copy_loop

@copy_loop
    %i_copy =l phi @init_done %i_copy_0, @copy_next %i_copy_next
    %cmp_copy =w csltl %i_copy, %n_sq
    jnz %cmp_copy, @copy_body, @copy_done

@copy_body
    %off_copy =l mul %i_copy, 8
    %ptr_a_copy =l add %a, %off_copy
    %val_a =l loadl %ptr_a_copy
    %ptr_tmp_copy =l add %tmp, %off_copy
    storel %val_a, %ptr_tmp_copy
    jmp @copy_next

@copy_next
    %i_copy_next =l add %i_copy, 1
    jmp @copy_loop

@copy_done
    %bit_0 =l copy 0
    jmp @exp_loop

@exp_loop
    %bit =l phi @copy_done %bit_0, @exp_next %bit_next
    %cmp_exp =w csltl %bit, 64
    jnz %cmp_exp, @exp_body, @exp_done

@exp_body
    %one_const =l copy 1
    %mask =l shl %one_const, %bit
    %masked =l and %k, %mask
    %is_set =w cnel %masked, 0
    jnz %is_set, @do_mult_out, @skip_mult_out

@do_mult_out
    %base_b1 =l copy %tmp
    %off_c1 =l mul %n_sq, 8
    %base_c1 =l add %tmp, %off_c1
    %mi1_0 =l copy 0
    jmp @m1_i_loop

@m1_i_loop
    %mi1 =l phi @do_mult_out %mi1_0, @m1_i_next %mi1_next
    %cmp_mi1 =w csltl %mi1, %n
    jnz %cmp_mi1, @m1_i_body, @m1_i_done

@m1_i_body
    %mj1_0 =l copy 0
    jmp @m1_j_loop

@m1_j_loop
    %mj1 =l phi @m1_i_body %mj1_0, @m1_j_next %mj1_next
    %cmp_mj1 =w csltl %mj1, %n
    jnz %cmp_mj1, @m1_j_body, @m1_j_done

@m1_j_body
    %sum1_0 =l copy 0
    %mt1_0 =l copy 0
    jmp @m1_t_loop

@m1_t_loop
    %mt1 =l phi @m1_j_body %mt1_0, @m1_t_next %mt1_next
    %sum1 =l phi @m1_j_body %sum1_0, @m1_t_next %sum1_next
    %cmp_mt1 =w csltl %mt1, %n
    jnz %cmp_mt1, @m1_t_body, @m1_t_done

@m1_t_body
    %a1_row_off =l mul %mi1, %n
    %a1_idx =l add %a1_row_off, %mt1
    %a1_off =l mul %a1_idx, 8
    %a1_ptr =l add %out, %a1_off
    %a1_val =l loadl %a1_ptr
    %b1_row_off =l mul %mt1, %n
    %b1_idx =l add %b1_row_off, %mj1
    %b1_off =l mul %b1_idx, 8
    %b1_ptr =l add %base_b1, %b1_off
    %b1_val =l loadl %b1_ptr
    %prod1 =l mul %a1_val, %b1_val
    %sum1_next =l add %sum1, %prod1
    jmp @m1_t_next

@m1_t_next
    %mt1_next =l add %mt1, 1
    jmp @m1_t_loop

@m1_t_done
    %c1_row_off =l mul %mi1, %n
    %c1_idx =l add %c1_row_off, %mj1
    %c1_off =l mul %c1_idx, 8
    %c1_ptr =l add %base_c1, %c1_off
    storel %sum1, %c1_ptr
    jmp @m1_j_next

@m1_j_next
    %mj1_next =l add %mj1, 1
    jmp @m1_j_loop

@m1_j_done
    jmp @m1_i_next

@m1_i_next
    %mi1_next =l add %mi1, 1
    jmp @m1_i_loop

@m1_i_done
    %cp1_0 =l copy 0
    jmp @cp1_loop

@cp1_loop
    %cp1 =l phi @m1_i_done %cp1_0, @cp1_next %cp1_next_val
    %cmp_cp1 =w csltl %cp1, %n_sq
    jnz %cmp_cp1, @cp1_body, @cp1_done

@cp1_body
    %cp1_src_off =l mul %cp1, 8
    %cp1_src_ptr =l add %base_c1, %cp1_src_off
    %cp1_val =l loadl %cp1_src_ptr
    %cp1_dst_off =l mul %cp1, 8
    %cp1_dst_ptr =l add %out, %cp1_dst_off
    storel %cp1_val, %cp1_dst_ptr
    jmp @cp1_next

@cp1_next
    %cp1_next_val =l add %cp1, 1
    jmp @cp1_loop

@cp1_done
    jmp @mult_out_done

@skip_mult_out
    jmp @mult_out_done

@mult_out_done
    %base_a2 =l copy %tmp
    %base_b2 =l copy %tmp
    %off_c2 =l mul %n_sq, 8
    %base_c2 =l add %tmp, %off_c2
    %mi2_0 =l copy 0
    jmp @m2_i_loop

@m2_i_loop
    %mi2 =l phi @mult_out_done %mi2_0, @m2_i_next %mi2_next
    %cmp_mi2 =w csltl %mi2, %n
    jnz %cmp_mi2, @m2_i_body, @m2_i_done

@m2_i_body
    %mj2_0 =l copy 0
    jmp @m2_j_loop

@m2_j_loop
    %mj2 =l phi @m2_i_body %mj2_0, @m2_j_next %mj2_next
    %cmp_mj2 =w csltl %mj2, %n
    jnz %cmp_mj2, @m2_j_body, @m2_j_done

@m2_j_body
    %sum2_0 =l copy 0
    %mt2_0 =l copy 0
    jmp @m2_t_loop

@m2_t_loop
    %mt2 =l phi @m2_j_body %mt2_0, @m2_t_next %mt2_next
    %sum2 =l phi @m2_j_body %sum2_0, @m2_t_next %sum2_next
    %cmp_mt2 =w csltl %mt2, %n
    jnz %cmp_mt2, @m2_t_body, @m2_t_done

@m2_t_body
    %a2_row_off =l mul %mi2, %n
    %a2_idx =l add %a2_row_off, %mt2
    %a2_off =l mul %a2_idx, 8
    %a2_ptr =l add %base_a2, %a2_off
    %a2_val =l loadl %a2_ptr
    %b2_row_off =l mul %mt2, %n
    %b2_idx =l add %b2_row_off, %mj2
    %b2_off =l mul %b2_idx, 8
    %b2_ptr =l add %base_b2, %b2_off
    %b2_val =l loadl %b2_ptr
    %prod2 =l mul %a2_val, %b2_val
    %sum2_next =l add %sum2, %prod2
    jmp @m2_t_next

@m2_t_next
    %mt2_next =l add %mt2, 1
    jmp @m2_t_loop

@m2_t_done
    %c2_row_off =l mul %mi2, %n
    %c2_idx =l add %c2_row_off, %mj2
    %c2_off =l mul %c2_idx, 8
    %c2_ptr =l add %base_c2, %c2_off
    storel %sum2, %c2_ptr
    jmp @m2_j_next

@m2_j_next
    %mj2_next =l add %mj2, 1
    jmp @m2_j_loop

@m2_j_done
    jmp @m2_i_next

@m2_i_next
    %mi2_next =l add %mi2, 1
    jmp @m2_i_loop

@m2_i_done
    %cp2_0 =l copy 0
    jmp @cp2_loop

@cp2_loop
    %cp2 =l phi @m2_i_done %cp2_0, @cp2_next %cp2_next_val
    %cmp_cp2 =w csltl %cp2, %n_sq
    jnz %cmp_cp2, @cp2_body, @cp2_done

@cp2_body
    %cp2_src_off =l mul %cp2, 8
    %cp2_src_ptr =l add %base_c2, %cp2_src_off
    %cp2_val =l loadl %cp2_src_ptr
    %cp2_dst_off =l mul %cp2, 8
    %cp2_dst_ptr =l add %tmp, %cp2_dst_off
    storel %cp2_val, %cp2_dst_ptr
    jmp @cp2_next

@cp2_next
    %cp2_next_val =l add %cp2, 1
    jmp @cp2_loop

@cp2_done
    jmp @exp_next

@exp_next
    %bit_next =l add %bit, 1
    jmp @exp_loop

@exp_done
    %i_zero_0 =l copy 0
    jmp @zero_loop

@zero_loop
    %i_zero =l phi @exp_done %i_zero_0, @zero_next %i_zero_next
    %cmp_zero =w csltl %i_zero, %two_n_sq
    jnz %cmp_zero, @zero_body, @zero_done

@zero_body
    %off_zero =l mul %i_zero, 8
    %ptr_zero =l add %tmp, %off_zero
    storel 0, %ptr_zero
    jmp @zero_next

@zero_next
    %i_zero_next =l add %i_zero, 1
    jmp @zero_loop

@zero_done
    ret 0
}
