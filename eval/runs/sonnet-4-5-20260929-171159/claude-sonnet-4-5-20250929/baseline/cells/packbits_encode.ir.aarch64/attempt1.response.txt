export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop.0

@loop.0
    %i.loop =l phi @start 0, @repeat.done %i.after.repeat, @literal.done %i.after.literal
    %out_pos.loop =l phi @start 0, @repeat.done %out_pos.after.repeat, @literal.done %out_pos.after.literal
    %cond.loop =w csltl %i.loop, %n
    jnz %cond.loop, @run.start, @done

@run.start
    %src.ptr.run =l add %src, %i.loop
    %first.run =w loadub %src.ptr.run
    %run.idx.start =l add %i.loop, 1
    jmp @run.loop.0

@run.loop.0
    %run.count =l phi @run.start 1, @run.increment %run.count.next
    %run.idx =l phi @run.start %run.idx.start, @run.increment %run.idx.next
    %run.in.range =w csltl %run.idx, %n
    jnz %run.in.range, @run.check.limit, @run.done

@run.check.limit
    %run.at.limit =w csgesl %run.count, 128
    jnz %run.at.limit, @run.done, @run.check.byte

@run.check.byte
    %run.cur.ptr =l add %src, %run.idx
    %run.cur =w loadub %run.cur.ptr
    %run.same =w ceqw %run.cur, %first.run
    jnz %run.same, @run.increment, @run.done

@run.increment
    %run.count.next =l add %run.count, 1
    %run.idx.next =l add %run.idx, 1
    jmp @run.loop.0

@run.done
    %is.repeat =w csgesl %run.count, 3
    jnz %is.repeat, @emit.repeat, @emit.literal.start

@emit.repeat
    %needed.repeat =l add %out_pos.loop, 2
    %fits.repeat =w cslel %needed.repeat, %cap
    jnz %fits.repeat, @repeat.ok, @fail

@repeat.ok
    %header.tmp =l sub 257, %run.count
    %header =w copy %header.tmp
    %out.ptr.header =l add %out, %out_pos.loop
    storeb %header, %out.ptr.header
    %out_pos.after.header =l add %out_pos.loop, 1
    %out.ptr.byte =l add %out, %out_pos.after.header
    storeb %first.run, %out.ptr.byte
    %out_pos.after.byte =l add %out_pos.after.header, 1
    %i.after.run =l add %i.loop, %run.count
    jmp @repeat.done

@repeat.done
    %i.after.repeat =l copy %i.after.run
    %out_pos.after.repeat =l copy %out_pos.after.byte
    jmp @loop.0

@emit.literal.start
    jmp @literal.scan.0

@literal.scan.0
    %j =l phi @emit.literal.start %i.loop, @literal.advance %j.next
    %j.in.range =w csltl %j, %n
    jnz %j.in.range, @literal.check.len, @literal.emit

@literal.check.len
    %diff =l sub %j, %i.loop
    %len.ok =w csltl %diff, 128
    jnz %len.ok, @literal.run.start, @literal.emit

@literal.run.start
    %src.j.ptr =l add %src, %j
    %first.j =w loadub %src.j.ptr
    %run.j.idx.start =l add %j, 1
    jmp @literal.run.loop.0

@literal.run.loop.0
    %run.j.count =l phi @literal.run.start 1, @literal.run.increment %run.j.count.next
    %run.j.idx =l phi @literal.run.start %run.j.idx.start, @literal.run.increment %run.j.idx.next
    %run.j.in.range =w csltl %run.j.idx, %n
    jnz %run.j.in.range, @literal.run.check.limit, @literal.run.done

@literal.run.check.limit
    %run.j.at.limit =w csgesl %run.j.count, 128
    jnz %run.j.at.limit, @literal.run.done, @literal.run.check.byte

@literal.run.check.byte
    %run.j.cur.ptr =l add %src, %run.j.idx
    %run.j.cur =w loadub %run.j.cur.ptr
    %run.j.same =w ceqw %run.j.cur, %first.j
    jnz %run.j.same, @literal.run.increment, @literal.run.done

@literal.run.increment
    %run.j.count.next =l add %run.j.count, 1
    %run.j.idx.next =l add %run.j.idx, 1
    jmp @literal.run.loop.0

@literal.run.done
    %run.j.small =w csltl %run.j.count, 3
    jnz %run.j.small, @literal.advance, @literal.emit

@literal.advance
    %j.next =l add %j, 1
    jmp @literal.scan.0

@literal.emit
    %L =l sub %j, %i.loop
    %L.plus.1 =l add %L, 1
    %total.needed =l add %out_pos.loop, %L.plus.1
    %fits.literal =w cslel %total.needed, %cap
    jnz %fits.literal, @literal.ok, @fail

@literal.ok
    %header.L.tmp =l sub %L, 1
    %header.L =w copy %header.L.tmp
    %out.ptr.L.header =l add %out, %out_pos.loop
    storeb %header.L, %out.ptr.L.header
    %out_pos.after.L.header =l add %out_pos.loop, 1
    jmp @literal.copy.0

@literal.copy.0
    %copy.idx =l phi @literal.ok %i.loop, @literal.copy.do %copy.idx.next
    %out_pos.copy =l phi @literal.ok %out_pos.after.L.header, @literal.copy.do %out_pos.copy.next
    %copy.cond =w csltl %copy.idx, %j
    jnz %copy.cond, @literal.copy.do, @literal.copy.done

@literal.copy.do
    %src.copy.ptr =l add %src, %copy.idx
    %byte.copy =w loadub %src.copy.ptr
    %out.copy.ptr =l add %out, %out_pos.copy
    storeb %byte.copy, %out.copy.ptr
    %copy.idx.next =l add %copy.idx, 1
    %out_pos.copy.next =l add %out_pos.copy, 1
    jmp @literal.copy.0

@literal.copy.done
    jmp @literal.done

@literal.done
    %i.after.literal =l copy %j
    %out_pos.after.literal =l copy %out_pos.copy
    jmp @loop.0

@fail
    ret -1

@done
    ret %out_pos.loop
}
