collatz_max:
    xor eax, eax
    mov rcx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov r8, rcx
    xor r9d, r9d
.Linner:
    cmp r8, 1
    je .Linner_done
    test r8, 1
    jz .Leven
    lea r8, [r8 + r8*2 + 1]
    inc r9
    jmp .Linner
.Leven:
    shr r8, 1
    inc r9
    jmp .Linner
.Linner_done:
    cmp r9, rax
    cmovg rax, r9
    inc rcx
    jmp .Louter
.Ldone:
    ret
