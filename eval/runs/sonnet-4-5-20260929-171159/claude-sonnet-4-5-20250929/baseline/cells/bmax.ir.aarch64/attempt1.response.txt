export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    %one =l copy 1
    jmp @loop

@loop
    %i =l phi @start %one, @next %i2
    %max =l phi @start %first, @next %max2
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %greater =w csgtl %val, %max
    jnz %greater, @update, @noupdate

@update
    jmp @next

@noupdate
    jmp @next

@next
    %max2 =l phi @update %val, @noupdate %max
    %i2 =l add %i, 1
    jmp @loop

@end
    ret %max
}
