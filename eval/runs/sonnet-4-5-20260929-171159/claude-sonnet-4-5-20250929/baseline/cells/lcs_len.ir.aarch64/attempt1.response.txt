export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init_check

@init_check
    %j.init =l phi @start 0, @init_body %j.init.next
    %cond_init =w csgtl %j.init, %n
    jnz %cond_init, @outer_start, @init_body

@init_body
    %idx.init =l shl %j.init, 3
    %addr.init =l add %row, %idx.init
    storel 0, %addr.init
    %j.init.next =l add %j.init, 1
    jmp @init_check

@outer_start
    jmp @outer_check

@outer_check
    %i =l phi @outer_start 0, @inner_done %i.next
    %cond_outer =w csltl %i, %m
    jnz %cond_outer, @outer_body, @done

@outer_body
    %x_ptr =l add %x, %i
    %x_val =w loadub %x_ptr
    jmp @inner_check

@inner_check
    %j =l phi @outer_body 1, @inner_update %j.next
    %diag =l phi @outer_body 0, @inner_update %diag.next
    %cond_inner =w csgtl %j, %n
    jnz %cond_inner, @inner_done, @inner_body

@inner_body
    %j_idx =l shl %j, 3
    %j_ptr =l add %row, %j_idx
    %up =l loadl %j_ptr
    %j_m1 =l sub %j, 1
    %y_ptr =l add %y, %j_m1
    %y_val =w loadub %y_ptr
    %eq =w ceqw %x_val, %y_val
    jnz %eq, @case_match, @case_nomatch

@case_match
    %val_match =l add %diag, 1
    jmp @inner_update

@case_nomatch
    %jm1_idx =l shl %j_m1, 3
    %jm1_ptr =l add %row, %jm1_idx
    %left =l loadl %jm1_ptr
    %cmp =w csgtl %up, %left
    jnz %cmp, @use_up, @use_left

@use_up
    jmp @inner_update

@use_left
    jmp @inner_update

@inner_update
    %val =l phi @case_match %val_match, @use_up %up, @use_left %left
    storel %val, %j_ptr
    %diag.next =l copy %up
    %j.next =l add %j, 1
    jmp @inner_check

@inner_done
    %i.next =l add %i, 1
    jmp @outer_check

@done
    %n_idx =l shl %n, 3
    %n_ptr =l add %row, %n_idx
    %res =l loadl %n_ptr
    ret %res
}
