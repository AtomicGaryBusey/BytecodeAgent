export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @return_zero, @check_direction
@check_direction
    %dst_lt_src =w csltl %dst, %src
    jnz %dst_lt_src, @fwd_init, @bwd_init
@fwd_init
    %i0 =l copy 0
    jmp @fwd_check
@fwd_check
    %i =l phi @fwd_init %i0, @fwd_body %i_next
    %cond =w csltl %i, %n
    jnz %cond, @fwd_body, @return_zero
@fwd_body
    %offset =l mul %i, 8
    %src_addr =l add %src, %offset
    %dst_addr =l add %dst, %offset
    %val =l loadl %src_addr
    storel %val, %dst_addr
    %i_next =l add %i, 1
    jmp @fwd_check
@bwd_init
    %j0 =l sub %n, 1
    jmp @bwd_check
@bwd_check
    %j =l phi @bwd_init %j0, @bwd_body %j_next
    %cond_bwd =w csltl %j, 0
    jnz %cond_bwd, @return_zero, @bwd_body
@bwd_body
    %offset_bwd =l mul %j, 8
    %src_addr_bwd =l add %src, %offset_bwd
    %dst_addr_bwd =l add %dst, %offset_bwd
    %val_bwd =l loadl %src_addr_bwd
    storel %val_bwd, %dst_addr_bwd
    %j_next =l sub %j, 1
    jmp @bwd_check
@return_zero
    ret 0
}
