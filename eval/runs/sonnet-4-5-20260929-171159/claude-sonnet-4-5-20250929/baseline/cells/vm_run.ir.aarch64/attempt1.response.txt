export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    jmp @loop

@loop
    %pc =l phi @start 0, @push %pc_push, @add %pc_add, @sub %pc_sub, @mul %pc_mul, @dup %pc_dup, @swap %pc_swap, @drop %pc_drop, @over %pc_over, @lt %pc_lt, @jz_merge %pc_jz, @jmp %pc_jmp, @default %pc_default
    %sp =l phi @start 0, @push %sp_push, @add %sp_add, @sub %sp_sub, @mul %sp_mul, @dup %sp_dup, @swap %sp_swap, @drop %sp_drop, @over %sp_over, @lt %sp_lt, @jz_merge %sp_jz, @jmp %sp_jmp, @default %sp_default
    %inrange =w csltl %pc, %len
    jnz %inrange, @fetch, @done

@fetch
    %pc_off =l shl %pc, 3
    %code_ptr =l add %code, %pc_off
    %op =l loadl %code_ptr
    %is0 =w ceql %op, 0
    jnz %is0, @halt, @t1
@t1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @t2
@t2
    %is2 =w ceql %op, 2
    jnz %is2, @add, @t3
@t3
    %is3 =w ceql %op, 3
    jnz %is3, @sub, @t4
@t4
    %is4 =w ceql %op, 4
    jnz %is4, @mul, @t5
@t5
    %is5 =w ceql %op, 5
    jnz %is5, @dup, @t6
@t6
    %is6 =w ceql %op, 6
    jnz %is6, @swap, @t7
@t7
    %is7 =w ceql %op, 7
    jnz %is7, @drop, @t8
@t8
    %is8 =w ceql %op, 8
    jnz %is8, @over, @t9
@t9
    %is9 =w ceql %op, 9
    jnz %is9, @lt, @t10
@t10
    %is10 =w ceql %op, 10
    jnz %is10, @jz, @t11
@t11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp, @default

@halt
    jmp @done

@push
    %pc1 =l add %pc, 1
    %pc1_off =l shl %pc1, 3
    %op_ptr =l add %code, %pc1_off
    %val =l loadl %op_ptr
    %sp_off_push =l shl %sp, 3
    %stk_ptr_push =l add %stack, %sp_off_push
    storel %val, %stk_ptr_push
    %pc_push =l add %pc, 2
    %sp_push =l add %sp, 1
    jmp @loop

@add
    %sp_m1_add =l sub %sp, 1
    %off_b_add =l shl %sp_m1_add, 3
    %ptr_b_add =l add %stack, %off_b_add
    %b_add =l loadl %ptr_b_add
    %sp_m2_add =l sub %sp_m1_add, 1
    %off_a_add =l shl %sp_m2_add, 3
    %ptr_a_add =l add %stack, %off_a_add
    %a_add =l loadl %ptr_a_add
    %res_add =l add %a_add, %b_add
    storel %res_add, %ptr_a_add
    %pc_add =l add %pc, 1
    %sp_add =l add %sp_m2_add, 1
    jmp @loop

@sub
    %sp_m1_sub =l sub %sp, 1
    %off_b_sub =l shl %sp_m1_sub, 3
    %ptr_b_sub =l add %stack, %off_b_sub
    %b_sub =l loadl %ptr_b_sub
    %sp_m2_sub =l sub %sp_m1_sub, 1
    %off_a_sub =l shl %sp_m2_sub, 3
    %ptr_a_sub =l add %stack, %off_a_sub
    %a_sub =l loadl %ptr_a_sub
    %res_sub =l sub %a_sub, %b_sub
    storel %res_sub, %ptr_a_sub
    %pc_sub =l add %pc, 1
    %sp_sub =l add %sp_m2_sub, 1
    jmp @loop

@mul
    %sp_m1_mul =l sub %sp, 1
    %off_b_mul =l shl %sp_m1_mul, 3
    %ptr_b_mul =l add %stack, %off_b_mul
    %b_mul =l loadl %ptr_b_mul
    %sp_m2_mul =l sub %sp_m1_mul, 1
    %off_a_mul =l shl %sp_m2_mul, 3
    %ptr_a_mul =l add %stack, %off_a_mul
    %a_mul =l loadl %ptr_a_mul
    %res_mul =l mul %a_mul, %b_mul
    storel %res_mul, %ptr_a_mul
    %pc_mul =l add %pc, 1
    %sp_mul =l add %sp_m2_mul, 1
    jmp @loop

@dup
    %sp_m1_dup =l sub %sp, 1
    %off_dup =l shl %sp_m1_dup, 3
    %ptr_dup =l add %stack, %off_dup
    %val_dup =l loadl %ptr_dup
    %sp_off_dup =l shl %sp, 3
    %ptr_new_dup =l add %stack, %sp_off_dup
    storel %val_dup, %ptr_new_dup
    %pc_dup =l add %pc, 1
    %sp_dup =l add %sp, 1
    jmp @loop

@swap
    %sp_m1_swap =l sub %sp, 1
    %off_top_swap =l shl %sp_m1_swap, 3
    %ptr_top_swap =l add %stack, %off_top_swap
    %top_swap =l loadl %ptr_top_swap
    %sp_m2_swap =l sub %sp_m1_swap, 1
    %off_sec_swap =l shl %sp_m2_swap, 3
    %ptr_sec_swap =l add %stack, %off_sec_swap
    %sec_swap =l loadl %ptr_sec_swap
    storel %sec_swap, %ptr_top_swap
    storel %top_swap, %ptr_sec_swap
    %pc_swap =l add %pc, 1
    %sp_swap =l copy %sp
    jmp @loop

@drop
    %pc_drop =l add %pc, 1
    %sp_drop =l sub %sp, 1
    jmp @loop

@over
    %sp_m2_over =l sub %sp, 2
    %off_over =l shl %sp_m2_over, 3
    %ptr_over =l add %stack, %off_over
    %val_over =l loadl %ptr_over
    %sp_off_over =l shl %sp, 3
    %ptr_new_over =l add %stack, %sp_off_over
    storel %val_over, %ptr_new_over
    %pc_over =l add %pc, 1
    %sp_over =l add %sp, 1
    jmp @loop

@lt
    %sp_m1_lt =l sub %sp, 1
    %off_b_lt =l shl %sp_m1_lt, 3
    %ptr_b_lt =l add %stack, %off_b_lt
    %b_lt =l loadl %ptr_b_lt
    %sp_m2_lt =l sub %sp_m1_lt, 1
    %off_a_lt =l shl %sp_m2_lt, 3
    %ptr_a_lt =l add %stack, %off_a_lt
    %a_lt =l loadl %ptr_a_lt
    %cmp_lt =w csltl %a_lt, %b_lt
    %res_lt =l extsw %cmp_lt
    storel %res_lt, %ptr_a_lt
    %pc_lt =l add %pc, 1
    %sp_lt =l add %sp_m2_lt, 1
    jmp @loop

@jz
    %sp_m1_jz =l sub %sp, 1
    %off_jz =l shl %sp_m1_jz, 3
    %ptr_jz =l add %stack, %off_jz
    %cond_jz =l loadl %ptr_jz
    %is_zero =w ceql %cond_jz, 0
    jnz %is_zero, @jz_take, @jz_fall
@jz_take
    %pc1_jz =l add %pc, 1
    %pc1_off_jz =l shl %pc1_jz, 3
    %tgt_ptr_jz =l add %code, %pc1_off_jz
    %tgt_jz =l loadl %tgt_ptr_jz
    %pc_jz_t =l copy %tgt_jz
    %sp_jz_t =l copy %sp_m1_jz
    jmp @jz_merge
@jz_fall
    %pc_jz_f =l add %pc, 2
    %sp_jz_f =l copy %sp_m1_jz
    jmp @jz_merge
@jz_merge
    %pc_jz =l phi @jz_take %pc_jz_t, @jz_fall %pc_jz_f
    %sp_jz =l phi @jz_take %sp_jz_t, @jz_fall %sp_jz_f
    jmp @loop

@jmp
    %pc1_jmp =l add %pc, 1
    %pc1_off_jmp =l shl %pc1_jmp, 3
    %tgt_ptr_jmp =l add %code, %pc1_off_jmp
    %tgt_jmp =l loadl %tgt_ptr_jmp
    %pc_jmp =l copy %tgt_jmp
    %sp_jmp =l copy %sp
    jmp @loop

@default
    %pc_default =l add %pc, 1
    %sp_default =l copy %sp
    jmp @loop

@done
    %sp_done =l phi @loop %sp, @halt %sp
    %is_empty =w ceql %sp_done, 0
    jnz %is_empty, @empty, @nonempty

@empty
    %ret_empty =l copy 0
    jmp @clear_start

@nonempty
    %top_idx =l sub %sp_done, 1
    %top_off =l shl %top_idx, 3
    %top_ptr =l add %stack, %top_off
    %ret_nonempty =l loadl %top_ptr
    jmp @clear_start

@clear_start
    %retval =l phi @empty %ret_empty, @nonempty %ret_nonempty
    jmp @clear_loop

@clear_loop
    %clear_i =l phi @clear_start %sp_done, @clear_body %clear_i_next
    %is_done =w csltl %clear_i, %cap
    jnz %is_done, @clear_body, @return

@clear_body
    %clear_off =l shl %clear_i, 3
    %clear_ptr =l add %stack, %clear_off
    storel 0, %clear_ptr
    %clear_i_next =l add %clear_i, 1
    jmp @clear_loop

@return
    ret %retval
}
