export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @body %i.next
    %count =l phi @start 0, @body %count.next
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %is_greater =w cugtl %val, %t
    %inc_ext =l extuw %is_greater
    %count.next =l add %count, %inc_ext
    %i.next =l add %i, 1
    jmp @loop

@end
    ret %count
}
