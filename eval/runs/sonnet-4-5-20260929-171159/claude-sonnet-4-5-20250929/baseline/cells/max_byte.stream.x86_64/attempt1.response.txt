max_byte:
    xor eax, eax
    test rsi, rsi
    jz .Lreturn
    cmp rsi, 16
    jb .Lscalar_loop
    pxor xmm0, xmm0
.Lsimd_loop:
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    cmp rsi, 16
    jae .Lsimd_loop
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
    test rsi, rsi
    jz .Lreturn
.Lscalar_loop:
    movzx edx, byte ptr [rdi]
    cmp eax, edx
    cmovb eax, edx
    inc rdi
    dec rsi
    jnz .Lscalar_loop
.Lreturn:
    ret
