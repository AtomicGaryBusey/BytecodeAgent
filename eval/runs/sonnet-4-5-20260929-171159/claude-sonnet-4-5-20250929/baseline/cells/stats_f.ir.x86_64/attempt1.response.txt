export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %y0_and =l and %y0, 1
    %y0_odd =w ceql %y0_and, 1
    %odd0 =l extsw %y0_odd
    %one =l copy 1
    jmp @loop

@loop
    %i =l phi @start %one, @endif_max %i_next
    %min =l phi @start %y0, @endif_max %min_next
    %max =l phi @start %y0, @endif_max %max_next
    %sum =l phi @start %y0, @endif_max %sum_next
    %odd =l phi @start %odd0, @endif_max %odd_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %ai =l loadl %ptr
    %yi =l call %f(l %ai)
    %is_less =w csltl %yi, %min
    jnz %is_less, @then_min, @else_min

@then_min
    jmp @endif_min

@else_min
    jmp @endif_min

@endif_min
    %min_next =l phi @then_min %yi, @else_min %min
    %is_greater =w csgtl %yi, %max
    jnz %is_greater, @then_max, @else_max

@then_max
    jmp @endif_max

@else_max
    jmp @endif_max

@endif_max
    %max_next =l phi @then_max %yi, @else_max %max
    %sum_next =l add %sum, %yi
    %yi_and =l and %yi, 1
    %yi_odd =w ceql %yi_and, 1
    %yi_odd_l =l extsw %yi_odd
    %odd_next =l add %odd, %yi_odd_l
    %i_next =l add %i, 1
    jmp @loop

@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
