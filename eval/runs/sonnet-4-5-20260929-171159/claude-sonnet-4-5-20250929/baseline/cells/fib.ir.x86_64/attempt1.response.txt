export function l $fib(l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @ret_zero, @check_one
@ret_zero
    ret 0
@check_one
    %is_one =w ceql %n, 1
    jnz %is_one, @ret_one, @init
@ret_one
    ret 1
@init
    jmp @loop
@loop
    %a =l phi @init 0, @body %newa
    %b =l phi @init 1, @body %newb
    %i =l phi @init 2, @body %newi
    %cond =w csgtl %i, %n
    jnz %cond, @done, @body
@body
    %sum =l add %a, %b
    %newa =l copy %b
    %newb =l copy %sum
    %newi =l add %i, 1
    jmp @loop
@done
    ret %b
}
