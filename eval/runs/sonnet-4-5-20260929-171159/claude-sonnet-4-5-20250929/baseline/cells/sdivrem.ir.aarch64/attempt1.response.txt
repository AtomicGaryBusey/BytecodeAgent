export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %is_zero =w ceql %b, 0
    jnz %is_zero, @div_by_zero, @check_overflow

@div_by_zero
    storel %a, %rem
    ret 0

@check_overflow
    %int64_min =l copy -9223372036854775808
    %is_min =w ceql %a, %int64_min
    jnz %is_min, @check_neg_one, @normal_div

@check_neg_one
    %is_neg_one =w ceql %b, -1
    jnz %is_neg_one, @overflow_case, @normal_div

@overflow_case
    storel 0, %rem
    %min =l copy -9223372036854775808
    ret %min

@normal_div
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
