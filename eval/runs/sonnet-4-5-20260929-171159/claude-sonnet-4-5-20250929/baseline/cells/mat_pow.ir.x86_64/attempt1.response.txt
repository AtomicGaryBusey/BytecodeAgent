export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n2 =l mul %n, %n
    %i0 =l copy 0
    jmp @init_loop
@init_loop
    %i1 =l phi @start %i0, @init_body %i2
    %icond =w csltl %i1, %n2
    jnz %icond, @init_body, @init_done
@init_body
    %i1s =l mul %i1, 8
    %iaddr =l add %out, %i1s
    storel 0, %iaddr
    %i2 =l add %i1, 1
    jmp @init_loop
@init_done
    %d0 =l copy 0
    jmp @diag_loop
@diag_loop
    %d1 =l phi @init_done %d0, @diag_body %d2
    %dcond =w csltl %d1, %n
    jnz %dcond, @diag_body, @diag_done
@diag_body
    %drow =l mul %d1, %n
    %didx =l add %drow, %d1
    %didxs =l mul %didx, 8
    %daddr =l add %out, %didxs
    storel 1, %daddr
    %d2 =l add %d1, 1
    jmp @diag_loop
@diag_done
    %c0 =l copy 0
    jmp @copy_loop
@copy_loop
    %c1 =l phi @diag_done %c0, @copy_body %c2
    %ccond =w csltl %c1, %n2
    jnz %ccond, @copy_body, @copy_done
@copy_body
    %c1s =l mul %c1, 8
    %asrc =l add %a, %c1s
    %aval =l loadl %asrc
    %tdst =l add %tmp, %c1s
    storel %aval, %tdst
    %c2 =l add %c1, 1
    jmp @copy_loop
@copy_done
    %b0 =l copy 0
    jmp @bit_loop
@bit_loop
    %b1 =l phi @copy_done %b0, @bit_next %b2
    %bcond =w csltl %b1, 64
    jnz %bcond, @bit_body, @bit_done
@bit_body
    %kshr =l shr %k, %b1
    %kand =l and %kshr, 1
    %kbitw =w copy %kand
    jnz %kbitw, @do_mult, @skip_mult
@do_mult
    %mi0 =l copy 0
    jmp @mult_i
@mult_i
    %mi1 =l phi @do_mult %mi0, @mult_i_next %mi2
    %micond =w csltl %mi1, %n
    jnz %micond, @mult_i_body, @mult_i_done
@mult_i_body
    %mj0 =l copy 0
    jmp @mult_j
@mult_j
    %mj1 =l phi @mult_i_body %mj0, @mult_j_next %mj2
    %mjcond =w csltl %mj1, %n
    jnz %mjcond, @mult_j_body, @mult_j_done
@mult_j_body
    %mt0 =l copy 0
    %msum0 =l copy 0
    jmp @mult_t
@mult_t
    %mt1 =l phi @mult_j_body %mt0, @mult_t_body %mt2
    %msum1 =l phi @mult_j_body %msum0, @mult_t_body %msum2
    %mtcond =w csltl %mt1, %n
    jnz %mtcond, @mult_t_body, @mult_t_done
@mult_t_body
    %maidx0 =l mul %mi1, %n
    %maidx1 =l add %maidx0, %mt1
    %maidxs =l mul %maidx1, 8
    %maaddr =l add %out, %maidxs
    %maval =l loadl %maaddr
    %mbidx0 =l mul %mt1, %n
    %mbidx1 =l add %mbidx0, %mj1
    %mbidxs =l mul %mbidx1, 8
    %mbaddr =l add %tmp, %mbidxs
    %mbval =l loadl %mbaddr
    %mprod =l mul %maval, %mbval
    %msum2 =l add %msum1, %mprod
    %mt2 =l add %mt1, 1
    jmp @mult_t
@mult_t_done
    %mcidx0 =l mul %mi1, %n
    %mcidx1 =l add %mcidx0, %mj1
    %mcidx2 =l add %mcidx1, %n2
    %mcidxs =l mul %mcidx2, 8
    %mcaddr =l add %tmp, %mcidxs
    storel %msum1, %mcaddr
    %mj2 =l add %mj1, 1
    jmp @mult_j_next
@mult_j_next
    jmp @mult_j
@mult_j_done
    %mi2 =l add %mi1, 1
    jmp @mult_i_next
@mult_i_next
    jmp @mult_i
@mult_i_done
    %cp0 =l copy 0
    jmp @copy_res
@copy_res
    %cp1 =l phi @mult_i_done %cp0, @copy_res_body %cp2
    %cpcond =w csltl %cp1, %n2
    jnz %cpcond, @copy_res_body, @copy_res_done
@copy_res_body
    %cpsrc_idx =l add %cp1, %n2
    %cpsrc_idxs =l mul %cpsrc_idx, 8
    %cpsrc_addr =l add %tmp, %cpsrc_idxs
    %cpval =l loadl %cpsrc_addr
    %cpdst_idxs =l mul %cp1, 8
    %cpdst_addr =l add %out, %cpdst_idxs
    storel %cpval, %cpdst_addr
    %cp2 =l add %cp1, 1
    jmp @copy_res
@copy_res_done
    jmp @skip_mult
@skip_mult
    %bnext0 =l add %b1, 1
    %sqcond =w csltl %bnext0, 64
    jnz %sqcond, @do_square, @skip_square
@do_square
    %si0 =l copy 0
    jmp @sq_i
@sq_i
    %si1 =l phi @do_square %si0, @sq_i_next %si2
    %sicond =w csltl %si1, %n
    jnz %sicond, @sq_i_body, @sq_i_done
@sq_i_body
    %sj0 =l copy 0
    jmp @sq_j
@sq_j
    %sj1 =l phi @sq_i_body %sj0, @sq_j_next %sj2
    %sjcond =w csltl %sj1, %n
    jnz %sjcond, @sq_j_body, @sq_j_done
@sq_j_body
    %st0 =l copy 0
    %ssum0 =l copy 0
    jmp @sq_t
@sq_t
    %st1 =l phi @sq_j_body %st0, @sq_t_body %st2
    %ssum1 =l phi @sq_j_body %ssum0, @sq_t_body %ssum2
    %stcond =w csltl %st1, %n
    jnz %stcond, @sq_t_body, @sq_t_done
@sq_t_body
    %saidx0 =l mul %si1, %n
    %saidx1 =l add %saidx0, %st1
    %saidxs =l mul %saidx1, 8
    %saaddr =l add %tmp, %saidxs
    %saval =l loadl %saaddr
    %sbidx0 =l mul %st1, %n
    %sbidx1 =l add %sbidx0, %sj1
    %sbidxs =l mul %sbidx1, 8
    %sbaddr =l add %tmp, %sbidxs
    %sbval =l loadl %sbaddr
    %sprod =l mul %saval, %sbval
    %ssum2 =l add %ssum1, %sprod
    %st2 =l add %st1, 1
    jmp @sq_t
@sq_t_done
    %scidx0 =l mul %si1, %n
    %scidx1 =l add %scidx0, %sj1
    %scidx2 =l add %scidx1, %n2
    %scidxs =l mul %scidx2, 8
    %scaddr =l add %tmp, %scidxs
    storel %ssum1, %scaddr
    %sj2 =l add %sj1, 1
    jmp @sq_j_next
@sq_j_next
    jmp @sq_j
@sq_j_done
    %si2 =l add %si1, 1
    jmp @sq_i_next
@sq_i_next
    jmp @sq_i
@sq_i_done
    %sqcp0 =l copy 0
    jmp @sq_copy
@sq_copy
    %sqcp1 =l phi @sq_i_done %sqcp0, @sq_copy_body %sqcp2
    %sqcpcond =w csltl %sqcp1, %n2
    jnz %sqcpcond, @sq_copy_body, @sq_copy_done
@sq_copy_body
    %sqcpsrc_idx =l add %sqcp1, %n2
    %sqcpsrc_idxs =l mul %sqcpsrc_idx, 8
    %sqcpsrc_addr =l add %tmp, %sqcpsrc_idxs
    %sqcpval =l loadl %sqcpsrc_addr
    %sqcpdst_idxs =l mul %sqcp1, 8
    %sqcpdst_addr =l add %tmp, %sqcpdst_idxs
    storel %sqcpval, %sqcpdst_addr
    %sqcp2 =l add %sqcp1, 1
    jmp @sq_copy
@sq_copy_done
    jmp @skip_square
@skip_square
    %b2 =l copy %bnext0
    jmp @bit_next
@bit_next
    jmp @bit_loop
@bit_done
    %z0 =l copy 0
    %tmpsz =l mul %n2, 2
    jmp @zero_loop
@zero_loop
    %z1 =l phi @bit_done %z0, @zero_body %z2
    %zcond =w csltl %z1, %tmpsz
    jnz %zcond, @zero_body, @zero_done
@zero_body
    %z1s =l mul %z1, 8
    %zaddr =l add %tmp, %z1s
    storel 0, %zaddr
    %z2 =l add %z1, 1
    jmp @zero_loop
@zero_done
    ret 0
}
