export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init_start

@init_start
    %j0 =l copy 0
    jmp @init_check

@init_check
    %j.init =l phi @init_start %j0, @init_next %j.init.inc
    %n.p1 =l add %n, 1
    %init.cond =w csltl %j.init, %n.p1
    jnz %init.cond, @init_body, @outer_start

@init_body
    %j.init.off =l mul %j.init, 8
    %j.init.ptr =l add %row, %j.init.off
    storel 0, %j.init.ptr
    jmp @init_next

@init_next
    %j.init.inc =l add %j.init, 1
    jmp @init_check

@outer_start
    %i0 =l copy 0
    jmp @outer_check

@outer_check
    %i =l phi @outer_start %i0, @inner_done %i.inc
    %outer.cond =w csltl %i, %m
    jnz %outer.cond, @outer_body, @finish

@outer_body
    %diag0 =l copy 0
    %j0.inner =l copy 1
    jmp @inner_check

@inner_check
    %j =l phi @outer_body %j0.inner, @inner_next %j.inc
    %diag =l phi @outer_body %diag0, @inner_next %diag.new
    %n.p1.b =l add %n, 1
    %inner.cond =w csltl %j, %n.p1.b
    jnz %inner.cond, @inner_body, @inner_done

@inner_body
    %j.off =l mul %j, 8
    %j.ptr =l add %row, %j.off
    %up =l loadl %j.ptr
    %xi.ptr =l add %x, %i
    %xi =w loadub %xi.ptr
    %j.m1 =l sub %j, 1
    %yj.ptr =l add %y, %j.m1
    %yj =w loadub %yj.ptr
    %match =w ceqw %xi, %yj
    jnz %match, @case_match, @case_nomatch

@case_match
    %val.match =l add %diag, 1
    jmp @update

@case_nomatch
    %jm1.off =l mul %j.m1, 8
    %jm1.ptr =l add %row, %jm1.off
    %left =l loadl %jm1.ptr
    %cmp =w csgtl %up, %left
    jnz %cmp, @take_up, @take_left

@take_up
    %val.up =l copy %up
    jmp @update

@take_left
    %val.left =l copy %left
    jmp @update

@update
    %newval =l phi @case_match %val.match, @take_up %val.up, @take_left %val.left
    storel %newval, %j.ptr
    jmp @inner_next

@inner_next
    %diag.new =l copy %up
    %j.inc =l add %j, 1
    jmp @inner_check

@inner_done
    %i.inc =l add %i, 1
    jmp @outer_check

@finish
    %n.off =l mul %n, 8
    %n.ptr =l add %row, %n.off
    %result =l loadl %n.ptr
    ret %result
}
