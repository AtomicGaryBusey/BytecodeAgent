export function l $shift_sat(l %x, l %n) {
@start
    %cmp =w csltl %n, 64
    jnz %cmp, @n_lt_64, @n_ge_64

@n_lt_64
    %left =l shl %x, %n
    %right =l sar %x, %n
    %result =l add %left, %right
    ret %result

@n_ge_64
    %is_neg =w csltl %x, 0
    jnz %is_neg, @x_neg, @x_nonneg

@x_neg
    %neg_one =l copy -1
    ret %neg_one

@x_nonneg
    %zero =l copy 0
    ret %zero
}
