export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop

@loop
    %count =l phi @start 0, @loop_body %count_next
    %i =l phi @start 0, @loop_body %i_next
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done

@loop_body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %elem =l loadl %addr
    %is_greater =w cugtl %elem, %t
    %is_greater_l =l extuw %is_greater
    %count_next =l add %count, %is_greater_l
    %i_next =l add %i, 1
    jmp @loop

@done
    ret %count
}
