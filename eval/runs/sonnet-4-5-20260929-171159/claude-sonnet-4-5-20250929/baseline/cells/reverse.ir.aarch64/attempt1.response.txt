export function l $reverse(l %a, l %n) {
@start
    %left =l copy 0
    %nm1 =l sub %n, 1
    %right =l copy %nm1
    jmp @check

@check
    %cmp =w csltl %left, %right
    jnz %cmp, @body, @end

@body
    %loff =l mul %left, 8
    %roff =l mul %right, 8
    %lptr =l add %a, %loff
    %rptr =l add %a, %roff
    %lval =l loadl %lptr
    %rval =l loadl %rptr
    storel %rval, %lptr
    storel %lval, %rptr
    %left =l add %left, 1
    %right =l sub %right, 1
    jmp @check

@end
    ret 0
}
