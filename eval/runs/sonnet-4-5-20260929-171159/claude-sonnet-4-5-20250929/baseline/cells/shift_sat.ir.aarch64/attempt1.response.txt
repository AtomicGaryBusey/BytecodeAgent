export function l $shift_sat(l %x, l %n) {
@start
    %cmp64 =w csgel %n, 64
    jnz %cmp64, @n_ge_64, @n_lt_64
@n_lt_64
    %left =l shl %x, %n
    %right =l sar %x, %n
    %result =l add %left, %right
    ret %result
@n_ge_64
    %x_neg =w csltl %x, 0
    jnz %x_neg, @x_negative, @x_nonnegative
@x_negative
    ret -1
@x_nonnegative
    ret 0
}
