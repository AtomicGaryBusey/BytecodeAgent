export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @check

@check
    %i =l phi @start 0, @body %i_next
    %count =l phi @start 0, @body %count_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %i_offset =l mul %i, 8
    %src_addr =l add %src, %i_offset
    %src_val =l loadl %src_addr
    %result =l call %f(l %src_val)
    %dst_addr =l add %dst, %i_offset
    storel %result, %dst_addr
    %is_less =w csltl %result, %src_val
    %is_less_ext =l extsw %is_less
    %count_next =l add %count, %is_less_ext
    %i_next =l add %i, 1
    jmp @check

@end
    ret %count
}
