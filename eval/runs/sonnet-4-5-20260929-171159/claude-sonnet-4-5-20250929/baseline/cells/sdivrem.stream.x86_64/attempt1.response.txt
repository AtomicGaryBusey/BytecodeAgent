sdivrem:
    mov r8, rdx
    test rsi, rsi
    jz .Ldiv_by_zero
    movabs rax, 0x8000000000000000
    cmp rdi, rax
    jne .Lnormal_div
    cmp rsi, -1
    je .Loverflow
.Lnormal_div:
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [r8], rdx
    ret
.Ldiv_by_zero:
    xor eax, eax
    mov qword ptr [r8], rdi
    ret
.Loverflow:
    xor edx, edx
    mov qword ptr [r8], rdx
    ret
