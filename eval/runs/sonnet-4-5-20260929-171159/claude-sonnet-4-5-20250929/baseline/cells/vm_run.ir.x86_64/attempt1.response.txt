export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pc_ptr =l alloc8 8
    %depth_ptr =l alloc8 8
    storel 0, %pc_ptr
    storel 0, %depth_ptr
    jmp @loop

@loop
    %pc =l loadl %pc_ptr
    %depth =l loadl %depth_ptr
    %done_cond =w ceql %pc, %len
    jnz %done_cond, @done, @fetch

@fetch
    %pc_offset =l mul %pc, 8
    %instr_ptr =l add %code, %pc_offset
    %opcode =l loadl %instr_ptr
    %is0 =w ceql %opcode, 0
    jnz %is0, @done, @try1

@try1
    %is1 =w ceql %opcode, 1
    jnz %is1, @op_push, @try2

@try2

    jnz %is2, @op_add, @try3

@try3

@try4

@try8

@try9

@try10

    jnz %is10, @op_jz, @try11

@try11

    jnz %is11, @op_jmp, @op_unknown

@op_push

    %depth_push =l loadl %depth_ptr
    %pc1_push =l add %pc_push, 1
    %pc1_off_push =l mul %pc1_push, 8
    %val_ptr_push =l add %code, %pc1_off_push
    %val_push =l loadl %val_ptr_push
    %depth_off_push =l mul %depth_push, 8
    %stack_ptr_push =l add %stack, %depth_off_push
    storel %val_push, %stack_ptr_push
    %depth_next_push =l add %depth_push, 1
    %pc_next_push =l add %pc_push, 2
    storel %depth_next_push, %depth_ptr
    storel %pc_next_push, %pc_ptr
    jmp @loop

@op_add

    %depth_add =l loadl %depth_ptr

    %depth2_add =l sub %depth_add, 2

    %a_ptr_add =l add %stack, %a_off_add
    %a_add =l loadl %a_ptr_add
    %b_off_add =l mul %depth1_add, 8
    %b_ptr_add =l add %stack, %b_off_add
    %b_add =l loadl %b_ptr_add
    %res_add =l add %a_add, %b_add
    storel %res_add, %a_ptr_add
    %depth_next_add =l sub %depth_add, 1
    %pc_next_add =l add %pc_add, 1

    jmp @loop

@op_sub

    %a_sub =l loadl %a_ptr_sub
    %b_off_sub =l mul %depth1_sub, 8
    %b_ptr_sub =l add %stack, %b_off_sub
    %b_sub =l loadl %b_ptr_sub
    %res_sub =l sub %a_sub, %b_sub
    storel %res_sub, %a_ptr_sub
    %depth_next_sub =l sub %depth_sub, 1
    %pc_next_sub =l add %pc_sub, 1
    storel %depth_next_sub, %depth_ptr
    storel %pc_next_sub, %pc_ptr
    jmp @loop

@op_mul
    %pc_mul =l loadl %pc_ptr
    %depth_mul =l loadl %depth_ptr
    %depth1_mul =l sub %depth_mul, 1
    %depth2_mul =l sub %depth_mul, 2
    %a_off_mul =l mul %depth2_mul, 8
    %a_ptr_mul =l add %stack, %a_off_mul
    %a_mul =l loadl %a_ptr_mul

    %b_ptr_mul =l add %stack, %b_off_mul

    %depth_next_mul =l sub %depth_mul, 1

    storel %depth_next_mul, %depth_ptr
    storel %pc_next_mul, %pc_ptr

    %depth_dup =l loadl %depth_ptr

    %top_off_dup =l mul %depth1_dup, 8
    %top_ptr_dup =l add %stack, %top_off_dup

    %new_off_dup =l mul %depth_dup, 8
    %new_ptr_dup =l add %stack, %new_off_dup

    storel %top_swap, %sec_ptr_swap

    %depth_drop =l loadl %depth_ptr
    %depth_next_drop =l sub %depth_drop, 1

    %depth_over =l loadl %depth_ptr

    %sec_over =l loadl %sec_ptr_over
    %new_off_over =l mul %depth_over, 8

    %depth_next_over =l add %depth_over, 1
    %pc_next_over =l add %pc_over, 1
    storel %depth_next_over, %depth_ptr
    storel %pc_next_over, %pc_ptr
    jmp @loop

    %depth_lt =l loadl %depth_ptr
    %depth1_lt =l sub %depth_lt, 1
    %depth2_lt =l sub %depth_lt, 2
    %a_off_lt =l mul %depth2_lt, 8
    %a_ptr_lt =l add %stack, %a_off_lt
    %a_lt =l loadl %a_ptr_lt
    %b_off_lt =l mul %depth1_lt, 8
    %b_ptr_lt =l add %stack, %b_off_lt
    %b_lt =l loadl %b_ptr_lt
    %cmp_lt =w csltl %a_lt, %b_lt
    %res_lt =l extsw %cmp_lt
    storel %res_lt, %a_ptr_lt
    %depth_next_lt =l sub %depth_lt, 1
    %pc_next_lt =l add %pc_lt, 1
    storel %depth_next_lt, %depth_ptr
    storel %pc_next_lt, %pc_ptr
    jmp @loop

    %pc_jz =l loadl %pc_ptr
    %depth_jz =l loadl %depth_ptr
    %depth1_jz =l sub %depth_jz, 1
    %top_off_jz =l mul %depth1_jz, 8
    %top_ptr_jz =l add %stack, %top_off_jz
    %v_jz =l loadl %top_ptr_jz
    %depth_next_jz =l sub %depth_jz, 1
    storel %depth_next_jz, %depth_ptr
    %is_zero_jz =w ceql %v_jz, 0
    jnz %is_zero_jz, @jz_take, @jz_skip

    %pc_jz_take =l loadl %pc_ptr

    %pc1_off_jz_take =l mul %pc1_jz_take, 8
    %tgt_ptr_jz_take =l add %code, %pc1_off_jz_take
    %tgt_jz_take =l loadl %tgt_ptr_jz_take
    storel %tgt_jz_take, %pc_ptr
    jmp @loop

@jz_skip

    %pc_next_jz_skip =l add %pc_jz_skip, 2
    storel %pc_next_jz_skip, %pc_ptr
    jmp @loop

@op_jmp
    %pc_jmp =l loadl %pc_ptr

    %tgt_ptr_jmp =l add %code, %pc1_off_jmp
    %tgt_jmp =l loadl %tgt_ptr_jmp
    storel %tgt_jmp, %pc_ptr
    jmp @loop

@op_unknown
    %pc_unk =l loadl %pc_ptr
    %pc_next_unk =l add %pc_unk, 1
    storel %pc_next_unk, %pc_ptr
    jmp @loop

@done
    %depth_done =l loadl %depth_ptr
    %clear_ptr =l alloc8 8
    storel %depth_done, %clear_ptr
    jmp @clear_loop

@clear_loop
    %clear_idx =l loadl %clear_ptr
    %not_done =w csltl %clear_idx, %cap
    jnz %not_done, @clear_one, @return_value

@clear_one
    %idx_clear =l loadl %clear_ptr
    %off_clear =l mul %idx_clear, 8
    %ptr_clear =l add %stack, %off_clear
    storel 0, %ptr_clear
    %idx_next_clear =l add %idx_clear, 1
    storel %idx_next_clear, %clear_ptr
    jmp @clear_loop

@return_value
    %depth_ret =l loadl %depth_ptr
    %is_empty_ret =w ceql %depth_ret, 0
    jnz %is_empty_ret, @ret_zero, @ret_top

@ret_zero
    ret 0

@ret_top
    %depth_top =l loadl %depth_ptr
    %top_idx_ret =l sub %depth_top, 1
    %top_off_ret =l mul %top_idx_ret, 8
    %top_ptr_ret =l add %stack, %top_off_ret
    %top_ret =l loadl %top_ptr_ret
    ret %top_ret
}
