export function l $bmin_empty(l %a, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @empty, @nonempty

@empty
    ret 9223372036854775807

@nonempty
    %min0 =l loadl %a
    %i0 =l copy 1
    jmp @loop

@loop
    %i =l phi @nonempty %i0, @loop_next %i_next
    %min =l phi @nonempty %min0, @loop_next %min_updated
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done

@loop_body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %is_less =w csltl %val, %min
    jnz %is_less, @update, @no_update

@update
    %new_min =l copy %val
    jmp @loop_next

@no_update
    %old_min =l copy %min
    jmp @loop_next

@loop_next
    %min_updated =l phi @update %new_min, @no_update %old_min
    %i_next =l add %i, 1
    jmp @loop

@done
    ret %min
}
