export function l $fill_down(l %a, l %n, l %start) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @loop_body %next_i
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @done

@loop_body
    %value =l sub %start, %i
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    storel %value, %addr
    %next_i =l add %i, 1
    jmp @loop

@done
    ret %n
}
