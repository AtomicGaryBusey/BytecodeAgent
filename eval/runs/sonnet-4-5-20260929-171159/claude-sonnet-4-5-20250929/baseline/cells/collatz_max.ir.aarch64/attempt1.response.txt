export function l $collatz_max(l %n) {
@start
    jmp @loop

@loop
    %k =l phi @start 1, @next %k.1
    %max =l phi @start 0, @next %max.1
    %done =w csgtl %k, %n
    jnz %done, @ret, @body

@body
    jmp @coll

@coll
    %x =l phi @body %k, @cmerge %x.1
    %cnt =l phi @body 0, @cmerge %cnt.1
    %eq1 =w ceql %x, 1
    jnz %eq1, @cdone, @cstep

@cstep
    %mod2 =l rem %x, 2
    %even =w ceql %mod2, 0
    jnz %even, @ceven, @codd

@ceven
    %xn =l div %x, 2
    %cn =l add %cnt, 1
    jmp @cmerge

@codd
    %xm3 =l mul %x, 3
    %xn2 =l add %xm3, 1
    %cn2 =l add %cnt, 1
    jmp @cmerge

@cmerge
    %x.1 =l phi @ceven %xn, @codd %xn2
    %cnt.1 =l phi @ceven %cn, @codd %cn2
    jmp @coll

@cdone
    %gt =w csgtl %cnt, %max
    jnz %gt, @upd, @noupd

@upd
    jmp @next

@noupd
    jmp @next

@next
    %max.1 =l phi @upd %cnt, @noupd %max
    %k.1 =l add %k, 1
    jmp @loop

@ret
    ret %max
}
