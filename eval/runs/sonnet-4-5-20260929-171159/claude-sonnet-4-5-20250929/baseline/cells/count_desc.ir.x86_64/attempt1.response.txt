export function l $count_desc(l %a, l %n) {
@start
    %cmp_n_lt_2 =w csltl %n, 2
    jnz %cmp_n_lt_2, @return_zero, @init

@return_zero
    ret 0

@init
    %n_minus_1 =l sub %n, 1
    jmp @loop

@loop
    %i =l phi @init 0, @next %i_next
    %count =l phi @init 0, @next %count_next
    %i_lt_n_minus_1 =w csltl %i, %n_minus_1
    jnz %i_lt_n_minus_1, @loop_body, @return_count

@loop_body
    %i_times_8 =l mul %i, 8
    %addr_i =l add %a, %i_times_8
    %val_i =l loadl %addr_i
    %i_plus_1 =l add %i, 1
    %i_plus_1_times_8 =l mul %i_plus_1, 8
    %addr_i_plus_1 =l add %a, %i_plus_1_times_8
    %val_i_plus_1 =l loadl %addr_i_plus_1
    %is_desc =w csgtl %val_i, %val_i_plus_1
    jnz %is_desc, @increment, @no_increment

@increment
    %count_inc =l add %count, 1
    jmp @next

@no_increment
    jmp @next

@next
    %count_next =l phi @increment %count_inc, @no_increment %count
    %i_next =l add %i, 1
    jmp @loop

@return_count
    ret %count
}
