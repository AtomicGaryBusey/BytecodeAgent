export function l $count_desc(l %a, l %n) {
@start
    %cmp0 = csltl %n, 2
    jnz %cmp0, @ret_zero, @loop_init
@ret_zero
    ret 0
@loop_init
    %limit = sub %n, 1
    jmp @loop_head
@loop_head
    %i = phi @loop_init 0, @loop_body %i_next
    %count = phi @loop_init 0, @loop_body %count_next
    %cmp1 = csltl %i, %limit
    jnz %cmp1, @loop_body, @loop_end
@loop_body
    %offset_i = mul %i, 8
    %addr_i = add %a, %offset_i
    %val_i = loadl %addr_i
    %i_next = add %i, 1
    %offset_i1 = mul %i_next, 8
    %addr_i1 = add %a, %offset_i1
    %val_i1 = loadl %addr_i1
    %is_desc = csgtl %val_i, %val_i1
    %is_desc_l = extuw %is_desc
    %count_next = add %count, %is_desc_l
    jmp @loop_head
@loop_end
    ret %count
}
