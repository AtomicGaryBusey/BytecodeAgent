export function l $count_desc(l %a, l %n) {
@start
    %cmp0 =w csltl %n, 2
    jnz %cmp0, @ret_zero, @loop_init
@ret_zero
    ret 0
@loop_init
    %limit =l sub %n, 1
    jmp @loop_head
@loop_head
    %i =l phi @loop_init 0, @loop_body %i_next
    %count =l phi @loop_init 0, @loop_body %count_next
    %cmp1 =w csltl %i, %limit
    jnz %cmp1, @loop_body, @loop_end
@loop_body
    %offset_i =l mul %i, 8
    %addr_i =l add %a, %offset_i
    %val_i =l loadl %addr_i
    %i_next =l add %i, 1
    %offset_i1 =l mul %i_next, 8
    %addr_i1 =l add %a, %offset_i1
    %val_i1 =l loadl %addr_i1
    %is_desc =w csgtl %val_i, %val_i1
    %is_desc_l =l extuw %is_desc
    %count_next =l add %count, %is_desc_l
    jmp @loop_head
@loop_end
    ret %count
}
