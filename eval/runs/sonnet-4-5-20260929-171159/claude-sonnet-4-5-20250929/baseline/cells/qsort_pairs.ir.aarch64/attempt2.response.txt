export function l $qsort_pairs(l %a, l %n) {
@start
    %cond =w csgtl %n, 1
    jnz %cond, @setup, @return_zero

@return_zero
    ret 0

@setup
    %pivot_idx =l sub %n, 1
    %pivot_offset1 =l shl %pivot_idx, 1
    %pivot_offset =l shl %pivot_offset1, 3
    %pivot_addr =l add %a, %pivot_offset
    %pivot_key =l loadl %pivot_addr
    %pivot_val_addr =l add %pivot_addr, 8
    %pivot_val =l loadl %pivot_val_addr
    %limit =l sub %n, 1
    jmp @loop_init

@loop_init
    %i_init =l copy -1
    %j_init =l copy 0
    jmp @loop_test

@loop_test
    %i =l phi @loop_init %i_init, @loop_next %i_next
    %j =l phi @loop_init %j_init, @loop_next %j_next
    %j_cond =w csltl %j, %limit
    jnz %j_cond, @loop_body, @after_loop

@loop_body
    %j_offset1 =l shl %j, 1
    %j_offset =l shl %j_offset1, 3
    %j_addr =l add %a, %j_offset
    %j_key =l loadl %j_addr
    %j_val_addr =l add %j_addr, 8
    %j_val =l loadl %j_val_addr
    %key_lt =w csltl %j_key, %pivot_key
    jnz %key_lt, @do_swap, @no_swap_check

@no_swap_check
    %key_eq =w ceql %j_key, %pivot_key
    jnz %key_eq, @val_check, @no_swap

@val_check
    %val_lt =w csltl %j_val, %pivot_val
    jnz %val_lt, @do_swap, @no_swap

@do_swap
    %i_inc =l add %i, 1
    %i_offset1 =l shl %i_inc, 1
    %i_offset =l shl %i_offset1, 3
    %i_addr =l add %a, %i_offset
    %i_key =l loadl %i_addr
    %i_val_addr =l add %i_addr, 8
    %i_val =l loadl %i_val_addr
    storel %j_key, %i_addr
    storel %i_key, %j_addr
    storel %j_val, %i_val_addr
    storel %i_val, %j_val_addr
    jmp @swap_next

@swap_next
    %i_after_swap =l copy %i_inc
    %j_after_swap =l add %j, 1
    jmp @loop_next

@no_swap
    %i_after_no_swap =l copy %i
    %j_after_no_swap =l add %j, 1
    jmp @loop_next

@loop_next
    %i_next =l phi @swap_next %i_after_swap, @no_swap %i_after_no_swap
    %j_next =l phi @swap_next %j_after_swap, @no_swap %j_after_no_swap
    jmp @loop_test

@after_loop
    %i_final =l add %i, 1
    %i_final_offset1 =l shl %i_final, 1
    %i_final_offset =l shl %i_final_offset1, 3
    %i_final_addr =l add %a, %i_final_offset
    %i_final_key =l loadl %i_final_addr
    %i_final_val_addr =l add %i_final_addr, 8
    %i_final_val =l loadl %i_final_val_addr
    storel %pivot_key, %i_final_addr
    storel %pivot_val, %i_final_val_addr
    storel %i_final_key, %pivot_addr
    storel %i_final_val, %pivot_val_addr
    %dummy1 =l call $qsort_pairs(l %a, l %i_final)
    %right_idx =l add %i_final, 1
    %right_offset1 =l shl %right_idx, 1
    %right_offset =l shl %right_offset1, 3
    %right_addr =l add %a, %right_offset
    %right_size =l sub %n, %right_idx
    %dummy2 =l call $qsort_pairs(l %right_addr, l %right_size)
    ret 0
}
