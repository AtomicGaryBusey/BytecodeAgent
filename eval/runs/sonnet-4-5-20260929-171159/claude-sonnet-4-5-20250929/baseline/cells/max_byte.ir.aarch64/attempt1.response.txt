export function l $max_byte(l %p, l %n) {
@start
    %is_zero =w ceql %n, 0
    jnz %is_zero, @ret_zero, @loop_start
@ret_zero
    ret 0
@loop_start
    jmp @loop_cond
@loop_cond
    %i =l phi @loop_start 0, @loop_next %i_next
    %max =l phi @loop_start 0, @loop_next %max_next
    %cmp =w csltl %i, %n
    jnz %cmp, @loop_body, @loop_end
@loop_body
    %addr =l add %p, %i
    %byte =l loadub %addr
    %gt =w cugtl %byte, %max
    jnz %gt, @update_max, @keep_max
@update_max
    %max_1 =l copy %byte
    jmp @loop_next
@keep_max
    %max_2 =l copy %max
    jmp @loop_next
@loop_next
    %max_next =l phi @update_max %max_1, @keep_max %max_2
    %i_next =l add %i, 1
    jmp @loop_cond
@loop_end
    ret %max
}
