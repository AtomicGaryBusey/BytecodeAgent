binary_search:
    test rsi, rsi
    jz .not_found
    xor rax, rax
    lea rcx, [rsi - 1]
.loop:
    cmp rax, rcx
    jg .not_found
    mov r8, rax
    add r8, rcx
    shr r8, 1
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .found
    jl .search_right
    lea rcx, [r8 - 1]
    jmp .loop
.search_right:
    lea rax, [r8 + 1]
    jmp .loop
.found:
    mov rax, r8
    ret
.not_found:
    mov rax, -1
    ret
