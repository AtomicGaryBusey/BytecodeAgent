export function l $binary_search(l %a, l %n, l %key) {
@start
    %init_left =l copy 0
    %init_right =l sub %n, 1
    jmp @loop

@loop
    %left =l phi @start %init_left, @go_right %left_next, @go_left %left_next2
    %right =l phi @start %init_right, @go_right %right_next, @go_left %right_next2
    %cond =w csgtl %left, %right
    jnz %cond, @notfound, @body

@body
    %diff =l sub %right, %left
    %half =l sar %diff, 1
    %mid =l add %left, %half
    %offset =l mul %mid, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %eq =w ceql %val, %key
    jnz %eq, @found, @check_less

@check_less
    %lt =w csltl %val, %key
    jnz %lt, @go_right, @go_left

@go_right
    %left_next =l add %mid, 1
    %right_next =l copy %right
    jmp @loop

@go_left
    %left_next2 =l copy %left
    %right_next2 =l sub %mid, 1
    jmp @loop

@found
    ret %mid

@notfound
    %neg1 =l copy -1
    ret %neg1
}
