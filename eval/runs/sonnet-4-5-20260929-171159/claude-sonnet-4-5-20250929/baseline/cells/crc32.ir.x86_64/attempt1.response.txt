export function l $crc32(l %p, l %n) {
@start
    jmp @loop_outer

@loop_outer
    %crc =w phi @start 0xFFFFFFFF, @loop_outer_next %crc_after_inner
    %i =l phi @start 0, @loop_outer_next %i_next
    %cond_outer =w cultl %i, %n
    jnz %cond_outer, @loop_outer_body, @done

@loop_outer_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc_xor =w xor %crc, %byte
    jmp @loop_inner

@loop_inner
    %crc_iter =w phi @loop_outer_body %crc_xor, @loop_inner_next %crc_inner_next
    %j =w phi @loop_outer_body 0, @loop_inner_next %j_next
    %cond_inner =w cultw %j, 8
    jnz %cond_inner, @loop_inner_body2, @loop_outer_next

@loop_inner_body2
    %bit =w and %crc_iter, 1
    jnz %bit, @crc_odd, @crc_even

@crc_odd
    %crc_shr_odd =w shr %crc_iter, 1
    %crc_odd_result =w xor %crc_shr_odd, 0xEDB88320
    jmp @loop_inner_next

@crc_even
    %crc_even_result =w shr %crc_iter, 1
    jmp @loop_inner_next

@loop_inner_next
    %crc_inner_next =w phi @crc_odd %crc_odd_result, @crc_even %crc_even_result
    %j_next =w add %j, 1
    jmp @loop_inner

@loop_outer_next
    %crc_after_inner =w copy %crc_iter
    %i_next =l add %i, 1
    jmp @loop_outer

@done
    %result =w xor %crc, 0xFFFFFFFF
    %result_l =l extuw %result
    ret %result_l
}
