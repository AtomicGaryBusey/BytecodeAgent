bubble_sort:
    cmp rsi, 2
    jl .Ldone
    xor rcx, rcx
    lea r11, [rsi - 1]
.Louter:
    cmp rcx, r11
    jge .Ldone
    xor rdx, rdx
    lea r8, [rsi - 2]
    sub r8, rcx
.Linner:
    cmp rdx, r8
    jg .Linner_end
    mov r9, qword ptr [rdi + rdx*8]
    mov r10, qword ptr [rdi + rdx*8 + 8]
    cmp r9, r10
    jle .Lno_swap
    mov qword ptr [rdi + rdx*8], r10
    mov qword ptr [rdi + rdx*8 + 8], r9
.Lno_swap:
    inc rdx
    jmp .Linner
.Linner_end:
    inc rcx
    jmp .Louter
.Ldone:
    xor rax, rax
    ret
