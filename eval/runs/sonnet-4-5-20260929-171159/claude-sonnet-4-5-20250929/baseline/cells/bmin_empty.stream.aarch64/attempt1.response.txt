bmin_empty:
    cbz x1, .Lempty
    ldr x2, [x0]
    mov x3, #1
    cmp x3, x1
    b.ge .Lreturn
.Lloop:
    ldr x4, [x0, x3, lsl #3]
    cmp x4, x2
    csel x2, x4, x2, lt
    add x3, x3, #1
    cmp x3, x1
    b.lt .Lloop
.Lreturn:
    mov x0, x2
    ret
.Lempty:
    movn x0, #0x8000, lsl #48
    ret
