export function l $popcount(l %x) {
@start
    jmp @loop

@loop
    %x.loop =l phi @start %x, @body %x.new
    %count.loop =l phi @start 0, @body %count.new
    %is_zero =w ceql %x.loop, 0
    jnz %is_zero, @end, @body

@body
    %x_minus_1 =l sub %x.loop, 1
    %x.new =l and %x.loop, %x_minus_1
    %count.new =l add %count.loop, 1
    jmp @loop

@end
    ret %count.loop
}
