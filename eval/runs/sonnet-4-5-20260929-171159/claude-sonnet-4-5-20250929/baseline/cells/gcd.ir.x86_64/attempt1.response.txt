export function l $gcd(l %a, l %b) {
@start
    jmp @loop

@loop
    %a_curr =l phi @start %a, @body %a_next
    %b_curr =l phi @start %b, @body %b_next
    %is_zero =w ceql %b_curr, 0
    jnz %is_zero, @end, @body

@body
    %a_next =l copy %b_curr
    %b_next =l rem %a_curr, %b_curr
    jmp @loop

@end
    ret %a_curr
}
