export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%mask =l copy 4294967295
	%sh32 =l copy 32
	%n =l add %na, %nb
	jmp @zloop
@zloop
	%zi =l phi @start %zero, @zbody %zi1
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zbody, @iloop
@zbody
	%zoff =l mul %zi, %eight
	%zp =l add %r, %zoff
	storel %zero, %zp
	%zi1 =l add %zi, %one
	jmp @zloop
@iloop
	%i =l phi @zloop %zero, @jdone %i1
	%icmp =w csltl %i, %na
	jnz %icmp, @ibody, @sloop
@ibody
	%ioff =l mul %i, %eight
	%ap =l add %a, %ioff
	%ai =l loadl %ap
	jmp @jloop
@jloop
	%j =l phi @ibody %zero, @jbody %j1
	%carry =l phi @ibody %zero, @jbody %cout
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @jbody, @jdone
@jbody
	%joff =l mul %j, %eight
	%bp =l add %b, %joff
	%bj =l loadl %bp
	%al =l and %ai, %mask
	%ah =l shr %ai, %sh32
	%bl =l and %bj, %mask
	%bh =l shr %bj, %sh32
	%p0 =l mul %al, %bl
	%p1 =l mul %al, %bh
	%p2 =l mul %ah, %bl
	%p3 =l mul %ah, %bh
	%mid =l add %p1, %p2
	%midcw =w cultl %mid, %p1
	%midc =l extuw %midcw
	%midlo =l shl %mid, %sh32
	%lo =l add %p0, %midlo
	%locw =w cultl %lo, %p0
	%loc =l extuw %locw
	%midhi =l shr %mid, %sh32
	%midcsh =l shl %midc, %sh32
	%hi0 =l add %p3, %midhi
	%hi1 =l add %hi0, %midcsh
	%hi =l add %hi1, %loc
	%ij =l add %i, %j
	%ijoff =l mul %ij, %eight
	%rp =l add %r, %ijoff
	%rij =l loadl %rp
	%t =l add %rij, %lo
	%c1w =w cultl %t, %rij
	%c1 =l extuw %c1w
	%t2 =l add %t, %carry
	%c2w =w cultl %t2, %t
	%c2 =l extuw %c2w
	storel %t2, %rp
	%ctmp =l add %hi, %c1
	%cout =l add %ctmp, %c2
	%j1 =l add %j, %one
	jmp @jloop
@jdone
	%inb =l add %i, %nb
	%inboff =l mul %inb, %eight
	%rp2 =l add %r, %inboff
	storel %carry, %rp2
	%i1 =l add %i, %one
	jmp @iloop
@sloop
	%k =l phi @iloop %n, @scheck %km1
	%kgt =w csltl %zero, %k
	jnz %kgt, @scheck, @sret
@scheck
	%km1 =l sub %k, %one
	%koff =l mul %km1, %eight
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kis0 =w ceql %kv, %zero
	jnz %kis0, @sloop, @sretk
@sret
	ret %zero
@sretk
	ret %k
}
