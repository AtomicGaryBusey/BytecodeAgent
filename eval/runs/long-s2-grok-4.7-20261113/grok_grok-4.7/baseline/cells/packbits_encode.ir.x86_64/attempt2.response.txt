export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	storel 0, %pi
	storel 0, %po
	jmp @main
@main
	%i =l loadl %pi
	%ilt =w csltl %i, %n
	jnz %ilt, @dorun, @success
@success
	%odone =l loadl %po
	ret %odone
@fail
	%neg =l copy -1
	ret %neg
@dorun
	%sp =l add %src, %i
	%b =w loadub %sp
	%r0 =w copy 1
	%k0 =l add %i, 1
	jmp @run1
@run1
	%r =w phi @dorun %r0, @run1inc %r1
	%k =l phi @dorun %k0, @run1inc %k1
	%klt =w csltl %k, %n
	jnz %klt, @run1ck, @run1done
@run1ck
	%rlt =w csltw %r, 128
	jnz %rlt, @run1cmp, @run1done
@run1cmp
	%kp =l add %src, %k
	%bk =w loadub %kp
	%eq =w ceqw %bk, %b
	jnz %eq, @run1inc, @run1done
@run1inc
	%r1 =w add %r, 1
	%k1 =l add %k, 1
	jmp @run1
@run1done
	%ge =w csgew %r, 3
	jnz %ge, @repeat, @literal
@repeat
	%o1 =l loadl %po
	%need =l add %o1, 2
	%fits =w cslel %need, %cap
	jnz %fits, @repeatw, @fail
@repeatw
	%hdr =w sub 257, %r
	%op =l add %out, %o1
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %b, %op1
	%onew =l add %o1, 2
	storel %onew, %po
	%rl =l extuw %r
	%inew =l add %i, %rl
	storel %inew, %pi
	jmp @main
@literal
	%j0 =l copy %i
	jmp @litscan
@litscan
	%j =l phi @literal %j0, @litadv %j1
	%jlt =w csltl %j, %n
	jnz %jlt, @litlen, @litemit
@litlen
	%len =l sub %j, %i
	%lenlt =w csltl %len, 128
	jnz %lenlt, @litrun, @litemit
@litrun
	%jp =l add %src, %j
	%bj =w loadub %jp
	%r20 =w copy 1
	%k20 =l add %j, 1
	jmp @run2
@run2
	%r2 =w phi @litrun %r20, @run2inc %r21
	%k2 =l phi @litrun %k20, @run2inc %k21
	%k2lt =w csltl %k2, %n
	jnz %k2lt, @run2ck, @run2done
@run2ck
	%r2lt =w csltw %r2, 128
	jnz %r2lt, @run2cmp, @run2done
@run2cmp
	%k2p =l add %src, %k2
	%bk2 =w loadub %k2p
	%eq2 =w ceqw %bk2, %bj
	jnz %eq2, @run2inc, @run2done
@run2inc
	%r21 =w add %r2, 1
	%k21 =l add %k2, 1
	jmp @run2
@run2done
	%r2ge =w csgew %r2, 3
	jnz %r2ge, @litemit, @litadv
@litadv
	%j1 =l add %j, 1
	jmp @litscan
@litemit
	%L =l sub %j, %i
	%sz =l add %L, 1
	%o2 =l loadl %po
	%need2 =l add %o2, %sz
	%fits2 =w cslel %need2, %cap
	jnz %fits2, @litwrite, @fail
@litwrite
	%Lm1 =l sub %L, 1
	%hdr2 =w copy %Lm1
	%op2 =l add %out, %o2
	storeb %hdr2, %op2
	%c0 =l copy 0
	jmp @copy
@copy
	%c =l phi @litwrite %c0, @copyinc %c1
	%clt =w csltl %c, %L
	jnz %clt, @copybyte, @copydone
@copybyte
	%si =l add %i, %c
	%sp2 =l add %src, %si
	%sb =w loadub %sp2
	%base =l add %o2, 1
	%di =l add %base, %c
	%dp =l add %out, %di
	storeb %sb, %dp
	jmp @copyinc
@copyinc
	%c1 =l add %c, 1
	jmp @copy
@copydone
	%onew2 =l add %o2, %sz
	storel %onew2, %po
	storel %j, %pi
	jmp @main
}
