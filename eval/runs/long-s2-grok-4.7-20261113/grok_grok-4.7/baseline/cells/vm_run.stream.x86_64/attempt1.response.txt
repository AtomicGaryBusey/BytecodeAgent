vm_run:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
.Lloop:
cmp rbx, r13
je .Ldone
mov rax, qword ptr [r12 + rbx*8]
test rax, rax
jz .Ldone
cmp rax, 1
je .Lpush
cmp rax, 2
je .Ladd
cmp rax, 3
je .Lsub
cmp rax, 4
je .Lmul
cmp rax, 5
je .Ldup
cmp rax, 6
je .Lswap
cmp rax, 7
je .Ldrop
cmp rax, 8
je .Lover
cmp rax, 9
je .Llt
cmp rax, 10
je .Ljz
cmp rax, 11
je .Ljmp
inc rbx
jmp .Lloop
.Lpush:
mov rax, qword ptr [r12 + rbx*8 + 8]
mov qword ptr [r14 + rbp*8], rax
inc rbp
add rbx, 2
jmp .Lloop
.Ladd:
mov rax, qword ptr [r14 + rbp*8 - 16]
add rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Lsub:
mov rax, qword ptr [r14 + rbp*8 - 16]
sub rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Lmul:
mov rax, qword ptr [r14 + rbp*8 - 16]
imul rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Ldup:
mov rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8], rax
inc rbp
inc rbx
jmp .Lloop
.Lswap:
mov rax, qword ptr [r14 + rbp*8 - 8]
mov rcx, qword ptr [r14 + rbp*8 - 16]
mov qword ptr [r14 + rbp*8 - 16], rax
mov qword ptr [r14 + rbp*8 - 8], rcx
inc rbx
jmp .Lloop
.Ldrop:
dec rbp
inc rbx
jmp .Lloop
.Lover:
mov rax, qword ptr [r14 + rbp*8 - 16]
mov qword ptr [r14 + rbp*8], rax
inc rbp
inc rbx
jmp .Lloop
.Llt:
mov rax, qword ptr [r14 + rbp*8 - 16]
cmp rax, qword ptr [r14 + rbp*8 - 8]
setl al
movzx rax, al
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Ljz:
dec rbp
cmp qword ptr [r14 + rbp*8], 0
je .Ljz_taken
add rbx, 2
jmp .Lloop
.Ljz_taken:
mov rbx, qword ptr [r12 + rbx*8 + 8]
jmp .Lloop
.Ljmp:
mov rbx, qword ptr [r12 + rbx*8 + 8]
jmp .Lloop
.Ldone:
xor eax, eax
test rbp, rbp
jz .Lzero
mov rax, qword ptr [r14 + rbp*8 - 8]
.Lzero:
mov rcx, rbp
.Lzero_loop:
cmp rcx, r15
jge .Lepilogue
mov qword ptr [r14 + rcx*8], 0
inc rcx
jmp .Lzero_loop
.Lepilogue:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
