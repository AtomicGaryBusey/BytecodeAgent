mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rsi
    mov r12, rdx
    mov r13, rcx
    mov r14, r8
    mov r15, rdi
    mov rax, rbx
    imul rax, rbx
    mov rbp, rax
    xor r11, r11
.Lid_i:
    cmp r11, rbx
    jge .Lcpy_init
    xor rcx, rcx
.Lid_j:
    cmp rcx, rbx
    jge .Lid_next_i
    mov rsi, r11
    imul rsi, rbx
    add rsi, rcx
    xor eax, eax
    cmp r11, rcx
    jne .Lid_st
    mov eax, 1
.Lid_st:
    mov qword ptr [r13 + rsi*8], rax
    inc rcx
    jmp .Lid_j
.Lid_next_i:
    inc r11
    jmp .Lid_i
.Lcpy_init:
    xor rcx, rcx
.Lcpy:
    cmp rcx, rbp
    jge .Lexp
    mov rax, qword ptr [r15 + rcx*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    jmp .Lcpy
.Lexp:
    test r12, r12
    jz .Lzero_all
    test r12, 1
    jz .Lsq_shift
    mov r8, r13
    mov r9, r14
    mov r10, rbp
    shl r10, 3
    add r10, r14
    call .Lmatmul
    xor rcx, rcx
.Lcopy_out:
    cmp rcx, rbp
    jge .Lsq_shift
    mov rax, qword ptr [r10 + rcx*8]
    mov qword ptr [r13 + rcx*8], rax
    inc rcx
    jmp .Lcopy_out
.Lsq_shift:
    shr r12, 1
    jz .Lzero_all
    mov r8, r14
    mov r9, r14
    mov r10, rbp
    shl r10, 3
    add r10, r14
    call .Lmatmul
    xor rcx, rcx
.Lcopy_base:
    cmp rcx, rbp
    jge .Lexp
    mov rax, qword ptr [r10 + rcx*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    jmp .Lcopy_base
.Lzero_all:
    mov rcx, rbp
    add rcx, rbp
    xor eax, eax
    xor rdx, rdx
.Lzero:
    cmp rdx, rcx
    jge .Lret
    mov qword ptr [r14 + rdx*8], rax
    inc rdx
    jmp .Lzero
.Lret:
    xor eax, eax
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lmatmul:
    xor r11, r11
.Lmm_i:
    cmp r11, rbx
    jge .Lmm_ret
    xor rcx, rcx
.Lmm_j:
    cmp rcx, rbx
    jge .Lmm_next_i
    xor rax, rax
    xor rdx, rdx
.Lmm_t:
    cmp rdx, rbx
    jge .Lmm_st
    mov rsi, r11
    imul rsi, rbx
    add rsi, rdx
    mov rsi, qword ptr [r8 + rsi*8]
    mov rdi, rdx
    imul rdi, rbx
    add rdi, rcx
    mov rdi, qword ptr [r9 + rdi*8]
    imul rsi, rdi
    add rax, rsi
    inc rdx
    jmp .Lmm_t
.Lmm_st:
    mov rsi, r11
    imul rsi, rbx
    add rsi, rcx
    mov qword ptr [r10 + rsi*8], rax
    inc rcx
    jmp .Lmm_j
.Lmm_next_i:
    inc r11
    jmp .Lmm_i
.Lmm_ret:
    ret
