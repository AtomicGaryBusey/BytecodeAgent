export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%tmpsz =l mul %nn, 2
	%zero =l copy 0
	%one =l copy 1
	jmp @zout
@zout
	%zi =l phi @start %zero, @zoutb %zinext
	%zc =w csltl %zi, %nn
	jnz %zc, @zoutb, @diag
@zoutb
	%zoff =l mul %zi, 8
	%zptr =l add %out, %zoff
	storel %zero, %zptr
	%zinext =l add %zi, 1
	jmp @zout
@diag
	%di =l phi @zout %zero, @diagb %dinext
	%dc =w csltl %di, %n
	jnz %dc, @diagb, @pow
@diagb
	%din =l mul %di, %n
	%didx =l add %din, %di
	%doff =l mul %didx, 8
	%dptr =l add %out, %doff
	storel %one, %dptr
	%dinext =l add %di, 1
	jmp @diag
@pow
	%p =l phi @diag %zero, @powinc %pnext
	%pc =w csltl %p, %k
	jnz %pc, @muli, @ztmp
@muli
	%i =l phi @pow %zero, @mulinext %inext
	%ic =w csltl %i, %n
	jnz %ic, @mulj, @pcopy
@mulj
	%j =l phi @muli %zero, @mulst %jnext
	%jc =w csltl %j, %n
	jnz %jc, @mult, @mulinext
@mult
	%t =l phi @mulj %zero, @multb %tnext
	%sum =l phi @mulj %zero, @multb %sumnext
	%tc =w csltl %t, %n
	jnz %tc, @multb, @mulst
@multb
	%in =l mul %i, %n
	%ai =l add %in, %t
	%aoff =l mul %ai, 8
	%ap =l add %out, %aoff
	%av =l loadl %ap
	%tn =l mul %t, %n
	%bi =l add %tn, %j
	%boff =l mul %bi, 8
	%bp =l add %a, %boff
	%bv =l loadl %bp
	%prod =l mul %av, %bv
	%sumnext =l add %sum, %prod
	%tnext =l add %t, 1
	jmp @mult
@mulst
	%in2 =l mul %i, %n
	%ci =l add %in2, %j
	%coff =l mul %ci, 8
	%cp =l add %tmp, %coff
	storel %sum, %cp
	%jnext =l add %j, 1
	jmp @mulj
@mulinext
	%inext =l add %i, 1
	jmp @muli
@pcopy
	%ci2 =l phi @muli %zero, @pcopyb %ci2next
	%cc =w csltl %ci2, %nn
	jnz %cc, @pcopyb, @powinc
@pcopyb
	%co2 =l mul %ci2, 8
	%sp =l add %tmp, %co2
	%sv =l loadl %sp
	%dp =l add %out, %co2
	storel %sv, %dp
	%ci2next =l add %ci2, 1
	jmp @pcopy
@powinc
	%pnext =l add %p, 1
	jmp @pow
@ztmp
	%ti =l phi @pow %zero, @ztmpb %tinext
	%ztc =w csltl %ti, %tmpsz
	jnz %ztc, @ztmpb, @done
@ztmpb
	%toff =l mul %ti, 8
	%tp =l add %tmp, %toff
	storel %zero, %tp
	%tinext =l add %ti, 1
	jmp @ztmp
@done
	ret %zero
}
