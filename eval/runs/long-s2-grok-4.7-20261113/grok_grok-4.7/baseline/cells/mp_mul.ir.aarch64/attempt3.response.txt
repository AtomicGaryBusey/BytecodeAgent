export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l const 0
	%one =l const 1
	%eight =l const 8
	%mask =l const 4294967295
	%thirtytwo =l const 32
	%n =l add %na, %nb
	jmp @zcond
@zcond
	%zi =l phi @start %zero, @zbody %znext
	%zlt =w csltl %zi, %n
	jnz %zlt, @zbody, @zdone
@zbody
	%zoff =l mul %zi, %eight
	%zptr =l add %r, %zoff
	storel %zero, %zptr
	%znext =l add %zi, %one
	jmp @zcond
@zdone
	%nais0 =w ceql %na, %zero
	jnz %nais0, @retzero, @nbcheck
@nbcheck
	%nbis0 =w ceql %nb, %zero
	jnz %nbis0, @retzero, @mulinit
@retzero
	ret %zero
@mulinit
	jmp @iloop
@iloop
	%i =l phi @mulinit %zero, @jdone %inext
	%ilt =w csltl %i, %na
	jnz %ilt, @iload, @scinit
@iload
	%ioff =l mul %i, %eight
	%ap =l add %a, %ioff
	%ai =l loadl %ap
	jmp @jloop
@jloop
	%j =l phi @iload %zero, @jbody %jnext
	%carry =l phi @iload %zero, @jbody %cout
	%jlt =w csltl %j, %nb
	jnz %jlt, @jbody, @jdone
@jbody
	%joff =l mul %j, %eight
	%bp =l add %b, %joff
	%bj =l loadl %bp
	%xlo =l and %ai, %mask
	%xhi =l shr %ai, %thirtytwo
	%ylo =l and %bj, %mask
	%yhi =l shr %bj, %thirtytwo
	%p0 =l mul %xlo, %ylo
	%p1 =l mul %xlo, %yhi
	%p2 =l mul %xhi, %ylo
	%p3 =l mul %xhi, %yhi
	%mid =l add %p1, %p2
	%midc =w cultl %mid, %p1
	%midcl =l extuw %midc
	%midlo =l and %mid, %mask
	%midhi =l shr %mid, %thirtytwo
	%midlosh =l shl %midlo, %thirtytwo
	%lo =l add %p0, %midlosh
	%loc =w cultl %lo, %p0
	%locl =l extuw %loc
	%midcsh =l shl %midcl, %thirtytwo
	%h1 =l add %p3, %midhi
	%h2 =l add %h1, %midcsh
	%hi =l add %h2, %locl
	%idx =l add %i, %j
	%idoff =l mul %idx, %eight
	%rp =l add %r, %idoff
	%rv =l loadl %rp
	%s1 =l add %rv, %lo
	%c1 =w cultl %s1, %rv
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	storel %s2, %rp
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%csum =l add %c1l, %c2l
	%cout =l add %hi, %csum
	%jnext =l add %j, %one
	jmp @jloop
@jdone
	%iend =l add %i, %nb
	%eoff =l mul %iend, %eight
	%eptr =l add %r, %eoff
	storel %carry, %eptr
	%inext =l add %i, %one
	jmp @iloop
@scinit
	jmp @scond
@scond
	%k =l phi @scinit %n, @sdec %kdec
	%kpos =w csltl %zero, %k
	jnz %kpos, @sbody, @retzero
@sbody
	%kdec =l sub %k, %one
	%koff =l mul %kdec, %eight
	%kptr =l add %r, %koff
	%kv =l loadl %kptr
	%kzero =w ceql %kv, %zero
	jnz %kzero, @sdec, @retk
@sdec
	jmp @scond
@retk
	ret %k
}
