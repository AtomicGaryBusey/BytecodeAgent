export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dpp =l alloc8 8
	%zero =l copy 0
	storel %zero, %pcp
	storel %zero, %dpp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch
@fetch
	%pc8 =l mul %pc, 8
	%cip =l add %code, %pc8
	%op =l loadl %cip
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @t1
@t1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @t2
@t2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @t3
@t3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @t4
@t4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @t5
@t5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @t6
@t6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @t7
@t7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @t8
@t8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @t9
@t9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @t10
@t10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @t11
@t11
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @other
@push
	%ppc =l add %pc, 1
	%ppc8 =l mul %ppc, 8
	%pp =l add %code, %ppc8
	%pv =l loadl %pp
	%pd =l loadl %dpp
	%pd8 =l mul %pd, 8
	%pa =l add %stack, %pd8
	storel %pv, %pa
	%pd1 =l add %pd, 1
	storel %pd1, %dpp
	%ppc2 =l add %pc, 2
	storel %ppc2, %pcp
	jmp @loop
@add
	%ad =l loadl %dpp
	%ad1 =l sub %ad, 1
	%ad2 =l sub %ad, 2
	%ad18 =l mul %ad1, 8
	%ad28 =l mul %ad2, 8
	%abp =l add %stack, %ad18
	%aap =l add %stack, %ad28
	%ab =l loadl %abp
	%aa =l loadl %aap
	%asum =l add %aa, %ab
	storel %asum, %aap
	storel %ad1, %dpp
	%apc =l add %pc, 1
	storel %apc, %pcp
	jmp @loop
@sub
	%sd =l loadl %dpp
	%sd1 =l sub %sd, 1
	%sd2 =l sub %sd, 2
	%sd18 =l mul %sd1, 8
	%sd28 =l mul %sd2, 8
	%sbp =l add %stack, %sd18
	%sap =l add %stack, %sd28
	%sb =l loadl %sbp
	%sa =l loadl %sap
	%sdiff =l sub %sa, %sb
	storel %sdiff, %sap
	storel %sd1, %dpp
	%spc =l add %pc, 1
	storel %spc, %pcp
	jmp @loop
@mul
	%md =l loadl %dpp
	%md1 =l sub %md, 1
	%md2 =l sub %md, 2
	%md18 =l mul %md1, 8
	%md28 =l mul %md2, 8
	%mbp =l add %stack, %md18
	%map =l add %stack, %md28
	%mb =l loadl %mbp
	%ma =l loadl %map
	%mprod =l mul %ma, %mb
	storel %mprod, %map
	storel %md1, %dpp
	%mpc =l add %pc, 1
	storel %mpc, %pcp
	jmp @loop
@dup
	%dd =l loadl %dpp
	%dd1 =l sub %dd, 1
	%dd18 =l mul %dd1, 8
	%dd8 =l mul %dd, 8
	%dfrom =l add %stack, %dd18
	%dto =l add %stack, %dd8
	%dv =l loadl %dfrom
	storel %dv, %dto
	%ddn =l add %dd, 1
	storel %ddn, %dpp
	%dpc =l add %pc, 1
	storel %dpc, %pcp
	jmp @loop
@swap
	%wd =l loadl %dpp
	%w1 =l sub %wd, 1
	%w2 =l sub %wd, 2
	%w18 =l mul %w1, 8
	%w28 =l mul %w2, 8
	%wp1 =l add %stack, %w18
	%wp2 =l add %stack, %w28
	%wv1 =l loadl %wp1
	%wv2 =l loadl %wp2
	storel %wv2, %wp1
	storel %wv1, %wp2
	%wpc =l add %pc, 1
	storel %wpc, %pcp
	jmp @loop
@drop
	%xd =l loadl %dpp
	%xd1 =l sub %xd, 1
	storel %xd1, %dpp
	%xpc =l add %pc, 1
	storel %xpc, %pcp
	jmp @loop
@over
	%od =l loadl %dpp
	%od2 =l sub %od, 2
	%od28 =l mul %od2, 8
	%od8 =l mul %od, 8
	%osrc =l add %stack, %od28
	%odst =l add %stack, %od8
	%ov =l loadl %osrc
	storel %ov, %odst
	%odn =l add %od, 1
	storel %odn, %dpp
	%opc =l add %pc, 1
	storel %opc, %pcp
	jmp @loop
@lt
	%ld =l loadl %dpp
	%ld1 =l sub %ld, 1
	%ld2 =l sub %ld, 2
	%ld18 =l mul %ld1, 8
	%ld28 =l mul %ld2, 8
	%lbp =l add %stack, %ld18
	%lap =l add %stack, %ld28
	%lb =l loadl %lbp
	%la =l loadl %lap
	%lc =w csltl %la, %lb
	%lr =l extuw %lc
	storel %lr, %lap
	storel %ld1, %dpp
	%lpc =l add %pc, 1
	storel %lpc, %pcp
	jmp @loop
@jz
	%jd =l loadl %dpp
	%jd1 =l sub %jd, 1
	storel %jd1, %dpp
	%jd18 =l mul %jd1, 8
	%jvp =l add %stack, %jd18
	%jv =l loadl %jvp
	%jpc1 =l add %pc, 1
	%jpc18 =l mul %jpc1, 8
	%jtp =l add %code, %jpc18
	%jt =l loadl %jtp
	%jzero =w ceql %jv, 0
	jnz %jzero, @jz1, @jz0
@jz1
	storel %jt, %pcp
	jmp @loop
@jz0
	%jpc2 =l add %pc, 2
	storel %jpc2, %pcp
	jmp @loop
@jmp
	%gpc1 =l add %pc, 1
	%gpc8 =l mul %gpc1, 8
	%gtp =l add %code, %gpc8
	%gt =l loadl %gtp
	storel %gt, %pcp
	jmp @loop
@other
	%epc =l add %pc, 1
	storel %epc, %pcp
	jmp @loop
@finish
	%fd =l loadl %dpp
	%fempty =w ceql %fd, 0
	jnz %fempty, @ret0, @rettop
@ret0
	jmp @zero
@rettop
	%rd1 =l sub %fd, 1
	%rd8 =l mul %rd1, 8
	%rp =l add %stack, %rd8
	%rtop =l loadl %rp
	jmp @zero
@zero
	%result =l phi @ret0 %zero, @rettop %rtop
	jmp @zloop
@zloop
	%zi =l phi @zero %fd, @zstep %zi1
	%zmore =w cultl %zi, %cap
	jnz %zmore, @zstep, @ret
@zstep
	%zi8 =l mul %zi, 8
	%zp =l add %stack, %zi8
	storel %zero, %zp
	%zi1 =l add %zi, 1
	jmp @zloop
@ret
	ret %result
}
