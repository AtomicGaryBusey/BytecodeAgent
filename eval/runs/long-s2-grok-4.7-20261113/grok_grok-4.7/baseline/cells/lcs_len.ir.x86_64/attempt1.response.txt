export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zinit =l copy 0
	jmp @zloop
@zloop
	%zi =l phi @start %zinit, @zbody %zi1
	%zpast =w csltl %n, %zi
	jnz %zpast, @zdone, @zbody
@zbody
	%zoff =l shl %zi, 3
	%zp =l add %row, %zoff
	storel %zinit, %zp
	%zi1 =l add %zi, 1
	jmp @zloop
@zdone
	%iinit =l copy 0
	jmp @oloop
@oloop
	%i =l phi @zdone %iinit, @oinc %i1
	%imore =w csltl %i, %m
	jnz %imore, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%jinit =l copy 1
	%dinit =l copy 0
	jmp @iloop
@iloop
	%j =l phi @obody %jinit, @istore %j1
	%diag =l phi @obody %dinit, @istore %diag1
	%jpast =w csltl %n, %j
	jnz %jpast, @oinc, @ibody
@ibody
	%joff =l shl %j, 3
	%jp =l add %row, %joff
	%up =l loadl %jp
	%jm1 =l sub %j, 1
	%loff =l shl %jm1, 3
	%lp =l add %row, %loff
	%left =l loadl %lp
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%nv_m =l add %diag, 1
	%up_m =l copy %up
	jmp @istore
@nomatch
	%gt =w csltl %left, %up
	jnz %gt, @takeup, @takeleft
@takeup
	%nv_u =l copy %up
	%up_u =l copy %up
	jmp @istore
@takeleft
	%nv_l =l copy %left
	%up_l =l copy %up
	jmp @istore
@istore
	%nv =l phi @match %nv_m, @takeup %nv_u, @takeleft %nv_l
	%diag1 =l phi @match %up_m, @takeup %up_u, @takeleft %up_l
	storel %nv, %jp
	%j1 =l add %j, 1
	jmp @iloop
@oinc
	%i1 =l add %i, 1
	jmp @oloop
@done
	%noff =l shl %n, 3
	%np =l add %row, %noff
	%res =l loadl %np
	ret %res
}
