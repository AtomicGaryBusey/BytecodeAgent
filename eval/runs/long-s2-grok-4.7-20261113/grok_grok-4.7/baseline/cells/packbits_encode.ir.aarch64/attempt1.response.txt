export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i0 =l copy 0
	%o0 =l copy 0
	jmp @outer
@outer
	%i =l phi @start %i0, @write_rep %i_rep, @lit_done %i_lit
	%o =l phi @start %o0, @write_rep %o_rep, @lit_done %o_lit
	%atend =w csgel %i, %n
	jnz %atend, @ret_ok, @run_init
@ret_ok
	ret %o
@ret_fail
	%neg1 =l copy -1
	ret %neg1
@run_init
	%ip =l add %src, %i
	%bv =w loadub %ip
	%k0 =l copy %i
	%c0 =w copy 0
	jmp @runloop
@runloop
	%k =l phi @run_init %k0, @run_cont %k1
	%c =w phi @run_init %c0, @run_cont %c1
	%kdone =w csgel %k, %n
	jnz %kdone, @run_end, @run_cap
@run_cap
	%cfull =w csgew %c, 128
	jnz %cfull, @run_end, @run_byte
@run_byte
	%kp =l add %src, %k
	%kb =w loadub %kp
	%eqb =w ceqw %kb, %bv
	jnz %eqb, @run_cont, @run_end
@run_cont
	%k1 =l add %k, 1
	%c1 =w add %c, 1
	jmp @runloop
@run_end
	%isrep =w csgew %c, 3
	jnz %isrep, @do_rep, @do_lit
@do_rep
	%needr =l add %o, 2
	%badrep =w csgtl %needr, %cap
	jnz %badrep, @ret_fail, @write_rep
@write_rep
	%t257 =w copy 257
	%hdr =w sub %t257, %c
	%op =l add %out, %o
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %bv, %op1
	%cl =l extuw %c
	%i_rep =l add %i, %cl
	%o_rep =l add %o, 2
	jmp @outer
@do_lit
	%j0 =l copy %i
	%len0 =w copy 0
	jmp @lit_scan
@lit_scan
	%j =l phi @do_lit %j0, @lit_adv %j1
	%len =w phi @do_lit %len0, @lit_adv %len1
	%jdone =w csgel %j, %n
	jnz %jdone, @lit_emit, @lit_lenchk
@lit_lenchk
	%lenfull =w csgew %len, 128
	jnz %lenfull, @lit_emit, @lrun_init
@lrun_init
	%jp =l add %src, %j
	%jb =w loadub %jp
	%lk0 =l copy %j
	%lc0 =w copy 0
	jmp @lrun
@lrun
	%lk =l phi @lrun_init %lk0, @lrun_cont %lk1
	%lc =w phi @lrun_init %lc0, @lrun_cont %lc1
	%lkdone =w csgel %lk, %n
	jnz %lkdone, @lrun_end, @lrun_cap
@lrun_cap
	%lcfull =w csgew %lc, 128
	jnz %lcfull, @lrun_end, @lrun_byte
@lrun_byte
	%lkp =l add %src, %lk
	%lkb =w loadub %lkp
	%leq =w ceqw %lkb, %jb
	jnz %leq, @lrun_cont, @lrun_end
@lrun_cont
	%lk1 =l add %lk, 1
	%lc1 =w add %lc, 1
	jmp @lrun
@lrun_end
	%longrun =w csgew %lc, 3
	jnz %longrun, @lit_emit, @lit_adv
@lit_adv
	%j1 =l add %j, 1
	%len1 =w add %len, 1
	jmp @lit_scan
@lit_emit
	%Ll =l extuw %len
	%psz =l add %Ll, 1
	%needl =l add %o, %psz
	%badlit =w csgtl %needl, %cap
	jnz %badlit, @ret_fail, @lit_write
@lit_write
	%onew =w copy 1
	%lhdr =w sub %len, %onew
	%lop =l add %out, %o
	storeb %lhdr, %lop
	%s0 =w copy 0
	jmp @copyloop
@copyloop
	%s =w phi @lit_write %s0, @copy_body %s1
	%scmp =w csgew %s, %len
	jnz %scmp, @lit_done, @copy_body
@copy_body
	%sl =l extuw %s
	%si =l add %i, %sl
	%sp =l add %src, %si
	%sb =w loadub %sp
	%soff =l add %sl, 1
	%doff =l add %o, %soff
	%dp =l add %out, %doff
	storeb %sb, %dp
	%s1 =w add %s, 1
	jmp @copyloop
@lit_done
	%i_lit =l copy %j
	%o_lit =l copy %needl
	jmp @outer
}
