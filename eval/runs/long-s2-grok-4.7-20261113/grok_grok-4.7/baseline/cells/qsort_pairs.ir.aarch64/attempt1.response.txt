export function l $qsort_pairs(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%sixteen =l copy 16
	%eight =l copy 8
	%le1 =w cslel %n, %one
	jnz %le1, @ret0, @part
@ret0
	ret %zero
@part
	%nm1 =l sub %n, %one
	%poff =l mul %nm1, %sixteen
	%ppk =l add %a, %poff
	%ppv =l add %ppk, %eight
	%pkey =l loadl %ppk
	%pval =l loadl %ppv
	jmp @loop
@loop
	%j =l phi @part %zero, @next %j1
	%i =l phi @part %zero, @next %i1
	%cont =w csltl %j, %nm1
	jnz %cont, @cmp, @place
@cmp
	%joff =l mul %j, %sixteen
	%jpk =l add %a, %joff
	%jpv =l add %jpk, %eight
	%jkey =l loadl %jpk
	%jval =l loadl %jpv
	%klt =w csltl %jkey, %pkey
	jnz %klt, @take, @kge
@kge
	%kgt =w csgtl %jkey, %pkey
	jnz %kgt, @skip, @vcmp
@vcmp
	%vle =w cslel %jval, %pval
	jnz %vle, @take, @skip
@take
	%ioff =l mul %i, %sixteen
	%ipk =l add %a, %ioff
	%ipv =l add %ipk, %eight
	%ikey =l loadl %ipk
	%ival =l loadl %ipv
	storel %jkey, %ipk
	storel %jval, %ipv
	storel %ikey, %jpk
	storel %ival, %jpv
	%i2 =l add %i, %one
	jmp @next
@skip
	%i3 =l copy %i
	jmp @next
@next
	%i1 =l phi @take %i2, @skip %i3
	%j1 =l add %j, %one
	jmp @loop
@place
	%ioff2 =l mul %i, %sixteen
	%ipk2 =l add %a, %ioff2
	%ipv2 =l add %ipk2, %eight
	%ikey2 =l loadl %ipk2
	%ival2 =l loadl %ipv2
	storel %pkey, %ipk2
	storel %pval, %ipv2
	storel %ikey2, %ppk
	storel %ival2, %ppv
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, %one
	%rn =l sub %n, %ip1
	%roff =l mul %ip1, %sixteen
	%rp =l add %a, %roff
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	ret %zero
}
