export function l $qsort_pairs(l %a, l %n) {
@start
    %small =w csltl %n, 2
    jnz %small, @done, @dosort
@done
    ret 0
@dosort
    %nm1 =l sub %n, 1
    %pivbase =l mul %nm1, 16
    %pivaddr =l add %a, %pivbase
    %pkey =l loadl %pivaddr
    %pvadd =l add %pivaddr, 8
    %pval =l loadl %pvadd
    jmp @loop
@loop
    %j =l phi @dosort 0, @nextj %jnext
    %i =l phi @dosort 0, @nextj %inext
    %cont =w csltl %j, %nm1
    jnz %cont, @body, @afterloop
@body
    %jbase =l mul %j, 16
    %jaddr =l add %a, %jbase
    %jkey =l loadl %jaddr
    %jvadd =l add %jaddr, 8
    %jval =l loadl %jvadd
    %klt =w csltl %jkey, %pkey
    jnz %klt, @issmall, @checkeq
@checkeq
    %keq =w ceql %jkey, %pkey
    jnz %keq, @cmpval, @notsmall
@cmpval
    %vlt =w csltl %jval, %pval
    jnz %vlt, @issmall, @notsmall
@issmall
    %ibase =l mul %i, 16
    %iaddr =l add %a, %ibase
    %ikey =l loadl %iaddr
    %ivadd =l add %iaddr, 8
    %ival =l loadl %ivadd
    storel %jkey, %iaddr
    storel %jval, %ivadd
    storel %ikey, %jaddr
    storel %ival, %jvadd
    %inew =l add %i, 1
    jmp @nextj
@notsmall
    %isame =l copy %i
    jmp @nextj
@nextj
    %inext =l phi @issmall %inew, @notsmall %isame
    %jnext =l add %j, 1
    jmp @loop
@afterloop
    %ibase2 =l mul %i, 16
    %iaddr2 =l add %a, %ibase2
    %ikey2 =l loadl %iaddr2
    %ivadd2 =l add %iaddr2, 8
    %ival2 =l loadl %ivadd2
    storel %pkey, %iaddr2
    storel %pval, %ivadd2
    storel %ikey2, %pivaddr
    storel %ival2, %pvadd
    %r1 =l call $qsort_pairs(l %a, l %i)
    %ip1 =l add %i, 1
    %roff =l mul %ip1, 16
    %rptr =l add %a, %roff
    %nmi =l sub %n, %i
    %rcount =l sub %nmi, 1
    %r2 =l call $qsort_pairs(l %rptr, l %rcount)
    ret 0
}
