export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%si =l phi @start %src, @advance %si_n
	%rem =l phi @start %n, @advance %rem_n
	%oi =l phi @start %out, @advance %oi_n
	%rc =l phi @start %cap, @advance %rc_n
	%wr =l phi @start 0, @advance %wr_n
	%empty =w ceql %rem, 0
	jnz %empty, @done, @hdr
@done
	ret %wr
@hdr
	%h =w loadsb %si
	%si1 =l add %si, 1
	%rem1 =l sub %rem, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @donop, @notnop
@donop
	jmp @advance
@notnop
	%isneg =w csltw %h, 0
	jnz %isneg, @repeat, @literal
@literal
	%lcntw =w add %h, 1
	%lcnt =l extuw %lcntw
	%lsok =w culel %lcnt, %rem1
	jnz %lsok, @lit_chk, @err2
@lit_chk
	%lfits =w culel %lcnt, %rc
	jnz %lfits, @lcopy, @pcopy
@lcopy
	%li =l phi @lit_chk 0, @lstep %li2
	%ls =l phi @lit_chk %si1, @lstep %ls2
	%ld =l phi @lit_chk %oi, @lstep %ld2
	%ldone =w ceql %li, %lcnt
	jnz %ldone, @litdone, @lstep
@lstep
	%lb =w loadub %ls
	storeb %lb, %ld
	%ls2 =l add %ls, 1
	%ld2 =l add %ld, 1
	%li2 =l add %li, 1
	jmp @lcopy
@litdone
	%si_lit =l add %si1, %lcnt
	%rem_lit =l sub %rem1, %lcnt
	%oi_lit =l add %oi, %lcnt
	%rc_lit =l sub %rc, %lcnt
	%wr_lit =l add %wr, %lcnt
	jmp @advance
@pcopy
	%pi =l phi @lit_chk 0, @pstep %pi2
	%ps =l phi @lit_chk %si1, @pstep %ps2
	%pd =l phi @lit_chk %oi, @pstep %pd2
	%pdone =w ceql %pi, %rc
	jnz %pdone, @err1, @pstep
@pstep
	%pb =w loadub %ps
	storeb %pb, %pd
	%ps2 =l add %ps, 1
	%pd2 =l add %pd, 1
	%pi2 =l add %pi, 1
	jmp @pcopy
@repeat
	%rshort =w ceql %rem1, 0
	jnz %rshort, @err2, @rload
@rload
	%rb =w loadub %si1
	%rcntw =w sub 1, %h
	%rcnt =l extuw %rcntw
	%rfits =w culel %rcnt, %rc
	jnz %rfits, @rcopy, @rpcopy
@rcopy
	%ri =l phi @rload 0, @rstep %ri2
	%rd =l phi @rload %oi, @rstep %rd2
	%rdone =w ceql %ri, %rcnt
	jnz %rdone, @repdone, @rstep
@rstep
	storeb %rb, %rd
	%rd2 =l add %rd, 1
	%ri2 =l add %ri, 1
	jmp @rcopy
@repdone
	%si_rep =l add %si1, 1
	%rem_rep =l sub %rem1, 1
	%oi_rep =l add %oi, %rcnt
	%rc_rep =l sub %rc, %rcnt
	%wr_rep =l add %wr, %rcnt
	jmp @advance
@rpcopy
	%rpi =l phi @rload 0, @rpstep %rpi2
	%rpd =l phi @rload %oi, @rpstep %rpd2
	%rpdone =w ceql %rpi, %rc
	jnz %rpdone, @err1, @rpstep
@rpstep
	storeb %rb, %rpd
	%rpd2 =l add %rpd, 1
	%rpi2 =l add %rpi, 1
	jmp @rpcopy
@advance
	%si_n =l phi @donop %si1, @litdone %si_lit, @repdone %si_rep
	%rem_n =l phi @donop %rem1, @litdone %rem_lit, @repdone %rem_rep
	%oi_n =l phi @donop %oi, @litdone %oi_lit, @repdone %oi_rep
	%rc_n =l phi @donop %rc, @litdone %rc_lit, @repdone %rc_rep
	%wr_n =l phi @donop %wr, @litdone %wr_lit, @repdone %wr_rep
	jmp @loop
@err1
	%em1 =l copy -1
	ret %em1
@err2
	%em2 =l copy -2
	ret %em2
}
