export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
	jmp @zero
@zero
	%j =l phi @start %j0, @zbody %jn
	%zdone =w csgtl %j, %n
	jnz %zdone, @iinit, @zbody
@zbody
	%zoff =l mul %j, 8
	%zp =l add %row, %zoff
	storel 0, %zp
	%jn =l add %j, 1
	jmp @zero
@iinit
	%i0 =l copy 0
	jmp @iloop
@iloop
	%i =l phi @iinit %i0, @jdone %inext
	%idone =w csgel %i, %m
	jnz %idone, @done, @ibody
@ibody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%diag0 =l copy 0
	%j1 =l copy 1
	jmp @jloop
@jloop
	%j2 =l phi @ibody %j1, @store %j2n
	%diag =l phi @ibody %diag0, @store %diagout
	%jdonew =w csgtl %j2, %n
	jnz %jdonew, @jdone, @jbody
@jbody
	%jm1 =l sub %j2, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xi, %yb
	%joff =l mul %j2, 8
	%rp =l add %row, %joff
	%up =l loadl %rp
	%loff =l mul %jm1, 8
	%lp =l add %row, %loff
	%left =l loadl %lp
	%d1 =l add %diag, 1
	%gt =w csgtl %up, %left
	jnz %gt, @takeup, @takeleft
@takeup
	%maxup =l copy %up
	jmp @amax
@takeleft
	%maxleft =l copy %left
	jmp @amax
@amax
	%mv =l phi @takeup %maxup, @takeleft %maxleft
	jnz %eq, @takeeq, @takenot
@takeeq
	%veq =l copy %d1
	jmp @store
@takenot
	%vne =l copy %mv
	jmp @store
@store
	%newv =l phi @takeeq %veq, @takenot %vne
	storel %newv, %rp
	%diagout =l copy %up
	%j2n =l add %j2, 1
	jmp @jloop
@jdone
	%inext =l add %i, 1
	jmp @iloop
@done
	%noff =l mul %n, 8
	%np =l add %row, %noff
	%res =l loadl %np
	ret %res
}
