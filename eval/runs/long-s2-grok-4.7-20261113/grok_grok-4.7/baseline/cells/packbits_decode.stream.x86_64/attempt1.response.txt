packbits_decode:
    lea rsi, [rdi + rsi]
    mov r9, rcx
.Lloop:
    cmp rdi, rsi
    jae .Ldone
    movzx eax, byte ptr [rdi]
    inc rdi
    cmp al, 128
    je .Lloop
    ja .Lrep
    inc eax
    mov r10d, eax
    mov r11, rsi
    sub r11, rdi
    cmp r10, r11
    ja .Lsrcerr
    cmp r10, rcx
    ja .Llit_over
    sub rcx, r10
.Llit_copy:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec r10
    jnz .Llit_copy
    jmp .Lloop
.Llit_over:
    test rcx, rcx
    jz .Lcaperr
.Llit_over_lp:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec rcx
    jnz .Llit_over_lp
    jmp .Lcaperr
.Lrep:
    mov r10d, 257
    sub r10d, eax
    cmp rdi, rsi
    jae .Lsrcerr
    movzx eax, byte ptr [rdi]
    inc rdi
    cmp r10, rcx
    ja .Lrep_over
    sub rcx, r10
.Lrep_copy:
    mov byte ptr [rdx], al
    inc rdx
    dec r10
    jnz .Lrep_copy
    jmp .Lloop
.Lrep_over:
    test rcx, rcx
    jz .Lcaperr
.Lrep_over_lp:
    mov byte ptr [rdx], al
    inc rdx
    dec rcx
    jnz .Lrep_over_lp
    jmp .Lcaperr
.Ldone:
    mov rax, r9
    sub rax, rcx
    ret
.Lsrcerr:
    mov rax, -2
    ret
.Lcaperr:
    mov rax, -1
    ret
