export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc0 =l copy 0
	%sp0 =l copy 0
	jmp @loop
@loop
	%pc =l phi @start %pc0, @continue %pcn
	%sp =l phi @start %sp0, @continue %spn
	%atend =w ceql %pc, %len
	jnz %atend, @halt, @fetch
@fetch
	%pcoff =l mul %pc, 8
	%pcaddr =l add %code, %pcoff
	%op =l loadl %pcaddr
	%is0 =w ceql %op, 0
	jnz %is0, @halt, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @c2
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @c3
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @c4
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @c5
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @c6
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @c7
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @c8
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @c9
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @c10
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @c11
@c11
	%is11 =w ceql %op, 11
	jnz %is11, @dojmp, @def
@push
	%pc1p =l add %pc, 1
	%offp =l mul %pc1p, 8
	%addrp =l add %code, %offp
	%vp =l loadl %addrp
	%offsp =l mul %sp, 8
	%addrsp =l add %stack, %offsp
	storel %vp, %addrsp
	%spush =l add %sp, 1
	%pcpush =l add %pc, 2
	jmp @continue
@add
	%sp1a =l sub %sp, 1
	%sp2a =l sub %sp, 2
	%offba =l mul %sp1a, 8
	%addrba =l add %stack, %offba
	%ba =l loadl %addrba
	%offaa =l mul %sp2a, 8
	%addraa =l add %stack, %offaa
	%aa =l loadl %addraa
	%suma =l add %aa, %ba
	storel %suma, %addraa
	%pca =l add %pc, 1
	jmp @continue
@sub
	%sp1s =l sub %sp, 1
	%sp2s =l sub %sp, 2
	%offbs =l mul %sp1s, 8
	%addrbs =l add %stack, %offbs
	%bs =l loadl %addrbs
	%offas =l mul %sp2s, 8
	%addras =l add %stack, %offas
	%as =l loadl %addras
	%diffs =l sub %as, %bs
	storel %diffs, %addras
	%pcs =l add %pc, 1
	jmp @continue
@mul
	%sp1m =l sub %sp, 1
	%sp2m =l sub %sp, 2
	%offbm =l mul %sp1m, 8
	%addrbm =l add %stack, %offbm
	%bm =l loadl %addrbm
	%offam =l mul %sp2m, 8
	%addram =l add %stack, %offam
	%am =l loadl %addram
	%prodm =l mul %am, %bm
	storel %prodm, %addram
	%pcm =l add %pc, 1
	jmp @continue
@dup
	%sp1d =l sub %sp, 1
	%offd =l mul %sp1d, 8
	%addrd =l add %stack, %offd
	%vd =l loadl %addrd
	%offnd =l mul %sp, 8
	%addrnd =l add %stack, %offnd
	storel %vd, %addrnd
	%spdup =l add %sp, 1
	%pcdup =l add %pc, 1
	jmp @continue
@swap
	%sp1w =l sub %sp, 1
	%sp2w =l sub %sp, 2
	%offbw =l mul %sp1w, 8
	%addrbw =l add %stack, %offbw
	%bw =l loadl %addrbw
	%offaw =l mul %sp2w, 8
	%addraw =l add %stack, %offaw
	%aw =l loadl %addraw
	storel %bw, %addraw
	storel %aw, %addrbw
	%pcw =l add %pc, 1
	%spw =l copy %sp
	jmp @continue
@drop
	%spdrop =l sub %sp, 1
	%pcdrop =l add %pc, 1
	jmp @continue
@over
	%sp2o =l sub %sp, 2
	%offo =l mul %sp2o, 8
	%addro =l add %stack, %offo
	%vo =l loadl %addro
	%offno =l mul %sp, 8
	%addrno =l add %stack, %offno
	storel %vo, %addrno
	%spover =l add %sp, 1
	%pcover =l add %pc, 1
	jmp @continue
@lt
	%sp1l =l sub %sp, 1
	%sp2l =l sub %sp, 2
	%offbl =l mul %sp1l, 8
	%addrbl =l add %stack, %offbl
	%bl =l loadl %addrbl
	%offal =l mul %sp2l, 8
	%addral =l add %stack, %offal
	%al =l loadl %addral
	%cmpl =w csltl %al, %bl
	%resl =l extsw %cmpl
	storel %resl, %addral
	%pcl =l add %pc, 1
	jmp @continue
@jz
	%sp1j =l sub %sp, 1
	%offj =l mul %sp1j, 8
	%addrj =l add %stack, %offj
	%vj =l loadl %addrj
	%pc1j =l add %pc, 1
	%offt =l mul %pc1j, 8
	%addrt =l add %code, %offt
	%tj =l loadl %addrt
	%isz =w ceql %vj, 0
	jnz %isz, @jz_taken, @jz_not
@jz_taken
	%pcjt =l copy %tj
	%spjt =l copy %sp1j
	jmp @continue
@jz_not
	%pcjz =l add %pc, 2
	%spjn =l copy %sp1j
	jmp @continue
@dojmp
	%pc1g =l add %pc, 1
	%offg =l mul %pc1g, 8
	%addrg =l add %code, %offg
	%tg =l loadl %addrg
	%spg =l copy %sp
	jmp @continue
@def
	%pcdef =l add %pc, 1
	%spdef =l copy %sp
	jmp @continue
@continue
	%pcn =l phi @push %pcpush, @add %pca, @sub %pcs, @mul %pcm, @dup %pcdup, @swap %pcw, @drop %pcdrop, @over %pcover, @lt %pcl, @jz_taken %pcjt, @jz_not %pcjz, @dojmp %tg, @def %pcdef
	%spn =l phi @push %spush, @add %sp1a, @sub %sp1s, @mul %sp1m, @dup %spdup, @swap %spw, @drop %spdrop, @over %spover, @lt %sp1l, @jz_taken %spjt, @jz_not %spjn, @dojmp %spg, @def %spdef
	jmp @loop
@halt
	%zconst =l copy 0
	%iinit =l copy %sp
	jmp @zloop
@zloop
	%i =l phi @halt %iinit, @zstep %inext
	%zeq =w ceql %i, %cap
	jnz %zeq, @zdone, @zstep
@zstep
	%ioff =l mul %i, 8
	%iaddr =l add %stack, %ioff
	storel %zconst, %iaddr
	%inext =l add %i, 1
	jmp @zloop
@zdone
	%isempty =w ceql %sp, 0
	jnz %isempty, @empty, @top
@empty
	%r0 =l copy 0
	ret %r0
@top
	%last =l sub %sp, 1
	%lastoff =l mul %last, 8
	%lastaddr =l add %stack, %lastoff
	%rv =l loadl %lastaddr
	ret %rv
}
