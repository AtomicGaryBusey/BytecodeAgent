packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lmain:
    cmp rbx, r13
    jge .Ldone
    movzx r8d, byte ptr [r12 + rbx]
    lea r9, [rbx + 128]
    cmp r9, r13
    cmova r9, r13
    lea rsi, [rbx + 1]
.Lrun1:
    cmp rsi, r9
    jge .Lrun1_done
    movzx edx, byte ptr [r12 + rsi]
    cmp edx, r8d
    jne .Lrun1_done
    inc rsi
    jmp .Lrun1
.Lrun1_done:
    mov rax, rsi
    sub rax, rbx
    cmp rax, 3
    jl .Lstart_lit
    lea rcx, [rbp + 2]
    cmp rcx, r15
    jg .Lfail
    mov edx, 257
    sub edx, eax
    mov byte ptr [r14 + rbp], dl
    mov byte ptr [r14 + rbp + 1], r8b
    add rbp, 2
    add rbx, rax
    jmp .Lmain
.Lstart_lit:
    mov r11, rbx
.Llit_scan:
    cmp r11, r13
    jge .Llit_emit
    mov rax, r11
    sub rax, rbx
    cmp rax, 128
    jge .Llit_emit
    movzx r8d, byte ptr [r12 + r11]
    lea r9, [r11 + 128]
    cmp r9, r13
    cmova r9, r13
    lea rsi, [r11 + 1]
.Lrun2:
    cmp rsi, r9
    jge .Lrun2_done
    movzx edx, byte ptr [r12 + rsi]
    cmp edx, r8d
    jne .Lrun2_done
    inc rsi
    jmp .Lrun2
.Lrun2_done:
    mov rax, rsi
    sub rax, r11
    cmp rax, 3
    jge .Llit_emit
    inc r11
    jmp .Llit_scan
.Llit_emit:
    mov rax, r11
    sub rax, rbx
    mov r8, rbp
    add r8, rax
    inc r8
    cmp r8, r15
    jg .Lfail
    mov edx, eax
    dec edx
    mov byte ptr [r14 + rbp], dl
    inc rbp
    lea rsi, [r12 + rbx]
    lea rdi, [r14 + rbp]
    xor ecx, ecx
.Lcopy:
    cmp rcx, rax
    jge .Lcopy_done
    movzx edx, byte ptr [rsi + rcx]
    mov byte ptr [rdi + rcx], dl
    inc rcx
    jmp .Lcopy
.Lcopy_done:
    add rbp, rax
    mov rbx, r11
    jmp .Lmain
.Ldone:
    mov rax, rbp
    jmp .Lret
.Lfail:
    mov rax, -1
.Lret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
