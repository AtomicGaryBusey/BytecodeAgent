export function l $base64_encode(l %src, l %n, l %out) {
@start
	%nfull =l udiv %n, 3
	%rem =l urem %n, 3
	%neg75 =w sub 0, 75
	%neg15 =w sub 0, 15
	jmp @gloop
@gloop
	%gi =l phi @start 0, @ccont %gi1
	%lt =w csltl %gi, %nfull
	jnz %lt, @dofull, @chkrem
@dofull
	%off =l mul %gi, 3
	%p0 =l add %src, %off
	%p1 =l add %p0, 1
	%p2 =l add %p0, 2
	%b0 =w loadub %p0
	%b1 =w loadub %p1
	%b2 =w loadub %p2
	%b0s =w shl %b0, 16
	%b1s =w shl %b1, 8
	%ab =w or %b0s, %b1s
	%valf =w or %ab, %b2
	%outf =l mul %gi, 4
	jmp @cloop
@chkrem
	%nz =w cnel %rem, 0
	jnz %nz, @dopart, @done
@dopart
	%poff =l mul %nfull, 3
	%pp0 =l add %src, %poff
	%pb0 =w loadub %pp0
	%istwo =w ceql %rem, 2
	jnz %istwo, @part2, @part1
@part1
	%val1 =w shl %pb0, 16
	%real1 =w copy 2
	jmp @partjoin
@part2
	%pp1 =l add %pp0, 1
	%pb1 =w loadub %pp1
	%pb0s =w shl %pb0, 16
	%pb1s =w shl %pb1, 8
	%val2 =w or %pb0s, %pb1s
	%real2 =w copy 3
	jmp @partjoin
@partjoin
	%valp =w phi @part1 %val1, @part2 %val2
	%realp =w phi @part1 %real1, @part2 %real2
	%outp =l mul %nfull, 4
	jmp @cloop
@cloop
	%k =w phi @dofull 0, @partjoin 0, @cnext %k1
	%val =w phi @dofull %valf, @partjoin %valp, @cnext %valn
	%real =w phi @dofull 4, @partjoin %realp, @cnext %realn
	%obase =l phi @dofull %outf, @partjoin %outp, @cnext %obasen
	%mode =w phi @dofull 1, @partjoin 0, @cnext %moden
	%klt =w csltw %k, 4
	jnz %klt, @cdobody, @cend
@cdobody
	%kltreal =w csltw %k, %real
	jnz %kltreal, @cenc, @cpad
@cpad
	jmp @cstore
@cenc
	%k6 =w mul %k, 6
	%sh =w sub 18, %k6
	%shifted =w shr %val, %sh
	%sext =w and %shifted, 63
	%d25 =w sub 25, %sext
	%g25 =w sar %d25, 31
	%m6 =w and %g25, 6
	%d51 =w sub 51, %sext
	%g51 =w sar %d51, 31
	%m75 =w and %g51, %neg75
	%d61 =w sub 61, %sext
	%g61 =w sar %d61, 31
	%m15 =w and %g61, %neg15
	%d62 =w sub 62, %sext
	%g62 =w sar %d62, 31
	%m3 =w and %g62, 3
	%cA =w add %sext, 65
	%cB =w add %cA, %m6
	%cC =w add %cB, %m75
	%cD =w add %cC, %m15
	%chenc =w add %cD, %m3
	jmp @cstore
@cstore
	%ch =w phi @cpad 61, @cenc %chenc
	%kl =l extuw %k
	%koff =l add %obase, %kl
	%dp =l add %out, %koff
	storeb %ch, %dp
	jmp @cnext
@cnext
	%k1 =w add %k, 1
	%valn =w copy %val
	%realn =w copy %real
	%obasen =l copy %obase
	%moden =w copy %mode
	jmp @cloop
@cend
	%cont =w ceqw %mode, 1
	jnz %cont, @ccont, @done
@ccont
	%gi1 =l add %gi, 1
	jmp @gloop
@done
	%n2 =l add %n, 2
	%q =l udiv %n2, 3
	%rv =l mul %q, 4
	ret %rv
}
