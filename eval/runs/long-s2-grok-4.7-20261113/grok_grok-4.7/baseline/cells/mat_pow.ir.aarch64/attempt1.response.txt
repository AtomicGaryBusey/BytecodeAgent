export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	jmp @id_i
@id_i
	%ii =l phi @start 0, @id_body %ii1
	%iddone =w ceql %ii, %nn
	jnz %iddone, @pow_i, @id_body
@id_body
	%row =l udiv %ii, %n
	%col =l urem %ii, %n
	%isd =w ceql %row, %col
	%iv =l extuw %isd
	%ioff =l shl %ii, 3
	%iaddr =l add %out, %ioff
	storel %iv, %iaddr
	%ii1 =l add %ii, 1
	jmp @id_i
@pow_i
	%pc =l phi @id_i 0, @after_copy %pc1
	%pdone =w ceql %pc, %k
	jnz %pdone, @zero_i, @m_i
@m_i
	%i =l phi @pow_i 0, @m_i_inc %i1
	%idone =w ceql %i, %n
	jnz %idone, @copy_i, @m_j
@m_i_inc
	%i1 =l add %i, 1
	jmp @m_i
@m_j
	%j =l phi @m_i 0, @m_store %j1
	%jdone =w ceql %j, %n
	jnz %jdone, @m_i_inc, @m_t
@m_t
	%t =l phi @m_j 0, @m_acc %t1
	%sum =l phi @m_j 0, @m_acc %sum1
	%tdone =w ceql %t, %n
	jnz %tdone, @m_store, @m_acc
@m_acc
	%in =l mul %i, %n
	%xi =l add %in, %t
	%xoff =l shl %xi, 3
	%xaddr =l add %out, %xoff
	%xv =l loadl %xaddr
	%tn =l mul %t, %n
	%yi =l add %tn, %j
	%yoff =l shl %yi, 3
	%yaddr =l add %a, %yoff
	%yv =l loadl %yaddr
	%prod =l mul %xv, %yv
	%sum1 =l add %sum, %prod
	%t1 =l add %t, 1
	jmp @m_t
@m_store
	%zn =l mul %i, %n
	%zi =l add %zn, %j
	%zoff =l shl %zi, 3
	%zaddr =l add %tmp, %zoff
	storel %sum, %zaddr
	%j1 =l add %j, 1
	jmp @m_j
@copy_i
	%ci =l phi @m_i 0, @copy_b %ci1
	%cdone =w ceql %ci, %nn
	jnz %cdone, @after_copy, @copy_b
@copy_b
	%coff =l shl %ci, 3
	%cs =l add %tmp, %coff
	%cd =l add %out, %coff
	%cv =l loadl %cs
	storel %cv, %cd
	%ci1 =l add %ci, 1
	jmp @copy_i
@after_copy
	%pc1 =l add %pc, 1
	jmp @pow_i
@zero_i
	%zx =l phi @pow_i 0, @zero_b %zx1
	%zdone =w ceql %zx, %nn2
	jnz %zdone, @done, @zero_b
@zero_b
	%zboff =l shl %zx, 3
	%zbaddr =l add %tmp, %zboff
	storel 0, %zbaddr
	%zx1 =l add %zx, 1
	jmp @zero_i
@done
	ret 0
}
