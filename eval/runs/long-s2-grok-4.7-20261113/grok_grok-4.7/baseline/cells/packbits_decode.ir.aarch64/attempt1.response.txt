export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%si_init =l copy 0
	%oi_init =l copy 0
	jmp @hdr
@hdr
	%si =l phi @entry %si_init, @nop %si_nop, @lit_done %si_lit, @rep_done %si_rep
	%oi =l phi @entry %oi_init, @nop %oi_nop, @lit_done %oi_lit, @rep_done %oi_rep
	%has =w csltl %si, %n
	jnz %has, @read_hdr, @done
@done
	ret %oi
@read_hdr
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @check_lit
@nop
	%si_nop =l copy %si1
	%oi_nop =l copy %oi
	jmp @hdr
@check_lit
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%hl =l extsw %h
	%lcount =l add %hl, 1
	%lrem =l sub %n, %si1
	%lok =w cslel %lcount, %lrem
	jnz %lok, @lit_prep, @err_src
@lit_prep
	%loi_in =l copy %oi
	%lsi_in =l copy %si1
	jmp @lit_loop
@lit_loop
	%li =l phi @lit_prep 0, @lit_write %li_next
	%loi =l phi @lit_prep %loi_in, @lit_write %loi_next
	%lsi =l phi @lit_prep %lsi_in, @lit_write %lsi_next
	%lmore =w csltl %li, %lcount
	jnz %lmore, @lit_byte, @lit_done
@lit_done
	%si_lit =l copy %lsi
	%oi_lit =l copy %loi
	jmp @hdr
@lit_byte
	%lroom =w csltl %loi, %cap
	jnz %lroom, @lit_write, @err_cap
@lit_write
	%lsp =l add %src, %lsi
	%lb =w loadub %lsp
	%ldp =l add %out, %loi
	storeb %lb, %ldp
	%lsi_next =l add %lsi, 1
	%loi_next =l add %loi, 1
	%li_next =l add %li, 1
	jmp @lit_loop
@repeat
	%rok =w csltl %si1, %n
	jnz %rok, @rep_load, @err_src
@rep_load
	%rsp =l add %src, %si1
	%rb =w loadub %rsp
	%si_after =l add %si1, 1
	%hr =l extsw %h
	%rcount =l sub 1, %hr
	%roi_in =l copy %oi
	jmp @rep_loop
@rep_loop
	%ri =l phi @rep_load 0, @rep_write %ri_next
	%roi =l phi @rep_load %roi_in, @rep_write %roi_next
	%rmore =w csltl %ri, %rcount
	jnz %rmore, @rep_byte, @rep_done
@rep_done
	%si_rep =l copy %si_after
	%oi_rep =l copy %roi
	jmp @hdr
@rep_byte
	%rroom =w csltl %roi, %cap
	jnz %rroom, @rep_write, @err_cap
@rep_write
	%rdp =l add %out, %roi
	storeb %rb, %rdp
	%roi_next =l add %roi, 1
	%ri_next =l add %ri, 1
	jmp @rep_loop
@err_src
	%err2 =l copy -2
	ret %err2
@err_cap
	%err1 =l copy -1
	ret %err1
}
