qsort_pairs:
cmp rsi, 1
jg .Lqs_work
xor eax, eax
ret
.Lqs_work:
push rbx
push r12
push r15
mov rbx, rdi
mov r12, rsi
mov rax, r12
shr rax, 1
shl rax, 4
mov rcx, r12
dec rcx
shl rcx, 4
mov rdx, qword ptr [rbx + rax]
mov r8, qword ptr [rbx + rcx]
mov qword ptr [rbx + rax], r8
mov qword ptr [rbx + rcx], rdx
mov rdx, qword ptr [rbx + rax + 8]
mov r8, qword ptr [rbx + rcx + 8]
mov qword ptr [rbx + rax + 8], r8
mov qword ptr [rbx + rcx + 8], rdx
mov r8, qword ptr [rbx + rcx]
mov r9, qword ptr [rbx + rcx + 8]
xor r15, r15
xor rax, rax
.Lqs_loop:
mov rdx, r12
dec rdx
cmp rax, rdx
jge .Lqs_pdone
mov rcx, rax
shl rcx, 4
mov rdx, qword ptr [rbx + rcx]
cmp rdx, r8
jl .Lqs_le
jg .Lqs_gt
mov rdx, qword ptr [rbx + rcx + 8]
cmp rdx, r9
jg .Lqs_gt
.Lqs_le:
mov rdx, r15
shl rdx, 4
cmp rdx, rcx
je .Lqs_inc_i
mov r10, qword ptr [rbx + rdx]
mov r11, qword ptr [rbx + rcx]
mov qword ptr [rbx + rdx], r11
mov qword ptr [rbx + rcx], r10
mov r10, qword ptr [rbx + rdx + 8]
mov r11, qword ptr [rbx + rcx + 8]
mov qword ptr [rbx + rdx + 8], r11
mov qword ptr [rbx + rcx + 8], r10
.Lqs_inc_i:
inc r15
.Lqs_gt:
inc rax
jmp .Lqs_loop
.Lqs_pdone:
mov rax, r12
dec rax
shl rax, 4
mov rcx, r15
shl rcx, 4
mov r10, qword ptr [rbx + rcx]
mov r11, qword ptr [rbx + rax]
mov qword ptr [rbx + rcx], r11
mov qword ptr [rbx + rax], r10
mov r10, qword ptr [rbx + rcx + 8]
mov r11, qword ptr [rbx + rax + 8]
mov qword ptr [rbx + rcx + 8], r11
mov qword ptr [rbx + rax + 8], r10
cmp r15, 1
jle .Lqs_noleft
mov rdi, rbx
mov rsi, r15
call qsort_pairs
.Lqs_noleft:
mov rax, r15
inc rax
mov rcx, r12
sub rcx, rax
cmp rcx, 1
jle .Lqs_done
shl rax, 4
lea rdi, [rbx + rax]
mov rsi, rcx
call qsort_pairs
.Lqs_done:
xor eax, eax
pop r15
pop r12
pop rbx
ret
