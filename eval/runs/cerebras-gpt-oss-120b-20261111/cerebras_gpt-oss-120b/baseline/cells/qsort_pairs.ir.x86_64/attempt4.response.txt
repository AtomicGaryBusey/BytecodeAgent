export function l $qsort_pairs(l %a, l %n) {
@entry:
    %cond_small =w csltl %n, l 2
    jnz %cond_small, @ret

    %pivot_idx =l sdiv %n, l 2
    %pivot_off =l mul %pivot_idx, l 16
    %pivot_ptr =l add %a, %pivot_off
    %pk =l loadl %pivot_ptr
    %pv =l loadl %pivot_ptr, l 8

    %n_minus1 =l sub %n, l 1
    jmp @loop

@loop:
    %i =l phi l -1, %i_next
    %j =l phi l 0, %j_next
    %cond_j =w csltl %j, %n_minus1
    jz %cond_j, @after_loop

    %off_j =l mul %j, l 16
    %ptr_j =l add %a, %off_j
    %key_j =l loadl %ptr_j
    %val_j =l loadl %ptr_j, l 8

    %key_lt =w csltl %key_j, %pk
    %key_eq =w ceql %key_j, %pk
    %val_lt =w csltl %val_j, %pv
    %val_eq =w ceql %val_j, %pv
    %val_le =w orw %val_lt, %val_eq
    %cond_eq =w andw %key_eq, %val_le
    %cond =w orw %key_lt, %cond_eq
    jz %cond, @no_swap

    %i1 =l add %i, l 1
    %off_i1 =l mul %i1, l 16
    %ptr_i1 =l add %a, %off_i1
    %key_i1 =l loadl %ptr_i1
    %val_i1 =l loadl %ptr_i1, l 8
    storel %key_j, %ptr_i1
    storel %val_j, %ptr_i1, l 8
    storel %key_i1, %ptr_j
    storel %val_i1, %ptr_j, l 8
    %i_next =l %i1
    jmp @inc_j

@no_swap:
    %i_next =l %i
    jmp @inc_j

@inc_j:
    %j_next =l add %j, l 1
    jmp @loop

@after_loop:
    %i1 =l add %i, l 1
    %off_i1 =l mul %i1, l 16
    %ptr_i1 =l add %a, %off_i1
    %key_i1 =l loadl %ptr_i1
    %val_i1 =l loadl %ptr_i1, l 8
    storel %pk, %ptr_i1
    storel %pv, %ptr_i1, l 8
    storel %key_i1, %pivot_ptr
    storel %val_i1, %pivot_ptr, l 8

    %left_n =l %i1
    %right_n_tmp =l sub %n, %i1
    %right_n =l sub %right_n_tmp, l 1

    %cond_left =w csltl l 1, %left_n
    jnz %cond_left, @call_left
    jmp @maybe_right

@call_left:
    %r1 =l call $qsort_pairs(l %a, l %left_n)
    jmp @maybe_right

@maybe_right:
    %cond_right =w csltl l 1, %right_n
    jz %cond_right, @ret
    %right_ptr =l add %ptr_i1, l 16
    %r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    jmp @ret

@ret:
    ret l 0
}
