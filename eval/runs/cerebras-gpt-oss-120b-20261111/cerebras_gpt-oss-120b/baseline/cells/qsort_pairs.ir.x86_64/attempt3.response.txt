export function l $qsort_pairs(l %a, l %n) {
@entry:
    %cond_small =w csltl %n, l 2
    jnz %cond_small, @ret

    %half =l sdiv %n, l 2
    %pivot_off =l mul %half, l 16
    %pivot_ptr =l add %a, %pivot_off
    %pk =l loadl %pivot_ptr
    %pv =l loadl %pivot_ptr, l 8

    %i0 =l 0
    %j0 =l sub %n, l 1
    jmp @partition(l %i0, l %j0)

@partition(l %i, l %j):
    %off_i =l mul %i, l 16
    %ptr_i =l add %a, %off_i
    %key_i =l loadl %ptr_i
    %val_i =l loadl %ptr_i, l 8
    %c1_i =w csltl %key_i, %pk
    %c2_i =w ceql %key_i, %pk
    %c3_i =w csltl %val_i, %pv
    %c4_i =w andw %c2_i, %c3_i
    %cond_i =w orw %c1_i, %c4_i
    %i_next =l add %i, l 1
    jnz %cond_i, @partition(l %i_next, l %j)

    %off_j =l mul %j, l 16
    %ptr_j =l add %a, %off_j
    %key_j =l loadl %ptr_j
    %val_j =l loadl %ptr_j, l 8
    %c1_j =w csltl %pk, %key_j
    %c2_j =w ceql %pk, %key_j
    %c3_j =w csltl %pv, %val_j
    %c4_j =w andw %c2_j, %c3_j
    %cond_j =w orw %c1_j, %c4_j
    %j_next =l sub %j, l 1
    jnz %cond_j, @partition(l %i, l %j_next)

    %j1 =l add %j, l 1
    %cmp =w csltl %i, %j1
    jz %cmp, @after_partition(l %i, l %j)

    %off_i2 =l mul %i, l 16
    %ptr_i2 =l add %a, %off_i2
    %off_j2 =l mul %j, l 16
    %ptr_j2 =l add %a, %off_j2
    %key_i2 =l loadl %ptr_i2
    %val_i2 =l loadl %ptr_i2, l 8
    %key_j2 =l loadl %ptr_j2
    %val_j2 =l loadl %ptr_j2, l 8
    storel %key_i2, %ptr_j2
    storel %val_i2, %ptr_j2, l 8
    storel %key_j2, %ptr_i2
    storel %val_j2, %ptr_i2, l 8
    %i_next2 =l add %i, l 1
    %j_next2 =l sub %j, l 1
    jmp @partition(l %i_next2, l %j_next2)

@after_partition(l %i, l %j):
    %left_n =l add %j, l 1
    %right_n =l sub %n, %i
    %off_i =l mul %i, l 16
    %right_ptr =l add %a, %off_i
    %cond_left =w csltl l 0, %left_n
    jnz %cond_left, @call_left(l %left_n, l %right_n, l %right_ptr)
    jmp @maybe_right(l %right_n, l %right_ptr)

@call_left(l %left_n, l %right_n, l %right_ptr):
    %r1 =l call $qsort_pairs(l %a, l %left_n)
    jmp @maybe_right(l %right_n, l %right_ptr)

@maybe_right(l %right_n, l %right_ptr):
    %cond_right =w csltl l 0, %right_n
    jnz %cond_right, @call_right(l %right_ptr, l %right_n)
    jmp @ret

@call_right(l %right_ptr, l %right_n):
    %r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    jmp @ret

@ret:
    ret l 0
}
