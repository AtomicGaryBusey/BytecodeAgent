export function l $qsort_pairs(l %a, l %n) {
@entry:
    %small =w csltl l 0, %n
    %one_or_more =w csltl %n, l 2
    jnz %one_or_more, @ret

    %half =l sdiv %n, l 2
    %pivot_off =l mul %half, l 16
    %pivot_ptr =l add %a, %pivot_off
    %pk =l loadl %pivot_ptr
    %pv =l loadl %pivot_ptr, l 8

    %j0 =l sub %n, l 1
    jmp @loop(l 0, %j0)

@loop(l %i, l %j) {
    %off_i =l mul %i, l 16
    %ptr_i =l add %a, %off_i
    %key_i =l loadl %ptr_i
    %val_i =l loadl %ptr_i, l 8
    %c1_i =w csltl %key_i, %pk
    %c2_i =w ceql %key_i, %pk
    %c3_i =w csltl %val_i, %pv
    %c4_i =w andw %c2_i, %c3_i
    %cond_i =w orw %c1_i, %c4_i
    %i_inc =l add %i, l 1
    jnz %cond_i, @loop(%i_inc, %j)

    %off_j =l mul %j, l 16
    %ptr_j =l add %a, %off_j
    %key_j =l loadl %ptr_j
    %val_j =l loadl %ptr_j, l 8
    %c1_j =w csltl %pk, %key_j
    %c2_j =w ceql %pk, %key_j
    %c3_j =w csltl %pv, %val_j
    %c4_j =w andw %c2_j, %c3_j
    %cond_j =w orw %c1_j, %c4_j
    %j_dec =l sub %j, l 1
    jnz %cond_j, @loop(%i, %j_dec)

    %j1 =l add %j, l 1
    %cond_ij =w csltl %i, %j1
    jz %cond_ij, @after_partition

    %key_i2 =l loadl %ptr_i
    %val_i2 =l loadl %ptr_i, l 8
    %key_j2 =l loadl %ptr_j
    %val_j2 =l loadl %ptr_j, l 8
    storel %key_i2, %ptr_j
    storel %val_i2, %ptr_j, l 8
    storel %key_j2, %ptr_i
    storel %val_j2, %ptr_i, l 8

    %i_next =l add %i, l 1
    %j_next =l sub %j, l 1
    jmp @loop(%i_next, %j_next)
}

@after_partition:
    %left_n =l add %j, l 1
    %right_n =l sub %n, %i
    %tmp_off =l mul %i, l 16
    %right_ptr =l add %a, %tmp_off
    %left_cond =w csltl l 0, %left_n
    jnz %left_cond, @call_left
    jmp @maybe_right

@call_left:
    %r1 =l call $qsort_pairs(l %a, l %left_n)
    jmp @maybe_right

@maybe_right:
    %right_cond =w csltl l 0, %right_n
    jnz %right_cond, @call_right
    jmp @ret

@call_right:
    %r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    jmp @ret

@ret:
    ret l 0
}
