mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8                     # local slot for total count
    mov r9, rsi                     # r9 = n
    mov r10, rdx                    # r10 = k (exponent counter)
    # compute total = n*n
    mov rax, r9
    imul rax, r9
    mov rbx, rax                    # rbx = total
    mov [rsp], rbx                  # store total on stack
    # copy matrix a into base (tmp)
    mov rsi, rdi                    # rsi = a (source)
    mov rdi, r8                     # rdi = tmp (dest)
    mov rcx, rbx
    xor rax, rax
.Lcopy_a:
    mov rdx, qword ptr [rsi + rax*8]
    mov qword ptr [rdi + rax*8], rdx
    inc rax
    cmp rax, rcx
    jne .Lcopy_a
    # initialize out = identity
    mov rcx, r9                     # rcx = n
    mov rdx, rbx                    # rdx = total
    xor rax, rax
.Lzero_out:
    mov qword ptr [r8 + rax*8 + rbx*8], 0   # out starts after tmp (tmp + total)
    inc r
