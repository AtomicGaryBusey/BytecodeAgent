mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r9, r8                     # r9 = base pointer (tmp)
    mov rbx, rsi
    imul rbx, rsi                  # rbx = total = n*n
    lea r10, [r8 + rbx*8]          # r10 = workspace pointer (tmp + total)
    # initialize out = identity
    xor r12, r12                   # i = 0
.Linit_i:
    mov r14, r12
    imul r14, rsi                  # r14 = i*n
    xor r13, r13                   # j = 0
.Linit_j:
    mov rax, r14
    add rax, r13                   # index = i*n + j
    lea rdx, [rcx + rax*8]         # dest = out + index*8
    cmp r12, r13
    jne .Lstore0
    mov qword ptr [rdx], 1
    jmp .Linc_j
.Lstore0:
    mov qword ptr [rdx], 0
.Linc_j:
    inc r13
    cmp r13, rsi
    jl .Linit_j
    inc r12
    cmp r12, rsi
    jl .Linit_i
    # exponent loop
    mov r11, rdx                   # r11 = k (exponent)
.Lexp_loop:
    test r11, r11
    jz .Lexp_done
    test r11, 1
    jz .Lskip_mul_res
    # result = result * base -> workspace
    mov rdi, rcx                  # X = result (out)
    mov rdx, r9                   # Y = base
    mov r8, r10                   # Z = workspace
    call .Lmultiply
    # copy workspace to result
    xor r12, r12
.Lcopy_res:
    mov rax, qword ptr [r10 + r12*8]
    mov qword ptr [rcx + r12*8], rax
    inc r12
    cmp r12, rbx
    jl .Lcopy_res
.Lskip_mul_res:
    # base = base * base -> workspace
    mov rdi, r9                  # X = base
    mov rdx, r9                  # Y = base
    mov r8, r10                  # Z = workspace
    call .Lmultiply
    # copy workspace to base
    xor r12, r12
.Lcopy_base:
    mov rax, qword ptr [r10 + r12*8]
    mov qword ptr [r9 + r12*8], rax
    inc r12
    cmp r12, rbx
    jl .Lcopy_base
    shr r11, 1
    jmp .Lexp_loop
.Lexp_done:
    # zero tmp and workspace
    xor r12, r12
.Lzero_tmp:
    mov qword ptr [r9 + r12*8], 0
    mov qword ptr [r10 + r12*8], 0
    inc r12
    cmp r12, rbx
    jl .Lzero_tmp
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lmultiply:
    # multiply matrix X (rdi) by Y (rdx) into Z (r8), n in rsi, total in rbx
    xor r12, r12                   # i = 0
.Lmul_i:
    xor r13, r13                   # j = 0
.Lmul_j:
    xor r14, r14                   # t = 0
    xor r15, r15                   # sum = 0
.Lmul_t:
    mov rax, r12
    imul rax, rsi                  # i*n
    add rax, r14                  # i*n + t
    lea rax, [rdi + rax*8]         # &X[i*n + t]
    mov rax, qword ptr [rax]
    mov rbx, r14
    imul rbx, rsi                  # t*n
    add rbx, r13                  # t*n + j
    lea rbx, [rdx + rbx*8]         # &Y[t*n + j]
    mov rbx, qword ptr [rbx]
    imul rax, rbx                 # a * b (low 64 bits)
    add r15, rax                  # sum += product
    inc r14
    cmp r14, rsi
    jl .Lmul_t
    mov rax, r12
    imul rax, rsi                  # i*n
    add rax, r13                  # i*n + j
    lea rax, [r8 + rax*8]          # &Z[i*n + j]
    mov qword ptr [rax], r15
    inc r13
    cmp r13, rsi
    jl .Lmul_j
    inc r12
    cmp r12, rsi
    jl .Lmul_i
    ret
