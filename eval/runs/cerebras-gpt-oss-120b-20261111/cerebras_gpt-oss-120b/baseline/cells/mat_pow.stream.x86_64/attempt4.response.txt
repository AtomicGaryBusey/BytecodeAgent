mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rbp
    mov rbx, rsi
    imul rbx, rsi                # rbx = total = n*n
    lea r9, [r8 + rbx*8]         # r9 = buffer1 (tmp + total)

    # initialize out = identity
    xor r12, r12                 # i = 0
.Linit_i:
    xor r13, r13                 # j = 0
.Linit_j:
    mov rax, r12
    imul rax, rsi                # i*n
    add rax, r13                 # index = i*n + j
    lea rdx, [rcx + rax*8]       # dest = out + index*8
    cmp r12, r13
    jne .Lset0
    mov qword ptr [rdx], 1
    jmp .Linc_j
.Lset0:
    mov qword ptr [rdx], 0
.Linc_j:
    inc r13
    cmp r13, rsi
    jl .Linit_j
    inc r12
    cmp r12, rsi
    jl .Linit_i

    # copy a into buffer0 (tmp)
    xor r12, r12
.Lcopy_a:
    mov rax, qword ptr [rdi + r12*8]
    mov qword ptr [r8 + r12*8], rax
    inc r12
    cmp r12, rbx
    jl .Lcopy_a

    mov r14, rdx                # r14 = exponent k

.Lexp_loop:
    test r14, r14
    jz .Lexp_done
    test r14, 1
    jz .Lskip_mul_res

    # result = result * base -> buffer1
    push r14
    mov rdi, rcx                # X = out
    mov rdx, r8                 # Y = base
    mov r8, r9                  # Z = buffer1
    call .Lmultiply
    pop r14

    # copy buffer1 to out
    xor r12, r12
.Lcopy_res:
    mov rax, qword ptr [r9 + r12*8]
    mov qword ptr [rcx + r12*8], rax
    inc r12
    cmp r12, rbx
    jl .Lcopy_res

.Lskip_mul_res:
    # base = base * base -> buffer1
    push r14
    mov rdi, r8                 # X = base
    mov rdx, r8                 # Y = base
    mov r8, r9                  # Z = buffer1
    call .Lmultiply
    pop r14

    # copy buffer1 to base
    xor r12, r12
.Lcopy_base:
    mov rax, qword ptr [r9 + r12*8]
    mov qword ptr [r8 + r12*8], rax
    inc r12
    cmp r12, rbx
    jl .Lcopy_base

    shr r14, 1
    jmp .Lexp_loop

.Lexp_done:
    # zero temporary buffers
    xor r12, r12
.Lzero_tmp:
    mov qword ptr [r8 + r12*8], 0
    mov qword ptr [r9 + r12*8], 0
    inc r12
    cmp r12, rbx
    jl .Lzero_tmp

    xor rax, rax
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

.Lmultiply:
    xor r12, r12                # i = 0
.Lmul_i:
    xor r13, r13                # j = 0
.Lmul_j:
    xor r15, r15                # sum = 0
    xor r14, r14                # t = 0
.Lmul_t:
    mov rax, r12
    imul rax, rsi               # i*n
    add rax, r14                # i*n + t
    lea rax, [rdi + rax*8]      # &X[i][t]
    mov rax, qword ptr [rax]

    mov rbx, r14
    imul rbx, rsi               # t*n
    add rbx, r13                # t*n + j
    lea rbx, [rdx + rbx*8]      # &Y[t][j]
    mov rbx, qword ptr [rbx]

    imul rax, rbx               # product (low 64)
    add r15, rax                # sum += product

    inc r14
    cmp r14, rsi
    jl .Lmul_t

    mov rax, r12
    imul rax, rsi               # i*n
    add rax, r13                # i*n + j
    lea rax, [r8 + rax*8]       # &Z[i][j]
    mov qword ptr [rax], r15

    inc r13
    cmp r13, rsi
    jl .Lmul_j

    inc r12
    cmp r12, rsi
    jl .Lmul_i
    ret
