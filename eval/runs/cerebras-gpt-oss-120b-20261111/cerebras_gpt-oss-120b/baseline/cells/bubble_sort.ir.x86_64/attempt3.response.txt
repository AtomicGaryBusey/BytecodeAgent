export function l $bubble_sort(l %a, l %n) {
@entry:
    %zero = const 0
    %one = const 1
    %eight = const 8
    %i0 = const 0
    jmp @outer %i0

@outer(%i):
    %n1 = subl %n, %one
    %c_i = csltl %i, %n1
    jnz %c_i, @inner %i %zero, @ret

@inner(%i, %j):
    %n1 = subl %n, %one
    %limit = subl %n1, %i
    %c_j = csltl %j, %limit
    jnz %c_j, @compare %i %j, @outer_inc %i

@compare(%i, %j):
    %j8 = mull %j, %eight
    %addr = addl %a, %j8
    %addr_next = addl %addr, %eight
    %v1 = loadl %addr
    %v2 = loadl %addr_next
    %c = csltl %v2, %v1
    jnz %c, @swap %i %j, @inner_inc %i %j

@swap(%i, %j):
    %j8 = mull %j, %eight
    %addr = addl %a, %j8
    %addr_next = addl %addr, %eight
    %v1 = loadl %addr
    %v2 = loadl %addr_next
    storel %v2, %addr
    storel %v1, %addr_next
    jmp @inner_inc %i %j

@inner_inc(%i, %j):
    %j1 = addl %j, %one
    jmp @inner %i %j1

@outer_inc(%i):
    %i1 = addl %i, %one
    jmp @outer %i1

@ret:
    ret l %zero
}
