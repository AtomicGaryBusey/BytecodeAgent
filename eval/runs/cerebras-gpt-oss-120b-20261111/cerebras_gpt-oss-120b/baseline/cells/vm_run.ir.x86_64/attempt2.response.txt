export function l vm_run(l %code, l %len, l %stack, l %cap) {
@start:
    %pc = imm 0
    %sp = imm 0
    jmp @loop

@loop:
    %cond_end = csltl %pc, %len
    jz %cond_end, @exit
    %pc8 = mull %pc, imm 8
    %op_addr = addl %pc8, %code
    %op = loadl %op_addr
    %is_halt = ceql %op, imm 0
    jnz %is_halt, @exit
    %is_push = ceql %op, imm 1
    jnz %is_push, @push
    %is_add = ceql %op, imm 2
    jnz %is_add, @add
    %is_sub = ceql %op, imm 3
    jnz %is_sub, @sub
    %is_mul = ceql %op, imm 4
    jnz %is_mul, @mul
    %is_dup = ceql %op, imm 5
    jnz %is_dup, @dup
    %is_swap = ceql %op, imm 6
    jnz %is_swap, @swap
    %is_drop = ceql %op, imm 7
    jnz %is_drop, @drop
    %is_over = ceql %op, imm 8
    jnz %is_over, @over
    %is_lt = ceql %op, imm 9
    jnz %is_lt, @lt
    %is_jz = ceql %op, imm 10
    jnz %is_jz, @jz
    %is_jmp = ceql %op, imm 11
    jnz %is_jmp, @jmp
    %pc = addl %pc, imm 1
    jmp @loop

@push:
    %pc1 = addl %pc, imm 1
    %pc1_8 = mull %pc1, imm 8
    %opaddr = addl %pc1_8, %code
    %val = loadl %opaddr
    %sp8 = mull %sp, imm 8
    %stkaddr = addl %sp8, %stack
    storel %val, %stkaddr
    %sp = addl %sp, imm 1
    %pc = addl %pc, imm 2
    jmp @loop

@add:
    %sp1 = subl %sp, imm 1
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp1_8 = mull %sp1, imm 8
    %addr_b = addl %sp1_8, %stack
    %b = loadl %addr_b
    %res = addl %a, %b
    storel %res, %addr_a
    %sp = subl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@sub:
    %sp1 = subl %sp, imm 1
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp1_8 = mull %sp1, imm 8
    %addr_b = addl %sp1_8, %stack
    %b = loadl %addr_b
    %res = subl %a, %b
    storel %res, %addr_a
    %sp = subl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@mul:
    %sp1 = subl %sp, imm 1
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp1_8 = mull %sp1, imm 8
    %addr_b = addl %sp1_8, %stack
    %b = loadl %addr_b
    %res = mull %a, %b
    storel %res, %addr_a
    %sp = subl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@dup:
    %sp1 = subl %sp, imm 1
    %sp1_8 = mull %sp1, imm 8
    %addr_top = addl %sp1_8, %stack
    %val = loadl %addr_top
    %sp8 = mull %sp, imm 8
    %addr_new = addl %sp8, %stack
    storel %val, %addr_new
    %sp = addl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@swap:
    %sp1 = subl %sp, imm 1
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp1_8 = mull %sp1, imm 8
    %addr_b = addl %sp1_8, %stack
    %b = loadl %addr_b
    storel %b, %addr_a
    storel %a, %addr_b
    %pc = addl %pc, imm 1
    jmp @loop

@drop:
    %sp = subl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@over:
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp8 = mull %sp, imm 8
    %addr_new = addl %sp8, %stack
    storel %a, %addr_new
    %sp = addl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@lt:
    %sp1 = subl %sp, imm 1
    %sp2 = subl %sp, imm 2
    %sp2_8 = mull %sp2, imm 8
    %addr_a = addl %sp2_8, %stack
    %a = loadl %addr_a
    %sp1_8 = mull %sp1, imm 8
    %addr_b = addl %sp1_8, %stack
    %b = loadl %addr_b
    %condw = csltl %a, %b
    %cond = zextw %condw
    storel %cond, %addr_a
    %sp = subl %sp, imm 1
    %pc = addl %pc, imm 1
    jmp @loop

@jz:
    %pc1 = addl %pc, imm 1
    %pc1_8 = mull %pc1, imm 8
    %opaddr = addl %pc1_8, %code
    %target = loadl %opaddr
    %sp1 = subl %sp, imm 1
    %sp1_8 = mull %sp1, imm 8
    %addr_top = addl %sp1_8, %stack
    %val = loadl %addr_top
    %sp = %sp1
    %is_zero = ceql %val, imm 0
    jnz %is_zero, @jz_target
    %pc = addl %pc, imm 2
    jmp @loop
@jz_target:
    %pc = %target
    jmp @loop

@jmp:
    %pc1 = addl %pc, imm 1
    %pc1_8 = mull %pc1, imm 8
    %opaddr = addl %pc1_8, %code
    %target = loadl %opaddr
    %pc = %target
    jmp @loop

@exit:
    %i = %sp
@zero_loop:
    %condz = csltl %i, %cap
    jz %condz, @after_zero
    %i8 = mull %i, imm 8
    %addr = addl %i8, %stack
    storel 0, %addr
    %i = addl %i, imm 1
    jmp @zero_loop
@after_zero:
    %sp_eq0 = ceql %sp, imm 0
    jnz %sp_eq0, @ret_zero
    %sp1 = subl %sp, imm 1
    %sp1_8 = mull %sp1, imm 8
    %addr_top = addl %sp1_8, %stack
    %top = loadl %addr_top
    ret %top
@ret_zero:
    ret 0
}
