xxh64:
    // constants
    movz    x4, 0xCA87, lsl #0          // P1 low
    movk    x4, 0x85EB, lsl #16
    movk    x4, 0x79B1, lsl #32
    movk    x4, 0x9E37, lsl #48        // x4 = P1
    movz    x5, 0xEB4F, lsl #0          // P2 low
    movk    x5, 0xD4EB, lsl #16
    movk    x5, 0x27D4, lsl #32
    movk    x5, 0xC2B2, lsl #48        // x5 = P2
    movz    x6, 0x79F9, lsl #0          // P3 low
    movk    x6, 0x9E37, lsl #16
    movk    x6, 0x67B1, lsl #32
    movk    x6, 0x1656, lsl #48        // x6 = P3
    movz    x7, 0xAE63, lsl #0          // P4 low
    movk    x7, 0xC2B2, lsl #16
    movk    x7, 0xCA77, lsl #32
    movk    x7, 0x85EB, lsl #48        // x7 = P4
    movz    x8, 0x67C5, lsl #0          // P5 low
    movk    x8, 0x1656, lsl #16
    movk    x8, 0xEB2F, lsl #32
    movk    x8, 0x27D4, lsl #48        // x8 = P5

    // save original length
    mov     x9, x1                     // x9 = n_orig

    // decide if we have stripes
    cmp     x1, #32
    blt     .Lno_stripes

    // initialize accumulators
    add     x10, x2, x4                // v1 = seed + P1
    add     x10, x10, x5               // v1 = seed + P1 + P2
    add     x11, x2, x5                // v2 = seed + P2
    mov     x12, x2                    // v3 = seed
    sub     x13, x2, x4                // v4 = seed - P1

.Lstripe_loop:
    // load 4 * 8 bytes and update accumulators
    ldr     x14, [x0], #8
    mul     x15, x14, x5
    add     x15, x15, x10
    ror     x15, x15, #33
    mul     x10, x15, x4

    ldr     x14, [x0], #8
    mul     x15, x14, x5
    add     x15, x15, x11
    ror     x15, x15, #33
    mul     x11, x15, x4

    ldr     x14, [x0], #8
    mul     x15, x14, x5
    add     x15, x15, x12
    ror     x15, x15, #33
    mul     x12, x15, x4

    ldr     x14, [x0], #8
    mul     x15, x14, x5
    add     x15, x15, x13
    ror     x15, x15, #33
    mul     x13, x15, x4

    sub     x1, x1, #32
    cmp     x1, #32
    bge     .Lstripe_loop

    // combine accumulators into h
    ror     x14, x10, #63            // rotl(v1,1)
    ror     x15, x11, #57            // rotl(v2,7)
    ror     x16, x12, #52            // rotl(v3,12)
    ror     x17, x13, #46            // rotl(v4,18)
    add     x14, x14, x15
    add     x14, x14, x16
    add     x14, x14, x17            // x14 = h
    // merge with each accumulator
    // merge(v1)
    mul     x15, x10, x5
    ror     x15, x15, #33
    mul     x15, x15, x4
    eor     x14, x14, x15
    mul     x14, x14, x4
    add     x14, x14, x7
    // merge(v2)
    mul     x15, x11, x5
    ror     x15, x15, #33
    mul     x15, x15, x4
    eor     x14, x14, x15
    mul     x14, x14, x4
    add     x14, x14, x7
    // merge(v3)
    mul     x15, x12, x5
    ror     x15, x15, #33
    mul     x15, x15, x4
    eor     x14, x14, x15
    mul     x14, x14, x4
    add     x14, x14, x7
    // merge(v4)
    mul     x15, x13, x5
    ror     x15, x15, #33
    mul     x15, x15, x4
    eor     x14, x14, x15
    mul     x14, x14, x4
    add     x14, x14, x7
    b       .Lafter_stripes

.Lno_stripes:
    add     x14, x2, x8               // h = seed + P5

.Lafter_stripes:
    // add length
    add     x14, x14, x9

    // remaining bytes processing
.Lwhile8:
    cmp     x1, #8
    blt     .Lwhile4
    ldr     x15, [x0], #8
    mul     x15, x15, x5
    ror     x15, x15, #33
    mul     x15, x15, x4
    eor     x14, x14, x15
    ror     x14, x14, #37            // rotl 27
    mul     x14, x14, x4
    add     x14, x14, x7
    sub     x1, x1, #8
    b       .Lwhile8

.Lwhile4:
    cmp     x1, #4
    blt     .Lwhile1
    ldr     w15, [x0], #4
    mul     x15, x15, x4
    eor     x14, x14, x15
    ror     x14, x14, #41            // rotl 23
    mul     x14, x14, x5
    add     x14, x14, x6
    sub     x1, x1, #4
    b       .Lwhile4

.Lwhile1:
    cbz     x1, .Lfinal
.Lbyte_loop:
    ldrb    w15, [x0], #1
    mul     x15, x15, x8
    eor     x14, x14, x15
    ror     x14, x14, #53            // rotl 11
    mul     x14, x14, x4
    sub     x1, x1, #1
    cbnz    x1, .Lbyte_loop

.Lfinal:
    // avalanche
    mov     x15, x14
    lsr     x15, x15, #33
    eor     x14, x14, x15
    mul     x14, x14, x5
    mov     x15, x14
    lsr     x15, x15, #29
    eor     x14, x14, x15
    mul     x14, x14, x6
    mov     x15, x14
    lsr     x15, x15, #32
    eor     x14, x14, x15
    mov     x0, x14
    ret
