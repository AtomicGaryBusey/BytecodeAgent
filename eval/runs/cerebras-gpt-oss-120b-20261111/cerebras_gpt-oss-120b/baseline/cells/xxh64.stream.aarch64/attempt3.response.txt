xxh64:
    // constants
    movz    x19, 0xCA87, lsl #0          // P1 low
    movk    x19, 0x85EB, lsl #16
    movk    x19, 0x79B1, lsl #32
    movk    x19, 0x9E37, lsl #48        // x19 = P1
    movz    x20, 0xEB4F, lsl #0          // P2 low
    movk    x20, 0xD4EB, lsl #16
    movk    x20, 0x27D4, lsl #32
    movk    x20, 0xC2B2, lsl #48        // x20 = P2
    movz    x21, 0x79F9, lsl #0          // P3 low
    movk    x21, 0x9E37, lsl #16
    movk    x21, 0x67B1, lsl #32
    movk    x21, 0x1656, lsl #48        // x21 = P3
    movz    x22, 0xAE63, lsl #0          // P4 low
    movk    x22, 0xC2B2, lsl #16
    movk    x22, 0xCA77, lsl #32
    movk    x22, 0x85EB, lsl #48        // x22 = P4
    movz    x23, 0x67C5, lsl #0          // P5 low
    movk    x23, 0x1656, lsl #16
    movk    x23, 0xEB2F, lsl #32
    movk    x23, 0x27D4, lsl #48        // x23 = P5

    // save original length
    mov     x24, x1                     // x24 = n_orig

    // decide if we have stripes
    cmp     x1, #32
    blt     .Lno_stripes

    // initialize accumulators
    add     x9, x2, x19                // v1 = seed + P1
    add     x9, x9, x20                // v1 = seed + P1 + P2
    add     x10, x2, x20               // v2 = seed + P2
    mov     x11, x2                    // v3 = seed
    sub     x12, x2, x19               // v4 = seed - P1

.Lstripe_loop:
    // v1
    ldr     x13, [x0], #8
    mul     x14, x13, x20
    add     x14, x14, x9
    ror     x14, x14, #33
    mul     x9, x14, x19
    // v2
    ldr     x13, [x0], #8
    mul     x14, x13, x20
    add     x14, x14, x10
    ror     x14, x14, #33
    mul     x10, x14, x19
    // v3
    ldr     x13, [x0], #8
    mul     x14, x13, x20
    add     x14, x14, x11
    ror     x14, x14, #33
    mul     x11, x14, x19
    // v4
    ldr     x13, [x0], #8
    mul     x14, x13, x20
    add     x14, x14, x12
    ror     x14, x14, #33
    mul     x12, x14, x19

    sub     x1, x1, #32
    cmp     x1, #32
    bge     .Lstripe_loop

    // combine accumulators into h
    ror     x13, x9, #63               // rotl(v1,1)
    ror     x14, x10, #57              // rotl(v2,7)
    ror     x15, x11, #52              // rotl(v3,12)
    ror     x16, x12, #46              // rotl(v4,18)
    add     x13, x13, x14
    add     x13, x13, x15
    add     x13, x13, x16              // x13 = h

    // merge with v1
    mul     x14, x9, x20
    ror     x14, x14, #33
    mul     x14, x14, x19              // round(0,v1)
    eor     x13, x13, x14
    mul     x13, x13, x19
    add     x13, x13, x22

    // merge with v2
    mul     x14, x10, x20
    ror     x14, x14, #33
    mul     x14, x14, x19
    eor     x13, x13, x14
    mul     x13, x13, x19
    add     x13, x13, x22

    // merge with v3
    mul     x14, x11, x20
    ror     x14, x14, #33
    mul     x14, x14, x19
    eor     x13, x13, x14
    mul     x13, x13, x19
    add     x13, x13, x22

    // merge with v4
    mul     x14, x12, x20
    ror     x14, x14, #33
    mul     x14, x14, x19
    eor     x13, x13, x14
    mul     x13, x13, x19
    add     x13, x13, x22

    b       .Lafter_stripes

.Lno_stripes:
    add     x13, x2, x23               // h = seed + P5

.Lafter_stripes:
    // add length
    add     x13, x13, x24

.Lwhile8:
    cmp     x1, #8
    blt     .Lwhile4
    ldr     x14, [x0], #8
    mul     x15, x14, x20
    ror     x15, x15, #33
    mul     x15, x15, x19              // round(0,val)
    eor     x13, x13, x15
    ror     x13, x13, #37              // rotl 27
    mul     x13, x13, x19
    add     x13, x13, x22
    sub     x1, x1, #8
    b       .Lwhile8

.Lwhile4:
    cmp     x1, #4
    blt     .Lwhile1
    ldr     w14, [x0], #4
    mul     x15, x14, x19              // read32 * P1
    eor     x13, x13, x15
    ror     x13, x13, #41              // rotl 23
    mul     x13, x13, x20
    add     x13, x13, x21
    sub     x1, x1, #4
    b       .Lwhile4

.Lwhile1:
    cbz     x1, .Lfinal
.Lbyte_loop:
    ldrb    w14, [x0], #1
    mul     x15, x14, x23
    eor     x13, x13, x15
    ror     x13, x13, #53              // rotl 11
    mul     x13, x13, x19
    sub     x1, x1, #1
    cbnz    x1, .Lbyte_loop

.Lfinal:
    // avalanche
    mov     x14, x13
    lsr     x14, x14, #33
    eor     x13, x13, x14
    mul     x13, x13, x20
    mov     x14, x13
    lsr     x14, x14, #29
    eor     x13, x13, x14
    mul     x13, x13, x21
    mov     x14, x13
    lsr     x14, x14, #32
    eor     x13, x13, x14
    mov     x0, x13
    ret
