xxh64:
    // constants
    movz    x9,  #0xCA87, lsl #0          // P1 low
    movk    x9,  #0x85EB, lsl #16
    movk    x9,  #0x79B1, lsl #32
    movk    x9,  #0x9E37, lsl #48        // x9 = P1
    movz    x10, #0xEB4F, lsl #0          // P2 low
    movk    x10, #0xD4EB, lsl #16
    movk    x10, #0x27D4, lsl #32
    movk    x10, #0xC2B2, lsl #48        // x10 = P2
    movz    x11, #0xF9F9, lsl #0          // P3 low (0xF9F9?) actually 0xF9F9 from 0x165667B19E3779F9 low 16 bits = 0x79F9? Wait correct value:
    // P3 = 0x165667B19E3779F9
    movz    x11, #0x79F9, lsl #0
    movk    x11, #0x9E37, lsl #16
    movk    x11, #0x67B1, lsl #32
    movk    x11, #0x1656, lsl #48        // x11 = P3
    movz    x12, #0xAE63, lsl #0          // P4 low
    movk    x12, #0xC2B2, lsl #16
    movk    x12, #0xCA77, lsl #32
    movk    x12, #0x85EB, lsl #48        // x12 = P4
    movz    x13, #0x67C5, lsl #0          // P5 low
    movk    x13, #0x1656, lsl #16
    movk    x13, #0xEB2F, lsl #32
    movk    x13, #0x27D4, lsl #48        // x13 = P5

    mov     x5,  x0                      // keep pointer in x5
    mov     x14, x1                      // save original length (n_orig)

    cmp     x1, #32
    blt     .Lno_stripes

    // initialize accumulators
    add     x15, x2, x9                 // v1 = seed + P1
    add     x15, x15, x10               // v1 = seed + P1 + P2
    add     x16, x2, x10                // v2 = seed + P2
    mov     x17, x2                     // v3 = seed
    sub     x18, x2, x9                 // v4 = seed - P1

.Lstripe_loop:
    // v1
    ldr     x19, [x5], #8
    mul     x20, x19, x10
    add     x20, x20, x15
    ror     x20, x20, #33               // rotl 31
    mul     x15, x20, x9

    // v2
    ldr     x19, [x5], #8
    mul     x20, x19, x10
    add     x20, x20, x16
    ror     x20, x20, #33
    mul     x16, x20, x9

    // v3
    ldr     x19, [x5], #8
    mul     x20, x19, x10
    add     x20, x20, x17
    ror     x20, x20, #33
    mul     x17, x20, x9

    // v4
    ldr     x19, [x5], #8
    mul     x20, x19, x10
    add     x20, x20, x18
    ror     x20, x20, #33
    mul     x18, x20, x9

    sub     x1, x1, #32
    cmp     x1, #32
    bge     .Lstripe_loop

    // combine accumulators into h (x0)
    ror     x19, x15, #63               // rotl(v1,1)
    ror     x20, x16, #57               // rotl(v2,7)
    ror     x21, x17, #52               // rotl(v3,12)
    ror     x22, x18, #46               // rotl(v4,18)
    add     x0,  x19, x20
    add     x0,  x0,  x21
    add     x0,  x0,  x22               // h = sum

    // merge with v1
    mul     x19, x15, x10
    ror     x19, x19, #33
    mul     x19, x19, x9               // round0(v1)
    eor     x0,  x0,  x19
    mul     x0,  x0,  x9
    add     x0,  x0,  x12

    // merge with v2
    mul     x19, x16, x10
    ror     x19, x19, #33
    mul     x19, x19, x9
    eor     x0,  x0,  x19
    mul     x0,  x0,  x9
    add     x0,  x0,  x12

    // merge with v3
    mul     x19, x17, x10
    ror     x19, x19, #33
    mul     x19, x19, x9
    eor     x0,  x0,  x19
    mul     x0,  x0,  x9
    add     x0,  x0,  x12

    // merge with v4
    mul     x19, x18, x10
    ror     x19, x19, #33
    mul     x19, x19, x9
    eor     x0,  x0,  x19
    mul     x0,  x0,  x9
    add     x0,  x0,  x12

    b       .Lafter_stripes

.Lno_stripes:
    add     x0, x2, x13                 // h = seed + P5

.Lafter_stripes:
    // add length
    add     x0, x0, x14

.Lwhile8:
    cmp     x1, #8
    blt     .Lwhile4
    ldr     x19, [x5], #8
    mul     x20, x19, x10
    ror     x20, x20, #33
    mul     x20, x20, x9               // round0(q)
    eor     x0,  x0,  x20
    ror     x0,  x0,  #37              // rotl 27
    mul     x0,  x0,  x9
    add     x0,  x0,  x12
    sub     x1,  x1,  #8
    b       .Lwhile8

.Lwhile4:
    cmp     x1, #4
    blt     .Lwhile1
    ldr     w19, [x5], #4
    mul     x20, x19, x9               // read32 * P1
    eor     x0,  x0,  x20
    ror     x0,  x0,  #41              // rotl 23
    mul     x0,  x0,  x10
    add     x0,  x0,  x11
    sub     x1,  x1,  #4
    b       .Lwhile4

.Lwhile1:
    cbz     x1, .Lfinal
.Lbyte_loop:
    ldrb    w19, [x5], #1
    mul     x20, x19, x13              // c * P5
    eor     x0,  x0,  x20
    ror     x0,  x0,  #53              // rotl 11
    mul     x0,  x0,  x9
    sub     x1,  x1,  #1
    cbnz    x1, .Lbyte_loop

.Lfinal:
    // avalanche
    mov     x19, x0
    lsr     x19, x19, #33
    eor     x0,  x0,  x19
    mul     x0,  x0,  x10
    mov     x19, x0
    lsr     x19, x19, #29
    eor     x0,  x0,  x19
    mul     x0,  x0,  x11
    mov     x19, x0
    lsr     x19, x19, #32
    eor     x0,  x0,  x19
    ret
