isqrt:
    cbz x0, .Lzero
    mov x1, #1
    mov x2, #0xffffffff
    mov x5, #0
.Lloop:
    cmp x1, x2
    bgt .Ldone
    add x3, x1, x2
    lsr x3, x3, #1
    udiv x4, x0, x3
    cmp x3, x4
    ble .Lif
    sub x2, x3, #1
    b .Lloop
.Lif:
    mov x5, x3
    add x1, x3, #1
    b .Lloop
.Ldone:
    mov x0, x5
    ret
.Lzero:
    mov x0, #0
    ret
