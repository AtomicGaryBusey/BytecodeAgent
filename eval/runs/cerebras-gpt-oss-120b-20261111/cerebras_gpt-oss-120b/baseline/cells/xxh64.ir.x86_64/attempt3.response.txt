    %rot2 = shr %sum, $33
    %rotl = or %rot, %rot2

    %rot4b = shr %v4, $46
    %h = add %h, (or %rot4, %rot4b)

    // merge h with v1..v4
    // merge macro: h = (h ^ round(0,v)) * P1 + P4
    %zero = $0

    // merge v1
    %tmp = mul %v1, $-4417276706812531889
    %rot = shl %tmp, $31

    %rot = shl %tmp, $31

    %h = add %h, $-8796714831421723037

    %rot = shl %tmp, $31

    %r0 = mul %r0, $-7046029288634856825
    %h = xor %h, %r0
    %h = mul %h, $-7046029288634856825
    %h = add %h, $-8796714831421723037
    jmp @post_h_init

@no_stripes
    %h = add %seed, $2870177450012600261          // seed + P5
    %q = %p
    %len = %n

@post_h_init

@while8

    jnz %c8, @while4
    %val = loadq %q
    // round(0, val)
    %tmp = mul %val, $-4417276706812531889
    %rot = shl %tmp, $31
    %rot2 = shr %tmp, $33
    %r0 = or %rot, %rot2
    %r0 = mul %r0, $-7046029288634856825
    %h = xor %h, %r0

    %h = add %h, $-8796714831421723037
    %q = add %q, $8

    jmp @while8

@while4

    jnz %c4, @while1
    %val32 = loaduw %q

    jmp @while4

@while1

    jnz %c1, @finalize
    %c = loadb %q

    %c_mul = mul %c0, $2870177450012600261       // * P5

    %rothb = shr %h, $53
    %h = or %roth, %rothb
