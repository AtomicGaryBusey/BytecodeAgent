bmin_empty:
    mov rax, 0x7FFFFFFFFFFFFFFF
    test rsi, rsi
    je .Lret
    mov rcx, rsi
    mov rdx, qword ptr [rdi]
    mov rax, rdx
    dec rcx
    je .Lret
    add rdi, 8
.Lloop:
    mov rdx, qword ptr [rdi]
    cmp rdx, rax
    jl .Lupdate
    jmp .Lnext
.Lupdate:
    mov rax, rdx
.Lnext:
    add rdi, 8
    dec rcx
    jne .Lloop
.Lret:
    ret
