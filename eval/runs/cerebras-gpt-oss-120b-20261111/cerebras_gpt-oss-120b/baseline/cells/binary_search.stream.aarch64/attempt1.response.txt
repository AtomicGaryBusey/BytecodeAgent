binary_search:
    // x0 = a, x1 = n, x2 = key
    mov     x3, #0                     // low = 0
    sub     x4, x1, #1                 // high = n-1
.Lloop:
    cmp     x3, x4
    bgt     .Lnotfound
    add     x5, x3, x4                 // x5 = low + high
    asr     x5, x5, #1                 // mid = (low+high)/2
    ldr     x6, [x0, x5, LSL #3]       // value = a[mid]
    cmp     x6, x2
    beq     .Lfound
    bgt     .Lhigh
    // value < key
    add     x3, x5, #1                 // low = mid + 1
    b       .Lloop
.Lhigh:
    sub     x4, x5, #1                 // high = mid - 1
    b       .Lloop
.Lfound:
    mov     x0, x5                     // return index
    ret
.Lnotfound:
    mov     x0, #-1                    // return -1
    ret
