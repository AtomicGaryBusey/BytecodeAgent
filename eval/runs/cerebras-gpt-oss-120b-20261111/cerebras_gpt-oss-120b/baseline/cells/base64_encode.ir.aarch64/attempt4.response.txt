export function l $base64_encode(l %src, l %n, l %dst) {
@entry
    %full3 =l udiv %n, 3
    %k =l 0
    jmp @loop_cond

@loop_cond
    %c_loop =w csltl %k, %full3
    jnz %c_loop @process3
    jmp @remainder

@process3
    %src_off =l mul %k, 3
    %dst_off =l mul %k, 4

    %p0 =l add %src, %src_off
    %b0 =l loadub %p0
    %p1 =l add %p0, 1
    %b1 =l loadub %p1
    %p2 =l add %p0, 2
    %b2 =l loadub %p2

    %v0 =l shl %b0, 16
    %v1 =l shl %b1, 8
    %v =l or %v0, %v1
    %v =l or %v, %b2

    %t0 =l shr %v, 18
    %idx0 =l and %t0, 63
    %t1 =l shr %v, 12
    %idx1 =l and %t1, 63
    %t2 =l shr %v, 6
    %idx2 =l and %t2, 63
    %idx3 =l and %v, 63

    %d0 =l add %dst, %dst_off
    %d1 =l add %d0, 1
    %d2 =l add %d0, 2
    %d3 =l add %d0, 3

    # map idx0
    %c =w csltl %idx0, 26
    jnz %c @map0_a
    %c =w csltl %idx0, 52
    jnz %c @map0_b
    %c =w csltl %idx0, 62
    jnz %c @map0_c
    %c =w ceql %idx0, 62
    jnz %c @map0_plus
    jmp @map0_slash
@map0_a
    %ch =l add %idx0, 65
    jmp @store0
@map0_b
    %ch =l add %idx0, 71
    jmp @store0
@map0_c
    %ch =l add %idx0, -4
    jmp @store0
@map0_plus
    %ch =l 43
    jmp @store0
@map0_slash
    %ch =l 47
@store0
    storeb %ch, %d0

    # map idx1
    %c =w csltl %idx1, 26
    jnz %c @map1_a
    %c =w csltl %idx1, 52
    jnz %c @map1_b
    %c =w csltl %idx1, 62
    jnz %c @map1_c
    %c =w ceql %idx1, 62
    jnz %c @map1_plus
    jmp @map1_slash
@map1_a
    %ch =l add %idx1, 65
    jmp @store1
@map1_b
    %ch =l add %idx1, 71
    jmp @store1
@map1_c
    %ch =l add %idx1, -4
    jmp @store1
@map1_plus
    %ch =l 43
    jmp @store1
@map1_slash
    %ch =l 47
@store1
    storeb %ch, %d1

    # map idx2
    %c =w csltl %idx2, 26
    jnz %c @map2_a
    %c =w csltl %idx2, 52
    jnz %c @map2_b
    %c =w csltl %idx2, 62
    jnz %c @map2_c
    %c =w ceql %idx2, 62
    jnz %c @map2_plus
    jmp @map2_slash
@map2_a
    %ch =l add %idx2, 65
    jmp @store2
@map2_b
    %ch =l add %idx2, 71
    jmp @store2
@map2_c
    %ch =l add %idx2, -4
    jmp @store2
@map2_plus
    %ch =l 43
    jmp @store2
@map2_slash
    %ch =l 47
@store2
    storeb %ch, %d2

    # map idx3
    %c =w csltl %idx3, 26
    jnz %c @map3_a
    %c =w csltl %idx3, 52
    jnz %c @map3_b
    %c =w csltl %idx3, 62
    jnz %c @map3_c
    %c =w ceql %idx3, 62
    jnz %c @map3_plus
    jmp @map3_slash
@map3_a
    %ch =l add %idx3, 65
    jmp @store3
@map3_b
    %ch =l add %idx3, 71
    jmp @store3
@map3_c
    %ch =l add %idx3, -4
    jmp @store3
@map3_plus
    %ch =l 43
    jmp @store3
@map3_slash
    %ch =l 47
@store3
    storeb %ch, %d3

    %k =l add %k, 1
    jmp @loop_cond

@remainder
    %rem =l urem %n, 3
    %c1 =w ceql %rem, 1
    jnz %c1 @rem1
    %c2 =w ceql %rem, 2
    jnz %c2 @rem2
    jmp @done

@rem1
    %src_off =l mul %full3, 3
    %p0 =l add %src, %src_off
    %b0 =l loadub %p0

    %idx0 =l shr %b0, 2
    %tmp =l and %b0, 3
    %idx1 =l shl %tmp, 4

    %dst_off =l mul %full3, 4
    %d0 =l add %dst, %dst_off
    %d1 =l add %d0, 1
    %d2 =l add %d0, 2
    %d3 =l add %d0, 3

    # map idx0
    %c =w csltl %idx0, 26
    jnz %c @r1a
    %c =w csltl %idx0, 52
    jnz %c @r1b
    %c =w csltl %idx0, 62
    jnz %c @r1c
    %c =w ceql %idx0, 62
    jnz %c @r1_plus
    jmp @r1_slash
@r1a
    %ch =l add %idx0, 65
    jmp @r1store0
@r1b
    %ch =l add %idx0, 71
    jmp @r1store0
@r1c
    %ch =l add %idx0, -4
    jmp @r1store0
@r1_plus
    %ch =l 43
    jmp @r1store0
@r1_slash
    %ch =l 47
@r1store0
    storeb %ch, %d0

    # map idx1
    %c =w csltl %idx1, 26
    jnz %c @r1a1
    %c =w csltl %idx1, 52
    jnz %c @r1b1
    %c =w csltl %idx1, 62
    jnz %c @r1c1
    %c =w ceql %idx1, 62
    jnz %c @r1_plus1
    jmp @r1_slash1
@r1a1
    %ch =l add %idx1, 65
    jmp @r1store1
@r1b1
    %ch =l add %idx1, 71
    jmp @r1store1
@r1c1
    %ch =l add %idx1, -4
    jmp @r1store1
@r1_plus1
    %ch =l 43
    jmp @r1store1
@r1_slash1
    %ch =l 47
@r1store1
    storeb %ch, %d1

    %cpad =l 61
    storeb %cpad, %d2
    storeb %cpad, %d3
    jmp @done

@rem2
    %src_off =l mul %full3, 3
    %p0 =l add %src, %src_off
    %b0 =l loadub %p0
    %p1 =l add %p0, 1
    %b1 =l loadub %p1

    %idx0 =l shr %b0, 2
    %tmp0 =l and %b0, 3
    %tmp1 =l shr %b1, 4
    %idx1 =l or %tmp0, %tmp1
    %idx1 =l shl %idx1, 4
    %tmp2 =l and %b1, 15
    %idx2 =l shl %tmp2, 2

    %dst_off =l mul %full3, 4
    %d0 =l add %dst, %dst_off
    %d1 =l add %d0, 1
    %d2 =l add %d0, 2
    %d3 =l add %d0, 3

    # map idx0
    %c =w csltl %idx0, 26
    jnz %c @r2a
    %c =w csltl %idx0, 52
    jnz %c @r2b
    %c =w csltl %idx0, 62
    jnz %c @r2c
    %c =w ceql %idx0, 62
    jnz %c @r2_plus
    jmp @r2_slash
@r2a
    %ch =l add %idx0, 65
    jmp @r2store0
@r2b
    %ch =l add %idx0, 71
    jmp @r2store0
@r2c
    %ch =l add %idx0, -4
    jmp @r2store0
@r2_plus
    %ch =l 43
    jmp @r2store0
@r2_slash
    %ch =l 47
@r2store0
    storeb %ch, %d0

    # map idx1
    %c =w csltl %idx1, 26
    jnz %c @r2a1
    %c =w csltl %idx1, 52
    jnz %c @r2b1
    %c =w csltl %idx1, 62
    jnz %c @r2c1
    %c =w ceql %idx1, 62
    jnz %c @r2_plus1
    jmp @r2_slash1
@r2a1
    %ch =l add %idx1, 65
    jmp @r2store1
@r2b1
    %ch =l add %idx1, 71
    jmp @r2store1
@r2c1
    %ch =l add %idx1, -4
    jmp @r2store1
@r2_plus1
    %ch =l 43
    jmp @r2store1
@r2_slash1
    %ch =l 47
@r2store1
    storeb %ch, %d1

    # map idx2
    %c =w csltl %idx2, 26
    jnz %c @r2a2
    %c =w csltl %idx2, 52
    jnz %c @r2b2
    %c =w csltl %idx2, 62
    jnz %c @r2c2
    %c =w ceql %idx2, 62
    jnz %c @r2_plus2
    jmp @r2_slash2
@r2a2
    %ch =l add %idx2, 65
    jmp @r2store2
@r2b2
    %ch =l add %idx2, 71
    jmp @r2store2
@r2c2
    %ch =l add %idx2, -4
    jmp @r2store2
@r2_plus2
    %ch =l 43
    jmp @r2store2
@r2_slash2
    %ch =l 47
@r2store2
    storeb %ch, %d2

    %cpad =l 61
    storeb %cpad, %d3
    jmp @done

@done
    %tmp =l add %n, 2
    %groups =l udiv %tmp, 3
    %out_len =l mul %groups, 4
    ret %out_len
}
