export function l $base64_encode(l %src, l %n, l %dst) {
@entry
	%full3 =l divl %n, 3
	%k =l 0
	jmp @loop_cond

@loop_cond
	%c_loop =w csltl %k, %full3
	jnz %c_loop @process3
	jmp @remainder

@process3
	%src_off =l mul %k, 3
	%dst_off =l mul %k, 4

	%p0 =l add %src, %src_off
	%b0 =l loadub %p0
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%p2 =l add %p0, 2
	%b2 =l loadub %p2

	%v0 =l shll %b0, 16
	%v1 =l shll %b1, 8
	%v =l or %v0, %v1
	%v =l or %v, %b2

	%idx0 =l and (shrl %v, 18), 63
	%idx1 =l and (shrl %v, 12), 63
	%idx2 =l and (shrl %v, 6), 63
	%idx3 =l and %v, 63

	%d0 =l add %dst, %dst_off
	%d1 =l add %d0, 1
	%d2 =l add %d0, 2
	%d3 =l add %d0, 3

	# map idx0 -> ch0
	%c1 =w csltl %idx0, 26
	jnz %c1 @map0_a
	%c2 =w csltl %idx0, 52
	jnz %c2 @map0_b
	%c3 =w csltl %idx0, 62
	jnz %c3 @map0_c
	%c62 =w ceql %idx0, 62
	jnz %c62 @map0_plus
	jmp @map0_slash

@map0_a
	%ch =l add %idx0, 65
	jmp @map0_done
@map0_b
	%ch =l add %idx0, 71
	jmp @map0_done
@map0_c
	%ch =l add %idx0, -4
	jmp @map0_done
@map0_plus
	%ch =l 43
	jmp @map0_done
@map0_slash
	%ch =l 47
@map0_done
	storeb %ch, %d0

	# map idx1 -> ch1
	%c1 =w csltl %idx1, 26
	jnz %c1 @map1_a
	%c2 =w csltl %idx1, 52
	jnz %c2 @map1_b
	%c3 =w csltl %idx1, 62
	jnz %c3 @map1_c
	%c62 =w ceql %idx1, 62
	jnz %c62 @map1_plus
	jmp @map1_slash

@map1_a
	%ch =l add %idx1, 65
	jmp @map1_done
@map1_b
	%ch =l add %idx1, 71
	jmp @map1_done
@map1_c
	%ch =l add %idx1, -4
	jmp @map1_done
@map1_plus
	%ch =l 43
	jmp @map1_done
@map1_slash
	%ch =l 47
@map1_done
	storeb %ch, %d1

	# map idx2 -> ch2
	%c1 =w csltl %idx2, 26
	jnz %c1 @map2_a
	%c2 =w csltl %idx2, 52
	jnz %c2 @map2_b
	%c3 =w csltl %idx2, 62
	jnz %c3 @map2_c
	%c62 =w ceql %idx2, 62
	jnz %c62 @map2_plus
	jmp @map2_slash

@map2_a
	%ch =l add %idx2, 65
	jmp @map2_done
@map2_b
	%ch =l add %idx2, 71
	jmp @map2_done
@map2_c
	%ch =l add %idx2, -4
	jmp @map2_done
@map2_plus
	%ch =l 43
	jmp @map2_done
@map2_slash
	%ch =l 47
@map2_done
	storeb %ch, %d2

	# map idx3 -> ch3
	%c1 =w csltl %idx3, 26
	jnz %c1 @map3_a
	%c2 =w csltl %idx3, 52
	jnz %c2 @map3_b
	%c3 =w csltl %idx3, 62
	jnz %c3 @map3_c
	%c62 =w ceql %idx3, 62
	jnz %c62 @map3_plus
	jmp @map3_slash

@map3_a
	%ch =l add %idx3, 65
	jmp @map3_done
@map3_b
	%ch =l add %idx3, 71
	jmp @map3_done
@map3_c
	%ch =l add %idx3, -4
	jmp @map3_done
@map3_plus
	%ch =l 43
	jmp @map3_done
@map3_slash
	%ch =l 47
@map3_done
	storeb %ch, %d3

	%k =l add %k, 1
	jmp @loop_cond

@remainder
	%rem =l modl %n, 3
	%c1 =w ceql %rem, 1
	jnz %c1 @rem1
	%c2 =w ceql %rem, 2
	jnz %c2 @rem2
	jmp @done

@rem1
	%src_off =l mul %full3, 3
	%p0 =l add %src, %src_off
	%b0 =l loadub %p0

	%idx0 =l shr %b0, 2
	%tmp =l and %b0, 3
	%idx1 =l shll %tmp, 4

	%dst_off =l mul %full3, 4
	%d0 =l add %dst, %dst_off
	%d1 =l add %d0, 1
	%d2 =l add %d0, 2
	%d3 =l add %d0, 3

	# map idx0
	%c =w csltl %idx0, 26
	jnz %c @r1a
	%c =w csltl %idx0, 52
	jnz %c @r1b
	%c =w csltl %idx0, 62
	jnz %c @r1c
	%c =w ceql %idx0, 62
	jnz %c @r1plus
	jmp @r1slash

@r1a
	%ch =l add %idx0, 65
	jmp @r1done
@r1b
	%ch =l add %idx0, 71
	jmp @r1done
@r1c
	%ch =l add %idx0, -4
	jmp @r1done
@r1plus
	%ch =l 43
	jmp @r1done
@r1slash
	%ch =l 47
@r1done
	storeb %ch, %d0

	# map idx1
	%c =w csltl %idx1, 26
	jnz %c @r1a2
	%c =w csltl %idx1, 52
	jnz %c @r1b2
	%c =w csltl %idx1, 62
	jnz %c @r1c2
	%c =w ceql %idx1, 62
	jnz %c @r1plus2
	jmp @r1slash2

@r1a2
	%ch =l add %idx1, 65
	jmp @r1done2
@r1b2
	%ch =l add %idx1, 71
	jmp @r1done2
@r1c2
	%ch =l add %idx1, -4
	jmp @r1done2
@r1plus2
	%ch =l 43
	jmp @r1done2
@r1slash2
	%ch =l 47
@r1done2
	storeb %ch, %d1

	# padding
	storeb 61, %d2   # '='
	storeb 61, %d3
	jmp @done

@rem2
	%src_off =l mul %full3, 3
	%p0 =l add %src, %src_off
	%b0 =l loadub %p0
	%p1 =l add %p0, 1
	%b1 =l loadub %p1

	%idx0 =l shr %b0, 2
	%tmp0 =l and %b0, 3
	%tmp1 =l shr %b1, 4
	%idx1 =l or (shll %tmp0, 4), %tmp1
	%tmp2 =l and %b1, 15
	%idx2 =l shll %tmp2, 2

	%dst_off =l mul %full3, 4
	%d0 =l add %dst, %dst_off
	%d1 =l add %d0, 1
	%d2 =l add %d0, 2
	%d3 =l add %d0, 3

	# map idx0
	%c =w csltl %idx0, 26
	jnz %c @r2a
	%c =w csltl %idx0, 52
	jnz %c @r2b
	%c =w csltl %idx0, 62
	jnz %c @r2c
	%c =w ceql %idx0, 62
	jnz %c @r2plus
	jmp @r2slash

@r2a
	%ch =l add %idx0, 65
	jmp @r2done
@r2b
	%ch =l add %idx0, 71
	jmp @r2done
@r2c
	%ch =l add %idx0, -4
	jmp @r2done
@r2plus
	%ch =l 43
	jmp @r2done
@r2slash
	%ch =l 47
@r2done
	storeb %ch, %d0

	# map idx1
	%c =w csltl %idx1, 26
	jnz %c @r2a1
	%c =w csltl %idx1, 52
	jnz %c @r2b1
	%c =w csltl %idx1, 62
	jnz %c @r2c1
	%c =w ceql %idx1, 62
	jnz %c @r2plus1
	jmp @r2slash1

@r2a1
	%ch =l add %idx1, 65
	jmp @r2done1
@r2b1
	%ch =l add %idx1, 71
	jmp @r2done1
@r2c1
	%ch =l add %idx1, -4
	jmp @r2done1
@r2plus1
	%ch =l 43
	jmp @r2done1
@r2slash1
	%ch =l 47
@r2done1
	storeb %ch, %d1

	# map idx2
	%c =w csltl %idx2, 26
	jnz %c @r2a2
	%c =w csltl %idx2, 52
	jnz %c @r2b2
	%c =w csltl %idx2, 62
	jnz %c @r2c2
	%c =w ceql %idx2, 62
	jnz %c @r2plus2
	jmp @r2slash2

@r2a2
	%ch =l add %idx2, 65
	jmp @r2done2
@r2b2
	%ch =l add %idx2, 71
	jmp @r2done2
@r2c2
	%ch =l add %idx2, -4
	jmp @r2done2
@r2plus2
	%ch =l 43
	jmp @r2done2
@r2slash2
	%ch =l 47
@r2done2
	storeb %ch, %d2

	# padding
	storeb 61, %d3   # '='
	jmp @done

@done
	%groups =l divl (add %n, 2), 3
	%out_len =l mul %groups, 4
	ret %out_len
}
