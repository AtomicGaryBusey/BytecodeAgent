export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
%c0 =l copy 0
%c1 =l copy 1
%c8 =l copy 8
%i =l copy %c0
%cnt =l copy %c0
jmp @cond

@cond
%cond =w csltl %i, %n
jnz %cond, @body
jmp @exit

@body
%off =l mul %i, %c8
%src_addr =l add %src, %off
%x =l loadl %src_addr
%y =l call %f(l %x)
%dst_addr =l add %dst, %off
storel %y, %dst_addr
%ltw =w csltl %y, %x
%lt =l zext %ltw
%cnt =l add %cnt, %lt
%i =l add %i, %c1
jmp @cond

@exit
ret %cnt
}
