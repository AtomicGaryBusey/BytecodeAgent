export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
%rsp =l subl %rsp, 8
%i =l const 0
%cnt =l const 0
jmp @cond

@cond
%c =w csltl %i, %n
jnz %c, @body
jmp @exit

@body
%off =l mull %i, 8
%src_addr =l addl %src, %off
%x =l loadl %src_addr
%y =l call %f(l %x)
%dst_addr =l addl %dst, %off
storel %y, %dst_addr
%ltw =w csltl %y, %x
%lt =l zextl %ltw
%cnt =l addl %cnt, %lt
%i =l addl %i, 1
jmp @cond

@exit
%rsp =l addl %rsp, 8
ret %cnt
}
