export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
%sp =l sub %sp, 8
%i =l imm 0
%cnt =l imm 0
jmp @cond

@cond
%c =w cslt %i, %n
jnz %c, @body
jmp @exit

@body
%off =l mul %i, 8
%src_addr =l add %src, %off
%x =l loadl %src_addr
%y =l call %f(l %x)
%dst_addr =l add %dst, %off
storel %y, %dst_addr
%ltw =w cslt %y, %x
%lt =l zext %ltw
%cnt =l add %cnt, %lt
%i =l add %i, 1
jmp @cond

@exit
%sp =l add %sp, 8
ret %cnt
}
