export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
%rsp =l sub %rsp, 8
%i =l const 0
%cnt =l const 0
jmp @cond

@cond
%c =w cslt %i, %n
jnz %c, @body
jmp @exit

@body
%off =l mul %i, 8
%src_addr =l add %src, %off
%x =l load %src_addr
%y =l call %f(l %x)
%dst_addr =l add %dst, %off
store %y, %dst_addr
%ltw =w cslt %y, %x
%lt =l zext %ltw
%cnt =l add %cnt, %lt
%i =l add %i, 1
jmp @cond

@exit
%rsp =l add %rsp, 8
ret %cnt
}
