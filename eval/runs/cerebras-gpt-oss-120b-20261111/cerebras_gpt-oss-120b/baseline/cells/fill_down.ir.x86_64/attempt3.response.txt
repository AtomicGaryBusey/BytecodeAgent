export function l $fill_down(l %a, l %n, l %start) {
@entry
    int l 0, %i
    int l 1, %c1
    int l 3, %c3
    jmp @loop

@loop
    cslt %i, %n, %c
    jnz %c, @body
    jmp @exit

@body
    sub %start, %i, %val
    shl %i, %c3, %off
    add %a, %off, %ptr
    storel %val, %ptr
    add %i, %c1, %i
    jmp @loop

@exit
    ret %n
}
