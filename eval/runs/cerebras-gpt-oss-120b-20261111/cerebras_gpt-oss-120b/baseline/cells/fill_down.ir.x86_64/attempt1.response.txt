export function l $fill_down(l %a, l %n, l %start) {
@entry
    constl 0, %i
    jmp @loop

@loop
    csltl %i, %n, %c
    jnz %c, @body
    jmp @exit

@body
    subl %start, %i, %val
    shll %i, 3, %off
    addl %a, %off, %ptr
    storel %val, %ptr
    constl 1, %one
    addl %i, %one, %i
    jmp @loop

@exit
    ret %n
}
