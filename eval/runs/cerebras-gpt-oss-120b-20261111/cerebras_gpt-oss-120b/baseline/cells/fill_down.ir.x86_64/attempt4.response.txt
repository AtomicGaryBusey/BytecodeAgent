export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop

@loop
    %i = phi 0, @entry, %i_next, @body
    %c = cslt %i, %n
    jnz %c, @body
    jmp @exit

@body
    %val = sub %start, %i
    %off = shl %i, 3
    %ptr = add %a, %off
    storel %val, %ptr
    %i_next = add %i, 1
    jmp @loop

@exit
    ret %n
}
