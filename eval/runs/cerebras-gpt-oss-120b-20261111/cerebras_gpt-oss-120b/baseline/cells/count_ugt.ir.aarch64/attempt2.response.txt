export function l $count_ugt(l a, l n, l t)
@entry
    %i0 = const 0
    %cnt0 = const 0
    jmp %Lloop

@Lloop
    %i = phi %i0, @entry, %i1, %Lnext
    %cnt = phi %cnt0, @entry, %cnt1, %Lnext
    %cond = csltl %i, %n
    jnz %cond, %Lcheck
    ret %cnt

@Lcheck
    %off = mul %i, 8
    %ptr = addl %a, %off
    %val = loadl %ptr
    %gt = csltu %t, %val
    jnz %gt, %Linc
    jmp %Lnext

@Linc
    %cnt1 = addl %cnt, 1
    jmp %Lnext

@Lnext
    %i1 = addl %i, 1
    %cnt1 = phi %cnt, %Lcheck, %cnt1, %Linc
    jmp %Lloop
