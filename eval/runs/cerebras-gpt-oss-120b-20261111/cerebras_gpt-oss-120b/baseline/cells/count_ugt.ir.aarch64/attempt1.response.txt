export function l $count_ugt(l a, l n, l t)
@entry
    %i0 = const 0
    %cnt0 = const 0
    jmp %L0

@L0
    %i = phi %i0, @entry, %i1, @Lnext
    %cnt = phi %cnt0, @entry, %cnt2, @Lnext
    %cond = csltl %i, n
    jnz %cond, %L1
    ret %cnt

@L1
    %offset = mul %i, 8
    %ptr = addl a, %offset
    %val = loadl %ptr
    %gt = csltu t, %val
    jnz %gt, %Linc
    jmp %Lnext

@Linc
    %cnt_inc = addl %cnt, 1
    jmp %Lnext

@Lnext
    %cnt2 = phi %cnt, @L1, %cnt_inc, @Linc
    %i1 = addl %i, 1
    jmp %L0
