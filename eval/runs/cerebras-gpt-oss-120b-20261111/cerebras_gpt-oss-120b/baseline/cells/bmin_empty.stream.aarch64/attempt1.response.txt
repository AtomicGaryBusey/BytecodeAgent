bmin_empty:
    cbz x1, .Lmax
    ldr x2, [x0]          // load a[0] as current min
    add x3, x0, #8        // x3 = a + 8 (pointer to a[1])
    sub x4, x1, #1        // x4 = n - 1 (remaining elements)
    cbz x4, .Lret_min
.Lloop:
    ldr x5, [x3], #8      // load next element, post‑increment pointer
    cmp x5, x2
    csel x2, x5, x2, lt   // if a[i] < min, update min
    subs x4, x4, #1
    bne .Lloop
.Lret_min:
    mov x0, x2
    ret
.Lmax:
    mov x0, #-1
    lsr x0, x0, #1
    ret
