bubble_sort:
    cmp x1, #1
    ble .Lreturn
    mov x9, #0
.Louter:
    sub x11, x1, #1
    sub x11, x11, x9
    cbz x11, .Louter_done
    mov x10, #0
.Linner:
    lsl x14, x10, #3
    add x15, x14, #8
    ldr x12, [x0, x14]
    ldr x13, [x0, x15]
    cmp x12, x13
    ble .Lskip_swap
    str x13, [x0, x14]
    str x12, [x0, x15]
.Lskip_swap:
    add x10, x10, #1
    cmp x10, x11
    blt .Linner
    add x9, x9, #1
    b .Louter
.Louter_done:
.Lreturn:
    mov x0, #0
    ret
