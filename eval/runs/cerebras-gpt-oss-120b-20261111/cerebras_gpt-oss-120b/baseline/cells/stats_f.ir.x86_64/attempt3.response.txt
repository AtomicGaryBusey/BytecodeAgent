export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
    %c0 =l const 0
    %c1 =l const 1
    %c8 =l const 8
    %c16 =l const 16

    %addr0 =l add %a, %c0
    %aval0 =l loadl %addr0
    %y0 =l call %f(l %aval0)

    %min0 =l mov %y0
    %max0 =l mov %y0
    %sum0 =l mov %y0
    %odd0 =l and %y0, %c1

    %i0 =l const 1

    jmp @loop_hdr

@loop_hdr:
    %i =l phi %i0, @entry, %i2, @loop_end
    %min =l phi %min0, @entry, %min2, @loop_end
    %max =l phi %max0, @entry, %max2, @loop_end
    %sum =l phi %sum0, @entry, %sum2, @loop_end
    %oddcnt =l phi %odd0, @entry, %oddcnt2, @loop_end

    %cond =w csltl %i, %n
    jnz %cond, @loop_body
    jmp @done

@loop_body:
    %off =l mul %i, %c8
    %addr =l add %a, %off
    %aval =l loadl %addr
    %y =l call %f(l %aval)

    %odd =l and %y, %c1
    %oddcnt2 =l add %oddcnt, %odd
    %sum2 =l add %sum, %y

    %cmin =w csltl %y, %min
    jnz %cmin, @cmin_true
    jmp @cmin_false

@cmin_true:
    %min2 =l mov %y
    jmp @cmax_check

@cmin_false:
    %min2 =l mov %min
    jmp @cmax_check

@cmax_check:
    %cmax =w csltl %max, %y
    jnz %cmax, @cmax_true
    jmp @cmax_false

@cmax_true:
    %max2 =l mov %y
    jmp @loop_end

@cmax_false:
    %max2 =l mov %max
    jmp @loop_end

@loop_end:
    %i2 =l add %i, %c1
    jmp @loop_hdr

@done:
    %out0 =l add %out, %c0
    storel %min, %out0
    %out1 =l add %out, %c8
    storel %max, %out1
    %out2 =l add %out, %c16
    storel %sum, %out2
    ret %oddcnt
}
