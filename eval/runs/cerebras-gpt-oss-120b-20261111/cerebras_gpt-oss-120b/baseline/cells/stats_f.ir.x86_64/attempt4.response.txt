export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
    %c0 =l const 0
    %c1 =l const 1
    %c8 =l const 8
    %c16 =l const 16

    %addr0 =l add %a, %c0
    %aval0 =l loadl %addr0
    %y0 =l call %f(l %aval0)

    %min0 =l mov %y0
    %max0 =l mov %y0
    %sum0 =l mov %y0
    %odd0 =l and %y0, %c1

    %i0 =l const 1

    jmp @loop_hdr

@loop_hdr:
    %i =l phi %i0, @entry, %i_next, @loop_end
    %min =l phi %min0, @entry, %min_next, @loop_end
    %max =l phi %max0, @entry, %max_next, @loop_end
    %sum =l phi %sum0, @entry, %sum_next, @loop_end
    %oddcnt =l phi %odd0, @entry, %oddcnt_next, @loop_end

    %cond =w csltl %i, %n
    jnz %cond, @loop_body
    jmp @done

@loop_body:
    %off =l mul %i, %c8
    %addr =l add %a, %off
    %aval =l loadl %addr
    %y =l call %f(l %aval)

    %odd =l and %y, %c1
    %oddcnt_next =l add %oddcnt, %odd

    %sum_next =l add %sum, %y

    %cmin =w csltl %y, %min
    jnz %cmin, @min_update
    jmp @min_keep

@min_update:
    %min_next =l mov %y
    jmp @max_check

@min_keep:
    %min_next =l mov %min
    jmp @max_check

@max_check:
    %cmax =w csltl %max, %y
    jnz %cmax, @max_update
    jmp @max_keep

@max_update:
    %max_next =l mov %y
    jmp @loop_end

@max_keep:
    %max_next =l mov %max
    jmp @loop_end

@loop_end:
    %i_next =l add %i, %c1
    jmp @loop_hdr

@done:
    %out0 =l add %out, %c0
    storel %min, %out0
    %out1 =l add %out, %c8
    storel %max, %out1
    %out2 =l add %out, %c16
    storel %sum, %out2
    ret %oddcnt
}
