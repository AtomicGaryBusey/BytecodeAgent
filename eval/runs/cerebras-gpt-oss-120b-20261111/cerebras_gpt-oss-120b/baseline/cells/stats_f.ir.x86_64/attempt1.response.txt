export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
    %rsp =l subl %rsp, 8
    %c0 =l const 0
    %c1 =l const 1
    %c8 =l const 8
    %c16 =l const 16
    %addr0 =l addl %a, %c0
    %aval0 =l loadl %addr0
    %y0 =l call %f(l %aval0)
    %min0 =l mov %y0
    %max0 =l mov %y0
    %sum0 =l mov %y0
    %odd0 =l andl %y0, %c1
    %i0 =l const 1
    jmp @loop_hdr

@loop_hdr:
    %i =l phi %i0, @entry, %i_next, @loop_body
    %min =l phi %min0, @entry, %min_next, @loop_body
    %max =l phi %max0, @entry, %max_next, @loop_body
    %sum =l phi %sum0, @entry, %sum_next, @loop_body
    %oddcnt =l phi %odd0, @entry, %oddcnt_next, @loop_body
    %cond =w csltl %i, %n
    jnz %cond, @loop_body
    jmp @done

@loop_body:
    %off =l shl %i, 3
    %addr =l addl %a, %off
    %aval =l loadl %addr
    %y =l call %f(l %aval)
    %odd =l andl %y, %c1
    %oddcnt_next =l addl %oddcnt, %odd
    %sum_next =l addl %sum, %y
    %cmin =w csltl %y, %min
    jnz %cmin, @cmin_true
    jmp @cmin_false

@cmin_true:
    jmp @cmin_merge

@cmin_false:
    jmp @cmin_merge

@cmin_merge:
    %min_next =l phi %y, @cmin_true, %min, @cmin_false
    %cmax =w csltl %max, %y
    jnz %cmax, @cmax_true
    jmp @cmax_false

@cmax_true:
    jmp @cmax_merge

@cmax_false:
    jmp @cmax_merge

@cmax_merge:
    %max_next =l phi %y, @cmax_true, %max, @cmax_false
    %i_next =l addl %i, %c1
    jmp @loop_hdr

@done:
    %out0 =l addl %out, %c0
    storel %min, %out0
    %out1 =l addl %out, %c8
    storel %max, %out1
    %out2 =l addl %out, %c16
    storel %sum, %out2
    %rsp =l addl %rsp, 8
    ret %oddcnt
}
