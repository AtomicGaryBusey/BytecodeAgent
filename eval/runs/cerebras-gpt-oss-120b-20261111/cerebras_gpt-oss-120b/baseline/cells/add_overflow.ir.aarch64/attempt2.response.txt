export function l $add_overflow(l %a, l %b, l %out) {
@start {
    %sum = addl %a, %b
    storel %sum, %out
    %sign_a = csltl %a, 0
    %sign_b = csltl %b, 0
    %sign_s = csltl %sum, 0
    %diff_ab = xorw %sign_a, %sign_b
    %not_same = subw 1, %diff_ab
    %diff_as = xorw %sign_a, %sign_s
    %ov = andw %not_same, %diff_as
    %res = zextl %ov
    ret %res
}
}
