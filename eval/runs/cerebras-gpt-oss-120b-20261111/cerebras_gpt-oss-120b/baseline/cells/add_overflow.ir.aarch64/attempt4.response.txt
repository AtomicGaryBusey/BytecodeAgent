export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum = add %a, %b
    storel %sum, %out
    %sign_a = cslt %a, 0
    %sign_b = cslt %b, 0
    %sign_s = cslt %sum, 0
    %xor_ab = xorw %sign_a, %sign_b
    %same_ab = subw 1, %xor_ab
    %xor_as = xorw %sign_a, %sign_s
    %ov_w = andw %same_ab, %xor_as
    %ov = zextl %ov_w
    ret %ov
}
