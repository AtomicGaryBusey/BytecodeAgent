export function l count_desc(l a, l n) {
@entry
    %n = arg n
    %a = arg a
    %c0 = const l 0
    %c1 = const l 1
    %c2 = const l 2
    %c8 = const l 8
    %lt2 = csltl %n, %c2
    jnz %lt2, @ret0

    %last = subl %n, %c1
    %i = const l 0
    %cnt = const l 0
    jmp @loop

@loop
    %cond = csltl %i, %last
    jnz %cond, @body
    jmp @exit

@body
    %off = mull %i, %c8
    %p1 = addl %a, %off
    %x = loadl %p1
    %off2 = addl %off, %c8
    %p2 = addl %a, %off2
    %y = loadl %p2
    %gt = csgtl %x, %y
    %gtl = sextw %gt
    %cnt = addl %cnt, %gtl
    %i = addl %i, %c1
    jmp @loop

@exit
    ret %cnt

@ret0
    ret %c0
}
