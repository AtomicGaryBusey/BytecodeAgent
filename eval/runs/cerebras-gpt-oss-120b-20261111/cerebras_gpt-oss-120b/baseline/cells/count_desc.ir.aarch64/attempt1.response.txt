export function l $count_desc(l a, l n) {
@entry
    %n = arg n
    %a = arg a
    %c2 = const l 2
    %c0 = const l 0
    %c1 = const l 1
    %c8 = const l 8
    %lt2 = csltl %n, %c2
    jnz %lt2, @ret0

    %last = subl %n, %c1        // n-1 comparisons
    %i = const l 0
    %cnt = const l 0
    jmp @loop

@loop
    %cond = csltl %i, %last
    jnz %cond, @body
    jmp @exit

@body
    %off = mul %i, %c8
    %p1 = add %a, %off
    %x = loadl %p1
    %off2 = add %off, %c8
    %p2 = add %a, %off2
    %y = loadl %p2
    %gt = csgtl %x, %y          // w (0/1)
    %gtl = sextw %gt            // extend to l
    %cnt = addl %cnt, %gtl
    %i = addl %i, %c1
    jmp @loop

@exit
    ret %cnt

@ret0
    ret %c0
}
