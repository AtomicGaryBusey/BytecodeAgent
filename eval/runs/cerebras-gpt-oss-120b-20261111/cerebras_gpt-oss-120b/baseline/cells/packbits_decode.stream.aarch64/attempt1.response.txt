packbits_decode:
    mov     x5, #0                 // src_idx
    mov     x6, #0                 // out_idx
.Lloop_start:
    cmp     x5, x1
    b.ge    .Ldone
    ldrb    w4, [x0, x5]           // header byte
    add     x5, x5, #1
    sxtb    w4, w4                 // sign-extend
    cmp     w4, #-128
    beq     .Lnoop
    cmp     w4, #0
    b.ge    .Lliteral
    // repeat packet
    cmp     x5, x1
    b.ge    .Lerror_src
    ldrb    w8, [x0, x5]           // repeat value
    add     x5, x5, #1
    mov     w9, #1
    sub     w9, w9, w4            // w9 = 1 - h
    uxtw    x9, w9                 // x9 = repeat count
    sub     x10, x3, x6           // remaining capacity
    cmp     x9, x10
    b.gt    .Loverflow_repeat
    mov     x11, #0
.Lrepeat_loop:
    cmp     x11, x9
    b.ge    .Lrepeat_done
    strb    w8, [x2, x6]
    add     x6, x6, #1
    add     x11, x11, #1
    b       .Lrepeat_loop
.Lrepeat_done:
    b       .Lloop_start
.Lliteral:
    add     w12, w4, #1           // w12 = literal length
    uxtw    x12, w12              // x12 = literal length
    add     x13, x5, x12          // x13 = src_idx + len
    cmp     x13, x1
    b.gt    .Lerror_src
    sub     x10, x3, x6           // remaining capacity
    cmp     x12, x10
    b.gt    .Loverflow_literal
    mov     x11, #0
.Lcopy_loop:
    cmp     x11, x12
    b.ge    .Lcopy_done
    ldrb    w4, [x0, x5]
    strb    w4, [x2, x6]
    add     x5, x5, #1
    add     x6, x6, #1
    add     x11, x11, #1
    b       .Lcopy_loop
.Lcopy_done:
    b       .Lloop_start
.Lnoop:
    b       .Lloop_start
.Loverflow_literal:
    mov     x11, #0
.Loverflow_copy_loop:
    cmp     x11, x10
    b.ge    .Lreturn_error
    ldrb    w4, [x0, x5]
    strb    w4, [x2, x6]
    add     x5, x5, #1
    add     x6, x6, #1
    add     x11, x11, #1
    b       .Loverflow_copy_loop
.Loverflow_repeat:
    mov     x11, #0
.Loverflow_repeat_loop:
    cmp     x11, x10
    b.ge    .Lreturn_error
    strb    w8, [x2, x6]
    add     x6, x6, #1
    add     x11, x11, #1
    b       .Loverflow_repeat_loop
.Lerror_src:
    mov     x0, #-2
    ret
.Lreturn_error:
    mov     x0, #-1
    ret
.Ldone:
    mov     x0, x6
    ret
