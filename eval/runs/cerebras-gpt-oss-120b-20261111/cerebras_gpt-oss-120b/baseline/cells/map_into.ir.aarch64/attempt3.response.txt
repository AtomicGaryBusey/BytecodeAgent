export function l $map_into(l, l, l, l) {
@entry
  %dst =l arg 0
  %src =l arg 1
  %n   =l arg 2
  %f   =l arg 3
  %i0  =l const 0
  %cnt0 =l const 0
  %one =l const 1
  jmp @Lcond

@Lcond
  %i   =l phi %i0 @entry, %i_next @Lbody
  %cnt =l phi %cnt0 @entry, %cnt_next @Lbody
  %c   =w csltl %i, %n
  jnz %c @Lbody @Lexit

@Lbody
  %off      =l shll %i, 3
  %src_addr =l addl %src, %off
  %val      =l loadl %src_addr
  %res      =l call %f (l %val)
  %dst_addr =l addl %dst, %off
  storel %res, %dst_addr
  %lt_w     =w csltl %res, %val
  %lt_l     =l zextw %lt_w
  %cnt_next =l addl %cnt, %lt_l
  %i_next   =l addl %i, %one
  jmp @Lcond

@Lexit
  ret %cnt
}
