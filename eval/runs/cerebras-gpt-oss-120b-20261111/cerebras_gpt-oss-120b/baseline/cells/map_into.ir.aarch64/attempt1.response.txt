export function l $map_into(l dst, l src, l n, l f) {
@entry
  %dst =l arg 0
  %src =l arg 1
  %n =l arg 2
  %f =l arg 3
  %i =l const 0
  %cnt =l const 0
  %one =l const 1
  jmp .Lcond
@Lcond
  %c =w csltl %i, %n
  jnz %c, .Lbody, .Lexit
@Lbody
  %off =l shll %i, 3
  %src_addr =l addl %src, %off
  %val =l loadl %src_addr
  %res =l call %f(l %val)
  %dst_addr =l addl %dst, %off
  storel %res, %dst_addr
  %lt =w csltl %res, %val
  %lt_l =l zextw %lt
  %cnt =l addl %cnt, %lt_l
  %i =l addl %i, %one
  jmp .Lcond
@Lexit
  ret %cnt
}
