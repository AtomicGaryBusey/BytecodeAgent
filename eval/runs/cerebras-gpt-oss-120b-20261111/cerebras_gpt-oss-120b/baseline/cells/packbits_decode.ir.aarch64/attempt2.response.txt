export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
%zero =l 0
%neg1 =l -1
%neg2 =l -2
%neg128 =l -128

%i =l 0
%o =l 0
jmp @check_src

@check_src
%src_rem =l sub %n, %i
%src_has =w csgt %src_rem, %zero
jnz %src_has, @read_hdr
jmp @done

@read_hdr
%hdr_ptr =l add %src, %i
%h =w loadsb %hdr_ptr
%h64 =l sextw %h

%is_nop =w ceql %h64, %neg128
jnz %is_nop, @nop

%is_lit =w csgel %h64, %zero
jnz %is_lit, @literal

# ---- repeat packet -------------------------------------------------
%src_need =l 2               # header + repeat byte
%src_ok =w csgel %src_rem, %src_need
jnz %src_ok, @repeat_body
ret %neg2

@repeat_body
%rep_ptr =l add %src, add %i, 1
%rep =w loadub %rep_ptr
%rep64 =l sextw %rep
%cnt =l sub 1, %h64          # repeat count = 1 - h (h negative)

%i =l add %i, 2              # consume header and repeat byte

%out_rem =l sub %cap, %o
%exceed =w csgt %cnt, %out_rem
jnz %exceed, @repeat_overflow

%rcnt =l mov %cnt
@repeat_copy
%rcnt_zero =w csgel %rcnt, %zero
jnz %rcnt_zero, @loop
%out_addr =l add %out, %o
storeb %rep, %out_addr
%o =l add %o, 1
%rcnt =l sub %rcnt, 1
jmp @repeat_copy

@repeat_overflow
%rem =l mov %out_rem
jnz %rem, @repeat_copy_over
ret %neg1

@repeat_copy_over
%out_addr =l add %out, %o
storeb %rep, %out_addr
%o =l add %o, 1
%rem =l sub %rem, 1
jnz %rem, @repeat_copy_over
ret %neg1

# ---- literal packet ------------------------------------------------
@literal
%cnt =l add %h64, 1               # literal count
%src_need =l add %cnt, 1          # header + literals
%src_ok =w csgel %src_rem, %src_need
jnz %src_ok, @literal_body
ret %neg2

@literal_body
%i =l add %i, 1                    # skip header

%out_rem =l sub %cap, %o
%exceed =w csgt %cnt, %out_rem
jnz %exceed, @literal_overflow

%lcnt =l mov %cnt
@literal_copy
%lcnt_zero =w csgel %lcnt, %zero
jnz %lcnt_zero, @loop
%src_addr =l add %src, %i
%b =w loadub %src_addr
%out_addr =l add %out, %o
storeb %b, %out_addr
%i =l add %i, 1
%o =l add %o, 1
%lcnt =l sub %lcnt, 1
jmp @literal_copy

@literal_overflow
%rem =l mov %out_rem
jnz %rem, @literal_copy_over
ret %neg1

@literal_copy_over
%src_addr =l add %src, %i
%b =w loadub %src_addr
%out_addr =l add %out, %o
storeb %b, %out_addr
%i =l add %i, 1
%o =l add %o, 1
%rem =l sub %rem, 1
jnz %rem, @literal_copy_over
ret %neg1

# ---- nop packet ----------------------------------------------------
@nop
%i =l add %i, 1
jmp @check_src

@loop
jmp @check_src

@done
ret %o
}
