export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
%zero =l const 0
%neg1 =l const -1
%neg2 =l const -2
%neg128 =l const -128

%i =l const 0
%o =l const 0
jmp @loop

@loop
%src_rem =l sub %n, %i
%src_done =csgel %src_rem, %zero
jnz %src_done, @done

%src_hdr_ptr =l add %src, %i
%h =w loadsb %src_hdr_ptr
%h64 =l sextw %h

%is_neg128 =ceql %h64, %neg128
jnz %is_neg128, @nop

%is_nonneg =csgel %h64, %zero
jnz %is_nonneg, @literal

# ---- repeat packet -------------------------------------------------
%src_need =l const 2
%src_ok =csgel %src_rem, %src_need
jnz %src_ok, @repeat_body
ret %neg2

@repeat_body
%src_rep_ptr =l add %src, add %i, 1
%rep =w loadub %src_rep_ptr
%rep64 =l sextw %rep
%cnt =l sub 1, %h64        # count = 1 - h (h is negative)

%i =l add %i, 2            # consume header and repeat byte

%out_rem =l sub %cap, %o
%exceed =csgt %cnt, %out_rem
jnz %exceed, @repeat_overflow

# copy repeat byte %cnt times
%rcnt =l mov %cnt
@repeat_copy
%rcnt_zero =csgel %rcnt, %zero
jnz %rcnt_zero, @loop
%out_addr =l add %out, %o
storeb %rep, %out_addr
%o =l add %o, 1
%rcnt =l sub %rcnt, 1
jmp @repeat_copy

@repeat_overflow
%rem =l mov %out_rem
jnz %rem, @repeat_copy_over
ret %neg1

@repeat_copy_over
%out_addr =l add %out, %o
storeb %rep, %out_addr
%o =l add %o, 1
%rem =l sub %rem, 1
jnz %rem, @repeat_copy_over
ret %neg1

# ---- literal packet ------------------------------------------------
@literal
%cnt =l add %h64, 1               # literal count
%src_need =l add %cnt, 1          # header + literals
%src_ok =csgel %src_rem, %src_need
jnz %src_ok, @literal_body
ret %neg2

@literal_body
%i =l add %i, 1                    # skip header

%out_rem =l sub %cap, %o
%exceed =csgt %cnt, %out_rem
jnz %exceed, @literal_overflow

%lcnt =l mov %cnt
@literal_copy
%lcnt_zero =csgel %lcnt, %zero
jnz %lcnt_zero, @loop
%src_addr =l add %src, %i
%b =w loadub %src_addr
%out_addr =l add %out, %o
storeb %b, %out_addr
%i =l add %i, 1
%o =l add %o, 1
%lcnt =l sub %lcnt, 1
jmp @literal_copy

@literal_overflow
%rem =l mov %out_rem
jnz %rem, @literal_copy_over
ret %neg1

@literal_copy_over
%src_addr =l add %src, %i
%b =w loadub %src_addr
%out_addr =l add %out, %o
storeb %b, %out_addr
%i =l add %i, 1
%o =l add %o, 1
%rem =l sub %rem, 1
jnz %rem, @literal_copy_over
ret %neg1

# ---- nop packet ----------------------------------------------------
@nop
%i =l add %i, 1
jmp @loop

@done
ret %o
}
