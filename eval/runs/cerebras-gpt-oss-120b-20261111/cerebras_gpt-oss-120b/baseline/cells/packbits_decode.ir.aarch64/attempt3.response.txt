export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry:
    %zero =l 0
    %neg1 =l -1
    %neg2 =l -2
    %neg128 =l -128

    %i =l 0
    %o =l 0
    jmp @check_src

@check_src:
    %src_rem =l sub %n, %i
    %has_src =w csgt %src_rem, %zero
    jnz %has_src, @read_hdr
    ret %o

@read_hdr:
    %hdr_ptr =l add %src, %i
    %h =w loadsb %hdr_ptr
    %h64 =l sextw %h

    %is_nop =w ceql %h64, %neg128
    jnz %is_nop, @nop

    %is_literal =w csgte %h64, %zero
    jnz %is_literal, @literal

    // repeat packet (h negative, not -128)
    // need at least header + repeat byte
    %src_need =l 2
    %src_ok =w csgte %src_rem, %src_need
    jnz %src_ok, @repeat_body
    ret %neg2

@repeat_body:
    %rep_ptr =l add %src, %i
    %rep_ptr =l add %rep_ptr, 1
    %rep =w loadub %rep_ptr
    %cnt =l sub 1, %h64          // repeat count = 1 - h (h negative)

    %i =l add %i, 2              // consume header and repeat byte

    %out_rem =l sub %cap, %o
    %exceed =w csgt %cnt, %out_rem
    jnz %exceed, @repeat_overflow

    // copy all repeats
    %rcnt =l %cnt
@repeat_copy_loop:
    %rcnt_pos =w csgt %rcnt, %zero
    jnz %rcnt_pos, @repeat_copy_store
    jmp @check_src

@repeat_copy_store:
    %out_addr =l add %out, %o
    storeb %rep, %out_addr
    %o =l add %o, 1
    %rcnt =l sub %rcnt, 1
    jmp @repeat_copy_loop

@repeat_overflow:
    %out_rem =l sub %cap, %o
    %rem =l %out_rem
    %rem_zero =w ceql %rem, %zero
    jnz %rem_zero, @return_neg1
@repeat_copy_over:
    %out_addr =l add %out, %o
    storeb %rep, %out_addr
    %o =l add %o, 1
    %rem =l sub %rem, 1
    %rem_pos =w csgt %rem, %zero
    jnz %rem_pos, @repeat_copy_over
@return_neg1:
    ret %neg1

@nop:
    %i =l add %i, 1
    jmp @check_src

@literal:
    %cnt =l add %h64, 1               // literal count
    %src_need =l add %cnt, 1          // header + literals
    %src_ok =w csgte %src_rem, %src_need
    jnz %src_ok, @literal_body
    ret %neg2

@literal_body:
    %i =l add %i, 1                    // skip header
    %out_rem =l sub %cap, %o
    %exceed =w csgt %cnt, %out_rem
    jnz %exceed, @literal_overflow

    %lcnt =l %cnt
@literal_copy_loop:
    %lcnt_pos =w csgt %lcnt, %zero
    jnz %lcnt_pos, @literal_copy_store
    jmp @check_src

@literal_copy_store:
    %src_addr =l add %src, %i
    %b =w loadub %src_addr
    %out_addr =l add %out, %o
    storeb %b, %out_addr
    %i =l add %i, 1
    %o =l add %o, 1
    %lcnt =l sub %lcnt, 1
    jmp @literal_copy_loop

@literal_overflow:
    %out_rem =l sub %cap, %o
    %rem =l %out_rem
    %rem_zero =w ceql %rem, %zero
    jnz %rem_zero, @return_neg1
@literal_copy_over:
    %src_addr =l add %src, %i
    %b =w loadub %src_addr
    %out_addr =l add %out, %o
    storeb %b, %out_addr
    %i =l add %i, 1
    %o =l add %o, 1
    %rem =l sub %rem, 1
    %rem_pos =w csgt %rem, %zero
    jnz %rem_pos, @literal_copy_over
    jmp @return_neg1
}
