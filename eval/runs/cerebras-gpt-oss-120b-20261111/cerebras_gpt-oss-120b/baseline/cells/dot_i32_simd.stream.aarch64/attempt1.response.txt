dot_i32_simd:
    mov     x4, #0                          // scalar sum = 0
    eor     v0.16b, v0.16b, v0.16b          // vector accumulator = 0
.Lvec_loop:
    cmp     x2, #4
    blt     .Lscalar_loop
    ld1     {v1.4s}, [x0], #16              // load 4 int32 from a
    ld1     {v2.4s}, [x1], #16              // load 4 int32 from b
    smlal   v0.2d, v1.2s, v2.2s             // multiply low halves and accumulate
    smlal2  v0.2d, v1.4s, v2.4s             // multiply high halves and accumulate
    sub     x2, x2, #4
    b       .Lvec_loop
.Lscalar_loop:
    umov    x5, v0.d[0]                     // extract lane 0
    umov    x6, v0.d[1]                     // extract lane 1
    add     x4, x4, x5
    add     x4, x4, x6
    cbz     x2, .Ldone
.Lrem_loop:
    ldr     w5, [x0], #4
    ldr     w6, [x1], #4
    sxtw    x5, w5
    sxtw    x6, w6
    mul     x7, x5, x6
    add     x4, x4, x7
    sub     x2, x2, #1
    cbnz    x2, .Lrem_loop
.Ldone:
    mov     x0, x4
    ret
