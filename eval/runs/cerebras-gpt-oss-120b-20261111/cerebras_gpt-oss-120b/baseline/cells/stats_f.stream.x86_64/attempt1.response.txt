stats_f:
    push r12
    push r13
    push r14
    push r15
    push rbx
    push rbp
    sub rsp, 8
    mov r12, rdx
    mov r13, rcx
    mov r11, rdi
    mov r14, 0x7FFFFFFFFFFFFFFF
    mov r15, 0x8000000000000000
    xor rbx, rbx
    xor r8, r8
    xor rdx, rdx
.Lloop:
    cmp rdx, rsi
    jge .Lend
    mov rdi, [r11 + rdx*8]
    call r12
    add rbx, rax
    test rax, 1
    jz .Lskipodd
    inc r8
.Lskipodd:
    cmp rax, r14
    jge .Lskipmin
    mov r14, rax
.Lskipmin:
    cmp rax, r15
    jle .Lskipmax
    mov r15, rax
.Lskipmax:
    inc rdx
    jmp .Lloop
.Lend:
    mov [r13 + 0], r14
    mov [r13 + 8], r15
    mov [r13 + 16], rbx
    mov rax, r8
    add rsp, 8
    pop rbp
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
