stats_f:
    push r12
    push r13
    push r14
    push r15
    push rbx
    sub rsp, 32
    mov r12, rdx                 # f pointer
    mov r13, rcx                 # out pointer
    mov r14, rdi                 # a pointer
    mov r15, rsi                 # n counter
    mov rax, 0x7FFFFFFFFFFFFFFF
    mov qword ptr [rsp], rax    # min = INT64_MAX
    mov rax, 0x8000000000000000
    mov qword ptr [rsp+8], rax  # max = INT64_MIN
    xor rax, rax
    mov qword ptr [rsp+16], rax # sum = 0
    mov qword ptr [rsp+24], rax # oddcnt = 0
.Lloop:
    test r15, r15
    jz .Lend
    mov rax, qword ptr [r14]    # load a[i]
    call r12                    # y = f(a[i])
    cmp rax, qword ptr [rsp]    # update min
    jge .Lskip_min
    mov qword ptr [rsp], rax
.Lskip_min:
    cmp rax, qword ptr [rsp+8]  # update max
    jle .Lskip_max
    mov qword ptr [rsp+8], rax
.Lskip_max:
    add qword ptr [rsp+16], rax # sum += y
    test rax, 1
    jz .Lskip_odd
    inc qword ptr [rsp+24]      # oddcnt++
.Lskip_odd:
    add r14, 8                  # a++
    dec r15                     # n--
    jmp .Lloop
.Lend:
    mov rdx, qword ptr [rsp]      # min
    mov qword ptr [r13], rdx
    mov rdx, qword ptr [rsp+8]    # max
    mov qword ptr [r13+8], rdx
    mov rdx, qword ptr [rsp+16]   # sum
    mov qword ptr [r13+16], rdx
    mov rax, qword ptr [rsp+24]   # return oddcnt
    add rsp, 32
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
