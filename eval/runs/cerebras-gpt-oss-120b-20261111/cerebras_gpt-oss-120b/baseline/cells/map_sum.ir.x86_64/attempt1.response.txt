export function l $map_sum(l %a, l %n, l %f) {
@entry
%sum =l const 0
%i =l const 0
jmp .Lcond

.Lcond
%c =w csltl %i, %n
jnz %c, .Lbody
jmp .Lend

.Lbody
%off =l mul %i, 8
%addr =l add %a, %off
%elem =l loadl %addr
%v =l call %f(l %elem)
%sum =l add %sum, %v
%i =l add %i, 1
jmp .Lcond

.Lend
ret %sum
}
