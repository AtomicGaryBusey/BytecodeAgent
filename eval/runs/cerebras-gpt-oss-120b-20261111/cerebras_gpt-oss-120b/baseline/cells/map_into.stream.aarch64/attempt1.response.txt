map_into:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    mov x19, x3          // f
    mov x20, x1          // src
    mov x21, x0          // dst
    mov x22, #0          // count
    mov x23, #0          // i
    mov x25, x2          // n
.Lloop:
    cmp x23, x25
    b.ge .Lendloop
    ldr x0, [x20, x23, lsl #3]   // src[i] -> arg
    blr x19
    str x0, [x21, x23, lsl #3]   // dst[i] = result
    ldr x1, [x20, x23, lsl #3]   // reload src[i] for compare
    cmp x0, x1
    b.lt .Linc
    b .Lnext
.Linc:
    add x22, x22, #1
.Lnext:
    add x23, x23, #1
    b .Lloop
.Lendloop:
    mov x0, x22
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
