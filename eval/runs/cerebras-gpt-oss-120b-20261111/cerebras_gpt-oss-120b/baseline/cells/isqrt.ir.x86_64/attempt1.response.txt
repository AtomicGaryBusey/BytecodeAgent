export function l $isqrt(l n) {
	@entry
		%low0 = constl 0
		%high0 = constl 4294967296
		jmp @loop(%low0, %high0)

	@loop(l low, l high)
		%cond = cslt %low, %high
		jnz %cond, @body(%low, %high)
		jmp @done(%low)

	@body(l low, l high)
		%sum = addl %low, %high
		%mid = srl %sum, constl 1
		%mid_sq = mull %mid, %mid
		%le = cule %mid_sq, %n
		%mid_inc = addl %mid, constl 1
		%low1 = sel %le, %mid_inc, %low
		%high1 = sel %le, %high, %mid
		jmp @loop(%low1, %high1)

	@done(l low)
		%res = subl %low, constl 1
		ret %res
}
