export function l $isqrt(l n) {
    @entry
        %zero = constl 0
        %one = constl 1
        %n_is_zero = ceql %n, %zero
        jnz %n_is_zero, @ret0

        %x0 = %n
        %n_div_x0 = udivl %n, %x0
        %sum0 = addl %x0, %n_div_x0
        %x1 = srl %sum0, constl 1

        %n_div_x1 = udivl %n, %x1
        %sum1 = addl %x1, %n_div_x1
        %x2 = srl %sum1, constl 1

        %n_div_x2 = udivl %n, %x2
        %sum2 = addl %x2, %n_div_x2
        %x3 = srl %sum2, constl 1

        %n_div_x3 = udivl %n, %x3
        %sum3 = addl %x3, %n_div_x3
        %x4 = srl %sum3, constl 1

        %n_div_x4 = udivl %n, %x4
        %sum4 = addl %x4, %n_div_x4
        %x5 = srl %sum4, constl 1

        %n_div_x5 = udivl %n, %x5
        %sum5 = addl %x5, %n_div_x5
        %x6 = srl %sum5, constl 1

        jmp @adjust(%x6)

    @adjust(l x)
        %sq = mull %x, %x
        %gt = cult %sq, %n
        %x_dec = subl %x, constl 1
        %x1 = sel %gt, %x_dec, %x

        %xp1 = addl %x1, constl 1
        %sqp1 = mull %xp1, %xp1
        %le = cule %sqp1, %n
        %res = sel %le, %xp1, %x1
        ret %res

    @ret0
        ret constl 0
}
