crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
    mov rcx, rsi
.Lbyte_loop:
    xor al, byte ptr [rdi]
    mov rdx, 8
.Lbit_loop:
    test eax, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lbit_next
.Lshift:
    shr eax, 1
.Lbit_next:
    dec rdx
    jne .Lbit_loop
    inc rdi
    dec rcx
    jne .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
