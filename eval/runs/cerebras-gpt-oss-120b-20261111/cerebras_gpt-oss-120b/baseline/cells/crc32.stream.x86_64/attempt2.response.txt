crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Lret
    mov rcx, rsi
.Lbyte:
    xor al, byte ptr [rdi]
    inc rdi
    mov ebx, 8
.Lbit:
    test eax, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lnext
.Lshift:
    shr eax, 1
.Lnext:
    dec ebx
    jnz .Lbit
    dec rcx
    jnz .Lbyte
.Lret:
    xor eax, 0xFFFFFFFF
    ret
