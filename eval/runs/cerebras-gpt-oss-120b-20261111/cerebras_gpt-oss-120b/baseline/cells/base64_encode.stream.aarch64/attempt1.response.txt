base64_encode:
    // preserve original length
    mov     x9, x1                     // x9 = n
    // main loop, process groups of 3 bytes
.Lloop:
    cmp     x1, #3
    blt     .Lafter_loop
    ldrb    w3, [x0], #1               // b0
    ldrb    w4, [x0], #1               // b1
    ldrb    w5, [x0], #1               // b2
    // combine into 24-bit value in w6
    lsl     w6, w3, #16
    lsl     w7, w4, #8
    orr     w6, w6, w7
    orr     w6, w6, w5
    // extract indices
    lsr     w8, w6, #18
    and     w8, w8, #0x3F              // idx0
    lsr     w9, w6, #12
    and     w9, w9, #0x3F              // idx1
    lsr     w10, w6, #6
    and     w10, w10, #0x3F            // idx2
    and     w11, w6, #0x3F             // idx3
    // encode idx0 in w8
    mov     w12, w8
    add     w12, w12, #65
    mov     w13, w8
    add     w13, w13, #71
    cmp     w8, #26
    csel    w12, w12, w13, lo
    mov     w13, w8
    sub     w13, w13, #4
    cmp     w8, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w8, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w8, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx1 in w9
    mov     w12, w9
    add     w12, w12, #65
    mov     w13, w9
    add     w13, w13, #71
    cmp     w9, #26
    csel    w12, w12, w13, lo
    mov     w13, w9
    sub     w13, w13, #4
    cmp     w9, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w9, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w9, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx2 in w10
    mov     w12, w10
    add     w12, w12, #65
    mov     w13, w10
    add     w13, w13, #71
    cmp     w10, #26
    csel    w12, w12, w13, lo
    mov     w13, w10
    sub     w13, w13, #4
    cmp     w10, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w10, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w10, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx3 in w11
    mov     w12, w11
    add     w12, w12, #65
    mov     w13, w11
    add     w13, w13, #71
    cmp     w11, #26
    csel    w12, w12, w13, lo
    mov     w13, w11
    sub     w13, w13, #4
    cmp     w11, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w11, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w11, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    sub     x1, x1, #3
    b       .Lloop
.Lafter_loop:
    // handle remaining bytes (0,1,2)
    cmp     x1, #0
    beq     .Lcompute_len
    cmp     x1, #1
    beq     .Ltail1
    // remaining must be 2
.Ltail2:
    ldrb    w3, [x0]                  // b0
    ldrb    w4, [x0, #1]              // b1
    // idx0 = b0 >> 2
    lsr     w5, w3, #2
    and     w5, w5, #0x3F
    // idx1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and     w6, w3, #0x3
    lsl     w6, w6, #4
    lsr     w7, w4, #4
    orr     w6, w6, w7
    and     w6, w6, #0x3F
    // idx2 = (b1 & 0xF) << 2
    and     w7, w4, #0xF
    lsl     w7, w7, #2
    and     w7, w7, #0x3F
    // encode idx0 (w5)
    mov     w12, w5
    add     w12, w12, #65
    mov     w13, w5
    add     w13, w13, #71
    cmp     w5, #26
    csel    w12, w12, w13, lo
    mov     w13, w5
    sub     w13, w13, #4
    cmp     w5, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w5, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w5, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx1 (w6)
    mov     w12, w6
    add     w12, w12, #65
    mov     w13, w6
    add     w13, w13, #71
    cmp     w6, #26
    csel    w12, w12, w13, lo
    mov     w13, w6
    sub     w13, w13, #4
    cmp     w6, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w6, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w6, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx2 (w7)
    mov     w12, w7
    add     w12, w12, #65
    mov     w13, w7
    add     w13, w13, #71
    cmp     w7, #26
    csel    w12, w12, w13, lo
    mov     w13, w7
    sub     w13, w13, #4
    cmp     w7, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w7, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w7, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // store padding '='
    mov     w12, #'='
    strb    w12, [x2], #1
    b       .Lcompute_len
.Ltail1:
    ldrb    w3, [x0]                  // b0
    // idx0 = b0 >> 2
    lsr     w4, w3, #2
    and     w4, w4, #0x3F
    // idx1 = (b0 & 0x3) << 4
    and     w5, w3, #0x3
    lsl     w5, w5, #4
    and     w5, w5, #0x3F
    // encode idx0 (w4)
    mov     w12, w4
    add     w12, w12, #65
    mov     w13, w4
    add     w13, w13, #71
    cmp     w4, #26
    csel    w12, w12, w13, lo
    mov     w13, w4
    sub     w13, w13, #4
    cmp     w4, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w4, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w4, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // encode idx1 (w5)
    mov     w12, w5
    add     w12, w12, #65
    mov     w13, w5
    add     w13, w13, #71
    cmp     w5, #26
    csel    w12, w12, w13, lo
    mov     w13, w5
    sub     w13, w13, #4
    cmp     w5, #52
    csel    w12, w12, w13, lo
    mov     w13, #'+'
    cmp     w5, #62
    csel    w12, w12, w13, eq
    mov     w13, #'/'
    cmp     w5, #63
    csel    w12, w12, w13, eq
    strb    w12, [x2], #1
    // store padding "=="
    mov     w12, #'='
    strb    w12, [x2], #1
    strb    w12, [x2], #1
    b       .Lcompute_len
.Lcompute_len:
    // result = ((n + 2) / 3) * 4
    add     x10, x9, #2
    mov     x11, #3
    udiv    x12, x10, x11
    lsl     x0, x12, #2
    ret
