base64_encode:
    // x0 = src, x1 = n, x2 = out
    mov     x9, x1                     // preserve original n
    // main loop processing 3-byte groups
.Lloop:
    cmp     x1, #3
    blt     .Lafter_loop
    ldrb    w3, [x0], #1               // b0
    ldrb    w4, [x0], #1               // b1
    ldrb    w5, [x0], #1               // b2
    // combine into 24-bit value
    lsl     w6, w3, #16
    lsl     w7, w4, #8
    orr     w6, w6, w7
    orr     w6, w6, w5
    // extract 4 indices
    lsr     w8, w6, #18
    and     w8, w8, #0x3F
    lsr     w9, w6, #12
    and     w9, w9, #0x3F
    lsr     w10, w6, #6
    and     w10, w10, #0x3F
    and     w11, w6, #0x3F
    // encode w8
    mov     w12, w8
    add     w12, w12, #65
    add     w13, w8, #71
    cmp     w8, #26
    csel    w12, w12, w13, cs
    sub     w13, w8, #4
    cmp     w8, #52
    csel    w12, w12, w13, cs
    cmp     w8, #62
    csel    w12, w12, #'+' , eq
    cmp     w8, #63
