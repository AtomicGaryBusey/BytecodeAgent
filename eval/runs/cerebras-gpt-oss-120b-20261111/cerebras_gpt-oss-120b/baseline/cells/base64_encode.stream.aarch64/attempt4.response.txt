base64_encode:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x1
.Lloop:
    cmp x1, #3
    blt .Ltail
    ldrb w4, [x0], #1
    ldrb w5, [x0], #1
    ldrb w6, [x0], #1
    mov w7, w4
    lsl w7, w7, #16
    mov w8, w5
    lsl w8, w8, #8
    orr w7, w7, w8
    orr w7, w7, w6
    ubfx w8, w7, #18, #6
    ubfx w9, w7, #12, #6
    ubfx w10, w7, #6, #6
    ubfx w11, w7, #0, #6
    // encode w8
    mov w12, w8
    add w12, w12, #65
    mov w13, w8
    add w13, w13, #71
    mov w14, w8
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w8, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w8, #63
    csel w12, w12, w15, eq
    cmp w8, #26
    csel w12, w12, w13, cs
    cmp w8, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // encode w9
    mov w12, w9
    add w12, w12, #65
    mov w13, w9
    add w13, w13, #71
    mov w14, w9
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w9, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w9, #63
    csel w12, w12, w15, eq
    cmp w9, #26
    csel w12, w12, w13, cs
    cmp w9, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // encode w10
    mov w12, w10
    add w12, w12, #65
    mov w13, w10
    add w13, w13, #71
    mov w14, w10
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w10, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w10, #63
    csel w12, w12, w15, eq
    cmp w10, #26
    csel w12, w12, w13, cs
    cmp w10, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // encode w11
    mov w12, w11
    add w12, w12, #65
    mov w13, w11
    add w13, w13, #71
    mov w14, w11
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w11, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w11, #63
    csel w12, w12, w15, eq
    cmp w11, #26
    csel w12, w12, w13, cs
    cmp w11, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    sub x1, x1, #3
    b .Lloop
.Ltail:
    cbz x1, .Ldone
    ldrb w4, [x0]               // b0
    ubfx w8, w4, #2, #6         // i0
    // encode i0
    mov w12, w8
    add w12, w12, #65
    mov w13, w8
    add w13, w13, #71
    mov w14, w8
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w8, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w8, #63
    csel w12, w12, w15, eq
    cmp w8, #26
    csel w12, w12, w13, cs
    cmp w8, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    cmp x1, #1
    beq .Ltail_one
    // two bytes case
    ldrb w5, [x0, #1]           // b1
    and w6, w4, #3
    lsl w6, w6, #4
    lsr w7, w5, #4
    orr w9, w6, w7               // i1
    and w10, w5, #0xF
    lsl w10, w10, #2            // i2
    // encode i1
    mov w12, w9
    add w12, w12, #65
    mov w13, w9
    add w13, w13, #71
    mov w14, w9
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w9, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w9, #63
    csel w12, w12, w15, eq
    cmp w9, #26
    csel w12, w12, w13, cs
    cmp w9, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // encode i2
    mov w12, w10
    add w12, w12, #65
    mov w13, w10
    add w13, w13, #71
    mov w14, w10
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w10, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w10, #63
    csel w12, w12, w15, eq
    cmp w10, #26
    csel w12, w12, w13, cs
    cmp w10, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // padding '='
    mov w12, #0x3d
    strb w12, [x2], #1
    b .Ldone
.Ltail_one:
    // one byte case: i1 = (b0 &3) <<4
    and w5, w4, #3
    lsl w9, w5, #4
    // encode i1
    mov w12, w9
    add w12, w12, #65
    mov w13, w9
    add w13, w13, #71
    mov w14, w9
    sub w14, w14, #4
    mov w15, #0x2b
    cmp w9, #62
    csel w12, w12, w15, eq
    mov w15, #0x2f
    cmp w9, #63
    csel w12, w12, w15, eq
    cmp w9, #26
    csel w12, w12, w13, cs
    cmp w9, #52
    csel w12, w12, w14, cs
    strb w12, [x2], #1
    // two padding '='
    mov w12, #0x3d
    strb w12, [x2], #1
    strb w12, [x2], #1
    b .Ldone
.Ldone:
    add x9, x19, #2
    mov x10, #3
    udiv x9, x9, x10
    lsl x0, x9, #2
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
