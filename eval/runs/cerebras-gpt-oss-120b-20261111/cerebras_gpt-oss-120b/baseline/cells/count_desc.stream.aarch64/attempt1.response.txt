count_desc:
    cmp     x1, #2
    blt     .Lret0
    sub     x5, x1, #1          // limit = n-1
    mov     x2, #0              // i = 0
    mov     x3, #0              // count = 0
.Lloop:
    ldr     x6, [x0, x2, LSL #3]    // a[i]
    add     x8, x2, #1
    ldr     x7, [x0, x8, LSL #3]    // a[i+1]
    cmp     x6, x7
    ble     .Lskipinc
    add     x3, x3, #1
.Lskipinc:
    add     x2, x2, #1
    cmp     x2, x5
    blt     .Lloop
    mov     x0, x3
    ret
.Lret0:
    mov     x0, #0
    ret
