export function l $max_byte(l %p, l %n) {
@entry
    %zero =l const 0
    %one =l const 1
    %cmp0 =w ceql %n, %zero
    jnz %cmp0, @ret0

    ; load first byte as initial max
    %max0 =l loadub %p

    %i0 =l const 1
    %n0 =l %n
    jmp @loop

@loop
    %cond =w cltl %i0, %n0
    jz %cond, @ret

    %addr =l add %p, %i0
    %v =l loadub %addr
    %gt =w csgtl %v, %max0
    %max0 =l sel %gt, %v, %max0

    %i0 =l add %i0, %one
    jmp @loop

@ret
    ret %max0

@ret0
    ret %zero
}
