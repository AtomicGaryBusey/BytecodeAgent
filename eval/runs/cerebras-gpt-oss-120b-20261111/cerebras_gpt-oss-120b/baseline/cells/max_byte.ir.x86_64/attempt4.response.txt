export function l $max_byte(l %p, l %n) {
@entry
    w %iszero = ceql %n, 0
    jnz %iszero, @ret0

    w %max = loadub %p
    l %i = add 0, 1
    jmp @loop

@loop
    w %cond = csgtl %i, %n
    jz %cond, @ret

    l %addr = add %p, %i
    w %v = loadub %addr
    w %gt = cugtw %v, %max
    w %max = sel %gt, %v, %max

    l %i = add %i, 1
    jmp @loop

@ret
    l %maxl = zextw %max
    ret %maxl

@ret0
    ret 0
}
