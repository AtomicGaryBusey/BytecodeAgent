export function l $max_byte(l %p, l %n) {
@entry
    w %iszero = ceql %n, 0
    jnz %iszero, @ret0

    l %max = loadub %p
    l %i = add 0, 1
    jmp @loop

@loop
    w %cond = cultl %i, %n
    jz %cond, @ret

    l %addr = add %p, %i
    l %v = loadub %addr
    w %gt = csgtl %v, %max
    l %max = sel %gt, %v, %max

    l %i = add %i, 1
    jmp @loop

@ret
    ret %max

@ret0
    ret 0
}
