export function l $max_byte(l %p, l %n) {
@entry
    %zero =l 0
    %one =l 1
    %cmp0 =w ceql %n, %zero
    jnz %cmp0, @ret0

    %max =l loadub %p
    %i =l 1
    jmp @loop

@loop
    %cond =w cultl %i, %n
    jz %cond, @ret

    %addr =l add %p, %i
    %v =l loadub %addr
    %gt =w csgtl %v, %max
    %max =l sel %gt, %v, %max

    %i =l add %i, %one
    jmp @loop

@ret
    ret %max

@ret0
    ret %zero
}
