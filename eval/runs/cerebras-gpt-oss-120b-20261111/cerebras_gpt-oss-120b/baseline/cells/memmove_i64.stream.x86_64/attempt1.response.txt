memmove_i64:
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    je .Ldone
    ja .Lbackward
.Lforward:
    mov rcx, rdx
    mov r8, rsi
    mov r9, rdi
.Lfwd_loop:
    mov r10, qword ptr [r8]
    mov qword ptr [r9], r10
    add r8, 8
    add r9, 8
    dec rcx
    jnz .Lfwd_loop
    jmp .Ldone
.Lbackward:
    mov rcx, rdx
    lea r8, [rsi + rcx*8 - 8]
    lea r9, [rdi + rcx*8 - 8]
.Lbwd_loop:
    mov r10, qword ptr [r8]
    mov qword ptr [r9], r10
    sub r8, 8
    sub r9, 8
    dec rcx
    jnz .Lbwd_loop
.Ldone:
    ret
