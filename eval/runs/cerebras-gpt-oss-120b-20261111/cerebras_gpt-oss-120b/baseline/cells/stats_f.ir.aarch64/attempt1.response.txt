export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %zero =l add $0, $0
    %i0 =l add $1, $0               # i = 1 after first element
    %off0 =l shll $0, $3
    %ptr0 =l add %a, %off0
    %x0 =l loadl %ptr0
    %y0 =l call %f(l %x0)
    %min0 =l %y0
    %max0 =l %y0
    %sum0 =l %y0
    %cnt0 =l andl %y0, $1

    %cond =w csltl %i0, %n
    jnz %cond, %body
    jmp %store

@body
    %off =l shll %i0, $3
    %ptr =l add %a, %off
    %x =l loadl %ptr
    %y =l call %f(l %x)

    %sum1 =l add %sum0, %y
    %oddtmp =l andl %y, $1
    %cnt1 =l add %cnt0, %oddtmp

    %lt =w csltl %y, %min0
    jnz %lt, %min_update
    jmp %min_keep

@min_update
    %min1 =l %y
    jmp %min_join

@min_keep
    %min1 =l %min0
    jmp %min_join

@min_join
    %gt =w csgtl %y, %max0
    jnz %gt, %max_update
    jmp %max_keep

@max_update
    %max1 =l %y
    jmp %max_join

@max_keep
    %max1 =l %max0
    jmp %max_join

@max_join
    %i1 =l add %i0, $1

    %cond2 =w csltl %i1, %n
    jnz %cond2, %body
    jmp %store

@store
    %out0 =l %out
    %out1 =l add %out, $8
    %out2 =l add %out, $16
    storel %min1, %out0
    storel %max1, %out1
    storel %sum1, %out2
    ret %cnt1
}
