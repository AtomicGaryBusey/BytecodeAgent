export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %min0 =l %y0
    %max0 =l %y0
    %sum0 =l %y0
    %cnt0 =l and %y0, $1
    %i0 =l add $1, $0
    jmp @loop(%i0, %min0, %max0, %sum0, %cnt0)

@loop(%i, %min, %max, %sum, %cnt)
    %cond =w csltl %i, %n
    jnz %cond, @body
    jmp @store(%min, %max, %sum, %cnt)

@body
    %off =l shl %i, $3
    %ptr =l add %a, %off
    %x =l loadl %ptr
    %y =l call %f(l %x)

    %sum1 =l add %sum, %y
    %odd =l and %y, $1
    %cnt1 =l add %cnt, %odd

    %lt =w csltl %y, %min
    jnz %lt, @min_upd
    jmp @min_keep

@min_upd
    jmp @min_merge(%y, %max, %sum1, %cnt1, %i, %y)

@min_keep
    jmp @min_merge(%min, %max, %sum1, %cnt1, %i, %y)

@min_merge(%min2, %max2, %sum2, %cnt2, %i2, %y2)
    %gt =w csgtl %y2, %max2
    jnz %gt, @max_upd
    jmp @max_keep

@max_upd
    jmp @max_merge(%min2, %y2, %sum2, %cnt2, %i2)

@max_keep
    jmp @max_merge(%min2, %max2, %sum2, %cnt2, %i2)

@max_merge(%min3, %max3, %sum3, %cnt3, %i3)
    %i1 =l add %i3, $1
    jmp @loop(%i1, %min3, %max3, %sum3, %cnt3)

@store(%min, %max, %sum, %cnt)
    %out0 =l %out
    %out1 =l add %out, $8
    %out2 =l add %out, $16
    storel %min, %out0
    storel %max, %out1
    storel %sum, %out2
    ret %cnt
}
