binary_search:
    cmp rsi, 0
    je .Lnotfound
    mov rcx, 0
    mov r8, rsi
    dec r8
.Lloop:
    cmp rcx, r8
    jg .Lnotfound
    lea r9, [rcx + r8]
    shr r9, 1
    mov rax, qword ptr [rdi + r9*8]
    cmp rax, rdx
    je .Lfound
    jl .Lless
    dec r9
    mov r8, r9
    jmp .Lloop
.Lless:
    inc r9
    mov rcx, r9
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnotfound:
    mov rax, -1
    ret
