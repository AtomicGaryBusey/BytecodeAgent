mp_mul:
    // arguments: x0=a, x1=na, x2=b, x3=nb, x4=r
    add     x5, x1, x3                // total = na + nb
    // zero r[0..total-1]
    mov     x6, #0                    // idx = 0
.Lzero_loop:
    cmp     x6, x5
    b.ge    .Lzero_done
    str     xzr, [x4, x6, lsl #3]
    add     x6, x6, #1
    b       .Lzero_loop
.Lzero_done:
    // if na==0 or nb==0 return 0
    cbz     x1, .Lreturn_zero
    cbz     x3, .Lreturn_zero
    // outer loop i = 0 .. na-1
    mov     x6, #0                    // i
.Louter:
    cmp     x6, x1
    b.ge    .Louter_done
    ldr     x8, [x0, x6, lsl #3]      // a[i]
    mov     x12, #0                   // carry = 0
    mov     x7, #0                    // j
.Linner:
    cmp     x7, x3
    b.ge    .Linner_done
    ldr     x9, [x2, x7, lsl #3]      // b[j]
    mul     x10, x8, x9               // low 64 bits
    umulh   x11, x8, x9               // high 64 bits
    add     x13, x6, x7
    lsl     x13, x13, #3              // offset = (i+j)*8
    ldr     x14, [x4, x13]            // r[i+j]
    adds    x15, x14, x10            // sum = r + low
    cset    x16, cs                  // carry1
    adcs    x15, x15, x12            // sum += carry_in
    cset    x17, cs                  // carry2
    adds    x12, x11, x16            // newcarry = high + carry1
    adcs    x12, x12, x17            // + carry2
    str     x15, [x4, x13]            // store result limb
    add     x7, x7, #1
    b       .Linner
.Linner_done:
    // store final carry to r[i+nb]
    add     x13, x6, x3
    lsl     x13, x13, #3
    str     x12, [x4, x13]
    add     x6, x6, #1
    b       .Louter
.Louter_done:
    // find most significant non‑zero limb
    mov     x6, x5
    subs    x6, x6, #1               // x6 = total-1
    blt     .Lall_zero               // total == 0
.Ltrim:
    ldr     x0, [x4, x6, lsl #3]
    cbz     x0, .Lnext
    add     x0, x6, #1               // return index+1
    b       .Lreturn
.Lnext:
    sub     x6, x6, #1
    bge     .Ltrim
.Lall_zero:
    mov     x0, #0
.Lreturn:
    ret
.Lreturn_zero:
    mov     x0, #0
    ret
