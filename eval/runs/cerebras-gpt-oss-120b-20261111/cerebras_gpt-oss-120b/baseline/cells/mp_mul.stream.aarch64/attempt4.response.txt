mp_mul:
    // x0=a, x1=na, x2=b, x3=nb, x4=r
    add     x5, x1, x3                // x5 = total = na+nb
    // zero r[0..total-1]
    mov     x6, #0                    // idx = 0
.Lzero:
    cmp     x6, x5
    b.ge    .Lzero_done
    str     xzr, [x4, x6, lsl #3]
    add     x6, x6, #1
    b       .Lzero
.Lzero_done:
    // if na==0 or nb==0 return 0 (r already zero)
    cbz     x1, .Lret0
    cbz     x3, .Lret0

    // outer loop i = 0 .. na-1
    mov     x7, #0                    // i
.Louter:
    cmp     x7, x1
    b.ge    .Louter_done
    ldr     x8, [x0, x7, lsl #3]      // a[i]
    mov     x9, #0                    // carry = 0
    mov     x10, #0                   // j
.Linner:
    cmp     x10, x3
    b.ge    .Linner_done
    ldr     x11, [x2, x10, lsl #3]    // b[j]
    mul     x12, x8, x11              // low 64
    umulh   x13, x8, x11              // high 64
    add     x14, x7, x10
    lsl     x14, x14, #3              // offset = (i+j)*8
    ldr     x15, [x4, x14]            // r[i+j]
    adds    x16, x15, x12            // r + low, set C1
    adcs    x16, x16, x9             // + carry_in, set C2
    // new carry = high + C1 + C2
    cset    x17, cs                  // C2 -> x17 (0/1)
    cset    x18, cs                  // after adds, C1 is in flags; need capture before adcs? capture now:
    // actually capture C1 before adcs:
    // redo: use separate flags
    // we'll recompute:
    adds    x15, x15, x12            // r+low
    cset    x18, cs                  // C1
    adcs    x16, x15, x9             // + carry_in
    cset    x17, cs                  // C2
    adds    x9, x13, x18             // high + C1
    adcs    x9, x9, x17              // + C2  => new carry in x9
    str     x16, [x4, x14]           // store result limb
    add     x10, x10, #1
    b       .Linner
.Linner_done:
    // propagate remaining carry starting at k = i+nb
    add     x14, x7, x3
    lsl     x14, x14, #3              // offset = (i+nb)*8
.Lcarry:
    cbz     x9, .Lcarry_done
    ldr     x15, [x4, x14]
    adds    x16, x15, x9
    cset    x9, cs                   // new carry
    str     x16, [x4, x14]
    add     x14, x14, #8
    b       .Lcarry
.Lcarry_done:
    add     x7, x7, #1
    b       .Louter
.Louter_done:
    // find most significant non‑zero limb among total limbs
    mov     x6, x5                    // x6 = total
    subs    x6, x6, #1                // idx = total-1
    b.lt    .Lall_zero
.Ltrim:
    ldr     x0, [x4, x6, lsl #3]
    cbnz    x0, .Lfound
    subs    x6, x6, #1
    bge     .Ltrim
.Lall_zero:
    mov     x0, #0
    ret
.Lfound:
    add     x0, x6, #1
    ret
.Lret0:
    mov     x0, #0
    ret
