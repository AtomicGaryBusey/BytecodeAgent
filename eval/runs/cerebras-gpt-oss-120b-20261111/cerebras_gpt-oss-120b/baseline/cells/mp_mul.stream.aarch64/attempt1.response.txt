mp_mul:
    // Prologue: save fp, lr and callee‑saved registers
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    // Move arguments to callee‑saved registers
    mov x19, x0          // a
    mov x20, x1          // na
    mov x21, x2          // b
    mov x22, x3          // nb
    mov x23, x4          // r

    // total = na + nb
    add x9,  x20, x22    // x9 = total limbs (na+nb)

    // Zero r[0..total-1]
    mov x25, #0
.Lzero_loop:
    cmp x25, x9
    b.ge .Lzero_done
    str xzr, [x23, x25, lsl #3]
    add x25, x25, #1
    b .Lzero_loop
.Lzero_done:

    // If na==0 or nb==0, result is zero
    cbz x20, .Lreturn
    cbz x22, .Lreturn

    // Outer loop over i = 0 .. na-1
    mov x26, #0                 // i
.Louter:
    cmp x26, x20
    b.ge .Louter_done

    ldr x9, [x19, x26, lsl #3] // x9 = a[i]

    mov x27, #0                // carry = 0
    mov x28, #0                // j

    // Inner loop over j = 0 .. nb-1
.Linner:
    cmp x28, x22
    b.ge .Linner_done

    ldr x10, [x21, x28, lsl #3]   // x10 = b[j]

    mul  x11, x9, x10            // low 64 bits
    umulh x12, x9, x10           // high 64 bits

    add  x13, x26, x28
    lsl  x13, x13, #3
    ldr  x14, [x23, x13]         // r[i+j]

    adds x15, x14, x11           // sum = r + low
    cset x16, cs                 // carry1 = overflow?
    adds x15, x15, x27           // sum += carry
    cset x17, cs                 // carry2 = overflow?

    adds x18, x12, x16           // newcarry = high + carry1
    adc  x18, x18, x17           // + carry2 (and possible overflow)

    str  x15, [x23, x13]         // store result limb
    mov  x27, x18                // carry = newcarry

    add  x28, x28, #1
    b .Linner
.Linner_done:
    // Store final carry to r[i+nb]
    add  x13, x26, x22
    lsl  x13, x13, #3
    str  x27, [x23, x13]

    add  x26, x26, #1
    b .Louter
.Louter_done:

    // Find most significant non‑zero limb
    mov x25, x9
    subs x25, x25, #1            // start at total-1
    b.lt .Lall_zero              // total == 0

.Ltrim:
    ldr x0, [x23, x25, lsl #3]
    cbz x0, .Lnext
    add x0, x25, #1              // return index+1
    b .Lreturn
.Lnext:
    sub x25, x25, #1
    bge .Ltrim
.Lall_zero:
    mov x0, #0

.Lreturn:
    // Epilogue: restore callee‑saved registers and return
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
