export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
    %src_end =l add %src %n
    %src_ptr =l %src
    %out_ptr =l %out
    %written =l 0
    %neg1 =l const -1
    %neg2 =l const -2
    jmp @check

@check:
    %has_src =w cslt %src_ptr %src_end
    jnz %has_src, @read_header, @done

@read_header:
    %h =w loadsb %src_ptr
    %src_ptr =l add %src_ptr 1
    %h_ge0 =w csge %h 0
    %is_neg128 =w ceql %h -128
    jnz %h_ge0, @literal, @neg_check

@neg_check:
    jnz %is_neg128, @check, @repeat

@literal:
    %need =w add %h 1
    %need_l =l zextw %need
    %src_needed_end =l add %src_ptr %need_l
    %src_ok =w csle %src_needed_end %src_end
    jnz %src_ok, @literal_src_ok, @error_src

@literal_src_ok:
    %new_total =l add %written %need_l
    %cap_ok =w csle %new_total %cap
    jnz %cap_ok, @literal_copy, @literal_overflow

@literal_copy:
    %i =l 0
    jmp @literal_copy_loop

@literal_copy_loop:
    %cond =w cslt %i %need_l
    jnz %cond, @literal_copy_body, @literal_copy_done

@literal_copy_body:
    %src_off =l add %src_ptr %i
    %b =b loadub %src_off
    %out_off =l add %out_ptr %i
    storeb %b, %out_off
    %i =l add %i 1
    jmp @literal_copy_loop

@literal_copy_done:
    %src_ptr =l add %src_ptr %need_l
    %out_ptr =l add %out_ptr %need_l
    %written =l add %written %need_l
    jmp @check

@literal_overflow:
    %remain =l sub %cap %written
    %i =l 0
    jmp @literal_overflow_loop

@literal_overflow_loop:
    %cond =w cslt %i %remain
    jnz %cond, @literal_overflow_body, @literal_overflow_done

@literal_overflow_body:
    %src_off =l add %src_ptr %i
    %b =b loadub %src_off
    %out_off =l add %out_ptr %i
    storeb %b, %out_off
    %i =l add %i 1
    jmp @literal_overflow_loop

@literal_overflow_done:
    ret %neg1

@repeat:
    %src_has =w cslt %src_ptr %src_end
    jnz %src_has, @repeat_have, @error_src

@repeat_have:
    %b =b loadub %src_ptr
    %src_ptr =l add %src_ptr 1
    %neg_h =w sub 0 %h
    %repeat =w add %neg_h 1
    %repeat_l =l zextw %repeat
    %new_total =l add %written %repeat_l
    %cap_ok =w csle %new_total %cap
    jnz %cap_ok, @repeat_copy, @repeat_overflow

@repeat_copy:
    %i =l 0
    jmp @repeat_copy_loop

@repeat_copy_loop:
    %cond =w cslt %i %repeat_l
    jnz %cond, @repeat_copy_body, @repeat_copy_done

@repeat_copy_body:
    %out_off =l add %out_ptr %i
    storeb %b, %out_off
    %i =l add %i 1
    jmp @repeat_copy_loop

@repeat_copy_done:
    %out_ptr =l add %out_ptr %repeat_l
    %written =l add %written %repeat_l
    jmp @check

@repeat_overflow:
    %remain =l sub %cap %written
    %i =l 0
    jmp @repeat_overflow_loop

@repeat_overflow_loop:
    %cond =w cslt %i %remain
    jnz %cond, @repeat_overflow_body, @repeat_overflow_done

@repeat_overflow_body:
    %out_off =l add %out_ptr %i
    storeb %b, %out_off
    %i =l add %i 1
    jmp @repeat_overflow_loop

@repeat_overflow_done:
    ret %neg1

@error_src:
    ret %neg2

@done:
    ret %written
}
