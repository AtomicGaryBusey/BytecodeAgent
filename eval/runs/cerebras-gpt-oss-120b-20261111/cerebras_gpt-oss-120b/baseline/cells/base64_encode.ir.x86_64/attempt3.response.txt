export function l $base64_encode(l, l, l) {
@entry
    %src =l arg0
    %n =l arg1
    %out =l arg2
    %i =l const 0
    %dst =l %out
    %ret =l const 0
    goto %loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, %process
    goto %done

@process
    %remain =l sub %n, %i
    %cond_full =w csltl %remain, 3
    jnz %cond_full, %partial   // remain < 3 -> partial
    // full 3‑byte block
    %addr0 =l add %src, %i
    %b0 =l loadub %addr0
    %addr1 =l add %i, 1
    %addr1 =l add %src, %addr1
    %b1 =l loadub %addr1
    %addr2 =l add %i, 2
    %addr2 =l add %src, %addr2
    %b2 =l loadub %addr2

    %val0 =l shl %b0, 16
    %val1 =l shl %b1, 8
    %val =l add %val0, %val1
    %val =l add %val, %b2

    %i0 =l shr %val, 18
    %i1 =l shr %val, 12
    %i1 =l and %i1, 63
    %i2 =l shr %val, 6
    %i2 =l and %i2, 63
    %i3 =l and %val, 63

    %c0 =l call $map_char(%i0)
    storeb %c0, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c1 =l call $map_char(%i1)
    storeb %c1, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c2 =l call $map_char(%i2)
    storeb %c2, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c3 =l call $map_char(%i3)
    storeb %c3, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %i =l add %i, 3
    goto %loop

@partial
    %cond1 =w ceql %remain, 1
    jnz %cond1, %partial1   // remain == 1
    // remain == 2
    %addr0 =l add %src, %i
    %b0 =l loadub %addr0
    %addr1 =l add %i, 1
    %addr1 =l add %src, %addr1
    %b1 =l loadub %addr1

    %i0 =l shr %b0, 2
    %tmp =l and %b0, 3
    %i1 =l shl %tmp, 4
    %tmp2 =l shr %b1, 4
    %i1 =l add %i1, %tmp2
    %tmp3 =l and %b1, 15
    %i2 =l shl %tmp3, 2

    %c0 =l call $map_char(%i0)
    storeb %c0, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c1 =l call $map_char(%i1)
    storeb %c1, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c2 =l call $map_char(%i2)
    storeb %c2, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %pad =l const 61
    storeb %pad, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    goto %done

@partial1
    // remain == 1
    %addr0 =l add %src, %i
    %b0 =l loadub %addr0

    %i0 =l shr %b0, 2
    %tmp =l and %b0, 3
    %i1 =l shl %tmp, 4

    %c0 =l call $map_char(%i0)
    storeb %c0, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %c1 =l call $map_char(%i1)
    storeb %c1, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %pad =l const 61
    storeb %pad, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1
    storeb %pad, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    goto %done

@done
    ret %ret
}

export function l $map_char(l) {
@mc_entry
    %i =l arg0
    %c =l const 0
    %t0 =w csltl %i, 26
    jnz %t0, %mc_a
    %t1 =w csltl %i, 52
    jnz %t1, %mc_b
    %t2 =w csltl %i, 62
    jnz %t2, %mc_c
    %t3 =w ceql %i, 62
    jnz %t3, %mc_plus
    // i == 63
    %c =l const 47
    ret %c
@mc_a
    %c =l add %i, 65
    ret %c
@mc_b
    %c =l add %i, 71
    ret %c
@mc_c
    %c =l sub %i, 4
    ret %c
@mc_plus
    %c =l const 43
    ret %c
}
