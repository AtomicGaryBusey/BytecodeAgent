export function l $base64_encode(l src, l n, l out) {
@entry
    %i =l const 0
    %dst =l out
    %ret =l const 0
    goto %loop

@loop
    %remain =l sub n %i
    %cond_done =w csltl %remain, 1
    jnz %cond_done, %done

    %cond_full =w csltl %remain, 3
    jnz %cond_full, %partial

    // full 3-byte block
    %b0 =l loadub %src, %i
    %addr1 =l add %i, 1
    %b1 =l loadub %src, %addr1
    %addr2 =l add %i, 2
    %b2 =l loadub %src, %addr2

    %val0 =l shl %b0, 16
    %val1 =l shl %b1, 8
    %val =l add %val0, %val1
    %val =l add %val, %b2

    %i0 =l shr %val, 18
    %i1 =l shr %val, 12
    %i1 =l and %i1, 63
    %i2 =l shr %val, 6
    %i2 =l and %i2, 63
    %i3 =l and %val, 63

    // output i0
    %t0 =w csltl %i0, 26
    jnz %t0, %store0_a
    %t1 =w csltl %i0, 52
    jnz %t1, %store0_b
    %t2 =w csltl %i0, 62
    jnz %t2, %store0_c
    %t3 =w ceql %i0, 62
    jnz %t3, %store0_plus
    %c =l const 47
    goto %store0_store
@store0_a
    %c =l add %i0, 65
    goto %store0_store
@store0_b
    %c =l add %i0, 71
    goto %store0_store
@store0_c
    %c =l sub %i0, 4
    goto %store0_store
@store0_plus
    %c =l const 43
@store0_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i1
    %t0 =w csltl %i1, 26
    jnz %t0, %store1_a
    %t1 =w csltl %i1, 52
    jnz %t1, %store1_b
    %t2 =w csltl %i1, 62
    jnz %t2, %store1_c
    %t3 =w ceql %i1, 62
    jnz %t3, %store1_plus
    %c =l const 47
    goto %store1_store
@store1_a
    %c =l add %i1, 65
    goto %store1_store
@store1_b
    %c =l add %i1, 71
    goto %store1_store
@store1_c
    %c =l sub %i1, 4
    goto %store1_store
@store1_plus
    %c =l const 43
@store1_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i2
    %t0 =w csltl %i2, 26
    jnz %t0, %store2_a
    %t1 =w csltl %i2, 52
    jnz %t1, %store2_b
    %t2 =w csltl %i2, 62
    jnz %t2, %store2_c
    %t3 =w ceql %i2, 62
    jnz %t3, %store2_plus
    %c =l const 47
    goto %store2_store
@store2_a
    %c =l add %i2, 65
    goto %store2_store
@store2_b
    %c =l add %i2, 71
    goto %store2_store
@store2_c
    %c =l sub %i2, 4
    goto %store2_store
@store2_plus
    %c =l const 43
@store2_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i3
    %t0 =w csltl %i3, 26
    jnz %t0, %store3_a
    %t1 =w csltl %i3, 52
    jnz %t1, %store3_b
    %t2 =w csltl %i3, 62
    jnz %t2, %store3_c
    %t3 =w ceql %i3, 62
    jnz %t3, %store3_plus
    %c =l const 47
    goto %store3_store
@store3_a
    %c =l add %i3, 65
    goto %store3_store
@store3_b
    %c =l add %i3, 71
    goto %store3_store
@store3_c
    %c =l sub %i3, 4
    goto %store3_store
@store3_plus
    %c =l const 43
@store3_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    %i =l add %i, 3
    goto %loop

@partial
    %cond1 =w ceql %remain, 1
    jnz %cond1, %partial1

    // remain == 2
    %b0 =l loadub %src, %i
    %addr1 =l add %i, 1
    %b1 =l loadub %src, %addr1

    %i0 =l shr %b0, 2
    %tmp =l and %b0, 3
    %i1 =l shl %tmp, 4
    %tmp2 =l shr %b1, 4
    %i1 =l add %i1, %tmp2
    %tmp3 =l and %b1, 15
    %i2 =l shl %tmp3, 2

    // output i0
    %t0 =w csltl %i0, 26
    jnz %t0, %p2_store0_a
    %t1 =w csltl %i0, 52
    jnz %t1, %p2_store0_b
    %t2 =w csltl %i0, 62
    jnz %t2, %p2_store0_c
    %t3 =w ceql %i0, 62
    jnz %t3, %p2_store0_plus
    %c =l const 47
    goto %p2_store0_store
@p2_store0_a
    %c =l add %i0, 65
    goto %p2_store0_store
@p2_store0_b
    %c =l add %i0, 71
    goto %p2_store0_store
@p2_store0_c
    %c =l sub %i0, 4
    goto %p2_store0_store
@p2_store0_plus
    %c =l const 43
@p2_store0_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i1
    %t0 =w csltl %i1, 26
    jnz %t0, %p2_store1_a
    %t1 =w csltl %i1, 52
    jnz %t1, %p2_store1_b
    %t2 =w csltl %i1, 62
    jnz %t2, %p2_store1_c
    %t3 =w ceql %i1, 62
    jnz %t3, %p2_store1_plus
    %c =l const 47
    goto %p2_store1_store
@p2_store1_a
    %c =l add %i1, 65
    goto %p2_store1_store
@p2_store1_b
    %c =l add %i1, 71
    goto %p2_store1_store
@p2_store1_c
    %c =l sub %i1, 4
    goto %p2_store1_store
@p2_store1_plus
    %c =l const 43
@p2_store1_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i2
    %t0 =w csltl %i2, 26
    jnz %t0, %p2_store2_a
    %t1 =w csltl %i2, 52
    jnz %t1, %p2_store2_b
    %t2 =w csltl %i2, 62
    jnz %t2, %p2_store2_c
    %t3 =w ceql %i2, 62
    jnz %t3, %p2_store2_plus
    %c =l const 47
    goto %p2_store2_store
@p2_store2_a
    %c =l add %i2, 65
    goto %p2_store2_store
@p2_store2_b
    %c =l add %i2, 71
    goto %p2_store2_store
@p2_store2_c
    %c =l sub %i2, 4
    goto %p2_store2_store
@p2_store2_plus
    %c =l const 43
@p2_store2_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // padding '='
    storeb %b, %dst   // %b is constant '=' (ASCII 61)
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    goto %done

@partial1
    // remain == 1
    %b0 =l loadub %src, %i

    %i0 =l shr %b0, 2
    %tmp =l and %b0, 3
    %i1 =l shl %tmp, 4

    // output i0
    %t0 =w csltl %i0, 26
    jnz %t0, %p1_store0_a
    %t1 =w csltl %i0, 52
    jnz %t1, %p1_store0_b
    %t2 =w csltl %i0, 62
    jnz %t2, %p1_store0_c
    %t3 =w ceql %i0, 62
    jnz %t3, %p1_store0_plus
    %c =l const 47
    goto %p1_store0_store
@p1_store0_a
    %c =l add %i0, 65
    goto %p1_store0_store
@p1_store0_b
    %c =l add %i0, 71
    goto %p1_store0_store
@p1_store0_c
    %c =l sub %i0, 4
    goto %p1_store0_store
@p1_store0_plus
    %c =l const 43
@p1_store0_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // output i1
    %t0 =w csltl %i1, 26
    jnz %t0, %p1_store1_a
    %t1 =w csltl %i1, 52
    jnz %t1, %p1_store1_b
    %t2 =w csltl %i1, 62
    jnz %t2, %p1_store1_c
    %t3 =w ceql %i1, 62
    jnz %t3, %p1_store1_plus
    %c =l const 47
    goto %p1_store1_store
@p1_store1_a
    %c =l add %i1, 65
    goto %p1_store1_store
@p1_store1_b
    %c =l add %i1, 71
    goto %p1_store1_store
@p1_store1_c
    %c =l sub %i1, 4
    goto %p1_store1_store
@p1_store1_plus
    %c =l const 43
@p1_store1_store
    storeb %c, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    // padding "=="
    storeb %b, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1
    storeb %b, %dst
    %dst =l add %dst, 1
    %ret =l add %ret, 1

    goto %done

@done
    ret %ret
}
