export function l $bmin_empty(l a, l n) {
@entry:
  %cond = ceql %n, 0
  jnz %cond, @ret_max
  %min_start = loadl %a
  %i_start = 1
  jmp @loop_cond

@loop_cond:
  %i0 = phi l %i_start, %i1
  %min0 = phi l %min_start, %min2
  %cmp = csltl %i0, %n
  jnz %cmp, @loop_body
  jmp @ret_min

@loop_body:
  %off = shll %i0, 3
  %addr = addl %a, %off
  %val = loadl %addr
  %lt = csltl %val, %min0
  jnz %lt, @take_val
  jmp @keep_min

@take_val:
  %min2 = copy %val
  jmp @inc_i

@keep_min:
  %min2 = copy %min0
  jmp @inc_i

@inc_i:
  %i1 = addl %i0, 1
  jmp @loop_cond

@ret_min:
  ret %min0

@ret_max:
  ret 9223372036854775807
}
