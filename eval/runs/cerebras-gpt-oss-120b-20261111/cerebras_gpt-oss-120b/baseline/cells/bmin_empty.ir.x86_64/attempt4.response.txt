export function l bmin_empty(l a, l n) {
@entry:
    %cond = ceql %n, 0
    jnz %cond, @ret_max
    %min0 = loadl %a
    %i0 = copy 1
    jmp @loop_cond

@loop_cond:
    %i = phi l %i0, @entry, %i_next, @inc
    %min = phi l %min0, @entry, %min_next, @inc
    %cmp = csltl %i, %n
    jnz %cmp, @loop_body
    jmp @ret_min

@loop_body:
    %off = shll %i, 3
    %addr = addl %a, %off
    %val = loadl %addr
    %lt = csltl %val, %min
    jnz %lt, @set_min
    %min_next = copy %min
    jmp @inc

@set_min:
    %min_next = copy %val
    jmp @inc

@inc:
    %i_next = addl %i, 1
    jmp @loop_cond

@ret_min:
    ret %min

@ret_max:
    ret 9223372036854775807
}
