export function q $count_desc(q a, q n) {
@entry:
  %a = arg
  %n = arg
  %lt2 = csltq %n, 2
  jnz %lt2, @ret0
  %limit = sub %n, 1
  jmp @loop

@loop:
  %i = phi 0, @entry, %i_next, @next
  %cnt = phi 0, @entry, %cnt_next, @next
  %c = csltq %i, %limit
  jz %c, @end

  %i8 = mul %i, 8
  %addr1 = add %a, %i8
  %v1 = loadq %addr1

  %i1 = add %i, 1
  %i1_8 = mul %i1, 8
  %addr2 = add %a, %i1_8
  %v2 = loadq %addr2

  %desc = csltq %v2, %v1
  jnz %desc, @inc
  %cnt_next = %cnt
  jmp @next

@inc:
  %cnt_next = add %cnt, 1
  jmp @next

@next:
  %i_next = add %i, 1
  jmp @loop

@end:
  ret %cnt

@ret0:
  ret 0
}
