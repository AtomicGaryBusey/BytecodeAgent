export function l $count_desc(l a, l n) {
@entry:
  %n = arg n
  %cond_lt2 = csltl %n, 2
  jnz %cond_lt2, @ret0

  %limit = subl %n, 1
  %i = 0
  %cnt = 0
@loop_cond:
  %c = csltl %i, %limit
  jz %c, @end
@loop_body:
  %i8 = mul %i, 8
  %addr1 = addl a, %i8
  %v1 = loadl %addr1
  %i1 = addl %i, 1
  %i1_8 = mul %i1, 8
  %addr2 = addl a, %i1_8
  %v2 = loadl %addr2
  %desc = csltl %v2, %v1
  jnz %desc, @inc
  jmp @next
@inc:
  %cnt_inc = addl %cnt, 1
  jmp @next
@next:
  %cnt = phi %cnt, %cnt_inc
  %i = addl %i, 1
  jmp @loop_cond
@ret0:
  ret 0
@end:
  ret %cnt
}
