export function l $count_desc(l a, l n) {
@entry:
  %a = arg
  %n = arg
  %lt2 = csltl %n, 2
  jnz %lt2, @ret0

  %limit = sub %n, 1
  %zero = const 0
  jmp @loop

@loop:
  %i = phi %zero, @entry, %i_next, @next
  %cnt = phi %zero, @entry, %cnt_next, @next
  %c = csltl %i, %limit
  jz %c, @end

  %i8 = mul %i, 8
  %addr1 = add %a, %i8
  %v1 = loadl %addr1

  %i1 = add %i, 1
  %i1_8 = mul %i1, 8
  %addr2 = add %a, %i1_8
  %v2 = loadl %addr2

  %desc = csltl %v2, %v1
  jnz %desc, @inc

  %cnt_next = %cnt
  jmp @next

@inc:
  %cnt_next = add %cnt, 1
  jmp @next

@next:
  %i_next = add %i, 1
  jmp @loop

@end:
  ret %cnt

@ret0:
  ret 0
}
