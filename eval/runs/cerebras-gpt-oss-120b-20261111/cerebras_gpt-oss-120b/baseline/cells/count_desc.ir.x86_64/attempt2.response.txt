export function l $count_desc(l a, l n) {
@entry:
  %a = arg a
  %n = arg n
  %lt2 = csltl %n, 2
  jnz %lt2, @ret0
  jmp @loop

@loop:
  %i = phi 0, @entry, %i1, @next
  %cnt = phi 0, @entry, %cnt1, @next
  %limit = subl %n, 1
  %c = csltl %i, %limit
  jz %c, @end

  %i8 = mul %i, 8
  %addr1 = addl %a, %i8
  %v1 = loadl %addr1
  %i1tmp = addl %i, 1
  %i1_8 = mul %i1tmp, 8
  %addr2 = addl %a, %i1_8
  %v2 = loadl %addr2
  %desc = csltl %v2, %v1
  jnz %desc, @inc

  %cnt1 = %cnt
  jmp @next

@inc:
  %cnt1 = addl %cnt, 1
  jmp @next

@next:
  %i1 = addl %i, 1
  jmp @loop

@end:
  ret %cnt

@ret0:
  ret 0
}
