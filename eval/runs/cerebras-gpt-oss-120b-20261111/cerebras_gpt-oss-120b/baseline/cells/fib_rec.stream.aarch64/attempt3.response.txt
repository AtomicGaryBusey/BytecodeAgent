fib_rec:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cmp x0, #1
    ble .Lbase
    sub sp, sp, #16
    str x0, [sp, #8]          // save n
    sub x0, x0, #1
    bl fib_rec
    str x0, [sp, #0]          // save result1
    ldr x0, [sp, #8]          // reload n
    sub x0, x0, #2
    bl fib_rec
    ldr x1, [sp, #0]
    add x0, x0, x1
    add sp, sp, #16
    b .Lret
.Lbase:
    // x0 already holds 0 or 1
.Lret:
    ldp x29, x30, [sp], #16
    ret
