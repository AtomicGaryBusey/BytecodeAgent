map_sum:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rdx
    mov r13, rsi
    xor r14, r14
    test r13, r13
    jz .Ldone
.Lloop:
    mov rax, qword ptr [rbx]
    mov rdi, rax
    call r12
    add r14, rax
    add rbx, 8
    dec r13
    jne .Lloop
.Ldone:
    mov rax, r14
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
