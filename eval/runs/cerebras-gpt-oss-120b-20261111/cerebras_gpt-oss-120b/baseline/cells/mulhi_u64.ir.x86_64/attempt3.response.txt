export function l $mulhi_u64(l %a, l %b) {
@start
%a_lo = and %a, 0xffffffff
%a_hi = shr %a, 32
%b_lo = and %b, 0xffffffff
%b_hi = shr %b, 32

%p0 = mul %a_lo, %b_lo
%carry0 = shr %p0, 32

%p1 = mul %a_lo, %b_hi
%p2 = mul %a_hi, %b_lo

%tmp = add %p1, %p2
%ov1 = csltul %tmp, %p1
%carry1 = zextw %ov1

%mid = add %tmp, %carry0
%ov2 = csltul %mid, %tmp
%carry2 = zextw %ov2

%total_carry = add %carry1, %carry2

%mid_hi = shr %mid, 32

%hi = mul %a_hi, %b_hi
%hi = add %hi, %mid_hi
%hi = add %hi, %total_carry

ret %hi
}
