export function l $mulhi_u64(l %a, l %b) {
@start
%a_lo = and.l %a, 0xffffffff
%a_hi = shr.l %a, 32
%b_lo = and.l %b, 0xffffffff
%b_hi = shr.l %b, 32

%p0 = mul.l %a_lo, %b_lo
%carry0 = shr.l %p0, 32

%p1 = mul.l %a_lo, %b_hi
%p2 = mul.l %a_hi, %b_lo

%tmp = add.l %p1, %p2
%ov1 = csltul.l %tmp, %p1
%carry1 = zextw %ov1

%mid = add.l %tmp, %carry0
%ov2 = csltul.l %mid, %tmp
%carry2 = zextw %ov2

%total_carry = add.l %carry1, %carry2

%mid_hi = shr.l %mid, 32

%hi = mul.l %a_hi, %b_hi
%hi = add.l %hi, %mid_hi
%hi = add.l %hi, %total_carry

ret %hi
}
