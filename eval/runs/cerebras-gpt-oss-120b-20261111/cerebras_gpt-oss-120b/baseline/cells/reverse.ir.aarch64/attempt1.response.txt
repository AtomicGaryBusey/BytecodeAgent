export function l $reverse(l a, l n) {
@entry
  %half = sarl l n, 1
  %i = copy l 0
  %ptr_i = copy l a
  %tmp = sub l n, 1
  %offset = mul l %tmp, 8
  %ptr_j = add l a, %offset
  jmp @loop

@loop
  %cond = csltl l %i, %half
  jnz %cond, @body
  jmp @ret0

@body
  %vi = loadl l %ptr_i
  %vj = loadl l %ptr_j
  storel l %vj, %ptr_i
  storel l %vi, %ptr_j
  %ptr_i = add l %ptr_i, 8
  %ptr_j = sub l %ptr_j, 8
  %i = add l %i, 1
  jmp @loop

@ret0
  ret l 0
}
