export function l $reverse(l %a, l %n) {
@entry
  %i0 = copy 0
  %tmp = sub %n, 1
  %off = mul %tmp, 8
  %ptr_j0 = add %a, %off
  %ptr_i0 = copy %a
  %half = sarl %n, 1
  jmp @loop(%i0, %ptr_i0, %ptr_j0, %half)

@loop(%i, %ptr_i, %ptr_j, %half)
  %cond = csltl %i, %half
  jnz %cond, @body(%i, %ptr_i, %ptr_j, %half)
  jmp @ret0

@body(%i, %ptr_i, %ptr_j, %half)
  %vi = loadl %ptr_i
  %vj = loadl %ptr_j
  storel %vj, %ptr_i
  storel %vi, %ptr_j
  %i1 = add %i, 1
  %ptr_i1 = add %ptr_i, 8
  %ptr_j1 = sub %ptr_j, 8
  jmp @loop(%i1, %ptr_i1, %ptr_j1, %half)

@ret0
  ret l 0
}
