export function l $reverse(l, l) {
@entry
  %a = copy l %0
  %n = copy l %1
  %half = sarl l %n, 1
  %i0 = copy l 0
  %ptr_i0 = copy l %a
  %tmp = sub l %n, 1
  %offset = mul l %tmp, 8
  %ptr_j0 = add l %a, %offset
  jmp @loop(%i0, %ptr_i0, %ptr_j0, %half)

@loop(%i, %ptr_i, %ptr_j, %half)
  %cond = csltl l %i, %half
  jnz %cond, @body(%i, %ptr_i, %ptr_j, %half)
  jmp @ret0

@body(%i, %ptr_i, %ptr_j, %half)
  %vi = loadl l %ptr_i
  %vj = loadl l %ptr_j
  storel l %vj, %ptr_i
  storel l %vi, %ptr_j
  %i1 = add l %i, 1
  %ptr_i1 = add l %ptr_i, 8
  %ptr_j1 = sub l %ptr_j, 8
  jmp @loop(%i1, %ptr_i1, %ptr_j1, %half)

@ret0
  ret l 0
}
