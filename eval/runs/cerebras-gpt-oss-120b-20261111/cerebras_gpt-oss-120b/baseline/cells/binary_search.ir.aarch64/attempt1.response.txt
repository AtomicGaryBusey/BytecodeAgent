export function l $binary_search(l a, l n, l key)
{
    @entry
    %zero = constl 0
    %n_eq0 = ceql %n, %zero
    jnz %n_eq0, @ret_minus1

    %n1 = subl %n, 1
    jmp @loop(0, %n1)

    @loop(%lo, %hi)
    %cond = csltl %lo, %hi
    jnz %cond, @mid(%lo, %hi)
    jmp @after(%lo)

    @mid(%lo, %hi)
    %sum = addl %lo, %hi
    %mid = srl %sum, 1
    %mid8 = shll %mid, 3
    %addr = addl %a, %mid8
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @ret_mid(%mid)
    %lt = csltl %val, %key
    jnz %lt, @set_lo(%mid, %hi)
    jmp @set_hi(%mid, %lo)

    @set_lo(%mid, %hi)
    %mid1 = addl %mid, 1
    jmp @loop(%mid1, %hi)

    @set_hi(%mid, %lo)
    %mid1 = subl %mid, 1
    jmp @loop(%lo, %mid1)

    @after(%lo)
    %cond2 = csltl %lo, %n
    jnz %cond2, @check(%lo)
    jmp @ret_minus1

    @check(%lo)
    %addr2 = addl %a, shll %lo, 3
    %val2 = loadl %addr2
    %eq2 = ceql %val2, %key
    jnz %eq2, @ret_lo(%lo)
    jmp @ret_minus1

    @ret_mid(%mid)
    ret %mid

    @ret_lo(%lo)
    ret %lo

    @ret_minus1
    %minus1 = constl -1
    ret %minus1
}
