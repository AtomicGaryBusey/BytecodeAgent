export function l $binary_search(l a, l n, l key)
{
    @entry
    %zero = constl 0
    %one = constl 1
    %neg1 = constl -1
    %three = constl 3
    %cond_zero = ceql %n, %zero
    jnz %cond_zero, @ret_minus1

    %hi0 = subl %n, %one
    jmp @loop %zero %hi0

    @loop(%lo, %hi)
    %cond_done = csltl %hi, %lo
    jnz %cond_done, @ret_minus1

    %sum = addl %lo, %hi
    %mid = srl %sum, %one
    %mid8 = shll %mid, %three
    %addr = addl %a, %mid8
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @found %mid
    %lt = csltl %val, %key
    jnz %lt, @set_lo %mid %hi
    jmp @set_hi %mid %lo

    @set_lo(%mid, %hi)
    %mid1 = addl %mid, %one
    jmp @loop %mid1 %hi

    @set_hi(%mid, %lo)
    %mid1 = subl %mid, %one
    jmp @loop %lo %mid1

    @found(%mid)
    ret %mid

    @ret_minus1
    ret %neg1
}
