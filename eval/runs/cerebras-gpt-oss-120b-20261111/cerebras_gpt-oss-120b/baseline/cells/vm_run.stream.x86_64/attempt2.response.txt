vm_run:
    push r12
    push r13
    push r14
    push r15
    mov r14, rdi                # code base
    mov r15, rdx                # stack base
    mov r10, rcx                # cap
    xor r12, r12                # pc = 0
    xor r13, r13                # sp = 0
.Lloop:
    cmp r12, rsi
    je .Lfinish
    mov rax, qword ptr [r14 + r12*8]
    cmp rax, 0
    je .Lhalt
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc r12
    jmp .Lloop
.Lhalt:
    jmp .Lfinish
.Lpush:
    mov rdx, qword ptr [r14 + r12*8 + 8]
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    add r12, 2
    jmp .Lloop
.Ladd:
    dec r13
    mov rax, qword ptr [r15 + r13*8]   # b
    dec r13
    mov rdx, qword ptr [r15 + r13*8]   # a
    add rdx, rax
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    inc r12
    jmp .Lloop
.Lsub:
    dec r13
    mov rax, qword ptr [r15 + r13*8]   # b
    dec r13
    mov rdx, qword ptr [r15 + r13*8]   # a
    sub rdx, rax
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    inc r12
    jmp .Lloop
.Lmul:
    dec r13
    mov rdx, qword ptr [r15 + r13*8]   # b
    dec r13
    mov rax, qword ptr [r15 + r13*8]   # a
    mul rdx
    mov qword ptr [r15 + r13*8], rax
    inc r13
    inc r12
    jmp .Lloop
.Ldup:
    mov rdx, qword ptr [r15 + r13*8 - 8]
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    inc r12
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [r15 + r13*8 - 8]
    mov rdx, qword ptr [r15 + r13*8 - 16]
    mov qword ptr [r15 + r13*8 - 8], rdx
    mov qword ptr [r15 + r13*8 - 16], rax
    inc r12
    jmp .Lloop
.Ldrop:
    dec r13
    inc r12
    jmp .Lloop
.Lover:
    mov rdx, qword ptr [r15 + r13*8 - 16]
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    inc r12
    jmp .Lloop
.Llt:
    dec r13
    mov rax, qword ptr [r15 + r13*8]   # b
    dec r13
    mov rdx, qword ptr [r15 + r13*8]   # a
    cmp rdx, rax
    setl al
    movzx rdx, al
    mov qword ptr [r15 + r13*8], rdx
    inc r13
    inc r12
    jmp .Lloop
.Ljz:
    dec r13
    mov rax, qword ptr [r15 + r13*8]   # v
    mov rcx, r12
    inc rcx
    mov rcx, qword ptr [r14 + rcx*8]   # t
    test rax, rax
    jne .Ljz_notzero
    mov r12, rcx
    jmp .Lloop
.Ljz_notzero:
    add r12, 2
    jmp .Lloop
.Ljmp:
    mov rcx, r12
    inc rcx
    mov r12, qword ptr [r14 + rcx*8]   # t
    jmp .Lloop
.Lfinish:
    test r13, r13
    je .Lsetzero
    mov rax, qword ptr [r15 + r13*8 - 8]
    jmp .Lzero_start
.Lsetzero:
    xor rax, rax
.Lzero_start:
    mov r8, r13
.Lzero_loop:
    cmp r8, r10
    jge .Lzero_done
    mov qword ptr [r15 + r8*8], 0
    inc r8
    jmp .Lzero_loop
.Lzero_done:
    pop r15
    pop r14
    pop r13
    pop r12
    ret
