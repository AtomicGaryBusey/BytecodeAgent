mp_mul:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi                # a pointer
    mov r13, rsi                # na
    mov r14, rdx                # b pointer
    mov r15, rcx                # nb
    mov r10, r8                 # r pointer
    mov rdx, r13
    add rdx, r15                # total = na+nb
    xor rcx, rcx
    .Lzero_loop:
    cmp rcx, rdx
    jae .Lzero_done
    mov qword ptr [r10 + rcx*8], 0
    inc rcx
    jmp .Lzero_loop
    .Lzero_done:
    xor rax, rax                # i = 0
    .Li_loop:
    cmp rax, r13
    jae .Lmul_done
    mov r11, qword ptr [r12 + rax*8]   # a_i
    xor r8, r8                 # carryPrev = 0
    xor rbx, rbx               # j = 0
    .Lj_loop:
    cmp rbx, r15
    jae .Lpost_j
    lea r9, [rax + rbx]        # idx = i + j
    mov rsi, qword ptr [r14 + rbx*8]   # b_j
    mov rax, r11
    mul rsi                    # rdx:rax = a_i * b_j
    add rax, qword ptr [r10 + r9*8]
    adc rdx, 0
    add rax, r8
    adc rdx, 0
    mov qword ptr [r10 + r9*8], rax
    mov r8, rdx                # new carryPrev
    inc rbx
    jmp .Lj_loop
    .Lpost_j:
    lea r9, [rax + r15]        # idx = i + nb
    mov qword ptr [r10 + r9*8], r8
    inc rax
    jmp .Li_loop
    .Lmul_done:
    mov rbx, rdx               # rbx = total
    xor rax, rax
    .Lscan_loop:
    test rbx, rbx
    jz .Lreturn_zero
    dec rbx
    mov rcx, qword ptr [r10 + rbx*8]
    test rcx, rcx
    jnz .Lreturn_nonzero
    jmp .Lscan_loop
    .Lreturn_nonzero:
    inc rbx
    mov rax, rbx
    jmp .Lend
    .Lreturn_zero:
    xor rax, rax
    .Lend:
    pop r15
    pop r14
    pop r13
    pop r12
    ret
