    push r12
    push r13
    push r14
    push r15

    mov r13, rsi
    mov r14, rdx
    mov r15, rcx

    mov r10, r13

.Lzero:
    cmp rcx, r10
    jae .Lzero_done
    mov qword ptr [r9 + rcx*8], 0
    inc rcx
    jmp .Lzero
.Lzero_done:
    xor r11, r11
.Li_loop:
    cmp r11, r13
    jae .Lmul_done
    mov r8, qword ptr [r12 + r11*8]
    xor rdi, rdi
    xor rsi, rsi
.Lj_loop:
    cmp rsi, r15
    jae .Lpost_j
    mov rcx, qword ptr [r14 + rsi*8]
    mov rax, r8
    mul rcx
    lea rbx, [r11 + rsi]
    add rax, qword ptr [r9 + rbx*8]
    adc rdx, 0
    add rax, rdi
    adc rdx, 0
    mov qword ptr [r9 + rbx*8], rax
    mov rdi, rdx
    inc rsi
    jmp .Lj_loop
.Lpost_j:
    lea rbx, [r11 + r15]
    mov qword ptr [r9 + rbx*8], rdi
    inc r11
    jmp .Li_loop
.Lmul_done:
    mov rbx, r10
    xor rax, rax
.Lscan:
    test rbx, rbx
    je .Lreturn_zero
    dec rbx
    mov rcx, qword ptr [r9 + rbx*8]
    test rcx, rcx
    jne .Lfound_nonzero
    jmp .Lscan
.Lfound_nonzero:
    inc rbx
    mov rax, rbx
    jmp .Lend
.Lreturn_zero:
    xor rax, rax
.Lend:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
