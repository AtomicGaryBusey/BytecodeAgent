export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %i = 0
    %outpos = 0
    jmp %main_check

@main_loop
    // compute run length at %i
    %run = 0
    %addr0 = addl %src, %i
    %byte0 = loadub %addr0

    @run_loop
    %cond1 = csltl %run, 128
    %tmp1 = addl %i, %run
    %cond2 = csltl %tmp1, %n
    %cond12 = and %cond1, %cond2
    %addr_run = addl %src, %tmp1
    %byte_run = loadub %addr_run
    %eq = ceql %byte_run, %byte0
    %cond = and %cond12, %eq
    jnz %cond, %run_inc
    jmp %run_done

    @run_inc
    %run = addl %run, 1
    jmp %run_loop

    @run_done
    // decide repeat or literal
    %run_lt3 = csltl %run, 3
    jnz %run_lt3, %literal_path
    // ----- repeat packet -----
    %remain = subl %cap, %outpos
    %need = 2
    %fail_rep = csltl %remain, %need
    jnz %fail_rep, %fail_ret

    %tmp257 = 257
    %header = subl %tmp257, %run
    %addr_out = addl %out, %outpos
    storeb %header, %addr_out
    %addr_out1 = addl %addr_out, 1
    storeb %byte0, %addr_out1

    %outpos = addl %outpos, 2
    %i = addl %i, %run
    jmp %main_check

@literal_path
    %lit_start = %i
    %j = %i

    @lit_loop
    %cond_jn = csltl %j, %n
    %len = subl %j, %lit_start
    %cond_len = csltl %len, 128
    %continue = and %cond_jn, %cond_len
    // compute run at %j up to 3
    %runj = 0
    %addr_j0 = addl %src, %j
    %byte_j0 = loadub %addr_j0

    @runj_loop
    %c1 = csltl %runj, 3
    %tmpj = addl %j, %runj
    %c2 = csltl %tmpj, %n
    %c12 = and %c1, %c2
    %addr_rj = addl %src, %tmpj
    %byte_rj = loadub %addr_rj
    %eqj = ceql %byte_rj, %byte_j0
    %c = and %c12, %eqj
    jnz %c, %runj_inc
    jmp %runj_done

    @runj_inc
    %runj = addl %runj, 1
    jmp %runj_loop

    @runj_done
    %runj_ge3 = csltl %runj, 3   // 1 if <3, 0 if >=3
    %cond_continue = and %continue, %runj_ge3
    jnz %cond_continue, %lit_advance
    jmp %lit_exit

    @lit_advance
    %j = addl %j, 1
    jmp %lit_loop

    @lit_exit
    %L = subl %j, %lit_start
    %remain_lit = subl %cap, %outpos
    %need_lit = addl 1, %L
    %fail_lit = csltl %remain_lit, %need_lit
    jnz %fail_lit, %fail_ret

    %header_lit = subl %L, 1
    %addr_out_lit = addl %out, %outpos
    storeb %header_lit, %addr_out_lit

    %k = %lit_start
    @copy_loop
    %cond_k = csltl %k, %j
    jnz %cond_k, %copy_body
    jmp %copy_done

    @copy_body
    %addr_src_k = addl %src, %k
    %b = loadub %addr_src_k
    %offset = subl %k, %lit_start
    %addr_dst = addl %addr_out_lit, 1
    %addr_dst_k = addl %addr_dst, %offset
    storeb %b, %addr_dst_k
    %k = addl %k, 1
    jmp %copy_loop

    @copy_done
    %outpos = addl %outpos, %need_lit
    %i = %j
    jmp %main_check

@fail_ret
    %ret = -1
    ret %ret

@main_check
    %cond_main = csltl %i, %n
    jnz %cond_main, %main_loop
    %ret = %outpos
    ret %ret
}
