export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %i = constl 0
    %outpos = constl 0
@L0
    %cond_main = csltl %i, %n
    jnz %cond_main, @L1
    ret %outpos
@L1
    %run = constl 0
    %addr0 = addl %src, %i
    %byte0 = loadub %addr0
@run_loop
    %c1 = csltl %run, constl 128
    %tmp = addl %i, %run
    %c2 = csltl %tmp, %n
    %c12 = and %c1, %c2
    jnz %c12, @run_check
    jmp @run_done
@run_check
    %addr = addl %src, %tmp
    %byte = loadub %addr
    %eq = ceql %byte, %byte0
    jnz %eq, @run_inc
    jmp @run_done
@run_inc
    %run = addl %run, constl 1
    jmp @run_loop
@run_done
    %run_lt3 = csltl %run, constl 3
    jnz %run_lt3, @literal_start
    ; repeat packet
    %remain = subl %cap, %outpos
    %need2 = constl 2
    %cap_fail = csltl %remain, %need2
    jnz %cap_fail, @fail
    %header = subl constl 257, %run
    %addr_out = addl %out, %outpos
    storeb %header, %addr_out
    %addr_out1 = addl %addr_out, constl 1
    storeb %byte0, %addr_out1
    %outpos = addl %outpos, constl 2
    %i = addl %i, %run
    jmp @L0
@literal_start
    %lit_start = %i
    %j = %i
@lit_loop
    %cond_jn = csltl %j, %n
    %len = subl %j, %lit_start
    %cond_len = csltl %len, constl 128
    %continue = and %cond_jn, %cond_len
    jnz %continue, @check_runj
    jmp @literal_done
@check_runj
    %runj = constl 0
    %addrj0 = addl %src, %j
    %bytej0 = loadub %addrj0
@runj_loop
    %c1j = csltl %runj, constl 3
    %tmpj = addl %j, %runj
    %c2j = csltl %tmpj, %n
    %c12j = and %c1j, %c2j
    jnz %c12j, @runj_check
    jmp @runj_done
@runj_check
    %addrj = addl %src, %tmpj
    %bytej = loadub %addrj
    %eqj = ceql %bytej, %bytej0
    jnz %eqj, @runj_inc
    jmp @runj_done
@runj_inc
    %runj = addl %runj, constl 1
    jmp @runj_loop
@runj_done
    %runj_lt3 = csltl %runj, constl 3
    %cond_continue = and %continue, %runj_lt3
    jnz %cond_continue, @lit_advance
    jmp @literal_done
@lit_advance
    %j = addl %j, constl 1
    jmp @lit_loop
@literal_done
    %L = subl %j, %lit_start
    %remain2 = subl %cap, %outpos
    %needL = addl constl 1, %L
    %cap_fail2 = csltl %remain2, %needL
    jnz %cap_fail2, @fail
    %headerL = subl %L, constl 1
    %addr_outL = addl %out, %outpos
    storeb %headerL, %addr_outL
    %k = %lit_start
@copy_loop
    %cond_k = csltl %k, %j
    jnz %cond_k, @copy_body
    jmp @copy_done
@copy_body
    %addr_src = addl %src, %k
    %b = loadub %addr_src
    %offset = subl %k, %lit_start
    %dst = addl %addr_outL, constl 1
    %dst_k = addl %dst, %offset
    storeb %b, %dst_k
    %k = addl %k, constl 1
    jmp @copy_loop
@copy_done
    %outpos = addl %outpos, %needL
    %i = %j
    jmp @L0
@fail
    ret constl -1
}
