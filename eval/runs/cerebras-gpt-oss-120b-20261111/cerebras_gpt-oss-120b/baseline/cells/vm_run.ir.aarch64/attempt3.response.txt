export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry:
    %pc = lit 0
    %sp = lit 0
    jmp @loop

@loop:
    %c_pc_lt_len = csltl %pc, %len
    jnz %c_pc_lt_len, @fetch
    jmp @end

@fetch:
    %pc8 = mul %pc, 8
    %addr = add %code, %pc8
    %op = loadl %addr

    %t_halt = ceql %op, 0
    jnz %t_halt, @halt
    %t_push = ceql %op, 1
    jnz %t_push, @push
    %t_add = ceql %op, 2
    jnz %t_add, @add
    %t_sub = ceql %op, 3
    jnz %t_sub, @sub
    %t_mul = ceql %op, 4
    jnz %t_mul, @mul
    %t_dup = ceql %op, 5
    jnz %t_dup, @dup
    %t_swap = ceql %op, 6
    jnz %t_swap, @swap
    %t_drop = ceql %op, 7
    jnz %t_drop, @drop
    %t_over = ceql %op, 8
    jnz %t_over, @over
    %t_lt = ceql %op, 9
    jnz %t_lt, @lt
    %t_jz = ceql %op, 10
    jnz %t_jz, @jz
    %t_jmp = ceql %op, 11
    jnz %t_jmp, @jmp
    %pc = add %pc, 1
    jmp @loop

@halt:
    jmp @end

@push:
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %addr1 = add %code, %pc1_8
    %val = loadl %addr1
    %sp8 = mul %sp, 8
    %saddr = add %stack, %sp8
    storel %val, %saddr
    %sp = add %sp, 1
    %pc = add %pc, 2
    jmp @loop

@add:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %sum = add %a, %b
    storel %sum, %aaddr
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@sub:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %diff = sub %a, %b
    storel %diff, %aaddr
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@mul:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %prod = mul %a, %b
    storel %prod, %aaddr
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@dup:
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %taddr = add %stack, %sp1_8
    %val = loadl %taddr
    %sp8 = mul %sp, 8
    %daddr = add %stack, %sp8
    storel %val, %daddr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@swap:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %addr_top = add %stack, %sp1_8
    %addr_next = add %stack, %sp2_8
    %v1 = loadl %addr_top
    %v2 = loadl %addr_next
    storel %v2, %addr_top
    storel %v1, %addr_next
    %pc = add %pc, 1
    jmp @loop

@drop:
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@over:
    %sp2 = sub %sp, 2
    %sp2_8 = mul %sp2, 8
    %src = add %stack, %sp2_8
    %val = loadl %src
    %sp8 = mul %sp, 8
    %dst = add %stack, %sp8
    storel %val, %dst
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@lt:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %cmp_w = csltl %a, %b
    %cmp_l = zextl %cmp_w
    storel %cmp_l, %aaddr
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@jz:
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %vaddr = add %stack, %sp1_8
    %v = loadl %vaddr
    %sp = sub %sp, 1
    %cond = ceql %v, 0
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    %pc_next = add %pc, 2
    jnz %cond, @jz_zero
    %pc = %pc_next
    jmp @loop
@jz_zero:
    %pc = %t
    jmp @loop

@jmp:
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    %pc = %t
    jmp @loop

@end:
    %i = %sp
@zero_loop:
    %c_i_cap = csltl %i, %cap
    jnz %c_i_cap, @zero_body
    jmp @final
@zero_body:
    %i8 = mul %i, 8
    %addrz = add %stack, %i8
    storel 0, %addrz
    %i = add %i, 1
    jmp @zero_loop
@final:
    %empty = ceql %sp, 0
    jnz %empty, @ret0
    %top_idx = sub %sp, 1
    %top8 = mul %top_idx, 8
    %topaddr = add %stack, %top8
    %topval = loadl %topaddr
    ret %topval
@ret0:
    ret 0
}
