export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry:
    %pc = lit 0
    %sp = lit 0
    jmp @loop

@loop:
    %c_lt_len = csltl %pc, %len
    jnz %c_lt_len, @fetch
    jmp @end

@fetch:
    %pc8 = mul %pc, 8
    %addr = add %code, %pc8
    %op = loadl %addr

    %t0 = ceql %op, 0
    jnz %t0, @halt
    %t1 = ceql %op, 1
    jnz %t1, @push
    %t2 = ceql %op, 2
    jnz %t2, @add
    %t3 = ceql %op, 3
    jnz %t3, @sub
    %t4 = ceql %op, 4
    jnz %t4, @mul
    %t5 = ceql %op, 5
    jnz %t5, @dup
    %t6 = ceql %op, 6
    jnz %t6, @swap
    %t7 = ceql %op, 7
    jnz %t7, @drop
    %t8 = ceql %op, 8
    jnz %t8, @over
    %t9 = ceql %op, 9
    jnz %t9, @lt
    %t10 = ceql %op, 10
    jnz %t10, @jz
    %t11 = ceql %op, 11
    jnz %t11, @jmp
    %pc = add %pc, 1
    jmp @loop

@halt:
    jmp @end

@push:
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %addr1 = add %code, %pc1_8
    %val = loadl %addr1
    %sp8 = mul %sp, 8
    %saddr = add %stack, %sp8
    storel %saddr, %val
    %sp = add %sp, 1
    %pc = add %pc, 2
    jmp @loop

@add:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %sum = add %a, %b
    storel %aaddr, %sum
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@sub:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %diff = sub %a, %b
    storel %aaddr, %diff
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@mul:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %prod = mul %a, %b
    storel %aaddr, %prod
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@dup:
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %taddr = add %stack, %sp1_8
    %val = loadl %taddr
    %sp8 = mul %sp, 8
    %daddr = add %stack, %sp8
    storel %daddr, %val
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@swap:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %addr_top = add %stack, %sp1_8
    %addr_next = add %stack, %sp2_8
    %v1 = loadl %addr_top
    %v2 = loadl %addr_next
    storel %addr_top, %v2
    storel %addr_next, %v1
    %pc = add %pc, 1
    jmp @loop

@drop:
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@over:
    %sp2 = sub %sp, 2
    %sp2_8 = mul %sp2, 8
    %src = add %stack, %sp2_8
    %val = loadl %src
    %sp8 = mul %sp, 8
    %dst = add %stack, %sp8
    storel %dst, %val
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@lt:
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %cmp = csltl %a, %b
    %r = zextw %cmp
    storel %aaddr, %r
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@jz:
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %vaddr = add %stack, %sp1_8
    %v = loadl %vaddr
    %sp = sub %sp, 1
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    %cond = ceql %v, 0
    %is_zero = zextw %cond
    %pc_next = add %pc, 2
    %pc = add %pc, %is_zero * %t
    %pc = sub %pc, %is_zero * %pc_next
    // if is_zero==1 => pc = t, else pc = pc+2
    %pc = sub %pc_next, %pc   // Actually simpler: use select
    %z = sub %pc_next, %pc   // Not correct. We'll use conditional jump.

    // Use conditional jump
    jnz %cond, @jz_nonzero
    %pc = %t
    jmp @loop
@jz_nonzero:
    %pc = add %pc, 2
    jmp @loop

@jmp:
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    %pc = %t
    jmp @loop

@end:
    // zero remaining slots
    %i = %sp
@zero_loop:
    %c_i_cap = csltl %i, %cap
    jnz %c_i_cap, @zero_body
    jmp @final

@zero_body:
    %i8 = mul %i, 8
    %addrz = add %stack, %i8
    storel %addrz, 0
    %i = add %i, 1
    jmp @zero_loop

@final:
    %empty = ceql %sp, 0
    jnz %empty, @ret0
    %top_idx = sub %sp, 1
    %top8 = mul %top_idx, 8
    %topaddr = add %stack, %top8
    %topval = loadl %topaddr
    ret %topval

@ret0:
    ret 0
}
