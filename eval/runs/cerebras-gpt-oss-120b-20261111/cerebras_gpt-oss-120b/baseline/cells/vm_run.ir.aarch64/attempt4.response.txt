export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start:
    %pc0 = lit 0
    %sp0 = lit 0
    jmp @loop(%pc0, %sp0)

@loop(%pc, %sp):
    %c = csltl %pc, %len
    jnz %c, @fetch(%pc, %sp)
    jmp @end(%pc, %sp)

@fetch(%pc, %sp):
    %pc8 = mul %pc, 8
    %addr = add %code, %pc8
    %op = loadl %addr

    %t_halt = ceql %op, 0
    jnz %t_halt, @end(%pc, %sp)

    %t_push = ceql %op, 1
    jnz %t_push, @push(%pc, %sp)

    %t_add = ceql %op, 2
    jnz %t_add, @add(%pc, %sp)

    %t_sub = ceql %op, 3
    jnz %t_sub, @sub(%pc, %sp)

    %t_mul = ceql %op, 4
    jnz %t_mul, @mul(%pc, %sp)

    %t_dup = ceql %op, 5
    jnz %t_dup, @dup(%pc, %sp)

    %t_swap = ceql %op, 6
    jnz %t_swap, @swap(%pc, %sp)

    %t_drop = ceql %op, 7
    jnz %t_drop, @drop(%pc, %sp)

    %t_over = ceql %op, 8
    jnz %t_over, @over(%pc, %sp)

    %t_lt = ceql %op, 9
    jnz %t_lt, @lt(%pc, %sp)

    %t_jz = ceql %op, 10
    jnz %t_jz, @jz(%pc, %sp)

    %t_jmp = ceql %op, 11
    jnz %t_jmp, @jmp(%pc, %sp)

    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp)

@push(%pc, %sp):
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %valaddr = add %code, %pc1_8
    %val = loadl %valaddr
    %sp8 = mul %sp, 8
    %saddr = add %stack, %sp8
    storel %val, %saddr
    %sp1 = add %sp, 1
    %pc2 = add %pc, 2
    jmp @loop(%pc2, %sp1)

@add(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %sum = add %a, %b
    storel %sum, %aaddr
    %sp1n = sub %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@sub(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %diff = sub %a, %b
    storel %diff, %aaddr
    %sp1n = sub %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@mul(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %prod = mul %a, %b
    storel %prod, %aaddr
    %sp1n = sub %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@dup(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %src = add %stack, %sp1_8
    %val = loadl %src
    %sp8 = mul %sp, 8
    %dst = add %stack, %sp8
    storel %val, %dst
    %sp1n = add %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@swap(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %addr_top = add %stack, %sp1_8
    %addr_next = add %stack, %sp2_8
    %v1 = loadl %addr_top
    %v2 = loadl %addr_next
    storel %v2, %addr_top
    storel %v1, %addr_next
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp)

@drop(%pc, %sp):
    %sp1n = sub %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@over(%pc, %sp):
    %sp2 = sub %sp, 2
    %sp2_8 = mul %sp2, 8
    %src = add %stack, %sp2_8
    %val = loadl %src
    %sp8 = mul %sp, 8
    %dst = add %stack, %sp8
    storel %val, %dst
    %sp1n = add %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@lt(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp2 = sub %sp, 2
    %sp1_8 = mul %sp1, 8
    %sp2_8 = mul %sp2, 8
    %baddr = add %stack, %sp1_8
    %aaddr = add %stack, %sp2_8
    %b = loadl %baddr
    %a = loadl %aaddr
    %cmp_w = csltl %a, %b
    %cmp_l = zextl %cmp_w
    storel %cmp_l, %aaddr
    %sp1n = sub %sp, 1
    %pc1 = add %pc, 1
    jmp @loop(%pc1, %sp1n)

@jz(%pc, %sp):
    %sp1 = sub %sp, 1
    %sp1_8 = mul %sp1, 8
    %vaddr = add %stack, %sp1_8
    %v = loadl %vaddr
    %sp1n = sub %sp, 1
    %cond = ceql %v, 0
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    %pc_jump = %t
    %pc_next = add %pc, 2
    %pc_next_sel = phi l %pc_next, %pc_jump
    %pc_next_sel = phi l %pc_jump, %pc_next   // placeholder, will be overridden by condition
    jnz %cond, @jz_zero(%pc_jump, %sp1n)
    jmp @loop(%pc_next, %sp1n)

@jz_zero(%pc_target, %sp):
    jmp @loop(%pc_target, %sp)

@jmp(%pc, %sp):
    %pc1 = add %pc, 1
    %pc1_8 = mul %pc1, 8
    %taddr = add %code, %pc1_8
    %t = loadl %taddr
    jmp @loop(%t, %sp)

@end(%pc, %sp):
    %i0 = %sp
@zero_loop(%i):
    %c = csltl %i, %cap
    jnz %c, @zero_body(%i)
    jmp @final(%sp)

@zero_body(%i):
    %i8 = mul %i, 8
    %addrz = add %stack, %i8
    storel 0, %addrz
    %i1 = add %i, 1
    jmp @zero_loop(%i1)

@final(%sp):
    %empty = ceql %sp, 0
    jnz %empty, @ret0
    %top_idx = sub %sp, 1
    %top8 = mul %top_idx, 8
    %topaddr = add %stack, %top8
    %topval = loadl %topaddr
    ret %topval

@ret0:
    ret 0
}
