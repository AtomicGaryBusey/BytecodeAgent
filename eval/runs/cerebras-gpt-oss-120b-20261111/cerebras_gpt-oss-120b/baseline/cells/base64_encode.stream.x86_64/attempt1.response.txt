base64_encode:
    xor r9, r9                     # output length counter = 0
    mov rcx, rsi                  # rcx = n (bytes remaining)
    test rcx, rcx
    je .Ldone                     # if n==0, finish
.Lmain_loop:
    cmp rcx, 3
    jl .Lremainder
    mov al, byte ptr [rdi]        # a0
    mov bl, byte ptr [rdi+1]      # a1
    mov bh, byte ptr [rdi+2]      # a2
    # v0 = a0 >> 2
    movzx eax, al
    shr eax, 2
    # map v0 -> char
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmap0_store
    add edx, 6
    cmp eax, 52
    jb .Lmap0_store
    add edx, -63
    cmp eax, 62
    jb .Lmap0_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmap0_store
    mov dl, '/'
.Lmap0_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v1 = ((a0 & 3) << 4) | (a1 >> 4)
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    # map v1
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmap1_store
    add edx, 6
    cmp eax, 52
    jb .Lmap1_store
    add edx, -63
    cmp eax, 62
    jb .Lmap1_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmap1_store
    mov dl, '/'
.Lmap1_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v2 = ((a1 & 0xF) << 2) | (a2 >> 6)
    movzx eax, bl
    and eax, 0xF
    shl eax, 2
    movzx ebx, bh
    shr ebx, 6
    or eax, ebx
    # map v2
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmap2_store
    add edx, 6
    cmp eax, 52
    jb .Lmap2_store
    add edx, -63
    cmp eax, 62
    jb .Lmap2_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmap2_store
    mov dl, '/'
.Lmap2_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v3 = a2 & 0x3F
    movzx eax, bh
    and eax, 0x3F
    # map v3
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmap3_store
    add edx, 6
    cmp eax, 52
    jb .Lmap3_store
    add edx, -63
    cmp eax, 62
    jb .Lmap3_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmap3_store
    mov dl, '/'
.Lmap3_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    add rdi, 3
    sub rcx, 3
    jmp .Lmain_loop
.Lremainder:
    cmp rcx, 0
    je .Ldone
    cmp rcx, 1
    je .Lrem1
    # rcx == 2
    mov al, byte ptr [rdi]
    mov bl, byte ptr [rdi+1]
    # v0 = a0 >> 2
    movzx eax, al
    shr eax, 2
    # map v0
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmapr0_store
    add edx, 6
    cmp eax, 52
    jb .Lmapr0_store
    add edx, -63
    cmp eax, 62
    jb .Lmapr0_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmapr0_store
    mov dl, '/'
.Lmapr0_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v1 = ((a0 & 3) << 4) | (a1 >> 4)
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    # map v1
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmapr1_store
    add edx, 6
    cmp eax, 52
    jb .Lmapr1_store
    add edx, -63
    cmp eax, 62
    jb .Lmapr1_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmapr1_store
    mov dl, '/'
.Lmapr1_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v2 = ((a1 & 0xF) << 2)
    movzx eax, bl
    and eax, 0xF
    shl eax, 2
    # map v2
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmapr2_store
    add edx, 6
    cmp eax, 52
    jb .Lmapr2_store
    add edx, -63
    cmp eax, 62
    jb .Lmapr2_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmapr2_store
    mov dl, '/'
.Lmapr2_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # padding '='
    mov byte ptr [rdx], '='
    inc rdx
    inc r9
    jmp .Ldone
.Lrem1:
    mov al, byte ptr [rdi]
    # v0 = a0 >> 2
    movzx eax, al
    shr eax, 2
    # map v0
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmapr10_store
    add edx, 6
    cmp eax, 52
    jb .Lmapr10_store
    add edx, -63
    cmp eax, 62
    jb .Lmapr10_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmapr10_store
    mov dl, '/'
.Lmapr10_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # v1 = (a0 & 3) << 4
    movzx eax, al
    and eax, 3
    shl eax, 4
    # map v1
    mov edx, eax
    add edx, 65
    cmp eax, 26
    jb .Lmapr11_store
    add edx, 6
    cmp eax, 52
    jb .Lmapr11_store
    add edx, -63
    cmp eax, 62
    jb .Lmapr11_store
    mov dl, '+'
    cmp eax, 63
    jne .Lmapr11_store
    mov dl, '/'
.Lmapr11_store:
    mov byte ptr [rdx], dl
    inc rdx
    inc r9
    # padding "=="
    mov byte ptr [rdx], '='
    inc rdx
    inc r9
    mov byte ptr [rdx], '='
    inc rdx
    inc r9
    jmp .Ldone
.Ldone:
    mov rax, r9
    ret
