base64_encode:
    xor r9, r9                     # length = 0
    mov r8, rdx                    # r8 = out pointer
    mov rcx, rsi                   # rcx = n (bytes remaining)
    test rcx, rcx
    je .Ldone
.Lmain_loop:
    cmp rcx, 3
    jl .Lremainder
    # load three bytes
    mov al, byte ptr [rdi]         # a0
    mov bl, byte ptr [rdi+1]       # a1
    mov bh, byte ptr [rdi+2]       # a2
    # v0 = a0 >> 2
    movzx eax, al
    shr eax, 2
    # map v0 -> char in al
    cmp eax, 26
    jb .Lcase0
    cmp eax, 52
    jb .Lcase1
    cmp eax, 62
    jb .Lcase2
    je .Lcase3
    mov al, '/'                     # v == 63
    jmp .Lstore_char
.Lcase3:
    mov al, '+'                     # v == 62
    jmp .Lstore_char
.Lcase2:
    sub eax, 52
    add al, '0'                     # '0' + (v-52)
    jmp .Lstore_char
.Lcase1:
    sub eax, 26
    add al, 'a'                     # 'a' + (v-26)
    jmp .Lstore_char
.Lcase0:
    add al, 'A'                     # 'A' + v
.Lstore_char:
    mov byte ptr [r8], al
    inc r8
    inc r9
    # v1 = ((a0 & 3) << 4) | (a1 >> 4)
    mov al, byte ptr [rdi]         # a0
    mov bl, byte ptr [rdi+1]       # a1
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    # map v1
    cmp eax, 26
    jb .Lcase0
    cmp eax, 52
    jb .Lcase1
    cmp eax, 62
    jb .Lcase2
    je .Lcase3
    mov al, '/'                     # v == 63
    jmp .Lstore_char
    # v2 = ((a1 & 0xF) << 2) | (a2 >> 6)
.Lcase3:
    mov al
