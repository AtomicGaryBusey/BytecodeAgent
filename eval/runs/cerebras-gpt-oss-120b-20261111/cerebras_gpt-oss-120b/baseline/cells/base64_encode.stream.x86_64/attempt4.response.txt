base64_encode:
    mov r8, rdx                # out pointer
    xor r9, r9                 # output length counter
    test rsi, rsi
    je .Ldone
.Lmain_loop:
    cmp rsi, 3
    jb .Lremainder
    movzx eax, byte ptr [rdi]          # b0
    movzx ebx, byte ptr [rdi+1]        # b1
    movzx ecx, byte ptr [rdi+2]        # b2
    # v0 = b0 >> 2
    mov r10d, eax
    shr r10d, 2
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA0
    cmp r11d, 52
    jl .Lmapa0
    cmp r11d, 62
    jl .Lmap0_0
    cmp r11d, 63
    je .Lmapslash0
    mov r11b, '+'
    jmp .Lstore0
.LmapA0:    add r11b, 'A'
    jmp .Lstore0
.Lmapa0:    sub r11b, 26
    add r11b, 'a'
    jmp .Lstore0
.Lmap0_0:  sub r11b, 52
    add r11b, '0'
    jmp .Lstore0
.Lmapslash0: mov r11b, '/'
    jmp .Lstore0
.Lstore0:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    # v1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, ebx
    shr r11d, 4
    or  r10d, r11d
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA1
    cmp r11d, 52
    jl .Lmapa1
    cmp r11d, 62
    jl .Lmap0_1
    cmp r11d, 63
    je .Lmapslash1
    mov r11b, '+'
    jmp .Lstore1

    jmp .Lstore1
.Lmapa1:   sub r11b, 26
    add r11b, 'a'
    jmp .Lstore1
.Lmap0_1: sub r11b, 52
    add r11b, '0'
    jmp .Lstore1
.Lmapslash1: mov r11b, '/'
    jmp .Lstore1
.Lstore1:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    # v2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    mov r10d, ebx
    and r10d, 0xF
    shl r10d, 2
    mov r11d, ecx
    shr r11d, 6
    or  r10d, r11d
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA2
    cmp r11d, 52
    jl .Lmapa2
    cmp r11d, 62
    jl .Lmap0_2
    cmp r11d, 63
    je .Lmapslash2
    mov r11b, '+'
    jmp .Lstore2
.LmapA2:   add r11b, 'A'
    jmp .Lstore2
.Lmapa2:   sub r11b, 26
    add r11b, 'a'
    jmp .Lstore2
.Lmap0_2: sub r11b, 52
    add r11b, '0'
    jmp .Lstore2
.Lmapslash2: mov r11b, '/'
    jmp .Lstore2
.Lstore2:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    # v3 = b2 & 0x3F
    mov r10d, ecx
    and r10d, 0x3F
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA3
    cmp r11d, 52
    jl .Lmapa3
    cmp r11d, 62
    jl .Lmap0_3
    cmp r11d, 63
    je .Lmapslash3
    mov r11b, '+'
    jmp .Lstore3
.LmapA3:   add r11b, 'A'
    jmp .Lstore3
.Lmapa3:   sub r11b, 26
    add r11b, 'a'
    jmp .Lstore3
.Lmap0_3: sub r11b, 52
    add r11b, '0'
    jmp .Lstore3
.Lmapslash3: mov r11b, '/'
    jmp .Lstore3
.Lstore3:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    add rdi, 3
    sub rsi, 3
    jmp .Lmain_loop
.Lremainder:
    cmp rsi, 0
    je .Ldone
    cmp rsi, 1
    je .Lrem1
    # rsi == 2
    movzx eax, byte ptr [rdi]          # b0
    movzx ebx, byte ptr [rdi+1]        # b1
    # v0 = b0 >> 2
    mov r10d, eax
    shr r10d, 2
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA_R0
    cmp r11d, 52
    jl .Lmapa_R0
    cmp r11d, 62
    jl .Lmap0_R0
    cmp r11d, 63
    je .Lmapslash_R0

    jmp .Lstore_R0

    add r11b, '0'

    mov byte ptr [r8], r11b

    # v1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov r10d, eax
    and r10d, 3

    add r11b, 'a'

    add r11b, '0'
    jmp .Lstore_R1

    jmp .Lstore_R1
.Lstore_R1:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    # v2 = ((b1 & 0xF) << 2)
    mov r10d, ebx
    and r10d, 0xF
    shl r10d, 2

    jl .LmapA_R2

    add r11b, '0'
    jmp .Lstore_R10
.Lmapslash_R10: mov r11b, '/'
    jmp .Lstore_R10
.Lstore_R10:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    # v1 = ((b0 & 3) << 4)
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, r10d
    cmp r11d, 26
    jl .LmapA_R11
    cmp r11d, 52
    jl .Lmapa_R11
    cmp r11d, 62

    add r11b, '0'
    jmp .Lstore_R11

    jmp .Lstore_R11
.Lstore_R11:
    mov byte ptr [r8], r11b
    inc r8
    inc r9
    mov byte ptr [r8], '='
    inc r8
    inc r9
    mov byte ptr [r8], '='
    inc r8
    inc r9
    jmp .Ldone
