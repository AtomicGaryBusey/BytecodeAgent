base64_encode:
    xor r9, r9
    mov r8, rdx
    mov rcx, rsi
    test rcx, rcx
    je .Ldone
.Lmain_loop:
    cmp rcx, 3
    jl .Lremainder
    mov al, byte ptr [rdi]
    mov bl, byte ptr [rdi+1]
    mov bh, byte ptr [rdi+2]
    # v0
    movzx eax, al
    shr eax, 2
    cmp eax, 26
    jb .Lmap0_A
    cmp eax, 52
    jb .Lmap0_B
    cmp eax, 62
    jb .Lmap0_C
    cmp eax, 63
    je .Lmap0_D
    mov al, '+'                # v==62
    jmp .Lstore0
.Lmap0_A: add al, 'A'
    jmp .Lstore0
.Lmap0_B: sub eax, 26
    add al, 'a'
    jmp .Lstore0
.Lmap0_C: sub eax, 52
    add al, '0'
    jmp .Lstore0
.Lmap0_D: mov al, '/'
.Lstore0:
    mov byte ptr [r8], al
    inc r8
    inc r9
    # v1
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    cmp eax, 26
    jb .Lmap1_A
    cmp eax, 52
    jb .Lmap1_B
    cmp eax, 62
    jb .Lmap1_C
    cmp eax, 63

.Lstore1:

    inc r9
    # v2
    movzx eax, bl
    and eax, 0xF
    shl eax, 2
    movzx ebx, bh

    or eax, ebx

    jb .Lmap2_A
    cmp eax, 52
    jb .Lmap2_B
    cmp eax, 62
    jb .Lmap2_C
