mat_pow:
    // x0=a, x1=n, x2=k, x3=out, x4=tmp
    // compute n_sq and size in bytes
    mov     x5, x1
    mul     x5, x5, x1          // x5 = n_sq
    lsl     x6, x5, #3          // x6 = n_sq * 8 (size_bytes)
    add     x7, x4, x6          // x7 = temp buffer (second half)

    // copy a -> base buffer (x4)
    mov     x8, x5              // count = n_sq
    mov     x9, x0              // src = a
    mov     x10, x4             // dst = base
.Lcopy_a:
    cbz     x8, .Lcopy_a_done
    ldr     x11, [x9], #8
    str     x11, [x10], #8
    sub     x8, x8, #1
    b       .Lcopy_a
.Lcopy_a_done:

    // initialize out = identity matrix
    mov     x8, #0              // i = 0
.Lid_i:
    cmp     x8, x1
    bge     .Lid_done
    mov     x9, #0              // j = 0
.Lid_j:
    cmp     x9, x1
    bge     .Lid_j_next
    mul     x10, x8, x1
    add     x10, x10, x9
    lsl     x10, x10, #3
    add     x10, x3, x10
    mov     x11, #0
    cmp     x8, x9
    cset    x11, eq
    str     x11, [x10]
    add     x9, x9, #1
    b       .Lid_j
.Lid_j_next:
    add     x8, x8, #1
    b       .Lid_i
.Lid_done:

    // exponentiation loop
.Lexp_loop:
    cbz     x2, .Lexp_done
    and     x8, x2, #1
    cbz     x8, .Lskip_mul_out

    // out = out * base -> tmp buffer (x7)
    mov     x12, x3             // A = out
    mov     x13, x4             // B = base
    mov     x14, x7             // C = tmp
    // multiplication A*B -> C
    mov     x15, #0             // i = 0
.Lmul_out_i:
    cmp     x15, x1
    bge     .Lmul_out_i_done
    mov     x16, #0             // j = 0
.Lmul_out_j:
    cmp     x16, x1
    bge     .Lmul_out_j_done
    mov     x17, #0             // sum = 0
    mov     x18, #0             // t = 0
.Lmul_out_t:
    cmp     x18, x1
    bge     .Lmul_out_t_done
    // load A[i][t]
    mul     x19, x15, x1
    add     x19, x19, x18
    lsl     x19, x19, #3
    add     x19, x12, x19
    ldr     x19, [x19]
    // load B[t][j]
    mul     x20, x18, x1
    add     x20, x20, x16
    lsl     x20, x20, #3
    add     x20, x13, x20
    ldr     x20, [x20]
    mul     x21, x19, x20
    add     x17, x17, x21
    add     x18, x18, #1
    b       .Lmul_out_t
.Lmul_out_t_done:
    // store sum to C[i][j]
    mul     x19, x15, x1
    add     x19, x19, x16
    lsl     x19, x19, #3
    add     x19, x14, x19
    str     x17, [x19]
    add     x16, x16, #1
    b       .Lmul_out_j
.Lmul_out_j_done:
    add     x15, x15, #1
    b       .Lmul_out_i
.Lmul_out_i_done:

    // copy tmp (x7) -> out (x3)
    mov     x22, x5
    mov     x23, x7
    mov     x24, x3
.Lcopy_out:
    cbz     x22, .Lcopy_out_done
    ldr     x25, [x23], #8
    str     x25, [x24], #8
    sub     x22, x22, #1
    b       .Lcopy_out
.Lcopy_out_done:

.Lskip_mul_out:
    // base = base * base -> tmp buffer (x7)
    mov     x12, x4             // A = base
    mov     x13, x4             // B = base
    mov     x14, x7             // C = tmp
    // multiplication A*B -> C (reuse same loop labels with new prefix)
    mov     x15, #0             // i = 0
.Lmul_base_i:
    cmp     x15, x1
    bge     .Lmul_base_i_done
    mov     x16, #0             // j = 0
.Lmul_base_j:
    cmp     x16, x1
    bge     .Lmul_base_j_done
    mov     x17, #0             // sum = 0
    mov     x18, #0             // t = 0
.Lmul_base_t:
    cmp     x18, x1
    bge     .Lmul_base_t_done
    // load A[i][t]
    mul     x19, x15, x1
    add     x19, x19, x18
    lsl     x19, x19, #3
    add     x19, x12, x19
    ldr     x19, [x19]
    // load B[t][j]
    mul     x20, x18, x1
    add     x20, x20, x16
    lsl     x20, x20, #3
    add     x20, x13, x20
    ldr     x20, [x20]
    mul     x21, x19, x20
    add     x17, x17, x21
    add     x18, x18, #1
    b       .Lmul_base_t
.Lmul_base_t_done:
    // store sum to C[i][j]
    mul     x19, x15, x1
    add     x19, x19, x16
    lsl     x19, x19, #3
    add     x19, x14, x19
    str     x17, [x19]
    add     x16, x16, #1
    b       .Lmul_base_j
.Lmul_base_j_done:
    add     x15, x15, #1
    b       .Lmul_base_i
.Lmul_base_i_done:

    // copy tmp (x7) -> base (x4)
    mov     x22, x5
    mov     x23, x7
    mov     x24, x4
.Lcopy_base:
    cbz     x22, .Lcopy_base_done
    ldr     x25, [x23], #8
    str     x25, [x24], #8
    sub     x22, x22, #1
    b       .Lcopy_base
.Lcopy_base_done:

    // shift exponent
    lsr     x2, x2, #1
    b       .Lexp_loop
.Lexp_done:

    // zero tmp (both buffers)
    add     x8, x5, x5          // total elements = 2*n_sq
    mov     x9, x4
.Lzero_tmp:
    cbz     x8, .Lzero_done
    str     xzr, [x9], #8
    sub     x8, x8, #1
    b       .Lzero_tmp
.Lzero_done:
    mov     x0, #0
    ret
