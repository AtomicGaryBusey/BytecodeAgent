mat_pow:
    // x0 = a, x1 = n, x2 = k, x3 = out, x4 = tmp
    // compute n_sq and size_bytes
    mov x5, x1                // n
    mul x7, x5, x5            // n_sq
    lsl x8, x7, #3            // size_bytes = n_sq * 8
    add x9, x4, x8            // second buffer = tmp + size_bytes

    // if k == 0 -> identity and zero tmp
    cbz x2, .Lk_zero

    // copy a into base buffer (tmp)
    mov x10, x7               // count = n_sq
    mov x11, x0               // src = a
    mov x12, x4               // dst = base
.Lcopy_a_loop:
    cbz x10, .Lcopy_a_done
    ldr x13, [x11], #8
    str x13, [x12], #8
    sub x10, x10, #1
    b .Lcopy_a_loop
.Lcopy_a_done:

    // initialize out = identity
    mov x14, #0               // i = 0
.Lid_i_loop:
    cmp x14, x5
    bge .Lid_i_done
    mul x15, x14, x5         // i*n
    lsl x15, x15, #3         // byte offset
    add x16, x3, x15         // row_ptr = out + i*n*8
    mov x17, #0              // j = 0
.Lid_j_loop:
    cmp x17, x5
    bge .Lid_j_done
    lsl x18, x17, #3
    add x19, x16, x18        // element ptr
    mov x20, #0
    cmp x14, x17
    cset x20, eq
    str x20, [x19]
    add x17, x17, #1
    b .Lid_j_loop
.Lid_j_done:
    add x14, x14, #1
    b .Lid_i_loop
.Lid_i_done:

    // exponentiation loop
.Lexp_loop:
    cbz x2, .Lexp_done
    and x21, x2, #1
    cbz x21, .Lskip_mul_out
    // out = out * base -> product in buffer (x9)
    // A = out (x3), B = base (x4), C = buffer (x9)
    // multiplication 1
    mov x22, x3
    mov x23, x4
    mov x24, x9
    mov x25, #0                // i = 0
.Lmul1_i_loop:
    cmp x25, x5
    bge .Lmul1_i_done
    mul x26, x25, x5
    lsl x26, x26, #3
    add x27, x22, x26         // a_row = A + i*n*8
    add x28, x24, x26         // c_row = C + i*n*8
    mov x29, #0                // j = 0
.Lmul1_j_loop:
    cmp x29, x5
    bge .Lmul1_j_done
    mov x30, #0                // sum = 0
    mov x31, #0                // t = 0
.Lmul1_t_loop:
    cmp x31, x5
    bge .Lmul1_t_done
    // load a = A[i][t]
    lsl x0, x31, #3
    add x0, x27, x0
    ldr x
