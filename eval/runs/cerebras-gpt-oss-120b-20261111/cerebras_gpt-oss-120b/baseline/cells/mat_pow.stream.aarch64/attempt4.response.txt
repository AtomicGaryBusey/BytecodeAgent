mat_pow:
    // compute n_sq and size bytes, work pointer
    mul     x5, x1, x1
    lsl     x6, x5, #3
    add     x7, x4, x6

    // copy a -> base
    mov     x9, x5
    mov     x10, x0
    mov     x11, x4
.Lcopy_a:
    cbz     x9, .Lcopy_a_done
    ldr     x12, [x10], #8
    str     x12, [x11], #8
    sub     x9, x9, #1
    b       .Lcopy_a
.Lcopy_a_done:

    // zero out out matrix
    mov     x9, x5
    mov     x10, x3
.Lzero_out:
    cbz     x9, .Lzero_out_done
    str     xzr, [x10], #8
    sub     x9, x9, #1
    b       .Lzero_out
.Lzero_out_done:

    // set identity diagonal
    mov     x12, #0
.Ldiag:
    cmp     x12, x1
    bge     .Ldiag_done
    mul     x13, x12, x1
    add     x13, x13, x12
    lsl     x13, x13, #3
    add     x13, x3, x13
    mov     x14, #1
    str     x14, [x13]
    add     x12, x12, #1
    b       .Ldiag
.Ldiag_done:

.Lexp_loop:
    cbz     x2, .Lexp_done
    and     x9, x2, #1
    cbz     x9, .Lskip_mul_out

    // out = out * base -> work
    mov     x10, x3
    mov     x11, x4
    mov     x12, x7
    mov     x13, #0
.Lmul_out_i:
    cmp     x13, x1
    bge     .Lmul_out_i_done
    mov     x14, #0
.Lmul_out_j:
    cmp     x14, x1
    bge     .Lmul_out_j_done
    mov     x16, xzr
    mov     x15, #0
.Lmul_out_t:
    cmp     x15, x1
    bge     .Lmul_out_t_done
    mul     x17, x13, x1
    add     x17, x17, x15
    lsl     x17, x17, #3
    add     x17, x10, x17
    ldr     x0, [x17]
    mul     x18, x15, x1
    add     x18, x18, x14
    lsl     x18, x18, #3
    add     x18, x11, x18
    ldr     x8, [x18]
    mul     x18, x0, x8
    add     x16, x16, x18
    add     x15, x15, #1
    b       .Lmul_out_t
.Lmul_out_t_done:
    mul     x17, x13, x1
    add     x17, x17, x14
    lsl     x17, x17, #3
    add     x17, x12, x17
    str     x16, [x17]
    add     x14, x14, #1
    b       .Lmul_out_j
.Lmul_out_j_done:
    add     x13, x13, #1
    b       .Lmul_out_i
.Lmul_out_i_done:

    // copy work -> out
    mov     x9, x5
    mov     x10, x7
    mov     x11, x3
.Lcopy_work_out:
    cbz     x9, .Lcopy_work_out_done
    ldr     x0, [x10], #8
    str     x0, [x11], #8
    sub     x9, x9, #1
    b       .Lcopy_work_out
.Lcopy_work_out_done:

.Lskip_mul_out:
    // base = base * base -> work
    mov     x10, x4
    mov     x11, x4
    mov     x12, x7
    mov     x13, #0
.Lmul_base_i:
    cmp     x13, x1
    bge     .Lmul_base_i_done
    mov     x14, #0
.Lmul_base_j:
    cmp     x14, x1
    bge     .Lmul_base_j_done
    mov     x16, xzr
    mov     x15, #0
.Lmul_base_t:
    cmp     x15, x1
    bge     .Lmul_base_t_done
    mul     x17, x13, x1
    add     x17, x17, x15
    lsl     x17, x17, #3
    add     x17, x10, x17
    ldr     x0, [x17]
    mul     x18, x15, x1
    add     x18, x18, x14
    lsl     x18, x18, #3
    add     x18, x11, x18
    ldr     x8, [x18]
    mul     x18, x0, x8
    add     x16, x16, x18
    add     x15, x15, #1
    b       .Lmul_base_t
.Lmul_base_t_done:
    mul     x17, x13, x1
    add     x17, x17, x14
    lsl     x17, x17, #3
    add     x17, x12, x17
    str     x16, [x17]
    add     x14, x14, #1
    b       .Lmul_base_j
.Lmul_base_j_done:
    add     x13, x13, #1
    b       .Lmul_base_i
.Lmul_base_i_done:

    // copy work -> base
    mov     x9, x5
    mov     x10, x7
    mov     x11, x4
.Lcopy_work_base:
    cbz     x9, .Lcopy_work_base_done
    ldr     x0, [x10], #8
    str     x0, [x11], #8
    sub     x9, x9, #1
    b       .Lcopy_work_base
.Lcopy_work_base_done:

    // shift exponent
    lsr     x2, x2, #1
    b       .Lexp_loop

.Lexp_done:
    // zero tmp buffers
    add     x9, x5, x5
    mov     x10, x4
.Lzero_tmp:
    cbz     x9, .Lzero_tmp_done
    str     xzr, [x10], #8
    sub     x9, x9, #1
    b       .Lzero_tmp
.Lzero_tmp_done:
    mov     x0, #0
    ret
