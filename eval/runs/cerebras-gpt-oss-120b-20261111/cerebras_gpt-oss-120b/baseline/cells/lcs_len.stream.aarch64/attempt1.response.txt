lcs_len:
    // preserve input pointers
    mov x10, x0          // x pointer
    mov x11, x2          // y pointer

    // zero row[0..n]
    mov x8, #0
.Lzero_loop:
    cmp x8, x3
    bgt .Lzero_done
    add x9, x4, x8, LSL #3
    str xzr, [x9]
    add x8, x8, #1
    b .Lzero_loop
.Lzero_done:

    // i = 0
    mov x5, #0
.Louter:
    cmp x5, x1
    bge .Lfinish

    // load x[i] byte
    add x0, x10, x5
    ldrb w0, [x0]

    // diag = 0
    mov x6, #0

    // j = 1
    mov x8, #1
.Linner:
    cmp x8, x3
    bgt .Linner_done

    // address of row[j]
    add x9, x4, x8, LSL #3
    // up = row[j]
    ldr x7, [x9]

    // y[j-1] byte
    sub x12, x8, #1
    add x13, x11, x12
    ldrb w12, [x13]

    // compare bytes
    cmp w0, w12
    bne .Lnot_eq

    // equal: row[j] = diag + 1
    add x6, x6, #1
    str x6, [x9]
    // diag = up
    mov x6, x7
    b .Linner_next

.Lnot_eq:
    // left = row[j-1]
    sub x12, x8, #1
    add x14, x4, x12, LSL #3
    ldr x14, [x14]          // x14 = left

    // max(up, left) -> x15
    cmp x7, x14
    csel x15, x7, x14, gt
    str x15, [x9]

    // diag = up
    mov x6, x7

.Linner_next:
    add x8, x8, #1
    b .Linner

.Linner_done:
    add x5, x5, #1
    b .Louter

.Lfinish:
    // result = row[n]
    add x0, x4, x3, LSL #3
    ldr x0, [x0]
    ret
