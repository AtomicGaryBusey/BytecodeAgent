stats_f:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    mov x19, x2          // f pointer
    mov x20, x0          // a pointer
    mov x21, x1          // n (remaining count)
    mov x22, x3          // out pointer
    // first element
    ldr x0, [x20], #8
    blr x19
    mov x23, x0          // min
    mov x24, x0          // max
    mov x25, x0          // sum
    mov x26, #0          // odd count
    tbz x0, #0, .skip_inc0
    add x26, x26, #1
.skip_inc0:
    subs x21, x21, #1
    beq .store_results
.loop:
    ldr x0, [x20], #8
    blr x19
    add x25, x25, x0
    cmp x0, x23
    csel x23, x0, x23, lt
    cmp x0, x24
    csel x24, x0, x24, gt
    tbz x0, #0, .skip_inc
    add x26, x26, #1
.skip_inc:
    subs x21, x21, #1
    b.ne .loop
.store_results:
    str x23, [x22]
    str x24, [x22, #8]
    str x25, [x22, #16]
    mov x0, x26
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
