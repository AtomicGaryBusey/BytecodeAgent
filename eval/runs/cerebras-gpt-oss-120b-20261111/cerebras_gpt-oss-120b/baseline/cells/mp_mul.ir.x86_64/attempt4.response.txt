export function l $mp_mul(l a, l na, l b, l nb, l r) {
    %zero_l = l 0
    %mask32 = l 4294967295
    %two32 = l 4294967296

    %total = l add na, nb

    // initialize r[0..total-1] = 0
    %i0 = l 0
@init_loop
    %c_init = csltl %i0, %total
    jnz %c_init, @init_body
    jmp @init_done
@init_body
    %off_i0 = l mul %i0, 8
    %addr_i0 = l add r, %off_i0
    storel %zero_l, %addr_i0
    %i0 = l add %i0, 1
    jmp @init_loop
@init_done

    // main multiplication loops
    %i = l 0
@i_loop
    %c_i = csltl %i, na
    jnz %c_i, @i_body
    jmp @i_done
@i_body
    %off_ai = l mul %i, 8
    %addr_ai = l add a, %off_ai
    %a_val = l loadl %addr_ai

    %j = l 0
@j_loop
    %c_j = csltl %j, nb
    jnz %c_j, @j_body
    jmp @j_end
@j_body
    %off_bj = l mul %j, 8
    %addr_bj = l add b, %off_bj
    %b_val = l loadl %addr_bj

    // split a_val and b_val
    %x0 = l and %a_val, %mask32
    %x1 = l shr %a_val, 32
    %y0 = l and %b_val, %mask32
    %y1 = l shr %b_val, 32

    %p0 = l mul %x0, %y0
    %p1 = l mul %x0, %y1
    %p2 = l mul %x1, %y0
    %p3 = l mul %x1, %y1

    %mid = l add %p1, %p2
    %mid_low = l and %mid, %mask32
    %mid_high = l shr %mid, 32
    %mid_low_shift = l mul %mid_low, %two32
    %lo = l add %p0, %mid_low_shift
    %c1 = w cultl %lo, %p0
    %hi_tmp = l add %p3, %mid_high
    %hi = l add %hi_tmp, (zextw %c1)

    // pos = i + j
    %pos = l add %i, %j

    // add lo to r[pos]
    %off0 = l mul %pos, 8
    %addr0 = l add r, %off0
    %old0 = l loadl %addr0
    %sum0 = l add %old0, %lo
    %c_lo = w cultl %sum0, %old0
    storel %sum0, %addr0

    // add hi + c_lo to r[pos+1]
    %pos1 = l add %pos, 1
    %off1 = l mul %pos1, 8
    %addr1 = l add r, %off1
    %old1 = l loadl %addr1
    %sum1_tmp = l add %old1, %hi
    %c_hi = w cultl %sum1_tmp, %old1
    %sum1 = l add %sum1_tmp, (zextw %c_lo)
    %c2 = w cultl %sum1, %sum1_tmp
    %carry = w or %c_hi, %c2
    storel %sum1, %addr1

    // propagate carry if any
    %prop_idx = l add %pos1, 1
    %carry_w = %carry
@prop_loop
    %c_prop = csltl %prop_idx, %total
    jnz %c_prop, @prop_body
    jmp @prop_done
@prop_body
    %offp = l mul %prop_idx, 8
    %addrp = l add r, %offp
    %oldp = l loadl %addrp
    %sump = l add %oldp, (zextw %carry_w)
    %c_new = w cultl %sump, %oldp
    storel %sump, %addrp
    %carry_w = %c_new
    %prop_idx = l add %prop_idx, 1
    %c_zero = w eq %carry_w, %zero_l
    jnz %c_zero, @prop_done
    jmp @prop_loop
@prop_done

    %j = l add %j, 1
    jmp @j_loop
@j_end
    %i = l add %i, 1
    jmp @i_loop
@i_done

    // find significant length
    %k = l %total
@len_loop
    %c_kzero = w eq %k, %zero_l
    jnz %c_kzero, @len_done
    %k1 = l sub %k, 1
    %offk = l mul %k1, 8
    %addrk = l add r, %offk
    %valk = l loadl %addrk
    %c_nonzero = w neql %valk, %zero_l
    jnz %c_nonzero, @len_done
    %k = l sub %k, 1
    jmp @len_loop
@len_done
    ret %k
}
