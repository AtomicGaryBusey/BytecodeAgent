memmove_i64:
    cbz x2, .Lzero_ret
    cmp x0, x1
    bgt .Lbackward
.Lforward_loop:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    bne .Lforward_loop
    b .Lzero_ret
.Lbackward:
    add x3, x0, x2, lsl #3
    add x4, x1, x2, lsl #3
.Lbackward_loop:
    sub x3, x3, #8
    sub x4, x4, #8
    ldr x5, [x4]
    str x5, [x3]
    subs x2, x2, #1
    bne .Lbackward_loop
.Lzero_ret:
    mov x0, #0
    ret
