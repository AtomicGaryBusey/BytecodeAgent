dot_i32_simd:
    xor rax, rax
    xorps xmm0, xmm0
    mov r8, rdx
    mov rcx, rdx
    shr rcx, 1
    test rcx, rcx
    jz .Lscalar_tail
.Lvec_loop:
    movdqu xmm1, [rdi]
    movdqu xmm2, [rsi]
    pmuludq xmm1, xmm2
    paddq xmm0, xmm1
    add rdi, 8
    add rsi, 8
    dec rcx
    jnz .Lvec_loop
    movq rax, xmm0
    movhlps xmm1, xmm0
    movq rdx, xmm1
    add rax, rdx
.Lscalar_tail:
    mov rcx, r8
    and rcx, 1
    test rcx, rcx
    jz .Ldone
    movsxd rdx, dword ptr [rdi]
    movsxd rcx, dword ptr [rsi]
    imul rdx, rcx
    add rax, rdx
.Ldone:
    ret
