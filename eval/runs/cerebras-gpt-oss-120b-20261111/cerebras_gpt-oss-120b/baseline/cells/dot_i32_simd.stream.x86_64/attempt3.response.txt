dot_i32_simd:
    xor rax, rax
    xorps xmm5, xmm5
    mov r8, rdx
    mov rcx, rdx
    shr rcx, 2
    test rcx, rcx
    jz .Lscalar_tail
.Lvec_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pmuludq xmm0, xmm1
    paddq xmm5, xmm0
    pshufd xmm2, xmm2, 0xEE
    pshufd xmm3, xmm3, 0xEE
    pmuludq xmm2, xmm3
    paddq xmm5, xmm2
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lvec_loop
    movq rax, xmm5
    movhlps xmm0, xmm5
    movq rdx, xmm0
    add rax, rdx
.Lscalar_tail:
    mov rcx, r8
    and rcx, 3
    test rcx, rcx
    jz .Ldone
.Lscalar_loop:
    movsxd rdx, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul rdx, r9
    add rax, rdx
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Lscalar_loop
.Ldone:
    ret
