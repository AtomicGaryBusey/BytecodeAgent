dot_i32_simd:
    xor rax, rax
    xorps xmm0, xmm0
    mov rcx, rdx
    shr rcx, 2
    test rcx, rcx
    jz .Lscalar_tail
.Lvec_loop:
    movdqu xmm1, [rdi]
    movdqu xmm2, [rsi]
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lvec_loop
    movdqa xmm1, xmm0
    pshufd xmm1, xmm1, 0x4E
    paddq xmm0, xmm1
    movhlps xmm1, xmm0
    paddq xmm0, xmm1
    movq rax, xmm0
.Lscalar_tail:
    mov rcx, rdx
    and rcx, 3
    test rcx, rcx
    jz .Ldone
.Lscalar_loop:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Lscalar_loop
.Ldone:
    ret
