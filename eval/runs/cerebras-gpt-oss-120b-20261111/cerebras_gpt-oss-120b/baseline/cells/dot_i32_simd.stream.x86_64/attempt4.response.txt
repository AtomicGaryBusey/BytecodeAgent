dot_i32_simd:
    xor rax, rax
    xorps xmm5, xmm5
    mov r8, rdx
    mov rcx, rdx
    shr rcx, 2
    jz .Lscalar_tail
.Lvec_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuludq xmm2, xmm3
    paddq xmm5, xmm2
    pshufd xmm0, xmm0, 0xEE
    pshufd xmm1, xmm1, 0xEE
    pmovsxdq xmm4, xmm0
    pmovsxdq xmm6, xmm1
    pmuludq xmm4, xmm6
    paddq xmm5, xmm4
    add rdi, 16
    add rsi, 16
    dec rcx
    jne .Lvec_loop
    movq rax, xmm5
    movhlps xmm0, xmm5
    movq rdx, xmm0
    add rax, rdx
.Lscalar_tail:
    mov rcx, r8
    and rcx, 3
    jz .Ldone
.Lscalar_loop:
    movsxd rdx, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul rdx, r9
    add rax, rdx
    add rdi, 4
    add rsi, 4
    dec rcx
    jne .Lscalar_loop
.Ldone:
    ret
