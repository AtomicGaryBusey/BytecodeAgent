export function l $crc32(l %p, l %n)
@entry
%i =l const 0
%crc =w const 0xFFFFFFFF
jmp @loop_cond

@loop_cond
%cnd =l csltl %i, %n
jnz %cnd, @loop_body
jmp @after_loop

@loop_body
%addr =l add %p, %i
%byte =w loadub %addr
%crc =w xor %crc, %byte
%bit =l const 0
jmp @bit_loop_cond

@bit_loop_cond
%cndb =l csltl %bit, const 8
jnz %cndb, @bit_body
jmp @loop_inc

@bit_body
%lsb =w and %crc, const 1
%crc =w shr %crc, 1
%poly =w mul %lsb, const 0xEDB88320
%crc =w xor %crc, %poly
%bit =l add %bit, const 1
jmp @bit_loop_cond

@loop_inc
%i =l add %i, const 1
jmp @loop_cond

@after_loop
%crc =w xor %crc, const 0xFFFFFFFF
%res =l extuw %crc
ret %res
