map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rcx
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
    xor rbx, rbx
    test r15, r15
    jz .Ldone
.Lloop:
    mov rax, qword ptr [r14]
    mov rdi, rax
    call r12
    mov qword ptr [r13], rax
    mov rdx, qword ptr [r14]
    cmp rax, rdx
    jl .Linc
.Lnext:
    add r13, 8
    add r14, 8
    dec r15
    jnz .Lloop
    jmp .Ldone
.Linc:
    inc rbx
    jmp .Lnext
.Ldone:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
