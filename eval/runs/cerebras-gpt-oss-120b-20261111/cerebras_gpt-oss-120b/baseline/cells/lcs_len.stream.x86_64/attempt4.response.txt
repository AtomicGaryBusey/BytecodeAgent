lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, r8
    mov r9, rdx
    mov r10, rcx
    mov r11, rsi
    xor r12, r12
.Lzloop:
    cmp r12, r10
    jg .Lzdone
    mov qword ptr [rbx + r12*8], 0
    inc r12
    jmp .Lzloop
.Lzdone:
    xor r12, r12
.Louter:
    cmp r12, r11
    jge .Ldone
    movzx eax, byte ptr [rdi + r12]
    xor r14, r14
    xor r13, r13
.Linner:
    inc r13
    cmp r13, r10
    jg .Linner_done
    mov r15, qword ptr [rbx + r13*8]
    movzx edx, byte ptr [r9 + r13 - 1]
    cmp al, dl
    je .Lmatch
    mov rax, qword ptr [rbx + r13*8 - 8]
    mov rdx, r15
    cmp rax, rdx
    cmovg rdx, rax
    mov qword ptr [rbx + r13*8], rdx
    mov r14, r15
    jmp .Linner_next
.Lmatch:
    lea rax, [r14 + 1]
    mov qword ptr [rbx + r13*8], rax
    mov r14, r15
.Linner_next:
    jmp .Linner
.Linner_done:
    inc r12
    jmp .Louter
.Ldone:
    mov rax, qword ptr [rbx + r10*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
