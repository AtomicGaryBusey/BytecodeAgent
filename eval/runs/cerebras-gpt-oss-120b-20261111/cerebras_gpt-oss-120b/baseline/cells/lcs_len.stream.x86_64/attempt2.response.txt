lcs_len:
    push rbx
    push r9
    push r10
    push r11
    push r12
    push r13
    push r14
    push r15
    mov rbx, r8                # row pointer
    mov r9, rdx                # y pointer
    mov r10, rcx               # n
    mov r11, rsi               # m
    xor r12, r12               # i = 0
.Linit:
    cmp r12, r10
    jg .Linit_done
    mov qword ptr [rbx + r12*8], 0
    inc r12
    jmp .Linit
.Linit_done:
    xor r12, r12               # i = 0
.Louter:
    cmp r12, r11
    jge .Ldone_outer
    movzx eax, byte ptr [rdi + r12]   # x[i] into al
    xor r14, r14               # diag = 0
    mov r13, 1                 # j = 1
.Linner:
    cmp r13, r10
    jg .Linner_done
    mov r15, qword ptr [rbx + r13*8]   # up = row[j]
    movzx r14d, byte ptr [r9 + r13 - 1]   # y[j-1] into r14b
    cmp al, r14b
    jne .Lelse
    lea rax, [r14 + 1]         # diag + 1
    mov qword ptr [rbx + r13*8], rax
    mov r14, r15               # diag = up
    jmp .Linner_next
.Lelse:
    mov rax, qword ptr [rbx + r13*8 - 8]   # left = row[j-1]
    cmp rax, r15
    cmovg r15, rax            # r15 = max(left, up)
    mov qword ptr [rbx + r13*8], r15
    mov r14, r15              # diag = up (original up in r15 before max)
.Linner_next:
    inc r13
    jmp .Linner
.Linner_done:
    inc r12
    jmp .Louter
.Ldone_outer:
    mov rax, qword ptr [rbx + r10*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop r11
    pop r10
    pop r9
    pop rbx
    ret
