lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, r8          # row pointer
    mov r9, rdx          # y pointer
    mov r10, rcx         # n
    mov r11, rsi         # m
    xor r12, r12         # i = 0
.Lzero:
    cmp r12, r10
    ja .Lzero_done
    mov qword ptr [rbx + r12*8], 0
    inc r12
    jmp .Lzero
.Lzero_done:
    xor r12, r12         # i = 0
.Louter:
    cmp r12, r11
    jae .Ldone
    movzx eax, byte ptr [rdi + r12]
    xor r14, r14         # diag = 0
    xor r13, r13         # j = 0
.Linner:
    inc r13
    cmp r13, r10
    ja .Linner_done
    mov r15, qword ptr [rbx + r13*8]          # up = row[j]
    movzx edx, byte ptr [r9 + r13 - 1]        # y[j-1]
    cmp al, dl
    jne .Lelse
    lea rax, [r14 + 1]                        # diag + 1
    mov qword ptr [rbx + r13*8], rax
    mov r14, r15                              # diag = up
    jmp .Linner_next
.Lelse:
    mov rax, qword ptr [rbx + r13*8 - 8]      # left = row[j-1]
    mov rdx, r15
    cmp rax, rdx
    cmovg rdx, rax                           # rdx = max(left, up)
    mov qword ptr [rbx + r13*8], rdx
    mov r14, r15                              # diag = up
.Linner_next:
    jmp .Linner
.Linner_done:
    inc r12
    jmp .Louter
.Ldone:
    mov rax, qword ptr [rbx + r10*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
