lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, r8                # row pointer
    mov r12, rdx               # y pointer
    mov r13, rcx               # n
    mov r14, rsi               # m
    xor r15, r15               # idx = 0
.Linit_loop:
    cmp r15, r13
    jg .Linit_done
    mov qword ptr [rbx + r15*8], 0
    inc r15
    jmp .Linit_loop
.Linit_done:
    xor r15, r15               # i = 0
.Louter:
    cmp r15, r14
    jge .Ldone_outer
    movzx eax, byte ptr [rdi + r15]   # x[i] into al
    xor rdx, rdx               # diag = 0
    mov rcx, 1                 # j = 1
.Linner:
    cmp rcx, r13
    jg .Linner_done
    mov r11, qword ptr [rbx + rcx*8]         # up = row[j]
    movzx r9d, byte ptr [r12 + rcx - 1]      # y[j-1]
    cmp al, r9b
    jne .Lelse
    lea rax, [rdx + 1]                      # diag + 1
    mov qword ptr [rbx + rcx*8], rax
    mov rdx, r11                            # diag = up
    jmp .Linner_next
.Lelse:
    mov r10, qword ptr [rbx + rcx*8 - 8]    # row[j-1]
    mov rax, r11
    cmp r10, r11
    cmovg rax, r10                         # rax = max(row[j], row[j-1])
    mov qword ptr [rbx + rcx*8], rax
    mov rdx, r11                            # diag = up
.Linner_next:
    inc rcx
    jmp .Linner
.Linner_done:
    inc r15
    jmp .Louter
.Ldone_outer:
    mov rax, qword ptr [rbx + r13*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
