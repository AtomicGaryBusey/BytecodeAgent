qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi                # save original n
    mov r13, rdi                # save original base pointer
    cmp rsi, 1
    jle .Lreturn
    mov rax, rsi
    shr rax, 1                  # rax = n/2
    shl rax, 4                  # rax = (n/2)*16
    lea rdx, [rdi + rax]        # pivot address
    mov r8, qword ptr [rdx]     # pivot key
    mov r9, qword ptr [rdx + 8] # pivot value
    xor rbx, rbx                # i = 0
    mov rcx, rsi
    dec rcx                     # j = n-1
.Lpartition:
    # advance i while a[i] < pivot
.Lleft_loop:
    mov rax, rbx
    shl rax, 4
    lea rax, [rdi + rax]
    mov r10, qword ptr [rax]          # key_i
    mov r11, qword ptr [rax + 8]      # val_i
    mov r14, r10
    cmp r14, r8
    jl .Linc_i
    jg .Lstop_left
    cmp r11, r9
    jl .Linc_i
    jmp .Lstop_left
.Linc_i:
    inc rbx
    cmp rbx, rcx
    jle .Lleft_loop
.Lstop_left:
    # retreat j while a[j] > pivot
.Lright_loop:
    mov rax, rcx
    shl rax, 4
    lea rax, [rdi + rax]
    mov r10, qword ptr [rax]          # key_j
    mov r11, qword ptr [rax + 8]      # val_j
    mov r14, r10
    cmp r14, r8
    jg .Ldec_j
    jl .Lstop_right
    cmp r11, r9
    jg .Ldec_j
    jmp .Lstop_right
.Ldec_j:
    dec rcx
    cmp rbx, rcx
    jge .Lright_loop
.Lstop_right:
    cmp rbx, rcx
    jg .Lafter_partition
    # swap a[i] and a[j]
    mov rax, rbx
    shl rax, 4
    lea rax, [rdi + rax]        # addr_i
    mov rdx, rcx
    shl rdx, 4
    lea rdx, [rdi + rdx]        # addr_j
    mov r10, qword ptr [rax]           # key_i
    mov r11, qword ptr [rax + 8]       # val_i
    mov r14, qword ptr [rdx]           # key_j
    mov r15, qword ptr [rdx + 8]       # val_j
    mov qword ptr [rax], r14
    mov qword ptr [rax + 8], r15
    mov qword ptr [rdx], r10
    mov qword ptr [rdx + 8], r11
    inc rbx
    dec rcx
    jmp .Lpartition
.Lafter_partition:
    # left part size = rcx + 1
    mov rdx, rcx
    inc rdx                     # rdx = left_n
    cmp rdx, 1
    jle .Lskip_left
    mov rsi, rdx                # arg2 = left_n
    mov rdi, r13                # arg1 = base
    call qsort_pairs
.Lskip_left:
    # right part size = original_n - rbx
    mov rdx, r12
    sub rdx, rbx                # rdx = right_n
    cmp rdx, 1
    jle .Lreturn
    lea rdi, [r13 + rbx*16]     # arg1 = base + i*16
    mov rsi, rdx
