qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    cmp rsi, 1
    jle .Lreturn
    mov rax, rsi
    shr rax, 1
    shl rax, 4
    lea rdx, [rdi + rax]
    mov r8, qword ptr [rdx]
    mov r9, qword ptr [rdx + 8]
    xor rbx, rbx                 # i = 0
    mov rcx, rsi
    dec rcx                      # j = n-1
.Lpartition_loop:
    cmp rbx, rcx
    jg .Lafter_partition
.Linc_loop:
    mov rdx, rbx
    shl rdx, 4
    lea rdx, [rdi + rdx]
    mov r10, qword ptr [rdx]
    mov r11, qword ptr [rdx + 8]
    cmp r10, r8
    jl .Linc_step
    jg .Linc_done
    cmp r11, r9
    jl .Linc_step
    jmp .Linc_done
.Linc_step:
    inc rbx
    cmp rbx, rcx
    jle .Linc_loop
    jmp .Lafter_partition
.Linc_done:
    ; proceed to decrement side
