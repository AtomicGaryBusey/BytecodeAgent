qsort_pairs:
    push r12
    push r13
    push r14
    push r15
    cmp rsi, 1
    jle .Lreturn
    mov rax, rsi
    shr rax, 1
    mov rdx, rax
    shl rdx, 4
    lea rdx, [rdi + rdx]
    mov r8, [rdx]
    mov r9, [rdx + 8]
    xor rbx, rbx
    mov rcx, rsi
    dec rcx
.Lloop:
.Linc_i:
    mov rdx, rbx
    shl rdx, 4
    lea rdx, [rdi + rdx]
    mov rax, [rdx]
    mov r10, [rdx + 8]
    cmp rax, r8
    jl .Linc_i_done
    jg .Linc_i_done2
    cmp r10, r9
    jl .Linc_i_done
    jge .Linc_i_done2
.Linc_i_done:
    inc rbx
    cmp rbx, rcx
    jle .Linc_i
    jmp .Lafter_partition
.Linc_i_done2:
.Ldec_j:
    mov rdx, rcx
    shl rdx, 4
    lea rdx, [rdi + rdx]
    mov rax, [rdx]
    mov r10, [rdx + 8]
    cmp rax, r8
    jg .Ldec_j_done
    jl .Ldec_j_done2
    cmp r10, r9
    jg .Ldec_j_done
    jle .Ldec_j_done2
.Ldec_j_done:
    dec rcx
    cmp rbx, rcx
    jge .Lafter_partition
    jmp .Ldec_j
.Ldec_j_done2:
    cmp rbx, rcx
    jg .Lafter_partition
    mov rdx, rbx
    shl rdx, 4
    lea rdx, [rdi + rdx]
    mov rax, rcx
    shl rax, 4
    lea rax, [rdi + rax]
    mov r11, [rdx]
    mov r12, [rdx + 8]
    mov r13, [rax]
    mov r14, [rax + 8]
    mov [rdx], r13
    mov [rdx + 8], r14
    mov [rax], r11
    mov [rax + 8], r12
    inc rbx
    dec rcx
    jmp .Lloop
.Lafter_partition:
    mov r12, rbx
    mov r13, rsi
    mov rdx, r13
    sub rdx, r12
    mov r15, rdx
    mov r14, rdi
    mov rdx, r12
    shl rdx, 4
    add r14, rdx
    mov rdi, rdi
    mov rsi, r12
    cmp rsi, 1
    jle .Lskip_left
    sub rsp, 8
    call qsort_pairs
    add rsp, 8
.Lskip_left:
    mov rdi, r14
    mov rsi, r15
    cmp rsi, 1
    jle .Lskip_right
    sub rsp, 8
    call qsort_pairs
    add rsp, 8
.Lskip_right:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    ret
.Lreturn:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    ret
