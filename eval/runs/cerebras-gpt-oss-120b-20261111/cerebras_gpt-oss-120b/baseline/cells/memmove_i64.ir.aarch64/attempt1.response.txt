export function l $memmove_i64(l %dst, l %src, l %n) {
    %src_plus_n = addl %src, %n
    %cond1 = csltl %dst, %src
    %cond2 = csltl %src_plus_n, %dst
    %forward = or %cond1, %cond2
    jnz %forward, @forward, @backward

@forward:
    %i = copyl $0
@loop_fwd:
    %c = csltl %i, %n
    jnz %c, @body_fwd, @end_fwd
@body_fwd:
    %off = mull %i, $8
    %src_addr = addl %src, %off
    %val = loadl %src_addr
    %dst_addr = addl %dst, %off
    storel %val, %dst_addr
    %i = addl %i, $1
    jmp @loop_fwd
@end_fwd:
    jmp @ret

@backward:
    %i = copyl %n
@loop_back:
    %c = csgtl %i, $0
    jnz %c, @body_back, @end_back
@body_back:
    %i1 = subl %i, $1
    %off = mull %i1, $8
    %src_addr = addl %src, %off
    %val = loadl %src_addr
    %dst_addr = addl %dst, %off
    storel %val, %dst_addr
    %i = copyl %i1
    jmp @loop_back
@end_back:
    jmp @ret

@ret:
    %r = copyl $0
    ret %r
}
