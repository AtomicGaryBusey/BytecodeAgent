export function l memmove_i64(l %dst, l %src, l %n) {
@entry:
    %src_end_off = mull %n, $8
    %src_end = addl %src, %src_end_off
    %cond1 = csltl %dst, %src
    %cond2 = csltl %src_end, %dst
    %forward = or %cond1, %cond2
    jnz %forward, @forward
    jmp @backward

@forward:
    %i = copyl $0
@loop_fwd:
    %c = csltl %i, %n
    jnz %c, @body_fwd
    jmp @end_fwd
@body_fwd:
    %off = mull %i, $8
    %src_addr = addl %src, %off
    %val = loadl %src_addr
    %dst_addr = addl %dst, %off
    storel %val, %dst_addr
    %i = addl %i, $1
    jmp @loop_fwd
@end_fwd:
    jmp @ret

@backward:
    %i = copyl %n
@loop_back:
    %c = csltl $0, %i
    jnz %c, @body_back
    jmp @end_back
@body_back:
    %i1 = subl %i, $1
    %off = mull %i1, $8
    %src_addr = addl %src, %off
    %val = loadl %src_addr
    %dst_addr = addl %dst, %off
    storel %val, %dst_addr
    %i = copyl %i1
    jmp @loop_back
@end_back:
    jmp @ret

@ret:
    %r = copyl $0
    ret %r
}
