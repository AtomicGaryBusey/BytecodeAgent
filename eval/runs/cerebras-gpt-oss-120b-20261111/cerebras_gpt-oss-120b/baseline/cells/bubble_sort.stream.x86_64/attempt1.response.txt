bubble_sort:
    cmp rsi, 1
    jle .Lreturn
    mov r11, rsi
    dec r11
    xor rcx, rcx
.Louter:
    cmp rcx, r11
    jge .Lreturn
    mov r8, rsi
    sub r8, rcx
    dec r8
    xor r10, r10
.Linner:
    cmp r10, r8
    jge .Linner_done
    mov rax, qword ptr [rdi + r10*8]
    mov r9, qword ptr [rdi + r10*8 + 8]
    cmp rax, r9
    jle .Lskip_swap
    mov qword ptr [rdi + r10*8], r9
    mov qword ptr [rdi + r10*8 + 8], rax
.Lskip_swap:
    inc r10
    jmp .Linner
.Linner_done:
    inc rcx
    jmp .Louter
.Lreturn:
    xor rax, rax
    ret
