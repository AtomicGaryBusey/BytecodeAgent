vm_run:
    mov     x4, #0                     // pc = 0
    mov     x5, #0                     // depth = 0
.Lloop:
    cmp     x4, x1
    b.ge    .Lfinish
    ldr     x9, [x0, x4, lsl #3]       // opcode
    cmp     x9, #0
    b.eq    .Lhalt
    cmp     x9, #1
    b.eq    .Lpush
    cmp     x9, #2
    b.eq    .Ladd
    cmp     x9, #3
    b.eq    .Lsub
    cmp     x9, #4
    b.eq    .Lmul
    cmp     x9, #5
    b.eq    .Ldup
    cmp     x9, #6
    b.eq    .Lswap
    cmp     x9, #7
    b.eq    .Ldrop
    cmp     x9, #8
    b.eq    .Lover
    cmp     x9, #9
    b.eq    .Llt
    cmp     x9, #10
    b.eq    .Ljz
    cmp     x9, #11
    b.eq    .Ljmp
    // unknown opcode: advance pc
    add     x4, x4, #1
    b       .Lloop
.Lhalt:
    b       .Lfinish
.Lpush:
    // load immediate value
    add     x6, x4, #1
    ldr     x7, [x0, x6, lsl #3]
    // store to stack[depth]
    lsl     x8, x5, #3
    str     x7, [x2, x8]
    add     x5, x5, #1
    add     x4, x4, #2
    b       .Lloop
.Ladd:
    // a = stack[depth-2], b = stack[depth-1]
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // b
    sub     x6, x5, #2
    lsl     x6, x6, #3
    ldr     x8, [x2, x6]               // a
    add     x8, x8, x7
    str     x8, [x2, x6]               // store result
    sub     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Lsub:
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // b
    sub     x6, x5, #2
    lsl     x6, x6, #3
    ldr     x8, [x2, x6]               // a
    sub     x8, x8, x7
    str     x8, [x2, x6]
    sub     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Lmul:
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // b
    sub     x6, x5, #2
    lsl     x6, x6, #3
    ldr     x8, [x2, x6]               // a
    mul     x8, x8, x7
    str     x8, [x2, x6]
    sub     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Ldup:
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // top
    lsl     x8, x5, #3
    str     x7, [x2, x8]
    add     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Lswap:
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // b
    sub     x8, x5, #2
    lsl     x8, x8, #3
    ldr     x9, [x2, x8]               // a
    str     x7, [x2, x8]
    str     x9, [x2, x6]
    add     x4, x4, #1
    b       .Lloop
.Ldrop:
    sub     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Lover:
    sub     x6, x5, #2
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // value below top
    lsl     x8, x5, #3
    str     x7, [x2, x8]
    add     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Llt:
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // b
    sub     x6, x5, #2
    lsl     x6, x6, #3
    ldr     x8, [x2, x6]               // a
    cmp     x8, x7
    cset    w9, lt
    str     x9, [x2, x6]
    sub     x5, x5, #1
    add     x4, x4, #1
    b       .Lloop
.Ljz:
    // load top value
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x7, [x2, x6]               // v
    sub     x5, x5, #1
    cbz     x7, .Ljz_zero
    // not zero
    add     x4, x4, #2
    b       .Lloop
.Ljz_zero:
    add     x8, x4, #1
    ldr     x9, [x0, x8, lsl #3]       // t
    mov     x4, x9
    b       .Lloop
.Ljmp:
    add     x8, x4, #1
    ldr     x9, [x0, x8, lsl #3]       // t
    mov     x4, x9
    b       .Lloop
.Lfinish:
    cbz     x5, .Lret_zero
    sub     x6, x5, #1
    lsl     x6, x6, #3
    ldr     x0, [x2, x6]               // result
    b       .Lzero_fill
.Lret_zero:
    mov     x0, #0
.Lzero_fill:
    mov     x9, x5                     // i = depth
.Lzero_loop:
    cmp     x9, x3
    b.ge    .Lend
    lsl     x10, x9, #3
    str     xzr, [x2, x10]
    add     x9, x9, #1
    b       .Lzero_loop
.Lend:
    ret
