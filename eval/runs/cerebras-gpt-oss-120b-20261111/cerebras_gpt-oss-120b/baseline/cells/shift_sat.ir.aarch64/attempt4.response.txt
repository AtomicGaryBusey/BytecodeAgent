export function l $shift_sat(l %x, l %n) {
@entry:
    %c64 = constl 64
    %lt64 = csltl %n, %c64
    jnz %lt64, @small
@big:
    %c0 = constl 0
    %neg = csltl %x, %c0
    jnz %neg, @neg
@pos:
    %shl0 = constl 0
    %sar0 = constl 0
    %res0 = addl %shl0, %sar0
    ret %res0
@neg:
    %shl0 = constl 0
    %sar1 = constl -1
    %res1 = addl %shl0, %sar1
    ret %res1
@small:
    %shl = shl %x, %n
    %sar = sar %x, %n
    %res = addl %shl, %sar
    ret %res
}
