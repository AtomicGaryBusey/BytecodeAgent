export function l $shift_sat(l %x, l %n) {
@entry:
    %lt64 = csltl %n, 64
    jnz %lt64, @small, @big

@small:
    %shl = shl %x, %n
    %sar = sar %x, %n
    %res = addl %shl, %sar
    ret %res

@big:
    %neg = csltl %x, 0
    jnz %neg, @neg, @pos

@neg:
    %shl = constl 0
    %sar = constl -1
    %res = addl %shl, %sar
    ret %res

@pos:
    %shl = constl 0
    %sar = constl 0
    %res = addl %shl, %sar
    ret %res
}
