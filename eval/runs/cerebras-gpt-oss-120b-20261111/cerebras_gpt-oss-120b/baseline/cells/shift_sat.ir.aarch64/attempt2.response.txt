export function l $shift_sat(l %x, l %n) {
@entry:
    %lt64 = csltl %n, 64
    jnz %lt64, @shift_ok, @shift_big

@shift_ok:
    %shl = shll %x, %n
    %sar = sar %x, %n
    %res = addl %shl, %sar
    ret %res

@shift_big:
    %isneg = csltl %x, 0
    jnz %isneg, @neg_big, @pos_big

@neg_big:
    %shl = constl 0
    %sar = constl -1
    %res = addl %shl, %sar
    ret %res

@pos_big:
    %shl = constl 0
    %sar = constl 0
    %res = addl %shl, %sar
    ret %res
}
