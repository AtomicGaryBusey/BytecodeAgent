export function l $shift_sat(l %x, l %n) {
@entry:
    %lt64 = csltl %n, 64
    jnz %lt64, @then_shift, @else_shift

@then_shift:
    %shl = shll %x, %n
    %sar = sar %x, %n
    %res = addl %shl, %sar
    ret %res

@else_shift:
    %isneg = csltl %x, 0
    jnz %isneg, @else_neg, @else_pos

@else_neg:
    %right = constl -1
    %shl = constl 0
    %res = addl %shl, %right
    ret %res

@else_pos:
    %right = constl 0
    %shl = constl 0
    %res = addl %shl, %right
    ret %res
}
