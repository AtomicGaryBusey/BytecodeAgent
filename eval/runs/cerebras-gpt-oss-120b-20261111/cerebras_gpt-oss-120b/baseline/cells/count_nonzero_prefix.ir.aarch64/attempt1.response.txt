export function l $count_nonzero_prefix(l a, l n) {
@start
    constl 0 %i
    constl 0 %zero
    constl 1 %one
    constl 8 %eight
    jmp @loop

@loop
    csltl %i, %n, %cond_lt
    jnz %cond_lt, @load
    jmp @exit

@load
    mul %i, %eight, %off
    addl %a, %off, %addr
    loadl %addr, %val
    ceql %val, %zero, %is_zero
    jnz %is_zero, @exit
    addl %i, %one, %i_next
    copyl %i_next, %i
    jmp @loop

@exit
    retl %i
}
