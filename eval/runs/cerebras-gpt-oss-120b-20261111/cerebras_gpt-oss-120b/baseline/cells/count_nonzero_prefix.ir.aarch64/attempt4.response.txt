export function l $count_nonzero_prefix(l %a, l %n) {
@start
    constl 0, %i
    constl 0, %zero
    constl 1, %one
    constl 8, %eight
    jmp @cond

@cond
    csltl %i, %n, %c1
    jnz %c1, @load
    jmp @exit

@load
    mul %i, %eight, %off
    addl %a, %off, %addr
    loadl %addr, %val
    ceql %val, %zero, %c2
    jnz %c2, @exit
    addl %i, %one, %i_next
    copyl %i_next, %i
    jmp @cond

@exit
    retl %i
}
