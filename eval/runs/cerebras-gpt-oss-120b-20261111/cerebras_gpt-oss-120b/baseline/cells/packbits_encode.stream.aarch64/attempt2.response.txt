packbits_encode:
    // x0 = src, x1 = n, x2 = out, x3 = cap
    mov     x4, #0                     // i = 0
    mov     x5, #0                     // outpos = 0
.Louter:
    cmp     x4, x1
    b.cs    .Ldone
    // compute run length, capped at 128
    mov     x7, #1                     // run = 1
.Lrun:
    add     x8, x4, x7                 // idx = i + run
    cmp     x8, x1
    b.cs    .Lrun_done
    cmp     x7, #128
    b.hs    .Lrun_done
    ldrb    w12, [x0, x8]              // src[idx]
    ldrb    w13, [x0, x4]              // src[i]
    cmp     w12, w13
    b.ne    .Lrun_done
    add     x7, x7, #1
    b       .Lrun
.Lrun_done:
    cmp     x7, #3
    b.lt    .Lliteral_start
    // repeat packet
    sub     x8, x3, x5                 // remaining = cap - outpos
    cmp     x8, #2
    b.lt    .Lreturn_minus_one
    sub     w12, w7, #1                // w12 = run-1
    neg     w12, w12                   // header = -(run-1)
    strb    w12, [x2, x5]              // write header
    add     x5, x5, #1
    ldrb    w13, [x0, x4]              // repeat byte
    strb    w13, [x2, x5]              // write byte
    add     x5, x5, #1
    add     x4, x4, x7                 // i += run
    b       .Louter
.Lliteral_start:
    mov     x9, x4                     // lit_start = i
    mov     x10, #0                    // lit_len = 0
.Lliteral_loop:
    cmp     x4, x1
    b.cs    .Lemit_literal
    cmp     x10, #128
    b.hs    .Lemit_literal
    // check for run >=3 starting at i
    add     x8, x4, #2
    cmp     x8, x1
    b.cs    .Ladd_literal
    // load three bytes
    ldrb    w12, [x0, x4]              // src[i]
    add     x13, x4, #1
    ldrb    w13, [x0, x13]             // src[i+1]
    add     x14, x4, #2
    ldrb    w14, [x0, x14]             // src[i+2]
    cmp     w12, w13
    b.ne    .Ladd_literal
    cmp     w12, w14
    b.ne    .Ladd_literal
    // run of 3 found, stop literal
    b       .Lemit_literal
.Ladd_literal:
    add     x10, x10, #1               // lit_len++
    add     x4, x4, #1                 // i++
    b       .Lliteral_loop
.Lemit_literal:
    // lit_len is in x10, must be >=1
    // check output space: need 1+lit_len bytes
    add     x12, x10, #1               // needed = lit_len+1
    sub     x8, x3, x5                 // remaining = cap - outpos
    cmp     x8, x12
    b.lt    .Lreturn_minus_one
    // write header = lit_len-1
    sub     w13, w10, #1
    strb    w13, [x2, x5]
    add     x5, x5, #1
    // copy literal bytes
    mov     x11, #0                    // copy_offset = 0
.Lcopy_loop:
    cmp     x11, x10
    b.eq    .Lcopy_done
    add     x12, x9, x11               // src offset
    ldrb    w13, [x0, x12]
    add     x14, x5, x11               // out offset
    strb    w13, [x2, x14]
    add     x11, x11, #1
    b       .Lcopy_loop
.Lcopy_done:
    add     x5, x5, x10                // outpos += lit_len
    b       .Louter
.Lreturn_minus_one:
    mov     x0, #-1
    ret
.Ldone:
    mov     x0, x5
    ret
