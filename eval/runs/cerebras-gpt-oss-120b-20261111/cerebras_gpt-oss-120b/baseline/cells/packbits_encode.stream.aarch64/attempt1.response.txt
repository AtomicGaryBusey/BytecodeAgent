packbits_encode:
    mov     x4, #0                     // i = 0
    mov     x5, #0                     // outpos = 0
.Louter:
    cmp     x4, x1
    b.cs    .Ldone
    // compute run length (x7)
    mov     x7, #1
.Lrun:
    add     x8, x4, x7                 // idx = i + run
    cmp     x8, x1
    b.cs    .Lrun_done
    cmp     x7, #128
    b.hs    .Lrun_done
    ldrb    w12, [x0, x8]
    ldrb    w13, [x0, x4]
    cmp     w12, w13
    b.ne    .Lrun_done
    add     x7, x7, #1
    b       .Lrun
.Lrun_done:
    cmp     x7, #3
    b.lt    .Lliteral_start
    // repeat packet
    sub     x8, x3, x5                 // remaining = cap - outpos
    cmp     x8, #2
    b.lt    .Lreturn_minus_one
    sub     w12, w7, #1                // w12 = run-1
    neg     w12, w12                   // header = -(run-1)
    strb    w12, [x2, x5]              // write header
    add     x5, x5, #1
    ldrb    w13, [x0, x4]              // repeat byte
    strb    w13, [x2, x5]              // write byte
    add     x5, x5, #1
    add     x4, x4, x7                 // i += run
    b       .Louter
.Lliteral_start:
    mov     x11, #0                    // lit_len = 0
    mov     x9, x4                     // lit_start = i
.Lliteral_loop:
    cmp     x4, x1
    b.cs    .Lemit_literal
    cmp     x11, #128
    b.hs    .Lemit_literal
    // check for run >= 3 starting at i
    add     x8, x4, #2
    cmp     x8, x1
    b.hs    .Ladd_literal
    ldrb    w14, [x0, x4]
    ldrb    w15, [x0, x4, #1]
    ldrb    w16, [x0
