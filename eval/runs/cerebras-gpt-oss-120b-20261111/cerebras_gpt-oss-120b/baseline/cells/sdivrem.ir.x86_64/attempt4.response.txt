export function l $sdivrem(l %a, l %b, l %rem)
{
@entry
    %c0 = cnst l 0
    %cmin = cnst l 0x8000000000000000
    %cminus1 = cnst l -1

    %b_eq0 = ceql %b, %c0
    jnz %b_eq0, @b_zero

    %a_eq_min = ceql %a, %cmin
    %b_eq_m1 = ceql %b, %cminus1
    %cond2 = wand %a_eq_min, %b_eq_m1
    jnz %cond2, @min_div_minus1

    %q = div %a, %b
    %r = rem %a, %b
    storel %r, %rem
    ret %q

@b_zero
    storel %a, %rem
    ret %c0

@min_div_minus1
    %zero = cnst l 0
    storel %zero, %rem
    ret %cmin
}
