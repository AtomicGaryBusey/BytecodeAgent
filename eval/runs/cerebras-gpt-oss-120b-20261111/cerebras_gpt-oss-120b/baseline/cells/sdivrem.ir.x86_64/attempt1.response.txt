export function l $sdivrem(l %a, l %b, l %rem)
{
@entry
    %b_eq0 = ceql %b, 0
    jnz %b_eq0, @b_zero

    %a_eq_min = ceql %a, 0x8000000000000000
    %b_eq_m1 = ceql %b, -1
    %cond2 = wand %a_eq_min, %b_eq_m1
    jnz %cond2, @min_div_minus1

    %q = div %a, %b
    %r = rem %a, %b
    storel %r, %rem
    ret %q

@b_zero

    %r = %a
    storel %r, %rem
    ret %q

@min_div_minus1

    storel %r, %rem
    ret %q
}
