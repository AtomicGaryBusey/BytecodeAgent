packbits_decode:
    # rdi = src_ptr, rsi = n, rdx = out_ptr, rcx = cap
    mov     r8, rdi                 # r8 = src_ptr
    add     r8, rsi                 # r8 = src_end
    mov     r9, rdx                 # r9 = out_ptr
    add     r9, rcx                 # r9 = out_end
    xor     rax, rax                # rax = bytes_written = 0
.Lloop:
    cmp     rdi, r8
    jae     .Lreturn                # src exhausted
    mov     al, byte ptr [rdi]     # load header byte
    inc     rdi
    movsx   ebp, al                # sign‑extend header
    cmp     ebp, -128
    je      .Lloop                 # NOP
    cmp     ebp, 0
    jl      .Lrepeat               # negative header -> repeat packet
    # ---------- literal packet ----------
    # need = header + 1
    mov     edx, ebp
    add     edx, 1                  # edx = need (32‑bit)
    movzx   r10d, edx               # r10 = need (64‑bit)
