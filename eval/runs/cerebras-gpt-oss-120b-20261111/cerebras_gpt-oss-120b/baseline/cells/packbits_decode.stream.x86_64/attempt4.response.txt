packbits_decode:
    push    rbx
    mov     r8, rdi                 # r8 = src_end = src + n
    add     r8, rsi
    mov     r9, rdx                 # r9 = out_end = out + cap
    add     r9, rcx
    xor     rax, rax                # bytes_written = 0
.Lloop:
    cmp     rdi, r8
    jae     .Ldone
    movsx   r10d, byte ptr [rdi]    # signed header
    inc     rdi
    cmp     r10d, -128
    je      .Lloop                  # no-op packet
    test    r10d, r10d
    jl      .Lrepeat
    # literal packet
    mov     r11d, r10d
    add     r11d, 1                 # r11d = length
    lea     r12, [rdi + r11]        # src + length
    cmp     r12, r8
    ja      .Lsrc_err
    mov     r12, r9
    sub     r12, rdx                # remaining output space
    cmp     r11d, r12d
    jbe     .Lcopy_literal_full
    # overflow: copy only remaining space then return -1
    test    r12, r12
    je      .Lreturn_minus_one
    mov     rcx, r12
.Lcopy_literal_partial:
    mov     bl, byte ptr [rdi]
    mov     byte ptr [rdx], bl
    inc     rdi
    inc     rdx
    dec     rcx
    jnz     .Lcopy_literal_partial
    mov     rax, -1
    jmp     .Lexit
.Lcopy_literal_full:
    mov     rcx, r11
.Lcopy_literal_loop:
    mov     bl, byte ptr [rdi]
    mov     byte ptr [rdx], bl
    inc     rdi
    inc     rdx
    dec     rcx
    jnz     .Lcopy_literal_loop
    add     rax, r11
    jmp     .Lloop
.Lrepeat:
    # r10d is negative header
    mov     r11d, r10d
    neg     r11d
    inc     r11d                    # r11d = count = 1 - header
    cmp     rdi, r8
    jae     .Lsrc_err
    mov     bl, byte ptr [rdi]      # repeat value
    inc     rdi
    mov     r12, r9
    sub     r12, rdx                # remaining output space
    cmp     r11d, r12d
    jbe     .Lwrite_repeat_full
    test    r12, r12
    je      .Lreturn_minus_one
    mov     rcx, r12
.Lcopy_repeat_partial:
    mov     byte ptr [rdx], bl
    inc     rdx
    dec     rcx
    jnz     .Lcopy_repeat_partial
    mov     rax, -1
    jmp     .Lexit
.Lwrite_repeat_full:
    mov     rcx, r11
.Lcopy_repeat_loop:
    mov     byte ptr [rdx], bl
    inc     rdx
    dec     rcx
    jnz     .Lcopy_repeat_loop
    add     rax, r11
    jmp     .Lloop
.Lsrc_err:
    mov     rax, -2
    jmp     .Lexit
.Lreturn_minus_one:
    mov     rax, -1
    jmp     .Lexit
.Ldone:
    xor     rax, rax               # ensure zero if no bytes written
    # (rax already holds total bytes_written)
.Lexit:
    pop     rbx
    ret
