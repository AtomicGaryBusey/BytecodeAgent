packbits_decode:
    mov     r8, rdi                 # r8 = src_end = src + n
    add     r8, rsi
    mov     r9, rdx                 # r9 = out_end = out + cap
    add     r9, rcx
    xor     rax, rax                # bytes_written = 0
.Lloop:
    cmp     rdi, r8
    jae     .Ldone
    mov     al, byte ptr [rdi]
    inc     rdi
    movsx   r10d, al                # signed header in r10d
    cmp     r10d, -128
    je      .Lloop                  # no-op packet
    cmp     r10d, 0
    jl      .Lrepeat
    # literal packet
    add     r10d, 1                 # r10d = length
    lea     r11, [rdi + r10]        # src + length
    cmp     r11, r8
    ja      .Lsrc_err
    mov     r11, r9
    sub     r11, rdx                # remaining output space
    cmp     r10d, r11d
    jbe     .Lcopy_literal_full
    # overflow: copy only remaining space then return -1
    test    r11, r11
    je      .Lreturn_minus_one
    mov     rcx, r11
.Lcopy_literal_partial:
    mov     al, byte ptr [rdi]
    mov     byte ptr [rdx], al
    inc     rdi
    inc     rdx
    dec     rcx
    jnz     .Lcopy_literal_partial
    mov     rax, -1
    ret
.Lcopy_literal_full:
    mov     rcx, r10
.Lcopy_literal_loop:
    test    rcx, rcx
    je      .Lafter_literal
    mov     al, byte ptr [rdi]
    mov     byte ptr [rdx], al
    inc     rdi
    inc     rdx
    dec     rcx
    jmp     .Lcopy_literal_loop
.Lafter_literal:
    add     rax, r10
    jmp     .Lloop
.Lrepeat:
    # r10d is negative header
    mov     r10d, r10d
    neg     r10d
    inc     r10d                    # r10d = count = 1 - header
    cmp     rdi, r8
    jae     .Lsrc_err
    mov     sil, byte ptr [rdi]    # repeat value
    inc     rdi
    mov     r11, r9
    sub     r11, rdx                # remaining output space
    cmp     r10d, r11d
    jbe     .Lwrite_repeat_full
    test    r11, r11
    je      .Lreturn_minus_one
    mov     rcx, r11
.Lcopy_repeat_partial:
    mov     byte ptr [rdx], sil
    inc     rdx
    dec     rcx
    jnz     .Lcopy_repeat_partial
    mov     rax, -1
    ret
.Lwrite_repeat_full:
    mov     rcx, r10
.Lcopy_repeat_loop:
    test    rcx, rcx
    je      .Lafter_repeat
    mov     byte ptr [rdx], sil
    inc     rdx
    dec     rcx
    jmp     .Lcopy_repeat_loop
.Lafter_repeat:
    add     rax, r10
    jmp     .Lloop
.Lsrc_err:
    mov     rax, -2
    ret
.Lreturn_minus_one:
    mov     rax, -1
    ret
.Ldone:
    ret
