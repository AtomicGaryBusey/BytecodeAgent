packbits_decode:
    push rbp
    mov r8, rdi
    add r8, rsi
    mov r9, rdx
    add r9, rcx
    xor rax, rax
.Ltop:
    cmp rdi, r8
    jae .Ldone
    mov al, byte ptr [rdi]
    inc rdi
    movsx ebp, al
    cmp ebp, -128
    je .Ltop
    cmp ebp, 0
    jl .Lrepeat
    # literal packet
    lea ebx, [rbp + 1]          # need
    mov r10d, ebx               # save need
    mov r11, rdi
    add r11, rbx
    cmp r11, r8
    ja .Lsrc_err
    mov r11, rdx
    add r11, rbx
    cmp r11, r9
    ja .Lliteral_overflow
.Lcopy_lit:
    test ebx, ebx
    jz .Lafter_literal
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec ebx
    jmp .Lcopy_lit
.Lafter_literal:
    add rax, r10
    jmp .Ltop
.Lrepeat:
    mov ebx, 1
    sub ebx, ebp                # count
    mov r10d, ebx               # save count
    cmp rdi, r8
    jae .Lsrc_err
    mov bl, byte ptr [rdi]
    inc rdi
    mov r11, rdx
    add r11, rbx
    cmp r11, r9
    ja .Lrepeat_overflow
.Lwrite_rep:
    test ebx, ebx
    jz .Lafter_repeat
    mov byte ptr [rdx], bl
    inc rdx
    dec ebx
    jmp .Lwrite_rep
.Lafter_repeat:
    add rax, r10
    jmp .Ltop
.Lsrc_err:
    mov rax, -2
    pop rbp
    ret
.Lliteral_overflow:
    mov rcx, r9
    sub rcx, rdx                # rcx = remaining output
    test rcx, rcx
    jz .Lreturn_minus_one
.Lcopy_rem_loop:
    test rcx, rcx
    jz .Lreturn_minus_one
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec rcx
    jmp .Lcopy_rem_loop
.Lrepeat_overflow:
    mov rcx, r9
    sub rcx, rdx                # rcx = remaining output
    test rcx, rcx
    jz .Lreturn_minus_one
.Lfill_rep_loop:
    test rcx, rcx
    jz .Lreturn_minus_one
    mov byte ptr [rdx], bl
    inc rdx
    dec rcx
    jmp .Lfill_rep_loop
.Lreturn_minus_one:
    mov rax, -1
    pop rbp
    ret
.Ldone:
    pop rbp
    ret
