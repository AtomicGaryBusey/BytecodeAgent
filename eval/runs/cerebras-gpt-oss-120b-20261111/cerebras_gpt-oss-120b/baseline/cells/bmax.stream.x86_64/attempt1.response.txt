bmax:
    mov rdx, rsi                # rdx = n
    mov rax, qword ptr [rdi]    # rax = a[0] (initial max)
    dec rdx                     # rdx = n-1 (remaining elements)
    je .Ldone                   # if n==1, done
    lea rdi, [rdi + 8]          # rdi points to a[1]
.Lloop:
    mov rcx, qword ptr [rdi]    # rcx = current element
    cmp rcx, rax
    jle .Lskip
    mov rax, rcx                # update max
.Lskip:
    add rdi, 8                  # advance pointer
    dec rdx
    jne .Lloop
.Ldone:
    ret
