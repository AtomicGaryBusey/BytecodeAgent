export function l $bmax(l %a, l %n) {
@entry
    %zero = const l 0
    %off0 = mul l %zero 8
    %addr0 = add l %a %off0
    %max0 = loadl %addr0
    %i0 = const l 1
    jmp @loop

@loop
    %max = phi l %max0 @entry %max1 @body
    %i = phi l %i0 @entry %i1 @body
    %cond = csltl %i %n
    jnz %cond @body @exit

@body
    %off = mul l %i 8
    %addr = add l %a %off
    %cur = loadl %addr
    %gt = csltl %max %cur
    %max1 = sel %gt %cur %max
    %i1 = add l %i 1
    jmp @loop

@exit
    ret l %max
}
