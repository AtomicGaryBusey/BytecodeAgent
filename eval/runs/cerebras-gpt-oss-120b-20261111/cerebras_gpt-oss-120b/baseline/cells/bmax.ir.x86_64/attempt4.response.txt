export function l bmax(l %a, l %n) {
@entry
    %max0 = loadl %a
    %i0 = iconst l 1
    jmp @loop

@loop
    %max = phi l %max0 @entry %max1 @body
    %i = phi l %i0 @entry %i1 @body
    %cond = csltl %i %n
    jnz %cond @body @exit

@body
    %eight = iconst l 8
    %off = mul l %i %eight
    %addr = add l %a %off
    %cur = loadl %addr
    %gt = csltl %max %cur
    %max1 = sel l %gt %cur %max
    %i1 = add l %i 1
    jmp @loop

@exit
    ret l %max
}
