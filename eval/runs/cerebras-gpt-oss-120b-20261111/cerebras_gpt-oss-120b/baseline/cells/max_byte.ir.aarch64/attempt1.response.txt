export function l $max_byte(l %p, l %n) {
@entry
%zero =l const 0
%one =l const 1
%i =l phi %zero, @entry, %i_next, @loop
%max =l phi %zero, @entry, %max_next, @loop
%cond =w csltl %i, %n
jnz %cond, @loop
jmp @ret

@loop
%addr =l add %p, %i
%b =l loadub %addr
%cmp =w csltl %max, %b
%max_next =l sel %cmp, %b, %max
%i_next =l add %i, %one
jmp @entry

@ret
ret %max
}
