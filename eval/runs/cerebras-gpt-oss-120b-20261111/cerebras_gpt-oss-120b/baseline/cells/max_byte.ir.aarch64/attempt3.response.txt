export function l $max_byte(l %p, l %n) {
@entry
%zero_l =l 0
%zero_w =w 0
%one_l =l 1
jmp @loop

@loop
%i =l phi %zero_l @entry %i_next @loop
%max =w phi %zero_w @entry %max_next @loop
%cond =w csltl %i, %n
jnz %cond, @body
jmp @ret

@body
%addr =l add %p, %i
%b =w loadub %addr
%cmp =w csltw %max, %b
%max_next =w sel %cmp, %b, %max
%i_next =l add %i, %one_l
jmp @loop

@ret
%max_l =l cvtwl %max
ret %max_l
}
