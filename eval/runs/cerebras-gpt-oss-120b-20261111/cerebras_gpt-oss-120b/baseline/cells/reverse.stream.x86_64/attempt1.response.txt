reverse:
    mov rcx, 0
    mov rdx, rsi
    dec rdx
    cmp rcx, rdx
    jge .Ldone
.Lloop:
    mov rax, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rdx*8]
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rdx*8], rax
    inc rcx
    dec rdx
    cmp rcx, rdx
    jl .Lloop
.Ldone:
    xor rax, rax
    ret
