export function l $binary_search(l a, l n, l key) {
@entry
  %n1 = sub %n, 1
  %low0 = const 0
  %high0 = %n1
  jmp @loop %low0 %high0

@loop %low %high
  %cond = csltl %high %low
  jnz %cond, @notfound
  %sum = add %low %high
  %mid = shr %sum, 1
  %mid8 = mul %mid, 8
  %addr = add %a, %mid8
  %val = loadl %addr
  %eq = ceql %key %val
  jnz %eq, @found %mid
  %key_lt_val = csltl %key %val
  jnz %key_lt_val, @adjust_high %low %mid
  %mid1 = add %mid, 1
  jmp @loop %mid1 %high

@adjust_high %low %mid
  %mid1 = sub %mid, 1
  jmp @loop %low %mid1

@found %idx
  ret %idx

@notfound
  %neg1 = const -1
  ret %neg1
}
