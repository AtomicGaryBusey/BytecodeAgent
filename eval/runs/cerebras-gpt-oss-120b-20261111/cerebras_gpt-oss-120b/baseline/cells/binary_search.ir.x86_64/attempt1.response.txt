export function l $binary_search(l a, l n, l key) {
@entry
  %low0 = copy 0
  %n1 = sub %n, 1
  %high0 = copy %n1
  jmp @test

@test
  %cond_lt = csltl %high0 %low0
  jnz %cond_lt, @notfound
  %sum = add %low0 %high0
  %mid = shrl %sum, 1
  %mid8 = mul %mid, 8
  %addr = add %a, %mid8
  %val = loadl %addr
  %eq = ceql %key %val
  jnz %eq, @found
  %key_lt_val = csltl %key %val
  jnz %key_lt_val, @adjust_high
  %mid1 = add %mid, 1
  %low0 = copy %mid1
  jmp @test

@adjust_high
  %mid1 = sub %mid, 1
  %high0 = copy %mid1
  jmp @test

@found
  ret %mid

@notfound
  %neg1 = const -1
  ret %neg1
}
