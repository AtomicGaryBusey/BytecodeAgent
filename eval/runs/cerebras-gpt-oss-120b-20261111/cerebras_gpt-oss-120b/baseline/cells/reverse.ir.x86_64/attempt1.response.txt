export function l $reverse(l %a, l %n) {
@start:
    %zero = const l 0
    %one  = const l 1
    %half = csrl %n, %one
    %i    = const l 0
@loop:
    %cond = csltl %i, %half
    jz %cond, @ret
    %tmp1 = subl %n, %one
    %idx2 = subl %tmp1, %i
    %off1 = shll %i, 3
    %off2 = shll %idx2, 3
    %ptr1 = addl %a, %off1
    %ptr2 = addl %a, %off2
    %v1 = loadl %ptr1
    %v2 = loadl %ptr2
    storel %v2, %ptr1
    storel %v1, %ptr2
    %i = addl %i, %one
    jmp @loop
@ret:
    ret %zero
}
