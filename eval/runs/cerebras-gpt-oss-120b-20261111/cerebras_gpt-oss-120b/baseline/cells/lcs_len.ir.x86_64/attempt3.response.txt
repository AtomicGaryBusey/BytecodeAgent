export function l lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start:
    %zero =l const 0
    %one =l const 1
    %eight =l const 8
    %n1 =l add %n %one

    %j =l %zero
@init_loop:
    %cinit =w csltl %j %n1
    jnz %cinit @init_body
    jmp @outer_start
@init_body:
    %off =l mul %j %eight
    %ptr =l add %row %off
    storel %zero %ptr
    %j =l add %j %one
    jmp @init_loop

@outer_start:
    %i =l %zero
@outer_cond:
    %couter =w csltl %i %m
    jnz %couter @outer_body
    %off_n =l mul %n %eight
    %ptr_n =l add %row %off_n
    %res =l loadl %ptr_n
    ret %res

@outer_body:
    %xi_ptr =l add %x %i
    %diag =l %zero
    %j =l %one
@inner_cond:
    %cinner =w csltl %j %n1
    jnz %cinner @inner_body
    %i =l add %i %one
    jmp @outer_cond

@inner_body:
    %off_j =l mul %j %eight
    %ptr_j =l add %row %off_j
    %off_jm1 =l sub %off_j %eight
    %ptr_jm1 =l add %row %off_jm1

    %up =l loadl %ptr_j

    %xb =w loadub %xi_ptr
    %jm1 =l sub %j %one
    %yj_ptr =l add %y %jm1
    %yb =w loadub %yj_ptr

    %eq =w ceql %xb %yb
    jnz %eq @eq_case

@neq_case:
    %left =l loadl %ptr_jm1
    %lt =w csltl %left %up
    jnz %lt @use_up
@use_left:
    %new =l %left
    jmp @store_new
@use_up:
    %new =l %up
    jmp @store_new

@eq_case:
    %new =l add %diag %one

@store_new:
    storel %new %ptr_j

    %diag =l %up

    %j =l add %j %one
    jmp @inner_cond
}
