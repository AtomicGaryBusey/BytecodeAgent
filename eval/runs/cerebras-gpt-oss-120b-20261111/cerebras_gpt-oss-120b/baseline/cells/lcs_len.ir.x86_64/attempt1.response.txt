export function l $lcs_len(l %x, l %m, l %y, l %n, l %row)
@start
    %zero =l const 0
    %one =l const 1
    %eight =l const 8
    %j0 =l const 0
    %j1 =l const 1
    %i0 =l const 0
    %cond =w const 0

    // initialize row[0..n] = 0
    %j =l %j0
@init_loop
    %cmp_jn =csltl %j %n
    jnz %cmp_jn @init_body
    jmp @outer_start
@init_body
    %off =l mul %j %eight
    %ptr =l add %row %off
    storel %zero %ptr
    %j_next =l add %j %one
    %j =l %j_next
    jmp @init_loop

@outer_start
    %i =l %i0
@outer_loop_cond
    %cmp_i =csltl %i %m
    jnz %cmp_i @outer_body
    // finished, return row[n]
    %off_n =l mul %n %eight
    %ptr_n =l add %row %off_n
    %result =l loadl %ptr_n
    ret %result

@outer_body
    // compute base pointer for x[i]
    %xi_ptr =l add %x %i          // x[i] address (bytes)
    %diag =l %zero                // diag = 0
    %j =l %j1                     // j = 1
@inner_loop_cond
    %cmp_jn =csltl %j %n
    jnz %cmp_jn @inner_body
    // inner loop done, move to next i
    %i_next =l add %i %one
    %i =l %i_next
    jmp @outer_loop_cond

@inner_body
    // compute addresses for row[j] and row[j-1]
    %off_j =l mul %j %eight
    %ptr_j =l add %row %off_j
    %off_j1 =l mul %j %eight      // row[j-1] offset = (j-1)*8 = j*8 - 8
    %off_jm1 =l sub %off_j %eight
    %ptr_jm1 =l add %row %off_jm1

    // load current row values
    %up =l loadl %ptr_j

    // load bytes x[i] and y[j-1]
    %xb =w loadub %xi_ptr
    %jm1 =l sub %j %one
    %yj_ptr =l add %y %jm1
    %yb =w loadub %yj_ptr

    %eq =w ceql %xb %yb
    jnz %eq @eq_case

@neq_case
    %left =l loadl %ptr_jm1
    %lt =w csltl %left %up
    jnz %lt @use_up
@use_left
    %new =l %left
    jmp @store_new
@use_up
    %new =l %up
    jmp @store_new

@eq_case
    %new =l add %diag %one

@store_new
    storel %new %ptr_j

    // set diag = up for next iteration
    %diag =l %up

    // increment j
    %j_next =l add %j %one
    %j =l %j_next
    jmp @inner_loop_cond
