export function l $mat_pow(l a, l n, l k, l out, l tmp)
{
	%zero = const 0
	%one  = const 1
	%eight = const 8

	/* if k == 0 goto identity */
	%k_eq0 = ceql %k %zero
	jnz %k_eq0, .Lidentity

	/* cur = tmp ; acc = tmp + n*n*8 */
	%nn   = mul %n %n
	%nn8  = mul %nn %eight
	%cur  = %tmp
	%acc  = add %tmp %nn8

	/* copy a -> cur (nn elements) */
	%idxc = const 0
	%cntc = %nn
	br .Lcopy_cur
.Lcopy_cur:
	%condc = csltl %idxc %cntc
	jnz %condc, .Lcopy_cur_body
	br .Lcopy_cur_done
.Lcopy_cur_body:
	%offc   = mul %idxc %eight
	%src    = add %a %offc
	%val    = loadl %src
	%dst    = add %cur %offc
	storel %val %dst
	%idxc   = add %idxc %one
	br .Lcopy_cur
.Lcopy_cur_done:

	/* acc = identity */
	%ii = const 0
	br .Linit_acc_i
.Linit_acc_i:
	%condi = csltl %ii %n
	jnz %condi, .Linit_acc_i_body
	br .Linit_done
.Linit_acc_i_body:
	%jj = const 0
	br .Linit_acc_j
.Linit_acc_j:
	%condj = csltl %jj %n
	jnz %condj, .Linit_acc_j_body
	%ii = add %ii %one
	br .Linit_acc_i
.Linit_acc_j_body:
	%eqij = ceql %ii %jj
	%addr_off = mul (add (mul %ii %n) %jj) %eight
	%addr = add %acc %addr_off
	jnz %eqij, .Lstore_one
	storel %zero %addr
	br .Lnext_j
.Lstore_one:
	storel %one %addr
	br .Lnext_j
.Lnext_j:
	%jj = add %jj %one
	br .Linit_acc_j

	/* exponentiation loop */
	%exp_k = %k
	br .Lexp_loop
.Lexp_loop:
	%condk = csltl %exp_k %one
	jnz %condk, .Lexp_done
	%odd = and %exp_k %one
	%oddnz = csltl %zero %odd   // odd != 0
	jnz %oddnz, .Lmultiply_acc_cur
	/* else skip multiplication */
	br .Lskip_mul
.Lmultiply_acc_cur:
	/* acc = acc * cur -> tmp (reuse cur space as temp) */
	%temp = %cur               // temp will hold product
	%i0 = const 0
	br .Lmul_i
.Lmul_i:
	%cond_i0 = csltl %i0 %n
	jnz %cond_i0, .Lmul_i_body
	br .Lmul_i_done
.Lmul_i_body:
	%j0 = const 0
	br .Lmul_j
.Lmul_j:
	%cond_j0 = csltl %j0 %n
	jnz %cond_j0, .Lmul_j_body
	%i0 = add %i0 %one
	br .Lmul_i
.Lmul_j_body:
	%sum = const 0
	%t0 = const 0
	br .Lmul_t
.Lmul_t:
	%cond_t = csltl %t0 %n
	jnz %cond_t, .Lmul_t_body
	/* store sum */
	%offij = mul (add (mul %i0 %n) %j0) %eight
	%dest = add %temp %offij
	storel %sum %dest
	%j0 = add %j0 %one
	br .Lmul_j
.Lmul_t_body:
	%off_a = mul (add (mul %i0 %n) %t0) %eight
	%ptr_a = add %acc %off_a
	%val_a = loadl %ptr_a
	%off_b = mul (add (mul %t0 %n) %j0) %eight
	%ptr_b = add %cur %off_b
	%val_b = loadl %ptr_b
	%prod = mul %val_a %val_b
	%sum = add %sum %prod
	%t0 = add %t0 %one
	br .Lmul_t
.Lmul_i_done:
	/* copy temp back to acc */
	%idxc2 = const 0
	%cntc2 = %nn
	br .Lcopy_back
.Lcopy_back:
	%condc2 = csltl %idxc2 %cntc2
	jnz %condc2, .Lcopy_back_body
	br .Lskip_mul_done
.Lcopy_back_body:
	%offc2 = mul %idxc2 %eight
	%src2 = add %temp %offc2
	%val2 = loadl %src2
	%dst2 = add %acc %offc2
	storel %val2 %dst2
	%idxc2 = add %idxc2 %one
	br .Lcopy_back
.Lskip_mul_done:
	/* cur = cur * cur */
	%temp2 = %cur            // reuse cur as temp
	%i1 = const 0
	br .Lsq_i
.Lsq_i:
	%cond_i1 = csltl %i1 %n
	jnz %cond_i1, .Lsq_i_body
	br .Lsq_done
.Lsq_i_body:
	%j1 = const 0
	br .Lsq_j
.Lsq_j:
	%cond_j1 = csltl %j1 %n
	jnz %cond_j1, .Lsq_j_body
	%i1 = add %i1 %one
	br .Lsq_i
.Lsq_j_body:
	%sum1 = const 0
	%t1 = const 0
	br .Lsq_t
.Lsq_t:
	%cond_t1 = csltl %t1 %n
	jnz %cond_t1, .Lsq_t_body
	%offij1 = mul (add (mul %i1 %n) %j1) %eight
	%dest1 = add %temp2 %offij1
	storel %sum1 %dest1
	%j1 = add %j1 %one
	br .Lsq_j
.Lsq_t_body:
	%off_c1 = mul (add (mul %i1 %n) %t1) %eight
	%ptr_c1 = add %cur %off_c1
	%val_c1 = loadl %ptr_c1
	%off_c2 = mul (add (mul %t1 %n) %j1) %eight
	%ptr_c2 = add %cur %off_c2
	%val_c2 = loadl %ptr_c2
	%prod1 = mul %val_c1 %val_c2
	%sum1 = add %sum1 %prod1
	%t1 = add %t1 %one
	br .Lsq_t
.Lsq_done:
	/* copy temp2 back to cur */
	%idxc3 = const 0
	%cntc3 = %nn
	br .Lcopy_cur2
.Lcopy_cur2:
	%condc3 = csltl %idxc3 %cntc3
	jnz %condc3, .Lcopy_cur2_body
	br .Lexp_continue
.Lcopy_cur2_body:
	%offc3 = mul %idxc3 %eight
	%src3 = add %temp2 %offc3
	%val3 = loadl %src3
	%dst3 = add %cur %offc3
	storel %val3 %dst3
	%idxc3 = add %idxc3 %one
	br .Lcopy_cur2

.Lexp_continue:
	%exp_k = lshr %exp_k %one
	br .Lexp_loop
.Lexp_done:
	/* copy acc -> out */
	%idxo = const 0
	%cnto = %nn
	br .Lcopy_out
.Lcopy_out:
	%condo = csltl %idxo %cnto
	jnz %condo, .Lcopy_out_body
	br .Lzero_tmp
.Lcopy_out_body:
	%offo = mul %idxo %eight
	%srco = add %acc %offo
	%valo = loadl %srco
	%dsto = add %out %offo
	storel %valo %dsto
	%idxo = add %idxo %one
	br .Lcopy_out

	/* zero tmp (both matrices) */
.Lzero_tmp:
	%idxz = const 0
	%cntz = %nn8               /* bytes = n*n*8 */
	br .Lzero_loop
.Lzero_loop:
	%condz = csltl %idxz %cntz
	jnz %condz, .Lzero_body
	br .Lreturn
.Lzero_body:
	%addrz = add %tmp %idxz
	storel %zero %addrz
	%idxz = add %idxz %eight
	br .Lzero_loop

	/* identity case */
.Lidentity:
	%ii0 = const 0
	br .Lid_i
.Lid_i:
	%condii = csltl %ii0 %n
	jnz %condii, .Lid_i_body
	br .Lreturn
.Lid_i_body:
	%jj0 = const 0
	br .Lid_j
.Lid_j:
	%condjj = csltl %jj0 %n
	jnz %condjj, .Lid_j_body
	%ii0 = add %ii0 %one
	br .Lid_i
.Lid_j_body:
	%eqij0 = ceql %ii0 %jj0
	%addr_off0 = mul (add (mul %ii0 %n) %jj0) %eight
	%addr0 = add %out %addr_off0
	jnz %eqij0, .Lid_one
	storel %zero %addr0
	br .Lid_next_j
.Lid_one:
	storel %one %addr0
	br .Lid_next_j
.Lid_next_j:
	%jj0 = add %jj0 %one
	br .Lid_j

.Lreturn:
	ret %zero
}
