export function mat_pow(l a, l n, l k, l out, l tmp)
{
    %zero = const 0
    %one = const 1
    %eight = const 8
    %sixteen = const 16

    %nn = mul %n %n
    %nn8 = mul %nn %eight

    %k_eq0 = ceql %k %zero
    jnz %k_eq0 @Lidentity

    /* cur = tmp ; acc = tmp + nn*8 */
    %cur = %tmp
    %acc = add %tmp %nn8

    /* copy a -> cur (nn elements) */
    %idxc = const 0
    jmp @Lcopy_cur
@Lcopy_cur:
    %condc = csltl %idxc %nn
    jnz %condc @Lcopy_cur_body
    jmp @Lcopy_cur_done
@Lcopy_cur_body:
    %offc = mul %idxc %eight
    %srcc = add %a %offc
    %valc = loadl %srcc
    %dstc = add %cur %offc
    storel %valc %dstc
    %idxc = add %idxc %one
    jmp @Lcopy_cur
@Lcopy_cur_done:

    /* acc = identity */
    %ii = const 0
    jmp @Linit_i
@Linit_i:
    %cond_i = csltl %ii %n
    jnz %cond_i @Linit_i_body
    jmp @Linit_done
@Linit_i_body:
    %jj = const 0
    jmp @Linit_j
@Linit_j:
    %cond_j = csltl %jj %n
    jnz %cond_j @Linit_j_body
    %ii = add %ii %one
    jmp @Linit_i
@Linit_j_body:
    %eqij = ceql %ii %jj
    %off_ij = mul %ii %n
    %off_ij = add %off_ij %jj
    %off_ij = mul %off_ij %eight
    %addr = add %acc %off_ij
    jnz %eqij @Lstore_one
    storel %zero %addr
    jmp @Lnext_j
@Lstore_one:
    storel %one %addr
@Lnext_j:
    %jj = add %jj %one
    jmp @Linit_j
@Linit_done:

    /* exponentiation loop (repeated multiplication) */
    %expk = %k
    jmp @Lexp_loop
@Lexp_loop:
    %condk = csltl %expk %one
    jnz %condk @Lexp_done

    /* acc = acc * a -> cur */
    %left = %acc
    %right = %a
    %dest = %cur

    %i = const 0
    jmp @Lmul_i
@Lmul_i:
    %cond_i = csltl %i %n
    jnz %cond_i @Lmul_i_body
    jmp @Lmul_i_done
@Lmul_i_body:
    %j = const 0
    jmp @Lmul_j
@Lmul_j:
    %cond_j = csltl %j %n
    jnz %cond_j @Lmul_j_body
    %i = add %i %one
    jmp @Lmul_i
@Lmul_j_body:
    %sum = const 0
    %t = const 0
    jmp @Lmul_t
@Lmul_t:
    %cond_t = csltl %t %n
    jnz %cond_t @Lmul_t_body
    /* store sum */
    %off_ij = mul %i %n
    %off_ij = add %off_ij %j
    %off_ij = mul %off_ij %eight
    %addr_dest = add %dest %off_ij
    storel %sum %addr_dest
    %j = add %j %one
    jmp @Lmul_j
@Lmul_t_body:
    %off_a = mul %i %n
    %off_a = add %off_a %t
    %off_a = mul %off_a %eight
    %ptr_a = add %left %off_a
    %val_a = loadl %ptr_a
    %off_b = mul %t %n
    %off_b = add %off_b %j
    %off_b = mul %off_b %eight
    %ptr_b = add %right %off_b
    %val_b = loadl %ptr_b
    %prod = mul %val_a %val_b
    %sum = add %sum %prod
    %t = add %t %one
    jmp @Lmul_t
@Lmul_i_done:

    /* copy cur -> acc */
    %idxc2 = const 0
    jmp @Lcopy_back
@Lcopy_back:
    %condc2 = csltl %idxc2 %nn
    jnz %condc2 @Lcopy_back_body
    jmp @Lexp_continue
@Lcopy_back_body:
    %offc2 = mul %idxc2 %eight
    %srcc2 = add %cur %offc2
    %valc2 = loadl %srcc2
    %dstc2 = add %acc %offc2
    storel %valc2 %dstc2
    %idxc2 = add %idxc2 %one
    jmp @Lcopy_back
@Lexp_continue:
    %expk = sub %expk %one
    jmp @Lexp_loop

@Lexp_done:

    /* copy acc -> out */
    %idxo = const 0
    jmp @Lcopy_out
@Lcopy_out:
    %condo = csltl %idxo %nn
    jnz %condo @Lcopy_out_body
    jmp @Lzero_tmp
@Lcopy_out_body:
    %offo = mul %idxo %eight
    %srco = add %acc %offo
    %valo = loadl %srco
    %dsto = add %out %offo
    storel %valo %dsto
    %idxo = add %idxo %one
    jmp @Lcopy_out

    /* zero tmp (both matrices) */
@Lzero_tmp:
    %total_bytes = mul %nn %sixteen
    %idxz = const 0
    jmp @Lzero_loop
@Lzero_loop:
    %condz = csltl %idxz %total_bytes
    jnz %condz @Lzero_body
    jmp @Lreturn
@Lzero_body:
    %addrz = add %tmp %idxz
    storel %zero %addrz
    %idxz = add %idxz %eight
    jmp @Lzero_loop

    /* identity case */
@Lidentity:
    %ii0 = const 0
    jmp @Lid_i
@Lid_i:
    %condii = csltl %ii0 %n
    jnz %condii @Lid_i_body
    jmp @Lzero_tmp
@Lid_i_body:
    %jj0 = const 0
    jmp @Lid_j
@Lid_j:
    %condjj = csltl %jj0 %n
    jnz %condjj @Lid_j_body
    %ii0 = add %ii0 %one
    jmp @Lid_i
@Lid_j_body:
    %eqij0 = ceql %ii0 %jj0
    %off_ij0 = mul %ii0 %n
    %off_ij0 = add %off_ij0 %jj0
    %off_ij0 = mul %off_ij0 %eight
    %addr0 = add %out %off_ij0
    jnz %eqij0 @Lid_one
    storel %zero %addr0
    jmp @Lid_next_j
@Lid_one:
    storel %one %addr0
@Lid_next_j:
    %jj0 = add %jj0 %one
    jmp @Lid_j

@Lreturn:
    ret %zero
}
