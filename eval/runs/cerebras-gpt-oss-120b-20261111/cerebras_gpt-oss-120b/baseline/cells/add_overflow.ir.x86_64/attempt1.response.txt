export function l $add_overflow(l a, l b, l out) {
@start
%sum = addl %a, %b
storel %sum, %out

%zero_l = cnstl 0
%zero_w = cnstw 0
%one_w = cnstw 1

%a_neg = csltl %a, %zero_l
%b_neg = csltl %b, %zero_l
%sum_neg = csltl %sum, %zero_l

%xor_ab = xorw %a_neg, %b_neg
%same_sign = subw %one_w, %xor_ab

%xor_sum_a = xorw %sum_neg, %a_neg
%ov_w = andw %same_sign, %xor_sum_a

%ov_l = zextw %ov_w
ret %ov_l
}
