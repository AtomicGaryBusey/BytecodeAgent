export function l $map_sum(l %a, l %n, l %f) {
@entry
	%sum =l const 0
	%i =l const 0
	%c =w csltl %i, %n
	jnz %c, @loop
	ret %sum

@loop
	%off =l mulli %i, 8
	%addr =l addl %a, %off
	%elem =l loadl %addr
	%rv =l call %f(l %elem)
	%sum =l addl %sum, %rv
	%i =l addli %i, 1
	%c2 =w csltl %i, %n
	jnz %c2, @loop
	ret %sum
}
