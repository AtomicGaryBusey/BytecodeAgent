export function l $mat_pow(l a, l n, l k, l out, l tmp) {
    %a = arg 0
    %n = arg 1
    %k = arg 2
    %out = arg 3
    %tmp = arg 4

    %zero = const 0
    %one = const 1
    %size = mul %n, %n            // n*n
    %bytes = mul %size, const 8   // bytes per matrix
    %base = %tmp                   // base matrix at tmp
    %prod = add %tmp, %bytes       // product matrix at tmp+bytes

    // copy a -> base
    %i0 = const 0
    br %i0, @cpy_a

@cpy_a
    %i = phi %i0, @cpy_a, %i_next, @cpy_a_done
    %cnd_i = csltl %i, %size
    jnz %cnd_i, @cpy_body
    jmp @cpy_a_done

@cpy_body
    %off = shl %i, 3
    %src = add %a, %off
    %dst = add %base, %off
    %val = loadl %src
    storel %val, %dst
    %i_next = add %i, %one
    jmp @cpy_a

@cpy_a_done

    // init out to identity
    %j0 = const 0
    br %j0, @init_out

@init_out
    %j = phi %j0, @init_out, %j_next, @init_out_done
    %cnd_j = csltl %j, %size
    jnz %cnd_j, @init_body
    jmp @init_out_done

@init_body
    %n1 = add %n, %one
    %rem = mod %j, %n1
    %isdiag = ceql %rem, %zero
    %val = sel %isdiag, %one, %zero
    %offj = shl %j, 3
    %outptr = add %out, %offj
    storel %val, %outptr
    %j_next = add %j, %one
    jmp @init_out

@init_out_done

    // exponentiation loop
    %k_cur = phi %k, @loop_entry, %k_next, @loop_end
    %cnd_k = csltl %zero, %k_cur
    jnz %cnd_k, @loop_body
    jmp @after_loop

@loop_body
    // if (k & 1) out = out * base
    %k_and1 = and %k_cur, %one
    %is_one = ceql %k_and1, %one
    jnz %is_one, @mult_out
    jmp @skip_mult_out

@mult_out
    // prod = out * base
    // i loop
    %i0b = const 0
    br %i0b, @i_loop_mul

@i_loop_mul
    %i = phi %i0b, @i_loop_mul, %i_next, @i_done_mul
    %cnd_i = csltl %i, %n
    jnz %cnd_i, @j_loop_mul
    jmp @i_done_mul

@j_loop_mul
    %j1 = phi %zero, @j_loop_mul, %j_next, @j_done_mul
    %cnd_j = csltl %j1, %n
    jnz %cnd_j, @t_loop_start_mul
    // copy prod to out
    %c0 = const 0
    br %c0, @copy_prod_out

@copy_prod_out
    %ci = phi %c0, @copy_prod_out, %ci_next, @copy_done_out
    %cnd_c = csltl %ci, %size
    jnz %cnd_c, @copy_body_out
    jmp @copy_done_out

@copy_body_out
    %offc = shl %ci, 3
    %srcc = add %prod, %offc
    %dstc = add %out, %offc
    %v = loadl %srcc
    storel %v, %dstc
    %ci_next = add %ci, %one
    jmp @copy_prod_out

@copy_done_out
    %j_next = add %j1, %one
    jmp @j_loop_mul

@t_loop_start_mul
    %sum = const 0
    %t0 = const 0
    br %t0, @t_loop_mul

@t_loop_mul
    %t = phi %t0, @t_loop_mul, %t_next, @t_done_mul
    %cnd_t = csltl %t, %n
    jnz %cnd_t, @t_body_mul
    jmp @t_done_mul

@t_body_mul
    %i_n = mul %i, %n
    %i_n_t = add %i_n, %t
    %offL = shl %i_n_t, 3
    %Laddr = add %out, %offL
    %aval = loadl %Laddr

    %t_n = mul %t, %n
    %t_n_j = add %t_n, %j1
    %offR = shl %t_n_j, 3
    %Raddr = add %base, %offR
    %bval = loadl %Raddr

    %prodval = mul %aval, %bval
    %sum = add %sum, %prodval

    %t_next = add %t, %one
    jmp @t_loop_mul

@t_done_mul
    %i_n2 = mul %i, %n
    %i_n_j = add %i_n2, %j1
    %offD = shl %i_n_j, 3
    %Daddr = add %prod, %offD
    storel %sum, %Daddr
    %j_next = add %j1, %one
    jmp @j_loop_mul

@i_done_mul

    jmp @square_base

@skip_mult_out
    // no multiplication, just proceed to square base
    jmp @square_base

@square_base
    // prod = base * base
    %i0c = const 0
    br %i0c, @i_loop_sq

@i_loop_sq
    %i2 = phi %i0c, @i_loop_sq, %i2_next, @i2_done
    %cnd_i3 = csltl %i2, %n
    jnz %cnd_i3, @j_loop_sq
    jmp @i2_done

@j_loop_sq
    %j2 = phi %zero, @j_loop_sq, %j2_next, @j2_done
    %cnd_j3 = csltl %j2, %n
    jnz %cnd_j3, @t_loop_start_sq
    // copy prod to base
    %c1 = const 0
    br %c1, @copy_prod_base

@copy_prod_base
    %ci2 = phi %c1, @copy_prod_base, %ci2_next, @copy_base_done
    %cnd_c2 = csltl %ci2, %size
    jnz %cnd_c2, @copy_body_base
    jmp @copy_base_done

@copy_body_base
    %offc2 = shl %ci2, 3
    %srcc2 = add %prod, %offc2
    %dstc2 = add %base, %offc2
    %v2 = loadl %srcc2
    storel %v2, %dstc2
    %ci2_next = add %ci2, %one
    jmp @copy_prod_base

@copy_base_done
    %j2_next = add %j2, %one
    jmp @j_loop_sq

@t_loop_start_sq
    %sum2 = const 0
    %t0b = const 0
    br %t0b, @t_loop_sq

@t_loop_sq
    %t2 = phi %t0b, @t_loop_sq, %t2_next, @t2_done
    %cnd_t2 = csltl %t2, %n
    jnz %cnd_t2, @t_body_sq
    jmp @t2_done

@t_body_sq
    %i_n4 = mul %i2, %n
    %i_n_t2 = add %i_n4, %t2
    %offL2 = shl %i_n_t2, 3
    %Laddr2 = add %base, %offL2
    %aval2 = loadl %Laddr2

    %t_n4 = mul %t2, %n
    %t_n_j2 = add %t_n4, %j2
    %offR2 = shl %t_n_j2, 3
    %Raddr2 = add %base, %offR2
    %bval2 = loadl %Raddr2

    %prodval2 = mul %aval2, %bval2
    %sum2 = add %sum2, %prodval2

    %t2_next = add %t2, %one
    jmp @t_loop_sq

@t2_done
    %i_n5 = mul %i2, %n
    %i_n_j2b = add %i_n5, %j2
    %offD2 = shl %i_n_j2b, 3
    %Daddr2 = add %prod, %offD2
    storel %sum2, %Daddr2
    %j2_next = add %j2, %one
    jmp @j_loop_sq

@i2_done

    // shift k
    %k_next = shr %k_cur, %one
    jmp @loop_entry

@loop_entry
    %k_cur = phi %k, @loop_entry, %k_next, @loop_end

@loop_end

@after_loop
    // zero out tmp (2*size elements)
    %total = mul %size, const 2
    %z0 = const 0
    br %z0, @zero_loop

@zero_loop
    %z = phi %z0, @zero_loop, %z_next, @zero_done
    %cnd_z = csltl %z, %total
    jnz %cnd_z, @zero_body
    jmp @zero_done

@zero_body
    %offz = shl %z, 3
    %dstz = add %tmp, %offz
    storel %zero, %dstz
    %z_next = add %z, %one
    jmp @zero_loop

@zero_done
    ret %zero
}
