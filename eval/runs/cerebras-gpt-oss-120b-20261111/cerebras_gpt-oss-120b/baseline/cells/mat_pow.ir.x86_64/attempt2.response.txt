export function l mat_pow(l a, l n, l k, l out, l tmp) {
    %a = arg 0
    %n = arg 1
    %k = arg 2
    %out = arg 3
    %tmp = arg 4

    %zero = const 0
    %one = const 1
    %size = mul %n, %n            // n*n
    %bytes = mul %size, const 8   // bytes per matrix

    // copy a -> base (tmp)
    %i0 = const 0
    br %i0, @cpy_a

@cpy_a
    %i = phi %i0, @cpy_a, %i_next, @cpy_a_done
    %cnd_i = csltl %i, %size
    jnz %cnd_i, @cpy_body
    jmp @cpy_a_done

@cpy_body
    %off = shl %i, 3
    %src = add %a, %off
    %dst = add %tmp, %off
    %val = loadl %src
    storel %val, %dst
    %i_next = add %i, %one
    jmp @cpy_a

@cpy_a_done

    // init out to identity
    %j0 = const 0
    br %j0, @init_out

@init_out
    %j = phi %j0, @init_out, %j_next, @init_out_done
    %cnd_j = csltl %j, %size
    jnz %cnd_j, @init_body
    jmp @init_out_done

@init_body
    %n1 = add %n, %one
    %rem = mod %j, %n1
    %isdiag = ceql %rem, %zero
    %val = sel %isdiag, %one, %zero
    %offj = shl %j, 3
    %outptr = add %out, %offj
    storel %val, %outptr
    %j_next = add %j, %one
    jmp @init_out

@init_out_done

    // exponentiation loop
    %k_cur = phi %k, @loop_entry, %k_next, @loop_end
    %cnd_k = csltl %zero, %k_cur
    jnz %cnd_k, @loop_body
    jmp @after_loop

@loop_body
    // if (k & 1) out = out * base
    %k_and1 = and %k_cur, %one
    %is_one = ceql %k_and1, %one
    jnz %is_one, @mult_out
    jmp @skip_mult_out

@mult_out
    // prod = out * base
    %prod = add %tmp, %bytes
    // i loop
    %i0b = const 0
    br %i0b, @i_loop

@i_loop
    %i = phi %i0b, @i_loop, %i_next, @i_done
    %cnd_i2 = csltl %i, %n
    jnz %cnd_i2, @j_loop
    jmp @i_done

@j_loop
    %j1 = phi %zero, @j_loop, %j_next, @j_done
    %cnd_j2 = csltl %j1, %n
    jnz %cnd_j2, @t_loop_start
    // copy prod to out
    %c0 = const 0
    br %c0, @copy_prod_out

@copy_prod_out
    %ci = phi %c0, @copy_prod_out, %ci_next, @copy_done_out
    %cnd_c = csltl %ci, %size
    jnz %cnd_c, @copy_body_out
    jmp @copy_done_out

@copy_body_out
    %offc = shl %ci, 3
    %srcc = add %prod, %offc
    %dstc = add %out, %offc
    %v = loadl %srcc
    storel %v, %dstc
    %ci_next = add %ci, %one
    jmp @copy_prod_out

@copy_done_out
    %j_next = add %j1, %one
    jmp @j_loop

@t_loop_start
    %sum = const 0
    %t0 = const 0
    br %t0, @t_loop

@t_loop
    %t = phi %t0, @t_loop, %t_next, @t_done
    %cnd_t = csltl %t, %n
    jnz %cnd_t, @t_body
    jmp @t_done

@t_body
    %i_n = mul %i, %n
    %i_n_t = add %i_n, %t
    %offL = shl %i_n_t, 3
    %Laddr = add %out, %offL
    %aval = loadl %Laddr

    %t_n = mul %t, %n
    %t_n_j = add %t_n, %j1
    %offR = shl %t_n_j, 3
    %Raddr = add %tmp, %offR
    %bval = loadl %Raddr

    %prodval = mul %aval, %bval
    %sum = add %sum, %prodval

    %t_next = add %t, %one
    jmp @t_loop

@t_done
    %i_n2 = mul %i, %n
    %i_n_j = add %i_n2, %j1
    %offD = shl %i_n_j, 3
    %Daddr = add %prod, %offD
    storel %sum, %Daddr
    %j_next = add %j1, %one
    jmp @j_loop

@i_done

    // square base -> prod
    // prod already at %prod
    // i loop for square
    %i0c = const 0
    br %i0c, @i_loop_sq

@i_loop_sq
    %i2 = phi %i0c, @i_loop_sq, %i2_next, @i2_done
    %cnd_i3 = csltl %i2, %n
    jnz %cnd_i3, @j_loop_sq
    jmp @i2_done

@j_loop_sq
    %j2 = phi %zero, @j_loop_sq, %j2_next, @j2_done
    %cnd_j3 = csltl %j2, %n
    jnz %cnd_j3, @t_loop_sq_start
    // copy prod to base
    %c1 = const 0
    br %c1, @copy_prod_base

@copy_prod_base
    %ci2 = phi %c1, @copy_prod_base, %ci2_next, @copy_base_done
    %cnd_c2 = csltl %ci2, %size
    jnz %cnd_c2, @copy_body_base
    jmp @copy_base_done

@copy_body_base
    %offc2 = shl %ci2, 3
    %srcc2 = add %prod, %offc2
    %dstc2 = add %tmp, %offc2
    %v2 = loadl %srcc2
    storel %v2, %dstc2
    %ci2_next = add %ci2, %one
    jmp @copy_prod_base

@copy_base_done
    %j2_next = add %j2, %one
    jmp @j_loop_sq

@t_loop_sq_start
    %sum2 = const 0
    %t0b = const 0
    br %t0b, @t_loop_sq

@t_loop_sq
    %t2 = phi %t0b, @t_loop_sq, %t2_next, @t2_done
    %cnd_t2 = csltl %t2, %n
    jnz %cnd_t2, @t_body_sq
    jmp @t2_done

@t_body_sq
    %i_n4 = mul %i2, %n
    %i_n_t2 = add %i_n4, %t2
    %offL2 = shl %i_n_t2, 3
    %Laddr2 = add %tmp, %offL2
    %aval2 = loadl %Laddr2

    %t_n4 = mul %t2, %n
    %t_n_j2 = add %t_n4, %j2
    %offR2 = shl %t_n_j2, 3
    %Raddr2 = add %tmp, %offR2
    %bval2 = loadl %Raddr2

    %prodval2 = mul %aval2, %bval2
    %sum2 = add %sum2, %prodval2

    %t2_next = add %t2, %one
    jmp @t_loop_sq

@t2_done
    %i_n5 = mul %i2, %n
    %i_n_j2b = add %i_n5, %j2
    %offD2 = shl %i_n_j2b, 3
    %Daddr2 = add %prod, %offD2
    storel %sum2, %Daddr2
    %j2_next = add %j2, %one
    jmp @j_loop_sq

@i2_done

    // shift k
    %k_next = shr %k_cur, %one
    jmp @loop_entry

@skip_mult_out
    // square base without updating out
    %prod2 = add %tmp, %bytes
    // i loop for square into prod2
    %i0d = const 0
    br %i0d, @i_loop2

@i_loop2
    %i3 = phi %i0d, @i_loop2, %i3_next, @i3_done
    %cnd_i4 = csltl %i3, %n
    jnz %cnd_i4, @j_loop2
    jmp @i3_done

@j_loop2
    %j3 = phi %zero, @j_loop2, %j3_next, @j3_done
    %cnd_j4 = csltl %j3, %n
    jnz %cnd_j4, @t_loop2_start
    // copy prod2 to base
    %c2 = const 0
    br %c2, @copy_prod_base2

@copy_prod_base2
    %ci3 = phi %c2, @copy_prod_base2, %ci3_next, @copy_base_done2
    %cnd_c3 = csltl %ci3, %size
    jnz %cnd_c3, @copy_body_base2
    jmp @copy_base_done2

@copy_body_base2
    %offc3 = shl %ci3, 3
    %srcc3 = add %prod2, %offc3
    %dstc3 = add %tmp, %offc3
    %v3 = loadl %srcc3
    storel %v3, %dstc3
    %ci3_next = add %ci3, %one
    jmp @copy_prod_base2

@copy_base_done2
    %j3_next = add %j3, %one
    jmp @j_loop2

@t_loop2_start
    %sum3 = const 0
    %t0c = const 0
    br %t0c, @t_loop2

@t_loop2
    %t3 = phi %t0c, @t_loop2, %t3_next, @t3_done
    %cnd_t3 = csltl %t3, %n
    jnz %cnd_t3, @t_body2
    jmp @t3_done

@t_body2
    %i_n6 = mul %i3, %n
    %i_n_t3 = add %i_n6, %t3
    %offL3 = shl %i_n_t3, 3
    %Laddr3 = add %tmp, %offL3
    %aval3 = loadl %Laddr3

    %t_n6 = mul %t3, %n
    %t_n_j3 = add %t_n6, %j3
    %offR3 = shl %t_n_j3, 3
    %Raddr3 = add %tmp, %offR3
    %bval3 = loadl %Raddr3

    %prodval3 = mul %aval3, %bval3
    %sum3 = add %sum3, %prodval3

    %t3_next = add %t3, %one
    jmp @t_loop2

@t3_done
    %i_n7 = mul %i3, %n
    %i_n_j3b = add %i_n7, %j3
    %offD3 = shl %i_n_j3b, 3
    %Daddr3 = add %prod2, %offD3
    storel %sum3, %Daddr3
    %j3_next = add %j3, %one
    jmp @j_loop2

@i3_done
    // shift k
    %k_next = shr %k_cur, %one
    jmp @loop_entry

@loop_entry
    %k_cur = phi %k, @loop_entry, %k_next, @loop_end

@loop_end

@after_loop
    // zero out tmp (2*size elements)
    %total = mul %size, const 2
    %z0 = const 0
    br %z0, @zero_loop

@zero_loop
    %z = phi %z0, @zero_loop, %z_next, @zero_done
    %cnd_z = csltl %z, %total
    jnz %cnd_z, @zero_body
    jmp @zero_done

@zero_body
    %offz = shl %z, 3
    %dstz = add %tmp, %offz
    storel %zero, %dstz
    %z_next = add %z, %one
    jmp @zero_loop

@zero_done
    ret %zero
}
