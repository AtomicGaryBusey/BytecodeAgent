packbits_encode:
    xor r9, r9                     # written = 0
.Lmain_loop:
    cmp rsi, 0
    je .Lreturn_success
    cmp rsi, 3
    jl .Lliteral_start
    mov al, byte ptr [rdi]        # first byte
    cmp al, byte ptr [rDI+1]
    jne .Lliteral_start
    cmp al, byte ptr [rDI+2]
    jne .Lliteral_start
    xor r8, r8                     # run_len = 0
.Lrun_len_loop:
    cmp r8, 128
    jae .Lrun_len_done
    cmp r8, rsi
    jae .Lrun_len_done
    mov r10b, byte ptr [rdi+r8]
    cmp r10b, al
    jne .Lrun_len_done
    inc r8
    jmp .Lrun_len_loop
.Lrun_len_done:
    mov al, 1
    sub al, r8b                    # header = 1 - run_len
    cmp rcx, 2
    jl .Lfail
    mov byte ptr [rdx], al
    mov al, byte ptr [rdi]        # repeat byte
    mov byte ptr [rdx+1], al
    add rdi, r8
    add rdx, 2
    sub rsi, r8
    sub rcx, 2
    add r9, 2
    jmp .Lmain_loop
.Lliteral_start:
    xor r8, r8                     # lit_len = 0
.Lliteral_scan:
    cmp r8, rsi
    je .Lemit_literal
    cmp r8, 128
    jae .Lemit_literal
    mov rax, rsi
    sub rax, r8
    cmp rax, 3
    jl .Lcontinue_literal
    mov al, byte ptr [rdi+r8]
    cmp al, byte ptr [rdi+r8+1]
    jne .Lcontinue_literal
    cmp al, byte ptr [rdi+r8+2]
    jne .Lcontinue_literal
    jmp .Lemit_literal
.Lcontinue_literal:
    inc r8
    jmp .Lliteral_scan
.Lemit_literal:
    mov rax, r8
    inc rax                         # packet size = lit_len + 1
    cmp rcx, rax
    jl .Lfail
    mov al, r8b
    dec al
    mov byte ptr [rdx], al          # header = lit_len-1
    test r8, r8
    je .Lafter_copy
    xor rax, rax
.Lcopy_loop:
    mov r10b, byte ptr [rdi+rax]
    mov byte ptr [rdx+1+rax], r10b
    inc rax
    cmp rax, r8
    jne .Lcopy_loop
.Lafter_copy:
    add rdi, r8
    add rdx, r8
    inc rdx                         # advance past header
    sub rsi, r8
    sub rcx, r8
    dec rcx                         # header byte
    add r9, r8
    inc r9
    jmp .Lmain_loop
.Lfail:
    mov rax, -1
    ret
.Lreturn_success:
    mov rax, r9
    ret
