packbits_encode:
    push r12
    mov r12, rdx                 # out_start
    mov r8, rdi                  # src_ptr
    mov r9, rsi                  # n_remaining
    mov r10, rdx                 # out_ptr
    mov r11, rcx                 # cap_remaining
.Lmain_loop:
    cmp r9, 0
    je .Ldone
    cmp r9, 3
    jl .Lliteral_start
    mov al, byte ptr [r8]
    cmp al, byte ptr [r8+1]
    jne .Lliteral_start
    cmp al, byte ptr [r8+2]
    jne .Lliteral_start
    mov rax, 3                   # run length accumulator
.Lrun_len:
    cmp rax, 128
    jae .Lrun_done
    cmp rax, r9
    jae .Lrun_done
    mov sil, byte ptr [r8+rax]
    cmp sil, al
    jne .Lrun_done
    inc rax
    jmp .Lrun_len
.Lrun_done:
    cmp r11, 2
    jl .Lfail
    mov dl, 1
    sub dl, al                   # header = 1 - run_len
    mov byte ptr [r10], dl
    mov byte ptr [r10+1], al
    add r8, rax
    add r10, 2
    sub r9, rax
    sub r11, 2
    jmp .Lmain_loop
.Lliteral_start:
    xor rdx, rdx                 # literal_len = 0
.Lliteral_scan:
    cmp rdx, r9
    je .Lemit_literal
    cmp rdx, 128
    jae .Lemit_literal
    mov rcx, r9
    sub rcx, rdx
    cmp rcx, 3
    jl .Lcontinue_literal
    mov al, byte ptr [r8+rdx]
    mov sil, byte ptr [r8+rdx+1]
    cmp al, sil
    jne .Lcontinue_literal
    mov sil, byte ptr [r8+rdx+2]
    cmp al, sil
    jne .Lcontinue_literal
    jmp .Lemit_literal
.Lcontinue_literal:
    inc rdx
    jmp .Lliteral_scan
.Lemit_literal:
    mov rcx, rdx                 # rcx = literal_len
    inc rcx                      # packet size = 1 + literal_len
    cmp r11, rcx
    jl .Lfail
    sub dl, dl                   # clear dl
    mov dl, dlb?                 # placeholder removed
    mov dl, rdx
