packbits_encode:
    push r12
    mov r12, rdx                # out_start
    mov r8, rdi                 # src_ptr
    mov r9, rdx                 # dst_ptr
    mov r10, rsi                # remaining n
    mov r11, rcx                # remaining cap
.Lmain_loop:
    cmp r10, 0
    je .Ldone
    cmp r10, 3
    jl .Lliteral_start
    mov al, byte ptr [r8]
    cmp al, byte ptr [r8+1]
    jne .Lliteral_start
    cmp al, byte ptr [r8+2]
    jne .Lliteral_start
    mov rax, 3                  # run length accumulator
.Lrun_len:
    cmp rax, 128
    jae .Lrun_done
    cmp rax, r10
    jae .Lrun_done
    mov bl, byte ptr [r8+rax]
    cmp bl, al
    jne .Lrun_done
    inc rax
    jmp .Lrun_len
.Lrun_done:
    # rax = run length (>=3)
    cmp r11, 2
    jl .Lfail
    mov rcx, rax
    mov rdx, 257
    sub rdx, rcx                # header = 257 - run
    mov byte ptr [r9], dl
    mov byte ptr [r9+1], al
    add r8, rax
    add r9, 2
    sub r10, rax
    sub r11, 2
    jmp .Lmain_loop
.Lliteral_start:
    xor rdx, rdx                # literal_len = 0
.Lliteral_scan:
    cmp rdx, r10
    je .Lemit_literal
    cmp rdx, 128
    jae .Lemit_literal
    mov rcx, r10
    sub rcx, rdx
    cmp rcx, 3
    jl .Lcontinue_literal
    mov al, byte ptr [r8+rdx]
    cmp al, byte ptr [r8+rdx+1]
    jne .Lcontinue_literal
    cmp al, byte ptr [r8+rdx+2]
    jne .Lcontinue_literal
    jmp .Lemit_literal
.Lcontinue_literal:
    inc rdx
    jmp .Lliteral_scan
.Lemit_literal:
    mov rcx, rdx
    inc rcx                     # packet size = 1 + literal_len
    cmp r11, rcx
    jl .Lfail
    mov bl, dl
    sub bl, 1                   # header = literal_len - 1
    mov byte ptr [r9], bl
    mov rcx, rdx                # copy count = literal_len
    test rcx, rcx
    je .Lafter_copy
    mov rsi, r8
    lea rdi, [r9+1]
    rep movsb
.Lafter_copy:
    add r9, 1
    add r9, rdx
    add r8, rdx
    sub r10, rdx
    sub r11, rcx                # rcx = 1 + literal_len
    jmp .Lmain_loop
.Lfail:
    mov rax, -1
    pop r12
    ret
.Ldone:
    mov rax, r9
    sub rax, r12                # bytes written = dst_ptr - out_start
    pop r12
    ret
