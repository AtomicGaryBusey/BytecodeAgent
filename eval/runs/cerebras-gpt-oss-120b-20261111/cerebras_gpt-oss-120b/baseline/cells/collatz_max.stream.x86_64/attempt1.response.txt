collatz_max:
    xor rax, rax
    mov rdx, 1
outer_loop:
    cmp rdx, rdi
    jg end_outer
    mov r8, rdx
    xor rcx, rcx
inner_loop:
    cmp r8, 1
    je after_inner
    test r8, 1
    jnz odd_case
    shr r8, 1
    inc rcx
    jmp inner_loop
odd_case:
    lea r8, [r8*2 + r8 + 1]
    inc rcx
    jmp inner_loop
after_inner:
    cmp rcx, rax
    jle continue_outer
    mov rax, rcx
continue_outer:
    inc rdx
    jmp outer_loop
end_outer:
    ret
