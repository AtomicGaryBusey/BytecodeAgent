export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
	%i =l 0
	%pos =l 0
	jmp @loop

@loop:
	%c_i_n = csltl %i, %n
	jnz %c_i_n, @process, @exit

@process:
	%run =l 0
	%j =l %i
	jmp @run_check

@run_check:
	%c_j_n = csltl %j, %n
	%c_run_lim = csltl %run, 128
	%c_cond = andw %c_j_n, %c_run_lim
	jnz %c_cond, @run_body, @run_done

@run_body:
	%src_i_ptr =addl %src, %i
	%b_i =w loadub %src_i_ptr
	%src_j_ptr =addl %src, %j
	%b_j =w loadub %src_j_ptr
	%eq = ceql %b_i, %b_j
	jnz %eq, @run_inc, @run_done

@run_inc:
	%run =l addl %run, 1
	%j =l addl %j, 1
	jmp @run_check

@run_done:
	%c_ge3 = csltl 3, %run       ; run >= 3 ?
	jnz %c_ge3, @repeat_path, @literal_path

@repeat_path:
	%need2 =l 2
	%rem =l subl %cap, %pos
	%rem_need =l subl %rem, %need2
	%c_space = csltl 0, %rem_need
	jnz %c_space, @repeat_write, @out_of_space

@repeat_write:
	%header =w subw 1, %run
	%out_ptr =addl %out, %pos
	storeb %header, %out_ptr
	%out_ptr1 =addl %out_ptr, 1
	%src_i_ptr2 =addl %src, %i
	%b_i2 =w loadub %src_i_ptr2
	storeb %b_i2, %out_ptr1
	%pos =l addl %pos, %need2
	%i =l addl %i, %run
	jmp @loop

@literal_path:
	%len =l 0
	%j =l %i
	jmp @lit_check

@lit_check:
	%c_len_lim = csltl %len, 128
	%c_j_n = csltl %j, %n
	%c_cond2 = andw %c_len_lim, %c_j_n
	jnz %c_cond2, @lit_body, @lit_done

@lit_body:
	%j_plus2 =l addl %j, 2
	%c_jp2_n = csltl %j_plus2, %n
	jnz %c_jp2_n, @check_run3, @inc_literal

@check_run3:
	%src_j_ptr2 =addl %src, %j
	%b0 =w loadub %src_j_ptr2
	%b1 =w loadub addl %src_j_ptr2, 1
	%b2 =w loadub addl %src_j_ptr2, 2
	%eq01 = ceql %b0, %b1
	%eq02 = ceql %b0, %b2
	%run3 = andw %eq01, %eq02
	jnz %run3, @lit_done, @inc_literal

@inc_literal:
	%len =l addl %len, 1
	%j =l addl %j, 1
	jmp @lit_check

@lit_done:
	%need =l addl %len, 1
	%rem2 =l subl %cap, %pos
	%rem_need2 =l subl %rem2, %need
	%c_space2 = csltl 0, %rem_need2
	jnz %c_space2, @lit_write, @out_of_space

@lit_write:
	%header_l = w subw %len, 1
	%out_ptr_l =addl %out, %pos
	storeb %header_l, %out_ptr_l
	%k =l 0
	jmp @copy_check

@copy_check:
	%c_k_len = csltl %k, %len
	jnz %c_k_len, @copy_body, @copy_done

@copy_body:
	%src_off =addl %i, %k
	%src_ptr =addl %src, %src_off
	%b =w loadub %src_ptr
	%dst_off =addl %pos, 1
	%dst_off2 =addl %dst_off, %k
	%dst_ptr =addl %out, %dst_off2
	storeb %b, %dst_ptr
	%k =l addl %k, 1
	jmp @copy_check

@copy_done:
	%pos =l addl %pos, %need
	%i =l addl %i, %len
	jmp @loop

@out_of_space:
	%ret =l -1
	ret %ret

@exit:
	%ret =l %pos
	ret %ret
}
