export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %i =l 0
    %pos =l 0
    jmp @loop

@loop:
    %c_i_n = csltl %i, %n
    jz %c_i_n, @exit
    %src_i_ptr =addl %src, %i
    %b_i =w loadub %src_i_ptr

    ; compute run length starting at i, limited to 128
    %run =l 0
    %j =l %i
    jmp @runlen_loop

@runlen_loop:
    %c_j_n = csltl %j, %n
    jz %c_j_n, @runlen_done
    %src_j_ptr =addl %src, %j
    %b_j =w loadub %src_j_ptr
    %eq = ceql %b_i, %b_j
    jz %eq, @runlen_done
    %run =l addl %run, 1
    %j =l addl %j, 1
    %c_run_lim = csltl %run, 128
    jnz %c_run_lim, @runlen_loop
    jmp @runlen_done

@runlen_done:
    %c_ge3 = csltl 2, %run          ; run >= 3 ?
    jnz %c_ge3, @repeat_packet

    ; ---- literal packet ----
    %len =l 0
    %j =l %i
    jmp @lit_loop

@lit_loop:
    %c_len_lim = csltl %len, 128
    jz %c_len_lim, @lit_done
    %c_j_n = csltl %j, %n
    jz %c_j_n, @lit_done

    %n_minus2 =l subl %n, 2
    %c_j2 = csltl %j, %n_minus2
    jz %c_j2, @lit_continue   ; not enough bytes for a run of 3

    %ptr_j =addl %src, %j
    %b0 =w loadub %ptr_j
    %b1 =w loadub addl %ptr_j, 1
    %eq01 = ceql %b0, %b1
    %b2 =w loadub addl %ptr_j, 2
    %eq02 = ceql %b0, %b2
    %run_ge3 = andw %eq01, %eq02
    %c_run_ge3 = csltl %run_ge3, 1   ; true if run_ge3 == 0
    jnz %c_run_ge3, @lit_continue
    jmp @lit_done

@lit_continue:
    %len =l addl %len, 1
    %j =l addl %j, 1
    jmp @lit_loop

@lit_done:
    ; need len+1 bytes space
    %need =l addl %len, 1
    %rem =l subl %cap, %pos
    %rem1 =l addl %rem, 1
    %c_fit = csltl %need, %rem1
    jz %c_fit, @out_of_space

    ; header = len-1
    %header =w subw %len, 1
    %out_ptr =addl %out, %pos
    storeb %header, %out_ptr

    ; copy literal bytes
    %k =l 0
    jmp @copy_loop

@copy_loop:
    %c_k_len = csltl %k, %len
    jz %c_k_len, @copy_done
    %src_off =addl %i, %k
    %src_ptr =addl %src, %src_off
    %b =w loadub %src_ptr
    %dst_off =addl %pos, 1
    %dst_off2 =addl %dst_off, %k
    %dst_ptr =addl %out, %dst_off2
    storeb %b, %dst_ptr
    %k =l addl %k, 1
    jmp @copy_loop

@copy_done:
    %pos =l addl %pos, %need
    %i =l addl %i, %len
    jmp @loop

@repeat_packet:
    ; need 2 bytes space
    %need =l 2
    %rem =l subl %cap, %pos
    %rem1 =l addl %rem, 1
    %c_fit = csltl %need, %rem1
    jz %c_fit, @out_of_space

    %header =w subw 257, %run
    %out_ptr =addl %out, %pos
    storeb %header, %out_ptr
    %out_ptr1 =addl %out_ptr, 1
    storeb %b_i, %out_ptr1

    %pos =l addl %pos, 2
    %i =l addl %i, %run
    jmp @loop

@out_of_space:
    %ret =l -1
    ret %ret

@exit:
    %ret =l %pos
    ret %ret
}
