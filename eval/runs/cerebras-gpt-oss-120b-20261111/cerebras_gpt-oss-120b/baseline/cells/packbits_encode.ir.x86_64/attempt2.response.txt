export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %i =l 0
    %pos =l 0
    jmp @loop

@loop:
    %c_i_n = csltl %i, %n
    jmp %c_i_n, @main, @exit

@main:
    %src_i_ptr =addl %src, %i
    %b_i =w loadub %src_i_ptr

    ; compute run length
    %run =l 0
    %j =l %i
    jmp @run_check

@run_check:
    %c_j_n = csltl %j, %n
    %c_run_lim = csltl %run, 128
    %c_runcond = andw %c_j_n, %c_run_lim
    jmp %c_runcond, @run_body, @run_done

@run_body:
    %src_j_ptr =addl %src, %j
    %b_j =w loadub %src_j_ptr
    %eq = ceql %b_i, %b_j
    jmp %eq, @run_inc, @run_done

@run_inc:
    %run =l addl %run, 1
    %j =l addl %j, 1
    jmp @run_check

@run_done:
    %c_ge3 = csltl 2, %run        ; run >= 3 ?
    jmp %c_ge3, @repeat_packet, @literal_start

@repeat_packet:
    ; need 2 bytes
    %need2 =l 2
    %rem =l subl %cap, %pos
    %c_space = csltl %rem, %need2
    jmp %c_space, @out_of_space, @repeat_write

@repeat_write:
    %header =w subw 1, %run       ; 257 - run  == 1 - run (mod 256)
    %out_ptr =addl %out, %pos
    storeb %header, %out_ptr
    %out_ptr1 =addl %out_ptr, 1
    storeb %b_i, %out_ptr1
    %pos =l addl %pos, %need2
    %i =l addl %i, %run
    jmp @loop

@literal_start:
    %len =l 0
    %j =l %i
    jmp @lit_check

@lit_check:
    %c_len_lim = csltl %len, 128
    %c_j_n = csltl %j, %n
    %c_cond = andw %c_len_lim, %c_j_n
    jmp %c_cond, @lit_body, @lit_done

@lit_body:
    %n_minus2 =l subl %n, 2
    %c_enough = csltl %j, %n_minus2
    jmp %c_enough, @lit_check_run, @lit_done

@lit_check_run:
    %ptr_j =addl %src, %j
    %b0 =w loadub %ptr_j
    %b1 =w loadub addl %ptr_j, 1
    %b2 =w loadub addl %ptr_j, 2
    %eq01 = ceql %b0, %b1
    %eq02 = ceql %b0, %b2
    %run_ge3 = andw %eq01, %eq02
    jmp %run_ge3, @lit_done, @lit_continue

@lit_continue:
    %len =l addl %len, 1
    %j =l addl %j, 1
    jmp @lit_check

@lit_done:
    %need =l addl %len, 1
    %rem =l subl %cap, %pos
    %c_space = csltl %rem, %need
    jmp %c_space, @out_of_space, @lit_write

@lit_write:
    %header =w subw %len, 1
    %out_ptr =addl %out, %pos
    storeb %header, %out_ptr

    %k =l 0
    jmp @copy_check

@copy_check:
    %c_k_len = csltl %k, %len
    jmp %c_k_len, @copy_body, @copy_done

@copy_body:
    %src_off =addl %i, %k
    %src_ptr =addl %src, %src_off
    %b =w loadub %src_ptr
    %dst_off =addl %pos, 1
    %dst_off2 =addl %dst_off, %k
    %dst_ptr =addl %out, %dst_off2
    storeb %b, %dst_ptr
    %k =l addl %k, 1
    jmp @copy_check

@copy_done:
    %pos =l addl %pos, %need
    %i =l addl %i, %len
    jmp @loop

@out_of_space:
    %ret =l -1
    ret %ret

@exit:
    %ret =l %pos
    ret %ret
}
