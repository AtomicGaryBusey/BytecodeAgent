export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
	%i =l 0
	%pos =l 0
	@L0:
		%c_i_n = csltl %i, %n
		jnz %c_i_n, L_process, L_exit

	@L_process:
		%run =l 0
		%j =l %i
		@L_run_check:
			%c_j_n = csltl %j, %n
			%c_run_lim = csltl %run, 128
			%c_cond = andw %c_j_n, %c_run_lim
			jnz %c_cond, L_run_body, L_run_done

		@L_run_body:
			%src_i_ptr =addl %src, %i
			%b_i =w loadub %src_i_ptr
			%src_j_ptr =addl %src, %j
			%b_j =w loadub %src_j_ptr
			%eq = ceql %b_i, %b_j
			jnz %eq, L_run_inc, L_run_done

		@L_run_inc:
			%run =l addl %run, 1
			%j =l addl %j, 1
			jmp L_run_check

		@L_run_done:
			%c_lt3 = csltl %run, 3
			%c_ge3 = notw %c_lt3
			jnz %c_ge3, L_repeat_path, L_literal_path

		@L_repeat_path:
			%need =l 2
			%rem =l subl %cap, %pos
			%rem_need =l subl %rem, %need
			%c_space = csltl 0, %rem_need
			jnz %c_space, L_repeat_write, L_out_of_space

		@L_repeat_write:
			%header =w subw 1, %run
			%out_ptr =addl %out, %pos
			storeb %header, %out_ptr
			%out_ptr1 =addl %out_ptr, 1
			%src_i_ptr2 =addl %src, %i
			%b_i2 =w loadub %src_i_ptr2
			storeb %b_i2, %out_ptr1
			%pos =l addl %pos, %need
			%i =l addl %i, %run
			jmp L0

		@L_literal_path:
			%len =l 0
			%j2 =l %i
			@L_lit_check:
				%c_len_lim = csltl %len, 128
				%c_j_n2 = csltl %j2, %n
				%c_cond2 = andw %c_len_lim, %c_j_n2
				jnz %c_cond2, L_lit_body, L_lit_done

			@L_lit_body:
				%j_plus2 =l addl %j2, 2
				%c_jp2_n = csltl %j_plus2, %n
				jnz %c_jp2_n, L_check_run3, L_inc_literal

			@L_check_run3:
				%src_j_ptr2 =addl %src, %j2
				%b0 =w loadub %src_j_ptr2
				%b1 =w loadub addl %src_j_ptr2, 1
				%b2 =w loadub addl %src_j_ptr2, 2
				%eq01 = ceql %b0, %b1
				%eq02 = ceql %b0, %b2
				%run3 = andw %eq01, %eq02
				jnz %run3, L_lit_done, L_inc_literal

			@L_inc_literal:
				%len =l addl %len, 1
				%j2 =l addl %j2, 1
				jmp L_lit_check

			@L_lit_done:
				%need_l =l addl %len, 1
				%rem2 =l subl %cap, %pos
				%rem_need2 =l subl %rem2, %need_l
				%c_space2 = csltl 0, %rem_need2
				jnz %c_space2, L_lit_write, L_out_of_space

			@L_lit_write:
				%header_l =w subw %len, 1
				%out_ptr_l =addl %out, %pos
				storeb %header_l, %out_ptr_l
				%k =l 0
				@L_copy_check:
					%c_k_len = csltl %k, %len
					jnz %c_k_len, L_copy_body, L_copy_done

				@L_copy_body:
					%src_off =addl %i, %k
					%src_ptr =addl %src, %src_off
					%b =w loadub %src_ptr
					%dst_off =addl %pos, 1
					%dst_off2 =addl %dst_off, %k
					%dst_ptr =addl %out, %dst_off2
					storeb %b, %dst_ptr
					%k =l addl %k, 1
					jmp L_copy_check

				@L_copy_done:
					%pos =l addl %pos, %need_l
					%i =l addl %i, %len
					jmp L0

	@L_out_of_space:
		%ret =l -1
		ret %ret

	@L_exit:
		%ret =l %pos
		ret %ret
}
