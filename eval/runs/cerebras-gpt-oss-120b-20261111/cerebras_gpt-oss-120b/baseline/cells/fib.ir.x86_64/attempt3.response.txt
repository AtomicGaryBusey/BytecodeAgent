export function l $fib(l %n) {
@entry:
  %c2 = constl 2
  %c0 = constl 0
  %c1 = constl 1
  %cond = csltl %n, %c2
  jnz %cond, @ret_small
  %a = %c0
  %b = %c1
  %i = %c2
  jmp @loop
@loop:
  %exit = csgtl %i, %n
  jnz %exit, @after
  %c = addl %a, %b
  %a = %b
  %b = %c
  %i = addl %i, %c1
  jmp @loop
@after:
  ret %b
@ret_small:
  ret %n
}
