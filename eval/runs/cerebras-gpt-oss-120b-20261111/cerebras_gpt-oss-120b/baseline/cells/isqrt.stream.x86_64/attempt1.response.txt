isqrt:
    xor r9, r9
    mov rcx, 0x100000000
.Lloop:
    cmp r9, rcx
    jae .Ldone
    lea r8, [r9 + rcx]
    shr r8, 1
    mov rax, r8
    mul r8
    cmp rdx, 0
    jne .Lsethigh
    cmp rax, rdi
    jbe .Lsetlow
.Lsethigh:
    mov rcx, r8
    jmp .Lloop
.Lsetlow:
    lea r9, [r8 + 1]
    jmp .Lloop
.Ldone:
    lea rax, [r9 - 1]
    ret
