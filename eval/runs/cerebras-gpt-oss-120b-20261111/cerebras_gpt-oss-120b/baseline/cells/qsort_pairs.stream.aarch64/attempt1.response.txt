qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    // save original parameters
    mov x21, x0
    mov x22, x1
    // base case n <= 1
    cmp x1, #1
    ble .Lreturn0
    // pivot = middle element
    lsr x2, x1, #1
    lsl x3, x2, #4
    add x4, x0, x3
    ldr x19, [x4]          // pivot key
    ldr x20, [x4, #8]      // pivot value
    // i = -1, j = n
    mov x9, #-1
    mov x10, x1
.Linc_loop:
    add x9, x9, #1
    lsl x12, x9, #4
    add x5, x0, x12
    ldr x6, [x5]           // key_i
    ldr x7, [x5, #8]       // val_i
    cmp x6, x19
    blt .Linc_loop
    bgt .Ldec_loop
    cmp x7, x20
    blt .Linc_loop
    // key_i == pivot_key && val_i >= pivot_val
.Ldec_loop:
    sub x10, x10, #1
    lsl x12, x10, #4
    add x13, x0, x12
    ldr x14, [x13]         // key_j
    ldr x15, [x13, #8]     // val_j
    cmp x14, x19
    bgt .Ldec_loop
    blt .Lcompare_swap
    cmp x15, x20
    bgt .Ldec_loop
    // key_j == pivot_key && val_j <= pivot_val
.Lcompare_swap:
    cmp x9, x10
    bge .Lpartition_done
    // swap records i and j
    str x14, [x5]
    str x15, [x5, #8]
    str x6, [x13]
    str x7, [x13, #8]
    b .Linc_loop
.Lpartition_done:
    // left_n = j + 1
    add x11, x10, #1
    // right_n = original_n - left_n
    sub x12, x22, x11
    // offset = left_n * 16
    lsl x13, x11, #4
    // recurse on left part
    cmp x11, #1
    ble .Lskip_left
    mov x0, x21
    mov x1, x11
    bl qsort_pairs
.Lskip_left:
    // recurse on right part
    cmp x12, #1
    ble .Lreturn0
    add x0, x21, x13
    mov x1, x12
    bl qsort_pairs
.Lreturn0:
    mov x0, #0
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
