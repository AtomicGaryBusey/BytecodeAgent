    stp x29, x30, [sp, #0]
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    mov x29, sp
    mov x21, x0
    mov x22, x1
    cmp x22, #1
    ble .Lexit
    lsr x2, x22, #1
    lsl x3, x2, #4
    add x4, x21, x3
    ldr x19, [x4]
    ldr x20, [x4, #8]
    mov x9, #-1
    mov x10, x22
.Lpartition_loop:
.Linc:

    lsl x11, x9, #4
    add x5, x21, x11
    ldr x6, [x5]
    ldr x7, [x5, #8]
    cmp x6, x19
    blt .Linc
    bgt .Ldec
    cmp x7, x20
    blt .Linc
    b .Ldec
.Ldec:
    sub x10, x10, #1
    lsl x12, x10, #4
    add x13, x21, x12
    ldr x14, [x13]
    ldr x15, [x13, #8]
    cmp x14, x19
    bgt .Ldec
    blt .Lswap_check
    cmp x15, x20
    bgt .Ldec
.Lswap_check:
    cmp x9, x10
    bge .Lpartition_done
    str x14, [x5]
    str x15, [x5, #8]
    str x6, [x13]
    str x7, [x13, #8]
    b .Linc
.Lpartition_done:
    add x11, x10, #1
    sub x12, x22, x11
    cmp x11, #1
    ble .Lskip_left
    mov x0, x21
    mov x1, x11
    bl qsort_pairs
.Lskip_left:
    cmp x12, #1
    ble .Lexit
    lsl x13, x11, #4
    add x0, x21, x13
    mov x1, x12
    bl qsort_pairs
.Lexit:
    mov x0, #0
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp, #0]
    add sp, sp, #64
    ret
