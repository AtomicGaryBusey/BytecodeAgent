qsort_pairs:
    sub sp, sp, #48
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]
    str x30, [sp, #32]

    mov x21, x0            // base pointer
    mov x22, x1            // n

    cmp x1, #1
    ble .Lreturn

    // pivot = middle record
    lsr x2, x1, #1
    lsl x3, x2, #4
    add x4, x0, x3
    ldr x19, [x4]          // pivot key
    ldr x20, [x4, #8]      // pivot value

    // i = -1, j = n
    mov x9, #-1            // i
    mov x10, x1            // j = n

.Lpartition_loop:
    // increment i
.Linc_loop:
    add x9, x9, #1
    lsl x11, x9, #4
    add x5, x0, x11
    ldr x6, [x5]           // key_i
    ldr x7, [x5, #8]       // val_i
    cmp x6, x19
    blt .Linc_loop
    bgt .Ldec_loop
    cmp x7, x20
    blt .Linc_loop
    b .Ldec_loop

.Ldec_loop:
    sub x10, x10, #1
    lsl x12, x10, #4
    add x13, x0, x12
    ldr x14, [x13]         // key_j
    ldr x15, [x13, #8]     // val_j
    cmp x14, x19
    bgt .Ldec_loop
    blt .Lcompare_swap
    cmp x15, x20
    bgt .Ldec_loop

.Lcompare_swap:
    cmp x9, x10
    bge .Lpartition_done
    // swap records i and j
    str x14, [x5]
    str x15, [x5, #8]
    str x6, [x13]
    str x7, [x13, #8]
    b .Linc_loop

.Lpartition_done:
    // left_n = j + 1
    add x11, x10, #1
    // right_n = n - left_n
    sub x12, x22, x11
    // offset = left_n * 16
    lsl x13, x11, #4

    // recurse left part if left_n > 1
    cmp x11, #1
    ble .Lskip_left
    mov x0, x21
    mov x1, x11
    bl qsort_pairs
.Lskip_left:
    // recurse right part if right_n > 1
    cmp x12, #1
    ble .Lreturn
    add x0, x21, x13
    mov x1, x12
    bl qsort_pairs

.Lreturn:
    mov x0, #0
    ldr x30, [sp, #32]
    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    add sp, sp, #48
    ret
