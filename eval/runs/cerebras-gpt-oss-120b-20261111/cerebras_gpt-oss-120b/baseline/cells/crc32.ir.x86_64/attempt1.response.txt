export function l $crc32(l %p, l %n) {
@start:
    %n32 =w extuw %n
    %crc =w const 0xffffffff
    %i =w const 0
    jmp @cond

@cond:
    %c =w csltl %i, %n32
    jnz %c, @body
    jmp @done

@body:
    %i64 =l extuw %i
    %addr =l add %p, %i64
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %k =w const 0
    jmp @inner_cond

@inner_cond:
    %c2 =w csltl %k, const 8
    jnz %c2, @inner_body
    jmp @inner_end

@inner_body:
    %lsb =w and %crc, const 1
    %crc_shr =w shr %crc, const 1
    %poly =w mul %lsb, const 0xEDB88320
    %crc =w xor %crc_shr, %poly
    %k =w add %k, const 1
    jmp @inner_cond

@inner_end:
    %i =w add %i, const 1
    jmp @cond

@done:
    %crc =w xor %crc, const 0xffffffff
    %res =l extuw %crc
    ret %res
}
