export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
  %one = 1
  %n1 = addl %n, %one

  %j = 0
  %zero = 0
@init_loop
  %c = csltl %j %n1
  jnz %c, @init_body
  jmp @init_done
@init_body
  %off = shll %j, 3
  %addr = addl %row, %off
  storel %zero, %addr
  %j = addl %j, 1
  jmp @init_loop
@init_done

  %i = 0
@outer_loop
  %c2 = csltl %i %m
  jnz %c2, @outer_body
  jmp @outer_done
@outer_body
  %diag = 0
  %j = 1
@inner_loop
  %c3 = csltl %j %n1
  jnz %c3, @inner_body
  jmp @inner_done
@inner_body
  %xp = addl %x, %i
  %a =w loadub %xp
  %j1 = subl %j, 1
  %yp = addl %y, %j1
  %b =w loadub %yp
  %eq = ceql %a %b
  %offj = shll %j, 3
  %addrj = addl %row, %offj
  %up = loadl %addrj
  %offjm1 = shll %j1, 3
  %addrjm1 = addl %row, %offjm1
  %left = loadl %addrjm1
  jnz %eq, @eq_case
  %c_gt = csltl %left %up
  jnz %c_gt, @up_is_max
  %new = %left
  jmp @store_new
@up_is_max
  %new = %up
  jmp @store_new
@eq_case
  %new = addl %diag, 1
@store_new
  storel %new, %addrj
  %diag = %up
  %j = addl %j, 1
  jmp @inner_loop
@inner_done
  %i = addl %i, 1
  jmp @outer_loop
@outer_done
  %offn = shll %n, 3
  %addrn = addl %row, %offn
  %res = loadl %addrn
  ret %res
}
