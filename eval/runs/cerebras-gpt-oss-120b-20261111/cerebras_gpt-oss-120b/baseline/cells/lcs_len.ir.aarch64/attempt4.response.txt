export function lcs_len(l %x, l %m, l %y, l %n, l %row) {
    %zero = const 0
    %one = const 1
    %nplus1 = addl %n, %one
    jmp @init(0)

@init(%j):
    %c = csltl %j %nplus1
    jnz %c, @init_body(%j)
    jmp @outer(0)

@init_body(%j):
    %off = shll %j, 3
    %addr = addl %row, %off
    storel %zero, %addr
    %j_next = addl %j, %one
    jmp @init(%j_next)

@outer(%i):
    %c2 = csltl %i %m
    jnz %c2, @outer_body(%i)
    jmp @finish

@outer_body(%i):
    %diag = const 0
    jmp @inner(%i, 1, %diag)

@inner(%i, %j, %diag):
    %c3 = csltl %j %nplus1
    jnz %c3, @inner_body(%i, %j, %diag)
    %i_next = addl %i, %one
    jmp @outer(%i_next)

@inner_body(%i, %j, %diag):
    %xp = addl %x, %i
    %a =w loadub %xp
    %j1 = subl %j, %one
    %yp = addl %y, %j1
    %b =w loadub %yp
    %eq = ceql %a %b
    %offj = shll %j, 3
    %addrj = addl %row, %offj
    %up = loadl %addrj
    %offjm1 = shll %j1, 3
    %addrjm1 = addl %row, %offjm1
    %left = loadl %addrjm1
    jnz %eq, @eq_case(%i, %j, %diag, %up)
    %c_gt = csltl %left %up
    jnz %c_gt, @up_is_max(%i, %j, %diag, %up, %left)
    %new = %left
    jmp @store(%i, %j, %diag, %new, %up)

@eq_case(%i, %j, %diag, %up):
    %new = addl %diag, %one
    jmp @store(%i, %j, %diag, %new, %up)

@up_is_max(%i, %j, %diag, %up, %left):
    %new = %up
    jmp @store(%i, %j, %diag, %new, %up)

@store(%i, %j, %diag, %new, %up):
    %offj = shll %j, 3
    %addrj = addl %row, %offj
    storel %new, %addrj
    %j_next = addl %j, %one
    jmp @inner(%i, %j_next, %up)

@finish:
    %offn = shll %n, 3
    %addrn = addl %row, %offn
    %res = loadl %addrn
    ret %res
}
