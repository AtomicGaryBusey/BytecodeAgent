export function l $isqrt(l n)
@entry
    %zero = const l 0
    %two32 = const l 4294967296
    jmp @loop
@loop
    %lo = phi l %zero, @entry, %new_lo, @set_lo, %new_lo, @set_hi
    %hi = phi l %two32, @entry, %hi_next, @set_lo, %mid, @set_hi
    %cond = cultl %lo, %hi
    jnz %cond, @body
    %res = subl %lo, 1
    ret %res
@body
    %sum = addl %lo, %hi
    %mid = shrl %sum, 1
    %mid_eq_zero = ceql %mid, %zero
    jnz %mid_eq_zero, @set_lo
    %quot = udivl %n, %mid
    %gt = cugtl %mid, %quot
    jnz %gt, @set_hi
    jmp @set_lo
@set_lo
    %new_lo = addl %mid, 1
    %hi_next = %hi
    jmp @loop
@set_hi
    %new_lo = %lo
    jmp @loop
