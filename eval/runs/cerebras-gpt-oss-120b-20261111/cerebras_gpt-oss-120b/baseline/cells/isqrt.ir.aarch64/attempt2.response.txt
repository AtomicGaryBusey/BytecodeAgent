export function l $isqrt(l n)
@entry
    jmp @header
@header
    %lo = phi l 0, @entry, %lo_next, @set_lo, %lo_next, @set_hi
    %hi = phi l 4294967296, @entry, %hi_next, @set_lo, %hi_next, @set_hi
    %cond = cultl %lo, %hi
    jnz %cond, @body
    %res = subl %lo, 1
    ret %res
@body
    %sum = addl %lo, %hi
    %mid = shrl %sum, 1
    %mid2 = mull %mid, %mid
    %gt = cugt %mid2, %n
    jnz %gt, @set_hi
    jmp @set_lo
@set_lo
    %lo_next = addl %mid, 1
    %hi_next = %hi
    jmp @header
@set_hi
    %lo_next = %lo
    %hi_next = %mid
    jmp @header
