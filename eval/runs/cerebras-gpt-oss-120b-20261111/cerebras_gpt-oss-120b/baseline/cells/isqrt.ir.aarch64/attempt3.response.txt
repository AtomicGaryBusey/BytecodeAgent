export function l $isqrt(l n)
@entry
    %lo0 = const l 0
    %hi0 = const l 4294967296
    jmp @header
@header
    %lo = phi l %lo0, @entry, %lo_next, @set_lo, %lo_next, @set_hi
    %hi = phi l %hi0, @entry, %hi_next, @set_lo, %hi_next, @set_hi
    %cond = cultl %lo, %hi
    jnz %cond, @body
    %res = subl %lo, 1
    ret %res
@body
    %sum = addl %lo, %hi
    %mid = shrl %sum, 1
    %quot = udivl %n, %mid
    %gt = cugtl %mid, %quot
    jnz %gt, @set_hi
    jmp @set_lo
@set_lo
    %lo_next = addl %mid, 1
    %hi_next = %hi
    jmp @header
@set_hi
    %lo_next = %lo
    %hi_next = %mid
    jmp @header
