export function l $isqrt(l n)
@loop
    %lo = phi l 0, @entry, %lo_next, @set_lo, %lo_next, @set_hi
    %hi = phi l 4294967296, @entry, %hi_next, @set_lo, %hi_next, @set_hi
    %cond = cultl %lo, %hi
    jnz %cond, @body
    %res = subl %lo, 1
    ret %res
@entry
    jmp @loop
@body
    %sum = addl %lo, %hi
    %mid = shrl %sum, 1
    %mid2 = mull %mid, %mid
    %gt = cugt %mid2, %n
    %lo_inc = addl %mid, 1
    %hi_mid = %mid
    jnz %gt, @set_hi
    jmp @set_lo
@set_lo
    %lo_next = %lo_inc
    %hi_next = %hi
    jmp @loop
@set_hi
    %lo_next = %lo
    %hi_next = %hi_mid
    jmp @loop
