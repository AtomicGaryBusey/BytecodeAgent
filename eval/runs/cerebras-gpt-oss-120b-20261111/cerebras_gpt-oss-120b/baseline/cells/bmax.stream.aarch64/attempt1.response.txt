bmax:
    ldr x2, [x0]            // load a[0]
    subs x1, x1, #1        // n-1
    cbz x1, .Lreturn
    add x3, x0, #8         // ptr = a+1
.Lloop:
    ldr x4, [x3], #8       // load element, ptr+=8
    cmp x4, x2
    csel x2, x4, x2, gt
    subs x1, x1, #1
    bne .Lloop
.Lreturn:
    mov x0, x2
    ret
