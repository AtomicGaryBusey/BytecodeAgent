export function l $qsort_pairs(l %a, l %n) {
@entry
    %cond = cslel %n l 1
    jnz %cond, @ret0, @init

@init
    %mid = srl %n l 1
    %pivot_off = mulul %mid l 16
    %pivot_addr = addl %a %pivot_off
    %pkey = loadl %pivot_addr
    %pval = loadl (addl %pivot_addr l 8)
    %i0 = l 0
    %j0 = subl %n l 1
    br @loop

@loop
    %i = phi l %i0, @init, %i1, @inc_i, %i, @dec_j, %i2, @swap
    %j = phi l %j0, @init, %j, @inc_i, %j1, @dec_j, %j2, @swap
    %cond_ij = csleql %i %j
    jnz %cond_ij, @partition_body, @after

@partition_body
    %addr_i = addl %a, mulul %i l 16
    %key_i = loadl %addr_i
    %val_i = loadl (addl %addr_i l 8)
    %lt_key = csltl %key_i %pkey
    %eq_key = ceql %key_i %pkey
    %lt_val = csltl %val_i %pval
    %cond_i = orw %lt_key (andw %eq_key %lt_val)
    jnz %cond_i, @inc_i, @check_j

@inc_i
    %i1 = addl %i l 1
    br @loop

@check_j
    %addr_j = addl %a, mulul %j l 16
    %key_j = loadl %addr_j
    %val_j = loadl (addl %addr_j l 8)
    %gt_key = csltl %pkey %key_j
    %eq_key2 = ceql %key_j %pkey
    %gt_val = csltl %pval %val_j
    %cond_j = orw %gt_key (andw %eq_key2 %gt_val)
    jnz %cond_j, @dec_j, @swap

@dec_j
    %j1 = subl %j l 1
    br @loop

@swap
    %addr_i2 = addl %a, mulul %i l 16
    %addr_j2 = addl %a, mulul %j l 16
    %key_i2 = loadl %addr_i2
    %val_i2 = loadl (addl %addr_i2 l 8)
    %key_j2 = loadl %addr_j2
    %val_j2 = loadl (addl %addr_j2 l 8)
    storel %addr_i2 %key_j2
    storel (addl %addr_i2 l 8) %val_j2
    storel %addr_j2 %key_i2
    storel (addl %addr_j2 l 8) %val_i2
    %i2 = addl %i l 1
    %j2 = subl %j l 1
    br @loop

@after
    %left_n = addl %j l 1
    %right_a = addl %a, mulul %i l 16
    %right_n = subl %n %i
    %cond_left = csgtl %left_n l 0
    jnz %cond_left, @call_left, @right

@call_left
    %rleft =l call $qsort_pairs(l %a, l %left_n)
    br @right

@right
    %cond_right = csgtl %right_n l 0
    jnz %cond_right, @call_right, @ret

@call_right
    %rright =l call $qsort_pairs(l %right_a, l %right_n)
    br @ret

@ret
    retl l 0

@ret0
    retl l 0
}
