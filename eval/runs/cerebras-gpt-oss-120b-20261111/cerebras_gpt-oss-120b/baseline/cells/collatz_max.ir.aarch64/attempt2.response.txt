export function l $collatz_max(l %n) {
  %n1 = addl %n, 1
  %i = copy 1
  %max = copy 0
  jump @outer(%i, %max, %n1)
}
@outer(%i, %max, %n1) {
  %cond = csltl %i, %n1
  jnz %cond, @inner(%i, %max, %n1)
  ret %max
}
@inner(%i, %max, %n1) {
  %x = %i
  %steps = copy 0
  jump @inner_loop(%i, %max, %n1, %x, %steps)
}
@inner_loop(%i, %max, %n1, %x, %steps) {
  %eq = ceql %x, 1
  jnz %eq, @inner_done(%i, %max, %n1, %steps)
  %low = andl %x, 1
  %even = ceql %low, 0
  jnz %even, @even_case(%i, %max, %n1, %x, %steps)
  %tmp = mull %x, 3
  %x2 = addl %tmp, 1
  %steps2 = addl %steps, 1
  jump @inner_loop(%i, %max, %n1, %x2, %steps2)
}
@even_case(%i, %max, %n1, %x, %steps) {
  %x2 = shrli %x, 1
  %steps2 = addl %steps, 1
  jump @inner_loop(%i, %max, %n1, %x2, %steps2)
}
@inner_done(%i, %max, %n1, %steps) {
  %gt = csltl %max, %steps
  jnz %gt, @update_max(%i, %steps, %n1)
  jump @outer_next(%i, %max, %n1)
}
@update_max(%i, %steps, %n1) {
  jump @outer_next(%i, %steps, %n1)
}
@outer_next(%i, %max, %n1) {
  %i2 = addl %i, 1
  jump @outer(%i2, %max, %n1)
}
