export function l $collatz_max(l %n) {
  %n1 = addl %n, 1
  %i = copy 1
  %max = copy 0
  jmp @outer(%i,%max,%n1)
}
@outer(%i,%max,%n1) {
  %cond = csltl %i, %n1
  jnz %cond, @inner_start(%i,%max,%n1)
  ret %max
}
@inner_start(%i,%max,%n1) {
  %x = %i
  %steps = copy 0
  jmp @inner_loop(%i,%max,%n1,%x,%steps)
}
@inner_loop(%i,%max,%n1,%x,%steps) {
  %eq = ceql %x, 1
  jnz %eq, @inner_done(%i,%max,%n1,%steps)
  %odd = andl %x, 1
  %is_even = ceql %odd, 0
  jnz %is_even, @even(%i,%max,%n1,%x,%steps)
  %tmp = mull %x, 3
  %x2 = addl %tmp, 1
  %steps2 = addl %steps, 1
  jmp @inner_loop(%i,%max,%n1,%x2,%steps2)
}
@even(%i,%max,%n1,%x,%steps) {
  %x2 = shrl %x, 1
  %steps2 = addl %steps, 1
  jmp @inner_loop(%i,%max,%n1,%x2,%steps2)
}
@inner_done(%i,%max,%n1,%steps) {
  %gt = csltl %max, %steps
  jnz %gt, @update_max(%i,%steps,%n1)
  jmp @outer_next(%i,%max,%n1)
}
@update_max(%i,%steps,%n1) {
  jmp @outer_next(%i,%steps,%n1)
}
@outer_next(%i,%max,%n1) {
  %i2 = addl %i, 1
  jmp @outer(%i2,%max,%n1)
}
