sdivrem:
    test rsi, rsi
    jne .Ldiv_nonzero
    mov rax, 0
    mov [rdx], rdi
    ret
.Ldiv_nonzero:
    mov rcx, 0x8000000000000000
    cmp rdi, rcx
    jne .Ldo_idiv
    cmp rsi, -1
    jne .Ldo_idiv
    mov rax, rcx
    xor r8, r8
    mov [rdx], r8
    ret
.Ldo_idiv:
    mov r8, rdx
    mov rax, rdi
    cqo
    idiv rsi
    mov [r8], rdx
    ret
