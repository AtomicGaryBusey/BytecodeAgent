export function $bmax(l %a, l %n) : l {
@entry
    %max0 = loadl %a
    %i0 = constl 1
    br @cond

@cond
    %i = phi l %i0 @entry, %i_next @body
    %max = phi l %max0 @entry, %max_next @body
    %c = csltl %i, %n
    jnz %c, @body, @exit

@body
    %off = mul %i, 8
    %addr = add %a, %off
    %val = loadl %addr
    %gt = csltl %max, %val
    %max_next = sel %gt, %val, %max
    %i_next = add %i, 1
    br @cond

@exit
    ret %max
}
