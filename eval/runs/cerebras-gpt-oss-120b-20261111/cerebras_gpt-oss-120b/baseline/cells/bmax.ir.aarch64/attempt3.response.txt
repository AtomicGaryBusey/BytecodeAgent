export function l $bmax(l %a, l %n) {
@entry
    %max0 = loadl %a
    br @cond

@cond
    %i = phi l 0 @entry, %i1 @body
    %max = phi l %max0 @entry, %max1 @body
    %c = csltl %i, %n
    jnz %c, @body, @exit

@body
    %off = mul %i, 8
    %addr = add %a, %off
    %val = loadl %addr
    %gt = csltl %max, %val
    %max1 = sel %gt, %val, %max
    %i1 = add %i, 1
    br @cond

@exit
    ret %max
}
