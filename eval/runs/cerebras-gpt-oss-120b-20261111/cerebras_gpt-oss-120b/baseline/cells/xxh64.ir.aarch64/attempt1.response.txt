export function l $xxh64(l %p, l %n, l %seed) {
    @start
    %P1 =l const -7046029288634856825
    %P2 =l const -4417276706812521889
    %P3 =l const 1609587929392839161
    %P4 =l const -8796714831421723037
    %P5 =l const 2870177450012600261
    %zero =l const 0
    %c32 =l const 32
    %c8 =l const 8
    %c4 =l const 4
    %c1 =l const 1
    %c31 =l const 31
    %c33 =l const 33
    %c27 =l const 27
    %c37 =l const 37
    %c23 =l const 23
    %c41 =l const 41
    %c11 =l const 11
    %c53 =l const 53

    %cmp32 = w csltl %n %c32
    jnz %cmp32 small_case

    # big case
    %v1 = add %seed %P1
    %v1 = add %v1 %P2
    %v2 = add %seed %P2
    %v3 = add %seed %zero
    %v4 = sub %seed %P1

    %blocks = shr %n 5
    %i =l const 0
    %q = add %p %zero

stripe_loop:
    %condb = w csltl %i %blocks
    jnz %condb stripe_body
    jmp after_stripes

stripe_body:
    %addr0 = add %q %zero
    %addr1 = add %q %c8
    %addr2 = add %q %c8
    %addr2 = add %addr2 %c8
    %addr3 = add %q %c8
    %addr3 = add %addr3 %c8
    %addr3 = add %addr3 %c8
    %addr3 = add %addr3 %c8

    %load0 = l loadl %addr0
    %load1 = l loadl %addr1
    %load2 = l loadl %addr2
    %load3 = l loadl %addr3

    %tmp = mul %load0 %P2
    %tmp2 = add %v1 %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %v1 = mul %rot %P1

    %tmp = mul %load1 %P2
    %tmp2 = add %v2 %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %v2 = mul %rot %P1

    %tmp = mul %load2 %P2
    %tmp2 = add %v3 %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %v3 = mul %rot %P1

    %tmp = mul %load3 %P2
    %tmp2 = add %v4 %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %v4 = mul %rot %P1

    %q = add %q %c32
    %i = add %i %c1
    jmp stripe_loop

after_stripes:
    %h1 = or (shl %v1 %c1) (shr %v1 63)
    %h2 = or (shl %v2 %c7) (shr %v2 57)
    %h3 = or (shl %v3 %c12) (shr %v3 52)
    %h4 = or (shl %v4 %c18) (shr %v4 46)
    %h = add (add (add %h1 %h2) %h3) %h4

    # merge with v1
    %tmp = mul %v1 %P2
    %tmp2 = add %zero %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %round0 = mul %rot %P1
    %h = xor %h %round0
    %h = mul %h %P1
    %h = add %h %P4

    # merge with v2
    %tmp = mul %v2 %P2
    %tmp2 = add %zero %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %round0 = mul %rot %P1
    %h = xor %h %round0
    %h = mul %h %P1
    %h = add %h %P4

    # merge with v3
    %tmp = mul %v3 %P2
    %tmp2 = add %zero %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %round0 = mul %rot %P1
    %h = xor %h %round0
    %h = mul %h %P1
    %h = add %h %P4

    # merge with v4
    %tmp = mul %v4 %P2
    %tmp2 = add %zero %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %round0 = mul %rot %P1
    %h = xor %h %round0
    %h = mul %h %P1
    %h = add %h %P4

    jmp common

small_case:
    %h = add %seed %P5

common:
    %h = add %h %n

    %stripe_len = mul %blocks %c32
    %q = add %p %stripe_len
    %rem = sub %n %stripe_len

while8:
    %cond8 = w csltl %rem %c8
    jnz %cond8 after8
    %load = l loadl %q
    %tmp = mul %load %P2
    %tmp2 = add %zero %tmp
    %rot = or (shl %tmp2 %c31) (shr %tmp2 %c33)
    %round0 = mul %rot %P1
    %h = xor %h %round0
    %h = or (shl %h %c27) (shr %h %c37)
    %h = mul %h %P1
    %h = add %h %P4
    %q = add %q %c8
    %rem = sub %rem %c8
    jmp while8

after8:
while4:
    %cond4 = w csltl %rem %c4
    jnz %cond4 after4
    %load32 = l loaduw %q
    %tmp = mul %load32 %P1
    %h = xor %h %tmp
    %h = or (shl %h %c23) (shr %h %c41)
    %h = mul %h %P2
    %h = add %h %P3
    %q = add %q %c4
    %rem = sub %rem %c4
    jmp while4

after4:
while1:
    %cond1 = w csltl %rem %c1
    jnz %cond1 done
    %c = ub load %q
    %tmp = mul %c %P5
    %h = xor %h %tmp
    %h = or (shl %h %c11) (shr %h %c53)
    %h = mul %h %P1
    %q = add %q %c1
    %rem = sub %rem %c1
    jmp while1

done:
    %h = xor %h (shr %h 33)
    %h = mul %h %P2
    %h = xor %h (shr %h 29)
    %h = mul %h %P3
    %h = xor %h (shr %h 32)
    ret %h
}
