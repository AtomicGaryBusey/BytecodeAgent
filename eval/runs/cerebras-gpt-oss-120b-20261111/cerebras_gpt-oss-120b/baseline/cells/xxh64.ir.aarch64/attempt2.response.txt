export function l $xxh64(l %p, l %n, l %seed) {
    @start
    %P1 =l -7046029288634856825
    %P2 =l -4417276706812521889
    %P3 =l 1609587929392839161
    %P4 =l -8796714831421723037
    %P5 =l 2870177450012600261
    %zero =l 0
    %c1 =l 1
    %c4 =l 4
    %c8 =l 8
    %c11 =l 11
    %c23 =l 23
    %c27 =l 27
    %c31 =l 31
    %c33 =l 33
    %c37 =l 37
    %c41 =l 41
    %c53 =l 53
    %c32 =l 32

    %cmp32 = w csltl %n %c32
    jnz %cmp32 small_case

    # big case
    %v1 =l add %seed %P1
    %v1 =l add %v1 %P2
    %v2 =l add %seed %P2
    %v3 =l add %seed %zero
    %v4 =l sub %seed %P1

    %blocks =l shr %n 5        # n >> 5
    %i =l 0
    %q =l add %p %zero

stripe_loop:
    %condb = w csltl %i %blocks
    jnz %condb stripe_body
    jmp after_stripes

stripe_body:
    %addr0 =l %q
    %addr1 =l add %q %c8
    %addr2 =l add %addr1 %c8   # q+16
    %addr3 =l add %addr2 %c8   # q+24

    %load0 =l loadl %addr0
    %load1 =l loadl %addr1
    %load2 =l loadl %addr2
    %load3 =l loadl %addr3

    %tmp =l mul %load0 %P2
    %tmp =l add %v1 %tmp
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %v1 =l mul %rot %P1

    %tmp =l mul %load1 %P2
    %tmp =l add %v2 %tmp
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %v2 =l mul %rot %P1

    %tmp =l mul %load2 %P2
    %tmp =l add %v3 %tmp
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %v3 =l mul %rot %P1

    %tmp =l mul %load3 %P2
    %tmp =l add %v4 %tmp
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %v4 =l mul %rot %P1

    %q =l add %q %c32
    %i =l add %i %c1
    jmp stripe_loop

after_stripes:
    %h1 =l or (shl %v1 %c1) (shr %v1 63)
    %h2 =l or (shl %v2 %c1) (shr %v2 57)   # rotate left 7: 64-7=57
    %h3 =l or (shl %v3 %c1) (shr %v3 52)   # rotate left 12: 64-12=52
    %h4 =l or (shl %v4 %c1) (shr %v4 46)   # rotate left 18: 64-18=46
    %h =l add (add (add %h1 %h2) %h3) %h4

    # merge with v1
    %tmp =l mul %v1 %P2
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %round0 =l mul %rot %P1
    %h =l xor %h %round0
    %h =l mul %h %P1
    %h =l add %h %P4

    # merge with v2
    %tmp =l mul %v2 %P2
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %round0 =l mul %rot %P1
    %h =l xor %h %round0
    %h =l mul %h %P1
    %h =l add %h %P4

    # merge with v3
    %tmp =l mul %v3 %P2
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %round0 =l mul %rot %P1
    %h =l xor %h %round0
    %h =l mul %h %P1
    %h =l add %h %P4

    # merge with v4
    %tmp =l mul %v4 %P2
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %round0 =l mul %rot %P1
    %h =l xor %h %round0
    %h =l mul %h %P1
    %h =l add %h %P4

    jmp common

small_case:
    %h =l add %seed %P5

common:
    %h =l add %h %n

    %stripe_len =l mul %blocks %c32
    %q =l add %p %stripe_len
    %rem =l sub %n %stripe_len

while8:
    %cond8 = w csltl %rem %c8
    jnz %cond8 after8
    %load =l loadl %q
    %tmp =l mul %load %P2
    %rot =l or (shl %tmp %c31) (shr %tmp %c33)
    %round0 =l mul %rot %P1
    %h =l xor %h %round0
    %h =l or (shl %h %c27) (shr %h %c37)
    %h =l mul %h %P1
    %h =l add %h %P4
    %q =l add %q %c8
    %rem =l sub %rem %c8
    jmp while8

after8:
while4:
    %cond4 = w csltl %rem %c4
    jnz %cond4 after4
    %load32 =l loaduw %q
    %tmp =l mul %load32 %P1
    %h =l xor %h %tmp
    %h =l or (shl %h %c23) (shr %h %c41)
    %h =l mul %h %P2
    %h =l add %h %P3
    %q =l add %q %c4
    %rem =l sub %rem %c4
    jmp while4

after4:
while1:
    %cond1 = w csltl %rem %c1
    jnz %cond1 done
    %c =l loadb %q
    %tmp =l mul %c %P5
    %h =l xor %h %tmp
    %h =l or (shl %h %c11) (shr %h %c53)
    %h =l mul %h %P1
    %q =l add %q %c1
    %rem =l sub %rem %c1
    jmp while1

done:
    %h =l xor %h (shr %h 33)
    %h =l mul %h %P2
    %h =l xor %h (shr %h 29)
    %h =l mul %h %P3
    %h =l xor %h (shr %h 32)
    ret %h
}
