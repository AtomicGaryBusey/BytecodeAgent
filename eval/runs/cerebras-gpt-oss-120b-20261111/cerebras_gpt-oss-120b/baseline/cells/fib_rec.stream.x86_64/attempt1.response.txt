fib_rec:
    cmp rdi, 1
    jbe .Lbase
    sub rsp, 24
    mov qword ptr [rsp+8], rdi
    lea rdi, [rdi - 1]
    call fib_rec
    mov qword ptr [rsp+16], rax
    mov rdi, qword ptr [rsp+8]
    lea rdi, [rdi - 2]
    call fib_rec
    add rax, qword ptr [rsp+16]
    add rsp, 24
    ret
