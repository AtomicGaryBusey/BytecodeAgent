crc32:
    // load CRC polynomial constant into w7
    movz    w7, #0xEDB8, lsl #16    // w7 = 0xEDB80000
    movk    w7, #0x8320, lsl #0     // w7 = 0xEDB88320
    // crc = 0xFFFFFFFF
    mov     w4, #0xffffffff
    // x2 = p (pointer), x3 = n (remaining length)
    mov     x2, x0
    mov     x3, x1
    // if n == 0 jump to final xor
    cbz     x3, .Lfinal_xor
.Lbyte_loop:
    // load next byte, post-increment pointer
    ldrb    w5, [x2], #1
    // crc ^= byte
    eor     w4, w4, w5
    // inner loop: 8 bits
    mov     w6, #8
.Lbit_loop:
    // if (crc & 1) == 0 then branch to shift only
    tbz     w4, #0, .Lshift_only
    // LSB = 1: shift then xor polynomial
    lsr     w4, w4, #1
    eor     w4, w4, w7
    b       .Lbit_continue
.Lshift_only:
    // LSB = 0: just shift
    lsr     w4, w4, #1
.Lbit_continue:
    subs    w6, w6, #1
    b.ne    .Lbit_loop
    // decrement byte counter
    subs    x3, x3, #1
    b.ne    .Lbyte_loop
.Lfinal_xor:
    eor     w4, w4, #0xffffffff
