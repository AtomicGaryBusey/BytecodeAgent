export function l $gcd(l, l)
@entry
    %a0 = arg0
    %b0 = arg1
    jmp @loop(%a0, %b0)

@loop(l %a, l %b)
    %zero = constl 0
    %cond = ceql %b, %zero
    jnz %cond, @exit(%a), @calc(%a, %b)

@calc(l %a, l %b)
    %mod = umodl %a, %b
    %a1 = movl %b
    %b1 = movl %mod
    jmp @loop(%a1, %b1)

@exit(l %a)
    ret %a
