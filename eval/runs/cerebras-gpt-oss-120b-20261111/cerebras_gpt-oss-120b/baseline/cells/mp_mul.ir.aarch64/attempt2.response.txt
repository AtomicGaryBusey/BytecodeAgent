export function l mp_mul(l %a_ptr, l %na, l %b_ptr, l %nb, l %r_ptr) {
    %c0 =l const 0
    %c1 =l const 1
    %c32 =l const 32
    %mask32 =l const 0xffffffff

    %total =l add %na %nb
    %total_eq0 =w ceql %total %c0
    jnz %total_eq0, @ret0

    // zero r[0..total-1]
    %z_i0 =l const 0
    jmp @zero_cond %z_i0

@zero_cond %z_i
    %z_cond =w csltl %z_i %total
    jnz %z_cond, @zero_body
    jmp @zero_done

@zero_body %z_i
    %z_addr =l add %r_ptr %z_i
    storel %c0, %z_addr
    %z_i1 =l add %z_i %c1
    jmp @zero_cond %z_i1

@zero_done
    // if na==0 or nb==0 then product is zero, return 0
    %na_eq0 =w ceql %na %c0
    %nb_eq0 =w ceql %nb %c0
    %any_zero =w or %na_eq0 %nb_eq0
    jnz %any_zero, @ret0

    // outer loop i = 0..na-1
    %i0 =l const 0
    jmp @outer_cond %i0

@outer_cond %i
    %cond_i =w csltl %i %na
    jnz %cond_i, @outer_body
    jmp @outer_done

@outer_body %i
    %ai_addr =l add %a_ptr %i
    %ai =loadl %ai_addr
    %j0 =l const 0
    %carry0 =l const 0
    jmp @inner_cond %i %ai %j0 %carry0

@inner_cond %i %ai %j %carry
    %cond_j =w csltl %j %nb
    jnz %cond_j, @inner_body
    jmp @inner_done %i %carry

@inner_body %i %ai %j %carry
    %bj_addr =l add %b_ptr %j
    %bj =loadl %bj_addr

    %ai_lo =l and %ai %mask32
    %ai_hi =l shr %ai %c32
    %bj_lo =l and %bj %mask32
    %bj_hi =l shr %bj %c32

    %p0 =l mul %ai_lo %bj_lo
    %p1 =l mul %ai_lo %bj_hi
    %p2 =l mul %ai_hi %bj_lo
    %p3 =l mul %ai_hi %bj_hi

    %p1_sh =l shl %p1 %c32
    %p2_sh =l shl %p2 %c32
    %sum0 =l add %p0 %p1_sh
    %c_sum0 =w cultl %sum0 %p0
    %low =l add %sum0 %p2_sh
    %c_low =w cultl %low %sum0
    %carry_low =l add %c_sum0 %c_low

    %p1_hi =l shr %p1 %c32
    %p2_hi =l shr %p2 %c32
    %tmp1 =l add %p3 %p1_hi
    %c_tmp1 =w cultl %tmp1 %p3
    %tmp2 =l add %tmp1 %p2_hi
    %c_tmp2 =w cultl %tmp2 %tmp1
    %high =l add %tmp2 %carry_low
    %c_tmp3 =w cultl %high %tmp2
    %carry_high =l add %c_tmp1 %c_tmp2
    %carry_high =l add %carry_high %c_tmp3

    %r_idx =l add %i %j
    %r_addr =l add %r_ptr %r_idx
    %r_val =loadl %r_addr
    %sum1 =l add %r_val %low
    %c_sum1 =w cultl %sum1 %r_val
    %sum2 =l add %sum1 %carry
    %c_sum2 =w cultl %sum2 %sum1
    storel %sum2, %r_addr

    %new_carry1 =l add %high %c_sum1
    %new_carry2 =l add %new_carry1 %c_sum2
    %new_carry =l add %new_carry2 %carry_high

    %j1 =l add %j %c1
    jmp @inner_cond %i %ai %j1 %new_carry

@inner_done %i %carry
    %dest_idx =l add %i %nb
    %dest_addr =l add %r_ptr %dest_idx
    storel %carry, %dest_addr
    %i1 =l add %i %c1
    jmp @outer_cond %i1

@outer_done
    %k0 =l sub %total %c1
    jmp @find_cond %k0

@find_cond %k
    %addrk =l add %r_ptr %k
    %valk =loadl %addrk
    %iszero =w ceql %valk %c0
    %k_eq0 =w ceql %k %c0
    %both =w and %iszero %k_eq0
    jnz %both, @ret0
    jnz %iszero, @dec_k %k
    %ret =l add %k %c1
    ret %ret

@dec_k %k
    %k1 =l sub %k %c1
    jmp @find_cond %k1

@ret0
    ret %c0
}
