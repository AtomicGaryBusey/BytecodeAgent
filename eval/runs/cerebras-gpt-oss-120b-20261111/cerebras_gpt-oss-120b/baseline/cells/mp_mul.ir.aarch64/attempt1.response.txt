export function l $mp_mul(l %0, l %1, l %2, l %3, l %4) {
    %a_ptr =l %0
    %na =l %1
    %b_ptr =l %2
    %nb =l %3
    %r_ptr =l %4

    %c0 =l const 0
    %c1 =l const 1
    %c32 =l const 32
    %c32_mask =l const 0xffffffff

    %total =l add %na, %nb
    %total_eq0 =w ceql %total, %c0
    jnz %total_eq0, @ret0

    // zero r[0..total-1]
    %i =l const 0
@z_loop
    %cond_i =w csltl %i, %total
    jnz %cond_i, @z_body
    jmp @z_done
@z_body
    %addr =l add %r_ptr, %i
    storel %c0, %addr
    %i =l add %i, %c1
    jmp @z_loop
@z_done

    // if na==0 or nb==0 then product is zero, return 0
    %na_eq0 =w ceql %na, %c0
    %nb_eq0 =w ceql %nb, %c0
    %any_zero =w or %na_eq0, %nb_eq0
    jnz %any_zero, @ret0

    // outer loop i = 0..na-1
    %i =l const 0
@outer
    %cond_i =w csltl %i, %na
    jnz %cond_i, @outer_body
    jmp @outer_done
@outer_body
    %ai_addr =l add %a_ptr, %i
    %ai =loadl %ai_addr

    %j =l const 0
    %carry =l const 0
@inner
    %cond_j =w csltl %j, %nb
    jnz %cond_j, @inner_body
    jmp @inner_done
@inner_body
    %bj_addr =l add %b_ptr, %j
    %bj =loadl %bj_addr

    // split ai and bj
    %ai_lo =l and %ai, %c32_mask
    %ai_hi =l shr %ai, %c32
    %bj_lo =l and %bj, %c32_mask
    %bj_hi =l shr %bj, %c32

    // partial products
    %p0 =l mul %ai_lo, %bj_lo
    %p1 =l mul %ai_lo, %bj_hi
    %p2 =l mul %ai_hi, %bj_lo
    %p3 =l mul %ai_hi, %bj_hi

    // combine for low 64 bits
    %p1_sh =l shl %p1, %c32
    %p2_sh =l shl %p2, %c32
    %sum0 =l add %p0, %p1_sh
    %c_sum0 =w cultl %sum0, %p0
    %low =l add %sum0, %p2_sh
    %c_low =w cultl %low, %sum0
    %carry_low =l add %c_sum0, %c_low

    // high part
    %p1_hi =l shr %p1, %c32
    %p2_hi =l shr %p2, %c32
    %tmp1 =l add %p3, %p1_hi
    %c_tmp1 =w cultl %tmp1, %p3
    %tmp2 =l add %tmp1, %p2_hi
    %c_tmp2 =w cultl %tmp2, %tmp1
    %high =l add %tmp2, %carry_low
    %c_tmp3 =w cultl %high, %tmp2
    %carry_high =l add %c_tmp1, %c_tmp2
    %carry_high =l add %carry_high, %c_tmp3

    // add low to r[i+j] with carry
    %r_idx =l add %i, %j
    %r_addr =l add %r_ptr, %r_idx
    %r_val =loadl %r_addr
    %sum1 =l add %r_val, %low
    %c_sum1 =w cultl %sum1, %r_val
    %sum2 =l add %sum1, %carry
    %c_sum2 =w cultl %sum2, %sum1
    storel %sum2, %r_addr

    // new carry = high + c_sum1 + c_sum2 + carry_high
    %new_carry =l add %high, %c_sum1
    %new_carry =l add %new_carry, %c_sum2
    %carry =l add %new_carry, %carry_high

    %j =l add %j, %c1
    jmp @inner
@inner_done
    // store final carry to r[i+nb]
    %dest_idx =l add %i, %nb
    %dest_addr =l add %r_ptr, %dest_idx
    storel %carry, %dest_addr

    %i =l add %i, %c1
    jmp @outer
@outer_done

    // find most significant non‑zero limb
    %k =l sub %total, %c1
@find
    %addrk =l add %r_ptr, %k
    %valk =loadl %addrk
    %iszero =w ceql %valk, %c0
    %k_eq0 =w ceql %k, %c0
    %both =w and %iszero, %k_eq0
    jnz %both, @ret0
    jnz %iszero, @dec_k
    // non‑zero found
    %ret =l add %k, %c1
    ret %ret
@dec_k
    %k =l sub %k, %c1
    jmp @find
@ret0
    ret %c0
}
