export function l $fib(l %n) {
@entry
    %lt2 = csltl %n, 2
    jnz %lt2, @ret_small

    %a0 = constl 0
    %b0 = constl 1
    %i0 = constl 2
    %np1 = addl %n, constl 1
    jmp @loop(%a0, %b0, %i0)

@loop(%a, %b, %i)
    %cond = csltl %i, %np1
    jnz %cond, @body(%a, %b, %i)
    jmp @end(%b)

@body(%a, %b, %i)
    %tmp = addl %a, %b
    %a1 = %b
    %b1 = %tmp
    %i1 = addl %i, constl 1
    jmp @loop(%a1, %b1, %i1)

@end(%res)
    ret %res

@ret_small
    ret %n
}
