export function l $collatz_max(l n) {
  %n1 = addl %n (imm l 1)
  %i = imm l 1
  %max = imm l 0
  @outer
  %cond = csltl %i %n1
  jnz w %cond @body @exit
  @body
  %i_val = %i
  jmp @loop_start
  @loop_start
  jmp @loop_head
  @loop_head
  %x = phi l %i_val %loop_start %x_next_odd %odd_branch %x_next_even %even_branch
  %steps = phi l (imm l 0) %loop_start %steps_next_odd %odd_branch %steps_next_even %even_branch
  %eq = ceql %x (imm l 1)
  jnz w %eq @loop_end @odd_even_test
  @odd_even_test
  %odd = andl %x (imm l 1)
  jnz w %odd @odd_branch @even_branch
  @odd_branch
  %tmp = mul l %x (imm l 3)
  %x_next_odd = addl %tmp (imm l 1)
  %steps_next_odd = addl %steps (imm l 1)
  jmp @loop_head
  @even_branch
  %x_next_even = srll %x (imm l 1)
  %steps_next_even = addl %steps (imm l 1)
  jmp @loop_head
  @loop_end
  %gt = csltl %max %steps
  jnz w %gt @update_max @skip_max
  @update_max
  %max = %steps
  @skip_max
  %i = addl %i (imm l 1)
  jmp @outer
  @exit
  ret %max
}
