export function l $collatz_max(l n) {
@entry
%n1 = addl %n, imm l 1
jmp @outer_head

@outer_head
%i = phi l imm l 1 @entry %i_next @outer_continue
%max = phi l imm l 0 @entry %max_next @outer_continue
%cond = csltl %i, %n1
jnz %cond @outer_body @exit

@outer_body
%x = phi l %i @outer_head %x_next @odd_branch %x_next @even_branch
%steps = phi l imm l 0 @outer_head %steps_next @odd_branch %steps_next @even_branch
%eq = ceql %x, imm l 1
jnz %eq @inner_end @odd_even_test

@odd_even_test
%odd = andl %x, imm l 1
jnz %odd @odd_branch @even_branch

@odd_branch
%tmp = mull %x, imm l 3
%x_next = addl %tmp, imm l 1
%steps_next = addl %steps, imm l 1
jmp @outer_body

@even_branch
%x_next = srll %x, imm l 1
%steps_next = addl %steps, imm l 1
jmp @outer_body

@inner_end
%gt = csltl %max, %steps
jnz %gt @update_max @skip_max

@update_max
%max_next = %steps
jmp @outer_continue

@skip_max
%max_next = %max
jmp @outer_continue

@outer_continue
%i_next = addl %i, imm l 1
jmp @outer_head

@exit
ret %max
}
