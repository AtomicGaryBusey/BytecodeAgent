export function l $collatz_max(l n) {
    %n1 = addl %n (imm l 1)
    %p_i = alloc l 8
    %p_max = alloc l 8
    storel (imm l 1) %p_i
    storel (imm l 0) %p_max
    @outer
    %i = loadl %p_i
    %cond = csltl %i %n1
    jnz w %cond @body @exit
    @body
    %i_val = loadl %p_i
    %x_init = %i_val
    %steps_init = (imm l 0)
    jmp @inner_start
    @inner_start
    %x = phi l %x_init %x_next_odd %x_next_even
    %steps = phi l %steps_init %steps_next_odd %steps_next_even
    %eq = ceql %x (imm l 1)
    jnz w %eq @inner_end @inner_body
    @inner_body
    %odd = andl %x (imm l 1)
    jnz w %odd @odd_branch @even_branch
    @odd_branch
    %tmp = mul l %x (imm l 3)
    %x_next_odd = addl %tmp (imm l 1)
    %steps_next_odd = addl %steps (imm l 1)
    jmp @inner_start
    @even_branch
    %x_next_even = srll %x (imm l 1)
    %steps_next_even = addl %steps (imm l 1)
    jmp @inner_start
    @inner_end
    %max = loadl %p_max
    %gt = csltl %max %steps
    jnz w %gt @update_max @skip_max
    @update_max
    storel %steps %p_max
    jmp @skip_max
    @skip_max
    %i_next = addl %i (imm l 1)
    storel %i_next %p_i
    jmp @outer
    @exit
    %result = loadl %p_max
    ret %result
}
