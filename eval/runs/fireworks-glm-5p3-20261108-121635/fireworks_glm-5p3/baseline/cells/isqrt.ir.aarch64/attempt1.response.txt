export function l $isqrt(l %n) {
@start
	%lo0 = l copy 0
	%hi0 = l copy 4294967296
	%one = l copy 1
	%two = l copy 2
	jmp @loop
@loop
	%lo = l phi @start %lo0, @setlo %mid, @sethi %lo
	%hi = l phi @start %hi0, @setlo %hi, @sethi %mid
	%lp1 = l add %lo, %one
	%cont = w cultl %lp1, %hi
	jnz %cont, @body, @end
@body
	%sum = l add %lo, %hi
	%mid = l div %sum, %two
	%sq = l mul %mid, %mid
	%le = w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	jmp @loop
@sethi
	jmp @loop
@end
	ret %lo
}
