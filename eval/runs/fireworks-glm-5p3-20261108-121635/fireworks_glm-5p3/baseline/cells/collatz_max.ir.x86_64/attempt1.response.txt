export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%best =l copy 0
	jmp @outer
@outer
	%brk =w csltl %n, %k
	jnz %brk, @end, @body
@body
	%x =l copy %k
	%c =l copy 0
	jmp @inner
@inner
	%at1 =w ceql %x, 1
	jnz %at1, @innerend, @step
@step
	%lsb =l and %x, 1
	%odd =w ceql %lsb, 1
	jnz %odd, @odd, @even
@even
	%x =l div %x, 2
	jmp @cnt
@odd
	%t2 =l add %x, %x
	%t3 =l add %t2, %x
	%x =l add %t3, 1
	jmp @cnt
@cnt
	%c =l add %c, 1
	jmp @inner
@innerend
	%upd =w csltl %best, %c
	jnz %upd, @newbest, @inc
@newbest
	%best =l copy %c
	jmp @inc
@inc
	%k =l add %k, 1
	jmp @outer
@end
	ret %best
}
