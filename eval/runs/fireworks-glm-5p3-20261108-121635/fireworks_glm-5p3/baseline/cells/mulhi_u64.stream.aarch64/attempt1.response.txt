mulhi_u64:
    mov x2, x0              // t = a (a << 0), low word of shifted a
    mov x3, xzr             // u = 0, high word of shifted a
    mov x4, x1              // remaining low bits of b
    mov x5, xzr             // hi = 0
    mov x6, xzr             // lo = 0
.Lloop:
    tbz x4, #0, .Lnobit     // current bit of b clear?
    adds x6, x6, x2         // lo += t
    adc x5, x5, x3          // hi += u + carry
.Lnobit:
    lsr x4, x4, #1          // move to next bit of b
    cbz x4, .Ldone          // no set bits left
    lsr x7, x2, #63         // bit shifted out of t
    lsl x2, x2, #1          // t <<= 1
    lsl x3, x3, #1          // u <<= 1
    orr x3, x3, x7          // u |= carry-out of t
    b .Lloop
.Ldone:
    mov x0, x5              // result = high 64 bits
    ret
