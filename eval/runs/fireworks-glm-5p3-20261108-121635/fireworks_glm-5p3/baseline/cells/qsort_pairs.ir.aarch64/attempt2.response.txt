export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret, @init
@init
	%hi =l sub %n, 1
	%poff =l mul %hi, 16
	%pkaddr =l add %a, %poff
	%pvaddr =l add %pkaddr, 8
	%pk =l loadl %pkaddr
	%pv =l loadl %pvaddr
	%i =l copy 0
	%j =l copy 0
	jmp @loop
@loop
	%more =w csltl %j, %hi
	jnz %more, @body, @after
@body
	%joff =l mul %j, 16
	%jkaddr =l add %a, %joff
	%jvaddr =l add %jkaddr, 8
	%jk =l loadl %jkaddr
	%jv =l loadl %jvaddr
	%clt =w csltl %jk, %pk
	%ceq =w ceql %jk, %pk
	%cle =w cslel %jv, %pv
	%ceqle =w and %ceq, %cle
	%le =w or %clt, %ceqle
	jnz %le, @swap, @adv
@swap
	%ioff =l mul %i, 16
	%ikaddr =l add %a, %ioff
	%ivaddr =l add %ikaddr, 8
	%ik =l loadl %ikaddr
	%iv =l loadl %ivaddr
	storel %ik, %jkaddr
	storel %iv, %jvaddr
	storel %jk, %ikaddr
	storel %jv, %ivaddr
	%i =l add %i, 1
	jmp @adv
@adv
	%j =l add %j, 1
	jmp @loop
@after
	%ioff2 =l mul %i, 16
	%ipkaddr =l add %a, %ioff2
	%ipvaddr =l add %ipkaddr, 8
	%ik2 =l loadl %ipkaddr
	%iv2 =l loadl %ipvaddr
	storel %ik2, %pkaddr
	storel %iv2, %pvaddr
	storel %pk, %ipkaddr
	storel %pv, %ipvaddr
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%rptr =l add %a, %roff
	%rcnt =l sub %hi, %i
	%r2 =l call $qsort_pairs(l %rptr, l %rcnt)
	ret 0
@ret
	ret 0
}
