export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @up %lo1, @down %lo2
	%hi =l phi @start %n, @up %hi1, @down %hi2
	%c0 =w csltl %lo, %hi
	jnz %c0, @body, @notfound
@body
	%s =l add %lo, %hi
	%mid =l udiv %s, 2
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%e =w ceql %v, %key
	jnz %e, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @up, @down
@up
	%lo1 =l add %mid, 1
	%hi1 =l copy %hi
	jmp @loop
@down
	%lo2 =l copy %lo
	%hi2 =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	%r =l copy -1
	ret %r
}
