export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%n8 =l mul %n, 8
	%end =l add %a, %n8
	%p =l copy %a
	%i =l copy 0
	jmp @loop
@loop
	%more =w csltl %p, %end
	jnz %more, @body, @end
@body
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @next, @end
@next
	%p =l add %p, 8
	%i =l add %i, 1
	jmp @loop
@end
	ret %i
}
