export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%total =l add %n, 1
	%zero =l copy 0
	jmp @zchk
@zchk
	%z =l phi @start 0, @zbody %z1
	%zc =w csltl %z, %total
	jnz %zc, @zbody, @ichk
@zbody
	%zo =l mul %z, 8
	%zp =l add %row, %zo
	storel %zero, %zp
	%z1 =l add %z, 1
	jmp @zchk
@ichk
	%i =l phi @zchk 0, @idone %i1
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @fin
@ibody
	%xp =l add %x, %i
	%xi =w loadub %xp
	jmp @jchk
@jchk
	%j =l phi @ibody 1, @upd %j1
	%diag =l phi @ibody 0, @upd %up
	%jc =w csltl %j, %total
	jnz %jc, @jbody, @idone
@jbody
	%jo =l mul %j, 8
	%rj =l add %row, %jo
	%up =l loadl %rj
	%rprev =l sub %rj, 8
	%jp =l sub %j, 1
	%yp =l add %y, %jp
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@nomatch
	%left =l loadl %rprev
	%lt =w csltl %left, %up
	jnz %lt, @takeup, @takeleft
@takeup
	jmp @upd
@takeleft
	jmp @upd
@match
	%d1 =l add %diag, 1
	jmp @upd
@upd
	%val =l phi @takeup %up, @takeleft %left, @match %d1
	storel %val, %rj
	%j1 =l add %j, 1
	jmp @jchk
@idone
	%i1 =l add %i, 1
	jmp @ichk
@fin
	%noff =l mul %n, 8
	%np =l add %row, %noff
	%res =l loadl %np
	ret %res
}
