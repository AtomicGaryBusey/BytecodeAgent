reverse:
    xor r8d, r8d                      # i = 0
    lea r9, [rsi - 1]                 # j = n - 1
.Lloop:
    cmp r8, r9
    jge .Ldone
    mov rdx, qword ptr [rdi + r8*8]   # rdx = a[i]
    mov rcx, qword ptr [rdi + r9*8]   # rcx = a[j]
    mov qword ptr [rdi + r8*8], rcx
    mov qword ptr [rdi + r9*8], rdx
    inc r8
    dec r9
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
