export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%N =l mul %n, %n
	%N2 =l add %N, %N
	%one =l copy 1
	%zero =l copy 0
	%ii =l copy 0
@id_i
	%t1 =w csltl %ii, %n
	jnz %t1, @id_i_body, @id_i_end
@id_i_body
	%ridx =l mul %ii, %n
	%roff =l mul %ridx, 8

	%jj =l copy 0
@id_j
	%t2 =w csltl %jj, %n
	jnz %t2, @id_j_body, @id_j_end
@id_j_body

	jnz %t3, @id_one, @id_zero
@id_one
	storel %one, %op
	jmp @id_next
@id_zero
	storel %zero, %op
	jmp @id_next
@id_next
	%jj1 =l add %jj, 1
	%jj =l copy %jj1
	jmp @id_j
@id_j_end
	%ii1 =l add %ii, 1
	%ii =l copy %ii1
	jmp @id_i
@id_i_end
	%kk =l copy 0
@mul_k
	%t4 =w csltl %kk, %k
	jnz %t4, @mul_k_body, @mul_k_end
@mul_k_body

@mi

@mi_body

	%prow8 =l mul %prow, 8
	%orow2 =l add %out, %prow8

@mj

	jnz %t6, @mj_body, @mj_end
@mj_body

	%tt =l copy 0
@mt

@mt_body

	%ta8 =l mul %tai, 8
	%ael =l add %a, %ta8
	%av =l loadl %ael

	jmp @mt
@mt_end
	%di =l add %prow, %mj
	%di8 =l mul %di, 8
	%del =l add %tmp, %di8
	storel %acc, %del
	%mj1 =l add %mj, 1
	%mj =l copy %mj1
	jmp @mj
@mj_end
	%mi1 =l add %mi, 1
	%mi =l copy %mi1
	jmp @mi
@mi_end
	%cc =l copy 0
@mcpy
	%t8 =w csltl %cc, %N
	jnz %t8, @mcpy_body, @mcpy_end
@mcpy_body

	%cs =l add %tmp, %cc8
	%cv =l loadl %cs
	%cd =l add %out, %cc8
	storel %cv, %cd
	%cc1 =l add %cc, 1
	%cc =l copy %cc1
	jmp @mcpy
@mcpy_end
	%kk1 =l add %kk, 1
	%kk =l copy %kk1
	jmp @mul_k
@mul_k_end

@ztmp
	%t9 =w csltl %zz, %N2
	jnz %t9, @ztmp_body, @ztmp_end
@ztmp_body
	%zz8 =l mul %zz, 8
	%ze =l add %tmp, %zz8
	storel %zero, %ze
	%zz1 =l add %zz, 1
	%zz =l copy %zz1
	jmp @ztmp
@ztmp_end
	ret %zero
}
