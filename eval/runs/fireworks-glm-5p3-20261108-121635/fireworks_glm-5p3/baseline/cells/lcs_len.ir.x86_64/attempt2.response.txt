export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	jmp @zcond
@zcond
	%jz =l phi @entry 0, @zbody %jz1
	%cz =w csgtl %jz, %n
	jnz %cz, @iloop, @zbody
@zbody
	%za =l mul %jz, 8
	%zp =l add %row, %za
	storel 0, %zp
	%jz1 =l add %jz, 1
	jmp @zcond
@iloop
	%i =l phi @zcond 0, @inext %i1
	%ci =w csltl %i, %m
	jnz %ci, @ibody, @done
@ibody
	%xa =l add %x, %i
	%xi =w loadub %xa
	jmp @jcond
@jcond
	%j =l phi @ibody 1, @jnext %j1
	%diag =l phi @ibody 0, @jnext %up
	%left =l phi @ibody 0, @jnext %newv
	%cj =w cslel %j, %n
	jnz %cj, @jbody, @inext
@jbody
	%ya =l add %y, %j
	%ym =l sub %ya, 1
	%yj =w loadub %ym
	%ja =l mul %j, 8
	%rp =l add %row, %ja
	%up =l loadl %rp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@nomatch
	%lt =w csltl %up, %left
	jnz %lt, @takel, @takeu
@takel
	jmp @jnext
@takeu
	jmp @jnext
@match
	%mv =l add %diag, 1
	jmp @jnext
@jnext
	%newv =l phi @takel %left, @takeu %up, @match %mv
	storel %newv, %rp
	%j1 =l add %j, 1
	jmp @jcond
@inext
	%i1 =l add %i, 1
	jmp @iloop
@done
	%ra =l mul %n, 8
	%rp2 =l add %row, %ra
	%res =l loadl %rp2
	ret %res
}
