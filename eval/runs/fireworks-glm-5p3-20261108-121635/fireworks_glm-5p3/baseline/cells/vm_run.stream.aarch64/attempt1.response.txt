vm_run:
    mov x9, x0                    // code
    mov x10, x1                   // len
    mov x11, x2                   // stack
    mov x12, x3                   // cap
    mov x13, xzr                  // d = depth
    mov x14, xzr                  // pc
.Lloop:
    cmp x14, x10
    b.ge .Ldone
    ldr x15, [x9, x14, lsl #3]    // word = code[pc]
    cmp x15, #11
    b.hi .Lunknown
    adr x16, .Ltbl
    add x16, x16, x15, lsl #2
    br x16
.Lunknown:
    add x14, x14, #1
    b .Lloop
.Lhalt:
    b .Ldone
.Lpush:
    add x16, x14, #1
    ldr x17, [x9, x16, lsl #3]
    str x17, [x11, x13, lsl #3]
    add x13, x13, #1
    add x14, x14, #2
    b .Lloop
.Ladd:
    sub x13, x13, #1
    ldr x17, [x11, x13, lsl #3]   // b (old top)
    sub x16, x13, #1
    ldr x0, [x11, x16, lsl #3]    // a
    add x0, x0, x17
    str x0, [x11, x16, lsl #3]
    add x14, x14, #1
    b .Lloop
.Lsub:
    sub x13, x13, #1
    ldr x17, [x11, x13, lsl #3]
    sub x16, x13, #1
    ldr x0, [x11, x16, lsl #3]
    sub x0, x0, x17
    str x0, [x11, x16, lsl #3]
    add x14, x14, #1
    b .Lloop
.Lmul:
    sub x13, x13, #1
    ldr x17, [x11, x13, lsl #3]
    sub x16, x13, #1
    ldr x0, [x11, x16, lsl #3]
    mul x0, x0, x17
    str x0, [x11, x16, lsl #3]
    add x14, x14, #1
    b .Lloop
.Ldup:
    sub x16, x13, #1
    ldr x17, [x11, x16, lsl #3]
    str x17, [x11, x13, lsl #3]
    add x13, x13, #1
    add x14, x14, #1
    b .Lloop
.Lswap:
    sub x16, x13, #1
    sub x17, x13, #2
    ldr x0, [x11, x16, lsl #3]
    ldr x1, [x11, x17, lsl #3]
    str x1, [x11, x16, lsl #3]
    str x0, [x11, x17, lsl #3]
    add x14, x14, #1
    b .Lloop
.Ldrop:
    sub x13, x13, #1
    add x14, x14, #1
    b .Lloop
.Lover:
    sub x16, x13, #2
    ldr x17, [x11, x16, lsl #3]
    str x17, [x11, x13, lsl #3]
    add x13, x13, #1
    add x14, x14, #1
    b .Lloop
.Llt:
    sub x13, x13, #1
    ldr x17, [x11, x13, lsl #3]   // b
    sub x16, x13, #1
    ldr x0, [x11, x16, lsl #3]    // a
    cmp x0, x17
    cset x0, lt
    str x0, [x11, x16, lsl #3]
    add x14, x14, #1
    b .Lloop
.Ljz:
    sub x13, x13, #1
    ldr x17, [x11, x13, lsl #3]   // v
    add x16, x14, #1
    ldr x0, [x9, x16, lsl #3]     // t
    add x14, x14, #2
    cbnz x17, .Lloop
    mov x14, x0
    b .Lloop
.Ljmp:
    add x16, x14, #1
    ldr x14, [x9, x16, lsl #3]
    b .Lloop
.Ldone:
    mov x0, xzr
    cbz x13, .Lzero
    sub x16, x13, #1
    ldr x0, [x11, x16, lsl #3]
.Lzero:
    cmp x13, x12
    b.ge .Lret
    str xzr, [x11, x13, lsl #3]
    add x13, x13, #1
    b .Lzero
.Lret:
    ret
.Ltbl:
    b .Lhalt
    b .Lpush
    b .Ladd
    b .Lsub
    b .Lmul
    b .Ldup
    b .Lswap
    b .Ldrop
    b .Lover
    b .Llt
    b .Ljz
    b .Ljmp
