vm_run:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi                 # code
mov r13, rsi                 # len
mov r14, rdx                 # stack
mov r15, rcx                 # cap
xor ebx, ebx                 # pc = 0
xor ebp, ebp                 # depth d = 0
.Lloop:
cmp rbx, r13
jae .Ldone
mov rax, qword ptr [r12 + rbx*8]
cmp rax, 11
ja .Lother
cmp rax, 6
jb .LA
cmp rax, 9
jb .LB
cmp rax, 10
je .Ljz
jb .Llt
jmp .Ljmp
.LB:
cmp rax, 7
je .Ldrop
jb .Lswap
jmp .Lover
.LA:
cmp rax, 2
jb .LC
cmp rax, 4
jb .LD
je .Lmul
jmp .Ldup
.LC:
test rax, rax
jz .Lhalt
jmp .Lpush
.LD:
cmp rax, 2
je .Ladd
jmp .Lsub
.Lpush:
mov rax, qword ptr [r12 + rbx*8 + 8]
mov qword ptr [r14 + rbp*8], rax
inc rbp
add rbx, 2
jmp .Lloop
.Ladd:
mov rax, qword ptr [r14 + rbp*8 - 16]
add rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Lsub:
mov rax, qword ptr [r14 + rbp*8 - 16]
sub rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Lmul:
mov rax, qword ptr [r14 + rbp*8 - 16]
imul rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8 - 16], rax
dec rbp
inc rbx
jmp .Lloop
.Ldup:
mov rax, qword ptr [r14 + rbp*8 - 8]
mov qword ptr [r14 + rbp*8], rax
inc rbp
inc rbx
jmp .Lloop
.Lswap:
mov rax, qword ptr [r14 + rbp*8 - 8]
mov rcx, qword ptr [r14 + rbp*8 - 16]
mov qword ptr [r14 + rbp*8 - 16], rax
mov qword ptr [r14 + rbp*8 - 8], rcx
inc rbx
jmp .Lloop
.Ldrop:
dec rbp
inc rbx
jmp .Lloop
.Lover:
mov rax, qword ptr [r14 + rbp*8 - 16]
mov qword ptr [r14 + rbp*8], rax
inc rbp
inc rbx
jmp .Lloop
.Llt:
xor ecx, ecx
mov rax, qword ptr [r14 + rbp*8 - 16]
cmp rax, qword ptr [r14 + rbp*8 - 8]
setl cl
mov qword ptr [r14 + rbp*8 - 16], rcx
dec rbp
inc rbx
jmp .Lloop
.Ljz:
mov rax, qword ptr [r14 + rbp*8 - 8]
dec rbp
test rax, rax
jz .Ljzt
add rbx, 2
jmp .Lloop
.Ljzt:
mov rbx, qword ptr [r12 + rbx*8 + 8]
jmp .Lloop
.Ljmp:
mov rbx, qword ptr [r12 + rbx*8 + 8]
jmp .Lloop
.Lother:
inc rbx
jmp .Lloop
.Lhalt:
.Ldone:
mov rcx, rbp
.Lzero:
cmp rcx, r15
jae .Lres
mov qword ptr [r14 + rcx*8], 0
inc rcx
jmp .Lzero
.Lres:
xor eax, eax
test rbp, rbp
jz .Lout
mov rax, qword ptr [r14 + rbp*8 - 8]
.Lout:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
