dot_i32_simd:
    movi v0.16b, #0               // vector accumulator 0 (two 64-bit lanes)
    movi v1.16b, #0               // vector accumulator 1 (two 64-bit lanes)
    mov x6, #0                    // scalar accumulator for tail elements
    lsr x3, x2, #3                // number of full 8-element chunks
    cbz x3, .Ltail
.Lloop:
    ld1 {v2.4s}, [x0], #16        // a[i..i+3] (4-byte alignment is fine)
    ld1 {v3.4s}, [x1], #16        // b[i..i+3]
    ld1 {v4.4s}, [x0], #16        // a[i+4..i+7]
    ld1 {v5.4s}, [x1], #16        // b[i+4..i+7]
    smlal v0.2d, v2.2s, v3.2s     // a0*b0, a1*b1 -> lanes of v0
    smlal2 v0.2d, v2.4s, v3.4s    // a2*b2, a3*b3 -> lanes of v0
    smlal v1.2d, v4.2s, v5.2s     // a4*b4, a5*b5 -> lanes of v1
    smlal2 v1.2d, v4.4s, v5.4s    // a6*b6, a7*b7 -> lanes of v1
    subs x3, x3, #1
    b.ne .Lloop
.Ltail:
    and x3, x2, #7                // remaining element count
    cbz x3, .Lreduce
.Lscalar:
    ldr w7, [x0], #4
    ldr w8, [x1], #4
    smaddl x6, w7, w8, x6         // x6 += sext(a[i]) * sext(b[i])
    subs x3, x3, #1
    b.ne .Lscalar
.Lreduce:
    addp v0.2d, v0.2d, v1.2d
    addp d0, v0.2d                // horizontal sum of the two 64-bit lanes
    fmov x7, d0
    add x0, x6, x7
    ret
