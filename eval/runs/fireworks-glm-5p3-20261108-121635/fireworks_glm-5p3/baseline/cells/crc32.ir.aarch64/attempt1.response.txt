export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	jmp @loop
@loop
	%done =w ceql %n, 0
	jnz %done, @end, @body
@body
	%b =w loadub %p
	%crc =w xor %crc, %b
	%j =w copy 0
	jmp @bitloop
@bitloop
	%more =w csltw %j, 8
	jnz %more, @bit, @next
@bit
	%low =w and %crc, 1
	%crc =w shr %crc, 1
	%mask =w mul %low, 3988292384
	%crc =w xor %crc, %mask
	%j =w add %j, 1
	jmp @bitloop
@next
	%p =l add %p, 1
	%n =l sub %n, 1
	jmp @loop
@end
	%res =w xor %crc, 4294967295
	%out =l extuw %res
	ret %out
}
