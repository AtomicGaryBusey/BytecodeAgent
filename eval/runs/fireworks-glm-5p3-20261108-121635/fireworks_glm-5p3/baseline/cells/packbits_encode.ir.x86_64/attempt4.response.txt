export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %three =l copy 3
    %c128 =l copy 128
    %onew =w copy 1
    %c257 =w copy 257
    %neg1 =l sub %zero, %one
    %i =l copy 0
    %wd =l copy 0
    jmp @main
@main
    %more =w csltl %i, %n
    jnz %more, @runinit, @done
@done
    ret %wd
@runinit
    %pio =l add %src, %i
    %bi0 =w loadub %pio
    %r =l copy 1
    %rw =w copy 1
    %k =l copy %i
    jmp @runloop
@runloop
    %kn =l add %k, %one
    %ca =w csltl %kn, %n
    jnz %ca, @runb, @runend
@runb
    %cb =w csltl %r, %c128
    jnz %cb, @runc, @runend
@runc
    %pk =l add %src, %kn
    %bk =w loadub %pk
    %cc =w ceqw %bk, %bi0
    jnz %cc, @runadv, @runend
@runadv
    %rn =l add %r, %one
    %r =l copy %rn
    %rwn =w add %rw, %onew
    %rw =w copy %rwn
    %k =l copy %kn
    jmp @runloop
@runend
    %islit =w csltl %r, %three
    jnz %islit, @litinit, @repchk
@repchk
    %need1 =l add %wd, %two
    %bad1 =w csltl %cap, %need1
    jnz %bad1, @fail, @repwr
@fail
    ret %neg1
@repwr
    %hdrw =w sub %c257, %rw
    %p0 =l add %out, %wd
    storeb %hdrw, %p0
    %psrc =l add %src, %i
    %b0 =w loadub %psrc
    %wdn =l add %wd, %one
    %p1 =l add %out, %wdn
    storeb %b0, %p1
    %wd2 =l add %wd, %two
    %wd =l copy %wd2
    %in =l add %i, %r
    %i =l copy %in
    jmp @main
@litinit
    %j =l copy %i
    %len =w copy 0
    jmp @litloop
@litloop
    %cja =w csltl %j, %n
    jnz %cja, @litb, @litend
@litb
    %dj =l sub %j, %i
    %cjb =w csltl %dj, %c128
    jnz %cjb, @litc, @litend
@litc
    %j2 =l add %j, %two
    %cjc =w csltl %j2, %n
    jnz %cjc, @litd, @litadv
@litd
    %pj =l add %src, %j
    %bj =w loadub %pj
    %pj1 =l add %pj, %one
    %bj1 =w loadub %pj1
    %pj2 =l add %pj, %two
    %bj2 =w loadub %pj2
    %e1 =w ceqw %bj, %bj1
    jnz %e1, @lite, @litadv
@lite
    %e2 =w ceqw %bj1, %bj2
    jnz %e2, @litend, @litadv
@litadv
    %jn =l add %j, %one
    %j =l copy %jn
    %len2 =w add %len, %onew
    %len =w copy %len2
    jmp @litloop
@litend
    %lenl =l sub %j, %i
    %nt =l add %wd, %lenl
    %need2 =l add %nt, %one
    %bad2 =w csltl %cap, %need2
    jnz %bad2, @fail2, @litwr
@fail2
    ret %neg1
@litwr
    %hdr2 =w sub %len, %onew
    %ph =l add %out, %wd
    storeb %hdr2, %ph
    %t =l copy 0
    jmp @cploop
@cploop
    %tc =w csltl %t, %lenl
    jnz %tc, @cpbody, @litfin
@cpbody
    %si =l add %i, %t
    %ps =l add %src, %si
    %bv =w loadub %ps
    %di =l add %wd, %t
    %di2 =l add %di, %one
    %pd =l add %out, %di2
    storeb %bv, %pd
    %tn =l add %t, %one
    %t =l copy %tn
    jmp @cploop
@litfin
    %wa =l add %wd, %lenl
    %wd3 =l add %wa, %one
    %wd =l copy %wd3
    %i =l copy %j
    jmp @main
}
