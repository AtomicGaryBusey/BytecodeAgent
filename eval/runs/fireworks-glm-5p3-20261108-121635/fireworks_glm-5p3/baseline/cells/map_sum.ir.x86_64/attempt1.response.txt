export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%s =l phi @start 0, @iter %s1
	%i =l phi @start 0, @iter %i1
	%cond =w csltl %i, %n
	jnz %cond, @iter, @end
@iter
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%r =l call %f(l %x)
	%s1 =l add %s, %r
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %s
}
