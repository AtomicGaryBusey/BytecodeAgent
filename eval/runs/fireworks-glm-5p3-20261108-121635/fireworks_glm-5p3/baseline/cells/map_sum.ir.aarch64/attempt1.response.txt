export function l $map_sum(l %a, l %n, l %f) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %inext
	%s =l phi @entry 0, @body %snext
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%o =l mul %i, 8
	%p =l add %a, %o
	%x =l loadl %p
	%r =l call %f(l %x)
	%snext =l add %s, %r
	%inext =l add %i, 1
	jmp @loop
@end
	ret %s
}
