export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%diff =l sub %dst, %src
	%cpos =w csgtl %diff, %zero
	jnz %cpos, @chk, @fwd
@chk
	%lim =l mul %n, 8
	%cund =w csltl %diff, %lim
	jnz %cund, @bwd, @fwd
@fwd
	%ps =l copy %src
	%pd =l copy %dst
	%i =l copy %zero
@fwdloop
	%cmore =w csltl %i, %n
	jnz %cmore, @fwdbody, @done
@fwdbody
	%v =l loadl %ps
	storel %v, %pd
	%ps =l add %ps, 8
	%pd =l add %pd, 8
	%i =l add %i, 1
	jmp @fwdloop
@bwd
	%j =l sub %n, 1
@bwdloop
	%cge =w csgel %j, %zero
	jnz %cge, @bwdbody, @done
@bwdbody
	%off =l mul %j, 8
	%qs =l add %src, %off
	%qd =l add %dst, %off
	%wv =l loadl %qs
	storel %wv, %qd
	%j =l sub %j, 1
	jmp @bwdloop
@done
	%r =l copy 0
	ret %r
}
