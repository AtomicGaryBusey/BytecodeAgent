export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%z =w ceql %n, 0
	jnz %z, @done, @cmp2
@cmp2
	%lt =w cultl %dst, %src
	jnz %lt, @forward, @cmp3
@cmp3
	%eq =w ceql %dst, %src
	jnz %eq, @done, @backward
@forward
	%sf =l phi @cmp2 - %src, @forward - %sf2
	%df =l phi @cmp2 - %dst, @forward - %df2
	%cf =l phi @cmp2 - %n, @forward - %cf2
	%vf =l loadl %sf
	storel %vf, %df
	%sf2 =l add %sf, 8
	%df2 =l add %df, 8
	%cf2 =l sub %cf, 1
	%mf =w csgtl %cf2, 0
	jnz %mf, @forward, @done
@backward
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 8
	%sb0 =l add %src, %off
	%db0 =l add %dst, %off
	jmp @bwdloop
@bwdloop
	%sb =l phi @backward - %sb0, @bwdloop - %sb2
	%db =l phi @backward - %db0, @bwdloop - %db2
	%cb =l phi @backward - %n, @bwdloop - %cb2
	%vb =l loadl %sb
	storel %vb, %db
	%sb2 =l sub %sb, 8
	%db2 =l sub %db, 8
	%cb2 =l sub %cb, 1
	%mb =w csgtl %cb2, 0
	jnz %mb, @bwdloop, @done
@done
	%r =l copy 0
	ret %r
}
