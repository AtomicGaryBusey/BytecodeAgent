export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l add %nn, %nn
	%nn8 =l mul %nn, 8
	%pb =l add %tmp, %nn8
	%zero =l copy 0
	jmp @init_i
@init_i
	%i =l phi @start %zero, @init_i_next %i_next
	%ci =w csltl %i, %n
	jnz %ci, @init_j, @mul_t
@init_j
	%j =l phi @init_i %zero, @init_body %j_next
	%cj =w csltl %j, %n
	jnz %cj, @init_body, @init_i_next
@init_body
	%d1 =l mul %i, %n
	%d2 =l add %d1, %j
	%d3 =l mul %d2, 8
	%d4 =l add %tmp, %d3
	%eq =w ceql %i, %j
	%one =l extsw %eq
	storel %one, %d4
	%j_next =l add %j, 1
	jmp @init_j
@init_i_next
	%i_next =l add %i, 1
	jmp @init_i
@mul_t
	%t =l phi @init_i %zero, @mul_t_next %t_next
	%cur =l phi @init_i %tmp, @mul_t_next %ncur
	%oth =l phi @init_i %pb, @mul_t_next %noth
	%ct =w csltl %t, %k
	jnz %ct, @mul_i, @copy_out
@mul_i
	%mi =l phi @mul_t %zero, @mul_i_next %mi_next
	%cmi =w csltl %mi, %n
	jnz %cmi, @mul_j, @mul_t_next
@mul_j
	%mj =l phi @mul_i %zero, @mul_store %mj_next
	%cmj =w csltl %mj, %n
	jnz %cmj, @mul_s, @mul_i_next
@mul_s
	%ms =l phi @mul_j %zero, @mul_s_body %ms_next
	%acc =l phi @mul_j %zero, @mul_s_body %acc_next
	%cms =w csltl %ms, %n
	jnz %cms, @mul_s_body, @mul_store
@mul_s_body
	%e1 =l mul %mi, %n
	%e2 =l add %e1, %ms
	%e3 =l mul %e2, 8
	%e4 =l add %cur, %e3
	%rv =l loadl %e4
	%e5 =l mul %ms, %n
	%e6 =l add %e5, %mj
	%e7 =l mul %e6, 8
	%e8 =l add %a, %e7
	%av =l loadl %e8
	%pr =l mul %rv, %av
	%acc_next =l add %acc, %pr
	%ms_next =l add %ms, 1
	jmp @mul_s
@mul_store
	%f1 =l mul %mi, %n
	%f2 =l add %f1, %mj
	%f3 =l mul %f2, 8
	%f4 =l add %oth, %f3
	storel %acc, %f4
	%mj_next =l add %mj, 1
	jmp @mul_j
@mul_i_next
	%mi_next =l add %mi, 1
	jmp @mul_i
@mul_t_next
	%t_next =l add %t, 1
	%ncur =l copy %oth
	%noth =l copy %cur
	jmp @mul_t
@copy_out
	%oi =l phi @mul_t %zero, @copy_out_body %oi_next
	%coi =w csltl %oi, %nn
	jnz %coi, @copy_out_body, @zero
@copy_out_body
	%g1 =l mul %oi, 8
	%g2 =l add %cur, %g1
	%ov =l loadl %g2
	%g3 =l add %out, %g1
	storel %ov, %g3
	%oi_next =l add %oi, 1
	jmp @copy_out
@zero
	%zi =l phi @copy_out %zero, @zero_body %zi_next
	%czi =w csltl %zi, %nn2
	jnz %czi, @zero_body, @retblk
@zero_body
	%h1 =l mul %zi, 8
	%h2 =l add %tmp, %h1
	storel %zero, %h2
	%zi_next =l add %zi, 1
	jmp @zero
@retblk
	%r0 =l copy 0
	ret %r0
}
