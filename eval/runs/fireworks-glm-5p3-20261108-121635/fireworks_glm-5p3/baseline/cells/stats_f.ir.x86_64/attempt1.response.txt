export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%odd0 =l and %y0, 1
	%cnt =l copy %odd0
	%p =l add %a, 8
	%i =l copy 1
	jmp @loop
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum =l add %sum, %y
	%odd =l and %y, 1
	%cnt =l add %cnt, %odd
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
	jmp @chkmax
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @next
@setmax
	%max =l copy %y
	jmp @next
@next
	%p =l add %p, 8
	%i =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%out8 =l add %out, 8
	storel %max, %out8
	%out16 =l add %out, 16
	storel %sum, %out16
	ret %cnt
}
