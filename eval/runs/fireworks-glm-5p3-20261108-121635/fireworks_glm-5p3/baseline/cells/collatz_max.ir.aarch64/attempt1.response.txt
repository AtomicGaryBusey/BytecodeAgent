export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@outer
	%done =w csltl %n, %k
	jnz %done, @end, @body
@body
	%x =l copy %k
	%c =l copy 0
@inner
	%isone =w ceql %x, 1
	jnz %isone, @after, @step
@step
	%odd =l and %x, 1
	%isodd =w cnel %odd, 0
	jnz %isodd, @oddb, @evenb
@evenb
	%x =l shr %x, 1
	jmp @count
@oddb
	%t3 =l mul %x, 3
	%x =l add %t3, 1
	jmp @count
@count
	%c =l add %c, 1
	jmp @inner
@after
	%better =w csltl %max, %c
	jnz %better, @update, @next
@update
	%max =l copy %c
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@end
	ret %max
}
