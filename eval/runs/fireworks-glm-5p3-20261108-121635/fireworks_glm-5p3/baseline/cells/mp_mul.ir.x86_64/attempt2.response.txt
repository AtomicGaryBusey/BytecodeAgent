export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	jmp @zero
@zero
	%zi =l phi @entry 0, @zero_body %zin
	%zc =w cultl %zi, %n
	jnz %zc, @zero_body, @iloop
@zero_body
	%zo =l mul %zi, 8
	%zp =l add %r, %zo
	storel 0, %zp
	%zin =l add %zi, 1
	jmp @zero
@iloop
	%i =l phi @zero 0, @i_end %in
	%ic =w cultl %i, %na
	jnz %ic, @i_body, @cnt
@i_body
	%ao =l mul %i, 8
	%ap =l add %a, %ao
	%ai =l loadl %ap
	jmp @jloop
@jloop
	%j =l phi @i_body 0, @j_body %jn
	%c =l phi @i_body 0, @j_body %h2
	%jc =w cultl %j, %nb
	jnz %jc, @j_body, @i_end
@j_body
	%ij =l add %i, %j
	%ijo =l mul %ij, 8
	%ijp =l add %r, %ijo
	%rv =l loadl %ijp
	%bo =l mul %j, 8
	%bp =l add %b, %bo
	%bj =l loadl %bp
	%a0 =l and %ai, 4294967295
	%a1 =l shr %ai, 32
	%b0 =l and %bj, 4294967295
	%b1 =l shr %bj, 32
	%t0 =l mul %a0, %b0
	%t1 =l mul %a0, %b1
	%t2 =l mul %a1, %b0
	%t3 =l mul %a1, %b1
	%m0 =l shr %t0, 32
	%m1 =l and %t1, 4294967295
	%m2 =l and %t2, 4294967295
	%s01 =l add %m0, %m1
	%s02 =l add %s01, %m2
	%cy =l shr %s02, 32
	%l0 =l and %t0, 4294967295
	%l1 =l and %s02, 4294967295
	%l1s =l shl %l1, 32
	%pl =l or %l0, %l1s
	%n1 =l shr %t1, 32
	%n2 =l shr %t2, 32
	%h01 =l add %t3, %n1
	%h02 =l add %h01, %n2
	%ph =l add %h02, %cy
	%s1 =l add %pl, %rv
	%cc1 =w cultl %s1, %pl
	%cc1l =l extuw %cc1
	%s2 =l add %s1, %c
	%cc2 =w cultl %s2, %s1
	%cc2l =l extuw %cc2
	%h1 =l add %ph, %cc1l
	%h2 =l add %h1, %cc2l
	storel %s2, %ijp
	%jn =l add %j, 1
	jmp @jloop
@i_end
	%ipn =l add %i, %nb
	%ipno =l mul %ipn, 8
	%ip =l add %r, %ipno
	storel %c, %ip
	%in =l add %i, 1
	jmp @iloop
@cnt
	%k =l phi @iloop %n, @cnt_dec %kn
	%kz =w ceql %k, 0
	jnz %kz, @fin, @cnt_chk
@cnt_chk
	%km =l sub %k, 1
	%kmo =l mul %km, 8
	%kp =l add %r, %kmo
	%kv =l loadl %kp
	%kvz =w ceql %kv, 0
	jnz %kvz, @cnt_dec, @fin
@cnt_dec
	%kn =l sub %k, 1
	jmp @cnt
@fin
	ret %k
}
