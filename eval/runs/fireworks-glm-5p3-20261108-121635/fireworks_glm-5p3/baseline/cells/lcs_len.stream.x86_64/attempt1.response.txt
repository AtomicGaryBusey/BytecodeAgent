lcs_len:
    push rbx
    push r12
    push r13
    push r14
    # row[0..n] = 0
    xor r9d, r9d
.Lfill:
    mov qword ptr [r8 + r9*8], 0
    inc r9
    cmp r9, rcx
    jbe .Lfill
    lea rbx, [r8 + rcx*8]          # rbx = &row[n]
.Louter:
    test rsi, rsi
    jz .Ldone
    movzx r11d, byte ptr [rdi]     # c = x[i]
    inc rdi
    dec rsi
    lea rax, [r8 + 8]              # p = &row[1]
    mov r9, rdx                    # q = &y[0]
    xor r10d, r10d                 # diag = 0
    xor r12d, r12d                 # prev = row[0] = 0
    cmp rax, rbx
    ja .Louter                     # n == 0: no inner work for this i
.Linner:
    mov r13, qword ptr [rax]       # up = row[j]
    lea r14, [r10 + 1]             # candidate A = diag + 1
    mov rcx, r13                   # candidate B = up
    cmp r13, r12
    cmovl rcx, r12                 # B = max(up, prev)
    cmp r11b, byte ptr [r9]        # c == y[j-1] ?
    cmovne r14, rcx                # take B if bytes differ, else A
    mov qword ptr [rax], r14       # row[j] = new value
    mov r12, r14                   # prev = row[j]
    mov r10, r13                   # diag = up
    add rax, 8
    inc r9
    cmp rax, rbx
    jbe .Linner
    jmp .Louter
.Ldone:
    mov rax, qword ptr [rbx]       # result = row[n]
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
