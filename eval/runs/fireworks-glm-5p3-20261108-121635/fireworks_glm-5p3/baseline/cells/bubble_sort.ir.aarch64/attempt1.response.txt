export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%i =l copy 0
@outer
	%c1 =w csltl %i, %nm1
	jnz %c1, @inner_init, @exit
@inner_init
	%lim =l sub %nm1, %i
	%j =l copy 0
	jmp @inner
@inner
	%c2 =w csltl %j, %lim
	jnz %c2, @cmp, @after_inner
@cmp
	%jo =l mul %j, 8
	%p0 =l add %a, %jo
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @swap, @advance
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @advance
@advance
	%j =l add %j, 1
	jmp @inner
@after_inner
	%i =l add %i, 1
	jmp @outer
@exit
	ret 0
}
