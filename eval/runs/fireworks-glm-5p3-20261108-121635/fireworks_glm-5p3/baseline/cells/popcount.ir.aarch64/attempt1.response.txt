export function l $popcount(l %x) {
@start
	%m55 =l copy 0x5555555555555555
	%m33 =l copy 0x3333333333333333
	%m0f =l copy 0x0f0f0f0f0f0f0f0f
	%m01 =l copy 0x0101010101010101
	%a =l shr %x, 1
	%b =l and %a, %m55
	%c =l sub %x, %b
	%d =l shr %c, 2
	%e =l and %c, %m33
	%f =l and %d, %m33
	%g =l add %e, %f
	%h =l shr %g, 4
	%i =l add %g, %h
	%j =l and %i, %m0f
	%k =l mul %j, %m01
	%r =l shr %k, 56
	ret %r
}
