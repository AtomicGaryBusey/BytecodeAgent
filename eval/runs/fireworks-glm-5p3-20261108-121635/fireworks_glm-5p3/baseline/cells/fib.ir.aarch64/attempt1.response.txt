export function l $fib(l %n) {
@start
	%a0 =l copy 0
	%b0 =l copy 1
	jmp @loop
@loop
	%a =l phi @start %a0, @body %na
	%b =l phi @start %b0, @body %nb
	%c =l phi @start %n, @body %c1
	%z =w ceql %c, 0
	jnz %z, @done, @body
@body
	%na =l copy %b
	%nb =l add %a, %b
	%c1 =l sub %c, 1
	jmp @loop
@done
	ret %a
}
