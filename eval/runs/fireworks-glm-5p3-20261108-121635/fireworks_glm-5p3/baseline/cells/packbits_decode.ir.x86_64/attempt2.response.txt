export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%kslot =l alloc8 8
	%kslot2 =l alloc8 8
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%c257 =l copy 257
	storel %zero, %islot
	storel %zero, %oslot
	jmp @loop
@loop
	%i =l loadl %islot
	%cont =w csltl %i, %n
	jnz %cont, @readhdr, @finish
@readhdr
	%hdrp =l add %src, %i
	%hu =w loadub %hdrp
	%isnop =w ceqw %hu, 128
	jnz %isnop, @noop, @chk
@noop
	%iinc =l add %i, 1
	storel %iinc, %islot
	jmp @loop
@chk
	%islit =w cultw %hu, 128
	jnz %islit, @lprep, @rprep
@lprep
	%hul =l extuw %hu
	%llen =l add %hul, 1
	%ip1 =l add %i, 1
	%lneed =l add %ip1, %llen
	%lbad =w csltl %n, %lneed
	jnz %lbad, @err2, @linit
@linit
	%ldatap =l add %src, %ip1
	storel %zero, %kslot
	jmp @lcopy
@lcopy
	%k =l loadl %kslot
	%lmore =w csltl %k, %llen
	jnz %lmore, @lbyte, @ldone
@lbyte
	%lo =l loadl %oslot
	%lsrcp =l add %ldatap, %k
	%lval =w loadub %lsrcp
	%lfit =w csltl %lo, %cap
	jnz %lfit, @lwrite, @lskip
@lwrite
	%loutp =l add %out, %lo
	storeb %lval, %loutp
	jmp @lskip
@lskip
	%lo1 =l add %lo, 1
	storel %lo1, %oslot
	%lover =w csltl %cap, %lo1
	jnz %lover, @err1, @lstep
@lstep
	%lk1 =l add %k, 1
	storel %lk1, %kslot
	jmp @lcopy
@ldone
	storel %lneed, %islot
	jmp @loop
@rprep
	%rneed =l add %i, 2
	%rbad =w csltl %n, %rneed
	jnz %rbad, @err2, @rinit
@rinit
	%rip1 =l add %i, 1
	%rdatap =l add %src, %rip1
	%rval =w loadub %rdatap
	%rhul =l extuw %hu
	%rlen =l sub %c257, %rhul
	storel %zero, %kslot2
	jmp @rcopy
@rcopy
	%rk =l loadl %kslot2
	%rmore =w csltl %rk, %rlen
	jnz %rmore, @rbyte, @rdone
@rbyte
	%ro =l loadl %oslot
	%rfit =w csltl %ro, %cap
	jnz %rfit, @rwrite, @rskip
@rwrite
	%routp =l add %out, %ro
	storeb %rval, %routp
	jmp @rskip
@rskip
	%ro1 =l add %ro, 1
	storel %ro1, %oslot
	%rover =w csltl %cap, %ro1
	jnz %rover, @err1, @rstep
@rstep
	%rk1 =l add %rk, 1
	storel %rk1, %kslot2
	jmp @rcopy
@rdone
	storel %rneed, %islot
	jmp @loop
@err2
	%m2 =l sub %zero, %two
	ret %m2
@err1
	%m1 =l sub %zero, %one
	ret %m1
@finish
	%res =l loadl %oslot
	ret %res
}
