stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24                  # re-align stack: 6 pushes + 24 keeps rsp 0 mod 16 at calls
    mov rbx, rdi                 # rbx = current element pointer
    lea rbp, [rdi + rsi*8]       # rbp = end pointer
    mov r12, rdx                 # r12 = f
    mov r13, rcx                 # r13 = out
    mov r14, 0x7fffffffffffffff  # min = INT64_MAX
    mov r15, 0x8000000000000000  # max = INT64_MIN
    mov qword ptr [rsp], 0       # sum
    mov qword ptr [rsp+8], 0     # odd count
.Lloop:
    mov rdi, qword ptr [rbx]
    call r12
    cmp rax, r14
    cmovl r14, rax               # min = y if y < min
    cmp rax, r15
    cmovg r15, rax               # max = y if y > max
    add qword ptr [rsp], rax     # sum += y (wrapping)
    and rax, 1
    add qword ptr [rsp+8], rax   # odd += (y & 1)
    add rbx, 8
    cmp rbx, rbp
    jne .Lloop
    mov rax, qword ptr [rsp+8]   # return value: odd count
    mov qword ptr [r13], r14     # out[0] = min
    mov qword ptr [r13+8], r15   # out[1] = max
    mov rcx, qword ptr [rsp]
    mov qword ptr [r13+16], rcx  # out[2] = sum
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
