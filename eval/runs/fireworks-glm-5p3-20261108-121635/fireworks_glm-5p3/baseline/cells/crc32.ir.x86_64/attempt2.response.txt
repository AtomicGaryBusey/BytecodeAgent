export function l $crc32(l %p, l %n) {
@start
	%z =w copy 0
	%ones =w sub %z, 1
	%hi =w copy 60856
	%hi =w shl %hi, 16
	%poly =w or %hi, 33536
	%c =w copy %ones
	%i =l copy 0
	jmp @byteloop
@byteloop
	%cont =w csltl %i, %n
	jnz %cont, @byte, @done
@byte
	%pb =l add %p, %i
	%b =w loadub %pb
	%c =w xor %c, %b
	%k =w copy 0
	jmp @bitloop
@bitloop
	%kc =w csltw %k, 8
	jnz %kc, @bit, @nextbyte
@bit
	%odd =w and %c, 1
	%c =w shr %c, 1
	jnz %odd, @flip, @even
@flip
	%c =w xor %c, %poly
	jmp @even
@even
	%k =w add %k, 1
	jmp @bitloop
@nextbyte
	%i =l add %i, 1
	jmp @byteloop
@done
	%r =w xor %c, %ones
	%res =l extuw %r
	ret %res
}
