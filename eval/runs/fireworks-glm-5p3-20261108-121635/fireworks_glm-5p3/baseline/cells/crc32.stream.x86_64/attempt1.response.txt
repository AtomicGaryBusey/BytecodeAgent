crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
.Louter:
    movzx edx, byte ptr [rdi]
    inc rdi
    xor eax, edx
    mov r8d, 8
.Linner:
    shr eax, 1
    jnc .Lskip
    xor eax, 0xEDB88320
.Lskip:
    dec r8d
    jnz .Linner
    dec rsi
    jnz .Louter
.Ldone:
    not eax
    ret
