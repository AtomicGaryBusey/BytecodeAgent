lcs_len:
    // x0=x, x1=m, x2=y, x3=n, x4=row
    // zero row[0..n]
    add x5, x4, x3, lsl #3
    add x5, x5, #8              // x5 = &row[n+1] = end of row
    mov x6, x4
.Lzero:
    cmp x6, x5
    b.hs .Lzero_done
    str xzr, [x6], #8
    b .Lzero
.Lzero_done:
    cbz x1, .Ldone              // m == 0 -> nothing to update
    mov x8, x0                  // cursor into x
.Louter:
    ldrb w9, [x8], #1           // c = x[i]
    mov x10, #0                 // diag = 0
    mov x16, #0                 // left = row[0] = 0
    mov x11, x2                 // cursor into y
    add x12, x4, #8             // &row[1]
.Linner:
    cmp x12, x5
    b.hs .Linner_end
    ldr x14, [x12]              // up = row[j]
    ldrb w15, [x11], #1         // y[j-1]
    cmp w9, w15
    b.eq .Lmatch
    cmp x14, x16
    csel x16, x14, x16, hs      // newval = max(up, left)
    b .Lstore
.Lmatch:
    add x16, x10, #1            // newval = diag + 1
.Lstore:
    str x16, [x12]              // row[j] = newval
    mov x10, x14                // diag = up
    add x12, x12, #8
    b .Linner
.Linner_end:
    subs x1, x1, #1
    b.ne .Louter
.Ldone:
    ldr x0, [x4, x3, lsl #3]    // result = row[n]
    ret
