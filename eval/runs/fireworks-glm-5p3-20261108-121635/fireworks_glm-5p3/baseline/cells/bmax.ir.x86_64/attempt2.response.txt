export function l $bmax(l %a, l %n) {
@start
	%best =l loadl %a
	%i =l copy 1
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%lt =w csltl %best, %v
	jnz %lt, @update, @next
@update
	%best =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %best
}
