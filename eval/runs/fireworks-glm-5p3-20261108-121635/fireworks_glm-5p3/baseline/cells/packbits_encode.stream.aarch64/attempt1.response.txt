packbits_encode:
    mov x5, xzr              // i = 0
    mov x6, xzr              // o = 0
.Lmain:
    cmp x5, x1
    b.ge .Ldone              // i >= n -> finished
    ldrb w9, [x0, x5]        // b = src[i]
    mov x7, #1               // r = 1
.Lrun:
    cmp x7, #128
    b.ge .Lrundone
    add x10, x5, x7          // i + r
    cmp x10, x1
    b.ge .Lrundone           // would read past src[n-1]
    ldrb w11, [x0, x10]
    cmp w11, w9
    b.ne .Lrundone
    add x7, x7, #1
    b .Lrun
.Lrundone:
    cmp x7, #3
    b.lt .Llit
    add x10, x6, #2          // o + 2
    cmp x10, x3
    b.gt .Lfail              // repeat packet does not fit
    mov w11, #257
    sub w11, w11, w7         // header = 257 - r
    strb w11, [x2, x6]
    add x6, x6, #1
    strb w9, [x2, x6]        // b
    add x6, x6, #1
    add x5, x5, x7           // i += r
    b .Lmain
.Llit:
    mov x12, x5              // j = i
.Lscan:
    cmp x12, x1
    b.ge .Lemit
    sub x13, x12, x5
    cmp x13, #128
    b.ge .Lemit
    ldrb w14, [x0, x12]      // src[j]
    add x13, x12, #1
    cmp x13, x1
    b.ge .Ladv               // run(j) = 1
    ldrb w15, [x0, x13]
    cmp w15, w14
    b.ne .Ladv               // run(j) = 1
    add x13, x12, #2
    cmp x13, x1
    b.ge .Ladv               // run(j) = 2
    ldrb w15, [x0, x13]
    cmp w15, w14
    b.eq .Lemit              // run(j) >= 3, stop before j
.Ladv:
    add x12, x12, #1
    b .Lscan
.Lemit:
    sub x13, x12, x5         // L = j - i
    add x14, x6, x13
    add x14, x14, #1         // o + 1 + L
    cmp x14, x3
    b.gt .Lfail              // literal packet does not fit
    sub w15, w13, #1         // header = L - 1
    strb w15, [x2, x6]
    add x6, x6, #1
    mov x7, x5               // k = i
.Lcopy:
    cmp x7, x12
    b.ge .Lcopydone
    ldrb w9, [x0, x7]
    strb w9, [x2, x6]
    add x7, x7, #1
    add x6, x6, #1
    b .Lcopy
.Lcopydone:
    mov x5, x12              // i = j
    b .Lmain
.Lfail:
    movn x0, #0              // -1
    ret
.Ldone:
    mov x0, x6
    ret
