export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%ll =l mul %alo, %blo
	%lh =l mul %alo, %bhi
	%hl =l mul %ahi, %blo
	%hh =l mul %ahi, %bhi
	%lhh =l shr %lh, 32
	%hlh =l shr %hl, 32
	%llo =l and %lh, 4294967295
	%hlo =l and %hl, 4294967295
	%llh =l shr %ll, 32
	%s0 =l add %llo, %hlo
	%s1 =l add %s0, %llh
	%carry =l shr %s1, 32
	%t0 =l add %hh, %lhh
	%t1 =l add %t0, %hlh
	%t2 =l add %t1, %carry
	ret %t2
}
