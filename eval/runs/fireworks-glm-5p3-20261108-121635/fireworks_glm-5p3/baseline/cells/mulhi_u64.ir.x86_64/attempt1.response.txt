export function l $mulhi_u64(l %a, l %b) {
@start
	%alo = and %a, 4294967295
	%ahi = shr %a, 32
	%blo = and %b, 4294967295
	%bhi = shr %b, 32
	%p0 = mul %alo, %blo
	%p1 = mul %alo, %bhi
	%p2 = mul %ahi, %blo
	%p3 = mul %ahi, %bhi
	%p0hi = shr %p0, 32
	%p1hi = shr %p1, 32
	%p1lo = and %p1, 4294967295
	%p2hi = shr %p2, 32
	%p2lo = and %p2, 4294967295
	%s1 = add %p0hi, %p1lo
	%s2 = add %s1, %p2lo
	%cy = shr %s2, 32
	%r1 = add %p3, %p1hi
	%r2 = add %r1, %p2hi
	%r3 = add %r2, %cy
	ret %r3
}
