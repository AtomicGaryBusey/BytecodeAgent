bubble_sort:
    cmp x1, #1
    b.le .Ldone
    sub x1, x1, #1
.Louter:
    mov x2, x1
    mov x3, #0
    mov x4, #0
.Linner:
    ldr x6, [x0, x3, lsl #3]
    add x7, x3, #1
    ldr x8, [x0, x7, lsl #3]
    cmp x6, x8
    b.le .Lnoswap
    str x8, [x0, x3, lsl #3]
    str x6, [x0, x7, lsl #3]
    mov x4, #1
.Lnoswap:
    add x3, x3, #1
    subs x2, x2, #1
    b.ne .Linner
    cbz x4, .Ldone
    subs x1, x1, #1
    b.ne .Louter
.Ldone:
    mov x0, #0
    ret
