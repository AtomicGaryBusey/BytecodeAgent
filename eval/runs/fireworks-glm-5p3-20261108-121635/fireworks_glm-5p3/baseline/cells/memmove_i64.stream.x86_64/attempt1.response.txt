memmove_i64:
    test rdx, rdx
    je .Lret
    cmp rdi, rsi
    jae .Lbackward
    xor ecx, ecx
.Lforward:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jb .Lforward
    jmp .Lret
.Lbackward:
    lea rcx, [rdx - 1]
.Lback:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    dec rcx
    jns .Lback
.Lret:
    xor eax, eax
    ret
