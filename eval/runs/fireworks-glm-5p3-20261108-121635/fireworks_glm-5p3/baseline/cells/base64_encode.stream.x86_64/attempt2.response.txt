base64_encode:
    push rbx
    mov r8, rdi                  # r8 = src cursor
    mov r9, rdx                  # r9 = out cursor
    mov r10, rdx                 # r10 = out start
    xor ebx, ebx                 # pad count for final group
    test rsi, rsi
    jz .Ldone
.Lgroup:
    cmp rsi, 3
    jb .Ltail
    movzx eax, byte ptr [r8]
    movzx ecx, byte ptr [r8 + 1]
    movzx edx, byte ptr [r8 + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov r11d, eax                # 24-bit group value
    add r8, 3
    sub rsi, 3
    mov edi, 18
    jmp .Lquad
.Ltail:
    movzx eax, byte ptr [r8]
    shl eax, 16
    cmp rsi, 2
    jb .Ltail2
    movzx ecx, byte ptr [r8 + 1]
    shl ecx, 8
    or eax, ecx
.Ltail2:
    mov ebx, 3
    sub ebx, esi                 # pad chars needed: 2 or 1
    imul edi, esi, 6             # lowest shift used (6 or 12)
    mov ecx, 18
    sub ecx, edi                 # pre-shift amount
    shr eax, cl
    mov r11d, eax
    xor esi, esi                 # no further groups
.Lquad:
    mov eax, r11d
    mov ecx, edi
    shr eax, cl
    and eax, 63
    lea ecx, [rax + 65]          # 'A' + v
    lea edx, [rax + 71]          # 'a' + v - 26
    cmp eax, 26
    cmovge ecx, edx
    lea edx, [rax - 4]           # '0' + v - 52
    cmp eax, 52
    cmovge ecx, edx
    lea edx, [rax - 19]          # '+' for v == 62
    cmp eax, 62
    cmovge ecx, edx
    lea edx, [rax - 16]          # '/' for v == 63
    cmp eax, 63
    cmove ecx, edx
    mov byte ptr [r9], cl
    inc r9
    sub edi, 6
    jns .Lquad
    test rsi, rsi
    jnz .Lgroup
    test ebx, ebx
    jz .Ldone
    mov byte ptr [r9], 61        # '='
    cmp ebx, 2
    jb .Lpad1
    mov byte ptr [r9 + 1], 61
    add r9, 2
    jmp .Ldone
.Lpad1:
    inc r9
.Ldone:
    mov rax, r9
    sub rax, r10
    pop rbx
    ret
