export function l $vm_run(l %code, l %len, l %stk, l %cap) {
@start
	%pc =l copy $0
	%d =l copy $0
	jmp @loop
@loop
	%finw =w ceql %pc, %len
	jnz %finw, @halt, @fetch
@fetch
	%p8 =l mul %pc, $8
	%oadr =l add %code, %p8
	%op =l loadl %oadr
	%big =w cugtl %op, $11
	jnz %big, @other, @sw0
@sw0
	%c0 =w ceql %op, $0
	jnz %c0, @halt, @sw1
@sw1
	%c1 =w ceql %op, $1
	jnz %c1, @i_push, @sw2
@sw2
	%c2 =w ceql %op, $2
	jnz %c2, @i_add, @sw3
@sw3
	%c3 =w ceql %op, $3
	jnz %c3, @i_sub, @sw4
@sw4
	%c4 =w ceql %op, $4
	jnz %c4, @i_mul, @sw5
@sw5
	%c5 =w ceql %op, $5
	jnz %c5, @i_dup, @sw6
@sw6
	%c6 =w ceql %op, $6
	jnz %c6, @i_swap, @sw7
@sw7
	%c7 =w ceql %op, $7
	jnz %c7, @i_drop, @sw8
@sw8
	%c8 =w ceql %op, $8
	jnz %c8, @i_over, @sw9
@sw9
	%c9 =w ceql %op, $9
	jnz %c9, @i_lt, @sw10
@sw10
	%ca =w ceql %op, $10
	jnz %ca, @i_jz, @sw11
@sw11
	%cb =w ceql %op, $11
	jnz %cb, @i_jmp, @other
@i_push
	%oadrp =l add %oadr, $8
	%vp =l loadl %oadrp
	%dp8 =l mul %d, $8
	%padr =l add %stk, %dp8
	storel %vp, %padr
	%d =l add %d, $1
	%pc =l add %pc, $2
	jmp @loop
@i_add
	%am2 =l sub %d, $2
	%aa8 =l mul %am2, $8
	%aadr =l add %stk, %aa8
	%a1 =l loadl %aadr
	%aadr2 =l add %aadr, $8
	%a2 =l loadl %aadr2
	%ar =l add %a1, %a2
	storel %ar, %aadr
	%d =l sub %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_sub
	%sm2 =l sub %d, $2
	%sa8 =l mul %sm2, $8
	%sadr =l add %stk, %sa8
	%s1 =l loadl %sadr
	%sadr2 =l add %sadr, $8
	%s2 =l loadl %sadr2
	%sr =l sub %s1, %s2
	storel %sr, %sadr
	%d =l sub %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_mul
	%mm2 =l sub %d, $2
	%ma8 =l mul %mm2, $8
	%madr =l add %stk, %ma8
	%m1 =l loadl %madr
	%madr2 =l add %madr, $8
	%m2v =l loadl %madr2
	%mr =l mul %m1, %m2v
	storel %mr, %madr
	%d =l sub %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_dup
	%dm1u =l sub %d, $1
	%du8 =l mul %dm1u, $8
	%dadr =l add %stk, %du8
	%dv =l loadl %dadr
	%dtadr =l add %dadr, $8
	storel %dv, %dtadr
	%d =l add %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_swap
	%wm2 =l sub %d, $2
	%wa8 =l mul %wm2, $8
	%wadr =l add %stk, %wa8
	%w1 =l loadl %wadr
	%wadr2 =l add %wadr, $8
	%w2 =l loadl %wadr2
	storel %w2, %wadr
	storel %w1, %wadr2
	%pc =l add %pc, $1
	jmp @loop
@i_drop
	%d =l sub %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_over
	%om2 =l sub %d, $2
	%oa8 =l mul %om2, $8
	%oadr3 =l add %stk, %oa8
	%ov =l loadl %oadr3
	%otadr =l add %oadr3, $16
	storel %ov, %otadr
	%d =l add %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_lt
	%tm2 =l sub %d, $2
	%ta8 =l mul %tm2, $8
	%tadr =l add %stk, %ta8
	%t1 =l loadl %tadr
	%tadr2 =l add %tadr, $8
	%t2 =l loadl %tadr2
	%tlw =w csltl %t1, %t2
	%tlv =l extsw %tlw
	storel %tlv, %tadr
	%d =l sub %d, $1
	%pc =l add %pc, $1
	jmp @loop
@i_jz
	%zm1 =l sub %d, $1
	%za8 =l mul %zm1, $8
	%zadr =l add %stk, %za8
	%zv =l loadl %zadr
	%d =l sub %d, $1
	%zw =w ceql %zv, $0
	jnz %zw, @jz_t, @jz_f
@jz_t
	%zo =l add %oadr, $8
	%zt =l loadl %zo
	%pc =l copy %zt
	jmp @loop
@jz_f
	%pc =l add %pc, $2
	jmp @loop
@i_jmp
	%jo =l add %oadr, $8
	%jt =l loadl %jo
	%pc =l copy %jt
	jmp @loop
@other
	%pc =l add %pc, $1
	jmp @loop
@halt
	%dz =w ceql %d, $0
	jnz %dz, @h0, @h1
@h0
	%res =l copy $0
	jmp @zloop
@h1
	%hm1 =l sub %d, $1
	%ha8 =l mul %hm1, $8
	%hadr =l add %stk, %ha8
	%res =l loadl %hadr
	jmp @zloop
@zloop
	%zdn =w ceql %d, %cap
	jnz %zdn, @fin, @zbody
@zbody
	%zb8 =l mul %d, $8
	%zbadr =l add %stk, %zb8
	storel $0, %zbadr
	%d =l add %d, $1
	jmp @zloop
@fin
	ret %res
}
