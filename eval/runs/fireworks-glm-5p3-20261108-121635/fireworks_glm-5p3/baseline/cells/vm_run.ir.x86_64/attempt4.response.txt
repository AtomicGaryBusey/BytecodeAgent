export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%pc0 =l copy 0
	%sp0 =l copy 0
	jmp @loop
@loop
	%pc =l phi @entry %pc0, @h_push %npc_push, @h_add %npc_add, @h_sub %npc_sub, @h_mul %npc_mul, @h_dup %npc_dup, @h_swap %npc_swap, @h_drop %npc_drop, @h_over %npc_over, @h_lt %npc_lt, @h_jzt %npc_jzt, @h_jzf %npc_jzf, @h_jmp %npc_jmp, @h_other %npc_oth
	%sp =l phi @entry %sp0, @h_push %nsp_push, @h_add %nsp_add, @h_sub %nsp_sub, @h_mul %nsp_mul, @h_dup %nsp_dup, @h_swap %nsp_swap, @h_drop %nsp_drop, @h_over %nsp_over, @h_lt %nsp_lt, @h_jzt %nsp_jz, @h_jzf %nsp_jz, @h_jmp %nsp_jmp, @h_other %nsp_oth
	%run =w csltl %pc, %len
	jnz %run, @fetch, @fin
@fetch
	%pc8 =l mul %pc, 8
	%ip =l add %code, %pc8
	%op =l loadl %ip
	%q0 =w ceql %op, 0
	jnz %q0, @fin, @d1
@d1
	%q1 =w ceql %op, 1
	jnz %q1, @h_push, @d2
@d2
	%q2 =w ceql %op, 2
	jnz %q2, @h_add, @d3
@d3
	%q3 =w ceql %op, 3
	jnz %q3, @h_sub, @d4
@d4
	%q4 =w ceql %op, 4
	jnz %q4, @h_mul, @d5
@d5
	%q5 =w ceql %op, 5
	jnz %q5, @h_dup, @d6
@d6
	%q6 =w ceql %op, 6
	jnz %q6, @h_swap, @d7
@d7
	%q7 =w ceql %op, 7
	jnz %q7, @h_drop, @d8
@d8
	%q8 =w ceql %op, 8
	jnz %q8, @h_over, @d9
@d9
	%q9 =w ceql %op, 9
	jnz %q9, @h_lt, @d10
@d10
	%qa =w ceql %op, 10
	jnz %
