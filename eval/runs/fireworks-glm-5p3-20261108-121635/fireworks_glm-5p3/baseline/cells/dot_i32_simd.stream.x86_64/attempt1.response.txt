dot_i32_simd:
    xor eax, eax
    xor r10d, r10d
    pxor xmm4, xmm4
    mov r9, rdx
    shr r9, 2
    test r9, r9
    je .Lscalar
.Lvec:
    movdqu xmm0, xmmword ptr [rdi + r10*4]
    movdqu xmm1, xmmword ptr [rsi + r10*4]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pshufd xmm2, xmm2, 0xF5
    pshufd xmm3, xmm3, 0xF5
    pmuldq xmm0, xmm1
    pmuldq xmm2, xmm3
    paddq xmm0, xmm2
    paddq xmm4, xmm0
    add r10, 4
    dec r9
    jne .Lvec
.Lscalar:
    cmp r10, rdx
    jge .Lcombine
    movsxd rcx, dword ptr [rdi + r10*4]
    movsxd r11, dword ptr [rsi + r10*4]
    imul rcx, r11
    add rax, rcx
    inc r10
    jmp .Lscalar
.Lcombine:
    pshufd xmm0, xmm4, 0x0E
    paddq xmm4, xmm0
    movq rcx, xmm4
    add rax, rcx
    ret
