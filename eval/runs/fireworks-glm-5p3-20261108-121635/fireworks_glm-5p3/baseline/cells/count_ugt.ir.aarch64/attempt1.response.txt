export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%c =w phi @start 0, @body %c1
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%o =l mul %i, 8
	%p =l add %a, %o
	%v =l loadl %p
	%g =w cugtl %v, %t
	%c1 =w add %c, %g
	%i1 =l add %i, 1
	jmp @loop
@end
	%r =l extuw %c
	ret %r
}
