packbits_decode:
    add x1, x1, x0              // x1 = src + n (source end)
    mov x10, x2                 // x10 = out base
    add x3, x3, x2              // x3 = out + cap (output end)
.Lloop:
    cmp x0, x1                  // any input left?
    b.hs .Ldone
    ldrsb w4, [x0], #1          // h = signed header byte
    cmp w4, #0
    b.ge .Llit                  // 0..127: literal packet
    cmn w4, #128                // h == -128 ?
    b.eq .Lloop                 // no-op packet
    // repeat packet: -127 <= h <= -1
    cmp x0, x1                  // need one payload byte
    b.hs .Lsrcerr
    ldrb w6, [x0], #1           // b = byte to repeat
    neg w5, w4
    add w5, w5, #1              // R = 1 - h  (2..128)
    sub x7, x3, x2              // room left in out
    cmp x7, x5
    b.hs .Lrepfull              // room >= R
    mov x8, x7                  // write only what fits
    cbz x8, .Lcaperr
.Lreppart:
    strb w6, [x2], #1
    subs x8, x8, #1
    b.ne .Lreppart
    b .Lcaperr
.Lrepfull:
    mov x8, x5
.Lreploop:
    strb w6, [x2], #1
    subs x8, x8, #1
    b.ne .Lreploop
    b .Lloop
.Llit:
    add w5, w4, #1              // L = h + 1  (1..128)
    sub x6, x1, x0              // source bytes remaining
    cmp x6, x5
    b.lo .Lsrcerr               // not enough source bytes
    sub x7, x3, x2              // room left in out
    cmp x7, x5
    b.hs .Llitfull              // room >= L
    mov x8, x7                  // write only what fits
    cbz x8, .Lcaperr
.Llitpart:
    ldrb w9, [x0], #1
    strb w9, [x2], #1
    subs x8, x8, #1
    b.ne .Llitpart
    b .Lcaperr
.Llitfull:
    mov x8, x5
.Llitloop:
    ldrb w9, [x0], #1
    strb w9, [x2], #1
    subs x8, x8, #1
    b.ne .Llitloop
    b .Lloop
.Lsrcerr:
    movn x0, #1                 // return -2: truncated source
    ret
.Lcaperr:
    movn x0, #0                 // return -1: output exceeded cap
    ret
.Ldone:
    sub x0, x2, x10             // return bytes written
    ret
