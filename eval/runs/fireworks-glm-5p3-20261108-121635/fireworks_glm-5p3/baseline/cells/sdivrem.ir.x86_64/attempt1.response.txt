export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%z =w ceql %b, 0
	jnz %z, @bzero, @bchk
@bchk
	%bp1 =l add %b, 1
	%n =w ceql %bp1, 0
	jnz %n, @bnegone, @bnormal
@bzero
	storel %a, %rem
	ret 0
@bnegone
	%q =l sub 0, %a
	storel 0, %rem
	ret %q
@bnormal
	%quot =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %quot
}
