collatz_max:
    xor eax, eax                # max = 0
    mov rcx, 1                  # k = 1
.Louter:
    mov r8, rcx                 # x = k
    xor r9d, r9d                # steps = 0
.Linner:
    cmp r8, 1
    je .Lsteps_done
    test r8, 1
    jz .Leven
    lea r8, [r8 + r8*2 + 1]     # x = 3x + 1
    inc r9
    jmp .Linner
.Leven:
    shr r8, 1                   # x = x / 2
    inc r9
    jmp .Linner
.Lsteps_done:
    cmp r9, rax
    jle .Lnext
    mov rax, r9                 # new max
.Lnext:
    inc rcx
    cmp rcx, rdi
    jle .Louter
    ret
