collatz_max:
    mov x1, #0
    mov x2, #1
.Louter:
    cmp x2, x0
    b.gt .Ldone
    mov x3, x2
    mov x4, #0
.Linner:
    cmp x3, #1
    b.eq .Lnext
    tst x3, #1
    b.ne .Lodd
    lsr x3, x3, #1
    b .Lcount
.Lodd:
    add x3, x3, x3, lsl #1
    add x3, x3, #1
.Lcount:
    add x4, x4, #1
    b .Linner
.Lnext:
    cmp x4, x1
    b.le .Lskip
    mov x1, x4
.Lskip:
    add x2, x2, #1
    b .Louter
.Ldone:
    mov x0, x1
    ret
