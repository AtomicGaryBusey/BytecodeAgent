export function l $binary_search(l %a, l %n, l %key) {
@entry
	%hn =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @entry 0, @low %nlo, @high %lo
	%hi =l phi @entry %hn, @low %hi, @high %nhi
	%ok =w cslel %lo, %hi
	jnz %ok, @body, @notfound
@body
	%s =l add %lo, %hi
	%m =l div %s, 2
	%o =l mul %m, 8
	%p =l add %a, %o
	%v =l loadl %p
	%e =w ceql %v, %key
	jnz %e, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @low, @high
@low
	%nlo =l add %m, 1
	jmp @loop
@high
	%nhi =l sub %m, 1
	jmp @loop
@found
	ret %m
@notfound
	%r =l copy -1
	ret %r
}
