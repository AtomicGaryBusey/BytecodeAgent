max_byte:
    test rsi, rsi                # n == 0 -> return 0 without reading
    je .Lzero
    cmp rsi, 16
    jb .Lsmall
    mov rax, rsi
    and rax, -16                 # bytes covered by full 16-byte chunks
    xor ecx, ecx
    pxor xmm0, xmm0
.Lvec:
    movdqu xmm1, [rdi + rcx]
    pmaxub xmm0, xmm1
    add rcx, 16
    cmp rcx, rax
    jb .Lvec
    movdqa xmm1, xmm0            # horizontal max of the 16 bytes
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx edx, al                # edx = running max
    jmp .Ltail
.Lsmall:
    xor edx, edx
    xor ecx, ecx
.Ltail:
    cmp rcx, rsi
    jae .Ldone
.Lloop:
    movzx eax, byte ptr [rdi + rcx]
    cmp eax, edx
    cmova edx, eax
    inc rcx
    cmp rcx, rsi
    jb .Lloop
.Ldone:
    mov eax, edx
    ret
.Lzero:
    xor eax, eax
    ret
