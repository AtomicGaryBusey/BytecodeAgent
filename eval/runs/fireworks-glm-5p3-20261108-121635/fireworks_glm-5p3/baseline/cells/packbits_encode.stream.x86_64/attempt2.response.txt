packbits_encode:
    push rbx
    push r12
    xor r12d, r12d
    xor ebx, ebx
.Lmain:
    cmp rbx, rsi
    jae .Ldone
    movzx r8d, byte ptr [rdi + rbx]
    lea r11, [rbx + 1]
    mov r10d, 1
.Lrunloop:
    cmp r11, rsi
    jae .Lrunend
    cmp r10, 128
    jae .Lrunend
    movzx eax, byte ptr [rdi + r11]
    cmp al, r8b
    jne .Lrunend
    inc r10
    inc r11
    jmp .Lrunloop
.Lrunend:
    cmp r10, 3
    jb .Lliteral
    lea rax, [r12 + 2]
    cmp rax, rcx
    ja .Loverflow
    mov r11d, 257
    sub r11, r10
    mov byte ptr [rdx + r12], r11b
    mov byte ptr [rdx + r12 + 1], r8b
    add r12, 2
    add rbx, r10
    jmp .Lmain
.Lliteral:
    mov r10, rbx
.Lscan:
    cmp r10, rsi
    jae .Lemit
    mov r11, r10
    sub r11, rbx
    cmp r11, 128
    jae .Lemit
    lea r11, [r10 + 2]
    cmp r11, rsi
    jae .Ladv
    movzx eax, byte ptr [rdi + r10]
    movzx r9d, byte ptr [rdi + r10 + 1]
    cmp al, r9b
    jne .Ladv
    movzx r9d, byte ptr [rdi + r10 + 2]
    cmp al, r9b
    jne .Ladv
    jmp .Lemit
.Ladv:
    inc r10
    jmp .Lscan
.Lemit:
    mov r11, r10
    sub r11, rbx
    lea rax, [r12 + r11 + 1]
    cmp rax, rcx
    ja .Loverflow
    lea rax, [r11 - 1]
    mov byte ptr [rdx + r12], al
    inc r12
.Lcopy:
    cmp rbx, r10
    jae .Lcopydone
    movzx eax, byte ptr [rdi + rbx]
    mov byte ptr [rdx + r12], al
    inc rbx
    inc r12
    jmp .Lcopy
.Lcopydone:
    jmp .Lmain
.Loverflow:
    mov rax, -1
    jmp .Lout
.Ldone:
    mov rax, r12
.Lout:
    pop r12
    pop rbx
    ret
