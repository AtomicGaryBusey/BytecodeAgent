export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @iter %i1
	%j =l phi @start %j0, @iter %j1
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @end
@body
	%i8 =l mul %i, 8
	%pa =l add %a, %i8
	%va =l loadl %pa
	%j8 =l mul %j, 8
	%pb =l add %a, %j8
	%vb =l loadl %pb
	storel %vb, %pa
	storel %va, %pb
@iter
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@end
	ret 0
}
