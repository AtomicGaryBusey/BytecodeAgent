	export function l $count_desc(l %a, l %n) {
	@init
		%last =l subl %n, 1
		jmp @loop
	@loop
		%cnt =l phi @init 0, @body %cnt1
		%i =l phi @init 0, @body %i1
		%more = csltl %i, %last
		jnz %more, @body, @done
	@body
		%off =l mull %i, 8
		%p0 =l addl %a, %off
		%v0 =l loadl %p0
		%p1 =l addl %p0, 8
		%v1 =l loadl %p1
		%desc =w csgtl %v0, %v1
		%dv =l extuw %desc
		%cnt1 =l addl %cnt, %dv
		%i1 =l addl %i, 1
		jmp @loop
	@done
		ret %cnt
	}
