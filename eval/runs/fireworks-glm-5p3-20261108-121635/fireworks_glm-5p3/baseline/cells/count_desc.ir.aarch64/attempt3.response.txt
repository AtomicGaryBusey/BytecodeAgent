export function l $count_desc(l %a, l %n) {
@start
	%less =w csltl %n, 2
	jnz %less, @ret0, @init
@ret0
	ret 0
@init
	%i =l copy 0
	%cnt =l copy 0
	%m =l sub %n, 1
	jmp @loop
@loop
	%cont =w csltl %i, %m
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%ai =l add %a, %off
	%x0 =l loadl %ai
	%aip =l add %ai, 8
	%x1 =l loadl %aip
	%gt =w csgtl %x0, %x1
	%gtl =l extuw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
