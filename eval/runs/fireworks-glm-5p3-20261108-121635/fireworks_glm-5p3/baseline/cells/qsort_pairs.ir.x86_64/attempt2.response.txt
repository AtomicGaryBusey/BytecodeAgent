export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w cslel %n, 1
	jnz %small, @base, @init
@base
	%zero =l sub %n, %n
	ret %zero
@init
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 16
	%last =l add %a, %off
	%lastvp =l add %last, 8
	%pk =l loadl %last
	%pv =l loadl %lastvp
	jmp @loop
@loop
	%pi =l phi @init %a, @swapped %pinext, @kept %pisame
	%pj =l phi @init %a, @swapped %pjnext, @kept %pjnext2
	%done =w ceql %pj, %last
	jnz %done, @final, @scan
@scan
	%jk =l loadl %pj
	%jvp =l add %pj, 8
	%jv =l loadl %jvp
	%klt =w csltl %jk, %pk
	jnz %klt, @swapped, @cmpk
@cmpk
	%keq =w ceql %jk, %pk
	jnz %keq, @cmpv, @kept
@cmpv
	%vlt =w csltl %jv, %pv
	jnz %vlt, @swapped, @kept
@swapped
	%ik =l loadl %pi
	%ivp =l add %pi, 8
	%iv =l loadl %ivp
	storel %jk, %pi
	storel %jv, %ivp
	storel %ik, %pj
	storel %iv, %jvp
	%pinext =l add %pi, 16
	%pjnext =l add %pj, 16
	jmp @loop
@kept
	%pisame =l copy %pi
	%pjnext2 =l add %pj, 16
	jmp @loop
@final
	%fk =l loadl %pi
	%fvp =l add %pi, 8
	%fv =l loadl %fvp
	storel %pk, %pi
	storel %pv, %fvp
	storel %fk, %last
	storel %fv, %lastvp
	%diff =l sub %pi, %a
	%cl =l sar %diff, 4
	%r1 =l call $qsort_pairs(l %a, l %cl)
	%rp =l add %pi, 16
	%cr0 =l sub %n, %cl
	%cr =l sub %cr0, 1
	%r2 =l call $qsort_pairs(l %rp, l %cr)
	%z =l sub %r2, %r2
	ret %z
}
