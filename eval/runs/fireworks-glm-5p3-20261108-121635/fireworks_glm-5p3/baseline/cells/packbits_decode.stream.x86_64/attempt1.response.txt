packbits_decode:
    mov r8, rdi                    # p = src cursor
    lea r9, [rdi + rsi]            # send = src + n
    mov r10, rdx                   # q = out cursor
    lea r11, [rdx + rcx]           # oend = out + cap
    mov rdi, rdx                   # keep out base for the final count
.Lloop:
    cmp r8, r9
    jae .Ldone
    movsx eax, byte ptr [r8]       # h (signed)
    test eax, eax
    jns .Llit
    cmp al, 0x80
    je .Lnop
    lea rdx, [r8 + 2]              # packet needs header + value byte
    cmp rdx, r9
    ja .Lbadsrc
    neg rax
    inc rax                        # rax = 1 - h, run length 2..128
    movzx esi, byte ptr [r8 + 1]   # value byte
    mov rdx, r11
    sub rdx, r10                   # rdx = space left in out
    cmp rax, rdx
    jbe .Lrepfull
    mov rcx, rdx                   # only the fitting part gets written
    test rcx, rcx
    jz .Ltoobig
.Lreppart:
    mov [r10], sil
    inc r10
    dec rcx
    jnz .Lreppart
    jmp .Ltoobig
.Lrepfull:
    mov rcx, rax
.Lrepfill:
    mov [r10], sil
    inc r10
    dec rcx
    jnz .Lrepfill
    add r8, 2
    jmp .Lloop
.Llit:
    inc r8                         # r8 = payload start
    lea ecx, [rax + 1]             # ecx = h + 1, length 1..128
    lea rax, [r8 + rcx]            # end of payload
    cmp rax, r9
    ja .Lbadsrc
    mov rdx, r11
    sub rdx, r10                   # rdx = space left in out
    cmp rcx, rdx
    jbe .Llitcopy
    mov rcx, rdx                   # only the fitting part gets written
    test rcx, rcx
    jz .Ltoobig
.Llitpart:
    mov al, [r8]
    mov [r10], al
    inc r8
    inc r10
    dec rcx
    jnz .Llitpart
    jmp .Ltoobig
.Llitcopy:
    mov al, [r8]
    mov [r10], al
    inc r8
    inc r10
    dec rcx
    jnz .Llitcopy
    jmp .Lloop
.Lnop:
    inc r8
    jmp .Lloop
.Ldone:
    mov rax, r10
    sub rax, rdi
    ret
.Lbadsrc:
    mov rax, -2
    ret
.Ltoobig:
    mov rax, -1
    ret
