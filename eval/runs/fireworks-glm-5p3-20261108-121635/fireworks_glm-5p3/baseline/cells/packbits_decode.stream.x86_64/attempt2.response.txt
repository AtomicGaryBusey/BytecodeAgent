packbits_decode:
    mov r8, rdi                     # p = src cursor
    lea r9, [rdi + rsi]             # send = src + n
    mov r10, rdx                    # q = out cursor
    lea r11, [rdx + rcx]            # oend = out + cap
    mov rdi, rdx                    # remember out base for the final count
.Lloop:
    cmp r8, r9
    jae .Ldone
    movzx eax, byte ptr [r8]        # b = header byte, read unsigned
    cmp eax, 128
    je .Lnop                        # b == 128 (h == -128): no operation
    ja .Lrep                        # b in 129..255 (h in -127..-1): repeat
    inc r8                          # literal: r8 = payload start
    lea ecx, [rax + 1]              # ecx = b + 1 = length, 1..128
    lea rax, [r8 + rcx]             # rax = end of payload
    cmp rax, r9
    ja .Lbadsrc                     # payload would run past src + n
    mov rdx, r11
    sub rdx, r10                    # rdx = space left in out
    cmp rcx, rdx
    jbe .Llitcopy                   # whole literal fits
    mov rcx, rdx                    # copy only the part that fits
    test rcx, rcx
    jz .Ltoobig                     # cap already reached
.Llitpart:
    mov al, [r8]
    mov [r10], al
    inc r8
    inc r10
    dec rcx
    jnz .Llitpart
    jmp .Ltoobig
.Llitcopy:
    mov al, [r8]
    mov [r10], al
    inc r8
    inc r10
    dec rcx
    jnz .Llitcopy
    jmp .Lloop
.Lrep:
    lea rdx, [r8 + 2]               # packet needs header + value byte
    cmp rdx, r9
    ja .Lbadsrc                     # value byte would run past src + n
    movzx esi, byte ptr [r8 + 1]    # value to repeat
    mov edx, 257
    sub edx, eax                    # run length = 257 - b, 2..128
    mov rax, r11
    sub rax, r10                    # rax = space left in out
    cmp rdx, rax
    jbe .Lrepfull                   # whole run fits
    mov rcx, rax                    # write only the part that fits
    test rcx, rcx
    jz .Ltoobig                     # cap already reached
.Lreppart:
    mov [r10], sil
    inc r10
    dec rcx
    jnz .Lreppart
    jmp .Ltoobig
.Lrepfull:
    mov rcx, rdx
.Lrepfill:
    mov [r10], sil
    inc r10
    dec rcx
    jnz .Lrepfill
    add r8, 2
    jmp .Lloop
.Lnop:
    inc r8
    jmp .Lloop
.Ldone:
    mov rax, r10
    sub rax, rdi                    # bytes written
    ret
.Lbadsrc:
    mov rax, -2
    ret
.Ltoobig:
    mov rax, -1
    ret
