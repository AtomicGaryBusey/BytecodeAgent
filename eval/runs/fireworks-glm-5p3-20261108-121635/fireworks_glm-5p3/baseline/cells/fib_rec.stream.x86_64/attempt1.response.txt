fib_rec:
    push rbx
    push r12
    sub rsp, 8
    mov rbx, rdi
    cmp rbx, 2
    jl .Lbase
    lea rdi, [rbx - 1]
    call fib_rec
    mov r12, rax
    lea rdi, [rbx - 2]
    call fib_rec
    add rax, r12
    add rsp, 8
    pop r12
    pop rbx
    ret
.Lbase:
    mov rax, rbx
    add rsp, 8
    pop r12
    pop rbx
    ret
