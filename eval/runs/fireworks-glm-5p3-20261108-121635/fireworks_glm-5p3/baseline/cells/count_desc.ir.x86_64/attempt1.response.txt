export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%c0 =l copy 0
	%i0 =l copy 0
	jmp @loop
@loop
	%cnt =l phi @start %c0, @body %cnt1
	%i =l phi @start %i0, @body %i1
	%cont =w csltl %i, %lim
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%pa =l add %a, %off
	%v0 =l loadl %pa
	%pb =l add %pa, 8
	%v1 =l loadl %pb
	%dw =w csltl %v1, %v0
	%dl =l extsw %dw
	%cnt1 =l add %cnt, %dl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
