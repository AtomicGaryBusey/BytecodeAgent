map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r13, rdi
    mov rbp, rsi
    mov r14, rcx
    xor r15d, r15d
    lea r12, [rbp + rdx*8]
    jmp .Lcheck
.Lloop:
    mov rbx, qword ptr [rbp]
    mov rdi, rbx
    call r14
    mov qword ptr [r13], rax
    cmp rax, rbx
    setl al
    movzx eax, al
    add r15, rax
    add rbp, 8
    add r13, 8
.Lcheck:
    cmp rbp, r12
    jb .Lloop
    mov rax, r15
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
