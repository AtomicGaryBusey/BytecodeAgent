map_sum:
    test rsi, rsi
    je .Lret0
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r15, rdx
    mov r13, rdi
    lea r14, [rdi + rsi*8]
    xor ebx, ebx
.Lloop:
    mov rdi, qword ptr [r13]
    add r13, 8
    call r15
    add rbx, rax
    cmp r13, r14
    jne .Lloop
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lret0:
    xor eax, eax
    ret
