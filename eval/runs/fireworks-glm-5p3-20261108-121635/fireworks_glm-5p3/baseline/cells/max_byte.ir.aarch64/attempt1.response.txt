export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%c0 =w csltl %i, %n
	jnz %c0, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%lt =w csltl %max, %b
	jnz %lt, @update, @next
@update
	%max =l copy %b
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
