binary_search:
    xor rax, rax            # lo = 0
    mov rcx, rsi            # hi = n
.Lloop:
    cmp rax, rcx
    jge .Lnotfound
    lea r8, [rax + rcx]
    shr r8, 1               # mid = (lo + hi) / 2
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .Lfound
    jl .Lless
    mov rcx, r8             # a[mid] > key: hi = mid
    jmp .Lloop
.Lless:
    lea rax, [r8 + 1]       # a[mid] < key: lo = mid + 1
    jmp .Lloop
.Lfound:
    mov rax, r8
    ret
.Lnotfound:
    mov rax, -1
    ret
