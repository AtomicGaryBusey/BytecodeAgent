mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    mov rbx, rsi              # n
    mov rbp, rdx              # k
    mov r12, rcx              # out
    mov r13, r8               # tmp
    mov r14, rdi              # a
    # out = identity matrix
    mov rdi, r12
    xor r10d, r10d            # i
.Lid_i:
    xor r11d, r11d            # j
.Lid_j:
    xor eax, eax
    cmp r10, r11
    jne .Lid_st
    mov eax, 1
.Lid_st:
    mov qword ptr [rdi + r11*8], rax
    inc r11
    cmp r11, rbx
    jl .Lid_j
    lea rdi, [rdi + rbx*8]
    inc r10
    cmp r10, rbx
    jl .Lid_i
    # repeat k times: tmp = out * a ; out = tmp
    test rbp, rbp
    jz .Lfin
.Lmain:
    mov r9, r12               # xp = &out[i][0]
    mov rdi, r13              # dp = &tmp[i][0]
    xor r10d, r10d            # i
.Lm_i:
    xor r11d, r11d            # j
.Lm_j:
    lea rdx, [r14 + r11*8]    # &a[0][j]
    xor eax, eax              # acc
    xor ecx, ecx              # t
    xor r8d, r8d              # t*n
.Lm_t:
    mov rsi, qword ptr [r9 + rcx*8]
    imul rsi, qword ptr [rdx + r8*8]
    add rax, rsi
    add r8, rbx
    inc rcx
    cmp rcx, rbx
    jl .Lm_t
    mov qword ptr [rdi + r11*8], rax
    inc r11
    cmp r11, rbx
    jl .Lm_j
    lea r9, [r9 + rbx*8]
    lea rdi, [rdi + rbx*8]
    inc r10
    cmp r10, rbx
    jl .Lm_i
    # copy n*n elements tmp -> out
    mov rdx, rbx
    imul rdx, rbx
    xor ecx, ecx
.Lcopy:
    mov rax, qword ptr [r13 + rcx*8]
    mov qword ptr [r12 + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lcopy
    dec rbp
    jnz .Lmain
.Lfin:
    # zero all 2*n*n tmp elements
    mov rdx, rbx
    imul rdx, rbx
    add rdx, rdx
    xor ecx, ecx
.Lzero:
    mov qword ptr [r13 + rcx*8], 0
    inc rcx
    cmp rcx, rdx
    jl .Lzero
    xor eax, eax
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
