export function l $max_byte(l %p, l %n) {
@start
	%m =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%b =l loadub %p
	%lt =w csltl %m, %b
	jnz %lt, @update, @next
@update
	%m =l copy %b
@next
	%i =l add %i, 1
	%p =l add %p, 1
	jmp @loop
@done
	ret %m
}
