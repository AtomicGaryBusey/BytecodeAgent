    # bubble_sort(int64_t *a /*rdi*/, int64_t n /*rsi*/) -> 0
bubble_sort:
    xor eax, eax
    cmp rsi, 2
    jl .Lret
    mov rdx, rsi                # rdx = n-1 = inner count for first pass
    dec rdx
.Louter:
    xor r8d, r8d                # swapped = 0
    mov rcx, rdx                # comparisons this pass
    mov r10, rdi                # r10 = &a[j]
.Linner:
    mov rax, qword ptr [r10]
    mov r9, qword ptr [r10 + 8]
    cmp rax, r9
    jle .Lnoswap
    mov qword ptr [r10], r9
    mov qword ptr [r10 + 8], rax
    mov r8d, 1                  # swapped = 1
.Lnoswap:
    add r10, 8
    dec rcx
    jnz .Linner
    test r8d, r8d
    jz .Lret
    dec rdx
    jnz .Louter
.Lret:
    xor eax, eax
    ret
