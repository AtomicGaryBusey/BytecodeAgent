export function l $isqrt(l %n) {
@start
	%lo0 =l copy 0
	%hi0 =l copy 4294967296
	jmp @loop
@loop
	%lo =l phi @start %lo0, @le %mid, @gt %loc
	%hi =l phi @start %hi0, @le %hic, @gt %mid
	%d =l sub %hi, %lo
	%fin =w culel %d, 1
	jnz %fin, @end, @body
@body
	%s =l add %lo, %hi
	%mid =l udiv %s, 2
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @le, @gt
@le
	%hic =l copy %hi
	jmp @loop
@gt
	%loc =l copy %lo
	jmp @loop
@end
	ret %lo
}
