export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%cnt =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%da =l add %dst, %off
	%x =l loadl %sa
	%r =l call %f(l %x)
	storel %r, %da
	%lt =w csltl %r, %x
	%ltl =l extsw %lt
	%cnt =l add %cnt, %ltl
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
