sdivrem:
    cbz x1, .Lzero_divisor          // b == 0: q = 0, r = a
    cmn x1, #1
    b.ne .Lgeneral                  // b != -1
    mov x9, #1
    lsl x9, x9, #63                 // x9 = INT64_MIN
    cmp x0, x9
    b.eq .Loverflow                 // a == INT64_MIN && b == -1
.Lgeneral:
    sdiv x3, x0, x1                 // q = a / b, truncated toward zero
    msub x4, x3, x1, x0             // r = a - q * b, sign of a
    str x4, [x2]                    // *rem = r
    mov x0, x3                      // return q
    ret
.Loverflow:
    str xzr, [x2]                   // *rem = 0
    mov x0, x9                      // q = INT64_MIN
    ret
.Lzero_divisor:
    str x0, [x2]                    // *rem = a
    mov x0, xzr                     // q = 0
    ret
