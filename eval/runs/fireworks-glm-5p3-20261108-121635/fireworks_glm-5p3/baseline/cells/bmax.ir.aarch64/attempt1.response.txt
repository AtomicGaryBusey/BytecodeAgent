export function l $bmax(l %a, l %n) {
@start
	%m =l loadl %a
	%p =l add %a, 8
	%i =l copy 1
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%v =l loadl %p
	%lt =w csltl %m, %v
	jnz %lt, @new, @next
@new
	%m =l copy %v
	jmp @next
@next
	%p =l add %p, 8
	%i =l add %i, 1
	jmp @loop
@done
	ret %m
}
