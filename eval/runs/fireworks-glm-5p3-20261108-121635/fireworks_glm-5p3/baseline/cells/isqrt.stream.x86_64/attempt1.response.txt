isqrt:
    xor r8d, r8d              # lo = 0
    mov r9d, 0xFFFFFFFF       # hi = 2^32 - 1  (zero-extended)
.Lloop:
    cmp r8, r9
    jae .Ldone                # lo >= hi -> answer found
    lea r10, [r8 + r9 + 1]    # lo + hi + 1  (fits: <= 2^33 - 1)
    shr r10, 1                # mid = (lo + hi + 1) / 2, mid <= 2^32 - 1
    mov rax, r10
    mul rax                   # rdx:rax = mid*mid, fits in 64 bits (rdx = 0)
    cmp rax, rdi
    ja .Lbig                  # mid*mid > n -> hi = mid - 1
    mov r8, r10               # lo = mid
    jmp .Lloop
.Lbig:
    lea r9, [r10 - 1]         # hi = mid - 1
    jmp .Lloop
.Ldone:
    mov rax, r8
    ret
