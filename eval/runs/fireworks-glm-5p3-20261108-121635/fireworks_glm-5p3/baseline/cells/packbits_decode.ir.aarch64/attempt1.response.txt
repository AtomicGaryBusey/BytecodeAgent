export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%endp =l add %src, %n
	%spslot =l alloc8 8
	%wslot =l alloc8 8
	storel %src, %spslot
	%zero =l copy 0
	storel %zero, %wslot
	%one =l copy 1
	%two =l copy 2
	%k127 =w copy 127
	%k128 =w copy 128
	%k257 =l copy 257
	%neg1 =l sub %zero, %one
	%neg2 =l sub %zero, %two
	jmp @loop
@loop
	%sp1 =l loadl %spslot
	%cont =w cultl %sp1, %endp
	jnz %cont, @hdr, @done
@hdr
	%hu =w loadub %sp1
	%sp2 =l add %sp1, %one
	storel %sp2, %spslot
	%clit =w culew %hu, %k127
	jnz %clit, @lit, @chk
@chk
	%cnop =w ceqw %hu, %k128
	jnz %cnop, @loop, @rep
@rep
	%have =w cultl %sp2, %endp
	jnz %have, @rep2, @e2
@rep2
	%rbyte =w loadub %sp2
	%sp3 =l add %sp2, %one
	storel %sp3, %spslot
	%hul =l extuw %hu
	%rcnt =l sub %k257, %hul
	%rw0 =l loadl %wslot
	%rtgt =l add %rw0, %rcnt
	jmp @rloop
@rloop
	%rw1 =l loadl %wslot
	%rmore =w csltl %rw1, %rtgt
	jnz %rmore, @rbody, @loop
@rbody
	%rroom =w csltl %rw1, %cap
	jnz %rroom, @rstore, @e1
@rstore
	%rptr =l add %out, %rw1
	storeb %rbyte, %rptr
	%rw2 =l add %rw1, %one
	storel %rw2, %wslot
	jmp @rloop
@lit
	%hul2 =l extuw %hu
	%lcnt =l add %hul2, %one
	%lend =l add %sp2, %lcnt
	%lok =w culel %lend, %endp
	jnz %lok, @lloop, @e2
@lloop
	%sp4 =l loadl %spslot
	%lmore =w cultl %sp4, %lend
	jnz %lmore, @lbody, @loop
@lbody
	%lw3 =l loadl %wslot
	%lroom =w csltl %lw3, %cap
	jnz %lroom, @lstore, @e1
@lstore
	%lbyte =w loadub %sp4
	%lptr =l add %out, %lw3
	storeb %lbyte, %lptr
	%lw4 =l add %lw3, %one
	storel %lw4, %wslot
	%sp5 =l add %sp4, %one
	storel %sp5, %spslot
	jmp @lloop
@e1
	ret %neg1
@e2
	ret %neg2
@done
	%wfin =l loadl %wslot
	ret %wfin
}
