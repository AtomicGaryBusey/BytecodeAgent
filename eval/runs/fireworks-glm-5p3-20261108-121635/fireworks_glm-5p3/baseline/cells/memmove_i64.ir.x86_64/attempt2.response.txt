export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%isz =w ceql %n, %zero
	jnz %isz, @done, @cmp
@cmp
	%lt =w cultl %dst, %src
	jnz %lt, @fwdinit, @cmpeq
@cmpeq
	%eq =w ceql %dst, %src
	jnz %eq, @done, @bwdinit
@fwdinit
	%i0 =l copy 0
	jmp @fwdloop
@fwdloop
	%i =l phi @fwdinit %i0, @fwdbody %i1
	%cont =w csltl %i, %n
	jnz %cont, @fwdbody, @done
@fwdbody
	%off =l mul %i, 8
	%sa =l add %src, %off
	%da =l add %dst, %off
	%val =l loadl %sa
	storel %val, %da
	%i1 =l add %i, 1
	jmp @fwdloop
@bwdinit
	%j0 =l sub %n, 1
	jmp @bwdloop
@bwdloop
	%j =l phi @bwdinit %j0, @bwdbody %j1
	%stop =w csltl %j, %zero
	jnz %stop, @done, @bwdbody
@bwdbody
	%off2 =l mul %j, 8
	%sb =l add %src, %off2
	%db =l add %dst, %off2
	%val2 =l loadl %sb
	storel %val2, %db
	%j1 =l sub %j, 1
	jmp @bwdloop
@done
	ret %zero
}
