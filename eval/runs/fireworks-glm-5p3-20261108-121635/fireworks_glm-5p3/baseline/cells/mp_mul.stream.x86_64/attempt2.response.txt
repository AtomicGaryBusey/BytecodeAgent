mp_mul:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rsi
    mov r12, rcx
    mov r13, r8
    mov r14, rdi
    mov r15, rdx
    lea rdi, [rbx + r12]
    xor eax, eax
    xor r8d, r8d
.Lzero:
    cmp rax, rdi
    jae .Louter
    mov qword ptr [r13 + rax*8], 0
    inc rax
    jmp .Lzero
.Louter:
    cmp r8, rbx
    jae .Lcount
    xor r11d, r11d
    xor ecx, ecx
.Linner:
    cmp rcx, r12
    jae .Linner_end
    mov rax, qword ptr [r14 + r8*8]
    mul qword ptr [r15 + rcx*8]
    add rax, r11
    adc rdx, 0
    lea rsi, [r8 + rcx]
    add rax, qword ptr [r13 + rsi*8]
    adc rdx, 0
    mov qword ptr [r13 + rsi*8], rax
    mov r11, rdx
    inc rcx
    jmp .Linner
.Linner_end:
    lea rsi, [r8 + r12]
    mov qword ptr [r13 + rsi*8], r11
    inc r8
    jmp .Louter
.Lcount:
    mov rax, rdi
.Lcount_loop:
    test rax, rax
    jz .Ldone
    cmp qword ptr [r13 + rax*8 - 8], 0
    jne .Ldone
    dec rax
    jmp .Lcount_loop
.Ldone:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
