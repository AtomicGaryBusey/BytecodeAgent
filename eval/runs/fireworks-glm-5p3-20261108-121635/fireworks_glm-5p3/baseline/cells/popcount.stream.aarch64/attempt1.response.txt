popcount:
    movz x4, #0x0101
    movk x4, #0x0101, lsl #16
    movk x4, #0x0101, lsl #32
    movk x4, #0x0101, lsl #48     // x4 = 0x0101010101010101
    movz x1, #0x55
    mul x1, x4, x1                // x1 = 0x5555555555555555
    movz x2, #0x33
    mul x2, x4, x2                // x2 = 0x3333333333333333
    movz x3, #0x0f
    mul x3, x4, x3                // x3 = 0x0f0f0f0f0f0f0f0f
    lsr x5, x0, #1
    and x5, x5, x1
    sub x0, x0, x5                // pairs of bits
    lsr x5, x0, #2
    and x5, x5, x2
    and x6, x0, x2
    add x0, x5, x6                // nibbles
    lsr x5, x0, #4
    add x0, x0, x5
    and x0, x0, x3                // counts per byte
    mul x0, x0, x4                // sum bytes into top byte
    lsr x0, x0, #56
    ret
