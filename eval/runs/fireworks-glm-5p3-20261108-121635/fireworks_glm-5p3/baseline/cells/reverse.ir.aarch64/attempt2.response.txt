export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	jmp @loop
@loop
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @done
@body
	%i8 =l mul %i, 8
	%j8 =l mul %j, 8
	%ai =l add %a, %i8
	%aj =l add %a, %j8
	%vi =l loadl %ai
	%vj =l loadl %aj
	storel %vj, %ai
	storel %vi, %aj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	%r =l copy 0
	ret %r
}
