	export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
		%sum =l copy 0
		%cnt =l copy 0
		%i =l copy 0
		%min =l copy 0
		%max =l copy 0
		jmp @loop
@loop
		%cont =w csltl %i, %n
		jnz %cont, @body, @done
@body
		%off =l mul %i, 8
		%pa =l add %a, %off
		%av =l loadl %pa
		%y =l call %f(l %av)
		%sum =l add %sum, %y
		%oddl =l and %y, 1
		%cnt =l add %cnt, %oddl
		%first =w ceql %i, 0
		jnz %first, @init, @upd
@init
		%min =l copy %y
		%max =l copy %y
		jmp @next
@upd
		%ltmin =w csltl %y, %min
		jnz %ltmin, @setmin, @chkmax
@setmin
		%min =l copy %y
		jmp @chkmax
@chkmax
		%gtmax =w csgtl %y, %max
		jnz %gtmax, @setmax, @next
@setmax
		%max =l copy %y
		jmp @next
@next
		%i =l add %i, 1
		jmp @loop
@done
		storel %min, %out
		%out1 =l add %out, 8
		storel %max, %out1
		%out2 =l add %out, 16
		storel %sum, %out2
		ret %cnt
	}
