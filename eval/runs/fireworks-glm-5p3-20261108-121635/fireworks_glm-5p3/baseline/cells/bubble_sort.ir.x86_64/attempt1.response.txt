	export function l $bubble_sort(l %a, l %n) {
	@start
		%nm1 =l sub %n, 1
		%i =l copy 0
		jmp @outer
	@outer
		%oc =w csltl %i, %nm1
		jnz %oc, @outer_body, @end
	@outer_body
		%lim =l sub %nm1, %i
		%j =l copy 0
		jmp @inner
	@inner
		%ic =w csltl %j, %lim
		jnz %ic, @inner_body, @inner_end
	@inner_body
		%off =l mul %j, 8
		%pj =l add %a, %off
		%pj1 =l add %pj, 8
		%x =l loadl %pj
		%y =l loadl %pj1
		%sw =w csltl %y, %x
		jnz %sw, @swap, @next
	@swap
		storel %y, %pj
		storel %x, %pj1
		jmp @next
	@next
		%j =l add %j, 1
		jmp @inner
	@inner_end
		%i =l add %i, 1
		jmp @outer
	@end
		ret 0
	}
