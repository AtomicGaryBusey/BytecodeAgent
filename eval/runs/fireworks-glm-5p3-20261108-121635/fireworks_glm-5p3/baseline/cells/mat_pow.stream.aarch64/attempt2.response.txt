mat_pow:
    mov x5, x1                    // x5 = n
    mul x6, x1, x1                // x6 = n*n
    mov x7, x3                    // x7 = out
    mov x8, x4                    // x8 = X = tmp
    add x9, x4, x6, lsl #3        // x9 = Y = tmp + n*n*8
    mov x10, x2                   // x10 = k
    // out = zero matrix
    mov x12, x7
    mov x13, x6
    mov x14, #0
.Lzero_out:
    str x14, [x12], #8
    subs x13, x13, #1
    b.ne .Lzero_out
    // out = identity matrix
    add x12, x5, #1               // stride = n+1
    mov x13, #0
    mov x14, x5
    mov x15, #1
.Ldiag:
    str x15, [x7, x13, lsl #3]
    add x13, x13, x12
    subs x14, x14, #1
    b.ne .Ldiag
    cbz x10, .Lzero_tmp
    // X = copy of a
    mov x12, x0
    mov x13, x8
    mov x14, x6
.Lcopy:
    ldr x15, [x12], #8
    str x15, [x13], #8
    subs x14, x14, #1
    b.ne .Lcopy
.Lkloop:
    // Y = out * X
    mov x3, #0                    // i = 0
.Lmi:
    mul x12, x3, x5               // i*n
    add x13, x9, x12, lsl #3      // &Y[i*n]
    mov x4, #0                    // j = 0
.Lmj:
    add x16, x7, x12, lsl #3      // &out[i*n]
    add x17, x8, x4, lsl #3       // &X[j]
    mov x14, #0                   // accumulator
    mov x15, x5                   // t counter
.Lmt:
    ldr x0, [x16], #8             // out[i*n+t]
    ldr x1, [x17]                 // X[t*n+j]
    add x17, x17, x5, lsl #3
    mul x0, x0, x1
    add x14, x14, x0
    subs x15, x15, #1
    b.ne .Lmt
    str x14, [x13, x4, lsl #3]    // Y[i*n+j]
    add x4, x4, #1
    cmp x4, x5
    b.lt .Lmj
    add x3, x3, #1
    cmp x3, x5
    b.lt .Lmi
    // out = Y
    mov x12, x9
    mov x13, x7
    mov x14, x6
.Lcopy_out:
    ldr x15, [x12], #8
    str x15, [x13], #8
    subs x14, x14, #1
    b.ne .Lcopy_out
    subs x10, x10, #1
    b.ne .Lkloop
.Lzero_tmp:
    mov x12, x8
    lsl x13, x6, #1               // 2*n*n
    mov x14, #0
.Lztmp:
    str x14, [x12], #8
    subs x13, x13, #1
    b.ne .Lztmp
    mov x0, #0
    ret
