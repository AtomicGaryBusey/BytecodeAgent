export function l $base64_encode(l %src, l %n, l %out) {
@start
	%n2 =l add %n, 2
	%ng =l div %n2, 3
	%g =l copy 0
	jmp @grouploop
@grouploop
	%cont =w csltl %g, %ng
	jnz %cont, @groupbody, @done
@groupbody
	%base =l mul %g, 3
	%e3 =l add %base, 3
	%isfull =w cslel %e3, %n
	jnz %isfull, @fullgroup, @chk2
@chk2
	%e2 =l add %base, 2
	%is2 =w cslel %e2, %n
	jnz %is2, @group2, @group1
@fullgroup
	%p0 =l add %src, %base
	%b0 =w loadub %p0
	%p1 =l add %p0, 1
	%b1 =w loadub %p1
	%p2 =l add %p0, 2
	%b2 =w loadub %p2
	%s0 =w shl %b0, 16
	%s1 =w shl %b1, 8
	%v0 =w or %s0, %s1
	%v =w or %v0, %b2
	%k =w copy 4
	jmp @emit
@group2
	%p0 =l add %src, %base
	%b0 =w loadub %p0
	%p1 =l add %p0, 1
	%b1 =w loadub %p1
	%s0 =w shl %b0, 16
	%s1 =w shl %b1, 8
	%v =w or %s0, %s1
	%k =w copy 3
	jmp @emit
@group1
	%p0 =l add %src, %base
	%b0 =w loadub %p0
	%v =w shl %b0, 16
	%k =w copy 2
	jmp @emit
@emit
	%g4 =l mul %g, 4
	%outp =l add %out, %g4
	%j =w copy 0
	jmp @charloop
@charloop
	%isdata =w csltw %j, %k
	jnz %isdata, @datachar, @padchar
@datachar
	%sixj =w mul %j, 6
	%sh =w sub 18, %sixj
	%tmp =w shr %v, %sh
	%idx =w and %tmp, 63
	%u =w csltw %idx, 26
	%ge26 =w csgew %idx, 26
	%lt52 =w csltw %idx, 52
	%m =w and %ge26, %lt52
	%ge52 =w csgew %idx, 52
	%lt62 =w csltw %idx, 62
	%d =w and %ge52, %lt62
	%q62 =w ceqw %idx, 62
	%q63 =w ceqw %idx, 63
	%r1 =w add %idx, 65
	%c1 =w mul %u, %r1
	%r2 =w add %idx, 71
	%c2 =w mul %m, %r2
	%r3 =w sub %idx, 4
	%c3 =w mul %d, %r3
	%c4 =w mul %q62, 43
	%c5 =w mul %q63, 47
	%c6 =w add %c1, %c2
	%c7 =w add %c6, %c3
	%c8 =w add %c7, %c4
	%ch =w add %c8, %c5
	jmp @storech
@padchar
	%ch =w copy 61
	jmp @storech
@storech
	%jl =l extuw %j
	%dst =l add %outp, %jl
	storeb %ch, %dst
	%jn =w add %j, 1
	%j =w copy %jn
	%more =w csltw %j, 4
	jnz %more, @charloop, @groupnext
@groupnext
	%gn =l add %g, 1
	%g =l copy %gn
	jmp @grouploop
@done
	%res =l mul %ng, 4
	ret %res
}
