export function l $mulhi_u64(l %a, l %b) {
@start
	%al  =l and %a, 4294967295
	%ah  =l shr %a, 32
	%bl  =l and %b, 4294967295
	%bh  =l shr %b, 32
	%p0  =l mul %al, %bl
	%p1  =l mul %al, %bh
	%p2  =l mul %ah, %bl
	%p3  =l mul %ah, %bh
	%p1h =l shr %p1, 32
	%p1l =l and %p1, 4294967295
	%p2h =l shr %p2, 32
	%p2l =l and %p2, 4294967295
	%p0h =l shr %p0, 32
	%m   =l add %p1l, %p2l
	%t   =l add %m, %p0h
	%carry =l shr %t, 32
	%s1  =l add %p3, %p1h
	%s2  =l add %s1, %p2h
	%res =l add %s2, %carry
	ret %res
}
