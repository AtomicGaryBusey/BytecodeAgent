count_ugt:
    mov     x3, #0                  // running count
    cmp     x1, #0
    b.le    .Ldone
    lsr     x4, x1, #2              // number of 4-element blocks
    and     x5, x1, #3              // leftover elements
    dup     v0.2d, x2               // t broadcast to both lanes
    movi    v3.16b, #0
    movi    v4.16b, #0
    cbz     x4, .Ltail
.Lloop:
    ld1     {v1.2d, v2.2d}, [x0], #32
    cmhi    v5.2d, v1.2d, v0.2d     // all-ones where a[i] > t (unsigned)
    cmhi    v6.2d, v2.2d, v0.2d
    add     v3.2d, v3.2d, v5.2d     // accumulate -1 per hit
    add     v4.2d, v4.2d, v6.2d
    subs    x4, x4, #1
    b.ne    .Lloop
    add     v3.2d, v3.2d, v4.2d
    addp    v3.2d, v3.2d, v3.2d
    umov    x6, v3.d[0]
    neg     x6, x6                  // hits in vector part
    add     x3, x3, x6
.Ltail:
    cbz     x5, .Ldone
.Ltail_loop:
    ldr     x7, [x0], #8
    cmp     x7, x2
    cinc    x3, x3, hi
    subs    x5, x5, #1
    b.ne    .Ltail_loop
.Ldone:
    mov     x0, x3
    ret
