export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%cnt =l phi @start %c0, @body %c1
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%g =w cugtl %v, %t
	%gl =l extuw %g
	%c1 =l add %cnt, %gl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
