export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz = w ceql %b, 0
	jnz %bz, @divzero, @checkmin
@divzero
	%q0 = l copy 0
	storel %a, %rem
	ret %q0
@checkmin
	%bp1 = l add %b, 1
	%bneg1 = w ceql %bp1, 0
	%aa = l add %a, %a
	%awrap = w ceql %aa, 0
	%anot0 = w cnel %a, 0
	%ismin = w and %awrap, %anot0
	%special = w and %ismin, %bneg1
	jnz %special, @mincase, @normal
@mincase
	%z = l copy 0
	storel %z, %rem
	ret %a
@normal
	%q = l div %a, %b
	%r = l rem %a, %b
	storel %r, %rem
	ret %q
}
