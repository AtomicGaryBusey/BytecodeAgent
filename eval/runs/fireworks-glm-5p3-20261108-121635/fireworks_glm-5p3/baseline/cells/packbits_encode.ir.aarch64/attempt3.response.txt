export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%w =l copy 0
	jmp @main
@main
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@body
	%pi =l add %src, %i
	%base =w loadub %pi
	%r =l copy 1
	jmp @runloop
@runloop
	%c1 =w csltl %r, 128
	jnz %c1, @run1, @runend
@run1
	%ipr =l add %i, %r
	%c2 =w csltl %ipr, %n
	jnz %c2, @run2, @runend
@run2
	%pr =l add %src, %ipr
	%br =w loadub %pr
	%c3 =w ceqw %br, %base
	jnz %c3, @run3, @runend
@run3
	%r =l add %r, 1
	jmp @runloop
@runend
	%c4 =w csltl %r, 3
	jnz %c4, @literal, @repeat
@repeat
	%w2 =l add %w, 2
	%c5 =w cslel %w2, %cap
	jnz %c5, @repok, @fail
@repok
	%ow =l add %out, %w
	%rw =w copy %r
	%c257 =w copy 257
	%hdr =w sub %c257, %rw
	storeb %hdr, %ow
	%ow1 =l add %ow, 1
	storeb %base, %ow1
	%w =l copy %w2
	%i =l add %i, %r
	jmp @main
@fail
	%zero =l copy 0
	%one =l copy 1
	%res =l sub %zero, %one
	ret %res
@done
	ret %w
@literal
	%j =l copy %i
	jmp @litloop
@litloop
	%c6 =w csltl %j, %n
	jnz %c6, @lit1, @litdone
@lit1
	%d =l sub %j, %i
	%c7 =w csltl %d, 128
	jnz %c7, @lit2, @litdone
@lit2
	%j1 =l add %j, 1
	%c8 =w csltl %j1, %n
	jnz %c8, @lit3, @litadv
@lit3
	%j2 =l add %j, 2
	%c9 =w csltl %j2, %n
	jnz %c9, @lit4, @litadv
@lit4
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %src, %j1
	%bj1 =w loadub %pj1
	%ca =w ceqw %bj, %bj1
	jnz %ca, @lit5, @litadv
@lit5
	%pj2 =l add %src, %j2
	%bj2 =w loadub %pj2
	%cb =w ceqw %bj1, %bj2
	jnz %cb, @litdone, @litadv
@litadv
	%j =l add %j, 1
	jmp @litloop
@litdone
	%len =l sub %j, %i
	%len1 =l add %len, 1
	%need =l add %w, %len1
	%cc =w cslel %need, %cap
	jnz %cc, @litok, @fail
@litok
	%ow2 =l add %out, %w
	%lenw =w copy %len
	%lenm1 =w sub %lenw, 1
	storeb %lenm1, %ow2
	%k =l copy 0
	jmp @copyloop
@copyloop
	%cd =w csltl %k, %len
	jnz %cd, @copybody, @copydone
@copybody
	%ik =l add %i, %k
	%ps =l add %src, %ik
	%bk =w loadub %ps
	%wk =l add %w, %k
	%wk1 =l add %wk, 1
	%pd =l add %out, %wk1
	storeb %bk, %pd
	%k =l add %k, 1
	jmp @copyloop
@copydone
	%w =l add %w, %need
	%i =l copy %j
	jmp @main
}
