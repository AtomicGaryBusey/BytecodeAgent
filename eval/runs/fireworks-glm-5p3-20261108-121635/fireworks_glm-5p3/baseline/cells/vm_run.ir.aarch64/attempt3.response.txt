export function l $vm_run(l %code, l %len, l %stk, l %cap) {
@entry
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%run =w csltl %pc, %len
	jnz %run, @dispatch, @finish
@dispatch
	%opat =l mul %pc, 8
	%opad =l add %code, %opat
	%op =l loadl %opad
	%t0 =w ceql %op, 0
	jnz %t0, @finish, @q1
@q1
	%t1 =w ceql %op, 1
	jnz %t1, @op_push, @q2
@q2
	%t2 =w ceql %op, 2
	jnz %t2, @op_add, @q3
@q3
	%t3 =w ceql %op, 3
	jnz %t3, @op_sub, @q4
@q4
	%t4 =w ceql %op, 4
	jnz %t4, @op_mul, @q5
@q5
	%t5 =w ceql %op, 5
	jnz %t5, @op_dup, @q6
@q6
	%t6 =w ceql %op, 6
	jnz %t6, @op_swap, @q7
@q7
	%t7 =w ceql %op, 7
	jnz %t7, @op_drop, @q8
@q8
	%t8 =w ceql %op, 8
	jnz %t8, @op_over, @q9
@q9
	%t9 =w ceql %op, 9
	jnz %t9, @op_lt, @q10
@q10
	%t10 =w ceql %op, 10
	jnz %t10, @op_jz, @q11
@q11
	%t11 =w ceql %op, 11
	jnz %t11, @op_jmp, @op_other
@op_push
	%pu1 =l add %pc, 1
	%pu2 =l mul %pu1, 8
	%pu3 =l add %code, %pu2
	%pu4 =l loadl %pu3
	%pu5 =l mul %d, 8
	%pu6 =l add %stk, %pu5
	storel %pu4, %pu6
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@op_add
	%aa1 =l sub %d, 1
	%aa2 =l mul %aa1, 8
	%aa3 =l add %stk, %aa2
	%aav2 =l loadl %aa3
	%aa4 =l sub %d, 2
	%aa5 =l mul %aa4, 8
	%aa6 =l add %stk, %aa5
	%aav1 =l loadl %aa6
	%aa7 =l add %aav1, %aav2
	storel %aa7, %aa6
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_sub
	%sb1 =l sub %d, 1
	%sb2 =l mul %sb1, 8
	%sb3 =l add %stk, %sb2
	%sbv2 =l loadl %sb3
	%sb4 =l sub %d, 2
	%sb5 =l mul %sb4, 8
	%sb6 =l add %stk, %sb5
	%sbv1 =l loadl %sb6
	%sb7 =l sub %sbv1, %sbv2
	storel %sb7, %sb6
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_mul
	%mc1 =l sub %d, 1
	%mc2 =l mul %mc1, 8
	%mc3 =l add %stk, %mc2
	%mcv2 =l loadl %mc3
	%mc4 =l sub %d, 2
	%mc5 =l mul %mc4, 8
	%mc6 =l add %stk, %mc5
	%mcv1 =l loadl %mc6
	%mc7 =l mul %mcv1, %mcv2
	storel %mc7, %mc6
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_dup
	%du1 =l sub %d, 1
	%du2 =l mul %du1, 8
	%du3 =l add %stk, %du2
	%du4 =l loadl %du3
	%du5 =l mul %d, 8
	%du6 =l add %stk, %du5
	storel %du4, %du6
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_swap
	%sw1 =l sub %d, 1
	%sw2 =l mul %sw1, 8
	%sw3 =l add %stk, %sw2
	%swv2 =l loadl %sw3
	%sw4 =l sub %d, 2
	%sw5 =l mul %sw4, 8
	%sw6 =l add %stk, %sw5
	%swv1 =l loadl %sw6
	storel %swv1, %sw3
	storel %swv2, %sw6
	%pc =l add %pc, 1
	jmp @loop
@op_drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_over
	%ov1 =l sub %d, 2
	%ov2 =l mul %ov1, 8
	%ov3 =l add %stk, %ov2
	%ov4 =l loadl %ov3
	%ov5 =l mul %d, 8
	%ov6 =l add %stk, %ov5
	storel %ov4, %ov6
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_lt
	%lt1 =l sub %d, 1
	%lt2 =l mul %lt1, 8
	%lt3 =l add %stk, %lt2
	%ltv2 =l loadl %lt3
	%lt4 =l sub %d, 2
	%lt5 =l mul %lt4, 8
	%lt6 =l add %stk, %lt5
	%ltv1 =l loadl %lt6
	%ltc =w csltl %ltv1, %ltv2
	%lt7 =l extsw %ltc
	storel %lt7, %lt6
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@op_jz
	%jz1 =l add %pc, 1
	%jz2 =l mul %jz1, 8
	%jz3 =l add %code, %jz2
	%jzt =l loadl %jz3
	%jz4 =l sub %d, 1
	%jz5 =l mul %jz4, 8
	%jz6 =l add %stk, %jz5
	%jzv =l loadl %jz6
	%d =l sub %d, 1
	%jz7 =w ceql %jzv, 0
	jnz %jz7, @op_jz_yes, @op_jz_no
@op_jz_yes
	%pc =l copy %jzt
	jmp @loop
@op_jz_no
	%pc =l add %pc, 2
	jmp @loop
@op_jmp
	%jm1 =l add %pc, 1
	%jm2 =l mul %jm1, 8
	%jm3 =l add %code, %jm2
	%jmt =l loadl %jm3
	%pc =l copy %jmt
	jmp @loop
@op_other
	%pc =l add %pc, 1
	jmp @loop
@finish
	%zi =l copy %d
	%zv =l copy 0
	jmp @zloop
@zloop
	%zc =w csltl %zi, %cap
	jnz %zc, @zbody, @zdone
@zbody
	%za1 =l mul %zi, 8
	%za2 =l add %stk, %za1
	storel %zv, %za2
	%zi =l add %zi, 1
	jmp @zloop
@zdone
	%de =w ceql %d, 0
	jnz %de, @ret_zero, @ret_top
@ret_zero
	%rz =l copy 0
	ret %rz
@ret_top
	%rt1 =l sub %d, 1
	%rt2 =l mul %rt1, 8
	%rt3 =l add %stk, %rt2
	%rt4 =l loadl %rt3
	ret %rt4
}
