memmove_i64:
    cbz x2, .Lret
    cmp x0, x1
    b.eq .Lret
    b.lo .Lforward
    mov x3, x2
.Lback:
    subs x3, x3, #1
    b.lt .Lret
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    b .Lback
.Lforward:
    mov x3, xzr
.Lfwd:
    cmp x3, x2
    b.hs .Lret
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x3, x3, #1
    b .Lfwd
.Lret:
    mov x0, xzr
    ret
