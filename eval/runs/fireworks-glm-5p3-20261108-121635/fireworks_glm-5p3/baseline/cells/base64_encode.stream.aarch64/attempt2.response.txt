base64_encode:
    stp x29, x30, [sp, #-16]!
    mov x9, x0                    // src cursor
    mov x10, x1                   // bytes remaining
    mov x11, x2                   // out cursor
    mov x12, x2                   // out start
.Lloop3:
    cmp x10, #3
    b.lt .Ltail
    ldrb w13, [x9]                // b0
    ldrb w14, [x9, #1]            // b1
    ldrb w15, [x9, #2]            // b2
    add x9, x9, #3
    sub x10, x10, #3
    lsr w0, w13, #2               // v0
    bl .Ltochar
    strb w0, [x11]
    and w3, w13, #3
    lsl w3, w3, #4
    lsr w4, w14, #4
    orr w0, w3, w4                // v1
    bl .Ltochar
    strb w0, [x11, #1]
    and w3, w14, #0xf
    lsl w3, w3, #2
    lsr w4, w15, #6
    orr w0, w3, w4                // v2
    bl .Ltochar
    strb w0, [x11, #2]
    and w0, w15, #0x3f            // v3
    bl .Ltochar
    strb w0, [x11, #3]
    add x11, x11, #4
    b .Lloop3
.Ltail:
    cbz x10, .Lend
    cmp x10, #1
    b.ne .Ltwo
.Lone:
    ldrb w13, [x9]                // b0
    lsr w0, w13, #2               // v0
    bl .Ltochar
    strb w0, [x11]
    and w3, w13, #3
    lsl w0, w3, #4                // v1
    bl .Ltochar
    strb w0, [x11, #1]
    mov w1, #61                   // '='
    strb w1, [x11, #2]
    strb w1, [x11, #3]
    add x11, x11, #4
    b .Lend
.Ltwo:
    ldrb w13, [x9]                // b0
    ldrb w14, [x9, #1]            // b1
    lsr w0, w13, #2               // v0
    bl .Ltochar
    strb w0, [x11]
    and w3, w13, #3
    lsl w3, w3, #4
    lsr w4, w14, #4
    orr w0, w3, w4                // v1
    bl .Ltochar
    strb w0, [x11, #1]
    and w3, w14, #0xf
    lsl w0, w3, #2                // v2
    bl .Ltochar
    strb w0, [x11, #2]
    mov w1, #61                   // '='
    strb w1, [x11, #3]
    add x11, x11, #4
.Lend:
    sub x0, x11, x12              // number of chars written
    ldp x29, x30, [sp], #16
    ret
.Ltochar:
    cmp w0, #26                   // map 6-bit index in w0 to char
    add w1, w0, #65               // 'A' + v
    add w2, w0, #71               // 'a' + (v - 26)
    csel w1, w2, w1, ge
    cmp w0, #52
    sub w2, w0, #4                // '0' + (v - 52)
    csel w1, w2, w1, ge
    cmp w0, #62
    mov w2, #43                   // '+'
    csel w1, w2, w1, eq
    mov w2, #47                   // '/'
    csel w1, w2, w1, hi
    mov w0, w1
    ret
