export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %i1
	%cnt =l phi @entry 0, @body %cnt1
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%x =l loadl %sa
	%r =l call %f(l %x)
	%da =l add %dst, %off
	storel %r, %da
	%less =w csltl %r, %x
	%inc =l extsw %less
	%cnt1 =l add %cnt, %inc
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
