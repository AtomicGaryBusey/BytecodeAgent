export function l $collatz_max(%n) {
@start
	%best =l copy 0
	%k =l copy 1
@outer
	%c0 =w csgtl %k, %n
	jnz %c0, @done, @body
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%c1 =w ceql %x, 1
	jnz %c1, @cmp, @step
@step
	%r =l and %x, 1
	jnz %r, @odd, @even
@even
	%xe =l div %x, 2
	%x2 =l copy %xe
	jmp @next
@odd
	%m =l mul %x, 3
	%xo =l add %m, 1
	%x2 =l copy %xo
	jmp @next
@next
	%s2 =l add %steps, 1
	%steps =l copy %s2
	%x =l copy %x2
	jmp @inner
@cmp
	%c2 =w csgtl %steps, %best
	jnz %c2, @upd, @cont
@upd
	%best =l copy %steps
	jmp @cont
@cont
	%k2 =l add %k, 1
	%k =l copy %k2
	jmp @outer
@done
	ret %best
}
