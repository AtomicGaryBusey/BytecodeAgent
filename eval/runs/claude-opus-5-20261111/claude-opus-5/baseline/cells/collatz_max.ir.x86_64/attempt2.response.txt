export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%best =l phi @start 0, @cont %bestn
	%k =l phi @start 1, @cont %k2
	%c0 =w csgtl %k, %n
	jnz %c0, @done, @body
@body
	jmp @inner
@inner
	%x =l phi @body %k, @next %x2
	%steps =l phi @body 0, @next %s2
	%c1 =w ceql %x, 1
	jnz %c1, @cmp, @step
@step
	%r =l and %x, 1
	%rw =w cnel %r, 0
	jnz %rw, @odd, @even
@even
	%xe =l shr %x, 1
	jmp @next
@odd
	%m =l mul %x, 3
	%xo =l add %m, 1
	jmp @next
@next
	%x2 =l phi @even %xe, @odd %xo
	%s2 =l add %steps, 1
	jmp @inner
@cmp
	%c2 =w csgtl %steps, %best
	jnz %c2, @upd, @cont
@upd
	jmp @cont
@cont
	%bestn =l phi @upd %steps, @cmp %best
	%k2 =l add %k, 1
	jmp @outer
@done
	ret %best
}
