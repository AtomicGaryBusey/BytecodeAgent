export function l $base64_encode(%src, %n, %out) {
@start
	jmp @gloop
@gloop
	%i =l phi @start 0, @gend %in
	%o =l phi @start 0, @gend %on
	%cg =w csltl %i, %n
	jnz %cg, @group, @done
@group
	%op =l add %out, %o
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%c2 =w csgtl %rem, 1
	jnz %c2, @has1, @no1
@has1
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v1 =l or %v0, %s1
	%c3 =w csgtl %rem, 2
	jnz %c3, @has2, @no2
@has2
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @emit
@no2
	jmp @emit
@no1
	jmp @emit
@emit
	%v =l phi @has2 %v2, @no2 %v1, @no1 %v0
	%nc =l phi @has2 4, @no2 3, @no1 2
	jmp @jloop
@jloop
	%j =l phi @emit 0, @jnext %jn
	%dp =l add %op, %j
	%cj =w csltl %j, %nc
	jnz %cj, @idx, @pad
@idx
	%sh =l mul %j, 6
	%sh2 =l sub 18, %sh
	%vs =l shr %v, %sh2
	%ix =l and %vs, 63
	%t1 =w csltl %ix, 26
	jnz %t1, @m1, @m2
@m1
	%ch1 =l add %ix, 65
	storeb %ch1, %dp
	jmp @jnext
@m2
	%t2 =w csltl %ix, 52
	jnz %t2, @m3, @m4
@m3
	%ch3 =l add %ix, 71
	storeb %ch3, %dp
	jmp @jnext
@m4
	%t3 =w csltl %ix, 62
	jnz %t3, @m5, @m6
@m5
	%ch5 =l sub %ix, 4
	storeb %ch5, %dp
	jmp @jnext
@m6
	%t4 =w ceql %ix, 62
	jnz %t4, @m7, @m8
@m7
	storeb 43, %dp
	jmp @jnext
@m8
	storeb 47, %dp
	jmp @jnext
@pad
	storeb 61, %dp
	jmp @jnext
@jnext
	%jn =l add %j, 1
	%cl =w csltl %jn, 4
	jnz %cl, @jloop, @gend
@gend
	%in =l add %i, 3
	%on =l add %o, 4
	jmp @gloop
@done
	ret %o
}
