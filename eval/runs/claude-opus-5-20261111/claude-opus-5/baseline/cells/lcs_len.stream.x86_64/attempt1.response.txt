	push	rbx
	push	r12
	xor	eax, eax
.Lzero:

	inc	rax
	cmp	rax, rcx
	jle	.Lzero
	test	rsi, rsi
	jle	.Ldone
	xor	r9d, r9d
.Louter:
	movzx	r10d, byte ptr [rdi+r9]
	xor	r11d, r11d
	mov	eax, 1
	test	rcx, rcx
	jle	.Lnext
.Linner:
	mov	rbx, qword ptr [r8+rax*8]
	movzx	r12d, byte ptr [rdx+rax-1]
	cmp	r12b, r10b
	jne	.Lmax
	lea	r12, [r11+1]

	jmp	.Lcont
.Lmax:

	cmp	r12, rbx
	cmovl	r12, rbx
	mov	qword ptr [r8+rax*8], r12
.Lcont:

	inc	rax
	cmp	rax, rcx
	jle	.Linner
.Lnext:
	inc	r9
	cmp	r9, rsi
	jl	.Louter
.Ldone:
	mov	rax, qword ptr [r8+rcx*8]
	pop	r12
	pop	rbx
