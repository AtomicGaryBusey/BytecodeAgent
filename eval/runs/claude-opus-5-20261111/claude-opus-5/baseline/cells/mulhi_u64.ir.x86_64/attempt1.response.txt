export function l $mulhi_u64(l %a, l %b) {
@start
	%m =l copy 4294967295
	%a0 =l and %a, %m
	%a1 =l shr %a, 32
	%b0 =l and %b, %m
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%t0 =l shr %p00, 32
	%t1 =l and %p01, %m
	%t2 =l and %p10, %m
	%s0 =l add %t0, %t1
	%mid =l add %s0, %t2
	%c =l shr %mid, 32
	%h0 =l shr %p01, 32
	%h1 =l shr %p10, 32
	%r0 =l add %p11, %h0
	%r1 =l add %r0, %h1
	%r2 =l add %r1, %c
	ret %r2
}
