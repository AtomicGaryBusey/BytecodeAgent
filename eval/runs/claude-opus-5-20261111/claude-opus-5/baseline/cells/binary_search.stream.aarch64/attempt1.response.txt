binary_search:
    mov x4, x2                  // key
    mov x2, xzr                 // lo = 0
    mov x3, x1                  // hi = n
.Lloop:
    cmp x2, x3
    b.hs .Lfail
    add x9, x2, x3
    lsr x9, x9, #1              // mid
    ldr x10, [x0, x9, lsl #3]
    cmp x10, x4
    b.eq .Lfound
    b.lt .Lhigher
    mov x3, x9                  // hi = mid
    b .Lloop
.Lhigher:
    add x2, x9, #1              // lo = mid + 1
    b .Lloop
.Lfound:
    mov x0, x9
    ret
.Lfail:
    mov x0, #-1
    ret
