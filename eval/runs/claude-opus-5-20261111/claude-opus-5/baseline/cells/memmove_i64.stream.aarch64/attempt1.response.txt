memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.lo .Lfwd
    mov x3, x2
.Lbwd:
    sub x3, x3, #1
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    cbnz x3, .Lbwd
    b .Ldone
.Lfwd:
    mov x3, #0
.Lfwdloop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x3, x3, #1
    cmp x3, x2
    b.lo .Lfwdloop
.Ldone:
    mov x0, #0
    ret
