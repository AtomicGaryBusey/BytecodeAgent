export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %nlo, @ge %lokeep
	%hi =l phi @start %n, @less %hikeep, @ge %nhi
	%c1 =w csltl %lo, %hi
	jnz %c1, @body, @done
@body
	%d =l sub %hi, %lo
	%h =l div %d, 2
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%c2 =w csltl %v, %key
	jnz %c2, @less, @ge
@less
	%nlo =l add %mid, 1
	%hikeep =l copy %hi
	jmp @loop
@ge
	%nhi =l copy %mid
	%lokeep =l copy %lo
	jmp @loop
@done
	%c3 =w csltl %lo, %n
	jnz %c3, @chk, @fail
@chk
	%off2 =l mul %lo, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%c4 =w ceql %v2, %key
	jnz %c4, @found, @fail
@found
	ret %lo
@fail
	%m =l copy -1
	ret %m
}
