export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%c0 =l shr %p00, 32
	%l01 =l and %p01, %mask
	%l10 =l and %p10, %mask
	%m1 =l add %c0, %l01
	%mid =l add %m1, %l10
	%h01 =l shr %p01, 32
	%h10 =l shr %p10, 32
	%hmid =l shr %mid, 32
	%t1 =l add %p11, %h01
	%t2 =l add %t1, %h10
	%res =l add %t2, %hmid
	ret %res
}
