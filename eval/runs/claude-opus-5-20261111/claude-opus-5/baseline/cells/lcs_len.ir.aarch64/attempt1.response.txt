export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zcond
@zcond
	%zj =l phi @start 0, @zbody %zjn
	%c1 =w cslel %zj, %n
	jnz %c1, @zbody, @zdone
@zbody
	%zo =l mul %zj, 8
	%zp =l add %row, %zo
	storel 0, %zp
	%zjn =l add %zj, 1
	jmp @zcond
@zdone
	jmp @icond
@icond
	%i =l phi @zdone 0, @iend %in
	%c2 =w csltl %i, %m
	jnz %c2, @ibody, @fin
@ibody
	%xp =l add %x, %i
	%xc =w loadub %xp
	jmp @jcond
@jcond
	%j =l phi @ibody 1, @jend %jn
	%diag =l phi @ibody 0, @jend %up
	%c3 =w cslel %j, %n
	jnz %c3, @jbody, @iend
@jbody
	%jm =l sub %j, 1
	%yp =l add %y, %jm
	%yc =w loadub %yp
	%ro =l mul %j, 8
	%rp =l add %row, %ro
	%up =l loadl %rp
	%lp =l sub %rp, 8
	%left =l loadl %lp
	%eq =w ceqw %xc, %yc
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	storel %v1, %rp
	jmp @jend
@nomatch
	%c4 =w csgtl %left, %up
	jnz %c4, @useleft, @useup
@useleft
	storel %left, %rp
	jmp @jend
@useup
	storel %up, %rp
	jmp @jend
@jend
	%jn =l add %j, 1
	jmp @jcond
@iend
	%in =l add %i, 1
	jmp @icond
@fin
	%fo =l mul %n, 8
	%fp =l add %row, %fo
	%res =l loadl %fp
	ret %res
}
