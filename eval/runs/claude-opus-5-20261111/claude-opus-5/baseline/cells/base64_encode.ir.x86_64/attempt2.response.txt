export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i0 =l copy 0
	%o0 =l copy 0
	jmp @loop
@loop
	%ii =l phi @start %i0, @gend %inext
	%oo =l phi @start %o0, @gend %onext
	%left =l sub %n, %ii
	%c0 =w csgtl %left, 0
	jnz %c0, @body, @done
@body
	%k3 =w csgel %left, 3
	jnz %k3, @kis3, @klt3
@kis3
	%ka =l copy 3
	jmp @kdone
@klt3
	%kb =l copy %left
	jmp @kdone
@kdone
	%k =l phi @kis3 %ka, @klt3 %kb
	%p0 =l add %src, %ii
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%ge2 =w csgel %k, 2
	jnz %ge2, @ld1, @v1done
@ld1
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%vv1 =l or %v0, %s1
	jmp @v1done
@v1done
	%v1 =l phi @kdone %v0, @ld1 %vv1
	%ge3 =w csgel %k, 3
	jnz %ge3, @ld2, @v2done
@ld2
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%vv2 =l or %v1, %b2
	jmp @v2done
@v2done
	%val =l phi @v1done %v1, @ld2 %vv2
	%nch =l add %k, 1
	%j0 =l copy 0
	jmp @cloop
@cloop
	%j =l phi @v2done %j0, @cnext %jn
	%cmp =w csltl %j, %nch
	jnz %cmp, @cbody, @cend
@cbody
	%m6 =l mul %j, 6
	%sh =l sub 18, %m6
	%sv =l shr %val, %sh
	%v =l and %sv, 63
	%t1 =w csgel %v, 26
	%t1l =l extsw %t1
	%a1 =l mul %t1l, 6
	%t2 =w csgel %v, 52
	%t2l =l extsw %t2
	%a2 =l mul %t2l, 75
	%t3 =w csgel %v, 62
	%t3l =l extsw %t3
	%a3 =l mul %t3l, 15
	%t4 =w csgel %v, 63
	%t4l =l extsw %t4
	%a4 =l mul %t4l, 3
	%r1 =l add %v, 65
	%r2 =l add %r1, %a1
	%r3 =l sub %r2, %a2
	%r4 =l sub %r3, %a3
	%r5 =l add %r4, %a4
	%rw =w copy %r5
	%op =l add %out, %oo
	%op2 =l add %op, %j
	storeb %rw, %op2
	jmp @cnext
@cnext
	%jn =l add %j, 1
	jmp @cloop
@cend
	%pj0 =l copy %nch
	jmp @ploop
@ploop
	%pj =l phi @cend %pj0, @pnext %pjn
	%pc =w csltl %pj, 4
	jnz %pc, @pbody, @gend
@pbody
	%pp =l add %out, %oo
	%pp2 =l add %pp, %pj
	storeb 61, %pp2
	jmp @pnext
@pnext
	%pjn =l add %pj, 1
	jmp @ploop
@gend
	%inext =l add %ii, %k
	%onext =l add %oo, 4
	jmp @loop
@done
	ret %oo
}
