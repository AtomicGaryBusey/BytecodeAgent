export function l $bmax(l %a, l %n) {
@start
	%v0 =l loadl %a
	jmp @loop
@loop
	%m =l phi @start %v0, @body2 %mn
	%i =l phi @start 1, @body2 %i1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%o =l mul %i, 8
	%p =l add %a, %o
	%v =l loadl %p
	%g =w csgtl %v, %m
	jnz %g, @upd, @skip
@upd
	jmp @body2
@skip
	jmp @body2
@body2
	%mn =l phi @upd %v, @skip %m
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %m
}
