export function l $bmax(l %a, l %n) {
@start
	%m =l loadl %a
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%o =l mul %i, 8
	%p =l add %a, %o
	%v =l loadl %p
	%g =w csgtl %v, %m
	jnz %g, @upd, @next
@upd
	%m1 =l copy %v
	jmp @next2
@next
	%m2 =l copy %m
	jmp @next2
@next2
	%m =l phi @upd %m1, @next %m2
	%i =l add %i, 1
	jmp @loop
@end
	ret %m
}
