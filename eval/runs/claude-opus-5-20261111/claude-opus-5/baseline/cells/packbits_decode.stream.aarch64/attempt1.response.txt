packbits_decode:
    mov x4, #0                  // i = source index
    mov x5, #0                  // w = bytes written
.Lloop:
    cmp x4, x1
    b.ge .Ldone
    ldrsb x6, [x0, x4]          // h, sign extended
    add x4, x4, #1
    cmn x6, #128                // h == -128 ?
    b.eq .Lloop
    tbnz x6, #63, .Lrepeat
    add x7, x6, #1              // cnt = h + 1
    sub x8, x1, x4              // remaining source bytes
    cmp x8, x7
    b.lt .Lerr2
    sub x9, x3, x5              // room = cap - w
    cmp x7, x9
    csel x10, x7, x9, lt        // copy = min(cnt, room)
    add x12, x0, x4
    add x13, x2, x5
.Ll1:
    cbz x10, .Ll2
    ldrb w11, [x12], #1
    strb w11, [x13], #1
    sub x10, x10, #1
    b .Ll1
.Ll2:
    add x4, x4, x7
    cmp x7, x9
    b.gt .Lovf
    add x5, x5, x7
    b .Lloop
.Lrepeat:
    sub x7, xzr, x6
    add x7, x7, #1              // cnt = 1 - h
    sub x8, x1, x4
    cmp x8, #1
    b.lt .Lerr2
    ldrb w11, [x0, x4]
    add x4, x4, #1
    sub x9, x3, x5              // room = cap - w
    cmp x7, x9
    csel x10, x7, x9, lt
    add x13, x2, x5
.Lr1:
    cbz x10, .Lr2
    strb w11, [x13], #1
    sub x10, x10, #1
    b .Lr1
.Lr2:
    cmp x7, x9
    b.gt .Lovf
    add x5, x5, x7
    b .Lloop
.Ldone:
    mov x0, x5
    ret
.Lerr2:
    mov x0, #-2
    ret
.Lovf:
    mov x0, #-1
    ret
