collatz_max:
	xor eax, eax
	mov rsi, 1
.Louter:
	cmp rsi, rdi
	jg .Ldone
	mov rcx, rsi
	xor edx, edx
.Linner:
	cmp rcx, 1
	je .Lcheck
	test rcx, 1
	jnz .Lodd
	shr rcx, 1
	inc rdx
	jmp .Linner
.Lodd:
	lea rcx, [rcx + rcx*2 + 1]
	inc rdx
	jmp .Linner
.Lcheck:
	cmp rdx, rax
	jle .Lnext
	mov rax, rdx
.Lnext:
	inc rsi
	jmp .Louter
.Ldone:
	ret
