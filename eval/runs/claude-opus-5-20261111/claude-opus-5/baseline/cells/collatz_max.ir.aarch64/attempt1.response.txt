export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy 1
@outer
	%c1 =w csgtl %k, %n
	jnz %c1, @done, @body
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%c2 =w ceql %x, 1
	jnz %c2, @endinner, @step
@step
	%r =l and %x, 1
	jnz %r, @odd, @even
@even
	%xe =l div %x, 2
	%x =l copy %xe
	jmp @next
@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	%x =l copy %xo
	jmp @next
@next
	%steps =l add %steps, 1
	jmp @inner
@endinner
	%c3 =w csgtl %steps, %best
	jnz %c3, @upd, @noupd
@upd
	%best =l copy %steps
	jmp @noupd
@noupd
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
