export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%ce =w ceql %pc, %len
	jnz %ce, @done, @fetch
@fetch
	%fo =l mul %pc, 8
	%fp =l add %code, %fo
	%op =l loadl %fp
	%k0 =w ceql %op, 0
	jnz %k0, @done, @c1
@c1
	%k1 =w ceql %op, 1
	jnz %k1, @push, @c2
@c2
	%k2 =w ceql %op, 2
	jnz %k2, @opadd, @c3
@c3
	%k3 =w ceql %op, 3
	jnz %k3, @opsub, @c4
@c4
	%k4 =w ceql %op, 4
	jnz %k4, @opmul, @c5
@c5
	%k5 =w ceql %op, 5
	jnz %k5, @opdup, @c6
@c6
	%k6 =w ceql %op, 6
	jnz %k6, @opswap, @c7
@c7
	%k7 =w ceql %op, 7
	jnz %k7, @opdrop, @c8
@c8
	%k8 =w ceql %op, 8
	jnz %k8, @opover, @c9
@c9
	%k9 =w ceql %op, 9
	jnz %k9, @oplt, @c10
@c10
	%k10 =w ceql %op, 10
	jnz %k10, @opjz, @c11
@c11
	%k11 =w ceql %op, 11
	jnz %k11, @opjmp, @nextpc
@nextpc
	%n1 =l add %pc, 1
	storel %n1, %pcp
	jmp @loop
@push
	%pu1 =l add %pc, 1
	%pu2 =l mul %pu1, 8
	%pu3 =l add %code, %pu2
	%puv =l loadl %pu3
	%pud =l loadl %dp
	%pu4 =l mul %pud, 8
	%pu5 =l add %stack, %pu4
	storel %puv, %pu5
	%pu6 =l add %pud, 1
	storel %pu6, %dp
	%pu7 =l add %pc, 2
	storel %pu7, %pcp
	jmp @loop
@opadd
	%aad =l loadl %dp
	%aai =l sub %aad, 1
	%aao =l mul %aai, 8
	%aap =l add %stack, %aao
	%aab =l loadl %aap
	%aaj =l sub %aad, 2
	%aaq =l mul %aaj, 8
	%aar =l add %stack, %aaq
	%aaa =l loadl %aar
	%aas =l add %aaa, %aab
	storel %aas, %aar
	storel %aai, %dp
	%aan =l add %pc, 1
	storel %aan, %pcp
	jmp @loop
@opsub
	%ssd =l loadl %dp
	%ssi =l sub %ssd, 1
	%sso =l mul %ssi, 8
	%ssp =l add %stack, %sso
	%ssb =l loadl %ssp
	%ssj =l sub %ssd, 2
	%ssq =l mul %ssj, 8
	%ssr =l add %stack, %ssq
	%ssa =l loadl %ssr
	%sss =l sub %ssa, %ssb
	storel %sss, %ssr
	storel %ssi, %dp
	%ssn =l add %pc, 1
	storel %ssn, %pcp
	jmp @loop
@opmul
	%mmd =l loadl %dp
	%mmi =l sub %mmd, 1
	%mmo =l mul %mmi, 8
	%mmp =l add %stack, %mmo
	%mmb =l loadl %mmp
	%mmj =l sub %mmd, 2
	%mmq =l mul %mmj, 8
	%mmr =l add %stack, %mmq
	%mma =l loadl %mmr
	%mms =l mul %mma, %mmb
	storel %mms, %mmr
	storel %mmi, %dp
	%mmn =l add %pc, 1
	storel %mmn, %pcp
	jmp @loop
@oplt
	%ltd =l loadl %dp
	%lti =l sub %ltd, 1
	%lto =l mul %lti, 8
	%ltp =l add %stack, %lto
	%ltb =l loadl %ltp
	%ltj =l sub %ltd, 2
	%ltq =l mul %ltj, 8
	%ltr =l add %stack, %ltq
	%lta =l loadl %ltr
	%ltc =w csltl %lta, %ltb
	%lts =l extuw %ltc
	storel %lts, %ltr
	storel %lti, %dp
	%ltn =l add %pc, 1
	storel %ltn, %pcp
	jmp @loop
@opdup
	%dud =l loadl %dp
	%dui =l sub %dud, 1
	%duo =l mul %dui, 8
	%dup1 =l add %stack, %duo
	%duv =l loadl %dup1
	%duq =l mul %dud, 8
	%dur =l add %stack, %duq
	storel %duv, %dur
	%dun =l add %dud, 1
	storel %dun, %dp
	%dum =l add %pc, 1
	storel %dum, %pcp
	jmp @loop
@opswap
	%swd =l loadl %dp
	%swi =l sub %swd, 1
	%swo =l mul %swi, 8
	%swp =l add %stack, %swo
	%swb =l loadl %swp
	%swj =l sub %swd, 2
	%swq =l mul %swj, 8
	%swr =l add %stack, %swq
	%swa =l loadl %swr
	storel %swa, %swp
	storel %swb, %swr
	%swn =l add %pc, 1
	storel %swn, %pcp
	jmp @loop
@opdrop
	%drd =l loadl %dp
	%dri =l sub %drd, 1
	storel %dri, %dp
	%drn =l add %pc, 1
	storel %drn, %pcp
	jmp @loop
@opover
	%ovd =l loadl %dp
	%ovj =l sub %ovd, 2
	%ovq =l mul %ovj, 8
	%ovr =l add %stack, %ovq
	%ova =l loadl %ovr
	%ovo =l mul %ovd, 8
	%ovp =l add %stack, %ovo
	storel %ova, %ovp
	%ovn =l add %ovd, 1
	storel %ovn, %dp
	%ovm =l add %pc, 1
	storel %ovm, %pcp
	jmp @loop
@opjz
	%jzd =l loadl %dp
	%jzi =l sub %jzd, 1
	%jzo =l mul %jzi, 8
	%jzp =l add %stack, %jzo
	%jzv =l loadl %jzp
	storel %jzi, %dp
	%jzc =w ceql %jzv, 0
	jnz %jzc, @jztake, @jzskip
@jztake
	%jz1 =l add %pc, 1
	%jz2 =l mul %jz1, 8
	%jz3 =l add %code, %jz2
	%jz4 =l loadl %jz3
	storel %jz4, %pcp
	jmp @loop
@jzskip
	%jz5 =l add %pc, 2
	storel %jz5, %pcp
	jmp @loop
@opjmp
	%jm1 =l add %pc, 1
	%jm2 =l mul %jm1, 8
	%jm3 =l add %code, %jm2
	%jm4 =l loadl %jm3
	storel %jm4, %pcp
	jmp @loop
@done
	%fd =l loadl %dp
	storel %fd, %ip
	jmp @zloop
@zloop
	%zi =l loadl %ip
	%zc =w csltl %zi, %cap
	jnz %zc, @zbody, @zend
@zbody
	%zo =l mul %zi, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%zn =l add %zi, 1
	storel %zn, %ip
	jmp @zloop
@zend
	%rd =l loadl %dp
	%rc =w ceql %rd, 0
	jnz %rc, @retz, @retv
@retz
	ret 0
@retv
	%ri =l sub %rd, 1
	%ro =l mul %ri, 8
	%rp =l add %stack, %ro
	%rv =l loadl %rp
	ret %rv
}
