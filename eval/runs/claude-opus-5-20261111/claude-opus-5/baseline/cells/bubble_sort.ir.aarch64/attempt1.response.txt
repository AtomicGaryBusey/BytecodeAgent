export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%c0 =w csltl %i, %n
	jnz %c0, @outerbody, @done
@outerbody
	%lim =l sub %n, %i
	%lim1 =l sub %lim, 1
	%j =l copy 0
@inner
	%c1 =w csltl %j, %lim1
	jnz %c1, @innerbody, @innerdone
@innerbody
	%o0 =l mul %j, 8
	%p0 =l add %a, %o0
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%c2 =w csgtl %v0, %v1
	jnz %c2, @swap, @noswap
@swap
	storel %v1, %p0
	storel %v0, %p1
@noswap
	%j1 =l add %j, 1
	%j =l copy %j1
	jmp @inner
@innerdone
	%i1 =l add %i, 1
	%i =l copy %i1
	jmp @outer
@done
	ret 0
}
