export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%c0 =w csltl 0, %lim
	jnz %c0, @init, @zero
@init
	%p0 =l loadl %a
	jmp @body
@body
	%i =l phi @init 0, @body %next
	%prev =l phi @init %p0, @body %cur
	%cnt =l phi @init 0, @body %cnt2
	%next =l add %i, 1
	%off =l mul %next, 8
	%pp =l add %a, %off
	%cur =l loadl %pp
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt2 =l add %cnt, %gtl
	%c1 =w csltl %next, %lim
	jnz %c1, @body, @outl
@outl
	jmp @done
@zero
	jmp @done
@done
	%res =l phi @outl %cnt2, @zero 0
	ret %res
}
