export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%lim =l sub %n, 1
	%c0 =w csltl 0, %lim
	jnz %c0, @loop, @done
@loop
	%i =l copy 0
	%prev =l loadl %a
@body
	%next =l add %i, 1
	%p =l mul %next, 8
	%pp =l add %a, %p
	%cur =l loadl %pp
	%gt =w csgtl %prev, %cur
	%cnt2 =l add %cnt, %gt
	%cnt =l copy %cnt2
	%prev =l copy %cur
	%i =l copy %next
	%c1 =w csltl %i, %lim
	jnz %c1, @body, @done
@done
	ret %cnt
}
