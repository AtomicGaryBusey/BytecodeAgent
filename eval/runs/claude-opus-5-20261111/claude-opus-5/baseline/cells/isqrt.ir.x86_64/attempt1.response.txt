export function l $isqrt(l %n) {
@start
	%p =l alloc8 24
	%pn =l add %p, 0
	%pr =l add %p, 8
	%pb =l add %p, 16
	storel %n, %pn
	storel 0, %pr
	storel 4611686018427387904, %pb
	jmp @shiftcond
@shiftcond
	%b1 =l loadl %pb
	%n1 =l loadl %pn
	%c1 =w cugtl %b1, %n1
	jnz %c1, @shiftbody, @maincond
@shiftbody
	%b2 =l shr %b1, 2
	storel %b2, %pb
	jmp @shiftcond
@maincond
	%b3 =l loadl %pb
	jnz %b3, @mainbody, @done
@mainbody
	%n3 =l loadl %pn
	%r3 =l loadl %pr
	%s =l add %r3, %b3
	%c2 =w cugel %n3, %s
	jnz %c2, @sub, @nosub
@sub
	%n4 =l sub %n3, %s
	storel %n4, %pn
	%r4 =l shr %r3, 1
	%r5 =l add %r4, %b3
	storel %r5, %pr
	jmp @next
@nosub
	%r6 =l shr %r3, 1
	storel %r6, %pr
	jmp @next
@next
	%b4 =l shr %b3, 2
	storel %b4, %pb
	jmp @maincond
@done
	%rr =l loadl %pr
	ret %rr
}
