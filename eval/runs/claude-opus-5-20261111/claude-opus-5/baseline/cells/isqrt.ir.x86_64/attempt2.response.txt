export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%m =l phi @start 2147483648, @loop %m2
	%r =l phi @start 0, @loop %r2
	%t =l add %r, %m
	%q =l udiv %n, %t
	%c =w cugel %q, %t
	%cl =l extuw %c
	%d =l mul %cl, %m
	%r2 =l add %r, %d
	%m2 =l shr %m, 1
	%go =w cnel %m2, 0
	jnz %go, @loop, @end
@end
	ret %r2
}
