map_into:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	sub rsp, 8
	mov r13, rdi
	mov r14, rsi
	mov r15, rdx
	mov r12, rcx
	xor ebx, ebx
	xor ebp, ebp
	test r15, r15
	jle .Ldone
.Lloop:
	mov rdi, qword ptr [r14 + rbx*8]
	mov qword ptr [rsp], rdi
	call r12
	mov qword ptr [r13 + rbx*8], rax
	cmp rax, qword ptr [rsp]
	jge .Lskip
	add rbp, 1
.Lskip:
	add rbx, 1
	cmp rbx, r15
	jl .Lloop
.Ldone:
	mov rax, rbp
	add rsp, 8
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
