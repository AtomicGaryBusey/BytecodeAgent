export function l $qsort_pairs(l %a, l %n) {
@start
	%t0 =w csgtl %n, 1
	jnz %t0, @work, @done
@work
	%half =l div %n, 2
	%pm =l mul %half, 16
	%pp =l add %a, %pm
	%last =l sub %n, 1
	%lm =l mul %last, 16
	%lp =l add %a, %lm
	%lp8 =l add %lp, 8
	%pp8 =l add %pp, 8
	%ak =l loadl %pp
	%av =l loadl %pp8
	%bk =l loadl %lp
	%bv =l loadl %lp8
	storel %bk, %pp
	storel %bv, %pp8
	storel %ak, %lp
	storel %av, %lp8
	jmp @loop
@loop
	%i =l phi @work 0, @nxt %i3
	%j =l phi @work 0, @nxt %j2
	%cj =w csltl %j, %last
	jnz %cj, @body, @after
@body
	%jo =l mul %j, 16
	%jp =l add %a, %jo
	%jp8 =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jp8
	%c1 =w csltl %jk, %ak
	jnz %c1, @doswap, @chk2
@chk2
	%c2 =w ceql %jk, %ak
	jnz %c2, @chk3, @nxt
@chk3
	%c3 =w csltl %jv, %av
	jnz %c3, @doswap, @nxt
@doswap
	%io =l mul %i, 16
	%ip =l add %a, %io
	%ip8 =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%i2 =l add %i, 1
	jmp @nxt
@nxt
	%i3 =l phi @doswap %i2, @chk2 %i, @chk3 %i
	%j2 =l add %j, 1
	jmp @loop
@after
	%io2 =l mul %i, 16
	%ip2 =l add %a, %io2
	%ip28 =l add %ip2, 8
	%xk =l loadl %ip2
	%xv =l loadl %ip28
	%yk =l loadl %lp
	%yv =l loadl %lp8
	storel %yk, %ip2
	storel %yv, %ip28
	storel %xk, %lp
	storel %xv, %lp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ri =l add %i, 1
	%ro =l mul %ri, 16
	%rp =l add %a, %ro
	%rn =l sub %n, %ri
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	ret 0
@done
	ret 0
}
