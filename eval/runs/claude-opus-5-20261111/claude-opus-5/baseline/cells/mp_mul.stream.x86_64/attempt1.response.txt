mp_mul:
	push	rbx
	push	r12
	push	r13
	push	r14
	push	r15
	mov	r9, rdx
	lea	r10, [rsi + rcx]
	xor	r11d, r11d
.Lzero:
	cmp	r11, r10
	jge	.Lzerodone
	mov	qword ptr [r8 + r11*8], 0
	inc	r11
	jmp	.Lzero
.Lzerodone:
	test	rsi, rsi
	jle	.Lcount
	test	rcx, rcx
	jle	.Lcount
	xor	r11d, r11d
.Louter:
	mov	rdx, qword ptr [rdi + r11*8]
	xor	ebx, ebx
	xor	r12d, r12d
.Linner:
	mulx	r14, r13, qword ptr [r9 + r12*8]
	add	r13, rbx
	adc	r14, 0
	lea	r15, [r11 + r12]
	add	qword ptr [r8 + r15*8], r13
	adc	r14, 0
	mov	rbx, r14
	inc	r12
	cmp	r12, rcx
	jl	.Linner
	lea	r15, [r11 + rcx]
	mov	qword ptr [r8 + r15*8], rbx
	inc	r11
	cmp	r11, rsi
	jl	.Louter
.Lcount:
	mov	rax, r10
.Lcnt:
	test	rax, rax
	jle	.Lret
	cmp	qword ptr [r8 + rax*8 - 8], 0
	jne	.Lret
	dec	rax
	jmp	.Lcnt
.Lret:
	pop	r15
	pop	r14
	pop	r13
	pop	r12
	pop	rbx
	ret
