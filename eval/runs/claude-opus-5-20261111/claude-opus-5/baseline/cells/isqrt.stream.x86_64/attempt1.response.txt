isqrt:
	xor eax, eax                      # res = 0
	mov rcx, 1
	shl rcx, 62                       # bit = 4^31
.Lloop:
	mov rdx, rax
	add rdx, rcx                      # sum = res + bit  (provably < 2^64)
	cmp rdi, rdx
	jb .Lskip
	sub rdi, rdx                      # n -= res + bit
	shr rax, 1
	add rax, rcx                      # res = res/2 + bit
	jmp .Lnext
.Lskip:
	shr rax, 1                        # res = res/2
.Lnext:
	shr rcx, 2                        # bit >>= 2
	jnz .Lloop
	ret
