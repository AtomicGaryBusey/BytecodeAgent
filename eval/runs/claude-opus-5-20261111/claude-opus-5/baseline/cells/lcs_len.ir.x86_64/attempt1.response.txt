export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zloop
@zloop
	%jz =l phi @start 0, @zbody %jzn
	%c0 =w csgtl %jz, %n
	jnz %c0, @iinit, @zbody
@zbody
	%ofz =l mul %jz, 8
	%pz =l add %row, %ofz
	storel 0, %pz
	%jzn =l add %jz, 1
	jmp @zloop
@iinit
	jmp @iloop
@iloop
	%i =l phi @iinit 0, @iend %in
	%ci =w csltl %i, %m
	jnz %ci, @ibody, @done
@ibody
	%xp =l add %x, %i
	%xc =w loadub %xp
	jmp @jloop
@jloop
	%j =l phi @ibody 1, @jend %jn
	%diag =l phi @ibody 0, @jend %up
	%cj =w csgtl %j, %n
	jnz %cj, @iend, @jbody
@jbody
	%of =l mul %j, 8
	%pj =l add %row, %of
	%up =l loadl %pj
	%yp =l add %y, %j
	%ypm =l sub %yp, 1
	%yc =w loadub %ypm
	%eq =w ceqw %xc, %yc
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @store
@nomatch
	%pjm =l sub %pj, 8
	%left =l loadl %pjm
	%cm =w csgtl %left, %up
	jnz %cm, @useleft, @useup
@useleft
	jmp @store
@useup
	jmp @store
@store
	%v =l phi @match %v1, @useleft %left, @useup %up
	storel %v, %pj
	jmp @jend
@jend
	%jn =l add %j, 1
	jmp @jloop
@iend
	%in =l add %i, 1
	jmp @iloop
@done
	%ofn =l mul %n, 8
	%pn =l add %row, %ofn
	%r =l loadl %pn
	ret %r
}
