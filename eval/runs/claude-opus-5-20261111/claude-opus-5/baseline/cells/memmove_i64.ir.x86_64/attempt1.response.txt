export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c0 =w csgtl %n, 0
	jnz %c0, @check, @done
@check
	%fwd =w cultl %dst, %src
	jnz %fwd, @fwdinit, @bwdinit
@fwdinit
	%i =l copy 0
	jmp @fwdloop
@fwdloop
	%fp =l mul %i, 8
	%fs =l add %src, %fp
	%fd =l add %dst, %fp
	%fv =l loadl %fs
	storel %fv, %fd
	%i2 =l add %i, 1
	%fc =w csltl %i2, %n
	%i =l copy %i2
	jnz %fc, @fwdloop, @done
@bwdinit
	%j =l sub %n, 1
	jmp @bwdloop
@bwdloop
	%bp =l mul %j, 8
	%bs =l add %src, %bp
	%bd =l add %dst, %bp
	%bv =l loadl %bs
	storel %bv, %bd
	%j2 =l sub %j, 1
	%bc =w csgel %j2, 0
	%j =l copy %j2
	jnz %bc, @bwdloop, @done
@done
	ret 0
}
