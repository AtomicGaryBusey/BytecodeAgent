crc32:
	mov eax, 0xFFFFFFFF
	test rsi, rsi
	jle .Ldone
	xor ecx, ecx
.Lbyte:
	movzx edx, byte ptr [rdi + rcx]
	xor eax, edx
	mov r8d, 8
.Lbit:
	mov r9d, eax
	and r9d, 1
	shr eax, 1
	neg r9d
	and r9d, 0xEDB88320
	xor eax, r9d
	dec r8d
	jnz .Lbit
	inc rcx
	cmp rcx, rsi
	jl .Lbyte
.Ldone:
	not eax
	mov eax, eax
	ret
