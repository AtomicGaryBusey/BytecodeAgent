export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	storel 0, %ip
	storel 0, %op
@loop
	%i =l loadl %ip
	%c0 =w csgel %i, %n
	jnz %c0, @done, @hdr
@hdr
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i1 =l add %i, 1
	storel %i1, %ip
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @chk
@chk
	%isneg =w csltw %h, 0
	jnz %isneg, @rep, @lit
@lit
	%hl =l extsw %h
	%cnt =l add %hl, 1
	%send =l add %i1, %cnt
	%bad =w csgtl %send, %n
	jnz %bad, @err2, @litloop
@litloop
	%li =l loadl %ip
	%more =w csltl %li, %send
	jnz %more, @litbody, @loop
@litbody
	%lo =l loadl %op
	%full =w csgel %lo, %cap
	jnz %full, @err1, @litw
@litw
	%sp2 =l add %src, %li
	%bv =w loadub %sp2
	%dp =l add %out, %lo
	storeb %bv, %dp
	%li2 =l add %li, 1
	storel %li2, %ip
	%lo2 =l add %lo, 1
	storel %lo2, %op
	jmp @litloop
@rep
	%send2 =l add %i1, 1
	%bad2 =w csgtl %send2, %n
	jnz %bad2, @err2, @rep2
@rep2
	%rp =l add %src, %i1
	%rb =w loadub %rp
	storel %send2, %ip
	%hl2 =l extsw %h
	%cnt2 =l sub 1, %hl2
	%ostart =l loadl %op
	%oend =l add %ostart, %cnt2
	jmp @reploop
@reploop
	%ro =l loadl %op
	%more2 =w csltl %ro, %oend
	jnz %more2, @repbody, @loop
@repbody
	%full2 =w csgel %ro, %cap
	jnz %full2, @err1, @repw
@repw
	%dp2 =l add %out, %ro
	storeb %rb, %dp2
	%ro2 =l add %ro, 1
	storel %ro2, %op
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	%res =l loadl %op
	ret %res
}
