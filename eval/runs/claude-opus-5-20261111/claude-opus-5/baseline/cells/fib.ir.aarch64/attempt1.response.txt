export function l $fib(l %n) {
@start
	%c0 =w ceql %n, 0
	jnz %c0, @ret0, @loop_init
@ret0
	ret 0
@loop_init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 1
	jmp @loop
@loop
	%ai =l phi @loop_init %a, @body %an
	%bi =l phi @loop_init %b, @body %bn
	%ii =l phi @loop_init %i, @body %in
	%c1 =w csgel %ii, %n
	jnz %c1, @done, @body
@body
	%an =l copy %bi
	%bn =l add %ai, %bi
	%in =l add %ii, 1
	jmp @loop
@done
	ret %bi
}
