export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @right %loR, @left %loL
	%hi =l phi @start %n, @right %hiR, @left %hiL
	%c =w csltl %lo, %hi
	jnz %c, @body, @fail
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%loR =l add %mid, 1
	%hiR =l copy %hi
	jmp @loop
@left
	%loL =l copy %lo
	%hiL =l copy %mid
	jmp @loop
@found
	ret %mid
@fail
	ret -1
}
