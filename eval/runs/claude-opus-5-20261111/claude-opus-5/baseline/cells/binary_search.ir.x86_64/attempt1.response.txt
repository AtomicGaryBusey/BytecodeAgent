export function l $binary_search(%a : l, %n : l, %key : l) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @fail
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%lo2 =l add %mid, 1
	%lo =l copy %lo2
	jmp @loop
@left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@fail
	ret -1
}
