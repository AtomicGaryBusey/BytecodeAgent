export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nnb =l mul %nn, 8
	%prod =l add %tmp, %nnb
	%nn2 =l add %nn, %nn
	%np1 =l add %n, 1
	jmp @idz
@idz
	%zi =l phi @start 0, @idzb %zin
	%cz0 =w csltl %zi, %nn
	jnz %cz0, @idzb, @diag
@idzb
	%zo =l mul %zi, 8
	%zp =l add %out, %zo
	storel 0, %zp
	%zin =l add %zi, 1
	jmp @idz
@diag
	%di =l phi @idz 0, @diagb %din
	%cd0 =w csltl %di, %n
	jnz %cd0, @diagb, @cpy
@diagb
	%dix =l mul %di, %np1
	%do =l mul %dix, 8
	%dp =l add %out, %do
	storel 1, %dp
	%din =l add %di, 1
	jmp @diag
@cpy
	%ci =l phi @diag 0, @cpyb %cin
	%cc0 =w csltl %ci, %nn
	jnz %cc0, @cpyb, @loop
@cpyb
	%co =l mul %ci, 8
	%cs =l add %a, %co
	%cv =l loadl %cs
	%cdst =l add %tmp, %co
	storel %cv, %cdst
	%cin =l add %ci, 1
	jmp @cpy
@loop
	%kk =l phi @cpy %k, @afterm %kk2, @m2c %kk2
	%ck =w cnel %kk, 0
	jnz %ck, @bodytest, @zero
@bodytest
	%bit =l and %kk, 1
	%bitw =w cnel %bit, 0
	jnz %bitw, @m1i, @afterm
@m1i
	%i1 =l phi @bodytest 0, @m1iend %i1n
	%r1 =l mul %i1, %n
	%ci1 =w csltl %i1, %n
	jnz %ci1, @m1j, @m1c
@m1j
	%j1 =l phi @m1i 0, @m1store %j1n
	%cj1 =w csltl %j1, %n
	jnz %cj1, @m1t, @m1iend
@m1t
	%t1 =l phi @m1j 0, @m1tb %t1n
	%s1 =l phi @m1j 0, @m1tb %s1n
	%ct1 =w csltl %t1, %n
	jnz %ct1, @m1tb, @m1store
@m1tb
	%xi1 =l add %r1, %t1
	%xo1 =l mul %xi1, 8
	%xp1 =l add %out, %xo1
	%xv1 =l loadl %xp1
	%yr1 =l mul %t1, %n
	%yi1 =l add %yr1, %j1
	%yo1 =l mul %yi1, 8
	%yp1 =l add %tmp, %yo1
	%yv1 =l loadl %yp1
	%pm1 =l mul %xv1, %yv1
	%s1n =l add %s1, %pm1
	%t1n =l add %t1, 1
	jmp @m1t
@m1store
	%qi1 =l add %r1, %j1
	%qo1 =l mul %qi1, 8
	%qp1 =l add %prod, %qo1
	storel %s1, %qp1
	%j1n =l add %j1, 1
	jmp @m1j
@m1iend
	%i1n =l add %i1, 1
	jmp @m1i
@m1c
	%u1 =l phi @m1i 0, @m1cb %u1n
	%cu1 =w csltl %u1, %nn
	jnz %cu1, @m1cb, @afterm
@m1cb
	%uo1 =l mul %u1, 8
	%us1 =l add %prod, %uo1
	%uv1 =l loadl %us1
	%ud1 =l add %out, %uo1
	storel %uv1, %ud1
	%u1n =l add %u1, 1
	jmp @m1c
@afterm
	%kk2 =l shr %kk, 1
	%nzw =w cnel %kk2, 0
	jnz %nzw, @m2i, @loop
@m2i
	%i2 =l phi @afterm 0, @m2iend %i2n
	%r2 =l mul %i2, %n
	%ci2 =w csltl %i2, %n
	jnz %ci2, @m2j, @m2c
@m2j
	%j2 =l phi @m2i 0, @m2store %j2n
	%cj2 =w csltl %j2, %n
	jnz %cj2, @m2t, @m2iend
@m2t
	%t2 =l phi @m2j 0, @m2tb %t2n
	%s2 =l phi @m2j 0, @m2tb %s2n
	%ct2 =w csltl %t2, %n
	jnz %ct2, @m2tb, @m2store
@m2tb
	%xi2 =l add %r2, %t2
	%xo2 =l mul %xi2, 8
	%xp2 =l add %tmp, %xo2
	%xv2 =l loadl %xp2
	%yr2 =l mul %t2, %n
	%yi2 =l add %yr2, %j2
	%yo2 =l mul %yi2, 8
	%yp2 =l add %tmp, %yo2
	%yv2 =l loadl %yp2
	%pm2 =l mul %xv2, %yv2
	%s2n =l add %s2, %pm2
	%t2n =l add %t2, 1
	jmp @m2t
@m2store
	%qi2 =l add %r2, %j2
	%qo2 =l mul %qi2, 8
	%qp2 =l add %prod, %qo2
	storel %s2, %qp2
	%j2n =l add %j2, 1
	jmp @m2j
@m2iend
	%i2n =l add %i2, 1
	jmp @m2i
@m2c
	%u2 =l phi @m2i 0, @m2cb %u2n
	%cu2 =w csltl %u2, %nn
	jnz %cu2, @m2cb, @loop
@m2cb
	%uo2 =l mul %u2, 8
	%us2 =l add %prod, %uo2
	%uv2 =l loadl %us2
	%ud2 =l add %tmp, %uo2
	storel %uv2, %ud2
	%u2n =l add %u2, 1
	jmp @m2c
@zero
	%zj =l phi @loop 0, @zerob %zjn
	%cz1 =w csltl %zj, %nn2
	jnz %cz1, @zerob, @ret
@zerob
	%zjo =l mul %zj, 8
	%zjp =l add %tmp, %zjo
	storel 0, %zjp
	%zjn =l add %zj, 1
	jmp @zero
@ret
	ret 0
}
