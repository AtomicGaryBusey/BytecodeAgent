export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nnb =l mul %nn, 8
	%prod =l add %tmp, %nnb
	%np1 =l add %n, 1
	%nn2 =l add %nn, %nn
	jmp @initloop
@initloop
	%p =l phi @start 0, @initbody %pn
	%c0 =w csltl %p, %nn
	jnz %c0, @initbody, @mainloop
@initbody
	%poff =l mul %p, 8
	%pa =l add %a, %poff
	%po =l add %out, %poff
	%pt =l add %tmp, %poff
	%av =l loadl %pa
	storel %av, %pt
	%rm =l rem %p, %np1
	%cd =w ceql %rm, 0
	%dv =l extuw %cd
	storel %dv, %po
	%pn =l add %p, 1
	jmp @initloop
@mainloop
	%kc =l phi @initloop %k, @loopend %kn
	%cz =w cnel %kc, 0
	jnz %cz, @chkbit, @cleanup
@chkbit
	%bit =l and %kc, 1
	%cb =w cnel %bit, 0
	jnz %cb, @m_i, @s_i
@m_i
	%mi =l phi @chkbit 0, @m_iinc %mi1
	%min =l mul %mi, %n
	%cmi =w csltl %mi, %n
	jnz %cmi, @m_j, @m_copy
@m_j
	%mj =l phi @m_i 0, @m_jinc %mj1
	%cmj =w csltl %mj, %n
	jnz %cmj, @m_t, @m_iinc
@m_t
	%mt =l phi @m_j 0, @m_tbody %mt1
	%ms =l phi @m_j 0, @m_tbody %ms1
	%cmt =w csltl %mt, %n
	jnz %cmt, @m_tbody, @m_tdone
@m_tbody
	%mxi =l add %min, %mt
	%mxo =l mul %mxi, 8
	%mxp =l add %out, %mxo
	%mx =l loadl %mxp
	%mtn =l mul %mt, %n
	%myi =l add %mtn, %mj
	%myo =l mul %myi, 8
	%myp =l add %tmp, %myo
	%my =l loadl %myp
	%mm =l mul %mx, %my
	%ms1 =l add %ms, %mm
	%mt1 =l add %mt, 1
	jmp @m_t
@m_tdone
	%mdi =l add %min, %mj
	%mdo =l mul %mdi, 8
	%mdp =l add %prod, %mdo
	storel %ms, %mdp
	jmp @m_jinc
@m_jinc
	%mj1 =l add %mj, 1
	jmp @m_j
@m_iinc
	%mi1 =l add %mi, 1
	jmp @m_i
@m_copy
	%mp =l phi @m_i 0, @m_copyb %mp1
	%cmp =w csltl %mp, %nn
	jnz %cmp, @m_copyb, @s_i
@m_copyb
	%mco =l mul %mp, 8
	%mcs =l add %prod, %mco
	%mcd =l add %out, %mco
	%mcv =l loadl %mcs
	storel %mcv, %mcd
	%mp1 =l add %mp, 1
	jmp @m_copy
@s_i
	%si =l phi @chkbit 0, @m_copy 0, @s_iinc %si1
	%sin =l mul %si, %n
	%csi =w csltl %si, %n
	jnz %csi, @s_j, @s_copy
@s_j
	%sj =l phi @s_i 0, @s_jinc %sj1
	%csj =w csltl %sj, %n
	jnz %csj, @s_t, @s_iinc
@s_t
	%st =l phi @s_j 0, @s_tbody %st1
	%ss =l phi @s_j 0, @s_tbody %ss1
	%cst =w csltl %st, %n
	jnz %cst, @s_tbody, @s_tdone
@s_tbody
	%sxi =l add %sin, %st
	%sxo =l mul %sxi, 8
	%sxp =l add %tmp, %sxo
	%sx =l loadl %sxp
	%stn =l mul %st, %n
	%syi =l add %stn, %sj
	%syo =l mul %syi, 8
	%syp =l add %tmp, %syo
	%sy =l loadl %syp
	%sm =l mul %sx, %sy
	%ss1 =l add %ss, %sm
	%st1 =l add %st, 1
	jmp @s_t
@s_tdone
	%sdi =l add %sin, %sj
	%sdo =l mul %sdi, 8
	%sdp =l add %prod, %sdo
	storel %ss, %sdp
	jmp @s_jinc
@s_jinc
	%sj1 =l add %sj, 1
	jmp @s_j
@s_iinc
	%si1 =l add %si, 1
	jmp @s_i
@s_copy
	%sp =l phi @s_i 0, @s_copyb %sp1
	%csp =w csltl %sp, %nn
	jnz %csp, @s_copyb, @loopend
@s_copyb
	%sco =l mul %sp, 8
	%scs =l add %prod, %sco
	%scd =l add %tmp, %sco
	%scv =l loadl %scs
	storel %scv, %scd
	%sp1 =l add %sp, 1
	jmp @s_copy
@loopend
	%kn =l shr %kc, 1
	jmp @mainloop
@cleanup
	%zp =l phi @mainloop 0, @cleanupb %zp1
	%czp =w csltl %zp, %nn2
	jnz %czp, @cleanupb, @fin
@cleanupb
	%zo =l mul %zp, 8
	%zpp =l add %tmp, %zo
	storel 0, %zpp
	%zp1 =l add %zp, 1
	jmp @cleanup
@fin
	ret 0
}
