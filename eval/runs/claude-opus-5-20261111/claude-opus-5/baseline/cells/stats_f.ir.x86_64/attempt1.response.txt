export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%c0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @entry 1, @maxjoin %i2
	%mn =l phi @entry %y0, @maxjoin %mn2
	%mx =l phi @entry %y0, @maxjoin %mx2
	%sm =l phi @entry %y0, @maxjoin %sm2
	%cn =l phi @entry %c0, @maxjoin %cn2
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%off =l mul %i, 8
	%pa =l add %a, %off
	%x =l loadl %pa
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	jnz %lt, @minset, @minskip
@minset
	jmp @minjoin
@minskip
	jmp @minjoin
@minjoin
	%mn2 =l phi @minset %y, @minskip %mn
	%gt =w csgtl %y, %mx
	jnz %gt, @maxset, @maxskip
@maxset
	jmp @maxjoin
@maxskip
	jmp @maxjoin
@maxjoin
	%mx2 =l phi @maxset %y, @maxskip %mx
	%sm2 =l add %sm, %y
	%od =l and %y, 1
	%cn2 =l add %cn, %od
	%i2 =l add %i, 1
	jmp @loop
@done
	storel %mn, %out
	%o1 =l add %out, 8
	storel %mx, %o1
	%o2 =l add %out, 16
	storel %sm, %o2
	ret %cn
}
