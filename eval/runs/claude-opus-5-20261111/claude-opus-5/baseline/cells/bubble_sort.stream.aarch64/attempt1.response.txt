bubble_sort:
	cmp	x1, #2
	b.lt	.Ldone
	sub	x9, x1, #1		// outer limit i = n-1
.Louter:
	mov	x10, #0			// j = 0
	mov	x13, #0			// swapped flag
.Linner:
	cmp	x10, x9
	b.ge	.Linner_end
	ldr	x11, [x0, x10, lsl #3]
	add	x12, x10, #1
	ldr	x14, [x0, x12, lsl #3]
	cmp	x11, x14
	b.le	.Lnoswap
	str	x14, [x0, x10, lsl #3]
	str	x11, [x0, x12, lsl #3]
	mov	x13, #1
.Lnoswap:
	mov	x10, x12
	b	.Linner
.Linner_end:
	cbz	x13, .Ldone
	subs	x9, x9, #1
	b.gt	.Louter
.Ldone:
	mov	x0, #0
	ret
