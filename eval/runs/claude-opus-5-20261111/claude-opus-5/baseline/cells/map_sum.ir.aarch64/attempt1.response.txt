export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum1 =l add %sum, %r
	%i1 =l add %i, 1
	%sum =l copy %sum1
	%i =l copy %i1
	jmp @cond
@done
	ret %sum
}
