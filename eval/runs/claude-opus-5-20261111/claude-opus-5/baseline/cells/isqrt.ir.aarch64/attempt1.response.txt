export function l $isqrt(l %n) {
@start
	%res0 =l copy 0
	%bit0 =l copy 4611686018427387904
	jmp @loop
@loop
	%res =l phi @start %res0, @cont %resn
	%bit =l phi @start %bit0, @cont %bitn
	%nn =l phi @start %n, @cont %nnn
	%c =w cnel %bit, 0
	jnz %c, @body, @end
@body
	%t =l add %res, %bit
	%ge =w cugel %nn, %t
	jnz %ge, @yes, @no
@yes
	%n1 =l sub %nn, %t
	%r1 =l shr %res, 1
	%r1b =l add %r1, %bit
	jmp @cont
@no
	%n2 =l copy %nn
	%r2 =l shr %res, 1
	jmp @cont
@cont
	%nnn =l phi @yes %n1, @no %n2
	%resn =l phi @yes %r1b, @no %r2
	%bitn =l shr %bit, 2
	jmp @loop
@end
	ret %res
}
