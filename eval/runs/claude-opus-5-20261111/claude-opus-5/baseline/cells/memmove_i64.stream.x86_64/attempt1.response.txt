memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lfwd
    xor rcx, rcx
    mov rcx, rdx
    dec rcx
.Lbackloop:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    dec rcx
    jns .Lbackloop
    jmp .Ldone
.Lfwd:
    xor rcx, rcx
.Lfwdloop:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jl .Lfwdloop
.Ldone:
    xor eax, eax
    ret
