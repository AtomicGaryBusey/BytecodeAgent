export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start

	%pj =l alloc8 8

	storel 0, %pi
	jmp @zcond
@zcond

@zbody

@icond

	jnz %ic, @ibody, @idone
@ibody
	%ao =l mul %i, 8

	%av =l loadl %ap
	storel 0, %pc
	storel 0, %pj
	jmp @jcond
@jcond
