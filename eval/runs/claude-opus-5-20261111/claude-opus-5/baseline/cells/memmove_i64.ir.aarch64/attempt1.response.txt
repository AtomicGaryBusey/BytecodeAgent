export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c0 =w csgtl %n, 0
	jnz %c0, @check, @done
@check
	%cmp =w cugtl %dst, %src
	jnz %cmp, @back, @fwd
@fwd
	%i =l copy 0
	jmp @floop
@floop
	%fc =w csltl %i, %n
	jnz %fc, @fbody, @done
@fbody
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i2 =l add %i, 1
	%i =l copy %i2
	jmp @floop
@back
	%j =l sub %n, 1
	jmp @bloop
@bloop
	%bc =w csgel %j, 0
	jnz %bc, @bbody, @done
@bbody
	%off2 =l mul %j, 8
	%sp2 =l add %src, %off2
	%dp2 =l add %dst, %off2
	%v2 =l loadl %sp2
	storel %v2, %dp2
	%j2 =l sub %j, 1
	%j =l copy %j2
	jmp @bloop
@done
	ret 0
}
