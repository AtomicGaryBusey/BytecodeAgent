export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	%pj =l alloc8 8
	%pc =l alloc8 8
	%pt =l alloc8 8
	%pr =l alloc4 4
	%pk =l alloc4 4
	storel 0, %pi
	storel 0, %po
	jmp @main
@main
	%i =l loadl %pi
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@body
	%ap =l add %src, %i
	%bi =w loadub %ap
	storew 1, %pr
	jmp @rloop
@rloop
	%rw =w loadw %pr
	%rl =l loadsw %pr
	%i1 =l loadl %pi
	%t1 =l add %i1, %rl
	%cc1 =w csltl %t1, %n
	jnz %cc1, @rchk, @rend
@rchk
	%cc2 =w csltw %rw, 128
	jnz %cc2, @rchk2, @rend
@rchk2
	%p2 =l add %src, %t1
	%b2 =w loadub %p2
	%cc3 =w ceqw %b2, %bi
	jnz %cc3, @rinc, @rend
@rinc
	%rw2 =w add %rw, 1
	storew %rw2, %pr
	jmp @rloop
@rend
	%rwf =w loadw %pr
	%cc4 =w csgew %rwf, 3
	jnz %cc4, @rep, @lit
@rep
	%o1 =l loadl %po
	%o2 =l add %o1, 2
	%cc5 =w csgtl %o2, %cap
	jnz %cc5, @fail, @rep2
@rep2
	%hp =l add %out, %o1
	%hv =w sub 257, %rwf
	storeb %hv, %hp
	%bp =l add %hp, 1
	storeb %bi, %bp
	storel %o2, %po
	%rl2 =l loadsw %pr
	%ipre =l loadl %pi
	%ni =l add %ipre, %rl2
	storel %ni, %pi
	jmp @main
@lit
	%li =l loadl %pi
	storel %li, %pj
	jmp @ltest
@ltest
	%j =l loadl %pj
	%lc1 =w csltl %j, %n
	jnz %lc1, @ltest2, @litdone
@ltest2
	%li2 =l loadl %pi
	%d =l sub %j, %li2
	%lc2 =w csltl %d, 128
	jnz %lc2, @ltest3, @litdone
@ltest3
	%jp =l add %src, %j
	%bj =w loadub %jp
	storew 1, %pk
	jmp @kloop
@kloop
	%kw =w loadw %pk
	%kc0 =w csltw %kw, 3
	jnz %kc0, @kchk, @kend
@kchk
	%kl =l loadsw %pk
	%jj =l loadl %pj
	%tk =l add %jj, %kl
	%kc1 =w csltl %tk, %n
	jnz %kc1, @kchk2, @kend
@kchk2
	%kp =l add %src, %tk
	%bk =w loadub %kp
	%kc2 =w ceqw %bk, %bj
	jnz %kc2, @kinc, @kend
@kinc
	%kw2 =w add %kw, 1
	storew %kw2, %pk
	jmp @kloop
@kend
	%kf =w loadw %pk
	%kc3 =w csgew %kf, 3
	jnz %kc3, @litdone, @ladv
@ladv
	%j2 =l loadl %pj
	%j3 =l add %j2, 1
	storel %j3, %pj
	jmp @ltest
@litdone
	%jf =l loadl %pj
	%if =l loadl %pi
	%L =l sub %jf, %if
	%of =l loadl %po
	%need =l add %L, 1
	%end =l add %of, %need
	%lc3 =w csgtl %end, %cap
	jnz %lc3, @fail, @lwrite
@lwrite
	%hp2 =l add %out, %of
	%hl =l sub %L, 1
	storel %hl, %pt
	%hw2 =w loadw %pt
	storeb %hw2, %hp2
	storel 0, %pc
	jmp @cloop
@cloop
	%c =l loadl %pc
	%cc9 =w csltl %c, %L
	jnz %cc9, @cbody, @cend
@cbody
	%sp2 =l add %if, %c
	%sp3 =l add %src, %sp2
	%bv =w loadub %sp3
	%dp1 =l add %of, 1
	%dp2 =l add %dp1, %c
	%dp3 =l add %out, %dp2
	storeb %bv, %dp3
	%c2 =l add %c, 1
	storel %c2, %pc
	jmp @cloop
@cend
	storel %end, %po
	storel %jf, %pi
	jmp @main
@fail
	ret -1
@done
	%of2 =l loadl %po
	ret %of2
}
