export function l $qsort_pairs(l %a, l %n) {
@start
	%si =l alloc8 8
	%sj =l alloc8 8
	%big =w csgtl %n, 1
	jnz %big, @work, @done
@work
	%nm1 =l sub %n, 1
	%h =l div %nm1, 2
	%hb =l mul %h, 16
	%pp =l add %a, %hb
	%pk =l loadl %pp
	%ppv =l add %pp, 8
	%pv =l loadl %ppv
	%m1 =l sub 0, 1
	storel %m1, %si
	storel %n, %sj
	jmp @outer
@outer
	%io =l loadl %si
	%io1 =l add %io, 1
	storel %io1, %si
	jmp @iloop
@iloop
	%i =l loadl %si
	%ib =l mul %i, 16
	%ipp =l add %a, %ib
	%ik =l loadl %ipp
	%ivp =l add %ipp, 8
	%iv =l loadl %ivp
	%il =w csltl %ik, %pk
	%ie =w ceql %ik, %pk
	%ivl =w csltl %iv, %pv
	%iand =w and %ie, %ivl
	%ir =w or %il, %iand
	jnz %ir, @inc, @jpre
@inc
	%i2 =l add %i, 1
	storel %i2, %si
	jmp @iloop
@jpre
	%jo =l loadl %sj
	%jo1 =l sub %jo, 1
	storel %jo1, %sj
	jmp @jloop
@jloop
	%j =l loadl %sj
	%jb =l mul %j, 16
	%jpp =l add %a, %jb
	%jk =l loadl %jpp
	%jvp =l add %jpp, 8
	%jv =l loadl %jvp
	%jl =w csltl %pk, %jk
	%je =w ceql %pk, %jk
	%jvl =w csltl %pv, %jv
	%jand =w and %je, %jvl
	%jr =w or %jl, %jand
	jnz %jr, @jdec, @chk
@jdec
	%j2 =l sub %j, 1
	storel %j2, %sj
	jmp @jloop
@chk
	%ci =l loadl %si
	%cj =l loadl %sj
	%cross =w csgel %ci, %cj
	jnz %cross, @part, @swap
@swap
	storel %jk, %ipp
	storel %jv, %ivp
	storel %ik, %jpp
	storel %iv, %jvp
	jmp @outer
@part
	%ln =l add %cj, 1
	%r1 =l call $qsort_pairs(l %a, l %ln)
	%roff =l mul %ln, 16
	%rp =l add %a, %roff
	%rn =l sub %n, %ln
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	jmp @done
@done
	ret 0
}
