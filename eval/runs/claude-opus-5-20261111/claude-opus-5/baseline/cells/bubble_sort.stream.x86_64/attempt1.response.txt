bubble_sort:
    xor rax, rax
    cmp rsi, 2
    jl .Ldone
    mov rcx, rsi
    dec rcx
.Louter:
    xor rdx, rdx
    xor r8, r8
.Linner:
    mov r9, qword ptr [rdi + r8*8]
    mov r10, qword ptr [rdi + r8*8 + 8]
    cmp r9, r10
    jle .Lnoswap
    mov qword ptr [rdi + r8*8], r10
    mov qword ptr [rdi + r8*8 + 8], r9
    mov rdx, 1
.Lnoswap:
    inc r8
    cmp r8, rcx
    jl .Linner
    dec rcx
    test rdx, rdx
    jz .Ldone
    cmp rcx, 1
    jge .Louter
.Ldone:
    xor rax, rax
    ret
