export function l $bubble_sort(l %a, l %n) {
@start
	jmp @oloop
@oloop
	%i =l phi @start 0, @onext %i1
	%c1 =w csltl %i, %n
	jnz %c1, @obody, @end
@obody
	%rem =l sub %n, %i
	%lim =l sub %rem, 1
	jmp @iloop
@iloop
	%j =l phi @obody 0, @inext %j1
	%c2 =w csltl %j, %lim
	jnz %c2, @ibody, @onext
@ibody
	%off =l mul %j, 8
	%p1 =l add %a, %off
	%p2 =l add %p1, 8
	%x =l loadl %p1
	%y =l loadl %p2
	%c3 =w csgtl %x, %y
	jnz %c3, @swap, @inext
@swap
	storel %y, %p1
	storel %x, %p2
	jmp @inext
@inext
	%j1 =l add %j, 1
	jmp @iloop
@onext
	%i1 =l add %i, 1
	jmp @oloop
@end
	ret 0
}
