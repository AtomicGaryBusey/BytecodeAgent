binary_search:
    xor rax, rax
    test rsi, rsi
    jle .Lfail
    xor r8, r8
    mov r9, rsi
.Lloop:
    cmp r8, r9
    jge .Lfail
    mov rcx, r8
    add rcx, r9
    shr rcx, 1
    mov r10, qword ptr [rdi + rcx*8]
    cmp r10, rdx
    je .Lfound
    jl .Lgo_right
    mov r9, rcx
    jmp .Lloop
.Lgo_right:
    lea r8, [rcx + 1]
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lfail:
    mov rax, -1
    ret
