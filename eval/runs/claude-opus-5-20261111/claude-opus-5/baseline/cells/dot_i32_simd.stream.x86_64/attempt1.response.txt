dot_i32_simd:
pxor xmm4, xmm4
xor rax, rax
xor r8, r8
mov rcx, rdx
and rcx, -4
je .Ltail
.Lvec:
movdqu xmm0, [rdi + r8*4]
movdqu xmm1, [rsi + r8*4]
movdqa xmm2, xmm0
movdqa xmm3, xmm1
pmuldq xmm2, xmm3
psrlq xmm0, 32
psrlq xmm1, 32
pmuldq xmm0, xmm1
paddq xmm4, xmm2
paddq xmm4, xmm0
add r8, 4
cmp r8, rcx
jb .Lvec
.Ltail:
movdqa xmm0, xmm4
psrldq xmm0, 8
paddq xmm4, xmm0
movq rax, xmm4
.Lscalar:
cmp r8, rdx
jge .Ldone
movsxd r9, dword ptr [rdi + r8*4]
movsxd r10, dword ptr [rsi + r8*4]
imul r9, r10
add rax, r9
inc r8
jmp .Lscalar
.Ldone:
ret
