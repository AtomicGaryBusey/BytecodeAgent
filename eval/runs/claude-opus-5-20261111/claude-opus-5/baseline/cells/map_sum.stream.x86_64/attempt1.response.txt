map_sum:
	push	rbx
	push	rbp
	push	r12
	push	r13
	push	r14
	xor	eax, eax
	test	rsi, rsi
	jle	.Ldone
	mov	r12, rdi
	mov	r13, rsi
	mov	r14, rdx
	xor	ebx, ebx
	xor	ebp, ebp
.Lloop:
	mov	rdi, qword ptr [r12 + rbx*8]
	call	r14
	add	rbp, rax
	add	rbx, 1
	cmp	rbx, r13
	jl	.Lloop
	mov	rax, rbp
.Ldone:
	pop	r14
	pop	r13
	pop	r12
	pop	rbp
	pop	rbx
	ret
