export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%si =l alloc8 8
	%sw =l alloc8 8
	%sj =l alloc8 8
	%sr =l alloc8 8
	%sk =l alloc8 8
	%sq =l alloc8 8
	storel 0, %si
	storel 0, %sw
	jmp @loop
@loop
	%i =l loadl %si
	%cc =w csltl %i, %n
	jnz %cc, @body, @done
@body
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	storel 1, %sr
	%k0 =l add %i, 1
	storel %k0, %sk
	jmp @rl
@rl
	%k =l loadl %sk
	%r1 =l loadl %sr
	%t1 =w csltl %k, %n
	jnz %t1, @rl2, @rldone
@rl2
	%t2 =w csltl %r1, 128
	jnz %t2, @rl3, @rldone
@rl3
	%pk =l add %src, %k
	%bk =w loadub %pk
	%t3 =w ceqw %bk, %b0
	jnz %t3, @rl4, @rldone
@rl4
	%rn =l add %r1, 1
	storel %rn, %sr
	%kn =l add %k, 1
	storel %kn, %sk
	jmp @rl
@rldone
	%r =l loadl %sr
	%t4 =w csgel %r, 3
	jnz %t4, @rep, @lit
@rep
	%w =l loadl %sw
	%w2 =l add %w, 2
	%t5 =w csgtl %w2, %cap
	jnz %t5, @fail, @rep2
@rep2
	%rw =w loadw %sr
	%h =w sub 257, %rw
	%o1 =l add %out, %w
	storeb %h, %o1
	%o2 =l add %o1, 1
	%bb =w loadub %p0
	storeb %bb, %o2
	storel %w2, %sw
	%in =l add %i, %r
	storel %in, %si
	jmp @loop
@lit
	storel %i, %sj
	jmp @litloop
@litloop
	%j =l loadl %sj
	%u1 =w csltl %j, %n
	jnz %u1, @litc2, @litend
@litc2
	%d =l sub %j, %i
	%u2 =w csltl %d, 128
	jnz %u2, @litc3, @litend
@litc3
	%pj =l add %src, %j
	%bj =w loadub %pj
	storel 1, %sr
	%kj =l add %j, 1
	storel %kj, %sk
	jmp @jl
@jl
	%jk =l loadl %sk
	%jr =l loadl %sr
	%v1 =w csltl %jk, %n
	jnz %v1, @jl2, @jldone
@jl2
	%v2 =w csltl %jr, 3
	jnz %v2, @jl3, @jldone
@jl3
	%pjk =l add %src, %jk
	%bjk =w loadub %pjk
	%v3 =w ceqw %bjk, %bj
	jnz %v3, @jl4, @jldone
@jl4
	%jrn =l add %jr, 1
	storel %jrn, %sr
	%jkn =l add %jk, 1
	storel %jkn, %sk
	jmp @jl
@jldone
	%jr2 =l loadl %sr
	%v4 =w csgel %jr2, 3
	jnz %v4, @litend, @litadv
@litadv
	%jn =l add %j, 1
	storel %jn, %sj
	jmp @litloop
@litend
	%jf =l loadl %sj
	%L =l sub %jf, %i
	%w3 =l loadl %sw
	%need =l add %w3, 1
	%need2 =l add %need, %L
	%v5 =w csgtl %need2, %cap
	jnz %v5, @fail, @emit
@emit
	%hp =l add %out, %w3
	%iw =w loadw %si
	%jw =w loadw %sj
	%Lmw =w sub %jw, %iw
	%Lm1 =w sub %Lmw, 1
	storeb %Lm1, %hp
	storel 0, %sq
	jmp @cp
@cp
	%ci =l loadl %sq
	%cz =w csltl %ci, %L
	jnz %cz, @cp2, @cpdone
@cp2
	%sp =l add %src, %i
	%sp2 =l add %sp, %ci
	%cb =w loadub %sp2
	%dp0 =l add %out, %w3
	%dp1 =l add %dp0, 1
	%dp =l add %dp1, %ci
	storeb %cb, %dp
	%cin =l add %ci, 1
	storel %cin, %sq
	jmp @cp
@cpdone
	storel %need2, %sw
	storel %jf, %si
	jmp @loop
@fail
	%m1 =l copy -1
	ret %m1
@done
	%wf =l loadl %sw
	ret %wf
}
