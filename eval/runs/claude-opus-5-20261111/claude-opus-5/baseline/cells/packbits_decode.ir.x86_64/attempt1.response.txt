export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	%pe =l alloc8 8
	%pb =l alloc8 8
	storel 0, %pi
	storel 0, %po
	storel 0, %pe
	storel 0, %pb
	jmp @loop
@loop
	%i =l loadl %pi
	%c0 =w csgel %i, %n
	jnz %c0, @done, @rdhdr
@rdhdr
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i1 =l add %i, 1
	storel %i1, %pi
	%c1 =w csltw %h, 0
	jnz %c1, @neg, @lit
@lit
	%hl =l extsw %h
	%cnt =l add %hl, 1
	%end =l add %i1, %cnt
	%c2 =w csgtl %end, %n
	jnz %c2, @err2, @litgo
@litgo
	storel %end, %pe
	jmp @litloop
@litloop
	%li =l loadl %pi
	%le =l loadl %pe
	%c3 =w csgel %li, %le
	jnz %c3, @loop, @litbody
@litbody
	%lo =l loadl %po
	%c4 =w csgel %lo, %cap
	jnz %c4, @err1, @litwr
@litwr
	%lsp =l add %src, %li
	%lv =w loadub %lsp
	%ldp =l add %out, %lo
	storeb %lv, %ldp
	%li2 =l add %li, 1
	storel %li2, %pi
	%lo2 =l add %lo, 1
	storel %lo2, %po
	jmp @litloop
@neg
	%c5 =w ceqw %h, -128
	jnz %c5, @loop, @rep
@rep
	%c6 =w csgel %i1, %n
	jnz %c6, @err2, @repgo
@repgo
	%bp =l add %src, %i1
	%bv =w loadub %bp
	storeb %bv, %pb
	%i2 =l add %i1, 1
	storel %i2, %pi
	%hl2 =l extsw %h
	%rc =l sub 1, %hl2
	%ro0 =l loadl %po
	%rend =l add %ro0, %rc
	storel %rend, %pe
	jmp @reploop
@reploop
	%ro =l loadl %po
	%re =l loadl %pe
	%c7 =w csgel %ro, %re
	jnz %c7, @loop, @repbody
@repbody
	%c8 =w csgel %ro, %cap
	jnz %c8, @err1, @repwr
@repwr
	%rv =w loadub %pb
	%rdp =l add %out, %ro
	storeb %rv, %rdp
	%ro2 =l add %ro, 1
	storel %ro2, %po
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	%res =l loadl %po
	ret %res
}
