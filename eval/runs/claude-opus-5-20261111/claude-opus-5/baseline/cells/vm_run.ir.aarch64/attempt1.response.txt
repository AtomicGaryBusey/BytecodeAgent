export function l $vm_run(l %code, l %len, l %stk, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%e =w ceql %pc, %len
	jnz %e, @done, @exec
@exec
	%o1 =l mul %pc, 8
	%o2 =l add %code, %o1
	%op =l loadl %o2
	%c0 =w ceql %op, 0
	jnz %c0, @done, @d1
@d1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @d2
@d2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @d3
@d3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @d4
@d4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @d5
@d5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @d6
@d6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @d7
@d7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @d8
@d8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @d9
@d9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @d10
@d10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @d11
@d11
	%c11 =w ceql %op, 11
	jnz %c11, @jmp, @next1
@push
	%ppc =l loadl %pcp
	%poa =l add %ppc, 1
	%pom =l mul %poa, 8
	%pop =l add %code, %pom
	%pv =l loadl %pop
	%pd =l loadl %dp
	%psa =l mul %pd, 8
	%psp =l add %stk, %psa
	storel %pv, %psp
	%pd1 =l add %pd, 1
	storel %pd1, %dp
	%pnc =l add %ppc, 2
	storel %pnc, %pcp
	jmp @loop
@add
	%ad =l loadl %dp
	%ad1 =l sub %ad, 1
	%aba =l mul %ad1, 8
	%abp =l add %stk, %aba
	%ab =l loadl %abp
	%ad2 =l sub %ad, 2
	%aaa =l mul %ad2, 8
	%aap =l add %stk, %aaa
	%aa =l loadl %aap
	%ar =l add %aa, %ab
	storel %ar, %aap
	storel %ad1, %dp
	jmp @next1
@sub
	%sd =l loadl %dp
	%sd1 =l sub %sd, 1
	%sba =l mul %sd1, 8
	%sbp =l add %stk, %sba
	%sb =l loadl %sbp
	%sd2 =l sub %sd, 2
	%saa =l mul %sd2, 8
	%sap =l add %stk, %saa
	%sa =l loadl %sap
	%sr =l sub %sa, %sb
	storel %sr, %sap
	storel %sd1, %dp
	jmp @next1
@mul
	%md =l loadl %dp
	%md1 =l sub %md, 1
	%mba =l mul %md1, 8
	%mbp =l add %stk, %mba
	%mb =l loadl %mbp
	%md2 =l sub %md, 2
	%maa =l mul %md2, 8
	%map =l add %stk, %maa
	%ma =l loadl %map
	%mr =l mul %ma, %mb
	storel %mr, %map
	storel %md1, %dp
	jmp @next1
@lt
	%ld =l loadl %dp
	%ld1 =l sub %ld, 1
	%lba =l mul %ld1, 8
	%lbp =l add %stk, %lba
	%lb =l loadl %lbp
	%ld2 =l sub %ld, 2
	%laa =l mul %ld2, 8
	%lap =l add %stk, %laa
	%la =l loadl %lap
	%lw =w csltl %la, %lb
	%lr =l extuw %lw
	storel %lr, %lap
	storel %ld1, %dp
	jmp @next1
@dup
	%ud =l loadl %dp
	%ud1 =l sub %ud, 1
	%uta =l mul %ud1, 8
	%utp =l add %stk, %uta
	%uv =l loadl %utp
	%una =l mul %ud, 8
	%unp =l add %stk, %una
	storel %uv, %unp
	%ud2 =l add %ud, 1
	storel %ud2, %dp
	jmp @next1
@swap
	%wd =l loadl %dp
	%wd1 =l sub %wd, 1
	%wa1 =l mul %wd1, 8
	%wp1 =l add %stk, %wa1
	%wv1 =l loadl %wp1
	%wd2 =l sub %wd, 2
	%wa2 =l mul %wd2, 8
	%wp2 =l add %stk, %wa2
	%wv2 =l loadl %wp2
	storel %wv2, %wp1
	storel %wv1, %wp2
	jmp @next1
@drop
	%rd =l loadl %dp
	%rd1 =l sub %rd, 1
	storel %rd1, %dp
	jmp @next1
@over
	%vd =l loadl %dp
	%vd2 =l sub %vd, 2
	%vaa =l mul %vd2, 8
	%vap =l add %stk, %vaa
	%vv =l loadl %vap
	%vna =l mul %vd, 8
	%vnp =l add %stk, %vna
	storel %vv, %vnp
	%vd1 =l add %vd, 1
	storel %vd1, %dp
	jmp @next1
@jz
	%zd =l loadl %dp
	%zd1 =l sub %zd, 1
	%zva =l mul %zd1, 8
	%zvp =l add %stk, %zva
	%zv =l loadl %zvp
	storel %zd1, %dp
	%zc =w ceql %zv, 0
	jnz %zc, @jmp, @jzskip
@jzskip
	%kpc =l loadl %pcp
	%k2 =l add %kpc, 2
	storel %k2, %pcp
	jmp @loop
@jmp
	%jpc =l loadl %pcp
	%jta =l add %jpc, 1
	%jtm =l mul %jta, 8
	%jtp =l add %code, %jtm
	%jt =l loadl %jtp
	storel %jt, %pcp
	jmp @loop
@next1
	%npc =l loadl %pcp
	%nn =l add %npc, 1
	storel %nn, %pcp
	jmp @loop
@done
	%fd =l loadl %dp
	storel %fd, %ip
	jmp @zchk
@zchk
	%fi =l loadl %ip
	%fc =w csltl %fi, %cap
	jnz %fc, @zb, @ret
@zb
	%fa =l mul %fi, 8
	%fp =l add %stk, %fa
	storel 0, %fp
	%fi1 =l add %fi, 1
	storel %fi1, %ip
	jmp @zchk
@ret
	%td =l loadl %dp
	%tc =w ceql %td, 0
	jnz %tc, @retz, @rett
@retz
	ret 0
@rett
	%tm =l sub %td, 1
	%ta =l mul %tm, 8
	%tp =l add %stk, %ta
	%tv =l loadl %tp
	ret %tv
}
