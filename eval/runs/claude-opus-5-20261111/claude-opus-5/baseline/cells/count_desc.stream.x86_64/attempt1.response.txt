count_desc:

    cmp rsi, 2
    jl .Ldone
    sub rsi, 1
    xor ecx, ecx
    mov rdx, qword ptr [rdi]
.Lloop:
    mov r8, qword ptr [rdi + rcx*8 + 8]
    cmp rdx, r8
    setg r9b
    movzx r9d, r9b

    mov rdx, r8
    add rcx, 1
    cmp rcx, rsi
    jl .Lloop
.Ldone:
    ret
