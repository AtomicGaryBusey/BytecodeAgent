isqrt:
    mov x1, #0                  // res = 0
    mov x2, #1
    lsl x2, x2, #62             // bit = 1 << 62
.Lshift:
    cmp x2, x0                  // while bit > n: bit >>= 2
    b.ls .Lloop
    lsr x2, x2, #2
    cbnz x2, .Lshift
.Lloop:
    cbz x2, .Ldone
    add x3, x1, x2              // t = res + bit
    cmp x0, x3
    b.lo .Lelse
    sub x0, x0, x3              // n -= t
    lsr x1, x1, #1
    add x1, x1, x2              // res = (res >> 1) + bit
    b .Lnext
.Lelse:
    lsr x1, x1, #1              // res >>= 1
.Lnext:
    lsr x2, x2, #2              // bit >>= 2
    b .Lloop
.Ldone:
    mov x0, x1
    ret
