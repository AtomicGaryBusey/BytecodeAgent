export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%lim =l sub %n, 1
	%c0 =w csgtl %lim, 0
	jnz %c0, @loop, @end
@loop
	%p =l mul %i, 8
	%p1 =l add %a, %p
	%x =l loadl %p1
	%p2 =l add %p1, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%cnt =l add %cnt, %ge
	%i =l add %i, 1
	%c =w csltl %i, %lim
	jnz %c, @loop, @end
@end
	ret %cnt
}
