export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @loop
@loop
	%c1 =w csltl %i, %n
	jnz %c1, @body, @done
@body
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crc =w xor %crc, %b
	%k =w copy 0
	jmp @bit
@bit
	%c2 =w csltw %k, 8
	jnz %c2, @bitbody, @bitdone
@bitbody
	%lo =w and %crc, 1
	%sh =w shr %crc, 1
	jnz %lo, @xorit, @noxor
@xorit
	%crc =w xor %sh, 3988292384
	jmp @bitnext
@noxor
	%crc =w copy %sh
	jmp @bitnext
@bitnext
	%k =w add %k, 1
	jmp @bit
@bitdone
	%i =l add %i, 1
	jmp @loop
@done
	%f =w xor %crc, 4294967295
	%r =l extuw %f
	ret %r
}
