export function l $max_byte(l %p, l %n) {
@start
	%best =l copy 0
	%i =l copy 0
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%a =l add %p, %i
	%b =l loadub %a
	%gt =w cugtl %b, %best
	jnz %gt, @upd, @next
@upd
	%best =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @cond
@end
	ret %best
}
