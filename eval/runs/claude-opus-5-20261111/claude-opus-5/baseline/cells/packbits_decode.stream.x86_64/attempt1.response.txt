packbits_decode:
	xor	r8d, r8d
	xor	r9d, r9d
.Lloop:
	cmp	r8, rsi
	jge	.Ldone
	movsx	r10d, byte ptr [rdi + r8]
	inc	r8
	cmp	r10d, -128
	je	.Lloop
	test	r10d, r10d
	js	.Lrepeat
	inc	r10d
	mov	r11, r8
	add	r11, r10
	cmp	r11, rsi
	jg	.Lerr2
.Llit:
	test	r10, r10
	jz	.Lloop
	cmp	r9, rcx
	jae	.Lerr1
	mov	al, byte ptr [rdi + r8]
	mov	byte ptr [rdx + r9], al
	inc	r8
	inc	r9
	dec	r10
	jmp	.Llit
.Lrepeat:
	mov	r11d, 1
	sub	r11d, r10d
	cmp	r8, rsi
	jge	.Lerr2
	movzx	eax, byte ptr [rdi + r8]
	inc	r8
.Lrep:
	test	r11, r11
	jz	.Lloop
	cmp	r9, rcx
	jae	.Lerr1
	mov	byte ptr [rdx + r9], al
	inc	r9
	dec	r11
	jmp	.Lrep
.Ldone:
	mov	rax, r9
	ret
.Lerr1:
	mov	rax, -1
	ret
.Lerr2:
	mov	rax, -2
	ret
