export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %empty =w ceql %n, 0
    jnz %empty, @done, @choose
@choose
    %backward =w cultl %src, %dst
    jnz %backward, @back, @forward
@forward
    %fi =l phi @choose 0, @forward %fnext
    %foff =l mul %fi, 8
    %fsrc =l add %src, %foff
    %fdst =l add %dst, %foff
    %fval =l loadl %fsrc
    storel %fval, %fdst
    %fnext =l add %fi, 1
    %fmore =w csltl %fnext, %n
    jnz %fmore, @forward, @done
@back
    %bi =l phi @choose %n, @back %bnext
    %bnext =l sub %bi, 1
    %boff =l mul %bnext, 8
    %bsrc =l add %src, %boff
    %bdst =l add %dst, %boff
    %bval =l loadl %bsrc
    storel %bval, %bdst
    %bmore =w csgtl %bnext, 0
    jnz %bmore, @back, @done
@done
    ret 0
}
