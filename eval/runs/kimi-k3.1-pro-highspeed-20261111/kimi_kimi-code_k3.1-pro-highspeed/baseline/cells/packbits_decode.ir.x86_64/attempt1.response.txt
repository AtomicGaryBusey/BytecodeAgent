export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%ip =l alloc8 8
	%wp =l alloc8 8
	storel 0, %ip
	storel 0, %wp
@loop
	%i0 =l loadl %ip
	%finished =w csgel %i0, %n
	jnz %finished, @return, @packet
@packet
	%hp =l add %src, %i0
	%h32 =w loadsb %hp
	%h =l extsw %h32
	%i1 =l add %i0, 1
	storel %i1, %ip
	%is_nop =w ceql %h, -128
	jnz %is_nop, @loop, @not_nop
@not_nop
	%is_literal =w csgel %h, 0
	jnz %is_literal, @literal, @repeat
@literal
	%lit_count =l add %h, 1
	%src_left =l sub %n, %i1
	%lit_short =w csltl %src_left, %lit_count
	jnz %lit_short, @source_error, @literal_ready
@literal_ready
	%lit_end =l add %i1, %lit_count
@literal_loop
	%li =l loadl %ip
	%lit_done =w ceql %li, %lit_end
	jnz %lit_done, @loop, @literal_output
@literal_output
	%lw0 =l loadl %wp
	%lit_full =w csgel %lw0, %cap
	jnz %lit_full, @capacity_error, @literal_write
@literal_write
	%lsp =l add %src, %li
	%lv =w loadub %lsp
	%ldp =l add %out, %lw0
	storeb %lv, %ldp
	%li1 =l add %li, 1
	%lw1 =l add %lw0, 1
	storel %li1, %ip
	storel %lw1, %wp
	jmp @literal_loop
@repeat
	%value_present =w csltl %i1, %n
	jnz %value_present, @repeat_ready, @source_error
@repeat_ready
	%rep_count =l sub 1, %h
	%rvp =l add %src, %i1
	%rv =w loadub %rvp
	%i2 =l add %i1, 1
	storel %i2, %ip
	%rw0 =l loadl %wp
	%rep_end =l add %rw0, %rep_count
@repeat_loop
	%rw =l loadl %wp
	%rep_done =w ceql %rw, %rep_end
	jnz %rep_done, @loop, @repeat_output
@repeat_output
	%rep_full =w csgel %rw, %cap
	jnz %rep_full, @capacity_error, @repeat_write
@repeat_write
	%rdp =l add %out, %rw
	storeb %rv, %rdp
	%rw1 =l add %rw, 1
	storel %rw1, %wp
	jmp @repeat_loop
@source_error
	ret -2
@capacity_error
	ret -1
@return
	%result =l loadl %wp
	ret %result
}
