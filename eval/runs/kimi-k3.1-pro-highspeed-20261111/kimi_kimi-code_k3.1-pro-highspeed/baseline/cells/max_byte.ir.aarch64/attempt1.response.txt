export function l $max_byte(l %p, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @loop
@loop
	%i =l phi @start 0, @next %inext
	%max =l phi @start 0, @next %mnext
	%done =w ceql %i, %n
	jnz %done, @return, @body
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w cugtl %b, %max
	jnz %greater, @update, @skip
@update
	jmp @next
@skip
	jmp @next
@next
	%mnext =l phi @update %b, @skip %max
	%inext =l add %i, 1
	jmp @loop
@return
	ret %max
@zero
	ret 0
}
