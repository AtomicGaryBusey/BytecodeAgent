export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @loop
@loop
	%i =l phi @start 0, @next %i_next
	%count =l phi @start 0, @next %count_next
	%offset =l mul %i, 8
	%src_ptr =l add %src, %offset
	%x =l loadl %src_ptr
	%r =l call %f(l %x)
	%dst_ptr =l add %dst, %offset
	storel %r, %dst_ptr
	%less =w csltl %r, %x
	%increment =l extuw %less
	jmp @next
@next
	%count_next =l add %count, %increment
	%i_next =l add %i, 1
	%more =w csltl %i_next, %n
	jnz %more, @loop, @done
@done
	ret %count_next
@zero
	ret 0
}
