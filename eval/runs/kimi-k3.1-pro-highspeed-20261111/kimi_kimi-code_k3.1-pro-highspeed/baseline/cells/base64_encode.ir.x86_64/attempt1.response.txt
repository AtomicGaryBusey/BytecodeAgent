export function l $base64_encode(l %src, l %n, l %out) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @empty, @group

@group
	%left =l phi @start %n, @groupdone %leftnext
	%sp =l phi @start %src, @groupdone %spnext
	%op =l phi @start %out, @groupdone %opnext
	%small =w csltl %left, 3
	jnz %small, @partial, @full

@full
	%b0f =l loadub %sp
	%sp1f =l add %sp, 1
	%b1f =l loadub %sp1f
	%sp2f =l add %sp, 2
	%b2f =l loadub %sp2f
	%x0f =l shl %b0f, 16
	%x1f =l shl %b1f, 8
	%xf0 =l or %x0f, %x1f
	%xf =l or %xf0, %b2f
	jmp @havegroup

@partial
	%b0p =l loadub %sp
	%xbase =l shl %b0p, 16
	%two =w ceql %left, 2
	jnz %two, @partial2, @partial1

@partial1
	%xp1 =l copy %xbase
	jmp @havegroup

@partial2
	%sp1p =l add %sp, 1
	%b1p =l loadub %sp1p
	%x1p =l shl %b1p, 8
	%xp2 =l or %xbase, %x1p
	jmp @havegroup

@havegroup
	%x =l phi @full %xf, @partial1 %xp1, @partial2 %xp2
	%rem =l phi @full 3, @partial1 1, @partial2 2
	jmp @emit

@emit
	%j =l phi @havegroup 0, @advance %jnext
	%opcur =l phi @havegroup %op, @advance %opnext
	%xw =l phi @havegroup %x, @advance %xwnext
	%nreal =l add %rem, 1
	%real =w csltl %j, %nreal
	jnz %real, @mapindex, @pad

@pad
	storeb 61, %opcur
	jmp @advance

@mapindex
	%top =l shr %xw, 18
	%v =l and %top, 63
	%isupper =w csltl %v, 26
	jnz %isupper, @mapupper, @notupper

@notupper
	%islower =w csltl %v, 52
	jnz %islower, @maplower, @notlower

@notlower
	%isdigit =w csltl %v, 62
	jnz %isdigit, @mapdigit, @notdigit

@notdigit
	%isplus =w ceql %v, 62
	jnz %isplus, @mapplus, @mapslash

@mapupper
	%cu =l add %v, 65
	jmp @storechar

@maplower
	%cl =l add %v, 71
	jmp @storechar

@mapdigit
	%cd =l sub %v, 4
	jmp @storechar

@mapplus
	%cp =l copy 43
	jmp @storechar

@mapslash
	%cs =l copy 47
	jmp @storechar

@storechar
	%c =l phi @mapupper %cu, @maplower %cl, @mapdigit %cd, @mapplus %cp, @mapslash %cs
	storeb %c, %opcur
	jmp @advance

@advance
	%jnext =l add %j, 1
	%opnext =l add %opcur, 1
	%xwnext =l shl %xw, 6
	%more =w csltl %jnext, 4
	jnz %more, @emit, @groupdone

@groupdone
	%spnext =l add %sp, %rem
	%leftnext =l sub %left, %rem
	%finished =w ceql %leftnext, 0
	jnz %finished, @done, @group

@done
	%written =l sub %opnext, %out
	ret %written

@empty
	ret 0
}
