export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%np1 =l add %n, 1
	jmp @init
@init
	%j =l phi @start 0, @init_body %jnext
	%initcond =w csltl %j, %np1
	jnz %initcond, @init_body, @outer_setup
@init_body
	%joff =l mul %j, 8
	%jaddr =l add %row, %joff
	storel 0, %jaddr
	%jnext =l add %j, 1
	jmp @init
@outer_setup
	jmp @outer
@outer
	%i =l phi @outer_setup 0, @outer_next %inext
	%outercond =w csltl %i, %m
	jnz %outercond, @outer_body, @done
@outer_body
	%xaddr =l add %x, %i
	%xb =w loadub %xaddr
	jmp @inner
@inner
	%k =l phi @outer_body 1, @inner_next %knext
	%diag =l phi @outer_body 0, @inner_next %up
	%innercond =w csltl %k, %np1
	jnz %innercond, @inner_body, @outer_next
@inner_body
	%koff =l mul %k, 8
	%kaddr =l add %row, %koff
	%up =l loadl %kaddr
	%km1 =l sub %k, 1
	%km1off =l mul %km1, 8
	%leftaddr =l add %row, %km1off
	%left =l loadl %leftaddr
	%yaddr =l add %y, %km1
	%yb =w loadub %yaddr
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%matchval =l add %diag, 1
	storel %matchval, %kaddr
	jmp @inner_next
@nomatch
	%leftgreater =w csltl %up, %left
	jnz %leftgreater, @take_left, @take_up
@take_left
	storel %left, %kaddr
	jmp @inner_next
@take_up
	storel %up, %kaddr
	jmp @inner_next
@inner_next
	%knext =l add %k, 1
	jmp @inner
@outer_next
	%inext =l add %i, 1
	jmp @outer
@done
	%noff =l mul %n, 8
	%naddr =l add %row, %noff
	%result =l loadl %naddr
	ret %result
}
