export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@init
	%i =l copy 0
	%count =l copy 0
	%limit =l sub %n, 1
@loop
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%desc =w csgtl %x, %y
	jnz %desc, @increment, @next
@increment
	%count =l add %count, 1
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@zero
	ret 0
@done
	ret %count
}
