export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%n_cur =l phi @start %n, @next %n_next
	%res_cur =l phi @start 0, @next %res_next
	%bit_cur =l phi @start 4611686018427387904, @next %bit_next
	%done =w ceql %bit_cur, 0
	jnz %done, @end, @body
@body
	%sum =l add %res_cur, %bit_cur
	%too_large =w cultl %n_cur, %sum
	jnz %too_large, @else, @then
@then
	%n_then =l sub %n_cur, %sum
	%half_then =l shr %res_cur, 1
	%res_then =l add %half_then, %bit_cur
	jmp @next
@else
	%res_else =l shr %res_cur, 1
	jmp @next
@next
	%n_next =l phi @then %n_then, @else %n_cur
	%res_next =l phi @then %res_then, @else %res_else
	%bit_next =l shr %bit_cur, 2
	jmp @loop
@end
	ret %res_cur
}
