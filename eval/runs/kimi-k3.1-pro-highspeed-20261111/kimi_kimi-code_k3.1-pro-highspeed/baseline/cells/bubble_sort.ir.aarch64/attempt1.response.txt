export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%limit0 =l sub %n, 1
	jmp @outer
@outer
	%limit =l phi @init %limit0, @pass_done %limit_next
	%has_work =w csltl 0, %limit
	jnz %has_work, @pass_init, @done
@pass_init
	%j0 =l copy 0
	jmp @inner
@inner
	%j =l phi @pass_init %j0, @next %j_next
	%more =w csltl %j, %limit
	jnz %more, @body, @pass_done
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%gt =w csltl %y, %x
	jnz %gt, @swap, @next
@swap
	storel %y, %p
	storel %x, %q
	jmp @next
@next
	%j_next =l add %j, 1
	jmp @inner
@pass_done
	%limit_next =l sub %limit, 1
	jmp @outer
@done
	ret 0
}
