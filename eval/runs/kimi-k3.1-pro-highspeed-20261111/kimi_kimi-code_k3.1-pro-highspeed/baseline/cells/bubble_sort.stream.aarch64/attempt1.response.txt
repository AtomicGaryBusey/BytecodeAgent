bubble_sort:
cmp x1, #2
b.lo .Ldone
.Louter:
sub x3, x1, #1
mov x2, #0
.Linner:
ldr x4, [x0, x2, lsl #3]
add x5, x2, #1
ldr x5, [x0, x5, lsl #3]
cmp x4, x5
b.le .Lnoswap
str x5, [x0, x2, lsl #3]
add x6, x2, #1
str x4, [x0, x6, lsl #3]
.Lnoswap:
add x2, x2, #1
cmp x2, x3
b.lo .Linner
sub x1, x1, #1
cmp x1, #2
b.hs .Louter
.Ldone:
mov x0, #0
ret
