dot_i32_simd:
pxor xmm4, xmm4
pxor xmm5, xmm5
xor r8d, r8d
mov rcx, rdx
cmp rcx, 4
jl .Ltail
.Lloop:
movdqu xmm0, xmmword ptr [rdi]
movdqu xmm1, xmmword ptr [rsi]
movdqa xmm2, xmm0
movdqa xmm3, xmm1
pmovsxdq xmm0, xmm0
pmovsxdq xmm1, xmm1
pmuldq xmm0, xmm1
paddq xmm4, xmm0
psrldq xmm2, 8
psrldq xmm3, 8
pmovsxdq xmm2, xmm2
pmovsxdq xmm3, xmm3
pmuldq xmm2, xmm3
paddq xmm5, xmm2
add rdi, 16
add rsi, 16
sub rcx, 4
cmp rcx, 4
jge .Lloop
.Ltail:
test rcx, rcx
jz .Lreduce
.Ltail_loop:
movsxd r9, dword ptr [rdi]
movsxd r10, dword ptr [rsi]
imul r9, r10
add r8, r9
add rdi, 4
add rsi, 4
dec rcx
jnz .Ltail_loop
.Lreduce:
paddq xmm4, xmm5
movdqa xmm0, xmm4
psrldq xmm0, 8
paddq xmm4, xmm0
movq rax, xmm4
add rax, r8
ret
