packbits_decode:
mov r8, rdi
lea r9, [rdi + rsi]
mov r10, rdx
lea r11, [rdx + rcx]
xor eax, eax
.Lpacket:
cmp r8, r9
je .Ldone
movzx ecx, byte ptr [r8]
inc r8
test cl, cl
js .Lnegative
inc ecx
mov rsi, r9
sub rsi, r8
cmp rsi, rcx
jb .Lsrc_error
mov rsi, r11
sub rsi, r10
cmp rsi, rcx
jb .Lliteral_partial
add rax, rcx
.Lliteral_loop:
mov sil, byte ptr [r8]
mov byte ptr [r10], sil
inc r8
inc r10
dec rcx
jnz .Lliteral_loop
jmp .Lpacket
.Lliteral_partial:
test rsi, rsi
jz .Loverflow
.Lliteral_partial_loop:
mov dl, byte ptr [r8]
mov byte ptr [r10], dl
inc r8
inc r10
dec rsi
jnz .Lliteral_partial_loop
jmp .Loverflow
.Lnegative:
cmp cl, -128
je .Lpacket
cmp r8, r9
je .Lsrc_error
mov edx, 257
sub edx, ecx
mov ecx, edx
movzx esi, byte ptr [r8]
inc r8
mov rdx, r11
sub rdx, r10
cmp rdx, rcx
jb .Lrepeat_partial
add rax, rcx
.Lrepeat_loop:
mov byte ptr [r10], sil
inc r10
dec rcx
jnz .Lrepeat_loop
jmp .Lpacket
.Lrepeat_partial:
test rdx, rdx
jz .Loverflow
.Lrepeat_partial_loop:
mov byte ptr [r10], sil
inc r10
dec rdx
jnz .Lrepeat_partial_loop
.Loverflow:
mov rax, -1
ret
.Lsrc_error:
mov rax, -2
ret
.Ldone:
ret
