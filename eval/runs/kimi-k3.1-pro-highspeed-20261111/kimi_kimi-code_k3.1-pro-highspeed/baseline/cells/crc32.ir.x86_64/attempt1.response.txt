export function l $crc32(l %p, l %n) {
@entry
	jmp @outer
@outer
	%i =l phi @entry 0, @next %i1
	%crc =w phi @entry 4294967295, @next %crc1
	%done =w ceql %i, %n
	jnz %done, @finish, @byte
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc0 =w xor %crc, %b
	jmp @bits
@bits
	%k =w phi @byte 0, @bit %k1
	%c =w phi @byte %crc0, @bit %c1
	%bitsdone =w ceqw %k, 8
	jnz %bitsdone, @next, @bit
@bit
	%lsb =w and %c, 1
	%mask0 =w sub 0, %lsb
	%mask =w and %mask0, 3988292384
	%shifted =w shr %c, 1
	%c1 =w xor %shifted, %mask
	%k1 =w add %k, 1
	jmp @bits
@next
	%crc1 =w copy %c
	%i1 =l add %i, 1
	jmp @outer
@finish
	%final =w xor %crc, 4294967295
	%res =l extuw %final
	ret %res
}
