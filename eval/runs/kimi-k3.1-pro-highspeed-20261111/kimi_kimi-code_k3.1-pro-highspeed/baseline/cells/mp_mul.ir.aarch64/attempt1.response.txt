export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%empty =w ceql %n, 0
	jnz %empty, @return_zero, @zero_start
@zero_start
	jmp @zero_head
@zero_head
	%zi =l phi @zero_start 0, @zero_body %zinext
	%zdone =w ceql %zi, %n
	jnz %zdone, @check_empty, @zero_body
@zero_body
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zinext =l add %zi, 1
	jmp @zero_head
@check_empty
	%aempty =w ceql %na, 0
	jnz %aempty, @trim_start, @check_b
@check_b
	%bempty =w ceql %nb, 0
	jnz %bempty, @trim_start, @outer_start
@outer_start
	jmp @outer_head
@outer_head
	%i =l phi @outer_start 0, @inner_done %inext
	%idone =w ceql %i, %na
	jnz %idone, @trim_start, @outer_body
@outer_body
	%aoff =l mul %i, 8
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	jmp @inner_head
@inner_head
	%j =l phi @outer_body 0, @inner_body %jnext
	%carry =l phi @outer_body 0, @inner_body %carrynext
	%jdone =w ceql %j, %nb
	jnz %jdone, @inner_done, @inner_body
@inner_body
	%joff =l mul %j, 8
	%bp =l add %b, %joff
	%bj =l loadl %bp
	%ij =l add %i, %j
	%rijoff =l mul %ij, 8
	%rp =l add %r, %rijoff
	%old =l loadl %rp
	%a0 =l and %ai, 4294967295
	%a1 =l shr %ai, 32
	%b0 =l and %bj, 4294967295
	%b1 =l shr %bj, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%t =l add %p10, %p00hi
	%tlo =l and %t, 4294967295
	%thi =l shr %t, 32
	%w1 =l add %p01, %tlo
	%w1hi =l shr %w1, 32
	%hi0 =l add %p11, %thi
	%hi =l add %hi0, %w1hi
	%w1shift =l shl %w1, 32
	%p00lo =l and %p00, 4294967295
	%lo =l add %w1shift, %p00lo
	%sum0 =l add %old, %lo
	%c0 =w cultl %sum0, %old
	%c0l =l extuw %c0
	%sum1 =l add %sum0, %carry
	%c1 =w cultl %sum1, %carry
	%c1l =l extuw %c1
	%carry0 =l add %hi, %c0l
	%carrynext =l add %carry0, %c1l
	storel %sum1, %rp
	%jnext =l add %j, 1
	jmp @inner_head
@inner_done
	%endindex =l add %i, %nb
	%endoff =l mul %endindex, 8
	%endp =l add %r, %endoff
	storel %carry, %endp
	%inext =l add %i, 1
	jmp @outer_head
@trim_start
	jmp @trim_head
@trim_head
	%k =l phi @trim_start %n, @trim_dec %km1
	%kzero =w ceql %k, 0
	jnz %kzero, @trim_return, @trim_check
@trim_check
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kp =l add %r, %koff
	%limb =l loadl %kp
	%limbzero =w ceql %limb, 0
	jnz %limbzero, @trim_dec, @trim_return
@trim_dec
	jmp @trim_head
@trim_return
	ret %k
@return_zero
	ret 0
}
