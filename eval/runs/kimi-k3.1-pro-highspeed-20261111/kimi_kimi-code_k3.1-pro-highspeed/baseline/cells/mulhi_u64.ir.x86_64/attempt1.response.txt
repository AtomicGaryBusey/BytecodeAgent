export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%k0 =l shr %p00, 32
	%p10 =l mul %a1, %b0
	%t0 =l add %p10, %k0
	%w1 =l and %t0, 4294967295
	%w2 =l shr %t0, 32
	%p01 =l mul %a0, %b1
	%t1 =l add %p01, %w1
	%k1 =l shr %t1, 32
	%p11 =l mul %a1, %b1
	%sum =l add %p11, %w2
	%hi =l add %sum, %k1
	ret %hi
}
