export function l $fill_down(l %a, l %n, l %start) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %next
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%value =l sub %start, %i
	%offset =l mul %i, 8
	%p =l add %a, %offset
	storel %value, %p
	%next =l add %i, 1
	jmp @loop
@done
	ret %n
}
