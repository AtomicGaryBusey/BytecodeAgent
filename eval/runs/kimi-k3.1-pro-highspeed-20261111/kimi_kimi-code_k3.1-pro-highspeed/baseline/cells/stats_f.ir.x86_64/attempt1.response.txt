export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @cont %inext
	%min =l phi @start %y0, @cont %min2
	%max =l phi @start %y0, @cont %max2
	%sum =l phi @start %y0, @cont %sum2
	%count =l phi @start %odd0, @cont %count2
	%more =w csltl %i, %n
	jnz %more, @body, @finish
@body
	%offset =l mul %i, 8
	%ap =l add %a, %offset
	%x =l loadl %ap
	%y =l call %f(l %x)
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%count2 =l add %count, %odd
	%inext =l add %i, 1
	%ismin =w csltl %y, %min
	jnz %ismin, @setmin, @checkmax
@setmin
	jmp @checkmax
@checkmax
	%min2 =l phi @body %min, @setmin %y
	%ismax =w csltl %max, %y
	jnz %ismax, @setmax, @cont
@setmax
	jmp @cont
@cont
	%max2 =l phi @checkmax %max, @setmax %y
	jmp @loop
@finish
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
