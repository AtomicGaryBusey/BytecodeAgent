export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @done, @setup
@done
	ret 0
@setup
	%last =l sub %n, 1
	%lastoff =l mul %last, 16
	%pivot =l add %a, %lastoff
	%pivotvp =l add %pivot, 8
	%pk =l loadl %pivot
	%pv =l loadl %pivotvp
	jmp @loop
@loop
	%i =l phi @setup 0, @advance %inext
	%j =l phi @setup 0, @advance %jnext
	%finished =w ceql %j, %last
	jnz %finished, @partitioned, @inspect
@inspect
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jvp =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jvp
	%keyless =w csltl %jk, %pk
	jnz %keyless, @take, @equalkey
@equalkey
	%keyeq =w ceql %jk, %pk
	jnz %keyeq, @comparevalue, @skip
@comparevalue
	%valueless =w csltl %jv, %pv
	jnz %valueless, @take, @skip
@take
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ivp =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%itake =l add %i, 1
	jmp @advance
@skip
	jmp @advance
@advance
	%inext =l phi @take %itake, @skip %i
	%jnext =l add %j, 1
	jmp @loop
@partitioned
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ivp2 =l add %ip2, 8
	%ik2 =l loadl %ip2
	%iv2 =l loadl %ivp2
	storel %pk, %ip2
	storel %pv, %ivp2
	storel %ik2, %pivot
	storel %iv2, %pivotvp
	%rl =l call $qsort_pairs(l %a, l %i)
	%rightindex =l add %i, 1
	%rightn =l sub %n, %rightindex
	%rightoff =l mul %rightindex, 16
	%rightp =l add %a, %rightoff
	%rr =l call $qsort_pairs(l %rightp, l %rightn)
	ret 0
}
