mp_mul:
push rbx
push r12
push r13
mov r12, rdx
mov r13, rcx
lea r9, [rsi + rcx]
xor eax, eax
.Lzero_loop:
cmp rax, r9
jae .Lzero_done
mov qword ptr [r8 + rax*8], 0
inc rax
jmp .Lzero_loop
.Lzero_done:
xor r10d, r10d
.Louter_loop:
cmp r10, rsi
jae .Lsignificant
xor r11d, r11d
xor ebx, ebx
.Linner_loop:
cmp r11, r13
jae .Linner_done
mov rax, qword ptr [rdi + r10*8]
mul qword ptr [r12 + r11*8]
add rax, rbx
adc rdx, 0
lea rcx, [r10 + r11]
add rax, qword ptr [r8 + rcx*8]
adc rdx, 0
mov qword ptr [r8 + rcx*8], rax
mov rbx, rdx
inc r11
jmp .Linner_loop
.Linner_done:
lea rcx, [r10 + r13]
mov qword ptr [r8 + rcx*8], rbx
inc r10
jmp .Louter_loop
.Lsignificant:
mov rax, r9
.Lsignificant_loop:
test rax, rax
jz .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lsignificant_loop
.Ldone:
pop r13
pop r12
pop rbx
ret
