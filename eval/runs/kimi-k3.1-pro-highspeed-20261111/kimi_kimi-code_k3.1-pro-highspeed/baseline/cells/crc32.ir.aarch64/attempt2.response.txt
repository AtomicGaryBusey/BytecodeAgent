export function l $crc32(l %p, l %n) {
@entry
	%init =w copy -1
	%izero =l copy 0
	jmp @outer
@outer
	%crc =w phi @entry %init, @advance %crcbyte
	%i =l phi @entry %izero, @advance %inext
	%have =w csltl %i, %n
	jnz %have, @startbyte, @done
@startbyte
	%addr =l add %p, %i
	%byte =w loadub %addr
	%x =w xor %crc, %byte
	%jzero =w copy 0
	jmp @inner
@inner
	%c =w phi @startbyte %x, @step %cnext
	%j =w phi @startbyte %jzero, @step %jnext
	%low =w and %c, 1
	%mask =w sub 0, %low
	%poly =w and %mask, -306674912
	%half =w shr %c, 1
	%cnext =w xor %half, %poly
	jmp @step
@step
	%jnext =w add %j, 1
	%again =w csltw %jnext, 8
	jnz %again, @inner, @advance
@advance
	%crcbyte =w copy %cnext
	%inext =l add %i, 1
	jmp @outer
@done
	%resultw =w xor %crc, -1
	%result =l extuw %resultw
	ret %result
}
