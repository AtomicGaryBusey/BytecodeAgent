export function l $max_byte(l %p, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @init
@init
	%i0 =l copy 0
	%max0 =l copy 0
	jmp @loop
@loop
	%i =l phi @init %i0, @next %i1
	%max =l phi @init %max0, @next %max1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w csltl %max, %b
	jnz %greater, @update, @keep
@update
	%newmax =l copy %b
	jmp @next
@keep
	%oldmax =l copy %max
	jmp @next
@next
	%max1 =l phi @update %newmax, @keep %oldmax
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %max
@zero
	ret 0
}
