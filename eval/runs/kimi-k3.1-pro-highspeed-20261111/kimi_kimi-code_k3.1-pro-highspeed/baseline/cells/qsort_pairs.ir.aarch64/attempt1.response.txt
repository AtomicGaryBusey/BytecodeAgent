export function l $qsort_pairs(l %a, l %n) {
@entry
    %islot =l alloc8 8
    %jslot =l alloc8 8
    %small =w cslel %n, 1
    jnz %small, @return, @setup
@setup
    storel 0, %islot
    storel 0, %jslot
    %last =l sub %n, 1
    %pivotoffset =l mul %last, 16
    %pivot =l add %a, %pivotoffset
    %pivotkey =l loadl %pivot
    %pivotvalueaddress =l add %pivot, 8
    %pivotvalue =l loadl %pivotvalueaddress
    jmp @test
@test
    %j =l loadl %jslot
    %more =w csltl %j, %last
    jnz %more, @body, @finish
@body
    %joffset =l mul %j, 16
    %jrecord =l add %a, %joffset
    %jkey =l loadl %jrecord
    %keyless =w csltl %jkey, %pivotkey
    jnz %keyless, @swap, @keynotless
@keynotless
    %keyeEqual =w ceql %jkey, %pivotkey
    jnz %keyeEqual, @comparevalue, @advance
@comparevalue
    %jvalueaddress =l add %jrecord, 8
    %jvalue =l loadl %jvalueaddress
    %valueless =w csltl %jvalue, %pivotvalue
    jnz %valueless, @swap, @advance
@swap
    %i =l loadl %islot
    %ioffset =l mul %i, 16
    %irecord =l add %a, %ioffset
    %ivalueaddress =l add %irecord, 8
    %ikey =l loadl %irecord
    %ivalue =l loadl %ivalueaddress
    %swapjvalueaddress =l add %jrecord, 8
    %swapjvalue =l loadl %swapjvalueaddress
    storel %jkey, %irecord
    storel %swapjvalue, %ivalueaddress
    storel %ikey, %jrecord
    storel %ivalue, %swapjvalueaddress
    %inext =l add %i, 1
    storel %inext, %islot
    jmp @advance
@advance
    %jnext =l add %j, 1
    storel %jnext, %jslot
    jmp @test
@finish
    %ifinal =l loadl %islot
    %ifinaloffset =l mul %ifinal, 16
    %ifinalrecord =l add %a, %ifinaloffset
    %ifinalvalueaddress =l add %ifinalrecord, 8
    %ifinalkey =l loadl %ifinalrecord
    %ifinalvalue =l loadl %ifinalvalueaddress
    storel %pivotkey, %ifinalrecord
    storel %pivotvalue, %ifinalvalueaddress
    storel %ifinalkey, %pivot
    storel %ifinalvalue, %pivotvalueaddress
    %rightpointer =l add %ifinalrecord, 16
    %rightcount =l sub %last, %ifinal
    %leftresult =l call $qsort_pairs(l %a, l %ifinal)
    %rightresult =l call $qsort_pairs(l %rightpointer, l %rightcount)
    jmp @return
@return
    ret 0
}
