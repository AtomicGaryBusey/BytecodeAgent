export function l $count_ugt(l %a, l %n, l %t) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %inext
	%count =l phi @entry 0, @body %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w cugtl %value, %t
	%increment =l extuw %greater
	%countnext =l add %count, %increment
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
