export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%p =l phi @start %src, @group_done %pnext
	%r =l phi @start %n, @group_done %rnext
	%o =l phi @start %out, @group_done %onext_group
	%small =w csltl %r, 3
	jnz %small, @tail, @group
@group
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%b0s =l shl %b0, 16
	%b1s =l shl %b1, 8
	%x01 =l or %b0s, %b1s
	%value =l or %x01, %b2
	jmp @emit
@emit
	%i =l phi @group 0, @emit_next %inext
	%sh =l phi @group 18, @emit_next %shnext
	%oe =l phi @group %o, @emit_next %oenext
	%shifted =l shr %value, %sh
	%v =l and %shifted, 63
	%lt26 =w csltl %v, 26
	%ge26w =w sub 1, %lt26
	%lt52 =w csltl %v, 52
	%ge52w =w sub 1, %lt52
	%eq62 =w ceql %v, 62
	%eq63 =w ceql %v, 63
	%c0 =l add %v, 65
	%ge26 =l extsw %ge26w
	%d26 =l mul %ge26, 6
	%c1 =l add %c0, %d26
	%ge52 =l extsw %ge52w
	%d52 =l mul %ge52, 75
	%c2 =l sub %c1, %d52
	%eq62l =l extsw %eq62
	%d62 =l mul %eq62l, 15
	%c3 =l sub %c2, %d62
	%eq63l =l extsw %eq63
	%d63 =l mul %eq63l, 12
	%c4 =l sub %c3, %d63
	storeb %c4, %oe
	jmp @emit_next
@emit_next
	%oenext =l add %oe, 1
	%inext =l add %i, 1
	%shnext =l sub %sh, 6
	%more =w csltl %inext, 4
	jnz %more, @emit, @group_done
@group_done
	%pnext =l add %p, 3
	%rnext =l sub %r, 3
	%onext_group =l add %o, 4
	jmp @loop
@tail
	%empty =w ceql %r, 0
	jnz %empty, @return, @tail_nonzero
@tail_nonzero
	%tb0 =l loadub %p
	%tb0s =l shl %tb0, 16
	%two =w ceql %r, 2
	jnz %two, @tail_two, @tail_one
@tail_two
	%tp1 =l add %p, 1
	%tb1 =l loadub %tp1
	%tb1s =l shl %tb1, 8
	%tvalue2 =l or %tb0s, %tb1s
	jmp @tail_setup
@tail_one
	jmp @tail_setup
@tail_setup
	%tvalue =l phi @tail_two %tvalue2, @tail_one %tb0s
	%chars =l phi @tail_two 3, @tail_one 2
	jmp @tail_emit
@tail_emit
	%ti =l phi @tail_setup 0, @tail_next %tinext
	%tsh =l phi @tail_setup 18, @tail_next %tshnext
	%toe =l phi @tail_setup %o, @tail_next %toenext
	%isdata =w csltl %ti, %chars
	jnz %isdata, @tail_data, @tail_pad
@tail_data
	%tshifted =l shr %tvalue, %tsh
	%tv =l and %tshifted, 63
	%tlt26 =w csltl %tv, 26
	%tge26w =w sub 1, %tlt26
	%tlt52 =w csltl %tv, 52
	%tge52w =w sub 1, %tlt52
	%teq62 =w ceql %tv, 62
	%teq63 =w ceql %tv, 63
	%tc0 =l add %tv, 65
	%tge26 =l extsw %tge26w
	%td26 =l mul %tge26, 6
	%tc1 =l add %tc0, %td26
	%tge52 =l extsw %tge52w
	%td52 =l mul %tge52, 75
	%tc2 =l sub %tc1, %td52
	%teq62l =l extsw %teq62
	%td62 =l mul %teq62l, 15
	%tc3 =l sub %tc2, %td62
	%teq63l =l extsw %teq63
	%td63 =l mul %teq63l, 12
	%tc4 =l sub %tc3, %td63
	storeb %tc4, %toe
	jmp @tail_next
@tail_pad
	storeb 61, %toe
	jmp @tail_next
@tail_next
	%toenext =l add %toe, 1
	%tinext =l add %ti, 1
	%tshnext =l sub %tsh, 6
	%tmore =w csltl %tinext, 4
	jnz %tmore, @tail_emit, @tail_finish
@tail_finish
	%len1 =l sub %toenext, %out
	ret %len1
@return
	%len0 =l sub %o, %out
	ret %len0
}
