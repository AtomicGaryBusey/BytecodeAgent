export function l $collatz_max(l %n) {
@entry
    %kp =l alloc8 8
    %maxp =l alloc8 8
    %xp =l alloc8 8
    %sp =l alloc8 8
    storel 1, %kp
    storel 0, %maxp
    jmp @outer
@outer
    %k =l loadl %kp
    %finished =w csltl %n, %k
    jnz %finished, @done, @init
@init
    storel %k, %xp
    storel 0, %sp
    jmp @inner
@inner
    %x =l loadl %xp
    %isone =w ceql %x, 1
    jnz %isone, @consider, @test
@test
    %r =l and %x, 1
    %iseven =w ceql %r, 0
    jnz %iseven, @even, @odd
@even
    %xe =l div %x, 2
    storel %xe, %xp
    %se =l loadl %sp
    %se1 =l add %se, 1
    storel %se1, %sp
    jmp @inner
@odd
    %x3 =l mul %x, 3
    %xo =l add %x3, 1
    storel %xo, %xp
    %so =l loadl %sp
    %so1 =l add %so, 1
    storel %so1, %sp
    jmp @inner
@consider
    %steps =l loadl %sp
    %max =l loadl %maxp
    %better =w csltl %max, %steps
    jnz %better, @update, @next
@update
    storel %steps, %maxp
    jmp @next
@next
    %k1 =l add %k, 1
    storel %k1, %kp
    jmp @outer
@done
    %result =l loadl %maxp
    ret %result
}
