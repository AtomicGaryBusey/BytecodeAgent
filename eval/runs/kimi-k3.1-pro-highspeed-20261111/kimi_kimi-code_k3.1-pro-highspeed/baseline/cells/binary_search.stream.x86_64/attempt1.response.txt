binary_search:
    xor eax, eax
.Lloop:
    cmp rax, rsi
    jae .Lnot_found
    mov rcx, rsi
    sub rcx, rax
    shr rcx, 1
    add rcx, rax
    mov r8, qword ptr [rdi + rcx*8]
    cmp rdx, r8
    je .Lfound
    jl .Llower
    lea rax, [rcx + 1]
    jmp .Lloop
.Llower:
    mov rsi, rcx
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lnot_found:
    mov rax, -1
    ret
