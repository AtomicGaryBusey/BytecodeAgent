export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @push %pc_push, @add %pc_add, @sub %pc_sub, @mul %pc_mul, @dup %pc_dup, @swap %pc_swap, @drop %pc_drop, @over %pc_over, @lt %pc_lt, @jz_taken %jz_target, @jz_fall %pc_jz_fall, @jump %jump_target, @default %pc_default
	%depth =l phi @start 0, @push %depth_push, @add %depth_add, @sub %depth_sub, @mul %depth_mul, @dup %depth_dup, @swap %depth, @drop %depth_drop, @over %depth_over, @lt %depth_lt, @jz_taken %depth_jz, @jz_fall %depth_jz, @jump %depth, @default %depth
	%at_end =w ceql %pc, %len
	jnz %at_end, @finish, @fetch
@fetch
	%pc_offset =l mul %pc, 8
	%instruction_ptr =l add %code, %pc_offset
	%op =l loadl %instruction_ptr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @halt, @test_push
@test_push
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @test_add
@test_add
	%is_add =w ceql %op, 2
	jnz %is_add, @add, @test_sub
@test_sub
	%is_sub =w ceql %op, 3
	jnz %is_sub, @sub, @test_mul
@test_mul
	%is_mul =w ceql %op, 4
	jnz %is_mul, @mul, @test_dup
@test_dup
	%is_dup =w ceql %op, 5
	jnz %is_dup, @dup, @test_swap
@test_swap
	%is_swap =w ceql %op, 6
	jnz %is_swap, @swap, @test_drop
@test_drop
	%is_drop =w ceql %op, 7
	jnz %is_drop, @drop, @test_over
@test_over
	%is_over =w ceql %op, 8
	jnz %is_over, @over, @test_lt
@test_lt
	%is_lt =w ceql %op, 9
	jnz %is_lt, @lt, @test_jz
@test_jz
	%is_jz =w ceql %op, 10
	jnz %is_jz, @jz, @test_jmp
@test_jmp
	%is_jmp =w ceql %op, 11
	jnz %is_jmp, @jump, @default
@push
	%push_operand_pc =l add %pc, 1
	%push_operand_offset =l mul %push_operand_pc, 8
	%push_operand_ptr =l add %code, %push_operand_offset
	%push_value =l loadl %push_operand_ptr
	%push_stack_offset =l mul %depth, 8
	%push_stack_ptr =l add %stack, %push_stack_offset
	storel %push_value, %push_stack_ptr
	%pc_push =l add %pc, 2
	%depth_push =l add %depth, 1
	jmp @loop
@add
	%add_b_index =l sub %depth, 1
	%add_a_index =l sub %depth, 2
	%add_b_offset =l mul %add_b_index, 8
	%add_a_offset =l mul %add_a_index, 8
	%add_b_ptr =l add %stack, %add_b_offset
	%add_a_ptr =l add %stack, %add_a_offset
	%add_b =l loadl %add_b_ptr
	%add_a =l loadl %add_a_ptr
	%add_value =l add %add_a, %add_b
	storel %add_value, %add_a_ptr
	%pc_add =l add %pc, 1
	%depth_add =l sub %depth, 1
	jmp @loop
@sub
	%sub_b_index =l sub %depth, 1
	%sub_a_index =l sub %depth, 2
	%sub_b_offset =l mul %sub_b_index, 8
	%sub_a_offset =l mul %sub_a_index, 8
	%sub_b_ptr =l add %stack, %sub_b_offset
	%sub_a_ptr =l add %stack, %sub_a_offset
	%sub_b =l loadl %sub_b_ptr
	%sub_a =l loadl %sub_a_ptr
	%sub_value =l sub %sub_a, %sub_b
	storel %sub_value, %sub_a_ptr
	%pc_sub =l add %pc, 1
	%depth_sub =l sub %depth, 1
	jmp @loop
@mul
	%mul_b_index =l sub %depth, 1
	%mul_a_index =l sub %depth, 2
	%mul_b_offset =l mul %mul_b_index, 8
	%mul_a_offset =l mul %mul_a_index, 8
	%mul_b_ptr =l add %stack, %mul_b_offset
	%mul_a_ptr =l add %stack, %mul_a_offset
	%mul_b =l loadl %mul_b_ptr
	%mul_a =l loadl %mul_a_ptr
	%mul_value =l mul %mul_a, %mul_b
	storel %mul_value, %mul_a_ptr
	%pc_mul =l add %pc, 1
	%depth_mul =l sub %depth, 1
	jmp @loop
@dup
	%dup_source_index =l sub %depth, 1
	%dup_source_offset =l mul %dup_source_index, 8
	%dup_destination_offset =l mul %depth, 8
	%dup_source_ptr =l add %stack, %dup_source_offset
	%dup_destination_ptr =l add %stack, %dup_destination_offset
	%dup_value =l loadl %dup_source_ptr
	storel %dup_value, %dup_destination_ptr
	%pc_dup =l add %pc, 1
	%depth_dup =l add %depth, 1
	jmp @loop
@swap
	%swap_b_index =l sub %depth, 1
	%swap_a_index =l sub %depth, 2
	%swap_b_offset =l mul %swap_b_index, 8
	%swap_a_offset =l mul %swap_a_index, 8
	%swap_b_ptr =l add %stack, %swap_b_offset
	%swap_a_ptr =l add %stack, %swap_a_offset
	%swap_b =l loadl %swap_b_ptr
	%swap_a =l loadl %swap_a_ptr
	storel %swap_a, %swap_b_ptr
	storel %swap_b, %swap_a_ptr
	%pc_swap =l add %pc, 1
	jmp @loop
@drop
	%depth_drop =l sub %depth, 1
	%pc_drop =l add %pc, 1
	jmp @loop
@over
	%over_source_index =l sub %depth, 2
	%over_source_offset =l mul %over_source_index, 8
	%over_destination_offset =l mul %depth, 8
	%over_source_ptr =l add %stack, %over_source_offset
	%over_destination_ptr =l add %stack, %over_destination_offset
	%over_value =l loadl %over_source_ptr
	storel %over_value, %over_destination_ptr
	%pc_over =l add %pc, 1
	%depth_over =l add %depth, 1
	jmp @loop
@lt
	%lt_b_index =l sub %depth, 1
	%lt_a_index =l sub %depth, 2
	%lt_b_offset =l mul %lt_b_index, 8
	%lt_a_offset =l mul %lt_a_index, 8
	%lt_b_ptr =l add %stack, %lt_b_offset
	%lt_a_ptr =l add %stack, %lt_a_offset
	%lt_b =l loadl %lt_b_ptr
	%lt_a =l loadl %lt_a_ptr
	%lt_comparison =w csltl %lt_a, %lt_b
	%lt_value =l extuw %lt_comparison
	storel %lt_value, %lt_a_ptr
	%pc_lt =l add %pc, 1
	%depth_lt =l sub %depth, 1
	jmp @loop
@jz
	%jz_value_index =l sub %depth, 1
	%jz_value_offset =l mul %jz_value_index, 8
	%jz_value_ptr =l add %stack, %jz_value_offset
	%jz_value =l loadl %jz_value_ptr
	%jz_operand_pc =l add %pc, 1
	%jz_operand_offset =l mul %jz_operand_pc, 8
	%jz_operand_ptr =l add %code, %jz_operand_offset
	%jz_target =l loadl %jz_operand_ptr
	%depth_jz =l sub %depth, 1
	%jz_is_zero =w ceql %jz_value, 0
	jnz %jz_is_zero, @jz_taken, @jz_fall
@jz_taken
	jmp @loop
@jz_fall
	%pc_jz_fall =l add %pc, 2
	jmp @loop
@jump
	%jump_operand_pc =l add %pc, 1
	%jump_operand_offset =l mul %jump_operand_pc, 8
	%jump_operand_ptr =l add %code, %jump_operand_offset
	%jump_target =l loadl %jump_operand_ptr
	jmp @loop
@default
	%pc_default =l add %pc, 1
	jmp @loop
@halt
	jmp @finish
@finish
	%final_depth =l phi @loop %depth, @halt %depth
	%stack_empty =w ceql %final_depth, 0
	jnz %stack_empty, @empty, @nonempty
@empty
	jmp @zero_entry
@nonempty
	%top_index =l sub %final_depth, 1
	%top_offset =l mul %top_index, 8
	%top_ptr =l add %stack, %top_offset
	%top_value =l loadl %top_ptr
	jmp @zero_entry
@zero_entry
	%result =l phi @empty 0, @nonempty %top_value
	jmp @zero_loop
@zero_loop
	%zero_index =l phi @zero_entry %final_depth, @zero_body %zero_next
	%zero_done =w ceql %zero_index, %cap
	jnz %zero_done, @return, @zero_body
@zero_body
	%zero_offset =l mul %zero_index, 8
	%zero_ptr =l add %stack, %zero_offset
	storel 0, %zero_ptr
	%zero_next =l add %zero_index, 1
	jmp @zero_loop
@return
	ret %result
}
