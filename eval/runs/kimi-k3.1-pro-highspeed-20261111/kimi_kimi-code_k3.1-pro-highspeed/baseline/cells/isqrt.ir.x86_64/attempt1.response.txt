export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%r =l phi @start 0, @next %rn
	%rem =l phi @start %n, @next %remn
	%bit =l phi @start 4611686018427387904, @next %bitn
	%done =w ceql %bit, 0
	jnz %done, @end, @body
@body
	%sum =l add %r, %bit
	%take =w culel %sum, %rem
	jnz %take, @taken, @skipped
@taken
	%remt =l sub %rem, %sum
	%halft =l shr %r, 1
	%rt =l add %halft, %bit
	jmp @next
@skipped
	%rs =l shr %r, 1
	jmp @next
@next
	%rn =l phi @taken %rt, @skipped %rs
	%remn =l phi @taken %remt, @skipped %rem
	%bitn =l shr %bit, 2
	jmp @loop
@end
	ret %r
}
