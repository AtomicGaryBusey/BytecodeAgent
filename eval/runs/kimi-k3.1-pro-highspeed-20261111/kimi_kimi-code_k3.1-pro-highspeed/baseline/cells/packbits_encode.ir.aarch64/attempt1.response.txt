export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%rslot =l alloc4 4
	%jslot =l alloc8 8
	%kslot =l alloc8 8
	storel 0, %islot
	storel 0, %oslot
	jmp @outer

@outer
	%i0 =l loadl %islot
	%more =w csltl %i0, %n
	jnz %more, @begin, @success

@begin
	%ip =l add %src, %i0
	%b =w loadub %ip
	storew 1, %rslot
	%i_plus_1 =l add %i0, 1
	storel %i_plus_1, %jslot
	jmp @run_cond

@run_cond
	%rj =l loadl %jslot
	%rj_in =w csltl %rj, %n
	jnz %rj_in, @run_cap, @run_done

@run_cap
	%rr =w loadw %rslot
	%rr_can =w csltw %rr, 128
	jnz %rr_can, @run_byte, @run_done

@run_byte
	%rbp =l add %src, %rj
	%rb =w loadub %rbp
	%rsame =w ceqw %rb, %b
	jnz %rsame, @run_inc, @run_done

@run_inc
	%rr1 =w add %rr, 1
	%rj1 =l add %rj, 1
	storew %rr1, %rslot
	storel %rj1, %jslot
	jmp @run_cond

@run_done
	%r_final =w loadw %rslot
	%not_repeat =w csltw %r_final, 3
	jnz %not_repeat, @literal_init, @repeat_check

@repeat_check
	%rep_outpos =l loadl %oslot
	%rep_room =l sub %cap, %rep_outpos
	%rep_too_big =w csltl %rep_room, 2
	jnz %rep_too_big, @fail, @repeat_write

@repeat_write
	%rep_header =w sub 257, %r_final
	%rep_dest =l add %out, %rep_outpos
	storeb %rep_header, %rep_dest
	%rep_value_dest =l add %rep_dest, 1
	storeb %b, %rep_value_dest
	%rep_next_out =l add %rep_outpos, 2
	storel %rep_next_out, %oslot
	%rep_next_i =l loadl %jslot
	storel %rep_next_i, %islot
	jmp @outer

@literal_init
	%lit_i0 =l loadl %islot
	%lit_first_next =l add %lit_i0, 1
	storel %lit_first_next, %jslot
	jmp @lit_cond

@lit_cond
	%lit_j =l loadl %jslot
	%lit_i =l loadl %islot
	%lit_has =w csltl %lit_j, %n
	jnz %lit_has, @lit_len_check, @lit_ready

@lit_len_check
	%lit_len_so_far =l sub %lit_j, %lit_i
	%lit_can =w csltl %lit_len_so_far, 128
	jnz %lit_can, @lit_run_init, @lit_ready

@lit_run_init
	%lit_base_ptr =l add %src, %lit_j
	%lit_b =w loadub %lit_base_ptr
	storew 1, %rslot
	%lit_k0 =l add %lit_j, 1
	storel %lit_k0, %kslot
	jmp @lit_run_cond

@lit_run_cond
	%lit_k =l loadl %kslot
	%lit_k_in =w csltl %lit_k, %n
	jnz %lit_k_in, @lit_run_cap, @lit_run_done

@lit_run_cap
	%lit_r =w loadw %rslot
	%lit_r_can =w csltw %lit_r, 128
	jnz %lit_r_can, @lit_run_byte, @lit_run_done

@lit_run_byte
	%lit_run_ptr =l add %src, %lit_k
	%lit_run_b =w loadub %lit_run_ptr
	%lit_same =w ceqw %lit_run_b, %lit_b
	jnz %lit_same, @lit_run_inc, @lit_run_done

@lit_run_inc
	%lit_r1 =w add %lit_r, 1
	%lit_k1 =l add %lit_k, 1
	storew %lit_r1, %rslot
	storel %lit_k1, %kslot
	jmp @lit_run_cond

@lit_run_done
	%lit_r_final =w loadw %rslot
	%lit_run_small =w csltw %lit_r_final, 3
	jnz %lit_run_small, @lit_advance, @lit_ready

@lit_advance
	%lit_j1 =l add %lit_j, 1
	storel %lit_j1, %jslot
	jmp @lit_cond

@lit_ready
	%lit_end =l loadl %jslot
	%lit_start =l loadl %islot
	%lit_len =l sub %lit_end, %lit_start
	%lit_need =l add %lit_len, 1
	%lit_outpos =l loadl %oslot
	%lit_room =l sub %cap, %lit_outpos
	%lit_too_big =w csltl %lit_room, %lit_need
	jnz %lit_too_big, @fail, @lit_write

@lit_write
	%lit_header =l sub %lit_len, 1
	%lit_dest =l add %out, %lit_outpos
	storeb %lit_header, %lit_dest
	storel 0, %kslot
	jmp @lit_copy_cond

@lit_copy_cond
	%copy_k =l loadl %kslot
	%copy_more =w csltl %copy_k, %lit_len
	jnz %copy_more, @lit_copy_body, @lit_copy_done

@lit_copy_body
	%copy_src_off =l add %lit_start, %copy_k
	%copy_src =l add %src, %copy_src_off
	%copy_val =w loadub %copy_src
	%copy_dst_base =l add %lit_outpos, 1
	%copy_dst_off =l add %copy_dst_base, %copy_k
	%copy_dst =l add %out, %copy_dst_off
	storeb %copy_val, %copy_dst
	%copy_k1 =l add %copy_k, 1
	storel %copy_k1, %kslot
	jmp @lit_copy_cond

@lit_copy_done
	%lit_next_out =l add %lit_outpos, %lit_need
	storel %lit_next_out, %oslot
	storel %lit_end, %islot
	jmp @outer

@success
	%result =l loadl %oslot
	ret %result

@fail
	ret -1
}
