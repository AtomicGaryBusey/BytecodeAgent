memmove_i64:
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jbe .Lforward
lea rcx, [rdx - 1]
.Lbackward:
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
sub rcx, 1
jns .Lbackward
jmp .Ldone
.Lforward:
xor ecx, ecx
.Lforward_loop:
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
add rcx, 1
cmp rcx, rdx
jb .Lforward_loop
.Ldone:
xor eax, eax
ret
