export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%count0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @maxjoin %inext
	%min =l phi @start %y0, @maxjoin %minnext
	%max =l phi @start %y0, @maxjoin %maxnext
	%sum =l phi @start %y0, @maxjoin %sumnext
	%count =l phi @start %count0, @maxjoin %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%address =l add %a, %offset
	%x =l loadl %address
	%y =l call %f(l %x)
	%sumnext =l add %sum, %y
	%odd =l and %y, 1
	%countnext =l add %count, %odd
	%isless =w csltl %y, %min
	jnz %isless, @newmin, @minjoin
@newmin
	jmp @minjoin
@minjoin
	%minnext =l phi @body %min, @newmin %y
	%isgreater =w csltl %max, %y
	jnz %isgreater, @newmax, @maxjoin
@newmax
	jmp @maxjoin
@maxjoin
	%maxnext =l phi @minjoin %max, @newmax %y
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%outmax =l add %out, 8
	storel %max, %outmax
	%outsum =l add %out, 16
	storel %sum, %outsum
	ret %count
}
