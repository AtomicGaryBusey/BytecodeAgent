export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero_test

@zero_test
	%zi =l phi @start 0, @zero_body %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zero_body, @outer_init

@zero_body
	%zoff =l shl %zi, 3
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi1 =l add %zi, 1
	jmp @zero_test

@outer_init
	jmp @outer_test

@outer_test
	%i =l phi @outer_init 0, @outer_end %i_next
	%ic =w csltl %i, %na
	jnz %ic, @inner_init, @scan_init

@inner_init
	%aoff =l shl %i, 3
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	jmp @inner_test

@inner_test
	%j =l phi @inner_init 0, @inner_body %j1
	%carry =l phi @inner_init 0, @inner_body %newcarry
	%jc =w csltl %j, %nb
	jnz %jc, @inner_body, @outer_end

@inner_body
	%boff =l shl %j, 3
	%bp =l add %b, %boff
	%bj =l loadl %bp

	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32

	%m0 =l mul %x0, %y0
	%m1 =l mul %x0, %y1
	%m2 =l mul %x1, %y0
	%m3 =l mul %x1, %y1

	%m1lo =l shl %m1, 32
	%m1hi =l shr %m1, 32
	%lo1 =l add %m0, %m1lo
	%c1 =w cultl %lo1, %m0
	%c1x =l extuw %c1
	%hi1 =l add %m3, %m1hi
	%hi2 =l add %hi1, %c1x

	%m2lo =l shl %m2, 32
	%m2hi =l shr %m2, 32
	%lo2 =l add %lo1, %m2lo
	%c2 =w cultl %lo2, %lo1
	%c2x =l extuw %c2
	%hi3 =l add %hi2, %m2hi
	%hi4 =l add %hi3, %c2x

	%ij =l add %i, %j
	%roff =l shl %ij, 3
	%rp =l add %r, %roff
	%old =l loadl %rp

	%lo3 =l add %lo2, %old
	%c3 =w cultl %lo3, %lo2
	%c3x =l extuw %c3
	%hi5 =l add %hi4, %c3x

	%lo4 =l add %lo3, %carry
	%c4 =w cultl %lo4, %lo3
	%c4x =l extuw %c4
	%newcarry =l add %hi5, %c4x

	storel %lo4, %rp
	%j1 =l add %j, 1
	jmp @inner_test

@outer_end
	%endindex =l add %i, %nb
	%endoff =l shl %endindex, 3
	%endp =l add %r, %endoff
	storel %carry, %endp
	%i_next =l add %i, 1
	jmp @outer_test

@scan_init
	jmp @scan_test

@scan_test
	%k =l phi @scan_init %n, @scan_dec %k1
	%kzero =w ceql %k, 0
	jnz %kzero, @ret, @scan_check

@scan_check
	%kminus =l sub %k, 1
	%koff =l shl %kminus, 3
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kvzero =w ceql %kv, 0
	jnz %kvzero, @scan_dec, @ret

@scan_dec
	%k1 =l sub %k, 1
	jmp @scan_test

@ret
	ret %k
}
