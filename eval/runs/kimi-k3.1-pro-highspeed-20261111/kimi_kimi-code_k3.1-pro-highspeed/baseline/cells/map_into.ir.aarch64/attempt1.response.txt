export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%is_empty =w ceql %n, 0
	jnz %is_empty, @empty, @loop
@loop
	%i =l phi @entry 0, @next %next_i
	%count =l phi @entry 0, @next %next_count
	%offset =l mul %i, 8
	%src_addr =l add %src, %offset
	%x =l loadl %src_addr
	%r =l call %f(l %x)
	%dst_addr =l add %dst, %offset
	storel %r, %dst_addr
	%less =w csltl %r, %x
	%less_l =l extuw %less
	%next_count =l add %count, %less_l
	%next_i =l add %i, 1
	jmp @next
@next
	%done =w ceql %next_i, %n
	jnz %done, @exit, @loop
@exit
	ret %next_count
@empty
	ret 0
}
