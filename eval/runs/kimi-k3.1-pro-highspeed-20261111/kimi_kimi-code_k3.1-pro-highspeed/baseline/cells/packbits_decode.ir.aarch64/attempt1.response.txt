export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet_loop

@packet_loop
	%pos =l phi @start 0, @packet_advance %next_pos
	%total =l phi @start 0, @packet_advance %next_total
	%at_end =w csgel %pos, %n
	jnz %at_end, @success, @header

@success
	ret %total

@header
	%header_ptr =l add %src, %pos
	%h =w loadsb %header_ptr
	%after_header =l add %pos, 1
	%is_nop =w ceqw %h, -128
	jnz %is_nop, @nop_packet, @classify

@classify
	%is_negative =w csltw %h, 0
	jnz %is_negative, @repeat_packet, @literal_packet

@nop_packet
	jmp @packet_advance

@packet_advance
	%next_pos =l phi @nop_packet %after_header, @literal_done %literal_end, @repeat_done %repeat_end
	%next_total =l phi @nop_packet %total, @literal_done %literal_total, @repeat_done %repeat_total
	jmp @packet_loop

@literal_packet
	%lit_count_w =w add %h, 1
	%lit_count =l extuw %lit_count_w
	%lit_source_left =l sub %n, %after_header
	%lit_truncated =w csgtl %lit_count, %lit_source_left
	jnz %lit_truncated, @error_truncated, @lit_capacity

@lit_capacity
	%lit_space =l sub %cap, %total
	%lit_overflow =w csgtl %lit_count, %lit_space
	jnz %lit_overflow, @lit_overflow_start, @lit_copy_start

@lit_copy_start
	%lit_data_src =l add %src, %after_header
	%lit_data_dst =l add %out, %total
	jmp @lit_copy_loop

@lit_copy_loop
	%lit_offset =l phi @lit_copy_start 0, @lit_copy_body %lit_offset_next
	%lit_copy_complete =w csgel %lit_offset, %lit_count
	jnz %lit_copy_complete, @literal_done, @lit_copy_body

@lit_copy_body
	%lit_src_ptr =l add %lit_data_src, %lit_offset
	%lit_dst_ptr =l add %lit_data_dst, %lit_offset
	%lit_byte =w loadub %lit_src_ptr
	storeb %lit_byte, %lit_dst_ptr
	%lit_offset_next =l add %lit_offset, 1
	jmp @lit_copy_loop

@literal_done
	%literal_end =l add %after_header, %lit_count
	%literal_total =l add %total, %lit_count
	jmp @packet_advance

@lit_overflow_start
	%lit_over_src =l add %src, %after_header
	%lit_over_dst =l add %out, %total
	jmp @lit_overflow_loop

@lit_overflow_loop
	%lit_over_offset =l phi @lit_overflow_start 0, @lit_overflow_body %lit_over_next
	%lit_fill_complete =w csgel %lit_over_offset, %lit_space
	jnz %lit_fill_complete, @error_overflow, @lit_overflow_body

@lit_overflow_body
	%lit_over_src_ptr =l add %lit_over_src, %lit_over_offset
	%lit_over_dst_ptr =l add %lit_over_dst, %lit_over_offset
	%lit_over_byte =w loadub %lit_over_src_ptr
	storeb %lit_over_byte, %lit_over_dst_ptr
	%lit_over_next =l add %lit_over_offset, 1
	jmp @lit_overflow_loop

@repeat_packet
	%repeat_has_value =w csltl %after_header, %n
	jnz %repeat_has_value, @repeat_value, @error_truncated

@repeat_value
	%repeat_value_ptr =l add %src, %after_header
	%repeat_byte =w loadub %repeat_value_ptr
	%repeat_end =l add %after_header, 1
	%repeat_count_w =w sub 1, %h
	%repeat_count =l extuw %repeat_count_w
	%repeat_space =l sub %cap, %total
	%repeat_overflow =w csgtl %repeat_count, %repeat_space
	jnz %repeat_overflow, @repeat_overflow_start, @repeat_fill_start

@repeat_fill_start
	%repeat_dst =l add %out, %total
	jmp @repeat_fill_loop

@repeat_fill_loop
	%repeat_offset =l phi @repeat_fill_start 0, @repeat_fill_body %repeat_offset_next
	%repeat_fill_complete =w csgel %repeat_offset, %repeat_count
	jnz %repeat_fill_complete, @repeat_done, @repeat_fill_body

@repeat_fill_body
	%repeat_dst_ptr =l add %repeat_dst, %repeat_offset
	storeb %repeat_byte, %repeat_dst_ptr
	%repeat_offset_next =l add %repeat_offset, 1
	jmp @repeat_fill_loop

@repeat_done
	%repeat_total =l add %total, %repeat_count
	jmp @packet_advance

@repeat_overflow_start
	%repeat_over_dst =l add %out, %total
	jmp @repeat_overflow_loop

@repeat_overflow_loop
	%repeat_over_offset =l phi @repeat_overflow_start 0, @repeat_overflow_body %repeat_over_next
	%repeat_over_complete =w csgel %repeat_over_offset, %repeat_space
	jnz %repeat_over_complete, @error_overflow, @repeat_overflow_body

@repeat_overflow_body
	%repeat_over_dst_ptr =l add %repeat_over_dst, %repeat_over_offset
	storeb %repeat_byte, %repeat_over_dst_ptr
	%repeat_over_next =l add %repeat_over_offset, 1
	jmp @repeat_overflow_loop

@error_overflow
	ret -1

@error_truncated
	ret -2
}
