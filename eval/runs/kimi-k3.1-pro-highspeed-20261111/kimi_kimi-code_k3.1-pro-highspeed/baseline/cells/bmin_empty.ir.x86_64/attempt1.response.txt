export function l $bmin_empty(l %a, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @empty, @nonempty
@empty
	ret 9223372036854775807
@nonempty
	%first =l loadl %a
	%one =l copy 1
	jmp @check
@check
	%i =l phi @nonempty %one, @next %nexti
	%cur =l phi @nonempty %first, @next %nextmin
	%atend =w ceql %i, %n
	jnz %atend, @done, @body
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%less =w csltl %v, %cur
	jnz %less, @take, @keep
@take
	jmp @next
@keep
	jmp @next
@next
	%nextmin =l phi @take %v, @keep %cur
	%nexti =l add %i, 1
	jmp @check
@done
	ret %cur
}
