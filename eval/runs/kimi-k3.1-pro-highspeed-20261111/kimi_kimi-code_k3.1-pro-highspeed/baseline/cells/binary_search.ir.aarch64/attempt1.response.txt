export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @right %next_lo, @left %same_lo
	%hi =l phi @start %n, @right %same_hi, @left %mid
	%nonempty =w csltl %lo, %hi
	jnz %nonempty, @body, @not_found
@body
	%range =l sub %hi, %lo
	%half =l div %range, 2
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%address =l add %a, %offset
	%value =l loadl %address
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @right, @left
@right
	%next_lo =l add %mid, 1
	%same_hi =l copy %hi
	jmp @loop
@left
	%same_lo =l copy %lo
	jmp @loop
@found
	ret %mid
@not_found
	ret -1
}
