export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%bound =l sub %n, 1
	jmp @outer(%bound)
@outer(l %bound)
	%finished =w ceql %bound, 0
	jnz %finished, @done, @body
@body
	jmp @inner(0)
@inner(l %i)
	%inside =w csltl %i, %bound
	jnz %inside, @compare, @outer_next
@compare
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%offset_next =l add %offset, 8
	%q =l add %a, %offset_next
	%y =l loadl %q
	%greater =w csltl %y, %x
	%i_next =l add %i, 1
	jnz %greater, @swap, @continue
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner(%i_next)
@continue
	jmp @inner(%i_next)
@outer_next
	%bound_next =l sub %bound, 1
	jmp @outer(%bound_next)
@done
	ret 0
}
