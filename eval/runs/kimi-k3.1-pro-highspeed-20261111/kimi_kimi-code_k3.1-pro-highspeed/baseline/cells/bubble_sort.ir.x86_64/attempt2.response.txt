export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%bound0 =l sub %n, 1
	jmp @outer
@outer
	%bound =l phi @init %bound0, @outer_next %bound_next
	%finished =w ceql %bound, 0
	jnz %finished, @done, @body
@body
	jmp @inner
@inner
	%i =l phi @body 0, @swap %i_next, @continue %i_next
	%inside =w csltl %i, %bound
	jnz %inside, @compare, @outer_next
@compare
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%offset_next =l add %offset, 8
	%q =l add %a, %offset_next
	%y =l loadl %q
	%greater =w csltl %y, %x
	%i_next =l add %i, 1
	jnz %greater, @swap, @continue
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner
@continue
	jmp @inner
@outer_next
	%bound_next =l sub %bound, 1
	jmp @outer
@done
	ret 0
}
