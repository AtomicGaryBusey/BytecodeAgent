crc32:
mov eax, 0xFFFFFFFF
xor ecx, ecx
.Lbyte_loop:
cmp rcx, rsi
jae .Ldone
movzx edx, byte ptr [rdi + rcx]
xor eax, edx
mov r8d, 8
.Lbit_loop:
test eax, 1
jz .Lshift
shr eax, 1
xor eax, 0xEDB88320
jmp .Lnext_bit
.Lshift:
shr eax, 1
.Lnext_bit:
dec r8d
jnz .Lbit_loop
inc rcx
jmp .Lbyte_loop
.Ldone:
not eax
ret
