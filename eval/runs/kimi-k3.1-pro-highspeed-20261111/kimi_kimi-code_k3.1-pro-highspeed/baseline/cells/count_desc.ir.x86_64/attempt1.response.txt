export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@init
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %i1
	%count =l phi @init 0, @body %count1
	%more =w csltl %i, %last
	jnz %more, @body, @exit
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csgtl %x, %y
	%inc =l extuw %gt
	%count1 =l add %count, %inc
	%i1 =l add %i, 1
	jmp @loop
@exit
	ret %count
@zero
	ret 0
}
