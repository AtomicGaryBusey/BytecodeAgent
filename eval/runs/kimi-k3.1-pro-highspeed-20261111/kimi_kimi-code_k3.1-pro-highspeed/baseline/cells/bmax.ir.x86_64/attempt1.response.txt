export function l $bmax(l %a, l %n) {
@start
	%initial =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %inext
	%m =l phi @start %initial, @next %mnext
	%done =w ceql %i, %n
	jnz %done, @return, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csltl %m, %v
	jnz %gt, @update, @next
@update
	jmp @next
@next
	%mnext =l phi @body %m, @update %v
	%inext =l add %i, 1
	jmp @loop
@return
	ret %m
}
