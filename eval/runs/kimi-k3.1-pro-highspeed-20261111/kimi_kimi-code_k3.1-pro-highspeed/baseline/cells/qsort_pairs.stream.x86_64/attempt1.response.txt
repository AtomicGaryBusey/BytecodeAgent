qsort_pairs:
push rbx
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
cmp r13, 1
jle .Ldone
mov rax, r13
shr rax, 1
lea rdx, [rax + rax]
mov r14, qword ptr [r12 + rdx*8]
mov r15, qword ptr [r12 + rdx*8 + 8]
xor ebx, ebx
xor esi, esi
mov rcx, r13
.Lpartition_loop:
cmp rsi, rcx
jge .Lpartition_done
lea rdx, [rsi + rsi]
mov rax, qword ptr [r12 + rdx*8]
cmp rax, r14
jl .Lless
jg .Lgreater
mov rax, qword ptr [r12 + rdx*8 + 8]
cmp rax, r15
jl .Lless
jg .Lgreater
inc rsi
jmp .Lpartition_loop
.Lless:
lea rdx, [rbx + rbx]
lea rdi, [rsi + rsi]
mov r8, qword ptr [r12 + rdx*8]
mov r9, qword ptr [r12 + rdx*8 + 8]
mov r10, qword ptr [r12 + rdi*8]
mov r11, qword ptr [r12 + rdi*8 + 8]
mov qword ptr [r12 + rdx*8], r10
mov qword ptr [r12 + rdx*8 + 8], r11
mov qword ptr [r12 + rdi*8], r8
mov qword ptr [r12 + rdi*8 + 8], r9
inc rbx
inc rsi
jmp .Lpartition_loop
.Lgreater:
dec rcx
lea rdx, [rsi + rsi]
lea rdi, [rcx + rcx]
mov r8, qword ptr [r12 + rdx*8]
mov r9, qword ptr [r12 + rdx*8 + 8]
mov r10, qword ptr [r12 + rdi*8]
mov r11, qword ptr [r12 + rdi*8 + 8]
mov qword ptr [r12 + rdx*8], r10
mov qword ptr [r12 + rdx*8 + 8], r11
mov qword ptr [r12 + rdi*8], r8
mov qword ptr [r12 + rdi*8 + 8], r9
jmp .Lpartition_loop
.Lpartition_done:
mov r14, rbx
mov r15, rcx
mov rdi, r12
mov rsi, r14
call qsort_pairs
lea rax, [r15 + r15]
lea rdi, [r12 + rax*8]
mov rsi, r13
sub rsi, r15
call qsort_pairs
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
