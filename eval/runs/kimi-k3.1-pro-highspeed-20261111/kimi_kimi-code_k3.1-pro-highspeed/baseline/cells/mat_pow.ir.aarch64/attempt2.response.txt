export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%count =l mul %n, %n
	%count_bytes =l mul %count, 8
	%prod =l add %tmp, %count_bytes
	jmp @init
@init
	%idx =l phi @start 0, @init_body %idx_next
	%init_done =w ceql %idx, %count
	jnz %init_done, @diag, @init_body
@init_body
	%offset =l mul %idx, 8
	%src =l add %a, %offset
	%value =l loadl %src
	%base_ptr =l add %tmp, %offset
	storel %value, %base_ptr
	%out_ptr =l add %out, %offset
	storel 0, %out_ptr
	%idx_next =l add %idx, 1
	jmp @init
@diag
	%di =l phi @init 0, @diag_body %di_next
	%diag_done =w ceql %di, %n
	jnz %diag_done, @power, @diag_body
@diag_body
	%diag_row =l mul %di, %n
	%diag_idx =l add %diag_row, %di
	%diag_offset =l mul %diag_idx, 8
	%diag_ptr =l add %out, %diag_offset
	storel 1, %diag_ptr
	%di_next =l add %di, 1
	jmp @diag
@power
	%e =l phi @diag 0, @copy_done %e_next
	%power_done =w ceql %e, %k
	jnz %power_done, @zero_setup, @mult_init
@mult_init
	jmp @outer
@outer
	%i =l phi @mult_init 0, @outer_next %i_next
	%outer_done =w ceql %i, %n
	jnz %outer_done, @copy_setup, @row_init
@row_init
	jmp @inner
@inner
	%j =l phi @row_init 0, @inner_next %j_next
	%inner_done =w ceql %j, %n
	jnz %inner_done, @outer_next, @dot_init
@dot_init
	jmp @dot
@dot
	%t =l phi @dot_init 0, @dot_body %t_next
	%sum =l phi @dot_init 0, @dot_body %sum_next
	%dot_done =w ceql %t, %n
	jnz %dot_done, @store_product, @dot_body
@dot_body
	%x_row =l mul %i, %n
	%x_idx =l add %x_row, %t
	%x_offset =l mul %x_idx, 8
	%x_ptr =l add %out, %x_offset
	%x =l loadl %x_ptr
	%y_row =l mul %t, %n
	%y_idx =l add %y_row, %j
	%y_offset =l mul %y_idx, 8
	%y_ptr =l add %tmp, %y_offset
	%y =l loadl %y_ptr
	%term =l mul %x, %y
	%sum_next =l add %sum, %term
	%t_next =l add %t, 1
	jmp @dot
@store_product
	%p_row =l mul %i, %n
	%p_idx =l add %p_row, %j
	%p_offset =l mul %p_idx, 8
	%p_ptr =l add %prod, %p_offset
	storel %sum, %p_ptr
	jmp @inner_next
@inner_next
	%j_next =l add %j, 1
	jmp @inner
@outer_next
	%i_next =l add %i, 1
	jmp @outer
@copy_setup
	jmp @copy
@copy
	%ci =l phi @copy_setup 0, @copy_body %ci_next
	%copy_finished =w ceql %ci, %count
	jnz %copy_finished, @copy_done, @copy_body
@copy_body
	%c_offset =l mul %ci, 8
	%c_src =l add %prod, %c_offset
	%c_value =l loadl %c_src
	%c_dst =l add %out, %c_offset
	storel %c_value, %c_dst
	%ci_next =l add %ci, 1
	jmp @copy
@copy_done
	%e_next =l add %e, 1
	jmp @power
@zero_setup
	%total =l add %count, %count
	jmp @zero
@zero
	%zi =l phi @zero_setup 0, @zero_body %zi_next
	%zero_done =w ceql %zi, %total
	jnz %zero_done, @finish, @zero_body
@zero_body
	%z_offset =l mul %zi, 8
	%z_ptr =l add %tmp, %z_offset
	storel 0, %z_ptr
	%zi_next =l add %zi, 1
	jmp @zero
@finish
	ret 0
}
