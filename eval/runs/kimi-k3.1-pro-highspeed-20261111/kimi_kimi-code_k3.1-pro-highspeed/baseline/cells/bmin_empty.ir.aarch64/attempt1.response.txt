export function l $bmin_empty(l %a, l %n) {
@start

@loop
	%i =l phi @start 0, @next %i1
	%min =l phi @start 9223372036854775807, @next %min1
	%done =w ceql %i, %n
	jnz %done, @end, @body
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%less =w csltl %value, %min
	jnz %less, @update, @keep
@update

@keep

@next

@end
