export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00_hi =l shr %p00, 32
	%p01_lo =l and %p01, 4294967295
	%p10_lo =l and %p10, 4294967295
	%t0 =l add %p00_hi, %p01_lo
	%t =l add %t0, %p10_lo
	%p01_hi =l shr %p01, 32
	%p10_hi =l shr %p10, 32
	%cross_hi =l add %p01_hi, %p10_hi
	%t_hi =l shr %t, 32
	%h0 =l add %p11, %cross_hi
	%h1 =l add %h0, %t_hi
	ret %h1
}
