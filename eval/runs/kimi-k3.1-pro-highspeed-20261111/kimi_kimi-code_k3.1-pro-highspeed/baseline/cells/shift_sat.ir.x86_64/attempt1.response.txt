export function l $shift_sat(l %x, l %n) {
@entry
	%in_range =w csltl %n, 64
	jnz %in_range, @small, @large
@small
	%left =l shl %x, %n
	%right =l sar %x, %n
	%sum =l add %left, %right
	ret %sum
@large
	%is_negative =w csltl %x, 0
	jnz %is_negative, @negative, @nonnegative
@negative
	ret -1
@nonnegative
	ret 0
}
