export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%pcslot =l alloc8 8
	%dslot =l alloc8 8
	%islot =l alloc8 8
	storel 0, %pcslot
	storel 0, %dslot
	jmp @loop

@loop
	%pc =l loadl %pcslot
	%atend =w ceql %pc, %len
	jnz %atend, @stop, @fetch

@fetch
	%pcoff =l mul %pc, 8
	%iaddr =l add %code, %pcoff
	%op =l loadl %iaddr
	%ishalt =w ceql %op, 0
	jnz %ishalt, @stop, @checkpush

@checkpush
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @checkadd

@checkadd
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @checksub

@checksub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @checkmul

@checkmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @checkdup

@checkdup
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @checkswap

@checkswap
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @checkdrop

@checkdrop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @checkover

@checkover
	%isover =w ceql %op, 8
	jnz %isover, @over, @checklt

@checklt
	%islt =w ceql %op, 9
	jnz %islt, @lt, @checkjz

@checkjz
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @checkjmp

@checkjmp
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jmp, @next

@push
	%pcp =l loadl %pcslot
	%oppc =l add %pcp, 1
	%opoff =l mul %oppc, 8
	%opaddr =l add %code, %opoff
	%pushval =l loadl %opaddr
	%dp =l loadl %dslot
	%dpoff =l mul %dp, 8
	%dpaddr =l add %stack, %dpoff
	storel %pushval, %dpaddr
	%dpnew =l add %dp, 1
	storel %dpnew, %dslot
	%pcpnew =l add %pcp, 2
	storel %pcpnew, %pcslot
	jmp @loop

@add
	%da =l loadl %dslot
	%dai1 =l sub %da, 1
	%dai2 =l sub %da, 2
	%abo =l mul %dai1, 8
	%abp =l add %stack, %abo
	%abv =l loadl %abp
	%aao =l mul %dai2, 8
	%aap =l add %stack, %aao
	%aav =l loadl %aap
	%asum =l add %aav, %abv
	storel %asum, %aap
	storel %dai1, %dslot
	jmp @next

@sub
	%ds =l loadl %dslot
	%dsi1 =l sub %ds, 1
	%dsi2 =l sub %ds, 2
	%sbo =l mul %dsi1, 8
	%sbp =l add %stack, %sbo
	%sbv =l loadl %sbp
	%sao =l mul %dsi2, 8
	%sap =l add %stack, %sao
	%sav =l loadl %sap
	%sdiff =l sub %sav, %sbv
	storel %sdiff, %sap
	storel %dsi1, %dslot
	jmp @next

@mul
	%dm =l loadl %dslot
	%dmi1 =l sub %dm, 1
	%dmi2 =l sub %dm, 2
	%mbo =l mul %dmi1, 8
	%mbp =l add %stack, %mbo
	%mbv =l loadl %mbp
	%mao =l mul %dmi2, 8
	%map =l add %stack, %mao
	%mav =l loadl %map
	%mprod =l mul %mav, %mbv
	storel %mprod, %map
	storel %dmi1, %dslot
	jmp @next

@dup
	%dd =l loadl %dslot
	%ddi =l sub %dd, 1
	%ddso =l mul %ddi, 8
	%ddsp =l add %stack, %ddso
	%ddv =l loadl %ddsp
	%dddo =l mul %dd, 8
	%dddp =l add %stack, %dddo
	storel %ddv, %dddp
	%ddnew =l add %dd, 1
	storel %ddnew, %dslot
	jmp @next

@swap
	%dsw =l loadl %dslot
	%dswi1 =l sub %dsw, 1
	%dswi2 =l sub %dsw, 2
	%swo1 =l mul %dswi1, 8
	%swp1 =l add %stack, %swo1
	%swv1 =l loadl %swp1
	%swo2 =l mul %dswi2, 8
	%swp2 =l add %stack, %swo2
	%swv2 =l loadl %swp2
	storel %swv2, %swp1
	storel %swv1, %swp2
	jmp @next

@drop
	%ddr =l loadl %dslot
	%ddrnew =l sub %ddr, 1
	storel %ddrnew, %dslot
	jmp @next

@over
	%do =l loadl %dslot
	%doi =l sub %do, 2
	%doso =l mul %doi, 8
	%dosp =l add %stack, %doso
	%dov =l loadl %dosp
	%dodo =l mul %do, 8
	%dodp =l add %stack, %dodo
	storel %dov, %dodp
	%donew =l add %do, 1
	storel %donew, %dslot
	jmp @next

@lt
	%dl =l loadl %dslot
	%dli1 =l sub %dl, 1
	%dli2 =l sub %dl, 2
	%lbo =l mul %dli1, 8
	%lbp =l add %stack, %lbo
	%lbv =l loadl %lbp
	%lao =l mul %dli2, 8
	%lap =l add %stack, %lao
	%lav =l loadl %lap
	%lcmp =w csltl %lav, %lbv
	%lval =l extsw %lcmp
	storel %lval, %lap
	storel %dli1, %dslot
	jmp @next

@jz
	%pcj =l loadl %pcslot
	%dj =l loadl %dslot
	%dji =l sub %dj, 1
	%jvo =l mul %dji, 8
	%jvp =l add %stack, %jvo
	%jv =l loadl %jvp
	storel %dji, %dslot
	%jopc =l add %pcj, 1
	%joo =l mul %jopc, 8
	%jop =l add %code, %joo
	%jt =l loadl %jop
	%jzero =w ceql %jv, 0
	jnz %jzero, @jztaken, @jznext

@jztaken
	storel %jt, %pcslot
	jmp @loop

@jznext
	%pcjn =l add %pcj, 2
	storel %pcjn, %pcslot
	jmp @loop

@jmp
	%pcjm =l loadl %pcslot
	%jmopc =l add %pcjm, 1
	%jmoo =l mul %jmopc, 8
	%jmop =l add %code, %jmoo
	%jmt =l loadl %jmop
	storel %jmt, %pcslot
	jmp @loop

@next
	%pcn =l loadl %pcslot
	%pcnnew =l add %pcn, 1
	storel %pcnnew, %pcslot
	jmp @loop

@stop
	%df =l loadl %dslot
	storel %df, %islot
	jmp @clearloop

@clearloop
	%ci =l loadl %islot
	%cend =w ceql %ci, %cap
	jnz %cend, @return, @clearbody

@clearbody
	%cio =l mul %ci, 8
	%cip =l add %stack, %cio
	storel 0, %cip
	%cin =l add %ci, 1
	storel %cin, %islot
	jmp @clearloop

@return
	%rz =w ceql %df, 0
	jnz %rz, @retzero, @retvalue

@retzero
	ret 0

@retvalue
	%ridx =l sub %df, 1
	%roff =l mul %ridx, 8
	%raddr =l add %stack, %roff
	%rval =l loadl %raddr
	ret %rval
}
