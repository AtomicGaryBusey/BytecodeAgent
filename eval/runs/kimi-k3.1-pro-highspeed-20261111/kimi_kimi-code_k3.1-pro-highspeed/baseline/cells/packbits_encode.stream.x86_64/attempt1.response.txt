packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
.Louter:
cmp rbx, r13
jge .Lsuccess
lea rsi, [r12 + rbx]
mov r10b, byte ptr [rsi]
mov r8, r13
sub r8, rbx
cmp r8, 128
jbe .Lrun_limit_ready
mov r8d, 128
.Lrun_limit_ready:
mov r9d, 1
.Lrun_loop:
cmp r9, r8
jge .Lrun_done
cmp byte ptr [rsi + r9], r10b
jne .Lrun_done
inc r9
jmp .Lrun_loop
.Lrun_done:
cmp r9, 3
jge .Lrepeat
xor r8d, r8d
mov rcx, rbx
.Lliteral_test:
lea rsi, [r12 + rcx]
mov r10b, byte ptr [rsi]
mov rax, r13
sub rax, rcx
cmp rax, 128
jbe .Lliteral_limit_ready
mov eax, 128
.Lliteral_limit_ready:
mov r9d, 1
.Lliteral_run_loop:
cmp r9, rax
jge .Lliteral_run_done
cmp byte ptr [rsi + r9], r10b
jne .Lliteral_run_done
inc r9
jmp .Lliteral_run_loop
.Lliteral_run_done:
cmp r9, 3
jge .Lliteral_ready
inc rcx
inc r8
cmp rcx, r13
jge .Lliteral_ready
cmp r8, 128
jge .Lliteral_ready
jmp .Lliteral_test
.Lliteral_ready:
mov rax, r15
sub rax, rbp
lea rdx, [r8 + 1]
cmp rax, rdx
jb .Lfailure
lea rax, [r8 - 1]
mov byte ptr [r14 + rbp], al
lea rsi, [r12 + rbx]
lea rdi, [r14 + rbp + 1]
xor edx, edx
.Lliteral_copy:
cmp rdx, r8
jge .Lliteral_copied
mov al, byte ptr [rsi + rdx]
mov byte ptr [rdi + rdx], al
inc rdx
jmp .Lliteral_copy
.Lliteral_copied:
lea rbp, [rbp + r8 + 1]
mov rbx, rcx
jmp .Louter
.Lrepeat:
mov rax, r15
sub rax, rbp
cmp rax, 2
jb .Lfailure
mov eax, 1
sub rax, r9
mov byte ptr [r14 + rbp], al
mov byte ptr [r14 + rbp + 1], r10b
add rbp, 2
add rbx, r9
jmp .Louter
.Lsuccess:
mov rax, rbp
jmp .Lepilogue
.Lfailure:
mov rax, -1
.Lepilogue:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
