export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%two_nn =l add %nn, %nn
	jmp @zero_out_loop

@zero_out_loop
	%zi =l phi @start 0, @zero_out_body %zi_next
	%zc =w csltl %zi, %nn
	jnz %zc, @zero_out_body, @diag_loop

@zero_out_body
	%zoff =l mul %zi, 8
	%zp =l add %out, %zoff
	storel 0, %zp
	%zi_next =l add %zi, 1
	jmp @zero_out_loop

@diag_loop
	%d =l phi @zero_out_loop 0, @diag_body %d_next
	%dc =w csltl %d, %n
	jnz %dc, @diag_body, @exp_loop

@diag_body
	%dn =l mul %d, %n
	%didx =l add %dn, %d
	%doff =l mul %didx, 8
	%dp =l add %out, %doff
	storel 1, %dp
	%d_next =l add %d, 1
	jmp @diag_loop

@exp_loop
	%e =l phi @diag_loop 0, @copy_done %e_next
	%ec =w csltl %e, %k
	jnz %ec, @i_loop, @cleanup_loop

@i_loop
	%i =l phi @exp_loop 0, @j_done %i_next
	%ic =w csltl %i, %n
	jnz %ic, @j_loop, @copy_loop

@j_loop
	%j =l phi @i_loop 0, @t_done %j_next
	%jc =w csltl %j, %n
	jnz %jc, @t_loop, @j_done

@t_loop
	%t =l phi @j_loop 0, @t_body %t_next
	%sum =l phi @j_loop 0, @t_body %sum_next
	%tc =w csltl %t, %n
	jnz %tc, @t_body, @t_done

@t_body
	%in =l mul %i, %n
	%xidx =l add %in, %t
	%xoff =l mul %xidx, 8
	%xp =l add %out, %xoff
	%x =l loadl %xp
	%tn =l mul %t, %n
	%yidx =l add %tn, %j
	%yoff =l mul %yidx, 8
	%yp =l add %a, %yoff
	%y =l loadl %yp
	%prod =l mul %x, %y
	%sum_next =l add %sum, %prod
	%t_next =l add %t, 1
	jmp @t_loop

@t_done
	%ridx =l add %in, %j
	%roff =l mul %ridx, 8
	%rp =l add %tmp, %roff
	storel %sum, %rp
	%j_next =l add %j, 1
	jmp @j_loop

@j_done
	%i_next =l add %i, 1
	jmp @i_loop

@copy_loop
	%c =l phi @i_loop 0, @copy_body %c_next
	%cc =w csltl %c, %nn
	jnz %cc, @copy_body, @copy_done

@copy_body
	%coff =l mul %c, 8
	%csrc =l add %tmp, %coff
	%cdst =l add %out, %coff
	%cval =l loadl %csrc
	storel %cval, %cdst
	%c_next =l add %c, 1
	jmp @copy_loop

@copy_done
	%e_next =l add %e, 1
	jmp @exp_loop

@cleanup_loop
	%q =l phi @exp_loop 0, @cleanup_body %q_next
	%qc =w csltl %q, %two_nn
	jnz %qc, @cleanup_body, @done

@cleanup_body
	%qoff =l mul %q, 8
	%qp =l add %tmp, %qoff
	storel 0, %qp
	%q_next =l add %q, 1
	jmp @cleanup_loop

@done
	ret 0
}
