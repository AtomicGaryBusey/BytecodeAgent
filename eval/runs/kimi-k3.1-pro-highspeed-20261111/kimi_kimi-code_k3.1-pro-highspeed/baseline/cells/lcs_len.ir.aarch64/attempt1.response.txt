export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	storel 0, %row
	jmp @init
@init
	%i0 =l phi @start 1, @init_body %i0_next
	%init_done =w csgtl %i0, %n
	jnz %init_done, @outer, @init_body
@init_body
	%off0 =l mul %i0, 8
	%addr0 =l add %row, %off0
	storel 0, %addr0
	%i0_next =l add %i0, 1
	jmp @init
@outer
	%i =l phi @init 0, @outer_next %i_next
	%outer_done =w ceql %i, %m
	jnz %outer_done, @finish, @outer_body
@outer_body
	%xaddr =l add %x, %i
	%xbyte =w loadub %xaddr
	jmp @inner
@inner
	%j =l phi @outer_body 1, @after_update %j_next
	%diag =l phi @outer_body 0, @after_update %up
	%inner_done =w csgtl %j, %n
	jnz %inner_done, @outer_next, @inner_body
@inner_body
	%joff =l mul %j, 8
	%rowj =l add %row, %joff
	%up =l loadl %rowj
	%jm1 =l sub %j, 1
	%yaddr =l add %y, %jm1
	%ybyte =w loadub %yaddr
	%match =w ceqw %xbyte, %ybyte
	jnz %match, @set_match, @set_nomatch
@set_match
	%newval =l add %diag, 1
	storel %newval, %rowj
	jmp @after_update
@set_nomatch
	%prevoff =l sub %joff, 8
	%prevaddr =l add %row, %prevoff
	%left =l loadl %prevaddr
	%up_greater =w csgtl %up, %left
	jnz %up_greater, @store_up, @store_left
@store_up
	storel %up, %rowj
	jmp @after_update
@store_left
	storel %left, %rowj
	jmp @after_update
@after_update
	%j_next =l add %j, 1
	jmp @inner
@outer_next
	%i_next =l add %i, 1
	jmp @outer
@finish
	%noff =l mul %n, 8
	%result_addr =l add %row, %noff
	%result =l loadl %result_addr
	ret %result
}
