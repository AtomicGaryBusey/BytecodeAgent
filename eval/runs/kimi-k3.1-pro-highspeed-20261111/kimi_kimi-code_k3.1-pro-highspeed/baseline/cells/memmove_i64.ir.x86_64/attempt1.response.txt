export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @choose
@choose
	%goforward =w cultl %dst, %src
	jnz %goforward, @forward, @backward
@forward
	%fi =l phi @choose 0, @forward %fi1
	%foff =l mul %fi, 8
	%fs =l add %src, %foff
	%fd =l add %dst, %foff
	%fv =l loadl %fs
	storel %fv, %fd
	%fi1 =l add %fi, 1
	%fdone =w ceql %fi1, %n
	jnz %fdone, @done, @forward
@backward
	%bi =l phi @choose %n, @backward %bi1
	%bi1 =l sub %bi, 1
	%boff =l mul %bi1, 8
	%bs =l add %src, %boff
	%bd =l add %dst, %boff
	%bv =l loadl %bs
	storel %bv, %bd
	%bmore =w cnel %bi1, 0
	jnz %bmore, @backward, @done
@done
	ret 0
}
