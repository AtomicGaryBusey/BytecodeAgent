export function l $collatz_max(l %n) {
@entry
	jmp @outer
@outer
	%k =l phi @entry 1, @update %k_next
	%max =l phi @entry 0, @update %max_next
	%in_range =w cslel %k, %n
	jnz %in_range, @init, @done
@init
	jmp @collatz
@collatz
	%x =l phi @init %k, @step_end %x_next
	%steps =l phi @init 0, @step_end %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @candidate, @branch
@branch
	%low =l and %x, 1
	%is_odd =w ceql %low, 1
	jnz %is_odd, @odd, @even
@even
	%x_even =l div %x, 2
	jmp @step_end
@odd
	%x_triple =l mul %x, 3
	%x_odd =l add %x_triple, 1
	jmp @step_end
@step_end
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @collatz
@candidate
	%greater =w csltl %max, %steps
	jnz %greater, @newmax, @keepmax
@newmax
	jmp @update
@keepmax
	jmp @update
@update
	%max_next =l phi @newmax %steps, @keepmax %max
	%k_next =l add %k, 1
	jmp @outer
@done
	ret %max
}
