export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @cond
@cond
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @done
@body
	%ioff =l mul %i, 8
	%ip =l add %a, %ioff
	%joff =l mul %j, 8
	%jp =l add %a, %joff
	%iv =l loadl %ip
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @cond
@done
	ret 0
}
