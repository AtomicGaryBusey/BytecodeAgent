export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i_next
	%in_bounds =w csltl %i, %n
	jnz %in_bounds, @body, @exit
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%is_zero =w ceql %value, 0
	jnz %is_zero, @exit, @next
@next
	%i_next =l add %i, 1
	jmp @loop
@exit
	ret %i
}
