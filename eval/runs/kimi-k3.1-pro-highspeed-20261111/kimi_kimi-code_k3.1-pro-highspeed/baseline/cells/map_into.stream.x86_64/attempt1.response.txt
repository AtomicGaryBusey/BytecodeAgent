map_into:
push rbx
push rbp
push r12
push r13
push r14
mov r12, rdi
mov r13, rsi
mov rbp, rdx
mov r14, rcx
xor ebx, ebx
test rbp, rbp
je .Ldone
.Lloop:
mov rdi, qword ptr [r13]
call r14
mov qword ptr [r12], rax
cmp rax, qword ptr [r13]
jge .Lnot_less
inc rbx
.Lnot_less:
add r12, 8
add r13, 8
dec rbp
jne .Lloop
.Ldone:
mov rax, rbx
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
