export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%jslot =l alloc8 8
	%rslot =l alloc8 8
	%kslot =l alloc8 8
	%lslot =l alloc8 8
	%bslot =l alloc4 4
	storel 0, %islot
	storel 0, %oslot
	jmp @outer

@outer
	%i0 =l loadl %islot
	%atend =w csgel %i0, %n
	jnz %atend, @done, @outer_run_init

@done
	%ans =l loadl %oslot
	ret %ans

@outer_run_init
	storel 1, %rslot
	%k0 =l add %i0, 1
	storel %k0, %kslot
	%p0 =l add %src, %i0
	%v0 =w loadub %p0
	storew %v0, %bslot
	jmp @outer_run_cond

@outer_run_cond
	%k1 =l loadl %kslot
	%more0 =w csltl %k1, %n
	jnz %more0, @outer_run_cap, @outer_run_done

@outer_run_cap
	%r1 =l loadl %rslot
	%under0 =w csltl %r1, 128
	jnz %under0, @outer_run_byte, @outer_run_done

@outer_run_byte
	%k2 =l loadl %kslot
	%p1 =l add %src, %k2
	%v1 =w loadub %p1
	%oldb0 =w loadw %bslot
	%same0 =w ceqw %v1, %oldb0
	jnz %same0, @outer_run_take, @outer_run_done

@outer_run_take
	%r2 =l loadl %rslot
	%r3 =l add %r2, 1
	storel %r3, %rslot
	%k3 =l loadl %kslot
	%k4 =l add %k3, 1
	storel %k4, %kslot
	jmp @outer_run_cond

@outer_run_done
	%r4 =l loadl %rslot
	%isrep0 =w csgel %r4, 3
	jnz %isrep0, @repeat_check, @literal_init

@repeat_check
	%o0 =l loadl %oslot
	%room0 =l sub %cap, %o0
	%fit0 =w csgel %room0, 2
	jnz %fit0, @repeat_write, @error

@repeat_write
	%o1 =l loadl %oslot
	%r5 =l loadl %rslot
	%r5w =w copy %r5
	%hdr0 =w sub 257, %r5w
	%q0 =l add %out, %o1
	storeb %hdr0, %q0
	%o2 =l add %o1, 1
	%q1 =l add %out, %o2
	%vb0 =w loadw %bslot
	storeb %vb0, %q1
	%o3 =l add %o1, 2
	storel %o3, %oslot
	%i1 =l loadl %islot
	%i2 =l add %i1, %r5
	storel %i2, %islot
	jmp @outer

@literal_init
	%i3 =l loadl %islot
	storel %i3, %jslot
	storel 0, %lslot
	jmp @literal_cond

@literal_cond
	%j0 =l loadl %jslot
	%jlt0 =w csltl %j0, %n
	jnz %jlt0, @literal_len_check, @literal_emit

@literal_len_check
	%len0 =l loadl %lslot
	%lenlt0 =w csltl %len0, 128
	jnz %lenlt0, @literal_run_init, @literal_emit

@literal_run_init
	storel 1, %rslot
	%k5 =l add %j0, 1
	storel %k5, %kslot
	%p2 =l add %src, %j0
	%v2 =w loadub %p2
	storew %v2, %bslot
	jmp @literal_run_cond

@literal_run_cond
	%k6 =l loadl %kslot
	%more1 =w csltl %k6, %n
	jnz %more1, @literal_run_cap, @literal_run_done

@literal_run_cap
	%r6 =l loadl %rslot
	%under1 =w csltl %r6, 128
	jnz %under1, @literal_run_byte, @literal_run_done

@literal_run_byte
	%k7 =l loadl %kslot
	%p3 =l add %src, %k7
	%v3 =w loadub %p3
	%oldb1 =w loadw %bslot
	%same1 =w ceqw %v3, %oldb1
	jnz %same1, @literal_run_take, @literal_run_done

@literal_run_take
	%r7 =l loadl %rslot
	%r8 =l add %r7, 1
	storel %r8, %rslot
	%k8 =l loadl %kslot
	%k9 =l add %k8, 1
	storel %k9, %kslot
	jmp @literal_run_cond

@literal_run_done
	%r9 =l loadl %rslot
	%isrep1 =w csgel %r9, 3
	jnz %isrep1, @literal_emit, @literal_take

@literal_take
	%j1 =l loadl %jslot
	%j2 =l add %j1, 1
	storel %j2, %jslot
	%len1 =l loadl %lslot
	%len2 =l add %len1, 1
	storel %len2, %lslot
	jmp @literal_cond

@literal_emit
	%len3 =l loadl %lslot
	%packet0 =l add %len3, 1
	%o4 =l loadl %oslot
	%room1 =l sub %cap, %o4
	%fit1 =w csgel %room1, %packet0
	jnz %fit1, @literal_write, @error

@literal_write
	%len4 =l loadl %lslot
	%o5 =l loadl %oslot
	%q2 =l add %out, %o5
	%len4w =w copy %len4
	%hdr1 =w sub %len4w, 1
	storeb %hdr1, %q2
	storel 0, %kslot
	jmp @copy_cond

@copy_cond
	%k10 =l loadl %kslot
	%len5 =l loadl %lslot
	%copymore =w csltl %k10, %len5
	jnz %copymore, @copy_body, @copy_done

@copy_body
	%k11 =l loadl %kslot
	%i4 =l loadl %islot
	%sidx =l add %i4, %k11
	%p4 =l add %src, %sidx
	%v4 =w loadub %p4
	%o6 =l loadl %oslot
	%didx0 =l add %o6, 1
	%didx =l add %didx0, %k11
	%q3 =l add %out, %didx
	storeb %v4, %q3
	%k12 =l add %k11, 1
	storel %k12, %kslot
	jmp @copy_cond

@copy_done
	%o7 =l loadl %oslot
	%len6 =l loadl %lslot
	%packet1 =l add %len6, 1
	%o8 =l add %o7, %packet1
	storel %o8, %oslot
	%j3 =l loadl %jslot
	storel %j3, %islot
	jmp @outer

@error
	ret -1
}
