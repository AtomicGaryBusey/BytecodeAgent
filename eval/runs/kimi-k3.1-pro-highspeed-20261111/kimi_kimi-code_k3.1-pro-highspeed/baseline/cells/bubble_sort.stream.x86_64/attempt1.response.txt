bubble_sort:
mov rax, rsi
cmp rax, 2
jl .Ldone
.Louter:
mov rcx, 1
.Linner:
mov rdx, qword ptr [rdi + rcx*8 - 8]
mov r8, qword ptr [rdi + rcx*8]
cmp rdx, r8
jle .Lnoswap
mov qword ptr [rdi + rcx*8 - 8], r8
mov qword ptr [rdi + rcx*8], rdx
.Lnoswap:
add rcx, 1
cmp rcx, rax
jl .Linner
sub rax, 1
cmp rax, 1
jg .Louter
.Ldone:
xor eax, eax
ret
