lcs_len:
push rbx
xor r9d, r9d
xor eax, eax
.Lzero:
cmp r9, rcx
jg .Lzero_done
mov qword ptr [r8 + r9*8], rax
inc r9
jmp .Lzero
.Lzero_done:
xor r9d, r9d
.Louter:
cmp r9, rsi
jge .Ldone
xor ebx, ebx
mov r10d, 1
.Linner:
cmp r10, rcx
jg .Louter_next
mov r11, qword ptr [r8 + r10*8]
movzx eax, byte ptr [rdi + r9]
cmp al, byte ptr [rdx + r10 - 1]
jne .Lne
lea rax, [rbx + 1]
mov qword ptr [r8 + r10*8], rax
jmp .Linner_next
.Lne:
mov rax, qword ptr [r8 + r10*8 - 8]
cmp rax, r11
jge .Lmax_done
mov rax, r11
.Lmax_done:
mov qword ptr [r8 + r10*8], rax
.Linner_next:
mov rbx, r11
inc r10
jmp .Linner
.Louter_next:
inc r9
jmp .Louter
.Ldone:
mov rax, qword ptr [r8 + rcx*8]
pop rbx
ret
