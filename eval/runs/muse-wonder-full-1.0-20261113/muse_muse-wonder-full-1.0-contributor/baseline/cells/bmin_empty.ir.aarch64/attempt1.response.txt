export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @inc %i_next
	%min =l phi @start 9223372036854775807, @inc %min_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%lt =w csltl %val, %min
	jnz %lt, @upd, @keep
@upd
	jmp @inc
@keep
	jmp @inc
@inc
	%min_next =l phi @upd %val, @keep %min
	%i_next =l add %i, 1
	jmp @loop
@end
	ret %min
}
