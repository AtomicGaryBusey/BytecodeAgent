export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %slotidx =l alloc8 8
    %sloti =l alloc8 8
    %slotj =l alloc8 8
    %slotcarry =l alloc8 8
    %slotk =l alloc8 8
    %total =l add %na, %nb
    storel 0, %slotidx
    jmp @zcond
@zcond
    %zi =l loadl %slotidx
    %zc =w csltl %zi, %total
    jnz %zc, @zbody, @zdone
@zbody
    %zib =l loadl %slotidx
    %zoff =l mul %zib, 8
    %zptr =l add %r, %zoff
    storel 0, %zptr
    %znext =l add %zib, 1
    storel %znext, %slotidx
    jmp @zcond
@zdone
    %nazero =w ceql %na, 0
    jnz %nazero, @siginit, @chnb
@chnb
    %nbzero =w ceql %nb, 0
    jnz %nbzero, @siginit, @oinit
@oinit
    storel 0, %sloti
    jmp @ocond
@ocond
    %oi =l loadl %sloti
    %oc =w csltl %oi, %na
    jnz %oc, @obody, @siginit
@obody
    storel 0, %slotcarry
    storel 0, %slotj
    jmp @icond
@icond
    %ij =l loadl %slotj
    %ic =w csltl %ij, %nb
    jnz %ic, @ibody, @idone
@ibody
    %bi =l loadl %sloti
    %bj =l loadl %slotj
    %bcarry =l loadl %slotcarry
    %aoff =l mul %bi, 8
    %aptr =l add %a, %aoff
    %ai =l loadl %aptr
    %boff =l mul %bj, 8
    %bptr =l add %b, %boff
    %bv =l loadl %bptr
    %ijsum =l add %bi, %bj
    %roff =l mul %ijsum, 8
    %rptr =l add %r, %roff
    %rv =l loadl %rptr
    %xlo =l and %ai, 4294967295
    %xhi =l shr %ai, 32
    %ylo =l and %bv, 4294967295
    %yhi =l shr %bv, 32
    %p0 =l mul %xlo, %ylo
    %p1 =l mul %xlo, %yhi
    %p2 =l mul %xhi, %ylo
    %p3 =l mul %xhi, %yhi
    %p1shl =l shl %p1, 32
    %s1 =l add %p0, %p1shl
    %caw =w cultl %s1, %p0
    %ca =l extuw %caw
    %p2shl =l shl %p2, 32
    %plo =l add %s1, %p2shl
    %cbw =w cultl %plo, %s1
    %cb =l extuw %cbw
    %p1shr =l shr %p1, 32
    %p2shr =l shr %p2, 32
    %hi1 =l add %p3, %p1shr
    %hi2 =l add %hi1, %p2shr
    %hi3 =l add %hi2, %ca
    %phimul =l add %hi3, %cb
    %sum1 =l add %plo, %rv
    %c1w =w cultl %sum1, %plo
    %c1 =l extuw %c1w
    %sum =l add %sum1, %bcarry
    %c2w =w cultl %sum, %sum1
    %c2 =l extuw %c2w
    storel %sum, %rptr
    %nc1 =l add %phimul, %c1
    %ncarry =l add %nc1, %c2
    storel %ncarry, %slotcarry
    %jnext =l add %bj, 1
    storel %jnext, %slotj
    jmp @icond
@idone
    %di =l loadl %sloti
    %dcarry =l loadl %slotcarry
    %didx =l add %di, %nb
    %doff =l mul %didx, 8
    %dptr =l add %r, %doff
    %dold =l loadl %dptr
    %dnew =l add %dold, %dcarry
    storel %dnew, %dptr
    %inext =l add %di, 1
    storel %inext, %sloti
    jmp @ocond
@siginit
    storel %total, %slotk
    jmp @sigcond
@sigcond
    %k =l loadl %slotk
    %keqz =w ceql %k, 0
    jnz %keqz, @done, @sigbody
@sigbody
    %kb =l loadl %slotk
    %km1 =l sub %kb, 1
    %koff =l mul %km1, 8
    %kptr =l add %r, %koff
    %kv =l loadl %kptr
    %kzero =w ceql %kv, 0
    jnz %kzero, @sigdec, @done
@sigdec
    %kd =l loadl %slotk
    %kdec =l sub %kd, 1
    storel %kdec, %slotk
    jmp @sigcond
@done
    %res =l loadl %slotk
    ret %res
}
