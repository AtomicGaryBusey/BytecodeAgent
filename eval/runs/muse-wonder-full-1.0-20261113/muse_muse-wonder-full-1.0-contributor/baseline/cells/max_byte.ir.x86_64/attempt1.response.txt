export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%max =l phi @start 0, @next %maxnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @update, @next
@update
	jmp @next
@next
	%maxnext =l phi @body %max, @update %b
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
