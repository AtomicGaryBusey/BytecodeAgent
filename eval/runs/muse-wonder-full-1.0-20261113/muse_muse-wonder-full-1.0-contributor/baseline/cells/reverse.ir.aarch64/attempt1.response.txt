export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%j =l phi @start %j0, @body %j1
	%cond =w csltl %i, %j
	jnz %cond, @body, @end
@body
	%offi =l mul %i, 8
	%ptri =l add %a, %offi
	%offj =l mul %j, 8
	%ptrj =l add %a, %offj
	%vali =l loadl %ptri
	%valj =l loadl %ptrj
	storel %valj, %ptri
	storel %vali, %ptrj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@end
	ret 0
}
