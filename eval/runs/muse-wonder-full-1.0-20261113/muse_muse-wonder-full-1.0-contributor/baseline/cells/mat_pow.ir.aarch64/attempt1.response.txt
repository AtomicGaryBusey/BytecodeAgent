export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%pptr =l add %tmp, %nn8
	%tnn =l add %nn, %nn
	%vidx =l alloc8 8
	%vi =l alloc8 8
	%vj =l alloc8 8
	%vt =l alloc8 8
	%vcnt =l alloc8 8
	%vsum =l alloc8 8
	storel 0, %vidx
	jmp @ca_cond
@ca_cond
	%ca_i =l loadl %vidx
	%ca_c =w csltl %ca_i, %nn
	jnz %ca_c, @ca_body, @id_init_i
@ca_body
	%ca_i2 =l loadl %vidx
	%ca_off =l mul %ca_i2, 8
	%ca_sa =l add %a, %ca_off
	%ca_v =l loadl %ca_sa
	%ca_da =l add %tmp, %ca_off
	storel %ca_v, %ca_da
	%ca_nx =l add %ca_i2, 1
	storel %ca_nx, %vidx
	jmp @ca_cond
@id_init_i
	storel 0, %vi
	jmp @id_i_cond
@id_i_cond
	%idi =l loadl %vi
	%idic =w csltl %idi, %n
	jnz %idic, @id_j_init, @main_init
@id_j_init
	storel 0, %vj
	jmp @id_j_cond
@id_j_cond
	%idj =l loadl %vj
	%idjc =w csltl %idj, %n
	jnz %idjc, @id_body, @id_i_inc
@id_body
	%idb_i =l loadl %vi
	%idb_j =l loadl %vj
	%idb_row =l mul %idb_i, %n
	%idb_idx =l add %idb_row, %idb_j
	%idb_off =l mul %idb_idx, 8
	%idb_addr =l add %out, %idb_off
	%idb_eq =w ceql %idb_i, %idb_j
	%idb_val =l extuw %idb_eq
	storel %idb_val, %idb_addr
	%idb_nj =l add %idb_j, 1
	storel %idb_nj, %vj
	jmp @id_j_cond
@id_i_inc
	%idii =l loadl %vi
	%idin =l add %idii, 1
	storel %idin, %vi
	jmp @id_i_cond
@main_init
	storel 0, %vcnt
	jmp @main_cond
@main_cond
	%mc =l loadl %vcnt
	%mcc =w csltl %mc, %k
	jnz %mcc, @mm_i_init, @zero_init
@mm_i_init
	storel 0, %vi
	jmp @mm_i_cond
@mm_i_cond
	%mmi =l loadl %vi
	%mmic =w csltl %mmi, %n
	jnz %mmic, @mm_j_init, @cp_init
@mm_j_init
	storel 0, %vj
	jmp @mm_j_cond
@mm_j_cond
	%mmj =l loadl %vj
	%mmjc =w csltl %mmj, %n
	jnz %mmjc, @mm_t_init, @mm_i_inc
@mm_t_init
	storel 0, %vt
	storel 0, %vsum
	jmp @mm_t_cond
@mm_t_cond
	%mmt =l loadl %vt
	%mmtc =w csltl %mmt, %n
	jnz %mmtc, @mm_t_body, @mm_store
@mm_t_body
	%tb_i =l loadl %vi
	%tb_j =l loadl %vj
	%tb_t =l loadl %vt
	%tb_ir =l mul %tb_i, %n
	%tb_ai =l add %tb_ir, %tb_t
	%tb_ao =l mul %tb_ai, 8
	%tb_aa =l add %out, %tb_ao
	%tb_av =l loadl %tb_aa
	%tb_tr =l mul %tb_t, %n
	%tb_bi =l add %tb_tr, %tb_j
	%tb_bo =l mul %tb_bi, 8
	%tb_ba =l add %tmp, %tb_bo
	%tb_bv =l loadl %tb_ba
	%tb_pr =l mul %tb_av, %tb_bv
	%tb_s =l loadl %vsum
	%tb_ns =l add %tb_s, %tb_pr
	storel %tb_ns, %vsum
	%tb_nt =l add %tb_t, 1
	storel %tb_nt, %vt
	jmp @mm_t_cond
@mm_store
	%ms_i =l loadl %vi
	%ms_j =l loadl %vj
	%ms_r =l mul %ms_i, %n
	%ms_idx =l add %ms_r, %ms_j
	%ms_off =l mul %ms_idx, 8
	%ms_addr =l add %pptr, %ms_off
	%ms_v =l loadl %vsum
	storel %ms_v, %ms_addr
	%ms_nj =l add %ms_j, 1
	storel %ms_nj, %vj
	jmp @mm_j_cond
@mm_i_inc
	%mii =l loadl %vi
	%miin =l add %mii, 1
	storel %miin, %vi
	jmp @mm_i_cond
@cp_init
	storel 0, %vidx
	jmp @cp_cond
@cp_cond
	%cpi =l loadl %vidx
	%cpic =w csltl %cpi, %nn
	jnz %cpic, @cp_body, @main_inc
@cp_body
	%cpb_i =l loadl %vidx
	%cpb_off =l mul %cpb_i, 8
	%cpb_sa =l add %pptr, %cpb_off
	%cpb_v =l loadl %cpb_sa
	%cpb_da =l add %out, %cpb_off
	storel %cpb_v, %cpb_da
	%cpb_nx =l add %cpb_i, 1
	storel %cpb_nx, %vidx
	jmp @cp_cond
@main_inc
	%mni =l loadl %vcnt
	%mnin =l add %mni, 1
	storel %mnin, %vcnt
	jmp @main_cond
@zero_init
	storel 0, %vidx
	jmp @zero_cond
@zero_cond
	%zi =l loadl %vidx
	%zic =w csltl %zi, %tnn
	jnz %zic, @zero_body, @done
@zero_body
	%zb_i =l loadl %vidx
	%zb_off =l mul %zb_i, 8
	%zb_a =l add %tmp, %zb_off
	storel 0, %zb_a
	%zb_nx =l add %zb_i, 1
	storel %zb_nx, %vidx
	jmp @zero_cond
@done
	ret 0
}
