export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @next %i_next
	%max =l phi @start %max0, @next %max_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%cmp =w csgtl %val, %max
	jnz %cmp, @update, @keep
@update
	%max_upd =l copy %val
	jmp @next
@keep
	%max_keep =l copy %max
	jmp @next
@next
	%max_next =l phi @update %max_upd, @keep %max_keep
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %max
}
