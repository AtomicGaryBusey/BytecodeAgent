export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, 4294967295
	%p1hi =l shr %p1, 32
	%p2lo =l and %p2, 4294967295
	%p2hi =l shr %p2, 32
	%p3lo =l and %p3, 4294967295
	%p3hi =l shr %p3, 32
	%t1a =l add %p0hi, %p1lo
	%t1 =l add %t1a, %p2lo
	%carry2 =l shr %t1, 32
	%t2a =l add %p1hi, %p2hi
	%t2b =l add %t2a, %p3lo
	%t2 =l add %t2b, %carry2
	%r2 =l and %t2, 4294967295
	%carry3 =l shr %t2, 32
	%r3 =l add %p3hi, %carry3
	%r3s =l shl %r3, 32
	%hi =l add %r2, %r3s
	ret %hi
}
