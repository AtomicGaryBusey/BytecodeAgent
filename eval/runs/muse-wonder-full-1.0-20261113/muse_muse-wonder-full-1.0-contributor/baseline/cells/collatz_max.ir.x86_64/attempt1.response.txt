export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @outer_next %k_next
	%max =l phi @start 0, @outer_next %max_next
	%outer_cond =w cslel %k, %n
	jnz %outer_cond, @inner, @done
@inner
	%x =l phi @outer %k, @inner_step %x_next
	%steps =l phi @outer 0, @inner_step %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @inner_done, @inner_body
@inner_body
	%par =l and %x, 1
	%is_odd =w ceql %par, 1
	jnz %is_odd, @odd, @even
@even
	%x_even =l div %x, 2
	jmp @inner_step
@odd
	%x_tmp =l mul %x, 3
	%x_odd =l add %x_tmp, 1
	jmp @inner_step
@inner_step
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner
@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @update, @no_update
@update
	jmp @outer_next
@no_update
	jmp @outer_next
@outer_next
	%max_next =l phi @update %steps, @no_update %max
	%k_next =l add %k, 1
	jmp @outer
@done
	ret %max
}
