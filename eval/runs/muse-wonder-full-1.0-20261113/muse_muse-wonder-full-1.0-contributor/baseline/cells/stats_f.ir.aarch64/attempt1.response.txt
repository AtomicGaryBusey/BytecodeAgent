export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%one =l copy 1
	%sh =w copy 63
	%bit63 =l shl %one, %sh
	%min0 =l sub %bit63, 1
	%max0 =l copy %bit63
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @start %zero, @next %inext
	%min =l phi @start %min0, @next %minnext
	%max =l phi @start %max0, @next %maxnext
	%sum =l phi @start %zero, @next %sumnext
	%odd =l phi @start %zero, @next %oddnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%sumnext =l add %sum, %y
	%isodd =l and %y, 1
	%oddnext =l add %odd, %isodd
	%ltmin =w csltl %y, %min
	jnz %ltmin, @min_y, @min_old
@min_y
	jmp @min_done
@min_old
	jmp @min_done
@min_done
	%minnext =l phi @min_y %y, @min_old %min
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @max_y, @max_old
@max_y
	jmp @next
@max_old
	jmp @next
@next
	%maxnext =l phi @max_y %y, @max_old %max
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
