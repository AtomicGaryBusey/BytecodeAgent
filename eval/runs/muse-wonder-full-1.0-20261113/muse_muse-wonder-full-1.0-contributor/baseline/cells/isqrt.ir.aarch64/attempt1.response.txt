export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @next %lo2
	%hi =l phi @start 4294967296, @next %hi2
	%diff =l sub %hi, %lo
	%c =w cugtl %diff, 1
	jnz %c, @body, @end
@body
	%sum =l add %lo, %hi
	%mid =l udiv %sum, 2
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @lo_up, @hi_down
@lo_up
	%lo_a =l copy %mid
	%hi_a =l copy %hi
	jmp @next
@hi_down
	%lo_b =l copy %lo
	%hi_b =l copy %mid
	jmp @next
@next
	%lo2 =l phi @lo_up %lo_a, @hi_down %lo_b
	%hi2 =l phi @lo_up %hi_a, @hi_down %hi_b
	jmp @loop
@end
	ret %lo
}
