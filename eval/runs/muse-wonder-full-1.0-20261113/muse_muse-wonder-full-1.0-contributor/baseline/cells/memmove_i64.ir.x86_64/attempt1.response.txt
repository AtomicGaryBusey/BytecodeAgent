export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%islt =w cultl %dst, %src
	jnz %islt, @fwd_init, @bwd_init
@fwd_init
	jmp @fwd_loop
@fwd_loop
	%fi =l phi @fwd_init 0, @fwd_body %fi_next
	%fcmp =w csltl %fi, %n
	jnz %fcmp, @fwd_body, @done
@fwd_body
	%foff =l mul %fi, 8
	%fsrcp =l add %src, %foff
	%fdstp =l add %dst, %foff
	%fv =l loadl %fsrcp
	storel %fv, %fdstp
	%fi_next =l add %fi, 1
	jmp @fwd_loop
@bwd_init
	%bi_start =l sub %n, 1
	jmp @bwd_loop
@bwd_loop
	%bi =l phi @bwd_init %bi_start, @bwd_body %bi_next
	%bneg =w csltl %bi, 0
	jnz %bneg, @done, @bwd_body
@bwd_body
	%boff =l mul %bi, 8
	%bsrcp =l add %src, %boff
	%bdstp =l add %dst, %boff
	%bv =l loadl %bsrcp
	storel %bv, %bdstp
	%bi_next =l sub %bi, 1
	jmp @bwd_loop
@done
	ret 0
}
