bubble_sort:
cmp rsi, 1
jle .Ldone
mov rcx, rsi
dec rcx
.Louter:
xor edx, edx
.Linner:
mov rax, qword ptr [rdi + rdx*8]
mov r8, qword ptr [rdi + rdx*8 + 8]
cmp rax, r8
jle .Lnoswap
mov qword ptr [rdi + rdx*8], r8
mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
inc rdx
cmp rdx, rcx
jl .Linner
dec rcx
cmp rcx, 0
jg .Louter
.Ldone:
xor eax, eax
ret
