export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @incnop %i1, @litdone %ilitnext, @repdone %irepnext
	%o =l phi @start 0, @incnop %o, @litdone %olitnext, @repdone %orepnext
	%has =w csltl %i, %n
	jnz %has, @packet, @retok
@retok
	ret %o
@packet
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @incnop, @checksign
@incnop
	jmp @loop
@checksign
	%isneg =w csltw %h, 0
	jnz %isneg, @repeat, @literal
@literal
	%cntl0 =l extsw %h
	%cnt =l add %cntl0, 1
	%rem =l sub %n, %i1
	%srcok =w cslel %cnt, %rem
	jnz %srcok, @litcap, @retm2lit
@retm2lit
	ret -2
@litcap
	%oend =l add %o, %cnt
	%fits =w cslel %oend, %cap
	jnz %fits, @litcopyinit, @littruncinit
@litcopyinit
	jmp @litcopyloop
@litcopyloop
	%k =l phi @litcopyinit 0, @litcopybody %k1
	%klt =w csltl %k, %cnt
	jnz %klt, @litcopybody, @litdone
@litcopybody
	%sp0 =l add %src, %i1
	%sp =l add %sp0, %k
	%b =w loadub %sp
	%dp0 =l add %out, %o
	%dp =l add %dp0, %k
	storeb %b, %dp
	%k1 =l add %k, 1
	jmp @litcopyloop
@litdone
	%ilitnext =l add %i1, %cnt
	%olitnext =l add %o, %cnt
	jmp @loop
@littruncinit
	%avail =l sub %cap, %o
	jmp @littruncloop
@littruncloop
	%tk =l phi @littruncinit 0, @littruncbody %tk1
	%tklt =w csltl %tk, %avail
	jnz %tklt, @littruncbody, @retm1lit
@littruncbody
	%tsp0 =l add %src, %i1
	%tsp =l add %tsp0, %tk
	%tb =w loadub %tsp
	%tdp0 =l add %out, %o
	%tdp =l add %tdp0, %tk
	storeb %tb, %tdp
	%tk1 =l add %tk, 1
	jmp @littruncloop
@retm1lit
	ret -1
@repeat
	%remr =l sub %n, %i1
	%srcokr =w csgel %remr, 1
	jnz %srcokr, @repcap, @retm2rep
@retm2rep
	ret -2
@repcap
	%hl =l extsw %h
	%cntr =l sub 1, %hl
	%rp =l add %src, %i1
	%rb =w loadub %rp
	%oendr =l add %o, %cntr
	%fitsr =w cslel %oendr, %cap
	jnz %fitsr, @repcopyinit, @reptruncinit
@repcopyinit
	jmp @repcopyloop
@repcopyloop
	%rk =l phi @repcopyinit 0, @repcopybody %rk1
	%rklt =w csltl %rk, %cntr
	jnz %rklt, @repcopybody, @repdone
@repcopybody
	%rdp0 =l add %out, %o
	%rdp =l add %rdp0, %rk
	storeb %rb, %rdp
	%rk1 =l add %rk, 1
	jmp @repcopyloop
@repdone
	%irepnext =l add %i1, 1
	%orepnext =l add %o, %cntr
	jmp @loop
@reptruncinit
	%availr =l sub %cap, %o
	jmp @reptruncloop
@reptruncloop
	%trk =l phi @reptruncinit 0, @reptruncbody %trk1
	%trklt =w csltl %trk, %availr
	jnz %trklt, @reptruncbody, @retm1rep
@reptruncbody
	%trdp0 =l add %out, %o
	%trdp =l add %trdp0, %trk
	storeb %rb, %trdp
	%trk1 =l add %trk, 1
	jmp @reptruncloop
@retm1rep
	ret -1
}
