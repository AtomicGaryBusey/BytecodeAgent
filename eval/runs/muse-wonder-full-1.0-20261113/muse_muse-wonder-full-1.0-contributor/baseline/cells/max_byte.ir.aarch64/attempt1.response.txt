export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%max =l phi @start 0, @next %maxnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @useb, @usemax
@useb
	jmp @next
@usemax
	jmp @next
@next
	%maxnext =l phi @useb %b, @usemax %max
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
