export function l $qsort_pairs(l %a, l %n) {
@start
%plt =l alloc8 8
%pi =l alloc8 8
%pgt =l alloc8 8
%ppk =l alloc8 8
%ppv =l alloc8 8
%small =w csltl %n, 2
jnz %small, @done, @init
@init
%mid =l div %n, 2
%midoff =l mul %mid, 16
%midaddr =l add %a, %midoff
%pk0 =l loadl %midaddr
%midaddr1 =l add %midaddr, 8
%pv0 =l loadl %midaddr1
storel %pk0, %ppk
storel %pv0, %ppv
storel 0, %plt
storel 0, %pi
storel %n, %pgt
jmp @loop
@loop
%i1 =l loadl %pi
%gt1 =l loadl %pgt
%cont =w csltl %i1, %gt1
jnz %cont, @body, @pdone
@body
%ioff =l mul %i1, 16
%iaddr =l add %a, %ioff
%k1 =l loadl %iaddr
%iaddr1 =l add %iaddr, 8
%v1 =l loadl %iaddr1
%pk1 =l loadl %ppk
%pv1 =l loadl %ppv
%klt =w csltl %k1, %pk1
jnz %klt, @less, @checkkgt
@checkkgt
%kgt =w csltl %pk1, %k1
jnz %kgt, @greater, @keyeq
@keyeq
%vlt =w csltl %v1, %pv1
jnz %vlt, @less, @checkvgt
@checkvgt
%vgt =w csltl %pv1, %v1
jnz %vgt, @greater, @equal
@less
%lt1 =l loadl %plt
%ltoff =l mul %lt1, 16
%ltaddr =l add %a, %ltoff
%ltk =l loadl %ltaddr
%ltaddr1 =l add %ltaddr, 8
%ltv =l loadl %ltaddr1
storel %k1, %ltaddr
storel %v1, %ltaddr1
storel %ltk, %iaddr
storel %ltv, %iaddr1
%ltnew =l add %lt1, 1
storel %ltnew, %plt
%inew =l add %i1, 1
storel %inew, %pi
jmp @loop
@greater
%gt2 =l loadl %pgt
%gtnew =l sub %gt2, 1
storel %gtnew, %pgt
%gtoff =l mul %gtnew, 16
%gtaddr =l add %a, %gtoff
%gtk =l loadl %gtaddr
%gtaddr1 =l add %gtaddr, 8
%gtv =l loadl %gtaddr1
storel %k1, %gtaddr
storel %v1, %gtaddr1
storel %gtk, %iaddr
storel %gtv, %iaddr1
jmp @loop
@equal
%ieqnew =l add %i1, 1
storel %ieqnew, %pi
jmp @loop
@pdone
%ltf =l loadl %plt
%gtf =l loadl %pgt
%r1 =l call $qsort_pairs(l %a, l %ltf)
%rightoff =l mul %gtf, 16
%rightptr =l add %a, %rightoff
%rightn =l sub %n, %gtf
%r2 =l call $qsort_pairs(l %rightptr, l %rightn)
jmp @done
@done
ret 0
}
