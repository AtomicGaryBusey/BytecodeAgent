export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	%idxp =l alloc8 8
	%retp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch
@fetch
	%pcoff =l mul %pc, 8
	%codeptr =l add %code, %pcoff
	%op =l loadl %codeptr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @chk1
@chk1
	%is1 =w ceql %op, 1
	jnz %is1, @do_push, @chk2
@chk2
	%is2 =w ceql %op, 2
	jnz %is2, @do_add, @chk3
@chk3
	%is3 =w ceql %op, 3
	jnz %is3, @do_sub, @chk4
@chk4
	%is4 =w ceql %op, 4
	jnz %is4, @do_mul, @chk5
@chk5
	%is5 =w ceql %op, 5
	jnz %is5, @do_dup, @chk6
@chk6
	%is6 =w ceql %op, 6
	jnz %is6, @do_swap, @chk7
@chk7
	%is7 =w ceql %op, 7
	jnz %is7, @do_drop, @chk8
@chk8
	%is8 =w ceql %op, 8
	jnz %is8, @do_over, @chk9
@chk9
	%is9 =w ceql %op, 9
	jnz %is9, @do_lt, @chk10
@chk10
	%is10 =w ceql %op, 10
	jnz %is10, @do_jz, @chk11
@chk11
	%is11 =w ceql %op, 11
	jnz %is11, @do_jmp, @do_def
@do_push
	%pc1_push =l add %pc, 1
	%offop_push =l mul %pc1_push, 8
	%ptrop_push =l add %code, %offop_push
	%v_push =l loadl %ptrop_push
	%sp_push =l loadl %spp
	%off_push =l mul %sp_push, 8
	%ptr_push =l add %stack, %off_push
	storel %v_push, %ptr_push
	%spn_push =l add %sp_push, 1
	storel %spn_push, %spp
	%pcn_push =l add %pc, 2
	storel %pcn_push, %pcp
	jmp @loop
@do_add
	%sp_add =l loadl %spp
	%bidx_add =l sub %sp_add, 1
	%boff_add =l mul %bidx_add, 8
	%bptr_add =l add %stack, %boff_add
	%b_add =l loadl %bptr_add
	%aidx_add =l sub %sp_add, 2
	%aoff_add =l mul %aidx_add, 8
	%aptr_add =l add %stack, %aoff_add
	%a_add =l loadl %aptr_add
	%r_add =l add %a_add, %b_add
	storel %r_add, %aptr_add
	%spn_add =l sub %sp_add, 1
	storel %spn_add, %spp
	%pcn_add =l add %pc, 1
	storel %pcn_add, %pcp
	jmp @loop
@do_sub
	%sp_sub =l loadl %spp
	%bidx_sub =l sub %sp_sub, 1
	%boff_sub =l mul %bidx_sub, 8
	%bptr_sub =l add %stack, %boff_sub
	%b_sub =l loadl %bptr_sub
	%aidx_sub =l sub %sp_sub, 2
	%aoff_sub =l mul %aidx_sub, 8
	%aptr_sub =l add %stack, %aoff_sub
	%a_sub =l loadl %aptr_sub
	%r_sub =l sub %a_sub, %b_sub
	storel %r_sub, %aptr_sub
	%spn_sub =l sub %sp_sub, 1
	storel %spn_sub, %spp
	%pcn_sub =l add %pc, 1
	storel %pcn_sub, %pcp
	jmp @loop
@do_mul
	%sp_mul =l loadl %spp
	%bidx_mul =l sub %sp_mul, 1
	%boff_mul =l mul %bidx_mul, 8
	%bptr_mul =l add %stack, %boff_mul
	%b_mul =l loadl %bptr_mul
	%aidx_mul =l sub %sp_mul, 2
	%aoff_mul =l mul %aidx_mul, 8
	%aptr_mul =l add %stack, %aoff_mul
	%a_mul =l loadl %aptr_mul
	%r_mul =l mul %a_mul, %b_mul
	storel %r_mul, %aptr_mul
	%spn_mul =l sub %sp_mul, 1
	storel %spn_mul, %spp
	%pcn_mul =l add %pc, 1
	storel %pcn_mul, %pcp
	jmp @loop
@do_dup
	%sp_dup =l loadl %spp
	%tidx_dup =l sub %sp_dup, 1
	%toff_dup =l mul %tidx_dup, 8
	%tptr_dup =l add %stack, %toff_dup
	%t_dup =l loadl %tptr_dup
	%off_dup =l mul %sp_dup, 8
	%ptr_dup =l add %stack, %off_dup
	storel %t_dup, %ptr_dup
	%spn_dup =l add %sp_dup, 1
	storel %spn_dup, %spp
	%pcn_dup =l add %pc, 1
	storel %pcn_dup, %pcp
	jmp @loop
@do_swap
	%sp_swap =l loadl %spp
	%bidx_swap =l sub %sp_swap, 1
	%boff_swap =l mul %bidx_swap, 8
	%bptr_swap =l add %stack, %boff_swap
	%b_swap =l loadl %bptr_swap
	%aidx_swap =l sub %sp_swap, 2
	%aoff_swap =l mul %aidx_swap, 8
	%aptr_swap =l add %stack, %aoff_swap
	%a_swap =l loadl %aptr_swap
	storel %b_swap, %aptr_swap
	storel %a_swap, %bptr_swap
	%pcn_swap =l add %pc, 1
	storel %pcn_swap, %pcp
	jmp @loop
@do_drop
	%sp_drop =l loadl %spp
	%spn_drop =l sub %sp_drop, 1
	storel %spn_drop, %spp
	%pcn_drop =l add %pc, 1
	storel %pcn_drop, %pcp
	jmp @loop
@do_over
	%sp_over =l loadl %spp
	%aidx_over =l sub %sp_over, 2
	%aoff_over =l mul %aidx_over, 8
	%aptr_over =l add %stack, %aoff_over
	%a_over =l loadl %aptr_over
	%off_over =l mul %sp_over, 8
	%ptr_over =l add %stack, %off_over
	storel %a_over, %ptr_over
	%spn_over =l add %sp_over, 1
	storel %spn_over, %spp
	%pcn_over =l add %pc, 1
	storel %pcn_over, %pcp
	jmp @loop
@do_lt
	%sp_lt =l loadl %spp
	%bidx_lt =l sub %sp_lt, 1
	%boff_lt =l mul %bidx_lt, 8
	%bptr_lt =l add %stack, %boff_lt
	%b_lt =l loadl %bptr_lt
	%aidx_lt =l sub %sp_lt, 2
	%aoff_lt =l mul %aidx_lt, 8
	%aptr_lt =l add %stack, %aoff_lt
	%a_lt =l loadl %aptr_lt
	%cmp_lt =w csltl %a_lt, %b_lt
	%r_lt =l extuw %cmp_lt
	storel %r_lt, %aptr_lt
	%spn_lt =l sub %sp_lt, 1
	storel %spn_lt, %spp
	%pcn_lt =l add %pc, 1
	storel %pcn_lt, %pcp
	jmp @loop
@do_jz
	%pc1_jz =l add %pc, 1
	%offt_jz =l mul %pc1_jz, 8
	%ptrt_jz =l add %code, %offt_jz
	%t_jz =l loadl %ptrt_jz
	%sp_jz =l loadl %spp
	%vidx_jz =l sub %sp_jz, 1
	%voff_jz =l mul %vidx_jz, 8
	%vptr_jz =l add %stack, %voff_jz
	%v_jz =l loadl %vptr_jz
	%spn_jz =l sub %sp_jz, 1
	storel %spn_jz, %spp
	%iszero_jz =w ceql %v_jz, 0
	jnz %iszero_jz, @jz_taken, @jz_skip
@jz_taken
	storel %t_jz, %pcp
	jmp @loop
@jz_skip
	%pcn_jz =l add %pc, 2
	storel %pcn_jz, %pcp
	jmp @loop
@do_jmp
	%pc1_jmp =l add %pc, 1
	%offt_jmp =l mul %pc1_jmp, 8
	%ptrt_jmp =l add %code, %offt_jmp
	%t_jmp =l loadl %ptrt_jmp
	storel %t_jmp, %pcp
	jmp @loop
@do_def
	%pcn_def =l add %pc, 1
	storel %pcn_def, %pcp
	jmp @loop
@finish
	%spf =l loadl %spp
	%has =w csltl 0, %spf
	jnz %has, @fin_has, @fin_empty
@fin_has
	%topidx =l sub %spf, 1
	%topoff =l mul %topidx, 8
	%topptr =l add %stack, %topoff
	%topv =l loadl %topptr
	storel %topv, %retp
	jmp @fin_init
@fin_empty
	storel 0, %retp
	jmp @fin_init
@fin_init
	%dinit =l loadl %spp
	storel %dinit, %idxp
	jmp @zero_loop
@zero_loop
	%i =l loadl %idxp
	%ilt =w csltl %i, %cap
	jnz %ilt, @zero_body, @done
@zero_body
	%ioff =l mul %i, 8
	%iptr =l add %stack, %ioff
	storel 0, %iptr
	%inext =l add %i, 1
	storel %inext, %idxp
	jmp @zero_loop
@done
	%rv =l loadl %retp
	ret %rv
}
