export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @goup %lo2, @godown %lo
	%hi =l phi @start %n, @goup %hi, @godown %mid
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@found
	ret %mid
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @goup, @godown
@goup
	%lo2 =l add %mid, 1
	jmp @loop
@godown
	jmp @loop
@notfound
	ret -1
}
