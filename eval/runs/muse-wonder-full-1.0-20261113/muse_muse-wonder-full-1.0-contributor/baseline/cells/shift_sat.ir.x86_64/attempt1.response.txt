export function l $shift_sat(l %x, l %n) {
@start
	jmp @cond
@cond
	%lft =l phi @start %x, @body %ln
	%rgt =l phi @start %x, @body %rn
	%i =l phi @start 0, @body %in
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%ln =l shl %lft, 1
	%rn =l sar %rgt, 1
	%in =l add %i, 1
	jmp @cond
@done
	%res =l add %lft, %rgt
	ret %res
}
