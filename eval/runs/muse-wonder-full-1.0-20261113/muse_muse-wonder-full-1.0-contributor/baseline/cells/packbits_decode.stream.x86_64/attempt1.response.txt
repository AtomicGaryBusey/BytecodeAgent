packbits_decode:
xor r8, r8
xor r9, r9
.Lloop:
cmp r8, rsi
jge .Ldone
movsx r10, byte ptr [rdi + r8]
inc r8
cmp r10, -128
je .Lloop
test r10, r10
jns .Lliteral
mov r11, 1
sub r11, r10
cmp r8, rsi
jge .Lerr2
movzx eax, byte ptr [rdi + r8]
inc r8
mov r10, r9
add r10, r11
cmp r10, rcx
jg .Lrepeat_fill
.Lrepeat_write:
test r11, r11
jz .Lloop
mov byte ptr [rdx + r9], al
inc r9
dec r11
jmp .Lrepeat_write
.Lrepeat_fill:
cmp r9, rcx
jge .Lerr1
mov byte ptr [rdx + r9], al
inc r9
jmp .Lrepeat_fill
.Lliteral:
inc r10
mov r11, r8
add r11, r10
cmp r11, rsi
jg .Lerr2
mov r11, r9
add r11, r10
cmp r11, rcx
jg .Lliteral_fill
.Lliteral_copy:
test r10, r10
jz .Lloop
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r10
jmp .Lliteral_copy
.Lliteral_fill:
cmp r9, rcx
jge .Lerr1
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
jmp .Lliteral_fill
.Ldone:
mov rax, r9
ret
.Lerr1:
mov rax, -1
ret
.Lerr2:
mov rax, -2
ret
