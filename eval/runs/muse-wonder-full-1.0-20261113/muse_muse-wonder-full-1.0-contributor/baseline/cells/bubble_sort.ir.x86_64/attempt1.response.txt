export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @outer_inc %i_next
	%i1 =l add %i, 1
	%ocond =w csltl %i1, %n
	jnz %ocond, @inner, @done
@inner
	%j =l phi @outer 0, @inner_inc %j_next
	%lim =l sub %n, %i
	%lim1 =l sub %lim, 1
	%icond =w csltl %j, %lim1
	jnz %icond, @body, @outer_inc
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @inner_inc
@swap
	storel %y, %p
	storel %x, %p1
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner
@outer_inc
	%i_next =l add %i, 1
	jmp @outer
@done
	ret 0
}
