export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%zero =l copy 0
	%is_zero =w ceql %b, %zero
	jnz %is_zero, @divzero, @check_overflow
@divzero
	storel %a, %rem
	ret %zero
@check_overflow
	%negmax =l copy -9223372036854775807
	%min =l sub %negmax, 1
	%is_min =w ceql %a, %min
	jnz %is_min, @check_neg1, @normal
@check_neg1
	%neg1 =l copy -1
	%is_neg1 =w ceql %b, %neg1
	jnz %is_neg1, @overflow, @normal
@overflow
	storel %zero, %rem
	ret %a
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
