export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @cont %lo2
	%hi =l phi @start 4294967296, @cont %hi2
	%cond =w cultl %lo, %hi
	jnz %cond, @body, @done
@body
	%sum =l add %lo, %hi
	%sum1 =l add %sum, 1
	%mid =l shr %sum1, 1
	%q =l udiv %n, %mid
	%le =w culel %mid, %q
	jnz %le, @up_lo, @up_hi
@up_lo
	jmp @cont
@up_hi
	%hi_next =l sub %mid, 1
	jmp @cont
@cont
	%lo2 =l phi @up_lo %mid, @up_hi %lo
	%hi2 =l phi @up_lo %hi, @up_hi %hi_next
	jmp @loop
@done
	ret %lo
}
