export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%x =l loadl %sptr
	%y =l call %f(l %x)
	%dptr =l add %dst, %off
	storel %y, %dptr
	%lt =w csltl %y, %x
	%ltl =l extuw %lt
	%cntnext =l add %cnt, %ltl
	%inext =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
