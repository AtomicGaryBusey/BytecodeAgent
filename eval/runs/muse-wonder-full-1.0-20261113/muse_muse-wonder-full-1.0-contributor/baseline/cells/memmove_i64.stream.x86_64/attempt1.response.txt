memmove_i64:
cmp rdx, 0
jle .Ldone
cmp rdi, rsi
je .Ldone
jb .Lforward
mov rcx, rdx
.Lback:
cmp rcx, 0
jle .Ldone
dec rcx
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
jmp .Lback
.Lforward:
xor ecx, ecx
.Lfwd:
cmp rcx, rdx
jge .Ldone
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
jmp .Lfwd
.Ldone:
xor eax, eax
ret
