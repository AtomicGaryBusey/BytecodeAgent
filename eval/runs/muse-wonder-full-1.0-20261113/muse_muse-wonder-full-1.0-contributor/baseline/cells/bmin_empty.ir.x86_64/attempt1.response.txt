export function l $bmin_empty(l %a, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @empty, @nonempty
@empty
	%neg =l copy -1
	%max =l shr %neg, 1
	ret %max
@nonempty
	%min0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @nonempty 1, @next %inext
	%min =l phi @nonempty %min0, @next %minnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%lt =w csltl %v, %min
	jnz %lt, @take, @keep
@take
	jmp @next
@keep
	jmp @next
@next
	%minnext =l phi @take %v, @keep %min
	%inext =l add %i, 1
	jmp @loop
@end
	ret %min
}
