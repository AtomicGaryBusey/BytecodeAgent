export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer_cond
@outer_cond
	%i =l phi @start 0, @packet_done %i2
	%o =l phi @start 0, @packet_done %o2
	%cond_outer =w csltl %i, %n
	jnz %cond_outer, @run_i_init, @done
@done
	ret %o
@run_i_init
	%p_i =l add %src, %i
	%b_i =w loadub %p_i
	%k0 =l add %i, 1
	jmp @run_i_loop
@run_i_loop
	%k =l phi @run_i_init %k0, @run_i_inc %k_next
	%c_kn =w csltl %k, %n
	jnz %c_kn, @run_i_check128, @run_i_done
@run_i_check128
	%kdiff_i =l sub %k, %i
	%c_k128 =w csltl %kdiff_i, 128
	jnz %c_k128, @run_i_checkbyte, @run_i_done
@run_i_checkbyte
	%p_k =l add %src, %k
	%b_k =w loadub %p_k
	%c_keq =w ceqw %b_k, %b_i
	jnz %c_keq, @run_i_inc, @run_i_done
@run_i_inc
	%k_next =l add %k, 1
	jmp @run_i_loop
@run_i_done
	%k_f =l phi @run_i_loop %k, @run_i_check128 %k, @run_i_checkbyte %k
	%r_i =l sub %k_f, %i
	%c_rlt3 =w csltl %r_i, 3
	jnz %c_rlt3, @literal_init, @emit_repeat
@emit_repeat
	%o_rep_end =l add %o, 2
	%c_rep_nofit =w csltl %cap, %o_rep_end
	jnz %c_rep_nofit, @fail, @emit_repeat_write
@emit_repeat_write
	%r_w =w copy %r_i
	%hdr_r =w sub 257, %r_w
	%p_o =l add %out, %o
	storeb %hdr_r, %p_o
	%p_o1 =l add %p_o, 1
	storeb %b_i, %p_o1
	%o_rep =l copy %o_rep_end
	%i_rep =l add %i, %r_i
	jmp @packet_done
@literal_init
	%j0 =l copy %i
	jmp @lit_scan_loop
@lit_scan_loop
	%j =l phi @literal_init %j0, @lit_scan_inc %j_next
	%c_jn =w csltl %j, %n
	jnz %c_jn, @lit_scan_check128, @lit_scan_done
@lit_scan_check128
	%jdiff =l sub %j, %i
	%c_j128 =w csltl %jdiff, 128
	jnz %c_j128, @run_j_init, @lit_scan_done
@run_j_init
	%p_j =l add %src, %j
	%b_j =w loadub %p_j
	%kj0 =l add %j, 1
	jmp @run_j_loop
@run_j_loop
	%kj =l phi @run_j_init %kj0, @run_j_inc %kj_next
	%c_kjn =w csltl %kj, %n
	jnz %c_kjn, @run_j_check128, @run_j_done
@run_j_check128
	%kjdiff =l sub %kj, %j
	%c_kj128 =w csltl %kjdiff, 128
	jnz %c_kj128, @run_j_checkbyte, @run_j_done
@run_j_checkbyte
	%p_kj =l add %src, %kj
	%b_kj =w loadub %p_kj
	%c_kjeq =w ceqw %b_kj, %b_j
	jnz %c_kjeq, @run_j_inc, @run_j_done
@run_j_inc
	%kj_next =l add %kj, 1
	jmp @run_j_loop
@run_j_done
	%kj_f =l phi @run_j_loop %kj, @run_j_check128 %kj, @run_j_checkbyte %kj
	%r_j =l sub %kj_f, %j
	%c_rjlt3 =w csltl %r_j, 3
	jnz %c_rjlt3, @lit_scan_inc, @lit_scan_done
@lit_scan_inc
	%j_next =l add %j, 1
	jmp @lit_scan_loop
@lit_scan_done
	%j_f =l phi @lit_scan_loop %j, @lit_scan_check128 %j, @run_j_done %j
	%lit_len =l sub %j_f, %i
	%pkt_sz =l add %lit_len, 1
	%o_end_lit =l add %o, %pkt_sz
	%c_lit_nofit =w csltl %cap, %o_end_lit
	jnz %c_lit_nofit, @fail, @literal_write
@literal_write
	%lit_len_w =w copy %lit_len
	%hdr_lit =w sub %lit_len_w, 1
	%p_o_lit =l add %out, %o
	storeb %hdr_lit, %p_o_lit
	%t0 =l copy 0
	jmp @copy_loop
@copy_loop
	%t =l phi @literal_write %t0, @copy_body %t_next
	%c_t =w csltl %t, %lit_len
	jnz %c_t, @copy_body, @literal_write_done
@copy_body
	%s_idx =l add %i, %t
	%p_s =l add %src, %s_idx
	%byte_copy =w loadub %p_s
	%d_base =l add %o, 1
	%d_idx =l add %d_base, %t
	%p_d =l add %out, %d_idx
	storeb %byte_copy, %p_d
	%t_next =l add %t, 1
	jmp @copy_loop
@literal_write_done
	%i_lit_next =l copy %j_f
	%o_lit_next =l copy %o_end_lit
	jmp @packet_done
@packet_done
	%i2 =l phi @emit_repeat_write %i_rep, @literal_write_done %i_lit_next
	%o2 =l phi @emit_repeat_write %o_rep, @literal_write_done %o_lit_next
	jmp @outer_cond
@fail
	%neg1 =l copy -1
	ret %neg1
}
