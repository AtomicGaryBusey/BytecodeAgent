export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%ppc =l alloc8 8
	%pd =l alloc8 8
	%pi =l alloc8 8
	storel 0, %ppc
	storel 0, %pd
	storel 0, %pi
	jmp @loop
@loop
	%pc0 =l loadl %ppc
	%cloop =w csltl %pc0, %len
	jnz %cloop, @fetch, @finish
@fetch
	%offop =l mul %pc0, 8
	%addrop =l add %code, %offop
	%op =l loadl %addrop
	%c0 =w ceql %op, 0
	jnz %c0, @finish, @chk1
@chk1
	%c1 =w ceql %op, 1
	jnz %c1, @do_push, @chk2
@chk2
	%c2 =w ceql %op, 2
	jnz %c2, @do_add, @chk3
@chk3
	%c3 =w ceql %op, 3
	jnz %c3, @do_sub, @chk4
@chk4
	%c4 =w ceql %op, 4
	jnz %c4, @do_mul, @chk5
@chk5
	%c5 =w ceql %op, 5
	jnz %c5, @do_dup, @chk6
@chk6
	%c6 =w ceql %op, 6
	jnz %c6, @do_swap, @chk7
@chk7
	%c7 =w ceql %op, 7
	jnz %c7, @do_drop, @chk8
@chk8
	%c8 =w ceql %op, 8
	jnz %c8, @do_over, @chk9
@chk9
	%c9 =w ceql %op, 9
	jnz %c9, @do_lt, @chk10
@chk10
	%c10 =w ceql %op, 10
	jnz %c10, @do_jz, @chk11
@chk11
	%c11 =w ceql %op, 11
	jnz %c11, @do_jmp, @do_default
@do_push
	%push_pc1 =l add %pc0, 1
	%push_offc =l mul %push_pc1, 8
	%push_addrc =l add %code, %push_offc
	%push_val =l loadl %push_addrc
	%push_d0 =l loadl %pd
	%push_offs =l mul %push_d0, 8
	%push_addrs =l add %stack, %push_offs
	storel %push_val, %push_addrs
	%push_d1 =l add %push_d0, 1
	storel %push_d1, %pd
	%push_pcnew =l add %pc0, 2
	storel %push_pcnew, %ppc
	jmp @loop
@do_add
	%add_d0 =l loadl %pd
	%add_ib =l sub %add_d0, 1
	%add_ia =l sub %add_d0, 2
	%add_offa =l mul %add_ia, 8
	%add_addra =l add %stack, %add_offa
	%add_a =l loadl %add_addra
	%add_offb =l mul %add_ib, 8
	%add_addrb =l add %stack, %add_offb
	%add_b =l loadl %add_addrb
	%add_res =l add %add_a, %add_b
	storel %add_res, %add_addra
	%add_d1 =l sub %add_d0, 1
	storel %add_d1, %pd
	%add_pcnew =l add %pc0, 1
	storel %add_pcnew, %ppc
	jmp @loop
@do_sub
	%sub_d0 =l loadl %pd
	%sub_ib =l sub %sub_d0, 1
	%sub_ia =l sub %sub_d0, 2
	%sub_offa =l mul %sub_ia, 8
	%sub_addra =l add %stack, %sub_offa
	%sub_a =l loadl %sub_addra
	%sub_offb =l mul %sub_ib, 8
	%sub_addrb =l add %stack, %sub_offb
	%sub_b =l loadl %sub_addrb
	%sub_res =l sub %sub_a, %sub_b
	storel %sub_res, %sub_addra
	%sub_d1 =l sub %sub_d0, 1
	storel %sub_d1, %pd
	%sub_pcnew =l add %pc0, 1
	storel %sub_pcnew, %ppc
	jmp @loop
@do_mul
	%mul_d0 =l loadl %pd
	%mul_ib =l sub %mul_d0, 1
	%mul_ia =l sub %mul_d0, 2
	%mul_offa =l mul %mul_ia, 8
	%mul_addra =l add %stack, %mul_offa
	%mul_a =l loadl %mul_addra
	%mul_offb =l mul %mul_ib, 8
	%mul_addrb =l add %stack, %mul_offb
	%mul_b =l loadl %mul_addrb
	%mul_res =l mul %mul_a, %mul_b
	storel %mul_res, %mul_addra
	%mul_d1 =l sub %mul_d0, 1
	storel %mul_d1, %pd
	%mul_pcnew =l add %pc0, 1
	storel %mul_pcnew, %ppc
	jmp @loop
@do_dup
	%dup_d0 =l loadl %pd
	%dup_ib =l sub %dup_d0, 1
	%dup_offb =l mul %dup_ib, 8
	%dup_addrb =l add %stack, %dup_offb
	%dup_b =l loadl %dup_addrb
	%dup_offt =l mul %dup_d0, 8
	%dup_addrt =l add %stack, %dup_offt
	storel %dup_b, %dup_addrt
	%dup_d1 =l add %dup_d0, 1
	storel %dup_d1, %pd
	%dup_pcnew =l add %pc0, 1
	storel %dup_pcnew, %ppc
	jmp @loop
@do_swap
	%swp_d0 =l loadl %pd
	%swp_ib =l sub %swp_d0, 1
	%swp_ia =l sub %swp_d0, 2
	%swp_offa =l mul %swp_ia, 8
	%swp_addra =l add %stack, %swp_offa
	%swp_a =l loadl %swp_addra
	%swp_offb =l mul %swp_ib, 8
	%swp_addrb =l add %stack, %swp_offb
	%swp_b =l loadl %swp_addrb
	storel %swp_b, %swp_addra
	storel %swp_a, %swp_addrb
	%swp_pcnew =l add %pc0, 1
	storel %swp_pcnew, %ppc
	jmp @loop
@do_drop
	%drp_d0 =l loadl %pd
	%drp_d1 =l sub %drp_d0, 1
	storel %drp_d1, %pd
	%drp_pcnew =l add %pc0, 1
	storel %drp_pcnew, %ppc
	jmp @loop
@do_over
	%ovr_d0 =l loadl %pd
	%ovr_ia =l sub %ovr_d0, 2
	%ovr_offa =l mul %ovr_ia, 8
	%ovr_addra =l add %stack, %ovr_offa
	%ovr_a =l loadl %ovr_addra
	%ovr_offt =l mul %ovr_d0, 8
	%ovr_addrt =l add %stack, %ovr_offt
	storel %ovr_a, %ovr_addrt
	%ovr_d1 =l add %ovr_d0, 1
	storel %ovr_d1, %pd
	%ovr_pcnew =l add %pc0, 1
	storel %ovr_pcnew, %ppc
	jmp @loop
@do_lt
	%lt_d0 =l loadl %pd
	%lt_ib =l sub %lt_d0, 1
	%lt_ia =l sub %lt_d0, 2
	%lt_offa =l mul %lt_ia, 8
	%lt_addra =l add %stack, %lt_offa
	%lt_a =l loadl %lt_addra
	%lt_offb =l mul %lt_ib, 8
	%lt_addrb =l add %stack, %lt_offb
	%lt_b =l loadl %lt_addrb
	%lt_c =w csltl %lt_a, %lt_b
	%lt_res =l extuw %lt_c
	storel %lt_res, %lt_addra
	%lt_d1 =l sub %lt_d0, 1
	storel %lt_d1, %pd
	%lt_pcnew =l add %pc0, 1
	storel %lt_pcnew, %ppc
	jmp @loop
@do_jz
	%jz_d0 =l loadl %pd
	%jz_ib =l sub %jz_d0, 1
	%jz_offb =l mul %jz_ib, 8
	%jz_addrb =l add %stack, %jz_offb
	%jz_v =l loadl %jz_addrb
	storel %jz_ib, %pd
	%jz_pc1 =l add %pc0, 1
	%jz_offc =l mul %jz_pc1, 8
	%jz_addrc =l add %code, %jz_offc
	%jz_t =l loadl %jz_addrc
	%jz_c =w ceql %jz_v, 0
	jnz %jz_c, @jz_take, @jz_fall
@jz_take
	storel %jz_t, %ppc
	jmp @loop
@jz_fall
	%jz_pcnew =l add %pc0, 2
	storel %jz_pcnew, %ppc
	jmp @loop
@do_jmp
	%jmp_pc1 =l add %pc0, 1
	%jmp_offc =l mul %jmp_pc1, 8
	%jmp_addrc =l add %code, %jmp_offc
	%jmp_t =l loadl %jmp_addrc
	storel %jmp_t, %ppc
	jmp @loop
@do_default
	%def_pcnew =l add %pc0, 1
	storel %def_pcnew, %ppc
	jmp @loop
@finish
	%fin_d0 =l loadl %pd
	storel %fin_d0, %pi
	jmp @zloop
@zloop
	%z_i0 =l loadl %pi
	%z_c =w csltl %z_i0, %cap
	jnz %z_c, @zbody, @ret_chk
@zbody
	%z_off =l mul %z_i0, 8
	%z_addr =l add %stack, %z_off
	storel 0, %z_addr
	%z_i1 =l add %z_i0, 1
	storel %z_i1, %pi
	jmp @zloop
@ret_chk
	%ret_d0 =l loadl %pd
	%ret_c =w ceql %ret_d0, 0
	jnz %ret_c, @ret_zero, @ret_top
@ret_zero
	ret 0
@ret_top
	%rt_itop =l sub %ret_d0, 1
	%rt_off =l mul %rt_itop, 8
	%rt_addr =l add %stack, %rt_off
	%rt_val =l loadl %rt_addr
	ret %rt_val
}
