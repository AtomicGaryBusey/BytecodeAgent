export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_loop
@init_loop
	%k =l phi @start 0, @init_body %k_next
	%init_cond =w cslel %k, %n
	jnz %init_cond, @init_body, @outer_loop
@init_body
	%k_off =l mul %k, 8
	%k_ptr =l add %row, %k_off
	storel 0, %k_ptr
	%k_next =l add %k, 1
	jmp @init_loop
@outer_loop
	%i =l phi @init_loop 0, @outer_next %i_next
	%i_cond =w csltl %i, %m
	jnz %i_cond, @inner_loop, @done
@inner_loop
	%j =l phi @outer_loop 1, @inner_next %j_next
	%diag =l phi @outer_loop 0, @inner_next %diag_next
	%j_cond =w cslel %j, %n
	jnz %j_cond, @inner_body, @outer_next
@inner_body
	%j_off =l mul %j, 8
	%row_j_ptr =l add %row, %j_off
	%up =l loadl %row_j_ptr
	%x_ptr =l add %x, %i
	%xb =w loadub %x_ptr
	%j_minus1 =l sub %j, 1
	%y_ptr =l add %y, %j_minus1
	%yb =w loadub %y_ptr
	%ceq =w ceqw %xb, %yb
	jnz %ceq, @match, @nomatch
@match
	%match_val =l add %diag, 1
	storel %match_val, %row_j_ptr
	jmp @inner_next
@nomatch
	%left_off =l mul %j_minus1, 8
	%row_left_ptr =l add %row, %left_off
	%left =l loadl %row_left_ptr
	%up_ge_left =w csgel %up, %left
	jnz %up_ge_left, @use_up, @use_left
@use_up
	storel %up, %row_j_ptr
	jmp @inner_next
@use_left
	storel %left, %row_j_ptr
	jmp @inner_next
@inner_next
	%diag_next =l copy %up
	%j_next =l add %j, 1
	jmp @inner_loop
@outer_next
	%i_next =l add %i, 1
	jmp @outer_loop
@done
	%n_off =l mul %n, 8
	%row_n_ptr =l add %row, %n_off
	%result =l loadl %row_n_ptr
	ret %result
}
