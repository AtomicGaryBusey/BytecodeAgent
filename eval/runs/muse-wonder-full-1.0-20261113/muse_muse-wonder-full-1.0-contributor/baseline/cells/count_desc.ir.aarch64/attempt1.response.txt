export function l $count_desc(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%i1 =l add %i, 1
	%c =w csltl %i1, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v0 =l loadl %p
	%p1 =l add %p, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l copy %i1
	jmp @loop
@end
	ret %cnt
}
