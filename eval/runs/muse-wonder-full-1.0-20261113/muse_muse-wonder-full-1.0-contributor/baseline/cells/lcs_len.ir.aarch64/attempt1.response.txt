export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_loop
@init_loop
	%j =l phi @start 0, @init_body %j_next
	%init_cond =w cslel %j, %n
	jnz %init_cond, @init_body, @init_done
@init_body
	%j_off =l mul %j, 8
	%j_ptr =l add %row, %j_off
	storel 0, %j_ptr
	%j_next =l add %j, 1
	jmp @init_loop
@init_done
	jmp @outer_loop
@outer_loop
	%i =l phi @init_done 0, @outer_next %i_next
	%outer_cond =w csltl %i, %m
	jnz %outer_cond, @outer_body, @done
@outer_body
	%x_ptr =l add %x, %i
	%xi =w loadub %x_ptr
	jmp @inner_loop
@inner_loop
	%jj =l phi @outer_body 1, @inner_next %jj_next
	%diag =l phi @outer_body 0, @inner_next %up
	%inner_cond =w cslel %jj, %n
	jnz %inner_cond, @inner_body, @outer_next
@inner_body
	%jj_off =l mul %jj, 8
	%rowj_ptr =l add %row, %jj_off
	%up =l loadl %rowj_ptr
	%jm1 =l sub %jj, 1
	%y_ptr =l add %y, %jm1
	%yj =w loadub %y_ptr
	%eq =w ceqw %xi, %yj
	jnz %eq, @eq_block, @ne_block
@eq_block
	%new_eq =l add %diag, 1
	jmp @inner_store
@ne_block
	%jm1_off =l mul %jm1, 8
	%rowjm1_ptr =l add %row, %jm1_off
	%left =l loadl %rowjm1_ptr
	%ge =w csgel %up, %left
	jnz %ge, @max_up, @max_left
@max_up
	jmp @inner_store
@max_left
	jmp @inner_store
@inner_store
	%newval =l phi @eq_block %new_eq, @max_up %up, @max_left %left
	storel %newval, %rowj_ptr
	jmp @inner_next
@inner_next
	%jj_next =l add %jj, 1
	jmp @inner_loop
@outer_next
	%i_next =l add %i, 1
	jmp @outer_loop
@done
	%n_off =l mul %n, 8
	%rown_ptr =l add %row, %n_off
	%res =l loadl %rown_ptr
	ret %res
}
