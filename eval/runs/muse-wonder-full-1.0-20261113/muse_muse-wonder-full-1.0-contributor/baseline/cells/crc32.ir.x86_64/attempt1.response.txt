export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @next %i1
	%crc =w phi @start -1, @next %crc2
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crcx =w xor %crc, %b
	jmp @inner
@inner
	%j =l phi @body 0, @ibody %j1
	%ci =w phi @body %crcx, @ibody %cnext
	%icmp =w csltl %j, 8
	jnz %icmp, @ibody, @next
@ibody
	%bit =w and %ci, 1
	%shift =w shr %ci, 1
	%mask =w sub 0, %bit
	%pm =w and %mask, -306674912
	%cnext =w xor %shift, %pm
	%j1 =l add %j, 1
	jmp @inner
@next
	%crc2 =w copy %ci
	%i1 =l add %i, 1
	jmp @outer
@done
	%final =w xor %crc, -1
	%res =l extuw %final
	ret %res
}
