export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @outer_next %i_next
	%crc =w phi @start -1, @outer_next %c
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crc1 =w xor %crc, %b
	jmp @inner
@inner
	%k =l phi @body 0, @inner_merge %k_next
	%c =w phi @body %crc1, @inner_merge %c_next
	%kcond =w csltl %k, 8
	jnz %kcond, @inner_body, @outer_next
@inner_body
	%bit =w and %c, 1
	%shifted =w shr %c, 1
	jnz %bit, @inner_xor, @inner_noxor
@inner_xor
	%c_xor =w xor %shifted, -306674912
	jmp @inner_merge
@inner_noxor
	jmp @inner_merge
@inner_merge
	%c_next =w phi @inner_xor %c_xor, @inner_noxor %shifted
	%k_next =l add %k, 1
	jmp @inner
@outer_next
	%i_next =l add %i, 1
	jmp @outer
@done
	%final =w xor %crc, -1
	%res =l extuw %final
	ret %res
}
