sdivrem:
mov r8, rdx
test rsi, rsi
jz .Lzero
cmp rsi, -1
je .Lcheck
.Ldiv:
mov rax, rdi
mov rdx, rax
sar rdx, 63
idiv rsi
mov qword ptr [r8], rdx
ret
.Lzero:
mov qword ptr [r8], rdi
xor eax, eax
ret
.Lcheck:
mov rcx, 1
shl rcx, 63
cmp rdi, rcx
jne .Ldiv
xor ecx, ecx
mov qword ptr [r8], rcx
mov rax, rdi
ret
