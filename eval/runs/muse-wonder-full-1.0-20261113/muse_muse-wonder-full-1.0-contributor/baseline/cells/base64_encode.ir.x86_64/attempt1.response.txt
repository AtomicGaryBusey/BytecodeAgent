export function l $base64_encode(l %src, l %n, l %out) {
@start
jmp @outer
@outer
%i =l phi @start 0, @next %inext
%o =l phi @start 0, @next %onext
%couter =w csltl %i, %n
jnz %couter, @body, @done
@done
ret %o
@body
%rem =l sub %n, %i
%p0 =l add %src, %i
%b0 =l loadub %p0
%lt2 =w csltl %rem, 2
jnz %lt2, @no1, @load1
@load1
%ip1 =l add %i, 1
%p1 =l add %src, %ip1
%b1load =l loadub %p1
jmp @after1
@no1
jmp @after1
@after1
%b1 =l phi @load1 %b1load, @no1 0
%lt3 =w csltl %rem, 3
jnz %lt3, @no2, @load2
@load2
%ip2 =l add %i, 2
%p2 =l add %src, %ip2
%b2load =l loadub %p2
jmp @after2
@no2
jmp @after2
@after2
%b2 =l phi @load2 %b2load, @no2 0
%b0sh =l shl %b0, 16
%b1sh =l shl %b1, 8
%vtmp =l or %b0sh, %b1sh
%v =l or %vtmp, %b2
%lt3l =l extuw %lt3
%ge3l =l xor %lt3l, 1
%remp1 =l add %rem, 1
%ndge =l mul %ge3l, 4
%ndlt =l mul %lt3l, %remp1
%ndata =l add %ndge, %ndlt
%stge =l mul %ge3l, 3
%stlt =l mul %lt3l, %rem
%step =l add %stge, %stlt
jmp @inner
@inner
%k =l phi @after2 0, @innernext %knext
%cur =l phi @after2 %v, @innernext %cur2
%klt4 =w csltl %k, 4
jnz %klt4, @innerbody, @next
@innerbody
%shifted =l shr %cur, 18
%idx =l and %shifted, 63
%lt26 =w csltl %idx, 26
%lt52 =w csltl %idx, 52
%lt62 =w csltl %idx, 62
%eq62 =w ceql %idx, 62
%eq63 =w ceql %idx, 63
%up1 =l extuw %lt26
%lt52l =l extuw %lt52
%lt62l =l extuw %lt62
%plus1 =l extuw %eq62
%slash1 =l extuw %eq63
%lo1 =l sub %lt52l, %up1
%dig1 =l sub %lt62l, %lt52l
%upchar =l add %idx, 65
%lochar =l add %idx, 71
%digchar =l sub %idx, 4
%mup =l mul %up1, %upchar
%mlo =l mul %lo1, %lochar
%mdig =l mul %dig1, %digchar
%mplus =l mul %plus1, 43
%mslash =l mul %slash1, 47
%msum1 =l add %mup, %mlo
%msum2 =l add %msum1, %mdig
%msum3 =l add %msum2, %mplus
%mapped =l add %msum3, %mslash
%isdata =w csltl %k, %ndata
%isdatal =l extuw %isdata
%pad1 =l xor %isdatal, 1
%chdata =l mul %isdatal, %mapped
%chpad =l mul %pad1, 61
%ch =l add %chdata, %chpad
%chw =w copy %ch
%ok =l add %o, %k
%pout =l add %out, %ok
storeb %chw, %pout
jmp @innernext
@innernext
%knext =l add %k, 1
%cur2 =l shl %cur, 6
jmp @inner
@next
%inext =l add %i, %step
%onext =l add %o, 4
jmp @outer
}
