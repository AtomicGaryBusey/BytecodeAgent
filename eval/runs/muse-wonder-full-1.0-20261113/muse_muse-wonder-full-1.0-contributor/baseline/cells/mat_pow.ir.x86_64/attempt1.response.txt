export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%tmp2 =l add %tmp, %nn8
	%two_nn =l add %nn, %nn
	%p_iter =l alloc8 8
	%p_idx =l alloc8 8
	%p_i =l alloc8 8
	%p_j =l alloc8 8
	%p_t =l alloc8 8
	%p_sum =l alloc8 8
	storel 0, %p_idx
	jmp @zero_out_cond
@zero_out_cond
	%zo_idx =l loadl %p_idx
	%zo_c =w csltl %zo_idx, %nn
	jnz %zo_c, @zero_out_body, @diag_init
@zero_out_body
	%zo_off =l mul %zo_idx, 8
	%zo_ptr =l add %out, %zo_off
	storel 0, %zo_ptr
	%zo_next =l add %zo_idx, 1
	storel %zo_next, %p_idx
	jmp @zero_out_cond
@diag_init
	storel 0, %p_i
	jmp @diag_cond
@diag_cond
	%d_i =l loadl %p_i
	%d_c =w csltl %d_i, %n
	jnz %d_c, @diag_body, @copy_init
@diag_body
	%d_in =l mul %d_i, %n
	%d_pos =l add %d_in, %d_i
	%d_off =l mul %d_pos, 8
	%d_ptr =l add %out, %d_off
	storel 1, %d_ptr
	%d_next =l add %d_i, 1
	storel %d_next, %p_i
	jmp @diag_cond
@copy_init
	storel 0, %p_idx
	jmp @copy_cond
@copy_cond
	%cp_idx =l loadl %p_idx
	%cp_c =w csltl %cp_idx, %nn
	jnz %cp_c, @copy_body, @iter_init
@copy_body
	%cp_off =l mul %cp_idx, 8
	%cp_srcp =l add %a, %cp_off
	%cp_val =l loadl %cp_srcp
	%cp_dstp =l add %tmp, %cp_off
	storel %cp_val, %cp_dstp
	%cp_next =l add %cp_idx, 1
	storel %cp_next, %p_idx
	jmp @copy_cond
@iter_init
	storel 0, %p_iter
	jmp @iter_cond
@iter_cond
	%it_iter =l loadl %p_iter
	%it_c =w csltl %it_iter, %k
	jnz %it_c, @mul_i_init, @cleanup_init
@mul_i_init
	storel 0, %p_i
	jmp @mul_i_cond
@mul_i_cond
	%mi_i =l loadl %p_i
	%mi_c =w csltl %mi_i, %n
	jnz %mi_c, @mul_j_init, @copyback_init
@mul_j_init
	storel 0, %p_j
	jmp @mul_j_cond
@mul_j_cond
	%mj_j =l loadl %p_j
	%mj_c =w csltl %mj_j, %n
	jnz %mj_c, @mul_t_init, @mul_i_next
@mul_t_init
	storel 0, %p_sum
	storel 0, %p_t
	jmp @mul_t_cond
@mul_t_cond
	%mt_t =l loadl %p_t
	%mt_c =w csltl %mt_t, %n
	jnz %mt_c, @mul_t_body, @mul_store
@mul_t_body
	%mtb_i =l loadl %p_i
	%mtb_j =l loadl %p_j
	%mtb_t =l loadl %p_t
	%mtb_in =l mul %mtb_i, %n
	%mtb_p1 =l add %mtb_in, %mtb_t
	%mtb_off1 =l mul %mtb_p1, 8
	%mtb_ptr1 =l add %out, %mtb_off1
	%mtb_v1 =l loadl %mtb_ptr1
	%mtb_tn =l mul %mtb_t, %n
	%mtb_p2 =l add %mtb_tn, %mtb_j
	%mtb_off2 =l mul %mtb_p2, 8
	%mtb_ptr2 =l add %tmp, %mtb_off2
	%mtb_v2 =l loadl %mtb_ptr2
	%mtb_prod =l mul %mtb_v1, %mtb_v2
	%mtb_sum =l loadl %p_sum
	%mtb_nsum =l add %mtb_sum, %mtb_prod
	storel %mtb_nsum, %p_sum
	%mtb_nextt =l add %mtb_t, 1
	storel %mtb_nextt, %p_t
	jmp @mul_t_cond
@mul_store
	%ms_i =l loadl %p_i
	%ms_j =l loadl %p_j
	%ms_in =l mul %ms_i, %n
	%ms_pos =l add %ms_in, %ms_j
	%ms_off =l mul %ms_pos, 8
	%ms_ptr =l add %tmp2, %ms_off
	%ms_sum =l loadl %p_sum
	storel %ms_sum, %ms_ptr
	%ms_nextj =l add %ms_j, 1
	storel %ms_nextj, %p_j
	jmp @mul_j_cond
@mul_i_next
	%min_i =l loadl %p_i
	%min_next =l add %min_i, 1
	storel %min_next, %p_i
	jmp @mul_i_cond
@copyback_init
	storel 0, %p_idx
	jmp @copyback_cond
@copyback_cond
	%cb_idx =l loadl %p_idx
	%cb_c =w csltl %cb_idx, %nn
	jnz %cb_c, @copyback_body, @iter_next
@copyback_body
	%cb_off =l mul %cb_idx, 8
	%cb_srcp =l add %tmp2, %cb_off
	%cb_val =l loadl %cb_srcp
	%cb_dstp =l add %out, %cb_off
	storel %cb_val, %cb_dstp
	%cb_next =l add %cb_idx, 1
	storel %cb_next, %p_idx
	jmp @copyback_cond
@iter_next
	%in_iter =l loadl %p_iter
	%in_next =l add %in_iter, 1
	storel %in_next, %p_iter
	jmp @iter_cond
@cleanup_init
	storel 0, %p_idx
	jmp @cleanup_cond
@cleanup_cond
	%cl_idx =l loadl %p_idx
	%cl_c =w csltl %cl_idx, %two_nn
	jnz %cl_c, @cleanup_body, @done
@cleanup_body
	%cl_off =l mul %cl_idx, 8
	%cl_ptr =l add %tmp, %cl_off
	storel 0, %cl_ptr
	%cl_next =l add %cl_idx, 1
	storel %cl_next, %p_idx
	jmp @cleanup_cond
@done
	ret 0
}
