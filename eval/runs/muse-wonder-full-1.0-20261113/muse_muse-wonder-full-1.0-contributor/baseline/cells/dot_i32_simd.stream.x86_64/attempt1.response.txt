dot_i32_simd:
pxor xmm0, xmm0
xor ecx, ecx
.Lvloop:
lea r8, [rcx + 4]
cmp r8, rdx
jg .Lvdone
movdqu xmm1, [rdi + rcx*4]
movdqu xmm2, [rsi + rcx*4]
pmovsxdq xmm3, xmm1
pmovsxdq xmm4, xmm2
pmuldq xmm3, xmm4
paddq xmm0, xmm3
psrldq xmm1, 8
psrldq xmm2, 8
pmovsxdq xmm3, xmm1
pmovsxdq xmm4, xmm2
pmuldq xmm3, xmm4
paddq xmm0, xmm3
add rcx, 4
jmp .Lvloop
.Lvdone:
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
.Lsloop:
cmp rcx, rdx
jge .Ldone
movsxd r8, dword ptr [rdi + rcx*4]
movsxd r9, dword ptr [rsi + rcx*4]
imul r8, r9
add rax, r8
add rcx, 1
jmp .Lsloop
.Ldone:
ret
