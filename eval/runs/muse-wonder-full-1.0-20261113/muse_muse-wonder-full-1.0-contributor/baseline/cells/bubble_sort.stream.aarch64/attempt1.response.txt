bubble_sort:

b.lt .Ldone
mov x9, #0
.Louter:
sub x11, x1, #1
sub x11, x11, x9
cmp x11, #0
b.le .Ldone
mov x10, #0
.Linner:
cmp x10, x11
b.ge .Linner_done
ldr x12, [x0, x10, lsl #3]
add x14, x10, #1
ldr x13, [x0, x14, lsl #3]
cmp x12, x13
b.le .Lno_swap

str x12, [x0, x14, lsl #3]
.Lno_swap:
add x10, x10, #1

.Linner_done:
add x9, x9, #1
b .Louter
.Ldone:
mov x0, #0
ret
