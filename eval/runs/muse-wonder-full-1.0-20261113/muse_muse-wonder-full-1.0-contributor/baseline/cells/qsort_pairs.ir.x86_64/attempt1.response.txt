export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @sort
@sort
	%last =l sub %n, 1
	%loff =l mul %last, 16
	%lptr =l add %a, %loff
	%pk =l loadl %lptr
	%lptrv =l add %lptr, 8
	%pv =l loadl %lptrv
	jmp @loop
@loop
	%i =l phi @sort 0, @advance %inext
	%j =l phi @sort 0, @advance %jnext
	%cl =w csltl %j, %last
	jnz %cl, @body, @after
@body
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%kj =l loadl %jptr
	%jptrv =l add %jptr, 8
	%vj =l loadl %jptrv
	%c1 =w csltl %kj, %pk
	%c2 =w ceql %kj, %pk
	%c3 =w csltl %vj, %pv
	%c23 =w and %c2, %c3
	%less =w or %c1, %c23
	jnz %less, @swap, @advance
@swap
	%ioff =l mul %i, 16
	%iptr =l add %a, %ioff
	%ki =l loadl %iptr
	%iptrv =l add %iptr, 8
	%vi =l loadl %iptrv
	storel %kj, %iptr
	storel %vj, %iptrv
	storel %ki, %jptr
	storel %vi, %jptrv
	%iinc =l add %i, 1
	jmp @advance
@advance
	%inext =l phi @body %i, @swap %iinc
	%jnext =l add %j, 1
	jmp @loop
@after
	%pioff =l mul %i, 16
	%piptr =l add %a, %pioff
	%pki =l loadl %piptr
	%piptrv =l add %piptr, 8
	%pvi =l loadl %piptrv
	storel %pk, %piptr
	storel %pv, %piptrv
	storel %pki, %lptr
	storel %pvi, %lptrv
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%rn =l sub %n, %ip1
	%roff =l mul %ip1, 16
	%rptr =l add %a, %roff
	%r2 =l call $qsort_pairs(l %rptr, l %rn)
	jmp @done
@done
	ret 0
}
