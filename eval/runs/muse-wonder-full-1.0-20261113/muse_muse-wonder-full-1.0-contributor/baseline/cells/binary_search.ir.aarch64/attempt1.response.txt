export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @back %lo2
	%hi =l phi @start %n, @back %hi2
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @noteq
@found
	ret %mid
@noteq
	%lt =w csltl %val, %key
	jnz %lt, @ltblock, @gtblock
@ltblock
	%lo_next =l add %mid, 1
	jmp @back
@gtblock
	jmp @back
@back
	%lo2 =l phi @ltblock %lo_next, @gtblock %lo
	%hi2 =l phi @ltblock %hi, @gtblock %mid
	jmp @loop
@notfound
	%neg =l sub 0, 1
	ret %neg
}
