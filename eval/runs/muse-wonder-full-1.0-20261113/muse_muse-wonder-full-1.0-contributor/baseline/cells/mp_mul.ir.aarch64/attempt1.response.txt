export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
 %pi =l alloc8 8
 %pj =l alloc8 8
 %pcarry =l alloc8 8
 %pk =l alloc8 8
 %total =l add %na, %nb
 storel 0, %pi
 jmp @initcond
@initcond
 %ici =l loadl %pi
 %icc =w csltl %ici, %total
 jnz %icc, @initbody, @initdone
@initbody
 %ibi =l loadl %pi
 %iboff =l mul %ibi, 8
 %ibptr =l add %r, %iboff
 storel 0, %ibptr
 %ibnext =l add %ibi, 1
 storel %ibnext, %pi
 jmp @initcond
@initdone
 storel 0, %pi
 jmp @outercond
@outercond
 %oci =l loadl %pi
 %occ =w csltl %oci, %na
 jnz %occ, @outerbody, @outerdone
@outerbody
 storel 0, %pcarry
 storel 0, %pj
 jmp @innercond
@innercond
 %icj =l loadl %pj
 %ijc =w csltl %icj, %nb
 jnz %ijc, @innerbody, @innerdone
@innerbody
 %nbi =l loadl %pi
 %nbj =l loadl %pj
 %nbcarry =l loadl %pcarry
 %aioff =l mul %nbi, 8
 %aiptr =l add %a, %aioff
 %aval =l loadl %aiptr
 %bjoff =l mul %nbj, 8
 %bjptr =l add %b, %bjoff
 %bval =l loadl %bjptr
 %ma0 =l and %aval, 4294967295
 %ma1 =l shr %aval, 32
 %mb0 =l and %bval, 4294967295
 %mb1 =l shr %bval, 32
 %mp0 =l mul %ma0, %mb0
 %mp1 =l mul %ma0, %mb1
 %mp2 =l mul %ma1, %mb0
 %mp3 =l mul %ma1, %mb1
 %mp0lo =l and %mp0, 4294967295
 %mp0hi =l shr %mp0, 32
 %mp1lo =l and %mp1, 4294967295
 %mp1hi =l shr %mp1, 32
 %mp2lo =l and %mp2, 4294967295
 %mp2hi =l shr %mp2, 32
 %mp3lo =l and %mp3, 4294967295
 %mp3hi =l shr %mp3, 32
 %ms1a =l add %mp0hi, %mp1lo
 %ms1 =l add %ms1a, %mp2lo
 %mw1 =l and %ms1, 4294967295
 %mc1 =l shr %ms1, 32
 %ms2a =l add %mp1hi, %mp2hi
 %ms2b =l add %ms2a, %mp3lo
 %ms2 =l add %ms2b, %mc1
 %mw2 =l and %ms2, 4294967295
 %mc2 =l shr %ms2, 32
 %mw3 =l add %mp3hi, %mc2
 %mlosh =l shl %mw1, 32
 %mlo =l add %mp0lo, %mlosh
 %mhish =l shl %mw3, 32
 %mhi =l add %mw2, %mhish
 %sidx =l add %nbi, %nbj
 %soff =l mul %sidx, 8
 %sptr =l add %r, %soff
 %rval =l loadl %sptr
 %t1 =l add %rval, %mlo
 %c1w =w cultl %t1, %rval
 %c1 =l extuw %c1w
 %t2 =l add %t1, %nbcarry
 %c2w =w cultl %t2, %t1
 %c2 =l extuw %c2w
 %nca =l add %mhi, %c1
 %ncarry =l add %nca, %c2
 storel %t2, %sptr
 storel %ncarry, %pcarry
 %nbnext =l add %nbj, 1
 storel %nbnext, %pj
 jmp @innercond
@innerdone
 %idi =l loadl %pi
 %idcarry =l loadl %pcarry
 %ididx =l add %idi, %nb
 %idoff =l mul %ididx, 8
 %idptr =l add %r, %idoff
 %idold =l loadl %idptr
 %idnew =l add %idold, %idcarry
 storel %idnew, %idptr
 %idinext =l add %idi, 1
 storel %idinext, %pi
 jmp @outercond
@outerdone
 storel %total, %pk
 jmp @trimcond
@trimcond
 %tck =l loadl %pk
 %tcc =w csgtl %tck, 0
 jnz %tcc, @trimbody, @trimdone
@trimbody
 %tbk =l loadl %pk
 %tbkm1 =l sub %tbk, 1
 %tboff =l mul %tbkm1, 8
 %tbptr =l add %r, %tboff
 %tbval =l loadl %tbptr
 %tbz =w ceql %tbval, 0
 jnz %tbz, @trimdec, @trimdone
@trimdec
 %tdk =l loadl %pk
 %tdkm1 =l sub %tdk, 1
 storel %tdkm1, %pk
 jmp @trimcond
@trimdone
 %retv =l loadl %pk
 ret %retv
}
