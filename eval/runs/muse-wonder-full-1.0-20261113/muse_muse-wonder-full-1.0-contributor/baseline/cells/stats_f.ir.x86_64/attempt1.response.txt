export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @merged %inext
	%min =l phi @start %y0, @merged %minnext
	%max =l phi @start %y0, @merged %maxnext
	%sum =l phi @start %y0, @merged %sumnext
	%odd =l phi @start %odd0, @merged %oddnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%ai =l loadl %ptr
	%y =l call %f(l %ai)
	%ismin =w csltl %y, %min
	jnz %ismin, @setmin, @keepmin
@setmin
	jmp @minmerged
@keepmin
	jmp @minmerged
@minmerged
	%min1 =l phi @setmin %y, @keepmin %min
	%ismax =w csgtl %y, %max
	jnz %ismax, @setmax, @keepmax
@setmax
	jmp @merged
@keepmax
	jmp @merged
@merged
	%max1 =l phi @setmax %y, @keepmax %max
	%minnext =l copy %min1
	%maxnext =l copy %max1
	%sumnext =l add %sum, %y
	%isodd =l and %y, 1
	%oddnext =l add %odd, %isodd
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
