export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %inext
	%cnt =l phi @init 0, @body %cntnext
	%done =w csgel %i, %limit
	jnz %done, @end, @body
@end
	ret %cnt
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%gt =w csgtl %x, %y
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l add %i, 1
	jmp @loop
}
