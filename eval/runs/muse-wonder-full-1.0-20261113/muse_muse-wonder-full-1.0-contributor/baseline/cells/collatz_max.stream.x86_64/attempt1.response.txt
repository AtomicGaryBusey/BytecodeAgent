collatz_max:
mov rax, 0
mov rsi, 1
.Louter:
cmp rsi, rdi
jg .Ldone
mov rcx, rsi
mov rdx, 0
.Linner:
cmp rcx, 1
je .Linner_done
test rcx, 1
jz .Leven
imul rcx, rcx, 3
inc rcx
jmp .Lstep
.Leven:
shr rcx, 1
.Lstep:
inc rdx
jmp .Linner
.Linner_done:
cmp rdx, rax
jle .Lnext
mov rax, rdx
.Lnext:
inc rsi
jmp .Louter
.Ldone:
ret
