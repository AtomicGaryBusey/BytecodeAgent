export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @fwd, @bwd
@fwd
	%i =l phi @start 0, @fwdb %inext
	%fcond =w csltl %i, %n
	jnz %fcond, @fwdb, @end
@fwdb
	%foff =l mul %i, 8
	%fsp =l add %src, %foff
	%fv =l loadl %fsp
	%fdp =l add %dst, %foff
	storel %fv, %fdp
	%inext =l add %i, 1
	jmp @fwd
@bwd
	%j =l phi @start %n, @bwdb %jnext
	%bcond =w csgtl %j, 0
	jnz %bcond, @bwdb, @end
@bwdb
	%jnext =l sub %j, 1
	%boff =l mul %jnext, 8
	%bsp =l add %src, %boff
	%bv =l loadl %bsp
	%bdp =l add %dst, %boff
	storel %bv, %bdp
	jmp @bwd
@end
	ret 0
}
