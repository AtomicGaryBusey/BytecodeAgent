export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%si =l phi @start 0, @advance %sia
	%oi =l phi @start 0, @advance %oia
	%has =w csltl %si, %n
	jnz %has, @header, @retok
@retok
	ret %oi
@header
	%psrc =l add %src, %si
	%h =w loadsb %psrc
	%si1 =l add %si, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @sign
@nop
	jmp @advance
@sign
	%nonneg =w csgew %h, 0
	jnz %nonneg, @literal, @repeat
@literal
	%hl =l extsw %h
	%litlen =l add %hl, 1
	%litend =l add %si1, %litlen
	%litok =w cslel %litend, %n
	jnz %litok, @litcap, @retm2
@litcap
	%litoiend =l add %oi, %litlen
	%litcapok =w cslel %litoiend, %cap
	jnz %litcapok, @litcopyinit, @littruncinit
@litcopyinit
	jmp @litcopyloop
@litcopyloop
	%i =l phi @litcopyinit 0, @litcopybody %inext
	%ilt =w csltl %i, %litlen
	jnz %ilt, @litcopybody, @litdone
@litcopybody
	%lsidx =l add %si1, %i
	%lsptr =l add %src, %lsidx
	%lbyte =w loadub %lsptr
	%loidx =l add %oi, %i
	%loptr =l add %out, %loidx
	storeb %lbyte, %loptr
	%inext =l add %i, 1
	jmp @litcopyloop
@litdone
	%litsiend =l copy %litend
	%litoiend2 =l copy %litoiend
	jmp @advance
@littruncinit
	%litrem =l sub %cap, %oi
	jmp @littruncloop
@littruncloop
	%j =l phi @littruncinit 0, @littruncbody %jnext
	%jlt =w csltl %j, %litrem
	jnz %jlt, @littruncbody, @retm1
@littruncbody
	%tsidx =l add %si1, %j
	%tsptr =l add %src, %tsidx
	%tbyte =w loadub %tsptr
	%toidx =l add %oi, %j
	%toptr =l add %out, %toidx
	storeb %tbyte, %toptr
	%jnext =l add %j, 1
	jmp @littruncloop
@repeat
	%hr =l extsw %h
	%replen =l sub 1, %hr
	%repend =l add %si1, 1
	%repok =w cslel %repend, %n
	jnz %repok, @repl, @retm2
@repl
	%rsptr =l add %src, %si1
	%rbyte =w loadub %rsptr
	%repsiend =l copy %repend
	%repoiend =l add %oi, %replen
	%repcapok =w cslel %repoiend, %cap
	jnz %repcapok, @repcopyinit, @reptruncinit
@repcopyinit
	jmp @repcopyloop
@repcopyloop
	%k =l phi @repcopyinit 0, @repcopybody %knext
	%klt =w csltl %k, %replen
	jnz %klt, @repcopybody, @repdone
@repcopybody
	%roidx =l add %oi, %k
	%roptr =l add %out, %roidx
	storeb %rbyte, %roptr
	%knext =l add %k, 1
	jmp @repcopyloop
@repdone
	%repsiend2 =l copy %repsiend
	%repoiend2 =l copy %repoiend
	jmp @advance
@reptruncinit
	%reprem =l sub %cap, %oi
	jmp @reptruncloop
@reptruncloop
	%m =l phi @reptruncinit 0, @reptruncbody %mnext
	%mlt =w csltl %m, %reprem
	jnz %mlt, @reptruncbody, @retm1
@reptruncbody
	%rtoidx =l add %oi, %m
	%rtoptr =l add %out, %rtoidx
	storeb %rbyte, %rtoptr
	%mnext =l add %m, 1
	jmp @reptruncloop
@advance
	%sia =l phi @nop %si1, @litdone %litsiend, @repdone %repsiend2
	%oia =l phi @nop %oi, @litdone %litoiend2, @repdone %repoiend2
	jmp @loop
@retm1
	ret -1
@retm2
	ret -2
}
