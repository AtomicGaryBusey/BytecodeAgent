export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @outerinc %inext
	%n1 =l sub %n, 1
	%ocond =w csltl %i, %n1
	jnz %ocond, @inner, @done
@inner
	%j =l phi @outer 0, @innerinc %jnext
	%limit =l sub %n1, %i
	%icond =w csltl %j, %limit
	jnz %icond, @body, @outerinc
@body
	%off0 =l mul %j, 8
	%p0 =l add %a, %off0
	%v0 =l loadl %p0
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @swap, @innerinc
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @innerinc
@innerinc
	%jnext =l add %j, 1
	jmp @inner
@outerinc
	%inext =l add %i, 1
	jmp @outer
@done
	ret 0
}
