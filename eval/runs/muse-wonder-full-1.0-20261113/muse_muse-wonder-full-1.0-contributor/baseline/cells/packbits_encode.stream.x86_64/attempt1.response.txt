packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor r15, r15
xor rbp, rbp
.Louter:
cmp r15, r12
jge .Ldone
movzx r11, byte ptr [rbx + r15]
mov r10, 1
.Lrun_loop:
cmp r10, 128
jge .Lrun_done
lea rax, [r15 + r10]
cmp rax, r12
jge .Lrun_done
movzx ecx, byte ptr [rbx + rax]
cmp rcx, r11
jne .Lrun_done
inc r10
jmp .Lrun_loop
.Lrun_done:
cmp r10, 3
jl .Lliteral
lea rax, [rbp + 2]
cmp rax, r14
jg .Lfail
mov rax, 257
sub rax, r10
mov byte ptr [r13 + rbp], al
mov byte ptr [r13 + rbp + 1], r11b
add rbp, 2
add r15, r10
jmp .Louter
.Lliteral:
mov r8, r15
.Llit_scan:
cmp r8, r12
jge .Llit_done
mov rax, r8
sub rax, r15
cmp rax, 128
jge .Llit_done
lea rax, [r8 + 2]
cmp rax, r12
jge .Llit_adv
movzx eax, byte ptr [rbx + r8]
movzx ecx, byte ptr [rbx + r8 + 1]
cmp rax, rcx
jne .Llit_adv
movzx ecx, byte ptr [rbx + r8 + 2]
cmp rax, rcx
je .Llit_done
.Llit_adv:
inc r8
jmp .Llit_scan
.Llit_done:
mov r9, r8
sub r9, r15
lea rax, [rbp + 1]
add rax, r9
cmp rax, r14
jg .Lfail
mov rax, r9
dec rax
mov byte ptr [r13 + rbp], al
inc rbp
xor r10, r10
.Lcopy_loop:
cmp r10, r9
jge .Lcopy_done
lea rax, [r15 + r10]
movzx ecx, byte ptr [rbx + rax]
lea rax, [rbp + r10]
mov byte ptr [r13 + rax], cl
inc r10
jmp .Lcopy_loop
.Lcopy_done:
add rbp, r9
mov r15, r8
jmp .Louter
.Ldone:
mov rax, rbp
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
.Lfail:
mov rax, -1
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
