export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @outer_next %k_next
	%max =l phi @start 0, @outer_next %max_new
	%done =w csgtl %k, %n
	jnz %done, @finish, @inner
@inner
	%x =l phi @outer %k, @inner_join %x_next
	%steps =l phi @outer 0, @inner_join %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @inner_done, @inner_body
@inner_body
	%parity =l and %x, 1
	%is_odd =w cnel %parity, 0
	jnz %is_odd, @odd, @even
@even
	%x_even =l div %x, 2
	jmp @inner_join
@odd
	%tmp =l mul %x, 3
	%x_odd =l add %tmp, 1
	jmp @inner_join
@inner_join
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner
@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @update, @keep
@update
	jmp @outer_next
@keep
	jmp @outer_next
@outer_next
	%max_new =l phi @update %steps, @keep %max
	%k_next =l add %k, 1
	jmp @outer
@finish
	ret %max
}
