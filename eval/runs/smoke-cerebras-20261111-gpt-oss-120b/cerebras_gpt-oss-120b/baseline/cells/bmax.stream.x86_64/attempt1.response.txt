bmax:
    mov rax, qword ptr [rdi]            # load first element as max
    mov rcx, rsi
    sub rcx, 1
    jz .Ldone
    lea rdx, [rdi + 8]                  # pointer to second element
.Lloop:
    mov r8, qword ptr [rdx]
    cmp r8, rax
    jle .Lskip
    mov rax, r8
.Lskip:
    add rdx, 8
    dec rcx
    jnz .Lloop
.Ldone:
    ret
