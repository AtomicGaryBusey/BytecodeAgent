

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536
.equ ERR_PREFIX_LEN, 9
.equ OK_LEN, 3
.equ NL_LEN, 1

.section .rodata
err_prefix: .ascii "error at "
ok_msg: .byte 0x6F, 0x6B, 0x0A
nl_msg: .byte 0x0A

.section .bss
.align 8
input: .space 65536
.align 8
numbuf: .space 32
.align 8
input_len: .zero 8
pos: .zero 8
err_pos: .zero 8

.text
.globl _start
_start:
    xor r12, r12
    lea r13, [rip + input]
read_loop:
    mov rdi, STDIN
    lea rsi, [r13 + r12]
    mov rdx, MAX_INPUT
    sub rdx, r12
    mov rax, SYS_READ
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jae read_done
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], r12
    mov qword ptr [rip + pos], 0

    call parse_value
    test eax, eax
    jz print_error

    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je print_ok

    mov qword ptr [rip + err_pos], rax
    jmp print_error

fail_at_pos:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret

parse_ws:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
ws_loop:
    cmp rsi, rdx
    jae ws_done
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x20
    je ws_skip
    cmp eax, 0x09
    je ws_skip
    cmp eax, 0x0A
    je ws_skip
    cmp eax, 0x0D
    je ws_skip
    jmp ws_done
ws_skip:
    inc rsi
    jmp ws_loop
ws_done:
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

expect_char:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos
    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, r8d
    jne fail_at_pos
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

expect_hex_digit:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos
    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]

    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    jbe hex_ok

    mov ecx, eax
    sub ecx, 0x61
    cmp ecx, 5
    jbe hex_ok

    mov ecx, eax
    sub ecx, 0x41
    cmp ecx, 5
    jbe hex_ok

    jmp fail_at_pos
hex_ok:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

parse_value:
    call parse_ws

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos

    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]

    cmp eax, 0x7B
    je parse_object
    cmp eax, 0x5B
    je parse_array
    cmp eax, 0x22
    je parse_string
    cmp eax, 0x74
    je parse_true
    cmp eax, 0x66
    je parse_false
    cmp eax, 0x6E
    je parse_null
    cmp eax, 0x2D
    je parse_number

    sub eax, 0x30
    cmp eax, 9
    ja fail_at_pos
    jmp parse_number

parse_true:
    mov rsi, qword ptr [rip + pos]
    inc rsi
    mov qword ptr [rip + pos], rsi

    mov r8d, 0x72
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x75
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x65
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov eax, 1
    ret

parse_false:
    mov rsi, qword ptr [rip + pos]
    inc rsi
    mov qword ptr [rip + pos], rsi

    mov r8d, 0x61
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x6C
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x73
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x65
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov eax, 1
    ret

parse_null:
    mov rsi, qword ptr [rip + pos]
    inc rsi
    mov qword ptr [rip + pos], rsi

    mov r8d, 0x75
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x6C
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov r8d, 0x6C
    call expect_char
    test eax, eax
    jz fail_at_pos

    mov eax, 1
    ret

parse_string:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x22
    jne fail_at_pos

    inc rsi
    mov qword ptr [rip + pos], rsi

str_loop:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x22
    je str_close
    cmp eax, 0x5C
    je str_escape
    cmp eax, 0x20
    jb fail_at_pos

    inc rsi
    mov qword ptr [rip + pos], rsi
    jmp str_loop

str_close:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

str_escape:
    inc rsi
    mov qword ptr [rip + pos], rsi

    cmp rsi, rdx
    jae fail_at_pos
    movzx eax, byte ptr [rdi + rsi]

    cmp eax, 0x22
    je str_escape_ok
    cmp eax, 0x5C
    je str_escape_ok
    cmp eax, 0x2F
    je str_escape_ok
    cmp eax, 0x62
    je str_escape_ok
    cmp eax, 0x66
    je str_escape_ok
    cmp eax, 0x6E
    je str_escape_ok
    cmp eax, 0x72
    je str_escape_ok
    cmp eax, 0x74
    je str_escape_ok
    cmp eax, 0x75
    je str_hex4

    jmp fail_at_pos

str_escape_ok:
    inc rsi
    mov qword ptr [rip + pos], rsi
    jmp str_loop

str_hex4:
    inc rsi
    mov qword ptr [rip + pos], rsi

    call expect_hex_digit
    test eax, eax
    jz fail_at_pos

    call expect_hex_digit
    test eax, eax
    jz fail_at_pos

    call expect_hex_digit
    test eax, eax
    jz fail_at_pos

    call expect_hex_digit
    test eax, eax
    jz fail_at_pos

    jmp str_loop

parse_number:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x2D
    jne num_no_minus

    inc rsi
    mov qword ptr [rip + pos], rsi

num_no_minus:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x30
    je num_zero

    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja fail_at_pos

    inc rsi
    mov qword ptr [rip + pos], rsi

num_digits:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae num_after_int

    movzx eax, byte ptr [rdi + rsi]
    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja num_after_int

    inc rsi
    mov qword ptr [rip + pos], rsi
    jmp num_digits

num_zero:
    inc rsi
    mov qword ptr [rip + pos], rsi

num_after_int:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae num_exp

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x2E
    jne num_exp

    inc rsi
    mov qword ptr [rip + pos], rsi

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja fail_at_pos

    inc rsi
    mov qword ptr [rip + pos], rsi

num_frac_digits:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae num_exp

    movzx eax, byte ptr [rdi + rsi]
    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja num_exp

    inc rsi
    mov qword ptr [rip + pos], rsi
    jmp num_frac_digits

num_exp:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae num_success

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x65
    je num_do_exp
    cmp eax, 0x45
    jne num_success

num_do_exp:
    inc rsi
    mov qword ptr [rip + pos], rsi

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x2B
    je num_exp_sign
    cmp eax, 0x2D
    jne num_exp_digit

num_exp_sign:
    inc rsi
    mov qword ptr [rip + pos], rsi

num_exp_digit:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae fail_at_pos

    movzx eax, byte ptr [rdi + rsi]
    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja fail_at_pos

    inc rsi
    mov qword ptr [rip + pos], rsi

num_exp_digits:
    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    lea rdi, [rip + input]
    cmp rsi, rdx
    jae num_success

    movzx eax, byte ptr [rdi + rsi]
    mov ecx, eax
    sub ecx, 0x30
    cmp ecx, 9
    ja num_success

    inc rsi
    mov qword ptr [rip + pos], rsi
    jmp num_exp_digits

num_success:
    mov eax, 1
    ret

parse_array:
    mov rsi, qword ptr [rip + pos]
    inc rsi
    mov qword ptr [rip + pos], rsi

    call parse_ws

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos

    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x5D
    je arr_close_first

    call parse_value
    test eax, eax
    jz fail_at_pos

arr_loop:
    call parse_ws

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos

    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x5D
    je arr_close
    cmp eax, 0x2C
    je arr_comma

    jmp fail_at_pos

arr_comma:
    inc rsi
    mov qword ptr [rip + pos], rsi

    call parse_value
    test eax, eax
    jz fail_at_pos

    jmp arr_loop

arr_close:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

arr_close_first:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

parse_object:
    mov rsi, qword ptr [rip + pos]
    inc rsi
    mov qword ptr [rip + pos], rsi

    call parse_ws

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos

    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x7D
    je obj_close_first

    call parse_member
    test eax, eax
    jz fail_at_pos

obj_loop:
    call parse_ws

    mov rsi, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rsi, rdx
    jae fail_at_pos

    lea rdi, [rip + input]
    movzx eax, byte ptr [rdi + rsi]
    cmp eax, 0x7D
    je obj_close
    cmp eax, 0x2C
    je obj_comma

    jmp fail_at_pos

obj_comma:
    inc rsi
    mov qword ptr [rip + pos], rsi

    call parse_member
    test eax, eax
    jz fail_at_pos

    jmp obj_loop

obj_close:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

obj_close_first:
    inc rsi
    mov qword ptr [rip + pos], rsi
    mov eax, 1
    ret

parse_member:
    call parse_ws

    call parse_string
    test eax, eax
    jz fail_at_pos

    call parse_ws

    mov r8d, 0x3A
    call expect_char
    test eax, eax
    jz fail_at_pos

    call parse_value
    test eax, eax
    jz fail_at_pos

    mov eax, 1
    ret

write_all:
    test rdx, rdx
    jz write_all_ret
write_all_loop:
    mov rax, SYS_WRITE
    syscall
    cmp rax, 0
    jle write_all_ret
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_all_loop
write_all_ret:
    ret

print_num:
    test rax, rax
    jne print_num_nonzero

    lea rsi, [rip + numbuf + 31]
    mov byte ptr [rsi], 0x30
    mov rdi, STDOUT
    mov rdx, 1
    call write_all
    ret

print_num_nonzero:
    lea rcx, [rip + numbuf + 32]
    mov r8, 10
print_num_loop:
    xor edx, edx
    div r8
    add edx, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz print_num_loop

    mov rsi, rcx
    lea r10, [rip + numbuf + 32]
    sub r10, rcx
    mov rdx, r10
    mov rdi, STDOUT
    call write_all
    ret

print_error:
    mov rdi, STDOUT
    lea rsi, [rip + err_prefix]
    mov rdx, ERR_PREFIX_LEN
    call write_all

    mov rax, qword ptr [rip + err_pos]
    call print_num

    mov rdi, STDOUT
    lea rsi, [rip + nl_msg]
    mov rdx, NL_LEN
    call write_all

    mov rax, SYS_EXIT
    mov edi, 1
    syscall

print_ok:
    mov rdi, STDOUT
    lea rsi, [rip + ok_msg]
    mov rdx, OK_LEN
    call write_all

    mov rax, SYS_EXIT
    xor edi, edi
    syscall
