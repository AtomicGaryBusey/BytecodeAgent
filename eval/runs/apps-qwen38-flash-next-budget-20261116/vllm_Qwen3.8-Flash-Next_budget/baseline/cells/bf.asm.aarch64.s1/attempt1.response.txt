

.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 65536
.equ OUT_SIZE, 65536
.equ STACK_LIMIT, 65536
.equ BANG, 33
.equ LT, 60
.equ GT, 62
.equ PLUS, 43
.equ MINUS, 45
.equ DOT, 46
.equ COMMA, 44
.equ OPEN, 91
.equ CLOSE, 93

.section .bss
.p2align 3
input_buf:
.zero 65536
.p2align 3
cmd_buf:
.zero 65536
.p2align 3
match_arr:
.zero 262144
.p2align 3
br_stack:
.zero 262144
.p2align 3
tape:
.zero 30000
.p2align 3
out_buf:
.zero 65536

.section .text
.p2align 2
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, cmd_buf
    add x20, x20, :lo12:cmd_buf
    adrp x21, out_buf
    add x21, x21, :lo12:out_buf
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x23, match_arr
    add x23, x23, :lo12:match_arr
    adrp x24, br_stack
    add x24, x24, :lo12:br_stack

    mov x25, #0
read_loop:
    cmp x25, #MAX_INPUT
    b.ge read_done
    add x1, x19, x25
    mov x2, #MAX_INPUT
    sub x2, x2, x25
    mov x0, #0
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    ble read_done
    add x25, x25, x0
    cmp x25, #MAX_INPUT
    blt read_loop
    mov x25, #MAX_INPUT
read_done:

    mov x9, #0
scan_loop:
    cmp x9, x25
    b.ge scan_notfound
    ldrb w10, [x19, x9]
    cmp w10, #BANG
    beq scan_found
    add x9, x9, #1
    b scan_loop
scan_found:
    mov x11, x9
    add x15, x9, #1
    b prep_start
scan_notfound:
    mov x11, x25
    mov x15, x25

prep_start:
    mov x14, x25
    mov x26, #0
    mov x17, #0
    mov x13, #0
pre_loop:
    cmp x17, x11
    b.ge pre_done
    ldrb w10, [x19, x17]

    cmp w10, #OPEN
    b.eq prep_open
    cmp w10, #CLOSE
    b.eq prep_close

    cmp w10, #LT
    b.eq prep_store
    cmp w10, #GT
    b.eq prep_store
    cmp w10, #PLUS
    b.eq prep_store
    cmp w10, #MINUS
    b.eq prep_store
    cmp w10, #DOT
    b.eq prep_store
    cmp w10, #COMMA
    b.eq prep_store

    add x17, x17, #1
    b pre_loop

prep_store:
    strb w10, [x20, x26]
    add x26, x26, #1
    add x17, x17, #1
    b pre_loop

prep_open:
    cmp x13, #STACK_LIMIT
    b.ge error_exit
    str w26, [x24, x13, lsl #2]
    add x13, x13, #1
    strb w10, [x20, x26]
    add x26, x26, #1
    add x17, x17, #1
    b pre_loop

prep_close:
    cmp x13, #0
    b.eq error_exit
    sub x13, x13, #1
    ldr w12, [x24, x13, lsl #2]
    add w7, w26, #1
    str w7, [x23, x12, lsl #2]
    add w7, w12, #1
    str w7, [x23, x26, lsl #2]
    strb w10, [x20, x26]
    add x26, x26, #1
    add x17, x17, #1
    b pre_loop

pre_done:
    cmp x13, #0
    b.ne error_exit

    mov x16, #0
    mov x17, #0
    mov x28, #0

exec_loop:
    cmp x17, x26
    b.ge exec_done
    ldrb w0, [x20, x17]

    cmp w0, #GT
    b.eq exec_gt
    cmp w0, #LT
    b.eq exec_lt
    cmp w0, #PLUS
    b.eq exec_plus
    cmp w0, #MINUS
    b.eq exec_minus
    cmp w0, #DOT
    b.eq exec_dot
    cmp w0, #COMMA
    b.eq exec_comma
    cmp w0, #OPEN
    b.eq exec_open
    cmp w0, #CLOSE
    b.eq exec_close

    add x17, x17, #1
    b exec_loop

exec_gt:
    add x16, x16, #1
    add x17, x17, #1
    b exec_loop

exec_lt:
    sub x16, x16, #1
    add x17, x17, #1
    b exec_loop

exec_plus:
    ldrb w1, [x22, x16]
    add w1, w1, #1
    strb w1, [x22, x16]
    add x17, x17, #1
    b exec_loop

exec_minus:
    ldrb w1, [x22, x16]
    sub w1, w1, #1
    strb w1, [x22, x16]
    add x17, x17, #1
    b exec_loop

exec_dot:
    ldrb w1, [x22, x16]
    strb w1, [x21, x28]
    add x28, x28, #1
    cmp x28, #OUT_SIZE
    b.lt dot_noflush
    mov x0, #1
    mov x1, x21
    mov x2, #OUT_SIZE
    mov x8, #SYS_WRITE
    svc #0
    mov x28, #0
dot_noflush:
    add x17, x17, #1
    b exec_loop

exec_comma:
    cmp x15, x14
    b.ge comma_zero
    ldrb w1, [x19, x15]
    add x15, x15, #1
    strb w1, [x22, x16]
    add x17, x17, #1
    b exec_loop
comma_zero:
    strb wzr, [x22, x16]
    add x17, x17, #1
    b exec_loop

exec_open:
    ldrb w1, [x22, x16]
    cbnz w1, exec_open_nonzero
    ldr w17, [x23, x17, lsl #2]
    b exec_loop
exec_open_nonzero:
    add x17, x17, #1
    b exec_loop

exec_close:
    ldrb w1, [x22, x16]
    cbz w1, exec_close_zero
    ldr w17, [x23, x17, lsl #2]
    b exec_loop
exec_close_zero:
    add x17, x17, #1
    b exec_loop

exec_done:
    cmp x28, #0
    b.eq exit_zero
    mov x0, #1
    mov x1, x21
    mov x2, x28
    mov x8, #SYS_WRITE
    svc #0
exit_zero:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

error_exit:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0
