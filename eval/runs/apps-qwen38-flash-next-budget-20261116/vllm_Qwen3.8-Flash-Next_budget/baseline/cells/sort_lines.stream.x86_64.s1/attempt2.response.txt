

.text
.globl _start
.align 16
_start:
    xor r12, r12
read_loop:
    cmp r12, 32768
    jb read_main
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + scratch]
    mov rdx, 1
    syscall
    test rax, rax
    jle read_done
    jmp read_loop
read_main:
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rbp, rbp
    lea r14, [rip + input_buf]
    mov r15, r14
    mov rbx, r14
    add rbx, r12
parse_loop:
    cmp r14, rbx
    jae parse_final
    cmp byte ptr [r14], 10
    jne parse_next
    mov rcx, r14
    sub rcx, r15
    cmp rbp, 1000
    jae parse_after_store
    mov rsi, rbp
    shl rsi, 4
    lea r10, [rip + line_info]
    mov qword ptr [r10 + rsi], r15
    mov qword ptr [r10 + rsi + 8], rcx
    inc rbp
parse_after_store:
    lea r15, [r14 + 1]
parse_next:
    inc r14
    jmp parse_loop
parse_final:
    cmp r15, rbx
    jae parse_skip_final
    cmp rbp, 1000
    jae parse_skip_final
    mov rcx, rbx
    sub rcx, r15
    mov rsi, rbp
    shl rsi, 4
    lea r10, [rip + line_info]
    mov qword ptr [r10 + rsi], r15
    mov qword ptr [r10 + rsi + 8], rcx
    inc rbp
parse_skip_final:
    mov r12, 1
    cmp r12, rbp
    jae sort_done
sort_outer:
    mov rbx, r12
    shl rbx, 4
    lea r10, [rip + line_info]
    mov r14, qword ptr [r10 + rbx]
    mov r15, qword ptr [r10 + rbx + 8]
    mov r13, r12
    dec r13
sort_inner:
    test r13, r13
    jl sort_place
    mov rbx, r13
    shl rbx, 4
    lea r10, [rip + line_info]
    mov rdi, qword ptr [r10 + rbx]
    mov rsi, qword ptr [r10 + rbx + 8]
    mov rdx, r14
    mov rcx, r15
    call cmp_line
    test rax, rax
    jle sort_place
    mov rbx, r13
    shl rbx, 4
    lea r10, [rip + line_info]
    mov rax, qword ptr [r10 + rbx]
    mov qword ptr [r10 + rbx + 16], rax
    mov rax, qword ptr [r10 + rbx + 8]
    mov qword ptr [r10 + rbx + 24], rax
    dec r13
    jmp sort_inner
sort_place:
    lea r10, [rip + line_info]
    mov rbx, r13
    inc rbx
    shl rbx, 4
    mov qword ptr [r10 + rbx], r14
    mov qword ptr [r10 + rbx + 8], r15
    inc r12
    cmp r12, rbp
    jb sort_outer
sort_done:
    lea r14, [rip + out_buf]
    mov r15, r14
    xor r13, r13
out_loop:
    cmp r13, rbp
    jae write_loop
    mov rbx, r13
    shl rbx, 4
    lea r10, [rip + line_info]
    mov rsi, qword ptr [r10 + rbx]
    mov rcx, qword ptr [r10 + rbx + 8]
    mov rdi, r15
    test rcx, rcx
    je out_nl
    rep movsb
out_nl:
    mov byte ptr [rdi], 10
    inc rdi
    mov r15, rdi
    inc r13
    jmp out_loop
write_loop:
    mov rax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    sub rdx, r14
    test rdx, rdx
    je exit_ok
    syscall
    test rax, rax
    jle exit_ok
    add r14, rax
    cmp r14, r15
    jb write_loop
exit_ok:
    mov rax, 60
    xor edi, edi
    syscall

cmp_line:
    mov r8, rsi
    mov r9, rcx
    mov r10, r8
    cmp r10, r9
    cmovae r10, r9
    mov r11, r10
    cmp r11, 8
    jb cmp_tail
cmp_chunk:
    mov rax, qword ptr [rdi]
    mov rbx, qword ptr [rdx]
    cmp rax, rbx
    jne cmp_diff
    add rdi, 8
    add rdx, 8
    sub r11, 8
    cmp r11, 8
    jae cmp_chunk
    jmp cmp_tail
cmp_diff:
    xor rax, rbx
    tzcnt rax, rax
    shr rax, 3
    movzx ecx, byte ptr [rdi + rax]
    movzx ebx, byte ptr [rdx + rax]
    cmp ecx, ebx
    jb cmp_less
    ja cmp_greater
    jmp cmp_greater
cmp_tail:
    test r11, r11
    je cmp_by_len
cmp_byte:
    movzx ecx, byte ptr [rdi]
    movzx ebx, byte ptr [rdx]
    cmp ecx, ebx
    jb cmp_less
    ja cmp_greater
    inc rdi
    inc rdx
    dec r11
    jnz cmp_byte
cmp_by_len:
    cmp r8, r9
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov rax, -1
    ret
cmp_greater:
    mov rax, 1
    ret

.bss
.align 16
input_buf: .zero 32768
line_info: .zero 16000
out_buf: .zero 33824
scratch: .zero 8
