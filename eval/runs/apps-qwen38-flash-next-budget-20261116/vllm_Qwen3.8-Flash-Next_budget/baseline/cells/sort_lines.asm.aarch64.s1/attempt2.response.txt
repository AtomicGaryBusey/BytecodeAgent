

.text
.globl _start
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ READ, 63
.equ WRITE, 64
.equ EXIT, 93

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #MAX_INPUT
    bhs read_full
    mov x0, #0
    add x1, x19, x20
    mov x2, #MAX_INPUT
    subs x2, x2, x20
    mov x8, #READ
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    blo read_loop
    mov x20, #MAX_INPUT
    b read_loop

read_full:
    cmp x20, #MAX_INPUT
    blo read_loop
    mov x20, #MAX_INPUT
    mov x0, #0
    add x1, x19, x20
    mov x2, #0
    mov x8, #READ
    svc #0
    b read_done

read_done:
    adrp x22, records
    add x22, x22, :lo12:records
    mov x23, #0
    mov x24, #0
    mov x25, #0

parse_loop:
    cmp x25, x20
    bhs parse_final
    ldrb w5, [x19, x25]
    cmp w5, #10
    bne parse_inc

    cmp x23, #MAX_LINES
    bhs parse_newline_end
    add x6, x19, x24
    subs x7, x25, x24
    stp x6, x7, [x22]
    add x22, x22, #16
    add x23, x23, #1

parse_newline_end:
    add x24, x25, #1

parse_inc:
    add x25, x25, #1
    b parse_loop

parse_final:
    cmp x24, x20
    bhs parse_done
    cmp x23, #MAX_LINES
    bhs parse_done
    add x6, x19, x24
    subs x7, x20, x24
    stp x6, x7, [x22]
    add x22, x22, #16
    add x23, x23, #1

parse_done:
    mov x10, x23
    cmp x10, #1
    bls output

    adrp x12, records
    add x12, x12, :lo12:records
    adrp x13, records_tmp
    add x13, x13, :lo12:records_tmp
    mov x11, #1

sort_outer:
    cmp x11, x10
    bhs sort_done
    mov x14, #0

merge_outer:
    cmp x14, x10
    bhs merge_done

    add x20, x14, x11
    cmp x20, x10
    csel x20, x20, x10, hi

    add x21, x20, x11
    cmp x21, x10
    csel x21, x21, x10, hi
    mov x22, x21

    add x15, x12, x14, lsl #4
    add x16, x12, x20, lsl #4
    mov x17, x16
    add x18, x12, x21, lsl #4
    add x19, x13, x14, lsl #4

merge_loop:
    cmp x15, x16
    bhs merge_left_exhausted
    cmp x17, x18
    bhs merge_right_exhausted

    ldp x0, x2, [x15]
    ldp x1, x3, [x17]
    bl leq
    cbz x0, copy_one_right

copy_one_left:
    ldp x20, x21, [x15]
    stp x20, x21, [x19]
    add x15, x15, #16
    add x19, x19, #16
    b merge_loop

copy_one_right:
    ldp x20, x21, [x17]
    stp x20, x21, [x19]
    add x17, x17, #16
    add x19, x19, #16
    b merge_loop

merge_right_exhausted:
    cmp x15, x16
    bhs merge_advance
    ldp x20, x21, [x15]
    stp x20, x21, [x19]
    add x15, x15, #16
    add x19, x19, #16
    b merge_right_exhausted

merge_left_exhausted:
    cmp x17, x18
    bhs merge_advance
    ldp x20, x21, [x17]
    stp x20, x21, [x19]
    add x17, x17, #16
    add x19, x19, #16
    b merge_left_exhausted

merge_advance:
    mov x14, x22
    b merge_outer

merge_done:
    mov x20, #0

copy_back_loop:
    cmp x20, x10
    bhs copy_back_done
    add x21, x13, x20, lsl #4
    add x22, x12, x20, lsl #4
    ldp x23, x24, [x21]
    stp x23, x24, [x22]
    add x20, x20, #1
    b copy_back_loop

copy_back_done:
    add x11, x11, x11
    b sort_outer

sort_done:

output:
    adrp x12, records
    add x12, x12, :lo12:records
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, #0
    mov x23, #0

output_loop:
    cmp x23, x10
    bhs output_write

    add x24, x12, x23, lsl #4
    ldr x1, [x24]
    ldr x2, [x24, #8]

    add x5, x21, x22
    mov x4, #0

copy_output_line:
    cmp x4, x2
    bhs copy_output_done
    ldrb w7, [x1, x4]
    strb w7, [x5, x4]
    add x4, x4, #1
    b copy_output_line

copy_output_done:
    add x22, x22, x2
    add x5, x21, x22
    mov w6, #10
    strb w6, [x5]
    add x22, x22, #1
    add x23, x23, #1
    b output_loop

output_write:
    cmp x22, #0
    beq exit_ok

    mov x0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x2, x22

write_all:
    cbz x2, exit_ok
    mov x8, #WRITE
    svc #0
    cmp x0, #0
    ble exit_ok
    add x1, x1, x0
    subs x2, x2, x0
    b write_all

leq:
    mov x4, x2
    cmp x4, x3
    csel x4, x4, x3, lo
    mov x8, #0
    cbz x4, leq_len

leq_loop:
    ldrb w5, [x0, x8]
    ldrb w6, [x1, x8]
    cmp w5, w6
    bne leq_found
    add x8, x8, #1
    subs x4, x4, #1
    bne leq_loop

leq_len:
    cmp x2, x3
    cset x0, ls
    ret

leq_found:
    cset x0, lo
    ret

exit_ok:
    mov x0, #0
    mov x8, #EXIT
    svc #0
    b exit_ok

.bss
.balign 16
input_buf: .zero 32769
.balign 16
records: .zero 16000
.balign 16
records_tmp: .zero 16000
.balign 16
output_buf: .zero 33792
