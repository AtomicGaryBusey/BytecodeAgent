

.text
.p2align 2
.globl _start
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ READ_SYS, 63
.equ WRITE_SYS, 64
.equ EXIT_SYS, 93

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #MAX_INPUT
    bhs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #MAX_INPUT
    subs x2, x2, x20
    mov x8, #READ_SYS
    svc #0
    cmp x0, #0
    blt read_done
    beq read_done
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    blo read_loop
    mov x20, #MAX_INPUT
    b read_done

read_done:
    adrp x21, records
    add x21, x21, :lo12:records
    mov x22, #0
    mov x23, #0
    mov x24, #0

parse_loop:
    cmp x24, x20
    bhs parse_final
    ldrb w5, [x19, x24]
    cmp w5, #10
    bne parse_inc
    cmp x22, #MAX_LINES
    bhs parse_newline_skip
    add x5, x19, x23
    subs x6, x24, x23
    stp x5, x6, [x21]
    add x21, x21, #16
    add x22, x22, #1

parse_newline_skip:
    add x23, x24, #1

parse_inc:
    add x24, x24, #1
    b parse_loop

parse_final:
    cmp x23, x20
    bhs parse_done
    cmp x22, #MAX_LINES
    bhs parse_done
    add x5, x19, x23
    subs x6, x20, x23
    stp x5, x6, [x21]
    add x21, x21, #16
    add x22, x22, #1

parse_done:
    adrp x19, records
    add x19, x19, :lo12:records
    mov x20, x22
    cmp x20, #1
    bls output

    mov x21, #1

sort_i_loop:
    cmp x21, x20
    bhs sort_done
    add x27, x19, x21, lsl #4
    ldp x22, x23, [x27]
    mov x24, x21
    subs x24, x24, #1

sort_j_loop:
    cmp x24, #0
    bmi sort_place
    add x27, x19, x24, lsl #4
    ldp x25, x26, [x27]

    mov x4, x23
    cmp x4, x26
    csel x4, x4, x26, lo
    mov x0, #0
    cbz x4, compare_len

cmp_loop:
    ldrb w1, [x22, x0]
    ldrb w2, [x25, x0]
    cmp w1, w2
    bne cmp_diff
    add x0, x0, #1
    subs x4, x4, #1
    bne cmp_loop

compare_len:
    cmp x23, x26
    cset x3, lo
    b compare_end

cmp_diff:
    cmp w1, w2
    cset x3, lo

compare_end:
    cbz x3, sort_place

    add x27, x24, #1
    add x27, x19, x27, lsl #4
    stp x25, x26, [x27]
    sub x24, x24, #1
    b sort_j_loop

sort_place:
    add x24, x24, #1
    add x27, x19, x24, lsl #4
    stp x22, x23, [x27]
    add x21, x21, #1
    b sort_i_loop

sort_done:

output:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, #0
    mov x23, #0

output_loop:
    cmp x23, x20
    bhs output_write
    add x24, x19, x23, lsl #4
    ldp x25, x26, [x24]
    add x27, x21, x22
    mov x28, #0

copy_line_loop:
    cmp x28, x26
    bhs copy_line_done
    ldrb w5, [x25, x28]
    strb w5, [x27, x28]
    add x28, x28, #1
    b copy_line_loop

copy_line_done:
    add x22, x22, x26
    add x27, x21, x22
    mov w5, #10
    strb w5, [x27]
    add x22, x22, #1
    add x23, x23, #1
    b output_loop

output_write:
    cmp x22, #0
    beq exit_ok
    mov x0, #1
    mov x1, x21
    mov x2, x22

write_all:
    cbz x2, exit_ok
    mov x8, #WRITE_SYS
    svc #0
    cmp x0, #0
    ble exit_ok
    add x1, x1, x0
    subs x2, x2, x0
    b write_all

exit_ok:
    mov x0, #0
    mov x8, #EXIT_SYS
    svc #0
    b exit_ok

.bss
.balign 16
input_buf: .zero 32769
.balign 16
records: .zero 16000
.balign 16
output_buf: .zero 40000
