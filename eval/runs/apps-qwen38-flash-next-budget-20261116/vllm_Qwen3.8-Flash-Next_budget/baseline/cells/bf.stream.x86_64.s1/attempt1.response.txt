

.globl _start

.equ IN_CAP, 65536
.equ OUT_CAP, 65536

.text
_start:
    xor rbp, rbp
    lea r9, [rip + inbuf]
    mov rdi, 0
read_again:
    cmp rbp, IN_CAP
    jae read_done
    lea rsi, [r9 + rbp]
    mov rdx, IN_CAP
    sub rdx, rbp
    xor eax, eax
    syscall
    test rax, rax
    jg read_add
    cmp rax, 0
    je read_done
    cmp rax, -4
    je read_again
    jmp read_done
read_add:
    add rbp, rax
    jmp read_again
read_done:
    mov r12, rbp
    xor rbp, rbp

    xor r13, r13
find_loop:
    cmp r13, r12
    jae no_bang
    cmp byte ptr [r9 + r13], 33
    je bang_found
    inc r13
    jmp find_loop
bang_found:
    lea r10, [r9 + r13 + 1]
    mov r15, r9
    add r15, r12
    jmp scan_brackets
no_bang:
    mov r13, r12
    lea r10, [r9 + r12]
    mov r15, r10
    jmp scan_brackets

scan_brackets:
    lea r12, [rip + match_table]
    xor rbp, rbp
    xor r11, r11
balance_loop:
    cmp r11, r13
    jae balance_done
    movzx eax, byte ptr [r9 + r11]
    cmp eax, 91
    je balance_left
    cmp eax, 93
    je balance_right
    inc r11
    jmp balance_loop
balance_left:
    push r11
    inc rbp
    inc r11
    jmp balance_loop
balance_right:
    test rbp, rbp
    je exit_one
    dec rbp
    pop rbx
    mov edx, r11d
    inc edx
    mov dword ptr [r12 + rbx*4], edx
    lea edx, [rbx + 1]
    mov dword ptr [r12 + r11*4], edx
    inc r11
    jmp balance_loop
balance_done:
    test rbp, rbp
    jne exit_one
    xor rbp, rbp
    xor rbx, rbx
    lea r8, [rip + tape]
    lea r14, [rip + outbuf]

exec_loop:
    cmp rbx, r13
    jae exec_end
    movzx eax, byte ptr [r9 + rbx]
    cmp eax, 46
    je cmd_dot
    ja exec_after_dot
    cmp eax, 44
    je cmd_comma
    ja cmd_minus
    cmp eax, 43
    je cmd_plus
    jmp cmd_other

exec_after_dot:
    cmp eax, 91
    je cmd_lbrack
    jb exec_mid
    cmp eax, 93
    je cmd_rbrack
    jmp cmd_other

exec_mid:
    cmp eax, 62
    je cmd_gt
    jb exec_lt
    jmp cmd_other
exec_lt:
    cmp eax, 60
    je cmd_lt
    jmp cmd_other

cmd_plus:
    inc byte ptr [r8]
    inc rbx
    jmp exec_loop
cmd_minus:
    dec byte ptr [r8]
    inc rbx
    jmp exec_loop
cmd_gt:
    inc r8
    inc rbx
    jmp exec_loop
cmd_lt:
    dec r8
    inc rbx
    jmp exec_loop
cmd_comma:
    cmp r10, r15
    jae comma_zero
    movzx eax, byte ptr [r10]
    inc r10
    mov byte ptr [r8], al
    inc rbx
    jmp exec_loop
comma_zero:
    mov byte ptr [r8], 0
    inc rbx
    jmp exec_loop
cmd_dot:
    movzx eax, byte ptr [r8]
    mov byte ptr [r14 + rbp], al
    inc rbp
    cmp rbp, OUT_CAP
    jae dot_flush
    inc rbx
    jmp exec_loop
dot_flush:
    mov rsi, r14
dot_flush_again:
    test rbp, rbp
    je dot_done
    mov rdi, 1
    mov rdx, rbp
    mov rax, 1
    syscall
    test rax, rax
    jle dot_flush_error
    sub rbp, rax
    add rsi, rax
    jmp dot_flush_again
dot_flush_error:
    xor rbp, rbp
dot_done:
    inc rbx
    jmp exec_loop

cmd_lbrack:
    cmp byte ptr [r8], 0
    jne lbrack_inc
    mov edx, dword ptr [r12 + rbx*4]
    mov rbx, rdx
    jmp exec_loop
lbrack_inc:
    inc rbx
    jmp exec_loop
cmd_rbrack:
    cmp byte ptr [r8], 0
    je rbrack_inc
    mov edx, dword ptr [r12 + rbx*4]
    mov rbx, rdx
    jmp exec_loop
rbrack_inc:
    inc rbx
    jmp exec_loop
cmd_other:
    inc rbx
    jmp exec_loop

exec_end:
    test rbp, rbp
    je exit_zero
    mov rsi, r14
end_flush_again:
    test rbp, rbp
    je exit_zero
    mov rdi, 1
    mov rdx, rbp
    mov rax, 1
    syscall
    test rax, rax
    jle exit_zero
    sub rbp, rax
    add rsi, rax
    jmp end_flush_again

exit_zero:
    mov rdi, 0
    mov rax, 60
    syscall
    jmp exit_zero
exit_one:
    mov rdi, 1
    mov rax, 60
    syscall
    jmp exit_one

.bss
.align 16
inbuf:
.space 65536
.align 16
match_table:
.space 262144
.align 16
outbuf:
.space 65536
.align 16
tape:
.space 30000
