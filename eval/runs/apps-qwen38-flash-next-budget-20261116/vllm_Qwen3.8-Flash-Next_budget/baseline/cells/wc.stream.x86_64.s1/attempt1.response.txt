

.text
.globl _start
_start:
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d
    xor r12d, r12d
    lea rbp, [rip + inbuf]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbp
    mov edx, 65536
    mov r15, rdx
    syscall
    test rax, rax
    jz process_end
    js read_loop
    cmp rax, r15
    jbe read_clamped
    mov rax, r15
read_clamped:
    mov rdx, rax
    add rbx, rax
    mov rsi, rbp

chunk_loop:
    test rdx, rdx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rdx
    cmp eax, 10
    je is_line
    cmp eax, 32
    je is_ws
    cmp eax, 9
    je is_ws
    cmp eax, 11
    je is_ws
    cmp eax, 12
    je is_ws
    cmp eax, 13
    je is_ws
    test r12, r12
    jnz chunk_loop
    mov r12d, 1
    inc r14
    jmp chunk_loop

is_ws:
    xor r12d, r12d
    jmp chunk_loop

is_line:
    inc r13
    xor r12d, r12d
    jmp chunk_loop

process_end:
    lea rbp, [rip + outbuf]
    sub rsp, 64

    lea r8, [rsp + 63]
    mov r10, r8
    mov rax, r13
    mov rcx, 10
conv1_digits:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz conv1_digits
    mov rsi, r8
    mov r9, r10
conv1_copy:
    cmp rsi, r9
    jae conv1_done
    mov al, byte ptr [rsi]
    mov byte ptr [rbp], al
    inc rsi
    inc rbp
    jmp conv1_copy
conv1_done:
    mov byte ptr [rbp], 32
    inc rbp

    lea r8, [rsp + 63]
    mov r10, r8
    mov rax, r14
    mov rcx, 10
conv2_digits:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz conv2_digits
    mov rsi, r8
    mov r9, r10
conv2_copy:
    cmp rsi, r9
    jae conv2_done
    mov al, byte ptr [rsi]
    mov byte ptr [rbp], al
    inc rsi
    inc rbp
    jmp conv2_copy
conv2_done:
    mov byte ptr [rbp], 32
    inc rbp

    lea r8, [rsp + 63]
    mov r10, r8
    mov rax, rbx
    mov rcx, 10
conv3_digits:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz conv3_digits
    mov rsi, r8
    mov r9, r10
conv3_copy:
    cmp rsi, r9
    jae conv3_done
    mov al, byte ptr [rsi]
    mov byte ptr [rbp], al
    inc rsi
    inc rbp
    jmp conv3_copy
conv3_done:
    mov byte ptr [rbp], 10
    inc rbp

    add rsp, 64
    lea rsi, [rip + outbuf]
    mov rdx, rbp
    sub rdx, rsi

write_loop:
    mov rdi, 1
    mov rax, 1
    syscall
    test rax, rax
    js write_loop
    jz exit_status
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_loop

exit_status:
    xor edi, edi
    mov rax, 60
    syscall
    mov rax, 231
    syscall

.bss
.align 16
inbuf: .zero 65536
outbuf: .zero 256
