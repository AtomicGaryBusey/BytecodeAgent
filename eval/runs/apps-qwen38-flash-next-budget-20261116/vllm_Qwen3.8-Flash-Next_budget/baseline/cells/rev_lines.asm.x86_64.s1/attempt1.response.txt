

.equ MAXINPUT, 65536
.equ READLIMIT, 65537
.equ BUFSIZE, 65537

.text
.globl _start
_start:
    xor ebx, ebx
    lea r12, [rip + buf]

read_again:
    cmp rbx, READLIMIT
    jae process_input

    mov eax, 0
    xor edi, edi
    mov rsi, r12
    add rsi, rbx
    mov rdx, READLIMIT
    sub rdx, rbx
    syscall

    test rax, rax
    jle process_input

    add rbx, rax
    jmp read_again

process_input:
    mov r13, rbx
    cmp r13, MAXINPUT
    jbe process_len
    mov r13, MAXINPUT
process_len:
    xor r14, r14

scan_lines:
    mov r15, r14

scan_byte:
    cmp r15, r13
    jae final_line

    cmp byte ptr [r12 + r15], 10
    je found_newline

    inc r15
    jmp scan_byte

found_newline:
    mov rbx, r14
    mov rbp, r15

reverse_line:
    lea rax, [rbx + 1]
    cmp rax, rbp
    jae reverse_done

    movzx ecx, byte ptr [r12 + rbx]
    movzx edx, byte ptr [r12 + rbp - 1]
    mov byte ptr [r12 + rbp - 1], cl
    mov byte ptr [r12 + rbx], dl

    inc rbx
    dec rbp
    jmp reverse_line

reverse_done:
    lea r14, [r15 + 1]
    jmp scan_lines

final_line:
    cmp r14, r13
    jae write_output

    mov rbx, r14
    mov rbp, r13

reverse_final:
    lea rax, [rbx + 1]
    cmp rax, rbp
    jae append_newline

    movzx ecx, byte ptr [r12 + rbx]
    movzx edx, byte ptr [r12 + rbp - 1]
    mov byte ptr [r12 + rbp - 1], cl
    mov byte ptr [r12 + rbx], dl

    inc rbx
    dec rbp
    jmp reverse_final

append_newline:
    mov byte ptr [r12 + r13], 10
    inc r13

write_output:
    xor rbp, rbp

write_loop:
    mov rbx, r13
    sub rbx, rbp
    test rbx, rbx
    je exit_success

    mov eax, 1
    mov edi, 1
    mov rsi, r12
    add rsi, rbp
    mov rdx, rbx
    syscall

    test rax, rax
    jle exit_success

    add rbp, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
buf:
    .zero BUFSIZE
