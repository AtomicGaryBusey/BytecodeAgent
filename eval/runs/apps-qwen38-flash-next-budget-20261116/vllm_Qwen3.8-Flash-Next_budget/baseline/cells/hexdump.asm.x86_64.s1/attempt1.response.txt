

.text
.globl _start
_start:
    lea r9, [rip + hexdig]
    xor r12, r12
    lea r13, [rip + inbuf]
    mov r14, 65536
read_loop:
    mov rdi, 0
    mov rsi, r13
    mov rdx, r14
    mov eax, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    add r13, rax
    sub r14, rax
    cmp r14, 0
    jle read_done
    jmp read_loop
read_done:
    lea r13, [rip + inbuf]
    lea rbp, [rip + linebuf]
    xor rbx, rbx
row_loop:
    cmp rbx, r12
    jae print_final
    mov r14, rbp
    mov rdi, rbx
    mov rsi, r14
    call hex8
    mov r14, rsi
    mov word ptr [r14], 0x2020
    add r14, 2
    mov r15, r12
    sub r15, rbx
    cmp r15, 16
    jbe row_n_done
    mov r15, 16
row_n_done:
    xor r8, r8
    mov r11, 8
first_slot_loop:
    cmp r8, r15
    jae first_blank
    movzx eax, byte ptr [r13 + r8]
    mov edx, eax
    shr edx, 4
    and edx, 15
    movzx rdx, byte ptr [r9 + rdx]
    mov byte ptr [r14], dl
    mov ecx, eax
    and ecx, 15
    movzx rcx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + 1], cl
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
    jmp first_next
first_blank:
    mov word ptr [r14], 0x2020
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
first_next:
    inc r8
    dec r11
    jnz first_slot_loop
    mov byte ptr [r14], 0x20
    inc r14
    mov r8, 8
    mov r11, 8
second_slot_loop:
    cmp r8, r15
    jae second_blank
    movzx eax, byte ptr [r13 + r8]
    mov edx, eax
    shr edx, 4
    and edx, 15
    movzx rdx, byte ptr [r9 + rdx]
    mov byte ptr [r14], dl
    mov ecx, eax
    and ecx, 15
    movzx rcx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + 1], cl
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
    jmp second_next
second_blank:
    mov word ptr [r14], 0x2020
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
second_next:
    inc r8
    dec r11
    jnz second_slot_loop
    mov byte ptr [r14], 0x20
    inc r14
    mov byte ptr [r14], 0x7C
    inc r14
    xor r8, r8
char_loop:
    cmp r8, r15
    jae char_done
    movzx eax, byte ptr [r13 + r8]
    cmp eax, 0x20
    jl char_dot
    cmp eax, 0x7E
    jg char_dot
    mov byte ptr [r14], al
    inc r14
    jmp char_next
char_dot:
    mov byte ptr [r14], 0x2E
    inc r14
char_next:
    inc r8
    jmp char_loop
char_done:
    mov byte ptr [r14], 0x7C
    inc r14
    mov byte ptr [r14], 0x0A
    inc r14
    mov rsi, rbp
    mov rdx, r14
    sub rdx, rbp
    call write_buf
    add rbx, r15
    add r13, r15
    jmp row_loop
print_final:
    mov rdi, r12
    mov rsi, rbp
    call hex8
    mov byte ptr [rsi], 0x0A
    inc rsi
    mov rdx, rsi
    sub rdx, rbp
    mov rsi, rbp
    call write_buf
    mov eax, 60
    xor edi, edi
    syscall
    mov eax, 231
    xor edi, edi
    syscall
hex8:
    lea r10, [rip + hexdig]
    mov eax, edi
    mov edx, eax
    shr edx, 28
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi], dl
    mov edx, eax
    shr edx, 24
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 1], dl
    mov edx, eax
    shr edx, 20
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 2], dl
    mov edx, eax
    shr edx, 16
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 3], dl
    mov edx, eax
    shr edx, 12
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 4], dl
    mov edx, eax
    shr edx, 8
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 5], dl
    mov edx, eax
    shr edx, 4
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 6], dl
    mov edx, eax
    and edx, 15
    movzx rdx, byte ptr [r10 + rdx]
    mov byte ptr [rsi + 7], dl
    add rsi, 8
    ret
write_buf:
    mov rdi, 1
write_loop:
    mov eax, 1
    syscall
    cmp rax, rdx
    jae write_done
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    ret
.section .rodata
.p2align 4
hexdig:
.ascii "0123456789abcdef"
.bss
.p2align 4
inbuf:
.zero 65536
.p2align 4
linebuf:
.zero 128
