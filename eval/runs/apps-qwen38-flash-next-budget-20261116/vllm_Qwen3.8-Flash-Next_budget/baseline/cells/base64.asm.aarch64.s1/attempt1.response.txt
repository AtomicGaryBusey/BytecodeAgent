

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ NEWLINE, 10
.equ EQUALS, 61
.equ LINE_LEN, 76
.equ INBUF_SIZE, 49152
.equ OUTBUF_SIZE, 70000

.text
.align 16
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x21, x20
    mov x2, #INBUF_SIZE
    subs x2, x2, x20
    beq encode
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    ble encode
    add x20, x20, x0
    mov x9, #INBUF_SIZE
    cmp x20, x9
    blt read_loop
encode:
    adrp x14, base64_alphabet
    add x14, x14, :lo12:base64_alphabet
    mov x5, x21
    add x6, x21, x20
    mov x7, x22
    mov x13, #0
full_loop:
    sub x9, x6, x5
    cmp x9, #3
    blt partial
    ldrb w10, [x5]
    ldrb w11, [x5, #1]
    ldrb w12, [x5, #2]
    mov x16, #0
    orr x16, x16, x10, lsl #16
    orr x16, x16, x11, lsl #8
    orr x16, x16, x12
    ubfx x3, x16, #18, #6
    ldrb w3, [x14, x3]
    bl emit_char
    ubfx x3, x16, #12, #6
    ldrb w3, [x14, x3]
    bl emit_char
    ubfx x3, x16, #6, #6
    ldrb w3, [x14, x3]
    bl emit_char
    and x3, x16, #63
    ldrb w3, [x14, x3]
    bl emit_char
    add x5, x5, #3
    b full_loop
partial:
    cmp x9, #0
    beq finish
    cmp x9, #1
    b.eq one_byte
    ldrb w10, [x5]
    ldrb w11, [x5, #1]
    mov x16, #0
    orr x16, x16, x10, lsl #16
    orr x16, x16, x11, lsl #8
    ubfx x3, x16, #18, #6
    ldrb w3, [x14, x3]
    bl emit_char
    ubfx x3, x16, #12, #6
    ldrb w3, [x14, x3]
    bl emit_char
    ubfx x3, x16, #6, #6
    ldrb w3, [x14, x3]
    bl emit_char
    mov w3, #EQUALS
    bl emit_char
    b finish
one_byte:
    ldrb w10, [x5]
    mov x16, #0
    orr x16, x16, x10, lsl #16
    ubfx x3, x16, #18, #6
    ldrb w3, [x14, x3]
    bl emit_char
    ubfx x3, x16, #12, #6
    ldrb w3, [x14, x3]
    bl emit_char
    mov w3, #EQUALS
    bl emit_char
    mov w3, #EQUALS
    bl emit_char
finish:
    cmp x13, #0
    beq output
    mov w3, #NEWLINE
    strb w3, [x7], #1
output:
    sub x2, x7, x22
    cmp x2, #0
    beq exit_ok
    mov x1, x22
write_loop:
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    ble exit_ok
    cmp x0, x2
    b.ge exit_ok
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
emit_char:
    strb w3, [x7], #1
    add x13, x13, #1
    cmp x13, #LINE_LEN
    b.ne emit_done
    mov w3, #NEWLINE
    strb w3, [x7], #1
    mov x13, #0
emit_done:
    ret

.section .rodata
.align 16
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buf:
    .zero INBUF_SIZE
.align 16
output_buf:
    .zero OUTBUF_SIZE
