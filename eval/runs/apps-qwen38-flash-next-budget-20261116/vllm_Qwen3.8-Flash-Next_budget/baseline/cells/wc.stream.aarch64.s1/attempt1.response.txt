

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ CHUNK, 4096

.section .rodata
.balign 8
spc_str:
.byte 0x20
.balign 8
nl_str:
.byte 0x0A

.bss
.balign 16
inbuf:
.zero CHUNK
.balign 16
print_buf:
.zero 64

.text
.globl _start

_start:
  adrp x19, inbuf
  add x19, x19, :lo12:inbuf
  mov x20, #0
  mov x21, #0
  mov x22, #0
  mov x23, #0
read_loop:
  mov x8, #SYS_READ
  mov x0, #0
  mov x1, x19
  mov x2, #CHUNK
  svc #0
  cmp x0, #0
  ble done_read
  add x20, x20, x0
  mov x24, x0
  mov x5, x19
proc_loop:
  ldrb w3, [x5], #1
  cmp w3, #0x0A
  bne proc_after_nl
  add x21, x21, #1
proc_after_nl:
  cmp w3, #9
  beq proc_ws
  cmp w3, #10
  beq proc_ws
  cmp w3, #11
  beq proc_ws
  cmp w3, #12
  beq proc_ws
  cmp w3, #13
  beq proc_ws
  cmp w3, #32
  beq proc_ws
  cbnz x23, proc_next
  add x22, x22, #1
  mov x23, #1
  b proc_next
proc_ws:
  mov x23, #0
proc_next:
  subs x24, x24, #1
  bgt proc_loop
  b read_loop
done_read:
  mov x0, x21
  bl print_uint
  adrp x1, spc_str
  add x1, x1, :lo12:spc_str
  mov x2, #1
  bl write_buf
  mov x0, x22
  bl print_uint
  adrp x1, spc_str
  add x1, x1, :lo12:spc_str
  mov x2, #1
  bl write_buf
  mov x0, x20
  bl print_uint
  adrp x1, nl_str
  add x1, x1, :lo12:nl_str
  mov x2, #1
  bl write_buf
  mov x8, #SYS_EXIT
  mov x0, #0
  svc #0
exit_stop:
  b exit_stop

print_uint:
  cbz x0, pu_zero
  adrp x1, print_buf
  add x1, x1, :lo12:print_buf
  mov x2, #0
  mov x4, #10
  mov x3, #63
  add x3, x1, x3
pu_fill:
  udiv x5, x0, x4
  msub x0, x4, x5, x0
  add x2, x2, #1
  add x6, x0, #0x30
  strb w6, [x3]
  subs x3, x3, #1
  mov x0, x5
  cbnz x0, pu_fill
  add x1, x3, #1
pu_write:
  cmp x2, #0
  ble pu_done
  mov x8, #SYS_WRITE
  mov x0, #1
  svc #0
  ble pu_done
  cmp x0, x2
  bge pu_done
  add x1, x1, x0
  subs x2, x2, x0
  bgt pu_write
pu_done:
  ret
pu_zero:
  adrp x1, print_buf
  add x1, x1, :lo12:print_buf
  mov x2, #1
  mov x3, #0x30
  strb w3, [x1]
pu_zero_write:
  mov x8, #SYS_WRITE
  mov x0, #1
  svc #0
  ble pu_zero_done
  cmp x0, x2
  bge pu_zero_done
  add x1, x1, x0
  subs x2, x2, x0
  bgt pu_zero_write
pu_zero_done:
  ret

write_buf:
  cmp x2, #0
  ble wb_done
wb_loop:
  mov x8, #SYS_WRITE
  mov x0, #1
  svc #0
  ble wb_done
  cmp x0, x2
  bge wb_done
  add x1, x1, x0
  subs x2, x2, x0
  bgt wb_loop
wb_done:
  ret
