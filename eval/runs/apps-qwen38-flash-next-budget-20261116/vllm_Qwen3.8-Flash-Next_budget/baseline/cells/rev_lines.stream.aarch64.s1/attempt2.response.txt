

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ NEWLINE, 10

.text
.globl _start
.balign 4
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf

    mov x24, #0
    movk x24, #1, lsl #16
    mov x21, #0

read_loop:
    cmp x21, x24
    b.ge read_done
    mov x0, #0
    add x1, x19, x21
    sub x2, x24, x21
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.gt read_append
    b read_done

read_append:
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0

process_line:
    cmp x22, x21
    b.ge finish

    mov x2, x22
    mov x3, x21

scan_loop:
    cmp x2, x3
    b.ge scan_done
    ldrb w4, [x19, x2]
    cmp w4, #NEWLINE
    b.eq scan_found
    add x2, x2, #1
    b scan_loop

scan_found:
    mov x5, x2
    add x6, x2, #1
    b reverse_line

scan_done:
    mov x5, x3
    mov x6, x3

reverse_line:
    sub x7, x5, x22
    mov x12, x7
    add x11, x20, x23
    cbz x12, copy_done
    sub x10, x5, #1
    add x10, x10, x19

copy_loop:
    cbz x12, copy_done
    ldrb w4, [x10]
    strb w4, [x11]
    add x11, x11, #1
    sub x12, x12, #1
    cbz x12, copy_done
    sub x10, x10, #1
    b copy_loop

copy_done:
    mov w4, #NEWLINE
    strb w4, [x11]
    add x23, x23, x7
    add x23, x23, #1
    mov x22, x6
    b process_line

finish:
    cbz x23, exit_program
    mov x1, x20
    mov x2, x23

write_loop:
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.gt write_progress
    b exit_program

write_progress:
    add x1, x1, x0
    sub x2, x2, x0
    cbz x2, exit_program
    b write_loop

exit_program:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 4
inbuf: .space 65536
.balign 4
outbuf: .space 131072
