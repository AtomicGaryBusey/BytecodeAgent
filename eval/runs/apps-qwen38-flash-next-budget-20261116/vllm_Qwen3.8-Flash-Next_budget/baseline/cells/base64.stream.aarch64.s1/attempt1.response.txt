

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ SYS_exit_group, 94
.equ INBUF_SIZE, 49152
.equ OUTBUF_SIZE, 70000
.equ LINE_LEN, 76

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.p2align 4
inbuf:
.zero INBUF_SIZE

.p2align 4
outbuf:
.zero OUTBUF_SIZE

.text
.p2align 2
.globl _start
_start:
 mov x19, #0
 adrp x21, inbuf
 add x21, x21, :lo12:inbuf
 mov x11, #INBUF_SIZE
read_loop:
 cmp x19, x11
 bge read_done
 add x1, x21, x19
 mov x0, #0
 mov x2, x11
 sub x2, x2, x19
 mov x8, #SYS_read
 svc #0
 cmp x0, #0
 ble read_done
 add x19, x19, x0
 b read_loop

read_done:
 cbz x19, exit_ok
 adrp x24, inbuf
 add x24, x24, :lo12:inbuf
 mov x20, x19
 adrp x26, outbuf
 add x26, x26, :lo12:outbuf
 mov x22, x26
 mov x27, #0
 adrp x3, alphabet
 add x3, x3, :lo12:alphabet

enc_loop3:
 cmp x20, #3
 blt enc_tail
 ldrb w0, [x24], #1
 ldrb w1, [x24], #1
 ldrb w2, [x24], #1
 sub x20, x20, #3
 lsl x0, x0, #16
 lsl x1, x1, #8
 orr x5, x0, x1
 orr x5, x5, x2
 ubfx x0, x5, #18, #6
 bl append_char
 ubfx x0, x5, #12, #6
 bl append_char
 ubfx x0, x5, #6, #6
 bl append_char
 and x0, x5, #63
 bl append_char
 b enc_loop3

enc_tail:
 cbz x20, enc_final
 cmp x20, #1
 bne enc_tail2
 ldrb w0, [x24], #1
 mov x2, x0
 ubfx w0, w0, #2, #6
 bl append_char
 and x0, x2, #3
 lsl x0, x0, #4
 bl append_char
 mov w0, #61
 bl append_raw
 mov w0, #61
 bl append_raw
 b enc_final

enc_tail2:
 ldrb w0, [x24], #1
 ldrb w1, [x24], #1
 mov x2, x0
 ubfx w0, w0, #2, #6
 bl append_char
 and x0, x2, #3
 lsl x0, x0, #4
 ubfx x4, x1, #4, #4
 orr x0, x0, x4
 bl append_char
 and x0, x1, #15
 lsl x0, x0, #2
 bl append_char
 mov w0, #61
 bl append_raw

enc_final:
 cbz x27, write_output
 mov w0, #10
 strb w0, [x26], #1

write_output:
 mov x1, x22
 sub x2, x26, x22
 cbz x2, exit_ok
write_loop:
 mov x0, #1
 mov x8, #SYS_write
 svc #0
 cmp x0, #0
 ble exit_ok
 add x1, x1, x0
 sub x2, x2, x0
 cbnz x2, write_loop

exit_ok:
 mov x0, #0
 mov x8, #SYS_exit
 svc #0
 mov x0, #0
 mov x8, #SYS_exit_group
 svc #0
exit_loop:
 b exit_loop

.p2align 2
append_char:
 add x6, x3, x0
 ldrb w7, [x6]
 strb w7, [x26], #1
 add x27, x27, #1
 cmp x27, #LINE_LEN
 bne ac_done
 mov w7, #10
 strb w7, [x26], #1
 mov x27, #0
ac_done:
 ret

.p2align 2
append_raw:
 strb w0, [x26], #1
 add x27, x27, #1
 cmp x27, #LINE_LEN
 bne ar_done
 mov w7, #10
 strb w7, [x26], #1
 mov x27, #0
ar_done:
 ret
