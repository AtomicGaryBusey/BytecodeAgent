

.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    movz x2, #1, lsl #16
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le print_done
    add x21, x21, x0
    mov x16, x0
    adrp x17, inbuf
    add x17, x17, :lo12:inbuf
byte_loop:
    ldrb w4, [x17], #1
    sub w5, w4, #9
    cmp w5, #4
    b.hi not_ws_range
    mov x22, #0
    cmp w4, #10
    b.ne not_newline
    add x19, x19, #1
not_newline:
    b byte_next
not_ws_range:
    cmp w4, #32
    b.ne not_space
    mov x22, #0
    b byte_next
not_space:
    cbz x22, new_word
    b byte_next
new_word:
    add x20, x20, #1
    mov x22, #1
byte_next:
    subs x16, x16, #1
    b.ne byte_loop
    b read_loop
print_done:
    mov x0, x19
    bl print_num
    mov x0, #32
    bl put_byte
    mov x0, x20
    bl print_num
    mov x0, #32
    bl put_byte
    mov x0, x21
    bl print_num
    mov x0, #10
    bl put_byte
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    adrp x2, out_len
    add x2, x2, :lo12:out_len
    ldr x2, [x2]
write_again:
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le final_exit
    add x1, x1, x0
    cmp x2, x0
    b.le final_exit
    subs x2, x2, x0
    b write_again
final_exit:
    mov x0, #0
    mov x8, #93
    svc #0

print_num:
    mov x2, x0
    adrp x1, num_buf
    add x1, x1, :lo12:num_buf
    mov x5, #32
    add x5, x1, x5
    mov x6, #10
    cbz x2, num_zero
num_loop:
    udiv x8, x2, x6
    msub x7, x8, x6, x2
    add x7, x7, #48
    sub x5, x5, #1
    strb w7, [x5]
    mov x2, x8
    cbnz x2, num_loop
    b num_copy
num_zero:
    sub x5, x5, #1
    mov w7, #48
    strb w7, [x5]
num_copy:
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    adrp x9, out_len
    add x9, x9, :lo12:out_len
    ldr x4, [x9]
    add x10, x1, x4
    adrp x3, num_buf
    add x3, x3, :lo12:num_buf
    add x3, x3, #32
    sub x11, x3, x5
    mov x13, x11
num_cp:
    cbz x11, num_done
    ldrb w12, [x5], #1
    strb w12, [x10], #1
    subs x11, x11, #1
    b.ne num_cp
num_done:
    add x4, x4, x13
    str x4, [x9]
    ret

put_byte:
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    adrp x2, out_len
    add x2, x2, :lo12:out_len
    ldr x3, [x2]
    add x4, x1, x3
    strb w0, [x4]
    add x3, x3, #1
    str x3, [x2]
    ret

.bss
.p2align 3
