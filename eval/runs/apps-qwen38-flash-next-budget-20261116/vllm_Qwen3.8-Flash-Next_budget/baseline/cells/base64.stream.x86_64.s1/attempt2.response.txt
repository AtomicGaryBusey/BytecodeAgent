

.globl _start
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 262144

.text
_start:
    lea r12, [rip + in_buf]
    lea r15, [rip + out_buf]
    xor r13d, r13d
    xor ebp, ebp
    lea rbx, [rip + alphabet]
read_loop:
    mov rdx, IN_BUF_SIZE
    sub rdx, r13
    je read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r13]
    syscall
    cmp rax, -11
    je read_loop
    cmp rax, -4
    je read_loop
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov rax, r13
    xor edx, edx
    mov ecx, 3
    div ecx
    mov r8, rax
    mov r9, r12
group_loop:
    test r8, r8
    je tail
    movzx eax, byte ptr [r9]
    movzx ecx, byte ptr [r9 + 1]
    movzx edx, byte ptr [r9 + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov r14d, eax
    mov ecx, r14d
    shr ecx, 18
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    mov ecx, r14d
    shr ecx, 12
    and ecx, 63
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    mov ecx, r14d
    shr ecx, 6
    and ecx, 63
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    mov ecx, r14d
    and ecx, 63
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    add r9, 3
    dec r8
    jmp group_loop
tail:
    lea r10, [r12 + r13]
    sub r10, r9
    cmp r10, 2
    je tail2
    cmp r10, 1
    je tail1
    jmp end_encode
tail1:
    movzx eax, byte ptr [r9]
    mov r14d, eax
    mov ecx, r14d
    shr ecx, 2
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    mov ecx, r14d
    and ecx, 3
    shl ecx, 4
    movzx eax, byte ptr [rbx + rcx]
    call emit_char
    mov al, 61
    call emit_char
    mov al, 61
    call emit_char
    jmp end_encode
tail2:
    movzx eax, byte ptr [r9]
    movzx ecx, byte ptr [r9 + 1]
    mov r14d, eax
    mov r10d, ecx
    mov edx, r14d
    shr edx, 2
    movzx eax, byte ptr [rbx + rdx]
    call emit_char
    mov edx, r14d
    and edx, 3
    shl edx, 4
    mov ecx, r10d
    shr ecx, 4
    or edx, ecx
    movzx eax, byte ptr [rbx + rdx]
    call emit_char
    mov edx, r10d
    and edx, 15
    shl edx, 2
    movzx eax, byte ptr [rbx + rdx]
    call emit_char
    mov al, 61
    call emit_char
    jmp end_encode
end_encode:
    lea rax, [rip + out_buf]
    cmp r15, rax
    je exit_zero
    test rbp, rbp
    je write_start
    mov byte ptr [r15], 10
    inc r15
write_start:
    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    je exit_zero
write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    cmp rax, -11
    je write_loop
    cmp rax, -4
    je write_loop
    test rax, rax
    jle write_done
    cmp rax, rdx
    jae write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
exit_zero:
    xor edi, edi
    mov rax, 60
    syscall
emit_char:
    mov byte ptr [r15], al
    inc r15
    inc rbp
    cmp rbp, 76
    jne 1f
    mov byte ptr [r15], 10
    inc r15
    xor ebp, ebp
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.p2align 4
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE
