

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ NL, 10
.equ INBUF_SIZE, 65536
.equ BUF_SIZE, 65537

.text
.globl _start

_start:
    xor ebx, ebx

read_loop:
    mov rdi, 0
    lea rsi, [rip + buf]
    add rsi, rbx
    mov rdx, INBUF_SIZE
    sub rdx, rbx
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    js read_err
    add rbx, rax
    cmp rbx, INBUF_SIZE
    jb read_loop
    mov rbx, INBUF_SIZE
    jmp read_done

read_err:
    cmp rax, -4
    je read_loop
    jmp exit0

read_done:
    mov r10, rbx
    test r10, r10
    je exit0

    lea rdi, [rip + buf]
    xor r8, r8
    xor r9, r9

scan_loop:
    cmp r9, r10
    jae scan_end
    movzx eax, byte ptr [rdi + r9]
    cmp eax, NL
    jne no_nl

do_reverse_scan:
    mov rax, r9
    sub rax, r8
    cmp rax, 1
    jle continue_scan

    lea rcx, [rdi + r8]
    lea rdx, [rdi + r9]
    dec rdx

scan_reverse_loop:
    cmp rcx, rdx
    jae continue_scan
    mov al, byte ptr [rcx]
    mov bl, byte ptr [rdx]
    mov byte ptr [rdx], al
    mov byte ptr [rcx], bl
    inc rcx
    dec rdx
    jmp scan_reverse_loop

continue_scan:
    mov r8, r9
    inc r8
    inc r9
    jmp scan_loop

no_nl:
    inc r9
    jmp scan_loop

scan_end:
    mov rcx, r10
    dec rcx
    movzx eax, byte ptr [rdi + rcx]
    cmp eax, NL
    je write_output

    mov r9, r10
    mov byte ptr [rdi + r10], NL
    inc r10

    mov rax, r9
    sub rax, r8
    cmp rax, 1
    jle write_output

    lea rcx, [rdi + r8]
    lea rdx, [rdi + r9]
    dec rdx

final_reverse_loop:
    cmp rcx, rdx
    jae write_output
    mov al, byte ptr [rcx]
    mov bl, byte ptr [rdx]
    mov byte ptr [rdx], al
    mov byte ptr [rcx], bl
    inc rcx
    dec rdx
    jmp final_reverse_loop

write_output:
    mov rsi, rdi
    mov rdi, 1
    mov rdx, r10
    mov rax, SYS_WRITE
    syscall

exit0:
    mov rdi, 0
    mov rax, SYS_EXIT
    syscall

.bss
.balign 16
buf:
    .zero BUF_SIZE
