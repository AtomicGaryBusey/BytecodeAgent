

.section .rodata
.balign 16
error_str:
    .ascii "error\n"
error_len = . - error_str

.bss
.balign 16
input_buf:
    .zero 65536
eval_stack:
    .zero 16384
num_buf:
    .zero 64

.text
.globl _start

_start:
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    adrp x23, eval_stack
    add x23, x23, :lo12:eval_stack

    mov x27, x23
    mov x8, #16384
    add x27, x27, x8

    mov x25, #0
    movk x25, #1, lsl #16

    mov x28, #0
    movk x28, #32768, lsl #48

    mov x19, #0

read_loop:
    cmp x19, x25
    bge read_done
    mov x0, #0
    mov x1, x24
    add x1, x1, x19
    mov x2, x25
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0
    cmp x19, #0
    beq prog_exit

line_start:
    mov x21, x23
    mov x22, #0

line_scan:
    cmp x20, x19
    bge end_line
    ldrb w7, [x24, x20]
    cmp w7, #10
    beq end_line
    cmp w7, #32
    beq advance_sep
    cmp w7, #9
    bne token_begin

advance_sep:
    add x20, x20, #1
    b line_scan

token_begin:
    mov x4, x20

token_scan:
    add x20, x20, #1
    cmp x20, x19
    bge token_end
    ldrb w7, [x24, x20]
    cmp w7, #10
    beq token_end
    cmp w7, #32
    beq token_end
    cmp w7, #9
    beq token_end
    b token_scan

token_end:
    mov x5, x20
    mov x6, x5
    sub x6, x6, x4

    cmp x6, #1
    bne lit
    ldrb w7, [x4]
    cmp w7, #43
    beq op_add
    cmp w7, #45
    beq op_sub
    cmp w7, #42
    beq op_mul
    cmp w7, #47
    beq op_div
    cmp w7, #37
    beq op_mod
    b error_skip

lit:
    cmp x6, #0
    beq error_skip
    ldrb w7, [x4]
    cmp w7, #45
    bne lit_no_minus

    cmp x6, #19
    bgt error_skip
    add x8, x4, #1
    mov x9, x5
    cmp x8, x9
    beq error_skip
    b lit_parse

lit_no_minus:
    cmp x6, #18
    bgt error_skip
    mov x8, x4
    mov x9, x5
    cmp x8, x9
    beq error_skip

lit_parse:
    mov x10, #0
    mov x11, #10

lit_loop:
    cmp x8, x9
    b.ge lit_done
    ldrb w7, [x8]
    cmp w7, #48
    blo error_skip
    cmp w7, #57
    bhi error_skip
    sub w7, w7, #48
    mul x10, x10, x11
    add x10, x10, x7
    add x8, x8, #1
    b lit_loop

lit_done:
    ldrb w7, [x4]
    cmp w7, #45
    bne lit_store
    neg x10, x10

lit_store:
    cmp x21, x27
    bhs error_skip
    str x10, [x21]
    add x21, x21, #8
    b line_scan

op_add:
    sub x7, x21, x23
    cmp x7, #16
    blt error_skip
    sub x21, x21, #16
    ldp x0, x1, [x21]
    add x0, x0, x1
    b op_store

op_sub:
    sub x7, x21, x23
    cmp x7, #16
    blt error
