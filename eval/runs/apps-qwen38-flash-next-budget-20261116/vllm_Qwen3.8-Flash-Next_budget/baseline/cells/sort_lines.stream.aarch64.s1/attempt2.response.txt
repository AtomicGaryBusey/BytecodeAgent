

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.text
.globl _start
_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
read_next:
    cmp x19, #MAX_INPUT
    b.hs read_done
    mov x0, #STDIN
    add x1, x20, x19
    mov x2, #MAX_INPUT
    sub x2, x2, x19
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_next
read_done:
    adrp x25, records
    add x25, x25, :lo12:records
    mov x22, x20
    add x23, x20, x19
    mov x24, x20
    mov x26, #0
parse_scan:
    cmp x22, x23
    b.hs parse_final
    ldrb w5, [x22]
    add x22, x22, #1
    cmp w5, #10
    b.ne parse_scan
    sub x6, x22, x24
    sub x6, x6, #1
    cmp x26, #MAX_LINES
    b.hs parse_next_line
    add x7, x25, x26, lsl #4
    str x24, [x7]
    str x6, [x7, #8]
    add x26, x26, #1
parse_next_line:
    mov x24, x22
    b parse_scan
parse_final:
    cmp x24, x23
    b.hs parse_done
    sub x6, x23, x24
    cmp x26, #MAX_LINES
    b.hs parse_done
    add x7, x25, x26, lsl #4
    str x24, [x7]
    str x6, [x7, #8]
    add x26, x26, #1
parse_done:
    mov x19, x26
    mov x20, #1
sort_outer:
    cmp x20, x19
    b.hs sort_done
    add x21, x25, x20, lsl #4
    ldp x22, x23, [x21]
sort_inner:
    cmp x21, x25
    b.ls sort_place
    sub x24, x21, #16
    ldp x1, x2, [x24]
    mov x4, x23
    cmp x2, x23
    b.hs cmp_chunk_start
    mov x4, x2
cmp_chunk_start:
    mov x5, #0
cmp_chunk:
    cmp x5, x4
    b.hs cmp_after_bytes
    sub x7, x4, x5
    cmp x7, #8
    b.lo cmp_tail
    ldr x6, [x22, x5]
    ldr x7, [x1, x5]
    rev x6, x6
    rev x7, x7
    cmp x6, x7
    b.lo cmp_v_less
    b.hi cmp_v_greater
    add x5, x5, #8
    b cmp_chunk
cmp_tail:
    cmp x5, x4
    b.hs cmp_after_bytes
    ldrb w6, [x22, x5]
    ldrb w7, [x1, x5]
    cmp w6, w7
    b.lo cmp_v_less
    b.hi cmp_v_greater
    add x5, x5, #1
    b cmp_tail
cmp_after_bytes:
    cmp x23, x2
    b.lo cmp_v_less
    b cmp_v_greater
cmp_v_less:
    stp x1, x2, [x21]
    mov x21, x24
    b sort_inner
cmp_v_greater:
    b sort_place
sort_place:
    stp x22, x23, [x21]
    add x20, x20, #1
    b sort_outer
sort_done:
    mov x20, #0
    adrp x24, newline
    add x24, x24, :lo12:newline
output_loop:
    cmp x20, x19
    b.hs output_done
    add x21, x25, x20, lsl #4
    ldr x22, [x21]
    ldr x23, [x21, #8]
output_write:
    cbz x23, output_nl_setup
    mov x0, #STDOUT
    mov x1, x22
    mov x2, x23
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le output_nl_setup
    add x22, x22, x0
    sub x23, x23, x0
    b output_write
output_nl_setup:
    mov x22, x24
    mov x23, #1
output_nl_write:
    cbz x23, output_next
    mov x0, #STDOUT
    mov x1, x22
    mov x2, x23
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le output_next
    add x22, x22, x0
    sub x23, x23, x0
    b output_nl_write
output_next:
    add x20, x20, #1
    b output_loop
output_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
exit_forever:
    b exit_forever

.bss
.align 16
input_buf:
    .zero 32768
.align 16
records:
    .zero 16000

.data
.align 16
newline:
    .byte 10
