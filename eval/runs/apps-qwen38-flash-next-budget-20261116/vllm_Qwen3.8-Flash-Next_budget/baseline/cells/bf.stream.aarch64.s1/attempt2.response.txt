

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ TAPE_MAX, 29999

.text
.globl _start
_start:
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    mov x3, #0
    movz x19, #1, lsl #16
read_next:
    cmp x3, x19
    bge read_done
    add x1, x5, x3
    sub x2, x19, x3
    mov x0, #0
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    ble read_done
    add x3, x3, x0
    b read_next
read_done:
    mov x19, x3
    mov x6, #0
find_sep:
    cmp x6, x19
    bge no_sep
    ldrb w4, [x5, x6]
    cmp w4, #0x21
    beq sep_found
    add x6, x6, #1
    b find_sep
sep_found:
    mov x15, x6
    add x17, x6, #1
    sub x12, x19, x17
    mov x11, #0
    b init_balance
no_sep:
    mov x15, x19
    mov x17, x19
    mov x12, #0
    mov x11, #0
init_balance:
    adrp x16, match
    add x16, x16, :lo12:match
    adrp x22, br_stack
    add x22, x22, :lo12:br_stack
    mov x21, #0
    mov x6, #0
balance_next:
    cmp x6, x15
    bge balance_done
    ldrb w4, [x5, x6]
    cmp w4, #0x5b
    beq is_open
    cmp w4, #0x5d
    beq is_close
    add x6, x6, #1
    b balance_next
is_open:
    str w6, [x22, x21, lsl #2]
    add x21, x21, #1
    add x6, x6, #1
    b balance_next
is_close:
    cmp x21, #0
    beq err_exit
    sub x21, x21, #1
    ldr w1, [x22, x21, lsl #2]
    add x2, x6, #1
    str w2, [x16, x1, lsl #2]
    add x1, x1, #1
    str w1, [x16, x6, lsl #2]
    add x6, x6, #1
    b balance_next
balance_done:
    cmp x21, #0
    bne err_exit
    adrp x10, tape
    add x10, x10, :lo12:tape
    adrp x13, outbuf
    add x13, x13, :lo12:outbuf
    movz x20, #1, lsl #16
    movz x21, #TAPE_MAX
    mov x6, #0
    mov x9, #0
    mov x14, #0
run_next:
    cmp x6, x15
    bge run_done
    mov x7, x6
    add x6, x6, #1
    ldrb w4, [x5, x7]
    cmp w4, #0x3e
    beq cmd_gt
    cmp w4, #0x3c
    beq cmd_lt
    cmp w4, #0x2b
    beq cmd_plus
    cmp w4, #0x2d
    beq cmd_minus
    cmp w4, #0x2e
    beq cmd_dot
    cmp w4, #0x2c
    beq cmd_comma
    cmp w4, #0x5b
    beq cmd_open
    cmp w4, #0x5d
    beq cmd_close
    b run_next
cmd_gt:
    cmp x9, x21
    bge run_next
    add x9, x9, #1
    b run_next
cmd_lt:
    cmp x9, #0
    beq run_next
    sub x9, x9, #1
    b run_next
cmd_plus:
    ldrb w0, [x10, x9]
    add w0, w0, #1
    and w0, w0, #0xff
    strb w0, [x10, x9]
    b run_next
cmd_minus:
    ldrb w0, [x10, x9]
    sub w0, w0, #1
    and w0, w0, #0xff
    strb w0, [x10, x9]
    b run_next
cmd_dot:
    cmp x14, x20
    bge run_next
    ldrb w0, [x10, x9]
    add x2, x13, x14
    strb w0, [x2]
    add x14, x14, #1
    b run_next
cmd_comma:
    cmp x11, x12
    bge comma_eof
    add x7, x17, x11
    ldrb w0, [x5, x7]
    add x11, x11, #1
    strb w0, [x10, x9]
    b run_next
comma_eof:
    mov w0, #0
    strb w0, [x10, x9]
    b run_next
cmd_open:
    ldrb w0, [x10, x9]
    cmp w0, #0
    beq open_jump
    b run_next
open_jump:
    ldr w6, [x16, x7, lsl #2]
    b run_next
cmd_close:
    ldrb w0, [x10, x9]
    cmp w0, #0
    bne close_jump
    b run_next
close_jump:
    ldr w6, [x16, x7, lsl #2]
    b run_next
run_done:
    cmp x14, #0
    beq exit_ok
    mov x1, x13
    mov x2, x14
write_next:
    cmp x2, #0
    ble exit_ok
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    ble exit_ok
    add x1, x1, x0
    sub x2, x2, x0
    b write_next
exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
err_exit:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buf:
    .zero MAX_INPUT
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
outbuf:
    .zero MAX_INPUT
.balign 16
match:
    .zero 262144
.balign 16
br_stack:
    .zero 262144
