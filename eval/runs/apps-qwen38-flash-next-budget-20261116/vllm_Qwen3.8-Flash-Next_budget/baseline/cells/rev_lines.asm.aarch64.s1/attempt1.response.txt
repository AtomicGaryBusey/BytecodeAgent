

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 94
.equ INPUT_MAX, 65536
.equ NL_BYTE, 10

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x0, #STDIN
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, read_done
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, x2
    bgt read_cap
    cmp x0, #0
    bgt read_acc
    b read_done
read_cap:
    mov x0, x2
read_acc:
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    mov x22, #0
    mov x24, #0
process_loop:
    cmp x22, x20
    bge process_final
    add x0, x19, x22
    ldrb w1, [x0]
    cmp w1, #NL_BYTE
    bne process_inc
    mov x0, x21
    mov x1, x22
    bl reverse_segment
    mov x21, x22
    add x21, x21, #1
    add x22, x22, #1
    b process_loop
process_inc:
    add x22, x22, #1
    b process_loop
process_final:
    cmp x21, x20
    bge process_no_partial
    mov x0, x21
    mov x1, x20
    bl reverse_segment
    mov x24, #1
    b write_all_main
process_no_partial:
    mov x24, #0
write_all_main:
    mov x0, x19
    mov x1, x20
    bl write_all
    cbz x24, exit_now
    adrp x0, nl
    add x0, x0, :lo12:nl
    mov x1, #1
    bl write_all
exit_now:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    b exit_now

reverse_segment:
    sub x4, x1, x0
    cmp x4, #2
    blt reverse_segment_done
    mov x5, x19
    add x5, x5, x0
    mov x6, x19
    add x6, x6, x1
    sub x6, x6, #1
reverse_segment_loop:
    ldrb w7, [x5]
    ldrb w3, [x6]
    strb w7, [x6]
    strb w3, [x5]
    add x5, x5, #1
    sub x6, x6, #1
    cmp x5, x6
    blt reverse_segment_loop
reverse_segment_done:
    ret

write_all:
    mov x4, x0
    mov x5, x1
write_all_loop:
    cmp x5, #0
    ble write_all_done
    mov x0, #STDOUT
    mov x1, x4
    mov x2, x5
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    ble write_all_done
    add x4, x4, x0
    sub x5, x5, x0
    b write_all_loop
write_all_done:
    ret

.section .rodata
nl: .byte NL_BYTE

.bss
inbuf: .zero INPUT_MAX
