

.bss
in_buf:
    .zero 65536
prog:
    .zero 65536
tape:
    .zero 30000
match:
    .zero 262144
bstack:
    .zero 262144
out_buf:
    .zero 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12d, 65536
    jae read_done
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + in_buf]
    lea r13, [rdi + r12]
    mov r14, r13
    lea rsi, [rip + prog]
    xor r15, r15
    mov rbx, rdi
filter_loop:
    cmp rbx, r13
    jae filter_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x21
    je bang_found
    cmp al, 0x3e
    je filter_store
    cmp al, 0x3c
    je filter_store
    cmp al, 0x2b
    je filter_store
    cmp al, 0x2d
    je filter_store
    cmp al, 0x2e
    je filter_store
    cmp al, 0x2c
    je filter_store
    cmp al, 0x5b
    je filter_store
    cmp al, 0x5d
    je filter_store
    inc rbx
    jmp filter_loop
bang_found:
    mov r14, rbx
    inc r14
    jmp filter_done
filter_store:
    mov byte ptr [rsi + r15], al
    inc r15
    inc rbx
    jmp filter_loop
filter_done:
    mov r12, r15
    lea r10, [rip + match]
    lea r15, [rip + bstack]
    mov rbp, r15
    xor rbx, rbx
match_loop:
    cmp rbx, r12
    jae match_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x5b
    je match_push
    cmp al, 0x5d
    je match_pop
    inc rbx
    jmp match_loop
match_push:
    mov dword ptr [rbp], ebx
    add rbp, 4
    inc rbx
    jmp match_loop
match_pop:
    cmp rbp, r15
    je error_exit
    sub rbp, 4
    mov ecx, dword ptr [rbp]
    mov dword ptr [r10 + rcx*4], ebx
    mov dword ptr [r10 + rbx*4], ecx
    inc rbx
    jmp match_loop
match_done:
    cmp rbp, r15
    jne error_exit
    xor rbx, rbx
    lea rbp, [rip + tape]
    lea r9, [rip + out_buf]
    mov r8, r9
exec_loop:
    cmp rbx, r12
    jae exec_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x3e
    je exec_gt
    cmp al, 0x3c
    je exec_lt
    cmp al, 0x2b
    je exec_plus
    cmp al, 0x2d
    je exec_minus
    cmp al, 0x2e
    je exec_dot
    cmp al, 0x2c
    je exec_comma
    cmp al, 0x5b
    je exec_lbr
    cmp al, 0x5d
    je exec_rbr
    inc rbx
    jmp exec_loop
exec_gt:
    inc rbx
    inc rbp
    jmp exec_loop
exec_lt:
    inc rbx
    dec rbp
    jmp exec_loop
exec_plus:
    inc rbx
    inc byte ptr [rbp]
    jmp exec_loop
exec_minus:
    inc rbx
    dec byte ptr [rbp]
    jmp exec_loop
exec_dot:
    movzx eax, byte ptr [rbp]
    mov byte ptr [r8], al
    inc rbx
    inc r8
    jmp exec_loop
exec_comma:
    inc rbx
    cmp r14, r13
    jae exec_comma_zero
    movzx eax, byte ptr [r14]
    mov byte ptr [rbp], al
    inc r14
    jmp exec_loop
exec_comma_zero:
    mov byte ptr [rbp], 0
    jmp exec_loop
exec_lbr:
    mov edx, ebx
    cmp byte ptr [rbp], 0
    je exec_lbr_zero
    inc rbx
    jmp exec_loop
exec_lbr_zero:
    mov eax, dword ptr [r10 + rdx*4]
    mov ebx, eax
    inc rbx
    jmp exec_loop
exec_rbr:
    mov edx, ebx
    cmp byte ptr [rbp], 0
    je exec_rbr_zero
    mov eax, dword ptr [r10 + rdx*4]
    mov ebx, eax
    inc rbx
    jmp exec_loop
exec_rbr_zero:
    inc rbx
    jmp exec_loop
exec_done:
    mov rdx, r8
    sub rdx, r9
    test rdx, rdx
    jz exit_ok
    mov rsi, r9
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    cmp rdx, rax
    jbe exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
error_exit:
    mov edi, 1
    mov eax, 60
    syscall
