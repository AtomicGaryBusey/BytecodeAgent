

.bss
input_buf:
    .zero 65536
cmds:
    .zero 65536
tape:
    .zero 30000
match:
    .zero 262144
bstack:
    .zero 262144
output_buf:
    .zero 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12d, 65536
    jae read_done
    mov edx, 65536
    sub edx, r12d
    test edx, edx
    je read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + input_buf]
    mov r13, r12
    xor rbx, rbx
find_loop:
    cmp rbx, r13
    jae no_bang
    cmp byte ptr [rdi + rbx], 0x21
    je bang_found
    inc rbx
    jmp find_loop
bang_found:
    mov r14, rbx
    lea r12, [rdi + rbx]
    inc r12
    add r13, rdi
    jmp compress
no_bang:
    mov r14, rbx
    lea r12, [rdi + r13]
    add r13, rdi
compress:
    lea rsi, [rip + cmds]
    xor rbx, rbx
    xor r15, r15
compress_loop:
    cmp rbx, r14
    jae compress_done
    movzx eax, byte ptr [rdi + rbx]
    cmp al, 0x3e
    je store_cmd
    cmp al, 0x3c
    je store_cmd
    cmp al, 0x2b
    je store_cmd
    cmp al, 0x2d
    je store_cmd
    cmp al, 0x2e
    je store_cmd
    cmp al, 0x2c
    je store_cmd
    cmp al, 0x5b
    je store_cmd
    cmp al, 0x5d
    je store_cmd
    inc rbx
    jmp compress_loop
store_cmd:
    mov byte ptr [rsi + r15], al
    inc r15
    inc rbx
    jmp compress_loop
compress_done:
    mov r14, r15
parse_match:
    lea r10, [rip + match]
    lea r15, [rip + bstack]
    mov rbp, r15
    xor rbx, rbx
match_loop:
    cmp rbx, r14
    jae stack_check
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x5b
    je push_br
    cmp al, 0x5d
    je pop_br
    inc rbx
    jmp match_loop
push_br:
    mov dword ptr [rbp], ebx
    add rbp, 4
    inc rbx
    jmp match_loop
pop_br:
    cmp rbp, r15
    je error_exit
    sub rbp, 4
    mov ecx, dword ptr [rbp]
    mov dword ptr [r10 + rcx*4], ebx
    mov dword ptr [r10 + rbx*4], ecx
    inc rbx
    jmp match_loop
stack_check:
    cmp rbp, r15
    jne error_exit
    xor rbx, rbx
    xor rbp, rbp
    lea r11, [rip + tape]
    lea r9, [rip + output_buf]
    mov r8, r9
exec_loop:
    cmp rbx, r14
    jae exec_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x3e
    je cmd_gt
    cmp al, 0x3c
    je cmd_lt
    cmp al, 0x2b
    je cmd_plus
    cmp al, 0x2d
    je cmd_minus
    cmp al, 0x2e
    je cmd_dot
    cmp al, 0x2c
    je cmd_comma
    cmp al, 0x5b
    je cmd_lbr
    cmp al, 0x5d
    je cmd_rbr
    inc rbx
    jmp exec_loop
cmd_gt:
    inc rbx
    inc rbp
    jmp exec_loop
cmd_lt:
    dec rbx
    dec rbp
    jmp exec_loop
cmd_plus:
    inc rbx
    inc byte ptr [r11 + rbp]
    jmp exec_loop
cmd_minus:
    dec rbx
    dec byte ptr [r11 + rbp]
    jmp exec_loop
cmd_dot:
    movzx eax, byte ptr [r11 + rbp]
    mov byte ptr [r8], al
    inc rbx
    inc r8
    jmp exec_loop
cmd_comma:
    inc rbx
    cmp r12, r13
    jae comma_zero
    movzx eax, byte ptr [r12]
    mov byte ptr [r11 + rbp], al
    inc r12
    jmp exec_loop
comma_zero:
    mov byte ptr [r11 + rbp], 0
    jmp exec_loop
cmd_lbr:
    mov edx, ebx
    cmp byte ptr [r11 + rbp], 0
    jne lbr_nonzero
    mov eax, dword ptr [r10 + rdx*4]
    mov ebx, eax
    inc rbx
    jmp exec_loop
lbr_nonzero:
    mov ebx, edx
    inc rbx
    jmp exec_loop
cmd_rbr:
    mov edx, ebx
    cmp byte ptr [r11 + rbp], 0
    je rbr_zero
    mov eax, dword ptr [r10 + rdx*4]
    mov ebx, eax
    inc rbx
    jmp exec_loop
rbr_zero:
    mov ebx, edx
    inc rbx
    jmp exec_loop
exec_done:
    mov rdx, r8
    sub rdx, r9
    test rdx, rdx
    jz do_exit
    mov rsi, r9
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    cmp rdx, rax
    jbe do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    xor edi, edi
    mov eax, 60
    syscall
error_exit:
    mov edi, 1
    mov eax, 60
    syscall
