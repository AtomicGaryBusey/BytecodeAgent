

.globl _start

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="

.bss
.align 16
in_buf:
    .zero 49152
.align 16
out_buf:
    .zero 70000

.text
_start:
    lea rsi, [rip + in_buf]
    xor r12d, r12d
read_loop:
    mov rdx, 49152
    sub rdx, r12
    cmp rdx, 0
    jle read_done
    mov eax, 0
    mov edi, 0
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    add r12, rax
    cmp r12, 49152
    jae read_done
    jmp read_loop
read_done:
    lea rbx, [rip + base64_alphabet]
    lea rsi, [rip + in_buf]
    lea rbp, [rip + out_buf]
    mov rdi, rbp
    xor r14d, r14d

encode_loop:
    cmp r12, 3
    jb encode_tail
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov r8d, eax

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    call emit_char

    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    call emit_char

    mov ecx, r8d
    and ecx, 63
    call emit_char

    add rsi, 3
    sub r12, 3
    jmp encode_loop

encode_tail:
    cmp r12, 1
    je encode_1
    cmp r12, 2
    je encode_2
    jmp encode_finish

encode_2:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    mov r8d, eax

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    call emit_char

    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    call emit_char

    mov ecx, 64
    call emit_char
    jmp encode_finish

encode_1:
    movzx eax, byte ptr [rsi]
    mov r8d, eax
    shl r8d, 16

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    call emit_char

    mov ecx, 64
    call emit_char

    mov ecx, 64
    call emit_char
    jmp encode_finish

encode_finish:
    test r14d, r14d
    je write_length
    mov byte ptr [rdi], 10
    inc rdi

write_length:
    sub rdi, rbp
    mov r13, rdi
    test r13, r13
    je exit_zero
    mov rsi, rbp

write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub r13, rax
    cmp r13, 0
    jg write_loop

write_done:
exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
    jmp exit_zero

emit_char:
    cmp r14d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
1:
    movzx eax, byte ptr [rbx + rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc r14d
    ret
