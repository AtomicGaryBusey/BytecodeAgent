

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ SYS_EXIT_GROUP, 231
.equ STDIN, 0
.equ STDOUT, 1
.equ STDERR, 2
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 524288
.equ STACK_MAX, 2048
.equ VAL_STACK_SIZE, 16384
.equ ERROR_LEN, 6
.equ CHAR_SP, 0x20
.equ CHAR_TAB, 0x09
.equ CHAR_NL, 0x0A
.equ CHAR_PLUS, 0x2B
.equ CHAR_MINUS, 0x2D
.equ CHAR_MUL, 0x2A
.equ CHAR_DIV, 0x2F
.equ CHAR_MOD, 0x25
.equ CHAR_ZERO, 0x30
.equ CHAR_NINE, 0x39

.section .rodata
.align 8
llong_min_val:
.quad -9223372036854775808
.align 1
error_msg:
.ascii "error"
.byte CHAR_NL
minus_msg:
.byte CHAR_MINUS
zero_msg:
.byte CHAR_ZERO
newline_msg:
.byte CHAR_NL

.bss
.align 16
input_buf:
.zero IN_BUF_SIZE
.align 16
out_buf:
.zero OUT_BUF_SIZE
.align 8
out_len:
.zero 8
in_len:
.zero 8
.align 8
val_stack:
.zero VAL_STACK_SIZE

.text
.globl _start
_start:
  cld
  xor eax, eax
  xor edi, edi
  xor ebx, ebx
read_loop:
  mov rdx, IN_BUF_SIZE
  sub rdx, rbx
  jle read_done
  lea rsi, [rip + input_buf]
  add rsi, rbx
  syscall
  test rax, rax
  jle read_done
  cmp rax, rdx
  jbe read_add
  mov rax, rdx
read_add:
  add rbx, rax
  jmp read_loop
read_done:
  mov r12, rbx
  mov qword ptr [rip + in_len], rbx
  lea rbx, [rip + input_buf]
  xor ebp, ebp
main_loop:
  cmp rbp, r12
  jae main_exit
  mov rdi, rbx
  add rdi, rbp
  mov rsi, rbx
  add rsi, r12
  call eval_line
  sub rax, rbx
  mov rbp, rax
  jmp main_loop
main_exit:
  call flush_output
  xor edi, edi
  mov eax, SYS_EXIT
  syscall
  xor edi, edi
  mov eax, SYS_EXIT_GROUP
  syscall
main_exit_halt:
  jmp main_exit_halt

eval_line:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, rdi
  mov r12, rsi
  xor ebp, ebp
  xor r13, r13
eval_scan:
  cmp rbx, r12
  jae eval_finish
  movzx eax, byte ptr [rbx]
  cmp eax, CHAR_SP
  je eval_sep
  cmp eax, CHAR_TAB
  je eval_sep
  cmp eax, CHAR_NL
  je eval_finish
  cmp ebp, 0
  jne eval_skip_to_nl
  jmp eval_parse_token
eval_sep:
  inc rbx
  jmp eval_scan
eval_skip_to_nl:
  cmp rbx, r12
  jae eval_finish
  cmp byte ptr [rbx], CHAR_NL
  je eval_finish
  inc rbx
  jmp eval_skip_to_nl
eval_parse_token:
  mov r14, rbx
eval_tok_scan:
  cmp r14, r12
  jae eval_tok_done
  movzx eax, byte ptr [r14]
  cmp eax, CHAR_SP
  je eval_tok_done
  cmp eax, CHAR_TAB
  je eval_tok_done
  cmp eax, CHAR_NL
  je eval_tok_done
  inc r14
  jmp eval_tok_scan
eval_tok_done:
  mov rsi, r14
  sub rsi, rbx
  movzx edx, byte ptr [rbx]
  cmp edx, CHAR_PLUS
  je eval_plus_tok
  cmp edx, CHAR_MINUS
  je eval_minus_tok
  cmp edx, CHAR_MUL
  je eval_mul_tok
  cmp edx, CHAR_DIV
  je eval_div_tok
  cmp edx, CHAR_MOD
  je eval_mod_tok
  cmp edx, CHAR_ZERO
  jb eval_invalid_token
  cmp edx, CHAR_NINE
  ja eval_invalid_token
  jmp eval_digit_literal
eval_plus_tok:
  cmp rsi, 1
  jne eval_invalid_token
  cmp r13, 2
  jb eval_underflow
  lea rax, [rip + val_stack]
  mov rcx, r13
  dec rcx
  mov rdx, rcx
  shl rdx, 3
  add rax, rdx
  mov r8, [rax]
  mov r9, [rax - 8]
  add r9, r8
  mov [rax - 8], r9
  dec r13
  mov rbx, r14
  jmp eval_scan
eval_sub_tok:
  cmp r13, 2
  jb eval_underflow
  lea rax, [rip + val_stack]
  mov rcx, r13
  dec rcx
  mov rdx, rcx
  shl rdx, 3
  add rax, rdx
  mov r8, [rax]
  mov r9, [rax - 8]
  sub r9, r8
  mov [rax - 8], r9
  dec r13
  mov rbx, r14
  jmp eval_scan
eval_mul_tok:
  cmp rsi, 1
  jne eval_invalid_token
  cmp r13, 2
  jb eval_underflow
  lea rax, [rip + val_stack]
  mov rcx, r13
  dec rcx
  mov rdx, rcx
  shl rdx, 3
  add rax, rdx
  mov r8, [rax]
  mov r9, [rax - 8]
  imul r9, r8
  mov [rax - 8], r9
  dec r13
  mov rbx, r14
  jmp eval_scan
eval_div_tok:
  cmp rsi, 1
  jne eval_invalid_token
  cmp r13, 2
  jb eval_underflow
  lea rax, [rip + val_stack]
  mov rcx, r13
  dec rcx
  mov rdx, rcx
  shl rdx, 3
  add rax, rdx
  mov r10, rax
  sub r10, 8
  mov r8, [rax]
  mov r9, [rax - 8]
  cmp r8, 0
  je eval_invalid_token
  mov rax, [rip + llong_min_val]
  cmp r9, rax
  jne eval_div_ok
  cmp r8, -1
  je eval_invalid_token
eval_div_ok:
  mov rax, r9
  cqo
  mov rcx, r8
  idiv rcx
  mov [r10], rax
  dec r13
  mov rbx, r14
  jmp eval_scan
eval_mod_tok:
  cmp rsi, 1
  jne eval_invalid_token
  cmp r13, 2
  jb eval_underflow
  lea rax, [rip + val_stack]
  mov rcx, r13
  dec rcx
  mov rdx, rcx
  shl rdx, 3
  add rax, rdx
  mov r10, rax
  sub r10, 8
  mov r8, [rax]
  mov r9, [rax - 8]
  cmp r8, 0
  je eval_invalid_token
  mov rax, [rip + llong_min_val]
  cmp r9, rax
  jne eval_mod_ok
  cmp r8, -1
  je eval_invalid_token
eval_mod_ok:
  mov rax, r9
  cqo
  mov rcx, r8
  idiv rcx
  mov [r10], rdx
  dec r13
  mov rbx, r14
  jmp eval_scan
eval_minus_tok:
  cmp rsi, 1
  je eval_sub_tok
  cmp rsi, 19
  ja eval_invalid_token
  mov r15d, 1
  mov r8, rbx
  inc r8
  jmp eval_parse_num
eval_digit_literal:
  cmp rsi, 18
  ja eval_invalid_token
  mov r15d, 0
  mov r8, rbx
eval_parse_num:
  xor eax, eax
eval_num_loop:
  cmp r8, r14
  jae eval_num_done
  movzx edx, byte ptr [r8]
  sub edx, CHAR_ZERO
  cmp edx, 9
  ja eval_invalid_token
  imul rax, rax, 10
  add rax, rdx
  inc r8
  jmp eval_num_loop
eval_num_done:
  test r15d, r15d
  je eval_no_neg
  neg rax
eval_no_neg:
  cmp r13, STACK_MAX
  jae eval_underflow
  lea rcx, [rip + val_stack]
  mov rdx, r13
  shl rdx, 3
  mov [rcx + rdx], rax
  inc r13
  mov rbx, r14
  jmp eval_scan
eval_underflow:
eval_invalid_token:
  mov rbx, r14
  mov ebp, 1
  jmp eval_skip_to_nl
eval_finish:
  cmp ebp, 0
  jne eval_out_error
  cmp r13, 1
  jne eval_out_error
  lea rdi, [rip + val_stack]
  mov rdi, [rdi]
  call print_value
  jmp eval_after_output
eval_out_error:
  lea rdi, [rip + error_msg]
  mov esi, ERROR_LEN
  call out_bytes
eval_after_output:
  cmp rbx, r12
  jae eval_ret
  cmp byte ptr [rbx], CHAR_NL
  jne eval_ret
  inc rbx
eval_ret:
  mov rax, rbx
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

print_value:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 64
  mov rbx, rdi
  test rbx, rbx
  jns pv_pos
  lea rdi, [rip + minus_msg]
  mov esi, 1
  call out_bytes
  neg rbx
pv_pos:
  test rbx, rbx
  jne pv_gen
  lea rdi, [rip + zero_msg]
  mov esi, 1
  call out_bytes
  jmp pv_nl
pv_gen:
  mov r14, rsp
  add r14, 64
pv_loop:
  mov rax, rbx
  xor edx, edx
  mov rcx, 10
  div rcx
  add edx, CHAR_ZERO
  dec r14
  mov byte ptr [r14], dl
  mov rbx, rax
  test rbx, rbx
  jnz pv_loop
  mov rdi, r14
  mov rsi, rsp
  add rsi, 64
  sub rsi, r14
  call out_bytes
pv_nl:
  lea rdi, [rip + newline_msg]
  mov esi, 1
  call out_bytes
  add rsp, 64
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

out_bytes:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  test rsi, rsi
  je out_bytes_done
  mov rbp, rdi
  mov r12, rsi
  cmp r12, OUT_BUF_SIZE
  jbe out_append
  mov rbx, qword ptr [rip + out_len]
  test rbx, rbx
  je out_direct
  mov rdi, STDOUT
  lea rsi, [rip + out_buf]
  mov rdx, rbx
  call write_all
  mov qword ptr [rip + out_len], 0
out_direct:
  mov rdi, STDOUT
  mov rsi, rbp
  mov rdx, r12
  call write_all
  jmp out_bytes_done
out_append:
  mov rbx, qword ptr [rip + out_len]
  mov r13, OUT_BUF_SIZE
  sub r13, rbx
  cmp r13, r12
  jae out_copy
  mov rdi, STDOUT
  lea rsi, [rip + out_buf]
  mov rdx, rbx
  call write_all
  mov qword ptr [rip + out_len], 0
  xor rbx, rbx
out_copy:
  lea rdi, [rip + out_buf]
  add rdi, rbx
  mov rsi, rbp
  mov rcx, r12
  rep movsb
  add rbx, r12
  mov qword ptr [rip + out_len], rbx
out_bytes_done:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

write_all:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, rdx
  mov r12, rsi
  mov r13, rdi
  test rbx, rbx
  je write_all_done
write_all_loop:
  mov eax, SYS_WRITE
  mov rdi, r13
  mov rsi, r12
  mov rdx, rbx
  syscall
  test rax, rax
  jle write_all_done
  cmp rax, rbx
  jbe write_all_clamped
  mov rax, rbx
write_all_clamped:
  add r12, rax
  sub rbx, rax
  test rbx, rbx
  jne write_all_loop
write_all_done:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

flush_output:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, qword ptr [rip + out_len]
  test rbx, rbx
  je flush_done
  mov rdi, STDOUT
  lea rsi, [rip + out_buf]
  mov rdx, rbx
  call write_all
  mov qword ptr [rip + out_len], 0
flush_done:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
