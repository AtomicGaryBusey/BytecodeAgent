

.globl _start

.section .rodata
hexdigits: .ascii "0123456789abcdef"

.bss
.align 16
inbuf: .zero 65536
.align 16
outbuf: .zero 327680

.text
.align 16
_start:
    lea r12, [rip + inbuf]
    lea r13, [rip + outbuf]
    lea r8, [rip + hexdigits]
    xor ebx, ebx

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [r12 + rbx*1]
    mov edx, 65536
    sub rdx, rbx
    jle read_done
    syscall
    cmp rax, 0
    je read_done
    jl read_error
    add rbx, rax
    jmp read_loop

read_error:
    cmp rax, -4
    je read_loop
    jmp read_done

read_done:
    mov rbp, r13
    xor r14d, r14d

row_loop:
    cmp r14, rbx
    jae final_offset

    mov rsi, r12
    add rsi, r14

    mov r15d, 16
    mov eax, ebx
    sub eax, r14d
    cmp eax, 16
    jge full_n
    mov r15d, eax
full_n:
    mov ecx, 28
offset_loop:
    mov eax, r14d
    shr eax, cl
    and eax, 15
    movzx eax, byte ptr [r8 + rax*1]
    mov byte ptr [r13], al
    inc r13
    sub ecx, 4
    cmp ecx, 0
    jge offset_loop

    mov byte ptr [r13], 0x20
    inc r13
    mov byte ptr [r13], 0x20
    inc r13

    xor ecx, ecx
slot_loop:
    cmp ecx, 16
    je slot_done

    cmp ecx, r15d
    jae slot_spaces

    movzx eax, byte ptr [rsi + rcx*1]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx*1]
    mov byte ptr [r13], dl
    inc r13

    and eax, 15
    movzx eax, byte ptr [r8 + rax*1]
    mov byte ptr [r13], al
    inc r13

    mov byte ptr [r13], 0x20
    inc r13
    jmp slot_next

slot_spaces:
    mov byte ptr [r13], 0x20
    inc r13
    mov byte ptr [r13], 0x20
    inc r13
    mov byte ptr [r13], 0x20
    inc r13

slot_next:
    cmp ecx, 7
    jne slot_no_extra
    mov byte ptr [r13], 0x20
    inc r13
slot_no_extra:
    inc ecx
    jmp slot_loop

slot_done:
    mov byte ptr [r13], 0x20
    inc r13
    mov byte ptr [r13], 0x7c
    inc r13

    xor ecx, ecx
char_loop:
    cmp ecx, r15d
    jge char_done

    movzx eax, byte ptr [rsi + rcx*1]
    cmp eax, 0x20
    jb char_dot
    cmp eax, 0x7e
    ja char_dot

    mov byte ptr [r13], al
    inc r13
    jmp char_next

char_dot:
    mov byte ptr [r13], 0x2e
    inc r13

char_next:
    inc ecx
    jmp char_loop

char_done:
    mov byte ptr [r13], 0x7c
    inc r13
    mov byte ptr [r13], 0x0a
    inc r13

    add r14, 16
    jmp row_loop

final_offset:
    mov rdx, rbx
    mov ecx, 28
final_hex_loop:
    mov eax, edx
    shr eax, cl
    and eax, 15
    movzx eax, byte ptr [r8 + rax*1]
    mov byte ptr [r13], al
    inc r13
    sub ecx, 4
    cmp ecx, 0
    jge final_hex_loop

    mov byte ptr [r13], 0x0a
    inc r13

    mov rsi, rbp
    mov rdx, r13
    sub rdx, rbp
    mov edi, 1

write_loop:
    test rdx, rdx
    jle exit_success
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    je exit_success
    jl write_error
    add rsi, rax
    sub rdx, rax
    jmp write_loop

write_error:
    cmp rax, -4
    je write_loop
    jmp exit_success

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_success
