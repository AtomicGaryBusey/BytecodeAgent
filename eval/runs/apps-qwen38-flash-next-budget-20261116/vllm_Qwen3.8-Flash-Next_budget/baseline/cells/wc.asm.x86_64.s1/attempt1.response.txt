

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 65536
.equ TEMP_SIZE, 32

.bss
.align 16
inbuf: .zero BUF_SIZE
outbuf: .zero 64
tempbuf: .zero TEMP_SIZE

.text
.globl _start
_start:
    xor rbx, rbx
    xor r12, r12
    xor r13, r13
    xor r14, r14

read_loop:
    mov eax, SYS_READ
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jle done_output
    add r13, rax
    lea rdi, [rip + inbuf]
    mov r10, rdi
    add r10, rax

proc_loop:
    cmp rdi, r10
    jae read_loop
    movzx eax, byte ptr [rdi]
    cmp al, 10
    jne not_nl
    inc rbx
not_nl:
    cmp al, 32
    je is_ws
    sub eax, 9
    cmp eax, 4
    jbe is_ws
    test r14, r14
    jnz proc_next
    inc r12
    mov r14, 1
proc_next:
    inc rdi
    jmp proc_loop
is_ws:
    xor r14, r14
    inc rdi
    jmp proc_loop

done_output:
    lea rbp, [rip + outbuf]

    mov rdi, rbx
    call print_num
    mov byte ptr [rbp], 32
    inc rbp

    mov rdi, r12
    call print_num
    mov byte ptr [rbp], 32
    inc rbp

    mov rdi, r13
    call print_num
    mov byte ptr [rbp], 10
    inc rbp

    lea rsi, [rip + outbuf]
    mov rdx, rbp
    sub rdx, rsi

write_loop:
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_loop

exit_program:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

print_num:
    test rdi, rdi
    jne print_num_conv
    mov byte ptr [rbp], 48
    inc rbp
    ret

print_num_conv:
    mov r8, rdi
    lea r10, [rip + tempbuf]
    add r10, TEMP_SIZE
    mov rdi, r10
    mov r9, 10

print_num_loop:
    mov rax, r8
    xor edx, edx
    div r9
    mov r8, rax
    add edx, 48
    dec rdi
    mov byte ptr [rdi], dl
    test r8, r8
    jnz print_num_loop
    mov rax, rbp

print_num_copy:
    movzx ecx, byte ptr [rdi]
    mov byte ptr [rax], cl
    inc rdi
    inc rax
    cmp rdi, r10
    jb print_num_copy
    mov rbp, rax
    ret
