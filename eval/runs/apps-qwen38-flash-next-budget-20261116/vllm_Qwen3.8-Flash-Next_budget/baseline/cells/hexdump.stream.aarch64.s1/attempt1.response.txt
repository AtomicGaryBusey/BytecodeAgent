

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536
.equ IN_BUF_SIZE, 65537
.equ OUT_SIZE, 327680
.equ ROW_BYTES, 16
.equ SPACE, 0x20
.equ PIPE, 0x7c
.equ DOT, 0x2e
.equ NEWLINE, 0x0a
.equ PRINT_LO, 0x20
.equ PRINT_HI, 0x7e

.text
.globl _start
_start:
    movz x27, #1, lsl #16
    add x27, x27, #1
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    adrp x21, out_buf
    add x21, x21, :lo12:out_buf
    mov x20, x21
    mov x22, #0
read_loop:
    sub x2, x27, x22
    mov x1, x19
    add x1, x1, x22
    mov x0, #STDIN
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x22, x22, x0
    cmp x22, x27
    b.lt read_loop
    mov x22, x27
read_done:
    sub x4, x27, #1
    cmp x22, x4
    b.le no_clamp
    mov x22, x4
no_clamp:
    adrp x26, hexdigits
    add x26, x26, :lo12:hexdigits
    mov x23, #0
row_loop:
    cmp x23, x22
    b.ge final
    sub x4, x22, x23
    cmp x4, #ROW_BYTES
    b.ge set16
    mov x24, x4
    b len_done
set16:
    mov x24, #ROW_BYTES
len_done:
    add x25, x19, x23

    mov x1, x23
    mov x17, #28
1:
    lsr x0, x1, x17
    and x0, x0, #15
    ldrb w3, [x26, x0]
    strb w3, [x20], #1
    subs x17, x17, #4
    b.ge 1b

    mov w1, #SPACE
    strb w1, [x20], #1
    strb w1, [x20], #1

    mov x16, #0
byte_loop:
    cmp x16, x24
    b.ge pad_slot
    ldrb w5, [x25, x16]
    ubfx w4, w5, #4, #4
    and w5, w5, #15
    ldrb w1, [x26, x4]
    strb w1, [x20], #1
    ldrb w1, [x26, x5]
    strb w1, [x20], #1
    mov w1, #SPACE
    strb w1, [x20], #1
    b slot_done
pad_slot:
    mov w1, #SPACE
    strb w1, [x20], #1
    strb w1, [x20], #1
    strb w1, [x20], #1
slot_done:
    cmp x16, #7
    b.ne next_byte
    mov w1, #SPACE
    strb w1, [x20], #1
next_byte:
    add x16, x16, #1
    cmp x16, #ROW_BYTES
    b.lt byte_loop

    mov w1, #SPACE
    strb w1, [x20], #1
    mov w1, #PIPE
    strb w1, [x20], #1

    mov x16, #0
char_loop:
    ldrb w5, [x25, x16]
    cmp w5, #PRINT_LO
    b.lt dot
    cmp w5, #PRINT_HI
    b.gt dot
    strb w5, [x20], #1
    b char_next
dot:
    mov w1, #DOT
    strb w1, [x20], #1
char_next:
    add x16, x16, #1
    cmp x16, x24
    b.lt char_loop

    mov w1, #PIPE
    strb w1, [x20], #1
    mov w1, #NEWLINE
    strb w1, [x20], #1

    add x23, x23, x24
    b row_loop

final:
    mov x1, x22
    mov x17, #28
1:
    lsr x0, x1, x17
    and x0, x0, #15
    ldrb w3, [x26, x0]
    strb w3, [x20], #1
    subs x17, x17, #4
    b.ge 1b

    mov w1, #NEWLINE
    strb w1, [x20], #1

write_loop:
    mov x0, #STDOUT
    mov x1, x21
    sub x2, x20, x21
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_done
    add x21, x21, x0
    cmp x21, x20
    b.lt write_loop
write_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
.p2align 4
hexdigits:
    .ascii "0123456789abcdef"

.bss
.p2align 4
in_buf:
    .space IN_BUF_SIZE
.p2align 4
out_buf:
    .space OUT_SIZE
