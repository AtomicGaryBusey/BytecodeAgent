

.equ INPUT_MAX, 32768
.equ LINE_MAX, 1000
.equ LINE_BYTES, 16000
.equ OUT_MAX, 65536

.text
.globl _start
_start:
    xor r15d, r15d
read_loop:
    cmp r15, INPUT_MAX
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov edx, INPUT_MAX
    sub edx, r15d
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    mov rdi, rsi
    add rdi, r15
    lea rdx, [rip + line_array]
    xor r14d, r14d
parse_loop:
    cmp rsi, rdi
    jae parse_set_count
    mov rcx, rsi
parse_scan:
    cmp rcx, rdi
    jae parse_final
    cmp byte ptr [rcx], 10
    je parse_newline
    inc rcx
    jmp parse_scan
parse_newline:
    mov r8, rcx
    sub r8, rsi
    cmp r14, LINE_MAX
    jae parse_skip
    mov qword ptr [rdx], rsi
    mov qword ptr [rdx + 8], r8
    add rdx, 16
    inc r14
parse_skip:
    lea rsi, [rcx + 1]
    jmp parse_loop
parse_final:
    cmp rsi, rdi
    jae parse_set_count
    mov r8, rdi
    sub r8, rsi
    cmp r14, LINE_MAX
    jae parse_set_count
    mov qword ptr [rdx], rsi
    mov qword ptr [rdx + 8], r8
    add rdx, 16
    inc r14
parse_set_count:
    test r14, r14
    jz exit_ok

    lea rbx, [rip + line_array]
    mov r15, 1
sort_outer:
    cmp r15, r14
    jae sort_done
    mov rdi, r15
    shl rdi, 4
    add rdi, rbx
    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]
    mov rbp, r15
    dec rbp
sort_inner:
    test rbp, rbp
    js sort_place_inc
    mov rdi, rbp
    shl rdi, 4
    add rdi, rbx
    mov rsi, qword ptr [rdi + 8]
    mov rdx, qword ptr [rdi]
    mov r12, r11
    call cmp_lines
    cmp eax, 1
    jne sort_place_inc
    mov r9, rbp
    inc r9
    mov rdi, r9
    shl rdi, 4
    add rdi, rbx
    mov rax, qword ptr [rdi - 16]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rdi - 8]
    mov qword ptr [rdi + 8], rax
    dec rbp
    jmp sort_inner
sort_place_inc:
    inc rbp
sort_place:
    mov rdi, rbp
    shl rdi, 4
    add rdi, rbx
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    inc r15
    jmp sort_outer
sort_done:

    lea rdi, [rip + out_buf]
    mov rsi, rbx
    mov rcx, r14
copy_loop:
    test rcx, rcx
    jz out_done
    mov rdx, qword ptr [rsi]
    mov r8, qword ptr [rsi + 8]
    mov r9, r8
copy_bytes:
    test r9, r9
    jz copy_nl
    movzx eax, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdx
    inc rdi
    dec r9
    jnz copy_bytes
copy_nl:
    mov byte ptr [rdi], 10
    inc rdi
    add rsi, 16
    dec rcx
    jmp copy_loop
out_done:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
    mov eax, 231
    syscall
    jmp exit_ok

cmp_lines:
    mov r8, rsi
    cmp r8, r12
    cmovae r8, r12
    test r8, r8
    je cmp_done_len
cmp_chunk:
    cmp r8, 8
    jb cmp_tail
    mov rax, qword ptr [rdi]
    mov rcx, qword ptr [rdx]
    xor rax, rcx
    jnz cmp_diff
    add rdi, 8
    add rdx, 8
    sub r8, 8
    jmp cmp_chunk
cmp_tail:
    test r8, r8
    jz cmp_done_len
cmp_byte_loop:
    movzx eax, byte ptr [rdi]
    movzx ecx, byte ptr [rdx]
    cmp eax, ecx
    jb cmp_less
    ja cmp_greater
    inc rdi
    inc rdx
    dec r8
    jnz cmp_byte_loop
cmp_done_len:
    cmp rsi, r12
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_diff:
    tzcnt r9, rax
    shr r9, 3
    add rdi, r9
    add rdx, r9
    movzx eax, byte ptr [rdi]
    movzx ecx, byte ptr [rdx]
    cmp eax, ecx
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret

.bss
.align 16
input_buf:
    .zero INPUT_MAX
line_array:
    .zero LINE_BYTES
out_buf:
    .zero OUT_MAX
