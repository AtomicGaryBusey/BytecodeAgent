

.equ INPUT_MAX, 32768
.equ OUT_MAX, 65536

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, INPUT_MAX
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, INPUT_MAX
    jb read_loop
read_done:
    mov qword ptr [input_len], rbx

    mov rbx, qword ptr [input_len]
    lea rcx, [rip + input_buf]
    mov r10, rcx
    add rcx, rbx
    lea rdi, [rip + line_array]
    mov rsi, r10
    xor r8d, r8d
parse_loop:
    cmp rbx, rcx
    je parse_final
    cmp byte ptr [rbx], 10
    jne parse_not_nl
    mov r9, rbx
    sub r9, rsi
    mov qword ptr [rdi], rsi
    mov qword ptr [rdi + 8], r9
    add rdi, 16
    inc r8
    lea rsi, [rbx + 1]
    inc rbx
    jmp parse_loop
parse_not_nl:
    inc rbx
    jmp parse_loop
parse_final:
    cmp rsi, rcx
    jae parse_set_count
    mov r9, rcx
    sub r9, rsi
    mov qword ptr [rdi], rsi
    mov qword ptr [rdi + 8], r9
    inc r8
parse_set_count:
    mov qword ptr [line_count], r8
    test r8, r8
    jz exit_ok

    lea rbx, [rip + line_array]
    mov r15, 1
sort_outer:
    cmp r15, qword ptr [line_count]
    jae sort_done
    lea rsi, [rbx + r15*16]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov rbp, r15
    dec rbp
sort_inner:
    lea rdi, [rbx + rbp*16]
    mov rsi, qword ptr [rdi + 8]
    mov rdx, r10
    mov r12, r11
    mov rdi, qword ptr [rdi]
    call cmp_lines
    cmp eax, 1
    jne sort_place
    lea r9, [rbx + rbp*16]
    mov rax, qword ptr [r9]
    mov qword ptr [r9 + 16], rax
    mov rax, qword ptr [r9 + 8]
    mov qword ptr [r9 + 24], rax
    dec rbp
    test rbp, rbp
    jns sort_inner
sort_place:
    lea r9, [rbx + rbp*16 + 16]
    mov qword ptr [r9], r10
    mov qword ptr [r9 + 8], r11
    inc r15
    jmp sort_outer
sort_done:

    mov r10, qword ptr [line_count]
    lea rbp, [rip + out_buf]
    xor edx, edx
out_loop:
    cmp rdx, r10
    jae out_done
    lea r9, [rbx + rdx*16]
    mov rsi, qword ptr [r9]
    mov rcx, qword ptr [r9 + 8]
    mov rdi, rbp
out_copy:
    test rcx, rcx
    jz out_copied
out_copy_byte:
    movzx eax, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz out_copy_byte
out_copied:
    mov rbp, rdi
    mov byte ptr [rbp], 10
    inc rbp
    inc edx
    jmp out_loop
out_done:
    lea r8, [rip + out_buf]
    mov rdx, rbp
    sub rdx, r8
    lea rsi, [rip + out_buf]
write_loop:
    test rdx, rdx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
    mov eax, 231
    syscall
    jmp exit_ok

cmp_lines:
    mov r8, rsi
    cmp r8, r12
    jbe 1f
    mov r8, r12
1:
    test r8, r8
    jz cmp_done
cmp_chunk:
    cmp r8, 8
    jb cmp_byte
    mov rax, qword ptr [rdi]
    mov r9, qword ptr [rdx]
    xor rax, r9
    jz cmp_next
    tzcnt r13, rax
    shr r13, 3
    movzx eax, byte ptr [rdi + r13]
    movzx r9d, byte ptr [rdx + r13]
    cmp eax, r9d
    jb cmp_less
    ja cmp_greater
    jmp cmp_byte
cmp_next:
    add rdi, 8
    add rdx, 8
    sub r8, 8
    jmp cmp_chunk
cmp_byte:
    test r8, r8
    jz cmp_done
cmp_byte_loop:
    movzx eax, byte ptr [rdi]
    movzx r9d, byte ptr [rdx]
    cmp eax, r9d
    jb cmp_less
    ja cmp_greater
    inc rdi
    inc rdx
    dec r8
    jnz cmp_byte_loop
cmp_done:
    mov r8, rsi
    cmp r8, r12
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret

.bss
.align 16
input_buf:
    .zero INPUT_MAX
line_array:
    .zero 16000
out_buf:
    .zero OUT_MAX
input_len:
    .zero 8
line_count:
    .zero 8
