export function l $crc32(l %p, l %n) {
@entry
    %i =l const 0
    %crc =w const 0xFFFFFFFF
    %one =l const 1
    %eight =l const 8
    %poly =w const 0xEDB88320
    %n64 =l %n
    jmp @outer_check

@outer_check
    %c1 =l csltl %i, %n64
    jnz %c1, @outer_body
    jmp @final

@outer_body
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %j =l const 0
    jmp @inner_check

@inner_check
    %c2 =l csltl %j, %eight
    jnz %c2, @inner_body
    jmp @outer_next

@inner_body
    %lsb =w and %crc, 1
    %c3 =w ne %lsb, 0
    jnz %c3, @inner_if
    %crc =w shr %crc, 1
    jmp @inner_inc

@inner_if
    %crc =w shr %crc, 1
    %crc =w xor %crc, %poly
    jmp @inner_inc

@inner_inc
    %j =l add %j, %one
    jmp @inner_check

@outer_next
    %i =l add %i, %one
    jmp @outer_check

@final
    %crc =w xor %crc, 0xFFFFFFFF
    %result =l zextw %crc
    ret %result
}
