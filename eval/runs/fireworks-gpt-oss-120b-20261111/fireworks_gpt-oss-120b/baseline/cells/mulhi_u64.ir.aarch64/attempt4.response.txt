export function l $mulhi_u64(l a, l b)
@start
%t0 = and %a, 0xffffffff
%t1 = shr %a, 32
%t2 = and %b, 0xffffffff
%t3 = shr %b, 32
%p0 = mul %t0, %t2
%p1 = mul %t0, %t3
%p2 = mul %t1, %t2
%p3 = mul %t1, %t3
%p0_hi = shr %p0, 32
%p1_lo = and %p1, 0xffffffff
%p2_lo = and %p2, 0xffffffff
%c0 = add %p0_hi, %p1_lo
%c1 = add %c0, %p2_lo
%c_hi = shr %c1, 32
%p1_hi = shr %p1, 32
%p2_hi = shr %p2, 32
%h0 = add %p3, %p1_hi
%h1 = add %h0, %p2_hi
%high = add %h1, %c_hi
ret %high
