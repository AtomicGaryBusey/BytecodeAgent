export function l $mulhi_u64(l a, l b)
@start
%a0 = and a, 0xffffffff
%a1 = shr a, 32
%b0 = and b, 0xffffffff
%b1 = shr b, 32
%p0 = mul %a0, %b0
%p1 = mul %a0, %b1
%p2 = mul %a1, %b0
%p3 = mul %a1, %b1
%p0_hi = shr %p0, 32
%p1_lo = and %p1, 0xffffffff
%p2_lo = and %p2, 0xffffffff
%carry0 = add %p0_hi, %p1_lo
%carry = add %carry0, %p2_lo
%carry_hi = shr %carry, 32
%p1_hi = shr %p1, 32
%p2_hi = shr %p2, 32
%high0 = add %p3, %p1_hi
%high1 = add %high0, %p2_hi
%high = add %high1, %carry_hi
ret %high
