export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %i =l 0
    %o =l 0
    jmp @L_main_check

@L_main_check:
    %cond_i_lt_n = csltl %i, %n
    jnz %cond_i_lt_n, @L_main_body
    %ret =l %o
    ret %ret

@L_main_body:
    %src_i_addr = addl %src, %i
    %first =w loadub %src_i_addr
    %run =l 0
    jmp @L_run_check

@L_run_check:
    %run_eq_128 = ceql %run, 128
    jnz %run_eq_128, @L_run_end
    %idx = addl %i, %run
    %idx_lt_n = csltl %idx, %n
    jnz %idx_lt_n, @L_run_continue
    jmp @L_run_end

@L_run_continue:
    %addr = addl %src, %idx
    %b =w loadub %addr
    %eq = ceql %b, %first
    jnz %eq, @L_run_inc
    jmp @L_run_end

@L_run_inc:
    %run = addl %run, 1
    jmp @L_run_check

@L_run_end:
    %run_lt_3 = csltl %run, 3
    jnz %run_lt_3, @L_literal_start
    %need = addl %o, 2
    %diff = subl %need, %cap
    %ok = csltl %diff, 1
    jnz %ok, @L_repeat_store
    %ret =l -1
    ret %ret

@L_repeat_store:
    %header = subl 257, %run
    %tmp_out = addl %out, %o
    storeb %header, %tmp_out
    %out_val = addl %tmp_out, 1
    %src_val = addl %src, %i
    %val =w loadub %src_val
    storeb %val, %out_val
    %i = addl %i, %run
    %o = addl %o, 2
    jmp @L_main_check

@L_literal_start:
    %j =l %i
    %len =l 0
    jmp @L_lit_check

@L_lit_check:
    %j_lt_n = csltl %j, %n
    jnz %j_lt_n, @L_lit_len_check
    jmp @L_lit_end

@L_lit_len_check:
    %len_lt_128 = csltl %len, 128
    jnz %len_lt_128, @L_lit_run_check
    jmp @L_lit_end

@L_lit_run_check:
    %j_plus2 = addl %j, 2
    %has_three = csltl %j_plus2, %n
    jnz %has_three, @L_lit_three_check
    jmp @L_lit_extend

@L_lit_three_check:
    %addr0 = addl %src, %j
    %b0 =w loadub %addr0
    %tmp1 = addl %src, %j
    %addr1 = addl %tmp1, 1
    %b1 =w loadub %addr1
    %addr2 = addl %tmp1, 2
    %b2 =w loadub %addr2
    %eq01 = ceql %b0, %b1
    %eq02 = ceql %b0, %b2
    %run3 = andl %eq01, %eq02
    jnz %run3, @L_lit_end
    jmp @L_lit_extend

@L_lit_extend:
    %j = addl %j, 1
    %len = addl %len, 1
    jmp @L_lit_check

@L_lit_end:
    %len_plus1 = addl %len, 1
    %need = addl %o, %len_plus1
    %diff = subl %need, %cap
    %ok = csltl %diff, 1
    jnz %ok, @L_literal_store
    %ret =l -1
    ret %ret

@L_literal_store:
    %header = subl %len, 1
    %tmp_out = addl %out, %o
    storeb %header, %tmp_out
    %k =l 0
    jmp @L_copy_check

@L_copy_check:
    %k_lt_len = csltl %k, %len
    jnz %k_lt_len, @L_copy_body
    jmp @L_copy_done

@L_copy_body:
    %src_base = addl %src, %i
    %src_addr = addl %src_base, %k
    %b =w loadub %src_addr
    %dst_base = addl %out, %o
    %dst_base1 = addl %dst_base, 1
    %dst_addr = addl %dst_base1, %k
    storeb %b, %dst_addr
    %k = addl %k, 1
    jmp @L_copy_check

@L_copy_done:
    %i = addl %i, %len
    %o = addl %o, %len_plus1
    jmp @L_main_check
}
