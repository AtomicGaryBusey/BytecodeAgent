export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry:
    %i =l 0
    %o =l 0
    jmp @check_main

@check_main:
    %cond_i_lt_n = csltl %i, %n
    jnz %cond_i_lt_n, @main_body
    ret %o

@main_body:
    %src_i_addr = addl %src, %i
    %first =w loadub %src_i_addr
    %run =l 0
    jmp @run_check

@run_check:
    %run_eq_128 = ceql %run, 128
    jnz %run_eq_128, @run_end
    %idx = addl %i, %run
    %idx_lt_n = csltl %idx, %n
    jnz %idx_lt_n, @run_continue
    jmp @run_end

@run_continue:
    %addr = addl %src, %idx
    %b =w loadub %addr
    %eq = ceql %b, %first
    jnz %eq, @run_inc
    jmp @run_end

@run_inc:
    %run = addl %run, 1
    jjmp @run_check

@run_end:
    %run_lt_3 = csltl %run, 3
    jnz %run_lt_3, @literal_start
    %need = addl %o, 2
    %diff = subl %need, %cap
    %ok = csltl %diff, 1
    jnz %ok, @repeat_store
    %retneg =l -1
    ret %retneg

@repeat_store:
    %header = subl 257, %run
    %out_ptr = addl %out, %o
    storeb %header, %out_ptr
    %out_val = addl %out_ptr, 1
    %src_val = addl %src, %i
    %val =w loadub %src_val
    storeb %val, %out_val
    %i = addl %i, %run
    %o = addl %o, 2
    jmp @check_main

@literal_start:
    %j =l %i
    %len =l 0
    jmp @literal_check

@literal_check:
    %j_lt_n = csltl %j, %n
    jnz %j_lt_n, @literal_len_check
    jmp @literal_end

@literal_len_check:
    %len_lt_128 = csltl %len, 128
    jnz %len_lt_128, @literal_run_check
    jmp @literal_end

@literal_run_check:
    %j_plus2 = addl %j, 2
    %has_three = csltl %j_plus2, %n
    jnz %has_three, @literal_three_check
    jmp @literal_extend

@literal_three_check:
    %src_j = addl %src, %j
    %b0 =w loadub %src_j
    %src_j1 = addl %src_j, 1
    %b1 =w loadub %src_j1
    %src_j2 = addl %src_j, 2
    %b2 =w loadub %src_j2
    %eq01 = ceql %b0, %b1
    %eq02 = ceql %b0, %b2
    %run3 = andl %eq01, %eq02
    jnz %run3, @literal_end
    jmp @literal_extend

@literal_extend:
    %j = addl %j, 1
    %len = addl %len, 1
    jmp @literal_check

@literal_end:
    %len_plus1 = addl %len, 1
    %need = addl %o, %len_plus1
    %diff = subl %need, %cap
    %ok = csltl %diff, 1
    jnz %ok, @literal_store
    %retneg =l -1
    ret %retneg

@literal_store:
    %header = subl %len, 1
    %out_ptr = addl %out, %o
    storeb %header, %out_ptr
    %k =l 0
    jmp @copy_check

@copy_check:
    %k_lt_len = csltl %k, %len
    jnz %k_lt_len, @copy_body
    jmp @copy_done

@copy_body:
    %src_base = addl %src, %i
    %src_addr = addl %src_base, %k
    %b =w loadub %src_addr
    %out_base = addl %out, %o
    %out_base1 = addl %out_base, 1
    %dst_addr = addl %out_base1, %k
    storeb %b, %dst_addr
    %k = addl %k, 1
    jmp @copy_check

@copy_done:
    %i = addl %i, %len
    %o = addl %o, %len_plus1
    jmp @check_main
}
