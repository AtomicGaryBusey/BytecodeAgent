map_sum:
    push rbx
    push r12
    push r13
    push r14
    sub rsp, 8
    mov rbx, rdi
    mov r12, rsi
    mov r14, rdx
    xor r13, r13
.Lloop:
    test r12, r12
    jz .Ldone
    mov rdi, qword ptr [rbx]
    call r14
    add r13, rax
    add rbx, 8
    dec r12
    jmp .Lloop
.Ldone:
    mov rax, r13
    add rsp, 8
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
