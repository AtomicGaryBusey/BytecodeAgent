export function l $bmin_empty(l a, l n) {
@entry:
    w %is_zero = ceql n, 0
    jnz %is_zero, @return_max

    l %min = loadl a
    l %i = constl 1
    jmp @loop

@loop:
    w %cond = csltl %i, n
    jnz %cond, @body
    jmp @done

@body:
    l %idx8 = mull %i, constl 8
    l %ptr = addl a, %idx8
    l %val = loadl %ptr
    w %less = csltl %val, %min
    jnz %less, @update_min
    jmp @inc_i

@update_min:
    %min = %val
    jmp @inc_i

@inc_i:
    l %i = addl %i, constl 1
    jmp @loop

@done:
    ret %min

@return_max:
    l %max = constl 0x7FFFFFFFFFFFFFFF
    ret %max
}
