export function l $bmin_empty(l a, l n) {
@entry:
    w %is_zero = ceql n, constl 0
    jnz %is_zero, @return_max

    l %min = loadl a
    l %i = constl 1
    jmp @loop

@loop:
    w %cond = csltl %i, n
    jnz %cond, @body
    jmp @done

@body:
    l %idx = mul %i, constl 8
    l %ptr = add a, %idx
    l %val = loadl %ptr
    w %less = csltl %val, %min
    jnz %less, @update
    jmp @inc

@update:
    %min = %val
    jmp @inc

@inc:
    l %i = add %i, constl 1
    jmp @loop

@done:
    ret %min

@return_max:
    l %max = constl 0x7FFFFFFFFFFFFFFF
    ret %max
}
