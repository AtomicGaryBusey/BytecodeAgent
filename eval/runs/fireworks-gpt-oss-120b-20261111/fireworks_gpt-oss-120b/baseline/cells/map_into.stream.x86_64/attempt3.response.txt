map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov r12, rsi                     # src pointer
    mov r13, rdi                     # dst pointer
    mov r14, rdx                     # n
    mov r15, rcx                     # f pointer
    xor rax, rax
    mov qword ptr [rsp], rax         # count = 0
    mov qword ptr [rsp+8], r15       # store f pointer
    xor rbx, rbx                     # i = 0
.Lloop:
    cmp rbx, r14
    jae .Lend
    mov rdx, qword ptr [r12 + rbx*8] # src[i] -> rdx
    mov rdi, rdx                     # argument for f
    mov rax, qword ptr [rsp+8]       # load f pointer
    call rax
    mov qword ptr [r13 + rbx*8], rax # dst[i] = result
    mov rdx, qword ptr [r12 + rbx*8] # reload src[i] for compare
    cmp rax, rdx
    jl .Linc
.Lnext:
    inc rbx
    jmp .Lloop
.Linc:
    inc qword ptr [rsp]              # count++
    jmp .Lnext
.Lend:
    mov rax, qword ptr [rsp]         # return count
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
