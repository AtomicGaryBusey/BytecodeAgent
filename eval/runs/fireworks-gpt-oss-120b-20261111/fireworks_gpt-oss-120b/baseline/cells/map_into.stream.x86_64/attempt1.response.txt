map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rcx
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
    xor rbx, rbx
    xor rax, rax
.Lloop:
    cmp rbx, r15
    jae .Lend
    mov rdx, qword ptr [r14 + rbx*8]
    sub rsp, 8
    push rdx
    mov rdi, rdx
    call r12
    mov qword ptr [r13 + rbx*8], rax
    pop rcx
    add rsp, 8
    cmp rax, rcx
    jl .Linc
    jmp .Lnext
.Linc:
    inc rax
.Lnext:
    inc rbx
    jmp .Lloop
.Lend:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
