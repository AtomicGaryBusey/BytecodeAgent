export function l $count_desc(l a, l n) {
@entry
    %c2 = constl 2
    %cond_len = csltl %n, %c2
    jnz %cond_len, @ret0

    %cnt0 = constl 0
    %i0 = constl 0
    jmp @loop %cnt0 %i0

@loop %cnt %i
    %c1 = constl 1
    %limit = subl %n, %c1
    %cond = csltl %i, %limit
    jz %cond, @ret

    %c8 = constl 8
    %off = mull %i, %c8
    %addr1 = addl %a, %off
    %v1 = loadl %addr1

    %off2 = addl %off, %c8
    %addr2 = addl %a, %off2
    %v2 = loadl %addr2

    %gt = csltl %v2, %v1
    %cnt1 = addl %cnt, %c1
    %cnt2 = sel %gt, %cnt1, %cnt

    %i1 = addl %i, %c1
    jmp @loop %cnt2 %i1

@ret
    ret %cnt

@ret0
    ret 0
}
