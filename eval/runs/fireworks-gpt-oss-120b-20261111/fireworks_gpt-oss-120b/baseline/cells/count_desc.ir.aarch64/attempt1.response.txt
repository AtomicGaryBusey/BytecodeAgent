export function l $count_desc(l a, l n) {
@entry
    %c2 = constl 2
    %cond_n = csltl %n, %c2
    jnz %cond_n, @ret0

    %i = constl 0
    %cnt = constl 0
    jmp @loop

@loop
    %c1 = constl 1
    %limit = subl %n, %c1
    %cond_loop = csltl %i, %limit
    jz %cond_loop, @ret

    %c8 = constl 8
    %addr = mul %i, %c8
    %p = addl %a, %addr
    %v1 = loadl %p

    %addr2 = addl %addr, %c8
    %p2 = addl %a, %addr2
    %v2 = loadl %p2

    %gt = csltl %v2, %v1
    %cnt_inc = addl %cnt, %c1
    %cnt = sel %gt, %cnt_inc, %cnt

    %i = addl %i, %c1
    jmp @loop

@ret
    ret %cnt

@ret0
    ret 0
}
