export function l count_desc(l p, l len) {
@entry
    %c2 = constl 2
    %cond_len = csltl %len, %c2
    jnz %cond_len, @ret0

    %i = constl 0
    %cnt = constl 0
    jmp @loop

@loop
    %c1 = constl 1
    %limit = subl %len, %c1
    %cond_loop = csltl %i, %limit
    jz %cond_loop, @ret

    %c8 = constl 8
    %off = mull %i, %c8
    %addr1 = addl %p, %off
    %v1 = loadl %addr1

    %off2 = addl %off, %c8
    %addr2 = addl %p, %off2
    %v2 = loadl %addr2

    %gt = csltl %v2, %v1
    %cnt_inc = addl %cnt, %c1
    %cnt = sel %gt, %cnt_inc, %cnt

    %i = addl %i, %c1
    jmp @loop

@ret
    ret %cnt

@ret0
    ret 0
}
