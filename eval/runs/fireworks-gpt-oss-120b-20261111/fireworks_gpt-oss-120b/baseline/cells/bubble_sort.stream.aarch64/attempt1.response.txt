bubble_sort:
    cmp x1, #1
    ble .Lreturn
    sub x3, x1, #1
    mov x2, #0
.Louter:
    cmp x2, x3
    bge .Lreturn
    sub x4, x3, x2
    mov x5, #0
.Linner:
    cmp x5, x4
    bge .Louter_next
    add x6, x0, x5, lsl #3
    ldr x7, [x6]
    ldr x8, [x6, #8]
    cmp x7, x8
    ble .Lnoswap
    str x8, [x6]
    str x7, [x6, #8]
.Lnoswap:
    add x5, x5, #1
    b .Linner
.Louter_next:
    add x2, x2, #1
    b .Louter
.Lreturn:
    mov x0, #0
    ret
