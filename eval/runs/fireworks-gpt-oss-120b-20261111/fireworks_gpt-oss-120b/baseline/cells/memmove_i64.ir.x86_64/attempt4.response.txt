export function l $memmove_i64(l %dst, l %src, l %n) {
@entry:
  %zero = const 0
  %one = const 1
  %three = const 3
  %dst_lt_src = csltl %dst %src
  jnz %dst_lt_src @forward_entry
  jmp @backward_entry

@forward_entry:
  %i0 = const 0
  jmp @forward_loop

@forward_loop:
  %i = phi l %i0 @forward_entry, %i_next @forward_body
  %c = csltl %i %n
  jnz %c @forward_body
  jmp @ret

@forward_body:
  %off = shl %i %three
  %src_addr = add %src %off
  %val = loadl %src_addr
  %dst_addr = add %dst %off
  storel %val %dst_addr
  %i_next = add %i %one
  jmp @forward_loop

@backward_entry:
  %i0 = mov %n
  jmp @backward_loop

@backward_loop:
  %i = phi l %i0 @backward_entry, %i_next @backward_body
  %c = csltl %zero %i
  jnz %c @backward_body
  jmp @ret

@backward_body:
  %i1 = sub %i %one
  %off = shl %i1 %three
  %src_addr = add %src %off
  %val = loadl %src_addr
  %dst_addr = add %dst %off
  storel %val %dst_addr
  %i_next = mov %i1
  jmp @backward_loop

@ret:
  ret %zero
}
