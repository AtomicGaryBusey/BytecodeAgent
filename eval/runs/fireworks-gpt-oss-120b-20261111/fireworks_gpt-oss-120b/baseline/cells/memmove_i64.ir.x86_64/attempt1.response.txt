export function l $memmove_i64(l %dst, l %src, l %n) {
    %zero = const 0
    %dir = csltl %dst %src
    jnz %dir, @forward
    jmp @backward

@forward:
    %i = const 0
    jmp @lf_check

@lf_check:
    %c = csltl %i %n
    jnz %c, @lf_body
    jmp @ret

@lf_body:
    %off = shl %i, 3
    %src_addr = add %src, %off
    %val = loadl %src_addr
    %dst_addr = add %dst, %off
    storel %val, %dst_addr
    %i = add %i, 1
    jmp @lf_check

@backward:
    %i = %n
    jmp @lb_check

@lb_check:
    %c = csltl %zero %i
    jnz %c, @lb_body
    jmp @ret

@lb_body:
    %i1 = sub %i, 1
    %off = shl %i1, 3
    %src_addr = add %src, %off
    %val = loadl %src_addr
    %dst_addr = add %dst, %off
    storel %val, %dst_addr
    %i = %i1
    jmp @lb_check

@ret:
    ret %zero
}
