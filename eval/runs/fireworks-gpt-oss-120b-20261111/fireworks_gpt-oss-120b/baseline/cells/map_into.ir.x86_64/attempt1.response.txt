export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    addsp -8
    %cnt =l iconst 0
    %i =l iconst 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @out

@body
    %off =l mul %i, 8
    %src_addr =l add %src, %off
    %val =l loadl %src_addr
    %res =l call %f(l %val)
    %dst_addr =l add %dst, %off
    storel %res, %dst_addr
    %lt =w csltl %res, %val
    %cnt_inc =l add %cnt, 1
    %cnt =l sel %lt, %cnt_inc, %cnt
    %i =l add %i, 1
    jmp @loop

@out
    addsp 8
    ret %cnt
}
