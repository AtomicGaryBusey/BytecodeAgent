export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %i =l iconst 0
    %cnt =l iconst 0
    jmp @loop, %i, %cnt

@loop(l %i, l %cnt)
    %cond =w csltl %i, %n
    jnz %cond, @body, %i, %cnt
    jmp @out, %cnt

@body(l %i, l %cnt)
    %off =l mul %i, 8
    %src_addr =l add %src, %off
    %val =l loadl %src_addr
    %res =l call %f(l %val)
    %dst_addr =l add %dst, %off
    storel %res, %dst_addr
    %lt =w csltl %res, %val
    %cnt_inc =l add %cnt, 1
    %newcnt =l sel %lt, %cnt_inc, %cnt
    %i_next =l add %i, 1
    jmp @loop, %i_next, %newcnt

@out(l %cnt)
    ret %cnt
}
