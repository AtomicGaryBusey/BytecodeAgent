    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x19, x0
    mov x20, x1

.Lloop_start:

    ble .Lreturn
    asr x21, x20, #1
    add x22, x19, x21, LSL #4
    ldr x23, [x22]

    ldr x26, [x19, #8]
    str x23, [x19]
    str x24, [x19, #8]
    str x25, [x22]
    str x26, [x22, #8]
    mov x27, #1
    sub x28, x20, #1

    bgt .Lpart_done
    add x5, x19, x27, LSL #4

    add x6, x19, x27, LSL #4

    cmp x6, x24

    ldr x8, [x8, #8]

    bgt .Ldec_j

    add x10, x19, x28, LSL #4
    str x7, [x9]
    str x8, [x9, #8]
    str x5, [x10]
    str x6, [x10, #8]

    sub x28, x28, #1
    b .Lpart_loop
.Ldec_j:

    b .Lpart_loop
.Lpart_done:

    beq .Lafter_swap
    add x11, x19, x28, LSL #4
    ldr x12, [x11]
    ldr x13, [x11, #8]

    str x24, [x11, #8]

    str x13, [x19, #8]

    cmp x14, #1
    ble .Lskip_left
    stp x19, x20, [sp, #-16]!
    mov x0, x19
    mov x1, x14
    bl qsort_pairs
    ldp x19, x20, [sp], #16
.Lskip_left:

    add x19, x19, x15, LSL #4
    sub x20, x20, x15
    b .Lloop_start
.Lreturn:
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
