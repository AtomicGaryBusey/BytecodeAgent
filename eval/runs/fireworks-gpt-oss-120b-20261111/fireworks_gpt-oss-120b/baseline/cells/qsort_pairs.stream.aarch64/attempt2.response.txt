qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    ble .Lreturn

    // choose pivot = middle element
    asr x21, x20, #1                // idx = n/2
    lsl x21, x21, #4                // idx*16
    add x22, x19, x21               // &pivot
    ldr x21, [x22]                  // pivot.key
    ldr x22, [x22, #8]              // pivot.val

    // swap pivot with first element
    ldr x23, [x19]                  // a[0].key
    ldr x24, [x19, #8]              // a[0].val
    str x21, [x19]
    str x22, [x19, #8]
    str x23, [x22]
    str x24, [x22, #8]

    mov x25, #1                     // i = 1
    sub x26, x20, #1                // j = n-1

.Lpartition_loop:
    cmp x25, x26
    bgt .Lpartition_done

    // load a[i]
    add x27, x19, x25, LSL #4
    ldr x28, [x27]                  // key_i
    ldr x29, [x27, #8]              // val_i
    // compare (key_i, val_i) < pivot
    cmp x28, x21
    blt .Linc_i
    bgt .Lcheck_j
    cmp x29, x22
    blt .Linc_i
    b .Lcheck_j

.Linc_i:
    add x25, x25, #1
    b .Lpartition_loop

.Lcheck_j:
    // load a[j]
    add x30, x19, x26, LSL #4
    ldr x31, [x30]                  // key_j
