qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    ble .Lreturn

    // choose pivot = middle element
    asr x8, x20, #1
    lsl x8, x8, #4
    add x9, x19, x8
    ldr x21, [x9]          // pivot.key
    ldr x22, [x9, #8]      // pivot.val

    // swap pivot with first element
    ldr x10, [x19]         // a0.key
    ldr x11, [x19, #8]     // a0.val
    str x21, [x19]
    str x22, [x19, #8]
    str x10, [x9]
    str x11, [x9, #8]

    // partition
    mov x23, #1            // i = 1
    sub x24, x20, #1       // j = n-1

.Lpartition_loop:
    cmp x23, x24
    bgt .Lpartition_done

    // advance i while a[i] < pivot
.Linner_i:
    add x12, x19, x23, LSL #4
    ldr x13, [x12]
    ldr x14, [x12, #8]
    cmp x13, x21
    blt .Linc_i
    bgt .Linner_j
    cmp x14, x22
    blt .Linc_i
    // equal, stop advancing i
    b .Linner_j

.Linc_i:
    add x23, x23, #1
    b .Linner_i

    // decrement j while a[j] > pivot
.Linner_j:
    add x15, x19, x24, LSL #4
    ldr x16, [x15]
    ldr x17, [x15, #8]
    cmp x16, x21
    bgt .Ldec_j
    blt .Lswap_check
    cmp x17, x22
    bgt .Ldec_j
    // equal, stop decrementing j
    b .Lswap_check

.Ldec_j:
    sub x24, x24, #1
    b .Linner_j

.Lswap_check:
    cmp x23, x24
    bgt .Lpartition_done
    // swap a[i] and a[j]
    str x16, [x12]
    str x17, [x12, #8]
    str x13, [x15]
    str x14, [x15, #8]
    add x23, x23, #1
    sub x24, x24, #1
    b .Lpartition_loop

.Lpartition_done:
    // place pivot in its final position at index j
    add x12, x19, x24, LSL #4
    ldr x13, [x12]
    ldr x14, [x12, #8]
    str x21, [x12]
    str x22, [x12, #8]
    str x13, [x19]
    str x14, [x19, #8]

    // left size = j
    mov x25, x24
    // right size = n - j - 1
    sub x26, x20, x24
    sub x26, x26, #1

    // recurse on left part if size > 1
    cmp x25, #1
    ble .Lskip_left
    mov x0, x19
    mov x1, x25
    bl qsort_pairs
.Lskip_left:
    // recurse on right part if size > 1
    cmp x26, #1
    ble .Lreturn
    add x8, x24, #1
    lsl x8, x8, #4
    add x0, x19, x8
    mov x1, x26
    bl qsort_pairs

.Lreturn:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
