qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    ble .Lreturn

    // pivot = middle element
    asr x9, x20, #1           // mid index
    lsl x9, x9, #4            // bytes offset
    add x10, x19, x9          // pivot address
    ldr x11, [x10]            // pivot.key
    ldr x12, [x10, #8]        // pivot.val

    // swap pivot with first element
    ldr x13, [x19]
    ldr x14, [x19, #8]
    str x11, [x19]
    str x12, [x19, #8]
    str x13, [x10]
    str x14, [x10, #8]

    // i = 1, j = n-1
    mov x15, #1               // i
    sub x16, x20, #1          // j

.Lpartition_loop:
    cmp x15, x16
    bgt .Lpartition_done

    // advance i while a[i] < pivot
.Linner_i:
    lsl x17, x15, #4
    add x18, x19, x17
    ldr x0, [x18]             // cur.key
    ldr x1, [x18, #8]         // cur.val
    cmp x0, x11
    blt .Linc_i
    bgt .Linner_j
    cmp x1, x12
    blt .Linc_i
    b .Linner_j

.Linc_i:
    add x15, x15, #1
    b .Linner_i

    // decrement j while a[j] > pivot
.Linner_j:
    lsl x17, x16, #4
    add x18, x19, x17
    ldr x2, [x18]
    ldr x3, [x18, #8]
    cmp x2, x11
    bgt .Ldec_j
    blt .Lswap_check
    cmp x3, x12
    bgt .Ldec_j
    b .Lswap_check

.Ldec_j:
    sub x16, x16, #1
    b .Linner_j

    // swap a[i] and a[j]
.Lswap_check:
    cmp x15, x16
    bgt .Lpartition_done
    // compute addresses
    lsl x17, x15, #4
    add x4, x19, x17          // addr_i
    lsl x18, x16, #4
    add x5, x19, x18          // addr_j
    ldr x0, [x4]
    ldr x1, [x4, #8]
    ldr x2, [x5]
    ldr x3, [x5, #8]
    str x2, [x4]
    str x3, [x4, #8]
    str x0, [x5]
    str x1, [x5, #8]
    add x15, x15, #1
    sub x16, x16, #1
    b .Lpartition_loop

.Lpartition_done:
    // place pivot at position j
    lsl x17, x16, #4
    add x5, x19, x17          // addr_j
    ldr x0, [x5]
    ldr x1, [x5, #8]
    str x11, [x5]
    str x12, [x5, #8]
    str x0, [x19]
    str x1, [x19, #8]

    // left size = j
    mov x6, x16
    // right size = n - j - 1
    sub x7, x20, x16
    sub x7, x7, #1

    // recurse on left part if size > 1
    cmp x6, #1
    ble .Lskip_left
    mov x0, x19
    mov x1, x6
    bl qsort_pairs
.Lskip_left:
    // recurse on right part if size > 1
    cmp x7, #1
    ble .Lreturn
    add x8, x16, #1
    lsl x8, x8, #4
    add x0, x19, x8
    mov x1, x7
    bl qsort_pairs

.Lreturn:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
