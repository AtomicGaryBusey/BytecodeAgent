export function l $max_byte(l %p, l %n) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    jmp @loop

@loop:
    %i =l phi %i0 %i_next
    %max =l phi %max0 %max_next
    %c =w csltl %i %n
    jnz %c @body @exit

@body:
    %addr =l add %p %i
    %b =l loadub %addr
    %c2 =w csltl %max %b
    %max_next =l sel %c2 %b %max
    %i_next =l add %i 1
    jmp @loop

@exit:
    ret %max
}
