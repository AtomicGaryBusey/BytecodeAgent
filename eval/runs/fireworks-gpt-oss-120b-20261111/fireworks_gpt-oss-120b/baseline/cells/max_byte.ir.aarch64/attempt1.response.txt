export function l $max_byte(l %p, l %n) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    %c0 =w csltl %i0 %n
    jnz %c0 @loop @exit

@loop:
    %i =l phi %i0, %i_next
    %max =l phi %max0, %max_next
    %addr =l add %p %i
    %b =l loadub %addr
    %c =w csltl %max %b
    %max_next =l sel %c %b %max
    %i_next =l add %i 1
    %cond =w csltl %i_next %n
    jnz %cond @loop @exit

@exit:
    %res =l phi %max0, %max
    ret %res
}
