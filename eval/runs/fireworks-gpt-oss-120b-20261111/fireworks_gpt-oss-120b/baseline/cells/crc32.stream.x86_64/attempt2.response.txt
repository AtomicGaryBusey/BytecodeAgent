crc32:
    mov     eax, -1
    test    rsi, rsi
    jz      .Lfinish
.Lbyte_loop:
    movzx   edx, byte ptr [rdi]
    inc     rdi
    xor     eax, edx
    mov     ecx, 8
.Lbit_loop:
    test    al, 1
    jz      .Lshift
    shr     eax, 1
    xor     eax, 0xEDB88320
    jmp     .Lnextbit
.Lshift:
    shr     eax, 1
.Lnextbit:
    dec     ecx
    jnz     .Lbit_loop
    dec     rsi
    jnz     .Lbyte_loop
.Lfinish:
    xor     eax, -1
    ret
