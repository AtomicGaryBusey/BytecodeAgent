lcs_len:
    xor rax, rax
    xor r10, r10
.Linit:
    cmp r10, rcx
    jge .Linit_done
    mov qword ptr [r8 + r10*8], rax
    inc r10
    jmp .Linit
.Linit_done:
    xor r11, r11
.Louter:
    cmp r11, rsi
    jge .Louter_done
    mov al, byte ptr [rdi + r11]
    xor r12, r12
    xor r10, r10
.Linner:
    inc r10
    cmp r10, rcx
    jg .Linner_done
    mov r14, qword ptr [r8 + r10*8]
    mov dl, byte ptr [rdx + r10 - 1]
    cmp al, dl
    je .Leq
    mov r15, qword ptr [r8 + r10*8 - 8]
    mov r13, r14
    cmp r15, r14
    cmovg r13, r15
    jmp .Lstore_common
.Leq:
    lea r13, [r12 + 1]
.Lstore_common:
    mov qword ptr [r8 + r10*8], r13
    mov r12, r14
    jmp .Linner
.Linner_done:
    inc r11
    jmp .Louter
.Louter_done:
    mov rax, qword ptr [r8 + rcx*8]
    ret
