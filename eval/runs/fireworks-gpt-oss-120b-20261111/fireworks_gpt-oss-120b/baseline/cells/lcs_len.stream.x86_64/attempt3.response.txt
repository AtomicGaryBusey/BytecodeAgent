lcs_len:
    push r12
    push r13
    push r14
    push r15
    xor rax, rax
    xor r9, r9
.Linit:
    cmp r9, rcx
    jg .Linit_done
    mov qword ptr [r8 + r9*8], rax
    inc r9
    jmp .Linit
.Linit_done:
    xor r10, r10
.Louter:
    cmp r10, rsi
    jge .Louter_done
    mov al, byte ptr [rdi + r10]
    xor r11, r11
    xor r9, r9
.Linner:
    inc r9
    cmp r9, rcx
    jg .Linner_done
    movzx r15d, byte ptr [rdx + r9 - 1]
    mov r12, qword ptr [r8 + r9*8]          # up
    mov r13, qword ptr [r8 + r9*8 - 8]      # left
    cmp al, r15b
    je .Leq
    mov r14, r12
    cmp r13, r12
    cmovg r14, r13
    mov qword ptr [r8 + r9*8], r14
    jmp .Lstore_done
.Leq:
    lea r14, [r11 + 1]
    mov qword ptr [r8 + r9*8], r14
.Lstore_done:
    mov r11, r12
    jmp .Linner
.Linner_done:
    inc r10
    jmp .Louter
.Louter_done:
    mov rax, qword ptr [r8 + rcx*8]
    pop r15
    pop r14
    pop r13
    pop r12
    ret
