export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
  %zero = lit 0
  %one = lit 1
  %eight = lit 8
  %limit_n = addl %n, %one          # n+1
  # init row[0..n] = 0
  br @init_loop %zero
@init_loop
  %j = phi %zero, %j_inc
  %c = csltl %j, %limit_n
  jnz %c, @init_body, @after_init
@init_body
  %off = mul %j, %eight
  %addr = addl %row, %off
  storel %addr, %zero
  %j_inc = addl %j, %one
  br @init_loop %j_inc
@after_init
  # outer loop over i
  br @outer_loop %zero
@outer_loop
  %i = phi %zero, %i_next
  %c_i = csltl %i, %m
  jnz %c_i, @outer_body, @after_outer
@outer_body
  # inner loop over j = 1..n, diag = 0
  br @inner_loop %one %zero
@inner_loop
  %j = phi %one, %j_next
  %diag = phi %zero, %diag_next
  %c_j = csltl %j, %limit_n
  jnz %c_j, @inner_body, @after_inner
@inner_body
  %off_j = mul %j, %eight
  %addr_j = addl %row, %off_j
  %up = loadl %addr_j
  %j1 = subl %j, %one
  %off_j1 = mul %j1, %eight
  %addr_j1 = addl %row, %off_j1
  %left = loadl %addr_j1
  %gt_up_left = csgtl %up, %left
  jnz %gt_up_left, @max_up, @max_left
@max_up
  br @max_end
@max_left
  br @max_end
@max_end
  %max = phi %up, %left
  %x_off = addl %x, %i
  %xb = w loadub %x_off
  %y_off = addl %y, %j1
  %yb = w loadub %y_off
  %eq = ceql %xb, %yb
  jnz %eq, @eq_true, @eq_false
@eq_true
  %diag_plus1 = addl %diag, %one
  br @after_eq
@eq_false
  br @after_eq
@after_eq
  %new = phi %diag_plus1, %max
  storel %addr_j, %new
  %j_next = addl %j, %one
  %diag_next = %up
  br @inner_loop %j_next %diag_next
@after_inner
  %i_next = addl %i, %one
  br @outer_loop %i_next
@after_outer
  %off_n = mul %n, %eight
  %addr_n = addl %row, %off_n
  %res = loadl %addr_n
  ret %res
}
