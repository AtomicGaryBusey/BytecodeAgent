export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
  %zero = lit 0
  %one = lit 1
  %eight = lit 8

  %n1 = addl %n, %one          # n+1

  # init row[0..n] = 0
  br @init %zero
@init
  %j = phi %zero %entry, %j_next %init_body
  %c = csltl %j, %n1
  jnz %c, @init_body, @after_init
@init_body
  %off = mul %j, %eight
  %addr = addl %row, %off
  storel %zero %addr
  %j_next = addl %j, %one
  br @init %j_next
@after_init

  # outer loop over i = 0 .. m-1
  br @outer %zero
@outer
  %i = phi %zero %entry, %i_next %outer_body
  %c_i = csltl %i, %m
  jnz %c_i, @outer_body, @after_outer
@outer_body
  # inner loop over j = 1 .. n, diag = 0
  br @inner %one %zero
@inner
  %j = phi %one %outer_body, %j_next %inner_body
  %diag = phi %zero %outer_body, %diag_next %inner_body
  %c_j = csltl %j, %n1
  jnz %c_j, @inner_body, @after_inner
@inner_body
  # up = row[j]
  %offj = mul %j, %eight
  %addrj = addl %row, %offj
  %up = loadl %addrj

  # left = row[j-1]
  %j1 = subl %j, %one
  %offj1 = mul %j1, %eight
  %addrj1 = addl %row, %offj1
  %left = loadl %addrj1

  # max = (up > left) ? up : left
  %gt = csgtl %up, %left
  jnz %gt, @max_up, @max_left
@max_up
  %max_up_val = phi %up %max_up, %left %max_left
  br @max_merge
@max_left
  %max_left_val = phi %left %max_left, %up %max_up
  br @max_merge
@max_merge
  %max = phi %max_up_val %max_up, %max_left_val %max_left

  # compare x[i] and y[j-1]
  %xoff = addl %x, %i
  %xb = w loadub %xoff
  %yoff = addl %y, %j1
  %yb = w loadub %yoff
  %eq = ceql %xb, %yb
  jnz %eq, @eq_true, @eq_false
@eq_true
  %new_eq = addl %diag, %one
  br @new_merge
@eq_false
  %new_eq = phi %diag %eq_false, %max %eq_true   # dummy, will be replaced
  br @new_merge
@new_merge
  %new = phi %new_eq %eq_true, %max %eq_false

  storel %new %addrj

  %j_next = addl %j, %one
  %diag_next = %up
  br @inner %j_next %diag_next

@after_inner
  %i_next = addl %i, %one
  br @outer %i_next
@after_outer
  %offn = mul %n, %eight
  %addrn = addl %row, %offn
  %res = loadl %addrn
  ret %res
}
