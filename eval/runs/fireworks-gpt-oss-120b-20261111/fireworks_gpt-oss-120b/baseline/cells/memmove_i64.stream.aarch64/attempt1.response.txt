memmove_i64:
    // compare dst and src
    cmp x0, x1
    b.lt .Lforward
    // compute src_end = src + n*8
    add x5, x1, x2, LSL #3
    cmp x0, x5
    b.ge .Lforward
    // overlapping, copy backwards
    b .Lbackward

.Lforward:
    mov x3, #0
.Lfwd_loop:
    cmp x3, x2
    b.eq .Lreturn
    ldr x4, [x1, x3, LSL #3]
    str x4, [x0, x3, LSL #3]
    add x3, x3, #1
    b .Lfwd_loop

.Lbackward:
    mov x3, x2
.Lbwd_loop:
    subs x3, x3, #1
    ldr x4, [x1, x3, LSL #3]
    str x4, [x0, x3, LSL #3]
    cbnz x3, .Lbwd_loop

.Lreturn:
    mov x0, #0
    ret
