mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rsi            # n
    mov r13, rdx            # exponent k
    mov r14, rcx            # out pointer
    mov r15, r8             # tmp0 pointer

    mov r12, rbx
    imul r12, rbx           # n_sq = n*n

    # copy a to tmp0 (base matrix)
    xor rax, rax
.Lcopy_a:
    cmp rax, r12
    jge .Lcopy_a_done
    mov rdx, [rdi + rax*8]
    mov [r15 + rax*8], rdx
    inc rax
    jmp .Lcopy_a
.Lcopy_a_done:

    # initialize out to identity matrix
    xor r8, r8              # i = 0
.Linit_i:
    cmp r8, rbx
    jge .Linit_done
    xor r9, r9              # j = 0
.Linit_j:
    cmp r9, rbx
    jge .Lnext_i
    mov rax, r8
    imul rax, rbx
    add rax, r9             # index = i*n + j
    xor rdx, rdx
    cmp r8, r9
    jne .Lset_id
    mov rdx, 1
.Lset_id:
    mov [r14 + rax*8], rdx
    inc r9
    jmp .Linit_j
.Lnext_i:
    inc r8
    jmp .Linit_i
.Linit_done:

    # exponentiation loop
.Lexp_loop:
    test r13, r13
    jz .Lexp_done

    test r13, 1
    jz .Lskip_mul_res
    # res = res * base  (out * base -> tmp1)
    push r14                # save out pointer
    mov rdi, r14            # A = out
    mov rsi, r15            # B = base
    # compute tmp1 pointer in rdx
    lea rdx, [r15 + r12*8]  # Dest = tmp1
    call .Lmultiply
    pop r14                 # restore out pointer
    # copy tmp1 -> out
    lea r10, [r15 + r12*8]  # tmp1 base
    xor rax, rax
.Lcopy_res:
    cmp rax, r12
    jge .Lcopy_res_done
    mov rcx, [r10 + rax*8]
    mov [r14 + rax*8], rcx
    inc rax
    jmp .Lcopy_res
.Lcopy_res_done:
.Lskip_mul_res:

    # base = base * base  (base * base -> tmp1)
    push r15                # save base pointer
    mov rdi, r15            # A = base
    mov rsi, r15            # B = base
    lea rdx, [r15 + r12*8]  # Dest = tmp1
    call .Lmultiply
    pop r15                 # restore base pointer
    # copy tmp1 -> base
    lea r10, [r15 + r12*8]  # tmp1 base
    xor rax, rax
.Lcopy_base:
    cmp rax, r12
    jge .Lcopy_base_done
    mov rcx, [r10 + rax*8]
    mov [r15 + rax*8], rcx
    inc rax
    jmp .Lcopy_base
.Lcopy_base_done:

    shr r13, 1
    jmp .Lexp_loop
.Lexp_done:

    # zero tmp0
    xor rax, rax
.Lzero_tmp0:
    cmp rax, r12
    jge .Lzero_tmp0_done
    mov qword ptr [r15 + rax*8], 0
    inc rax
    jmp .Lzero_tmp0
.Lzero_tmp0_done:

    # zero tmp1
    lea r10, [r15 + r12*8]  # tmp1 base
    xor rax, rax
.Lzero_tmp1:
    cmp rax, r12
    jge .Lzero_tmp1_done
    mov qword ptr [r10 + rax*8], 0
    inc rax
    jmp .Lzero_tmp1
.Lzero_tmp1_done:

    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# multiply matrix: A in rdi, B in rsi, Dest in rdx
.Lmultiply:
    xor r8, r8                # i = 0
.Lmul_i:
    cmp r8, rbx
    jge .Lmul_done
    xor r9, r9                # j = 0
.Lmul_j:
    cmp r9, rbx
    jge .Lnext_i_mul
    mov rax, r8
    imul rax, rbx
    add rax, r9               # dest offset = i*n + j
    xor r11, r11              # accumulator
    xor r10, r10              # t = 0
.Lmul_t:
    cmp r10, rbx
    jge .Lstore
    # load A[i*n + t] into rax
    mov rax, r8
    imul rax, rbx
    add rax, r10
    mov rax, [rdi + rax*8]    # rax = A element
    # load B[t*n + j] into rcx
    mov rcx, r10
    imul rcx, rbx
    add rcx, r9
    mov rcx, [rsi + rcx*8]    # rcx = B element
    imul rax, rcx             # signed multiply, low 64 bits
    add r11, rax
    inc r10
    jmp .Lmul_t
.Lstore:
    # store accumulator into Dest
    mov rax, r8
    imul rax, rbx
    add rax, r9
    mov [rdx + rax*8], r11
    inc r9
    jmp .Lmul_j
.Lnext_i_mul:
    inc r8
    jmp .Lmul_i
.Lmul_done:
    ret
