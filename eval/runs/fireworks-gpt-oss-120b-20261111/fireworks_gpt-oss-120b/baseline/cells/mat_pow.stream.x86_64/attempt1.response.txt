mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rsi                # n
    mov r10, rbx
    imul r10, r10               # n_sq = n*n

    mov r12, r8                 # tmp0 pointer
    lea r13, [r12 + r10*8]      # tmp1 pointer
    mov r14, rcx                # out pointer

    # copy a to tmp0
    xor rax, rax
.Lcopy_a:
    cmp rax, r10
    jge .Lcopy_a_done
    mov rdx, [rdi + rax*8]
    mov [r12 + rax*8], rdx
    inc rax
    jmp .Lcopy_a
.Lcopy_a_done:

    # initialize out to identity
    xor r8, r8                  # i = 0
.Linit_i:
    cmp r8, rbx
    jge .Linit_done
    xor r9, r9                  # j = 0
.Linit_j:
    cmp r9, rbx
    jge .Linit_next_i
    mov rdx, r8
    imul rdx, rdx, rbx          # i*n
