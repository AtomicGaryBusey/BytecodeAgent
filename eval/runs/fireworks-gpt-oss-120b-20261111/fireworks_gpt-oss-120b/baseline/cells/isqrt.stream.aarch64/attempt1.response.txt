isqrt:
    movz    x1, #0
    movz    x2, #0x4000, lsl #48
.Lloop:
    adds    x3, x1, x2
    cmp     x0, x3
    b.cc    .Lskip_sub
    sub     x0, x0, x3
    lsr     x1, x1, #1
    add     x1, x1, x2
    b       .Lafter_sub
.Lskip_sub:
    lsr     x1, x1, #1
.Lafter_sub:
    lsr     x2, x2, #2
    cbnz    x2, .Lloop
    mov     x0, x1
    ret
