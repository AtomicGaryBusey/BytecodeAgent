export function l $fill_down(l a, l n, l start) {
@entry:
    %i = copy 0
    %c = csltl %i, %n
    jnz %c, @loop
    jump @exit

@loop:
    %off = shll %i, 3
    %addr = add %a, %off
    %val = sub %start, %i
    storel %val, %addr
    %i = add %i, 1
    %c = csltl %i, %n
    jnz %c, @loop
    jump @exit

@exit:
    ret %n
}
