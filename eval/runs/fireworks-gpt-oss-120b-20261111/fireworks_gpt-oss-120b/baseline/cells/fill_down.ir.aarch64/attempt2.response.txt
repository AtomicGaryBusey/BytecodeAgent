export function l $fill_down(l p0, l p1, l p2) {
@entry:
    %i = copy 0
    %c = csltl %i, p1
    jnz %c, @loop
    jump @exit

@loop:
    %off = shll %i, 3
    %addr = add p0, %off
    %val = sub p2, %i
    storel %val, %addr
    %i = add %i, 1
    %c = csltl %i, p1
    jnz %c, @loop
    jump @exit

@exit:
    ret p1
}
