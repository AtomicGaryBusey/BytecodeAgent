export function l $fill_down(l a, l n, l start) {
@entry:
    %i = copy 0
    jump @cond

@cond:
    %cmp = csltl %i, n
    jnz %cmp, @body
    jump @exit

@body:
    %off = shll %i, 3
    %p = add a, %off
    %val = sub start, %i
    storel %val, %p
    %i2 = add %i, 1
    %i = copy %i2
    jump @cond

@exit:
    ret n
}
