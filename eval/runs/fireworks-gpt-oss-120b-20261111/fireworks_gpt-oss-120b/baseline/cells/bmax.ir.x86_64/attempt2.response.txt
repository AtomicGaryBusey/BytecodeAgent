export function l $bmax(l a, l n) {
@entry
	%i = copyl 0
	%v0 = loadl %a,0
	%max = copyl %v0
	%i = addl %i,1
	%cond0 = csltl %i,%n
	jnz %cond0,@loop
	ret %max

@loop
	%off = mull %i,8
	%p = addl %a,%off
	%v = loadl %p,0
	%cmp = csltl %max,%v
	%max = sel %cmp,%v,%max
	%i = addl %i,1
	%cond1 = csltl %i,%n
	jnz %cond1,@loop
	ret %max
}
