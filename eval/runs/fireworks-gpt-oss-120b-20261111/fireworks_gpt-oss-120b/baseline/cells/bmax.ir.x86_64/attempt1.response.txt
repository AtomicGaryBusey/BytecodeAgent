export function l $bmax(l a, l n) {
@entry
	%i0 = copyl 0
	%first = loadl a,0
	%max0 = copyl %first
	%i1 = addl %i0,1
	%cond0 = csltl %i1,n
	jnz %cond0,@loop
	ret %max0

@loop
	%off0 = mul %i1,8
	%p0 = addl a,%off0
	%val0 = loadl %p0,0
	%cmp0 = csltl %max0,%val0
	%max1 = sel %cmp0,%val0,%max0
	%i2 = addl %i1,1
	%cond1 = csltl %i2,n
	%max0 = copyl %max1
	%i1 = copyl %i2
	jnz %cond1,@loop
	ret %max0
}
