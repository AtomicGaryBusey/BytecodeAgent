export function l $bmax(l, l) {
@entry
	%i = copyl 0
	%max = loadl %0,0
	%i = addl %i,1
	%cond = csltl %i, %1
	jnz %cond, @loop
	ret %max

@loop
	%off = mull %i,8
	%p = addl %0, %off
	%v = loadl %p,0
	%cmp = csltl %max, %v
	%max = sel %cmp, %v, %max
	%i = addl %i,1
	%cond = csltl %i, %1
	jnz %cond, @loop
	ret %max
}
