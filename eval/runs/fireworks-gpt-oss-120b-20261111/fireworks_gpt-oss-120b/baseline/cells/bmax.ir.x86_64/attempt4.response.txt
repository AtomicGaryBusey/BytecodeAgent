export function l $bmax(l %a, l %n) {
@entry
	%i0 = copyl 0
	%max0 = loadl %a,0
	jmp @loop

@loop
	%i = phi %i0, @entry, %i1, @loop
	%max = phi %max0, @entry, %max1, @loop
	%cond = csltl %i, %n
	jnz %cond, @body
	ret %max

@body
	%off = mull %i,8
	%p = addl %a, %off
	%v = loadl %p,0
	%cmp = csltl %max, %v
	%max1 = sel %cmp, %v, %max
	%i1 = addl %i,1
	jmp @loop
}
