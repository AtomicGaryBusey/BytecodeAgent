export function $xxh64(l %p, l %n, l %seed) {
@entry:
    %P1 =l -7046029288634856825          // 0x9E3779B185EBCA87
    %P2 =l -4417276706812531889          // 0xC2B2AE3D27D4EB4F
    %P3 =l 1609587929392839161           // 0x165667B19E3779F9
    %P4 =l -8775519303254910029          // 0x85EBCA77C2B2AE63
    %P5 =l 2851391565950198965           // 0x27D4EB2F165667C5

    %len32 =l and %n, -32                // n & ~31
    %n_lt_32 =w csltul %n, 32
    jnz %n_lt_32, @small

    // ------- n >= 32 -------
    %v1 =l add %seed, %P1
    %v1 =l add %v1, %P2
    %v2 =l add %seed, %P2
    %v3 =l mov %seed
    %v4 =l sub %seed, %P1

    %q =l mov %p
    %end =l add %p, %len32

@loop32:
    %cond =w csltul %q, %end
    jnz %cond, @body32
    br @after32

@body32:
    // v1
    %a0 =l loadl %q
    %t0 =l mul %a0, %P2
    %t1 =l add %v1, %t0
    %rot =l shl %t1, 31
    %rot2 =l lshr %t1, 33
    %rot =l or %rot, %rot2
    %v1 =l mul %rot, %P1
    // v2
    %q8 =l add %q, 8
    %a1 =l loadl %q8
    %t0 =l mul %a1, %P2
    %t1 =l add %v2, %t0
    %rot =l shl %t1, 31
    %rot2 =l lshr %t1, 33
    %rot =l or %rot, %rot2
    %v2 =l mul %rot, %P1
    // v3
    %q16 =l add %q, 16
    %a2 =l loadl %q16
    %t0 =l mul %a2, %P2
    %t1 =l add %v3, %t0
    %rot =l shl %t1, 31
    %rot2 =l lshr %t1, 33
    %rot =l or %rot, %rot2
    %v3 =l mul %rot, %P1
    // v4
    %q24 =l add %q, 24
    %a3 =l loadl %q24
    %t0 =l mul %a3, %P2
    %t1 =l add %v4, %t0
    %rot =l shl %t1, 31
    %rot2 =l lshr %t1, 33
    %rot =l or %rot, %rot2
    %v4 =l mul %rot, %P1

    %q =l add %q, 32
    br @loop32

@after32:
    // h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18)
    %tmp =l shl %v1, 1
    %tmp2 =l lshr %v1, 63
    %r1 =l or %tmp, %tmp2
    %tmp =l shl %v2, 7
    %tmp2 =l lshr %v2, 57
    %r2 =l or %tmp, %tmp2
    %tmp =l shl %v3, 12
    %tmp2 =l lshr %v3, 52
    %r3 =l or %tmp, %tmp2
    %tmp =l shl %v4, 18
    %tmp2 =l lshr %v4, 46
    %r4 =l or %tmp, %tmp2
    %h =l add %r1, %r2
    %h =l add %h, %r3
    %h =l add %h, %r4

    // merge with v1
    %tmp =l mul %v1, %P2
    %tmp =l shl %tmp, 31
    %tmp2 =l lshr %tmp, 33
    %tmp =l or %tmp, %tmp2
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4
    // merge with v2
    %tmp =l mul %v2, %P2
    %tmp =l shl %tmp, 31
    %tmp2 =l lshr %tmp, 33
    %tmp =l or %tmp, %tmp2
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4
    // merge with v3
    %tmp =l mul %v3, %P2
    %tmp =l shl %tmp, 31
    %tmp2 =l lshr %tmp, 33
    %tmp =l or %tmp, %tmp2
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4
    // merge with v4
    %tmp =l mul %v4, %P2
    %tmp =l shl %tmp, 31
    %tmp2 =l lshr %tmp, 33
    %tmp =l or %tmp, %tmp2
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4
    br @posth

@small:
    %h =l add %seed, %P5
    br @posth

@posth:
    %h =l add %h, %n

    %q =l add %p, %len32
    %rem =l sub %n, %len32

@loop8:
    %cond8 =w csltul %rem, 8
    jnz %cond8, @after8
    %x8 =l loadl %q
    // round(0, x8)
    %t =l mul %x8, %P2
    %t =l shl %t, 31
    %t2 =l lshr %t, 33
    %t =l or %t, %t2
    %t =l mul %t, %P1
    %h =l xor %h, %t
    // rotl 27 then *P1 + P4
    %rot =l shl %h, 27
    %rot2 =l lshr %h, 37
    %rot =l or %rot, %rot2
    %h =l mul %rot, %P1
    %h =l add %h, %P4
    %q =l add %q, 8
    %rem =l sub %rem, 8
    br @loop8

@after8:
    %cond4 =w csltul %rem, 4
    jnz %cond4, @after4
    %x4 =l loaduw %q
    %h =l xor %h, %x4
    %h =l mul %h, %P1
    %rot =l shl %h, 23
    %rot2 =l lshr %h, 41
    %rot =l or %rot, %rot2
    %h =l mul %rot, %P2
    %h =l add %h, %P3
    %q =l add %q, 4
    %rem =l sub %rem, 4
    br @after4

@after4:
    %cond1 =w csltul %rem, 1
    jnz %cond1, @final

@byte_loop:
    %c =l loadub %q
    %tmp =l mul %c, %P5
    %h =l xor %h, %tmp
    %rot =l shl %h, 11
    %rot2 =l lshr %h, 53
    %rot =l or %rot, %rot2
    %h =l mul %rot, %P1
    %q =l add %q, 1
    %rem =l sub %rem, 1
    %cond =w csltul %rem, 1
    jnz %cond, @final
    br @byte_loop

@final:
    // avalanche
    %t =l lshr %h, 33
    %h =l xor %h, %t
    %h =l mul %h, %P2
    %t =l lshr %h, 29
    %h =l xor %h, %t
    %h =l mul %h, %P3
    %t =l lshr %h, 32
    %h =l xor %h, %t
    ret %h
}
