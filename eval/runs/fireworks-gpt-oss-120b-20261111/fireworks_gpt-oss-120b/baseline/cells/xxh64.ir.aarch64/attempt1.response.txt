export function $xxh64(l %p, l %n, l %seed) {
    %P1 =l -7046029288634856825          // 0x9E3779B185EBCA87
    %P2 =l -4417276706812531889          // 0xC2B2AE3D27D4EB4F
    %P3 =l 1609587929392839161           // 0x165667B19E3779F9
    %P4 =l -8775519303254910029          // 0x85EBCA77C2B2AE63
    %P5 =l 2851391565950198965           // 0x27D4EB2F165667C5
    %zero =l 0

    %n_lt_32 =w csltul %n, 32
    jnz %n_lt_32, @small

    // >= 32 bytes
    %v1 =l add %seed, %P1
    %v1 =l add %v1, %P2
    %v2 =l add %seed, %P2
    %v3 =l mov %seed
    %v4 =l sub %seed, %P1

    %len32 =l and %n, -32               // n & ~31
    %q =l mov %p
    %end =l add %p, %len32

@loop32:
    %cond =w csltul %q, %end
    jnz %cond, @body32
    br @after32

@body32:
    // v1
    %a0 =l loadul %q
    %t0 =l mul %a0, %P2
    %t1 =l add %v1, %t0
    %t2 =l shll %t1, 31
    %t3 =l shrl %t1, 33
    %t4 =l or %t2, %t3
    %v1 =l mul %t4, %P1
    // v2
    %q8 =l add %q, 8
    %a1 =l loadul %q8
    %t0 =l mul %a1, %P2
    %t1 =l add %v2, %t0
    %t2 =l shll %t1, 31
    %t3 =l shrl %t1, 33
    %t4 =l or %t2, %t3
    %v2 =l mul %t4, %P1
    // v3
    %q16 =l add %q, 16
    %a2 =l loadul %q16
    %t0 =l mul %a2, %P2
    %t1 =l add %v3, %t0
    %t2 =l shll %t1, 31
    %t3 =l shrl %t1, 33
    %t4 =l or %t2, %t3
    %v3 =l mul %t4, %P1
    // v4
    %q24 =l add %q, 24
    %a3 =l loadul %q24
    %t0 =l mul %a3, %P2
    %t1 =l add %v4, %t0
    %t2 =l shll %t1, 31
    %t3 =l shrl %t1, 33
    %t4 =l or %t2, %t3
    %v4 =l mul %t4, %P1
    // advance
    %q =l add %q, 32
    br @loop32

@after32:
    // h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18)
    %r1 =l shll %v1, 1
    %r1 =l or %r1, (shrl %v1, 63)
    %r2 =l shll %v2, 7
    %r2 =l or %r2, (shrl %v2, 57)
    %r3 =l shll %v3, 12
    %r3 =l or %r3, (shrl %v3, 52)
    %r4 =l shll %v4, 18
    %r4 =l or %r4, (shrl %v4, 46)
    %h =l add %r1, %r2
    %h =l add %h, %r3
    %h =l add %h, %r4

    // merge(h, v1..v4)
    // round0(v) = rotl(v*P2,31)*P1
    %tmp =l mul %v1, %P2
    %tmp =l shll %tmp, 31
    %tmp =l or %tmp, (shrl %tmp, 33)
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4

    %tmp =l mul %v2, %P2
    %tmp =l shll %tmp, 31
    %tmp =l or %tmp, (shrl %tmp, 33)
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4

    %tmp =l mul %v3, %P2
    %tmp =l shll %tmp, 31
    %tmp =l or %tmp, (shrl %tmp, 33)
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4

    %tmp =l mul %v4, %P2
    %tmp =l shll %tmp, 31
    %tmp =l or %tmp, (shrl %tmp, 33)
    %tmp =l mul %tmp, %P1
    %h =l xor %h, %tmp
    %h =l mul %h, %P1
    %h =l add %h, %P4
    br @posth

@small:
    %h =l add %seed, %P5

@posth:
    // h += n
    %h =l add %h, %n

    // remaining bytes processing
    %q =l add %p, %len32
    %rem =l sub %n, %len32

@loop8:
    %cond8 =w csltul %rem, 8
    jnz %cond8, @after8
    %x8 =l loadul %q
    // round(0,x8)
    %t0 =l mul %x8, %P2
    %t1 =l shll %t0, 31
    %t2 =l or %t1, (shrl %t0, 33)
    %t3 =l mul %t2, %P1
    %h =l xor %h, %t3
    %h =l shll %h, 27
    %h =l or %h, (shrl %h, 37)
    %h =l mul %h, %P1
    %h =l add %h, %P4
    %q =l add %q, 8
    %rem =l sub %rem, 8
    br @loop8

@after8:
    %cond4 =w csltul %rem, 4
    jnz %cond4, @after4
    %x4 =l loaduw %q
    %tmp =l mul %x4, %P1
    %h =l xor %h, %tmp
    %h =l shll %h, 23
    %h =l or %h, (shrl %h, 41)
    %h =l mul %h, %P2
    %h =l add %h, %P3
    %q =l add %q, 4
    %rem =l sub %rem, 4
    br @after4

@after4:
    %cond1 =w csltul %rem, 1
    jnz %cond1, @final

@byte_loop:
    %c =l loadub %q
    %tmp =l mul %c, %P5
    %h =l xor %h, %tmp
    %h =l shll %h, 11
    %h =l or %h, (shrl %h, 53)
    %h =l mul %h, %P1
    %q =l add %q, 1
    %rem =l sub %rem, 1
    %cond =w csltul %rem, 1
    jnz %cond, @final
    br @byte_loop

@final:
    // avalanche
    %t =l shrl %h, 33
    %h =l xor %h, %t
    %h =l mul %h, %P2
    %t =l shrl %h, 29
    %h =l xor %h, %t
    %h =l mul %h, %P3
    %t =l shrl %h, 32
    %h =l xor %h, %t
    ret %h
}
