sdivrem:
    cbz x1, .L_bzero
    mov x3, #-1
    cmp x1, x3
    b.ne .L_normal
.L_bneg1:
    mov x4, #0x8000
    lsl x4, x4, #48
    cmp x0, x4
    b.ne .L_bneg1_not_min
    // a == INT64_MIN && b == -1
    mov x5, #0
    str x5, [x2]
    mov x0, x4
    ret
.L_bneg1_not_min:
    // b == -1, a != INT64_MIN
    mov x5, #0
    str x5, [x2]
    sub x0, xzr, x0
    ret
.L_bzero:
    // b == 0
    str x0, [x2]
    mov x0, #0
    ret
.L_normal:
    // sign masks
    asr x3, x0, #63          // sign_a
    asr x4, x1, #63          // sign_b
    // abs_a = (a ^ sign_a) - sign_a
    eor x5, x0, x3
    sub x5, x5, x3
    // abs_b = (b ^ sign_b) - sign_b
    eor x6, x1, x4
    sub x6, x6, x4
    // unsigned division
    udiv x7, x5, x6
    // remainder = abs_a - quotient*abs_b
    mul x8, x7, x6
    sub x8, x5, x8
    // adjust quotient sign
    eor x9, x3, x4          // xor of signs
    cbz x9, .L_qpos
    sub x0, xzr, x7         // negative quotient
    b .L_qdone
.L_qpos:
    mov x0, x7
.L_qdone:
    // adjust remainder sign (sign of a)
    cbz x3, .L_rpos
    sub x8, xzr, x8
.L_rpos:
    str x8, [x2]
    ret
