    cmp x1, #-1
    b.ne .L_normal
    // b == -1
    mov x4, #0
    movk x4, #0x8000, LSL #48
    cmp x0, x4
    b.ne .L_bneg_not_min
    // a == INT64_MIN
    mov x5, #0
    str x5, [x2]
    mov x0, x4
    ret
.L_bneg_not_min:
    // a != INT64_MIN
    rsb x0, x0, #0
