export function $binary_search(l a, l n, l key) {
@entry:
    sub %high0, %n, 1
    add %low0, 0, 0
    jmp @loop

@loop:
    %low = phi %low0, @entry, %low_next, @low_update
    %high = phi %high0, @entry, %high_next, @high_update
    csltl %cond, %high, %low
    jnz %cond, @notfound
    add %mid_sum, %low, %high
    shrl %mid, %mid_sum, 1
    mul %mid_off, %mid, 8
    add %addr, %a, %mid_off
    loadl %val, %addr
    ceql %eq, %val, %key
    jnz %eq, @found
    csltl %lt, %val, %key
    jnz %lt, @low_update
    sub %high_next, %mid, 1
    jmp @high_update

@low_update:
    add %low_next, %mid, 1
    jmp @loop

@high_update:
    jmp @loop

@found:
    ret %mid

@notfound:
    ret -1
}
