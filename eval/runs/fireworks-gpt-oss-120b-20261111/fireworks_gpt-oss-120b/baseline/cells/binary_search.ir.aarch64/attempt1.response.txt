export function l $binary_search(l a, l n, l key) {
@entry:
    sub %high, n, 1
    add %low, 0, 0
@loop:
    csltl %exit, %high, %low
    jnz %exit, @notfound
    add %mid_sum, %low, %high
    shrl %mid, %mid_sum, 1
    mul %mid_off, %mid, 8
    add %addr, a, %mid_off
    loadl %val, %addr
    ceql %eq, %val, key
    jnz %eq, @found
    csltl %lt, %val, key
    jnz %lt, @low_update
    sub %high, %mid, 1
    jmp @loop
@low_update:
    add %low, %mid, 1
    jmp @loop
@found:
    ret %mid
@notfound:
    ret -1
}
