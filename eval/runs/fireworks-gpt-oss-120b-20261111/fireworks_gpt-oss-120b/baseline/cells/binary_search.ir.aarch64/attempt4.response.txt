export function $binary_search(l a, l n, l key) {
@entry:
    add %low0, 0, 0
    sub %high0, %n, 1
    jmp @loop

@loop:
    %low = phi %low0, @entry, %low_next, @low_update, %low, @high_update
    %high = phi %high0, @entry, %high, @low_update, %high_next, @high_update
    csltl %cond, %high, %low
    jnz %cond, @notfound
    sub %diff, %high, %low
    shrl %half, %diff, 1
    add %mid, %low, %half
    mul %mid_off, %mid, 8
    add %addr, %a, %mid_off
    loadl %val, %addr
    ceql %eq, %val, %key
    jnz %eq, @found
    csltl %lt, %val, %key
    jnz %lt, @low_update
    sub %high_next, %mid, 1
    jmp @high_update

@low_update:
    add %low_next, %mid, 1
    jmp @loop

@high_update:
    jmp @loop

@found:
    ret %mid

@notfound:
    ret -1
}
