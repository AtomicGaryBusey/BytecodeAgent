fib:
    cmp rdi, 1
    jle .Lbase
    xor r9, r9            # a = 0
    mov r8, 1             # b = 1
    mov rcx, rdi          # rcx = n
    dec rcx               # loop count = n-1
.Lloop:
    test rcx, rcx
    je .Ldone
    mov rax, r9           # rax = a
    add rax, r8           # rax = a + b (next)
    mov r9, r8            # a = b
    mov r8, rax           # b = next
    dec rcx
    jmp .Lloop
.Ldone:
    mov rax, r8           # result = b
    ret
.Lbase:
    mov rax, rdi          # result = n (0 or 1)
    ret
