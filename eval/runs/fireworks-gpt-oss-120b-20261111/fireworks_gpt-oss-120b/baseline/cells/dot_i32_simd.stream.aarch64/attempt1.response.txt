dot_i32_simd:
    eor    v0.16b, v0.16b, v0.16b          // accumulator vector = 0
    mov    x3, x2                         // x3 = n
    lsr    x4, x3, #1                     // x4 = n / 2 (pair count)
    cbz    x4, .Ltail                     // if no pairs, skip vector loop

.Lvec_loop:
    ld1    {v1.2s}, [x0], #8              // load 2 int32 from a, post-increment
    ld1    {v2.2s}, [x1], #8              // load 2 int32 from b, post-increment
    smlal  v0.2d, v1.2s, v2.2s            // accumulator += a * b
    subs   x4, x4, #1
    bne    .Lvec_loop

.Ltail:
    umov   x6, v0.d[0]                    // lower lane
    umov   x7, v0.d[1]                    // upper lane
    add    x8, x6, x7                     // x8 = vector sum
    and    x5, x3, #1                     // x5 = n & 1 (remaining element?)
    cbz    x5, .Lfinish

    ldr    w6, [x0]                       // load remaining a[i]
    ldr    w7, [x1]                       // load remaining b[i]
    sxtw   x6, w6
    sxtw   x7, w7
    mul    x6, x6, x7
    add    x8, x8, x6

.Lfinish:
    mov    x0, x8                         // return sum
    ret
