export function l $isqrt(l n) {
@entry:
    %c0 = const l 32
    %res0 = const l 0
    %bit0 = const l 1
    %bit0 = shll %bit0, const l 62
    jmp @loop

@loop:
    %c = phi l (%c0, @entry), (%c1, @after)
    %res = phi l (%res0, @entry), (%res2, @after)
    %ncur = phi l (n, @entry), (%n2, @after)
    %bit = phi l (%bit0, @entry), (%bit1, @after)

    %c_eq0 = ceql %c, const l 0
    jnz %c_eq0, @exit

    %sum = addl %res, %bit
    %lt = cultl %ncur, %sum
    %ge = notw %lt
    jnz %ge, @true

@false:
    %res1 = shrl %res, const l 1
    %n1 = %ncur
    jmp @after

@true:
    %n1 = subl %ncur, %sum
    %rshift = shrl %res, const l 1
    %res1 = addl %rshift, %bit
    jmp @after

@after:
    %bit1 = shrl %bit, const l 2
    %c1 = subl %c, const l 1
    %n2 = %n1
    %res2 = %res1
    jmp @loop

@exit:
    ret %res
}
