export function l $isqrt(l n) {
@entry:
    %bit0 = const l 1
    %bit0 = shll %bit0, const l 62
    %res0 = const l 0
    %n0   = n
    jmp @adjust_bit

@adjust_bit:
    %bit   = phi l (%bit0, @entry), (%bit1, @shift)
    %ncur  = phi l (%n0, @entry), (%n0, @shift)
    %gt    = cultl %ncur, %bit        // ncur < bit  => bit > n
    jnz %gt, @shift
    jmp @loop_start

@shift:
    %bit1 = shrl %bit, const l 2
    jmp @adjust_bit

@loop_start:
    %res   = const l 0
    %ncur2 = %ncur
    %bit2  = %bit
    jmp @loop

@loop:
    %bit   = phi l (%bit2, @loop_start), (%bit_next, @after)
    %res   = phi l (%res, @loop_start), (%res_next, @after)
    %ncur  = phi l (%ncur2, @loop_start), (%n_next, @after)

    %tmp   = addl %res, %bit
    %lt    = cultl %ncur, %tmp
    %ge    = notw %lt
    jnz %ge, @take

@skip:
    %res_next = shrl %res, const l 1
    %n_next   = %ncur
    jmp @after

@take:
    %n_next   = subl %ncur, %tmp
    %res_shift = shrl %res, const l 1
    %res_next  = addl %res_shift, %bit
    jmp @after

@after:
    %bit_next = shrl %bit, const l 2
    %done = ceql %bit_next, const l 0
    jnz %done, @exit
    jmp @loop

@exit:
    ret %res
}
