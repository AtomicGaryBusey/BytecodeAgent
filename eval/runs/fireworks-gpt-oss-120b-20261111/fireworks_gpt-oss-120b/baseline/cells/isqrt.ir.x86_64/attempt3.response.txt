export function l $isqrt(l n) {
@entry:
    %low0 = const l 0
    %high0 = const l 0xFFFFFFFF
    %ans0 = const l 0
    jmp @loop

@loop:
    %low = phi l (%low0, @entry), (%low_big, @mid_big), (%low_ok, @mid_ok), (%low_z, @mid_zero)
    %high = phi l (%high0, @entry), (%high_big, @mid_big), (%high_ok, @mid_ok), (%high_z, @mid_zero)
    %ans = phi l (%ans0, @entry), (%ans_big, @mid_big), (%ans_ok, @mid_ok), (%ans_z, @mid_zero)

    %cond = culgtl %low, %high
    jnz %cond, @exit

    %sum = addl %low, %high
    %mid = shrl %sum, const l 1

    %mid_is_zero = ceql %mid, const l 0
    jnz %mid_is_zero, @mid_zero

    %div = udivl %n, %mid
    %mid_gt = culgtl %mid, %div
    jnz %mid_gt, @mid_big

@mid_ok:
    %ans_ok = %mid
    %low_ok = addl %mid, const l 1
    %high_ok = %high
    jmp @loop

@mid_big:
    %ans_big = %ans
    %high_big = subl %mid, const l 1
    %low_big = %low
    jmp @loop

@mid_zero:
    %ans_z = const l 0
    %low_z = const l 1
    %high_z = %high
    jmp @loop

@exit:
    ret %ans
}
