export function l $fib(l %n) {
  %zero = constl 0
  %one = constl 1
  %cond0 = ceql %n, %zero
  jnz %cond0, @ret0
  %cond1 = ceql %n, %one
  jnz %cond1, @ret1
  %i0 = constl 2
  %a0 = constl 0
  %b0 = constl 1
  jmp @loop, %i0, %a0, %b0

@ret0:
  ret %zero

@ret1:
  ret %one

@loop(l %i, l %a, l %b) {
  %gt = csltl %n, %i
  jnz %gt, @done, %b
  %next = addl %a, %b
  %i1 = addl %i, %one
  jmp @loop, %i1, %b, %next
}

@done(l %result) {
  ret %result
}
}
