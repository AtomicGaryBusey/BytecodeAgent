memmove_i64:
    test rdx, rdx
    jz .Lret0
    mov r8, rdx
    shl r8, 3
    mov r9, rsi
    add r9, r8
    cmp rdi, rsi
    jb .Lforward
    cmp rdi, r9
    jae .Lforward
.Lbackward:
    std
    lea rsi, [rsi + r8 - 8]
    lea rdi, [rdi + r8 - 8]
    mov rcx, rdx
    rep movsq
    cld
    xor eax, eax
    ret
.Lforward:
    cld
    mov rcx, rdx
    rep movsq
    xor eax, eax
    ret
.Lret0:
    xor eax, eax
    ret
