export function l $count_ugt(l %a, l %n, l %t) {
@entry:
    %cnt = constl 0
    %i = constl 0
    %one = constl 1
    %eight = constl 8
@loop:
    %cond = csltl %i %n
    jz %cond, @done
    %off = mul %i %eight
    %ptr = add %a %off
    %val = loadl %ptr
    %gt = cugtl %val %t
    jnz %gt, @inc
@post:
    %i = add %i %one
    jmp @loop
@inc:
    %cnt = add %cnt %one
    jmp @post
@done:
    ret %cnt
}
