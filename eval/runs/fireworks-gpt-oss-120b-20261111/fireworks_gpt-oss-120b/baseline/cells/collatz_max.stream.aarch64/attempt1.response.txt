collatz_max:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x9, #0                // max = 0
    mov x10, #1               // k = 1
.Louter_loop:
    cmp x10, x0
    bgt .Ldone_outer
    mov x11, #0               // steps = 0
    mov x12, x10              // x = k
.Linner_loop:
    cmp x12, #1
    beq .Linner_done
    and x13, x12, #1          // test low bit
    cbz x13, .Leven
    add x13, x12, x12, LSL #1 // x*3
    add x12, x13, #1          // x = 3*x+1
    b .Lstep_inc
.Leven:
    lsr x12, x12, #1          // x = x/2
.Lstep_inc:
    add x11, x11, #1          // steps++
    b .Linner_loop
.Linner_done:
    cmp x11, x9
    ble .Lskip_max_update
    mov x9, x11               // max = steps
.Lskip_max_update:
    add x10, x10, #1          // k++
    b .Louter_loop
.Ldone_outer:
    mov x0, x9                // return max
    ldp x29, x30, [sp], #16
    ret
