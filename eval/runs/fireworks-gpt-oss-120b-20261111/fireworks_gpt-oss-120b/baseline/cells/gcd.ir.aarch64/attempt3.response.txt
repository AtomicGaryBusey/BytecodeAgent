export function l $gcd(l %a, l %b) {
@entry:
    %zero = const 0
    %b_is_zero = ceql %b, %zero
    jnz %b_is_zero, @ret_a
    %a_is_zero = ceql %a, %zero
    jnz %a_is_zero, @ret_b
    jmp @loop(%a, %b)

@loop(%x, %y):
    %y_is_zero = ceql %y, %zero
    jnz %y_is_zero, @ret_x(%x)
    %r = mod %x, %y
    jmp @loop(%y, %r)

@ret_a:
    ret %a

@ret_b:
    ret %b

@ret_x(%res):
    ret %res
}
