mp_mul:
    // x0=a, x1=na, x2=b, x3=nb, x4=r
    // total_len = na + nb
    add     x5, x1, x3                // x5 = total_len

    // zero r[0..total_len-1]
    mov     x6, #0                    // offset = 0
    cbz     x5, .Lzero_done
.Lzero_loop:
    str     xzr, [x4, x6]
    add     x6, x6, #8
    subs    x5, x5, #1
    bne     .Lzero_loop
.Lzero_done:
    // restore total_len in x5
    add     x5, x1, x3

    // outer loop i = 0 .. na-1
    mov     x6, #0                    // i = 0
    cbz     x1, .Louter_done
.Louter:
    // load a[i]
    ldr     x11, [x0, x6, lsl #3]     // a_i in x11
    // r_base = r + i*8
    add     x12, x4, x6, lsl #3       // base pointer for this i
    // carry = 0
    mov     x10, xzr                  // carry

    // inner loop j = 0 .. nb-1
    mov     x9, #0                    // j = 0
    cbz     x3, .Linner_done
.Linner:
    // load b[j]
    ldr     x13, [x2, x9, lsl #3]     // b_j in x13
    // multiply a_i * b_j
    mul     x14, x11, x13             // lo product
    umulh   x15, x11, x13             // hi product

    // sum = r[i+j] + lo
    ldr     x16, [x12, x9, lsl #3]    // r_ij
    adds    x16, x16, x14            // r_ij + lo, sets C flag
    // add carry
    adc     x17, xzr, xzr             // x17 = 0 + C flag (carry from previous adds)
    adds    x16, x16, x10            // add previous carry
    adc     x18, x15, xzr            // hi + overflow from low adds
    adds    x18, x18, x17            // add carry flag from first adds
    // store result low
    str     x16, [x12, x9, lsl #3]
    // new carry for next iteration
    mov     x10, x18

    add     x9, x9, #1
    cmp     x9, x3
    bne     .Linner
.Linner_done:
    // propagate remaining carry to r[i+nb] upwards
    // ptr = r + (i+nb)*8
    add     x12, x12, x3, lsl #3      // x12 points to r[i+nb]
.Lprop:
    // add carry to current limb
    ldr     x16, [x12]
    adds    x16, x16, x10
    str     x16, [x12]
    // if overflow, continue upward
    bcs     .Lprop_next
    b       .Louter_next
.Lprop_next:
    // set carry = 1 and move to next limb
    mov     x10, #1
    add     x12, x12, #8
    b       .Lprop

.Louter_next:
    add     x6, x6, #1
    cmp     x6, x1
    bne     .Louter
.Louter_done:

    // find most significant non‑zero limb
    cbz     x5, .Lret_zero            // total_len == 0
    sub     x6, x5, #1                // idx = total_len-1
    add     x12, x4, x6, lsl #3       // ptr = r + idx*8
.Lfind_loop:
    ldr     x10, [x12]
    cbnz    x10, .Lfound_nonzero
    cbz     x6, .Lret_zero
    sub     x6, x6, #1
    sub     x12, x12, #8
    b       .Lfind_loop
.Lfound_nonzero:
    add     x0, x6, #1                // return idx+1
    b       .Lret
.Lret_zero:
    mov     x0, #0
.Lret:
    ret
