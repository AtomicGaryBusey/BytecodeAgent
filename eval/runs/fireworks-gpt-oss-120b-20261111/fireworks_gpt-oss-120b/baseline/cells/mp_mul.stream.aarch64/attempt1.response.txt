mp_mul:
    // x0=a, x1=na, x2=b, x3=nb, x4=r
    add     x5, x1, x3                // total_len = na + nb
    // zero r[0..total_len-1]
    mov     x6, x5                    // counter
    mov     x9, x4                    // dest pointer
.Lzero_loop:
    cbz     x6, .Lzero_done
    str     xzr, [x9], #8
    sub     x6, x6, #1
    b       .Lzero_loop
.Lzero_done:
    // outer loop i = 0 .. na-1
    mov     x6, #0                    // i
.Louter:
    cmp     x6, x1
    beq     .Louter_done
    // load a[i]
    ldr     x9, [x0, x6, lsl #3]      // a_i
    // set inner loop base pointer for r[i+*]
    add     x10, x4, x6, lsl #3       // r_base = r + i*8
    mov     x7, #0                    // j
    mov     x8, #0                    // carry = 0
.Linner:
    cmp     x7, x3
    beq     .Linner_done
    // load b[j]
    ldr     x11, [x2, x7, lsl #3]     // b_j
    // multiply a_i * b_j
    mul     x13, x9, x11              // low product
    umulh   x14, x9, x11              // high product
    // load current r[i+j]
    ldr     x15, [x10, x7, lsl #3]    // r_ij
    // add low part
    adds    x15, x15, x13
    // add high part + carry + carry flag
    adc     x14, x14, x8
    // store low result back
    str     x15, [x10, x7, lsl #3]
    // update carry for next iteration
    mov     x8, x14
    add     x7, x7, #1
    b       .Linner
.Linner_done:
    // store final carry to r[i+nb]
    add     x15, x4, x6, lsl #3
    add     x15, x15, x3, lsl #3
    str     x8, [x15]
    // increment i
    add     x6, x6, #1
    b       .Louter
.Louter_done:
    // find most significant non‑zero limb
    cbz     x5, .Lret_zero            // total_len == 0
    sub     x6, x5, #1                // idx = total_len-1
    add     x9, x4, x6, lsl #3        // ptr = r + idx*8
.Lfind_loop:
    ldr     x10, [x9]
    cbnz    x10, .Lfound_nonzero
    cbz     x6, .Lret_zero
    sub     x6, x6, #1
    sub     x9, x9, #8
    b       .Lfind_loop
.Lfound_nonzero:
    add     x0, x6, #1                // return idx+1
    b       .Lret
.Lret_zero:
    mov     x0, #0
.Lret:
    ret
