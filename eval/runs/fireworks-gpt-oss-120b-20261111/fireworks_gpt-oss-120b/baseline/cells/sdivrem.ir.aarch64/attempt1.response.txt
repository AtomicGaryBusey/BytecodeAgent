export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bz = ceql %b, 0
    jnz %bz, @bzero

    %amin = ceql %a, -9223372036854775808
    %bneg1 = ceql %b, -1
    %ov = and %amin, %bneg1
    jnz %ov, @overflow

    %q = div %a, %b
    %r = rem %a, %b
    storel %r, %rem
    ret %q

@bzero
    %q = 0
    %r = %a
    storel %r, %rem
    ret %q

@overflow
    %q = -9223372036854775808
    %r = 0
    storel %r, %rem
    ret %q
}
