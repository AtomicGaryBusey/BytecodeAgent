export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bz = ceql %b, 0l
    jnz %bz, @bzero

    %amin = ceql %a, -9223372036854775808l
    %bneg1 = ceql %b, -1l
    %ov = and %amin, %bneg1
    jnz %ov, @overflow

    %q = div %a, %b
    %r = rem %a, %b
    storel %r, %rem
    ret %q

@bzero
    %q = 0l
    %r = %a
    storel %r, %rem
    ret %q

@overflow
    %q = -9223372036854775808l
    %r = 0l
    storel %r, %rem
    ret %q
}
