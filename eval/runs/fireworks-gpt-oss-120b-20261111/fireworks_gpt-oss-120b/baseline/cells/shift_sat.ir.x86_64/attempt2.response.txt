export function l $shift_sat(l %x, l %n) {
@entry
    %c64 = const 64
    %cond_lt64 = csltl %n, %c64
    jnz %cond_lt64, @lt64, @ge64

@lt64
    %lshift = shl %x, %n
    %rshift = ashr %x, %n
    %sum = add %lshift, %rshift
    jmp @exit(%sum)

@ge64
    %c0 = const 0
    %isneg = csltl %x, %c0
    jnz %isneg, @neg, @nonneg

@neg
    %cneg1 = const -1
    jmp @exit(%cneg1)

@nonneg
    %c0b = const 0
    jmp @exit(%c0b)

@exit(l %res)
    ret %res
}
