export function l $shift_sat(l %x, l %n) {
@entry
    %cond_lt64 = cslt %n, 64
    jnz %cond_lt64, @lt64, @ge64

@lt64
    %lsh = shl %x, %n
    %rsh = ashr %x, %n
    %sum = add %lsh, %rsh
    ret %sum

@ge64
    %cond_neg = cslt %x, 0
    jnz %cond_neg, @neg, @nonneg

@neg
    %minus1 = sub 0, 1
    ret %minus1

@nonneg
    %zero = sub 0, 0
    ret %zero
}
