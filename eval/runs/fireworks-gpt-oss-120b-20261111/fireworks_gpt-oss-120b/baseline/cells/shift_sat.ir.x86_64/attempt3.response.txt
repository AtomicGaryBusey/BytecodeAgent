export function l $shift_sat(l %x, l %n) {
@entry
    %c64 = imm 64
    %cond_lt64 = cslt %n, %c64
    jnz %cond_lt64, @lt64, @ge64

@lt64
    %lshift = shl %x, %n
    %rshift = ashr %x, %n
    %sum = add %lshift, %rshift
    ret %sum

@ge64
    %c0 = imm 0
    %isneg = cslt %x, %c0
    jnz %isneg, @neg, @nonneg

@neg
    %cneg1 = imm -1
    ret %cneg1

@nonneg
    %c0b = imm 0
    ret %c0b
}
