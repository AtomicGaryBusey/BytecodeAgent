export function l $shift_sat(l %x, l %n) {
@entry
    %cond_lt64 = csltl %n, const 64
    jnz %cond_lt64, @lt64, @ge64

@lt64
    %lshift = shl %x, %n
    %rshift = ashr %x, %n
    %sum = add %lshift, %rshift
    jmp @exit(%sum)

@ge64
    %isneg = csltl %x, const 0
    jnz %isneg, @neg, @nonneg

@neg
    %neg_one = const -1
    jmp @exit(%neg_one)

@nonneg
    %zero = const 0
    jmp @exit(%zero)

@exit(l %res)
    ret %res
}
