fib:
    cbz x0, .Lreturn0
    cmp x0, #1
    beq .Lreturn1
    mov x1, #0
    mov x2, #1
    mov x3, #2
.Lloop:
    cmp x3, x0
    bgt .Lend
    add x4, x1, x2
    mov x1, x2
    mov x2, x4
    add x3, x3, #1
    b .Lloop
.Lend:
    mov x0, x2
    ret
.Lreturn0:
    mov x0, #0
    ret
.Lreturn1:
    mov x0, #1
    ret
