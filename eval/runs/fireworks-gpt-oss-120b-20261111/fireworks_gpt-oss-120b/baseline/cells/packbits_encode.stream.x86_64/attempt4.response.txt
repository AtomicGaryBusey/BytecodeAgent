packbits_encode:
    push r12
    push r13
    xor r8, r8
    xor r9, r9
.main_loop:
    cmp r8, rsi
    jae .done
    mov al, byte ptr [rdi + r8]
    mov r12b, al
    xor r10, r10
    inc r10
    mov r11, r8
.run_check:
    cmp r10, 128
    jae .run_done
    inc r11
    cmp r11, rsi
    jae .run_done
    cmp al, byte ptr [rdi + r11]
    jne .run_done
    inc r10
    jmp .run_check
.run_done:
    cmp r10, 3
    jl .literal_start
    lea rax, [r9 + 2]
    cmp rax, rcx
    ja .fail
    mov bl, r10b
    mov al, 1
    sub al, bl
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r12b
    add r9, 2
    add r8, r10
    jmp .main_loop
.literal_start:
    mov r13, r8
    xor r10, r10
.literal_scan:
    cmp r13, rsi
    jae .literal_end
    mov rax, r13
    sub rax, r8
    cmp rax, 128
    jae .literal_end
    mov al, byte ptr [rdi + r13]
    mov r11, r13
    mov r14, 1
.run_check_lit:
    cmp r14, 3
    jae .run_ge3_lit
    inc r11
    cmp r11, rsi
    jae .run_lt3_lit
    cmp al, byte ptr [rdi + r11]
    jne .run_lt3_lit
    inc r14
    jmp .run_check_lit
.run_ge3_lit:
    jmp .literal_end
.run_lt3_lit:
    inc r13
    jmp .literal_scan
.literal_end:
    mov rax, r13
    sub rax, r8
    cmp rax, 0
    je .done
    lea rbx, [r9 + rax + 1]
    cmp rbx, rcx
    ja .fail
    mov bl, al
    dec bl
    mov byte ptr [rdx + r9], bl
    mov r14, rax
    mov r11, r8
    mov r12, r9
.copy_loop:
    test r14, r14
    jz .copy_done
    mov al, byte ptr [rdi + r11]
    mov byte ptr [rdx + r12 + 1], al
    inc r11
    inc r12
    dec r14
    jmp .copy_loop
.copy_done:
    add r9, rax
    inc r9
    add r8, rax
    jmp .main_loop
.fail:
    mov rax, -1
    pop r13
    pop r12
    ret
.done:
    mov rax, r9
    pop r13
    pop r12
    ret
