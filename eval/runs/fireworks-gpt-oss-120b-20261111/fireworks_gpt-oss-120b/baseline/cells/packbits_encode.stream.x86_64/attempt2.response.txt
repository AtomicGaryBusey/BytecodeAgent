packbits_encode:
    xor r8, r8                # i = 0
    xor r9, r9                # out_written = 0
.main_loop:
    cmp r8, rsi               # i vs n
    jae .done
    mov al, byte ptr [rdi + r8]   # first byte of run / literal
    mov r10, r8
    mov r11, 1                # run length
.run_check:
    cmp r11, 128
    jae .run_done
    inc r10
    cmp r10, rsi
    jae .run_done
    cmp al, byte ptr [rdi + r10]
    jne .run_done
    inc r11
    jmp .run_check
.run_done:
    cmp r11, 3
    jl .literal_start
    # repeat packet
    lea rax, [r9 + 2]
    cmp rax, rcx
    ja .fail
    mov r10d, 0x100
    sub r10d, r11d            # header byte in r10b
    mov byte ptr [rdx + r9], r10b
    mov byte ptr [rdx + r9 + 1], al
    add r8, r11               # advance source index
    add r9, 2                 # advance output index
    jmp .main_loop
.literal_start:
    mov r10, r8               # j = i
    xor r11, r11              # lit_len = 0
.literal_loop:
    cmp r10, rsi
    jae .literal_end
    cmp r11, 128
    jae .literal_end
    mov rax, r10
    add rax, 2
    cmp rax, rsi
    jae .include_byte        # not enough bytes for a run of 3
    mov bl, byte ptr [rdi + r10]
    cmp bl, byte ptr [rdi + r10 + 1]
    jne .include_byte
    cmp bl, byte ptr [rdi + r10 + 2]
    jne .include_byte
    jmp .literal_end         # run of 3 found, stop literal
.include_byte:
    inc r11                   # lit_len++
    inc r10                   # j++
    jmp .literal_loop
.literal_end:
    test r11, r11
    jz .main_loop             # nothing to emit, continue
    lea rax, [r9 + r11 + 1]
    cmp rax, rcx
    ja .fail
    mov r10d, r11d
    dec r10d                  # header = lit_len-1
    mov byte ptr [rdx + r9], r10b
    # copy literal bytes
.copy_loop:
    test r11, r11
    jz .copy_done
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    inc r8
    inc r9
    dec r11
    jmp .copy_loop
.copy_done:
    inc r9                    # account for header byte already written
    jmp .main_loop
.fail:
    mov rax, -1
    ret
.done:
    mov rax, r9
    ret
