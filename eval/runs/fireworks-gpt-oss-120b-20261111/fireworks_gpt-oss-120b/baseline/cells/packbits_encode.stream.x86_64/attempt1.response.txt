packbits_encode:
    xor r8, r8
    xor r9, r9
.Lmain_loop:
    cmp r8, rsi
    jae .Ldone
    mov r14b, byte ptr [rdi + r8]
    mov r10, r8
    mov r11, 1
.Lrun_repeat:
    cmp r11, 128
    jae .Lrun_repeat_done
    inc r10
    cmp r10, rsi
    jae .Lrun_repeat_done
    cmp r14b, byte ptr [rdi + r10]
    jne .Lrun_repeat_done
    inc r11
    jmp .Lrun_repeat
.Lrun_repeat_done:
    cmp r11, 3
    jl .Lstart_literal
    mov rax, r9
    add rax, 2
    cmp rax, rcx
    ja .Lfail
    mov eax, 0x101
    sub eax, r11d
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r14b
    add r8, r11
    add r9, 2
    jmp .Lmain_loop
.Lstart_literal:
    mov r10, r8
    xor r13, r13
.Lliteral_scan:
    cmp r10, rsi
    jae .Lliteral_end
    cmp r13, 128
    jae .Lliteral_end
    mov rax, r10
    add rax, 2
    cmp rax, rsi
    jae .Lno_run3
    mov al, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Lno_run3
    cmp al, byte ptr [rdi + r10 + 2]
    jne .Lno_run3
    jmp .Lliteral_end
.Lno_run3:
    inc r13
    inc r10
    jmp .Lliteral_scan
.Lliteral_end:
    test r13, r13
    jz .Lmain_loop
    mov rax, r9
    add rax, r13
    inc rax
    cmp rax, rcx
    ja .Lfail
    mov r10d, r13d
    dec r10d
    mov byte ptr [rdx + r9], r10b
    inc r9
    mov rax, r13
.Lcopy_loop:
    test rax, rax
    jz .Lcopy_done
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    dec rax
    jmp .Lcopy_loop
.Lcopy_done:
    jmp .Lmain_loop
.Lfail:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
