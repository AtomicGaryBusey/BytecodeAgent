packbits_encode:
    push r12
    push r13
    xor r8, r8                    # i = 0
    xor r9, r9                    # out_pos = 0
.main_loop:
    cmp r8, rsi
    jae .done
    # compute run length starting at i
    mov al, byte ptr [rdi + r8]   # first byte in al
    mov r10, 1                    # run_len = 1
    mov r11, r8                   # pos = i
.run_check:
    cmp r10, 128
    jae .run_done
    inc r11
    cmp r11, rsi
    jae .run_done
    cmp al, byte ptr [rdi + r11]
    jne .run_done
    inc r10
    jmp .run_check
.run_done:
    cmp r10, 3
    jl .literal_start
    # repeat packet
    lea rax, [r9 + 2]
    cmp rax, rcx
    ja .fail
    mov al, 0x101
