export function q $count_ugt(q a, q n, q t) {
@entry:
    %i = copy 0
    %cnt = copy 0
@loop:
    %cond = csltq %i, n
    jnz %cond, @body
    jmp @end
@body:
    %off = mul %i, 8
    %p = add a, %off
    %val = loadq %p
    %gt = cultq t, %val
    %inc = sel %gt, 1, 0
    %cnt = add %cnt, %inc
    %i = add %i, 1
    jmp @loop
@end:
    ret %cnt
}
