export function l $count_ugt(l a, l n, l t) {
@entry:
    %i = copy 0
    %cnt = copy 0
@loop:
    %cond = ultl %i, n
    jnz %cond, @body
    jmp @end
@body:
    %off = mul %i, 8
    %p = add a, %off
    %val = loadl %p
    %gt = ultl t, %val
    %inc = sel %gt, 1, 0
    %cnt = add %cnt, %inc
    %i = add %i, 1
    jmp @loop
@end:
    ret %cnt
}
