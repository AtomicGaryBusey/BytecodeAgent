export function l $popcount(l)
@entry
    %v0 = %0
    %cnt0 = constl 0
    %zero = constl 0
    jmp @loop(%cnt0, %v0, %zero)

@loop(%cnt:l, %v:l, %zero:l)
    %cond = cneql %v, %zero
    jnz %cond, @body(%cnt, %v, %zero)
    jmp @exit(%cnt)

@body(%cnt:l, %v:l, %zero:l)
    %cnt1 = addl %cnt, 1
    %t = subl %v, 1
    %v1 = andl %v, %t
    jmp @loop(%cnt1, %v1, %zero)

@exit(%cnt:l)
    ret %cnt
