binary_search:
    push rbx
    mov rbx, 0
    mov rcx, rsi
.Lloop:
    cmp rbx, rcx
    jae .Lnotfound
    mov r8, rcx
    sub r8, rbx
    shr r8, 1
    add r8, rbx
    mov rax, qword ptr [rdi + r8*8]
    cmp rdx, rax
    je .Lfound
    jl .Lless
    lea rbx, [r8 + 1]
    jmp .Lloop
.Lless:
    mov rcx, r8
    jmp .Lloop
.Lfound:
    mov rax, r8
    pop rbx
    ret
.Lnotfound:
    mov rax, -1
    pop rbx
    ret
