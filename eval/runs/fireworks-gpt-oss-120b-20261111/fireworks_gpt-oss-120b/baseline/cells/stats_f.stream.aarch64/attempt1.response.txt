    mov x21, x1          // n
    mov x27, x2          // f

    mov x23, x0          // min
    mov x24, x0          // max
    mov x25, x0          // sum

    add x26, x26, #1
.Linit_done:
    mov x22, #1          // i = 1

    bge .Lafter_loop
    ldr x0, [x19, x22, LSL #3]
    blr x27
    add x25, x25, x0
    cmp x0, x23
    csel x23, x0, x23, lt
    cmp x0, x24

    stp x23, x24, [x20]
    str x25, [x20, #16]
