export function l $mat_pow(l a, l n, l k, l out, l tmp) {
    @entry
        %zero = constl 0
        %one  = constl 1
        %two  = constl 2
        %nn   = mul %n, %n                // n*n
        %total = mul %nn, %two           // 2*n*n

        // initialize result = identity matrix in out
        jmp @id_i(%zero)

    // identity matrix loops
    @id_i(%i: l)
        %c_i = csltl %i, %n
        jnz %c_i, @id_j(%i, %zero), @copy_base

    @id_j(%i: l, %j: l)
        %c_j = csltl %j, %n
        jnz %c_j, @id_set(%i, %j), @inc_i(%i)

    @id_set(%i: l, %j: l)
        %off_i = mul %i, %n
        %off   = add %off_i, %j
        %addr  = add %out, %off
        %eq    = ceql %i, %j
        %val   = sel %eq, %one, %zero
        storel %val, *%addr
        %j_nxt = add %j, %one
        jmp @id_j(%i, %j_nxt)

    @inc_i(%i: l)
        %i_nxt = add %i, %one
        jmp @id_i(%i_nxt)

    // copy original matrix a into tmp (first nn slots)
    @copy_base
        jmp @copy_base_loop(%zero)

    @copy_base_loop(%c: l)
        %cnd = csltl %c, %nn
        jnz %cnd, @copy_base_body(%c), @while
    @copy_base_body(%c: l)
        %src = add %a, %c
        %val = loadl *%src
        %dst = add %tmp, %c
        storel %val, *%dst
        %c_nxt = add %c, %one
        jmp @copy_base_loop(%c_nxt)

    // main exponentiation loop
    @while
        %c_k = csltl %zero, %k
        jnz %c_k, @while_body, @final_zero_tmp

    @while_body
        // if (k & 1) multiply result *= base
        %bit  = and %k, %one
        %odd  = csltl %zero, %bit
        jnz %odd, @mult_res, @skip_mult

    @mult_res
        %prod = add %tmp, %nn          // product location
        %X = %out
        %Y = %tmp
        %Dst = %prod
        jmp @mul_i(%zero)

        // copy product to out
        %src_base = %prod
        jmp @copy_to_out(%zero)

    @skip_mult
        nop

        // square base: base = base * base
        %prod2 = add %tmp, %nn
        %X = %tmp
        %Y = %tmp
        %Dst = %prod2
        jmp @mul_i(%zero)

        // copy product to base (tmp)
        %src_base = %prod2
        jmp @copy_to_tmp(%zero)

        // shift k >>= 1
        %k = lshr %k, %one
        jmp @while

    // copy product to out loop
    @copy_to_out(%c: l)
        %cnd = csltl %c, %nn
        jnz %cnd, @copy_to_out_body(%c), @while
    @copy_to_out_body(%c: l)
        %src = add %src_base, %c
        %val = loadl *%src
        %dst = add %out, %c
        storel %val, *%dst
        %c_nxt = add %c, %one
        jmp @copy_to_out(%c_nxt)

    // copy product to base (tmp) loop
    @copy_to_tmp(%c: l)
        %cnd = csltl %c, %nn
        jnz %cnd, @copy_to_tmp_body(%c), @while
    @copy_to_tmp_body(%c: l)
        %src = add %src_base, %c
        %val = loadl *%src
        %dst = add %tmp, %c
        storel %val, *%dst
        %c_nxt = add %c, %one
        jmp @copy_to_tmp(%c_nxt)

    // matrix multiplication: X * Y -> Dst
    @mul_i(%i: l)
        %c_i = csltl %i, %n
        jnz %c_i, @mul_j(%i, %zero), @after_mul

    @mul_j(%i: l, %j: l)
        %c_j = csltl %j, %n
        jnz %c_j, @mul_t(%i, %j, %zero, %zero), @inc_i(%i)

    @mul_t(%i: l, %j: l, %t: l, %sum: l)
        %c_t = csltl %t, %n
        jnz %c_t, @mul_acc(%i, %j, %t, %sum), @store_mul(%i, %j, %sum)

    @mul_acc(%i: l, %j: l, %t: l, %sum: l)
        %x_off = mul %i, %n
        %x_idx = add %x_off, %t
        %x_addr = add %X, %x_idx
        %x_val = loadl *%x_addr
        %y_off = mul %t, %n
        %y_idx = add %y_off, %j
        %y_addr = add %Y, %y_idx
        %y_val = loadl *%y_addr
        %prod = mul %x_val, %y_val
        %sum2 = add %sum, %prod
        %t_nxt = add %t, %one
        jmp @mul_t(%i, %j, %t_nxt, %sum2)

    @store_mul(%i: l, %j: l, %sum: l)
        %off_i = mul %i, %n
        %off   = add %off_i, %j
        %dst_addr = add %Dst, %off
        storel %sum, *%dst_addr
        %j_nxt = add %j, %one
        jmp @mul_j(%i, %j_nxt)

    @inc_i(%i: l)
        %i_nxt = add %i, %one
        jmp @mul_i(%i_nxt)

    @after_mul
        nop

    // zero out tmp (2*nn elements)
    @final_zero_tmp
        jmp @zero_tmp_loop(%zero)

    @zero_tmp_loop(%c: l)
        %cnd = csltl %c, %total
        jnz %cnd, @zero_tmp_body(%c), @ret

    @zero_tmp_body(%c: l)
        %dst = add %tmp, %c
        storel %zero, *%dst
        %c_nxt = add %c, %one
        jmp @zero_tmp_loop(%c_nxt)

    @ret
        %r = constl 0
        ret %r
}
