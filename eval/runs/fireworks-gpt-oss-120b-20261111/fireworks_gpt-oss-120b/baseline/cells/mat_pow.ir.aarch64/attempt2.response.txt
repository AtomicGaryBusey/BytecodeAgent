export function l mat_pow(l a, l n, l k, l out, l tmp) {
    @entry
    %zero = constl 0
    %one = constl 1
    %nn = mul %n, %n
    jmp @init_i(%zero)

    @init_i(%i: l)
        %cond_i = csltl %i, %n
        jnz %cond_i, @init_j(%i, %zero), @copy_a

    @init_j(%i: l, %j: l)
        %cond_j = csltl %j, %n
        jnz %cond_j, @init_set(%i, %j), @inc_i(%i)

    @init_set(%i: l, %j: l)
        %off_i = mul %i, %n
        %off = add %off_i, %j
        %addr = add %out, %off
        %eq = ceql %i, %j
        %val = sel %eq, %one, %zero
        storel %val, *%addr
        %j_next = add %j, %one
        jmp @init_j(%i, %j_next)

    @inc_i(%i: l)
        %i_next = add %i, %one
        jmp @init_i(%i_next)

    @copy_a
        jmp @copy_loop(%zero)

    @copy_loop(%cnt: l)
        %cond = csltl %cnt, %nn
        jnz %cond, @copy_body(%cnt), @pow_setup

    @copy_body(%cnt: l)
        %src = add %a, %cnt
        %val = loadl *%src
        %dst = add %tmp, %cnt
        storel %val, *%dst
        %cnt_next = add %cnt, %one
        jmp @copy_loop(%cnt_next)

    @pow_setup
        jmp @pow_loop(%k)

    @pow_loop(%k_cur: l)
        %condk = csltl %zero, %k_cur
        jnz %condk, @pow_body(%k_cur), @done

    @pow_body(%k_cur: l)
        %bit = and %k_cur, %one
        %odd = csltl %zero, %bit
        jnz %odd, @mul_res(%k_cur), @skip_mul_res(%k_cur)

    @skip_mul_res(%k_cur: l)
        jmp @square_base(%k_cur)

    @mul_res(%k_cur: l)
        %tmp1 = add %tmp, %nn
        jmp @mul_res_i(%zero, %out, %tmp, %tmp1)

    @mul_res_i(%i: l, %X: l, %Y: l, %Dst: l)
        %cond_i = csltl %i, %n
        jnz %cond_i, @mul_res_j(%i, %zero, %X, %Y, %Dst), @after_mul_res(%k_cur)

    @mul_res_j(%i: l, %j: l, %X: l, %Y: l, %Dst: l)
        %cond_j = csltl %j, %n
        jnz %cond_j, @mul_res_compute(%i, %j, %X, %Y, %Dst), @inc_mul_res_i(%i, %X, %Y, %Dst)

    @mul_res_compute(%i: l, %j: l, %X: l, %Y: l, %Dst: l)
        %sum = constl 0
        jmp @mul_res_t(%i, %j, %sum, %zero, %X, %Y, %Dst)

    @mul_res_t(%i: l, %j: l, %sum: l, %t: l, %X: l, %Y: l, %Dst: l)
        %cond_t = csltl %t, %n
        jnz %cond_t, @mul_res_acc(%i, %j, %sum, %t, %X, %Y, %Dst), @store_mul_res(%i, %j, %sum, %X, %Y, %Dst)

    @mul_res_acc(%i: l, %j: l, %sum: l, %t: l, %X: l, %Y: l, %Dst: l)
        %x_off = mul %i, %n
        %x_idx = add %x_off, %t
        %x_addr = add %X, %x_idx
        %x_val = loadl *%x_addr
        %y_off = mul %t, %n
        %y_idx = add %y_off, %j
        %y_addr = add %Y, %y_idx
        %y_val = loadl *%y_addr
        %prod = mul %x_val, %y_val
        %sum2 = add %sum, %prod
        %t_next = add %t, %one
        jmp @mul_res_t(%i, %j, %sum2, %t_next, %X, %Y, %Dst)

    @store_mul_res(%i: l, %j: l, %sum: l, %X: l, %Y: l, %Dst: l)
        %off_i = mul %i, %n
        %off = add %off_i, %j
        %dst_addr = add %Dst, %off
        storel %sum, *%dst_addr
        %j_next = add %j, %one
        jmp @mul_res_j(%i, %j_next, %X, %Y, %Dst)

    @inc_mul_res_i(%i: l, %X: l, %Y: l, %Dst: l)
        %i_next = add %i, %one
        jmp @mul_res_i(%i_next, %X, %Y, %Dst)

    @after_mul_res(%k_cur: l)
        %tmp1 = add %tmp, %nn
        jmp @copy_back(%zero, %tmp1)

    @copy_back(%cnt: l, %src_base: l)
        %cond = csltl %cnt, %nn
        jnz %cond, @copy_back_body(%cnt, %src_base), @square_base(%k_cur)

    @copy_back_body(%cnt: l, %src_base: l)
        %src = add %src_base, %cnt
        %val = loadl *%src
        %dst = add %out, %cnt
        storel %val, *%dst
        %cnt_next = add %cnt, %one
        jmp @copy_back(%cnt_next, %src_base)

    @square_base(%k_cur: l)
        %tmp1 = add %tmp, %nn
        jmp @square_i(%zero, %tmp, %tmp, %tmp1)

    @square_i(%i: l, %X: l, %Y: l, %Dst: l)
        %cond_i = csltl %i, %n
        jnz %cond_i, @square_j(%i, %zero, %X, %Y, %Dst), @after_square(%k_cur)

    @square_j(%i: l, %j: l, %X: l, %Y: l, %Dst: l)
        %cond_j = csltl %j, %n
        jnz %cond_j, @square_compute(%i, %j, %X, %Y, %Dst), @inc_square_i(%i, %X, %Y, %Dst)

    @square_compute(%i: l, %j: l, %X: l, %Y: l, %Dst: l)
        %sum = constl 0
        jmp @square_t(%i, %j, %sum, %zero, %X, %Y, %Dst)

    @square_t(%i: l, %j: l, %sum: l, %t: l, %X: l, %Y: l, %Dst: l)
        %cond_t = csltl %t, %n
        jnz %cond_t, @square_acc(%i, %j, %sum, %t, %X, %Y, %Dst), @store_square(%i, %j, %sum, %X, %Y, %Dst)

    @square_acc(%i: l, %j: l, %sum: l, %t: l, %X: l, %Y: l, %Dst: l)
        %x_off = mul %i, %n
        %x_idx = add %x_off, %t
        %x_addr = add %X, %x_idx
        %x_val = loadl *%x_addr
        %y_off = mul %t, %n
        %y_idx = add %y_off, %j
        %y_addr = add %Y, %y_idx
        %y_val = loadl *%y_addr
        %prod = mul %x_val, %y_val
        %sum2 = add %sum, %prod
        %t_next = add %t, %one
        jmp @square_t(%i, %j, %sum2, %t_next, %X, %Y, %Dst)

    @store_square(%i: l, %j: l, %sum: l, %X: l, %Y: l, %Dst: l)
        %off_i = mul %i, %n
        %off = add %off_i, %j
        %dst_addr = add %Dst, %off
        storel %sum, *%dst_addr
        %j_next = add %j, %one
        jmp @square_j(%i, %j_next, %X, %Y, %Dst)

    @inc_square_i(%i: l, %X: l, %Y: l, %Dst: l)
        %i_next = add %i, %one
        jmp @square_i(%i_next, %X, %Y, %Dst)

    @after_square(%k_cur: l)
        %tmp1 = add %tmp, %nn
        jmp @copy_base_back(%zero, %tmp1)

    @copy_base_back(%cnt: l, %src_base: l)
        %cond = csltl %cnt, %nn
        jnz %cond, @copy_base_body(%cnt, %src_base), @shift_k(%k_cur)

    @copy_base_body(%cnt: l, %src_base: l)
        %src = add %src_base, %cnt
        %val = loadl *%src
        %dst = add %tmp, %cnt
        storel %val, *%dst
        %cnt_next = add %cnt, %one
        jmp @copy_base_back(%cnt_next, %src_base)

    @shift_k(%k_cur: l)
        %k_next = lshr %k_cur, 1
        jmp @pow_loop(%k_next)

    @done
        %total = mul %nn, %two
        %two = constl 2
        jmp @zero_tmp(%zero, %total)

    @zero_tmp(%cnt: l, %total: l)
        %cond = csltl %cnt, %total
        jnz %cond, @zero_body(%cnt), @ret

    @zero_body(%cnt: l)
        %dst = add %tmp, %cnt
        storel %zero, *%dst
        %cnt_next = add %cnt, %one
        jmp @zero_tmp(%cnt_next, %total)

    @ret
        %ret = constl 0
        ret %ret
}
