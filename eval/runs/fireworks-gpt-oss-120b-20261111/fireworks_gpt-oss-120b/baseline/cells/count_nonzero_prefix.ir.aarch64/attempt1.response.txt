export function l $count_nonzero_prefix(l a, l n) {
@entry:
    %i = copy 0
    jmp @loop

@loop:
    %cmp = csltl %i, n
    jnz %cmp, @load
    ret n

@load:
    %off = mul %i, 8
    %ptr = add a, %off
    %val = loadl %ptr
    %iszero = ceql %val, 0
    jnz %iszero, @ret_i
    %i1 = add %i, 1
    %i = copy %i1
    jmp @loop

@ret_i:
    ret %i
}
