export function l $xxh64(l %p, l %n, l %seed) {
@entry
	%c32 =l 32
	%c8 =l 8
	%c4 =l 4
	%c1 =l 1
	%len =l %n
	%ptr =l %p

	%P1 =l -7046029288634856825
	%P2 =l -4417276706812521889
	%P3 =l 1609587929392839161
	%P4 =l -8796714831421723037
	%P5 =l 2870177450012600261

	%cond_ge32 =w csltl %len, %c32
	jnz %cond_ge32, @no_stripes

	/* n >= 32: init v1..v4 */
	%v1 =l %seed + %P1 + %P2
	%v2 =l %seed + %P2
	%v3 =l %seed
	%v4 =l %seed - %P1

	%rem =l %len

@stripe_loop
	%cond =w csltl %rem, %c32
	jnz %cond, @after_stripes

	%load1 =l loadl %ptr
	%load2 =l loadl %ptr+8
	%load3 =l loadl %ptr+16
	%load4 =l loadl %ptr+24

	/* round(v1, load1) */
	%t =l %load1 * %P2
	%t =l %v1 + %t
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%v1 =l %rot * %P1

	/* round(v2, load2) */
	%t =l %load2 * %P2
	%t =l %v2 + %t
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%v2 =l %rot * %P1

	/* round(v3, load3) */
	%t =l %load3 * %P2
	%t =l %v3 + %t
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%v3 =l %rot * %P1

	/* round(v4, load4) */
	%t =l %load4 * %P2
	%t =l %v4 + %t
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%v4 =l %rot * %P1

	%ptr =l %ptr + 32
	%rem =l %rem - 32
	jmp @stripe_loop

@after_stripes
	/* h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18) */
	%r1 =l %v1 shl 1
	%r1 =l %r1 or %v1 shr 63
	%r2 =l %v2 shl 7
	%r2 =l %r2 or %v2 shr 57
	%r3 =l %v3 shl 12
	%r3 =l %r3 or %v3 shr 52
	%r4 =l %v4 shl 18
	%r4 =l %r4 or %v4 shr 46
	%h =l %r1 + %r2 + %r3 + %r4

	/* merge(h, v1) */
	%t =l %v1 * %P2
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%h =l %h * %P1 + %P4

	/* merge(h, v2) */
	%t =l %v2 * %P2
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%h =l %h * %P1 + %P4

	/* merge(h, v3) */
	%t =l %v3 * %P2
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%h =l %h * %P1 + %P4

	/* merge(h, v4) */
	%t =l %v4 * %P2
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%h =l %h * %P1 + %P4

	jmp @post_stripes

@no_stripes
	%h =l %seed + %P5
	%rem =l %len
	%ptr =l %p

@post_stripes
	%h =l %h + %len

/* process remaining 8-byte blocks */
@while8
	%cond =w csltl %rem, %c8
	jnz %cond, @after8
	%load =l loadl %ptr
	%t =l %load * %P2
	%rot =l %t shl 31
	%rot2 =l %t shr 33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%r =l %h shl 27
	%r =l %r or %h shr 37
	%h =l %r * %P1 + %P4
	%ptr =l %ptr + 8
	%rem =l %rem - 8
	jmp @while8

@after8
	%cond =w csltl %rem, %c4
	jnz %cond, @after4
	%load32 =l loaduw %ptr
	%tmp =l %load32 * %P1
	%h =l %h xor %tmp
	%r =l %h shl 23
	%r =l %r or %h shr 41
	%h =l %r * %P2 + %P3
	%ptr =l %ptr + 4
	%rem =l %rem - 4

@after4
@while1
	%cond =w csltl %rem, %c1
	jnz %cond, @final
	%byte =l loadb %ptr
	%tmp =l %byte * %P5
	%h =l %h xor %tmp
	%r =l %h shl 11
	%r =l %r or %h shr 53
	%h =l %r * %P1
	%ptr =l %ptr + 1
	%rem =l %rem - 1
	jmp @while1

@final
	%h =l %h xor %h shr 33
	%h =l %h * %P2
	%h =l %h xor %h shr 29
	%h =l %h * %P3
	%h =l %h xor %h shr 32
	ret %h
}
