export function l $xxh64(l %p, l %n, l %seed) {
@entry
	%len =l %n
	%ptr =l %p
	%c32 =l 32
	%c8 =l 8
	%c4 =l 4
	%c1 =l 1
	%P1 =l -7046029288634856825
	%P2 =l -4417276706812521889
	%P3 =l 1609587929392839161
	%P4 =l -8796714831421723037
	%P5 =l 2870177450012600261
	%c31 =l 31
	%c33 =l 33
	%c27 =l 27
	%c23 =l 23
	%c11 =l 11
	%c33s =l 33
	%c29 =l 29
	%c32s =l 32

	%cond_ge32 =csltl %len, %c32
	jnz %cond_ge32, @no_stripes
	/* n >= 32 */
	%v1 =l %seed
	%tmp =l %P1
	%v1 =l %v1 + %tmp
	%tmp =l %P2
	%v1 =l %v1 + %tmp

	%v2 =l %seed
	%v2 =l %v2 + %P2

	%v3 =l %seed

	%v4 =l %seed
	%v4 =l %v4 - %P1

	%rem =l %len
@stripe_loop
	%cond =csltl %rem, %c32
	jnz %cond, @after_stripes

	%load1 =l loadl %ptr
	%load2 =l loadl %ptr+8
	%load3 =l loadl %ptr+16
	%load4 =l loadl %ptr+24

	/* round(v1, load1) */
	%t =l %load1 * %P2
	%t =l %v1 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%v1 =l %rot * %P1

	/* round(v2, load2) */
	%t =l %load2 * %P2
	%t =l %v2 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%v2 =l %rot * %P1

	/* round(v3, load3) */
	%t =l %load3 * %P2
	%t =l %v3 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%v3 =l %rot * %P1

	/* round(v4, load4) */
	%t =l %load4 * %P2
	%t =l %v4 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%v4 =l %rot * %P1

	%ptr =l %ptr + %c32
	%rem =l %rem - %c32
	jmp @stripe_loop
@after_stripes
	/* h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18) */
	%rot1 =l %v1 shl 1
	%rot1 =l %rot1 or %v1 shr 63
	%rot2 =l %v2 shl 7
	%rot2 =l %rot2 or %v2 shr 57
	%rot3 =l %v3 shl 12
	%rot3 =l %rot3 or %v3 shr 52
	%rot4 =l %v4 shl 18
	%rot4 =l %rot4 or %v4 shr 46
	%h =l %rot1 + %rot2 + %rot3 + %rot4

	/* merge(h, v1) */
	%tmp0 =l 0
	%t =l %v1 * %P2
	%t =l %tmp0 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%r0v1 =l %rot * %P1
	%h =l %h xor %r0v1
	%h =l %h * %P1 + %P4

	/* merge(h, v2) */
	%t =l %v2 * %P2
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%r0v2 =l %rot * %P1
	%h =l %h xor %r0v2
	%h =l %h * %P1 + %P4

	/* merge(h, v3) */
	%t =l %v3 * %P2
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%r0v3 =l %rot * %P1
	%h =l %h xor %r0v3
	%h =l %h * %P1 + %P4

	/* merge(h, v4) */
	%t =l %v4 * %P2
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%r0v4 =l %rot * %P1
	%h =l %h xor %r0v4
	%h =l %h * %P1 + %P4

	jmp @post_stripes
@no_stripes
	%h =l %seed + %P5
@post_stripes
	%h =l %h + %len

	%rem =l %rem   // remaining bytes after stripe loop (if any), %rem already holds leftover count
@while8
	%cond =csltl %rem, %c8
	jnz %cond, @after_while8
	%load =l loadl %ptr
	/* round(0, load) */
	%t =l %load * %P2
	%t =l 0 + %t
	%rot =l %t shl %c31
	%rot2 =l %t shr %c33
	%rot =l %rot or %rot2
	%rnd =l %rot * %P1
	%h =l %h xor %rnd
	%roth =l %h shl %c27
	%roth =l %roth or %h shr %c37   // 64-27 = 37
	%h =l %roth * %P1 + %P4
	%ptr =l %ptr + %c8
	%rem =l %rem - %c8
	jmp @while8
@after_while8
	%cond =csltl %rem, %c4
	jnz %cond, @after_while4
	%load4 =l loaduw %ptr
	%tmp =l %load4 * %P1
	%h =l %h xor %tmp
	%roth =l %h shl %c23
	%roth =l %roth or %h shr %c41   // 64-23 = 41
	%h =l %roth * %P2 + %P3
	%ptr =l %ptr + %c4
	%rem =l %rem - %c4
@after_while4
@while1
	%cond =csltl %rem, %c1
	jnz %cond, @final_avalanche
	%byte =l loadub %ptr
	%tmp =l %byte * %P5
	%h =l %h xor %tmp
	%roth =l %h shl %c11
	%roth =l %roth or %h shr %c53   // 64-11 = 53
	%h =l %roth * %P1
	%ptr =l %ptr + %c1
	%rem =l %rem - %c1
	jmp @while1
@final_avalanche
	%h =l %h xor %h shr %c33s
	%h =l %h * %P2
	%h =l %h xor %h shr %c29
	%h =l %h * %P3
	%h =l %h xor %h shr %c32s
	ret %h
}
