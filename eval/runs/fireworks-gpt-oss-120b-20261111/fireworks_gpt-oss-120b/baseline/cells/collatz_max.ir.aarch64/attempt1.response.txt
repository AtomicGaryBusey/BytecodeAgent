export function l $collatz_max(l n) {
    %zero = constl 0
    %one = constl 1
    jump %outer(%one, %zero)

@outer(i, max):
    %n1 = addl %n, %one
    %cond = csltl %i, %n1
    jnz %cond, %inner_start(i, max), %ret(max)

@inner_start(i, max):
    jump %collatz(%i, %zero)

@collatz(x, steps):
    %cond1 = ceql %x, %one
    jnz %cond1, %collatz_done(%x, %steps), %collatz_step(%x, %steps)

@collatz_step(x, steps):
    %steps1 = addl %steps, %one
    %odd = andl %x, %one
    %is_even = ceql %odd, %zero
    jnz %is_even, %collatz_even(%x, %steps1), %collatz_odd(%x, %steps1)

@collatz_even(x, steps1):
    %x_next = shrul %x, %one
    jump %collatz(%x_next, %steps1)

@collatz_odd(x, steps1):
    %tmp = mulsl %x, constl 3
    %x_next = addl %tmp, %one
    jump %collatz(%x_next, %steps1)

@collatz_done(x, steps):
    jump %after_collatz(%steps, i, max)

@after_collatz(steps, i, max):
    %cmp = csltl %max, %steps
    jnz %cmp, %set_max(%steps, %i), %keep_max(%max, %i)

@set_max(newmax, i):
    jump %inc_i(%newmax, %i)

@keep_max(oldmax, i):
    jump %inc_i(%oldmax, %i)

@inc_i(max, i):
    %i_next = addl %i, %one
    jump %outer(%i_next, %max)

@ret(max):
    ret %max
}
