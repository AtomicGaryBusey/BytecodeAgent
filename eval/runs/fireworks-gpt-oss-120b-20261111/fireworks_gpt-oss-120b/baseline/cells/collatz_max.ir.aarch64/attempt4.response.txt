export function l $collatz_max(l n) {
    %zero = constl 0
    %one = constl 1
    %i0 = constl 1
    %max0 = constl 0
    jump %outer(%i0, %max0)

@outer(%i, %max):
    %n1 = addl %n, %one
    %cond = csltl %i, %n1
    jnz %cond, %start_collatz(%i, %max), %ret(%max)

@start_collatz(%i, %max):
    jump %collatz(%i, %zero, %i, %max)

@collatz(%x, %steps, %i, %max):
    %is_one = ceql %x, %one
    jnz %is_one, %after_collatz(%steps, %i, %max), %collatz_step(%x, %steps, %i, %max)

@collatz_step(%x, %steps, %i, %max):
    %steps1 = addl %steps, %one
    %odd = andl %x, %one
    %is_even = ceql %odd, %zero
    jnz %is_even, %collatz_even(%x, %steps1, %i, %max), %collatz_odd(%x, %steps1, %i, %max)

@collatz_even(%x, %steps1, %i, %max):
    %x_next = shrl %x, %one
    jump %collatz(%x_next, %steps1, %i, %max)

@collatz_odd(%x, %steps1, %i, %max):
    %tmp = mull %x, constl 3
    %x_next = addl %tmp, %one
    jump %collatz(%x_next, %steps1, %i, %max)

@after_collatz(%steps, %i, %max):
    %cmp = csltl %max, %steps
    jnz %cmp, %inc_i(%steps, %i), %inc_i(%max, %i)

@inc_i(%newmax, %i):
    %i_next = addl %i, %one
    jump %outer(%i_next, %newmax)

@ret(%max):
    ret %max
}
