export function l $collatz_max(l n) {
    %zero = constl 0
    %one = constl 1
    jump %outer(%one, %zero)

@outer(%i, %max):
    %n1 = addl %n, %one
    %cond = csltl %i, %n1
    jnz %cond, %collatz_start(%i, %max), %ret(%max)

@collatz_start(%i, %max):
    jump %collatz(%i, %zero, %i, %max)

@collatz(%x, %steps, %i, %max):
    %is_one = ceql %x, %one
    jnz %is_one, %after_collatz(%steps, %i, %max), %collatz_step(%x, %steps, %i, %max)

@collatz_step(%x, %steps, %i, %max):
    %steps1 = addl %steps, %one
    %odd = andl %x, %one
    %is_even = ceql %odd, %zero
    jnz %is_even, %collatz_even(%x, %steps1, %i, %max), %collatz_odd(%x, %steps1, %i, %max)

@collatz_even(%x, %steps1, %i, %max):
    %x_next = shrul %x, %one
    jump %collatz(%x_next, %steps1, %i, %max)

@collatz_odd(%x, %steps1, %i, %max):
    %tmp = mulsl %x, constl 3
    %x_next = addl %tmp, %one
    jump %collatz(%x_next, %steps1, %i, %max)

@after_collatz(%steps, %i, %max):
    %cmp = csltl %max, %steps
    jnz %cmp, %update_max(%steps, %i), %keep_max(%max, %i)

@update_max(%newmax, %i):
    jump %inc_i(%newmax, %i)

@keep_max(%oldmax, %i):
    jump %inc_i(%oldmax, %i)

@inc_i(%max, %i):
    %i_next = addl %i, %one
    jump %outer(%i_next, %max)

@ret(%max):
    ret %max
}
