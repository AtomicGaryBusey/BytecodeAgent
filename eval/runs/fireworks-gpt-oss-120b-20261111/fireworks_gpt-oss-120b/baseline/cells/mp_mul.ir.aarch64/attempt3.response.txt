export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry:
	%len = add %na, %nb
	%zero = 0
	%cond_len = csltl %zero, %len
	jnz w %cond_len, @zero_loop
	ret %zero

@zero_loop:
	%z = phi l %zero, @entry, %z_next, @zero_body
	%cond_z = csltl %z, %len
	jnz w %cond_z, @zero_body
	jmp @outer_entry

@zero_body:
	%off_z = shl %z, 3
	%addr_z = add %r, %off_z
	storel %zero, %addr_z
	%z_next = add %z, 1
	jmp @zero_loop

@outer_entry:
	%i = phi l %zero, @zero_loop, %i_next, @outer_inc
	%cond_i = csltl %i, %na
	jnz w %cond_i, @outer_body
	jmp @outer_done

@outer_body:
	%j = phi l %zero, @outer_entry, %j_next, @inner_inc
	%carry = phi l %zero, @outer_entry, %carry_next, @inner_inc
	%cond_j = csltl %j, %nb
	jnz w %cond_j, @inner_body
	%idxc = add %i, %nb
	%offc = shl %idxc, 3
	%addrc = add %r, %offc
	%oldc = loadl %addrc
	%sumc = add %oldc, %carry
	storel %sumc, %addrc
	jmp @outer_inc

@inner_body:
	%off_i = shl %i, 3
	%addr_ai = add %a, %off_i
	%ai = loadl %addr_ai
	%off_j = shl %j, 3
	%addr_bj = add %b, %off_j
	%bj = loadl %addr_bj

	%ai_lo = and %ai, 0xffffffff
	%ai_hi = shr %ai, 32
	%bj_lo = and %bj, 0xffffffff
	%bj_hi = shr %bj, 32
	%p0 = mul %ai_lo, %bj_lo
	%p1 = mul %ai_lo, %bj_hi
	%p2 = mul %ai_hi, %bj_lo
	%p3 = mul %ai_hi, %bj_hi
	%mid = add %p1, %p2
	%mid_lo = and %mid, 0xffffffff
	%mid_hi = shr %mid, 32
	%mid_lo_sh = shl %mid_lo, 32
	%low = add %p0, %mid_lo_sh
	%c_low1 = w cultl %low, %p0
	%high_mid = add %p3, %mid_hi
	%high = add %high_mid, %c_low1

	%idx = add %i, %j
	%off_idx = shl %idx, 3
	%addr_r = add %r, %off_idx
	%old = loadl %addr_r
	%sum0 = add %old, %low
	%c0 = w cultl %sum0, %old
	%sum = add %sum0, %carry
	%c1 = w cultl %sum, %sum0
	%new_carry = add %high, %c0
	%new_carry = add %new_carry, %c1
	storel %sum, %addr_r
	jmp @inner_inc

@inner_inc:
	%j_next = add %j, 1
	%carry_next = %new_carry
	jmp @outer_body

@outer_inc:
	%i_next = add %i, 1
	jmp @outer_entry

@outer_done:
	jmp @trim_loop

@trim_loop:
	%k = phi l %len, @outer_done, %k_next, @trim_dec
	%cond_k = csltl %zero, %k
	jnz w %cond_k, @trim_body
	ret %zero

@trim_body:
	%km1 = sub %k, 1
	%offk = shl %km1, 3
	%addrk = add %r, %offk
	%valk = loadl %addrk
	%iszero = ceql %valk, %zero
	jnz w %iszero, @trim_dec
	ret %k

@trim_dec:
	%k_next = sub %k, 1
	jmp @trim_loop
}
