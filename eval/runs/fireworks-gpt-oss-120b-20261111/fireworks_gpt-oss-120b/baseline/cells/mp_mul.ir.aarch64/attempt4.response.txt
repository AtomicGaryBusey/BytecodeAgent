export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry:
	%zero = 0
	%len = add %na, %nb
	%i0 = %zero
	jnz w (csltl %i0, %len), @zero_body
	jmp @outer_entry

@zero_body:
	%off = shl %i0, 3
	%addr = add %r, %off
	storel %zero, %addr
	%i_next = add %i0, 1
	%cond = csltl %i_next, %len
	jnz w %cond, @zero_body
	jmp @outer_entry

@outer_entry:
	%i = phi l %zero, @zero_body, %i_next, @outer_inc
	%cond_i = csltl %i, %na
	jnz w %cond_i, @outer_body
	jmp @outer_done

@outer_body:
	%carry0 = %zero
	%j0 = %zero
	jmp @inner_entry

@inner_entry:
	%j = phi l %zero, @outer_body, %j_next, @inner_body
	%carry = phi l %zero, @outer_body, %carry_next, @inner_body
	%cond_j = csltl %j, %nb
	jnz w %cond_j, @inner_body
	%idxc = add %i, %nb
	%carry_val = %carry
	jmp @carry_prop_entry

@inner_body:
	%off_i = shl %i, 3
	%addr_ai = add %a, %off_i
	%ai = loadl %addr_ai

	%off_j = shl %j, 3
	%addr_bj = add %b, %off_j
	%bj = loadl %addr_bj

	%mask = 0xffffffff
	%ai_lo = and %ai, %mask
	%ai_hi = shr %ai, 32
	%bj_lo = and %bj, %mask
	%bj_hi = shr %bj, 32

	%p0 = mul %ai_lo, %bj_lo
	%p1 = mul %ai_lo, %bj_hi
	%p2 = mul %ai_hi, %bj_lo
	%p3 = mul %ai_hi, %bj_hi

	%mid = add %p1, %p2
	%mid_lo = and %mid, %mask
	%mid_hi = shr %mid, 32
	%mid_lo_sh = shl %mid_lo, 32
	%low_tmp = add %p0, %mid_lo_sh
	%c_low = w cultl %low_tmp, %p0
	%low = %low_tmp

	%high_tmp = add %p3, %mid_hi
	%high = add %high_tmp, %c_low

	%idx = add %i, %j
	%off_idx = shl %idx, 3
	%addr_r = add %r, %off_idx
	%old = loadl %addr_r
	%sum0 = add %old, %low
	%c0 = w cultl %sum0, %old
	%sum = add %sum0, %carry
	%c1 = w cultl %sum, %sum0
	%new_carry = add %high, %c0
	%new_carry = add %new_carry, %c1

	storel %sum, %addr_r

	%j_next = add %j, 1
	%carry_next = %new_carry
	jmp @inner_entry

@carry_prop_entry:
	%idx = phi l %idxc, @inner_body, %idx_next, @carry_body
	%c = phi l %carry_val, @inner_body, %c_next, @carry_body
	%cond_idx = csltl %idx, %len
	%cond_c = csltl %zero, %c
	%go = and %cond_idx, %cond_c
	jnz w %go, @carry_body
	jmp @outer_inc

@carry_body:
	%off_idx = shl %idx, 3
	%addr_idx = add %r, %off_idx
	%oldc = loadl %addr_idx
	%sumc = add %oldc, %c
	%c_next = w cultl %sumc, %oldc
	storel %sumc, %addr_idx
	%idx_next = add %idx, 1
	jmp @carry_prop_entry

@outer_inc:
	%i_next = add %i, 1
	jmp @outer_entry

@outer_done:
	%k0 = %len
	jmp @trim_entry

@trim_entry:
	%k = phi l %len, @outer_done, %k_next, @trim_body
	%cond_k = csltl %zero, %k
	jnz w %cond_k, @trim_body
	ret %zero

@trim_body:
	%km1 = sub %k, 1
	%offk = shl %km1, 3
	%addrk = add %r, %offk
	%valk = loadl %addrk
	%iszero = ceql %valk, %zero
	jnz w %iszero, @trim_dec
	ret %k

@trim_dec:
	%k_next = sub %k, 1
	jmp @trim_body
}
