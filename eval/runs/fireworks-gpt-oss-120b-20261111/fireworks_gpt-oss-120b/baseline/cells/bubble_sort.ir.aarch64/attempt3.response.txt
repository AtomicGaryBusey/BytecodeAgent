export function l $bubble_sort(l %a, l %n) {
@entry
    jmp @outer_cond(const 0)

@outer_cond(l %i)
    %n1 = subl %n, const 1
    %c = csltl %i, %n1
    jnz %c, @inner_start(%i), @ret

@inner_start(l %i)
    %i1 = addl %i, const 1
    %lim = subl %n, %i1
    %j0 = const 0
    jmp @inner_cond(%i, %j0, %lim)

@inner_cond(l %i, l %j, l %lim)
    %c = csltl %j, %lim
    jnz %c, @compare_swap(%i, %j, %lim), @outer_next(%i)

@compare_swap(l %i, l %j, l %lim)
    %off = mull %j, const 8
    %ptr = addl %a, %off
    %ptr1 = addl %ptr, const 8
    %v = loadl %ptr
    %v1 = loadl %ptr1
    %swap = csltl %v1, %v
    jnz %swap, @do_swap(%ptr, %ptr1, %v, %v1), @inc_j(%i, %j, %lim)

@do_swap(l %ptr, l %ptr1, l %v, l %v1)
    storel %v, %ptr1
    storel %v1, %ptr
    jmp @inc_j(%i, %j, %lim)

@inc_j(l %i, l %j, l %lim)
    %j1 = addl %j, const 1
    jmp @inner_cond(%i, %j1, %lim)

@outer_next(l %i)
    %i1 = addl %i, const 1
    jmp @outer_cond(%i1)

@ret
    ret const 0
}
