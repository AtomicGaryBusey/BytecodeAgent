export function l $bubble_sort(l %a, l %n) {
@entry
  %c0 = const 0
  %c1 = const 1
  %c8 = const 8
  %i0 = const 0
  jmp @outer_cond

@outer_cond
  %n1 = subl %n %c1
  %i_lt_n1 = csltl %i0 %n1
  jnz %i_lt_n1, @outer_body
  ret %c0

@outer_body
  %j0 = const 0
  %n_i = subl %n %i0
  %inner_limit = subl %n_i %c1          ; n - i - 1
  jmp @inner_cond

@inner_cond
  %j_lt_lim = csltl %j0 %inner_limit
  jnz %j_lt_lim, @inner_body
  %i_next = addl %i0 %c1
  %i0 = %i_next
  jmp @outer_cond

@inner_body
  %off_j = mul %j0 %c8
  %ptr_j = addl %a %off_j
  %off_j1 = addl %off_j %c8
  %ptr_j1 = addl %a %off_j1
  %val_j = loadl %ptr_j
  %val_j1 = loadl %ptr_j1
  %swap_cond = csltl %val_j1 %val_j   ; val_j > val_j1 ?
  jnz %swap_cond, @do_swap
  %j_next = addl %j0 %c1
  %j0 = %j_next
  jmp @inner_cond

@do_swap
  storel %val_j, %ptr_j1
  storel %val_j1, %ptr_j
  %j_next = addl %j0 %c1
  %j0 = %j_next
  jmp @inner_cond
}
