export function l $bubble_sort(l %a, l %n) {
@entry
  jmp @outer_cond(const 0)

@outer_cond(%i)
  %n1 = subl %n, const 1
  %cond = csltl %i, %n1
  jnz %cond, @outer_body(%i), @ret

@ret
  ret const 0

@outer_body(%i)
  %n_i = subl %n, %i
  %inner_limit = subl %n_i, const 1
  jmp @inner_cond(%i, const 0, %inner_limit)

@inner_cond(%i, %j, %lim)
  %condj = csltl %j, %lim
  jnz %condj, @inner_body(%i, %j, %lim), @outer_next(%i)

@outer_next(%i)
  %i_next = addl %i, const 1
  jmp @outer_cond(%i_next)

@inner_body(%i, %j, %lim)
  %off = mull %j, const 8
  %ptr = addl %a, %off
  %off1 = addl %off, const 8
  %ptr1 = addl %a, %off1
  %val = loadl %ptr
  %val1 = loadl %ptr1
  %swap = csltl %val1, %val
  jnz %swap, @do_swap(%i, %j, %lim, %val, %val1, %ptr, %ptr1), @inc_j(%i, %j, %lim)

@do_swap(%i, %j, %lim, %val, %val1, %ptr, %ptr1)
  storel %val, %ptr1
  storel %val1, %ptr
  jmp @inc_j(%i, %j, %lim)

@inc_j(%i, %j, %lim)
  %j_next = addl %j, const 1
  jmp @inner_cond(%i, %j_next, %lim)
}
