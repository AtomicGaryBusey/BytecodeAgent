packbits_decode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    cld
    mov rbx, rdi
    lea r10, [rdi + rsi]
    mov r12, rdx
    lea r13, [rdx + rcx]
    xor r14, r14
.Lmain_loop:
    cmp rbx, r10
    jae .Lreturn_written
    movsx r8d, byte ptr [rbx]
    inc rbx
    cmp r8d, 0
    jge .Lliteral
    cmp r8d, -128
    je .Lnoop
    # repeat packet
    cmp rbx, r10
    jae .Lerror_src
    mov al, byte ptr [rbx]
    inc rbx
    mov eax, 1
    sub eax, r8d
    mov ecx, eax
    mov r15, r13
    sub r15, r12
    cmp r15, rcx
    jae .Lrepeat_full
    test r15, r15
    jz .Lerror_out_cap
    mov rcx, r15
    mov rdi, r12
    rep stosb
    add r12, r15
    add r14, r15
    mov rax, -1
    jmp .Lcleanup
.Lrepeat_full:
    mov rdi, r12
    rep stosb
    add r12, rcx
    add r14, rcx
    jmp .Lmain_loop
.Lliteral:
    movsx rcx, r8d
    inc rcx
    mov r15, r10
    sub r15, rbx
    cmp r15, rcx
    jb .Lerror_src
    mov r15, r13
    sub r15, r12
    cmp r15, rcx
    jae .Lliteral_full
    test r15, r15
    jz .Lerror_out_cap
    mov rdx, r15
    mov rsi, rbx
    mov rdi, r12
    mov rcx, r15
    rep movsb
    add rbx, r15
    add r12, r15
    add r14, r15
    mov rax, -1
    jmp .Lcleanup
.Lliteral_full:
    mov rsi, rbx
    mov rdi, r12
    rep movsb
    add rbx, rcx
    add r12, rcx
    add r14, rcx
    jmp .Lmain_loop
.Lnoop:
    jmp .Lmain_loop
.Lerror_src:
    mov rax, -2
    jmp .Lcleanup
.Lerror_out_cap:
    mov rax, -1
    jmp .Lcleanup
.Lreturn_written:
    mov rax, r14
.Lcleanup:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
