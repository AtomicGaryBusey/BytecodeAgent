packbits_decode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    cld
    mov rbx, rdi                 # src pointer
    lea r12, [rdi + rsi]         # src_end
    mov r13, rdx                 # dst pointer
    lea r14, [rdx + rcx]         # dst_end
    xor r15, r15                 # written count
.Lloop:
    cmp rbx, r12
    jae .Lreturn_written
    movsx r8d, byte ptr [rbx]    # header signed
    inc rbx
    cmp r8d, 0
    jge .Lliteral
    cmp r8d, -128
    je .Lnoop
    # repeat packet
    cmp rbx, r12
    jae .Lerror_src
    mov al, byte ptr [rbx]       # repeat value
    inc rbx
    mov edi, 1
    sub edi, r8d                 # count = 1 - header
    mov r9, r14
    sub r9, r13                  # dst_remaining
    cmp r9, rdi
    jae .Lrepeat_full
    test r9, r9
    jz .Lreturn_cap_exceed
    mov rcx, r9
    mov rdi, r13
    rep stosb
    add r13, r9
    add r15, r9
    mov rax, -1
    jmp .Lcleanup
.Lrepeat_full:
    mov r10, rdi                 # preserve count
    mov rcx, rdi
    mov rdi, r13
    rep stosb
    add r13, r10
    add r15, r10
    jmp .Lloop
.Lliteral:
    mov ecx, r8d
    inc ecx                      # length = header + 1
    mov r9, r12
    sub r9, rbx                  # src_remaining
    cmp r9, rcx
    jb .Lerror_src
    mov r9, r14
    sub r9, r13                  # dst_remaining
    cmp r9, rcx
    jae .Lliteral_full
    test r9, r9
    jz .Lreturn_cap_exceed
    mov r10, r9                  # preserve copy amount
    mov rsi, rbx
    mov rdi, r13
    mov rcx, r9
    rep movsb
    add rbx, r10
    add r13, r10
    add r15, r10
    mov rax, -1
    jmp .Lcleanup
.Lliteral_full:
    mov r10, rcx                 # preserve length
    mov rsi, rbx
    mov rdi, r13
    rep movsb
    add rbx, r10
    add r13, r10
    add r15, r10
    jmp .Lloop
.Lnoop:
    jmp .Lloop
.Lerror_src:
    mov rax, -2
    jmp .Lcleanup
.Lreturn_cap_exceed:
    mov rax, -1
    jmp .Lcleanup
.Lreturn_written:
    mov rax, r15
.Lcleanup:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
