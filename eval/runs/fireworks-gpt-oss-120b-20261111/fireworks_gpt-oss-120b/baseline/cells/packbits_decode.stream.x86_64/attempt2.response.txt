packbits_decode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    cld
    mov rbx, rdi                  # src pointer
    lea r10, [rdi + rsi]          # src_end = src + n
    mov r12, rdx                  # out pointer
    lea r13, [rdx + rcx]          # out_end = out + cap
    xor r14, r14                  # bytes_written = 0
.Lmain_loop:
    cmp rbx, r10
    jae .Lreturn_written
    movzx eax, byte ptr [rbx]     # header unsigned
    inc rbx
    movsx r8d, al                 # signed header in r8d
    cmp r8d, 0
    jge .Lliteral
    cmp r8d, -128
    je .Lnoop
    # repeat packet
    cmp rbx, r10
    jae .Lerror_src
    mov al, byte ptr [rbx]        # repeat value
    inc rbx
    mov ebx, 1
    sub ebx, r8d                  # repeat count = 1 - h
    # remaining output capacity
    mov r15, r13
    sub r15, r12                  # r15 = out_end - out
    cmp r15, rbx
    ja .Lrepeat_full
    test r15, r15
    jz .Lerror_out_cap
    # partial write
    mov rcx, r15
    mov rdi, r12
    rep stosb
    mov rax, -1
    jmp .Lcleanup
.Lrepeat_full:
    mov rcx, rbx
    mov rdi, r12
    rep stosb
    add r14, rbx
    mov r12, rdi
    jmp .Lmain_loop
.Lliteral:
    lea ebx, [r8d + 1]            # literal count = h + 1
    # ensure enough source bytes
    mov r15, r10
    sub r15, rbx
    cmp r15, ebx
