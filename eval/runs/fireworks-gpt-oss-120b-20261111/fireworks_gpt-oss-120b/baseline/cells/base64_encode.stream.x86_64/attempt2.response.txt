base64_encode:
    mov r8, rsi                # save original n
    mov r9, rdi                # src pointer copy
    mov r10, rdx               # out pointer copy
    mov rsi, rsi               # remaining count (in rsi)
.Lfull:
    cmp rsi, 3
    jl .Ltail
    movzx rax, byte ptr [r9]   # b0
    movzx rcx, byte ptr [r9+1] # b1
    movzx r11, byte ptr [r9+2] # b2
    shl rax, 16
    shl rcx, 8
    or rax, rcx
    or rax, r11                # rax = 24‑bit value
    # char 0
    mov rcx, rax
    shr rcx, 18
    and rcx, 0x3F
    cmp rcx, 26
    jb .Lmap0A
    cmp rcx, 52
    jb .Lmap0a
    cmp rcx, 62
    jb .Lmap0d
    cmp rcx, 63
    je .Lmap0s
    mov al, '+'
    jmp .Lstore0
.Lmap0s:
    mov al, '/'
    jmp .Lstore0
.Lmap0d:
    add rcx, -4
    jmp .Lstore0
.Lmap0a:
    add rcx, 71
    jmp .Lstore0
.Lmap0A:
    add rcx, 65
.Lstore0:
    mov byte ptr [r10], cl
    inc r10
    # char 1
    mov rcx, rax
    shr rcx, 12
    and rcx, 0x3F
    cmp rcx, 26
    jb .Lmap1A
    cmp rcx, 52
    jb .Lmap1a

    jb .Lmap2A

    jb .Lmap2a

    jmp .Lstore3
.Lmap3a:
    add rcx, 71
    jmp .Lstore3
.Lmap3A:
    add rcx, 65
.Lstore3:
    mov byte ptr [r10], cl
    inc r10
    add r9, 3
    sub rsi, 3
    jmp .Lfull
.Ltail:
    cmp rsi, 0
    je .Ldone
    cmp rsi, 1
    je .Lone
    # two bytes remaining
    movzx rax, byte ptr [r9]
    movzx rcx, byte ptr [r9+1]
    shl rax, 16
    shl rcx, 8
    or rax, rcx                # rax = 16‑bit value in high bits
    # idx0
    mov rcx, rax
    shr rcx, 18

    cmp rcx, 26
    jb .Lmap4A

    jb .Lmap4a

    inc r10
    mov byte ptr [r10], '='
    inc r10
    mov byte ptr [r10], '='
    inc r10
.Ldone:
    mov rax, r8
    add rax, 2
    xor rdx, rdx
    mov rcx, 3
    div rcx
    imul rax, 4
    ret
