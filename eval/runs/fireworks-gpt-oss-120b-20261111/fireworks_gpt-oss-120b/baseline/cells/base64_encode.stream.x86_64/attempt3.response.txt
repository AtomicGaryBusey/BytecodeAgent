base64_encode:
    mov     r8, rsi                # save original n
    # rdi = src, rsi = n, rdx = out
.Lfull_loop:
    cmp     rsi, 3
    jb      .Ltail
    movzx   rax, byte ptr [rdi]          # b0
    movzx   rcx, byte ptr [rdi+1]        # b1
    movzx   r9,  byte ptr [rdi+2]        # b2
    shl     rax, 16
    shl     rcx, 8
    or      rax, rcx
    or      rax, r9                      # rax = 24‑bit value
    # char 0
    mov     rcx, rax
    shr     rcx, 18
    and     rcx, 0x3F
    call    .Lmap_and_store
    # char 1
    mov     rcx, rax
    shr     rcx, 12
    and     rcx, 0x3F
    call    .Lmap_and_store
    # char 2
    mov     rcx, rax
    shr     rcx, 6
    and     rcx, 0x3F
    call    .Lmap_and_store
    # char 3
    mov     rcx, rax
    and     rcx, 0x3F
    call    .Lmap_and_store
    add     rdi, 3
    sub     rsi, 3
    jmp     .Lfull_loop
.Ltail:
    cmp     rsi, 0
    je      .Ldone
    cmp     rsi, 1
    je      .Lone_byte
    # two bytes remaining
    movzx   rax, byte ptr [rdi]          # b0
    movzx   rcx, byte ptr [rdi+1]        # b1
    shl     rax, 16
    shl     rcx, 8
    or      rax, rcx                     # rax = 16‑bit value in high bits
    # idx0
    mov     rcx, rax
    shr     rcx, 18
    and     rcx, 0x3F
    call    .Lmap_and_store
    # idx1
    mov     rcx, rax
    shr     rcx, 12
    and     rcx, 0x3F
    call    .Lmap_and_store
    # idx2
    mov     rcx, rax
    shr     rcx, 6
    and     rcx, 0x3F
    call    .Lmap_and_store
    # padding
    mov     byte ptr [rdx], '='
    inc     rdx
    jmp     .Ldone
.Lone_byte:
    movzx   rax, byte ptr [rdi]          # b0
    shl     rax, 16
    # idx0
    mov     rcx, rax
    shr     rcx, 18
    and     rcx, 0x3F
    call    .Lmap_and_store
    # idx1
    mov     rcx, rax
    shr     rcx, 12
    and     rcx, 0x3F
    call    .Lmap_and_store
    # padding x2
    mov     byte ptr [rdx], '='
    inc     rdx
    mov     byte ptr [rdx], '='
    inc     rdx
    jmp     .Ldone
.Lmap_and_store:
    # rcx = index 0..63, rdx = out ptr
    cmp     rcx, 26
    jb      .Lmap_A
    cmp     rcx, 52
    jb      .Lmap_a
    cmp     rcx, 62
    jb      .Lmap_0
    # rcx is 62 or 63
    cmp     rcx, 62
    je      .Lmap_plus
    mov     rcx, '/'                # 63
    jmp     .Lstore_char
.Lmap_plus:
    mov     rcx, '+'
    jmp     .Lstore_char
.Lmap_A:
    add     rcx, 65                 # 'A'
    jmp     .Lstore_char
.Lmap_a:
    add     rcx, 71                 # 'a' - 26 + 65 = 97, but 71 works: 26+71=97
    jmp     .Lstore_char
.Lmap_0:
    add     rcx, -4                 # '0' - 52 = -4
    # fall through
.Lstore_char:
    mov     byte ptr [rdx], cl
    inc     rdx
    ret
.Ldone:
    mov     rax, r8
    add     rax, 2
    xor     rdx, rdx
    mov     rcx, 3
    div     rcx
    imul    rax, 4
    ret
