base64_encode:
    mov r8, rdi                # src pointer
    mov r9, rdx                # out pointer
    mov r10, rsi               # remaining byte count

.Lloop:
    cmp r10, 3
    jl .Ltail

    mov al, byte ptr [r8]      # b0
    mov bl, byte ptr [r8+1]    # b1
    mov cl, byte ptr [r8+2]    # b2

    movzx rax, al
    shl rax, 16
    movzx rbx, bl
    shl rbx, 8
    or rax, rbx
    movzx rcx, cl
    or rax, rcx                # rax = 24‑bit value

    # ---- output first char (bits 18‑23) ----
    mov rcx, rax
    shr rcx, 18
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap0A
    cmp rcx, 52
    jb .Lmap0a
    cmp rcx, 62
    jb .Lmap0d
    cmp rcx, 63
    je .Lmap0s
    mov al, '+'
    jmp .Lstore0
.Lmap0s:
    mov al, '/'
    jmp .Lstore0
.Lmap0d:
    add al, -4
    jmp .Lstore0
.Lmap0a:
    add al, 71
    jmp .Lstore0
.Lmap0A:
    add al, 65
.Lstore0:
    mov byte ptr [r9], al
    inc r9

    # ---- output second char (bits 12‑17) ----
    mov rcx, rax
    shr rcx, 12
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap1A
    cmp rcx, 52
    jb .Lmap1a
    cmp rcx, 62
    jb .Lmap1d
    cmp rcx, 63
    je .Lmap1s
    mov al, '+'
    jmp .Lstore1
.Lmap1s:
    mov al, '/'
    jmp .Lstore1
.Lmap1d:
    add al, -4
    jmp .Lstore1
.Lmap1a:
    add al, 71
    jmp .Lstore1
.Lmap1A:
    add al, 65
.Lstore1:
    mov byte ptr [r9], al
    inc r9

    # ---- output third char (bits 6‑11) ----
    mov rcx, rax
    shr rcx, 6
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap2A
    cmp rcx, 52
    jb .Lmap2a
    cmp rcx, 62
    jb .Lmap2d
    cmp rcx, 63
    je .Lmap2s
    mov al, '+'
    jmp .Lstore2
.Lmap2s:
    mov al, '/'
    jmp .Lstore2
.Lmap2d:
    add al, -4
    jmp .Lstore2
.Lmap2a:
    add al, 71
    jmp .Lstore2
.Lmap2A:
    add al, 65
.Lstore2:
    mov byte ptr [r9], al
    inc r9

    # ---- output fourth char (bits 0‑5) ----
    mov rcx, rax
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap3A
    cmp rcx, 52
    jb .Lmap3a
    cmp rcx, 62
    jb .Lmap3d
    cmp rcx, 63
    je .Lmap3s
    mov al, '+'
    jmp .Lstore3
.Lmap3s:
    mov al, '/'
    jmp .Lstore3
.Lmap3d:
    add al, -4
    jmp .Lstore3
.Lmap3a:
    add al, 71
    jmp .Lstore3
.Lmap3A:
    add al, 65
.Lstore3:
    mov byte ptr [r9], al
    inc r9

    add r8, 3
    sub r10, 3
    jmp .Lloop

.Ltail:
    cmp r10, 0
    je .Ldone
    cmp r10, 1
    je .Lone

    # two remaining bytes
    mov al, byte ptr [r8]
    mov bl, byte ptr [r8+1]
    movzx rax, al
    shl rax, 16
    movzx rbx, bl
    shl rbx, 8
    or rax, rbx                # rax = 16‑bit value

    # idx0
    mov rcx, rax
    shr rcx, 18
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap4A
    cmp rcx, 52
    jb .Lmap4a
    cmp rcx, 62
    jb .Lmap4d
    cmp rcx, 63
    je .Lmap4s
    mov al, '+'
    jmp .Lstore4
.Lmap4s:
    mov al, '/'
    jmp .Lstore4
.Lmap4d:
    add al, -4
    jmp .Lstore4
.Lmap4a:
    add al, 71
    jmp .Lstore4
.Lmap4A:
    add al, 65
.Lstore4:
    mov byte ptr [r9], al
    inc r9

    # idx1
    mov rcx, rax
    shr rcx, 12
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap5A
    cmp rcx, 52
    jb .Lmap5a
    cmp rcx, 62
    jb .Lmap5d
    cmp rcx, 63
    je .Lmap5s
    mov al, '+'
    jmp .Lstore5
.Lmap5s:
    mov al, '/'
    jmp .Lstore5
.Lmap5d:
    add al, -4
    jmp .Lstore5
.Lmap5a:
    add al, 71
    jmp .Lstore5
.Lmap5A:
    add al, 65
.Lstore5:
    mov byte ptr [r9], al
    inc r9

    # idx2
    mov rcx, rax
    shr rcx, 6
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap6A
    cmp rcx, 52
    jb .Lmap6a
    cmp rcx, 62
    jb .Lmap6d
    cmp rcx, 63
    je .Lmap6s
    mov al, '+'
    jmp .Lstore6
.Lmap6s:
    mov al, '/'
    jmp .Lstore6
.Lmap6d:
    add al, -4
    jmp .Lstore6
.Lmap6a:
    add al, 71
    jmp .Lstore6
.Lmap6A:
    add al, 65
.Lstore6:
    mov byte ptr [r9], al
    inc r9

    mov byte ptr [r9], '='
    inc r9
    jmp .Ldone

.Lone:
    mov al, byte ptr [r8]
    movzx rax, al
    shl rax, 16

    # idx0
    mov rcx, rax
    shr rcx, 18
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap7A
    cmp rcx, 52
    jb .Lmap7a
    cmp rcx, 62
    jb .Lmap7d
    cmp rcx, 63
    je .Lmap7s
    mov al, '+'
    jmp .Lstore7
.Lmap7s:
    mov al, '/'
    jmp .Lstore7
.Lmap7d:
    add al, -4
    jmp .Lstore7
.Lmap7a:
    add al, 71
    jmp .Lstore7
.Lmap7A:
    add al, 65
.Lstore7:
    mov byte ptr [r9], al
    inc r9

    # idx1
    mov rcx, rax
    shr rcx, 12
    and rcx, 0x3F
    mov al, cl
    cmp rcx, 26
    jb .Lmap8A
    cmp rcx, 52
    jb .Lmap8a
    cmp rcx, 62
    jb .Lmap8d
    cmp rcx, 63
    je .Lmap8s
    mov al, '+'
    jmp .Lstore8
.Lmap8s:
    mov al, '/'
    jmp .Lstore8
.Lmap8d:
    add al, -4
    jmp .Lstore8
.Lmap8a:
    add al, 71
    jmp .Lstore8
.Lmap8A:
    add al, 65
.Lstore8:
    mov byte ptr [r9], al
    inc r9

    mov byte ptr [r9], '='
    inc r9
    mov byte ptr [r9], '='
    inc r9
    jmp .Ldone

.Ldone:
    lea rax, [rsi+2]
    xor rdx, rdx
    mov rcx, 3
    div rcx
    imul rax, 4
    ret
