export function l $map_char(l %idx) {
@entry
    %isA = csltl %idx, 26
    %cA = addl %idx, 65
    %sub26 = subl %idx, 26
    %isB = csltl %idx, 52
    %cB = addl %sub26, 97
    %cAB = sel %isA, %cA, %cB
    %isC = csltl %idx, 62
    %sub52 = subl %idx, 52
    %cC = addl %sub52, 48
    %cABC = sel %isC, %cAB, 43
    %is63 = ceql %idx, 63
    %c = sel %is63, 47, %cABC
    ret %c
}

export function l $base64_encode(l %src, l %n, l %out) {
@entry
    %zero = constl 0
    %full = udivl %n, 3
    %rem = umodl %n, 3
    %out_len_full = mulul %full, 4
    %rem_gt0 = csltl %zero, %rem
    %add4 = sel %rem_gt0, 4, 0
    %out_len = addl %out_len_full, %add4
    %i0 = constl 0
    jnz 1, @loop

@loop
    %i = phi %i0, %i_next
    %cond = csltl %i, %full
    jnz %cond, @body
    jnz 1, @rem

@body
    %src_off = mulul %i, 3
    %src_addr = addl %src, %src_off
    %b0 = loadub %src_addr
    %b1 = loadub (addl %src_addr, 1)
    %b2 = loadub (addl %src_addr, 2)
    %v0 = shll %b0, 16
    %v1 = shll %b1, 8
    %v24 = orl %v0, %v1
    %v24 = orl %v24, %b2
    %idx0 = shrul %v24, 18
    %idx1 = andl (shrul %v24, 12), 63
    %idx2 = andl (shrul %v24, 6), 63
    %idx3 = andl %v24, 63

    %c0 = call $map_char(%idx0)
    %c1 = call $map_char(%idx1)
    %c2 = call $map_char(%idx2)
    %c3 = call $map_char(%idx3)

    %dst_off = mulul %i, 4
    %dst_addr = addl %out, %dst_off
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb %c3, (addl %dst_addr, 3)

    %i_next = addl %i, 1
    jnz 1, @loop

@rem
    %rem_eq0 = ceql %rem, 0
    jnz %rem_eq0, @end

    %rem_eq1 = ceql %rem, 1
    jnz %rem_eq1, @rem1

    %rem_eq2 = ceql %rem, 2
    jnz %rem_eq2, @rem2

@rem1
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src
    %b0 = loadub %src_addr
    %v = shll %b0, 16
    %idx0 = shrul %v, 18
    %idx1 = andl (shrul %v, 12), 63

    %c0 = call $map_char(%idx0)
    %c1 = call $map_char(%idx1)

    %base_dst = mulul %full, 4
    %dst_addr = addl %out, %base_dst
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb 61, (addl %dst_addr, 2)
    storeb 61, (addl %dst_addr, 3)
    jnz 1, @end

@rem2
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src
    %b0 = loadub %src_addr
    %b1 = loadub (addl %src_addr, 1)
    %v0 = shll %b0, 16
    %v1 = shll %b1, 8
    %v = orl %v0, %v1
    %idx0 = shrul %v, 18
    %idx1 = andl (shrul %v, 12), 63
    %idx2 = andl (shrul %v, 6), 63

    %c0 = call $map_char(%idx0)
    %c1 = call $map_char(%idx1)
    %c2 = call $map_char(%idx2)

    %base_dst = mulul %full, 4
    %dst_addr = addl %out, %base_dst
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb 61, (addl %dst_addr, 3)
    jnz 1, @end

@end
    ret %out_len
}
