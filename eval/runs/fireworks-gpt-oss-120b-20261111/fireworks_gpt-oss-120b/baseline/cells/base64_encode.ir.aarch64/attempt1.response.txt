export function l $base64_encode(l %src, l %n, l %out)
{
    %full = divl %n, 3
    %rem = modl %n, 3

    %out_len_full = mulul %full, 4
    %rem_gt0 = csltl 0, %rem          // 0 < rem ?
    %add4 = sel %rem_gt0, 4, 0
    %out_len = addl %out_len_full, %add4

    %g0 = constl 0
    @loop
    %g = phi %g0, %g_next
    %cond = csltl %g, %full
    jnz %cond, @body
    jnz 1, @rem

    @body
    %src_off = mulul %g, 3
    %src_addr = addl %src, %src_off
    %b0 = loadub %src_addr
    %b1 = loadub (addl %src_addr, 1)
    %b2 = loadub (addl %src_addr, 2)

    %v0 = shll %b0, 16
    %v1 = shll %b1, 8
    %v24 = orl %v0, %v1
    %v24 = orl %v24, %b2

    %idx0 = shrul %v24, 18
    %idx1 = andl (shrul %v24, 12), 63
    %idx2 = andl (shrul %v24, 6), 63
    %idx3 = andl %v24, 63

    // map idx0 -> c0
    %c0_tmp = addl %idx0, 65
    %c0_cond_a = csltl %idx0, 26
    %c0_tmp2 = addl (subl %idx0, 26), 97
    %c0_tmp = sel %c0_cond_a, %c0_tmp, %c0_tmp2
    %c0_cond_b = csltl %idx0, 52
    %c0_tmp3 = addl (subl %idx0, 52), 48
    %c0_tmp = sel %c0_cond_b, %c0_tmp, %c0_tmp3
    %c0_cond_c = csltl %idx0, 62
    %c0_tmp = sel %c0_cond_c, %c0_tmp, 43
    %c0_cond_eq63 = ceql %idx0, 63
    %c0 = sel %c0_cond_eq63, 47, %c0_tmp

    // map idx1 -> c1
    %c1_tmp = addl %idx1, 65
    %c1_cond_a = csltl %idx1, 26
    %c1_tmp2 = addl (subl %idx1, 26), 97
    %c1_tmp = sel %c1_cond_a, %c1_tmp, %c1_tmp2
    %c1_cond_b = csltl %idx1, 52
    %c1_tmp3 = addl (subl %idx1, 52), 48
    %c1_tmp = sel %c1_cond_b, %c1_tmp, %c1_tmp3
    %c1_cond_c = csltl %idx1, 62
    %c1_tmp = sel %c1_cond_c, %c1_tmp, 43
    %c1_cond_eq63 = ceql %idx1, 63
    %c1 = sel %c1_cond_eq63, 47, %c1_tmp

    // map idx2 -> c2
    %c2_tmp = addl %idx2, 65
    %c2_cond_a = csltl %idx2, 26
    %c2_tmp2 = addl (subl %idx2, 26), 97
    %c2_tmp = sel %c2_cond_a, %c2_tmp, %c2_tmp2
    %c2_cond_b = csltl %idx2, 52
    %c2_tmp3 = addl (subl %idx2, 52), 48
    %c2_tmp = sel %c2_cond_b, %c2_tmp, %c2_tmp3
    %c2_cond_c = csltl %idx2, 62
    %c2_tmp = sel %c2_cond_c, %c2_tmp, 43
    %c2_cond_eq63 = ceql %idx2, 63
    %c2 = sel %c2_cond_eq63, 47, %c2_tmp

    // map idx3 -> c3
    %c3_tmp = addl %idx3, 65
    %c3_cond_a = csltl %idx3, 26
    %c3_tmp2 = addl (subl %idx3, 26), 97
    %c3_tmp = sel %c3_cond_a, %c3_tmp, %c3_tmp2
    %c3_cond_b = csltl %idx3, 52
    %c3_tmp3 = addl (subl %idx3, 52), 48
    %c3_tmp = sel %c3_cond_b, %c3_tmp, %c3_tmp3
    %c3_cond_c = csltl %idx3, 62
    %c3_tmp = sel %c3_cond_c, %c3_tmp, 43
    %c3_cond_eq63 = ceql %idx3, 63
    %c3 = sel %c3_cond_eq63, 47, %c3_tmp

    %dst_off = mulul %g, 4
    %dst_addr = addl %out, %dst_off
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb %c3, (addl %dst_addr, 3)

    %g_next = addl %g, 1
    jnz 1, @loop

    @rem
    %rem_eq0 = ceql %rem, 0
    jnz %rem_eq0, @end

    %rem_eq1 = ceql %rem, 1
    jnz %rem_eq1, @rem1

    %rem_eq2 = ceql %rem, 2
    jnz %rem_eq2, @rem2

    @rem1
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src
    %b0 = loadub %src_addr
    %v = shll %b0, 16

    %idx0 = shrul %v, 18
    %idx1 = andl (shrul %v, 12), 63

    // map idx0 -> c0
    %c0_tmp = addl %idx0, 65
    %c0_cond_a = csltl %idx0, 26
    %c0_tmp2 = addl (subl %idx0, 26), 97
    %c0_tmp = sel %c0_cond_a, %c0_tmp, %c0_tmp2
    %c0_cond_b = csltl %idx0, 52
    %c0_tmp3 = addl (subl %idx0, 52), 48
    %c0_tmp = sel %c0_cond_b, %c0_tmp, %c0_tmp3
    %c0_cond_c = csltl %idx0, 62
    %c0_tmp = sel %c0_cond_c, %c0_tmp, 43
    %c0_cond_eq63 = ceql %idx0, 63
    %c0 = sel %c0_cond_eq63, 47, %c0_tmp

    // map idx1 -> c1
    %c1_tmp = addl %idx1, 65
    %c1_cond_a = csltl %idx1, 26
    %c1_tmp2 = addl (subl %idx1, 26), 97
    %c1_tmp = sel %c1_cond_a, %c1_tmp, %c1_tmp2
    %c1_cond_b = csltl %idx1, 52
    %c1_tmp3 = addl (subl %idx1, 52), 48
    %c1_tmp = sel %c1_cond_b, %c1_tmp, %c1_tmp3
    %c1_cond_c = csltl %idx1, 62
    %c1_tmp = sel %c1_cond_c, %c1_tmp, 43
    %c1_cond_eq63 = ceql %idx1, 63
    %c1 = sel %c1_cond_eq63, 47, %c1_tmp

    %base_dst = mulul %full, 4
    %dst_addr = addl %out, %base_dst
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb 61, (addl %dst_addr, 2)   // '='
    storeb 61, (addl %dst_addr, 3)   // '='
    jnz 1, @end

    @rem2
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src
    %b0 = loadub %src_addr
    %b1 = loadub (addl %src_addr, 1)
    %v0 = shll %b0, 16
    %v1 = shll %b1, 8
    %v = orl %v0, %v1

    %idx0 = shrul %v, 18
    %idx1 = andl (shrul %v, 12), 63
    %idx2 = andl (shrul %v, 6), 63

    // map idx0 -> c0
    %c0_tmp = addl %idx0, 65
    %c0_cond_a = csltl %idx0, 26
    %c0_tmp2 = addl (subl %idx0, 26), 97
    %c0_tmp = sel %c0_cond_a, %c0_tmp, %c0_tmp2
    %c0_cond_b = csltl %idx0, 52
    %c0_tmp3 = addl (subl %idx0, 52), 48
    %c0_tmp = sel %c0_cond_b, %c0_tmp, %c0_tmp3
    %c0_cond_c = csltl %idx0, 62
    %c0_tmp = sel %c0_cond_c, %c0_tmp, 43
    %c0_cond_eq63 = ceql %idx0, 63
    %c0 = sel %c0_cond_eq63, 47, %c0_tmp

    // map idx1 -> c1
    %c1_tmp = addl %idx1, 65
    %c1_cond_a = csltl %idx1, 26
    %c1_tmp2 = addl (subl %idx1, 26), 97
    %c1_tmp = sel %c1_cond_a, %c1_tmp, %c1_tmp2
    %c1_cond_b = csltl %idx1, 52
    %c1_tmp3 = addl (subl %idx1, 52), 48
    %c1_tmp = sel %c1_cond_b, %c1_tmp, %c1_tmp3
    %c1_cond_c = csltl %idx1, 62
    %c1_tmp = sel %c1_cond_c, %c1_tmp, 43
    %c1_cond_eq63 = ceql %idx1, 63
    %c1 = sel %c1_cond_eq63, 47, %c1_tmp

    // map idx2 -> c2
    %c2_tmp = addl %idx2, 65
    %c2_cond_a = csltl %idx2, 26
    %c2_tmp2 = addl (subl %idx2, 26), 97
    %c2_tmp = sel %c2_cond_a, %c2_tmp, %c2_tmp2
    %c2_cond_b = csltl %idx2, 52
    %c2_tmp3 = addl (subl %idx2, 52), 48
    %c2_tmp = sel %c2_cond_b, %c2_tmp, %c2_tmp3
    %c2_cond_c = csltl %idx2, 62
    %c2_tmp = sel %c2_cond_c, %c2_tmp, 43
    %c2_cond_eq63 = ceql %idx2, 63
    %c2 = sel %c2_cond_eq63, 47, %c2_tmp

    %base_dst = mulul %full, 4
    %dst_addr = addl %out, %base_dst
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb 61, (addl %dst_addr, 3)   // '='
    jnz 1, @end

    @end
    ret %out_len
}
