export function l $base64_encode(l %src, l %n, l %out) {
    %full = divl %n, 3
    %rem = modl %n, 3
    %out_len_full = mulul %full, 4
    %rem_gt0 = csltl 0, %rem
    %add4 = sel %rem_gt0, 4, 0
    %out_len = addl %out_len_full, %add4

    %i0 = constl 0
    @loop
    %i = phi %i0, %i_next
    %cond = csltl %i, %full
    jnz %cond, @body
    jnz 1, @rem

    @body
    %src_off = mulul %i, 3
    %src_addr = addl %src, %src_off
    %b0 = loadub %src_addr
    %b1 = loadub (addl %src_addr, 1)
    %b2 = loadub (addl %src_addr, 2)

    %v24 = orl %v24, %b2
    %idx0 = shrul %v24, 18
    %idx1 = andl (shrul %v24, 12), 63
    %idx2 = andl (shrul %v24, 6), 63
    %idx3 = andl %v24, 63

    %c0a = addl %idx0, 65
    %c0cond_a = csltl %idx0, 26
    %c0b = addl (subl %idx0, 26), 97

    %c0c = addl (subl %idx0, 52), 48

    %c0cond_c = csltl %idx0, 62
    %c0tmp3 = sel %c0cond_c, %c0tmp2, 43

    %c2c = addl (subl %idx2, 52), 48

    %c2tmp3 = sel %c2cond_c, %c2tmp2, 43
    %c2cond_eq = ceql %idx2, 63
    %c2 = sel %c2cond_eq, 47, %c2tmp3

    %c3a = addl %idx3, 65
    %c3cond_a = csltl %idx3, 26
    %c3b = addl (subl %idx3, 26), 97
    %c3tmp = sel %c3cond_a, %c3a, %c3b
    %c3cond_b = csltl %idx3, 52
    %c3c = addl (subl %idx3, 52), 48
    %c3tmp2 = sel %c3cond_b, %c3tmp, %c3c
    %c3cond_c = csltl %idx3, 62
    %c3tmp3 = sel %c3cond_c, %c3tmp2, 43
    %c3cond_eq = ceql %idx3, 63
    %c3 = sel %c3cond_eq, 47, %c3tmp3

    %dst_off = mulul %i, 4
    %dst_addr = addl %out, %dst_off
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb %c3, (addl %dst_addr, 3)

    %i_next = addl %i, 1
    jnz 1, @loop

    @rem
    %rem_eq0 = ceql %rem, 0
    jnz %rem_eq0, @end

    %rem_eq1 = ceql %rem, 1
    jnz %rem_eq1, @rem1

    %rem_eq2 = ceql %rem, 2
    jnz %rem_eq2, @rem2

    @rem1
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src
    %b0 = loadub %src_addr
    %v = shll %b0, 16
    %idx0 = shrul %v, 18
    %idx1 = andl (shrul %v, 12), 63

    %c0a = addl %idx0, 65
    %c0cond_a = csltl %idx0, 26
    %c0b = addl (subl %idx0, 26), 97
    %c0tmp = sel %c0cond_a, %c0a, %c0b
    %c0cond_b = csltl %idx0, 52
    %c0c = addl (subl %idx0, 52), 48

    %c0cond_eq = ceql %idx0, 63
    %c0 = sel %c0cond_eq, 47, %c0tmp3

    %c1a = addl %idx1, 65
    %c1cond_a = csltl %idx1, 26
    %c1b = addl (subl %idx1, 26), 97
    %c1tmp = sel %c1cond_a, %c1a, %c1b
    %c1cond_b = csltl %idx1, 52
    %c1c = addl (subl %idx1, 52), 48
    %c1tmp2 = sel %c1cond_b, %c1tmp, %c1c
    %c1cond_c = csltl %idx1, 62
    %c1tmp3 = sel %c1cond_c, %c1tmp2, 43
    %c1cond_eq = ceql %idx1, 63
    %c1 = sel %c1cond_eq, 47, %c1tmp3

    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb 61, (addl %dst_addr, 2)
    storeb 61, (addl %dst_addr, 3)
    jnz 1, @end

    @rem2
    %base_src = mulul %full, 3
    %src_addr = addl %src, %base_src

    %b1 = loadub (addl %src_addr, 1)
    %v0 = shll %b0, 16
    %v1 = shll %b1, 8
    %v = orl %v0, %v1
    %idx0 = shrul %v, 18

    %idx2 = andl (shrul %v, 6), 63

    %c0a = addl %idx0, 65
    %c0cond_a = csltl %idx0, 26
    %c0b = addl (subl %idx0, 26), 97
    %c0tmp = sel %c0cond_a, %c0a, %c0b
    %c0cond_b = csltl %idx0, 52

    %c0tmp2 = sel %c0cond_b, %c0tmp, %c0c

    %c0cond_eq = ceql %idx0, 63
    %c0 = sel %c0cond_eq, 47, %c0tmp3

    %c1a = addl %idx1, 65
    %c1cond_a = csltl %idx1, 26

    %c1c = addl (subl %idx1, 52), 48
    %c1tmp2 = sel %c1cond_b, %c1tmp, %c1c
    %c1cond_c = csltl %idx1, 62
    %c1tmp3 = sel %c1cond_c, %c1tmp2, 43
    %c1cond_eq = ceql %idx1, 63
    %c1 = sel %c1cond_eq, 47, %c1tmp3

    %c2a = addl %idx2, 65
    %c2cond_a = csltl %idx2, 26
    %c2b = addl (subl %idx2, 26), 97
    %c2tmp = sel %c2cond_a, %c2a, %c2b
    %c2cond_b = csltl %idx2, 52
    %c2c = addl (subl %idx2, 52), 48
    %c2tmp2 = sel %c2cond_b, %c2tmp, %c2c
    %c2cond_c = csltl %idx2, 62
    %c2tmp3 = sel %c2cond_c, %c2tmp2, 43
    %c2cond_eq = ceql %idx2, 63
    %c2 = sel %c2cond_eq, 47, %c2tmp3

    %base_dst = mulul %full, 4
    %dst_addr = addl %out, %base_dst
    storeb %c0, %dst_addr
    storeb %c1, (addl %dst_addr, 1)
    storeb %c2, (addl %dst_addr, 2)
    storeb 61, (addl %dst_addr, 3)
    jnz 1, @end

    @end
    ret %out_len
}
