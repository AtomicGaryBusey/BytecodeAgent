bmax:
    ldr x2, [x0]          // load first element as current max
    subs x1, x1, #1       // n--
    beq .Ldone
    add x3, x0, #8        // pointer to second element
.Lloop:
    ldr x4, [x3], #8      // load element, post-increment pointer
    cmp x4, x2
    csel x2, x4, x2, gt
    subs x1, x1, #1
    bne .Lloop
.Ldone:
    mov x0, x2
    ret
