bubble_sort:
    push rbx
    xor ebx, ebx                # i = 0
    cmp rsi, 2
    jl .return
.outer_loop:
    mov rdx, rsi                # rdx = n
    dec rdx                     # rdx = n-1
    sub rdx, rbx                # rdx = n-1-i (inner limit)
    xor ecx, ecx                # j = 0
.inner_loop:
    cmp rcx, rdx
    jge .inner_done
    mov rax, [rdi + rcx*8]      # a[j]
    mov r10, [rdi + rcx*8 + 8]  # a[j+1]
    cmp rax, r10
    jle .no_swap
    mov [rdi + rcx*8], r10
    mov [rdi + rcx*8 + 8], rax
.no_swap:
    inc rcx
    jmp .inner_loop
.inner_done:
    inc rbx
    cmp rbx, rsi
    jl .outer_loop
.return:
    xor eax, eax                # return 0
    pop rbx
    ret
