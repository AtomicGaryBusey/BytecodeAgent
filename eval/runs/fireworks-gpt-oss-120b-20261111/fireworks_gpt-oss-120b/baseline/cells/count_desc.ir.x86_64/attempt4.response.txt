export function l $count_desc(l %a, l %n) {
@entry:
  %c2 = copyl 2
  %lt2 = csltl %n, %c2
  jnz %lt2, @ret0

  %c0 = copyl 0
  %cnt = copyl %c0
  %i = copyl %c0
  %c1 = copyl 1
  %c3 = copyl 3
  %limit = subl %n, %c1

@loop_cond:
  %cond = csltl %i, %limit
  jz %cond, @end_loop

@loop_body:
  %off = shll %i, %c3
  %ptr = addl %a, %off
  %val = loadl %ptr

  %i1 = addl %i, %c1
  %off1 = shll %i1, %c3
  %ptr1 = addl %a, %off1
  %val1 = loadl %ptr1

  %desc_w = csltl %val1, %val
  %inc = zextw %desc_w
  %cnt = addl %cnt, %inc

  %i = copyl %i1
  jmp @loop_cond

@end_loop:
  ret %cnt

@ret0:
  %c0 = copyl 0
  ret %c0
}
