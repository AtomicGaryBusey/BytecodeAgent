export function l $count_desc(l %a, l %n) {
entry:
  %c2 = copyl 2
  %n_lt2 = csltl %n, %c2
  jnz %n_lt2, return0

  %c0 = copyl 0
  %cnt = copyl %c0
  %i = copyl %c0
  %c1 = copyl 1
  %limit = subl %n, %c1

loop:
  %cond = csltl %i, %limit
  jz %cond, endloop

  %c3 = copyl 3
  %off = shll %i, %c3
  %ptr = addl %a, %off
  %val = loadl %ptr

  %i1 = addl %i, %c1
  %off1 = shll %i1, %c3
  %ptr1 = addl %a, %off1
  %val1 = loadl %ptr1

  %desc = csltl %val1, %val
  %desc_l = copyl %desc
  %cnt = addl %cnt, %desc_l

  %i = copyl %i1
  jmp loop

endloop:
  ret %cnt

return0:
  ret %c0
}
