export function l $count_desc(l %a, l %n) {
  @entry:
    %n_lt2 = csltl %n, 2
    jnz %n_lt2, @return0

    %cnt = copyl 0
    %i = copyl 0
    %limit = subl %n, 1

  @loop:
    %cond = csltl %i, %limit
    jz %cond, @endloop

    %off = shll %i, 3
    %ptr = addl %a, %off
    %val = loadl %ptr

    %i1 = addl %i, 1
    %off1 = shll %i1, 3
    %ptr1 = addl %a, %off1
    %val1 = loadl %ptr1

    %desc = csltl %val1, %val
    %cnt = addl %cnt, %desc

    %i = copyl %i1
    jmp @loop

  @endloop:
    ret %cnt

  @return0:
    ret 0
}
