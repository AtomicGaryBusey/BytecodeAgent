export function l $reverse(l %a, l %n) {
@entry:
    %i0 = copy 0
    %j0 = sub %n, 1
    jmp @loop

@loop:
    %i = phi l %i0, @entry, %i_next, @swap
    %j = phi l %j0, @entry, %j_next, @swap
    %cond = slt %i, %j
    jnz %cond, @swap, @exit

@swap:
    %offi = shl %i, 3
    %ptri = add %a, %offi
    %offj = shl %j, 3
    %ptrj = add %a, %offj
    %val_i = load %ptri
    %val_j = load %ptrj
    store %val_j, %ptri
    store %val_i, %ptrj
    %i_next = add %i, 1
    %j_next = sub %j, 1
    jmp @loop

@exit:
    %zero = copy 0
    ret %zero
}
