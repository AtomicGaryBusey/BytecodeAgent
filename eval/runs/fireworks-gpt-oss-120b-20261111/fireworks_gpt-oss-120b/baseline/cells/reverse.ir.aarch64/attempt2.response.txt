export function l $reverse(l %a, l %n) {
@entry
    %i0 = copyl 0
    %j0 = subl %n, 1
    jmp @loop

@loop
    %i = phi l %i0, @entry, %i_next, @swap
    %j = phi l %j0, @entry, %j_next, @swap
    %cond = csltl %i, %j
    jnz %cond, @swap, @exit

@swap
    %offi = shll %i, 3
    %ptri = addl %a, %offi
    %offj = shll %j, 3
    %ptrj = addl %a, %offj
    %val_i = loadl %ptri
    %val_j = loadl %ptrj
    storel %val_j, %ptri
    storel %val_i, %ptrj
    %i_next = addl %i, 1
    %j_next = subl %j, 1
    jmp @loop

@exit
    %zero = copyl 0
    ret %zero
}
