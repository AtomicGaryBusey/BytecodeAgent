export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
    jmp @loop(l const 0, l const 0)

@loop(l %i, l %o) {
    %cond =w cltl %i, %n
    jnz %cond, @proc_header(%i, %o)
    %res =l %o
    ret %res
}

@proc_header(l %i, l %o) {
    %ph =l add %src, %i
    %h =w loadsb %ph
    %i1 =l add %i, l const 1
    %isneg =w csltl %h, 0
    jnz %isneg, @neg_handle(%h, %i1, %o)

    ; literal packet
    %h_u =l zextw %h
    %cnt =l add %h_u, l const 1

    %avail =l sub %n, %i1
    %bad =w cltl %avail, %cnt
    jnz %bad, @err2

    %space =l sub %cap, %o
    %overflow =w cltl %space, %cnt
    jnz %overflow, @literal_overflow(%i1, %o, %cnt, %space)

    jmp @lit_copy(%i1, %o, %cnt, l const 0)
}

@lit_copy(l %src_pos, l %out_pos, l %cnt, l %j) {
    %cnd =w cltl %j, %cnt
    jnz %cnd, @lit_body(%src_pos, %out_pos, %cnt, %j)

    %new_i =l add %src_pos, %cnt
    %new_o =l add %out_pos, %cnt
    jmp @loop(%new_i, %new_o)
}

@lit_body(l %src_pos, l %out_pos, l %cnt, l %j) {
    %src_off =l add %src_pos, %j
    %src_ptr =l add %src, %src_off
    %byte =w loadub %src_ptr
    %dst_off =l add %out_pos, %j
    %dst_ptr =l add %out, %dst_off
    storeb %byte, %dst_ptr
    %j1 =l add %j, l const 1
    jmp @lit_copy(%src_pos, %out_pos, %cnt, %j1)
}

@literal_overflow(l %src_pos, l %out_pos, l %cnt, l %space) {
    %write =l %space
    jmp @lit_over(%src_pos, %out_pos, %write, l const 0)
}

@lit_over(l %src_pos, l %out_pos, l %write, l %j) {
    %cnd =w cltl %j, %write
    jnz %cnd, @lit_over_body(%src_pos, %out_pos, %write, %j)

    %res =l const -1
    ret %res
}

@lit_over_body(l %src_pos, l %out_pos, l %write, l %j) {
    %src_off =l add %src_pos, %j
    %src_ptr =l add %src, %src_off
    %byte =w loadub %src_ptr
    %dst_off =l add %out_pos, %j
    %dst_ptr =l add %out, %dst_off
    storeb %byte, %dst_ptr
    %j1 =l add %j, l const 1
    jmp @lit_over(%src_pos, %out_pos, %write, %j1)
}

@neg_handle(l %h, l %i, l %o) {
    %h_eq128 =w ceql %h, -128
    jnz %h_eq128, @loop(%i, %o)

    %need =l const 1
    %avail =l sub %n, %i
    %bad =w cltl %avail, %need
    jnz %bad, @err2

    %val_ptr =l add %src, %i
    %val =w loadub %val_ptr
    %i1 =l add %i, l const 1

    %h_se =l sextw %h
    %cnt =l sub l const 1, %h_se

    %space =l sub %cap, %o
    %overflow =w cltl %space, %cnt
    jnz %overflow, @repeat_overflow(%i1, %o, %cnt, %space, %val)

    jmp @repeat_copy(%i1, %o, %cnt, %val, l const 0)
}

@repeat_copy(l %src_pos, l %out_pos, l %cnt, l %val, l %k) {
    %cnd =w cltl %k, %cnt
    jnz %cnd, @repeat_body(%src_pos, %out_pos, %cnt, %val, %k)

    %new_o =l add %out_pos, %cnt
    jmp @loop(%src_pos, %new_o)
}

@repeat_body(l %src_pos, l %out_pos, l %cnt, l %val, l %k) {
    %dst_off =l add %out_pos, %k
    %dst_ptr =l add %out, %dst_off
    storeb %val, %dst_ptr
    %k1 =l add %k, l const 1
    jmp @repeat_copy(%src_pos, %out_pos, %cnt, %val, %k1)
}

@repeat_overflow(l %src_pos, l %out_pos, l %cnt, l %space, l %val) {
    %write =l %space
    jmp @repeat_over(%src_pos, %out_pos, %write, %val, l const 0)
}

@repeat_over(l %src_pos, l %out_pos, l %write, l %val, l %k) {
    %cnd =w cltl %k, %write
    jnz %cnd, @repeat_over_body(%src_pos, %out_pos, %write, %val, %k)

    %res =l const -1
    ret %res
}

@repeat_over_body(l %src_pos, l %out_pos, l %write, l %val, l %k) {
    %dst_off =l add %out_pos, %k
    %dst_ptr =l add %out, %dst_off
    storeb %val, %dst_ptr
    %k1 =l add %k, l const 1
    jmp @repeat_over(%src_pos, %out_pos, %write, %val, %k1)
}

@err2 {
    %res =l const -2
    ret %res
}
}
