export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
	%i =l const 0
	%o =l const 0
	%res =l const 0
	jmp @loop

@loop:
	%cond =w cltl %i, %n
	jnz %cond, @proc_header
	%res =l %o
	jmp @end

@proc_header:
	%ph =l add %src, %i
	%h =w loadsb %ph
	%i1 =l add %i, 1
	%i =l %i1

	%isneg =w csltl %h, 0
	jnz %isneg, @negcase

	%h_l =l zextw %h
	%cnt =l add %h_l, 1

	%avail =l sub %n, %i
	%bad =w cltl %avail, %cnt
	jnz %bad, @err2

	%space =l sub %cap, %o
	%overflow =w cltl %space, %cnt
	jnz %overflow, @literal_overflow

	%j =l const 0
@lit_copy_loop:
	%cnd =w cltl %j, %cnt
	jnz %cnd, @lit_copy_body
	%i =l add %i, %cnt
	%o =l add %o, %cnt
	jmp @loop

@lit_copy_body:
	%src_off =l add %i, %j
	%src_ptr =l add %src, %src_off
	%byte =w loadub %src_ptr
	%dst_off =l add %o, %j
	%dst_ptr =l add %out, %dst_off
	storeb %byte, %dst_ptr
	%j =l add %j, 1
	jmp @lit_copy_loop

@literal_overflow:
	%write =l %space
	%j2 =l const 0
@lit_copy_over_loop:
	%cnd2 =w cltl %j2, %write
	jnz %cnd2, @lit_copy_over_body
	%res =l const -1
	jmp @end

@lit_copy_over_body:
	%src_off2 =l add %i, %j2
	%src_ptr2 =l add %src, %src_off2
	%byte2 =w loadub %src_ptr2
	%dst_off2 =l add %o, %j2
	%dst_ptr2 =l add %out, %dst_off2
	storeb %byte2, %dst_ptr2
	%j2 =l add %j2, 1
	jmp @lit_copy_over_loop

@negcase:
	%h_eq_neg128 =w ceql %h, -128
	jnz %h_eq_neg128, @loop

	%need_src =l const 1
	%avail2 =l sub %n, %i
	%bad2 =w cltl %avail2, %need_src
	jnz %bad2, @err2

	%rep_ptr =l add %src, %i
	%val =w loadub %rep_ptr
	%i =l add %i, 1

	%h_l2 =l sextw %h
	%cnt2 =l sub 1, %h_l2

	%space2 =l sub %cap, %o
	%overflow2 =w cltl %space2, %cnt2
	jnz %overflow2, @repeat_overflow

	%k =l const 0
@repeat_loop:
	%cnd3 =w cltl %k, %cnt2
	jnz %cnd3, @repeat_body
	%o =l add %o, %cnt2
	jmp @loop

@repeat_body:
	%dst_off3 =l add %out, %o
	%dst_ptr3 =l add %dst_off3, %k
	storeb %val, %dst_ptr3
	%k =l add %k, 1
	jmp @repeat_loop

@repeat_overflow:
	%write2 =l %space2
	%k2 =l const 0
@repeat_over_loop:
	%cnd4 =w cltl %k2, %write2
	jnz %cnd4, @repeat_over_body
	%res =l const -1
	jmp @end

@repeat_over_body:
	%dst_off4 =l add %out, %o
	%dst_ptr4 =l add %dst_off4, %k2
	storeb %val, %dst_ptr4
	%k2 =l add %k2, 1
	jmp @repeat_over_loop

@err2:
	%res =l const -2
	jmp @end

@end:
	ret %res
}
