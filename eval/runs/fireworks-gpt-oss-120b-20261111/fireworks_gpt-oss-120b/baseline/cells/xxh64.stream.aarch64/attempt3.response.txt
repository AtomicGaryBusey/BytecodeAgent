xxh64:
    // x0 = p, x1 = n, x2 = seed
    // load constants
    movz    x9,  #0xCA87                 // P1 low 16 bits
    movk    x9,  #0x85EB, lsl #16
    movk    x9,  #0x79B1, lsl #32
    movk    x9,  #0x9E37, lsl #48        // P1
    movz    x10, #0xEB4F                 // P2 low
    movk    x10, #0x27D4, lsl #16
    movk    x10, #0xAE3D, lsl #32
    movk    x10, #0xC2B2, lsl #48        // P2
    movz    x11, #0x79F9                 // P3 low
    movk    x11, #0x9E37, lsl #16
    movk    x11, #0x67B1, lsl #32
    movk    x11, #0x1656, lsl #48        // P3
    movz    x12, #0xAE63                 // P4 low
    movk    x12, #0xC2B2, lsl #16
    movk    x12, #0xCA77, lsl #32
    movk    x12, #0x85EB, lsl #48        // P4
    movz    x13, #0x67C5                 // P5 low
    movk    x13, #0x1656, lsl #16
    movk    x13, #0xEB2F, lsl #32
    movk    x13, #0x27D4, lsl #48        // P5

    // test for large input (n >= 32)
    cmp     x1, #32
    blt     .Lsmall_input

    // ------- large input path -------
    // v1 = seed + P1 + P2
    add     x14, x2, x9
    add     x14, x14, x10
    // v2 = seed + P2
    add     x15, x2, x10
    // v3 = seed
    mov     x16, x2
    // v4 = seed - P1
    sub     x17, x2, x9

    // number of 32‑byte stripes = n >> 5
    lsr     x18, x1, #5          // x18 = stripe count
    mov     x3, x0               // x3 = ptr = p

.Lstripe_loop:
    cbz     x18, .Lafter_stripes

    // lane 1
    ldr     x4, [x3], #8
    // round(v1, x4)
    mul     x5, x4, x10          // x5 = x4 * P2
    add     x14, x14, x5
    ror     x14, x14, #33        // rotl 31
    mul     x14, x14, x9

    // lane 2
    ldr     x4, [x3], #8
    mul     x5, x4, x10
    add     x15, x15, x5
    ror     x15, x15, #33
    mul     x15, x15, x9

    // lane 3
    ldr     x4, [x3], #8
    mul     x5, x4, x10
    add     x16, x16, x5
    ror     x16, x16, #33
    mul     x16, x16, x9

    // lane 4
    ldr     x4, [x3], #8
    mul     x5, x4, x10
    add     x17, x17, x5
    ror     x17, x17, #33
    mul     x17, x17, x9

    sub     x18, x18, #1
    b       .Lstripe_loop

.Lafter_stripes:
    // h = rotl(v1,1) + rotl(v2,7) + rotl(v3,12) + rotl(v4,18)
    mov     x0, x14
    ror     x0, x0, #63          // rotl 1
    mov     x4, x15
    ror     x4, x4, #57          // rotl 7
    add     x0, x0, x4
    mov     x4, x16
    ror     x4, x4, #52          // rotl 12
    add     x0, x0, x4
    mov     x4, x17
    ror     x4, x4, #46          // rotl 18
    add     x0, x0, x4

    // merge h with v1..v4
    // merge macro: h = (h ^ round(0,v)) * P1 + P4
    // round(0,v) = rotl(v * P2,31) * P1
    // first v1
    mul     x4, x14, x10          // v1*P2
    ror     x4, x4, #33           // rotl 31
    mul     x4, x4, x9            // *P1
    eor     x0, x0, x4
    mul     x0, x0, x9
    add     x0, x0, x12

    // v2
    mul     x4, x15, x10
    ror     x4, x4, #33
    mul     x4, x4, x9
    eor     x0, x0, x4
    mul     x0, x0, x9
    add     x0, x0, x12

    // v3
    mul     x4, x16, x10
    ror     x4, x4, #33
    mul     x4, x4, x9
    eor     x0, x0, x4
    mul     x0, x0, x9
    add     x0, x0, x12

    // v4
    mul     x4, x17, x10
    ror     x4, x4, #33
    mul     x4, x4, x9
    eor     x0, x0, x4
    mul     x0, x0, x9
    add     x0, x0, x12

    b       .Lpost_merge

.Lsmall_input:
    // h = seed + P5
    add     x0, x2, x13

.Lpost_merge:
    // h += n
    add     x0, x0, x1

    // processed = (n >= 32) ? (n & ~31) : 0
    mov     x4, x1
    cmp     x1, #32
    blt     .Lno_processed
    and     x4, x1, #-32          // clear low 5 bits
.Lno_processed:
    // q = p + processed
    add     x3, x0, #0            // preserve h in x0, compute ptr in x3
    add     x3, x0, #0            // dummy to keep register usage aligned
    add     x3, x0, #0            // (will be overwritten)
    add     x3, x0, #0            // (placeholder)
    // correct computation:
    add     x3, x0, #0            // n/a, redo:
    // Actually we need pointer; p is in x0 at entry, saved? Use x5 to hold original p
    mov     x5, x0                // x5 = p (original pointer)
    add     x3, x5, x4            // x3 = p + processed
    // remaining = n - processed
    sub     x6, x1, x4            // x6 = remaining
    // ----- process 8‑byte chunks -----
.Lwhile8:
    cmp     x6, #8
    blt     .Lafter8
    ldr     x7, [x3], #8
    // round(0, x7)
    mul     x8, x7, x10
    ror     x8, x8, #33
    mul     x8, x8, x9
    eor     x0, x0, x8
    // rotl 27
    ror     x0, x0, #37          // 64-27 = 37
    mul     x0, x0, x9
    add     x0, x0, x12
    sub     x6, x6, #8
    b       .Lwhile8

.Lafter8:
    // ----- process 4‑byte chunk -----
    cmp     x6, #4
    blt     .Lafter4
    ldr     w7, [x3], #4
    uxtw    x8, w7
    mul     x8, x8, x9
    eor     x0, x0, x8
    // rotl 23
    ror     x0, x0, #41          // 64-23 = 41
    mul     x0, x0, x10
    add     x0, x0, x11
    sub     x6, x6, #4

.Lafter4:
    // ----- process remaining bytes -----
.Lbyte_loop:
    cbz     x6, .Lfinal_mix
    ldrb    w7, [x3], #1
    uxtw    x8, w7
    mul     x8, x8, x13
    eor     x0, x0, x8
    // rotl 11
    ror     x0, x0, #53          // 64-11 = 53
    mul     x0, x0, x9
    sub     x6, x6, #1
    b       .Lbyte_loop

.Lfinal_mix:
    // final avalanche
    lsr     x7, x0, #33
    eor     x0, x0, x7
    mul     x0, x0, x10
    lsr     x7, x0, #29
    eor     x0, x0, x7
    mul     x0, x0, x11
    lsr     x7, x0, #32
    eor     x0, x0, x7
    ret
