xxh64:
    // save original arguments
    mov     x24, x0                 // p
    mov     x25, x1                 // n
    mov     x26, x2                 // seed
    // load constants
    movz    x10, #0xCA87            // P1 low
    movk    x10, #0x85EB, lsl #16
    movk    x10, #0x79B1, lsl #32
    movk    x10, #0x9E37, lsl #48   // P1
    movz    x11, #0xEB4F            // P2 low
    movk    x11, #0x27D4, lsl #16
    movk    x11, #0xAE3D, lsl #32
    movk    x11, #0xC2B2, lsl #48   // P2
    movz    x12, #0x79F9            // P3 low
    movk    x12, #0x9E37, lsl #16
    movk    x12, #0x67B1, lsl #32
    movk    x12, #0x1656, lsl #48   // P3
    movz    x13, #0xAE63            // P4 low
    movk    x13, #0xC2B2, lsl #16
    movk    x13, #0xCA77, lsl #32
    movk    x13, #0x85EB, lsl #48   // P4
    movz    x14, #0x67C5            // P5 low
    movk    x14, #0x1656, lsl #16
    movk    x14, #0xEB2F, lsl #32
    movk    x14, #0x27D4, lsl #48   // P5
    // choose path based on length
    cmp     x25, #32
    b.lt    .Lsmall
    // large input path
    // v1 = seed + P1 + P2
    add     x9,  x26, x10
    add     x9,  x9,  x11
    // v2 = seed + P2
    add     x15, x26, x11
    // v3 = seed
    mov     x16, x26
    // v4 = seed - P1
    sub     x17, x26, x10
    // stripe loop
    lsr     x18, x25, #5            // number of 32‑byte stripes
    mov     x19, x24                // q = p
    cbz     x18, .Lafter_stripes
.Lstripe_loop:
    // lane 1
    ldr     x20, [x19], #8
    mul     x21, x20, x11
    add     x9,  x9,  x21
    ror     x9,  x9, #33
    mul     x9,  x9,  x10
    // lane 2
    ldr     x20, [x19], #8
    mul     x21, x20, x11
    add     x15, x15, x21
    ror     x15, x15, #33
    mul     x15, x15, x10
    // lane 3
    ldr     x20, [x19], #8
    mul     x21, x20, x11
    add     x16, x16, x21
    ror     x16, x16, #33
    mul     x16, x16, x10
    // lane 4
    ldr     x20, [x19], #8
    mul     x21, x20, x11
    add     x17, x17, x21
    ror     x17, x17, #33
    mul     x17, x17, x10
    subs    x18, x18, #1
    b.ne    .Lstripe_loop
.Lafter_stripes:
    // h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18)
    mov     x0, x9
    ror     x0, x0, #63            // rotl 1
    mov     x21, x15
    ror     x21, x21, #57          // rotl 7
    add     x0, x0, x21
    mov     x21, x16
    ror     x21, x21, #52          // rotl 12
    add     x0, x0, x21
    mov     x21, x17
    ror     x21, x21, #46          // rotl 18
    add     x0, x0, x21
    // merge with v1..v4
    // merge(h,v) = (h ^ round(0,v))*P1 + P4
    // round(0,v)
    mul     x21, x9,  x11
    ror     x21, x21, #33
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    mul     x0,  x0,  x10
    add     x0,  x0,  x13
    mul     x21, x15, x11
    ror     x21, x21, #33
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    mul     x0,  x0,  x10
    add     x0,  x0,  x13
    mul     x21, x16, x11
    ror     x21, x21, #33
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    mul     x0,  x0,  x10
    add     x0,  x0,  x13
    mul     x21, x17, x11
    ror     x21, x21, #33
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    mul     x0,  x0,  x10
    add     x0,  x0,  x13
    b       .Lfinal_h
.Lsmall:
    // h = seed + P5
    add     x0, x26, x14
.Lfinal_h:
    // h += n
    add     x0, x0, x25
    // processed = (n >= 32) ? (n & ~31) : 0
    mov     x27, x25
    cmp     x25, #32
    b.lt    .Lno_processed
    lsr     x27, x25, #5
    lsl     x27, x27, #5
.Lno_processed:
    // q = p + processed
    add     x28, x24, x27
    // remaining = n - processed
    sub     x29, x25, x27
    // while at least 8 bytes
    cmp     x29, #8
    blt     .Lrem4
.Lloop8:
    ldr     x20, [x28], #8
    mul     x21, x20, x11
    ror     x21, x21, #33
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    ror     x0,  x0, #37          // rotl 27
    mul     x0,  x0,  x10
    add     x0,  x0,  x13
    sub     x29, x29, #8
    cmp     x29, #8
    bge     .Lloop8
.Lrem4:
    cmp     x29, #4
    blt     .Lbyte_loop
    ldr     w20, [x28], #4
    uxtw    x21, w20
    mul     x21, x21, x10
    eor     x0,  x0,  x21
    ror     x0,  x0, #41          // rotl 23
    mul     x0,  x0,  x11
    add     x0,  x0,  x12
    sub     x29, x29, #4
.Lbyte_loop:
    cbz     x29, .Lfinal_mix
    ldrb    w20, [x28], #1
    uxtw    x21, w20
    mul     x21, x21, x14
    eor     x0,  x0,  x21
    ror     x0,  x0, #53          // rotl 11
    mul     x0,  x0,  x10
    sub     x29, x29, #1
    b       .Lbyte_loop
.Lfinal_mix:
    lsr     x21, x0, #33
    eor     x0, x0, x21
    mul     x0, x0, x11
    lsr     x21, x0, #29
    eor     x0, x0, x21
    mul     x0, x0, x12
    lsr     x21, x0, #32
    eor     x0, x0, x21
    ret
