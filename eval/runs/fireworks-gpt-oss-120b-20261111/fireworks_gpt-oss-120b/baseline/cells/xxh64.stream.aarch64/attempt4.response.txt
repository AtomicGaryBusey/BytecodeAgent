xxh64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    mov x19, x0          // p
    mov x20, x1          // n
    mov x21, x2          // seed
    // constants
    movz x3, #0xCA87
    movk x3, #0xEB85, lsl #16
    movk x3, #0x79B1, lsl #32
    movk x3, #0x9E37, lsl #48   // P1
    movz x4, #0xEB4F
    movk x4, #0x27D4, lsl #16
    movk x4, #0xAE3D, lsl #32
    movk x4, #0xC2B2, lsl #48   // P2
    movz x5, #0x79F9
    movk x5, #0x9E37, lsl #16
    movk x5, #0x67B1, lsl #32
    movk x5, #0x1656, lsl #48   // P3
    movz x6, #0xAE63
    movk x6, #0xC2B2, lsl #16
    movk x6, #0xCA77, lsl #32
    movk x6, #0x85EB, lsl #48   // P4
    movz x7, #0x67C5
    movk x7, #0x1656, lsl #16
    movk x7, #0xEB2F, lsl #32
    movk x7, #0x27D4, lsl #48   // P5
    cmp x20, #32
    blt .Lsmall
    // large input path
    add x8, x21, x3
    add x8, x8, x4           // v1 = seed + P1 + P2
    add x9, x21, x4          // v2 = seed + P2
    mov x10, x21             // v3 = seed
    sub x11, x21, x3         // v4 = seed - P1
    mov x12, x19             // ptr
    and x13, x20, #-32       // processed = n & ~31
    add x14, x19, x13        // end = p + processed
.Lstripe_loop:
    cmp x12, x14
    beq .Lafter_stripes
    ldr x15, [x12], #8
    mul x16, x15, x4
    add x8, x8, x16
    ror x8, x8, #33
    mul x8, x8, x3
    ldr x15, [x12], #8
    mul x16, x15, x4
    add x9, x9, x16
    ror x9, x9, #33
    mul x9, x9, x3
    ldr x15, [x12], #8
    mul x16, x15, x4
    add x10, x10, x16
    ror x10, x10, #33
    mul x10, x10, x3
    ldr x15, [x12], #8
    mul x16, x15, x4
    add x11, x11, x16
    ror x11, x11, #33
    mul x11, x11, x3
    b .Lstripe_loop
.Lafter_stripes:
    mov x0, x8
    ror x0, x0, #63          // rotl(v1,1)
    mov x15, x9
    ror x15, x15, #57        // rotl(v2,7)
    add x0, x0, x15
    mov x15, x10
    ror x15, x15, #52        // rotl(v3,12)
    add x0, x0, x15
    mov x15, x11
    ror x15, x15, #46        // rotl(v4,18)
    add x0, x0, x15
    // merge v1..v4
    mul x15, x8, x4
    ror x15, x15, #33
    mul x15, x15, x3
    eor x0, x0, x15
    mul x0, x0, x3
    add x0, x0, x6
    mul x15, x9, x4
    ror x15, x15, #33
    mul x15, x15, x3
    eor x0, x0, x15
    mul x0, x0, x3
    add x0, x0, x6
    mul x15, x10, x4
    ror x15, x15, #33
    mul x15, x15, x3
    eor x0, x0, x15
    mul x0, x0, x3
    add x0, x0, x6
    mul x15, x11, x4
    ror x15, x15, #33
    mul x15, x15, x3
    eor x0, x0, x15
    mul x0, x0, x3
    add x0, x0, x6
    b .Lpost_small
.Lsmall:
    add x0, x21, x7          // h = seed + P5
.Lpost_small:
    add x0, x0, x20          // h += n
    // pointer to remaining data
    add x12, x19, x13        // x12 = p + processed
    sub x13, x20, x13        // x13 = remaining (0..31)
.Lwhile8:
    cmp x13, #8
    blt .Lafter8
    ldr x15, [x12], #8
    mul x16, x15, x4
    ror x16, x16, #33
    mul x16, x16, x3
    eor x0, x0, x16
    ror x0, x0, #37          // rotl 27
    mul x0, x0, x3
    add x0, x0, x6
    sub x13, x13, #8
    b .Lwhile8
.Lafter8:
    cmp x13, #4
    blt .Lafter4
    ldr w15, [x12], #4
    uxtw x16, w15
    mul x16, x16, x3
    eor x0, x0, x16
    ror x0, x0, #41          // rotl 23
    mul x0, x0, x4
    add x0, x0, x5
    sub x13, x13, #4
.Lafter4:
.Lbyte_loop:
    cbz x13, .Lfinal
    ldrb w15, [x12], #1
    uxtw x16, w15
    mul x16, x16, x7
    eor x0, x0, x16
    ror x0, x0, #53          // rotl 11
    mul x0, x0, x3
    sub x13, x13, #1
    b .Lbyte_loop
.Lfinal:
    lsr x15, x0, #33
    eor x0, x0, x15
    mul x0, x0, x4
    lsr x15, x0, #29
    eor x0, x0, x15
    mul x0, x0, x5
    lsr x15, x0, #32
    eor x0, x0, x15
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
