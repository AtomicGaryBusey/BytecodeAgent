qsort_pairs:
    push r14
    mov r14, rdi
    cmp rsi, 1
    jle .ret_zero
.top:
    mov r8, rsi
    shr r8, 1
    mov rax, r8
    shl rax, 4
    add rax, r14
    mov rdx, qword ptr [rax]          # pivot key
    mov rcx, qword ptr [rax+8]        # pivot value
    xor r10, r10                      # i = 0
    mov r11, rsi
    dec r11                           # j = n-1
.part_loop:
    mov rax, r10
    shl rax, 4
    add rax, r14                      # ptr_i
    mov r12, qword ptr [rax]          # key_i
    cmp r12, rdx
    jl .inc_i
    jne .check_j
    mov r12, qword ptr [rax+8]        # value_i
    cmp r12, rcx
    jl .inc_i
.check_j:
    mov r13, r11
    shl r13, 4
    add r13, r14                      # ptr_j
    mov r12, qword ptr [r13]          # key_j
    cmp r12, rdx
    jg .dec_j
    jne .swap_check
    mov r12, qword ptr [r13+8]        # value_j
    cmp r12, rcx
    jg .dec_j
.swap_check:
    cmp r10, r11
    jg .after_part
    mov r9, qword ptr [rax]           # temp key
    mov r12, qword ptr [r13]
    mov qword ptr [rax], r12
    mov qword ptr [r13], r9
    mov r9, qword ptr [rax+8]         # temp value
    mov r12, qword ptr [r13+8]
    mov qword ptr [rax+8], r12
    mov qword ptr [r13+8], r9
    inc r10
    dec r11
    jmp .part_loop
.inc_i:
    inc r10
    jmp .part_loop
.dec_j:
    dec r11
    jmp .part_loop
.after_part:
    lea r12, [r11+1]                  # left_n
    mov r13, rsi
    sub r13, r10                      # right_n
    cmp r12, r13
    jle .recurse_left_first
    test r13, r13
    jz .recurse_left_only
    sub rsp, 8
    mov qword ptr [rsp], r12         # save left_n
    mov rax, r10
    shl rax, 4
    add rax, r14
    mov rdi, rax
    mov rsi, r13
    call qsort_pairs
    mov r12, qword ptr [rsp]
    add rsp, 8
    mov rsi, r12
    mov rdi, r14
    jmp .top
.recurse_left_only:
    test r12, r12
    jz .ret_zero
    mov rsi, r12
    mov rdi, r14
    jmp .top
.recurse_left_first:
    test r12, r12
    jz .recurse_right_only
    sub rsp, 8
    mov qword ptr [rsp], r13         # save right_n
    mov rdi, r14
    mov rsi, r12
    call qsort_pairs
    mov r13, qword ptr [rsp]
    add rsp, 8
    test r13, r13
    jz .ret_zero
    mov rax, r10
    shl rax, 4
    add rax, r14
    mov rdi, rax
    mov rsi, r13
    jmp .top
.recurse_right_only:
    test r13, r13
    jz .ret_zero
    mov rax, r10
    shl rax, 4
    add rax, r14
    mov rdi, rax
    mov rsi, r13
    jmp .top
.ret_zero:
    xor rax, rax
    pop r14
    ret
