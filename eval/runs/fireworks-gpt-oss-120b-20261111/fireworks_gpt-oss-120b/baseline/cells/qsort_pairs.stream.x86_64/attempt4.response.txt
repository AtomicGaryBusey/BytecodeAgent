qsort_pairs:
    push rbx
    mov rbx, rdi                     # rbx = base pointer a
    cmp rsi, 1
    jle .ret0
    # choose pivot = middle element
    mov rax, rsi
    shr rax, 1                       # rax = n/2
    lea rdx, [rax*8]
    add rdx, rdx                     # rdx = idx*16
    mov r10, [rbx + rdx]             # pivot key
    mov r11, [rbx + rdx + 8]         # pivot value
    xor r8, r8                       # i = 0
    mov r9, rsi
    dec r9                           # j = n-1
.partition_loop:
    cmp r8, r9
    jg .after_partition
.left_scan:
    mov rax, r8
    lea rax, [rax*8]
    add rax, rax                     # rax = i*16
    mov r12, [rbx + rax]             # key_i
    mov r13, [rbx + rax + 8]         # val_i
    cmp r12, r10
    jl .inc_i
    jne .left_done
    cmp r13, r11
    jl .inc_i
.left_done:
    jmp .right_scan
.inc_i:
    inc r8
    cmp r8, r9
    jle .left_scan
    jmp .after_partition
.right_scan:
    mov rax, r9
    lea rax, [rax*8]
    add rax, rax                     # rax = j*16
    mov r12, [rbx + rax]             # key_j
    mov r13, [rbx + rax + 8]         # val_j
    cmp r12, r10
    jg .dec_j
    jne .right_done
    cmp r13, r11
    jg .dec_j
.right_done:
    # swap records i and j
    mov rax, r8
    lea rax, [rax*8]
    add rax, rax                     # rax = i*16
    mov rdx, r9
    lea rdx, [rdx*8]
    add rdx, rdx                     # rdx = j*16
    mov rcx, [rbx + rax]             # key_i
    mov rsi, [rbx + rax + 8]         # val_i
    mov r14, [rbx + rdx]             # key_j
    mov r15, [rbx + rdx + 8]         # val_j
    mov [rbx + rax], r14
    mov [rbx + rax + 8], r15
    mov [rbx + rdx], rcx
    mov [rbx + rdx + 8], rsi
    inc r8
    dec r9
    jmp .partition_loop
.dec_j:
    dec r9
    cmp r8, r9
    jle .right_scan
    jmp .after_partition
.after_partition:
    # left_len = j+1, right_len = n - i
    mov rax, r9
    inc rax                         # rax = left_len
    mov rcx, rsi
    sub rcx, r8                     # rcx = right_len
    cmp rax, rcx
    jle .left_smaller
    # right part is smaller
    mov r12, rax                    # save left_len
    test rcx, rcx
    jz .skip_right_call
    mov rdx, r8
    lea rdx, [rdx*8]
    add rdx, rdx
    lea rdi, [rbx + rdx]
    mov rsi, rcx
    call qsort_pairs
.skip_right_call:
    test r12, r12
    jz .done_recursions
    mov rdi, rbx
    mov rsi, r12
    call qsort_pairs
    jmp .done_recursions
.left_smaller:
    mov r12, rax        # save left_len
    mov r13, rcx        # save right_len
    test r12, r12
    jz .skip_left_call
    mov rdi, rbx
    mov rsi, r12
    call qsort_pairs
.skip_left_call:
    test r13, r13
    jz .done_recursions
    mov rdx, r12
    lea rdx, [rdx*8]
    add rdx, rdx
    lea rdi, [rbx + rdx]
    mov rsi, r13
    call qsort_pairs
.done_recursions:
.ret0:
    xor rax, rax
    pop rbx
    ret
