export function l $qsort_pairs(l %a, l %n) {
    @entry
    %zero =l const 0
    %one  =l const 1
    %two  =l const 2
    %sixteen =l const 16
    %twentyfour =l const 24

    %cond_n_lt2 = csltl %n, %two
    jnz %cond_n_lt2, @ret0

    sub %sp, %twentyfour          // align stack, allocate 24 bytes

    %pivot_idx = udiv %n, %two
    %pivot_off = mul %pivot_idx, %sixteen
    %pivot_ptr = add %a, %pivot_off

    %pivot_key = loadl (%pivot_ptr)
    %pivot_val = loadl 8(%pivot_ptr)

    storel %pivot_key, (%sp)
    storel %pivot_val, 8(%sp)

    %i =l const 0
    %j =l sub %n, %one

    @outer_loop
        // advance i while record[i] < pivot
        @inc_i
            %i_off = mul %i, %sixteen
            %i_ptr = add %a, %i_off
            %ik = loadl (%i_ptr)
            %iv = loadl 8(%i_ptr)

            %ck  = csltl %ik, %pivot_key
            %ceq = ceql %ik, %pivot_key
            %cv  = csltl %iv, %pivot_val
            %cond_i_lt = or %ck, (and %ceq, %cv)
            jnz %cond_i_lt, @inc_i_inc
            jmp @done_inc_i
        @inc_i_inc
            %i = add %i, %one
            jmp @inc_i
        @done_inc_i

        // advance j while record[j] > pivot
        @dec_j
            %j_off = mul %j, %sixteen
            %j_ptr = add %a, %j_off
            %jk = loadl (%j_ptr)
            %jv = loadl 8(%j_ptr)

            %ckg  = csltl %pivot_key, %jk
            %ceqg = ceql %pivot_key, %jk
            %cvg  = csltl %pivot_val, %jv
            %cond_j_gt = or %ckg, (and %ceqg, %cvg)
            jnz %cond_j_gt, @dec_j_dec
            jmp @done_dec_j
        @dec_j_dec
            %j = sub %j, %one
            jmp @dec_j
        @done_dec_j

        // if i >= j break
        %cmpij = csltl %i, %j
        jz %cmpij, @after_partition

        // swap records i and j
        storel %jk, (%i_ptr)
        storel %jv, 8(%i_ptr)
        storel %ik, (%j_ptr)
        storel %iv, 8(%j_ptr)

        %i = add %i, %one
        %j = sub %j, %one
        jmp @outer_loop
    @after_partition

    %left_n  = %i
    %right_n = sub %n, %i

    %cond_left = csltl %zero, %left_n
    jnz %cond_left, @recur_left
    jmp @skip_left
    @recur_left
        call $qsort_pairs(l %a, l %left_n)
    @skip_left

    %cond_right = csltl %zero, %right_n
    jnz %cond_right, @recur_right
    jmp @skip_right
    @recur_right
        %right_off = mul %i, %sixteen
        %right_ptr = add %a, %right_off
        call $qsort_pairs(l %right_ptr, l %right_n)
    @skip_right

    add %sp, %twentyfour

    @ret0
    ret l 0
}
