export function l $qsort_pairs(l %a, l %n) {
@entry
    %zero =l 0
    %one =l 1
    %two =l 2
    %eight =l 8
    %sixteen =l 16

    %cond_n_lt2 = csltl %n, %two
    jnz %cond_n_lt2, @ret

    %pivot_idx = udiv %n, %two
    %pivot_off = mul %pivot_idx, %sixteen
    %pivot_ptr = add %a, %pivot_off
    %pivot_key = loadl (%pivot_ptr)
    %pivot_val = loadl 8(%pivot_ptr)

    %i =l %zero
    %j =l sub %n, %one

@loop
    %i_off = mul %i, %sixteen
    %i_ptr = add %a, %i_off
    %ik = loadl (%i_ptr)
    %iv = loadl 8(%i_ptr)

    %ck = csltl %ik, %pivot_key
    %ceq = ceql %ik, %pivot_key
    %cv = csltl %iv, %pivot_val
    %eq_and_cv = and %ceq, %cv
    %i_lt = or %ck, %eq_and_cv
    jnz %i_lt, @inc_i

    %j_off = mul %j, %sixteen
    %j_ptr = add %a, %j_off
    %jk = loadl (%j_ptr)
    %jv = loadl 8(%j_ptr)

    %ckg = csltl %pivot_key, %jk
    %ceqg = ceql %pivot_key, %jk
    %cvg = csltl %pivot_val, %jv
    %eq_and_cvg = and %ceqg, %cvg
    %j_gt = or %ckg, %eq_and_cvg
    jnz %j_gt, @dec_j

    %cmpij = csltl %i, %j
    jz %cmpij, @after_partition

    storel %jk, (%i_ptr)
    storel %jv, 8(%i_ptr)
    storel %ik, (%j_ptr)
    storel %iv, 8(%j_ptr)

    %i = add %i, %one
    %j = sub %j, %one
    jmp @loop

@inc_i
    %i = add %i, %one
    jmp @loop

@dec_j
    %j = sub %j, %one
    jmp @loop

@after_partition
    %left_n = %i
    %right_n = sub %n, %i

    %cond_left = csltl %zero, %left_n
    jnz %cond_left, @recur_left
    jmp @skip_left
@recur_left
    %sp = sub %sp, %eight
    %tmp =l call $qsort_pairs(l %a, l %left_n)
    %sp = add %sp, %eight
@skip_left

    %cond_right = csltl %zero, %right_n
    jnz %cond_right, @recur_right
    jmp @skip_right
@recur_right
    %right_off = mul %i, %sixteen
    %right_ptr = add %a, %right_off
    %sp = sub %sp, %eight
    %tmp2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    %sp = add %sp, %eight
@skip_right

@ret
    ret l %zero
}
