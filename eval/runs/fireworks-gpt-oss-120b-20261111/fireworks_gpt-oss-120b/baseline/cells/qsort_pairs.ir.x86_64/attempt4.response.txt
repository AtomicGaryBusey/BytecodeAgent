export function l $qsort_pairs(l %a, l %n) {
@entry
    %zero =l iconst 0
    %one =l iconst 1
    %two =l iconst 2
    %eight =l iconst 8
    %sixteen =l iconst 16

    %cond_n_lt2 =w csltl %n, %two
    jnz %cond_n_lt2, @ret

    %pivot_idx =l udiv %n, %two
    %pivot_off =l mul %pivot_idx, %sixteen
    %pivot_ptr =l add %a, %pivot_off
    %pivot_key =l loadl (%pivot_ptr)
    %pivot_val =l loadl 8(%pivot_ptr)

    %i =l copy %zero
    %j =l sub %n, %one

@loop
    %i_off =l mul %i, %sixteen
    %i_ptr =l add %a, %i_off
    %ik =l loadl (%i_ptr)
    %iv =l loadl 8(%i_ptr)

    %ck =w csltl %ik, %pivot_key
    %ceq =w ceql %ik, %pivot_key
    %cv =w csltl %iv, %pivot_val
    %eq_and_cv =w and %ceq, %cv
    %i_lt =w or %ck, %eq_and_cv
    jnz %i_lt, @inc_i

    %j_off =l mul %j, %sixteen
    %j_ptr =l add %a, %j_off
    %jk =l loadl (%j_ptr)
    %jv =l loadl 8(%j_ptr)

    %ckg =w csltl %pivot_key, %jk
    %ceqg =w ceql %pivot_key, %jk
    %cvg =w csltl %pivot_val, %jv
    %eq_and_cvg =w and %ceqg, %cvg
    %j_gt =w or %ckg, %eq_and_cvg
    jnz %j_gt, @dec_j

    %cmpij =w csltl %i, %j
    jz %cmpij, @after_partition

    storel %jk, (%i_ptr)
    storel %jv, 8(%i_ptr)
    storel %ik, (%j_ptr)
    storel %iv, 8(%j_ptr)

    %i =l add %i, %one
    %j =l sub %j, %one
    jmp @loop

@inc_i
    %i =l add %i, %one
    jmp @loop

@dec_j
    %j =l sub %j, %one
    jmp @loop

@after_partition
    %left_n =l copy %i
    %right_n =l sub %n, %i

    %cond_left =w csltl %zero, %left_n
    jnz %cond_left, @recur_left
    jmp @skip_left
@recur_left
    %sp = sub %sp, %eight
    %tmp =l call $qsort_pairs(l %a, l %left_n)
    %sp = add %sp, %eight
@skip_left

    %cond_right =w csltl %zero, %right_n
    jnz %cond_right, @recur_right
    jmp @skip_right
@recur_right
    %right_off =l mul %i, %sixteen
    %right_ptr =l add %a, %right_off
    %sp = sub %sp, %eight
    %tmp2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    %sp = add %sp, %eight
@skip_right

@ret
    ret l %zero
}
