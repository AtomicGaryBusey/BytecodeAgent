base64_encode:
    // compute groups = (n+2)/3
    add     x4, x1, #2
    mov     x5, #3
    udiv    x5, x4, x5          // x5 = groups
    lsl     x6, x5, #2          // x6 = output length (return value)

    // x0 = src, x2 = out, x1 = n (remaining bytes)
    mov     x3, x1              // x3 = remaining count

    // main loop: process full 3-byte chunks
.Lloop:
    cmp     x3, #3
    b.lt    .Lremainder

    // load three bytes
    ldrb    w7, [x0], #1
    ldrb    w8, [x0], #1
    ldrb    w9, [x0], #1

    // idx0 = b0 >> 2
    lsr     w10, w7, #2
    // idx1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and     w11, w7, #0x3
    lsl     w11, w11, #4
    lsr     w12, w8, #4
    orr     w11, w11, w12
    // idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and     w13, w8, #0xF
    lsl     w13, w13, #2
    lsr     w14, w9, #6
    orr     w13, w13, w14
    // idx3 = b2 & 0x3F
    and     w14, w9, #0x3F

    // map idx0 -> char in w10
    cmp     w10, #26
    b.lt    .Lmap0_AZ
    cmp     w10, #52
    b.lt    .Lmap0_az
    cmp     w10, #62
    b.lt    .Lmap0_09
    cmp     w10, #62
    b.eq    .Lmap0_plus
    b       .Lmap0_slash
.Lmap0_AZ:
    add     w10, w10, #65
    b       .Lstore0
.Lmap0_az:
    add     w10, w10, #71
    b       .Lstore0
.Lmap0_09:
    sub     w10, w10, #4
    b       .Lstore0
.Lmap0_plus:
    mov     w10, #43
    b       .Lstore0
.Lmap0_slash:
    mov     w10, #47
.Lstore0:
    strb    w10, [x2], #1

    // map idx1 -> char in w11
    cmp     w11, #26
    b.lt    .Lmap1_AZ
    cmp     w11, #52
    b.lt    .Lmap1_az
    cmp     w11, #62
    b.lt    .Lmap1_09
    cmp     w11, #62
    b.eq    .Lmap1_plus
    b       .Lmap1_slash
.Lmap1_AZ:
    add     w11, w11, #65
    b       .Lstore1
.Lmap1_az:
    add     w11, w11, #71
    b       .Lstore1
.Lmap1_09:
    sub     w11, w11, #4
    b       .Lstore1
.Lmap1_plus:
    mov     w11, #43
    b       .Lstore1
.Lmap1_slash:
    mov     w11, #47
.Lstore1:
    strb    w11, [x2], #1

    // map idx2 -> char in w13
    cmp     w13, #26
    b.lt    .Lmap2_AZ
    cmp     w13, #52
    b.lt    .Lmap2_az
    cmp     w13, #62
    b.lt    .Lmap2_09
    cmp     w13, #62
    b.eq    .Lmap2_plus
    b       .Lmap2_slash
.Lmap2_AZ:
    add     w13, w13, #65
    b       .Lstore2
.Lmap2_az:
    add     w13, w13, #71
    b       .Lstore2
.Lmap2_09:
    sub     w13, w13, #4
    b       .Lstore2
.Lmap2_plus:
    mov     w13, #43
    b       .Lstore2
.Lmap2_slash:
    mov     w13, #47
.Lstore2:
    strb    w13, [x2], #1

    // map idx3 -> char in w14
    cmp     w14, #26
    b.lt    .Lmap3_AZ
    cmp     w14, #52
    b.lt    .Lmap3_az
    cmp     w14, #62
    b.lt    .Lmap3_09
    cmp     w14, #62
    b.eq    .Lmap3_plus
    b       .Lmap3_slash
.Lmap3_AZ:
    add     w14, w14, #65
    b       .Lstore3
.Lmap3_az:
    add     w14, w14, #71
    b       .Lstore3
.Lmap3_09:
    sub     w14, w14, #4
    b       .Lstore3
.Lmap3_plus:
    mov     w14, #43
    b       .Lstore3
.Lmap3_slash:
    mov     w14, #47
.Lstore3:
    strb    w14, [x2], #1

    // decrement remaining count
    sub     x3, x3, #3
    b       .Lloop

    // handle remaining 0,1,2 bytes
.Lremainder:
    cmp     x3, #0
    beq     .Ldone

    cmp     x3, #1
    beq     .Lrem1

    // remaining == 2
.Lrem2:
    ldrb    w7, [x0], #1
    ldrb    w8, [x0]            // last byte, no post-inc needed

    // idx0 = b0 >> 2
    lsr     w10, w7, #2
    // idx1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and     w11, w7, #0x3
    lsl     w11, w11, #4
    lsr     w12, w8, #4
    orr     w11, w11, w12
    // idx2 = (b1 & 0xF) << 2
    and     w13, w8, #0xF
    lsl     w13, w13, #2

    // map idx0
    cmp     w10, #26
    b.lt    .LmapR0_AZ
    cmp     w10, #52
    b.lt    .LmapR0_az
    cmp     w10, #62
    b.lt    .LmapR0_09
    cmp     w10, #62
    b.eq    .LmapR0_plus
    b       .LmapR0_slash
.LmapR0_AZ:
    add     w10, w10, #65
    b       .LstoreR0
.LmapR0_az:
    add     w10, w10, #71
    b       .LstoreR0
.LmapR0_09:
    sub     w10, w10, #4
    b       .LstoreR0
.LmapR0_plus:
    mov     w10, #43
    b       .LstoreR0
.LmapR0_slash:
    mov     w10, #47
.LstoreR0:
    strb    w10, [x2], #1

    // map idx1
    cmp     w11, #26
    b.lt    .LmapR1_AZ
    cmp     w11, #52
    b.lt    .LmapR1_az
    cmp     w11, #62
    b.lt    .LmapR1_09
    cmp     w11, #62
    b.eq    .LmapR1_plus
    b       .LmapR1_slash
.LmapR1_AZ:
    add     w11, w11, #65
    b       .LstoreR1
.LmapR1_az:
    add     w11, w11, #71
    b       .LstoreR1
.LmapR1_09:
    sub     w11, w11, #4
    b       .LstoreR1
.LmapR1_plus:
    mov     w11, #43
    b       .LstoreR1
.LmapR1_slash:
    mov     w11, #47
.LstoreR1:
    strb    w11, [x2], #1

    // map idx2
    cmp     w13, #26
    b.lt    .LmapR2_AZ
    cmp     w13, #52
    b.lt    .LmapR2_az
    cmp     w13, #62
    b.lt    .LmapR2_09
    cmp     w13, #62
    b.eq    .LmapR2_plus
    b       .LmapR2_slash
.LmapR2_AZ:
    add     w13, w13, #65
    b       .LstoreR2
.LmapR2_az:
    add     w13, w13, #71
    b       .LstoreR2
.LmapR2_09:
    sub     w13, w13, #4
    b       .LstoreR2
.LmapR2_plus:
    mov     w13, #43
    b       .LstoreR2
.LmapR2_slash:
    mov     w13, #47
.LstoreR2:
    strb    w13, [x2], #1

    // two padding '=' characters
    mov     w0, #61
    strb    w0, [x2], #1
    strb    w0, [x2], #1
    b       .Ldone

    // remaining == 1
.Lrem1:
    ldrb    w7, [x0]            // single byte

    // idx0 = b0 >> 2
    lsr     w10, w7, #2
    // idx1 = (b0 & 0x3) << 4
    and     w11, w7, #0x3
    lsl     w11, w11, #4

    // map idx0
    cmp     w10, #26
    b.lt    .Lmap1R0_AZ
    cmp     w10, #52
    b.lt    .Lmap1R0_az
    cmp     w10, #62
    b.lt    .Lmap1R0_09
    cmp     w10, #62
    b.eq    .Lmap1R0_plus
    b       .Lmap1R0_slash
.Lmap1R0_AZ:
    add     w10, w10, #65
    b       .Lstore1R0
.Lmap1R0_az:
    add     w10, w10, #71
    b       .Lstore1R0
.Lmap1R0_09:
    sub     w10, w10, #4
    b       .Lstore1R0
.Lmap1R0_plus:
    mov     w10, #43
    b       .Lstore1R0
.Lmap1R0_slash:
    mov     w10, #47
.Lstore1R0:
    strb    w10, [x2], #1

    // map idx1
    cmp     w11, #26
    b.lt    .Lmap1R1_AZ
    cmp     w11, #52
    b.lt    .Lmap1R1_az
    cmp     w11, #62
    b.lt    .Lmap1R1_09
    cmp     w11, #62
    b.eq    .Lmap1R1_plus
    b       .Lmap1R1_slash
.Lmap1R1_AZ:
    add     w11, w11, #65
    b       .Lstore1R1
.Lmap1R1_az:
    add     w11, w11, #71
    b       .Lstore1R1
.Lmap1R1_09:
    sub     w11, w11, #4
    b       .Lstore1R1
.Lmap1R1_plus:
    mov     w11, #43
    b       .Lstore1R1
.Lmap1R1_slash:
    mov     w11, #47
.Lstore1R1:
    strb    w11, [x2], #1

    // two padding '=' characters
    mov     w0, #61
    strb    w0, [x2], #1
    strb    w0, [x2], #1
    b       .Ldone

.Ldone:
    // return length in x0
    mov     x0, x6
    ret
