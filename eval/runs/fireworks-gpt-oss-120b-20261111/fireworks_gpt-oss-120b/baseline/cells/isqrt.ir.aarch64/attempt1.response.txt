export function l $isqrt(l %n) {
  %low0 = constl 0
  %high0 = constl 4294967296
  jmp @loop

@loop
  %low = phi l(%low0, %entry, %low_next, %merge)
  %high = phi l(%high0, %entry, %high_next, %merge)
  %cond = cul %low, %high
  jz %cond, @exit
  %sum = addl %low, %high
  %mid = shrl %sum, constl 1
  %den = udivl %n, %mid
  %gt = cugt %mid, %den
  jnz %gt, @gt_branch
  jmp @le_branch

@le_branch
  %newlow = addl %mid, constl 1
  %low_next = %newlow
  %high_next = %high
  jmp @merge

@gt_branch
  %newhigh = %mid
  %low_next = %low
  %high_next = %newhigh
  jmp @merge

@merge
  %low_next = phi l(%newlow, %le_branch, %low, %gt_branch)
  %high_next = phi l(%high, %le_branch, %newhigh, %gt_branch)
  jmp @loop

@exit
  %res = subl %low, constl 1
  ret %res
}
