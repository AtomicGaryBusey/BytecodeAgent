export function l $isqrt(l %n) {
@entry
  %zero = constl 0
  %one = constl 1
  %high0 = constl 4294967296
  jmp @loop

@loop
  %low = phi l(%zero, @entry, %low_next, @merge)
  %high = phi l(%high0, @entry, %high_next, @merge)
  %cond = cult %low %high
  jz %cond, @exit
  %sum = addl %low %high
  %mid = shrl %sum %one
  %mid_is_zero = ceql %mid %zero
  jnz %mid_is_zero, @mid_zero
  jmp @mid_nonzero

@mid_zero
  %low_next = %one
  %high_next = %high
  jmp @merge

@mid_nonzero
  %div = udivl %n %mid
  %lt = cult %div %mid
  jnz %lt, @gt
  jmp @le

@gt
  %low_next = %low
  %high_next = %mid
  jmp @merge

@le
  %low_next = addl %mid %one
  %high_next = %high
  jmp @merge

@merge
  jmp @loop

@exit
  %res = subl %low %one
  ret %res
}
