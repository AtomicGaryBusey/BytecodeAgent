export function l $isqrt(l %n) {
@entry
  %zero = const 0
  %one = const 1
  %high0 = const 4294967296
  jmp @loop

@loop
  %low = phi l(%zero, @entry, %low_next, @merge)
  %high = phi l(%high0, @entry, %high_next, @merge)
  %cond = ult %low %high
  jz %cond, @exit
  %sum = add %low %high
  %mid = shr %sum %one
  %mid_is_zero = eq %mid %zero
  jnz %mid_is_zero, @mid_zero

  %div = udiv %n %mid
  %lt = ult %div %mid
  jnz %lt, @gt

  %low_next = add %mid %one
  %high_next = %high
  jmp @merge

@mid_zero
  %low_next = %one
  %high_next = %high
  jmp @merge

@gt
  %low_next = %low
  %high_next = %mid
  jmp @merge

@merge
  jmp @loop

@exit
  %res = sub %low %one
  ret %res
}
