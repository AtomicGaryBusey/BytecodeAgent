    jz .done
    mov r8, rsi
    shr r8, 1
    xor rcx, rcx
.Lloop:
    cmp rcx, r8
    jge .done
    mov r9, rsi
    dec r9
    sub r9, rcx
    mov rax, qword ptr [rdi + rcx*8]
    mov r10, qword ptr [rdi + r9*8]
    mov qword ptr [rdi + rcx*8], r10
    mov qword ptr [rdi + r9*8], rax
    inc rcx
    jmp .Lloop
.done:
    xor rax, rax
    ret
