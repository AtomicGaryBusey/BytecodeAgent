export function l $fib(l) {
@entry:
  %c2 = addl 0, 2
  %c0 = addl 0, 0
  %cond = csltl %0, %c2
  jnz %cond, @ret_small
  jmp @preloop
@preloop:
  %a0 = addl 0, 0
  %b0 = addl 0, 1
  %i0 = addl %0, -1
  jmp @loop
@loop:
  %a = phi %a0, @preloop, %a_next, @loop
  %b = phi %b0, @preloop, %b_next, @loop
  %i = phi %i0, @preloop, %i_next, @loop
  %gt = csgtl %i, %c0
  jnz %gt, @body
  jmp @exit
@body:
  %c = addl %a, %b
  %a_next = %b
  %b_next = %c
  %i_next = addl %i, -1
  jmp @loop
@exit:
  ret %b
@ret_small:
  ret %0
}
