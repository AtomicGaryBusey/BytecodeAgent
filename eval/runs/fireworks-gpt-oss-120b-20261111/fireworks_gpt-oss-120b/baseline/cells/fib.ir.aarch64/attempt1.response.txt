export function l $fib(l n) {
@entry:
 %c0 = csltl n, 2
 jnz %c0, @ret_n

 %i0 = addl 0, 2
 %a0 = addl 0, 0
 %b0 = addl 0, 1
 jmp @loop

@loop:
 %i = phi %i0, %i_next
 %a = phi %a0, %a_next
 %b = phi %b0, %b_next
 %gt = csgtl %i, n
 jnz %gt, @exit
 %c = addl %a, %b
 %i_next = addl %i, 1
 %a_next = %b
 %b_next = %c
 jmp @loop

@exit:
 ret %b

@ret_n:
 ret n
}
