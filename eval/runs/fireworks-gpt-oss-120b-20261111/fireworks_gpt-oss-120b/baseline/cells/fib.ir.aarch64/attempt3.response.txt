export function l $fib(l) {
@entry:
 %c0 = csltl %0, 2
 jnz %c0, @ret_small
 %i_init = addl 2, 0
 %a_init = addl 0, 0
 %b_init = addl 0, 1
 jmp @loop
@loop:
 %i = phi %i_init, %i_next
 %a = phi %a_init, %a_next
 %b = phi %b_init, %b_next
 %gt = csgtl %i, %0
 jnz %gt, @exit
 %c = addl %a, %b
 %i_next = addl %i, 1
 %a_next = %b
 %b_next = %c
 jmp @loop
@exit:
 ret %b
@ret_small:
 ret %0
}
