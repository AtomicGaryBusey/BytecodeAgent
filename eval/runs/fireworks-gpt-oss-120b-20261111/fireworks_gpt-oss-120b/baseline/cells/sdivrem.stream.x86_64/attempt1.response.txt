sdivrem:
    mov r8, rdx
    mov rax, rdi
    test rsi, rsi
    je .b_zero
    mov rdx, rdi
    mov rcx, 0x8000000000000000
    cmp rdx, rcx
    jne .do_divide
    cmp rsi, -1
    jne .do_divide
    mov rax, 0x8000000000000000
    xor rdx, rdx
    mov qword ptr [r8], rdx
    ret
.b_zero:
    xor rax, rax
    mov qword ptr [r8], rdi
    ret
.do_divide:
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [r8], rdx
    ret
