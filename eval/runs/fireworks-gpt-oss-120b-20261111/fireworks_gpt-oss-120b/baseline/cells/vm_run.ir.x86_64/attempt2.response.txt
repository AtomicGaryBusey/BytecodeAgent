export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry:
    %pc = imm 0
    %sp = imm 0
    br @loop

@loop:
    %pc_ge_len = csgeul %pc, %len
    jnz %pc_ge_len, @done

    %pc8 = mul %pc, imm 8
    %opc_addr = add %code, %pc8
    %op = loadl %opc_addr

    %is_halt = ceql %op, imm 0
    jnz %is_halt, @done

    %is_push = ceql %op, imm 1
    jnz %is_push, @push

    %is_add = ceql %op, imm 2
    jnz %is_add, @add

    %is_sub = ceql %op, imm 3
    jnz %is_sub, @sub

    %is_mul = ceql %op, imm 4
    jnz %is_mul, @mul

    %is_dup = ceql %op, imm 5
    jnz %is_dup, @dup

    %is_swap = ceql %op, imm 6
    jnz %is_swap, @swap

    %is_drop = ceql %op, imm 7
    jnz %is_drop, @drop

    %is_over = ceql %op, imm 8
    jnz %is_over, @over

    %is_lt = ceql %op, imm 9
    jnz %is_lt, @lt

    %is_jz = ceql %op, imm 10
    jnz %is_jz, @jz

    %is_jmp = ceql %op, imm 11
    jnz %is_jmp, @jmp

    %pc = add %pc, imm 1
    br @loop

@push:
    %pc1 = add %pc, imm 1
    %pc1_8 = mul %pc1, imm 8
    %val_addr = add %code, %pc1_8
    %val = loadl %val_addr
    %sp8 = mul %sp, imm 8
    %stk_addr = add %stack, %sp8
    storel %val, %stk_addr
    %sp = add %sp, imm 1
    %pc = add %pc, imm 2
    br @loop

@add:
    %sp = sub %sp, imm 1
    %sp8b = mul %sp, imm 8
    %b_addr = add %stack, %sp8b
    %b = loadl %b_addr
    %sp = sub %sp, imm 1
    %sp8a = mul %sp, imm 8
    %a_addr = add %stack, %sp8a
    %a = loadl %a_addr
    %c = add %a, %b
    %sp8c = mul %sp, imm 8
    %c_addr = add %stack, %sp8c
    storel %c, %c_addr
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@sub:
    %sp = sub %sp, imm 1
    %sp8b = mul %sp, imm 8
    %b_addr = add %stack, %sp8b
    %b = loadl %b_addr
    %sp = sub %sp, imm 1
    %sp8a = mul %sp, imm 8
    %a_addr = add %stack, %sp8a
    %a = loadl %a_addr
    %c = sub %a, %b
    %sp8c = mul %sp, imm 8
    %c_addr = add %stack, %sp8c
    storel %c, %c_addr
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@mul:
    %sp = sub %sp, imm 1
    %sp8b = mul %sp, imm 8
    %b_addr = add %stack, %sp8b
    %b = loadl %b_addr
    %sp = sub %sp, imm 1
    %sp8a = mul %sp, imm 8
    %a_addr = add %stack, %sp8a
    %a = loadl %a_addr
    %c = mul %a, %b
    %sp8c = mul %sp, imm 8
    %c_addr = add %stack, %sp8c
    storel %c, %c_addr
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@dup:
    %sp1 = sub %sp, imm 1
    %sp1_8 = mul %sp1, imm 8
    %src = add %stack, %sp1_8
    %val = loadl %src
    %sp8 = mul %sp, imm 8
    %dst = add %stack, %sp8
    storel %val, %dst
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@swap:
    %idx_top = sub %sp, imm 1
    %idx2 = sub %sp, imm 2
    %addr_top_tmp = mul %idx_top, imm 8
    %addr_top = add %stack, %addr_top_tmp
    %addr2_tmp = mul %idx2, imm 8
    %addr2 = add %stack, %addr2_tmp
    %val_top = loadl %addr_top
    %val2 = loadl %addr2
    storel %val2, %addr_top
    storel %val_top, %addr2
    %pc = add %pc, imm 1
    br @loop

@drop:
    %sp = sub %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@over:
    %idx2 = sub %sp, imm 2
    %addr2_tmp = mul %idx2, imm 8
    %addr2 = add %stack, %addr2_tmp
    %val = loadl %addr2
    %sp8 = mul %sp, imm 8
    %dst = add %stack, %sp8
    storel %val, %dst
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@lt:
    %sp = sub %sp, imm 1
    %sp8b = mul %sp, imm 8
    %b_addr = add %stack, %sp8b
    %b = loadl %b_addr
    %sp = sub %sp, imm 1
    %sp8a = mul %sp, imm 8
    %a_addr = add %stack, %sp8a
    %a = loadl %a_addr
    %c = csltl %a, %b
    %sp8c = mul %sp, imm 8
    %c_addr = add %stack, %sp8c
    storel %c, %c_addr
    %sp = add %sp, imm 1
    %pc = add %pc, imm 1
    br @loop

@jz:
    %sp = sub %sp, imm 1
    %sp8c = mul %sp, imm 8
    %cond_addr = add %stack, %sp8c
    %cond = loadl %cond_addr
    %pc1 = add %pc, imm 1
    %pc1_8 = mul %pc1, imm 8
    %t_addr = add %code, %pc1_8
    %t = loadl %t_addr
    %is_zero = ceql %cond, imm 0
    jnz %is_zero, @jz_set
    %pc = add %pc, imm 2
    br @loop

@jz_set:
    %pc = %t
    br @loop

@jmp:
    %pc1 = add %pc, imm 1
    %pc1_8 = mul %pc1, imm 8
    %t_addr = add %code, %pc1_8
    %t = loadl %t_addr
    %pc = %t
    br @loop

@done:
    %i = %sp
    br @zero_fill

@zero_fill:
    %condz = csltl %i, %cap
    jnz %condz, @zero_body
    br @finalize

@zero_body:
    %addrz_tmp = mul %i, imm 8
    %addrz = add %stack, %addrz_tmp
    storel imm 0, %addrz
    %i = add %i, imm 1
    br @zero_fill

@finalize:
    %has = csltl imm 0, %sp
    jnz %has, @ret_nonempty
    br @ret_zero

@ret_zero:
    %val0 = imm 0
    br @ret_merge

@ret_nonempty:
    %sp1 = sub %sp, imm 1
    %addr_top_tmp = mul %sp1, imm 8
    %addr_top = add %stack, %addr_top_tmp
    %val1 = loadl %addr_top
    br @ret_merge

@ret_merge:
    %ret = phi l (%val0, @ret_zero), (%val1, @ret_nonempty)
    ret %ret
}
