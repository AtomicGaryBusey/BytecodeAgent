export function l $map_sum(l %a, l %n, l %f) {
@entry:
    %c0 =l const 0
    %c1 =l const 1
    %sum =l copy %c0
    %i =l copy %c0
    jmp @cond
@cond:
    %c =w csltl %i, %n
    jnz %c, @body
    jmp @exit
@body:
    %ofs =l shll %i, 3
    %p =l add %a, %ofs
    %val =l loadl %p
    %r =l call %f(l %val)
    %sum =l add %sum, %r
    %i =l add %i, %c1
    jmp @cond
@exit:
    ret %sum
}
