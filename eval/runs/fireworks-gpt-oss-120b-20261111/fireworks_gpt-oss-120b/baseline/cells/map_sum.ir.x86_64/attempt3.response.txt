export function l $map_sum(l %a, l %n, l %f) {
@entry:
    %sum =l copy const 0;
    %i =l copy const 0;
    %c1 =l copy const 1;
    jmp @cond;
@cond:
    %c =w csltl %i, %n;
    jnz %c, @body;
    jmp @exit;
@body:
    %ofs =l mul %i, const 8;
    %p =l add %a, %ofs;
    %val =l loadl %p;
    %r =l call %f(l %val);
    %sum =l add %sum, %r;
    %i =l add %i, %c1;
    jmp @cond;
@exit:
    ret %sum;
}
