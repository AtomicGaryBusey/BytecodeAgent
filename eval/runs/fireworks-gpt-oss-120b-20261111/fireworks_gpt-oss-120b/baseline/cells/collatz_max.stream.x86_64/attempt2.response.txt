collatz_max:
    push rbx
    xor ebx, ebx
    xor ecx, ecx
.Lk_loop:
    inc ecx
    cmp rcx, rdi
    jg .Ldone_k
    mov rdx, rcx
    xor r9d, r9d
.Linner:
    cmp rdx, 1
    je .Linner_done
    test rdx, 1
    jz .Leven
    imul rdx, rdx, 3
    inc rdx
    jmp .Linc_step
.Leven:
    shr rdx, 1
.Linc_step:
    inc r9d
    jmp .Linner
.Linner_done:
    cmp r9d, ebx
    jle .Lk_loop_continue
    mov ebx, r9d
.Lk_loop_continue:
    jmp .Lk_loop
.Ldone_k:
    mov eax, ebx
    pop rbx
    ret
