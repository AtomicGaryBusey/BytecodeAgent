collatz_max:
    xor ebx, ebx                     # max = 0
    xor ecx, ecx                     # k = 0
.Lk_loop:
    inc ecx                          # k++
    cmp ecx, edi
    ja .Ldone_k                      # if k > n, exit
    mov rdx, rcx                     # x = k
    xor r9d, r9d                     # steps = 0
.Linner:
    cmp rdx, 1
    je .Linner_done
    test rdx, 1
    jz .Leven
    imul rdx, rdx, 3                 # x = 3*x
    add rdx, 1                       # x = 3*x + 1
    jmp .Linc_step
.Leven:
    shr rdx, 1                       # x = x/2
.Linc_step:
    inc r9d
    jmp .Linner
.Linner_done:
    cmp r9d, ebx
    jle .Lk_loop_continue
    mov ebx, r9d                     # update max
.Lk_loop_continue:
    jmp .Lk_loop
.Ldone_k:
    mov eax, ebx
    ret
