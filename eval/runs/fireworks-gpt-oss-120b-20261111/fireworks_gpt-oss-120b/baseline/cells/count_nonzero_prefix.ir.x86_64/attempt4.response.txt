export function $count_nonzero_prefix(l, l) {
@start
    %i = phi l 0, @start, %i_next, @loop
    %c = csltl %i, %1
    jnz %c, @check, @ret

@check
    %ofs = mul %i, 8
    %addr = addl %0, %ofs
    %val = loadl %addr
    %zero = ceql %val, 0
    jnz %zero, @ret, @loop

@loop
    %i_next = addl %i, 1
    jmp @start

@ret
    ret %i
}
