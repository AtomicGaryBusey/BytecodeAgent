export function $count_nonzero_prefix(l a, l n) {
@start
    %i = phi l 0, @start, %i_next, @loop
    %c1 = csltl %i, n
    jnz %c1, @check, @ret

@check
    %addr = addl a, mul %i, 8
    %val = loadl %addr
    %zero = ceql %val, 0
    jnz %zero, @ret, @loop

@loop
    %i_next = addl %i, 1
    jmp @start

@ret
    ret %i
}
