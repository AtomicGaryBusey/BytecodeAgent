export function l $count_nonzero_prefix(l a, l n) {
@entry
    jmp @loop

@loop
    %i = phi l 0, @entry, %i_next, @inc
    %c1 = csltl %i, n
    jnz %c1, @check, @ret

@check
    %addr = addl a, mul %i, 8
    %val = loadl %addr
    %zero = ceql %val, 0
    jnz %zero, @ret, @inc

@inc
    %i_next = addl %i, 1
    jmp @loop

@ret
    ret %i
}
