export function l $qsort_pairs(l %a, l %n) {
@entry:
    %zero =l const 0
    %one =l const 1
    %two =l const 2
    %eight =l const 8
    %sixteen =l const 16

    %lt2 =w csltl %n, %two
    jnz %lt2, @ret0(%zero)

    ; pivot = middle element
    %pivot_idx =l udivl %n, %two
    %pivot_off =l mul %pivot_idx, %sixteen
    %pivot_key_addr =l add %a, %pivot_off
    %pivot_key =l loadl %pivot_key_addr
    %pivot_val_addr =l add %pivot_key_addr, %eight
    %pivot_val =l loadl %pivot_val_addr

    %i0 =l const 0
    %j0 =l subl %n, %one
    jmp @loop(%i0 %j0)

@loop(%i:l %j:l):
    %i_gt_j =w csltl %j, %i
    jnz %i_gt_j, @part_done(%i %j)

    ; load record i
    %off_i =l mul %i, %sixteen
    %addr_i_key =l add %a, %off_i
    %key_i =l loadl %addr_i_key
    %addr_i_val =l add %addr_i_key, %eight
    %val_i =l loadl %addr_i_val

    ; i condition (record[i] < pivot)
    %c_key_lt =w csltl %key_i, %pivot_key
    %c_key_eq =w ceql %key_i, %pivot_key
    %c_val_lt =w csltl %val_i, %pivot_val
    %c_i_lt_tmp =w andw %c_key_eq %c_val_lt
    %c_i_lt =w orw %c_key_lt %c_i_lt_tmp

    %i1 =l add %i, %one
    %i_new =l sel %c_i_lt %i1 %i

    ; load record j
    %off_j =l mul %j, %sixteen
    %addr_j_key =l add %a, %off_j
    %key_j =l loadl %addr_j_key
    %addr_j_val =l add %addr_j_key, %eight
    %val_j =l loadl %addr_j_val

    ; j condition (record[j] > pivot)
    %c_key_gt =w csltl %pivot_key, %key_j
    %c_key_eq2 =w ceql %pivot_key, %key_j
    %c_val_gt =w csltl %pivot_val, %val_j
    %c_j_gt_tmp =w andw %c_key_eq2 %c_val_gt
    %c_j_gt =w orw %c_key_gt %c_j_gt_tmp

    %j1 =l subl %j, %one
    %j_new =l sel %c_j_gt %j1 %j

    ; check if i_new > j_new
    %i_gt_j2 =w csltl %j_new, %i_new
    jnz %i_gt_j2, @part_done(%i_new %j_new)

    ; swap i_new and j_new
    jmp @swap(%i_new %j_new)

@swap(%i_sw:l %j_sw:l):
    ; load i record
    %off_i =l mul %i_sw, %sixteen
    %addr_i_key =l add %a, %off_i
    %key_i =l loadl %addr_i_key
    %addr_i_val =l add %addr_i_key, %eight
    %val_i =l loadl %addr_i_val

    ; load j record
    %off_j =l mul %j_sw, %sixteen
    %addr_j_key =l add %a, %off_j
    %key_j =l loadl %addr_j_key
    %addr_j_val =l add %addr_j_key, %eight
    %val_j =l loadl %addr_j_val

    ; store swapped
    storel %addr_i_key %key_j
    storel %addr_i_val %val_j
    storel %addr_j_key %key_i
    storel %addr_j_val %val_i

    %i_next =l add %i_sw, %one
    %j_next =l subl %j_sw, %one
    jmp @loop(%i_next %j_next)

@part_done(%i_fin:l %j_fin:l):
    %left_sz =l add %j_fin, %one
    %right_sz =l subl %n, %i_fin

    ; recurse left if left_sz > 1
    %left_gt1 =w csltl %one, %left_sz
    jnz %left_gt1, @recurse_left

    jmp @maybe_right

@recurse_left:
    %_ =l call $qsort_pairs(l %a, l %left_sz)
    jmp @maybe_right

@maybe_right:
    ; recurse right if right_sz > 1
    %right_gt1 =w csltl %one, %right_sz
    jnz %right_gt1, @recurse_right
    jmp @ret0

@recurse_right:
    %off_i =l mul %i_fin, %sixteen
    %ptr_r =l add %a, %off_i
    %_ =l call $qsort_pairs(l %ptr_r, l %right_sz)
    jmp @ret0

@ret0(%ret:l):
    ret %ret
}
