export function l $qsort_pairs(l %a, l %n) {
@entry:
    %zero =l const 0
    %one =l const 1
    %two =l const 2
    %sixteen =l const 16

    ; if n < 2 return 0
    %lt2 =w csltl %n, %two
    jnz %lt2, @ret0

    ; pivot index = n / 2
    %pivot_idx =l udivl %n, %two
    %pivot_off =l mul %pivot_idx, %sixteen
    %pivot_key_addr =l add %a, %pivot_off
    %pivot_key =l loadl %pivot_key_addr
    %pivot_val_addr =l add %pivot_key_addr, %one
    %pivot_val =l loadl %pivot_val_addr

    ; i = 0, j = n-1
    %i =l const 0
    %j =l subl %n, %one

@partition_loop:
    ; if i > j break
    %i_gt_j =w csltl %j, %i
    jnz %i_gt_j, @partition_done

    ; advance i while rec[i] < pivot
@advance_i:
    ; if i > j break
    %i_gt_j2 =w csltl %j, %i
    jnz %i_gt_j2, @after_advance_i

    %off_i =l mul %i, %sixteen
    %addr_i_key =l add %a, %off_i
    %key_i =l loadl %addr_i_key
    %addr_i_val =l add %addr_i_key, %one
    %val_i =l loadl %addr_i_val

    ; key_i < pivot_key ?
    %c_key_lt =w csltl %key_i, %pivot_key
    %c_key_eq =w ceql %key_i, %pivot_key
    %c_val_lt =w csltl %val_i, %pivot_val
    %c_key_eq_and_val_lt =w andw %c_key_eq, %c_val_lt
    %c_i_lt =w orw %c_key_lt, %c_key_eq_and_val_lt
    jnz %c_i_lt, @inc_i
    jmp @after_advance_i

@inc_i:
    %i =l add %i, %one
    jmp @advance_i

@after_advance_i:
    ; retreat j while rec[j] > pivot
@retreat_j:
    ; if i > j break
    %i_gt_j3 =w csltl %j, %i
    jnz %i_gt_j3, @after_retreat_j

    %off_j =l mul %j, %sixteen
    %addr_j_key =l add %a, %off_j
    %key_j =l loadl %addr_j_key
    %addr_j_val =l add %addr_j_key, %one
    %val_j =l loadl %addr_j_val

    ; key_j > pivot_key ?
    %c_key_gt =w csltl %pivot_key, %key_j
    %c_key_eq2 =w ceql %pivot_key, %key_j
    %c_val_gt =w csltl %pivot_val, %val_j
    %c_key_eq_and_val_gt =w andw %c_key_eq2, %c_val_gt
    %c_j_gt =w orw %c_key_gt, %c_key_eq_and_val_gt
    jnz %c_j_gt, @dec_j
    jmp @after_retreat_j

@dec_j:
    %j =l subl %j, %one
    jmp @retreat_j

@after_retreat_j:
    ; if i <= j swap
    %i_le_j =w csltl %j, %i
    notw %i_le_j, %i_le_j
    jnz %i_le_j, @swap
    jmp @partition_loop

@swap:
    ; load i record
    %off_i2 =l mul %i, %sixteen
    %addr_i_key2 =l add %a, %off_i2
    %key_i2 =l loadl %addr_i_key2
    %addr_i_val2 =l add %addr_i_key2, %one
    %val_i2 =l loadl %addr_i_val2
    ; load j record
    %off_j2 =l mul %j, %sixteen
    %addr_j_key2 =l add %a, %off_j2
    %key_j2 =l loadl %addr_j_key2
    %addr_j_val2 =l add %addr_j_key2, %one
    %val_j2 =l loadl %addr_j_val2
    ; swap
    storel %addr_i_key2, %key_j2
    storel %addr_i_val2, %val_j2
    storel %addr_j_key2, %key_i2
    storel %addr_j_val2, %val_i2
    ; i++, j--
    %i =l add %i, %one
    %j =l subl %j, %one
    jmp @partition_loop

@partition_done:
    ; left size = j+1
    %left_sz =l add %j, %one
    ; right size = n - i
    %right_sz =l subl %n, %i

    ; recurse left if left_sz > 1
    %gt1_left =w csltl %one, %left_sz
    jnz %gt1_left, @recurse_left
    jmp @maybe_right

@recurse_left:
    %r_left =l call $qsort_pairs(l %a, l %left_sz)
    jmp @maybe_right

@maybe_right:
    ; recurse right if right_sz > 1
    %gt1_right =w csltl %one, %right_sz
    jnz %gt1_right, @recurse_right
    jmp @ret0

@recurse_right:
    %off_i3 =l mul %i, %sixteen
    %ptr_r =l add %a, %off_i3
    %r_right =l call $qsort_pairs(l %ptr_r, l %right_sz)
    jmp @ret0

@ret0:
    ret %zero
}
