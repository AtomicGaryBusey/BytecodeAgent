packbits_decode:
    mov     x4, x0                     // src_ptr
    add     x5, x0, x1                 // src_end = src + n
    mov     x6, x2                     // out_ptr
    add     x7, x2, x3                 // out_end = out + cap
.Lloop:
    cmp     x4, x5
    bge     .Lend_success
    ldrb    w9, [x4], #1               // load header, post-increment src_ptr
    sxtb    w9, w9                     // sign‑extend header
    cmp     w9, #-128
    beq     .Lloop                     // h == -128 : no‑op
    cmp     w9, #0
    bge     .Lliteral                  // h >= 0 : literal packet
    // ----- repeat packet (h < 0, h != -128) -----
    cmp     x4, x5
    bge     .Lerror_src                // need repeat byte
    ldrb    w11, [x4], #1              // repeat value, advance src_ptr
    neg     w10, w9                    // -h
    add     w10, w10, #1               // count = 1 - h
    sub     x12, x7, x6                // remaining capacity
    cmp     x12, x10
    bhs     .Lrepeat_full
    // not enough capacity
    cmp     x12, #0
    beq     .Lreturn_minus1
    mov     x13, x12
.Lrepeat_copy_partial:
    strb    w11, [x6], #1
    subs    x13, x13, #1
    bne     .Lrepeat_copy_partial
    b       .Lreturn_minus1
.Lrepeat_full:
    mov     x13, x10
.Lrepeat_copy:
    strb    w11, [x6], #1
    subs    x13, x13, #1
    bne     .Lrepeat_copy
    b       .Lloop
.Lliteral:
    // length = h + 1
    add     w10, w9, #1                // length in w10
    add     x11, x4, x10               // src_ptr + length
    cmp     x11, x5
    bhi     .Lerror_src                // not enough source bytes
    sub     x12, x7, x6                // remaining capacity
    cmp     x12, x10
    bhs     .Lliteral_full
    // not enough capacity
    cmp     x12, #0
    beq     .Lreturn_minus1
    mov     x13, x12
.Lliteral_copy_partial:
    ldrb    w14, [x4], #1
    strb    w14, [x6], #1
    subs    x13, x13, #1
    bne     .Lliteral_copy_partial
    b       .Lreturn_minus1
.Lliteral_full:
    mov     x13, x10
.Lliteral_copy:
    ldrb    w14, [x4], #1
    strb    w14, [x6], #1
    subs    x13, x13, #1
    bne     .Lliteral_copy
    b       .Lloop
.Lerror_src:
    mov     x0, #-2
    ret
.Lreturn_minus1:
    mov     x0, #-1
    ret
.Lend_success:
    sub     x0, x6, x2                 // bytes_written = out_ptr - out_start
    ret
