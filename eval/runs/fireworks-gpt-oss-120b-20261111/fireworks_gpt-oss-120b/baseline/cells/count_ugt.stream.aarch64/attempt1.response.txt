count_ugt:
    mov     x3, #0                 // i = 0
    mov     x4, #0                 // count = 0
    cbz     x1, .Ldone             // if n == 0, finish
.Lloop:
    cmp     x3, x1
    b.ge    .Ldone
    ldr     x5, [x0, x3, lsl #3]   // load a[i]
    cmp     x5, x2
    b.le    .Lskipinc
    add     x4, x4, #1             // count++
.Lskipinc:
    add     x3, x3, #1             // i++
    b       .Lloop
.Ldone:
    mov     x0, x4                 // return count
    ret
