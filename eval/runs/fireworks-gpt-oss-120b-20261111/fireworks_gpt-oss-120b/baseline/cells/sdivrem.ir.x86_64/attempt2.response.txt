export function l $sdivrem(l %a, l %b, l %rem) {
@entry:
  %zero = constl 0
  %minus_one = constl -1
  %int64_min = constl -9223372036854775808
  %b_is_zero = ceql %b, %zero
  jnz %b_is_zero, @case_bzero
  %a_is_min = ceql %a, %int64_min
  %b_is_minus_one = ceql %b, %minus_one
  %special = andl %a_is_min, %b_is_minus_one
  jnz %special, @case_overflow
  %q = divl %a, %b
  %r = reml %a, %b
  storel %r, %rem
  ret %q
@case_bzero:
  storel %a, %rem
  ret %zero
@case_overflow:
  %zero_rem = constl 0
  storel %zero_rem, %rem
  ret %int64_min
}
