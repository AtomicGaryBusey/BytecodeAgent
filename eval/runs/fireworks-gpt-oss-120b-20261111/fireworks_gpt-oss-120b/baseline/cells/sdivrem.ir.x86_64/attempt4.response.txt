export function l $sdivrem(l %a, l %b, l %rem) {
@entry:
  %zero = constl 0
  %minus_one = constl -1
  %b_is_zero = ceql %b, %zero
  jnz %b_is_zero, @case_bzero
  %neg_a = subl %zero, %a
  %a_is_min = ceql %a, %neg_a
  %b_is_minus_one = ceql %b, %minus_one
  %special = andw %a_is_min, %b_is_minus_one
  jnz %special, @case_overflow
  %q = divl %a, %b
  %r = reml %a, %b
  storel %r, %rem
  ret %q
@case_bzero:
  storel %a, %rem
  ret %zero
@case_overflow:
  %one = constl 1
  %bits63 = constl 63
  %int64_min = shl %one, %bits63
  storel %zero, %rem
  ret %int64_min
}
