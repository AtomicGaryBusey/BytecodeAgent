mat_pow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    mov x19, x3               // out
    mov x20, x4               // tmp
    mov x21, x0               // a
    mov x22, x1               // n
    mov x23, x2               // k
    mul x24, x22, x22         // count = n*n
    add x25, x20, x24, lsl #3 // tmp2 = tmp + count*8
    // copy a -> base (tmp first half)
    mov x27, #0
.Lcopy_a:
    cmp x27, x24
    b.eq .Lcopy_a_done
    lsl x28, x27, #3
    ldr x0, [x21, x28]
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lcopy_a
.Lcopy_a_done:
    // zero out out matrix
    mov x27, #0
.Lzero_out:
    cmp x27, x24
    b.eq .Lzero_out_done
    lsl x28, x27, #3
    mov x0, #0
    str x0, [x19, x28]
    add x27, x27, #1
    b .Lzero_out
.Lzero_out_done:
    // set identity in out
    mov x27, #0
.Ldiag:
    cmp x27, x22
    b.eq .Ldiag_done
    mul x28, x27, x22
    add x28, x28, x27
    lsl x28, x28, #3
    mov x0, #1
    str x0, [x19, x28]
    add x27, x27, #1
    b .Ldiag
.Ldiag_done:
    cmp x23, #0
    beq .Lzero_tmp
.Lexp_loop:
    and x0, x23, #1
    cbz x0, .Lskip_mul
    adr x26, .Lafter_mul
    mov x5, x19
    mov x6, x20
    mov x7, x25
    b .Mmult
.Lafter_mul:
    // copy tmp2 -> out
    mov x27, #0
.Lcopy_res:
    cmp x27, x24
    b.eq .Lcopy_res_done
    lsl x28, x27, #3
    ldr x0, [x25, x28]
    str x0, [x19, x28]
    add x27, x27, #1
    b .Lcopy_res
.Lcopy_res_done:
.Lskip_mul:
    // square base: base *= base -> tmp2
    adr x26, .Lafter_sq
    mov x5, x20
    mov x6, x20
    mov x7, x25
    b .Mmult
.Lafter_sq:
    // copy tmp2 -> base
    mov x27, #0
.Lcopy_base:
    cmp x27, x24
    b.eq .Lcopy_base_done
    lsl x28, x27, #3
    ldr x0, [x25, x28]
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lcopy_base
.Lcopy_base_done:
    lsr x23, x23, #1
    cmp x23, #0
    b.ne .Lexp_loop
.Lzero_tmp:
    // zero first half of tmp (base)
    mov x27, #0
.Lzero_tmp1:
    cmp x27, x24
    b.eq .Lzero_tmp2
    lsl x28, x27, #3
    mov x0, #0
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lzero_tmp1
.Lzero_tmp2:
    // zero second half of tmp (tmp2)
    mov x27, #0
.Lzero_tmp2_loop:
    cmp x27, x24
    b.eq .Lend_zero
    lsl x28, x27, #3
    mov x0, #0
    str x0, [x25, x28]
    add x27, x27, #1
    b .Lzero_tmp2_loop
.Lend_zero:
    mov x0, #0
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
.Mmult:
    mov x8, #0                     // i
.Mmult_i:
    cmp x8, x22
    b.eq .Mmult_done_i
    mov x9, #0                     // j
.Mmult_j:
    cmp x9, x22
    b.eq .Mmult_next_i
    mul x10, x8, x22
    add x10, x10, x9
    lsl x10, x10, #3                // dest offset
    mov x11, #0                     // sum
    mov x12, #0                     // t
.Mmult_t:
    cmp x12, x22
    b.eq .Mmult_store
    mul x13, x8, x22
    add x13, x13, x12
    lsl x13, x13, #3
    ldr x14, [x5, x13]              // a[i][t]
    mul x15, x12, x22
    add x15, x15, x9
    lsl x15, x15, #3
    ldr x16, [x6, x15]              // b[t][j]
    mul x17, x14, x16
    adds x11, x11, x17              // sum += product (wrap)
    add x12, x12, #1
    b .Mmult_t
.Mmult_store:
    str x11, [x7, x10]
    add x9, x9, #1
    b .Mmult_j
.Mmult_next_i:
    add x8, x8, #1
    b .Mmult_i
.Mmult_done_i:
    br x26
