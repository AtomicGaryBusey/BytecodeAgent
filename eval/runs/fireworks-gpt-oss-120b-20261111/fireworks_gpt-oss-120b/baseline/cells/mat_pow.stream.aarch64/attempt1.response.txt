mat_pow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    mov x19, x3          // out
    mov x20, x4          // tmp (base)
    mov x21, x0          // a
    mov x22, x1          // n
    mov x23, x2          // k
    // count = n * n
    mul x25, x22, x22    // x25 = element count
    // copy a -> base (tmp)
    mov x26, #0
.Lcopy_a:
    cmp x26, x25
    b.eq .Lcopy_a_done
    lsl x27, x26, #3
    ldr x0, [x21, x27]
    str x0, [x20, x27]
    add x26, x26, #1
    b .Lcopy_a
.Lcopy_a_done:
    // zero out result matrix
    mov x26, #0
.Lzero_out:
    cmp x26, x25
    b.eq .Lzero_out_done
    lsl x27, x26, #3
    mov x0, #0
    str x0, [x19, x27]
    add x26, x26, #1
    b .Lzero_out
.Lzero_out_done:
    // set identity
    mov x28, #0
.Ldiag:
    cmp x28, x22
    b.eq .Ldiag_done
    mul x27, x28, x22
    add x27, x27, x28
    lsl x27, x27, #3
    mov x0, #1
    str x0, [x19, x27]
    add x28, x28, #1
    b .Ldiag
.Ldiag_done:
    // if k == 0 skip exponentiation
    cmp x23, #0
    b.eq .Lcleanup
.Lexp_loop:
    // if (k & 1) multiply result *= base
    and x0, x23, #1
    cbz x0, .Lskip_mul_res
    // tmp2 = tmp + n*n*8
    add x24, x20, x25, lsl #3
    // multiply result (out) * base (tmp) -> tmp2
    mov x5, x19
    mov x6, x20
    mov x7, x24
    b .Mmult
.Lskip_mul_res:
    // square base: base = base * base
    add x24, x20, x25, lsl #3
    mov x5, x20
    mov x6, x20
    mov x7, x24
    b .Mmult
    // after multiplication, copy tmp2 to appropriate destination
.Mcopy_res:
    mov x5, x24
    mov x6, x19
    mov x27, #0
.Lcopy_res_loop:
    cmp x27, x25
    b.eq .Lafter_copy_res
    lsl x28, x27, #3
    ldr x0, [x5, x28]
    str x0, [x6, x28]
    add x27, x27, #1
    b .Lcopy_res_loop
.Lafter_copy_res:
    // copy tmp2 to base
    mov x5, x24
    mov x6, x20
    mov x27, #0
.Lcopy_base_loop:
    cmp x27, x25
    b.eq .Lafter_copy_base
    lsl x28, x27, #3
    ldr x0, [x5, x28]
    str x0, [x6, x28]
    add x27, x27, #1
    b .Lcopy_base_loop
.Lafter_copy_base:
    // shift k >>= 1
    lsr x23, x23, #1
    cmp x23, #0
    b.ne .Lexp_loop
    b .Lcleanup
.Mmult:
    // x5 = src1, x6 = src2, x7 = dest, x22 = n
    mov x8, #0                // i
.Lm_i:
    cmp x8, x22
    b.eq .Lm_done_i
    mov x9, #0                // j
.Lm_j:
    cmp x9, x22
    b.eq .Lm_next_i
    mov x11, #0               // sum
    mov x10, #0               // t
.Lm_t:
    cmp x10, x22
    b.eq .Lm_store
    mul x12, x8, x22
    add x12, x12, x10
    lsl x12, x12, #3
    ldr x13, [x5, x12]
    mul x14, x10, x22
    add x14, x14, x9
    lsl x14, x14, #3
    ldr x15, [x6, x14]
    mul x13, x13, x15
    adds x11, x11, x13
    add x10, x10, #1
    b .Lm_t
.Lm_store:
    mul x12, x8, x22
    add x12, x12, x9
    lsl x12, x12, #3
    str x11, [x7, x12]
    add x9, x9, #1
    b .Lm_j
.Lm_next_i:
    add x8, x8, #1
    b .Lm_i
.Lm_done_i:
    // Determine which copy to perform next
    // If we came from result multiply, copy to out; else copy to base
    // Use x0 (was set before branching) to decide: if (k & 1) we branched here, x0 was non-zero
    // We'll inspect x0: zero means base squaring, non-zero means result multiply
    cbz x0, .Mcopy_base
    b .Mcopy_res
.Mcopy_base:
    // copy tmp2 to base (already defined)
    b .Lcopy_base_loop
.Mcopy_res:
