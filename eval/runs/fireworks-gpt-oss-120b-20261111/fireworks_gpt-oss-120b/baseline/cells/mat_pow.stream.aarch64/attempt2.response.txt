mat_pow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    // x19 = out, x20 = tmp, x21 = a, x22 = n, x23 = k
    mov x19, x3
    mov x20, x4
    mov x21, x0
    mov x22, x1
    mov x23, x2
    // count = n*n
    mul x24, x22, x22          // x24 = element count
    // copy a -> base (tmp first half)
    mov x27, #0
.Lcopy_a_loop:
    cmp x27, x24
    b.eq .Lcopy_a_done
    lsl x28, x27, #3
    ldr x0, [x21, x28]
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lcopy_a_loop
.Lcopy_a_done:
    // zero out result matrix (out)
    mov x27, #0
.Lzero_out_loop:
    cmp x27, x24
    b.eq .Lzero_out_done
    lsl x28, x27, #3
    mov x0, #0
    str x0, [x19, x28]
    add x27, x27, #1
    b .Lzero_out_loop
.Lzero_out_done:
    // set identity in out
    mov x27, #0
.Lset_diag_loop:
    cmp x27, x22
    b.eq .Lset_diag_done
    mul x28, x27, x22
    add x28, x28, x27
    lsl x28, x28, #3
    mov x0, #1
    str x0, [x19, x28]
    add x27, x27, #1
    b .Lset_diag_loop
.Lset_diag_done:
    // if k == 0 skip exponentiation
    cmp x23, #0
    beq .Lzero_tmp
    // tmp2 = tmp + count*8
    add x25, x20, x24, lsl #3
.Lexp_loop:
    // if (k & 1) multiply out *= base -> tmp2
    and x0, x23, #1
    cbz x0, .Lskip_mul
    // set return address after multiplication
    adr x26, .Lafter_mul_res
    mov x5, x19          // src1 = out
    mov x6, x20          // src2 = base
    mov x7, x25          // dest = tmp2
    b .Mmult
.Lafter_mul_res:
    // copy tmp2 -> out
    mov x27, #0
.Lcopy_res_loop:
    cmp x27, x24
    b.eq .Lafter_copy_res
    lsl x28, x27, #3
    ldr x0, [x25, x28]
    str x0, [x19, x28]
    add x27, x27, #1
    b .Lcopy_res_loop
.Lafter_copy_res:
.Lskip_mul:
    // square base: base *= base -> tmp2
    adr x26, .Lafter_square
    mov x5, x20          // src1 = base
    mov x6, x20          // src2 = base
    mov x7, x25          // dest = tmp2
    b .Mmult
.Lafter_square:
    // copy tmp2 -> base
    mov x27, #0
.Lcopy_base_loop:
    cmp x27, x24
    b.eq .Lafter_copy_base
    lsl x28, x27, #3
    ldr x0, [x25, x28]
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lcopy_base_loop
.Lafter_copy_base:
    // k >>= 1
    lsr x23, x23, #1
    cmp x23, #0
    b.ne .Lexp_loop
    // exponentiation done, zero tmp
.Lzero_tmp:
    // zero first half of tmp (base)
    mov x27, #0
.Lzero_tmp1_loop:
    cmp x27, x24
    b.eq .Lzero_tmp2_loop
    lsl x28, x27, #3
    mov x0, #0
    str x0, [x20, x28]
    add x27, x27, #1
    b .Lzero_tmp1_loop
.Lzero_tmp2_loop:
    // zero second half of tmp (tmp2)
    mov x27, #0
.Lzero_tmp2_loop:
