export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %zero = copy 0
    %i0 = copy %zero
    %outpos0 = copy %zero
    jmp %outer_cond

@outer_cond:
    %i = phi %i0, %i_next
    %outpos = phi %outpos0, %outpos_next
    %c_i_n = csltl %i, %n
    jnz %c_i_n, %outer_body
    ret %outpos

@outer_body:
    %src_i = add %src, %i
    %base = w loadub %src_i
    %run0 = copy 1
    jmp %run_check

@run_check:
    %run = phi %run0, %run_inc
    %i_plus_run = add %i, %run
    %c1 = csltl %i_plus_run, %n
    %c2 = csltl %run, 128
    %c = and %c1, %c2
    jnz %c, %run_eq
    jmp %run_done

@run_eq:
    %cur_ptr = add %src, %i_plus_run
    %cur = w loadub %cur_ptr
    %eq = ceql %cur, %base
    jnz %eq, %run_inc
    jmp %run_done

@run_inc:
    %run_inc = add %run, 1
    jmp %run_check

@run_done:
    %c_lt3 = csltl %run, 3
    jnz %c_lt3, %literal_start
    %need = add %outpos, 2
    %c_over = csltl %cap, %need
    jnz %c_over, %fail
    %header = sub 257, %run
    %out_ptr = add %out, %outpos
    storeb %header, %out_ptr
    %out_ptr1 = add %out_ptr, 1
    storeb %base, %out_ptr1
    %outpos_next = add %outpos, 2
    %i_next = add %i, %run
    jmp %outer_cond

@literal_start:
    %lit_start = copy %i
    %lit_len0 = copy %zero
    jmp %literal_check

@literal_check:
    %lit_len = phi %lit_len0, %lit_len_next
    %i_cur = phi %i, %i_next2
    %c1 = csltl %i_cur, %n
    %c2 = csltl %lit_len, 128
    %c = and %c1, %c2
    jnz %c, %literal_run_check
    jmp %literal_done

@literal_run_check:
    %remain = sub %n, %i_cur
    %c3 = csltl %remain, 3
    jnz %c3, %literal_add_byte
    %p0 = add %src, %i_cur
    %b0 = w loadub %p0
    %p1 = add %p0, 1
    %b1 = w loadub %p1
    %p2 = add %p0, 2
    %b2 = w loadub %p2
    %eq01 = ceql %b0, %b1
    %eq12 = ceql %b1, %b2
    %run3 = and %eq01, %eq12
    jnz %run3, %literal_done
    jmp %literal_add_byte

@literal_add_byte:
    %i_next2 = add %i_cur, 1
    %lit_len_next = add %lit_len, 1
    jmp %literal_check

@literal_done:
    %need2 = add %outpos, 1
    %need2 = add %need2, %lit_len
    %c_over2 = csltl %cap, %need2
    jnz %c_over2, %fail
    %header_lit = sub %lit_len, 1
    %out_ptr2 = add %out, %outpos
    storeb %header_lit, %out_ptr2
    %copy_i0 = copy %zero
    jmp %copy_check

@copy_check:
    %copy_i = phi %copy_i0, %copy_i_next
    %c_copy = csltl %copy_i, %lit_len
    jnz %c_copy, %copy_body
    jmp %copy_done

@copy_body:
    %src_addr = add %src, %lit_start
    %src_addr = add %src_addr, %copy_i
    %byte = w loadub %src_addr
    %dst_addr = add %out, %outpos
    %dst_addr = add %dst_addr, 1
    %dst_addr = add %dst_addr, %copy_i
    storeb %byte, %dst_addr
    %copy_i_next = add %copy_i, 1
    jmp %copy_check

@copy_done:
    %outpos_next = add %outpos, 1
    %outpos_next = add %outpos_next, %lit_len
    %i_next = %i_cur
    jmp %outer_cond

@fail:
    ret -1
}
