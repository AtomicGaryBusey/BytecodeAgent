export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
  %i0 = copy 0
  %outpos0 = copy 0
  jmp @loop

@loop:
  %i = phi %i0, %i_next
  %outpos = phi %outpos0, %outpos_next
  %cond_end = csltl %i, %n
  jnz %cond_end, @run_start
  ret %outpos

@run_start:
  %run0 = copy 1
  %base0 = w loadub (add %src, %i)
  jmp @run_check

@run_check:
  %run = phi %run0, %run_inc
  %next = add %i, %run
  %c1 = csltl %next, %n
  %c2 = csltl %run, 128
  %c = and %c1, %c2
  jnz %c, @run_cmp
  jmp @run_done

@run_cmp:
  %ptr = add %src, %next
  %cur = w loadub %ptr
  %eq = ceql %cur, %base0
  jnz %eq, @run_inc
  jmp @run_done

@run_inc:
  %run_inc = add %run, 1
  jmp @run_check

@run_done:
  %run_ge3 = csltl 3, %run    // run >= 3 ?
  jnz %run_ge3, @emit_repeat
  jmp @literal_start

@emit_repeat:
  %need = add %outpos, 2
  %cap_lt_need = csltl %cap, %need
  jnz %cap_lt_need, @fail
  %header = sub 257, %run
  %outptr = add %out, %outpos
  storeb %header, %outptr
  %outptr1 = add %outptr, 1
  storeb %base0, %outptr1
  %i_next = add %i, %run
  %outpos_next = add %outpos, 2
  jmp @loop

@literal_start:
  %lit_start = copy %i
  %lit_len0 = copy 0
  jmp @literal_check

@literal_check:
  %lit_len = phi %lit_len0, %lit_len_inc
  %j = add %lit_start, %lit_len
  %c1 = csltl %j, %n
  %c2 = csltl %lit_len, 128
  %c = and %c1, %c2
  jnz %c, @literal_run_check
  jmp @literal_done

@literal_run_check:
  %run2_0 = copy 1
  %base2 = w loadub (add %src, %j)
  jmp @run2_check

@run2_check:
  %run2 = phi %run2_0, %run2_inc
  %next2 = add %j, %run2
  %c1b = csltl %next2, %n
  %c2b = csltl %run2, 128
  %c_b = and %c1b, %c2b
  jnz %c_b, @run2_cmp
  jmp @run2_end

@run2_cmp:
  %ptr2 = add %src, %next2
  %cur2 = w loadub %ptr2
  %eq2 = ceql %cur2, %base2
  jnz %eq2, @run2_inc
  jmp @run2_end

@run2_inc:
  %run2_inc = add %run2, 1
  jmp @run2_check

@run2_end:
  %run2_ge3 = csltl 3, %run2   // run2 >= 3 ?
  jnz %run2_ge3, @literal_stop
  jmp @literal_inc

@literal_inc:
  %lit_len_inc = add %lit_len, 1
  jmp @literal_check

@literal_stop:
  jmp @literal_done

@literal_done:
  %need2 = add %outpos, 1
  %need2 = add %need2, %lit_len
  %cap_lt_need2 = csltl %cap, %need2
  jnz %cap_lt_need2, @fail
  %header_lit = sub %lit_len, 1
  %outptr2 = add %out, %outpos
  storeb %header_lit, %outptr2
  %copy_i0 = copy 0
  jmp @copy_loop

@copy_loop:
  %copy_i = phi %copy_i0, %copy_i_next
  %c_copy = csltl %copy_i, %lit_len
  jnz %c_copy, @copy_body
  jmp @copy_end

@copy_body:
  %src_addr = add %src, %lit_start
  %src_addr = add %src_addr, %copy_i
  %byte = w loadub %src_addr
  %dst_addr = add %out, %outpos
  %dst_addr = add %dst_addr, 1
  %dst_addr = add %dst_addr, %copy_i
  storeb %byte, %dst_addr
  %copy_i_next = add %copy_i, 1
  jmp @copy_loop

@copy_end:
  %i_next = add %lit_start, %lit_len
  %outpos_next = add %outpos, 1
  %outpos_next = add %outpos_next, %lit_len
  jmp @loop

@fail:
  ret -1
}
