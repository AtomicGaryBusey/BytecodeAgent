export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
    %zero = copy 0
    %i0 = copy %zero
    %outpos0 = copy %zero
    jmp %loop_cond

@loop_cond:
    %i = phi %i0, %i_next
    %outpos = phi %outpos0, %outpos_next
    %c_i_n = csltl %i, %n
    jnz %c_i_n, %run_start
    ret %outpos

@run_start:
    %src_i = add %src, %i
    %base = w loadub %src_i
    %run0 = copy 1
    jmp %run_loop

@run_loop:
    %run = phi %run0, %run_inc
    %i_plus_run = add %i, %run
    %c1 = csltl %i_plus_run, %n
    %c2 = csltl %run, 128
    %c = and %c1, %c2
    jnz %c, %run_eq
    jmp %run_done

@run_eq:
    %ptr = add %src, %i_plus_run
    %cur = w loadub %ptr
    %eq = ceql %cur, %base
    jnz %eq, %run_inc
    jmp %run_done

@run_inc:
    %run_inc = add %run, 1
    jmp %run_loop

@run_done:
    %c_lt3 = csltl %run, 3
    jnz %c_lt3, %literal_start
    %need = add %outpos, 2
    %c_over = csltl %cap, %need
    jnz %c_over, %fail
    %header = sub 257, %run
    %outptr = add %out, %outpos
    storeb %header, %outptr
    %outptr1 = add %outptr, 1
    storeb %base, %outptr1
    %outpos_next = add %outpos, 2
    %i_next = add %i, %run
    jmp %loop_cond

@literal_start:
    %lit_start = copy %i
    %lit_len0 = copy %zero
    jmp %literal_cond

@literal_cond:
    %lit_len = phi %lit_len0, %lit_len_inc
    %j = add %lit_start, %lit_len
    %c1 = csltl %j, %n
    %c2 = csltl %lit_len, 128
    %c = and %c1, %c2
    jnz %c, %literal_check_run
    jmp %literal_done

@literal_check_run:
    %run2_0 = copy 1
    %src_j = add %src, %j
    %base2 = w loadub %src_j
    jmp %literal_run_loop

@literal_run_loop:
    %run2 = phi %run2_0, %run2_inc
    %j_plus_run = add %j, %run2
    %c1b = csltl %j_plus_run, %n
    %c2b = csltl %run2, 128
    %c_b = and %c1b, %c2b
    jnz %c_b, %literal_run_eq
    jmp %literal_no_run

@literal_run_eq:
    %ptr2 = add %src, %j_plus_run
    %cur2 = w loadub %ptr2
    %eq2 = ceql %cur2, %base2
    jnz %eq2, %literal_run_inc
    jmp %literal_no_run

@literal_run_inc:
    %run2_inc = add %run2, 1
    jmp %literal_run_loop

@literal_no_run:
    %c_ge3 = csltl 2, %run2   // run2 >= 3 ?
    jnz %c_ge3, %literal_done
    jmp %literal_add

@literal_add:
    %lit_len_inc = add %lit_len, 1
    jmp %literal_cond

@literal_done:
    %need2 = add %outpos, 1
    %need2 = add %need2, %lit_len
    %c_over2 = csltl %cap, %need2
    jnz %c_over2, %fail
    %header_lit = sub %lit_len, 1
    %outptr2 = add %out, %outpos
    storeb %header_lit, %outptr2
    %copy_i0 = copy %zero
    jmp %copy_loop

@copy_loop:
    %copy_i = phi %copy_i0, %copy_i_next
    %c_copy = csltl %copy_i, %lit_len
    jnz %c_copy, %copy_body
    jmp %copy_done

@copy_body:
    %src_addr = add %src, %lit_start
    %src_addr = add %src_addr, %copy_i
    %byte = w loadub %src_addr
    %dst_addr = add %out, %outpos
    %dst_addr = add %dst_addr, 1
    %dst_addr = add %dst_addr, %copy_i
    storeb %byte, %dst_addr
    %copy_i_next = add %copy_i, 1
    jmp %copy_loop

@copy_done:
    %outpos_next = add %outpos, 1
    %outpos_next = add %outpos_next, %lit_len
    %i_next = add %lit_start, %lit_len
    jmp %loop_cond

@fail:
    ret -1
}
