export function bubble_sort(l a, l n) l {
@entry:
    %zero = constl 0
    %one = constl 1
    %n_is_one_or_zero = csltl %n, %one
    jnz %n_is_one_or_zero, @ret

    %i0 = constl 0
    jmp @outer_cond

@outer_cond:
    %i = phi %i0, %i_next
    %nminus1 = subl %n, %one
    %c_i = csltl %i, %nminus1
    jnz %c_i, @inner_start
    jmp @ret

@inner_start:
    %tmp1 = subl %n, %i
    %inner_limit = subl %tmp1, %one
    %j0 = constl 0
    jmp @inner_cond

@inner_cond:
    %j = phi %j0, %j_next
    %c_j = csltl %j, %inner_limit
    jnz %c_j, @cmp_swap
    jmp @outer_inc

@cmp_swap:
    %shift1 = lsl %j, 3
    %addr1 = addl %a, %shift1
    %val1 = loadl (%addr1)
    %j1 = addl %j, %one
    %shift2 = lsl %j1, 3
    %addr2 = addl %a, %shift2
    %val2 = loadl (%addr2)
    %gt = cgtl %val1, %val2
    jnz %gt, @do_swap
    jmp @inner_inc

@do_swap:
    storel %val2, (%addr1)
    storel %val1, (%addr2)
    jmp @inner_inc

@inner_inc:
    %j_next = addl %j, %one
    jmp @inner_cond

@outer_inc:
    %i_next = addl %i, %one
    jmp @outer_cond

@ret:
    retl %zero
}
