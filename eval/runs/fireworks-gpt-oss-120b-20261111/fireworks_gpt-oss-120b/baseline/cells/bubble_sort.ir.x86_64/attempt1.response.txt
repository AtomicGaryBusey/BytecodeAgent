export function l $bubble_sort(l a, l n) {
@entry:
    %n1 = subl %n, 1
    %cond = csltl %n1, 0
    jnz %cond, @ret

    %i0 = constl 0
    jmp @outer_cond

@outer_cond:
    %i = phi %i0, %i_next
    %i_end = subl %n, 1
    %c_i = csltl %i, %i_end
    jnz %c_i, @inner_start
    jmp @ret

@inner_start:
    %inner_tmp = subl %n, %i
    %inner_limit = subl %inner_tmp, 1
    %j0 = constl 0
    jmp @inner_cond

@inner_cond:
    %j = phi %j0, %j_next
    %c_j = csltl %j, %inner_limit
    jnz %c_j, @cmp_swap
    jmp @outer_next

@cmp_swap:
    %shift1 = lsl %j, 3
    %addr1 = addl %a, %shift1
    %val1 = loadl (%addr1)
    %j1 = addl %j, 1
    %shift2 = lsl %j1, 3
    %addr2 = addl %a, %shift2
    %val2 = loadl (%addr2)
    %gt = cgtl %val1, %val2
    jnz %gt, @do_swap
    jmp @inner_next

@do_swap:
    storel %val2, (%addr1)
    storel %val1, (%addr2)
    jmp @inner_next

@inner_next:
    %j_next = addl %j, 1
    jmp @inner_cond

@outer_next:
    %i_next = addl %i, 1
    jmp @outer_cond

@ret:
    retl 0
}
