export function l $base64_encode(l a0, l a1, l a2) {
    @entry
    %src = l %a0
    %n = l %a1
    %out = l %a2

    %full = l div %n, 3
    %rem = l mod %n, 3
    %rem_nz = w csltl 0, %rem
    %out_groups = add %full, %rem_nz
    %out_len = mul %out_groups, 4

    %i = l 0
    %out_ptr = l %out

    @full_loop
    %c_full = w csltl %i, %full
    jnz %c_full, @full_body, @after_full

    @full_body
    %src_off = mul %i, 3
    %p0 = add %src, %src_off
    %b0 = l loadub %p0
    %p1 = add %p0, 1
    %b1 = l loadub %p1
    %p2 = add %p0, 2
    %b2 = l loadub %p2

    %v0 = srl %b0, 2
    %t0 = and %b0, 3
    %t1 = sll %t0, 4
    %v1 = or %t1, srl %b1, 4
    %t2 = and %b1, 15
    %t3 = sll %t2, 2
    %v2 = or %t3, srl %b2, 6
    %v3 = and %b2, 63

    // encode v0
    %c0 = add %v0, 65
    %c0_lt26 = w csltl %v0, 26
    jnz %c0_lt26, @store0_char, @c0_step1
    @c0_step1
    %c0 = add %v0, 71
    %c0_lt52 = w csltl %v0, 52
    jnz %c0_lt52, @store0_char, @c0_step2
    @c0_step2
    %c0 = add %v0, -4
    %c0_lt62 = w csltl %v0, 62
    jnz %c0_lt62, @store0_char, @c0_step3
    @c0_step3
    %c0_eq62 = w ceql %v0, 62
    jnz %c0_eq62, @store0_plus, @store0_slash
    @store0_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1
    @store0_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1
    @store0_char
    storeb %c0, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1

    // encode v1
    @store1
    %c1 = add %v1, 65
    %c1_lt26 = w csltl %v1, 26
    jnz %c1_lt26, @store1_char, @c1_step1
    @c1_step1
    %c1 = add %v1, 71
    %c1_lt52 = w csltl %v1, 52
    jnz %c1_lt52, @store1_char, @c1_step2
    @c1_step2
    %c1 = add %v1, -4
    %c1_lt62 = w csltl %v1, 62
    jnz %c1_lt62, @store1_char, @c1_step3
    @c1_step3
    %c1_eq62 = w ceql %v1, 62
    jnz %c1_eq62, @store1_plus, @store1_slash
    @store1_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2
    @store1_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2
    @store1_char
    storeb %c1, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2

    // encode v2
    @store2
    %c2 = add %v2, 65
    %c2_lt26 = w csltl %v2, 26
    jnz %c2_lt26, @store2_char, @c2_step1
    @c2_step1
    %c2 = add %v2, 71
    %c2_lt52 = w csltl %v2, 52
    jnz %c2_lt52, @store2_char, @c2_step2
    @c2_step2
    %c2 = add %v2, -4
    %c2_lt62 = w csltl %v2, 62
    jnz %c2_lt62, @store2_char, @c2_step3
    @c2_step3
    %c2_eq62 = w ceql %v2, 62
    jnz %c2_eq62, @store2_plus, @store2_slash
    @store2_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store3
    @store2_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store3
    @store2_char
    storeb %c2, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store3

    // encode v3
    @store3
    %c3 = add %v3, 65
    %c3_lt26 = w csltl %v3, 26
    jnz %c3_lt26, @store3_char, @c3_step1
    @c3_step1
    %c3 = add %v3, 71
    %c3_lt52 = w csltl %v3, 52
    jnz %c3_lt52, @store3_char, @c3_step2
    @c3_step2
    %c3 = add %v3, -4
    %c3_lt62 = w csltl %v3, 62
    jnz %c3_lt62, @store3_char, @c3_step3
    @c3_step3
    %c3_eq62 = w ceql %v3, 62
    jnz %c3_eq62, @store3_plus, @store3_slash
    @store3_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @next_iter
    @store3_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @next_iter
    @store3_char
    storeb %c3, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @next_iter

    @next_iter
    %i = add %i, 1
    jnz 1, @full_loop

    @after_full
    %rem_zero = w ceql %rem, 0
    jnz %rem_zero, @done, @rem_nonzero

    @rem_nonzero
    %rem_one = w ceql %rem, 1
    jnz %rem_one, @rem1, @rem2

    // remainder 1 byte
    @rem1
    %src_off1 = mul %full, 3
    %p0r = add %src, %src_off1
    %b0r = l loadub %p0r

    %v0r = srl %b0r, 2
    %t0r = and %b0r, 3
    %v1r = sll %t0r, 4

    // encode v0r
    %c0r = add %v0r, 65
    %c0r_lt26 = w csltl %v0r, 26
    jnz %c0r_lt26, @store0r_char, @c0r_step1
    @c0r_step1
    %c0r = add %v0r, 71
    %c0r_lt52 = w csltl %v0r, 52
    jnz %c0r_lt52, @store0r_char, @c0r_step2
    @c0r_step2
    %c0r = add %v0r, -4
    %c0r_lt62 = w csltl %v0r, 62
    jnz %c0r_lt62, @store0r_char, @c0r_step3
    @c0r_step3
    %c0r_eq62 = w ceql %v0r, 62
    jnz %c0r_eq62, @store0r_plus, @store0r_slash
    @store0r_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1r
    @store0r_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1r
    @store0r_char
    storeb %c0r, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1r

    // encode v1r
    @store1r
    %c1r = add %v1r, 65
    %c1r_lt26 = w csltl %v1r, 26
    jnz %c1r_lt26, @store1r_char, @c1r_step1
    @c1r_step1
    %c1r = add %v1r, 71
    %c1r_lt52 = w csltl %v1r, 52
    jnz %c1r_lt52, @store1r_char, @c1r_step2
    @c1r_step2
    %c1r = add %v1r, -4
    %c1r_lt62 = w csltl %v1r, 62
    jnz %c1r_lt62, @store1r_char, @c1r_step3
    @c1r_step3
    %c1r_eq62 = w ceql %v1r, 62
    jnz %c1r_eq62, @store1r_plus, @store1r_slash
    @store1r_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad1r
    @store1r_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad1r
    @store1r_char
    storeb %c1r, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad1r

    @pad1r
    storeb 61, %out_ptr
    %out_ptr = add %out_ptr, 1
    storeb 61, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @done

    // remainder 2 bytes
    @rem2
    %src_off2 = mul %full, 3
    %p0s = add %src, %src_off2
    %b0s = l loadub %p0s
    %p1s = add %p0s, 1
    %b1s = l loadub %p1s

    %v0s = srl %b0s, 2
    %t0s = and %b0s, 3
    %t1s = sll %t0s, 4
    %v1s = or %t1s, srl %b1s, 4
    %t2s = and %b1s, 15
    %v2s = sll %t2s, 2

    // encode v0s
    %c0s = add %v0s, 65
    %c0s_lt26 = w csltl %v0s, 26
    jnz %c0s_lt26, @store0s_char, @c0s_step1
    @c0s_step1
    %c0s = add %v0s, 71
    %c0s_lt52 = w csltl %v0s, 52
    jnz %c0s_lt52, @store0s_char, @c0s_step2
    @c0s_step2
    %c0s = add %v0s, -4
    %c0s_lt62 = w csltl %v0s, 62
    jnz %c0s_lt62, @store0s_char, @c0s_step3
    @c0s_step3
    %c0s_eq62 = w ceql %v0s, 62
    jnz %c0s_eq62, @store0s_plus, @store0s_slash
    @store0s_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1s
    @store0s_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1s
    @store0s_char
    storeb %c0s, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store1s

    // encode v1s
    @store1s
    %c1s = add %v1s, 65
    %c1s_lt26 = w csltl %v1s, 26
    jnz %c1s_lt26, @store1s_char, @c1s_step1
    @c1s_step1
    %c1s = add %v1s, 71
    %c1s_lt52 = w csltl %v1s, 52
    jnz %c1s_lt52, @store1s_char, @c1s_step2
    @c1s_step2
    %c1s = add %v1s, -4
    %c1s_lt62 = w csltl %v1s, 62
    jnz %c1s_lt62, @store1s_char, @c1s_step3
    @c1s_step3
    %c1s_eq62 = w ceql %v1s, 62
    jnz %c1s_eq62, @store1s_plus, @store1s_slash
    @store1s_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2s
    @store1s_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2s
    @store1s_char
    storeb %c1s, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @store2s

    // encode v2s
    @store2s
    %c2s = add %v2s, 65
    %c2s_lt26 = w csltl %v2s, 26
    jnz %c2s_lt26, @store2s_char, @c2s_step1
    @c2s_step1
    %c2s = add %v2s, 71
    %c2s_lt52 = w csltl %v2s, 52
    jnz %c2s_lt52, @store2s_char, @c2s_step2
    @c2s_step2
    %c2s = add %v2s, -4
    %c2s_lt62 = w csltl %v2s, 62
    jnz %c2s_lt62, @store2s_char, @c2s_step3
    @c2s_step3
    %c2s_eq62 = w ceql %v2s, 62
    jnz %c2s_eq62, @store2s_plus, @store2s_slash
    @store2s_plus
    storeb 43, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad2s
    @store2s_slash
    storeb 47, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad2s
    @store2s_char
    storeb %c2s, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @pad2s

    @pad2s
    storeb 61, %out_ptr
    %out_ptr = add %out_ptr, 1
    jnz 1, @done

    @done
    ret %out_len
}
