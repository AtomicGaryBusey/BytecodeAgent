export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
entry:
    %n_sq = mul %n %n
    %nsq8 = mul %n_sq 8
    %one = 1
    %zero = 0
    %eight = 8
    jmp %id_entry

id_entry:
    %i_id = phi l 0 %id_entry %i_id_next %id_i_inc %i_id_next
    %cond_i_id = csltl %i_id %n
    jnz %cond_i_id, %id_body, %id_done

id_body:
    %j_id = phi l 0 %id_body %j_id_next %id_set %j_id_next
    %cond_j_id = csltl %j_id %n
    jnz %cond_j_id, %id_set, %id_i_inc

id_set:
    %eq = ceql %i_id %j_id
    %val = select %eq %one %zero
    %i_n = mul %i_id %n
    %idx = add %i_n %j_id
    %off = mul %idx %eight
    %addr = add %out %off
    storel %val %addr
    %j_id_next = add %j_id %one
    jmp %id_body with %j_id=%j_id_next

id_i_inc:
    %i_id_next = add %i_id %one
    jmp %id_entry with %i_id=%i_id_next %j_id=0

id_done:
    %p = phi l 0 %copy_entry %p_next %copy_done %p_next
    %cond_p = csltl %p %n_sq
    jnz %cond_p, %copy_body, %copy_done

copy_body:
    %off_a = mul %p %eight
    %addr_a = add %a %off_a
    %val_a = loadl %addr_a
    %addr_base = add %tmp %off_a
    storel %val_a %addr_base
    %p_next = add %p %one
    jmp %copy_entry with %p=%p_next

copy_done:
    %exp = phi l %k %exp_entry %exp_next %exp_body %exp_next
    %cond_exp = csltl %exp %one
    jnz %cond_exp, %exp_done, %exp_body

exp_body:
    %odd = and %exp %one
    %odd_w = cneql %odd %zero
    jnz %odd_w, %mult_out, %skip_mult

mult_out:
    %base_ptr = %tmp
    %out_ptr = %out
    %dest_ptr = add %tmp %nsq8
    %i_mul = phi l 0 %mul_i_entry %i_mul_next %mul_i_inc %i_mul_next
    %cond_i_mul = csltl %i_mul %n
    jnz %cond_i_mul, %mul_j_entry, %mul_done

mul_j_entry:
    %j_mul = phi l 0 %mul_j_entry %j_mul_next %mul_j_inc %j_mul_next
    %cond_j_mul = csltl %j_mul %n
    jnz %cond_j_mul, %mul_t_entry, %mul_i_inc

mul_t_entry:
    %t_mul = phi l 0 %mul_t_entry %t_mul_next %mul_t_inc %t_mul_next
    %sum = phi l 0 %mul_t_entry %sum_next %mul_t_inc %sum_next
    %cond_t_mul = csltl %t_mul %n
    jnz %cond_t_mul, %mul_compute, %mul_store

mul_compute:
    %i_n = mul %i_mul %n
    %idx_a = add %i_n %t_mul
    %off_a = mul %idx_a %eight
    %addr_a = add %out_ptr %off_a
    %val_a = loadl %addr_a
    %t_n = mul %t_mul %n
    %idx_b = add %t_n %j_mul
    %off_b = mul %idx_b %eight
    %addr_b = add %base_ptr %off_b
    %val_b = loadl %addr_b
    %prod = mul %val_a %val_b
    %new_sum = add %sum %prod
    %t_mul_next = add %t_mul %one
    jmp %mul_t_entry with %t_mul=%t_mul_next %sum=%new_sum

mul_store:
    %i_n2 = mul %i_mul %n
    %idx_c = add %i_n2 %j_mul
    %off_c = mul %idx_c %eight
    %addr_c = add %dest_ptr %off_c
    storel %sum %addr_c
    %j_mul_next = add %j_mul %one
    jmp %mul_j_entry with %j_mul=%j_mul_next

mul_i_inc:
    %i_mul_next = add %i_mul %one
    jmp %mul_i_entry with %i_mul=%i_mul_next %j_mul=0

mul_done:
    %p_c = phi l 0 %copy_res_entry %p_c_next %copy_res_done %p_c_next
    %cond_p_c = csltl %p_c %n_sq
    jnz %cond_p_c, %copy_res_body, %copy_res_done

copy_res_body:
    %off = mul %p_c %eight
    %addr_src = add %dest_ptr %off
    %val = loadl %addr_src
    %addr_dst = add %out %off
    storel %val %addr_dst
    %p_c_next = add %p_c %one
    jmp %copy_res_entry with %p_c=%p_c_next

copy_res_done:
    jmp %skip_mult

skip_mult:
    %base_ptr2 = %tmp
    %dest_ptr2 = add %tmp %nsq8
    %i_sq = phi l 0 %sq_i_entry %i_sq_next %sq_i_inc %i_sq_next
    %cond_i_sq = csltl %i_sq %n
    jnz %cond_i_sq, %sq_j_entry, %sq_done

sq_j_entry:
    %j_sq = phi l 0 %sq_j_entry %j_sq_next %sq_j_inc %j_sq_next
    %cond_j_sq = csltl %j_sq %n
    jnz %cond_j_sq, %sq_t_entry, %sq_i_inc

sq_t_entry:
    %t_sq = phi l 0 %sq_t_entry %t_sq_next %sq_t_inc %t_sq_next
    %sum_sq = phi l 0 %sq_t_entry %sum_sq_next %sq_t_inc %sum_sq_next
    %cond_t_sq = csltl %t_sq %n
    jnz %cond_t_sq, %sq_compute, %sq_store

sq_compute:
    %i_n3 = mul %i_sq %n
    %idx_x = add %i_n3 %t_sq
    %off_x = mul %idx_x %eight
    %addr_x = add %base_ptr2 %off_x
    %val_x = loadl %addr_x
    %t_n3 = mul %t_sq %n
    %idx_y = add %t_n3 %j_sq
    %off_y = mul %idx_y %eight
    %addr_y = add %base_ptr2 %off_y
    %val_y = loadl %addr_y
    %prod2 = mul %val_x %val_y
    %new_sum2 = add %sum_sq %prod2
    %t_sq_next = add %t_sq %one
    jmp %sq_t_entry with %t_sq=%t_sq_next %sum_sq=%new_sum2

sq_store:
    %i_n4 = mul %i_sq %n
    %idx_z = add %i_n4 %j_sq
    %off_z = mul %idx_z %eight
    %addr_z = add %dest_ptr2 %off_z
    storel %sum_sq %addr_z
    %j_sq_next = add %j_sq %one
    jmp %sq_j_entry with %j_sq=%j_sq_next

sq_i_inc:
    %i_sq_next = add %i_sq %one
    jmp %sq_i_entry with %i_sq=%i_sq_next %j_sq=0

sq_done:
    %p_bc = phi l 0 %copy_base_entry %p_bc_next %copy_base_done %p_bc_next
    %cond_p_bc = csltl %p_bc %n_sq
    jnz %cond_p_bc, %copy_base_body, %copy_base_done

copy_base_body:
    %off_bc = mul %p_bc %eight
    %addr_src2 = add %dest_ptr2 %off_bc
    %val_src2 = loadl %addr_src2
    %addr_dst2 = add %base_ptr2 %off_bc
    storel %val_src2 %addr_dst2
    %p_bc_next = add %p_bc %one
    jmp %copy_base_entry with %p_bc=%p_bc_next

copy_base_done:
    %exp_next = lsr %exp %one
    jmp %exp_entry with %exp=%exp_next

exp_done:
    %p_z = phi l 0 %zero_entry %p_z_next %zero_done %p_z_next
    %cond_p_z = csltl %p_z %n_sq
    jnz %cond_p_z, %zero_body, %zero_done

zero_body:
    %off_z = mul %p_z %eight
    %addr_z = add %tmp %off_z
    storel %zero %addr_z
    %p_z_next = add %p_z %one
    jmp %zero_entry with %p_z=%p_z_next

zero_done:
    ret 0
}
