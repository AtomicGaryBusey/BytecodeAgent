export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
%0:
    %n_sq = mul %n %n
    %eight = const 8
    %one = const 1
    %zero = const 0
    %nsq8 = mul %n_sq %eight
    jmp %i_entry

%i_entry:
    %i = phi l 0 %0 %i_next %i_inc
    %cond_i = csltl %i %n
    jnz %cond_i %j_entry %i_done

%j_entry:
    %j = phi l 0 %i_entry %j_next %j_inc
    %cond_j = csltl %j %n
    jnz %cond_j %set_val %j_inc

%set_val:
    %eq = ceql %i %j
    %val = select %eq %one %zero
    %i_n = mul %i %n
    %idx = add %i_n %j
    %off = mul %idx %eight
    %addr = add %out %off
    storel %val %addr
    %j_next = add %j %one
    jmp %j_entry

%j_inc:
    %i_next = add %i %one
    jmp %i_entry

%i_done:
    %p = phi l 0 %i_done %p_next %p_inc
    %cond_p = csltl %p %n_sq
    jnz %cond_p %copy_a %copy_done

%copy_a:
    %off_a = mul %p %eight
    %addr_a = add %a %off_a
    %val_a = loadl %addr_a
    %addr_b = add %tmp %off_a
    storel %val_a %addr_b
    %p_next = add %p %one
    jmp %p

%copy_done:
    %exp = phi l %k %copy_done %exp_next %exp_body
    %cond_exp = csltl %exp %one
    jnz %cond_exp %exp_done %exp_body

%exp_body:
    %odd = and %exp %one
    %odd_w = cneql %odd %zero
    jnz %odd_w %mult_out %skip_mult

%mult_out:
    %src1 = %out
    %src2 = %tmp
    %dst = add %tmp %nsq8
    jmp %mul_i_entry

%skip_mult:
    %src1 = %tmp
    %src2 = %tmp
    %dst = add %tmp %nsq8
    jmp %mul_i_entry

%mul_i_entry:
    %i_mul = phi l 0 %mult_out %i_mul_next %i_mul_inc
    %cond_i_mul = csltl %i_mul %n
    jnz %cond_i_mul %mul_j_entry %mul_done

%mul_j_entry:
    %j_mul = phi l 0 %mul_i_entry %j_mul_next %j_mul_inc
    %cond_j_mul = csltl %j_mul %n
    jnz %cond_j_mul %mul_t_entry %mul_i_inc

%mul_t_entry:
    %t_mul = phi l 0 %mul_j_entry %t_mul_next %t_mul_inc
    %sum = phi l 0 %mul_j_entry %sum_next %t_mul_inc
    %cond_t_mul = csltl %t_mul %n
    jnz %cond_t_mul %mul_compute %mul_store

%mul_compute:
    %i_n = mul %i_mul %n
    %idx_a = add %i_n %t_mul
    %off_a = mul %idx_a %eight
    %addr_a = add %src1 %off_a
    %val_a = loadl %addr_a
    %t_n = mul %t_mul %n
    %idx_b = add %t_n %j_mul
    %off_b = mul %idx_b %eight
    %addr_b = add %src2 %off_b
    %val_b = loadl %addr_b
    %prod = mul %val_a %val_b
    %new_sum = add %sum %prod
    %t_mul_next = add %t_mul %one
    jmp %mul_t_entry %t_mul=%t_mul_next %sum=%new_sum

%mul_store:
    %i_n2 = mul %i_mul %n
    %idx_c = add %i_n2 %j_mul
    %off_c = mul %idx_c %eight
    %addr_c = add %dst %off_c
    storel %sum %addr_c
    %j_mul_next = add %j_mul %one
    jmp %mul_j_entry %j_mul=%j_mul_next

%i_mul_inc:
    %i_mul_next = add %i_mul %one
    jmp %mul_i_entry %i_mul=%i_mul_next %j_mul=0

%mul_done:
    // If we just multiplied out *= B (odd case), copy result back to out
    %odd_check = cneql %odd %zero
    %copy_dest = select %odd_check %dst %src1
    // copy %dst to %copy_dest if odd, else copy %dst to %src2 (base squaring)
    %p_c = phi l 0 %mul_done %p_c_next %p_c_inc
    %cond_p_c = csltl %p_c %n_sq
    jnz %cond_p_c %copy_res_body %copy_res_done

%copy_res_body:
    %off = mul %p_c %eight
    %addr_src = add %dst %off
    %val = loadl %addr_src
    %addr_dst = add %copy_dest %off
    storel %val %addr_dst
    %p_c_next = add %p_c %one
    jmp %mul_done %p_c=%p_c_next

%copy_res_done:
    // square base: tmp = tmp * tmp
    %exp_next = lsr %exp %one
    jmp %exp_body %exp=%exp_next

%exp_done:
    // zero out tmp (both buffers)
    %z = phi l 0 %exp_done %z_next %z_inc
    %cond_z = csltl %z %n_sq
    jnz %cond_z %zero_body %zero_done

%zero_body:
    %off_z = mul %z %eight
    %addr_z = add %tmp %off_z
    storel %zero %addr_z
    %z_next = add %z %one
    jmp %exp_done %z=%z_next

%zero_done:
    ret 0
}
