export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
%zero = const 0
%one = const 1
%eight = const 8

%nsq = mul %n %n
%nsq8 = mul %nsq %eight
%prod = add %tmp %nsq8

; ---- identity matrix in out ----
%id_i_entry:
    %i = phi l 0 %entry %i_next %id_j_inc
    %cond_i = csltl %i %n
    jnz %cond_i %id_j_entry %id_done

%id_j_entry:
    %j = phi l 0 %id_i_entry %j_next %id_j_inc2
    %cond_j = csltl %j %n
    jnz %cond_j %id_set %id_j_inc2

%id_set:
    %eq = ceql %i %j
    %val = select l %eq %one %zero
    %i_n = mul %i %n
    %idx = add %i_n %j
    %off = mul %idx %eight
    %addr = add %out %off
    storel %val %addr
    %j_next = add %j %one
    jmp %id_j_entry

%id_j_inc2:
    %i_next = add %i %one
    jmp %id_i_entry %i=%i_next

%id_done:
    ; ---- copy a to tmp (base matrix) ----
    jmp %copy_entry

%copy_entry:
    %p = phi l 0 %id_done %p_next %copy_body
    %cond_p = csltl %p %nsq
    jnz %cond_p %copy_body %exp_entry

%copy_body:
    %off_a = mul %p %eight
    %addr_a = add %a %off_a
    %val_a = loadl %addr_a
    %addr_tmp = add %tmp %off_a
    storel %val_a %addr_tmp
    %p_next = add %p %one
    jmp %copy_entry %p=%p_next

%exp_entry:
    %exp = phi l %k %copy_entry %exp_next %square_done
    %cond_exp = csltl %zero %exp
    jnz %cond_exp %exp_body %exp_exit

%exp_body:
    %odd = and %exp %one
    %odd_w = cneql %odd %zero
    %is_out_mul = %odd_w
    %src1 = %out
    %src2 = %tmp
    jmp %mul_i_entry

%mul_i_entry:
    %i_mul = phi l 0 %exp_body %i_next %i_inc
    %cond_i = csltl %i_mul %n
    jnz %cond_i %mul_j_entry %mul_done

%mul_j_entry:
    %j_mul = phi l 0 %mul_i_entry %j_next %j_inc
    %cond_j = csltl %j_mul %n
    jnz %cond_j %mul_t_entry %j_inc

%mul_t_entry:
    %t_mul = phi l 0 %mul_j_entry %t_next %t_inc
    %sum = phi l 0 %mul_j_entry %sum_next %t_inc
    %cond_t = csltl %t_mul %n
    jnz %cond_t %mul_compute %mul_store

%mul_compute:
    %i_n = mul %i_mul %n
    %idx_a = add %i_n %t_mul
    %off_a = mul %idx_a %eight
    %addr_a = add %src1 %off_a
    %val_a = loadl %addr_a
    %t_n = mul %t_mul %n
    %idx_b = add %t_n %j_mul
    %off_b = mul %idx_b %eight
    %addr_b = add %src2 %off_b
    %val_b = loadl %addr_b
    %prod_mul = mul %val_a %val_b
    %new_sum = add %sum %prod_mul
    %t_next = add %t_mul %one
    jmp %mul_t_entry %t=%t_next %sum=%new_sum

%mul_store:
    %i_n2 = mul %i_mul %n
    %idx_c = add %i_n2 %j_mul
    %off_c = mul %idx_c %eight
    %addr_c = add %prod %off_c
    storel %sum %addr_c
    %j_next = add %j_mul %one
    jmp %mul_j_entry %j=%j_next

%j_inc:
    %i_next = add %i_mul %one
    jmp %mul_i_entry %i=%i_next

%mul_done:
    %cond_out = %is_out_mul
    jnz %cond_out %copy_out %copy_base

%copy_out:
    %p = phi l 0 %copy_out %p_next %copy_out_body
    %cond_p = csltl %p %nsq
    jnz %cond_p %copy_out_body %copy_base

%copy_out_body:
    %off = mul %p %eight
    %src = add %prod %off
    %val = loadl %src
    %dst_out = add %out %off
    storel %val %dst_out
    %p_next = add %p %one
    jmp %copy_out %p=%p_next

%copy_base:
    %p2 = phi l 0 %copy_base %p2_next %copy_base_body
    %cond_p2 = csltl %p2 %nsq
    jnz %cond_p2 %copy_base_body %square_done

%copy_base_body:
    %off = mul %p2 %eight
    %src = add %prod %off
    %val = loadl %src
    %dst_base = add %tmp %off
    storel %val %dst_base
    %p2_next = add %p2 %one
    jmp %copy_base %p2=%p2_next

%square_done:
    %exp_next = lsr %exp %one
    jmp %exp_entry %exp=%exp_next

%exp_exit:
    ; ---- zero out tmp and prod buffers ----
    jmp %zero_entry

%zero_entry:
    %z = phi l 0 %exp_exit %z_next %zero_body
    %cond_z = csltl %z %nsq
    jnz %cond_z %zero_body %zero_done

%zero_body:
    %off = mul %z %eight
    %addr_tmp = add %tmp %off
    storel %zero %addr_tmp
    %addr_prod = add %prod %off
    storel %zero %addr_prod
    %z_next = add %z %one
    jmp %zero_entry %z=%z_next

%zero_done:
    ret 0
}
