export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
    // Initialize result matrix (out) to identity
    %i0 = phi l 0 %entry_i0 %j_next_i %i0_inc
    %j0 = phi l 0 %i_entry_j %j_next %j0_inc
    %cond_i = csltl %i0 %n
    jnz %cond_i, %i_body, %done_i

%i_body:
    %cond_j = csltl %j0 %n
    jnz %cond_j, %j_body, %i_next

%j_body:
    %eq = ceql %i0 %j0
    %val = select %eq, 1, 0
    %i_n = mul %i0 %n
    %idx = add %i_n %j0
    %off = mul %idx 8
    %addr = add %out %off
    storel %val %addr
    %j_next = add %j0 1
    jmp %i_body with %i0=%i0 %j0=%j_next

%i_next:
    %i0_inc = add %i0 1
    jmp %entry_i0 with %i0=%i0_inc %j0=0

%done_i:

    // Copy matrix a into base (tmp base region)
    %idx0 = phi l 0 %copy_entry %idx0_inc
    %cond_idx = csltl %idx0 %n
    %n_sq = mul %n %n
    %cond_idx = csltl %idx0 %n_sq
    jnz %cond_idx, %copy_body, %copy_done

%copy_body:
    %off_a = mul %idx0 8
    %addr_a = add %a %off_a
    %val_a = loadl %addr_a
    %off_base = mul %idx0 8
    %addr_base = add %tmp %off_base
    storel %val_a %addr_base
    %idx0_inc = add %idx0 1
    jmp %copy_entry with %idx0=%idx0_inc

%copy_done:

    // exponent loop
    %exp0 = phi l %k %exp_entry %exp_next
    %cond_exp = csltl %exp0 1
    jnz %cond_exp, %exp_done, %exp_body

%exp_body:
    %odd = cneql %exp0 0
    %odd = csltl %odd %exp0   // test low bit: (exp0 & 1) != 0
    jnz %odd, %mult_out_base, %skip_mult_out

%mult_out_base:
    // out = out * base => store in tmp region after base (size n*n)
    %base_ptr = add %tmp 0
    %out_ptr = %out
    %dest_ptr = add %tmp %n_sq 8   // tmp + n*n*8
    // multiplication loops (out * base -> dest_ptr)
    %i1 = phi l 0 %mult_i_entry %i1_inc
    %j1 = phi l 0 %mult_i_entry %j1_inc
    %cond_i1 = csltl %i1 %n
    jnz %cond_i1, %mult_j_loop, %mult_done

%mult_j_loop:
    %cond_j1 = csltl %j1 %n
    jnz %cond_j1, %mult_t_loop, %i1_inc

%mult_t_loop:
    %t1 = phi l 0 %mult_t_entry %t1_inc
    %sum1 = phi l 0 %mult_t_entry %sum1_inc
    %cond_t1 = csltl %t1 %n
    jnz %cond_t1, %mult_t_body, %mult_t_done

%mult_t_body:
    %i_n1 = mul %i1 %n
    %idx1 = add %i_n1 %t1
    %off_x = mul %idx1 8
    %addr_x = add %out_ptr %off_x
    %val_x = loadl %addr_x
    %t_n1 = mul %t1 %n
    %idx2 = add %t_n1 %j1
    %off_y = mul %idx2 8
    %addr_y = add %base_ptr %off_y
    %val_y = loadl %addr_y
    %prod = mul %val_x %val_y
    %new_sum = add %sum1 %prod
    %t1_inc = add %t1 1
    jmp %mult_t_loop with %t1=%t1_inc %sum1=%new_sum

%mult_t_done:
    %i_n2 = mul %i1 %n
    %idx3 = add %i_n2 %j1
    %off_z = mul %idx3 8
    %addr_z = add %dest_ptr %off_z
    storel %sum1 %addr_z
    %j1_inc = add %j1 1
    jmp %mult_j_loop with %i1=%i1 %j1=%j1_inc

%i1_inc:
    %i1_inc = add %i1 1
    jmp %mult_i_entry with %i1=%i1_inc %j1=0

%mult_done:
    // copy dest_ptr to out (result)
    %idx_c = phi l 0 %copy_res_entry %idx_c_inc
    %cond_c = csltl %idx_c %n_sq
    jnz %cond_c, %copy_res_body, %copy_res_done

%copy_res_body:
    %off_src = mul %idx_c 8
    %addr_src = add %dest_ptr %off_src
    %val_src = loadl %addr_src
    %addr_dst = add %out %off_src
    storel %val_src %addr_dst
    %idx_c_inc = add %idx_c 1
    jmp %copy_res_entry with %idx_c=%idx_c_inc

%copy_res_done:
    jmp %skip_mult_out

%skip_mult_out:

    // base = base * base => store in tmp region after base
    %base_ptr2 = add %tmp 0
    %dest_ptr2 = add %tmp %n_sq 8
    // multiplication loops (base * base -> dest_ptr2)
    %i2 = phi l 0 %sq_i_entry %i2_inc
    %j2 = phi l 0 %sq_i_entry %j2_inc
    %cond_i2 = csltl %i2 %n
    jnz %cond_i2, %sq_j_loop, %sq_done

%sq_j_loop:
    %cond_j2 = csltl %j2 %n
    jnz %cond_j2, %sq_t_loop, %i2_inc

%sq_t_loop:
    %t2 = phi l 0 %sq_t_entry %t2_inc
    %sum2 = phi l 0 %sq_t_entry %sum2_inc
    %cond_t2 = csltl %t2 %n
    jnz %cond_t2, %sq_t_body, %sq_t_done

%sq_t_body:
    %i_n3 = mul %i2 %n
    %idx4 = add %i_n3 %t2
    %off_x2 = mul %idx4 8
    %addr_x2 = add %base_ptr2 %off_x2
    %val_x2 = loadl %addr_x2
    %t_n3 = mul %t2 %n
    %idx5 = add %t_n3 %j2
    %off_y2 = mul %idx5 8
    %addr_y2 = add %base_ptr2 %off_y2
    %val_y2 = loadl %addr_y2
    %prod2 = mul %val_x2 %val_y2
    %new_sum2 = add %sum2 %prod2
    %t2_inc = add %t2 1
    jmp %sq_t_loop with %t2=%t2_inc %sum2=%new_sum2

%sq_t_done:
    %i_n4 = mul %i2 %n
    %idx6 = add %i_n4 %j2
    %off_z2 = mul %idx6 8
    %addr_z2 = add %dest_ptr2 %off_z2
    storel %sum2 %addr_z2
    %j2_inc = add %j2 1
    jmp %sq_j_loop with %i2=%i2 %j2=%j2_inc

%i2_inc:
    %i2_inc = add %i2 1
    jmp %sq_i_entry with %i2=%i2_inc %j2=0

%sq_done:
    // copy dest_ptr2 back to base
    %idx_bc = phi l 0 %copy_base_entry %idx_bc_inc
    %cond_bc = csltl %idx_bc %n_sq
    jnz %cond_bc, %copy_base_body, %copy_base_done

%copy_base_body:
    %off_src2 = mul %idx_bc 8
    %addr_src2 = add %dest_ptr2 %off_src2
    %val_src2 = loadl %addr_src2
    %addr_dst2 = add %base_ptr2 %off_src2
    storel %val_src2 %addr_dst2
    %idx_bc_inc = add %idx_bc 1
    jmp %copy_base_entry with %idx_bc=%idx_bc_inc

%copy_base_done:
    // shift exponent
    %exp_next = lsr %exp0 1
    jmp %exp_entry with %exp0=%exp_next

%exp_done:

    // Zero out tmp (both regions)
    %idx_z = phi l 0 %zero_entry %idx_z_inc
    %cond_z = csltl %idx_z %n_sq
    jnz %cond_z, %zero_body, %zero_done

%zero_body:
    %off_z = mul %idx_z 8
    %addr_z = add %tmp %off_z
    storel 0 %addr_z
    %idx_z_inc = add %idx_z 1
    jmp %zero_entry with %idx_z=%idx_z_inc

%zero_done:

    ret 0
}
