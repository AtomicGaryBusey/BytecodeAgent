vm_run:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Lend
    mov rax, [rdi + r8*8]
    cmp rax, 0
    je .Lhalt
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc r8
    jmp .Lloop
.Lhalt:
    jmp .Lend
.Lpush:
    mov rbx, [rdi + r8*8 + 8]
    mov [rdx + r9*8], rbx
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    dec r9
    mov rax, [rdx + r9*8]          # b
    dec r9
    mov rbx, [rdx + r9*8]          # a
    add rax, rbx
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Lsub:
    dec r9
    mov rax, [rdx + r9*8]          # b
    dec r9
    mov rbx, [rdx + r9*8]          # a
    sub rbx, rax
    mov [rdx + r9*8], rbx
    inc r9
    inc r8
    jmp .Lloop
.Lmul:
    dec r9
    mov rbx, [rdx + r9*8]          # b
    dec r9
    mov rax, [rdx + r9*8]          # a
    mul rbx
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Ldup:
    mov rax, [rdx + r9*8 - 8]
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov rax, [rdx + r9*8 - 8]
    mov rbx, [rdx + r9*8 - 16]
    mov [rdx + r9*8 - 8], rbx
    mov [rdx + r9*8 - 16], rax
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov rax, [rdx + r9*8 - 16]
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    dec r9
    mov rbx, [rdx + r9*8]          # b
    dec r9
    mov rax, [rdx + r9*8]          # a
    cmp rax, rbx
    setl al
    movzx rax, al
    mov [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Ljz:
    dec r9
    mov rax, [rdx + r9*8]          # v
    cmp rax, 0
    jne .Ljz_notzero
    mov r8, [rdi + r8*8 + 8]
    jmp .Lloop
.Ljz_notzero:
    add r8, 2
    jmp .Lloop
.Ljmp:
    mov r8, [rdi + r8*8 + 8]
    jmp .Lloop
.Lend:
    mov r10, r9
    cmp r10, rcx
    jae .Lzero_done
.Lzero_loop:
    mov [rdx + r10*8], 0
