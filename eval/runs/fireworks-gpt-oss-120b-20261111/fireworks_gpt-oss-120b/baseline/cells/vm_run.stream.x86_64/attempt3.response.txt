vm_run:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Lend
    mov rax, [rdi + r8*8]
    cmp rax, 0
    je .Lend
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc r8
    jmp .Lloop
.Lpush:
    mov r10, [rdi + r8*8 + 8]
    mov [rdx + r9*8], r10
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    dec r9
    mov r10, [rdx + r9*8]
    dec r9
    mov r11, [rdx + r9*8]
    add r10, r11
    mov [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lsub:
    dec r9
    mov r10, [rdx + r9*8]
    dec r9
    mov r11, [rdx + r9*8]
    sub r11, r10
    mov [rdx + r9*8], r11
    inc r9
    inc r8
    jmp .Lloop
.Lmul:
    dec r9
    mov r10, [rdx + r9*8]
    dec r9
    mov r11, [rdx + r9*8]
    imul r10, r11
    mov [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Ldup:
    mov r10, [rdx + r9*8 - 8]
    mov [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov r10, [rdx + r9*8 - 8]
    mov r11, [rdx + r9*8 - 16]
    mov [rdx + r9*8 - 8], r11
    mov [rdx + r9*8 - 16], r10
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov r10, [rdx + r9*8 - 16]
    mov [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    dec r9
    mov r10, [rdx + r9*8]
    dec r9
    mov r11, [rdx + r9*8]
    cmp r11, r10
    setl al
    movzx r10, al
    mov [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Ljz:
    dec r9
    mov r10, [rdx + r9*8]
    cmp r10, 0
    jne .Ljz_notzero
    mov r8, [rdi + r8*8 + 8]
    jmp .Lloop
.Ljz_notzero:
    add r8, 2
    jmp .Lloop
.Ljmp:
    mov r8, [rdi + r8*8 + 8]
    jmp .Lloop
.Lend:
    mov r10, r9
    cmp r10, rcx
    jae .Lzero_done
.Lzero_loop:
    mov qword ptr [rdx + r10*8], 0
    inc r10
    cmp r10, rcx
    jb .Lzero_loop
.Lzero_done:
    cmp r9, 0
    je .Lresult_zero
    mov rax, [rdx + r9*8 - 8]
    ret
.Lresult_zero:
    xor rax, rax
    ret
