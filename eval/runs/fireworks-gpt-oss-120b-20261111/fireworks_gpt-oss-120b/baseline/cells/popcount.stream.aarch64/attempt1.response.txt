    // x0 = input
    // step 1: x = x - ((x>>1) & 0x5555555555555555)
    lsr     x1, x0, #1
    and     x1, x1, #0x5555555555555555
    sub     x0, x0, x1
    // step 2: x = (x & 0x3333333333333333) + ((x>>2) & 0x3333333333333333)
    lsr     x1, x0, #2
    and     x2, x0, #0x3333333333333333
    and     x1, x1, #0x3333333333333333
    add     x0, x2, x1
    // step 3: x = (x + (x>>4)) & 0x0F0F0F0F0F0F0F0F
    lsr     x1, x0, #4
    add     x0, x0, x1
    and     x0, x0, #0x0F0F0F0F0F0F0F0F
    // step 4: x = x * 0x0101010101010101
    movz    x1, #0x0101, lsl #0
    movk    x1, #0x0101, lsl #16
    movk    x1, #0x0101, lsl #32
    movk    x1, #0x0101, lsl #48
    mul     x0, x0, x1
    // result = high byte
    lsr     x0, x0, #56
    ret
