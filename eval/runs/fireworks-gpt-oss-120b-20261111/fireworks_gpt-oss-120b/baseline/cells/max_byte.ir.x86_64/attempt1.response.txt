export function l $max_byte(l %p, l %n)
    %zero =l copy 0
    br @loop(%zero, %zero, %p, %n)

@loop(%i, %max, %p, %n)
    %cond =l csltl %i, %n
    jnz %cond, @body(%i, %max, %p, %n)
    ret %max

@body(%i, %max, %p, %n)
    %b =l loadub %p
    %cmp =l csltl %max, %b
    %newmax =l sel %cmp %b %max
    %i2 =l add %i, 1
    %p2 =l add %p, 1
    br @loop(%i2, %newmax, %p2, %n)
