mp_mul:
    push r12
    push r13
    push r14
    push r15
    push rbx
    mov r12, rdx            # b pointer
    mov r13, r8             # r pointer
    mov r14, rcx            # nb (constant)
    mov rax, rsi
    add rax, r14            # total = na + nb
    mov rdx, rax            # preserve total in rdx
    xor rbx, rbx
.Lzero_loop:
    cmp rbx, rdx
    jge .Lzero_done
    mov qword ptr [r13 + rbx*8], 0
    inc rbx
    jmp .Lzero_loop
.Lzero_done:
    test rsi, rsi
    jz .Lreturn_len_zero
    test r14, r14
    jz .Lreturn_len_zero
    xor rbx, rbx            # outer i = 0
.Louter:
    cmp rbx, rsi
    jge .Louter_done
    mov rax, qword ptr [rdi + rbx*8]   # a_i
    xor r15, r15            # carry = 0
    xor r9, r9              # inner j = 0
.Linner:
    cmp r9, r14
    jge .Linner_done
    mov rdx, qword ptr [r12 + r9*8]    # b_j
    mul rdx                               # rdx:rax = a_i * b_j
    lea r10, [rbx + r9]                  # index i+j
    mov r11, qword ptr [r13 + r10*8]    # existing r[i+j]
    add rax, r11
    adc rdx, 0
    add rax, r15
    adc rdx, 0
    mov qword ptr [r13 + r10*8], rax   # store low
    mov r15, rdx                       # new carry
    inc r9
    jmp .Linner
.Linner_done:
    lea r10, [rbx + r14]                 # index i+nb
    mov qword ptr [r13 + r10*8], r15    # store carry
    inc rbx
    jmp .Louter
.Louter_done:
    mov rax, rdx            # total length
    dec rax                 # start from last index
    xor rcx, rcx            # result length = 0
.Lscan:
    cmp rax, -1
    jl .Lreturn_len
    mov r10, qword ptr [r13 + rax*8]
    test r10, r10
    jne .Lfound
    dec rax
    jmp .Lscan
.Lfound:
    inc rax
    mov rcx, rax
.Lreturn_len:
    mov rax, rcx
    jmp .Lepilogue
.Lreturn_len_zero:
    xor rax, rax
.Lepilogue:
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
