mp_mul:
    mov r15, rcx
    mov r14, rsi
    mov r12, rdx
    mov rax, rsi
    add rax, r15
    xor r9, r9
.Lzero_loop:
    cmp r9, rax
    jge .Lzero_done
    mov qword ptr [r8 + r9*8], 0
    inc r9
    jmp .Lzero_loop
.Lzero_done:
    test rsi, rsi
    jz .Lreturn_zero
    test r15, r15
    jz .Lreturn_zero
    xor r9, r9
.Louter:
    cmp r9, rsi
    jge .Louter_done
    mov rax, qword ptr [rdi + r9*8]
    lea r13, [r8 + r9*8]
    xor r11, r11
    xor r10, r10
.Linner:
    cmp r10, r15
    jge .Linner_done
    mov rdx, qword ptr [r12 + r10*8]
    mul rdx
    lea r14, [r13 + r10*8]
    mov rcx, qword ptr [r14]
    add rax, rcx
    adc rdx, 0
    add rax, r11
    adc rdx, 0
    mov qword ptr [r14], rax
    mov r11, rdx
    inc r10
    jmp .Linner
.Linner_done:
    mov rax, r9
    add rax, r15
    lea r14, [r8 + rax*8]
    mov qword ptr [r14], r11
    inc r9
    jmp .Louter
.Louter_done:
    mov rax, rsi
    add rax, r15
    dec rax
    xor rcx, rcx
.Lscan:
    cmp rax, -1
    jl .Lreturn_len
    mov rdx, qword ptr [r8 + rax*8]
    test rdx, rdx
    jne .Lfound
    dec rax
    jmp .Lscan
.Lfound:
    inc rax
    mov rcx, rax
.Lreturn_len:
    mov rax, rcx
    ret
.Lreturn_zero:
    xor rax, rax
    ret
