mp_mul:
    push r12
    push r13
    push r14
    push r15
    push rbx
    mov r12, rdi          # a pointer
    mov r13, rdx          # b pointer
    mov r14, r8           # r pointer
    mov r15, rcx          # nb
    mov rbx, rsi          # na
    lea r11, [rbx + r15]  # total = na + nb
    xor rax, rax
.Lzero_loop:
    cmp rax, r11
    jge .Lzero_done
    mov qword ptr [r14 + rax*8], 0
    inc rax
    jmp .Lzero_loop
.Lzero_done:
    test rbx, rbx
    jz .Lreturn_len_zero
    test r15, r15
    jz .Lreturn_len_zero
    xor rcx, rcx          # i = 0
.Louter:
    cmp rcx, rbx
    jge .Louter_done
    mov rax, qword ptr [r12 + rcx*8]   # a_i
    xor rsi, rsi          # carry = 0
    xor rdi, rdi          # j = 0
.Linner:
    cmp rdi, r15
    jge .Linner_done
    mov r8, qword ptr [r13 + rdi*8]    # b_j
    mul r8                                 # rdx:rax = a_i * b_j
    lea r9, [rcx + rdi]                 # index i+j
    mov r10, qword ptr [r14 + r9*8]     # existing result
    add rax, r10
    adc rdx, 0
    add rax, rsi
    adc rdx, 0
    mov qword ptr [r14 + r9*8], rax
    mov rsi, rdx
    inc rdi
    jmp .Linner
.Linner_done:
    lea r9, [rcx + r15]                 # index i+nb
    mov qword ptr [r14 + r9*8], rsi
    inc rcx
    jmp .Louter
.Louter_done:
    mov rax, r11
    dec rax                               # start from total-1
    xor rdx, rdx                           # result length = 0
.Lscan:
    cmp rax, -1
    jl .Lreturn_len
    mov r8, qword ptr [r14 + rax*8]
    test r8, r8
    jne .Lfound
    dec rax
    jmp .Lscan
.Lfound:
    inc rax
    mov rdx, rax
.Lreturn_len:
    mov rax, rdx
    jmp .Lepilogue
.Lreturn_len_zero:
    xor rax, rax
.Lepilogue:
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
