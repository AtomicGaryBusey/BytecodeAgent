mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi          # a pointer
    mov r14, rsi          # na
    mov r12, rdx          # b pointer
    mov r15, rcx          # nb
    mov r13, r8           # r pointer
    test r14, r14
    jz .Lreturn_zero
    test r15, r15
    jz .Lreturn_zero
    # zero result array
    mov rax, r14
    add rax, r15          # total = na+nb
    xor rcx, rcx          # i = 0
.Lzero_loop:
    cmp rcx, rax
    jge .Lzero_done
    mov qword ptr [r13 + rcx*8], 0
    inc rcx
    jmp .Lzero_loop
.Lzero_done:
    xor r9, r9            # i = 0
.Louter:
    cmp r9, r14
    jge .Louter_done
    xor r11, r11          # carry = 0
    xor r10, r10          # j = 0
.Linner:
    cmp r10, r15
    jge .Linner_done
    mov rax, qword ptr [rbx + r9*8]   # a[i]
    mul qword ptr [r12 + r10*8]      # rdx:rax = a[i] * b[j]
    mov rcx, rdx                     # high part
    lea rsi, [r9 + r10]              # idx = i+j
    add rax, qword ptr [r13 + rsi*8]
    adc rcx, 0
    add rax, r11
    adc rcx, 0
    mov qword ptr [r13 + rsi*8], rax
    mov r11, rcx
    inc r10
    jmp .Linner
.Linner_done:
    lea rsi, [r9 + r15]              # i+nb
    mov qword ptr [r13 + rsi*8], r11
    inc r9
    jmp .Louter
.Louter_done:
    # determine highest non‑zero limb
    mov rax, r14
    add rax, r15          # total = na+nb
    dec rax               # idx = total-1
    xor rcx, rcx          # result length = 0
.Lscan:
    cmp rax, -1
    jl .Lreturn_len
    mov rdx, qword ptr [r13 + rax*8]
    test rdx, rdx
    jne .Lfound_nonzero
    dec rax
    jmp .Lscan
.Lfound_nonzero:
    inc rax
    mov rcx, rax
.Lreturn_len:
    mov rax, rcx
    jmp .Lcleanup
.Lreturn_zero:
    xor rax, rax
.Lcleanup:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
