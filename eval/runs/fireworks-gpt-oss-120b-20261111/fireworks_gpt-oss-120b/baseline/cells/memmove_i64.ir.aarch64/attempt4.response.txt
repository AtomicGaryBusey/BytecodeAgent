export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %zero = const 0
    %one = const 1
    %eight = const 8

    %n_eq_zero = ceql %n, %zero
    jnz %n_eq_zero, @return

    %dst_lt_src = csltl %dst, %src
    jnz %dst_lt_src, @forward_entry
    jmp @backward_entry

@forward_entry
    %i0_f = const 0
    jmp @forward_loop

@forward_loop
    %i_f = phi %i0_f %i_next_f
    %cond_f = csltl %i_f, %n
    jnz %cond_f, @forward_body
    jmp @return

@forward_body
    %off_f = mul %i_f, %eight
    %src_addr_f = add %src, %off_f
    %val_f = loadl %src_addr_f
    %dst_addr_f = add %dst, %off_f
    storel %val_f, %dst_addr_f
    %i_next_f = add %i_f, %one
    jmp @forward_loop

@backward_entry
    %i0_b = sub %n, %one
    jmp @backward_loop

@backward_loop
    %i_b = phi %i0_b %i_next_b
    %off_b = mul %i_b, %eight
    %src_addr_b = add %src, %off_b
    %val_b = loadl %src_addr_b
    %dst_addr_b = add %dst, %off_b
    storel %val_b, %dst_addr_b
    %i_next_b = sub %i_b, %one
    %cond_b = csgel %i_b, %zero
    jnz %cond_b, @backward_loop
    jmp @return

@return
    %retval = const 0
    retl %retval
}
