export function l memmove_i64(l %dst, l %src, l %n) {
@entry
    %zero = constl 0
    %n_eq_zero = ceql %n, %zero
    jnz %n_eq_zero, @return

    %dst_lt_src = csltl %dst, %src
    jnz %dst_lt_src, @forward_entry

@backward_entry
    %i0 = sub %n, 1
    jmp @backward_loop

@backward_loop
    %i = phi %i0, %i_next
    %off = mul %i, 8
    %src_addr = add %src, %off
    %val = loadl %src_addr
    %dst_addr = add %dst, %off
    storel %val, %dst_addr
    %i_next = sub %i, 1
    %cond = csgel %i, %zero
    jnz %cond, @backward_loop, @return

@forward_entry
    %i0_f = constl 0
    jmp @forward_loop

@forward_loop
    %i_f = phi %i0_f, %i_next_f
    %cond_f = csltl %i_f, %n
    jnz %cond_f, @forward_body, @return

@forward_body
    %off_f = mul %i_f, 8
    %src_addr_f = add %src, %off_f
    %val_f = loadl %src_addr_f
    %dst_addr_f = add %dst, %off_f
    storel %val_f, %dst_addr_f
    %i_next_f = add %i_f, 1
    jmp @forward_loop

@return
    %retval = constl 0
    retl %retval
}
