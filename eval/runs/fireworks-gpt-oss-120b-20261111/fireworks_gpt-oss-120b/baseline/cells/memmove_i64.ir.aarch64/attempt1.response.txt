export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %cmp = csltl %dst, %src
    jnz %cmp, @forward_entry, @backward_entry

@forward_entry
    %i0 = constl 0
    jmp @forward_loop

@forward_loop
    %i = phi %i0, %i_next
    %cond = csltl %i, %n
    jnz %cond, @forward_body, @return

@forward_body
    %off = mul %i, 8
    %src_addr = add %src, %off
    %val = loadl %src_addr
    %dst_addr = add %dst, %off
    storel %val, %dst_addr
    %i_next = add %i, 1
    jmp @forward_loop

@backward_entry
    %zero = constl 0
    %nzero = ceql %n, %zero
    jnz %nzero, @return
    %i0 = sub %n, 1
    jmp @backward_loop

@backward_loop
    %i = phi %i0, %i_prev
    %cond = csgel %i, %zero
    jnz %cond, @backward_body, @return

@backward_body
    %off = mul %i, 8
    %src_addr = add %src, %off
    %val = loadl %src_addr
    %dst_addr = add %dst, %off
    storel %val, %dst_addr
    %i_prev = sub %i, 1
    jmp @backward_loop

@return
    %retval = constl 0
    retl %retval
}
