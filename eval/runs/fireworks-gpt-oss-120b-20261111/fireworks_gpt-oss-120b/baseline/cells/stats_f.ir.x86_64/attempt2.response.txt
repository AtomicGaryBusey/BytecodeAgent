export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
    %i =l const 0
    %min =l const 9223372036854775807
    %max =l const -9223372036854775808
    %sum =l const 0
    %oddcnt =l const 0
    jmp @cond

@body:
    %shift =l shll %i, const 3
    %ptr =l add %a, %shift
    %val =l loadl %ptr
    %y =l call %f(l %val)
    %sum =l add %sum, %y

    %odd =l and %y, const 1
    %odd_is_zero =w csltl %odd, const 1
    jnz %odd_is_zero, @skip_inc_odd
    %oddcnt =l add %oddcnt, const 1
@skip_inc_odd:

    %lt =w csltl %y, %min
    jnz %lt, @newmin
    %gt =w csgtl %y, %max
    jnz %gt, @newmax
    jmp @inc_i

@newmin:
    %min =l %y
    jmp @inc_i

@newmax:
    %max =l %y
    jmp @inc_i

@inc_i:
    %i =l add %i, const 1
@cond:
    %c =w csltl %i, %n
    jnz %c, @body

    %out0 =l add %out, const 0
    storel %min, %out0
    %out1 =l add %out, const 8
    storel %max, %out1
    %out2 =l add %out, const 16
    storel %sum, %out2
    ret %oddcnt
}
