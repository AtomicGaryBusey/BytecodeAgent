export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
    %i =l const 0
    %min =l const 9223372036854775807
    %max =l const -9223372036854775808
    %sum =l const 0
    %oddcnt =l const 0
    jmp @loop_cond

@loop_body:
    %shift =l shll %i, 3
    %ptr =l add %a, %shift
    %val =l loadl %ptr
    %y =l call %f(l %val)
    %sum =l add %sum, %y

    %odd =l and %y, 1
    %odd_is_zero =w csltl %odd, 1
    jnz %odd_is_zero, @inc_skip
    %oddcnt =l add %oddcnt, 1
@inc_skip:

    %lt =w csltl %y, %min
    jnz %lt, @newmin
    %gt =w csgtl %y, %max
    jnz %gt, @newmax
    jmp @inc_i

@newmin:
    %min =l mov %y
    jmp @inc_i

@newmax:
    %max =l mov %y
    // fallthrough

@inc_i:
    %i =l add %i, 1
    jmp @loop_cond

@loop_cond:
    %cond =w csltl %i, %n
    jnz %cond, @loop_body

    %out0 =l add %out, 0
    storel %min, %out0
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %oddcnt
}
