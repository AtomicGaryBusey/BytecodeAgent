export function l $binary_search(l %a, l %n, l %key) {
@entry:
	%low = copyl 0
	%high = copyl %n
	jmp @loop

@loop:
	%cond = csltl %low, %high
	jnz %cond, @body, @exit

@body:
	%sum = addl %low, %high
	%mid = shrl %sum, 1
	%mid8 = shll %mid, 3
	%addr = addl %a, %mid8
	%val = loadl %addr
	%eq = ceql %val, %key
	jnz %eq, @found, @cmp

@cmp:
	%lt = csltl %val, %key
	jnz %lt, @lt, @gt

@lt:
	%mid1 = addl %mid, 1
	%low = copyl %mid1
	jmp @loop

@gt:
	%high = copyl %mid
	jmp @loop

@found:
	ret %mid

@exit:
	%minus1 = copyl -1
	ret %minus1
}
