export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry:
    %pc0 = const 0
    %sp0 = const 0
    br @loop, %pc0, %sp0

@loop(%pc %sp):
    %cond = csltl %pc, %len
    jnz %cond @fetch, %pc, %sp
    br @exit, %sp

@fetch(%pc %sp):
    %off = shl %pc, 3
    %addr = add %code, %off
    %op = loadl %addr

    %halt = ceql %op, const 0
    jnz %halt @exit, %sp
    %push = ceql %op, const 1
    jnz %push @push, %pc, %sp
    %add = ceql %op, const 2
    jnz %add @add, %pc, %sp
    %sub = ceql %op, const 3
    jnz %sub @sub, %pc, %sp
    %mul = ceql %op, const 4
    jnz %mul @mul, %pc, %sp
    %dup = ceql %op, const 5
    jnz %dup @dup, %pc, %sp
    %swap = ceql %op, const 6
    jnz %swap @swap, %pc, %sp
    %drop = ceql %op, const 7
    jnz %drop @drop, %pc, %sp
    %over = ceql %op, const 8
    jnz %over @over, %pc, %sp
    %lt = ceql %op, const 9
    jnz %lt @lt, %pc, %sp
    %jz = ceql %op, const 10
    jnz %jz @jz, %pc, %sp
    %jmp = ceql %op, const 11
    jnz %jmp @jmp, %pc, %sp
    %pc1 = add %pc, const 1
    br @loop, %pc1, %sp

@push(%pc %sp):
    %pc1 = add %pc, const 1
    %off1 = shl %pc1, 3
    %addr1 = add %code, %off1
    %val = loadl %addr1
    %sp8 = shl %sp, 3
    %saddr = add %stack, %sp8
    storel %val, %saddr
    %sp1 = add %sp, const 1
    %pc2 = add %pc, const 2
    br @loop, %pc2, %sp1

@add(%pc %sp):
    %spb = sub %sp, const 1
    %offb = shl %spb, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    %spa = sub %spb, const 1
    %offa = shl %spa, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    %res = add %a, %b
    storel %res, %aaddr
    %spn = add %spa, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@sub(%pc %sp):
    %spb = sub %sp, const 1
    %offb = shl %spb, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    %spa = sub %spb, const 1
    %offa = shl %spa, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    %res = sub %a, %b
    storel %res, %aaddr
    %spn = add %spa, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@mul(%pc %sp):
    %spb = sub %sp, const 1
    %offb = shl %spb, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    %spa = sub %spb, const 1
    %offa = shl %spa, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    %res = imul %a, %b
    storel %res, %aaddr
    %spn = add %spa, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@dup(%pc %sp):
    %sp1 = sub %sp, const 1
    %offt = shl %sp1, 3
    %taddr = add %stack, %offt
    %t = loadl %taddr
    %sp8 = shl %sp, 3
    %daddr = add %stack, %sp8
    storel %t, %daddr
    %spn = add %sp, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@swap(%pc %sp):
    %sp1 = sub %sp, const 1
    %offb = shl %sp1, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    %sp2 = sub %sp, const 2
    %offa = shl %sp2, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    storel %a, %baddr
    storel %b, %aaddr
    %pc1 = add %pc, const 1
    br @loop, %pc1, %sp

@drop(%pc %sp):
    %spn = sub %sp, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@over(%pc %sp):
    %sp2 = sub %sp, const 2
    %offv = shl %sp2, 3
    %vaddr = add %stack, %offv
    %v = loadl %vaddr
    %sp8 = shl %sp, 3
    %taddr = add %stack, %sp8
    storel %v, %taddr
    %spn = add %sp, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@lt(%pc %sp):
    %spb = sub %sp, const 1
    %offb = shl %spb, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    %spa = sub %spb, const 1
    %offa = shl %spa, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    %cmp = csltl %a, %b
    %one = const 1
    %zero = const 0
    %res = sel %cmp, %one, %zero
    storel %res, %aaddr
    %spn = add %spa, const 1
    %pc1 = add %pc, const 1
    br @loop, %pc1, %spn

@jz(%pc %sp):
    %pc1 = add %pc, const 1
    %offt = shl %pc1, 3
    %taddr = add %code, %offt
    %t = loadl %taddr
    %spv = sub %sp, const 1
    %offv = shl %spv, 3
    %vaddr = add %stack, %offv
    %v = loadl %vaddr
    %cond = ceql %v, const 0
    %pc2 = add %pc, const 2
    %pcn = sel %cond, %t, %pc2
    br @loop, %pcn, %spv

@jmp(%pc %sp):
    %pc1 = add %pc, const 1
    %offt = shl %pc1, 3
    %taddr = add %code, %offt
    %t = loadl %taddr
    br @loop, %t, %sp

@exit(%sp):
    %empty = ceql %sp, const 0
    jnz %empty @retval_zero, %sp
    %sp1 = sub %sp, const 1
    %offt = shl %sp1, 3
    %taddr = add %stack, %offt
    %top = loadl %taddr
    br @zero_loop, const 0, %cap, %stack, %top

@retval_zero(%sp):
    br @zero_loop, const 0, %cap, %stack, const 0

@zero_loop(%i %c %stk %retval):
    %cond = csltl %i, %c
    jnz %cond @zero_body, %i, %c, %stk, %retval
    br @ret, %retval

@zero_body(%i %c %stk %retval):
    %off = shl %i, 3
    %addr = add %stk, %off
    storel const 0, %addr
    %i1 = add %i, const 1
    br @zero_loop, %i1, %c, %stk, %retval

@ret(%retval):
    ret %retval
}
