export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry:
    %pc0 = const 0
    %sp0 = const 0
    %zero = const 0
    br @loop, %pc0, %sp0

@loop(%pc, %sp):
    ; if pc >= len -> exit
    %pc_lt_len = csltl %pc, %len
    jnz %pc_lt_len @fetch
    br @exit, %sp

@fetch:
    %pc8 = shl %pc, 3
    %caddr = add %code, %pc8
    %op = loadl %caddr

    ; opcode dispatch
    %is_halt = ceql %op, const 0
    jnz %is_halt @exit, %sp

    %is_push = ceql %op, const 1
    jnz %is_push @push, %pc, %sp

    %is_add = ceql %op, const 2
    jnz %is_add @binop, %pc, %sp, const 2

    %is_sub = ceql %op, const 3
    jnz %is_sub @binop, %pc, %sp, const 3

    %is_mul = ceql %op, const 4
    jnz %is_mul @binop, %pc, %sp, const 4

    %is_dup = ceql %op, const 5
    jnz %is_dup @dup, %pc, %sp

    %is_swap = ceql %op, const 6
    jnz %is_swap @swap, %pc, %sp

    %is_drop = ceql %op, const 7
    jnz %is_drop @drop, %pc, %sp

    %is_over = ceql %op, const 8
    jnz %is_over @over, %pc, %sp

    %is_lt = ceql %op, const 9
    jnz %is_lt @lt, %pc, %sp

    %is_jz = ceql %op, const 10
    jnz %is_jz @jz, %pc, %sp

    %is_jmp = ceql %op, const 11
    jnz %is_jmp @jmp, %pc, %sp

    ; default: pc += 1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp

@push(%pc, %sp):
    ; load immediate
    %pc1 = add %pc, const 1
    %off1 = shl %pc1, 3
    %addr1 = add %code, %off1
    %val = loadl %addr1
    ; store to stack[sp]
    %sp8 = shl %sp, 3
    %saddr = add %stack, %sp8
    storel %val, %saddr
    ; sp++, pc+=2
    %sp1 = add %sp, const 1
    %pc2 = add %pc, const 2
    br @loop, %pc2, %sp1

@binop(%pc, %sp, %kind):
    ; pop b
    %sp1 = sub %sp, const 1
    %offb = shl %sp1, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    ; pop a
    %sp2 = sub %sp1, const 1
    %offa = shl %sp2, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    ; compute result
    %cond_add = ceql %kind, const 2
    %cond_sub = ceql %kind, const 3
    %cond_mul = ceql %kind, const 4
    %tmp_add = add %a, %b
    %tmp_sub = sub %a, %b
    %tmp_mul = imul %a, %b
    %res1 = sel %cond_add, %tmp_add, %zero
    %res2 = sel %cond_sub, %tmp_sub, %res1
    %res = sel %cond_mul, %tmp_mul, %res2
    ; store result at a's slot (index %sp2)
    storel %res, %aaddr
    ; sp = sp2 + 1
    %sp_new = add %sp2, const 1
    ; pc += 1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp_new

@dup(%pc, %sp):
    ; load top
    %sp1 = sub %sp, const 1
    %offt = shl %sp1, 3
    %taddr = add %stack, %offt
    %t = loadl %taddr
    ; store duplicate
    %sp8 = shl %sp, 3
    %daddr = add %stack, %sp8
    storel %t, %daddr
    ; sp++, pc+=1
    %sp1 = add %sp, const 1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp1

@swap(%pc, %sp):
    ; a = stack[sp-2], b = stack[sp-1]
    %sp2 = sub %sp, const 2
    %offa = shl %sp2, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    %sp1 = sub %sp, const 1
    %offb = shl %sp1, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    ; swap
    storel %a, %baddr
    storel %b, %aaddr
    ; pc+=1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp

@drop(%pc, %sp):
    ; sp--
    %sp1 = sub %sp, const 1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp1

@over(%pc, %sp):
    ; val = stack[sp-2]
    %sp2 = sub %sp, const 2
    %offv = shl %sp2, 3
    %vaddr = add %stack, %offv
    %v = loadl %vaddr
    ; store at top
    %sp8 = shl %sp, 3
    %taddr = add %stack, %sp8
    storel %v, %taddr
    ; sp++, pc+=1
    %sp1 = add %sp, const 1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp1

@lt(%pc, %sp):
    ; pop b
    %sp1 = sub %sp, const 1
    %offb = shl %sp1, 3
    %baddr = add %stack, %offb
    %b = loadl %baddr
    ; pop a
    %sp2 = sub %sp1, const 1
    %offa = shl %sp2, 3
    %aaddr = add %stack, %offa
    %a = loadl %aaddr
    ; compute (a < b) signed
    %cmp = csltl %a, %b
    %one = const 1
    %zero = const 0
    %res = sel %cmp, %one, %zero
    ; store result at a's slot
    storel %res, %aaddr
    ; sp = sp2 + 1
    %sp_new = add %sp2, const 1
    ; pc+=1
    %pc_next = add %pc, const 1
    br @loop, %pc_next, %sp_new

@jz(%pc, %sp):
    ; load target t
    %pc1 = add %pc, const 1
    %offt = shl %pc1, 3
    %taddr = add %code, %offt
    %t = loadl %taddr
    ; pop v
    %sp1 = sub %sp, const 1
    %offv = shl %sp1, 3
    %vaddr = add %stack, %offv
    %v = loadl %vaddr
    ; test v==0
    %cond = ceql %v, const 0
    %pc_two = add %pc, const 2
    %pc_next = sel %cond, %t, %pc_two
    br @loop, %pc_next, %sp1

@jmp(%pc, %sp):
    ; load target
    %pc1 = add %pc, const 1
    %offt = shl %pc1, 3
    %taddr = add %code, %offt
    %t = loadl %taddr
    br @loop, %t, %sp

@exit(%sp):
    ; determine return value
    %sp_eq0 = ceql %sp, const 0
    jnz %sp_eq0 @zero_ret, %sp

    ; non-empty: load top
    %sp1 = sub %sp, const 1
    %offt = shl %sp1, 3
    %taddr = add %stack, %offt
    %top = loadl %taddr
    br @zero_fill, %sp, %cap, %stack, %top

@zero_ret:
    ; empty stack: return 0 after zeroing whole buffer
    br @zero_fill, const 0, %cap, %stack, const 0

@zero_fill(%i, %c, %stk, %retval):
    %cond = csltl %i, %c
    jnz %cond @zero_body, %i, %c, %stk, %retval
    br @ret, %retval

@zero_body(%i, %c, %stk, %retval):
    %off = shl %i, 3
    %addr = add %stk, %off
    storel const 0, %addr
    %i1 = add %i, const 1
    br @zero_fill, %i1, %c, %stk, %retval

@ret(%retval):
    ret %retval
}
