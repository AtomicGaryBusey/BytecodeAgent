export function l $fib_rec(l %n) {
@entry:
    %c0 =w ceql %n, $0
    jnz %c0, @zero
    %c1 =w ceql %n, $1
    jnz %c1, @one
    %n1 =l sub %n, $1
    %r1 =l call $fib_rec(l %n1)
    %n2 =l sub %n, $2
    %r2 =l call $fib_rec(l %n2)
    %sum =l add %r1, %r2
    ret %sum
@zero:
    ret $0
@one:
    ret $1
}
