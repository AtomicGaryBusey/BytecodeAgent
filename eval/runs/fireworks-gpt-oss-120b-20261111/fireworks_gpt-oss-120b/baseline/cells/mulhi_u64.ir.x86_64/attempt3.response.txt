export function l $mulhi_u64(l a, l b)
@entry:
  %a_lo = and a, 0xffffffff
  %a_hi = shr a, 32
  %b_lo = and b, 0xffffffff
  %b_hi = shr b, 32

  %p0 = mul %a_lo, %b_lo
  %p1 = mul %a_lo, %b_hi
  %p2 = mul %a_hi, %b_lo
  %p3 = mul %a_hi, %b_hi

  %t = shr %p0, 32

  %p1_lo = and %p1, 0xffffffff
  %p2_lo = and %p2, 0xffffffff

  %w0 = add %p1_lo, %p2_lo
  %w1 = add %w0, %t

  %carry = shr %w1, 32

  %p1_hi = shr %p1, 32
  %p2_hi = shr %p2, 32

  %h0 = add %p3, %p1_hi
  %h1 = add %h0, %p2_hi
  %high = add %h1, %carry

  ret %high
