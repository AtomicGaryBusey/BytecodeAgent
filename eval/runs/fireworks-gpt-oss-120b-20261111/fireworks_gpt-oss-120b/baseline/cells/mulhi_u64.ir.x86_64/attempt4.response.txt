export function l $mulhi_u64(l a, l b)
@entry:
  %a_lo = and a, 0xffffffffl
  %a_hi = shr a, 32l
  %b_lo = and b, 0xffffffffl
  %b_hi = shr b, 32l

  %p0 = mul %a_lo, %b_lo
  %p1 = mul %a_lo, %b_hi
  %p2 = mul %a_hi, %b_lo
  %p3 = mul %a_hi, %b_hi

  %t = shr %p0, 32l

  %p1_lo = and %p1, 0xffffffffl
  %p2_lo = and %p2, 0xffffffffl

  %w0 = add %p1_lo, %p2_lo
  %w1 = add %w0, %t

  %carry = shr %w1, 32l

  %p1_hi = shr %p1, 32l
  %p2_hi = shr %p2, 32l

  %h0 = add %p3, %p1_hi
  %h1 = add %h0, %p2_hi
  %high = add %h1, %carry

  ret %high
