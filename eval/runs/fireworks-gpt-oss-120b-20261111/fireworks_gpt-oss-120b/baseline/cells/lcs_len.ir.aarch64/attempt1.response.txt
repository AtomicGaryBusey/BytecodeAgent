export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
    %n1 = add %n, 1

    // zero row[0..n]
    @zrow_start:
        %jz = phi l (0, %zrow_start) (%jz_next, %zrow_body)
        %condz = csltl %jz, %n1
        jnz %condz, %zrow_body, %zrow_done

    @zrow_body:
        %offz = shl %jz, 3
        %ptrz = add %row, %offz
        storel %ptrz, 0
        %jz_next = add %jz, 1
        jmp %zrow_start

    @zrow_done:
    // outer loop over i
    @outer_start:
        %i = phi l (0, %outer_start) (%i_next, %outer_body)
        %condi = csltl %i, %m
        jnz %condi, %outer_body, %outer_done

    @outer_body:
        %xi_ptr = add %x, %i
        %xb =w loadub %xi_ptr
        // start inner loop
        jmp %inner_loop

    // inner loop over j
    @inner_loop:
        %j = phi l (1, %outer_body) (%j_next, %inner_loop)
        %diag = phi l (0, %outer_body) (%up, %inner_loop)

        %offj = shl %j, 3
        %ptr = add %row, %offj
        %up = loadl %ptr

        %jm1 = sub %j, 1
        %yptr = add %y, %jm1
        %yb =w loadub %yptr

        %eq = ceql %xb, %yb
        jnz %eq, %eq_true, %eq_false

    @eq_true:
        %val = add %diag, 1
        jmp %after_eq

    @eq_false:
        %offjm1 = shl %jm1, 3
        %ptrjm1 = add %row, %offjm1
        %left = loadl %ptrjm1
        %cmp = csltl %left, %up      // left < up -> up is larger
        jnz %cmp, %use_up, %use_left

    @use_up:
        %val = %up
        jmp %after_eq

    @use_left:
        %val = %left
        jmp %after_eq

    @after_eq:
        storel %ptr, %val
        %j_next = add %j, 1
        %condj = csltl %j_next, %n1
        jnz %condj, %inner_loop, %inner_done

    @inner_done:
        %i_next = add %i, 1
        jmp %outer_start

    @outer_done:
        %offn = shl %n, 3
        %ptrn = add %row, %offn
        %res = loadl %ptrn
        ret %res
}
