export function lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    %n1 = add %n, 1
    jmp init_row

@init_row
    %j = phi l (0, entry) (%j_next, init_row_body)
    %cond = csltl %j, %n1
    jnz %cond, init_row_body, init_done

@init_row_body
    %off = shl %j, 3
    %ptr = add %row, %off
    storel 0, %ptr
    %j_next = add %j, 1
    jmp init_row

@init_done
    jmp outer

@outer
    %i = phi l (0, init_done) (%i_next, outer_body)
    %condi = csltl %i, %m
    jnz %condi, outer_body, outer_done

@outer_body
    %xi_ptr = add %x, %i
    %xb =w loadub %xi_ptr
    jmp inner

@inner
    %j = phi l (1, outer_body) (%j_next, inner_body)
    %diag = phi l (0, outer_body) (%up_next, inner_body)
    %offj = shl %j, 3
    %ptrj = add %row, %offj
    %up = loadl %ptrj
    %jm1 = sub %j, 1
    %yptr = add %y, %jm1
    %yb =w loadub %yptr
    %eq = ceql %xb, %yb
    jnz %eq, eq_true, eq_false

@eq_true
    %val = add %diag, 1
    jmp after_eq

@eq_false
    %offjm1 = shl %jm1, 3
    %ptrjm1 = add %row, %offjm1
    %left = loadl %ptrjm1
    %cmp = csltl %left, %up
    jnz %cmp, use_up, use_left

@use_up
    %val = %up
    jmp after_eq

@use_left
    %val = %left
    jmp after_eq

@after_eq
    storel %val, %ptrj
    %up_next = %up
    %j_next = add %j, 1
    %condj = csltl %j_next, %n1
    jnz %condj, inner_body, inner_done

@inner_body
    jmp inner

@inner_done
    %i_next = add %i, 1
    jmp outer

@outer_done
    %offn = shl %n, 3
    %ptrn = add %row, %offn
    %res = loadl %ptrn
    ret %res
}
