export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
    %n1 = add %n, 1
    @zrow_start:
        %jz = phi l (0, %zrow_start) (%jz_next, %zrow_body)
        %condz = csltl %jz, %n1
        jnz %condz, %zrow_body, %zrow_done
    @zrow_body:
        %offz = shl %jz, 3
        %ptrz = add %row, %offz
        storel %ptrz, 0
        %jz_next = add %jz, 1
        jmp %zrow_start
    @zrow_done:
        jmp %outer_start
    @outer_start:
        %i = phi l (0, %outer_start) (%i_next, %inner_done)
        %condi = csltl %i, %m
        jnz %condi, %outer_body, %outer_done
    @outer_body:
        %xi_ptr = add %x, %i
        %xb =w loadub %xi_ptr
        jmp %inner_start
    @inner_start:
        %j = phi l (1, %outer_body) (%j_next, %after_eq)
        %diag = phi l (0, %outer_body) (%up_val, %after_eq)
        %offj = shl %j, 3
        %ptr = add %row, %offj
        %up_val = loadl %ptr
        %jm1 = sub %j, 1
        %yptr = add %y, %jm1
        %yb =w loadub %yptr
        %eq = ceql %xb, %yb
        jnz %eq, %eq_true, %eq_false
    @eq_true:
        %val = add %diag, 1
        jmp %after_eq
    @eq_false:
        %offjm1 = shl %jm1, 3
        %ptrjm1 = add %row, %offjm1
        %left = loadl %ptrjm1
        %cmp = csltl %left, %up_val
        jnz %cmp, %use_up, %use_left
    @use_up:
        %val = %up_val
        jmp %after_eq
    @use_left:
        %val = %left
        jmp %after_eq
    @after_eq:
        storel %ptr, %val
        %j_next = add %j, 1
        %condj = csltl %j_next, %n1
        jnz %condj, %inner_start, %inner_done
    @inner_done:
        %i_next = add %i, 1
        jmp %outer_start
    @outer_done:
        %offn = shl %n, 3
        %ptrn = add %row, %offn
        %res = loadl %ptrn
        ret %res
}
