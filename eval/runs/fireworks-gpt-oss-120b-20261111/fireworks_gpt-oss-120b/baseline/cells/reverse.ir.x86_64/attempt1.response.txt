export function l $reverse(l %a, l %n) {
    @entry:
        %two = constl 2
        %half = divl %n, %two
        %i0 = constl 0
        br %loop

    @loop:
        %i = phi l (%i0, %entry), (%i_next, %loop)
        %cond = csltl %i, %half
        jnz %cond, %body, %exit

    @body:
        %idx1 = mul %i, constl 8
        %addr1 = add %a, %idx1
        %val1 = loadl %addr1

        %n_minus_one = sub %n, constl 1
        %j = sub %n_minus_one, %i
        %idx2 = mul %j, constl 8
        %addr2 = add %a, %idx2
        %val2 = loadl %addr2

        storel %val2, %addr1
        storel %val1, %addr2

        %i_next = add %i, constl 1
        br %loop

    @exit:
        ret constl 0
}
