export function l $reverse(l %a, l %n) {
@entry:
    %c0 = constl 0
    %c1 = constl 1
    %c2 = constl 2
    %c8 = constl 8
    %half = divl %n, %c2
    br %loop

@loop:
    %i = phi l (%c0, %entry), (%i_next, %loop)
    %cond = csltl %i, %half
    jnz %cond, %body, %exit

@body:
    %idx1 = mul %i, %c8
    %addr1 = add %a, %idx1
    %val1 = loadl %addr1

    %n1 = sub %n, %c1
    %j = sub %n1, %i
    %idx2 = mul %j, %c8
    %addr2 = add %a, %idx2
    %val2 = loadl %addr2

    storel %val2, %addr1
    storel %val1, %addr2

    %i_next = add %i, %c1
    br %loop

@exit:
    ret %c0
}
