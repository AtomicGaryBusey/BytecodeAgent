export function l $bmax(l a, l n) {
@entry
    %i0 = constl 0
    %first = loadl a
    %max0 = copyl %first
    %i1 = addl %i0, constl 1
    %cond0 = csltl %i1, n
    jnz %cond0 @loop
    ret %max0
@loop
    %off = mul %i1, constl 8
    %addr = addl a, %off
    %val = loadl %addr
    %cmp = csltl %max0, %val
    %newmax = sel %cmp %val %max0
    %max0 = copyl %newmax
    %i1 = addl %i1, constl 1
    %cond = csltl %i1, n
    jnz %cond @loop
    ret %max0
}
