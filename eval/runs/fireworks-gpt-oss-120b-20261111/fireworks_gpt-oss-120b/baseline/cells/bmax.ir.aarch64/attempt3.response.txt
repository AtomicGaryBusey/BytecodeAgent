export function l bmax(l a, l n) {
@entry
    %i = constl 1
    %first = loadl %a
    %max = copyl %first
    jmp @check
@check
    %cond = csltl %i, %n
    jnz %cond @loop
    jmp @exit
@loop
    %off = mull %i, constl 8
    %addr = addl %a, %off
    %val = loadl %addr
    %cmp = csltl %max, %val
    %max = sel %cmp %val %max
    %i = addl %i, constl 1
    jmp @check
@exit
    ret %max
}
