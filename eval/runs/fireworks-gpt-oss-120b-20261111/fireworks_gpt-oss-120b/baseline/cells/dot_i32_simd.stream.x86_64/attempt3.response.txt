dot_i32_simd:
    xor rax, rax
    mov rcx, rdx
    mov r8, rcx
    shr r8, 2
    mov r9, rcx
    and r9, 3
    pxor xmm0, xmm0
    test r8, r8
    jz .Lscalar_tail
.Lvec_loop:
    movdqu xmm2, [rdi]          # load 4 a values
    movdqu xmm1, [rsi]          # load 4 b values
    # low two elements
    pmovsxdq xmm3, xmm2         # sign-extend low 2 a's to 64-bit
    pmovsxdq xmm4, xmm1         # sign-extend low 2 b's to 64-bit
    pmuludq xmm3, xmm4          # multiply low 2 pairs
    # high two elements
    psrldq xmm2, 8
    psrldq xmm1, 8
    pmovsxdq xmm5, xmm2         # sign-extend high 2 a's
    pmovsxdq xmm6, xmm1         # sign-extend high 2 b's
    pmuludq xmm5, xmm6          # multiply high 2 pairs
    # accumulate
    paddq xmm0, xmm3
    paddq xmm0, xmm5
    add rdi, 16
    add rsi, 16
    dec r8
    jne .Lvec_loop
    movq rdx, xmm0
    pextrq rcx, xmm0, 1
    add rdx, rcx
    add rax, rdx
.Lscalar_tail:
    test r9, r9
    jz .Ldone
.Lrem_loop:
    movsxd rcx, dword ptr [rdi]
    movsxd rdx, dword ptr [rsi]
    imul rcx, rdx
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec r9
    jne .Lrem_loop
.Ldone:
    ret
