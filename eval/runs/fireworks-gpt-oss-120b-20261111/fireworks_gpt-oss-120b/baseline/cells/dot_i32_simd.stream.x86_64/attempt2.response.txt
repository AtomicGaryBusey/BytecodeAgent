dot_i32_simd:
    xor rax, rax
    mov rcx, rdx
    mov r8, rcx
    shr r8, 2
    mov r9, rcx
    and r9, 3
    test r8, r8
    jz .Lscalar_tail
    pxor xmm0, xmm0
.Lvec_loop:
    movdqu xmm1, [rdi]
    movdqu xmm2, [rsi]
    movdqa xmm3, xmm1
    movdqa xmm4, xmm2
    pmuludq xmm3, xmm4
    psrldq xmm1, 4
    psrldq xmm2, 4
    pmuludq xmm1, xmm2
    paddq xmm0, xmm3
    paddq xmm0, xmm1
    add rdi, 16
    add rsi, 16
    dec r8
    jne .Lvec_loop
    movq rdx, xmm0
    pextrq rdx, xmm0, 1
    add rax, rdx
.Lscalar_tail:
    test r9, r9
    jz .Ldone
.Lrem_loop:
    movsxd rdx, dword ptr [rdi]
    movsxd rcx, dword ptr [rsi]
    imul rdx, rcx
    add rax, rdx
    add rdi, 4
    add rsi, 4
    dec r9
    jne .Lrem_loop
.Ldone:
    ret
