dot_i32_simd:
    xor rax, rax
    mov rcx, rdx
    mov r8, rcx
    shr r8, 2
    and rcx, 3
    pxor xmm0, xmm0
    test r8, r8
    jz .Lscalar
.Lvec_loop:
    movdqu xmm1, [rdi]
    movdqu xmm2, [rsi]
    pmovsxdq xmm3, xmm1
    pmovsxdq xmm4, xmm2
    pmuludq xmm3, xmm4
    paddq xmm0, xmm3
    psrldq xmm1, 8
    psrldq xmm2, 8
    pmovsxdq xmm5, xmm1
    pmovsxdq xmm6, xmm2
    pmuludq xmm5, xmm6
    paddq xmm0, xmm5
    add rdi, 16
    add rsi, 16
    dec r8
    jne .Lvec_loop
.Lscalar:
    test rcx, rcx
    jz .Ldone
.Lrem_loop:
    movsxd rdx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rdx, r8
    add rax, rdx
    add rdi, 4
    add rsi, 4
    dec rcx
    jne .Lrem_loop
.Ldone:
    mov rdx, xmm0
