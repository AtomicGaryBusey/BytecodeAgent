dot_i32_simd:
    xor rax, rax
    mov rcx, rdx
    mov r9, rcx
    shr r9, 2
    jz .Lscalar
.Lvec_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pextrd edx, xmm0, 0
    movsxd r10, edx
    pextrd edx, xmm1, 0
    imul r10, rdx
    add rax, r10
    pextrd edx, xmm0, 1
    movsxd r10, edx
    pextrd edx, xmm1, 1
    imul r10, rdx
    add rax, r10
    pextrd edx, xmm0, 2
    movsxd r10, edx
    pextrd edx, xmm1, 2
    imul r10, rdx
    add rax, r10
    pextrd edx, xmm0, 3
    movsxd r10, edx
    pextrd edx, xmm1, 3
    imul r10, rdx
    add rax, r10
    add rdi, 16
    add rsi, 16
    dec r9
    jne .Lvec_loop
.Lscalar:
    and rcx, 3
    jz .Ldone
.Lrem_loop:
    movsxd r10, dword ptr [rdi]
    movsxd r11, dword ptr [rsi]
    imul r10, r11
    add rax, r10
    add rdi, 4
    add rsi, 4
    dec rcx
    jne .Lrem_loop
.Ldone:
    ret
