export function l $map_sum(l %a, l %n, l %f) {
@entry:
  %i0 =l imm 0
  %sum0 =l imm 0
  jmp @loop

@loop:
  %i =l phi %i0 %i_next
  %sum =l phi %sum0 %sum_next
  %cond =w csltl %i, %n
  jnz %cond, @body
  ret %sum

@body:
  %offset =l mul %i, 8
  %ptr =l add %a, %offset
  %val =l loadl %ptr
  %res =l call %f(l %val)
  %sum_next =l add %sum, %res
  %i_next =l add %i, 1
  jmp @loop
}
