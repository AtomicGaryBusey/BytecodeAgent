packbits_encode:
    // x0=src, x1=n, x2=out, x3=cap
    mov     x4, #0                 // i = 0
    mov     x5, #0                 // outpos = 0
.Lmain:
    cmp     x4, x1
    b.ge    .Ldone
    // ---- compute run at i ----
    add     x6, x0, x4             // src+i
    ldrb    w7, [x6]               // first byte
    mov     x8, #1                 // run = 1
    add     x9, x6, #1             // ptr = src+i+1
    sub     x10, x1, x4            // remaining = n-i
    sub     x10, x10, #1           // remaining after first byte
.Lrun_check:
    cmp     x8, #128
    b.ge    .Lrun_done
    cbz     x10, .Lrun_done
    ldrb    w11, [x9]
    cmp     w11, w7
    b.ne    .Lrun_done
    add     x8, x8, #1
    add     x9, x9, #1
    sub     x10, x10, #1
    b       .Lrun_check
.Lrun_done:
    // if run >= 3 -> repeat packet
    cmp     x8, #3
    b.lt    .Lliteral_start
    // repeat packet needs 2 bytes
    add     x12, x5, #2
    cmp     x12, x3
    b.gt    .Lreturn_minus_one
    // header = 257 - run
    mov     w13, #257
    sub     w13, w13, w8
    add     x14, x2, x5            // out+outpos
    strb    w13, [x14]             // store header
    add     x14, x14, #1
    strb    w7, [x14]              // store repeated byte
    add     x5, x5, #2
    add     x4, x4, x8
    b       .Lmain
.Lliteral_start:
    // literal_len = 0
    mov     x15, #0
    // literal_start = i (in x4)
.Lliteral_loop:
    // idx = i + literal_len
    add     x16, x4, x15
    cmp     x16, x1
    b.ge    .Lliteral_done
    cmp     x15, #128
    b.ge    .Lliteral_done
    // compute run at idx
    add     x6, x0, x16
    ldrb    w7, [x6]
    mov     x8, #1
    add     x9, x6, #1
    sub     x10, x1, x16
    sub     x10, x10, #1
.Lrun2_check:
    cmp     x8, #128
    b.ge    .Lrun2_done
    cbz     x10, .Lrun2_done
    ldrb    w11, [x9]
    cmp     w11, w7
    b.ne    .Lrun2_done
    add     x8, x8, #1
    add     x9, x9, #1
    sub     x10, x10, #1
    b       .Lrun2_check
.Lrun2_done:
    cmp     x8, #3
    b.ge    .Lliteral_done
    // safe to include this byte in literal
    add     x15, x15, #1
    b       .Lliteral_loop
.Lliteral_done:
    // literal_len in x15 (>=1)
    add     x12, x5, #1
    add     x12, x12, x15          // needed = outpos +1+len
    cmp     x12, x3
    b.gt    .Lreturn_minus_one
    // store header = literal_len-1
    sub     w13, w15, #1
    add     x14, x2, x5
    strb    w13, [x14]
    // copy literal bytes
    add     x6, x0, x4             // src + i
    add     x14, x2, x5
    add     x14, x14, #1           // dest = out+outpos+1
    mov     x17, #0
.Lcopy_loop:
    cmp     x17, x15
    b.ge    .Lcopy_done
    ldrb    w18, [x6, x17]
    strb    w18, [x14, x17]
    add     x17, x17, #1
    b       .Lcopy_loop
.Lcopy_done:
    add     x5, x5, #1
    add     x5, x5, x15            // outpos += 1+len
    add     x4, x4, x15            // i += len
    b       .Lmain
.Lreturn_minus_one:
    mov     x0, #-1
    ret
.Ldone:
    mov     x0, x5
    ret
