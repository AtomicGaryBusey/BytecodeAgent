packbits_encode:
    mov     x4, #0                     // i = 0
    mov     x5, #0                     // outpos = 0
    mov     x6, x0                     // src base
    mov     x7, x2                     // out base
    mov     x8, x1                     // n
    mov     x9, x3                     // cap
.Lmain:
    cmp     x4, x8
    b.ge    .Ldone
    // compute run length at i
    add     x10, x6, x4                // ptr = src + i
    ldrb    w11, [x10]                 // first byte
    mov     x12, #1                    // count = 1
    add     x13, x10, #1               // next ptr
    sub     x14, x8, x4                // remaining = n - i
.Lrun_loop:
    cmp     x12, #128
    b.ge    .Lrun_done
    cmp     x12, x14
    b.ge    .Lrun_done
    ldrb    w15, [x13]
    cmp     w15, w11
    b.ne    .Lrun_done
    add     x12, x12, #1
    add     x13, x13, #1
    b       .Lrun_loop
.Lrun_done:
    cmp     x12, #3
    b.lt    .Lliteral_start
    // repeat packet
    add     x15, x5, #2
    cmp     x15, x9
    b.gt    .Lreturn_minus_one
    // header = 257 - run
    mov     w0, #257
    sub     w0, w0, w12
    add     x0, x7, x5                 // dest = out + outpos
    strb    w0, [x0]                   // store header
    add     x0, x0, #1
    strb    w11, [x0]                  // store repeat byte
    add     x5, x5, #2                 // outpos += 2
    add     x4, x4, x12                // i += run
    b       .Lmain
.Lliteral_start:
    mov     x17, #0                    // literal_len = 0
.Lliteral_loop:
    add     x18, x4, x17               // idx = i + literal_len
    cmp     x18, x8
    b.ge    .Lliteral_done
    cmp     x17, #128
    b.ge    .Lliteral_done
    // compute run at idx
    add     x10, x6, x18
    ldrb    w11, [x10]
    mov     x12, #1
    add     x13, x10, #1
    sub     x14, x8, x18
.Lrun2_loop:
    cmp     x12, #128
    b.ge    .Lrun2_done
    cmp     x12, x14
    b.ge    .Lrun2_done
    ldrb    w15, [x13]
    cmp     w15, w11
    b.ne    .Lrun2_done
    add     x12, x12, #1
    add     x13, x13, #1
    b       .Lrun2_loop
.Lrun2_done:
    cmp     x12, #3
    b.ge    .Lliteral_done
    add     x17, x17, #1
    b       .Lliteral_loop
.Lliteral_done:
    // ensure we have at least one literal byte
    // (the loop guarantees literal_len >= 1)
    add     x15, x5, #1
    add     x15, x15, x17
    cmp     x15, x9
    b.gt    .Lreturn_minus_one
    // header = literal_len - 1
    sub     w0, w17, #1
    add     x0, x7, x5                 // dest = out + outpos
    strb    w0, [x0]                   // store header
    // copy literal bytes
    add     x10, x6, x4                // src_ptr = src + i
    add     x11, x7, x5
    add     x11, x11, #1               // dest_ptr = out + outpos + 1
    mov     x12, #0
.Lcopy_loop:
    cmp     x12, x17
    b.ge    .Lcopy_done
    ldrb    w0, [x10, x12]
    strb    w0, [x11, x12]
    add     x12, x12, #1
    b       .Lcopy_loop
.Lcopy_done:
    add     x5, x5, #1
    add     x5, x5, x17                // outpos += 1 + literal_len
    add     x4, x4, x17                // i += literal_len
    b       .Lmain
.Lreturn_minus_one:
    mov     x0, #-1
    ret
.Ldone:
    mov     x0, x5
    ret
