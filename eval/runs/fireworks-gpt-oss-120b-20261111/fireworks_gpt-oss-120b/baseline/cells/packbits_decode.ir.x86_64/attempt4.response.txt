export function l $packbits_decode(l %src, l %n, l %out, l %cap)
{
    @entry
        %src_end = add %src, %n
        %out_end = add %out, %cap
        %written = const 0
        %src_ptr = mov %src
        %out_ptr = mov %out
        jmp @loop

    @loop
        %has_src = csltl %src_ptr, %src_end
        %no_src = ceql %has_src, 0
        jnz %no_src, @done

        %h = loadsb %src_ptr
        %src_ptr = add %src_ptr, 1

        %is_nop = ceql %h, const -128
        jnz %is_nop, @loop

        %is_neg = csltl %h, const 0
        jnz %is_neg, @repeat_case
        jmp @literal_case

    @literal_case
        %cnt = add %h, const 1          // bytes to copy

        %src_needed = add %src_ptr, %cnt
        %src_lt_end = csltl %src_needed, %src_end
        %src_eq_end = ceql %src_needed, %src_end
        %src_ok = or %src_lt_end, %src_eq_end
        %src_fail = ceql %src_ok, 0
        jnz %src_fail, @err_src

        %space_left = sub %out_end, %out_ptr
        %space_ok = csltl const 0, %space_left   // space_left > 0
        %space_fail = ceql %space_ok, 0
        jnz %space_fail, @err_out

        %space_lt_cnt = csltl %space_left, %cnt
        jnz %space_lt_cnt, @literal_partial

        // full copy
        %copy_src = mov %src_ptr
        %copy_dst = mov %out_ptr
        %rem = mov %cnt
        jmp @literal_copy_full

    @literal_copy_full
        %cond = csltl const 0, %rem
        jnz %cond, @literal_copy_full_body
        %src_ptr = mov %copy_src
        %out_ptr = mov %copy_dst
        %written = add %written, %cnt
        jmp @loop

    @literal_copy_full_body
        %b = loadub %copy_src
        storeb %b, %copy_dst
        %copy_src = add %copy_src, 1
        %copy_dst = add %copy_dst, 1
        %rem = sub %rem, 1
        jmp @literal_copy_full

    @literal_partial
        %copy_src = mov %src_ptr
        %copy_dst = mov %out_ptr
        %rem = mov %space_left
        jmp @literal_copy_partial

    @literal_copy_partial
        %cond = csltl const 0, %rem
        jnz %cond, @literal_copy_partial_body
        %src_ptr = add %src_ptr, %space_left
        %out_ptr = mov %out_end
        jmp @err_out

    @literal_copy_partial_body
        %b = loadub %copy_src
        storeb %b, %copy_dst
        %copy_src = add %copy_src, 1
        %copy_dst = add %copy_dst, 1
        %rem = sub %rem, 1
        jmp @literal_copy_partial

    @repeat_case
        %repeat = sub const 1, %h        // repeat count = 1 - h

        %src_has_val = csltl %src_ptr, %src_end
        %src_no_val = ceql %src_has_val, 0
        jnz %src_no_val, @err_src

        %val = loadub %src_ptr
        %src_ptr = add %src_ptr, 1

        %space_left = sub %out_end, %out_ptr
        %space_ok = csltl const 0, %space_left
        %space_fail = ceql %space_ok, 0
        jnz %space_fail, @err_out

        %space_lt_rep = csltl %space_left, %repeat
        jnz %space_lt_rep, @repeat_partial

        // full repeat
        %copy_dst = mov %out_ptr
        %rem = mov %repeat
        jmp @repeat_copy_full

    @repeat_copy_full
        %cond = csltl const 0, %rem
        jnz %cond, @repeat_copy_full_body
        %out_ptr = add %out_ptr, %repeat
        %written = add %written, %repeat
        jmp @loop

    @repeat_copy_full_body
        storeb %val, %copy_dst
        %copy_dst = add %copy_dst, 1
        %rem = sub %rem, 1
        jmp @repeat_copy_full

    @repeat_partial
        %copy_dst = mov %out_ptr
        %rem = mov %space_left
        jmp @repeat_copy_partial

    @repeat_copy_partial
        %cond = csltl const 0, %rem
        jnz %cond, @repeat_copy_partial_body
        %out_ptr = mov %out_end
        jmp @err_out

    @repeat_copy_partial_body
        storeb %val, %copy_dst
        %copy_dst = add %copy_dst, 1
        %rem = sub %rem, 1
        jmp @repeat_copy_partial

    @done
        ret %written

    @err_src
        %neg2 = const -2
        ret %neg2

    @err_out
        %neg1 = const -1
        ret %neg1
}
