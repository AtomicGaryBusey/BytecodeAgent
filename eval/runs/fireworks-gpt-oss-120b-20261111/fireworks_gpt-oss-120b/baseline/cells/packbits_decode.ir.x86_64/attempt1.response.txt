export function l $packbits_decode(l src, l n, l out, l cap)
{
    @entry
        %src_end =add src, n
        %out_end =add out, cap
        %written =const 0
        %src_ptr =mov src
        %out_ptr =mov out
        jmp %loop

    @loop
        %has_src =csltl %src_ptr, %src_end
        %no_src =ceql %has_src, 0
        jnz %no_src, %done

        %h =loadsb %src_ptr
        %src_ptr =add %src_ptr, 1

        %is_nop =ceql %h, -128
        jnz %is_nop, %loop

        %is_neg =csltl %h, 0
        %is_pos =ceql %is_neg, 0
        jnz %is_pos, %literal_case
        jmp %repeat_case

    @literal_case
        %cnt =add %h, 1           // bytes to copy

        // check enough source bytes
        %src_needed =add %src_ptr, %cnt
        %src_ok =csltl %src_needed, %src_end
        %src_eq =ceql %src_needed, %src_end
        %src_ok =or %src_ok, %src_eq
        %src_fail =ceql %src_ok, 0
        jnz %src_fail, %err_src

        // compute space left
        %space_left =sub %out_end, %out_ptr
        // if space_left <= 0 goto overflow_no_write
        %space_le0 =csltl %space_left, 1
        jnz %space_le0, %err_out

        // if space_left < cnt -> partial copy
        %space_lt_cnt =csltl %space_left, %cnt
        jnz %space_lt_cnt, %literal_partial

        // enough space: full copy
        %src_cur =mov %src_ptr
        %dst_cur =mov %out_ptr
        %rem =mov %cnt
        jmp %literal_copy_full

    @literal_copy_full
        %rem_nz =csltl 0, %rem
        %rem_eq =ceql %rem, 0
        %rem_done =or %rem_eq, %rem_nz
        jnz %rem_done, %literal_copy_full_body
        // rem == 0
        %src_ptr =mov %src_cur
        %out_ptr =mov %dst_cur
        %written =add %written, %cnt
        jmp %loop

    @literal_copy_full_body
        %b =loadub %src_cur
        storeb %b, %dst_cur
        %src_cur =add %src_cur, 1
        %dst_cur =add %dst_cur, 1
        %rem =sub %rem, 1
        jmp %literal_copy_full

    @literal_partial
        // write up to space_left bytes, then overflow
        %src_cur =mov %src_ptr
        %dst_cur =mov %out_ptr
        %rem =mov %space_left
        jmp %literal_copy_partial

    @literal_copy_partial
        %rem_nz =csltl 0, %rem
        %rem_eq =ceql %rem, 0
        %rem_done =or %rem_eq, %rem_nz
        jnz %rem_done, %literal_copy_partial_body
        // rem == 0
        %src_ptr =add %src_ptr, %space_left
        %out_ptr =mov %out_end
        // written already equals cap, ignore exact count
        jmp %err_out

    @literal_copy_partial_body
        %b =loadub %src_cur
        storeb %b, %dst_cur
        %src_cur =add %src_cur, 1
        %dst_cur =add %dst_cur, 1
        %rem =sub %rem, 1
        jmp %literal_copy_partial

    @repeat_case
        // repeat packet: repeat count = 1 - h
        %repeat =sub 1, %h

        // need one byte value
        %src_has_val =csltl %src_ptr, %src_end
        %src_no_val =ceql %src_has_val, 0
        jnz %src_no_val, %err_src

        %val =loadub %src_ptr
        %src_ptr =add %src_ptr, 1

        // compute space left
        %space_left =sub %out_end, %out_ptr
        %space_le0 =csltl %space_left, 1
        jnz %space_le0, %err_out

        // if space_left < repeat -> partial
        %space_lt_rep =csltl %space_left, %repeat
        jnz %space_lt_rep, %repeat_partial

        // enough space: full repeat
        %dst_cur =mov %out_ptr
        %rem =mov %repeat
        jmp %repeat_copy_full

    @repeat_copy_full
        %rem_nz =csltl 0, %rem
        %rem_eq =ceql %rem, 0
        %rem_done =or %rem_eq, %rem_nz
        jnz %rem_done, %repeat_copy_full_body
        // rem == 0
        %out_ptr =add %out_ptr, %repeat
        %written =add %written, %repeat
        jmp %loop

    @repeat_copy_full_body
        storeb %val, %dst_cur
        %dst_cur =add %dst_cur, 1
        %rem =sub %rem, 1
        jmp %repeat_copy_full

    @repeat_partial
        // write up to space_left bytes, then overflow
        %dst_cur =mov %out_ptr
        %rem =mov %space_left
        jmp %repeat_copy_partial

    @repeat_copy_partial
        %rem_nz =csltl 0, %rem
        %rem_eq =ceql %rem, 0
        %rem_done =or %rem_eq, %rem_nz
        jnz %rem_done, %repeat_copy_partial_body
        // rem == 0
        %out_ptr =mov %out_end
        jmp %err_out

    @repeat_copy_partial_body
        storeb %val, %dst_cur
        %dst_cur =add %dst_cur, 1
        %rem =sub %rem, 1
        jmp %repeat_copy_partial

    @done
        ret %written

    @err_src
        ret -2

    @err_out
        ret -1
}
