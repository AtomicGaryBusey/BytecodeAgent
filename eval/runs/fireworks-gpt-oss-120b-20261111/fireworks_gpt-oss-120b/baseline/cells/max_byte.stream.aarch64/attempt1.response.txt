max_byte:
    cbz x1, .Lzero
    ldrb w2, [x0]               // initial max = p[0]
    add x3, x0, #1              // x3 = p + 1
    sub x1, x1, #1              // remaining = n-1
.Lloop:
    cbz x1, .Ldone
    ldrb w4, [x3], #1           // load byte and increment pointer
    cmp w4, w2
    csel w2, w4, w2, hi         // if w4 > w2, update max
    sub x1, x1, #1
    b .Lloop
.Ldone:
    uxtw x0, w2                 // zero‑extend result
    ret
.Lzero:
    mov x0, #0
    ret
