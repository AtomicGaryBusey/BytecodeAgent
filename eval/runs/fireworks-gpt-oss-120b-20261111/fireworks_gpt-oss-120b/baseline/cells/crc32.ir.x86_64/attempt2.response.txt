export function l $crc32(l %p, l %n) {
@start
    %poly =w const 0xedb88320
    %xorout =w const 0xffffffff
    %crc =w const 0xffffffff
    %i =l const 0
@outer
    %c1 =w csltl %i, %n
    jnz %c1, @body
    jmp @end
@body
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %j =l const 0
@bit_loop
    %c2 =w csltl %j, 8
    jnz %c2, @bit_body
    jmp @after_bit
@bit_body
    %lsb =w and %crc, 1
    %zero =w ceql %lsb, 0
    jnz %zero, @shift_only
    %crc =w shr %crc, 1
    %crc =w xor %crc, %poly
    jmp @inc_j
@shift_only
    %crc =w shr %crc, 1
@inc_j
    %j =l add %j, 1
    jmp @bit_loop
@after_bit
    %i =l add %i, 1
    jmp @outer
@end
    %crc =w xor %crc, %xorout
    %res =l extuw %crc
    ret %res
}
