export function l $crc32(l %p, l %n) {
@start
    %poly =w 3988292384
    %xorout =w 4294967295
    %crc =w 4294967295
    %i =l 0
@outer_cond
    %c =w csltl %i, %n
    jnz %c, @outer_body
    jmp @after_outer
@outer_body
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %j =l 0
@bit_cond
    %c2 =w csltl %j, 8
    jnz %c2, @bit_body
    jmp @after_bit
@bit_body
    %lsb =w and %crc, 1
    jnz %lsb, @poly_branch
    %crc =w shr %crc, 1
    jmp @inc_j
@poly_branch
    %crc =w shr %crc, 1
    %crc =w xor %crc, %poly
@inc_j
    %j =l add %j, 1
    jmp @bit_cond
@after_bit
    %i =l add %i, 1
    jmp @outer_cond
@after_outer
    %crc =w xor %crc, %xorout
    %res =l extuw %crc
    ret %res
}
