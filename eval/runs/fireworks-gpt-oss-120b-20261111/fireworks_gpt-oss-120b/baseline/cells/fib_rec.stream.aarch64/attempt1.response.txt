fib_rec:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #16
    cmp x0, #1
    b.le .Lreturn_n
    str x0, [sp, #8]            // save original n
    sub x0, x0, #1              // n-1
    bl fib_rec
    str x0, [sp, #0]            // store fib(n-1)
    ldr x0, [sp, #8]            // reload original n
    sub x0, x0, #2              // n-2
    bl fib_rec
    ldr x1, [sp, #0]            // load fib(n-1)
    add x0, x0, x1              // fib(n-1)+fib(n-2)
    add sp, sp, #16
    ldp x29, x30, [sp], #16
    ret
.Lreturn_n:
    // n <= 1, result already in x0
    add sp, sp, #16
    ldp x29, x30, [sp], #16
    ret
