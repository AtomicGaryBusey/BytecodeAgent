export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
    %zero =l 0
    %one =l 1
    %mask32 =l 0xffffffff
    %total =l add %na, %nb

    %i =l 0
@zero_loop:
    %c_i =w csltl %i, %total
    jnz %c_i, @zero_body, @zero_done
@zero_body:
    %off_i =l shl %i, 3
    %addr_i =l add %r, %off_i
    storel %zero, %addr_i
    %i =l add %i, %one
    jmp @zero_loop
@zero_done:

    %c_na0 =w csltl %na, %one
    %c_nb0 =w csltl %nb, %one
    %any_zero =w or %c_na0, %c_nb0
    jnz %any_zero, @len_calc, @mul_start

@mul_start:
    %i =l 0
@i_loop:
    %c_i =w csltl %i, %na
    jnz %c_i, @i_body, @i_done
@i_body:
    %ai_off =l shl %i, 3
    %ai_addr =l add %a, %ai_off
    %ai =l loadl %ai_addr
    %carry =l 0
    %j =l 0
@j_loop:
    %c_j =w csltl %j, %nb
    jnz %c_j, @j_body, @j_end
@j_body:
    %bj_off =l shl %j, 3
    %bj_addr =l add %b, %bj_off
    %bj =l loadl %bj_addr

    %ai_low =l and %ai, %mask32
    %ai_high =l shr %ai, 32
    %bj_low =l and %bj, %mask32
    %bj_high =l shr %bj, 32

    %p0 =l mul %ai_low, %bj_low
    %p1 =l mul %ai_low, %bj_high
    %p2 =l mul %ai_high, %bj_low
    %p3 =l mul %ai_high, %bj_high

    %mid =l add %p1, %p2
    %c_mid =w cultl %mid, %p1

    %mid_shl =l shl %mid, 32
    %low_sum =l add %p0, %mid_shl
    %c_low =w cultl %low_sum, %p0

    %mid_shr =l shr %mid, 32
    %c_mid_l =l zextw %c_mid
    %c_low_l =l zextw %c_low
    %high_tmp =l add %p3, %mid_shr
    %high_tmp =l add %high_tmp, %c_mid_l
    %high =l add %high_tmp, %c_low_l

    %ij =l add %i, %j
    %off_ij =l shl %ij, 3
    %addr_ij =l add %r, %off_ij
    %r_old =l loadl %addr_ij

    %sum1 =l add %r_old, %low_sum
    %c1 =w cultl %sum1, %r_old

    %sum2 =l add %sum1, %carry
    %c2 =w cultl %sum2, %sum1

    %c1_l =l zextw %c1
    %c2_l =l zextw %c2
    %new_carry =l add %high, %c1_l
    %new_carry =l add %new_carry, %c2_l

    storel %sum2, %addr_ij
    %carry =l %new_carry

    %j =l add %j, %one
    jmp @j_loop
@j_end:
    %dest_idx =l add %i, %nb
    %dest_off =l shl %dest_idx, 3
    %dest_addr =l add %r, %dest_off
    %dest_old =l loadl %dest_addr
    %dest_new =l add %dest_old, %carry
    storel %dest_new, %dest_addr

    %i =l add %i, %one
    jmp @i_loop
@i_done:

@len_calc:
    %len =l add %na, %nb
@len_loop:
    %c_len_pos =w csltl %zero, %len
    jnz %c_len_pos, @len_body, @len_ret
@len_body:
    %idx =l sub %len, %one
    %off_idx =l shl %idx, 3
    %addr_idx =l add %r, %off_idx
    %val =l loadl %addr_idx
    %nz =w csneql %val, %zero
    jnz %nz, @len_ret, @len_dec
@len_dec:
    %len =l sub %len, %one
    jmp @len_loop
@len_ret:
    ret %len
}
