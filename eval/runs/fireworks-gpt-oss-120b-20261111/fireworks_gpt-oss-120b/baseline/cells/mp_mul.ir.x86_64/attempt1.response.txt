export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
    %zero =l 0
    %one =l 1
    %mask32 =l 0xffffffff
    %total =l add %na, %nb

    // zero r[0..total-1]
    %i =l 0
    @zero_loop:
        %c_i =w csltl %i, %total
        jnz %c_i, @zero_body, @zero_done
    @zero_body:
        %off_i =l shl %i, 3
        %addr_i =l add %r, %off_i
        storel %zero, %addr_i
        %i_next =l add %i, %one
        %i =l %i_next
        jmp @zero_loop
    @zero_done:

    // if na==0 or nb==0 skip multiplication (result stays zero)
    %c_na0 =w csltl %na, %one
    %c_nb0 =w csltl %nb, %one
    %c_any =w csltl %c_na0, %c_nb0   // dummy to use both? not needed

    %i =l 0
    @i_loop:
        %c_i =w csltl %i, %na
        jnz %c_i, @i_body, @i_done
    @i_body:
        %ai_off =l shl %i, 3
        %ai_addr =l add %a, %ai_off
        %ai =l loadl %ai_addr
        %carry =l 0
        %j =l 0
        @j_loop:
            %c_j =w csltl %j, %nb
            jnz %c_j, @j_body, @j_end
        @j_body:
            %bj_off =l shl %j, 3
            %bj_addr =l add %b, %bj_off
            %bj =l loadl %bj_addr

            // split limbs
            %ai_low =l and %ai, %mask32
            %ai_high =l shr %ai, 32
            %bj_low =l and %bj, %mask32
            %bj_high =l shr %bj, 32

            // partial products
            %p0 =l mul %ai_low, %bj_low
            %p1 =l mul %ai_low, %bj_high
            %p2 =l mul %ai_high, %bj_low
            %p3 =l mul %ai_high, %bj_high

            // combine middle terms
            %mid =l add %p1, %p2
            %c_mid =w cultl %mid, %p1

            %mid_shl =l shl %mid, 32
            %low_sum =l add %p0, %mid_shl
            %c_low =w cultl %low_sum, %p0

            %mid_shr =l shr %mid, 32
            %high_tmp =l add %p3, %mid_shr
            %high_tmp =l add %high_tmp, (zextw %c_mid)
            %high =l add %high_tmp, (zextw %c_low)

            // add to result limb
            %ij =l add %i, %j
            %off_ij =l shl %ij, 3
            %addr_ij =l add %r, %off_ij
            %r_old =l loadl %addr_ij

            %sum1 =l add %r_old, %low_sum
            %c1 =w cultl %sum1, %r_old

            %sum2 =l add %sum1, %carry
            %c2 =w cultl %sum2, %sum1

            %new_carry_tmp =l add %high, (zextw %c1)
            %new_carry =l add %new_carry_tmp, (zextw %c2)

            storel %sum2, %addr_ij
            %carry =l %new_carry

            %j_next =l add %j, %one
            %j =l %j_next
            jmp @j_loop
        @j_end:
            %dest_idx =l add %i, %nb
            %dest_off =l shl %dest_idx, 3
            %dest_addr =l add %r, %dest_off
            storel %carry, %dest_addr

        %i_next =l add %i, %one
        %i =l %i_next
        jmp @i_loop
    @i_done:

    // compute length of significant limbs
    %len =l add %na, %nb
    @len_loop:
        %c_len =w csltl %zero, %len   // len > 0 ?
        jnz %c_len, @len_body, @len_finish
    @len_body:
        %idx =l sub %len, %one
        %off_idx =l shl %idx, 3
        %addr_idx =l add %r, %off_idx
        %val =l loadl %addr_idx
        %nz =w csneql %val, %zero
        jnz %nz, @len_finish, @len_dec
    @len_dec:
        %len =l sub %len, %one
        jmp @len_loop
    @len_finish:
        ret %len
}
