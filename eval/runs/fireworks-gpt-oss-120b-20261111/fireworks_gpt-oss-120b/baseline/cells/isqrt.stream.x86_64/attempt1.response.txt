isqrt:
    mov rsi, rdi
    xor rdx, rdx
    mov rcx, 0xffffffff
.Lloop:
    cmp rdx, rcx
    jae .Ldone
    mov r8, rdx
    add r8, rcx
    inc r8
    shr r8, 1
    mov rax, r8
    mul r8
    test rdx, rdx
    jne .Lhigh
    cmp rax, rsi
    jbe .Llow
.Lhigh:
    mov rcx, r8
    dec rcx
    jmp .Lloop
.Llow:
    mov rdx, r8
    jmp .Lloop
.Ldone:
    mov rax, rdx
    ret
