count_desc:
    mov     rax, rsi                # rax = n
    cmp     rax, 1
    jle     .Lzero
    mov     rcx, rsi                # rcx = n
    dec     rcx                     # rcx = n-1 (comparisons)
    xor     rdx, rdx                # rdx = i = 0
    xor     rax, rax                # rax = count = 0
.Lloop:
    mov     r8, qword ptr [rdi + rdx*8]
    mov     r9, qword ptr [rdi + rdx*8 + 8]
    cmp     r8, r9
    jle     .Lskip
    inc     rax                     # count++
.Lskip:
    inc     rdx                     # i++
    dec     rcx                     # remaining--
    jnz     .Lloop
    ret
.Lzero:
    xor     rax, rax
    ret
