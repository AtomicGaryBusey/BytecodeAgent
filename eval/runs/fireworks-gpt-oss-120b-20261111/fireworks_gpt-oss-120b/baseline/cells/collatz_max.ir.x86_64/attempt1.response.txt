export function l $collatz_max(l %n) {
@entry:
    %i0 = constl 1
    %max0 = constl 0
    jmp @outer

@outer:
    %i = phi %i0, %i_next
    %max = phi %max0, %max_next
    %gt = csgtl %i, %n
    jnz %gt, @ret
    %x0 = %i
    %cnt0 = constl 0
    jmp @inner

@inner:
    %x = phi %x0, %x_next
    %cnt = phi %cnt0, %cnt_next
    %eq = ceql %x, 1
    jnz %eq, @inner_end
    %odd = andl %x, 1
    %isodd = ceql %odd, 1
    jnz %isodd, @odd
    jmp @even

@odd:
    %three = mull %x, 3
    %next = addl %three, 1
    jmp @inner_body

@even:
    %next = shrl %x, 1
    jmp @inner_body

@inner_body:
    %x_next = %next
    %cnt_next = addl %cnt, 1
    jmp @inner

@inner_end:
    %gt_cnt = csgtl %cnt, %max
    %max_next = sel %gt_cnt, %cnt, %max
    %i_next = addl %i, 1
    jmp @outer

@ret:
    ret %max
}
