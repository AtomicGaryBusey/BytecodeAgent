export function l $collatz_max(l %n) {
@entry:
    %i0 = constl 1
    %max0 = constl 0
    jmp @outer

@outer:
    %i = phi %i0 %i_next
    %max = phi %max0 %max_next
    %done = sgtl %i %n
    jnz %done @ret
    %x0 = %i
    %cnt0 = constl 0
    jmp @inner

@inner:
    %x = phi %x0 %x_next
    %cnt = phi %cnt0 %cnt_next
    %is_one = ceql %x 1
    jnz %is_one @inner_end
    %odd = andl %x 1
    %isodd = ceql %odd 1
    jnz %isodd @odd
    jmp @even

@odd:
    %tmp = mull %x 3
    %next = addl %tmp 1
    jmp @inner_body

@even:
    %next = shrl %x 1
    jmp @inner_body

@inner_body:
    %x_next = %next
    %cnt_next = addl %cnt 1
    jmp @inner

@inner_end:
    %gt = sgtl %cnt %max
    %max_next = sel %gt %cnt %max
    %i_next = addl %i 1
    jmp @outer

@ret:
    ret %max
}
