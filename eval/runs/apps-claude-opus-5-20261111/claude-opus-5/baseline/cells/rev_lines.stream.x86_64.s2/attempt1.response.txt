.globl _start

.text
_start:
    xor r12, r12                    # total bytes read

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r8, [rip + inbuf]
    lea r9, [rip + outbuf]
    xor rbx, rbx                    # input index (line start)
    xor r10, r10                    # output index

outer:
    cmp rbx, r12
    jae done

    mov rdi, rbx                    # scan for newline
scan:
    cmp rdi, r12
    jae scan_end
    cmp byte ptr [r8 + rdi], 10
    je scan_end
    inc rdi
    jmp scan
scan_end:
    mov rsi, rdi                    # save line end
    mov rdx, rdi

rev:
    cmp rdx, rbx
    jbe rev_done
    dec rdx
    mov al, byte ptr [r8 + rdx]
    mov byte ptr [r9 + r10], al
    inc r10
    jmp rev
rev_done:
    mov byte ptr [r9 + r10], 10
    inc r10
    lea rbx, [rsi + 1]
    jmp outer

done:
    test r10, r10
    jz fin
    mov eax, 1                      # write
    mov edi, 1                      # fd 1
    lea rsi, [rip + outbuf]
    mov rdx, r10
    syscall

fin:
    mov eax, 231                    # exit_group
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 65552
