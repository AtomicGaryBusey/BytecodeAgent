.globl _start

.text
_start:
        // x19 = input buffer base
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        adrp    x20, outbuf
        add     x20, x20, :lo12:outbuf
        mov     x21, #0                 // bytes read so far

read_loop:
        mov     x0, #65536
        sub     x2, x0, x21             // space left
        cbz     x2, read_done
        add     x1, x19, x21
        mov     x0, #0
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        ble     read_done
        add     x21, x21, x0
        b       read_loop

read_done:
        mov     x22, #0                 // input index
        mov     x23, #0                 // output index

outer:
        cmp     x22, x21
        b.hs    build_done
        mov     x24, x22                // scanner
scan:
        cmp     x24, x21
        b.hs    scan_done
        ldrb    w0, [x19, x24]
        cmp     w0, #10
        b.eq    scan_done
        add     x24, x24, #1
        b       scan
scan_done:
        // line bytes are [x22, x24)
        mov     x25, x24                // copy cursor (exclusive)
rev:
        cmp     x25, x22
        b.ls    rev_done
        sub     x25, x25, #1
        ldrb    w0, [x19, x25]
        strb    w0, [x20, x23]
        add     x23, x23, #1
        b       rev
rev_done:
        mov     w0, #10
        strb    w0, [x20, x23]
        add     x23, x23, #1
        add     x22, x24, #1
        b       outer

build_done:
        mov     x26, #0                 // bytes written
write_loop:
        cmp     x26, x23
        b.hs    write_done
        mov     x0, #1
        add     x1, x20, x26
        sub     x2, x23, x26
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        ble     write_done
        add     x26, x26, x0
        b       write_loop

write_done:
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       write_done

.bss
.balign 16
inbuf:
        .zero   65536
.balign 16
outbuf:
        .zero   65544
