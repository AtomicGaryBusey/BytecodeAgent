.globl _start

.section .rodata
b64tab:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
.zero 49152
.balign 16
outbuf:
.zero 70016

.text
_start:
    // x19 = inbuf, x20 = bytes read
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
    mov     x26, #49152

read_loop:
    sub     x2, x26, x20
    cbz     x2, read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x27, x21                // save out base
    adrp    x24, b64tab
    add     x24, x24, :lo12:b64tab
    mov     x22, x19                // src
    mov     x23, x20                // remaining

outer:
    cbz     x23, finish
    mov     x25, #57
    cmp     x23, x25
    csel    x25, x23, x25, lo
    sub     x23, x23, x25

grp:
    cmp     x25, #3
    b.lo    tail
    ldrb    w0, [x22], #1
    ldrb    w1, [x22], #1
    ldrb    w2, [x22], #1
    lsl     w3, w0, #16
    orr     w3, w3, w1, lsl #8
    orr     w3, w3, w2
    lsr     w4, w3, #18
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    lsr     w4, w3, #12
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    lsr     w4, w3, #6
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    and     w4, w3, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    sub     x25, x25, #3
    b       grp

tail:
    cbz     x25, eol
    cmp     x25, #1
    b.eq    one_byte
    // two bytes remain
    ldrb    w0, [x22], #1
    ldrb    w1, [x22], #1
    lsl     w3, w0, #8
    orr     w3, w3, w1
    lsr     w4, w3, #10
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    lsr     w4, w3, #4
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    lsl     w4, w3, #2
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    mov     w4, #61
    strb    w4, [x21], #1
    b       eol

one_byte:
    ldrb    w0, [x22], #1
    lsr     w4, w0, #2
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    lsl     w4, w0, #4
    and     w4, w4, #63
    ldrb    w4, [x24, x4]
    strb    w4, [x21], #1
    mov     w4, #61
    strb    w4, [x21], #1
    strb    w4, [x21], #1

eol:
    mov     w4, #10
    strb    w4, [x21], #1
    b       outer

finish:
    sub     x20, x21, x27           // total out length
    mov     x19, x27

write_loop:
    cbz     x20, do_exit
    mov     x8, #64
    mov     x0, #1
    mov     x1, x19
    mov     x2, x20
    svc     #0
    cmp     x0, #0
    b.le    do_exit
    add     x19, x19, x0
    sub     x20, x20, x0
    b       write_loop

do_exit:
    mov     x8, #93
    mov     x0, #0
    svc     #0
    b       do_exit
