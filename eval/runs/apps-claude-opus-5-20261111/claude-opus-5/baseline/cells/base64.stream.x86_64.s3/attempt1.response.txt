.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49168
outbuf:
    .zero 66560

.text
_start:
    xor r12, r12                  # total bytes read
read_loop:
    xor eax, eax                  # read
    xor edi, edi                  # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 49168
    sub rdx, r12
    test rdx, rdx
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r10, [rip + alphabet]
    mov rcx, r12                  # remaining input bytes
    test rcx, rcx
    jz write_out

line_loop:
    mov rbx, rcx
    cmp rbx, 57
    jbe 1f
    mov rbx, 57
1:
    sub rcx, rbx

grp_loop:
    cmp rbx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx r8d, byte ptr [rsi + 2]
    add rsi, 3
    sub rbx, 3

    mov r9d, eax
    shr r9d, 2
    mov r11b, byte ptr [r10 + r9]
    mov byte ptr [rdi], r11b

    and eax, 3
    shl eax, 4
    mov r9d, edx
    shr r9d, 4
    or eax, r9d
    mov r11b, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], r11b

    and edx, 15
    shl edx, 2
    mov r9d, r8d
    shr r9d, 6
    or edx, r9d
    mov r11b, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], r11b

    and r8d, 63
    mov r11b, byte ptr [r10 + r8]
    mov byte ptr [rdi + 3], r11b

    add rdi, 4
    jmp grp_loop

tail:
    test rbx, rbx
    jz line_end

    movzx eax, byte ptr [rsi]
    mov r9d, eax
    shr r9d, 2
    mov r11b, byte ptr [r10 + r9]
    mov byte ptr [rdi], r11b
    and eax, 3
    shl eax, 4
    cmp rbx, 2
    jb one_byte

    movzx edx, byte ptr [rsi + 1]
    mov r9d, edx
    shr r9d, 4
    or eax, r9d
    mov r11b, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], r11b
    and edx, 15
    shl edx, 2
    mov r11b, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], r11b
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    jmp line_end

one_byte:
    mov r11b, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], r11b
    mov word ptr [rdi + 2], 0x3d3d
    add rdi, 4

line_end:
    mov byte ptr [rdi], 10
    inc rdi
    test rcx, rcx
    jnz line_loop

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop

done:
    mov eax, 60
    xor edi, edi
    syscall
