.globl _start

.section .bss
.align 64
buf:
.zero 65536
.zero 64
outbuf:
.zero 64

.text
_start:
    # r12 = lines, r13 = words, r14 = bytes
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d              # r15 = in_word flag (0 = not in word)

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    mov rcx, rax
    add r14, rax
    lea rsi, [rip + buf]
scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    # check newline
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    # whitespace check: 0x20, 0x09..0x0D
    cmp al, 32
    je is_ws
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe is_ws
    # word byte
    test r15d, r15d
    jnz next_byte
    mov r15d, 1
    inc r13
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    dec rcx
    jnz scan_loop
    jmp read_loop

done_read:
    # format output into outbuf
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    mov r8, rdi                 # r8 = end position (one past last written going backwards)
    # print bytes
    mov rax, r14
    call emit
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call emit
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call emit

    # write
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
    hlt

# emit: write decimal of rax backwards starting at [rdi-1], update rdi
emit:
    mov r9, 10
emit_loop:
    xor edx, edx
    div r9
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz emit_loop
    ret
