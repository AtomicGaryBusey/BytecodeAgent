.globl _start

.text
_start:
    # ---- read all of standard input ----
    xor     r12d, r12d                  # total bytes read
rd_loop:
    mov     edx, 65536
    sub     edx, r12d
    jz      rd_done
    lea     rsi, [rip + inbuf]
    add     rsi, r12
    xor     eax, eax                    # read
    xor     edi, edi                    # fd 0
    syscall
    test    rax, rax
    jle     rd_done
    add     r12, rax
    jmp     rd_loop
rd_done:

    # ---- locate first '!' ----
    lea     rbx, [rip + inbuf]
    xor     rcx, rcx
fb_loop:
    cmp     rcx, r12
    jae     fb_none
    cmp     byte ptr [rbx + rcx], 0x21
    je      fb_found
    inc     rcx
    jmp     fb_loop
fb_none:
    mov     r13, r12                    # program length
    mov     r14, r12                    # input cursor (exhausted)
    jmp     validate
fb_found:
    mov     r13, rcx
    lea     r14, [rcx + 1]

    # ---- validate brackets, build jump table ----
validate:
    lea     rbp, [rip + bstack]
    lea     r10, [rip + jtab]
    xor     rcx, rcx                    # index
    xor     edx, edx                    # depth
v_loop:
    cmp     rcx, r13
    jae     v_done
    movzx   eax, byte ptr [rbx + rcx]
    cmp     al, 0x5b                    # '['
    je      v_open
    cmp     al, 0x5d                    # ']'
    je      v_close
v_next:
    inc     rcx
    jmp     v_loop
v_open:
    mov     dword ptr [rbp + rdx*4], ecx
    inc     edx
    jmp     v_next
v_close:
    test    edx, edx
    jz      v_err
    dec     edx
    mov     eax, dword ptr [rbp + rdx*4]
    lea     edi, [rcx + 1]
    mov     dword ptr [r10 + rax*4], edi
    lea     edi, [rax + 1]
    mov     dword ptr [r10 + rcx*4], edi
    jmp     v_next
v_done:
    test    edx, edx
    jnz     v_err

    # ---- execute ----
    lea     r15, [rip + tape]           # data pointer
    lea     r8, [rip + outbuf]
    xor     r9d, r9d                    # output length
    lea     rbp, [rip + ctab]
    xor     rcx, rcx                    # instruction pointer
exec:
    cmp     rcx, r13
    jae     exec_done
    movzx   eax, byte ptr [rbx + rcx]
    inc     rcx
    movzx   eax, byte ptr [rbp + rax]
    jmp     qword ptr [rbp + rax*8 + 256]

h_nop:
    jmp     exec
h_plus:
    inc     byte ptr [r15]
    jmp     exec
h_minus:
    dec     byte ptr [r15]
    jmp     exec
h_right:
    inc     r15
    jmp     exec
h_left:
    dec     r15
    jmp     exec
h_dot:
    movzx   eax, byte ptr [r15]
    mov     byte ptr [r8 + r9], al
    inc     r9
    jmp     exec
h_comma:
    xor     eax, eax
    cmp     r14, r12
    jae     1f
    movzx   eax, byte ptr [rbx + r14]
    inc     r14
1:
    mov     byte ptr [r15], al
    jmp     exec
h_lb:
    cmp     byte ptr [r15], 0
    jne     exec
    mov     ecx, dword ptr [r10 + rcx*4 - 4]
    jmp     exec
h_rb:
    cmp     byte ptr [r15], 0
    je      exec
    mov     ecx, dword ptr [r10 + rcx*4 - 4]
    jmp     exec

exec_done:
    test    r9, r9
    jz      1f
    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + outbuf]
    mov     rdx, r9
    syscall
1:
    mov     eax, 231
    xor     edi, edi
    syscall

v_err:
    mov     eax, 231
    mov     edi, 1
    syscall

.section .rodata
.balign 8
ctab:
    .zero   43
    .byte   1                           # 43 '+'
    .byte   7                           # 44 ','
    .byte   2                           # 45 '-'
    .byte   6                           # 46 '.'
    .zero   13
    .byte   4                           # 60 '<'
    .zero   1
    .byte   3                           # 62 '>'
    .zero   28
    .byte   5                           # 91 '['
    .zero   1
    .byte   8                           # 93 ']'
    .zero   162
jmptab:
    .quad   h_nop
    .quad   h_plus
    .quad   h_minus
    .quad   h_right
    .quad   h_left
    .quad   h_lb
    .quad   h_dot
    .quad   h_comma
    .quad   h_rb

.bss
.balign 16
inbuf:
    .zero   65544
tape:
    .zero   30016
outbuf:
    .zero   65544
jtab:
    .zero   262144
bstack:
    .zero   262152
