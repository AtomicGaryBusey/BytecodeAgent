.globl _start

    .text
_start:
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0

read_loop:
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cmp     x2, #0
    b.eq    read_done
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    mov     x21, #0
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.ne    do_error

    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
// skip_ws : advance x21 over space, tab, lf, cr
// ---------------------------------------------------------------
skip_ws:
sw_loop:
    cmp     x21, x20
    b.hs    sw_done
    ldrb    w0, [x19, x21]
    cmp     w0, #0x20
    b.eq    sw_adv
    cmp     w0, #0x09
    b.eq    sw_adv
    cmp     w0, #0x0a
    b.eq    sw_adv
    cmp     w0, #0x0d
    b.ne    sw_done
sw_adv:
    add     x21, x21, #1
    b       sw_loop
sw_done:
    ret

// ---------------------------------------------------------------
// parse_value
// ---------------------------------------------------------------
parse_value:
    stp     x29, x30, [sp, #-16]!
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7b
    b.eq    pv_obj
    cmp     w0, #0x5b
    b.eq    pv_arr
    cmp     w0, #0x22
    b.eq    pv_str
    cmp     w0, #0x74
    b.eq    pv_true
    cmp     w0, #0x66
    b.eq    pv_false
    cmp     w0, #0x6e
    b.eq    pv_null
    cmp     w0, #0x2d
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pv_num
    b       do_error

pv_obj:
    bl      parse_object
    b       pv_ret
pv_arr:
    bl      parse_array
    b       pv_ret
pv_str:
    bl      parse_string
    b       pv_ret
pv_num:
    bl      parse_number
    b       pv_ret
pv_true:
    adrp    x0, s_true
    add     x0, x0, :lo12:s_true
    mov     x1, #4
    bl      match_lit
    b       pv_ret
pv_false:
    adrp    x0, s_false
    add     x0, x0, :lo12:s_false
    mov     x1, #5
    bl      match_lit
    b       pv_ret
pv_null:
    adrp    x0, s_null
    add     x0, x0, :lo12:s_null
    mov     x1, #4
    bl      match_lit
pv_ret:
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// match_lit : x0 = ptr, x1 = count
// ---------------------------------------------------------------
match_lit:
ml_loop:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w2, [x19, x21]
    ldrb    w3, [x0], #1
    cmp     w2, w3
    b.ne    do_error
    add     x21, x21, #1
    subs    x1, x1, #1
    b.ne    ml_loop
    ret

// ---------------------------------------------------------------
// parse_object
// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1
    bl      skip_ws
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7d
    b.eq    po_end

po_loop:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x22
    b.ne    do_error
    bl      parse_string
    bl      skip_ws
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x3a
    b.ne    do_error
    add     x21, x21, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2c
    b.ne    po_close
    add     x21, x21, #1
    bl      skip_ws
    b       po_loop
po_close:
    cmp     w0, #0x7d
    b.ne    do_error
po_end:
    add     x21, x21, #1
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse_array
// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1
    bl      skip_ws
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5d
    b.eq    pa_end

pa_loop:
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2c
    b.ne    pa_close
    add     x21, x21, #1
    bl      skip_ws
    b       pa_loop
pa_close:
    cmp     w0, #0x5d
    b.ne    do_error
pa_end:
    add     x21, x21, #1
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse_string : x21 points at the opening quote
// ---------------------------------------------------------------
parse_string:
    add     x21, x21, #1
ps_loop:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x22
    b.eq    ps_end
    cmp     w0, #0x5c
    b.eq    ps_esc
    cmp     w0, #0x20
    b.lo    do_error
    add     x21, x21, #1
    b       ps_loop
ps_end:
    add     x21, x21, #1
    ret

ps_esc:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x75
    b.eq    ps_u
    cmp     w0, #0x22
    b.eq    ps_esc_ok
    cmp     w0, #0x5c
    b.eq    ps_esc_ok
    cmp     w0, #0x2f
    b.eq    ps_esc_ok
    cmp     w0, #0x62
    b.eq    ps_esc_ok
    cmp     w0, #0x66
    b.eq    ps_esc_ok
    cmp     w0, #0x6e
    b.eq    ps_esc_ok
    cmp     w0, #0x72
    b.eq    ps_esc_ok
    cmp     w0, #0x74
    b.eq    ps_esc_ok
    b       do_error
ps_esc_ok:
    add     x21, x21, #1
    b       ps_loop

ps_u:
    add     x21, x21, #1
    mov     x1, #4
ps_u_loop:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    sub     w2, w0, #0x30
    cmp     w2, #9
    b.ls    ps_u_ok
    orr     w2, w0, #0x20
    sub     w2, w2, #0x61
    cmp     w2, #5
    b.hi    do_error
ps_u_ok:
    add     x21, x21, #1
    subs    x1, x1, #1
    b.ne    ps_u_loop
    b       ps_loop

// ---------------------------------------------------------------
// parse_number
// ---------------------------------------------------------------
parse_number:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2d
    b.ne    pn_int
    add     x21, x21, #1
pn_int:
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x30
    b.ne    pn_nz
    add     x21, x21, #1
    b       pn_frac
pn_nz:
    sub     w1, w0, #0x31
    cmp     w1, #8
    b.hi    do_error
    add     x21, x21, #1
pn_int_loop:
    cmp     x21, x20
    b.hs    pn_done
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    pn_frac
    add     x21, x21, #1
    b       pn_int_loop

pn_frac:
    cmp     x21, x20
    b.hs    pn_done
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2e
    b.ne    pn_exp
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    do_error
pn_frac_loop:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_done
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pn_frac_loop

pn_exp:
    cmp     x21, x20
    b.hs    pn_done
    ldrb    w0, [x19, x21]
    orr     w0, w0, #0x20
    cmp     w0, #0x65
    b.ne    pn_done
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2b
    b.eq    pn_sign
    cmp     w0, #0x2d
    b.ne    pn_exp_d
pn_sign:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    do_error
    ldrb    w0, [x19, x21]
pn_exp_d:
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    do_error
pn_exp_loop:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_done
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pn_exp_loop
pn_done:
    ret

// ---------------------------------------------------------------
// do_error : prints "error at <x21>\n", exits 1
// ---------------------------------------------------------------
do_error:
    mov     x22, x21
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #9
    mov     x0, #1
    mov     x8, #64
    svc     #0

    adrp    x1, numbuf
    add     x1, x1, :lo12:numbuf
    add     x3, x1, #24
    mov     w6, #10
    sub     x3, x3, #1
    strb    w6, [x3]
    mov     x4, x22
    mov     x5, #10
de_loop:
    udiv    x6, x4, x5
    msub    x7, x6, x5, x4
    add     w7, w7, #0x30
    sub     x3, x3, #1
    strb    w7, [x3]
    mov     x4, x6
    cbnz    x4, de_loop

    adrp    x2, numbuf
    add     x2, x2, :lo12:numbuf
    add     x2, x2, #24
    sub     x2, x2, x3
    mov     x1, x3
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

    .section .rodata
    .balign 8
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"

    .bss
    .balign 16
buf:
    .zero 65536
    .balign 16
numbuf:
    .zero 32
