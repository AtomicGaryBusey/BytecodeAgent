.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16

    .zero 65600

    .zero 324096

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0

read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0

    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    adrp    x23, hextab
    add     x23, x23, :lo12:hextab
    mov     x22, #0

row_loop:

    b.hs    rows_done
    sub     x24, x20, x22
    cmp     x24, #16
    b.ls    1f
    mov     x24, #16

    add     x26, x19, x22
    mov     w0, w22
    bl      hex8
    mov     w3, #32
    strb    w3, [x21], #1
    strb    w3, [x21], #1
    mov     x25, #0

col_loop:

    b.hs    col_blank
    ldrb    w0, [x26, x25]
    lsr     w2, w0, #4
    ldrb    w3, [x23, x2]
    strb    w3, [x21], #1
    and     w2, w0, #15
    ldrb    w3, [x23, x2]
    strb    w3, [x21], #1
    mov     w3, #32
    strb    w3, [x21], #1

col_blank:
    mov     w3, #32
    strb    w3, [x21], #1
    strb    w3, [x21], #1
    strb    w3, [x21], #1
col_next:
    cmp     x25, #7
    b.ne    2f
    mov     w3, #32
    strb    w3, [x21], #1

    add     x25, x25, #1
    cmp     x25, #16
    b.lo    col_loop

    mov     w3, #32
    strb    w3, [x21], #1
    mov     w3, #124
    strb    w3, [x21], #1
    mov     x25, #0

    cmp     x25, x24
    b.hs    ch_done
    ldrb    w0, [x26, x25]
    sub     w1, w0, #32
    cmp     w1, #0x5e
    b.ls    3f
    mov     w0, #46

    strb    w0, [x21], #1
    add     x25, x25, #1
    b       ch_loop

    mov     w3, #124
    strb    w3, [x21], #1
    mov     w3, #10
    strb    w3, [x21], #1
    add     x22, x22, #16

    mov     w0, w20
    bl      hex8
    mov     w3, #10
    strb    w3, [x21], #1

    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x21, x1

    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    wdone
    add     x1, x1, x0
    sub     x2, x2, x0
    b       write_loop

    mov     x8, #93
    svc     #0
    b       wdone

hex8:
    mov     w1, #28
1:
    lsr     w2, w0, w1
    and     w2, w2, #15
    ldrb    w3, [x23, x2]
    strb    w3, [x21], #1
