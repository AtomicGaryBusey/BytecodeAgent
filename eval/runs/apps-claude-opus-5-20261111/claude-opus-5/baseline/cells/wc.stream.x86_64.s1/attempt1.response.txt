.bss
.align 16
buf:
.zero 65536
outbuf:
.zero 64

.text
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # prev was whitespace flag (1 = whitespace)
    mov r15d, 1

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax

    lea rsi, [rip + buf]
    lea rcx, [rsi + rax]    # end pointer
    lea rbx, [rip + ws_table]

scan:
    cmp rsi, rcx
    je read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    movzx edx, byte ptr [rbx + rax]
    # dl = 1 if whitespace
    test dl, dl
    jnz is_ws
    # word byte: if prev was whitespace, new word
    test r15d, r15d
    jz scan
    inc r13
    xor r15d, r15d
    jmp scan
is_ws:
    mov r15d, 1
    jmp scan

done:
    # format output
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    dec rdi

    mov rax, r14
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r13
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r12
    call emit

    inc rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall

# emit: write decimal of rax backwards starting at [rdi], rdi left pointing one before first digit
emit:
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz 1b
    ret
