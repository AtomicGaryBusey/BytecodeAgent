.globl _start

.equ INCAP, 65552
.equ OFLUSH, 8192

.text
_start:
    lea     r15, [rip + vstack]
    lea     rbp, [rip + obuf]
    xor     r12d, r12d

rdloop:
    mov     rdx, INCAP
    sub     rdx, r12
    jz      rddone
    lea     rsi, [rip + inbuf]
    add     rsi, r12
    xor     eax, eax
    xor     edi, edi
    syscall
    test    rax, rax
    jle     rddone
    add     r12, rax
    jmp     rdloop
rddone:
    xor     r13d, r13d

lineloop:
    cmp     r13, r12
    jae     alldone
    lea     rsi, [rip + inbuf]
    add     rsi, r13
    mov     rdi, rsi
    lea     rcx, [rip + inbuf]
    add     rcx, r12
nlscan:
    cmp     rdi, rcx
    jae     nlfound
    cmp     byte ptr [rdi], 10
    je      nlfound
    inc     rdi
    jmp     nlscan
nlfound:
    lea     rax, [rip + inbuf]
    mov     r13, rdi
    sub     r13, rax
    cmp     rdi, rcx
    jae     noskip
    inc     r13
noskip:
    xor     r14d, r14d

tokloop:
    cmp     rsi, rdi
    jae     endtokens
    movzx   eax, byte ptr [rsi]
    cmp     al, 32
    je      tokadv
    cmp     al, 9
    je      tokadv
    jmp     havetok
tokadv:
    inc     rsi
    jmp     tokloop

havetok:
    mov     r8, rsi
tscan:
    cmp     rsi, rdi
    jae     tscanend
    movzx   eax, byte ptr [rsi]
    cmp     al, 32
    je      tscanend
    cmp     al, 9
    je      tscanend
    inc     rsi
    jmp     tscan
tscanend:
    mov     r9, rsi
    mov     rcx, r9
    sub     rcx, r8
    cmp     rcx, 1
    jne     notop
    movzx   eax, byte ptr [r8]
    cmp     al, 43
    je      isop
    cmp     al, 45
    je      isop
    cmp     al, 42
    je      isop
    cmp     al, 47
    je      isop
    cmp     al, 37
    je      isop

notop:
    mov     rdx, r8
    xor     r10d, r10d
    movzx   eax, byte ptr [rdx]
    cmp     al, 45
    jne     nosign
    mov     r10d, 1
    inc     rdx
nosign:
    mov     rcx, r9
    sub     rcx, rdx
    cmp     rcx, 1
    jb      toerror
    cmp     rcx, 18
    ja      toerror
    xor     eax, eax
digloop:
    movzx   r11d, byte ptr [rdx]
    sub     r11d, 48
    cmp     r11d, 9
    ja      toerror
    lea     rax, [rax + rax*4]
    add     rax, rax
    add     rax, r11
    inc     rdx
    cmp     rdx, r9
    jb      digloop
    test    r10d, r10d
    jz      nonegate
    neg     rax
nonegate:
    mov     qword ptr [r15 + r14*8], rax
    inc     r14
    jmp     tokloop

isop:
    mov     r11d, eax
    cmp     r14, 2
    jb      toerror
    dec     r14
    mov     rcx, qword ptr [r15 + r14*8]
    mov     rax, qword ptr [r15 + r14*8 - 8]
    cmp     r11b, 43
    je      opadd
    cmp     r11b, 45
    je      opsub
    cmp     r11b, 42
    je      opmul
    cmp     r11b, 47
    je      opdiv
    jmp     opmod

opadd:
    add     rax, rcx
    jmp     dostore
opsub:
    sub     rax, rcx
    jmp     dostore
opmul:
    imul    rax, rcx
    jmp     dostore
opdiv:
    test    rcx, rcx
    jz      toerror
    cmp     rcx, -1
    jne     divok
    mov     rdx, 1
    shl     rdx, 63
    cmp     rax, rdx
    je      toerror
divok:
    cqo
    idiv    rcx
    jmp     dostore
opmod:
    test    rcx, rcx
    jz      toerror
    cmp     rcx, -1
    jne     modok
    mov     rdx, 1
    shl     rdx, 63
    cmp     rax, rdx
    je      toerror
modok:
    cqo
    idiv    rcx
    mov     rax, rdx
    jmp     dostore

dostore:
    mov     qword ptr [r15 + r14*8 - 8], rax
    jmp     tokloop

endtokens:
    cmp     r14, 1
    jne     toerror
    mov     rax, qword ptr [r15]

    test    rax, rax
    jns     posnum
    mov     byte ptr [rbp], 45
    inc     rbp
    neg     rax
posnum:
    sub     rsp, 32
    lea     r8, [rsp + 32]
    mov     ecx, 10
cvtloop:
    xor     edx, edx
    div     rcx
    add     dl, 48
    dec     r8
    mov     byte ptr [r8], dl
    test    rax, rax
    jnz     cvtloop
cpyloop:
    movzx   eax, byte ptr [r8]
    mov     byte ptr [rbp], al
    inc     rbp
    inc     r8
    lea     rcx, [rsp + 32]
    cmp     r8, rcx
    jb      cpyloop
    add     rsp, 32
    mov     byte ptr [rbp], 10
    inc     rbp
    jmp     afterline

toerror:
    mov     dword ptr [rbp], 0x6f727265
    mov     word ptr [rbp + 4], 0x0a72
    add     rbp, 6

afterline:
    lea     rax, [rip + obuf]
    mov     rdx, rbp
    sub     rdx, rax
    cmp     rdx, OFLUSH
    jb      lineloop
    mov     rsi, rax
    mov     eax, 1
    mov     edi, 1
    syscall
    lea     rbp, [rip + obuf]
    jmp     lineloop

alldone:
    lea     rsi, [rip + obuf]
    mov     rdx, rbp
    sub     rdx, rsi
    test    rdx, rdx
    jz      noflush
    mov     eax, 1
    mov     edi, 1
    syscall
noflush:
    mov     eax, 60
    xor     edi, edi
    syscall
    jmp     noflush

.bss
.balign 16
inbuf:
    .zero 65552
.balign 16
obuf:
    .zero 8256
.balign 16
vstack:
    .zero 8192
