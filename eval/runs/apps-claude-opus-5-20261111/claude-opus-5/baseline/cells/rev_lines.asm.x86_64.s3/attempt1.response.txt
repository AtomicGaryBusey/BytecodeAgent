.globl _start

.text
_start:
    xor r12, r12                    # total bytes read

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz  read_done
    lea rsi, [rip + buf]
    add rsi, r12
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbp, [rip + buf]
    lea r14, [rip + outbuf]
    xor rbx, rbx                    # input position
    xor r13, r13                    # output position

line_loop:
    cmp rbx, r12
    jae finish
    mov rcx, rbx

scan:
    cmp rcx, r12
    jae noeol
    cmp byte ptr [rbp + rcx], 10
    je  found
    inc rcx
    jmp scan

noeol:
    mov rdx, r12
    jmp emit

found:
    mov rdx, rcx

emit:
    mov rax, rdx

rev:
    cmp rax, rbx
    jbe rev_done
    dec rax
    mov r8b, byte ptr [rbp + rax]
    mov byte ptr [r14 + r13], r8b
    inc r13
    jmp rev

rev_done:
    mov byte ptr [r14 + r13], 10
    inc r13
    lea rbx, [rdx + 1]
    jmp line_loop

finish:
    xor r15, r15                    # bytes written

write_loop:
    mov rdx, r13
    sub rdx, r15
    jz  done
    lea rsi, [rip + outbuf]
    add rsi, r15
    mov eax, 1                      # write
    mov edi, 1                      # fd 1
    syscall
    test rax, rax
    jle done
    add r15, rax
    jmp write_loop

done:
    mov eax, 231                    # exit_group
    xor edi, edi
    syscall
    jmp done

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 65544
