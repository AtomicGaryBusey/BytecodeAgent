.globl _start

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, xzr
    movz    x22, #0x1, lsl #16          // read capacity = 65536

rd_loop:
    sub     x2, x22, x20
    cbz     x2, rd_done
    mov     x0, xzr
    add     x1, x19, x20
    mov     x8, #63                     // read
    svc     #0
    cmp     x0, #0
    b.le    rd_done
    add     x20, x20, x0
    b       rd_loop
rd_done:
    strb    wzr, [x19, x20]             // sentinel

    mov     x21, xzr
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.ne    fail

    mov     x0, #1
    adrp    x1, okmsg
    add     x1, x1, :lo12:okmsg
    mov     x2, #3
    mov     x8, #64                     // write
    svc     #0
    mov     x0, xzr
    mov     x8, #93                     // exit
    svc     #0

// ---------------------------------------------------------------
// error reporting: offset in x21, never returns
// ---------------------------------------------------------------
fail:
    mov     x0, #1
    adrp    x1, errmsg
    add     x1, x1, :lo12:errmsg
    mov     x2, #9
    mov     x8, #64
    svc     #0

    adrp    x1, scratch
    add     x1, x1, :lo12:scratch
    add     x23, x1, #32
    mov     x1, x23
    mov     w0, #10
    strb    w0, [x1, #-1]!
    mov     x0, x21
    mov     x2, #10
1:  udiv    x3, x0, x2
    msub    x4, x3, x2, x0
    add     x4, x4, #48
    strb    w4, [x1, #-1]!
    mov     x0, x3
    cbnz    x0, 1b

    sub     x2, x23, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0

// ---------------------------------------------------------------
// skip_ws: advance x21 over 0x20 0x09 0x0A 0x0D
// ---------------------------------------------------------------
skip_ws:
1:  ldrb    w0, [x19, x21]
    cmp     w0, #0x20
    b.eq    2f
    cmp     w0, #0x09
    b.eq    2f
    cmp     w0, #0x0a
    b.eq    2f
    cmp     w0, #0x0d
    b.eq    2f
    ret
2:  add     x21, x21, #1
    b       1b

// ---------------------------------------------------------------
// parse_value
// ---------------------------------------------------------------
parse_value:
    stp     x29, x30, [sp, #-16]!
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7b                   // {
    b.eq    pv_obj
    cmp     w0, #0x5b                   // [
    b.eq    pv_arr
    cmp     w0, #0x22                   // "
    b.eq    pv_str
    cmp     w0, #0x74                   // t
    b.eq    pv_true
    cmp     w0, #0x66                   // f
    b.eq    pv_false
    cmp     w0, #0x6e                   // n
    b.eq    pv_null
    cmp     w0, #0x2d                   // -
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    pv_num
    b       fail

pv_num:
    bl      parse_number
    ldp     x29, x30, [sp], #16
    ret
pv_str:
    bl      parse_string
    ldp     x29, x30, [sp], #16
    ret
pv_obj:
    bl      parse_object
    ldp     x29, x30, [sp], #16
    ret
pv_arr:
    bl      parse_array
    ldp     x29, x30, [sp], #16
    ret

pv_true:
    adrp    x0, s_true
    add     x0, x0, :lo12:s_true
    b       pv_kw
pv_false:
    adrp    x0, s_false
    add     x0, x0, :lo12:s_false
    b       pv_kw
pv_null:
    adrp    x0, s_null
    add     x0, x0, :lo12:s_null
pv_kw:
1:  ldrb    w1, [x0], #1
    cbz     w1, 2f
    ldrb    w2, [x19, x21]
    cmp     w1, w2
    b.ne    fail
    add     x21, x21, #1
    b       1b
2:  ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse_string: x21 points at the opening quote (leaf)
// ---------------------------------------------------------------
parse_string:
    add     x21, x21, #1
1:  ldrb    w0, [x19, x21]
    cmp     w0, #0x22                   // "
    b.eq    2f
    cmp     w0, #0x5c                   // backslash
    b.eq    3f
    cmp     w0, #0x20
    b.lo    fail
    add     x21, x21, #1
    b       1b
2:  add     x21, x21, #1
    ret

3:  add     x21, x21, #1
    ldrb    w0, [x19, x21]
    cmp     w0, #0x22                   // "
    b.eq    4f
    cmp     w0, #0x5c                   // backslash
    b.eq    4f
    cmp     w0, #0x2f                   // /
    b.eq    4f
    cmp     w0, #0x62                   // b
    b.eq    4f
    cmp     w0, #0x66                   // f
    b.eq    4f
    cmp     w0, #0x6e                   // n
    b.eq    4f
    cmp     w0, #0x72                   // r
    b.eq    4f
    cmp     w0, #0x74                   // t
    b.eq    4f
    cmp     w0, #0x75                   // u
    b.eq    5f
    b       fail
4:  add     x21, x21, #1
    b       1b

5:  add     x21, x21, #1
    mov     w3, #4
6:  ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    7f
    orr     w1, w0, #0x20
    sub     w1, w1, #0x61
    cmp     w1, #6
    b.hs    fail
7:  add     x21, x21, #1
    sub     w3, w3, #1
    cbnz    w3, 6b
    b       1b

// ---------------------------------------------------------------
// parse_number (leaf)
// ---------------------------------------------------------------
parse_number:
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2d                   // -
    b.ne    1f
    add     x21, x21, #1
    ldrb    w0, [x19, x21]
1:  cmp     w0, #0x30                   // 0
    b.ne    2f
    add     x21, x21, #1
    b       4f
2:  sub     w1, w0, #0x31               // 1..9
    cmp     w1, #9
    b.hs    fail
3:  add     x21, x21, #1
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    3b

4:  ldrb    w0, [x19, x21]
    cmp     w0, #0x2e                   // .
    b.ne    6f
    add     x21, x21, #1
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.hs    fail
5:  add     x21, x21, #1
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    5b

6:  cmp     w0, #0x65                   // e
    b.eq    7f
    cmp     w0, #0x45                   // E
    b.ne    9f
7:  add     x21, x21, #1
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2b                   // +
    b.eq    10f
    cmp     w0, #0x2d                   // -
    b.ne    11f
10: add     x21, x21, #1
    ldrb    w0, [x19, x21]
11: sub     w1, w0, #0x30
    cmp     w1, #10
    b.hs    fail
12: add     x21, x21, #1
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    12b
9:  ret

// ---------------------------------------------------------------
// parse_object: x21 points at '{'
// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1
    bl      skip_ws
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7d                   // }
    b.eq    8f
1:  ldrb    w0, [x19, x21]
    cmp     w0, #0x22                   // "
    b.ne    fail
    bl      parse_string
    bl      skip_ws
    ldrb    w0, [x19, x21]
    cmp     w0, #0x3a                   // :
    b.ne    fail
    add     x21, x21, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2c                   // ,
    b.ne    2f
    add     x21, x21, #1
    bl      skip_ws
    b       1b
2:  cmp     w0, #0x7d                   // }
    b.ne    fail
8:  add     x21, x21, #1
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse_array: x21 points at '['
// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1
    bl      skip_ws
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5d                   // ]
    b.eq    8f
1:  bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2c                   // ,
    b.ne    2f
    add     x21, x21, #1
    bl      skip_ws
    b       1b
2:  cmp     w0, #0x5d                   // ]
    b.ne    fail
8:  add     x21, x21, #1
    ldp     x29, x30, [sp], #16
    ret

.section .rodata
.align 3
okmsg:
    .ascii "ok\n"
errmsg:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"

.bss
.align 4
scratch:
    .zero 32
inbuf:
    .zero 65552
