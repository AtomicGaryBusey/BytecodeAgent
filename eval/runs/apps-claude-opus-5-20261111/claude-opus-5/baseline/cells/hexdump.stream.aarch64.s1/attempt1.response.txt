.globl _start

.section .rodata
hextab:
.ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
.zero 65552
.balign 16
outbuf:
.zero 324096

.text

// x0 = value (32-bit), appends 8 lowercase hex digits at [x21]
put_hex8:
    mov     w9, #28
1:
    lsr     w10, w0, w9
    and     w10, w10, #0xf
    ldrb    w11, [x23, x10]
    strb    w11, [x21], #1
    subs    w9, w9, #4
    b.ge    1b
    ret

_start:
    adrp    x23, hextab
    add     x23, x23, :lo12:hextab
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf

    mov     x20, #0
    mov     w24, #0x10000
rd_loop:
    sub     x2, x24, x20
    cbz     x2, rd_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    rd_done
    add     x20, x20, x0
    b       rd_loop
rd_done:

    mov     x22, #0
row_loop:
    cmp     x22, x20
    b.hs    rows_done

    sub     x24, x20, x22
    mov     x25, #16
    cmp     x24, #16
    csel    x24, x24, x25, lo

    mov     w0, w22
    bl      put_hex8
    mov     w11, #32
    strb    w11, [x21], #1
    strb    w11, [x21], #1

    mov     x26, #0
2:
    cmp     x26, x24
    b.hs    3f
    add     x9, x19, x22
    ldrb    w10, [x9, x26]
    lsr     w11, w10, #4
    ldrb    w11, [x23, x11]
    strb    w11, [x21], #1
    and     w11, w10, #0xf
    ldrb    w11, [x23, x11]
    strb    w11, [x21], #1
    mov     w11, #32
    strb    w11, [x21], #1
    b       4f
3:
    mov     w11, #32
    strb    w11, [x21], #1
    strb    w11, [x21], #1
    strb    w11, [x21], #1
4:
    cmp     x26, #7
    b.ne    5f
    mov     w11, #32
    strb    w11, [x21], #1
5:
    add     x26, x26, #1
    cmp     x26, #16
    b.lo    2b

    mov     w11, #32
    strb    w11, [x21], #1
    mov     w11, #124
    strb    w11, [x21], #1

    mov     x26, #0
6:
    cmp     x26, x24
    b.hs    7f
    add     x9, x19, x22
    ldrb    w10, [x9, x26]
    sub     w11, w10, #0x20
    cmp     w11, #0x5f
    mov     w12, #46
    csel    w10, w10, w12, lo
    strb    w10, [x21], #1
    add     x26, x26, #1
    b       6b
7:
    mov     w11, #124
    strb    w11, [x21], #1
    mov     w11, #10
    strb    w11, [x21], #1

    add     x22, x22, #16
    b       row_loop

rows_done:
    mov     w0, w20
    bl      put_hex8
    mov     w11, #10
    strb    w11, [x21], #1

    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x21, x1
wr_loop:
    cbz     x2, wr_done
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    wr_done
    add     x1, x1, x0
    sub     x2, x2, x0
    b       wr_loop
wr_done:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       wr_done
