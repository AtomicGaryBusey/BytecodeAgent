.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.equ INCAP, 65536
.equ OUTCAP, 4096

.section .rodata
errstr:
    .ascii "error\n"

.bss
    .balign 8
inbuf:
    .zero 65544
    .balign 8
outbuf:
    .zero 4112
    .balign 8
estack:
    .zero 262400

.text
.globl _start

// ---------------------------------------------------------------
// flush_out: write out buffer, reset count.  clobbers x0,x1,x2,x8
// preserves x3-x18
// ---------------------------------------------------------------
flush_out:
    cbz x26, fo_done
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x25
    mov x2, x26
    svc #0
    mov x26, #0
fo_done:
    ret

// ---------------------------------------------------------------
// out_write(x1 = ptr, x2 = len): append bytes to output buffer
// clobbers x0,x1,x2,x3,x4,x5,x6,x8,x9,x10
// ---------------------------------------------------------------
out_write:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x9, x1
    mov x10, x2
    add x3, x26, x10
    mov x4, #OUTCAP
    cmp x3, x4
    b.ls ow_copy
    bl flush_out
ow_copy:
    add x5, x25, x26
    add x26, x26, x10
ow_loop:
    cbz x10, ow_done
    ldrb w6, [x9], #1
    strb w6, [x5], #1
    sub x10, x10, #1
    b ow_loop
ow_done:
    ldp x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// emit_num(x0 = signed value): print decimal value + newline
// ---------------------------------------------------------------
emit_num:
    stp x29, x30, [sp, #-48]!
    mov x29, sp
    mov x14, x0
    mov x15, #0
    cmp x14, #0
    b.ge en_pos
    mov x15, #1
    neg x14, x14
en_pos:
    add x16, sp, #40
    mov w2, #10
    strb w2, [x16]
    mov x17, #10
en_dig:
    udiv x1, x14, x17
    msub x2, x1, x17, x14
    add w2, w2, #48
    sub x16, x16, #1
    strb w2, [x16]
    mov x14, x1
    cbnz x14, en_dig
    cbz x15, en_nosign
    mov w2, #45
    sub x16, x16, #1
    strb w2, [x16]
en_nosign:
    add x2, sp, #41
    sub x2, x2, x16
    mov x1, x16
    bl out_write
    ldp x29, x30, [sp], #48
    ret

// ---------------------------------------------------------------
// emit_error: print "error\n"
// ---------------------------------------------------------------
emit_error:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    adrp x1, errstr
    add x1, x1, :lo12:errstr
    mov x2, #6
    bl out_write
    ldp x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// entry point
// ---------------------------------------------------------------
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x2, #INCAP
    subs x2, x2, x20
    b.ls read_done
    mov x8, #SYS_read
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:

    adrp x22, estack
    add x22, x22, :lo12:estack
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x26, #0
    mov x21, #0

line_loop:
    cmp x21, x20
    b.hs all_done
    mov x27, x21
    mov x28, x21
find_nl:
    cmp x28, x20
    b.hs got_line
    ldrb w0, [x19, x28]
    cmp w0, #10
    b.eq got_line
    add x28, x28, #1
    b find_nl
got_line:
    add x21, x28, #1

    // evaluate line [x27, x28)
    mov x23, #0
    mov x9, x27
tok_loop:
    cmp x9, x28
    b.hs tok_end
    ldrb w0, [x19, x9]
    cmp w0, #32
    b.eq tok_skip
    cmp w0, #9
    b.eq tok_skip
    b have_tok
tok_skip:
    add x9, x9, #1
    b tok_loop
have_tok:
    mov x10, x9
scan_tok:
    cmp x9, x28
    b.hs tok_got
    ldrb w0, [x19, x9]
    cmp w0, #32
    b.eq tok_got
    cmp w0, #9
    b.eq tok_got
    add x9, x9, #1
    b scan_tok
tok_got:
    sub x11, x9, x10
    cmp x11, #1
    b.ne not_op
    ldrb w0, [x19, x10]
    cmp w0, #43
    b.eq do_op
    cmp w0, #45
    b.eq do_op
    cmp w0, #42
    b.eq do_op
    cmp w0, #47
    b.eq do_op
    cmp w0, #37
    b.eq do_op
not_op:
    mov x12, x10
    mov w13, #0
    ldrb w0, [x19, x12]
    cmp w0, #45
    b.ne num_nosign
    mov w13, #1
    add x12, x12, #1
num_nosign:
    sub x14, x9, x12
    cmp x14, #1
    b.lo line_error
    cmp x14, #18
    b.hi line_error
    mov x15, #0
    mov x16, #10
dig_loop:
    cmp x12, x9
    b.hs dig_done
    ldrb w0, [x19, x12]
    sub w0, w0, #48
    cmp w0, #9
    b.hi line_error
    madd x15, x15, x16, x0
    add x12, x12, #1
    b dig_loop
dig_done:
    cbz w13, push_val
    neg x15, x15
push_val:
    str x15, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

do_op:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #2
    ldr x1, [x22, x23, lsl #3]
    add x12, x23, #1
    ldr x2, [x22, x12, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cmp w0, #47
    b.eq op_div
op_mod:
    cbz x2, line_error
    cmn x2, #1
    b.ne mod_ok
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.eq line_error
mod_ok:
    sdiv x3, x1, x2
    msub x15, x3, x2, x1
    b push_res
op_div:
    cbz x2, line_error
    cmn x2, #1
    b.ne div_ok
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.eq line_error
div_ok:
    sdiv x15, x1, x2
    b push_res
op_add:
    add x15, x1, x2
    b push_res
op_sub:
    sub x15, x1, x2
    b push_res
op_mul:
    mul x15, x1, x2
push_res:
    str x15, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

tok_end:
    cmp x23, #1
    b.ne line_error
    ldr x0, [x22]
    bl emit_num
    b line_loop

line_error:
    bl emit_error
    b line_loop

all_done:
    bl flush_out
    mov x8, #SYS_exit
    mov x0, #0
    svc #0
    b all_done
