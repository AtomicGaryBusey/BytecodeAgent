    .globl _start

    .text
_start:
    // ---- read all of standard input into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // bytes read so far
1:
    mov     x2, #65536
    subs    x2, x2, x20
    b.ls    2f
    mov     x8, #63                 // read
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x20, x20, x0
    b       1b
2:
    add     x20, x19, x20           // x20 = end of input
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf  // x21 = output cursor
    adrp    x23, valstack
    add     x23, x23, :lo12:valstack

    // ---- loop over lines ----
line_loop:
    cmp     x19, x20
    b.hs    all_done
    // find end of line
    mov     x22, x19
1:
    cmp     x22, x20
    b.hs    2f
    ldrb    w0, [x22]
    cmp     w0, #10
    b.eq    2f
    add     x22, x22, #1
    b       1b
2:
    // process line [x19, x22)
    mov     x24, #0                 // value stack depth
    mov     x28, x19                // token scanner

tok_loop:
    // skip separators
1:
    cmp     x28, x22
    b.hs    tok_end
    ldrb    w0, [x28]
    cmp     w0, #32
    b.eq    2f
    cmp     w0, #9
    b.ne    3f
2:
    add     x28, x28, #1
    b       1b
3:
    mov     x25, x28                // token start
1:
    cmp     x28, x22
    b.hs    2f
    ldrb    w0, [x28]
    cmp     w0, #32
    b.eq    2f
    cmp     w0, #9
    b.eq    2f
    add     x28, x28, #1
    b       1b
2:
    mov     x26, x28                // token end
    sub     x1, x26, x25
    cmp     x1, #1
    b.ne    is_num
    ldrb    w0, [x25]
    cmp     w0, #43                 // '+'
    b.eq    do_op
    cmp     w0, #45                 // '-'
    b.eq    do_op
    cmp     w0, #42                 // '*'
    b.eq    do_op
    cmp     w0, #47                 // '/'
    b.eq    do_op
    cmp     w0, #37                 // '%'
    b.eq    do_op

is_num:
    mov     x2, x25
    mov     w3, #0
    ldrb    w0, [x2]
    cmp     w0, #45                 // '-'
    b.ne    1f
    mov     w3, #1
    add     x2, x2, #1
1:
    subs    x1, x26, x2
    b.le    tok_invalid
    cmp     x1, #18
    b.gt    tok_invalid
    mov     x4, #0
    mov     x5, #10
2:
    ldrb    w0, [x2], #1
    sub     w0, w0, #48
    cmp     w0, #9
    b.hi    tok_invalid
    madd    x4, x4, x5, x0
    cmp     x2, x26
    b.lo    2b
    cbz     w3, 3f
    neg     x4, x4
3:
    str     x4, [x23, x24, lsl #3]
    add     x24, x24, #1
    b       tok_loop

do_op:
    cmp     x24, #2
    b.lt    tok_invalid
    sub     x24, x24, #2
    add     x7, x23, x24, lsl #3
    ldp     x5, x6, [x7]            // x5 = a, x6 = b
    cmp     w0, #43
    b.eq    op_add
    cmp     w0, #45
    b.eq    op_sub
    cmp     w0, #42
    b.eq    op_mul
    cmp     w0, #47
    b.eq    op_div
    // '%'
    cbz     x6, tok_invalid
    cmn     x6, #1
    b.ne    1f
    mov     x9, #1
    lsl     x9, x9, #63
    cmp     x5, x9
    b.eq    tok_invalid
1:
    sdiv    x9, x5, x6
    msub    x4, x9, x6, x5
    b       op_store
op_add:
    add     x4, x5, x6
    b       op_store
op_sub:
    sub     x4, x5, x6
    b       op_store
op_mul:
    mul     x4, x5, x6
    b       op_store
op_div:
    cbz     x6, tok_invalid
    cmn     x6, #1
    b.ne    1f
    mov     x9, #1
    lsl     x9, x9, #63
    cmp     x5, x9
    b.eq    tok_invalid
1:
    sdiv    x4, x5, x6
op_store:
    str     x4, [x7]
    add     x24, x24, #1
    b       tok_loop

tok_end:
    cmp     x24, #1
    b.ne    tok_invalid
    ldr     x0, [x23]
    bl      print_num
    mov     w1, #10
    strb    w1, [x21], #1
    b       next_line

tok_invalid:
    adrp    x1, errmsg
    add     x1, x1, :lo12:errmsg
    mov     x2, #6
1:
    ldrb    w3, [x1], #1
    strb    w3, [x21], #1
    subs    x2, x2, #1
    b.ne    1b

next_line:
    cmp     x22, x20
    b.hs    1f
    add     x19, x22, #1
    b       line_loop
1:
    mov     x19, x20
    b       line_loop

all_done:
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x21, x1
    cbz     x2, 2f
1:
    mov     x8, #64                 // write
    mov     x0, #1
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x1, x1, x0
    subs    x2, x2, x0
    b.gt    1b
2:
    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0
    b       2b

// ---- print signed 64-bit value in x0 into output buffer at x21 ----
print_num:
    mov     x1, x0
    tbz     x1, #63, 1f
    mov     w2, #45                 // '-'
    strb    w2, [x21], #1
    neg     x1, x1
1:
    sub     sp, sp, #32
    mov     x3, sp
    mov     x4, #10
    mov     x5, #0
2:
    udiv    x6, x1, x4
    msub    x7, x6, x4, x1
    add     w7, w7, #48
    strb    w7, [x3, x5]
    add     x5, x5, #1
    mov     x1, x6
    cbnz    x1, 2b
3:
    sub     x5, x5, #1
    ldrb    w7, [x3, x5]
    strb    w7, [x21], #1
    cbnz    x5, 3b
    add     sp, sp, #32
    ret

    .section .rodata
errmsg:
    .ascii  "error\n"

    .bss
    .balign 16
inbuf:
    .zero   65536
    .balign 16
outbuf:
    .zero   400000
    .balign 16
valstack:
    .zero   8192
