.globl _start

.text
_start:
    lea     rbx, [rip + buf]
    mov     r13, rbx
    lea     r14, [rip + buf_end]
read_loop:
    mov     rsi, r13
    mov     rdx, r14
    sub     rdx, r13
    jle     read_done
    xor     eax, eax
    xor     edi, edi
    syscall
    test    rax, rax
    jle     read_done
    add     r13, rax
    jmp     read_loop
read_done:
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jne     fail
    lea     rsi, [rip + okmsg]
    mov     edx, 3
    mov     eax, 1
    mov     edi, 1
    syscall
    mov     eax, 231
    xor     edi, edi
    syscall
    jmp     .

# ---------------- skip whitespace ----------------
skip_ws:
    cmp     rbx, r13
    jae     2f
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x20
    je      1f
    cmp     al, 0x09
    je      1f
    cmp     al, 0x0a
    je      1f
    cmp     al, 0x0d
    jne     2f
1:
    inc     rbx
    jmp     skip_ws
2:
    ret

# ---------------- value ----------------
parse_value:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x7b
    je      parse_object
    cmp     al, 0x5b
    je      parse_array
    cmp     al, 0x22
    je      parse_string
    cmp     al, 0x2d
    je      parse_number
    cmp     al, 0x30
    jb      1f
    cmp     al, 0x39
    jbe     parse_number
1:
    cmp     al, 0x74
    je      kw_true
    cmp     al, 0x66
    je      kw_false
    cmp     al, 0x6e
    je      kw_null
    jmp     fail

# ---------------- keywords ----------------
kw_true:
    lea     rsi, [rip + str_true]
    mov     ecx, 4
    jmp     kw_match
kw_false:
    lea     rsi, [rip + str_false]
    mov     ecx, 5
    jmp     kw_match
kw_null:
    lea     rsi, [rip + str_null]
    mov     ecx, 4
kw_match:
1:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, byte ptr [rsi]
    jne     fail
    inc     rbx
    inc     rsi
    dec     ecx
    jnz     1b
    ret

# ---------------- string ----------------
parse_string:
    inc     rbx
1:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x22
    je      9f
    cmp     al, 0x5c
    je      3f
    cmp     al, 0x20
    jb      fail
    inc     rbx
    jmp     1b
9:
    inc     rbx
    ret
3:
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x75
    je      5f
    cmp     al, 0x22
    je      4f
    cmp     al, 0x5c
    je      4f
    cmp     al, 0x2f
    je      4f
    cmp     al, 0x62
    je      4f
    cmp     al, 0x66
    je      4f
    cmp     al, 0x6e
    je      4f
    cmp     al, 0x72
    je      4f
    cmp     al, 0x74
    je      4f
    jmp     fail
4:
    inc     rbx
    jmp     1b
5:
    inc     rbx
    mov     ecx, 4
6:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    mov     edx, eax
    sub     dl, 0x30
    cmp     dl, 9
    jbe     str_hexok
    or      al, 0x20
    sub     al, 0x61
    cmp     al, 5
    ja      fail
str_hexok:
    inc     rbx
    dec     ecx
    jnz     6b
    jmp     1b

# ---------------- number ----------------
parse_number:
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2d
    jne     num_int
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
num_int:
    cmp     al, 0x30
    jne     num_nz
    inc     rbx
    jmp     num_frac
num_nz:
    cmp     al, 0x31
    jb      fail
    cmp     al, 0x39
    ja      fail
    inc     rbx
num_intdig:
    cmp     rbx, r13
    jae     num_done
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      num_frac
    cmp     al, 0x39
    ja      num_frac
    inc     rbx
    jmp     num_intdig
num_frac:
    cmp     rbx, r13
    jae     num_done
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2e
    jne     num_exp
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      fail
    cmp     al, 0x39
    ja      fail
num_fracdig:
    inc     rbx
    cmp     rbx, r13
    jae     num_done
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      num_exp
    cmp     al, 0x39
    ja      num_exp
    jmp     num_fracdig
num_exp:
    cmp     al, 0x65
    je      num_exp2
    cmp     al, 0x45
    jne     num_done
num_exp2:
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2b
    je      num_sign
    cmp     al, 0x2d
    jne     num_expdig0
num_sign:
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
num_expdig0:
    cmp     al, 0x30
    jb      fail
    cmp     al, 0x39
    ja      fail
num_expdig:
    inc     rbx
    cmp     rbx, r13
    jae     num_done
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      num_done
    cmp     al, 0x39
    ja      num_done
    jmp     num_expdig
num_done:
    ret

# ---------------- object ----------------
parse_object:
    inc     rbx
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    cmp     byte ptr [rbx], 0x7d
    jne     obj_member
    inc     rbx
    ret
obj_member:
    cmp     byte ptr [rbx], 0x22
    jne     fail
    call    parse_string
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    cmp     byte ptr [rbx], 0x3a
    jne     fail
    inc     rbx
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x7d
    je      obj_end
    cmp     al, 0x2c
    jne     fail
    inc     rbx
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    jmp     obj_member
obj_end:
    inc     rbx
    ret

# ---------------- array ----------------
parse_array:
    inc     rbx
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    cmp     byte ptr [rbx], 0x5d
    jne     arr_elem
    inc     rbx
    ret
arr_elem:
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x5d
    je      arr_end
    cmp     al, 0x2c
    jne     fail
    inc     rbx
    call    skip_ws
    jmp     arr_elem
arr_end:
    inc     rbx
    ret

# ---------------- failure ----------------
fail:
    lea     rax, [rip + buf]
    sub     rbx, rax
    lea     rdi, [rip + msgbuf]
    lea     rsi, [rip + errpfx]
    mov     rax, qword ptr [rsi]
    mov     qword ptr [rdi], rax
    movzx   eax, byte ptr [rsi + 8]
    mov     byte ptr [rdi + 8], al
    add     rdi, 9
    lea     rsi, [rip + scratch_end]
    mov     rax, rbx
    mov     ecx, 10
1:
    xor     edx, edx
    div     rcx
    add     dl, 0x30
    dec     rsi
    mov     byte ptr [rsi], dl
    test    rax, rax
    jnz     1b
    lea     r8, [rip + scratch_end]
2:
    movzx   eax, byte ptr [rsi]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     rsi
    cmp     rsi, r8
    jne     2b
    mov     byte ptr [rdi], 10
    inc     rdi
    lea     rsi, [rip + msgbuf]
    mov     rdx, rdi
    sub     rdx, rsi
    mov     eax, 1
    mov     edi, 1
    syscall
    mov     eax, 231
    mov     edi, 1
    syscall
    jmp     .

.section .rodata
okmsg:
    .ascii  "ok\n"
errpfx:
    .ascii  "error at "
str_true:
    .ascii  "true"
str_false:
    .ascii  "false"
str_null:
    .ascii  "null"
    .quad   0

.bss
.align 16
buf:
    .zero   65600
buf_end:
msgbuf:
    .zero   64
scratch:
    .zero   32
scratch_end:
    .zero   8
