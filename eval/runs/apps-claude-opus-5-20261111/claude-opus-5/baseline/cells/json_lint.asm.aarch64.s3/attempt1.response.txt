.globl _start

// ---------------------------------------------------------------
//  json_lint : validate stdin as one RFC 8259 JSON document
//  prints "ok\n" (status 0) or "error at N\n" (status 1)
// ---------------------------------------------------------------

        .section .rodata
s_err:  .ascii  "error at "
s_ok:   .ascii  "ok\n"
lit_rue:  .ascii "rue"
lit_alse: .ascii "alse"
lit_ull:  .ascii "ull"

        .bss
        .balign 16
buf:    .zero   65792
msgbuf: .zero   64

        .text

// ---------------------------------------------------------------
_start:
        adrp    x19, buf
        add     x19, x19, :lo12:buf
        mov     x20, xzr                // length read so far

rd_loop:
        movz    x3, #1, lsl #16         // 65536
        add     x3, x3, #256            // 65792 = buffer size
        sub     x2, x3, x20             // space remaining
        cbz     x2, rd_done
        mov     x0, xzr                 // fd 0
        add     x1, x19, x20
        mov     x8, #63                 // read
        svc     #0
        cmp     x0, #0
        b.le    rd_done
        add     x20, x20, x0
        b       rd_loop
rd_done:

        mov     x21, xzr                // current offset
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        cmp     x21, x20
        b.ne    fail

        // ---- success ----
        adrp    x1, s_ok
        add     x1, x1, :lo12:s_ok
        mov     x2, #3
        mov     x0, #1
        mov     x8, #64                 // write
        svc     #0
        mov     x0, xzr
        mov     x8, #93                 // exit
        svc     #0
        b       .

// ---------------------------------------------------------------
// fail: x21 = error offset.  Never returns.
fail:
        adrp    x0, msgbuf
        add     x0, x0, :lo12:msgbuf
        adrp    x1, s_err
        add     x1, x1, :lo12:s_err
        mov     w2, #9
f_cpy:
        ldrb    w3, [x1], #1
        strb    w3, [x0], #1
        subs    w2, w2, #1
        b.ne    f_cpy

        sub     sp, sp, #32
        mov     x5, sp
        mov     x4, x21
        mov     x6, xzr
        mov     x7, #10
f_dig:
        udiv    x9, x4, x7
        msub    x10, x9, x7, x4
        add     w10, w10, #48
        strb    w10, [x5, x6]
        add     x6, x6, #1
        mov     x4, x9
        cbnz    x4, f_dig
f_rev:
        sub     x6, x6, #1
        ldrb    w3, [x5, x6]
        strb    w3, [x0], #1
        cbnz    x6, f_rev

        mov     w3, #10                 // '\n'
        strb    w3, [x0], #1

        adrp    x1, msgbuf
        add     x1, x1, :lo12:msgbuf
        sub     x2, x0, x1
        mov     x0, #1
        mov     x8, #64                 // write
        svc     #0
        mov     x0, #1
        mov     x8, #93                 // exit
        svc     #0
        b       .

// ---------------------------------------------------------------
// skip_ws : advance x21 over space/tab/lf/cr
skip_ws:
sw_loop:
        cmp     x21, x20
        b.hs    sw_done
        ldrb    w0, [x19, x21]
        cmp     w0, #0x20
        b.eq    sw_next
        cmp     w0, #0x09
        b.eq    sw_next
        cmp     w0, #0x0a
        b.eq    sw_next
        cmp     w0, #0x0d
        b.eq    sw_next
        b       sw_done
sw_next:
        add     x21, x21, #1
        b       sw_loop
sw_done:
        ret

// ---------------------------------------------------------------
// parse_value : tail-dispatch, no frame
parse_value:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x7b               // '{'
        b.eq    parse_object
        cmp     w0, #0x5b               // '['
        b.eq    parse_array
        cmp     w0, #0x22               // '"'
        b.eq    parse_string
        cmp     w0, #0x74               // 't'
        b.eq    pv_true
        cmp     w0, #0x66               // 'f'
        b.eq    pv_false
        cmp     w0, #0x6e               // 'n'
        b.eq    pv_null
        cmp     w0, #0x2d               // '-'
        b.eq    parse_number
        sub     w1, w0, #48
        cmp     w1, #10
        b.lo    parse_number
        b       fail

pv_true:
        adrp    x0, lit_rue
        add     x0, x0, :lo12:lit_rue
        mov     w1, #3
        b       match_lit
pv_false:
        adrp    x0, lit_alse
        add     x0, x0, :lo12:lit_alse
        mov     w1, #4
        b       match_lit
pv_null:
        adrp    x0, lit_ull
        add     x0, x0, :lo12:lit_ull
        mov     w1, #3
        b       match_lit

// match_lit : x0 = rest of literal, w1 = its length; first byte
// at x21 already known to match.
match_lit:
        add     x21, x21, #1
ml_loop:
        cbz     w1, ml_done
        cmp     x21, x20
        b.hs    fail
        ldrb    w2, [x19, x21]
        ldrb    w3, [x0], #1
        cmp     w2, w3
        b.ne    fail
        add     x21, x21, #1
        sub     w1, w1, #1
        b       ml_loop
ml_done:
        ret

// ---------------------------------------------------------------
// parse_string : x21 points at the opening quote
parse_string:
        add     x21, x21, #1
ps_loop:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x22               // '"'
        b.eq    ps_end
        cmp     w0, #0x5c               // '\'
        b.eq    ps_esc
        cmp     w0, #0x20
        b.lo    fail
        add     x21, x21, #1
        b       ps_loop
ps_end:
        add     x21, x21, #1
        ret
ps_esc:
        add     x21, x21, #1
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x75               // 'u'
        b.eq    ps_u
        cmp     w0, #0x22               // '"'
        b.eq    ps_simple
        cmp     w0, #0x5c               // '\'
        b.eq    ps_simple
        cmp     w0, #0x2f               // '/'
        b.eq    ps_simple
        cmp     w0, #0x62               // 'b'
        b.eq    ps_simple
        cmp     w0, #0x66               // 'f'
        b.eq    ps_simple
        cmp     w0, #0x6e               // 'n'
        b.eq    ps_simple
        cmp     w0, #0x72               // 'r'
        b.eq    ps_simple
        cmp     w0, #0x74               // 't'
        b.eq    ps_simple
        b       fail
ps_simple:
        add     x21, x21, #1
        b       ps_loop
ps_u:
        add     x21, x21, #1
        mov     w1, #4
ps_hex:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        sub     w2, w0, #48             // '0'
        cmp     w2, #10
        b.lo    ps_hok
        and     w3, w0, #0xdf
        sub     w2, w3, #65             // 'A'
        cmp     w2, #6
        b.hs    fail
ps_hok:
        add     x21, x21, #1
        subs    w1, w1, #1
        b.ne    ps_hex
        b       ps_loop

// ---------------------------------------------------------------
// parse_number : x21 points at '-' or a digit
parse_number:
        ldrb    w0, [x19, x21]
        cmp     w0, #0x2d               // '-'
        b.ne    pn_int
        add     x21, x21, #1
pn_int:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #48                 // '0'
        b.eq    pn_zero
        sub     w1, w0, #49             // '1'
        cmp     w1, #9
        b.hs    fail
        add     x21, x21, #1
pn_digits:
        cmp     x21, x20
        b.hs    pn_done
        ldrb    w0, [x19, x21]
        sub     w1, w0, #48
        cmp     w1, #10
        b.hs    pn_frac
        add     x21, x21, #1
        b       pn_digits
pn_zero:
        add     x21, x21, #1
pn_frac:
        cmp     x21, x20
        b.hs    pn_done
        ldrb    w0, [x19, x21]
        cmp     w0, #0x2e               // '.'
        b.ne    pn_exp
        add     x21, x21, #1
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        sub     w1, w0, #48
        cmp     w1, #10
        b.hs    fail
        add     x21, x21, #1
pn_fdig:
        cmp     x21, x20
        b.hs    pn_done
        ldrb    w0, [x19, x21]
        sub     w1, w0, #48
        cmp     w1, #10
        b.hs    pn_exp
        add     x21, x21, #1
        b       pn_fdig
pn_exp:
        cmp     x21, x20
        b.hs    pn_done
        ldrb    w0, [x19, x21]
        cmp     w0, #0x65               // 'e'
        b.eq    pn_e
        cmp     w0, #0x45               // 'E'
        b.ne    pn_done
pn_e:
        add     x21, x21, #1
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x2b               // '+'
        b.eq    pn_esign
        cmp     w0, #0x2d               // '-'
        b.ne    pn_edig1
pn_esign:
        add     x21, x21, #1
pn_edig1:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        sub     w1, w0, #48
        cmp     w1, #10
        b.hs    fail
        add     x21, x21, #1
pn_edigs:
        cmp     x21, x20
        b.hs    pn_done
        ldrb    w0, [x19, x21]
        sub     w1, w0, #48
        cmp     w1, #10
        b.hs    pn_done
        add     x21, x21, #1
        b       pn_edigs
pn_done:
        ret

// ---------------------------------------------------------------
// parse_object : x21 points at '{'
parse_object:
        stp     x29, x30, [sp, #-16]!
        add     x21, x21, #1
        bl      skip_ws
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x7d               // '}'
        b.eq    po_end
po_member:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x22               // '"'
        b.ne    fail
        bl      parse_string
        bl      skip_ws
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x3a               // ':'
        b.ne    fail
        add     x21, x21, #1
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x7d               // '}'
        b.eq    po_end
        cmp     w0, #0x2c               // ','
        b.ne    fail
        add     x21, x21, #1
        bl      skip_ws
        b       po_member
po_end:
        add     x21, x21, #1
        ldp     x29, x30, [sp], #16
        ret

// ---------------------------------------------------------------
// parse_array : x21 points at '['
parse_array:
        stp     x29, x30, [sp, #-16]!
        add     x21, x21, #1
        bl      skip_ws
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x5d               // ']'
        b.eq    pa_end
pa_elem:
        bl      parse_value
        bl      skip_ws
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x5d               // ']'
        b.eq    pa_end
        cmp     w0, #0x2c               // ','
        b.ne    fail
        add     x21, x21, #1
        bl      skip_ws
        b       pa_elem
pa_end:
        add     x21, x21, #1
        ldp     x29, x30, [sp], #16
        ret
