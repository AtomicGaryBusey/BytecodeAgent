.globl _start

.text
_start:
    lea rsi, [rip + buf]
    xor r12d, r12d                  # total bytes read
rloop:
    mov edx, 65536
    sub rdx, r12
    jz rdone
    lea rsi, [rip + buf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle rdone
    add r12, rax
    jmp rloop
rdone:
    lea rsi, [rip + buf]

    # find first '!'
    xor rcx, rcx
scan:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp scan
no_bang:
    mov r10, r12                    # prog_len
    mov r13, rsi                    # input ptr
    mov r14, rsi                    # input end (empty)
    jmp compact
found_bang:
    mov r10, rcx
    lea r13, [rsi + rcx + 1]
    lea r14, [rsi + r12]

    # strip non-command bytes in place
compact:
    xor rcx, rcx
    xor rdx, rdx
cloop:
    cmp rcx, r10
    jae cdone
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x2b
    je ckeep
    cmp al, 0x2d
    je ckeep
    cmp al, 0x3c
    je ckeep
    cmp al, 0x3e
    je ckeep
    cmp al, 0x2e
    je ckeep
    cmp al, 0x2c
    je ckeep
    cmp al, 0x5b
    je ckeep
    cmp al, 0x5d
    je ckeep
    jmp cloop
ckeep:
    mov byte ptr [rsi + rdx], al
    inc rdx
    jmp cloop
cdone:
    mov r10, rdx

    # bracket matching
    lea r12, [rip + jmptab]
    lea r9, [rip + brkstk]
    xor rcx, rcx
    xor r8d, r8d
bloop:
    cmp rcx, r10
    jae bdone
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je bopen
    cmp al, 0x5d
    je bclose
bnext:
    inc rcx
    jmp bloop
bopen:
    mov dword ptr [r9 + r8*4], ecx
    inc r8
    jmp bnext
bclose:
    test r8, r8
    jz berr
    dec r8
    mov eax, dword ptr [r9 + r8*4]
    mov dword ptr [r12 + rax*4], ecx
    mov dword ptr [r12 + rcx*4], eax
    jmp bnext
bdone:
    test r8, r8
    jnz berr

    # run
    lea rbp, [rip + tape]
    lea r15, [rip + outbuf]
    xor ebx, ebx
run:
    cmp rbx, r10
    jae finish
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    cmp al, 0x2b
    je o_inc
    cmp al, 0x2d
    je o_dec
    cmp al, 0x3e
    je o_right
    cmp al, 0x3c
    je o_left
    cmp al, 0x5b
    je o_lbr
    cmp al, 0x5d
    je o_rbr
    cmp al, 0x2e
    je o_dot
    jmp o_comma

o_inc:
    inc byte ptr [rbp]
    jmp run
o_dec:
    dec byte ptr [rbp]
    jmp run
o_right:
    inc rbp
    jmp run
o_left:
    dec rbp
    jmp run
o_lbr:
    cmp byte ptr [rbp], 0
    jne run
    lea rax, [rbx - 1]
    mov eax, dword ptr [r12 + rax*4]
    lea rbx, [rax + 1]
    jmp run
o_rbr:
    cmp byte ptr [rbp], 0
    je run
    lea rax, [rbx - 1]
    mov eax, dword ptr [r12 + rax*4]
    lea rbx, [rax + 1]
    jmp run
o_dot:
    mov al, byte ptr [rbp]
    mov byte ptr [r15], al
    inc r15
    lea rax, [rip + outbuf]
    add rax, 65536
    cmp r15, rax
    jb run
    call flush
    jmp run
o_comma:
    cmp r13, r14
    jae o_comma_zero
    movzx eax, byte ptr [r13]
    inc r13
    mov byte ptr [rbp], al
    jmp run
o_comma_zero:
    mov byte ptr [rbp], 0
    jmp run

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall
    mov eax, 231
    xor edi, edi
    syscall

berr:
    mov eax, 60
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall

flush:
    push rsi
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    jz flush_done
    mov eax, 1
    mov edi, 1
    syscall
flush_done:
    pop rsi
    lea r15, [rip + outbuf]
    ret

.bss
.balign 16
buf:
    .zero 65544
.balign 16
tape:
    .zero 30016
.balign 16
jmptab:
    .zero 262144
.balign 16
brkstk:
    .zero 262144
.balign 16
outbuf:
    .zero 65552
