.globl _start

.section .rodata
hexdig:
.ascii "0123456789abcdef"

.bss
.align 16
inbuf:
.zero 65536
.align 16
outbuf:
.zero 327680

.text
_start:
    xor r12, r12            # total length
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea rbx, [rip + outbuf]     # output pointer
    xor r13, r13                # offset
    lea r8, [rip + hexdig]

row_loop:
    cmp r13, r12
    jae rows_done
    # n = min(16, total - off)
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    # print offset: 8 hex digits
    mov eax, r13d
    mov ecx, 8
off_digits:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    dec ecx
    jnz off_digits
    mov word ptr [rbx], 0x2020
    add rbx, 2

    # hex bytes
    lea rsi, [rip + inbuf]
    add rsi, r13
    xor rcx, rcx
hex_loop:
    cmp rcx, r14
    jae hex_pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    and eax, 15
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rbx + 1], al
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp hex_next
hex_pad:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
hex_next:
    cmp rcx, 7
    jne 1f
    mov byte ptr [rbx], 0x20
    inc rbx
1:
    inc rcx
    cmp rcx, 16
    jb hex_loop

    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2

    xor rcx, rcx
asc_loop:
    cmp rcx, r14
    jae asc_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb 1f
    cmp al, 0x7e
    jbe 2f
1:
    mov al, 0x2e
2:
    mov byte ptr [rbx], al
    inc rbx
    inc rcx
    jmp asc_loop
asc_done:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2

    add r13, r14
    jmp row_loop

rows_done:
    # final total length line
    mov eax, r12d
    mov ecx, 8
tot_digits:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    dec ecx
    jnz tot_digits
    mov byte ptr [rbx], 0x0a
    inc rbx

    # write out
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall
