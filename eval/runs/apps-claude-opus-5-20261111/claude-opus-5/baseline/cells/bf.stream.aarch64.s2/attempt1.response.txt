.globl _start

.text

_start:
    // ---- read all of standard input into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x28, #0                 // total bytes read
    movz    x26, #1, lsl #16        // 65536
1:
    sub     x2, x26, x28
    cbz     x2, 2f
    mov     x0, #0
    add     x1, x19, x28
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x28, x28, x0
    b       1b
2:
    // ---- locate first '!' ----
    mov     x21, #0
3:
    cmp     x21, x28
    b.hs    4f
    ldrb    w0, [x19, x21]
    cmp     w0, #33
    b.eq    5f
    add     x21, x21, #1
    b       3b
4:
    // no '!' : program is everything, input empty
    mov     x20, x28
    add     x23, x19, x28
    add     x24, x19, x28
    b       6f
5:
    mov     x20, x21                // program length
    add     x23, x19, x21
    add     x23, x23, #1            // input start
    add     x24, x19, x28           // input end
6:
    // ---- bracket matching ----
    adrp    x27, matchtab
    add     x27, x27, :lo12:matchtab
    adrp    x10, bstack
    add     x10, x10, :lo12:bstack
    mov     x9, #0                  // stack depth
    mov     x0, #0                  // scan position
7:
    cmp     x0, x20
    b.hs    11f
    ldrb    w1, [x19, x0]
    cmp     w1, #91
    b.eq    8f
    cmp     w1, #93
    b.eq    9f
10:
    add     x0, x0, #1
    b       7b
8:
    str     x0, [x10, x9, lsl #3]
    add     x9, x9, #1
    b       10b
9:
    cbz     x9, bad
    sub     x9, x9, #1
    ldr     x2, [x10, x9, lsl #3]
    str     w0, [x27, x2, lsl #2]
    mov     w3, w2
    str     w3, [x27, x0, lsl #2]
    b       10b
11:
    cbnz    x9, bad

    // ---- set up machine state ----
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x26, outbuf
    add     x26, x26, :lo12:outbuf
    mov     x25, #0                 // output count
    movz    x9, #1, lsl #16         // 65536
    mov     x21, #0                 // pc

run:
    cmp     x21, x20
    b.hs    done
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #43                 // '+'
    b.eq    c_inc
    cmp     w0, #45                 // '-'
    b.eq    c_dec
    cmp     w0, #62                 // '>'
    b.eq    c_right
    cmp     w0, #60                 // '<'
    b.eq    c_left
    cmp     w0, #46                 // '.'
    b.eq    c_out
    cmp     w0, #44                 // ','
    b.eq    c_in
    cmp     w0, #91                 // '['
    b.eq    c_open
    cmp     w0, #93                 // ']'
    b.eq    c_close
    b       run

c_inc:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       run
c_dec:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       run
c_right:
    add     x22, x22, #1
    b       run
c_left:
    sub     x22, x22, #1
    b       run
c_out:
    ldrb    w1, [x22]
    strb    w1, [x26, x25]
    add     x25, x25, #1
    cmp     x25, x9
    b.lo    run
    bl      flush
    b       run
c_in:
    cmp     x23, x24
    b.hs    12f
    ldrb    w1, [x23], #1
    strb    w1, [x22]
    b       run
12:
    strb    wzr, [x22]
    b       run
c_open:
    ldrb    w1, [x22]
    cbnz    w1, run
    sub     x2, x21, #1
    ldr     w3, [x27, x2, lsl #2]
    add     x21, x3, #1
    b       run
c_close:
    ldrb    w1, [x22]
    cbz     w1, run
    sub     x2, x21, #1
    ldr     w3, [x27, x2, lsl #2]
    add     x21, x3, #1
    b       run

done:
    bl      flush
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       done

bad:
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       bad

flush:
    cbz     x25, 13f
    mov     x0, #1
    mov     x1, x26
    mov     x2, x25
    mov     x8, #64
    svc     #0
    mov     x25, #0
13:
    ret

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
matchtab:
    .zero 262144
tape:
    .zero 30016
bstack:
    .zero 8192
