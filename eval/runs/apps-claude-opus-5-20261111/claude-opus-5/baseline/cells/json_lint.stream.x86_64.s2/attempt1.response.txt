.globl _start

.text
_start:
    lea r12, [rip + buf]
rd_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    lea rdx, [rip + buf_end]
    sub rdx, r12
    test rdx, rdx
    jle rd_done
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    lea rbp, [rip + buf]
    mov rbx, rbp
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp hang

# ---------------- failure path ----------------
fail:
    sub rbx, rbp
    lea rdi, [rip + msgbuf_end]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, rbx
    mov ecx, 10
fl_dig:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz fl_dig
    sub rdi, 9
    lea rsi, [rip + errtxt]
    mov rdx, rdi
    mov ecx, 9
fl_cp:
    mov al, byte ptr [rsi]
    mov byte ptr [rdx], al
    inc rsi
    inc rdx
    dec ecx
    jnz fl_cp
    lea rdx, [rip + msgbuf_end]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall
hang:
    jmp hang

# ---------------- whitespace ----------------
skip_ws:
sw_loop:
    cmp rbx, r12
    jae sw_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0a
    je sw_adv
    cmp al, 0x0d
    je sw_adv
    jmp sw_done
sw_adv:
    inc rbx
    jmp sw_loop
sw_done:
    ret

# ---------------- value ----------------
parse_value:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb pv_lit
    cmp al, 0x39
    jbe parse_number
pv_lit:
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6e
    je pv_null
    jmp fail
pv_true:
    lea rsi, [rip + s_true]
    jmp match_lit
pv_false:
    lea rsi, [rip + s_false]
    jmp match_lit
pv_null:
    lea rsi, [rip + s_null]
    jmp match_lit

match_lit:
ml_loop:
    movzx eax, byte ptr [rsi]
    test al, al
    jz ml_done
    cmp rbx, r12
    jae fail
    cmp al, byte ptr [rbx]
    jne fail
    inc rbx
    inc rsi
    jmp ml_loop
ml_done:
    ret

# ---------------- string ----------------
parse_string:
    inc rbx
ps_loop:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5c
    je ps_esc
    cmp al, 0x20
    jb fail
    inc rbx
    jmp ps_loop
ps_end:
    inc rbx
    ret
ps_esc:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x75
    je ps_u
    lea rsi, [rip + s_esc]
ps_eset:
    movzx ecx, byte ptr [rsi]
    test cl, cl
    jz fail
    inc rsi
    cmp cl, al
    jne ps_eset
    inc rbx
    jmp ps_loop
ps_u:
    inc rbx
    mov edx, 4
ps_hex:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    mov ecx, eax
    or ecx, 0x20
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe ps_hok
    cmp cl, 0x61
    jb fail
    cmp cl, 0x66
    ja fail
ps_hok:
    inc rbx
    dec edx
    jnz ps_hex
    jmp ps_loop

# ---------------- number ----------------
parse_number:
    movzx eax, byte ptr [rbx]
    cmp al, 0x2d
    jne pn_int
    inc rbx
    cmp rbx, r12
    jae fail
pn_int:
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jne pn_nz
    inc rbx
    jmp pn_frac
pn_nz:
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
pn_idig:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    inc rbx
    jmp pn_idig
pn_frac:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x2e
    jne pn_exp
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
pn_fdig:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    inc rbx
    jmp pn_fdig
pn_exp:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x65
    je pn_esign
    cmp al, 0x45
    jne pn_done
pn_esign:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x2b
    je pn_eskip
    cmp al, 0x2d
    jne pn_echk
pn_eskip:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
pn_echk:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
pn_edig:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    ja pn_done
    inc rbx
    jmp pn_edig
pn_done:
    ret

# ---------------- object ----------------
parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    cmp byte ptr [rbx], 0x7d
    jne po_member
    inc rbx
    ret
po_member:
    cmp rbx, r12
    jae fail
    cmp byte ptr [rbx], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp rbx, r12
    jae fail
    cmp byte ptr [rbx], 0x3a
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x2c
    je po_next
    cmp al, 0x7d
    jne fail
    inc rbx
    ret
po_next:
    inc rbx
    call skip_ws
    jmp po_member

# ---------------- array ----------------
parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    cmp byte ptr [rbx], 0x5d
    jne pa_elem
    inc rbx
    ret
pa_elem:
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x2c
    je pa_next
    cmp al, 0x5d
    jne fail
    inc rbx
    ret
pa_next:
    inc rbx
    call skip_ws
    jmp pa_elem

.section .rodata
okmsg:
    .ascii "ok\n"
errtxt:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"
s_esc:
    .byte 0x22, 0x5c, 0x2f, 0x62, 0x66, 0x6e, 0x72, 0x74, 0x00

.bss
.balign 16
buf:
    .zero 70000
buf_end:
msgbuf:
    .zero 32
msgbuf_end:
    .zero 16
