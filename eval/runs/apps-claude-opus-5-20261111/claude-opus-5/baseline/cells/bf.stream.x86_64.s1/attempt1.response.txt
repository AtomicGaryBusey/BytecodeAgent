.globl _start

.text
_start:
    xor r8, r8                      # total bytes read

read_loop:
    mov edx, 65536
    sub rdx, r8
    jz  read_done
    lea rsi, [rip + inbuf]
    add rsi, r8
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r8, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx

find_loop:
    cmp rcx, r8
    je  no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je  found_bang
    inc rcx
    jmp find_loop

no_bang:
    mov r9, r8                      # program length
    mov r13, r8                     # input start (empty)
    mov r14, r8                     # input end
    jmp scan_init

found_bang:
    mov r9, rcx
    lea r13, [rcx + 1]
    mov r14, r8

scan_init:
    lea rbp, [rip + jmptbl]
    lea rdi, [rip + bstack]
    xor ecx, ecx                    # pc
    xor edx, edx                    # bracket stack depth

scan_loop:
    cmp rcx, r9
    je  scan_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je  is_open
    cmp al, 0x5d
    je  is_close

scan_next:
    inc rcx
    jmp scan_loop

is_open:
    cmp rdx, 32768
    jae bad
    mov dword ptr [rdi + rdx*4], ecx
    inc rdx
    jmp scan_next

is_close:
    test rdx, rdx
    jz  bad
    dec rdx
    mov eax, dword ptr [rdi + rdx*4]
    mov dword ptr [rbp + rax*4], ecx
    mov dword ptr [rbp + rcx*4], eax
    jmp scan_next

scan_end:
    test rdx, rdx
    jnz bad

    lea r12, [rip + tape]
    add r13, rsi
    add r14, rsi
    xor ebx, ebx
    xor r15d, r15d

exec:
    cmp rbx, r9
    jae done
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    cmp al, 0x2b
    je  op_incc
    cmp al, 0x2d
    je  op_decc
    cmp al, 0x3e
    je  op_incp
    cmp al, 0x3c
    je  op_decp
    cmp al, 0x5b
    je  op_open
    cmp al, 0x5d
    je  op_close
    cmp al, 0x2e
    je  op_out
    cmp al, 0x2c
    je  op_in
    jmp exec

op_incc:
    inc byte ptr [r12]
    jmp exec

op_decc:
    dec byte ptr [r12]
    jmp exec

op_incp:
    inc r12
    jmp exec

op_decp:
    dec r12
    jmp exec

op_open:
    cmp byte ptr [r12], 0
    jne exec
    lea rax, [rbx - 1]
    mov ebx, dword ptr [rbp + rax*4]
    inc rbx
    jmp exec

op_close:
    cmp byte ptr [r12], 0
    je  exec
    lea rax, [rbx - 1]
    mov ebx, dword ptr [rbp + rax*4]
    inc rbx
    jmp exec

op_out:
    lea rdi, [rip + outbuf]
    mov al, byte ptr [r12]
    mov byte ptr [rdi + r15], al
    inc r15
    cmp r15, 65536
    jb  exec
    call flush_out
    jmp exec

op_in:
    cmp r13, r14
    jae in_zero
    movzx eax, byte ptr [r13]
    inc r13
    mov byte ptr [r12], al
    jmp exec

in_zero:
    mov byte ptr [r12], 0
    jmp exec

done:
    call flush_out
    xor edi, edi
    mov eax, 231
    syscall
    mov eax, 60
    syscall

bad:
    mov edi, 1
    mov eax, 231
    syscall
    mov eax, 60
    syscall

flush_out:
    test r15, r15
    jz  2f
    push rsi
    push rbx
    lea rbx, [rip + outbuf]
1:
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r15
    syscall
    test rax, rax
    jle 3f
    add rbx, rax
    sub r15, rax
    jnz 1b
3:
    pop rbx
    pop rsi
    xor r15d, r15d
2:
    ret

.bss
.balign 64
inbuf:
    .zero 65536
.balign 64
outbuf:
    .zero 65536
.balign 64
tape:
    .zero 30016
.balign 64
jmptbl:
    .zero 262144
.balign 64
bstack:
    .zero 131072
