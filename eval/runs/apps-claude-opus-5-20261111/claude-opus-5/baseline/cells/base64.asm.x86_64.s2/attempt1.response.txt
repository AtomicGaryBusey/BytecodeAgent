.globl _start

.section .rodata
alpha:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
.zero 49160
.balign 16
outbuf:
.zero 70016

.text

# encode one full group of 3 bytes: rsi -> input (advanced by 3),
# rdi -> output (advanced by 4), rbx = alphabet base
enc3:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]
    add rsi, 3
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov ecx, eax
    shr ecx, 18
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], cl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], cl
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+2], cl
    mov ecx, eax
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+3], cl
    add rdi, 4
    ret

_start:
    xor r12d, r12d
read_loop:
    mov rax, 49152
    sub rax, r12
    jz read_done
    mov rdx, rax
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz finish_exit

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alpha]
    mov r13, r12

chunk_loop:
    cmp r13, 57
    jb tail
    mov r14d, 19
1:
    call enc3
    dec r14d
    jnz 1b
    mov byte ptr [rdi], 10
    inc rdi
    sub r13, 57
    jmp chunk_loop

tail:
    test r13, r13
    jz write_out

    mov rax, r13
    xor edx, edx
    mov ecx, 3
    div rcx
    mov r15, rdx
    test rax, rax
    jz tail_rem
2:
    call enc3
    dec rax
    jnz 2b

tail_rem:
    test r15, r15
    jz tail_nl
    cmp r15, 1
    je one_byte

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    mov ecx, eax
    shr ecx, 18
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], cl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], cl
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+2], cl
    mov byte ptr [rdi+3], 61
    add rdi, 4
    jmp tail_nl

one_byte:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov ecx, eax
    shr ecx, 18
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], cl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov cl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], cl
    mov byte ptr [rdi+2], 61
    mov byte ptr [rdi+3], 61
    add rdi, 4

tail_nl:
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
write_loop:
    test r13, r13
    jz finish_exit
    mov rdx, r13
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish_exit
    add rsi, rax
    sub r13, rax
    jmp write_loop

finish_exit:
    mov eax, 231
    xor edi, edi
    syscall
