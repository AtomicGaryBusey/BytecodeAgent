.globl _start

.section .rodata
alpha:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
.zero 960
.balign 16
outbuf:
.zero 1280

.text

# encode one group of 3 bytes
# rsi -> input (advanced by 3), rdi -> output (advanced by 4), rbx = alphabet
# clobbers rax, rcx, rdx
enc3:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]
    add rsi, 3
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov ecx, eax
    shr ecx, 18
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], dl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], dl
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+2], dl
    mov ecx, eax
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+3], dl
    add rdi, 4
    ret

_start:
    lea rbx, [rip + alpha]
    xor r14d, r14d                  # eof flag

outer:
    xor r13d, r13d                  # bytes in inbuf

fill:
    cmp r13, 912
    jae filled
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, 912
    sub rdx, r13
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle fill_eof
    add r13, rax
    jmp fill
fill_eof:
    mov r14d, 1

filled:
    test r13, r13
    jz check_done

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]

lines:
    cmp r13, 57
    jb tail
    mov r15d, 19
1:
    call enc3
    dec r15d
    jnz 1b
    mov byte ptr [rdi], 10
    inc rdi
    sub r13, 57
    jmp lines

tail:
    test r13, r13
    jz flush

    mov eax, r13d
    xor edx, edx
    mov ecx, 3
    div ecx
    mov r12d, edx                   # remainder 0,1,2
    mov r15d, eax                   # full groups
    test r15d, r15d
    jz tail_rem
2:
    call enc3
    dec r15d
    jnz 2b

tail_rem:
    test r12d, r12d
    jz tail_nl
    cmp r12d, 1
    je one_byte

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    mov ecx, eax
    shr ecx, 18
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], dl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], dl
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+2], dl
    mov byte ptr [rdi+3], 61
    add rdi, 4
    jmp tail_nl

one_byte:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov ecx, eax
    shr ecx, 18
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi], dl
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbx+rcx]
    mov byte ptr [rdi+1], dl
    mov byte ptr [rdi+2], 61
    mov byte ptr [rdi+3], 61
    add rdi, 4

tail_nl:
    mov byte ptr [rdi], 10
    inc rdi

flush:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
write_loop:
    test r13, r13
    jz check_done
    mov rdx, r13
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle check_done
    add rsi, rax
    sub r13, rax
    jmp write_loop

check_done:
    test r14d, r14d
    jz outer

    mov eax, 231
    xor edi, edi
    syscall
    hlt
