.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
.balign 16
outbuf:
    .zero 70000

.text
_start:
    xor r12d, r12d

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 49152
    sub edx, r12d
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alpha]
    mov rcx, r12
    xor r8d, r8d

main:
    cmp rcx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3
    sub rcx, 3

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi + 2], dl

    mov edx, eax
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi + 3], dl

    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne main
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp main

tail:
    test rcx, rcx
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 2
    jb tail_one
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

tail_one:
    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi + 1], dl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

    cmp rcx, 2
    jb tail_done
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi + 2], dl

tail_done:
    add rdi, 4
    add r8d, 4

finish:
    test r8d, r8d
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 231
    xor edi, edi
    syscall
