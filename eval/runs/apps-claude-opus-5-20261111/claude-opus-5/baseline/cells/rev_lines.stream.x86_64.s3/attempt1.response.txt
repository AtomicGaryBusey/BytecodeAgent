.globl _start

.text
_start:
    xor r12, r12                    # total bytes read

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor rcx, rcx                    # start index of current line

proc_loop:
    cmp rcx, r12
    jae finish
    mov rsi, rcx

scan_loop:
    cmp rsi, r12
    jae copy_start
    cmp byte ptr [rbx + rsi], 10
    je copy_start
    inc rsi
    jmp scan_loop

copy_start:
    mov rdx, rsi

copy_loop:
    cmp rdx, rcx
    jbe copy_done
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp copy_loop

copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rcx, [rsi + 1]
    jmp proc_loop

finish:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi                    # total output length

write_loop:
    test r13, r13
    jz done
    mov eax, 1                      # write
    mov edi, 1                      # fd 1
    mov rdx, r13
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r13, rax
    jmp write_loop

done:
    mov eax, 231                    # exit_group
    xor edi, edi
    syscall
    jmp done

.bss
.balign 64
inbuf:
    .zero 65536
.balign 64
outbuf:
    .zero 65600
