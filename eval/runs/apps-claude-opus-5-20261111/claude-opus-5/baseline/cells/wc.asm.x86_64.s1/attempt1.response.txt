.text
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in-word flag
    lea rbp, [rip + ws_table]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax
    lea rsi, [rip + buf]
    lea rbx, [rsi + rax]

scan_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne 1f
    inc r12
1:
    movzx ecx, byte ptr [rbp + rax]
    test ecx, ecx
    jnz is_ws
    # word byte
    test r15, r15
    jnz 2f
    mov r15d, 1
    inc r13
2:
    inc rsi
    cmp rsi, rbx
    jb scan_loop
    jmp read_loop
is_ws:
    xor r15d, r15d
    inc rsi
    cmp rsi, rbx
    jb scan_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call putnum
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call putnum
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call putnum
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall

# rax = value, rdi = dest buffer; returns rdi advanced
putnum:
    mov r8, rdi
    mov r9d, 10
    lea rcx, [rsp - 64]
    mov rsi, rcx
1:
    xor edx, edx
    div r9
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [r8], al
    inc r8
    inc rsi
    cmp rsi, rcx
    jb 2b
    mov rdi, r8
    ret
