.globl _start

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .zero 65552
outbuf:
    .zero 330000

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea rdi, [rip + outbuf]
    xor r14, r14
row_loop:
    cmp r14, r12
    jae rows_done
    mov rbx, r12
    sub rbx, r14
    cmp rbx, 16
    jbe 1f
    mov ebx, 16
1:
    mov eax, r14d
    call put8hex
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hex_loop:
    cmp rcx, rbx
    jae pad_slot
    lea rsi, [rip + inbuf]
    add rsi, r14
    movzx eax, byte ptr [rsi + rcx]
    lea r8, [rip + hexdig]
    mov edx, eax
    shr edx, 4
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp slot_next
pad_slot:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
slot_next:
    cmp rcx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc rcx
    cmp rcx, 16
    jb hex_loop

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
ascii_loop:
    cmp rcx, rbx
    jae ascii_done
    lea rsi, [rip + inbuf]
    add rsi, r14
    movzx eax, byte ptr [rsi + rcx]
    mov edx, 0x2e
    cmp al, 0x20
    jb 3f
    cmp al, 0x7e
    ja 3f
    mov edx, eax
3:
    mov byte ptr [rdi], dl
    inc rdi
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    add r14, 16
    jmp row_loop

rows_done:
    mov eax, r12d
    call put8hex
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov r15, rdi
    sub r15, rsi
write_loop:
    test r15, r15
    jle write_done
    mov eax, 1
    mov edi, 1
    mov rdx, r15
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub r15, rax
    jmp write_loop
write_done:
    mov eax, 231
    xor edi, edi
    syscall
    jmp write_done

put8hex:
    lea r9, [rip + hexdig]
    mov r10d, 8
    mov r11d, eax
5:
    rol r11d, 4
    mov eax, r11d
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r10d
    jnz 5b
    ret
