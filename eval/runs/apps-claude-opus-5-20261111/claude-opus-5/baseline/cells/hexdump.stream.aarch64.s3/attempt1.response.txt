.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 324096

.text

// ---- put8hex: write 8 lowercase hex digits of w24 to [x21], advancing x21
put8hex:
    mov     w9, #28
1:
    lsr     w10, w24, w9
    and     w10, w10, #0xf
    ldrb    w11, [x23, x10]
    strb    w11, [x21], #1
    subs    w9, w9, #4
    b.ge    1b
    ret

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x23, hextab
    add     x23, x23, :lo12:hextab
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf

    mov     x20, #0                 // total bytes read

read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    mov     x22, #0                 // current offset

row_loop:
    cmp     x22, x20
    b.hs    rows_done

    // offset field
    mov     w24, w22
    bl      put8hex
    mov     w13, #32
    strb    w13, [x21], #1
    strb    w13, [x21], #1

    // n = min(16, total - offset)
    sub     x25, x20, x22
    cmp     x25, #16
    b.ls    1f
    mov     x25, #16
1:
    mov     x26, #0                 // i

col_loop:
    cmp     x26, x25
    b.hs    pad_col
    add     x9, x19, x22
    ldrb    w10, [x9, x26]
    lsr     w11, w10, #4
    ldrb    w11, [x23, x11]
    strb    w11, [x21], #1
    and     w12, w10, #0xf
    ldrb    w12, [x23, x12]
    strb    w12, [x21], #1
    mov     w13, #32
    strb    w13, [x21], #1
    b       after_col
pad_col:
    mov     w13, #32
    strb    w13, [x21], #1
    strb    w13, [x21], #1
    strb    w13, [x21], #1
after_col:
    cmp     x26, #7
    b.ne    no_gap
    mov     w13, #32
    strb    w13, [x21], #1
no_gap:
    add     x26, x26, #1
    cmp     x26, #16
    b.lo    col_loop

    // " |"
    mov     w13, #32
    strb    w13, [x21], #1
    mov     w13, #124
    strb    w13, [x21], #1

    mov     x26, #0
char_loop:
    cmp     x26, x25
    b.hs    char_done
    add     x9, x19, x22
    ldrb    w10, [x9, x26]
    cmp     w10, #0x20
    b.lo    1f
    cmp     w10, #0x7e
    b.hi    1f
    b       2f
1:
    mov     w10, #46
2:
    strb    w10, [x21], #1
    add     x26, x26, #1
    b       char_loop
char_done:

    mov     w13, #124
    strb    w13, [x21], #1
    mov     w13, #10
    strb    w13, [x21], #1

    add     x22, x22, #16
    b       row_loop

rows_done:
    mov     w24, w20
    bl      put8hex
    mov     w13, #10
    strb    w13, [x21], #1

    // write out
    adrp    x27, outbuf
    add     x27, x27, :lo12:outbuf
    sub     x28, x21, x27

write_loop:
    cbz     x28, write_done
    mov     x0, #1
    mov     x1, x27
    mov     x2, x28
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    write_done
    add     x27, x27, x0
    sub     x28, x28, x0
    b       write_loop
write_done:

    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       write_done
