.globl _start

.text
_start:
    // x19 = lines, x20 = words, x21 = bytes, x22 = in_word flag
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #16384
    svc #0
    cmp x0, #0
    ble done
    mov x3, x0                  // count
    add x21, x21, x0            // bytes += n
    adrp x1, buf
    add x1, x1, :lo12:buf

byte_loop:
    ldrb w4, [x1], #1
    // check whitespace
    cmp w4, #0x20
    b.eq is_ws
    sub w5, w4, #9
    cmp w5, #4
    b.hi not_ws
    // 0x09..0x0D range
    cmp w4, #0x0A
    b.ne is_ws
    add x19, x19, #1
is_ws:
    mov x22, #0
    b next_byte
not_ws:
    cmp x22, #0
    b.ne next_byte
    mov x22, #1
    add x20, x20, #1
next_byte:
    subs x3, x3, #1
    b.ne byte_loop
    b read_loop

done:
    // format output into obuf
    adrp x23, obuf
    add x23, x23, :lo12:obuf
    mov x24, x23

    mov x0, x19
    bl print_num
    mov w1, #0x20
    strb w1, [x24], #1
    mov x0, x20
    bl print_num
    mov w1, #0x20
    strb w1, [x24], #1
    mov x0, x21
    bl print_num
    mov w1, #0x0A
    strb w1, [x24], #1

    sub x2, x24, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// x0 = number, appends decimal digits at x24 (advancing it)
print_num:
    sub sp, sp, #32
    mov x5, sp
    mov x6, #0
    mov x7, #10
pn_div:
    udiv x9, x0, x7
    msub x10, x9, x7, x0
    add w10, w10, #0x30
    strb w10, [x5, x6]
    add x6, x6, #1
    mov x0, x9
    cmp x0, #0
    b.ne pn_div
pn_out:
    sub x6, x6, #1
    ldrb w10, [x5, x6]
    strb w10, [x24], #1
    cmp x6, #0
    b.ne pn_out
    add sp, sp, #32
    ret

.bss
.balign 16
buf:
    .zero 16384
obuf:
    .zero 64
