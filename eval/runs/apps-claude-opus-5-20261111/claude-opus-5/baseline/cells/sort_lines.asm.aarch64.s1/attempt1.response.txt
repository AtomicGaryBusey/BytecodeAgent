.globl _start

.text
_start:
    // ---- read all of stdin into buf ----
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0                 // total bytes read
    mov     x5, #32768              // capacity
1:
    subs    x2, x5, x20
    b.ls    2f
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #1
    b.lt    2f
    add     x20, x20, x0
    b       1b
2:

    // ---- split into lines ----
    adrp    x21, lines
    add     x21, x21, :lo12:lines
    mov     x22, #0                 // n lines
    mov     x23, #0                 // start of current line
    mov     x24, #0                 // scan index
    mov     x6, #4096               // MAXLINES
1:
    cmp     x24, x20
    b.hs    3f
    ldrb    w0, [x19, x24]
    add     x24, x24, #1
    cmp     w0, #10
    b.ne    1b
    cmp     x22, x6
    b.hs    2f
    add     x0, x19, x23
    sub     x1, x24, x23
    sub     x1, x1, #1
    lsl     x2, x22, #4
    add     x2, x21, x2
    stp     x0, x1, [x2]
    add     x22, x22, #1
2:
    mov     x23, x24
    b       1b
3:
    cmp     x23, x20
    b.hs    4f
    cmp     x22, x6
    b.hs    4f
    add     x0, x19, x23
    sub     x1, x20, x23
    lsl     x2, x22, #4
    add     x2, x21, x2
    stp     x0, x1, [x2]
    add     x22, x22, #1
4:

    // ---- bottom-up merge sort over the line table ----
    adrp    x26, tmp
    add     x26, x26, :lo12:tmp
    mov     x25, #1                 // width
msort_outer:
    cmp     x25, x22
    b.hs    msort_done
    mov     x27, #0                 // lo
msort_inner:
    cmp     x27, x22
    b.hs    msort_pass_done
    add     x0, x27, x25            // mid
    cmp     x0, x22
    csel    x0, x0, x22, ls
    add     x1, x27, x25, lsl #1    // hi
    cmp     x1, x22
    csel    x1, x1, x22, ls
    mov     x2, x27                 // index into run A
    mov     x3, x0                  // index into run B
    lsl     x4, x27, #4
    add     x4, x26, x4             // destination pointer in tmp
merge_loop:
    cmp     x2, x0
    b.hs    take_b
    cmp     x3, x1
    b.hs    take_a
    // compare lines[x2] and lines[x3]
    lsl     x9, x2, #4
    add     x9, x21, x9
    ldp     x10, x11, [x9]          // ptrA, lenA
    lsl     x9, x3, #4
    add     x9, x21, x9
    ldp     x12, x13, [x9]          // ptrB, lenB
    cmp     x11, x13
    csel    x14, x11, x13, ls       // min length
    mov     x15, #0
cmp_loop:
    cmp     x15, x14
    b.hs    cmp_tie
    ldrb    w16, [x10, x15]
    ldrb    w17, [x12, x15]
    cmp     w16, w17
    b.lo    take_a
    b.hi    take_b
    add     x15, x15, #1
    b       cmp_loop
cmp_tie:
    cmp     x11, x13
    b.ls    take_a
    b       take_b
take_a:
    lsl     x9, x2, #4
    add     x9, x21, x9
    ldp     x10, x11, [x9]
    stp     x10, x11, [x4], #16
    add     x2, x2, #1
    b       merge_loop
take_b:
    cmp     x3, x1
    b.hs    merge_done
    lsl     x9, x3, #4
    add     x9, x21, x9
    ldp     x10, x11, [x9]
    stp     x10, x11, [x4], #16
    add     x3, x3, #1
    b       merge_loop
merge_done:
    add     x27, x27, x25, lsl #1
    b       msort_inner
msort_pass_done:
    lsl     x10, x22, #4
    mov     x9, #0
1:
    cmp     x9, x10
    b.hs    2f
    ldr     x11, [x26, x9]
    str     x11, [x21, x9]
    add     x9, x9, #8
    b       1b
2:
    lsl     x25, x25, #1
    b       msort_outer
msort_done:

    // ---- build output buffer ----
    adrp    x28, outbuf
    add     x28, x28, :lo12:outbuf
    mov     x10, x28
    mov     x9, #0
1:
    cmp     x9, x22
    b.hs    3f
    lsl     x11, x9, #4
    add     x11, x21, x11
    ldp     x12, x13, [x11]
2:
    cbz     x13, 4f
    ldrb    w14, [x12], #1
    strb    w14, [x10], #1
    sub     x13, x13, #1
    b       2b
4:
    mov     w14, #10
    strb    w14, [x10], #1
    add     x9, x9, #1
    b       1b
3:

    // ---- write it out ----
    sub     x2, x10, x28
    mov     x1, x28
1:
    cbz     x2, 2f
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #1
    b.lt    2f
    add     x1, x1, x0
    subs    x2, x2, x0
    b       1b
2:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       2b
