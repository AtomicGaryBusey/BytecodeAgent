.globl _start

.text
_start:
    // ---- read all of stdin into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #32768
    subs    x2, x2, x20
    b.ls    read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // ---- split into lines: table of (offset, length) ----
    adrp    x21, lines
    add     x21, x21, :lo12:lines
    mov     x22, #0                 // line count
    mov     x23, #0                 // start of current line
    mov     x24, #0                 // scan index
parse_loop:
    cmp     x24, x20
    b.hs    parse_tail
    ldrb    w0, [x19, x24]
    add     x24, x24, #1
    cmp     w0, #10
    b.ne    parse_loop
    sub     x1, x24, #1
    sub     x1, x1, x23
    lsl     x2, x22, #4
    add     x2, x21, x2
    stp     x23, x1, [x2]
    add     x22, x22, #1
    mov     x23, x24
    b       parse_loop
parse_tail:
    cmp     x23, x20
    b.hs    parse_done
    sub     x1, x20, x23
    lsl     x2, x22, #4
    add     x2, x21, x2
    stp     x23, x1, [x2]
    add     x22, x22, #1
parse_done:
    cbz     x22, do_exit

    // ---- bottom-up merge sort ----
    adrp    x25, tmp
    add     x25, x25, :lo12:tmp
    mov     x26, #1                 // width
sort_outer:
    cmp     x26, x22
    b.hs    sort_done
    mov     x27, #0                 // block start
sort_mid:
    cmp     x27, x22
    b.hs    sort_next
    mov     x10, x27                // a
    add     x11, x27, x26           // aend
    cmp     x11, x22
    csel    x11, x11, x22, lo
    mov     x12, x11                // b
    add     x13, x27, x26
    add     x13, x13, x26           // bend
    cmp     x13, x22
    csel    x13, x13, x22, lo
    mov     x14, x27                // out index
merge_loop:
    cmp     x10, x11
    b.hs    merge_a_done
    cmp     x12, x13
    b.hs    merge_take_a
    lsl     x0, x10, #4
    add     x0, x21, x0
    lsl     x1, x12, #4
    add     x1, x21, x1
    bl      cmp_lines
    cmp     x0, #0
    b.le    merge_take_a
    lsl     x1, x12, #4
    add     x1, x21, x1
    ldp     x2, x3, [x1]
    lsl     x1, x14, #4
    add     x1, x25, x1
    stp     x2, x3, [x1]
    add     x12, x12, #1
    add     x14, x14, #1
    b       merge_loop
merge_take_a:
    lsl     x1, x10, #4
    add     x1, x21, x1
    ldp     x2, x3, [x1]
    lsl     x1, x14, #4
    add     x1, x25, x1
    stp     x2, x3, [x1]
    add     x10, x10, #1
    add     x14, x14, #1
    b       merge_loop
merge_a_done:
    cmp     x12, x13
    b.hs    merge_fin
    lsl     x1, x12, #4
    add     x1, x21, x1
    ldp     x2, x3, [x1]
    lsl     x1, x14, #4
    add     x1, x25, x1
    stp     x2, x3, [x1]
    add     x12, x12, #1
    add     x14, x14, #1
    b       merge_a_done
merge_fin:
    add     x27, x27, x26
    add     x27, x27, x26
    b       sort_mid
sort_next:
    mov     x0, x21
    mov     x21, x25
    mov     x25, x0
    lsl     x26, x26, #1
    b       sort_outer
sort_done:

    // ---- build output ----
    adrp    x28, outbuf
    add     x28, x28, :lo12:outbuf
    mov     x14, #0                 // output length
    mov     x10, #0
out_loop:
    cmp     x10, x22
    b.hs    out_done
    lsl     x0, x10, #4
    add     x0, x21, x0
    ldp     x1, x2, [x0]
    add     x1, x19, x1
    add     x3, x28, x14
copy_loop:
    cbz     x2, copy_done
    ldrb    w4, [x1], #1
    strb    w4, [x3], #1
    sub     x2, x2, #1
    b       copy_loop
copy_done:
    mov     w4, #10
    strb    w4, [x3], #1
    sub     x14, x3, x28
    add     x10, x10, #1
    b       out_loop
out_done:

    // ---- write it all out ----
    mov     x15, #0
write_loop:
    cmp     x15, x14
    b.hs    do_exit
    mov     x0, #1
    add     x1, x28, x15
    sub     x2, x14, x15
    mov     x8, #64
    svc     #0
    cmp     x0, #1
    b.lt    do_exit
    add     x15, x15, x0
    b       write_loop

do_exit:
    mov     x0, #0
    mov     x8, #93
    svc     #0

// x0 = entry A, x1 = entry B -> x0 negative/zero/positive
cmp_lines:
    ldp     x2, x3, [x0]
    ldp     x4, x5, [x1]
    add     x2, x19, x2
    add     x4, x19, x4
    cmp     x3, x5
    csel    x6, x3, x5, lo
    mov     x7, #0
1:  cmp     x7, x6
    b.hs    2f
    ldrb    w8, [x2, x7]
    ldrb    w9, [x4, x7]
    add     x7, x7, #1
    cmp     w8, w9
    b.eq    1b
    sub     x0, x8, x9
    ret
2:  sub     x0, x3, x5
    ret
