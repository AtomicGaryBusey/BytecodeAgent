.globl _start

.equ INSZ, 32784
.equ MAXL, 2048

.text
_start:
    // ---- read all of stdin into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
1:
    mov     x2, #INSZ
    sub     x2, x2, x20
    cbz     x2, 2f
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x20, x20, x0
    b       1b
2:
    // ---- split into lines ----
    adrp    x22, lines
    add     x22, x22, :lo12:lines
    mov     x21, #0                 // line count
    mov     x23, #0                 // start of current line
    mov     x24, #0                 // scan index
3:
    cmp     x24, x20
    b.hs    5f
    ldrb    w9, [x19, x24]
    add     x24, x24, #1
    cmp     w9, #10
    b.ne    3b
    mov     x13, #MAXL
    cmp     x21, x13
    b.hs    6f
    add     x10, x19, x23
    sub     x11, x24, #1
    sub     x11, x11, x23
    add     x12, x22, x21, lsl #4
    stp     x10, x11, [x12]
    add     x21, x21, #1
    mov     x23, x24
    b       3b
5:
    cmp     x23, x20
    b.hs    6f
    mov     x13, #MAXL
    cmp     x21, x13
    b.hs    6f
    add     x10, x19, x23
    sub     x11, x20, x23
    add     x12, x22, x21, lsl #4
    stp     x10, x11, [x12]
    add     x21, x21, #1
6:
    // ---- bottom-up merge sort ----
    mov     x19, x22                // src array
    adrp    x20, tmparr
    add     x20, x20, :lo12:tmparr  // dst array
    cmp     x21, #2
    b.lo    ms_done
    mov     x22, #1                 // width
ms_outer:
    cmp     x22, x21
    b.hs    ms_done
    mov     x23, #0                 // block start
ms_block:
    cmp     x23, x21
    b.hs    ms_blockdone
    add     x27, x23, x22
    cmp     x27, x21
    csel    x27, x27, x21, ls       // mid
    add     x28, x23, x22, lsl #1
    cmp     x28, x21
    csel    x28, x28, x21, ls       // hi
    mov     x24, x23                // a
    mov     x25, x27                // b
    mov     x26, x23                // out
ms_merge:
    cmp     x24, x27
    b.hs    ms_tail_b
    cmp     x25, x28
    b.hs    ms_tail_a
    add     x9, x19, x24, lsl #4
    ldp     x11, x12, [x9]
    add     x10, x19, x25, lsl #4
    ldp     x13, x14, [x10]
    mov     x0, x11
    mov     x1, x12
    mov     x2, x13
    mov     x3, x14
    bl      cmp_le
    add     x15, x20, x26, lsl #4
    cbz     w0, 1f
    stp     x11, x12, [x15]
    add     x24, x24, #1
    b       2f
1:
    stp     x13, x14, [x15]
    add     x25, x25, #1
2:
    add     x26, x26, #1
    b       ms_merge
ms_tail_a:
    cmp     x24, x27
    b.hs    ms_blockend
    add     x9, x19, x24, lsl #4
    ldp     x11, x12, [x9]
    add     x15, x20, x26, lsl #4
    stp     x11, x12, [x15]
    add     x24, x24, #1
    add     x26, x26, #1
    b       ms_tail_a
ms_tail_b:
    cmp     x25, x28
    b.hs    ms_blockend
    add     x9, x19, x25, lsl #4
    ldp     x11, x12, [x9]
    add     x15, x20, x26, lsl #4
    stp     x11, x12, [x15]
    add     x25, x25, #1
    add     x26, x26, #1
    b       ms_tail_b
ms_blockend:
    add     x23, x23, x22, lsl #1
    b       ms_block
ms_blockdone:
    mov     x9, x19
    mov     x19, x20
    mov     x20, x9
    lsl     x22, x22, #1
    b       ms_outer
ms_done:
    // ---- build output buffer ----
    adrp    x22, outbuf
    add     x22, x22, :lo12:outbuf
    mov     x23, #0                 // out length
    mov     x24, #0                 // line index
1:
    cmp     x24, x21
    b.hs    4f
    add     x9, x19, x24, lsl #4
    ldp     x10, x11, [x9]          // ptr, len
    mov     x12, #0
2:
    cmp     x12, x11
    b.hs    3f
    ldrb    w13, [x10, x12]
    strb    w13, [x22, x23]
    add     x12, x12, #1
    add     x23, x23, #1
    b       2b
3:
    mov     w13, #10
    strb    w13, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    b       1b
4:
    // ---- write output ----
    mov     x19, x22
    mov     x20, x23
5:
    cbz     x20, 6f
    mov     x0, #1
    mov     x1, x19
    mov     x2, x20
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    6f
    add     x19, x19, x0
    sub     x20, x20, x0
    b       5b
6:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       6b

// cmp_le: x0=p1 x1=l1 x2=p2 x3=l2 -> w0 = 1 if line1 <= line2
cmp_le:
    mov     x4, x1
    cmp     x4, x3
    csel    x4, x4, x3, ls
    mov     x5, #0
1:
    cmp     x5, x4
    b.hs    2f
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    cmp     w6, w7
    b.ne    3f
    add     x5, x5, #1
    b       1b
2:
    cmp     x1, x3
    cset    w0, ls
    ret
3:
    cset    w0, lo
    ret

.bss
.balign 16
inbuf:
    .zero 32784
.balign 16
lines:
    .zero 32768
.balign 16
tmparr:
    .zero 32768
.balign 16
outbuf:
    .zero 34848
