.globl _start

.text
_start:
    xor     r12, r12                    # total bytes read

read_loop:
    lea     rsi, [rip + buf]
    add     rsi, r12
    mov     rdx, 32768
    sub     rdx, r12
    test    rdx, rdx
    jz      read_done
    xor     eax, eax                    # read
    xor     edi, edi                    # fd 0
    syscall
    test    rax, rax
    jle     read_done
    add     r12, rax
    jmp     read_loop

read_done:
    # ---- split into lines ----
    lea     r15, [rip + buf]            # input base
    lea     r13, [rip + lines]          # cursor in line table
    xor     r14, r14                    # line count
    xor     rbx, rbx                    # i

split_loop:
    cmp     rbx, r12
    jae     split_done
    mov     rcx, rbx                    # start of line

scan_loop:
    cmp     rbx, r12
    jae     store_line
    cmp     byte ptr [r15 + rbx], 10
    je      store_nl
    inc     rbx
    jmp     scan_loop

store_nl:
    mov     rax, rcx
    add     rax, r15
    mov     qword ptr [r13], rax
    mov     rdx, rbx
    sub     rdx, rcx
    mov     qword ptr [r13 + 8], rdx
    add     r13, 16
    inc     r14
    inc     rbx
    jmp     split_loop

store_line:
    mov     rax, rcx
    add     rax, r15
    mov     qword ptr [r13], rax
    mov     rdx, rbx
    sub     rdx, rcx
    mov     qword ptr [r13 + 8], rdx
    add     r13, 16
    inc     r14
    jmp     split_loop

split_done:
    # ---- bottom-up merge sort ----
    mov     rbp, r14                    # n
    cmp     rbp, 1
    jbe     output
    mov     r12, 1                      # width

pass_loop:
    cmp     r12, rbp
    jae     output
    lea     r13, [rip + lines]
    lea     r15, [rip + tmp]
    xor     rbx, rbx                    # lo

outer_loop:
    cmp     rbx, rbp
    jae     pass_end
    mov     rsi, rbx
    add     rsi, r12                    # mid
    cmp     rsi, rbp
    jbe     1f
    mov     rsi, rbp
1:
    lea     rdi, [rbx + r12*2]          # hi
    cmp     rdi, rbp
    jbe     1f
    mov     rdi, rbp
1:
    mov     rax, rbx                    # ia
    mov     rcx, rsi                    # ib
    mov     rdx, rbx                    # out

merge_loop:
    cmp     rax, rsi
    jae     take_b
    cmp     rcx, rdi
    jae     take_a
    lea     r8, [r13 + rax*16]
    l
