.globl _start

.text
_start:
    xor     r12d, r12d                  # total bytes read

read_loop:
    mov     rdx, 32768
    sub     rdx, r12
    jz      read_done
    lea     rsi, [rip + buf]
    add     rsi, r12
    xor     eax, eax                    # read
    xor     edi, edi                    # fd 0
    syscall
    test    rax, rax
    jle     read_done
    add     r12, rax
    jmp     read_loop

read_done:
    # ---- split input into lines ----
    lea     r15, [rip + buf]            # input base
    xor     r14d, r14d                  # line count
    xor     ebx, ebx                    # byte index

split_loop:
    cmp     rbx, r12
    jae     split_done
    mov     r8, rbx                     # start index of this line

scan_loop:
    cmp     rbx, r12
    jae     store_line
    cmp     byte ptr [r15 + rbx], 10
    je      store_nl
    inc     rbx
    jmp     scan_loop

store_nl:
    lea     r9, [r15 + r8]
    mov     qword ptr [ptrs + r14*8], r9
    mov     r10, rbx
    sub     r10, r8
    mov     qword ptr [lens + r14*8], r10
    inc     r14
    inc     rbx
    jmp     split_loop

store_line:
    lea     r9, [r15 + r8]
    mov     qword ptr [ptrs + r14*8], r9
    mov     r10, rbx
    sub     r10, r8
    mov     qword ptr [lens + r14*8], r10
    inc     r14
    jmp     split_loop

split_done:
    mov     rbp, r14                    # n = number of lines
    cmp     rbp, 1
    jbe     output
    mov     r12d, 1                     # width

pass_loop:
    cmp     r12, rbp
    jae     output
    xor     ebx, ebx                    # lo

outer_loop:
    cmp     rbx, rbp
    jae     pass_end
    mov     rsi, rbx
    add     rsi, r12                    # mid
    cmp     rsi, rbp
    jbe     1f
    mov     rsi, rbp
1:
    mov     rdi, rbx
    add     rdi, r12
    add     rdi, r12                    # hi
    cmp     rdi, rbp
    jbe     1f
    mov     rdi, rbp
1:
    mov     rax, rbx                    # ia
    mov     rcx, rsi                    # ib
    mov     rdx, rbx                    # out index

merge_loop:
    cmp     rax, rsi
    jae     tb_check
    cmp     rcx, rdi
    jae     take_a
    # ---- compare line rax with line rcx ----
    mov     r8, qword ptr [ptrs + rax*8]
    mov     r9, qword ptr [lens + rax*8]
    mov     r10, qword ptr [ptrs + rcx*8]
    mov     r11, qword ptr [lens + rcx*8]
    mov     r14, r9
    cmp     r11, r14
    jae     1f
    mov     r14, r11
1:
    add     r14, r8                     # end pointer for A over min length

cmp_loop:
    cmp     r8, r14
    jae     cmp_prefix
    movzx   r15d, byte ptr [r8]
    cmp     r15b, byte ptr [r10]
    jne     cmp_diff
    inc     r8
    inc     r10
    jmp     cmp_loop

cmp_diff:
    jb      take_a
    jmp     take_b

cmp_prefix:
    cmp     r9, r11
    jbe     take_a
    jmp     take_b

tb_check:
    cmp     rcx, rdi
    jae     merge_done
    jmp     take_b

take_a:
    mov     r8, qword ptr [ptrs + rax*8]
    mov     r9, qword ptr [lens + rax*8]
    mov     qword ptr [tptrs + rdx*8], r8
    mov     qword ptr [tlens + rdx*8], r9
    inc     rax
    inc     rdx
    jmp     merge_loop

take_b:
    mov     r8, qword ptr [ptrs + rcx*8]
    mov     r9, qword ptr [lens + rcx*8]
    mov     qword ptr [tptrs + rdx*8], r8
    mov     qword ptr [tlens + rdx*8], r9
    inc     rcx
    inc     rdx
    jmp     merge_loop

merge_done:
    add     rbx, r12
    add     rbx, r12
    jmp     outer_loop

pass_end:
    xor     eax, eax
copy_back:
    cmp     rax, rbp
    jae     copy_done
    mov     r8, qword ptr [tptrs + rax*8]
    mov     r9, qword ptr [tlens + rax*8]
    mov     qword ptr [ptrs + rax*8], r8
    mov     qword ptr [lens + rax*8], r9
    inc     rax
    jmp     copy_back
copy_done:
    add     r12, r12
    jmp     pass_loop

output:
    lea     r13, [rip + outbuf]
    xor     ebx, ebx

out_loop:
    cmp     rbx, rbp
    jae     out_write
    mov     rsi, qword ptr [ptrs + rbx*8]
    mov     rdx, qword ptr [lens + rbx*8]
copy_bytes:
    test    rdx, rdx
    jz      1f
    movzx   eax, byte ptr [rsi]
    mov     byte ptr [r13], al
    inc     rsi
    inc     r13
    dec     rdx
    jmp     copy_bytes
1:
    mov     byte ptr [r13], 10
    inc     r13
    inc     rbx
    jmp     out_loop

out_write:
    lea     rsi, [rip + outbuf]
    mov     rdx, r13
    sub     rdx, rsi

write_loop:
    test    rdx, rdx
    jz      finish
    mov     eax, 1                      # write
    mov     edi, 1                      # fd 1
    syscall
    test    rax, rax
    jle     finish
    add     rsi, rax
    sub     rdx, rax
    jmp     write_loop

finish:
    mov     eax, 231                    # exit_group
    xor     edi, edi
    syscall
    mov     eax, 60                     # exit
    xor     edi, edi
    syscall

.bss
.balign 16
buf:
    .zero 32768
.balign 16
ptrs:
    .zero 8192
.balign 16
lens:
    .zero 8192
.balign 16
tptrs:
    .zero 8192
.balign 16
tlens:
    .zero 8192
.balign 16
outbuf:
    .zero 34816
