.globl _start

.text
_start:
    // ---- read all of stdin into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x10, #0                 // total bytes read
1:
    mov     x2, #65536
    subs    x2, x2, x10
    b.le    2f
    mov     x0, #0
    add     x1, x19, x10
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x10, x10, x0
    b       1b
2:
    // ---- find first '!' ----
    mov     x21, #0
3:
    cmp     x21, x10
    b.eq    4f
    ldrb    w0, [x19, x21]
    cmp     w0, #0x21
    b.eq    5f
    add     x21, x21, #1
    b       3b
4:
    mov     x9, x10                 // prog_len = total
    mov     x11, x10                // input offset = total (empty)
    b       6f
5:
    mov     x9, x21                 // prog_len = index of '!'
    add     x11, x21, #1            // input offset
6:
    mov     x20, x9                 // x20 = prog_len
    add     x24, x19, x11           // x24 = input pointer
    add     x25, x19, x10           // x25 = input end

    // ---- bracket check + jump table ----
    adrp    x23, jtab
    add     x23, x23, :lo12:jtab
    adrp    x11, bstack
    add     x11, x11, :lo12:bstack
    mov     x12, #0                 // stack depth
    mov     x21, #0
7:
    cmp     x21, x20
    b.eq    10f
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5b
    b.eq    8f
    cmp     w0, #0x5d
    b.eq    9f
    add     x21, x21, #1
    b       7b
8:
    str     w21, [x11, x12, lsl #2]
    add     x12, x12, #1
    add     x21, x21, #1
    b       7b
9:
    cbz     x12, 20f
    sub     x12, x12, #1
    ldr     w13, [x11, x12, lsl #2]
    str     w21, [x23, x13, lsl #2]
    str     w13, [x23, x21, lsl #2]
    add     x21, x21, #1
    b       7b
10:
    cbnz    x12, 20f

    // ---- run ----
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x26, obuf
    add     x26, x26, :lo12:obuf
    mov     x27, #0                 // output count
    mov     x21, #0                 // pc
run:
    cmp     x21, x20
    b.hs    done
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #0x2b
    b.eq    c_plus
    cmp     w0, #0x2d
    b.eq    c_minus
    cmp     w0, #0x3e
    b.eq    c_right
    cmp     w0, #0x3c
    b.eq    c_left
    cmp     w0, #0x2e
    b.eq    c_dot
    cmp     w0, #0x5b
    b.eq    c_lb
    cmp     w0, #0x5d
    b.eq    c_rb
    cmp     w0, #0x2c
    b.eq    c_comma
    b       run

c_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       run
c_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       run
c_right:
    add     x22, x22, #1
    b       run
c_left:
    sub     x22, x22, #1
    b       run
c_dot:
    ldrb    w1, [x22]
    strb    w1, [x26, x27]
    add     x27, x27, #1
    cmp     x27, #4096
    b.lo    run
    mov     x0, #1
    mov     x1, x26
    mov     x2, x27
    mov     x8, #64
    svc     #0
    mov     x27, #0
    b       run
c_comma:
    cmp     x24, x25
    b.hs    11f
    ldrb    w1, [x24], #1
    strb    w1, [x22]
    b       run
11:
    strb    wzr, [x22]
    b       run
c_lb:
    ldrb    w1, [x22]
    cbnz    w1, run
    sub     x9, x21, #1
    ldr     w2, [x23, x9, lsl #2]
    add     x21, x2, #1
    b       run
c_rb:
    ldrb    w1, [x22]
    cbz     w1, run
    sub     x9, x21, #1
    ldr     w2, [x23, x9, lsl #2]
    add     x21, x2, #1
    b       run

done:
    cbz     x27, 12f
    mov     x0, #1
    mov     x1, x26
    mov     x2, x27
    mov     x8, #64
    svc     #0
12:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       12b

20:
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       20b

.bss
.balign 16
inbuf:
    .zero 65552
.balign 16
tape:
    .zero 30016
.balign 16
jtab:
    .zero 262144
.balign 16
bstack:
    .zero 16384
.balign 16
obuf:
    .zero 4096
