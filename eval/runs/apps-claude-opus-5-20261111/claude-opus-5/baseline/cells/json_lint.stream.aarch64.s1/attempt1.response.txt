    add     x19, x19, :lo12:buf

    sub     x2, x2, x20

    bl      skip_ws

    add     x1, x1, :lo12:s_ok
    mov     x2, #3
    mov     x0, #1
    mov     x8, #64

    mov     x0, #1

    mov     x0, #1

    mov     x0, #1
    mov     x8, #93
    svc     #0

// ---- skip whitespace (leaf) ----

    ldrb    w0, [x19, x21]
    cmp     w0, #0x20
    b.eq    sw_adv
    cmp     w0, #0x09
    b.eq    sw_adv
    cmp     w0, #0x0a
    b.eq    sw_adv
    cmp     w0, #0x0d
    b.ne    sw_done

    add     x21, x21, #1
    b       skip_ws

    ret

// ---- match literal: x0 = asciz pointer (leaf) ----

    ldrb    w1, [x0], #1
    cbz     w1, ml_done
    ldrb    w2, [x19, x21]
    cmp     w1, w2
    b.ne    fail
    add     x21, x21, #1
    b       match_lit

    ret

// ---- value ----
parse_value:
    stp     x29, x30, [sp, #-16]!
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7b
    b.eq    pv_obj
    cmp     w0, #0x5b
    b.eq    pv_arr
    cmp     w0, #0x22
    b.eq    pv_str
    cmp     w0, #0x74
    b.eq    pv_true
    cmp     w0, #0x66
    b.eq    pv_false
    cmp     w0, #0x6e
    b.eq    pv_null
    cmp     w0, #0x2d
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #10
    b.lo    pv_num
    b       fail

    bl      parse_object
    b       pv_ret
pv_arr:
    bl      parse_array
    b       pv_ret
pv_str:
    bl      parse_string
    b       pv_ret
pv_num:
    bl      parse_number
    b       pv_ret

    add     x0, x0, :lo12:lit_true

pv_false:

    add     x0, x0, :lo12:lit_false
    bl      match_lit
    b       pv_ret

    add     x0, x0, :lo12:lit_null
    bl      match_lit

    ldp     x29, x30, [sp], #16
    ret

// ---- string (leaf), buf[pos] == '"' ----

    ldrb    w0, [x19, x21]
    cmp     w0, #0x22
    b.eq    ps_end

    b       ps_loop

    ret

    ldrb    w0, [x19, x21]
    cmp     w0, #0x75
    b.eq    ps_hex
    cmp     w0, #0x22
    b.eq    ps_e1
    cmp     w0, #0x5c
    b.eq    ps_e1
    cmp     w0, #0x2f
    b.eq    ps_e1
    cmp     w0, #0x62
    b.eq    ps_e1
    cmp     w0, #0x66
    b.eq    ps_e1
    cmp     w0, #0x6e
    b.eq    ps_e1
    cmp     w0, #0x72
    b.eq    ps_e1
    cmp     w0, #0x74
