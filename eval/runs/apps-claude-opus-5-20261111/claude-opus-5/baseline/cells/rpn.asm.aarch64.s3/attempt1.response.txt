.globl _start

.equ INBUF_SIZE, 65536
.equ OUT_LIMIT, 65536

.text

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x22, x23
    adrp    x24, vstack
    add     x24, x24, :lo12:vstack

    mov     x20, #0
rd_loop:
    mov     x2, #INBUF_SIZE
    subs    x2, x2, x20
    b.le    rd_done
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    rd_done
    add     x20, x20, x0
    b       rd_loop
rd_done:

    mov     x21, x19
    add     x25, x19, x20
main_loop:
    cmp     x21, x25
    b.hs    main_done
    mov     x26, x21
1:  cmp     x26, x25
    b.hs    2f
    ldrb    w0, [x26]
    cmp     w0, #10
    b.eq    2f
    add     x26, x26, #1
    b       1b
2:  mov     x0, x21
    mov     x1, x26
    bl      proc_line
    add     x21, x26, #1
    b       main_loop
main_done:
    bl      flush
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       main_done


// proc_line(x0 = line start, x1 = line end)
proc_line:
    stp     x29, x30, [sp, #-16]!
    mov     x27, x0
    mov     x28, x1
    mov     x9, #0
tok_loop:
1:  cmp     x27, x28
    b.hs    tok_end
    ldrb    w0, [x27]
    cmp     w0, #32
    b.eq    2f
    cmp     w0, #9
    b.ne    3f
2:  add     x27, x27, #1
    b       1b
3:  mov     x10, x27
4:  cmp     x27, x28
    b.hs    5f
    ldrb    w0, [x27]
    cmp     w0, #32
    b.eq    5f
    cmp     w0, #9
    b.eq    5f
    add     x27, x27, #1
    b       4b
5:  mov     x11, x27
    sub     x12, x11, x10
    cmp     x12, #1
    b.ne    lit_parse
    ldrb    w0, [x10]
    cmp     w0, #43                 // '+'
    b.eq    do_op
    cmp     w0, #45                 // '-'
    b.eq    do_op
    cmp     w0, #42                 // '*'
    b.eq    do_op
    cmp     w0, #47                 // '/'
    b.eq    do_op
    cmp     w0, #37                 // '%'
    b.eq    do_op
    b       lit_parse

do_op:
    cmp     x9, #2
    b.lt    line_error
    sub     x9, x9, #2
    ldr     x2, [x24, x9, lsl #3]
    add     x3, x9, #1
    ldr     x3, [x24, x3, lsl #3]
    cmp     w0, #43                 // '+'
    b.eq    op_add
    cmp     w0, #45                 // '-'
    b.eq    op_sub
    cmp     w0, #42                 // '*'
    b.eq    op_mul
    cmp     w0, #47                 // '/'
    b.eq    op_div
    b       op_mod

op_add:
    add     x5, x2, x3
    b       push_val
op_sub:
    sub     x5, x2, x3
    b       push_val
op_mul:
    mul     x5, x2, x3
    b       push_val
op_div:
    cbz     x3, line_error
    cmn     x3, #1
    b.ne    1f
    mov     x4, #1
    lsl     x4, x4, #63
    cmp     x2, x4
    b.eq    line_error
1:  sdiv    x5, x2, x3
    b       push_val
op_mod:
    cbz     x3, line_error
    cmn     x3, #1
    b.ne    1f
    mov     x4, #1
    lsl     x4, x4, #63
    cmp     x2, x4
    b.eq    line_error
1:  sdiv    x5, x2, x3
    msub    x5, x5, x3, x2
    b       push_val

push_val:
    str     x5, [x24, x9, lsl #3]
    add     x9, x9, #1
    b       tok_loop

lit_parse:
    mov     x13, x10
    mov     x14, #0
    ldrb    w0, [x13]
    cmp     w0, #45                 // '-'
    b.ne    1f
    mov     x14, #1
    add     x13, x13, #1
1:  sub     x12, x11, x13
    cmp     x12, #1
    b.lt    line_error
    cmp     x12, #18
    b.gt    line_error
    mov     x5, #0
    mov     x15, #10
2:  ldrb    w0, [x13], #1
    sub     w0, w0, #48
    cmp     w0, #9
    b.hi    line_error
    madd    x5, x5, x15, x0
    cmp     x13, x11
    b.lo    2b
    cbz     x14, 3f
    neg     x5, x5
3:  str     x5, [x24, x9, lsl #3]
    add     x9, x9, #1
    b       tok_loop

tok_end:
    cmp     x9, #1
    b.ne    line_error
    ldr     x0, [x24]
    bl      print_num
    ldp     x29, x30, [sp], #16
    ret
line_error:
    bl      print_error
    ldp     x29, x30, [sp], #16
    ret


// flush: write out pending output bytes (leaf, clobbers x0,x1,x2,x8)
flush:
    sub     x2, x22, x23
    mov     x1, x23
1:  cbz     x2, 2f
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #1
    b.lt    2f
    add     x1, x1, x0
    sub     x2, x2, x0
    b       1b
2:  mov     x22, x23
    ret


print_error:
    stp     x29, x30, [sp, #-16]!
    sub     x0, x22, x23
    mov     x1, #OUT_LIMIT
    cmp     x0, x1
    b.lo    1f
    bl      flush
1:  mov     w0, #0x65               // 'e'
    strb    w0, [x22], #1
    mov     w0, #0x72               // 'r'
    strb    w0, [x22], #1
    strb    w0, [x22], #1
    mov     w0, #0x6f               // 'o'
    strb    w0, [x22], #1
    mov     w0, #0x72               // 'r'
    strb    w0, [x22], #1
    mov     w0, #10
    strb    w0, [x22], #1
    ldp     x29, x30, [sp], #16
    ret


// print_num(x0 = signed value)
print_num:
    stp     x29, x30, [sp, #-16]!
    mov     x16, x0
    sub     x0, x22, x23
    mov     x1, #OUT_LIMIT
    cmp     x0, x1
    b.lo    1f
    bl      flush
1:  mov     x17, #0
    tbz     x16, #63, 2f
    mov     x17, #1
    neg     x16, x16
2:  sub     sp, sp, #32
    add     x3, sp, #32
    mov     x4, #10
3:  udiv    x5, x16, x4
    msub    x6, x5, x4, x16
    add     x6, x6, #48
    sub     x3, x3, #1
    strb    w6, [x3]
    mov     x16, x5
    cbnz    x16, 3b
    cbz     x17, 4f
    mov     w6, #0x2d               // '-'
    sub     x3, x3, #1
    strb    w6, [x3]
4:  add     x7, sp, #32
5:  ldrb    w6, [x3], #1
    strb    w6, [x22], #1
    cmp     x3, x7
    b.lo    5b
    mov     w6, #10
    strb    w6, [x22], #1
    add     sp, sp, #32
    ldp     x29, x30, [sp], #16
    ret


.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 65600
.balign 16

    .zero 262160
