.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
    .balign 16
buf:
    .zero 131072
outbuf:
    .zero 64

.text

// x21 = buffer base, x20 = length, x19 = current offset
_start:
    adrp x21, buf
    add  x21, x21, :lo12:buf
    mov  x20, #0

read_loop:
    movz x2, #2, lsl #16
    sub  x2, x2, x20
    cbz  x2, read_done
    mov  x0, #0
    add  x1, x21, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    mov  x19, #0
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.ne fail

    // print "ok\n", exit 0
    adrp x1, msg_ok
    add  x1, x1, :lo12:msg_ok
    mov  x0, #1
    mov  x2, #3
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0

fail:
    // write "error at "
    adrp x1, msg_err
    add  x1, x1, :lo12:msg_err
    mov  x0, #1
    mov  x2, #9
    mov  x8, #64
    svc  #0
    // decimal of x19 into outbuf, newline at outbuf+31
    adrp x3, outbuf
    add  x3, x3, :lo12:outbuf
    add  x4, x3, #31
    mov  w5, #10
    strb w5, [x4]
    mov  x6, x4
    mov  x1, x19
    mov  x7, #10
1:  udiv x2, x1, x7
    msub x5, x2, x7, x1
    add  w5, w5, #0x30
    sub  x6, x6, #1
    strb w5, [x6]
    mov  x1, x2
    cbnz x1, 1b
    add  x2, x3, #32
    sub  x2, x2, x6
    mov  x1, x6
    mov  x0, #1
    mov  x8, #64
    svc  #0
    mov  x0, #1
    mov  x8, #93
    svc  #0

// ---------------------------------------------------------------
// skip_ws: advance x19 over space, tab, lf, cr
skip_ws:
1:  cmp  x19, x20
    b.hs 2f
    ldrb w0, [x21, x19]
    cmp  w0, #0x20
    b.eq 3f
    cmp  w0, #0x09
    b.eq 3f
    cmp  w0, #0x0a
    b.eq 3f
    cmp  w0, #0x0d
    b.ne 2f
3:  add  x19, x19, #1
    b    1b
2:  ret

// ---------------------------------------------------------------
// match_lit: x0 = pointer to nul-terminated literal
match_lit:
1:  ldrb w1, [x0], #1
    cbz  w1, 2f
    cmp  x19, x20
    b.hs fail
    ldrb w2, [x21, x19]
    cmp  w1, w2
    b.ne fail
    add  x19, x19, #1
    b    1b
2:  ret

// ---------------------------------------------------------------
// parse_string: byte at x19 is known to be '"'
parse_string:
    add  x19, x19, #1
ps_loop:
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x22
    b.eq ps_done
    cmp  w0, #0x5c
    b.eq ps_esc
    cmp  w0, #0x20
    b.lo fail
    add  x19, x19, #1
    b    ps_loop
ps_done:
    add  x19, x19, #1
    ret
ps_esc:
    add  x19, x19, #1
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x75
    b.eq ps_u
    cmp  w0, #0x22
    b.eq ps_ok1
    cmp  w0, #0x5c
    b.eq ps_ok1
    cmp  w0, #0x2f
    b.eq ps_ok1
    cmp  w0, #0x62
    b.eq ps_ok1
    cmp  w0, #0x66
    b.eq ps_ok1
    cmp  w0, #0x6e
    b.eq ps_ok1
    cmp  w0, #0x72
    b.eq ps_ok1
    cmp  w0, #0x74
    b.eq ps_ok1
    b    fail
ps_ok1:
    add  x19, x19, #1
    b    ps_loop
ps_u:
    add  x19, x19, #1
    mov  w3, #4
ps_uloop:
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.lo ps_uok
    orr  w1, w0, #0x20
    sub  w1, w1, #0x61
    cmp  w1, #6
    b.hs fail
ps_uok:
    add  x19, x19, #1
    subs w3, w3, #1
    b.ne ps_uloop
    b    ps_loop

// ---------------------------------------------------------------
// parse_number: byte at x19 is known to be '-' or a digit
parse_number:
    ldrb w0, [x21, x19]
    cmp  w0, #0x2d
    b.ne pn_int
    add  x19, x19, #1
pn_int:
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x30
    b.eq pn_zero
    sub  w1, w0, #0x31
    cmp  w1, #9
    b.hs fail
    add  x19, x19, #1
pn_digits:
    cmp  x19, x20
    b.hs pn_done
    ldrb w0, [x21, x19]
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.hs pn_frac
    add  x19, x19, #1
    b    pn_digits
pn_zero:
    add  x19, x19, #1
pn_frac:
    cmp  x19, x20
    b.hs pn_done
    ldrb w0, [x21, x19]
    cmp  w0, #0x2e
    b.ne pn_exp
    add  x19, x19, #1
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.hs fail
1:  add  x19, x19, #1
    cmp  x19, x20
    b.hs pn_done
    ldrb w0, [x21, x19]
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.lo 1b
pn_exp:
    cmp  x19, x20
    b.hs pn_done
    ldrb w0, [x21, x19]
    orr  w0, w0, #0x20
    cmp  w0, #0x65
    b.ne pn_done
    add  x19, x19, #1
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x2b
    b.eq 2f
    cmp  w0, #0x2d
    b.ne 3f
2:  add  x19, x19, #1
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
3:  sub  w1, w0, #0x30
    cmp  w1, #10
    b.hs fail
4:  add  x19, x19, #1
    cmp  x19, x20
    b.hs pn_done
    ldrb w0, [x21, x19]
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.lo 4b
pn_done:
    ret

// ---------------------------------------------------------------
parse_value:
    stp  x29, x30, [sp, #-16]!
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x7b
    b.eq pv_obj
    cmp  w0, #0x5b
    b.eq pv_arr
    cmp  w0, #0x22
    b.eq pv_str
    cmp  w0, #0x74
    b.eq pv_true
    cmp  w0, #0x66
    b.eq pv_false
    cmp  w0, #0x6e
    b.eq pv_null
    cmp  w0, #0x2d
    b.eq pv_num
    sub  w1, w0, #0x30
    cmp  w1, #10
    b.lo pv_num
    b    fail

pv_str:
    bl   parse_string
    b    pv_ret
pv_num:
    bl   parse_number
    b    pv_ret
pv_true:
    adrp x0, lit_true
    add  x0, x0, :lo12:lit_true
    bl   match_lit
    b    pv_ret
pv_false:
    adrp x0, lit_false
    add  x0, x0, :lo12:lit_false
    bl   match_lit
    b    pv_ret
pv_null:
    adrp x0, lit_null
    add  x0, x0, :lo12:lit_null
    bl   match_lit
    b    pv_ret

pv_arr:
    add  x19, x19, #1
    bl   skip_ws
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x5d
    b.eq pa_close
pa_elem:
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x5d
    b.eq pa_close
    cmp  w0, #0x2c
    b.ne fail
    add  x19, x19, #1
    bl   skip_ws
    b    pa_elem
pa_close:
    add  x19, x19, #1
    b    pv_ret

pv_obj:
    add  x19, x19, #1
    bl   skip_ws
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x7d
    b.eq po_close
po_member:
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x22
    b.ne fail
    bl   parse_string
    bl   skip_ws
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x3a
    b.ne fail
    add  x19, x19, #1
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.hs fail
    ldrb w0, [x21, x19]
    cmp  w0, #0x7d
    b.eq po_close
    cmp  w0, #0x2c
    b.ne fail
    add  x19, x19, #1
    bl   skip_ws
    b    po_member
po_close:
    add  x19, x19, #1

pv_ret:
    ldp  x29, x30, [sp], #16
    ret
