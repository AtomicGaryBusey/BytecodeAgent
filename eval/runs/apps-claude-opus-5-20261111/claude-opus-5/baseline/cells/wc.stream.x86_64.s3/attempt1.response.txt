.bss
.balign 64
buf:
.zero 65536
outbuf:
.zero 64

.text
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in_word flag (0 = not in word)

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read

    mov rcx, rax            # count
    add r14, rax            # bytes
    lea rsi, [rip + buf]
    lea rbx, [rip + ws_table]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    movzx edx, byte ptr [rbx + rax]
    # edx = 1 if whitespace
    test edx, edx
    jnz is_ws
    # word byte
    test r15, r15
    jnz skip_inc
    inc r13
    mov r15d, 1
    jmp skip_inc
is_ws:
    xor r15d, r15d
skip_inc:
    dec rcx
    jnz scan_loop
    jmp read_loop

done_read:
    # format output
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    dec rdi

    mov rax, r14
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r13
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r12
    call emit

    inc rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall

# emit: write decimal of rax backwards starting at [rdi], leaving rdi one before first digit
emit:
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz 1b
    ret
