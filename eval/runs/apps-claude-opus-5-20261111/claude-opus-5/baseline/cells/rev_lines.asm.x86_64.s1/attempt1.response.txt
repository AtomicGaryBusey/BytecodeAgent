.globl _start

.text
_start:
    lea rsi, [rip + inbuf]
    xor r12, r12                 # total bytes read

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov eax, 0
    xor edi, edi
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13                 # line start index
    xor r14, r14                 # output length

scan_loop:
    cmp r13, r12
    jae flush_tail
    mov rbx, r13                 # find newline
find_nl:
    cmp rbx, r12
    jae no_nl
    lea rsi, [rip + inbuf]
    mov al, byte ptr [rsi + rbx]
    cmp al, 10
    je got_nl
    inc rbx
    jmp find_nl

got_nl:
    # line is [r13, rbx)
    mov rcx, rbx                 # src cursor (exclusive end)
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    add rdi, r14
copy_rev:
    cmp rcx, r13
    jbe copy_done
    dec rcx
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    jmp copy_rev
copy_done:
    mov byte ptr [rdi], 10
    inc r14
    lea r13, [rbx + 1]
    jmp scan_loop

no_nl:
    # final line [r13, r12)
    mov rcx, r12
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    add rdi, r14
copy_rev2:
    cmp rcx, r13
    jbe copy_done2
    dec rcx
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    jmp copy_rev2
copy_done2:
    mov byte ptr [rdi], 10
    inc r14
    mov r13, r12

flush_tail:
    xor rbx, rbx                 # bytes written
write_loop:
    cmp rbx, r14
    jae done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jle done
    add rbx, rax
    jmp write_loop

done:
    mov eax, 60
    xor edi, edi
    syscall
    jmp done

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65552
