.globl _start

.section .rodata
hexdig:
.ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
.zero 65536
.balign 16
outbuf:
.zero 400000

.text

# emit 8 hex digits of eax at [r13], advancing r13
emit8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [rbp + rdx]
    mov byte ptr [r13], dl
    inc r13
    dec ecx
    jnz 1b
    ret

_start:
    lea r14, [rip + inbuf]
    lea r13, [rip + outbuf]
    lea rbp, [rip + hexdig]
    xor ebx, ebx

read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r14
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor r12d, r12d

row_loop:
    cmp r12, rbx
    jae rows_done

    mov eax, r12d
    call emit8

    mov word ptr [r13], 0x2020
    add r13, 2

    mov rax, rbx
    sub rax, r12
    cmp rax, 16
    jbe 1f
    mov eax, 16
1:
    mov r15d, eax

    xor ecx, ecx
hex_loop:
    cmp ecx, 8
    jne 2f
    mov byte ptr [r13], 0x20
    inc r13
2:
    cmp ecx, r15d
    jae pad_slot
    mov rax, r12
    add rax, rcx
    movzx eax, byte ptr [r14 + rax]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rbp + rdx]
    mov byte ptr [r13], dl
    and eax, 15
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [r13 + 1], al
    mov byte ptr [r13 + 2], 0x20
    add r13, 3
    jmp 3f
pad_slot:
    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x20
    mov byte ptr [r13 + 2], 0x20
    add r13, 3
3:
    inc ecx
    cmp ecx, 16
    jb hex_loop

    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x7c
    add r13, 2

    xor ecx, ecx
ch_loop:
    cmp ecx, r15d
    jae ch_done
    mov rax, r12
    add rax, rcx
    movzx eax, byte ptr [r14 + rax]
    cmp al, 0x20
    jb 4f
    cmp al, 0x7e
    jbe 5f
4:
    mov al, 0x2e
5:
    mov byte ptr [r13], al
    inc r13
    inc ecx
    jmp ch_loop
ch_done:
    mov byte ptr [r13], 0x7c
    mov byte ptr [r13 + 1], 10
    add r13, 2

    add r12, 16
    jmp row_loop

rows_done:
    mov eax, ebx
    call emit8
    mov byte ptr [r13], 10
    inc r13

    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall
