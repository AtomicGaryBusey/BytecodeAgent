.globl _start

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x20, #0

read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    mov     x22, #0
    mov     x23, #0

scan_loop:
    cmp     x22, x20
    b.ge    tail
    ldrb    w0, [x19, x22]
    cmp     w0, #10
    b.ne    scan_next
    bl      emit
    add     x23, x22, #1
scan_next:
    add     x22, x22, #1
    b       scan_loop

tail:
    cmp     x23, x20
    b.eq    done
    bl      emit

done:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       done

// emit: reverse bytes inbuf[x23 .. x22) into outbuf, append newline, write
emit:
    sub     x3, x22, x23
    add     x4, x19, x22
    mov     x5, #0
1:
    cmp     x5, x3
    b.ge    2f
    ldrb    w0, [x4, #-1]!
    strb    w0, [x21, x5]
    add     x5, x5, #1
    b       1b
2:
    mov     w0, #10
    strb    w0, [x21, x3]
    add     x2, x3, #1
    mov     x0, #1
    mov     x1, x21
    mov     x8, #64
    svc     #0
    ret

.bss
.balign 16
inbuf:

outbuf:
