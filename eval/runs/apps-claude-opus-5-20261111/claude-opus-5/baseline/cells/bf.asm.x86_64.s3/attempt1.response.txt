.globl _start

.text
_start:
    xor     r8d, r8d                  # total bytes read
rdloop:
    mov     edx, 65536
    sub     rdx, r8
    jz      rddone
    lea     rsi, [rip + inbuf]
    add     rsi, r8
    xor     eax, eax                  # read
    xor     edi, edi                  # fd 0
    syscall
    cmp     rax, 0
    jle     rddone
    add     r8, rax
    jmp     rdloop
rddone:

    lea     rdi, [rip + inbuf]        # input base
    xor     ecx, ecx
findloop:
    cmp     rcx, r8
    je      nobang
    cmp     byte ptr [rdi + rcx], 0x21
    je      foundbang
    inc     rcx
    jmp     findloop
foundbang:
    mov     r12, rcx                  # program length
    lea     r14, [rdi + rcx + 1]      # program input pointer
    lea     r15, [rdi + r8]           # program input end
    jmp     gotprog
nobang:
    mov     r12, r8
    lea     r14, [rdi + r8]
    mov     r15, r14
gotprog:

    # ---- bracket check and jump table ----
    xor     ecx, ecx                  # index
    xor     r9d, r9d                  # stack depth
valloop:
    cmp     rcx, r12
    je      valend
    movzx   eax, byte ptr [rdi + rcx]
    cmp     al, 91
    je      vpush
    cmp     al, 93
    je      vpop
    inc     rcx
    jmp     valloop
vpush:
    mov     dword ptr [bstack + r9*4], ecx
    inc     r9
    inc     rcx
    jmp     valloop
vpop:
    test    r9, r9
    jz      err
    dec     r9
    mov     edx, dword ptr [bstack + r9*4]
    lea     eax, [rcx + 1]
    mov     dword ptr [jumptab + rdx*4], eax
    lea     eax, [rdx + 1]
    mov     dword ptr [jumptab + rcx*4], eax
    inc     rcx
    jmp     valloop
valend:
    test    r9, r9
    jnz     err

    # ---- build dispatch table ----
    lea     rbp, [rip + disptab]
    lea     rax, [rip + h_def]
    mov     ecx, 256
1:
    mov     qword ptr [rbp + rcx*8 - 8], rax
    dec     rcx
    jnz     1b
    lea     rax, [rip + h_plus]
    mov     qword ptr [rbp + 43*8], rax
    lea     rax, [rip + h_comma]
    mov     qword ptr [rbp + 44*8], rax
    lea     rax, [rip + h_minus]
    mov     qword ptr [rbp + 45*8], rax
    lea     rax, [rip + h_dot]
    mov     qword ptr [rbp + 46*8], rax
    lea     rax, [rip + h_left]
    mov     qword ptr [rbp + 60*8], rax
    lea     rax, [rip + h_right]
    mov     qword ptr [rbp + 62*8], rax
    lea     rax, [rip + h_open]
    mov     qword ptr [rbp + 91*8], rax
    lea     rax, [rip + h_close]
    mov     qword ptr [rbp + 93*8], rax

    # ---- run ----
    lea     r8, [rip + inbuf]         # program base
    lea     rbx, [r8]                 # program counter
    add     r12, r8                   # program end pointer
    lea     r13, [rip + tape]         # data pointer
    xor     r10d, r10d                # output buffer fill

run:
    cmp     rbx, r12
    jae     done
    movzx   eax, byte ptr [rbx]
    inc     rbx
    jmp     qword ptr [rbp + rax*8]

h_right:
    inc     r13
    jmp     run
h_left:
    dec     r13
    jmp     run
h_plus:
    inc     byte ptr [r13]
    jmp     run
h_minus:
    dec     byte ptr [r13]
    jmp     run
h_dot:
    movzx   eax, byte ptr [r13]
    lea     rdi, [rip + outbuf]
    mov     byte ptr [rdi + r10], al
    inc     r10
    cmp     r10, 65536
    jb      run
    call    flushout
    jmp     run
h_comma:
    cmp     r14, r15
    jae     1f
    movzx   eax, byte ptr [r14]
    inc     r14
    mov     byte ptr [r13], al
    jmp     run
1:
    mov     byte ptr [r13], 0
    jmp     run
h_open:
    cmp     byte ptr [r13], 0
    jne     run
    mov     rcx, rbx
    sub     rcx, r8
    dec     rcx
    mov     eax, dword ptr [jumptab + rcx*4]
    lea     rbx, [r8 + rax]
    jmp     run
h_close:
    cmp     byte ptr [r13], 0
    je      run
    mov     rcx, rbx
    sub     rcx, r8
    dec     rcx
    mov     eax, dword ptr [jumptab + rcx*4]
    lea     rbx, [r8 + rax]
    jmp     run
h_def:
    jmp     run

done:
    call    flushout
    mov     eax, 60
    xor     edi, edi
    syscall
    jmp     done

err:
    mov     eax, 60
    mov     edi, 1
    syscall
    jmp     err

flushout:
    test    r10, r10
    jz      2f
1:
    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + outbuf]
    mov     rdx, r10
    syscall
    cmp     rax, 0
    jle     2f
    sub     r10, rax
    jnz     1b
2:
    xor     r10d, r10d
    ret

.bss
.balign 16
inbuf:
    .zero 65544
.balign 16
tape:
    .zero 30016
.balign 16
jumptab:
    .zero 262144
.balign 16
bstack:
    .zero 262144
.balign 16
outbuf:
    .zero 65536
.balign 16
disptab:
    .zero 2048
