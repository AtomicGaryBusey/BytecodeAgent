.globl _start

.text
_start:
    // ---- read all of stdin into buf ----
    adrp x19, buf
    add  x19, x19, :lo12:buf
    mov  x20, #0                  // bytes read so far
    mov  x21, #32768              // capacity
1:
    subs x2, x21, x20
    b.eq 2f
    mov  x0, #0
    add  x1, x19, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le 2f
    add  x20, x20, x0
    b    1b
2:
    // ---- split into lines ----
    adrp x22, starts
    add  x22, x22, :lo12:starts
    adrp x23, lens
    add  x23, x23, :lo12:lens
    mov  x24, #0                  // n lines
    mov  x25, #0                  // current line start
    mov  x26, #0                  // scan position
3:
    cmp  x26, x20
    b.hs 4f
    ldrb w0, [x19, x26]
    add  x26, x26, #1
    cmp  w0, #10
    b.ne 3b
    str  x25, [x22, x24, lsl #3]
    sub  x1, x26, #1
    sub  x1, x1, x25
    str  x1, [x23, x24, lsl #3]
    add  x24, x24, #1
    mov  x25, x26
    b    3b
4:
    cmp  x25, x20
    b.hs 5f
    str  x25, [x22, x24, lsl #3]
    sub  x1, x20, x25
    str  x1, [x23, x24, lsl #3]
    add  x24, x24, #1
5:
    // ---- init index array ----
    adrp x27, idx
    add  x27, x27, :lo12:idx
    adrp x28, tmp
    add  x28, x28, :lo12:tmp
    mov  x0, #0
6:
    cmp  x0, x24
    b.hs 7f
    str  x0, [x27, x0, lsl #3]
    add  x0, x0, #1
    b    6b
7:
    cbz  x24, exit_prog

    // ---- bottom-up merge sort over idx ----
    mov  x9, #1                   // width
pass_loop:
    cmp  x9, x24
    b.hs sorted
    mov  x10, #0                  // lo
seg_loop:
    cmp  x10, x24
    b.hs pass_done
    add  x11, x10, x9             // mid
    cmp  x11, x24
    csel x11, x11, x24, lo
    add  x12, x10, x9, lsl #1     // hi
    cmp  x12, x24
    csel x12, x12, x24, lo
    mov  x13, x10                 // i
    mov  x14, x11                 // j
    mov  x15, x10                 // k
merge_loop:
    cmp  x13, x11
    b.hs take_right_all
    cmp  x14, x12
    b.hs take_left_all
    ldr  x0, [x27, x13, lsl #3]
    ldr  x1, [x27, x14, lsl #3]
    bl   le_cmp
    cbz  x0, take_right
take_left:
    ldr  x0, [x27, x13, lsl #3]
    str  x0, [x28, x15, lsl #3]
    add  x13, x13, #1
    add  x15, x15, #1
    b    merge_loop
take_right:
    ldr  x0, [x27, x14, lsl #3]
    str  x0, [x28, x15, lsl #3]
    add  x14, x14, #1
    add  x15, x15, #1
    b    merge_loop
take_left_all:
    cmp  x13, x11
    b.hs seg_done
    ldr  x0, [x27, x13, lsl #3]
    str  x0, [x28, x15, lsl #3]
    add  x13, x13, #1
    add  x15, x15, #1
    b    take_left_all
take_right_all:
    cmp  x14, x12
    b.hs seg_done
    ldr  x0, [x27, x14, lsl #3]
    str  x0, [x28, x15, lsl #3]
    add  x14, x14, #1
    add  x15, x15, #1
    b    take_right_all
seg_done:
    add  x10, x10, x9, lsl #1
    b    seg_loop
pass_done:
    mov  x0, #0
8:
    cmp  x0, x24
    b.hs 9f
    ldr  x1, [x28, x0, lsl #3]
    str  x1, [x27, x0, lsl #3]
    add  x0, x0, #1
    b    8b
9:
    lsl  x9, x9, #1
    b    pass_loop

sorted:
    // ---- build output ----
    adrp x5, outbuf
    add  x5, x5, :lo12:outbuf
    mov  x6, #0                   // out length
    mov  x7, #0                   // i
o1:
    cmp  x7, x24
    b.hs o3
    ldr  x0, [x27, x7, lsl #3]
    ldr  x1, [x22, x0, lsl #3]
    ldr  x2, [x23, x0, lsl #3]
    add  x1, x19, x1
    mov  x3, #0
o2:
    cmp  x3, x2
    b.hs o2d
    ldrb w4, [x1, x3]
    strb w4, [x5, x6]
    add  x3, x3, #1
    add  x6, x6, #1
    b    o2
o2d:
    mov  w4, #10
    strb w4, [x5, x6]
    add  x6, x6, #1
    add  x7, x7, #1
    b    o1
o3:
    mov  x20, #0
o4:
    cmp  x20, x6
    b.hs exit_prog
    mov  x0, #1
    add  x1, x5, x20
    sub  x2, x6, x20
    mov  x8, #64
    svc  #0
    cmp  x0, #1
    b.lt exit_prog
    add  x20, x20, x0
    b    o4

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0
    b    exit_prog

// le_cmp: x0 = left index, x1 = right index
// returns x0 = 1 if line[left] <= line[right], else 0
// uses x0-x7 only
le_cmp:
    ldr  x2, [x22, x0, lsl #3]
    ldr  x3, [x23, x0, lsl #3]
    ldr  x4, [x22, x1, lsl #3]
    ldr  x5, [x23, x1, lsl #3]
    add  x2, x19, x2
    add  x4, x19, x4
    cmp  x3, x5
    csel x6, x3, x5, lo
    mov  x7, #0
lc1:
    cmp  x7, x6
    b.hs lc2
    ldrb w0, [x2, x7]
    ldrb w1, [x4, x7]
    add  x7, x7, #1
    cmp  w0, w1
    b.eq lc1
    cset w0, lo
    ret
lc2:
    cmp  x3, x5
    cset w0, ls
    ret

.bss
.balign 16
buf:
    .zero 32784
.balign 8
starts:
    .zero 8192
lens:
    .zero 8192
idx:
    .zero 8192
tmp:
    .zero 8192
.balign 16

    .zero 33792
