.globl _start

.text
_start:
    lea     r12, [rip + buf]
    xor     r13, r13

read_loop:
    mov     rsi, r12
    add     rsi, r13
    mov     rdx, BUFSZ
    sub     rdx, r13
    jbe     read_done
    xor     eax, eax
    xor     edi, edi
    syscall
    test    rax, rax
    jle     read_done
    add     r13, rax
    jmp     read_loop

read_done:
    xor     ebx, ebx
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jne     fail

    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + s_ok]
    mov     edx, 3
    syscall
    mov     eax, 231
    xor     edi, edi
    syscall
    jmp     .

# ---------------------------------------------------------------
# skip_ws : advance rbx over space, tab, lf, cr
# ---------------------------------------------------------------
skip_ws:
1:  cmp     rbx, r13
    jae     2f
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 0x20
    je      3f
    cmp     al, 0x09
    je      3f
    cmp     al, 0x0a
    je      3f
    cmp     al, 0x0d
    jne     2f
3:  inc     rbx
    jmp     1b
2:  ret

# ---------------------------------------------------------------
# parse_value
# ---------------------------------------------------------------
parse_value:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 123
    je      parse_object
    cmp     al, 91
    je      parse_array
    cmp     al, 34
    je      parse_string
    cmp     al, 116
    je      pv_true
    cmp     al, 102
    je      pv_false
    cmp     al, 110
    je      pv_null
    cmp     al, 45
    je      parse_number
    cmp     al, 48
    jb      fail
    cmp     al, 57
    jbe     parse_number
    jmp     fail

pv_true:
    lea     rsi, [rip + s_true]
    mov     ecx, 4
    jmp     match_lit
pv_false:
    lea     rsi, [rip + s_false]
    mov     ecx, 5
    jmp     match_lit
pv_null:
    lea     rsi, [rip + s_null]
    mov     ecx, 4
    jmp     match_lit

match_lit:
1:  cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, byte ptr [rsi]
    jne     fail
    inc     rbx
    inc     rsi
    dec     ecx
    jnz     1b
    ret

# ---------------------------------------------------------------
# parse_string : buf[rbx] == '"'
# ---------------------------------------------------------------
parse_string:
    inc     rbx
ps_loop:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 34
    je      ps_end
    cmp     al, 92
    je      ps_esc
    cmp     al, 0x20
    jb      fail
    inc     rbx
    jmp     ps_loop
ps_end:
    inc     rbx
    ret

ps_esc:
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 117
    je      ps_u
    lea     rsi, [rip + esc_tbl]
    mov     ecx, 8
1:  cmp     al, byte ptr [rsi]
    je      2f
    inc     rsi
    dec     ecx
    jnz     1b
    jmp     fail
2:  inc     rbx
    jmp     ps_loop

ps_u:
    inc     rbx
    mov     ecx, 4
1:  cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      fail
    cmp     al, 57
    jbe     2f
    or      al, 0x20
    cmp     al, 97
    jb      fail
    cmp     al, 102
    ja      fail
2:  inc     rbx
    dec     ecx
    jnz     1b
    jmp     ps_loop

# ---------------------------------------------------------------
# parse_number
# ---------------------------------------------------------------
parse_number:
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 45
    jne     pn_int
    inc     rbx
    cmp     rbx, r13
    jae     fail
pn_int:
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      fail
    ja      pn_nonzero
    inc     rbx
    jmp     pn_frac
pn_nonzero:
    cmp     al, 57
    ja      fail
    inc     rbx
pn_d1:
    cmp     rbx, r13
    jae     pn_done
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      pn_frac
    cmp     al, 57
    ja      pn_frac
    inc     rbx
    jmp     pn_d1

pn_frac:
    cmp     rbx, r13
    jae     pn_done
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 46
    jne     pn_exp
    inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      fail
    cmp     al, 57
    ja      fail
    inc     rbx
pn_fd:
    cmp     rbx, r13
    jae     pn_done
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      pn_exp
    cmp     al, 57
    ja      pn_exp
    inc     rbx
    jmp     pn_fd

pn_exp:
    cmp     rbx, r13
    jae     pn_done
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 101
    je      1f
    cmp     al, 69
    jne     pn_done
1:  inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 43
    je      2f
    cmp     al, 45
    jne     3f
2:  inc     rbx
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
3:  cmp     al, 48
    jb      fail
    cmp     al, 57
    ja      fail
    inc     rbx
pn_ed:
    cmp     rbx, r13
    jae     pn_done
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 48
    jb      pn_done
    cmp     al, 57
    ja      pn_done
    inc     rbx
    jmp     pn_ed

pn_done:
    ret

# ---------------------------------------------------------------
# parse_object : buf[rbx] == '{'
# ---------------------------------------------------------------
parse_object:
    inc     rbx
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 125
    je      po_close
po_member:
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 34
    jne     fail
    call    parse_string
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 58
    jne     fail
    inc     rbx
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 44
    je      po_comma
    cmp     al, 125
    jne     fail
po_close:
    inc     rbx
    ret
po_comma:
    inc     rbx
    call    skip_ws
    jmp     po_member

# ---------------------------------------------------------------
# parse_array : buf[rbx] == '['
# ---------------------------------------------------------------
parse_array:
    inc     rbx
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 93
    je      pa_close
pa_elem:
    call    parse_value
    call    skip_ws
    cmp     rbx, r13
    jae     fail
    movzx   eax, byte ptr [r12 + rbx]
    cmp     al, 44
    je      pa_comma
    cmp     al, 93
    jne     fail
pa_close:
    inc     rbx
    ret
pa_comma:
    inc     rbx
    call    skip_ws
    jmp     pa_elem

# ---------------------------------------------------------------
# fail : print "error at <rbx>\n", exit 1
# ---------------------------------------------------------------
fail:
    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + s_err]
    mov     edx, 9
    syscall

    lea     rdi, [rip + outbuf]
    add     rdi, 31
    mov     byte ptr [rdi], 10
    mov     rax, rbx
    mov     ecx, 10
1:  xor     edx, edx
    div     rcx
    add     dl, 48
    dec     rdi
    mov     byte ptr [rdi], dl
    test    rax, rax
    jnz     1b

    mov     rsi, rdi
    lea     rdx, [rip + outbuf]
    add     rdx, 32
    sub     rdx, rsi
    mov     eax, 1
    mov     edi, 1
    syscall

    mov     eax, 231
    mov     edi, 1
    syscall
    jmp     .

.section .rodata
s_ok:
    .ascii "ok\n"
s_err:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"
esc_tbl:
    .byte 34, 92, 47, 98, 102, 110, 114, 116

.bss
.balign 16
outbuf:
    .zero 32
.balign 16
buf:
    .zero 70016
.equ BUFSZ, 70000
