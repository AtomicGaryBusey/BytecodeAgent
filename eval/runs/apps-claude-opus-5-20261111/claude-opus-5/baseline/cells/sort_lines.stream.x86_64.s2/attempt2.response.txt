.globl _start

.text
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    jle read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz done

    # scan lines, build table of (offset,length) pairs
    xor r13d, r13d
    xor rcx, rcx
    lea rbx, [rip + buf]
    lea rbp, [rip + table]
scan_loop:
    cmp rcx, r12
    jae scan_done
    mov r8, rcx
scan_inner:
    cmp rcx, r12
    jae scan_eol
    cmp byte ptr [rbx + rcx], 10
    je scan_eol
    inc rcx
    jmp scan_inner
scan_eol:
    mov r9, rcx
    sub r9, r8
    mov rax, r13
    shl rax, 4
    add rax, rbp
    mov qword ptr [rax], r8
    mov qword ptr [rax + 8], r9
    inc r13
    inc rcx
    jmp scan_loop
scan_done:
    cmp r13, 2
    jb output

    # insertion sort over table entries
    mov r14, 1
sort_outer:
    cmp r14, r13
    jae output
    mov rax, r14
    shl rax, 4
    add rax, rbp
    mov r10, qword ptr [rax]        # key offset
    mov r11, qword ptr [rax + 8]    # key length
    mov r15, r14
sort_inner:
    test r15, r15
    jz sort_place
    mov rax, r15
    dec rax
    shl rax, 4
    add rax, rbp
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    # compare line(rdi,rsi) with key(r10,r11): need prev > key to shift
    push r10
    push r11
    push rax
    call cmp_lines
    pop rax
    pop r11
    pop r10
    test eax, eax
    jle sort_place
    # shift prev up one slot
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov qword ptr [rax + 16], rdi
    mov qword ptr [rax + 24], rsi
    dec r15
    jmp sort_inner
sort_place:
    mov rax, r15
    shl rax, 4
    add rax, rbp
    mov qword ptr [rax], r10
    mov qword ptr [rax + 8], r11
    inc r14
    jmp sort_outer

# cmp_lines: (rdi=offA, rsi=lenA) vs (r10=offB, r11=lenB) -> eax <0,0,>0
cmp_lines:
    lea r8, [rip + buf]
    add rdi, r8
    add r8, r10
    mov rcx, rsi
    cmp rcx, r11
    jbe 1f
    mov rcx, r11
1:
    xor rdx, rdx
2:
    cmp rdx, rcx
    jae 3f
    movzx eax, byte ptr [rdi + rdx]
    movzx r9d, byte ptr [r8 + rdx]
    cmp eax, r9d
    jb cmp_lt
    ja cmp_gt
    inc rdx
    jmp 2b
3:
    cmp rsi, r11
    jb cmp_lt
    ja cmp_gt
    xor eax, eax
    ret
cmp_lt:
    mov eax, -1
    ret
cmp_gt:
    mov eax, 1
    ret

output:
    # build output buffer
    lea rdi, [rip + outbuf]
    xor r14d, r14d
    lea rbx, [rip + buf]
    lea rbp, [rip + table]
out_loop:
    cmp r14, r13
    jae out_write
    mov rax, r14
    shl rax, 4
    add rax, rbp
    mov rsi, qword ptr [rax]
    mov rcx, qword ptr [rax + 8]
    add rsi, rbx
out_copy:
    test rcx, rcx
    jz out_nl
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp out_copy
out_nl:
    mov byte ptr [rdi], 10
    inc rdi
    inc r14
    jmp out_loop
out_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
buf:
    .zero 32768
outbuf:
    .zero 34816
table:
    .zero 16016
