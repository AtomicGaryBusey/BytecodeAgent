.globl _start

.text
_start:
    # read all of stdin into buf
    lea rsi, [rip + buf]
    xor r12d, r12d          # total bytes
read_loop:
    xor eax, eax            # read
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 32769
    sub rdx, r12
    test rdx, rdx
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz done

    # scan lines: build table of (offset, length)
    xor r13d, r13d          # line count
    xor rcx, rcx            # cursor
    lea rbx, [rip + buf]
    lea rbp, [rip + table]
scan_loop:
    cmp rcx, r12
    jae scan_done
    mov r8, rcx             # start of line
scan_inner:
    cmp rcx, r12
    jae scan_eol
    cmp byte ptr [rbx + rcx], 10
    je scan_eol
    inc rcx
    jmp scan_inner
scan_eol:
    mov r9, rcx
    sub r9, r8              # length
    mov qword ptr [rbp + r13*8*2], r8
    mov
