.globl _start

.section .rodata
hextab:
.ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
.zero 65536
.balign 16
obuf:
.zero 324096

.text
_start:
    xor r12, r12                      # total bytes read
read_loop:
    mov edx, 65536
    sub edx, r12d
    test edx, edx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea r8, [rip + hextab]
    lea rbx, [rip + obuf]
    xor r13, r13                      # current offset

row_loop:
    cmp r13, r12
    jae rows_done

    # offset: 8 hex digits
    mov eax, r13d
    mov ecx, 32
1:
    sub ecx, 4
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    test ecx, ecx
    jnz 1b

    mov word ptr [rbx], 0x2020
    add rbx, 2

    # n = min(16, total - off)
    mov rbp, r12
    sub rbp, r13
    cmp rbp, 16
    jbe 2f
    mov rbp, 16
2:
    lea rsi, [rip + inbuf]
    add rsi, r13

    xor ecx, ecx
3:
    cmp rcx, rbp
    jb 4f
    mov word ptr [rbx], 0x2020
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp 5f
4:
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rbx + 1], al
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
5:
    cmp rcx, 7
    jne 6f
    mov byte ptr [rbx], 0x20
    inc rbx
6:
    inc rcx
    cmp rcx, 16
    jb 3b

    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2

    xor ecx, ecx
7:
    cmp rcx, rbp
    jae 9f
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 8f
    mov eax, 0x2e
8:
    mov byte ptr [rbx], al
    inc rbx
    inc rcx
    jmp 7b
9:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 10
    add rbx, 2

    add r13, 16
    jmp row_loop

rows_done:
    mov eax, r12d
    mov ecx, 32
10:
    sub ecx, 4
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    test ecx, ecx
    jnz 10b
    mov byte ptr [rbx], 10
    inc rbx

    lea rsi, [rip + obuf]
    mov rdx, rbx
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 231
    xor edi, edi
    syscall
