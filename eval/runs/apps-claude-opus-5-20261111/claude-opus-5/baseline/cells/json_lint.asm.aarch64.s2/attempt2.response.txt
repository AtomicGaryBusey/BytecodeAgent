.globl _start

.text
_start:
    adrp    x21, inbuf
    add     x21, x21, :lo12:inbuf
    mov     x19, #0
    movz    x22, #2, lsl #16            // buffer size 131072
rdloop:
    mov     x0, #0
    add     x1, x21, x19
    mov     x2, x22
    sub     x2, x2, x19
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    rdend
    add     x19, x19, x0
    b       rdloop
rdend:
    mov     x20, x19
    mov     x19, #0

    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.ne    fail

    // success
    mov     x0, #1
    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
// skip whitespace
skip_ws:
1:  cmp     x19, x20
    b.hs    2f
    ldrb    w1, [x21, x19]
    cmp     w1, #0x20
    b.eq    3f
    cmp     w1, #0x09
    b.eq    3f
    cmp     w1, #0x0a
    b.eq    3f
    cmp     w1, #0x0d
    b.ne    2f
3:  add     x19, x19, #1
    b       1b
2:  ret

// ---------------------------------------------------------------
// parse a value at x19
parse_value:
    stp     x29, x30, [sp, #-16]!
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x7b                   // '{'
    b.eq    pv_obj
    cmp     w1, #0x5b                   // '['
    b.eq    pv_arr
    cmp     w1, #0x22                   // '"'
    b.eq    pv_str
    cmp     w1, #0x2d                   // '-'
    b.eq    pv_num
    cmp     w1, #0x30                   // '0'
    b.lo    1f
    cmp     w1, #0x39                   // '9'
    b.ls    pv_num
1:  cmp     w1, #0x74                   // 't'
    b.eq    pv_true
    cmp     w1, #0x66                   // 'f'
    b.eq    pv_false
    cmp     w1, #0x6e                   // 'n'
    b.eq    pv_null
    b       fail

pv_obj:
    bl      parse_object
    ldp     x29, x30, [sp], #16
    ret
pv_arr:
    bl      parse_array
    ldp     x29, x30, [sp], #16
    ret
pv_str:
    bl      parse_string
    ldp     x29, x30, [sp], #16
    ret
pv_num:
    bl      parse_number
    ldp     x29, x30, [sp], #16
    ret
pv_true:
    adrp    x2, lit_true
    add     x2, x2, :lo12:lit_true
    mov     w3, #4
    b       pv_lit
pv_false:
    adrp    x2, lit_false
    add     x2, x2, :lo12:lit_false
    mov     w3, #5
    b       pv_lit
pv_null:
    adrp    x2, lit_null
    add     x2, x2, :lo12:lit_null
    mov     w3, #4
pv_lit:
1:  cmp     x19, x20
    b.hs    fail
    ldrb    w4, [x21, x19]
    ldrb    w5, [x2], #1
    cmp     w4, w5
    b.ne    fail
    add     x19, x19, #1
    sub     w3, w3, #1
    cbnz    w3, 1b
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse string, x19 points at the opening quote
parse_string:
    add     x19, x19, #1
ps_loop:
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x22                   // '"'
    b.eq    ps_end
    cmp     w1, #0x5c                   // '\'
    b.eq    ps_esc
    cmp     w1, #0x20
    b.lo    fail
    add     x19, x19, #1
    b       ps_loop
ps_end:
    add     x19, x19, #1
    ret
ps_esc:
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x75                   // 'u'
    b.eq    ps_u
    cmp     w1, #0x22
    b.eq    ps_ok1
    cmp     w1, #0x5c
    b.eq    ps_ok1
    cmp     w1, #0x2f
    b.eq    ps_ok1
    cmp     w1, #0x62
    b.eq    ps_ok1
    cmp     w1, #0x66
    b.eq    ps_ok1
    cmp     w1, #0x6e
    b.eq    ps_ok1
    cmp     w1, #0x72
    b.eq    ps_ok1
    cmp     w1, #0x74
    b.eq    ps_ok1
    b       fail
ps_ok1:
    add     x19, x19, #1
    b       ps_loop
ps_u:
    add     x19, x19, #1
    mov     w3, #4
1:  cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.ls    2f
    orr     w4, w1, #0x20
    sub     w4, w4, #0x61
    cmp     w4, #5
    b.hi    fail
2:  add     x19, x19, #1
    sub     w3, w3, #1
    cbnz    w3, 1b
    b       ps_loop

// ---------------------------------------------------------------
// parse number at x19
parse_number:
    ldrb    w1, [x21, x19]
    cmp     w1, #0x2d                   // '-'
    b.ne    1f
    add     x19, x19, #1
1:  cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.hi    fail
    add     x19, x19, #1
    cbz     w4, pn_frac
pn_digits:
    cmp     x19, x20
    b.hs    pn_done
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.hi    pn_frac
    add     x19, x19, #1
    b       pn_digits
pn_frac:
    cmp     x19, x20
    b.hs    pn_done
    ldrb    w1, [x21, x19]
    cmp     w1, #0x2e                   // '.'
    b.ne    pn_exp
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.hi    fail
2:  add     x19, x19, #1
    cmp     x19, x20
    b.hs    pn_done
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.ls    2b
pn_exp:
    cmp     x19, x20
    b.hs    pn_done
    ldrb    w1, [x21, x19]
    orr     w2, w1, #0x20
    cmp     w2, #0x65                   // 'e'
    b.ne    pn_done
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x2b                   // '+'
    b.eq    3f
    cmp     w1, #0x2d                   // '-'
    b.ne    4f
3:  add     x19, x19, #1
4:  cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.hi    fail
5:  add     x19, x19, #1
    cmp     x19, x20
    b.hs    pn_done
    ldrb    w1, [x21, x19]
    sub     w4, w1, #0x30
    cmp     w4, #9
    b.ls    5b
pn_done:
    ret

// ---------------------------------------------------------------
// parse object, x19 points at '{'
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1
    bl      skip_ws
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x7d                   // '}'
    b.eq    po_close
po_member:
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x22                   // '"'
    b.ne    fail
    bl      parse_string
    bl      skip_ws
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x3a                   // ':'
    b.ne    fail
    add     x19, x19, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x7d
    b.eq    po_close
    cmp     w1, #0x2c                   // ','
    b.ne    fail
    add     x19, x19, #1
    bl      skip_ws
    b       po_member
po_close:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse array, x19 points at '['
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1
    bl      skip_ws
    cmp     x19, x20
    b.hs    pa_elem
    ldrb    w1, [x21, x19]
    cmp     w1, #0x5d                   // ']'
    b.eq    pa_close
pa_elem:
    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.hs    fail
    ldrb    w1, [x21, x19]
    cmp     w1, #0x5d
    b.eq    pa_close
    cmp     w1, #0x2c                   // ','
    b.ne    fail
    add     x19, x19, #1
    bl      skip_ws
    b       pa_elem
pa_close:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// error exit, x19 = offset
fail:
    mov     x0, #1
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #9
    mov     x8, #64
    svc     #0

    adrp    x3, numbuf
    add     x3, x3, :lo12:numbuf
    mov     x5, #31
    mov     w7, #10
    strb    w7, [x3, x5]
    mov     x2, x19
    mov     x10, #10
1:  udiv    x6, x2, x10
    msub    x7, x6, x10, x2
    add     w7, w7, #0x30
    sub     x5, x5, #1
    strb    w7, [x3, x5]
    mov     x2, x6
    cbnz    x2, 1b

    add     x1, x3, x5
    mov     x2, #32
    sub     x2, x2, x5
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
.section .rodata
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "

// ---------------------------------------------------------------
.bss
    .balign 16

    .zero 32
    .balign 16

    .zero 131072
