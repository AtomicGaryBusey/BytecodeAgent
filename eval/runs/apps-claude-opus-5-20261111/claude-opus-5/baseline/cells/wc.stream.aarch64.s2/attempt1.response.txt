.globl _start

.text
_start:
    mov     x19, #0                     // lines
    mov     x20, #0                     // words
    mov     x21, #0                     // bytes
    mov     x22, #0                     // in-word flag

read_loop:
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x0, #0
    mov     x2, #4096
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    input_done
    add     x21, x21, x0
    add     x2, x1, x0

byte_loop:
    ldrb    w3, [x1], #1
    cmp     w3, #0x20
    b.eq    is_ws
    sub     w4, w3, #9
    cmp     w4, #4
    b.ls    is_ws
    cbnz    x22, next_byte
    mov     x22, #1
    add     x20, x20, #1
next_byte:
    cmp     x1, x2
    b.lo    byte_loop
    b       read_loop

is_ws:
    cmp     w3, #0x0a
    b.ne    is_ws2
    add     x19, x19, #1
is_ws2:
    mov     x22, #0
    b       next_byte

input_done:
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x24, x23

    mov     x0, x19
    mov     x1, x24
    bl      tonum
    mov     x24, x1
    mov     w0, #0x20
    strb    w0, [x24], #1

    mov     x0, x20
    mov     x1, x24
    bl      tonum
    mov     x24, x1
    mov     w0, #0x20
    strb    w0, [x24], #1

    mov     x0, x21
    mov     x1, x24
    bl      tonum
    mov     x24, x1
    mov     w0, #0x0a
    strb    w0, [x24], #1

    sub     x2, x24, x23
    mov     x1, x23
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0
    mov     x0, #0
    mov     x8, #94
    svc     #0

// x0 = value, x1 = destination pointer; returns x1 past last digit
tonum:
    sub     sp, sp, #32
    add     x2, sp, #32
    add     x6, sp, #32
    mov     x3, #10
1:
    udiv    x4, x0, x3
    msub    x5, x4, x3, x0
    add     w5, w5, #0x30
    strb    w5, [x2, #-1]!
    mov     x0, x4
    cbnz    x0, 1b
2:
    ldrb    w7, [x2], #1
    strb    w7, [x1], #1
    cmp     x2, x6
    b.lo    2b
    add     sp, sp, #32
    ret

.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 128
