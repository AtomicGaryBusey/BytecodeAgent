.globl _start

.section .bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov w22, #0                 // in-word flag

read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    ble done
    add x21, x21, x0
    adrp x1, buf
    add x1, x1, :lo12:buf
    add x3, x1, x0

byte_loop:
    ldrb w4, [x1], #1
    cmp w4, #10
    cinc x19, x19, eq
    cmp w4, #32
    b.eq is_ws
    sub w5, w4, #9
    cmp w5, #5
    b.lo is_ws
    cbnz w22, next_byte
    mov w22, #1
    add x20, x20, #1
    b next_byte
is_ws:
    mov w22, #0
next_byte:
    cmp x1, x3
    b.lo byte_loop
    b read_loop

done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x19
    bl itoa
    mov w5, #32
    strb w5, [x1], #1
    mov x0, x20
    bl itoa
    mov w5, #32
    strb w5, [x1], #1
    mov x0, x21
    bl itoa
    mov w5, #10
    strb w5, [x1], #1
    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// x0 = value, x1 = destination pointer, returns advanced x1
itoa:
    sub sp, sp, #32
    add x2, sp, #31
    mov x3, #10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    cbnz x0, 1b
    add x6, sp, #31
2:
    ldrb w5, [x2], #1
    strb w5, [x1], #1
    cmp x2, x6
    b.lo 2b
    add sp, sp, #32
    ret
