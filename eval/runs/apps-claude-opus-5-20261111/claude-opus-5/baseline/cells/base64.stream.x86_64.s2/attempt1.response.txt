.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 64
inbuf:
    .zero 65536
    .balign 64
outbuf:
    .zero 73728

.text
_start:
    xor r12, r12                      # bytes read so far

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jbe read_done
    xor eax, eax                      # read
    xor edi, edi                      # fd 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alphabet]
    mov rcx, r12
    xor r8d, r8d                      # chars in current line

enc3:
    cmp rcx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3
    sub rcx, 3

    mov r9d, eax
    shr r9d, 18
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], r9b

    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 2], r9b

    mov r9d, eax
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 3], r9b

    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne enc3
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp enc3

tail:
    test rcx, rcx
    jz finish
    cmp rcx, 1
    je tail1

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

    mov r9d, eax
    shr r9d, 18
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], r9b

    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 2], r9b

    mov byte ptr [rdi + 3], 61
    jmp tailend

tail1:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov r9d, eax
    shr r9d, 18
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], r9b

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

tailend:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne finish
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

finish:
    test r8d, r8d
    jz write_start
    mov byte ptr [rdi], 10
    inc rdi

write_start:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi                      # remaining bytes

write_loop:
    test r13, r13
    jz done
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r13, rax
    jmp write_loop

done:
    mov eax, 60
    xor edi, edi
    syscall
    mov eax, 231
    xor edi, edi
    syscall
