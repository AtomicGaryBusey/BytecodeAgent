.globl _start

.text
_start:
    // ---- read all of standard input into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x28, #0                 // total bytes read
1:
    movz    x2, #1, lsl #16         // 65536
    subs    x2, x2, x28
    b.eq    2f
    add     x1, x19, x28
    mov     x0, #0
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x28, x28, x0
    b       1b
2:
    // ---- split program / input at first '!' (0x21) ----
    mov     x20, x28                // program length
    mov     x26, x28                // input position
    mov     x0, #0
3:
    cmp     x0, x28
    b.hs    4f
    ldrb    w1, [x19, x0]
    cmp     w1, #0x21
    b.eq    5f
    add     x0, x0, #1
    b       3b
5:
    mov     x20, x0
    add     x26, x0, #1
4:
    mov     x27, x28                // input end

    // ---- validate brackets and build jump table ----
    adrp    x23, jumps
    add     x23, x23, :lo12:jumps
    adrp    x4, bstack
    add     x4, x4, :lo12:bstack
    mov     x0, #0                  // i
    mov     x1, #0                  // bracket stack depth
6:
    cmp     x0, x20
    b.hs    9f
    ldrb    w2, [x19, x0]
    cmp     w2, #0x5b               // '['
    b.eq    7f
    cmp     w2, #0x5d               // ']'
    b.eq    8f
    add     x0, x0, #1
    b       6b
7:
    str     w0, [x4, x1, lsl #2]
    add     x1, x1, #1
    add     x0, x0, #1
    b       6b
8:
    cbz     x1, bad
    sub     x1, x1, #1
    ldr     w3, [x4, x1, lsl #2]
    add     x0, x0, #1
    str     w0, [x23, x3, lsl #2]   // '[' jumps to just after ']'
    add     w5, w3, #1
    sub     x6, x0, #1
    str     w5, [x23, x6, lsl #2]   // ']' jumps to just after '['
    b       6b
9:
    cbnz    x1, bad

    // ---- run ----
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x24, outbuf
    add     x24, x24, :lo12:outbuf
    mov     x25, #0                 // pending output bytes
    mov     x21, #0                 // pc

run:
    cmp     x21, x20
    b.hs    done
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #0x2b               // '+'
    b.eq    op_plus
    cmp     w0, #0x3e               // '>'
    b.eq    op_right
    cmp     w0, #0x2d               // '-'
    b.eq    op_minus
    cmp     w0, #0x3c               // '<'
    b.eq    op_left
    cmp     w0, #0x5b               // '['
    b.eq    op_lb
    cmp     w0, #0x5d               // ']'
    b.eq    op_rb
    cmp     w0, #0x2e               // '.'
    b.eq    op_dot
    cmp     w0, #0x2c               // ','
    b.eq    op_comma
    b       run

op_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       run
op_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       run
op_right:
    add     x22, x22, #1
    b       run
op_left:
    sub     x22, x22, #1
    b       run
op_lb:
    ldrb    w1, [x22]
    cbnz    w1, run
    sub     x2, x21, #1
    ldr     w21, [x23, x2, lsl #2]
    b       run
op_rb:
    ldrb    w1, [x22]
    cbz     w1, run
    sub     x2, x21, #1
    ldr     w21, [x23, x2, lsl #2]
    b       run
op_dot:
    ldrb    w1, [x22]
    strb    w1, [x24, x25]
    add     x25, x25, #1
    cmp     x25, #4096
    b.lo    run
    bl      flush
    b       run
op_comma:
    cmp     x26, x27
    b.hs    1f
    ldrb    w1, [x19, x26]
    add     x26, x26, #1
    strb    w1, [x22]
    b       run
1:
    strb    wzr, [x22]
    b       run

done:
    bl      flush
    mov     x0, #0
    mov     x8, #93
    svc     #0

bad:
    mov     x0, #1
    mov     x8, #93
    svc     #0

flush:
    cbz     x25, 1f
    mov     x0, #1
    mov     x1, x24
    mov     x2, x25
    mov     x8, #64                 // write
    svc     #0
    mov     x25, #0
1:
    ret

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
tape:
    .zero 30016
.balign 16
outbuf:
    .zero 4096
.balign 16
jumps:
    .zero 262144
.balign 16
bstack:
    .zero 8192
