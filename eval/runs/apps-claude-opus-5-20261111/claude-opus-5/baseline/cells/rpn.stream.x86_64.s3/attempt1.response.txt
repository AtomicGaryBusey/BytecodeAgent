.globl _start

.text
_start:
    xor r12d, r12d
rd_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 70000
    sub rdx, r12
    jle rd_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    lea r13, [rip + inbuf]
    lea r14, [r13 + r12]

line_loop:
    cmp r13, r14
    jae all_done
    mov rsi, r13
    mov rdi, r13
scan_nl:
    cmp rdi, r14
    jae nl_none
    cmp byte ptr [rdi], 10
    je nl_found
    inc rdi
    jmp scan_nl
nl_found:
    lea r13, [rdi + 1]
    jmp eval_line
nl_none:
    mov r13, r14

eval_line:
    xor r8d, r8d
tok_loop:
    cmp rsi, rdi
    jae tok_end
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je tok_skip
    cmp al, 9
    jne tok_start
tok_skip:
    inc rsi
    jmp tok_loop
tok_start:
    mov r9, rsi
tok_scan:
    inc rsi
    cmp rsi, rdi
    jae tok_scan_done
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je tok_scan_done
    cmp al, 9
    je tok_scan_done
    jmp tok_scan
tok_scan_done:
    mov r10, rsi
    sub r10, r9
    cmp r10, 1
    jne parse_num
    movzx eax, byte ptr [r9]
    cmp al, 43
    je op_do
    cmp al, 45
    je op_do
    cmp al, 42
    je op_do
    cmp al, 47
    je op_do
    cmp al, 37
    je op_do
    jmp parse_num

op_do:
    cmp r8, 2
    jb err
    lea r11, [rip + vstack]
    mov rcx, [r11 + r8*8 - 8]
    mov rbx, [r11 + r8*8 - 16]
    dec r8
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne mod_go
    mov rax, 1
    shl rax, 63
    cmp rbx, rax
    je err
mod_go:
    mov rax, rbx
    cqo
    idiv rcx
    mov rbx, rdx
    jmp op_store
op_div:
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne div_go
    mov rax, 1
    shl rax, 63
    cmp rbx, rax
    je err
div_go:
    mov rax, rbx
    cqo
    idiv rcx
    mov rbx, rax
    jmp op_store
op_add:
    add rbx, rcx
    jmp op_store
op_sub:
    sub rbx, rcx
    jmp op_store
op_mul:
    imul rbx, rcx
op_store:
    lea r11, [rip + vstack]
    mov [r11 + r8*8 - 8], rbx
    jmp tok_loop

parse_num:
    mov rcx, r9
    xor r11d, r11d
    movzx eax, byte ptr [rcx]
    cmp al, 45
    jne pn_nn
    mov r11d, 1
    inc rcx
pn_nn:
    xor edx, edx
    xor ebx, ebx
pn_dig:
    cmp rcx, rsi
    jae pn_digdone
    movzx eax, byte ptr [rcx]
    sub eax, 48
    cmp eax, 9
    ja err
    inc rcx
    inc rbx
    cmp rbx, 18
    ja err
    lea rdx, [rdx + rdx*4]
    add rdx, rdx
    add rdx, rax
    jmp pn_dig
pn_digdone:
    test rbx, rbx
    jz err
    test r11d, r11d
    jz pn_push
    neg rdx
pn_push:
    cmp r8, 1000
    jae err
    lea r11, [rip + vstack]
    mov [r11 + r8*8], rdx
    inc r8
    jmp tok_loop

tok_end:
    cmp r8, 1
    jne err
    lea r11, [rip + vstack]
    mov rax, [r11]
    call print_num
    jmp line_loop

err:
    lea rsi, [rip + errmsg]
    mov ecx, 6
    call append
    jmp line_loop

all_done:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_num:
    lea r10, [rip + numbuf]
    add r10, 24
    mov byte ptr [r10], 10
    mov r9, r10
    inc r9
    xor r11d, r11d
    test rax, rax
    jns pn_pos
    mov r11d, 1
    neg rax
pn_pos:
    mov rcx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r10
    mov byte ptr [r10], dl
    test rax, rax
    jnz pn_loop
    test r11d, r11d
    jz pn_nosign
    dec r10
    mov byte ptr [r10], 45
pn_nosign:
    mov rsi, r10
    mov rcx, r9
    sub rcx, r10
    call append
    ret

append:
    mov rdi, [rip + outlen]
    lea rdx, [rip + outbuf]
    add rdi, rdx
ap_cp:
    test rcx, rcx
    jz ap_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp ap_cp
ap_done:
    lea rdx, [rip + outbuf]
    sub rdi, rdx
    mov qword ptr [rip + outlen], rdi
    cmp rdi, 65472
    jb ap_ret
    call flush
ap_ret:
    ret

flush:
    mov rdx, qword ptr [rip + outlen]
    lea rsi, [rip + outbuf]
fl_loop:
    test rdx, rdx
    jz fl_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle fl_done
    add rsi, rax
    sub rdx, rax
    jmp fl_loop
fl_done:
    mov qword ptr [rip + outlen], 0
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 8
outlen:
    .zero 8
numbuf:
    .zero 32
vstack:
    .zero 8192
outbuf:
    .zero 65536
inbuf:
    .zero 70016
