.globl _start

.text
_start:
    xor     ebx, ebx
rd_loop:
    lea     rsi, [rip + buf]
    add     rsi, rbx
    mov     edx, 70000
    sub     edx, ebx
    jbe     rd_done
    xor     eax, eax
    xor     edi, edi
    syscall
    test    rax, rax
    jle     rd_done
    add     rbx, rax
    jmp     rd_loop
rd_done:
    lea     r12, [rip + buf]
    add     r12, rbx
    lea     rbx, [rip + buf]
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r12
    jne     fail

    lea     rsi, [rip + msg_ok]
    mov     edx, 3
    mov     eax, 1
    mov     edi, 1
    syscall
    xor     edi, edi
    mov     eax, 60
    syscall
    hlt

# ---------------------------------------------------------------
# skip whitespace:  rbx = cursor, r12 = end
# ---------------------------------------------------------------
skip_ws:
sw_loop:
    cmp     rbx, r12
    jae     sw_ret
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x20
    je      sw_adv
    cmp     al, 0x09
    je      sw_adv
    cmp     al, 0x0A
    je      sw_adv
    cmp     al, 0x0D
    je      sw_adv
sw_ret:
    ret
sw_adv:
    inc     rbx
    jmp     sw_loop

# ---------------------------------------------------------------
# parse one value at rbx
# ---------------------------------------------------------------
parse_value:
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x7B
    je      parse_object
    cmp     al, 0x5B
    je      parse_array
    cmp     al, 0x22
    je      parse_string
    cmp     al, 0x74
    je      do_true
    cmp     al, 0x66
    je      do_false
    cmp     al, 0x6E
    je      do_null
    cmp     al, 0x2D
    je      parse_number
    cmp     al, 0x30
    jb      fail
    cmp     al, 0x39
    ja      fail
    jmp     parse_number

do_true:
    lea     rdi, [rip + lit_true]
    jmp     match_lit
do_false:
    lea     rdi, [rip + lit_false]
    jmp     match_lit
do_null:
    lea     rdi, [rip + lit_null]
    jmp     match_lit

match_lit:
ml_loop:
    movzx   eax, byte ptr [rdi]
    test    al, al
    je      ml_ret
    cmp     rbx, r12
    jae     fail
    cmp     al, byte ptr [rbx]
    jne     fail
    inc     rbx
    inc     rdi
    jmp     ml_loop
ml_ret:
    ret

# ---------------------------------------------------------------
# string: [rbx] is known to be '"'
# ---------------------------------------------------------------
parse_string:
    inc     rbx
ps_loop:
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x22
    je      ps_end
    cmp     al, 0x5C
    je      ps_esc
    cmp     al, 0x20
    jb      fail
    inc     rbx
    jmp     ps_loop
ps_esc:
    inc     rbx
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x75
    je      ps_hex4
    cmp     al, 0x22
    je      ps_esc_ok
    cmp     al, 0x5C
    je      ps_esc_ok
    cmp     al, 0x2F
    je      ps_esc_ok
    cmp     al, 0x62
    je      ps_esc_ok
    cmp     al, 0x66
    je      ps_esc_ok
    cmp     al, 0x6E
    je      ps_esc_ok
    cmp     al, 0x72
    je      ps_esc_ok
    cmp     al, 0x74
    je      ps_esc_ok
    jmp     fail
ps_esc_ok:
    inc     rbx
    jmp     ps_loop
ps_hex4:
    inc     rbx
    mov     ecx, 4
ps_hex_loop:
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      fail
    cmp     al, 0x39
    jbe     ps_hex_ok
    or      al, 0x20
    cmp     al, 0x61
    jb      fail
    cmp     al, 0x66
    ja      fail
ps_hex_ok:
    inc     rbx
    dec     ecx
    jnz     ps_hex_loop
    jmp     ps_loop
ps_end:
    inc     rbx
    ret

# ---------------------------------------------------------------
# number: [rbx] is '-' or a digit
# ---------------------------------------------------------------
parse_number:
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2D
    jne     pn_int
    inc     rbx
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
pn_int:
    cmp     al, 0x30
    je      pn_zero
    cmp     al, 0x31
    jb      fail
    cmp     al, 0x39
    ja      fail
    inc     rbx
pn_dig_loop:
    cmp     rbx, r12
    jae     pn_ret
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      pn_frac
    cmp     al, 0x39
    ja      pn_frac
    inc     rbx
    jmp     pn_dig_loop
pn_zero:
    inc     rbx
pn_frac:
    cmp     rbx, r12
    jae     pn_ret
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2E
    jne     pn_exp
    inc     rbx
    call    need_digits
pn_exp:
    cmp     rbx, r12
    jae     pn_ret
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x65
    je      pn_exp_go
    cmp     al, 0x45
    jne     pn_ret
pn_exp_go:
    inc     rbx
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x2B
    je      pn_exp_sign
    cmp     al, 0x2D
    jne     pn_exp_dig
pn_exp_sign:
    inc     rbx
pn_exp_dig:
    call    need_digits
pn_ret:
    ret

need_digits:
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      fail
    cmp     al, 0x39
    ja      fail
nd_loop:
    inc     rbx
    cmp     rbx, r12
    jae     nd_ret
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x30
    jb      nd_ret
    cmp     al, 0x39
    ja      nd_ret
    jmp     nd_loop
nd_ret:
    ret

# ---------------------------------------------------------------
# object: [rbx] is known to be '{'
# ---------------------------------------------------------------
parse_object:
    inc     rbx
    call    skip_ws
    cmp     rbx, r12
    jae     fail
    cmp     byte ptr [rbx], 0x7D
    je      po_end
po_member:
    cmp     rbx, r12
    jae     fail
    cmp     byte ptr [rbx], 0x22
    jne     fail
    call    parse_string
    call    skip_ws
    cmp     rbx, r12
    jae     fail
    cmp     byte ptr [rbx], 0x3A
    jne     fail
    inc     rbx
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x7D
    je      po_end
    cmp     al, 0x2C
    jne     fail
    inc     rbx
    call    skip_ws
    jmp     po_member
po_end:
    inc     rbx
    ret

# ---------------------------------------------------------------
# array: [rbx] is known to be '['
# ---------------------------------------------------------------
parse_array:
    inc     rbx
    call    skip_ws
    cmp     rbx, r12
    jae     fail
    cmp     byte ptr [rbx], 0x5D
    je      pa_end
pa_elem:
    call    parse_value
    call    skip_ws
    cmp     rbx, r12
    jae     fail
    movzx   eax, byte ptr [rbx]
    cmp     al, 0x5D
    je      pa_end
    cmp     al, 0x2C
    jne     fail
    inc     rbx
    call    skip_ws
    jmp     pa_elem
pa_end:
    inc     rbx
    ret

# ---------------------------------------------------------------
# error reporting: offset = rbx - buf
# ---------------------------------------------------------------
fail:
    lea     rax, [rip + buf]
    sub     rbx, rax
    lea     rdi, [rip + outbuf]
    mov     rax, qword ptr [rip + msg_err]
    mov     qword ptr [rdi], rax
    mov     byte ptr [rdi + 8], 0x20
    add     rdi, 9
    mov     rax, rbx
    lea     rsi, [rip + tmpend]
    mov     ecx, 10
fl_div:
    xor     edx, edx
    div     rcx
    add     dl, 0x30
    dec     rsi
    mov     byte ptr [rsi], dl
    test    rax, rax
    jnz     fl_div
fl_copy:
    lea     rdx, [rip + tmpend]
    cmp     rsi, rdx
    jae     fl_done
    mov     al, byte ptr [rsi]
    mov     byte ptr [rdi], al
    inc     rsi
    inc     rdi
    jmp     fl_copy
fl_done:
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    lea     rsi, [rip + outbuf]
    mov     rdx, rdi
    sub     rdx, rsi
    mov     eax, 1
    mov     edi, 1
    syscall
    mov     eax, 60
    mov     edi, 1
    syscall
    hlt

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at"
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
.align 16
buf:
    .zero 70016
outbuf:
    .zero 64
tmp:
    .zero 32
tmpend:
    .zero 16
