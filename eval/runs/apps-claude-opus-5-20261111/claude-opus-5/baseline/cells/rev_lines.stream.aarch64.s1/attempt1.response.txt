.globl _start

.bss
.balign 16
inbuf:
.zero 65536
outbuf:
.zero 65536

.text
_start:
    // read all input into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                      // total bytes read
    mov x21, #65536

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, done

    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, #0                      // out index
    mov x24, #0                      // line start index

line_loop:
    cmp x24, x20
    b.ge flush
    // find newline from x24
    mov x25, x24
scan:
    cmp x25, x20
    b.ge scan_end
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.eq scan_end
    add x25, x25, #1
    b scan
scan_end:
    // line is [x24, x25)
    mov x26, x25
rev:
    cmp x26, x24
    b.le rev_done
    sub x26, x26, #1
    ldrb w0, [x19, x26]
    strb w0, [x22, x23]
    add x23, x23, #1
    b rev
rev_done:
    mov w0, #10
    strb w0, [x22, x23]
    add x23, x23, #1
    add x24, x25, #1
    b line_loop

flush:
    mov x0, #1
    mov x1, x22
    mov x2, x23
    mov x8, #64
    svc #0

done:
    mov x0, #0
    mov x8, #93
    svc #0
