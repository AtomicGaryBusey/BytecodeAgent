.globl _start

.text
_start:
    xor r12, r12                    # lines
    xor r13, r13                    # words
    xor r14, r14                    # bytes
    xor r15, r15                    # in-word flag
    lea rbx, [rip + wstab]

read_loop:
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea r9, [rsi + rax]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    movzx edx, byte ptr [rbx + rax]
    test edx, edx
    jnz 2f
    test r15, r15
    jnz 3f
    inc r13
    mov r15d, 1
    jmp 3f
2:
    xor r15d, r15d
3:
    cmp rsi, r9
    jb byte_loop
    jmp read_loop

done:
    lea rdi, [rip + obuf]
    add rdi, 64
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call emit
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call emit
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call emit

    lea rdx, [rip + obuf]
    add rdx, 64
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall
    hlt

emit:
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.section .rodata
wstab:
    .zero 9
    .byte 1, 1, 1, 1, 1
    .zero 18
    .byte 1
    .zero 223

.bss
.balign 16
buf:
    .zero 65536
obuf:
    .zero 64
