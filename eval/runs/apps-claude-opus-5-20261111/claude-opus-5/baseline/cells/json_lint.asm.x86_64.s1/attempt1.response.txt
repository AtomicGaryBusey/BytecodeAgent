.globl _start

.text
_start:
    lea r12, [rip + buf]
    xor rbx, rbx
rd_loop:
    mov rdx, 69999
    sub rdx, rbx
    jbe rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd_loop
rd_done:
    mov r13, rbx
    xor rbx, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# ---------------- error reporting ----------------
fail:
    lea rsi, [rip + numbuf + 24]
    mov byte ptr [rsi], 0x0A
    mov rax, rbx
    mov ecx, 10
f_dig:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz f_dig
    mov r14, rsi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    lea rdx, [rip + numbuf + 25]
    sub rdx, r14
    syscall
    mov eax, 231
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

# ---------------- whitespace ----------------
skip_ws:
    cmp rbx, r13
    jae sw_ret
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    jne sw_ret
sw_adv:
    inc rbx
    jmp skip_ws
sw_ret:
    ret

# ---------------- value ----------------
parse_value:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    jmp parse_number

pv_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp match_lit
pv_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp match_lit
pv_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
match_lit:
ml_loop:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, byte ptr [rsi]
    jne fail
    inc rbx
    inc rsi
    dec ecx
    jnz ml_loop
    ret

# ---------------- string ----------------
parse_string:
    inc rbx
ps_loop:
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb fail
    inc rbx
    jmp ps_loop
ps_end:
    inc rbx
    ret
ps_esc:
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x75
    je ps_u
    lea rsi, [rip + esc_set]
    mov ecx, 8
ps_e2:
    cmp al, byte ptr [rsi]
    je ps_e3
    inc rsi
    dec ecx
    jnz ps_e2
    jmp fail
ps_e3:
    inc rbx
    jmp ps_loop
ps_u:
    inc rbx
    mov ecx, 4
ps_u2:
    movzx eax, byte ptr [r12 + rbx]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    jbe ps_u3
    or eax, 0x20
    sub eax, 0x61
    cmp eax, 5
    ja fail
ps_u3:
    inc rbx
    dec ecx
    jnz ps_u2
    jmp ps_loop

# ---------------- number ----------------
parse_number:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x2D
    jne pn_int
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
pn_int:
    cmp al, 0x30
    jne pn_nz
    inc rbx
    jmp pn_frac
pn_nz:
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
pn_d1:
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    jbe pn_d1
pn_frac:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x2E
    jne pn_exp
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
pn_fd:
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    jbe pn_fd
pn_exp:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x65
    je pn_e
    cmp al, 0x45
    jne pn_done
pn_e:
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x2B
    je pn_sign
    cmp al, 0x2D
    jne pn_ed0
pn_sign:
    inc rbx
pn_ed0:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
pn_ed:
    inc rbx
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    jbe pn_ed
pn_done:
    ret

# ---------------- object ----------------
parse_object:
    inc rbx
    call skip_ws
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x7D
    je po_close
po_member:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x22
    jne fail
    call parse_string
    call skip_ws
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x3A
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    jne fail
po_close:
    inc rbx
    ret
po_comma:
    inc rbx
    call skip_ws
    jmp po_member

# ---------------- array ----------------
parse_array:
    inc rbx
    call skip_ws
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x5D
    je pa_close
pa_elem:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    jne fail
pa_close:
    inc rbx
    ret
pa_comma:
    inc rbx
    call skip_ws
    jmp pa_elem

.section .rodata
okmsg:
    .ascii "ok\n"
errmsg:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"
esc_set:
    .byte 0x22, 0x5C, 0x2F, 0x62, 0x66, 0x6E, 0x72, 0x74

.bss
.balign 16
numbuf:
    .zero 32
.balign 16
buf:
    .zero 70016
