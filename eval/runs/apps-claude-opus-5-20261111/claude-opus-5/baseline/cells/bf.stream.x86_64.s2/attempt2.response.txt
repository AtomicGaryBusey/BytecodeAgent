.globl _start

.text
_start:
    xor r12d, r12d
rdloop:
    mov rdx, 65536
    sub rdx, r12
    jle rddone
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle rddone
    add r12, rax
    jmp rdloop
rddone:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
fbang:
    cmp rcx, r12
    jae nobang
    cmp byte ptr [rsi + rcx], 33
    je gotbang
    inc rcx
    jmp fbang
nobang:
    mov rdx, r12
    mov r13, r12
    jmp havesplit
gotbang:
    mov rdx, rcx
    lea r13, [rcx + 1]
havesplit:
    lea r14, [rsi + r13]
    lea r15, [rsi + r12]

    lea rdi, [rip + prog]
    lea r8, [rip + jumptab]
    lea r9, [rip + bstack]
    lea r11, [rip + cmdtab]
    xor ecx, ecx
    xor ebx, ebx
    xor r10d, r10d
filt:
    cmp rcx, rdx
    jae filtdone
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    movzx eax, byte ptr [r11 + rax]
    test al, al
    jz filt
    cmp al, 7
    je fpush
    cmp al, 8
    je fpop
    mov byte ptr [rdi + rbx], al
    inc rbx
    jmp filt
fpush:
    mov dword ptr [r9 + r10*4], ebx
    inc r10
    mov byte ptr [rdi + rbx], al
    inc rbx
    jmp filt
fpop:
    test r10, r10
    jz badexit
    dec r10
    mov ebp, dword ptr [r9 + r10*4]
    mov byte ptr [rdi + rbx], al
    inc rbx
    mov dword ptr [r8 + rbp*4], ebx
    lea eax, [rbp + 1]
    mov dword ptr [r8 + rbx*4 - 4], eax
    jmp filt
filtdone:
    test r10, r10
    jnz badexit
    mov byte ptr [rdi + rbx], 0

    mov r13, rdi
    mov r12, r8
    mov rbx, rdi
    lea rbp, [rip + tape]
    lea r9, [rip + dispatch]
    lea r10, [rip + outbuf]

nextop:
    movzx eax, byte ptr [rbx]
    inc rbx
    jmp qword ptr [r9 + rax*8]

op_inc_ptr:
    inc rbp
    jmp nextop
op_dec_ptr:
    dec rbp
    jmp nextop
op_inc_val:
    inc byte ptr [rbp]
    jmp nextop
op_dec_val:
    dec byte ptr [rbp]
    jmp nextop
op_out:
    mov al, byte ptr [rbp]
    mov byte ptr [r10], al
    inc r10
    lea rax, [rip + outbuf]
    add rax, 65536
    cmp r10, rax
    jb nextop
    call flush
    jmp nextop
op_in:
    xor eax, eax
    cmp r14, r15
    jae 1f
    movzx eax, byte ptr [r14]
    inc r14
1:  mov byte ptr [rbp], al
    jmp nextop
op_lbr:
    cmp byte ptr [rbp], 0
    jne nextop
    mov rax, rbx
    sub rax, r13
    dec rax
    mov eax, dword ptr [r12 + rax*4]
    lea rbx, [r13 + rax]
    jmp nextop
op_rbr:
    cmp byte ptr [rbp], 0
    je nextop
    mov rax, rbx
    sub rax, r13
    dec rax
    mov eax, dword ptr [r12 + rax*4]
    lea rbx, [r13 + rax]
    jmp nextop

op_end:
    call flush
    mov eax, 60
    xor edi, edi
    syscall
    jmp op_end

badexit:
    mov eax, 60
    mov edi, 1
    syscall
    jmp badexit

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r10
    sub rdx, rsi
    jz 1f
    mov eax, 1
    mov edi, 1
    syscall
    lea r10, [rip + outbuf]
1:  ret

.section .rodata
.align 8
dispatch:
    .quad op_end
    .quad op_inc_ptr
    .quad op_dec_ptr
    .quad op_inc_val
    .quad op_dec_val
    .quad op_out
    .quad op_in
    .quad op_lbr
    .quad op_rbr

cmdtab:
    .zero 43
    .byte 3, 6, 4, 5
    .zero 13
    .byte 2
    .byte 0
    .byte 1
    .zero 28
    .byte 7
    .byte 0
    .byte 8
    .zero 162

.bss
.align 16
inbuf:
    .zero 65536
prog:
    .zero 65544
jumptab:
    .zero 262152
bstack:
    .zero 262144
tape:
    .zero 30016
outbuf:
    .zero 65552
