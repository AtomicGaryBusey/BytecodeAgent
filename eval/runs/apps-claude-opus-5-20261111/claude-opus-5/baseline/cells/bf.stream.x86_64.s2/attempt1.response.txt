.globl _start

.text
_start:
    # ---- read all of standard input into inbuf ----
    xor r12d, r12d                  # total bytes read
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65540
    sub rdx, r12
    test rdx, rdx
    jle read_done
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # ---- locate the first '!' ----
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rsi + rcx], 33
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov rdx, r12                    # program length
    mov r13, r12                    # input start index (empty)
    jmp have_split
found_bang:
    mov rdx, rcx                    # program length
    lea r13, [rcx + 1]              # input start index
have_split:
    lea r14, [rsi + r13]            # input pointer
    lea r15, [rsi + r12]            # input end

    # ---- filter commands and build the jump table ----
    lea rdi, [rip + prog]
    lea r8, [rip + jumptab]
    lea r9, [rip + bstack]
    xor ecx, ecx                    # source index
    xor ebx, ebx                    # destination count
    xor r10d, r10d                  # bracket stack depth
filt:
    cmp rcx, rdx
    jae filt_done
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 62
    je k0
