.globl _start

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .zero 65536
    .balign 16
outbuf:
    .zero 80

.text

# write all of rsi, length rdx, to fd 1
wrout:
    push rbx
    mov rbx, rsi
1:
    test rdx, rdx
    jz 2f
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    sub rdx, rax
    jmp 1b
2:
    pop rbx
    ret

_start:
    # read all input
    xor r12, r12                 # total length
1:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz 2f
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    # r13 = current offset
    xor r13, r13

row_loop:
    cmp r13, r12
    jae done

    mov r14, r12
    sub r14, r13                 # remaining
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    # build line in outbuf
    lea rdi, [rip + outbuf]
    lea rbx, [rip + hexdig]

    # 8 hex digits of offset
    mov eax, r13d
    mov ecx, 8
2:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 2b

    mov byte ptr [rdi], 32
    mov byte ptr [rdi+1], 32
    add rdi, 2

    # hex bytes
    lea rsi, [rip + inbuf]
    add rsi, r13
    xor rcx, rcx
3:
    cmp rcx, r14
    jae 4f
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], al
    mov byte ptr [rdi+2], 32
    add rdi, 3
    jmp 5f
4:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi+1], 32
    mov byte ptr [rdi+2], 32
    add rdi, 3
5:
    cmp rcx, 7
    jne 6f
    mov byte ptr [rdi], 32
    inc rdi
6:
    inc rcx
    cmp rcx, 16
    jb 3b

    mov byte ptr [rdi], 32
    mov byte ptr [rdi+1], 124
    add rdi, 2

    xor rcx, rcx
7:
    cmp rcx, r14
    jae 8f
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb 9f
    cmp al, 0x7e
    jbe 10f
9:
    mov eax, 46
10:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    jmp 7b
8:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi+1], 10
    add rdi, 2

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    call wrout

    add r13, r14
    jmp row_loop

done:
    lea rdi, [rip + outbuf]
    lea rbx, [rip + hexdig]
    mov eax, r12d
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    mov byte ptr [rdi], 10

    lea rsi, [rip + outbuf]
    mov rdx, 9
    call wrout

    mov eax, 231
    xor edi, edi
    syscall
