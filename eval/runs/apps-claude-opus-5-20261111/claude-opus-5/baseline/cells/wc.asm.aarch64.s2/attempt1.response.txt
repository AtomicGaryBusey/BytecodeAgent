.globl _start

.section .bss
.balign 16
buf:
.zero 65536
outbuf:
.zero 64

.text
_start:
    mov     x19, #0                 // lines
    mov     x20, #0                 // words
    mov     x21, #0                 // bytes
    mov     x22, #0                 // in_word flag

    adrp    x25, buf
    add     x25, x25, :lo12:buf

read_loop:
    mov     x0, #0
    mov     x1, x25
    mov     x2, #65536
    mov     x8, #63
    svc     #0

    cmp     x0, #0
    b.le    read_done

    add     x21, x21, x0
    mov     x23, x0
    mov     x24, x25

inner:
    ldrb    w0, [x24], #1
    cmp     w0, #0x0a
    b.ne    1f
    add     x19, x19, #1
1:
    cmp     w0, #0x20
    b.eq    2f
    sub     w1, w0, #9
    cmp     w1, #4
    b.ls    2f
    cbnz    x22, 3f
    mov     x22, #1
    add     x20, x20, #1
    b       3f
2:
    mov     x22, #0
3:
    subs    x23, x23, #1
    b.ne    inner

    b       read_loop

read_done:
    adrp    x26, outbuf
    add     x26, x26, :lo12:outbuf
    mov     x27, x26

    mov     x0, x19
    mov     x1, x27
    bl      itoa
    mov     x27, x1
    mov     w0, #0x20
    strb    w0, [x27], #1

    mov     x0, x20
    mov     x1, x27
    bl      itoa
    mov     x27, x1
    mov     w0, #0x20
    strb    w0, [x27], #1

    mov     x0, x21
    mov     x1, x27
    bl      itoa
    mov     x27, x1
    mov     w0, #0x0a
    strb    w0, [x27], #1

    sub     x2, x27, x26
    mov     x0, #1
    mov     x1, x26
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       read_done

// x0 = value, x1 = output pointer; returns x1 advanced past digits
itoa:
    sub     sp, sp, #48
    mov     x6, sp
    mov     x4, #32
    mov     x2, #10
1:
    udiv    x3, x0, x2
    msub    x5, x3, x2, x0
    add     w5, w5, #48
    sub     x4, x4, #1
    strb    w5, [x6, x4]
    mov     x0, x3
    cbnz    x0, 1b
2:
    ldrb    w5, [x6, x4]
    strb    w5, [x1], #1
    add     x4, x4, #1
    cmp     x4, #32
    b.lt    2b
    add     sp, sp, #48
    ret
