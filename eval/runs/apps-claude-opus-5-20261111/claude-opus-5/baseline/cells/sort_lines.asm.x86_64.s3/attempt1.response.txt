.globl _start

.section .bss
.balign 16
buf:
.zero 40000
.balign 16
lines:
.zero 16400
.balign 16
aux:
.zero 16400
.balign 16
outbuf:
.zero 48000
.balign 8
v_lo:
.zero 8
v_mid:
.zero 8
v_hi:
.zero 8
v_p:
.zero 8
v_q:
.zero 8
v_k:
.zero 8

.text

_start:
    # ---- read all of stdin into buf ----
    xor r12, r12                      # total bytes read
read_loop:
    mov rax, 40000
    sub rax, r12
    test rax, rax
    jz read_done
    mov rdx, rax
    lea rsi, [rip + buf]
    add rsi, r12
    xor rdi, rdi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # ---- split into lines ----
    lea rbx, [rip + buf]
    xor r14, r14                      # i
    xor r13, r13                      # n
    xor r15, r15                      # start
parse_loop:
    cmp r14, r12
    jae parse_tail
    movzx eax, byte ptr [rbx + r14]
    cmp al, 10
    jne parse_next
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    lea rax, [rbx + r15]
    mov qword ptr [rdi], rax
    mov rcx, r14
    sub rcx, r15
    mov qword ptr [rdi + 8], rcx
    inc r13
    mov r15, r14
    inc r15
parse_next:
    inc r14
    jmp parse_loop
parse_tail:
    cmp r15, r12
    jae parse_done
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    lea rax, [rbx + r15]
    mov qword ptr [rdi], rax
    mov rcx, r12
    sub rcx, r15
    mov qword ptr [rdi + 8], rcx
    inc r13
parse_done:

    # ---- bottom-up merge sort over lines[0 .. r13) ----
    cmp r13, 1
    jbe sort_done
    mov rbx, 1                        # width
width_loop:
    cmp rbx, r13
    jae sort_done
    xor r14, r14                      # i
run_loop:
    cmp r14, r13
    jae width_next
    mov qword ptr [rip + v_lo], r14
    lea rax, [r14 + rbx]
    mov rcx, r13
    cmp rax, rcx
    cmova rax, rcx
    mov qword ptr [rip + v_mid], rax
    mov rdx, rbx
    add rdx, rbx
    add rdx, r14
    cmp rdx, rcx
    cmova rdx, rcx
    mov qword ptr [rip + v_hi], rdx
    cmp rax, rdx
    jae run_next
    call merge
run_next:
    lea r14, [r14 + rbx*2]
    jmp run_loop
width_next:
    add rbx, rbx
    jmp width_loop
sort_done:

    # ---- build output ----
    lea rdi, [rip + outbuf]
    xor r14, r14
out_loop:
    cmp r14, r13
    jae out_done
    mov rax, r14
    shl rax, 4
    lea rbx, [rip + lines]
    add rbx, rax
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r14
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    xor edi, edi
    mov eax, 231
    syscall
    xor edi, edi
    mov eax, 60
    syscall
hang:
    jmp hang

# merge lines[v_lo .. v_mid) and lines[v_mid .. v_hi) -> aux, copy back
merge:
    mov rax, qword ptr [rip + v_lo]
    mov qword ptr [rip + v_p], rax
    mov qword ptr [rip + v_k], rax
    mov rax, qword ptr [rip + v_mid]
    mov qword ptr [rip + v_q], rax
merge_loop:
    mov rax, qword ptr [rip + v_k]
    cmp rax, qword ptr [rip + v_hi]
    jae merge_copy
    mov rax, qword ptr [rip + v_p]
    cmp rax, qword ptr [rip + v_mid]
    jae merge_take_q
    mov rax, qword ptr [rip + v_q]
    cmp rax, qword ptr [rip + v_hi]
    jae merge_take_p
    mov rax, qword ptr [rip + v_p]
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    mov rax, qword ptr [rip + v_q]
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    call cmp_lines
    test eax, eax
    jg merge_take_q
merge_take_p:
    mov rax, qword ptr [rip + v_p]
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rdx, qword ptr [rsi]
    mov rcx, qword ptr [rsi + 8]
    mov rax, qword ptr [rip + v_k]
    shl rax, 4
    lea rdi, [rip + aux]
    add rdi, rax
    mov qword ptr [rdi], rdx
    mov qword ptr [rdi + 8], rcx
    inc qword ptr [rip + v_p]
    inc qword ptr [rip + v_k]
    jmp merge_loop
merge_take_q:
    mov rax, qword ptr [rip + v_q]
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rdx, qword ptr [rsi]
    mov rcx, qword ptr [rsi + 8]
    mov rax, qword ptr [rip + v_k]
    shl rax, 4
    lea rdi, [rip + aux]
    add rdi, rax
    mov qword ptr [rdi], rdx
    mov qword ptr [rdi + 8], rcx
    inc qword ptr [rip + v_q]
    inc qword ptr [rip + v_k]
    jmp merge_loop
merge_copy:
    mov rax, qword ptr [rip + v_lo]
    shl rax, 4
    lea rsi, [rip + aux]
    add rsi, rax
    lea rdi, [rip + lines]
    add rdi, rax
    mov rcx, qword ptr [rip + v_hi]
    sub rcx, qword ptr [rip + v_lo]
    shl rcx, 1
    rep movsq
    ret

# cmp_lines: rdi = &A, rsi = &B -> eax < 0 if A < B, 0 if equal, > 0 if A > B
cmp_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov rcx, r9
    cmp r11, rcx
    cmovb rcx, r11
    xor rdx, rdx
cmp_byte_loop:
    cmp rdx, rcx
    jae cmp_by_len
    movzx eax, byte ptr [r8 + rdx]
    movzx edi, byte ptr [r10 + rdx]
    cmp eax, edi
    jne cmp_diff
    inc rdx
    jmp cmp_byte_loop
cmp_diff:
    sub eax, edi
    ret
cmp_by_len:
    mov eax, r9d
    sub eax, r11d
    ret
