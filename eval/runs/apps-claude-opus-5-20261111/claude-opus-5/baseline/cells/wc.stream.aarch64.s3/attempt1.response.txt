.globl _start

.text
_start:
    mov     x19, #0                 // lines
    mov     x20, #0                 // words
    mov     x21, #0                 // bytes
    mov     x22, #0                 // in-word flag

read_loop:
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x0, #0
    mov     x2, #4096
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x21, x21, x0
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    add     x3, x1, x0

byte_loop:
    cmp     x1, x3
    b.eq    read_loop
    ldrb    w4, [x1], #1
    cmp     w4, #0x0a
    cset    w5, eq
    add     x19, x19, x5
    cmp     w4, #0x20
    b.eq    is_ws
    sub     w6, w4, #0x09
    cmp     w6, #4
    b.ls    is_ws
    cbnz    x22, byte_loop
    mov     x22, #1
    add     x20, x20, #1
    b       byte_loop
is_ws:
    mov     x22, #0
    b       byte_loop

done:
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x0, x19
    bl      itoa
    mov     w0, #0x20
    strb    w0, [x23], #1
    mov     x0, x20
    bl      itoa
    mov     w0, #0x20
    strb    w0, [x23], #1
    mov     x0, x21
    bl      itoa
    mov     w0, #0x0a
    strb    w0, [x23], #1

    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x23, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0
    mov     x0, #0
    mov     x8, #94
    svc     #0

// x0 = value, appends decimal digits at [x23], advances x23
itoa:
    sub     sp, sp, #48
    mov     x4, sp
    add     x4, x4, #40
    mov     x5, #10
    mov     x6, #0
1:
    udiv    x7, x0, x5
    msub    x9, x7, x5, x0
    add     w9, w9, #48
    strb    w9, [x4], #-1
    add     x6, x6, #1
    mov     x0, x7
    cbnz    x0, 1b
    add     x4, x4, #1
2:
    ldrb    w9, [x4], #1
    strb    w9, [x23], #1
    subs    x6, x6, #1
    b.ne    2b
    add     sp, sp, #48
    ret

.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 128
