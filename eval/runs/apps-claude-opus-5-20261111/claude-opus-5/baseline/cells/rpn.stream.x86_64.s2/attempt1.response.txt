.globl _start

.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    mov edx, 65536
    sub rdx, r13
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r15, [r12 + r13]
    mov rbx, r12
    xor r14, r14

line_loop:
    cmp rbx, r15
    jae all_done
    mov rbp, rbx
find_nl:
    cmp rbp, r15
    jae found_nl
    cmp byte ptr [rbp], 10
    je found_nl
    inc rbp
    jmp find_nl
found_nl:
    xor r13d, r13d

tok_loop:
    cmp rbx, rbp
    jae tok_end
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je 1f
    cmp al, 9
    jne tok_start_found
1:
    inc rbx
    jmp tok_loop

tok_start_found:
    mov rsi, rbx
scan_tok:
    cmp rbx, rbp
    jae scan_done
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je scan_done
    cmp al, 9
    je scan_done
    inc rbx
    jmp scan_tok
scan_done:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne not_op
    movzx eax, byte ptr [rsi]
    cmp al, 43
    je do_op
    cmp al, 45
    je do_op
    cmp al, 42
    je do_op
    cmp al, 47
    je do_op
    cmp al, 37
    je do_op

not_op:
    mov rdx, rsi
    xor r8d, r8d
    movzx eax, byte ptr [rdx]
    cmp al, 45
    jne 1f
    mov r8d, 1
    inc rdx
1:
    mov r9, rbx
    sub r9, rdx
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor rax, rax
2:
    movzx ecx, byte ptr [rdx]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    lea rax, [rax + rax*4]
    lea rax, [rax + rax]
    add rax, rcx
    inc rdx
    cmp rdx, rbx
    jb 2b
    test r8d, r8d
    jz 3f
    neg rax
3:
    lea rdi, [rip + vstack]
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

do_op:
    mov r9d, eax
    cmp r13, 2
    jb line_error
    lea rdi, [rip + vstack]
    mov r10, qword ptr [rdi + r13*8 - 8]
    mov r11, qword ptr [rdi + r13*8 - 16]
    dec r13
    cmp r9b, 43
    je op_add
    cmp r9b, 45
    je op_sub
    cmp r9b, 42
    je op_mul
    test r10, r10
    jz line_error
    cmp r10, -1
    jne 1f
    mov rax, 0x8000000000000000
    cmp r11, rax
    je line_error
1:
    mov rax, r11
    cqo
    idiv r10
    cmp r9b, 37
    jne store_res
    mov rax, rdx
    jmp store_res
op_add:
    mov rax, r11
    add rax, r10
    jmp store_res
op_sub:
    mov rax, r11
    sub rax, r10
    jmp store_res
op_mul:
    mov rax, r11
    imul rax, r10
store_res:
    lea rdi, [rip + vstack]
    mov qword ptr [rdi + r13*8 - 8], rax
    jmp tok_loop

tok_end:
    cmp r13, 1
    jne line_error
    lea rdi, [rip + vstack]
    mov rax, qword ptr [rdi]
    lea rdi, [rip + numbuf + 24]
    xor r8d, r8d
    test rax, rax
    jns 1f
    mov r8d, 1
    neg rax
1:
    mov r9, 10
2:
    xor edx, edx
    div r9
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    test r8d, r8d
    jz 3f
    dec rdi
    mov byte ptr [rdi], 45
3:
    lea rcx, [rip + outbuf]
    add rcx, r14
    lea rsi, [rip + numbuf + 24]
4:
    movzx eax, byte ptr [rdi]
    mov byte ptr [rcx], al
    inc rdi
    inc rcx
    inc r14
    cmp rdi, rsi
    jb 4b
    mov byte ptr [rcx], 10
    inc r14
    jmp next_line

line_error:
    lea rcx, [rip + outbuf]
    add rcx, r14
    mov eax, 0x6f727265
    mov dword ptr [rcx], eax
    mov word ptr [rcx + 4], 0x0a72
    add r14, 6

next_line:
    cmp r14, 65536
    jb 1f
    call flush
1:
    lea rbx, [rbp + 1]
    jmp line_loop

all_done:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall

flush:
    test r14, r14
    jz flush_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r14
    syscall
    xor r14d, r14d
flush_done:
    ret

.bss
.align 16
inbuf:
    .zero 65600
outbuf:
    .zero 65600
vstack:
    .zero 16384
numbuf:
    .zero 32
