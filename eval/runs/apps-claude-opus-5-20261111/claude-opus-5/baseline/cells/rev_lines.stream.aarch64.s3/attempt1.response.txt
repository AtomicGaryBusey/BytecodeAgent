.globl _start

.text
_start:
    adrp    x20, inbuf
    add     x20, x20, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x19, #0
    mov     x26, #1
    lsl     x26, x26, #16          // 65536 = capacity

read_loop:
    cmp     x19, x26
    b.hs    read_done
    mov     x0, #0
    add     x1, x20, x19
    sub     x2, x26, x19
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x19, x19, x0
    b       read_loop
read_done:

    mov     x22, #0                // start index
    mov     x23, #0                // out length

outer:
    cmp     x22, x19
    b.hs    emit_done
    mov     x24, x22
scan:
    cmp     x24, x19
    b.hs    found
    ldrb    w0, [x20, x24]
    cmp     w0, #10
    b.eq    found
    add     x24, x24, #1
    b       scan
found:
    mov     x25, x24
rev:
    cmp     x25, x22
    b.ls    revdone
    sub     x25, x25, #1
    ldrb    w0, [x20, x25]
    strb    w0, [x21, x23]
    add     x23, x23, #1
    b       rev
revdone:
    mov     w0, #10
    strb    w0, [x21, x23]
    add     x23, x23, #1
    add     x22, x24, #1
    b       outer

emit_done:
    mov     x27, #0                // bytes written
write_loop:
    cmp     x27, x23
    b.hs    write_done
    mov     x0, #1
    add     x1, x21, x27
    sub     x2, x23, x27
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    write_done
    add     x27, x27, x0
    b       write_loop
write_done:

    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       write_done

.bss
.balign 16
inbuf:
    .zero   65536
outbuf:
    .zero   65544
