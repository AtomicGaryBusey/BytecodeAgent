.globl _start

.text
_start:
    # ---- read all of standard input ----
    xor r12d, r12d                  # total bytes read
read_loop:
    mov edx, 65536
    sub edx, r12d
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # ---- split program / input at first '!' ----
    lea rdi, [rip + inbuf]
    xor ecx, ecx
find_bang:
    cmp rcx, r12
    je no_bang
    cmp byte ptr [rdi + rcx], 0x21
    je got_bang
    inc rcx
    jmp find_bang
got_bang:
    lea r13, [rdi + rcx + 1]        # input start
    lea r14, [rdi + r12]            # input end
    mov r12, rcx                    # program length
    jmp have_split
no_bang:
    lea r13, [rdi + r12]
    mov r14, r13
have_split:

    # ---- validate brackets and build jump table ----
    lea rsi, [rip + jtab]
    lea r8, [rip + bstack]
    xor ecx, ecx                    # index
    xor edx, edx                    # stack depth
scan:
    cmp rcx, r12
    je scan_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x5B                    # '['
    je is_open
    cmp al, 0x5D                    # ']'
    je is_close
    inc rcx
    jmp scan
is_open:
    mov qword ptr [r8 + rdx*8], rcx
    inc rdx
    inc rcx
    jmp scan
is_close:
    test rdx, rdx
    jz bad
    dec rdx
    mov r9, qword ptr [r8 + rdx*8]
    lea rax, [rdi + rcx + 1]
    mov qword ptr [rsi + r9*8], rax
    lea rax, [rdi + r9 + 1]
    mov qword ptr [rsi + rcx*8], rax
    inc rcx
    jmp scan
scan_done:
    test rdx, rdx
    jnz bad

    # ---- set up interpreter state ----
    mov rbx, rdi                    # instruction pointer
    add r12, rdi                    # program end
    mov rcx, rdi
    shl rcx, 3
    lea r10, [rip + jtab]
    sub r10, rcx                    # jtab base biased by program base
    lea rbp, [rip + tape]
    lea r15, [rip + outbuf]
    lea r9, [rip + outbuf]
    add r9, 65536                   # output buffer limit

run:
    cmp rbx, r12
    jae finish
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 0x2B                    # '+'
    je c_plus
    cmp al, 0x2D                    # '-'
    je c_minus
    cmp al, 0x3E                    # '>'
    je c_gt
    cmp al, 0x3C                    # '<'
    je c_lt
    cmp al, 0x2E                    # '.'
    je c_dot
    cmp al, 0x5B                    # '['
    je c_open
    cmp al, 0x5D                    # ']'
    je c_close
    cmp al, 0x2C                    # ','
    je c_comma
    jmp run

c_plus:
    inc byte ptr [rbp]
    jmp run
c_minus:
    dec byte ptr [rbp]
    jmp run
c_gt:
    inc rbp
    jmp run
c_lt:
    dec rbp
    jmp run
c_open:
    cmp byte ptr [rbp], 0
    jne run
    mov rbx, qword ptr [r10 + rbx*8 - 8]
    jmp run
c_close:
    cmp byte ptr [rbp], 0
    je run
    mov rbx, qword ptr [r10 + rbx*8 - 8]
    jmp run
c_dot:
    mov al, byte ptr [rbp]
    mov byte ptr [r15], al
    inc r15
    cmp r15, r9
    jb run
    call flush
    jmp run
c_comma:
    cmp r13, r14
    jae c_comma_eof
    movzx eax, byte ptr [r13]
    inc r13
    mov byte ptr [rbp], al
    jmp run
c_comma_eof:
    mov byte ptr [rbp], 0
    jmp run

finish:
    call flush
    xor edi, edi
    mov eax, 231
    syscall
    jmp finish

bad:
    mov edi, 1
    mov eax, 231
    syscall
    jmp bad

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r15, [rip + outbuf]
    ret

.bss
.balign 16
inbuf:
    .zero 65552
.balign 16
tape:
    .zero 30016
.balign 16
outbuf:
    .zero 65552
.balign 16
bstack:
    .zero 8192
.balign 16
jtab:
    .zero 524288
