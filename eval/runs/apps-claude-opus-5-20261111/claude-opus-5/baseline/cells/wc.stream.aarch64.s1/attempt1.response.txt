.globl _start

.section .bss
.balign 16
buf:
.zero 4096
outbuf:
.zero 96

.text
_start:
    mov     x20, #0                 // lines
    mov     x21, #0                 // words
    mov     x22, #0                 // bytes
    mov     w23, #0                 // in-word flag

read_loop:
    mov     x0, #0
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #4096
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x22, x22, x0
    adrp    x10, buf
    add     x10, x10, :lo12:buf
    add     x11, x10, x0

byte_loop:
    ldrb    w5, [x10], #1
    cmp     w5, #0x0A
    b.ne    1f
    add     x20, x20, #1
1:
    cmp     w5, #0x20
    b.eq    2f
    sub     w6, w5, #9
    cmp     w6, #4
    b.ls    2f
    cbnz    w23, 3f
    add     x21, x21, #1
    mov     w23, #1
    b       3f
2:
    mov     w23, #0
3:
    cmp     x10, x11
    b.lo    byte_loop
    b       read_loop

done:
    adrp    x12, outbuf
    add     x12, x12, :lo12:outbuf
    add     x19, x12, #96           // end of buffer
    mov     x12, x19
    mov     w7, #0x0A
    sub     x12, x12, #1
    strb    w7, [x12]

    mov     x0, x22
    bl      emit
    mov     w7, #0x20
    sub     x12, x12, #1
    strb    w7, [x12]

    mov     x0, x21
    bl      emit
    mov     w7, #0x20
    sub     x12, x12, #1
    strb    w7, [x12]

    mov     x0, x20
    bl      emit

    mov     x1, x12
    sub     x2, x19, x12
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       done

emit:
    mov     x13, #10
1:
    udiv    x14, x0, x13
    msub    x15, x14, x13, x0
    add     w15, w15, #48
    sub     x12, x12, #1
    strb    w15, [x12]
    mov     x0, x14
    cbnz    x0, 1b
    ret
