.globl _start

.equ IN_SZ, 65536
.equ OUT_SZ, 65536

.text
_start:
    // set up output buffer pointer
    adrp    x26, outbuf
    add     x26, x26, :lo12:outbuf
    mov     x25, x26
    // value stack base
    adrp    x27, vstack
    add     x27, x27, :lo12:vstack
    // input buffer
    adrp    x23, inbuf
    add     x23, x23, :lo12:inbuf
    mov     x24, x23
    mov     x19, #IN_SZ
    add     x19, x23, x19          // buffer limit

rd_loop:
    sub     x2, x19, x24
    cbz     x2, rd_done
    mov     x0, #0
    mov     x1, x24
    mov     x8, #63
    svc     #0
    cmp     x0, #1
    b.lt    rd_done
    add     x24, x24, x0
    b       rd_loop
rd_done:

line_loop:
    cmp     x23, x24
    b.hs    all_done
    mov     x20, x23               // line start
    mov     x21, x23               // scanner
scan_nl:
    cmp     x21, x24
    b.hs    scan_done
    ldrb    w0, [x21]
    cmp     w0, #10
    b.eq    scan_done
    add     x21, x21, #1
    b       scan_nl
scan_done:
    mov     x22, x21               // next line start
    cmp     x22, x24
    b.hs    have_line
    add     x22, x22, #1
have_line:
    mov     x19, x20               // cursor
    mov     x28, #0                // stack depth

tok_loop:
skip_sep:
    cmp     x19, x21
    b.hs    tokens_done
    ldrb    w0, [x19]
    cmp     w0, #32
    b.eq    adv_sep
    cmp     w0, #9
    b.ne    tok_start
adv_sep:
    add     x19, x19, #1
    b       skip_sep

tok_start:
    mov     x1, x19                // token start
scan_tok:
    cmp     x19, x21
    b.hs    tok_end
    ldrb    w0, [x19]
    cmp     w0, #32
    b.eq    tok_end
    cmp     w0, #9
    b.eq    tok_end
    add     x19, x19, #1
    b       scan_tok
tok_end:
    sub     x2, x19, x1            // length
    cmp     x2, #1
    b.ne    lit_parse
    ldrb    w0, [x1]
    cmp     w0, #43                // '+'
    b.eq    do_op
    cmp     w0, #45                // '-'
    b.eq    do_op
    cmp     w0, #42                // '*'
    b.eq    do_op
    cmp     w0, #47                // '/'
    b.eq    do_op
    cmp     w0, #37                // '%'
    b.eq    do_op

lit_parse:
    mov     x3, #0                 // negative flag
    mov     x4, x1
    ldrb    w0, [x4]
    cmp     w0, #45                // '-'
    b.ne    no_sign
    mov     x3, #1
    add     x4, x4, #1
no_sign:
    sub     x5, x19, x4            // digit count
    cmp     x5, #1
    b.lt    line_error
    cmp     x5, #18
    b.gt    line_error
    mov     x6, #0
    mov     x7, #10
dig_loop:
    cmp     x4, x19
    b.hs    dig_done
    ldrb    w0, [x4], #1
    sub     w0, w0, #48
    cmp     w0, #9
    b.hi    line_error
    mul     x6, x6, x7
    add     x6, x6, x0
    b       dig_loop
dig_done:
    cbz     x3, lit_push
    neg     x6, x6
lit_push:
    str     x6, [x27, x28, lsl #3]
    add     x28, x28, #1
    b       tok_loop

do_op:
    cmp     x28, #2
    b.lt    line_error
    sub     x28, x28, #2
    ldr     x9, [x27, x28, lsl #3]         // a
    add     x10, x28, #1
    ldr     x11, [x27, x10, lsl #3]        // b
    cmp     w0, #43
    b.eq    op_add
    cmp     w0, #45
    b.eq    op_sub
    cmp     w0, #42
    b.eq    op_mul
    cmp     w0, #47
    b.eq    op_div

op_mod:
    cbz     x11, line_error
    cmn     x11, #1
    b.ne    do_mod
    mov     x12, #1
    lsl     x12, x12, #63
    cmp     x9, x12
    b.eq    line_error
do_mod:
    sdiv    x12, x9, x11
    msub    x6, x12, x11, x9
    b       push_res

op_div:
    cbz     x11, line_error
    cmn     x11, #1
    b.ne    do_div
    mov     x12, #1
    lsl     x12, x12, #63
    cmp     x9, x12
    b.eq    line_error
do_div:
    sdiv    x6, x9, x11
    b       push_res

op_add:
    add     x6, x9, x11
    b       push_res
op_sub:
    sub     x6, x9, x11
    b       push_res
op_mul:
    mul     x6, x9, x11
push_res:
    str     x6, [x27, x28, lsl #3]
    add     x28, x28, #1
    b       tok_loop

tokens_done:
    cmp     x28, #1
    b.ne    line_error
    ldr     x0, [x27]
    bl      emit_num
    b       line_end
line_error:
    bl      emit_err
line_end:
    adrp    x0, outbuf
    add     x0, x0, :lo12:outbuf
    sub     x1, x25, x0
    mov     x2, #(OUT_SZ - 64)
    cmp     x1, x2
    b.lo    no_flush
    bl      flush_out
no_flush:
    mov     x23, x22
    b       line_loop

all_done:
    bl      flush_out
    mov     x0, #0
    mov     x8, #93
    svc     #0
    mov     x0, #0
    mov     x8, #94
    svc     #0

// ---- append decimal representation of x0 to output buffer ----
emit_num:
    mov     x1, x0
    tbz     x1, #63, en_abs
    mov     w2, #45
    strb    w2, [x25], #1
    neg     x1, x1
en_abs:
    adrp    x3, numbuf
    add     x3, x3, :lo12:numbuf
    add     x3, x3, #24
    mov     x4, x3
    mov     x5, #10
en_conv:
    udiv    x6, x1, x5
    msub    x7, x6, x5, x1
    add     x7, x7, #48
    sub     x4, x4, #1
    strb    w7, [x4]
    mov     x1, x6
    cbnz    x1, en_conv
en_copy:
    ldrb    w7, [x4], #1
    strb    w7, [x25], #1
    cmp     x4, x3
    b.lo    en_copy
    mov     w7, #10
    strb    w7, [x25], #1
    ret

// ---- append "error\n" ----
emit_err:
    adrp    x1, errmsg
    add     x1, x1, :lo12:errmsg
    mov     x2, #6
ee_loop:
    ldrb    w3, [x1], #1
    strb    w3, [x25], #1
    subs    x2, x2, #1
    b.ne    ee_loop
    ret

// ---- write out and reset buffer ----
flush_out:
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x25, x1
    cbz     x2, fo_done
    mov     x0, #1
    mov     x8, #64
    svc     #0
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
fo_done:
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
numbuf:
    .zero 32
vstack:
    .zero 262400
