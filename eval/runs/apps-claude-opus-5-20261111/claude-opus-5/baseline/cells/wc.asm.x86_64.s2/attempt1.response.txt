buf:

.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # prev in word flag

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    mov rcx, rax
    add r14, rax
    lea rsi, [rip + buf]
    lea rbx, [rip + ws_table]
scan_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne no_nl
    inc r12
no_nl:
    movzx edx, byte ptr [rbx + rax]
    # edx = 1 if whitespace, 0 if word byte
    xor edx, 1               # edx = 1 if word byte
    mov eax, edx
    sub eax, r15d            # 1 - prev
    cmp eax, 1
    jne no_word
    inc r13
no_word:
    mov r15d, edx
    inc rsi
    dec rcx
    jnz scan_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    dec rdi
    mov rax, r14
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r13
    call emit
    mov byte ptr [rdi], 32
    dec rdi
    mov rax, r12
    call emit
    inc rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall

# emit: rax = value, rdi = pointer to last free byte (writes backwards), leaves rdi on next free
emit:
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz 1b
    ret
