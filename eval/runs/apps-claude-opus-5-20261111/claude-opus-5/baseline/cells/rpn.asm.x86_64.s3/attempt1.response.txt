.globl _start

.section .rodata
errmsg: .ascii "error\n"

.equ INCAP, 65536
.equ OUTCAP, 524288
.equ OUTLIM, OUTCAP - 32

.bss
.balign 16
inbuf:  .zero 65552
.balign 16
outbuf: .zero 524288
.balign 16
vstack: .zero 8192

.text

_start:
    xor ebx, ebx
read_loop:
    mov rdx, INCAP
    sub rdx, rbx
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [r12 + rbx]
    xor r14d, r14d

line_loop:
    cmp r12, r13
    jae all_done
    mov rbx, r12
scan_nl:
    cmp rbx, r13
    jae scan_nl_none
    cmp byte ptr [rbx], 10
    je scan_nl_hit
    inc rbx
    jmp scan_nl
scan_nl_none:
    mov rbp, rbx
    jmp scan_nl_out
scan_nl_hit:
    lea rbp, [rbx + 1]
scan_nl_out:
    xor r15d, r15d

tok_loop:
    cmp r12, rbx
    jae tok_end
    movzx eax, byte ptr [r12]
    cmp al, 32
    je tok_skip
    cmp al, 9
    je tok_skip
    jmp got_tok
tok_skip:
    inc r12
    jmp tok_loop

got_tok:
    mov r8, r12
tscan:
    cmp r12, rbx
    jae tscan_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je tscan_done
    cmp al, 9
    je tscan_done
    inc r12
    jmp tscan
tscan_done:
    mov r9, r12
    sub r9, r8

    cmp r9, 1
    jne not_op
    movzx eax, byte ptr [r8]
    cmp al, 43
    je do_op
    cmp al, 45
    je do_op
    cmp al, 42
    je do_op
    cmp al, 47
    je do_op
    cmp al, 37
    je do_op

not_op:
    mov r10, r8
    xor r11d, r11d
    cmp byte ptr [r10], 45
    jne no_sign
    mov r11d, 1
    inc r10
no_sign:
    mov rcx, r12
    sub rcx, r10
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
dig_loop:
    movzx edx, byte ptr [r10]
    sub dl, 48
    cmp dl, 9
    ja line_error
    lea rax, [rax + rax*4]
    lea rax, [rax + rax]
    add rax, rdx
    inc r10
    dec rcx
    jnz dig_loop
    test r11d, r11d
    jz push_res
    neg rax
    jmp push_res

do_op:
    cmp r15, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r15*8 - 8]
    mov rsi, qword ptr [rdx + r15*8 - 16]
    sub r15, 2
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div

op_mod:
    test rcx, rcx
    jz line_error
    mov rax, rsi
    cmp rcx, -1
    jne op_mod_go
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
op_mod_go:
    cqo
    idiv rcx
    mov rax, rdx
    jmp push_res

op_div:
    test rcx, rcx
    jz line_error
    mov rax, rsi
    cmp rcx, -1
    jne op_div_go
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
op_div_go:
    cqo
    idiv rcx
    jmp push_res

op_add:
    mov rax, rsi
    add rax, rcx
    jmp push_res
op_sub:
    mov rax, rsi
    sub rax, rcx
    jmp push_res
op_mul:
    mov rax, rsi
    imul rax, rcx
    jmp push_res

push_res:
    cmp r15, 1024
    jae line_error
    lea rdx, [rip + vstack]
    mov qword ptr [rdx + r15*8], rax
    inc r15
    jmp tok_loop

tok_end:
    cmp r15, 1
    jne line_error
    lea rdx, [rip + vstack]
    mov rax, qword ptr [rdx]
    call emit_num
    jmp next_line

line_error:
    call flush_check
    lea rdi, [rip + outbuf]
    add rdi, r14
    lea rsi, [rip + errmsg]
    mov ecx, 6
err_cp:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jnz err_cp
    add r14, 6
    jmp next_line

next_line:
    mov r12, rbp
    jmp line_loop

all_done:
    call do_flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp all_done

flush_check:
    cmp r14, OUTLIM
    jb fc_ret
    call do_flush
fc_ret:
    ret

do_flush:
    mov rdx, r14
    test rdx, rdx
    jz df_ret
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall
    xor r14d, r14d
df_ret:
    ret

emit_num:
    call flush_check
    lea rdi, [rip + outbuf]
    add rdi, r14
    test rax, rax
    jns en_pos
    mov byte ptr [rdi], 45
    inc rdi
    inc r14
    neg rax
en_pos:
    sub rsp, 32
    lea rsi, [rsp + 32]
    mov r8, 10
en_div:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz en_div
    lea rcx, [rsp + 32]
    sub rcx, rsi
    add r14, rcx
en_cp:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz en_cp
    mov byte ptr [rdi], 10
    inc r14
    add rsp, 32
    ret
