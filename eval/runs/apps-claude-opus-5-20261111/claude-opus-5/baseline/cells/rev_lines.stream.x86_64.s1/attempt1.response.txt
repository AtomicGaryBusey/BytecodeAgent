.globl _start

.text
_start:
    lea rsi, [rip + inbuf]
    xor r12, r12                  # total bytes read

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov eax, 0
    xor edi, edi
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13                  # line start index
    xor r14, r14                  # output length

scan_loop:
    cmp r13, r12
    jae finish
    # find newline from r13
    mov rbx, r13
find_nl:
    cmp rbx, r12
    jae no_nl
    lea rcx, [rip + inbuf]
    cmp byte ptr [rcx + rbx], 10
    je got_nl
    inc rbx
    jmp find_nl

no_nl:
    # final line [r13, r12)
    mov rbx, r12
got_nl:
    # emit bytes [r13, rbx) reversed, then newline
    mov rcx, rbx                  # rcx = one past end
rev_emit:
    cmp rcx, r13
    jbe rev_done
    dec rcx
    lea rdi, [rip + inbuf]
    movzx eax, byte ptr [rdi + rcx]
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], al
    inc r14
    jmp rev_emit
rev_done:
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], 10
    inc r14
    lea r13, [rbx + 1]
    jmp scan_loop

finish:
    test r14, r14
    jz do_exit
    xor rbx, rbx
write_loop:
    cmp rbx, r14
    jae do_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jle do_exit
    add rbx, rax
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65600
