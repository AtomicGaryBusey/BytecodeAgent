.globl _start

.text
_start:
    xor r12, r12            # total bytes read
read_loop:
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r15, [rip + buf]
    lea r14, [rip + out]
    xor r13, r13            # output length
    xor rbx, rbx            # current index into buf (line start)
process:
    cmp rbx, r12
    jae finish
    mov rcx, rbx
scan:
    cmp rcx, r12
    jae scan_end
    cmp byte ptr [r15 + rcx], 10
    je scan_end
    inc rcx
    jmp scan
scan_end:
    mov rdx, rcx            # save end index
    mov r9, rcx
    sub r9, rbx             # length
    mov rdi, r14
    add rdi, r13
    mov rsi, rdx
    dec rsi                 # last byte index
    mov r8, r9
copy_loop:
    test r8, r8
    jz copy_done
    mov al, byte ptr [r15 + rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    dec r8
    jmp copy_loop
copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    mov r13, rdi
    sub r13, r14
    mov rbx, rdx
    inc rbx
    jmp process

finish:
    xor r8, r8
write_loop:
    mov rdx, r13
    sub rdx, r8
    jz exit_prog
    mov rsi, r14
    add rsi, r8
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle exit_prog
    add r8, rax
    jmp write_loop

exit_prog:
    mov rax, 60
    xor edi, edi
    syscall
    hlt

.bss
.balign 16
buf:
    .zero 65536
out:
    .zero 65552
