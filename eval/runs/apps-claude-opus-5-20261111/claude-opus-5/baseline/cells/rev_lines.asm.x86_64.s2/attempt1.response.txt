.globl _start

.text
_start:
    lea rsi, [rip + buf]
    xor r12, r12            # total bytes read
read_loop:
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13            # output length
    xor rbx, rbx            # current index into buf (line start)
process:
    cmp rbx, r12
    jae finish
    # find newline starting at rbx
    mov rcx, rbx
scan:
    cmp rcx, r12
    jae scan_end
    cmp byte ptr [rip + buf + rcx], 10
    je scan_end
    inc rcx
    jmp scan
scan_end:
    # line is [rbx, rcx)
    mov rdx, rcx            # save end
    mov rax, rcx
    sub rax, rbx            # length
    # copy reversed into out at r13
    lea rdi, [rip + out]
    add rdi, r13
    mov rsi, rdx
    dec rsi                 # last byte index
    mov r8, rax
copy_loop:
    test r8, r8
    jz copy_done
    mov al, byte ptr [rip + buf + rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    dec r8
    jmp copy_loop
copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rax, [rip + out]
    sub rdi, rax
    mov r13, rdi
    # advance rbx past newline
    mov rbx, rdx
    inc rbx
    jmp process

finish:
    test r13, r13
    jz exit_prog
    lea rsi, [rip + out]
    xor r14, r14
write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out]
    add rsi, r14
    mov rdx, r13
    sub rdx, r14
    jz exit_prog
    syscall
    test rax, rax
    jle exit_prog
    add r14, rax
    jmp write_loop

exit_prog:
    mov rax, 60
    xor edi, edi
    syscall
    hlt

.bss
.balign 16
buf:
    .zero 65536
out:
    .zero 65540
