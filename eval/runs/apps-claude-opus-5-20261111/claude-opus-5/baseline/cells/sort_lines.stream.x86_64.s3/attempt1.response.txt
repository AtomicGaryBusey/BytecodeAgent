.globl _start

.text
_start:
    lea r14, [rip + inbuf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r14
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # split into lines
    lea rbp, [rip + lines]
    xor rbx, rbx
    xor r13, r13
scan_loop:
    cmp rbx, r12
    jae scan_done
    cmp r13, 1024
    jae scan_done
    mov r8, rbx
find_loop:
    cmp rbx, r12
    jae store_noeol
    cmp byte ptr [r14 + rbx], 10
    je store_eol
    inc rbx
    jmp find_loop
store_eol:
    mov rax, rbx
    sub rax, r8
    shl rax, 32
    or rax, r8
    mov qword ptr [rbp + r13*8], rax
    inc r13
    inc rbx
    jmp scan_loop
store_noeol:
    mov rax, r12
    sub rax, r8
    shl rax, 32
    or rax, r8
    mov qword ptr [rbp + r13*8], rax
    inc r13
scan_done:

    # bottom-up merge sort
    lea rsi, [rip + lines]
    lea rdi, [rip + aux]
    mov r15, 1
msort_loop:
    cmp r15, r13
    jae msort_done
    xor r8, r8
pass_loop:
    cmp r8, r13
    jae pass_done
    mov r9, r8
    add r9, r15
    cmp r9, r13
    jbe 1f
    mov r9, r13
1:
    lea r10, [r8 + r15*2]
    cmp r10, r13
    jbe 1f
    mov r10, r13
1:
    mov r11, r8
    mov rbx, r9
    mov rcx, r8
merge_loop:
    cmp rcx, r10
    jae merge_done
    cmp r11, r9
    jae take_b
    cmp rbx, r10
    jae take_a
    mov rax, qword ptr [rsi + r11*8]
    mov rdx, qword ptr [rsi + rbx*8]
    call cmpline
    test eax, eax
    jnz take_a
take_b:
    mov rax, qword ptr [rsi + rbx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rbx
    inc rcx
    jmp merge_loop
take_a:
    mov rax, qword ptr [rsi + r11*8]
    mov qword ptr [rdi + rcx*8], rax
    inc r11
    inc rcx
    jmp merge_loop
merge_done:
    lea r8, [r8 + r15*2]
    jmp pass_loop
pass_done:
    xchg rsi, rdi
    add r15, r15
    jmp msort_loop
msort_done:

    # build output buffer
    lea rdi, [rip + outbuf]
    xor rbx, rbx
out_loop:
    cmp rbx, r13
    jae out_done
    mov rax, qword ptr [rsi + rbx*8]
    mov ecx, eax
    shr rax, 32
    lea rdx, [r14 + rcx]
copy_loop:
    test rax, rax
    jz copy_done
    mov cl, byte ptr [rdx]
    mov byte ptr [rdi], cl
    inc rdx
    inc rdi
    dec rax
    jmp copy_loop
copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 231
    xor edi, edi
    syscall
    hlt

# rax = entry A, rdx = entry B -> eax = 1 if A <= B else 0
cmpline:
    push rcx
    push rbx
    mov ebp, eax
    add rbp, r14
    mov r12d, edx
    add r12, r14
    shr rax, 32
    shr rdx, 32
    mov rcx, rax
    cmp rcx, rdx
    cmova rcx, rdx
cl_loop:
    test rcx, rcx
    jz cl_pref
    mov bl, byte ptr [rbp]
    cmp bl, byte ptr [r12]
    jne cl_diff
    inc rbp
    inc r12
    dec rcx
    jmp cl_loop
cl_diff:
    setb al
    movzx eax, al
    pop rbx
    pop rcx
    ret
cl_pref:
    cmp rax, rdx
    setbe al
    movzx eax, al
    pop rbx
    pop rcx
    ret

.bss
.balign 16
inbuf:
    .zero 32769
.balign 16
lines:
    .zero 8192
.balign 16
aux:
    .zero 8192
.balign 16
outbuf:
    .zero 33800
