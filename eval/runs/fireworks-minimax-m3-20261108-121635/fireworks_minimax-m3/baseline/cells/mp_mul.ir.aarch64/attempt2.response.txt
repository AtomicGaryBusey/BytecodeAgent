export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
    %n =l add %na, %nb
    %i =l copy 0
@zero_loop
    %cond =w cltl %i, %n
    jnz %cond, @zero_body, @zero_end
@zero_body
    %off =l shl %i, 3
    %addr =l add %r, %off
    storel 0, %addr
    %i =l add %i, 1
    jmp @zero_loop
@zero_end
    %i =l copy 0
@outer_loop
    %cond_i =w cltl %i, %na
    jnz %cond_i, @outer_body, @outer_end
@outer_body
    %off_i =l shl %i, 3
    %a_i_addr =l add %a, %off_i
    %a_i =l loadl %a_i_addr
    %j =l copy 0
@inner_loop
        %cond_j =w cltl %j, %nb
        jnz %cond_j, @inner_body, @inner_end
@inner_body
        %off_j =l shl %j, 3
        %b_j_addr =l add %b, %off_j
        %b_j =l loadl %b_j_addr

        %a_lo =l and %a_i, 0xFFFFFFFF
        %a_hi =l shr %a_i, 32
        %b_lo =l and %b_j, 0xFFFFFFFF
        %b_hi =l shr %b_j, 32

        %p0 =l mul %a_lo, %b_lo
        %p1 =l mul %a_lo, %b_hi
        %p2 =l mul %a_hi, %b_lo
        %p3 =l mul %a_hi, %b_hi

        %t1 =l and %p1, 0xFFFFFFFF
        %t2 =l and %p2, 0xFFFFFFFF
        %t3 =l shr %p0, 32
        %c =l add %t1, %t2
        %c =l add %c, %t3

        %p0_low =l and %p0, 0xFFFFFFFF
        %c_low =l and %c, 0xFFFFFFFF
        %c_low_shl =l shl %c_low, 32
        %lo =l or %p0_low, %c_low_shl

        %p1_hi =l shr %p1, 32
        %p2_hi =l shr %p2, 32
        %c_hi =l shr %c, 32
        %hi =l add %p1_hi, %p2_hi
        %hi =l add %hi, %c_hi
        %hi =l add %hi, %p3

        %k =l add %i, %j
        %off_k =l shl %k, 3
        %r_k_addr =l add %r, %off_k
        %old_k =l loadl %r_k_addr
        %sum_k =l add %old_k, %lo
        %carry1 =l cultl %sum_k, %old_k
        storel %sum_k, %r_k_addr

        %k1 =l add %k, 1
        %off_k1 =l shl %k1, 3
        %r_k1_addr =l add %r, %off_k1
        %old_k1 =l loadl %r_k1_addr
        %sum_k1 =l add %old_k1, %hi
        %carry2 =l cultl %sum_k1, %old_k1
        %sum_k2 =l add %sum_k1, %carry1
        %carry3 =l cultl %sum_k2, %sum_k1
        storel %sum_k2, %r_k1_addr

        %carry =l add %carry2, %carry3

        %m =l add %k, 2
@prop_loop
            %cond_m =w cltl %m, %n
            jnz %cond_m, @prop_body, @prop_end
@prop_body
            %off_m =l shl %m, 3
            %m_addr =l add %r, %off_m
            %old_m =l loadl %m_addr
            %new_m =l add %old_m, %carry
            %carry =l cultl %new_m, %old_m
            storel %new_m, %m_addr
            %m =l add %m, 1
            jmp @prop_loop
@prop_end
        %j =l add %j, 1
        jmp @inner_loop
@inner_end
    %i =l add %i, 1
    jmp @outer_loop
@outer_end
    %zero_count =l copy 0
    %idx =l sub %n, 1
@scan_loop
    %cond_idx =w cltl %idx, 0
    jnz %cond_idx, @scan_end
    %off_idx =l shl %idx, 3
    %addr_idx =l add %r, %off_idx
    %val =l loadl %addr_idx
    %is_zero =w ceql %val, 0
    jnz %is_zero, @scan_continue, @scan_found
@scan_continue
    %zero_count =l add %zero_count, 1
    %idx =l sub %idx, 1
    jmp @scan_loop
@scan_found
    %res =l sub %n, %zero_count
    ret %res
@scan_end
    ret 0
}
