export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%zero_l =l copy 0
	%one_l =l copy 1
	%mask =l shl 1, 32
	%mask =l sub %mask, 1

	%i =l copy 0
@zero_cond
	%cond =w csltl %i, %n
	jnz %cond, @zero_body, @zero_end
@zero_body
	%off =l shl %i, 3
	%addr =l add %r, %off
	storel 0, %addr
	%i =l add %i, 1
	jmp @zero_cond
@zero_end

	%i =l copy 0
@outer_cond
	%cond_i =w csltl %i, %na
	jnz %cond_i, @outer_body, @outer_end
@outer_body
	%off_i =l shl %i, 3
	%a_i_addr =l add %a, %off_i
	%a_i =l loadl %a_i_addr
	%j =l copy 0
@inner_cond
	%cond_j =w csltl %j, %nb
	jnz %cond_j, @inner_body, @inner_end
@inner_body
	%off_j =l shl %j, 3
	%b_j_addr =l add %b, %off_j
	%b_j =l loadl %b_j_addr

	%a_lo =l and %a_i, %mask
	%a_hi =l shr %a_i, 32
	%b_lo =l and %b_j, %mask
	%b_hi =l shr %b_j, 32

	%p0 =l mul %a_lo, %b_lo
	%p1 =l mul %a_lo, %b_hi
	%p2 =l mul %a_hi, %b_lo
	%p3 =l mul %a_hi, %b_hi

	%sum12 =l add %p1, %p2
	%c12_w =w cultl %sum12, %p1
	%c12 =l cmov %c12_w, %one_l, %zero_l

	%X_lo =l and %p0, %mask
	%X_hi =l shr %p0, 32
	%Y_lo =l and %sum12, %mask

	%mid =l add %X_hi, %Y_lo
	%mid_lo =l and %mid, %mask
	%mid_hi =l shr %mid, 32

	%lo_low_part =l shl %mid_lo, 32
	%lo =l or %X_lo, %lo_low_part

	%sum12_hi =l shr %sum12, 32
	%c12_at32 =l shl %c12, 32
	%hi =l add %p3, %sum12_hi
	%hi =l add %hi, %mid_hi
	%hi =l add %hi, %c12_at32

	%k =l add %i, %j
	%off_k =l shl %k, 3
	%r_k_addr =l add %r, %off_k
	%old_k =l loadl %r_k_addr
	%sum_k =l add %old_k, %lo
	%c1_w =w cultl %sum_k, %old_k
	%c1_l =l cmov %c1_w, %one_l, %zero_l
	storel %sum_k, %r_k_addr

	%k1 =l add %k, 1
	%off_k1 =l shl %k1, 3
	%r_k1_addr =l add %r, %off_k1
	%old_k1 =l loadl %r_k1_addr
	%sum_k1 =l add %old_k1, %hi
	%c2_w =w cultl %sum_k1, %old_k1
	%sum_k1b =l add %sum_k1, %c1_l
	%c3_w =w cultl %sum_k1b, %sum_k1
	%c_or_w =w or %c2_w, %c3_w
	%c_tot_l =l cmov %c_or_w, %one_l, %zero_l
	storel %sum_k1b, %r_k1_addr

	%carry_l =l copy %c_tot_l
	%m =l add %k, 2
@prop_cond
	%cond_m =w csltl %m, %n
	jnz %cond_m, @prop_check, @prop_end
@prop_check
	%has_carry =w cnel %carry_l, 0
	jnz %has_carry, @prop_body, @prop_end
@prop_body
	%off_m =l shl %m, 3
	%m_addr =l add %r, %off_m
	%old_m =l loadl %m_addr
	%sum_m =l add %old_m, %carry_l
	%c_m_w =w cultl %sum_m, %old_m
	%new_carry_l =l cmov %c_m_w, %one_l, %zero_l
	storel %sum_m, %m_addr
	%carry_l =l copy %new_carry_l
	%m =l add %m, 1
	jmp @prop_cond
@prop_end
	%j =l add %j, 1
	jmp @inner_cond
@inner_end
	%i =l add %i, 1
	jmp @outer_cond
@outer_end

	%zero_count =l copy 0
	%idx =l sub %n, 1
@scan_cond
	%cond_idx =w csltl %idx, 0
	jnz %cond_idx, @scan_all_zero, @scan_check
@scan_check
	%off_idx =l shl %idx, 3
	%addr_idx =l add %r, %off_idx
	%val =l loadl %addr_idx
	%is_zero =w ceql %val, 0
	jnz %is_zero, @scan_continue, @scan_found
@scan_continue
	%zero_count =l add %zero_count, 1
	%idx =l sub %idx, 1
	jmp @scan_cond
@scan_all_zero
	ret 0
@scan_found
	%res =l sub %n, %zero_count
	ret %res
}
