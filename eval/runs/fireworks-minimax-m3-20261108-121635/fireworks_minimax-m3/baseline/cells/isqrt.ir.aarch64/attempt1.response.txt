export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967295
	jmp @cond
@cond
	%c =w cultl %hi, %lo
	jnz %c, @ret, @body
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%c2 =w cultl %n, %sq
	jnz %c2, @else, @then
@then
	%lo =l add %mid, 1
	jmp @cond
@else
	%hi =l sub %mid, 1
	jmp @cond
@ret
	ret %hi
}
