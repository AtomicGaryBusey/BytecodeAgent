bubble_sort:
    push rbx
    push r12
    push r13
    push r14
    mov r12, rsi
    mov r13, rdi
    test r12, r12
    jz .Ldone
    xor r14, r14
.Louter_check:
    cmp r14, r12
    jge .Ldone
    mov rcx, r12
    sub rcx, r14
    sub rcx, 1
    test rcx, rcx
    jle .Lnext_outer
    xor rbx, rbx
.Linner:
    mov rax, qword ptr [r13 + rbx*8]
    mov rdx, qword ptr [r13 + rbx*8 + 8]
    cmp rax, rdx
    jle .Lskip_swap
    mov qword ptr [r13 + rbx*8], rdx
    mov qword ptr [r13 + rbx*8 + 8], rax
.Lskip_swap:
    inc rbx
    cmp rbx, rcx
    jl .Linner
.Lnext_outer:
    inc r14
    jmp .Louter_check
.Ldone:
    xor rax, rax
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
