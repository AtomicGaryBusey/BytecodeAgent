crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
    mov r8, rdi
    mov rcx, rsi
.Lloop:
    movzx edx, byte ptr [r8]
    xor eax, edx
    mov r9d, 8
.Linner:
    shr eax, 1
    sbb edx, edx
    and edx, 0xEDB88320
    xor eax, edx
    dec r9d
    jnz .Linner
    inc r8
    dec rcx
    jnz .Lloop
.Ldone:
    not eax
    ret
