export function l $packbits_decode(l %src_arg, l %n_arg, l %out_arg, l %cap_arg) {
@start
    %src =l copy %src_arg
    %n =l copy %n_arg
    %out =l copy %out_arg
    %cap =l copy %cap_arg
    %cap_rem =l copy %cap
    jmp @loop

@loop
    %nz =w csltl 0, %n
    jnz %nz, @body, @finish_ok

@body
    %hb =w loadsb %src
    %src1 =l add %src, 1
    %n1 =l sub %n, 1
    %isno =w ceqw %hb, -128
    jnz %isno, @skip_one, @not_noop

@skip_one
    %src =l copy %src1
    %n =l copy %n1
    jmp @loop

@not_noop
    %neg =w csltw %hb, 0
    jnz %neg, @repeating, @literalising

@literalising
    %hl =l extsw %hb
    %cnt =l add %hl, 1
    %srcbad =w csltl %n1, %cnt
    jnz %srcbad, @err_src, @lit_cap_ok

@lit_cap_ok
    %capbad =w csltl %cap_rem, %cnt
    jnz %capbad, @lit_fill, @lit_copy

@lit_copy
    %i =l copy 0
    jmp @lit_copy_test

@lit_copy_test
    %go =w csltl %i, %cnt
    jnz %go, @lit_copy_do, @lit_copy_done

@lit_copy_do
    %sp =l add %src1, %i
    %dp =l add %out, %i
    %bv =w loadub %sp
    storeb %bv, %dp
    %i =l add %i, 1
    jmp @lit_copy_test

@lit_copy_done
    %src =l add %src1, %cnt
    %n =l sub %n1, %cnt
    %out =l add %out, %cnt
    %cap_rem =l sub %cap_rem, %cnt
    jmp @loop

@lit_fill
    %i2 =l copy 0
    jmp @lit_fill_test

@lit_fill_test
    %go2 =w csltl %i2, %cap_rem
    jnz %go2, @lit_fill_do, @lit_fill_done

@lit_fill_do
    %sp2 =l add %src1, %i2
    %dp2 =l add %out, %i2
    %bv2 =w loadub %sp2
    storeb %bv2, %dp2
    %i2 =l add %i2, 1
    jmp @lit_fill_test

@lit_fill_done
    %result =l copy -1
    jmp @end

@repeating
    %srcbad2 =w csltl %n1, 1
    jnz %srcbad2, @err_src, @rep_cap_ok

@rep_cap_ok
    %hl2 =l extsw %hb
    %cnt2 =l sub 1, %hl2
    %capbad2 =w csltl %cap_rem, %cnt2
    jnz %capbad2, @rep_fill, @rep_write

@rep_write
    %v =w loadub %src1
    %j =l copy 0
    jmp @rep_write_test

@rep_write_test
    %go3 =w csltl %j, %cnt2
    jnz %go3, @rep_write_do, @rep_write_done

@rep_write_do
    %dp3 =l add %out, %j
    storeb %v, %dp3
    %j =l add %j, 1
    jmp @rep_write_test

@rep_write_done
    %src =l add %src1, 1
    %n =l sub %n1, 1
    %out =l add %out, %cnt2
    %cap_rem =l sub %cap_rem, %cnt2
    jmp @loop

@rep_fill
    %v2 =w loadub %src1
    %j2 =l copy 0
    jmp @rep_fill_test

@rep_fill_test
    %go4 =w csltl %j2, %cap_rem
    jnz %go4, @rep_fill_do, @rep_fill_done

@rep_fill_do
    %dp4 =l add %out, %j2
    storeb %v2, %dp4
    %j2 =l add %j2, 1
    jmp @rep_fill_test

@rep_fill_done
    %result =l copy -1
    jmp @end

@err_src
    %result =l copy -2
    jmp @end

@finish_ok
    %result =l sub %cap, %cap_rem
    jmp @end

@end
    ret %result
}
