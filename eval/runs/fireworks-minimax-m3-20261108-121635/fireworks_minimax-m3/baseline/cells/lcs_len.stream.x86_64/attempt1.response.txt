lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, r8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rdi, rbx
    mov rcx, r15
    inc rcx
    xor eax, eax
    rep stosq
    test r13, r13
    jz .Lreturn
.Louter:
    xor r9d, r9d
    mov ecx, 1
.Linner:
    cmp rcx, r15
    jg .Linner_done
    mov rdi, [rbx + rcx*8]
    mov rsi, [rbx + rcx*8 - 8]
    movzx eax, byte ptr [r12]
    movzx edx, byte ptr [r14 + rcx - 1]
    cmp eax, edx
    jne .Lnotequal
    lea rax, [r9 + 1]
    jmp .Lstore
.Lnotequal:
    mov rax, rdi
    cmp rax, rsi
    jge .Lstore
    mov rax, rsi
.Lstore:
    mov [rbx + rcx*8], rax
    mov r9, rdi
    inc rcx
    jmp .Linner
.Linner_done:
    inc r12
    dec r13
    jnz .Louter
.Lreturn:
    mov rax, [rbx + r15*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
