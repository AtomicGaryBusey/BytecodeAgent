memmove_i64:
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    jbe .Lforward
    mov rcx, rdx
    dec rcx
    shl rcx, 3
    add rsi, rcx
    add rdi, rcx
    mov rcx, rdx
.Lrev_loop:
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    sub rsi, 8
    sub rdi, 8
    dec rcx
    jnz .Lrev_loop
    jmp .Ldone
.Lforward:
    mov rcx, rdx
.Lfwd_loop:
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    add rsi, 8
    add rdi, 8
    dec rcx
    jnz .Lfwd_loop
.Ldone:
    xor eax, eax
    ret
