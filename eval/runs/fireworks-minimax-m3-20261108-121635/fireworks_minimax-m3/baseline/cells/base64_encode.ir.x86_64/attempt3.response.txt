export function l $base64_encode(l %src, l %n, l %out) {
@start
    %i =l copy 0
    %j =l copy 0
    jmp @loop_init

@loop_init
    %d =l sub %n, %i
    %ge =w csgel %d, 3
    jnz %ge, @loop_body, @remain

@remain
    %d2 =l sub %n, %i
    %z =w ceql %d2, 0
    jnz %z, @ret, @remain_1
@remain_1
    %one =w ceql %d2, 1
    jnz %one, @case1, @case2

@loop_body
    %p =l add %src, %i
    %b0 =l loadub %p
    %b1 =l loadub (add %p, 1)
    %b2 =l loadub (add %p, 2)
    %v =l shl %b0, 16
    %v =l or %v, shl %b1, 8
    %v =l or %v, %b2
    %c0v =l shr %v, 18
    %c1v =l and (shr %v, 12), 63
    %c2v =l and (shr %v, 6), 63
    %c3v =l and %v, 63
    %t =w csltl %c0v, 26
    jnz %t, @c0_cap, @c0_nx

@c0_nx
    %t =w csltl %c0v, 52
    jnz %t, @c0_low, @c0_nx2
@c0_nx2
    %t =w csltl %c0v, 62
    jnz %t, @c0_dig, @c0_nx3
@c0_nx3
    %t =w ceql %c0v, 62
    jnz %t, @c0_plus, @c0_sl

@c0_cap
    %c0 =l add 'A', %c0v
    jmp @c0_done
@c0_low
    %c0 =l add 'a', sub %c0v, 26
    jmp @c0_done
@c0_dig
    %c0 =l add '0', sub %c0v, 52
    jmp @c0_done
@c0_plus
    %c0 =l copy '+'
    jmp @c0_done
@c0_sl
    %c0 =l copy '/'
    jmp @c0_done

@c0_done
    %o0 =l add %out, %j
    storeb %c0, %o0
    %t =w csltl %c1v, 26
    jnz %t, @c1_cap, @c1_nx

@c1_nx
    %t =w csltl %c1v, 52
    jnz %t, @c1_low, @c1_nx2
@c1_nx2
    %t =w csltl %c1v, 62
    jnz %t, @c1_dig, @c1_nx3
@c1_nx3
    %t =w ceql %c1v, 62
    jnz %t, @c1_plus, @c1_sl

@c1_cap
    %c1 =l add 'A', %c1v
    jmp @c1_done
@c1_low
    %c1 =l add 'a', sub %c1v, 26
    jmp @c1_done
@c1_dig
    %c1 =l add '0', sub %c1v, 52
    jmp @c1_done
@c1_plus
    %c1 =l copy '+'
    jmp @c1_done
@c1_sl
    %c1 =l copy '/'
    jmp @c1_done

@c1_done
    %o1 =l add %o0, 1
    storeb %c1, %o1
    %t =w csltl %c2v, 26
    jnz %t, @c2_cap, @c2_nx

@c2_nx
    %t =w csltl %c2v, 52
    jnz %t, @c2_low, @c2_nx2
@c2_nx2
    %t =w csltl %c2v, 62
    jnz %t, @c2_dig, @c2_nx3
@c2_nx3
    %t =w ceql %c2v, 62
    jnz %t, @c2_plus, @c2_sl

@c2_cap
    %c2 =l add 'A', %c2v
    jmp @c2_done
@c2_low
    %c2 =l add 'a', sub %c2v, 26
    jmp @c2_done
@c2_dig
    %c2 =l add '0', sub %c2v, 52
    jmp @c2_done
@c2_plus
    %c2 =l copy '+'
    jmp @c2_done
@c2_sl
    %c2 =l copy '/'
    jmp @c2_done

@c2_done
    %o2 =l add %o0, 2
    storeb %c2, %o2
    %t =w csltl %c3v, 26
    jnz %t, @c3_cap, @c3_nx

@c3_nx
    %t =w csltl %c3v, 52
    jnz %t, @c3_low, @c3_nx2
@c3_nx2
    %t =w csltl %c3v, 62
    jnz %t, @c3_dig, @c3_nx3
@c3_nx3
    %t =w ceql %c3v, 62
    jnz %t, @c3_plus, @c3_sl

@c3_cap
    %c3 =l add 'A', %c3v
    jmp @c3_done
@c3_low
    %c3 =l add 'a', sub %c3v, 26
    jmp @c3_done
@c3_dig
    %c3 =l add '0', sub %c3v, 52
    jmp @c3_done
@c3_plus
    %c3 =l copy '+'
    jmp @c3_done
@c3_sl
    %c3 =l copy '/'
    jmp @c3_done

@c3_done
    %o3 =l add %o0, 3
    storeb %c3, %o3
    %i =l add %i, 3
    %j =l add %j, 4
    jmp @loop_init

@case1
    %p1 =l add %src, %i
    %cl =l loadub %p1
    %w1 =l shl %cl, 16
    %a0v =l shr %w1, 18
    %a1v =l and (shr %w1, 12), 63
    %t =w csltl %a0v, 26
    jnz %t, @a0_cap, @a0_nx

@a0_nx
    %t =w csltl %a0v, 52
    jnz %t, @a0_low, @a0_nx2
@a0_nx2
    %t =w csltl %a0v, 62
    jnz %t, @a0_dig, @a0_nx3
@a0_nx3
    %t =w ceql %a0v, 62
    jnz %t, @a0_plus, @a0_sl

@a0_cap
    %a0 =l add 'A', %a0v
    jmp @a0_done
@a0_low
    %a0 =l add 'a', sub %a0v, 26
    jmp @a0_done
@a0_dig
    %a0 =l add '0', sub %a0v, 52
    jmp @a0_done
@a0_plus
    %a0 =l copy '+'
    jmp @a0_done
@a0_sl
    %a0 =l copy '/'
    jmp @a0_done

@a0_done
    %pa0 =l add %out, %j
    storeb %a0, %pa0
    %t =w csltl %a1v, 26
    jnz %t, @a1_cap, @a1_nx

@a1_nx
    %t =w csltl %a1v, 52
    jnz %t, @a1_low, @a1_nx2
@a1_nx2
    %t =w csltl %a1v, 62
    jnz %t, @a1_dig, @a1_nx3
@a1_nx3
    %t =w ceql %a1v, 62
    jnz %t, @a1_plus, @a1_sl

@a1_cap
    %a1 =l add 'A', %a1v
    jmp @a1_done
@a1_low
    %a1 =l add 'a', sub %a1v, 26
    jmp @a1_done
@a1_dig
    %a1 =l add '0', sub %a1v, 52
    jmp @a1_done
@a1_plus
    %a1 =l copy '+'
    jmp @a1_done
@a1_sl
    %a1 =l copy '/'
    jmp @a1_done

@a1_done
    %pa1 =l add %pa0, 1
    storeb %a1, %pa1
    %pad =l copy '='
    %pa2 =l add %pa0, 2
    storeb %pad, %pa2
    %pa3 =l add %pa0, 3
    storeb %pad, %pa3
    %j =l add %j, 4
    jmp @ret

@case2
    %p2 =l add %src, %i
    %dl =l loadub %p2
    %dm =l loadub (add %p2, 1)
    %w2 =l or (shl %dl, 16), shl %dm, 8
    %k0v =l shr %w2, 18
    %k1v =l and (shr %w2, 12), 63
    %k2v =l and (shr %w2, 6), 63
    %t =w csltl %k0v, 26
    jnz %t, @k0_cap, @k0_nx

@k0_nx
    %t =w csltl %k0v, 52
    jnz %t, @k0_low, @k0_nx2
@k0_nx2
    %t =w csltl %k0v, 62
    jnz %t, @k0_dig, @k0_nx3
@k0_nx3
    %t =w ceql %k0v, 62
    jnz %t, @k0_plus, @k0_sl

@k0_cap
    %k0 =l add 'A', %k0v
    jmp @k0_done
@k0_low
    %k0 =l add 'a', sub %k0v, 26
    jmp @k0_done
@k0_dig
    %k0 =l add '0', sub %k0v, 52
    jmp @k0_done
@k0_plus
    %k0 =l copy '+'
    jmp @k0_done
@k0_sl
    %k0 =l copy '/'
    jmp @k0_done

@k0_done
    %pk0 =l add %out, %j
    storeb %k0, %pk0
    %t =w csltl %k1v, 26
    jnz %t, @k1_cap, @k1_nx

@k1_nx
    %t =w csltl %k1v, 52
    jnz %t, @k1_low, @k1_nx2
@k1_nx2
    %t =w csltl %k1v, 62
    jnz %t, @k1_dig, @k1_nx3
@k1_nx3
    %t =w ceql %k1v, 62
    jnz %t, @k1_plus, @k1_sl

@k1_cap
    %k1 =l add 'A', %k1v
    jmp @k1_done
@k1_low
    %k1 =l add 'a', sub %k1v, 26
    jmp @k1_done
@k1_dig
    %k1 =l add '0', sub %k1v, 52
    jmp @k1_done
@k1_plus
    %k1 =l copy '+'
    jmp @k1_done
@k1_sl
    %k1 =l copy '/'
    jmp @k1_done

@k1_done
    %pk1 =l add %pk0, 1
    storeb %k1, %pk1
    %t =w csltl %k2v, 26
    jnz %t, @k2_cap, @k2_nx

@k2_nx
    %t =w csltl %k2v, 52
    jnz %t, @k2_low, @k2_nx2
@k2_nx2
    %t =w csltl %k2v, 62
    jnz %t, @k2_dig, @k2_nx3
@k2_nx3
    %t =w ceql %k2v, 62
    jnz %t, @k2_plus, @k2_sl

@k2_cap
    %k2 =l add 'A', %k2v
    jmp @k2_done
@k2_low
    %k2 =l add 'a', sub %k2v, 26
    jmp @k2_done
@k2_dig
    %k2 =l add '0', sub %k2v, 52
    jmp @k2_done
@k2_plus
    %k2 =l copy '+'
    jmp @k2_done
@k2_sl
    %k2 =l copy '/'
    jmp @k2_done

@k2_done
    %pk2 =l add %pk0, 2
    storeb %k2, %pk2
    %pad2 =l copy '='
    %pk3 =l add %pk0, 3
    storeb %pad2, %pk3
    %j =l add %j, 4
    jmp @ret

@ret
    ret %j
}
