export function l $base64_encode(l %src, l %n, l %out) {
@start
    %init_i =l copy 0
    %init_j =l copy 0
    jmp @loop_init

@loop_init
    %i =l phi @start %init_i, @loop_body %next_i
    %j =l phi @start %init_j, @loop_body %next_j
    %d =l sub %n, %i
    %ge =w csgel %d, 3
    jnz %ge, @loop_body, @remain

@remain
    %d2 =l sub %n, %i
    %z =w ceql %d2, 0
    jnz %z, @ret, @remain_1
@remain_1
    %one =w ceql %d2, 1
    jnz %one, @case1, @case2

@loop_body
    %p =l add %src, %i
    %b0 =l loadub %p
    %ap1 =l add %p, 1
    %b1 =l loadub %ap1
    %ap2 =l add %p, 2
    %b2 =l loadub %ap2
    %sh16 =l shl %b0, 16
    %sh8 =l shl %b1, 8
    %vx1 =l or %sh16, %sh8
    %vp =l or %vx1, %b2
    %shr18 =l shr %vp, 18
    %c0v =l and %shr18, 63
    %shr12 =l shr %vp, 12
    %c1v =l and %shr12, 63
    %shr6 =l shr %vp, 6
    %c2v =l and %shr6, 63
    %c3v =l and %vp, 63
    %tsta =w csltl %c0v, 26
    jnz %tsta, @c0_cap, @c0_nx

@c0_nx
    %tstb =w csltl %c0v, 52
    jnz %tstb, @c0_low, @c0_nx2
@c0_nx2
    %tstc =w csltl %c0v, 62
    jnz %tstc, @c0_dig, @c0_nx3
@c0_nx3
    %tstd =w ceql %c0v, 62
    jnz %tstd, @c0_plus, @c0_sl

@c0_cap
    %c0 =l add 'A', %c0v
    jmp @c0_done
@c0_low
    %c0 =l add 'a', sub %c0v, 26
    jmp @c0_done
@c0_dig
    %c0 =l add '0', sub %c0v, 52
    jmp @c0_done
@c0_plus
    %c0 =l copy 43
    jmp @c0_done
@c0_sl
    %c0 =l copy 47
    jmp @c0_done

@c0_done
    %o0 =l add %out, %j
    storeb %c0, %o0
    %tste =w csltl %c1v, 26
    jnz %tste, @c1_cap, @c1_nx

@c1_nx
    %tstf =w csltl %c1v, 52
    jnz %tstf, @c1_low, @c1_nx2
@c1_nx2
    %tstg =w csltl %c1v, 62
    jnz %tstg, @c1_dig, @c1_nx3
@c1_nx3
    %tsth =w ceql %c1v, 62
    jnz %tsth, @c1_plus, @c1_sl

@c1_cap
    %c1 =l add 'A', %c1v
    jmp @c1_done
@c1_low
    %c1 =l add 'a', sub %c1v, 26
    jmp @c1_done
@c1_dig
    %c1 =l add '0', sub %c1v, 52
    jmp @c1_done
@c1_plus
    %c1 =l copy 43
    jmp @c1_done
@c1_sl
    %c1 =l copy 47
    jmp @c1_done

@c1_done
    %ap1b =l add %o0, 1
    storeb %c1, %ap1b
    %tsti =w csltl %c2v, 26
    jnz %tsti, @c2_cap, @c2_nx

@c2_nx
    %tstj =w csltl %c2v, 52
    jnz %tstj, @c2_low, @c2_nx2
@c2_nx2
    %tstk =w csltl %c2v, 62
    jnz %tstk, @c2_dig, @c2_nx3
@c2_nx3
    %tstl =w ceql %c2v, 62
    jnz %tstl, @c2_plus, @c2_sl

@c2_cap
    %c2 =l add 'A', %c2v
    jmp @c2_done
@c2_low
    %c2 =l add 'a', sub %c2v, 26
    jmp @c2_done
@c2_dig
    %c2 =l add '0', sub %c2v, 52
    jmp @c2_done
@c2_plus
    %c2 =l copy 43
    jmp @c2_done
@c2_sl
    %c2 =l copy 47
    jmp @c2_done

@c2_done
    %ap2b =l add %o0, 2
    storeb %c2, %ap2b
    %tstm =w csltl %c3v, 26
    jnz %tstm, @c3_cap, @c3_nx

@c3_nx
    %tstn =w csltl %c3v, 52
    jnz %tstn, @c3_low, @c3_nx2
@c3_nx2
    %tsto =w csltl %c3v, 62
    jnz %tsto, @c3_dig, @c3_nx3
@c3_nx3
    %tstp =w ceql %c3v, 62
    jnz %tstp, @c3_plus, @c3_sl

@c3_cap
    %c3 =l add 'A', %c3v
    jmp @c3_done
@c3_low
    %c3 =l add 'a', sub %c3v, 26
    jmp @c3_done
@c3_dig
    %c3 =l add '0', sub %c3v, 52
    jmp @c3_done
@c3_plus
    %c3 =l copy 43
    jmp @c3_done
@c3_sl
    %c3 =l copy 47
    jmp @c3_done

@c3_done
    %ap3b =l add %o0, 3
    storeb %c3, %ap3b
    %next_i =l add %i, 3
    %next_j =l add %j, 4
    jmp @loop_init

@case1
    %pc1 =l add %src, %i
    %cl =l loadub %pc1
    %cls16 =l shl %cl, 16
    %cls18 =l shr %cls16, 18
    %a0v =l and %cls18, 63
    %cls12 =l shr %cls16, 12
    %a1v =l and %cls12, 63
    %tstq =w csltl %a0v, 26
    jnz %tstq, @a0_cap, @a0_nx

@a0_nx
    %tstr =w csltl %a0v, 52
    jnz %tstr, @a0_low, @a0_nx2
@a0_nx2
    %tsts =w csltl %a0v, 62
    jnz %tsts, @a0_dig, @a0_nx3
@a0_nx3
    %tstt =w ceql %a0v, 62
    jnz %tstt, @a0_plus, @a0_sl

@a0_cap
    %a0 =l add 'A', %a0v
    jmp @a0_done
@a0_low
    %a0 =l add 'a', sub %a0v, 26
    jmp @a0_done
@a0_dig
    %a0 =l add '0', sub %a0v, 52
    jmp @a0_done
@a0_plus
    %a0 =l copy 43
    jmp @a0_done
@a0_sl
    %a0 =l copy 47
    jmp @a0_done

@a0_done
    %pa0 =l add %out, %j
    storeb %a0, %pa0
    %tstu =w csltl %a1v, 26
    jnz %tstu, @a1_cap, @a1_nx

@a1_nx
    %tstv =w csltl %a1v, 52
    jnz %tstv, @a1_low, @a1_nx2
@a1_nx2
    %tstw =w csltl %a1v, 62
    jnz %tstw, @a1_dig, @a1_nx3
@a1_nx3
    %tstx =w ceql %a1v, 62
    jnz %tstx, @a1_plus, @a1_sl

@a1_cap
    %a1 =l add 'A', %a1v
    jmp @a1_done
@a1_low
    %a1 =l add 'a', sub %a1v, 26
    jmp @a1_done
@a1_dig
    %a1 =l add '0', sub %a1v, 52
    jmp @a1_done
@a1_plus
    %a1 =l copy 43
    jmp @a1_done
@a1_sl
    %a1 =l copy 47
    jmp @a1_done

@a1_done
    %pa1b =l add %pa0, 1
    storeb %a1, %pa1b
    %pad =l copy 61
    %pa2 =l add %pa0, 2
    storeb %pad, %pa2
    %pa3 =l add %pa0, 3
    storeb %pad, %pa3
    %retc1 =l add %j, 4
    jmp @ret

@case2
    %pc2 =l add %src, %i
    %dl =l loadub %pc2
    %pc2b =l add %pc2, 1
    %dm =l loadub %pc2b
    %dls16 =l shl %dl, 16
    %dms8 =l shl %dm, 8
    %wv =l or %dls16, %dms8
    %wvs18 =l shr %wv, 18
    %k0v =l and %wvs18, 63
    %wvs12 =l shr %wv, 12
    %k1v =l and %wvs12, 63
    %wvs6 =l shr %wv, 6
    %k2v =l and %wvs6, 63
    %tsty =w csltl %k0v, 26
    jnz %tsty, @k0_cap, @k0_nx

@k0_nx
    %tstz =w csltl %k0v, 52
    jnz %tstz, @k0_low, @k0_nx2
@k0_nx2
    %tstaa =w csltl %k0v, 62
    jnz %tstaa, @k0_dig, @k0_nx3
@k0_nx3
    %tstab =w ceql %k0v, 62
    jnz %tstab, @k0_plus, @k0_sl

@k0_cap
    %k0 =l add 'A', %k0v
    jmp @k0_done
@k0_low
    %k0 =l add 'a', sub %k0v, 26
    jmp @k0_done
@k0_dig
    %k0 =l add '0', sub %k0v, 52
    jmp @k0_done
@k0_plus
    %k0 =l copy 43
    jmp @k0_done
@k0_sl
    %k0 =l copy 47
    jmp @k0_done

@k0_done
    %pk0 =l add %out, %j
    storeb %k0, %pk0
    %tstac =w csltl %k1v, 26
    jnz %tstac, @k1_cap, @k1_nx

@k1_nx
    %tstad =w csltl %k1v, 52
    jnz %tstad, @k1_low, @k1_nx2
@k1_nx2
    %tstae =w csltl %k1v, 62
    jnz %tstae, @k1_dig, @k1_nx3
@k1_nx3
    %tstaf =w ceql %k1v, 62
    jnz %tstaf, @k1_plus, @k1_sl

@k1_cap
    %k1 =l add 'A', %k1v
    jmp @k1_done
@k1_low
    %k1 =l add 'a', sub %k1v, 26
    jmp @k1_done
@k1_dig
    %k1 =l add '0', sub %k1v, 52
    jmp @k1_done
@k1_plus
    %k1 =l copy 43
    jmp @k1_done
@k1_sl
    %k1 =l copy 47
    jmp @k1_done

@k1_done
    %pk1b =l add %pk0, 1
    storeb %k1, %pk1b
    %tstag =w csltl %k2v, 26
    jnz %tstag, @k2_cap, @k2_nx

@k2_nx
    %tstah =w csltl %k2v, 52
    jnz %tstah, @k2_low, @k2_nx2
@k2_nx2
    %tstai =w csltl %k2v, 62
    jnz %tstai, @k2_dig, @k2_nx3
@k2_nx3
    %tstaj =w ceql %k2v, 62
    jnz %tstaj, @k2_plus, @k2_sl

@k2_cap
    %k2 =l add 'A', %k2v
    jmp @k2_done
@k2_low
    %k2 =l add 'a', sub %k2v, 26
    jmp @k2_done
@k2_dig
    %k2 =l add '0', sub %k2v, 52
    jmp @k2_done
@k2_plus
    %k2 =l copy 43
    jmp @k2_done
@k2_sl
    %k2 =l copy 47
    jmp @k2_done

@k2_done
    %pk2b =l add %pk0, 2
    storeb %k2, %pk2b
    %pad2 =l copy 61
    %pk3 =l add %pk0, 3
    storeb %pad2, %pk3
    %retc2 =l add %j, 4
    jmp @ret

@ret
    %result =l phi @remain %j, @a1_done %retc1, @k2_done %retc2
    ret %result
}
