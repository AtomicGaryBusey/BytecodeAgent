export function l $base64_encode(l %src, l %n, l %out) {
@start
    %i =l copy 0
    %j =l copy 0
    jmp @loop_cond

@loop_cond
    %rem =l sub %n, %i
    %ge =w csgel %rem, 3
    jnz %ge, @loop_body
    jmp @after_loop

@loop_body
    %off0 =l copy %i
    %p0 =l add %src, %off0
    %b0 =l loadub %p0
    %off1 =l add %i, 1
    %p1 =l add %src, %off1
    %b1 =l loadub %p1
    %off2 =l add %i, 2
    %p2 =l add %src, %off2
    %b2 =l loadub %p2

    %v0 =l shl %b0, 16
    %v1 =l shl %b1, 8
    %v2 =l or %v0, %v1
    %v =l or %v2, %b2

    %c0v =l shr %v, 18
    %c1v =l shr %v, 12
    %c1v =l and %c1v, 63
    %c2v =l shr %v, 6
    %c2v =l and %c2v, 63
    %c3v =l and %v, 63

    # map c0
    %c0 =l copy 0
    %c0cond =w csltl %c0v, 26
    jnz %c0cond, @c0_cap
    %c0 =l add 'a', sub %c0v, 26
    %c0cond2 =w csltl %c0v, 52
    jnz %c0cond2, @c0_done
    %c0 =l add '0', sub %c0v, 52
    %c0cond3 =w csltl %c0v, 62
    jnz %c0cond3, @c0_done
    %c0eq62 =w ceql %c0v, 62
    jnz %c0eq62, @c0_plus
    %c0 =l copy '/'
    jmp @c0_done
@c0_plus
    %c0 =l copy '+'
    jmp @c0_done
@c0_cap
    %c0 =l add 'A', %c0v
    jmp @c0_done
@c0_done
    %op0 =l add %out, %j
    storeb %c0, %op0

    # map c1
    %c1 =l copy 0
    %c1cond =w csltl %c1v, 26
    jnz %c1cond, @c1_cap
    %c1 =l add 'a', sub %c1v, 26
    %c1cond2 =w csltl %c1v, 52
    jnz %c1cond2, @c1_done
    %c1 =l add '0', sub %c1v, 52
    %c1cond3 =w csltl %c1v, 62
    jnz %c1cond3, @c1_done
    %c1eq62 =w ceql %c1v, 62
    jnz %c1eq62, @c1_plus
    %c1 =l copy '/'
    jmp @c1_done
@c1_plus
    %c1 =l copy '+'
    jmp @c1_done
@c1_cap
    %c1 =l add 'A', %c1v
    jmp @c1_done
@c1_done
    %op1 =l add %op0, 1
    storeb %c1, %op1

    # map c2
    %c2 =l copy 0
    %c2cond =w csltl %c2v, 26
    jnz %c2cond, @c2_cap
    %c2 =l add 'a', sub %c2v, 26
    %c2cond2 =w csltl %c2v, 52
    jnz %c2cond2, @c2_done
    %c2 =l add '0', sub %c2v, 52
    %c2cond3 =w csltl %c2v, 62
    jnz %c2cond3, @c2_done
    %c2eq62 =w ceql %c2v, 62
    jnz %c2eq62, @c2_plus
    %c2 =l copy '/'
    jmp @c2_done
@c2_plus
    %c2 =l copy '+'
    jmp @c2_done
@c2_cap
    %c2 =l add 'A', %c2v
    jmp @c2_done
@c2_done
    %op2 =l add %op0, 2
    storeb %c2, %op2

    # map c3
    %c3 =l copy 0
    %c3cond =w csltl %c3v, 26
    jnz %c3cond, @c3_cap
    %c3 =l add 'a', sub %c3v, 26
    %c3cond2 =w csltl %c3v, 52
    jnz %c3cond2, @c3_done
    %c3 =l add '0', sub %c3v, 52
    %c3cond3 =w csltl %c3v, 62
    jnz %c3cond3, @c3_done
    %c3eq62 =w ceql %c3v, 62
    jnz %c3eq62, @c3_plus
    %c3 =l copy '/'
    jmp @c3_done
@c3_plus
    %c3 =l copy '+'
    jmp @c3_done
@c3_cap
    %c3 =l add 'A', %c3v
    jmp @c3_done
@c3_done
    %op3 =l add %op0, 3
    storeb %c3, %op3

    %i =l add %i, 3
    %j =l add %j, 4
    jmp @loop_cond

@after_loop
    %rem =l sub %n, %i
    %rem0 =w ceql %rem, 0
    jnz %rem0, @ret
    %rem1 =w ceql %rem, 1
    jnz %rem1, @case1
    jmp @case2

@case1
    %p_case1 =l add %src, %i
    %b0_case1 =l loadub %p_case1
    %v_case1 =l shl %b0_case1, 16
    %c0v_case1 =l shr %v_case1, 18
    %c1v_case1 =l shr %v_case1, 12
    %c1v_case1 =l and %c1v_case1, 63

    # map c0 (case1)
    %c0_case1 =l copy 0
    %c0cond_case1 =w csltl %c0v_case1, 26
    jnz %c0cond_case1, @case1_c0_cap
    %c0_case1 =l add 'a', sub %c0v_case1, 26
    %c0cond2_case1 =w csltl %c0v_case1, 52
    jnz %c0cond2_case1, @case1_c0_done
    %c0_case1 =l add '0', sub %c0v_case1, 52
    %c0cond3_case1 =w csltl %c0v_case1, 62
    jnz %c0cond3_case1, @case1_c0_done
    %c0eq62_case1 =w ceql %c0v_case1, 62
    jnz %c0eq62_case1, @case1_c0_plus
    %c0_case1 =l copy '/'
    jmp @case1_c0_done
@case1_c0_plus
    %c0_case1 =l copy '+'
    jmp @case1_c0_done
@case1_c0_cap
    %c0_case1 =l add 'A', %c0v_case1
    jmp @case1_c0_done
@case1_c0_done
    %op_case1 =l add %out, %j
    storeb %c0_case1, %op_case1

    # map c1 (case1)
    %c1_case1 =l copy 0
    %c1cond_case1 =w csltl %c1v_case1, 26
    jnz %c1cond_case1, @case1_c1_cap
    %c1_case1 =l add 'a', sub %c1v_case1, 26
    %c1cond2_case1 =w csltl %c1v_case1, 52
    jnz %c1cond2_case1, @case1_c1_done
    %c1_case1 =l add '0', sub %c1v_case1, 52
    %c1cond3_case1 =w csltl %c1v_case1, 62
    jnz %c1cond3_case1, @case1_c1_done
    %c1eq62_case1 =w ceql %c1v_case1, 62
    jnz %c1eq62_case1, @case1_c1_plus
    %c1_case1 =l copy '/'
    jmp @case1_c1_done
@case1_c1_plus
    %c1_case1 =l copy '+'
    jmp @case1_c1_done
@case1_c1_cap
    %c1_case1 =l add 'A', %c1v_case1
    jmp @case1_c1_done
@case1_c1_done
    %op1_case1 =l add %op_case1, 1
    storeb %c1_case1, %op1_case1
    %pad_case1 =l copy '='
    %op2_case1 =l add %op_case1, 2
    storeb %pad_case1, %op2_case1
    %op3_case1 =l add %op_case1, 3
    storeb %pad_case1, %op3_case1
    %j =l add %j, 4
    jmp @ret

@case2
    %p0_case2 =l add %src, %i
    %b0_case2 =l loadub %p0_case2
    %off1_case2 =l add %i, 1
    %p1_case2 =l add %src, %off1_case2
    %b1_case2 =l loadub %p1_case2
    %v0_case2 =l shl %b0_case2, 16
    %v1_case2 =l shl %b1_case2, 8
    %v_case2 =l or %v0_case2, %v1_case2
    %c0v_case2 =l shr %v_case2, 18
    %c1v_case2 =l shr %v_case2, 12
    %c1v_case2 =l and %c1v_case2, 63
    %c2v_case2 =l shr %v_case2, 6
    %c2v_case2 =l and %c2v_case2, 63

    %op_case2 =l add %out, %j

    # map c0 (case2)
    %c0_case2 =l copy 0
    %c0cond_case2 =w csltl %c0v_case2, 26
    jnz %c0cond_case2, @case2_c0_cap
    %c0_case2 =l add 'a', sub %c0v_case2, 26
    %c0cond2_case2 =w csltl %c0v_case2, 52
    jnz %c0cond2_case2, @case2_c0_done
    %c0_case2 =l add '0', sub %c0v_case2, 52
    %c0cond3_case2 =w csltl %c0v_case2, 62
    jnz %c0cond3_case2, @case2_c0_done
    %c0eq62_case2 =w ceql %c0v_case2, 62
    jnz %c0eq62_case2, @case2_c0_plus
    %c0_case2 =l copy '/'
    jmp @case2_c0_done
@case2_c0_plus
    %c0_case2 =l copy '+'
    jmp @case2_c0_done
@case2_c0_cap
    %c0_case2 =l add 'A', %c0v_case2
    jmp @case2_c0_done
@case2_c0_done
    storeb %c0_case2, %op_case2

    # map c1 (case2)
    %c1_case2 =l copy 0
    %c1cond_case2 =w csltl %c1v_case2, 26
    jnz %c1cond_case2, @case2_c1_cap
    %c1_case2 =l add 'a', sub %c1v_case2, 26
    %c1cond2_case2 =w csltl %c1v_case2, 52
    jnz %c1cond2_case2, @case2_c1_done
    %c1_case2 =l add '0', sub %c1v_case2, 52
    %c1cond3_case2 =w csltl %c1v_case2, 62
    jnz %c1cond3_case2, @case2_c1_done
    %c1eq62_case2 =w ceql %c1v_case2, 62
    jnz %c1eq62_case2, @case2_c1_plus
    %c1_case2 =l copy '/'
    jmp @case2_c1_done
@case2_c1_plus
    %c1_case2 =l copy '+'
    jmp @case2_c1_done
@case2_c1_cap
    %c1_case2 =l add 'A', %c1v_case2
    jmp @case2_c1_done
@case2_c1_done
    %op1_case2 =l add %op_case2, 1
    storeb %c1_case2, %op1_case2

    # map c2 (case2)
    %c2_case2 =l copy 0
    %c2cond_case2 =w csltl %c2v_case2, 26
    jnz %c2cond_case2, @case2_c2_cap
    %c2_case2 =l add 'a', sub %c2v_case2, 26
    %c2cond2_case2 =w csltl %c2v_case2, 52
    jnz %c2cond2_case2, @case2_c2_done
    %c2_case2 =l add '0', sub %c2v_case2, 52
    %c2cond3_case2 =w csltl %c2v_case2, 62
    jnz %c2cond3_case2, @case2_c2_done
    %c2eq62_case2 =w ceql %c2v_case2, 62
    jnz %c2eq62_case2, @case2_c2_plus
    %c2_case2 =l copy '/'
    jmp @case2_c2_done
@case2_c2_plus
    %c2_case2 =l copy '+'
    jmp @case2_c2_done
@case2_c2_cap
    %c2_case2 =l add 'A', %c2v_case2
    jmp @case2_c2_done
@case2_c2_done
    %op2_case2 =l add %op_case2, 2
    storeb %c2_case2, %op2_case2
    %pad_case2 =l copy '='
    %op3_case2 =l add %op_case2, 3
    storeb %pad_case2, %op3_case2
    %j =l add %j, 4
    jmp @ret

@ret
    ret %j
}
