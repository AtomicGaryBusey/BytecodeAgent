export function l $bmax(l %a, l %n) {
@start
	%cur = loadl %a
	%i = copy 1
@loop
	%cond = csltl %i, %n
	jnz %cond, @body, @end
@body
	%off = mul %i, 8
	%ptr = add %a, %off
	%val = loadl %ptr
	%isless = csltl %cur, %val
	jnz %isless, @update, @skip
@update
	%cur = copy %val
@skip
	%i = add %i, 1
	jmp @loop
@end
	ret %cur
}
