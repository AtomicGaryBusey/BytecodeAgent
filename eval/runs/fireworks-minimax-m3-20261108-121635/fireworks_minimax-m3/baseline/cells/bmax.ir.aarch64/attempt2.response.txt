export function l $bmax(l %a, l %n) {
@start
	%cur =l loadl %a
	%i =l copy 1
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @ret
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%isless =w csltl %cur, %val
	jnz %isless, @update, @cont
@update
	%cur =l copy %val
	jmp @cont
@cont
	%i =l add %i, 1
	jmp @loop
@ret
	ret %cur
}
