export function $memmove_i64(l %dst, l %src, l %n) -> l {
@start
  %lt = csltl %dst, %src
  jnz %lt, @copy_fwd, @copy_bwd

@copy_fwd
  jmp @loop_fwd(0)

@loop_fwd(l %i)
  %cond = csltl %i, %n
  jnz %cond, @body_fwd, @ret

@body_fwd
  %off = shl %i, 3
  %sp = add %src, %off
  %dp = add %dst, %off
  %val = loadl %sp
  storel %val, %dp
  %inext = add %i, 1
  jmp @loop_fwd(%inext)

@copy_bwd
  jmp @loop_bwd(%n)

@loop_bwd(l %j)
  %cond2 = csltl 0, %j
  jnz %cond2, @body_bwd, @ret

@body_bwd
  %jnext = sub %j, 1
  %off2 = shl %jnext, 3
  %sp2 = add %src, %off2
  %dp2 = add %dst, %off2
  %val2 = loadl %sp2
  storel %val2, %dp2
  jmp @loop_bwd(%jnext)

@ret
  ret 0
}
