export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum = addl %a, %b
	%sa = shrl %a, 63
	%sb = shrl %b, 63
	%ss = shrl %sum, 63
	%na = xor %sa, 1
	%nb = xor %sb, 1
	%ns = xor %ss, 1
	%neg = and %sa, %sb
	%neg = and %neg, %ns
	%pos = and %na, %nb
	%pos = and %pos, %ss
	%ret = or %neg, %pos
	storel %sum, %out
	ret %ret
}
