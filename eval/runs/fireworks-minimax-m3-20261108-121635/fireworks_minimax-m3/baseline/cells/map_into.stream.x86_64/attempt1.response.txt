map_into:
    sub rsp, 8
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rdx
    mov r14, rcx
    mov r15, rsi
    xor rbx, rbx
    xor rbp, rbp
.Lloop:
    cmp rbp, r13
    je .Ldone
    mov rdi, qword ptr [r15 + rbp*8]
    call r14
    mov qword ptr [r12 + rbp*8], rax
    cmp rax, qword ptr [r15 + rbp*8]
    jge .Lskip
    inc rbx
.Lskip:
    inc rbp
    jmp .Lloop
.Ldone:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    add rsp, 8
    ret
