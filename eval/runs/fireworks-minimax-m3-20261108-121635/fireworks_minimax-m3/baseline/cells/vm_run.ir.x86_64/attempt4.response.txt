export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc0 =l const 0
	%d0 =l const 0
	%cnt0 =l const 0
	%max =l const 5000
	jmp @check

@check
	%pc =l phi @start %pc0, @i_adv %pc_adv, @i_push %pc_push, @i_add %pc_add, @i_sub %pc_sub, @i_mul %pc_mul, @i_dup %pc_dup, @i_swap %pc_swap, @i_drop %pc_drop, @i_over %pc_over, @i_lt %pc_lt, @i_jz_t %pc_jzt, @i_jz_n %pc_jzn, @i_jmp %pc_jmp
	%d =l phi @start %d0, @i_adv %d_adv, @i_push %d_push, @i_add %d_add, @i_sub %d_sub, @i_mul %d_mul, @i_dup %d_dup, @i_swap %d_swap, @i_drop %d_drop, @i_over %d_over, @i_lt %d_lt, @i_jz_t %d_jzt, @i_jz_n %d_jzn, @i_jmp %d_jmp
	%cnt =l phi @start %cnt0, @i_adv %cnt_adv, @i_push %cnt_push, @i_add %cnt_add, @i_sub %cnt_sub, @i_mul %cnt_mul, @i_dup %cnt_dup, @i_swap %cnt_swap, @i_drop %cnt_drop, @i_over %cnt_over, @i_lt %cnt_lt, @i_jz_t %cnt_jzt, @i_jz_n %cnt_jzn, @i_jmp %cnt_jmp
	%c1 =w csltl %pc, %len
	jnz %c1, @check2, @zero

@check2
	%c2 =w csltl %cnt, %max
	jnz %c2, @dispatch, @zero

@dispatch
	%new_cnt =l add %cnt, 1
	%off =l mul %pc, 8
	%ptr =l add %code, %off
	%op =l loadl %ptr
	%w0 =w ceql %op, 0
	jnz %w0, @i_halt
	%w1 =w ceql %op, 1
	jnz %w1, @i_push
	%w2 =w ceql %op, 2
	jnz %w2, @i_add
	%w3 =w ceql %op, 3
	jnz %w3, @i_sub
	%w4 =w ceql %op, 4
	jnz %w4, @i_mul
	%w5 =w ceql %op, 5
	jnz %w5, @i_dup
	%w6 =w ceql %op, 6
	jnz %w6, @i_swap
	%w7 =w ceql %op, 7
	jnz %w7, @i_drop
	%w8 =w ceql %op, 8
	jnz %w8, @i_over
	%w9 =w ceql %op, 9
	jnz %w9, @i_lt
	%w10 =w ceql %op, 10
	jnz %w10, @i_jz
	%w11 =w ceql %op, 11
	jnz %w11, @i_jmp
	jmp @i_adv

@i_halt
	%d_halt =l copy %d
	jmp @zero

@i_push
	%tpc =l add %pc, 1
	%toff =l mul %tpc, 8
	%tptr =l add %code, %toff
	%v =l loadl %tptr
	%doff =l mul %d, 8
	%dptr =l add %stack, %doff
	storel %v, %dptr
	%pc_push =l add %pc, 2
	%d_push =l add %d, 1
	%cnt_push =l copy %new_cnt
	jmp @check

@i_add
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%b =l loadl %ptr1
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%a =l loadl %ptr2
	%res =l add %a, %b
	storel %res, %ptr2
	%pc_add =l add %pc, 1
	%d_add =l sub %d, 1
	%cnt_add =l copy %new_cnt
	jmp @check

@i_sub
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%b =l loadl %ptr1
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%a =l loadl %ptr2
	%res =l sub %a, %b
	storel %res, %ptr2
	%pc_sub =l add %pc, 1
	%d_sub =l sub %d, 1
	%cnt_sub =l copy %new_cnt
	jmp @check

@i_mul
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%b =l loadl %ptr1
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%a =l loadl %ptr2
	%res =l mul %a, %b
	storel %res, %ptr2
	%pc_mul =l add %pc, 1
	%d_mul =l sub %d, 1
	%cnt_mul =l copy %new_cnt
	jmp @check

@i_dup
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%v =l loadl %ptr1
	%doff =l mul %d, 8
	%dptr =l add %stack, %doff
	storel %v, %dptr
	%pc_dup =l add %pc, 1
	%d_dup =l add %d, 1
	%cnt_dup =l copy %new_cnt
	jmp @check

@i_swap
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%a =l loadl %ptr1
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%b =l loadl %ptr2
	storel %b, %ptr1
	storel %a, %ptr2
	%pc_swap =l add %pc, 1
	%d_swap =l copy %d
	%cnt_swap =l copy %new_cnt
	jmp @check

@i_drop
	%pc_drop =l add %pc, 1
	%d_drop =l sub %d, 1
	%cnt_drop =l copy %new_cnt
	jmp @check

@i_over
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%v =l loadl %ptr2
	%doff =l mul %d, 8
	%dptr =l add %stack, %doff
	storel %v, %dptr
	%pc_over =l add %pc, 1
	%d_over =l add %d, 1
	%cnt_over =l copy %new_cnt
	jmp @check

@i_lt
	%d1 =l sub %d, 1
	%off1 =l mul %d1, 8
	%ptr1 =l add %stack, %off1
	%b =l loadl %ptr1
	%d2 =l sub %d, 2
	%off2 =l mul %d2, 8
	%ptr2 =l add %stack, %off2
	%a =l loadl %ptr2
	%c =w csltl %a, %b
	storel %c, %ptr2
	%pc_lt =l add %pc, 1
	%d_lt =l sub %d, 1
	%cnt_lt =l copy %new_cnt
	jmp @check

@i_jz
	%tpc =l add %pc, 1
	%toff =l mul %tpc, 8
	%tptr =l add %code, %toff
	%t =l loadl %tptr
	%d1 =l sub %d, 1
	%voff =l mul %d1, 8
	%vaddr =l add %stack, %voff
	%v =l loadl %vaddr
	%is0 =w ceql %v, 0
	jnz %is0, @i_jz_t, @i_jz_n

@i_jz_t
	%pc_jzt =l copy %t
	%d_jzt =l sub %d, 1
	%cnt_jzt =l copy %new_cnt
	jmp @check

@i_jz_n
	%pc_jzn =l add %pc, 2
	%d_jzn =l sub %d, 1
	%cnt_jzn =l copy %new_cnt
	jmp @check

@i_jmp
	%tpc =l add %pc, 1
	%toff =l mul %tpc, 8
	%tptr =l add %code, %toff
	%t =l loadl %tptr
	%pc_jmp =l copy %t
	%d_jmp =l copy %d
	%cnt_jmp =l copy %new_cnt
	jmp @check

@i_adv
	%pc_adv =l add %pc, 1
	%d_adv =l copy %d
	%cnt_adv =l copy %new_cnt
	jmp @check

@zero
	%d_final =l phi @check %d, @i_halt %d_halt
	%i_init =l copy %d_final
	jmp @zloop

@zloop
	%i =l phi @zero %i_init, @zbody %i_next
	%c =w csltl %i, %cap
	jnz %c, @zbody, @ret_prep

@zbody
	%off =l mul %i, 8
	%addr =l add %stack, %off
	storel 0, %addr
	%i_next =l add %i, 1
	jmp @zloop

@ret_prep
	%c_ret =w csltl 0, %d_final
	jnz %c_ret, @ret_val, @ret_zero

@ret_zero
	ret 0

@ret_val
	%idx =l sub %d_final, 1
	%off =l mul %idx, 8
	%addr =l add %stack, %off
	%val =l loadl %addr
	ret %val
}
