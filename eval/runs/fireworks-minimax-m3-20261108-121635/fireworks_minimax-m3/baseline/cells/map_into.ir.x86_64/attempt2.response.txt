export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i0 =l copy 0
	%cnt0 =l copy 0
	jmp @cond

@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%dptr =l add %dst, %off
	%val =l loadl %sptr
	%fp =l copy %f
	%res =l call %fp(l %val)
	storel %res, %dptr
	%lt =w csltl %res, %val
	%inc =l extsw %lt
	%cnt1 =l add %cnt, %inc
	%i1 =l add %i, 1
	jmp @cond

@cond
	%i =l phi @start %i0, @body %i1
	%cnt =l phi @start %cnt0, @body %cnt1
	%c =w csltl %i, %n
	jnz %c, @body, @end

@end
	ret %cnt
}
