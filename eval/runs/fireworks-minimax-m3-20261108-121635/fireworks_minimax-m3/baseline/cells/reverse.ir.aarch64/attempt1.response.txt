export function l $reverse(l %a, l %n) {
@start
    %nm1 =l sub %n, 1
    %io =l copy 0
    %jo =l copy %nm1
    %c0 =w csltl %io, %jo
    jnz %c0, @body, @end

@body
    %i =l phi @start %io, @body %inew
    %j =l phi @start %jo, @body %jnew
    %offi =l mul %i, 8
    %offj =l mul %j, 8
    %pi =l add %a, %offi
    %pj =l add %a, %offj
    %vi =l loadl %pi
    %vj =l loadl %pj
    storel %vj, %pi
    storel %vi, %pj
    %inew =l add %i, 1
    %jnew =l sub %j, 1
    %c1 =w csltl %inew, %jnew
    jnz %c1, @body, @end

@end
    ret 0
}
