export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%nm1 =l add %n, 1
	%i0 =l copy 0
	jmp @zl_entry
@zl_entry
	%zi =l phi @start %i0, @zl_body %zinext
	%zc =w csltl %zi, %nm1
	jnz %zc, @zl_body, @zl_done
@zl_body
	%zio =l mul %zi, 8
	%zia =l add %row, %zio
	%zh =l copy 0
	storel %zh, %zia
	%zinext =l add %zi, 1
	jmp @zl_entry
@zl_done
	%istart =l copy 0
	jmp @ol_entry
@ol_entry
	%oi =l phi @zl_done %istart, @il_done %oinext
	%oc =w csltl %oi, %m
	jnz %oc, @ol_body, @ol_done
@ol_body
	%xia =l add %x, %oi
	%xi =w loadub %xia
	%dstart =l copy 0
	%jstart =l copy 1
	jmp @il_entry
@il_entry
	%j =l phi @ol_body %jstart, @il_step %jnext
	%diag =l phi @ol_body %dstart, @il_step %diagnext
	%ic =w csltl %j, %n
	jnz %ic, @il_body, @il_done
@il_body
	%jm1 =l sub %j, 1
	%yja =l add %y, %jm1
	%yj =w loadub %yja
	%rjo =l mul %j, 8
	%rja =l add %row, %rjo
	%up =l loadl %rja
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%nv =l add %diag, 1
	storel %nv, %rja
	jmp @il_step
@nomatch
	%rjmo =l mul %jm1, 8
	%rjma =l add %row, %rjmo
	%left =l loadl %rjma
	%gt =w csltl %left, %up
	jnz %gt, @maxu, @maxl
@maxu
	storel %up, %rja
	jmp @il_step
@maxl
	storel %left, %rja
	jmp @il_step
@il_step
	%diagnext =l copy %up
	%jnext =l add %j, 1
	jmp @il_entry
@il_done
	%oinext =l add %oi, 1
	jmp @ol_entry
@ol_done
	%fo =l mul %n, 8
	%fa =l add %row, %fo
	%res =l loadl %fa
	ret %res
}
