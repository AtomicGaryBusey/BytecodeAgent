export function l $vm_run(l %code, l %len, l %stack, l %cap) {

@start
	%pc = addl 0, 0
	%depth = addl 0, 0
	jmp @loop

@loop
	%ok = csltl %pc, %len
	jnz %ok, @exec, @halt

@exec
	%opaddr = addl %code, mull %pc, 8
	%op = loadl %opaddr
	switch %op, @op_default, [0 @halt, 1 @op_push, 2 @op_add, 3 @op_sub, 4 @op_mul, 5 @op_dup, 6 @op_swap, 7 @op_drop, 8 @op_over, 9 @op_lt, 10 @op_jz, 11 @op_jmp]

@op_push
	%popc = addl %pc, 1
	%paddr = addl %code, mull %popc, 8
	%pval = loadl %paddr
	%saddr = addl %stack, mull %depth, 8
	storel %pval, %saddr
	%depth = addl %depth, 1
	%pc = addl %pc, 2
	jmp @loop

@op_add
	%depth = subl %depth, 1
	%baddr = addl %stack, mull %depth, 8
	%b = loadl %baddr
	%depth = subl %depth, 1
	%aaddr = addl %stack, mull %depth, 8
	%a = loadl %aaddr
	%r = addl %a, %b
	storel %r, %aaddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_sub
	%depth = subl %depth, 1
	%baddr = addl %stack, mull %depth, 8
	%b = loadl %baddr
	%depth = subl %depth, 1
	%aaddr = addl %stack, mull %depth, 8
	%a = loadl %aaddr
	%r = subl %a, %b
	storel %r, %aaddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_mul
	%depth = subl %depth, 1
	%baddr = addl %stack, mull %depth, 8
	%b = loadl %baddr
	%depth = subl %depth, 1
	%aaddr = addl %stack, mull %depth, 8
	%a = loadl %aaddr
	%r = mull %a, %b
	storel %r, %aaddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_dup
	%ti = subl %depth, 1
	%taddr = addl %stack, mull %ti, 8
	%tval = loadl %taddr
	%saddr = addl %stack, mull %depth, 8
	storel %tval, %saddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_swap
	%i1 = subl %depth, 1
	%i2 = subl %depth, 2
	%addr1 = addl %stack, mull %i1, 8
	%addr2 = addl %stack, mull %i2, 8
	%v1 = loadl %addr1
	%v2 = loadl %addr2
	storel %v1, %addr2
	storel %v2, %addr1
	%pc = addl %pc, 1
	jmp @loop

@op_drop
	%depth = subl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_over
	%ti = subl %depth, 2
	%taddr = addl %stack, mull %ti, 8
	%tval = loadl %taddr
	%saddr = addl %stack, mull %depth, 8
	storel %tval, %saddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_lt
	%depth = subl %depth, 1
	%baddr = addl %stack, mull %depth, 8
	%b = loadl %baddr
	%depth = subl %depth, 1
	%aaddr = addl %stack, mull %depth, 8
	%a = loadl %aaddr
	%cmpw = csltl %a, %b
	%cmpl = extsw %cmpw
	storel %cmpl, %aaddr
	%depth = addl %depth, 1
	%pc = addl %pc, 1
	jmp @loop

@op_jz
	%popc = addl %pc, 1
	%paddr = addl %code, mull %popc, 8
	%t = loadl %paddr
	%depth = subl %depth, 1
	%vaddr = addl %stack, mull %depth, 8
	%v = loadl %vaddr
	%is0 = ceql %v, 0
	jnz %is0, @jz_take, @jz_skip

@jz_take
	%pc = addl %t, 0
	jmp @loop

@jz_skip
	%pc = addl %pc, 2
	jmp @loop

@op_jmp
	%popc = addl %pc, 1
	%paddr = addl %code, mull %popc, 8
	%pc = loadl %paddr
	jmp @loop

@op_default
	%pc = addl %pc, 1
	jmp @loop

@halt
	%i = addl %depth, 0
	jmp @zero_loop

@zero_loop
	%done = ceql %i, %cap
	jnz %done, @result, @zero_body

@zero_body
	%zaddr = addl %stack, mull %i, 8
	storel 0, %zaddr
	%i = addl %i, 1
	jmp @zero_loop

@result
	%isempty = ceql %depth, 0
	jnz %isempty, @ret_zero, @ret_top

@ret_top
	%ti = subl %depth, 1
	%taddr = addl %stack, mull %ti, 8
	%retval = loadl %taddr
	ret %retval

@ret_zero
	ret 0
}
