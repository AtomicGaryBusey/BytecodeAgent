export function $vm_run(l %code, l %len, l %stack, l %cap) -> l {

@start
	%pc = add 0, 0
	%depth = add 0, 0
	jmp @loop

@loop
	%ok = csltl %pc, %len
	jnz %ok, @exec, @halt

@exec
	%opaddr = add %code, mul %pc, 8
	%op = loadl %opaddr
	switch %op, @op_default, [0 @halt, 1 @op_push, 2 @op_add, 3 @op_sub, 4 @op_mul, 5 @op_dup, 6 @op_swap, 7 @op_drop, 8 @op_over, 9 @op_lt, 10 @op_jz, 11 @op_jmp]

@op_push
	%popc = add %pc, 1
	%paddr = add %code, mul %popc, 8
	%pval = loadl %paddr
	%saddr = add %stack, mul %depth, 8
	storel %pval, %saddr
	%depth = add %depth, 1
	%pc = add %pc, 2
	jmp @loop

@op_add
	%depth = sub %depth, 1
	%baddr = add %stack, mul %depth, 8
	%b = loadl %baddr
	%depth = sub %depth, 1
	%aaddr = add %stack, mul %depth, 8
	%a = loadl %aaddr
	%r = add %a, %b
	storel %r, %aaddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_sub
	%depth = sub %depth, 1
	%baddr = add %stack, mul %depth, 8
	%b = loadl %baddr
	%depth = sub %depth, 1
	%aaddr = add %stack, mul %depth, 8
	%a = loadl %aaddr
	%r = sub %a, %b
	storel %r, %aaddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_mul
	%depth = sub %depth, 1
	%baddr = add %stack, mul %depth, 8
	%b = loadl %baddr
	%depth = sub %depth, 1
	%aaddr = add %stack, mul %depth, 8
	%a = loadl %aaddr
	%r = mul %a, %b
	storel %r, %aaddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_dup
	%ti = sub %depth, 1
	%taddr = add %stack, mul %ti, 8
	%tval = loadl %taddr
	%saddr = add %stack, mul %depth, 8
	storel %tval, %saddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_swap
	%i1 = sub %depth, 1
	%i2 = sub %depth, 2
	%addr1 = add %stack, mul %i1, 8
	%addr2 = add %stack, mul %i2, 8
	%v1 = loadl %addr1
	%v2 = loadl %addr2
	storel %v1, %addr2
	storel %v2, %addr1
	%pc = add %pc, 1
	jmp @loop

@op_drop
	%depth = sub %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_over
	%ti = sub %depth, 2
	%taddr = add %stack, mul %ti, 8
	%tval = loadl %taddr
	%saddr = add %stack, mul %depth, 8
	storel %tval, %saddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_lt
	%depth = sub %depth, 1
	%baddr = add %stack, mul %depth, 8
	%b = loadl %baddr
	%depth = sub %depth, 1
	%aaddr = add %stack, mul %depth, 8
	%a = loadl %aaddr
	%cmpw = csltl %a, %b
	%cmpl = extsw %cmpw
	storel %cmpl, %aaddr
	%depth = add %depth, 1
	%pc = add %pc, 1
	jmp @loop

@op_jz
	%popc = add %pc, 1
	%paddr = add %code, mul %popc, 8
	%t = loadl %paddr
	%depth = sub %depth, 1
	%vaddr = add %stack, mul %depth, 8
	%v = loadl %vaddr
	%is0 = ceql %v, 0
	jnz %is0, @jz_take, @jz_skip

@jz_take
	%pc = add %t, 0
	jmp @loop

@jz_skip
	%pc = add %pc, 2
	jmp @loop

@op_jmp
	%popc = add %pc, 1
	%paddr = add %code, mul %popc, 8
	%pc = loadl %paddr
	jmp @loop

@op_default
	%pc = add %pc, 1
	jmp @loop

@halt
	%i = add %depth, 0
	jmp @zero_loop

@zero_loop
	%done = ceql %i, %cap
	jnz %done, @result, @zero_body

@zero_body
	%zaddr = add %stack, mul %i, 8
	storel 0, %zaddr
	%i = add %i, 1
	jmp @zero_loop

@result
	%isempty = ceql %depth, 0
	jnz %isempty, @ret_zero, @ret_top

@ret_top
	%ti = sub %depth, 1
	%taddr = add %stack, mul %ti, 8
	%retval = loadl %taddr
	ret %retval

@ret_zero
	ret 0
}
