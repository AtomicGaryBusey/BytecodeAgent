export function l $max_byte(l %p, l %n) {
	%zero =l copy 0
	%n_is_zero =w ceql %n, %zero
	jnz %n_is_zero, @ret_zero, @init

@init
	%max =l loadub %p
	%i =l copy 1
	jmp @check

@check
	%cont =w cultl %i, %n
	jnz %cont, @body, @ret_max

@body
	%addr =l add %p, %i
	%byte =l loadub %addr
	%is_greater =w cugtl %byte, %max
	jnz %is_greater, @update, @inc

@update
	%max =l copy %byte
	jmp @inc

@inc
	%i =l add %i, 1
	jmp @check

@ret_max
	ret %max

@ret_zero
	ret %zero
}
