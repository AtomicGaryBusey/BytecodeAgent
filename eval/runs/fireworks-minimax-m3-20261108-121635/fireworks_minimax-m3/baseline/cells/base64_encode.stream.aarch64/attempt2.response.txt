    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    sub sp, sp, #16

    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, xzr

    cmp x20, #3
    blt .Lhandle_remainder

.Lmain_loop:
    ldrb w3, [x19], #1
    ldrb w4, [x19], #1
    ldrb w5, [x19], #1

    lsr w6, w3, #2
    and w7, w3, #0x3
    lsl w7, w7, #4
    lsr w8, w4, #4
    orr w7, w7, w8

    and w9, w4, #0xF
    lsl w9, w9, #2
    lsr w10, w5, #6
    orr w9, w9, w10

    and w10, w5, #0x3F

    mov w3, w6
    adr x9, .Lret_idx0
    b .Lmap_index
.Lret_idx0:
    mov w3, w7
    adr x9, .Lret_idx1
    b .Lmap_index
.Lret_idx1:
    mov w3, w9
    adr x9, .Lret_idx2
    b .Lmap_index
.Lret_idx2:
    mov w3, w10
    adr x9, .Lret_idx3
    b .Lmap_index
.Lret_idx3:

    sub x20, x20, #3
    add x22, x22, #4
    cmp x20, #3
    b.ge .Lmain_loop

.Lhandle_remainder:
    cbz x20, .Ldone
    cmp x20, #1
    b.eq .Lrem1
    b .Lrem2

.Lrem1:
    ldrb w9, [x19]
    lsr w6, w9, #2
    and w7, w9, #0x3
    lsl w7, w7, #4

    mov w3, w6
    adr x9, .Lret_rem1_0
    b .Lmap_index
.Lret_rem1_0:
    mov w3, w7
    adr x9, .Lret_rem1_1
    b .Lmap_index
.Lret_rem1_1:
    mov w4, #0x3D
    strb w4, [x21], #1
    strb w4, [x21], #1
    add x22, x22, #4
    b .Ldone

.Lrem2:
    ldrb w9, [x19]
    ldrb w10, [x19, #1]
    lsr w6, w9, #2
    and w11, w9, #0x3
    lsl w11, w11, #4
    lsr w12, w10, #4
    orr w7, w11, w12
    and w13, w10, #0xF
    lsl w8, w13, #2

    mov w3, w6
    adr x9, .Lret_rem2_0
    b .Lmap_index
.Lret_rem2_0:
    mov w3, w7
    adr x9, .Lret_rem2_1
    b .Lmap_index
.Lret_rem2_1:
    mov w3, w8
    adr x9, .Lret_rem2_2
    b .Lmap_index
.Lret_rem2_2:
    mov w4, #0x3D
    strb w4, [x21], #1
    add x22, x22, #4
    b .Ldone

.Ldone:
    mov x0, x22
    add sp, sp, #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

.Lmap_index:
    mov w4, #0x41
    add w4, w4, w3
    cmp w3, #25
    b.le .Lmap_index_end
    mov w4, #0x61
    sub w3, w3, #26
    add w4, w4, w3
    cmp w3, #25
    b.le .Lmap_index_end
    mov w4, #0x30
    sub w3, w3, #26
    add w4, w4, w3
    cmp w3, #9
    b.le .Lmap_index_end
    cmp w3, #10
    b.eq .Lmap_index_plus
    mov w4, #0x2F
    b .Lmap_index_end
.Lmap_index_plus:
    mov w4, #0x2B
.Lmap_index_end:
    strb w4, [x21], #1
    br x9
