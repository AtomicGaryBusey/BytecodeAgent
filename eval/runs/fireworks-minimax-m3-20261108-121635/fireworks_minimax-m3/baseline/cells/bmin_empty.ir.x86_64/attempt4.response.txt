export function l $bmin_empty(l %a, l %n) {
@start
	%min =l sub 0, 1
	%min =l shr %min, 1
	%off =l copy 0
	%limit =l mul %n, 8
	jmp @loop

@loop
	%cond =l csltl %off, %limit
	jnz %cond, @body, @done

@body
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%isless =l csltl %val, %min
	jnz %isless, @update, @next

@update
	%min =l copy %val

@next
	%off =l add %off, 8
	jmp @loop

@done
	ret %min
}
