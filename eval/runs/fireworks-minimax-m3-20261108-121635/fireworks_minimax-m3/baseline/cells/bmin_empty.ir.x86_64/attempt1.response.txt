	%min = copy 0x7fffffffffffffff
	%off = copy 0

	jmp @loop

@loop
	%cond = csltl %off, %limit
	jnz %cond, @body, @done

@body
	%ptr = add %a, %off
	%val = loadl %ptr
	%isless = csltl %val, %min
	jnz %isless, @update, @next

@update
	%min = copy %val

@next
	%off = add %off, 8
	jmp @loop

@done
	ret %min
}
