export function l $bmin_empty(l %a, l %n) {
@start
	%min = sub 0l, 1l
	%min = shr %min, 1l
	%off = copy 0l
	%limit = mul %n, 8l
	jmp @loop

@loop
	%cond = csltl %off, %limit
	jnz %cond, @body, @done

@body
	%ptr = add %a, %off
	%val = loadl %ptr
	%isless = csltl %val, %min
	jnz %isless, @update, @next

@update
	%min = copy %val

@next
	%off = add %off, 8l
	jmp @loop

@done
	ret %min
}
