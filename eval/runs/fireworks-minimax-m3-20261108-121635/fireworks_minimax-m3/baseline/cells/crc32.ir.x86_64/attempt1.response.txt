export function l $crc32(l %p, l %n) {
@lstart
	%crc =w copy 0xFFFFFFFF
	%i =l copy 0
	jmp @loop_check
loop_body:
	%ptr_i =l add %p, %i
	%b =w loadub %ptr_i
	%crc =w xor %crc, %b
	%j =w copy 0
	jmp @bit_check
bit_body:
	%bit =w and %crc, 1
	%cond =w ceqw %bit, 1
	jnz %cond, @bit_one, @bit_zero
bit_one:
	%crc =w shr %crc, 1
	%crc =w xor %crc, 0xEDB88320
	jmp @bit_inc
bit_zero:
	%crc =w shr %crc, 1
	jmp @bit_inc
bit_inc:
	%j =w add %j, 1
	jmp @bit_check
bit_check:
	%jc =w csltw %j, 8
	jnz %jc, @bit_body, @after_loop
after_loop:
	%i =l add %i, 1
	jmp @loop_check
loop_check:
	%ic =w csltl %i, %n
	jnz %ic, @loop_body, @final
final:
	%crc =w xor %crc, 0xFFFFFFFF
	%ret =l extuw %crc
	ret %ret
}
