export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop

@loop
	%i =l phi @start 0, @loop_cont %next
	%ge =w csgel %i, %n
	jnz %ge, @retn, @body

@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%z =w ceql %v, 0
	jnz %z, @reti, @loop_cont

@loop_cont
	%next =l add %i, 1
	jmp @loop

@reti
	ret %i

@retn
	ret %n
}
