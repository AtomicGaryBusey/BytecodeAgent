export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%twonn =l mul %nn, 2
	%tmpnn =l mul %nn, 8
	%tmpcur =l copy %tmp
	%tmpscr =l add %tmp, %tmpnn
	%zero =l copy 0
	%one =l copy 1
	%i =l copy 0
	jmp @L1chk
@L1do
	%q =l div %i, %n
	%r =l rem %i, %n
	%d =w ceql %q, %r
	jnz %d, @L1one, @L1zro
@L1one
	%off =l mul %i, 8
	%p =l add %out, %off
	storel %one, %p
	jmp @L1inc
@L1zro
	%off =l mul %i, 8
	%p =l add %out, %off
	storel %zero, %p
	jmp @L1inc
@L1inc
	%i =l add %i, 1
	jmp @L1chk
@L1chk
	%c =w csltl %i, %nn
	jnz %c, @L1do, @L1end
@L1end
	%i =l copy 0
	jmp @L2chk
@L2do
	%off =l mul %i, 8
	%s =l add %a, %off
	%v =l loadl %s
	%d =l add %tmpcur, %off
	storel %v, %d
	%i =l add %i, 1
	jmp @L2chk
@L2chk
	%c =w csltl %i, %nn
	jnz %c, @L2do, @L2end
@L2end
	%bit =l copy 0
	jmp @L3chk
@L3do
	%mask =l shl %one, %bit
	%km =l and %k, %mask
	%is =w ceql %km, %zero
	jnz %is, @L3sqr, @L3res
@L3res
	%mA =l copy %out
	%mB =l copy %tmpcur
	%mC =l copy %tmpscr
	%mD =l copy %out
	%ph =l copy 1
	jmp @mulcp
@L3sqr
	%mA =l copy %tmpcur
	%mB =l copy %tmpcur
	%mC =l copy %tmpscr
	%mD =l copy %tmpcur
	%ph =l copy 2
	jmp @mulcp
@L3inc
	%bit =l add %bit, 1
	jmp @L3chk
@L3chk
	%c =w csltl %bit, 64
	jnz %c, @L3do, @L3end
@L3end
	%i =l copy 0
	jmp @L4chk
@L4do
	%off =l mul %i, 8
	%p =l add %tmp, %off
	storel %zero, %p
	%i =l add %i, 1
	jmp @L4chk
@L4chk
	%c =w csltl %i, %twonn
	jnz %c, @L4do, @L4end
@L4end
	ret %zero

@mulcp
	%mi =l copy 0
	jmp @Mic
@Mib
	%mj =l copy 0
	jmp @Mjc
@Mjb
	%sum =l copy 0
	%mt =l copy 0
	jmp @Mtc
@Mtb
	%ix =l mul %mi, %n
	%ix =l add %ix, %mt
	%oo =l mul %ix, 8
	%pa =l add %mA, %oo
	%va =l loadl %pa
	%jx =l mul %mt, %n
	%jx =l add %jx, %mj
	%oo =l mul %jx, 8
	%pb =l add %mB, %oo
	%vb =l loadl %pb
	%pr =l mul %va, %vb
	%sum =l add %sum, %pr
	%mt =l add %mt, 1
	jmp @Mtc
@Mtc
	%c =w csltl %mt, %n
	jnz %c, @Mtb, @Mte
@Mte
	%kx =l mul %mi, %n
	%kx =l add %kx, %mj
	%oo =l mul %kx, 8
	%pc =l add %mC, %oo
	storel %sum, %pc
	%mj =l add %mj, 1
	jmp @Mjc
@Mjc
	%c =w csltl %mj, %n
	jnz %c, @Mjb, @Mje
@Mje
	%mi =l add %mi, 1
	jmp @Mic
@Mic
	%c =w csltl %mi, %n
	jnz %c, @Mib, @Mie
@Mie
	%ci =l copy 0
	jmp @Cic
@Cib
	%off =l mul %ci, 8
	%s =l add %mC, %off
	%v =l loadl %s
	%d =l add %mD, %off
	storel %v, %d
	%ci =l add %ci, 1
	jmp @Cic
@Cic
	%c =w csltl %ci, %nn
	jnz %c, @Cib, @Cie
@Cie
	%t =w ceql %ph, 1
	jnz %t, @L3sqr, @L3inc
}
