mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov rbp, rdx
    mov r12, r8
    mov r13, rsi
    mov r14, rcx
    mov r15, r13
    add r15, r14
    mov rdi, r12
    mov rcx, r15
    xor rax, rax
    rep stosq
    test r13, r13
    jz .Lret0
    test r14, r14
    jz .Lret0
    shl r13, 3
    shl r14, 3
    xor r9, r9
.Louter:
    mov r11, r12
    add r11, r9
    mov rdx, qword ptr [rbx + r9]
    xor rcx, rcx
    xor r10, r10
.Linner:
    mov rax, qword ptr [rbp + r10]
    mulx rsi, rdi, rax
    add rdi, qword ptr [r11 + r10]
    adc rsi, 0
    add rdi, rcx
    adc rsi, 0
    mov qword ptr [r11 + r10], rdi
    mov rcx, rsi
    add r10, 8
    cmp r10, r14
    jb .Linner
    add rcx, qword ptr [r11 + r14]
    mov qword ptr [r11 + r14], rcx
    jnc .Lno_carry
    lea r10, [r11 + r14 + 8]
    mov rdi, r15
    shl rdi, 3
    add rdi, r12
.Lprop:
    cmp r10, rdi
    jae .Lprop_done
    adc qword ptr [r10], 0
    jnc .Lprop_done
    add r10, 8
    jmp .Lprop
.Lprop_done:
.Lno_carry:
    add r9, 8
    cmp r9, r13
    jb .Louter
    mov rcx, r15
    shl rcx, 3
    add rcx, r12
.Lfind:
    sub rcx, 8
    cmp rcx, r12
    jb .Lall_zero
    mov rdx, qword ptr [rcx]
    test rdx, rdx
    jnz .Lfound
    jmp .Lfind
.Lfound:
    mov rax, rcx
    sub rax, r12
    shr rax, 3
    inc rax
    jmp .Lret
.Lall_zero:
    xor rax, rax
.Lret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lret0:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
