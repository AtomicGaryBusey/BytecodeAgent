export function l $mulhi_u64(l %a, l %b) {
@start
	%al = and %a, 4294967295
	%ah = shr %a, 32
	%bl = and %b, 4294967295
	%bh = shr %b, 32

	%p0 = mul %al, %bl
	%p1 = mul %al, %bh
	%p2 = mul %ah, %bl
	%p3 = mul %ah, %bh

	%p0l = and %p0, 4294967295
	%p0h = shr %p0, 32
	%p1l = and %p1, 4294967295
	%p1h = shr %p1, 32
	%p2l = and %p2, 4294967295
	%p2h = shr %p2, 32
	%p3l = and %p3, 4294967295
	%p3h = shr %p3, 32

	%sum0 = add %p1l, %p2l
	%sum0l = and %sum0, 4294967295
	%carry0 = shr %sum0, 32

	%sum1a = add %p1h, %p2h
	%sum1 = add %sum1a, %carry0
	%sum1l = and %sum1, 4294967295
	%carry1 = shr %sum1, 32

	%mid = add %p0h, %sum0l
	%carry2 = shr %mid, 32

	%t1a = add %sum1l, %p3l
	%t1 = add %t1a, %carry2
	%t1l = and %t1, 4294967295
	%carry3 = shr %t1, 32

	%t2a = add %carry1, %p3h
	%t2 = add %t2a, %carry3
	%t2l = and %t2, 4294967295

	%high = shl %t2l, 32
	%res = or %high, %t1l

	ret %res
}
