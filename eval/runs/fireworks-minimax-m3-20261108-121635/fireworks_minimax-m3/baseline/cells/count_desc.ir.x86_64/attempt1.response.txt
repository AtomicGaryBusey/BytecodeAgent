export function l $count_desc(l %a, l %n) {
	%count = copy 0
	%limit = sub %n, 1
	%cond = csltl %limit, 0
	jnz %cond, @done
	%i = copy 0
	jmp @check

@check:
	%continue = csltl %i, %limit
	jnz %continue, @body, @done

@body:
	%offset = mul %i, 8
	%ptr1 = add %a, %offset
	%val1 = loadl %ptr1
	%offset2 = add %offset, 8
	%ptr2 = add %a, %offset2
	%val2 = loadl %ptr2
	%is_desc = csgtl %val1, %val2
	jnz %is_desc, @inc, @next

@inc:
	%count = add %count, 1

@next:
	%i = add %i, 1
	jmp @check

@done:
	ret %count
}
