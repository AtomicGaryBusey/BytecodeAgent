export function l $count_desc(l %a, l %n) {
@start
	%count =l copy 0
	%limit =l sub %n, 1
	%cond =w csltl %limit, 0
	jnz %cond, @done, @loop_init

@loop_init
	%i =l copy 0
	jmp @check

@check
	%continue =w csltl %i, %limit
	jnz %continue, @body, @done

@body
	%offset =l mul %i, 8
	%ptr1 =l add %a, %offset
	%val1 =l loadl %ptr1
	%offset2 =l add %offset, 8
	%ptr2 =l add %a, %offset2
	%val2 =l loadl %ptr2
	%is_desc =w csgtl %val1, %val2
	jnz %is_desc, @inc, @next

@inc
	%count =l add %count, 1
	jmp @next

@next
	%i =l add %i, 1
	jmp @check

@done
	ret %count
}
