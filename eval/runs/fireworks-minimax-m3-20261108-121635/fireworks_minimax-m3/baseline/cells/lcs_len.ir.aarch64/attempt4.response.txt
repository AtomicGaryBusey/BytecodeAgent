export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %lim_init =l add %n, 1
    %j0 =l copy 0
    jmp @init_check

@init_check
    %j =l phi @start %j0, @init_body %jnext
    %c1 =w csltl %j, %lim_init
    jnz %c1, @init_body, @outer_prep

@init_body
    %rp =l add %row, %j
    storel 0, %rp
    %jnext =l add %j, 1
    jmp @init_check

@outer_prep
    %i0 =l copy 0
    jmp @outer_check

@outer_check
    %i =l phi @outer_prep %i0, @outer_step %inext
    %co =w csltl %i, %m
    jnz %co, @outer_real, @end

@outer_real
    %d0 =l copy 0
    %jjstart =l copy 1
    %cn =w csltl 0, %n
    jnz %cn, @inner_body, @outer_step

@inner_body
    %jj =l phi @outer_real %jjstart, @inner_step %jjnext
    %diag =l phi @outer_real %d0, @inner_step %dnext

    %xp =l add %x, %i
    %xb =w loadub %xp
    %jm1 =l sub %jj, 1
    %yp =l add %y, %jm1
    %yb =w loadub %yp

    %rjp =l add %row, %jj
    %up =l loadl %rjp

    %eq =w ceqw %xb, %yb
    jnz %eq, @inner_eq, @inner_max

@inner_eq
    %nv =l add %diag, 1
    storel %nv, %rjp
    jmp @inner_step

@inner_max
    %rjmp =l add %row, %jm1
    %left =l loadl %rjmp
    %cm =w csltl %up, %left
    jnz %cm, @take_left, @take_up

@take_left
    storel %left, %rjp
    jmp @inner_step

@take_up
    storel %up, %rjp
    jmp @inner_step

@inner_step
    %dnext =l copy %up
    %jjnext =l add %jj, 1
    %ci =w csltl %jj, %n
    jnz %ci, @inner_body, @outer_step

@outer_step
    %inext =l add %i, 1
    jmp @outer_check

@end
    %res_p =l add %row, %n
    %res =l loadl %res_p
    ret %res
}
