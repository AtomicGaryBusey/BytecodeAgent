export function $collatz_max(l %n) -> l {
@start
	jmp @outer_check

@outer_check
	%k = phi @start 1, @update %k1, @skip %k1
	%max = phi @start 0, @update %max1, @skip %max1
	%cond = csgtl %k, %n
	jnz %cond, @return, @outer_body

@outer_body
	jmp @inner_check

@inner_check
	%x = phi @outer_body %k, @inner_cont %x_n
	%cnt = phi @outer_body 0, @inner_cont %cnt_n
	%is_one = ceql %x, 1
	jnz %is_one, @post_inner, @inner_step

@inner_step
	%mask = and %x, 1
	%is_even = ceql %mask, 0
	jnz %is_even, @even, @odd

@even
	%new_x = div %x, 2
	jmp @inner_cont

@odd
	%three = mul %x, 3
	%new_x = add %three, 1
	jmp @inner_cont

@inner_cont
	%x_n = phi @even %new_x, @odd %new_x
	%cnt_n = add %cnt, 1
	jmp @inner_check

@post_inner
	%gt = csltl %max, %cnt
	jnz %gt, @update, @skip

@update
	%max1 = copy %cnt
	%k1 = add %k, 1
	jmp @outer_check

@skip
	%max1 = copy %max
	%k1 = add %k, 1
	jmp @outer_check

@return
	ret %max
}
