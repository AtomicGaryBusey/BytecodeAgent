export function l $collatz_max(l %n) {
@start
	%k0 = copy 1
	%max0 = copy 0
	jmp @outer_check

@outer_check
	%k = phi @start %k0, @outer_step %k1
	%max = phi @start %max0, @outer_step %max1
	%done = csgtl %k, %n
	jnz %done, @return_max, @outer_body

@outer_body
	%x_init = copy %k
	%cnt_init = copy 0
	jmp @inner_check

@inner_check
	%x = phi @outer_body %x_init, @inner_cont %x_n
	%cnt = phi @outer_body %cnt_init, @inner_cont %cnt_n
	%is_one = ceql %x, 1
	jnz %is_one, @post_inner, @inner_step

@inner_step
	%mask = and %x, 1
	%is_even = ceql %mask, 0
	jnz %is_even, @inner_even, @inner_odd

@inner_even
	%new_x = div %x, 2
	jmp @inner_cont

@inner_odd
	%three = mul %x, 3
	%new_x = add %three, 1
	jmp @inner_cont

@inner_cont
	%x_n = phi @inner_even %new_x, @inner_odd %new_x
	%cnt_n = add %cnt, 1
	jmp @inner_check

@post_inner
	%ne_max = csltl %max, %cnt
	jnz %ne_max, @do_update, @skip_update

@do_update
	%next_max = copy %cnt
	jmp @outer_step

@skip_update
	%next_max = copy %max
	jmp @outer_step

@outer_step
	%max1 = phi @do_update %next_max, @skip_update %next_max
	%k1 = add %k, 1
	jmp @outer_check

@return_max
	ret %max
}
