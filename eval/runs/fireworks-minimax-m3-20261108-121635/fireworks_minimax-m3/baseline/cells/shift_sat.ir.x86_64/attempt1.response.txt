export function l $shift_sat(l %x, l %n) {
@start
	%lt64 =w csltl %n, 64
	jnz %lt64, @small_n, @large_n

@small_n
	%left =l shl %x, %n
	%right =l sar %x, %n
	jmp @merge(%left, %right)

@large_n
	%left =l copy 0
	%right =l sar %x, 63
	jmp @merge(%left, %right)

@merge(l %left, l %right)
	%res =l add %left, %right
	ret %res
}
