base64_encode:
    push r12
    push r13
    push r14
    push r15
    sub rsp, 64

    mov cl, 0
    mov al, 'A'
.LfillA:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 26
    jb .LfillA

    mov cl, 26
    mov al, 'a'
.Lfilla:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 52
    jb .Lfilla

    mov cl, 52
    mov al, '0'
.Lfill0:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 62
    jb .Lfill0

    mov byte ptr [rsp + 62], '+'
    mov byte ptr [rsp + 63], '/'

    mov r12, rdx
    mov r13, rsi

    mov rax, r13
    xor edx, edx
    mov ecx, 3
    div ecx
    mov r15, rax
    mov r14, rdx

.Lloop_full:
    test r15, r15
    jz .Lhandle_tail

    movzx eax, byte ptr [rdi]
    movzx ecx, byte ptr [rdi + 1]
    movzx edx, byte ptr [rdi + 2]
    add rdi, 3

    mov r8d, eax
    shr r8d, 2
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r9d, ecx
    shr r9d, 4
    or r8d, r9d
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, ecx
    and r8d, 0xF
    shl r8d, 2
    mov r9d, edx
    shr r9d, 6
    or r8d, r9d
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, edx
    and r8d, 0x3F
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    dec r15
    jmp .Lloop_full

.Lhandle_tail:
    test r14, r14
    jz .Ldone

    cmp r14, 1
    je .Lrem1

    movzx eax, byte ptr [rdi]
    movzx ecx, byte ptr [rdi + 1]

    mov r8d, eax
    shr r8d, 2
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r9d, ecx
    shr r9d, 4
    or r8d, r9d
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, ecx
    and r8d, 0xF
    shl r8d, 2
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov byte ptr [r12], '='
    inc r12
    jmp .Ldone

.Lrem1:
    movzx eax, byte ptr [rdi]

    mov r8d, eax
    shr r8d, 2
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r8b, byte ptr [rsp + r8]
    mov byte ptr [r12], r8b
    inc r12

    mov byte ptr [r12], '='
    inc r12
    mov byte ptr [r12], '='
    inc r12

.Ldone:
    mov rax, r13
    add rax, 2
    xor edx, edx
    mov ecx, 3
    div ecx
    shl rax, 2

    add rsp, 64
    pop r15
    pop r14
    pop r13
    pop r12
    ret
