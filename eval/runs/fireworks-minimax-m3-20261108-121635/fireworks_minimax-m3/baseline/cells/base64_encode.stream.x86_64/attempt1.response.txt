base64_encode:
    push r12
    push r13
    push r14
    sub rsp, 68

    mov r12, rsi
    mov rax, rsi
    xor edx, edx
    mov ecx, 3
    div ecx
    mov r13, rax
    mov r14, rdx

    mov ecx, 26
    mov al, 'A'
.LfillA:
    mov byte ptr [rsp + rcx - 1], al
    inc al
    dec ecx
    jnz .LfillA

    mov ecx, 26
    mov al, 'a'
.Lfilla:
    mov byte ptr [rsp + 26 + rcx - 1], al
    inc al
    dec ecx
    jnz .Lfilla

    mov ecx, 10
    mov al, '0'
.Lfill0:
    mov byte ptr [rsp + 52 + rcx - 1], al
    inc al
    dec ecx
    jnz .Lfill0

    mov byte ptr [rsp + 62], '+'
    mov byte ptr [rsp + 63], '/'

    test r13, r13
    jz .handle_tail

.loop_full:
    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    movzx r10d, byte ptr [rdi + 2]
    add rdi, 3

    mov al, r8b
    shr al, 2
    mov byte ptr [rsp + 64], al

    mov al, r8b
    and al, 3
    shl al, 4
    mov r11b, al
    mov al, r9b
    shr al, 4
    or al, r11b
    mov byte ptr [rsp + 65], al

    mov al, r9b
    and al, 0xF
    shl al, 2
    mov r11b, al
    mov al, r10b
    shr al, 6
    or al, r11b
    mov byte ptr [rsp + 66], al

    mov al, r10b
    and al, 0x3F
    mov byte ptr [rsp + 67], al

    xor ecx, ecx
.Lmap_full:
    movzx eax, byte ptr [rsp + 64 + rcx]
    mov al, byte ptr [rsp + rax]
    mov byte ptr [rdx], al
    inc rdx
    inc ecx
    cmp ecx, 4
    jb .Lmap_full

    dec r13
    test r13, r13
    jnz .loop_full

.handle_tail:
    test r14, r14
    jz .done

    cmp r14, 1
    je .rem1

    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    add rdi, 2

    mov al, r8b
    shr al, 2
    mov byte ptr [rsp + 64], al

    mov al, r8b
    and al, 3
    shl al, 4
    mov r11b, al
    mov al, r9b
    shr al, 4
    or al, r11b
    mov byte ptr [rsp + 65], al

    mov al, r9b
    and al, 0xF
    shl al, 2
    mov byte ptr [rsp + 66], al

    xor ecx, ecx
.Lmap_rem2:
    movzx eax, byte ptr [rsp + 64 + rcx]
    mov al, byte ptr [rsp + rax]
    mov byte ptr [rdx], al
    inc rdx
    inc ecx
    cmp ecx, 3
    jb .Lmap_rem2
    mov byte ptr [rdx], '='
    inc rdx
    jmp .done

.rem1:
    movzx r8d, byte ptr [rdi]
    add rdi, 1

    mov al, r8b
    shr al, 2
    mov byte ptr [rsp + 64], al

    mov al, r8b
    and al, 3
    shl al, 4
    mov byte ptr [rsp + 65], al

    xor ecx, ecx
.Lmap_rem1:
    movzx eax, byte ptr [rsp + 64 + rcx]
    mov al, byte ptr [rsp + rax]
    mov byte ptr [rdx], al
    inc rdx
    inc ecx
    cmp ecx, 2
    jb .Lmap_rem1
    mov byte ptr [rdx], '='
    inc rdx
    mov byte ptr [rdx], '='
    inc rdx

.done:
    mov rax, r13
    shl rax, 2
    test r14, r14
    jz .return
    add rax, 4
.return:
    add rsp, 68
    pop r14
    pop r13
    pop r12
    ret
