base64_encode:
    # rdi = src, rsi = n, rdx = out
    mov r10, rdx            # save out
    mov rax, rsi
    xor edx, edx
    mov rcx, 3
    div rcx                 # rax = n/3, rdx = n%3
    mov r8, rax             # full_groups
    mov r9, rdx             # remainder
    mov rdx, r10            # restore out

    test r8, r8
    jz .Lhandle_remainder

.Lfull_loop:
    movzx eax, byte [rdi]
