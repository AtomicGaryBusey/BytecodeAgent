base64_encode:
    push r12
    push r13
    push r14
    push r15
    sub rsp, 64

    # Fill uppercase A-Z
    mov cl, 0
    mov al, 'A'
.LfillA:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 26
    jb .LfillA

    # Fill lowercase a-z
    mov cl, 26
    mov al, 'a'
.Lfilla:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 52
    jb .Lfilla

    # Fill digits 0-9
    mov cl, 52
    mov al, '0'
.Lfill0:
    mov byte ptr [rsp + rcx], al
    inc al
    inc cl
    cmp cl, 62
    jb .Lfill0

    mov byte ptr [rsp + 62], '+'
    mov byte ptr [rsp + 63], '/'

    mov r12, rdx
    mov r13, rsi

    mov rax, r13
    xor edx, edx
    mov ecx, 3
    div ecx
    mov r15, rax
    mov r14, rdx

.Lloop_full:
    test r15, r15
    jz .Lhandle_tail

    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdi+1]
    movzx r10d, byte ptr [rdi+2]
    add rdi, 3

    # i0
    mov al, r8b
    shr al, 2
    movzx eax, al
    mov al, byte ptr [rsp + rax]
    mov byte ptr [r12], al
    inc r12

    # i1
    mov al, r8b
    and al, 3
    shl al, 4
    mov r11b, r9b
    shr r11b, 4
    or al, r11b
    movzx eax, al
    mov al, byte ptr [rsp + rax]
    mov byte ptr [r12], al
    inc r12

    # i2
    mov al, r9b
    and al, 0xF
    shl al, 2
    mov r11b, r10b
    shr r11b, 6
    or al, r11b
    movzx eax, al
    mov al, byte ptr [rsp + rax]
    mov byte ptr [r12], al
    inc r12

    # i3
    mov al, r10b
    and al, 0x3F
    movzx eax, al
    mov al, byte ptr [rsp + rax]
    mov byte ptr [r12], al
    inc r12

    dec r15
    jmp .Lloop_full

.Lhandle_tail:
    test r14, r14
    jz .Ldone

    cmp r14, 1
    je .Lrem1

    # remainder == 2
    movzx r8b, byte ptr [rdi]
