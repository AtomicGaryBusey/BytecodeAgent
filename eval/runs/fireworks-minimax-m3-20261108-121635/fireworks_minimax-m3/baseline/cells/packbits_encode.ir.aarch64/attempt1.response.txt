export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  jmp @loop_cond

@loop_cond
  %i =l phi @start 0, @do_repeat %i_rep, @copy_end %i_lit
  %written =l phi @start 0, @do_repeat %written_rep, @copy_end %written_lit
  %out_cur =l phi @start %out, @do_repeat %out_cur_rep, @copy_end %out_cur_lit
  %mcond =w csltl %i, %n
  jnz %mcond, @loop_body, @done

@loop_body
  jmp @run_start

@run_start
  %next0 =l add %i, 1
  jmp @run_check

@run_check
  %run =l phi @run_start 1, @run_inc %run2
  %j =l phi @run_start %i, @run_inc %j_run
  %next =l phi @run_start %next0, @run_inc %next_run
  %rc1 =w csltl %next, %n
  %rc2 =w csltl %run, 128
  %rcond =w and %rc1, %rc2
  jnz %rcond, @run_eq, @run_end

@run_eq
  %p1 =l add %src, %j
  %p2 =l add %src, %next
  %b1_run =w loadub %p1
  %b2_run =w loadub %p2
  %req =w ceqw %b1_run, %b2_run
  jnz %req, @run_inc, @run_end

@run_inc
  %run2 =l add %run, 1
  %j_run =l add %j, 1
  %next_run =l add %next, 1
  jmp @run_check

@run_end
  %ge3 =w csltl 2, %run
  jnz %ge3, @emit_repeat, @lit_start

@emit_repeat
  %rneed =l 2
  %rhave =l sub %cap, %written
  %rfit =w culel %rneed, %rhave
  jnz %rfit, @do_repeat, @fail

@do_repeat
  %hdr =l sub 257, %run
  storeb %hdr, %out_cur
  %val_ptr =l add %src, %i
  %val =w loadub %val_ptr
  %out_next =l add %out_cur, 1
  storeb %val, %out_next
  %i_rep =l add %i, %run
  %written_rep =l add %written, 2
  %out_cur_rep =l add %out_cur, 2
  jmp @loop_cond

@lit_start
  jmp @lit_check

@lit_check
  %j =l phi @lit_start %i, @lit_inc %j_lit
  %l =l phi @lit_start 0, @lit_inc %l_lit
  %lc1 =w csltl %j, %n
  %lc2 =w csltl %l, 128
  %lcond =w and %lc1, %lc2
  jnz %lcond, @lit_run, @lit_end

@lit_run
  %j2_idx =l add %j, 2
  %lc3 =w csltl %j2_idx, %n
  jnz %lc3, @lit_check_eq, @lit_inc

@lit_check_eq
  %p_j =l add %src, %j
  %b1_addr =l add %p_j, 1
  %b2_addr =l add %p_j, 2
  %b0_lit =w loadub %p_j
  %b1_lit =w loadub %b1_addr
  %b2_lit =w loadub %b2_addr
  %leq1 =w ceqw %b0_lit, %b1_lit
  %leq2 =w ceqw %b1_lit, %b2_lit
  %leq =w and %leq1, %leq2
  jnz %leq, @lit_end, @lit_inc

@lit_inc
  %j_lit =l add %j, 1
  %l_lit =l add %l, 1
  jmp @lit_check

@lit_end
  %lneed =l add %l, 1
  %lhave =l sub %cap, %written
  %lfit =w culel %lneed, %lhave
  jnz %lfit, @do_literal, @fail

@do_literal
  %src_i =l add %src, %i
  %dst_i =l add %out_cur, 1
  jmp @copy_check

@copy_check
  %k =l phi @do_literal 0, @copy_inc %k2
  %src_ptr =l phi @do_literal %src_i, @copy_inc %src_ptr2
  %dst_ptr =l phi @do_literal %dst_i, @copy_inc %dst_ptr2
  %ccond =w csltl %k, %l
  jnz %ccond, @copy_body, @copy_end

@copy_body
  %b =w loadub %src_ptr
  storeb %b, %dst_ptr
  jmp @copy_inc

@copy_inc
  %k2 =l add %k, 1
  %src_ptr2 =l add %src_ptr, 1
  %dst_ptr2 =l add %dst_ptr, 1
  jmp @copy_check

@copy_end
  %tmp =l add %written, %l
  %written_lit =l add %tmp, 1
  %tmp2 =l add %out_cur, %l
  %out_cur_lit =l add %tmp2, 1
  %i_lit =l add %i, %l
  jmp @loop_cond

@fail
  ret -1

@done
  ret %written
}
