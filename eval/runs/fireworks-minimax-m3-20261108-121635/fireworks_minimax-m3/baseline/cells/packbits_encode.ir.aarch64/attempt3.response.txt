export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  jmp @cond

@cond
  %i =l phi @start 0, @do_rep %i_r, @cend %i_l
  %w =l phi @start 0, @do_rep %w_r, @cend %w_l
  %oc =l phi @start %out, @do_rep %oc_r, @cend %oc_l
  %c0 =w csltl %i, %n
  jnz %c0, @body, @done

@body
  %ip1 =l add %i, 1
  jmp @rchk

@rchk
  %run =l phi @body 1, @rinc %run2
  %rj =l phi @body %i, @rinc %rj2
  %rn =l phi @body %ip1, @rinc %rn2
  %c1 =w csltl %rn, %n
  %c2 =w csltl %run, 128
  %cand =w and %c1, %c2
  jnz %cand, @req, @rend

@req
  %pa =l add %src, %rj
  %pb =l add %src, %rn
  %va =w loadub %pa
  %vb =w loadub %pb
  %eq =w ceqw %va, %vb
  jnz %eq, @rinc, @rend

@rinc
  %run2 =l add %run, 1
  %rj2 =l add %rj, 1
  %rn2 =l add %rn, 1
  jmp @rchk

@rend
  %ge3 =w csltl 2, %run
  jnz %ge3, @fits_rep, @lchk

@fits_rep
  %free =l sub %cap, %w
  %fits =w culel 2, %free
  jnz %fits, @do_rep, @fail

@do_rep
  %hsub =l sub 257, %run
  %hw =w copy %hsub
  storeb %hw, %oc
  %vp =l add %src, %i
  %v =w loadub %vp
  %ocp =l add %oc, 1
  storeb %v, %ocp
  %i_r =l add %i, %run
  %w_r =l add %w, 2
  %oc_r =l add %oc, 2
  jmp @cond

@lchk
  %lj =l phi @rend %i, @linc %lj2
  %ll =l phi @rend 0, @linc %ll2
  %c3 =w csltl %lj, %n
  %c4 =w csltl %ll, 128
  %cand2 =w and %c3, %c4
  jnz %cand2, @lrun, @lend

@lrun
  %ljp2 =l add %lj, 2
  %c5 =w csltl %ljp2, %n
  jnz %c5, @leq, @linc

@leq
  %pj =l add %src, %lj
  %pja =l add %pj, 1
  %pjb =l add %pj, 2
  %bv0 =w loadub %pj
  %bv1 =w loadub %pja
  %bv2 =w loadub %pjb
  %eq1 =w ceqw %bv0, %bv1
  %eq2 =w ceqw %bv1, %bv2
  %eq12 =w and %eq1, %eq2
  jnz %eq12, @lend, @linc

@linc
  %lj2 =l add %lj, 1
  %ll2 =l add %ll, 1
  jmp @lchk

@lend
  %free2 =l sub %cap, %w
  %lneed =l add %ll, 1
  %lfit =w culel %lneed, %free2
  jnz %lfit, @do_lit, @fail

@do_lit
  %hlsub =l sub %ll, 1
  %hlw =w copy %hlsub
  storeb %hlw, %oc
  %sp =l add %src, %i
  %dp =l add %oc, 1
  jmp @cchk

@cchk
  %k =l phi @do_lit 0, @cinc %k2
  %spp =l phi @do_lit %sp, @cinc %spp2
  %dpp =l phi @do_lit %dp, @cinc %dpp2
  %cc =w csltl %k, %ll
  jnz %cc, @cbody, @cend

@cbody
  %bv =w loadub %spp
  storeb %bv, %dpp
  jmp @cinc

@cinc
  %k2 =l add %k, 1
  %spp2 =l add %spp, 1
  %dpp2 =l add %dpp, 1
  jmp @cchk

@cend
  %i_l =l add %i, %ll
  %w_l_a =l add %w, %ll
  %w_l =l add %w_l_a, 1
  %oc_l_a =l add %oc, %ll
  %oc_l =l add %oc_l_a, 1
  jmp @cond

@fail
  ret -1

@done
  ret %w
}
