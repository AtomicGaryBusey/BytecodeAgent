export function l $qsort_pairs(l %a, l %n) {
    %cond0 =w csltl %n, 2
    jnz %cond0, @ret0, @start

@ret0:
    %ret_val =l copy 0
    ret %ret_val

@start:
    %hi =l sub %n, 1
    %pidx =l div %n, 2

    %off_p =l shl %pidx, 4
    %off_h =l shl %hi, 4
    %addr_p =l add %a, %off_p
    %addr_h =l add %a, %off_h

    %key_p =l loadl %addr_p
    %addr_p8 =l add %addr_p, 8
    %val_p =l loadl %addr_p8
    %key_h =l loadl %addr_h
    %addr_h8 =l add %addr_h, 8
    %val_h =l loadl %addr_h8

    storel %key_h, %addr_p
    storel %val_h, %addr_p8
    storel %key_p, %addr_h
    storel %val_p, %addr_h8

    %pkey =l loadl %addr_h
    %pval =l loadl %addr_h8

    %zero =l add 0, 0
    %i =l sub %zero, 1
    %j =l add 0, 0

    jmp @part_check

@part_check:
    %cond_j =w csltl %j, %hi
    jnz %cond_j, @part_body, @part_done

@part_body:
    %off_j =l shl %j, 4
    %addr_jk =l add %a, %off_j
    %key_j =l loadl %addr_jk
    %addr_jv =l add %addr_jk, 8
    %val_j =l loadl %addr_jv

    %k_lt =w csltl %key_j, %pkey
    %k_eq =w ceql %key_j, %pkey
    %v_le =w csltl %val_j, %pval
    %eq_and_le =w and %k_eq, %v_le
    %cond =w or %k_lt, %eq_and_le

    jnz %cond, @inc_i, @no_inc

@inc_i:
    %i =l add %i, 1
    %i_eq_j =w ceql %i, %j
    jnz %i_eq_j, @skip_swap, @do_swap

@do_swap:
    %off_i =l shl %i, 4
    %addr_ik =l add %a, %off_i
    %addr_iv =l add %addr_ik, 8

    %key_i =l loadl %addr_ik
    %val_i =l loadl %addr_iv

    storel %key_j, %addr_ik
    storel %val_j, %addr_iv
    storel %key_i, %addr_jk
    storel %val_i, %addr_jv

    jmp @inc_j

@skip_swap:
    jmp @inc_j

@no_inc:
    jmp @inc_j

@inc_j:
    %j =l add %j, 1
    jmp @part_check

@part_done:
    %p =l add %i, 1

    %off_p =l shl %p, 4
    %addr_pk =l add %a, %off_p
    %addr_pv =l add %addr_pk, 8

    %off_h =l shl %hi, 4
    %addr_hk =l add %a, %off_h
    %addr_hv =l add %addr_hk, 8

    %key_p =l loadl %addr_pk
    %val_p =l loadl %addr_pv
    %key_h =l loadl %addr_hk
    %val_h =l loadl %addr_hv

    storel %key_h, %addr_pk
    storel %val_h, %addr_pv
    storel %key_p, %addr_hk
    storel %val_p, %addr_hv

    %left_n =l copy %p
    %right_n =l sub %hi, %p

    %p1 =l add %p, 1
    %off_right =l shl %p1, 4
    %right_a =l add %a, %off_right

    %left_lt_right =w csltl %left_n, %right_n
    jnz %left_lt_right, @recurse_left, @recurse_right

@recurse_left:
    %r1 =l call $qsort_pairs(l %a, l %left_n)
    %a =l copy %right_a
    %n =l copy %right_n
    jmp @start

@recurse_right:
    %r2 =l call $qsort_pairs(l %right_a, l %right_n)
    %n =l copy %left_n
    jmp @start
}
