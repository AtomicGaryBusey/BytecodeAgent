export function l $qsort_pairs(l %a, l %n) {
@start
	jmp @check
@check
	%a_cur =l phi @start %a, @after_rec_l %a_loop_r, @after_rec_r %a_loop_l
	%n_cur =l phi @start %n, @after_rec_l %n_loop_r, @after_rec_r %n_loop_l
	%n_lt_2 =w csltl %n_cur, 2
	jnz %n_lt_2, @ret_zero, @prep
@ret_zero
	ret 0
@prep
	%mid =l div %n_cur, 2
	%last_idx =l sub %n_cur, 1
	%mid_eq_l =w ceql %mid, %last_idx
	jnz %mid_eq_l, @load_piv, @do_swap_piv
@do_swap_piv
	%mid_off =l mul %mid, 16
	%last_off =l mul %last_idx, 16
	%mp =l add %a_cur, %mid_off
	%lp =l add %a_cur, %last_off
	%mpp =l add %mp, 8
	%lpp =l add %lp, 8
	%mk =l loadl %mp
	%mv =l loadl %mpp
	%lk =l loadl %lp
	%lv =l loadl %lpp
	storel %mp, %lk
	storel %mpp, %lv
	storel %lp, %mk
	storel %lpp, %mv
	jmp @load_piv
@load_piv
	%lo2 =l mul %last_idx, 16
	%lptr =l add %a_cur, %lo2
	%lptr8 =l add %lptr, 8
	%piv_k =l loadl %lptr
	%piv_v =l loadl %lptr8
	jmp @iloop_init
@iloop_init
	%i_init =l copy -1
	jmp @iloop_check
@iloop_check
	%i_idx =l phi @iloop_init %i_init, @iloop_step %i_step
	%j_idx =l phi @iloop_init 0, @iloop_step %j_step
	%last_limit =l sub %n_cur, 1
	%j_lt_lim =w csltl %j_idx, %last_limit
	jnz %j_lt_lim, @iloop_body, @post_partition
@iloop_body
	%j_off =l mul %j_idx, 16
	%jp =l add %a_cur, %j_off
	%jpv =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jpv
	%k_lt_p =w csltl %jk, %piv_k
	%k_eq_p =w ceql %jk, %piv_k
	%v_lt_p =w csltl %jv, %piv_v
	%v_eq_p =w ceql %jv, %piv_v
	%v_le_p =w or %v_lt_p, %v_eq_p
	%keq_vle =w and %k_eq_p, %v_le_p
	%is_le =w or %k_lt_p, %keq_vle
	jnz %is_le, @inc, @iloop_step
@inc
	%i_new =l add %i_idx, 1
	%need_swap =w cnel %i_new, %j_idx
	jnz %need_swap, @do_swap, @iloop_step
@do_swap
	%i_off =l mul %i_new, 16
	%ip =l add %a_cur, %i_off
	%ipv =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ipv
	%jk_sw =l loadl %jp
	%jv_sw =l loadl %jpv
	storel %ip, %jk_sw
	storel %ipv, %jv_sw
	storel %jp, %ik
	storel %jpv, %iv
	jmp @iloop_step
@iloop_step
	%i_step =l phi @iloop_body %i_idx, @inc %i_new, @do_swap %i_new
	%j_step =l add %j_idx, 1
	jmp @iloop_check
@post_partition
	%ip1 =l add %i_idx, 1
	%fp_eq =w ceql %ip1, %last_idx
	jnz %fp_eq, @choose, @do_final_swap
@do_final_swap
	%ip1_off =l mul %ip1, 16
	%ip1_p =l add %a_cur, %ip1_off
	%ip1_p8 =l add %ip1_p, 8
	%fp_off =l mul %last_idx, 16
	%fp =l add %a_cur, %fp_off
	%fp8 =l add %fp, 8
	%ip1k =l loadl %ip1_p
	%ip1v =l loadl %ip1_p8
	%fpk =l loadl %fp
	%fpv =l loadl %fp8
	storel %ip1_p, %fpk
	storel %ip1_p8, %fpv
	storel %fp, %ip1k
	storel %fp8, %ip1v
	jmp @choose
@choose
	%left_n =l copy %ip1
	%rn1 =l sub %n_cur, %ip1
	%right_n =l sub %rn1, 1
	%rec_left =w csltl %left_n, %right_n
	jnz %rec_left, @after_rec_l, @after_rec_r
@after_rec_l
	%r_l =l call $qsort_pairs(l %a_cur, l %left_n)
	%a_lr_off =l mul %ip1, 16
	%a_lr_off2 =l add %a_lr_off, 16
	%a_loop_r =l add %a_cur, %a_lr_off2
	%n_loop_r =l copy %right_n
	jmp @check
@after_rec_r
	%a_cr_off =l mul %ip1, 16
	%a_cr_off2 =l add %a_cr_off, 16
	%a_call_r =l add %a_cur, %a_cr_off2
	%r_r =l call $qsort_pairs(l %a_call_r, l %right_n)
	%a_loop_l =l copy %a_cur
	%n_loop_l =l copy %left_n
	jmp @check
}
