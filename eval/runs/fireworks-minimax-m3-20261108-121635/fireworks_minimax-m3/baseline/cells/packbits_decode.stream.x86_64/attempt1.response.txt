packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r14, rsi
    mov r13, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lloop:
    cmp rbx, r14
    jae .Ldone
    movsx rax, byte ptr [r12 + rbx]
    inc rbx
    test rax, rax
    js .Lnegative
    lea rax, [rax + 1]
    mov rcx, rbx
    add rcx, rax
    cmp rcx, r14
    ja .Lerr_src
    mov rcx, rbp
    add rcx, rax
    cmp rcx, r15
    ja .Lover_literal
    lea rdi, [r13 + rbp]
    lea rsi, [r12 + rbx]
    mov rcx, rax
    rep movsb
    add rbx, rax
    add rbp, rax
    jmp .Lloop
.Lnegative:
    cmp rax, -128
    je .Lloop
    neg rax
    inc rax
    lea rcx, [rbx + 1]
    cmp rcx, r14
    ja .Lerr_src
    mov rcx, rbp
    add rcx, rax
    cmp rcx, r15
    ja .Lover_repeat
    mov r10, rax
    movzx eax, byte ptr [r12 + rbx]
    lea rdi, [r13 + rbp]
    mov rcx, r10
    rep stosb
    add rbx, 1
    add rbp, r10
    jmp .Lloop
.Lerr_src:
    mov rax, -2
    jmp .Lepilogue
.Lover_literal:
    mov rcx, r15
    sub rcx, rbp
    lea rdi, [r13 + rbp]
    lea rsi, [r12 + rbx]
    rep movsb
    mov rax, -1
    jmp .Lepilogue
.Lover_repeat:
    movzx eax, byte ptr [r12 + rbx]
    lea rdi, [r13 + rbp]
    mov rcx, r15
    sub rcx, rbp
    rep stosb
    mov rax, -1
    jmp .Lepilogue
.Ldone:
    mov rax, rbp
.Lepilogue:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
