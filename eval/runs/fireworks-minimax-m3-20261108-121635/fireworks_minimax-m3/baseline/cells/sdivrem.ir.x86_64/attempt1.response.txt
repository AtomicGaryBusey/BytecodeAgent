export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%iszero =w ceql %b, 0
	jnz %iszero, @bzero, @checkmin

@bzero
	%r0 =l copy %a
	storel %r0, %rem
	%q0 =l copy 0
	ret %q0

@checkmin
	%isamin =w ceql %a, -9223372036854775808
	%isbnegone =w ceql %b, -1
	%both =w and %isamin, %isbnegone
	jnz %both, @mincase, @normal

@mincase
	%r1 =l copy 0
	storel %r1, %rem
	%q1 =l copy -9223372036854775808
	ret %q1

@normal
	%q2 =l div %a, %b
	%r2 =l rem %a, %b
	storel %r2, %rem
	ret %q2
}
