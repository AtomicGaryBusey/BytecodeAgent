export function l $mulhi_u64(l %a, l %b) {
@l1
	%zero = mul 0, %a
	%neg1 = add -1, %zero
	%mask = shr 32, %neg1
	%alo = and %mask, %a
	%ahi = shr 32, %a
	%blo = and %mask, %b
	%bhi = shr 32, %b
	%p0 = mul %alo, %blo
	%p1 = mul %alo, %bhi
	%p2 = mul %ahi, %blo
	%p3 = mul %ahi, %bhi
	%p0hi = shr 32, %p0
	%p1lo = and %mask, %p1
	%p1hi = shr 32, %p1
	%p2lo = and %mask, %p2
	%p2hi = shr 32, %p2
	%p3lo = and %mask, %p3
	%p3hi = shr 32, %p3
	%s1 = add %p0hi, %p1lo
	%s1 = add %s1, %p2lo
	%s1hi = shr 32, %s1
	%s2 = add %s1hi, %p1hi
	%s2 = add %s2, %p2hi
	%s2 = add %s2, %p3lo
	%s2lo = and %mask, %s2
	%s2hi = shr 32, %s2
	%s3 = add %s2hi, %p3hi
	%s3lo = and %mask, %s3
	%hipart = shl 32, %s3lo
	%res = or %hipart, %s2lo
	ret %res
}
