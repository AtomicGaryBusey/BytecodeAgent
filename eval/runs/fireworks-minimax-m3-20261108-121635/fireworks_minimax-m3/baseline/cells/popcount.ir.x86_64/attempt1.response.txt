export function l $popcount(l %x) {
	%c55 = int64 0x5555555555555555
	%c33 = int64 0x3333333333333333
	%c0f = int64 0x0f0f0f0f0f0f0f0f
	%c01 = int64 0x0101010101010101

	%t1 = shr %x, 1
	%t2 = and %t1, %c55
	%x1 = sub %x, %t2

	%u1 = and %x1, %c33
	%t3 = shr %x1, 2
	%u2 = and %t3, %c33
	%x2 = add %u1, %u2

	%t4 = shr %x2, 4
	%x3 = add %x2, %t4
	%x3m = and %x3, %c0f

	%t5 = mul %x3m, %c01
	%r = shr %t5, 56
	ret %r
}
