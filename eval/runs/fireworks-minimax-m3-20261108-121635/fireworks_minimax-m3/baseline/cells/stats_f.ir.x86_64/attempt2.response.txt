export function l $stats_f(l %a, l %n, l %f, l %out) {
@first
	%one =l copy 1
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%yand0 =l and %y0, %one
	%isodd_w0 =w ceql %yand0, %one
	%isodd0 =l ext %isodd_w0
	%odd =l copy %isodd0
	%i_start =l copy 1
	jmp @loop

@loop
	%min =l phi @first %min, @update_sum %min
	%max =l phi @first %max, @update_sum %max
	%sum =l phi @first %sum, @update_sum %sum
	%odd =l phi @first %odd, @update_sum %new_odd
	%i =l phi @first %i_start, @update_sum %i_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @end

@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	jnz %ismin, @upd_min, @check_max

@upd_min
	%new_min =l copy %y
	jmp @check_max

@check_max
	%min =l phi @body %min, @upd_min %new_min
	%ismax =w csltl %max, %y
	jnz %ismax, @upd_max, @update_sum

@upd_max
	%new_max =l copy %y
	jmp @update_sum

@update_sum
	%min =l phi @check_max %min, @upd_max %min
	%max =l phi @check_max %max, @upd_max %new_max
	%sum =l add %sum, %y
	%yand =l and %y, %one
	%isodd_w =w ceql %yand, %one
	%isodd =l ext %isodd_w
	%new_odd =l add %odd, %isodd
	%i_next =l add %i, 1
	jmp @loop

@end
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
