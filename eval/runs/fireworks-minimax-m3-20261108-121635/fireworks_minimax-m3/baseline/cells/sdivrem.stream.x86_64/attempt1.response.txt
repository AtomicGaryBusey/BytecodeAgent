sdivrem:
    test rsi, rsi
    jz .Ldiv_by_zero
    
    cmp rsi, -1
    jne .Lnormal
    
    mov rax, 0x8000000000000000
    cmp rdi, rax
    jne .Lnormal
    
    mov qword ptr [rdx], 0
    ret

.Ldiv_by_zero:
    mov qword ptr [rdx], rdi
    xor eax, eax
    ret

.Lnormal:
    mov rax, rdi
    mov rcx, rdx
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
