export function l $count_ugt(l %a, l %n, l %t) {
    %cnt =l copy 0
    %i =l copy 0
    %ptr =l copy %a
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %elem =l loadl %ptr
    %gt =w cugtl %elem, %t
    jnz %gt, @inc, @next

@inc
    %cnt =l add %cnt, 1

@next
    %ptr =l add %ptr, 8
    %i =l add %i, 1
    jmp @loop

@end
    ret %cnt
}
