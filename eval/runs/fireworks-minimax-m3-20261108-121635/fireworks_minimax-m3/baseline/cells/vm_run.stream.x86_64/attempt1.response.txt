vm_run:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  xor edi, edi
  mov rbx, r14
.Lloop:
  cmp rdi, r13
  jge .Ldone
  mov rax, qword ptr [r12 + rdi*8]
  inc rdi
  cmp eax, 0
  je .Lhalt
  cmp eax, 1
  je .Lpush
  cmp eax, 2
  je .Ladd
  cmp eax, 3
  je .Lsub
  cmp eax, 4
  je .Lmul
  cmp eax, 5
  je .Ldup
  cmp eax, 6
  je .Lswap
  cmp eax, 7
  je .Ldrop
  cmp eax, 8
  je .Lover
  cmp eax, 9
  je .Llt
  cmp eax, 10
  je .Ljz
  cmp eax, 11
  je .Ljmp
  jmp .Lloop
.Lhalt:
  jmp .Ldone
.Lpush:
  mov rax, qword ptr [r12 + rdi*8]
  inc rdi
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lloop
.Ladd:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  add rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lloop
.Lsub:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  sub rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lloop
.Lmul:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  imul rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lloop
.Ldup:
  mov rax, qword ptr [rbx - 8]
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lloop
.Lswap:
  mov rax, qword ptr [rbx - 8]
  mov rcx, qword ptr [rbx - 16]
  mov qword ptr [rbx - 8], rcx
  mov qword ptr [rbx - 16], rax
  jmp .Lloop
.Ldrop:
  sub rbx, 8
  jmp .Lloop
.Lover:
  mov rax, qword ptr [rbx - 16]
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lloop
.Llt:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  cmp qword ptr [rbx - 8], rax
  mov rax, 0
  setl al
  mov qword ptr [rbx - 8], rax
  jmp .Lloop
.Ljz:
  sub rbx, 8
  mov rcx, qword ptr [rbx]
  mov rax, qword ptr [r12 + rdi*8]
  test rcx, rcx
  jnz .Lskip_op
  mov rdi, rax
  jmp .Lloop
.Lskip_op:
  inc rdi
  jmp .Lloop
.Ljmp:
  mov rax, qword ptr [r12 + rdi*8]
  mov rdi, rax
  jmp .Lloop
.Ldone:
  mov rcx, rbx
  sub rcx, r14
  shr rcx, 3
  mov r8, rcx
  mov rdx, rcx
.zero_loop:
  cmp rdx, r15
  jge .Lresult
  mov qword ptr [r14 + rdx*8], 0
  inc rdx
  jmp .zero_loop
.Lresult:
  test r8, r8
  jz .Lreturn_zero
  mov rax, qword ptr [r14 + r8*8 - 8]
  jmp .Lrestore
.Lreturn_zero:
  xor eax, eax
.Lrestore:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
