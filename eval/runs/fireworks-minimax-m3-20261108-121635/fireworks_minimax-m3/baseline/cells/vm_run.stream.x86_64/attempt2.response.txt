vm_run:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  xor edi, edi
  mov rbx, r14
  jmp .Lcheck
.Lloop:
  mov rax, qword ptr [r12 + rdi*8]
  inc rdi
  cmp rax, 0
  je .Lhalt
  cmp rax, 1
  je .Lpush
  cmp rax, 2
  je .Ladd
  cmp rax, 3
  je .Lsub
  cmp rax, 4
  je .Lmul
  cmp rax, 5
  je .Ldup
  cmp rax, 6
  je .Lswap
  cmp rax, 7
  je .Ldrop
  cmp rax, 8
  je .Lover
  cmp rax, 9
  je .Llt
  cmp rax, 10
  je .Ljz
  cmp rax, 11
  je .Ljmp
  jmp .Lcheck
.Lhalt:
  jmp .Ldone
.Lpush:
  mov rax, qword ptr [r12 + rdi*8]
  inc rdi
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lcheck
.Ladd:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  add rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lcheck
.Lsub:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  sub rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lcheck
.Lmul:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  imul rax, qword ptr [rbx - 8]
  mov qword ptr [rbx - 8], rax
  jmp .Lcheck
.Ldup:
  mov rax, qword ptr [rbx - 8]
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lcheck
.Lswap:
  mov rax, qword ptr [rbx - 8]
  mov rcx, qword ptr [rbx - 16]
  mov qword ptr [rbx - 8], rcx
  mov qword ptr [rbx - 16], rax
  jmp .Lcheck
.Ldrop:
  sub rbx, 8
  jmp .Lcheck
.Lover:
  mov rax, qword ptr [rbx - 16]
  mov qword ptr [rbx], rax
  add rbx, 8
  jmp .Lcheck
.Llt:
  sub rbx, 8
  mov rax, qword ptr [rbx]
  cmp qword ptr [rbx - 8], rax
  mov rax, 0
  setl al
  mov qword ptr [rbx - 8], rax
  jmp .Lcheck
.Ljz:
  sub rbx, 8
  mov rcx, qword ptr [rbx]
  mov rax, qword ptr [r12 + rdi*8]
  test rcx, rcx
  jnz .Lskip_op
  mov rdi, rax
  jmp .Lcheck
.Lskip_op:
  inc rdi
  jmp .Lcheck
.Ljmp:
  mov rax, qword ptr [r12 + rdi*8]
  mov rdi, rax
.Lcheck:
  cmp rdi, r13
  jl .Lloop
.Ldone:
  mov rcx, rbx
  sub rcx, r14
  shr rcx, 3
  mov rdx, rcx
  mov rax, rdx
.zero_loop:
  cmp rdx, r15
  jge .Lresult
  mov qword ptr [r14 + rdx*8], 0
  inc rdx
  jmp .zero_loop
.Lresult:
  test rax, rax
  jz .Lreturn_zero
  mov rax, qword ptr [r14 + rax*8 - 8]
  jmp .Lrestore
.Lreturn_zero:
  xor eax, eax
.Lrestore:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
