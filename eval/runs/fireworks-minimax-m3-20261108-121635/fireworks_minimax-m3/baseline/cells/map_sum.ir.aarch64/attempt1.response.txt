export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%s =l phi @start 0, @body %snext
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul 8, %i
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%r =l call %f(l %v)
	%snext =l add %s, %r
	%inext =l add %i, 1
	jmp @loop
@end
	ret %s
}
