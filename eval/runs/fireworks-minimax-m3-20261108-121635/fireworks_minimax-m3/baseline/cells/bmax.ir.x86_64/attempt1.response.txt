export function l $bmax(l %a, l %n) {
@start
    %max =l loadl %a
    %i =l copy 1
    %c0 =w csltl %i, %n
    jnz %c0, @loop, @done

@loop
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %gt =w csltl %max, %val
    jnz %gt, @update, @next

@update
    %max =l copy %val
    jmp @next

@next
    %i =l add %i, 1
    %c1 =w csltl %i, %n
    jnz %c1, @loop, @done

@done
    ret %max
}
