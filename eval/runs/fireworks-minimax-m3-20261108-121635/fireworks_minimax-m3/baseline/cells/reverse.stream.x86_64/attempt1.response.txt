reverse:
  test rsi, rsi
  jle .Ldone
  xor ecx, ecx
  mov rdx, rsi
  dec rdx
.Lloop:
  cmp rcx, rdx
  jge .Ldone
  mov rax, qword ptr [rdi + rcx*8]
  mov r8, qword ptr [rdi + rdx*8]
  mov qword ptr [rdi + rcx*8], r8
  mov qword ptr [rdi + rdx*8], rax
  inc rcx
  dec rdx
  jmp .Lloop
.Ldone:
  xor eax, eax
  ret
