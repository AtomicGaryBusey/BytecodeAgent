export function l $crc32(l %p, l %n) {
@start
    %crc =w copy 0xffffffff
    %i =l copy 0
    jmp @loop_test

@loop_test
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @loop_end

@loop_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc_in =w xor %crc, %byte
    %b0 =w and %crc_in, 1
    %m0 =w sub 0, %b0
    %p0 =w and 0xedb88320, %m0
    %s0 =w shr %crc_in, 1
    %crc0 =w xor %s0, %p0
    %b1 =w and %crc0, 1
    %m1 =w sub 0, %b1
    %p1 =w and 0xedb88320, %m1
    %s1 =w shr %crc0, 1
    %crc1 =w xor %s1, %p1
    %b2 =w and %crc1, 1
    %m2 =w sub 0, %b2
    %p2 =w and 0xedb88320, %m2
    %s2 =w shr %crc1, 1
    %crc2 =w xor %s2, %p2
    %b3 =w and %crc2, 1
    %m3 =w sub 0, %b3
    %p3 =w and 0xedb88320, %m3
    %s3 =w shr %crc2, 1
    %crc3 =w xor %s3, %p3
    %b4 =w and %crc3, 1
    %m4 =w sub 0, %b4
    %p4 =w and 0xedb88320, %m4
    %s4 =w shr %crc3, 1
    %crc4 =w xor %s4, %p4
    %b5 =w and %crc4, 1
    %m5 =w sub 0, %b5
    %p5 =w and 0xedb88320, %m5
    %s5 =w shr %crc4, 1
    %crc5 =w xor %s5, %p5
    %b6 =w and %crc5, 1
    %m6 =w sub 0, %b6
    %p6 =w and 0xedb88320, %m6
    %s6 =w shr %crc5, 1
    %crc6 =w xor %s6, %p6
    %b7 =w and %crc6, 1
    %m7 =w sub 0, %b7
    %p7 =w and 0xedb88320, %m7
    %s7 =w shr %crc6, 1
    %crc7 =w xor %s7, %p7
    %crc =w copy %crc7
    %i =l add %i, 1
    jmp @loop_test

@loop_end
    %crc_inv =w xor %crc, 0xffffffff
    %ret =l extuw %crc_inv
    ret %ret
}
