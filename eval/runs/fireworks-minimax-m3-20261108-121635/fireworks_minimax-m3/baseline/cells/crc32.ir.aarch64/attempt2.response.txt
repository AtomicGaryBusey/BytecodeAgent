export function l $crc32(l %p, l %n) {
@start
    %crc =w sub 0, 1
    %poly =w sub 0, 0x12477ce0
    %i =l copy 0
    jmp @outer_test

@outer_test
    %cond =w csltl %i, %n
    jnz %cond, @outer_body, @done

@outer_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %c =w xor %crc, %byte
    %j =l copy 0
    jmp @inner_test

@inner_test
    %jc =w csltl %j, 8
    jnz %jc, @inner_body, @inner_done

@inner_body
    %b =w and %c, 1
    %m =w sub 0, %b
    %p =w and %poly, %m
    %s =w shr %c, 1
    %c =w xor %s, %p
    %j =l add %j, 1
    jmp @inner_test

@inner_done
    %crc =w copy %c
    %i =l add %i, 1
    jmp @outer_test

@done
    %ones =w sub 0, 1
    %fin =w xor %crc, %ones
    %ret =l extuw %fin
    ret %ret
}
