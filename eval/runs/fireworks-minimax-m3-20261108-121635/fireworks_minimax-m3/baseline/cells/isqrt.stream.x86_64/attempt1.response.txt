isqrt:
    push rbx
    test rdi, rdi
    jz .Lzero
    mov rbx, 0x100000000
.Lstart:
    mov rax, rdi
    xor edx, edx
    div rbx
    add rax, rbx
    shr rax, 1
    cmp rax, rbx
    jae .Ldone
    mov rbx, rax
    jmp .Lstart
.Ldone:
    mov rax, rbx
    pop rbx
    ret
.Lzero:
    xor eax, eax
    pop rbx
    ret
