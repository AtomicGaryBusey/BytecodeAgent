dot_i32_simd:
    test rdx, rdx
    jz .Ldone
    mov rcx, rdx
    shr rcx, 2
    and rdx, 3
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rcx
    pxor xmm6, xmm6
    pxor xmm7, xmm7
    test r14, r14
    jz .Lskip_main
.Lmain_loop:
    movdqu xmm0, [r12]
    movdqu xmm1, [r13]
    pmovsxdq xmm2, xmm0
    psrldq xmm0, 8
    pmovsxdq xmm3, xmm0
    pmovsxdq xmm4, xmm1
    psrldq xmm1, 8
    pmovsxdq xmm5, xmm1
    pmuldq xmm2, xmm4
    pmuldq xmm3, xmm5
    paddq xmm6, xmm2
    paddq xmm7, xmm3
    add r12, 16
    add r13, 16
    dec r14
    jnz .Lmain_loop
.Lskip_main:
    movdqa xmm0, xmm6
    psrldq xmm0, 8
    paddq xmm6, xmm0
    movdqa xmm1, xmm7
    psrldq xmm1, 8
    paddq xmm7, xmm1
    paddq xmm6, xmm7
    movq rax, xmm6
    test rdx, rdx
    jz .Lfinish
.Lscalar:
    movsxd r8, dword ptr [r12]
    movsxd r9, dword ptr [r13]
    imul r8, r9
    add rax, r8
    add r12, 4
    add r13, 4
    dec rdx
    jnz .Lscalar
.Lfinish:
    pop r15
    pop r14
    pop r13
    pop r12
    ret
.Ldone:
    xor rax, rax
    ret
