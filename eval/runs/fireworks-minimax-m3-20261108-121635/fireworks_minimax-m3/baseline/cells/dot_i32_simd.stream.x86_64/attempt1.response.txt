dot_i32_simd:
    pxor xmm6, xmm6
    xor rax, rax
    test rdx, rdx
    jz .Lhsum
    mov r8, rdx
    shr r8, 2
    jz .Lrem
.Lvloop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    pshufd xmm4, xmm0, 0x0E
    pshufd xmm5, xmm1, 0x0E
    pmovsxdq xmm4, xmm4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm2, xmm4
    paddq xmm6, xmm2
    add rdi, 16
    add rsi, 16
    dec r8
    jnz .Lvloop
.Lrem:
    mov r9, rdx
    and r9, 3
    jz .Lhsum
.Lsloop:
    movsxd rcx, dword ptr [rdi]
    movsxd r10, dword ptr [rsi]
    imul rcx, r10
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec r9
    jnz .Lsloop
.Lhsum:
    psrldq xmm0, xmm6, 8
