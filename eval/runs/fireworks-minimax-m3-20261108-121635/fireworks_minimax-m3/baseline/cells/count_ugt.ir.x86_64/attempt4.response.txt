@start
	jmp @loop
@loop

@body
	%off = mul %i, 8
	%addr = add %a, %off
	%val = loadl %addr
	%gt = cugtl %val, %t
	jnz %gt, @true, @merge
@true
	%cnt_true = add %cnt, 1
	jmp @merge
@merge

	jmp @loop
@done
	ret %cnt
}
