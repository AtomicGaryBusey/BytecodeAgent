export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %n =l add %na, %nb
    %v0 =l alloc8
    %v1 =l alloc8
    %v2 =l alloc8
    %v3 =l alloc8
    %v4 =l alloc8
    %v5 =l alloc8
    storel 0, %v0
    jmp @zeroloop
@zeroloop
    %iz =l loadl %v0
    %c =w csltl %iz, %n
    jnz %c, @zerowrite, @zerodone
@zerowrite
    %off =l shl %iz, 3
    %p =l add %r, %off
    storel 0, %p
    %iz1 =l add %iz, 1
    storel %iz1, %v0
    jmp @zeroloop
@zerodone
    %na0 =w ceql %na, 0
    jnz %na0, @retzero, @checknb
@checknb
    %nbc =w ceql %nb, 0
    jnz %nbc, @retzero, @outerinit
@retzero
    ret 0
@outerinit
    storel 0, %v1
    jmp @outerloop
@outerloop
    %io =l loadl %v1
    %c1 =w csltl %io, %na
    jnz %c1, @outerbody, @outerdone
@outerbody
    storel 0, %v2
    jmp @innerloop
@innerloop
    %j =l loadl %v2
    %c2 =w csltl %j, %nb
    jnz %c2, @innerbody, @innerdone
@innerbody
    %aoff =l shl %io, 3
    %ap =l add %a, %aoff
    %av =l loadl %ap
    %boff =l shl %j, 3
    %bp =l add %b, %boff
    %bv =l loadl %bp
    %a0 =l and %av, 4294967295
    %a1 =l shr %av, 32
    %b0 =l and %bv, 4294967295
    %b1 =l shr %bv, 32
    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1
    %s12 =l add %p1, %p2
    %s12c =w cultl %s12, %p1
    %s12h =l shr %s12, 32
    %s12l =l and %s12, 4294967295
    %s12ls =l shl %s12l, 32
    %lo =l add %p0, %s12ls
    %loc =w cultl %lo, %p0
    %h1 =l add %p3, %s12h
    %h1c =w cultl %h1, %p3
    %h2 =l add %h1, %s12c
    %h2c =w cultl %h2, %h1
    %h3 =l add %h2, %loc
    %h3c =w cultl %h3, %h2
    %hor1 =w or %h1c, %h2c
    %hic =w or %hor1, %h3c
    %pos =l add %io, %j
    %rp1off =l shl %pos, 3
    %rp1p =l add %r, %rp1off
    %rp1v =l loadl %rp1p
    %newlo =l add %rp1v, %lo
    %nloc =w cultl %newlo, %rp1v
    storel %newlo, %rp1p
    %rp2off =l add %rp1off, 8
    %rp2p =l add %r, %rp2off
    %rp2v =l loadl %rp2p
    %t1 =l add %rp2v, %h3
    %t1c =w cultl %t1, %rp2v
    %t2 =l add %t1, %nloc
    %t2c =w cultl %t2, %t1
    %tor =w or %t1c, %t2c
    %totc =l add %tor, %hic
    storel %t2, %rp2p
    %pos2 =l add %pos, 2
    storel %pos2, %v3
    storel %totc, %v4
    jmp @proploop
@proploop
    %pc =l loadl %v4
    %pi =l loadl %v3
    %pcz =w ceql %pc, 0
    jnz %pcz, @propdone, @propchk
@propchk
    %pic =w csltl %pi, %n
    jnz %pic, @propbody, @propdone
@propbody
    %pioff =l shl %pi, 3
    %pip =l add %r, %pioff
    %piv =l loadl %pip
    %pinew =l add %piv, %pc
    %pinewc =w cultl %pinew, %piv
    storel %pinew, %pip
    %pinext =l add %pi, 1
    storel %pinext, %v3
    storel %pinewc, %v4
    jmp @proploop
@propdone
    %jnext =l add %j, 1
    storel %jnext, %v2
    jmp @innerloop
@innerdone
    %ionext =l add %io, 1
    storel %ionext, %v1
    jmp @outerloop
@outerdone
    storel %n, %v5
    jmp @scanloop
@scanloop
    %k =l loadl %v5
    %kz =w csltl 0, %k
    jnz %kz, @scanbody, @scandone
@scanbody
    %kk =l sub %k, 1
    %kkoff =l shl %kk, 3
    %kkp =l add %r, %kkoff
    %kkv =l loadl %kkp
    %kkz =w ceql %kkv, 0
    jnz %kkz, @scandec, @scandone
@scandec
    %k1 =l sub %k, 1
    storel %k1, %v5
    jmp @scanloop
@scandone
    %kr =l loadl %v5
    ret %kr
}
