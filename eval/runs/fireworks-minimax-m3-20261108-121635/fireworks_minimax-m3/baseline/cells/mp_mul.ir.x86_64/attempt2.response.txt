export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
    %zero =l copy 0
    %nsum =l add %na, %nb
    %k_alloc =l alloc8
    storel %zero, %k_alloc
    jmp @zero_start
@zero_start
    %k =l loadl %k_alloc
    %zcond =w csltl %k, %nsum
    jnz %zcond, @zero_body, @zero_end
@zero_body
    %koff =l shl %k, 3
    %kptr =l add %r, %koff
    storel %zero, %kptr
    %k1 =l add %k, 1
    storel %k1, %k_alloc
    jmp @zero_start
@zero_end
    %na0 =w ceql %na, %zero
    %nb0 =w ceql %nb, %zero
    %any0 =w or %na0, %nb0
    jnz %any0, @ret0, @main
@ret0
    ret %zero
@main
    %i_alloc =l alloc8
    storel %zero, %i_alloc
    jmp @outer_start
@outer_start
    %i =l loadl %i_alloc
    %icond =w csltl %i, %na
    jnz %icond, @outer_body, @outer_end
@outer_body
    %j_alloc =l alloc8
    storel %zero, %j_alloc
    jmp @inner_start
@inner_start
    %j =l loadl %j_alloc
    %jcond =w csltl %j, %nb
    jnz %jcond, @inner_body, @inner_end
@inner_body
    %aoff =l shl %i, 3
    %aptr =l add %a, %aoff
    %av =l loadl %aptr
    %boff =l shl %j, 3
    %bptr =l add %b, %boff
    %bv =l loadl %bptr
    %a0 =l and %av, 0xFFFFFFFF
    %a1 =l shr %av, 32
    %b0 =l and %bv, 0xFFFFFFFF
    %b1 =l shr %bv, 32
    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1
    %lo =l copy 0
    %hi =l copy 0
    %t1 =l add %lo, %p0
    %c1 =w cultl %t1, %lo
    %lo =l copy %t1
    %hi =l add %hi, %c1
    %p1_lo =l and %p1, 0xFFFFFFFF
    %p1_sh =l shl %p1_lo, 32
    %t2 =l add %lo, %p1_sh
    %c2 =w cultl %t2, %lo
    %lo =l copy %t2
    %hi =l add %hi, %c2
    %p1_hi =l shr %p1, 32
    %hi =l add %hi, %p1_hi
    %p2_lo =l and %p2, 0xFFFFFFFF
    %p2_sh =l shl %p2_lo, 32
    %t3 =l add %lo, %p2_sh
    %c3 =w cultl %t3, %lo
    %lo =l copy %t3
    %hi =l add %hi, %c3
    %p2_hi =l shr %p2, 32
    %hi =l add %hi, %p2_hi
    %hi =l add %hi, %p3
    %pos =l add %i, %j
    %posoff =l shl %pos, 3
    %rpos_ptr =l add %r, %posoff
    %rpos_val =l loadl %rpos_ptr
    %tlo =l add %rpos_val, %lo
    %clo =w cultl %tlo, %rpos_val
    storel %tlo, %rpos_ptr
    %carry =l copy %clo
    %pos1off =l add %posoff, 8
    %rpos1_ptr =l add %r, %pos1off
    %rpos1_val =l loadl %rpos1_ptr
    %thi1 =l add %rpos1_val, %hi
    %chi1 =w cultl %thi1, %rpos1_val
    %thi2 =l add %thi1, %carry
    %chi2 =w cultl %thi2, %thi1
    %carry_total =l copy %chi1
    %carry_total =l add %carry_total, %chi2
    storel %thi2, %rpos1_ptr
    %idx_alloc =l alloc8
    %idx =l add %pos, 2
    storel %idx, %idx_alloc
    %carr_alloc =l alloc8
    storel %carry_total, %carr_alloc
    jmp @prop_start
@prop_start
    %cval =l loadl %carr_alloc
    %cidx =l loadl %idx_alloc
    %czero =w ceql %cval, %zero
    jnz %czero, @prop_end, @prop_check
@prop_check
    %idxcond =w csltl %cidx, %nsum
    jnz %idxcond, @prop_body, @prop_end
@prop_body
    %ioff =l shl %cidx, 3
    %iptr =l add %r, %ioff
    %ival =l loadl %iptr
    %inew =l add %ival, %cval
    %inewc =w cultl %inew, %ival
    storel %inew, %iptr
    %inext =l add %cidx, 1
    storel %inext, %idx_alloc
    storel %inewc, %carr_alloc
    jmp @prop_start
@prop_end
    %j_next =l add %j, 1
    storel %j_next, %j_alloc
    jmp @inner_start
@inner_end
    %i_next =l add %i, 1
    storel %i_next, %i_alloc
    jmp @outer_start
@outer_end
    %k_ret =l copy %nsum
    jmp @scan_start
@scan_start
    %kpos =w csltl %zero, %k_ret
    jnz %kpos, @scan_body, @scan_end
@scan_body
    %kidx =l sub %k_ret, 1
    %koff2 =l shl %kidx, 3
    %kptr2 =l add %r, %koff2
    %kval =l loadl %kptr2
    %kzero =w ceql %kval, %zero
    jnz %kzero, @scan_dec, @scan_end
@scan_dec
    %k_ret =l sub %k_ret, 1
    jmp @scan_start
@scan_end
    ret %k_ret
}
