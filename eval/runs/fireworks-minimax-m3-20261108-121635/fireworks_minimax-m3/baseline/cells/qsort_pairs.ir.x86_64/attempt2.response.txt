export function l $qsort_pairs(l %a, l %n) {
@start
  %n_lt_2 =w csltl %n, 2
  jnz %n_lt_2, @ret0, @prep

@ret0
  ret 0

@prep
  %n1 =l sub %n, 1
  %m =l sar %n, 1
  %pivot_off =l mul %m, 16
  %pivot_addr =l add %a, %pivot_off
  %last_off =l mul %n1, 16
  %last_addr =l add %a, %last_off
  %k1 =l loadl %pivot_addr
  %pivot_addr1 =l add %pivot_addr, 8
  %v1 =l loadl %pivot_addr1
  %k2 =l loadl %last_addr
  %last_addr1 =l add %last_addr, 8
  %v2 =l loadl %last_addr1
  storel %k2, %pivot_addr
  storel %v2, %pivot_addr1
  storel %k1, %last_addr
  storel %v1, %last_addr1
  %pk =l loadl %last_addr
  %pv =l loadl %last_addr1
  %i_init =l copy -1
  %j_init =l copy 0
  jmp @loop_check

@loop_check
  %i =l phi @prep %i_init, @swap_inc %i_new, @loop_inc %i_pass
  %j =l phi @prep %j_init, @swap_inc %j_after_swap, @loop_inc %j_no_swap
  %j_lt_n1 =w csltl %j, %n1
  jnz %j_lt_n1, @loop_body, @final_swap

@loop_body
  %j_off =l mul %j, 16
  %j_addr =l add %a, %j_off
  %jk =l loadl %j_addr
  %j_addr1 =l add %j_addr, 8
  %jv =l loadl %j_addr1
  %lt_k =w csltl %jk, %pk
  %eq_k =w ceql %jk, %pk
  %lt_v =w csltl %jv, %pv
  %and_v =w and %eq_k, %lt_v
  %cond =w or %lt_k, %and_v
  jnz %cond, @swap_inc, @loop_inc

@swap_inc
  %i_new =l add %i, 1
  %i_new_off =l mul %i_new, 16
  %i_new_addr =l add %a, %i_new_off
  %i_new_addr1 =l add %i_new_addr, 8
  %ik =l loadl %i_new_addr
  %iv =l loadl %i_new_addr1
  storel %jk, %i_new_addr
  storel %jv, %i_new_addr1
  storel %ik, %j_addr
  storel %iv, %j_addr1
  %j_after_swap =l add %j, 1
  jmp @loop_check

@loop_inc
  %i_pass =l copy %i
  %j_no_swap =l add %j, 1
  jmp @loop_check

@final_swap
  %p =l add %i, 1
  %p_off =l mul %p, 16
  %p_addr =l add %a, %p_off
  %p_addr1 =l add %p_addr, 8
  %pk2 =l loadl %p_addr
  %pv2 =l loadl %p_addr1
  storel %pk, %p_addr
  storel %pv, %p_addr1
  storel %pk2, %last_addr
  storel %pv2, %last_addr1
  %left_n =l copy %p
  %right_start =l add %p, 1
  %right_off =l mul %right_start, 16
  %right_addr =l add %a, %right_off
  %right_n =l sub %n, %right_start
  %r0 =l call $qsort_pairs(l %a, l %left_n)
  %r1 =l call $qsort_pairs(l %right_addr, l %right_n)
  ret 0
}
