export function l $count_desc(l %a, l %n) {
@init
	%cnt =l alloc8 8
	%idx =l alloc8 8
	storel 0, %cnt
	storel 0, %idx
	%lim =l sub %n, 1

@loop
	%iv =l loadl %idx
	%c =w csltl %iv, %lim
	jnz %c, @body, @ret

@body
	%o0 =l mul %iv, 8
	%o1 =l add %o0, 8
	%p0 =l add %a, %o0
	%p1 =l add %a, %o1
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @inc, @next

@inc
	%cur =l loadl %cnt
	%new =l add %cur, 1
	storel %new, %cnt

@next
	%inew =l add %iv, 1
	storel %inew, %idx
	jmp @loop

@ret
	%r =l loadl %cnt
	ret %r
}
