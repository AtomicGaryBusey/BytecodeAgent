count_desc:
  test rsi, rsi
  jle .Lzero
  cmp rsi, 1
  je .Lzero
  mov r8, rsi
  dec r8
  mov r9, r8
  shr r9, 1
  shl r9, 1
  xor eax, eax
  test r9, r9
  jz .Lremainder
  xor ecx, ecx
.Lloop_sse:
  movdqu xmm0, [rdi + rcx*8]
  movdqu xmm1, [rdi + rcx*8 + 8]
  pcmpgtq xmm0, xmm1
  pmovmskb edx, xmm0
  popcnt edx, edx
  shr edx, 3
  add eax, edx
  add rcx, 2
  cmp rcx, r9
  jb .Lloop_sse
.Lremainder:
  test r8, 1
  jz .Ldone
  mov rdx, [rdi + r8*8 - 8]
  cmp rdx, [rdi + r8*8]
  jle .Ldone
  inc eax
.Ldone:
  ret
.Lzero:
  xor eax, eax
  ret
