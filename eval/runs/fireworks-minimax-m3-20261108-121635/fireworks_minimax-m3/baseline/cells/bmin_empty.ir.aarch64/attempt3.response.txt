export function l $bmin_empty(l %a, l %n) {
@start
%zero =l const 0
%one =l const 1
%eight =l const 8
%max =l const 0x7fffffffffffffff
%iszero =w ceql %n, %zero
jnz %iszero, @retmax, @init

@retmax
ret %max

@init
%min =l loadl 0, %a
%i =l copy %one
%ptr =l add %a, %eight
%bound =l sub %n, %one
jmp @loop

@loop
%done =w csgtl %i, %bound
jnz %done, @retmin, @body

@body
%val =l loadl 0, %ptr
%lt =w csltl %val, %min
jnz %lt, @upd, @inc

@upd
%min =l copy %val
jmp @inc

@inc
%i =l add %i, %one
%ptr =l add %ptr, %eight
jmp @loop

@retmin
ret %min
}
