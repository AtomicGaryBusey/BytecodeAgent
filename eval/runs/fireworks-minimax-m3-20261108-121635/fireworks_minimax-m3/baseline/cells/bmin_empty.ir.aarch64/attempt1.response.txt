export function l $bmin_empty(l %a, l %n) {
    %max =l const 0x7fffffffffffffff
    %is_zero =w ceql %n, 0
    jnz %is_zero, @return_max, @start

@start:
    %min =l loadl 0, %a
    %i =l const 1
    %ptr =l add %a, 8
    %bound =l sub %n, 1
    jmp @loop

@loop:
    %cond =w csgtl %i, %bound
    jnz %cond, @done, @body

@body:
    %val =l loadl 0, %ptr
    %cmp =w csltl %val, %min
    jnz %cmp, @update, @next

@update:
    %min =l copy %val
    jmp @next

@next:
    %i =l add %i, 1
    %ptr =l add %ptr, 8
    jmp @loop

@done:
    ret %min

@return_max:
    ret %max
}
