export function l $madd32(l %a, l %b, l %c) {
@start
	%a_high = shr %a, 32
	%a_msk = shl %a_high, 32
	%a_lo = sub %a, %a_msk
	%b_high = shr %b, 32
	%b_msk = shl %b_high, 32
	%b_lo = sub %b, %b_msk
	%c_high = shr %c, 32
	%c_msk = shl %c_high, 32
	%c_lo = sub %c, %c_msk
	%prod = mul %a_lo, %b_lo
	%sum = add %prod, %c_lo
	%s_high = shr %sum, 32
	%s_msk = shl %s_high, 32
	%res = sub %sum, %s_msk
	%shf = shl %res, 32
	%ret = sar %shf, 32
	ret %ret
}
