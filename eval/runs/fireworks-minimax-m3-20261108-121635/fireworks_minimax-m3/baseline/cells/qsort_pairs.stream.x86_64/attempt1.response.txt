qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
.loop:
    cmp r13, 1
    jle .return

    # pivot = last record
    mov rcx, r13
    dec rcx
    shl rcx, 4
    add rcx, r12
    mov r8, qword ptr [rcx]
    mov r9, qword ptr [rcx + 8]

    mov r14, -1          # i
    xor r15, r15         # j
    mov rcx, r13
    dec rcx              # limit = n-1

.partition_loop:
    cmp r15, rcx
    jge .end_partition_loop

    # load record j
    mov rdi, r15
    shl rdi, 4
    add rdi, r12
    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]

    # compare key, then value
    cmp r10, r8
    jl .swap_inc
    jne .next_j
    cmp r11, r9
    jle .swap_inc
    jmp .next_j

.swap_inc:
    inc r14
    cmp r14, r15
    je .next_j

    # swap records i and j
    mov rax, r14
    shl rax, 4
    add rax, r12

    movdqu xmm0, [rax]
    movdqu xmm1, [rdi]
    movdqu [rax], xmm1
    movdqu [rdi], xmm0

.next_j:
    inc r15
    jmp .partition_loop

.end_partition_loop:
    inc r14              # i+1 = pivot position
    # compute addresses for pivot swap
    mov rcx, r13
    dec rcx
    shl rcx, 4
    add rcx, r12         # pivot address
    mov rax, r14
    shl rax, 4
    add rax, r12         # i+1 address

    cmp rax, rcx
    je .skip_pivot_swap

    movdqu xmm2, [rax]
    movdqu xmm3, [rcx]
    movdqu [rax], xmm3
    movdqu [rcx], xmm2

.skip_pivot_swap:
    # right size = n - left_size - 1
    mov r15, r13
    sub r15, r14
    dec r15

    test r14, r14
    jz .skip_left_call

    # recurse on left part
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs

.skip_left_call:
    # loop on right part
    lea r12, [r12 + r14*16 + 16]
