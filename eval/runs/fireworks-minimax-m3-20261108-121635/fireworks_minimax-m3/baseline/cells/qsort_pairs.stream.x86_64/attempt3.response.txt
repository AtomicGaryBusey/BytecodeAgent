qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
.loop:
    cmp r13, 1
    jle .done
    lea rcx, [r13 - 1]
    shl rcx, 4
    add rcx, r12
    mov r8, [rcx]
    mov r9, [rcx + 8]
    mov r14, -1
    mov r15, 0
    lea rbx, [r13 - 1]
.ploop:
    cmp r15, rbx
    jge .ploop_end
    mov rdi, r15
    shl rdi, 4
    add rdi, r12
    mov r10, [rdi]
    mov r11, [rdi + 8]
    cmp r10, r8
    jl .swap
    jg .noswap
    cmp r11, r9
    jle .swap
.noswap:
    jmp .pinc
.swap:
    inc r14
    cmp r14, r15
    je .pinc
    mov rax, r14
    shl rax, 4
    add rax, r12
    movdqu xmm0, [rax]
    movdqu xmm1, [rdi]
    movdqu [rax], xmm1
    movdqu [rdi], xmm0
.pinc:
    inc r15
    jmp .ploop
.ploop_end:
    inc r14
    cmp r14, rbx
    je .pskip
    mov rax, r14
    shl rax, 4
    add rax, r12
    movdqu xmm0, [rax]
    movdqu xmm1, [rcx]
    movdqu [rax], xmm1
    movdqu [rcx], xmm0
.pskip:
    mov r15, r13
    sub r15, r14
    dec r15
    test r14, r14
    jz .nleft
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs
.nleft:
    lea rcx, [r14 + 1]
    shl rcx, 4
    add rcx, r12
    mov r12, rcx
    mov r13, r15
    jmp .loop
.done:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
