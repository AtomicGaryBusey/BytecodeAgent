qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
.loop:
    cmp r13, 1
    jle .return

    mov rcx, r13
    dec rcx
    shl rcx, 4
    add rcx, r12
    mov r8, qword ptr [rcx]
    mov r9, qword ptr [rcx + 8]

    mov r14, -1
    xor r15, r15
    mov rbx, r13
    dec rbx

.partition_loop:
    cmp r15, rbx
    jge .end_partition_loop

    mov rdi, r15
    shl rdi, 4
    add rdi, r12
    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]

    cmp r10, r8
    jl .do_swap
    jne .next_j
    cmp r11, r9
    jle .do_swap
    jmp .next_j

.do_swap:
    inc r14
    cmp r14, r15
    je .next_j
    mov rax, r14
    shl rax, 4
    add rax, r12
    movdqu xmm0, [rax]
    movdqu xmm1, [rdi]
    movdqu [rax], xmm1
    movdqu [rdi], xmm0

.next_j:
    inc r15
    jmp .partition_loop

.end_partition_loop:
    inc r14
    cmp r14, rbx
    je .skip_pivot_swap
    mov rax, r14
    shl rax, 4
    add rax, r12
    movdqu xmm2, [rax]
    movdqu xmm3, [rcx]
    movdqu [rax], xmm3
    movdqu [rcx], xmm2

.skip_pivot_swap:
    mov r15, r13
    sub r15, r14
    dec r15

    test r14, r14
    jz .no_left
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs

.no_left:
    mov rcx, r14
    shl rcx, 4
    add rcx, r12
    add rcx, 16
    mov r12, rcx
    mov r13, r15
    jmp .loop

.return:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
