export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@init
    %size_n =l mul %n, %n
    %size_tmp =l mul %size_n, 2
    %B_ptr =l add %tmp, %size_n
    jmp @zero_out_loop

@zero_out_loop
    %zi =l phi @init 0, @zero_out_body %zi_next
    %zcond =l csltl %zi, %size_n
    jnz %zcond, @zero_out_body, @zero_out_done

@zero_out_body
    %zaddr =l add %out, %zi
    storel 0, %zaddr
    %zi_next =l add %zi, 1
    jmp @zero_out_loop

@zero_out_done
    jmp @identity_loop

@identity_loop
    %di =l phi @zero_out_done 0, @diag_body %di_next
    %dcond =l csltl %di, %n
    jnz %dcond, @diag_body, @diag_done

@diag_body
    %didx =l mul %di, %n
    %didx =l add %didx, %di
    %daddr =l add %out, %didx
    storel 1, %daddr
    %di_next =l add %di, 1
    jmp @identity_loop

@diag_done
    jmp @k_check

@k_check
    %k_is_zero =l ceql %k, 0
    jnz %k_is_zero, @zero_tmp, @k_loop

@k_loop
    %iter =l phi @k_check 0, @post_multiply %iter_next
    %icond =l csltl %iter, %k
    jnz %icond, @copy_to_b, @zero_tmp

@copy_to_b
    %ci =l phi @k_loop 0, @copy_to_b_body %ci_next
    %cicond =l csltl %ci, %size_n
    jnz %cicond, @copy_to_b_body, @multiply

@copy_to_b_body
    %src =l add %out, %ci
    %val =l loadl %src
    %dst =l add %B_ptr, %ci
    storel %val, %dst
    %ci_next =l add %ci, 1
    jmp @copy_to_b

@multiply
    %mi =l phi @copy_to_b 0, @j_done %mi_next
    %micond =l csltl %mi, %n
    jnz %micond, @i_body, @i_done

@i_body
    %mj =l phi @multiply 0, @t_end %mj_next
    %mjcond =l csltl %mj, %n
    jnz %mjcond, @j_body, @j_done

@j_body
    %sum =l phi @i_body 0, @t_body %sum_new
    %mt =l phi @i_body 0, @t_body %mt_next
    %mtcond =l csltl %mt, %n
    jnz %mtcond, @t_body, @t_end

@t_body
    %idx_it =l mul %mi, %n
    %idx_it =l add %idx_it, %mt
    %B_it_addr =l add %B_ptr, %idx_it
    %B_it_val =l loadl %B_it_addr
    %idx_tj =l mul %mt, %n
    %idx_tj =l add %idx_tj, %mj
    %a_tj_addr =l add %a, %idx_tj
    %a_tj_val =l loadl %a_tj_addr
    %prod =l mul %B_it_val, %a_tj_val
    %sum_new =l add %sum, %prod
    %mt_next =l add %mt, 1
    jmp @j_body

@t_end
    %idx_ij =l mul %mi, %n
    %idx_ij =l add %idx_ij, %mj
    %out_ij_addr =l add %out, %idx_ij
    storel %sum, %out_ij_addr
    %mj_next =l add %mj, 1
    jmp @i_body

@j_done
    %mi_next =l add %mi, 1
    jmp @multiply

@i_done
    jmp @post_multiply

@post_multiply
    %iter_next =l add %iter, 1
    jmp @k_loop

@zero_tmp
    %zt =l phi @k_check 0, @k_loop 0, @zero_tmp_body %zt_next
    %ztcond =l csltl %zt, %size_tmp
    jnz %ztcond, @zero_tmp_body, @ret_block

@zero_tmp_body
    %ztaddr =l add %tmp, %zt
    storel 0, %ztaddr
    %zt_next =l add %zt, 1
    jmp @zero_tmp

@ret_block
    ret 0
}
