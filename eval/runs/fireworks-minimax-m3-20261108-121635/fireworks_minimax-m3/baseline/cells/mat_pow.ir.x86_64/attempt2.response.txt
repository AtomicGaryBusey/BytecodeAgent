export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %size_n =l mul %n, %n
    %size_2n =l mul %size_n, 2
    %B_ptr =l add %tmp, %size_n
    jmp @zero_out_loop

@zero_out_loop
    %zo =l phi @start 0, @zero_out_body %zo_next
    %zo_done =w csltl %zo, %size_n
    jnz %zo_done, @zero_out_body, @zero_out_done

@zero_out_body
    %zo_addr =l add %out, %zo
    storel 0, %zo_addr
    %zo_next =l add %zo, 1
    jmp @zero_out_loop

@zero_out_done
    jmp @ident_loop

@ident_loop
    %id =l phi @zero_out_done 0, @ident_body %id_next
    %id_done =w csltl %id, %n
    jnz %id_done, @ident_body, @ident_done

@ident_body
    %id_idx =l mul %id, %n
    %id_idx =l add %id_idx, %id
    %id_addr =l add %out, %id_idx
    storel 1, %id_addr
    %id_next =l add %id, 1
    jmp @ident_loop

@ident_done
    %k0 =w ceql %k, 0
    jnz %k0, @zero_tmp_loop, @iter_loop

@iter_loop
    %it =l phi @ident_done 0, @iter_done %it_next
    %it_cont =w csltl %it, %k
    jnz %it_cont, @copy_loop, @zero_tmp_loop

@copy_loop
    %cp =l phi @iter_loop 0, @copy_body %cp_next
    %cp_done =w csltl %cp, %size_n
    jnz %cp_done, @copy_body, @mult_i_loop

@copy_body
    %cp_src =l add %out, %cp
    %cp_val =l loadl %cp_src
    %cp_dst =l add %B_ptr, %cp
    storel %cp_val, %cp_dst
    %cp_next =l add %cp, 1
    jmp @copy_loop

@mult_i_loop
    %mi =l phi @copy_loop 0, @mult_i_done %mi_next
    %mi_cont =w csltl %mi, %n
    jnz %mi_cont, @mult_j_start, @iter_done

@mult_j_start
    %mj =l phi @mult_i_loop 0, @mult_j_done %mj_next
    jmp @mult_t_loop

@mult_t_loop
    %sum =l phi @mult_j_start 0, @mult_t_body %sum_new
    %mt =l phi @mult_j_start 0, @mult_t_body %mt_next
    %mt_cont =w csltl %mt, %n
    jnz %mt_cont, @mult_t_body, @mult_t_done

@mult_t_body
    %it_off =l mul %mi, %n
    %it_off =l add %it_off, %mt
    %B_it =l add %B_ptr, %it_off
    %Bi =l loadl %B_it
    %tj_off =l mul %mt, %n
    %tj_off =l add %tj_off, %mj
    %a_at =l add %a, %tj_off
    %aj =l loadl %a_at
    %prod =l mul %Bi, %aj
    %sum_new =l add %sum, %prod
    %mt_next =l add %mt, 1
    jmp @mult_t_loop

@mult_t_done
    %o_off =l mul %mi, %n
    %o_off =l add %o_off, %mj
    %out_ij =l add %out, %o_off
    storel %sum, %out_ij
    jmp @mult_j_done

@mult_j_done
    %mj_next =l add %mj, 1
    jmp @mult_j_start

@mult_i_done
    %mi_next =l add %mi, 1
    jmp @mult_i_loop

@iter_done
    %it_next =l add %it, 1
    jmp @iter_loop

@zero_tmp_loop
    %zt =l phi @ident_done 0, @iter_loop 0, @zero_tmp_body %zt_next
    %zt_cont =w csltl %zt, %size_2n
    jnz %zt_cont, @zero_tmp_body, @end

@zero_tmp_body
    %zt_addr =l add %tmp, %zt
    storel 0, %zt_addr
    %zt_next =l add %zt, 1
    jmp @zero_tmp_loop

@end
    ret 0
}
