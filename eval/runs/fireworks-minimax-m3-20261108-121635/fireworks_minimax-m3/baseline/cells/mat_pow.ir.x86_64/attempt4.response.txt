export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n2 =l mul %n, %n
    %sz =l mul %n2, 8
    %Bp =l add %tmp, %sz
    %tmp_sz =l mul %sz, 2
    jmp @zero_out

@zero_out
    %z =l phi @start 0, @zero_out_body %z_next
    %zlt =w cultl %z, %sz
    jnz %zlt, @zero_out_body, @ident_loop

@zero_out_body
    %zp =l add %out, %z
    storel 0, %zp
    %z_next =l add %z, 8
    jmp @zero_out

@ident_loop
    %id =l phi @zero_out 0, @ident_body %id_next
    %idlt =w cultl %id, %n
    jnz %idlt, @ident_body, @k_check

@ident_body
    %idn =l mul %id, %n
    %idx =l add %idn, %id
    %idxb =l mul %idx, 8
    %idp =l add %out, %idxb
    storel 1, %idp
    %id_next =l add %id, 1
    jmp @ident_loop

@k_check
    %k0 =w ceql %k, 0
    jnz %k0, @zero_tmp, @iter_loop

@iter_loop
    %it =l phi @k_check 0, @iter_done %it_next
    %itlt =w cultl %it, %k
    jnz %itlt, @copy_loop, @zero_tmp

@copy_loop
    %cp =l phi @iter_loop 0, @copy_body %cp_next
    %cplt =w cultl %cp, %sz
    jnz %cplt, @copy_body, @mul_i

@copy_body
    %cps =l add %out, %cp
    %cpv =l loadl %cps
    %cpd =l add %Bp, %cp
    storel %cpv, %cpd
    %cp_next =l add %cp, 8
    jmp @copy_loop

@mul_i
    %mi =l phi @copy_loop 0, @mul_i_done %mi_next
    %milt =w cultl %mi, %n
    jnz %milt, @mul_j, @iter_done

@mul_j
    %mj =l phi @mul_i 0, @mul_j_done %mj_next
    %mjlt =w cultl %mj, %n
    jnz %mjlt, @mul_t_init, @mul_i_done

@mul_t_init
    jmp @mul_t

@mul_t
    %sum =l phi @mul_t_init 0, @mul_t_body %sum_next
    %mt =l phi @mul_t_init 0, @mul_t_body %mt_next
    %mtlt =w cultl %mt, %n
    jnz %mtlt, @mul_t_body, @mul_t_end

@mul_t_body
    %min =l mul %mi, %n
    %mit =l add %min, %mt
    %bo1 =l mul %mit, 8
    %Ba =l add %Bp, %bo1
    %Bi =l loadl %Ba
    %mtn =l mul %mt, %n
    %mtj =l add %mtn, %mj
    %bo2 =l mul %mtj, 8
    %Aa =l add %a, %bo2
    %Aj =l loadl %Aa
    %prod =l mul %Bi, %Aj
    %sum_next =l add %sum, %prod
    %mt_next =l add %mt, 1
    jmp @mul_t

@mul_t_end
    %omin =l mul %mi, %n
    %omij =l add %omin, %mj
    %obo =l mul %omij, 8
    %op =l add %out, %obo
    storel %sum, %op
    jmp @mul_j_done

@mul_j_done
    %mj_next =l add %mj, 1
    jmp @mul_j

@mul_i_done
    %mi_next =l add %mi, 1
    jmp @mul_i

@iter_done
    %it_next =l add %it, 1
    jmp @iter_loop

@zero_tmp
    %zt =l phi @k_check 0, @iter_loop 0, @zero_tmp_body %zt_next
    %ztlt =w cultl %zt, %tmp_sz
    jnz %ztlt, @zero_tmp_body, @end

@zero_tmp_body
    %ztp =l add %tmp, %zt
    storel 0, %ztp
    %zt_next =l add %zt, 8
    jmp @zero_tmp

@end
    ret 0
}
