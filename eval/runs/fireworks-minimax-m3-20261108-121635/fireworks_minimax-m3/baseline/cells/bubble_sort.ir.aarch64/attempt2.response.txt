@check_i

jnz %cond_i, @outer_body, @ret_zero

@outer_body

%j = copy 0

@check_j

@inner_body

%v1 = loadl %p1
%p2 = add %p1, 8

%cond_swap = csltl %v2, %v1

@step_j

@step_i

}
