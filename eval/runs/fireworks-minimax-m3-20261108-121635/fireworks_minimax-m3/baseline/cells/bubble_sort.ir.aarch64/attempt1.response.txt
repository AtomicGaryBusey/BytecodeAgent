export function l $bubble_sort(l %a, l %n) {
	%n1 = sub %n, 1
	%i = copy 0
	jmp @check_outer

@check_outer
	%c1 = csltl %i, %n1
	jnz %c1, @outer_body, @ret_zero

@outer_body
	%bound = sub %n1, %i
	%j = copy 0
	jmp @check_inner

@check_inner
	%c2 = csltl %j, %bound
	jnz %c2, @inner_body, @outer_inc

@inner_body
	%idx1 = mul %j, 8
	%ptr1 = add %a, %idx1
	%val1 = loadl %ptr1
	%ptr2 = add %ptr1, 8
	%val2 = loadl %ptr2
	%c3 = csltl %val2, %val1
	jnz %c3, @do_swap, @inner_inc

@do_swap
	storel %val2, %ptr1
	storel %val1, %ptr2
	jmp @inner_inc

@inner_inc
	%j = add %j, 1
	jmp @check_inner

@outer_inc
	%i = add %i, 1
	jmp @check_outer

@ret_zero
	ret 0
}
