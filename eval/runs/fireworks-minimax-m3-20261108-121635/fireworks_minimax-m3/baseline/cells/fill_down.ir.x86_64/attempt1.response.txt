export function l $fill_down(l %a, l %n, l %start) {
@start
	%z = ceql %n, 0
	%init = copy 0
	jnz %z, @end, @loop
@loop
	%i = phi l %init, %next
	%c = csltl %i, %n
	jnz %c, @body, @end
@body
	%off = mul %i, 8
	%p = add %a, %off
	%v = sub %start, %i
	storel %v, %p
	%next = add %i, 1
	jmp @loop
@end
	ret %n
}
