export function l $isqrt(l %n) {
@start
	%r0 =l copy 0
	%bit0 =l copy 31
	jmp @loop
@loop
	%r =l phi @start %r0, @step %r_step
	%bit =l phi @start %bit0, @step %bit_step
	%mask =l shl 1, %bit
	%r2 =l or %r, %mask
	%prod =l mul %r2, %r2
	%cond =w culel %prod, %n
	jnz %cond, @set, @noset
@set
	%r_set =l copy %r2
	%bit_set =l sub %bit, 1
	jmp @step
@noset
	%r_noset =l copy %r
	%bit_noset =l sub %bit, 1
	jmp @step
@step
	%r_step =l phi @set %r_set, @noset %r_noset
	%bit_step =l phi @set %bit_set, @noset %bit_noset
	%stop =w csltl %bit_step, 0
	jnz %stop, @exit, @loop
@exit
	ret %r_step
}
