export function l $isqrt(l %n) {
@start
	%r =l copy 0
	%bit =w copy 31
	jmp @loop
@loop
	%r =l phi @start %r, @step %r_step
	%bit =w phi @start %bit, @step %bit_step
	%mask =l shl 1, %bit
	%r2 =l or %r, %mask
	%prod =l mul %r2, %r2
	%cond =w culel %prod, %n
	jnz %cond, @set, @noset
@set
	%r_next =l copy %r2
	%bit_next =w sub %bit, 1
	jmp @step
@noset
	%r_next =l copy %r
	%bit_next =w sub %bit, 1
	jmp @step
@step
	%r_step =l phi @set %r_next, @noset %r_next
	%bit_step =w phi @set %bit_next, @noset %bit_next
	%exit =w csltw %bit_step, 0
	jnz %exit, @exit, @loop
@exit
	ret %r_step
}
