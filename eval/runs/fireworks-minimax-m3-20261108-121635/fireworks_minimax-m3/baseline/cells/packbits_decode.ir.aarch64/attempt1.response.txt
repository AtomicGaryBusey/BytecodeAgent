export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%s =l copy %src
	%o =l copy %out
	%rem_s =l copy %n
	%rem_o =l copy %cap

@jloop
	%more =w csltl $0, %rem_s
	jnz %more, @body, @done_succ

@body
	%h_u =w loadub %s
	%is_nop =w ceqw %h_u, 128
	jnz %is_nop, @do_nop, @check_lit

@check_lit
	%is_lit =w csltw %h_u, 128
	jnz %is_lit, @do_lit, @do_rep

@do_nop
	%s =l add %s, 1
	%rem_s =l sub %rem_s, 1
	jmp @jloop

@do_lit
	%need =l addl %h_u, 2
	%src_short =w csltl %rem_s, %need
	jnz %src_short, @err_src, @lit_cap

@lit_cap
	%cnt =l addl %h_u, 1
	%rem_short =w csltl %rem_o, %cnt
	jnz %rem_short, @err_cap_lit, @lit_copy

@lit_copy
	%i =l copy 0
	%sp =l add %s, 1
	%dp =l copy %o
@lit_loop
	%done =w csgel %i, %cnt
	jnz %done, @lit_done, @lit_body

@lit_body
	%b =w loadub %sp
	storeb %b, %dp
	%sp =l add %sp, 1
	%dp =l add %dp, 1
	%i =l add %i, 1
	jmp @lit_loop

@lit_done
	%s =l add %s, %need
	%o =l add %o, %cnt
	%rem_s =l sub %rem_s, %need
	%rem_o =l sub %rem_o, %cnt
	jmp @jloop

@do_rep
	%need =l copy 2
	%src_short =w csltl %rem_s, %need
	jnz %src_short, @err_src, @rep_cap

@rep_cap
	%h_s =w loadsb %s
	%cnt =l subl $1, %h_s
	%rem_short =w csltl %rem_o, %cnt
	jnz %rem_short, @err_cap_rep, @rep_copy

@rep_copy
	%sp =l add %s, 1
	%b =w loadub %sp
	%i =l copy 0
	%dp =l copy %o
@rep_loop
	%done =w csgel %i, %cnt
	jnz %done, @rep_done, @rep_body

@rep_body
	storeb %b, %dp
	%dp =l add %dp, 1
	%i =l add %i, 1
	jmp @rep_loop

@rep_done
	%s =l add %s, 2
	%o =l add %o, %cnt
	%rem_s =l sub %rem_s, 2
	%rem_o =l sub %rem_o, %cnt
	jmp @jloop

@err_cap_lit
	%i =l copy 0
	%sp =l add %s, 1
	%dp =l copy %o
@ecl_loop
	%done =w csgel %i, %rem_o
	jnz %done, @err_cap_ret, @ecl_body

@ecl_body
	%b =w loadub %sp
	storeb %b, %dp
	%sp =l add %sp, 1
	%dp =l add %dp, 1
	%i =l add %i, 1
	jmp @ecl_loop

@err_cap_rep
	%sp =l add %s, 1
	%b =w loadub %sp
	%i =l copy 0
	%dp =l copy %o
@ecr_loop
	%done =w csgel %i, %rem_o
	jnz %done, @err_cap_ret, @ecr_body

@ecr_body
	storeb %b, %dp
	%dp =l add %dp, 1
	%i =l add %i, 1
	jmp @ecr_loop

@err_cap_ret
	%ret =l copy -1
	jmp @final

@err_src
	%ret =l copy -2
	jmp @final

@done_succ
	%ret =l sub %o, %out
	jmp @final

@final
	ret %ret
}
