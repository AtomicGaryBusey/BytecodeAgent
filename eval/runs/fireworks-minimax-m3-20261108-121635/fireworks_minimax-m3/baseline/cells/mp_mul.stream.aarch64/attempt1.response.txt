mp_mul:
    mov x9, x0           // a_ptr
    mov x10, x1          // na
    mov x11, x2          // b_ptr
    mov x12, x3          // nb
    mov x13, x4          // r_ptr
    cbz x10, .Lzero_case // if na == 0
    cbz x12, .Lzero_case // if nb == 0
    add x14, x10, x12    // total_limbs = na + nb
    mov x15, x14         // counter for zeroing
    mov x16, #0
.Lzero_loop:
    cbz x15, .Lzero_done
    sub x15, x15, #1
    str x16, [x13, x15, lsl #3]
    b .Lzero_loop
.Lzero_done:
    mov x15, #0          // i = 0
.Louter_loop:
    cmp x15, x10
    b.hs .Louter_done    // if i >= na
    mov x16, #0          // j = 0
.Linner_loop:
    cmp x16, x12
    b.hs .Linner_done    // if j >= nb
    ldr x19, [x9, x15, lsl #3]   // a[i]
    ldr x20, [x11, x16, lsl #3]  // b[j]
    mul x21, x19, x20             // lo = a[i] * b[j]
    umulh x22, x19, x20           // hi = (a[i] * b[j]) >> 64
    add x25, x15, x16             // idx = i + j
    add x17, x13, x25, lsl #3     // &r[idx]
    ldr x23, [x17]                // r[idx]
    adds x23, x23, x21            // r[idx] + lo
    str x23, [x17]                // store back
    cset x24, cs                  // carry from lo addition
    add x17, x17, #8              // &r[idx+1]
    ldr x23, [x17]                // r[idx+1]
    add x23, x23, x22             // r[idx+1] + hi
    adds x23, x23, x24            // + carry
    str x23, [x17]                // store back
    cset x24, cs                  // carry to propagate
    add x25, x25, #2              // k = idx+2
.Lpropagate:
    cbz x24, .Lprop_done           // if no carry, done
    cmp x25, x14
    b.hs .Lprop_done              // if k >= total, done (should not happen)
    ldr x23, [x13, x25, lsl #3]   // r[k]
    adds x23, x23, #1             // r[k] + carry
    str x23, [x13, x25, lsl #3]   // store back
    cset x24, cs                  // next carry
    add x25, x25, #1
    b .Lpropagate
.Lprop_done:
    add x16, x16, #1              // j++
    b .Linner_loop
.Linner_done:
    add x15, x15, #1              // i++
    b .Louter_loop
.Louter_done:
    mov x15, x14                  // start from total_limbs
.Lscan_loop:
    sub x15, x15, #1              // index = total_limbs - 1
    ldr x23, [x13, x15, lsl #3]   // r[index]
    cbnz x23, .Lfound_nonzero     // if non-zero, found
    cbz x15, .Lall_zero           // if index == 0 and still zero, all zero
    b .Lscan_loop
.Lfound_nonzero:
    add x0, x15, #1               // return index + 1
    ret
.Lall_zero:
    mov x0, #0
    ret
.Lzero_case:
    add x14, x10, x12             // total = na + nb
    mov x15, x14
    mov x16, #0
.Lzero_case_loop:
    cbz x15, .Lzero_case_done
    sub x15, x15, #1
    str x16, [x13, x15, lsl #3]
    b .Lzero_case_loop
.Lzero_case_done:
    mov x0, #0
    ret
