export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i =l alloc 8
	storel 0, %i
	%min =l alloc 8
	%max =l alloc 8
	%sum =l alloc 8
	%cnt =l alloc 8
	jmp @loop
@loop
	%idx =l loadl %i
	%cnd =w csltl %idx, %n
	jnz %cnd, @body, @done
@body
	%off =l mul %idx, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%y =l call %f(l %elem)
	%first =w ceql %idx, 0
	jnz %first, @init, @combine
@init
	storel %y, %min
	storel %y, %max
	storel 0, %sum
	storel 0, %cnt
	jmp @odd
@combine
	%curm =l loadl %min
	%lt =w csltl %y, %curm
	jnz %lt, @setmin, @chkmax
@setmin
	storel %y, %min
	jmp @chkmax
@chkmax
	%curx =l loadl %max
	%gt =w csgtl %y, %curx
	jnz %gt, @setmax, @sumit
@setmax
	storel %y, %max
	jmp @sumit
@sumit
	%curs =l loadl %sum
	%ns =l add %curs, %y
	storel %ns, %sum
	jmp @odd
@odd
	%bit =l and %y, 1
	%isodd =w ceql %bit, 1
	jnz %isodd, @inc, @next
@inc
	%curc =l loadl %cnt
	%nc =l add %curc, 1
	storel %nc, %cnt
	jmp @next
@next
	%curi =l loadl %i
	%ni =l add %curi, 1
	storel %ni, %i
	jmp @loop
@done
	%fm =l loadl %min
	%fx =l loadl %max
	%fs =l loadl %sum
	%fc =l loadl %cnt
	storel %fm, %out
	%p1 =l add %out, 8
	storel %fx, %p1
	%p2 =l add %out, 16
	storel %fs, %p2
	ret %fc
}
