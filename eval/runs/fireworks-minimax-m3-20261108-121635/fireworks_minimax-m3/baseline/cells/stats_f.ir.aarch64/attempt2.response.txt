export function l $stats_f(l %a, l %n, l %f, l %out) {
	%isz =w ceql %n, 1
	jnz %isz, @first, @rest
@first
	%elem =l loadl %a
	%y =l call %f(l %elem)
	storel %y, %out
	%p1 =l add %out, 8
	storel %y, %p1
	%p2 =l add %out, 16
	storel %y, %p2
	%bit =l and %y, 1
	ret %bit
@rest
	%nm1 =l sub %n, 1
	%cntn1 =l call $stats_f(l %a, %nm1, l %f, l %out)
	%off =l mul %nm1, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%y =l call %f(l %elem)
	%curm =l loadl %out
	%lt =w csltl %y, %curm
	jnz %lt, @setmin, @keepmin
@setmin
	storel %y, %out
	jmp @keepmin
@keepmin
	%p1 =l add %out, 8
	%curx =l loadl %p1
	%gt =w csgtl %y, %curx
	jnz %gt, @setmax, @keepmax
@setmax
	storel %y, %p1
	jmp @keepmax
@keepmax
	%p2 =l add %out, 16
	%curs =l loadl %p2
	%ns =l add %curs, %y
	storel %ns, %p2
	%bit =l and %y, 1
	%total =l add %cntn1, %bit
	ret %total
}
