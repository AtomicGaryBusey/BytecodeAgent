export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%isz =w ceql %n, 1
	jnz %isz, @first, @rest
@first
	%elem =l loadl %a
	%y =l call %f(l %elem)
	storel %y, %out
	%p1 =l add %out, 8
	storel %y, %p1
	%p2 =l add %out, 16
	storel %y, %p2
	%bit =l and %y, 1
	ret %bit
@rest
	%nm1 =l sub %n, 1
	%cntn1 =l call $stats_f(l %a, %nm1, l %f, l %out)
	%off =l mul %nm1, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%min =l loadl %out
	%ld1 =l add %out, 8
	%max =l loadl %ld1
	%ld2 =l add %out, 16
	%sum =l loadl %ld2
	%y =l call %f(l %elem)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @checkmax
@setmin
	storel %y, %out
	jmp @checkmax
@checkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @updatesum
@setmax
	%st1 =l add %out, 8
	storel %y, %st1
	jmp @updatesum
@updatesum
	%st2 =l add %out, 16
	%ns =l add %sum, %y
	storel %ns, %st2
	%bit =l and %y, 1
	%total =l add %cntn1, %bit
	ret %total
}
