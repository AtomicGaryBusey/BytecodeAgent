export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @back %nexti
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%iszero =w ceql %val, 0
	jnz %iszero, @end, @back
@back
	%nexti =l add %i, 1
	jmp @loop
@end
	ret %i
}
