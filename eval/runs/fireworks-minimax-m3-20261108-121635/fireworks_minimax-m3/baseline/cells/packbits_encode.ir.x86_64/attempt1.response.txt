export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
%written =l copy 0
%i =l copy 0
jmp @check_outer

@check_outer
%cond_outer =w csltl %i, %n
jnz %cond_outer, @calc_run, @return

@calc_run
%run_base =l add %src, %i
%run_b =w loadub %run_base
%run =l copy 1
%rem =l sub %n, %i
%cap_run =l copy 128
%rem_lt_128 =w csltl %rem, 128
jnz %rem_lt_128, @set_cap_rem, @run_loop_check
@set_cap_rem
%cap_run =l copy %rem
jmp @run_loop_check

@run_loop_check
%run_lt_cap =w csltl %run, %cap_run
jnz %run_lt_cap, @run_step, @run_after

@run_step
%next_i =l add %i, %run
%next_p =l add %src, %next_i
%next_b =w loadub %next_p
%eq =w ceqw %run_b, %next_b
jnz %eq, @run_inc, @run_after

@run_inc
%run =l add %run, 1
jmp @run_loop_check

@run_after
%run_ge_3 =w cslel 3, %run
jnz %run_ge_3, @emit_repeat, @emit_literal

@emit_repeat
%hdr_rep =l sub 257, %run
%avail_rep =l sub %cap, %written
%avail_lt_2 =w csltl %avail_rep, 2
jnz %avail_lt_2, @fail, @write_repeat

@write_repeat
%hdr_w =w trunc %hdr_rep
%hdr_p =l add %out, %written
storeb %hdr_w, %hdr_p
%byte_p =l add %hdr_p, 1
storeb %run_b, %byte_p
%written =l add %written, 2
%i =l add %i, %run
jmp @check_outer

@emit_literal
%avail_l0 =l sub %cap, %written
%av0_lt_2 =w csltl %avail_l0, 2
jnz %av0_lt_2, @fail, @lit_scan

@lit_scan
%j =l copy %i
jmp @lit_scan_check

@lit_scan_check
%j_lt_n =w csltl %j, %n
jnz %j_lt_n, @lit_check_len, @lit_finish

@lit_check_len
%len =l sub %j, %i
%len_lt_128 =w csltl %len, 128
jnz %len_lt_128, @lit_check_run, @lit_finish

@lit_check_run
%jp2 =l add %j, 2
%jp2_lt_n =w csltl %jp2, %n
jnz %jp2_lt_n, @lit_check_eq, @lit_advance

@lit_check_eq
%bp0 =l add %src, %j
%b0 =w loadub %bp0
%jp1 =l add %j, 1
%bp1 =l add %src, %jp1
%b1 =w loadub %bp1
%bp2 =l add %src, %jp2
%b2 =w loadub %bp2
%beq01 =w ceqw %b0, %b1
%beq02 =w ceqw %b0, %b2
%all =w and %beq01, %beq02
jnz %all, @lit_finish, @lit_advance

@lit_advance
%j =l add %j, 1
jmp @lit_scan_check

@lit_finish
%L =l sub %j, %i
%need_l =l add %L, 1
%avail_l =l sub %cap, %written
%avail_lt_need =w csltl %avail_l, %need_l
jnz %avail_lt_need, @fail, @write_literal

@write_literal
%hdr_lit =l sub %L, 1
%hdr_l_w =w trunc %hdr_lit
%hdr_p_l =l add %out, %written
storeb %hdr_l_w, %hdr_p_l
%k =l copy 0
jmp @copy_check

@copy_check
%k_lt_L =w csltl %k, %L
jnz %k_lt_L, @copy_body, @copy_done

@copy_body
%src_idx =l add %i, %k
%src_p =l add %src, %src_idx
%bc =w loadub %src_p
%dst_base =l add %written, 1
%dst_p =l add %dst_base, %k
%final_p =l add %out, %dst_p
storeb %bc, %final_p
%k =l add %k, 1
jmp @copy_check

@copy_done
%written =l add %written, %need_l
%i =l copy %j
jmp @check_outer

@fail
%written =l copy -1
jmp @return

@return
ret %written
}
