export function l $fill_down(l %a, l %n, l %start) {
@start
	jmp @loop(%n)

@loop(%cur)
	%cond =w csgtl %cur, 0
	jnz %cond, @body, @end

@body(%cur)
	%idx =l sub %n, %cur
	%val =l sub %start, %idx
	%off =l mul %idx, 8
	%ptr =l add %a, %off
	storel %val, %ptr
	%nxt =l sub %cur, 1
	jmp @loop(%nxt)

@end
	ret %n
}
