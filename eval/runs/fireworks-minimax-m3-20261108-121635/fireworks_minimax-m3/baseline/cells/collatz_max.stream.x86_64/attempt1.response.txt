    je .LInner_done
    test rbx, 1
    jz .LEven
    lea rbx, [rbx + rbx*2 + 1]
    jmp .LNext
.LEven:
    shr rbx, 1
.LNext:
    inc r14
    jmp .LInner

    cmp r14, rax
    jle .LSkip
    mov rax, r14
.LSkip:
    inc r13
    jmp .LOuter
.Ldone:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
