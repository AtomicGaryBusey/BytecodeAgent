    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!

    mov x19, x0
    mov x20, x1

    blt .Lreturn0
    sub x9, x20, #1
    lsl x10, x9, #4
    ldr x21, [x19, x10]
    add x10, x10, #8
    ldr x22, [x19, x10]

.Lpart_loop:

    lsl x10, x10, #4
    add x11, x19, x10
    cmp x24, x11

    blt .Lpart_swap
    bgt .Lpart_skip
    cmp x13, x22
    blt .Lpart_swap

    ldr x0, [x14]
    ldr x1, [x14, #8]
    ldr x2, [x24]
    ldr x3, [x24, #8]

    str x3, [x14, #8]
    str x0, [x24]
    str x1, [x24, #8]

.Lpart_done:

    lsl x10, x10, #4
    add x11, x19, x10
    add x14, x19, x23
    ldr x0, [x14]
    ldr x1, [x14, #8]
    ldr x2, [x11]
    ldr x3, [x11, #8]
    str x2, [x14]

    lsl x23, x23, #4
    sub x26, x20, x25

    mov x1, x26
    bl qsort_pairs
    mov x20, x25
    b .Lstart
.Lrecurse_left:

    bl qsort_pairs
    add x25, x25, #1
    lsl x25, x25, #4
    add x19, x19, x25
    mov x20, x26
    b .Lstart
.Lreturn0:

.Lepilogue:

    ldp x21, x22, [sp], #16
