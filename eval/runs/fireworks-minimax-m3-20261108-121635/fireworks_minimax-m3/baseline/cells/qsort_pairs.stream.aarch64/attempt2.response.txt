qsort_pairs:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    mov x29, sp
    mov x19, x0
    mov x20, x1
.Lstart:
    cmp x20, #1
    bls .Lreturn0
    lsr x26, x20, #1
    lsl x28, x26, #4
    ldr x24, [x19, x28]
    add x28, x28, #8
    ldr x25, [x19, x28]
    mov x21, #0
    sub x22, x20, #1
.Lpartition_loop:
    b .Lleft_find_loop
.Lleft_find_loop:
    cmp x21, x22
    b.gt .Lleft_done
    lsl x26, x21, #4
    ldr x23, [x19, x26]
    add x26, x26, #8
    ldr x27, [x19, x26]
    cmp x23, x24
    blt .Lleft_inc
    bgt .Lleft_done
    cmp x27, x25
    blt .Lleft_inc
    b .Lleft_done
.Lleft_inc:
    add x21, x21, #1
    b .Lleft_find_loop
.Lleft_done:
    b .Lright_find_loop
.Lright_find_loop:
    cmp x21, x22
    b.gt .Lright_done
    lsl x26, x22, #4
    ldr x23, [x19, x26]
    add x26, x26, #8
    ldr x27, [x19, x26]
    cmp x23, x24
    bgt .Lright_dec
    blt .Lright_done
    cmp x27, x25
    bgt .Lright_dec
    b .Lright_done
.Lright_dec:
    sub x22, x22, #1
    b .Lright_find_loop
.Lright_done:
    cmp x21, x22
    b.gt .Lpartition_end
    lsl x26, x21, #4
    ldr x3, [x19, x26]
    add x26, x26, #8
    ldr x4, [x19, x26]
    lsl x26, x22, #4
    ldr x5, [x19, x26]
    add x26, x26, #8
    ldr x6, [x19, x26]
    lsl x26, x21, #4
    str x5, [x19, x26]
    add x26, x26, #8
    str x6, [x19, x26]
    lsl x26, x22, #4
    str x3, [x19, x26]
    add x26, x26, #8
    str x4, [x19, x26]
    add x21, x21, #1
    sub x22, x22, #1
    b .Lpartition_loop
.Lpartition_end:
    sub x22, x20, x21
    cmp x21, x22
    b.le .Lcall_left
    lsl x27, x21, #4
    add x0, x19, x27
    mov x1, x22
    bl qsort_pairs
    mov x20, x21
    b .Lstart
.Lcall_left:
    mov x0, x19
    mov x1, x21
    bl qsort_pairs
    lsl x27, x21, #4
    add x19, x19, x27
    mov x20, x22
    b .Lstart
.Lreturn0:
    mov x0, #0
    b .Lepilogue
.Lepilogue:
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
