export function l $base64_encode(l %src, l %n, l %out) {
@start
    %stack =l alloc16 64
    %init_i =l copy 0
    jmp @init_cond
@init_cond
    %i =l phi @start %init_i, @init_store %i_next
    %cond_i =w csltl %i, 64
    jnz %cond_i, @init_body, @init_done
@init_body
    %ptr =l add %stack, %i
    %lt26 =w csltl %i, 26
    jnz %lt26, @init_lt26, @init_ge26
@init_lt26
    %ca1 =l add %i, 65
    jmp @init_store
@init_ge26
    %lt52 =w csltl %i, 52
    jnz %lt52, @init_lt52, @init_ge52
@init_lt52
    %ca2 =l add %i, 71
    jmp @init_store
@init_ge52
    %lt62 =w csltl %i, 62
    jnz %lt62, @init_lt62, @init_ge62
@init_lt62
    %ca3 =l sub %i, 4
    jmp @init_store
@init_ge62
    %eq62 =w ceql %i, 62
    jnz %eq62, @init_eq62, @init_eq63
@init_eq62
    %ca4 =l copy 43
    jmp @init_store
@init_eq63
    %ca5 =l copy 47
    jmp @init_store
@init_store
    %c =l phi @init_lt26 %ca1, @init_lt52 %ca2, @init_lt62 %ca3, @init_eq62 %ca4, @init_eq63 %ca5
    storeb %c, %ptr
    %i_next =l add %i, 1
    jmp @init_cond
@init_done
    %full =l div %n, 3
    %rem =l rem %n, 3
    %cnt0 =l copy 0
    jmp @loop_cond
@loop_cond
    %cnt =l phi @init_done %cnt0, @loop_body %cnt_next
    %cond_l =w csltl %cnt, %full
    jnz %cond_l, @loop_body, @loop_end
@loop_body
    %offb =l mul %cnt, 3
    %spb =l add %src, %offb
    %b0b =l loadub %spb
    %spb1 =l add %spb, 1
    %b1b =l loadub %spb1
    %spb2 =l add %spb, 2
    %b2b =l loadub %spb2
    %v0b =l mul %b0b, 65536
    %v1b =l mul %b1b, 256
    %v2b =l add %v0b, %v1b
    %v3b =l add %v2b, %b2b
    %idx0b_t =l shr %v3b, 18
    %idx0b =l and %idx0b_t, 63
    %idx1b_t =l shr %v3b, 12
    %idx1b =l and %idx1b_t, 63
    %idx2b_t =l shr %v3b, 6
    %idx2b =l and %idx2b_t, 63
    %idx3b =l and %v3b, 63
    %p0b =l add %stack, %idx0b
    %c0b =l loadub %p0b
    %p1b =l add %stack, %idx1b
    %c1b =l loadub %p1b
    %p2b =l add %stack, %idx2b
    %c2b =l loadub %p2b
    %p3b =l add %stack, %idx3b
    %c3b =l loadub %p3b
    %ooffb =l mul %cnt, 4
    %opb =l add %out, %ooffb
    storeb %c0b, %opb
    %opb1 =l add %opb, 1
    storeb %c1b, %opb1
    %opb2 =l add %opb, 2
    storeb %c2b, %opb2
    %opb3 =l add %opb, 3
    storeb %c3b, %opb3
    %cnt_next =l add %cnt, 1
    jmp @loop_cond
@loop_end
    %ooffe =l mul %full, 4
    %ope =l add %out, %ooffe
    %offe =l mul %full, 3
    %spe =l add %src, %offe
    %is_zero =w ceql %rem, 0
    jnz %is_zero, @ret_full, @rem_check
@ret_full
    %retf =l mul %full, 4
    ret %retf
@rem_check
    %is_one =w ceql %rem, 1
    jnz %is_one, @rem1, @rem2
@rem1
    %b01 =l loadub %spe
    %t01 =l shr %b01, 2
    %idx01 =l and %t01, 63
    %t11 =l and %b01, 3
    %idx11 =l mul %t11, 16
    %p01 =l add %stack, %idx01
    %c01 =l loadub %p01
    %p11 =l add %stack, %idx11
    %c11 =l loadub %p11
    storeb %c01, %ope
    %ope1 =l add %ope, 1
    storeb %c11, %ope1
    %ope2 =l add %ope, 2
    storeb 61, %ope2
    %ope3 =l add %ope, 3
    storeb 61, %ope3
    %ret1 =l mul %full, 4
    %fret1 =l add %ret1, 4
    ret %fret1
@rem2
    %b02 =l loadub %spe
    %spe1 =l add %spe, 1
    %b12 =l loadub %spe1
    %v02 =l mul %b02, 65536
    %v12 =l mul %b12, 256
    %v22 =l add %v02, %v12
    %idx02_t =l shr %v22, 18
    %idx02 =l and %idx02_t, 63
    %idx12_t =l shr %v22, 12
    %idx12 =l and %idx12_t, 63
    %idx22_t =l shr %v22, 6
    %idx22 =l and %idx22_t, 63
    %p02 =l add %stack, %idx02
    %c02 =l loadub %p02
    %p12 =l add %stack, %idx12
    %c12 =l loadub %p12
    %p22 =l add %stack, %idx22
    %c22 =l loadub %p22
    storeb %c02, %ope
    %ope1_2 =l add %ope, 1
    storeb %c12, %ope1_2
    %ope2_2 =l add %ope, 2
    storeb %c22, %ope2_2
    %ope3_2 =l add %ope, 3
    storeb 61, %ope3_2
    %ret2 =l mul %full, 4
    %fret2 =l add %ret2, 4
    ret %fret2
}
