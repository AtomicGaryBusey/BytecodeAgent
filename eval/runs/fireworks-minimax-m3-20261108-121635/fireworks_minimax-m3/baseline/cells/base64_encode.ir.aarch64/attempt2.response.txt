export function l $base64_encode(l %src, l %n, l %out) {
@start
    %stack =l alloc16 64
    %i =l copy 0
    jmp @init_cond
@init_cond
    %i =l phi @start %i, @init_store %i_next
    %cond =w csltl %i, 64
    jnz %cond, @init_body, @init_done
@init_body
    %ptr =l add %stack, %i
    %lt26 =w csltl %i, 26
    jnz %lt26, @init_lt26, @init_ge26
@init_lt26
    %c =l add %i, 65
    jmp @init_store
@init_ge26
    %lt52 =w csltl %i, 52
    jnz %lt52, @init_lt52, @init_ge52
@init_lt52
    %c =l add %i, 71
    jmp @init_store
@init_ge52
    %lt62 =w csltl %i, 62
    jnz %lt62, @init_lt62, @init_ge62
@init_lt62
    %c =l sub %i, 4
    jmp @init_store
@init_ge62
    %eq62 =w ceql %i, 62
    jnz %eq62, @init_eq62, @init_eq63
@init_eq62
    %c =l copy 43
    jmp @init_store
@init_eq63
    %c =l copy 47
    jmp @init_store
@init_store
    %c =l phi @init_lt26 %c, @init_lt52 %c, @init_lt62 %c, @init_eq62 %c, @init_eq63 %c
    %ptr =l phi @init_lt26 %ptr, @init_lt52 %ptr, @init_lt62 %ptr, @init_eq62 %ptr, @init_eq63 %ptr
    %i_in =l phi @init_lt26 %i, @init_lt52 %i, @init_lt62 %i, @init_eq62 %i, @init_eq63 %i
    storeb %c, %ptr
    %i_next =l add %i_in, 1
    jmp @init_cond
@init_done
    %full =l div %n, 3
    %rem =l rem %n, 3
    jmp @loop_cond
@loop_cond
    %cnt =l phi @init_done 0, @loop_body %cnt_next
    %full =l phi @init_done %full, @loop_body %full
    %cond =w csltl %cnt, %full
    jnz %cond, @loop_body, @loop_end
@loop_body
    %srcoff =l mul %cnt, 3
    %sp =l add %src, %srcoff
    %b0 =l loadub %sp
    %sp1 =l add %sp, 1
    %b1 =l loadub %sp1
    %sp2 =l add %sp, 2
    %b2 =l loadub %sp2
    %v0 =l mul %b0, 65536
    %v1 =l add %v0, %b1
    %v2 =l mul %v1, 256
    %v3 =l add %v2, %b2
    %t0 =l shr %v3, 18
    %idx0 =l and %t0, 63
    %t1 =l shr %v3, 12
    %idx1 =l and %t1, 63
    %t2 =l shr %v3, 6
    %idx2 =l and %t2, 63
    %idx3 =l and %v3, 63
    %p0 =l add %stack, %idx0
    %c0 =l loadub %p0
    %p1 =l add %stack, %idx1
    %c1 =l loadub %p1
    %p2 =l add %stack, %idx2
    %c2 =l loadub %p2
    %p3 =l add %stack, %idx3
    %c3 =l loadub %p3
    %outoff =l mul %cnt, 4
    %op =l add %out, %outoff
    storeb %c0, %op
    %op1 =l add %op, 1
    storeb %c1, %op1
    %op2 =l add %op, 2
    storeb %c2, %op2
    %op3 =l add %op, 3
    storeb %c3, %op3
    %cnt_next =l add %cnt, 1
    jmp @loop_cond
@loop_end
    %outoff =l mul %full, 4
    %op =l add %out, %outoff
    %srcoff =l mul %full, 3
    %sp =l add %src, %srcoff
    %is_zero =w ceql %rem, 0
    jnz %is_zero, @ret_full, @rem_check
@ret_full
    %ret =l mul %full, 4
    ret %ret
@rem_check
    %is_one =w ceql %rem, 1
    jnz %is_one, @rem1, @rem2
@rem1
    %b0 =l loadub %sp
    %t0 =l shr %b0, 2
    %idx0 =l and %t0, 63
    %t1 =l and %b0, 3
    %idx1 =l mul %t1, 16
    %p0 =l add %stack, %idx0
    %c0 =l loadub %p0
    %p1 =l add %stack, %idx1
    %c1 =l loadub %p1
    storeb %c0, %op
    %op1 =l add %op, 1
    storeb %c1, %op1
    %op2 =l add %op, 2
    storeb 61, %op2
    %op3 =l add %op, 3
    storeb 61, %op3
    %ret =l mul %full, 4
    %ret =l add %ret, 4
    ret %ret
@rem2
    %b0 =l loadub %sp
    %sp1 =l add %sp, 1
    %b1 =l loadub %sp1
    %v0 =l mul %b0, 65536
    %v1 =l mul %b1, 256
    %v2 =l add %v0, %v1
    %t0 =l shr %v2, 18
    %idx0 =l and %t0, 63
    %t1 =l shr %v2, 12
    %idx1 =l and %t1, 63
    %t2 =l shr %v2, 6
    %idx2 =l and %t2, 63
    %p0 =l add %stack, %idx0
    %c0 =l loadub %p0
    %p1 =l add %stack, %idx1
    %c1 =l loadub %p1
    %p2 =l add %stack, %idx2
    %c2 =l loadub %p2
    storeb %c0, %op
    %op1 =l add %op, 1
    storeb %c1, %op1
    %op2 =l add %op, 2
    storeb %c2, %op2
    %op3 =l add %op, 3
    storeb 61, %op3
    %ret =l mul %full, 4
    %ret =l add %ret, 4
    ret %ret
}
