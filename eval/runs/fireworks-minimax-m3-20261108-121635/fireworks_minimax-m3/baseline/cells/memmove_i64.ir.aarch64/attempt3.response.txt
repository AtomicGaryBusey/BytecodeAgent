export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%dst_lt_src =w cultl %dst, %src
	jnz %dst_lt_src, @fwd, @bwd

@fwd
	%i =l 0
	%cond =w cltl %i, %n
	jnz %cond, @fwd_body, @done

@fwd_body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l add %i, 1
	%cond =w cltl %i, %n
	jnz %cond, @fwd_body, @done

@bwd
	%i =l sub %n, 1
	%cond2 =w cltl %i, 0
	jnz %cond2, @done, @bwd_body

@bwd_body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l sub %i, 1
	%cond2 =w cltl %i, 0
	jnz %cond2, @done, @bwd_body

@done
	ret 0
}
