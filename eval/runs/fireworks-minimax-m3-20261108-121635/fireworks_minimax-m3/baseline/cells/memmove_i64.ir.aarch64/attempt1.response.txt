export function l $memmove_i64(l %dst, l %src, l %n) {
	%dst_lt_src =w cultl %dst, %src
	jnz %dst_lt_src, @fwd, @bwd

@fwd
	%i =l 0
	jmp @fwd_loop

@fwd_loop
	%cond =w csltl %i, %n
	jnz %cond, @fwd_body, @ret0

@fwd_body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l add %i, 1
	jmp @fwd_loop

@bwd
	%i =l sub %n, 1
	jmp @bwd_loop

@bwd_loop
	%cond =w csltl %i, 0
	jnz %cond, @ret0, @bwd_body

@bwd_body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l sub %i, 1
	jmp @bwd_loop

@ret0
	ret 0
}
