export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer_check

@outer_check
	%i =l phi @start 0, @outer_inc %i_next
	%t1 =l sub %n, %i
	%bound1 =l sub %t1, 1
	%c1 =w csltl 0, %bound1
	jnz %c1, @inner_check, @ret

@inner_check
	%j =l phi @outer_check 0, @inner_inc %j_next
	%t2 =l sub %n, %i
	%bound2 =l sub %t2, 1
	%c2 =w csltl %j, %bound2
	jnz %c2, @inner_body, @outer_inc

@inner_body
	%off =l mul %j, 8
	%addr =l add %a, %off
	%v0 =l loadl %addr
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%addr1 =l add %a, %off1
	%v1 =l loadl %addr1
	%gt =w csgtl %v0, %v1
	jnz %gt, @swap, @inner_inc

@swap
	storel %v1, %addr
	storel %v0, %addr1
	jmp @inner_inc

@inner_inc
	%j_next =l add %j, 1
	jmp @inner_check

@outer_inc
	%i_next =l add %i, 1
	jmp @outer_check

@ret
	ret 0
}
