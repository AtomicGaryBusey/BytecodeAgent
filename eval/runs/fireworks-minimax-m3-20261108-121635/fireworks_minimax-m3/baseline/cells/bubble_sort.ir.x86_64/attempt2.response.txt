export function l $bubble_sort(l %a, l %n) {
@start
	%c =w cslel %n, 1
	jnz %c, @ret, @init

@init
	%i =l add 0, 0
	jmp @outer_check

@outer_check
	%i =l phi @init %i, @outer_inc %i_inc
	%t =l sub %n, %i
	%bound =l sub %t, 1
	%c =w csltl 0, %bound
	jnz %c, @inner_init, @ret

@inner_init
	%i_pass =l add %i, 0
	%j =l add 0, 0
	jmp @inner_check

@inner_check
	%i =l phi @inner_init %i_pass, @inner_inc %i_p
	%j =l phi @inner_init %j, @inner_inc %j_next
	%t =l sub %n, %i
	%bound =l sub %t, 1
	%c =w csltl %j, %bound
	jnz %c, @inner_body, @outer_inc

@inner_body
	%off =l mul %j, 8
	%addr =l add %a, %off
	%v0 =l loadl %addr
	%next =l add %j, 1
	%off1 =l mul %next, 8
	%addr1 =l add %a, %off1
	%v1 =l loadl %addr1
	%gt =w csgtl %v0, %v1
	jnz %gt, @swap, @inner_inc

@swap
	storel %v1, %addr
	storel %v0, %addr1
	jmp @inner_inc

@inner_inc
	%i_p =l add %i, 0
	%j_next =l add %j, 1
	jmp @inner_check

@outer_inc
	%i_inc =l add %i, 1
	jmp @outer_check

@ret
	ret 0
}
