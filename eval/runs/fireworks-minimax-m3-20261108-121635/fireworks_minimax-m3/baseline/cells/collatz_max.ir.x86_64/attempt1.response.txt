export function l $collatz_max(l %n) {
@start
	%max = copy 0
	%i = copy 1
	jmp @outer_cond

@outer_cond
	%cond = cslel %i, %n
	jnz %cond, @outer_body, @outer_done

@outer_body
	%current = copy %i
	%steps = copy 0
	jmp @collatz_cond

@collatz_cond
	%is_one = ceql %current, 1
	jnz %is_one, @max_update, @coll_step

@coll_step
	%bit = and %current, 1
	%is_even = ceql %bit, 0
	jnz %is_even, @coll_even, @coll_odd

@coll_even
	%current = div %current, 2
	jmp @coll_continue

@coll_odd
	%tmp = mul %current, 3
	%current = add %tmp, 1
	jmp @coll_continue

@coll_continue
	%steps = add %steps, 1
	jmp @collatz_cond

@max_update
	%is_greater = csltl %max, %steps
	jnz %is_greater, @max_assign, @outer_next

@max_assign
	%max = copy %steps
	jmp @outer_next

@outer_next
	%i = add %i, 1
	jmp @outer_cond

@outer_done
	ret %max
}
