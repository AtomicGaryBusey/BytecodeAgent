export function l $collatz_max(l %n) {
@start
    %max0 = sub %n, %n
    %i0 = add %max0, 1
    jmp @outer_cond

@outer_cond
    %max = phi @start %max0, @max_assign %mx_new, @outer_next %mx_p
    %i = phi @start %i0, @max_assign %i_v1, @outer_next %i_v2
    %cond = csltel %i, %n
    jnz %cond, @outer_body, @outer_done

@outer_body
    %c0 = copy %i
    %s0 = sub %c0, %c0
    jmp @collatz_cond

@collatz_cond
    %current = phi @outer_body %c0, @coll_continue %cn
    %steps = phi @outer_body %s0, @coll_continue %sn
    %is_one = ceql %current, 1
    jnz %is_one, @max_update, @coll_step

@coll_step
    %bit = and %current, 1
    %is_even = ceql %bit, 0
    jnz %is_even, @coll_even, @coll_odd

@coll_even
    %cn_e = div %current, 2
    jmp @coll_continue

@coll_odd
    %tmp = mul %current, 3
    %cn_o = add %tmp, 1
    jmp @coll_continue

@coll_continue
    %cn = phi @coll_even %cn_e, @coll_odd %cn_o
    %sn = add %steps, 1
    jmp @collatz_cond

@max_update
    %i_n = add %i, 1
    %is_greater = csltl %max, %steps
    jnz %is_greater, @max_assign, @outer_next

@max_assign
    %mx_new = copy %steps
    %i_v1 = copy %i_n
    jmp @outer_cond

@outer_next
    %mx_p = copy %max
    %i_v2 = copy %i_n
    jmp @outer_cond

@outer_done
    ret %max
}
