export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo = copy 0
	%hi = copy %n
	jmp @loop

@loop
	%cond = csltl %lo, %hi
	jnz %cond, @body, @end

@body
	%diff = sub %hi, %lo
	%half = sar %diff, 1
	%mid = add %lo, %half
	%off = shl %mid, 3
	%addr = add %a, %off
	%val = loadl %addr
	%eq = ceql %val, %key
	jnz %eq, @found, @less

@less
	%lt = csltl %val, %key
	jnz %lt, @lo_upd, @hi_upd

@lo_upd
	%lo = add %mid, 1
	%hi = add %hi, 0
	jmp @loop

@hi_upd
	%hi = copy %mid
	%lo = add %lo, 0
	jmp @loop

@found
	ret %mid

@end
	%r = copy -1
	ret %r
}
