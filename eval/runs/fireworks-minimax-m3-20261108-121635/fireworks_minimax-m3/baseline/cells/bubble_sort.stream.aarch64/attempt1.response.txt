bubble_sort:
    cmp x1, #1
    ble .Ldone
    sub sp, sp, #32
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]
    mov x19, x0
    sub x20, x1, #1
    mov x21, #0
.Louter:
    cmp x21, x20
    bge .Lend
    sub x22, x20, x21
    mov x23, #0
.Linner:
    cmp x23, x22
    bge .Lnext_outer
    lsl x24, x23, #3
    add x25, x19, x24
    ldr x26, [x25]
    ldr x27, [x25, #8]
    cmp x26, x27
    ble .Lno_swap
    str x27, [x25]
    str x26, [x25, #8]
.Lno_swap:
    add x23, x23, #1
    b .Linner
.Lnext_outer:
    add x21, x21, #1
    b .Louter
.Lend:
    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    add sp, sp, #32
.Ldone:
    mov x0, #0
    ret
