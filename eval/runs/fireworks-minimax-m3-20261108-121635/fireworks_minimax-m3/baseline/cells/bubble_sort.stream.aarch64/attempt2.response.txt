bubble_sort:
    cmp x1, #1
    ble .Ldone
    mov x9, x0
    sub x10, x1, #1
    mov x11, #0
.Louter:
    cmp x11, x10
    bge .Lend
    sub x12, x10, x11
    mov x13, #0
.Linner:
    cmp x13, x12
    bge .Lnext_outer
    lsl x14, x13, #3
    add x15, x9, x14
    ldr x16, [x15]
    ldr x17, [x15, #8]
    cmp x16, x17
    ble .Lno_swap
    str x17, [x15]
    str x16, [x15, #8]
.Lno_swap:
    add x13, x13, #1
    b .Linner
.Lnext_outer:
    add x11, x11, #1
    b .Louter
.Lend:
.Ldone:
    mov x0, #0
    ret
