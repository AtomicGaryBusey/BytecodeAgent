export function l $reverse(l %a, l %n) {
@entry
	%c0 =l copy 0
	%c1 =l copy 1
	%c3 =l copy 3
	%i =l copy 0
	%j =l sub %n, %c1
	jmp @loop

@loop
	%cond =w csltl %i, %j
	jnz %cond, @body, @done

@body
	%off_i =l shl %i, %c3
	%off_j =l shl %j, %c3
	%ptr_i =l add %a, %off_i
	%ptr_j =l add %a, %off_j
	%val_i =l loadl %ptr_i
	%val_j =l loadl %ptr_j
	storel %val_j, %ptr_i
	storel %val_i, %ptr_j
	%i =l add %i, %c1
	%j =l sub %j, %c1
	jmp @loop

@done
	ret %c0
}
