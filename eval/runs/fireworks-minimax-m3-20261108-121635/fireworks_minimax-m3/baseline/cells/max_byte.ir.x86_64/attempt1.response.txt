export function l $max_byte(l %p, l %n) {
    %max =l copy 0
    %i =l copy 0
    jmp @loop
@loop
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %a =l add %p, %i
    %b =l loadub %a
    %t =w cultl %max, %b
    jnz %t, @upd, @next
@upd
    %max =l copy %b
@next
    %i =l add %i, 1
    jmp @loop
@done
    ret %max
}
