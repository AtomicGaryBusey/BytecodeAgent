memmove_i64:
    cbz x2, .Lzero
    cmp x0, x1
    b.eq .Lzero
    b.lo .Lforward
.Lbackward:
    add x1, x1, x2, lsl #3
    add x0, x0, x2, lsl #3
.Lbackward_loop:
    cbz x2, .Lzero
    ldr x3, [x1, #-8]!
    str x3, [x0, #-8]!
    sub x2, x2, #1
    b .Lbackward_loop

.Lforward:
.Lforward_loop:
    cbz x2, .Lzero
    ldr x3, [x1], #8
    str x3, [x0], #8
    sub x2, x2, #1
    b .Lforward_loop

.Lzero:
    mov x0, #0
    ret
