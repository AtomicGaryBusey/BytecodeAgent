    push r12
    push r13
    push r14
    push r15
    push rbp
    mov rbp, rsp
    sub rsp, 32
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rdi, [r12]
    call r14
    mov [rbp-8], rax
    mov [rbp-16], rax
    mov [rbp-24], rax
    and rax, 1
    mov [rbp-32], rax
    add r12, 8
    dec r13
.Lloop:
    test r13, r13
    jz .Lend
    mov rdi, [r12]
    call r14
    cmp rax, [rbp-8]
    jge .Lskip_min
    mov [rbp-8], rax
.Lskip_min:
    cmp rax, [rbp-16]
    jle .Lskip_max
    mov [rbp-16], rax
.Lskip_max:
    add [rbp-24], rax
    test rax, 1
    jz .Lskip_odd
    inc qword ptr [rbp-32]
.Lskip_odd:
    add r12, 8
    dec r13
    jmp .Lloop
.Lend:
    mov rax, [rbp-8]
    mov [r15], rax
    mov rax, [rbp-16]
    mov [r15+8], rax
    mov rax, [rbp-24]
    mov [r15+16], rax
    mov rax, [rbp-32]
    mov rsp, rbp
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    ret
