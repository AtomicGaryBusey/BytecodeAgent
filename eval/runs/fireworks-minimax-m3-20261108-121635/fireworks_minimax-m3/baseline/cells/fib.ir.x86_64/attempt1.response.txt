export function l $fib(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @is_zero, @not_zero

@is_zero
	ret 0

@not_zero
	jmp @loop_check

@loop_check
	%a =l phi @not_zero 0, @loop_body %a_next
	%b =l phi @not_zero 1, @loop_body %b_next
	%k =l phi @not_zero 1, @loop_body %k_next
	%cond =w csltl %k, %n
	jnz %cond, @loop_body, @loop_exit

@loop_body
	%c =l add %a, %b
	%a_next =l copy %b
	%b_next =l copy %c
	%k_next =l add %k, 1
	jmp @loop_check

@loop_exit
	ret %b
}
