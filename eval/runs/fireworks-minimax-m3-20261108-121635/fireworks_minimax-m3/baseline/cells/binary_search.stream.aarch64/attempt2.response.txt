binary_search:
    cbz x1, .Lnot_found
    mov x3, x0
    mov x4, x2
    mov x5, #0
    mov x6, x1
.Lloop:
    cmp x5, x6
    b.ge .Lnot_found
    add x7, x5, x6
    asr x7, x7, #1
    ldr x8, [x3, x7, lsl #3]
    cmp x8, x4
    b.eq .Lfound
    b.lt .Lless
    mov x6, x7
    b .Lloop
.Lless:
    add x5, x7, #1
    b .Lloop
.Lfound:
    mov x0, x7
    ret
.Lnot_found:
    mov x0, #-1
    ret
