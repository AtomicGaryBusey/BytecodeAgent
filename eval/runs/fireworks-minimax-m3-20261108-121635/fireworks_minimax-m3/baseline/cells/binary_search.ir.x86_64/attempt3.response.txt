export function l $binary_search(l %a, l %n, l %key) {
@start
	%result = copy -1
	%nz = ceql %n, 0
	jnz %nz, @done, @init
@init
	%lo = copy 0
	%hi = sub %n, 1
	jmp @cond
@cond
	%c1 = cslel %lo, %hi
	jnz %c1, @step, @done
@step
	%d = sub %hi, %lo
	%h = div %d, 2
	%m = add %lo, %h
	%off = mul %m, 8
	%adr = add %a, %off
	%v = loadl %adr
	%lt = csltl %v, %key
	jnz %lt, @rgt, @eq
@eq
	%cm = ceql %v, %key
	jnz %cm, @hit, @lft
@rgt
	%lo = add %m, 1
	jmp @cond
@lft
	%hi = sub %m, 1
	jmp @cond
@hit
	%result = copy %m
	jmp @done
@done
	ret %result
}
