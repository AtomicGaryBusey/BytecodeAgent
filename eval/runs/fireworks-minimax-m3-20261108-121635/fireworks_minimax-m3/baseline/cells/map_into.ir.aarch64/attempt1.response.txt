export function l $map_into(l %dst, l %src, l %n, l %f) {
@loop
    %i =l phi @loop 0, @merge %i.next
    %cnt =l phi @loop 0, @merge %cnt.next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %off =l mul %i, 8
    %sa =l add %src, %off
    %sv =l loadl %sa
    %rv =l call %f(l %sv)
    %da =l add %dst, %off
    storel %rv, %da
    %less =w csltl %rv, %sv
    jnz %less, @inc, @merge

@inc
    %cntplus =l add %cnt, 1
    jmp @merge

@merge
    %cnt.next =l phi @inc %cntplus, @body %cnt
    %i.next =l add %i, 1
    jmp @loop

@done
    ret %cnt
}
