export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @cont %i.next
    %cnt =l phi @start 0, @cont %cnt.next
    %cond =w csltl %i, %n
    jnz %cond, @step, @done

@step
    %off =l mul %i, 8
    %sa =l add %src, %off
    %sv =l loadl %sa
    %rv =l call %f(l %sv)
    %da =l add %dst, %off
    storel %rv, %da
    %less =w csltl %rv, %sv
    %i.next =l add %i, 1
    jnz %less, @inc, @cont

@inc
    %cnt.new =l add %cnt, 1
    jmp @cont

@cont
    %cnt.next =l phi @inc %cnt.new, @step %cnt
    jmp @loop

@done
    ret %cnt
}
