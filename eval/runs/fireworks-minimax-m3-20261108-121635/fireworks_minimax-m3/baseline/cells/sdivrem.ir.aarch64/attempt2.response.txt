export function l $sdivrem(l %a, l %b, l %rem) -> l {
@start
	%bzero = ceql %b, 0
	jnz %bzero, @b_zero, @check_overflow

@b_zero
	storel %a, %rem
	ret 0

@check_overflow
	%amin = ceql %a, -9223372036854775808
	%bneg1 = ceql %b, -1
	%flag = and %amin, %bneg1
	jnz %flag, @overflow, @do_div

@overflow
	storel 0, %rem
	ret -9223372036854775808

@do_div
	%q = div %a, %b
	%r = rem %a, %b
	storel %r, %rem
	ret %q
}
