export function l $sdivrem(l %a, l %b, l %rem) -> l {
@start
	%bzero = ceql %b, 0
	jnz %bzero, @b_zero, @check_overflow

@b_zero
	storel %a, %rem
	ret 0

@check_overflow
	%bneg1 = ceql %b, -1
	jnz %bneg1, @check_a, @do_div

@check_a
	%neg_a = subl 0, %a
	%eq_self = ceql %neg_a, %a
	%is_neg = csltl %a, 0
	%is_min = andw %eq_self, %is_neg
	jnz %is_min, @overflow, @do_div

@overflow
	storel 0, %rem
	ret %a

@do_div
	%q = div %a, %b
	%r = rem %a, %b
	storel %r, %rem
	ret %q
}
