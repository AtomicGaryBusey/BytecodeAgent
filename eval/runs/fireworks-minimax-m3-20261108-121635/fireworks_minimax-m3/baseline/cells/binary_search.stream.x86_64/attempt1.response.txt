binary_search:
    test rsi, rsi
    jz .Lnotfound
    mov rcx, rsi
    dec rcx
    xor r8, r8
.Lloop:
    cmp r8, rcx
    jg .Lnotfound
    mov rax, r8
    add rax, rcx
    sar rax, 1
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    je .Lfound
    jl .Lless
    dec rcx
    jmp .Lloop
.Lless:
    inc r8
    jmp .Lloop
.Lfound:
    ret
.Lnotfound:
    mov rax, -1
    ret
