export function l $map_sum(l %a, l %n, l %f) {
@start
    %z =l copy 0
    %sum =l copy %z
    %ptr =l copy %a
    %cnt =l copy %n
    %t =w ceql %n, %z
    jnz %t, @end
    jmp @loop

@loop
    %val =l loadl %ptr
    %rv =l call %f(l %val)
    %sum =l add %sum, %rv
    %ptr =l add %ptr, 8
    %cnt =l sub %cnt, 1
    %t2 =w ceql %cnt, %z
    jnz %t2, @end
    jmp @loop

@end
    ret %sum
}
