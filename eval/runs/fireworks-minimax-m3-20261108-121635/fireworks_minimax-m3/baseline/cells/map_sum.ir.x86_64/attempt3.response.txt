export function l $map_sum(l %a, l %n, l %f) {
@start
    %s =l copy 0
    %p =l copy %a
    %c =l copy %n
    %fptr =l copy %f
    %done =w ceql %n, 0
    jnz %done, @end
    jmp @loop

@loop
    %x =l loadl %p
    %y =l call %fptr(l %x)
    %s =l add %s, %y
    %p =l add %p, 8
    %c =l sub %c, 1
    %loopdone =w ceql %c, 0
    jnz %loopdone, @end
    jmp @loop

@end
    ret %s
}
