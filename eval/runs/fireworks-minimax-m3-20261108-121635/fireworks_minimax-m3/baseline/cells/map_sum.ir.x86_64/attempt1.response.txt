export function l $map_sum(l %a, l %n, l %f) {
    %s =l copy 0
    %fptr =l copy %f
    %p =l copy %a
    %c =l copy %n
    %done =w ceql %n, 0
    jnz %done, @end

    %loop:
        %x =l loadl %p
        %y =l call %fptr(l %x)
        %s =l add %s, %y
        %p =l add %p, 8
        %c =l sub %c, 1
        %done =w ceql %c, 0
        jnz %done, @end
        jmp %loop

    @end:
    ret %s
}
