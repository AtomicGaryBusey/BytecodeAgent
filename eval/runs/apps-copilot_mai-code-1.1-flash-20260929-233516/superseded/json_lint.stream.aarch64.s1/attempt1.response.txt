parse_number_digit_continue:
    add x21, x21, #1
    b parse_number_digit_loop
parse_number_after_int:
    cmp x21, x20
    b.hs parse_number_success
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x2E
    b.eq parse_number_frac
    cmp w22, #0x65
    b.eq parse_number_exp
    cmp w22, #0x45
    b.eq parse_number_exp
    b parse_number_success
parse_number_frac:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_fail_current
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x30
    b.lt parse_number_fail_current
    cmp w22, #0x39
    b.le parse_number_frac_digits
    b parse_number_fail_current
parse_number_frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_after_frac
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x30
    b.lt parse_number_after_frac
    cmp w22, #0x39
    b.le parse_number_frac_digits
parse_number_after_frac:
    cmp x21, x20
    b.hs parse_number_success
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x65
    b.eq parse_number_exp
    cmp w22, #0x45
    b.eq parse_number_exp
    b parse_number_success
parse_number_exp:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_fail_current
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x2B
    b.eq parse_number_exp_sign
    cmp w22, #0x2D
    b.eq parse_number_exp_sign
    b parse_number_exp_digit_check
parse_number_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_fail_current
parse_number_exp_digit_check:
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x30
    b.lt parse_number_fail_current
    cmp w22, #0x39
    b.le parse_number_exp_digit_loop
    b parse_number_fail_current
parse_number_exp_digit_loop:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_success
    add x22, x19, x21
    ldrb w22, [x22]
    cmp w22, #0x30
    b.lt parse_number_success
    cmp w22, #0x39
    b.le parse_number_exp_digit_loop
    b parse_number_success
parse_number_success:
    mov x0, x21
    mov x1, #0
    b parse_number_done
parse_number_fail_current:
    mov x0, x21
    mov x1, #1
    b parse_number_done
parse_number_fail:
    mov x0, x21
    mov x1, #1
parse_number_done:
    ldp x29, x30, [sp], #16
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
parse_number_invalid:
    mov x0, x21
    mov x1, #1
    b parse_number_done
