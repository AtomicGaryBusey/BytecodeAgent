    ldrb w23, [x23]

    cmp w23, #0x65
    b.eq parse_number_exp
    cmp w23, #0x45
    b.eq parse_number_exp

    b parse_number_success

parse_number_exp:
    add x22, x22, #1
    cmp x22, x20
    b.hs parse_number_fail_pos

    add x23, x19, x22
    ldrb w23, [x23]

    cmp w23, #0x2B
    b.eq parse_number_exp_sign
    cmp w23, #0x2D
    b.eq parse_number_exp_sign
    b parse_number_exp_digit_check

parse_number_exp_sign:
    add x22, x22, #1
    cmp x22, x20
    b.hs parse_number_fail_pos

parse_number_exp_digit_check:
    add x23, x19, x22
    ldrb w23, [x23]

    cmp w23, #0x30
    b.lt parse_number_fail_pos
    cmp w23, #0x39
    b.le parse_number_exp_digits

    b parse_number_fail_pos

parse_number_exp_digits:
    add x22, x22, #1

parse_number_exp_loop:
    cmp x22, x20
    b.hs parse_number_success

    add x23, x19, x22
    ldrb w23, [x23]

    cmp w23, #0x30
    b.lt parse_number_success
    cmp w23, #0x39
    b.le parse_number_exp_continue

    b parse_number_success

parse_number_exp_continue:
    add x22, x22, #1
    b parse_number_exp_loop

parse_number_success:
    mov x0, x22
    mov x1, #0
    b parse_number_done

parse_number_fail_pos:
    mov x0, x22
    mov x1, #1
    b parse_number_done

parse_number_fail_eof:
    mov x0, x22
    mov x1, #1
    b parse_number_done

parse_number_fail_at_pos:
    mov x0, x22
    mov x1, #1
    b parse_number_done

parse_number_invalid:
    mov x0, x21
    mov x1, #1

parse_number_done:
    ldp x29, x30, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
