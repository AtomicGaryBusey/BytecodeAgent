.text
.arch armv8-a

.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x23, x0
    cbz x23, done_input

    mov x24, x1
    mov x25, x23

process_loop:
    cbz x25, read_loop

    ldrb w26, [x24], #1
    add x21, x21, #1

    cmp w26, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    cmp w26, #0x20
    b.eq whitespace
    cmp w26, #0x09
    b.eq whitespace
    cmp w26, #0x0A
    b.eq whitespace
    cmp w26, #0x0B
    b.eq whitespace
    cmp w26, #0x0C
    b.eq whitespace
    cmp w26, #0x0D
    b.eq whitespace

non_whitespace:
    cbz x22, start_word
    b end_byte

start_word:
    add x20, x20, #1
    mov x22, #1
    b end_byte

whitespace:
    mov x22, #0

end_byte:
    sub x25, x25, #1
    b process_loop

done_input:
    mov x0, #1
    mov x1, x19
    bl write_dec

    mov x0, #1
    mov x1, #0x20
    bl write_byte

    mov x0, #1
    mov x1, x20
    bl write_dec

    mov x0, #1
    mov x1, #0x20
    bl write_byte

    mov x0, #1
    mov x1, x21
    bl write_dec

    mov x0, #1
    mov x1, #0x0A
    bl write_byte

    mov x0, #0
    mov x8, #93
    svc #0

write_dec:
    mov x9, x1
    mov x10, x0

    adrp x11, decbuf
    add x11, x11, :lo12:decbuf
    add x11, x11, #31

    mov x12, #0
    mov x13, #10

    cbz x9, zero_digit

digit_loop:
    cmp x9, #0
    b.eq digit_done

    udiv x14, x9, x13
    msub x15, x14, x13, x9
    add x15, x15, #48
    strb w15, [x11], #-1

    mov x9, x14
    add x12, x12, #1
    b digit_loop

zero_digit:
    mov w15, #48
    strb w15, [x11], #-1
    mov x12, #1

digit_done:
    add x1, x11, #1
    mov x2, x12
    mov x0, x10

write_dec_loop:
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt write_dec_fail
    cmp x0, x2
    b.eq write_dec_done
    sub x2, x2, x0
    add x1, x1, x0
    mov x0, x10
    b write_dec_loop

write_dec_fail:
    mov x0, #1
    mov x8, #93
    svc #0

write_dec_done:
    ret

write_byte:
    mov x9, x0

    adrp x10, tmpchar
    add x10, x10, :lo12:tmpchar
    strb w1, [x10]

    mov x0, x9
    mov x1, x10
    mov x2, #1

write_byte_loop:
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt write_byte_fail
    cmp x0, #1
    b.eq write_byte_done
    mov x0, x9
    b write_byte_loop

write_byte_fail:
    mov x0, #1
    mov x8, #93
    svc #0

write_byte_done:
    ret

.bss
    .align 4
input_buf:
    .space 65536

    .align 4
decbuf:
    .space 64

    .align 4
tmpchar:
    .space 1
