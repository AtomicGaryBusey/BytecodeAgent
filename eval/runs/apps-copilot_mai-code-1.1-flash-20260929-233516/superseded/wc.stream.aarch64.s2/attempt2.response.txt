_start:
    sub sp, sp, #4096

    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf

read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    svc #0

    cmp x0, #0
    b.lt error_exit
    beq done_reads

    mov x3, x0
    mov x4, xzr

process_loop:
    cmp x4, x3
    b.ge read_loop

    ldrb w5, [x23, x4]
    add x4, x4, #1
    add x21, x21, #1

    cmp w5, #0x20
    beq whitespace
    cmp w5, #0x09
    beq whitespace
    cmp w5, #0x0A
    beq newline
    cmp w5, #0x0B
    beq whitespace
    cmp w5, #0x0C
    beq whitespace
    cmp w5, #0x0D
    beq whitespace
    b not_whitespace

newline:
    add x19, x19, #1

whitespace:
    cmp x22, #0
    b.eq 1f
    add x20, x20, #1
    mov x22, xzr
1:
    b process_loop

not_whitespace:
    cmp x22, #0
    b.ne process_loop
    mov x22, #1
    b process_loop

done_reads:
    cmp x22, #0
    b.eq format_output
    add x20, x20, #1

format_output:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x22, x23

    mov x0, x19
    mov x1, x22
    bl print_dec
    mov x22, x1
    mov w9, #0x20
    strb w9, [x22], #1

    mov x0, x20
    mov x1, x22
    bl print_dec
    mov x22, x1
    mov w9, #0x20
    strb w9, [x22], #1

    mov x0, x21
    mov x1, x22
    bl print_dec
    mov x22, x1
    mov w9, #0x0A
    strb w9, [x22], #1

    mov x0, #1
    mov x1, x23
    sub x2, x22, x23
    bl write_all

    mov x8, #93
    mov x0, #0
    svc #0

error_exit:
    mov x0, #1
    mov x8, #93
    svc #0

print_dec:

    mov x2, sp
    mov x3, xzr

    cmp x0, #0
    b.ne 1f
    mov w4, #0x30
    strb w4, [x2]
    mov x3, #1
    b 2f

    b.eq 2f
    udiv x5, x0, #10
    msub x4, x5, #10, x0
    add x4, x4, #48
    strb w4, [x2, x3]
    add x3, x3, #1
    mov x0, x5
    b 1b

    mov x4, x3

    cmp x4, #0
    b.eq 4f
    sub x4, x4, #1
    ldrb w5, [x2, x4]
    strb w5, [x1], #1
    b 3b

    add sp, sp, #16
    mov x0, x3
    ret

write_all:
    mov x3, x0
    mov x4, x1
    mov x5, x2

    b.eq 2f

    mov x8, #64
    mov x0, x3
    mov x1, x4
    mov x2, x5

    cmp x0, #0
