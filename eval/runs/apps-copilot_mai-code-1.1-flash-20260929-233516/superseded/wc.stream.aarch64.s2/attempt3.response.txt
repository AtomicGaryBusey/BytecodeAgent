.text
.globl _start

_start:
    bl wc_main
    mov x8, #93
    mov x0, #0
    svc #0

wc_main:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    movz x2, #1, lsl #16
    svc #0
    cmp x0, #0
    b.lt error_exit
    beq done_input

    mov x23, x0
    mov x24, #0

process_loop:
    cmp x24, x23
    b.ge read_loop

    ldrb w25, [x1, x24]
    add x24, x24, #1
    add x21, x21, #1

    cmp w25, #0x0A
    b.eq newline_case
    cmp w25, #0x20
    b.eq whitespace_case
    cmp w25, #0x09
    b.eq whitespace_case
    cmp w25, #0x0B
    b.eq whitespace_case
    cmp w25, #0x0C
    b.eq whitespace_case
    cmp w25, #0x0D
    b.eq whitespace_case
    b non_whitespace_case

newline_case:
    add x19, x19, #1

whitespace_case:
    cmp x22, #0
    b.eq process_loop
    add x20, x20, #1
    mov x22, #0
    b process_loop

non_whitespace_case:
    cmp x22, #0
    b.ne process_loop
    mov x22, #1
    b process_loop

done_input:
    cmp x22, #0
    b.eq print_counts
    add x20, x20, #1

print_counts:
    mov x0, x19
    bl print_uint
    mov x0, #0x20
    bl write_char
    mov x0, x20
    bl print_uint
    mov x0, #0x20
    bl write_char
    mov x0, x21
    bl print_uint
    mov x0, #0x0A
    bl write_char

    ldp x29, x30, [sp], #16
    ret

error_exit:
    mov x0, #1
    mov x8, #93
    svc #0

print_uint:
    sub sp, sp, #32
    mov x9, sp
    add x9, x9, #32
    mov x12, x9
    mov x10, x0
    mov x11, #10
    cmp x10, #0
    b.ne 1f
    sub x9, x9, #1
    mov w13, #48
    strb w13, [x9]
    b 2f
1:
    cmp x10, #0
    b.eq 2f
    udiv x7, x10, x11
    mul x8, x7, x11
    sub x13, x10, x8
    add w13, w13, #48
    sub x9, x9, #1
    strb w13, [x9]
    mov x10, x7
    b 1b
2:
    mov x0, #1
    mov x1, x9
    sub x2, x12, x9
    mov x8, #64
    svc #0
    add sp, sp, #32
    ret

write_char:
    sub sp, sp, #16
    strb w0, [sp]
    mov x0, #1
    mov x1, sp
    mov x2, #1
    mov x8, #64
    svc #0
    add sp, sp, #16
    ret

.section .bss
.align 4
input_buf:
    .zero 65536
