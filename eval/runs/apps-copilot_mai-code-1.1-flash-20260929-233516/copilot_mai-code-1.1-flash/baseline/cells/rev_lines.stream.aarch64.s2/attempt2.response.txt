.text
.globl _start

_start:
    bl main
    mov x0, #0
    mov x8, #93
    svc #0

main:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x19, inbuf
    add x19, x19, :lo12:inbuf

    adrp x20, tmpbuf
    add x20, x20, :lo12:tmpbuf

    mov x21, #0
    movz x22, #1, lsl #16

read_loop:
    cmp x21, x22
    b.ge read_done
    mov x0, #0
    add x1, x19, x21
    sub x2, x22, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    mov x23, #0
    cmp x21, #0
    b.eq main_exit

process_loop:
    cmp x23, x21
    b.ge main_exit
    mov x24, x23
scan_line:
    cmp x24, x21
    b.ge final_line
    add x25, x19, x24
    ldrb w26, [x25]
    cmp w26, #10
    b.eq newline_found
    add x24, x24, #1
    b scan_line

newline_found:
    sub x25, x24, x23
    mov x0, x23
    mov x1, x25
    mov x2, x19
    mov x3, x20
    bl emit_line
    add x23, x24, #1
    b process_loop

final_line:
    sub x25, x21, x23
    cmp x25, #0
    b.eq main_exit
    mov x0, x23
    mov x1, x25
    mov x2, x19
    mov x3, x20
    bl emit_line

main_exit:
    ldp x29, x30, [sp], #16
    ret

emit_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // x0 = start index in input
    // x1 = line length
    // x2 = input buffer base
    // x3 = output buffer base
    cbz x1, emit_empty

    mov x4, #0
reverse_loop:
    cmp x4, x1
    b.ge reverse_done
    add x5, x0, x1
    sub x5, x5, #1
    sub x5, x5, x4
    add x5, x2, x5
    ldrb w6, [x5]
    add x7, x3, x4
    strb w6, [x7]
    add x4, x4, #1
    b reverse_loop

reverse_done:
    add x5, x3, x1
    mov w6, #10
    strb w6, [x5]

    mov x9, x1
    mov x0, #1
    mov x1, x3
    add x2, x9, #1
    bl sys_write

    ldp x29, x30, [sp], #16
    ret

emit_empty:
    mov w6, #10
    strb w6, [x3]

    mov x0, #1
    mov x1, x3
    mov x2, #1
    bl sys_write

    ldp x29, x30, [sp], #16
    ret

sys_write:
    mov x8, #64
    svc #0
    ret

.section .bss
.align 4
inbuf:
    .space 65536
tmpbuf:
    .space 131072
