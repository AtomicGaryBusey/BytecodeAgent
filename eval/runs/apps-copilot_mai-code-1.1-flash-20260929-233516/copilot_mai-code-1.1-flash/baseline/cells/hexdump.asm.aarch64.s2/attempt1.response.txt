.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
.globl _start
_start:
    mov x19, #0

read_loop:
    cmp x19, #65536
    b.hs read_done

    mov x8, #63
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    svc #0
    mov x20, x0

    cmp x20, #0
    b.lt read_error
    b.eq read_done

    add x19, x19, x20
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    mov x21, #0

row_loop:
    cmp x21, x19
    b.ge print_total

    mov x22, #16
    sub x23, x19, x21
    cmp x23, x22
    csel x22, x23, x22, lo

    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    add x24, x24, x21

    adrp x9, rowbuf
    add x9, x9, :lo12:rowbuf
    mov x10, #0

offset_digits:
    mov x11, #28

offset_loop:
    cmp x11, #0
    b.lt offset_done

    lsr x12, x21, x11
    and x12, x12, #0xf

    cmp x12, #10
    b.ge offset_letter
    add x12, x12, #'0'
    b offset_store

offset_letter:
    sub x12, x12, #10
    add x12, x12, #'a'

offset_store:
    strb w12, [x9, x10]
    add x10, x10, #1
    sub x11, x11, #4
    b offset_loop

offset_done:
    mov w12, #' '
    strb w12, [x9, x10]
    add x10, x10, #1
    strb w12, [x9, x10]
    add x10, x10, #1

    mov x11, #0

byte_print_loop:
    cmp x11, #16
    b.ge ascii_prefix

    cmp x11, x22
    b.ge blank_byte

    add x13, x24, x11
    ldrb w12, [x13]
    b emit_byte

blank_byte:
    mov w12, #' '
    strb w12, [x9, x10]
    add x10, x10, #1
    strb w12, [x9, x10]
    add x10, x10, #1
    strb w12, [x9, x10]
    add x10, x10, #1
    b slot_after

emit_byte:
    lsr w13, w12, #4
    cmp w13, #10
    b.ge hi_letter
    add w13, w13, #'0'
    b hi_store

hi_letter:
    sub w13, w13, #10
    add w13, w13, #'a'

hi_store:
    strb w13, [x9, x10]
    add x10, x10, #1

    and w13, w12, #0xf
    cmp w13, #10
    b.ge lo_letter
    add w13, w13, #'0'
    b lo_store

lo_letter:
    sub w13, w13, #10
    add w13, w13, #'a'

lo_store:
    strb w13, [x9, x10]
    add x10, x10, #1

    mov w13, #' '
    strb w13, [x9, x10]
    add x10, x10, #1

slot_after:
    cmp x11, #7
    b.ne slot_next

    mov w13, #' '
    strb w13, [x9, x10]
    add x10, x10, #1

slot_next:
    add x11, x11, #1
    b byte_print_loop

ascii_prefix:
    mov w12, #' '
    strb w12, [x9, x10]
    add x10, x10, #1
    mov w12, #'|'
    strb w12, [x9, x10]
    add x10, x10, #1

    mov x11, #0

ascii_loop:
    cmp x11, x22
    b.ge ascii_done

    add x13, x24, x11
    ldrb w12, [x13]

    cmp w12, #0x20
    b.lo ascii_dot
    cmp w12, #0x7e
    b.hi ascii_dot

    strb w12, [x9, x10]
    add x10, x10, #1
    b ascii_next

ascii_dot:
    mov w12, #'.'
    strb w12, [x9, x10]
    add x10, x10, #1

ascii_next:
    add x11, x11, #1
    b ascii_loop

ascii_done:
    mov w12, #'|'
    strb w12, [x9, x10]
    add x10, x10, #1
    mov w12, #10
    strb w12, [x9, x10]
    add x10, x10, #1

    mov x8, #64
    mov x0, #1
    adrp x1, rowbuf
    add x1, x1, :lo12:rowbuf
    mov x2, x10
    svc #0

    add x21, x21, x22
    b row_loop

print_total:
    adrp x9, rowbuf
    add x9, x9, :lo12:rowbuf
    mov x10, #0
    mov x11, #28

print_total_loop:
    cmp x11, #0
    b.lt print_total_done

    lsr x12, x19, x11
    and x12, x12, #0xf

    cmp x12, #10
    b.ge total_letter
    add x12, x12, #'0'
    b total_store

total_letter:
    sub x12, x12, #10
    add x12, x12, #'a'

total_store:
    strb w12, [x9, x10]
    add x10, x10, #1
    sub x11, x11, #4
    b print_total_loop

print_total_done:
    mov w12, #10
    strb w12, [x9, x10]
    add x10, x10, #1

    mov x8, #64
    mov x0, #1
    adrp x1, rowbuf
    add x1, x1, :lo12:rowbuf
    mov x2, x10
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.balign 16
input_buf:
    .space 65536

.balign 16
rowbuf:
    .space 128
