.globl _start

.section .bss
.align 8
input_buf: .space 65536
numbuf: .space 32
input_len: .zero 8
input_pos: .zero 8
error_at: .zero 8

.section .rodata
ok_msg: .ascii "ok\n"
err_prefix: .ascii "error at "
newline_byte: .byte 10
true_text: .ascii "true"
false_text: .ascii "false"
null_text: .ascii "null"

.text

_start:
    call read_all
    call parse_document
    test rax, rax
    jz .ok
    call print_error
    mov edi, 1
    mov eax, 60
    syscall
.ok:
    call print_ok
    xor edi, edi
    mov eax, 60
    syscall

read_all:
    mov qword ptr [rip + input_len], 0
.read_loop:
    lea rsi, [rip + input_buf]
    mov rax, qword ptr [rip + input_len]
    lea rsi, [rsi + rax]
    mov rdi, 0
    mov rdx, 65536
    sub rdx, rax
    mov rax, 0
    syscall
    cmp rax, 0
    je .done
    mov rbx, qword ptr [rip + input_len]
    add rbx, rax
    cmp rbx, 65536
    jbe .store
    mov rbx, 65536
.store:
    mov qword ptr [rip + input_len], rbx
    cmp rbx, 65536
    je .done
    jmp .read_loop
.done:
    ret

parse_document:
    mov qword ptr [rip + input_pos], 0
    call skip_ws
    call parse_value
    test rax, rax
    jnz .fail
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    je .ok
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.ok:
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

skip_ws:
    mov rax, qword ptr [rip + input_pos]
    lea rsi, [rip + input_buf]
    mov rdx, qword ptr [rip + input_len]
.loop:
    cmp rax, rdx
    jae .done
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je .advance
    cmp cl, 0x09
    je .advance
    cmp cl, 0x0A
    je .advance
    cmp cl, 0x0D
    je .advance
    jmp .done
.advance:
    inc rax
    jmp .loop
.done:
    mov qword ptr [rip + input_pos], rax
    ret

parse_value:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '{'
    je .object
    cmp cl, '['
    je .array
    cmp cl, '"'
    je .string
    cmp cl, '-'
    je .number
    cmp cl, '0'
    jb .check_literal
    cmp cl, '9'
    jbe .number
.check_literal:
    cmp cl, 't'
    je .literal_true
    cmp cl, 'f'
    je .literal_false
    cmp cl, 'n'
    je .literal_null
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.object:
    call parse_object
    ret
.array:
    call parse_array
    ret
.string:
    call parse_string
    ret
.number:
    call parse_number
    ret
.literal_true:
    call parse_true
    ret
.literal_false:
    call parse_false
    ret
.literal_null:
    call parse_null
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

parse_object:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '{'
    jne .invalid
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '}'
    je .close
.member_loop:
    call parse_string
    test rax, rax
    jnz .fail
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ':'
    jne .invalid_current
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    call parse_value
    test rax, rax
    jnz .fail
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    jne .end_member
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    jmp .member_loop
.end_member:
    cmp cl, '}'
    jne .invalid_current
.close:
    inc rax
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.invalid:
    mov qword ptr [rip + error_at], qword ptr [rip + input_pos]
    mov rax, 1
    ret
.invalid_current:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.fail:
    ret

parse_array:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '['
    jne .invalid
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ']'
    je .close
.value_loop:
    call parse_value
    test rax, rax
    jnz .fail
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    jne .end_value
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    jmp .value_loop
.end_value:
    cmp cl, ']'
    jne .invalid_current
.close:
    inc rax
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.invalid:
    mov qword ptr [rip + error_at], qword ptr [rip + input_pos]
    mov rax, 1
    ret
.invalid_current:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.fail:
    ret

parse_string:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    jne .invalid
    inc rax
    mov qword ptr [rip + input_pos], rax
.loop:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je .done
    cmp cl, '\\'
    je .escape
    cmp cl, 0x20
    jb .invalid_control
    inc qword ptr [rip + input_pos]
    jmp .loop
.escape:
    inc qword ptr [rip + input_pos]
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae .incomplete
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je .advance
    cmp cl, '\\'
    je .advance
    cmp cl, '/'
    je .advance
    cmp cl, 'b'
    je .advance
    cmp cl, 'f'
    je .advance
    cmp cl, 'n'
    je .advance
    cmp cl, 'r'
    je .advance
    cmp cl, 't'
    je .advance
    cmp cl, 'u'
    je .unicode
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.advance:
    inc qword ptr [rip + input_pos]
    jmp .loop
.unicode:
    mov r8, qword ptr [rip + input_pos]
    inc r8
    mov r9, r8
    add r9, 4
    cmp r9, qword ptr [rip + input_len]
    ja .incomplete
    mov rcx, 0
.unicode_loop:
    cmp rcx, 4
    je .unicode_done
    mov rax, r8
    add rax, rcx
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .unicode_invalid
    cmp dl, '9'
    jbe .unicode_ok
    cmp dl, 'a'
    jb .unicode_invalid
    cmp dl, 'f'
    jbe .unicode_ok
    cmp dl, 'A'
    jb .unicode_invalid
    cmp dl, 'F'
    jbe .unicode_ok
    jmp .unicode_invalid
.unicode_ok:
    inc rcx
    jmp .unicode_loop
.unicode_invalid:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.unicode_done:
    mov rax, r8
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    jmp .loop
.done:
    inc qword ptr [rip + input_pos]
    xor rax, rax
    ret
.invalid_control:
    mov qword ptr [rip + error_at], qword ptr [rip + input_pos]
    mov rax, 1
    ret
.invalid:
    mov qword ptr [rip + error_at], qword ptr [rip + input_pos]
    mov rax, 1
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

parse_true:
    mov rax, qword ptr [rip + input_pos]
    lea rsi, [rip + true_text]
    lea r9, [rip + input_buf]
    mov rdx, qword ptr [rip + input_len]
    lea rbx, [rax + 4]
    cmp rbx, rdx
    ja .incomplete
    mov rcx, 0
.loop:
    cmp rcx, 4
    je .ok
    movzx r8d, byte ptr [rsi + rcx]
    mov r10b, byte ptr [r9 + rax + rcx]
    cmp r8b, r10b
    jne .mismatch
    inc rcx
    jmp .loop
.mismatch:
    mov qword ptr [rip + error_at], rax
    add qword ptr [rip + error_at], rcx
    mov rax, 1
    ret
.ok:
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

parse_false:
    mov rax, qword ptr [rip + input_pos]
    lea rsi, [rip + false_text]
    lea r9, [rip + input_buf]
    mov rdx, qword ptr [rip + input_len]
    lea rbx, [rax + 5]
    cmp rbx, rdx
    ja .incomplete
    mov rcx, 0
.loop:
    cmp rcx, 5
    je .ok
    movzx r8d, byte ptr [rsi + rcx]
    mov r10b, byte ptr [r9 + rax + rcx]
    cmp r8b, r10b
    jne .mismatch
    inc rcx
    jmp .loop
.mismatch:
    mov qword ptr [rip + error_at], rax
    add qword ptr [rip + error_at], rcx
    mov rax, 1
    ret
.ok:
    add rax, 5
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

parse_null:
    mov rax, qword ptr [rip + input_pos]
    lea rsi, [rip + null_text]
    lea r9, [rip + input_buf]
    mov rdx, qword ptr [rip + input_len]
    lea rbx, [rax + 4]
    cmp rbx, rdx
    ja .incomplete
    mov rcx, 0
.loop:
    cmp rcx, 4
    je .ok
    movzx r8d, byte ptr [rsi + rcx]
    mov r10b, byte ptr [r9 + rax + rcx]
    cmp r8b, r10b
    jne .mismatch
    inc rcx
    jmp .loop
.mismatch:
    mov qword ptr [rip + error_at], rax
    add qword ptr [rip + error_at], rcx
    mov rax, 1
    ret
.ok:
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

parse_number:
    mov rax, qword ptr [rip + input_pos]
    mov rsi, qword ptr [rip + input_len]
    lea rdi, [rip + input_buf]
    cmp rax, rsi
    jae .incomplete
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '-'
    jne .after_sign
    inc rax
    cmp rax, rsi
    jae .incomplete
.after_sign:
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    je .zero
    cmp cl, '1'
    jb .invalid
    cmp cl, '9'
    jbe .int_digits
    jmp .invalid
.zero:
    inc rax
    cmp rax, rsi
    jae .after_int
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .after_int
    cmp cl, '9'
    jbe .invalid_leading_zero
    jmp .after_int
.int_digits:
    inc rax
.loop_int:
    cmp rax, rsi
    jae .after_int
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .after_int
    cmp cl, '9'
    jbe .int_inc
    jmp .after_int
.int_inc:
    inc rax
    jmp .loop_int
.invalid_leading_zero:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.after_int:
    cmp rax, rsi
    jae .done_valid
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '.'
    jne .exp_check
    inc rax
    cmp rax, rsi
    jae .incomplete
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .invalid_after_dot
    cmp cl, '9'
    jbe .frac_digits
    jmp .invalid_after_dot
.frac_digits:
    inc rax
.loop_frac:
    cmp rax, rsi
    jae .exp_check
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .exp_check
    cmp cl, '9'
    jbe .frac_inc
    jmp .exp_check
.frac_inc:
    inc rax
    jmp .loop_frac
.invalid_after_dot:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.exp_check:
    cmp rax, rsi
    jae .done_valid
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
.done_valid:
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret
.exp:
    inc rax
    cmp rax, rsi
    jae .incomplete
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '+'
    je .exp_sign
    cmp cl, '-'
    je .exp_sign
    jmp .exp_digit_check
.exp_sign:
    inc rax
    cmp rax, rsi
    jae .incomplete
.exp_digit_check:
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .invalid_exp
    cmp cl, '9'
    jbe .exp_digits
    jmp .invalid_exp
.exp_digits:
    inc rax
.loop_exp:
    cmp rax, rsi
    jae .done_valid
    movzx ecx, byte ptr [rdi + rax]
    cmp cl, '0'
    jb .done_valid
    cmp cl, '9'
    jbe .exp_inc
    jmp .done_valid
.exp_inc:
    inc rax
    jmp .loop_exp
.invalid_exp:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.invalid:
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret
.incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + error_at], rax
    mov rax, 1
    ret

print_ok:
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    mov rax, 1
    syscall
    ret

print_error:
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    mov rax, 1
    syscall
    lea rdi, [rip + numbuf + 31]
    mov rax, qword ptr [rip + error_at]
    mov rcx, 10
.loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .loop
    mov rsi, rdi
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    ret
