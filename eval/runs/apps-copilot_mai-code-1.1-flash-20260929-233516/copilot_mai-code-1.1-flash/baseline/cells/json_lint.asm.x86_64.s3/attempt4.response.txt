.globl _start

.section .bss
.align 8
input_buf: .zero 65536
input_len: .zero 8
input_pos: .zero 8
err_at: .zero 8
num_buf: .zero 32

.section .rodata
ok_msg: .ascii "ok\n"
err_prefix: .ascii "error at "
newline_byte: .byte 10
true_literal: .ascii "true"
false_literal: .ascii "false"
null_literal: .ascii "null"

.text

_start:
    call read_all
    call parse_document
    test rax, rax
    jnz fail_case
    call print_ok
    xor edi, edi
    mov eax, 60
    syscall

fail_case:
    call print_error
    mov edi, 1
    mov eax, 60
    syscall

read_all:
    mov qword ptr [rip + input_len], 0
read_loop:
    lea rsi, [rip + input_buf]
    mov rax, qword ptr [rip + input_len]
    lea rsi, [rsi + rax]
    mov edi, 0
    mov edx, 65536
    sub edx, eax
    xor eax, eax
    syscall
    cmp rax, 0
    je read_done
    mov r8, qword ptr [rip + input_len]
    add r8, rax
    cmp r8, 65536
    jbe read_store
    mov r8, 65536
read_store:
    mov qword ptr [rip + input_len], r8
    cmp r8, 65536
    je read_done
    jmp read_loop
read_done:
    ret

parse_document:
    mov qword ptr [rip + input_pos], 0
    call skip_ws
    call parse_value
    test rax, rax
    jnz parse_document_fail
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jne parse_document_extra
    xor rax, rax
    ret

parse_document_fail:
    mov rax, 1
    ret

parse_document_extra:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

skip_ws:
    mov rax, qword ptr [rip + input_pos]
skip_ws_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jae skip_ws_done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je skip_ws_advance
    cmp cl, 0x09
    je skip_ws_advance
    cmp cl, 0x0A
    je skip_ws_advance
    cmp cl, 0x0D
    je skip_ws_advance
    jmp skip_ws_done
skip_ws_advance:
    inc rax
    jmp skip_ws_loop
skip_ws_done:
    mov qword ptr [rip + input_pos], rax
    ret

parse_value:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jae parse_value_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '{'
    je parse_object
    cmp cl, '['
    je parse_array
    cmp cl, '"'
    je parse_string
    cmp cl, '-'
    je parse_number
    cmp cl, '0'
    jb parse_value_literal
    cmp cl, '9'
    jbe parse_number
parse_value_literal:
    cmp cl, 't'
    je parse_true
    cmp cl, 'f'
    je parse_false
    cmp cl, 'n'
    je parse_null
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_value_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_object:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jae parse_object_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '{'
    jne parse_object_bad
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '}'
    je parse_object_done

parse_object_loop:
    call parse_string
    test rax, rax
    jnz parse_object_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ':'
    jne parse_object_colon_bad
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    call parse_value
    test rax, rax
    jnz parse_object_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    jne parse_object_check_end
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '}'
    je parse_object_trailing_comma
    jmp parse_object_loop

parse_object_check_end:
    cmp cl, '}'
    jne parse_object_bad
parse_object_done:
    inc rax
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_object_bad:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_object_colon_bad:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_object_trailing_comma:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_object_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_object_return:
    ret

parse_array:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jae parse_array_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '['
    jne parse_array_bad
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ']'
    je parse_array_done

parse_array_loop:
    call parse_value
    test rax, rax
    jnz parse_array_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    jne parse_array_check_end
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ']'
    je parse_array_trailing_comma
    jmp parse_array_loop

parse_array_check_end:
    cmp cl, ']'
    jne parse_array_broken
parse_array_done:
    inc rax
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_array_bad:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_array_broken:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_array_trailing_comma:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_array_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_array_return:
    ret

parse_string:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp rax, rdx
    jae parse_string_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    jne parse_string_start_bad
    inc rax
    mov qword ptr [rip + input_pos], rax

parse_string_loop:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je parse_string_done
    cmp cl, '\\'
    je parse_string_escape
    cmp cl, 0x20
    jb parse_string_control
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp parse_string_loop

parse_string_escape:
    inc rax
    mov qword ptr [rip + input_pos], rax
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je parse_string_escape_ok
    cmp cl, '\\'
    je parse_string_escape_ok
    cmp cl, '/'
    je parse_string_escape_ok
    cmp cl, 'b'
    je parse_string_escape_ok
    cmp cl, 'f'
    je parse_string_escape_ok
    cmp cl, 'n'
    je parse_string_escape_ok
    cmp cl, 'r'
    je parse_string_escape_ok
    cmp cl, 't'
    je parse_string_escape_ok
    cmp cl, 'u'
    je parse_string_unicode
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_string_escape_ok:
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp parse_string_loop

parse_string_unicode:
    mov r8, rax
    inc r8
    mov rcx, 0
    mov r9, r8
    add r9, 4
    cmp r9, qword ptr [rip + input_len]
    ja parse_string_eof
parse_string_unicode_loop:
    cmp rcx, 4
    je parse_string_unicode_ok
    mov rdx, r8
    add rdx, rcx
    movzx eax, byte ptr [rsi + rdx]
    cmp al, '0'
    jb parse_string_unicode_bad
    cmp al, '9'
    jbe parse_string_unicode_next
    cmp al, 'a'
    jb parse_string_unicode_bad
    cmp al, 'f'
    jbe parse_string_unicode_next
    cmp al, 'A'
    jb parse_string_unicode_bad
    cmp al, 'F'
    jbe parse_string_unicode_next
    jmp parse_string_unicode_bad

parse_string_unicode_next:
    inc rcx
    jmp parse_string_unicode_loop

parse_string_unicode_ok:
    mov rax, r8
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    jmp parse_string_loop

parse_string_unicode_bad:
    mov qword ptr [rip + err_at], rdx
    mov rax, 1
    ret

parse_string_done:
    inc rax
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_string_control:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_string_start_bad:
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_string_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_true:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    lea rsi, [rip + true_literal]
    lea rdi, [rip + input_buf]
    mov r8, rax
    add r8, 4
    cmp r8, rdx
    ja parse_true_eof
    xor rcx, rcx
parse_true_loop:
    cmp rcx, 4
    je parse_true_ok
    mov r9b, byte ptr [rsi + rcx]
    mov r10b, byte ptr [rdi + rax + rcx]
    cmp r9b, r10b
    jne parse_true_mismatch
    inc rcx
    jmp parse_true_loop

parse_true_mismatch:
    mov qword ptr [rip + err_at], rax
    add qword ptr [rip + err_at], rcx
    mov rax, 1
    ret

parse_true_ok:
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_true_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_false:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    lea rsi, [rip + false_literal]
    lea rdi, [rip + input_buf]
    mov r8, rax
    add r8, 5
    cmp r8, rdx
    ja parse_false_eof
    xor rcx, rcx
parse_false_loop:
    cmp rcx, 5
    je parse_false_ok
    mov r9b, byte ptr [rsi + rcx]
    mov r10b, byte ptr [rdi + rax + rcx]
    cmp r9b, r10b
    jne parse_false_mismatch
    inc rcx
    jmp parse_false_loop

parse_false_mismatch:
    mov qword ptr [rip + err_at], rax
    add qword ptr [rip + err_at], rcx
    mov rax, 1
    ret

parse_false_ok:
    add rax, 5
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_false_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_null:
    mov rax, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    lea rsi, [rip + null_literal]
    lea rdi, [rip + input_buf]
    mov r8, rax
    add r8, 4
    cmp r8, rdx
    ja parse_null_eof
    xor rcx, rcx
parse_null_loop:
    cmp rcx, 4
    je parse_null_ok
    mov r9b, byte ptr [rsi + rcx]
    mov r10b, byte ptr [rdi + rax + rcx]
    cmp r9b, r10b
    jne parse_null_mismatch
    inc rcx
    jmp parse_null_loop

parse_null_mismatch:
    mov qword ptr [rip + err_at], rax
    add qword ptr [rip + err_at], rcx
    mov rax, 1
    ret

parse_null_ok:
    add rax, 4
    mov qword ptr [rip + input_pos], rax
    xor rax, rax
    ret

parse_null_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

parse_number:
    mov r8, qword ptr [rip + input_pos]
    mov rdx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp r8, rdx
    jae parse_number_eof
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '-'
    jne parse_number_int
    inc r8
    cmp r8, rdx
    jae parse_number_eof
    movzx ecx, byte ptr [rsi + r8]

parse_number_int:
    cmp cl, '0'
    je parse_number_zero
    cmp cl, '1'
    jb parse_number_bad
    cmp cl, '9'
    jbe parse_number_digits
    jmp parse_number_bad

parse_number_zero:
    inc r8
    cmp r8, rdx
    jae parse_number_after_int
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_after_int
    cmp cl, '9'
    jbe parse_number_bad
    jmp parse_number_after_int

parse_number_digits:
    inc r8
parse_number_digits_loop:
    cmp r8, rdx
    jae parse_number_after_int
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_after_int
    cmp cl, '9'
    jbe parse_number_digits_more
    jmp parse_number_after_int

parse_number_digits_more:
    inc r8
    jmp parse_number_digits_loop

parse_number_after_int:
    cmp r8, rdx
    jae parse_number_done
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '.'
    jne parse_number_exp_check
    inc r8
    cmp r8, rdx
    jae parse_number_eof
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_bad
    cmp cl, '9'
    jbe parse_number_frac_digits
    jmp parse_number_bad

parse_number_frac_digits:
    inc r8
parse_number_frac_loop:
    cmp r8, rdx
    jae parse_number_exp_check
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_exp_check
    cmp cl, '9'
    jbe parse_number_frac_more
    jmp parse_number_exp_check

parse_number_frac_more:
    inc r8
    jmp parse_number_frac_loop

parse_number_exp_check:
    cmp r8, rdx
    jae parse_number_done
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, 'e'
    je parse_number_exp
    cmp cl, 'E'
    je parse_number_exp
    jmp parse_number_done

parse_number_exp:
    inc r8
    cmp r8, rdx
    jae parse_number_eof
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '+'
    je parse_number_exp_sign
    cmp cl, '-'
    je parse_number_exp_sign
    jmp parse_number_exp_num

parse_number_exp_sign:
    inc r8
    cmp r8, rdx
    jae parse_number_eof

parse_number_exp_num:
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_bad
    cmp cl, '9'
    jbe parse_number_exp_digits
    jmp parse_number_bad

parse_number_exp_digits:
    inc r8
parse_number_exp_loop:
    cmp r8, rdx
    jae parse_number_done
    movzx ecx, byte ptr [rsi + r8]
    cmp cl, '0'
    jb parse_number_done
    cmp cl, '9'
    jbe parse_number_exp_more
    jmp parse_number_done

parse_number_exp_more:
    inc r8
    jmp parse_number_exp_loop

parse_number_done:
    mov qword ptr [rip + input_pos], r8
    xor rax, rax
    ret

parse_number_bad:
    mov qword ptr [rip + err_at], r8
    mov rax, 1
    ret

parse_number_eof:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + err_at], rax
    mov rax, 1
    ret

print_ok:
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov eax, 1
    syscall
    ret

print_error:
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    mov eax, 1
    syscall

    lea rdi, [rip + num_buf + 31]
    mov rax, qword ptr [rip + err_at]
    mov rcx, 10
print_error_digits:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz print_error_digits

    mov rsi, rdi
    lea rdx, [rip + num_buf + 32]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    lea rsi, [rip + newline_byte]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    ret
