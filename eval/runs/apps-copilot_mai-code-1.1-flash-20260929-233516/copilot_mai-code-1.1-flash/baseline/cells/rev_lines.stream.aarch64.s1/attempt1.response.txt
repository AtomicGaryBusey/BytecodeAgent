.text
.globl _start

_start:
    bl read_all
    mov x19, x0
    cmp x19, #0
    b.eq exit_zero

    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

    mov x21, #0
    mov x22, #0

scan_loop:
    cmp x22, x19
    b.hs final_line

    add x9, x20, x22
    ldrb w9, [x9]
    cmp w9, #10
    b.ne not_newline

    mov x0, x20
    mov x1, x21
    mov x2, x22
    bl reverse_segment

    mov x0, #1
    add x1, x20, x21
    sub x2, x22, x21
    bl write_all

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all

    add x21, x22, #1
    add x22, x22, #1
    b scan_loop

not_newline:
    add x22, x22, #1
    b scan_loop

final_line:
    cmp x21, x19
    b.hs exit_zero

    mov x0, x20
    mov x1, x21
    mov x2, x19
    bl reverse_segment

    mov x0, #1
    add x1, x20, x21
    sub x2, x19, x21
    bl write_all

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all

exit_zero:
    mov x8, #93
    mov x0, #0
    svc #0

_fail:
    mov x8, #93
    mov x0, #1
    svc #0

read_all:
    mov x9, #0
    adrp x10, input_buf
    add x10, x10, :lo12:input_buf

read_loop:
    cmp x9, #65536
    b.hs read_done

    mov x0, #0
    mov x1, x10
    add x1, x1, x9

    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x9
    mov x8, #63
    svc #0

    cmp x0, #0
    b.lt _fail
    beq read_done

    add x9, x9, x0
    b read_loop

read_done:
    mov x0, x9
    ret

write_all:
    mov x9, x0
    mov x10, x1
    mov x11, x2

write_loop:
    cmp x11, #0
    b.eq write_done

    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0

    cmp x0, #0
    b.le _fail

    sub x11, x11, x0
    add x10, x10, x0
    b write_loop

write_done:
    ret

reverse_segment:
    cmp x1, x2
    b.hs reverse_done

reverse_loop:
    sub x9, x2, x1
    cmp x9, #1
    b.ls reverse_done

    sub x10, x2, #1
    add x11, x0, x1
    add x12, x0, x10

    ldrb w13, [x11]
    ldrb w14, [x12]
    strb w14, [x11]
    strb w13, [x12]

    add x1, x1, #1
    sub x2, x2, #1
    b reverse_loop

reverse_done:
    ret

.section .rodata
newline:
    .byte 0x0A

.section .bss
.balign 8
input_buf:
    .space 65536
