.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ SYS_exit_group, 94

.equ STDIN, 0
.equ STDOUT, 1
.equ STDERR, 2

.equ CHR_SPACE, 32
.equ CHR_TAB, 9
.equ CHR_LF, 10
.equ CHR_VTAB, 11
.equ CHR_FF, 12
.equ CHR_CR, 13

.equ INPUT_BUF_SIZE, 4096
.equ OUTPUT_BUF_SIZE, 64

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, SYS_read
    mov x0, STDIN
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, INPUT_BUF_SIZE
    svc #0
    cmp x0, #0
    b.lt fail_exit
    b.eq done_read
    mov x23, x0
    add x21, x21, x23
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    add x25, x24, x23

process_chunk:
    ldrb w26, [x24], #1

    cmp w26, CHR_LF
    b.ne 1f
    add x19, x19, #1
1:
    cmp w26, CHR_SPACE
    b.eq 2f
    cmp w26, CHR_TAB
    b.eq 2f
    cmp w26, CHR_LF
    b.eq 2f
    cmp w26, CHR_VTAB
    b.eq 2f
    cmp w26, CHR_FF
    b.eq 2f
    cmp w26, CHR_CR
    b.eq 2f

    cbz x22, word_start
    b 3f

word_start:
    add x20, x20, #1
    mov x22, #1
    b 3f

2:
    mov x22, #0

3:
    cmp x24, x25
    b.ne process_chunk
    b read_loop

done_read:
    adrp x0, output_buf
    add x0, x0, :lo12:output_buf
    add x0, x0, OUTPUT_BUF_SIZE

    mov w1, CHR_LF
    bl emit_char_back
    mov x1, x21
    bl emit_unsigned_back
    mov w1, CHR_SPACE
    bl emit_char_back
    mov x1, x20
    bl emit_unsigned_back
    mov w1, CHR_SPACE
    bl emit_char_back
    mov x1, x19
    bl emit_unsigned_back

    mov x3, x0
    adrp x4, output_buf
    add x4, x4, :lo12:output_buf
    add x4, x4, OUTPUT_BUF_SIZE
    sub x2, x4, x3

write_loop:
    mov x8, SYS_write
    mov x0, STDOUT
    mov x1, x3
    svc #0
    cmp x0, #0
    b.le fail_exit
    sub x2, x2, x0
    add x3, x3, x0
    cbnz x2, write_loop
    b exit_ok

fail_exit:
    mov x8, SYS_exit
    mov x0, #1
    svc #0

exit_ok:
    mov x8, SYS_exit
    mov x0, #0
    svc #0

emit_char_back:
    sub x0, x0, #1
    strb w1, [x0]
    ret

emit_unsigned_back:
    mov x2, x1
    cbz x2, emit_zero_digit

1:
    mov x3, #10
    udiv x4, x2, x3
    msub x5, x4, x3, x2
    mov x2, x4
    add w5, w5, #48
    sub x0, x0, #1
    strb w5, [x0]
    cbnz x2, 1b
    ret

emit_zero_digit:
    sub x0, x0, #1
    mov w5, #48
    strb w5, [x0]
    ret

.section .bss
    .balign 4
input_buf:
    .space INPUT_BUF_SIZE
    .balign 4
output_buf:
    .space OUTPUT_BUF_SIZE
