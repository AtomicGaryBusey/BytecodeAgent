.globl _start

.text
_start:
    lea r12, [rip + stdin_buf]
    xor r13d, r13d

read_loop:
    mov rdx, 65536
    sub rdx, r13
    jbe read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    js error
    jz read_done
    add r13, rax
    cmp r13, 65536
    jae read_done
    jmp read_loop

read_done:
    mov r14d, r13d
    mov r15d, r13d
    xor ecx, ecx

scan_bang:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je bang_found
    inc rcx
    jmp scan_bang

bang_found:
    mov r14d, ecx
    lea r15, [rcx + 1]
    jmp validate

no_bang:
    mov r15d, r13d
    jmp validate

validate:
    lea rbx, [rip + bracket_stack]
    lea rcx, [rip + match_table]
    xor r8d, r8d
    xor esi, esi

validate_loop:
    cmp esi, r14d
    jae validate_done
    movzx eax, byte ptr [r12 + rsi]
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket
    inc esi
    jmp validate_loop

open_bracket:
    cmp r8d, 1000
    jae invalid
    mov eax, esi
    mov dword ptr [rbx + r8*4], eax
    inc r8d
    inc esi
    jmp validate_loop

close_bracket:
    test r8d, r8d
    jz invalid
    dec r8d
    mov eax, dword ptr [rbx + r8*4]
    mov edx, esi
    mov dword ptr [rcx + rax*4], edx
    mov dword ptr [rcx + rdx*4], eax
    inc esi
    jmp validate_loop

validate_done:
    test r8d, r8d
    jnz invalid

run:
    lea rbx, [rip + tape]
    xor r9d, r9d
    xor r10d, r10d
    mov r8d, r15d

run_loop:
    cmp r10d, r14d
    jae done
    movzx eax, byte ptr [r12 + r10]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    inc r10d
    jmp run_loop

cmd_gt:
    inc r9d
    inc r10d
    jmp run_loop

cmd_lt:
    dec r9d
    inc r10d
    jmp run_loop

cmd_plus:
    movzx eax, byte ptr [rbx + r9]
    inc al
    mov byte ptr [rbx + r9], al
    inc r10d
    jmp run_loop

cmd_minus:
    movzx eax, byte ptr [rbx + r9]
    dec al
    mov byte ptr [rbx + r9], al
    inc r10d
    jmp run_loop

cmd_dot:
    movzx eax, byte ptr [rbx + r9]
    mov byte ptr [rip + out_byte], al
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_byte]
    mov rdx, 1
    syscall
    inc r10d
    jmp run_loop

cmd_comma:
    cmp r8d, r13d
    jae comma_zero
    movzx eax, byte ptr [r12 + r8]
    inc r8d
    mov byte ptr [rbx + r9], al
    inc r10d
    jmp run_loop

comma_zero:
    mov byte ptr [rbx + r9], 0
    inc r10d
    jmp run_loop

cmd_open:
    movzx eax, byte ptr [rbx + r9]
    test al, al
    jnz open_continue
    lea rcx, [rip + match_table]
    mov eax, dword ptr [rcx + r10*4]
    mov r10d, eax
    inc r10d
    jmp run_loop

open_continue:
    inc r10d
    jmp run_loop

cmd_close:
    movzx eax, byte ptr [rbx + r9]
    test al, al
    jz close_continue
    lea rcx, [rip + match_table]
    mov eax, dword ptr [rcx + r10*4]
    mov r10d, eax
    inc r10d
    jmp run_loop

close_continue:
    inc r10d
    jmp run_loop

done:
    mov rax, 60
    xor edi, edi
    syscall

invalid:
error:
    mov rax, 60
    mov edi, 1
    syscall

.section .bss
.align 16
stdin_buf:
    .zero 65536

.align 16
bracket_stack:
    .zero 4000

.align 16
match_table:
    .zero 262144

.align 16
tape:
    .zero 30000

.align 16
out_byte:
    .zero 1
