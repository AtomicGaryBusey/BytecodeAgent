.globl _start

.text
_start:
    lea r12, [rip + stdin_buf]
    xor r13d, r13d
    lea rsi, [rip + stdin_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rdx, 4096
    syscall
    test rax, rax
    js error
    jz read_done
    add r13, rax
    cmp r13, 65536
    ja error
    add rsi, rax
    jmp read_loop

read_done:
    mov r11, r13
    xor ecx, ecx

scan_loop:
    cmp rcx, r11
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je bang_found
    inc rcx
    jmp scan_loop

bang_found:
    mov r13, rcx
    mov r10, r11
    sub r10, rcx
    dec r10
    lea r9, [r12 + rcx + 1]
    jmp validate

no_bang:
    mov r13, r11
    xor r10d, r10d
    lea r9, [r12 + r11]

validate:
    lea r14, [rip + match_table]
    lea r15, [rip + bracket_stack]
    xor r8d, r8d
    xor esi, esi

validate_loop:
    cmp rsi, r13
    jae validate_done
    mov al, byte ptr [r12 + rsi]
    cmp al, '['
    je open_br
    cmp al, ']'
    je close_br
    inc rsi
    jmp validate_loop

open_br:
    cmp r8, 1000
    jae invalid
    mov eax, esi
    mov dword ptr [r15 + r8*4], eax
    inc r8
    inc rsi
    jmp validate_loop

close_br:
    test r8, r8
    jz invalid
    dec r8
    mov eax, dword ptr [r15 + r8*4]
    mov ecx, esi
    mov dword ptr [r14 + rax*4], ecx
    mov dword ptr [r14 + rcx*4], eax
    inc rsi
    jmp validate_loop

validate_done:
    test r8, r8
    jnz invalid

run:
    lea r10, [r9 + r10]
    lea r14, [rip + match_table]
    lea r11, [rip + tape]
    xor rbp, rbp
    xor r8d, r8d

run_loop:
    cmp rbp, r13
    jae done
    movzx eax, byte ptr [r12 + rbp]

    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close

    inc rbp
    jmp run_loop

cmd_gt:
    inc r8
    inc rbp
    jmp run_loop

cmd_lt:
    dec r8
    inc rbp
    jmp run_loop

cmd_plus:
    mov dl, byte ptr [r11 + r8]
    inc dl
    mov byte ptr [r11 + r8], dl
    inc rbp
    jmp run_loop

cmd_minus:
    mov dl, byte ptr [r11 + r8]
    dec dl
    mov byte ptr [r11 + r8], dl
    inc rbp
    jmp run_loop

cmd_dot:
    mov dl, byte ptr [r11 + r8]
    mov byte ptr [rip + out_byte], dl
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_byte]
    mov rdx, 1
    syscall
    inc rbp
    jmp run_loop

cmd_comma:
    cmp r9, r10
    jae comma_zero
    mov dl, byte ptr [r9]
    inc r9
    mov byte ptr [r11 + r8], dl
    inc rbp
    jmp run_loop

comma_zero:
    mov byte ptr [r11 + r8], 0
    inc rbp
    jmp run_loop

cmd_open:
    movzx edx, byte ptr [r11 + r8]
    test dl, dl
    jnz open_continue
    mov eax, dword ptr [r14 + rbp*4]
    mov rbp, rax
    inc rbp
    jmp run_loop

open_continue:
    inc rbp
    jmp run_loop

cmd_close:
    movzx edx, byte ptr [r11 + r8]
    test dl, dl
    jz close_continue
    mov eax, dword ptr [r14 + rbp*4]
    mov rbp, rax
    inc rbp
    jmp run_loop

close_continue:
    inc rbp
    jmp run_loop

done:
    mov rax, 60
    xor edi, edi
    syscall

invalid:
    mov rax, 60
    mov edi, 1
    syscall

error:
    mov rax, 60
    mov edi, 1
    syscall

.section .bss
.align 16
stdin_buf:
    .zero 65536

.align 16
bracket_stack:
    .zero 1000 * 4

.align 16
match_table:
    .zero 65536 * 4

.align 16
tape:
    .zero 30000

.align 16
out_byte:
    .zero 1
