.globl _start

.equ INPUT_MAX, 49152
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
inbuf:
    .space INPUT_MAX
outbuf:
    .space 70000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r13
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    add r13, rax
    cmp r13, INPUT_MAX
    jb read_loop
read_done:
    cmp r13, 0
    je exit_now
    lea r12, [rip + inbuf]
    mov rbx, r13
    lea rbp, [rip + outbuf]
    mov r14, rbp
    xor r15, r15
    lea r11, [rip + alphabet]

encode_loop:
    cmp rbx, 3
    jb encode_final

    movzx rax, byte ptr [r12]
    movzx rdx, byte ptr [r12 + 1]
    movzx rcx, byte ptr [r12 + 2]
    add r12, 3
    sub rbx, 3

    mov r8, rax
    shr r8, 2
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rax
    and r8, 3
    shl r8, 4
    mov r9, rdx
    shr r9, 4
    or r8, r9
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rdx
    and r8, 15
    shl r8, 2
    mov r9, rcx
    shr r9, 6
    or r8, r9
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rcx
    and r8, 63
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    cmp rbx, 0
    jne encode_loop
    jmp encode_done

encode_final:
    cmp rbx, 0
    je encode_done
    cmp rbx, 1
    je final_one

    movzx rax, byte ptr [r12]
    movzx rdx, byte ptr [r12 + 1]

    mov r8, rax
    shr r8, 2
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rax
    and r8, 3
    shl r8, 4
    mov r9, rdx
    shr r9, 4
    or r8, r9
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rdx
    and r8, 15
    shl r8, 2
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov al, '='
    call emit_char
    jmp encode_done

final_one:
    movzx rax, byte ptr [r12]

    mov r8, rax
    shr r8, 2
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov r8, rax
    and r8, 3
    shl r8, 4
    movzx r8d, byte ptr [r11 + r8]
    mov al, r8b
    call emit_char

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
    jmp encode_done

encode_done:
    cmp r15, 0
    jne add_final_newline
    jmp write_output

add_final_newline:
    mov byte ptr [r14], 10
    inc r14

write_output:
    mov rdx, r14
    sub rdx, rbp
    mov rax, 1
    mov rdi, 1
    mov rsi, rbp
    syscall
    jmp exit_now

emit_char:
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne emit_ret
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
emit_ret:
    ret

exit_now:
    mov rax, 60
    xor rdi, rdi
    syscall
