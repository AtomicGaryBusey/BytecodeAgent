.globl _start

.equ INPUT_MAX, 49152
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
inbuf:
    .zero INPUT_MAX
outbuf:
    .zero 70000

.text

_start:
    lea r12, [rip + inbuf]
    xor r13, r13

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r13
    syscall
    cmp rax, 0
    je read_done
    add r13, rax
    add r12, rax
    cmp r13, INPUT_MAX
    jb read_loop

read_done:
    lea r10, [rip + alphabet]
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r14, r14
    xor r8, r8
    mov rbx, r13
    test rbx, rbx
    jz output

encode_loop:
    mov r9, rbx
    sub r9, r8
    cmp r9, 3
    jb encode_tail

    movzx eax, byte ptr [rsi + r8]
    movzx ecx, byte ptr [rsi + r8 + 1]
    movzx edx, byte ptr [rsi + r8 + 2]
    add r8, 3

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
1:
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r11d, ecx
    shr r11d, 4
    or r9d, r11d
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 2f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
2:
    mov r9d, ecx
    and r9d, 15
    shl r9d, 2
    mov r11d, edx
    shr r11d, 6
    or r9d, r11d
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 3f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
3:
    mov r9d, edx
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne encode_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
    jmp encode_loop

encode_tail:
    mov r9, rbx
    sub r9, r8
    cmp r9, 0
    je finalize
    cmp r9, 1
    je encode_one

    movzx eax, byte ptr [rsi + r8]
    movzx ecx, byte ptr [rsi + r8 + 1]

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 4f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
4:
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r11d, ecx
    shr r11d, 4
    or r9d, r11d
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 5f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
5:
    mov r9d, ecx
    and r9d, 15
    shl r9d, 2
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 6f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
6:
    mov byte ptr [rdi], 61
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne finalize
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
    jmp finalize

encode_one:
    movzx eax, byte ptr [rsi + r8]

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 7f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
7:
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 8f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
8:
    mov byte ptr [rdi], 61
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne 9f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
9:
    mov byte ptr [rdi], 61
    inc rdi
    inc r14
    cmp r14, LINE_LEN
    jne finalize
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
    jmp finalize

finalize:
    cmp r14, 0
    je output
    mov byte ptr [rdi], 10
    inc rdi

output:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1

write_all:
    syscall
    cmp rax, 0
    jl exit_now
    cmp rax, rdx
    je exit_now
    sub rdx, rax
    add rsi, rax
    jmp write_all

exit_now:
    mov rax, 60
    xor rdi, rdi
    syscall
