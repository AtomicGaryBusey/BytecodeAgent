.text
.globl _start

_start:
    // Read the entire stdin into input_buf.
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    mov x9, x0
    cmp x9, #0
    beq read_done
    add x19, x19, x9
    cmp x19, #65536
    b.lt read_loop

read_done:
    // Find the first ! byte in the input; if absent, whole input is program.
    mov x21, #0
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

find_bang:
    cmp x21, x19
    b.hs no_bang
    add x9, x22, x21
    ldrb w10, [x9]
    cmp w10, #0x21
    beq bang_found
    add x21, x21, #1
    b find_bang

bang_found:
    mov x20, x21
    add x21, x21, #1
    mov x24, x19
    sub x24, x24, x21
    adrp x25, input_buf
    add x25, x25, :lo12:input_buf
    add x25, x25, x21
    b bracket_setup

no_bang:
    mov x20, x19
    mov x24, #0
    adrp x25, input_buf
    add x25, x25, :lo12:input_buf
    add x25, x25, x19

bracket_setup:
    // Validate brackets and build match_table.
    adrp x26, match_table
    add x26, x26, :lo12:match_table
    adrp x27, bracket_stack
    add x27, x27, :lo12:bracket_stack
    mov x28, #0
    mov x29, #0

balance_loop:
    cmp x29, x20
    b.ge balance_end
    add x9, x22, x29
    ldrb w10, [x9]
    cmp w10, #0x5b
    b.ne balance_not_open
    add x11, x27, x28, lsl #1
    strh w29, [x11]
    add x28, x28, #1
    cmp x28, #1000
    b.hs exit_error
    b balance_inc

balance_not_open:
    cmp w10, #0x5d
    b.ne balance_inc
    cmp x28, #0
    b.eq exit_error
    sub x28, x28, #1
    add x11, x27, x28, lsl #1
    ldrh w9, [x11]
    add x10, x26, x9, lsl #1
    strh w29, [x10]
    add x10, x26, x29, lsl #1
    strh w9, [x10]
    b balance_inc

balance_inc:
    add x29, x29, #1
    b balance_loop

balance_end:
    cmp x28, #0
    b.ne exit_error

    // Interpreter state:
    // x19 = program_len
    // x20 = program_base
    // x21 = input_ptr
    // x22 = input_len
    // x23 = input_index
    // x24 = data_ptr
    // x25 = pc
    // x26 = temp
    // x27 = tape_base
    // x28 = match_table_base
    mov x19, x20
    mov x20, x22
    mov x21, x25
    mov x22, x24
    mov x23, #0
    mov x24, #0
    mov x25, #0
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, match_table
    add x28, x28, :lo12:match_table

exec_loop:
    cmp x25, x19
    b.ge exec_done
    add x9, x20, x25
    ldrb w26, [x9]

    cmp w26, #0x3e
    b.eq op_gt
    cmp w26, #0x3c
    b.eq op_lt
    cmp w26, #0x2b
    b.eq op_plus
    cmp w26, #0x2d
    b.eq op_minus
    cmp w26, #0x2e
    b.eq op_dot
    cmp w26, #0x2c
    b.eq op_comma
    cmp w26, #0x5b
    b.eq op_open
    cmp w26, #0x5d
    b.eq op_close

    add x25, x25, #1
    b exec_loop

op_gt:
    add x24, x24, #1
    add x25, x25, #1
    b exec_loop

op_lt:
    sub x24, x24, #1
    add x25, x25, #1
    b exec_loop

op_plus:
    add x9, x27, x24
    ldrb w26, [x9]
    add w26, w26, #1
    strb w26, [x9]
    add x25, x25, #1
    b exec_loop

op_minus:
    add x9, x27, x24
    ldrb w26, [x9]
    sub w26, w26, #1
    strb w26, [x9]
    add x25, x25, #1
    b exec_loop

op_dot:
    add x9, x27, x24
    ldrb w26, [x9]
    sub sp, sp, #16
    strb w26, [sp]
    mov x0, #1
    mov x1, sp
    mov x2, #1
    mov x8, #64
    svc #0
    add sp, sp, #16
    add x25, x25, #1
    b exec_loop

op_comma:
    cmp x23, x22
    b.ge comma_zero
    add x9, x21, x23
    ldrb w26, [x9]
    add x23, x23, #1
    b comma_store
comma_zero:
    mov w26, #0
comma_store:
    add x9, x27, x24
    strb w26, [x9]
    add x25, x25, #1
    b exec_loop

op_open:
    add x9, x27, x24
    ldrb w26, [x9]
    cmp w26, #0
    b.ne open_next
    add x9, x28, x25, lsl #1
    ldrh w26, [x9]
    add x25, x26, #1
    b exec_loop
open_next:
    add x25, x25, #1
    b exec_loop

op_close:
    add x9, x27, x24
    ldrb w26, [x9]
    cmp w26, #0
    b.eq close_next
    add x9, x28, x25, lsl #1
    ldrh w26, [x9]
    add x25, x26, #1
    b exec_loop
close_next:
    add x25, x25, #1
    b exec_loop

exec_done:
    mov x0, #0
    mov x8, #93
    svc #0

exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
    .align 8
input_buf:
    .space 65536

    .align 8
match_table:
    .space 131072

    .align 8
tape:
    .space 30000

    .align 8
bracket_stack:
    .space 2000
