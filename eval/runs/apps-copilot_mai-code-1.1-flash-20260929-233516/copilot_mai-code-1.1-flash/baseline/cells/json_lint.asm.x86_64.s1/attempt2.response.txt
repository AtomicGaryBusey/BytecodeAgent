.text
.globl _start

_start:
  call read_all_input
  mov qword ptr [rip + pos], 0
  call parse_document
  test eax, eax
  jnz ok_exit
  call write_error
  mov eax, 60
  mov edi, 1
  syscall

ok_exit:
  call write_ok
  mov eax, 60
  xor edi, edi
  syscall

read_all_input:
  mov r12, 0
1:
  mov eax, 0
  mov edi, 0
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 65536
  sub edx, r12d
  syscall
  cmp rax, 0
  jle 2f
  add r12, rax
  cmp r12, 65536
  jb 1b
2:
  mov qword ptr [rip + input_len], r12
  ret

write_ok:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, 3
  syscall
  ret

write_error:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + error_prefix]
  mov edx, 9
  syscall

  mov rax, qword ptr [rip + error_pos]
  lea r8, [rip + errbuf + 63]
  mov byte ptr [r8], 0
1:
  xor rdx, rdx
  mov rbx, 10
  div rbx
  add dl, '0'
  dec r8
  mov byte ptr [r8], dl
  test rax, rax
  jnz 1b

  lea r9, [rip + errbuf + 63]
  mov rsi, r8
  mov rdx, r9
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall

  lea rsi, [rip + newline]
  mov edx, 1
  mov eax, 1
  mov edi, 1
  syscall
  ret

parse_document:
  call skip_ws
  call parse_value
  test eax, eax
  jz 1f
  call skip_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jne 1f
  mov eax, 1
  ret
1:
  mov eax, 0
  ret

parse_value:
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae value_fail
  movzx eax, byte ptr [rip + input_buf + rax]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb 1f
  cmp al, '9'
  jbe parse_number
1:
  cmp al, 't'
  je parse_literal_true
  cmp al, 'f'
  je parse_literal_false
  cmp al, 'n'
  je parse_literal_null
value_fail:
  mov rax, qword ptr [rip + pos]
  mov qword ptr [rip + error_pos], rax
  xor eax, eax
  ret

skip_ws:
  mov rsi, qword ptr [rip + pos]
1:
  cmp rsi, qword ptr [rip + input_len]
  jae 2f
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 0x20
  je 3f
  cmp al, 0x09
  je 3f
  cmp al, 0x0A
  je 3f
  cmp al, 0x0D
  je 3f
  jmp 2f
3:
  inc rsi
  jmp 1b
2:
  mov qword ptr [rip + pos], rsi
  ret

parse_object:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '{'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '}'
  je obj_close
  call parse_string
  test eax, eax
  jz obj_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ':'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  call parse_value
  test eax, eax
  jz obj_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ','
  jne obj_close_check
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '}'
  je fail_pos_rsi
  jmp obj_member
obj_close_check:
  cmp al, '}'
  jne fail_pos_rsi
obj_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
obj_member:
  call parse_string
  test eax, eax
  jz obj_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ':'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  call parse_value
  test eax, eax
  jz obj_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ','
  jne obj_close_check
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '}'
  je fail_pos_rsi
  jmp obj_member
obj_fail:
  mov rax, qword ptr [rip + pos]
  mov qword ptr [rip + error_pos], rax
  xor eax, eax
  ret

parse_array:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '['
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ']'
  je arr_close
1:
  call parse_value
  test eax, eax
  jz arr_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ','
  jne arr_end
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, ']'
  je fail_pos_rsi
  jmp 1b
arr_end:
  cmp al, ']'
  jne fail_pos_rsi
arr_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
arr_fail:
  mov rax, qword ptr [rip + pos]
  mov qword ptr [rip + error_pos], rax
  xor eax, eax
  ret

parse_string:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '"'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
1:
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '"'
  je str_close
  cmp al, 0x5C
  jne str_raw
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '"'
  je 2f
  cmp al, 0x5C
  je 2f
  cmp al, '/'
  je 2f
  cmp al, 'b'
  je 2f
  cmp al, 'f'
  je 2f
  cmp al, 'n'
  je 2f
  cmp al, 'r'
  je 2f
  cmp al, 't'
  je 2f
  cmp al, 'u'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov rcx, 4
3:
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx edx, byte ptr [rip + input_buf + rsi]
  cmp dl, '0'
  jb fail_pos_rsi
  cmp dl, '9'
  jbe 4f
  cmp dl, 'a'
  jb fail_pos_rsi
  cmp dl, 'f'
  jbe 4f
  cmp dl, 'A'
  jb fail_pos_rsi
  cmp dl, 'F'
  jbe 4f
  jmp fail_pos_rsi
4:
  inc rsi
  mov qword ptr [rip + pos], rsi
  dec rcx
  jnz 3b
  jmp 1b
2:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
str_raw:
  cmp al, 0x20
  jb fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
str_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_literal_true:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 't'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'r'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'u'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'e'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_literal_false:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'f'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'a'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'l'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 's'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'e'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_literal_null:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'n'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'u'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'l'
  jne fail_pos_rsi
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'l'
  jne fail_pos_rsi
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_number:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '-'
  jne 1f
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
1:
  cmp al, '0'
  je num_zero
  cmp al, '1'
  jb fail_pos_rsi
  cmp al, '9'
  jbe num_int_digits
  jmp fail_pos_rsi
num_zero:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jnb num_after_int
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '0'
  jb num_after_int
  cmp al, '9'
  jbe fail_pos_rsi
num_after_int:
  cmp rsi, qword ptr [rip + input_len]
  jnb num_success
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '.'
  je num_frac
  cmp al, 'e'
  je num_exp
  cmp al, 'E'
  je num_exp
  jmp num_success
num_int_digits:
  inc rsi
  mov qword ptr [rip + pos], rsi
num_digit_loop:
  cmp rsi, qword ptr [rip + input_len]
  jnb num_after_int
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '0'
  jb num_after_int
  cmp al, '9'
  jbe 2f
  jmp num_after_int
2:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp num_digit_loop
num_frac:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '0'
  jb fail_pos_rsi
  cmp al, '9'
  jbe num_frac_digits
  jmp fail_pos_rsi
num_frac_digits:
  inc rsi
  mov qword ptr [rip + pos], rsi
num_frac_loop:
  cmp rsi, qword ptr [rip + input_len]
  jnb num_after_frac
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '0'
  jb num_after_frac
  cmp al, '9'
  jbe 3f
  jmp num_after_frac
3:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp num_frac_loop
num_after_frac:
  cmp rsi, qword ptr [rip + input_len]
  jnb num_success
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 'e'
  je num_exp
  cmp al, 'E'
  je num_exp
  jmp num_success
num_exp:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '+'
  je 4f
  cmp al, '-'
  je 4f
  jmp num_exp_digits
4:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae fail_end
  movzx eax, byte ptr [rip + input_buf + rsi]
num_exp_digits:
  cmp al, '0'
  jb fail_pos_rsi
  cmp al, '9'
  jbe num_exp_loop
  jmp fail_pos_rsi
num_exp_loop:
  inc rsi
  mov qword ptr [rip + pos], rsi
num_exp_body:
  cmp rsi, qword ptr [rip + input_len]
  jnb num_success
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, '0'
  jb num_success
  cmp al, '9'
  jbe 5f
  jmp num_success
5:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp num_exp_body
num_success:
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

fail_pos_rsi:
  mov qword ptr [rip + pos], rsi
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

fail_end:
  mov qword ptr [rip + pos], rsi
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

.section .rodata
ok_msg:
  .ascii "ok\n"
error_prefix:
  .ascii "error at "
newline:
  .byte 10

.data
input_len:
  .quad 0
pos:
  .quad 0
error_pos:
  .quad 0

.section .bss
input_buf:
  .space 65536
errbuf:
  .space 64
