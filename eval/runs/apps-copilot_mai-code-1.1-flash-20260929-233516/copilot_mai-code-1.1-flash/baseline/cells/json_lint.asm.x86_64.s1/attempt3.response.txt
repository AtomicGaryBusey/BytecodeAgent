.text
.globl _start

_start:
  call read_all_input
  mov qword ptr [rip + pos], 0
  call parse_document
  test eax, eax
  jnz .L_ok
  call write_error
  mov eax, 60
  mov edi, 1
  syscall
.L_ok:
  call write_ok
  mov eax, 60
  xor edi, edi
  syscall

read_all_input:
  xor r12, r12
  lea r10, [rip + input_buf]
1:
  mov eax, 0
  mov edi, 0
  mov rsi, r10
  add rsi, r12
  mov edx, 65536
  sub edx, r12d
  syscall
  cmp rax, 0
  jle 2f
  add r12, rax
  cmp r12, 65536
  jb 1b
2:
  mov qword ptr [rip + input_len], r12
  ret

write_ok:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, 3
  syscall
  ret

write_error:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + error_prefix]
  mov edx, 9
  syscall

  mov rax, qword ptr [rip + error_pos]
  lea r8, [rip + errbuf]
  add r8, 63
  mov byte ptr [r8], 0
1:
  xor rdx, rdx
  mov rcx, 10
  div rcx
  add dl, '0'
  dec r8
  mov byte ptr [r8], dl
  test rax, rax
  jnz 1b

  lea r11, [rip + errbuf]
  add r11, 63
  mov rdx, r11
  sub rdx, r8
  mov rsi, r8
  mov eax, 1
  mov edi, 1
  syscall

  lea rsi, [rip + newline]
  mov edx, 1
  mov eax, 1
  mov edi, 1
  syscall
  ret

parse_document:
  call skip_ws
  call parse_value
  test eax, eax
  jz .Lfail
  call skip_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jne .Lfail
  mov eax, 1
  ret
.Lfail:
  mov rax, qword ptr [rip + pos]
  mov qword ptr [rip + error_pos], rax
  xor eax, eax
  ret

parse_value:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb .L1
  cmp al, '9'
  jbe parse_number
.L1:
  cmp al, 't'
  je parse_literal_true
  cmp al, 'f'
  je parse_literal_false
  cmp al, 'n'
  je parse_literal_null
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret
.Lfail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

skip_ws:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
1:
  cmp rsi, qword ptr [rip + input_len]
  jae 2f
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 0x20
  je 3f
  cmp al, 0x09
  je 3f
  cmp al, 0x0A
  je 3f
  cmp al, 0x0D
  je 3f
  jmp 2f
3:
  inc rsi
  jmp 1b
2:
  mov qword ptr [rip + pos], rsi
  ret

parse_object:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '{'
  jne .Lfail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '}'
  je .Lclose
.Lmember:
  call parse_string
  test eax, eax
  jz .Lret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ':'
  jne .Lfail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  call parse_value
  test eax, eax
  jz .Lret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ','
  jne .Lcheck_close
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '}'
  je .Lfail_cur
  jmp .Lmember
.Lcheck_close:
  cmp al, '}'
  jne .Lfail_cur
.Lclose:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
.Lret:
  ret
.Lfail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret
.Lfail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_array:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end

  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ']'
  je .Lclose
1:
  call parse_value
  test eax, eax
  jz .Lret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ','

  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ']'
  je .Lfail_cur
  jmp 1b
.Lcheck_close:
  cmp al, ']'
  jne .Lfail_cur
.Lclose:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
.Lret:
  ret
.Lfail_cur:
  mov qword ptr [rip + error_pos], rsi

  ret
.Lfail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_string:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  jne .Lfail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
1:
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  je .Lclose
  cmp al, 0x5C
  jne .Lraw
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  je .Ladvance
  cmp al, 0x5C
  je .Ladvance
  cmp al, '/'
  je .Ladvance
  cmp al, 'b'
  je .Ladvance
  cmp al, 'f'
  je .Ladvance
  cmp al, 'n'
  je .Ladvance
  cmp al, 'r'
  je .Ladvance
  cmp al, 't'
  je .Ladvance
  cmp al, 'u'
  jne .Lfail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov rcx, 4
2:
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx edx, byte ptr [r10 + rsi]
  cmp dl, '0'
  jb .Lfail_cur
  cmp dl, '9'
  jbe 3f
  cmp dl, 'a'
  jb .Lfail_cur
  cmp dl, 'f'
  jbe 3f
  cmp dl, 'A'
  jb .Lfail_cur
  cmp dl, 'F'
  jbe 3f
  jmp .Lfail_cur
3:
  inc rsi
  mov qword ptr [rip + pos], rsi
  dec rcx
  jnz 2b
  jmp 1b
.Ladvance:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
.Lraw:
  cmp al, 0x20
  jb .Lfail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
.Lclose:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
.Lfail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret
.Lfail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_number:
  mov rsi, qword ptr [rip + pos]
  lea r10, [rip + input_buf]
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '-'
  jne .Lafter_minus
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
.Lafter_minus:
  cmp al, '0'
  je .Lzero
  cmp al, '1'
  jb .Lfail_cur
  cmp al, '9'
  jbe .Lint_digits
  jmp .Lfail_cur
.Lzero:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lafter_int
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb .Lafter_int
  cmp al, '9'
  jbe .Lfail_cur
.Lafter_int:
  cmp rsi, qword ptr [rip + input_len]
  jae .Ldone
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '.'
  je .Lfrac
  cmp al, 'e'
  je .Lexp
  cmp al, 'E'
  je .Lexp
  jmp .Ldone
.Lint_digits:
  inc rsi
  mov qword ptr [rip + pos], rsi
1:
  cmp rsi, qword ptr [rip + input_len]
  jae .Lafter_int
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb .Lafter_int
  cmp al, '9'
  jbe 2f
  jmp .Lafter_int
2:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
.Lfrac:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb .Lfail_cur
  cmp al, '9'
  jbe .Lfrac_digits
  jmp .Lfail_cur
.Lfrac_digits:
  inc rsi
  mov qword ptr [rip + pos], rsi
1:
  cmp rsi, qword ptr [rip + input_len]
  jae .Lafter_frac
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb .Lafter_frac
  cmp al, '9'
  jbe 2f
  jmp .Lafter_frac
2:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
.Lafter_frac:
  cmp rsi, qword ptr [rip + input_len]
  jae .Ldone
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'e'
  je .Lexp
  cmp al, 'E'
  je .Lexp
  jmp .Ldone
.Lexp:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '+'
  je 1f
  cmp al, '-'
  je 1f
  jmp .Lexp_digits
1:
  inc rsi
  cmp rsi, qword ptr [rip + input_len]
  jae .Lfail_end
  movzx eax, byte ptr [r10 + rsi]
.Lexp_digits:
  cmp al, '0'
  jb .Lfail_cur
  cmp al, '9'
  jbe .Lexp_loop
  jmp .Lfail_cur
.Lexp_loop:
  inc rsi
  mov qword ptr [rip + pos], rsi
1:
  cmp rsi, qword ptr [rip + input_len]
  jae .Ldone
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb .Ldone
  cmp al, '9'
  jbe 2f
  jmp .Ldone
2:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp 1b
.Ldone:
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret
.Lfail_cur:

  xor eax, eax
  ret
.Lfail_end:

  xor eax, eax
