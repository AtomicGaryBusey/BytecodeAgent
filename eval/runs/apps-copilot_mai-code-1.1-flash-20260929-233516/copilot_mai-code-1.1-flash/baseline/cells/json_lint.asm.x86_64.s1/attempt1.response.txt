.text
.globl _start

_start:
  call read_all_input
  mov qword ptr [pos], 0
  call parse_document
  test al, al
  jnz 1f
  call write_error
  mov eax, 60
  mov edi, 1
  syscall
1:
  call write_ok
  mov eax, 60
  xor edi, edi
  syscall

read_all_input:
  mov r12, 0
1:
  cmp r12, 65536
  je 2f
  mov eax, 0
  mov edi, 0
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 65536
  sub edx, r12d
  syscall
  test rax, rax
  jle 2f
  add r12, rax
  jmp 1b
2:
  mov qword ptr [input_len], r12
  ret

write_ok:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, 3
  syscall
  ret

write_error:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + error_prefix]
  mov edx, 9
  syscall
  mov rax, qword ptr [pos]
  lea r8, [rip + errbuf + 32]
  mov byte ptr [r8], 0
  mov r9, r8
1:
  xor rdx, rdx
  mov rbx, 10
  div rbx
  add dl, '0'
  dec r9
  mov byte ptr [r9], dl
  test rax, rax
  jnz 1b
  mov rdx, r8
  sub rdx, r9
  mov rsi, r9
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + newline]
  mov edx, 1
  syscall
  ret

parse_document:
  call skip_ws
  call parse_value
  test al, al
  jz 2f
  call skip_ws
  mov rax, qword ptr [pos]
  cmp rax, qword ptr [input_len]
  jne 2f
  mov al, 1
  ret
2:
  mov al, 0
  ret

parse_value:
  mov rax, qword ptr [pos]
  cmp rax, qword ptr [input_len]
  jae 1f
  movzx edx, byte ptr [rip + input_buf + rax]
  cmp dl, '{'
  je parse_object
  cmp dl, '['
  je parse_array
  cmp dl, '"'
  je parse_string
  cmp dl, '-'
  je parse_number
  cmp dl, '0'
  jb 2f
  cmp dl, '9'
  jbe parse_number
2:
  cmp dl, 't'
  je parse_literal_true
  cmp dl, 'f'
  je parse_literal_false
  cmp dl, 'n'
  je parse_literal_null
1:
  mov qword ptr [pos], rax
  xor eax, eax
  ret

skip_ws:
  mov rsi, qword ptr [pos]
1:
  cmp rsi, qword ptr [input_len]
  jae 2f
  movzx eax, byte ptr [rip + input_buf + rsi]
  cmp al, 0x20
  je 3f
  cmp al, 0x09
  je 3f
  cmp al, 0x0A
  je 3f
  cmp al, 0x0D
  je 3f
  jmp 2f
3:
  inc rsi
  jmp 1b
2:
  mov qword ptr [pos], rsi
  ret

2:
