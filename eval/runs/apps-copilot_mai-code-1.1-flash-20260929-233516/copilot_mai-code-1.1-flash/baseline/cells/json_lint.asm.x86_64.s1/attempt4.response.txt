.text
.globl _start

_start:
  call read_all_input
  mov qword ptr [rip + pos], 0
  call parse_document
  test eax, eax
  jnz .L_ok
  call write_error
  mov eax, 60
  mov edi, 1
  syscall
.L_ok:
  call write_ok
  mov eax, 60
  xor edi, edi
  syscall

read_all_input:
  xor r12, r12
  lea r10, [rip + input_buf]
1:
  mov eax, 0
  mov edi, 0
  mov rsi, r10
  add rsi, r12
  mov edx, 65536
  sub edx, r12d
  syscall
  cmp rax, 0
  je 2f
  add r12, rax
  cmp r12, 65536
  jb 1b
2:
  mov qword ptr [rip + input_len], r12
  ret

write_ok:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, 3
  syscall
  ret

write_error:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + error_prefix]
  mov edx, 9
  syscall

  mov rax, qword ptr [rip + error_pos]
  lea r8, [rip + errbuf + 63]
1:
  xor rdx, rdx
  mov rcx, 10
  div rcx
  add dl, '0'
  dec r8
  mov byte ptr [r8], dl
  test rax, rax
  jnz 1b

  lea r11, [rip + errbuf + 64]
  mov rdx, r11
  sub rdx, r8
  mov rsi, r8
  mov eax, 1
  mov edi, 1
  syscall

  lea rsi, [rip + newline]
  mov edx, 1
  mov eax, 1
  mov edi, 1
  syscall
  ret

parse_document:
  call skip_ws
  call parse_value
  test eax, eax
  jz parse_document_fail
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jne parse_document_fail_cur
  mov eax, 1
  ret

parse_document_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_document_fail:
  ret

skip_ws:
  mov rsi, qword ptr [rip + pos]
1:
  cmp rsi, qword ptr [rip + input_len]
  jae 2f
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 0x20
  je 3f
  cmp al, 0x09
  je 3f
  cmp al, 0x0A
  je 3f
  cmp al, 0x0D
  je 3f
  jmp 2f
3:
  inc rsi
  jmp 1b
2:
  mov qword ptr [rip + pos], rsi
  ret

parse_value:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_value_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb parse_value_test_literal
  cmp al, '9'
  jbe parse_number
parse_value_test_literal:
  cmp al, 't'
  je parse_literal_true
  cmp al, 'f'
  je parse_literal_false
  cmp al, 'n'
  je parse_literal_null
  jmp parse_value_fail_cur

parse_value_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_value_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_object:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_object_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '{'
  jne parse_object_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_object_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '}'
  je parse_object_close

parse_object_member_loop:
  call parse_string
  test eax, eax
  jz parse_object_ret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_object_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ':'
  jne parse_object_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  call parse_value
  test eax, eax
  jz parse_object_ret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_object_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ','
  jne parse_object_check_close
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_object_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '}'
  je parse_object_fail_cur
  jmp parse_object_member_loop

parse_object_check_close:
  cmp al, '}'
  jne parse_object_fail_cur

parse_object_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_object_ret:
  ret

parse_object_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_object_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_array:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_array_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '['
  jne parse_array_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_array_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ']'
  je parse_array_close

parse_array_loop:
  call parse_value
  test eax, eax
  jz parse_array_ret
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_array_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ','
  jne parse_array_check_close
  inc rsi
  mov qword ptr [rip + pos], rsi
  call skip_ws
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_array_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, ']'
  je parse_array_fail_cur
  jmp parse_array_loop

parse_array_check_close:
  cmp al, ']'
  jne parse_array_fail_cur

parse_array_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_array_ret:
  ret

parse_array_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_array_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_string:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_string_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  jne parse_string_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi

parse_string_loop:
  cmp rsi, qword ptr [rip + input_len]
  jae parse_string_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  je parse_string_close
  cmp al, 0x5C
  jne parse_string_raw
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_string_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '"'
  je parse_string_advance
  cmp al, 0x5C
  je parse_string_advance
  cmp al, '/'
  je parse_string_advance
  cmp al, 'b'
  je parse_string_advance
  cmp al, 'f'
  je parse_string_advance
  cmp al, 'n'
  je parse_string_advance
  cmp al, 'r'
  je parse_string_advance
  cmp al, 't'
  je parse_string_advance
  cmp al, 'u'
  jne parse_string_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov ecx, 4
parse_string_hex:
  cmp rsi, qword ptr [rip + input_len]
  jae parse_string_fail_end
  movzx edx, byte ptr [r10 + rsi]
  cmp dl, '0'
  jb parse_string_fail_cur
  cmp dl, '9'
  jbe parse_string_hex_ok
  cmp dl, 'a'
  jb parse_string_fail_cur
  cmp dl, 'f'
  jbe parse_string_hex_ok
  cmp dl, 'A'
  jb parse_string_fail_cur
  cmp dl, 'F'
  jbe parse_string_hex_ok
  jmp parse_string_fail_cur
parse_string_hex_ok:
  inc rsi
  mov qword ptr [rip + pos], rsi
  dec ecx
  jnz parse_string_hex
  jmp parse_string_loop

parse_string_advance:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp parse_string_loop

parse_string_raw:
  cmp al, 0x20
  jb parse_string_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp parse_string_loop

parse_string_close:
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_string_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_string_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_literal_true:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_true_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 't'
  jne parse_literal_true_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_true_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'r'
  jne parse_literal_true_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_true_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'u'
  jne parse_literal_true_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_true_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'e'
  jne parse_literal_true_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_literal_true_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_literal_true_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_literal_false:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_false_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'f'
  jne parse_literal_false_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_false_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'a'
  jne parse_literal_false_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_false_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'l'
  jne parse_literal_false_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_false_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 's'
  jne parse_literal_false_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_false_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'e'
  jne parse_literal_false_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_literal_false_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_literal_false_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_literal_null:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_literal_null_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'n'
  jne parse_literal_null_fail_cur
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]

  cmp al, 'u'
  jne parse_literal_null_fail_cur

  cmp rsi, qword ptr [rip + input_len]

  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]

parse_literal_null_fail_cur:

  ret

parse_literal_null_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_number:
  mov rsi, qword ptr [rip + pos]
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_fail_end
  lea r10, [rip + input_buf]
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '-'
  jne parse_number_after_minus
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_fail_end
  movzx eax, byte ptr [r10 + rsi]

parse_number_after_minus:
  cmp al, '0'
  je parse_number_zero
  cmp al, '1'
  jb parse_number_fail_cur
  cmp al, '9'
  jbe parse_number_nonzero
  jmp parse_number_fail_cur

parse_number_nonzero:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp parse_number_int_loop

parse_number_zero:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_after_int
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb parse_number_after_int
  cmp al, '9'
  jbe parse_number_fail_cur
  jmp parse_number_after_int

parse_number_int_loop:
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_after_int
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb parse_number_after_int
  cmp al, '9'
  jbe parse_number_int_consume
  jmp parse_number_after_int

parse_number_int_consume:
  inc rsi
  mov qword ptr [rip + pos], rsi
  jmp parse_number_int_loop

parse_number_after_int:
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_done
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '.'
  je parse_number_frac
  cmp al, 'e'
  je parse_number_exp
  cmp al, 'E'
  je parse_number_exp
  jmp parse_number_done

parse_number_frac:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb parse_number_fail_cur
  cmp al, '9'
  jbe parse_number_frac_loop
  jmp parse_number_fail_cur

parse_number_frac_loop:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_after_frac
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb parse_number_after_frac
  cmp al, '9'
  jbe parse_number_frac_loop
  jmp parse_number_after_frac

parse_number_after_frac:
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_done
  movzx eax, byte ptr [r10 + rsi]
  cmp al, 'e'
  je parse_number_exp
  cmp al, 'E'
  je parse_number_exp
  jmp parse_number_done

parse_number_exp:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_fail_end
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '+'
  je parse_number_exp_sign
  cmp al, '-'
  je parse_number_exp_sign
  jmp parse_number_exp_digits

parse_number_exp_sign:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_fail_end
  movzx eax, byte ptr [r10 + rsi]

parse_number_exp_digits:
  cmp al, '0'
  jb parse_number_fail_cur
  cmp al, '9'
  jbe parse_number_exp_loop
  jmp parse_number_fail_cur

parse_number_exp_loop:
  inc rsi
  mov qword ptr [rip + pos], rsi
  cmp rsi, qword ptr [rip + input_len]
  jae parse_number_done
  movzx eax, byte ptr [r10 + rsi]
  cmp al, '0'
  jb parse_number_done
  cmp al, '9'
  jbe parse_number_exp_loop
  jmp parse_number_done

parse_number_done:
  mov qword ptr [rip + pos], rsi
  mov eax, 1
  ret

parse_number_fail_cur:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret

parse_number_fail_end:
  mov qword ptr [rip + error_pos], rsi
  xor eax, eax
  ret
