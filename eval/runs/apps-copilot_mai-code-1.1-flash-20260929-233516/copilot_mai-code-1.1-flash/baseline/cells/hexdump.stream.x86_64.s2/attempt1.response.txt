.text
.globl _start
_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    cmp rax, 0
    jl read_error
    je read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_loop

read_error:
    mov eax, 60
    xor edi, edi
    mov dil, 1
    syscall

read_done:
    mov r13, r12
    cmp r13, 0
    je print_total_len
    xor r14, r14

row_loop:
    cmp r14, r13
    jae print_total_len
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jle row_len_ok
    mov r15, 16
row_len_ok:
    mov rdi, r14
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, r15
    call format_row
    mov rbx, rax
    mov rdi, 1
    lea rsi, [rip + linebuf]
    mov rdx, rbx
    call write_all
    add r14, r15
    jmp row_loop

print_total_len:
    mov rdi, r13
    lea rsi, [rip + linebuf]
    call hex8_to_buf
    mov byte ptr [rip + linebuf + 8], 10
    mov rdi, 1
    lea rsi, [rip + linebuf]
    mov rdx, 9
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

hex8_to_buf:
    push rbx
    mov rax, rdi
    lea rdx, [rip + hex_digits]
    mov rcx, 8
    lea r8, [rsi + 8]
1:
    mov rbx, rax
    and rbx, 0xF
    mov r9b, byte ptr [rdx + rbx]
    mov byte ptr [r8 - 1], r9b
    shr rax, 4
    dec rcx
    jnz 1b
    pop rbx
    ret

format_row:
    push rbx
    push r12
    push r13
    push r14
    push r15
    lea r15, [rip + linebuf]
    mov r12, rsi
    mov r13, rdx
    mov r14, r15
    mov rdi, rdi
    mov rsi, r15
    call hex8_to_buf
    mov byte ptr [r15 + 8], ' '
    mov byte ptr [r15 + 9], ' '
    add r14, 10
    xor rcx, rcx
1:
    cmp rcx, 16
    jae 2f
    cmp rcx, r13
    jb 3f
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    mov byte ptr [r14 + 2], ' '
    jmp 4f
3:
    movzx rax, byte ptr [r12 + rcx]
    mov rbx, rax
    shr rbx, 4
    lea rdx, [rip + hex_digits]
    mov r9b, byte ptr [rdx + rbx]
    mov byte ptr [r14], r9b
    mov rbx, rax
    and rbx, 0xF
    mov r9b, byte ptr [rdx + rbx]
    mov byte ptr [r14 + 1], r9b
    mov byte ptr [r14 + 2], ' '
4:
    add r14, 3
    cmp rcx, 7
    jne 5f
    mov byte ptr [r14], ' '
    inc r14
5:
    inc rcx
    jmp 1b
2:
    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], '|'
    inc r14
    xor rcx, rcx
6:
    cmp rcx, r13
    jae 7f
    movzx rax, byte ptr [r12 + rcx]
    cmp al, 0x20
    jb 8f
    cmp al, 0x7E
    ja 8f
    mov byte ptr [r14], al
    jmp 9f
8:
    mov byte ptr [r14], '.'
9:
    inc r14
    inc rcx
    jmp 6b
7:
    mov byte ptr [r14], '|'
    inc r14
    mov byte ptr [r14], 10
    inc r14
    mov rax, r14
    sub rax, r15
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

write_all:
    push r12
    mov r12, rdx
1:
    cmp r12, 0
    je 2f
    mov rax, 1
    syscall
    cmp rax, 0
    jl 3f
    cmp rax, r12
    je 2f
    sub r12, rax
    add rsi, rax
    jmp 1b
2:
    pop r12
    ret
3:
    mov eax, 60
    xor edi, edi
    mov dil, 1
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .space 65536
linebuf:
    .space 128
