.globl _start

.section .bss
.balign 16
input_buf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz done
    mov rbx, rax
    add r14, rbx
    mov rcx, rbx
    lea rsi, [rip + input_buf]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0A
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    test r15, r15
    jnz 2f
    inc r13
    mov r15, 1
2:
    dec rcx
    jnz process_loop
    jmp read_loop

whitespace:
    xor r15, r15
    dec rcx
    jnz process_loop
    jmp read_loop

done:
    lea r9, [rip + outbuf]
    mov rsi, r9

    mov rax, r12
    mov rbx, 10
    xor rcx, rcx
1:
    xor rdx, rdx
    div rbx
    push rdx
    inc rcx
    test rax, rax
    jnz 1b
2:
    pop rdx
    add dl, 0x30
    mov byte ptr [rsi], dl
    inc rsi
    dec rcx
    jnz 2b

    mov byte ptr [rsi], 0x20
    inc rsi

    mov rax, r13
    mov rbx, 10
    xor rcx, rcx
3:
    xor rdx, rdx
    div rbx
    push rdx
    inc rcx
    test rax, rax
    jnz 3b
4:
    pop rdx
    add dl, 0x30
    mov byte ptr [rsi], dl
    inc rsi
    dec rcx
    jnz 4b

    mov byte ptr [rsi], 0x20
    inc rsi

    mov rax, r14
    mov rbx, 10
    xor rcx, rcx
5:
    xor rdx, rdx
    div rbx
    push rdx
    inc rcx
    test rax, rax
    jnz 5b
6:
    pop rdx
    add dl, 0x30
    mov byte ptr [rsi], dl
    inc rsi
    dec rcx
    jnz 6b

    mov byte ptr [rsi], 0x0A
    inc rsi

    mov rdx, rsi
    sub rdx, r9

    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
