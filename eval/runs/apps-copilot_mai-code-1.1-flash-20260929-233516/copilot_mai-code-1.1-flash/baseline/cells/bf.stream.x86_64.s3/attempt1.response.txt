.globl _start

.section .bss
    .align 16
buf: .space 65536
tape: .space 30000
pair_table: .space 65536 * 4
stack: .space 1000 * 4
outbuf: .space 65536
program_len: .quad 0
input_ptr: .quad 0
input_len: .quad 0

.text
_start:
    lea r13, [rip + buf]
    xor r12, r12
    mov r14, r13
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r14
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add r14, rax
    jmp read_loop
read_done:
    xor r11, r11
    mov r10, r12
    cmp r10, 0
    je no_bang
find_bang:
    cmp r11, r10
    jae no_bang
    cmp byte ptr [r13 + r11], 0x21
    je bang_found
    inc r11
    jmp find_bang
bang_found:
    mov r9, r11
    lea r8, [r13 + r11 + 1]
    mov rax, r10
    sub rax, r11
    dec rax
    mov qword ptr [rip + input_len], rax
    mov qword ptr [rip + input_ptr], r8
    mov qword ptr [rip + program_len], r9
    jmp bracket_check
no_bang:
    mov r9, r10
    lea r8, [r13 + r10]
    mov qword ptr [rip + input_len], 0
    mov qword ptr [rip + input_ptr], r8
    mov qword ptr [rip + program_len], r9

bracket_check:
    lea rbx, [rip + stack]
    lea rsi, [rip + pair_table]
    xor r11, r11
    xor r15, r15
    mov r14, qword ptr [rip + program_len]
validate_loop:
    cmp r15, r14
    jae validate_end
    movzx eax, byte ptr [r13 + r15]
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket
    jmp validate_next
open_bracket:
    cmp r11, 1000
    jae invalid
    mov dword ptr [rbx + r11*4], r15d
    inc r11
    jmp validate_next
close_bracket:
    test r11, r11
    jz invalid
    dec r11
    mov eax, dword ptr [rbx + r11*4]
    mov dword ptr [rsi + rax*4], r15d
    mov dword ptr [rsi + r15*4], eax
validate_next:
    inc r15
    jmp validate_loop
validate_end:
    test r11, r11
    jnz invalid

    lea r14, [rip + buf]
    lea rbx, [rip + tape]
    lea rbp, [rip + outbuf]
    lea rsi, [rip + pair_table]
    mov r15, qword ptr [rip + program_len]
    mov r8, qword ptr [rip + input_ptr]
    mov r9, qword ptr [rip + input_len]
    xor r10, r10
    xor r11, r11
    xor r12, r12
    xor r13, r13

exec_loop:
    cmp r13, r15
    jae exec_done
    movzx eax, byte ptr [r14 + r13]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    inc r13
    jmp exec_loop

cmd_gt:
    inc r12
    inc r13
    jmp exec_loop

cmd_lt:
    dec r12
    inc r13
    jmp exec_loop

cmd_plus:
    movzx eax, byte ptr [rbx + r12]
    inc al
    mov byte ptr [rbx + r12], al
    inc r13
    jmp exec_loop

cmd_minus:
    movzx eax, byte ptr [rbx + r12]
    dec al
    mov byte ptr [rbx + r12], al
    inc r13
    jmp exec_loop

cmd_dot:
    movzx eax, byte ptr [rbx + r12]
    mov byte ptr [rbp + r11], al
    inc r11
    inc r13
    jmp exec_loop

cmd_comma:
    cmp r10, r9
    jb comma_input
    xor eax, eax
    jmp comma_store
comma_input:
    movzx eax, byte ptr [r8 + r10]
    inc r10
comma_store:
    mov byte ptr [rbx + r12], al
    inc r13
    jmp exec_loop

cmd_open:
    movzx eax, byte ptr [rbx + r12]
    test al, al
    jz open_jump
    inc r13
    jmp exec_loop
open_jump:
    mov eax, dword ptr [rsi + r13*4]
    inc eax
    mov r13, rax
    jmp exec_loop

cmd_close:
    movzx eax, byte ptr [rbx + r12]
    test al, al
    jnz close_jump
    inc r13
    jmp exec_loop
close_jump:
    mov eax, dword ptr [rsi + r13*4]
    inc eax
    mov r13, rax
    jmp exec_loop

exec_done:
    mov rax, 1
    mov rdi, 1
    mov rsi, rbp
    mov rdx, r11
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

invalid:
    mov rax, 60
    mov rdi, 1
    syscall
