.globl _start

.section .bss
    .align 16
buf: .space 65536
tape: .space 30000
pair_table: .space 65536 * 4
bracket_stack: .space 1001 * 4
output_buf: .space 65536
stdin_len: .space 8
program_len: .space 8
input_len: .space 8
input_pos: .space 8
input_base: .space 8
out_len: .space 8

.text

_start:
    xor r12, r12
    lea r13, [rip + buf]

read_loop:
    mov rdi, 0
    mov rsi, r13
    mov rdx, 65536
    sub rdx, r12
    cmp rdx, 0
    jle read_done
    mov rax, 0
    syscall
    test rax, rax
    js read_error
    jz read_done
    add r12, rax
    add r13, rax
    cmp r12, 65536
    je read_done
    jmp read_loop

read_done:
    mov qword ptr [rip + stdin_len], r12

    mov rbx, qword ptr [rip + stdin_len]
    xor rcx, rcx

find_bang:
    cmp rcx, rbx
    jae no_bang
    lea rax, [rip + buf]
    mov dl, byte ptr [rax + rcx]
    cmp dl, 0x21
    je bang_found
    inc rcx
    jmp find_bang

no_bang:
    mov qword ptr [rip + program_len], rbx
    mov qword ptr [rip + input_len], 0
    mov qword ptr [rip + input_pos], 0
    lea rax, [rip + buf]
    mov qword ptr [rip + input_base], rax
    jmp bracket_check

bang_found:
    mov qword ptr [rip + program_len], rcx
    mov rax, rbx
    sub rax, rcx
    dec rax
    mov qword ptr [rip + input_len], rax
    mov qword ptr [rip + input_pos], 0
    lea rax, [rip + buf]
    add rax, rcx
    inc rax
    mov qword ptr [rip + input_base], rax

bracket_check:
    xor r8, r8
    lea r9, [rip + bracket_stack]
    lea r10, [rip + pair_table]
    xor rcx, rcx

bc_loop:
    cmp rcx, qword ptr [rip + program_len]
    jae bc_done
    lea rax, [rip + buf]
    mov al, byte ptr [rax + rcx]
    cmp al, 0x5b
    jne bc_not_open
    mov dword ptr [r9 + r8*4], ecx
    inc r8
    jmp bc_next

bc_not_open:
    cmp al, 0x5d
    jne bc_next
    test r8, r8
    jz bracket_error
    dec r8
    mov eax, dword ptr [r9 + r8*4]
    mov dword ptr [r10 + rax*4], ecx
    mov dword ptr [r10 + rcx*4], eax

bc_next:
    inc rcx
    jmp bc_loop

bc_done:
    test r8, r8
    jnz bracket_error
    jmp interpreter

bracket_error:
    mov edi, 1
    mov eax, 60
    syscall

read_error:
    mov edi, 1
    mov eax, 60
    syscall

interpreter:
    xor rcx, rcx
    xor rdx, rdx
    xor r14, r14
    lea r11, [rip + buf]
    lea r12, [rip + tape]
    lea r13, [rip + output_buf]

interp_loop:
    cmp rcx, qword ptr [rip + program_len]
    jae interp_done
    mov al, byte ptr [r11 + rcx]
    cmp al, 0x3e
    je cmd_gt
    cmp al, 0x3c
    je cmd_lt
    cmp al, 0x2b
    je cmd_plus
    cmp al, 0x2d
    je cmd_minus
    cmp al, 0x2e
    je cmd_dot
    cmp al, 0x2c
    je cmd_comma
    cmp al, 0x5b
    je cmd_lbracket
    cmp al, 0x5d
    je cmd_rbracket
    inc rcx
    jmp interp_loop

cmd_gt:
    inc rdx
    inc rcx
    jmp interp_loop

cmd_lt:
    dec rdx
    inc rcx
    jmp interp_loop

cmd_plus:
    inc byte ptr [r12 + rdx]
    inc rcx
    jmp interp_loop

cmd_minus:
    dec byte ptr [r12 + rdx]
    inc rcx
    jmp interp_loop

cmd_dot:
    movzx eax, byte ptr [r12 + rdx]
    mov byte ptr [r13 + r14], al
    inc r14
    inc rcx
    jmp interp_loop

cmd_comma:
    mov rax, qword ptr [rip + input_pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae cmd_comma_zero
    mov rsi, qword ptr [rip + input_base]
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [r12 + rdx], al
    inc qword ptr [rip + input_pos]
    jmp cmd_comma_done

cmd_comma_zero:
    mov byte ptr [r12 + rdx], 0

cmd_comma_done:
    inc rcx
    jmp interp_loop

cmd_lbracket:
    movzx eax, byte ptr [r12 + rdx]
    test al, al
    jnz cmd_lbracket_continue
    mov rax, qword ptr [rip + pair_table]
    mov rsi, rcx
    lea rax, [rip + pair_table]
    mov eax, dword ptr [rax + rsi*4]
    lea rcx, [rax + 1]
    jmp interp_loop

cmd_lbracket_continue:
    inc rcx
    jmp interp_loop

cmd_rbracket:
    movzx eax, byte ptr [r12 + rdx]
    test al, al
    jz cmd_rbracket_continue
    mov rax, qword ptr [rip + pair_table]
    mov rsi, rcx
    lea rax, [rip + pair_table]
    mov eax, dword ptr [rax + rsi*4]
    lea rcx, [rax + 1]
    jmp interp_loop

cmd_rbracket_continue:
    inc rcx
    jmp interp_loop

interp_done:
    mov qword ptr [rip + out_len], r14
    mov rax, qword ptr [rip + out_len]
    test rax, rax
    jz exit_zero
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, rax
    mov rax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
