.globl _start

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
    jmp read_more

read_more:
    xor eax, eax
    mov edi, eax
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz done_output
    js done_output
    mov rcx, rax
    lea rbx, [rip + buffer]

process_chunk:
    mov dl, byte ptr [rbx]
    inc rbx
    inc r14

    cmp dl, 0x0A
    je newline_case
    cmp dl, 0x20
    je whitespace_case
    cmp dl, 0x09
    je whitespace_case
    cmp dl, 0x0B
    je whitespace_case
    cmp dl, 0x0C
    je whitespace_case
    cmp dl, 0x0D
    je whitespace_case

    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte

newline_case:
    inc r12
    xor r15, r15
    jmp next_byte

whitespace_case:
    xor r15, r15

next_byte:
    dec rcx
    jnz process_chunk
    jmp read_more

done_output:
    lea r10, [rip + outbuf]
    mov rdi, r10
    mov rax, r12
    call print_dec_to_ptr
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call print_dec_to_ptr
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call print_dec_to_ptr
    mov byte ptr [rdi], 0x0A
    inc rdi

    mov rdx, rdi
    sub rdx, r10
    lea rsi, [rip + outbuf]
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor edi, edi
    syscall

print_dec_to_ptr:
    mov r8, rax
    sub rsp, 32
    xor rcx, rcx

1:
    mov rax, r8
    xor edx, edx
    mov r9, 10
    div r9
    mov r8, rax
    add dl, 0x30
    mov byte ptr [rsp + rcx], dl
    inc rcx
    test r8, r8
    jnz 1b

2:
    dec rcx
    mov dl, byte ptr [rsp + rcx]
    mov byte ptr [rdi], dl
    inc rdi
    test rcx, rcx
    jnz 2b

    add rsp, 32
    mov rax, rdi
    ret

.section .bss
    .p2align 4
buffer:
    .zero 65536

outbuf:
    .zero 64
