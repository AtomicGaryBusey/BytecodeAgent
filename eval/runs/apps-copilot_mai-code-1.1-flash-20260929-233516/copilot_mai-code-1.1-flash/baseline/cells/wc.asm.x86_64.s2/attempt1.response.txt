.globl _start

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_more:
    mov rax, 0
    lea rdi, [rip + buffer]
    mov rsi, 65536
    syscall
    test rax, rax
    jz done_input
    mov rcx, rax
    lea rbx, [rip + buffer]

process_chunk:
    mov dl, byte ptr [rbx]
    inc rbx
    inc r14

    cmp dl, 10
    je newline_byte
    cmp dl, 32
    je whitespace_byte
    cmp dl, 9
    je whitespace_byte
    cmp dl, 11
    je whitespace_byte
    cmp dl, 12
    je whitespace_byte
    cmp dl, 13
    je whitespace_byte

    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte

newline_byte:
    inc r12
    xor r15, r15
    jmp next_byte

whitespace_byte:
    xor r15, r15

next_byte:
    dec rcx
    jnz process_chunk
    jmp read_more

done_input:
    lea r10, [rip + outbuf]
    mov rdi, r10
    mov rax, r12
    call print_dec_to_ptr
    mov byte ptr [rax], 32
    inc rax
    mov rdi, rax
    mov rax, r13
    call print_dec_to_ptr
    mov byte ptr [rax], 32
    inc rax
    mov rdi, rax
    mov rax, r14
    call print_dec_to_ptr
    mov byte ptr [rax], 10
    inc rax

    mov rdx, rax
    sub rdx, r10

    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    syscall

    mov rax, 60
    xor edi, edi
    syscall

print_dec_to_ptr:
    mov r8, rax
    sub rsp, 32
    xor rcx, rcx

1:
    mov rax, r8
    xor edx, edx
    mov rsi, 10
    div rsi
    mov r8, rax
    add dl, 48
    mov byte ptr [rsp + rcx], dl
    inc rcx
    test r8, r8
    jnz 1b

2:
    dec rcx
    mov dl, byte ptr [rsp + rcx]
    mov byte ptr [rdi], dl
    inc rdi
    test rcx, rcx
    jnz 2b

    add rsp, 32
    mov rax, rdi
    ret

.section .bss
    .p2align 4
buffer:
    .zero 65536

outbuf:
    .zero 64
