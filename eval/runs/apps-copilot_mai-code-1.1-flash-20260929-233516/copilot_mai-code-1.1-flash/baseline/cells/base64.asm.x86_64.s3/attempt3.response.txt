.text
.globl _start

_start:
    xor r12, r12

read_loop:
    xor rdi, rdi
    mov rax, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 49152
    jb read_loop

read_done:
    test r12, r12
    jz exit_zero

    xor r13, r13
    xor r14, r14
    lea r15, [rip + outbuf]
    lea rbx, [rip + alphabet]
    lea rsi, [rip + inbuf]
    xor rdx, rdx

encode_loop:
    mov rcx, r12
    sub rcx, rdx
    cmp rcx, 3
    jb encode_remainder

    movzx r8d, byte ptr [rsi + rdx]
    movzx r9d, byte ptr [rsi + rdx + 1]
    movzx r10d, byte ptr [rsi + rdx + 2]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    mov eax, r9d
    shr eax, 4
    and eax, 15
    or ecx, eax
    call emit_idx

    mov ecx, r9d
    and ecx, 15
    shl ecx, 2
    mov eax, r10d
    shr eax, 6
    and eax, 3
    or ecx, eax
    call emit_idx

    mov ecx, r10d
    and ecx, 63
    call emit_idx

    add rdx, 3
    jmp encode_loop

encode_remainder:
    mov rcx, r12
    sub rcx, rdx
    test rcx, rcx
    jz finalize
    cmp rcx, 1
    je rem_one

rem_two:
    movzx r8d, byte ptr [rsi + rdx]
    movzx r9d, byte ptr [rsi + rdx + 1]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    mov eax, r9d
    shr eax, 4
    and eax, 15
    or ecx, eax
    call emit_idx

    mov ecx, r9d
    and ecx, 15
    shl ecx, 2
    call emit_idx

    mov al, '='
    call emit_char

    mov al, '='
    call emit_char
    jmp finalize

rem_one:
    movzx r8d, byte ptr [rsi + rdx]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    call emit_idx

    mov al, '='
    call emit_char

    mov al, '='
    call emit_char

finalize:
    test r14, r14
    jz write_output
    mov byte ptr [r15 + r13], 10
    inc r13

write_output:
    test r13, r13
    jz exit_zero

    xor rbx, rbx

write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle exit_zero
    add rbx, rax
    cmp rbx, r13
    jb write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_idx:
    movzx eax, byte ptr [rbx + rcx]
    jmp emit_char

emit_char:
    mov byte ptr [r15 + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [r15 + r13], 10
    inc r13
    xor r14, r14
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .space 49152

.align 16
outbuf:
    .space 70000
