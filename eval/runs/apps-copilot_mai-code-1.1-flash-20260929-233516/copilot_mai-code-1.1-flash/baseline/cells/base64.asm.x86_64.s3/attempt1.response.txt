.text
.globl _start

_start:
    xor r11, r11
    lea r13, [rip + inbuf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    add rsi, r11
    mov rdx, 49152
    sub rdx, r11
    syscall
    cmp rax, 0
    je read_done
    add r11, rax
    cmp r11, 49152
    jae read_done
    jmp read_loop

read_done:
    cmp r11, 0
    je exit_now
    xor rbx, rbx
    xor r12, r12
    xor r14, r14
    lea r15, [rip + outbuf]
    lea rdi, [rip + inbuf]

encode_main:
    mov rax, r11
    sub rax, rbx
    cmp rax, 3
    jb encode_rem
    movzx rax, byte ptr [rdi + rbx]
    movzx rdx, byte ptr [rdi + rbx + 1]
    movzx r8, byte ptr [rdi + rbx + 2]
    mov rcx, rax
    shr rcx, 2
    and rcx, 63
    call emit_index
    mov rcx, rax
    and rcx, 3
    shl rcx, 4
    mov r9, rdx
    shr r9, 4
    and r9, 15
    or rcx, r9
    call emit_index
    mov rcx, rdx
    and rcx, 15
    shl rcx, 2
    mov r9, r8
    shr r9, 6
    and r9, 3
    or rcx, r9
    call emit_index
    mov rcx, r8
    and rcx, 63
    call emit_index
    add rbx, 3
    jmp encode_main

encode_rem:
    mov rax, r11
    sub rax, rbx
    cmp rax, 0
    je finalise
    cmp rax, 1
    je rem1

rem2:
    movzx rax, byte ptr [rdi + rbx]
    movzx rdx, byte ptr [rdi + rbx + 1]
    mov rcx, rax
    shr rcx, 2
    and rcx, 63
    call emit_index
    mov rcx, rax
    and rcx, 3
    shl rcx, 4
    mov r9, rdx
    shr r9, 4
    and r9, 15
    or rcx, r9
    call emit_index
    mov rcx, rdx
    and rcx, 15
    shl rcx, 2
    call emit_index
    mov rcx, 61
    call emit_char
    mov rcx, 61
    call emit_char
    jmp finalise

rem1:
    movzx rax, byte ptr [rdi + rbx]
    mov rcx, rax
    shr rcx, 2
    and rcx, 63
    call emit_index
    mov rcx, rax
    and rcx, 3
    shl rcx, 4
    call emit_index
    mov rcx, 61
    call emit_char
    mov rcx, 61
    call emit_char
    jmp finalise

finalise:
    cmp r14, 0
    je write_out
    call emit_nl

write_out:
    cmp r12, 0
    je exit_now
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r12
    syscall

exit_now:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_index:
    movzx rcx, byte ptr [rip + alphabet + rcx]
    jmp emit_char

emit_char:
    mov byte ptr [r15 + r12], cl
    inc r12
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [r15 + r12], 10
    inc r12
    xor r14, r14
1:
    ret

emit_nl:
    mov byte ptr [r15 + r12], 10
    inc r12
    xor r14, r14
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .space 49152
.align 16
outbuf:
    .space 70000
