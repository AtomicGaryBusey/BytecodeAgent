.text
.globl _start

_start:
    xor r12, r12
    lea rdi, [rip + inbuf]

read_loop:
    mov rax, 0
    mov rsi, rdi
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 49152
    jae read_done
    jmp read_loop

read_done:
    test r12, r12
    jz exit_zero

    xor rbx, rbx
    xor r13, r13
    xor r14, r14
    lea rsi, [rip + outbuf]
    lea r15, [rip + alphabet]
    lea rdi, [rip + inbuf]

encode_loop:
    mov rdx, r12
    sub rdx, rbx
    cmp rdx, 3
    jb encode_remainder

    movzx r8d, byte ptr [rdi + rbx]
    movzx r9d, byte ptr [rdi + rbx + 1]
    movzx r10d, byte ptr [rdi + rbx + 2]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    mov eax, r9d
    shr eax, 4
    and eax, 15
    or ecx, eax
    call emit_idx

    mov ecx, r9d
    and ecx, 15
    shl ecx, 2
    mov eax, r10d
    shr eax, 6
    and eax, 3
    or ecx, eax
    call emit_idx

    mov ecx, r10d
    and ecx, 63
    call emit_idx

    add rbx, 3
    jmp encode_loop

encode_remainder:
    mov rdx, r12
    sub rdx, rbx
    test rdx, rdx
    je finish
    cmp rdx, 1
    je rem_one

rem_two:
    movzx r8d, byte ptr [rdi + rbx]
    movzx r9d, byte ptr [rdi + rbx + 1]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    mov eax, r9d
    shr eax, 4
    and eax, 15
    or ecx, eax
    call emit_idx

    mov ecx, r9d
    and ecx, 15
    shl ecx, 2
    call emit_idx

    mov al, '='
    call emit_char

    mov al, '='
    call emit_char
    jmp finish

rem_one:
    movzx r8d, byte ptr [rdi + rbx]

    mov ecx, r8d
    shr ecx, 2
    and ecx, 63
    call emit_idx

    mov ecx, r8d
    and ecx, 3
    shl ecx, 4
    call emit_idx

    mov al, '='
    call emit_char

    mov al, '='
    call emit_char
    jmp finish

finish:
    test r14, r14
    jz write_output

    mov byte ptr [rsi + r13], 10
    inc r13

write_output:
    test r13, r13
    jz exit_zero

    xor rbx, rbx

write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle exit_zero
    add rbx, rax
    cmp rbx, r13
    jb write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_idx:
    movzx eax, byte ptr [r15 + rcx]
    jmp emit_char

emit_char:
    mov byte ptr [rsi + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [rsi + r13], 10
    inc r13
    xor r14, r14
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .space 49152

.align 16
outbuf:
    .space 70000
