.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 49152
    jb read_loop

read_done:
    test r12, r12
    jz exit_zero

    xor r13, r13
    xor r14, r14
    xor rsi, rsi

encode_loop:
    mov rax, r12
    sub rax, rsi
    cmp rax, 3
    jb encode_remainder

    movzx eax, byte ptr [inbuf + rsi]
    movzx ebx, byte ptr [inbuf + rsi + 1]
    movzx ecx, byte ptr [inbuf + rsi + 2]

    mov edx, eax
    shr edx, 2
    call emit_index

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov eax, ebx
    shr eax, 4
    and eax, 15
    or edx, eax
    call emit_index

    mov edx, ebx
    and edx, 15
    shl edx, 2
    mov eax, ecx
    shr eax, 6
    and eax, 3
    or edx, eax
    call emit_index

    mov edx, ecx
    and edx, 63
    call emit_index

    add rsi, 3
    jmp encode_loop

encode_remainder:
    mov rax, r12
    sub rax, rsi
    test rax, rax
    jz finalize
    cmp rax, 1
    je rem_one

rem_two:
    movzx eax, byte ptr [inbuf + rsi]
    movzx ebx, byte ptr [inbuf + rsi + 1]

    mov edx, eax
    shr edx, 2
    call emit_index

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov eax, ebx
    shr eax, 4
    and eax, 15
    or edx, eax
    call emit_index

    mov edx, ebx
    and edx, 15
    shl edx, 2
    call emit_index

    mov al, '='
    call emit_char
    jmp finalize

rem_one:
    movzx eax, byte ptr [inbuf + rsi]

    mov edx, eax
    shr edx, 2
    call emit_index

    mov edx, eax
    and edx, 3
    shl edx, 4
    call emit_index

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char

finalize:
    test r13, r13
    jz exit_zero

    test r14, r14
    jz write_output

    mov byte ptr [outbuf + r13], 10
    inc r13
    xor r14, r14

write_output:
    xor rbx, rbx

write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    syscall
    add rbx, rax
    cmp rbx, r13
    jb write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_index:
    mov rax, rdx
    lea r11, [rip + alphabet]
    movzx eax, byte ptr [r11 + rax]
    jmp emit_char

emit_char:
    mov byte ptr [outbuf + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [outbuf + r13], 10
    inc r13
    xor r14, r14
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .space 49152

.align 16
outbuf:
    .space 70000
