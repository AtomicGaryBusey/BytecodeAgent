.text
.globl _start

_start:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
read_loop:
    mov x0, #0
    mov x1, x9
    add x1, x1, x10
    mov x2, #32768
    sub x2, x2, x10
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x10, x10, x0
    cmp x10, #32768
    b.lt read_loop
read_done:
    adrp x12, line_descs
    add x12, x12, :lo12:line_descs
    mov x13, #0
    mov x11, #0
    cmp x10, #0
    b.eq no_lines
parse_outer:
    cmp x11, x10
    b.ge parse_done
    mov x14, x11
parse_inner:
    cmp x11, x10
    b.ge save_final
    ldrb w15, [x9, x11]
    cmp w15, #10
    b.eq save_line
    add x11, x11, #1
    b parse_inner
save_line:
    sub x15, x11, x14
    lsl x16, x13, #3
    add x16, x12, x16
    str w14, [x16]
    str w15, [x16, #4]
    add x13, x13, #1
    add x11, x11, #1
    b parse_outer
save_final:
    sub x15, x10, x14
    lsl x16, x13, #3
    add x16, x12, x16
    str w14, [x16]
    str w15, [x16, #4]
    add x13, x13, #1
    b sort_start
parse_done:
sort_start:
    adrp x16, line_count
    add x16, x16, :lo12:line_count
    str x13, [x16]
    mov x0, x12
    mov x1, x13
    bl sort_descs
    adrp x12, line_descs
    add x12, x12, :lo12:line_descs
    adrp x13, line_count
    add x13, x13, :lo12:line_count
    ldr x13, [x13]
    mov x14, #0
print_loop:
    cmp x14, x13
    b.ge exit_prog
    lsl x15, x14, #3
    add x15, x12, x15
    ldr w16, [x15]
    ldr w17, [x15, #4]
    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x16, uxtw
    mov x2, x17
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x14, x14, #1

no_lines:
    adrp x16, line_count
    add x16, x16, :lo12:line_count
    mov x17, #0
    str x17, [x16]
    b exit_prog

    mov x0, #0
    mov x8, #93
    svc #0

sort_descs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, #1
outer_loop:
    cmp x21, x20
    b.ge done_sort
    mov x22, x21
inner_loop:
    cmp x22, #0
    b.le inner_done
    sub x23, x22, #1
    lsl x23, x23, #3
    add x23, x19, x23
    lsl x24, x22, #3
    add x24, x19, x24
    mov x0, x23
    mov x1, x24
    bl cmp_desc
    cmp x0, #0
    b.le inner_done
    ldr x25, [x23]
    ldr x26, [x24]
    str x26, [x23]
    str x25, [x24]
    sub x22, x22, #1
    b inner_loop
inner_done:
    add x21, x21, #1
    b outer_loop
done_sort:
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

cmp_desc:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!

    ldr w19, [x0]
    ldr w20, [x0, #4]
    ldr w21, [x1]
    ldr w22, [x1, #4]

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x19, uxtw
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    add x24, x24, x21, uxtw

    cmp w20, w22
    csel w25, w20, w22, ls
    mov w26, #0
cmp_loop:
    cmp w26, w25
    b.hs cmp_len
    add x9, x23, x26
    ldrb w9, [x9]
    add x10, x24, x26
    ldrb w10, [x10]
    cmp w9, w10
    b.eq same_byte
    cset x0, hi
    cset x1, lo
    sub x0, x0, x1
    b cmp_done

    add w26, w26, #1
    b cmp_loop
cmp_len:
    cmp w20, w22
    b.lt len1_lt
    b.gt len1_gt
    mov x0, #0
    b cmp_done
len1_lt:
    mov x0, #1
    neg x0, x0
    b cmp_done
len1_gt:
    mov x0, #1
cmp_done:
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
