.globl _start
.text

_start:
    call read_all
    test rax, rax
    jz exit_zero
    mov rdi, rax
    call parse_lines
    test rax, rax
    jz exit_zero
    mov r12, rax
    mov rdi, rax
    call sort_lines
    mov rdi, r12
    call write_sorted
exit_zero:
    xor edi, edi
    mov rax, 60
    syscall

read_all:
    lea r8, [rip + input_buf]
    xor ebx, ebx
1:
    cmp rbx, 32768
    jae 2f
    mov rax, 0
    mov rdi, 0
    mov rsi, r8
    add rsi, rbx
    mov rdx, 32768
    sub rdx, rbx
    syscall
    test rax, rax
    jz 2f
    add rbx, rax
    cmp rbx, 32768
    jae 2f
    jmp 1b
2:
    mov rax, rbx
    ret

parse_lines:
    xor ecx, ecx
    lea r13, [rip + input_buf]
    mov r14, r13
    mov r15, r13
    add r15, rdi
.loop:
    cmp r14, r15
    jae .after_loop
    cmp byte ptr [r14], 0x0A
    jne .next
    mov rax, r14
    sub rax, r13
    mov qword ptr [rip + line_ptrs + rcx*8], r13
    mov dword ptr [rip + line_lens + rcx*4], eax
    inc rcx
    inc r14
    mov r13, r14
    jmp .loop
.next:
    inc r14
    jmp .loop
.after_loop:
    cmp r13, r15
    je .done
    mov rax, r15
    sub rax, r13
    mov qword ptr [rip + line_ptrs + rcx*8], r13
    mov dword ptr [rip + line_lens + rcx*4], eax
    inc rcx
.done:
    mov rax, rcx
    ret

sort_lines:
    cmp rdi, 1
    jle .done
    mov rcx, 1
.loop_i:
    cmp rcx, rdi
    jge .done
    mov r8, rcx
.loop_j:
    test r8, r8
    jz .next_i
    mov r9, r8
    dec r9
    mov rsi, r8
    mov rdi, r9
    call compare_indices
    test rax, rax
    jle .next_i
    lea r10, [rip + line_ptrs]
    lea r11, [rip + line_lens]
    mov rbx, qword ptr [r10 + r9*8]
    mov rdx, qword ptr [r10 + r8*8]
    mov qword ptr [r10 + r9*8], rdx
    mov qword ptr [r10 + r8*8], rbx
    mov eax, dword ptr [r11 + r9*4]
    mov edx, dword ptr [r11 + r8*4]
    mov dword ptr [r11 + r9*4], edx
    mov dword ptr [r11 + r8*4], eax
    dec r8
    jmp .loop_j
.next_i:
    inc rcx
    jmp .loop_i
.done:
    ret

compare_indices:
    lea r8, [rip + line_ptrs]
    lea r9, [rip + line_lens]
    mov r10, qword ptr [r8 + rdi*8]
    mov r11, qword ptr [r8 + rsi*8]
    mov eax, dword ptr [r9 + rdi*4]
    mov ebx, dword ptr [r9 + rsi*4]
    mov ecx, eax
    cmp eax, ebx
    jbe .len_min_done
    mov ecx, ebx
.len_min_done:
    xor edx, edx
.loop:
    cmp edx, ecx
    jae .after_loop
    movzx r8d, byte ptr [r10 + rdx]
    movzx r9d, byte ptr [r11 + rdx]
    cmp r8d, r9d
    je .next_byte
    cmp r8d, r9d
    jb .less
    mov rax, 1
    ret
.next_byte:
    inc edx
    jmp .loop
.less:
    mov rax, -1
    ret
.after_loop:
    cmp eax, ebx
    jb .less_len
    ja .greater_len
    xor eax, eax
    ret
.less_len:
    mov rax, -1
    ret
.greater_len:
    mov rax, 1
    ret

write_sorted:
    xor r8d, r8d
    lea r9, [rip + line_ptrs]
    lea r10, [rip + line_lens]
.loop:
    cmp r8d, edi
    jae .done
    mov rax, 1
    mov rdi, 1
    mov rsi, qword ptr [r9 + r8*8]
    mov rdx, dword ptr [r10 + r8*4]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r8d
    jmp .loop
.done:
    ret

.section .rodata
newline:
    .byte 0x0A

.section .bss
    .align 8
input_buf:
    .zero 32768
line_ptrs:
    .zero 1000 * 8
line_lens:
    .zero 1000 * 4
