.globl _start

.section .bss
    .align 8
input_buf:
    .space 32768
    .align 8
line_ptrs:
    .space 1000 * 8
    .align 4
line_lens:
    .space 1000 * 4

.section .rodata
newline:
    .byte 0x0A

.text

read_all:
    push rbx
    push r12
    xor ebx, ebx
    lea r12, [rip + input_buf]
1:
    cmp rbx, 32768
    jae 2f
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + rbx]
    mov rdx, 32768
    sub rdx, rbx
    syscall
    test rax, rax
    jz 2f
    add rbx, rax
    cmp rbx, 32768
    jae 2f
    jmp 1b
2:
    mov rax, rbx
    pop r12
    pop rbx
    ret

parse_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    xor r13d, r13d
    lea r14, [rip + input_buf]
    mov r15, r14
    mov r9, r14
    add r9, r12
    lea rbx, [rip + line_ptrs]
    lea rsi, [rip + line_lens]
.loop:
    cmp r15, r9
    jae .after
    cmp byte ptr [r15], 0x0A
    jne .advance
    mov qword ptr [rbx + r13*8], r15
    mov rax, r15
    sub rax, r14
    mov dword ptr [rsi + r13*4], eax
    inc r13
    inc r15
    jmp .loop
.advance:
    inc r15
    jmp .loop
.after:
    cmp r14, r9
    je .done
    cmp r15, r9
    je .done
    mov qword ptr [rbx + r13*8], r14
    mov rax, r9
    sub rax, r14
    mov dword ptr [rsi + r13*4], eax
    inc r13
.done:
    mov rax, r13
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

sort_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    cmp r12, 1
    jbe .done
    lea r15, [rip + line_ptrs]
    lea rbx, [rip + line_lens]
    mov r13, 1
.outer:
    cmp r13, r12
    jae .done
    mov r14, r13
.inner:
    test r14, r14
    jz .next_i
    mov r9, r14
    dec r9
    mov rsi, qword ptr [r15 + r9*8]
    mov edx, dword ptr [rbx + r9*4]
    mov rdi, qword ptr [r15 + r14*8]
    mov ecx, dword ptr [rbx + r14*4]
    mov r8d, edx
    mov r10d, r8d
    mov r9d, ecx
    cmp r10d, r9d
    cmovb r10d, r9d
    xor r11d, r11d
.compare_loop:
    cmp r11d, r10d
    jae .compare_len
    movzx eax, byte ptr [rsi + r11]
    movzx r8d, byte ptr [rdi + r11]
    cmp al, r8b
    jb .done_inner
    ja .swap
    inc r11d
    jmp .compare_loop
.compare_len:
    cmp edx, ecx
    jb .done_inner
    ja .swap
    jmp .done_inner
.swap:
    mov r8, qword ptr [r15 + r9*8]
    mov r10, qword ptr [r15 + r14*8]
    mov qword ptr [r15 + r9*8], r10
    mov qword ptr [r15 + r14*8], r8
    mov r8d, dword ptr [rbx + r9*4]
    mov r10d, dword ptr [rbx + r14*4]
    mov dword ptr [rbx + r9*4], r10d
    mov dword ptr [rbx + r14*4], r8d
    dec r14
    jmp .inner
.done_inner:
    jmp .next_i
.next_i:
    inc r13
    jmp .outer
.done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

write_sorted:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    xor r13d, r13d
    lea r14, [rip + line_ptrs]
    lea r15, [rip + line_lens]
.loop:
    cmp r13d, r12d
    jae .done
    mov rsi, qword ptr [r14 + r13*8]
    mov edx, dword ptr [r15 + r13*4]
    mov edi, 1
    mov rax, 1
    syscall
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov rax, 1
    syscall
    inc r13
    jmp .loop
.done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

_start:
    call read_all
    test rax, rax
    jz .exit
    mov rdi, rax
    call parse_lines
    mov rbx, rax
    test rbx, rbx
    jz .exit
    mov rdi, rbx
    call sort_lines
    mov rdi, rbx
    call write_sorted
.exit:
    xor edi, edi
    mov rax, 60
    syscall
