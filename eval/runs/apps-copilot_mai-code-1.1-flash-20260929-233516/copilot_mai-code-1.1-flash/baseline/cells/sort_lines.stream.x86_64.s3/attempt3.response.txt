.globl _start

.section .bss
    .align 16
input_buf:
    .space 32768
    .align 8
line_ptrs:
    .space 1000 * 8
    .align 4
line_lens:
    .space 1000 * 4

.section .rodata
newline:
    .byte 0x0A

.text

read_all:
    push rbx
    push r12
    push r13
    xor r12d, r12d
    lea r13, [rip + input_buf]

read_loop:
    cmp r12, 32768
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add r13, rax
    jmp read_loop

read_done:
    mov rax, r12
    pop r13
    pop r12
    pop rbx
    ret

parse_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    lea r14, [rip + input_buf]
    lea r15, [rip + line_ptrs]
    xor r13d, r13d
    mov rbx, r14
    xor ecx, ecx

parse_loop:
    cmp rcx, r12
    jae parse_after
    mov al, byte ptr [r14 + rcx]
    cmp al, 0x0A
    jne parse_next
    mov qword ptr [r15 + r13*8], rbx
    mov r9, rbx
    sub r9, r14
    mov rdx, rcx
    sub rdx, r9
    lea r9, [rip + line_lens]
    mov dword ptr [r9 + r13*4], edx
    inc r13
    lea rbx, [r14 + rcx + 1]
    inc rcx
    jmp parse_loop

parse_next:
    inc rcx
    jmp parse_loop

parse_after:
    mov r9, rbx
    sub r9, r14
    cmp r9, r12
    jae parse_done
    mov qword ptr [r15 + r13*8], rbx
    mov rdx, r12
    sub rdx, r9
    lea r9, [rip + line_lens]
    mov dword ptr [r9 + r13*4], edx
    inc r13

parse_done:
    mov rax, r13
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

sort_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    cmp r12, 1
    jbe sort_done
    mov r13, 1

sort_outer:
    cmp r13, r12
    jae sort_done
    mov r14, r13

sort_inner:
    test r14, r14
    jz sort_next
    lea r15, [rip + line_ptrs]
    lea rbx, [rip + line_lens]
    mov r10, r14
    dec r10
    mov rsi, qword ptr [r15 + r10*8]
    mov rdi, qword ptr [r15 + r14*8]
    mov ecx, dword ptr [rbx + r10*4]
    mov edx, dword ptr [rbx + r14*4]
    mov r8, rcx
    cmp r8, rdx
    jbe compare_min_set
    mov r8, rdx

compare_min_set:
    xor r9d, r9d

compare_loop:
    cmp r9, r8
    jae compare_len
    movzx eax, byte ptr [rsi + r9]
    movzx r11d, byte ptr [rdi + r9]
    cmp al, r11b
    jb sort_no_swap
    ja sort_swap
    inc r9
    jmp compare_loop

compare_len:
    cmp ecx, edx
    jb sort_no_swap
    ja sort_swap
    jmp sort_no_swap

sort_swap:
    mov r11, qword ptr [r15 + r10*8]
    mov rax, qword ptr [r15 + r14*8]
    mov qword ptr [r15 + r10*8], rax
    mov qword ptr [r15 + r14*8], r11
    mov r11d, dword ptr [rbx + r10*4]
    mov eax, dword ptr [rbx + r14*4]
    mov dword ptr [rbx + r10*4], eax
    mov dword ptr [rbx + r14*4], r11d
    dec r14
    jmp sort_inner

sort_no_swap:
    inc r13
    jmp sort_outer

sort_next:
    inc r13
    jmp sort_outer

sort_done:
    mov rax, r12
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_lines:
    push r12
    push r13
    push r14
    mov r12, rdi
    test r12, r12
    jz print_done
    xor r13d, r13d
    lea r14, [rip + line_ptrs]

print_loop:
    cmp r13, r12
    jae print_done
    mov rax, 1
    mov rdi, 1
    mov rsi, qword ptr [r14 + r13*8]
    mov edx, dword ptr [rip + line_lens + r13*4]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r13
    jmp print_loop

print_done:
    pop r14
    pop r13
    pop r12
    ret

_start:
    call read_all
    mov rdi, rax
    call parse_lines
    mov rdi, rax
    call sort_lines
    mov rdi, rax
    call print_lines
    xor edi, edi
    mov rax, 60
    syscall
