.globl _start

INPUT_MAX = 32768
MAX_LINES = 1000

.section .bss
    .align 16
input_buf:
    .space INPUT_MAX
    .align 8
line_ptrs:
    .space MAX_LINES * 8
    .align 4
line_lens:
    .space MAX_LINES * 4

.section .rodata
newline:
    .byte 0x0A

.text

read_all:
    push r12
    push r13
    push r14
    xor r12d, r12d
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add r13, rax
    jmp read_loop
read_done:
    mov rax, r12
    pop r14
    pop r13
    pop r12
    ret

parse_lines:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    xor r13d, r13d
    lea r14, [rip + input_buf]
    lea r15, [rip + line_ptrs]
    xor ebx, ebx
    xor ecx, ecx
parse_loop:
    cmp rcx, r12
    jae parse_tail
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 0x0A
    jne parse_next
    lea rax, [r14 + rbx]
    mov qword ptr [r15 + r13*8], rax
    mov rdx, rcx
    sub rdx, rbx
    lea r9, [rip + line_lens]
    mov dword ptr [r9 + r13*4], edx
    inc r13
    mov rbx, rcx
    inc rbx
    inc rcx
    jmp parse_loop
parse_next:
    inc rcx
    jmp parse_loop
parse_tail:
    cmp rbx, r12
    jae parse_done
    lea rax, [r14 + rbx]
    mov qword ptr [r15 + r13*8], rax
    mov rdx, r12
    sub rdx, rbx
    lea r9, [rip + line_lens]
    mov dword ptr [r9 + r13*4], edx
    inc r13
parse_done:
    mov rax, r13
    pop r15
    pop r14
    pop r13
    pop r12
    ret

compare_lines:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov r8, r14
    cmp r8, r15
    jbe compare_min
    mov r8, r15
compare_min:
    xor r9d, r9d
compare_loop:
    cmp r9, r8
    jae compare_len
    movzx eax, byte ptr [r12 + r9]
    movzx edx, byte ptr [r13 + r9]
    cmp al, dl
    jb line_less
    ja line_greater
    inc r9
    jmp compare_loop
compare_len:
    cmp r14, r15
    jb line_less
    ja line_greater
    xor eax, eax
    jmp compare_done
line_less:
    mov eax, -1
    jmp compare_done
line_greater:
    mov eax, 1
compare_done:
    pop r15
    pop r14
    pop r13
    pop r12
    ret

sort_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    cmp r12, 1
    jbe sort_done
    mov r13d, 1
sort_outer:
    cmp r13, r12
    jae sort_done
    mov r14, r13
sort_inner:
    test r14, r14
    jz sort_next
    lea rbx, [rip + line_ptrs]
    lea r15, [rip + line_lens]
    mov rdi, qword ptr [rbx + r14*8 - 8]
    mov rsi, qword ptr [rbx + r14*8]
    mov edx, dword ptr [r15 + r14*4 - 4]
    mov ecx, dword ptr [r15 + r14*4]
    call compare_lines
    test eax, eax
    jle sort_no_swap
    mov r10, qword ptr [rbx + r14*8 - 8]
    mov r11, qword ptr [rbx + r14*8]
    mov qword ptr [rbx + r14*8 - 8], r11
    mov qword ptr [rbx + r14*8], r10
    mov r10d, dword ptr [r15 + r14*4 - 4]
    mov r11d, dword ptr [r15 + r14*4]
    mov dword ptr [r15 + r14*4 - 4], r11d
    mov dword ptr [r15 + r14*4], r10d
    dec r14
    jmp sort_inner
sort_no_swap:
    inc r13
    jmp sort_outer
sort_next:
    inc r13
    jmp sort_outer
sort_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_lines:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    test r12, r12
    jz print_done
    lea r14, [rip + line_ptrs]
    lea r15, [rip + line_lens]
    xor r13d, r13d
print_loop:
    cmp r13, r12
    jae print_done
    mov rax, 1
    mov rdi, 1
    mov rsi, qword ptr [r14 + r13*8]
    mov edx, dword ptr [r15 + r13*4]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r13
    jmp print_loop
print_done:
    pop r15
    pop r14
    pop r13
    pop r12
    ret

_start:
    call read_all
    mov r12, rax
    test r12, r12
    jz exit_zero
    mov rdi, r12
    call parse_lines
    mov r13, rax
    mov rdi, r13
    call sort_lines
    mov rdi, r13
    call print_lines
exit_zero:
    xor edi, edi
    mov rax, 60
    syscall
