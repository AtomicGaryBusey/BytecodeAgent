.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    cmp rdx, 0
    jle read_done
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 49152
    jl read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

encode_loop:
    mov rax, r12
    sub rax, r13
    cmp rax, 3
    jb encode_tail

    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]
    movzx r10d, byte ptr [rsi + r13 + 2]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 1f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
1:
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 2f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
2:
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 3f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
3:
    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 4f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
4:
    add r13, 3
    jmp encode_loop

encode_tail:
    mov rax, r12
    sub rax, r13
    cmp rax, 2
    jne tail_one

    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 5f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
5:
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 6f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
6:
    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 7f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
7:
    mov al, '='
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 8f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
8:
    test r15, r15
    jz write_output
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    jmp write_output

tail_one:
    mov rax, r12
    sub rax, r13
    cmp rax, 1
    jne write_output

    movzx r8d, byte ptr [rsi + r13]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 9f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d
9:
    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [rip + base64_alphabet + rax]
    mov byte ptr [rdi + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 10f
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    xor r15d, r15d

    mov al, '='
    mov byte ptr [rdi + r14], al
    inc r14

    cmp r15, 76

    jz write_output

    inc r14

write_output:
    cmp r14, 0
    je exit_zero
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r14
    syscall

exit_zero:
    xor edi, edi
    mov rax, 60
    syscall
