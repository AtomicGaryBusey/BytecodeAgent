.section .rodata
.align 16
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000

.text
.globl _start

append_char:
    mov byte ptr [rbx + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne 1f
    mov byte ptr [rbx + r14], 10
    inc r14
    xor r15d, r15d
1:
    ret

_start:
    xor r12d, r12d

read_loop:
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    cmp rdx, 0
    jle read_done
    syscall
    cmp rax, 0
    js fail_exit
    je read_done
    add r12, rax
    cmp r12, 49152
    jl read_loop

read_done:
    cmp r12, 0
    je exit_zero

    lea rsi, [rip + inbuf]
    lea rbx, [rip + outbuf]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rcx, [rip + base64_alphabet]

encode_loop:
    mov rax, r12
    sub rax, r13
    cmp rax, 3
    jb encode_tail

    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]
    movzx r10d, byte ptr [rsi + r13 + 2]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [rcx + rax]
    call append_char

    add r13, 3
    jmp encode_loop

encode_tail:
    mov rax, r12
    sub rax, r13
    cmp rax, 2
    je encode_tail_2
    cmp rax, 1
    je encode_tail_1
    jmp finish_encoding

encode_tail_1:
    movzx r8d, byte ptr [rsi + r13]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov al, '='
    call append_char
    mov al, '='
    call append_char
    jmp finish_encoding

encode_tail_2:
    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [rcx + rax]
    call append_char

    mov al, '='
    call append_char

finish_encoding:
    cmp r15d, 0
    je do_write
    mov byte ptr [rbx + r14], 10
    inc r14

do_write:
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, r14
    syscall

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

fail_exit:
    mov rax, 60
    mov edi, 1
    syscall
