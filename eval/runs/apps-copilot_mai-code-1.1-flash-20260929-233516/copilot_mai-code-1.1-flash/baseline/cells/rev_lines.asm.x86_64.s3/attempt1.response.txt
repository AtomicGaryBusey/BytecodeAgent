.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    cmp rdx, 0
    jle read_done
    syscall
    cmp rax, 0
    jl fail_exit
    add r12, rax
    test rax, rax
    jne read_loop
    jmp read_done

fail_exit:
    mov rax, 60
    mov rdi, 1
    syscall

read_done:
    cmp r12, 0
    je exit_zero

    xor r13, r13
    xor r14, r14

process_loop:
    cmp r14, r12
    je final_line

    lea rbx, [rip + input_buf]
    add rbx, r14
    cmp byte ptr [rbx], 0x0A
    jne not_new_line

    mov rdi, r13
    mov rsi, r14
    sub rsi, r13
    call emit_reversed_line

    mov r13, r14
    inc r13
    inc r14
    jmp process_loop

not_new_line:
    inc r14
    jmp process_loop

final_line:
    mov rsi, r12
    sub rsi, r13
    cmp rsi, 0
    je exit_zero

    mov rdi, r13
    call emit_reversed_line
    jmp exit_zero

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_reversed_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rbp

    mov rbx, rdi
    mov r12, rsi

    cmp r12, 0
    je emit_newline_only

    lea rbp, [rip + input_buf]
    lea r13, [rbp + rbx]
    lea r14, [r13 + r12]
    dec r14

reverse_loop:
    cmp r13, r14
    jae after_reverse
    mov al, byte ptr [r13]
    mov dl, byte ptr [r14]
    mov byte ptr [r13], dl
    mov byte ptr [r14], al
    inc r13
    dec r14
    jmp reverse_loop

after_reverse:
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, r12
    call write_all

emit_newline:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all
    jmp emit_done

emit_newline_only:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all

emit_done:
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

write_all:
    cmp rdx, 0
    je write_done

write_loop:
    mov rax, 1
    syscall
    cmp rax, 0
    jl write_error
    test rax, rax
    jz write_error
    sub rdx, rax
    add rsi, rax
    cmp rdx, 0
    jne write_loop

write_done:
    ret

write_error:
    mov rax, 60
    mov rdi, 1
    syscall

.section .bss
.align 16
input_buf:
    .space 65536

.section .rodata
newline:
    .byte 0x0A
