.text
.globl _start

_start:
    lea r8, [rip + input_buf]
    lea r9, [rip + jump_to]
    lea r10, [rip + open_stack]
    lea r11, [rip + tape]
    mov r15, r11

    xor r12, r12
read_loop:
    cmp r12, 65536
    je read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r14, r14
    mov r13, r12
    mov rbp, r12

find_loop:
    cmp r14, r12
    jae no_bang
    cmp byte ptr [r8 + r14], 0x21
    je bang_found
    inc r14
    jmp find_loop

no_bang:
    jmp parse

bang_found:
    mov r13, r14
    mov rbp, r14
    inc rbp

parse:
    xor r14, r14
    xor rsi, rsi

parse_loop:
    cmp r14, r13
    je parse_done
    mov al, byte ptr [r8 + r14]
    cmp al, 0x5b
    je parse_open
    cmp al, 0x5d
    je parse_close
    inc r14
    jmp parse_loop

parse_open:
    mov dword ptr [r10 + rsi*4], r14d
    inc rsi
    inc r14
    jmp parse_loop

parse_close:
    test rsi, rsi
    jz unbalanced
    dec rsi
    mov eax, dword ptr [r10 + rsi*4]
    mov edx, r14d
    lea ecx, [rax + 1]
    mov dword ptr [r9 + rdx*4], ecx
    lea ecx, [rdx + 1]
    mov dword ptr [r9 + rax*4], ecx
    inc r14
    jmp parse_loop

parse_done:
    test rsi, rsi
    jnz unbalanced

    xor r14, r14
    xor rbx, rbx

exec_loop:
    cmp r14, r13
    jae exec_done
    mov al, byte ptr [r8 + r14]
    cmp al, 0x3e
    je ptr_right
    cmp al, 0x3c
    je ptr_left
    cmp al, 0x2b
    je cell_inc
    cmp al, 0x2d
    je cell_dec
    cmp al, 0x2e
    je output_byte
    cmp al, 0x2c
    je input_byte
    cmp al, 0x5b
    je lbracket
    cmp al, 0x5d
    je rbracket
    inc r14
    jmp exec_loop

ptr_right:
    inc r15
    inc r14
    jmp exec_loop

ptr_left:
    dec r15
    inc r14
    jmp exec_loop

cell_inc:
    mov dl, byte ptr [r15]
    inc dl
    mov byte ptr [r15], dl
    inc r14
    jmp exec_loop

cell_dec:
    mov dl, byte ptr [r15]
    dec dl
    mov byte ptr [r15], dl
    inc r14
    jmp exec_loop

output_byte:
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, 1
    syscall
    inc r14
    jmp exec_loop

input_byte:
    mov rax, rbp
    add rax, rbx
    cmp rax, r12
    jae input_zero
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [r15], dl
    inc rbx
    inc r14
    jmp exec_loop

input_zero:
    mov byte ptr [r15], 0
    inc r14
    jmp exec_loop

lbracket:
    mov dl, byte ptr [r15]
    test dl, dl
    jne lbracket_skip
    mov eax, dword ptr [r9 + r14*4]
    mov r14d, eax
    jmp exec_loop

lbracket_skip:
    inc r14
    jmp exec_loop

rbracket:
    mov dl, byte ptr [r15]
    test dl, dl
    je rbracket_skip
    mov eax, dword ptr [r9 + r14*4]
    mov r14d, eax
    jmp exec_loop

rbracket_skip:
    inc r14
    jmp exec_loop

exec_done:
    xor rdi, rdi
    mov rax, 60
    syscall

unbalanced:
    mov rdi, 1
    mov rax, 60
    syscall

.bss
.align 16
input_buf:
    .space 65536

.align 16
tape:
    .space 30000

.align 16
jump_to:
    .space 262144

.align 16
open_stack:
    .space 4000
