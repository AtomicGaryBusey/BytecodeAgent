.text
.align 2
.globl _start

_start:
    bl read_all
    bl parse_lines
    bl sort_lines
    bl write_output
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    mov x19, #0
read_all_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #32768
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_all_done
    add x19, x19, x0
    cmp x19, #32768
    b.ge read_all_done
    b read_all_loop
read_all_done:
    adrp x0, total_bytes
    add x0, x0, :lo12:total_bytes
    str x19, [x0]
    mov x0, x19
    ret

parse_lines:
    adrp x19, total_bytes
    add x19, x19, :lo12:total_bytes
    ldr x19, [x19]
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    add x21, x20, x19
    mov x22, x20
    mov x23, x20
    mov x24, #0
parse_loop:
    cmp x23, x21
    b.ge parse_done
    ldrb w25, [x23]
    cmp w25, #10
    b.ne parse_not_nl
    sub x26, x23, x22
    adrp x27, line_recs
    add x27, x27, :lo12:line_recs
    add x28, x27, x24, lsl #4
    stp x22, x26, [x28]
    add x24, x24, #1
    add x23, x23, #1
    mov x22, x23
    b parse_loop
parse_not_nl:
    add x23, x23, #1
    b parse_loop
parse_done:
    cmp x22, x21
    b.eq parse_no_final
    sub x26, x21, x22
    adrp x27, line_recs
    add x27, x27, :lo12:line_recs
    add x28, x27, x24, lsl #4
    stp x22, x26, [x28]
    add x24, x24, #1
parse_no_final:
    adrp x0, line_count
    add x0, x0, :lo12:line_count
    str x24, [x0]
    ret

sort_lines:
    adrp x19, line_count
    add x19, x19, :lo12:line_count
    ldr x19, [x19]
    cmp x19, #1
    ble sort_done
    mov x20, #1
outer_loop:
    cmp x20, x19
    b.ge sort_done
    adrp x21, line_recs
    add x21, x21, :lo12:line_recs
    add x22, x21, x20, lsl #4
    adrp x23, key_rec
    add x23, x23, :lo12:key_rec
    ldp x24, x25, [x22]
    stp x24, x25, [x23]
    sub x26, x20, #1
inner_loop:
    cmp x26, #0
    blt insert_key
    add x27, x21, x26, lsl #4
    ldp x0, x1, [x27]
    ldp x2, x3, [x23]
    mov x4, x0
    mov x5, x1
    mov x6, x2
    mov x7, x3
    mov x8, #0
compare_loop:
    cmp x8, x5
    b.hs a_exhausted
    cmp x8, x7
    b.hs b_exhausted
    add x9, x4, x8
    add x10, x6, x8
    ldrb w11, [x9]
    ldrb w12, [x10]
    cmp w11, w12
    b.lt a_less
    b.gt a_greater
    add x8, x8, #1
    b compare_loop
a_exhausted:
    cmp x8, x7
    b.lt a_less
    b.eq cmp_equal
    mov x14, #1
    b compare_done
b_exhausted:
    mov x14, #1
    b compare_done
a_less:
    mov x14, #-1
    b compare_done
a_greater:
    mov x14, #1
    b compare_done
cmp_equal:
    mov x14, #0
compare_done:
    cmp x14, #0
    b.gt shift_right
    b insert_key
shift_right:
    add x15, x21, x26, lsl #4
    add x16, x15, #16
    ldp x17, x18, [x15]
    stp x17, x18, [x16]
    sub x26, x26, #1
    b inner_loop
insert_key:
    add x15, x21, x26, lsl #4
    add x15, x15, #16
    ldp x17, x18, [x23]
    stp x17, x18, [x15]
    add x20, x20, #1
    b outer_loop
sort_done:
    ret

write_output:
    adrp x19, line_count
    add x19, x19, :lo12:line_count
    ldr x19, [x19]
    cmp x19, #0
    b.eq write_done
    mov x20, #0
write_loop:
    cmp x20, x19
    b.ge write_done
    adrp x21, line_recs
    add x21, x21, :lo12:line_recs
    add x22, x21, x20, lsl #4
    ldp x23, x24, [x22]
    mov x0, #1
    mov x1, x23
    mov x2, x24
write_line_bytes:
    cmp x2, #0
    b.eq write_line_done
    mov x8, #64
    svc #0
    cmp x0, #0
    ble write_line_done
    sub x2, x2, x0
    add x1, x1, x0
    b write_line_bytes
write_line_done:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
write_newline:
    cmp x2, #0
    b.eq newline_done
    mov x8, #64
    svc #0
    cmp x0, #0
    ble newline_done
    sub x2, x2, x0
    add x1, x1, x0
    b write_newline
newline_done:
    add x20, x20, #1
    b write_loop
write_done:
    mov x0, #0
    mov x8, #93
    svc #0
