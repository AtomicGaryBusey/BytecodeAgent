    add x0, x0, :lo12:line_count
    ldr x19, [x0]

    mov x20, #1

    b.ge sort_done
    adrp x21, line_recs
    add x21, x21, :lo12:line_recs
    add x21, x21, x20, lsl #4
    adrp x22, key_rec
    add x22, x22, :lo12:key_rec

    stp x23, x24, [x22]

    add x26, x26, :lo12:line_recs
    add x26, x26, x25, lsl #4

    mov x1, x22

    ldp x4, x5, [x1]
    mov x6, #0

    b.hs a_exhausted
    cmp x6, x5
    b.hs b_exhausted
    add x7, x2, x6
    add x8, x4, x6
    ldrb w9, [x7]
    ldrb w10, [x8]

    b.gt cmp_greater

    ret
