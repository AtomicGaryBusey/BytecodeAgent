    b.ge check_done
    add x26, x9, x25
    ldrb w26, [x26]
    cmp w26, #0x5b
    b.eq push_bracket
    cmp w26, #0x5d
    b.eq pop_bracket

    b check_loop

    str w25, [x30]
    add x29, x29, #1
    b check_next

    b.eq bracket_error
    sub x29, x29, #1
    add x30, x28, x29, lsl #2
    ldr w30, [x30]
    add x26, x27, x25, lsl #2
    str w30, [x26]
    add x26, x27, x30, lsl #2
    str w25, [x26]

    b.ne bracket_error

    adrp x19, tape
    add x19, x19, :lo12:tape
    mov x20, #0
    mov x25, #0
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf

    b.ge run_done

    add x26, x9, x25
    ldrb w26, [x26]

    cmp w26, #0x3e
    b.eq ptr_inc
    cmp w26, #0x3c
    b.eq ptr_dec
    cmp w26, #0x2b
    b.eq cell_inc
    cmp w26, #0x2d
    b.eq cell_dec
    cmp w26, #0x2e
    b.eq output_cell
    cmp w26, #0x2c
    b.eq input_cell
    cmp w26, #0x5b
    b.eq open_bracket
    cmp w26, #0x5d
    b.eq close_bracket

    b run_loop

    add x25, x25, #1
    b run_loop

    add x25, x25, #1

    cmp w30, #0
