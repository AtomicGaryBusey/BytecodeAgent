.section .bss
    .align 16
input_buf:
    .space 65536
row_buf:
    .space 128
total_buf:
    .space 16

.text
.globl _start

write_all:
    mov r8, rsi
    mov r9, rdx
1:
    test r9, r9
    jz 2f
    mov rax, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    cmp rax, 0
    jle 3f
    sub r9, rax
    add r8, rax
    jmp 1b
3:
    mov rax, 60
    mov rdi, 1
    syscall
2:
    ret

_start:
    xor r12, r12
    lea rbx, [rip + input_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    cmp rax, 0
    jl exit_error
    je read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_loop

read_done:
    test r12, r12
    jz print_total

    xor r13, r13

row_loop:
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_ok
    mov r14, 16

row_ok:
    lea rsi, [rip + row_buf]
    lea r9, [rip + hex_digits]

    mov rax, r13
    mov rcx, 8
1:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rsi + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz 1b

    mov byte ptr [rsi + 8], ' '
    mov byte ptr [rsi + 9], ' '
    mov r15, 10
    xor rcx, rcx

hex_loop:
    cmp rcx, 16
    jae hex_done
    cmp rcx, r14
    jae blank_slot

actual_slot:
    mov rax, r13
    add rax, rcx
    movzx eax, byte ptr [rbx + rax]
    mov r10, rax
    shr r10, 4
    mov dl, byte ptr [r9 + r10]
    mov byte ptr [rsi + r15], dl
    mov r10, rax
    and r10, 0xf
    mov dl, byte ptr [r9 + r10]
    mov byte ptr [rsi + r15 + 1], dl
    mov byte ptr [rsi + r15 + 2], ' '
    add r15, 3
    cmp rcx, 7
    jne 1f
    mov byte ptr [rsi + r15], ' '
    inc r15
1:
    jmp 2f

blank_slot:
    mov byte ptr [rsi + r15], ' '
    mov byte ptr [rsi + r15 + 1], ' '
    mov byte ptr [rsi + r15 + 2], ' '
    add r15, 3
    cmp rcx, 7
    jne 2f
    mov byte ptr [rsi + r15], ' '
    inc r15

2:
    inc rcx
    cmp rcx, 16
    jb hex_loop

hex_done:
    mov byte ptr [rsi + r15], ' '
    inc r15
    mov byte ptr [rsi + r15], '|'
    inc r15

    mov rcx, 0
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    mov rax, r13
    add rax, rcx
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rsi + r15], al
    jmp ascii_next

ascii_dot:
    mov byte ptr [rsi + r15], '.'

ascii_next:
    inc rcx
    inc r15
    jmp ascii_loop

ascii_done:
    mov byte ptr [rsi + r15], '|'
    inc r15
    mov byte ptr [rsi + r15], 10
    inc r15

    mov rdi, 1
    lea rsi, [rip + row_buf]
    mov rdx, r15
    call write_all

    add r13, r14
    cmp r13, r12
    jb row_loop

print_total:
    lea r9, [rip + hex_digits]
    lea rsi, [rip + total_buf]
    mov rax, r12
    mov rcx, 8
1:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rsi + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz 1b

    mov byte ptr [rsi + 8], 10
    mov rdi, 1
    lea rsi, [rip + total_buf]
    mov rdx, 9
    call write_all

    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"
