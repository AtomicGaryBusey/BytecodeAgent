.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDOUT, 1
.equ INPUT_SIZE, 65536
.equ ROW_SIZE, 16

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
.globl _start

write_all:
    mov r8, rdx
    mov r9, rsi
1:
    test r8, r8
    jz 2f
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    mov rsi, r9
    mov rdx, r8
    syscall
    cmp rax, 0
    jle 3f
    sub r8, rax
    add r9, rax
    jmp 1b
3:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall
2:
    ret

_start:
    xor r12, r12

read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    jl fail
    je read_done
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_loop

read_done:
    test r12, r12
    jz print_total

    xor r13, r13

row_loop:
    mov r14, r12
    sub r14, r13
    cmp r14, ROW_SIZE
    jbe row_len_ok
    mov r14, ROW_SIZE

row_len_ok:
    lea rsi, [rip + linebuf]
    mov rbx, r13
    lea r9, [rip + hex_digits]
    mov rcx, 8
1:
    dec rcx
    mov rax, rbx
    and rax, 0xF
    mov dl, byte ptr [r9 + rax]
    mov byte ptr [rsi + rcx], dl
    shr rbx, 4
    test rcx, rcx
    jnz 1b

    mov byte ptr [rsi + 8], ' '
    mov byte ptr [rsi + 9], ' '

    mov r10, 10
    xor rcx, rcx

hex_loop:
    cmp rcx, r14
    jae blank_slot
actual_slot:
    lea r8, [rip + input_buf]
    add r8, r13
    movzx rax, byte ptr [r8 + rcx]
    mov r8, rax
    shr r8, 4
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rsi + r10], dl
    mov r8, rax
    and r8, 0xF
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rsi + r10 + 1], dl
    mov byte ptr [rsi + r10 + 2], ' '
    cmp rcx, 7
    jne 2f
    mov byte ptr [rsi + r10 + 3], ' '
    add r10, 4
    jmp 3f
2:
    add r10, 3
3:
    inc rcx
    cmp rcx, ROW_SIZE
    jb hex_loop
    jmp after_hex

blank_slot:
    mov byte ptr [rsi + r10], ' '
    mov byte ptr [rsi + r10 + 1], ' '
    mov byte ptr [rsi + r10 + 2], ' '
    cmp rcx, 7
    jne 4f
    mov byte ptr [rsi + r10 + 3], ' '
    add r10, 4
    jmp 5f
4:
    add r10, 3
5:
    inc rcx
    cmp rcx, ROW_SIZE
    jb hex_loop

after_hex:
    mov byte ptr [rsi + r10], ' '
    mov byte ptr [rsi + r10 + 1], '|'
    mov r11, r10
    add r11, 2
    xor rcx, rcx

ascii_loop:
    cmp rcx, r14
    jae ascii_done
    lea r8, [rip + input_buf]
    add r8, r13
    movzx rax, byte ptr [r8 + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7E
    ja ascii_dot
    mov byte ptr [rsi + r11 + rcx], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rsi + r11 + rcx], '.'
ascii_next:
    inc rcx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rsi + r11 + rcx], '|'
    mov byte ptr [rsi + r11 + rcx + 1], 10

    mov rdx, r10
    add rdx, r14
    add rdx, 4
    lea rsi, [rip + linebuf]
    call write_all

    add r13, r14
    cmp r13, r12
    jb row_loop

print_total:
    lea rsi, [rip + totalbuf]
    mov rbx, r12
    lea r9, [rip + hex_digits]
    mov rcx, 8
6:
    dec rcx
    mov rax, rbx
    and rax, 0xF
    mov dl, byte ptr [r9 + rax]
    mov byte ptr [rsi + rcx], dl
    shr rbx, 4
    test rcx, rcx
    jnz 6b
    mov byte ptr [rsi + 8], 10
    mov rdx, 9
    lea rsi, [rip + totalbuf]
    call write_all

    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

fail:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall

.section .bss
    .align 16
input_buf:
    .space INPUT_SIZE
linebuf:
    .space 128
totalbuf:
    .space 16
