.text
.globl _start

_start:
    xor r15d, r15d

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r15
    mov rdx, 32768
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    cmp r15, 32768
    jb read_loop

read_done:
    mov r14, r15
    test r14, r14
    jz exit_zero

    xor r12d, r12d
    xor r13d, r13d

parse_loop:
    cmp r12, r14
    jae parse_done
    mov rbx, r12

scan_loop:
    cmp r12, r14
    jae parse_final
    lea rsi, [rip + input_buffer]
    movzx ecx, byte ptr [rsi + r12]
    cmp cl, 10
    je newline_found
    inc r12
    jmp scan_loop

newline_found:
    mov rax, r12
    sub rax, rbx
    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r13*8], rbx
    lea rsi, [rip + line_lens]
    mov qword ptr [rsi + r13*8], rax
    inc r13
    inc r12
    jmp parse_loop

parse_final:
    mov rax, r14
    sub rax, rbx
    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r13*8], rbx
    lea rsi, [rip + line_lens]
    mov qword ptr [rsi + r13*8], rax
    inc r13
    jmp sort_begin

parse_done:
sort_begin:
    mov r15d, 1

sort_loop:
    cmp r15d, r13d
    jae output_start
    lea rsi, [rip + line_ptrs]
    mov r10, qword ptr [rsi + r15*8]
    lea rsi, [rip + line_lens]
    mov r11, qword ptr [rsi + r15*8]
    mov r14d, r15d
    dec r14d

inner_loop:
    cmp r14d, 0
    jl insert_key
    lea rsi, [rip + line_ptrs]
    mov rbx, qword ptr [rsi + r14*8]
    lea rsi, [rip + line_lens]
    mov r12, qword ptr [rsi + r14*8]
    mov rdi, rbx
    mov rsi, r12
    mov rdx, r10
    mov rcx, r11
    call compare_lines
    cmp rax, 0
    jg move_right
    jmp insert_key

move_right:
    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r14*8 + 8], rbx
    lea rsi, [rip + line_lens]
    mov qword ptr [rsi + r14*8 + 8], r12
    dec r14
    jmp inner_loop

insert_key:
    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r14*8 + 8], r10
    lea rsi, [rip + line_lens]
    mov qword ptr [rsi + r14*8 + 8], r11
    inc r15d
    jmp sort_loop

output_start:
    xor r15d, r15d

output_loop:
    cmp r15d, r13d
    jae exit_zero
    lea rsi, [rip + line_ptrs]
    mov rbx, qword ptr [rsi + r15*8]
    lea rsi, [rip + line_lens]
    mov r12, qword ptr [rsi + r15*8]
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov rdx, r12
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_buf]
    mov rdx, 1
    syscall
    inc r15d
    jmp output_loop

compare_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9d, r9d

compare_loop:
    cmp r9, r8
    jae compare_len
    lea rax, [rip + input_buffer]
    movzx r10d, byte ptr [rax + rdi + r9]
    movzx r11d, byte ptr [rax + rdx + r9]
    cmp r10d, r11d
    je same_byte
    cmp r10d, r11d
    jb less
    mov rax, 1
    jmp compare_end

same_byte:
    inc r9
    jmp compare_loop

less:
    mov rax, -1
    jmp compare_end

compare_len:
    cmp rsi, rcx
    je equal
    jb len_less
    mov rax, 1
    jmp compare_end

len_less:
    mov rax, -1
    jmp compare_end

equal:
    xor rax, rax

compare_end:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
newline_buf:
    .byte 10

.section .bss
.align 8
input_buffer:
    .space 32768
.align 8
line_ptrs:
    .space 8000
.align 8
line_lens:
    .space 8000
