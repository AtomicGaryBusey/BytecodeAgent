.text
.globl _start

_start:
    xor r12d, r12d

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 32768
    jb read_loop

read_done:
    mov r13, r12
    test r13, r13
    jz exit_zero

    xor r15d, r15d
    xor r14d, r14d

parse_loop:
    cmp r14, r13
    jae parse_end
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r14]
    cmp al, 10
    jne parse_continue

    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r15*8], r13
    lea rsi, [rip + line_lens]
    mov rax, r14
    sub rax, r13
    mov qword ptr [rsi + r15*8], rax
    inc r15
    inc r14
    jmp parse_loop

parse_continue:
    inc r14
    jmp parse_loop

parse_end:
    cmp r13, r13
    je parse_done

    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r15*8], r13
    lea rsi, [rip + line_lens]
    mov rax, r12
    sub rax, r13
    mov qword ptr [rsi + r15*8], rax
    inc r15

parse_done:
    mov r12, r15
    cmp r12, 1
    jbe print_lines

    mov r13, 1

sort_outer:
    cmp r13, r12
    jae print_lines

    lea rsi, [rip + line_ptrs]
    mov r15, qword ptr [rsi + r13*8]
    lea rsi, [rip + line_lens]
    mov rbx, qword ptr [rsi + r13*8]

    mov r14, r13

sort_inner:
    cmp r14, 0
    je insert_key

    lea rsi, [rip + line_ptrs]
    mov rdi, qword ptr [rsi + (r14-1)*8]
    lea rsi, [rip + line_lens]
    mov rsi, qword ptr [rsi + (r14-1)*8]
    mov rdx, r15
    mov rcx, rbx
    call compare_lines

    cmp rax, 0
    jle insert_key

    lea rsi, [rip + line_ptrs]
    mov rax, qword ptr [rsi + (r14-1)*8]
    mov qword ptr [rsi + r14*8], rax

    lea rsi, [rip + line_lens]
    mov rax, qword ptr [rsi + (r14-1)*8]
    mov qword ptr [rsi + r14*8], rax

    dec r14
    jmp sort_inner

insert_key:
    lea rsi, [rip + line_ptrs]
    mov qword ptr [rsi + r14*8], r15
    lea rsi, [rip + line_lens]
    mov qword ptr [rsi + r14*8], rbx
    inc r13
    jmp sort_outer

print_lines:
    xor r13d, r13d

output_loop:
    cmp r13d, r12d
    jae exit_zero

    lea rsi, [rip + line_ptrs]
    mov rbx, qword ptr [rsi + r13*8]
    lea rsi, [rip + line_lens]
    mov rdx, qword ptr [rsi + r13*8]

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    inc r13
    jmp output_loop

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, rsi
    mov r9, rcx
    cmp r8, r9
    cmova r8, r9
    xor r10d, r10d

compare_loop:
    cmp r10, r8
    jae compare_len

    movzx eax, byte ptr [rdi + r10]
    movzx r11d, byte ptr [rdx + r10]
    cmp al, r11b
    jb less
    ja greater

    inc r10
    jmp compare_loop

less:
    mov rax, -1
    ret

greater:
    mov rax, 1
    ret

compare_len:
    cmp rsi, rcx
    jb shorter
    ja longer
    xor eax, eax
    ret

shorter:
    mov rax, -1
    ret

longer:
    mov rax, 1
    ret

.section .rodata
newline_byte:
    .byte 10

.section .bss
.align 8
input_buffer:
    .space 32768

.align 8
line_ptrs:
    .space 8000

.align 8
line_lens:
    .space 8000
