.text
.globl _start

_start:
    xor r13d, r13d

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov rdx, 32768
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    cmp r13, 32768
    jb read_loop

read_done:
    test r13d, r13d
    jz exit_zero

    xor r12d, r12d
    xor r14d, r14d
    xor ebx, ebx

parse_loop:
    cmp ebx, r13d
    jae parse_done
    movzx eax, byte ptr [rip + input_buffer + rbx]
    cmp al, 10
    jne parse_continue
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov qword ptr [line_ptrs + r12*8], rsi
    mov eax, ebx
    sub eax, r14d
    mov dword ptr [line_lens + r12*4], eax
    inc r12d
    mov r14d, ebx
    inc r14d

parse_continue:
    inc ebx
    jmp parse_loop

parse_done:
    cmp r14d, r13d
    jae sort_lines
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov qword ptr [line_ptrs + r12*8], rsi
    mov eax, r13d
    sub eax, r14d
    mov dword ptr [line_lens + r12*4], eax
    inc r12d

sort_lines:
    cmp r12d, 1
    jbe print_lines
    xor r13d, r13d

outer_loop:
    mov eax, r12d
    sub eax, r13d
    cmp eax, 1
    jbe print_lines
    xor r14d, r14d

inner_loop:
    mov eax, r12d
    sub eax, r13d
    dec eax
    cmp r14d, eax
    jae end_outer_loop

    mov rdi, qword ptr [line_ptrs + r14*8]
    mov esi, dword ptr [line_lens + r14*4]
    mov rdx, qword ptr [line_ptrs + r14*8 + 8]
    mov ecx, dword ptr [line_lens + r14*4 + 4]
    call compare_lines
    cmp eax, 1
    jle no_swap

    mov r8, qword ptr [line_ptrs + r14*8]
    mov r9, qword ptr [line_ptrs + r14*8 + 8]
    mov qword ptr [line_ptrs + r14*8], r9
    mov qword ptr [line_ptrs + r14*8 + 8], r8

    mov r8d, dword ptr [line_lens + r14*4]
    mov r9d, dword ptr [line_lens + r14*4 + 4]
    mov dword ptr [line_lens + r14*4], r9d
    mov dword ptr [line_lens + r14*4 + 4], r8d

no_swap:
    inc r14d
    jmp inner_loop

end_outer_loop:
    inc r13d
    jmp outer_loop

print_lines:
    xor r13d, r13d

print_loop:
    cmp r13d, r12d
    jae exit_zero

    mov rsi, qword ptr [line_ptrs + r13*8]
    mov edx, dword ptr [line_lens + r13*4]
    mov rdi, 1
    call write_all
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    call write_all
    inc r13d
    jmp print_loop

write_all:
    mov r8, rdx
    mov r9, rsi
    mov r10, rdi

write_all_loop:
    test r8, r8
    jz write_all_done
    mov rax, 1
    mov rdi, r10
    mov rsi, r9
    mov rdx, r8
    syscall
    cmp rax, 0
    jle write_all_done
    sub r8, rax
    add r9, rax
    jmp write_all_loop

write_all_done:
    ret

compare_lines:
    mov r8d, esi
    mov r9d, ecx
    cmp r8d, r9d
    jbe 1f
    mov r8d, r9d
1:
    xor eax, eax

compare_loop:
    cmp eax, r8d
    jae compare_len
    mov r10b, byte ptr [rdi + rax]
    mov r11b, byte ptr [rdx + rax]
    cmp r10b, r11b
    jb compare_less
    ja compare_greater
    inc eax
    jmp compare_loop

compare_len:
    cmp esi, ecx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
    .align 32
input_buffer:
    .space 32768

line_ptrs:
    .zero 1000 * 8

line_lens:
    .zero 1000 * 4
