.text
.globl _start

_start:
    xor r13d, r13d

read_input:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov rdx, 32768
    sub rdx, r13
    syscall
    test rax, rax
    jz parse_input
    add r13, rax
    cmp r13, 32768
    jb read_input

parse_input:
    xor r12d, r12d
    test r13d, r13d
    jz sort_lines

    xor ebx, ebx
    mov r14d, 0

parse_loop:
    cmp rbx, r13
    jae parse_done

    lea rcx, [rip + input_buffer]
    movzx eax, byte ptr [rcx + rbx]
    cmp al, 10
    jne parse_next

    mov rax, rbx
    sub rax, r14
    lea r9, [rip + line_ptrs]
    lea r10, [rip + line_lens]
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov qword ptr [r9 + r12*8], rsi
    mov dword ptr [r10 + r12*4], eax
    inc r12d
    mov r14d, ebx
    inc r14d

parse_next:
    inc rbx
    jmp parse_loop

parse_done:
    cmp r14d, r13d
    jae sort_lines

    mov rax, r13
    sub rax, r14
    lea r9, [rip + line_ptrs]
    lea r10, [rip + line_lens]
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov qword ptr [r9 + r12*8], rsi
    mov dword ptr [r10 + r12*4], eax
    inc r12d

sort_lines:
    cmp r12d, 1
    jbe print_lines

    xor r13d, r13d

outer_loop:
    mov eax, r12d
    sub eax, r13d
    cmp eax, 1
    jbe print_lines

    xor r14d, r14d

inner_loop:
    mov eax, r12d
    sub eax, r13d
    dec eax
    cmp r14d, eax
    jae end_outer_loop

    lea r15, [rip + line_ptrs]
    lea r10, [rip + line_lens]
    mov r11, r14
    shl r11, 3
    mov rdi, qword ptr [r15 + r11]
    mov rsi, qword ptr [r15 + r11 + 8]
    mov edx, dword ptr [r10 + r14*4]
    mov ecx, dword ptr [r10 + r14*4 + 4]
    call compare_lines
    cmp eax, 1
    jle no_swap

    lea r15, [rip + line_ptrs]
    lea r10, [rip + line_lens]
    mov r11, r14
    shl r11, 3
    mov r8, qword ptr [r15 + r11]
    mov r9, qword ptr [r15 + r11 + 8]
    mov qword ptr [r15 + r11], r9
    mov qword ptr [r15 + r11 + 8], r8
    mov r8d, dword ptr [r10 + r14*4]
    mov r9d, dword ptr [r10 + r14*4 + 4]
    mov dword ptr [r10 + r14*4], r9d
    mov dword ptr [r10 + r14*4 + 4], r8d

no_swap:
    inc r14d
    jmp inner_loop

end_outer_loop:
    inc r13d
    jmp outer_loop

print_lines:
    xor r13d, r13d

print_loop:
    cmp r13d, r12d
    jae exit_zero

    lea r15, [rip + line_ptrs]
    mov rsi, qword ptr [r15 + r13*8]
    lea r10, [rip + line_lens]
    mov edx, dword ptr [r10 + r13*4]

    mov rdi, 1
    call write_all

    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    call write_all

    inc r13d
    jmp print_loop

write_all:
    mov r8, rdx
    mov r9, rsi
    mov r10, rdi

write_all_loop:
    test r8, r8
    jz write_all_done

    mov rax, 1
    mov rdi, r10
    mov rsi, r9
    mov rdx, r8
    syscall
    cmp rax, 0
    jle write_all_done

    sub r8, rax
    add r9, rax
    jmp write_all_loop

write_all_done:
    ret

compare_lines:
    mov r9, rdx
    mov r10, rcx
    mov r8, r9
    cmp r8, r10
    jbe compare_min_done
    mov r8, r10

compare_min_done:
    xor eax, eax

compare_loop:
    cmp rax, r8
    jae compare_len_check

    mov dl, byte ptr [rdi + rax]
    mov sil, byte ptr [rsi + rax]
    cmp dl, sil
    jb compare_less
    ja compare_greater
    inc rax
    jmp compare_loop

compare_len_check:
    cmp r9, r10
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
    .align 32
input_buffer:
    .space 32768

line_ptrs:
    .zero 1000 * 8

line_lens:
    .zero 1000 * 4
