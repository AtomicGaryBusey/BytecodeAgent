.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94
.equ INPUT_SIZE, 65536
.equ STACK_SIZE, 16384

.text
.globl _start

_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0
read_loop:
  mov x8, #SYS_READ
  mov x0, #0
  add x1, x19, x20
  mov x2, #INPUT_SIZE
  sub x2, x2, x20
  cmp x2, #0
  b.eq read_done
  svc #0
  cmp x0, #0
  b.eq read_done
  cmp x0, #0
  b.lt read_error
  add x20, x20, x0
  cmp x20, #INPUT_SIZE
  b.eq read_done
  b read_loop
read_error:
  mov x8, #SYS_EXIT
  mov x0, #1
  svc #0

read_done:
  mov x23, x19
  add x24, x19, x20
parse_lines:
  cmp x23, x24
  b.hs all_done
  mov x25, x23
find_nl:
  cmp x25, x24
  b.hs no_nl
  ldrb w9, [x25]
  cmp w9, #0x0A
  b.eq nl_found
  add x25, x25, #1
  b find_nl
nl_found:
  mov x0, x23
  sub x1, x25, x23
  bl process_line
  add x23, x25, #1
  b parse_lines
no_nl:
  mov x0, x23
  sub x1, x24, x23
  bl process_line
  b all_done
all_done:
  mov x8, #SYS_EXIT
  mov x0, #0
  svc #0

process_line:
  adrp x21, stack_base
  add x21, x21, :lo12:stack_base
  mov x22, #0
  mov x2, x0
  add x3, x0, x1
line_loop:
  cmp x2, x3
  b.hs end_of_line
skip_sep:
  cmp x2, x3
  b.hs end_of_line
  ldrb w7, [x2]
  cmp w7, #0x20
  b.eq sep_inc
  cmp w7, #0x09
  b.eq sep_inc
  b token_start
sep_inc:
  add x2, x2, #1
  b skip_sep
token_start:
  mov x4, x2
token_scan:
  cmp x2, x3
  b.hs token_done
  ldrb w7, [x2]
  cmp w7, #0x20
  b.eq token_done
  cmp w7, #0x09
  b.eq token_done
  add x2, x2, #1
  b token_scan
token_done:
  sub x5, x2, x4
  ldrb w7, [x4]
  cmp x5, #1
  b.ne not_single_char
  cmp w7, #43
  b.eq op_plus
  cmp w7, #45
  b.eq op_minus
  cmp w7, #42
  b.eq op_mul
  cmp w7, #47
  b.eq op_div
  cmp w7, #37
  b.eq op_mod
  b invalid_token
not_single_char:
  cmp w7, #45
  b.eq parse_negative
  cmp w7, #43
  b.eq invalid_token
  b parse_positive
parse_negative:
  sub x5, x5, #1
  cmp x5, #0
  b.eq invalid_token
  cmp x5, #18
  b.hi invalid_token
  mov x13, #1
  add x11, x4, #1
  b parse_digits
parse_positive:
  cmp x5, #18
  b.hi invalid_token
  mov x13, #0
  mov x11, x4
parse_digits:
  mov x9, #0
  mov x10, #10
digit_accum:
  cmp x11, x2
  b.hs digit_done
  ldrb w7, [x11], #1
  cmp w7, #48
  b.lt invalid_token
  cmp w7, #57
  b.gt invalid_token
  sub x7, x7, #48
  madd x9, x9, x10, x7
  b digit_accum
digit_done:
  mov x0, x9
  cmp x13, #0
  b.eq push_value
  neg x0, x0
  b push_value
push_value:
  mov x7, x22
  lsl x7, x7, #3
  add x7, x21, x7
  str x0, [x7]
  add x22, x22, #1
  b line_loop
op_plus:
  cmp x22, #2
  b.lt line_error
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x11, [x7]
  sub x22, x22, #1
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x10, [x7]
  sub x22, x22, #1
  add x0, x10, x11
  b push_value
op_minus:
  cmp x22, #2
  b.lt line_error
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x11, [x7]
  sub x22, x22, #1
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x10, [x7]
  sub x22, x22, #1
  sub x0, x10, x11
  b push_value
op_mul:
  cmp x22, #2
  b.lt line_error
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x11, [x7]
  sub x22, x22, #1
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x10, [x7]
  sub x22, x22, #1
  mul x0, x10, x11
  b push_value
op_div:
  cmp x22, #2
  b.lt line_error
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x11, [x7]
  sub x22, x22, #1
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x10, [x7]
  sub x22, x22, #1
  cmp x11, #0
  b.eq line_error
  mov x12, #0x8000
  lsl x12, x12, #48
  cmp x10, x12
  b.ne div_ok
  cmp x11, #-1
  b.eq line_error
div_ok:
  sdiv x0, x10, x11
  b push_value
op_mod:
  cmp x22, #2
  b.lt line_error
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x11, [x7]
  sub x22, x22, #1
  sub x7, x22, #1
  lsl x7, x7, #3
  add x7, x21, x7
  ldr x10, [x7]
  sub x22, x22, #1
  cmp x11, #0
  b.eq line_error
  mov x12, #0x8000
  lsl x12, x12, #48
  cmp x10, x12
  b.ne mod_ok
  cmp x11, #-1
  b.eq line_error
mod_ok:
  sdiv x0, x10, x11
  msub x0, x0, x11, x10
  b push_value
end_of_line:
  cmp x22, #1
  b.ne line_error
  ldr x0, [x21]
  bl print_number
  ret
line_error:
  bl print_error
  ret
invalid_token:
  bl print_error
  ret

print_error:
  adrp x1, err_msg
  add x1, x1, :lo12:err_msg
  mov x2, #6
  mov x8, #SYS_WRITE
  mov x0, #1
  svc #0
  ret

print_number:
  mov x3, #0
  cmp x0, #0
  b.ge print_nonneg
  mov x3, #1
  mov x9, #0x8000
  lsl x9, x9, #48
  cmp x0, x9
  b.eq print_min_int64
  neg x0, x0
print_nonneg:
  sub sp, sp, #32
  mov x1, sp
  add x1, x1, #31
  mov x2, x0
  cmp x2, #0
  b.ne digits_loop
  mov x7, #48
  strb w7, [x1]
  mov x4, #1
  b after_digits
digits_loop:
  mov x5, #10
  udiv x6, x2, x5
  msub x7, x6, x5, x2
  add x7, x7, #48
  strb w7, [x1], #-1
  add x4, x4, #1
  mov x2, x6
  cmp x2, #0
  b.ne digits_loop
after_digits:
  cmp x3, #0
  b.eq no_sign
  sub x1, x1, #1
  mov w7, #45
  strb w7, [x1]
  add x4, x4, #1
no_sign:
  mov x2, x4
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  mov x9, #10
  strb w9, [sp]
  mov x1, sp
  mov x2, #1
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  add sp, sp, #32
  ret
print_min_int64:
  sub sp, sp, #32
  adrp x1, min_int64_str
  add x1, x1, :lo12:min_int64_str
  mov x2, #20
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  mov x9, #10
  strb w9, [sp]
  mov x1, sp
  mov x2, #1
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  add sp, sp, #32
  ret

.section .rodata
err_msg:
  .ascii "error\n"
min_int64_str:
  .ascii "-9223372036854775808"

.section .bss
.align 8
input_buf:
  .space INPUT_SIZE
.align 8
stack_base:
  .space STACK_SIZE
