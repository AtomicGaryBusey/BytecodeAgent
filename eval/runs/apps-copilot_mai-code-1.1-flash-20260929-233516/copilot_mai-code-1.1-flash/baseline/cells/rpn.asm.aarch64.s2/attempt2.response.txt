.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536
.equ STACK_SIZE, 16384

.text
.globl _start

_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0

read_loop:
  mov x9, #INPUT_SIZE
  sub x9, x9, x20
  cmp x9, #0
  b.le read_done
  mov x8, #SYS_READ
  mov x0, #0
  add x1, x19, x20
  mov x2, x9
  svc #0
  cmp x0, #0
  b.lt read_error
  b.eq read_done
  add x20, x20, x0
  b read_loop

read_error:
  mov x8, #SYS_EXIT
  mov x0, #1
  svc #0

read_done:
  mov x23, x19
  add x24, x19, x20
  b parse_lines

parse_lines:
  cmp x23, x24
  b.hs exit_ok
  mov x25, x23

find_nl:
  cmp x25, x24
  b.hs final_line
  ldrb w9, [x25]
  cmp w9, #0x0A
  b.eq newline_found
  add x25, x25, #1
  b find_nl

newline_found:
  mov x0, x23
  sub x1, x25, x23
  bl process_line
  add x23, x25, #1
  b parse_lines

final_line:
  cmp x23, x24
  b.hs exit_ok
  mov x0, x23
  sub x1, x24, x23
  bl process_line
  b exit_ok

exit_ok:
  mov x8, #SYS_EXIT
  mov x0, #0
  svc #0

process_line:
  adrp x21, stack_base
  add x21, x21, :lo12:stack_base
  mov x22, #0
  mov x2, x0
  add x3, x0, x1

line_loop:
  cmp x2, x3
  b.hs line_final

skip_sep:
  cmp x2, x3
  b.hs line_final
  ldrb w7, [x2]
  cmp w7, #0x20
  b.eq sep_inc
  cmp w7, #0x09
  b.eq sep_inc
  b token_start

sep_inc:
  add x2, x2, #1
  b skip_sep

token_start:
  mov x4, x2

scan_token:
  cmp x2, x3
  b.hs token_done
  ldrb w7, [x2]
  cmp w7, #0x20
  b.eq token_done
  cmp w7, #0x09
  b.eq token_done
  add x2, x2, #1
  b scan_token

token_done:
  sub x5, x2, x4
  ldrb w7, [x4]
  cmp x5, #1
  b.ne not_single
  cmp w7, #43
  b.eq op_plus
  cmp w7, #45
  b.eq op_minus
  cmp w7, #42
  b.eq op_mul
  cmp w7, #47
  b.eq op_div
  cmp w7, #37
  b.eq op_mod
  b invalid_token

not_single:
  cmp w7, #45
  b.eq parse_negative
  cmp w7, #43
  b.eq invalid_token
  b parse_positive

parse_negative:
  cmp x5, #19
  b.hi invalid_token
  sub x6, x5, #1
  cmp x6, #0
  b.eq invalid_token
  mov x13, #1
  add x9, x4, #1
  mov x10, #0
  b scan_digits

parse_positive:
  cmp x5, #19
  b.hi invalid_token
  mov x13, #0
  mov x9, x4
  mov x10, #0
  b scan_digits

scan_digits:
  cmp x9, x2
  b.hs digits_done
  ldrb w7, [x9], #1
  cmp w7, #48
  b.lt invalid_token
  cmp w7, #57
  b.gt invalid_token
  sub w7, w7, #48
  mov x11, #10
  madd x10, x10, x11, x7
  b scan_digits

digits_done:
  cmp x13, #0
  b.eq push_literal
  neg x10, x10

push_literal:
  mov x12, x22
  lsl x12, x12, #3
  add x12, x12, x21
  str x10, [x12]
  add x22, x22, #1
  b line_loop

op_plus:
  cmp x22, #2
  b.lt line_error
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x11, [x12]
  sub x22, x22, #1
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x10, [x12]
  sub x22, x22, #1
  add x10, x10, x11
  b push_result

op_minus:
  cmp x22, #2
  b.lt line_error
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x11, [x12]
  sub x22, x22, #1
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x10, [x12]
  sub x22, x22, #1
  sub x10, x10, x11
  b push_result

op_mul:
  cmp x22, #2
  b.lt line_error
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x11, [x12]
  sub x22, x22, #1
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x10, [x12]
  sub x22, x22, #1
  mul x10, x10, x11
  b push_result

op_div:
  cmp x22, #2
  b.lt line_error
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x11, [x12]
  sub x22, x22, #1
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x10, [x12]
  sub x22, x22, #1
  cmp x11, #0
  b.eq line_error
  mov x12, #0x8000
  lsl x12, x12, #48
  cmp x10, x12
  b.ne div_ok
  cmp x11, #-1
  b.eq line_error
div_ok:
  sdiv x10, x10, x11
  b push_result

op_mod:
  cmp x22, #2
  b.lt line_error
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x11, [x12]
  sub x22, x22, #1
  sub x12, x22, #1
  lsl x12, x12, #3
  add x12, x12, x21
  ldr x10, [x12]
  sub x22, x22, #1
  cmp x11, #0
  b.eq line_error
  mov x12, #0x8000
  lsl x12, x12, #48
  cmp x10, x12
  b.ne mod_ok
  cmp x11, #-1
  b.eq line_error
mod_ok:
  sdiv x12, x10, x11
  msub x10, x12, x11, x10
  b push_result

push_result:
  mov x12, x22
  lsl x12, x12, #3
  add x12, x12, x21
  str x10, [x12]
  add x22, x22, #1
  b line_loop

line_final:
  cmp x22, #1
  b.ne line_error
  ldr x0, [x21]
  bl print_number
  ret

line_error:
  bl print_error
  ret

invalid_token:
  bl print_error
  ret

print_error:
  adrp x1, err_msg
  add x1, x1, :lo12:err_msg
  mov x2, #6
  mov x8, #SYS_WRITE
  mov x0, #1
  svc #0
  ret

print_number:
  mov x9, #0x8000
  lsl x9, x9, #48
  cmp x0, x9
  b.eq print_min_int64
  mov x3, #0
  cmp x0, #0
  b.ge pos_num
  mov x3, #1
  neg x0, x0

pos_num:
  sub sp, sp, #32
  mov x1, sp
  add x1, x1, #31
  mov x2, x0
  mov x4, #0
  cmp x2, #0
  b.ne digits_loop
  mov w5, #48
  strb w5, [x1]
  mov x4, #1
  b after_digits

digits_loop:
  mov x5, #10
  udiv x6, x2, x5
  msub x7, x6, x5, x2
  add w7, w7, #48
  strb w7, [x1], #-1
  add x4, x4, #1
  mov x2, x6
  cmp x2, #0
  b.ne digits_loop

after_digits:
  cmp x3, #0
  b.eq no_sign
  sub x1, x1, #1
  mov w7, #45
  strb w7, [x1]
  add x4, x4, #1

no_sign:
  mov x0, #1
  mov x2, x4
  mov x8, #SYS_WRITE
  svc #0
  bl write_newline
  add sp, sp, #32
  ret

print_min_int64:
  adrp x1, min_int64_str
  add x1, x1, :lo12:min_int64_str
  mov x2, #20
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  bl write_newline
  ret

write_newline:
  adrp x1, newline
  add x1, x1, :lo12:newline
  mov x2, #1
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  ret

.section .rodata
err_msg:
  .ascii "error\n"

newline:
  .ascii "\n"

min_int64_str:
  .ascii "-9223372036854775808"

.section .bss
.align 8
input_buf:
  .space INPUT_SIZE

.align 8
stack_base:
  .space STACK_SIZE
