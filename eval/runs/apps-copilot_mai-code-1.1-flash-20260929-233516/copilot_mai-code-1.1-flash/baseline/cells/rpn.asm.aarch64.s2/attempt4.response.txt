.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536

_start:
  mov x29, xzr
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0

read_loop:
  cmp x20, #INPUT_SIZE
  b.hs read_done
  mov x8, #SYS_READ
  mov x0, #0
  add x1, x19, x20
  mov x2, #INPUT_SIZE
  sub x2, x2, x20
  svc #0
  cmp x0, #0
  b.lt read_error
  b.eq read_done
  add x20, x20, x0
  b read_loop

read_error:
  mov x8, #SYS_EXIT
  mov x0, #1
  svc #0

read_done:
  mov x21, x19
  add x22, x19, x20

process_lines:
  cmp x21, x22
  b.hs exit_ok
  mov x23, x21

find_nl:
  cmp x23, x22
  b.hs final_line
  ldrb w9, [x23]
  cmp w9, #0x0A
  b.eq newline_found
  add x23, x23, #1
  b find_nl

newline_found:
  sub sp, sp, #96
  stp x19, x20, [sp, #0]
  stp x21, x22, [sp, #16]
  stp x23, x24, [sp, #32]
  stp x25, x26, [sp, #48]
  stp x27, x28, [sp, #64]
  str x29, [sp, #80]
  mov x0, x21
  sub x1, x23, x21
  bl process_line
  ldr x29, [sp, #80]
  ldp x27, x28, [sp, #64]
  ldp x25, x26, [sp, #48]
  ldp x23, x24, [sp, #32]
  ldp x21, x22, [sp, #16]
  ldp x19, x20, [sp, #0]
  add sp, sp, #96
  add x21, x23, #1
  b process_lines

final_line:
  cmp x21, x22
  b.hs exit_ok
  sub sp, sp, #96
  stp x19, x20, [sp, #0]
  stp x21, x22, [sp, #16]
  stp x23, x24, [sp, #32]
  stp x25, x26, [sp, #48]
  stp x27, x28, [sp, #64]
  str x29, [sp, #80]
  mov x0, x21
  sub x1, x22, x21
  bl process_line
  ldr x29, [sp, #80]
  ldp x27, x28, [sp, #64]
  ldp x25, x26, [sp, #48]
  ldp x23, x24, [sp, #32]
  ldp x21, x22, [sp, #16]
  ldp x19, x20, [sp, #0]
  add sp, sp, #96
  b exit_ok

exit_ok:
  mov x8, #SYS_EXIT
  mov x0, #0
  svc #0

process_line:
  stp x29, x30, [sp, #-16]!
  mov x29, sp
  mov x19, x0
  mov x20, x1
  adrp x21, rpn_stack
  add x21, x21, :lo12:rpn_stack
  mov x22, #0
  add x23, x19, x20
  mov x24, x19
  b line_loop

line_loop:
  cmp x24, x23
  b.hs line_done

skip_sep:
  cmp x24, x23
  b.hs line_done
  ldrb w9, [x24]
  cmp w9, #0x20
  b.eq skip_inc
  cmp w9, #0x09
  b.eq skip_inc
  b token_start

skip_inc:
  add x24, x24, #1
  b skip_sep

token_start:
  mov x25, x24

scan_token:
  cmp x24, x23
  b.hs token_end
  ldrb w9, [x24]
  cmp w9, #0x20
  b.eq token_end
  cmp w9, #0x09
  b.eq token_end
  add x24, x24, #1
  b scan_token

token_end:
  sub x26, x24, x25
  cmp x26, #0
  b.eq invalid_line
  cmp x26, #19
  b.hi invalid_line
  ldrb w9, [x25]
  cmp x26, #1
  b.ne parse_literal
  cmp w9, #0x2B
  b.eq do_operator
  cmp w9, #0x2D
  b.eq do_operator
  cmp w9, #0x2A
  b.eq do_operator
  cmp w9, #0x2F
  b.eq do_operator
  cmp w9, #0x25
  b.eq do_operator
  b parse_literal

parse_literal:
  mov x27, x26
  mov x28, x25
  mov x9, #0
  mov x10, #0
  ldrb w11, [x28]
  cmp w11, #0x2D
  b.ne no_minus
  mov x9, #1
  add x28, x28, #1
  sub x27, x27, #1
  cmp x27, #0
  b.eq invalid_line
  b literal_digits
no_minus:
  cmp w11, #0x2B
  b.eq invalid_line
literal_digits:
  cmp x27, #18
  b.hi invalid_line
  cmp x27, #0
  b.eq literal_done
  ldrb w11, [x28], #1
  cmp w11, #0x30
  b.lt invalid_line
  cmp w11, #0x39
  b.gt invalid_line
  sub w11, w11, #0x30
  mov x12, #10
  madd x10, x10, x12, w11
  sub x27, x27, #1
  b literal_digits

literal_done:
  cmp x9, #0
  b.eq push_literal
  neg x10, x10

push_literal:
  add x11, x21, x22, lsl #3
  str x10, [x11]
  add x22, x22, #1
  b line_loop

do_operator:
  cmp x22, #2
  b.lt invalid_line
  sub x22, x22, #1
  add x11, x21, x22, lsl #3
  ldr x10, [x11]
  sub x22, x22, #1
  add x11, x21, x22, lsl #3
  ldr x12, [x11]

  cmp w9, #0x2B
  b.eq op_plus
  cmp w9, #0x2D
  b.eq op_minus
  cmp w9, #0x2A
  b.eq op_mul
  cmp w9, #0x2F
  b.eq op_div
  cmp w9, #0x25
  b.eq op_mod
  b invalid_line

op_plus:
  add x13, x12, x10
  b save_result

op_minus:
  sub x13, x12, x10
  b save_result

op_mul:
  mul x13, x12, x10
  b save_result

op_div:
  cmp x10, #0
  b.eq invalid_line
  mov x11, #0x8000
  lsl x11, x11, #48
  cmp x12, x11
  b.ne div_ok
  cmp x10, #-1
  b.eq invalid_line
div_ok:
  sdiv x13, x12, x10
  b save_result

op_mod:
  cmp x10, #0
  b.eq invalid_line
  mov x11, #0x8000
  lsl x11, x11, #48
  cmp x12, x11
  b.ne mod_ok
  cmp x10, #-1
  b.eq invalid_line
mod_ok:
  sdiv x11, x12, x10
  msub x13, x11, x10, x12
  b save_result

save_result:
  add x11, x21, x22, lsl #3
  str x13, [x11]
  add x22, x22, #1
  b line_loop

line_done:
  cmp x22, #1
  b.ne invalid_line
  ldr x0, [x21]
  bl print_number
  ldp x29, x30, [sp], #16
  ret

invalid_line:
  bl print_error
  ldp x29, x30, [sp], #16
  ret

print_error:
  stp x29, x30, [sp, #-16]!
  adrp x1, err_msg
  add x1, x1, :lo12:err_msg
  mov x2, #6
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  ldp x29, x30, [sp], #16
  ret

print_number:
  stp x29, x30, [sp, #-16]!
  mov x9, x0
  mov x10, #0x8000
  lsl x10, x10, #48
  cmp x9, x10
  b.eq print_min_int64
  cmp x9, #0
  b.ge print_positive
  mov x10, #1
  neg x9, x9
  b number_loop_setup
print_positive:
  mov x10, #0

number_loop_setup:
  sub sp, sp, #64
  mov x11, sp
  add x11, x11, #63
  mov x12, x9
  mov x13, #0
  cmp x12, #0
  b.ne number_loop
  mov w14, #48
  strb w14, [x11]
  sub x11, x11, #1
  mov x13, #1
  b number_done

number_loop:
  mov x16, #10
  udiv x15, x12, x16
  msub x14, x15, x16, x12
  add w14, w14, #48
  sub x11, x11, #1
  strb w14, [x11]
  mov x12, x15
  add x13, x13, #1
  cmp x12, #0
  b.ne number_loop

number_done:
  cmp x10, #0
  b.eq number_no_sign
  sub x11, x11, #1
  mov w14, #45
  strb w14, [x11]
  add x13, x13, #1

number_no_sign:
  mov x0, #1
  cmp x10, #0
  b.eq number_ptr
  mov x1, x11
  b write_number
number_ptr:
  add x1, x11, #1

write_number:
  mov x2, x13
  mov x8, #SYS_WRITE
  svc #0
  bl write_newline
  add sp, sp, #64
  ldp x29, x30, [sp], #16
  ret

print_min_int64:
  adrp x1, min_int64_str
  add x1, x1, :lo12:min_int64_str
  mov x2, #20
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  bl write_newline
  ldp x29, x30, [sp], #16
  ret

write_newline:
  stp x29, x30, [sp, #-16]!
  adrp x1, newline
  add x1, x1, :lo12:newline
  mov x2, #1
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  ldp x29, x30, [sp], #16
  ret

.section .rodata
err_msg:
  .ascii "error\n"

newline:
  .ascii "\n"

min_int64_str:
  .ascii "-9223372036854775808"

.section .bss
.align 8
input_buf:
  .space INPUT_SIZE

.align 8
rpn_stack:
  .space 32768
