.text
.globl _start

_start:
    bl read_all
    mov x19, x0
    mov x20, #0

line_scan:
    cmp x20, x19
    b.hs done_input

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    add x21, x9, x20
    mov x22, x20

scan_newline:
    cmp x22, x19
    b.hs final_line

    add x23, x9, x22
    ldrb w10, [x23]
    cmp w10, #10
    b.eq newline_found

    add x22, x22, #1
    b scan_newline

newline_found:
    mov x0, x21
    sub x1, x22, x20
    bl eval_line
    add x20, x22, #1
    b line_scan

final_line:
    cmp x20, x19
    b.hs done_input
    mov x22, x19
    mov x0, x21
    sub x1, x22, x20
    bl eval_line
    b done_input

done_input:
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0

read_loop:
    mov x0, #0
    mov x1, x9
    add x1, x1, x10
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x10
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    beq read_done
    add x10, x10, x0
    mov x11, #1
    lsl x11, x11, #16
    cmp x10, x11
    b.hs read_done
    b read_loop

read_done:
    mov x0, x10
    ret

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

eval_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    adrp x19, eval_stack
    add x19, x19, :lo12:eval_stack
    mov x20, #0
    mov x21, x0
    add x22, x0, x1
    mov x23, #0

line_loop:
    cmp x21, x22
    b.hs finish_line

skip_separators:
    cmp x21, x22
    b.hs finish_line
    ldrb w24, [x21]
    cmp w24, #0x20
    b.eq skip_one
    cmp w24, #0x09
    b.eq skip_one
    b scan_token

skip_one:
    add x21, x21, #1
    b skip_separators

scan_token:
    mov x24, x21

scan_token_end:
    cmp x21, x22
    b.hs token_done
    ldrb w25, [x21]
    cmp w25, #0x20
    b.eq token_done
    cmp w25, #0x09
    b.eq token_done
    add x21, x21, #1
    b scan_token_end

token_done:
    sub x25, x21, x24
    cmp x25, #1
    b.ne parse_literal

    ldrb w25, [x24]
    cmp w25, #43
    b.eq do_plus
    cmp w25, #45
    b.eq do_minus
    cmp w25, #42
    b.eq do_mul
    cmp w25, #47
    b.eq do_div
    cmp w25, #37
    b.eq do_mod
    b parse_literal

parse_literal:
    mov x26, x24
    mov x27, #0
    ldrb w25, [x26]
    cmp w25, #45
    b.ne parse_digits
    mov x27, #1
    add x26, x26, #1
    sub x25, x25, #0
    b parse_digits

parse_digits:
    cmp x25, #18
    b.hi token_invalid
    mov x28, #0

numeric_loop:
    cmp x26, x21
    b.hs numeric_done
    ldrb w9, [x26]
    cmp w9, #48
    b.lt token_invalid
    cmp w9, #57
    b.gt token_invalid
    sub w9, w9, #48
    mov x10, #10
    madd x28, x28, x10, x9
    add x26, x26, #1
    b numeric_loop

numeric_done:
    cmp x27, #0
    b.eq push_literal_value
    neg x28, x28
    b push_literal_value

push_literal_value:
    str x28, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_plus:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    add x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_minus:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    sub x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_mul:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    mul x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_div:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    cmp x2, #0
    b.eq line_error
    mov x9, #0x8000
    lsl x9, x9, #48
    cmp x1, x9
    b.ne division_ok
    mov x9, #-1
    cmp x2, x9
    b.eq line_error

division_ok:
    sdiv x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_mod:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    cmp x2, #0
    b.eq line_error
    mov x9, #0x8000
    lsl x9, x9, #48
    cmp x1, x9
    b.ne modulo_ok
    mov x9, #-1
    cmp x2, x9
    b.eq line_error

modulo_ok:
    sdiv x3, x1, x2
    msub x0, x3, x2, x1
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

token_invalid:
    mov x23, #1
    b finish_line

line_error:
    mov x23, #1
    b finish_line

finish_line:
    cmp x23, #0
    b.ne emit_error
    cmp x20, #1
    b.ne emit_error
    ldr x0, [x19]
    bl print_i64
    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    bl write_fd
    b eval_line_done

emit_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #6
    bl write_fd

eval_line_done:
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

print_i64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!

    mov x19, x0
    mov x20, #0

    mov x23, #0x8000
    lsl x23, x23, #48
    cmp x19, x23
    b.eq print_min

    cmp x19, #0
    b.ge positive_or_zero
    mov x20, #1
    neg x19, x19

positive_or_zero:
    adrp x21, print_buf
    add x21, x21, :lo12:print_buf
    add x22, x21, #32
    mov x23, x22

    cmp x19, #0
    b.ne print_digits
    sub x23, x23, #1
    mov w24, #48
    strb w24, [x23]
    b maybe_sign

print_digits:
    mov x24, #10
    udiv x25, x19, x24
    msub x24, x25, x24, x19
    add w24, w24, #48
    sub x23, x23, #1
    strb w24, [x23]
    mov x19, x25
    cmp x19, #0
    b.ne print_digits

maybe_sign:
    cmp x20, #0
    b.eq write_printbuf
    sub x23, x23, #1
    mov w24, #45
    strb w24, [x23]

write_printbuf:
    sub x2, x22, x23
    mov x1, x23
    mov x0, #1
    bl write_fd
    b print_i64_done

print_min:
    mov x0, #1
    adrp x1, min_str
    add x1, x1, :lo12:min_str
    mov x2, #20
    bl write_fd

print_i64_done:
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

write_fd:
    mov x8, #64
    svc #0
    ret

.section .rodata
err_msg:
    .asciz "error\n"

newline_byte:
    .byte 10

min_str:
    .asciz "-9223372036854775808"

.section .bss
.align 4
input_buf:
    .space 65536

.align 4
eval_stack:
    .space 16384

.align 4
print_buf:
    .space 32
