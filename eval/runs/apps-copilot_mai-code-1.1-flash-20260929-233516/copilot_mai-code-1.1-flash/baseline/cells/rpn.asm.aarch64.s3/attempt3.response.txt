.text
.globl _start

_start:
    bl read_all
    mov x19, x0
    mov x20, #0
    cmp x19, #0
    b.eq done_input

parse_loop:
    cmp x20, x19
    b.hs done_input

    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    add x23, x21, x20
    mov x22, x20

scan_line:
    cmp x22, x19
    b.hs final_line
    add x24, x21, x22
    ldrb w25, [x24]
    cmp w25, #10
    b.eq newline_found
    add x22, x22, #1
    b scan_line

newline_found:
    mov x0, x23
    sub x1, x22, x20
    bl eval_line
    add x20, x22, #1
    b parse_loop

final_line:
    mov x0, x23
    sub x1, x19, x20
    bl eval_line
    b done_input

done_input:
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_fail
    beq read_done
    add x20, x20, x0
    cmp x20, #65536
    b.hs read_done
    b read_loop

read_done:
    mov x0, x20
    ret

read_fail:
    mov x0, #1
    mov x8, #93
    svc #0

eval_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    adrp x19, eval_stack
    add x19, x19, :lo12:eval_stack
    mov x20, #0
    mov x21, x0
    add x22, x0, x1
    mov x23, #0

line_loop:
    cmp x21, x22
    b.hs finish_line

skip_sep:
    cmp x21, x22
    b.hs finish_line
    ldrb w24, [x21]
    cmp w24, #0x20
    b.eq skip_one
    cmp w24, #0x09
    b.eq skip_one
    b token_start

skip_one:
    add x21, x21, #1
    b skip_sep

token_start:
    mov x24, x21

token_scan:
    cmp x21, x22
    b.hs token_done
    ldrb w25, [x21]
    cmp w25, #0x20
    b.eq token_done
    cmp w25, #0x09
    b.eq token_done
    add x21, x21, #1
    b token_scan

token_done:
    sub x26, x21, x24
    cmp x26, #1
    b.ne parse_literal
    ldrb w25, [x24]
    cmp w25, #43
    b.eq do_plus
    cmp w25, #45
    b.eq do_minus
    cmp w25, #42
    b.eq do_mul
    cmp w25, #47
    b.eq do_div
    cmp w25, #37
    b.eq do_mod
    b parse_literal

parse_literal:
    mov x27, #0
    mov x28, x24
    ldrb w25, [x28]
    cmp w25, #45
    b.ne digit_count
    mov x27, #1
    add x28, x28, #1

digit_count:
    sub x9, x21, x28
    cmp x9, #0
    b.eq invalid_token
    cmp x9, #18
    b.hi invalid_token
    mov x10, #0
    mov x11, x28

digit_loop:
    cmp x11, x21
    b.hs literal_done
    ldrb w12, [x11]
    cmp w12, #48
    b.lt invalid_token
    cmp w12, #57
    b.gt invalid_token
    sub w12, w12, #48
    mov x13, #10
    madd x10, x10, x13, x12
    add x11, x11, #1
    b digit_loop

literal_done:
    cmp x27, #0
    b.eq push_literal
    neg x10, x10

push_literal:
    str x10, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_plus:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    add x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_minus:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    sub x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_mul:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    mul x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_div:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    cmp x2, #0
    b.eq invalid_token
    mov x9, #0x8000
    lsl x9, x9, #48
    cmp x1, x9
    b.ne div_ok
    mov x9, #-1
    cmp x2, x9
    b.eq invalid_token
div_ok:
    sdiv x0, x1, x2
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

do_mod:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    ldr x2, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x1, [x19, x20, lsl #3]
    cmp x2, #0
    b.eq invalid_token
    mov x9, #0x8000
    lsl x9, x9, #48
    cmp x1, x9
    b.ne mod_ok
    mov x9, #-1
    cmp x2, x9
    b.eq invalid_token
mod_ok:
    sdiv x3, x1, x2
    msub x0, x3, x2, x1
    str x0, [x19, x20, lsl #3]
    add x20, x20, #1
    b line_loop

invalid_token:
    mov x23, #1

finish_line:
    cmp x23, #0
    b.ne emit_error
    cmp x20, #1
    b.ne emit_error
    ldr x0, [x19]
    bl print_i64
    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    bl write_fd
    b eval_done

emit_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #6
    bl write_fd

eval_done:
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

write_fd:
    mov x8, #64
    svc #0
    ret

print_i64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!

    mov x19, x0
    mov x20, #0

    mov x9, #0x8000
    lsl x9, x9, #48
    cmp x19, x9
    b.eq print_min

    cmp x19, #0
    b.ge positive
    mov x20, #1
    neg x19, x19

positive:
    adrp x21, print_buf
    add x21, x21, :lo12:print_buf
    add x22, x21, #32
    mov x23, x22

    cmp x19, #0
    b.ne digit_loop2
    sub x23, x23, #1
    mov w24, #48
    strb w24, [x23]
    b maybe_sign

digit_loop2:
    mov x24, #10
    udiv x25, x19, x24
    msub x24, x25, x24, x19
    add w24, w24, #48
    sub x23, x23, #1
    strb w24, [x23]
    mov x19, x25
    cmp x19, #0
    b.ne digit_loop2

maybe_sign:
    cmp x20, #0
    b.eq write_out
    sub x23, x23, #1
    mov w24, #45
    strb w24, [x23]

write_out:
    sub x2, x22, x23
    mov x1, x23
    mov x0, #1
    bl write_fd
    b done_print

print_min:
    mov x0, #1
    adrp x1, min_str
    add x1, x1, :lo12:min_str
    mov x2, #20
    bl write_fd

done_print:
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

.section .rodata
err_msg:
    .asciz "error\n"

newline_byte:
    .byte 10

min_str:
    .asciz "-9223372036854775808"

.section .bss
.align 4
input_buf:
    .space 65536

.align 4
eval_stack:
    .space 4096

.align 4

    .space 32
