.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    cmp rax, 0
    jl fail_exit
    je read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    cmp r12, 0
    je exit_ok
    xor r14, r14

process_loop:
    cmp r14, r12
    jae exit_ok
    mov r15, r14

scan_line:
    cmp r15, r12
    jae final_line
    lea rbx, [rip + input_buffer]
    mov al, byte ptr [rbx + r15]
    cmp al, 0x0A
    je line_done
    inc r15
    jmp scan_line

line_done:
    mov rbx, r15
    sub rbx, r14
    test rbx, rbx
    jz emit_empty_line
    lea rdi, [rip + input_buffer]
    add rdi, r14
    mov rsi, rbx
    call reverse_range
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, rbx
    call write_all
    jmp emit_newline

emit_empty_line:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all

emit_newline:
    mov r14, r15
    inc r14
    jmp process_loop

final_line:
    mov rbx, r12
    sub rbx, r14
    test rbx, rbx
    je exit_ok
    lea rdi, [rip + input_buffer]
    add rdi, r14
    mov rsi, rbx
    call reverse_range
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, rbx
    call write_all
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all
    jmp exit_ok

reverse_range:
    cmp rsi, 1
    jbe reverse_done
    lea r8, [rdi + rsi]
    dec r8
    mov rcx, rsi
    shr rcx, 1
    jz reverse_done

reverse_loop:
    mov r9b, byte ptr [rdi]
    mov r10b, byte ptr [r8]
    mov byte ptr [rdi], r10b
    mov byte ptr [r8], r9b
    inc rdi
    dec r8
    dec rcx
    jnz reverse_loop

reverse_done:
    ret

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov rax, 1
    syscall
    cmp rax, 0
    jl fail_exit
    cmp rax, 0
    je fail_exit
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_loop

write_done:
    ret

fail_exit:
    mov rax, 60
    mov rdi, 1
    syscall

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
newline_byte:
    .byte 0x0A

.bss
    .align 16
input_buffer:
    .space 65536
