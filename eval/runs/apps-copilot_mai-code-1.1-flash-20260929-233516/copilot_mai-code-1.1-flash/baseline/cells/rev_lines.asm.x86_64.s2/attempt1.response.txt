.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz .input_too_large
    mov rax, 0
    syscall
    cmp rax, 0
    jl .fail
    je .done_reading
    add r12, rax
    cmp r12, 65536
    jb read_loop
.done_reading:
    xor r13, r13

process_loop:
    cmp r13, r12
    jae .exit_success
    mov r14, r13

scan_line:
    cmp r13, r12
    jae .final_line
    lea rsi, [rip + input_buffer]
    mov dl, byte ptr [rsi + r13]
    cmp dl, 0x0A
    je .line_found
    inc r13
    jmp scan_line

.line_found:
    mov r15, r13
    sub r15, r14
    cmp r15, 0
    jne .emit_segment
    jmp .emit_newline

.emit_segment:
    lea rdi, [rip + input_buffer]
    add rdi, r14
    mov rsi, r15
    call reverse_range
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, r15
    call write_all
    jmp .emit_newline

.emit_newline:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all
    inc r13
    jmp process_loop

.final_line:
    mov r15, r12
    sub r15, r14
    cmp r15, 0
    je .end_with_newline
    lea rdi, [rip + input_buffer]
    add rdi, r14
    mov rsi, r15
    call reverse_range
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, r15
    call write_all
.end_with_newline:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all
    jmp .exit_success

.exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

.fail:
    mov rax, 60
    mov rdi, 1
    syscall

.input_too_large:
    mov rax, 60
    mov rdi, 1
    syscall

reverse_range:
    cmp rsi, 1
    jbe .done
    mov rax, rdi
    add rax, rsi
    dec rax
    mov rcx, rsi
    shr rcx, 1
.loop:
    mov r8b, byte ptr [rdi]
    mov r9b, byte ptr [rax]
    mov byte ptr [rdi], r9b
    mov byte ptr [rax], r8b
    inc rdi
    dec rax
    dec rcx
    jnz .loop
.done:
    ret

write_all:
    test rdx, rdx
    jz .done
.loop:
    mov rax, 1
    syscall
    cmp rax, 0
    jl .fail
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz .loop
.done:
    ret
.fail:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
newline_byte:
    .byte 0x0A

.bss
    .align 16
input_buffer:
    .space 65536
