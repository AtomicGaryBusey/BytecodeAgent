.globl _start

.text
_start:
    xor r15d, r15d

read_more:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, 65536
    sub rdx, r15
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_more

read_done:
    lea r12, [rip + input_buf]
    mov r13, r12
    mov r14, r12
    add r14, r15

scan_lines:
    cmp r13, r14
    jae exit_program
    mov rbx, r13

scan_to_nl:
    cmp rbx, r14
    je eof_line
    cmp byte ptr [rbx], 0x0A
    je newline_line
    inc rbx
    jmp scan_to_nl

newline_line:
    mov rdi, r13
    mov rsi, rbx
    mov r15, rbx
    inc r15
    mov qword ptr [rip + line_flag], 1
    jmp eval_line

eof_line:
    mov rdi, r13
    mov rsi, r14
    mov r15, r14
    mov qword ptr [rip + line_flag], 0
    jmp eval_line

eval_line:
    lea rbx, [rip + stack_values]
    xor r12d, r12d
    mov r10, rdi
    mov r14, rsi

line_loop:
    cmp r10, r14
    jae finalize

skip_sep:
    cmp r10, r14
    jae finalize
    mov al, byte ptr [r10]
    cmp al, ' '
    je sep_inc
    cmp al, 0x09
    je sep_inc
    jmp parse_token

sep_inc:
    inc r10
    jmp skip_sep

parse_token:
    mov r13, r10
scan_token:
    cmp r10, r14
    jae token_done
    mov al, byte ptr [r10]
    cmp al, ' '
    je token_done
    cmp al, 0x09
    je token_done
    inc r10
    jmp scan_token

token_done:
    mov r9, r10
    sub r9, r13
    mov al, byte ptr [r13]

    cmp r9, 1
    jne token_not_single

    cmp al, '+'
    je do_add
    cmp al, '-'
    je do_sub
    cmp al, '*'
    je do_mul
    cmp al, '/'
    je do_div
    cmp al, '%'
    je do_mod
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    mov rdi, r13
    mov rsi, r9
    jmp parse_unsigned

token_not_single:
    cmp al, '-'
    jne token_maybe_unsigned
    mov rdi, r13
    add rdi, 1
    mov rsi, r9
    dec rsi
    jmp parse_signed

token_maybe_unsigned:
    cmp al, '+'
    je invalid_line
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    mov rdi, r13
    mov rsi, r9
    jmp parse_unsigned

parse_unsigned:
    cmp rsi, 19
    ja invalid_line
    xor rax, rax
    mov r8, 922337203685477580
    mov r9, 7
    mov r10, rdi
parse_unsigned_loop:
    test rsi, rsi
    jz push_unsigned
    movzx ecx, byte ptr [r10]
    cmp cl, '0'
    jb invalid_line
    cmp cl, '9'
    ja invalid_line
    sub cl, '0'
    cmp rax, r8
    ja invalid_line
    jne pu_continue
    cmp cl, r9b
    ja invalid_line
pu_continue:
    imul rax, rax, 10
    add rax, rcx
    inc r10
    dec rsi
    jmp parse_unsigned_loop

parse_signed:
    cmp rsi, 19
    ja invalid_line
    xor rax, rax
    mov r8, 922337203685477580
    mov r9, 8
    mov r10, rdi
parse_signed_loop:
    test rsi, rsi
    jz push_signed
    movzx ecx, byte ptr [r10]
    cmp cl, '0'
    jb invalid_line
    cmp cl, '9'
    ja invalid_line
    sub cl, '0'
    cmp rax, r8
    ja invalid_line
    jne ps_continue
    cmp cl, r9b
    ja invalid_line
ps_continue:
    imul rax, rax, 10
    add rax, rcx
    inc r10
    dec rsi
    jmp parse_signed_loop

push_unsigned:
    mov qword ptr [rbx + r12*8], rax
    inc r12
    jmp line_loop

push_signed:
    neg rax
    mov qword ptr [rbx + r12*8], rax
    inc r12
    jmp line_loop

do_add:
    cmp r12, 2
    jb invalid_line
    mov rcx, r12
    dec rcx
    mov rax, qword ptr [rbx + rcx*8]
    dec rcx
    mov rdx, qword ptr [rbx + rcx*8]
    add rdx, rax
    mov r8, r12
    sub r8, 2
    mov qword ptr [rbx + r8*8], rdx
    mov r12, r8
    inc r12
    jmp line_loop

do_sub:
    cmp r12, 2
    jb invalid_line
    mov rcx, r12
    dec rcx
    mov rax, qword ptr [rbx + rcx*8]
    dec rcx
    mov rdx, qword ptr [rbx + rcx*8]
    sub rdx, rax
    mov r8, r12
    sub r8, 2
    mov qword ptr [rbx + r8*8], rdx
    mov r12, r8
    inc r12
    jmp line_loop

do_mul:
    cmp r12, 2
    jb invalid_line
    mov rcx, r12
    dec rcx
    mov rax, qword ptr [rbx + rcx*8]
    dec rcx
    mov rdx, qword ptr [rbx + rcx*8]
    imul rdx, rax
    mov r8, r12
    sub r8, 2
    mov qword ptr [rbx + r8*8], rdx
    mov r12, r8
    inc r12
    jmp line_loop

do_div:
    cmp r12, 2
    jb invalid_line
    mov rcx, r12
    dec rcx
    mov r8, qword ptr [rbx + rcx*8]
    dec rcx
    mov r9, qword ptr [rbx + rcx*8]
    cmp r8, 0
    je invalid_line
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne div_ok
    cmp r8, -1
    je invalid_line
div_ok:
    mov rax, r9
    cqo
    idiv r8
    mov r8, r12
    sub r8, 2
    mov qword ptr [rbx + r8*8], rax
    mov r12, r8
    inc r12
    jmp line_loop

do_mod:
    cmp r12, 2
    jb invalid_line
    mov rcx, r12
    dec rcx
    mov r8, qword ptr [rbx + rcx*8]
    dec rcx
    mov r9, qword ptr [rbx + rcx*8]
    cmp r8, 0
    je invalid_line
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne mod_ok
    cmp r8, -1
    je invalid_line
mod_ok:
    mov rax, r9
    cqo
    idiv r8
    mov rax, rdx
    mov r8, r12
    sub r8, 2
    mov qword ptr [rbx + r8*8], rax
    mov r12, r8
    inc r12
    jmp line_loop

finalize:
    cmp r12, 1
    jne invalid_line
    mov rax, qword ptr [rbx]
    lea r15, [rip + output_buf + 63]
    mov byte ptr [r15], 0x0A
    mov r14, r15
    mov r13, 0
    cmp rax, 0
    jge value_nonneg
    mov r13, 1
    mov r8, 0x8000000000000000
    cmp rax, r8
    je value_min
    neg rax
    jmp value_digits
value_min:
    mov rax, 0x8000000000000000
value_digits:
    xor edx, edx
    mov r8, 10
value_loop:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, '0'
    dec r14
    mov byte ptr [r14], dl
    test rax, rax
    jnz value_loop
    cmp r13, 1
    jne value_done
    dec r14
    mov byte ptr [r14], '-'
value_done:
    mov rsi, r14
    lea rdx, [rip + output_buf + 64]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    jmp output_done

invalid_line:
    lea rsi, [rip + err_msg]
    mov rdx, 6
    mov rdi, 1
    mov rax, 1
    syscall

output_done:
    mov r13, r15
    mov rax, qword ptr [rip + line_flag]
    test rax, rax
    jz exit_program
    jmp scan_lines

value_nonneg:
    mov r13, 0
    jmp value_digits

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .ascii "error\n"

.section .bss
.align 8
input_buf:
    .zero 65536
.align 8
stack_values:
    .zero 8192
.align 8
output_buf:
    .zero 64
.align 8
line_flag:
    .zero 8
