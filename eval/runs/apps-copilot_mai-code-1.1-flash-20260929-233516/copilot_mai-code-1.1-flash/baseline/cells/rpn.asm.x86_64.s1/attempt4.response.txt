.globl _start

.text
_start:
    mov qword ptr [rip + in_len], 0
    call read_all_input

    mov rbx, qword ptr [rip + in_len]
    test rbx, rbx
    jz exit_ok

    lea r13, [rip + input_buf]
    lea r14, [rip + input_buf]
    add r14, rbx

line_loop:
    cmp r13, r14
    jae exit_ok

    mov rdi, r13
    mov rsi, r13

line_scan:
    cmp rsi, r14
    jae line_scan_eof

    cmp byte ptr [rsi], 0x0A
    je line_found_newline
    inc rsi
    jmp line_scan

line_found_newline:
    mov rsi, rsi
    call process_line
    mov r13, rsi
    inc r13
    jmp line_loop

line_scan_eof:
    mov rsi, r14
    mov rdi, r13
    call process_line
    jmp exit_ok

read_all_input:
    mov rbx, qword ptr [rip + in_len]
read_more:
    cmp rbx, 65536
    jae read_done

    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall

    test rax, rax
    jle read_done

    add rbx, rax
    mov qword ptr [rip + in_len], rbx
    jmp read_more

read_done:
    ret

process_line:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov r13, rdi
    mov r14, rsi
    lea rbx, [rip + line_stack]
    xor r12, r12
    xor r15, r15

line_top:
    cmp r13, r14
    jae finalize_line

skip_space:
    cmp r13, r14
    jae finalize_line
    movzx eax, byte ptr [r13]
    cmp al, ' '
    je skip_space_advance
    cmp al, 0x09
    je skip_space_advance
    jmp parse_token

skip_space_advance:
    inc r13
    jmp skip_space

parse_token:
    mov r8, r13

scan_token:
    cmp r13, r14
    jae token_done
    movzx eax, byte ptr [r13]
    cmp al, ' '
    je token_done
    cmp al, 0x09
    je token_done
    inc r13
    jmp scan_token

token_done:
    mov r9, r13
    sub r9, r8
    movzx eax, byte ptr [r8]

    cmp r9, 1
    jne parse_long_token

    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token

    movzx rax, byte ptr [r8]
    sub rax, '0'
    jmp push_literal

parse_long_token:
    cmp al, '-'
    jne check_plus
    mov rcx, r9
    dec rcx
    jz invalid_token
    cmp rcx, 18
    ja invalid_token

    mov rax, 0
    mov rsi, r8
    inc rsi
    xor rdx, rdx

neg_digit_loop:
    cmp rdx, rcx
    je neg_digit_done
    movzx r10d, byte ptr [rsi]
    cmp r10b, '0'
    jb invalid_token
    cmp r10b, '9'
    ja invalid_token
    imul rax, rax, 10
    movzx r10d, r10b
    sub r10d, '0'
    add rax, r10
    inc rsi
    inc rdx
    jmp neg_digit_loop

neg_digit_done:
    neg rax
    jmp push_literal

check_plus:
    cmp al, '+'
    je invalid_token
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token

    cmp r9, 18
    ja invalid_token

    mov rax, 0
    mov rsi, r8
    xor rdx, rdx

pos_digit_loop:
    cmp rdx, r9
    je pos_digit_done
    movzx r10d, byte ptr [rsi]
    cmp r10b, '0'
    jb invalid_token
    cmp r10b, '9'
    ja invalid_token
    imul rax, rax, 10
    movzx r10d, r10b
    sub r10d, '0'
    add rax, r10
    inc rsi
    inc rdx
    jmp pos_digit_loop

pos_digit_done:
    jmp push_literal

push_literal:
    mov qword ptr [rbx + r12*8], rax
    inc r12
    jmp line_top

invalid_token:
    mov r15, 1
    mov r13, r14
    jmp finalize_line

op_plus:
    cmp r12, 2
    jb invalid_token
    mov r8, r12
    sub r8, 1
    mov rax, qword ptr [rbx + r8*8]
    mov r8, r12
    sub r8, 2
    mov rdx, qword ptr [rbx + r8*8]
    add rdx, rax
    mov qword ptr [rbx + r8*8], rdx
    dec r12
    jmp line_top

op_minus:
    cmp r12, 2
    jb invalid_token
    mov r8, r12
    sub r8, 1
    mov rax, qword ptr [rbx + r8*8]
    mov r8, r12
    sub r8, 2
    mov rdx, qword ptr [rbx + r8*8]
    sub rdx, rax
    mov qword ptr [rbx + r8*8], rdx
    dec r12
    jmp line_top

op_mul:
    cmp r12, 2
    jb invalid_token
    mov r8, r12
    sub r8, 1
    mov rax, qword ptr [rbx + r8*8]
    mov r8, r12
    sub r8, 2
    mov rdx, qword ptr [rbx + r8*8]
    imul rdx, rax
    mov qword ptr [rbx + r8*8], rdx
    dec r12
    jmp line_top

op_div:
    cmp r12, 2
    jb invalid_token

    mov r8, r12
    sub r8, 1
    mov rdx, qword ptr [rbx + r8*8]
    test rdx, rdx
    jz invalid_token

    mov r8, r12
    sub r8, 2
    mov rax, qword ptr [rbx + r8*8]
    cmp rax, -9223372036854775808
    jne div_ok
    cmp rdx, -1
    je invalid_token

div_ok:
    mov r10, rdx
    cqo
    idiv r10
    mov qword ptr [rbx + r8*8], rax
    dec r12
    jmp line_top

op_mod:
    cmp r12, 2
    jb invalid_token

    mov r8, r12
    sub r8, 1
    mov rdx, qword ptr [rbx + r8*8]
    test rdx, rdx
    jz invalid_token

    mov r8, r12
    sub r8, 2
    mov rax, qword ptr [rbx + r8*8]
    cmp rax, -9223372036854775808
    jne mod_ok
    cmp rdx, -1
    je invalid_token

mod_ok:
    mov r10, rdx
    cqo
    idiv r10
    mov rax, rdx
    mov qword ptr [rbx + r8*8], rax
    dec r12
    jmp line_top

finalize_line:
    test r15, r15
    jnz print_error_only

    cmp r12, 1
    jne print_error_only

    mov rax, qword ptr [rbx]
    call print_signed_64
    jmp process_line_done

print_error_only:
    call print_error

process_line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_error:
    lea rsi, [rip + error_msg]
    mov rdx, 6
    mov rax, 1
    mov rdi, 1
    syscall
    ret

print_signed_64:
    lea r8, [rip + out_buf]
    add r8, 63
    mov byte ptr [r8], 0x0A
    test rax, rax
    jns print_abs

    cmp rax, -9223372036854775808
    je print_min_int

    neg rax
    dec r8
    mov byte ptr [r8], '-'

print_abs:
    mov r9, 10
digit_loop:
    xor rdx, rdx
    div r9
    mov r10b, dl
    add r10b, '0'
    dec r8
    mov byte ptr [r8], r10b
    test rax, rax
    jnz digit_loop

    mov rsi, r8
    lea rdx, [rip + out_buf]
    add rdx, 64
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    ret

print_min_int:
    lea rsi, [rip + min_int_msg]
    mov rdx, 21
    mov rax, 1
    mov rdi, 1
    syscall
    ret

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
error_msg:
    .ascii "error\n"
min_int_msg:
    .ascii "-9223372036854775808\n"

.section .bss
.align 8
in_len:
    .zero 8

.align 8
input_buf:
    .zero 65536

.align 8
line_stack:
    .zero 8192

.align 8
out_buf:
    .zero 64
