.globl _start

.text
_start:
    mov qword ptr [rip + in_len], 0

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rbx, qword ptr [rip + in_len]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    mov rbx, qword ptr [rip + in_len]
    add rbx, rax
    mov qword ptr [rip + in_len], rbx
    jmp read_loop

read_done:
    lea r13, [rip + input_buf]
    mov r15, qword ptr [rip + in_len]
    lea r14, [rip + input_buf]
    add r15, r14

process_lines:
    cmp r13, r15
    jae exit_ok
    mov r11, r13
    jmp find_line_end

find_line_end:
    cmp r11, r15
    jae line_end_eof
    cmp byte ptr [r11], 0x0A
    je line_end_nl
    inc r11
    jmp find_line_end

line_end_nl:
    mov rdx, r11
    lea r13, [r11 + 1]
    jmp eval_line

line_end_eof:
    mov rdx, r15
    mov r13, r15
    jmp eval_line

eval_line:
    lea rbp, [rip + stack_buf]
    xor r10d, r10d
    mov r11, r13
    mov r12, rdx
    jmp line_top

line_top:
    cmp r11, r12
    jae finalize_line
skip_separators:
    cmp r11, r12
    jae finalize_line
    mov al, byte ptr [r11]
    cmp al, ' '
    je sep_advance
    cmp al, 0x09
    je sep_advance
    jmp parse_token

sep_advance:
    inc r11
    jmp skip_separators

parse_token:
    mov r8, r11
scan_token:
    cmp r11, r12
    jae token_done
    mov al, byte ptr [r11]
    cmp al, ' '
    je token_done
    cmp al, 0x09
    je token_done
    inc r11
    jmp scan_token

token_done:
    mov r9, r11
    sub r9, r8
    mov al, byte ptr [r8]

    cmp r9, 1
    jne token_longer

    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    movzx rax, byte ptr [r8]
    sub rax, '0'
    jmp push_literal

token_longer:
    cmp al, '-'
    jne maybe_unsigned
    mov rbx, r9
    dec rbx
    cmp rbx, 0
    je invalid_line
    cmp rbx, 18
    ja invalid_line
    mov rax, 0
    mov rcx, r8
    inc rcx
    mov rdx, rbx
    jmp parse_negative_loop

maybe_unsigned:
    cmp al, '+'
    je invalid_line
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    mov rax, 0
    mov rcx, r8
    mov rdx, r9
    cmp rdx, 18
    ja invalid_line
    jmp parse_unsigned_loop

parse_negative_loop:
    test rdx, rdx
    jz finish_negative
    movzx rbx, byte ptr [rcx]
    cmp rbx, '0'
    jb invalid_line
    cmp rbx, '9'
    ja invalid_line
    sub rbx, '0'
    imul rax, rax, 10
    add rax, rbx
    inc rcx
    dec rdx
    jmp parse_negative_loop

finish_negative:
    neg rax
    jmp push_literal

parse_unsigned_loop:
    test rdx, rdx
    jz push_literal
    movzx rbx, byte ptr [rcx]
    cmp rbx, '0'
    jb invalid_line
    cmp rbx, '9'
    ja invalid_line
    sub rbx, '0'
    imul rax, rax, 10
    add rax, rbx
    inc rcx
    dec rdx
    jmp parse_unsigned_loop

push_literal:
    mov qword ptr [rbp + r10*8], rax
    inc r10
    jmp line_top

op_plus:
    cmp r10, 2
    jb invalid_line
    mov r8, r10
    dec r8
    mov rax, qword ptr [rbp + r8*8]
    dec r8
    mov rdx, qword ptr [rbp + r8*8]
    add rdx, rax
    mov r8, r10
    sub r8, 2
    mov qword ptr [rbp + r8*8], rdx
    mov r10, r8
    inc r10
    jmp line_top

op_minus:
    cmp r10, 2
    jb invalid_line
    mov r8, r10
    dec r8
    mov rax, qword ptr [rbp + r8*8]
    dec r8
    mov rdx, qword ptr [rbp + r8*8]
    sub rdx, rax
    mov r8, r10
    sub r8, 2
    mov qword ptr [rbp + r8*8], rdx
    mov r10, r8
    inc r10
    jmp line_top

op_mul:
    cmp r10, 2
    jb invalid_line
    mov r8, r10
    dec r8
    mov rax, qword ptr [rbp + r8*8]
    dec r8
    mov rdx, qword ptr [rbp + r8*8]
    imul rdx, rax
    mov r8, r10
    sub r8, 2
    mov qword ptr [rbp + r8*8], rdx
    mov r10, r8
    inc r10
    jmp line_top

op_div:
    cmp r10, 2
    jb invalid_line
    mov r8, r10
    dec r8
    mov r9, qword ptr [rbp + r8*8]
    dec r8
    mov rax, qword ptr [rbp + r8*8]
    cmp r9, 0
    je invalid_line
    mov rbx, 0x8000000000000000
    cmp rax, rbx
    jne div_ok
    cmp r9, -1
    je invalid_line
div_ok:
    mov rbx, r9
    cqo
    idiv rbx
    mov r8, r10
    sub r8, 2
    mov qword ptr [rbp + r8*8], rax
    mov r10, r8
    inc r10
    jmp line_top

op_mod:
    cmp r10, 2
    jb invalid_line
    mov r8, r10
    dec r8
    mov r9, qword ptr [rbp + r8*8]
    dec r8
    mov rax, qword ptr [rbp + r8*8]
    cmp r9, 0
    je invalid_line
    mov rbx, 0x8000000000000000
    cmp rax, rbx
    jne mod_ok
    cmp r9, -1
    je invalid_line
mod_ok:
    mov rbx, r9
    cqo
    idiv rbx
    mov rax, rdx
    mov r8, r10
    sub r8, 2
    mov qword ptr [rbp + r8*8], rax
    mov r10, r8
    inc r10
    jmp line_top

op_minus_operator:
    jmp op_minus

op_plus_operator:
    jmp op_plus

finalize_line:
    cmp r10, 1
    jne invalid_line
    mov rax, qword ptr [rbp]
    mov r15, qword ptr [rip + out_buf_end]
    lea r15, [rip + out_buf]
    add r15, 63
    mov byte ptr [r15], 0x0A
    mov r14, r15
    test rax, rax
    jns print_nonneg
    mov rbx, 0x8000000000000000
    cmp rax, rbx
    je print_min_int
    neg rax
    dec r14
    mov byte ptr [r14], '-'
    jmp print_digits

print_nonneg:
    jmp print_digits

print_digits:
    mov rbx, rax
    jmp digits_loop

digits_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec r14
    mov byte ptr [r14], dl
    test rax, rax
    jnz digits_loop
    lea rsi, [r14]
    lea rdx, [rip + out_buf]
    add rdx, 64
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    jmp process_lines

print_min_int:
    lea rsi, [rip + min_int_str]
    mov rdx, 21
    mov rdi, 1
    mov rax, 1
    syscall
    jmp process_lines

invalid_line:
    lea rsi, [rip + err_msg]
    mov rdx, 6
    mov rdi, 1
    mov rax, 1
    syscall
    jmp process_lines

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .ascii "error\n"
min_int_str:
    .ascii "-9223372036854775808\n"

.section .bss
.align 8
in_len:
    .zero 8
.align 8
input_buf:
    .zero 65536
.align 8
stack_buf:
    .zero 4096
.align 8
out_buf:
    .zero 64
.align 8
out_buf_end:
    .zero 8
