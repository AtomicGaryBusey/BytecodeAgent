.globl _start

.text
_start:
    xor r15, r15
    lea rsi, [rip + input_buf]
read_loop:
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r15
    jz read_done
    lea rsi, [rip + input_buf + r15]
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_size], r15
    lea r13, [rip + input_buf]
    lea r14, [rip + input_buf]
    add r14, r15

parse_loop:
    cmp r13, r14
    jae exit_program
    mov rbx, r13
scan_nl:
    cmp rbx, r14
    jae no_newline
    cmp byte ptr [rbx], 0x0A
    je found_newline
    inc rbx
    jmp scan_nl

found_newline:
    mov qword ptr [rip + temp_ptr], rbx
    inc qword ptr [rip + temp_ptr]
    mov qword ptr [rip + temp_flag], 1
    mov rdi, r13
    mov rsi, rbx
    jmp eval_line

no_newline:
    mov qword ptr [rip + temp_ptr], r14
    mov qword ptr [rip + temp_flag], 0
    mov rdi, r13
    mov rsi, r14
    jmp eval_line

eval_line:
    sub rsp, 64
    lea rbx, [rip + stack_values]
    xor r9d, r9d
    mov r10, rdi
    mov r11, rsi

line_scan:
    cmp r10, r11
    jae finalize_line

skip_sep:
    cmp r10, r11
    jae finalize_line
    mov al, byte ptr [r10]
    cmp al, ' '
    je inc_pos
    cmp al, '\t'
    je inc_pos
    jmp token_parse

inc_pos:
    inc r10
    jmp skip_sep

token_parse:
    mov r13, r10
scan_token:
    cmp r10, r11
    jae token_done
    mov al, byte ptr [r10]
    cmp al, ' '
    je token_done
    cmp al, '\t'
    je token_done
    inc r10
    jmp scan_token

token_done:
    mov rdx, r10
    sub rdx, r13
    mov al, byte ptr [r13]
    cmp al, '+'
    jne check_minus
    cmp rdx, 1
    jne invalid_token
    jmp op_plus

check_minus:
    cmp al, '-'
    jne check_mul
    cmp rdx, 1
    je op_minus
    mov r15, 1
    lea rsi, [r13 + 1]
    sub rdx, 1
    cmp rdx, 18
    ja invalid_token
    mov r12, 922337203685477580
    mov r13, 8
    jmp parse_decimal

check_mul:
    cmp al, '*'
    jne check_div
    cmp rdx, 1
    jne invalid_token
    jmp op_mul

check_div:
    cmp al, '/'
    jne check_mod
    cmp rdx, 1
    jne invalid_token
    jmp op_div

check_mod:
    cmp al, '%'
    jne check_digit
    cmp rdx, 1
    jne invalid_token
    jmp op_mod

check_digit:
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    mov r15, 0
    mov rsi, r13
    mov r12, 922337203685477580
    mov r13, 7
    jmp parse_decimal

parse_decimal:
    cmp rdx, 0
    je invalid_token
    mov rax, 0

digit_loop:
    cmp rdx, 0
    je digits_done
    movzx r8, byte ptr [rsi]
    cmp r8b, '0'
    jb invalid_token
    cmp r8b, '9'
    ja invalid_token
    sub r8, '0'
    cmp rax, r12
    ja invalid_token
    je digit_mod_check
    jmp digit_continue

digit_mod_check:
    cmp r8, r13
    ja invalid_token

digit_continue:
    imul rax, rax, 10
    add rax, r8
    inc rsi
    dec rdx
    jmp digit_loop

digits_done:
    cmp r15, 1
    jne push_literal
    neg rax

push_literal:
    mov qword ptr [rbx + r9*8], rax
    inc r9
    jmp line_scan

op_plus:
    cmp r9, 2
    jb invalid_operator
    mov r13, r9
    dec r13
    mov rax, qword ptr [rbx + r13*8]
    dec r13
    mov r8, qword ptr [rbx + r13*8]
    mov r9, r13
    add r8, rax
    mov rax, r8
    jmp push_literal

op_minus:
    cmp r9, 2
    jb invalid_operator
    mov r13, r9
    dec r13
    mov rax, qword ptr [rbx + r13*8]
    dec r13
    mov r8, qword ptr [rbx + r13*8]
    mov r9, r13
    sub r8, rax
    mov rax, r8
    jmp push_literal

op_mul:
    cmp r9, 2
    jb invalid_operator
    mov r13, r9
    dec r13
    mov rax, qword ptr [rbx + r13*8]
    dec r13
    mov r8, qword ptr [rbx + r13*8]
    mov r9, r13
    imul r8, rax
    mov rax, r8
    jmp push_literal

op_div:
    cmp r9, 2
    jb invalid_operator
    mov r13, r9
    dec r13
    mov rax, qword ptr [rbx + r13*8]
    dec r13
    mov r8, qword ptr [rbx + r13*8]
    mov r9, r13
    cmp rax, 0
    je invalid_operator
    mov r10, -9223372036854775808
    cmp r8, r10
    jne div_ok
    cmp rax, -1
    je invalid_operator
div_ok:
    mov r10, rax
    mov rax, r8
    mov r8, r10
    cqo
    idiv r8
    jmp push_literal

op_mod:
    cmp r9, 2
    jb invalid_operator
    mov r13, r9
    dec r13
    mov rax, qword ptr [rbx + r13*8]
    dec r13
    mov r8, qword ptr [rbx + r13*8]
    mov r9, r13
    cmp rax, 0
    je invalid_operator
    mov r10, -9223372036854775808
    cmp r8, r10
    jne mod_ok
    cmp rax, -1
    je invalid_operator
mod_ok:
    mov r10, rax
    mov rax, r8
    mov r8, r10
    cqo
    idiv r8
    mov rax, rdx
    jmp push_literal

finalize_line:
    cmp r9, 1
    jne invalid_operator
    mov rax, qword ptr [rbx]
    lea r10, [rsp + 64]
    mov r13, 0
    cmp rax, 0
    jge num_build
    mov r13, 1
    neg rax

num_build:
    mov r14, rax
    test r14, r14
    jz build_zero
build_digits:
    xor rdx, rdx
    mov rax, r14
    mov r8, 10
    div r8
    add dl, '0'
    dec r10
    mov byte ptr [r10], dl
    mov r14, rax
    test r14, r14
    jnz build_digits
    jmp put_sign

build_zero:
    dec r10
    mov byte ptr [r10], '0'

put_sign:
    cmp r13, 1
    jne number_done
    dec r10
    mov byte ptr [r10], '-'

number_done:
    mov rsi, r10
    lea rdx, [rsp + 64]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    syscall
    jmp eval_done

invalid_token:
    lea rsi, [rip + err_msg]
    mov rdx, 6
    mov rdi, 1
    mov rax, 1
    syscall
    jmp eval_done

invalid_operator:
    lea rsi, [rip + err_msg]
    mov rdx, 6
    mov rdi, 1
    mov rax, 1
    syscall

eval_done:
    add rsp, 64
    cmp qword ptr [rip + temp_flag], 0
    je eval_done_no_newline
    mov r13, qword ptr [rip + temp_ptr]
    jmp parse_loop

eval_done_no_newline:
    mov r13, qword ptr [rip + temp_ptr]
    jmp exit_program

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .asciz "error\n"
newline_byte:
    .byte 0x0A

.section .bss
.align 8
input_buf:
    .space 65536
.align 8
stack_values:
    .space 32768
.align 8
input_size:
    .quad 0
.align 8
temp_ptr:
    .quad 0
.align 8
temp_flag:
    .quad 0
