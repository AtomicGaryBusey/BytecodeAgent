.globl _start
.text

_start:
    mov x19, xzr
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #32768
    sub x2, x2, x19
    cmp x2, #0
    b.le read_done
    mov x8, #63
    svc #0
    mov x21, x0
    cmp x21, #0
    b.eq read_done
    b.lt read_error
    add x19, x19, x21
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    mov x21, xzr
    mov x23, xzr
    mov x22, x20
    adrp x28, line_ptrs
    add x28, x28, :lo12:line_ptrs
    adrp x29, line_lens
    add x29, x29, :lo12:line_lens
    add x24, x20, x19

parse_loop:
    cmp x21, x19
    b.hs parse_end
    add x26, x20, x21
    ldrb w27, [x26]
    cmp w27, #10
    b.ne parse_next
    sub x27, x26, x22
    add x30, x28, x23, lsl #3
    str x22, [x30]
    add x30, x29, x23, lsl #2
    str w27, [x30]
    add x23, x23, #1
    add x21, x21, #1
    add x22, x26, #1
    b parse_loop

parse_next:
    add x21, x21, #1
    b parse_loop

parse_end:
    cmp x22, x24
    b.hs sort_start
    sub x27, x24, x22
    add x30, x28, x23, lsl #3
    str x22, [x30]
    add x30, x29, x23, lsl #2
    str w27, [x30]
    add x23, x23, #1

sort_start:
    mov x21, #1
sort_outer:
    cmp x21, x23
    b.ge write_all
    mov x22, x21
sort_inner:
    cmp x22, #0
    b.le sort_next_outer
    sub x24, x22, #1

    add x26, x28, x24, lsl #3
    ldr x0, [x26]
    add x26, x29, x24, lsl #2
    ldr w1, [x26]
    add x26, x28, x22, lsl #3
    ldr x2, [x26]
    add x26, x29, x22, lsl #2
    ldr w3, [x26]

    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, xzr

cmp_loop:
    cmp x5, x4
    b.hs cmp_done
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.lo cmp_lt
    b.hi cmp_gt
    add x5, x5, #1
    b cmp_loop

cmp_lt:
    mov x6, #-1
    b compare_done

cmp_gt:
    mov x6, #1
    b compare_done

cmp_done:
    cmp x1, x3
    b.lo cmp_lt
    b.hi cmp_gt
    mov x6, xzr

compare_done:
    cmp x6, #0
    b.le no_swap

    add x26, x28, x24, lsl #3
    add x27, x28, x22, lsl #3
    ldr x5, [x26]
    ldr x6, [x27]
    str x6, [x26]
    str x5, [x27]

    add x26, x29, x24, lsl #2
    add x27, x29, x22, lsl #2
    ldr w5, [x26]
    ldr w6, [x27]
    str w6, [x26]
    str w5, [x27]

    sub x22, x22, #1
    b sort_inner

no_swap:
    add x21, x21, #1
    b sort_outer

sort_next_outer:
    add x21, x21, #1
    b sort_outer

write_all:
    mov x21, xzr
write_loop:
    cmp x21, x23
    b.ge exit_ok
    add x26, x28, x21, lsl #3
    ldr x1, [x26]
    add x26, x29, x21, lsl #2
    ldr w2, [x26]

    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0

    add x21, x21, #1
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
.balign 1
newline:
    .byte 10

.section .bss
.balign 8
input_buf:
    .space 32768

.balign 8
line_ptrs:
    .space 8000

.balign 8
line_lens:
    .space 4000
