.globl _start

.text
_start:
    xor r12, r12
    lea rbx, [rip + in_buf]
    lea r15, [rip + in_buf + 49152]

read_loop:
    cmp rbx, r15
    je read_done
    mov rdx, r15
    sub rdx, rbx
    cmp rdx, 4096
    jbe read_count_ok
    mov rdx, 4096

read_count_ok:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    syscall
    cmp rax, 0
    jl read_fail
    je read_done
    add rbx, rax
    add r12, rax
    jmp read_loop

read_done:
    mov qword ptr [rip + in_len], r12
    mov rbx, qword ptr [rip + in_len]
    test rbx, rbx
    jz no_input

    lea rsi, [rip + in_buf]
    lea rdi, [rip + enc_buf]
    lea r15, [rip + alphabet]
    xor rcx, rcx
    xor r9, r9

encode_loop:
    mov rax, rcx
    add rax, 3
    cmp rax, rbx
    jg encode_tail

    movzx r8d, byte ptr [rsi + rcx]
    movzx r10d, byte ptr [rsi + rcx + 1]
    movzx r11d, byte ptr [rsi + rcx + 2]

    mov eax, r8d
    shr eax, 2

    mov edx, r8d
    and edx, 3
    shl edx, 4

    mov r13d, r10d
    shr r13d, 4
    or edx, r13d

    mov r13d, r10d
    and r13d, 15
    shl r13d, 2

    mov r14d, r11d
    shr r14d, 6
    or r13d, r14d

    mov r14d, r11d
    and r14d, 63

    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + rdx]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + r13]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + r14]
    mov byte ptr [rdi + r9], al
    inc r9

    add rcx, 3
    jmp encode_loop

encode_tail:
    mov rax, rbx
    sub rax, rcx
    cmp rax, 2
    je tail_two

    movzx r8d, byte ptr [rsi + rcx]

    mov eax, r8d
    shr eax, 2

    mov edx, r8d
    and edx, 3
    shl edx, 4

    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + rdx]
    mov byte ptr [rdi + r9], al
    inc r9

    mov byte ptr [rdi + r9], 61
    inc r9
    mov byte ptr [rdi + r9], 61
    inc r9
    jmp encode_done

tail_two:
    movzx r8d, byte ptr [rsi + rcx]
    movzx r10d, byte ptr [rsi + rcx + 1]

    mov eax, r8d
    shr eax, 2

    mov edx, r8d
    and edx, 3
    shl edx, 4

    mov r13d, r10d
    shr r13d, 4
    or edx, r13d

    mov r13d, r10d
    and r13d, 15
    shl r13d, 2

    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + rdx]
    mov byte ptr [rdi + r9], al
    inc r9

    movzx eax, byte ptr [r15 + r13]
    mov byte ptr [rdi + r9], al
    inc r9

    mov byte ptr [rdi + r9], 61
    inc r9

encode_done:
    mov qword ptr [rip + enc_len], r9
    jmp format_output

no_input:
    mov qword ptr [rip + enc_len], 0
    mov qword ptr [rip + final_len], 0
    jmp exit_zero

format_output:
    xor r10, r10
    xor r11, r11
    mov rbx, qword ptr [rip + enc_len]
    test rbx, rbx
    jz zero_output

format_loop:
    mov rax, rbx
    sub rax, r11
    cmp rax, 76
    jbe format_last_line
    mov rdx, 76
    jmp format_line_continue

format_last_line:
    mov rdx, rax

format_line_continue:
    mov rcx, rdx
    lea rsi, [rip + enc_buf]
    add rsi, r11
    lea rdi, [rip + final_buf]
    add rdi, r10

copy_line:
    test rcx, rcx
    jz after_copy
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp copy_line

after_copy:
    mov rax, r10
    add rax, rdx
    lea rdi, [rip + final_buf]
    add rdi, rax
    mov byte ptr [rdi], 10
    add rax, 1
    add r11, rdx
    mov r10, rax
    mov rax, rbx
    cmp r11, rax
    jb format_loop
    jmp format_done

zero_output:
    mov qword ptr [rip + final_len], 0
    jmp exit_zero

format_done:
    mov qword ptr [rip + final_len], r10

write_output:
    xor r12, r12
    mov rbx, qword ptr [rip + final_len]
    test rbx, rbx
    jz exit_zero

write_loop:
    cmp r12, rbx
    jae exit_zero
    mov rdx, rbx
    sub rdx, r12
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + final_buf]
    add rsi, r12
    syscall
    cmp rax, 0
    jl write_fail
    je write_fail
    add r12, rax
    jmp write_loop

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

read_fail:
    mov rax, 60
    mov edi, 1
    syscall

write_fail:
    mov rax, 60
    mov edi, 1
    syscall

    .space 49152

    .align 16

    .space 65536

    .align 16

    .space 70000

    .align 8

    .zero 8
enc_len:
    .zero 8
final_len:
    .zero 8
