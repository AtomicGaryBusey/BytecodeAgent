.globl _start

.text
_start:
    xor r12, r12
    lea r13, [rip + in_buf]
    lea r14, [rip + in_buf]
    mov r15d, 49152
    add r14, r15

read_loop:
    mov rdx, r14
    sub rdx, r13
    cmp rdx, 4096
    jb 1f
    mov rdx, 4096
1:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    syscall
    cmp rax, 0
    jl fail
    je read_done
    add r13, rax
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_zero

    lea rsi, [rip + in_buf]
    mov r13, rsi
    add r13, r12
    lea rdi, [rip + enc_buf]
    xor r14, r14

encode_loop:
    mov rax, r13
    sub rax, rsi
    cmp rax, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi + 1]
    movzx ecx, byte ptr [rsi + 2]

    mov edx, eax
    shr edx, 2

    mov r8d, eax
    and r8d, 3
    shl r8d, 4

    mov r9d, ebx
    shr r9d, 4
    or r8d, r9d

    mov r9d, ebx
    and r9d, 15
    shl r9d, 2

    mov r10d, ecx
    shr r10d, 6
    or r9d, r10d

    mov r10d, ecx
    and r10d, 63

    lea r11, [rip + alphabet]

    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r8]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r9]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r10]
    mov byte ptr [rdi + r14], al
    inc r14

    add rsi, 3
    jmp encode_loop

encode_tail:
    mov rax, r13
    sub rax, rsi
    cmp rax, 2
    je tail_two
    cmp rax, 1
    je tail_one
    jmp format_output

tail_one:
    movzx eax, byte ptr [rsi]

    mov edx, eax
    shr edx, 2

    mov r8d, eax
    and r8d, 3
    shl r8d, 4

    lea r11, [rip + alphabet]

    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r8]
    mov byte ptr [rdi + r14], al
    inc r14

    mov byte ptr [rdi + r14], 61
    inc r14
    mov byte ptr [rdi + r14], 61
    inc r14
    jmp format_output

tail_two:
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi + 1]

    mov edx, eax
    shr edx, 2

    mov r8d, eax
    and r8d, 3
    shl r8d, 4

    mov r9d, ebx
    shr r9d, 4
    or r8d, r9d

    mov r9d, ebx
    and r9d, 15
    shl r9d, 2

    lea r11, [rip + alphabet]

    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r8]
    mov byte ptr [rdi + r14], al
    inc r14

    movzx eax, byte ptr [r11 + r9]
    mov byte ptr [rdi + r14], al
    inc r14

    mov byte ptr [rdi + r14], 61
    inc r14
    jmp format_output

format_output:
    xor r12, r12
    mov r13, r14
    lea rsi, [rip + enc_buf]
    lea rdi, [rip + final_buf]
    xor rbx, rbx

format_loop:
    mov rax, r13
    sub rax, rbx
    cmp rax, 76
    jbe format_last_chunk
    mov rdx, 76
    jmp format_continue

format_last_chunk:
    mov rdx, rax

format_continue:
    lea r8, [rip + enc_buf]
    add r8, rbx
    lea r9, [rip + final_buf]
    add r9, r12
    mov rcx, rdx

copy_chunk:
    test rcx, rcx
    jz copy_done
    mov al, byte ptr [r8]
    mov byte ptr [r9], al
    inc r8
    inc r9
    dec rcx
    jmp copy_chunk

copy_done:
    mov byte ptr [r9], 10
    inc r9
    add rbx, rdx
    mov rax, r12
    add rax, rdx
    add rax, 1
    mov r12, rax
    mov rax, r13
    cmp rbx, rax
    jb format_loop
    jmp write_output

write_output:
    xor r13, r13
    mov rbx, r12
    test rbx, rbx
    jz exit_zero

write_loop:
    cmp r13, rbx
    jae exit_zero
    mov rdx, rbx
    sub rdx, r13
    cmp rdx, 4096
    jbe 1f
    mov rdx, 4096
1:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + final_buf]
    add rsi, r13
    syscall
    cmp rax, 0
    jle fail
    add r13, rax
    jmp write_loop

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

fail:
    mov rax, 60
    mov edi, 1
    syscall
