.globl _start

.text
_start:
    lea rbx, [rip + in_buf]
    lea r13, [rip + in_buf]
    add r13, 49152
    xor r12, r12

read_loop:
    cmp rbx, r13
    jae read_done
    mov rdx, r13
    sub rdx, rbx
    cmp rdx, 4096
    jbe read_count_ok
    mov rdx, 4096

read_count_ok:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    syscall
    cmp rax, 0
    jl read_fail
    je read_done
    add rbx, rax
    add r12, rax
    jmp read_loop

read_done:
    mov qword ptr [in_len], r12

    mov rbx, qword ptr [in_len]
    test rbx, rbx
    jz encode_empty

    lea rcx, [rip + in_buf]
    lea rbp, [rip + alphabet]
    lea r14, [rip + enc_buf]
    xor r12, r12
    xor r13, r13

encode_loop:
    mov rax, r12
    add rax, 3
    cmp rax, qword ptr [in_len]
    jg encode_tail

    movzx r11d, byte ptr [rcx + r12]
    movzx eax, byte ptr [rcx + r12 + 1]
    movzx edx, byte ptr [rcx + r12 + 2]

    mov r8d, r11d
    shr r8d, 2

    mov r9d, r11d
    and r9d, 3
    shl r9d, 4

    mov r10d, eax
    shr r10d, 4
    or r9d, r10d

    mov r15d, eax
    and r15d, 15
    shl r15d, 2

    mov r10d, edx
    shr r10d, 6
    or r15d, r10d

    mov r10d, edx
    and r10d, 63

    movzx r11d, byte ptr [rbp + r8]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r9]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r15]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r10]
    mov byte ptr [r14 + r13], r11b
    inc r13

    add r12, 3
    jmp encode_loop

encode_tail:
    cmp r12, qword ptr [in_len]
    je encode_done

    mov rax, qword ptr [in_len]
    sub rax, r12
    cmp rax, 2
    je tail_two

    movzx r11d, byte ptr [rcx + r12]

    mov r8d, r11d
    shr r8d, 2

    mov r9d, r11d
    and r9d, 3
    shl r9d, 4

    movzx r11d, byte ptr [rbp + r8]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r9]
    mov byte ptr [r14 + r13], r11b
    inc r13

    mov byte ptr [r14 + r13], 61
    inc r13

    mov byte ptr [r14 + r13], 61
    inc r13
    jmp encode_done

tail_two:
    movzx r11d, byte ptr [rcx + r12]
    movzx eax, byte ptr [rcx + r12 + 1]

    mov r8d, r11d
    shr r8d, 2

    mov r9d, r11d
    and r9d, 3
    shl r9d, 4

    mov r10d, eax
    shr r10d, 4
    or r9d, r10d

    mov r15d, eax
    and r15d, 15
    shl r15d, 2

    movzx r11d, byte ptr [rbp + r8]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r9]
    mov byte ptr [r14 + r13], r11b
    inc r13

    movzx r11d, byte ptr [rbp + r15]
    mov byte ptr [r14 + r13], r11b
    inc r13

    mov byte ptr [r14 + r13], 61
    inc r13

encode_done:
    mov qword ptr [enc_len], r13
    jmp format_output

encode_empty:
    mov qword ptr [enc_len], 0

format_output:
    mov qword ptr [final_len], 0
    mov r12, 0
    mov r13, 0
    mov rcx, qword ptr [enc_len]
    test rcx, rcx
    jz write_final

format_loop:
    cmp r12, qword ptr [enc_len]
    je format_done

    mov rax, qword ptr [enc_len]
    sub rax, r12
    cmp rax, 76
    jg format_line_76
    mov rdx, rax
    jmp format_line_done

format_line_76:
    mov rdx, 76

format_line_done:
    lea rsi, [rip + enc_buf]
    add rsi, r12
    lea rdi, [rip + final_buf]
    add rdi, r13
    mov rcx, rdx
    rep movsb

    add r12, rdx
    add r13, rdx

    cmp r12, qword ptr [enc_len]
    je format_done

    mov byte ptr [rdi], 10
    inc r13
    jmp format_loop

format_done:
    mov qword ptr [final_len], r13

write_final:
    mov r12, 0
    mov rax, qword ptr [final_len]
    test rax, rax
    jz exit_zero

write_final_loop:
    mov rdx, qword ptr [final_len]
    sub rdx, r12
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + final_buf]
    add rsi, r12
    syscall
    cmp rax, 0
    jl write_fail
    add r12, rax
    cmp r12, qword ptr [final_len]
    jb write_final_loop

exit_zero:
    mov rax, 60
    xor edi, edi
    syscall

read_fail:
    mov rax, 60
    mov edi, 1
    syscall

write_fail:
    mov rax, 60
    mov edi, 1
    syscall
