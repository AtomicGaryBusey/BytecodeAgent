.text
.globl _start

_start:
    // Read all of stdin into stdin_buf.
    adrp x23, stdin_buf
    add x23, x23, :lo12:stdin_buf
    mov x19, #0
read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x23
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop
read_done:
    cmp x0, #0
    b.lt invalid_exit

    // Find first '!' in stdin.
    mov x20, #0
    mov x21, #-1
find_bang_loop:
    cmp x20, x19
    b.hs no_bang
    add x9, x23, x20
    ldrb w10, [x9]
    cmp w10, #0x21
    b.ne find_bang_next
    mov x21, x20
    b bang_found
find_bang_next:
    add x20, x20, #1
    b find_bang_loop

no_bang:
    mov x22, x19
    mov x24, #0
    mov x25, #0
    b validate_brackets

bang_found:
    mov x22, x20
    add x24, x23, x20
    add x24, x24, #1
    sub x25, x19, x20
    sub x25, x25, #1

validate_brackets:
    adrp x26, match_table
    add x26, x26, :lo12:match_table
    adrp x27, bracket_stack
    add x27, x27, :lo12:bracket_stack
    mov x28, #0
    mov x29, #0
validate_loop:
    cmp x29, x22
    b.hs validate_done
    add x9, x23, x29
    ldrb w10, [x9]
    cmp w10, #0x5b
    b.eq push_bracket
    cmp w10, #0x5d
    b.eq pop_bracket
validate_next:
    add x29, x29, #1
    b validate_loop

push_bracket:
    add x9, x27, x28, lsl #2
    str w29, [x9]
    add x28, x28, #1
    mov x9, #1000
    cmp x28, x9
    b.hi invalid_exit
    b validate_next

pop_bracket:
    cmp x28, #0
    b.eq invalid_exit
    sub x28, x28, #1
    add x9, x27, x28, lsl #2
    ldr w12, [x9]
    add x9, x26, x29, lsl #2
    str w12, [x9]
    add x9, x26, x12, lsl #2
    str w29, [x9]
    b validate_next

validate_done:
    cmp x28, #0
    b.ne invalid_exit

    // Execute BF program.
    mov x19, #0
    mov x20, #0
    mov x21, #0
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, output_buf
    add x28, x28, :lo12:output_buf

run_loop:
    cmp x19, x22
    b.hs run_done
    add x29, x23, x19
    ldrb w9, [x29]

    cmp w9, #0x3e
    b.eq ptr_plus
    cmp w9, #0x3c
    b.eq ptr_minus
    cmp w9, #0x2b
    b.eq cell_plus
    cmp w9, #0x2d
    b.eq cell_minus
    cmp w9, #0x2e
    b.eq output_cell
    cmp w9, #0x2c
    b.eq read_cell
    cmp w9, #0x5b
    b.eq open_bracket
    cmp w9, #0x5d
    b.eq close_bracket

    add x19, x19, #1
    b run_loop

ptr_plus:
    add x21, x21, #1
    mov x9, #30000
    cmp x21, x9
    b.hs invalid_exit
    add x19, x19, #1
    b run_loop

ptr_minus:
    cmp x21, #0
    b.eq invalid_exit
    sub x21, x21, #1
    add x19, x19, #1
    b run_loop

cell_plus:
    add x29, x27, x21
    ldrb w9, [x29]
    add w9, w9, #1
    strb w9, [x29]
    add x19, x19, #1
    b run_loop

cell_minus:
    add x29, x27, x21
    ldrb w9, [x29]
    sub w9, w9, #1
    strb w9, [x29]
    add x19, x19, #1
    b run_loop

output_cell:
    cmp x20, #65536
    b.hs invalid_exit
    add x29, x27, x21
    ldrb w9, [x29]
    add x29, x28, x20
    strb w9, [x29]
    add x20, x20, #1
    add x19, x19, #1
    b run_loop

read_cell:
    cmp x25, #0
    b.eq read_zero
    ldrb w9, [x24], #1
    sub x25, x25, #1
    b read_store
read_zero:
    mov w9, #0
read_store:
    add x29, x27, x21
    strb w9, [x29]
    add x19, x19, #1
    b run_loop

open_bracket:
    add x29, x27, x21
    ldrb w9, [x29]
    cmp w9, #0
    b.eq open_skip
    add x19, x19, #1
    b run_loop
open_skip:
    add x29, x26, x19, lsl #2
    ldr w9, [x29]
    add x19, x9, #1
    b run_loop

close_bracket:
    add x29, x27, x21
    ldrb w9, [x29]
    cmp w9, #0
    b.eq close_skip
    add x29, x26, x19, lsl #2
    ldr w9, [x29]
    add x19, x9, #1
    b run_loop
close_skip:
    add x19, x19, #1
    b run_loop

run_done:
    mov x8, #64
    mov x0, #1
    mov x1, x28
    mov x2, x20
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

invalid_exit:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.align 4
stdin_buf:
    .zero 65536
output_buf:
    .zero 65536
bracket_stack:
    .zero 4000
match_table:
    .zero 262144
tape:
    .zero 30000
