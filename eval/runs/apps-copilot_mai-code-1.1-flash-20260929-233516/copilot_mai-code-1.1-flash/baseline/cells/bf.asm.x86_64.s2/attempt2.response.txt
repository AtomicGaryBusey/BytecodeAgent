.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + total_read], rax

    xor rdx, rdx
    mov rcx, qword ptr [rip + total_read]
    lea r8, [rip + inputbuf]

scan_loop:
    cmp rdx, rcx
    jae no_bang
    movzx r9d, byte ptr [r8 + rdx]
    cmp r9b, 0x21
    je found_bang
    inc rdx
    jmp scan_loop

no_bang:
    mov qword ptr [rip + program_len], rcx
    mov qword ptr [rip + input_offset], rcx
    mov qword ptr [rip + input_len], 0
    jmp after_parse

found_bang:
    mov qword ptr [rip + program_len], rdx
    mov qword ptr [rip + input_offset], rdx
    add qword ptr [rip + input_offset], 1
    mov r9, rcx
    sub r9, rdx
    sub r9, 1
    mov qword ptr [rip + input_len], r9

after_parse:
    call validate_brackets
    test rax, rax
    jnz invalid_exit
    call run_program
    xor rdi, rdi
    mov rax, 60
    syscall

invalid_exit:
    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor r8, r8

read_loop:
    cmp r8, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r8
    lea rsi, [rip + inputbuf]
    add rsi, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    mov rax, r8
    ret

validate_brackets:
    xor rax, rax
    lea rbx, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    xor rsi, rsi
    xor rdi, rdi
    lea r8, [rip + stack]
    lea r9, [rip + match]

validate_loop:
    cmp rsi, rcx
    jae validate_end
    movzx edx, byte ptr [rbx + rsi]
    cmp dl, '['
    je bracket_open
    cmp dl, ']'
    je bracket_close
    inc rsi
    jmp validate_loop

bracket_open:
    cmp rdi, 1000
    jae validate_invalid
    mov word ptr [r8 + rdi*2], si
    inc rdi
    inc rsi
    jmp validate_loop

bracket_close:
    test rdi, rdi
    jz validate_invalid
    dec rdi
    movzx r10d, word ptr [r8 + rdi*2]
    mov word ptr [r9 + r10*2], si
    mov word ptr [r9 + rsi*2], r10w
    inc rsi
    jmp validate_loop

validate_invalid:
    mov rax, 1
    ret

validate_end:
    test rdi, rdi
    jz validate_ok
    mov rax, 1
    ret

validate_ok:
    xor rax, rax
    ret

run_program:
    xor r11, r11
    xor r12, r12
    xor r14, r14
    xor r15, r15
    lea r13, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    mov r9, qword ptr [rip + input_len]
    mov r10, qword ptr [rip + input_offset]

run_loop:
    cmp r14, rcx
    jae run_done

    movzx eax, byte ptr [r13 + r14]

    cmp al, '>'
    jne check_lt
    inc r11
    inc r14
    jmp run_loop

check_lt:
    cmp al, '<'
    jne check_plus
    dec r11
    inc r14
    jmp run_loop

check_plus:
    cmp al, '+'
    jne check_minus
    lea rbx, [rip + tape]
    mov dl, byte ptr [rbx + r11]
    add dl, 1
    mov byte ptr [rbx + r11], dl
    inc r14
    jmp run_loop

check_minus:
    cmp al, '-'
    jne check_dot
    lea rbx, [rip + tape]
    mov dl, byte ptr [rbx + r11]
    sub dl, 1
    mov byte ptr [rbx + r11], dl
    inc r14
    jmp run_loop

check_dot:
    cmp al, '.'
    jne check_comma
    lea rbx, [rip + tape]
    mov dl, byte ptr [rbx + r11]
    lea rsi, [rip + outputbuf]
    mov byte ptr [rsi + r15], dl
    inc r15
    inc r14
    jmp run_loop

check_comma:
    cmp al, ','
    jne check_open
    lea rbx, [rip + tape]
    cmp r12, r9
    jae comma_zero
    mov dl, byte ptr [r13 + r10 + r12]
    mov byte ptr [rbx + r11], dl
    inc r12
    inc r14
    jmp run_loop

comma_zero:
    xor dl, dl
    mov byte ptr [rbx + r11], dl
    inc r14
    jmp run_loop

check_open:
    cmp al, '['
    jne check_close
    lea rbx, [rip + tape]
    mov dl, byte ptr [rbx + r11]
    test dl, dl
    jnz open_continue
    lea rsi, [rip + match]
    movzx r14d, word ptr [rsi + r14*2]
    inc r14
    jmp run_loop

open_continue:
    inc r14
    jmp run_loop

check_close:
    cmp al, ']'
    jne ignore_other
    lea rbx, [rip + tape]
    mov dl, byte ptr [rbx + r11]
    test dl, dl
    jz close_continue
    lea rsi, [rip + match]
    movzx r14d, word ptr [rsi + r14*2]
    inc r14
    jmp run_loop

close_continue:
    inc r14
    jmp run_loop

ignore_other:
    inc r14
    jmp run_loop

run_done:
    mov qword ptr [rip + output_len], r15
    test r15, r15
    jz no_write
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outputbuf]
    mov rdx, r15
    syscall
    jmp after_write

no_write:
    xor rax, rax
    ret

after_write:
    xor rax, rax
    ret

.section .bss
    .align 16
inputbuf:
    .space 65536

    .align 16
tape:
    .space 30000

    .align 16
match:
    .space 131072

    .align 16
outputbuf:
    .space 65536

    .align 16
stack:
    .space 2000

    .align 8
total_read:
    .zero 8

program_len:
    .zero 8

input_offset:
    .zero 8

input_len:
    .zero 8

output_len:
    .zero 8
