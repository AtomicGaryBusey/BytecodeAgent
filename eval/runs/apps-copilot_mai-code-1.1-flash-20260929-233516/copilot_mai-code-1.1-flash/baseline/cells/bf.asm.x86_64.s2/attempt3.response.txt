.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + total_read], rax

    xor rcx, rcx
scan_loop:
    cmp rcx, qword ptr [rip + total_read]
    jae no_bang
    movzx r8d, byte ptr [rip + inputbuf + rcx]
    cmp r8b, 0x21
    je found_bang
    inc rcx
    jmp scan_loop

no_bang:
    mov rax, qword ptr [rip + total_read]
    mov qword ptr [rip + program_len], rax
    mov qword ptr [rip + input_offset], rax
    mov qword ptr [rip + input_len], 0
    jmp after_parse

found_bang:
    mov qword ptr [rip + program_len], rcx
    mov qword ptr [rip + input_offset], rcx
    inc qword ptr [rip + input_offset]
    mov rax, qword ptr [rip + total_read]
    sub rax, rcx
    dec rax
    mov qword ptr [rip + input_len], rax

after_parse:
    call validate_brackets
    test rax, rax
    jne invalid_exit
    call run_program
    xor rdi, rdi
    mov rax, 60
    syscall

invalid_exit:
    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor rbx, rbx

read_loop:
    cmp rbx, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov rax, rbx
    ret

validate_brackets:
    xor rax, rax
    lea rsi, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    xor r8, r8
    lea rbx, [rip + bracket_stack]
    lea r9, [rip + match]

validate_loop:
    cmp r8, 1000
    jae validate_invalid
    cmp rcx, 0
    je validate_end
    movzx edx, byte ptr [rsi]
    cmp dl, '['
    je bracket_open
    cmp dl, ']'
    je bracket_close
    inc rsi
    dec rcx
    jmp validate_loop

bracket_open:
    mov word ptr [rbx + r8*2], 0
    mov r10w, word ptr [rsi]
    ; keep source index in stack as actual byte offset
    ; need real index in rax, so store using a simple helper by scanning PC
    ; easier: rebuild scan with explicit index register
    jmp validate_begin_scan

validate_begin_scan:
    xor rdx, rdx
    mov rdx, qword ptr [rip + program_len]
    lea rsi, [rip + inputbuf]
    xor r10, r10
    xor r8, r8
    lea rbx, [rip + bracket_stack]
    lea r9, [rip + match]
    jmp validate_loop2

validate_loop2:
    cmp r10, rdx
    jae validate_done
    movzx eax, byte ptr [rsi + r10]
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket
    inc r10
    jmp validate_loop2

open_bracket:
    cmp r8, 1000
    jae validate_invalid
    mov word ptr [rbx + r8*2], r10w
    inc r8
    inc r10
    jmp validate_loop2

close_bracket:
    test r8, r8
    jz validate_invalid
    dec r8
    movzx r11d, word ptr [rbx + r8*2]
    mov word ptr [r9 + r11*2], r10w
    mov word ptr [r9 + r10*2], r11w
    inc r10
    jmp validate_loop2

validate_done:
    test r8, r8
    jnz validate_invalid
    xor rax, rax
    ret

validate_invalid:
    mov rax, 1
    ret

validate_end:
    test r8, r8
    jnz validate_invalid
    xor rax, rax
    ret

run_program:
    xor r11, r11
    xor r12, r12
    xor r13, r13
    xor r14, r14
    lea rbx, [rip + tape]
    lea rsi, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    mov r15, qword ptr [rip + input_offset]
    mov r8, qword ptr [rip + input_len]

run_loop:
    cmp r13, rcx
    jae run_done

    movzx eax, byte ptr [rsi + r13]

    cmp al, '>'
    jne check_lt
    inc r11
    inc r13
    jmp run_loop

check_lt:
    cmp al, '<'
    jne check_plus
    dec r11
    inc r13
    jmp run_loop

check_plus:
    cmp al, '+'
    jne check_minus
    mov dl, byte ptr [rbx + r11]
    add dl, 1
    mov byte ptr [rbx + r11], dl
    inc r13
    jmp run_loop

check_minus:
    cmp al, '-'
    jne check_dot
    mov dl, byte ptr [rbx + r11]
    sub dl, 1
    mov byte ptr [rbx + r11], dl
    inc r13
    jmp run_loop

check_dot:
    cmp al, '.'
    jne check_comma
    mov r9b, byte ptr [rbx + r11]
    lea rdx, [rip + outputbuf]
    mov byte ptr [rdx + r14], r9b
    inc r14
    inc r13
    jmp run_loop

check_comma:
    cmp al, ','
    jne check_open
    cmp r12, r8
    jae comma_zero
    lea r9, [rip + inputbuf]
    lea r10, [r9 + r15]
    mov dl, byte ptr [r10 + r12]
    mov byte ptr [rbx + r11], dl
    inc r12
    inc r13
    jmp run_loop

comma_zero:
    xor dl, dl
    mov byte ptr [rbx + r11], dl
    inc r13
    jmp run_loop

check_open:
    cmp al, '['
    jne check_close
    mov dl, byte ptr [rbx + r11]
    test dl, dl
    jnz open_continue
    lea r9, [rip + match]
    movzx r10d, word ptr [r9 + r13*2]
    inc r10
    mov r13, r10
    jmp run_loop

open_continue:
    inc r13
    jmp run_loop

check_close:
    cmp al, ']'
    jne ignore_other
    mov dl, byte ptr [rbx + r11]
    test dl, dl
    jz close_continue
    lea r9, [rip + match]
    movzx r10d, word ptr [r9 + r13*2]
    inc r10
    mov r13, r10
    jmp run_loop

close_continue:
    inc r13
    jmp run_loop

ignore_other:
    inc r13
    jmp run_loop

run_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outputbuf]
    mov rdx, r14
    syscall
    xor rdi, rdi
    mov rax, 60
    syscall

.section .bss
    .align 16
inputbuf:
    .space 65536

    .align 16
tape:
    .space 30000

    .align 16
match:
    .space 131072

    .align 16
outputbuf:
    .space 65536

    .align 16
bracket_stack:
    .space 2000

    .align 8
total_read:
    .zero 8

program_len:
    .zero 8

input_offset:
    .zero 8

input_len:
    .zero 8
