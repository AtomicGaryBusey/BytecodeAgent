.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + total_read], rax

    xor rcx, rcx
    mov rdx, qword ptr [rip + total_read]
    test rdx, rdx
    jz parse_done

scan_bang:
    cmp rcx, rdx
    jae no_bang
    movzx eax, byte ptr [rip + inputbuf + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp scan_bang

no_bang:
    mov qword ptr [rip + program_len], rdx
    mov qword ptr [rip + input_offset], rdx
    mov qword ptr [rip + input_len], 0
    jmp parse_done

found_bang:
    mov qword ptr [rip + program_len], rcx
    mov qword ptr [rip + input_offset], rcx
    inc qword ptr [rip + input_offset]
    mov rax, rdx
    sub rax, rcx
    dec rax
    mov qword ptr [rip + input_len], rax

parse_done:
    call validate_brackets
    test rax, rax
    jne invalid_exit
    call run_program
    xor rdi, rdi
    mov rax, 60
    syscall

invalid_exit:
    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor rbx, rbx

read_loop:
    cmp rbx, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov rax, rbx
    ret

validate_brackets:
    xor r12, r12
    xor r13, r13
    lea rdx, [rip + inputbuf]
    lea rbx, [rip + match]
    lea rsi, [rip + bracket_stack]
    mov rcx, qword ptr [rip + program_len]

validate_loop:
    cmp r13, rcx
    jae validate_done
    movzx eax, byte ptr [rdx + r13]
    cmp al, '['
    je bracket_open
    cmp al, ']'
    je bracket_close
    inc r13
    jmp validate_loop

bracket_open:
    cmp r12, 1000
    jae validate_invalid
    mov word ptr [rsi + r12*2], r13w
    inc r12
    inc r13
    jmp validate_loop

bracket_close:
    test r12, r12
    jz validate_invalid
    dec r12
    movzx r9d, word ptr [rsi + r12*2]
    mov word ptr [rbx + r9*2], r13w
    mov word ptr [rbx + r13*2], r9w
    inc r13
    jmp validate_loop

validate_done:
    test r12, r12
    jnz validate_invalid
    xor rax, rax
    ret

validate_invalid:
    mov rax, 1
    ret

run_program:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    lea rbx, [rip + tape]
    lea rdx, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    mov r8, qword ptr [rip + input_offset]
    lea r9, [rip + inputbuf]
    add r9, r8
    mov r15, 0
    lea r10, [rip + outputbuf]
    lea r11, [rip + match]

run_loop:
    cmp r13, rcx
    jae run_done

    movzx eax, byte ptr [rdx + r13]

    cmp al, '>'
    jne check_lt
    inc r12
    inc r13
    jmp run_loop

check_lt:
    cmp al, '<'
    jne check_plus
    dec r12
    inc r13
    jmp run_loop

check_plus:
    cmp al, '+'
    jne check_minus
    mov r11b, byte ptr [rbx + r12]
    add r11b, 1
    mov byte ptr [rbx + r12], r11b
    inc r13
    jmp run_loop

check_minus:
    cmp al, '-'
    jne check_dot
    mov r11b, byte ptr [rbx + r12]
    sub r11b, 1
    mov byte ptr [rbx + r12], r11b
    inc r13
    jmp run_loop

check_dot:
    cmp al, '.'
    jne check_comma
    mov r11b, byte ptr [rbx + r12]
    mov byte ptr [r10 + r14], r11b
    inc r14
    inc r13
    jmp run_loop

check_comma:
    cmp al, ','
    jne check_open
    mov rax, r15
    cmp rax, qword ptr [rip + input_len]
    jae comma_zero
    movzx eax, byte ptr [r9 + r15]
    mov byte ptr [rbx + r12], al
    inc r15
    inc r13
    jmp run_loop

comma_zero:
    mov byte ptr [rbx + r12], 0
    inc r13
    jmp run_loop

check_open:
    cmp al, '['
    jne check_close
    movzx eax, byte ptr [rbx + r12]
    test al, al
    jnz open_continue
    movzx eax, word ptr [r11 + r13*2]
    inc rax
    mov r13, rax
    jmp run_loop

open_continue:
    inc r13
    jmp run_loop

check_close:
    cmp al, ']'
    jne ignore_other
    movzx eax, byte ptr [rbx + r12]
    test al, al
    jz close_continue
    movzx eax, word ptr [r11 + r13*2]
    inc rax
    mov r13, rax
    jmp run_loop

close_continue:
    inc r13
    jmp run_loop

ignore_other:
    inc r13
    jmp run_loop

run_done:
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, r14
    syscall
    xor rdi, rdi
    mov rax, 60
    syscall

.section .bss
    .align 16
inputbuf:
    .space 65536

    .align 16
tape:
    .space 30000

    .align 16
match:
    .space 131072

    .align 16
bracket_stack:
    .space 2000

    .align 16
outputbuf:
    .space 65536

    .align 8
total_read:
    .zero 8

program_len:
    .zero 8

input_offset:
    .zero 8

input_len:
    .zero 8
