.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + total_read], rax

    xor rax, rax
    lea rsi, [rip + inputbuf]
    mov rcx, qword ptr [rip + total_read]
    xor rdx, rdx

scan_bang:
    cmp rdx, rcx
    jae no_bang
    movzx r8d, byte ptr [rsi + rdx]
    cmp r8b, 0x21
    je found_bang
    inc rdx
    jmp scan_bang

no_bang:
    mov qword ptr [rip + program_len], rcx
    mov qword ptr [rip + input_offset], rcx
    mov qword ptr [rip + input_len], 0
    jmp after_parse

found_bang:
    mov qword ptr [rip + program_len], rdx
    mov qword ptr [rip + input_offset], rdx
    add qword ptr [rip + input_offset], 1
    mov r8, rcx
    sub r8, rdx
    sub r8, 1
    mov qword ptr [rip + input_len], r8

after_parse:
    call validate_brackets
    test rax, rax
    jz run_ok
    mov rdi, 1
    jmp exit_now

run_ok:
    call run_program
    xor rdi, rdi

exit_now:
    mov rax, 60
    syscall

read_all:
    xor r9, r9

read_loop:
    cmp r9, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r9
    lea rsi, [rip + inputbuf]
    add rsi, r9
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop

read_done:
    mov rax, r9
    ret

validate_brackets:
    xor rax, rax
    lea rbx, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    xor rsi, rsi
    xor rdi, rdi
    lea r8, [rip + stack]
    lea r9, [rip + match]

validate_loop:
    cmp rsi, rcx
    jae validate_end
    movzx edx, byte ptr [rbx + rsi]
    cmp dl, '['
    je bracket_open
    cmp dl, ']'
    je bracket_close
    inc rsi
    jmp validate_loop

bracket_open:
    cmp rdi, 1000
    jae validate_invalid
    mov word ptr [r8 + rdi*2], si
    inc rdi
    inc rsi
    jmp validate_loop

bracket_close:
    test rdi, rdi
    jz validate_invalid
    dec rdi
    movzx r10d, word ptr [r8 + rdi*2]
    mov word ptr [r9 + r10*2], si
    mov word ptr [r9 + rsi*2], r10w
    inc rsi
    jmp validate_loop

validate_invalid:
    mov rax, 1
    ret

validate_end:
    test rdi, rdi
    jz validate_ok
    mov rax, 1
    ret

validate_ok:
    xor rax, rax
    ret

run_program:
    lea r15, [rip + inputbuf]
    mov rcx, qword ptr [rip + program_len]
    xor rdx, rdx
    xor r14, r14
    mov rdi, qword ptr [rip + input_len]
    mov r8, qword ptr [rip + input_offset]
    xor r9, r9
    xor r10, r10

run_loop:
    cmp r10, rcx
    jae run_output

    movzx eax, byte ptr [r15 + r10]

    cmp al, '>'
    jne check_lt
    inc rdx
    inc r10
    jmp run_loop

check_lt:
    cmp al, '<'
    jne check_plus
    dec rdx
    inc r10
    jmp run_loop

check_plus:
    cmp al, '+'
    jne check_minus
    lea r11, [rip + tape]
    mov bl, byte ptr [r11 + rdx]
    add bl, 1
    mov byte ptr [r11 + rdx], bl
    inc r10
    jmp run_loop

check_minus:
    cmp al, '-'
    jne check_dot
    lea r11, [rip + tape]
    mov bl, byte ptr [r11 + rdx]
    sub bl, 1
    mov byte ptr [r11 + rdx], bl
    inc r10
    jmp run_loop

check_dot:
    cmp al, '.'
    jne check_comma
    lea r11, [rip + tape]
    mov bl, byte ptr [r11 + rdx]
    lea r12, [rip + outputbuf]
    mov byte ptr [r12 + r9], bl
    inc r9
    inc r10
    jmp run_loop

check_comma:
    cmp al, ','
    jne check_open
    lea r11, [rip + tape]
    cmp r14, rdi
    jae comma_zero
    lea r12, [rip + inputbuf]
    mov bl, byte ptr [r12 + r8 + r14]
    mov byte ptr [r11 + rdx], bl
    inc r14
    inc r10
    jmp run_loop

comma_zero:
    xor bl, bl
    mov byte ptr [r11 + rdx], bl
    inc r10
    jmp run_loop

check_open:
    cmp al, '['
    jne check_close
    lea r11, [rip + tape]
    mov bl, byte ptr [r11 + rdx]
    test bl, bl
    jnz open_continue
    lea r12, [rip + match]
    movzx r10d, word ptr [r12 + r10*2]
    inc r10
    jmp run_loop

open_continue:
    inc r10
    jmp run_loop

check_close:
    cmp al, ']'
    jne ignore_other
    lea r11, [rip + tape]
    mov bl, byte ptr [r11 + rdx]
    test bl, bl
    jz close_continue
    lea r12, [rip + match]
    movzx r10d, word ptr [r12 + r10*2]
    inc r10
    jmp run_loop

close_continue:
    inc r10
    jmp run_loop

ignore_other:
    inc r10
    jmp run_loop

run_output:
    mov rax, 1
    lea rsi, [rip + outputbuf]
    mov rdx, r9
    mov rdi, 1
    syscall
    xor rax, rax
    ret

.section .bss
    .align 16
inputbuf:
    .space 65536

    .align 16
tape:
    .space 30000

    .align 16
match:
    .space 131072

    .align 16
outputbuf:
    .space 65536

    .align 16
stack:
    .space 2000

    .align 8
total_read:
    .quad 0

program_len:
    .quad 0

input_offset:
    .quad 0

input_len:
    .quad 0
