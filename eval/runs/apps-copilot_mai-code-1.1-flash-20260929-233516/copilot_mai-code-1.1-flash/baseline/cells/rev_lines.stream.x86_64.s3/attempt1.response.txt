.globl _start

.text
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    js exit_error
    jz read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop
read_done:
    mov r13, rbx
    test r13, r13
    jz exit_zero
    xor ebp, ebp
    xor r14d, r14d
line_loop:
    cmp r14, r13
    jae write_output
    mov rcx, r14
scan_newline:
    cmp rcx, r13
    je final_line
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x0A
    je line_found
    inc rcx
    jmp scan_newline

line_found:
    mov rdx, rcx
    sub rdx, r14
    mov r15, r14
    add r15, rdx
    dec r15
    lea rsi, [rip + outbuf]
    add rsi, rbp
rev_loop:
    cmp r15, r14
    jb rev_done
    mov al, byte ptr [r12 + r15]
    mov byte ptr [rsi], al
    inc rsi
    dec r15
    jmp rev_loop
rev_done:
    mov byte ptr [rsi], 0x0A
    inc rsi
    add rbp, rdx
    inc rbp
    mov r14, rcx
    inc r14
    jmp line_loop

final_line:
    mov rdx, r13
    sub rdx, r14
    mov r15, r14
    add r15, rdx
    dec r15
    lea rsi, [rip + outbuf]
    add rsi, rbp
final_rev_loop:
    cmp r15, r14
    jb final_rev_done
    mov al, byte ptr [r12 + r15]
    mov byte ptr [rsi], al
    inc rsi
    dec r15
    jmp final_rev_loop
final_rev_done:
    mov byte ptr [rsi], 0x0A
    add rbp, rdx
    inc rbp
    jmp write_output

write_output:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, rbp
    syscall
    jmp exit_zero

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall

exit_error:
    mov edi, 1
    mov eax, 60
    syscall

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65538
