.globl _start

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d

read_loop:
    mov eax, 0
    mov edi, 0
    mov rsi, r12
    add rsi, r13
    mov edx, 65536
    sub edx, r13d
    test edx, edx
    jz read_done
    syscall
    test rax, rax
    js exit_error
    jz read_done
    add r13, rax
    cmp r13, 65536
    jb read_loop

read_done:
    test r13, r13
    jz exit_zero

    xor r14, r14
    xor r15, r15
    lea r10, [rip + outbuf]

process_lines:
    cmp r14, r13
    jae write_output

    mov rbx, r14

scan_line:
    cmp rbx, r13
    je final_line

    mov al, byte ptr [r12 + rbx]
    cmp al, 0x0A
    je newline_found

    inc rbx
    jmp scan_line

newline_found:
    cmp rbx, r14
    je emit_newline

    mov rcx, rbx
    sub rcx, r14
    mov rsi, rbx
    dec rsi

reverse_line_loop:
    mov dl, byte ptr [r12 + rsi]
    mov byte ptr [r10 + r15], dl
    inc r15
    dec rcx
    jz emit_newline_after_reverse
    dec rsi
    jmp reverse_line_loop

emit_newline_after_reverse:
    mov byte ptr [r10 + r15], 0x0A
    inc r15
    mov r14, rbx
    inc r14
    jmp process_lines

emit_newline:
    mov byte ptr [r10 + r15], 0x0A
    inc r15
    mov r14, rbx
    inc r14
    jmp process_lines

final_line:
    mov rcx, r13
    sub rcx, r14
    jz write_output

    mov rsi, r13
    dec rsi

reverse_final_loop:
    mov dl, byte ptr [r12 + rsi]
    mov byte ptr [r10 + r15], dl
    inc r15
    dec rcx
    jz emit_final_newline
    dec rsi
    jmp reverse_final_loop

emit_final_newline:
    mov byte ptr [r10 + r15], 0x0A
    inc r15
    jmp write_output

write_output:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r15
    syscall
    jmp exit_zero

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall

exit_error:
    mov edi, 1
    mov eax, 60
    syscall

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65538
