.globl _start

.text
_start:
    xor ebx, ebx
    lea r12, [rip + inbuf]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    js exit_error
    jz read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop

read_done:
    test rbx, rbx
    jz exit_zero
    xor r13d, r13d
    xor r14d, r14d

line_loop:
    cmp r14, rbx
    jae write_output
    mov r15, r14

scan_newline:
    cmp r15, rbx
    je final_line
    mov al, byte ptr [r12 + r15]
    cmp al, 0x0A
    je line_found
    inc r15
    jmp scan_newline

line_found:
    mov rax, r15
    sub rax, r14
    mov rsi, r15
    dec rsi
    lea rdi, [rip + outbuf]
    add rdi, r13

reverse_line:
    cmp rsi, r14
    jb reverse_done
    mov dl, byte ptr [r12 + rsi]
    mov byte ptr [rdi], dl
    inc rdi
    dec rsi
    jmp reverse_line

reverse_done:
    mov byte ptr [rdi], 0x0A
    add r13, rax
    inc r13
    mov r14, r15
    inc r14
    jmp line_loop

final_line:
    mov rax, rbx
    sub rax, r14
    mov rsi, rbx
    dec rsi
    lea rdi, [rip + outbuf]
    add rdi, r13

reverse_final:
    cmp rsi, r14
    jb final_done
    mov dl, byte ptr [r12 + rsi]
    mov byte ptr [rdi], dl
    inc rdi
    dec rsi
    jmp reverse_final

final_done:
    mov byte ptr [rdi], 0x0A
    add r13, rax
    inc r13
    jmp write_output

write_output:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall
    jmp exit_zero

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall

exit_error:
    mov edi, 1
    mov eax, 60
    syscall

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65538
