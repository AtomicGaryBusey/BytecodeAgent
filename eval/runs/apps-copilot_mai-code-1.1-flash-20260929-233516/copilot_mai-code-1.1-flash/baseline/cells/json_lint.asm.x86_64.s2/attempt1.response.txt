.text
.globl _start

_start:
    lea rdi, [rip + input_buf]
    mov esi, 65536
    call read_all
    mov qword ptr [rip + input_len], rax

    call parse_document
    test rax, rax
    jnz .ok

    mov edi, 1
    mov eax, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    syscall

    sub rsp, 32
    lea r10, [rsp + 31]
    mov rax, r14
    mov rcx, 0
    test rax, rax
    jnz 1f
    mov byte ptr [rsp + 31], '0'
    mov rcx, 1
    mov r10, rsp
    add r10, 31
    jmp 2f
1:
    mov r11, rsp
    add r11, 31
    .digit_loop:
        xor edx, edx
        mov rsi, 10
        div rsi
        add dl, '0'
        mov byte ptr [r11], dl
        dec r11
        inc rcx
        test rax, rax
        jnz .digit_loop
    mov r10, r11
    inc r10
2:
    mov rsi, r10
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall

    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall

    add rsp, 32

    mov eax, 60
    mov edi, 1
    syscall

.ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall

    mov eax, 60
    xor edi, edi
    syscall

read_all:
    xor r8, r8
1:
    cmp r8, rsi
    jae 2f
    mov eax, 0
    mov edi, 0
    lea rsi, [rdi + r8]
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz 2f
    add r8, rax
    jmp 1b
2:
    mov rax, r8
    ret

parse_document:
    xor r14, r14
    call parse_ws
    mov r14, rax
    call parse_value
    cmp rax, -1
    je .fail
    mov r14, rax
    call parse_ws
    mov r14, rax
    mov rax, qword ptr [rip + input_len]
    cmp r14, rax
    jne .fail
    mov rax, 1
    ret
.fail:
    mov rax, 0
    ret

parse_ws:
    .loop:
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        jae .done
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r14]
        cmp al, 0x20
        je .sp
        cmp al, 0x09
        je .sp
        cmp al, 0x0A
        je .sp
        cmp al, 0x0D
        je .sp
        jmp .done
    .sp:
        inc r14
        jmp .loop
    .done:
        mov rax, r14
        ret

parse_value:
    mov rax, qword ptr [rip + input_len]
    cmp r14, rax
    jae .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r14]
    cmp al, 0x7B
    je .object
    cmp al, 0x5B
    je .array
    cmp al, 0x22
    je .string
    cmp al, 0x2D
    je .number
    cmp al, '0'
    jb .fail
    cmp al, '9'
    jbe .number
    cmp al, 't'
    je .true
    cmp al, 'f'
    je .false
    cmp al, 'n'
    je .null
    jmp .fail

.object:
    mov r15, r14
    inc r15
    mov r14, r15
    call parse_ws
    mov r14, rax
    mov rax, qword ptr [rip + input_len]
    cmp r14, rax
    je .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r14]
    cmp al, 0x7D
    je .obj_close
    .member_loop:
        mov r14, rax
        call parse_string
        cmp rax, -1
        je .fail
        mov r14, rax
        call parse_ws
        mov r14, rax
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r14]
        cmp al, 0x3A
        jne .fail
        inc r14
        call parse_ws
        mov r14, rax
        call parse_value
        cmp rax, -1
        je .fail
        mov r14, rax
        call parse_ws
        mov r14, rax
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        je .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r14]
        cmp al, 0x2C
        je .comma
        cmp al, 0x7D
        jne .fail
        inc r14
        mov rax, r14
        ret
    .comma:
        inc r14
        call parse_ws
        mov r14, rax
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        je .fail
        jmp .member_loop
    .obj_close:
        inc r14
        mov rax, r14
        ret

.array:
    mov r15, r14
    inc r15
    mov r14, r15
    call parse_ws
    mov r14, rax
    mov rax, qword ptr [rip + input_len]
    cmp r14, rax
    je .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r14]
    cmp al, 0x5D
    je .arr_close
    .elem_loop:
        mov r14, rax
        call parse_value
        cmp rax, -1
        je .fail
        mov r14, rax
        call parse_ws
        mov r14, rax
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        je .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r14]
        cmp al, 0x2C
        je .comma
        cmp al, 0x5D
        jne .fail
        inc r14
        mov rax, r14
        ret
    .comma:
        inc r14
        call parse_ws
        mov r14, rax
        mov rax, qword ptr [rip + input_len]
        cmp r14, rax
        je .fail
        jmp .elem_loop
    .arr_close:
        inc r14
        mov rax, r14
        ret

.string:
    mov r15, r14
    mov rax, qword ptr [rip + input_len]
    cmp r15, rax
    jae .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, 0x22
    jne .fail
    inc r15
    .loop:
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail_eof
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, 0x22
        je .done
        cmp al, 0x5C
        je .escape
        cmp al, 0x20
        jb .fail
        inc r15
        jmp .loop
    .escape:
        inc r15
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, 0x22
        je .adv
        cmp al, 0x5C
        je .adv
        cmp al, 0x2F
        je .adv
        cmp al, 0x62
        je .adv
        cmp al, 0x66
        je .adv
        cmp al, 0x6E
        je .adv
        cmp al, 0x72
        je .adv
        cmp al, 0x74
        je .adv
        cmp al, 0x75
        jne .fail
        mov r11, r15
        inc r11
        mov ecx, 4
    .hexloop:
        mov rax, qword ptr [rip + input_len]
        cmp r11, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r11]
        cmp al, '0'
        jb .fail
        cmp al, '9'
        jbe .hexok
        cmp al, 'a'
        jb .fail
        cmp al, 'f'
        jbe .hexok
        cmp al, 'A'
        jb .fail
        cmp al, 'F'
        jbe .hexok
        jmp .fail
    .hexok:
        inc r11
        dec ecx
        jne .hexloop
        mov r15, r11
        jmp .loop
    .adv:
        inc r15
        jmp .loop
    .done:
        inc r15
        mov r14, r15
        mov rax, r15
        ret
    .fail_eof:
        mov r14, r15
        mov rax, -1
        ret
    .fail:
        mov r14, r15
        mov rax, -1
        ret

.number:
    mov r15, r14
    mov rax, qword ptr [rip + input_len]
    cmp r15, rax
    jae .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, 0x2D
    jne .after_sign
    inc r15
    mov rax, qword ptr [rip + input_len]
    cmp r15, rax
    jae .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    jmp .mainnum
.after_sign:
    cmp al, '0'
    jne .mainnum
.mainnum:
    lea rax, [r15 + 1]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    ja .fail
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, '0'
    je .int_zero
    cmp al, '1'
    jb .fail
    cmp al, '9'
    ja .fail
    inc r15
    .digits_loop:
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .frac_or_exp
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, '0'
        jb .frac_or_exp
        cmp al, '9'
        ja .frac_or_exp
        inc r15
        jmp .digits_loop
    .int_zero:
        inc r15
    .frac_or_exp:
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .done
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, 0x2E
        jne .check_exp
        inc r15
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, '0'
        jb .fail
        cmp al, '9'
        ja .fail
        inc r15
        .frac_digits:
            mov rax, qword ptr [rip + input_len]
            cmp r15, rax
            jae .check_exp
            lea rbx, [rip + input_buf]
            mov al, byte ptr [rbx + r15]
            cmp al, '0'
            jb .check_exp
            cmp al, '9'
            ja .check_exp
            inc r15
            jmp .frac_digits
    .check_exp:
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .done
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, 'e'
        je .exp
        cmp al, 'E'
        je .exp
        jmp .done
    .exp:
        inc r15
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, 0x2B
        je .exp_sign
        cmp al, 0x2D
        je .exp_sign
    .exp_digits:
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail
        lea rbx, [rip + input_buf]
        mov al, byte ptr [rbx + r15]
        cmp al, '0'
        jb .fail
        cmp al, '9'
        ja .fail
        inc r15
        .exp_digits_loop:
            mov rax, qword ptr [rip + input_len]
            cmp r15, rax
            jae .done
            lea rbx, [rip + input_buf]
            mov al, byte ptr [rbx + r15]
            cmp al, '0'
            jb .done
            cmp al, '9'
            ja .done
            inc r15
            jmp .exp_digits_loop
    .exp_sign:
        inc r15
        mov rax, qword ptr [rip + input_len]
        cmp r15, rax
        jae .fail
        jmp .exp_digits
    .done:
        mov r14, r15
        mov rax, r15
        ret
    .fail:
        mov r14, r15
        mov rax, -1
        ret

.true:
    mov r15, r14
    lea rax, [r15 + 4]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    ja .fail_literal
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, 't'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'r'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'u'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'e'
    jne .fail_literal
    inc r15
    mov r14, r15
    mov rax, r15
    ret
.fail_literal:
    mov r14, r15
    mov rax, -1
    ret

.false:
    mov r15, r14
    lea rax, [r15 + 5]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    ja .fail_literal
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, 'f'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'a'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'l'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 's'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'e'
    jne .fail_literal
    inc r15
    mov r14, r15
    mov rax, r15
    ret

.null:
    mov r15, r14
    lea rax, [r15 + 4]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    ja .fail_literal
    lea rbx, [rip + input_buf]
    mov al, byte ptr [rbx + r15]
    cmp al, 'n'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'u'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'l'
    jne .fail_literal
    inc r15
    mov al, byte ptr [rbx + r15]
    cmp al, 'l'
    jne .fail_literal
    inc r15
    mov r14, r15
    mov rax, r15
    ret

.section .rodata
ok_msg:
    .asciz "ok\n"
err_prefix:
    .asciz "error at "
newline:
    .byte 10

.section .bss
    .p2align 4
input_buf:
    .space 65536
input_len:
    .quad 0
