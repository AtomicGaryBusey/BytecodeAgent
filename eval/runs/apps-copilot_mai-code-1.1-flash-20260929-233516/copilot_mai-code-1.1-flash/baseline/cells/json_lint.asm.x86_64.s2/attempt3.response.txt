.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + pos], 0
    call parse_document
    test rax, rax
    jnz .ok
    call write_error
    mov rax, 60
    mov rdi, 1
    syscall

.ok:
    call write_ok
    mov rax, 60
    xor edi, edi
    syscall

read_all:
    xor r12, r12
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz .done
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.done:
    mov qword ptr [rip + input_len], r12
    ret

write_ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    ret

write_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    syscall
    mov rax, qword ptr [rip + error_pos]
    call print_decimal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    ret

print_decimal:
    sub rsp, 32
    lea r13, [rsp + 31]
    mov rcx, 0
    mov r12, rax
.print_loop:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    mov byte ptr [r13], dl
    dec r13
    inc rcx
    test rax, rax
    jnz .print_loop
    inc r13
    mov rsi, r13
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    syscall
    add rsp, 32
    ret

fail_here:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    xor rax, rax
    ret

parse_document:
    call parse_ws
    call parse_value
    test rax, rax
    jz .fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne .fail
    mov rax, 1
    ret
.fail:
    call fail_here
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x20
    je .advance
    cmp al, 0x09
    je .advance
    cmp al, 0x0A
    je .advance
    cmp al, 0x0D
    je .advance
    ret
.advance:
    inc qword ptr [rip + pos]
    jmp parse_ws
.done:
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_value_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb .literal
    cmp al, '9'
    jbe parse_number
.literal:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp parse_value_fail

parse_value_fail:
    call fail_here
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '{'
    jne parse_object_fail
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '}'
    je .close
.loop:
    call parse_string
    test rax, rax
    jz .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ':'
    jne parse_object_fail
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    test rax, rax
    jz .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ','
    je .comma
    cmp al, '}'
    jne parse_object_fail
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_object_fail
    jmp .loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.done:
    ret
parse_object_fail:
    call fail_here
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '['
    jne parse_array_fail
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ']'
    je .close
.loop:
    call parse_value
    test rax, rax
    jz .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ','
    je .comma
    cmp al, ']'
    jne parse_array_fail
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_array_fail
    jmp .loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.done:
    ret
parse_array_fail:
    call fail_here
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    jne parse_string_fail
    inc qword ptr [rip + pos]
.loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    je .done
    cmp al, '\\'
    je .escape
    cmp al, 0x20
    jb parse_string_fail
    inc qword ptr [rip + pos]
    jmp .loop
.escape:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    je .advance
    cmp al, '\\'
    je .advance
    cmp al, '/'
    je .advance
    cmp al, 'b'
    je .advance
    cmp al, 'f'
    je .advance
    cmp al, 'n'
    je .advance
    cmp al, 'r'
    je .advance
    cmp al, 't'
    je .advance
    cmp al, 'u'
    jne parse_string_fail
    mov rdx, qword ptr [rip + pos]
    inc rdx
    mov rcx, 4
.hex_loop:
    cmp rdx, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rdx]
    cmp al, '0'
    jb parse_string_fail
    cmp al, '9'
    jbe .hex_ok
    cmp al, 'a'
    jb parse_string_fail
    cmp al, 'f'
    jbe .hex_ok
    cmp al, 'A'
    jb parse_string_fail
    cmp al, 'F'
    jbe .hex_ok
    jmp parse_string_fail
.hex_ok:
    inc rdx
    dec rcx
    jne .hex_loop
    mov qword ptr [rip + pos], rdx
    jmp .loop
.advance:
    inc qword ptr [rip + pos]
    jmp .loop
.done:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
parse_string_fail:
    call fail_here
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '-'
    je .minus
    cmp al, '0'
    je .zero
    cmp al, '1'
    jb parse_number_fail
    cmp al, '9'
    jbe .digits
    jmp parse_number_fail
.minus:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    je .zero
    cmp al, '1'
    jb parse_number_fail
    cmp al, '9'
    jbe .digits
    jmp parse_number_fail
.zero:
    inc qword ptr [rip + pos]
.after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '.'
    je .frac
    cmp al, 'e'
    je .exp
    cmp al, 'E'
    je .exp
.done:
    mov rax, 1
    ret
.digits:
    inc qword ptr [rip + pos]
.digits_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_int
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb .after_int
    cmp al, '9'
    ja .after_int
    inc qword ptr [rip + pos]
    jmp .digits_loop
.frac:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    inc qword ptr [rip + pos]
.frac_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_frac
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb .after_frac
    cmp al, '9'
    ja .after_frac
    inc qword ptr [rip + pos]
    jmp .frac_loop
.after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 'e'
    je .exp
    cmp al, 'E'
    je .exp
    mov rax, 1
    ret
.exp:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '+'
    je .exp_sign
    cmp al, '-'
    je .exp_sign
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    inc qword ptr [rip + pos]
    jmp .exp_digits
.exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    inc qword ptr [rip + pos]
.exp_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb .done
    cmp al, '9'
    ja .done
    inc qword ptr [rip + pos]
    jmp .exp_digits
parse_number_fail:
    call fail_here
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_literal_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 't'
    jne parse_literal_fail
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_literal_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 'r'
    jne parse_literal_fail
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_literal_fail

    movzx eax, byte ptr [rbx + rax]

    cmp rax, qword ptr [rip + input_len]
    jae parse_literal_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 'e'
    jne parse_literal_fail

    cmp al, 'f'
    jne parse_literal_fail
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_literal_fail
    lea rbx, [rip + input_buf]

    ret

.section .rodata
ok_msg:
    .asciz "ok\n"
err_prefix:
    .asciz "error at "
newline:
    .byte 10

.section .bss
    .p2align 4
input_buf:
    .space 65536
    .p2align 3
input_len:
    .space 8
pos:
    .space 8
error_pos:
    .space 8
