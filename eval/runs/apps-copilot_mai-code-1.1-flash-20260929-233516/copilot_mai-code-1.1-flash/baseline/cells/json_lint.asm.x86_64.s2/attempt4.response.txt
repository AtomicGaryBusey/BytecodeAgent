.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + pos], 0
    call parse_document
    test rax, rax
    jnz ok
    call write_error
    mov rax, 60
    mov rdi, 1
    syscall
ok:
    call write_ok
    mov rax, 60
    xor edi, edi
    syscall

read_all:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    mov qword ptr [rip + input_len], r12
    ret

write_ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    ret

write_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    syscall
    mov rax, qword ptr [rip + error_pos]
    call print_decimal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    ret

print_decimal:
    sub rsp, 32
    lea r13, [rsp + 31]
    mov rbx, 10
    mov r12, rax
    xor ecx, ecx
print_loop:
    xor edx, edx
    mov rax, r12
    div rbx
    mov r12, rax
    add dl, '0'
    mov byte ptr [r13], dl
    dec r13
    inc ecx
    test r12, r12
    jnz print_loop
    inc r13
    mov rsi, r13
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    syscall
    add rsp, 32
    ret

fail_here:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    xor rax, rax
    ret

parse_document:
    call parse_ws
    call parse_value
    test rax, rax
    jz document_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne document_fail
    mov rax, 1
    ret
document_fail:
    call fail_here
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae ws_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x20
    je ws_advance
    cmp al, 0x09
    je ws_advance
    cmp al, 0x0A
    je ws_advance
    cmp al, 0x0D
    je ws_advance
    ret
ws_advance:
    inc qword ptr [rip + pos]
    jmp parse_ws
ws_done:
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae value_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb value_literal
    cmp al, '9'
    jbe parse_number
value_literal:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp value_fail
value_fail:
    call fail_here
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '{'
    jne object_fail
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '}'
    je object_close
object_loop:
    call parse_string
    test rax, rax
    jz object_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ':'
    jne object_fail
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    test rax, rax
    jz object_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ','
    je object_comma
    cmp al, '}'
    jne object_fail
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
object_comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    jmp object_loop
object_close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
object_fail:
    call fail_here
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '['
    jne array_fail
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ']'
    je array_close
array_loop:
    call parse_value
    test rax, rax
    jz array_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, ','
    je array_comma
    cmp al, ']'
    jne array_fail
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
array_comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    jmp array_loop
array_close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
array_fail:
    call fail_here
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    jne string_fail
    inc qword ptr [rip + pos]
string_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    je string_done
    cmp al, '\\'
    je string_escape
    cmp al, 0x20
    jb string_fail
    inc qword ptr [rip + pos]
    jmp string_loop
string_escape:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '"'
    je string_escape_ok
    cmp al, '\\'
    je string_escape_ok
    cmp al, '/'
    je string_escape_ok
    cmp al, 'b'
    je string_escape_ok
    cmp al, 'f'
    je string_escape_ok
    cmp al, 'n'
    je string_escape_ok
    cmp al, 'r'
    je string_escape_ok
    cmp al, 't'
    je string_escape_ok
    cmp al, 'u'
    je string_unicode
    jmp string_fail
string_escape_ok:
    inc qword ptr [rip + pos]
    jmp string_loop
string_unicode:
    mov rdx, qword ptr [rip + pos]
    inc rdx
    xor ecx, ecx
string_hex_loop:
    cmp rdx, qword ptr [rip + input_len]
    jae string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rdx]
    cmp al, '0'
    jb string_fail
    cmp al, '9'
    jbe string_hex_ok
    cmp al, 'a'
    jb string_fail
    cmp al, 'f'
    jbe string_hex_ok
    cmp al, 'A'
    jb string_fail
    cmp al, 'F'
    jbe string_hex_ok
    jmp string_fail
string_hex_ok:
    inc rdx
    inc ecx
    cmp ecx, 4
    jb string_hex_loop
    mov qword ptr [rip + pos], rdx
    jmp string_loop
string_done:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
string_fail:
    call fail_here
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '-'
    je number_minus
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb number_fail
    cmp al, '9'
    jbe number_digits
    jmp number_fail
number_minus:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb number_fail
    cmp al, '9'
    jbe number_digits
    jmp number_fail
number_zero:
    inc qword ptr [rip + pos]
    jmp number_after_int
number_digits:
    inc qword ptr [rip + pos]
number_digits_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_after_int
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb number_after_int
    cmp al, '9'
    ja number_after_int
    inc qword ptr [rip + pos]
    jmp number_digits_loop
number_after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '.'
    je number_frac
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    je number_exp
number_done:
    mov rax, 1
    ret
number_frac:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb number_fail
    cmp al, '9'
    ja number_fail
    inc qword ptr [rip + pos]
number_frac_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_after_frac
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb number_after_frac
    cmp al, '9'
    ja number_after_frac
    inc qword ptr [rip + pos]
    jmp number_frac_loop
number_after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    je number_exp
    mov rax, 1
    ret
number_exp:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    je number_exp_sign
    cmp al, '0'
    jb number_fail
    cmp al, '9'
    ja number_fail
    inc qword ptr [rip + pos]
    jmp number_exp_digits
number_exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb number_fail
    cmp al, '9'
    ja number_fail
    inc qword ptr [rip + pos]
number_exp_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    ja number_done
    inc qword ptr [rip + pos]
    jmp number_exp_digits
number_fail:
    call fail_here
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_true_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 't'
