.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + pos], 0
    mov qword ptr [rip + status], 1
    mov qword ptr [rip + error_pos], 0
    call parse_document
    mov rax, qword ptr [rip + status]
    test rax, rax
    jnz _ok
    mov rax, qword ptr [rip + error_pos]
    mov r12, rax
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    syscall
    mov rax, r12
    call print_decimal_and_newline
    mov eax, 60
    mov edi, 1
    syscall
_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

read_all:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    mov qword ptr [rip + input_len], r12
    mov rax, r12
    ret

print_decimal_and_newline:
    sub rsp, 32
    lea r13, [rsp + 31]
    mov rcx, 0
pd_loop:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    mov byte ptr [r13], dl
    dec r13
    inc rcx
    test rax, rax
    jnz pd_loop
    inc r13
    mov rsi, r13
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    add rsp, 32
    ret

parse_document:
    call parse_ws
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_document_exit
    call parse_value
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_document_exit
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne parse_fail
    mov qword ptr [rip + status], 1
    ret
parse_fail:
    mov qword ptr [rip + status], 0
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    ret
parse_document_exit:
    ret

parse_ws:
ws_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae ws_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x20
    je ws_advance
    cmp al, 0x09
    je ws_advance
    cmp al, 0x0A
    je ws_advance
    cmp al, 0x0D
    je ws_advance
    jmp ws_done
ws_advance:
    inc qword ptr [rip + pos]
    jmp ws_loop
ws_done:
    ret

parse_value:
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_value_done
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    cmp al, '0'
    jb parse_literal_start
    cmp al, '9'
    jbe parse_number
parse_literal_start:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp parse_invalid_here
parse_value_done:
    ret

parse_invalid_here:
    mov qword ptr [rip + status], 0
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x7B
    jne parse_invalid_here
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x7D
    je obj_close
obj_member_loop:
    call parse_string
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_object_done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x3A
    jne parse_invalid_here
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_object_done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x2C
    je obj_comma
    cmp al, 0x7D
    jne parse_invalid_here
    inc qword ptr [rip + pos]
    ret
obj_comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    jmp obj_member_loop
obj_close:
    inc qword ptr [rip + pos]
    ret
parse_object_done:
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x5B
    jne parse_invalid_here
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x5D
    je arr_close
arr_value_loop:
    call parse_value
    mov rax, qword ptr [rip + status]
    test rax, rax
    jz parse_array_done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x2C
    je arr_comma
    cmp al, 0x5D
    jne parse_invalid_here
    inc qword ptr [rip + pos]
    ret
arr_comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_invalid_here
    jmp arr_value_loop
arr_close:
    inc qword ptr [rip + pos]
    ret
parse_array_done:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x22
    jne parse_string_fail
    inc qword ptr [rip + pos]
str_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail_eof
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x22
    je str_done
    cmp al, 0x5C
    je str_escape
    cmp al, 0x20
    jb parse_string_fail
    inc qword ptr [rip + pos]
    jmp str_loop
str_escape:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x22
    je str_escape_advance
    cmp al, 0x5C
    je str_escape_advance
    cmp al, 0x2F
    je str_escape_advance
    cmp al, 'b'
    je str_escape_advance
    cmp al, 'f'
    je str_escape_advance
    cmp al, 'n'
    je str_escape_advance
    cmp al, 'r'
    je str_escape_advance
    cmp al, 't'
    je str_escape_advance
    cmp al, 'u'
    jne parse_string_fail
    mov rdx, qword ptr [rip + pos]
    inc rdx
    mov rcx, 4
str_hex_loop:
    cmp rdx, qword ptr [rip + input_len]
    jae parse_string_fail
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rdx]
    cmp al, '0'
    jb parse_string_fail
    cmp al, '9'
    jbe str_hex_ok
    cmp al, 'a'
    jb parse_string_fail
    cmp al, 'f'
    jbe str_hex_ok
    cmp al, 'A'
    jb parse_string_fail
    cmp al, 'F'
    jbe str_hex_ok
    jmp parse_string_fail
str_hex_ok:
    inc rdx
    dec rcx
    jne str_hex_loop
    mov qword ptr [rip + pos], rdx
    jmp str_loop
str_escape_advance:
    inc qword ptr [rip + pos]
    jmp str_loop
str_done:
    inc qword ptr [rip + pos]
    ret
parse_string_fail_eof:
    mov qword ptr [rip + status], 0
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    ret
parse_string_fail:
    mov qword ptr [rip + status], 0
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + error_pos], rax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x2D
    je num_minus
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb parse_invalid_here
    cmp al, '9'
    jbe num_digits
    jmp parse_invalid_here

num_minus:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb parse_invalid_here
    cmp al, '9'
    jbe num_digits
    jmp parse_invalid_here

num_zero:
    inc qword ptr [rip + pos]
num_after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je num_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x2E
    je num_frac
    cmp al, 'e'
    je num_exp
    cmp al, 'E'
    je num_exp
num_done:
    ret

num_digits:
    inc qword ptr [rip + pos]
num_digits_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae num_after_int
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb num_after_int
    cmp al, '9'
    ja num_after_int
    inc qword ptr [rip + pos]
    jmp num_digits_loop

num_frac:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb parse_invalid_here
    cmp al, '9'
    ja parse_invalid_here
    inc qword ptr [rip + pos]
num_frac_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae num_after_frac
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb num_after_frac
    cmp al, '9'
    ja num_after_frac
    inc qword ptr [rip + pos]
    jmp num_frac_digits
num_after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je num_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 'e'
    je num_exp
    cmp al, 'E'
    je num_exp
    ret

num_exp:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, 0x2B
    je num_exp_sign
    cmp al, 0x2D
    je num_exp_sign
    cmp al, '0'
    jb parse_invalid_here
    cmp al, '9'
    ja parse_invalid_here
    inc qword ptr [rip + pos]
num_exp_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae num_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    ja num_done
    inc qword ptr [rip + pos]
    jmp num_exp_digits
num_exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    cmp al, '0'
    jb parse_invalid_here
    cmp al, '9'
    ja parse_invalid_here
    inc qword ptr [rip + pos]
    jmp num_exp_digits

parse_true:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_invalid_here
