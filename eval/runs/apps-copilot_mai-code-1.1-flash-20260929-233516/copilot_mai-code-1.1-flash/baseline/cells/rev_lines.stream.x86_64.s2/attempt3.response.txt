.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 65536
.equ LF, 10

.section .bss
input_buf:
    .zero BUF_SIZE

.section .rodata
newline:
    .byte LF

.text

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
1:
    test r10, r10
    jz 2f
    mov rax, SYS_write
    mov rdi, r8
    mov rsi, r9
    mov rdx, r10
    syscall
    cmp rax, 0
    jl 3f
    cmp rax, r10
    je 2f
    sub r10, rax
    add r9, rax
    jmp 1b
2:
    ret
3:
    mov rax, SYS_exit
    mov rdi, 1
    syscall

emit_line:
    mov r8, rdi
    mov r9, rsi
    mov rax, r9
    sub rax, r8
    jz 1f
    lea r10, [rip + input_buf]
    mov r11, r9
    dec r11
    mov r12, r8
2:
    cmp r12, r11
    jae 1f
    mov dl, byte ptr [r10 + r12]
    mov cl, byte ptr [r10 + r11]
    mov byte ptr [r10 + r12], cl
    mov byte ptr [r10 + r11], dl
    inc r12
    dec r11
    jmp 2b
1:
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, r9
    sub rdx, r8
    call write_all
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all
    ret

_start:
    xor r12, r12
read_more:
    cmp r12, BUF_SIZE
    jae process
    mov rax, SYS_read
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, BUF_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    jl exit_error
    je process
    add r12, rax
    jmp read_more

process:
    test r12, r12
    jz exit_ok
    xor r13, r13
    xor r14, r14
scan_lines:
    cmp r14, r12
    jae final_line
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + r14]
    cmp al, LF
    jne 1f
    mov rdi, r13
    mov rsi, r14
    call emit_line
    inc r14
    mov r13, r14
    jmp scan_lines
1:
    inc r14
    jmp scan_lines

final_line:
    cmp r13, r12
    je exit_ok
    mov rdi, r13
    mov rsi, r12
    call emit_line
    jmp exit_ok

exit_ok:
    mov rax, SYS_exit
    xor rdi, rdi
    syscall

exit_error:
    mov rax, SYS_exit
    mov rdi, 1
    syscall
