.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 65536
.equ LF, 10

.section .bss
    .align 16
input_buf:
    .zero BUF_SIZE

.section .rodata
newline:
    .byte LF

.text

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
1:
    test r10, r10
    jz 2f
    mov rax, SYS_write
    mov rdi, r8
    mov rsi, r9
    mov rdx, r10
    syscall
    cmp rax, 0
    jl 3f
    cmp rax, r10
    je 2f
    sub r10, rax
    add r9, rax
    jmp 1b
2:
    ret
3:
    mov rax, SYS_exit
    mov rdi, 1
    syscall

emit_line:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, r12
    mov r15, r13
    dec r15
1:
    cmp r14, r15
    jae 2f
    lea rbx, [rip + input_buf]
    mov dl, byte ptr [rbx + r14]
    mov cl, byte ptr [rbx + r15]
    mov byte ptr [rbx + r14], cl
    mov byte ptr [rbx + r15], dl
    inc r14
    dec r15
    jmp 1b
2:
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, r13
    sub rdx, r12
    call write_all
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all
    pop r15
    pop r14
    pop r13
    pop r12
    ret

_start:
    mov r12, 0
read_more:
    cmp r12, BUF_SIZE
    jae process
    mov rax, SYS_read
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, BUF_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    jl exit_error
    je process
    add r12, rax
    jmp read_more

process:
    test r12, r12
    jz exit_ok
    mov r13, 0
    mov r14, 0
scan:
    cmp r14, r12
    jae final
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + r14]
    cmp al, LF
    jne 1f
    mov rdi, r13
    mov rsi, r14
    call emit_line
    inc r14
    mov r13, r14
    jmp scan
1:
    inc r14
    jmp scan

final:
    cmp r13, r12
    je exit_ok
    mov rdi, r13
    mov rsi, r12
    call emit_line
    jmp exit_ok

exit_ok:
    mov rax, SYS_exit
    xor rdi, rdi
    syscall

exit_error:
    mov rax, SYS_exit
    mov rdi, 1
    syscall
