.globl _start

.section .bss
input_buf:
    .zero 65536

.section .rodata
newline:
    .byte 0x0A

.text

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
1:
    mov rax, 1
    mov rdi, r8
    mov rsi, r9
    mov rdx, r10
    syscall
    cmp rax, 0
    jl 2f
    cmp rax, r10
    je 2f
    sub r10, rax
    add r9, rax
    jmp 1b
2:
    ret

emit_line:
    mov r8, rdi
    mov r9, rsi
    mov rax, r9
    sub rax, r8
    jz 1f
    lea r10, [rip + input_buf]
    mov r11, r9
    dec r11
    mov r12, r8
2:
    cmp r12, r11
    jae 3f
    mov dl, byte ptr [r10 + r12]
    mov cl, byte ptr [r10 + r11]
    mov byte ptr [r10 + r12], cl
    mov byte ptr [r10 + r11], dl
    inc r12
    dec r11
    jmp 2b
3:
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, r9
    sub rdx, r8
    call write_all
1:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all
    ret

_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae process
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jl exit_zero
    je process
    add r12, rax
    jmp read_loop

process:
    test r12, r12
    jz exit_zero
    xor rbx, rbx
    xor r13, r13
scan_loop:
    cmp rbx, r12
    jae final_line
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rbx]
    cmp cl, 0x0A
    jne 1f
    mov rdi, r13
    mov rsi, rbx
    push rbx
    push r13
    call emit_line
    pop r13
    pop rbx
    inc rbx
    mov r13, rbx
    jmp scan_loop
1:
    inc rbx
    jmp scan_loop

final_line:
    mov rdi, r13
    mov rsi, r12
    cmp rdi, rsi
    je exit_zero
    push rbx
    push r13
    call emit_line
    pop r13
    pop rbx
    jmp exit_zero

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
