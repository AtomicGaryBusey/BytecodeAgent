.intel_syntax noprefix
.globl _start

.section .bss
input_buf:
    .zero 65536
out_byte:
    .zero 1

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jle read_done
    syscall
    cmp rax, 0
    jl error_exit
    je read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    test r12, r12
    jz exit_zero

    xor r14, r14
    xor r15, r15

scan_loop:
    cmp r14, r12
    jae finish_scan

    movzx ecx, byte ptr [rip + input_buf + r14]
    cmp cl, 0x0A
    jne not_newline

    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, r14
    sub rdx, r15
    call print_rev_segment
    call write_newline

    mov r15, r14
    inc r15
    inc r14
    jmp scan_loop

not_newline:
    inc r14
    jmp scan_loop

finish_scan:
    cmp r15, r12
    jae exit_zero

    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, r12
    sub rdx, r15
    call print_rev_segment
    call write_newline
    jmp exit_zero

print_rev_segment:
    test rdx, rdx
    jz 1f
    mov r9, rsi
    lea r8, [rsi + rdx]
    dec r8

1:
    movzx eax, byte ptr [r8]
    mov byte ptr [rip + out_byte], al
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_byte]
    mov rdx, 1
    syscall

    cmp r8, r9
    je 1f
    dec r8
    jmp 1b

1:
    mov rsi, r9
    ret

write_newline:
    mov byte ptr [rip + out_byte], 0x0A
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_byte]
    mov rdx, 1
    syscall
    ret

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

exit_zero:
    xor edi, edi
    mov rax, 60
    syscall
