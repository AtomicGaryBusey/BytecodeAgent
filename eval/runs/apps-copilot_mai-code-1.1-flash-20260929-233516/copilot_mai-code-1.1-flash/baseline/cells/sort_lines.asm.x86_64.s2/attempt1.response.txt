.globl _start

.equ INPUT_MAX, 32768
.equ ENTRY_SIZE, 16
.equ MAX_LINES, 1000

.section .rodata
newline_byte:
    .byte 10

.bss
input_buffer:
    .space INPUT_MAX
line_entries:
    .space MAX_LINES * ENTRY_SIZE

.text
_start:
    lea r9, [rip + input_buffer]
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r9
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    jz read_done
    syscall
    mov rbx, rax
    cmp rbx, 0
    je read_done
    add r12, rbx
    cmp r12, INPUT_MAX
    jb read_loop

read_done:
    lea r9, [rip + input_buffer]
    xor r10, r10
    xor r11, r11
    xor r13, r13
    lea r14, [rip + line_entries]

parse_loop:
    cmp r11, r12
    jae parse_done
    movzx eax, byte ptr [r9 + r11]
    cmp al, 10
    jne parse_next
    mov rax, r13
    lea rdi, [r14 + rax * ENTRY_SIZE]
    mov rdx, r9
    add rdx, r10
    mov qword ptr [rdi], rdx
    mov rdx, r11
    sub rdx, r10
    mov qword ptr [rdi + 8], rdx
    inc r13
    mov r10, r11
    inc r10
    inc r11
    jmp parse_loop

parse_next:
    inc r11
    jmp parse_loop

parse_done:
    cmp r10, r12
    jae parse_end
    mov rax, r13
    lea rdi, [r14 + rax * ENTRY_SIZE]
    mov rdx, r9
    add rdx, r10
    mov qword ptr [rdi], rdx
    mov rdx, r12
    sub rdx, r10
    mov qword ptr [rdi + 8], rdx
    inc r13

parse_end:
    mov rbp, r13
    mov r14, 1

sort_outer:
    cmp r14, rbp
    jae sort_done
    mov r15, r14

sort_inner:
    test r15, r15
    jz sort_inner_done

    lea rsi, [rip + line_entries]
    lea rdi, [rsi + r15 * ENTRY_SIZE - ENTRY_SIZE]
    lea rdx, [rsi + r15 * ENTRY_SIZE]

    mov r12, rdi
    mov r13, rdx

    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rdx]
    mov r11, qword ptr [rdx + 8]

    mov rax, r9
    cmp r9, r11
    jbe compare_bytes_start
    mov rax, r11

compare_bytes_start:
    xor rcx, rcx

compare_bytes:
    cmp rcx, rax
    jae compare_lengths
    movzx rsi, byte ptr [r8 + rcx]
    movzx rdx, byte ptr [r10 + rcx]
    cmp sil, dl
    jb no_swap
    ja swap_entries
    inc rcx
    jmp compare_bytes

compare_lengths:
    cmp r9, r11
    jb no_swap
    ja swap_entries
    jmp dec_j

no_swap:
    dec r15
    jmp sort_inner

swap_entries:
    mov rbx, qword ptr [r12]
    mov rbp, qword ptr [r12 + 8]
    mov rsi, qword ptr [r13]
    mov rdx, qword ptr [r13 + 8]
    mov qword ptr [r12], rsi
    mov qword ptr [r12 + 8], rdx
    mov qword ptr [r13], rbx
    mov qword ptr [r13 + 8], rbp

dec_j:
    dec r15
    jmp sort_inner

sort_inner_done:
    inc r14
    jmp sort_outer

sort_done:
    xor r14, r14

print_loop:
    cmp r14, rbp
    jae print_done

    lea rsi, [rip + line_entries]
    lea rdi, [rsi + r14 * ENTRY_SIZE]
    mov rbx, qword ptr [rdi]
    mov rdx, qword ptr [rdi + 8]

    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall

    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall

    inc r14
    jmp print_loop

print_done:
    mov rax, 60
    xor rdi, rdi
    syscall
