.globl _start

.equ INPUT_MAX, 32768
.equ ENTRY_SIZE, 16
.equ MAX_LINES, 1000

.bss
input_buffer:
    .space INPUT_MAX
line_entries:
    .space MAX_LINES * ENTRY_SIZE

.section .rodata
newline_byte:
    .byte 10

.text
_start:
    lea r12, [rip + input_buffer]
    xor rbp, rbp

read_loop:
    cmp rbp, INPUT_MAX
    jae read_done

    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    add rsi, rbp
    mov rdx, INPUT_MAX
    sub rdx, rbp
    syscall

    cmp rax, 0
    je read_done

    add rbp, rax
    cmp rbp, INPUT_MAX
    jb read_loop

read_done:
    lea rbx, [rip + line_entries]
    xor r9, r9
    xor r14, r14
    xor r15, r15

parse_loop:
    cmp r14, rbp
    jae parse_done

    movzx ecx, byte ptr [r12 + r14]
    cmp cl, 10
    jne parse_next

    mov rax, r9
    shl rax, 4
    lea rdx, [rbx + rax]

    mov rax, r12
    add rax, r15
    mov qword ptr [rdx], rax

    mov rax, r14
    sub rax, r15
    mov qword ptr [rdx + 8], rax

    inc r9
    mov r15, r14
    inc r15
    inc r14
    jmp parse_loop

parse_next:
    inc r14
    jmp parse_loop

parse_done:
    cmp r15, rbp
    jae parse_end

    mov rax, r9
    shl rax, 4
    lea rdx, [rbx + rax]

    mov rax, r12
    add rax, r15
    mov qword ptr [rdx], rax

    mov rax, rbp
    sub rax, r15
    mov qword ptr [rdx + 8], rax

    inc r9

parse_end:
    mov r14, 1

sort_outer:
    cmp r14, r9
    jae sort_done

    mov r15, r14

sort_inner:
    cmp r15, 0
    je sort_inner_done

    mov rax, r15
    shl rax, 4
    lea rsi, [rbx + rax]
    lea rdi, [rsi - 16]

    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]
    mov r8, qword ptr [rsi]
    mov rcx, qword ptr [rsi + 8]

    mov rax, r11
    cmp rax, rcx
    jbe compare_min_done
    mov rax, rcx

compare_min_done:
    xor rdx, rdx

compare_byte_loop:
    cmp rdx, rax
    jae compare_length_order

    movzx esi, byte ptr [r10 + rdx]
    movzx edi, byte ptr [r8 + rdx]

    cmp sil, dil
    jb no_swap
    ja do_swap

    inc rdx
    jmp compare_byte_loop

compare_length_order:
    cmp r11, rcx
    jb no_swap
    ja do_swap
    jmp no_swap

do_swap:
    mov rax, r15
    shl rax, 4
    lea rsi, [rbx + rax]
    lea rdi, [rsi - 16]

    mov rax, qword ptr [rdi]
    mov rdx, qword ptr [rdi + 8]
    mov rcx, qword ptr [rsi]
    mov r8, qword ptr [rsi + 8]

    mov qword ptr [rdi], rcx
    mov qword ptr [rdi + 8], r8
    mov qword ptr [rsi], rax
    mov qword ptr [rsi + 8], rdx

    dec r15
    jmp sort_inner

no_swap:
    jmp sort_inner_done

sort_inner_done:
    inc r14
    jmp sort_outer

sort_done:
    xor r14, r14

print_loop:
    cmp r14, r9
    jae print_done

    mov rax, r14
    shl rax, 4
    lea rsi, [rbx + rax]
    mov rdx, qword ptr [rsi + 8]
    mov rsi, qword ptr [rsi]

    mov rax, 1
    mov rdi, 1
    syscall

    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall

    inc r14
    jmp print_loop

print_done:
    mov rax, 60
    xor rdi, rdi
    syscall
