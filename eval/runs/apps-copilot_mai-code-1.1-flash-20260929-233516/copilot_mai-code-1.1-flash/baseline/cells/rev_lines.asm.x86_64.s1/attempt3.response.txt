.text
.globl _start

_start:
    call read_all
    mov r12, rax
    xor r13, r13

process_loop:
    cmp r13, r12
    jae exit_ok

    mov r14, r13

scan_loop:
    cmp r14, r12
    jae final_line

    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r14]
    cmp al, 0x0A
    je newline_found

    inc r14
    jmp scan_loop

newline_found:
    mov rdi, r13
    mov rsi, r14
    call reverse_range

    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov rdx, r14
    sub rdx, r13
    call write_all

    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all

    mov r13, r14
    inc r13
    jmp process_loop

final_line:
    cmp r13, r12
    je exit_ok

    mov rdi, r13
    mov rsi, r12
    call reverse_range

    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov rdx, r12
    sub rdx, r13
    call write_all

    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

read_all:
    push r12
    xor r12, r12

read_loop:
    cmp r12, 65536
    jae read_done

    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall

    cmp rax, 0
    jle read_done

    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    mov rax, r12
    pop r12
    ret

reverse_range:
    cmp rdi, rsi
    jae reverse_done

    lea r8, [rip + input_buffer]
    mov r9, rdi
    mov r10, rsi
    dec r10

reverse_loop:
    cmp r9, r10
    jae reverse_done

    mov al, byte ptr [r8 + r9]
    mov dl, byte ptr [r8 + r10]
    mov byte ptr [r8 + r9], dl
    mov byte ptr [r8 + r10], al

    inc r9
    dec r10
    jmp reverse_loop

reverse_done:
    ret

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx

write_loop:
    test r10, r10
    jz write_done

    mov rax, 1
    mov rdi, r8
    mov rsi, r9
    mov rdx, r10
    syscall

    cmp rax, 0
    jle write_fail

    sub r10, rax
    add r9, rax
    jmp write_loop

write_fail:
    mov rax, 60
    xor rdi, rdi
    syscall

write_done:
    ret

.section .rodata
.align 1
newline_byte:
    .byte 0x0A

.section .bss
.align 16
input_buffer:
    .space 65536
