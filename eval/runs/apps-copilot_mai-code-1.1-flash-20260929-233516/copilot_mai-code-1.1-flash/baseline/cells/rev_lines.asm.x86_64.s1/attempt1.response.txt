.text
.globl _start

_start:
    mov r12, 0

read_loop:
    cmp r12, 65536
    jae read_done

    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall

    cmp rax, 0
    je read_done

    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    mov rbx, 0

process_loop:
    cmp rbx, r12
    jge exit_program

    mov rcx, rbx

scan_loop:
    cmp rcx, r12
    jae no_nl

    lea rsi, [rip + input_buffer]
    add rsi, rcx
    movzx edx, byte ptr [rsi]
    cmp dl, 0x0A
    je found_nl

    inc rcx
    jmp scan_loop

found_nl:
    mov rsi, rbx
    mov rdi, rcx

    cmp rsi, rdi
    je empty_line

    dec rdi

reverse_loop:
    cmp rsi, rdi
    jge reverse_done

    lea r8, [rip + input_buffer]
    mov al, byte ptr [r8 + rsi]
    mov dl, byte ptr [r8 + rdi]
    mov byte ptr [r8 + rsi], dl
    mov byte ptr [r8 + rdi], al

    inc rsi
    dec rdi
    jmp reverse_loop

reverse_done:
    mov rdx, rcx
    sub rdx, rbx

    cmp rdx, 0
    je write_newline

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    syscall

write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    mov rbx, rcx
    inc rbx
    jmp process_loop

empty_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    mov rbx, rcx
    inc rbx
    jmp process_loop

no_nl:
    mov rsi, rbx
    mov rdi, r12

    cmp rsi, rdi
    je exit_program

    mov rdx, r12
    sub rdx, rbx

    mov rsi, rbx
    mov rdi, r12
    dec rdi

final_rev_loop:
    cmp rsi, rdi
    jge final_rev_done

    lea r8, [rip + input_buffer]
    mov al, byte ptr [r8 + rsi]
    mov dl, byte ptr [r8 + rdi]
    mov byte ptr [r8 + rsi], dl
    mov byte ptr [r8 + rdi], al

    inc rsi
    dec rdi
    jmp final_rev_loop

final_rev_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.align 8
input_buffer:
    .space 65536

.section .rodata
.align 1
newline_byte:
    .byte 0x0A
