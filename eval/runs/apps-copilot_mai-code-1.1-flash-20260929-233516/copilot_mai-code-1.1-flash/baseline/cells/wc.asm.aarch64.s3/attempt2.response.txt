.globl _start
.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x23, x0
    cmp x23, #0
    b.eq done

    add x21, x21, x23
    mov x24, #0

process_chunk:
    cmp x24, x23
    b.hs read_loop

    ldrb w10, [x1, x24]
    add x24, x24, #1

    cmp w10, #10
    cset x11, eq
    add x19, x19, x11

    mov x11, #0
    cmp w10, #0x20
    cset x12, eq
    orr x11, x11, x12
    cmp w10, #0x09
    cset x12, eq
    orr x11, x11, x12
    cmp w10, #0x0B
    cset x12, eq
    orr x11, x11, x12
    cmp w10, #0x0C
    cset x12, eq
    orr x11, x11, x12
    cmp w10, #0x0D
    cset x12, eq
    orr x11, x11, x12

    cmp x11, #0
    b.ne whitespace

    cmp x22, #0
    cset x12, eq
    add x20, x20, x12
    mov x22, #1
    b process_chunk

whitespace:
    mov x22, #0
    b process_chunk

done:
    mov x0, x19
    bl print_u64

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x20
    bl print_u64

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x21
    bl print_u64

    mov x0, #1
    adrp x1, newline_str
    add x1, x1, :lo12:newline_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_u64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32

    mov x10, x0
    mov x11, #0
    cmp x10, #0
    b.ne print_u64_digits

    mov w12, #48
    strb w12, [sp]
    mov x11, #1
    b print_u64_write

print_u64_digits:
    add x13, sp, #32

digits_loop:
    cmp x10, #0
    b.eq print_u64_write

    mov x14, #10
    udiv x15, x10, x14
    msub x16, x15, x14, x10
    mov w17, #48
    add w17, w17, w16
    sub x13, x13, #1
    strb w17, [x13]
    add x11, x11, #1
    mov x10, x15
    b digits_loop

print_u64_write:
    mov x0, #1
    mov x1, x13
    mov x2, x11
    mov x8, #64
    svc #0

    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret

.section .rodata
space_str:
    .ascii " "
newline_str:
    .ascii "\n"

.section .bss
.align 3
input_buf:
    .space 65536
