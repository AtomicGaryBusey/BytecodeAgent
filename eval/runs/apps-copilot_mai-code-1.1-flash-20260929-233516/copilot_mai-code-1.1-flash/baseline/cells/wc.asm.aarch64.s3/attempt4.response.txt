.globl _start
.text

_start:
    mov x19, #0          // lines
    mov x20, #0          // words
    mov x21, #0          // bytes
    mov x22, #0          // in_word

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x23, x0

    cmp x23, #0
    b.lt exit_error
    b.eq done

    add x21, x21, x23
    mov x24, #0

process_loop:
    cmp x24, x23
    b.hs read_loop

    ldrb w9, [x1, x24]
    add x24, x24, #1

    cmp w9, #10
    b.ne 1f
    add x19, x19, #1
    mov x22, #0
    b process_loop

1:
    cmp w9, #0x20
    b.eq whitespace
    cmp w9, #0x09
    b.eq whitespace
    cmp w9, #0x0B
    b.eq whitespace
    cmp w9, #0x0C
    b.eq whitespace
    cmp w9, #0x0D
    b.eq whitespace

    cmp x22, #0
    b.ne process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop

whitespace:
    mov x22, #0
    b process_loop

done:
    mov x0, x19
    bl print_u64

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x20
    bl print_u64

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x21
    bl print_u64

    mov x0, #1
    adrp x1, newline_str
    add x1, x1, :lo12:newline_str
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

print_u64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32

    mov x10, x0
    mov x11, sp
    add x11, x11, #31

    cmp x10, #0
    b.ne 1f

    mov w12, #48
    strb w12, [x11]
    mov x12, #1
    b 2f

1:
    mov x12, #0

3:
    cmp x10, #0
    b.eq 2f

    mov x13, #10
    udiv x14, x10, x13
    msub x15, x14, x13, x10
    mov w16, #48
    add w16, w16, w15
    sub x11, x11, #1
    strb w16, [x11]
    add x12, x12, #1
    mov x10, x14
    b 3b

2:
    mov x0, #1
    mov x1, x11
    mov x2, x12
    mov x8, #64
    svc #0

    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret

.section .rodata
space_str:
    .ascii " "
newline_str:
    .ascii "\n"

.section .bss
.align 4
input_buf:
    .space 65536
