.globl _start
.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    b read_loop

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_reading
    mov x3, x0
    add x21, x21, x3

    adrp x4, input_buf
    add x4, x4, :lo12:input_buf
    mov x5, #0

process_chunk:
    cmp x5, x3
    b.hs read_loop
    ldrb w6, [x4, x5]
    add x5, x5, #1

    cmp w6, #10
    cset x7, eq
    add x19, x19, x7

    mov x7, #0
    cmp w6, #0x20
    cset x8, eq
    orr x7, x7, x8
    cmp w6, #0x09
    cset x8, eq
    orr x7, x7, x8
    cmp w6, #0x0B
    cset x8, eq
    orr x7, x7, x8
    cmp w6, #0x0C
    cset x8, eq
    orr x7, x7, x8
    cmp w6, #0x0D
    cset x8, eq
    orr x7, x7, x8

    cmp x7, #0
    b.ne whitespace_case
    cmp x22, #0
    cset x8, eq
    add x20, x20, x8
    mov x22, #1
    b process_chunk

whitespace_case:
    mov x22, #0
    b process_chunk

done_reading:
    mov x0, x19
    bl print_uint32

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    bl write_buf

    mov x0, x20
    bl print_uint32

    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    bl write_buf

    mov x0, x21
    bl print_uint32

    mov x0, #1
    adrp x1, newline_str
    add x1, x1, :lo12:newline_str
    mov x2, #1
    bl write_buf

    mov x0, #0
    mov x8, #93
    svc #0

print_uint32:
    stp x19, x30, [sp, #-16]!
    mov x19, x0

    adrp x0, temp_digits
    add x0, x0, :lo12:temp_digits
    add x0, x0, #31
    mov x1, #0
    mov x2, x19
    cmp x2, #0
    b.ne 1f
    mov w3, #48
    sub x0, x0, #1
    strb w3, [x0]
    mov x1, #1
    b 2f

1:
    mov x4, #10
1_loop:
    cmp x2, #0
    b.eq 2f
    udiv x5, x2, x4
    msub x6, x5, x4, x2
    mov w7, #48
    add w7, w7, w6
    sub x0, x0, #1
    strb w7, [x0]
    add x1, x1, #1
    mov x2, x5
    b 1_loop

2:
    mov x3, x0
    mov x4, x1
    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #64
    svc #0

    ldp x19, x30, [sp], #16
    ret

write_buf:
    mov x8, #64
    svc #0
    ret

.section .rodata
space_str:
    .ascii " "
newline_str:
    .ascii "\n"

.section .bss
.align 3
input_buf:
    .space 65536
.align 3
temp_digits:
    .space 32
