.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + stdin_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    mov rcx, rax
    mov rsi, rbx

process_chunk:
    movzx r10d, byte ptr [rsi]
    cmp r10b, 0x0A
    jne 1f
    inc r12
1:
    cmp r10b, 0x20
    je whitespace
    cmp r10b, 0x09
    je whitespace
    cmp r10b, 0x0B
    je whitespace
    cmp r10b, 0x0C
    je whitespace
    cmp r10b, 0x0D
    je whitespace
    test r15d, r15d
    jnz 2f
    inc r13
    mov r15d, 1
    jmp 3f
2:
    mov r15d, 1
    jmp 3f
whitespace:
    xor r15d, r15d
3:
    inc rsi
    dec rcx
    jnz process_chunk
    jmp read_loop

read_done:
    mov rdi, r12
    call write_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space]
    mov rdx, 1
    syscall

    mov rdi, r13
    call write_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space]
    mov rdx, 1
    syscall

    mov rdi, r14
    call write_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    mov rax, 60
    xor edi, edi
    syscall

write_num:
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    lea r13, [rip + numbuf + 31]
    mov r14, 10

    test r12, r12
    jnz 1f
    mov byte ptr [r13], '0'
    dec r13
    jmp 2f

1:
    xor edx, edx
    mov rax, r12
    div r14
    add dl, '0'
    mov byte ptr [r13], dl
    dec r13
    mov r12, rax
    test r12, r12
    jnz 1b

2:
    lea rsi, [r13 + 1]
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi

    mov rax, 1
    mov rdi, 1
    syscall

    pop r15
    pop r14
    pop r13
    pop r12
    ret

.section .rodata
space:
    .byte ' '
newline:
    .byte 10

.bss
stdin_buf:
    .space 65536

numbuf:
    .space 32
