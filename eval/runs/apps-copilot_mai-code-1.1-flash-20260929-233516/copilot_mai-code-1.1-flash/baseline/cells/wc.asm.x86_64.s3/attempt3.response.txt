.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + stdin_buf]

read_loop:
    xor edi, edi
    mov eax, 0
    mov rsi, rbx
    mov edx, 65536
    syscall
    test rax, rax
    jz read_done
    mov rcx, rax
    add r14, rax
    mov rsi, rbx

process_loop:
    movzx r9d, byte ptr [rsi]

    cmp r9b, 0x0A
    jne 1f
    inc r12
1:
    cmp r9b, 0x20
    je whitespace
    cmp r9b, 0x09
    je whitespace
    cmp r9b, 0x0B
    je whitespace
    cmp r9b, 0x0C
    je whitespace
    cmp r9b, 0x0D
    je whitespace

    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    inc rsi
    dec rcx
    jnz process_loop
    jmp read_loop

read_done:
    mov rdi, r12
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space]
    mov edx, 1
    syscall

    mov rdi, r13
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space]
    mov edx, 1
    syscall

    mov rdi, r14
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

write_num:
    mov rax, rdi
    lea rsi, [rip + numbuf + 31]

    test rax, rax
    jnz 1f
    mov byte ptr [rsi], '0'
    jmp 3f

1:
    mov rcx, 10
2:
    test rax, rax
    jz 3f
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    jmp 2b

3:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi

    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
space:
    .byte ' '
newline:
    .byte 10

.bss
stdin_buf:
    .space 65536

numbuf:
    .space 32
