.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + stdin_buf]

read_loop:
    mov eax, 0
    mov edi, 0
    mov rsi, rbx
    mov edx, 65536
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    mov rcx, rax
    mov rsi, rbx

process_chunk:
    movzx r10d, byte ptr [rsi]

    cmp r10b, 0x0A
    jne 1f
    inc r12
1:
    cmp r10b, 0x20
    je whitespace
    cmp r10b, 0x09
    je whitespace
    cmp r10b, 0x0B
    je whitespace
    cmp r10b, 0x0C
    je whitespace
    cmp r10b, 0x0D
    je whitespace

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    inc rsi
    dec rcx
    jnz process_chunk
    jmp read_loop

read_done:
    mov rdi, r12
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space]
    mov edx, 1
    syscall

    mov rdi, r13
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space]
    mov edx, 1
    syscall

    mov rdi, r14
    call write_num

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

write_num:
    mov rax, rdi
    lea rsi, [rip + numbuf + 31]
    mov rcx, 10

    test rax, rax
    jnz 1f
    mov byte ptr [rsi], '0'
    jmp 2f

1:
    xor edx, edx
2:
    test rax, rax
    jz 3f
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    jmp 2b

3:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi

    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
space:
    .byte ' '
newline:
    .byte 10

.bss
stdin_buf:
    .space 65536

numbuf:
    .space 64
