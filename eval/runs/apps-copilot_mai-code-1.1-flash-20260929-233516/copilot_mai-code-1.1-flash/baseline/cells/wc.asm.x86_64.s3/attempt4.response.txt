.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + stdin_buf]

read_loop:
    xor edi, edi
    mov rsi, rbx
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jz .Ldone
    mov rcx, rax
    add r14, rax
    mov rsi, rbx

process_loop:
    movzx eax, byte ptr [rsi]

    cmp al, 0x0A
    jne .Lnot_newline
    inc r12d
.Lnot_newline:
    cmp al, 0x20
    je .Lwhitespace
    cmp al, 0x09
    je .Lwhitespace
    cmp al, 0x0B
    je .Lwhitespace
    cmp al, 0x0C
    je .Lwhitespace
    cmp al, 0x0D
    je .Lwhitespace

    test r15d, r15d
    jnz .Lnot_word_start
    inc r13d
    mov r15d, 1
.Lnot_word_start:
    jmp .Lnext_byte

.Lwhitespace:
    xor r15d, r15d

.Lnext_byte:
    inc rsi
    dec rcx
    jnz process_loop
    jmp read_loop

.Ldone:
    mov rdi, r12
    call print_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space_msg]
    mov edx, 1
    syscall

    mov rdi, r13
    call print_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space_msg]
    mov edx, 1
    syscall

    mov rdi, r14
    call print_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_msg]
    mov edx, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

print_uint:
    mov rax, rdi
    lea rsi, [rip + numbuf + 31]
    test rax, rax
    jnz .Lnonzero
    mov byte ptr [rsi], '0'
    jmp .Lwrite

.Lnonzero:
    mov rcx, 10
.Lloop:
    test rax, rax
    jz .Lwrite
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    jmp .Lloop

.Lwrite:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
space_msg:
    .byte ' '
newline_msg:
    .byte 10

.bss
stdin_buf:
    .space 65536

numbuf:
    .space 32
