.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
    .align 16
output_buf:
    .space 128

.text
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jl exit_error
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
dump_loop:
    cmp r13, r12
    jae dump_done
    mov r10, r12
    sub r10, r13
    cmp r10, 16
    jle 1f
    mov r10, 16
1:
    lea rbx, [rip + output_buf]
    mov rax, r13
    lea r8, [rbx + 7]
    mov rcx, 8
2:
    mov rdx, rax
    and rdx, 0x0f
    mov dl, byte ptr [rip + hex_table + rdx]
    mov byte ptr [r8], dl
    shr rax, 4
    dec r8
    loop 2b
    mov byte ptr [rbx + 8], ' '
    mov byte ptr [rbx + 9], ' '
    lea r8, [rbx + 10]
    lea r9, [rip + input_buf]
    add r9, r13
    xor rcx, rcx
3:
    cmp rcx, 16
    jae 4f
    cmp rcx, r10
    jae 5f
    movzx eax, byte ptr [r9]
    mov rdx, rax
    shr al, 4
    and rdx, 0x0f
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r8], al
    mov al, byte ptr [rip + hex_table + rdx]
    mov byte ptr [r8 + 1], al
    add r9, 1
    add r8, 3
    jmp 6f
5:
    mov byte ptr [r8], ' '
    mov byte ptr [r8 + 1], ' '
    mov byte ptr [r8 + 2], ' '
    add r8, 3
6:
    cmp rcx, 7
    jne 7f
    mov byte ptr [r8], ' '
    inc r8
7:
    inc rcx
    jmp 3b
4:
    mov byte ptr [r8], ' '
    inc r8
    mov byte ptr [r8], '|'
    inc r8
    lea r9, [rip + input_buf]
    add r9, r13
    mov rcx, r10
8:
    test rcx, rcx
    jz 9f
    movzx eax, byte ptr [r9]
    cmp al, 0x20
    jb 10f
    cmp al, 0x7e
    jbe 11f
10:
    mov byte ptr [r8], '.'
    jmp 12f
11:
    mov byte ptr [r8], al
12:
    inc r8
    inc r9
    dec rcx
    jmp 8b
9:
    mov byte ptr [r8], '|'
    inc r8
    mov byte ptr [r8], 10
    inc r8
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r8
    lea rbx, [rip + output_buf]
    sub rdx, rbx
    syscall
    add r13, r10
    jmp dump_loop
dump_done:
    mov rax, r12
    lea rbx, [rip + output_buf]
    add rbx, 7
    mov rcx, 8
13:
    mov rdx, rax
    and rdx, 0x0f
    mov dl, byte ptr [rip + hex_table + rdx]
    mov byte ptr [rbx], dl
    shr rax, 4
    dec rbx
    loop 13b
    mov byte ptr [rip + output_buf + 8], 10
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, 9
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
