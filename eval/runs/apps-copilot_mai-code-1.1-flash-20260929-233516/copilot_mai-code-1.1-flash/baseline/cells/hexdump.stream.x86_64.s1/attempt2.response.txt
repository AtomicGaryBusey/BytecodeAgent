.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
    .align 16
line_buf:
    .space 128

.text

_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jl exit_error
    je read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13

dump_loop:
    cmp r13, r12
    jae dump_done

    lea rsi, [rip + hex_table]
    lea rbx, [rip + line_buf]

    mov rax, r13
    lea rdx, [rbx + 7]
    mov rcx, 8
1:
    mov r9, rax
    and r9, 15
    mov r10b, byte ptr [rsi + r9]
    mov byte ptr [rdx], r10b
    shr rax, 4
    dec rdx
    dec rcx
    jnz 1b

    mov byte ptr [rbx + 8], ' '
    mov byte ptr [rbx + 9], ' '

    mov rdi, rbx
    add rdi, 10

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jle 2f
    mov r14, 16
2:
    xor rcx, rcx

hex_fill_loop:
    cmp rcx, 16
    jae hex_fill_done

    cmp rcx, r14
    jae 3f

    lea r11, [rip + input_buf]
    add r11, r13
    add r11, rcx
    movzx r9, byte ptr [r11]

    mov r8, r9
    shr r8, 4
    and r8, 15
    mov r10b, byte ptr [rsi + r8]
    mov byte ptr [rdi], r10b
    inc rdi

    mov r8, r9
    and r8, 15
    mov r10b, byte ptr [rsi + r8]
    mov byte ptr [rdi], r10b
    inc rdi

    mov byte ptr [rdi], ' '
    inc rdi
    jmp 4f

3:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

4:
    cmp rcx, 7
    jne 5f
    mov byte ptr [rdi], ' '
    inc rdi

5:
    inc rcx
    jmp hex_fill_loop

hex_fill_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor rcx, rcx

ascii_fill_loop:
    cmp rcx, r14
    jae ascii_fill_done

    lea r11, [rip + input_buf]
    add r11, r13
    add r11, rcx
    movzx r9, byte ptr [r11]

    cmp r9d, 0x20
    jb 10f
    cmp r9d, 0x7e
    ja 10f
    mov byte ptr [rdi], r9b
    jmp 11f

10:
    mov byte ptr [rdi], '.'

11:
    inc rdi
    inc rcx
    jmp ascii_fill_loop

ascii_fill_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, rbx

    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall

    add r13, r14
    jmp dump_loop

dump_done:
    lea rbx, [rip + line_buf]
    lea rsi, [rip + hex_table]
    mov rax, r12
    lea rdx, [rbx + 7]
    mov rcx, 8
6:
    mov r9, rax
    and r9, 15
    mov r10b, byte ptr [rsi + r9]
    mov byte ptr [rdx], r10b
    shr rax, 4
    dec rdx
    dec rcx
    jnz 6b

    mov byte ptr [rbx + 8], 10

    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, 9
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
