.section .rodata

.bss
    .balign 8

    .zero 65536
    .balign 8

    .balign 8

.text

.globl _start

    mov x0, #0
    mov x2, #1
    mov x3, #65536
    sub x2, x3, x9
    mov x8, #63

    b.eq 2f

    mov x8, #93

    cmp x0, #0

    mov x29, sp
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x10, x9, #64
    mov x11, #0
    mov x12, x0
    cmp x12, #0
    b.ge 1f
    mov x11, #1
    mov x13, #0x8000
    lsl x13, x13, #48
    cmp x12, x13
    b.eq 2f
    neg x12, x12
    b 3f
2:
    mov x12, x13
3:
    b 4f

    mov x12, x0
4:
    cmp x12, #0
    b.ne 5f
    sub x10, x10, #1
    mov w13, #48
    strb w13, [x10]

    b.eq 6f
    udiv x15, x12, x14
    msub x13, x15, x14, x12
    add x13, x13, #48
    sub x10, x10, #1

    cmp x11, #0
    b.eq 7f
    sub x10, x10, #1
    mov w13, #45
    strb w13, [x10]

    mov x0, #1
    mov x1, x10
    add x13, x9, #64
    sub x2, x13, x10
    bl write_all
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all
    ldp x29, x30, [sp], #16
    ret

eval_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    adrp x19, stack_mem
    add x19, x19, :lo12:stack_mem
    mov x20, #0
    mov x21, #0
    mov x22, x0
    mov x23, x1

line_loop:
    cmp x22, x23
    b.ge end_line
    ldrb w24, [x22]
    cmp w24, #0x20
    b.eq skip_sep
    cmp w24, #0x09
    b.eq skip_sep
    b parse_token

skip_sep:
    add x22, x22, #1
    b line_loop

parse_token:
    mov x25, x22

scan_token_end:
    cmp x22, x23
    b.ge token_end
    ldrb w24, [x22]
    cmp w24, #0x20
    b.eq token_end
    cmp w24, #0x09
    b.eq token_end
    add x22, x22, #1
    b scan_token_end

token_end:
    mov x24, x22
    sub x24, x24, x25

    cmp x24, #1
    b.ne literal_check
    ldrb w26, [x25]
    cmp w26, #43
    b.eq op_plus
    cmp w26, #45
    b.eq op_minus
    cmp w26, #42
    b.eq op_mul
    cmp w26, #47
    b.eq op_div
    cmp w26, #37
    b.eq op_mod
    b invalid_token

literal_check:
    mov x27, #0
    mov x28, x24
    ldrb w26, [x25]
    cmp w26, #45
    b.ne literal_digits
    mov x27, #1
    add x25, x25, #1
    sub x28, x28, #1

literal_digits:
    cmp x28, #0
    b.eq invalid_token
    cmp x28, #19
    b.gt invalid_token
    mov x24, x28
    mov x26, x25
    mov x14, #0
    mov x15, #0

digit_loop:
    cmp x15, x24
    b.ge digits_done
    ldrb w16, [x26], #1
    cmp w16, #48
    b.lt invalid_token
    cmp w16, #57
    b.gt invalid_token
    sub w16, w16, #48
    mov x17, #10
    madd x14, x14, x17, x16
    add x15, x15, #1
    b digit_loop

digits_done:
    cmp x27, #0
    b.eq push_value
    cmp x14, #0
    b.eq push_value
    neg x14, x14

push_value:
    add x16, x19, x20, lsl #3
    str x14, [x16]
    add x20, x20, #1
    b line_loop

invalid_token:
    mov x21, #1
    b end_line

op_plus:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x15, [x16]
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x16, [x16]
    add x14, x16, x15
    add x20, x20, #1
    add x17, x19, x20, lsl #3
    str x14, [x17]
    b line_loop

op_minus:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x15, [x16]
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x16, [x16]
    sub x14, x16, x15
    add x20, x20, #1
    add x17, x19, x20, lsl #3
    str x14, [x17]

op_mul:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x15, [x16]
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x16, [x16]
    mul x14, x16, x15
    add x20, x20, #1
    add x17, x19, x20, lsl #3

op_div:

    b.lt invalid_token
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x15, [x16]
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x16, [x16]
    cmp x15, #0
    b.eq div_error
    mov x17, #0x8000
    lsl x17, x17, #48
    cmp x16, x17
    b.ne 1f
    cmp x15, #-1
    b.ne 1f
    b div_error
1:
    sdiv x14, x16, x15
    add x20, x20, #1
    add x17, x19, x20, lsl #3
    str x14, [x17]
    b line_loop

op_mod:
    cmp x20, #2
    b.lt invalid_token
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x15, [x16]
    sub x20, x20, #1
    add x16, x19, x20, lsl #3
    ldr x16, [x16]
    cmp x15, #0
    b.eq div_error
    mov x17, #0x8000
    lsl x17, x17, #48
    cmp x16, x17
    b.ne 1f
    cmp x15, #-1
    b.ne 1f
    b div_error

    sdiv x14, x16, x15
    msub x14, x14, x15, x16
    add x20, x20, #1
    add x17, x19, x20, lsl #3
    str x14, [x17]

    mov x21, #1
    b end_line

end_line:
    cmp x21, #0
    b.ne print_error_final
    cmp x20, #1
    b.ne print_error_final
    sub x20, x20, #1
    add x17, x19, x20, lsl #3
    ldr x0, [x17]
    bl print_decimal_signed
    b restore_eval

    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

_start:
    bl read_all
    mov x9, x0
    mov x10, #0
    cmp x9, #0
    b.eq end_program

    cmp x10, x9
    b.ge end_program
    mov x11, x10

scan_line:
    cmp x10, x9
    b.ge final_line
    adrp x12, input_buf
    add x12, x12, :lo12:input_buf
    add x12, x12, x10
    ldrb w13, [x12]
    cmp w13, #10
    b.eq newline_hit
    add x10, x10, #1
    b scan_line

newline_hit:
    mov x0, x11
    mov x1, x10
    bl eval_line
    add x10, x10, #1
    b process_lines

final_line:
    cmp x11, x10
    b.ge end_program
    mov x0, x11
    mov x1, x10
    bl eval_line

end_program:
    mov x0, #0
    mov x8, #93
    svc #0
