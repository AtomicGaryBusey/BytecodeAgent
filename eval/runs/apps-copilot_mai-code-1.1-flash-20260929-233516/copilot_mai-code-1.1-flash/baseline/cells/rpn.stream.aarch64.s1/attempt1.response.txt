.section .rodata
error_msg:
    .ascii "error\n"

.bss
    .balign 8
input_buf:
    .space 65536

    .balign 8
stack_mem:
    .space 16384

    .balign 8
outbuf:
    .space 256

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #65536
    b.hs process_input
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    mov x21, x0
    cmp x21, #0
    b.eq process_input
    add x20, x20, x21
    b read_loop

process_input:
    mov x21, #0

parse_next_line:
    cmp x21, x20
    b.hs done_process
    mov x22, x21
    mov x23, x21

scan_to_newline:
    cmp x23, x20
    b.hs final_line
    ldrb w24, [x19, x23]
    cmp w24, #10
    b.eq newline_case
    add x23, x23, #1
    b scan_to_newline

newline_case:
    mov x24, x23
    sub x24, x24, x22
    cmp x24, #0
    b.eq empty_line
    b process_line

empty_line:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    b do_write_stdout

final_line:
    cmp x22, x20
    b.hs done_process
    mov x24, x20
    sub x24, x24, x22
    mov x23, x20
    b process_line

process_line:
    adrp x17, stack_mem
    add x17, x17, :lo12:stack_mem
    mov x25, #0
    mov x26, x22
    mov x27, x23

parse_token_loop:
    cmp x26, x27
    b.hs end_of_line_parse

skip_sep_loop:
    cmp x26, x27
    b.hs end_of_line_parse
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq skip_sep_inc
    cmp w9, #0x09
    b.eq skip_sep_inc
    b token_start

skip_sep_inc:
    add x26, x26, #1
    b skip_sep_loop

token_start:
    mov x28, x26

token_scan_loop:
    cmp x26, x27
    b.hs token_end
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq token_end
    cmp w9, #0x09
    b.eq token_end
    add x26, x26, #1
    b token_scan_loop

token_end:
    sub x30, x26, x28

    cmp x30, #1
    b.ne not_single

    ldrb w9, [x28]
    cmp w9, #43
    b.eq op_plus
    cmp w9, #45
    b.eq op_minus
    cmp w9, #42
    b.eq op_mul
    cmp w9, #47
    b.eq op_div
    cmp w9, #37
    b.eq op_mod
    b line_error

not_single:
    ldrb w9, [x28]
    cmp w9, #45
    b.ne parse_pos_candidate

    cmp x30, #1
    b.eq op_minus

    mov x9, x30
    sub x9, x9, #1
    cmp x9, #18
    b.hi line_error

    mov x10, x28
    add x10, x10, #1
    add x11, x28, x30
    mov x12, #0

neg_digit_loop:
    cmp x10, x11
    b.hs neg_digits_done
    ldrb w13, [x10]
    cmp w13, #48
    b.lt line_error
    cmp w13, #57
    b.gt line_error
    sub w13, w13, #48

    mov x14, x12
    lsl x12, x12, #3
    add x12, x12, x14, lsl #1
    add x12, x12, x13
    add x10, x10, #1
    b neg_digit_loop

neg_digits_done:
    neg x12, x12
    b push_literal

parse_pos_candidate:
    cmp x30, #18
    b.hi line_error
    mov x10, x28
    add x11, x28, x30
    mov x12, #0

pos_digit_loop:
    cmp x10, x11
    b.hs pos_digits_done
    ldrb w13, [x10]
    cmp w13, #48
    b.lt line_error
    cmp w13, #57
    b.gt line_error
    sub w13, w13, #48

    mov x14, x12
    lsl x12, x12, #3
    add x12, x12, x14, lsl #1
    add x12, x12, x13
    add x10, x10, #1
    b pos_digit_loop

pos_digits_done:
    b push_literal

push_literal:
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

op_plus:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x10, [x9]
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x11, [x9]
    add x12, x11, x10
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

op_minus:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x10, [x9]
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x11, [x9]
    sub x12, x11, x10
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

op_mul:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x10, [x9]
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x11, [x9]
    mul x12, x11, x10
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

op_div:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x10, [x9]
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x11, [x9]

    cmp x10, #0
    b.eq line_error

    mov x12, #0x8000
    lsl x12, x12, #48
    cmp x11, x12
    b.ne div_no_overflow
    mov x13, #-1
    cmp x10, x13
    b.eq line_error

div_no_overflow:
    sdiv x12, x11, x10
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

op_mod:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x10, [x9]
    sub x25, x25, #1
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    ldr x11, [x9]

    cmp x10, #0
    b.eq line_error

    mov x12, #0x8000
    lsl x12, x12, #48
    cmp x11, x12
    b.ne mod_no_overflow
    mov x13, #-1
    cmp x10, x13
    b.eq line_error

mod_no_overflow:
    sdiv x12, x11, x10
    msub x12, x12, x10, x11
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x12, [x9]
    add x25, x25, #1
    b parse_token_loop

end_of_line_parse:
    cmp x25, #1
    b.ne line_error
    ldr x9, [x17]
    b print_value

line_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    b do_write_stdout

print_value:
    adrp x10, outbuf
    add x10, x10, :lo12:outbuf
    add x12, x10, #256
    mov x11, x12
    mov x13, #0

    cmp x9, #0
    b.ge pval_nonneg
    mov x13, #1
    sub x9, xzr, x9

pval_nonneg:
    cmp x9, #0
    b.ne pval_build_digits

    sub x11, x11, #1
    mov w14, #48
    strb w14, [x11]
    b pval_sign_check

pval_build_digits:
    mov x14, #10

pval_digits_loop:
    cmp x9, #0
    b.eq pval_digits_done
    udiv x15, x9, x14
    msub x16, x15, x14, x9
    add w16, w16, #48
    sub x11, x11, #1
    strb w16, [x11]
    mov x9, x15
    b pval_digits_loop

pval_digits_done:
    cmp x13, #0
    b.eq pval_sign_check
    sub x11, x11, #1
    mov w16, #45
    strb w16, [x11]

pval_sign_check:
    sub x12, x12, #1
    mov w16, #10
    strb w16, [x12]

    mov x0, #1
    mov x1, x11
    sub x2, x12, x11
    add x2, x2, #1
    b do_write_stdout

do_write_stdout:
    mov x0, #1
    mov x8, #64
    svc #0
    mov x9, x0
    cmp x9, x2
    b.ge write_done
    add x1, x1, x9
    sub x2, x2, x9
    b do_write_stdout

write_done:
    b line_done

line_done:
    cmp x23, x20
    b.eq done_process
    add x21, x23, #1
    b parse_next_line

done_process:
    mov x0, #0
    mov x8, #93
    svc #0
