.section .rodata
error_msg:
    .ascii "error\n"
newline:
    .byte 10

.bss
    .balign 8
input_buf:
    .zero 65536
    .balign 8
stack_mem:
    .zero 16384

.text
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.globl _start

write_all:
    mov x8, #SYS_write
1:
    svc #0
    cmp x0, #0
    b.lt 9f
    sub x2, x2, x0
    add x1, x1, x0
    cmp x2, #0
    b.ne 1b
    ret
9:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_value:
    sub sp, sp, #32
    mov x11, sp
    mov x12, #0
    mov x14, x0
    cmp x14, #0
    b.ge 1f
    mov x12, #1
    mov x15, #0x8000
    lsl x15, x15, #48
    cmp x14, x15
    b.eq 2f
    neg x14, x14
    b 1f
2:
    mov x14, x15
1:
    mov x13, #0
    cmp x14, #0
    b.ne 3f
    sub x11, x11, #1
    mov w15, #48
    strb w15, [x11]
    mov x13, #1
    b 4f
3:
3_loop:
    cmp x14, #0
    b.eq 4f
    mov x15, #10
    udiv x16, x14, x15
    msub x17, x16, x15, x14
    add w17, w17, #48
    sub x11, x11, #1
    strb w17, [x11]
    mov x14, x16
    add x13, x13, #1
    b 3_loop
4:
    cmp x12, #0
    b.eq 5f
    sub x11, x11, #1
    mov w15, #45
    strb w15, [x11]
    add x13, x13, #1
5:
    mov x0, #1
    mov x1, x11
    mov x2, x13
    bl write_all
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all
    add sp, sp, #32
    ret

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
read_loop:
    cmp x20, #INPUT_SIZE
    b.hs read_done
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    cmp x20, #0
    b.eq exit_zero
    mov x21, #0
next_line:
    cmp x21, x20
    b.hs exit_zero
    mov x22, x21
    mov x23, x21
scan_to_nl:
    cmp x23, x20
    b.hs final_line
    ldrb w9, [x19, x23]
    cmp w9, #10
    b.eq newline_line
    add x23, x23, #1
    b scan_to_nl
newline_line:
    mov x24, x23
    sub x24, x24, x22
    mov x29, #1
    b process_line
final_line:
    mov x24, x20
    sub x24, x24, x22
    mov x29, #0
    b process_line

process_line:
    adrp x17, stack_mem
    add x17, x17, :lo12:stack_mem
    mov x25, #0
    mov x26, x22
    mov x27, x22
    add x27, x27, x24
parse_loop:
    cmp x26, x27
    b.hs parse_done
skip_sep:
    cmp x26, x27
    b.hs parse_done
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq skip_inc
    cmp w9, #0x09
    b.eq skip_inc
    b token_start
skip_inc:
    add x26, x26, #1
    b skip_sep

token_start:
    mov x28, x26
scan_token:
    cmp x26, x27
    b.hs token_end
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq token_end
    cmp w9, #0x09
    b.eq token_end
    add x26, x26, #1
    b scan_token
token_end:
    sub x24, x26, x28
    cmp x24, #1
    b.eq single_char

    ldrb w9, [x19, x28]
    cmp w9, #45
    b.eq parse_neg
    cmp w9, #43
    b.eq token_invalid
    cmp x24, #18
    b.hi token_invalid
    mov x0, #0
    mov x9, x28
    mov x10, x27
parse_positive_loop:
    cmp x9, x10
    b.hs parse_positive_done
    ldrb w11, [x19, x9]
    cmp w11, #48
    b.lt token_invalid
    cmp w11, #57
    b.gt token_invalid
    sub w11, w11, #48
    mov x12, #10
    mul x0, x0, x12
    add x0, x0, x11
    add x9, x9, #1
    b parse_positive_loop
parse_positive_done:
    b push_literal

single_char:
    ldrb w9, [x19, x28]
    cmp w9, #43
    b.eq op_plus
    cmp w9, #45
    b.eq op_minus
    cmp w9, #42
    b.eq op_mul
    cmp w9, #47
    b.eq op_div
    cmp w9, #37
    b.eq op_mod
    cmp w9, #48
    b.lt token_invalid
    cmp w9, #57
    b.gt token_invalid
    sub w0, w9, #48
    b push_literal

parse_neg:
    cmp x24, #19
    b.hi token_invalid
    cmp x24, #2
    b.lt token_invalid
    mov x0, #0
    mov x9, x28
    add x9, x9, #1
    mov x10, x27
parse_negative_loop:
    cmp x9, x10
    b.hs parse_negative_done
    ldrb w11, [x19, x9]
    cmp w11, #48
    b.lt token_invalid
    cmp w11, #57
    b.gt token_invalid
    sub w11, w11, #48
    mov x12, #10
    mul x0, x0, x12
    add x0, x0, x11
    add x9, x9, #1
    b parse_negative_loop
parse_negative_done:
    neg x0, x0
    b push_literal

push_literal:
    mov x9, x25
    lsl x9, x9, #3
    add x9, x9, x17
    str x0, [x9]
    add x25, x25, #1
    b parse_loop

op_plus:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    ldr x1, [x17, x25, lsl #3]
    sub x25, x25, #1
    ldr x0, [x17, x25, lsl #3]
    add x0, x0, x1
    str x0, [x17, x25, lsl #3]
    add x25, x25, #1
    b parse_loop

op_minus:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    ldr x1, [x17, x25, lsl #3]
    sub x25, x25, #1
    ldr x0, [x17, x25, lsl #3]
    sub x0, x0, x1
    str x0, [x17, x25, lsl #3]
    add x25, x25, #1
    b parse_loop

op_mul:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    ldr x1, [x17, x25, lsl #3]
    sub x25, x25, #1
    ldr x0, [x17, x25, lsl #3]
    mul x0, x0, x1
    str x0, [x17, x25, lsl #3]
    add x25, x25, #1
    b parse_loop

op_div:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    ldr x1, [x17, x25, lsl #3]
    sub x25, x25, #1
    ldr x0, [x17, x25, lsl #3]
    cmp x1, #0
    b.eq line_error
    mov x2, #0x8000
    lsl x2, x2, #48
    cmp x0, x2
    b.ne 1f
    cmp x1, #-1
    b.eq line_error
1:
    sdiv x0, x0, x1
    str x0, [x17, x25, lsl #3]
    add x25, x25, #1
    b parse_loop

op_mod:
    cmp x25, #2
    b.lt line_error
    sub x25, x25, #1
    ldr x1, [x17, x25, lsl #3]
    sub x25, x25, #1
    ldr x0, [x17, x25, lsl #3]
    cmp x1, #0
    b.eq line_error
    mov x2, #0x8000
    lsl x2, x2, #48
    cmp x0, x2
    b.ne 1f
    cmp x1, #-1
    b.eq line_error
1:
    sdiv x2, x0, x1
    msub x0, x2, x1, x0
    str x0, [x17, x25, lsl #3]
    add x25, x25, #1
    b parse_loop

parse_done:
    cmp x25, #1
    b.ne line_error
    ldr x0, [x17]
    bl print_value
    b line_after

token_invalid:
line_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    bl write_all

line_after:
    cmp x29, #0
    b.eq line_after_final
    add x21, x23, #1
    b next_line
line_after_final:
    mov x21, x20
    b next_line

exit_zero:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
