.section .rodata

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

    cmp x20, #65536
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

    mov x21, #0

    cmp x21, x20
    b.hs done
    mov x22, x21

    cmp x21, x20
    b.hs final_line
    ldrb w23, [x19, x21]
    cmp w23, #10
    b.eq newline_line
    add x21, x21, #1
    b scan_line

    mov x23, x21
    mov x24, x23
    sub x24, x24, x22
    mov x29, #1
    b process_line

    b.hs done
    mov x23, x20
    mov x24, x23
    sub x24, x24, x22
    mov x29, #0

    adrp x17, stack_mem
    add x17, x17, :lo12:stack_mem
    mov x25, #0
    mov x26, x22
    add x27, x22, x24

parse_loop:
    cmp x26, x27
    b.hs parse_done

skip_sep:
    cmp x26, x27
    b.hs parse_done
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq skip_inc
    cmp w9, #0x09
    b.eq skip_inc

    b skip_sep

    mov x28, x26

    cmp x26, x27
    b.hs token_end
    ldrb w9, [x19, x26]
    cmp w9, #0x20
    b.eq token_end
    cmp w9, #0x09
    b.eq token_end
    add x26, x26, #1
    b scan_token

    sub x24, x26, x28

    cmp x24, #1
    b.ne maybe_negative
    ldrb w9, [x19, x28]
    cmp w9, #43
    b.eq op_plus
    cmp w9, #45
    b.eq op_minus
    cmp w9, #42
    b.eq op_mul
    cmp w9, #47
    b.eq op_div
    cmp w9, #37
    b.eq op_mod
    cmp w9, #48
    b.lt token_invalid
    cmp w9, #57
    b.gt token_invalid
    sub w0, w9, #48
    b push_literal

    cmp w9, #45

    cmp x24, #19

    cmp w11, #57
