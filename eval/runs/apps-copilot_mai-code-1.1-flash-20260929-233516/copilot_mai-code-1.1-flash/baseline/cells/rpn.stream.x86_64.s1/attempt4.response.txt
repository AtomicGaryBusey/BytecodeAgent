.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
stack_mem:
    .zero 16384
.balign 16
tmp_buf:
    .zero 64

.section .rodata
err_msg:
    .ascii "error\n"
min_str:
    .ascii "-9223372036854775808\n"

.text
.globl _start

_start:
    lea r15, [rip + input_buf]
    xor r14, r14

read_more:
    cmp r14, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r15 + r14]
    mov rdx, 65536
    sub rdx, r14
    syscall
    cmp rax, 0
    je read_done
    add r14, rax
    jmp read_more

read_done:
    xor rbx, rbx

line_loop:
    cmp rbx, r14
    jge exit_ok
    mov rcx, rbx

scan_nl:
    cmp rcx, r14
    jge no_newline
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x0A
    je found_newline
    inc rcx
    jmp scan_nl

found_newline:
    mov rdi, rbx
    mov rsi, rcx
    call process_line
    mov rbx, rcx
    inc rbx
    jmp line_loop

no_newline:
    cmp rbx, r14
    jge exit_ok
    mov rdi, rbx
    mov rsi, r14
    call process_line
    jmp exit_ok

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

process_line:
    push r15
    push r14
    push r13
    push r12
    push rbx

    lea r15, [rip + input_buf]
    lea r12, [rip + stack_mem]
    xor r13, r13
    mov r14, rsi
    mov r10, rdi

parse_loop:
    cmp r10, r14
    jge after_tokens

skip_sep:
    cmp r10, r14
    jge after_tokens
    movzx eax, byte ptr [r15 + r10]
    cmp al, 0x20
    je sep_next
    cmp al, 0x09
    je sep_next
    jmp parse_token

sep_next:
    inc r10
    jmp skip_sep

parse_token:
    mov r11, r10

scan_token:
    cmp r11, r14
    jge token_done
    movzx eax, byte ptr [r15 + r11]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc r11
    jmp scan_token

token_done:
    mov rbx, r11
    sub rbx, r10

    cmp rbx, 1
    jne token_len_gt_1

    movzx eax, byte ptr [r15 + r10]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

    cmp al, '0'
    jb token_invalid
    cmp al, '9'
    ja token_invalid

    movzx rax, byte ptr [r15 + r10]
    sub rax, '0'
    mov qword ptr [r12 + r13*8], rax
    inc r13
    mov r10, r11
    jmp parse_loop

token_len_gt_1:
    movzx eax, byte ptr [r15 + r10]
    cmp al, '-'
    jne positive_token

    cmp rbx, 19
    ja token_invalid
    cmp rbx, 1
    jle token_invalid

    mov rcx, r10
    inc rcx
    mov rsi, 1
    jmp accumulate_digits

positive_token:
    cmp rbx, 18
    ja token_invalid
    cmp rbx, 0
    je token_invalid
    mov rcx, r10
    xor rsi, rsi

accumulate_digits:
    mov rax, 0
    mov rdx, rcx

digits_loop:
    cmp rdx, r11
    jge digits_done
    movzx ebx, byte ptr [r15 + rdx]
    cmp bl, '0'
    jb token_invalid
    cmp bl, '9'
    ja token_invalid
    sub bl, '0'
    imul rax, rax, 10
    add rax, rbx
    inc rdx
    jmp digits_loop

digits_done:
    test rsi, rsi
    jz push_literal
    neg rax

push_literal:
    mov qword ptr [r12 + r13*8], rax
    inc r13
    mov r10, r11
    jmp parse_loop

op_plus:
    cmp r13, 2
    jb line_error
    mov rbx, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    add rax, rbx
    sub r13, 1
    mov qword ptr [r12 + r13*8 - 8], rax
    mov r10, r11
    jmp parse_loop

op_minus:
    cmp r13, 2
    jb line_error
    mov rbx, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    sub rax, rbx
    sub r13, 1
    mov qword ptr [r12 + r13*8 - 8], rax
    mov r10, r11
    jmp parse_loop

op_mul:
    cmp r13, 2
    jb line_error
    mov rbx, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    imul rax, rbx
    sub r13, 1
    mov qword ptr [r12 + r13*8 - 8], rax
    mov r10, r11
    jmp parse_loop

op_div:
    cmp r13, 2
    jb line_error
    mov rbx, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    cmp rbx, 0
    je line_error
    mov rdx, -9223372036854775808
    cmp rax, rdx
    jne div_ok
    cmp rbx, -1
    je line_error
div_ok:
    cqo
    idiv rbx
    sub r13, 1
    mov qword ptr [r12 + r13*8 - 8], rax
    mov r10, r11
    jmp parse_loop

op_mod:
    cmp r13, 2
    jb line_error
    mov rbx, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    cmp rbx, 0
    je line_error
    mov rdx, -9223372036854775808
    cmp rax, rdx
    jne mod_ok
    cmp rbx, -1
    je line_error
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    sub r13, 1
    mov qword ptr [r12 + r13*8 - 8], rax
    mov r10, r11
    jmp parse_loop

token_invalid:
line_error:
    call print_error
    jmp process_line_end

after_tokens:
    cmp r13, 1
    jne line_error
    mov rax, qword ptr [r12]
    call print_result

process_line_end:
    pop rbx
    pop r12
    pop r13
    pop r14
    pop r15
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    ret

print_result:
    mov rdx, -9223372036854775808
    cmp rax, rdx
    je print_min_result

    lea r10, [rip + tmp_buf]
    add r10, 63
    mov byte ptr [r10], 10
    dec r10

    mov rdx, 0
    mov rbx, 0
    cmp rax, 0
    jge digits_loop2
    mov rbx, 1
    neg rax

digits_loop2:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    mov byte ptr [r10], dl
    dec r10
    test rax, rax
    jnz digits_loop2

    test rbx, rbx
    jz digits_done2
    mov byte ptr [r10], '-'
    dec r10

digits_done2:
    inc r10
    mov rsi, r10
    lea rdx, [rip + tmp_buf + 64]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    ret

print_min_result:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, 21
    syscall
    ret
