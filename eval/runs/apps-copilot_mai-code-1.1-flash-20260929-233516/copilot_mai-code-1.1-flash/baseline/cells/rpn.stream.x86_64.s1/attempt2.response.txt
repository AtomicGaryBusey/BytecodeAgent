.section .bss
.balign 16
inbuf:
    .zero 65536
.balign 16
stack:
    .zero 16384
.balign 16
outbuf:
    .zero 128

.section .rodata
err_msg:
    .ascii "error\n"
minus_char:
    .byte 45

.text
.globl _start

_start:
    lea r15, [rip + inbuf]
    lea r14, [rip + stack]
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r15
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 65536
    jl read_loop

read_done:
    xor r13, r13

process_lines:
    cmp r13, r12
    jge exit_ok
    mov rsi, r13
    mov r9, 0

scan_line:
    cmp r13, r12
    je end_of_line
    movzx eax, byte ptr [r15 + r13]
    cmp al, 10
    je end_of_line_nl
    inc r13
    jmp scan_line

end_of_line_nl:
    mov r9, 1

end_of_line:
    xor rbx, rbx
    mov rcx, rsi

parse_tokens:
    cmp rcx, r13
    jae after_tokens
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp parse_token

skip_sep:
    inc rcx
    jmp parse_tokens

parse_token:
    mov r8, rcx

token_loop:
    cmp rcx, r13
    jae token_ready
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x20
    je token_ready
    cmp al, 0x09
    je token_ready
    inc rcx
    jmp token_loop

token_ready:
    mov r11, rcx
    sub r11, r8

    cmp r11, 1
    jne parse_not_single

    movzx eax, byte ptr [r15 + r8]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp error_output

parse_not_single:
    cmp r11, 19
    ja error_output

    movzx eax, byte ptr [r15 + r8]
    cmp al, '-'
    jne parse_positive

    cmp r11, 1
    je error_output

    movzx eax, byte ptr [r15 + r8 + 1]
    cmp al, '0'
    jb error_output
    cmp al, '9'
    ja error_output

    xor rax, rax
    mov r10, 1

neg_digit_loop:
    cmp r10, r11
    jae neg_digits_done
    movzx ecx, byte ptr [r15 + r8 + r10]
    cmp cl, '0'
    jb error_output
    cmp cl, '9'
    ja error_output
    sub cl, '0'
    movzx ecx, cl
    imul rax, rax, 10
    add rax, rcx
    inc r10
    jmp neg_digit_loop

neg_digits_done:
    neg rax
    jmp push_value

parse_positive:
    cmp r11, 18
    ja error_output

    movzx eax, byte ptr [r15 + r8]
    cmp al, '0'
    jb error_output
    cmp al, '9'
    ja error_output

    xor rax, rax
    mov r10, 0

pos_digit_loop:
    cmp r10, r11
    jae pos_digits_done
    movzx ecx, byte ptr [r15 + r8 + r10]
    cmp cl, '0'
    jb error_output
    cmp cl, '9'
    ja error_output
    sub cl, '0'
    movzx ecx, cl
    imul rax, rax, 10
    add rax, rcx
    inc r10
    jmp pos_digit_loop

pos_digits_done:
    jmp push_value

push_value:
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    jmp parse_tokens

op_plus:
    cmp rbx, 2
    jb error_output
    mov rax, qword ptr [r14 + (rbx - 1)*8]
    mov rdx, qword ptr [r14 + (rbx - 2)*8]
    add rdx, rax
    dec rbx
    mov qword ptr [r14 + (rbx - 1)*8], rdx
    jmp parse_tokens

op_minus:
    cmp rbx, 2
    jb error_output
    mov rax, qword ptr [r14 + (rbx - 1)*8]
    mov rdx, qword ptr [r14 + (rbx - 2)*8]
    sub rdx, rax
    dec rbx
    mov qword ptr [r14 + (rbx - 1)*8], rdx
    jmp parse_tokens

op_mul:
    cmp rbx, 2
    jb error_output
    mov rax, qword ptr [r14 + (rbx - 1)*8]
    mov rdx, qword ptr [r14 + (rbx - 2)*8]
    imul rdx, rax
    dec rbx
    mov qword ptr [r14 + (rbx - 1)*8], rdx
    jmp parse_tokens

op_div:
    cmp rbx, 2
    jb error_output
    mov r10, qword ptr [r14 + (rbx - 1)*8]
    mov r11, qword ptr [r14 + (rbx - 2)*8]
    cmp r10, 0
    je error_output
    cmp r11, -9223372036854775808
    jne div_ok
    cmp r10, -1
    je error_output

div_ok:
    mov rax, r11
    cqo
    mov rcx, r10
    idiv rcx
    dec rbx
    mov qword ptr [r14 + (rbx - 1)*8], rax
    jmp parse_tokens

op_mod:
    cmp rbx, 2
    jb error_output
    mov r10, qword ptr [r14 + (rbx - 1)*8]
    mov r11, qword ptr [r14 + (rbx - 2)*8]
    cmp r10, 0
    je error_output
    cmp r11, -9223372036854775808
    jne mod_ok
    cmp r10, -1
    je error_output

mod_ok:
    mov rax, r11
    cqo
    mov rcx, r10
    idiv rcx
    mov rax, rdx
    dec rbx
    mov qword ptr [r14 + (rbx - 1)*8], rax
    jmp parse_tokens

after_tokens:
    cmp rbx, 1
    jne error_output
    mov rax, qword ptr [r14]
    jmp emit_number

error_output:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    jmp line_done

emit_number:
    mov rbx, 0
    cmp rax, 0
    jge emit_abs
    mov rbx, 1
    cmp rax, -9223372036854775808
    jne emit_neg
    mov rax, 1
    shl rax, 63
    jmp emit_digits

emit_neg:
    neg rax

emit_abs:
    ; rax is non-negative absolute value
emit_digits:
    lea r8, [rip + outbuf + 64]
    mov byte ptr [r8], 10
    dec r8

digit_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    mov byte ptr [r8], dl
    dec r8
    cmp rax, 0
    jne digit_loop

    cmp rbx, 1
    jne no_minus_out
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + minus_char]
    mov rdx, 1
    syscall

no_minus_out:
    mov rsi, r8
    inc rsi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp line_done

line_done:
    cmp r9, 0
    je process_lines
    inc r13
    jmp process_lines

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
