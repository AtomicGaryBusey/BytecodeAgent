.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
stack_mem:
    .zero 16384
.balign 16
output_buf:
    .zero 64

.section .rodata
err_msg:
    .ascii "error\n"
min_str:
    .ascii "-9223372036854775808\n"

.text
.globl _start

_start:
    lea r15, [rip + input_buf]
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r15
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    xor r13, r13

main_line_loop:
    cmp r13, r12
    jge exit_ok
    mov rdi, r13
    mov rsi, r13

scan_line_end:
    cmp rsi, r12
    jae line_done
    movzx eax, byte ptr [r15 + rsi]
    cmp al, 10
    je line_done
    inc rsi
    jmp scan_line_end

line_done:
    call process_line
    mov r13, rsi
    cmp r13, r12
    jae main_line_loop
    movzx eax, byte ptr [r15 + r13]
    cmp al, 10
    jne main_line_loop
    inc r13
    jmp main_line_loop

process_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    lea r14, [rip + stack_mem]
    xor rbx, rbx

    mov rcx, rdi

parse_skip:
    cmp rcx, rsi
    jae line_finish
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x20
    je add_sep
    cmp al, 0x09
    je add_sep
    jmp parse_token
add_sep:
    inc rcx
    jmp parse_skip

parse_token:
    mov r8, rcx

token_scan:
    cmp rcx, rsi
    jae token_done
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc rcx
    jmp token_scan

token_done:
    mov r10, rcx
    mov r9, r10
    sub r9, r8
    cmp r9, 1
    jne parse_non_single

    movzx eax, byte ptr [r15 + r8]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp parse_literal

parse_non_single:
    cmp r9, 19
    ja invalid_token
    movzx eax, byte ptr [r15 + r8]
    cmp al, '-'
    jne nonneg_len_check
    jmp parse_literal
nonneg_len_check:
    cmp r9, 19
    je invalid_token
    jmp parse_literal

parse_literal:
    cmp r9, 19
    ja invalid_token
    movzx eax, byte ptr [r15 + r8]
    cmp al, '-'
    jne pos_literal
    cmp r9, 1
    je invalid_token
    mov rdx, r8
    inc rdx
    mov r9, r9
    dec r9
    jmp validate_literal

pos_literal:
    cmp r9, 19
    je invalid_token
    mov rdx, r8
    mov r9, r9

validate_literal:
    xor rcx, rcx

validate_loop:
    cmp rcx, r9
    jae parse_value
    mov rax, rdx
    add rax, rcx
    movzx eax, byte ptr [r15 + rax]
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    inc rcx
    jmp validate_loop

parse_value:
    xor r11, r11
    xor rcx, rcx

accum_loop:
    cmp rcx, r9
    jae value_done
    mov rax, rdx
    add rax, rcx
    movzx eax, byte ptr [r15 + rax]
    sub eax, 48
    imul r11, r11, 10
    add r11, rax
    inc rcx
    jmp accum_loop

value_done:
    movzx eax, byte ptr [r15 + r8]
    cmp al, '-'
    jne push_value
    neg r11

push_value:
    mov rax, r11
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    mov rcx, r10
    jmp parse_skip

op_plus:
    cmp rbx, 2
    jb line_error
    mov rdx, qword ptr [r14 + rbx*8 - 8]
    mov rax, qword ptr [r14 + rbx*8 - 16]
    add rax, rdx
    sub rbx, 1
    mov qword ptr [r14 + rbx*8 - 8], rax
    mov rcx, r10
    jmp parse_skip

op_minus:
    cmp rbx, 2
    jb line_error
    mov rdx, qword ptr [r14 + rbx*8 - 8]
    mov rax, qword ptr [r14 + rbx*8 - 16]
    sub rax, rdx
    sub rbx, 1
    mov qword ptr [r14 + rbx*8 - 8], rax
    mov rcx, r10
    jmp parse_skip

op_mul:
    cmp rbx, 2
    jb line_error
    mov rdx, qword ptr [r14 + rbx*8 - 8]
    mov rax, qword ptr [r14 + rbx*8 - 16]
    imul rax, rdx
    sub rbx, 1
    mov qword ptr [r14 + rbx*8 - 8], rax
    mov rcx, r10
    jmp parse_skip

op_div:
    cmp rbx, 2
    jb line_error
    mov rsi, qword ptr [r14 + rbx*8 - 8]
    mov rax, qword ptr [r14 + rbx*8 - 16]
    cmp rsi, 0
    je line_error
    mov r11, -9223372036854775808
    cmp rax, r11
    jne div_ok
    cmp rsi, -1
    jne div_ok
    jmp line_error
div_ok:
    cqo
    idiv rsi
    sub rbx, 1
    mov qword ptr [r14 + rbx*8 - 8], rax
    mov rcx, r10
    jmp parse_skip

op_mod:
    cmp rbx, 2
    jb line_error
    mov rsi, qword ptr [r14 + rbx*8 - 8]
    mov rax, qword ptr [r14 + rbx*8 - 16]
    cmp rsi, 0
    je line_error
    mov r11, -9223372036854775808
    cmp rax, r11
    jne mod_ok
    cmp rsi, -1
    jne mod_ok
    jmp line_error
mod_ok:
    cqo
    idiv rsi
    mov rax, rdx
    sub rbx, 1
    mov qword ptr [r14 + rbx*8 - 8], rax
    mov rcx, r10
    jmp parse_skip

line_finish:
    cmp rbx, 1
    jne line_error
    mov rax, qword ptr [r14 + rbx*8 - 8]
    call print_signed
    jmp process_line_done

invalid_token:
line_error:
    call print_error

process_line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    ret

print_signed:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rax
    mov r11, -9223372036854775808
    cmp r12, r11
    je print_min_int

    lea r13, [rip + output_buf]
    add r13, 63
    mov byte ptr [r13], 10
    dec r13
    mov r14, 0

    cmp r12, 0
    jge digit_loop
    mov r14, 1
    neg r12

digit_loop:
    xor rdx, rdx
    mov rbx, 10
    mov rax, r12
    div rbx
    add dl, 48
    mov byte ptr [r13], dl
    dec r13
    mov r12, rax
    test r12, r12
    jnz digit_loop

    cmp r14, 1
    jne digits_done
    mov byte ptr [r13], '-'
    dec r13

digits_done:
    inc r13
    mov rsi, r13
    lea rdx, [rip + output_buf + 64]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_min_int:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, 21
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
