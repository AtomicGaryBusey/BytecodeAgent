.text
.globl _start

INPUT_MAX = 32768
LINE_MAX = 1000

.bss
.align 16
input_buf:
    .zero INPUT_MAX
ptrs_buf:
    .zero LINE_MAX * 8
lens_buf:
    .zero LINE_MAX * 8

.section .rodata
newline_byte:
    .byte 0x0A

.text

_start:
    xor r12d, r12d

read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INPUT_MAX
    jb read_loop

read_done:
    xor r13d, r13d
    cmp r12d, 0
    je exit_zero

    lea r15, [rip + input_buf]
    add r15, r12
    lea rsi, [rip + input_buf]

parse_loop:
    cmp rsi, r15
    jae parse_done
    mov rdi, rsi

scan_loop:
    cmp rdi, r15
    jae final_line
    cmp byte ptr [rdi], 0x0A
    je newline_found
    inc rdi
    jmp scan_loop

newline_found:
    mov rax, rdi
    sub rax, rsi
    lea rcx, [rip + ptrs_buf]
    mov qword ptr [rcx + r13*8], rsi
    lea rcx, [rip + lens_buf]
    mov qword ptr [rcx + r13*8], rax
    inc r13
    lea rsi, [rdi + 1]
    jmp parse_loop

final_line:
    cmp rsi, r15
    je parse_done
    mov rax, r15
    sub rax, rsi
    lea rcx, [rip + ptrs_buf]
    mov qword ptr [rcx + r13*8], rsi
    lea rcx, [rip + lens_buf]
    mov qword ptr [rcx + r13*8], rax
    inc r13
    jmp parse_done

parse_done:
    cmp r13d, 0
    je exit_zero
    mov r12d, 1

sort_outer:
    cmp r12d, r13d
    jae output_lines
    lea rax, [rip + ptrs_buf]
    mov r10, qword ptr [rax + r12*8]
    lea rax, [rip + lens_buf]
    mov r11, qword ptr [rax + r12*8]
    mov r9d, r12d
    dec r9d

sort_inner:
    cmp r9d, 0
    jl insert_at_zero
    lea rax, [rip + ptrs_buf]
    mov rdi, qword ptr [rax + r9*8]
    lea rax, [rip + lens_buf]
    mov rsi, qword ptr [rax + r9*8]

    mov r8, rsi
    mov rdx, r11
    cmp rdx, r8
    cmovb r8, r11
    xor ecx, ecx

compare_loop:
    cmp rcx, r8
    jae compare_lengths
    movzx eax, byte ptr [rdi + rcx]
    movzx ebx, byte ptr [r10 + rcx]
    cmp al, bl
    ja shift_key
    jb insert_after_j
    inc rcx
    jmp compare_loop

compare_lengths:
    cmp rsi, r11
    jb insert_after_j
    ja shift_key
    jmp insert_after_j

shift_key:
    lea rax, [rip + ptrs_buf]
    mov rdx, r9
    inc rdx
    shl rdx, 3
    mov rbx, qword ptr [rax + r9*8]
    mov qword ptr [rax + rdx], rbx
    lea rax, [rip + lens_buf]
    mov rbx, qword ptr [rax + r9*8]
    mov qword ptr [rax + rdx], rbx
    dec r9d
    jmp sort_inner

insert_after_j:
    lea rax, [rip + ptrs_buf]
    mov rdx, r9
    inc rdx
    shl rdx, 3
    mov qword ptr [rax + rdx], r10
    lea rax, [rip + lens_buf]
    mov qword ptr [rax + rdx], r11
    inc r12d
    jmp sort_outer

insert_at_zero:
    lea rax, [rip + ptrs_buf]
    mov qword ptr [rax], r10
    lea rax, [rip + lens_buf]
    mov qword ptr [rax], r11
    inc r12d
    jmp sort_outer

output_lines:
    xor r12d, r12d

output_loop:
    cmp r12d, r13d
    jae exit_zero

    lea rax, [rip + ptrs_buf]
    mov r9, qword ptr [rax + r12*8]
    lea rax, [rip + lens_buf]
    mov r10, qword ptr [rax + r12*8]

    test r10, r10
    jz write_newline

    mov r11, r10
    mov r14, r9

write_bytes_loop:
    test r11, r11
    jz write_newline
    mov rdi, 1
    mov rsi, r14
    mov rdx, r11
    mov rax, 1
    syscall
    cmp rax, 0
    jle exit_zero
    sub r11, rax
    add r14, rax
    jmp write_bytes_loop

write_newline:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r12d
    jmp output_loop

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
