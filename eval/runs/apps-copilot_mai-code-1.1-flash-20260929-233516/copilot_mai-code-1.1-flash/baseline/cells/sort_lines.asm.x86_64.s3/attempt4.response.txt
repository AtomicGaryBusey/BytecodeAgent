.text
.globl _start

INPUT_MAX = 32768
LINE_MAX = 1000

_start:
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_MAX
    jae read_done
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    js read_error
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    cmp r12, 0
    je exit_zero

    mov r13, 0
    lea r14, [rip + input_buf]
    lea r15, [rip + input_buf]
    add r15, r12

parse_loop:
    cmp r14, r15
    jae parse_done
    mov rbx, r14

scan_loop:
    cmp rbx, r15
    jae final_line
    cmp byte ptr [rbx], 0x0A
    je newline_found
    inc rbx
    jmp scan_loop

newline_found:
    mov rdx, rbx
    sub rdx, r14
    lea rax, [rip + ptrs_buf]
    mov qword ptr [rax + r13*8], r14
    lea rax, [rip + lens_buf]
    mov qword ptr [rax + r13*8], rdx
    inc r13
    lea r14, [rbx + 1]
    jmp parse_loop

final_line:
    cmp r14, r15
    je parse_done
    mov rdx, r15
    sub rdx, r14
    lea rax, [rip + ptrs_buf]
    mov qword ptr [rax + r13*8], r14
    lea rax, [rip + lens_buf]
    mov qword ptr [rax + r13*8], rdx
    inc r13
    jmp parse_done

parse_done:
    cmp r13, 0
    je exit_zero

    mov r12, 1

sort_outer:
    cmp r12, r13
    jae output_lines

    lea rax, [rip + ptrs_buf]
    mov rbx, qword ptr [rax + r12*8]
    lea rax, [rip + lens_buf]
    mov r10, qword ptr [rax + r12*8]

    mov r9, r12
    dec r9

sort_inner:
    cmp r9, 0
    jl insert_here

    lea rax, [rip + ptrs_buf]
    mov rdi, qword ptr [rax + r9*8]
    lea rax, [rip + lens_buf]
    mov rsi, qword ptr [rax + r9*8]
    mov rdx, rbx
    mov rcx, r10
    call compare_lines
    cmp rax, 0
    jle insert_here

    lea rax, [rip + ptrs_buf]
    mov r8, qword ptr [rax + r9*8]
    mov qword ptr [rax + (r9 + 1)*8], r8
    lea rax, [rip + lens_buf]
    mov r8, qword ptr [rax + r9*8]
    mov qword ptr [rax + (r9 + 1)*8], r8
    dec r9
    jmp sort_inner

insert_here:
    lea rax, [rip + ptrs_buf]
    mov qword ptr [rax + (r9 + 1)*8], rbx
    lea rax, [rip + lens_buf]
    mov qword ptr [rax + (r9 + 1)*8], r10
    inc r12
    jmp sort_outer

compare_lines:
    mov r8, rsi
    cmp r8, rcx
    cmovb r8, rcx
    xor r9d, r9d
    mov r10, rdx

compare_loop:
    cmp r9, r8
    jae compare_lengths
    movzx eax, byte ptr [rdi + r9]
    movzx edx, byte ptr [r10 + r9]
    cmp eax, edx
    jb compare_less
    ja compare_greater
    inc r9
    jmp compare_loop

compare_lengths:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

output_lines:
    mov r12d, 0

output_loop:
    cmp r12d, r13d
    jae exit_zero

    lea rax, [rip + ptrs_buf]
    mov rbx, qword ptr [rax + r12*8]
    lea rax, [rip + lens_buf]
    mov rcx, qword ptr [rax + r12*8]

    test rcx, rcx
    jz write_single_newline

    mov rdi, 1
    mov rsi, rbx
    mov rdx, rcx
    call write_all

    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all

    inc r12d
    jmp output_loop

write_single_newline:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all
    inc r12d
    jmp output_loop

write_all:
    test rdx, rdx
    jz write_all_done

write_all_loop:
    mov rax, 1
    syscall
    cmp rax, 0
    jle write_error
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_all_loop

write_all_done:
    ret

write_error:
    mov edi, 1
    mov eax, 60
    syscall

read_error:
    mov edi, 1
    mov eax, 60
    syscall

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline_byte:
    .byte 0x0A

.bss
.align 16
input_buf:
    .zero INPUT_MAX

.align 8
ptrs_buf:
    .zero LINE_MAX * 8

.align 8
lens_buf:
    .zero LINE_MAX * 8
