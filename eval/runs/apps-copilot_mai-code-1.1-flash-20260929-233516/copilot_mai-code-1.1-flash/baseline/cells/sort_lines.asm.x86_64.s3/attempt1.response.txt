.bss
.align 16
input_buf:
    .zero 32768
line_ary:
    .zero 16000

.text
.globl _start

cmp_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rsi]
    mov r10, qword ptr [rdi + 8]
    mov r11, qword ptr [rsi + 8]

    mov rax, r10
    cmp rax, r11
    cmovb rax, r11

    xor ecx, ecx
1:
    cmp rcx, rax
    jae 2f
    movzx edx, byte ptr [r8 + rcx]
    movzx ebx, byte ptr [r9 + rcx]
    cmp dl, bl
    jb 3f
    ja 4f
    inc rcx
    jmp 1b
2:
    cmp r10, r11
    jb 3f
    ja 4f
    xor eax, eax
    ret
3:
    mov eax, -1
    ret
4:
    mov eax, 1
    ret

_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d

read_loop:
    mov rdi, 0
    mov rsi, r12
    add rsi, r13
    mov rdx, 32768
    sub rdx, r13
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    cmp r13, 32768
    jb read_loop

read_done:
    lea rbx, [rip + input_buf]
    lea r14, [rbx + r13]
    xor r15, r15
    mov rsi, rbx

parse_loop:
    cmp rsi, r14
    je parse_done

    mov rdi, rsi

scan_loop:
    cmp rdi, r14
    je final_line
    cmp byte ptr [rdi], 0x0A
    je newline_found
    inc rdi
    jmp scan_loop

newline_found:
    mov rax, rdi
    sub rax, rsi
    mov qword ptr [rip + line_ary + r15*16], rsi
    mov qword ptr [rip + line_ary + r15*16 + 8], rax
    inc r15
    mov rsi, rdi
    inc rsi
    jmp parse_loop

final_line:
    cmp rsi, r14
    je parse_done
    mov rax, r14
    sub rax, rsi
    mov qword ptr [rip + line_ary + r15*16], rsi
    mov qword ptr [rip + line_ary + r15*16 + 8], rax
    inc r15
    jmp parse_done

parse_done:
    cmp r15, 1
    jbe output

    mov r12, 1

sort_outer:
    cmp r12, r15
    jae output

    mov r8, qword ptr [rip + line_ary + r12*16]
    mov r9, qword ptr [rip + line_ary + r12*16 + 8]
    mov r13, r12
    dec r13

sort_inner:
    cmp r13, -1
    jl sort_store

    lea rdi, [rip + line_ary + r13*16]
    lea rsi, [rip + line_ary + r12*16]
    call cmp_lines
    test eax, eax
    jle sort_store

    mov r10, qword ptr [rip + line_ary + r13*16]
    mov r11, qword ptr [rip + line_ary + r13*16 + 8]
    mov qword ptr [rip + line_ary + (r13 + 1)*16], r10
    mov qword ptr [rip + line_ary + (r13 + 1)*16 + 8], r11
    dec r13
    jmp sort_inner

sort_store:
    mov qword ptr [rip + line_ary + (r13 + 1)*16], r8
    mov qword ptr [rip + line_ary + (r13 + 1)*16 + 8], r9
    inc r12
    jmp sort_outer

output:
    xor r12, r12

output_loop:
    cmp r12, r15
    jae exit_zero

    mov rsi, qword ptr [rip + line_ary + r12*16]
    mov rdx, qword ptr [rip + line_ary + r12*16 + 8]
    mov rdi, 1
    mov rax, 1
    inc rdx
    syscall

    inc r12
    jmp output_loop

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
