.text
.globl _start

INPUT_MAX = 32768
LINE_MAX = 1000
LINE_ENTRY_SIZE = 16

.bss
.align 16
input_buf:
    .zero INPUT_MAX
line_ary:
    .zero LINE_MAX * LINE_ENTRY_SIZE

.section .rodata
newline_byte:
    .byte 0x0A

.text

_start:
    xor r13d, r13d

read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r13
    mov rdx, INPUT_MAX
    sub rdx, r13
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    cmp r13, INPUT_MAX
    jb read_loop

read_done:
    lea rbx, [rip + line_ary]
    lea r10, [rip + input_buf]
    lea r11, [r10 + r13]
    xor r14d, r14d
    mov rsi, r10

parse_loop:
    cmp rsi, r11
    je parse_done
    mov rdi, rsi

scan_loop:
    cmp rdi, r11
    je final_line
    cmp byte ptr [rdi], 0x0A
    je newline_found
    inc rdi
    jmp scan_loop

newline_found:
    mov rax, rdi
    sub rax, rsi
    mov qword ptr [rbx + r14*LINE_ENTRY_SIZE], rsi
    mov qword ptr [rbx + r14*LINE_ENTRY_SIZE + 8], rax
    inc r14
    lea rsi, [rdi + 1]
    jmp parse_loop

final_line:
    cmp rsi, r11
    je parse_done
    mov rax, r11
    sub rax, rsi
    mov qword ptr [rbx + r14*LINE_ENTRY_SIZE], rsi
    mov qword ptr [rbx + r14*LINE_ENTRY_SIZE + 8], rax
    inc r14
    jmp parse_done

parse_done:
    cmp r14, 1
    jbe output
    mov r12, 1

sort_outer:
    cmp r12, r14
    jae output

    mov r8, qword ptr [rbx + r12*LINE_ENTRY_SIZE]
    mov r9, qword ptr [rbx + r12*LINE_ENTRY_SIZE + 8]
    mov r13, r12
    dec r13

sort_inner:
    cmp r13, 0
    jl sort_store

    mov rdi, qword ptr [rbx + r13*LINE_ENTRY_SIZE]
    mov rsi, qword ptr [rbx + r13*LINE_ENTRY_SIZE + 8]
    mov rdx, r8
    mov rcx, r9

    xor r10d, r10d
    mov r11, rsi
    cmp r11, rcx
    cmovb r11, rcx
1:
    cmp r10, r11
    jae 2f
    movzx eax, byte ptr [rdi + r10]
    movzx ebx, byte ptr [rdx + r10]
    cmp al, bl
    jb 3f
    ja 4f
    inc r10
    jmp 1b
2:
    cmp rsi, rcx
    jb 3f
    ja 4f
    xor eax, eax
    jmp 5f
3:
    mov eax, -1
    jmp 5f
4:
    mov eax, 1
5:
    test eax, eax
    jle sort_store

    mov rax, qword ptr [rbx + r13*LINE_ENTRY_SIZE]
    mov rdx, qword ptr [rbx + r13*LINE_ENTRY_SIZE + 8]
    mov qword ptr [rbx + (r13 + 1)*LINE_ENTRY_SIZE], rax
    mov qword ptr [rbx + (r13 + 1)*LINE_ENTRY_SIZE + 8], rdx
    dec r13
    jmp sort_inner

sort_store:
    mov qword ptr [rbx + (r13 + 1)*LINE_ENTRY_SIZE], r8
    mov qword ptr [rbx + (r13 + 1)*LINE_ENTRY_SIZE + 8], r9
    inc r12
    jmp sort_outer

output:
    mov r15, r14
    xor r12d, r12d

output_loop:
    cmp r12, r15
    jae exit_zero

    mov r13, qword ptr [rbx + r12*LINE_ENTRY_SIZE]
    mov r14, qword ptr [rbx + r12*LINE_ENTRY_SIZE + 8]
    mov r11, r14
    mov r10, r13

write_line_bytes:
    test r11, r11
    jz write_newline
    mov rdi, 1
    mov rsi, r10
    mov rdx, r11
    mov rax, 1
    syscall
    cmp rax, 0
    jle write_newline
    sub r11, rax
    add r10, rax
    jmp write_line_bytes

write_newline:
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r12
    jmp output_loop

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
