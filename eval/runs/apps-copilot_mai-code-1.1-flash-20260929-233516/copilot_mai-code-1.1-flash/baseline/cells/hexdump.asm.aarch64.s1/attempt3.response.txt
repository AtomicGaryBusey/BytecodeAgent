.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 4
input_buf:
    .space 65536
    .align 4
line_buf:
    .space 256

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #65536
    b.hs read_done

    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0

    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, print_total

    mov x21, #0

row_loop:
    sub x22, x20, x21
    cmp x22, #16
    b.hs 1f
    mov x22, x22
    b 2f
1:
    mov x22, #16
2:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x24, x23

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    mov x26, #28
    mov x27, #8

offset_loop:
    lsr x28, x21, x26
    and x28, x28, #0xf
    add x29, x25, x28
    ldrb w29, [x29]
    strb w29, [x24], #1

    subs x27, x27, #1
    b.eq 3f
    sub x26, x26, #4
    b offset_loop

3:
    mov w29, #0x20
    strb w29, [x24], #1
    strb w29, [x24], #1

    mov x9, #0

hex_loop:
    cmp x9, #16
    b.hs hex_done

    cmp x9, x22
    b.ge hex_blank

    add x10, x19, x21
    add x10, x10, x9
    ldrb w11, [x10]

    lsr w12, w11, #4
    and w12, w12, #0xf
    add x13, x25, x12
    ldrb w14, [x13]
    strb w14, [x24], #1

    and w12, w11, #0xf
    add x13, x25, x12
    ldrb w14, [x13]
    strb w14, [x24], #1

    mov w14, #0x20
    strb w14, [x24], #1
    b hex_after

hex_blank:
    mov w14, #0x20
    strb w14, [x24], #1
    strb w14, [x24], #1
    strb w14, [x24], #1

hex_after:
    cmp x9, #7
    b.ne 4f
    mov w14, #0x20
    strb w14, [x24], #1

4:
    add x9, x9, #1
    b hex_loop

hex_done:
    mov w14, #0x20
    strb w14, [x24], #1
    mov w14, #0x7c
    strb w14, [x24], #1

    mov x9, #0

char_loop:
    cmp x9, x22
    b.hs char_done

    add x10, x19, x21
    add x10, x10, x9
    ldrb w11, [x10]

    cmp w11, #0x20
    b.lt dot_char
    cmp w11, #0x7e
    b.gt dot_char
    mov w14, w11
    b char_store

dot_char:
    mov w14, #0x2e

char_store:
    strb w14, [x24], #1
    add x9, x9, #1
    b char_loop

char_done:
    mov w14, #0x7c
    strb w14, [x24], #1
    mov w14, #0x0a
    strb w14, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64

write_loop:
    svc #0
    cbz x0, 5f
    sub x2, x2, x0
    cbz x2, 6f
    add x1, x1, x0
    b write_loop

5:
    b 6f

6:
    add x21, x21, x22
    cmp x21, x20
    b.lt row_loop

print_total:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x24, x23

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    mov x26, #28
    mov x27, #8

total_loop:
    lsr x28, x20, x26
    and x28, x28, #0xf
    add x29, x25, x28
    ldrb w29, [x29]
    strb w29, [x24], #1

    subs x27, x27, #1
    b.eq total_done
    sub x26, x26, #4
    b total_loop

total_done:
    mov w29, #0x0a
    strb w29, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64

total_write:
    svc #0
    cbz x0, 7f
    sub x2, x2, x0
    cbz x2, 8f
    add x1, x1, x0
    b total_write

7:
    b 8f

8:
    mov x0, #0
    mov x8, #93
    svc #0
