.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 4
input_buf:
    .space 4096
row_buf:
    .space 16
out_buf:
    .space 256

.text
.globl _start

emit_row:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    mov x3, x0
    mov x4, x1
    mov x5, x2

    adrp x6, out_buf
    add x6, x6, :lo12:out_buf
    mov x7, x6
    adrp x8, hex_digits
    add x8, x8, :lo12:hex_digits

    mov x9, #8
    mov x10, #28
emit_row_offset_loop:
    lsr x11, x3, x10
    and x11, x11, #0xf
    ldrb w12, [x8, x11]
    strb w12, [x7], #1
    subs x9, x9, #1
    sub x10, x10, #4
    b.ne emit_row_offset_loop

    mov w11, #0x20
    strb w11, [x7], #1
    strb w11, [x7], #1

    mov x9, #0
emit_row_slots:
    cmp x9, #16
    b.hs emit_row_after_slots

    cmp x9, x5
    b.ge emit_row_blank_slot

    ldrb w12, [x4, x9]
    lsr x11, x12, #4
    and x11, x11, #0xf
    ldrb w13, [x8, x11]
    strb w13, [x7], #1
    and x11, x12, #0xf
    ldrb w13, [x8, x11]
    strb w13, [x7], #1
    mov w13, #0x20
    strb w13, [x7], #1
    b emit_row_slot_done

emit_row_blank_slot:
    mov w13, #0x20
    strb w13, [x7], #1
    strb w13, [x7], #1
    strb w13, [x7], #1

emit_row_slot_done:
    cmp x9, #7
    b.ne emit_row_slot_continue
    mov w13, #0x20
    strb w13, [x7], #1

emit_row_slot_continue:
    add x9, x9, #1
    b emit_row_slots

emit_row_after_slots:
    mov w11, #0x20
    strb w11, [x7], #1
    mov w11, #0x7c
    strb w11, [x7], #1

    mov x9, #0
emit_row_char_loop:
    cmp x9, x5
    b.hs emit_row_after_chars

    ldrb w12, [x4, x9]
    cmp w12, #0x20
    b.lt emit_row_char_dot
    cmp w12, #0x7e
    b.gt emit_row_char_dot
    mov w11, w12
    b emit_row_char_store

emit_row_char_dot:
    mov w11, #0x2e

emit_row_char_store:
    strb w11, [x7], #1
    add x9, x9, #1
    b emit_row_char_loop

emit_row_after_chars:
    mov w11, #0x7c
    strb w11, [x7], #1
    mov w11, #0x0a
    strb w11, [x7], #1

    sub x2, x7, x6
    mov x0, #1
    mov x1, x6
    mov x8, #64
    svc #0

    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

emit_total:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    mov x2, x1
    adrp x3, hex_digits
    add x3, x3, :lo12:hex_digits
    mov x4, #8
    mov x5, #28

emit_total_digits:
    lsr x6, x0, x5
    and x6, x6, #0xf
    ldrb w7, [x3, x6]
    strb w7, [x2], #1
    subs x4, x4, #1
    sub x5, x5, #4
    b.ne emit_total_digits

    mov w7, #0x0a
    strb w7, [x2], #1

    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0

    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0

read_more:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #4096
    mov x8, #63
    svc #0
    mov x22, x0
    cbz x22, done_input

    mov x23, #0
    adrp x24, row_buf
    add x24, x24, :lo12:row_buf

process_chunk:
    cmp x23, x22
    b.hs read_more

    ldrb w25, [x1, x23]
    strb w25, [x24, x20]
    add x20, x20, #1
    add x19, x19, #1

    cmp x20, #16
    b.ne not_full_row

    stp x22, x23, [sp, #-16]!
    stp x24, x25, [sp, #-16]!
    mov x0, x21
    mov x1, x24
    mov x2, #16
    bl emit_row
    ldp x24, x25, [sp], #16
    ldp x22, x23, [sp], #16

    mov x21, x19
    mov x20, #0

not_full_row:
    add x23, x23, #1
    b process_chunk

done_input:
    cbz x20, print_total

    stp x22, x23, [sp, #-16]!
    stp x24, x25, [sp, #-16]!
    mov x0, x21
    adrp x1, row_buf
    add x1, x1, :lo12:row_buf
    mov x2, x20
    bl emit_row
    ldp x24, x25, [sp], #16
    ldp x22, x23, [sp], #16

print_total:
    mov x0, x19
    bl emit_total

    mov x0, #0
    mov x8, #93
    svc #0
