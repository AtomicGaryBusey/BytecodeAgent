.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 4
input_buf:
    .space 65536
    .align 4
line_buf:
    .space 128

.text
.globl _start

write_buf:
    mov x8, #64
    svc #0
    ret

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #65536
    b.hs read_done

    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x2, #65536
    sub x2, x2, x20

    mov x8, #63
    svc #0

    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    cmp x20, #0
    b.eq print_total

    mov x21, #0

print_rows:
    mov x22, x20
    sub x22, x22, x21
    cmp x22, #16
    b.lt row_len_done
    mov x22, #16

row_len_done:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x24, x23

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    mov x26, #28
    mov x27, #8

offset_loop:
    lsr x28, x21, x26
    and x28, x28, #0xf
    ldrb w29, [x25, x28]
    strb w29, [x24], #1

    subs x27, x27, #1
    b.eq offset_done
    sub x26, x26, #4
    b offset_loop

offset_done:
    mov w29, #0x20
    strb w29, [x24], #1
    strb w29, [x24], #1

    mov x9, #0

slot_loop:
    cmp x9, #16
    b.hs slots_done

    cmp x9, x22
    b.ge blank_slot

    add x10, x21, x9
    ldrb w11, [x19, x10]

    lsr w12, w11, #4
    and w12, w12, #0xf
    ldrb w13, [x25, w12]
    strb w13, [x24], #1

    and w12, w11, #0xf
    ldrb w13, [x25, w12]
    strb w13, [x24], #1

    mov w13, #0x20
    strb w13, [x24], #1
    b slot_after

blank_slot:
    mov w13, #0x20
    strb w13, [x24], #1
    strb w13, [x24], #1
    strb w13, [x24], #1

slot_after:
    cmp x9, #7
    b.ne slot_continue
    mov w13, #0x20
    strb w13, [x24], #1

slot_continue:
    add x9, x9, #1
    b slot_loop

slots_done:
    mov w13, #0x20
    strb w13, [x24], #1
    mov w13, #0x7c
    strb w13, [x24], #1

    mov x9, #0

char_loop:
    cmp x9, x22
    b.hs chars_done

    add x10, x21, x9
    ldrb w11, [x19, x10]

    cmp w11, #0x20
    b.lt dot_char
    cmp w11, #0x7e
    b.gt dot_char
    mov w13, w11
    b char_store

dot_char:
    mov w13, #0x2e

char_store:
    strb w13, [x24], #1
    add x9, x9, #1
    b char_loop

chars_done:
    mov w13, #0x7c
    strb w13, [x24], #1
    mov w13, #0x0a
    strb w13, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    bl write_buf

    add x21, x21, x22
    cmp x21, x20
    b.lt print_rows

print_total:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x24, x23

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    mov x26, #28
    mov x27, #8

total_loop:
    lsr x28, x20, x26
    and x28, x28, #0xf
    ldrb w29, [x25, x28]
    strb w29, [x24], #1

    subs x27, x27, #1
    b.eq total_done
    sub x26, x26, #4
    b total_loop

total_done:
    mov w29, #0x0a
    strb w29, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    bl write_buf

    mov x0, #0
    mov x8, #93
    svc #0
