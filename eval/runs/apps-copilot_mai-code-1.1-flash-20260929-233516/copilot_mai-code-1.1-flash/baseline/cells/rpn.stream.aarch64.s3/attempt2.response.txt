.text
.global _start

_start:
    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    bl read_all
    mov x19, x0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0

loop_lines:
    cmp x21, x19
    b.ge done_lines
    mov x22, x21

scan_line:
    cmp x22, x19
    b.ge final_chunk
    add x23, x20, x22
    ldrb w24, [x23]
    cmp w24, #10
    b.eq newline_hit
    add x22, x22, #1
    b scan_line

newline_hit:
    sub x1, x22, x21
    add x0, x20, x21
    bl process_line
    add x21, x22, #1
    b loop_lines

final_chunk:
    sub x1, x19, x21
    add x0, x20, x21
    bl process_line

done_lines:
    mov x8, #93
    mov x0, #0
    svc #0

read_all:
    mov x9, x0
    mov x10, #65536
    mov x11, #0

read_loop:
    mov x0, #0
    mov x1, x9
    mov x2, x10
    mov x8, #63
    svc #0
    mov x12, x0
    cmp x12, #0
    b.eq read_done
    add x9, x9, x12
    sub x10, x10, x12
    add x11, x11, x12
    cbz x10, read_done
    b read_loop

read_done:
    mov x0, x11
    ret

process_line:
    adrp x14, val_stack
    add x14, x14, :lo12:val_stack
    mov x2, #0
    mov x3, #0

parse_loop:
    cmp x3, x1
    b.ge end_parse
skip_sep:
    cmp x3, x1
    b.ge end_parse
    add x4, x0, x3
    ldrb w5, [x4]
    cmp w5, #0x20
    b.eq sep_next
    cmp w5, #0x09
    b.eq sep_next
    b scan_token

sep_next:
    add x3, x3, #1
    b skip_sep

scan_token:
    mov x4, x3

token_loop:
    cmp x3, x1
    b.ge token_done
    add x5, x0, x3
    ldrb w6, [x5]
    cmp w6, #0x20
    b.eq token_done
    cmp w6, #0x09
    b.eq token_done
    add x3, x3, #1
    b token_loop

token_done:
    sub x7, x3, x4
    cmp x7, #1
    b.ne parse_literal

    add x5, x0, x4
    ldrb w6, [x5]
    cmp w6, #'+'
    b.eq op_plus
    cmp w6, #'-'
    b.eq op_minus
    cmp w6, #'*'
    b.eq op_mul
    cmp w6, #'/'
    b.eq op_div
    cmp w6, #'%'
    b.eq op_mod

parse_literal:
    mov x13, #0
    add x9, x0, x4
    ldrb w8, [x9]
    cmp w8, #'-'
    b.ne lit_no_sign
    mov x13, #1
    add x9, x9, #1
    sub x7, x7, #1

lit_no_sign:
    cbz x7, invalid_line
    cmp x7, #18
    b.hi invalid_line
    mov x11, #0
    mov x10, #0

digit_loop:
    cmp x10, x7
    b.ge digit_done
    add x12, x9, x10
    ldrb w8, [x12]
    cmp w8, #'0'
    b.lt invalid_line
    cmp w8, #'9'
    b.gt invalid_line
    sub w8, w8, #'0'
    mov x12, #10
    madd x11, x11, x12, x8
    add x10, x10, #1
    b digit_loop

digit_done:
    cmp x13, #0
    b.eq push_value
    neg x11, x11

push_value:
    add x8, x14, x2, lsl #3
    str x11, [x8]
    add x2, x2, #1
    b parse_loop

op_plus:
    cmp x2, #2
    b.lt invalid_line
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x5, [x4]
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x6, [x4]
    add x6, x6, x5
    str x6, [x4]
    add x2, x2, #1
    b parse_loop

op_minus:
    cmp x2, #2
    b.lt invalid_line
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x5, [x4]
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x6, [x4]
    sub x6, x6, x5
    str x6, [x4]
    add x2, x2, #1
    b parse_loop

op_mul:
    cmp x2, #2
    b.lt invalid_line
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x5, [x4]
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x6, [x4]
    mul x6, x6, x5
    str x6, [x4]
    add x2, x2, #1
    b parse_loop

op_div:
    cmp x2, #2
    b.lt invalid_line
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x5, [x4]
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x6, [x4]
    cbz x5, invalid_line
    mov x8, #0x8000
    lsl x8, x8, #48
    cmp x6, x8
    b.ne 1f
    cmp x5, #-1
    b.eq invalid_line
1:
    sdiv x6, x6, x5
    str x6, [x4]
    add x2, x2, #1
    b parse_loop

op_mod:
    cmp x2, #2
    b.lt invalid_line
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x5, [x4]
    sub x2, x2, #1
    add x4, x14, x2, lsl #3
    ldr x6, [x4]
    cbz x5, invalid_line
    mov x8, #0x8000
    lsl x8, x8, #48
    cmp x6, x8
    b.ne 1f
    cmp x5, #-1
    b.eq invalid_line
1:
    sdiv x7, x6, x5
    msub x6, x7, x5, x6
    str x6, [x4]
    add x2, x2, #1
    b parse_loop

end_parse:
    cmp x2, #1
    b.ne invalid_line
    ldr x11, [x14]

    mov x10, x11
    mov x12, #0x8000
    lsl x12, x12, #48
    cmp x10, x12
    b.ne 1f
    mov x0, #1
    adrp x1, neg_min_str
    add x1, x1, :lo12:neg_min_str
    mov x2, #20
    mov x8, #64
    svc #0
    b write_newline
1:
    cmp x10, #0
    b.ge 2f
    mov x0, #1
    adrp x1, minus_char
    add x1, x1, :lo12:minus_char
    mov x2, #1
    mov x8, #64
    svc #0
    neg x10, x10
2:
    adrp x13, digit_buf
    add x13, x13, :lo12:digit_buf
    add x13, x13, #31
    mov x15, #0
    mov x16, x10

print_loop:
    cbz x16, print_digits_done
    mov x17, #10
    udiv x5, x16, x17
    msub x6, x5, x17, x16
    add w6, w6, #'0'
    sub x13, x13, #1
    strb w6, [x13]
    mov x16, x5
    add x15, x15, #1
    b print_loop

print_digits_done:
    cbz x15, print_zero
    mov x0, #1
    mov x1, x13
    mov x2, x15
    mov x8, #64
    svc #0
    b write_newline

print_zero:
    mov x0, #1
    adrp x1, zero_char
    add x1, x1, :lo12:zero_char
    mov x2, #1
    mov x8, #64
    svc #0

write_newline:
    mov x0, #1
    adrp x1, newline_msg
    add x1, x1, :lo12:newline_msg
    mov x2, #1
    mov x8, #64
    svc #0
    ret

invalid_line:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0
    ret

.section .rodata
minus_char:
    .byte 0x2d
zero_char:
    .byte 0x30
newline_msg:
    .byte 0x0a
error_msg:
    .ascii "error\n"
neg_min_str:
    .ascii "-9223372036854775808"

.section .bss
    .align 8
input_buf:
    .space 65536
val_stack:
    .space 8192
digit_buf:
    .space 32
