.text

.globl _start
_start:
    call read_all
    call parse_document
    test rax, rax
    jnz _start_ok
    mov rdi, qword ptr [rip + pos]
    call print_error
    mov edi, 1
    mov eax, 60
    syscall
_start_ok:
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

read_all:
    xor r12, r12
read_all_loop:
    mov rax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_all_done
    syscall
    test rax, rax
    jle read_all_done
    add r12, rax
    jmp read_all_loop
read_all_done:
    mov qword ptr [rip + input_len], r12
    ret

parse_document:
    mov qword ptr [rip + pos], 0
    call parse_ws
    call parse_value
    test rax, rax
    jz parse_document_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne parse_document_fail
    mov rax, 1
    ret
parse_document_fail:
    xor rax, rax
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
parse_ws_loop:
    cmp rax, rbx
    jae parse_ws_done
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, 0x20
    je parse_ws_advance
    cmp cl, 0x09
    je parse_ws_advance
    cmp cl, 0x0A
    je parse_ws_advance
    cmp cl, 0x0D
    je parse_ws_advance
    jmp parse_ws_done
parse_ws_advance:
    inc rax
    jmp parse_ws_loop
parse_ws_done:
    mov qword ptr [rip + pos], rax
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_value_eof_fail
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '{'
    je parse_object
    cmp cl, '['
    je parse_array
    cmp cl, '"'
    je parse_string
    cmp cl, '-'
    je parse_number
    cmp cl, '0'
    jb parse_value_bad
    cmp cl, '9'
    jbe parse_number
    cmp cl, 't'
    je parse_true
    cmp cl, 'f'
    je parse_false
    cmp cl, 'n'
    je parse_null
parse_value_bad:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_value_eof_fail:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae parse_object_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '{'
    jne parse_object_bad
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_object_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '}'
    je parse_object_close
parse_object_loop:
    call parse_string
    test rax, rax
    jz parse_object_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_object_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, ':'
    jne parse_object_bad_at
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    call parse_value
    test rax, rax
    jz parse_object_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_object_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, ','
    je parse_object_comma
    cmp cl, '}'
    je parse_object_close
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_object_comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_object_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '}'
    je parse_object_bad_at
    jmp parse_object_loop
parse_object_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_object_bad:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_object_bad_at:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_object_incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_object_fail:
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae parse_array_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '['
    jne parse_array_bad
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_array_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, ']'
    je parse_array_close
parse_array_loop:
    call parse_value
    test rax, rax
    jz parse_array_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_array_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, ','
    je parse_array_comma
    cmp cl, ']'
    je parse_array_close
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_array_comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_array_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, ']'
    je parse_array_bad_at
    jmp parse_array_loop
parse_array_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_array_bad:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_array_bad_at:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_array_incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_array_fail:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae parse_string_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '"'
    jne parse_string_fail
    inc rax
    mov qword ptr [rip + pos], rax
parse_string_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_string_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '"'
    je parse_string_done
    cmp cl, '\\'
    je parse_string_escape
    cmp cl, 0x20
    jb parse_string_fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
    jmp parse_string_loop
parse_string_escape:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_string_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '"'
    je parse_string_advance
    cmp cl, '\\'
    je parse_string_advance
    cmp cl, '/'
    je parse_string_advance
    cmp cl, 'b'
    je parse_string_advance
    cmp cl, 'f'
    je parse_string_advance
    cmp cl, 'n'
    je parse_string_advance
    cmp cl, 'r'
    je parse_string_advance
    cmp cl, 't'
    je parse_string_advance
    cmp cl, 'u'
    jne parse_string_fail_at_pos
    mov rdx, rax
    inc rdx
    mov r8, 4
parse_string_hex_loop:
    cmp rdx, qword ptr [rip + input_len]
    jae parse_string_incomplete
    movzx edi, byte ptr [rip + input_buf + rdx]
    cmp dil, '0'
    jb parse_string_fail_at_rdx
    cmp dil, '9'
    jbe parse_string_hex_ok
    cmp dil, 'a'
    jb parse_string_fail_at_rdx
    cmp dil, 'f'
    jbe parse_string_hex_ok
    cmp dil, 'A'
    jb parse_string_fail_at_rdx
    cmp dil, 'F'
    jbe parse_string_hex_ok
    jmp parse_string_fail_at_rdx
parse_string_hex_ok:
    inc rdx
    dec r8
    jnz parse_string_hex_loop
    mov rax, rdx
    mov qword ptr [rip + pos], rax
    jmp parse_string_loop
parse_string_advance:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp parse_string_loop
parse_string_done:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_string_incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_string_fail:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_string_fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_string_fail_at_rdx:
    mov qword ptr [rip + pos], rdx
    xor rax, rax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '-'
    jne parse_number_int
    inc rax
    cmp rax, rbx
    je parse_number_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
parse_number_int:
    cmp cl, '0'
    je parse_number_zero
    cmp cl, '1'
    jb parse_number_bad
    cmp cl, '9'
    jbe parse_number_int_digits
    jmp parse_number_bad
parse_number_zero:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_number_after_int
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '0'
    jb parse_number_after_int
    cmp cl, '9'
    jbe parse_number_bad_at_pos
    jmp parse_number_after_int
parse_number_int_digits:
    inc rax
    mov qword ptr [rip + pos], rax
parse_number_int_digits_loop:
    cmp rax, qword ptr [rip + input_len]
    je parse_number_after_int
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '0'
    jb parse_number_after_int
    cmp cl, '9'
    jbe parse_number_int_digits_advance
    jmp parse_number_after_int
parse_number_int_digits_advance:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp parse_number_int_digits_loop
parse_number_after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_number_ok
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '.'
    je parse_number_frac
    cmp cl, 'e'
    je parse_number_exp
    cmp cl, 'E'
    je parse_number_exp
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_number_frac:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_number_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '0'
    jb parse_number_bad_at_pos
    cmp cl, '9'
    jbe parse_number_frac_digits
    jmp parse_number_bad_at_pos
parse_number_frac_digits:
    inc rax
    mov qword ptr [rip + pos], rax
parse_number_frac_digits_loop:
    cmp rax, qword ptr [rip + input_len]
    je parse_number_frac_after
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '0'
    jb parse_number_frac_after
    cmp cl, '9'
    jbe parse_number_frac_digits_advance
    jmp parse_number_frac_after
parse_number_frac_digits_advance:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp parse_number_frac_digits_loop
parse_number_frac_after:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    je parse_number_ok
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, 'e'
    je parse_number_exp
    cmp cl, 'E'
    je parse_number_exp
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_number_exp:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_number_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '+'
    je parse_number_exp_sign
    cmp cl, '-'
    je parse_number_exp_sign
    jmp parse_number_exp_digits_start
parse_number_exp_sign:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_number_incomplete
    movzx ecx, byte ptr [rip + input_buf + rax]
parse_number_exp_digits_start:
    cmp cl, '0'
    jb parse_number_bad_at_pos
    cmp cl, '9'
    jbe parse_number_exp_digits_loop
    jmp parse_number_bad_at_pos
parse_number_exp_digits_loop:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    je parse_number_ok
    movzx ecx, byte ptr [rip + input_buf + rax]
    cmp cl, '0'
    jb parse_number_ok
    cmp cl, '9'
    jbe parse_number_exp_digits_loop
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
parse_number_bad_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_number_bad:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_number_incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
parse_number_ok:
    mov rax, 1
    ret

match_literal:
    mov r8, qword ptr [rip + pos]
    mov r9, qword ptr [rip + input_len]
    xor rcx, rcx
match_literal_loop:
    cmp rcx, rsi
    je match_literal_ok
    cmp r8, r9
    je match_literal_incomplete
    movzx eax, byte ptr [rip + input_buf + r8]
    movzx edx, byte ptr [rdi + rcx]
    cmp al, dl
    jne match_literal_bad
    inc r8
    inc rcx
    jmp match_literal_loop
match_literal_ok:
    mov qword ptr [rip + pos], r8
    mov rax, 1
    ret
match_literal_bad:
    mov qword ptr [rip + pos], r8
    xor rax, rax
    ret
match_literal_incomplete:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_true:
    lea rdi, [rip + true_lit]
    mov esi, 4
    jmp match_literal

parse_false:
    lea rdi, [rip + false_lit]
    mov esi, 5
    jmp match_literal

parse_null:
    lea rdi, [rip + null_lit]
    mov esi, 4
    jmp match_literal

number_to_ascii:
    lea r8, [rip + dec_buf + 31]
    mov byte ptr [r8], 0
    mov rax, rdi
    test rax, rax
    jz number_to_ascii_zero
number_to_ascii_loop:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz number_to_ascii_loop
    jmp number_to_ascii_done
number_to_ascii_zero:
    dec r8
    mov byte ptr [r8], '0'
number_to_ascii_done:
    mov rax, r8
    lea rdx, [rip + dec_buf + 31]
    sub rdx, r8
    ret

print_error:
    mov r12, rdx
    call number_to_ascii
    mov r13, rax
    mov r14, rdx
    lea rsi, [rip + err_prefix]
    mov edx, err_prefix_len
    mov edi, 1
    mov eax, 1
    syscall
    mov rsi, r13
    mov rdx, r14
    mov edi, 1
    mov eax, 1
    syscall
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    ret

.section .rodata
ok_msg:
    .asciz "ok\n"
ok_msg_len = . - ok_msg

err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix
newline:
    .byte 10

true_lit:
    .ascii "true"
false_lit:
    .ascii "false"
null_lit:
    .ascii "null"

.section .bss
.align 16
input_buf:
    .space 65536
input_len:
    .quad 0
pos:
    .quad 0
dec_buf:
    .space 32
