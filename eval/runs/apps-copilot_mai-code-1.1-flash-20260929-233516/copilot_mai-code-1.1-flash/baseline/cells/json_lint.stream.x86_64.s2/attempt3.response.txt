.text

.globl _start
_start:
    call read_all
    call parse_document
    test rax, rax
    jnz .ok
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + err_prefix]
    mov rdx, err_prefix_len
    mov edi, 1
    mov eax, 1
    syscall
    mov rax, qword ptr [rip + pos]
    call print_dec
    lea rsi, [rip + newline]
    mov rdx, 1
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
.ok:
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

read_all:
    xor r12, r12
.loop:
    mov rdx, 65536
    sub rdx, r12
    jz .done
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edi, 0
    xor eax, eax
    syscall
    test rax, rax
    jle .done
    add r12, rax
    jmp .loop
.done:
    mov qword ptr [rip + input_len], r12
    ret

parse_document:
    mov qword ptr [rip + pos], 0
    call parse_ws
    call parse_value
    test rax, rax
    jz .fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jne .fail_trailing
    mov rax, 1
    ret
.fail_trailing:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail:
    xor rax, rax
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
.loop:
    cmp rax, rbx
    jae .done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je .inc
    cmp cl, 0x09
    je .inc
    cmp cl, 0x0A
    je .inc
    cmp cl, 0x0D
    je .inc
    jmp .done
.inc:
    inc rax
    jmp .loop
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae .fail
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '{'
    je .do_object
    cmp cl, '['
    je .do_array
    cmp cl, '"'
    je .do_string
    cmp cl, '-'
    je .do_number
    cmp cl, '0'
    je .do_number
    cmp cl, '1'
    jb .fail
    cmp cl, '9'
    jbe .do_number
    cmp cl, 't'
    je .do_true
    cmp cl, 'f'
    je .do_false
    cmp cl, 'n'
    je .do_null
    jmp .fail
.do_object:
    call parse_object
    ret
.do_array:
    call parse_array
    ret
.do_string:
    call parse_string
    ret
.do_number:
    call parse_number
    ret
.do_true:
    call parse_true
    ret
.do_false:
    call parse_false
    ret
.do_null:
    call parse_null
    ret
.fail:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp rax, rbx
    jae .fail_end
    cmp byte ptr [rsi + rax], '{'
    jne .fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '}'
    je .close
    jmp .member
.member:
    call parse_string
    test rax, rax
    jz .ret_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ':'
    jne .fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    call parse_value
    test rax, rax
    jz .ret_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    je .comma
    cmp cl, '}'
    je .close
    jmp .fail_at_pos
.comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '}'
    je .fail_trailing
    jmp .member
.close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_trailing:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.ret_fail:
    xor rax, rax
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp rax, rbx
    jae .fail_end
    cmp byte ptr [rsi + rax], '['
    jne .fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ']'
    je .close
    jmp .value
.value:
    call parse_value
    test rax, rax
    jz .ret_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ','
    je .comma
    cmp cl, ']'
    je .close
    jmp .fail_at_pos
.comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, ']'
    je .fail_trailing
    jmp .value
.close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_trailing:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.ret_fail:
    xor rax, rax
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp rax, rbx
    jae .fail_end
    cmp byte ptr [rsi + rax], '"'
    jne .fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
.loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je .done
    cmp cl, '\\'
    je .escape
    cmp cl, 0x20
    jb .fail_at_pos
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .loop
.escape:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '"'
    je .advance
    cmp cl, '\\'
    je .advance
    cmp cl, '/'
    je .advance
    cmp cl, 'b'
    je .advance
    cmp cl, 'f'
    je .advance
    cmp cl, 'n'
    je .advance
    cmp cl, 'r'
    je .advance
    cmp cl, 't'
    je .advance
    cmp cl, 'u'
    je .escape_u
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.advance:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .loop
.escape_u:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rcx, 4
.hex_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .hex_fail
    cmp dl, '9'
    jbe .hex_ok
    cmp dl, 'a'
    jb .hex_fail
    cmp dl, 'f'
    jbe .hex_ok
    cmp dl, 'A'
    jb .hex_fail
    cmp dl, 'F'
    jbe .hex_ok
    jmp .hex_fail
.hex_ok:
    inc rax
    mov qword ptr [rip + pos], rax
    dec rcx
    jnz .hex_loop
    jmp .loop
.hex_fail:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.done:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    mov rcx, 0
.loop:
    cmp rcx, 4
    je .done
    cmp rax, rbx
    jae .fail_end
    mov dl, byte ptr [rsi + rax]
    cmp dl, byte ptr [rip + true_literal + rcx]
    jne .fail_at_pos
    inc rax
    inc rcx
    jmp .loop
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_false:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    mov rcx, 0
.loop:
    cmp rcx, 5
    je .done
    cmp rax, rbx
    jae .fail_end
    mov dl, byte ptr [rsi + rax]
    cmp dl, byte ptr [rip + false_literal + rcx]
    jne .fail_at_pos
    inc rax
    inc rcx
    jmp .loop
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_null:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    mov rcx, 0
.loop:
    cmp rcx, 4
    je .done
    cmp rax, rbx
    jae .fail_end
    mov dl, byte ptr [rsi + rax]
    cmp dl, byte ptr [rip + null_literal + rcx]
    jne .fail_at_pos
    inc rax
    inc rcx
    jmp .loop
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_pos:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    mov rbx, qword ptr [rip + input_len]
    lea rsi, [rip + input_buf]
    cmp rax, rbx
    jae .fail_end
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '-'
    jne .after_sign
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, rbx
    jae .fail_end
    movzx edx, byte ptr [rsi + rax]
.after_sign:
    cmp dl, '0'
    jne .nonzero
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, rbx
    jb .after_zero_check
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.after_zero_check:
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .after_zero_cont
    cmp dl, '9'
    jbe .fail_current
.after_zero_cont:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '.'
    je .frac
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.nonzero:
    cmp dl, '1'
    jb .fail_current
    cmp dl, '9'
    ja .fail_current
    jmp .int_digits
.int_digits:
    inc rax
    mov qword ptr [rip + pos], rax
.int_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_int
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .after_int
    cmp dl, '9'
    ja .after_int
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .int_loop
.after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '.'
    je .frac
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.frac:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .fail_current
    cmp dl, '9'
    ja .fail_current
    jmp .frac_digits
.frac_digits:
    inc rax
    mov qword ptr [rip + pos], rax
.frac_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_frac
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .after_frac
    cmp dl, '9'
    ja .after_frac
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .frac_loop
.after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx edx, byte ptr [rsi + rax]
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.exp:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '+'
    je .exp_sign
    cmp dl, '-'
    je .exp_sign
    jmp .exp_digits
.exp_sign:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_end
.exp_digits:
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .fail_current
    cmp dl, '9'
    ja .fail_current
    jmp .exp_num_loop
.exp_num_loop:
    inc rax
    mov qword ptr [rip + pos], rax
.exp_num_inner:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx edx, byte ptr [rsi + rax]
    cmp dl, '0'
    jb .done
    cmp dl, '9'
    ja .done
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .exp_num_inner
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_current:
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret
.fail_end:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + pos], rax
    xor rax, rax
    ret

print_dec:
    lea r8, [rip + dec_tmp + 31]
    mov byte ptr [r8], 0
    mov rcx, rax
    cmp rcx, 0
    jne .loop
    dec r8
    mov byte ptr [r8], '0'
    mov r9, 1
    jmp .write
.loop:
    xor rdx, rdx
    mov rax, rcx
    mov rbx, 10
    div rbx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    inc r9
    mov rcx, rax
    test rcx, rcx
    jnz .loop
.write:
    mov rsi, r8
    mov rdx, r9
    mov edi, 1
    mov eax, 1
    syscall
    ret
