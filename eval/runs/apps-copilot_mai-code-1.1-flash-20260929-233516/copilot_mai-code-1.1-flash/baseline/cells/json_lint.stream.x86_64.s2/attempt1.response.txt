.text

.globl _start
_start:
    call read_all
    xor edi, edi
    call parse_document
    test r8, r8
    jz _valid
    mov rdi, rax
    call print_error
    mov edi, 1
    jmp exit_status

_valid:
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    mov eax, 1
    mov edi, 1
    syscall
    xor edi, edi
    jmp exit_status

exit_status:
    mov eax, 60
    syscall

read_all:
    xor rbx, rbx
read_loop:
    mov rdx, 65536
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], rbx
    ret

parse_document:
    call parse_ws
    mov rdi, rax
    call parse_value
    test r8, r8
    jnz parse_document_fail
    mov rdi, rax
    call parse_ws
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jne parse_document_fail
    xor r8, r8
    ret

parse_document_fail:
    mov r8, 1
    ret

parse_ws:
parse_ws_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp rdi, rdx
    jae parse_ws_done
    movzx eax, byte ptr [rip + input_buf + rdi]
    cmp al, 0x20
    je parse_ws_advance
    cmp al, 0x09
    je parse_ws_advance
    cmp al, 0x0A
    je parse_ws_advance
    cmp al, 0x0D
    je parse_ws_advance
    jmp parse_ws_done
parse_ws_advance:
    inc rdi
    jmp parse_ws_loop
parse_ws_done:
    mov rax, rdi
    xor r8, r8
    ret

parse_value:
    mov rdx, qword ptr [rip + input_len]
    cmp rdi, rdx
    jae parse_value_fail_end
    movzx eax, byte ptr [rip + input_buf + rdi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_fail
    cmp al, '9'
    jbe parse_number
    cmp al, 't'
    je parse_literal_true
    cmp al, 'f'
    je parse_literal_false
    cmp al, 'n'
    je parse_literal_null
parse_value_fail:
    mov rax, rdi
    mov r8, 1
    ret
parse_value_fail_end:
    mov rax, rdx
    mov r8, 1
    ret

parse_object:
    mov r9, rdi
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_object_fail_end
    cmp byte ptr [rip + input_buf + r9], '{'
    jne parse_object_fail_cur
    inc r9
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_object_fail_end
    cmp byte ptr [rip + input_buf + r9], '}'
    je parse_object_close
parse_object_member_loop:
    mov rdi, r9
    call parse_string
    test r8, r8
    jnz parse_object_return_fail
    mov rdi, rax
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_object_fail_end
    cmp byte ptr [rip + input_buf + r9], ':'
    jne parse_object_fail_cur
    inc r9
    mov rdi, r9
    call parse_ws
    mov rdi, rax
    call parse_value
    test r8, r8
    jnz parse_object_return_fail
    mov r9, rax
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_object_fail_end
    cmp byte ptr [rip + input_buf + r9], ','
    jne parse_object_check_close
    inc r9
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_object_fail_end
    jmp parse_object_member_loop

parse_object_check_close:
    cmp byte ptr [rip + input_buf + r9], '}'
    jne parse_object_fail_cur
    inc r9
    mov rdi, r9
    call parse_ws
    mov rax, rax
    xor r8, r8
    ret

parse_object_close:
    inc r9
    mov rdi, r9
    call parse_ws
    mov rax, rax
    xor r8, r8
    ret

parse_object_fail_cur:
    mov rax, r9
    mov r8, 1
    ret

parse_object_fail_end:
    mov rax, qword ptr [rip + input_len]
    mov r8, 1
    ret

parse_object_return_fail:
    ret

parse_array:
    mov r9, rdi
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_array_fail_end
    cmp byte ptr [rip + input_buf + r9], '['
    jne parse_array_fail_cur
    inc r9
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_array_fail_end
    cmp byte ptr [rip + input_buf + r9], ']'
    je parse_array_close
parse_array_value_loop:
    mov rdi, r9
    call parse_value
    test r8, r8
    jnz parse_array_return_fail
    mov r9, rax
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_array_fail_end
    cmp byte ptr [rip + input_buf + r9], ','
    jne parse_array_check_close
    inc r9
    mov rdi, r9
    call parse_ws
    mov r9, rax
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_array_fail_end
    jmp parse_array_value_loop

parse_array_check_close:
    cmp byte ptr [rip + input_buf + r9], ']'
    jne parse_array_fail_cur
    inc r9
    mov rdi, r9
    call parse_ws
    mov rax, rax
    xor r8, r8
    ret

parse_array_close:
    inc r9
    mov rdi, r9
    call parse_ws
    mov rax, rax
    xor r8, r8
    ret

parse_array_fail_cur:
    mov rax, r9
    mov r8, 1
    ret

parse_array_fail_end:
    mov rax, qword ptr [rip + input_len]
    mov r8, 1
    ret

parse_array_return_fail:
    ret

parse_string:
    mov r9, rdi
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_string_fail_end
    cmp byte ptr [rip + input_buf + r9], '"'
    jne parse_string_fail_cur
    inc r9
parse_string_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_string_fail_end
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '"'
    je parse_string_ok
    cmp al, '\\'
    je parse_string_escape
    cmp al, 0x20
    jb parse_string_fail_cur
    inc r9
    jmp parse_string_loop

parse_string_escape:
    inc r9
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_string_fail_end
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '"'
    je parse_string_escape_ok
    cmp al, '\\'
    je parse_string_escape_ok
    cmp al, '/'
    je parse_string_escape_ok
    cmp al, 'b'
    je parse_string_escape_ok
    cmp al, 'f'
    je parse_string_escape_ok
    cmp al, 'n'
    je parse_string_escape_ok
    cmp al, 'r'
    je parse_string_escape_ok
    cmp al, 't'
    je parse_string_escape_ok
    cmp al, 'u'
    jne parse_string_fail_cur
    mov r10, r9
    inc r10
    mov r11, 4
parse_string_unicode_loop:
    cmp r10, qword ptr [rip + input_len]
    jae parse_string_fail_end
    movzx ecx, byte ptr [rip + input_buf + r10]
    cmp cl, '0'
    jb parse_string_unicode_fail
    cmp cl, '9'
    jbe parse_string_unicode_digit_ok
    cmp cl, 'a'
    jb parse_string_unicode_fail
    cmp cl, 'f'
    jbe parse_string_unicode_digit_ok
    cmp cl, 'A'
    jb parse_string_unicode_fail
    cmp cl, 'F'
    jbe parse_string_unicode_digit_ok
    jmp parse_string_unicode_fail
parse_string_unicode_digit_ok:
    inc r10
    dec r11
    jnz parse_string_unicode_loop
    mov r9, r10
    jmp parse_string_loop
parse_string_unicode_fail:
    mov rax, r10
    mov r8, 1
    ret

parse_string_escape_ok:
    inc r9
    jmp parse_string_loop

parse_string_ok:
    mov rax, r9
    xor r8, r8
    ret

parse_string_fail_cur:
    mov rax, r9
    mov r8, 1
    ret

parse_string_fail_end:
    mov rax, qword ptr [rip + input_len]
    mov r8, 1
    ret

match_literal:
    mov r9, rdi
    mov r10, rsi
    mov r11, rdx
match_literal_loop:
    test r11, r11
    jz match_literal_ok
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae match_literal_fail_end
    movzx eax, byte ptr [rip + input_buf + r9]
    movzx ecx, byte ptr [r10]
    cmp al, cl
    jne match_literal_fail_cur
    inc r9
    inc r10
    dec r11
    jmp match_literal_loop
match_literal_ok:
    mov rax, r9
    xor r8, r8
    ret
match_literal_fail_cur:
    mov rax, r9
    mov r8, 1
    ret
match_literal_fail_end:
    mov rax, qword ptr [rip + input_len]
    mov r8, 1
    ret

parse_literal_true:
    lea rsi, [rip + true_literal]
    mov rdx, true_literal_len
    call match_literal
    ret

parse_literal_false:
    lea rsi, [rip + false_literal]
    mov rdx, false_literal_len
    call match_literal
    ret

parse_literal_null:
    lea rsi, [rip + null_literal]
    mov rdx, null_literal_len
    call match_literal
    ret

parse_number:
    mov r9, rdi
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_fail_rdi
    cmp byte ptr [rip + input_buf + r9], '-'
    jne parse_number_after_minus
    inc r9
    cmp r9, rdx
    jae parse_number_fail_rdi
parse_number_after_minus:
    cmp r9, rdx
    jae parse_number_fail_rdi
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    je parse_number_zero
    cmp al, '1'
    jb parse_number_fail_rdi
    cmp al, '9'
    jbe parse_number_nonzero
    jmp parse_number_fail_rdi

parse_number_zero:
    inc r9
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_frac_exp
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_frac_exp
    cmp al, '9'
    jbe parse_number_fail_rdi
    jmp parse_number_frac_exp

parse_number_nonzero:
    inc r9
parse_number_nonzero_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_frac_exp
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_frac_exp
    cmp al, '9'
    jbe parse_number_nonzero_inc
    jmp parse_number_frac_exp
parse_number_nonzero_inc:
    inc r9
    jmp parse_number_nonzero_loop

parse_number_frac_exp:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_ok
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '.'
    jne parse_number_exponent
    inc r9
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_fail_rdi
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_fail_rdi
    cmp al, '9'
    jbe parse_number_frac_digits
    jmp parse_number_fail_rdi

parse_number_frac_digits:
    inc r9
parse_number_frac_digits_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_exponent
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_exponent
    cmp al, '9'
    jbe parse_number_frac_digits_inc
    jmp parse_number_exponent
parse_number_frac_digits_inc:
    inc r9
    jmp parse_number_frac_digits_loop

parse_number_exponent:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_ok
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, 'e'
    je parse_number_exp
    cmp al, 'E'
    je parse_number_exp
    jmp parse_number_ok

parse_number_exp:
    inc r9
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_fail_rdi
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '+'
    je parse_number_exp_sign
    cmp al, '-'
    je parse_number_exp_sign
    jmp parse_number_exp_digits_start
parse_number_exp_sign:
    inc r9
parse_number_exp_digits_start:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_fail_rdi
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_fail_rdi
    cmp al, '9'
    jbe parse_number_exp_digits
    jmp parse_number_fail_rdi
parse_number_exp_digits:
    inc r9
parse_number_exp_digits_loop:
    mov rdx, qword ptr [rip + input_len]
    cmp r9, rdx
    jae parse_number_ok
    movzx eax, byte ptr [rip + input_buf + r9]
    cmp al, '0'
    jb parse_number_ok
    cmp al, '9'
    jbe parse_number_exp_digits_inc
    jmp parse_number_ok
parse_number_exp_digits_inc:
    inc r9
    jmp parse_number_exp_digits_loop

parse_number_ok:
    mov rax, r9
    xor r8, r8
    ret

parse_number_fail_rdi:
    mov rax, r9
    mov r8, 1
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, err_prefix_len
    syscall

    mov rax, rdi
    lea r9, [rip + numbuf + 31]
    mov byte ptr [r9], 10
print_error_loop:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz print_error_loop

    lea r10, [rip + numbuf + 31]
    mov rdx, r10
    sub rdx, r9
    inc rdx
    lea rsi, [r9]
    mov rax, 1
    mov rdi, 1
    syscall
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg
err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix
true_literal:
    .ascii "true"
true_literal_len = . - true_literal
false_literal:
    .ascii "false"
false_literal_len = . - false_literal
null_literal:
    .ascii "null"
null_literal_len = . - null_literal

.bss
.align 16
input_buf:
    .space 65536
.align 8
input_len:
    .quad 0
.align 16
numbuf:
    .space 32
