.text

.globl _start
_start:
    call read_all
    call parse_document
    cmp qword ptr [rip + parse_status], 1
    jne .valid
    mov rax, qword ptr [rip + err_offset]
    call print_error
.valid:
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    mov eax, 1
    mov edi, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

read_all:
    xor r12, r12
.read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz .done
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle .done
    add r12, rax
    jmp .read_loop
.done:
    mov qword ptr [rip + input_len], r12
    ret

parse_document:
    mov qword ptr [rip + parse_status], 0
    mov qword ptr [rip + err_offset], 0
    mov qword ptr [rip + pos], 0
    call parse_ws
    call parse_value
    cmp qword ptr [rip + parse_status], 1
    jne 1f
    mov rax, qword ptr [rip + err_offset]
    ret
1:
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne .fail
    xor rax, rax
    ret
.fail:
    mov qword ptr [rip + err_offset], rax
    mov qword ptr [rip + parse_status], 1
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
.ws_loop:
    cmp rax, rdx
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 0x20
    je .inc
    cmp cl, 0x09
    je .inc
    cmp cl, 0x0A
    je .inc
    cmp cl, 0x0D
    je .inc
    jmp .done
.inc:
    inc rax
    jmp .ws_loop
.done:
    mov qword ptr [rip + pos], rax
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '{'
    je parse_object
    cmp cl, '['
    je parse_array
    cmp cl, '"'
    je parse_string
    cmp cl, '-'
    je parse_number
    cmp cl, '0'
    jb fail_at_current
    cmp cl, '9'
    jbe parse_number
    cmp cl, 't'
    je parse_true
    cmp cl, 'f'
    je parse_false
    cmp cl, 'n'
    je parse_null
    jmp fail_at_current

parse_object:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '}'
    je .close
.member_loop:
    call parse_string
    cmp qword ptr [rip + parse_status], 1
    je .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, ':'
    jne fail_at_current
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    cmp qword ptr [rip + parse_status], 1
    je .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '}'
    je .close
    cmp cl, ','
    jne fail_at_current
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '}'
    je fail_at_current
    jmp .member_loop
.close:
    inc qword ptr [rip + pos]
    ret
.done:
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, ']'
    je .close
.loop:
    call parse_value
    cmp qword ptr [rip + parse_status], 1
    je .done
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, ']'
    je .close
    cmp cl, ','
    jne fail_at_current
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, ']'
    je fail_at_current
    jmp .loop
.close:
    inc qword ptr [rip + pos]
    ret
.done:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '"'
    jne fail_at_current
    inc qword ptr [rip + pos]
.str_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '"'
    je .close
    cmp cl, 0x5C
    je .escape
    cmp cl, 0x20
    jb fail_at_current
    inc qword ptr [rip + pos]
    jmp .str_loop
.escape:
    mov rax, qword ptr [rip + pos]
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '"'
    je .advance
    cmp cl, 0x5C
    je .advance
    cmp cl, '/'
    je .advance
    cmp cl, 'b'
    je .advance
    cmp cl, 'f'
    je .advance
    cmp cl, 'n'
    je .advance
    cmp cl, 'r'
    je .advance
    cmp cl, 't'
    je .advance
    cmp cl, 'u'
    je .unicode
    mov qword ptr [rip + pos], rax
    call fail_at_current
    ret
.advance:
    add qword ptr [rip + pos], 2
    jmp .str_loop
.unicode:
    mov r8, qword ptr [rip + pos]
    add r8, 2
    mov rcx, 4
.unicode_loop:
    mov rax, r8
    cmp rax, qword ptr [rip + input_len]
    jae .unicode_fail
    lea rbx, [rip + input_buf]
    movzx edx, byte ptr [rbx + rax]
    cmp dl, '0'
    jb .unicode_fail
    cmp dl, '9'
    jbe .hex_ok
    cmp dl, 'a'
    jb .unicode_fail
    cmp dl, 'f'
    jbe .hex_ok
    cmp dl, 'A'
    jb .unicode_fail
    cmp dl, 'F'
    ja .unicode_fail
.hex_ok:
    inc r8
    dec rcx
    jnz .unicode_loop
    mov qword ptr [rip + pos], r8
    jmp .str_loop
.unicode_fail:
    mov qword ptr [rip + pos], rax
    call fail_at_current
    ret
.close:
    inc qword ptr [rip + pos]
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '-'
    jne .int_start
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb fail_at_current
    cmp cl, '9'
    ja fail_at_current
.int_start:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jne .nonzero
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    ret
.nonzero:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb .after_int
    cmp cl, '9'
    ja .after_int
    inc qword ptr [rip + pos]
    jmp .nonzero
.after_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    ret
.frac:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb fail_at_current
    cmp cl, '9'
    ja fail_at_current
.frac_digits:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_frac
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb .after_frac
    cmp cl, '9'
    ja .after_frac
    jmp .frac_digits
.after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    ret
.exp:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '+'
    je .exp_sign
    cmp cl, '-'
    je .exp_sign
    cmp cl, '0'
    jb fail_at_current
    cmp cl, '9'
    ja fail_at_current
.exp_digits:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb .done
    cmp cl, '9'
    ja .done
    jmp .exp_digits
.exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jb fail_at_current
    cmp cl, '9'
    ja fail_at_current
    jmp .exp_digits
.done:
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 't'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 1]
    cmp cl, 'r'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 2]
    cmp cl, 'u'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 3]
    cmp cl, 'e'
    jne fail_at_current
    add qword ptr [rip + pos], 4
    ret

parse_false:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 'f'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 1]
    cmp cl, 'a'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 2]
    cmp cl, 'l'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 3]
    cmp cl, 's'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 4]
    cmp cl, 'e'
    jne fail_at_current
    add qword ptr [rip + pos], 5
    ret

parse_null:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae fail_at_current
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 'n'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 1]
    cmp cl, 'u'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 2]
    cmp cl, 'l'
    jne fail_at_current
    movzx ecx, byte ptr [rbx + rax + 3]
    cmp cl, 'l'
    jne fail_at_current
    add qword ptr [rip + pos], 4
    ret

fail_at_current:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + err_offset], rax
    mov qword ptr [rip + parse_status], 1
    ret

print_error:
    mov qword ptr [rip + err_offset], rax
    lea rsi, [rip + error_prefix]
    mov rdx, error_prefix_len
    mov edi, 1
    mov eax, 1
    syscall
    mov rax, qword ptr [rip + err_offset]
    lea rdi, [rip + err_digits + 31]
    mov byte ptr [rdi], 10
    mov rcx, 0
    mov rbx, 10
    test rax, rax
    jnz .loop
    mov rcx, 1
    dec rdi
    mov byte ptr [rdi], '0'
    jmp .print
.loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc rcx
    test rax, rax
    jnz .loop
.print:
    lea rsi, [rdi]
    mov rdx, rcx
    mov edi, 1
    mov eax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
ok_msg:
    .asciz "ok\n"
ok_msg_len = . - ok_msg

error_prefix:
    .ascii "error at "
error_prefix_len = . - error_prefix

newline:
    .byte 10

.bss
pos:
    .quad 0
input_len:
    .quad 0
err_offset:
    .quad 0
parse_status:
    .quad 0
input_buf:
    .zero 65536
err_digits:
    .zero 32
