.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.bss
.align 8
input_buf:
    .zero INPUT_SIZE

.align 8
total_bytes:
    .zero 8

.align 8
line_ptrs:
    .zero MAX_LINES * 8

.align 8
line_lens:
    .zero MAX_LINES * 4

.align 8
line_count:
    .zero 8

.section .rodata
newline:
    .byte 10

.text
.globl _start

_start:
    call read_all
    call parse_lines
    call sort_lines
    call print_lines
    mov rax, SYS_EXIT
    xor edi, edi
    syscall

read_all:
    xor r12, r12
read_loop:
    cmp r12, INPUT_SIZE
    jae read_done
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_loop
read_done:
    mov qword ptr [rip + total_bytes], r12
    ret

parse_lines:
    mov qword ptr [rip + line_count], 0
    mov rcx, qword ptr [rip + total_bytes]
    test rcx, rcx
    jz parse_done
    xor r12, r12
    xor r13, r13
parse_loop:
    cmp r13, rcx
    jae parse_finalize
    lea rbx, [rip + input_buf]
    movzx r8d, byte ptr [rbx + r13]
    cmp r8b, 10
    jne parse_next
    mov rax, qword ptr [rip + line_count]
    lea rbx, [rip + line_ptrs]
    lea rdx, [rip + input_buf]
    lea rdi, [rdx + r12]
    mov qword ptr [rbx + rax*8], rdi
    lea rbx, [rip + line_lens]
    mov r8, r13
    sub r8, r12
    mov dword ptr [rbx + rax*4], r8d
    inc qword ptr [rip + line_count]
    mov r12, r13
    inc r12
parse_next:
    inc r13
    jmp parse_loop

parse_finalize:
    cmp r12, rcx
    je parse_done
    mov rax, qword ptr [rip + line_count]
    lea rbx, [rip + line_ptrs]
    lea rdx, [rip + input_buf]
    lea rdi, [rdx + r12]
    mov qword ptr [rbx + rax*8], rdi
    lea rbx, [rip + line_lens]
    mov r8, rcx
    sub r8, r12
    mov dword ptr [rbx + rax*4], r8d
    inc qword ptr [rip + line_count]
parse_done:
    ret

compare_lines:
    mov r8, rsi
    cmp rsi, rcx
    cmova r8, rcx
    xor r9d, r9d
compare_loop:
    cmp r9, r8
    jae compare_len
    movzx r10d, byte ptr [rdi + r9]
    movzx r11d, byte ptr [rdx + r9]
    cmp r10d, r11d
    jb compare_less
    ja compare_greater
    inc r9
    jmp compare_loop
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret
compare_len:
    cmp esi, ecx
    jb compare_less_len
    ja compare_greater_len
    xor eax, eax
    ret
compare_less_len:
    mov eax, -1
    ret
compare_greater_len:
    mov eax, 1
    ret

sort_lines:
    mov r15, qword ptr [rip + line_count]
    cmp r15, 1
    jbe sort_done
    dec r15
outer_loop:
    xor r13, r13
inner_loop:
    cmp r13, r15
    jae inner_done
    lea rbx, [rip + line_ptrs]
    lea r9, [rip + line_lens]
    mov rdi, qword ptr [rbx + r13*8]
    mov esi, dword ptr [r9 + r13*4]
    mov r14, r13
    inc r14
    mov rdx, qword ptr [rbx + r14*8]
    mov ecx, dword ptr [r9 + r14*4]
    call compare_lines
    cmp eax, 0
    jg swap_lines
    inc r13
    jmp inner_loop
swap_lines:
    mov rax, qword ptr [rbx + r13*8]
    mov r8, qword ptr [rbx + r14*8]
    mov qword ptr [rbx + r13*8], r8
    mov qword ptr [rbx + r14*8], rax
    mov eax, dword ptr [r9 + r13*4]
    mov r8d, dword ptr [r9 + r14*4]
    mov dword ptr [r9 + r13*4], r8d
    mov dword ptr [r9 + r14*4], eax
    inc r13
    jmp inner_loop
inner_done:
    test r15, r15
    jz sort_done
    dec r15
    jmp outer_loop
sort_done:
    ret

print_lines:
    mov r15, qword ptr [rip + line_count]
    test r15, r15
    jz print_done
    xor r13, r13
print_loop:
    cmp r13, r15
    jae print_done
    lea rbx, [rip + line_ptrs]
    lea r9, [rip + line_lens]
    mov rsi, qword ptr [rbx + r13*8]
    mov edx, dword ptr [r9 + r13*4]
    mov rax, SYS_WRITE
    mov rdi, 1
    syscall
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r13
    jmp print_loop
print_done:
    ret
