.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.bss
.align 8
input_buf:
    .zero INPUT_SIZE
.align 8
total_bytes:
    .zero 8
.align 8
line_ptrs:
    .zero MAX_LINES * 8
.align 4
line_lens:
    .zero MAX_LINES * 4
.align 8
line_count:
    .zero 8

.section .rodata
newline:
    .byte 10

.text

_start:
    call read_all
    call parse_lines
    call sort_lines
    call print_lines
    mov rax, SYS_EXIT
    xor edi, edi
    syscall

read_all:
    xor r12, r12
read_loop:
    cmp r12, INPUT_SIZE
    jae read_done
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_loop
read_done:
    mov qword ptr [rip + total_bytes], r12
    ret

parse_lines:
    mov qword ptr [rip + line_count], 0
    mov rbx, qword ptr [rip + total_bytes]
    test rbx, rbx
    jz parse_done
    xor r13, r13
    xor r14, r14
parse_loop:
    cmp r14, rbx
    jae parse_finish
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + r14]
    cmp cl, 10
    jne parse_next
    mov rdx, r14
    sub rdx, r13
    mov rax, qword ptr [rip + line_count]
    lea rdi, [rip + input_buf]
    add rdi, r13
    mov qword ptr [rip + line_ptrs + rax*8], rdi
    mov dword ptr [rip + line_lens + rax*4], edx
    inc qword ptr [rip + line_count]
    mov r13, r14
    inc r13
parse_next:
    inc r14
    jmp parse_loop
parse_finish:
    cmp r13, rbx
    jae parse_done
    mov rdx, rbx
    sub rdx, r13
    mov rax, qword ptr [rip + line_count]
    lea rdi, [rip + input_buf]
    add rdi, r13
    mov qword ptr [rip + line_ptrs + rax*8], rdi
    mov dword ptr [rip + line_lens + rax*4], edx
    inc qword ptr [rip + line_count]
parse_done:
    ret

compare_line_data:
    mov r8, rdi
    mov r9, rdx
    mov r10d, esi
    mov r11d, ecx
    mov eax, r10d
    cmp eax, r11d
    jbe compare_min_set
    mov eax, r11d
compare_min_set:
    mov ecx, eax
    test ecx, ecx
    jz compare_len
compare_byte_loop:
    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r9]
    cmp al, dl
    jb compare_less
    ja compare_greater
    inc r8
    inc r9
    dec ecx
    jne compare_byte_loop
compare_len:
    cmp r10d, r11d
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret

sort_lines:
    mov rcx, qword ptr [rip + line_count]
    cmp rcx, 1
    jbe sort_done
    mov r15, 1
sort_loop_i:
    cmp r15, rcx
    jae sort_done
    mov r13, r15
sort_loop_j:
    test r13, r13
    jz sort_after_swap
    mov rdi, qword ptr [rip + line_ptrs + (r13 - 1)*8]
    mov esi, dword ptr [rip + line_lens + (r13 - 1)*4]
    mov rdx, qword ptr [rip + line_ptrs + r13*8]
    mov ecx, dword ptr [rip + line_lens + r13*4]
    call compare_line_data
    test rax, rax
    jle sort_after_swap
    mov r8, qword ptr [rip + line_ptrs + (r13 - 1)*8]
    mov r9, qword ptr [rip + line_ptrs + r13*8]
    mov qword ptr [rip + line_ptrs + (r13 - 1)*8], r9
    mov qword ptr [rip + line_ptrs + r13*8], r8
    mov r8d, dword ptr [rip + line_lens + (r13 - 1)*4]
    mov r9d, dword ptr [rip + line_lens + r13*4]
    mov dword ptr [rip + line_lens + (r13 - 1)*4], r9d
    mov dword ptr [rip + line_lens + r13*4], r8d
    dec r13
    jmp sort_loop_j
sort_after_swap:
    inc r15
    jmp sort_loop_i
sort_done:
    ret

write_all:
    push r12
    push r13
    mov r12, rsi
    mov r13, rdx
write_all_loop:
    test r13, r13
    jz write_all_done
    mov rax, SYS_WRITE
    mov rsi, r12
    mov rdx, r13
    syscall
    cmp rax, 0
    je write_all_done
    add r12, rax
    sub r13, rax
    jmp write_all_loop
write_all_done:
    pop r13
    pop r12
    ret

print_lines:
    mov rcx, qword ptr [rip + line_count]
    xor r12, r12
print_loop:
    cmp r12, rcx
    jae print_done
    mov rdi, 1
    mov rsi, qword ptr [rip + line_ptrs + r12*8]
    mov edx, dword ptr [rip + line_lens + r12*4]
    call write_all
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all
    inc r12
    jmp print_loop
print_done:
    ret
