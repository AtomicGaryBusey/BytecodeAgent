.globl _start
.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000
.equ LINE_STRIDE, 16

.text
_start:
    call read_all
    call parse_lines
    call sort_lines
    call print_lines
    mov rax, 60
    xor rdi, rdi
    syscall

read_all:
    xor r12, r12
.loop:
    cmp r12, INPUT_SIZE
    jae .done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    cmp rdx, 0
    je .done
    syscall
    cmp rax, 0
    je .done
    add r12, rax
    jmp .loop
.done:
    mov qword ptr [rip + total_bytes], r12
    ret

store_line:
    mov rax, r9
    shl rax, 4
    lea rdi, [rip + line_array]
    add rdi, rax
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    inc r9
    ret

parse_lines:
    xor rbx, rbx
    xor r9, r9
    xor r13, r13
    mov r14, qword ptr [rip + total_bytes]
    cmp r14, 0
    je .done
.loop:
    cmp rbx, r14
    jae .end
    lea rax, [rip + input_buf]
    add rax, rbx
    cmp byte ptr [rax], 0x0A
    jne .next_byte
    lea r10, [rip + input_buf]
    add r10, r13
    mov r11, rbx
    sub r11, r13
    call store_line
    inc rbx
    mov r13, rbx
    jmp .loop
.next_byte:
    inc rbx
    jmp .loop
.end:
    cmp r13, r14
    je .done
    lea r10, [rip + input_buf]
    add r10, r13
    mov r11, r14
    sub r11, r13
    call store_line
.done:
    mov qword ptr [rip + line_count], r9
    ret

cmp_line_struct:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rsi]
    mov r10, qword ptr [rdi + 8]
    mov r11, qword ptr [rsi + 8]

    mov rax, r10
    cmp r10, r11
    cmova rax, r11

    xor rcx, rcx
.loop:
    cmp rcx, rax
    jae .len_cmp
    mov al, byte ptr [r8 + rcx]
    mov dl, byte ptr [r9 + rcx]
    cmp al, dl
    jne .diff
    inc rcx
    jmp .loop
.diff:
    cmp al, dl
    ja .gt
    mov eax, -1
    ret
.gt:
    mov eax, 1
    ret
.len_cmp:
    cmp r10, r11
    je .eq
    jb .lt
    mov eax, 1
    ret
.lt:
    mov eax, -1
    ret
.eq:
    xor eax, eax
    ret

swap_line_structs:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    mov qword ptr [rsi], r8
    mov qword ptr [rsi + 8], r9
    ret

sort_lines:
    mov r12, qword ptr [rip + line_count]
    cmp r12, 1
    jbe .done
    mov r13, 1
.outer:
    cmp r13, r12
    jae .done
.inner:
    cmp r13, 0
    je .next_i
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + line_array]
    add rdi, rax
    sub rdi, 16
    lea rsi, [rip + line_array]
    add rsi, rax
    call cmp_line_struct
    cmp eax, 0
    jle .next_i
    lea rdi, [rip + line_array]
    mov rax, r13
    shl rax, 4
    add rdi, rax
    sub rdi, 16
    lea rsi, [rip + line_array]
    add rsi, rax
    call swap_line_structs
    dec r13
    jmp .inner
.next_i:
    inc r13
    jmp .outer
.done:
    ret

print_lines:
    mov r12, qword ptr [rip + line_count]
    xor r13, r13
.loop:
    cmp r13, r12
    jae .done
    mov rax, r13
    shl rax, 4
    lea rsi, [rip + line_array]
    add rsi, rax
    mov r14, qword ptr [rsi]
    mov r15, qword ptr [rsi + 8]

    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, r15
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    inc r13
    jmp .loop
.done:
    ret

.section .rodata
.align 1
newline_byte:
    .byte 0x0A

.bss
.align 8
input_buf:
    .zero INPUT_SIZE

.align 8
total_bytes:
    .zero 8

.align 8
line_array:
    .zero MAX_LINES * LINE_STRIDE

.align 8
line_count:
    .zero 8
