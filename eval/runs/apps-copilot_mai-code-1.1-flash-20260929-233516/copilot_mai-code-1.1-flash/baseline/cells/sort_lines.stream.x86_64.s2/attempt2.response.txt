.globl _start

.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.bss
.align 8
input_buf:
    .zero INPUT_SIZE
.align 8
total_bytes:
    .zero 8
.align 8
line_ptrs:
    .zero MAX_LINES * 8
.align 4
line_lens:
    .zero MAX_LINES * 4
.align 8
line_count:
    .zero 8
.align 8
current_ptr:
    .zero 8
.align 4
current_len:
    .zero 4

.section .rodata
newline:
    .byte 10

.text
_start:
    call read_all
    call parse_lines
    call sort_lines
    call print_lines
    mov rax, 60
    xor edi, edi
    syscall

read_all:
    xor r12, r12
.loop:
    cmp r12, INPUT_SIZE
    jae .done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    je .done
    add r12, rax
    jmp .loop
.done:
    mov qword ptr [rip + total_bytes], r12
    ret

store_line:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r8*8], rdi
    lea rax, [rip + line_lens]
    mov dword ptr [rax + r8*4], esi
    ret

parse_lines:
    xor r13, r13
    mov r14, qword ptr [rip + total_bytes]
    cmp r14, 0
    je .done
    xor r15, r15
    xor rbx, rbx
.loop:
    cmp r15, r14
    jae .finalize
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r15]
    cmp cl, 0x0a
    jne .next
    mov r8, r13
    mov rdi, rax
    add rdi, rbx
    mov rsi, r15
    sub rsi, rbx
    call store_line
    mov rbx, r15
    inc rbx
    inc r13
.next:
    inc r15
    jmp .loop
.finalize:
    cmp rbx, r14
    jae .done
    mov r8, r13
    lea rax, [rip + input_buf]
    mov rdi, rax
    add rdi, rbx
    mov rsi, r14
    sub rsi, rbx
    call store_line
    inc r13
.done:
    mov qword ptr [rip + line_count], r13
    ret

cmp_lines:
    mov r8, rsi
    cmp r8, rcx
    jbe .len_limit
    mov r8, rcx
.len_limit:
    xor r9d, r9d
.loop:
    cmp r9d, r8d
    jae .check_len
    movzx eax, byte ptr [rdi + r9]
    movzx ebx, byte ptr [rdx + r9]
    cmp eax, ebx
    jb .less
    ja .greater
    inc r9
    jmp .loop
.check_len:
    cmp rsi, rcx
    jb .less
    ja .greater
    xor eax, eax
    ret
.less:
    mov eax, -1
    ret
.greater:
    mov eax, 1
    ret

sort_lines:
    mov rcx, qword ptr [rip + line_count]
    cmp rcx, 1
    jbe .done
    mov r12, 1
.outer:
    cmp r12, rcx
    jae .done
    mov r8, qword ptr [rip + line_ptrs + r12*8]
    mov r9d, dword ptr [rip + line_lens + r12*4]
    mov qword ptr [rip + current_ptr], r8
    mov dword ptr [rip + current_len], r9d
    mov r13, r12
    dec r13
.inner:
    cmp r13, -1
    je .insert
    mov r14, r13
    mov rdi, qword ptr [rip + line_ptrs + r14*8]
    mov esi, dword ptr [rip + line_lens + r14*4]
    mov rdx, qword ptr [rip + current_ptr]
    mov rcx, qword ptr [rip + current_len]
    call cmp_lines
    cmp eax, 0
    jle .insert
    mov rax, r13
    inc rax
    mov rdx, qword ptr [rip + line_ptrs + r13*8]
    mov qword ptr [rip + line_ptrs + rax*8], rdx
    mov ecx, dword ptr [rip + line_lens + r13*4]
    mov dword ptr [rip + line_lens + rax*4], ecx
    dec r13
    jmp .inner
.insert:
    inc r13
    mov rax, qword ptr [rip + current_ptr]
    mov qword ptr [rip + line_ptrs + r13*8], rax
    mov eax, dword ptr [rip + current_len]
    mov dword ptr [rip + line_lens + r13*4], eax
    inc r12
    jmp .outer
.done:
    ret

write_all:
    mov r8, rdx
    mov r9, rsi
    mov r10, rdi
.loop:
    cmp r8, 0
    je .done
    mov rax, 1
    mov rdi, r10
    mov rsi, r9
    mov rdx, r8
    syscall
    cmp rax, 0
    je .done
    cmp rax, r8
    je .done
    sub r8, rax
    add r9, rax
    jmp .loop
.done:
    ret

print_lines:
    mov r12, qword ptr [rip + line_count]
    cmp r12, 0
    je .done
    xor r13, r13
.loop:
    cmp r13, r12
    jae .done
    mov rdi, qword ptr [rip + line_ptrs + r13*8]
    mov esi, dword ptr [rip + line_lens + r13*4]
    mov rdx, rsi
    mov rsi, rdi
    mov edi, 1
    call write_all
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    inc r13
    jmp .loop
.done:
    ret
