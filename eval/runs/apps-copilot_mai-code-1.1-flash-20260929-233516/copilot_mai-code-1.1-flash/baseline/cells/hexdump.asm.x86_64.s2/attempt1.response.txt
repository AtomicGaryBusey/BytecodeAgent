.globl _start

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    add rsi, r12
    mov rdx, 4096
    syscall
    cmp rax, 0
    jl exit_fail
    je read_done
    add r12, rax
    cmp r12, 65536
    jbe read_loop
    mov r12, 65536
read_done:
    cmp r12, 0
    je print_total
    xor r13, r13
row_loop:
    mov rax, r12
    sub rax, r13
    mov r14, rax
    cmp r14, 16
    jbe row_ok
    mov r14, 16
row_ok:
    lea r11, [rip + output_buf]
    mov rdi, r11
    lea rbx, [rip + hex_digits]
    mov rdx, r13
    lea rsi, [rdi + 7]
    mov rcx, 8
offset_fill:
    mov rax, rdx
    and rax, 0xf
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rsi], al
    shr rdx, 4
    dec rsi
    dec rcx
    jnz offset_fill
    lea rdi, [r11 + 8]
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea r9, [rip + input_buf]
    add r9, r13
    xor rcx, rcx
hex_loop:
    cmp rcx, 16
    jae hex_done
    cmp rcx, r14
    jae hex_space
    movzx eax, byte ptr [r9]
    mov r8d, eax
    shr r8d, 4
    mov dl, byte ptr [rbx + r8]
    mov byte ptr [rdi], dl
    movzx eax, byte ptr [r9]
    and eax, 0xf
    mov dl, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    inc r9
    jmp after_slot
hex_space:
    mov dword ptr [rdi], 0x20202020
    add rdi, 3
after_slot:
    cmp rcx, 7
    jne slot_end
    mov byte ptr [rdi], 0x20
    inc rdi
slot_end:
    inc rcx
    jmp hex_loop
hex_done:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    lea r9, [rip + input_buf]
    add r9, r13
    mov rcx, r14
char_loop:
    movzx eax, byte ptr [r9]
    cmp al, 0x20
    jb char_dot
    cmp al, 0x7e
    ja char_dot
    mov byte ptr [rdi], al
    jmp char_next
char_dot:
    mov byte ptr [rdi], 0x2e
char_next:
    inc r9
    inc rdi
    dec rcx
    jnz char_loop
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rdx, rdi
    sub rdx, r11
    mov rax, 1
    mov rdi, 1
    mov rsi, r11
    syscall
    add r13, r14
    cmp r13, r12
    jb row_loop
    jmp print_total

print_total:
    lea r11, [rip + output_buf]
    lea rbx, [rip + hex_digits]
    mov rdx, r12
    lea rsi, [r11 + 7]
    mov rcx, 8
total_fill:
    mov rax, rdx
    and rax, 0xf
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rsi], al
    shr rdx, 4
    dec rsi
    dec rcx
    jnz total_fill
    mov byte ptr [r11 + 8], 0x0a
    mov rax, 1
    mov rdi, 1
    mov rsi, r11
    mov rdx, 9
    syscall
    mov rax, 60
    xor edi, edi
    syscall

exit_fail:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf:
    .space 65536
.align 16
output_buf:
    .space 256
