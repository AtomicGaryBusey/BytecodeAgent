.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
    .align 16
rowbuf:
    .space 128
    .align 16
outbuf:
    .space 16

.text
.globl _start

hex32_ascii:
    mov r8d, eax
    mov ecx, 8
    lea r9, [rip + hex_digits]
1:
    mov edx, r8d
    shr edx, 28
    and edx, 0xF
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl r8d, 4
    loop 1b
    ret

_start:
    xor r12, r12

read_loop:
    mov rax, 65536
    sub rax, r12
    jz done_read
    mov rdx, 4096
    cmp rax, rdx
    jae 1f
    mov rdx, rax
1:
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor rdi, rdi
    xor eax, eax
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    cmp r12, 65536
    jb read_loop

done_read:
    mov r13, r12
    xor rbp, rbp
    lea r14, [rip + input_buf]

dump_rows:
    test r12, r12
    jz final_line
    mov rcx, 16
    mov rax, r12
    cmp rax, rcx
    cmovb rcx, rax

    lea rsi, [rip + rowbuf]
    mov rdi, rsi
    mov eax, ebp
    call hex32_ascii
    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r8, r8

byte_loop:
    cmp r8d, 16
    jae byte_done
    cmp r8d, ecx
    jae missing_byte

    movzx eax, byte ptr [r14 + r8]
    mov r9b, al
    shr r9b, 4
    mov r10b, byte ptr [rip + hex_digits + r9]
    mov byte ptr [rdi], r10b
    inc rdi

    mov r9b, al
    and r9b, 0x0F
    mov r10b, byte ptr [rip + hex_digits + r9]
    mov byte ptr [rdi], r10b
    inc rdi

    mov byte ptr [rdi], ' '
    inc rdi
    cmp r8d, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc r8
    jmp byte_loop

missing_byte:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    inc r8
    jmp byte_loop

byte_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor r8, r8

char_loop:
    cmp r8d, ecx
    jae char_done
    movzx eax, byte ptr [r14 + r8]
    cmp al, 0x20
    jb dot
    cmp al, 0x7E
    ja dot
    mov byte ptr [rdi], al
    jmp char_next

dot:
    mov byte ptr [rdi], '.'

char_next:
    inc rdi
    inc r8
    jmp char_loop

char_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    mov r9, rcx
    mov rax, 1
    mov rdi, 1
    mov rdx, rdi
    sub rdx, rsi
    mov rsi, rsi
    syscall

    add r14, r9
    sub r12, r9
    add rbp, r9
    jmp dump_rows

final_line:
    lea rsi, [rip + outbuf]
    mov rdi, rsi
    mov eax, r13d
    call hex32_ascii
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

exit:
    mov rax, 60
    xor rdi, rdi
    syscall
