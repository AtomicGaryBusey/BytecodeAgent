.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
rowbuf:
    .space 128

.text
.globl _start

hex8:
    mov r8, rsi
    lea r9, [rip + hex_digits]
    mov rcx, 8
1:
    mov rax, r8
    shr rax, 60
    and eax, 15
    mov dl, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    shl r8, 4
    loop 1b
    ret

_start:
    xor r12d, r12d
    lea r11, [rip + input_buf]

read_loop:
    mov rax, 65536
    sub rax, r12
    jz done_read
    mov rdx, rax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    cmp r12, 65536
    jb read_loop

done_read:
    mov r13, r12
    xor ebp, ebp
    cmp r13, 0
    je total_line

dump_rows:
    mov rcx, 16
    mov rax, r13
    sub rax, rbp
    cmp rax, 16
    cmovb rcx, rax

    lea rdi, [rip + rowbuf]
    mov rsi, rbp
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2

    lea r9, [rip + hex_digits]
    xor rbx, rbx

byte_loop:
    cmp rbx, 16
    jae hex_done
    cmp rbx, rcx
    jae missing_hex

    movzx eax, byte ptr [r11 + rbp + rbx]
    mov r8b, al
    shr r8b, 4
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rdi], dl
    inc rdi
    mov r8b, al
    and r8b, 15
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp rbx, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc rbx
    jmp byte_loop

missing_hex:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp rbx, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc rbx
    jmp byte_loop

hex_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor rbx, rbx

char_loop:
    cmp rbx, rcx
    jae char_done
    movzx eax, byte ptr [r11 + rbp + rbx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    mov byte ptr [rdi], al
    jmp char_next
dot:
    mov byte ptr [rdi], '.'
char_next:
    inc rdi
    inc rbx
    jmp char_loop

char_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    add rbp, rcx
    cmp rbp, r13
    jb dump_rows

total_line:
    lea rdi, [rip + rowbuf]
    mov rsi, r13
    call hex8
    mov byte ptr [rdi], 10
    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    inc rdx
    mov rax, 1
    mov rdi, 1
    syscall

exit:
    mov rax, 60
    xor edi, edi
    syscall
