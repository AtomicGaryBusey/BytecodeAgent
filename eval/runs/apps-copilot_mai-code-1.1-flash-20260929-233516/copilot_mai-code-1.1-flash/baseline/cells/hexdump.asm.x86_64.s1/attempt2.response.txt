.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
    .align 16
rowbuf:
    .space 128

.text
.globl _start

hex8_to_ascii:
    mov r8d, esi
    lea r9, [rip + hex_digits]
    mov ecx, 8
1:
    mov eax, r8d
    shr eax, 28
    and eax, 0xF
    mov dl, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    shl r8d, 4
    loop 1b
    ret

_start:
    xor r12d, r12d

read_loop:
    mov eax, 65536
    sub eax, r12d
    jz done_read
    mov rdx, rax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    cmp r12d, 65536
    jb read_loop

done_read:
    mov r13d, r12d
    xor ebp, ebp
    mov r12, r13
    lea r14, [rip + input_buf]

dump_rows:
    test r12, r12
    jz final_line

    mov rcx, 16
    cmp rcx, r12
    cmovb rcx, r12

    lea r15, [rip + rowbuf]
    mov rdi, r15
    mov esi, ebp
    call hex8_to_ascii
    mov word ptr [rdi], 0x2020
    add rdi, 2

    lea r9, [rip + hex_digits]
    xor rbx, rbx

byte_loop:
    cmp rbx, 16
    jae byte_done
    cmp rbx, rcx
    jae missing_slot

    movzx eax, byte ptr [r14 + rbx]
    mov r8b, al
    shr r8b, 4
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rdi], dl
    inc rdi
    mov r8b, al
    and r8b, 0x0F
    mov dl, byte ptr [r9 + r8]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp rbx, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc rbx
    jmp byte_loop

missing_slot:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp rbx, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc rbx
    jmp byte_loop

byte_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor rbx, rbx

char_loop:
    cmp rbx, rcx
    jae char_done
    movzx eax, byte ptr [r14 + rbx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7E
    ja dot
    mov byte ptr [rdi], al
    jmp char_next
dot:
    mov byte ptr [rdi], '.'
char_next:
    inc rdi
    inc rbx
    jmp char_loop

char_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    mov rsi, r15
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    add r14, rcx
    sub r12, rcx
    add rbp, rcx
    jmp dump_rows

final_line:
    lea r15, [rip + rowbuf]
    mov rdi, r15
    mov esi, r13d
    call hex8_to_ascii
    mov byte ptr [rdi], 10
    inc rdi

    mov rsi, r15
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

exit:
    mov rax, 60
    xor edi, edi
    syscall
