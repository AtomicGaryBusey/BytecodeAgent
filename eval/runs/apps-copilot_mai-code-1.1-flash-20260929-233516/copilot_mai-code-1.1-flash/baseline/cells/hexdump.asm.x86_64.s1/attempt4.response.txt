.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buf:
    .space 65536
rowbuf:
    .space 128

.text
.globl _start

hex8:
    mov r8, rsi
    lea r9, [rip + hex_digits]
    mov rcx, 8
1:
    mov rax, r8
    shr rax, 60
    and al, 15
    mov dl, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    shl r8, 4
    loop 1b
    ret

write_all:
    mov r8, rdx
1:
    mov rax, 1
    mov rdx, r8
    syscall
    test rax, rax
    jz 9f
    cmp rax, r8
    je 9f
    sub r8, rax
    add rsi, rax
    jmp 1b
9:
    ret

_start:
    xor r12d, r12d
read_loop:
    mov rax, 65536
    sub rax, r12
    jz done_read
    mov rdx, rax
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    cmp r12, 65536
    jb read_loop
done_read:
    xor r13d, r13d
    cmp r12, 0
    je print_total
dump_rows:
    mov rbx, 16
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jb 1f
    jmp 2f
1:
    mov rbx, rax
2:
    lea rdi, [rip + rowbuf]
    mov rsi, r13
    call hex8
    mov byte ptr [rdi + 8], ' '
    mov byte ptr [rdi + 9], ' '
    add rdi, 10
    lea r9, [rip + input_buf]
    add r9, r13
    xor r10d, r10d
row_hex_loop:
    cmp r10, 16
    jae row_hex_done
    cmp r10, rbx
    jae row_missing_hex
    movzx eax, byte ptr [r9 + r10]
    mov dl, al
    shr dl, 4
    movzx edx, dl
    mov cl, byte ptr [rip + hex_digits + rdx]
    mov byte ptr [rdi], cl
    inc rdi
    mov dl, al
    and dl, 15
    movzx edx, dl
    mov cl, byte ptr [rip + hex_digits + rdx]
    mov byte ptr [rdi], cl
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp r10, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc r10
    jmp row_hex_loop
row_missing_hex:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    cmp r10, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc r10
    jmp row_hex_loop
row_hex_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    xor r10d, r10d
char_loop:
    cmp r10, rbx
    jae char_done
    movzx eax, byte ptr [r9 + r10]
    cmp al, 0x20
    jb 1f
    cmp al, 0x7e
    ja 1f
    mov byte ptr [rdi], al
    jmp 2f
1:
    mov byte ptr [rdi], '.'
2:
    inc rdi
    inc r10
    jmp char_loop
char_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rdi, 1
    call write_all
    add r13, rbx
    cmp r13, r12
    jb dump_rows
print_total:
    lea rdi, [rip + rowbuf]
    mov rsi, r12
    call hex8
    mov byte ptr [rdi], 10
    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    inc rdx
    mov rdi, 1
    call write_all
_exit:
    mov rax, 60
    xor edi, edi
    syscall
