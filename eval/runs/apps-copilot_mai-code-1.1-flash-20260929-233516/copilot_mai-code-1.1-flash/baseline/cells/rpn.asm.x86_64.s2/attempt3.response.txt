.text
.globl _start

.equ READ_BUF_SIZE, 65536
.equ ERR_MSG_LEN, 6
.equ STACK_ITEMS, 4096
.equ STACK_BYTES, STACK_ITEMS * 8

_start:
    xor r15, r15
    lea r14, [rip + input_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r14
    add rsi, r15
    mov rdx, READ_BUF_SIZE
    sub rdx, r15
    syscall
    cmp rax, 0
    je read_done
    add r15, rax
    cmp r15, READ_BUF_SIZE
    jb read_loop

read_done:
    mov qword ptr [rip + input_len], r15
    mov qword ptr [rip + next_pos], 0

main_loop:
    mov r15, qword ptr [rip + next_pos]
    mov rax, qword ptr [rip + input_len]
    cmp r15, rax
    jae done_input

    mov r12, r15
    mov r11, r15

scan_line_end:
    mov rax, qword ptr [rip + input_len]
    cmp r11, rax
    jae no_newline
    movzx eax, byte ptr [r14 + r11]
    cmp al, 0x0A
    je newline_found
    inc r11
    jmp scan_line_end

newline_found:
    mov qword ptr [rip + next_pos], r11
    inc qword ptr [rip + next_pos]
    jmp process_line

no_newline:
    mov r11, qword ptr [rip + input_len]
    mov qword ptr [rip + next_pos], r11
    jmp process_line

process_line:
    lea rbx, [rip + value_stack]
    xor r9, r9
    mov r10, r12

token_loop:
    cmp r10, r11
    jae end_of_line

    movzx eax, byte ptr [r14 + r10]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep

    mov r8, r10

scan_token:
    cmp r10, r11
    jae token_done
    movzx eax, byte ptr [r14 + r10]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc r10
    jmp scan_token

token_done:
    mov rdx, r10
    sub rdx, r8
    movzx eax, byte ptr [r14 + r8]

    cmp rdx, 1
    je single_char_token

    cmp al, '-'
    jne check_digit_start

    mov rdi, 1
    mov rsi, r8
    inc rsi
    dec rdx
    cmp rdx, 0
    je invalid_line
    jmp parse_digits

check_digit_start:
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    xor rdi, rdi
    mov rsi, r8
    jmp parse_digits

single_char_token:
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp invalid_line

parse_digits:
    xor rax, rax
    mov r13, 922337203685477580
    mov r15, 7

    cmp rdi, 0
    jne parse_neg_limit
    jmp parse_digit_loop

parse_neg_limit:
    mov r15, 8

parse_digit_loop:
    cmp rdx, 0
    je parse_done

    movzx ecx, byte ptr [r14 + rsi]
    cmp cl, '0'
    jb invalid_line
    cmp cl, '9'
    ja invalid_line
    sub cl, '0'

    cmp rax, r13
    ja invalid_line
    jne check_limit_done

    cmp cl, r15b
    ja invalid_line

check_limit_done:
    imul rax, rax, 10
    add rax, rcx
    inc rsi
    dec rdx
    jmp parse_digit_loop

parse_done:
    cmp rdi, 0
    je push_value
    neg rax

push_value:
    mov rdx, r9
    shl rdx, 3
    mov qword ptr [rbx + rdx], rax
    inc r9
    jmp token_loop

skip_sep:
    inc r10
    jmp token_loop

op_plus:
    cmp r9, 2
    jb invalid_line
    mov rax, qword ptr [rbx + r9*8 - 8]
    mov rcx, qword ptr [rbx + r9*8 - 16]
    add rcx, rax
    mov qword ptr [rbx + r9*8 - 16], rcx
    dec r9
    jmp token_loop

op_minus:
    cmp r9, 2
    jb invalid_line
    mov rax, qword ptr [rbx + r9*8 - 8]
    mov rcx, qword ptr [rbx + r9*8 - 16]
    sub rcx, rax
    mov qword ptr [rbx + r9*8 - 16], rcx
    dec r9
    jmp token_loop

op_mul:
    cmp r9, 2
    jb invalid_line
    mov rax, qword ptr [rbx + r9*8 - 16]
    imul rax, qword ptr [rbx + r9*8 - 8]
    mov qword ptr [rbx + r9*8 - 16], rax
    dec r9
    jmp token_loop

op_div:
    cmp r9, 2
    jb invalid_line
    mov rax, qword ptr [rbx + r9*8 - 16]
    mov rcx, qword ptr [rbx + r9*8 - 8]
    cmp rcx, 0
    je invalid_line
    mov r13, 0x8000000000000000
    cmp rax, r13
    jne div_ok
    cmp rcx, -1
    je invalid_line

div_ok:
    cqo
    idiv rcx
    mov qword ptr [rbx + r9*8 - 16], rax
    dec r9
    jmp token_loop

op_mod:
    cmp r9, 2
    jb invalid_line
    mov rax, qword ptr [rbx + r9*8 - 16]
    mov rcx, qword ptr [rbx + r9*8 - 8]
    cmp rcx, 0
    je invalid_line
    mov r13, 0x8000000000000000
    cmp rax, r13
    jne mod_ok
    cmp rcx, -1
    je invalid_line

mod_ok:
    cqo
    idiv rcx
    mov qword ptr [rbx + r9*8 - 16], rdx
    dec r9
    jmp token_loop

end_of_line:
    cmp r9, 1
    jne invalid_line
    mov rax, qword ptr [rbx]
    jmp print_value

invalid_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, ERR_MSG_LEN
    syscall
    jmp main_loop

print_value:
    mov r13, rax
    mov r12, 0
    cmp r13, 0
    jge value_nonneg
    mov r12, 1
    mov r11, 0x8000000000000000
    cmp r13, r11
    je value_special_min
    neg r13

value_nonneg:
    lea rsi, [rip + outbuf]
    add rsi, 32
    mov byte ptr [rsi], 0x0A
    mov rax, r13

value_digits:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne value_digits

    test r12, r12
    jz value_write
    dec rsi
    mov byte ptr [rsi], '-'

value_write:
    mov rax, 1
    mov rdi, 1
    lea rdx, [rip + outbuf]
    add rdx, 33
    sub rdx, rsi
    mov rsi, rsi
    syscall
    jmp main_loop

value_special_min:
    mov r13, 9223372036854775808
    jmp value_nonneg

done_input:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .asciz "error\n"

.section .bss
.align 8
input_len:
    .zero 8

.align 8
next_pos:
    .zero 8

.align 8
input_buf:
    .space READ_BUF_SIZE

.align 8
value_stack:
    .space STACK_BYTES

.align 8
outbuf:
    .space 64
