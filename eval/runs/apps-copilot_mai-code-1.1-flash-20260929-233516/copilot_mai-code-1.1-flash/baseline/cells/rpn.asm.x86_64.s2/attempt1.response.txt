.text
.globl _start

.equ READ_BUF_SIZE, 65536
.equ STACK_ITEMS, 4096
.equ STACK_BYTES, STACK_ITEMS * 8

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov rdx, READ_BUF_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, READ_BUF_SIZE
    jb read_loop

read_done:
    mov r13, 0

line_loop:
    cmp r13, r12
    jae done_input

    mov r14, r13
    mov r15, r13

find_newline:
    cmp r15, r12
    jae no_newline
    mov al, byte ptr [rip + stdin_buf + r15]
    cmp al, 0x0A
    je newline_found
    inc r15
    jmp find_newline

newline_found:
    mov rbx, r15
    lea r13, [r15 + 1]
    jmp process_line

no_newline:
    mov rbx, r12
    mov r13, r12

process_line:
    lea r8, [rip + stack_buf]
    xor r9, r9
    mov rcx, r14

token_loop:
skip_sep:
    cmp rcx, rbx
    jae end_of_line
    mov al, byte ptr [rip + stdin_buf + rcx]
    cmp al, 0x20
    je skip_sep_inc
    cmp al, 0x09
    je skip_sep_inc
    jmp token_start

skip_sep_inc:
    inc rcx
    jmp skip_sep

token_start:
    mov rsi, rcx
    mov rdx, rcx

token_scan:
    cmp rdx, rbx
    jae token_done
    mov al, byte ptr [rip + stdin_buf + rdx]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc rdx
    jmp token_scan

token_done:
    mov rdi, rdx
    mov rcx, rdi
    mov rdx, rdi
    sub rdx, rsi

    mov al, byte ptr [rip + stdin_buf + rsi]

    cmp rdx, 1
    jne not_single_char

    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp line_error

not_single_char:
    cmp al, '-'
    je parse_negative
    cmp al, '0'
    jb line_error
    cmp al, '9'
    ja line_error

    cmp rdx, 18
    ja line_error

    mov rax, 0
    mov r10, rsi

pos_digit_loop:
    cmp r10, rdi
    jae pos_done
    movzx r11d, byte ptr [rip + stdin_buf + r10]
    sub r11d, '0'
    imul rax, rax, 10
    add rax, r11
    inc r10
    jmp pos_digit_loop

pos_done:
    mov qword ptr [r8 + r9*8], rax
    inc r9
    jmp token_loop

parse_negative:
    cmp rdx, 2
    jb line_error
    cmp rdx, 19
    ja line_error

    mov rax, 0
    mov r10, rsi
    inc r10

neg_digit_loop:
    cmp r10, rdi
    jae neg_done
    movzx r11d, byte ptr [rip + stdin_buf + r10]
    cmp r11b, '0'
    jb line_error
    cmp r11b, '9'
    ja line_error
    sub r11d, '0'
    imul rax, rax, 10
    add rax, r11
    inc r10
    jmp neg_digit_loop

neg_done:
    neg rax
    mov qword ptr [r8 + r9*8], rax
    inc r9
    jmp token_loop

op_plus:
    cmp r9, 2
    jb line_error
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r11, qword ptr [r8 + rbx]
    dec r9
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r10, qword ptr [r8 + rbx]
    dec r9
    add r10, r11
    mov qword ptr [r8 + r9*8], r10
    inc r9
    jmp token_loop

op_minus:
    cmp r9, 2
    jb line_error
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r11, qword ptr [r8 + rbx]
    dec r9
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r10, qword ptr [r8 + rbx]
    dec r9
    sub r10, r11
    mov qword ptr [r8 + r9*8], r10
    inc r9
    jmp token_loop

op_mul:
    cmp r9, 2
    jb line_error
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r11, qword ptr [r8 + rbx]
    dec r9
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r10, qword ptr [r8 + rbx]
    dec r9
    imul r10, r11
    mov qword ptr [r8 + r9*8], r10
    inc r9
    jmp token_loop

op_div:
    cmp r9, 2
    jb line_error
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r11, qword ptr [r8 + rbx]
    dec r9
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r10, qword ptr [r8 + rbx]
    dec r9
    cmp r11, 0
    je line_error
    cmp r10, -9223372036854775808
    jne div_ok_check
    cmp r11, -1
    je line_error
div_ok_check:
    mov rax, r10
    cqo
    mov rcx, r11
    cqto
    idiv rcx
    mov r10, rax
    mov qword ptr [r8 + r9*8], r10
    inc r9
    jmp token_loop

op_mod:
    cmp r9, 2
    jb line_error
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r11, qword ptr [r8 + rbx]
    dec r9
    mov rbx, r9
    dec rbx
    shl rbx, 3
    mov r10, qword ptr [r8 + rbx]
    dec r9
    cmp r11, 0
    je line_error
    cmp r10, -9223372036854775808
    jne mod_ok_check
    cmp r11, -1
    je line_error
mod_ok_check:
    mov rax, r10
    cqo
    mov rcx, r11
    cqto
    idiv rcx
    mov r10, rdx
    mov qword ptr [r8 + r9*8], r10
    inc r9
    jmp token_loop

end_of_line:
    cmp r9, 1
    jne line_error
    mov rax, qword ptr [r8]
    jmp print_value

line_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    jmp after_line

print_value:
    mov r15, 0
    mov rbx, rax
    cmp rbx, 0
    jge print_value_positive
    mov r15, 1
    cmp rbx, 0x8000000000000000
    je print_value_min
    neg rbx
    jmp print_value_digits

print_value_min:
    mov rbx, 0x8000000000000000

print_value_positive:
print_value_digits:
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 0x0A
    ; rbx = abs(value) as unsigned, r15 = sign flag
    ; need loop that writes digits before newline
    ; if value == 0, we still produce one digit
    ; rbx is current value; rsi points to newline
    ; each iteration writes next digit before newline
digits_loop:
    mov rax, rbx
    xor rdx, rdx
    mov rcx, 10
    div rcx
    mov rbx, rax
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rbx, rbx
    jne digits_loop

    cmp r15, 0
    je write_value
    dec rsi
    mov byte ptr [rsi], '-'

write_value:
    mov rax, 1
    mov rdi, 1
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    syscall

after_line:
    jmp line_loop

done_input:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .asciz "error\n"

.section .bss
.align 8
stdin_buf:
    .space READ_BUF_SIZE

.align 8
stack_buf:
    .space STACK_BYTES

.align 8
outbuf:
    .space 32
