.text
.globl _start

.equ READ_BUF_SIZE, 65536
.equ STACK_ITEMS, 4096
.equ STACK_BYTES, STACK_ITEMS * 8
.equ ERR_MSG_LEN, 6
.equ MIN_STR_LEN, 21

_start:
    lea r14, [rip + input_buf]
    xor r15, r15

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r14 + r15]
    mov rdx, READ_BUF_SIZE
    sub rdx, r15
    syscall
    cmp rax, 0
    je read_done
    add r15, rax
    cmp r15, READ_BUF_SIZE
    jb read_loop

read_done:
    mov qword ptr [rip + input_len], r15
    xor r12, r12

main_loop:
    mov rax, qword ptr [rip + input_len]
    cmp r12, rax
    jae done_input

    mov r13, r12

scan_line_end:
    cmp r13, qword ptr [rip + input_len]
    jae no_newline
    movzx eax, byte ptr [r14 + r13]
    cmp al, 0x0A
    je newline_found
    inc r13
    jmp scan_line_end

newline_found:
    mov rax, r13
    mov qword ptr [rip + next_pos], rax
    inc qword ptr [rip + next_pos]
    jmp process_line

no_newline:
    mov rax, qword ptr [rip + input_len]
    mov qword ptr [rip + next_pos], rax
    jmp process_line

process_line:
    lea rbx, [rip + value_stack]
    xor r9, r9
    mov r10, r12

line_top:
    cmp r10, r13
    jae line_done

    movzx eax, byte ptr [r14 + r10]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp parse_token

skip_sep:
    inc r10
    jmp line_top

parse_token:
    mov r11, r10

scan_token:
    cmp r10, r13
    jae token_done
    movzx eax, byte ptr [r14 + r10]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc r10
    jmp scan_token

token_done:
    mov r8, r10
    sub r8, r11

    movzx eax, byte ptr [r14 + r11]

    cmp r8, 1
    je single_char_token

    cmp al, '-'
    jne parse_positive

    mov rdi, 1
    mov rsi, r11
    inc rsi
    mov rdx, r8
    dec rdx
    jmp parse_integer

parse_positive:
    mov rdi, 0
    mov rsi, r11
    mov rdx, r8
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line

parse_integer:
    cmp rdx, 0
    je invalid_line
    cmp rdx, 18
    ja invalid_line

    xor rax, rax

parse_digit_loop:
    cmp rdx, 0
    je parse_integer_done

    movzx ecx, byte ptr [r14 + rsi]
    cmp cl, '0'
    jb invalid_line
    cmp cl, '9'
    ja invalid_line
    sub cl, '0'

    imul rax, rax, 10
    add rax, rcx
    inc rsi
    dec rdx
    jmp parse_digit_loop

parse_integer_done:
    test rdi, rdi
    jz push_value
    neg rax
    jmp push_value

single_char_token:
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp invalid_line

push_value:
    mov rdx, r9
    shl rdx, 3
    mov qword ptr [rbx + rdx], rax
    inc r9
    jmp line_top

op_plus:
    cmp r9, 2
    jb invalid_line

    mov rdx, r9
    dec rdx
    shl rdx, 3
    mov rax, qword ptr [rbx + rdx]

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov rcx, qword ptr [rbx + rdx]

    add rcx, rax
    mov qword ptr [rbx + rdx], rcx
    dec r9
    jmp line_top

op_minus:
    cmp r9, 2
    jb invalid_line

    mov rdx, r9
    dec rdx
    shl rdx, 3
    mov rax, qword ptr [rbx + rdx]

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov rcx, qword ptr [rbx + rdx]

    sub rcx, rax
    mov qword ptr [rbx + rdx], rcx
    dec r9
    jmp line_top

op_mul:
    cmp r9, 2
    jb invalid_line

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov rax, qword ptr [rbx + rdx]

    mov rdx, r9
    dec rdx
    shl rdx, 3
    mov rcx, qword ptr [rbx + rdx]

    imul rax, rcx
    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov qword ptr [rbx + rdx], rax
    dec r9
    jmp line_top

op_div:
    cmp r9, 2
    jb invalid_line

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov rax, qword ptr [rbx + rdx]

    mov rdx, r9
    dec rdx
    shl rdx, 3
    mov rcx, qword ptr [rbx + rdx]

    cmp rcx, 0
    je invalid_line

    mov r8, 0x8000000000000000
    cmp rax, r8
    jne div_ok
    cmp rcx, -1
    je invalid_line

div_ok:
    cqo
    idiv rcx

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov qword ptr [rbx + rdx], rax
    dec r9
    jmp line_top

op_mod:
    cmp r9, 2
    jb invalid_line

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov rax, qword ptr [rbx + rdx]

    mov rdx, r9
    dec rdx
    shl rdx, 3
    mov rcx, qword ptr [rbx + rdx]

    cmp rcx, 0
    je invalid_line

    mov r8, 0x8000000000000000
    cmp rax, r8
    jne mod_ok
    cmp rcx, -1
    je invalid_line

mod_ok:
    cqo
    idiv rcx

    mov rdx, r9
    sub rdx, 2
    shl rdx, 3
    mov qword ptr [rbx + rdx], rdx
    dec r9
    jmp line_top

line_done:
    cmp r9, 1
    jne invalid_line

    mov rax, qword ptr [rbx]
    jmp emit_value

invalid_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, ERR_MSG_LEN
    syscall
    jmp advance_after_line

emit_value:
    mov r13, rax

    mov r11, 0x8000000000000000
    cmp r13, r11
    je emit_min_value

    cmp r13, 0
    jge value_nonneg
    mov r11, 1
    neg r13
    jmp value_setup

value_nonneg:
    xor r11, r11

value_setup:
    lea rsi, [rip + outbuf + 32]
    mov byte ptr [rsi], 0x0A
    dec rsi

value_digits:
    mov rax, r13
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    mov r13, rax
    test rax, rax
    jne value_digits

    test r11, r11
    jz value_write
    dec rsi
    mov byte ptr [rsi], '-'

value_write:
    lea rdx, [rip + outbuf + 33]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp advance_after_line

emit_min_value:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int64_str]
    mov rdx, MIN_STR_LEN
    syscall
    jmp advance_after_line

advance_after_line:
    mov r12, qword ptr [rip + next_pos]
    jmp main_loop

done_input:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
err_msg:
    .asciz "error\n"

min_int64_str:
    .asciz "-9223372036854775808\n"

.section .bss
.align 8
input_len:
    .zero 8

.align 8
next_pos:
    .zero 8

.align 8
input_buf:
    .space READ_BUF_SIZE

.align 8
value_stack:
    .space STACK_BYTES

.align 8
outbuf:
    .space 64
