.text
.globl _start

cmp_lines:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r8, rsi
    mov r9, rcx
    cmp r8, r9
    cmova r8, r9

    xor r10, r10
1:
    cmp r10, r8
    jae 2f
    movzx eax, byte ptr [rdi + r10]
    movzx ebx, byte ptr [rdx + r10]
    cmp eax, ebx
    jb 3f
    ja 4f
    inc r10
    jmp 1b
3:
    mov eax, -1
    jmp 9f
4:
    mov eax, 1
    jmp 9f
2:
    cmp rsi, rcx
    jb 5f
    ja 6f
    xor eax, eax
    jmp 9f
5:
    mov eax, -1
    jmp 9f
6:
    mov eax, 1
9:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

_start:
    xor r12, r12
1:
    cmp r12, 32768
    jae 2f
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    js 8f
    jz 2f
    add r12, rax
    jmp 1b
2:
    xor r15, r15
    xor r13, r13
    lea rbx, [rip + input_buf]

parse_loop:
    cmp r13, r12
    jae parse_done
    mov r14, r13
search_newline:
    cmp r14, r12
    jae parse_final
    movzx eax, byte ptr [rbx + r14]
    cmp al, 10
    je newline_found
    inc r14
    jmp search_newline

newline_found:
    mov rax, r14
    sub rax, r13
    lea rdx, [rip + input_buf]
    add rdx, r13
    lea rcx, [rip + line_ptrs]
    mov qword ptr [rcx + r15*8], rdx
    lea rcx, [rip + line_lens]
    mov qword ptr [rcx + r15*8], rax
    inc r15
    mov r13, r14
    inc r13
    jmp parse_loop

parse_final:
    cmp r13, r12
    jae parse_done
    mov rax, r12
    sub rax, r13
    lea rdx, [rip + input_buf]
    add rdx, r13
    lea rcx, [rip + line_ptrs]
    mov qword ptr [rcx + r15*8], rdx
    lea rcx, [rip + line_lens]
    mov qword ptr [rcx + r15*8], rax
    inc r15

parse_done:
    test r15, r15
    jz exit_success

    mov r14, 1
sort_outer:
    cmp r14, r15
    jae print_loop

    lea rax, [rip + line_ptrs]
    mov r12, qword ptr [rax + r14*8]
    lea rax, [rip + line_lens]
    mov rbp, qword ptr [rax + r14*8]

    mov r13, r14
    dec r13

sort_inner:
    cmp r13, 0
    jl insert_here

    lea rax, [rip + line_ptrs]
    mov rdi, qword ptr [rax + r13*8]
    lea rax, [rip + line_lens]
    mov rsi, qword ptr [rax + r13*8]
    mov rdx, r12
    mov rcx, rbp
    call cmp_lines

    cmp eax, 0
    jle insert_here

    lea rax, [rip + line_ptrs]
    mov r8, qword ptr [rax + r13*8]
    mov qword ptr [rax + r13*8 + 8], r8
    lea rax, [rip + line_lens]
    mov r8, qword ptr [rax + r13*8]
    mov qword ptr [rax + r13*8 + 8], r8

    dec r13
    jmp sort_inner

insert_here:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r13*8 + 8], r12
    lea rax, [rip + line_lens]
    mov qword ptr [rax + r13*8 + 8], rbp
    inc r14
    jmp sort_outer

print_loop:
    xor r14, r14
1:
    cmp r14, r15
    jae exit_success

    mov rdi, 1
    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + r14*8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + r14*8]
    mov rax, 1
    syscall

    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    syscall

    inc r14
    jmp 1b

exit_success:
    xor edi, edi
    mov rax, 60
    syscall

8:
    mov edi, 1
    mov rax, 60
    syscall

.section .bss
.align 8
input_buf:
    .zero 32768
line_ptrs:
    .zero 8000
line_lens:
    .zero 8000

.section .rodata
newline_byte:
    .byte 10
