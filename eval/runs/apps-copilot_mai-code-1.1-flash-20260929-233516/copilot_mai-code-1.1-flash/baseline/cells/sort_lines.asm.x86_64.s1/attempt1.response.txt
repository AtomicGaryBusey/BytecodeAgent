.text
.globl _start

cmp_lines:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9, r9
    jmp .loop_check
.loop:
    movzx eax, byte ptr [rdi + r9]
    movzx ebx, byte ptr [rdx + r9]
    cmp eax, ebx
    jb .lt
    ja .gt
    inc r9
.loop_check:
    cmp r9, r8
    jb .loop
    cmp rsi, rcx
    jb .neg1
    ja .pos1
    xor eax, eax
    jmp .done
.neg1:
    mov eax, -1
    jmp .done
.pos1:
    mov eax, 1
    jmp .done
.lt:
    mov eax, -1
    jmp .done
.gt:
    mov eax, 1
.done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

_start:
    xor r12, r12
    lea rbx, [rip + input_buf]
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    cmp r12, 32768
    je .read_done
    syscall
    test rax, rax
    js .exit_error
    jz .read_done
    add r12, rax
    cmp r12, 32768
    jb .read_loop
.read_done:
    xor r15, r15
    xor r14, r14
    cmp r12, 0
    je .sort_skip
.parse_outer:
    cmp r14, r12
    jae .parse_done
    mov r10, r14
    mov r11, r14
.scan_loop:
    cmp r11, r12
    je .line_no_newline
    movzx eax, byte ptr [rbx + r11]
    cmp al, 10
    je .line_with_newline
    inc r11
    jmp .scan_loop
.line_with_newline:
    mov rax, r11
    sub rax, r10
    lea rdx, [rbx + r10]
    mov qword ptr [rip + line_ptrs + r15*8], rdx
    mov qword ptr [rip + line_lens + r15*8], rax
    inc r15
    mov r14, r11
    inc r14
    cmp r14, r12
    jb .parse_outer
    jmp .parse_done
.line_no_newline:
    mov rax, r12
    sub rax, r10
    lea rdx, [rbx + r10]
    mov qword ptr [rip + line_ptrs + r15*8], rdx
    mov qword ptr [rip + line_lens + r15*8], rax
    inc r15
    jmp .parse_done
.parse_done:
.sort_skip:
    cmp r15, 2
    jb .print_loop
    mov r14, 1
.sort_outer:
    cmp r14, r15
    jae .print_loop
    mov r12, qword ptr [rip + line_ptrs + r14*8]
    mov r13, qword ptr [rip + line_lens + r14*8]
    mov rbx, r14
    dec rbx
.sort_inner:
    cmp rbx, 0
    jl .insert_here
    mov rdi, qword ptr [rip + line_ptrs + rbx*8]
    mov rsi, qword ptr [rip + line_lens + rbx*8]
    mov rdx, r12
    mov rcx, r13
    call cmp_lines
    cmp eax, 0
    jle .insert_here
    lea rcx, [rbx + 1]
    mov r8, qword ptr [rip + line_ptrs + rbx*8]
    mov qword ptr [rip + line_ptrs + rcx*8], r8
    mov r8, qword ptr [rip + line_lens + rbx*8]
    mov qword ptr [rip + line_lens + rcx*8], r8
    dec rbx
    jmp .sort_inner
.insert_here:
    lea rcx, [rbx + 1]
    mov qword ptr [rip + line_ptrs + rcx*8], r12
    mov qword ptr [rip + line_lens + rcx*8], r13
    inc r14
    jmp .sort_outer
.print_loop:
    xor r14, r14
.print_next:
    cmp r14, r15
    jae .exit_success
    mov rdi, 1
    mov rsi, qword ptr [rip + line_ptrs + r14*8]
    mov rdx, qword ptr [rip + line_lens + r14*8]
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r14
    jmp .print_next
.exit_success:
    xor edi, edi
    mov rax, 60
    syscall
.exit_error:
    mov edi, 1
    mov rax, 60
    syscall

.section .bss
.align 8
input_buf:
    .zero 32768
line_ptrs:
    .zero 8000
line_lens:
    .zero 8000

.section .rodata
newline_byte:
    .byte 10
