.globl _start

.text
_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + stdin_buf]
    xor rbx, rbx
    mov rcx, r12
scan_loop:
    cmp rbx, rcx
    jae no_bang
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x21
    je bang_found
    inc rbx
    jmp scan_loop
bang_found:
    mov r8, rbx
    mov r9, rbx
    inc r9
    mov r10, r12
    sub r10, r9
    jmp validate_start
no_bang:
    mov r8, r12
    xor r9, r9
    xor r10, r10
validate_start:
    xor r11d, r11d
    xor rbx, rbx
validate_loop:
    cmp rbx, r8
    jae validate_ok
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '['
    jne check_close
    cmp r11d, 1000
    jae invalid
    lea rdi, [rip + brk_stack]
    mov dword ptr [rdi + r11*4], ebx
    inc r11d
    jmp validate_next
check_close:
    cmp al, ']'
    jne validate_next
    test r11d, r11d
    je invalid
    dec r11d
    lea rdi, [rip + brk_stack]
    mov ecx, dword ptr [rdi + r11*4]
    lea rdi, [rip + match_table]
    mov dword ptr [rdi + rbx*4], ecx
    mov dword ptr [rdi + rcx*4], ebx
validate_next:
    inc rbx
    jmp validate_loop
validate_ok:
    test r11d, r11d
    jne invalid
    lea r15, [rip + stdin_buf]
    lea r14, [rip + tape]
    xor r13, r13
    xor r12, r12
    xor rbx, rbx
run_loop:
    cmp rbx, r8
    jae run_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '>'
    jne cmd_lt
    inc r13
    jmp next_pc
cmd_lt:
    cmp al, '<'
    jne cmd_plus
    dec r13
    jmp next_pc
cmd_plus:
    cmp al, '+'
    jne cmd_minus
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi]
    inc eax
    and eax, 0xff
    mov byte ptr [rsi], al
    jmp next_pc
cmd_minus:
    cmp al, '-'
    jne cmd_dot
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi]
    dec eax
    and eax, 0xff
    mov byte ptr [rsi], al
    jmp next_pc
cmd_dot:
    cmp al, '.'
    jne cmd_comma
    lea rsi, [r14 + r13]
    movzx edx, byte ptr [rsi]
    mov rdi, 1
    lea rsi, [rip + out_byte]
    mov byte ptr [rsi], dl
    mov rdx, 1
    mov rax, 1
    syscall
    jmp next_pc
cmd_comma:
    cmp al, ','
    jne cmd_open
    cmp r12, r10
    jae comma_zero
    lea rsi, [rip + stdin_buf]
    add rsi, r9
    add rsi, r12
    movzx eax, byte ptr [rsi]
    lea rsi, [r14 + r13]
    mov byte ptr [rsi], al
    inc r12
    jmp next_pc
comma_zero:
    lea rsi, [r14 + r13]
    mov byte ptr [rsi], 0
    jmp next_pc
cmd_open:
    cmp al, '['
    jne cmd_close
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi]
    test eax, eax
    jne next_pc
    lea rdi, [rip + match_table]
    mov eax, dword ptr [rdi + rbx*4]
    inc rax
    mov rbx, rax
    jmp run_loop
cmd_close:
    cmp al, ']'
    jne next_pc
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi]
    test eax, eax
    je next_pc
    lea rdi, [rip + match_table]
    mov eax, dword ptr [rdi + rbx*4]
    inc rax
    mov rbx, rax
    jmp run_loop
next_pc:
    inc rbx
    jmp run_loop
run_done:
    mov rax, 60
    xor rdi, rdi
    syscall
invalid:
    mov rax, 60
    mov rdi, 1
    syscall

.section .bss
stdin_buf:
    .space 65536
tape:
    .space 30000
brk_stack:
    .space 4000
match_table:
    .space 262144
out_byte:
    .space 1
