    cbnz x1, 1f

    mov x0, x19

    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    bl parse_string
    mov x20, x0
    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    bl parse_number
    mov x20, x0
    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    bl parse_true
    mov x20, x0
    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    bl parse_false
    mov x20, x0
    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    bl parse_null
    mov x20, x0
    mov x21, x1
    cbnz x21, 1f
    mov x0, x19
    mov x1, #0
    b 2f
1:
    mov x0, x20
    mov x1, x21
2:
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    adrp x10, g_buf
    add x10, x10, :lo12:g_buf
    adrp x11, g_len
    add x11, x11, :lo12:g_len
    ldr x11, [x11]
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x7B
    b.ne fail_pos
    add x19, x19, #1
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x7D
    b.eq object_done
member_loop:
    mov x0, x19
    bl parse_string
    mov x19, x0
    mov x20, x1
    cbnz x20, after_string
    mov x0, x19
    mov x1, #0
    b finish
after_string:
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x3A
    b.ne fail_pos
    add x19, x19, #1
    mov x0, x19
    bl parse_ws
    mov x19, x0
    mov x0, x19
    bl parse_value
    mov x19, x0
    mov x20, x1
    cbnz x20, after_value
    mov x0, x19
    mov x1, #0
    b finish
after_value:
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x2C
    b.eq comma
    cmp w12, #0x7D
    b.eq object_done
    b fail_pos
comma:
    add x19, x19, #1
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x7D
    b.eq fail_pos
    b member_loop

    add x19, x19, #1
    mov x0, x19
    mov x1, #1
    b finish
fail_pos:
    mov x0, x19
    mov x1, #0
finish:
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    adrp x10, g_buf
    add x10, x10, :lo12:g_buf
    adrp x11, g_len
    add x11, x11, :lo12:g_len
    ldr x11, [x11]
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x5B
    b.ne fail_pos
    add x19, x19, #1
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x5D
    b.eq array_done
array_loop:
    mov x0, x19
    bl parse_value
    mov x19, x0
    mov x20, x1
    cbnz x20, after_value
    mov x0, x19
    mov x1, #0
    b finish
after_value:
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x2C
    b.eq comma
    cmp w12, #0x5D
    b.eq array_done
    b fail_pos
comma:
    add x19, x19, #1
    mov x0, x19
    bl parse_ws
    mov x19, x0
    cmp x19, x11
    b.hs fail_pos
    ldrb w12, [x10, x19]
    cmp w12, #0x5D
    b.eq fail_pos
    b array_loop
array_done:
    add x19, x19, #1
    mov x0, x19
    mov x1, #1
    b finish
fail_pos:
    mov x0, x19
    mov x1, #0
finish:
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

parse_string:
    adrp x10, g_buf
    add x10, x10, :lo12:g_buf
    adrp x11, g_len
    add x11, x11, :lo12:g_len
    ldr x11, [x11]
    mov x9, x0
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x22
    b.ne fail_pos
    add x9, x9, #1
loop:
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x22
    b.eq close
    cmp w12, #0x5C
    b.eq escape
    cmp w12, #0x20
    b.lt fail_pos
    add x9, x9, #1
    b loop
escape:
    add x9, x9, #1
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x22
    b.eq cont
    cmp w12, #0x5C
    b.eq cont
    cmp w12, #0x2F
    b.eq cont
    cmp w12, #0x62
    b.eq cont
    cmp w12, #0x66
    b.eq cont
    cmp w12, #0x6E
    b.eq cont
    cmp w12, #0x72
    b.eq cont
    cmp w12, #0x74
    b.eq cont
    cmp w12, #0x75
    b.eq unicode
    b fail_pos
cont:
    add x9, x9, #1
    b loop
unicode:
    add x9, x9, #1
    mov x13, #0
u_loop:
    cmp x13, #4
    b.eq loop
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt fail_pos
    cmp w12, #0x39
    b.le hex_ok
    cmp w12, #0x41
    b.lt fail_pos
    cmp w12, #0x46
    b.le hex_ok
    cmp w12, #0x61
    b.lt fail_pos
    cmp w12, #0x66
    b.le hex_ok
    b fail_pos
hex_ok:
    add x9, x9, #1
    add x13, x13, #1
    b u_loop
close:
    add x9, x9, #1
    mov x0, x9
    mov x1, #1
    ret
fail_pos:
    mov x0, x9
    mov x1, #0
    ret
fail_end:
    mov x0, x11
    mov x1, #0
    ret

parse_number:
    adrp x10, g_buf
    add x10, x10, :lo12:g_buf
    adrp x11, g_len
    add x11, x11, :lo12:g_len
    ldr x11, [x11]
    mov x9, x0
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x2D
    b.ne 1f
    add x9, x9, #1
    cmp x9, x11
    b.hs fail_end
1:
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.eq zero_case
    cmp w12, #0x31
    b.lt fail_pos
    cmp w12, #0x39
    b.gt fail_pos
    add x9, x9, #1
    b digits
zero_case:
    add x9, x9, #1
    cmp x9, x11
    b.hs frac_or_exp_or_ok
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt frac_or_exp_or_ok
    cmp w12, #0x39
    b.gt frac_or_exp_or_ok
    b fail_pos
digits:
    cmp x9, x11
    b.hs frac_or_exp_or_ok
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt frac_or_exp_or_ok
    cmp w12, #0x39
    b.gt frac_or_exp_or_ok
    add x9, x9, #1
    b digits
frac_or_exp_or_ok:
    cmp x9, x11
    b.hs success
    ldrb w12, [x10, x9]
    cmp w12, #0x2E
    b.ne exp_part
    add x9, x9, #1
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt fail_pos
    cmp w12, #0x39
    b.gt fail_pos
    add x9, x9, #1
    b frac_digits
frac_digits:
    cmp x9, x11
    b.hs exp_or_done
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt exp_or_done
    cmp w12, #0x39
    b.gt exp_or_done
    add x9, x9, #1
    b frac_digits
exp_or_done:
    cmp x9, x11
    b.hs success
    ldrb w12, [x10, x9]
    cmp w12, #0x65
    b.eq exp_part
    cmp w12, #0x45
    b.eq exp_part
    b success
exp_part:
    add x9, x9, #1
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x2B
    b.eq sign
    cmp w12, #0x2D
    b.eq sign
    b exp_digits_start

    add x9, x9, #1
exp_digits_start:
    cmp x9, x11
    b.hs fail_end
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt fail_pos
    cmp w12, #0x39
    b.gt fail_pos
    add x9, x9, #1
    b exp_digits
exp_digits:
    cmp x9, x11
    b.hs success
    ldrb w12, [x10, x9]
    cmp w12, #0x30
    b.lt success
    cmp w12, #0x39
    b.gt success
    add x9, x9, #1
    b exp_digits
success:
    mov x0, x9
    mov x1, #1
    ret
fail_pos:
    mov x0, x9
    mov x1, #0
    ret
fail_end:
    mov x0, x11
    mov x1, #0
    ret

    mov x9, x0
    mov x10, x1
    mov x11, x2
    adrp x12, g_buf

    mov x14, #0
1:
    cmp x14, x11

    b.hs fail_end

    mov x2, x12

    add x1, x1, :lo12:err_suffix

    mov x0, #1
