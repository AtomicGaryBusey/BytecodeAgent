.text
.equ INT64_MAX, 9223372036854775807
.equ INT64_MIN_ABS, 9223372036854775808

.globl _start

_start:
    xor r12d, r12d
    lea r13, [rip + input_buffer]

read_loop:
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    mov r14, r12
    mov r12, 0

line_loop:
    cmp r12, r14
    jae exit_done
    mov rbx, r12

scan_line_end:
    cmp rbx, r14
    jae final_line
    mov al, byte ptr [input_buffer + rbx]
    cmp al, 10
    je newline_found
    inc rbx
    jmp scan_line_end

newline_found:
    mov rdi, r12
    mov rsi, rbx
    sub rsi, r12
    call process_line
    lea r12, [rbx + 1]
    jmp line_loop

final_line:
    mov rdi, r12
    mov rsi, r14
    sub rsi, r12
    call process_line
    jmp exit_done

exit_done:
    mov rax, 60
    xor edi, edi
    syscall

process_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi
    lea r14, [input_buffer]
    lea r15, [r14 + r12]
    lea rbp, [r15 + r13]
    mov qword ptr [stack_count], 0
    mov rbx, r14

parse_loop:
    cmp rbx, rbp
    jae parse_done
    mov al, byte ptr [rbx]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep

    mov rdi, rbx

scan_token:
    inc rbx
    cmp rbx, rbp
    jae token_end
    mov al, byte ptr [rbx]
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    jmp scan_token

token_end:
    mov rsi, rbx
    sub rsi, rdi
    call parse_token
    test al, al
    jnz parse_error
    jmp parse_loop

skip_sep:
    inc rbx
    jmp parse_loop

parse_done:
    mov rax, qword ptr [stack_count]
    cmp rax, 1
    jne parse_error
    mov rax, qword ptr [stack_data]
    call print_signed_64
    jmp process_line_done

parse_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [err_msg]
    mov rdx, 6
    syscall

process_line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

parse_token:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi

    cmp r13, 1
    jne maybe_number
    mov al, byte ptr [r12]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

maybe_number:
    mov r14, 0
    mov r15, r12
    mov rbx, r13

    cmp byte ptr [r15], '-'
    jne no_sign
    mov r14, 1
    inc r15
    dec rbx
    jz invalid_token
no_sign:
    cmp rbx, 0
    je invalid_token
    cmp rbx, 18
    ja invalid_token

    mov rax, 0
    mov rcx, 0
    mov r9, INT64_MAX
    mov r10, INT64_MIN_ABS
    mov r11, 7
    cmp r14, 0
    je num_limit_set
    mov r9, INT64_MIN_ABS
    mov r11, 8
num_limit_set:

digit_loop:
    cmp rcx, rbx
    je number_done
    movzx r8d, byte ptr [r15 + rcx]
    cmp r8b, '0'
    jb invalid_token
    cmp r8b, '9'
    ja invalid_token
    mov rdx, r8
    sub rdx, '0'

    cmp rax, 922337203685477580
    ja invalid_token
    jb digit_ok
    cmp rdx, r11
    ja invalid_token

digit_ok:
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    jmp digit_loop

number_done:
    cmp r14, 0
    je push_number
    neg rax
push_number:
    mov rdx, qword ptr [stack_count]
    mov qword ptr [stack_data + rdx*8], rax
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

op_plus:
    mov r15, qword ptr [stack_count]
    cmp r15, 2
    jb invalid_token
    mov rbx, qword ptr [stack_data + r15*8 - 8]
    mov rax, qword ptr [stack_data + r15*8 - 16]
    add rax, rbx
    sub r15, 2
    mov qword ptr [stack_data + r15*8], rax
    mov qword ptr [stack_count], r15
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

op_minus:
    mov r15, qword ptr [stack_count]
    cmp r15, 2
    jb invalid_token
    mov rbx, qword ptr [stack_data + r15*8 - 8]
    mov rax, qword ptr [stack_data + r15*8 - 16]
    sub rax, rbx
    sub r15, 2
    mov qword ptr [stack_data + r15*8], rax
    mov qword ptr [stack_count], r15
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

op_mul:
    mov r15, qword ptr [stack_count]
    cmp r15, 2
    jb invalid_token
    mov rbx, qword ptr [stack_data + r15*8 - 8]
    mov rax, qword ptr [stack_data + r15*8 - 16]
    imul rax, rbx
    sub r15, 2
    mov qword ptr [stack_data + r15*8], rax
    mov qword ptr [stack_count], r15
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

op_div:
    mov r15, qword ptr [stack_count]
    cmp r15, 2
    jb invalid_token
    mov rbx, qword ptr [stack_data + r15*8 - 8]
    mov rax, qword ptr [stack_data + r15*8 - 16]
    cmp rbx, 0
    je invalid_token
    mov rdx, -9223372036854775808
    cmp rax, rdx
    jne div_ok
    cmp rbx, -1
    je invalid_token
div_ok:
    cqo
    idiv rbx
    sub r15, 2
    mov qword ptr [stack_data + r15*8], rax
    mov qword ptr [stack_count], r15
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

op_mod:
    mov r15, qword ptr [stack_count]
    cmp r15, 2
    jb invalid_token
    mov rbx, qword ptr [stack_data + r15*8 - 8]
    mov rax, qword ptr [stack_data + r15*8 - 16]
    cmp rbx, 0
    je invalid_token
    mov rdx, -9223372036854775808
    cmp rax, rdx
    jne mod_ok
    cmp rbx, -1
    je invalid_token
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    sub r15, 2
    mov qword ptr [stack_data + r15*8], rax
    mov qword ptr [stack_count], r15
    inc qword ptr [stack_count]
    xor eax, eax
    jmp parse_token_done

invalid_token:
    mov eax, 1

parse_token_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_signed_64:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rax
    cmp r12, 0
    jl neg_value
    jmp prepare_print

neg_value:
    mov r15, -9223372036854775808
    cmp r12, r15
    je print_min
    neg r12
    sub rsp, 80
    mov byte ptr [rsp], '-'
    mov r13, 1
    jmp prepare_print2

print_min:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, 21
    syscall
    jmp print_done

prepare_print:
    sub rsp, 80
    xor r13, r13

prepare_print2:
    cmp r12, 0
    je print_zero
    lea r14, [rsp + 32 + 31]
    xor r15, r15

digit_to_buffer:
    mov rax, r12
    xor rdx, rdx
    mov rbx, 10
    div rbx
    mov r12, rax
    add dl, '0'
    mov byte ptr [r14], dl
    dec r14
    inc r15
    test r12, r12
    jnz digit_to_buffer

    lea rsi, [r14 + 1]
    lea rdi, [rsp + r13]
    mov rcx, r15
    rep movsb
    mov byte ptr [rdi], 10
    mov rdx, r13
    add rdx, r15
    inc rdx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rsp]
    syscall
    jmp print_done

print_zero:
    mov byte ptr [rsp], '0'
    mov byte ptr [rsp + 1], 10
    mov rax, 1
    mov rdi, 1
    lea rsi, [rsp]
    mov rdx, 2
    syscall

print_done:
    add rsp, 80
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.section .rodata
err_msg:
    .ascii "error\n"
min_str:
    .ascii "-9223372036854775808\n"

.bss
    .align 8
input_buffer:
    .space 65536
stack_data:
    .space 8192
stack_count:
    .zero 8
