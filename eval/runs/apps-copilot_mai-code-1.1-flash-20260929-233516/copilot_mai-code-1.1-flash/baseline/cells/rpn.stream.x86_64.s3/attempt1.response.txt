.text
.globl _start

_start:
    lea r13, [rip + input_buffer]
    xor r12d, r12d
read_loop:
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    lea r14, [rip + input_buffer]
    lea r15, [r14 + r12]
    mov r13, r14
line_loop:
    cmp r13, r15
    jae exit_ok
    mov rbx, r13
scan_line:
    cmp rbx, r15
    jae final_line
    mov al, byte ptr [rbx]
    cmp al, 10
    je newline_found
    inc rbx
    jmp scan_line

newline_found:
    mov rdi, r13
    mov rsi, rbx
    sub rsi, r13
    push r12
    push r13
    push r14
    push r15
    call process_line
    pop r15
    pop r14
    pop r13
    pop r12
    lea r13, [rbx + 1]
    jmp line_loop

final_line:
    mov rdi, r13
    mov rsi, r15
    sub rsi, r13
    push r12
    push r13
    push r14
    push r15
    call process_line
    pop r15
    pop r14
    pop r13
    pop r12
    jmp exit_ok

exit_ok:
    mov rax, 60
    xor edi, edi
    syscall

process_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r14, rdi
    mov r15, rsi
    mov qword ptr [rip + stack_count], 0
    xor r13d, r13d

parse_loop:
    cmp r13, r15
    jae parse_done

    mov al, byte ptr [r14 + r13]
    cmp al, 32
    je next_index
    cmp al, 9
    je next_index

    mov rbx, r13
token_scan:
    inc r13
    cmp r13, r15
    jae token_end
    mov al, byte ptr [r14 + r13]
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    jmp token_scan

token_end:
    mov rsi, r13
    sub rsi, rbx
    lea rdi, [r14 + rbx]
    call parse_token_and_apply
    test al, al
    jnz parse_error
    jmp parse_loop

next_index:
    inc r13
    jmp parse_loop

parse_done:
    mov rax, qword ptr [rip + stack_count]
    cmp rax, 1
    jne parse_error
    mov rdi, qword ptr [rip + stack_buffer]
    call print_signed_64
    jmp process_line_done

parse_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall

process_line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

parse_token_and_apply:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi

    cmp r13, 1
    jne parse_number

    mov al, byte ptr [r12]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

parse_number:
    xor r14d, r14d
    mov r15, r12
    mov rbx, r13

    cmp byte ptr [r15], '-'
    jne number_no_sign
    mov r14d, 1
    inc r15
    dec rbx
    jz parse_invalid
number_no_sign:
    cmp rbx, 0
    je parse_invalid
    cmp rbx, 18
    ja parse_invalid
    cmp rbx, 18
    jne digits_ok
    xor rdx, rdx
prefix_cmp:
    cmp rdx, 18
    je digits_ok
    movzx rax, byte ptr [r15 + rdx]
    movzx rcx, byte ptr [rip + max_prefix + rdx]
    cmp al, cl
    jb digits_ok
    ja parse_invalid
    inc rdx
    jmp prefix_cmp

digits_ok:
    xor rdx, rdx
digit_verify:
    cmp rdx, rbx
    je digit_done
    movzx rax, byte ptr [r15 + rdx]
    cmp al, '0'
    jb parse_invalid
    cmp al, '9'
    ja parse_invalid
    inc rdx
    jmp digit_verify

digit_done:
    xor rax, rax
    mov rcx, rbx
    mov rsi, r15
value_loop:
    test rcx, rcx
    jz value_done
    movzx rdx, byte ptr [rsi]
    sub dl, '0'
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jmp value_loop

value_done:
    cmp r14, 0
    je push_number
    neg rax
push_number:
    mov r8, qword ptr [rip + stack_count]
    lea r9, [rip + stack_buffer]
    mov qword ptr [r9 + r8*8], rax
    inc r8
    mov qword ptr [rip + stack_count], r8
    xor eax, eax
    jmp parse_token_end

op_plus:
    mov r10, 0
    jmp op_common
op_minus:
    mov r10, 1
    jmp op_common
op_mul:
    mov r10, 2
    jmp op_common
op_div:
    mov r10, 3
    jmp op_common
op_mod:
    mov r10, 4
    jmp op_common

op_common:
    mov rax, qword ptr [rip + stack_count]
    cmp rax, 2
    jb parse_invalid
    lea r14, [rip + stack_buffer]
    mov r8, qword ptr [r14 + rax*8 - 8]
    mov r9, qword ptr [r14 + rax*8 - 16]
    sub rax, 2
    mov qword ptr [rip + stack_count], rax

    cmp r10, 0
    je do_add
    cmp r10, 1
    je do_sub
    cmp r10, 2
    je do_mul
    cmp r10, 3
    je do_div
    cmp r10, 4
    je do_mod

do_add:
    mov rax, r9
    add rax, r8
    jmp op_push_result

do_sub:
    mov rax, r9
    sub rax, r8
    jmp op_push_result

do_mul:
    mov rax, r9
    imul rax, r8
    jmp op_push_result

do_div:
    cmp r8, 0
    je parse_invalid
    movabs r11, 0x8000000000000000
    cmp r9, r11
    jne div_ok
    cmp r8, -1
    jne div_ok
    jmp parse_invalid
div_ok:
    mov rax, r9
    cqo
    idiv r8
    jmp op_push_result

do_mod:
    cmp r8, 0
    je parse_invalid
    movabs r11, 0x8000000000000000
    cmp r9, r11
    jne mod_ok
    cmp r8, -1
    jne mod_ok
    jmp parse_invalid
mod_ok:
    mov rax, r9
    cqo
    idiv r8
    mov rax, rdx
    jmp op_push_result

op_push_result:
    mov r10, qword ptr [rip + stack_count]
    lea r11, [rip + stack_buffer]
    mov qword ptr [r11 + r10*8], rax
    inc r10
    mov qword ptr [rip + stack_count], r10
    xor eax, eax
    jmp parse_token_end

parse_invalid:
    mov eax, 1

parse_token_end:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_signed_64:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    lea r13, [rip + out_buffer]
    lea r14, [r13 + 63]
    mov byte ptr [r14], 10
    mov r15, r14
    mov rax, r12

    test rax, rax
    jns print_abs
    movabs rbx, 0x8000000000000000
    cmp rax, rbx
    je print_abs_min
    neg rax
    mov r11d, 1
    jmp print_digits

print_abs_min:
    movabs rax, 9223372036854775808
    mov r11d, 1
    jmp print_digits

print_abs:
    xor r11d, r11d
print_digits:
    test rax, rax
    jz print_zero
digits_loop:
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec r15
    mov byte ptr [r15], dl
    test rax, rax
    jnz digits_loop
    test r11d, r11d
    jz digits_done
    dec r15
    mov byte ptr [r15], '-'
digits_done:
    mov rdi, 1
    mov rsi, r15
    mov rdx, 64
    lea rbx, [r13 + 64]
    sub rdx, r15
    sub rdx, r13
    mov rdx, rbx
    sub rdx, r15
    mov rax, 1
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_zero:
    dec r15
    mov byte ptr [r15], '0'
    jmp digits_done

.section .rodata
err_msg:
    .ascii "error\n"

max_prefix:
    .ascii "922337203685477580"

.section .bss
.align 8
input_buffer:
    .zero 65536

stack_buffer:
    .zero 8192

stack_count:
    .zero 8

out_buffer:
    .zero 64
