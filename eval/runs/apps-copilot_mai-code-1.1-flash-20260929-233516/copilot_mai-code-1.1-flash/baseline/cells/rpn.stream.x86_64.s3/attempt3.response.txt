.intel_syntax noprefix

.text
.globl _start

_start:
    xor r12d, r12d

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    mov r13, r12
    xor r12d, r12d

main_loop:
    cmp r12, r13
    jae exit_program
    mov r14, r12

scan_line_end:
    cmp r14, r13
    jae final_line
    movzx eax, byte ptr [rip + input_buffer + r14]
    cmp al, 0x0A
    je line_done
    inc r14
    jmp scan_line_end

line_done:
    lea rdi, [rip + input_buffer]
    add rdi, r12
    mov rsi, r14
    sub rsi, r12
    call process_line
    mov r12, r14
    inc r12
    jmp main_loop

final_line:
    lea rdi, [rip + input_buffer]
    add rdi, r12
    mov rsi, r13
    sub rsi, r12
    test rsi, rsi
    jz exit_program
    call process_line
    jmp exit_program

exit_program:
    mov rax, 60
    xor edi, edi
    syscall

process_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov qword ptr [rip + stack_count], 0
    mov r12, rdi
    mov r13, rsi
    lea r14, [r12 + r13]

parse_loop:
    cmp r12, r14
    jae parse_done

skip_spaces:
    cmp r12, r14
    jae parse_done
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je next_byte
    cmp al, 0x09
    je next_byte
    jmp scan_token

next_byte:
    inc r12
    jmp skip_spaces

scan_token:
    mov r15, r12

scan_token_loop:
    cmp r12, r14
    jae token_end
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je token_end
    cmp al, 0x09
    je token_end
    inc r12
    jmp scan_token_loop

token_end:
    mov rdi, r15
    mov rsi, r12
    sub rsi, r15
    call parse_token
    test rax, rax
    jnz line_error
    jmp parse_loop

line_error:
    mov r12, r14
    jmp line_done_print

parse_done:
    mov rax, qword ptr [rip + stack_count]
    cmp rax, 1
    jne line_error
    mov rax, qword ptr [rip + stack]
    call print_signed_64
    jmp process_line_exit

line_done_print:
    call print_error

process_line_exit:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 6
    syscall
    ret

print_signed_64:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rax
    movabs r14, -9223372036854775808
    cmp r12, r14
    je print_signed_min
    mov r13, 0
    cmp r12, 0
    jge print_signed_nonneg
    mov r13, 1
    neg r12

print_signed_nonneg:
    lea r15, [rip + num_buf + 63]
    mov byte ptr [r15], 0x0A
    dec r15
    test r12, r12
    jnz print_signed_digits
    mov byte ptr [r15], '0'
    dec r15
    jmp print_signed_write

print_signed_digits:
    mov rax, r12
    mov rbx, 10

print_signed_loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    mov byte ptr [r15], dl
    dec r15
    mov r12, rax
    test rax, rax
    jne print_signed_loop
    cmp r13, 0
    je print_signed_write
    mov byte ptr [r15], '-'
    dec r15

print_signed_write:
    lea rsi, [r15 + 1]
    lea rax, [rip + num_buf + 64]
    sub rax, rsi
    mov rdx, rax
    mov rax, 1
    mov rdi, 1
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_signed_min:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, 21
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

parse_token:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi
    movzx eax, byte ptr [r12]
    cmp r13, 1
    jne parse_number
    cmp al, '+'
    je parse_operator
    cmp al, '-'
    je parse_operator
    cmp al, '*'
    je parse_operator
    cmp al, '/'
    je parse_operator
    cmp al, '%'
    je parse_operator

parse_number:
    xor rbx, rbx
    cmp byte ptr [r12], '-'
    jne parse_number_start
    mov rbx, 1
    inc r12
    dec r13
    jz parse_token_invalid

parse_number_start:
    cmp r13, 18
    ja parse_token_invalid
    movabs r14, 9223372036854775807
    cmp rbx, 1
    jne number_limit_ok
    movabs r14, -9223372036854775808

number_limit_ok:
    xor r15, r15
    mov rcx, r13

digit_loop:
    test rcx, rcx
    jz number_done
    movzx r8d, byte ptr [r12]
    cmp r8b, '0'
    jb parse_token_invalid
    cmp r8b, '9'
    ja parse_token_invalid
    sub r8d, '0'
    mov r9, r14
    sub r9, r8
    mov r10, 10
    xor rdx, rdx
    mov rax, r9
    div r10
    cmp r15, rax
    ja parse_token_invalid
    mov rax, r15
    imul rax, rax, 10
    add rax, r8
    mov r15, rax
    inc r12
    dec rcx
    jmp digit_loop

number_done:
    cmp rbx, 1
    jne push_number
    mov rax, r15
    neg rax
    mov r15, rax

push_number:
    mov rax, qword ptr [rip + stack_count]
    lea r10, [rip + stack]
    mov qword ptr [r10 + rax*8], r15
    inc qword ptr [rip + stack_count]
    xor eax, eax
    jmp parse_token_done

parse_operator:
    mov rbx, qword ptr [rip + stack_count]
    cmp rbx, 2
    jb parse_token_invalid
    dec rbx
    lea r10, [rip + stack]
    mov r8, qword ptr [r10 + rbx*8]
    dec rbx
    mov r9, qword ptr [r10 + rbx*8]
    mov rax, qword ptr [rip + stack_count]
    sub rax, 2
    mov qword ptr [rip + stack_count], rax

    cmp al, '+'
    je operator_add
    cmp al, '-'
    je operator_sub
    cmp al, '*'
    je operator_mul
    cmp al, '/'
    je operator_div
    cmp al, '%'
    je operator_mod

operator_add:
    mov rax, r9
    add rax, r8
    jmp push_result

operator_sub:
    mov rax, r9
    sub rax, r8
    jmp push_result

operator_mul:
    mov rax, r9
    imul rax, r8
    jmp push_result

operator_div:
    test r8, r8
    jz parse_token_invalid
    movabs r11, -9223372036854775808
    cmp r9, r11
    jne operator_div_ok
    cmp r8, -1
    jne operator_div_ok
    jmp parse_token_invalid

operator_div_ok:
    mov rax, r9
    cqo
    idiv r8
    jmp push_result

operator_mod:
    test r8, r8
    jz parse_token_invalid
    movabs r11, -9223372036854775808
    cmp r9, r11
    jne operator_mod_ok
    cmp r8, -1
    jne operator_mod_ok
    jmp parse_token_invalid

operator_mod_ok:
    mov rax, r9
    cqo
    idiv r8
    mov rax, rdx
    jmp push_result

push_result:
    mov rbx, qword ptr [rip + stack_count]
    lea r10, [rip + stack]
    mov qword ptr [r10 + rbx*8], rax
    inc qword ptr [rip + stack_count]
    xor eax, eax
    jmp parse_token_done

parse_token_invalid:
    mov rax, 1

parse_token_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
