.intel_syntax noprefix

.text
.globl _start

_start:
    lea rbp, [rip + stack]
    mov qword ptr [rip + line_error_flag], 0
    xor r12d, r12d

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    mov r13, r12
    xor r12d, r12d

main_loop:
    cmp r12, r13
    jae exit_ok

    mov r14, r12
    mov r15, r12

scan_line:
    cmp r15, r13
    jae scan_end
    movzx eax, byte ptr [rip + input_buffer + r15]
    cmp al, 10
    je newline_found
    inc r15
    jmp scan_line

newline_found:
    mov qword ptr [rip + line_nl_flag], 1
    jmp parse_setup

scan_end:
    mov r15, r13
    mov qword ptr [rip + line_nl_flag], 0

parse_setup:
    mov r10, r14
    xor r11d, r11d
    mov qword ptr [rip + line_error_flag], 0

parse_loop:
    cmp r10, r15
    jae after_parse
    movzx eax, byte ptr [rip + input_buffer + r10]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    jmp parse_token

skip_sep:
    inc r10
    jmp parse_loop

parse_token:
    mov r9, r10

scan_token_end:
    cmp r10, r15
    jae token_end
    movzx eax, byte ptr [rip + input_buffer + r10]
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    inc r10
    jmp scan_token_end

token_end:
    mov r8, r10
    sub r8, r9

    cmp r8, 1
    jne parse_literal

    movzx eax, byte ptr [rip + input_buffer + r9]
    cmp al, '+'
    je process_operator
    cmp al, '-'
    je process_operator
    cmp al, '*'
    je process_operator
    cmp al, '/'
    je process_operator
    cmp al, '%'
    je process_operator

    mov r10, r15
    mov qword ptr [rip + line_error_flag], 1
    jmp after_parse

parse_literal:
    mov rax, 0
    mov rcx, r8
    mov rdx, 0
    mov r10, r9

    movzx eax, byte ptr [rip + input_buffer + r9]
    cmp al, '-'
    jne literal_check_len

    mov rdx, 1
    dec rcx
    inc r10

literal_check_len:
    cmp rcx, 0
    jle invalid_literal
    cmp rcx, 18
    ja invalid_literal

digit_loop:
    cmp r10, r15
    jae digit_done
    movzx ebx, byte ptr [rip + input_buffer + r10]
    cmp bl, '0'
    jb invalid_literal
    cmp bl, '9'
    ja invalid_literal

    imul rax, rax, 10
    sub bl, '0'
    movzx rbx, bl
    add rax, rbx
    inc r10
    jmp digit_loop

digit_done:
    test rdx, rdx
    jz push_literal
    neg rax

push_literal:
    mov qword ptr [rbp + r11*8], rax
    inc r11
    jmp parse_loop

invalid_literal:
    mov r10, r15
    mov qword ptr [rip + line_error_flag], 1
    jmp after_parse

process_operator:
    cmp r11, 2
    jb line_error

    dec r11
    mov r10, qword ptr [rbp + r11*8]
    dec r11
    mov rax, qword ptr [rbp + r11*8]

    movzx ebx, byte ptr [rip + input_buffer + r9]

    cmp bl, '+'
    jne op_sub
    add rax, r10
    jmp op_push

op_sub:
    cmp bl, '-'
    jne op_mul
    sub rax, r10
    jmp op_push

op_mul:
    cmp bl, '*'
    jne op_div
    imul rax, r10
    jmp op_push

op_div:
    cmp bl, '/'
    jne op_mod
    test r10, r10
    jz line_error

    mov rdi, -9223372036854775808
    cmp rax, rdi
    jne op_div_no_overflow
    mov rsi, -1
    cmp r10, rsi
    je line_error

op_div_no_overflow:
    cqo
    idiv r10
    jmp op_push

op_mod:
    cmp bl, '%'
    jne line_error
    test r10, r10
    jz line_error

    mov rdi, -9223372036854775808
    cmp rax, rdi
    jne op_mod_no_overflow
    mov rsi, -1
    cmp r10, rsi
    je line_error

op_mod_no_overflow:
    cqo
    idiv r10
    mov rax, rdx
    jmp op_push

line_error:
    mov qword ptr [rip + line_error_flag], 1
    mov r10, r15
    jmp after_parse

op_push:
    mov qword ptr [rbp + r11*8], rax
    inc r11
    jmp parse_loop

after_parse:
    mov rax, qword ptr [rip + line_error_flag]
    test rax, rax
    jnz print_error

    cmp r11, 1
    jne print_error

    mov rdi, qword ptr [rbp]

    lea rsi, [rip + printbuf + 128]
    mov rax, rdi
    xor rbx, rbx
    test rax, rax
    jns num_nonneg

    mov rbx, 1
    mov rax, rdi
    add rax, 1
    neg rax
    add rax, 1

num_nonneg:
    cmp rax, 0
    jne num_digits
    dec rsi
    mov byte ptr [rsi], '0'
    jmp num_sign

num_digits:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz num_digits

num_sign:
    test rbx, rbx
    jz num_append
    dec rsi
    mov byte ptr [rsi], '-'

num_append:
    lea rdx, [rip + printbuf + 128]
    sub rdx, rsi
    mov byte ptr [rsi + rdx], 10
    inc rdx
    mov rax, 1
    mov rdi, 1
    syscall
    jmp advance_line

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 6
    syscall

advance_line:
    mov rax, qword ptr [rip + line_nl_flag]
    test rax, rax
    jz no_newline
    mov r12, r15
    inc r12
    jmp main_loop

no_newline:
    mov r12, r13
    jmp main_loop

exit_ok:
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
error_msg:
    .asciz "error\n"

.bss
line_nl_flag:
    .zero 8
line_error_flag:
    .zero 8
input_buffer:
    .space 65536
stack:
    .space 32768
printbuf:
    .space 128
