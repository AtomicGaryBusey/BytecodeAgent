.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.equ INPUT_LIMIT, 49152
.equ LINE_LIMIT, 76

.section .bss
.align 16
input_buf:
    .zero 65536
.align 16
output_buf:
    .zero 131072

.text
_start:
    xor rax, rax
    xor rbx, rbx
    xor rcx, rcx
    xor rdx, rdx
    xor rsi, rsi
    xor rdi, rdi
    cld

    lea r13, [rip + input_buf]
    xor r14, r14

read_loop:
    cmp r14, INPUT_LIMIT
    jae read_done

    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, 4096
    syscall

    cmp rax, 0
    je read_done

    add r14, rax
    add r13, rax
    jmp read_loop

read_done:
    lea r15, [rip + input_buf]
    lea rbp, [r15 + r14]

    lea r9, [rip + output_buf]
    xor r8, r8

encode_loop:
    mov rax, rbp
    sub rax, r15
    cmp rax, 3
    jb rem_group

    movzx eax, byte ptr [r15]
    movzx ecx, byte ptr [r15 + 1]
    movzx edx, byte ptr [r15 + 2]

    shl rax, 16
    shl rcx, 8
    or rax, rcx
    or rax, rdx

    mov rdx, rax
    shr rdx, 18
    and edx, 63
    call emit_char_from_index

    mov rdx, rax
    shr rdx, 12
    and edx, 63
    call emit_char_from_index

    mov rdx, rax
    shr rdx, 6
    and edx, 63
    call emit_char_from_index

    mov rdx, rax
    and edx, 63
    call emit_char_from_index

    add r15, 3
    jmp encode_loop

rem_group:
    mov rax, rbp
    sub rax, r15
    cmp rax, 0
    je finish_encode
    cmp rax, 2
    je rem_two

rem_one:
    movzx eax, byte ptr [r15]

    mov rdx, rax
    shr rdx, 2
    and edx, 63
    call emit_char_from_index

    mov rdx, rax
    and edx, 3
    shl edx, 4
    call emit_char_from_index

    call emit_eq
    call emit_eq
    jmp finish_encode

rem_two:
    movzx eax, byte ptr [r15]
    movzx ecx, byte ptr [r15 + 1]

    mov rdx, rax
    shr rdx, 2
    and edx, 63
    call emit_char_from_index

    mov rdx, rax
    and edx, 3
    shl edx, 4
    mov r11d, ecx
    shr r11d, 4
    and r11d, 15
    or edx, r11d
    call emit_char_from_index

    mov rdx, rcx
    and edx, 15
    shl edx, 2
    call emit_char_from_index

    call emit_eq
    jmp finish_encode

finish_encode:
    lea r12, [rip + output_buf]
    mov r13, r9
    sub r13, r12

    cmp r13, 0
    je exit_zero

write_loop:
    mov rax, 1
    mov rdi, 1
    mov rsi, r12
    mov rdx, r13
    syscall

    cmp rax, 0
    jle write_error

    add r12, rax
    sub r13, rax
    jnz write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

write_error:
    mov rax, 60
    mov rdi, 1
    syscall

emit_char_from_index:
    lea r10, [rip + alphabet]
    movzx r11d, dl
    mov dl, byte ptr [r10 + r11]
    mov byte ptr [r9], dl
    inc r9
    inc r8
    cmp r8, LINE_LIMIT
    jne 1f
    mov byte ptr [r9], 10
    inc r9
    xor r8, r8
1:
    ret

emit_eq:
    mov byte ptr [r9], 61
    inc r9
    inc r8
    cmp r8, LINE_LIMIT
    jne 1f
    mov byte ptr [r9], 10
    inc r9
    xor r8, r8
1:
    ret
