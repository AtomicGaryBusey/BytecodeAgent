.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 70000

.text

_start:
    cld
    xor r15, r15
    lea r14, [rip + input_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r14
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    add r14, rax
    jmp read_loop

read_done:
    test r15, r15
    jz exit_zero

    lea r14, [rip + input_buf]
    lea rbp, [rip + input_buf]
    add rbp, r15
    lea r12, [rip + output_buf]
    lea r9, [rip + alphabet]
    xor r13, r13

encode_loop:
    lea rbx, [r14 + 3]
    cmp rbx, rbp
    ja rem_group

    movzx eax, byte ptr [r14]
    movzx ebx, byte ptr [r14 + 1]
    shl rax, 16
    shl rbx, 8
    or rax, rbx
    movzx ebx, byte ptr [r14 + 2]
    or rax, rbx
    mov r10, rax

    mov rdx, r10
    shr rdx, 18
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r10
    shr rdx, 12
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r10
    shr rdx, 6
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r10
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    add r14, 3
    jmp encode_loop

rem_group:
    cmp r14, rbp
    je finalize

    mov rbx, rbp
    sub rbx, r14
    cmp rbx, 2
    je rem_two
    cmp rbx, 1
    je rem_one

rem_one:
    movzx eax, byte ptr [r14]
    mov r10, rax

    mov rdx, r10
    shr rdx, 2
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r10
    and edx, 3
    shl edx, 4
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov byte ptr [r12], 61
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov byte ptr [r12], 61
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    jmp finalize

rem_two:
    movzx eax, byte ptr [r14]
    movzx ebx, byte ptr [r14 + 1]
    mov r10, rax
    mov r11, rbx

    mov rdx, r10
    shr rdx, 2
    and edx, 63
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r10
    and edx, 3
    shl edx, 4
    mov ecx, r11d
    shr ecx, 4
    and ecx, 15
    or edx, ecx
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov rdx, r11
    and edx, 15
    shl edx, 2
    movzx r8d, byte ptr [r9 + rdx]
    mov byte ptr [r12], r8b
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    mov byte ptr [r12], 61
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    jmp finalize

finalize:
    cmp r13, 0
    je write_output
    mov byte ptr [r12], 10
    inc r12

write_output:
    lea r8, [rip + output_buf]
    mov rbx, r12
    sub rbx, r8
    test rbx, rbx
    jz exit_zero

write_loop:
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, rbx
    syscall
    add r8, rax
    sub rbx, rax
    jnz write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
