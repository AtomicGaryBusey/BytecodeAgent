.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 131072

.text

emit_symbol:
    lea rbx, [rip + alphabet]
    movzx ecx, dl
    mov dl, byte ptr [rbx + rcx]
    mov byte ptr [r12], dl
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    ret

emit_pad:
    mov byte ptr [r12], 61
    inc r12
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [r12], 10
    inc r12
    xor r13, r13
1:
    ret

_start:
    cld
    xor r15, r15
    lea r14, [rip + input_buf]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r14
    mov rdx, 4096
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    add r14, rax
    jmp read_loop

read_done:
    lea r14, [rip + input_buf]
    lea rbp, [rip + input_buf]
    add rbp, r15

    lea r12, [rip + output_buf]
    lea r9, [rip + output_buf]
    xor r13, r13

encode_loop:
    mov rbx, rbp
    sub rbx, r14
    cmp rbx, 3
    jb rem_group

    movzx eax, byte ptr [r14]
    movzx ebx, byte ptr [r14 + 1]
    shl rax, 16
    shl rbx, 8
    or rax, rbx
    movzx ebx, byte ptr [r14 + 2]
    or rax, rbx
    mov r15, rax

    mov rdx, r15
    shr rdx, 18
    and edx, 63
    call emit_symbol

    mov rdx, r15
    shr rdx, 12
    and edx, 63
    call emit_symbol

    mov rdx, r15
    shr rdx, 6
    and edx, 63
    call emit_symbol

    mov rdx, r15
    and edx, 63
    call emit_symbol

    add r14, 3
    jmp encode_loop

rem_group:
    mov rbx, rbp
    sub rbx, r14
    test rbx, rbx
    jz finalize
    cmp rbx, 2
    je rem_two

rem_one:
    movzx eax, byte ptr [r14]

    mov rdx, rax
    shr rdx, 2
    and edx, 63
    call emit_symbol

    mov rdx, rax
    and edx, 3
    shl edx, 4
    call emit_symbol

    call emit_pad
    call emit_pad
    jmp finalize

rem_two:
    movzx eax, byte ptr [r14]
    movzx ebx, byte ptr [r14 + 1]

    mov rdx, rax
    shr rdx, 2
    and edx, 63
    call emit_symbol

    mov rdx, rax
    and edx, 3
    shl edx, 4
    mov ecx, ebx
    shr ecx, 4
    and ecx, 15
    or edx, ecx
    call emit_symbol

    mov rdx, rbx
    and edx, 15
    shl edx, 2
    call emit_symbol

    call emit_pad
    jmp finalize

finalize:
    cmp r13, 0
    je write_output
    mov byte ptr [r12], 10
    inc r12

write_output:
    lea r10, [rip + output_buf]
    mov rbx, r12
    sub rbx, r10
    test rbx, rbx
    jz exit_zero

write_loop:
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, rbx
    syscall
    add r10, rax
    sub rbx, rax
    jnz write_loop

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
