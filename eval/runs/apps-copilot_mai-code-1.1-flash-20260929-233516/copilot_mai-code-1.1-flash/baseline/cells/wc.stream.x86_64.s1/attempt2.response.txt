.globl _start

.bss
.align 16
buffer:
    .zero 65536

.align 8
outputbuf:
    .zero 64

tmpbuf:
    .zero 32

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    mov eax, 0
    syscall
    test rax, rax
    jz done
    js exit_error
    mov rcx, rax
    add r14, rax
    lea rsi, [rip + buffer]

parse_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx

    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0A
    je newline
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace

    test r15, r15
    jnz parse_loop
    inc r13
    mov r15, 1
    jmp parse_loop

newline:
    inc r12
whitespace:
    xor r15, r15
    jmp parse_loop

exit_error:
    mov eax, 60
    mov edi, 1
    syscall

done:
    lea rbx, [rip + outputbuf]
    mov rdi, rbx

    mov rax, r12
    lea rsi, [rip + tmpbuf]
    xor rcx, rcx

conv_lines:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, '0'
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test rax, rax
    jnz conv_lines
    lea r11, [rsi + rcx]
    dec r11

copy_lines:
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    cmp r11, rsi
    je after_lines
    dec r11
    jmp copy_lines

after_lines:
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r13
    lea rsi, [rip + tmpbuf]
    xor rcx, rcx

conv_words:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, '0'
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test rax, rax
    jnz conv_words
    lea r11, [rsi + rcx]
    dec r11

copy_words:
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    cmp r11, rsi
    je after_words
    dec r11
    jmp copy_words

after_words:
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r14
    lea rsi, [rip + tmpbuf]
    xor rcx, rcx

conv_bytes:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, '0'
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test rax, rax
    jnz conv_bytes
    lea r11, [rsi + rcx]
    dec r11

copy_bytes:
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    cmp r11, rsi
    je after_bytes
    dec r11
    jmp copy_bytes

after_bytes:
    mov byte ptr [rdi], 0x0A
    inc rdi

    mov rdx, rdi
    sub rdx, rbx
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall

    mov eax, 60
    xor edi, edi
    syscall
