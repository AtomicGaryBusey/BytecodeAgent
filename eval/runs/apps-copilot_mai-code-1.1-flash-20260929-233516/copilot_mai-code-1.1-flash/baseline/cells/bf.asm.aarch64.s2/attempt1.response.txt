.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    mov x21, x0
    cmp x21, #0
    b.eq read_done
    cmp x21, #0
    b.lt read_error
    add x20, x20, x21
    cmp x20, #65536
    b.hi read_error
    b read_loop

read_done:
    mov x0, x19
    mov x1, #0
    mov x3, x20
scan_bang:
    cmp x1, x3
    b.ge no_bang
    ldrb w4, [x0, x1]
    cmp w4, #0x21
    b.eq bang_found
    add x1, x1, #1
    b scan_bang

no_bang:
    mov x21, x20
    add x22, x19, x20
    b after_split

bang_found:
    mov x21, x1
    add x1, x1, #1
    add x22, x19, x1

after_split:
    add x23, x19, x20

    adrp x24, jump
    add x24, x24, :lo12:jump
    adrp x25, stack
    add x25, x25, :lo12:stack

    mov x26, #0
    mov x27, #0

build_brackets:
    cmp x27, x21
    b.ge build_done
    ldrb w0, [x19, x27]
    cmp w0, #0x5b
    b.eq open_bracket
    cmp w0, #0x5d
    b.eq close_bracket
    add x27, x27, #1
    b build_brackets

open_bracket:
    cmp x26, #1000
    b.hs bracket_error
    add x0, x25, x26, lsl #2
    str w27, [x0]
    add x26, x26, #1
    add x27, x27, #1
    b build_brackets

close_bracket:
    cmp x26, #0
    b.eq bracket_error
    sub x26, x26, #1
    add x0, x25, x26, lsl #2
    ldr w1, [x0]
    str w27, [x24, x1, lsl #2]
    str w1, [x24, x27, lsl #2]
    add x27, x27, #1
    b build_brackets

build_done:
    cmp x26, #0
    b.ne bracket_error

    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x27, #0
    mov x28, #0
    mov x29, #0

run_loop:
    cmp x28, x21
    b.hs run_done
    add x0, x19, x28
    ldrb w0, [x0]

    cmp w0, #0x3e
    b.eq cmd_gt
    cmp w0, #0x3c
    b.eq cmd_lt
    cmp w0, #0x2b
    b.eq cmd_plus
    cmp w0, #0x2d
    b.eq cmd_minus
    cmp w0, #0x2e
    b.eq cmd_dot
    cmp w0, #0x2c
    b.eq cmd_comma
    cmp w0, #0x5b
    b.eq cmd_lbrack
    cmp w0, #0x5d
    b.eq cmd_rbrack

    add x28, x28, #1
    b run_loop

cmd_gt:
    add x29, x29, #1
    cmp x29, #30000
    b.hs data_error
    add x28, x28, #1
    b run_loop

cmd_lt:
    cmp x29, #0
    b.eq data_error
    sub x29, x29, #1
    add x28, x28, #1
    b run_loop

cmd_plus:
    add x0, x23, x29
    ldrb w1, [x0]
    add w1, w1, #1
    strb w1, [x0]
    add x28, x28, #1
    b run_loop

cmd_minus:
    add x0, x23, x29
    ldrb w1, [x0]
    sub w1, w1, #1
    strb w1, [x0]
    add x28, x28, #1
    b run_loop

cmd_dot:
    add x0, x23, x29
    ldrb w0, [x0]
    strb w0, [x26, x27]
    add x27, x27, #1
    add x28, x28, #1
    b run_loop

cmd_comma:
    cmp x21, x22
    b.hs comma_zero
    ldrb w0, [x21]
    add x21, x21, #1
    strb w0, [x23, x29]
    add x28, x28, #1
    b run_loop

comma_zero:
    mov w0, #0
    strb w0, [x23, x29]
    add x28, x28, #1
    b run_loop

cmd_lbrack:
    add x0, x23, x29
    ldrb w0, [x0]
    cmp w0, #0
    b.ne lbrack_continue
    ldr w1, [x24, x28, lsl #2]
    add x28, x1, #1
    b run_loop

lbrack_continue:
    add x28, x28, #1
    b run_loop

cmd_rbrack:
    add x0, x23, x29
    ldrb w0, [x0]
    cmp w0, #0
    b.eq rbrack_continue
    ldr w1, [x24, x28, lsl #2]
    add x28, x1, #1
    b run_loop

rbrack_continue:
    add x28, x28, #1
    b run_loop

run_done:
    cmp x27, #0
    b.eq exit_ok

    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #93
    svc #0

data_error:
    mov x0, #1
    mov x8, #93
    svc #0

.section .bss
    .align 4
input_buf:
    .space 65536

    .align 4
tape:
    .space 30000

    .align 4
jump:
    .space 65536 * 4

    .align 4
stack:
    .space 1000 * 4

    .align 4
output_buf:
    .space 65536
