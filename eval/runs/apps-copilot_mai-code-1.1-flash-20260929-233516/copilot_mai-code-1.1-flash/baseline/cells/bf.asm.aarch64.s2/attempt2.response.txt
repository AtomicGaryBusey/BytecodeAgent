.text
.globl _start

_start:
    mov x18, #0
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

read_loop:
    movz x28, #1, lsl #16
    cmp x18, x28
    b.hs read_done
    mov x0, #0
    add x1, x22, x18
    sub x2, x28, x18
    cmp x2, #4096
    b.le 1f
    mov x2, #4096
1:
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    b.eq read_done
    add x18, x18, x0
    b read_loop

read_done:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x23, x22
    add x24, x22, x18
scan_bang:
    cmp x19, x18
    b.hs no_bang
    ldrb w28, [x22, x19]
    cmp w28, #0x21
    b.eq bang_found
    add x19, x19, #1
    b scan_bang

bang_found:
    mov x20, x19
    add x19, x19, #1
    add x23, x22, x19
    b after_split

no_bang:
    mov x20, x18
    mov x23, x24

after_split:
    mov x21, x20
    mov x20, x18
    sub x20, x20, x21
    // x21 = program_len
    // x23 = input_pos
    // x24 = input_end
    // x18 = total input size
    adrp x25, jump_map
    add x25, x25, :lo12:jump_map
    adrp x26, bracket_stack
    add x26, x26, :lo12:bracket_stack
    mov x27, #0
    mov x28, #0
    mov x29, #0

build_loop:
    cmp x28, x21
    b.hs build_done
    ldrb w19, [x22, x28]
    cmp w19, #0x5b
    b.eq build_open
    cmp w19, #0x5d
    b.eq build_close
    add x28, x28, #1
    b build_loop

build_open:
    cmp x27, #1000
    b.hs bracket_error
    add x19, x26, x27, lsl #2
    str w28, [x19]
    add x27, x27, #1
    add x28, x28, #1
    b build_loop

build_close:
    cmp x27, #0
    b.eq bracket_error
    sub x27, x27, #1
    add x19, x26, x27, lsl #2
    ldr w29, [x19]
    add x19, x25, x29, lsl #2
    str w28, [x19]
    add x19, x25, x28, lsl #2
    str w29, [x19]
    add x28, x28, #1
    b build_loop

build_done:
    cmp x27, #0
    b.ne bracket_error

    mov x19, #0
    mov x20, #0
    adrp x21, tape
    add x21, x21, :lo12:tape
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x23, #0
    mov x24, #0
    adrp x25, input_buf
    add x25, x25, :lo12:input_buf
    mov x26, x23
    mov x27, #0
    mov x28, #0
    mov x29, #0
    adrp x30, jump_map
    add x30, x30, :lo12:jump_map

run_loop:
    cmp x19, x21
    b.hs run_done
    add x28, x25, x19
    ldrb w28, [x28]

    cmp w28, #0x3e
    b.eq cmd_gt
    cmp w28, #0x3c
    b.eq cmd_lt
    cmp w28, #0x2b
    b.eq cmd_plus
    cmp w28, #0x2d
    b.eq cmd_minus
    cmp w28, #0x2e
    b.eq cmd_dot
    cmp w28, #0x2c
    b.eq cmd_comma
    cmp w28, #0x5b
    b.eq cmd_lbrack
    cmp w28, #0x5d
    b.eq cmd_rbrack

    add x19, x19, #1
    b run_loop

cmd_gt:
    add x20, x20, #1
    movz x28, #0x7530
    cmp x20, x28
    b.hs data_error
    add x19, x19, #1
    b run_loop

cmd_lt:
    cmp x20, #0
    b.eq data_error
    sub x20, x20, #1
    add x19, x19, #1
    b run_loop

cmd_plus:
    add x28, x21, x20
    ldrb w28, [x28]
    add w28, w28, #1
    strb w28, [x21, x20]
    add x19, x19, #1
    b run_loop

cmd_minus:
    add x28, x21, x20
    ldrb w28, [x28]
    sub w28, w28, #1
    strb w28, [x21, x20]
    add x19, x19, #1
    b run_loop

cmd_dot:
    add x28, x21, x20
    ldrb w28, [x28]
    add x29, x22, x27
    strb w28, [x29]
    add x27, x27, #1
    add x19, x19, #1
    b run_loop

cmd_comma:
    cmp x23, x24
    b.hs comma_zero
    ldrb w28, [x23]
    add x23, x23, #1
    add x29, x21, x20
    strb w28, [x29]
    add x19, x19, #1
    b run_loop

comma_zero:
    add x29, x21, x20
    mov w28, #0
    strb w28, [x29]
    add x19, x19, #1
    b run_loop

cmd_lbrack:
    add x28, x21, x20
    ldrb w28, [x28]
    cmp w28, #0
    b.ne lbrack_continue
    add x28, x30, x19, lsl #2
    ldr w28, [x28]
    add x19, x28, #1
    b run_loop

lbrack_continue:
    add x19, x19, #1
    b run_loop

cmd_rbrack:
    add x28, x21, x20
    ldrb w28, [x28]
    cmp w28, #0
    b.eq rbrack_continue
    add x28, x30, x19, lsl #2
    ldr w28, [x28]
    add x19, x28, #1
    b run_loop

rbrack_continue:
    add x19, x19, #1
    b run_loop

run_done:
    cmp x27, #0
    b.eq exit_ok
    mov x0, #1
    mov x1, x22
    mov x2, x27
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #93
    svc #0

data_error:
    mov x0, #1
    mov x8, #93
    svc #0

.section .bss
    .align 4
input_buf:
    .space 65536

    .align 4
tape:
    .space 30000

    .align 4
jump_map:
    .space 65536 * 4

    .align 4
bracket_stack:
    .space 1000 * 4

    .align 4
output_buf:
    .space 65536
