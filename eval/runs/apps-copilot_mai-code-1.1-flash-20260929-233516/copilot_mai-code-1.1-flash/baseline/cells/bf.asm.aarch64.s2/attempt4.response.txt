.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ INPUT_LIMIT, 65536
.equ OUTPUT_LIMIT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_DEPTH, 1000

.equ BANG, 0x21
.equ LBRACK, 0x5b
.equ RBRACK, 0x5d
.equ GT, 0x3e
.equ LT, 0x3c
.equ PLUS, 0x2b
.equ MINUS, 0x2d
.equ DOT, 0x2e
.equ COMMA, 0x2c

_start:
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf
    mov x19, #0

read_loop:
    movz x11, #1, lsl #16
    cmp x19, x11
    b.hs read_done

    mov x0, #0
    add x1, x22, x19
    mov x2, #4096
    sub x3, x11, x19
    cmp x3, #4096
    b.hs 1f
    mov x2, x3
1:
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    b.lt fail_exit
    b.eq read_done

    add x19, x19, x0
    movz x11, #1, lsl #16
    cmp x19, x11
    b.hs read_done
    b read_loop

read_done:
    mov x20, #0
    cmp x20, x19
    b.hs no_bang

scan_loop:
    add x23, x22, x20
    ldrb w24, [x23]
    cmp w24, #BANG
    b.eq bang_found
    add x20, x20, #1
    cmp x20, x19
    b.lo scan_loop

no_bang:
    mov x20, x19
    mov x9, x22
    add x9, x9, x19
    mov x10, x9
    b validate_start

bang_found:
    mov x20, x20
    mov x9, x22
    add x9, x9, x20
    add x9, x9, #1
    mov x10, x22
    add x10, x10, x19
    b validate_start

validate_start:
    adrp x13, bracket_stack
    add x13, x13, :lo12:bracket_stack
    adrp x14, jump_map
    add x14, x14, :lo12:jump_map

    mov x15, #0
    mov x16, #0

validate_loop:
    cmp x16, x20
    b.hs validate_done

    add x17, x22, x16
    ldrb w18, [x17]

    cmp w18, #LBRACK
    b.eq validate_open
    cmp w18, #RBRACK
    b.eq validate_close

validate_next:
    add x16, x16, #1
    b validate_loop

validate_open:
    cmp x15, #MAX_DEPTH
    b.hs fail_exit
    add x17, x13, x15, lsl #2
    str w16, [x17]
    add x15, x15, #1
    b validate_next

validate_close:
    cmp x15, #0
    b.eq fail_exit
    sub x15, x15, #1
    add x17, x13, x15, lsl #2
    ldr w18, [x17]
    add x17, x14, x18, lsl #2
    str w16, [x17]
    add x17, x14, x16, lsl #2
    str w18, [x17]
    b validate_next

validate_done:
    cmp x15, #0
    b.ne fail_exit

    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    adrp x25, jump_map
    add x25, x25, :lo12:jump_map

    mov x26, #0
    mov x27, #0
    mov x28, #0

run_loop:
    cmp x26, x20
    b.hs run_done

    add x29, x22, x26
    ldrb w30, [x29]

    cmp w30, #GT
    b.eq cmd_gt
    cmp w30, #LT
    b.eq cmd_lt
    cmp w30, #PLUS
    b.eq cmd_plus
    cmp w30, #MINUS
    b.eq cmd_minus
    cmp w30, #DOT
    b.eq cmd_dot
    cmp w30, #COMMA
    b.eq cmd_comma
    cmp w30, #LBRACK
    b.eq cmd_lbrack
    cmp w30, #RBRACK
    b.eq cmd_rbrack

    add x26, x26, #1
    b run_loop

cmd_gt:
    add x27, x27, #1
    cmp x27, #TAPE_SIZE
    b.hs fail_exit
    add x26, x26, #1
    b run_loop

cmd_lt:
    cmp x27, #0
    b.eq fail_exit
    sub x27, x27, #1
    add x26, x26, #1
    b run_loop

cmd_plus:
    add x29, x23, x27
    ldrb w30, [x29]
    add w30, w30, #1
    strb w30, [x29]
    add x26, x26, #1
    b run_loop

cmd_minus:
    add x29, x23, x27
    ldrb w30, [x29]
    sub w30, w30, #1
    strb w30, [x29]
    add x26, x26, #1
    b run_loop

cmd_dot:
    add x29, x23, x27
    ldrb w30, [x29]
    add x29, x24, x28
    strb w30, [x29]
    add x28, x28, #1
    movz x11, #1, lsl #16
    cmp x28, x11
    b.hs fail_exit
    add x26, x26, #1
    b run_loop

cmd_comma:
    cmp x9, x10
    b.hs comma_zero
    ldrb w30, [x9], #1
    add x29, x23, x27
    strb w30, [x29]
    add x26, x26, #1
    b run_loop

comma_zero:
    add x29, x23, x27
    mov w30, #0
    strb w30, [x29]
    add x26, x26, #1
    b run_loop

cmd_lbrack:
    add x29, x23, x27
    ldrb w30, [x29]
    cmp w30, #0
    b.ne lbrack_skip
    add x29, x25, x26, lsl #2
    ldr w30, [x29]
    add x26, x30, #1
    b run_loop

lbrack_skip:
    add x26, x26, #1
    b run_loop

cmd_rbrack:
    add x29, x23, x27
    ldrb w30, [x29]
    cmp w30, #0
    b.eq rbrack_skip
    add x29, x25, x26, lsl #2
    ldr w30, [x29]
    add x26, x30, #1
    b run_loop

rbrack_skip:
    add x26, x26, #1
    b run_loop

run_done:
    cmp x28, #0
    b.eq exit_ok

    mov x0, #1
    mov x1, x24
    mov x2, x28
    mov x8, #SYS_WRITE
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

fail_exit:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.section .bss
    .align 4
input_buf:
    .space 65536

    .align 4
tape:
    .space 30000

    .align 4
jump_map:
    .space 262144

    .align 4
bracket_stack:
    .space 4000

    .align 4
output_buf:
    .space 65536
