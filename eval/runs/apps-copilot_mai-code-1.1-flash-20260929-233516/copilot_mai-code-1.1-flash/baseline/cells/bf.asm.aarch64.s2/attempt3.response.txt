.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_LIMIT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_DEPTH, 1000

.equ CH_BANG, 0x21
.equ CH_LBRACK, 0x5b
.equ CH_RBRACK, 0x5d
.equ CH_GT, 0x3e
.equ CH_LT, 0x3c
.equ CH_PLUS, 0x2b
.equ CH_MINUS, 0x2d
.equ CH_DOT, 0x2e
.equ CH_COMMA, 0x2c

_start:
    mov x18, #0
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf

read_loop:
    mov x0, #0
    add x1, x19, x18
    mov x2, #4096
    mov x3, #INPUT_LIMIT
    sub x3, x3, x18
    cmp x3, #4096
    b.hi 1f
    mov x2, x3
1:
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.lt fail_exit
    b.eq read_done
    add x18, x18, x0
    cmp x18, #INPUT_LIMIT
    b.hs read_done
    b read_loop

read_done:
    mov x22, #0
    cmp x22, x18
    b.hs no_bang

scan_bang:
    add x23, x19, x22
    ldrb w24, [x23]
    cmp w24, #CH_BANG
    b.eq bang_found
    add x22, x22, #1
    cmp x22, x18
    b.lo scan_bang

no_bang:
    mov x16, x18
    mov x20, x19
    add x21, x19, x18
    b after_bang

bang_found:
    mov x16, x22
    add x22, x22, #1
    mov x20, x19
    add x20, x20, x22
    add x21, x19, x18

after_bang:
    adrp x23, bracket_stack
    add x23, x23, :lo12:bracket_stack
    adrp x24, jump_map
    add x24, x24, :lo12:jump_map

    mov x22, #0
    mov x17, #0

build_loop:
    cmp x22, x16
    b.hs build_done

    add x25, x19, x22
    ldrb w26, [x25]

    cmp w26, #CH_LBRACK
    b.eq build_open
    cmp w26, #CH_RBRACK
    b.eq build_close

build_next:
    add x22, x22, #1
    b build_loop

build_open:
    cmp x17, #MAX_DEPTH
    b.hs bracket_fail
    add x27, x23, x17, lsl #2
    str w22, [x27]
    add x17, x17, #1
    b build_next

build_close:
    cmp x17, #0
    b.eq bracket_fail
    sub x17, x17, #1
    add x27, x23, x17, lsl #2
    ldr w28, [x27]

    add x27, x24, x28, lsl #2
    str w22, [x27]

    add x27, x24, x22, lsl #2
    str w28, [x27]

    b build_next

build_done:
    cmp x17, #0
    b.ne bracket_fail

    mov x22, #0
    mov x23, #0
    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    mov x26, #0
    adrp x27, jump_map
    add x27, x27, :lo12:jump_map

run_loop:
    cmp x22, x16
    b.hs run_done

    add x28, x19, x22
    ldrb w29, [x28]

    cmp w29, #CH_GT
    b.eq cmd_gt
    cmp w29, #CH_LT
    b.eq cmd_lt
    cmp w29, #CH_PLUS
    b.eq cmd_plus
    cmp w29, #CH_MINUS
    b.eq cmd_minus
    cmp w29, #CH_DOT
    b.eq cmd_dot
    cmp w29, #CH_COMMA
    b.eq cmd_comma
    cmp w29, #CH_LBRACK
    b.eq cmd_lbrack
    cmp w29, #CH_RBRACK
    b.eq cmd_rbrack

    add x22, x22, #1
    b run_loop

cmd_gt:
    add x23, x23, #1
    cmp x23, #TAPE_SIZE
    b.hs data_fail
    add x22, x22, #1
    b run_loop

cmd_lt:
    cmp x23, #0
    b.eq data_fail
    sub x23, x23, #1
    add x22, x22, #1
    b run_loop

cmd_plus:
    add x28, x24, x23
    ldrb w29, [x28]
    add w29, w29, #1
    strb w29, [x28]
    add x22, x22, #1
    b run_loop

cmd_minus:
    add x28, x24, x23
    ldrb w29, [x28]
    sub w29, w29, #1
    strb w29, [x28]
    add x22, x22, #1
    b run_loop

cmd_dot:
    add x28, x24, x23
    ldrb w29, [x28]
    add x28, x25, x26
    strb w29, [x28]
    add x26, x26, #1
    add x22, x22, #1
    b run_loop

cmd_comma:
    cmp x20, x21
    b.hs comma_zero
    ldrb w29, [x20], #1
    add x28, x24, x23
    strb w29, [x28]
    add x22, x22, #1
    b run_loop

comma_zero:
    add x28, x24, x23
    mov w29, #0
    strb w29, [x28]
    add x22, x22, #1
    b run_loop

cmd_lbrack:
    add x28, x24, x23
    ldrb w29, [x28]
    cmp w29, #0
    b.ne lbrack_skip
    add x28, x27, x22, lsl #2
    ldr w29, [x28]
    add x22, x29, #1
    b run_loop

lbrack_skip:
    add x22, x22, #1
    b run_loop

cmd_rbrack:
    add x28, x24, x23
    ldrb w29, [x28]
    cmp w29, #0
    b.eq rbrack_skip
    add x28, x27, x22, lsl #2
    ldr w29, [x28]
    add x22, x29, #1
    b run_loop

rbrack_skip:
    add x22, x22, #1
    b run_loop

run_done:
    cmp x26, #0
    b.eq exit_ok
    mov x0, #1
    mov x1, x25
    mov x2, x26
    mov x8, #SYS_WRITE
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

fail_exit:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

bracket_fail:
    b fail_exit

data_fail:
    b fail_exit

.section .bss
    .align 4
input_buf:
    .space 65536

    .align 4
tape:
    .space 30000

    .align 4
jump_map:
    .space 65536 * 4

    .align 4
bracket_stack:
    .space 1000 * 4

    .align 4
output_buf:
    .space 65536
