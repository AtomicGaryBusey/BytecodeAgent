.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    b read_chunk

read_chunk:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    movz x2, #1
    lsl x2, x2, #16
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done
    mov x23, x0
    mov x24, x1

process_chunk:
    cmp x23, #0
    b.eq read_chunk
    ldrb w9, [x24], #1
    sub x23, x23, #1
    add x21, x21, #1

    cmp w9, #0x0A
    b.eq newline_byte

    cmp w9, #0x20
    b.eq whitespace
    cmp w9, #0x09
    b.eq whitespace
    cmp w9, #0x0B
    b.eq whitespace
    cmp w9, #0x0C
    b.eq whitespace
    cmp w9, #0x0D
    b.eq whitespace

    cmp x22, #0
    b.ne process_chunk
    add x20, x20, #1
    mov x22, #1
    b process_chunk

whitespace:
    mov x22, #0
    b process_chunk

newline_byte:
    add x19, x19, #1
    mov x22, #0
    b process_chunk

done:
    mov x0, x19
    bl print_uint

    mov x0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x20
    bl print_uint

    mov x0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x21
    bl print_uint

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_uint:
    sub sp, sp, #32
    mov x9, x0
    add x10, sp, #31
    mov x11, #0

    cmp x9, #0
    b.ne 1f
    sub x10, x10, #1
    mov w12, #'0'
    strb w12, [x10]
    mov x11, #1
    b 2f

1:
    mov x13, #10
    b 3f

3:
    cmp x9, #0
    b.eq 2f
    udiv x14, x9, x13
    msub x15, x14, x13, x9
    add w15, w15, #'0'
    sub x10, x10, #1
    strb w15, [x10]
    mov x9, x14
    add x11, x11, #1
    b 3b

2:
    mov x0, #1
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0
    add sp, sp, #32
    ret

.section .rodata
space:
    .byte 0x20

newline:
    .byte 0x0A

.section .bss
    .align 4
inbuf:
    .space 65536
