.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    movz x2, #1
    lsl x2, x2, #16
    mov x8, #63
    svc #0
    cmp x0, #0
    beq done_read
    mov x3, x0
    add x21, x21, x3
    mov x4, x1
    mov x5, #0

process_loop:
    cmp x5, x3
    b.hs read_loop
    ldrb w9, [x4], #1
    add x5, x5, #1

    cmp w9, #0x0A
    b.eq line_case

    cmp w9, #0x20
    b.eq whitespace

    cmp w9, #0x09
    b.eq whitespace

    cmp w9, #0x0B
    b.eq whitespace

    cmp w9, #0x0C
    b.eq whitespace

    cmp w9, #0x0D
    b.eq whitespace

    cmp x22, #0
    b.ne process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop

whitespace:
    mov x22, #0
    b process_loop

line_case:
    add x19, x19, #1
    mov x22, #0
    b process_loop

done_read:
    mov x0, x19
    bl print_uint

    mov x0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x20
    bl print_uint

    mov x0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x21
    bl print_uint

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_uint:
    sub sp, sp, #32
    add x9, sp, #31
    mov x10, x0
    mov x11, #0

    cmp x10, #0
    b.ne 1f
    sub x9, x9, #1
    mov w12, #'0'
    strb w12, [x9]
    mov x11, #1
    b 2f

1:
    cmp x10, #0
    b.eq 2f
    udiv x12, x10, #10
    msub x13, x12, #10, x10
    add w13, w13, #'0'
    sub x9, x9, #1
    strb w13, [x9]
    mov x10, x12
    add x11, x11, #1
    b 1b

2:
    mov x0, #1
    mov x1, x9
    mov x2, x11
    mov x8, #64
    svc #0
    add sp, sp, #32
    ret

.section .rodata
space:
    .byte 0x20

newline:
    .byte 0x0A

.section .bss
    .align 4
inbuf:
    .space 65536
