.text
.globl _start

_start:
    mov x11, xzr
    mov x12, xzr
    mov x13, xzr
    mov x14, xzr

    adrp x10, input_buf
    add x10, x10, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x10
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x15, x0
    cmp x15, #0
    beq done_read

    mov x16, x10
    mov x17, x15

process_bytes:
    ldrb w18, [x16], #1
    add x13, x13, #1

    cmp w18, #10
    b.ne 1f
    add x11, x11, #1
1:
    cmp w18, #0x20
    beq whitespace
    cmp w18, #0x09
    beq whitespace
    cmp w18, #0x0A
    beq whitespace
    cmp w18, #0x0B
    beq whitespace
    cmp w18, #0x0C
    beq whitespace
    cmp w18, #0x0D
    beq whitespace

    mov x14, #1
    b next_byte

whitespace:
    cbnz x14, word_seen
    b next_byte

word_seen:
    add x12, x12, #1
    mov x14, xzr

next_byte:
    subs x17, x17, #1
    b.ne process_bytes
    b read_loop

done_read:
    cbnz x14, final_word
    b format_output

final_word:
    add x12, x12, #1

format_output:
    adrp x19, output_buf
    add x19, x19, :lo12:output_buf

    mov x0, x11
    mov x20, x19
    add x20, x20, #120
    mov x21, #0
    cbz x0, lines_zero
    b lines_loop

lines_zero:
    mov x21, #1
    sub x20, x20, #1
    mov w22, #48
    strb w22, [x20]
    b lines_done

lines_loop:
    mov x22, #10
    udiv x23, x0, x22
    msub x24, x23, x22, x0
    add w24, w24, #48
    sub x20, x20, #1
    strb w24, [x20]
    mov x0, x23
    add x21, x21, #1
    cbnz x0, lines_loop

lines_done:
    mov x25, x19
    mov x22, x20
    mov x23, x21
copy_lines:
    ldrb w24, [x22], #1
    strb w24, [x25], #1
    subs x23, x23, #1
    b.ne copy_lines
    mov w24, #32
    strb w24, [x25], #1
    mov x19, x25

    mov x0, x12
    mov x20, x19
    add x20, x20, #120
    mov x21, #0
    cbz x0, words_zero
    b words_loop

words_zero:
    mov x21, #1
    sub x20, x20, #1
    mov w22, #48
    strb w22, [x20]
    b words_done

words_loop:
    mov x22, #10
    udiv x23, x0, x22
    msub x24, x23, x22, x0
    add w24, w24, #48
    sub x20, x20, #1
    strb w24, [x20]
    mov x0, x23
    add x21, x21, #1
    cbnz x0, words_loop

words_done:
    mov x25, x19
    mov x22, x20
    mov x23, x21
copy_words:
    ldrb w24, [x22], #1
    strb w24, [x25], #1
    subs x23, x23, #1
    b.ne copy_words
    mov w24, #32
    strb w24, [x25], #1
    mov x19, x25

    mov x0, x13
    mov x20, x19
    add x20, x20, #120
    mov x21, #0
    cbz x0, bytes_zero
    b bytes_loop

bytes_zero:
    mov x21, #1
    sub x20, x20, #1
    mov w22, #48
    strb w22, [x20]
    b bytes_done

bytes_loop:
    mov x22, #10
    udiv x23, x0, x22
    msub x24, x23, x22, x0
    add w24, w24, #48
    sub x20, x20, #1
    strb w24, [x20]
    mov x0, x23
    add x21, x21, #1
    cbnz x0, bytes_loop

bytes_done:
    mov x25, x19
    mov x22, x20
    mov x23, x21
copy_bytes:
    ldrb w24, [x22], #1
    strb w24, [x25], #1
    subs x23, x23, #1
    b.ne copy_bytes
    mov w24, #10
    strb w24, [x25], #1
    mov x19, x25
    add x19, x19, #1

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x2, x19
    sub x2, x2, x1

write_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    cbz x0, write_fail
    cmp x0, x2
    beq exit_ok
    sub x2, x2, x0
    add x1, x1, x0
    b write_loop

write_fail:
    mov x0, #1
    mov x8, #93
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.balign 8
input_buf:
    .zero 65536

.balign 8
output_buf:
    .zero 128
