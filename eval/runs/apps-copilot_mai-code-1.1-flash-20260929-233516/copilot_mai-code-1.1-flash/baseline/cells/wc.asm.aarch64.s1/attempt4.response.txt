.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adr x1, input_buf
    mov x2, #BUF_SIZE
    mov x8, SYS_read
    svc #0
    mov x23, x0
    cmp x23, #0
    b.eq done

    mov x24, #0
process_loop:
    ldrb w25, [x1, x24]
    cmp w25, #0x20
    b.eq whitespace
    cmp w25, #0x09
    b.eq whitespace
    cmp w25, #0x0A
    b.eq newline
    cmp w25, #0x0B
    b.eq whitespace
    cmp w25, #0x0C
    b.eq whitespace
    cmp w25, #0x0D
    b.eq whitespace
    b non_whitespace

newline:
    add x19, x19, #1
whitespace:
    mov x22, #0
    b after_byte

non_whitespace:
    cbz x22, word_start
    b after_byte

word_start:
    add x20, x20, #1
    mov x22, #1

after_byte:
    add x24, x24, #1
    subs x23, x23, #1
    b.ne process_loop
    b read_loop

done:
    adr x23, output_buf
    mov x0, x23
    mov x1, x19
    bl append_decimal
    mov w2, #' '
    strb w2, [x0], #1
    mov x1, x20
    bl append_decimal
    mov w2, #' '
    strb w2, [x0], #1
    mov x1, x21
    bl append_decimal
    mov w2, #10
    strb w2, [x0], #1

    mov x24, x0
    sub x24, x24, x23
    mov x25, x23
write_loop:
    mov x0, #1
    mov x1, x25
    mov x2, x24
    mov x8, SYS_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    sub x24, x24, x0
    add x25, x25, x0
    cbnz x24, write_loop
    b exit_ok

append_decimal:
    mov x2, x0
    mov x3, x1
    adr x4, tmp_digits
    add x4, x4, #32
    mov x10, #0
    mov x11, #10
    cbz x3, zero_digit

digit_loop:
    udiv x5, x3, x11
    msub x6, x5, x11, x3
    add w6, w6, #48
    sub x4, x4, #1
    strb w6, [x4]
    add x10, x10, #1
    mov x3, x5
    cbnz x3, digit_loop
    b copy_digits

zero_digit:
    mov w6, #48
    sub x4, x4, #1
    strb w6, [x4]
    mov x10, #1

copy_digits:
    mov x5, x4
copy_loop:
    ldrb w6, [x5], #1
    strb w6, [x2], #1
    subs x10, x10, #1
    b.ne copy_loop
    mov x0, x2
    ret

exit_ok:
    mov x0, #0
    mov x8, SYS_exit
    svc #0

.section .bss
.balign 8
input_buf:
    .zero BUF_SIZE

.balign 8
output_buf:
    .zero 64

.balign 8
tmp_digits:
    .zero 32
