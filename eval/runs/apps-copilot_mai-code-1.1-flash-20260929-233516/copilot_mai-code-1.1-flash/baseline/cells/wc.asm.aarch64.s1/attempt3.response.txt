.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adr x1, input_buf
    mov x2, #65536
    mov x8, SYS_read
    svc #0
    mov x23, x0
    cmp x23, #0
    b.eq done_read

    adr x24, input_buf
    mov x25, x23

process_byte:
    ldrb w26, [x24], #1
    add x21, x21, #1

    cmp w26, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w26, #32
    b.eq whitespace
    cmp w26, #9
    b.eq whitespace
    cmp w26, #11
    b.eq whitespace
    cmp w26, #12
    b.eq whitespace
    cmp w26, #13
    b.eq whitespace

    cbnz x22, 2f
    add x20, x20, #1
    mov x22, #1
2:
    b next_byte

whitespace:
    mov x22, #0

next_byte:
    subs x25, x25, #1
    b.ne process_byte
    b read_loop

done_read:
    adr x0, output_buf
    mov x1, x19
    bl append_u32
    mov x1, #32
    strb w1, [x0], #1
    mov x1, x20
    bl append_u32
    mov x1, #32
    strb w1, [x0], #1
    mov x1, x21
    bl append_u32
    mov w1, #10
    strb w1, [x0], #1

    mov x23, x0
    adr x1, output_buf
    sub x2, x23, x1

write_loop:
    mov x0, #1
    mov x8, SYS_write
    svc #0
    cmp x0, #0
    ble write_error
    subs x2, x2, x0
    b.eq exit_ok
    add x1, x1, x0
    b write_loop

write_error:
    mov x0, #1
    mov x8, SYS_exit
    svc #0

exit_ok:
    mov x0, #0
    mov x8, SYS_exit
    svc #0

append_u32:
    mov x3, x0
    mov x4, x1
    adr x5, tmp_digits
    add x5, x5, #32
    mov x6, #0
    mov x7, x4
    cbz x7, zero_digits

digit_loop:
    mov x8, #10
    udiv x9, x7, x8
    msub x10, x9, x8, x7
    add w10, w10, #48
    sub x5, x5, #1
    strb w10, [x5]
    add x6, x6, #1
    mov x7, x9
    cbnz x7, digit_loop
    b copy_digits

zero_digits:
    mov w10, #48
    sub x5, x5, #1
    strb w10, [x5]
    mov x6, #1

copy_digits:
    mov x7, x5
    mov x8, x3

copy_inner:
    ldrb w9, [x7], #1
    strb w9, [x8], #1
    subs x6, x6, #1
    b.ne copy_inner
    mov x0, x8
    ret

.section .bss
.balign 8
input_buf:
    .zero 65536

.balign 8
output_buf:
    .zero 64

.balign 8
tmp_digits:
    .zero 32
