.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

_start:
    mov x11, #0
    mov x12, #0
    mov x13, #0
    mov x14, #0

read_loop:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x15, x0
    cmp x15, #0
    b.eq done_read

    mov x16, x1
    mov x17, x15

process_byte:
    ldrb w18, [x16], #1
    add x13, x13, #1

    cmp w18, #10
    b.ne 1f
    add x11, x11, #1
1:
    cmp w18, #32
    b.eq whitespace
    cmp w18, #9
    b.eq whitespace
    cmp w18, #11
    b.eq whitespace
    cmp w18, #12
    b.eq whitespace
    cmp w18, #13
    b.eq whitespace

    mov x14, #1
    b next_byte

whitespace:
    cbnz x14, word_done
    b next_byte

word_done:
    add x12, x12, #1
    mov x14, #0

next_byte:
    subs x17, x17, #1
    b.ne process_byte
    b read_loop

done_read:
    cbnz x14, final_word
    b format_output

final_word:
    add x12, x12, #1

format_output:
    adrp x19, output_buf
    add x19, x19, :lo12:output_buf

    mov x0, x11
    mov x1, x19
    mov x2, #32
    bl emit_dec
    mov x19, x0

    mov x0, x12
    mov x1, x19
    mov x2, #32
    bl emit_dec
    mov x19, x0

    mov x0, x13
    mov x1, x19
    mov x2, #10
    bl emit_dec
    mov x19, x0

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x19, x1
    mov x3, x1

write_loop:
    mov x0, #1
    mov x1, x3
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_fail
    sub x2, x2, x0
    cbz x2, exit_ok
    add x3, x3, x0
    b write_loop

write_fail:
    mov x0, #0
    mov x8, #93
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

emit_dec:
    adrp x9, temp_buf
    add x9, x9, :lo12:temp_buf
    add x10, x9, #32
    mov x11, #0

    cbz x0, 1f

0:
    mov x12, #10
    udiv x13, x0, x12
    msub x14, x13, x12, x0
    add w14, w14, #48
    sub x10, x10, #1
    strb w14, [x10]
    mov x0, x13
    add x11, x11, #1
    cbnz x0, 0b

1:
    mov x15, x1
    mov x16, x10
    mov x17, x11
    cbz x17, 2f

3:
    ldrb w14, [x16], #1
    strb w14, [x15], #1
    subs x17, x17, #1
    b.ne 3b

2:
    mov w14, w2
    strb w14, [x15], #1
    mov x0, x15
    ret

.section .bss
.balign 8
input_buf:
    .zero 65536

.balign 8
output_buf:
    .zero 64

.balign 8
temp_buf:
    .zero 32
