.text
.globl _start

_start:
    mov x10, #0
    mov x11, #0
    mov x12, #0
    mov x13, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq .Ldone_read
    add x12, x12, x0
    mov x14, x1
    add x15, x1, x0

process_chunk:
    cmp x14, x15
    b.hs read_loop

    ldrb w16, [x14], #1

    cmp w16, #0x0A
    b.ne .Lnot_newline
    add x10, x10, #1

.Lnot_newline:
    cmp w16, #0x20
    b.eq .Lwhitespace
    cmp w16, #0x09
    b.eq .Lwhitespace
    cmp w16, #0x0A
    b.eq .Lwhitespace
    cmp w16, #0x0B
    b.eq .Lwhitespace
    cmp w16, #0x0C
    b.eq .Lwhitespace
    cmp w16, #0x0D
    b.eq .Lwhitespace

    cmp x13, #0
    b.ne process_chunk
    add x11, x11, #1
    mov x13, #1
    b process_chunk

.Lwhitespace:
    mov x13, #0
    b process_chunk

.Ldone_read:
    mov x19, x10
    mov x20, x11
    mov x21, x12

    mov x0, x19
    bl print_dec

    mov x0, #1
    adr x1, space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x20
    bl print_dec

    mov x0, #1
    adr x1, space
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, x21
    bl print_dec

    mov x0, #1
    adr x1, newline
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_dec:
    sub sp, sp, #32
    mov x9, sp
    mov x10, #0

    cmp x0, #0
    b.ne .Lconvert_loop
    mov w11, #'0'
    strb w11, [sp]
    mov x10, #1
    b .Lprint_reverse

.Lconvert_loop:
    cmp x0, #0
    b.eq .Lprint_reverse

    mov x11, #10
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add x13, x13, #'0'
    strb w13, [sp, x10]
    add x10, x10, #1
    mov x0, x12
    b .Lconvert_loop

.Lprint_reverse:
    add x9, sp, x10
    sub x9, x9, #1
    mov x11, x10

.Lprint_loop:
    cmp x11, #0
    b.eq .Lprint_done

    ldrb w0, [x9]
    mov x1, x9
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0

    sub x9, x9, #1
    sub x11, x11, #1
    b .Lprint_loop

.Lprint_done:
    add sp, sp, #32
    ret

.section .rodata
space:
    .byte 0x20

newline:
    .byte 0x0A

.section .bss
    .balign 16
input_buf:
    .zero 4096
