.globl _start

.section .text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jz end_of_input
    mov rbx, rax
    xor rcx, rcx

process_chunk:
    cmp rcx, rbx
    jae read_loop
    movzx r8d, byte ptr [rsi + rcx]
    inc r14

    cmp r8b, 0x0A
    jne not_newline
    inc r12

not_newline:
    mov al, r8b
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0A
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace

    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte

whitespace:
    xor r15, r15

next_byte:
    inc rcx
    jmp process_chunk

end_of_input:
    test r15, r15
    jz generate_output
    inc r13

generate_output:
    lea rdi, [rip + output_buffer]
    xor rbx, rbx

    mov rax, r12
    call convert_uint
    mov r8, rsi
    mov rdx, rcx
copy_lines:
    mov al, byte ptr [r8]
    mov byte ptr [rdi + rbx], al
    inc r8
    inc rbx
    dec rdx
    jnz copy_lines
    mov byte ptr [rdi + rbx], ' '
    inc rbx

    mov rax, r13
    call convert_uint
    mov r8, rsi
    mov rdx, rcx
copy_words:
    mov al, byte ptr [r8]
    mov byte ptr [rdi + rbx], al
    inc r8
    inc rbx
    dec rdx
    jnz copy_words
    mov byte ptr [rdi + rbx], ' '
    inc rbx

    mov rax, r14
    call convert_uint
    mov r8, rsi
    mov rdx, rcx
copy_bytes:
    mov al, byte ptr [r8]
    mov byte ptr [rdi + rbx], al
    inc r8
    inc rbx
    dec rdx
    jnz copy_bytes
    mov byte ptr [rdi + rbx], 0x0A
    inc rbx

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

convert_uint:
    mov r8, rax
    lea rsi, [rip + temp_buffer + 31]
    xor rcx, rcx
    cmp r8, 0
    jne convert_uint_loop
    mov byte ptr [rsi], '0'
    mov rcx, 1
    ret

convert_uint_loop:
    mov rax, r8
    mov r9, 10
    jmp convert_uint_loop_body

convert_uint_loop_body:
    test rax, rax
    jz convert_uint_done
    xor rdx, rdx
    div r9
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc rcx
    jmp convert_uint_loop_body

convert_uint_done:
    ret

.section .bss
.align 16
input_buffer:
    .space 65536
.align 16
output_buffer:
    .space 64
.align 16
temp_buffer:
    .space 32
