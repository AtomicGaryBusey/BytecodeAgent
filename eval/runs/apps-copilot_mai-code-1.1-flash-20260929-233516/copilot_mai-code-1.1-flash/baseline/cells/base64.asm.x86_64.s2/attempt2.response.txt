.globl _start

.equ INPUT_LIMIT, 49152
.equ LINE_WIDTH, 76

.text

_start:
    xor r15, r15

read_loop:
    cmp r15, INPUT_LIMIT
    je read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, INPUT_LIMIT
    sub rdx, r15
    syscall
    cmp rax, 0
    jl read_error
    je read_done
    add r15, rax
    cmp r15, INPUT_LIMIT
    je read_done
    jmp read_loop

read_error:
    mov rax, 60
    mov rdi, 1
    syscall

read_done:
    test r15, r15
    jz exit_zero

    xor r12, r12
    lea r13, [rip + input_buf]
    lea r14, [rip + out_buf]
    xor r11, r11
    xor r10, r10
    lea r9, [rip + base64_alphabet]

encode_loop:
    mov rax, r15
    sub rax, r12
    cmp rax, 3
    jb final_group

    movzx eax, byte ptr [r13 + r12]
    movzx ecx, byte ptr [r13 + r12 + 1]
    movzx edx, byte ptr [r13 + r12 + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    and ecx, 63
    movzx ebx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    movzx ebx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    movzx ebx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov ecx, eax
    and ecx, 63
    movzx ebx, byte ptr [r9 + rcx]
    mov byte ptr [r14 + r11], bl
    inc r11

    add r10, 4
    cmp r10, LINE_WIDTH
    jne 1f
    mov byte ptr [r14 + r11], 10
    inc r11
    xor r10, r10
1:
    add r12, 3
    jmp encode_loop

final_group:
    mov rax, r15
    sub rax, r12

    cmp rax, 2
    je final_two
    cmp rax, 1
    je final_one
    jmp finalize_output

final_two:
    movzx eax, byte ptr [r13 + r12]
    movzx ecx, byte ptr [r13 + r12 + 1]

    mov edx, eax
    shr edx, 2
    and edx, 63
    movzx ebx, byte ptr [r9 + rdx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov esi, ecx
    shr esi, 4
    and esi, 15
    or edx, esi
    movzx ebx, byte ptr [r9 + rdx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov edx, ecx
    and edx, 15
    shl edx, 2
    movzx ebx, byte ptr [r9 + rdx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov byte ptr [r14 + r11], 61
    inc r11

    add r10, 4
    cmp r10, LINE_WIDTH
    jne 2f
    mov byte ptr [r14 + r11], 10
    inc r11
    xor r10, r10
2:
    jmp finalize_output

final_one:
    movzx eax, byte ptr [r13 + r12]

    mov edx, eax
    shr edx, 2
    and edx, 63
    movzx ebx, byte ptr [r9 + rdx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov edx, eax
    and edx, 3
    shl edx, 4
    movzx ebx, byte ptr [r9 + rdx]
    mov byte ptr [r14 + r11], bl
    inc r11

    mov byte ptr [r14 + r11], 61
    inc r11
    mov byte ptr [r14 + r11], 61
    inc r11

    add r10, 4
    cmp r10, LINE_WIDTH
    jne 3f
    mov byte ptr [r14 + r11], 10
    inc r11
    xor r10, r10
3:
    jmp finalize_output

finalize_output:
    test r10, r10
    je output_write

    mov byte ptr [r14 + r11], 10
    inc r11

output_write:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r11
    syscall
    jmp exit_zero

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
.align 16
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buf:
    .space 49152

.align 16
out_buf:
    .space 70000
