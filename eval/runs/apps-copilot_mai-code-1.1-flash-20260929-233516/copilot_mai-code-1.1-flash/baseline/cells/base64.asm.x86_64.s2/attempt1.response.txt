.globl _start

.equ INPUT_LIMIT, 49152
.equ LINE_WIDTH, 76

.text
_start:
    xor rbx, rbx

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf + rbx]
    mov rdx, INPUT_LIMIT
    sub rdx, rbx
    syscall
    cmp rax, 0
    jl read_error
    je read_done
    add rbx, rax
    cmp rbx, INPUT_LIMIT
    je read_done
    jmp read_loop

read_error:
    mov rax, 60
    mov rdi, 1
    syscall

read_done:
    test rbx, rbx
    jz exit_zero

    lea r10, [rip + out_buf]
    xor r15, r15
    xor r14, r14
    mov r12, rbx
    xor rbx, rbx
    lea r11, [rip + base64_alphabet]

loop_encode:
    mov eax, ebx
    add eax, 3
    cmp eax, r12d
    jg final_group

    movzx eax, byte ptr [rip + input_buf + rbx]
    movzx ecx, byte ptr [rip + input_buf + rbx + 1]
    movzx edx, byte ptr [rip + input_buf + rbx + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    and ecx, 63
    movzx r13d, byte ptr [r11 + rcx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 1f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
1:
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    movzx r13d, byte ptr [r11 + rcx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 2f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
2:
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    movzx r13d, byte ptr [r11 + rcx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 3f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
3:
    mov ecx, eax
    and ecx, 63
    movzx r13d, byte ptr [r11 + rcx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 4f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
4:
    add rbx, 3
    jmp loop_encode

final_group:
    mov eax, r12d
    sub eax, ebx

    cmp eax, 2
    je final_two
    cmp eax, 1
    je final_one
    jmp finish_encode

final_two:
    movzx eax, byte ptr [rip + input_buf + rbx]
    movzx ecx, byte ptr [rip + input_buf + rbx + 1]

    mov edx, eax
    shr edx, 2
    and edx, 63
    movzx r13d, byte ptr [r11 + rdx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 5f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
5:
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov esi, ecx
    shr esi, 4
    and esi, 15
    or edx, esi
    movzx r13d, byte ptr [r11 + rdx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 6f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
6:
    mov edx, ecx
    and edx, 15
    shl edx, 2
    movzx r13d, byte ptr [r11 + rdx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 7f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
7:
    mov r13b, 61
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 8f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
8:
    jmp finish_encode

final_one:
    movzx eax, byte ptr [rip + input_buf + rbx]

    mov edx, eax
    shr edx, 2
    and edx, 63
    movzx r13d, byte ptr [r11 + rdx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 9f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
9:
    mov edx, eax
    and edx, 3
    shl edx, 4
    movzx r13d, byte ptr [r11 + rdx]
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 10f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
10:
    mov r13b, 61
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 11f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
11:
    mov r13b, 61
    mov byte ptr [r10 + r15], r13b
    inc r15
    inc r14
    cmp r14, LINE_WIDTH
    jne 12f
    mov byte ptr [r10 + r15], 10
    inc r15
    xor r14, r14
12:
    jmp finish_encode

finish_encode:
    test r14, r14
    jz write_output
    mov r13b, 10
    mov byte ptr [r10 + r15], r13b
    inc r15

write_output:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r15
    syscall
    jmp exit_zero

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
.align 16
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buf:
    .space 49152
.align 16
out_buf:
    .space 70000
