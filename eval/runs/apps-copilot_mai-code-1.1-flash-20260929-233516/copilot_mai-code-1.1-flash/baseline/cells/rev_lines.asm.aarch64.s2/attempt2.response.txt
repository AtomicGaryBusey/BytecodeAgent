.globl _start

.section .bss
.align 4
input_buf:
    .space 65536

.section .rodata
.align 4
newline_byte:
    .byte 0x0A

.text

_start:
    bl read_all
    mov x19, x0
    cbz x19, exit_ok
    mov x20, #0

scan_loop:
    cmp x20, x19
    b.hs final_line_check

    mov x21, x20
    mov x22, x20

scan_nl:
    cmp x22, x19
    b.hs final_line

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x22
    ldrb w24, [x23]
    cmp w24, #10
    b.eq found_nl
    add x22, x22, #1
    b scan_nl

found_nl:
    sub x24, x22, x21
    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    add x0, x0, x21
    mov x1, x24
    bl reverse_range

    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x21
    mov x2, x24
    bl write_all

    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    bl write_all

    add x20, x22, #1
    b scan_loop

final_line:
    sub x24, x19, x21
    cbz x24, exit_ok

    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    add x0, x0, x21
    mov x1, x24
    bl reverse_range

    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x21
    mov x2, x24
    bl write_all

    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    bl write_all
    b exit_ok

final_line_check:
    mov x21, x20
    sub x24, x19, x21
    cbz x24, exit_ok
    b final_line

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    mov x9, #0

read_loop:
    cmp x9, #65536
    b.hs read_done

    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x9
    mov x2, #65536
    sub x2, x2, x9
    mov x8, #63
    svc #0

    cmp x0, #0
    b.lt read_error
    b.eq read_done

    add x9, x9, x0
    b read_loop

read_done:
    mov x0, x9
    ret

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

write_all:
    mov x9, x0
    mov x10, x1
    mov x11, x2

write_loop:
    cbz x11, write_done

    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0

    cmp x0, #0
    b.le write_error

    sub x11, x11, x0
    add x10, x10, x0
    b write_loop

write_done:
    ret

write_error:
    mov x0, #1
    mov x8, #93
    svc #0

reverse_range:
    cmp x1, #2
    b.lt reverse_done

    sub x2, x1, #1
    add x2, x0, x2
    mov x3, x0

reverse_loop:
    cmp x3, x2
    b.hs reverse_done

    ldrb w4, [x3]
    ldrb w5, [x2]
    strb w5, [x3]
    strb w4, [x2]

    add x3, x3, #1
    sub x2, x2, #1
    b reverse_loop

reverse_done:
    ret
