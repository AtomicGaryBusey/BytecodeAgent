.globl _start

.section .bss
.align 4
input_buf:
    .space 65536

.text
_start:
    bl read_all
    mov x19, x0
    cbz x19, exit_ok
    mov x20, #0

line_loop:
    cmp x20, x19
    b.ge exit_ok

    mov x21, x20
    mov x22, x20

scan_nl:
    cmp x22, x19
    b.ge final_line
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x22
    ldrb w24, [x23]
    cmp w24, #10
    b.eq found_nl
    add x22, x22, #1
    b scan_nl

found_nl:
    mov x24, x22
    sub x24, x24, x21

    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    add x0, x0, x21
    mov x1, x24
    bl reverse_range

    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x21
    mov x2, x24
    bl write_all

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all

    add x20, x22, #1
    b line_loop

final_line:
    mov x24, x19
    sub x24, x24, x21

    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    add x0, x0, x21
    mov x1, x24
    bl reverse_range

    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x21
    mov x2, x24
    bl write_all

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    mov x19, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    b.eq read_done
    add x19, x19, x0
    cmp x19, #65536
    b.ge read_done
    b read_loop

read_done:
    mov x0, x19
    ret

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

write_all:
    mov x19, x0
    mov x20, x1
    mov x21, x2

write_loop:
    cbz x21, write_done
    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_error
    sub x21, x21, x0
    add x20, x20, x0
    b write_loop

write_done:
    mov x0, #0
    ret

write_error:
    mov x0, #1
    mov x8, #93
    svc #0

reverse_range:
    cmp x1, #1
    b.ls reverse_done
    sub x2, x1, #1
    mov x3, x0
    add x4, x0, x2

reverse_loop:
    cmp x3, x4
    b.hs reverse_done
    ldrb w5, [x3]
    ldrb w6, [x4]
    strb w6, [x3]
    strb w5, [x4]
    add x3, x3, #1
    sub x4, x4, #1
    b reverse_loop

reverse_done:
    ret

.section .rodata
.align 4
newline:
    .byte 0x0A
