.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf
    mov x21, #0

read_loop:
    cmp x21, #65536
    b.ge read_done
    mov x8, #63
    mov x0, #0
    mov x1, x19
    add x1, x1, x21
    mov x2, #65536
    sub x2, x2, x21
    svc #0
    cmp x0, #0
    b.lt read_fail
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_fail:
    mov x8, #93
    mov x0, #1
    svc #0

read_done:
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x23, x21
    b.ge final_line
    ldrb w24, [x19, x23]
    cmp w24, #10
    b.ne not_newline

    sub x10, x23, x22
    cmp x10, #0
    b.eq line_empty

    add x11, x22, x10
    sub x11, x11, #1
    mov x12, #0

reverse_loop:
    cmp x12, x10
    b.ge reverse_done
    ldrb w9, [x19, x11]
    strb w9, [x20, x12]
    sub x11, x11, #1
    add x12, x12, #1
    b reverse_loop

reverse_done:
    mov x0, #1
    mov x1, x20
    mov x2, x10

do_write_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_fail
    sub x2, x2, x0
    add x1, x1, x0
    cmp x2, #0
    b.ne do_write_loop
    b write_newline

line_empty:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    b write_newline

write_newline:
    mov x8, #64
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1

newline_write_loop:
    svc #0
    cmp x0, #0
    b.le write_fail
    sub x2, x2, x0
    add x1, x1, x0
    cmp x2, #0
    b.ne newline_write_loop

    add x22, x23, #1
    add x23, x23, #1
    b scan_loop

not_newline:
    add x23, x23, #1
    b scan_loop

final_line:
    cmp x22, x21
    b.ge exit_ok

    sub x10, x21, x22
    cmp x10, #0
    b.eq final_empty

    add x11, x22, x10
    sub x11, x11, #1
    mov x12, #0

final_reverse_loop:
    cmp x12, x10
    b.ge final_reverse_done
    ldrb w9, [x19, x11]
    strb w9, [x20, x12]
    sub x11, x11, #1
    add x12, x12, #1
    b final_reverse_loop

final_reverse_done:
    mov x0, #1
    mov x1, x20
    mov x2, x10

final_write_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_fail
    sub x2, x2, x0
    add x1, x1, x0
    cmp x2, #0
    b.ne final_write_loop
    b final_write_newline

final_empty:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    b final_write_newline

final_write_newline:
    mov x8, #64
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1

final_newline_loop:
    svc #0
    cmp x0, #0
    b.le write_fail
    sub x2, x2, x0
    add x1, x1, x0
    cmp x2, #0
    b.ne final_newline_loop

exit_ok:
    mov x8, #93
    mov x0, #0
    svc #0

write_fail:
    mov x8, #93
    mov x0, #1
    svc #0
