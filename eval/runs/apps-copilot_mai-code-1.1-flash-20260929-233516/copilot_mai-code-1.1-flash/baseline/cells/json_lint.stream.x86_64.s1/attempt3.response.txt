.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + pos], 0
    call parse_document
    test rax, rax
    jnz .ok
    call print_error
    mov edi, 1
    jmp .exit
.ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    xor edi, edi
.exit:
    mov rax, 60
    syscall

read_all:
    mov qword ptr [rip + input_len], 0
.read_loop:
    lea rsi, [rip + input_buf]
    mov r8, qword ptr [rip + input_len]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    mov rax, 0
    mov rdi, 0
    syscall
    cmp rax, 0
    je .done
    mov r8, qword ptr [rip + input_len]
    add r8, rax
    mov qword ptr [rip + input_len], r8
    jmp .read_loop
.done:
    ret

parse_document:
    call parse_ws
    call parse_value
    test rax, rax
    jz .ret
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne .fail
    mov rax, 1
    ret
.fail:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
.ret:
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
.ws_loop:
    cmp rax, qword ptr [rip + input_len]
    jae .ws_done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je .ws_skip
    cmp cl, 0x09
    je .ws_skip
    cmp cl, 0x0A
    je .ws_skip
    cmp cl, 0x0D
    je .ws_skip
    mov qword ptr [rip + pos], rax
    ret
.ws_skip:
    inc rax
    jmp .ws_loop
.ws_done:
    mov qword ptr [rip + pos], rax
    ret

parse_value:
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7B
    je .object
    cmp cl, 0x5B
    je .array
    cmp cl, 0x22
    je .string
    cmp cl, 0x2D
    je .number
    cmp cl, '0'
    jb .bad
    cmp cl, '9'
    jbe .number
    cmp cl, 't'
    je .true
    cmp cl, 'f'
    je .false
    cmp cl, 'n'
    je .null
.bad:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.object:
    call parse_object
    ret
.array:
    call parse_array
    ret
.string:
    call parse_string
    ret
.number:
    call parse_number
    ret
.true:
    call parse_true
    ret
.false:
    call parse_false
    ret
.null:
    call parse_null
    ret
.eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7B
    jne .fail_here
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7D
    je .close
.obj_loop:
    call parse_string
    test rax, rax
    jz .ret
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x3A
    jne .fail_here
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    test rax, rax
    jz .ret
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je .comma
    cmp cl, 0x7D
    je .close
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7D
    je .fail_here
    jmp .obj_loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_here:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
.ret:
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5B
    jne .fail_here
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5D
    je .close
.arr_loop:
    call parse_value
    test rax, rax
    jz .ret
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je .comma
    cmp cl, 0x5D
    je .close
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5D
    je .fail_here
    jmp .arr_loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_here:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
.ret:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    jne .fail_here
    inc qword ptr [rip + pos]
.str_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je .done
    cmp cl, 0x5C
    je .escape
    cmp cl, 0x00
    jb .fail_here
    cmp cl, 0x1F
    jbe .fail_here
    inc qword ptr [rip + pos]
    jmp .str_loop
.escape:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je .advance
    cmp cl, 0x5C
    je .advance
    cmp cl, 0x2F
    je .advance
    cmp cl, 'b'
    je .advance
    cmp cl, 'f'
    je .advance
    cmp cl, 'n'
    je .advance
    cmp cl, 'r'
    je .advance
    cmp cl, 't'
    je .advance
    cmp cl, 'u'
    jne .fail_here
    inc qword ptr [rip + pos]
    call parse_hex4
    test rax, rax
    jz .ret
    jmp .str_loop
.advance:
    inc qword ptr [rip + pos]
    jmp .str_loop
.done:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_here:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
.ret:
    ret

parse_hex4:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + input_len]
    cmp rax, rdx
    jae .fail_eof
    lea rsi, [rip + input_buf]
    mov rcx, 4
.hex_loop:
    cmp rax, rdx
    jae .fail_eof
    movzx r8d, byte ptr [rsi + rax]
    cmp r8b, '0'
    jb .fail_here
    cmp r8b, '9'
    jbe .hex_next
    cmp r8b, 'a'
    jb .fail_here
    cmp r8b, 'f'
    jbe .hex_next
    cmp r8b, 'A'
    jb .fail_here
    cmp r8b, 'F'
    jbe .hex_next
    jmp .fail_here
.hex_next:
    inc rax
    dec rcx
    jnz .hex_loop
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_here:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rdx
    xor rax, rax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2D
    jne .int_start
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_at_current
    movzx ecx, byte ptr [rsi + rax]
.int_start:
    cmp cl, '0'
    je .zero_case
    cmp cl, '1'
    jb .fail_at_current
    cmp cl, '9'
    jbe .nonzero_case
    jmp .fail_at_current
.zero_case:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .after_zero_check
    cmp cl, '9'
    jbe .fail_at_current
.after_zero_check:
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .success
.nonzero_case:
    call advance_digits
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .success
.frac:
    mov r8, rax
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_at_current
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .fail_at_current
    cmp cl, '9'
    jbe .frac_digits
    jmp .fail_at_current
.frac_digits:
    call advance_digits
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .success
.exp:
    mov r8, rax
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_at_r8
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '+'
    je .exp_sign
    cmp cl, '-'
    je .exp_sign
    cmp cl, '0'
    jb .fail_at_current
    cmp cl, '9'
    jbe .exp_digits
    jmp .fail_at_current
.exp_sign:
    mov r9, rax
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_at_r9
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .fail_at_current
    cmp cl, '9'
    jbe .exp_digits
    jmp .fail_at_current
.exp_digits:
    call advance_digits
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.success:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.fail_at_current:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_at_r8:
    mov qword ptr [rip + error_offset], r8
    xor rax, rax
    ret
.fail_at_r9:
    mov qword ptr [rip + error_offset], r9
    xor rax, rax
    ret

advance_digits:
    mov r8, qword ptr [rip + input_len]
.ad_loop:
    cmp rax, r8
    jae .ad_done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .ad_done
    cmp cl, '9'
    ja .ad_done
    inc rax
    jmp .ad_loop
.ad_done:
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + true_lit]
    mov rcx, 4
.true_loop:
    test rcx, rcx
    jz .true_ok
    lea rdi, [rip + input_buf]
    movzx r8d, byte ptr [rdi + rax]
    movzx r9d, byte ptr [rsi]
    cmp r8b, r9b
    jne .true_fail
    inc rax
    inc rsi
    dec rcx
    jmp .true_loop
.true_ok:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.true_fail:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_false:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + false_lit]
    mov rcx, 5
.false_loop:
    test rcx, rcx
    jz .false_ok
    lea rdi, [rip + input_buf]
    movzx r8d, byte ptr [rdi + rax]
    movzx r9d, byte ptr [rsi]
    cmp r8b, r9b
    jne .false_fail
    inc rax
    inc rsi
    dec rcx
    jmp .false_loop
.false_ok:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.false_fail:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_null:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + null_lit]
    mov rcx, 4
.null_loop:
    test rcx, rcx
    jz .null_ok
    lea rdi, [rip + input_buf]
    movzx r8d, byte ptr [rdi + rax]
    movzx r9d, byte ptr [rsi]
    cmp r8b, r9b
    jne .null_fail
    inc rax
    inc rsi
    dec rcx
    jmp .null_loop
.null_ok:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.null_fail:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    syscall
    mov rax, qword ptr [rip + error_offset]
    lea rdi, [rip + out_buf + 31]
    mov byte ptr [rdi], 10
    mov rbx, 10
.dec_loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .dec_loop
    mov rsi, rdi
    lea rdx, [rip + out_buf + 32]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    ret

.section .rodata
ok_msg:
    .asciz "ok\n"
err_prefix:
    .ascii "error at "
true_lit:
    .ascii "true"
false_lit:
    .ascii "false"
null_lit:
    .ascii "null"

.section .bss
.align 8
input_buf:
    .space 65536
input_len:
    .quad 0
pos:
    .quad 0
error_offset:
    .quad 0
out_buf:
    .space 32
