.text
.globl _start

_start:
  call read_all
  call parse_document
  test eax, eax
  jnz .ok
  call print_error
  mov edi, 1
  jmp .exit
.ok:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + ok_msg]
  mov rdx, 3
  syscall
  xor edi, edi
.exit:
  mov rax, 60
  syscall

read_all:
  mov qword ptr [rip + input_len], 0
.read_loop:
  mov rax, qword ptr [rip + input_len]
  cmp rax, 65536
  jae .done
  mov rax, 0
  mov rdi, 0
  lea r8, [rip + input_buf]
  lea rsi, [r8 + rax]
  mov rax, qword ptr [rip + input_len]
  lea rsi, [r8 + rax]
  mov rdx, 65536
  sub rdx, rax
  syscall
  cmp rax, 0
  jle .done
  mov rbx, qword ptr [rip + input_len]
  add rbx, rax
  mov qword ptr [rip + input_len], rbx
  jmp .read_loop
.done:
  ret

parse_document:
  call parse_ws
  call parse_value
  test eax, eax
  jz .ret
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jne .fail_here
  mov eax, 1
  ret
.fail_here:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
.ret:
  ret

parse_ws:
  mov rax, qword ptr [rip + pos]
.loop:
  cmp rax, qword ptr [rip + input_len]
  jae .done
  lea r8, [rip + input_buf]
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x20
  je .skip
  cmp cl, 0x09
  je .skip
  cmp cl, 0x0A
  je .skip
  cmp cl, 0x0D
  je .skip
  mov qword ptr [rip + pos], rax
  ret
.skip:
  inc rax
  jmp .loop
.done:
  mov qword ptr [rip + pos], rax
  ret

parse_value:
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  lea r8, [rip + input_buf]
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x7B
  je parse_object
  cmp cl, 0x5B
  je parse_array
  cmp cl, 0x22
  je parse_string
  cmp cl, 0x2D
  je parse_number
  cmp cl, 0x30
  jb .bad
  cmp cl, 0x39
  jbe parse_number
  cmp cl, 0x74
  je parse_true
  cmp cl, 0x66
  je parse_false
  cmp cl, 0x6E
  je parse_null
.bad:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

parse_object:
  mov rax, qword ptr [rip + pos]
  inc rax
  mov qword ptr [rip + pos], rax
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  lea r8, [rip + input_buf]
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x7D
  je .done
.loop:
  call parse_string
  test eax, eax
  jz .ret
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x3A
  jne .bad_colon
  inc rax
  mov qword ptr [rip + pos], rax
  call parse_ws
  call parse_value
  test eax, eax
  jz .ret
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x2C
  je .next
  cmp cl, 0x7D
  jne .bad_end
  inc rax
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.next:
  inc rax
  mov qword ptr [rip + pos], rax
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x7D
  je .bad_trailing
  jmp .loop
.done:
  inc rax
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.bad_colon:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.bad_end:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.bad_trailing:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
.ret:
  ret

parse_array:
  mov rax, qword ptr [rip + pos]
  inc rax
  mov qword ptr [rip + pos], rax
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  lea r8, [rip + input_buf]
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x5D
  je .done
.loop:
  call parse_value
  test eax, eax
  jz .ret
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x2C
  je .next
  cmp cl, 0x5D
  jne .bad_end
  inc rax
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.next:
  inc rax
  mov qword ptr [rip + pos], rax
  call parse_ws
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x5D
  je .bad_trailing
  jmp .loop
.done:
  inc rax
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.bad_end:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.bad_trailing:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
.ret:
  ret

parse_string:
  mov rax, qword ptr [rip + pos]
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  lea r8, [rip + input_buf]
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x22
  jne .fail
  inc rax
.loop:
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x22
  je .done
  cmp cl, 0x5C
  je .escape
  cmp cl, 0x00
  jb .bad
  inc rax
  jmp .loop
.escape:
  inc rax
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x22
  je .next
  cmp cl, 0x5C
  je .next
  cmp cl, 0x2F
  je .next
  cmp cl, 0x62
  je .next
  cmp cl, 0x66
  je .next
  cmp cl, 0x6E
  je .next
  cmp cl, 0x72
  je .next
  cmp cl, 0x74
  je .next
  cmp cl, 0x75
  je .unicode
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.next:
  inc rax
  jmp .loop
.unicode:
  inc rax
  mov rbx, 4
.hex_loop:
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx ecx, byte ptr [r8 + rax]
  cmp cl, 0x30
  jb .hex_bad
  cmp cl, 0x39
  jbe .hex_ok
  cmp cl, 0x61
  jb .hex_bad
  cmp cl, 0x66
  jbe .hex_ok
  cmp cl, 0x41
  jb .hex_bad
  cmp cl, 0x46
  jbe .hex_ok
.hex_bad:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.hex_ok:
  inc rax
  dec rbx
  jnz .hex_loop
  jmp .loop
.done:
  inc rax
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.fail:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.bad:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

parse_true:
  mov rax, qword ptr [rip + pos]
  lea r8, [rip + input_buf]
  lea r9, [rip + true_str]
  mov rcx, 4
.loop:
  cmp rcx, 0
  je .done
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx edx, byte ptr [r9]
  movzx esi, byte ptr [r8 + rax]
  cmp sil, dl
  jne .mismatch
  inc rax
  inc r9
  dec rcx
  jmp .loop
.done:
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.mismatch:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

parse_false:
  mov rax, qword ptr [rip + pos]
  lea r8, [rip + input_buf]
  lea r9, [rip + false_str]
  mov rcx, 5
.loop:
  cmp rcx, 0
  je .done
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx edx, byte ptr [r9]
  movzx esi, byte ptr [r8 + rax]
  cmp sil, dl
  jne .mismatch
  inc rax
  inc r9
  dec rcx
  jmp .loop
.done:
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.mismatch:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

parse_null:
  mov rax, qword ptr [rip + pos]
  lea r8, [rip + input_buf]
  lea r9, [rip + null_str]
  mov rcx, 4
.loop:
  cmp rcx, 0
  je .done
  cmp rax, qword ptr [rip + input_len]
  jae .eof
  movzx edx, byte ptr [r9]
  movzx esi, byte ptr [r8 + rax]
  cmp sil, dl
  jne .mismatch
  inc rax
  inc r9
  dec rcx
  jmp .loop
.done:
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.mismatch:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

parse_number:
  mov rax, qword ptr [rip + pos]
  mov r8, qword ptr [rip + input_len]
  lea r9, [rip + input_buf]
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x2D
  jne .after_sign
  inc rax
  cmp rax, r8
  jae .eof
.after_sign:
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  je .zero
  cmp cl, 0x31
  jb .fail
  cmp cl, 0x39
  ja .fail
  inc rax
.int_loop:
  cmp rax, r8
  jae .done
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  jb .after_int
  cmp cl, 0x39
  jbe .int_ok
.after_int:
  cmp cl, 0x2E
  je .frac
  cmp cl, 0x65
  je .exp
  cmp cl, 0x45
  je .exp
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.int_ok:
  inc rax
  jmp .int_loop
.zero:
  inc rax
  cmp rax, r8
  jae .done
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  jb .after_zero
  cmp cl, 0x39
  jbe .fail
.after_zero:
  cmp cl, 0x2E
  je .frac
  cmp cl, 0x65
  je .exp
  cmp cl, 0x45
  je .exp
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.frac:
  inc rax
  cmp rax, r8
  jae .frac_eof
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  jb .frac_fail
  cmp cl, 0x39
  ja .frac_fail
.frac_loop:
  inc rax
  cmp rax, r8
  jae .done
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  jb .after_frac
  cmp cl, 0x39
  jbe .frac_ok
.after_frac:
  cmp cl, 0x65
  je .exp
  cmp cl, 0x45
  je .exp
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.frac_ok:
  inc rax
  jmp .frac_loop
.exp:
  inc rax
  cmp rax, r8
  jae .exp_eof
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x2B
  je .exp_sign
  cmp cl, 0x2D
  je .exp_sign
  jmp .exp_digit
.exp_sign:
  inc rax
  cmp rax, r8
  jae .exp_eof
  movzx ecx, byte ptr [r9 + rax]
.exp_digit:
  cmp cl, 0x30
  jb .exp_fail
  cmp cl, 0x39
  ja .exp_fail
.exp_digit_loop:
  inc rax
  cmp rax, r8
  jae .done
  movzx ecx, byte ptr [r9 + rax]
  cmp cl, 0x30
  jb .done
  cmp cl, 0x39
  jbe .exp_digit_ok
.done:
  mov qword ptr [rip + pos], rax
  mov eax, 1
  ret
.exp_digit_ok:
  inc rax
  jmp .exp_digit_loop
.fail:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.frac_fail:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.frac_eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.exp_fail:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.exp_eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret
.eof:
  mov qword ptr [rip + error_offset], rax
  xor eax, eax
  ret

print_error:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + err_prefix]
  mov rdx, 9
  syscall
  mov rax, qword ptr [rip + error_offset]
  lea rsi, [rip + decbuf + 31]
  mov byte ptr [rsi], 0x0A
  mov rbx, 10
.dec_loop:
  xor rdx, rdx
  div rbx
  add dl, 0x30
  dec rsi
  mov byte ptr [rsi], dl
  test rax, rax
  jnz .dec_loop
  mov rax, 1
  mov rdi, 1
  lea rdx, [rip + decbuf + 32]
  sub rdx, rsi
  syscall
  ret

.section .rodata
err_prefix:
  .ascii "error at "
ok_msg:
  .ascii "ok\n"
true_str:
  .ascii "true"
false_str:
  .ascii "false"
null_str:
  .ascii "null"

.section .bss
.align 8
input_buf:
  .space 65536
.align 8
input_len:
  .quad 0
pos:
  .quad 0
error_offset:
  .quad 0
decbuf:
  .space 32
