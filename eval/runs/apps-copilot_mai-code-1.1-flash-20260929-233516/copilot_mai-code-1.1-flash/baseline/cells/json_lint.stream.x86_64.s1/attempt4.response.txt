.text
.globl _start

_start:
    mov qword ptr [rip + pos], 0
    call read_all
    mov qword ptr [rip + pos], 0
    call parse_document
    test rax, rax
    jnz .ok
    call print_error
    mov edi, 1
    mov eax, 60
    syscall
.ok:
    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov eax, 1
    mov edi, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

read_all:
    mov qword ptr [rip + input_len], 0
    mov rbx, 0
.read_loop:
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    mov eax, 0
    mov edi, 0
    syscall
    cmp rax, 0
    je .done
    add rbx, rax
    mov qword ptr [rip + input_len], rbx
    cmp rbx, 65536
    je .done
    jmp .read_loop
.done:
    ret

parse_document:
    mov qword ptr [rip + pos], 0
    call parse_ws
    call parse_value
    test rax, rax
    jz .fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne .fail_extra
    mov rax, 1
    ret
.fail_extra:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail:
    xor rax, rax
    ret

parse_ws:
    mov rax, qword ptr [rip + pos]
.ws_loop:
    cmp rax, qword ptr [rip + input_len]
    jae .done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je .skip
    cmp cl, 0x09
    je .skip
    cmp cl, 0x0A
    je .skip
    cmp cl, 0x0D
    je .skip
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret
.skip:
    inc rax
    jmp .ws_loop
.done:
    mov qword ptr [rip + pos], rax
    mov rax, 1
    ret

parse_value:
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7B
    je .object
    cmp cl, 0x5B
    je .array
    cmp cl, 0x22
    je .string
    cmp cl, 0x2D
    je .number
    cmp cl, '0'
    jb .fail_char
    cmp cl, '9'
    jbe .number
    cmp cl, 't'
    je .true
    cmp cl, 'f'
    je .false
    cmp cl, 'n'
    je .null
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.object:
    call parse_object
    ret
.array:
    call parse_array
    ret
.string:
    call parse_string
    ret
.number:
    call parse_number
    ret
.true:
    call parse_true
    ret
.false:
    call parse_false
    ret
.null:
    call parse_null
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_object:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7B
    jne .fail_char
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7D
    je .close
.loop:
    call parse_string
    test rax, rax
    jz .done_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x3A
    jne .fail_char
    inc qword ptr [rip + pos]
    call parse_ws
    call parse_value
    test rax, rax
    jz .done_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je .comma
    cmp cl, 0x7D
    je .close
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7D
    je .fail_char
    jmp .loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.done_fail:
    ret

parse_array:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5B
    jne .fail_char
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5D
    je .close
.loop:
    call parse_value
    test rax, rax
    jz .done_fail
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je .comma
    cmp cl, 0x5D
    je .close
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.comma:
    inc qword ptr [rip + pos]
    call parse_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5D
    je .fail_char
    jmp .loop
.close:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.done_fail:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    jne .fail_char
    inc qword ptr [rip + pos]
.loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je .done
    cmp cl, 0x5C
    je .escape
    cmp cl, 0x00
    jb .fail_char
    cmp cl, 0x1F
    jbe .fail_char
    inc qword ptr [rip + pos]
    jmp .loop
.escape:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je .escaped
    cmp cl, 0x5C
    je .escaped
    cmp cl, 0x2F
    je .escaped
    cmp cl, 'b'
    je .escaped
    cmp cl, 'f'
    je .escaped
    cmp cl, 'n'
    je .escaped
    cmp cl, 'r'
    je .escaped
    cmp cl, 't'
    je .escaped
    cmp cl, 'u'
    je .unicode
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.escaped:
    inc qword ptr [rip + pos]
    jmp .loop
.unicode:
    inc qword ptr [rip + pos]
    mov r9, 0
.check_digit:
    cmp r9, 4
    je .unicode_ok
    mov rax, qword ptr [rip + pos]
    add rax, r9
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .bad_hex
    cmp cl, '9'
    jbe .digit_ok
    cmp cl, 'a'
    jb .bad_hex
    cmp cl, 'f'
    jbe .digit_ok
    cmp cl, 'A'
    jb .bad_hex
    cmp cl, 'F'
    jbe .digit_ok
.bad_hex:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.digit_ok:
    inc r9
    jmp .check_digit
.unicode_ok:
    add qword ptr [rip + pos], 4
    jmp .loop
.done:
    inc qword ptr [rip + pos]
    mov rax, 1
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + input_buf]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2D
    jne .start_int
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
.start_int:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    je .zero
    cmp cl, '1'
    jb .fail_char
    cmp cl, '9'
    jbe .digits
    jmp .fail_char
.zero:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jb .check_after_zero
    jmp .frac_or_exp_or_done
.check_after_zero:
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .frac_or_exp_or_done
    cmp cl, '9'
    jbe .fail_char
.frac_or_exp_or_done:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .done
.digits:
    inc qword ptr [rip + pos]
.loop_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_digits
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .after_digits
    cmp cl, '9'
    jbe .continue_digits
.after_digits:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '.'
    je .frac
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .done
.continue_digits:
    inc qword ptr [rip + pos]
    jmp .loop_digits
.frac:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .fail_char
    cmp cl, '9'
    jbe .frac_digits
    jmp .fail_char
.frac_digits:
    inc qword ptr [rip + pos]
.loop_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .after_frac
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .after_frac
    cmp cl, '9'
    jbe .continue_frac
.after_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'e'
    je .exp
    cmp cl, 'E'
    je .exp
    jmp .done
.continue_frac:
    inc qword ptr [rip + pos]
    jmp .loop_frac
.exp:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '+'
    je .exp_sign
    cmp cl, '-'
    je .exp_sign
    jmp .exp_digits_check
.exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
.exp_digits_check:
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .fail_char
    cmp cl, '9'
    jbe .exp_digits
    jmp .fail_char
.exp_digits:
    inc qword ptr [rip + pos]
.loop_exp:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .done
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '0'
    jb .done
    cmp cl, '9'
    jbe .continue_exp
    jmp .done
.continue_exp:
    inc qword ptr [rip + pos]
    jmp .loop_exp
.done:
    mov rax, 1
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_true:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 't'
    jne .fail_char
    mov rax, qword ptr [rip + pos]
    add rax, 1
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'r'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'u'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'e'
    jne .fail_char
    add qword ptr [rip + pos], 4
    mov rax, 1
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_false:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'f'
    jne .fail_char
    mov rax, qword ptr [rip + pos]
    add rax, 1
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'a'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'l'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 's'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'e'
    jne .fail_char
    add qword ptr [rip + pos], 5
    mov rax, 1
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

parse_null:
    mov rax, qword ptr [rip + pos]
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'n'
    jne .fail_char
    mov rax, qword ptr [rip + pos]
    add rax, 1
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'u'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'l'
    jne .fail_char
    inc rax
    cmp rax, qword ptr [rip + input_len]
    jae .fail_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 'l'
    jne .fail_char
    add qword ptr [rip + pos], 4
    mov rax, 1
    ret
.fail_eof:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret
.fail_char:
    mov qword ptr [rip + error_offset], rax
    xor rax, rax
    ret

print_error:
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    mov eax, 1
    mov edi, 1
    syscall
    mov rax, qword ptr [rip + error_offset]
    lea rsi, [rip + numbuf]
    add rsi, 31
    mov rcx, 0
    cmp rax, 0
    jne .digits
    mov byte ptr [rsi], '0'
    mov rcx, 1
    jmp .write
.digits:
    mov rdx, 0
    mov rbx, 10
    div rbx
    add dl, '0'
    mov byte ptr [rsi], dl
    dec rsi
    inc rcx
    cmp rax, 0
    jne .digits
    inc rsi
.write:
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov eax, 1
    mov edi, 1
    syscall
    ret

ok_msg:
    .ascii "ok\n"

    .ascii "error at "

    .byte 0x0A

    .zero 65536

    .zero 8
