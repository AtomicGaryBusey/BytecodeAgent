.section .text
.globl _start

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x21, hex_digits
    add x21, x21, :lo12:hex_digits
    b read_loop

read_loop:
    mov x0, #65536
    sub x0, x0, x19
    cbz x0, read_done
    mov x1, x20
    add x1, x1, x19
    mov x2, x0
    mov x8, #63
    svc #0
    cmp x0, #0
    blt exit_error
    beq read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, print_total
    mov x22, #0

row_loop:
    mov x23, x19
    sub x23, x23, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, lt

    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, x27

    mov x0, x22
    mov x1, x28
    bl emit_offset_hex
    mov x28, x0

    mov x26, #0

hex_loop:
    cmp x26, #16
    b.hs hex_done
    cmp x26, x23
    b.hs hex_missing

hex_present:
    add x0, x20, x22
    add x0, x0, x26
    ldrb w0, [x0]

    lsr w1, w0, #4
    and w1, w1, #0xf
    add x2, x21, x1
    ldrb w1, [x2]
    strb w1, [x28], #1

    and w1, w0, #0xf
    add x2, x21, x1
    ldrb w1, [x2]
    strb w1, [x28], #1

    mov w1, #0x20
    strb w1, [x28], #1

    cmp x26, #7
    b.ne hex_continue
    mov w1, #0x20
    strb w1, [x28], #1
    b hex_continue

hex_missing:
    mov w1, #0x20
    strb w1, [x28], #1
    strb w1, [x28], #1
    strb w1, [x28], #1

    cmp x26, #7
    b.ne hex_continue
    mov w1, #0x20
    strb w1, [x28], #1

hex_continue:
    add x26, x26, #1
    b hex_loop

hex_done:
    mov w1, #0x20
    strb w1, [x28], #1
    mov w1, #'|'
    strb w1, [x28], #1

    mov x26, #0

char_loop:
    cmp x26, x23
    b.ge char_done

    add x0, x20, x22
    add x0, x0, x26
    ldrb w0, [x0]

    cmp w0, #0x20
    b.lt char_dot
    cmp w0, #0x7e
    b.gt char_dot
    strb w0, [x28], #1
    b char_next

char_dot:
    mov w0, #46
    strb w0, [x28], #1

char_next:
    add x26, x26, #1
    b char_loop

char_done:
    mov w1, #'|'
    strb w1, [x28], #1
    mov w1, #10
    strb w1, [x28], #1

    mov x0, #1
    mov x1, x27
    sub x2, x28, x27
    bl write_all

    add x22, x22, x23
    cmp x22, x19
    b.lt row_loop

print_total:
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, x27
    mov x2, x19
    mov x3, #28

total_hex_loop:
    lsr x4, x2, x3
    and x4, x4, #0xf
    add x4, x4, x21
    ldrb w4, [x4]
    strb w4, [x28], #1
    subs x3, x3, #4
    b.ge total_hex_loop

    mov w4, #10
    strb w4, [x28], #1

    mov x0, #1
    mov x1, x27
    sub x2, x28, x27
    bl write_all
    b exit_clean

emit_offset_hex:
    mov x2, x0
    mov x3, #28

offset_hex_loop:
    lsr x4, x2, x3
    and x4, x4, #0xf
    add x4, x4, x21
    ldrb w4, [x4]
    strb w4, [x1], #1
    subs x3, x3, #4
    b.ge offset_hex_loop

    mov w4, #0x20
    strb w4, [x1], #1
    strb w4, [x1], #1
    mov x0, x1
    ret

write_all:
    mov x8, #64

write_all_loop:
    svc #0
    cmp x0, #0
    ble exit_error
    sub x2, x2, x0
    cbz x2, write_all_done
    add x1, x1, x0
    b write_all_loop

write_all_done:
    ret

exit_clean:
    mov x0, #0
    mov x8, #93
    svc #0

exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 256
