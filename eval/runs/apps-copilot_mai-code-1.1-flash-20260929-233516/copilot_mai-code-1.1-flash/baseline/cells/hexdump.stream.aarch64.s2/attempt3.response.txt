.section .text
.globl _start

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x21, hex_digits
    add x21, x21, :lo12:hex_digits
    b read_all

read_all:
    mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt exit_error
    beq read_done
    add x19, x19, x0
    b read_all

read_done:
    cbz x19, print_total
    mov x22, #0

row_loop:
    sub x23, x19, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, ls

    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, x27

    mov x0, x27
    mov x1, x22
    bl emit_offset_hex
    mov x28, x0

    mov x26, #0

hex_loop:
    cmp x26, #16
    b.ge hex_done
    cmp x26, x23
    b.ge hex_missing

hex_present:
    add x5, x20, x22
    add x5, x5, x26
    ldrb w5, [x5]

    lsr w6, w5, #4
    and w6, w6, #0xf
    add x6, x21, x6
    ldrb w7, [x6]
    strb w7, [x28], #1

    and w6, w5, #0xf
    add x6, x21, x6
    ldrb w7, [x6]
    strb w7, [x28], #1

    mov w7, #0x20
    strb w7, [x28], #1
    b hex_after

hex_missing:
    mov w7, #0x20
    strb w7, [x28], #1
    strb w7, [x28], #1
    strb w7, [x28], #1

hex_after:
    cmp x26, #7
    b.ne hex_continue
    mov w7, #0x20
    strb w7, [x28], #1

hex_continue:
    add x26, x26, #1
    b hex_loop

hex_done:
    mov w7, #0x20
    strb w7, [x28], #1
    mov w7, #'|'
    strb w7, [x28], #1

    mov x26, #0

char_loop:
    cmp x26, x23
    b.ge char_done

    add x5, x20, x22
    add x5, x5, x26
    ldrb w5, [x5]

    cmp w5, #0x20
    b.lt char_dot
    cmp w5, #0x7e
    b.gt char_dot
    b char_store

char_dot:
    mov w5, #0x2e

char_store:
    strb w5, [x28], #1
    add x26, x26, #1
    b char_loop

char_done:
    mov w7, #'|'
    strb w7, [x28], #1
    mov w7, #0x0a
    strb w7, [x28], #1

    mov x0, #1
    mov x1, x27
    sub x2, x28, x27
    bl write_all

    add x22, x22, x23
    cmp x22, x19
    b.lt row_loop

print_total:
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, x27

    mov x0, x27
    mov x1, x19
    bl emit_total_hex
    mov x28, x0

    mov x0, #1
    mov x1, x27
    sub x2, x28, x27
    bl write_all
    b exit_clean

emit_offset_hex:
    mov x2, x0
    mov x3, x1
    mov x4, #28

1:
    lsr x5, x3, x4
    and x5, x5, #0xf
    add x5, x5, x21
    ldrb w5, [x5]
    strb w5, [x2], #1
    subs x4, x4, #4
    b.ge 1b

    mov w5, #0x20
    strb w5, [x2], #1
    strb w5, [x2], #1

    mov x0, x2
    ret

emit_total_hex:
    mov x2, x0
    mov x3, x1
    mov x4, #28

1:
    lsr x5, x3, x4
    and x5, x5, #0xf
    add x5, x5, x21
    ldrb w5, [x5]
    strb w5, [x2], #1
    subs x4, x4, #4
    b.ge 1b

    mov w5, #0x0a
    strb w5, [x2], #1

    mov x0, x2
    ret

write_all:
    mov x8, #64

1:
    svc #0
    cmp x0, #0
    b.lt exit_error
    cbz x0, exit_error
    sub x2, x2, x0
    cbz x2, 2f
    add x1, x1, x0
    b 1b

2:
    ret

exit_clean:
    mov x0, #0
    mov x8, #93
    svc #0

exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 256
