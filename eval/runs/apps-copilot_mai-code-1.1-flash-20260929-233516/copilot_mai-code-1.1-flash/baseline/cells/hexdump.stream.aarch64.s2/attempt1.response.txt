.section .text
.globl _start
_start:
    mov x19, #0
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x22
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    cmp x2, #0
    ble read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    blt exit_error
    beq read_done
    add x19, x19, x0
    b read_loop

read_done:
    cmp x19, #0
    beq print_total_line

    mov x20, #0
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

row_loop:
    sub x23, x19, x20
    mov x27, #16
    cmp x23, x27
    csel x23, x23, x27, lt

    mov x24, x25

    mov x26, x20
    mov x27, #28

offset_hex:
    lsr x28, x26, x27
    and x28, x28, #0xf
    adrp x29, hex_digits
    add x29, x29, :lo12:hex_digits
    add x29, x29, x28
    ldrb w28, [x29]
    strb w28, [x24], #1
    subs x27, x27, #4
    b.ge offset_hex

    mov w28, #0x20
    strb w28, [x24], #1
    strb w28, [x24], #1

    mov x21, #0

hex_slots:
    cmp x21, #16
    b.ge hex_slots_done
    cmp x21, x23
    b.hs hex_missing

hex_present:
    add x26, x22, x20
    add x26, x26, x21
    ldrb w26, [x26]

    adrp x29, hex_digits
    add x29, x29, :lo12:hex_digits
    lsr w27, w26, #4
    and w27, w27, #0xf
    add x28, x29, x27
    ldrb w27, [x28]
    strb w27, [x24], #1

    and w27, w26, #0xf
    add x28, x29, x27
    ldrb w27, [x28]
    strb w27, [x24], #1

    mov w27, #0x20
    strb w27, [x24], #1
    b hex_slot_after

hex_missing:
    mov w27, #0x20
    strb w27, [x24], #1
    strb w27, [x24], #1
    strb w27, [x24], #1

hex_slot_after:
    cmp x21, #7
    b.ne hex_slot_next
    mov w27, #0x20
    strb w27, [x24], #1

hex_slot_next:
    add x21, x21, #1
    b hex_slots

hex_slots_done:
    mov w27, #0x20
    strb w27, [x24], #1
    mov w27, #'|'
    strb w27, [x24], #1

    mov x21, #0

char_dump:
    cmp x21, x23
    b.ge char_dump_done

    add x26, x22, x20
    add x26, x26, x21
    ldrb w26, [x26]

    cmp w26, #0x20
    b.lt char_dot
    cmp w26, #0x7e
    b.gt char_dot
    strb w26, [x24], #1
    b char_continue

char_dot:
    mov w26, #46
    strb w26, [x24], #1

char_continue:
    add x21, x21, #1
    b char_dump

char_dump_done:
    mov w27, #'|'
    strb w27, [x24], #1
    mov w27, #10
    strb w27, [x24], #1

    sub x2, x24, x25
    mov x0, #1
    mov x1, x25

write_output_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    ble exit_error
    sub x2, x2, x0
    cmp x2, #0
    beq row_done
    add x1, x1, x0
    b write_output_loop

row_done:
    add x20, x20, x23
    cmp x20, x19
    b.lt row_loop

print_total_line:
    mov x24, x25
    mov x26, x19
    mov x27, #28

total_hex:
    lsr x28, x26, x27
    and x28, x28, #0xf
    adrp x29, hex_digits
    add x29, x29, :lo12:hex_digits
    add x29, x29, x28
    ldrb w28, [x29]
    strb w28, [x24], #1
    subs x27, x27, #4
    b.ge total_hex

    mov w28, #10
    strb w28, [x24], #1

    sub x2, x24, x25
    mov x0, #1
    mov x1, x25

final_write_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    ble exit_error
    sub x2, x2, x0
    cmp x2, #0
    beq final_exit
    add x1, x1, x0
    b final_write_loop

final_exit:
    mov x0, #0
    mov x8, #93
    svc #0

exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 256
