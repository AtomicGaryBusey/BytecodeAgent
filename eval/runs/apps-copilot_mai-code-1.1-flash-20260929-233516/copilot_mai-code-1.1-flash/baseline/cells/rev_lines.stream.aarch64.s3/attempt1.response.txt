.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .rodata
newline:
    .byte 10

.bss
    .align 3
input_buf:
    .space 65536
tmp_buf:
    .space 65536

.text
_start:
    sub sp, sp, #4096

    mov x19, #0

read_loop:
    cmp x19, #65536
    b.hs read_done

    mov x8, #SYS_READ
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #0
    movk x2, #1, lsl #16
    sub x2, x2, x19
    svc #0

    cmp x0, #0
    b.lt read_error
    beq read_done

    add x19, x19, x0
    b read_loop

read_done:
    cmp x19, #0
    b.eq exit_ok

    mov x20, #0
    mov x21, #0

scan_loop:
    cmp x21, x19
    b.hs final_line

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x21
    ldrb w24, [x23]
    cmp w24, #10
    b.ne scan_next

    sub x22, x21, x20
    cbz x22, write_empty_line

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x20
    adrp x24, tmp_buf
    add x24, x24, :lo12:tmp_buf
    mov x25, #0

copy_loop:
    cmp x25, x22
    b.hs reverse_loop
    ldrb w26, [x23, x25]
    strb w26, [x24, x25]
    add x25, x25, #1
    b copy_loop

reverse_loop:
    mov x25, #0
    sub x26, x22, #1

rev_swap:
    cmp x25, x26
    b.hs write_segment
    add x27, x24, x25
    add x28, x24, x26
    ldrb w29, [x27]
    ldrb w30, [x28]
    strb w29, [x28]
    strb w30, [x27]
    add x25, x25, #1
    sub x26, x26, #1
    b rev_swap

write_segment:
    mov x9, #1
    mov x10, x24
    mov x11, x22

write_seg_loop:
    cbz x11, newline_out
    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_error
    subs x11, x11, x0
    add x10, x10, x0
    b write_seg_loop

newline_out:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_error

    add x20, x21, #1
    add x21, x21, #1
    b scan_loop

write_empty_line:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_error

    add x20, x21, #1
    add x21, x21, #1
    b scan_loop

scan_next:
    add x21, x21, #1
    b scan_loop

final_line:
    cmp x20, x19
    b.hs exit_ok

    sub x22, x19, x20
    cbz x22, exit_ok

    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x20
    adrp x24, tmp_buf
    add x24, x24, :lo12:tmp_buf
    mov x25, #0

copy_last:
    cmp x25, x22
    b.hs reverse_last
    ldrb w26, [x23, x25]
    strb w26, [x24, x25]
    add x25, x25, #1
    b copy_last

reverse_last:
    mov x25, #0
    sub x26, x22, #1

rev_last_swap:
    cmp x25, x26
    b.hs write_last
    add x27, x24, x25
    add x28, x24, x26
    ldrb w29, [x27]
    ldrb w30, [x28]
    strb w29, [x28]
    strb w30, [x27]
    add x25, x25, #1
    sub x26, x26, #1
    b rev_last_swap

write_last:
    mov x9, #1
    mov x10, x24
    mov x11, x22

write_last_loop:
    cbz x11, newline_last
    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_error
    subs x11, x11, x0
    add x10, x10, x0
    b write_last_loop

newline_last:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le write_error

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

read_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

write_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0
