.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + g_pos], 0
    call parse_document
    test rax, rax
    jnz .error
    lea rsi, [rip + msg_ok]
    mov rdx, 3
    mov rax, 1
    mov rdi, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

.error:
    lea rsi, [rip + msg_err_prefix]
    mov rdx, 9
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, [rip + g_err]
    lea r10, [rip + numbuf + 31]
    mov byte ptr [r10], 0
    mov rcx, 10
1:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec r10
    mov byte ptr [r10], dl
    test rax, rax
    jnz 1b
    lea r11, [rip + numbuf + 32]
    mov rdx, r11
    sub rdx, r10
    mov rsi, r10
    mov rax, 1
    mov rdi, 1
    syscall
    lea rsi, [rip + msg_newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor r12, r12
1:
    cmp r12, 65536
    jae .done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle .done
    add r12, rax
    cmp r12, 65536
    jl 1b
.done:
    mov qword ptr [rip + g_len], r12
    ret

parse_document:
    call parse_ws
    call parse_value
    test rax, rax
    jnz .fail
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    je .ok
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.ok:
    xor rax, rax
    ret
.fail:
    ret

parse_ws:
    mov rax, [rip + g_pos]
1:
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae 2f
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x20
    je 3f
    cmp dl, 0x09
    je 3f
    cmp dl, 0x0A
    je 3f
    cmp dl, 0x0D
    je 3f
    jmp 2f
3:
    inc rax
    jmp 1b
2:
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret

parse_value:
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '{'
    je .object
    cmp dl, '['
    je .array
    cmp dl, '"'
    je .string
    cmp dl, '-'
    je .number
    cmp dl, '0'
    jb .fail
    cmp dl, '9'
    jbe .number
    cmp dl, 't'
    je .true
    cmp dl, 'f'
    je .false
    cmp dl, 'n'
    je .null
    jmp .fail
.object:
    call parse_object
    ret
.array:
    call parse_array
    ret
.string:
    call parse_string
    ret
.number:
    call parse_number
    ret
.true:
    call parse_true
    ret
.false:
    call parse_false
    ret
.null:
    call parse_null
    ret
.eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.fail:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_object:
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '{'
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '}'
    je .done
.loop:
    call parse_string
    test rax, rax
    jnz .fail
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ':'
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    call parse_value
    test rax, rax
    jnz .fail
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '}'
    je .done
    cmp dl, ','
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    jmp .loop
.done:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
.bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.fail:
    ret

parse_array:
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '['
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ']'
    je .done
.loop:
    call parse_value
    test rax, rax
    jnz .fail
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ']'
    je .done
    cmp dl, ','
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    jmp .loop
.done:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
.bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.fail:
    ret

parse_string:
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '"'
    jne .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
.loop:
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '"'
    je .done
    cmp dl, '\\'
    je .escape
    cmp dl, 0x00
    jb .bad
    cmp dl, 0x20
    jb .bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp .loop
.escape:
    inc rax
    mov qword ptr [rip + g_pos], rax
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '"'
    je .advance
    cmp dl, '\\'
    je .advance
    cmp dl, '/'
    je .advance
    cmp dl, 'b'
    je .advance
    cmp dl, 'f'
    je .advance
    cmp dl, 'n'
    je .advance
    cmp dl, 'r'
    je .advance
    cmp dl, 't'
    je .advance
    cmp dl, 'u'
    je .unicode
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.advance:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp .loop
.unicode:
    inc rax
    mov qword ptr [rip + g_pos], rax
    mov rcx, 4
.hex_loop:
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea r11, [rip + input_buf]
    movzx r8d, byte ptr [r11 + rax]
    cmp r8b, '0'
    jb .bad
    cmp r8b, '9'
    jbe .hex_ok
    cmp r8b, 'a'
    jb .bad
    cmp r8b, 'f'
    jbe .hex_ok
    cmp r8b, 'A'
    jb .bad
    cmp r8b, 'F'
    jbe .hex_ok
    jmp .bad
.hex_ok:
    inc rax
    dec rcx
    jnz .hex_loop
    mov qword ptr [rip + g_pos], rax
    jmp .loop
.done:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
.bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
.eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_number:
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '-'
    jne .no_sign
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    je .zero
    cmp dl, '1'
    jb .bad
    cmp dl, '9'
    jbe .nonzero
    jmp .bad
.no_sign:
    cmp dl, '0'
    je .zero
    cmp dl, '1'
    jb .bad
    cmp dl, '9'
    jbe .nonzero
    jmp .bad

.zero:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae .done
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .after_zero
    cmp dl, '9'
    jbe .bad
.after_zero:
    cmp dl, '.'
    je .frac
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret

.nonzero:
    inc rax
    mov qword ptr [rip + g_pos], rax
.loop_int:
    cmp rax, [rip + g_len]
    jae .after_int
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .after_int
    cmp dl, '9'
    jbe .advance_int
    jmp .after_int
.advance_int:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp .loop_int
.after_int:
    cmp rax, [rip + g_len]
    jae .done
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '.'
    je .frac
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret

.frac:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .bad
    cmp dl, '9'
    jbe .frac_digits
    jmp .bad
.frac_digits:
    inc rax
    mov qword ptr [rip + g_pos], rax
.frac_loop:
    cmp rax, [rip + g_len]
    jae .done
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .after_frac
    cmp dl, '9'
    jbe .advance_frac
    jmp .after_frac
.advance_frac:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp .frac_loop
.after_frac:
    cmp dl, 'e'
    je .exp
    cmp dl, 'E'
    je .exp
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret

.exp:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '+'
    je .exp_sign
    cmp dl, '-'
    je .exp_sign
    jmp .exp_digit_start
.exp_sign:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae .eof
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
.exp_digit_start:
    cmp dl, '0'
    jb .bad
    cmp dl, '9'
    jbe .exp_digits
    jmp .bad
.exp_digits:
    inc rax
    mov qword ptr [rip + g_pos], rax
.exp_digits_loop:
    cmp rax, [rip + g_len]
    jae .done
    lea rcx, [rip + input_buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .done
    cmp dl, '9'
    jbe .advance_exp
    jmp .done
.advance_exp:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp .exp_digits_loop

.bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

.eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

.done:
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
