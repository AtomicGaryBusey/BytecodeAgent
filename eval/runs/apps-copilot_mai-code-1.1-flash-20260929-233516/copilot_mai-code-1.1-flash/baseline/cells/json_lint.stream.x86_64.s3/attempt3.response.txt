.text
.globl _start

_start:
    call read_all
    mov qword ptr [rip + g_pos], 0
    call parse_document
    test rax, rax
    jnz _error
    lea rsi, [rip + msg_ok]
    mov rdx, 3
    mov rax, 1
    mov rdi, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

_error:
    lea rsi, [rip + msg_err_prefix]
    mov rdx, 9
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, [rip + g_err]
    lea r10, [rip + numbuf + 31]
    mov byte ptr [r10], 0
    mov rcx, 10
_error_digit_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec r10
    mov byte ptr [r10], dl
    test rax, rax
    jnz _error_digit_loop

    lea r11, [rip + numbuf + 32]
    mov rdx, r11
    sub rdx, r10
    mov rsi, r10
    mov rax, 1
    mov rdi, 1
    syscall

    lea rsi, [rip + msg_newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall

    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor r12, r12
read_all_loop:
    cmp r12, 65536
    jae read_all_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_all_done
    add r12, rax
    jmp read_all_loop
read_all_done:
    mov qword ptr [rip + g_len], r12
    ret

parse_document:
    call parse_ws
    call parse_value
    test rax, rax
    jnz parse_document_fail
    call parse_ws
    mov rax, [rip + g_pos]
    mov rbx, [rip + g_len]
    cmp rax, rbx
    je parse_document_ok
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_document_ok:
    xor rax, rax
    ret
parse_document_fail:
    ret

parse_ws:
    mov rax, [rip + g_pos]
parse_ws_loop:
    cmp rax, [rip + g_len]
    jae parse_ws_done
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, 0x20
    je parse_ws_advance
    cmp r8b, 0x09
    je parse_ws_advance
    cmp r8b, 0x0A
    je parse_ws_advance
    cmp r8b, 0x0D
    je parse_ws_advance
    jmp parse_ws_done
parse_ws_advance:
    inc rax
    jmp parse_ws_loop
parse_ws_done:
    mov qword ptr [rip + g_pos], rax
    ret

parse_value:
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_value_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '{'
    je parse_value_object
    cmp r8b, '['
    je parse_value_array
    cmp r8b, '"'
    je parse_value_string
    cmp r8b, '-'
    je parse_value_number
    cmp r8b, '0'
    jb parse_value_fail
    cmp r8b, '9'
    jbe parse_value_number
    cmp r8b, 't'
    je parse_value_true
    cmp r8b, 'f'
    je parse_value_false
    cmp r8b, 'n'
    je parse_value_null
    jmp parse_value_fail
parse_value_object:
    call parse_object
    ret
parse_value_array:
    call parse_array
    ret
parse_value_string:
    call parse_string
    ret
parse_value_number:
    call parse_number
    ret
parse_value_true:
    call parse_literal_true
    ret
parse_value_false:
    call parse_literal_false
    ret
parse_value_null:
    call parse_literal_null
    ret
parse_value_fail:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_value_eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_object:
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_object_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '{'
    jne parse_object_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_object_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '}'
    je parse_object_close
parse_object_member_loop:
    call parse_string
    test rax, rax
    jnz parse_object_fail
    call parse_ws
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_object_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, ':'
    jne parse_object_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    call parse_value
    test rax, rax
    jnz parse_object_fail
    call parse_ws
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_object_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '}'
    je parse_object_close
    cmp r8b, ','
    jne parse_object_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    jmp parse_object_member_loop
parse_object_close:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
parse_object_bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_object_eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_object_fail:
    ret

parse_array:
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_array_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '['
    jne parse_array_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_array_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, ']'
    je parse_array_close
parse_array_value_loop:
    call parse_value
    test rax, rax
    jnz parse_array_fail
    call parse_ws
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_array_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, ']'
    je parse_array_close
    cmp r8b, ','
    jne parse_array_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    call parse_ws
    jmp parse_array_value_loop
parse_array_close:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
parse_array_bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_array_eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_array_fail:
    ret

parse_string:
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_string_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '"'
    jne parse_string_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
parse_string_loop:
    cmp rax, [rip + g_len]
    jae parse_string_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '"'
    je parse_string_close
    cmp r8b, '\\'
    je parse_string_escape
    cmp r8b, 0x00
    jb parse_string_bad
    cmp r8b, 0x20
    jb parse_string_bad
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp parse_string_loop
parse_string_escape:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_string_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '"'
    je parse_string_advance
    cmp r8b, '\\'
    je parse_string_advance
    cmp r8b, '/'
    je parse_string_advance
    cmp r8b, 'b'
    je parse_string_advance
    cmp r8b, 'f'
    je parse_string_advance
    cmp r8b, 'n'
    je parse_string_advance
    cmp r8b, 'r'
    je parse_string_advance
    cmp r8b, 't'
    je parse_string_advance
    cmp r8b, 'u'
    je parse_string_unicode
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_string_advance:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp parse_string_loop
parse_string_unicode:
    inc rax
    mov qword ptr [rip + g_pos], rax
    mov rcx, 4
parse_string_hex_loop:
    cmp rax, [rip + g_len]
    jae parse_string_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    jb parse_string_bad
    cmp r8b, '9'
    jbe parse_string_hex_ok
    cmp r8b, 'a'
    jb parse_string_bad
    cmp r8b, 'f'
    jbe parse_string_hex_ok
    cmp r8b, 'A'
    jb parse_string_bad
    cmp r8b, 'F'
    jbe parse_string_hex_ok
    jmp parse_string_bad
parse_string_hex_ok:
    inc rax
    dec rcx
    jnz parse_string_hex_loop
    mov qword ptr [rip + g_pos], rax
    jmp parse_string_loop
parse_string_close:
    inc rax
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
parse_string_bad:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_string_eof:
    mov rax, [rip + g_len]
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_number:
    mov rax, [rip + g_pos]
    cmp rax, [rip + g_len]
    jae parse_number_eof
    lea r9, [rip + input_buf]
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '-'
    jne parse_number_no_sign
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_number_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    je parse_number_zero
    cmp r8b, '1'
    jb parse_number_bad
    cmp r8b, '9'
    jbe parse_number_nonzero
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_number_no_sign:
    cmp r8b, '0'
    je parse_number_zero
    cmp r8b, '1'
    jb parse_number_bad
    cmp r8b, '9'
    jbe parse_number_nonzero
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_number_zero:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_number_success
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    jb parse_number_after_zero
    cmp r8b, '9'
    jbe parse_number_zero_digit_error
parse_number_after_zero:
    cmp r8b, '.'
    je parse_number_frac
    cmp r8b, 'e'
    je parse_number_exp
    cmp r8b, 'E'
    je parse_number_exp
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
parse_number_zero_digit_error:
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret

parse_number_nonzero:
    inc rax
    mov qword ptr [rip + g_pos], rax
parse_number_int_loop:
    cmp rax, [rip + g_len]
    jae parse_number_after_int
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    jb parse_number_after_int
    cmp r8b, '9'
    jbe parse_number_int_advance
    jmp parse_number_after_int
parse_number_int_advance:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp parse_number_int_loop
parse_number_after_int:
    cmp rax, [rip + g_len]
    jae parse_number_success
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '.'
    je parse_number_frac
    cmp r8b, 'e'
    je parse_number_exp
    cmp r8b, 'E'
    je parse_number_exp
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret

parse_number_frac:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_number_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    jb parse_number_bad
    cmp r8b, '9'
    jbe parse_number_frac_digits
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_number_frac_digits:
    inc rax
    mov qword ptr [rip + g_pos], rax
parse_number_frac_loop:
    cmp rax, [rip + g_len]
    jae parse_number_success
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '0'
    jb parse_number_frac_after
    cmp r8b, '9'
    jbe parse_number_frac_advance
    jmp parse_number_frac_after
parse_number_frac_advance:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp parse_number_frac_loop
parse_number_frac_after:
    cmp r8b, 'e'
    je parse_number_exp
    cmp r8b, 'E'
    je parse_number_exp
    mov qword ptr [rip + g_pos], rax

    ret

parse_number_exp:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_number_eof
    movzx r8d, byte ptr [r9 + rax]
    cmp r8b, '+'
    je parse_number_exp_sign
    cmp r8b, '-'
    je parse_number_exp_sign
    jmp parse_number_exp_digits_start
parse_number_exp_sign:
    inc rax
    mov qword ptr [rip + g_pos], rax
    cmp rax, [rip + g_len]
    jae parse_number_eof
    movzx r8d, byte ptr [r9 + rax]
parse_number_exp_digits_start:
    cmp r8b, '0'
    jb parse_number_bad
    cmp r8b, '9'
    jbe parse_number_exp_digits
    mov qword ptr [rip + g_err], rax
    mov rax, 1
    ret
parse_number_exp_digits:
    inc rax
    mov qword ptr [rip + g_pos], rax
parse_number_exp_loop:
    cmp rax, [rip + g_len]
    jae parse_number_success

    cmp r8b, '0'

    jmp parse_number_success
parse_number_exp_advance:
    inc rax
    mov qword ptr [rip + g_pos], rax
    jmp parse_number_exp_loop

parse_number_success:
    mov qword ptr [rip + g_pos], rax
    xor rax, rax
    ret
parse_number_bad:

    mov rax, 1
    ret
parse_number_eof:
