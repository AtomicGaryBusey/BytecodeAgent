  add x18, x18, :lo12:stdinbuf

  add x1, x1, x19
  mov x2, #65536

  mov x8, #63

  cmp x0, #0
  b.lt err_exit
  b.eq read_done

  mov x20, #0
  mov x21, #-1

  cmp x20, x19
  b.hs scan_done
  add x22, x18, x20
  ldrb w23, [x22]
  cmp w23, #0x21
  b.eq found_delim
  add x20, x20, #1
  b scan_loop

  b scan_done

  b.lt no_delim
  mov x28, x21
  add x21, x21, #1
  mov x24, x18
  add x24, x24, x21
  sub x25, x19, x21

  mov x28, x19
  mov x24, x18
  add x24, x24, x19
  mov x25, #0

  add x26, x26, :lo12:match_table
  adrp x27, brstack
  add x27, x27, :lo12:brstack
  mov x29, #0
  mov x30, #0
  cmp x28, #0
  b.eq exec_start

  cmp x30, x28
  b.hs bracket_done
  add x16, x18, x30
  ldrb w17, [x16]
  cmp w17, #0x5b
  b.ne not_open
  cmp x29, #1000
  b.hs err_exit
  add x16, x27, x29, lsl #2
  str w30, [x16]
  add x29, x29, #1
  b bracket_next

  b.ne bracket_next
  cmp x29, #0
  b.eq err_exit
  sub x29, x29, #1
  add x16, x27, x29, lsl #2
  ldr w17, [x16]
  add x16, x26, x17, lsl #2
  str w30, [x16]
  add x16, x26, x30, lsl #2
  str w17, [x16]

  add x30, x30, #1
  b bracket_loop
bracket_done:
  cmp x29, #0
  b.ne err_exit

  adrp x22, tape
  add x22, x22, :lo12:tape
  mov x20, #0
  mov x21, #0
  adrp x23, outbuf
  add x23, x23, :lo12:outbuf
  mov x29, #0
  mov x30, #0
exec_loop:
  cmp x21, x28
  b.hs exec_done
  add x16, x18, x21
  ldrb w17, [x16]
  cmp w17, #0x3e
  b.eq cmd_gt
  cmp w17, #0x3c
  b.eq cmd_lt
  cmp w17, #0x2b
  b.eq cmd_plus
  cmp w17, #0x2d
  b.eq cmd_minus
  cmp w17, #0x2e
  b.eq cmd_dot
  cmp w17, #0x2c
  b.eq cmd_comma
  cmp w17, #0x5b
  b.eq cmd_lbrack
  cmp w17, #0x5d
  b.eq cmd_rbrack
  add x21, x21, #1
  b exec_loop

  add x20, x20, #1
  add x21, x21, #1
  b exec_loop

cmd_lt:
  sub x20, x20, #1
  add x21, x21, #1
  b exec_loop

  ldrb w17, [x16]
  add w17, w17, #1
  strb w17, [x16]
  add x21, x21, #1
  b exec_loop

  ldrb w17, [x16]
  sub w17, w17, #1
  strb w17, [x16]
  add x21, x21, #1
  b exec_loop

  b.hs err_exit
  add x16, x22, x20
  ldrb w17, [x16]
  add x16, x23, x29
  strb w17, [x16]
  add x29, x29, #1
  add x21, x21, #1
  b exec_loop

  b.hs comma_eof
  add x16, x24, x30
  ldrb w17, [x16]
  add x30, x30, #1

  add x21, x21, #1

  ldrb w17, [x16]
  cmp w17, #0
  b.ne lbrack_skip
  add x16, x26, x21, lsl #2
  ldr w17, [x16]
  add x21, x17, #1
  b exec_loop

  add x21, x21, #1
  b exec_loop

cmd_rbrack:
  add x16, x22, x20
  ldrb w17, [x16]
  cmp w17, #0
  b.eq rbrack_skip
  add x16, x26, x21, lsl #2
  ldr w17, [x16]
  add x21, x17, #1
  b exec_loop

  add x21, x21, #1
  b exec_loop

  add x1, x1, :lo12:outbuf
