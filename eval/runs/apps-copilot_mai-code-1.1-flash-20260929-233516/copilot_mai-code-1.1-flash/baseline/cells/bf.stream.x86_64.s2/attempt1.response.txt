.globl _start

.text
_start:
    lea r12, [rip + stdin_buf]
    mov r13, 0
read_loop:
    cmp r13, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rdx, 4096
    lea rsi, [r12 + r13]
    syscall
    cmp rax, 0
    je read_done
    add r13, rax
    cmp r13, 65536
    jb read_done
    jb read_loop
read_done:
    mov r14, r13
    lea r15, [r12 + r13]
    mov r11, 0
    mov rcx, 0
scan_loop:
    cmp rcx, r13
    jae no_bang
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x21
    je bang_found
    inc rcx
    jmp scan_loop
bang_found:
    mov r14, rcx
    lea r15, [r12 + rcx + 1]
    mov r11, r13
    sub r11, rcx
    dec r11
    jmp after_split
no_bang:
after_split:
    lea r11, [r15 + r11]
    mov r8, 0
    mov r9, 0
validate_loop:
    cmp r8, r14
    jae validate_ok
    movzx r10d, byte ptr [r12 + r8]
    cmp r10b, 0x5b
    je validate_open
    cmp r10b, 0x5d
    je validate_close
    inc r8
    jmp validate_loop
validate_open:
    cmp r9, 1000
    jae validate_fail
    mov dword ptr [rip + stack + r9*4], r8d
    inc r9
    inc r8
    jmp validate_loop
validate_close:
    cmp r9, 0
    je validate_fail
    dec r9
    mov eax, dword ptr [rip + stack + r9*4]
    mov dword ptr [rip + match_table + rax*4], r8d
    mov dword ptr [rip + match_table + r8*4], eax
    inc r8
    jmp validate_loop
validate_ok:
    cmp r9, 0
    jne validate_fail
    mov rbx, r12
    lea r12, [rip + tape]
    mov r13, 0
    mov r10, r14
    mov r14, 0
interp_loop:
    cmp r14, r10
    jae interp_done
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x3e
    je op_gt
    cmp al, 0x3c
    je op_lt
    cmp al, 0x2b
    je op_plus
    cmp al, 0x2d
    je op_minus
    cmp al, 0x2e
    je op_dot
    cmp al, 0x2c
    je op_comma
    cmp al, 0x5b
    je op_lbracket
    cmp al, 0x5d
    je op_rbracket
    inc r14
    jmp interp_loop
op_gt:
    inc r13
    inc r14
    jmp interp_loop
op_lt:
    dec r13
    inc r14
    jmp interp_loop
op_plus:
    movzx eax, byte ptr [r12 + r13]
    add al, 1
    mov byte ptr [r12 + r13], al
    inc r14
    jmp interp_loop
op_minus:
    movzx eax, byte ptr [r12 + r13]
    sub al, 1
    mov byte ptr [r12 + r13], al
    inc r14
    jmp interp_loop
op_dot:
    mov rdi, 1
    lea rsi, [r12 + r13]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r14
    jmp interp_loop
op_comma:
    cmp r15, r11
    jae comma_zero
    movzx eax, byte ptr [r15]
    inc r15
    mov byte ptr [r12 + r13], al
    inc r14
    jmp interp_loop
comma_zero:
    mov byte ptr [r12 + r13], 0
    inc r14
    jmp interp_loop
op_lbracket:
    movzx eax, byte ptr [r12 + r13]
    test al, al
    jnz lbracket_continue
    mov eax, dword ptr [rip + match_table + r14*4]
    mov r14d, eax
    inc r14
    jmp interp_loop
lbracket_continue:
    inc r14
    jmp interp_loop
op_rbracket:
    movzx eax, byte ptr [r12 + r13]
    test al, al
    jz rbracket_continue
    mov eax, dword ptr [rip + match_table + r14*4]
    mov r14d, eax
    inc r14
    jmp interp_loop
rbracket_continue:
    inc r14
    jmp interp_loop
interp_done:
    mov rax, 60
    xor edi, edi
    syscall
validate_fail:
    mov rax, 60
    mov rdi, 1
    syscall
