.globl _start

.text
_start:
    lea r12, [rip + stdin_buf]
    xor r13, r13

read_loop:
    cmp r13, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r13
    lea rsi, [r12 + r13]
    syscall
    cmp rax, 0
    je read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r14, r13
    xor r15, r15
    mov rcx, 0

scan_loop:
    cmp rcx, r13
    jae no_bang
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x21
    je bang_found
    inc rcx
    jmp scan_loop

bang_found:
    mov r14, rcx
    lea r15, [r12 + rcx + 1]
    jmp split_done

no_bang:
    mov r14, r13
    lea r15, [r12 + r13]

split_done:
    lea r11, [r12 + r13]
    lea rbx, [rip + match_table]
    lea r10, [rip + open_stack]
    xor r9, r9
    xor rcx, rcx

validate_loop:
    cmp rcx, r14
    jae validate_ok
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x5b
    je validate_open
    cmp al, 0x5d
    je validate_close
    inc rcx
    jmp validate_loop

validate_open:
    cmp r9, 1000
    jae validate_fail
    mov dword ptr [r10 + r9*4], ecx
    inc r9
    inc rcx
    jmp validate_loop

validate_close:
    test r9, r9
    jz validate_fail
    dec r9
    mov eax, dword ptr [r10 + r9*4]
    mov dword ptr [rbx + rax*4], ecx
    mov dword ptr [rbx + rcx*4], eax
    inc rcx
    jmp validate_loop

validate_ok:
    test r9, r9
    jnz validate_fail

    lea rdi, [rip + tape]
    lea r8, [rip + output_buf]
    xor r9, r9
    mov rsi, r12
    lea r13, [r12 + r14]
    mov r10, r15

exec_loop:
    cmp rsi, r13
    jae exec_done
    movzx eax, byte ptr [rsi]

    cmp al, 0x3e
    jne check_lt
    inc rdi
    inc rsi
    jmp exec_loop

check_lt:
    cmp al, 0x3c
    jne check_plus
    dec rdi
    inc rsi
    jmp exec_loop

check_plus:
    cmp al, 0x2b
    jne check_minus
    movzx eax, byte ptr [rdi]
    inc al
    mov byte ptr [rdi], al
    inc rsi
    jmp exec_loop

check_minus:
    cmp al, 0x2d
    jne check_dot
    movzx eax, byte ptr [rdi]
    dec al
    mov byte ptr [rdi], al
    inc rsi
    jmp exec_loop

check_dot:
    cmp al, 0x2e
    jne check_comma
    movzx eax, byte ptr [rdi]
    mov byte ptr [r8 + r9], al
    inc r9
    inc rsi
    jmp exec_loop

check_comma:
    cmp al, 0x2c
    jne check_open
    cmp r10, r11
    jae comma_zero
    movzx eax, byte ptr [r10]
    inc r10
    mov byte ptr [rdi], al
    inc rsi
    jmp exec_loop

comma_zero:
    mov byte ptr [rdi], 0
    inc rsi
    jmp exec_loop

check_open:
    cmp al, 0x5b
    jne check_close
    movzx eax, byte ptr [rdi]
    test al, al
    jne open_continue
    lea rdx, [rip + match_table]
    mov rax, rsi
    sub rax, r12
    mov eax, dword ptr [rdx + rax*4]
    lea rsi, [r12 + rax + 1]
    jmp exec_loop

open_continue:
    inc rsi
    jmp exec_loop

check_close:
    cmp al, 0x5d
    jne other
    movzx eax, byte ptr [rdi]
    test al, al
    je close_continue
    lea rdx, [rip + match_table]
    mov rax, rsi
    sub rax, r12
    mov eax, dword ptr [rdx + rax*4]
    lea rsi, [r12 + rax + 1]
    jmp exec_loop

close_continue:
    inc rsi
    jmp exec_loop

other:
    inc rsi
    jmp exec_loop

exec_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r9
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

validate_fail:
    mov rax, 60
    mov rdi, 1
    syscall

.section .bss
    .align 8
stdin_buf:
    .space 65536
    .align 8
tape:
    .space 30000
    .align 8
output_buf:
    .space 65536
    .align 4
match_table:
    .space 262144
    .align 4
open_stack:
    .space 4000
