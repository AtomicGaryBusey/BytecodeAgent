.text
.globl _start
.p2align 2

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    cmp x19, #65536
    b.ge process_lines

    mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0

    cmp x0, #0
    b.lt fail
    b.eq process_lines

    add x19, x19, x0
    b read_loop

process_lines:
    mov x21, #0
    cmp x21, x19
    b.ge exit0

line_loop:
    mov x22, x21

find_newline:
    cmp x21, x19
    b.ge final_line

    add x23, x20, x21
    ldrb w24, [x23]
    cmp w24, #10
    b.eq newline_hit

    add x21, x21, #1
    b find_newline

newline_hit:
    sub x26, x21, x22
    cmp x26, #0
    b.eq newline_only

    adrp x28, output_buf
    add x28, x28, :lo12:output_buf
    mov x27, #0
    sub x23, x21, #1

reverse_loop:
    cmp x23, x22
    b.lt reverse_done

    add x24, x20, x23
    ldrb w25, [x24]

    add x24, x28, x27
    strb w25, [x24]

    sub x23, x23, #1
    add x27, x27, #1
    b reverse_loop

reverse_done:
    mov x0, #1
    mov x1, x28
    mov x2, x26
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt fail

    mov x0, #1
    adrp x1, newline_data
    add x1, x1, :lo12:newline_data
    mov x2, #1
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt fail

    add x21, x21, #1
    b line_loop

newline_only:
    mov x0, #1
    adrp x1, newline_data
    add x1, x1, :lo12:newline_data
    mov x2, #1
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt fail

    add x21, x21, #1
    b line_loop

final_line:
    sub x26, x19, x22
    cmp x26, #0
    b.eq exit0

    adrp x28, output_buf
    add x28, x28, :lo12:output_buf
    mov x27, #0
    sub x23, x19, #1

final_reverse:
    cmp x23, x22
    b.lt final_done

    add x24, x20, x23
    ldrb w25, [x24]

    add x24, x28, x27
    strb w25, [x24]

    sub x23, x23, #1
    add x27, x27, #1
    b final_reverse

final_done:
    mov x0, #1
    mov x1, x28
    mov x2, x26
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt fail

    mov x0, #1
    adrp x1, newline_data
    add x1, x1, :lo12:newline_data
    mov x2, #1
    mov x8, #64
    svc #0

    cmp x0, #0
    b.lt fail

    b exit0

exit0:
    mov x0, #0
    mov x8, #93
    svc #0

fail:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
.align 2
newline_data:
    .byte 10

.section .bss
.align 4
input_buf:
    .zero 65536
.align 4
output_buf:
    .zero 65536
