  add x0, x0, :lo12:input_idx

  str x1, [x0]

  add x0, x0, :lo12:error_offset

  str x1, [x0]

  cmp x0, #0

  add x9, x9, :lo12:err_num_buf
  add x9, x9, #32

  udiv x5, x3, x4

  mov x8, #SYS_WRITE
  svc #0

  mov x8, #SYS_EXIT
  svc #0

  cmp w3, #'\\'

  cmp w3, #'/'

  cmp w3, #'f'
