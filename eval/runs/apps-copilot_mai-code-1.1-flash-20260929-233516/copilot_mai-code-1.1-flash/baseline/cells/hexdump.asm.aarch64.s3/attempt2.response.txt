.globl _start

.text

_start:
    bl read_all
    bl dump_rows
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
1:
    cmp x10, #65536
    b.hs 2f
    mov x11, #65536
    sub x12, x11, x10
    mov x11, #4096
    cmp x12, x11
    csel x2, x12, x11, ls
    mov x0, #0
    mov x1, x9
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq 2f
    b.lt 3f
    add x9, x9, x0
    add x10, x10, x0
    b 1b
2:
    mov x0, x10
    ret
3:
    mov x0, #1
    mov x8, #93
    svc #0

dump_rows:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x19, x0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, #0

row_loop:
    cmp x20, x19
    b.hs final_line

    sub x22, x19, x20
    mov x24, #16
    cmp x22, x24
    csel x22, x22, x24, ls

    adrp x18, outbuf
    add x18, x18, :lo12:outbuf
    mov x23, x18

    mov x0, x20
    mov x1, x23
    bl emit_hex8
    mov x23, x0

    mov w24, #0x20
    strb w24, [x23], #1
    strb w24, [x23], #1

    mov x24, #0
hex_loop:
    cmp x24, #16
    b.ge hex_done

    cmp x24, x22
    b.ge blank_hex

    add x9, x21, x20
    add x9, x9, x24
    ldrb w0, [x9]
    mov x1, x23
    bl emit_hex2
    mov x23, x0

    mov w9, #0x20
    strb w9, [x23], #1
    b check_extra

blank_hex:
    mov w9, #0x20
    strb w9, [x23], #1
    strb w9, [x23], #1
    strb w9, [x23], #1

check_extra:
    cmp x24, #7
    b.ne no_extra
    mov w9, #0x20
    strb w9, [x23], #1

no_extra:
    add x24, x24, #1
    b hex_loop

hex_done:
    mov w9, #0x20
    strb w9, [x23], #1
    mov w9, #0x7c
    strb w9, [x23], #1

    mov x24, #0
ascii_loop:
    cmp x24, x22
    b.ge ascii_done

    add x9, x21, x20
    add x9, x9, x24
    ldrb w0, [x9]
    cmp w0, #0x20
    b.lt ascii_dot
    cmp w0, #0x7e
    b.gt ascii_dot
    b ascii_store

ascii_dot:
    mov w0, #0x2e

ascii_store:
    strb w0, [x23], #1
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w9, #0x7c
    strb w9, [x23], #1
    mov w9, #0x0a
    strb w9, [x23], #1

    sub x2, x23, x18
    mov x0, #1
    mov x1, x18
    bl write_all

    add x20, x20, x22
    b row_loop

final_line:
    adrp x18, outbuf
    add x18, x18, :lo12:outbuf
    mov x0, x19
    mov x1, x18
    bl emit_hex8
    mov x23, x0
    mov w9, #0x0a
    strb w9, [x23], #1

    sub x2, x23, x18
    mov x0, #1
    mov x1, x18
    bl write_all

    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret

write_all:
    mov x9, x0
    mov x10, x1
    mov x11, x2
1:
    cmp x11, #0
    b.eq 2f
    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 3f
    sub x11, x11, x0
    add x10, x10, x0
    b 1b
2:
    ret
3:
    mov x0, #1
    mov x8, #93
    svc #0

emit_hex8:
    mov x9, x1
    add x9, x9, #7
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits
    mov x11, #8
1:
    and w12, w0, #0xf
    add x13, x10, x12
    ldrb w12, [x13]
    strb w12, [x9]
    sub x9, x9, #1
    lsr x0, x0, #4
    sub x11, x11, #1
    cbnz x11, 1b
    add x0, x1, #8
    ret

emit_hex2:
    adrp x9, hex_digits
    add x9, x9, :lo12:hex_digits
    ubfx w10, w0, #4, #4
    add x11, x9, x10
    ldrb w12, [x11]
    strb w12, [x1]
    and w10, w0, #0xf
    add x11, x9, x10
    ldrb w12, [x11]
    strb w12, [x1, #1]
    add x0, x1, #2
    ret

.section .rodata
.balign 8
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 65536

.balign 16
outbuf:
    .space 128
