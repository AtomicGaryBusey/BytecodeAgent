.globl _start

.text

_start:
    bl read_all
    mov x19, x0
    mov x0, x19
    bl dump_rows
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
1:
    cmp x20, #65536
    b.hs 2f
    mov x21, #65536
    sub x21, x21, x20
    mov x22, #4096
    cmp x21, x22
    csel x2, x21, x22, ls
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq 2f
    cmp x0, #0
    b.lt 3f
    add x20, x20, x0
    b 1b
2:
    mov x0, x20
    ret
3:
    mov x0, #1
    mov x8, #93
    svc #0

dump_rows:
    mov x19, x0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, #0
row_loop:
    cmp x20, x19
    b.hs final_line

    sub x22, x19, x20
    mov x23, #16
    cmp x22, x23
    csel x22, x22, x23, ls

    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x9, x24

    mov x0, x20
    mov x1, x9
    bl emit_hex8
    mov x9, x0

    mov w10, #0x20
    strb w10, [x9], #1
    strb w10, [x9], #1

    mov x23, #0
hex_loop:
    cmp x23, #16
    b.ge hex_done

    cmp x23, x22
    b.ge blank_hex

    add x10, x21, x20
    add x10, x10, x23
    ldrb w0, [x10]
    mov x1, x9
    bl emit_hex2
    mov x9, x0

    mov w10, #0x20
    strb w10, [x9], #1
    b after_hex

blank_hex:
    mov w10, #0x20
    strb w10, [x9], #1
    strb w10, [x9], #1
    strb w10, [x9], #1

after_hex:
    cmp x23, #7
    b.ne no_extra
    mov w10, #0x20
    strb w10, [x9], #1

no_extra:
    add x23, x23, #1
    b hex_loop

hex_done:
    mov w10, #0x20
    strb w10, [x9], #1
    mov w10, #0x7c
    strb w10, [x9], #1

    mov x23, #0
ascii_loop:
    cmp x23, x22
    b.ge ascii_done

    add x10, x21, x20
    add x10, x10, x23
    ldrb w0, [x10]
    cmp w0, #0x20
    b.lt ascii_dot
    cmp w0, #0x7e
    b.gt ascii_dot
    b ascii_store

ascii_dot:
    mov w0, #0x2e

ascii_store:
    strb w0, [x9], #1
    add x23, x23, #1
    b ascii_loop

ascii_done:
    mov w10, #0x7c
    strb w10, [x9], #1
    mov w10, #0x0a
    strb w10, [x9], #1

    sub x2, x9, x24
    mov x0, #1
    mov x1, x24
    bl write_all

    add x20, x20, x22
    b row_loop

final_line:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x0, x19
    mov x1, x24
    bl emit_hex8
    mov x9, x0
    mov w10, #0x0a
    strb w10, [x9], #1

    sub x2, x9, x24
    mov x0, #1
    mov x1, x24
    bl write_all
    ret

write_all:
    mov x9, x0
    mov x10, x1
    mov x11, x2
1:
    cmp x11, #0
    b.eq 2f
    mov x0, x9
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 3f
    sub x11, x11, x0
    add x10, x10, x0
    b 1b
2:
    ret
3:
    mov x0, #1
    mov x8, #93
    svc #0

emit_hex8:
    mov x9, x1
    add x9, x9, #7
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits
    mov x11, #8
1:
    and w12, w0, #0xf
    add x13, x10, x12
    ldrb w12, [x13]
    strb w12, [x9]
    sub x9, x9, #1
    lsr x0, x0, #4
    sub x11, x11, #1
    cbnz x11, 1b
    add x0, x1, #8
    ret

emit_hex2:
    adrp x9, hex_digits
    add x9, x9, :lo12:hex_digits
    ubfx w10, w0, #4, #4
    add x11, x9, x10
    ldrb w12, [x11]
    strb w12, [x1]
    and w10, w0, #0xf
    add x11, x9, x10
    ldrb w12, [x11]
    strb w12, [x1, #1]
    add x0, x1, #2
    ret

.section .rodata
.balign 8
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536

.balign 16
outbuf:
    .zero 256
