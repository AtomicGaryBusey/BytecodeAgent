.text
.globl _start

_start:
    mov x19, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x19
    cmp x2, #0
    b.le read_done
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    beq read_done
    add x19, x19, x0
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    mov x20, #0
    cmp x19, #0
    b.eq print_total

print_rows:
    cmp x20, x19
    b.hs print_total

    sub x2, x19, x20
    cmp x2, #16
    b.le 1f
    mov x2, #16
1:
    mov x3, x2

    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    mov x5, x4

    mov x6, #8
    mov x7, x20

offset_loop:
    lsr x8, x7, #28
    and x8, x8, #0xf
    adr x9, hex_digits
    add x9, x9, x8
    ldrb w10, [x9]
    strb w10, [x4], #1
    lsl x7, x7, #4
    sub x6, x6, #1
    cbnz x6, offset_loop

    mov w8, #0x20
    strb w8, [x4], #1
    strb w8, [x4], #1

    mov x6, #0

    cmp x6, #16
    b.ge ascii_prefix
    cmp x6, x3
    b.ge missing_byte

    add x7, x20, x6
    adrp x8, inbuf
    add x8, x8, :lo12:inbuf
    add x8, x8, x7
    ldrb w9, [x8]

    lsr w10, w9, #4
    adr x11, hex_digits
    add x11, x11, x10
    ldrb w12, [x11]
    strb w12, [x4], #1

    and w10, w9, #0xf
    adr x11, hex_digits
    add x11, x11, x10
    ldrb w12, [x11]
    strb w12, [x4], #1

    mov w10, #0x20
    strb w10, [x4], #1
    cmp x6, #7
    b.ne 1f
    strb w10, [x4], #1
1:
    add x6, x6, #1
    b byte_loop

missing_byte:
    mov w10, #0x20
    strb w10, [x4], #1
    strb w10, [x4], #1
    strb w10, [x4], #1
    cmp x6, #7
    b.ne 1f
    strb w10, [x4], #1
1:
    add x6, x6, #1
    b byte_loop

ascii_prefix:
    mov w10, #0x20
    strb w10, [x4], #1
    mov w10, #0x7c
    strb w10, [x4], #1

    mov x6, #0

    cmp x6, x3
    b.ge ascii_done

    add x7, x20, x6
    adrp x8, inbuf
    add x8, x8, :lo12:inbuf
    add x8, x8, x7
    ldrb w9, [x8]

    cmp w9, #0x20
    b.lt ascii_dot
    cmp w9, #0x7e
    b.hi ascii_dot
    strb w9, [x4], #1
    b ascii_next

ascii_dot:
    mov w10, #0x2e
    strb w10, [x4], #1

ascii_next:
    add x6, x6, #1
    b ascii_loop

ascii_done:
    mov w10, #0x7c
    strb w10, [x4], #1
    mov w10, #0x0a
    strb w10, [x4], #1

    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x4, x1
    mov x8, #64
    svc #0

    add x20, x20, x3
    b print_rows

    add x4, x4, :lo12:outbuf

    mov x6, #8
    mov x7, x19

total_loop:
    lsr x8, x7, #28
    and x8, x8, #0xf
    adr x9, hex_digits
    add x9, x9, x8
    ldrb w10, [x9]
    strb w10, [x4], #1
    lsl x7, x7, #4
    sub x6, x6, #1
    cbnz x6, total_loop

    mov w10, #0x0a
    strb w10, [x4], #1

    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x4, x1
    mov x8, #64
    svc #0

exit_done:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
