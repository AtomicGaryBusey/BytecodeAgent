.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x19
    cmp x2, #0
    b.le read_done
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.lt read_error
    beq read_done
    add x19, x19, x0
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    cmp x19, #0
    b.eq print_total
    mov x20, #0

print_rows:
    cmp x20, x19
    b.hs print_total

    sub x2, x19, x20
    cmp x2, #16
    b.le row_length_done
    mov x2, #16

row_length_done:
    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    mov x12, x4

    mov x6, #8
    mov x7, x20

offset_digits_loop:
    lsr x8, x7, #28
    and x8, x8, #0xf
    adrp x9, hex_digits
    add x9, x9, :lo12:hex_digits
    add x9, x9, x8
    ldrb w10, [x9]
    strb w10, [x4], #1
    lsl x7, x7, #4
    sub x6, x6, #1
    cbnz x6, offset_digits_loop

    mov w9, #0x20
    strb w9, [x4], #1
    strb w9, [x4], #1

    mov x6, #0

slot_loop:
    cmp x6, #16
    b.ge ascii_prefix
    add x7, x20, x6
    cmp x7, x19
    b.hs missing_slot

    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x7
    ldrb w8, [x1]

    lsr w9, w8, #4
    and w9, w9, #0xf
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits
    add x10, x10, x9
    ldrb w11, [x10]
    strb w11, [x4], #1

    and w9, w8, #0xf
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits
    add x10, x10, x9
    ldrb w11, [x10]
    strb w11, [x4], #1

    mov w9, #0x20
    strb w9, [x4], #1
    cmp x6, #7
    b.ne slot_continue
    strb w9, [x4], #1

slot_continue:
    add x6, x6, #1
    b slot_loop

missing_slot:
    mov w9, #0x20
    strb w9, [x4], #1
    strb w9, [x4], #1
    strb w9, [x4], #1
    cmp x6, #7
    b.ne missing_continue
    strb w9, [x4], #1

missing_continue:
    add x6, x6, #1
    b slot_loop

ascii_prefix:
    mov w9, #0x20
    strb w9, [x4], #1
    mov w9, #0x7c
    strb w9, [x4], #1

    mov x6, #0

ascii_loop:
    cmp x6, x2
    b.ge ascii_done
    add x7, x20, x6
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x7
    ldrb w8, [x1]

    cmp w8, #0x20
    b.lt ascii_dot
    cmp w8, #0x7e
    b.hi ascii_dot
    strb w8, [x4], #1
    b ascii_next

ascii_dot:
    mov w9, #0x2e
    strb w9, [x4], #1

ascii_next:
    add x6, x6, #1
    b ascii_loop

ascii_done:
    mov w9, #0x7c
    strb w9, [x4], #1
    mov w9, #0x0a
    strb w9, [x4], #1

    sub x5, x4, x12
    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x2, x5
    mov x8, #64
    svc #0

    add x20, x20, x2
    b print_rows

print_total:
    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    mov x6, #8
    mov x7, x19

total_digit_loop:
    lsr x8, x7, #28
    and x8, x8, #0xf
    adrp x9, hex_digits
    add x9, x9, :lo12:hex_digits
    add x9, x9, x8
    ldrb w10, [x9]
    strb w10, [x4], #1
    lsl x7, x7, #4
    sub x6, x6, #1
    cbnz x6, total_digit_loop

    mov w9, #0x0a
    strb w9, [x4], #1

    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x4, x1
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 128
