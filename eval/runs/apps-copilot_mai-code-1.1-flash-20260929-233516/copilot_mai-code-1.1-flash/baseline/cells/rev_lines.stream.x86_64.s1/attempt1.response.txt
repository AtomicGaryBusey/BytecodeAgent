.intel_syntax noprefix

.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jl read_error
    je done_reading
    add r12, rax
    cmp r12, 65536
    jb read_loop

done_reading:
    cmp r12, 0
    je exit_zero

    lea r15, [rip + input_buffer]
    xor r13, r13

process_loop:
    cmp r13, r12
    jae exit_zero

    mov r14, r13

scan_line:
    cmp r14, r12
    jae final_line
    mov dl, byte ptr [r15 + r14]
    cmp dl, 0x0A
    je line_found
    inc r14
    jmp scan_line

line_found:
    mov r8, r14
    sub r8, r13

    mov r9, r14
    dec r9
    lea r10, [rip + output_scratch]
    xor rcx, rcx

copy_line:
    cmp rcx, r8
    jae emit_line
    mov dl, byte ptr [r15 + r9]
    mov byte ptr [r10 + rcx], dl
    dec r9
    inc rcx
    jmp copy_line

emit_line:
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    mov r13, r14
    inc r13
    jmp process_loop

final_line:
    mov r8, r12
    sub r8, r13

    mov r9, r12
    dec r9
    lea r10, [rip + output_scratch]
    xor rcx, rcx

copy_final:
    cmp rcx, r8
    jae emit_final
    mov dl, byte ptr [r15 + r9]
    mov byte ptr [r10 + rcx], dl
    dec r9
    inc rcx
    jmp copy_final

emit_final:
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    jmp exit_zero

read_error:
    mov rax, 60
    mov rdi, 1
    syscall

exit_zero:
    xor edi, edi
    mov rax, 60
    syscall

.section .bss
.align 16
input_buffer:
    .space 65536

output_scratch:
    .space 65536

.section .rodata
newline_byte:
    .byte 0x0A
