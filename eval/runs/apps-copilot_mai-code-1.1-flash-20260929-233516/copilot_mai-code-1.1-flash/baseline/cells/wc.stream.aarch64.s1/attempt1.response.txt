.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, done_read

    mov x23, x0
    mov x24, x1
    mov x25, x23

process_chunk:
    ldrb w26, [x24], #1
    add x21, x21, #1

    cmp w26, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    cmp w26, #0x20
    b.eq ws
    cmp w26, #0x09
    b.eq ws
    cmp w26, #0x0A
    b.eq ws
    cmp w26, #0x0B
    b.eq ws
    cmp w26, #0x0C
    b.eq ws
    cmp w26, #0x0D
    b.eq ws

non_ws:
    cbz x22, word_start
    b next_byte

word_start:
    add x20, x20, #1
    mov x22, #1
    b next_byte

ws:
    mov x22, #0

next_byte:
    subs x25, x25, #1
    b.ne process_chunk
    b read_loop

done_read:
    mov x0, x19
    bl print_dec
    mov x0, #0x20
    bl put_char
    mov x0, x20
    bl print_dec
    mov x0, #0x20
    bl put_char
    mov x0, x21
    bl print_dec
    mov x0, #0x0A
    bl put_char
    mov x0, #0
    mov x8, #93
    svc #0

print_dec:
    adrp x9, tmpbuf
    add x9, x9, :lo12:tmpbuf
    mov x10, x9
    add x11, x9, #32
    mov x12, #10

    cbz x0, zero_digits

digit_loop:
    cbz x0, digit_done
    udiv x13, x0, x12
    msub x14, x13, x12, x0
    add w14, w14, #'0'
    sub x11, x11, #1
    strb w14, [x11]
    mov x0, x13
    b digit_loop

zero_digits:
    mov w14, #'0'
    sub x11, x11, #1
    strb w14, [x11]

digit_done:
    add x15, x10, #32
    sub x2, x15, x11
    mov x0, #1
    mov x1, x11
    mov x8, #64
    svc #0
    ret

put_char:
    adrp x9, charbuf
    add x9, x9, :lo12:charbuf
    strb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    ret

.section .bss
.balign 16
inbuf:
    .zero 4096

.balign 16
tmpbuf:
    .zero 32

.balign 8
charbuf:
    .zero 8
