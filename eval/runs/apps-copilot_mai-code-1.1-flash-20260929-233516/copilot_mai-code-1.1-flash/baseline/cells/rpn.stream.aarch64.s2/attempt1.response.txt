.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_CAPACITY, 65536
.equ STACK_WORDS, 1024
.equ STACK_BYTES, STACK_WORDS * 8

_start:
    mov x29, xzr
    mov x30, xzr

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
    movz x11, #1, lsl #16

read_loop:
    cmp x10, x11
    b.hs read_done

    mov x0, #0
    add x1, x9, x10
    sub x2, x11, x10
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    b.eq read_done
    add x10, x10, x0
    b read_loop

read_done:
    mov x12, x10
    mov x13, #0

parse_lines:
    cmp x13, x12
    b.hs parse_done

    mov x14, x13
find_nl:
    cmp x14, x12
    b.hs no_nl

    add x15, x9, x14
    ldrb w16, [x15]
    cmp w16, #10
    b.eq found_nl

    add x14, x14, #1
    b find_nl

found_nl:
    mov x17, x13
    mov x18, x14
    add x14, x14, #1
    b process_line

no_nl:
    mov x17, x13
    mov x18, x12
    mov x14, x12
    b process_line

process_line:
    adrp x19, stack
    add x19, x19, :lo12:stack
    mov x20, #0
    mov x7, #0
    mov x21, x17

token_loop:
    cmp x21, x18
    b.hs token_loop_done

skip_space:
    cmp x21, x18
    b.hs token_loop_done
    add x15, x9, x21
    ldrb w16, [x15]
    cmp w16, #0x20
    b.eq skip_space_next
    cmp w16, #0x09
    b.eq skip_space_next
    b start_token

skip_space_next:
    add x21, x21, #1
    b skip_space

start_token:
    mov x22, x21

scan_token:
    cmp x21, x18
    b.hs token_parsed
    add x15, x9, x21
    ldrb w16, [x15]
    cmp w16, #0x20
    b.eq token_parsed
    cmp w16, #0x09
    b.eq token_parsed
    add x21, x21, #1
    b scan_token

token_parsed:
    sub x23, x21, x22

    cmp x23, #1
    b.ne maybe_int

    add x15, x9, x22
    ldrb w16, [x15]

    cmp w16, #43
    b.eq op_plus
    cmp w16, #45
    b.eq op_minus
    cmp w16, #42
    b.eq op_mul
    cmp w16, #47
    b.eq op_div
    cmp w16, #37
    b.eq op_mod
    b maybe_int

maybe_int:
    mov x24, x23
    mov x26, #0
    add x15, x9, x22
    ldrb w16, [x15]
    cmp w16, #45
    b.ne num_digits
    mov x26, #1
    sub x24, x24, #1
    cmp x24, #0
    b.eq token_err
    add x22, x22, #1

num_digits:
    cmp x24, #0
    b.eq token_err
    cmp x24, #18
    b.gt token_err

    mov x28, #0
    mov x27, #0

num_loop:
    cmp x28, x24
    b.hs num_done
    add x15, x9, x22
    add x15, x15, x28
    ldrb w16, [x15]
    cmp w16, #48
    b.lt token_err
    cmp w16, #57
    b.gt token_err
    sub w16, w16, #48
    mov x29, x27
    mov x30, #10
    madd x27, x29, x30, x16
    add x28, x28, #1
    b num_loop

num_done:
    cmp x26, #0
    b.eq push_literal
    neg x27, x27

push_literal:
    str x27, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

op_plus:
    cmp x20, #2
    b.lt op_error
    sub x20, x20, #1
    ldr x25, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x24, [x19, x20, lsl #3]
    add x24, x24, x25
    str x24, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

op_minus:
    cmp x20, #2
    b.lt op_error
    sub x20, x20, #1
    ldr x25, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x24, [x19, x20, lsl #3]
    sub x24, x24, x25
    str x24, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

op_mul:
    cmp x20, #2
    b.lt op_error
    sub x20, x20, #1
    ldr x25, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x24, [x19, x20, lsl #3]
    mul x24, x24, x25
    str x24, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

op_div:
    cmp x20, #2
    b.lt op_error
    sub x20, x20, #1
    ldr x25, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x24, [x19, x20, lsl #3]

    cmp x25, #0
    b.eq op_error

    movz x26, #0x8000, lsl #48
    cmp x24, x26
    b.ne div_ok
    cmp x25, #-1
    b.eq op_error

div_ok:
    sdiv x24, x24, x25
    str x24, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

op_mod:
    cmp x20, #2
    b.lt op_error
    sub x20, x20, #1
    ldr x25, [x19, x20, lsl #3]
    sub x20, x20, #1
    ldr x24, [x19, x20, lsl #3]

    cmp x25, #0
    b.eq op_error

    movz x26, #0x8000, lsl #48
    cmp x24, x26
    b.ne mod_ok
    cmp x25, #-1
    b.eq op_error

mod_ok:
    sdiv x26, x24, x25
    msub x24, x26, x25, x24
    str x24, [x19, x20, lsl #3]
    add x20, x20, #1
    b token_loop

token_err:
    mov x7, #1
    b line_end

op_error:
    mov x7, #1
    b line_end

token_loop_done:
line_end:
    cmp x7, #0
    b.ne emit_error
    cmp x20, #1
    b.ne emit_error

    ldr x24, [x19]

    adrp x9, printbuf
    add x9, x9, :lo12:printbuf
    add x10, x9, #65
    mov x11, x24
    mov x12, x10
    mov x13, #10
    mov x14, x11

    cmp x14, #0
    b.ge print_abs_positive
    add x14, x14, #1
    neg x14, x14
    add x14, x14, #1
    b print_digits

print_abs_positive:
    mov x14, x11

print_digits:
    mov x15, x14
    mov x16, #0

print_digit_loop:
    cmp x15, #0
    b.eq print_digit_zero
    udiv x17, x15, x13
    msub x18, x17, x13, x15
    add x18, x18, #48
    sub x12, x12, #1
    strb w18, [x12]
    mov x15, x17
    b print_digit_loop

print_digit_zero:
    mov w18, #'0'
    sub x12, x12, #1
    strb w18, [x12]

    cmp x11, #0
    b.ge print_newline
    sub x12, x12, #1
    mov w18, #45
    strb w18, [x12]

print_newline:
    mov w18, #10
    add x13, x9, #64
    strb w18, [x13]

    sub x2, x13, x12
    add x2, x2, #1
    mov x0, #1
    mov x1, x12
    mov x8, #SYS_WRITE
    svc #0

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x13, x14
    b parse_lines

emit_error:
    adrp x1, errstr
    add x1, x1, :lo12:errstr
    mov x2, #6
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x13, x14
    b parse_lines

parse_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

    .bss
    .balign 8
input_buf:
    .space INPUT_CAPACITY

    .balign 8
stack:
    .space STACK_BYTES

    .balign 8
printbuf:
    .space 65

    .section .rodata
errstr:
    .asciz "error\n"
