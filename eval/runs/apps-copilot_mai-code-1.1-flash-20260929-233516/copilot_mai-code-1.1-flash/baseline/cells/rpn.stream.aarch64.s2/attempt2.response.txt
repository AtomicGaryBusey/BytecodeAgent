.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_CAPACITY, 65536
.equ STACK_CAPACITY, 8192
.equ OUTBUF_SIZE, 64

_start:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0

read_loop:
    cmp x10, #INPUT_CAPACITY
    b.hs read_done
    mov x0, #0
    add x1, x9, x10
    mov x2, #INPUT_CAPACITY
    sub x2, x2, x10
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.lt bad_exit
    b.eq read_done
    add x10, x10, x0
    b read_loop

bad_exit:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

read_done:
    mov x11, #0

outer_loop:
    cmp x11, x10
    b.hs end_input
    mov x12, x11
    mov x13, x11

find_nl:
    cmp x13, x10
    b.hs no_nl
    add x15, x9, x13
    ldrb w16, [x15]
    cmp w16, #10
    b.eq newline_found
    add x13, x13, #1
    b find_nl

newline_found:
    mov x14, x13
    add x13, x13, #1
    mov x19, x13
    b process_line

no_nl:
    mov x14, x10
    mov x19, x10
    b process_line

process_line:
    mov x20, #0
    mov x21, x12
    mov x22, x14
    mov x23, #0

eval_skip:
    cmp x21, x22
    b.hs finish_line
    add x15, x9, x21
    ldrb w16, [x15]
    cmp w16, #0x20
    b.eq skip_one
    cmp w16, #0x09
    b.eq skip_one
    b parse_token

skip_one:
    add x21, x21, #1
    b eval_skip

parse_token:
    mov x24, x21
    mov x25, x21

scan_token:
    cmp x25, x22
    b.hs token_done
    add x15, x9, x25
    ldrb w16, [x15]
    cmp w16, #0x20
    b.eq token_done
    cmp w16, #0x09
    b.eq token_done
    add x25, x25, #1
    b scan_token

token_done:
    sub x26, x25, x24
    cmp x26, #1
    b.ne parse_literal
    add x15, x9, x24
    ldrb w27, [x15]
    cmp w27, #43
    b.eq do_op
    cmp w27, #45
    b.eq do_op
    cmp w27, #42
    b.eq do_op
    cmp w27, #47
    b.eq do_op
    cmp w27, #37
    b.eq do_op
    b parse_literal

parse_literal:
    mov x27, #1
    mov x28, x24
    mov x29, x26
    add x15, x9, x24
    ldrb w16, [x15]
    cmp w16, #45
    b.ne check_digits
    mov x27, #-1
    add x28, x28, #1
    sub x29, x29, #1
    cmp x29, #0
    b.eq line_error

check_digits:
    cmp x29, #18
    b.gt line_error
    mov x30, #0

digit_loop:
    cmp x29, #0
    b.eq digits_done
    add x15, x9, x28
    ldrb w16, [x15]
    cmp w16, #48
    b.lt line_error
    cmp w16, #57
    b.gt line_error
    sub w16, w16, #48
    mov x17, x30
    mov x18, #10
    madd x30, x17, x18, x16
    add x28, x28, #1
    sub x29, x29, #1
    b digit_loop

digits_done:
    cmp x27, #-1
    b.ne push_literal
    neg x30, x30

push_literal:
    adrp x17, stack
    add x17, x17, :lo12:stack
    add x18, x17, x20, lsl #3
    str x30, [x18]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

line_error:
    mov x23, #1
    mov x21, x22
    b finish_line

do_op:
    cmp w27, #43
    b.eq op_plus
    cmp w27, #45
    b.eq op_minus
    cmp w27, #42
    b.eq op_mul
    cmp w27, #47
    b.eq op_div
    cmp w27, #37
    b.eq op_mod
    b line_error

op_plus:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x31, [x17, x20, lsl #3]
    sub x20, x20, #1
    ldr x30, [x17, x20, lsl #3]
    add x30, x30, x31
    str x30, [x17, x20, lsl #3]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

op_minus:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x31, [x17, x20, lsl #3]
    sub x20, x20, #1
    ldr x30, [x17, x20, lsl #3]
    sub x30, x30, x31
    str x30, [x17, x20, lsl #3]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

op_mul:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x31, [x17, x20, lsl #3]
    sub x20, x20, #1
    ldr x30, [x17, x20, lsl #3]
    mul x30, x30, x31
    str x30, [x17, x20, lsl #3]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

op_div:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x31, [x17, x20, lsl #3]
    sub x20, x20, #1
    ldr x30, [x17, x20, lsl #3]
    cmp x31, #0
    b.eq line_error
    movz x28, #0x8000, lsl #48
    cmp x30, x28
    b.ne div_ok
    cmp x31, #-1
    b.eq line_error
div_ok:
    sdiv x30, x30, x31
    str x30, [x17, x20, lsl #3]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

op_mod:
    cmp x20, #2
    b.lt line_error
    sub x20, x20, #1
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x31, [x17, x20, lsl #3]
    sub x20, x20, #1
    ldr x30, [x17, x20, lsl #3]
    cmp x31, #0
    b.eq line_error
    movz x28, #0x8000, lsl #48
    cmp x30, x28
    b.ne mod_ok
    cmp x31, #-1
    b.eq line_error
mod_ok:
    sdiv x28, x30, x31
    msub x30, x28, x31, x30
    str x30, [x17, x20, lsl #3]
    add x20, x20, #1
    mov x21, x25
    b eval_skip

finish_line:
    cmp x23, #0
    b.ne emit_error
    cmp x20, #1
    b.ne emit_error
    adrp x17, stack
    add x17, x17, :lo12:stack
    ldr x24, [x17]
    b print_value

emit_error:
    adrp x1, errstr
    add x1, x1, :lo12:errstr
    mov x2, #6
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    b next_line

print_value:
    mov x11, x24
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x13, x9, #OUTBUF_SIZE
    mov x10, x13
    cmp x11, #0
    b.ge value_nonneg
    movz x12, #0x8000, lsl #48
    cmp x11, x12
    b.eq print_min_value
    neg x11, x11
    mov x12, #1
    b build_digits

value_nonneg:
    mov x12, #0
    b build_digits

print_min_value:
    adrp x1, min_neg_str
    add x1, x1, :lo12:min_neg_str
    mov x2, #21
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    b next_line

build_digits:
    mov x14, x11
    cmp x14, #0
    b.ne digit_proc
    sub x10, x10, #1
    mov w15, #48
    strb w15, [x10]
    b add_sign

digit_proc:
    cmp x14, #0
    b.eq add_sign
    mov x17, #10
    udiv x16, x14, x17
    msub x15, x16, x17, x14
    add w15, w15, #48
    sub x10, x10, #1
    strb w15, [x10]
    mov x14, x16
    b digit_proc

add_sign:
    cmp x12, #0
    b.eq add_newline
    sub x10, x10, #1
    mov w15, #45
    strb w15, [x10]

add_newline:
    sub x13, x13, #1
    mov w15, #10
    strb w15, [x13]
    mov x0, #1
    mov x1, x10
    add x2, x13, #1
    sub x2, x2, x10
    mov x8, #SYS_WRITE
    svc #0

next_line:
    mov x11, x19
    b outer_loop

end_input:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 8
input_buf:
    .space INPUT_CAPACITY

.balign 8
stack:
    .space STACK_CAPACITY

.balign 8
outbuf:
    .space OUTBUF_SIZE

.section .rodata
errstr:
    .asciz "error\n"

min_neg_str:
    .asciz "-9223372036854775808\n"
