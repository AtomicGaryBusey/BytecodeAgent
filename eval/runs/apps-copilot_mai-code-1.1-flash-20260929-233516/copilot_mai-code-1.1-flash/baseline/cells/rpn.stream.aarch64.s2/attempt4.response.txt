.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 65536
.equ EVAL_STACK_SIZE, 65536
.equ OUTBUF_SIZE, 256

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #INPUT_MAX
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.lt read_fail
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_fail:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

read_done:
    mov x21, #0

process_lines:
    cmp x21, x20
    b.hs exit_ok
    mov x22, x21

find_newline:
    cmp x22, x20
    b.hs no_newline
    ldrb w23, [x19, x22]
    cmp w23, #10
    b.eq newline_found
    add x22, x22, #1
    b find_newline

newline_found:
    mov x24, x22
    b eval_line

no_newline:
    mov x24, x20

eval_line:
    adrp x26, eval_stack
    add x26, x26, :lo12:eval_stack
    mov x27, #0
    mov x28, #0
    mov x29, x21
    mov x30, x24

line_scan:
    cmp x29, x30
    b.hs line_done
skip_separators:
    cmp x29, x30
    b.hs line_done
    ldrb w9, [x19, x29]
    cmp w9, #0x20
    b.eq skip_sep_next
    cmp w9, #0x09
    b.eq skip_sep_next
    b parse_token

skip_sep_next:
    add x29, x29, #1
    b skip_separators

parse_token:
    mov x9, x29
    add x29, x29, #1

scan_token:
    cmp x29, x30
    b.hs token_end
    ldrb w10, [x19, x29]
    cmp w10, #0x20
    b.eq token_end
    cmp w10, #0x09
    b.eq token_end
    add x29, x29, #1
    b scan_token

token_end:
    sub x10, x29, x9
    cmp x10, #1
    b.ne parse_literal
    ldrb w11, [x19, x9]
    cmp w11, #43
    b.eq do_operator
    cmp w11, #45
    b.eq do_operator
    cmp w11, #42
    b.eq do_operator
    cmp w11, #47
    b.eq do_operator
    cmp w11, #37
    b.eq do_operator

parse_literal:
    mov x11, #0
    mov x12, x9
    ldrb w13, [x19, x12]
    cmp w13, #45
    b.ne digit_loop_start
    mov x11, #1
    add x12, x12, #1
    cmp x12, x29
    b.hs invalid_token

digit_loop_start:
    mov x13, #0
    mov x14, #0

digit_loop:
    cmp x12, x29
    b.hs digits_done
    ldrb w15, [x19, x12]
    cmp w15, #48
    b.lt invalid_token
    cmp w15, #57
    b.gt invalid_token
    add x13, x13, #1
    cmp x13, #20
    b.gt invalid_token
    mov x16, #10
    mul x17, x14, x16
    sub w15, w15, #48
    add x14, x17, x15
    add x12, x12, #1
    b digit_loop

digits_done:
    cmp x13, #0
    b.eq invalid_token
    cmp x11, #0
    b.eq positive_check

negative_check:
    mov x15, #0
    movk x15, #0x8000, lsl #48
    cmp x14, x15
    b.hi invalid_token
    sub x14, xzr, x14
    b push_literal

positive_check:
    mov x15, #0xffff
    movk x15, #0xffff, lsl #16
    movk x15, #0xffff, lsl #32
    movk x15, #0x7fff, lsl #48
    cmp x14, x15
    b.hi invalid_token

push_literal:
    str x14, [x26, x27, lsl #3]
    add x27, x27, #1
    b line_scan

do_operator:
    cmp x27, #2
    b.lt invalid_token
    sub x27, x27, #1
    ldr x14, [x26, x27, lsl #3]
    sub x27, x27, #1
    ldr x15, [x26, x27, lsl #3]
    cmp w11, #43
    b.eq op_plus
    cmp w11, #45
    b.eq op_minus
    cmp w11, #42
    b.eq op_mul
    cmp w11, #47
    b.eq op_div
    cmp w11, #37
    b.eq op_mod
    b invalid_token

op_plus:
    add x16, x15, x14
    b store_result

op_minus:
    sub x16, x15, x14
    b store_result

op_mul:
    mul x16, x15, x14
    b store_result

op_div:
    cmp x14, #0
    b.eq line_invalid
    mov x17, #0
    movk x17, #0x8000, lsl #48
    cmp x15, x17
    b.ne div_ok
    cmp x14, #-1
    b.ne div_ok
    b line_invalid

div_ok:
    sdiv x16, x15, x14
    b store_result

op_mod:
    cmp x14, #0
    b.eq line_invalid
    mov x17, #0
    movk x17, #0x8000, lsl #48
    cmp x15, x17
    b.ne mod_ok
    cmp x14, #-1
    b.ne mod_ok
    b line_invalid

mod_ok:
    sdiv x16, x15, x14
    msub x16, x16, x14, x15
    b store_result

store_result:
    str x16, [x26, x27, lsl #3]
    add x27, x27, #1
    b line_scan

line_invalid:
    mov x28, #1
    b line_done

invalid_token:
    mov x28, #1
    b line_done

line_done:
    cmp x28, #0
    b.ne emit_error
    cmp x27, #1
    b.ne emit_error
    ldr x0, [x26]

    mov x4, x0
    mov x11, #0
    cmp x4, #0
    b.ge value_nonneg
    mov x11, #1
    mov x10, #0
    movk x10, #0x8000, lsl #48
    cmp x4, x10
    b.eq print_min_value
    neg x4, x4

value_nonneg:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x3, x1, #OUTBUF_SIZE

digit_loop_out:
    cmp x4, #0
    b.eq digits_out_done
    mov x5, #10
    udiv x6, x4, x5
    msub x7, x6, x5, x4
    add w7, w7, #'0'
    sub x3, x3, #1
    strb w7, [x3]
    mov x4, x6
    b digit_loop_out

digits_out_done:
    cmp x11, #0
    b.eq no_sign
    sub x3, x3, #1
    mov w7, #'-'
    strb w7, [x3]

no_sign:
    mov x0, #1
    mov x2, x1
    add x2, x2, #OUTBUF_SIZE
    sub x2, x2, x3
    mov x1, x3
    mov x8, #SYS_WRITE
    svc #0
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    b continue_after_line

print_min_value:
    mov x0, #1
    adrp x1, min_value
    add x1, x1, :lo12:min_value
    mov x2, #20
    mov x8, #SYS_WRITE
    svc #0
    b continue_after_line

emit_error:
    mov x0, #1
    adrp x1, msg_error
    add x1, x1, :lo12:msg_error
    mov x2, #6
    mov x8, #SYS_WRITE
    svc #0

continue_after_line:
    cmp x24, x20
    b.eq end_of_input_reached
    add x21, x24, #1
    b process_lines

end_of_input_reached:
    mov x21, x20
    b exit_ok

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
msg_error:
    .asciz "error\n"
newline:
    .byte 10
min_value:
    .asciz "-9223372036854775808"

.section .bss
.balign 8
input_buf:
    .space INPUT_MAX

.balign 8
eval_stack:
    .space EVAL_STACK_SIZE

.balign 8
outbuf:
    .space OUTBUF_SIZE
