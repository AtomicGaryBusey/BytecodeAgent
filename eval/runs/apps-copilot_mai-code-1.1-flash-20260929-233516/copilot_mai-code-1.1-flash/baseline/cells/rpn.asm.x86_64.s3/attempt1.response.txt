.globl _start

.text
_start:
    lea r15, [rip + stack_top]
    xor r13d, r13d
    lea r12, [rip + input_buf]

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r13
    mov edx, 65536
    sub edx, r13d
    cmp edx, 0
    jle done_read
    syscall
    cmp rax, 0
    je done_read
    add r13, rax
    jmp read_loop

done_read:
    mov r14, r13
    cmp r13, 0
    je exit_ok

line_loop:
    cmp r12, r14
    jae exit_ok
    mov rbx, r12

find_line_end:
    cmp rbx, r14
    jae line_found
    cmp byte ptr [rbx], 10
    je line_found
    inc rbx
    jmp find_line_end

line_found:
    jmp process_line

process_return:
    cmp r10, r14
    je final_line_end
    mov r12, r10
    inc r12
    jmp line_loop

final_line_end:
    mov r12, r14
    jmp exit_ok

process_line:
    mov r9, 0
    lea r15, [rip + stack_top]
    mov rbx, r12
    mov r10, rbx
    mov rax, r14
    sub rax, r12
    ; rax is not used below; r10 is line_end in main; restore r10 = r14? no, line end is in r14, not r10
    ; we need to know current line end while parsing:
    ; r10 is line_end and is set before jumping here, so keep it.
    ; simply continue.
parse_next:
    cmp rbx, r10
    jae line_finalize
skip_sep:
    cmp rbx, r10
    jae line_finalize
    mov dl, byte ptr [rbx]
    cmp dl, 0x20
    je sep_inc
    cmp dl, 0x09
    je sep_inc
    jmp parse_token
sep_inc:
    inc rbx
    jmp skip_sep

parse_token:
    mov rsi, rbx
    mov rdi, rbx
token_scan:
    cmp rdi, r10
    je token_done
    mov dl, byte ptr [rdi]
    cmp dl, 0x20
    je token_done
    cmp dl, 0x09
    je token_done
    inc rdi
    jmp token_scan

token_done:
    mov rcx, rdi
    sub rcx, rsi
    mov al, byte ptr [rsi]

    cmp al, '+'
    jne check_minus
    cmp rcx, 1
    jne invalid_line
    jmp do_add

check_minus:
    cmp al, '-'
    jne check_star
    cmp rcx, 1
    je do_sub

    mov rdx, rcx
    dec rdx
    cmp rdx, 18
    ja invalid_line

    mov rbx, rsi
    inc rbx
neg_digit_check:
    cmp rbx, rdi
    je neg_parse
    mov dl, byte ptr [rbx]
    cmp dl, '0'
    jb invalid_line
    cmp dl, '9'
    ja invalid_line
    inc rbx
    jmp neg_digit_check

neg_parse:
    xor eax, eax
    mov rbx, rsi
    inc rbx
neg_loop:
    cmp rbx, rdi
    je neg_done
    movzx r8d, byte ptr [rbx]
    sub r8d, '0'
    imul rax, rax, 10
    add rax, r8
    inc rbx
    jmp neg_loop

neg_done:
    neg rax
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

check_star:
    cmp al, '*'
    jne check_div
    cmp rcx, 1
    jne invalid_line
    jmp do_mul

check_div:
    cmp al, '/'
    jne check_mod
    cmp rcx, 1
    jne invalid_line
    jmp do_div

check_mod:
    cmp al, '%'
    jne check_literal
    cmp rcx, 1
    jne invalid_line
    jmp do_mod

check_literal:
    cmp al, '0'
    jb invalid_line
    cmp al, '9'
    ja invalid_line
    cmp rcx, 18
    ja invalid_line

    xor eax, eax
    mov rbx, rsi
pos_loop:
    cmp rbx, rdi
    je pos_done
    movzx r8d, byte ptr [rbx]
    sub r8d, '0'
    imul rax, rax, 10
    add rax, r8
    inc rbx
    jmp pos_loop

pos_done:
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

invalid_line:
    mov r9, 1
    jmp line_finalize

do_add:
    cmp r11, 2
    jb invalid_line
    mov r8, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    add rax, r8
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_sub:
    cmp r11, 2
    jb invalid_line
    mov r8, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    sub rax, r8
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_mul:
    cmp r11, 2
    jb invalid_line
    mov r8, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    imul rax, r8
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_div:
    cmp r11, 2
    jb invalid_line
    mov r8, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    test r8, r8
    jz invalid_line
    cmp rax, -9223372036854775808
    jne div_not_min
    cmp r8, -1
    je invalid_line
div_not_min:
    cqo
    idiv r8
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_mod:
    cmp r11, 2
    jb invalid_line
    mov r8, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    test r8, r8
    jz invalid_line
    cmp rax, -9223372036854775808
    jne mod_not_min
    cmp r8, -1
    je invalid_line
mod_not_min:
    cqo
    idiv r8
    mov rax, rdx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

line_finalize:
    cmp r9, 0
    jne line_error
    cmp r11, 1
    jne line_error
    mov rax, qword ptr [r15]
    jmp print_value

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall
    jmp line_done

print_value:
    mov r8, rax
    mov r9, 0
    cmp r8, 0
    jge value_nonneg
    mov r9, 1
    cmp r8, -9223372036854775808
    jne value_neg_normal
    mov r8, 0x8000000000000000
    jmp value_digits
value_neg_normal:
    neg r8
    jmp value_digits
value_nonneg:
    ; r8 already positive
value_digits:
    lea r10, [rip + outbuf + 63]
    mov byte ptr [r10], 10
    dec r10
value_loop:
    mov rax, r8
    xor edx, edx
    mov r11, 10
    div r11
    mov r8, rax
    movzx eax, dl
    add eax, '0'
    dec r10
    mov byte ptr [r10], al
    cmp r8, 0
    jne value_loop

    cmp r9, 0
    je value_no_sign
    dec r10
    mov byte ptr [r10], '-'
value_no_sign:
    lea r11, [rip + outbuf + 64]
    sub r11, r10
    mov rdx, r11
    mov rsi, r10
    mov eax, 1
    mov edi, 1
    syscall

line_done:
    cmp r10, r14
    je final_line_end
    mov r12, r10
    inc r12
    jmp line_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok

.section .rodata
error_msg:
    .ascii "error\n"

.section .bss
    .align 8
input_buf:
    .space 65536
stack_mem:
    .space 16384
stack_top = stack_mem + 16384
outbuf:
    .space 64
