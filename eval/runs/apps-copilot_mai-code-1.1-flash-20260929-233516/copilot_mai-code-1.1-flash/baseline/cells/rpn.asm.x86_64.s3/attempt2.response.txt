.globl _start

.text
_start:
    lea r15, [rip + stack_mem + 16384]
    lea r12, [rip + input_buf]
    xor r13d, r13d

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r13
    mov edx, 65536
    sub edx, r13d
    cmp edx, 0
    jle done_read
    syscall
    cmp rax, 0
    je done_read
    add r13, rax
    cmp r13, 65536
    jb read_loop

done_read:
    cmp r13, 0
    je exit_ok
    lea r14, [rip + input_buf]
    add r14, r13
    mov r13, r14
    lea r12, [rip + input_buf]

line_loop:
    cmp r12, r13
    jae exit_ok
    mov rbx, r12

scan_line_end:
    cmp rbx, r13
    je end_of_file_line
    cmp byte ptr [rbx], 10
    je newline_line
    inc rbx
    jmp scan_line_end

newline_line:
    mov r14, rbx
    mov r10, 1
    jmp process_line

end_of_file_line:
    mov r14, r13
    xor r10d, r10d

process_line:
    lea r15, [rip + stack_mem + 16384]
    xor r11, r11
    xor r9d, r9d
    mov rbx, r12

parse_next:
    cmp rbx, r14
    jae line_finalize
skip_separators:
    cmp rbx, r14
    jae line_finalize
    mov al, byte ptr [rbx]
    cmp al, 0x20
    je sep_inc
    cmp al, 0x09
    je sep_inc
    jmp parse_token

sep_inc:
    inc rbx
    jmp skip_separators

parse_token:
    mov rsi, rbx

scan_token:
    cmp rbx, r14
    je token_done
    mov al, byte ptr [rbx]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc rbx
    jmp scan_token

token_done:
    mov rdi, rbx
    mov rcx, rdi
    sub rcx, rsi
    mov al, byte ptr [rsi]

    cmp al, '+'
    jne check_minus
    cmp rcx, 1
    jne invalid_token
    jmp do_add

check_minus:
    cmp al, '-'
    jne check_mul
    cmp rcx, 1
    je do_sub
    cmp rcx, 2
    jb invalid_token
    cmp rcx, 19
    ja invalid_token
    mov rbx, rsi
    inc rbx
    jmp check_negative_digits

check_mul:
    cmp al, '*'
    jne check_div
    cmp rcx, 1
    jne invalid_token
    jmp do_mul

check_div:
    cmp al, '/'
    jne check_mod
    cmp rcx, 1
    jne invalid_token
    jmp do_div

check_mod:
    cmp al, '%'
    jne check_digit_or_invalid
    cmp rcx, 1
    jne invalid_token
    jmp do_mod

check_digit_or_invalid:
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    cmp rcx, 18
    ja invalid_token
    xor rax, rax
    mov rbx, rsi
pos_digit_loop:
    cmp rbx, rdi
    je pos_digit_done
    movzx r8d, byte ptr [rbx]
    sub r8d, '0'
    imul rax, rax, 10
    add rax, r8
    inc rbx
    jmp pos_digit_loop

pos_digit_done:
    jmp push_value

check_negative_digits:
    cmp rbx, rdi
    je negative_digits_done
    mov dl, byte ptr [rbx]
    cmp dl, '0'
    jb invalid_token
    cmp dl, '9'
    ja invalid_token
    inc rbx
    jmp check_negative_digits

negative_digits_done:
    xor rax, rax
    mov rbx, rsi
    inc rbx

negative_parse_loop:
    cmp rbx, rdi
    je negative_parse_done
    movzx r8d, byte ptr [rbx]
    sub r8d, '0'
    imul rax, rax, 10
    add rax, r8
    inc rbx
    jmp negative_parse_loop

negative_parse_done:
    neg rax
    jmp push_value

push_value:
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

invalid_token:
    mov r9d, 1
    jmp line_finalize

do_add:
    cmp r11, 2
    jb invalid_token
    mov rbx, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    add rax, rbx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_sub:
    cmp r11, 2
    jb invalid_token
    mov rbx, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    sub rax, rbx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_mul:
    cmp r11, 2
    jb invalid_token
    mov rbx, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    imul rax, rbx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_div:
    cmp r11, 2
    jb invalid_token
    mov rbx, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    test rbx, rbx
    jz invalid_token
    cmp rax, -9223372036854775808
    jne div_not_min
    cmp rbx, -1
    je invalid_token
div_not_min:
    cqo
    idiv rbx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

do_mod:
    cmp r11, 2
    jb invalid_token
    mov rbx, qword ptr [r15]
    mov rax, qword ptr [r15 + 8]
    add r15, 16
    sub r11, 2
    test rbx, rbx
    jz invalid_token
    cmp rax, -9223372036854775808
    jne mod_not_min
    cmp rbx, -1
    je invalid_token
mod_not_min:
    cqo
    idiv rbx
    mov rax, rdx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11
    mov rbx, rdi
    jmp parse_next

line_finalize:
    cmp r9d, 0
    jne line_error
    cmp r11, 1
    jne line_error
    mov rax, qword ptr [r15]
    jmp print_value

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall
    jmp line_done

print_value:
    mov r8, rax
    xor r9d, r9d
    cmp r8, 0
    jl value_negative
    jmp value_digits

value_negative:
    mov r9d, 1
    cmp r8, -9223372036854775808
    jne value_abs
    mov r8, 0x8000000000000000
    jmp value_digits

value_abs:
    neg r8

value_digits:
    lea rdi, [rip + outbuf + 64]
    mov byte ptr [rdi - 1], 10
    mov rsi, rdi
    mov rax, r8

digit_loop:
    mov rax, r8
    xor edx, edx
    mov rcx, 10
    div rcx
    mov r8, rax
    movzx eax, dl
    add eax, '0'
    dec rdi
    mov byte ptr [rdi], al
    test r8, r8
    jne digit_loop

    cmp r9d, 0
    je no_sign
    dec rdi
    mov byte ptr [rdi], '-'

no_sign:
    mov rdx, rsi
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

line_done:
    cmp r10, 0
    je end_of_input_line
    mov r12, r14
    inc r12
    jmp line_loop

end_of_input_line:
    mov r12, r13
    jmp exit_ok

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok

.section .rodata
error_msg:
    .ascii "error\n"

.section .bss
    .align 8
input_buf:
    .space 65536
stack_mem:
    .space 16384
outbuf:
    .space 64
