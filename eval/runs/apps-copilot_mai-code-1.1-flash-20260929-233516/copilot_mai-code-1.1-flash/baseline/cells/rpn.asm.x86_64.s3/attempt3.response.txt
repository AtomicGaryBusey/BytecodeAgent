.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]

read_input:
    cmp r12d, 65536
    jae read_done
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_input

read_done:
    cmp r12d, 0
    je exit_ok
    mov r13, 0
    lea r15, [rip + eval_stack + 16384]

line_loop:
    cmp r13, r12
    jae exit_ok
    mov rbx, r13
    mov rcx, r13

scan_line:
    cmp rcx, r12
    jae no_newline
    lea rsi, [rip + input_buf]
    add rsi, rcx
    cmp byte ptr [rsi], 0x0A
    je found_newline
    inc rcx
    jmp scan_line

found_newline:
    mov rdi, rcx
    mov r10d, 1
    jmp process_line

no_newline:
    mov rdi, r12
    xor r10d, r10d

process_line:
    lea r15, [rip + eval_stack + 16384]
    xor r11d, r11d
    xor r9d, r9d
    mov r14, rbx

line_scan_loop:
    cmp r14, rdi
    jae line_finalize
skip_separators:
    cmp r14, rdi
    jae line_finalize
    movzx eax, byte ptr [r14]
    cmp al, 0x20
    je sep_inc
    cmp al, 0x09
    je sep_inc
    jmp parse_token

sep_inc:
    inc r14
    jmp skip_separators

parse_token:
    mov rsi, r14
scan_token:
    cmp r14, rdi
    jae token_done
    movzx eax, byte ptr [r14]
    cmp al, 0x20
    je token_done
    cmp al, 0x09
    je token_done
    inc r14
    jmp scan_token

token_done:
    mov r10, r14
    sub r10, rsi
    movzx eax, byte ptr [rsi]

    cmp al, '+'
    jne check_minus
    cmp r10, 1
    jne token_error
    jmp op_plus

check_minus:
    cmp al, '-'
    jne check_mul
    cmp r10, 1
    je op_minus
    cmp r10, 2
    jb token_error
    cmp r10, 19
    ja token_error
    mov r8, rsi
    inc r8
check_digits_after_minus:
    cmp r8, r14
    je parse_negative_literal
    movzx eax, byte ptr [r8]
    cmp al, '0'
    jb token_error
    cmp al, '9'
    ja token_error
    inc r8
    jmp check_digits_after_minus

parse_negative_literal:
    mov rax, 0
    mov r8, rsi
    inc r8
negative_literal_loop:
    cmp r8, r14
    je negative_literal_done
    movzx ecx, byte ptr [r8]
    sub ecx, '0'
    imul rax, rax, 10
    add rax, rcx
    inc r8
    jmp negative_literal_loop

negative_literal_done:
    neg rax
    jmp push_literal

check_mul:
    cmp al, '*'
    jne check_div
    cmp r10, 1
    jne token_error
    jmp op_mul

check_div:
    cmp al, '/'
    jne check_mod
    cmp r10, 1
    jne token_error
    jmp op_div

check_mod:
    cmp al, '%'
    jne check_number
    cmp r10, 1
    jne token_error
    jmp op_mod

check_number:
    cmp al, '0'
    jb token_error
    cmp al, '9'
    ja token_error
    cmp r10, 19
    ja token_error
    mov rax, 0
    mov r8, rsi
positive_literal_loop:
    cmp r8, r14
    je push_literal
    movzx ecx, byte ptr [r8]
    cmp ecx, '0'
    jb token_error
    cmp ecx, '9'
    ja token_error
    sub ecx, '0'
    imul rax, rax, 10
    add rax, rcx
    inc r8
    jmp positive_literal_loop

push_literal:
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11d
    jmp line_scan_loop

token_error:
    mov r9d, 1
    jmp line_finalize

op_plus:
    cmp r11d, 2
    jb token_error
    mov rax, qword ptr [r15]
    mov r8, qword ptr [r15 + 8]
    add r15, 16
    sub r11d, 2
    add r8, rax
    sub r15, 8
    mov qword ptr [r15], r8
    inc r11d
    jmp line_scan_loop

op_minus:
    cmp r11d, 2
    jb token_error
    mov rax, qword ptr [r15]
    mov r8, qword ptr [r15 + 8]
    add r15, 16
    sub r11d, 2
    sub r8, rax
    sub r15, 8
    mov qword ptr [r15], r8
    inc r11d
    jmp line_scan_loop

op_mul:
    cmp r11d, 2
    jb token_error
    mov rax, qword ptr [r15]
    mov r8, qword ptr [r15 + 8]
    add r15, 16
    sub r11d, 2
    imul r8, rax
    sub r15, 8
    mov qword ptr [r15], r8
    inc r11d
    jmp line_scan_loop

op_div:
    cmp r11d, 2
    jb token_error
    mov rax, qword ptr [r15]
    mov r8, qword ptr [r15 + 8]
    add r15, 16
    sub r11d, 2
    test rax, rax
    jz token_error
    mov r10, -9223372036854775808
    cmp r8, r10
    jne div_ok
    mov r10, -1
    cmp rax, r10
    je token_error
div_ok:
    mov r10, rax
    mov rax, r8
    cqo
    mov rbx, r10
    idiv rbx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11d
    jmp line_scan_loop

op_mod:
    cmp r11d, 2
    jb token_error
    mov rax, qword ptr [r15]
    mov r8, qword ptr [r15 + 8]
    add r15, 16
    sub r11d, 2
    test rax, rax
    jz token_error
    mov r10, -9223372036854775808
    cmp r8, r10
    jne mod_ok
    mov r10, -1
    cmp rax, r10
    je token_error
mod_ok:
    mov r10, rax
    mov rax, r8
    cqo
    mov rbx, r10
    idiv rbx
    mov rax, rdx
    sub r15, 8
    mov qword ptr [r15], rax
    inc r11d
    jmp line_scan_loop

line_finalize:
    cmp r9d, 0
    jne line_error
    cmp r11d, 1
    jne line_error
    mov rax, qword ptr [r15]
    jmp print_value

line_error:
    lea rsi, [rip + error_msg]
    mov edx, 6
    mov edi, 1
    mov eax, 1
    syscall
    jmp after_line

print_value:
    mov r11, rax
    mov r9d, 0
    cmp r11, 0
    jge value_nonneg
    mov r9d, 1
    mov r10, -9223372036854775808
    cmp r11, r10
    jne value_abs
    mov r11, 0x8000000000000000
    jmp value_digits_init
value_abs:
    neg r11
value_digits_init:
value_nonneg:
    lea rsi, [rip + tmp_digits]
    xor rcx, rcx
value_digits_loop:
    mov rax, r11
    xor rdx, rdx
    mov r8, 10
    div r8
    mov r11, rax
    mov rax, rdx
    add rax, '0'
    mov byte ptr [rsi + rcx], al
    inc rcx
    test r11, r11
    jne value_digits_loop
    cmp r9d, 0
    je no_sign
    mov al, '-'
    mov byte ptr [rsi + rcx], al
    inc rcx
no_sign:
    lea rdi, [rip + outbuf]
    xor r8d, r8d
reverse_loop:
    cmp r8d, ecx
    jae reverse_done
    mov r10d, ecx
    sub r10d, r8d
    dec r10d
    mov al, byte ptr [rsi + r10]
    mov byte ptr [rdi + r8], al
    inc r8d
    jmp reverse_loop
reverse_done:
    mov byte ptr [rdi + rcx], 0x0A
    mov rsi, rdi
    mov rdx, rcx
    inc rdx
    mov eax, 1
    mov edi, 1
    syscall

after_line:
    cmp r10d, 0
    je exit_ok
    mov r13, rdi
    inc r13
    jmp line_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok

.section .rodata
error_msg:
    .asciz "error\n"

.section .bss
    .align 8
input_buf:
    .space 65536
eval_stack:
    .space 16384
tmp_digits:
    .space 64
outbuf:
    .space 128
