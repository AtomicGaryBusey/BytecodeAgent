.globl _start

.text
_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13
process_next_line:
    cmp r13, r12
    jae exit_ok
    mov r14, r13
find_newline:
    cmp r14, r12
    jae final_line
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r14]
    cmp cl, 0x0A
    je newline_found
    inc r14
    jmp find_newline

newline_found:
    mov r11, 1
    jmp eval_line

final_line:
    mov r14, r12
    cmp r13, r14
    je exit_ok
    xor r11, r11

eval_line:
    xor r8, r8
    xor r9, r9
    lea rbp, [rip + eval_stack]
    mov r10, r13
    jmp line_scan

line_scan:
    cmp r10, r14
    jae line_finalize
skip_sep:
    cmp r10, r14
    jae line_finalize
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r10]
    cmp cl, 0x20
    je sep_inc
    cmp cl, 0x09
    je sep_inc
    jmp parse_token

sep_inc:
    inc r10
    jmp skip_sep

parse_token:
    mov rsi, r10
scan_token:
    cmp r10, r14
    jae token_done
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r10]
    cmp cl, 0x20
    je token_done
    cmp cl, 0x09
    je token_done
    inc r10
    jmp scan_token

token_done:
    mov rbx, r10
    sub rbx, rsi
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + rsi]

    cmp rbx, 1
    jne check_literal_start
    cmp cl, '+'
    je op_plus
    cmp cl, '-'
    je op_minus
    cmp cl, '*'
    je op_mul
    cmp cl, '/'
    je op_div
    cmp cl, '%'
    je op_mod

check_literal_start:
    cmp cl, '-'
    jne check_positive
    cmp rbx, 19
    ja token_error
    mov rdi, rsi
    inc rdi
    xor rax, rax
neg_digits:
    cmp rdi, r10
    jae neg_done
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rdi]
    cmp cl, '0'
    jb token_error
    cmp cl, '9'
    ja token_error
    sub cl, '0'
    movzx ecx, cl
    imul rax, rax, 10
    add rax, rcx
    inc rdi
    jmp neg_digits

neg_done:
    neg rax
    jmp push_literal

check_positive:
    cmp cl, '0'
    jb token_error
    cmp cl, '9'
    ja token_error
    cmp rbx, 18
    ja token_error
    mov rdi, rsi
    xor rax, rax
pos_digits:
    cmp rdi, r10
    jae pos_done
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rdi]
    cmp cl, '0'
    jb token_error
    cmp cl, '9'
    ja token_error
    sub cl, '0'
    movzx ecx, cl
    imul rax, rax, 10
    add rax, rcx
    inc rdi
    jmp pos_digits

pos_done:
    jmp push_literal

push_literal:
    mov qword ptr [rbp + r9*8], rax
    inc r9
    jmp line_scan

token_error:
    mov r8, 1
    jmp line_finalize

op_plus:
    cmp r9, 2
    jb token_error
    lea rdx, [rbp + r9*8 - 8]
    mov rsi, qword ptr [rdx]
    lea rdx, [rbp + r9*8 - 16]
    mov rdi, qword ptr [rdx]
    add rdi, rsi
    mov qword ptr [rbp + r9*8 - 16], rdi
    dec r9
    jmp line_scan

op_minus:
    cmp r9, 2
    jb token_error
    lea rdx, [rbp + r9*8 - 8]
    mov rsi, qword ptr [rdx]
    lea rdx, [rbp + r9*8 - 16]
    mov rdi, qword ptr [rdx]
    sub rdi, rsi
    mov qword ptr [rbp + r9*8 - 16], rdi
    dec r9
    jmp line_scan

op_mul:
    cmp r9, 2
    jb token_error
    lea rdx, [rbp + r9*8 - 8]
    mov rsi, qword ptr [rdx]
    lea rdx, [rbp + r9*8 - 16]
    mov rdi, qword ptr [rdx]
    imul rdi, rsi
    mov qword ptr [rbp + r9*8 - 16], rdi
    dec r9
    jmp line_scan

op_div:
    cmp r9, 2
    jb token_error
    lea rdx, [rbp + r9*8 - 8]
    mov rsi, qword ptr [rdx]
    lea rdx, [rbp + r9*8 - 16]
    mov rdi, qword ptr [rdx]
    test rsi, rsi
    jz token_error
    mov rax, -9223372036854775808
    cmp rdi, rax
    jne div_ok
    cmp rsi, -1
    je token_error
div_ok:
    mov rax, rdi
    cqo
    mov rbx, rsi
    idiv rbx
    mov qword ptr [rbp + r9*8 - 16], rax
    dec r9
    jmp line_scan

op_mod:
    cmp r9, 2
    jb token_error
    lea rdx, [rbp + r9*8 - 8]
    mov rsi, qword ptr [rdx]
    lea rdx, [rbp + r9*8 - 16]
    mov rdi, qword ptr [rdx]
    test rsi, rsi
    jz token_error
    mov rax, -9223372036854775808
    cmp rdi, rax
    jne mod_ok
    cmp rsi, -1
    je token_error
mod_ok:
    mov rax, rdi
    cqo
    mov rbx, rsi
    idiv rbx
    mov rax, rdx
    mov qword ptr [rbp + r9*8 - 16], rax
    dec r9
    jmp line_scan

line_finalize:
    test r8, r8
    jnz line_error
    cmp r9, 1
    jne line_error
    mov rax, qword ptr [rbp]
    jmp print_value

line_error:
    lea rsi, [rip + error_msg]
    mov edx, 6
    mov edi, 1
    mov eax, 1
    syscall
    jmp after_line

print_value:
    lea rsi, [rip + tmp_digits]
    xor r8, r8
    xor r10, r10
    test rax, rax
    jns positive_value
    mov r10, 1
    cmp rax, -9223372036854775808
    jne value_abs
    mov r11, 0x8000000000000000
    jmp digits_loop
value_abs:
    neg rax
    mov r11, rax
    jmp digits_loop

positive_value:
    mov r11, rax

digits_loop:
    mov rax, r11
    xor rdx, rdx
    mov rcx, 10
    div rcx
    mov r11, rax
    mov rax, rdx
    add rax, '0'
    mov byte ptr [rsi + r8], al
    inc r8
    test r11, r11
    jne digits_loop

    test r10, r10
    jz no_sign
    mov al, '-'
    mov byte ptr [rsi + r8], al
    inc r8

no_sign:
    lea rdi, [rip + outbuf]
    xor rcx, rcx
reverse_loop:
    cmp rcx, r8
    jae reverse_done
    mov rdx, r8
    sub rdx, rcx
    dec rdx
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi + rcx], al
    inc rcx
    jmp reverse_loop

reverse_done:
    mov byte ptr [rdi + r8], 0x0A
    mov edx, r8d
    inc edx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall

after_line:
    test r11, r11
    jnz next_line_after
    jmp exit_ok

next_line_after:
    mov r13, r14
    inc r13
    jmp process_next_line

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok

.section .rodata
error_msg:
    .ascii "error\n"

.section .bss
    .align 8
input_buf:
    .space 65536
eval_stack:
    .space 8192
tmp_digits:
    .space 64
outbuf:
    .space 128
