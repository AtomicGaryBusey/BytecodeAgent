.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ SYS_exit_group, 94

.equ STDOUT, 1
.equ INPUT_MAX, 65536
.equ READ_SIZE, 4096
.equ LINEBUF_SIZE, 128

_start:
    bl read_all
    bl dump_rows
    bl print_total
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

read_all:
    adrp x10, input_buf
    add x10, x10, :lo12:input_buf
    adrp x11, total_bytes
    add x11, x11, :lo12:total_bytes
    mov x12, #0
    str x12, [x11]
1:
    ldr x13, [x11]
    cmp x13, #INPUT_MAX
    b.hs 9f
    mov x2, #READ_SIZE
    mov x3, #INPUT_MAX
    sub x3, x3, x13
    cmp x2, x3
    b.le 2f
    mov x2, x3
2:
    mov x0, #0
    mov x8, #SYS_read
    add x1, x10, x13
    svc #0
    cmp x0, #0
    b.lt 100f
    cbz x0, 9f
    add x13, x13, x0
    str x13, [x11]
    cmp x13, #INPUT_MAX
    b.hs 9f
    b 1b
9:
    ret
100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

dump_rows:
    adrp x10, total_bytes
    add x10, x10, :lo12:total_bytes
    ldr x19, [x10]
    cbz x19, 9f
    mov x20, #0
1:
    mov x0, x20
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x20
    mov x2, x19
    sub x2, x2, x20
    mov x3, #16
    cmp x2, x3
    b.le 2f
    mov x2, x3
2:
    bl print_row
    add x20, x20, #16
    cmp x20, x19
    b.lt 1b
9:
    ret

print_row:
    adrp x10, line_buf
    add x10, x10, :lo12:line_buf
    mov x11, x0
    mov x12, x10
    add x13, x10, #8
    mov x14, #8
1:
    and x15, x11, #0xF
    cmp x15, #10
    b.lo 2f
    sub x15, x15, #10
    add x15, x15, #'a'
    b 3f
2:
    add x15, x15, #'0'
3:
    sub x13, x13, #1
    strb w15, [x13]
    lsr x11, x11, #4
    subs x14, x14, #1
    b.ne 1b

    mov x15, x10
    add x15, x15, #8
    mov w14, #' '
    strb w14, [x15], #1
    strb w14, [x15], #1

    mov x16, #0
    mov x17, x1
    mov x18, x2
    mov w19, #' '
4:
    cmp x16, #16
    b.ge 5f
    cmp x16, x18
    b.ge 6f
    ldrb w14, [x17, x16]
    ubfx x20, x14, #4, #4
    cmp x20, #10
    b.lo 7f
    sub x20, x20, #10
    add x20, x20, #'a'
    b 8f
7:
    add x20, x20, #'0'
8:
    and x21, x14, #0xF
    cmp x21, #10
    b.lo 9f
    sub x21, x21, #10
    add x21, x21, #'a'
    b 10f
9:
    add x21, x21, #'0'
10:
    strb w20, [x15], #1
    strb w21, [x15], #1
    strb w19, [x15], #1
    cmp x16, #7
    b.ne 11f
    strb w19, [x15], #1
11:
    add x16, x16, #1
    b 4b

6:
    mov w14, #' '
    strb w14, [x15], #1
    mov w14, #'|'
    strb w14, [x15], #1

    mov x16, #0
12:
    cmp x16, x18
    b.ge 13f
    ldrb w14, [x17, x16]
    cmp w14, #0x20
    b.lt 14f
    cmp w14, #0x7E
    b.gt 14f
    b 15f
14:
    mov w14, #'.'
15:
    strb w14, [x15], #1
    add x16, x16, #1
    b 12b
13:
    mov w14, #'|'
    strb w14, [x15], #1
    mov w14, #10
    strb w14, [x15], #1

    mov x0, #STDOUT
    mov x1, x10
    sub x2, x15, x10
    bl write_all
    ret

5:
    mov w14, #' '
    strb w14, [x15], #1
    mov w14, #'|'
    strb w14, [x15], #1
    mov x16, #0
16:
    cmp x16, x18
    b.ge 17f
    ldrb w14, [x17, x16]
    cmp w14, #0x20
    b.lt 18f
    cmp w14, #0x7E
    b.gt 18f
    b 19f
18:
    mov w14, #'.'
19:
    strb w14, [x15], #1
    add x16, x16, #1
    b 16b
17:
    mov w14, #'|'
    strb w14, [x15], #1
    mov w14, #10
    strb w14, [x15], #1

    mov x0, #STDOUT
    mov x1, x10
    sub x2, x15, x10
    bl write_all
    ret

write_all:
    mov x19, x0
    mov x20, x1
    mov x21, x2
1:
    cbz x21, 9f
    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.lt 100f
    sub x21, x21, x0
    add x20, x20, x0
    b 1b
9:
    ret
100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_total:
    adrp x10, total_bytes
    add x10, x10, :lo12:total_bytes
    ldr x11, [x10]
    adrp x12, line_buf
    add x12, x12, :lo12:line_buf
    mov x13, x11
    mov x14, x12
    add x15, x12, #8
    mov x16, #8
1:
    and x17, x13, #0xF
    cmp x17, #10
    b.lo 2f
    sub x17, x17, #10
    add x17, x17, #'a'
    b 3f
2:
    add x17, x17, #'0'
3:
    sub x15, x15, #1
    strb w17, [x15]
    lsr x13, x13, #4
    subs x16, x16, #1
    b.ne 1b
    mov w17, #10
    strb w17, [x12, #8]
    mov x0, #STDOUT
    mov x1, x12
    mov x2, #9
    bl write_all
    ret

.section .bss
.align 4
input_buf:
    .zero INPUT_MAX
total_bytes:
    .zero 8
line_buf:
    .zero LINEBUF_SIZE
