.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ INPUT_MAX, 65536
.equ READ_CHUNK, 4096
.equ ROW_BYTES, 16

_start:
    bl read_all
    bl print_rows
    bl print_total
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

read_all:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
1:
    cmp x10, #INPUT_MAX
    b.hs 9f

    mov x0, #STDIN
    mov x1, x9
    add x1, x1, x10
    mov x2, #READ_CHUNK
    mov x8, #SYS_read
    svc #0

    cmp x0, #0
    b.lt 100f
    cbz x0, 9f

    add x10, x10, x0
    b 1b

9:
    adrp x11, total_len
    add x11, x11, :lo12:total_len
    str x10, [x11]
    ret

100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_rows:
    adrp x9, total_len
    add x9, x9, :lo12:total_len
    ldr x19, [x9]
    cbz x19, 9f

    mov x20, #0
1:
    cmp x20, x19
    b.ge 9f

    mov x0, x20
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x20

    sub x2, x19, x20
    cmp x2, #ROW_BYTES
    b.le 2f
    mov x2, #ROW_BYTES
2:
    str x30, [sp, #-16]!
    bl print_row
    ldr x30, [sp], #16

    add x20, x20, #ROW_BYTES
    b 1b

9:
    ret

print_row:
    adrp x9, line_buf
    add x9, x9, :lo12:line_buf

    mov x15, x9
    add x15, x15, #8
    mov x16, x15
    mov x17, x0
    mov x18, #8

offset_digits:
    and w19, w17, #0xF
    cmp w19, #10
    b.lo offset_digit
    sub w19, w19, #10
    add w19, w19, #'a'
    b offset_store

offset_digit:
    add w19, w19, #'0'

offset_store:
    sub x16, x16, #1
    strb w19, [x16]
    lsr x17, x17, #4
    subs x18, x18, #1
    b.ne offset_digits

    add x15, x16, #8
    mov w19, #' '
    strb w19, [x15], #1
    strb w19, [x15], #1

    mov x20, x1
    mov x21, x2
    mov x22, #0

byte_loop:
    cmp x22, #ROW_BYTES
    b.ge bytes_done
    cmp x22, x21
    b.ge pad_loop

    ldrb w23, [x20, x22]

    ubfx w24, w23, #4, #4
    cmp w24, #10
    b.lo active_hi0
    sub w24, w24, #10
    add w24, w24, #'a'
    b active_hi_store

active_hi0:
    add w24, w24, #'0'

active_hi_store:
    strb w24, [x15], #1

    and w24, w23, #0xF
    cmp w24, #10
    b.lo active_lo0
    sub w24, w24, #10
    add w24, w24, #'a'
    b active_lo_store

active_lo0:
    add w24, w24, #'0'

active_lo_store:
    strb w24, [x15], #1

    mov w24, #' '
    strb w24, [x15], #1
    cmp x22, #7
    b.ne active_next
    strb w24, [x15], #1

active_next:
    add x22, x22, #1
    b byte_loop

pad_loop:
    mov w24, #' '
    strb w24, [x15], #1
    strb w24, [x15], #1
    strb w24, [x15], #1
    cmp x22, #7
    b.ne pad_next
    strb w24, [x15], #1

pad_next:
    add x22, x22, #1
    b byte_loop

bytes_done:
    mov w24, #' '
    strb w24, [x15], #1
    mov w24, #'|'
    strb w24, [x15], #1

    mov x22, #0

char_loop:
    cmp x22, x21
    b.ge char_done

    ldrb w23, [x20, x22]
    cmp w23, #0x20
    b.lt char_dot
    cmp w23, #0x7E
    b.gt char_dot
    b char_store

char_dot:
    mov w23, #'.'

char_store:
    strb w23, [x15], #1
    add x22, x22, #1
    b char_loop

char_done:
    mov w23, #'|'
    strb w23, [x15], #1
    mov w23, #10
    strb w23, [x15], #1

    sub x22, x15, x9
    mov x0, #STDOUT
    mov x1, x9
    mov x2, x22

    str x30, [sp, #-16]!
    bl write_all
    ldr x30, [sp], #16
    ret

write_all:
    mov x19, x0
    mov x20, x1
    mov x21, x2

write_loop:
    cbz x21, write_done

    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_write
    svc #0

    cmp x0, #0
    b.le write_err

    sub x21, x21, x0
    add x20, x20, x0
    b write_loop

write_done:
    ret

write_err:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_total:
    adrp x9, total_len
    add x9, x9, :lo12:total_len
    ldr x10, [x9]

    adrp x11, line_buf
    add x11, x11, :lo12:line_buf
    add x12, x11, #8
    mov x13, #8

total_loop:
    and w14, w10, #0xF
    cmp w14, #10
    b.lo total_digit
    sub w14, w14, #10
    add w14, w14, #'a'
    b total_store

total_digit:
    add w14, w14, #'0'

total_store:
    sub x12, x12, #1
    strb w14, [x12]
    lsr x10, x10, #4
    subs x13, x13, #1
    b.ne total_loop

    mov w14, #10
    strb w14, [x11, #8]

    mov x0, #STDOUT
    mov x1, x11
    mov x2, #9

    str x30, [sp, #-16]!
    bl write_all
    ldr x30, [sp], #16
    ret

.section .bss
.align 4
input_buf:
    .zero 65536

total_len:
    .zero 8

line_buf:
    .zero 128
