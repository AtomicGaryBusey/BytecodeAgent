.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDOUT, 1

.equ INPUT_MAX, 65536
.equ READ_CHUNK, 4096
.equ ROW_BYTES, 16

_start:
    bl read_all
    bl dump_rows
    bl print_total
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

read_all:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
1:
    cmp x10, #INPUT_MAX
    b.hs 9f

    mov x2, #READ_CHUNK
    mov x3, #INPUT_MAX
    sub x3, x3, x10
    cmp x2, x3
    b.le 2f
    mov x2, x3
2:
    mov x0, #0
    mov x1, x9
    add x1, x1, x10
    mov x8, #SYS_read
    svc #0

    cmp x0, #0
    b.lt 100f
    cbz x0, 9f

    add x10, x10, x0
    b 1b

9:
    adrp x11, total_bytes
    add x11, x11, :lo12:total_bytes
    str x10, [x11]
    ret

100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

dump_rows:
    adrp x9, total_bytes
    add x9, x9, :lo12:total_bytes
    ldr x19, [x9]
    cbz x19, 9f

    mov x20, #0
1:
    cmp x20, x19
    b.ge 9f

    mov x0, x20
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x20

    mov x2, x19
    sub x2, x2, x20
    cmp x2, #ROW_BYTES
    b.le 2f
    mov x2, #ROW_BYTES
2:
    bl print_row
    add x20, x20, #ROW_BYTES
    b 1b

9:
    ret

print_row:
    adrp x9, line_buf
    add x9, x9, :lo12:line_buf

    mov x10, x9
    add x10, x10, #8
    mov x11, x0
    mov x12, #8
1:
    and x13, x11, #0xF
    cmp x13, #10
    b.lo 2f
    sub x13, x13, #10
    add x13, x13, #'a'
    b 3f
2:
    add x13, x13, #'0'
3:
    sub x10, x10, #1
    strb w13, [x10]
    lsr x11, x11, #4
    subs x12, x12, #1
    b.ne 1b

    add x13, x9, #8
    mov w14, #' '
    strb w14, [x13], #1
    strb w14, [x13], #1

    mov x15, #0
    mov x16, x1
    mov x17, x2
4:
    cmp x15, #16
    b.ge 5f
    cmp x15, x17
    b.ge 6f

    ldrb w18, [x16, x15]

    ubfx w19, w18, #4, #4
    cmp w19, #10
    b.lo 7f
    sub w19, w19, #10
    add w19, w19, #'a'
    b 8f
7:
    add w19, w19, #'0'
8:
    strb w19, [x13], #1

    and w20, w18, #0xF
    cmp w20, #10
    b.lo 9f
    sub w20, w20, #10
    add w20, w20, #'a'
    b 10f
9:
    add w20, w20, #'0'
10:
    strb w20, [x13], #1

    mov w21, #' '
    strb w21, [x13], #1

    cmp x15, #7
    b.ne 11f
    strb w21, [x13], #1
11:
    add x15, x15, #1
    b 4b

6:
    mov w21, #' '
    strb w21, [x13], #1
    strb w21, [x13], #1
    strb w21, [x13], #1
    cmp x15, #7
    b.ne 12f
    strb w21, [x13], #1
12:
    add x15, x15, #1
    b 4b

5:
    mov w21, #' '
    strb w21, [x13], #1
    mov w21, #'|'
    strb w21, [x13], #1

    mov x15, #0
    mov x16, x1
    mov x17, x2
13:
    cmp x15, x17
    b.ge 14f

    ldrb w18, [x16, x15]
    cmp w18, #0x20
    b.lt 15f
    cmp w18, #0x7E
    b.gt 15f
    b 16f
15:
    mov w18, #'.'
16:
    strb w18, [x13], #1
    add x15, x15, #1
    b 13b

14:
    mov w18, #'|'
    strb w18, [x13], #1
    mov w18, #10
    strb w18, [x13], #1

    mov x0, #STDOUT
    mov x1, x9
    mov x2, x13
    sub x2, x2, x9
    bl write_all
    ret

write_all:
    mov x19, x0
    mov x20, x1
    mov x21, x2
1:
    cbz x21, 9f

    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_write
    svc #0

    cmp x0, #0
    b.lt 100f
    cbz x0, 100f

    sub x21, x21, x0
    add x20, x20, x0
    b 1b

9:
    ret

100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_total:
    adrp x9, total_bytes
    add x9, x9, :lo12:total_bytes
    ldr x10, [x9]

    adrp x11, line_buf
    add x11, x11, :lo12:line_buf
    add x12, x11, #8
    mov x13, #8
1:
    and x14, x10, #0xF
    cmp x14, #10
    b.lo 2f
    sub x14, x14, #10
    add x14, x14, #'a'
    b 3f
2:
    add x14, x14, #'0'
3:
    sub x12, x12, #1
    strb w14, [x12]
    lsr x10, x10, #4
    subs x13, x13, #1
    b.ne 1b

    mov w14, #10
    strb w14, [x11, #8]

    mov x0, #STDOUT
    mov x1, x11
    mov x2, #9
    bl write_all
    ret

.section .bss
.align 4
input_buf:
    .zero 65536
total_bytes:
    .zero 8
line_buf:
    .zero 128
