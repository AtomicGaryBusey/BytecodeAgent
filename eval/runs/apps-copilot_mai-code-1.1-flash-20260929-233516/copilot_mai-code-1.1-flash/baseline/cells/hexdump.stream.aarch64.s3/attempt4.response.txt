.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ INPUT_MAX, 65536
.equ READ_CHUNK, 4096
.equ ROW_BYTES, 16

_start:
    bl read_all
    bl dump_all
    bl print_total
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

read_all:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, #0
1:
    cmp x10, #INPUT_MAX
    b.hs 9f

    mov x0, #STDIN
    mov x1, x9
    add x1, x1, x10
    mov x2, #READ_CHUNK
    mov x11, #INPUT_MAX
    sub x11, x11, x10
    cmp x11, x2
    b.hs 2f
    mov x2, x11
2:
    mov x8, #SYS_read
    svc #0

    cmp x0, #0
    b.lt 100f
    cbz x0, 9f

    add x10, x10, x0
    b 1b

9:
    adrp x11, total_len
    add x11, x11, :lo12:total_len
    str x10, [x11]
    ret

100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

dump_all:
    adrp x9, total_len
    add x9, x9, :lo12:total_len
    ldr x19, [x9]
    cbz x19, 9f
    mov x20, #0
1:
    cmp x20, x19
    b.ge 9f

    mov x0, x20
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x20
    mov x2, x19
    sub x2, x2, x20
    cmp x2, #ROW_BYTES
    b.le 2f
    mov x2, #ROW_BYTES
2:
    str x30, [sp, #-16]!
    bl print_row
    ldr x30, [sp], #16
    add x20, x20, #ROW_BYTES
    b 1b

9:
    ret

print_row:
    adrp x9, line_buf
    add x9, x9, :lo12:line_buf
    mov x15, x9
    mov x10, x0
    mov x11, #8
    mov x12, x9
    add x12, x12, #8

offset_digits:
    and w14, w10, #0xF
    cmp w14, #10
    b.lo 1f
    sub w14, w14, #10
    add w14, w14, #0x61
    b 2f
1:
    add w14, w14, #0x30
2:
    sub x12, x12, #1
    strb w14, [x12]
    lsr x10, x10, #4
    subs x11, x11, #1
    b.ne offset_digits

    mov x13, #8
3:
    ldrb w14, [x12], #1
    strb w14, [x15], #1
    subs x13, x13, #1
    b.ne 3b

    mov w14, #0x20
    strb w14, [x15], #1
    strb w14, [x15], #1

    mov x20, x1
    mov x21, x2
    mov x22, #0

hex_loop:
    cmp x22, #ROW_BYTES
    b.ge hex_done
    cmp x22, x21
    b.ge hex_pad

    ldrb w23, [x20, x22]
    ubfx w24, w23, #4, #4
    cmp w24, #10
    b.lo 4f
    sub w24, w24, #10
    add w24, w24, #0x61
    b 5f
4:
    add w24, w24, #0x30
5:
    strb w24, [x15], #1

    and w24, w23, #0xF
    cmp w24, #10
    b.lo 6f
    sub w24, w24, #10
    add w24, w24, #0x61
    b 7f
6:
    add w24, w24, #0x30
7:
    strb w24, [x15], #1

    mov w24, #0x20
    strb w24, [x15], #1
    cmp x22, #7
    b.ne 8f
    strb w24, [x15], #1
8:
    add x22, x22, #1
    b hex_loop

hex_pad:
    mov w24, #0x20
    strb w24, [x15], #1
    strb w24, [x15], #1
    strb w24, [x15], #1
    cmp x22, #7
    b.ne 9f
    strb w24, [x15], #1
9:
    add x22, x22, #1
    b hex_loop

hex_done:
    mov w24, #0x20
    strb w24, [x15], #1
    mov w24, #0x7C
    strb w24, [x15], #1

    mov x22, #0
char_loop:
    cmp x22, x21
    b.ge char_done

    ldrb w23, [x20, x22]
    cmp w23, #0x20
    b.lt 10f
    cmp w23, #0x7E
    b.gt 10f
    b 11f
10:
    mov w23, #0x2E
11:
    strb w23, [x15], #1
    add x22, x22, #1
    b char_loop

char_done:
    mov w24, #0x7C
    strb w24, [x15], #1
    mov w24, #0x0A
    strb w24, [x15], #1

    sub x22, x15, x9
    mov x0, #STDOUT
    mov x1, x9
    mov x2, x22
    str x30, [sp, #-16]!
    bl write_all
    ldr x30, [sp], #16
    ret

write_all:
    mov x19, x0
    mov x20, x1
    mov x21, x2

1:
    cbz x21, 9f
    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_write
    svc #0

    cmp x0, #0
    b.lt 100f
    cbz x0, 100f

    sub x21, x21, x0
    add x20, x20, x0
    b 1b

9:
    ret

100:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_total:
    adrp x9, total_len
    add x9, x9, :lo12:total_len
    ldr x10, [x9]

    adrp x11, line_buf
    add x11, x11, :lo12:line_buf
    mov x12, x11
    add x12, x12, #8
    mov x13, #8

1:
    and w14, w10, #0xF
    cmp w14, #10
    b.lo 2f
    sub w14, w14, #10
    add w14, w14, #0x61
    b 3f
2:
    add w14, w14, #0x30
3:
    sub x12, x12, #1
    strb w14, [x12]
    lsr x10, x10, #4
    subs x13, x13, #1
    b.ne 1b

    mov w14, #0x0A
    strb w14, [x11, #8]

    mov x0, #STDOUT
    mov x1, x11
    mov x2, #9
    str x30, [sp, #-16]!
    bl write_all
    ldr x30, [sp], #16
    ret

.section .bss
.align 4
input_buf:
    .zero 65536

total_len:
    .zero 8

line_buf:
    .zero 128
