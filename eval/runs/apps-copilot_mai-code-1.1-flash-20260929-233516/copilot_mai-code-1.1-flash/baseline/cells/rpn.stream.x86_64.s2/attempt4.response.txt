.text
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_SIZE, 65536
.equ STACK_SIZE, 32768

.section .bss
.align 8
input_buf:
    .space INPUT_SIZE

.align 8
stack_values:
    .space STACK_SIZE

.align 8
output_buf:
    .space 32

.section .rodata
error_msg:
    .ascii "error\n"

newline:
    .byte 0x0A

min_int_str:
    .ascii "-9223372036854775808"

.text

_start:
    xor ebp, ebp
    lea r15, [rip + input_buf]

read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbp
    mov rdx, INPUT_SIZE
    sub rdx, rbp
    syscall
    cmp rax, 0
    jl .exit_fail
    je .read_done
    add rbp, rax
    cmp rbp, INPUT_SIZE
    jb read_loop

.read_done:
    cmp rbp, 0
    je .exit_ok
    xor r12d, r12d

process_lines:
    cmp r12, rbp
    jae .exit_ok
    mov r13, r12

scan_line:
    cmp r13, rbp
    jae .line_ready
    movzx eax, byte ptr [r15 + r13]
    cmp al, 0x0A
    je .line_ready
    inc r13
    jmp scan_line

.line_ready:
    xor r14d, r14d
    xor r9d, r9d
    mov rbx, r12

parse_loop:
    cmp rbx, r13
    jae .line_end

skip_space:
    cmp rbx, r13
    jae .line_end
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 0x20
    je .skip_inc
    cmp al, 0x09
    je .skip_inc
    jmp .token_start

.skip_inc:
    inc rbx
    jmp skip_space

.token_start:
    mov rsi, rbx
    mov rcx, rbx

scan_token:
    cmp rcx, r13
    jae .token_done
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 0x20
    je .token_done
    cmp al, 0x09
    je .token_done
    inc rcx
    jmp scan_token

.token_done:
    mov rdx, rcx
    sub rdx, rsi
    movzx eax, byte ptr [r15 + rsi]

    cmp rdx, 1
    jne .not_single

.single_char:
    cmp al, '+'
    je .op_plus
    cmp al, '-'
    je .op_minus
    cmp al, '*'
    je .op_mul
    cmp al, '/'
    je .op_div
    cmp al, '%'
    je .op_mod
    jmp .number_process

.not_single:
    cmp al, '-'
    jne .number_process
    cmp rdx, 19
    ja .line_error
    mov r8, rcx
    mov r10, rsi
    inc r10
    xor r11d, r11d
    mov r11d, 1
    xor eax, eax
    jmp .parse_digits

.number_process:
    cmp rdx, 19
    ja .line_error
    mov r8, rcx
    mov r10, rsi
    xor r11d, r11d
    xor eax, eax

.parse_digits:
    cmp r10, r8
    je .digits_done
    movzx ecx, byte ptr [r15 + r10]
    cmp cl, '0'
    jb .line_error
    cmp cl, '9'
    ja .line_error
    sub cl, '0'
    movzx ecx, cl
    imul rax, rax, 10
    add rax, rcx
    inc r10
    jmp .parse_digits

.digits_done:
    cmp r11d, 1
    jne .push_value
    neg rax

.push_value:
    lea rsi, [rip + stack_values]
    mov qword ptr [rsi + r14*8], rax
    inc r14
    mov rbx, r8
    jmp parse_loop

.op_plus:
    cmp r14, 2
    jb .line_error
    lea rsi, [rip + stack_values]
    mov rdx, r14
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov rdx, r14
    sub rdx, 2
    mov rbx, qword ptr [rsi + rdx*8]
    sub r14, 2
    add rbx, rax
    mov qword ptr [rsi + r14*8], rbx
    inc r14
    mov rbx, rcx
    jmp parse_loop

.op_minus:
    cmp r14, 2
    jb .line_error
    lea rsi, [rip + stack_values]
    mov rdx, r14
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov rdx, r14
    sub rdx, 2
    mov rbx, qword ptr [rsi + rdx*8]
    sub r14, 2
    sub rbx, rax
    mov qword ptr [rsi + r14*8], rbx
    inc r14
    mov rbx, rcx
    jmp parse_loop

.op_mul:
    cmp r14, 2
    jb .line_error
    lea rsi, [rip + stack_values]
    mov rdx, r14
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov rdx, r14
    sub rdx, 2
    mov rbx, qword ptr [rsi + rdx*8]
    sub r14, 2
    imul rbx, rax
    mov qword ptr [rsi + r14*8], rbx
    inc r14
    mov rbx, rcx
    jmp parse_loop

.op_div:
    cmp r14, 2
    jb .line_error
    lea rsi, [rip + stack_values]
    mov rdx, r14
    dec rdx
    mov rbx, qword ptr [rsi + rdx*8]
    mov rdx, r14
    sub rdx, 2
    mov rax, qword ptr [rsi + rdx*8]
    sub r14, 2
    cmp rbx, 0
    je .line_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne .div_ok
    cmp rbx, -1
    jne .div_ok
    jmp .line_error

.div_ok:
    cqo
    idiv rbx
    mov qword ptr [rsi + r14*8], rax
    inc r14
    mov rbx, rcx
    jmp parse_loop

.op_mod:
    cmp r14, 2
    jb .line_error
    lea rsi, [rip + stack_values]
    mov rdx, r14
    dec rdx
    mov rbx, qword ptr [rsi + rdx*8]
    mov rdx, r14
    sub rdx, 2
    mov rax, qword ptr [rsi + rdx*8]
    sub r14, 2
    cmp rbx, 0
    je .line_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne .mod_ok
    cmp rbx, -1
    jne .mod_ok
    jmp .line_error

.mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    mov qword ptr [rsi + r14*8], rax
    inc r14
    mov rbx, rcx
    jmp parse_loop

.line_error:
    mov r9d, 1
    mov rbx, r13
    jmp .line_end

.line_end:
    cmp r9d, 1
    je .emit_error
    cmp r14, 1
    jne .emit_error

    lea rsi, [rip + stack_values]
    mov r8, qword ptr [rsi]

    lea r9, [rip + output_buf + 31]
    mov byte ptr [r9], 0
    mov r10, 0
    mov r11, 0

    cmp r8, 0
    jne .not_zero

    mov byte ptr [r9], '0'
    mov r11, 1
    jmp .write_value

.not_zero:
    cmp r8, 0
    jge .positive
    mov rdx, 0x8000000000000000
    cmp r8, rdx
    je .special_min
    neg r8
    mov r10, 1

.positive:
    mov rax, r8

.digits_loop:
    test rax, rax
    jz .digits_done
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    inc r11
    jmp .digits_loop

.digits_done:
    cmp r10, 1
    jne .write_value
    dec r9
    mov byte ptr [r9], '-'
    inc r11
    jmp .write_value

.special_min:
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, 20
    syscall
    cmp rax, 0
    jle .exit_fail
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    cmp rax, 0
    jle .exit_fail
    jmp .after_line

.write_value:
    mov r10, r9
    mov r11, r11
.write_value_loop:
    mov rax, SYS_WRITE
    mov rdi, 1
    mov rsi, r10
    mov rdx, r11
    syscall
    cmp rax, 0
    jle .exit_fail
    sub r11, rax
    add r10, rax
    jne .write_value_loop

    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    cmp rax, 0
    jle .exit_fail
    jmp .after_line

.emit_error:
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 6
    syscall
    cmp rax, 0
    jle .exit_fail

.after_line:
    mov r12, r13
    cmp r13, rbp
    jae .next_line
    movzx eax, byte ptr [r15 + r13]
    cmp al, 0x0A
    jne .next_line
    inc r12

.next_line:
    jmp process_lines

.exit_ok:
    mov rax, SYS_EXIT
    xor edi, edi
    syscall

.exit_fail:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall
