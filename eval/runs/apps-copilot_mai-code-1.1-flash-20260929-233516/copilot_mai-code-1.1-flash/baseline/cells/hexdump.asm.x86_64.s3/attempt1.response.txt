.globl _start

MAX_INPUT = 65536

.text
_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jb read_loop
read_done:
    xor r14, r14
row_loop:
    cmp r14, r12
    jae print_total
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_len_ok
    mov r15, 16
row_len_ok:
    lea rdi, [rip + row_buffer]
    mov r8, r14
    mov r9, 7
offset_loop:
    mov r10, r8
    and r10, 0x0f
    movzx r11d, byte ptr [rip + hex_digits + r10]
    mov byte ptr [rdi + r9], r11b
    shr r8, 4
    dec r9
    jns offset_loop
    lea rdi, [rdi + 8]
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    xor rcx, rcx
hex_loop:
    cmp rcx, 16
    je ascii_part
    cmp rcx, r15
    jae blank_byte
    movzx r8, byte ptr [rsi + rcx]
    mov r9, r8
    shr r9, 4
    movzx r10d, byte ptr [rip + hex_digits + r9]
    mov byte ptr [rdi], r10b
    inc rdi
    mov r9, r8
    and r9, 0x0f
    movzx r10d, byte ptr [rip + hex_digits + r9]
    mov byte ptr [rdi], r10b
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    jmp after_slot
blank_byte:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
after_slot:
    cmp rcx, 7
    jne not_extra
    mov byte ptr [rdi], ' '
    inc rdi
not_extra:
    inc rcx
    jmp hex_loop
ascii_part:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    xor rcx, rcx
ascii_loop:
    cmp rcx, r15
    jae ascii_done
    movzx r8, byte ptr [rsi + rcx]
    cmp r8b, 0x20
    jb dot_char
    cmp r8b, 0x7e
    ja dot_char
    mov byte ptr [rdi], r8b
    jmp ascii_after
dot_char:
    mov byte ptr [rdi], '.'
ascii_after:
    inc rdi
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + row_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    add r14, r15
    jmp row_loop

print_total:
    lea rdi, [rip + final_buf]
    mov r8, r12
    mov r9, 7
total_loop:
    mov r10, r8
    and r10, 0x0f
    movzx r11d, byte ptr [rip + hex_digits + r10]
    mov byte ptr [rdi + r9], r11b
    shr r8, 4
    dec r9
    jns total_loop
    lea rsi, [rip + final_buf]
    mov byte ptr [rsi + 8], 10
    mov rax, 1
    mov rdi, 1
    mov rdx, 9
    syscall
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buffer:
    .space 65536
row_buffer:
    .space 96
final_buf:
    .space 16
