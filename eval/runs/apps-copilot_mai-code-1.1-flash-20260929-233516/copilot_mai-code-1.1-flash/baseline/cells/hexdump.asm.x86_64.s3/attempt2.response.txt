.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buffer]

read_more:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    xor r14d, r14d

row_loop:
    cmp r14, r12
    jae print_total
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_len_ok
    mov r15, 16

row_len_ok:
    lea rdi, [rip + row_buffer]
    lea rsi, [rip + hex_digits]
    mov rax, r14
    mov rcx, 28

offset_fill:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xf
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub rcx, 4
    jns offset_fill

    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

    lea r11, [r13 + r14]
    xor rcx, rcx

hex_loop:
    cmp rcx, 16
    je ascii_write_start
    cmp rcx, r15
    jae blank_byte

    movzx eax, byte ptr [r11 + rcx]
    mov rdx, rax
    shr rdx, 4
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    jmp after_slot

blank_byte:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

after_slot:
    cmp rcx, 7
    jne slot_continue
    mov byte ptr [rdi], ' '
    inc rdi

slot_continue:
    inc rcx
    jmp hex_loop

ascii_write_start:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    xor rcx, rcx

ascii_loop:
    cmp rcx, r15
    jae ascii_done
    movzx eax, byte ptr [r11 + rcx]
    cmp al, 0x20
    jb dot_char
    cmp al, 0x7e
    ja dot_char
    mov byte ptr [rdi], al
    jmp ascii_after

dot_char:
    mov byte ptr [rdi], '.'

ascii_after:
    inc rdi
    inc rcx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + row_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    add r14, r15
    jmp row_loop

print_total:
    lea rdi, [rip + final_buf]
    mov rax, r12
    lea rsi, [rip + hex_digits]
    mov rcx, 28

total_fill:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xf
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub rcx, 4
    jns total_fill

    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + final_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

exit_now:
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
input_buffer:
    .space 65536

row_buffer:
    .space 96

final_buf:
    .space 16
