    b.eq .L_check_l2
    b .L_fail_end

.L_check_n:
    cmp w14, #0x6E
    b.ne .L_fail_pos
    b .L_next

.L_check_u:
    cmp w14, #0x75
    b.ne .L_fail_pos
    b .L_next

.L_check_l:
    cmp w14, #0x6C
    b.ne .L_fail_pos
    b .L_next

.L_check_l2:
    cmp w14, #0x6C
    b.ne .L_fail_pos
    b .L_next

.L_next:
    add x12, x12, #1
    b .L_loop

.L_ok:
    add x19, x19, #4
    mov x0, x19
    b .L_done

.L_fail_pos:
    mov x0, x13
    bl fail_at
    b .L_done

.L_fail_end:
    mov x0, x13
    bl fail_at
.L_done:
    ldp x29, x30, [sp], #16
    ret

print_error_at:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #64
    mov x19, x0

    adrp x9, err_prefix
    add x9, x9, :lo12:err_prefix
    mov x10, #0
    mov x20, sp

.L_prefix_loop:
    ldrb w11, [x9, x10]
    cbz w11, .L_prefix_done
    strb w11, [x20, x10]
    add x10, x10, #1
    b .L_prefix_loop

.L_prefix_done:
    mov x21, sp
    add x21, x21, #32
    add x22, x21, #15
    mov x23, #10
    mov x24, x19

    cmp x24, #0
    b.ne .L_digit_loop
    mov w25, #0x30
    strb w25, [x22]
    mov x26, #1
    b .L_write_digits

.L_digit_loop:
    cmp x24, #0
    b.eq .L_write_digits
    udiv x27, x24, x23
    msub x28, x27, x23, x24
    add w28, w28, #0x30
    strb w28, [x22]
    sub x22, x22, #1
    add x26, x26, #1
    mov x24, x27
    b .L_digit_loop

.L_write_digits:
    add x22, x22, #1
    mov x0, #1
    mov x1, sp
    mov x2, x10
    mov x8, #64
    svc #0

    mov x0, #1
    mov x1, x22
    mov x2, x26
    mov x8, #64
    svc #0

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0

    add sp, sp, #64
    ldp x29, x30, [sp], #16
    ret
