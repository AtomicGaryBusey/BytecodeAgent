.globl _start
.text

_start:
    xor rbx, rbx
    xor r12, r12
    xor r13, r13
    xor r14, r14
    sub rsp, 65568

read_loop:
    xor rax, rax
    mov rdi, rax
    lea rsi, [rsp]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_count
    mov r15, rax
    add r13, r15
    mov rcx, r15
    lea rsi, [rsp]

process_bytes:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 0x0A
    jne not_newline
    inc rbx

not_newline:
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    test r14, r14
    jnz process_bytes
    inc r12
    mov r14, 1
    jmp process_bytes

whitespace:
    xor r14, r14
    jmp process_bytes

done_count:
    lea rsi, [rsp + 65536 + 31]
    mov byte ptr [rsi], 0x0A

    mov rax, r13
    test rax, rax
    jz bytes_zero
bytes_loop:
    xor rdx, rdx
    mov r8, 10
    div r8
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz bytes_loop
    jmp bytes_done
bytes_zero:
    dec rsi
    mov byte ptr [rsi], 0x30
bytes_done:
    dec rsi
    mov byte ptr [rsi], 0x20

    mov rax, r12
    test rax, rax
    jz words_zero
words_loop:
    xor rdx, rdx
    mov r8, 10
    div r8
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz words_loop
    jmp words_done
words_zero:
    dec rsi
    mov byte ptr [rsi], 0x30
words_done:
    dec rsi
    mov byte ptr [rsi], 0x20

    mov rax, rbx
    test rax, rax
    jz lines_zero
lines_loop:
    xor rdx, rdx
    mov r8, 10
    div r8
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz lines_loop
    jmp output_ready
lines_zero:
    dec rsi
    mov byte ptr [rsi], 0x30

output_ready:
    lea rdx, [rsp + 65568]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
