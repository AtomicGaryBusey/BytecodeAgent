.globl _start
.text

_start:
    xor rbx, rbx
    xor r12, r12
    xor r13, r13
    xor r14, r14
    sub rsp, 65536 + 64

read_loop:
    xor rax, rax
    mov rdi, rax
    lea rsi, [rsp + 64]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_input
    mov rcx, rax
    add r13, rax
    lea rsi, [rsp + 64]

process_chunk:
    test rcx, rcx
    jz read_loop
    movzx r8d, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp r8b, 0x0A
    jne not_newline
    inc rbx

not_newline:
    cmp r8b, 0x20
    je whitespace
    cmp r8b, 0x09
    je whitespace
    cmp r8b, 0x0B
    je whitespace
    cmp r8b, 0x0C
    je whitespace
    cmp r8b, 0x0D
    je whitespace
    test r14, r14
    jnz process_chunk
    inc r12
    mov r14, 1
    jmp process_chunk

whitespace:
    xor r14, r14
    jmp process_chunk

done_input:
    lea rsi, [rsp + 63]
    mov byte ptr [rsi], 0x0A
    dec rsi
    mov rax, r13
    call emit_dec
    mov byte ptr [rsi], 0x20
    dec rsi
    mov rax, r12
    call emit_dec
    mov byte ptr [rsi], 0x20
    dec rsi
    mov rax, rbx
    call emit_dec
    inc rsi
    lea rdx, [rsp + 64]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

emit_dec:
    mov r8, 10
    test rax, rax
    jnz emit_dec_loop
    mov byte ptr [rsi], 0x30
    dec rsi
    ret

emit_dec_loop:
    xor rdx, rdx
    div r8
    add dl, 0x30
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz emit_dec_loop
    ret
