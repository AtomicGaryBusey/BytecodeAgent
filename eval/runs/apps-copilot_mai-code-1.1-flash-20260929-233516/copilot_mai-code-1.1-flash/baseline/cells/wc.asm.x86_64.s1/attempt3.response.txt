.globl _start

.bss
    .align 16
input_buf:
    .space 65536
output_buf:
    .space 64

.text

emit_dec_left:
    mov r8, 10
    mov r9, rdi
    test rax, rax
    jnz 1f
    dec r9
    mov byte ptr [r9], 0x30
    mov rax, r9
    ret
1:
    xor rdx, rdx
    div r8
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 1b
    mov rax, r9
    ret

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor rax, rax
    mov rdi, rax
    lea rsi, [rip + input_buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz done_input
    mov rcx, rax
    add r14, rax
    lea rsi, [rip + input_buf]

process_chunk:
    test rcx, rcx
    jz read_loop
    movzx r8d, byte ptr [rsi]
    inc rsi
    dec rcx

    cmp r8b, 0x0A
    je newline_char

    cmp r8b, 0x20
    je whitespace
    cmp r8b, 0x09
    je whitespace
    cmp r8b, 0x0B
    je whitespace
    cmp r8b, 0x0C
    je whitespace
    cmp r8b, 0x0D
    je whitespace

    test r15, r15
    jnz process_chunk
    inc r13
    mov r15, 1
    jmp process_chunk

newline_char:
    inc r12
    xor r15, r15
    jmp process_chunk

whitespace:
    xor r15, r15
    jmp process_chunk

done_input:
    lea rsi, [rip + output_buf]
    add rsi, 64
    mov byte ptr [rsi - 1], 0x0A
    mov rdi, rsi
    dec rdi
    mov rax, r14
    call emit_dec_left
    mov rdi, rax
    mov byte ptr [rdi - 1], 0x20
    dec rdi
    mov rax, r13
    call emit_dec_left
    mov rdi, rax
    mov byte ptr [rdi - 1], 0x20
    dec rdi
    mov rax, r12
    call emit_dec_left
    mov rsi, rax
    lea rdx, [rip + output_buf]
    add rdx, 64
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
