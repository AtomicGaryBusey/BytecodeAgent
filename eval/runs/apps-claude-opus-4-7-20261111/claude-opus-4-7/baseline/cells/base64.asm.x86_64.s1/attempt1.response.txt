.globl _start

.bss
.align 16
inbuf: .zero 49152
outbuf: .zero 70000

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_prog

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alphabet]
    mov rcx, r12
    xor r8, r8

encode_loop:
    cmp rcx, 3
    jl final_group

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]
    movzx r9d, byte ptr [rsi+2]

    mov r10d, eax
    shr r10d, 2
    mov r11b, byte ptr [rbx + r10]
    mov byte ptr [rdi], r11b

    and eax, 3
    shl eax, 4
    mov r10d, edx
    shr r10d, 4
    or eax, r10d
    mov r11b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r11b

    and edx, 15
    shl edx, 2
    mov r10d, r9d
    shr r10d, 6
    or edx, r10d
    mov r11b, byte ptr [rbx + rdx]
    mov byte ptr [rdi+2], r11b

    and r9d, 63
    mov r11b, byte ptr [rbx + r9]
    mov byte ptr [rdi+3], r11b

    add rsi, 3
    add rdi, 4
    sub rcx, 3
    add r8, 4
    cmp r8, 76
    jne no_nl
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r8, r8
no_nl:
    jmp encode_loop

final_group:
    test rcx, rcx
    jz after_final
    cmp rcx, 1
    je final_1

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]

    mov r10d, eax
    shr r10d, 2
    mov r11b, byte ptr [rbx + r10]
    mov byte ptr [rdi], r11b

    and eax, 3
    shl eax, 4
    mov r10d, edx
    shr r10d, 4
    or eax, r10d
    mov r11b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r11b

    and edx, 15
    shl edx, 2
    mov r11b, byte ptr [rbx + rdx]
    mov byte ptr [rdi+2], r11b

    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    add r8, 4
    jmp after_final

final_1:
    movzx eax, byte ptr [rsi]
    mov r10d, eax
    shr r10d, 2
    mov r11b, byte ptr [rbx + r10]
    mov byte ptr [rdi], r11b

    and eax, 3
    shl eax, 4
    mov r11b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r11b

    mov byte ptr [rdi+2], 0x3D
    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    add r8, 4

after_final:
    test r8, r8
    jz skip_nl
    mov byte ptr [rdi], 0x0A
    inc rdi
skip_nl:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall

exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
