.section .bss
.balign 16
input_buf:   .zero 65536
output_buf:  .zero 524288
rpn_stack:   .zero 8192

.text
.globl _start

    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    add x27, x19, #16, lsl #12

    subs x2, x27, x20
    b.eq read_done
    mov x8, #63
    mov x0, #0
    mov x1, x20
    svc #0
    cbz x0, read_done

    add x20, x20, x0
    b read_loop

    adrp x25, rpn_stack
    add x25, x25, :lo12:rpn_stack
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x21, x19

next_line:
    cmp x21, x20
    b.hs all_done
    mov x22, x21
scan_nl:
    cmp x21, x20
    b.hs eol_hit
    ldrb w0, [x21]
    add x21, x21, #1
    cmp w0, #0x0A
    b.ne scan_nl
    sub x23, x21, #1
    b have_line
eol_hit:
    mov x23, x21

have_line:
    mov x24, x25
    mov x9, x22

token_loop:
    cmp x9, x23
    b.hs end_line
    ldrb w0, [x9]
    cmp w0, #0x20
    b.eq skip_sep
    cmp w0, #0x09
    b.eq skip_sep
    b find_token_end
skip_sep:
    add x9, x9, #1
    b token_loop

find_token_end:
    mov x10, x9
ftt:
    cmp x10, x23
    b.hs ft_done
    ldrb w0, [x10]
    cmp w0, #0x20
    b.eq ft_done
    cmp w0, #0x09
    b.eq ft_done
    add x10, x10, #1
    b ftt
ft_done:
    sub x11, x10, x9
    cmp x11, #1
    b.ne try_number
    ldrb w12, [x9]
    cmp w12, #'+'
    b.eq do_op
    cmp w12, #'-'
    b.eq do_op
    cmp w12, #'*'
    b.eq do_op
    cmp w12, #'/'
    b.eq do_op
    cmp w12, #'%'
    b.eq do_op

try_number:
    mov x12, x9
    mov x13, #0
    ldrb w14, [x12]
    cmp w14, #'-'
    b.ne tn_no_sign
    mov x13, #1
    add x12, x12, #1
tn_no_sign:
    cmp x12, x10
    b.eq line_error_jmp
    mov x15, x12
    mov x14, #0
    mov x17, #10
tn_digit:
    cmp x12, x10
    b.hs tn_done
    ldrb w16, [x12]
    sub w16, w16, #'0'
    cmp w16, #9
    b.hi line_error_jmp
    madd x14, x14, x17, x16
    add x12, x12, #1
    b tn_digit
tn_done:
    sub x16, x12, x15
    cmp x16, #18
    b.hi line_error_jmp
    cbz x13, tn_no_neg
    neg x14, x14
tn_no_neg:
    str x14, [x24], #8
    mov x9, x10
    b token_loop

line_error_jmp:
    b line_error

do_op:
    sub x15, x24, x25
    cmp x15, #16
    b.lo line_error_jmp
    ldr x14, [x24, #-8]
    ldr x15, [x24, #-16]
    sub x24, x24, #16
    cmp w12, #'+'
    b.eq op_add
    cmp w12, #'-'
    b.eq op_sub
    cmp w12, #'*'
    b.eq op_mul
    cmp w12, #'/'
    b.eq op_div
op_mod:
    cbz x14, line_error_jmp
    mov x16, #0
    movk x16, #0x8000, lsl #48
    cmp x15, x16
    b.ne 1f
    cmn x14, #1
    b.eq line_error_jmp
1:  sdiv x17, x15, x14
    msub x15, x17, x14, x15
    b op_push

op_add:
    add x15, x15, x14
    b op_push
op_sub:
    sub x15, x15, x14
    b op_push
op_mul:
    mul x15, x15, x14
    b op_push
op_div:
    cbz x14, line_error_jmp
    mov x16, #0
    movk x16, #0x8000, lsl #48
    cmp x15, x16
    b.ne 2f
    cmn x14, #1
    b.eq line_error_jmp
2:  sdiv x15, x15, x14
op_push:
    str x15, [x24], #8
    mov x9, x10
    b token_loop

end_line:
    sub x15, x24, x25
    cmp x15, #8
    b.ne line_error
    ldr x0, [x25]
    tbz x0, #63, pn_pos
    mov w1, #'-'
    strb w1, [x26], #1
    neg x0, x0
pn_pos:
    sub sp, sp, #32
    add x2, sp, #32
    mov x3, x2
    mov x4, #10
pn_loop:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #'0'
    sub x3, x3, #1
    strb w6, [x3]
    mov x0, x5
    cbnz x0, pn_loop
pn_copy:
    ldrb w5, [x3], #1
    strb w5, [x26], #1
    cmp x3, x2
    b.lo pn_copy
    mov w5, #0x0A
    strb w5, [x26], #1
    add sp, sp, #32
    b next_line

line_error:
    mov w0, #'e'
    strb w0, [x26], #1
    mov w0, #'r'
    strb w0, [x26], #1
    strb w0, [x26], #1
    mov w0, #'o'
    strb w0, [x26], #1
    mov w0, #'r'
    strb w0, [x26], #1
    mov w0, #0x0A
    strb w0, [x26], #1
    b next_line

all_done:
    adrp x0, output_buf
    add x0, x0, :lo12:output_buf
    sub x2, x26, x0
    cbz x2, exit_prog
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0
exit_prog:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_prog
