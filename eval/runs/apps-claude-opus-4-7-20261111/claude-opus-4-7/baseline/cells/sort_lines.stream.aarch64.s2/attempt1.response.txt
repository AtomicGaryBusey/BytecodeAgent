```asm
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .rodata
nl_byte: .byte 0x0A

.section .bss
.align 4
input_buf: .zero 40000
line_off:  .zero 8192
line_len:  .zero 8192
tmp_off:   .zero 8192
tmp_len:   .zero 8192

.text
.globl _start
_start:
    adrp x20, input_buf
    add  x20, x20, :lo12:input_buf
    mov  x19, #0
.Lread_loop:
    mov  w2, #40000
    sub  x2, x2, x19
    cbz  x2, .Lread_done
    mov  x0, #0
    add  x1, x20, x19
    mov  x8, #SYS_read
    svc  #0
    cmp  x0, #0
    b.le .Lread_done
    add  x19, x19, x0
    b    .Lread_loop
.Lread_done:

    adrp x22, line_off
    add  x22, x22, :lo12:line_off
    adrp x23, line_len
    add  x23, x23, :lo12:line_len
    mov  x21, #0
    mov  x24, #0
    mov  x25, #0

.Lparse_loop:
    cmp  x25, x19
    b.ge .Lparse_check_tail
    ldrb w0, [x20, x25]
    cmp  w0, #0x0A
    b.ne .Lparse_cont
    str  x24, [x22, x21, lsl #3]
    sub  x0, x25, x24
    str  x0, [x23, x21, lsl #3]
    add  x21, x21, #1
    add  x25, x25, #1
    mov  x24, x25
    b    .Lparse_loop
.Lparse_cont:
    add  x25, x25, #1
    b    .Lparse_loop

.Lparse_check_tail:
    cmp  x24, x19
    b.ge .Lparse_done
    str  x24, [x22, x21, lsl #3]
    sub  x0, x19, x24
    str  x0, [x23, x21, lsl #3]
    add  x21, x21, #1
.Lparse_done:

    cbz  x21, .Ldo_exit

    adrp x26, tmp_off
    add  x26, x26, :lo12:tmp_off
    adrp x27, tmp_len
    add  x27, x27, :lo12:tmp_len

    mov  x28, #1
.Lms_loop:
    cmp  x28, x21
    b.ge .Lms_done

    mov  x0, #0
.Lms_pair:
    cmp  x0, x21
    b.ge .Lms_next_size
    add  x1, x0, x28
    cmp  x1, x21
    csel x1, x1, x21, lt
    add  x2, x1, x28
    cmp  x2, x21
    csel x2, x2, x21, lt

    stp  x0, x1, [sp, #-32]!
    stp  x2, x28, [sp, #16]

    mov  x3, x0
    mov  x4, x0
    mov  x5, x1
.Lms_merge:
    cmp  x4, x1
    b.ge .Lms_r_only
    cmp  x5, x2
    b.ge .Lms_l_only
    ldr  x6, [x22, x4, lsl #3]
    ldr  x7, [x23, x4, lsl #3]
    ldr  x9, [x22, x5, lsl #3]
    ldr  x10, [x23, x5, lsl #3]
    cmp  x7, x10
    csel x11, x7, x10, lt
    mov  x12, #0
.Lcmp_byte:
    cmp  x12, x11
    b.ge .Lcmp_prefix
    ldrb w13, [x20, x6]
    ldrb w14, [x20, x9]
    add  x6, x6, #1
    add  x9, x9, #1
    add  x12, x12, #1
    cmp  w13, w14
    b.lt .Ltake_left
    b.gt .Ltake_right
    b    .Lcmp_byte
.Lcmp_prefix:
    cmp  x7, x10
    b.le .Ltake_left
    b    .Ltake_right
.Ltake_left:
    ldr  x6, [x22, x4, lsl #3]
    ldr  x7, [x23, x4, lsl #3]
    str  x6, [x26, x3, lsl #3]
    str  x7, [x27, x3, lsl #3]
    add  x3, x3, #1
    add  x4, x4, #1
    b    .Lms_merge
.Ltake_right:
    ldr  x9, [x22, x5, lsl #3]
    ldr  x10, [x23, x5, lsl #3]
    str  x9, [x26, x3, lsl #3]
    str  x10, [x27, x3, lsl #3]
    add  x3, x3, #1
    add  x5, x5, #1
    b    .Lms_merge
.Lms_r_only:
    cmp  x5, x2
    b.ge .Lms_m_done
    ldr  x9, [x22, x5, lsl #3]
    ldr  x10, [x23, x5, lsl #3]
    str  x9, [x26, x3, lsl #3]
    str  x10, [x27, x3, lsl #3]
    add  x3, x3, #1
    add  x5, x5, #1
    b    .Lms_r_only
.Lms_l_only:
    cmp  x4, x1
    b.ge .Lms_m_done
    ldr  x6, [x22, x4, lsl #3]
    ldr  x7, [x23, x4, lsl #3]
    str  x6, [x26, x3, lsl #3]
    str  x7, [x27, x3, lsl #3]
    add  x3, x3, #1
    add  x4, x4, #1
    b    .Lms_l_only
.Lms_m_done:
    ldp  x2, x28, [sp, #16]
    ldp  x0, x1, [sp], #32
    add  x0, x0, x28, lsl #1
    b    .Lms_pair

.Lms_next_size:
    mov  x9, x22
    mov  x22, x26
    mov  x26, x9
    mov  x9, x23
    mov  x23, x27
    mov  x27, x9
    lsl  x28, x28, #1
    b    .Lms_loop

.Lms_done:
    mov  x24, #0
.Lout_loop:
    cmp  x24, x21
    b.ge .Lout_done
    ldr  x1, [x22, x24, lsl #3]
    ldr  x2, [x23, x24, lsl #3]
    add  x1, x20, x1
    cbz  x2, .Lwrite_nl
    mov  x0, #1
    mov  x8, #SYS_write
    svc  #0
.Lwrite_nl:
    adrp x1, nl_byte
    add  x1, x1, :lo12:nl_byte
    mov  x0, #1
    mov  x2, #1
    mov  x8, #SYS_write
    svc  #0
    add  x24, x24, #1
    b    .Lout_loop
.Lout_done:

.Ldo_exit:
    mov  x0, #0
    mov  x8, #SYS_exit
    svc  #0
    b    .Ldo_exit
```
