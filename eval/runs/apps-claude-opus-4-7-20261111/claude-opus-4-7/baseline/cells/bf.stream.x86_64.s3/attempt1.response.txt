```
.globl _start

.section .bss
.align 16
input_buf:     .zero 65536
.align 16
clean_prog:    .zero 65536
.align 16
tape:          .zero 30000
.align 16
jump_table:    .zero 262144
.align 16
bracket_stack: .zero 4000
.align 16
output_buf:    .zero 65536

.section .text
_start:
    # --- Read all of stdin ---
    xor ebx, ebx
.Lread:
    mov edx, 65536
    sub edx, ebx
    jz .Lread_done
    lea rsi, [rip + input_buf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    js .Lerr
    jz .Lread_done
    add ebx, eax
    jmp .Lread
.Lread_done:
    mov r12d, ebx

    # --- Find first '!' ---
    lea rbx, [rip + input_buf]
    xor ecx, ecx
.Lfind:
    cmp ecx, r12d
    jae .Lno_bang
    cmp byte ptr [rbx + rcx], 33
    je .Lfound
    inc ecx
    jmp .Lfind
.Lno_bang:
    mov r13d, r12d
    mov r14d, r12d
    jmp .Lafter
.Lfound:
    mov r13d, ecx
    lea r14d, [rcx + 1]
.Lafter:
    mov r15d, r12d

    # --- Validate brackets, build clean program and jump table ---
    lea rsi, [rip + bracket_stack]
    lea rdi, [rip + jump_table]
    lea r9, [rip + clean_prog]
    xor ecx, ecx
    xor r10d, r10d
    xor edx, edx
.Lvloop:
    cmp ecx, r13d
    jae .Lvdone
    movzx eax, byte ptr [rbx + rcx]
    inc ecx
    cmp al, 43
    je .Lvcmd
    cmp al, 45
    je .Lvcmd
    cmp al, 62
    je .Lvcmd
    cmp al, 60
    je .Lvcmd
    cmp al, 46
    je .Lvcmd
    cmp al, 44
    je .Lvcmd
    cmp al, 91
    je .Lvlbr
    cmp al, 93
    je .Lvrbr
    jmp .Lvloop
.Lvcmd:
    mov [r9 + r10], al
    inc r10d
    jmp .Lvloop
.Lvlbr:
    cmp edx, 1000
    jae .Lerr
    mov [rsi + rdx*4], r10d
    inc edx
    mov [r9 + r10], al
    inc r10d
    jmp .Lvloop
.Lvrbr:
    test edx, edx
    jz .Lerr
    dec edx
    mov r8d, [rsi + rdx*4]
    mov [rdi + r8*4], r10d
    mov [rdi + r10*4], r8d
    mov [r9 + r10], al
    inc r10d
    jmp .Lvloop
.Lvdone:
    test edx, edx
    jnz .Lerr
    mov r13d, r10d

    # --- Execute ---
    lea r11, [rip + input_buf]
    mov rbx, r9
    lea rbp, [rip + tape]
    lea rsi, [rip + output_buf]
    xor ecx, ecx
    xor r12d, r12d
.Lexec:
    cmp ecx, r13d
    jae .Ldone
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 43
    je .Lop_plus
    cmp al, 45
    je .Lop_minus
    cmp al, 62
    je .Lop_gt
    cmp al, 60
    je .Lop_lt
    cmp al, 91
    je .Lop_lbr
    cmp al, 93
    je .Lop_rbr
    cmp al, 46
    je .Lop_dot
    # otherwise it's ','
    cmp r14d, r15d
    jae .Lcomma_eof
    movzx eax, byte ptr [r11 + r14]
    mov [rbp], al
    inc r14d
    inc ecx
    jmp .Lexec
.Lcomma_eof:
    mov byte ptr [rbp], 0
    inc ecx
    jmp .Lexec
.Lop_plus:
    inc byte ptr [rbp]
    inc ecx
    jmp .Lexec
.Lop_minus:
    dec byte ptr [rbp]
    inc ecx
    jmp .Lexec
.Lop_gt:
    inc rbp
    inc ecx
    jmp .Lexec
.Lop_lt:
    dec rbp
    inc ecx
    jmp .Lexec
.Lop_lbr:
    cmp byte ptr [rbp], 0
    jne .Llbr_pass
    mov ecx, [rdi + rcx*4]
.Llbr_pass:
    inc ecx
    jmp .Lexec
.Lop_rbr:
    cmp byte ptr [rbp], 0
    je .Lrbr_pass
    mov ecx, [rdi + rcx*4]
.Lrbr_pass:
    inc ecx
    jmp .Lexec
.Lop_dot:
    mov al, [rbp]
    mov [rsi + r12], al
    inc r12d
    inc ecx
    jmp .Lexec

.Ldone:
    test r12d, r12d
    jz .Lexit
    mov edx, r12d
    mov edi, 1
    mov eax, 1
    syscall
.Lexit:
    xor edi, edi
    mov eax, 60
    syscall

.Lerr:
    mov edi, 1
    mov eax, 60
    syscall
```
