```
.bss
.balign 16
buf: .zero 4096
outbuf: .zero 128

.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    syscall
    test rax, rax
    jz done_read
    js done_read

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]

proc_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_nl
    inc r12
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    ja is_word
is_ws:
    xor r15d, r15d
    jmp proc_next
is_word:
    test r15d, r15d
    jnz proc_next
    inc r13
    mov r15d, 1
proc_next:
    dec rcx
    jnz proc_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf + 127]
    mov byte ptr [rdi], 0x0A
    dec rdi

    mov rax, r14
    call write_num
    mov byte ptr [rdi], 0x20
    dec rdi

    mov rax, r13
    call write_num
    mov byte ptr [rdi], 0x20
    dec rdi

    mov rax, r12
    call write_num

    inc rdi
    lea rax, [rip + outbuf + 128]
    mov rdx, rax
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

write_num:
    mov rcx, 10
    test rax, rax
    jnz wn_loop
    mov byte ptr [rdi], 0x30
    dec rdi
    ret
wn_loop:
    xor rdx, rdx
    div rcx
    add dl, 0x30
    mov [rdi], dl
    dec rdi
    test rax, rax
    jnz wn_loop
    ret
```
