.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_INPUT, 32768

.text
_start:
    mov x19, #0

read_loop:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x0, #0
    mov x2, #MAX_INPUT
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x21, line_table
    add x21, x21, :lo12:line_table
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

    mov x20, #0
    mov x9, #0
    mov x10, #0

parse_loop:
    cmp x9, x19
    b.eq parse_end
    ldrb w11, [x22, x9]
    cmp w11, #0x0A
    b.ne parse_cont
    sub w12, w9, w10
    str w10, [x21, x20, lsl #3]
    add x13, x21, x20, lsl #3
    str w12, [x13, #4]
    add x20, x20, #1
    add x9, x9, #1
    mov x10, x9
    b parse_loop

parse_cont:
    add x9, x9, #1
    b parse_loop

parse_end:
    cmp x9, x10
    b.eq do_sort
    sub w12, w9, w10
    str w10, [x21, x20, lsl #3]
    add x13, x21, x20, lsl #3
    str w12, [x13, #4]
    add x20, x20, #1

do_sort:
    cmp x20, #2
    b.lt do_output

    adrp x29, line_table2
    add x29, x29, :lo12:line_table2

    mov x19, #1

msort_outer:
    cmp x19, x20
    b.ge msort_done

    mov x23, #0

msort_inner:
    cmp x23, x20
    b.ge msort_swap

    add x24, x23, x19
    cmp x24, x20
    csel x24, x24, x20, ls

    add x25, x24, x19
    cmp x25, x20
    csel x25, x25, x20, ls

    mov x26, x23
    mov x27, x24
    mov x28, x23

merge_loop:
    cmp x26, x24
    b.eq merge_copy_right
    cmp x27, x25
    b.eq merge_copy_left

    ldr x0, [x21, x26, lsl #3]
    ldr x1, [x21, x27, lsl #3]
    bl compare_lines
    cmp x0, #0
    b.gt take_right

    ldr x0, [x21, x26, lsl #3]
    str x0, [x29, x28, lsl #3]
    add x26, x26, #1
    add x28, x28, #1
    b merge_loop

take_right:
    ldr x0, [x21, x27, lsl #3]
    str x0, [x29, x28, lsl #3]
    add x27, x27, #1
    add x28, x28, #1
    b merge_loop

merge_copy_right:
    cmp x27, x25
    b.eq merge_done
    ldr x0, [x21, x27, lsl #3]
    str x0, [x29, x28, lsl #3]
    add x27, x27, #1
    add x28, x28, #1
    b merge_copy_right

merge_copy_left:
    cmp x26, x24
    b.eq merge_done
    ldr x0, [x21, x26, lsl #3]
    str x0, [x29, x28, lsl #3]
    add x26, x26, #1
    add x28, x28, #1
    b merge_copy_left

merge_done:
    mov x23, x25
    b msort_inner

msort_swap:
    mov x9, x21
    mov x21, x29
    mov x29, x9
    lsl x19, x19, #1
    b msort_outer

msort_done:

do_output:
    mov x23, #0

output_loop:
    cmp x23, x20
    b.ge do_exit

    ldr x9, [x21, x23, lsl #3]
    mov w10, w9
    lsr x11, x9, #32

    cbz x11, skip_write
    mov x0, #1
    add x1, x22, x10
    mov x2, x11
    mov x8, #SYS_write
    svc #0

skip_write:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    add x23, x23, #1
    b output_loop

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

compare_lines:
    mov w2, w0
    lsr x3, x0, #32
    mov w4, w1
    lsr x5, x1, #32

    cmp x3, x5
    csel x6, x3, x5, lo

    mov x7, #0

cmp_loop:
    cmp x7, x6
    b.eq cmp_lens
    add x9, x22, x2
    ldrb w10, [x9, x7]
    add x9, x22, x4
    ldrb w11, [x9, x7]
    cmp w10, w11
    b.ne cmp_diff
    add x7, x7, #1
    b cmp_loop

cmp_diff:
    sub w0, w10, w11
    sxtw x0, w0
    ret

cmp_lens:
    sub x0, x3, x5
    ret

.section .rodata
newline: .byte 0x0A

.bss
.balign 16
input_buf: .zero 33000
line_table: .zero 8000
line_table2: .zero 8000
