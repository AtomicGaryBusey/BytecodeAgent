    add     x19, x19, :lo12:inbuf
    mov     x20, #0
    movz    x21, #1, lsl #16

    mov     x8, #SYS_read
    mov     x0, #0
    add     x1, x19, x20

    b.le    read_done
    add     x20, x20, x0

    mov     x23, #0
    mov     x24, #0
    mov     x25, #0

    b.ge    after_scan
    ldrb    w0, [x19, x25]
    cmp     w0, #0x0A
    b.eq    got_newline
    add     x25, x25, #1

    cmp     x26, x24
    b.lt    rev_done1
    ldrb    w0, [x19, x26]
    strb    w0, [x22, x23]

    sub     x26, x26, #1
    b       rev_copy1

    strb    w0, [x22, x23]
    add     x23, x23, #1
    add     x25, x25, #1
    mov     x24, x25

    b.ge    write_out
    sub     x26, x20, #1
rev_copy2:
    cmp     x26, x24
    b.lt    rev_done2
    ldrb    w0, [x19, x26]
    strb    w0, [x22, x23]
    add     x23, x23, #1
    sub     x26, x26, #1
    b       rev_copy2

    strb    w0, [x22, x23]

    mov     x1, x22
    mov     x2, x23
    mov     x27, #0

    sub     x2, x23, x27
    svc     #0
    cmp     x0, #0
    b.le    do_exit
    add     x27, x27, x0
    b       write_loop

do_exit:
    mov     x8, #SYS_exit
    mov     x0, #0
    svc     #0
    b       do_exit
