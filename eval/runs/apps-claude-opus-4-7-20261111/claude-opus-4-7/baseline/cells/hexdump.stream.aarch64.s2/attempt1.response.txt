.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 400000

.text

// write_hex8: value in x0, writes 8 lowercase hex digits via x22
write_hex8:
    mov x3, x0
    mov x1, #28
1:  lsr x2, x3, x1
    and x2, x2, #0xf
    ldrb w2, [x23, x2]
    strb w2, [x22], #1
    subs x1, x1, #4
    b.ge 1b
    ret

// write_hex2: byte value in w0, writes 2 lowercase hex digits via x22
write_hex2:
    ubfx x1, x0, #4, #4
    and x2, x0, #0xf
    ldrb w1, [x23, x1]
    strb w1, [x22], #1
    ldrb w2, [x23, x2]
    strb w2, [x22], #1
    ret

_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add  x21, x21, :lo12:output_buf
    adrp x23, hex_digits
    add  x23, x23, :lo12:hex_digits
    mov  x22, x21
    mov  x20, #0

read_loop:
    mov  x2, #65536
    sub  x2, x2, x20
    cbz  x2, read_done
    add  x1, x19, x20
    mov  x0, #0
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    mov  x24, #0

row_loop:
    cmp  x24, x20
    b.hs after_rows

    mov  x27, x24

    sub  x25, x20, x24
    cmp  x25, #16
    b.lo 1f
    mov  x25, #16
1:

    mov  x0, x27
    bl   write_hex8

    mov  w0, #0x20
    strb w0, [x22], #1
    strb w0, [x22], #1

    mov  x26, #0
byte_loop:
    cmp  x26, #16
    b.hs byte_done

    cmp  x26, x25
    b.hs byte_pad

    ldrb w0, [x19, x27]
    add  x27, x27, #1
    bl   write_hex2
    mov  w0, #0x20
    strb w0, [x22], #1
    b    byte_check

byte_pad:
    mov  w0, #0x20
    strb w0, [x22], #1
    strb w0, [x22], #1
    strb w0, [x22], #1

byte_check:
    cmp  x26, #7
    b.ne byte_next
    mov  w0, #0x20
    strb w0, [x22], #1
byte_next:
    add  x26, x26, #1
    b    byte_loop

byte_done:
    mov  w0, #0x20
    strb w0, [x22], #1
    mov  w0, #0x7c
    strb w0, [x22], #1

    mov  x26, #0
ascii_loop:
    cmp  x26, x25
    b.hs ascii_done
    ldrb w0, [x19, x24]
    add  x24, x24, #1
    cmp  w0, #0x20
    b.lo ascii_dot
    cmp  w0, #0x7e
    b.hi ascii_dot
    b    ascii_put
ascii_dot:
    mov  w0, #0x2e
ascii_put:
    strb w0, [x22], #1
    add  x26, x26, #1
    b    ascii_loop

ascii_done:
    mov  w0, #0x7c
    strb w0, [x22], #1
    mov  w0, #0x0a
    strb w0, [x22], #1

    b    row_loop

after_rows:
    mov  x0, x20
    bl   write_hex8
    mov  w0, #0x0a
    strb w0, [x22], #1

    sub  x2, x22, x21
    mov  x1, x21
    mov  x0, #1
    mov  x8, #64
    svc  #0

    mov  x0, #0
    mov  x8, #93
    svc  #0
    b    .
