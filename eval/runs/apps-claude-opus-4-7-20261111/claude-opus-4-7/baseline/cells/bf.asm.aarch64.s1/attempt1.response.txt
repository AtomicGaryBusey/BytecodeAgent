```
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 8
stdin_buf:    .zero 65536
tape:         .zero 30000
.balign 4
jump_table:   .zero 262144
bracket_stack:.zero 4000
out_buf:      .zero 65536

.text
_start:
    // Phase 1: read all stdin into stdin_buf
    adrp x19, stdin_buf
    add  x19, x19, :lo12:stdin_buf
    mov  x9, #0
    mov  x10, #0x10000
read_loop:
    cmp  x9, x10
    b.ge read_done
    mov  x8, #SYS_read
    mov  x0, #0
    add  x1, x19, x9
    sub  x2, x10, x9
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x9, x9, x0
    b    read_loop
read_done:
    // x9 = total bytes read; find first '!'
    mov  x11, x9
    mov  x12, #0
find_bang_loop:
    cmp  x12, x11
    b.ge find_bang_no
    ldrb w13, [x19, x12]
    cmp  w13, #0x21
    b.eq find_bang_found
    add  x12, x12, #1
    b    find_bang_loop
find_bang_found:
    mov  x20, x12
    add  x25, x19, x12
    add  x25, x25, #1
    add  x26, x19, x11
    b    split_done
find_bang_no:
    mov  x20, x11
    add  x25, x19, x11
    mov  x26, x25
split_done:
    // Phase 2: validate brackets, build jump table
    adrp x21, jump_table
    add  x21, x21, :lo12:jump_table
    adrp x14, bracket_stack
    add  x14, x14, :lo12:bracket_stack
    mov  x12, #0
    mov  x13, #0
val_loop:
    cmp  x12, x20
    b.ge val_done
    ldrb w15, [x19, x12]
    cmp  w15, #0x5B
    b.eq val_open
    cmp  w15, #0x5D
    b.eq val_close
    add  x12, x12, #1
    b    val_loop
val_open:
    cmp  x13, #1000
    b.ge err_exit
    str  w12, [x14, x13, lsl #2]
    add  x13, x13, #1
    add  x12, x12, #1
    b    val_loop
val_close:
    cbz  x13, err_exit
    sub  x13, x13, #1
    ldr  w15, [x14, x13, lsl #2]
    str  w12, [x21, x15, lsl #2]
    str  w15, [x21, x12, lsl #2]
    add  x12, x12, #1
    b    val_loop
val_done:
    cbnz x13, err_exit

    // Phase 3: interpret
    adrp x22, tape
    add  x22, x22, :lo12:tape
    adrp x27, out_buf
    add  x27, x27, :lo12:out_buf
    mov  x23, #0
    mov  x24, #0
    mov  x28, #0

interp_loop:
    cmp  x24, x20
    b.ge interp_done
    ldrb w0, [x19, x24]
    add  x24, x24, #1

    cmp  w0, #0x3E
    b.eq op_gt
    cmp  w0, #0x3C
    b.eq op_lt
    cmp  w0, #0x2B
    b.eq op_plus
    cmp  w0, #0x2D
    b.eq op_minus
    cmp  w0, #0x2E
    b.eq op_dot
    cmp  w0, #0x2C
    b.eq op_comma
    cmp  w0, #0x5B
    b.eq op_open
    cmp  w0, #0x5D
    b.eq op_close
    b    interp_loop

op_gt:
    add  x23, x23, #1
    b    interp_loop
op_lt:
    sub  x23, x23, #1
    b    interp_loop
op_plus:
    ldrb w1, [x22, x23]
    add  w1, w1, #1
    strb w1, [x22, x23]
    b    interp_loop
op_minus:
    ldrb w1, [x22, x23]
    sub  w1, w1, #1
    strb w1, [x22, x23]
    b    interp_loop
op_dot:
    ldrb w1, [x22, x23]
    strb w1, [x27, x28]
    add  x28, x28, #1
    b    interp_loop
op_comma:
    cmp  x25, x26
    b.ge comma_zero
    ldrb w1, [x25], #1
    strb w1, [x22, x23]
    b    interp_loop
comma_zero:
    strb wzr, [x22, x23]
    b    interp_loop
op_open:
    ldrb w1, [x22, x23]
    cbnz w1, interp_loop
    sub  x1, x24, #1
    ldr  w1, [x21, x1, lsl #2]
    add  x24, x1, #1
    b    interp_loop
op_close:
    ldrb w1, [x22, x23]
    cbz  w1, interp_loop
    sub  x1, x24, #1
    ldr  w1, [x21, x1, lsl #2]
    add  x24, x1, #1
    b    interp_loop

interp_done:
    cbz  x28, do_exit
    mov  x8, #SYS_write
    mov  x0, #1
    mov  x1, x27
    mov  x2, x28
    svc  #0
do_exit:
    mov  x8, #SYS_exit
    mov  x0, #0
    svc  #0
err_exit:
    mov  x8, #SYS_exit
    mov  x0, #1
    svc  #0
```
