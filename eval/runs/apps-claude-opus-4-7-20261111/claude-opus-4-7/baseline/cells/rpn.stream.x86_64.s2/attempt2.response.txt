.section .bss
.balign 16
input_buf: .zero 65536
.balign 16
output_buf: .zero 524288
.balign 16
rpn_stack: .zero 8192
.balign 16
num_buf: .zero 32

.text
.globl _start

_start:
    xor r12, r12
.Lread_loop:
    mov rdx, 65536
    sub rdx, r12
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [input_buf + r12]
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    xor r13, r13
    xor rbx, rbx

.Lline_loop:
    cmp rbx, r12
    jae .Lall_done
    mov r14, rbx
.Lfind_nl:
    cmp rbx, r12
    jae .Leol_noeol
    cmp byte ptr [input_buf + rbx], 10
    je .Leol_nl
    inc rbx
    jmp .Lfind_nl
.Leol_nl:
    mov r15, rbx
    inc rbx
    jmp .Lprocess
.Leol_noeol:
    mov r15, rbx

.Lprocess:
    xor ebp, ebp
    mov r8, r14

.Ltok_loop:
    cmp r8, r15
    jae .Ltok_done
    mov al, byte ptr [input_buf + r8]
    cmp al, 32
    je .Lskip_sep
    cmp al, 9
    jne .Lparse_token
.Lskip_sep:
    inc r8
    jmp .Ltok_loop

.Lparse_token:
    mov r9, r8
.Lfind_tok_end:
    cmp r9, r15
    jae .Lgot_token
    mov al, byte ptr [input_buf + r9]
    cmp al, 32
    je .Lgot_token
    cmp al, 9
    je .Lgot_token
    inc r9
    jmp .Lfind_tok_end

.Lgot_token:
    mov rcx, r9
    sub rcx, r8
    cmp rcx, 1
    jne .Lparse_number
    mov al, byte ptr [input_buf + r8]
    cmp al, 43
    je .Ldo_add
    cmp al, 45
    je .Ldo_sub
    cmp al, 42
    je .Ldo_mul
    cmp al, 47
    je .Ldo_div
    cmp al, 37
    je .Ldo_mod

.Lparse_number:
    mov rsi, r8
    xor r10, r10
    xor r11d, r11d
    mov al, byte ptr [input_buf + rsi]
    cmp al, 45
    jne .Lpn_check
    mov r11d, 1
    inc rsi
.Lpn_check:
    mov rcx, r9
    sub rcx, rsi
    test rcx, rcx
    jz .Lline_error
    cmp rcx, 18
    ja .Lline_error
.Lpn_loop:
    cmp rsi, r9
    jae .Lpn_done
    movzx eax, byte ptr [input_buf + rsi]
    sub eax, 48
    cmp eax, 9
    ja .Lline_error
    imul r10, r10, 10
    add r10, rax
    inc rsi
    jmp .Lpn_loop
.Lpn_done:
    test r11d, r11d
    jz .Lpn_push
    neg r10
.Lpn_push:
    mov qword ptr [rpn_stack + rbp*8], r10
    inc rbp
    mov r8, r9
    jmp .Ltok_loop

.Ldo_add:
    cmp rbp, 2
    jb .Lline_error
    mov rax, qword ptr [rpn_stack + rbp*8 - 16]
    add rax, qword ptr [rpn_stack + rbp*8 - 8]
    mov qword ptr [rpn_stack + rbp*8 - 16], rax
    dec rbp
    mov r8, r9
    jmp .Ltok_loop

.Ldo_sub:
    cmp rbp, 2
    jb .Lline_error
    mov rax, qword ptr [rpn_stack + rbp*8 - 16]
    sub rax, qword ptr [rpn_stack + rbp*8 - 8]
    mov qword ptr [rpn_stack + rbp*8 - 16], rax
    dec rbp
    mov r8, r9
    jmp .Ltok_loop

.Ldo_mul:
    cmp rbp, 2
    jb .Lline_error
    mov rax, qword ptr [rpn_stack + rbp*8 - 16]
    imul rax, qword ptr [rpn_stack + rbp*8 - 8]
    mov qword ptr [rpn_stack + rbp*8 - 16], rax
    dec rbp
    mov r8, r9
    jmp .Ltok_loop

.Ldo_div:
    cmp rbp, 2
    jb .Lline_error
    mov rax, qword ptr [rpn_stack + rbp*8 - 16]
    mov rcx, qword ptr [rpn_stack + rbp*8 - 8]
    test rcx, rcx
    jz .Lline_error
    cmp rcx, -1
    jne .Ldiv_ok
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je .Lline_error
.Ldiv_ok:
    cqo
    idiv rcx
    mov qword ptr [rpn_stack + rbp*8 - 16], rax
    dec rbp
    mov r8, r9
    jmp .Ltok_loop

.Ldo_mod:
    cmp rbp, 2
    jb .Lline_error
    mov rax, qword ptr [rpn_stack + rbp*8 - 16]
    mov rcx, qword ptr [rpn_stack + rbp*8 - 8]
    test rcx, rcx
    jz .Lline_error
    cmp rcx, -1
    jne .Lmod_ok
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je .Lline_error
.Lmod_ok:
    cqo
    idiv rcx
    mov qword ptr [rpn_stack + rbp*8 - 16], rdx
    dec rbp
    mov r8, r9
    jmp .Ltok_loop

.Ltok_done:
    cmp rbp, 1
    jne .Lline_error
    mov rax, qword ptr [rpn_stack]
    test rax, rax
    jns .Lfmt_pos
    mov byte ptr [output_buf + r13], 45
    inc r13
    neg rax
.Lfmt_pos:
    lea rsi, [rip + num_buf + 32]
    mov rcx, 10
.Lfmt_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .Lfmt_loop
    lea rdi, [rip + num_buf + 32]
.Lfmt_copy:
    cmp rsi, rdi
    jae .Lfmt_nl
    mov al, byte ptr [rsi]
    mov byte ptr [output_buf + r13], al
    inc r13
    inc rsi
    jmp .Lfmt_copy
.Lfmt_nl:
    mov byte ptr [output_buf + r13], 10
    inc r13
    jmp .Lline_loop

.Lline_error:
    mov byte ptr [output_buf + r13 + 0], 101
    mov byte ptr [output_buf + r13 + 1], 114
    mov byte ptr [output_buf + r13 + 2], 114
    mov byte ptr [output_buf + r13 + 3], 111
    mov byte ptr [output_buf + r13 + 4], 114
    mov byte ptr [output_buf + r13 + 5], 10
    add r13, 6
    jmp .Lline_loop

.Lall_done:
    xor rbx, rbx
.Lwrite_loop:
    cmp rbx, r13
    jae .Lwrite_done
    mov eax, 1
    mov edi, 1
    lea rsi, [output_buf + rbx]
    mov rdx, r13
    sub rdx, rbx
    syscall
    test rax, rax
    jle .Lwrite_done
    add rbx, rax
    jmp .Lwrite_loop
.Lwrite_done:
    mov eax, 60
    xor edi, edi
    syscall
    jmp .Lwrite_done
