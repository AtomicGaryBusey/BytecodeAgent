.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
.zero 49152
outbuf:
.zero 70000

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 49152
    sub rdx, r12
    jz read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbp, [rip + alphabet]
    xor rcx, rcx
    xor rbx, rbx

encode_loop:
    mov r11, r12
    sub r11, rcx
    cmp r11, 3
    jb encode_tail

    movzx r8d, byte ptr [rsi + rcx]
    movzx r9d, byte ptr [rsi + rcx + 1]
    movzx r10d, byte ptr [rsi + rcx + 2]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi], al

    mov eax, r8d
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 1], al

    mov eax, r9d
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 2], al

    mov eax, r10d
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 3], al

    add rdi, 4
    add rcx, 3
    add rbx, 4
    cmp rbx, 76
    jne encode_loop
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rbx, rbx
    jmp encode_loop

encode_tail:
    test r11, r11
    jz final_newline
    cmp r11, 1
    je tail_1

    movzx r8d, byte ptr [rsi + rcx]
    movzx r9d, byte ptr [rsi + rcx + 1]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi], al

    mov eax, r8d
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 1], al

    mov eax, r9d
    shl eax, 2
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 2], al

    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add rbx, 4
    jmp final_newline

tail_1:
    movzx r8d, byte ptr [rsi + rcx]

    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi], al

    mov eax, r8d
    shl eax, 4
    and eax, 0x3F
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [rdi + 1], al

    mov byte ptr [rdi + 2], 0x3D
    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add rbx, 4

final_newline:
    test rbx, rbx
    jz write_output
    mov byte ptr [rdi], 0x0A
    inc rdi

write_output:
    lea rax, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rax
    jz do_exit
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    syscall

do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
