.globl _start

.bss
.p2align 4
input_buf: .zero 32800
lines: .zero 16000
line_count: .zero 8
out_buf: .zero 40000

.text

compare_lines:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9, r9
.Lcmp_loop:
    cmp r9, r8
    jae .Lcmp_eq
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    sub eax, r10d
    jnz .Lcmp_ret
    inc r9
    jmp .Lcmp_loop
.Lcmp_eq:
    mov eax, esi
    sub eax, ecx
.Lcmp_ret:
    ret

sift_down:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    lea rbx, [rip + lines]
.Lsd_loop:
    lea r14, [r12 + r12 + 1]
    cmp r14, r13
    jge .Lsd_done
    mov r15, r14
    lea rax, [r14 + 1]
    cmp rax, r13
    jge .Lsd_check_i
    mov r8, r14
    shl r8, 4
    mov rdi, qword ptr [rbx + r8]
    mov rsi, qword ptr [rbx + r8 + 8]
    lea r8, [r14 + 1]
    shl r8, 4
    mov rdx, qword ptr [rbx + r8]
    mov rcx, qword ptr [rbx + r8 + 8]
    call compare_lines
    test eax, eax
    jge .Lsd_check_i
    lea r15, [r14 + 1]
.Lsd_check_i:
    mov r8, r15
    shl r8, 4
    mov rdi, qword ptr [rbx + r8]
    mov rsi, qword ptr [rbx + r8 + 8]
    mov r8, r12
    shl r8, 4
    mov rdx, qword ptr [rbx + r8]
    mov rcx, qword ptr [rbx + r8 + 8]
    call compare_lines
    test eax, eax
    jle .Lsd_done
    mov r8, r12
    shl r8, 4
    mov r9, r15
    shl r9, 4
    mov rax, qword ptr [rbx + r8]
    mov rcx, qword ptr [rbx + r8 + 8]
    mov rdx, qword ptr [rbx + r9]
    mov rsi, qword ptr [rbx + r9 + 8]
    mov qword ptr [rbx + r8], rdx
    mov qword ptr [rbx + r8 + 8], rsi
    mov qword ptr [rbx + r9], rax
    mov qword ptr [rbx + r9 + 8], rcx
    mov r12, r15
    jmp .Lsd_loop
.Lsd_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

_start:
    xor r12, r12
.Lread_loop:
    mov rdx, 32768
    sub rdx, r12
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbp, [rip + lines]
    lea rbx, [rip + input_buf]
.Lparse_loop:
    cmp r15, r12
    jge .Lparse_last
    cmp byte ptr [rbx + r15], 0x0A
    jne .Lparse_cont
    lea rax, [rbx + r14]
    mov qword ptr [rbp + r13], rax
    mov rax, r15
    sub rax, r14
    mov qword ptr [rbp + r13 + 8], rax
    add r13, 16
    inc r15
    mov r14, r15
    jmp .Lparse_loop
.Lparse_cont:
    inc r15
    jmp .Lparse_loop
.Lparse_last:
    cmp r14, r15
    jge .Lparse_done
    lea rax, [rbx + r14]
    mov qword ptr [rbp + r13], rax
    mov rax, r15
    sub rax, r14
    mov qword ptr [rbp + r13 + 8], rax
    add r13, 16
.Lparse_done:
    mov rax, r13
    shr rax, 4
    mov qword ptr [rip + line_count], rax
    mov rbx, rax
    cmp rbx, 1
    jle .Lsort_done
    mov r14, rbx
    shr r14, 1
    test r14, r14
    jz .Lsort_phase2
    dec r14
.Lbuild_loop:
    mov rdi, r14
    mov rsi, rbx
    call sift_down
    test r14, r14
    jz .Lsort_phase2
    dec r14
    jmp .Lbuild_loop
.Lsort_phase2:
    mov r14, rbx
    dec r14
.Lsort_loop2:
    test r14, r14
    jz .Lsort_done
    lea rbp, [rip + lines]
    mov r8, r14
    shl r8, 4
    mov rax, qword ptr [rbp]
    mov rcx, qword ptr [rbp + 8]
    mov rdx, qword ptr [rbp + r8]
    mov rsi, qword ptr [rbp + r8 + 8]
    mov qword ptr [rbp], rdx
    mov qword ptr [rbp + 8], rsi
    mov qword ptr [rbp + r8], rax
    mov qword ptr [rbp + r8 + 8], rcx
    xor edi, edi
    mov rsi, r14
    call sift_down
    dec r14
    jmp .Lsort_loop2
.Lsort_done:
    mov rbx, qword ptr [rip + line_count]
    xor r13, r13
    lea r12, [rip + lines]
    lea r14, [rip + out_buf]
    xor r15, r15
.Lout_loop:
    cmp r13, rbx
    jge .Lwrite
    mov r8, r13
    shl r8, 4
    mov rsi, qword ptr [r12 + r8]
    mov rcx, qword ptr [r12 + r8 + 8]
    xor r9, r9
.Lcopy_loop:
    cmp r9, rcx
    jge .Lcopy_done
    mov al, byte ptr [rsi + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    inc r9
    jmp .Lcopy_loop
.Lcopy_done:
    mov byte ptr [r14 + r15], 0x0A
    inc r15
    inc r13
    jmp .Lout_loop
.Lwrite:
    test r15, r15
    jz .Lexit
    xor r13, r13
.Lwrite_loop:
    cmp r13, r15
    jge .Lexit
    mov eax, 1
    mov edi, 1
    lea rsi, [r14 + r13]
    mov rdx, r15
    sub rdx, r13
    syscall
    test rax, rax
    jle .Lexit
    add r13, rax
    jmp .Lwrite_loop
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall
