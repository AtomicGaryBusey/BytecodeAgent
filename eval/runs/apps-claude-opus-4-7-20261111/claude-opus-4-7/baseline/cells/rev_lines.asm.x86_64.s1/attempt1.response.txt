.globl _start

.bss
.align 16
inbuf:  .zero 65536
.align 16
outbuf: .zero 65537

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rbx, rbx
    xor r13, r13
    xor rcx, rcx
process_loop:
    cmp rcx, r12
    jae finalize
    mov al, byte ptr [rip + inbuf + rcx]
    cmp al, 0x0A
    je emit_line
    inc rcx
    jmp process_loop
emit_line:
    mov rdi, rcx
reverse_loop:
    cmp rdi, rbx
    jbe end_reverse
    dec rdi
    mov al, byte ptr [rip + inbuf + rdi]
    mov byte ptr [rip + outbuf + r13], al
    inc r13
    jmp reverse_loop
end_reverse:
    mov byte ptr [rip + outbuf + r13], 0x0A
    inc r13
    inc rcx
    mov rbx, rcx
    jmp process_loop
finalize:
    cmp rbx, r12
    jae do_write
    mov rdi, r12
reverse_final:
    cmp rdi, rbx
    jbe end_final
    dec rdi
    mov al, byte ptr [rip + inbuf + rdi]
    mov byte ptr [rip + outbuf + r13], al
    inc r13
    jmp reverse_final
end_final:
    mov byte ptr [rip + outbuf + r13], 0x0A
    inc r13
do_write:
    xor rbx, rbx
write_loop:
    cmp rbx, r13
    jae do_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    syscall
    test rax, rax
    jle do_exit
    add rbx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
