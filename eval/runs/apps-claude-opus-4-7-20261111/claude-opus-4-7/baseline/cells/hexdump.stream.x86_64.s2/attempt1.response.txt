```
.globl _start

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.bss
.align 16
input_buf: .zero 65536
output_buf: .zero 400000

.text
_start:
    lea r12, [rip + input_buf]
    mov r13d, 65536
.Lread_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    sub r13, rax
    jnz .Lread_loop
.Lread_done:
    lea rax, [rip + input_buf]
    sub r12, rax

    lea rdi, [rip + output_buf]
    lea rsi, [rip + input_buf]
    xor ebx, ebx

.Lrow_loop:
    cmp rbx, r12
    jae .Ldone_rows

    mov rbp, r12
    sub rbp, rbx
    cmp rbp, 16
    jbe .Ln_ok
    mov ebp, 16
.Ln_ok:

    mov r10d, 8
.Loff:
    dec r10d
    mov ecx, r10d
    shl ecx, 2
    mov eax, ebx
    shr eax, cl
    and eax, 0xf
    mov al, byte ptr [hex_chars + rax]
    mov [rdi], al
    inc rdi
    test r10d, r10d
    jnz .Loff

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi+1], 0x20
    add rdi, 2

    xor r10d, r10d
.Lbyte_loop:
    cmp r10, rbp
    jae .Lpad
    movzx eax, byte ptr [rsi + r10]
    mov edx, eax
    shr eax, 4
    and edx, 0xf
    mov al, byte ptr [hex_chars + rax]
    mov dl, byte ptr [hex_chars + rdx]
    mov [rdi], al
    mov [rdi+1], dl
    mov byte ptr [rdi+2], 0x20
    jmp .Lbyte_done
.Lpad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi+1], 0x20
    mov byte ptr [rdi+2], 0x20
.Lbyte_done:
    add rdi, 3
    cmp r10, 7
    jne .Lno_mid
    mov byte ptr [rdi], 0x20
    inc rdi
.Lno_mid:
    inc r10
    cmp r10, 16
    jb .Lbyte_loop

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi+1], 0x7c
    add rdi, 2

    xor r10d, r10d
.Lascii_loop:
    cmp r10, rbp
    jae .Lascii_done
    movzx eax, byte ptr [rsi + r10]
    cmp al, 0x20
    jb .Ldot
    cmp al, 0x7e
    ja .Ldot
    jmp .Lstore_char
.Ldot:
    mov al, 0x2e
.Lstore_char:
    mov [rdi], al
    inc rdi
    inc r10
    jmp .Lascii_loop
.Lascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi+1], 0x0a
    add rdi, 2

    add rsi, rbp
    add rbx, rbp
    jmp .Lrow_loop

.Ldone_rows:
    mov r10d, 8
.Lfinal:
    dec r10d
    mov ecx, r10d
    shl ecx, 2
    mov eax, r12d
    shr eax, cl
    and eax, 0xf
    mov al, byte ptr [hex_chars + rax]
    mov [rdi], al
    inc rdi
    test r10d, r10d
    jnz .Lfinal
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
```
