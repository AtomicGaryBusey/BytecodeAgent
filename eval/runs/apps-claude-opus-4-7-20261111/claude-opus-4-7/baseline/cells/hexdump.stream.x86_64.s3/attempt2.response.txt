.globl _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .zero 65536
.align 16
output_buf: .zero 400000

.text
_start:
    xor r12, r12
.Lread_loop:
    mov eax, 65536
    sub rax, r12
    jz .Lread_done
    mov rdx, rax
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:
    lea rbx, [rip + output_buf]
    lea r15, [rip + hex_digits]
    lea r11, [rip + input_buf]
    xor r13, r13
.Lrow_loop:
    cmp r13, r12
    jge .Lfinal

    mov eax, r13d
    mov ecx, 28
.Loff_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r15 + rdx]
    mov [rbx], dl
    inc rbx
    sub ecx, 4
    jns .Loff_loop

    mov word ptr [rbx], 0x2020
    add rbx, 2

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    mov rsi, r11
    add rsi, r13

    xor rcx, rcx
.Lhex_loop:
    cmp rcx, r14
    jae .Lhex_pad_slot
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    and eax, 15
    movzx edx, byte ptr [r15 + rdx]
    movzx eax, byte ptr [r15 + rax]
    mov [rbx], dl
    mov [rbx + 1], al
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp .Lafter_slot
.Lhex_pad_slot:
    mov dword ptr [rbx], 0x20202020
    add rbx, 3
.Lafter_slot:
    cmp rcx, 7
    jne 2f
    mov byte ptr [rbx], 0x20
    inc rbx
2:
    inc rcx
    cmp rcx, 16
    jb .Lhex_loop

    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2

    xor rcx, rcx
.Lascii_loop:
    cmp rcx, r14
    jae .Lascii_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb .Lnot_print
    cmp al, 0x7e
    ja .Lnot_print
    jmp .Lemit_char
.Lnot_print:
    mov al, 0x2e
.Lemit_char:
    mov [rbx], al
    inc rbx
    inc rcx
    jmp .Lascii_loop
.Lascii_done:

    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2

    add r13, 16
    jmp .Lrow_loop

.Lfinal:
    mov eax, r12d
    mov ecx, 28
.Lfoff_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r15 + rdx]
    mov [rbx], dl
    inc rbx
    sub ecx, 4
    jns .Lfoff_loop
    mov byte ptr [rbx], 0x0a
    inc rbx

    lea rsi, [rip + output_buf]
    mov rdx, rbx
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall
