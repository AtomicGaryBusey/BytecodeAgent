.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_end:
.equ msg_ok_len, msg_ok_end - msg_ok

msg_err:
    .ascii "error at "
msg_err_end:
.equ msg_err_len, msg_err_end - msg_err

.section .bss
.balign 16
input_buf:
    .zero 65600
err_buf:
    .zero 64

.text
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0
    mov x25, #65536

read_loop:
    sub x2, x25, x21
    cbz x2, read_done
    mov x8, #63
    mov x0, #0
    add x1, x20, x21
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop
read_done:

    mov x19, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x21
    b.ne error_exit

    mov x8, #64
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #msg_ok_len
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0
    b .

error_exit:
    mov x8, #64
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #msg_err_len
    svc #0

    adrp x22, err_buf
    add x22, x22, :lo12:err_buf
    add x23, x22, #31
    mov w0, #0x0A
    strb w0, [x23]
    sub x24, x23, #1
    mov x0, x19
    mov x1, #10
.Lnum:
    udiv x2, x0, x1
    msub x3, x2, x1, x0
    add w3, w3, #0x30
    strb w3, [x24]
    sub x24, x24, #1
    mov x0, x2
    cbnz x0, .Lnum
    add x24, x24, #1
    mov x8, #64
    mov x0, #1
    mov x1, x24
    sub x2, x23, x24
    add x2, x2, #1
    svc #0
    mov x8, #93
    mov x0, #1
    svc #0
    b .

skip_ws:
.Lws_loop:
    cmp x19, x21
    b.hs .Lws_done
    ldrb w0, [x20, x19]
    cmp w0, #0x20
    b.eq .Lws_adv
    cmp w0, #0x09
    b.eq .Lws_adv
    cmp w0, #0x0A
    b.eq .Lws_adv
    cmp w0, #0x0D
    b.eq .Lws_adv
    b .Lws_done
.Lws_adv:
    add x19, x19, #1
    b .Lws_loop
.Lws_done:
    ret

parse_value:
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x7B
    b.eq parse_object
    cmp w0, #0x5B
    b.eq parse_array
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #0x74
    b.eq parse_true
    cmp w0, #0x66
    b.eq parse_false
    cmp w0, #0x6E
    b.eq parse_null
    cmp w0, #0x2D
    b.eq parse_number
    cmp w0, #0x30
    b.lt error_exit
    cmp w0, #0x39
    b.gt error_exit
    b parse_number

parse_true:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x72
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x75
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x65
    b.ne error_exit
    add x19, x19, #1
    ret

parse_false:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x61
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x6C
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x73
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x65
    b.ne error_exit
    add x19, x19, #1
    ret

parse_null:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x75
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x6C
    b.ne error_exit
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x6C
    b.ne error_exit
    add x19, x19, #1
    ret

parse_string:
    add x19, x19, #1
.Lstr_loop:
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x22
    b.eq .Lstr_end
    cmp w0, #0x5C
    b.eq .Lstr_esc
    cmp w0, #0x20
    b.lt error_exit
    add x19, x19, #1
    b .Lstr_loop
.Lstr_end:
    add x19, x19, #1
    ret
.Lstr_esc:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x22
    b.eq .Lstr_esc_simple
    cmp w0, #0x5C
    b.eq .Lstr_esc_simple
    cmp w0, #0x2F
    b.eq .Lstr_esc_simple
    cmp w0, #0x62
    b.eq .Lstr_esc_simple
    cmp w0, #0x66
    b.eq .Lstr_esc_simple
    cmp w0, #0x6E
    b.eq .Lstr_esc_simple
    cmp w0, #0x72
    b.eq .Lstr_esc_simple
    cmp w0, #0x74
    b.eq .Lstr_esc_simple
    cmp w0, #0x75
    b.eq .Lstr_esc_u
    b error_exit
.Lstr_esc_simple:
    add x19, x19, #1
    b .Lstr_loop
.Lstr_esc_u:
    add x19, x19, #1
    mov w1, #4
.Lstr_u_loop:
    cbz w1, .Lstr_loop
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.lt error_exit
    cmp w0, #0x39
    b.le .Lstr_u_ok
    cmp w0, #0x41
    b.lt error_exit
    cmp w0, #0x46
    b.le .Lstr_u_ok
    cmp w0, #0x61
    b.lt error_exit
    cmp w0, #0x66
    b.gt error_exit
.Lstr_u_ok:
    add x19, x19, #1
    sub w1, w1, #1
    b .Lstr_u_loop

parse_number:
    ldrb w0, [x20, x19]
    cmp w0, #0x2D
    b.ne .Lnum_int
    add x19, x19, #1
.Lnum_int:
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.eq .Lnum_zero
    cmp w0, #0x31
    b.lt error_exit
    cmp w0, #0x39
    b.gt error_exit
    add x19, x19, #1
.Lnum_digits:
    cmp x19, x21
    b.hs .Lnum_done
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.lt .Lnum_check_frac
    cmp w0, #0x39
    b.gt .Lnum_check_frac
    add x19, x19, #1
    b .Lnum_digits
.Lnum_zero:
    add x19, x19, #1
.Lnum_check_frac:
    cmp x19, x21
    b.hs .Lnum_done
    ldrb w0, [x20, x19]
    cmp w0, #0x2E
    b.ne .Lnum_check_exp
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.lt error_exit
    cmp w0, #0x39
    b.gt error_exit
    add x19, x19, #1
.Lnum_frac_digits:
    cmp x19, x21
    b.hs .Lnum_done
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.lt .Lnum_check_exp
    cmp w0, #0x39
    b.gt .Lnum_check_exp
    add x19, x19, #1
    b .Lnum_frac_digits
.Lnum_check_exp:
    cmp x19, x21
    b.hs .Lnum_done
    ldrb w0, [x20, x19]
    cmp w0, #0x65
    b.eq .Lnum_exp
    cmp w0, #0x45
    b.ne .Lnum_done
.Lnum_exp:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
    cmp w0, #0x2B
    b.eq .Lnum_exp_sign
    cmp w0, #0x2D
    b.eq .Lnum_exp_sign
    b .Lnum_exp_first_digit
.Lnum_exp_sign:
    add x19, x19, #1
    cmp x19, x21
    b.hs error_exit
    ldrb w0, [x20, x19]
.Lnum_exp_first_digit:
    cmp w0, #0x30
    b.lt error_exit
    cmp w0, #0x39
    b.gt error_exit
    add x19, x19, #1
.Lnum_exp_digits:
    cmp x19, x21
    b.hs .Lnum_done
    ldrb w0, [x20, x19]
    cmp w0, #0x30
    b.lt .Lnum_done
    cmp w0, #0x39
    b.gt .Lnum_done
    add x19, x19, #1
    b .Lnum_exp_digits
.Lnum_done:
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x21
    b.hs .Lobj_err
    ldrb w0, [x20, x19]
    cmp w0, #0x7D
    b.eq .Lobj_done
.Lobj_member:
    cmp x19, x21
    b.hs .Lobj_err
    ldrb w0, [x20, x19]
    cmp w0, #0x22
    b.ne .Lobj_err
    bl parse_string
    bl skip_ws
    cmp x19, x21
    b.hs .Lobj_err
    ldrb w0, [x20, x19]
    cmp w0, #0x3A
    b.ne .Lobj_err
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x21
    b.hs .Lobj_err
    ldrb w0, [x20, x19]
    cmp w0, #0x2C
    b.ne .Lobj_check_close
    add x19, x19, #1
    bl skip_ws
    b .Lobj_member
.Lobj_check_close:
    cmp w0, #0x7D
    b.ne .Lobj_err
.Lobj_done:
    add x19, x19, #1
    ldp x29, x30, [sp], #16
    ret
.Lobj_err:
    b error_exit

parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x21
    b.hs .Larr_err
    ldrb w0, [x20, x19]
    cmp w0, #0x5D
    b.eq .Larr_done
.Larr_elem:
    bl parse_value
    bl skip_ws
    cmp x19, x21
    b.hs .Larr_err
    ldrb w0, [x20, x19]
    cmp w0, #0x2C
    b.ne .Larr_check_close
    add x19, x19, #1
    bl skip_ws
    b .Larr_elem
.Larr_check_close:
    cmp w0, #0x5D
    b.ne .Larr_err
.Larr_done:
    add x19, x19, #1
    ldp x29, x30, [sp], #16
    ret
.Larr_err:
    b error_exit
