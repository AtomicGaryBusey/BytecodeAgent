_start:
    mov x19, #0                  // total bytes read
    adrp x20, in_buf
    add x20, x20, :lo12:in_buf
    adrp x23, out_buf
    add x23, x23, :lo12:out_buf

    cmp x19, x24
    b.hs read_done
    mov x0, #0                   // fd
    add x1, x20, x19             // buf ptr
    sub x2, x24, x19             // count
    mov x8, #63

    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0                  // start of current line
    mov x22, #0                  // scan index

line_loop:
    cmp x22, x19
    b.hs final_line
    ldrb w0, [x20, x22]
    cmp w0, #0x0A
    b.eq emit_line
    add x22, x22, #1
    b line_loop

emit_line:
    mov x4, #0                   // out index
    mov x5, x22                  // read ptr (exclusive)
rev1:

    b.ls rev1_done
    sub x5, x5, #1

    strb w6, [x23, x4]
    add x4, x4, #1
    b rev1

    strb w6, [x23, x4]
    add x4, x4, #1
    mov x0, #1
    mov x1, x23
    mov x2, x4
    mov x8, #64
    svc #0
    add x22, x22, #1
    mov x21, x22
    b line_loop

final_line:
    cmp x21, x22
    b.eq done
    mov x4, #0
    mov x5, x22

    cmp x5, x21
    b.ls rev2_done
    sub x5, x5, #1
    ldrb w6, [x20, x5]
    strb w6, [x23, x4]

    add x4, x4, #1
    mov x0, #1
    mov x1, x23
    mov x2, x4
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
    b done
