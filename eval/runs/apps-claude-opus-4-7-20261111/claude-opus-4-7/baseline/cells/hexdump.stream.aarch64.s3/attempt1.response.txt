.section .bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 400000

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21

    mov x23, #0

row_loop:
    cmp x23, x20
    b.ge write_final

    sub x24, x20, x23
    cmp x24, #16
    b.le n_ok
    mov x24, #16
n_ok:

    mov x0, x23
    mov x1, x22
    bl write_hex8
    mov x22, x1

    mov w9, #0x20
    strb w9, [x22], #1
    strb w9, [x22], #1

    mov x25, #0
hex_loop:
    cmp x25, #8
    b.ne no_extra
    mov w9, #0x20
    strb w9, [x22], #1
no_extra:
    cmp x25, #16
    b.ge hex_done
    cmp x25, x24
    b.ge pad_slot

    add x9, x19, x23
    ldrb w0, [x9, x25]
    adrp x9, hex_chars
    add x9, x9, :lo12:hex_chars
    lsr w10, w0, #4
    and w10, w10, #0xf
    ldrb w11, [x9, x10]
    strb w11, [x22], #1
    and w10, w0, #0xf
    ldrb w11, [x9, x10]
    strb w11, [x22], #1
    mov w9, #0x20
    strb w9, [x22], #1
    b hex_next
pad_slot:
    mov w9, #0x20
    strb w9, [x22], #1
    strb w9, [x22], #1
    strb w9, [x22], #1
hex_next:
    add x25, x25, #1
    b hex_loop
hex_done:

    mov w9, #0x20
    strb w9, [x22], #1
    mov w9, #0x7c
    strb w9, [x22], #1

    mov x25, #0
char_loop:
    cmp x25, x24
    b.ge char_done
    add x9, x19, x23
    ldrb w0, [x9, x25]
    cmp w0, #0x20
    b.lt dot_char
    cmp w0, #0x7e
    b.gt dot_char
    strb w0, [x22], #1
    b char_next
dot_char:
    mov w9, #0x2e
    strb w9, [x22], #1
char_next:
    add x25, x25, #1
    b char_loop
char_done:

    mov w9, #0x7c
    strb w9, [x22], #1
    mov w9, #0x0a
    strb w9, [x22], #1

    add x23, x23, #16
    b row_loop

write_final:
    mov x0, x20
    mov x1, x22
    bl write_hex8
    mov x22, x1
    mov w9, #0x0a
    strb w9, [x22], #1

    sub x2, x22, x21
    mov x1, x21
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// write_hex8: x0 = value (low 32 bits), x1 = output ptr; returns ptr in x1
write_hex8:
    adrp x2, hex_chars
    add x2, x2, :lo12:hex_chars
    mov x3, #28
hex8_loop:
    lsr x4, x0, x3
    and x4, x4, #0xf
    ldrb w5, [x2, x4]
    strb w5, [x1], #1
    cmp x3, #0
    b.eq hex8_done
    sub x3, x3, #4
    b hex8_loop
hex8_done:
    ret
