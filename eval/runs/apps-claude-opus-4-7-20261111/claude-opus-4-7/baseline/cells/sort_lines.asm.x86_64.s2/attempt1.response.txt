.section .rodata
nl_byte: .byte 0x0A

.bss
.align 16
input_buf: .zero 32784
.align 4
line_off:  .zero 4004
line_len:  .zero 4004
idx_a:     .zero 4004
idx_b:     .zero 4004

.text
.globl _start

compare_lines:
    mov r8d,  dword ptr [rip + line_off + rsi*4]
    mov r9d,  dword ptr [rip + line_len + rsi*4]
    mov r10d, dword ptr [rip + line_off + rdi*4]
    mov r11d, dword ptr [rip + line_len + rdi*4]
    mov rcx, r9
    cmp rcx, r11
    jbe 1f
    mov rcx, r11
1:
    test rcx, rcx
    jz 2f
    lea rax, [rip + input_buf]
    add r8, rax
    add r10, rax
3:
    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r10]
    sub eax, edx
    jne 9f
    inc r8
    inc r10
    dec rcx
    jnz 3b
2:
    mov eax, r9d
    sub eax, r11d
9:
    ret

_start:
    xor r12d, r12d
read_loop:
    mov edx, 32768
    sub edx, r12d
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor ebx, ebx
    xor ebp, ebp
parse_loop:
    cmp rbx, r12
    jae parse_end
    cmp byte ptr [rip + input_buf + rbx], 0x0A
    jne parse_next
    mov dword ptr [rip + line_off + r13*4], ebp
    mov rax, rbx
    sub rax, rbp
    mov dword ptr [rip + line_len + r13*4], eax
    inc r13
    lea rbp, [rbx + 1]
parse_next:
    inc rbx
    jmp parse_loop
parse_end:
    cmp rbp, r12
    jae no_tail
    mov dword ptr [rip + line_off + r13*4], ebp
    mov rax, r12
    sub rax, rbp
    mov dword ptr [rip + line_len + r13*4], eax
    inc r13
no_tail:
    test r13, r13
    jz exit0

    xor ecx, ecx
init_idx:
    mov dword ptr [rip + idx_a + rcx*4], ecx
    inc ecx
    cmp ecx, r13d
    jb init_idx

    lea r12, [rip + idx_a]
    lea rbp, [rip + idx_b]
    mov r14d, 1
msort_outer:
    cmp r14, r13
    jae msort_done
    xor ebx, ebx
pass_loop:
    cmp rbx, r13
    jae pass_end
    mov r15, rbx
    add r15, r14
    cmp r15, r13
    jbe 1f
    mov r15, r13
1:
    mov rax, rbx
    add rax, r14
    add rax, r14
    cmp rax, r13
    jbe 2f
    mov rax, r13
2:
    push rax
    mov rsi, rbx
    mov rdi, r15
    mov rcx, rbx
merge_inner:
    cmp rsi, r15
    jae left_done
    cmp rdi, qword ptr [rsp]
    jae right_done
    push rsi
    push rdi
    push rcx
    mov esi, dword ptr [r12 + rsi*4]
    mov edi, dword ptr [r12 + rdi*4]
    call compare_lines
    pop rcx
    pop rdi
    pop rsi
    test eax, eax
    jg take_right
    mov eax, dword ptr [r12 + rsi*4]
    mov dword ptr [rbp + rcx*4], eax
    inc rsi
    inc rcx
    jmp merge_inner
take_right:
    mov eax, dword ptr [r12 + rdi*4]
    mov dword ptr [rbp + rcx*4], eax
    inc rdi
    inc rcx
    jmp merge_inner
left_done:
    cmp rdi, qword ptr [rsp]
    jae merge_inner_done
    mov eax, dword ptr [r12 + rdi*4]
    mov dword ptr [rbp + rcx*4], eax
    inc rdi
    inc rcx
    jmp left_done
right_done:
    cmp rsi, r15
    jae merge_inner_done
    mov eax, dword ptr [r12 + rsi*4]
    mov dword ptr [rbp + rcx*4], eax
    inc rsi
    inc rcx
    jmp right_done
merge_inner_done:
    pop rbx
    jmp pass_loop
pass_end:
    xchg r12, rbp
    shl r14, 1
    jmp msort_outer
msort_done:
    xor ebx, ebx
out_loop:
    cmp rbx, r13
    jae exit0
    mov eax, dword ptr [r12 + rbx*4]
    mov r15d, dword ptr [rip + line_len + rax*4]
    mov ebp, dword ptr [rip + line_off + rax*4]
    test r15, r15
    jz write_nl
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + input_buf]
    add rsi, rbp
    mov edx, r15d
    syscall
write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_byte]
    mov edx, 1
    syscall
    inc rbx
    jmp out_loop

exit0:
    mov eax, 60
    xor edi, edi
    syscall
