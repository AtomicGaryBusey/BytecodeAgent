```
.globl _start

.section .rodata
.align 16
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:  .zero 49152
.align 16
outbuf: .zero 70000

.text
_start:
    xor r12, r12                    # total bytes read
read_loop:
    mov eax, 0                      # read
    mov edi, 0                      # stdin
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total input bytes
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r11, [rip + alphabet]
    mov rcx, r12
    xor r10, r10                    # chars on current line

encode_loop:
    cmp rcx, 3
    jl encode_tail

    movzx eax, byte ptr [rsi]
    movzx r8d, byte ptr [rsi+1]
    movzx r9d, byte ptr [rsi+2]
    shl eax, 16
    shl r8d, 8
    or eax, r8d
    or eax, r9d

    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], bl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+1], bl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+2], bl

    mov edx, eax
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+3], bl

    add rsi, 3
    add rdi, 4
    sub rcx, 3
    add r10, 4
    cmp r10, 76
    jl no_nl
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10, r10
no_nl:
    jmp encode_loop

encode_tail:
    test rcx, rcx
    jz done_encoding
    cmp rcx, 1
    je tail_1

    # tail_2
    movzx eax, byte ptr [rsi]
    movzx r8d, byte ptr [rsi+1]
    shl eax, 16
    shl r8d, 8
    or eax, r8d

    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], bl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+1], bl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+2], bl

    mov byte ptr [rdi+3], 61
    add rdi, 4
    add r10, 4
    jmp add_final_nl

tail_1:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], bl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov bl, byte ptr [r11 + rdx]
    mov byte ptr [rdi+1], bl

    mov byte ptr [rdi+2], 61
    mov byte ptr [rdi+3], 61
    add rdi, 4
    add r10, 4
    jmp add_final_nl

done_encoding:
    test r12, r12
    jz exit_prog
    test r10, r10
    jz write_output
    mov byte ptr [rdi], 0x0A
    inc rdi
    jmp write_output

add_final_nl:
    mov byte ptr [rdi], 0x0A
    inc rdi

write_output:
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov rdx, rdi
    test rdx, rdx
    jz exit_prog
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
```
