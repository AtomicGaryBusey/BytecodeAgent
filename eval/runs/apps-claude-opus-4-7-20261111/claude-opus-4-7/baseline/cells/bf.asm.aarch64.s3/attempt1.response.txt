.globl _start

.section .bss
.balign 8
input_buf:
    .zero 65536
tape:
    .zero 30000
.balign 4
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
output_buf:
    .zero 65536

.text
_start:
    // Read all of stdin into input_buf, length in x25
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x25, #0
.Lread_loop:
    mov x2, #65536
    sub x2, x2, x25
    cbz x2, .Lread_done
    mov x8, #63
    mov x0, #0
    add x1, x21, x25
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x25, x25, x0
    b .Lread_loop
.Lread_done:

    // Find first '!' (33). x23 = program length, x24 = input position
    mov x23, x25
    mov x24, x25
    mov x9, #0
.Lfind_bang:
    cmp x9, x25
    b.ge .Lno_bang
    ldrb w10, [x21, x9]
    cmp w10, #33
    b.eq .Lfound_bang
    add x9, x9, #1
    b .Lfind_bang
.Lfound_bang:
    mov x23, x9
    add x24, x9, #1
.Lno_bang:

    // Validate brackets and build jump table
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x12, #0
    mov x9, #0
.Lval_loop:
    cmp x9, x23
    b.ge .Lval_done
    ldrb w10, [x21, x9]
    cmp w10, #91
    b.eq .Lval_open
    cmp w10, #93
    b.eq .Lval_close
    b .Lval_next
.Lval_open:
    cmp x12, #1000
    b.ge .Lerror
    str w9, [x11, x12, lsl #2]
    add x12, x12, #1
    b .Lval_next
.Lval_close:
    cbz x12, .Lerror
    sub x12, x12, #1
    ldr w13, [x11, x12, lsl #2]
    add w14, w9, #1
    str w14, [x26, x13, lsl #2]
    add w14, w13, #1
    str w14, [x26, x9, lsl #2]
.Lval_next:
    add x9, x9, #1
    b .Lval_loop
.Lval_done:
    cbnz x12, .Lerror

    // Execute
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x28, output_buf
    add x28, x28, :lo12:output_buf
    mov x27, #0
    mov x19, #0
    mov x20, #0
.Lexec_loop:
    cmp x19, x23
    b.ge .Lexec_done
    ldrb w10, [x21, x19]
    add x19, x19, #1
    cmp w10, #62
    b.eq .Lop_right
    cmp w10, #60
    b.eq .Lop_left
    cmp w10, #43
    b.eq .Lop_plus
    cmp w10, #45
    b.eq .Lop_minus
    cmp w10, #46
    b.eq .Lop_dot
    cmp w10, #44
    b.eq .Lop_comma
    cmp w10, #91
    b.eq .Lop_open
    cmp w10, #93
    b.eq .Lop_close
    b .Lexec_loop
.Lop_right:
    add x20, x20, #1
    b .Lexec_loop
.Lop_left:
    sub x20, x20, #1
    b .Lexec_loop
.Lop_plus:
    ldrb w10, [x22, x20]
    add w10, w10, #1
    strb w10, [x22, x20]
    b .Lexec_loop
.Lop_minus:
    ldrb w10, [x22, x20]
    sub w10, w10, #1
    strb w10, [x22, x20]
    b .Lexec_loop
.Lop_dot:
    ldrb w10, [x22, x20]
    strb w10, [x28, x27]
    add x27, x27, #1
    b .Lexec_loop
.Lop_comma:
    cmp x24, x25
    b.ge .Lcomma_zero
    ldrb w10, [x21, x24]
    add x24, x24, #1
    strb w10, [x22, x20]
    b .Lexec_loop
.Lcomma_zero:
    strb wzr, [x22, x20]
    b .Lexec_loop
.Lop_open:
    ldrb w10, [x22, x20]
    cbnz w10, .Lexec_loop
    sub x9, x19, #1
    ldr w19, [x26, x9, lsl #2]
    b .Lexec_loop
.Lop_close:
    ldrb w10, [x22, x20]
    cbz w10, .Lexec_loop
    sub x9, x19, #1
    ldr w19, [x26, x9, lsl #2]
    b .Lexec_loop
.Lexec_done:
    cbz x27, .Lskip_write
    mov x8, #64
    mov x0, #1
    mov x1, x28
    mov x2, x27
    svc #0
.Lskip_write:
    mov x8, #93
    mov x0, #0
    svc #0
.Lerror:
    mov x8, #93
    mov x0, #1
    svc #0
