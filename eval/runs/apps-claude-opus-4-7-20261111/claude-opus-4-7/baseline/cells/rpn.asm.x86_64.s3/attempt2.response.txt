.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536
.align 16
stack_area:
    .zero 8192

.section .rodata
err_str:
    .ascii "error\n"

.text
_start:
    xor r12, r12
.Lrl:
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz .Ldr
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle .Ldr
    add r12, rax
    jmp .Lrl
.Ldr:
    lea r13, [rip + input_buf]
    lea r14, [r13 + r12]
.Lpr:
    cmp r13, r14
    jae .Lad
    mov r15, r13
.Lfn:
    cmp r15, r14
    jae .Leo
    cmp byte ptr [r15], 0x0A
    je .Leo
    inc r15
    jmp .Lfn
.Leo:
    call eval_line
    cmp r15, r14
    jae .Lad
    lea r13, [r15 + 1]
    jmp .Lpr
.Lad:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    push rbx
    push rbp
    xor ebx, ebx
    mov rbp, r13
.Ltl:
    cmp rbp, r15
    jae .Lte
    mov al, [rbp]
    cmp al, 0x20
    je .Lts
    cmp al, 0x09
    je .Lts
    mov r8, rbp
.Lfe:
    inc rbp
    cmp rbp, r15
    jae .Lh
    mov al, [rbp]
    cmp al, 0x20
    je .Lh
    cmp al, 0x09
    je .Lh
    jmp .Lfe
.Lts:
    inc rbp
    jmp .Ltl
.Lh:
    mov rcx, rbp
    sub rcx, r8
    cmp rcx, 1
    jne .Lt
    mov al, [r8]
    cmp al, 0x2B
    je .Loa
    cmp al, 0x2D
    je .Los
    cmp al, 0x2A
    je .Lom
    cmp al, 0x2F
    je .Lod
    cmp al, 0x25
    je .Lop
.Lt:
    xor r9d, r9d
    mov rsi, r8
    cmp byte ptr [rsi], 0x2D
    jne .Lnn
    mov r9d, 1
    inc rsi
    dec rcx
.Lnn:
    test rcx, rcx
    jz .Ler
    cmp rcx, 18
    ja .Ler
    xor edx, edx
    mov r10, rsi
    add r10, rcx
.Ldl:
    cmp rsi, r10
    jae .Ldd
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 9
    ja .Ler
    lea rdx, [rdx + rdx*4]
    lea rdx, [rax + rdx*2]
    inc rsi
    jmp .Ldl
.Ldd:
    test r9d, r9d
    jz .Lpu
    neg rdx
.Lpu:
    cmp rbx, 1024
    jae .Ler
    mov [stack_area + rbx*8], rdx
    inc rbx
    jmp .Ltl

.Loa:
    cmp rbx, 2
    jb .Ler
    mov rax, [stack_area + rbx*8 - 8]
    add [stack_area + rbx*8 - 16], rax
    dec rbx
    jmp .Ltl
.Los:
    cmp rbx, 2
    jb .Ler
    mov rax, [stack_area + rbx*8 - 8]
    sub [stack_area + rbx*8 - 16], rax
    dec rbx
    jmp .Ltl
.Lom:
    cmp rbx, 2
    jb .Ler
    mov rax, [stack_area + rbx*8 - 8]
    imul rax, [stack_area + rbx*8 - 16]
    mov [stack_area + rbx*8 - 16], rax
    dec rbx
    jmp .Ltl
.Lod:
    cmp rbx, 2
    jb .Ler
    mov r11, [stack_area + rbx*8 - 8]
    mov rax, [stack_area + rbx*8 - 16]
    test r11, r11
    jz .Ler
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne .Lok1
    cmp r11, -1
    je .Ler
.Lok1:
    cqo
    idiv r11
    mov [stack_area + rbx*8 - 16], rax
    dec rbx
    jmp .Ltl
.Lop:
    cmp rbx, 2
    jb .Ler
    mov r11, [stack_area + rbx*8 - 8]
    mov rax, [stack_area + rbx*8 - 16]
    test r11, r11
    jz .Ler
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne .Lok2
    cmp r11, -1
    je .Ler
.Lok2:
    cqo
    idiv r11
    mov [stack_area + rbx*8 - 16], rdx
    dec rbx
    jmp .Ltl

.Lte:
    cmp rbx, 1
    jne .Ler
    mov rax, [stack_area]
    call print_int
    jmp .Ldn
.Ler:
    lea rsi, [rip + err_str]
    mov edx, 6
    mov eax, 1
    mov edi, 1
    syscall
.Ldn:
    pop rbp
    pop rbx
    ret

print_int:
    sub rsp, 32
    lea r8, [rsp + 31]
    mov byte ptr [r8], 0x0A
    dec r8
    test rax, rax
    jnz 1f
    mov byte ptr [r8], 0x30
    dec r8
    jmp 3f
1:
    mov r9, rax
    jns 2f
    neg rax
2:
    mov ecx, 10
5:
    xor edx, edx
    div rcx
    add dl, 0x30
    mov [r8], dl
    dec r8
    test rax, rax
    jnz 5b
    test r9, r9
    jns 3f
    mov byte ptr [r8], 0x2D
    dec r8
3:
    inc r8
    lea rdx, [rsp + 32]
    sub rdx, r8
    mov rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    add rsp, 32
    ret
