```
.globl _start

.section .bss
.align 16
buffer: .zero 65536

.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at "

.text

_start:
    lea r14, [rip + buffer]
    mov rbx, r14
    mov r15d, 65536
read_loop:
    test r15, r15
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r15
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    sub r15, rax
    jmp read_loop
read_done:
    mov r13, rbx
    mov rbx, r14

    call skip_ws
    cmp rbx, r13
    je fail
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne fail

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

fail:
    sub rbx, r14
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov rax, rbx
    sub rsp, 64
    lea rdi, [rsp + 32]
    mov byte ptr [rdi], 0x0A
    lea r8, [rsp + 32]
    test rax, rax
    jnz digit_loop
    dec rdi
    mov byte ptr [rdi], 0x30
    jmp print_num
digit_loop:
    test rax, rax
    jz print_num
    mov rcx, 10
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    jmp digit_loop
print_num:
    mov rax, r8
    sub rax, rdi
    inc rax
    mov rdx, rax
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
.sw_loop:
    cmp rbx, r13
    je .sw_ret
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je .sw_adv
    cmp al, 0x09
    je .sw_adv
    cmp al, 0x0A
    je .sw_adv
    cmp al, 0x0D
    je .sw_adv
.sw_ret:
    ret
.sw_adv:
    inc rbx
    jmp .sw_loop

parse_value:
    movzx eax, byte ptr [rbx]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    jmp parse_number

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x7D
    je .po_close
.po_member:
    cmp byte ptr [rbx], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x3A
    jne fail
    inc rbx
    call skip_ws
    cmp rbx, r13
    je fail
    call parse_value
    call skip_ws
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x7D
    je .po_close
    cmp byte ptr [rbx], 0x2C
    jne fail
    inc rbx
    call skip_ws
    cmp rbx, r13
    je fail
    jmp .po_member
.po_close:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x5D
    je .pa_close
.pa_elem:
    call parse_value
    call skip_ws
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x5D
    je .pa_close
    cmp byte ptr [rbx], 0x2C
    jne fail
    inc rbx
    call skip_ws
    cmp rbx, r13
    je fail
    jmp .pa_elem
.pa_close:
    inc rbx
    ret

parse_string:
    inc rbx
.ps_loop:
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x22
    je .ps_end
    cmp al, 0x5C
    je .ps_esc
    cmp al, 0x20
    jb fail
    inc rbx
    jmp .ps_loop
.ps_end:
    inc rbx
    ret
.ps_esc:
    inc rbx
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x22
    je .ps_esc_ok
    cmp al, 0x5C
    je .ps_esc_ok
    cmp al, 0x2F
    je .ps_esc_ok
    cmp al, 0x62
    je .ps_esc_ok
    cmp al, 0x66
    je .ps_esc_ok
    cmp al, 0x6E
    je .ps_esc_ok
    cmp al, 0x72
    je .ps_esc_ok
    cmp al, 0x74
    je .ps_esc_ok
    cmp al, 0x75
    je .ps_u
    jmp fail
.ps_esc_ok:
    inc rbx
    jmp .ps_loop
.ps_u:
    inc rbx
    mov ecx, 4
.ps_u_loop:
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe .ps_u_ok
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    jbe .ps_u_ok
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
.ps_u_ok:
    inc rbx
    dec ecx
    jnz .ps_u_loop
    jmp .ps_loop

parse_number:
    cmp byte ptr [rbx], 0x2D
    jne .pn_no_sign
    inc rbx
    cmp rbx, r13
    je fail
.pn_no_sign:
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    je .pn_int_zero
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
.pn_int_digits:
    cmp rbx, r13
    je .pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb .pn_frac
    cmp al, 0x39
    ja .pn_frac
    inc rbx
    jmp .pn_int_digits
.pn_int_zero:
    inc rbx
.pn_frac:
    cmp rbx, r13
    je .pn_done
    cmp byte ptr [rbx], 0x2E
    jne .pn_exp
    inc rbx
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
.pn_frac_digits:
    cmp rbx, r13
    je .pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb .pn_exp
    cmp al, 0x39
    ja .pn_exp
    inc rbx
    jmp .pn_frac_digits
.pn_exp:
    cmp rbx, r13
    je .pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x65
    je .pn_exp_start
    cmp al, 0x45
    je .pn_exp_start
    jmp .pn_done
.pn_exp_start:
    inc rbx
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
    cmp al, 0x2B
    je .pn_exp_sign
    cmp al, 0x2D
    je .pn_exp_sign
    jmp .pn_exp_digit_check
.pn_exp_sign:
    inc rbx
    cmp rbx, r13
    je fail
    movzx eax, byte ptr [rbx]
.pn_exp_digit_check:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rbx
.pn_exp_digits:
    cmp rbx, r13
    je .pn_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x30
    jb .pn_done
    cmp al, 0x39
    ja .pn_done
    inc rbx
    jmp .pn_exp_digits
.pn_done:
    ret

parse_true:
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x72
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x75
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x65
    jne fail
    inc rbx
    ret

parse_false:
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x61
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x6C
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x73
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x65
    jne fail
    inc rbx
    ret

parse_null:
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x75
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x6C
    jne fail
    inc rbx
    cmp rbx, r13
    je fail
    cmp byte ptr [rbx], 0x6C
    jne fail
    inc rbx
    ret
```
