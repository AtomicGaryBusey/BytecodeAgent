.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:  .zero 49152
.align 16
outbuf: .zero 70000

.text
_start:
    lea r12, [rip + inbuf]
    mov r13, r12
read_loop:
    mov rax, 0
    xor edi, edi
    mov rsi, r13
    mov rdx, r12
    add rdx, 49152
    sub rdx, r13
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r14, r13
    sub r14, r12
    test r14, r14
    jz do_exit

    lea rbx, [rip + outbuf]
    mov r15, rbx
    xor r10d, r10d
    lea rcx, [rip + alphabet]
    mov rsi, r12
    mov rdi, r13

encode_loop:
    mov rax, rdi
    sub rax, rsi
    cmp rax, 3
    jl tail

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx eax, byte ptr [rsi + 2]
    add rsi, 3

    mov edx, r8d
    shr edx, 2
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx], dl

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov r11d, r9d
    shr r11d, 4
    or edx, r11d
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx + 1], dl

    mov edx, r9d
    and edx, 0xf
    shl edx, 2
    mov r11d, eax
    shr r11d, 6
    or edx, r11d
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx + 2], dl

    and eax, 0x3f
    mov al, byte ptr [rcx + rax]
    mov byte ptr [rbx + 3], al

    add rbx, 4
    add r10d, 4
    cmp r10d, 76
    jl encode_loop

    mov byte ptr [rbx], 0x0a
    inc rbx
    xor r10d, r10d
    jmp encode_loop

tail:
    test eax, eax
    jz final_newline
    cmp eax, 1
    je tail1

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]

    mov edx, r8d
    shr edx, 2
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx], dl

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov r11d, r9d
    shr r11d, 4
    or edx, r11d
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx + 1], dl

    mov edx, r9d
    and edx, 0xf
    shl edx, 2
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx + 2], dl

    mov byte ptr [rbx + 3], 0x3d
    add rbx, 4
    jmp emit_nl

tail1:
    movzx r8d, byte ptr [rsi]
    mov edx, r8d
    shr edx, 2
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx], dl

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov dl, byte ptr [rcx + rdx]
    mov byte ptr [rbx + 1], dl

    mov byte ptr [rbx + 2], 0x3d
    mov byte ptr [rbx + 3], 0x3d
    add rbx, 4
    jmp emit_nl

final_newline:
    test r10d, r10d
    jz do_write
emit_nl:
    mov byte ptr [rbx], 0x0a
    inc rbx

do_write:
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, rbx
    sub rdx, r15
    syscall

do_exit:
    mov rax, 60
    xor edi, edi
    syscall
