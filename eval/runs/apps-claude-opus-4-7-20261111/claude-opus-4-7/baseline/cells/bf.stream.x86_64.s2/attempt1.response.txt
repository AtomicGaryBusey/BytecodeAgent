.bss
.align 8
input_buf:
    .zero 65536
.align 8
tape:
    .zero 30000
.align 8
output_buf:
    .zero 65536
.align 8
bracket_stack:
    .zero 4000
.align 8
jump_table:
    .zero 262144

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
.Lread_loop:
    mov edx, 65536
    sub edx, r12d
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    add rsi, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:
    xor ecx, ecx
.Lfind:
    cmp rcx, r12
    jge .Lno_bang
    cmp byte ptr [rbx + rcx], 33
    je .Lfound_bang
    inc rcx
    jmp .Lfind
.Lno_bang:
    mov r13, r12
    lea r15, [rip + input_buf]
    xor r14d, r14d
    jmp .Lvalidate
.Lfound_bang:
    mov r13, rcx
    lea r15, [rbx + rcx + 1]
    mov r14, r12
    sub r14, rcx
    dec r14
.Lvalidate:
    lea rcx, [rip + bracket_stack]
    xor edx, edx
    xor esi, esi
.Lvloop:
    cmp rsi, r13
    jge .Lvdone
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 91
    je .Lvopen
    cmp al, 93
    je .Lvclose
    inc rsi
    jmp .Lvloop
.Lvopen:
    mov dword ptr [rcx + rdx*4], esi
    inc edx
    inc rsi
    jmp .Lvloop
.Lvclose:
    test edx, edx
    jz .Lerr
    dec edx
    mov edi, dword ptr [rcx + rdx*4]
    lea eax, [rsi + 1]
    mov dword ptr [jump_table + rdi*4], eax
    lea eax, [rdi + 1]
    mov dword ptr [jump_table + rsi*4], eax
    inc rsi
    jmp .Lvloop
.Lvdone:
    test edx, edx
    jnz .Lerr

    lea r12, [rip + tape]
    xor ebp, ebp
    lea r8, [rip + output_buf]
    xor r9d, r9d

.Lexec:
    cmp rbp, r13
    jge .Lfinish
    movzx eax, byte ptr [rbx + rbp]
    inc rbp
    cmp al, 62
    je .Lgt
    cmp al, 60
    je .Llt
    cmp al, 43
    je .Lplus
    cmp al, 45
    je .Lminus
    cmp al, 46
    je .Ldot
    cmp al, 44
    je .Lcomma
    cmp al, 91
    je .Lopen
    cmp al, 93
    je .Lclose
    jmp .Lexec

.Lgt:
    inc r12
    jmp .Lexec
.Llt:
    dec r12
    jmp .Lexec
.Lplus:
    inc byte ptr [r12]
    jmp .Lexec
.Lminus:
    dec byte ptr [r12]
    jmp .Lexec
.Ldot:
    mov al, byte ptr [r12]
    mov byte ptr [r8 + r9], al
    inc r9d
    jmp .Lexec
.Lcomma:
    test r14, r14
    jz .Lcomma_zero
    mov al, byte ptr [r15]
    inc r15
    dec r14
    mov byte ptr [r12], al
    jmp .Lexec
.Lcomma_zero:
    mov byte ptr [r12], 0
    jmp .Lexec
.Lopen:
    cmp byte ptr [r12], 0
    jne .Lexec
    dec rbp
    mov ebp, dword ptr [jump_table + rbp*4]
    jmp .Lexec
.Lclose:
    cmp byte ptr [r12], 0
    je .Lexec
    dec rbp
    mov ebp, dword ptr [jump_table + rbp*4]
    jmp .Lexec

.Lfinish:
    test r9d, r9d
    jz .Lexit
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov edx, r9d
    syscall
.Lexit:
    mov eax, 231
    xor edi, edi
    syscall

.Lerr:
    mov eax, 231
    mov edi, 1
    syscall
