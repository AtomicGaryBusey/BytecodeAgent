.globl _start

.section .rodata
ws_table:
    .byte 0,0,0,0,0,0,0,0,0
    .byte 1,1,1,1,1

.bss
.align 16
inbuf:
    .zero 65536
.align 16
outbuf:
    .zero 80

.text
_start:
    xor r12, r12
read_loop:
    mov edx, 65536
    sub rdx, r12
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [inbuf + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r8, r8
    xor r9, r9
    xor r10d, r10d
    xor ecx, ecx
count_loop:
    cmp rcx, r12
    jae count_done
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 0x0A
    jne 1f
    inc r8
1:
    movzx edx, byte ptr [ws_table + rax]
    test edx, edx
    jz 4f
    xor r10d, r10d
    jmp 3f
4:
    test r10d, r10d
    jnz 3f
    inc r9
    mov r10d, 1
3:
    inc rcx
    jmp count_loop
count_done:
    lea rdi, [outbuf + 80]
    dec rdi
    mov byte ptr [rdi], 0x0A

    mov rax, r12
    call itoa
    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r9
    call itoa
    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r8
    call itoa

    mov rsi, rdi
    lea rax, [outbuf + 80]
    sub rax, rdi
    mov rdx, rax
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

itoa:
    test rax, rax
    jnz itoa_loop
    dec rdi
    mov byte ptr [rdi], 0x30
    ret
itoa_loop:
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz itoa_loop
    ret
