```
.globl _start

.bss
.balign 16
input_buf: .zero 65536
output_buf: .zero 400000
rpn_stack: .zero 4096
num_buf: .zero 32

.text
_start:
    xor r14, r14
rd_loop:
    mov rdx, 65536
    sub rdx, r14
    jbe rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r14
    syscall
    test rax, rax
    jle rd_done
    add r14, rax
    jmp rd_loop
rd_done:
    lea r13, [rip + input_buf]
    add r13, r14
    lea r15, [rip + input_buf]
    lea r12, [rip + output_buf]

ml_loop:
    cmp r15, r13
    jae ml_done
    mov rcx, r15
f_nl:
    cmp rcx, r13
    jae no_nl
    cmp byte ptr [rcx], 0x0A
    je y_nl
    inc rcx
    jmp f_nl
y_nl:
    mov r10, rcx
    lea rbp, [rcx + 1]
    jmp ev_line
no_nl:
    mov r10, r13
    mov rbp, r13
ev_line:
    lea r11, [rip + rpn_stack]
    mov rsi, r15

tk_loop:
sp_skip:
    cmp rsi, r10
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je sp_next
    cmp al, 0x09
    jne hv_tok
sp_next:
    inc rsi
    jmp sp_skip
hv_tok:
    mov rdi, rsi
te_scan:
    cmp rdi, r10
    jae te_got
    movzx eax, byte ptr [rdi]
    cmp al, 0x20
    je te_got
    cmp al, 0x09
    je te_got
    inc rdi
    jmp te_scan
te_got:
    mov r9, rdi
    sub r9, rsi
    cmp r9, 1
    jne try_num
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B
    je op_add
    cmp al, 0x2D
    je op_sub
    cmp al, 0x2A
    je op_mul
    cmp al, 0x2F
    je op_div
    cmp al, 0x25
    je op_mod

try_num:
    mov rcx, rsi
    xor r8d, r8d
    cmp byte ptr [rcx], 0x2D
    jne no_neg
    mov r8d, 1
    inc rcx
no_neg:
    mov rbx, rcx
    xor eax, eax
dg_loop:
    cmp rcx, rdi
    jae dg_end
    movzx edx, byte ptr [rcx]
    sub dl, 0x30
    cmp dl, 9
    ja err
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    jmp dg_loop
dg_end:
    mov rdx, rcx
    sub rdx, rbx
    test rdx, rdx
    jz err
    cmp rdx, 18
    ja err
    test r8d, r8d
    jz pn
    neg rax
pn:
    mov qword ptr [r11], rax
    add r11, 8
    mov rsi, rdi
    jmp tk_loop

op_add:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 16
    jb err
    mov rax, qword ptr [r11 - 8]
    add qword ptr [r11 - 16], rax
    sub r11, 8
    mov rsi, rdi
    jmp tk_loop

op_sub:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 16
    jb err
    mov rax, qword ptr [r11 - 8]
    sub qword ptr [r11 - 16], rax
    sub r11, 8
    mov rsi, rdi
    jmp tk_loop

op_mul:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 16
    jb err
    mov rax, qword ptr [r11 - 8]
    imul rax, qword ptr [r11 - 16]
    mov qword ptr [r11 - 16], rax
    sub r11, 8
    mov rsi, rdi
    jmp tk_loop

op_div:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 16
    jb err
    mov rcx, qword ptr [r11 - 8]
    test rcx, rcx
    jz err
    mov rax, qword ptr [r11 - 16]
    cmp rcx, -1
    jne div_do
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je err
div_do:
    cqo
    idiv rcx
    mov qword ptr [r11 - 16], rax
    sub r11, 8
    mov rsi, rdi
    jmp tk_loop

op_mod:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 16
    jb err
    mov rcx, qword ptr [r11 - 8]
    test rcx, rcx
    jz err
    mov rax, qword ptr [r11 - 16]
    cmp rcx, -1
    jne mod_do
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je err
mod_do:
    cqo
    idiv rcx
    mov qword ptr [r11 - 16], rdx
    sub r11, 8
    mov rsi, rdi
    jmp tk_loop

line_end:
    lea rax, [rip + rpn_stack]
    mov rcx, r11
    sub rcx, rax
    cmp rcx, 8
    jne err
    mov rax, qword ptr [r11 - 8]
    test rax, rax
    jns pos
    mov byte ptr [r12], 0x2D
    inc r12
    neg rax
pos:
    lea rbx, [rip + num_buf + 31]
    mov rsi, rbx
nl_loop:
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, 0x30
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz nl_loop
    inc rsi
cp_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [r12], al
    inc r12
    cmp rsi, rbx
    je cp_done
    inc rsi
    jmp cp_loop
cp_done:
    mov byte ptr [r12], 0x0A
    inc r12
    jmp nx_line

err:
    mov byte ptr [r12], 0x65
    mov byte ptr [r12+1], 0x72
    mov byte ptr [r12+2], 0x72
    mov byte ptr [r12+3], 0x6F
    mov byte ptr [r12+4], 0x72
    mov byte ptr [r12+5], 0x0A
    add r12, 6
nx_line:
    mov r15, rbp
    jmp ml_loop

ml_done:
    lea rsi, [rip + output_buf]
    mov rdx, r12
    sub rdx, rsi
wr_loop:
    test rdx, rdx
    jz exit_p
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_p
    add rsi, rax
    sub rdx, rax
    jmp wr_loop
exit_p:
    mov eax, 60
    xor edi, edi
    syscall
```
