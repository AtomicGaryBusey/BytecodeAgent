.globl _start

.section .bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 400000

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    cmp x20, x2
    b.ge read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    sub x2, x2, x20
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, #0
    adrp x24, hex_digits
    add x24, x24, :lo12:hex_digits

row_loop:
    cmp x23, x20
    b.ge write_final

    sub x25, x20, x23
    cmp x25, #16
    b.le 1f
    mov x25, #16
1:
    mov x9, #28
offset_hex_loop:
    lsr x10, x23, x9
    and x10, x10, #0xf
    ldrb w11, [x24, x10]
    strb w11, [x22], #1
    subs x9, x9, #4
    b.ge offset_hex_loop

    mov w9, #0x20
    strb w9, [x22], #1
    strb w9, [x22], #1

    mov x27, #0
byte_loop:
    cmp x27, #16
    b.eq ascii_start

    cmp x27, x25
    b.ge pad_byte

    add x28, x19, x23
    ldrb w11, [x28, x27]
    lsr x12, x11, #4
    ldrb w12, [x24, x12]
    strb w12, [x22], #1
    and x12, x11, #0xf
    ldrb w12, [x24, x12]
    strb w12, [x22], #1
    mov w9, #0x20
    strb w9, [x22], #1
    b after_byte

pad_byte:
    mov w9, #0x20
    strb w9, [x22], #1
    strb w9, [x22], #1
    strb w9, [x22], #1

after_byte:
    add x27, x27, #1
    cmp x27, #8
    b.ne byte_loop
    mov w9, #0x20
    strb w9, [x22], #1
    b byte_loop

ascii_start:
    mov w9, #0x20
    strb w9, [x22], #1
    mov w9, #0x7c
    strb w9, [x22], #1

    mov x27, #0
ascii_loop:
    cmp x27, x25
    b.eq ascii_end
    add x28, x19, x23
    ldrb w11, [x28, x27]
    cmp w11, #0x20
    b.lt ascii_dot
    cmp w11, #0x7e
    b.gt ascii_dot
    b ascii_write
ascii_dot:
    mov w11, #0x2e
ascii_write:
    strb w11, [x22], #1
    add x27, x27, #1
    b ascii_loop

ascii_end:
    mov w9, #0x7c
    strb w9, [x22], #1
    mov w9, #0x0a
    strb w9, [x22], #1

    add x23, x23, #16
    b row_loop

write_final:
    mov x9, #28
final_hex_loop:
    lsr x10, x20, x9
    and x10, x10, #0xf
    ldrb w11, [x24, x10]
    strb w11, [x22], #1
    subs x9, x9, #4
    b.ge final_hex_loop
    mov w9, #0x0a
    strb w9, [x22], #1

    sub x2, x22, x21
    mov x8, #64
    mov x0, #1
    mov x1, x21
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
