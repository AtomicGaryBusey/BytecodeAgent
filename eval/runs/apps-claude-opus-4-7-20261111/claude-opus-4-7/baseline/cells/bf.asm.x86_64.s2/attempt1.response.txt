```
.globl _start

.section .bss
.balign 8
input_buf: .zero 65536
ops_buf:   .zero 65536
.balign 4
jump_table: .zero 262144
bracket_stack: .zero 8000
tape: .zero 30000
output_buf: .zero 65536

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    xor ecx, ecx
find_bang:
    cmp ecx, ebx
    je no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc ecx
    jmp find_bang
no_bang:
    mov r12d, ebx
    lea r13, [rsi + rbx]
    xor r14d, r14d
    jmp build_start
found_bang:
    mov r12d, ecx
    lea r13, [rsi + rcx + 1]
    mov r14d, ebx
    sub r14d, ecx
    dec r14d
build_start:
    lea rdi, [rip + ops_buf]
    lea r9, [rip + jump_table]
    lea r11, [rip + bracket_stack]
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d
build_loop:
    cmp edx, r12d
    je build_done
    movzx eax, byte ptr [rsi + rdx]
    inc edx
    cmp al, 0x3E
    je store_op
    cmp al, 0x3C
    je store_op
    cmp al, 0x2B
    je store_op
    cmp al, 0x2D
    je store_op
    cmp al, 0x2E
    je store_op
    cmp al, 0x2C
    je store_op
    cmp al, 0x5B
    je push_br
    cmp al, 0x5D
    je pop_br
    jmp build_loop
store_op:
    mov [rdi + r8], al
    inc r8d
    jmp build_loop
push_br:
    mov [rdi + r8], al
    mov [r11 + rcx*4], r8d
    inc ecx
    inc r8d
    jmp build_loop
pop_br:
    test ecx, ecx
    jz err
    dec ecx
    mov eax, [r11 + rcx*4]
    mov byte ptr [rdi + r8], 0x5D
    mov [r9 + rax*4], r8d
    mov [r9 + r8*4], eax
    inc r8d
    jmp build_loop
build_done:
    test ecx, ecx
    jnz err

    mov r12d, r8d
    mov rsi, rdi
    mov rdx, r9
    lea rbx, [rip + tape]
    lea r15, [rip + output_buf]
    mov rbp, r15
    xor ecx, ecx
interp_loop:
    cmp ecx, r12d
    jae interp_done
    movzx eax, byte ptr [rsi + rcx]
    inc ecx
    cmp al, 0x3E
    je op_gt
    cmp al, 0x3C
    je op_lt
    cmp al, 0x2B
    je op_plus
    cmp al, 0x2D
    je op_minus
    cmp al, 0x2E
    je op_dot
    cmp al, 0x2C
    je op_comma
    cmp al, 0x5B
    je op_lbracket
op_rbracket:
    cmp byte ptr [rbx], 0
    je interp_loop
    mov ecx, [rdx + rcx*4 - 4]
    inc ecx
    jmp interp_loop
op_gt:
    inc rbx
    jmp interp_loop
op_lt:
    dec rbx
    jmp interp_loop
op_plus:
    inc byte ptr [rbx]
    jmp interp_loop
op_minus:
    dec byte ptr [rbx]
    jmp interp_loop
op_dot:
    mov al, [rbx]
    mov [rbp], al
    inc rbp
    jmp interp_loop
op_comma:
    test r14d, r14d
    jz comma_zero
    mov al, [r13]
    mov [rbx], al
    inc r13
    dec r14d
    jmp interp_loop
comma_zero:
    mov byte ptr [rbx], 0
    jmp interp_loop
op_lbracket:
    cmp byte ptr [rbx], 0
    jne interp_loop
    mov ecx, [rdx + rcx*4 - 4]
    inc ecx
    jmp interp_loop
interp_done:
    sub rbp, r15
    jz do_exit0
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, rbp
    syscall
do_exit0:
    mov eax, 60
    xor edi, edi
    syscall
err:
    mov eax, 60
    mov edi, 1
    syscall
```
