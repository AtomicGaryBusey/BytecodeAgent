    jz read_done

    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall

    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor rbx, rbx
    lea r14, [rip + input_buf]
    lea r15, [rip + output_buf]
process:
    cmp rbx, r12
    jae process_done
    mov rcx, rbx
find_nl:

    jae found_end
    cmp byte ptr [r14 + rcx], 0x0A

    inc rcx
    jmp find_nl

    inc r13
    jmp rev1
rev1_done:

    inc r13
    lea rbx, [rcx + 1]
    jmp process
found_end:
    cmp rbx, r12
    jae process_done
    mov rdx, r12
rev2:
    cmp rdx, rbx
    je rev2_done
    dec rdx
    mov al, byte ptr [r14 + rdx]

    mov rsi, r15
    mov rbx, r13
write_loop:
    test rbx, rbx
    jz do_exit
    mov rdx, rbx
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rbx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit
