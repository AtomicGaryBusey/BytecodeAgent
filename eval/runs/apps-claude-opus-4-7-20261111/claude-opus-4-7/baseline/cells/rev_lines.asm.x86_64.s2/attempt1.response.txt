```
.globl _start

.bss
.balign 16
inbuf: .zero 65536
outbuf: .zero 65537

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rax, 0
    mov rdi, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor rcx, rcx
    mov r8, rsi
process_loop:
    cmp rcx, r12
    jae process_end
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x0A
    je found_newline
    inc rcx
    jmp process_loop
found_newline:
    lea r9, [rsi + rcx]
reverse_loop1:
    cmp r9, r8
    jbe emit_nl1
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_loop1
emit_nl1:
    mov byte ptr [rdi], 0x0A
    inc rdi
    inc rcx
    lea r8, [rsi + rcx]
    jmp process_loop
process_end:
    lea r9, [rsi + rcx]
    cmp r9, r8
    jbe write_out
reverse_loop2:
    cmp r9, r8
    jbe emit_nl2
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_loop2
emit_nl2:
    mov byte ptr [rdi], 0x0A
    inc rdi
write_out:
    lea rax, [rip + outbuf]
    sub rdi, rax
    jz do_exit
    mov rdx, rdi
    mov rsi, rax
    mov rax, 1
    mov rdi, 1
    syscall
do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
```
