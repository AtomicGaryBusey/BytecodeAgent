.globl _start

.bss
.align 16
input_buf: .zero 65536
.align 16
out_buf: .zero 128

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor rax, rax
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor rcx, rcx
    lea rbx, [rip + input_buf]

count_loop:
    cmp rcx, r12
    jae count_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 0x0A
    jne 1f
    inc r13
1:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    test r15, r15
    jnz count_loop
    mov r15d, 1
    inc r14
    jmp count_loop
is_ws:
    xor r15, r15
    jmp count_loop

count_done:
    lea rdi, [rip + out_buf + 127]
    mov byte ptr [rdi], 0x0A
    dec rdi
    mov rax, r12
    call put_num
    mov byte ptr [rdi], 0x20
    dec rdi
    mov rax, r14
    call put_num
    mov byte ptr [rdi], 0x20
    dec rdi
    mov rax, r13
    call put_num
    inc rdi
    mov rsi, rdi
    lea rdx, [rip + out_buf + 128]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

put_num:
    test rax, rax
    jnz pn_loop
    mov byte ptr [rdi], 0x30
    dec rdi
    ret
pn_loop:
    test rax, rax
    jz pn_done
    mov r9, 10
    xor rdx, rdx
    div r9
    add dl, 0x30
    mov byte ptr [rdi], dl
    dec rdi
    jmp pn_loop
pn_done:
    ret
