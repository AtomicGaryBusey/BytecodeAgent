.bss

_start:
    mov x19, #0            // total bytes
    mov x20, #0            // lines
    mov x21, #0            // words
    mov x22, #0            // in_word flag

    adrp x23, buf
    add x23, x23, :lo12:buf
    mov x24, #65536        // remaining capacity

read_loop:
    cbz x24, done_read
    mov x8, #63
    mov x0, #0
    mov x1, x23
    mov x2, x24
    svc #0

    cmp x0, #0
    b.le done_read

    add x19, x19, x0
    mov x25, x23
    add x26, x23, x0

proc_loop:
    cmp x25, x26
    b.eq proc_done
    ldrb w9, [x25], #1

    cmp w9, #0x0A
    b.ne not_nl
    add x20, x20, #1
not_nl:
    cmp w9, #0x20
    b.eq is_ws
    cmp w9, #0x09
    b.lo not_ws
    cmp w9, #0x0D
    b.ls is_ws
not_ws:
    cbnz x22, proc_loop
    mov x22, #1
    add x21, x21, #1
    b proc_loop
is_ws:
    mov x22, #0
    b proc_loop

proc_done:
    add x23, x23, x0
    sub x24, x24, x0
    b read_loop

done_read:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x27, x25

    mov x0, x20
    bl write_dec
    mov w9, #0x20
    strb w9, [x25], #1

    mov x0, x21
    bl write_dec
    mov w9, #0x20
    strb w9, [x25], #1

    mov x0, x19
    bl write_dec
    mov w9, #0x0A
    strb w9, [x25], #1

    sub x2, x25, x27
    mov x1, x27
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// write_dec: x0 = number, x25 = output ptr (advanced)
write_dec:
    sub sp, sp, #32
    add x14, sp, #31
    mov x10, x14
    mov x11, #10

    cbnz x0, wd_loop
    mov w9, #0x30
    strb w9, [x25], #1
    add sp, sp, #32
    ret

wd_loop:
    cbz x0, wd_copy
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add w13, w13, #0x30
    sub x10, x10, #1
    strb w13, [x10]
    mov x0, x12
    b wd_loop

wd_copy:
    cmp x10, x14
    b.hs wd_end
    ldrb w9, [x10], #1
    strb w9, [x25], #1
    b wd_copy
wd_end:
    add sp, sp, #32
    ret
