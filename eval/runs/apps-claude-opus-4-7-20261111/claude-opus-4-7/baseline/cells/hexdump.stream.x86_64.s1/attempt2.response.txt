    mov edx, 65536
    sub rdx, r12
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]

    jle read_done
    add r12, rax
    jmp read_loop

    xor r15, r15
    lea rbp, [rip + hex_digits]

row_loop:

    jae final_offset

    mov r10, rbx
    add r10, r15

    mov r8, r12
    sub r8, r15
    cmp r8, 16
    jbe 1f

    jmp 3b

    dec ecx
    jnz 9b

    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
