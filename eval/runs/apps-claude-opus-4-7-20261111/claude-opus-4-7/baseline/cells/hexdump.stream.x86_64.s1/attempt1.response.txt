.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
.align 16
input_buf: .zero 65536
output_buf: .zero 524288

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12, r12
read_loop:
    mov edx, 65536
    sub rdx, r12
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + output_buf]
    mov r14, r13
    xor r15, r15

row_loop:
    cmp r15, r12
    jae rows_done

    mov r8, r12
    sub r8, r15
    cmp r8, 16
    jbe 1f
    mov r8d, 16
1:
    mov eax, r15d
    mov ecx, 8
2:
    rol eax, 4
    mov edx, eax
    and edx, 15
