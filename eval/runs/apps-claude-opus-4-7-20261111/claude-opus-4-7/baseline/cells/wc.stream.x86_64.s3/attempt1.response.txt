.globl _start

.bss
.align 16
input_buf:
.zero 65536
out_buf:
.zero 128

.text
_start:
    xor r12, r12
read_loop:
    mov edx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rbx, rbx
    xor rbp, rbp
    xor r13d, r13d
    xor r14, r14
    lea r15, [rip + input_buf]
count_loop:
    cmp r14, r12
    jae count_done
    movzx eax, byte ptr [r15 + r14]
    inc r14
    cmp al, 0x0A
    jne not_nl
    inc rbx
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    jbe is_ws
not_ws:
    test r13d, r13d
    jnz count_loop
    inc rbp
    mov r13d, 1
    jmp count_loop
is_ws:
    xor r13d, r13d
    jmp count_loop
count_done:
    lea rdi, [rip + out_buf]
    mov rax, rbx
    call write_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, rbp
    call write_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r12
    call write_dec
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

write_dec:
    mov r8, rdi
    mov r9d, 10
wd_loop:
    xor edx, edx
    div r9
    add dl, 0x30
    mov [rdi], dl
    inc rdi
    test rax, rax
    jnz wd_loop
    mov rcx, rdi
    dec rcx
wd_rev:
    cmp r8, rcx
    jae wd_done
    mov al, [r8]
    mov dl, [rcx]
    mov [r8], dl
    mov [rcx], al
    inc r8
    dec rcx
    jmp wd_rev
wd_done:
    ret
