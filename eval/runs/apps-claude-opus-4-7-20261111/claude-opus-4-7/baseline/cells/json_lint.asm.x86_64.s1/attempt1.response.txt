.section .rodata
ok_msg: .ascii "ok\n"

.bss
.align 16
input_buf: .zero 65536
out_buf:   .zero 64
dec_tmp:   .zero 32

.text
.globl _start

_start:
    lea rbp, [rip + input_buf]
    xor rbx, rbx
    xor r12, r12
.Lread:
    mov rdx, 65536
    sub rdx, r12
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbp + r12]
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread
.Lread_done:
    call skip_ws
    call parse_value
    jc emit_error
    call skip_ws
    cmp rbx, r12
    jne emit_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp rbx, r12
    jae .Lsw_ret
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 0x20
    je .Lsw_adv
    cmp al, 0x09
    je .Lsw_adv
    cmp al, 0x0A
    je .Lsw_adv
    cmp al, 0x0D
    jne .Lsw_ret
.Lsw_adv:
    inc rbx
    jmp skip_ws
.Lsw_ret:
    ret

parse_value:
    cmp rbx, r12
    jae .Lpv_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb .Lpv_kw
    cmp al, '9'
    jbe parse_number
.Lpv_kw:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
.Lpv_err:
    stc
    ret

parse_true:
    lea rdi, [rbx + 1]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'r'
    jne .Lkw_f
    lea rdi, [rbx + 2]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'u'
    jne .Lkw_f
    lea rdi, [rbx + 3]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'e'
    jne .Lkw_f
    add rbx, 4
    clc
    ret

parse_false:
    lea rdi, [rbx + 1]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'a'
    jne .Lkw_f
    lea rdi, [rbx + 2]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'l'
    jne .Lkw_f
    lea rdi, [rbx + 3]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 's'
    jne .Lkw_f
    lea rdi, [rbx + 4]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'e'
    jne .Lkw_f
    add rbx, 5
    clc
    ret

parse_null:
    lea rdi, [rbx + 1]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'u'
    jne .Lkw_f
    lea rdi, [rbx + 2]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'l'
    jne .Lkw_f
    lea rdi, [rbx + 3]
    cmp rdi, r12
    jae .Lkw_f
    cmp byte ptr [rbp + rdi], 'l'
    jne .Lkw_f
    add rbx, 4
    clc
    ret

.Lkw_f:
    mov rbx, rdi
    stc
    ret

parse_number:
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '-'
    jne .Lnp_i
    inc rbx
    cmp rbx, r12
    jae .Lnerr
    movzx eax, byte ptr [rbp + rbx]
.Lnp_i:
    cmp al, '0'
    jb .Lnerr
    cmp al, '9'
    ja .Lnerr
    inc rbx
    cmp al, '0'
    je .Lnp_frac
.Lnp_d:
    cmp rbx, r12
    jae .Lnok
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '0'
    jb .Lnp_frac
    cmp al, '9'
    ja .Lnp_frac
    inc rbx
    jmp .Lnp_d
.Lnp_frac:
    cmp rbx, r12
    jae .Lnok
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '.'
    jne .Lnp_exp
    inc rbx
    cmp rbx, r12
    jae .Lnerr
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '0'
    jb .Lnerr
    cmp al, '9'
    ja .Lnerr
    inc rbx
.Lnp_fd:
    cmp rbx, r12
    jae .Lnok
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '0'
    jb .Lnp_exp
    cmp al, '9'
    ja .Lnp_exp
    inc rbx
    jmp .Lnp_fd
.Lnp_exp:
    cmp rbx, r12
    jae .Lnok
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 'e'
    je .Lnp_e2
    cmp al, 'E'
    jne .Lnok
.Lnp_e2:
    inc rbx
    cmp rbx, r12
    jae .Lnerr
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '+'
    je .Lnp_esign
    cmp al, '-'
    jne .Lnp_e4
.Lnp_esign:
    inc rbx
    cmp rbx, r12
    jae .Lnerr
    movzx eax, byte ptr [rbp + rbx]
.Lnp_e4:
    cmp al, '0'
    jb .Lnerr
    cmp al, '9'
    ja .Lnerr
    inc rbx
.Lnp_ed:
    cmp rbx, r12
    jae .Lnok
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '0'
    jb .Lnok
    cmp al, '9'
    ja .Lnok
    inc rbx
    jmp .Lnp_ed
.Lnok:
    clc
    ret
.Lnerr:
    stc
    ret

parse_string:
    inc rbx
.Lps_l:
    cmp rbx, r12
    jae .Lps_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '"'
    je .Lps_close
    cmp al, 0x5C
    je .Lps_esc
    cmp al, 0x20
    jb .Lps_err
    inc rbx
    jmp .Lps_l
.Lps_esc:
    inc rbx
    cmp rbx, r12
    jae .Lps_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '"'
    je .Lps_eok
    cmp al, 0x5C
    je .Lps_eok
    cmp al, '/'
    je .Lps_eok
    cmp al, 'b'
    je .Lps_eok
    cmp al, 'f'
    je .Lps_eok
    cmp al, 'n'
    je .Lps_eok
    cmp al, 'r'
    je .Lps_eok
    cmp al, 't'
    je .Lps_eok
    cmp al, 'u'
    je .Lps_eu
    jmp .Lps_err
.Lps_eok:
    inc rbx
    jmp .Lps_l
.Lps_eu:
    inc rbx
    mov rcx, 4
.Lps_ul:
    cmp rbx, r12
    jae .Lps_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '0'
    jb .Lps_err
    cmp al, '9'
    jbe .Lps_hok
    cmp al, 'A'
    jb .Lps_err
    cmp al, 'F'
    jbe .Lps_hok
    cmp al, 'a'
    jb .Lps_err
    cmp al, 'f'
    ja .Lps_err
.Lps_hok:
    inc rbx
    dec rcx
    jnz .Lps_ul
    jmp .Lps_l
.Lps_close:
    inc rbx
    clc
    ret
.Lps_err:
    stc
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae .Larr_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, ']'
    je .Larr_close
.Larr_it:
    call parse_value
    jc .Larr_pass
    call skip_ws
    cmp rbx, r12
    jae .Larr_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, ']'
    je .Larr_close
    cmp al, ','
    jne .Larr_err
    inc rbx
    call skip_ws
    jmp .Larr_it
.Larr_close:
    inc rbx
    clc
    ret
.Larr_err:
    stc
    ret
.Larr_pass:
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae .Lobj_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '}'
    je .Lobj_close
.Lobj_memb:
    cmp rbx, r12
    jae .Lobj_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '"'
    jne .Lobj_err
    call parse_string
    jc .Lobj_pass
    call skip_ws
    cmp rbx, r12
    jae .Lobj_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, ':'
    jne .Lobj_err
    inc rbx
    call skip_ws
    call parse_value
    jc .Lobj_pass
    call skip_ws
    cmp rbx, r12
    jae .Lobj_err
    movzx eax, byte ptr [rbp + rbx]
    cmp al, '}'
    je .Lobj_close
    cmp al, ','
    jne .Lobj_err
    inc rbx
    call skip_ws
    jmp .Lobj_memb
.Lobj_close:
    inc rbx
    clc
    ret
.Lobj_err:
    stc
    ret
.Lobj_pass:
    ret

emit_error:
    mov byte ptr [rip + out_buf + 0], 'e'
    mov byte ptr [rip + out_buf + 1], 'r'
    mov byte ptr [rip + out_buf + 2], 'r'
    mov byte ptr [rip + out_buf + 3], 'o'
    mov byte ptr [rip + out_buf + 4], 'r'
    mov byte ptr [rip + out_buf + 5], 0x20
    mov byte ptr [rip + out_buf + 6], 'a'
    mov byte ptr [rip + out_buf + 7], 't'
    mov byte ptr [rip + out_buf + 8], 0x20
    mov rax, rbx
    lea r8, [rip + dec_tmp + 31]
    mov ecx, 10
.Lde:
    xor edx, edx
    div rcx
    add dl, '0'
    mov [r8], dl
    dec r8
    test rax, rax
    jnz .Lde
    inc r8
    lea r9, [rip + dec_tmp + 32]
    sub r9, r8
    lea rdi, [rip + out_buf + 9]
    mov rsi, r8
    mov rcx, r9
    rep movsb
    mov byte ptr [rdi], 0x0A
    lea rdx, [r9 + 10]
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    syscall
    mov eax, 60
    mov edi, 1
    syscall
