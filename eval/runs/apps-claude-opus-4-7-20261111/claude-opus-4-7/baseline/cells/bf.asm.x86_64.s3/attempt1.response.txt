```asm
.globl _start

.bss
.balign 8
input_buf: .zero 65536
tape: .zero 30000
.balign 4
out_buf: .zero 65536
.balign 4
bracket_match: .zero 262144
bracket_stack: .zero 4000

.text
_start:
    xor r12, r12
read_loop:
    xor rax, rax
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    mov r13, r12

    xor rcx, rcx
    lea rbx, [rip + input_buf]
find_bang:
    cmp rcx, r13
    je no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov rsi, r13
    mov r12, r13
    jmp validate
found_bang:
    mov rsi, rcx
    lea r12, [rcx + 1]

validate:
    lea r8, [rip + bracket_stack]
    lea r11, [rip + bracket_match]
    xor rcx, rcx
    xor r9, r9
vloop:
    cmp rcx, rsi
    je vend
    mov al, [rbx + rcx]
    cmp al, 0x5b
    je v_open
    cmp al, 0x5d
    je v_close
    inc rcx
    jmp vloop
v_open:
    mov [r8 + r9*4], ecx
    inc r9
    inc rcx
    jmp vloop
v_close:
    test r9, r9
    jz exit_err
    dec r9
    mov edx, [r8 + r9*4]
    mov [r11 + rdx*4], ecx
    mov [r11 + rcx*4], edx
    inc rcx
    jmp vloop
vend:
    test r9, r9
    jnz exit_err

    lea r14, [rip + tape]
    xor r15, r15
    xor rcx, rcx

run_loop:
    cmp rcx, rsi
    jae run_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2b
    je op_plus
    cmp al, 0x2d
    je op_minus
    cmp al, 0x3e
    je op_right
    cmp al, 0x3c
    je op_left
    cmp al, 0x2e
    je op_dot
    cmp al, 0x2c
    je op_comma
    cmp al, 0x5b
    je op_open
    cmp al, 0x5d
    je op_close
    inc rcx
    jmp run_loop

op_plus:
    inc byte ptr [r14]
    inc rcx
    jmp run_loop
op_minus:
    dec byte ptr [r14]
    inc rcx
    jmp run_loop
op_right:
    inc r14
    inc rcx
    jmp run_loop
op_left:
    dec r14
    inc rcx
    jmp run_loop
op_dot:
    mov al, [r14]
    lea r10, [rip + out_buf]
    mov [r10 + r15], al
    inc r15
    inc rcx
    jmp run_loop
op_comma:
    cmp r12, r13
    jae comma_eof
    mov al, [rbx + r12]
    mov [r14], al
    inc r12
    inc rcx
    jmp run_loop
comma_eof:
    mov byte ptr [r14], 0
    inc rcx
    jmp run_loop
op_open:
    cmp byte ptr [r14], 0
    jne after_inc
    mov edx, [r11 + rcx*4]
    lea rcx, [rdx + 1]
    jmp run_loop
op_close:
    cmp byte ptr [r14], 0
    je after_inc
    mov edx, [r11 + rcx*4]
    lea rcx, [rdx + 1]
    jmp run_loop
after_inc:
    inc rcx
    jmp run_loop

run_done:
    test r15, r15
    jz exit_ok
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r15
    syscall
exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

exit_err:
    mov rax, 60
    mov rdi, 1
    syscall
```
