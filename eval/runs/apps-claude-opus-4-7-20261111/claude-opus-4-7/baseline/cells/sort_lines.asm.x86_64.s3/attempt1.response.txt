.globl _start

.bss
.p2align 4
input_buf: .zero 32768
.p2align 4
lines: .zero 8192
.p2align 4
qs_stack: .zero 16384
.p2align 4
out_buf: .zero 40960

.text

cmp_lines:
    mov r8d, dword ptr [rdi]
    mov r9d, dword ptr [rdi+4]
    mov r10d, dword ptr [rsi]
    mov r11d, dword ptr [rsi+4]
    mov eax, r9d
    cmp r11d, eax
    cmovl eax, r11d
.Lcmpl:
    test eax, eax
    jz .Lcmp_prefix_eq
    movzx ecx, byte ptr [input_buf + r8]
    movzx edx, byte ptr [input_buf + r10]
    cmp ecx, edx
    jne .Lcmp_diff
    inc r8
    inc r10
    dec eax
    jmp .Lcmpl
.Lcmp_diff:
    sub ecx, edx
    mov eax, ecx
    ret
.Lcmp_prefix_eq:
    mov eax, r9d
    sub eax, r11d
    ret

_start:
    xor r12d, r12d
.Lread:
    cmp r12d, 32768
    jae .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [input_buf + r12]
    mov edx, 32768
    sub edx, r12d
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread
.Lread_done:
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d
.Lparse:
    cmp r14, r12
    jae .Lparse_end
    mov al, byte ptr [input_buf + r14]
    inc r14
    cmp al, 0x0A
    jne .Lparse
    cmp rbx, 1000
    jae .Lparse_skip
    mov dword ptr [lines + rbx*8], r13d
    mov eax, r14d
    dec eax
    sub eax, r13d
    mov dword ptr [lines + rbx*8 + 4], eax
    inc rbx
.Lparse_skip:
    mov r13d, r14d
    jmp .Lparse
.Lparse_end:
    cmp r13, r12
    jae .Lno_trail
    cmp rbx, 1000
    jae .Lno_trail
    mov dword ptr [lines + rbx*8], r13d
    mov eax, r12d
    sub eax, r13d
    mov dword ptr [lines + rbx*8 + 4], eax
    inc rbx
.Lno_trail:
    cmp rbx, 1
    jbe .Loutput

    lea r15, [qs_stack]
    mov dword ptr [r15], 0
    mov eax, ebx
    dec eax
    mov dword ptr [r15+4], eax
    add r15, 8

.Lqs_loop:
    lea rax, [qs_stack]
    cmp r15, rax
    jbe .Loutput
    sub r15, 8
    mov r12d, dword ptr [r15]
    mov r13d, dword ptr [r15+4]

.Lqs_part:
    cmp r12d, r13d
    jge .Lqs_loop

    mov eax, r12d
    add eax, r13d
    shr eax, 1
    mov rcx, qword ptr [lines + rax*8]
    mov rdx, qword ptr [lines + r13*8]
    mov qword ptr [lines + rax*8], rdx
    mov qword ptr [lines + r13*8], rcx

    mov r14d, r12d
    dec r14d
    mov ebp, r12d
.Lqs_inner:
    cmp ebp, r13d
    jge .Lqs_inner_end
    lea rdi, [lines + rbp*8]
    lea rsi, [lines + r13*8]
    call cmp_lines
    test eax, eax
    jg .Lqs_skip
    inc r14d
    mov rcx, qword ptr [lines + r14*8]
    mov rdx, qword ptr [lines + rbp*8]
    mov qword ptr [lines + r14*8], rdx
    mov qword ptr [lines + rbp*8], rcx
.Lqs_skip:
    inc ebp
    jmp .Lqs_inner
.Lqs_inner_end:
    inc r14d
    mov rcx, qword ptr [lines + r14*8]
    mov rdx, qword ptr [lines + r13*8]
    mov qword ptr [lines + r14*8], rdx
    mov qword ptr [lines + r13*8], rcx

    mov eax, r14d
    inc eax
    cmp eax, r13d
    jg .Lqs_no_right
    mov dword ptr [r15], eax
    mov dword ptr [r15+4], r13d
    add r15, 8
.Lqs_no_right:
    mov r13d, r14d
    dec r13d
    jmp .Lqs_part

.Loutput:
    xor r12d, r12d
    xor r13d, r13d
.Lout_loop:
    cmp r13, rbx
    jae .Lout_write
    mov eax, dword ptr [lines + r13*8]
    mov ecx, dword ptr [lines + r13*8 + 4]
    lea rsi, [input_buf + rax]
    lea rdi, [out_buf + r12]
    mov rdx, rcx
    rep movsb
    add r12, rdx
    mov byte ptr [out_buf + r12], 0x0A
    inc r12
    inc r13
    jmp .Lout_loop
.Lout_write:
    xor r13d, r13d
.Lwrite_loop:
    cmp r13, r12
    jae .Lexit
    mov eax, 1
    mov edi, 1
    lea rsi, [out_buf + r13]
    mov rdx, r12
    sub rdx, r13
    syscall
    test rax, rax
    jle .Lexit
    add r13, rax
    jmp .Lwrite_loop
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall
