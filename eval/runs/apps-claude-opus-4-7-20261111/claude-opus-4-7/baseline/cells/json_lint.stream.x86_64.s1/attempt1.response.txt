.globl _start

.bss
.align 16
input_buf:
.zero 65600
out_buf:
.zero 64

.section .rodata
true_lit:    .ascii "true"
false_lit:   .ascii "false"
null_lit:    .ascii "null"
ok_msg:      .ascii "ok\n"

.text

_start:
  lea r12, [rip + input_buf]
  xor r13, r13
.read_loop:
  mov rdx, 65536
  sub rdx, r13
  jz .read_done
  lea rsi, [r12 + r13]
  xor eax, eax
  xor edi, edi
  syscall
  test rax, rax
  jle .read_done
  add r13, rax
  jmp .read_loop
.read_done:
  mov r14, -1
  xor edi, edi
  call skip_ws
  mov rdi, rax
  call parse_value
  test rax, rax
  js .err_exit
  mov rdi, rax
  call skip_ws
  cmp rax, r13
  je .ok_exit
  mov r14, rax
.err_exit:
  lea rdi, [rip + out_buf]
  mov byte ptr [rdi+0], 0x65
  mov byte ptr [rdi+1], 0x72
  mov byte ptr [rdi+2], 0x72
  mov byte ptr [rdi+3], 0x6F
  mov byte ptr [rdi+4], 0x72
  mov byte ptr [rdi+5], 0x20
  mov byte ptr [rdi+6], 0x61
  mov byte ptr [rdi+7], 0x74
  mov byte ptr [rdi+8], 0x20
  add rdi, 9
  mov rax, r14
  test rax, rax
  jnz .dec_nz
  mov byte ptr [rdi], 0x30
  inc rdi
  jmp .dec_dn
.dec_nz:
  sub rsp, 32
  mov rsi, rsp
  mov r8, 10
.dec_lp:
  xor edx, edx
  div r8
  add dl, 0x30
  mov byte ptr [rsi], dl
  inc rsi
  test rax, rax
  jnz .dec_lp
.dec_rv:
  dec rsi
  mov dl, byte ptr [rsi]
  mov byte ptr [rdi], dl
  inc rdi
  cmp rsi, rsp
  ja .dec_rv
  add rsp, 32
.dec_dn:
  mov byte ptr [rdi], 10
  inc rdi
  lea rsi, [rip + out_buf]
  mov rdx, rdi
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall
  mov edi, 1
  mov eax, 60
  syscall
.ok_exit:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, 3
  syscall
  xor edi, edi
  mov eax, 60
  syscall

skip_ws:
  mov rax, rdi
.sw1:
  cmp rax, r13
  jae .sw0
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x20
  je .sw2
  cmp cl, 0x0A
  je .sw2
  cmp cl, 0x0D
  je .sw2
  cmp cl, 0x09
  jne .sw0
.sw2:
  inc rax
  jmp .sw1
.sw0:
  ret

parse_value:
  cmp rdi, r13
  jae .pv_fail
  movzx eax, byte ptr [r12 + rdi]
  cmp al, 0x7B
  je parse_object
  cmp al, 0x5B
  je parse_array
  cmp al, 0x22
  je parse_string
  cmp al, 0x74
  je .pv_true
  cmp al, 0x66
  je .pv_false
  cmp al, 0x6E
  je .pv_null
  cmp al, 0x2D
  je parse_number
  cmp al, 0x30
  jb .pv_fail
  cmp al, 0x39
  jbe parse_number
.pv_fail:
  mov r14, rdi
  mov rax, -1
  ret
.pv_true:
  lea rsi, [rip + true_lit]
  mov edx, 4
  jmp parse_lit
.pv_false:
  lea rsi, [rip + false_lit]
  mov edx, 5
  jmp parse_lit
.pv_null:
  lea rsi, [rip + null_lit]
  mov edx, 4
  jmp parse_lit

parse_lit:
  xor ecx, ecx
.pl1:
  cmp rcx, rdx
  jae .pl_ok
  lea rax, [rdi + rcx]
  cmp rax, r13
  jae .pl_fail
  mov r8b, byte ptr [r12 + rax]
  cmp r8b, byte ptr [rsi + rcx]
  jne .pl_fail
  inc rcx
  jmp .pl1
.pl_ok:
  lea rax, [rdi + rcx]
  ret
.pl_fail:
  mov r14, rax
  mov rax, -1
  ret

parse_string:
  lea rax, [rdi + 1]
.ps1:
  cmp rax, r13
  jae .ps_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x22
  je .ps_end
  cmp cl, 0x5C
  je .ps_esc
  cmp cl, 0x20
  jb .ps_fail
  inc rax
  jmp .ps1
.ps_end:
  inc rax
  ret
.ps_esc:
  inc rax
  cmp rax, r13
  jae .ps_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x22
  je .ps_smpl
  cmp cl, 0x5C
  je .ps_smpl
  cmp cl, 0x2F
  je .ps_smpl
  cmp cl, 0x62
  je .ps_smpl
  cmp cl, 0x66
  je .ps_smpl
  cmp cl, 0x6E
  je .ps_smpl
  cmp cl, 0x72
  je .ps_smpl
  cmp cl, 0x74
  je .ps_smpl
  cmp cl, 0x75
  je .ps_uni
  jmp .ps_fail
.ps_smpl:
  inc rax
  jmp .ps1
.ps_uni:
  inc rax
  mov ecx, 4
.psu1:
  cmp rax, r13
  jae .ps_fail
  movzx edx, byte ptr [r12 + rax]
  cmp dl, 0x30
  jb .ps_fail
  cmp dl, 0x39
  jbe .psu_ok
  cmp dl, 0x61
  jb .psu_up
  cmp dl, 0x66
  jbe .psu_ok
  jmp .ps_fail
.psu_up:
  cmp dl, 0x41
  jb .ps_fail
  cmp dl, 0x46
  ja .ps_fail
.psu_ok:
  inc rax
  dec ecx
  jnz .psu1
  jmp .ps1
.ps_fail:
  mov r14, rax
  mov rax, -1
  ret

parse_number:
  mov rax, rdi
  cmp byte ptr [r12 + rax], 0x2D
  jne .pn_nm
  inc rax
.pn_nm:
  cmp rax, r13
  jae .pn_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  je .pn_zero
  cmp cl, 0x31
  jb .pn_fail
  cmp cl, 0x39
  ja .pn_fail
  inc rax
.pn_digits:
  cmp rax, r13
  jae .pn_frac
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  jb .pn_frac
  cmp cl, 0x39
  ja .pn_frac
  inc rax
  jmp .pn_digits
.pn_zero:
  inc rax
.pn_frac:
  cmp rax, r13
  jae .pn_done
  cmp byte ptr [r12 + rax], 0x2E
  jne .pn_exp
  inc rax
  cmp rax, r13
  jae .pn_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  jb .pn_fail
  cmp cl, 0x39
  ja .pn_fail
  inc rax
.pn_fd:
  cmp rax, r13
  jae .pn_done
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  jb .pn_exp
  cmp cl, 0x39
  ja .pn_exp
  inc rax
  jmp .pn_fd
.pn_exp:
  cmp rax, r13
  jae .pn_done
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x65
  je .pn_e
  cmp cl, 0x45
  jne .pn_done
.pn_e:
  inc rax
  cmp rax, r13
  jae .pn_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x2B
  je .pn_esign
  cmp cl, 0x2D
  jne .pn_ed1
.pn_esign:
  inc rax
.pn_ed1:
  cmp rax, r13
  jae .pn_fail
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  jb .pn_fail
  cmp cl, 0x39
  ja .pn_fail
  inc rax
.pn_ed:
  cmp rax, r13
  jae .pn_done
  movzx ecx, byte ptr [r12 + rax]
  cmp cl, 0x30
  jb .pn_done
  cmp cl, 0x39
  ja .pn_done
  inc rax
  jmp .pn_ed
.pn_done:
  ret
.pn_fail:
  mov r14, rax
  mov rax, -1
  ret

parse_object:
  inc rdi
  call skip_ws
  mov rdi, rax
  cmp rdi, r13
  jae .po_fail
  cmp byte ptr [r12 + rdi], 0x7D

.po_mem:

  mov rdi, rax
  call skip_ws
  mov rdi, rax
  cmp rdi, r13
  jae .po_fail
  cmp byte ptr [r12 + rdi], 0x3A
  jne .po_fail
  inc rdi

  mov rdi, rax

  test rax, rax

  mov rdi, rax
  call skip_ws

  cmp rdi, r13

  movzx eax, byte ptr [r12 + rdi]
  cmp al, 0x7D

  cmp al, 0x2C
