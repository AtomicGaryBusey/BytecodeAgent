.globl _start

.bss
.align 16
inbuf: .zero 65536
outbuf: .zero 65537

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor rcx, rcx
    xor rbx, rbx
    xor r8, r8
line_scan:
    cmp r8, r12
    je check_tail
    mov al, byte ptr [rsi + r8]
    cmp al, 0x0A
    je found_newline
    inc r8
    jmp line_scan
found_newline:
    mov r9, r8
    dec r9
reverse_copy:
    cmp r9, rcx
    jl end_reverse
    mov al, byte ptr [rsi + r9]
    mov byte ptr [rdi + rbx], al
    inc rbx
    dec r9
    jmp reverse_copy
end_reverse:
    mov byte ptr [rdi + rbx], 0x0A
    inc rbx
    inc r8
    mov rcx, r8
    jmp line_scan
check_tail:
    cmp rcx, r12
    je write_output
    mov r9, r12
    dec r9
tail_reverse:
    cmp r9, rcx
    jl end_tail
    mov al, byte ptr [rsi + r9]
    mov byte ptr [rdi + rbx], al
    inc rbx
    dec r9
    jmp tail_reverse
end_tail:
    mov byte ptr [rdi + rbx], 0x0A
    inc rbx
write_output:
    xor r13, r13
write_loop:
    cmp r13, rbx
    jge exit_prog
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, r13
    mov rdx, rbx
    sub rdx, r13
    syscall
    test rax, rax
    js exit_prog
    jz exit_prog
    add r13, rax
    jmp write_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
