```asm
.globl _start

.section .bss
.balign 16
input_buf: .zero 131072
out_buf: .zero 524288
num_stack: .zero 8192
num_scratch: .zero 32

.section .text

_start:
    lea r12, [rip + input_buf]
    mov r13, r12
.Lread_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, 65536
    syscall
    test rax, rax
    jle .Lread_done
    add r13, rax
    jmp .Lread_loop
.Lread_done:
    lea r14, [rip + out_buf]
    mov rbx, r12

.Lline_loop:
    cmp rbx, r13
    jae .Lprint_and_exit

    mov rcx, rbx
.Lfnl:
    cmp rcx, r13
    jae .Lnl_found
    cmp byte ptr [rcx], 0x0a
    je .Lnl_found
    inc rcx
    jmp .Lfnl
.Lnl_found:
    mov r15, rcx
    mov r9, rcx
    cmp r9, r13
    jae .Lno_nl
    inc r9
.Lno_nl:
    xor r10d, r10d

.Ltok_loop:
.Lskip_sep:
    cmp rbx, r15
    jae .Lline_end
    mov al, byte ptr [rbx]
    cmp al, 0x20
    je .Lssi
    cmp al, 0x09
    jne .Ltok_find
.Lssi:
    inc rbx
    jmp .Lskip_sep
.Ltok_find:
    mov rdi, rbx
.Lfe:
    cmp rbx, r15
    jae .Lfe_done
    mov al, byte ptr [rbx]
    cmp al, 0x20
    je .Lfe_done
    cmp al, 0x09
    je .Lfe_done
    inc rbx
    jmp .Lfe
.Lfe_done:
    mov rsi, rbx
    sub rsi, rdi
    cmp rsi, 1
    jne .Lparse_num
    mov al, byte ptr [rdi]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod
.Lparse_num:
    xor r8d, r8d
    mov al, byte ptr [rdi]
    cmp al, '-'
    jne .Lno_sign
    inc rdi
    mov r8d, 1
.Lno_sign:
    mov rcx, rbx
    sub rcx, rdi
    test rcx, rcx
    jz .Lerror
    cmp rcx, 18
    jg .Lerror
    xor eax, eax
.Ldig_loop:
    cmp rdi, rbx
    jae .Ldig_done
    movzx edx, byte ptr [rdi]
    sub edx, '0'
    cmp edx, 9
    ja .Lerror
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc rdi
    jmp .Ldig_loop
.Ldig_done:
    test r8d, r8d
    jz .Lpush
    neg rax
.Lpush:
    mov qword ptr [num_stack + r10*8], rax
    inc r10
    jmp .Ltok_loop

.Lop_add:
    cmp r10, 2
    jl .Lerror
    mov rax, qword ptr [num_stack + r10*8 - 8]
    add qword ptr [num_stack + r10*8 - 16], rax
    dec r10
    jmp .Ltok_loop

.Lop_sub:
    cmp r10, 2
    jl .Lerror
    mov rax, qword ptr [num_stack + r10*8 - 8]
    sub qword ptr [num_stack + r10*8 - 16], rax
    dec r10
    jmp .Ltok_loop

.Lop_mul:
    cmp r10, 2
    jl .Lerror
    mov rax, qword ptr [num_stack + r10*8 - 16]
    imul rax, qword ptr [num_stack + r10*8 - 8]
    mov qword ptr [num_stack + r10*8 - 16], rax
    dec r10
    jmp .Ltok_loop

.Lop_div:
    cmp r10, 2
    jl .Lerror
    mov rax, qword ptr [num_stack + r10*8 - 16]
    mov rcx, qword ptr [num_stack + r10*8 - 8]
    test rcx, rcx
    jz .Lerror
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne .Ldiv_ok
    cmp rcx, -1
    je .Lerror
.Ldiv_ok:
    cqo
    idiv rcx
    mov qword ptr [num_stack + r10*8 - 16], rax
    dec r10
    jmp .Ltok_loop

.Lop_mod:
    cmp r10, 2
    jl .Lerror
    mov rax, qword ptr [num_stack + r10*8 - 16]
    mov rcx, qword ptr [num_stack + r10*8 - 8]
    test rcx, rcx
    jz .Lerror
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne .Lmod_ok
    cmp rcx, -1
    je .Lerror
.Lmod_ok:
    cqo
    idiv rcx
    mov qword ptr [num_stack + r10*8 - 16], rdx
    dec r10
    jmp .Ltok_loop

.Lerror:
    mov rbx, r9
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp .Lline_loop

.Lline_end:
    cmp r10, 1
    jne .Lerror
    mov rax, qword ptr [num_stack]
    test rax, rax
    jns .Lfmt_pos
    mov byte ptr [r14], '-'
    inc r14
    neg rax
.Lfmt_pos:
    lea rdi, [rip + num_scratch + 32]
    mov rsi, rdi
    mov rcx, 10
.Lfmt_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .Lfmt_loop
.Lcopy:
    cmp rsi, rdi
    jae .Lcopy_done
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc rsi
    inc r14
    jmp .Lcopy
.Lcopy_done:
    mov byte ptr [r14], 0x0a
    inc r14
    mov rbx, r9
    jmp .Lline_loop

.Lprint_and_exit:
    lea rsi, [rip + out_buf]
    mov rdx, r14
    sub rdx, rsi
.Lwr_loop:
    test rdx, rdx
    jz .Lwr_done
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle .Lwr_done
    add rsi, rax
    sub rdx, rax
    jmp .Lwr_loop
.Lwr_done:
    mov rax, 60
    xor edi, edi
    syscall
```
