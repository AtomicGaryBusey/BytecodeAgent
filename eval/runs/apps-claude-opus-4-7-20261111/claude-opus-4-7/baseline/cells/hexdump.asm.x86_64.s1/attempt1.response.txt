.globl _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
.align 16
input_buf: .zero 65536
output_buf: .zero 524288

.text
_start:
    lea r15, [rip + input_buf]
    lea rbp, [rip + hex_digits]
    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov rsi, r15
    add rsi, r12
    xor rax, rax
    xor rdi, rdi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + output_buf]
    xor r14, r14

format_loop:
    cmp r14, r12
    jae format_done

    mov rcx, r12
    sub rcx, r14
    cmp rcx, 16
    jbe row_len_ok
    mov rcx, 16
row_len_ok:
    mov eax, r14d
    mov r9, 7
off_loop:
    mov r10d, eax
    and r10d, 0xf
    movzx r10d, byte ptr [rbp + r10]
    mov byte ptr [r13 + r9], r10b
    shr eax, 4
    dec r9
    jns off_loop
    add r13, 8

    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x20
    add r13, 2

    xor r8, r8
hex_loop:
    cmp r8, 16
    je hex_done
    cmp r8, rcx
    jae hex_pad
    movzx eax, byte ptr [r15 + r14 + r8]
    mov r10, rax
    shr r10, 4
    movzx r10d, byte ptr [rbp + r10]
    mov byte ptr [r13], r10b
    mov r10, rax
    and r10, 0xf
    movzx r10d, byte ptr [rbp + r10]
    mov byte ptr [r13 + 1], r10b
    mov byte ptr [r13 + 2], 0x20
    jmp hex_next
hex_pad:
    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x20
    mov byte ptr [r13 + 2], 0x20
hex_next:
    add r13, 3
    inc r8
    cmp r8, 8
    jne hex_loop
    mov byte ptr [r13], 0x20
    inc r13
    jmp hex_loop
hex_done:
    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x7c
    add r13, 2

    xor r8, r8
asc_loop:
    cmp r8, rcx
    jae asc_done
    movzx eax, byte ptr [r15 + r14 + r8]
    cmp al, 0x20
    jb use_dot
    cmp al, 0x7e
    jbe store_ch
use_dot:
    mov al, 0x2e
store_ch:
    mov byte ptr [r13], al
    inc r13
    inc r8
    jmp asc_loop
asc_done:
    mov byte ptr [r13], 0x7c
    mov byte ptr [r13 + 1], 0x0a
    add r13, 2

    add r14, 16
    jmp format_loop

format_done:
    mov eax, r12d
    mov r9, 7
final_off:
    mov r10d, eax
    and r10d, 0xf
    movzx r10d, byte ptr [rbp + r10]
    mov byte ptr [r13 + r9], r10b
    shr eax, 4
    dec r9
    jns final_off
    add r13, 8
    mov byte ptr [r13], 0x0a
    inc r13

    lea rsi, [rip + output_buf]
    mov rdx, r13
    sub rdx, rsi
write_out:
    test rdx, rdx
    jz write_done
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_out
write_done:
    mov rax, 60
    xor rdi, rdi
    syscall
