.globl _start

.bss
.align 16
input_buf: .zero 33000
.align 16
lines_arr: .zero 16384
.align 16
output_buf: .zero 34000

.text
_start:
    lea rbx, [rip + input_buf]
    mov r12, rbx
    mov r13, 32768
.read_loop:
    test r13, r13
    jz .read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    sub r13, rax
    jmp .read_loop
.read_done:
    lea r15, [rip + lines_arr]
    mov r13, rbx
    mov r14, rbx
    xor ebp, ebp
.parse_loop:
    cmp r14, r12
    jae .parse_end
    cmp byte ptr [r14], 0x0A
    jne .parse_inc
    mov [r15], r13
    mov rax, r14
    sub rax, r13
    mov [r15+8], rax
    add r15, 16
    inc rbp
    inc r14
    mov r13, r14
    jmp .parse_loop
.parse_inc:
    inc r14
    jmp .parse_loop
.parse_end:
    cmp r13, r12
    je .no_trail
    mov [r15], r13
    mov rax, r12
    sub rax, r13
    mov [r15+8], rax
    inc rbp
.no_trail:
    cmp rbp, 2
    jl .skip_sort
    lea rdi, [rip + lines_arr]
    mov rax, rbp
    dec rax
    shl rax, 4
    lea rsi, [rdi + rax]
    call quicksort
.skip_sort:
    lea r12, [rip + lines_arr]
    lea r13, [rip + output_buf]
    mov r14, r13
    test rbp, rbp
    jz .out_done
.out_loop:
    mov rsi, [r12]
    mov rcx, [r12+8]
    test rcx, rcx
    jz .copy_done
.copy_byte:
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc rsi
    inc r14
    dec rcx
    jnz .copy_byte
.copy_done:
    mov byte ptr [r14], 0x0A
    inc r14
    add r12, 16
    dec rbp
    jnz .out_loop
.out_done:
    mov rdx, r14
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    jz .do_exit
    mov eax, 1
    mov edi, 1
    syscall
.do_exit:
    mov eax, 60
    xor edi, edi
    syscall

quicksort:
    cmp rdi, rsi
    jae .qs_ret
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov rax, r13
    sub rax, r12
    cmp rax, 32
    jb .no_m3
    shr rax, 5
    shl rax, 4
    lea rbx, [r12 + rax]
    mov rdi, r12
    mov rsi, rbx
    call compare_lines
    test rax, rax
    jle .m3_1
    mov rax, [r12]
    mov rcx, [r12+8]
    mov rdx, [rbx]
    mov r8, [rbx+8]
    mov [r12], rdx
    mov [r12+8], r8
    mov [rbx], rax
    mov [rbx+8], rcx
.m3_1:
    mov rdi, r12
    mov rsi, r13
    call compare_lines
    test rax, rax
    jle .m3_2
    mov rax, [r12]
    mov rcx, [r12+8]
    mov rdx, [r13]
    mov r8, [r13+8]
    mov [r12], rdx
    mov [r12+8], r8
    mov [r13], rax
    mov [r13+8], rcx
.m3_2:
    mov rdi, rbx
    mov rsi, r13
    call compare_lines
    test rax, rax
    jle .m3_3
    mov rax, [rbx]
    mov rcx, [rbx+8]
    mov rdx, [r13]
    mov r8, [r13+8]
    mov [rbx], rdx
    mov [rbx+8], r8
    mov [r13], rax
    mov [r13+8], rcx
.m3_3:
    mov rax, [r12]
    mov rcx, [r12+8]
    mov rdx, [rbx]
    mov r8, [rbx+8]
    mov [r12], rdx
    mov [r12+8], r8
    mov [rbx], rax
    mov [rbx+8], rcx
.no_m3:
    mov r14, [r12]
    mov r15, [r12+8]
    mov rbx, r12
    sub rbx, 16
    mov rbp, r13
    add rbp, 16
.ho_outer:
.ho_inc:
    add rbx, 16
    mov rdi, rbx
    call compare_to_pivot
    test rax, rax
    js .ho_inc
.ho_dec:
    sub rbp, 16
    mov rdi, rbp
    call compare_to_pivot
    test rax, rax
    jg .ho_dec
    cmp rbx, rbp
    jae .ho_done
    mov rax, [rbx]
    mov rcx, [rbx+8]
    mov rdx, [rbp]
    mov r8, [rbp+8]
    mov [rbx], rdx
    mov [rbx+8], r8
    mov [rbp], rax
    mov [rbp+8], rcx
    jmp .ho_outer
.ho_done:
    mov rdi, r12
    mov rsi, rbp
    call quicksort
    lea rdi, [rbp+16]
    mov rsi, r13
    call quicksort
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.qs_ret:
    ret

compare_lines:
    mov r8, [rdi]
    mov r9, [rdi+8]
    mov r10, [rsi]
    mov r11, [rsi+8]
    mov rcx, r9
    cmp rcx, r11
    jbe .cl_m
    mov rcx, r11
.cl_m:
    xor edx, edx
.cl_loop:
    cmp rdx, rcx
    jae .cl_end
    movzx eax, byte ptr [r8 + rdx]
    movzx edi, byte ptr [r10 + rdx]
    sub rax, rdi
    jnz .cl_ret
    inc rdx
    jmp .cl_loop
.cl_end:
    mov rax, r9
    sub rax, r11
.cl_ret:
    ret

compare_to_pivot:
    mov r8, [rdi]
    mov r9, [rdi+8]
    mov rcx, r9
    cmp rcx, r15
    jbe .cp_m
    mov rcx, r15
.cp_m:
    xor edx, edx
.cp_loop:
    cmp rdx, rcx
    jae .cp_end
    movzx eax, byte ptr [r8 + rdx]
    movzx edi, byte ptr [r14 + rdx]
    sub rax, rdi
    jnz .cp_ret
    inc rdx
    jmp .cp_loop
.cp_end:
    mov rax, r9
    sub rax, r15
.cp_ret:
    ret
