```
.section .bss
.align 16
input_buf: .zero 65536
.align 4
jump_table: .zero 262144
.align 4
bracket_stack: .zero 4000
tape: .zero 30000
out_buf: .zero 65536
.align 8
input_len: .zero 8
prog_len: .zero 8
input_start: .zero 8

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + input_len], r12

    xor rcx, rcx
    lea rsi, [rip + input_buf]
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov qword ptr [rip + prog_len], r12
    mov qword ptr [rip + input_start], r12
    jmp validate_brackets
found_bang:
    mov qword ptr [rip + prog_len], rcx
    inc rcx
    mov qword ptr [rip + input_start], rcx

validate_brackets:
    mov rcx, qword ptr [rip + prog_len]
    xor rbx, rbx
    xor rbp, rbp
    lea rsi, [rip + input_buf]
    lea rdi, [rip + bracket_stack]
    lea r8, [rip + jump_table]
scan_loop:
    cmp rbx, rcx
    jae scan_done
    mov al, byte ptr [rsi + rbx]
    cmp al, 0x5B
    je open_br
    cmp al, 0x5D
    je close_br
    inc rbx
    jmp scan_loop
open_br:
    cmp rbp, 1000
    jae exit_error
    mov dword ptr [rdi + rbp*4], ebx
    inc rbp
    inc rbx
    jmp scan_loop
close_br:
    test rbp, rbp
    jz exit_error
    dec rbp
    mov eax, dword ptr [rdi + rbp*4]
    mov dword ptr [r8 + rax*4], ebx
    mov dword ptr [r8 + rbx*4], eax
    inc rbx
    jmp scan_loop
scan_done:
    test rbp, rbp
    jnz exit_error

    xor rbx, rbx
    xor rbp, rbp
    mov rcx, qword ptr [rip + prog_len]
    lea rsi, [rip + input_buf]
    lea rdi, [rip + tape]
    lea r8, [rip + jump_table]
    mov r9, qword ptr [rip + input_start]
    mov r10, qword ptr [rip + input_len]
    lea r11, [rip + out_buf]
    xor r13, r13
    mov r14, r9

exec_loop:
    cmp rbx, rcx
    jae exec_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x2B
    je op_plus
    cmp al, 0x2D
    je op_minus
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x5B
    je op_lb
    cmp al, 0x5D
    je op_rb
    cmp al, 0x2E
    je op_dot
    cmp al, 0x2C
    je op_comma
    inc rbx
    jmp exec_loop

op_plus:
    inc byte ptr [rdi + rbp]
    inc rbx
    jmp exec_loop
op_minus:
    dec byte ptr [rdi + rbp]
    inc rbx
    jmp exec_loop
op_right:
    inc rbp
    inc rbx
    jmp exec_loop
op_left:
    dec rbp
    inc rbx
    jmp exec_loop
op_dot:
    mov al, byte ptr [rdi + rbp]
    mov byte ptr [r11 + r13], al
    inc r13
    inc rbx
    jmp exec_loop
op_comma:
    cmp r14, r10
    jae comma_eof
    mov al, byte ptr [rsi + r14]
    inc r14
    mov byte ptr [rdi + rbp], al
    inc rbx
    jmp exec_loop
comma_eof:
    mov byte ptr [rdi + rbp], 0
    inc rbx
    jmp exec_loop
op_lb:
    cmp byte ptr [rdi + rbp], 0
    jne op_lb_through
    mov ebx, dword ptr [r8 + rbx*4]
    inc rbx
    jmp exec_loop
op_lb_through:
    inc rbx
    jmp exec_loop
op_rb:
    cmp byte ptr [rdi + rbp], 0
    je op_rb_through
    mov ebx, dword ptr [r8 + rbx*4]
    inc rbx
    jmp exec_loop
op_rb_through:
    inc rbx
    jmp exec_loop

exec_done:
    test r13, r13
    jz exit_ok
    xor r12, r12
write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    add rsi, r12
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jle exit_ok
    add r12, rax
    cmp r12, r13
    jb write_loop
exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
```
