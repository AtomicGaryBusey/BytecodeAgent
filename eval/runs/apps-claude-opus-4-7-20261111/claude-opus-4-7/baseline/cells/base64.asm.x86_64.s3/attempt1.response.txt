.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buf: .zero 49152
output_buf: .zero 70000

.text
_start:
    xor r12, r12
.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    jz .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread

.Lread_done:
    test r12, r12
    jz .Lexit

    lea rbx, [rip + alphabet]
    lea rsi, [rip + input_buf]
    lea rdi, [rip + output_buf]
    xor ecx, ecx
    mov r13, r12

.Lloop3:
    cmp r13, 3
    jl .Ltail
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]
    movzx r8d, byte ptr [rsi+2]
    add rsi, 3
    sub r13, 3

    mov r9, rax
    shr r9, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    shl eax, 4
    mov r10, rdx
    shr r10, 4
    or eax, r10d
    and eax, 0x3F
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], al

    shl edx, 2
    mov r10, r8
    shr r10, 6
    or edx, r10d
    and edx, 0x3F
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi+2], dl

    and r8d, 0x3F
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi+3], r8b

    add rdi, 4
    add ecx, 4
    cmp ecx, 76
    jne .Lloop3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor ecx, ecx
    jmp .Lloop3

.Ltail:
    test r13, r13
    jz .Lfinal
    cmp r13, 2
    je .Ltail2

    movzx eax, byte ptr [rsi]
    mov r9, rax
    shr r9, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    shl eax, 4
    and eax, 0x3F
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], al

    mov byte ptr [rdi+2], 0x3D
    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    jmp .Lfinal_nl

.Ltail2:
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]
    mov r9, rax
    shr r9, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [rdi], r9b

    shl eax, 4
    mov r10, rdx
    shr r10, 4
    or eax, r10d
    and eax, 0x3F
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], al

    shl edx, 2
    and edx, 0x3F
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi+2], dl

    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    jmp .Lfinal_nl

.Lfinal:
    test ecx, ecx
    jz .Lwrite
.Lfinal_nl:
    mov byte ptr [rdi], 0x0A
    inc rdi

.Lwrite:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall
