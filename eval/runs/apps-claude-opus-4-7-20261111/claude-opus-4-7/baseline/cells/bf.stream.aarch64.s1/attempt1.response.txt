```
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 8
input_buf: .zero 65536
tape: .zero 30000
.balign 8
compact_code: .zero 65536
.balign 8
bracket_stack: .zero 8000
.balign 8
bracket_table: .zero 262144

.text
_start:
    // Read all of stdin into input_buf
    mov x19, #0              // total bytes read
.Lrd:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x0, #0
    movz x2, #1, lsl #16     // 65536
    sub x2, x2, x19
    cbz x2, .Lrd_end
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le .Lrd_end
    add x19, x19, x0
    b .Lrd
.Lrd_end:
    // x19 = total length
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    // Find first '!'
    mov x21, #0
.Lfb:
    cmp x21, x19
    b.hs .Lnb
    ldrb w0, [x20, x21]
    cmp w0, #0x21
    b.eq .Lgb
    add x21, x21, #1
    b .Lfb
.Lnb:
    mov x22, x19             // input_start position
    mov x23, #0              // input_len
    b .Lparse
.Lgb:
    // x21 = position of '!' = prog_len
    add x22, x21, #1
    sub x23, x19, x22
.Lparse:
    // Parse program: compact commands and build bracket match table
    adrp x27, compact_code
    add x27, x27, :lo12:compact_code
    adrp x28, bracket_stack
    add x28, x28, :lo12:bracket_stack
    adrp x25, bracket_table
    add x25, x25, :lo12:bracket_table
    mov x26, #0              // bracket stack depth
    mov x4, #0               // scan index
    mov x3, #0               // compact index
.Lpl:
    cmp x4, x21
    b.hs .Lpe
    ldrb w1, [x20, x4]
    cmp w1, #0x5B            // [
    b.eq .Lpl_lb
    cmp w1, #0x5D            // ]
    b.eq .Lpl_rb
    cmp w1, #0x3E            // >
    b.eq .Lpl_cmd
    cmp w1, #0x3C            // <
    b.eq .Lpl_cmd
    cmp w1, #0x2B            // +
    b.eq .Lpl_cmd
    cmp w1, #0x2D            // -
    b.eq .Lpl_cmd
    cmp w1, #0x2E            // .
    b.eq .Lpl_cmd
    cmp w1, #0x2C            // ,
    b.eq .Lpl_cmd
    add x4, x4, #1
    b .Lpl
.Lpl_cmd:
    strb w1, [x27, x3]
    add x3, x3, #1
    add x4, x4, #1
    b .Lpl
.Lpl_lb:
    strb w1, [x27, x3]
    str w3, [x28, x26, lsl #2]
    add x26, x26, #1
    add x3, x3, #1
    add x4, x4, #1
    b .Lpl
.Lpl_rb:
    cbz x26, .Lerr
    sub x26, x26, #1
    strb w1, [x27, x3]
    ldr w2, [x28, x26, lsl #2]
    str w3, [x25, x2, lsl #2]
    str w2, [x25, x3, lsl #2]
    add x3, x3, #1
    add x4, x4, #1
    b .Lpl
.Lpe:
    cbnz x26, .Lerr

    // Setup execution state
    adrp x24, tape
    add x24, x24, :lo12:tape

    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x22, x1, x22         // input read pointer (absolute)
    add x23, x22, x23        // input end pointer (absolute)

    mov x20, x27             // compact code base
    mov x21, x3              // compact length
    mov x19, #0              // PC
.Lex:
    cmp x19, x21
    b.hs .Ldone
    ldrb w0, [x20, x19]
    cmp w0, #0x3E
    b.eq .Lr
    cmp w0, #0x3C
    b.eq .Ll
    cmp w0, #0x2B
    b.eq .Lp
    cmp w0, #0x2D
    b.eq .Lm
    cmp w0, #0x2E
    b.eq .Ldo
    cmp w0, #0x2C
    b.eq .Lco
    cmp w0, #0x5B
    b.eq .LL
    b .LR
.Lr:
    add x24, x24, #1
    add x19, x19, #1
    b .Lex
.Ll:
    sub x24, x24, #1
    add x19, x19, #1
    b .Lex
.Lp:
    ldrb w0, [x24]
    add w0, w0, #1
    strb w0, [x24]
    add x19, x19, #1
    b .Lex
.Lm:
    ldrb w0, [x24]
    sub w0, w0, #1
    strb w0, [x24]
    add x19, x19, #1
    b .Lex
.Ldo:
    mov x0, #1
    mov x1, x24
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x19, x19, #1
    b .Lex
.Lco:
    cmp x22, x23
    b.hs .Lcz
    ldrb w0, [x22], #1
    strb w0, [x24]
    add x19, x19, #1
    b .Lex
.Lcz:
    strb wzr, [x24]
    add x19, x19, #1
    b .Lex
.LL:
    ldrb w0, [x24]
    cbnz w0, .LL_in
    ldr w0, [x25, x19, lsl #2]
    add x19, x0, #1
    b .Lex
.LL_in:
    add x19, x19, #1
    b .Lex
.LR:
    ldrb w0, [x24]
    cbz w0, .LR_out
    ldr w0, [x25, x19, lsl #2]
    add x19, x0, #1
    b .Lex
.LR_out:
    add x19, x19, #1
    b .Lex
.Ldone:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
.Lerr:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
```
