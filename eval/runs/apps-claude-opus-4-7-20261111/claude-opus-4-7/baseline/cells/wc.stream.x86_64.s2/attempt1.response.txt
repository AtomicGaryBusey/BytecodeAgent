.globl _start

.bss
.align 16
input_buf: .zero 65536
out_buf: .zero 64

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rsi, [rip + input_buf]
    xor rbx, rbx
count_loop:
    cmp rbx, r12
    jae count_done
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    cmp al, 0x0A
    jne not_nl
    inc r13
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    ja not_ws
is_ws:
    xor r15d, r15d
    jmp count_loop
not_ws:
    test r15, r15
    jnz count_loop
    mov r15d, 1
    inc r14
    jmp count_loop
count_done:
    lea rdi, [rip + out_buf + 63]
    mov byte ptr [rdi], 0x0A
    dec rdi
    mov rax, r12
    call write_num
    mov byte ptr [rdi], 0x20
    dec rdi
    mov rax, r14
    call write_num
    mov byte ptr [rdi], 0x20
    dec rdi
    mov rax, r13
    call write_num
    inc rdi
    lea rax, [rip + out_buf + 64]
    sub rax, rdi
    mov rdx, rax
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

write_num:
    test rax, rax
    jnz wn_loop
    mov byte ptr [rdi], 0x30
    dec rdi
    ret
wn_loop:
    test rax, rax
    jz wn_done
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 0x30
    mov byte ptr [rdi], dl
    dec rdi
    jmp wn_loop
wn_done:
    ret
