.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at "

.bss
.balign 16
input_buf: .zero 65536
out_buf: .zero 64

.text
.globl _start

_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0
  mov x21, #65536
read_loop:
  cmp x20, x21
  beq read_done
  mov x8, #63
  mov x0, #0
  add x1, x19, x20
  sub x2, x21, x20
  svc #0
  cmp x0, #0
  beq read_done
  add x20, x20, x0
  b read_loop
read_done:
  mov x21, x20
  mov x20, #0

  bl skip_ws
  cmp x20, x21
  beq error_out
  bl parse_value
  cbnz w0, error_out
  bl skip_ws
  cmp x20, x21
  bne error_out

  mov x8, #64
  mov x0, #1
  adrp x1, ok_msg
  add x1, x1, :lo12:ok_msg
  mov x2, #3
  svc #0
  mov x8, #93
  mov x0, #0
  svc #0

error_out:
  mov x8, #64
  mov x0, #1
  adrp x1, err_msg
  add x1, x1, :lo12:err_msg
  mov x2, #9
  svc #0
  adrp x9, out_buf
  add x9, x9, :lo12:out_buf
  add x10, x9, #32
  sub x10, x10, #1
  mov w11, #0x0A
  strb w11, [x10]
  mov x12, x20
  mov x14, #10
digit_loop:
  udiv x13, x12, x14
  msub x15, x13, x14, x12
  add w15, w15, #0x30
  sub x10, x10, #1
  strb w15, [x10]
  mov x12, x13
  cbnz x12, digit_loop
  mov x8, #64
  mov x0, #1
  mov x1, x10
  add x2, x9, #32
  sub x2, x2, x10
  svc #0
  mov x8, #93
  mov x0, #1
  svc #0

skip_ws:
1:
  cmp x20, x21
  beq 2f
  ldrb w9, [x19, x20]
  cmp w9, #0x20
  beq 3f
  cmp w9, #0x09
  beq 3f
  cmp w9, #0x0A
  beq 3f
  cmp w9, #0x0D
  beq 3f
  b 2f
3:
  add x20, x20, #1
  b 1b
2:
  ret

parse_value:
  ldrb w9, [x19, x20]
  cmp w9, #0x7B
  beq parse_object
  cmp w9, #0x5B
  beq parse_array
  cmp w9, #0x22
  beq parse_string
  cmp w9, #0x74
  beq parse_true
  cmp w9, #0x66
  beq parse_false
  cmp w9, #0x6E
  beq parse_null
  cmp w9, #0x2D
  beq parse_number
  cmp w9, #0x30
  blt pv_err
  cmp w9, #0x39
  bgt pv_err
  b parse_number
pv_err:
  mov w0, #1
  ret

parse_string:
  add x20, x20, #1
ps_loop:
  cmp x20, x21
  beq ps_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x22
  beq ps_done
  cmp w9, #0x5C
  beq ps_escape
  cmp w9, #0x20
  blt ps_err
  add x20, x20, #1
  b ps_loop
ps_escape:
  add x20, x20, #1
  cmp x20, x21
  beq ps_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x22
  beq ps_simple
  cmp w9, #0x5C
  beq ps_simple
  cmp w9, #0x2F
  beq ps_simple
  cmp w9, #0x62
  beq ps_simple
  cmp w9, #0x66
  beq ps_simple
  cmp w9, #0x6E
  beq ps_simple
  cmp w9, #0x72
  beq ps_simple
  cmp w9, #0x74
  beq ps_simple
  cmp w9, #0x75
  beq ps_u
  b ps_err
ps_simple:
  add x20, x20, #1
  b ps_loop
ps_u:
  add x20, x20, #1
  mov w11, #4
ps_u_loop:
  cmp x20, x21
  beq ps_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x30
  blt ps_err
  cmp w9, #0x39
  ble ps_u_ok
  cmp w9, #0x41
  blt ps_err
  cmp w9, #0x46
  ble ps_u_ok
  cmp w9, #0x61
  blt ps_err
  cmp w9, #0x66
  bgt ps_err
ps_u_ok:
  add x20, x20, #1
  sub w11, w11, #1
  cbnz w11, ps_u_loop
  b ps_loop
ps_done:
  add x20, x20, #1
  mov w0, #0
  ret
ps_eof_err:
  mov x20, x21
ps_err:
  mov w0, #1
  ret

parse_number:
  ldrb w9, [x19, x20]
  cmp w9, #0x2D
  bne pn_after_minus
  add x20, x20, #1
  cmp x20, x21
  beq pn_eof_err
  ldrb w9, [x19, x20]
pn_after_minus:
  cmp w9, #0x30
  blt pn_err
  cmp w9, #0x39
  bgt pn_err
  cmp w9, #0x30
  bne pn_nonzero
  add x20, x20, #1
  b pn_check_frac
pn_nonzero:
  add x20, x20, #1
pn_int_digits:
  cmp x20, x21
  beq pn_ok
  ldrb w9, [x19, x20]
  cmp w9, #0x30
  blt pn_check_frac_w9
  cmp w9, #0x39
  bgt pn_check_frac_w9
  add x20, x20, #1
  b pn_int_digits
pn_check_frac:
  cmp x20, x21
  beq pn_ok
  ldrb w9, [x19, x20]
pn_check_frac_w9:
  cmp w9, #0x2E
  beq pn_frac
  cmp w9, #0x65
  beq pn_exp
  cmp w9, #0x45
  beq pn_exp
  b pn_ok
pn_frac:
  add x20, x20, #1
  cmp x20, x21
  beq pn_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x30
  blt pn_err
  cmp w9, #0x39
  bgt pn_err
  add x20, x20, #1
pn_frac_loop:
  cmp x20, x21
  beq pn_ok
  ldrb w9, [x19, x20]
  cmp w9, #0x30
  blt pn_check_exp_w9
  cmp w9, #0x39
  bgt pn_check_exp_w9
  add x20, x20, #1
  b pn_frac_loop
pn_check_exp_w9:
  cmp w9, #0x65
  beq pn_exp
  cmp w9, #0x45
  beq pn_exp
  b pn_ok
pn_exp:
  add x20, x20, #1
  cmp x20, x21
  beq pn_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x2B
  beq pn_exp_sign
  cmp w9, #0x2D
  beq pn_exp_sign
  b pn_exp_digit_check
pn_exp_sign:
  add x20, x20, #1
  cmp x20, x21
  beq pn_eof_err
  ldrb w9, [x19, x20]
pn_exp_digit_check:
  cmp w9, #0x30
  blt pn_err
  cmp w9, #0x39
  bgt pn_err
  add x20, x20, #1
pn_exp_loop:
  cmp x20, x21
  beq pn_ok
  ldrb w9, [x19, x20]
  cmp w9, #0x30
  blt pn_ok
  cmp w9, #0x39
  bgt pn_ok
  add x20, x20, #1
  b pn_exp_loop
pn_ok:
  mov w0, #0
  ret
pn_eof_err:
  mov x20, x21
pn_err:
  mov w0, #1
  ret

match_byte_adv:
  cmp x20, x21
  beq mb_eof_err
  ldrb w9, [x19, x20]
  cmp w9, w10
  bne mb_err
  add x20, x20, #1
  mov w0, #0
  ret
mb_eof_err:
  mov x20, x21
mb_err:
  mov w0, #1
  ret

parse_true:
  stp x29, x30, [sp, #-16]!
  mov w10, #0x74
  bl match_byte_adv
  cbnz w0, pt_r
  mov w10, #0x72
  bl match_byte_adv
  cbnz w0, pt_r
  mov w10, #0x75
  bl match_byte_adv
  cbnz w0, pt_r
  mov w10, #0x65
  bl match_byte_adv
pt_r:
  ldp x29, x30, [sp], #16
  ret

parse_false:
  stp x29, x30, [sp, #-16]!
  mov w10, #0x66
  bl match_byte_adv
  cbnz w0, pf_r
  mov w10, #0x61
  bl match_byte_adv
  cbnz w0, pf_r
  mov w10, #0x6C
  bl match_byte_adv
  cbnz w0, pf_r
  mov w10, #0x73
  bl match_byte_adv
  cbnz w0, pf_r
  mov w10, #0x65
  bl match_byte_adv
pf_r:
  ldp x29, x30, [sp], #16
  ret

parse_null:
  stp x29, x30, [sp, #-16]!
  mov w10, #0x6E
  bl match_byte_adv
  cbnz w0, pnu_r
  mov w10, #0x75
  bl match_byte_adv
  cbnz w0, pnu_r
  mov w10, #0x6C
  bl match_byte_adv
  cbnz w0, pnu_r
  mov w10, #0x6C
  bl match_byte_adv
pnu_r:
  ldp x29, x30, [sp], #16
  ret

parse_object:
  stp x29, x30, [sp, #-16]!
  add x20, x20, #1
  bl skip_ws
  cmp x20, x21
  beq po_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x7D
  beq po_close
po_member:
  cmp w9, #0x22
  bne po_err
  bl parse_string
  cbnz w0, po_fail
  bl skip_ws
  cmp x20, x21
  beq po_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x3A
  bne po_err
  add x20, x20, #1
  bl skip_ws
  cmp x20, x21
  beq po_eof_err
  bl parse_value
  cbnz w0, po_fail
  bl skip_ws
  cmp x20, x21
  beq po_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x2C
  beq po_comma
  cmp w9, #0x7D
  beq po_close
  b po_err
po_comma:
  add x20, x20, #1
  bl skip_ws
  cmp x20, x21
  beq po_eof_err
  ldrb w9, [x19, x20]
  b po_member
po_close:
  add x20, x20, #1
  mov w0, #0
  b po_ret
po_eof_err:
  mov x20, x21
po_err:
  mov w0, #1
po_fail:
po_ret:
  ldp x29, x30, [sp], #16
  ret

parse_array:
  stp x29, x30, [sp, #-16]!
  add x20, x20, #1
  bl skip_ws
  cmp x20, x21
  beq pa_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x5D
  beq pa_close
pa_elem:
  bl parse_value
  cbnz w0, pa_fail
  bl skip_ws
  cmp x20, x21
  beq pa_eof_err
  ldrb w9, [x19, x20]
  cmp w9, #0x2C
  beq pa_comma
  cmp w9, #0x5D
  beq pa_close
  b pa_err
pa_comma:
  add x20, x20, #1
  bl skip_ws
  cmp x20, x21
  beq pa_eof_err
  b pa_elem
pa_close:
  add x20, x20, #1
  mov w0, #0
  b pa_ret
pa_eof_err:
  mov x20, x21
pa_err:
  mov w0, #1
pa_fail:
pa_ret:
  ldp x29, x30, [sp], #16
  ret
