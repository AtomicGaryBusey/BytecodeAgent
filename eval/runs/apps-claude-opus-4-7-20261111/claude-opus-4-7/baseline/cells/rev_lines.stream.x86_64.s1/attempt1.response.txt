```
.globl _start

.bss
.align 16
buf: .zero 65544

.text
_start:
    lea rbp, [rip + buf]
    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbp
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    js exit_err
    add r12, rax
    jmp read_loop

read_done:
    xor rbx, rbx

line_loop:
    cmp rbx, r12
    jae exit_ok
    mov rcx, rbx
find_nl:
    cmp rcx, r12
    jae no_nl
    cmp byte ptr [rbp + rcx], 0x0A
    je found_nl
    inc rcx
    jmp find_nl

found_nl:
    mov rdi, rbx
    lea rsi, [rcx - 1]
rev1:
    cmp rdi, rsi
    jge rev1_done
    mov al, byte ptr [rbp + rdi]
    mov dl, byte ptr [rbp + rsi]
    mov byte ptr [rbp + rdi], dl
    mov byte ptr [rbp + rsi], al
    inc rdi
    dec rsi
    jmp rev1
rev1_done:
    mov r13, rcx
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    add rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    inc rdx
    syscall
    lea rbx, [r13 + 1]
    jmp line_loop

no_nl:
    mov rdi, rbx
    lea rsi, [r12 - 1]
rev2:
    cmp rdi, rsi
    jge rev2_done
    mov al, byte ptr [rbp + rdi]
    mov dl, byte ptr [rbp + rsi]
    mov byte ptr [rbp + rdi], dl
    mov byte ptr [rbp + rsi], al
    inc rdi
    dec rsi
    jmp rev2
rev2_done:
    mov byte ptr [rbp + r12], 0x0A
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    add rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    inc rdx
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

exit_err:
    mov eax, 60
    mov edi, 1
    syscall
```
