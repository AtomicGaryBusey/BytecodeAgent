.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 16
input_buf:
    .zero 65536
tape:
    .zero 30000
.balign 16
jump_map:
    .zero 262144
bracket_stack:
    .zero 16000
output_buf:
    .zero 4096

.text
.globl _start
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, #0
    movz x21, #0x0001, lsl #16          // 65536
.Lread_loop:
    cmp  x20, x21
    b.ge .Lread_done
    mov  x8, #SYS_read
    mov  x0, #0
    add  x1, x19, x20
    sub  x2, x21, x20
    svc  #0
    cmp  x0, #0
    b.eq .Lread_done
    b.lt .Lexit_err
    add  x20, x20, x0
    b    .Lread_loop
.Lread_done:
    mov  x0, #0
.Lfind:
    cmp  x0, x20
    b.ge .Lno_bang
    ldrb w1, [x19, x0]
    cmp  w1, #0x21
    b.eq .Lfound
    add  x0, x0, #1
    b    .Lfind
.Lfound:
    mov  x21, x0
    add  x22, x19, x0
    add  x22, x22, #1
    add  x23, x19, x20
    b    .Lcheck
.Lno_bang:
    mov  x21, x20
    add  x22, x19, x20
    mov  x23, x22
.Lcheck:
    adrp x25, jump_map
    add  x25, x25, :lo12:jump_map
    adrp x6,  bracket_stack
    add  x6,  x6, :lo12:bracket_stack
    mov  x7,  #0                        // depth
    mov  x0,  #0                        // pc
.Lcheck_loop:
    cmp  x0, x21
    b.ge .Lcheck_done
    ldrb w1, [x19, x0]
    cmp  w1, #0x5B
    b.eq .Lcheck_open
    cmp  w1, #0x5D
    b.eq .Lcheck_close
    add  x0, x0, #1
    b    .Lcheck_loop
.Lcheck_open:
    str  x0, [x6, x7, lsl #3]
    add  x7, x7, #1
    add  x0, x0, #1
    b    .Lcheck_loop
.Lcheck_close:
    cbz  x7, .Lexit_err
    sub  x7, x7, #1
    ldr  x2, [x6, x7, lsl #3]
    str  w0, [x25, x2, lsl #2]
    str  w2, [x25, x0, lsl #2]
    add  x0, x0, #1
    b    .Lcheck_loop
.Lcheck_done:
    cbnz x7, .Lexit_err

    adrp x24, tape
    add  x24, x24, :lo12:tape
    mov  x26, #0
    mov  x27, x24
    adrp x28, output_buf
    add  x28, x28, :lo12:output_buf
    mov  x29, #0
    mov  w17, #0x1000                   // 4096

.Lrun_loop:
    cmp  x26, x21
    b.ge .Lrun_done
    ldrb w0, [x19, x26]
    cmp  w0, #0x3E
    b.eq .Lop_right
    cmp  w0, #0x3C
    b.eq .Lop_left
    cmp  w0, #0x2B
    b.eq .Lop_plus
    cmp  w0, #0x2D
    b.eq .Lop_minus
    cmp  w0, #0x2E
    b.eq .Lop_dot
    cmp  w0, #0x2C
    b.eq .Lop_comma
    cmp  w0, #0x5B
    b.eq .Lop_lbracket
    cmp  w0, #0x5D
    b.eq .Lop_rbracket
    add  x26, x26, #1
    b    .Lrun_loop

.Lop_right:
    add  x27, x27, #1
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_left:
    sub  x27, x27, #1
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_plus:
    ldrb w0, [x27]
    add  w0, w0, #1
    strb w0, [x27]
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_minus:
    ldrb w0, [x27]
    sub  w0, w0, #1
    strb w0, [x27]
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_dot:
    ldrb w0, [x27]
    strb w0, [x28, x29]
    add  x29, x29, #1
    cmp  x29, x17
    b.lt .Ldot_cont
    mov  x8, #SYS_write
    mov  x0, #1
    mov  x1, x28
    mov  x2, x29
    svc  #0
    mov  x29, #0
.Ldot_cont:
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_comma:
    cmp  x22, x23
    b.ge .Lcomma_eof
    ldrb w0, [x22], #1
    strb w0, [x27]
    add  x26, x26, #1
    b    .Lrun_loop
.Lcomma_eof:
    strb wzr, [x27]
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_lbracket:
    ldrb w0, [x27]
    cbnz w0, .Llb_cont
    ldr  w1, [x25, x26, lsl #2]
    add  x26, x1, #1
    b    .Lrun_loop
.Llb_cont:
    add  x26, x26, #1
    b    .Lrun_loop
.Lop_rbracket:
    ldrb w0, [x27]
    cbz  w0, .Lrb_cont
    ldr  w1, [x25, x26, lsl #2]
    add  x26, x1, #1
    b    .Lrun_loop
.Lrb_cont:
    add  x26, x26, #1
    b    .Lrun_loop

.Lrun_done:
    cbz  x29, .Lexit_ok
    mov  x8, #SYS_write
    mov  x0, #1
    mov  x1, x28
    mov  x2, x29
    svc  #0
.Lexit_ok:
    mov  x8, #SYS_exit
    mov  x0, #0
    svc  #0

.Lexit_err:
    mov  x8, #SYS_exit
    mov  x0, #1
    svc  #0
