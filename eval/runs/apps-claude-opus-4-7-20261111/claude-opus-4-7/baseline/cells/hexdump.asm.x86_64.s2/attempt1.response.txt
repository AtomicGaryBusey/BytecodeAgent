.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:  .zero 65536
.balign 16
output_buf: .zero 400000

.text
.globl _start
_start:
    xor r12, r12

read_loop:
    mov edx, 65536
    sub edx, r12d
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov eax, 0
    mov edi, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + output_buf]
    lea r14, [rip + input_buf]
    lea rbp, [rip + hex_digits]
    xor r13, r13

row_loop:
    mov rax, r12
    sub rax, r13
    jz final_line
    mov ecx, 16
    cmp rax, rcx
    cmovb ecx, eax
    mov r15d, ecx

    mov edx, r13d
    mov ecx, 8
1:
    rol edx, 4
    mov eax, edx
    and eax, 0xf
    movzx eax, byte ptr [rbp + rax]
    mov [rbx], al
    inc rbx
    dec ecx
    jnz 1b

    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    add rbx, 2

    xor r8d, r8d
2:
    cmp r8d, r15d
    jge 3f
    lea rax, [r14 + r13]
    movzx edx, byte ptr [rax + r8]
    mov eax, edx
    shr eax, 4
    movzx eax, byte ptr [rbp + rax]
    mov [rbx], al
    mov eax, edx
    and eax, 0xf
    movzx eax, byte ptr [rbp + rax]
    mov [rbx + 1], al
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp 4f
3:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
4:
    cmp r8d, 7
    jne 5f
    mov byte ptr [rbx], 0x20
    inc rbx
5:
    inc r8d
    cmp r8d, 16
    jl 2b

    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2

    xor r8d, r8d

    jge 8f
    lea rax, [r14 + r13]

    cmp eax, 0x20

9:
    mov [rbx], al
    inc rbx
    inc r8d
    jmp 6b
8:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2

    add r13, r15
    jmp row_loop

final_line:
    mov edx, r12d
    mov ecx, 8
10:
    rol edx, 4
    mov eax, edx
    and eax, 0xf
    movzx eax, byte ptr [rbp + rax]
    mov [rbx], al
    inc rbx
    dec ecx
    jnz 10b
    mov byte ptr [rbx], 0x0a
    inc rbx

    lea rsi, [rip + output_buf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
