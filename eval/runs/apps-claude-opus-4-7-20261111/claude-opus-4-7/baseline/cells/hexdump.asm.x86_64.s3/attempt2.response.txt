.globl _start

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.bss
.align 16
input_buf: .zero 65536
.align 16
output_buf: .zero 524288

.text
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r14, [rip + output_buf]
    lea rbp, [rip + hex_chars]
    xor r15, r15

row_loop:
    cmp r15, r13
    jge rows_done

    mov rbx, r13
    sub rbx, r15
    cmp rbx, 16
    jbe got_n
    mov rbx, 16
got_n:
    mov rax, r15
    mov rcx, 7
hex_off:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r14 + rcx], dl
    shr rax, 4
    dec rcx
    jns hex_off
    add r14, 8
    mov byte ptr [r14], 0x20
    mov byte ptr [r14 + 1], 0x20
    add r14, 2

    mov r8, r12
    add r8, r15
    xor rcx, rcx
hex_byte_loop:
    cmp rcx, 16
    jge hex_byte_done
    cmp rcx, rbx
    jge hex_byte_space
    movzx rax, byte ptr [r8 + rcx]
    mov rdx, rax
    shr rdx, 4
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r14], dl
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r14 + 1], dl
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
    jmp hex_byte_next
hex_byte_space:
    mov byte ptr [r14], 0x20
    mov byte ptr [r14 + 1], 0x20
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
hex_byte_next:
    inc rcx
    cmp rcx, 8
    jne no_extra
    mov byte ptr [r14], 0x20
    inc r14
no_extra:
    jmp hex_byte_loop
hex_byte_done:
    mov byte ptr [r14], 0x20
    mov byte ptr [r14 + 1], 0x7c
    add r14, 2

    xor rcx, rcx
char_loop:
    cmp rcx, rbx
    jge char_done
    movzx rax, byte ptr [r8 + rcx]
    cmp al, 0x20
    jb char_dot
    cmp al, 0x7e
    ja char_dot
    jmp char_write
char_dot:
    mov al, 0x2e
char_write:
    mov byte ptr [r14], al
    inc r14
    inc rcx
    jmp char_loop
char_done:
    mov byte ptr [r14], 0x7c
    mov byte ptr [r14 + 1], 0x0a
    add r14, 2

    add r15, 16
    jmp row_loop

rows_done:
    mov rax, r13
    mov rcx, 7
hex_fin:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r14 + rcx], dl
    shr rax, 4
    dec rcx
    jns hex_fin
    mov byte ptr [r14 + 8], 0x0a
    add r14, 9

    lea rsi, [rip + output_buf]
    mov rdx, r14
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov rax, 1
    mov rdi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov rax, 60
    xor rdi, rdi
    syscall
