.globl _start

.bss
.p2align 4
inbuf:
    .zero 40000
.p2align 4
lines:
    .zero 8008
.p2align 4
outbuf:
    .zero 40000

.text

_start:
    xor r12, r12
.Lread_loop:
    mov edx, 40000
    sub edx, r12d
    jle .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [inbuf + r12]
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:

    xor r13, r13
    xor rsi, rsi
    xor rdi, rdi
.Lparse_loop:
    cmp rdi, r12
    jae .Lparse_tail
    cmp byte ptr [inbuf + rdi], 0x0A
    je .Lparse_nl
    inc rdi
    jmp .Lparse_loop
.Lparse_nl:
    mov rax, r13
    shl rax, 3
    mov [lines + rax], esi
    mov rcx, rdi
    sub rcx, rsi
    mov [lines + rax + 4], ecx
    inc r13
    inc rdi
    mov rsi, rdi
    jmp .Lparse_loop
.Lparse_tail:
    cmp rsi, r12
    jae .Lparse_done
    mov rax, r13
    shl rax, 3
    mov [lines + rax], esi
    mov rcx, r12
    sub rcx, rsi
    mov [lines + rax + 4], ecx
    inc r13
.Lparse_done:
    test r13, r13
    jz .Ldo_exit

    mov rbp, 1
.Lsort_outer:
    cmp rbp, r13
    jge .Lsort_done
    mov rax, rbp
    shl rax, 3
    mov r14, [lines + rax]
    mov rbx, rbp
    dec rbx
.Lsort_inner:
    test rbx, rbx
    js .Lsort_place
    mov rax, rbx
    shl rax, 3
    mov r15, [lines + rax]

    mov r8d, r14d
    lea rsi, [inbuf + r8]
    mov rcx, r14
    shr rcx, 32

    mov r8d, r15d
    lea rdi, [inbuf + r8]
    mov rdx, r15
    shr rdx, 32

    call less
    test eax, eax
    jz .Lsort_place

    mov rax, rbx
    shl rax, 3
    mov [lines + rax + 8], r15
    dec rbx
    jmp .Lsort_inner
.Lsort_place:
    mov rax, rbx
    shl rax, 3
    mov [lines + rax + 8], r14
    inc rbp
    jmp .Lsort_outer
.Lsort_done:

    xor r12, r12
    xor rbp, rbp
.Lout_loop:
    cmp rbp, r13
    jge .Lout_done
    mov rax, rbp
    shl rax, 3
    mov r8d, [lines + rax]
    mov ecx, [lines + rax + 4]
    lea rsi, [inbuf + r8]
    lea rdi, [outbuf + r12]
    add r12, rcx
    rep movsb
    mov byte ptr [outbuf + r12], 0x0A
    inc r12
    inc rbp
    jmp .Lout_loop
.Lout_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r12
    syscall

.Ldo_exit:
    mov eax, 60
    xor edi, edi
    syscall

less:
    mov r8, rcx
    mov r9, rdx
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz .Lless_eq
    repe cmpsb
    jb .Lless_true
    ja .Lless_false
.Lless_eq:
    cmp r8, r9
    jb .Lless_true
.Lless_false:
    xor eax, eax
    ret
.Lless_true:
    mov eax, 1
    ret
