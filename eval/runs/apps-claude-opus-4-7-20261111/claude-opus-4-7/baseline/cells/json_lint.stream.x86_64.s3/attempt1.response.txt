```
.globl _start

.section .rodata
prefix:
    .ascii "error at "
newline:
    .byte 0x0a
ok_msg:
    .ascii "ok\n"

.bss
buffer:
    .zero 65536
digbuf:
    .zero 32

.text
_start:
    lea rbx, [rip + buffer]
    mov r12, rbx
read_loop:
    lea rax, [rip + buffer + 65536]
    sub rax, r12
    jz read_done
    mov rdx, rax
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rsi, rbx
    mov rdi, r12

    call skip_ws
    call parse_value
    jc parse_error
    call skip_ws
    cmp rsi, rdi
    jne parse_error

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

parse_error:
    sub rsi, rbx
    mov rax, rsi

    lea rdi, [rip + digbuf + 16]
    mov rcx, 10
conv_loop:
    xor rdx, rdx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conv_loop

    mov r12, rdi
    lea rax, [rip + digbuf + 16]
    sub rax, rdi
    mov r13, rax

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + prefix]
    mov rdx, 9
    syscall

    mov rax, 1
    mov rdi, 1
    mov rsi, r12
    mov rdx, r13
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    mov rax, 60
    mov rdi, 1
    syscall

skip_ws:
sw_loop:
    cmp rsi, rdi
    je sw_done
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je sw_cont
    cmp al, 0x09
    je sw_cont
    cmp al, 0x0a
    je sw_cont
    cmp al, 0x0d
    je sw_cont
sw_done:
    ret
sw_cont:
    inc rsi
    jmp sw_loop

parse_value:
    cmp rsi, rdi
    je pv_fail
    mov al, byte ptr [rsi]
    cmp al, 0x22
    je parse_string
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6e
    je parse_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb pv_fail
    cmp al, 0x39
    ja pv_fail
    jmp parse_number
pv_fail:
    stc
    ret

parse_string:
    inc rsi
ps_loop:
    cmp rsi, rdi
    je ps_fail
    mov al, byte ptr [rsi]
    cmp al, 0x22
    je ps_close
    cmp al, 0x5c
    je ps_esc
    cmp al, 0x20
    jb ps_fail
    inc rsi
    jmp ps_loop
ps_close:
    inc rsi
    clc
    ret
ps_esc:
    inc rsi
    cmp rsi, rdi
    je ps_fail
    mov al, byte ptr [rsi]
    cmp al, 0x22
    je ps_esc_ok
    cmp al, 0x5c
    je ps_esc_ok
    cmp al, 0x2f
    je ps_esc_ok
    cmp al, 0x62
    je ps_esc_ok
    cmp al, 0x66
    je ps_esc_ok
    cmp al, 0x6e
    je ps_esc_ok
    cmp al, 0x72
    je ps_esc_ok
    cmp al, 0x74
    je ps_esc_ok
    cmp al, 0x75
    je ps_esc_u
ps_fail:
    stc
    ret
ps_esc_ok:
    inc rsi
    jmp ps_loop
ps_esc_u:
    inc rsi
    mov ecx, 4
ps_esc_u_loop:
    cmp rsi, rdi
    je ps_fail
    mov al, byte ptr [rsi]
    cmp al, 0x30
    jb ps_fail
    cmp al, 0x39
    jbe ps_esc_u_next
    cmp al, 0x41
    jb ps_fail
    cmp al, 0x46
    jbe ps_esc_u_next
    cmp al, 0x61
    jb ps_fail
    cmp al, 0x66
    ja ps_fail
ps_esc_u_next:
    inc rsi
    dec ecx
    jnz ps_esc_u_loop
    jmp ps_loop

parse_number:
    mov al, byte ptr [rsi]
    cmp al, 0x2d
    jne pn_int_start
    inc rsi
    cmp rsi, rdi
    je pn_fail
    mov al, byte ptr [rsi]
pn_int_start:
    cmp al, 0x30
    je pn_zero
    cmp al, 0x31
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc rsi
pn_int_loop:
    cmp rsi, rdi
    je pn_done
    mov al, byte ptr [rsi]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    inc rsi
    jmp pn_int_loop
pn_zero:
    inc rsi
    cmp rsi, rdi
    je pn_done
    mov al, byte ptr [rsi]
pn_frac:
    cmp al, 0x2e
    jne pn_exp_chk
    inc rsi
    cmp rsi, rdi
    je pn_fail
    mov al, byte ptr [rsi]
    cmp al, 0x30
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc rsi
pn_frac_loop:
    cmp rsi, rdi
    je pn_done
    mov al, byte ptr [rsi]
    cmp al, 0x30
    jb pn_exp_chk
    cmp al, 0x39
    ja pn_exp_chk
    inc rsi
    jmp pn_frac_loop
pn_exp_chk:
    cmp al, 0x65
    je pn_exp
    cmp al, 0x45
    je pn_exp
    jmp pn_done
pn_exp:
    inc rsi
    cmp rsi, rdi
    je pn_fail
    mov al, byte ptr [rsi]
    cmp al, 0x2b
    je pn_exp_sign
    cmp al, 0x2d
    je pn_exp_sign
    jmp pn_exp_d
pn_exp_sign:
    inc rsi
    cmp rsi, rdi
    je pn_fail
    mov al, byte ptr [rsi]
pn_exp_d:
    cmp al, 0x30
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc rsi
pn_exp_loop:
    cmp rsi, rdi
    je pn_done
    mov al, byte ptr [rsi]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    ja pn_done
    inc rsi
    jmp pn_exp_loop
pn_done:
    clc
    ret
pn_fail:
    stc
    ret

parse_true:
    cmp rsi, rdi
    je pt_fail
    cmp byte ptr [rsi], 0x74
    jne pt_fail
    inc rsi
    cmp rsi, rdi
    je pt_fail
    cmp byte ptr [rsi], 0x72
    jne pt_fail
    inc rsi
    cmp rsi, rdi
    je pt_fail
    cmp byte ptr [rsi], 0x75
    jne pt_fail
    inc rsi
    cmp rsi, rdi
    je pt_fail
    cmp byte ptr [rsi], 0x65
    jne pt_fail
    inc rsi
    clc
    ret
pt_fail:
    stc
    ret

parse_false:
    cmp rsi, rdi
    je pf_fail
    cmp byte ptr [rsi], 0x66
    jne pf_fail
    inc rsi
    cmp rsi, rdi
    je pf_fail
    cmp byte ptr [rsi], 0x61
    jne pf_fail
    inc rsi
    cmp rsi, rdi
    je pf_fail
    cmp byte ptr [rsi], 0x6c
    jne pf_fail
    inc rsi
    cmp rsi, rdi
    je pf_fail
    cmp byte ptr [rsi], 0x73
    jne pf_fail
    inc rsi
    cmp rsi, rdi
    je pf_fail
    cmp byte ptr [rsi], 0x65
    jne pf_fail
    inc rsi
    clc
    ret
pf_fail:
    stc
    ret

parse_null:
    cmp rsi, rdi
    je pnl_fail
    cmp byte ptr [rsi], 0x6e
    jne pnl_fail
    inc rsi
    cmp rsi, rdi
    je pnl_fail
    cmp byte ptr [rsi], 0x75
    jne pnl_fail
    inc rsi
    cmp rsi, rdi
    je pnl_fail
    cmp byte ptr [rsi], 0x6c
    jne pnl_fail
    inc rsi
    cmp rsi, rdi
    je pnl_fail
    cmp byte ptr [rsi], 0x6c
    jne pnl_fail
    inc rsi
    clc
    ret
pnl_fail:
    stc
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je po_fail
    cmp byte ptr [rsi], 0x7d
    je po_close
po_member:
    cmp byte ptr [rsi], 0x22
    jne po_fail
    call parse_string
    jc po_ret_fail
    call skip_ws
    cmp rsi, rdi
    je po_fail
    cmp byte ptr [rsi], 0x3a
    jne po_fail
    inc rsi
    call skip_ws
    call parse_value
    jc po_ret_fail
    call skip_ws
    cmp rsi, rdi
    je po_fail
    mov al, byte ptr [rsi]
    cmp al, 0x7d
    je po_close
    cmp al, 0x2c
    jne po_fail
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je po_fail
    jmp po_member
po_close:
    inc rsi
    clc
    ret
po_fail:
    stc
po_ret_fail:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je pa_fail
    cmp byte ptr [rsi], 0x5d
    je pa_close
pa_value:
    call parse_value
    jc pa_ret_fail
    call skip_ws
    cmp rsi, rdi
    je pa_fail
    mov al, byte ptr [rsi]
    cmp al, 0x5d
    je pa_close
    cmp al, 0x2c
    jne pa_fail
    inc rsi
    call skip_ws
    jmp pa_value
pa_close:
    inc rsi
    clc
    ret
pa_fail:
    stc
pa_ret_fail:
    ret
```
