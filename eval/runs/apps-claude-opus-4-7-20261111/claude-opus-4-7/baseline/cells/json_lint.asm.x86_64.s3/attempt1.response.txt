.globl _start

_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
.read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz .read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle .read_done
    add r13, rax
    jmp .read_loop
.read_done:
    xor ebx, ebx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne .err_at_rbx_main
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
.err_at_rbx_main:
    mov rax, rbx
    jmp .error

skip_ws:
.sw_loop:
    cmp rbx, r13
    jae .sw_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x20
    je .sw_adv
    cmp al, 0x09
    je .sw_adv
    cmp al, 0x0A
    je .sw_adv
    cmp al, 0x0D
    je .sw_adv
    jmp .sw_done
.sw_adv:
    inc rbx
    jmp .sw_loop
.sw_done:
    ret

parse_value:
    cmp rbx, r13
    jae .pv_err_eof
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb .pv_err
    cmp al, '9'
    jbe parse_number
    cmp al, 't'
    je .pv_true
    cmp al, 'f'
    je .pv_false
    cmp al, 'n'
    je .pv_null
.pv_err:
    mov rax, rbx
    jmp .error
.pv_err_eof:
    mov rax, r13
    jmp .error
.pv_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp parse_literal_impl
.pv_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp parse_literal_impl
.pv_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp parse_literal_impl

parse_literal_impl:
    xor edx, edx
.pli_loop:
    cmp rbx, r13
    jae .pli_err
    mov al, byte ptr [rsi + rdx]
    cmp al, byte ptr [r12 + rbx]
    jne .pli_err
    inc rbx
    inc rdx
    cmp rdx, rcx
    jb .pli_loop
    ret
.pli_err:
    mov rax, rbx
    jmp .error

parse_string:
    inc rbx
.ps_loop:
    cmp rbx, r13
    jae .ps_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '"'
    je .ps_done
    cmp al, '\\'
    je .ps_esc
    cmp al, 0x20
    jb .ps_err
    inc rbx
    jmp .ps_loop
.ps_done:
    inc rbx
    ret
.ps_esc:
    inc rbx
    cmp rbx, r13
    jae .ps_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '"'
    je .ps_adv
    cmp al, '\\'
    je .ps_adv
    cmp al, '/'
    je .ps_adv
    cmp al, 'b'
    je .ps_adv
    cmp al, 'f'
    je .ps_adv
    cmp al, 'n'
    je .ps_adv
    cmp al, 'r'
    je .ps_adv
    cmp al, 't'
    je .ps_adv
    cmp al, 'u'
    je .ps_u
    jmp .ps_err
.ps_adv:
    inc rbx
    jmp .ps_loop
.ps_u:
    inc rbx
    mov ecx, 4
.ps_hex:
    cmp rbx, r13
    jae .ps_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb .ps_err
    cmp al, '9'
    jbe .ps_hex_ok
    cmp al, 'A'
    jb .ps_err
    cmp al, 'F'
    jbe .ps_hex_ok
    cmp al, 'a'
    jb .ps_err
    cmp al, 'f'
    ja .ps_err
.ps_hex_ok:
    inc rbx
    dec ecx
    jnz .ps_hex
    jmp .ps_loop
.ps_err:
    mov rax, rbx
    jmp .error

parse_number:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '-'
    jne .pn_main
    inc rbx
    cmp rbx, r13
    jae .pn_err
    movzx eax, byte ptr [r12 + rbx]
.pn_main:
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    cmp al, '0'
    je .pn_zero
    inc rbx
.pn_d1:
    cmp rbx, r13
    jae .pn_frac
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb .pn_frac
    cmp al, '9'
    ja .pn_frac
    inc rbx
    jmp .pn_d1
.pn_zero:
    inc rbx
.pn_frac:
    cmp rbx, r13
    jae .pn_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '.'
    jne .pn_exp
    inc rbx
    cmp rbx, r13
    jae .pn_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    inc rbx
.pn_fd:
    cmp rbx, r13
    jae .pn_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb .pn_exp
    cmp al, '9'
    ja .pn_exp
    inc rbx
    jmp .pn_fd
.pn_exp:
    cmp rbx, r13
    jae .pn_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 'e'
    je .pn_e
    cmp al, 'E'
    je .pn_e
    jmp .pn_done
.pn_e:
    inc rbx
    cmp rbx, r13
    jae .pn_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '+'
    je .pn_esign
    cmp al, '-'
    je .pn_esign
    jmp .pn_edig1
.pn_esign:
    inc rbx
    cmp rbx, r13
    jae .pn_err
    movzx eax, byte ptr [r12 + rbx]
.pn_edig1:
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    inc rbx
.pn_edigs:
    cmp rbx, r13
    jae .pn_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb .pn_done
    cmp al, '9'
    ja .pn_done
    inc rbx
    jmp .pn_edigs
.pn_done:
    ret
.pn_err:
    mov rax, rbx
    jmp .error

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae .po_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '}'
    je .po_end
.po_member:
    cmp rbx, r13
    jae .po_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '"'
    jne .po_err
    call parse_string
    call skip_ws
    cmp rbx, r13
    jae .po_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, ':'
    jne .po_err
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae .po_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '}'
    je .po_end
    cmp al, ','
    jne .po_err
    inc rbx
    call skip_ws
    jmp .po_member
.po_end:
    inc rbx
    ret
.po_err:
    mov rax, rbx
    jmp .error

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae .pa_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, ']'
    je .pa_end
.pa_value:

    call skip_ws
    cmp rbx, r13
    jae .pa_err
    movzx eax, byte ptr [r12 + rbx]
    cmp al, ']'
    je .pa_end
    cmp al, ','
    jne .pa_err
    inc rbx
    call skip_ws
    jmp .pa_value
.pa_end:
    inc rbx
    ret
.pa_err:
    mov rax, rbx
    jmp .error

    mov byte ptr [r8 + 31], 0x0A
    mov r9d, 30
    mov ecx, 10

    lea rsi, [rip + err_prefix]
    mov edx, 9
    syscall
    lea rsi, [r8 + r9]
    mov edx, 32

    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
