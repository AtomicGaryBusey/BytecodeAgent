.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
	lea rbx, [rip + inbuf]
	mov r12, rbx
	mov r13d, 49152

read_loop:
	xor eax, eax
	xor edi, edi
	mov rsi, rbx
	mov edx, r13d
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	sub r13d, eax
	jnz read_loop

read_done:
	mov r14, rbx
	sub r14, r12
	test r14, r14
	jz exit_ok

	mov rsi, r12
	lea rdi, [rip + outbuf]
	xor r15d, r15d

encode_loop:
	cmp r14, 3
	jb encode_tail

	movzx eax, byte ptr [rsi]
	shl eax, 16
	movzx ecx, byte ptr [rsi + 1]
	shl ecx, 8
	or eax, ecx
	movzx ecx, byte ptr [rsi + 2]
	or eax, ecx
	add rsi, 3
	sub r14, 3

	mov ecx, eax
	shr ecx, 18
	and ecx, 63
	call emit
	mov ecx, eax
	shr ecx, 12
	and ecx, 63
	call emit
	mov ecx, eax
	shr ecx, 6
	and ecx, 63
	call emit
	mov ecx, eax
	and ecx, 63
	call emit
	jmp encode_loop

encode_tail:
	test r14, r14
	jz finish_out
	cmp r14, 1
	je tail_1

	movzx eax, byte ptr [rsi]
	shl eax, 16
	movzx ecx, byte ptr [rsi + 1]
	shl ecx, 8
	or eax, ecx
	mov ecx, eax
	shr ecx, 18
	and ecx, 63
	call emit
	mov ecx, eax
	shr ecx, 12
	and ecx, 63
	call emit
	mov ecx, eax
	shr ecx, 6
	and ecx, 63
	call emit
	mov ecx, 64
	call emit
	jmp finish_out

tail_1:
	movzx eax, byte ptr [rsi]
	shl eax, 16
	mov ecx, eax
	shr ecx, 18
	and ecx, 63
	call emit
	mov ecx, eax
	shr ecx, 12
	and ecx, 63
	call emit
	mov ecx, 64
	call emit
	mov ecx, 64
	call emit

finish_out:
	test r15d, r15d
	jz do_write
	mov byte ptr [rdi], 10
	inc rdi

do_write:
	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall

exit_ok:
	mov eax, 60
	xor edi, edi
	syscall

emit:
	push rax
	cmp ecx, 64
	jae 1f
	lea rax, [rip + alphabet]
	mov al, byte ptr [rax + rcx]
	jmp 2f
1:
	mov al, '='
2:
	mov byte ptr [rdi], al
	inc rdi
	inc r15d
	cmp r15d, 76
	jb 3f
	mov byte ptr [rdi], 10
	inc rdi
	xor r15d, r15d
3:
	pop rax
	ret

.bss
.align 16
inbuf:
.space 49152
outbuf:
.space 70000
