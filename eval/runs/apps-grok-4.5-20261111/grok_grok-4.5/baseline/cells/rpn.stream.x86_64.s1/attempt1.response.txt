.globl _start

.section .rodata
error_msg:
.ascii "error\n"
min_digits:
.ascii "9223372036854775808"

.bss
.align 16
input_buf:
.zero 65536
.align 8
rpn_stack:
.zero 8192
.align 8
out_buf:
.zero 524288

.text
_start:
lea rsi, [rip + input_buf]
mov rdx, 65536
mov r12, rsi
.read_loop:
mov rax, 0
xor rdi, rdi
syscall
cmp rax, 0
jle .read_done
add rsi, rax
sub rdx, rax
jnz .read_loop
.read_done:
mov r13, rsi
lea r14, [rip + out_buf]
mov rbx, r12
.line_loop:
cmp rbx, r13
jae .done_all
mov rsi, rbx
.find_nl:
cmp rsi, r13
jae .is_last
cmp byte ptr [rsi], 0x0A
je .got_nl
inc rsi
jmp .find_nl
.got_nl:
mov r8, rsi
lea r9, [rsi + 1]
jmp .prep_line
.is_last:
mov r8, r13
mov r9, r13
.prep_line:
mov rax, r8
sub rax, rbx
cmp rax, 1000
ja .line_error
lea rbp, [rip + rpn_stack]
mov r15, rbp
mov rsi, rbx
.token_loop:
cmp rsi, r8
jae .tokens_done
movzx eax, byte ptr [rsi]
cmp al, 0x20
je .skip1
cmp al, 0x09
je .skip1
jmp .start_tok
.skip1:
inc rsi
jmp .token_loop
.start_tok:
mov rdi, rsi
.tok_end_loop:
cmp rsi, r8
jae .tok_ended
movzx eax, byte ptr [rsi]
cmp al, 0x20
je .tok_ended
cmp al, 0x09
je .tok_ended
inc rsi
jmp .tok_end_loop
.tok_ended:
mov rcx, rsi
sub rcx, rdi
cmp rcx, 1
jne .try_number
movzx r10d, byte ptr [rdi]
cmp r10b, '+'
je .is_op
cmp r10b, '-'
je .is_op
cmp r10b, '*'
je .is_op
cmp r10b, '/'
je .is_op
cmp r10b, '%'
je .is_op
.try_number:
mov r10, rdi
xor r11d, r11d
cmp byte ptr [r10], '-'
jne .num_no_sign
inc r10
mov r11d, 1
cmp r10, rsi
je .line_error
.num_no_sign:
cmp r10, rsi
je .line_error
xor eax, eax
xor ecx, ecx
.num_loop:
cmp r10, rsi
jae .num_done
movzx edx, byte ptr [r10]
cmp dl, '0'
jb .line_error
cmp dl, '9'
ja .line_error
inc ecx
cmp ecx, 18
ja .line_error
imul rax, rax, 10
sub dl, '0'
add rax, rdx
inc r10
jmp .num_loop
.num_done:
test r11d, r11d
jz .do_push
neg rax
.do_push:
mov qword ptr [r15], rax
add r15, 8
jmp .token_loop
.is_op:
mov rax, r15
sub rax, rbp
cmp rax, 16
jb .line_error
sub r15, 8
mov rcx, qword ptr [r15]
sub r15, 8
mov rax, qword ptr [r15]
cmp r10b, '+'
je .c_add
cmp r10b, '-'
je .c_sub
cmp r10b, '*'
je .c_mul
cmp r10b, '/'
je .c_div
test rcx, rcx
jz .line_error
mov r11, 0x8000000000000000
cmp rax, r11
jne 1f
cmp rcx, -1
je .line_error
1:
cqo
idiv rcx
mov rax, rdx
jmp .do_push
.c_div:
test rcx, rcx
jz .line_error
mov r11, 0x8000000000000000
cmp rax, r11
jne 1f
cmp rcx, -1
je .line_error
1:
cqo
idiv rcx
jmp .do_push
.c_add:
add rax, rcx
jmp .do_push
.c_sub:
sub rax, rcx
jmp .do_push
.c_mul:
imul rax, rcx
jmp .do_push
.tokens_done:
mov rax, r15
sub rax, rbp
cmp rax, 8
jne .line_error
mov rax, qword ptr [r15 - 8]
call print_i64
jmp .next_line
.line_error:
lea rsi, [rip + error_msg]
mov ecx, 6
.err_copy:
mov al, byte ptr [rsi]
mov byte ptr [r14], al
inc rsi
inc r14
dec ecx
jnz .err_copy
.next_line:
mov rbx, r9
jmp .line_loop
.done_all:
lea rsi, [rip + out_buf]
mov rdx, r14
sub rdx, rsi
jz .exit0
.write_more:
mov rax, 1
mov rdi, 1
syscall
cmp rax, 0
jle .exit0
add rsi, rax
sub rdx, rax
jnz .write_more
.exit0:
mov rax, 60
xor rdi, rdi
syscall

print_i64:
test rax, rax
jnz .p_nz
mov byte ptr [r14], '0'
inc r14
jmp .p_addnl
.p_nz:
bt rax, 63
jnc .p_positive
mov byte ptr [r14], '-'
inc r14
mov r11, 0x8000000000000000
cmp rax, r11
jne .p_negate
lea rsi, [rip + min_digits]
mov ecx, 19
.p_cpmin:
mov al, byte ptr [rsi]
mov byte ptr [r14], al
inc rsi
inc r14
dec ecx
jnz .p_cpmin
jmp .p_addnl
.p_negate:
neg rax
.p_positive:
sub rsp, 32
mov rdi, rsp
xor ecx, ecx
.p_extract:
mov r11, 10
xor rdx, rdx
div r11
add dl, '0'
mov byte ptr [rdi + rcx], dl
inc ecx
test rax, rax
jnz .p_extract
.p_reverse:
dec ecx
mov al, byte ptr [rdi + rcx]
mov byte ptr [r14], al
inc r14
test ecx, ecx
jnz .p_reverse
add rsp, 32
.p_addnl:
mov byte ptr [r14], 0x0A
inc r14
ret
