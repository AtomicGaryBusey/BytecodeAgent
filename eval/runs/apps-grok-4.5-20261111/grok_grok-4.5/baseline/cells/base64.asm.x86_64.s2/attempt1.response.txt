.globl _start
.equ INPUT_MAX, 49152

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input:
.zero INPUT_MAX
output:
.zero 70000

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, INPUT_MAX
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    xor r13, r13
    lea r15, [rip + alphabet]

encode_loop:
    cmp r12, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx
    add rsi, 3
    sub r12, 3

    mov r8d, eax

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    jmp encode_loop

encode_tail:
    cmp r12, 0
    je encode_finish
    cmp r12, 1
    je encode_1

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    mov r8d, eax

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov al, '='
    call emit_char
    jmp encode_finish

encode_1:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov r8d, eax

    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    mov al, byte ptr [r15 + rcx]
    call emit_char

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char

encode_finish:
    test r13, r13
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz exit0
    mov eax, 1
    mov edi, 1
    syscall

exit0:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r13, r13
1:
    ret
