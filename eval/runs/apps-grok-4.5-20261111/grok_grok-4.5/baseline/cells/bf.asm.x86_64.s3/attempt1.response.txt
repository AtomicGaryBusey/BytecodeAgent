.globl _start
.text
_start:
    lea r8, [rip + stdin_buf]
    mov rsi, r8
    mov edx, 65536
read_loop:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov rbx, rsi
    sub rbx, r8
    mov rsi, r8
    mov rcx, rbx
    test rcx, rcx
    jz empty_input
find_bang:
    cmp byte ptr [rsi], 0x21
    je found_bang
    inc rsi
    dec rcx
    jnz find_bang
    mov r9, r8
    mov r10, rbx
    mov r14, rsi
    mov r15, rsi
    jmp validate_setup
found_bang:
    mov r9, r8
    mov r10, rsi
    sub r10, r8
    lea r14, [rsi + 1]
    lea r15, [r8 + rbx]
    jmp validate_setup
empty_input:
    mov r9, r8
    xor r10, r10
    mov r14, r8
    mov r15, r8
validate_setup:
    mov r11, rsp
    xor r12, r12
validate_loop:
    cmp r12, r10
    jae validate_end
    movzx eax, byte ptr [r9 + r12]
    cmp al, '['
    je v_open
    cmp al, ']'
    je v_close
    inc r12
    jmp validate_loop
v_open:
    mov rax, r11
    sub rax, rsp
    cmp rax, 8000
    jae unbalanced
    push r12
    inc r12
    jmp validate_loop
v_close:
    cmp rsp, r11
    je unbalanced
    pop rax
    lea rdi, [rip + jumps]
    mov qword ptr [rdi + rax*8], r12
    mov qword ptr [rdi + r12*8], rax
    inc r12
    jmp validate_loop
validate_end:
    cmp rsp, r11
    jne unbalanced
    xor r12, r12
    xor r13, r13
    lea rbx, [rip + tape]
interp_loop:
    cmp r12, r10
    jae exit_ok
    movzx eax, byte ptr [r9 + r12]
    cmp al, '>'
    je do_right
    cmp al, '<'
    je do_left
    cmp al, '+'
    je do_plus
    cmp al, '-'
    je do_minus
    cmp al, '.'
    je do_dot
    cmp al, ','
    je do_comma
    cmp al, '['
    je do_lbrack
    cmp al, ']'
    je do_rbrack
    inc r12
    jmp interp_loop
do_right:
    inc r13
    inc r12
    jmp interp_loop
do_left:
    dec r13
    inc r12
    jmp interp_loop
do_plus:
    inc byte ptr [rbx + r13]
    inc r12
    jmp interp_loop
do_minus:
    dec byte ptr [rbx + r13]
    inc r12
    jmp interp_loop
do_dot:
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r13]
    mov edx, 1
    syscall
    inc r12
    jmp interp_loop
do_comma:
    cmp r14, r15
    jae comma_eof
    mov al, byte ptr [r14]
    mov byte ptr [rbx + r13], al
    inc r14
    inc r12
    jmp interp_loop
comma_eof:
    mov byte ptr [rbx + r13], 0
    inc r12
    jmp interp_loop
do_lbrack:
    cmp byte ptr [rbx + r13], 0
    jne lbrack_next
    lea rdi, [rip + jumps]
    mov r12, qword ptr [rdi + r12*8]
lbrack_next:
    inc r12
    jmp interp_loop
do_rbrack:
    cmp byte ptr [rbx + r13], 0
    je rbrack_next
    lea rdi, [rip + jumps]
    mov r12, qword ptr [rdi + r12*8]
rbrack_next:
    inc r12
    jmp interp_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov rsp, r11
    mov eax, 60
    mov edi, 1
    syscall
.bss
.align 8
stdin_buf:
    .zero 65536
tape:
    .zero 30000
jumps:
    .zero 524288
