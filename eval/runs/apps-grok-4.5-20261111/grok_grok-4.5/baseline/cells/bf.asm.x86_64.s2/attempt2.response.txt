.globl _start

.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NEST, 1000

.text
_start:
    lea rsi, [rip + input_buf]
    mov edx, BUF_SIZE
read_loop:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    lea r8, [rip + input_buf]
    mov rcx, rsi
    sub rcx, r8

    mov rdi, r8
    mov rdx, rcx
find_bang:
    test rdx, rdx
    jz no_bang
    cmp byte ptr [rdi], 0x21
    je found_bang
    inc rdi
    dec rdx
    jmp find_bang

found_bang:
    mov r12, r8
    mov r13, rdi
    sub r13, r8
    lea r14, [rdi + 1]
    mov r15, rdx
    dec r15
    jmp parse_done

no_bang:
    mov r12, r8
    mov r13, rcx
    xor r15, r15

parse_done:
    lea rbp, [rip + bracket_stack]
    xor ebx, ebx
    xor ecx, ecx
validate_loop:
    cmp rcx, r13
    jge validate_end
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '['
    je do_open
    cmp al, ']'
    je do_close
    inc rcx
    jmp validate_loop

do_open:
    cmp ebx, MAX_NEST
    jge unbalanced
    mov dword ptr [rbp + rbx*4], ecx
    inc ebx
    inc rcx
    jmp validate_loop

do_close:
    test ebx, ebx
    jz unbalanced
    dec ebx
    mov edx, dword ptr [rbp + rbx*4]
    lea rsi, [rip + jump_table]
    mov dword ptr [rsi + rdx*4], ecx
    mov dword ptr [rsi + rcx*4], edx
    inc rcx
    jmp validate_loop

validate_end:
    test ebx, ebx
    jnz unbalanced

    xor ebx, ebx
    xor r9, r9
    lea r10, [rip + tape]
    lea r8, [rip + jump_table]

run_loop:
    cmp r9, r13
    jge exit_ok
    movzx eax, byte ptr [r12 + r9]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_lbrack
    cmp al, ']'
    je cmd_rbrack
    inc r9
    jmp run_loop

cmd_gt:
    inc ebx
    inc r9
    jmp run_loop

cmd_lt:
    dec ebx
    inc r9
    jmp run_loop

cmd_plus:
    inc byte ptr [r10 + rbx]
    inc r9
    jmp run_loop

cmd_minus:
    dec byte ptr [r10 + rbx]
    inc r9
    jmp run_loop

cmd_dot:
    lea rsi, [r10 + rbx]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r9
    jmp run_loop

cmd_comma:
    test r15, r15
    jz comma_eof
    mov al, byte ptr [r14]
    mov byte ptr [r10 + rbx], al
    inc r14
    dec r15
    jmp comma_done
comma_eof:
    mov byte ptr [r10 + rbx], 0
comma_done:
    inc r9
    jmp run_loop

cmd_lbrack:
    cmp byte ptr [r10 + rbx], 0
    jne lbrack_next
    mov r9d, dword ptr [r8 + r9*4]
lbrack_next:
    inc r9
    jmp run_loop

cmd_rbrack:
    cmp byte ptr [r10 + rbx], 0
    je rbrack_next
    mov r9d, dword ptr [r8 + r9*4]
rbrack_next:
    inc r9
    jmp run_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
input_buf:
    .zero BUF_SIZE
tape:
    .zero TAPE_SIZE
jump_table:
    .zero BUF_SIZE * 4
bracket_stack:
    .zero MAX_NEST * 4
