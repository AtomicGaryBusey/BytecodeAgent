```
.globl _start

.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NEST, 1000

.text
_start:
    # Read all of stdin into input_buf
    lea rsi, [rip + input_buf]
    mov edx, BUF_SIZE
read_loop:
    xor eax, eax          # sys_read
    xor edi, edi          # fd 0
    syscall
    test rax, rax
    jle read_done         # 0 = EOF, negative = error treat as EOF
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    lea r8, [rip + input_buf]
    mov rcx, rsi
    sub rcx, r8           # rcx = total_len

    # Find first '!'
    mov rdi, r8           # scan ptr
    mov rdx, rcx          # remaining
find_bang:
    test rdx, rdx
    jz no_bang
    cmp byte ptr [rdi], 0x21
    je found_bang
    inc rdi
    dec rdx
    jmp find_bang

found_bang:
    mov r12, r8           # prog_start
    mov r13, rdi
    sub r13, r8           # prog_len
    lea r14, [rdi + 1]    # input_start
    mov r15, rdx
    dec r15               # input_len = remaining after '!'
    jmp parse_done

no_bang:
    mov r12, r8
    mov r13, rcx          # prog_len = total
    xor r15, r15          # input_len = 0
    # r14 unused

parse_done:
    # Validate brackets and build jump_table
    # r12=prog, r13=len
    lea rbp, [rip + bracket_stack]
    xor ebx, ebx          # depth
    xor ecx, ecx          # i
validate_loop:
    cmp rcx, r13
    jge validate_end
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '['
    je do_open
    cmp al, ']'
    je do_close
    inc rcx
    jmp validate_loop

do_open:
    cmp ebx, MAX_NEST
    jge unbalanced
    mov dword ptr [rbp + rbx*4], ecx
    inc ebx
    inc rcx
    jmp validate_loop

do_close:
    test ebx, ebx
    jz unbalanced
    dec ebx
    mov edx, dword ptr [rbp + rbx*4]  # open pos
    lea rsi, [rip + jump_table]
    mov dword ptr [rsi + rdx*4], ecx  # [ -> ]
    mov dword ptr [rsi + rcx*4], edx  # ] -> [
    inc rcx
    jmp validate_loop

validate_end:
    test ebx, ebx
    jnz unbalanced

    # Run the program
    # r12=prog, r13=prog_len, r14=input_ptr, r15=input_len
    xor ebx, ebx          # tape offset
    xor r9, r9            # pc
    lea r10, [rip + tape]
    lea r11, [rip + jump_table]

run_loop:
    cmp r9, r13
    jge exit_ok
    movzx eax, byte ptr [r12 + r9]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_lbrack
    cmp al, ']'
    je cmd_rbrack
    inc r9
    jmp run_loop

cmd_gt:
    inc ebx
    inc r9
    jmp run_loop

cmd_lt:
    dec ebx
    inc r9
    jmp run_loop

cmd_plus:
    inc byte ptr [r10 + rbx]
    inc r9
    jmp run_loop

cmd_minus:
    dec byte ptr [r10 + rbx]
    inc r9
    jmp run_loop

cmd_dot:
    lea rsi, [r10 + rbx]
    mov edx, 1
    mov edi, 1
    mov eax, 1            # sys_write
    syscall
    inc r9
    jmp run_loop

cmd_comma:
    test r15, r15
    jz comma_eof
    mov al, byte ptr [r14]
    mov byte ptr [r10 + rbx], al
    inc r14
    dec r15
    jmp comma_done
comma_eof:
    mov byte ptr [r10 + rbx], 0
comma_done:
    inc r9
    jmp run_loop

cmd_lbrack:
    cmp byte ptr [r10 + rbx], 0
    jne lbrack_next
    mov r9d, dword ptr [r11 + r9*4]
lbrack_next:
    inc r9
    jmp run_loop

cmd_rbrack:
    cmp byte ptr [r10 + rbx], 0
    je rbrack_next
    mov r9d, dword ptr [r11 + r9*4]
rbrack_next:
    inc r9
    jmp run_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
input_buf:
    .zero BUF_SIZE
tape:
    .zero TAPE_SIZE
jump_table:
    .zero BUF_SIZE * 4
bracket_stack:
    .zero MAX_NEST * 4
```
