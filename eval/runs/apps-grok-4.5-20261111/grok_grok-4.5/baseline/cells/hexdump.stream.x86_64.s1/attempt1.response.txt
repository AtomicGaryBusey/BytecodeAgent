.globl _start
.text
_start:
	xor	ebx, ebx
	lea	r12, [rip + inbuf]
.read_loop:
	mov	eax, 0
	mov	edi, 0
	lea	rsi, [r12 + rbx]
	mov	rdx, 65536
	sub	rdx, rbx
	jz	.read_done
	syscall
	test	rax, rax
	jle	.read_done
	add	rbx, rax
	jmp	.read_loop
.read_done:
	xor	r14, r14
.row_loop:
	cmp	r14, rbx
	jae	.print_final
	mov	r9, rbx
	sub	r9, r14
	cmp	r9, 16
	jbe	1f
	mov	r9, 16
1:
	lea	rdi, [rip + outbuf]
	mov	eax, r14d
	mov	ecx, 8
2:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	lea	r8, [rip + hexdigits]
	mov	dl, [r8 + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ecx
	jnz	2b
	mov	ax, 0x2020
	stosw
	xor	esi, esi
.bytes_loop:
	cmp	esi, r9d
	jae	.pad
	lea	r8, [rip + inbuf]
	movzx	eax, byte ptr [r8 + r14 + rsi]
	mov	edx, eax
	shr	edx, 4
	lea	r8, [rip + hexdigits]
	mov	dl, [r8 + rdx]
	mov	[rdi], dl
	inc	rdi
	and	eax, 15
	mov	dl, [r8 + rax]
	mov	[rdi], dl
	inc	rdi
	mov	al, 0x20
	stosb
	jmp	.after_byte
.pad:
	mov	ax, 0x2020
	stosw
	mov	al, 0x20
	stosb
.after_byte:
	cmp	esi, 7
	jne	.noextra
	mov	al, 0x20
	stosb
.noextra:
	inc	esi
	cmp	esi, 16
	jb	.bytes_loop
	mov	al, 0x20
	stosb
	mov	al, '|'
	stosb
	xor	esi, esi
.ascii_loop:
	cmp	esi, r9d
	jae	.end_ascii
	lea	r8, [rip + inbuf]
	movzx	eax, byte ptr [r8 + r14 + rsi]
	cmp	al, 0x20
	jb	.dot
	cmp	al, 0x7e
	ja	.dot
	jmp	.putchar
.dot:
	mov	al, '.'
.putchar:
	stosb
	inc	esi
	jmp	.ascii_loop
.end_ascii:
	mov	al, '|'
	stosb
	mov	al, 10
	stosb
	lea	rax, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rax
	mov	eax, 1
	mov	edi, 1
	lea	rsi, [rip + outbuf]
	syscall
	add	r14, r9
	jmp	.row_loop
.print_final:
	lea	rdi, [rip + outbuf]
	mov	eax, ebx
	mov	ecx, 8
3:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	lea	r8, [rip + hexdigits]
	mov	dl, [r8 + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ecx
	jnz	3b
	mov	al, 10
	stosb
	mov	eax, 1
	mov	edi, 1
	lea	rsi, [rip + outbuf]
	mov	edx, 9
	syscall
	mov	eax, 60
	xor	edi, edi
	syscall
.section .rodata
hexdigits:
	.ascii "0123456789abcdef"
.section .bss
.align 16
inbuf:
	.zero 65536
outbuf:
	.zero 80
