.globl _start
.text
_start:
	xor	ebx, ebx
	lea	r12, [rip + inbuf]
.read_loop:
	mov	eax, 0
	mov	edi, 0
	lea	rsi, [r12 + rbx]
	mov	rdx, 65536
	sub	rdx, rbx
	jz	.read_done
	syscall
	test	rax, rax
	jle	.read_done
	add	rbx, rax
	jmp	.read_loop
.read_done:
	xor	r14, r14
.row_loop:
	cmp	r14, rbx
	jae	.print_final
	mov	r9, rbx
	sub	r9, r14
	cmp	r9, 16
	jbe	1f
	mov	r9, 16
1:
	lea	rdi, [rip + outbuf]
	mov	eax, r14d
	mov	ecx, 8
2:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	lea	r8, [rip + hexdigits]
	mov	dl, [r8 + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ecx
	jnz	2b
	mov	ax, 0x2020
	stosw
	xor	esi, esi
.bytes_loop:
	cmp	esi, 16
	jae	.do_ascii
	cmp	esi, r9d
	jae	.do_pad
	mov	rax, r14
	add	rax, rsi
	movzx	eax, byte ptr [r12 + rax]
	mov	ecx, eax
	shr	ecx, 4
	lea	r8, [rip + hexdigits]
	mov	cl, [r8 + rcx]
	mov	[rdi], cl
	inc	rdi
	and	eax, 15
	mov	al, [r8 + rax]
	stosb
	mov	al, 0x20
	stosb
	jmp	.after_byte
.do_pad:
	mov	al, 0x20
	stosb
	stosb
	stosb
.after_byte:
	inc	esi
	cmp	esi, 8
	jne	.bytes_loop
	mov	al, 0x20
	stosb
	jmp	.bytes_loop
.do_ascii:
	mov	al, 0x20
	stosb
	mov	al, '|'
	stosb
	xor	esi, esi
.ascii_loop:
	cmp	esi, r9d
	jae	.ascii_end
	mov	rax, r14
	add	rax, rsi
	movzx	eax, byte ptr [r12 + rax]
	cmp	al, 0x20
	jb	.is_dot
	cmp	al, 0x7e
	ja	.is_dot
	jmp	.put_char
.is_dot:
	mov	al, '.'
.put_char:
	stosb
	inc	esi
	jmp	.ascii_loop
.ascii_end:
	mov	al, '|'
	stosb
	mov	al, 0x0a
	stosb
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1
	mov	edi, 1
	syscall
	add	r14, r9
	jmp	.row_loop
.print_final:
	lea	rdi, [rip + outbuf]
	mov	eax, ebx
	mov	ecx, 8
3:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	lea	r8, [rip + hexdigits]
	mov	dl, [r8 + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ecx
	jnz	3b
	mov	al, 0x0a
	stosb
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1
	mov	edi, 1
	syscall
	mov	eax, 60
	xor	edi, edi
	syscall
.section .rodata
hexdigits:
	.ascii "0123456789abcdef"
.bss
.align 16
inbuf:
	.zero 65536
outbuf:
	.zero 128
