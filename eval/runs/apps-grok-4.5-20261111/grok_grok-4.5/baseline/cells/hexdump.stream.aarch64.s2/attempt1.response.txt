.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x27, hextab
    add x27, x27, :lo12:hextab
    mov x21, #0

print_loop:
    cmp x21, x20
    b.hs print_final
    sub x22, x20, x21
    cmp x22, #16
    b.ls 1f
    mov x22, #16
1:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23

    mov x0, x21
    mov x1, x24
    bl write_hex8
    mov x24, x1

    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1

    mov x25, #0
hex_loop:
    cmp x25, #16
    b.ge hex_done
    cmp x25, #8
    b.ne 2f
    mov w0, #' '
    strb w0, [x24], #1
2:
    cmp x25, x22
    b.ge pad_slot
    add x26, x19, x21
    ldrb w0, [x26, x25]
    lsr w1, w0, #4
    ldrb w1, [x27, x1]
    strb w1, [x24], #1
    and w1, w0, #0xf
    ldrb w1, [x27, x1]
    strb w1, [x24], #1
    mov w0, #' '
    strb w0, [x24], #1
    b next_hex
pad_slot:
    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1
next_hex:
    add x25, x25, #1
    b hex_loop

hex_done:
    mov w0, #' '
    strb w0, [x24], #1
    mov w0, #'|'
    strb w0, [x24], #1

    mov x25, #0
ascii_loop:
    cmp x25, x22
    b.ge ascii_done
    add x26, x19, x21
    ldrb w0, [x26, x25]
    cmp w0, #0x20
    b.lt do_dot
    cmp w0, #0x7e
    b.gt do_dot
    b store_asc
do_dot:
    mov w0, #'.'
store_asc:
    strb w0, [x24], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w0, #'|'
    strb w0, [x24], #1
    mov w0, #'\n'
    strb w0, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #SYS_write
    svc #0

    add x21, x21, #16
    b print_loop

print_final:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x0, x20
    mov x1, x23
    bl write_hex8
    mov w0, #'\n'
    strb w0, [x23, #8]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

write_hex8:
    mov w3, #28
1:
    lsr w4, w0, w3
    and w4, w4, #0xf
    ldrb w5, [x27, x4]
    strb w5, [x1], #1
    subs w3, w3, #4
    b.ge 1b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .space 65536
outbuf:
    .space 128
