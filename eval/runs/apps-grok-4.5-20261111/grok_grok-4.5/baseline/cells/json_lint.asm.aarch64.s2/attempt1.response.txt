.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ OK_LEN, 3
.equ ERR_PREFIX_LEN, 9

.text
.globl _start
_start:
    mov x29, sp

    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x22, x19
    movz x23, #1, lsl #16
    add x23, x19, x23

.Lread_loop:
    sub x2, x23, x22
    cbz x2, .Lread_done
    mov x0, #0
    mov x1, x22
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x22, x22, x0
    b .Lread_loop

.Lread_done:
    mov x21, x22
    mov x20, x19

    bl skip_ws
    cmp x20, x21
    b.hs fail
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne fail

success:
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    movz x2, #OK_LEN
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

fail:
    sub x22, x20, x19
    mov x0, #1
    adrp x1, err_prefix
    add x1, x1, :lo12:err_prefix
    movz x2, #ERR_PREFIX_LEN
    mov x8, #SYS_WRITE
    svc #0
    sub sp, sp, #32
    add x1, sp, #32
    mov x2, x22
    cbnz x2, .Litoa_loop
    mov w3, #'0'
    strb w3, [x1, #-1]!
    b .Litoa_done
.Litoa_loop:
    movz x4, #10
    udiv x5, x2, x4
    msub x6, x5, x4, x2
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    mov x2, x5
    cbnz x2, .Litoa_loop
.Litoa_done:
    mov x0, #1
    add x2, sp, #32
    sub x2, x2, x1
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    movz x2, #1
    mov x8, #SYS_WRITE
    svc #0
    movz x0, #1
    movz x8, #SYS_EXIT
    svc #0

skip_ws:
    cmp x20, x21
    b.hs .Lsw_done
    ldrb w0, [x20]
    cmp w0, #0x20
    b.eq .Lsw_adv
    cmp w0, #0x09
    b.eq .Lsw_adv
    cmp w0, #0x0A
    b.eq .Lsw_adv
    cmp w0, #0x0D
    b.eq .Lsw_adv
.Lsw_done:
    ret
.Lsw_adv:
    add x20, x20, #1
    b skip_ws

parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'{'
    b.eq .Pv_obj
    cmp w0, #'['
    b.eq .Pv_arr
    cmp w0, #'"'
    b.eq .Pv_str
    cmp w0, #'t'
    b.eq .Pv_true
    cmp w0, #'f'
    b.eq .Pv_false
    cmp w0, #'n'
    b.eq .Pv_null
    cmp w0, #'-'
    b.eq .Pv_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls .Pv_num
    b fail
.Pv_obj:
    bl parse_object
    b .Pv_done
.Pv_arr:
    bl parse_array
    b .Pv_done
.Pv_str:
    bl parse_string
    b .Pv_done
.Pv_true:
    bl parse_true
    b .Pv_done
.Pv_false:
    bl parse_false
    b .Pv_done
.Pv_null:
    bl parse_null
    b .Pv_done
.Pv_num:
    bl parse_number
.Pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_true:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'r'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'e'
    b.ne fail
    add x20, x20, #1
    ret

parse_false:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'a'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'s'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'e'
    b.ne fail
    add x20, x20, #1
    ret

parse_null:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    ret

parse_string:
    add x20, x20, #1
.Lstr_loop:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq .Lstr_end
    cmp w0, #'\\'
    b.eq .Lstr_esc
    cmp w0, #0x20
    b.lo fail
    add x20, x20, #1
    b .Lstr_loop
.Lstr_esc:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq .Lstr_esc_ok
    cmp w0, #'\\'
    b.eq .Lstr_esc_ok
    cmp w0, #'/'
    b.eq .Lstr_esc_ok
    cmp w0, #'b'
    b.eq .Lstr_esc_ok
    cmp w0, #'f'
    b.eq .Lstr_esc_ok
    cmp w0, #'n'
    b.eq .Lstr_esc_ok
    cmp w0, #'r'
    b.eq .Lstr_esc_ok
    cmp w0, #'t'
    b.eq .Lstr_esc_ok
    cmp w0, #'u'
    b.eq .Lstr_uni
    b fail
.Lstr_esc_ok:
    add x20, x20, #1
    b .Lstr_loop
.Lstr_uni:
    add x20, x20, #1
    movz x1, #4
.Lstr_hex:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    sub w2, w0, #'0'
    cmp w2, #10
    b.lo .Lstr_hex_ok
    orr w3, w0, #0x20
    sub w2, w3, #'a'
    cmp w2, #6
    b.hs fail
.Lstr_hex_ok:
    add x20, x20, #1
    subs x1, x1, #1
    b.ne .Lstr_hex
    b .Lstr_loop
.Lstr_end:
    add x20, x20, #1
    ret

parse_number:
    ldrb w0, [x20]
    cmp w0, #'-'
    b.ne .Lnum_int
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
.Lnum_int:
    ldrb w0, [x20]
    cmp w0, #'0'
    b.eq .Lnum_zero
    sub w1, w0, #'1'
    cmp w1, #9
    b.hs fail
    add x20, x20, #1
.Lnum_digits:
    cmp x20, x21
    b.hs .Lnum_after_int
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs .Lnum_after_int
    add x20, x20, #1
    b .Lnum_digits
.Lnum_zero:
    add x20, x20, #1
.Lnum_after_int:
    cmp x20, x21
    b.hs .Lnum_done
    ldrb w0, [x20]
    cmp w0, #'.'
    b.ne .Lnum_exp
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs fail
    add x20, x20, #1
.Lnum_frac_digits:
    cmp x20, x21
    b.hs .Lnum_done
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs .Lnum_exp
    add x20, x20, #1
    b .Lnum_frac_digits
.Lnum_exp:
    cmp x20, x21
    b.hs .Lnum_done
    ldrb w0, [x20]
    cmp w0, #'e'
    b.eq .Lnum_exp_ok
    cmp w0, #'E'
    b.ne .Lnum_done
.Lnum_exp_ok:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'+'
    b.eq .Lnum_exp_sign
    cmp w0, #'-'
    b.ne .Lnum_exp_need
.Lnum_exp_sign:
    add x20, x20, #1
.Lnum_exp_need:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs fail
    add x20, x20, #1
.Lnum_exp_digits:
    cmp x20, x21
    b.hs .Lnum_done
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs .Lnum_done
    add x20, x20, #1
    b .Lnum_exp_digits
.Lnum_done:
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'}'
    b.ne .Lobj_member
    add x20, x20, #1
    b .Lobj_done
.Lobj_member:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'"'
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #':'
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #','
    b.eq .Lobj_comma
    cmp w0, #'}'
    b.ne fail
    add x20, x20, #1
    b .Lobj_done
.Lobj_comma:
    add x20, x20, #1
    bl skip_ws
    b .Lobj_member
.Lobj_done:
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #']'
    b.ne .Larr_elem
    add x20, x20, #1
    b .Larr_done
.Larr_elem:
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #','
    b.eq .Larr_comma
    cmp w0, #']'
    b.ne fail
    add x20, x20, #1
    b .Larr_done
.Larr_comma:
    add x20, x20, #1
    bl skip_ws
    b .Larr_elem
.Larr_done:
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
nl_msg:
    .ascii "\n"

.bss
.balign 16
buffer:
    .space 65536
