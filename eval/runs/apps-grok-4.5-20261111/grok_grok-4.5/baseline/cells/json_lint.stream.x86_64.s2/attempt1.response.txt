```
.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_msg:
    .ascii "error at "
err_len = . - err_msg
nl_msg:
    .ascii "\n"

.bss
.align 16
buffer:
    .space 65536
numbuf:
    .space 32

.text
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + buffer]
    mov r13, rbx
    xor r14, r14

    call skip_ws
    cmp r14, r13
    je do_error
    call parse_value
    test eax, eax
    jnz do_error
    call skip_ws
    cmp r14, r13
    jne do_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, err_len
    syscall
    mov rax, r14
    call print_uint
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_msg]
    mov edx, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

print_uint:
    lea rdi, [rip + numbuf + 31]
    mov ecx, 10
pu_loop:
    dec rdi
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pu_loop
    lea rdx, [rip + numbuf + 31]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    ret

skip_ws:
sw_loop:
    cmp r14, r13
    jae sw_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    je sw_adv
    jmp sw_done
sw_adv:
    inc r14
    jmp sw_loop
sw_done:
    ret

parse_value:
    cmp r14, r13
    jae pv_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '{'
    je pv_obj
    cmp al, '['
    je pv_arr
    cmp al, '"'
    je pv_str
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je pv_num
    cmp al, '0'
    jb pv_fail
    cmp al, '9'
    jbe pv_num
pv_fail:
    mov eax, 1
    ret
pv_obj:
    call parse_object
    ret
pv_arr:
    call parse_array
    ret
pv_str:
    call parse_string
    ret
pv_num:
    call parse_number
    ret
pv_true:
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'r'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'u'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'e'
    jne pv_fail
    inc r14
    xor eax, eax
    ret
pv_false:
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'a'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'l'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 's'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'e'
    jne pv_fail
    inc r14
    xor eax, eax
    ret
pv_null:
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'u'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'l'
    jne pv_fail
    inc r14
    cmp r14, r13
    jae pv_fail
    cmp byte ptr [r12 + r14], 'l'
    jne pv_fail
    inc r14
    xor eax, eax
    ret

parse_string:
    inc r14
ps_loop:
    cmp r14, r13
    jae ps_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je ps_done
    cmp al, '\\'
    je ps_esc
    cmp al, 0x20
    jb ps_fail
    inc r14
    jmp ps_loop
ps_done:
    inc r14
    xor eax, eax
    ret
ps_fail:
    mov eax, 1
    ret
ps_esc:
    inc r14
    cmp r14, r13
    jae ps_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je ps_esc_ok
    cmp al, '\\'
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    je ps_uni
    jmp ps_fail
ps_esc_ok:
    inc r14
    jmp ps_loop
ps_uni:
    inc r14
    mov ecx, 4
ps_hex:
    cmp r14, r13
    jae ps_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb ps_fail
    cmp al, '9'
    jbe ps_hex_ok
    cmp al, 'A'
    jb ps_fail
    cmp al, 'F'
    jbe ps_hex_ok
    cmp al, 'a'
    jb ps_fail
    cmp al, 'f'
    jbe ps_hex_ok
    jmp ps_fail
ps_hex_ok:
    inc r14
    dec ecx
    jnz ps_hex
    jmp ps_loop

parse_number:
    cmp r14, r13
    jae pn_fail
    cmp byte ptr [r12 + r14], '-'
    jne pn_int
    inc r14
    cmp r14, r13
    jae pn_fail
pn_int:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    je pn_zero
    cmp al, '1'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
pn_digits:
    inc r14
    cmp r14, r13
    jae pn_after
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb pn_after
    cmp al, '9'
    ja pn_after
    jmp pn_digits
pn_zero:
    inc r14
pn_after:
    cmp r14, r13
    jae pn_ok
    cmp byte ptr [r12 + r14], '.'
    jne pn_exp
    inc r14
    cmp r14, r13
    jae pn_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
pn_frac:
    inc r14
    cmp r14, r13
    jae pn_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb pn_exp
    cmp al, '9'
    ja pn_exp
    jmp pn_frac
pn_exp:
    cmp r14, r13
    jae pn_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, 'e'
    je pn_do_exp
    cmp al, 'E'
    je pn_do_exp
    jmp pn_ok
pn_do_exp:
    inc r14
    cmp r14, r13
    jae pn_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '+'
    je pn_sign
    cmp al, '-'
    je pn_sign
    jmp pn_edig
pn_sign:
    inc r14
    cmp r14, r13
    jae pn_fail
pn_edig:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
pn_edigs:
    inc r14
    cmp r14, r13
    jae pn_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb pn_ok
    cmp al, '9'
    ja pn_ok
    jmp pn_edigs
pn_ok:
    xor eax, eax
    ret
pn_fail:
    mov eax, 1
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae po_fail
    cmp byte ptr [r12 + r14], '}'
    je po_end
po_member:
    cmp byte ptr [r12 + r14], '"'
    jne po_fail
    call parse_string
    test eax, eax
    jnz po_ret
    call skip_ws
    cmp r14, r13
    jae po_fail
    cmp byte ptr [r12 + r14], ':'
    jne po_fail
    inc r14
    call skip_ws
    call parse_value
    test eax, eax
    jnz po_ret
    call skip_ws
    cmp r14, r13
    jae po_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, '}'
    je po_end
    cmp al, ','
    jne po_fail
    inc r14
    call skip_ws
    cmp r14, r13
    jae po_fail
    jmp po_member
po_end:
    inc r14
    xor eax, eax
    ret
po_fail:
    mov eax, 1
po_ret:
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae pa_fail
    cmp byte ptr [r12 + r14], ']'
    je pa_end
pa_val:
    call parse_value
    test eax, eax
    jnz pa_ret
    call skip_ws
    cmp r14, r13
    jae pa_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, ']'
    je pa_end
    cmp al, ','
    jne pa_fail
    inc r14
    call skip_ws
    cmp r14, r13
    jae pa_fail
    jmp pa_val
pa_end:
    inc r14
    xor eax, eax
    ret
pa_fail:
    mov eax, 1
pa_ret:
    ret
```
