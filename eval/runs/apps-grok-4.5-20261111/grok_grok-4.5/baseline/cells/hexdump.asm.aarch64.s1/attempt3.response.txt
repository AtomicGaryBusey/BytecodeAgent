.globl _start

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    cmp x20, x2
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

dump_loop:
    cmp x21, x20
    b.hs dump_done
    sub x22, x20, x21
    cmp x22, #16
    mov x23, #16
    csel x22, x22, x23, lo
    add x0, x19, x21
    mov x1, x21
    mov x2, x22
    bl format_line
    mov x2, x0
    mov x0, #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    mov x8, #64
    svc #0
    add x21, x21, x22
    b dump_loop

dump_done:
    adrp x0, line_buf
    add x0, x0, :lo12:line_buf
    mov w1, w20
    bl append_hex32
    mov w1, #10
    strb w1, [x0]
    mov x0, #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// format_line(x0=data, x1=offset, x2=n) -> x0=length
format_line:
    stp x29, x30, [sp, #-48]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    mov x19, x0
    mov x20, x1
    mov x21, x2
    adrp x22, line_buf
    add x22, x22, :lo12:line_buf
    mov x0, x22
    mov w1, w20
    bl append_hex32
    mov w1, #32
    strb w1, [x0], #1
    strb w1, [x0], #1
    mov x3, #0
hex_loop:
    cmp x3, #16
    b.ge hex_done
    cmp x3, #8
    b.ne 1f
    mov w1, #32
    strb w1, [x0], #1
1:
    cmp x3, x21
    b.ge pad_slot
    ldrb w1, [x19, x3]
    bl append_hex_byte
    mov w1, #32
    strb w1, [x0], #1
    b next_hex
pad_slot:
    mov w1, #32
    strb w1, [x0], #1
    strb w1, [x0], #1
    strb w1, [x0], #1
next_hex:
    add x3, x3, #1
    b hex_loop
hex_done:
    mov w1, #32
    strb w1, [x0], #1
    mov w1, #124
    strb w1, [x0], #1
    mov x3, #0
ascii_loop:
    cmp x3, x21
    b.ge ascii_done
    ldrb w1, [x19, x3]
    cmp w1, #32
    b.lt make_dot
    cmp w1, #127
    b.ge make_dot
    b store_char
make_dot:
    mov w1, #46
store_char:
    strb w1, [x0], #1
    add x3, x3, #1
    b ascii_loop
ascii_done:
    mov w1, #124
    strb w1, [x0], #1
    mov w1, #10
    strb w1, [x0], #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    sub x0, x0, x1
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #48
    ret

// append_hex32(x0=dest, w1=value) -> x0=dest+8
append_hex32:
    adrp x9, hexdigits
    add x9, x9, :lo12:hexdigits
    mov w10, #28
1:
    lsr w11, w1, w10
    and w11, w11, #15
    ldrb w11, [x9, x11]
    strb w11, [x0], #1
    subs w10, w10, #4
    b.pl 1b
    ret

// append_hex_byte(x0=dest, w1=byte) -> x0=dest+2
append_hex_byte:
    adrp x9, hexdigits
    add x9, x9, :lo12:hexdigits
    lsr w10, w1, #4
    and w10, w10, #15
    ldrb w10, [x9, x10]
    strb w10, [x0], #1
    and w1, w1, #15
    ldrb w1, [x9, x1]
    strb w1, [x0], #1
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .space 65536
line_buf:
    .space 80
