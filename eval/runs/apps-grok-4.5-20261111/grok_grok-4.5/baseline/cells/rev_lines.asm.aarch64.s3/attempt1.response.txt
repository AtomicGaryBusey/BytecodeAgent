.globl _start

.equ BUF_SIZE, 65536
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 16
input:
	.space 65536
.balign 16
output:
	.space 65537

.text
.balign 4
_start:
	adrp	x20, input
	add	x20, x20, :lo12:input
	mov	x19, #0

read_loop:
	mov	x2, #BUF_SIZE
	cmp	x19, x2
	b.hs	read_done
	mov	x0, #0
	add	x1, x20, x19
	sub	x2, x2, x19
	mov	x8, #SYS_read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x19, x19, x0
	b	read_loop

read_done:
	adrp	x21, output
	add	x21, x21, :lo12:output
	mov	x22, #0
	mov	x23, #0

process_loop:
	cmp	x23, x19
	b.hs	process_done
	mov	x24, x23

find_nl:
	cmp	x23, x19
	b.hs	got_line
	ldrb	w0, [x20, x23]
	cmp	w0, #10
	b.eq	got_line
	add	x23, x23, #1
	b	find_nl

got_line:
	sub	x25, x23, x24
	cbz	x25, rev_done
	add	x0, x24, x25
	sub	x0, x0, #1
	add	x0, x20, x0
	add	x1, x21, x22
	mov	x2, x25
rev_loop:
	ldrb	w3, [x0], #-1
	strb	w3, [x1], #1
	subs	x2, x2, #1
	b.ne	rev_loop
rev_done:
	add	x22, x22, x25
	mov	w0, #10
	strb	w0, [x21, x22]
	add	x22, x22, #1
	cmp	x23, x19
	b.hs	process_loop
	add	x23, x23, #1
	b	process_loop

process_done:
	mov	x0, #1
	mov	x1, x21
	mov	x2, x22
	mov	x8, #SYS_write
	svc	#0
	mov	x0, #0
	mov	x8, #SYS_exit
	svc	#0
