.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x21, x19
    mov x20, x19

read_loop:
    mov x3, #65536
    sub x2, x20, x21
    sub x2, x3, x2
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x19, x21
    bl skip_ws
    cmp x19, x20
    b.eq err_eof
    bl parse_value
    cbnz x0, err_pos
    bl skip_ws
    cmp x19, x20
    b.ne err_pos

    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #ok_msg_len
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

err_eof:
    mov x19, x20
err_pos:
    sub x22, x19, x21
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #error_prefix_len
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, x22
    bl print_uint
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

skip_ws:
    cmp x19, x20
    b.eq sw_done
    ldrb w0, [x19]
    cmp w0, #0x20
    b.eq sw_adv
    cmp w0, #0x09
    b.eq sw_adv
    cmp w0, #0x0A
    b.eq sw_adv
    cmp w0, #0x0D
    b.eq sw_adv
    b sw_done
sw_adv:
    add x19, x19, #1
    b skip_ws
sw_done:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x19, x20
    b.eq pv_fail
    ldrb w0, [x19]
    cmp w0, #'{'
    b.eq pv_object
    cmp w0, #'['
    b.eq pv_array
    cmp w0, #'"'
    b.eq pv_string
    cmp w0, #'-'
    b.eq pv_number
    cmp w0, #'0'
    b.lt pv_check_lit
    cmp w0, #'9'
    b.le pv_number
pv_check_lit:
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    b pv_fail
pv_object:
    bl parse_object
    b pv_done
pv_array:
    bl parse_array
    b pv_done
pv_string:
    bl parse_string
    b pv_done
pv_number:
    bl parse_number
    b pv_done
pv_true:
    bl parse_true
    b pv_done
pv_false:
    bl parse_false
    b pv_done
pv_null:
    bl parse_null
    b pv_done
pv_fail:
    mov x0, #1
pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.eq po_fail
    ldrb w0, [x19]
    cmp w0, #'}'
    b.eq po_end
po_member:
    cmp x19, x20
    b.eq po_fail
    ldrb w0, [x19]
    cmp w0, #'"'
    b.ne po_fail
    bl parse_string
    cbnz x0, po_ret
    bl skip_ws
    cmp x19, x20
    b.eq po_fail
    ldrb w0, [x19]
    cmp w0, #':'
    b.ne po_fail
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    cbnz x0, po_ret
    bl skip_ws
    cmp x19, x20
    b.eq po_fail
    ldrb w0, [x19]
    cmp w0, #'}'
    b.eq po_end
    cmp w0, #','
    b.ne po_fail
    add x19, x19, #1
    bl skip_ws
    b po_member
po_end:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
po_fail:
    mov x0, #1
po_ret:
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.eq pa_fail
    ldrb w0, [x19]
    cmp w0, #']'
    b.eq pa_end
pa_elem:
    bl parse_value
    cbnz x0, pa_ret
    bl skip_ws
    cmp x19, x20
    b.eq pa_fail
    ldrb w0, [x19]
    cmp w0, #']'
    b.eq pa_end
    cmp w0, #','
    b.ne pa_fail
    add x19, x19, #1
    bl skip_ws
    b pa_elem
pa_end:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
pa_fail:
    mov x0, #1
pa_ret:
    ldp x29, x30, [sp], #16
    ret

parse_string:
    add x19, x19, #1
ps_loop:
    cmp x19, x20
    b.eq ps_fail
    ldrb w0, [x19]
    cmp w0, #'"'
    b.eq ps_end
    cmp w0, #'\\'
    b.eq ps_escape
    cmp w0, #0x20
    b.lt ps_fail
    add x19, x19, #1
    b ps_loop
ps_escape:
    add x19, x19, #1
    cmp x19, x20
    b.eq ps_fail
    ldrb w0, [x19]
    cmp w0, #'"'
    b.eq ps_esc_ok
    cmp w0, #'\\'
    b.eq ps_esc_ok
    cmp w0, #'/'
    b.eq ps_esc_ok
    cmp w0, #'b'
    b.eq ps_esc_ok
    cmp w0, #'f'
    b.eq ps_esc_ok
    cmp w0, #'n'
    b.eq ps_esc_ok
    cmp w0, #'r'
    b.eq ps_esc_ok
    cmp w0, #'t'
    b.eq ps_esc_ok
    cmp w0, #'u'
    b.eq ps_unicode
    b ps_fail
ps_esc_ok:
    add x19, x19, #1
    b ps_loop
ps_unicode:
    add x19, x19, #1
    mov w1, #4
ps_hex:
    cmp x19, x20
    b.eq ps_fail
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt ps_fail
    cmp w0, #'9'
    b.le ps_hex_ok
    cmp w0, #'A'
    b.lt ps_fail
    cmp w0, #'F'
    b.le ps_hex_ok
    cmp w0, #'a'
    b.lt ps_fail
    cmp w0, #'f'
    b.le ps_hex_ok
    b ps_fail
ps_hex_ok:
    add x19, x19, #1
    subs w1, w1, #1
    b.ne ps_hex
    b ps_loop
ps_end:
    add x19, x19, #1
    mov x0, #0
    ret
ps_fail:
    mov x0, #1
    ret

parse_number:
    ldrb w0, [x19]
    cmp w0, #'-'
    b.ne pn_int
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_fail
    ldrb w0, [x19]
pn_int:
    cmp w0, #'0'
    b.eq pn_zero
    cmp w0, #'1'
    b.lt pn_fail
    cmp w0, #'9'
    b.gt pn_fail
pn_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_ok
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pn_after_int
    cmp w0, #'9'
    b.gt pn_after_int
    b pn_digits
pn_zero:
    add x19, x19, #1
pn_after_int:
    cmp x19, x20
    b.eq pn_ok
    ldrb w0, [x19]
    cmp w0, #'.'
    b.eq pn_frac
    cmp w0, #'e'
    b.eq pn_exp
    cmp w0, #'E'
    b.eq pn_exp
    b pn_ok
pn_frac:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_fail
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pn_fail
    cmp w0, #'9'
    b.gt pn_fail
pn_frac_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_ok
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pn_after_frac
    cmp w0, #'9'
    b.gt pn_after_frac
    b pn_frac_digits
pn_after_frac:
    cmp w0, #'e'
    b.eq pn_exp
    cmp w0, #'E'
    b.eq pn_exp
    b pn_ok
pn_exp:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_fail
    ldrb w0, [x19]
    cmp w0, #'+'
    b.eq pn_exp_sign
    cmp w0, #'-'
    b.eq pn_exp_sign
    b pn_exp_digit
pn_exp_sign:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_fail
pn_exp_digit:
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pn_fail
    cmp w0, #'9'
    b.gt pn_fail
pn_exp_digits:
    add x19, x19, #1
    cmp x19, x20
    b.eq pn_ok
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pn_ok
    cmp w0, #'9'
    b.gt pn_ok
    b pn_exp_digits
pn_ok:
    mov x0, #0
    ret
pn_fail:
    mov x0, #1
    ret

parse_true:
    add x19, x19, #1
    cmp x19, x20
    b.eq pt_fail
    ldrb w0, [x19]
    cmp w0, #'r'
    b.ne pt_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pt_fail
    ldrb w0, [x19]
    cmp w0, #'u'
    b.ne pt_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pt_fail
    ldrb w0, [x19]
    cmp w0, #'e'
    b.ne pt_fail
    add x19, x19, #1
    mov x0, #0
    ret
pt_fail:
    mov x0, #1
    ret

parse_false:
    add x19, x19, #1
    cmp x19, x20
    b.eq pf_fail
    ldrb w0, [x19]
    cmp w0, #'a'
    b.ne pf_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pf_fail
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pf_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pf_fail
    ldrb w0, [x19]
    cmp w0, #'s'
    b.ne pf_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pf_fail
    ldrb w0, [x19]
    cmp w0, #'e'
    b.ne pf_fail
    add x19, x19, #1
    mov x0, #0
    ret
pf_fail:
    mov x0, #1
    ret

parse_null:
    add x19, x19, #1
    cmp x19, x20
    b.eq pnl_fail
    ldrb w0, [x19]
    cmp w0, #'u'
    b.ne pnl_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pnl_fail
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pnl_fail
    add x19, x19, #1
    cmp x19, x20
    b.eq pnl_fail
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pnl_fail
    add x19, x19, #1
    mov x0, #0
    ret
pnl_fail:
    mov x0, #1
    ret

print_uint:
    stp x29, x30, [sp, #-48]!
    add x29, sp, #48
    mov x1, x29
    mov x2, #10
    mov x3, x0
    cbnz x3, pu_loop
    sub x1, x1, #1
    mov w4, #'0'
    strb w4, [x1]
    b pu_out
pu_loop:
    udiv x4, x3, x2
    msub x5, x4, x2, x3
    add w5, w5, #'0'
    sub x1, x1, #1
    strb w5, [x1]
    mov x3, x4
    cbnz x3, pu_loop
pu_out:
    mov x0, #1
    sub x2, x29, x1
    mov x8, #SYS_write
    svc #0
    ldp x29, x30, [sp], #48
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg
error_prefix:
    .ascii "error at "
error_prefix_len = . - error_prefix
nl_msg:
    .ascii "\n"

.bss
.balign 16
input_buf:
    .space 65536
