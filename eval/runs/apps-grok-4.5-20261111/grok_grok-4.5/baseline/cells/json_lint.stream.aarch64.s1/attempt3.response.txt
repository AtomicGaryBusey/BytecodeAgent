.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.equ INPUT_MAX, 65536
.equ DEPTH_MAX, 1000

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x21, x19
    mov x20, x19

read_loop:
    mov x3, #INPUT_MAX
    sub x2, x20, x21
    sub x2, x3, x2
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x19, x21
    mov x22, #0
    bl skip_ws
    cmp x19, x20
    b.eq err_eof
    bl parse_value
    cbnz x0, err_pos
    bl skip_ws
    cmp x19, x20
    b.ne err_pos

    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

err_eof:
    mov x19, x20
err_pos:
    sub x23, x19, x21
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, x23
    bl print_num
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

print_num:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    mov x1, sp
    add x1, x1, #24
    mov x2, #0
    cmp x0, #0
    b.ne pn_loop
    mov w3, #'0'
    strb w3, [x1], #-1
    add x2, x2, #1
    b pn_done
pn_loop:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    strb w5, [x1], #-1
    add x2, x2, #1
    mov x0, x4
    cbnz x0, pn_loop
pn_done:
    add x1, x1, #1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov sp, x29
    ldp x29, x30, [sp], #16
    ret

skip_ws:
1:
    cmp x19, x20
    b.hs 2f
    ldrb w0, [x19]
    cmp w0, #0x20
    b.eq 3f
    cmp w0, #0x09
    b.eq 3f
    cmp w0, #0x0A
    b.eq 3f
    cmp w0, #0x0D
    b.eq 3f
    ret
3:
    add x19, x19, #1
    b 1b
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cmp x19, x20
    b.hs pv_err
    ldrb w0, [x19]
    cmp w0, #'{'
    b.eq pv_obj
    cmp w0, #'['
    b.eq pv_arr
    cmp w0, #'"'
    b.eq pv_str
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    cmp w0, #'-'
    b.eq pv_num
    cmp w0, #'0'
    b.lt pv_err
    cmp w0, #'9'
    b.gt pv_err
    b pv_num
pv_obj:
    bl parse_object
    b pv_done
pv_arr:
    bl parse_array
    b pv_done
pv_str:
    bl parse_string
    b pv_done
pv_true:
    bl parse_true
    b pv_done
pv_false:
    bl parse_false
    b pv_done
pv_null:
    bl parse_null
    b pv_done
pv_num:
    bl parse_number
    b pv_done
pv_err:
    mov x0, #1
pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_true:
    ldrb w0, [x19]
    cmp w0, #'t'
    b.ne pt_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pt_err
    ldrb w0, [x19]
    cmp w0, #'r'
    b.ne pt_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pt_err
    ldrb w0, [x19]
    cmp w0, #'u'
    b.ne pt_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pt_err
    ldrb w0, [x19]
    cmp w0, #'e'
    b.ne pt_err
    add x19, x19, #1
    mov x0, #0
    ret
pt_err:
    mov x0, #1
    ret

parse_false:
    ldrb w0, [x19]
    cmp w0, #'f'
    b.ne pf_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pf_err
    ldrb w0, [x19]
    cmp w0, #'a'
    b.ne pf_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pf_err
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pf_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pf_err
    ldrb w0, [x19]
    cmp w0, #'s'
    b.ne pf_err
    add x19, x19, #1
    cmp x19, x20
    b.hs pf_err
    ldrb w0, [x19]
    cmp w0, #'e'
    b.ne pf_err
    add x19, x19, #1
    mov x0, #0
    ret
pf_err:
    mov x0, #1
    ret

parse_null:
    ldrb w0, [x19]
    cmp w0, #'n'
    b.ne pn_err2
    add x19, x19, #1
    cmp x19, x20
    b.hs pn_err2
    ldrb w0, [x19]
    cmp w0, #'u'
    b.ne pn_err2
    add x19, x19, #1
    cmp x19, x20
    b.hs pn_err2
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pn_err2
    add x19, x19, #1
    cmp x19, x20
    b.hs pn_err2
    ldrb w0, [x19]
    cmp w0, #'l'
    b.ne pn_err2
    add x19, x19, #1
    mov x0, #0
    ret
pn_err2:
    mov x0, #1
    ret

parse_number:
    stp x29, x30, [sp, #-16]!
    cmp x19, x20
    b.hs pnum_err
    ldrb w0, [x19]
    cmp w0, #'-'
    b.ne pnum_int
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_err
    ldrb w0, [x19]
pnum_int:
    cmp w0, #'0'
    b.eq pnum_zero
    cmp w0, #'1'
    b.lt pnum_err
    cmp w0, #'9'
    b.gt pnum_err
    add x19, x19, #1
pnum_digits:
    cmp x19, x20
    b.hs pnum_frac
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pnum_frac
    cmp w0, #'9'
    b.gt pnum_frac
    add x19, x19, #1
    b pnum_digits
pnum_zero:
    add x19, x19, #1
pnum_frac:
    cmp x19, x20
    b.hs pnum_exp
    ldrb w0, [x19]
    cmp w0, #'.'
    b.ne pnum_exp
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_err
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pnum_err
    cmp w0, #'9'
    b.gt pnum_err
    add x19, x19, #1
pnum_frac_digits:
    cmp x19, x20
    b.hs pnum_exp
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pnum_exp
    cmp w0, #'9'
    b.gt pnum_exp
    add x19, x19, #1
    b pnum_frac_digits
pnum_exp:
    cmp x19, x20
    b.hs pnum_ok
    ldrb w0, [x19]
    cmp w0, #'e'
    b.eq pnum_exp_go
    cmp w0, #'E'
    b.ne pnum_ok
pnum_exp_go:
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_err
    ldrb w0, [x19]
    cmp w0, #'+'
    b.eq pnum_exp_sign
    cmp w0, #'-'
    b.eq pnum_exp_sign
    b pnum_exp_digit
pnum_exp_sign:
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_err
    ldrb w0, [x19]
pnum_exp_digit:
    cmp w0, #'0'
    b.lt pnum_err
    cmp w0, #'9'
    b.gt pnum_err
    add x19, x19, #1
pnum_exp_digits:
    cmp x19, x20
    b.hs pnum_ok
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt pnum_ok
    cmp w0, #'9'
    b.gt pnum_ok
    add x19, x19, #1
    b pnum_exp_digits
pnum_ok:
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
pnum_err:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_string:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x19]
    cmp w0, #'"'
    b.ne ps_err
    add x19, x19, #1
ps_loop:
    cmp x19, x20
    b.hs ps_err
    ldrb w0, [x19]
    cmp w0, #'"'
    b.eq ps_end
    cmp w0, #'\\'
    b.eq ps_esc
    cmp w0, #0x20
    b.lt ps_err
    add x19, x19, #1
    b ps_loop
ps_esc:
    add x19, x19, #1
    cmp x19, x20
    b.hs ps_err
    ldrb w0, [x19]
    cmp w0, #'"'
    b.eq ps_esc_ok
    cmp w0, #'\\'
    b.eq ps_esc_ok
    cmp w0, #'/'
    b.eq ps_esc_ok
    cmp w0, #'b'
    b.eq ps_esc_ok
    cmp w0, #'f'
    b.eq ps_esc_ok
    cmp w0, #'n'
    b.eq ps_esc_ok
    cmp w0, #'r'
    b.eq ps_esc_ok
    cmp w0, #'t'
    b.eq ps_esc_ok
    cmp w0, #'u'
    b.eq ps_unicode
    b ps_err
ps_esc_ok:
    add x19, x19, #1
    b ps_loop
ps_unicode:
    add x19, x19, #1
    mov x1, #4
ps_u_loop:
    cmp x19, x20
    b.hs ps_err
    ldrb w0, [x19]
    cmp w0, #'0'
    b.lt ps_u_af
    cmp w0, #'9'
    b.le ps_u_ok
ps_u_af:
    cmp w0, #'A'
    b.lt ps_u_af2
    cmp w0, #'F'
    b.le ps_u_ok
ps_u_af2:
    cmp w0, #'a'
    b.lt ps_err
    cmp w0, #'f'
    b.gt ps_err
ps_u_ok:
    add x19, x19, #1
    sub x1, x1, #1
    cbnz x1, ps_u_loop
    b ps_loop
ps_end:
    add x19, x19, #1
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret
ps_err:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-32]!
    str x22, [sp, #16]
    mov x29, sp
    add x22, x22, #1
    cmp x22, #DEPTH_MAX
    b.gt po_err
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs po_err
    ldrb w0, [x19]
    cmp w0, #'}'
    b.eq po_empty
po_member:
    bl parse_string
    cbnz x0, po_err
    bl skip_ws
    cmp x19, x20
    b.hs po_err
    ldrb w0, [x19]
    cmp w0, #':'
    b.ne po_err
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    cbnz x0, po_err
    bl skip_ws
    cmp x19, x20
    b.hs po_err
    ldrb w0, [x19]
    cmp w0, #','
    b.eq po_comma
    cmp w0, #'}'
    b.eq po_end
    b po_err
po_comma:
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs po_err
    b po_member
po_empty:
po_end:
    add x19, x19, #1
    mov x0, #0
    b po_done
po_err:
    mov x0, #1
po_done:
    ldr x22, [sp, #16]
    ldp x29, x30, [sp], #32
    ret

parse_array:
    stp x29, x30, [sp, #-32]!
    str x22, [sp, #16]
    mov x29, sp
    add x22, x22, #1
    cmp x22, #DEPTH_MAX
    b.gt pa_err
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs pa_err
    ldrb w0, [x19]
    cmp w0, #']'
    b.eq pa_empty
pa_value:
    bl parse_value
    cbnz x0, pa_err
    bl skip_ws
    cmp x19, x20
    b.hs pa_err
    ldrb w0, [x19]
    cmp w0, #','
    b.eq pa_comma
    cmp w0, #']'
    b.eq pa_end
    b pa_err
pa_comma:
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.hs pa_err
    b pa_value
pa_empty:
pa_end:
    add x19, x19, #1
    mov x0, #0
    b pa_done
pa_err:
    mov x0, #1
pa_done:
    ldr x22, [sp, #16]
    ldp x29, x30, [sp], #32
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
nl_msg:
    .ascii "\n"

.bss
.align 4
input_buf:
    .zero 65536
