.globl _start

.text
_start:
	lea r12, [rip + buffer]
	lea r13, [r12 + 65536]
read_loop:
	mov rax, 0
	xor rdi, rdi
	mov rsi, r12
	mov rdx, r13
	sub rdx, r12
	jz read_done
	syscall
	test rax, rax
	jle read_done
	add r12, rax
	jmp read_loop
read_done:
	lea r14, [rip + buffer]
	mov r15, r12
	mov rbx, r14
	call parse_document
	jc do_error
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + msg_ok]
	mov rdx, msg_ok_len
	syscall
	mov rax, 60
	xor rdi, rdi
	syscall

do_error:
	mov r8, rbx
	sub r8, r14
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + msg_err]
	mov rdx, msg_err_len
	syscall
	lea rdi, [rip + numbuf + 30]
	mov byte ptr [rdi], 10
	mov rax, r8
	mov r9, 10
	test rax, rax
	jnz dec_loop
	dec rdi
	mov byte ptr [rdi], '0'
	jmp dec_done
dec_loop:
	xor rdx, rdx
	div r9
	add dl, '0'
	dec rdi
	mov byte ptr [rdi], dl
	test rax, rax
	jnz dec_loop
dec_done:
	lea rdx, [rip + numbuf + 30]
	sub rdx, rdi
	inc rdx
	mov rsi, rdi
	mov rax, 1
	mov rdi, 1
	syscall
	mov rax, 60
	mov rdi, 1
	syscall

skip_ws:
	cmp rbx, r15
	jae skip_ws_done
	movzx eax, byte ptr [rbx]
	cmp al, 0x20
	je skip_ws_adv
	cmp al, 0x09
	je skip_ws_adv
	cmp al, 0x0A
	je skip_ws_adv
	cmp al, 0x0D
	je skip_ws_adv
skip_ws_done:
	ret
skip_ws_adv:
	inc rbx
	jmp skip_ws

parse_document:
	call skip_ws
	cmp rbx, r15
	je parse_doc_fail
	call parse_value
	jc parse_doc_fail
	call skip_ws
	cmp rbx, r15
	jne parse_doc_fail
	clc
	ret
parse_doc_fail:
	stc
	ret

parse_value:
	cmp rbx, r15
	jae parse_val_fail
	movzx eax, byte ptr [rbx]
	cmp al, '{'
	je parse_object
	cmp al, '['
	je parse_array
	cmp al, '"'
	je parse_string
	cmp al, 't'
	je parse_true
	cmp al, 'f'
	je parse_false
	cmp al, 'n'
	je parse_null
	cmp al, '-'
	je parse_number
	cmp al, '0'
	jb parse_val_fail
	cmp al, '9'
	ja parse_val_fail
	jmp parse_number
parse_val_fail:
	stc
	ret

parse_true:
	mov ecx, 4
	lea rsi, [rip + lit_true]
	jmp match_literal

parse_false:
	mov ecx, 5
	lea rsi, [rip + lit_false]
	jmp match_literal

parse_null:
	mov ecx, 4
	lea rsi, [rip + lit_null]
	jmp match_literal

match_literal:
	test ecx, ecx
	jz match_ok
	cmp rbx, r15
	jae match_fail
	mov al, byte ptr [rbx]
	cmp al, byte ptr [rsi]
	jne match_fail
	inc rbx
	inc rsi
	dec ecx
	jmp match_literal
match_ok:
	clc
	ret
match_fail:
	stc
	ret

parse_object:
	inc rbx
	call skip_ws
	cmp rbx, r15
	jae obj_fail
	cmp byte ptr [rbx], '}'
	je obj_end
obj_member:
	call parse_string
	jc obj_fail
	call skip_ws
	cmp rbx, r15
	jae obj_fail
	cmp byte ptr [rbx], ':'
	jne obj_fail
	inc rbx
	call skip_ws
	call parse_value
	jc obj_fail
	call skip_ws
	cmp rbx, r15
	jae obj_fail
	movzx eax, byte ptr [rbx]
	cmp al, '}'
	je obj_end
	cmp al, ','
	jne obj_fail
	inc rbx
	call skip_ws
	jmp obj_member
obj_end:
	inc rbx
	clc
	ret
obj_fail:
	stc
	ret

parse_array:
	inc rbx
	call skip_ws
	cmp rbx, r15
	jae arr_fail
	cmp byte ptr [rbx], ']'
	je arr_end
arr_value:
	call parse_value
	jc arr_fail
	call skip_ws
	cmp rbx, r15
	jae arr_fail
	movzx eax, byte ptr [rbx]
	cmp al, ']'
	je arr_end
	cmp al, ','
	jne arr_fail
	inc rbx
	call skip_ws
	jmp arr_value
arr_end:
	inc rbx
	clc
	ret
arr_fail:
	stc
	ret

parse_string:
	cmp rbx, r15
	jae str_fail
	cmp byte ptr [rbx], '"'
	jne str_fail
	inc rbx
str_loop:
	cmp rbx, r15
	jae str_fail
	movzx eax, byte ptr [rbx]
	cmp al, '"'
	je str_done
	cmp al, 0x5C
	je str_escape
	cmp al, 0x20
	jb str_fail
	inc rbx
	jmp str_loop
str_escape:
	inc rbx
	cmp rbx, r15
	jae str_fail
	movzx eax, byte ptr [rbx]
	cmp al, '"'
	je str_esc_ok
	cmp al, 0x5C
	je str_esc_ok
	cmp al, '/'
	je str_esc_ok
	cmp al, 'b'
	je str_esc_ok
	cmp al, 'f'
	je str_esc_ok
	cmp al, 'n'
	je str_esc_ok
	cmp al, 'r'
	je str_esc_ok
	cmp al, 't'
	je str_esc_ok
	cmp al, 'u'
	je str_unicode
	jmp str_fail
str_esc_ok:
	inc rbx
	jmp str_loop
str_unicode:
	inc rbx
	mov ecx, 4
str_u_loop:
	cmp rbx, r15
	jae str_fail
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb str_fail
	cmp al, '9'
	jbe str_u_ok
	cmp al, 'A'
	jb str_fail
	cmp al, 'F'
	jbe str_u_ok
	cmp al, 'a'
	jb str_fail
	cmp al, 'f'
	jbe str_u_ok
	jmp str_fail
str_u_ok:
	inc rbx
	dec ecx
	jnz str_u_loop
	jmp str_loop
str_done:
	inc rbx
	clc
	ret
str_fail:
	stc
	ret

parse_number:
	cmp rbx, r15
	jae num_fail
	cmp byte ptr [rbx], '-'
	jne num_int
	inc rbx
num_int:
	cmp rbx, r15
	jae num_fail
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	je num_zero
	cmp al, '1'
	jb num_fail
	cmp al, '9'
	ja num_fail
	inc rbx
num_digits:
	cmp rbx, r15
	jae num_after_int
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb num_after_int
	cmp al, '9'
	ja num_after_int
	inc rbx
	jmp num_digits
num_zero:
	inc rbx
num_after_int:
	cmp rbx, r15
	jae num_ok
	cmp byte ptr [rbx], '.'
	jne num_check_exp
	inc rbx
	cmp rbx, r15
	jae num_fail
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb num_fail
	cmp al, '9'
	ja num_fail
	inc rbx
num_frac:
	cmp rbx, r15
	jae num_check_exp
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb num_check_exp
	cmp al, '9'
	ja num_check_exp
	inc rbx
	jmp num_frac
num_check_exp:
	cmp rbx, r15
	jae num_ok
	movzx eax, byte ptr [rbx]
	cmp al, 'e'
	je num_do_exp
	cmp al, 'E'
	je num_do_exp
	jmp num_ok
num_do_exp:
	inc rbx
	cmp rbx, r15
	jae num_fail
	movzx eax, byte ptr [rbx]
	cmp al, '+'
	je num_exp_sign
	cmp al, '-'
	je num_exp_sign
	jmp num_exp_digits
num_exp_sign:
	inc rbx
num_exp_digits:
	cmp rbx, r15
	jae num_fail
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb num_fail
	cmp al, '9'
	ja num_fail
	inc rbx
num_exp_more:
	cmp rbx, r15
	jae num_ok
	movzx eax, byte ptr [rbx]
	cmp al, '0'
	jb num_ok
	cmp al, '9'
	ja num_ok
	inc rbx
	jmp num_exp_more
num_ok:
	clc
	ret
num_fail:
	stc
	ret

.section .rodata
msg_ok:
	.ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err:
	.ascii "error at "
msg_err_len = . - msg_err
lit_true:
	.ascii "true"
lit_false:
	.ascii "false"
lit_null:
	.ascii "null"

.bss
.align 8
buffer:
	.space 65536
numbuf:
	.space 32
