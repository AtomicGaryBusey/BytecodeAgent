.globl _start
.text
_start:
  lea rbx, [rip + input_buf]
  mov rsi, rbx
  mov r12, 32768
read_loop:
  xor eax, eax
  xor edi, edi
  mov rdx, r12
  syscall
  test rax, rax
  jle read_done
  add rsi, rax
  sub r12, rax
  jnz read_loop
read_done:
  mov r14, rsi
  sub r14, rbx
  lea r15, [rip + lines]
  xor r13d, r13d
  mov rsi, rbx
  xor ecx, ecx
parse_loop:
  cmp rcx, r14
  jae parse_done
  cmp byte ptr [rbx + rcx], 10
  jne parse_cont
  mov qword ptr [r15], rsi
  mov rax, rcx
  add rax, rbx
  sub rax, rsi
  mov qword ptr [r15 + 8], rax
  add r15, 16
  inc r13
  lea rsi, [rbx + rcx + 1]
parse_cont:
  inc rcx
  jmp parse_loop
parse_done:
  lea rax, [rbx + r14]
  cmp rsi, rax
  jae parse_no_final
  mov qword ptr [r15], rsi
  sub rax, rsi
  mov qword ptr [r15 + 8], rax
  inc r13
parse_no_final:
  test r13, r13
  jz do_exit
  lea r12, [rip + lines]
  mov r8, r13
  dec r8
  xor ebx, ebx
outer_loop:
  cmp rbx, r8
  jge sort_done
  mov rcx, r8
  sub rcx, rbx
  xor edx, edx
inner_loop:
  cmp rdx, rcx
  jge outer_next
  mov rdi, rdx
  shl rdi, 4
  add rdi, r12
  lea rsi, [rdi + 16]
  push rbx
  push rcx
  push rdx
  push r8
  push r12
  push r13
  call cmp_lines
  pop r13
  pop r12
  pop r8
  pop rdx
  pop rcx
  pop rbx
  test rax, rax
  jle no_swap
  mov rdi, rdx
  shl rdi, 4
  add rdi, r12
  lea rsi, [rdi + 16]
  mov rax, qword ptr [rdi]
  xchg rax, qword ptr [rsi]
  mov qword ptr [rdi], rax
  mov rax, qword ptr [rdi + 8]
  xchg rax, qword ptr [rsi + 8]
  mov qword ptr [rdi + 8], rax
no_swap:
  inc rdx
  jmp inner_loop
outer_next:
  inc rbx
  jmp outer_loop
sort_done:
  xor ebx, ebx
print_loop:
  cmp rbx, r13
  jge do_exit
  mov rax, rbx
  shl rax, 4
  add rax, r12
  mov rsi, qword ptr [rax]
  mov rdx, qword ptr [rax + 8]
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + newline]
  mov edx, 1
  syscall
  inc rbx
  jmp print_loop
do_exit:
  mov eax, 60
  xor edi, edi
  syscall
cmp_lines:
  mov r8, qword ptr [rdi]
  mov r9, qword ptr [rdi + 8]
  mov r10, qword ptr [rsi]
  mov r11, qword ptr [rsi + 8]
  mov rcx, r9
  cmp rcx, r11
  cmova rcx, r11
  test rcx, rcx
  jz 2f
1:
  movzx eax, byte ptr [r8]
  movzx edx, byte ptr [r10]
  cmp eax, edx
  jne 3f
  inc r8
  inc r10
  dec rcx
  jnz 1b
2:
  cmp r9, r11
  jb 4f
  ja 5f
  xor eax, eax
  ret
3:
  jb 4f
5:
  mov rax, 1
  ret
4:
  mov rax, -1
  ret
.section .rodata
newline:
.byte 10
.bss
.align 8
input_buf:
.space 32768
.align 16
lines:
.space 16000
