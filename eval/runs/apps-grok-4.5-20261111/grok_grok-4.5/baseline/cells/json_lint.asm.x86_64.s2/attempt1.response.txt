.globl _start

.text
_start:
	lea rbx, [rip + buffer]
	xor r12d, r12d
.Lread:
	mov eax, 65536
	sub eax, r12d
	jz .Lread_done
	mov edx, eax
	lea rsi, [rbx + r12]
	xor edi, edi
	xor eax, eax
	syscall
	test rax, rax
	jle .Lread_done
	add r12, rax
	jmp .Lread
.Lread_done:
	lea r15, [rbx + r12]
	mov rsi, rbx
	call skip_ws
	cmp rsi, r15
	jae fail
	call parse_value
	call skip_ws
	cmp rsi, r15
	jne fail
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + msg_ok]
	mov edx, 3
	syscall
	mov eax, 60
	xor edi, edi
	syscall

fail:
	mov r8, rsi
	sub r8, rbx
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + msg_err]
	mov edx, msg_err_len
	syscall
	sub rsp, 32
	lea rdi, [rsp + 32]
	mov rax, r8
	mov ecx, 10
	test rax, rax
	jnz .Lconv
	dec rdi
	mov byte ptr [rdi], '0'
	jmp .Lwrite_num
.Lconv:
.Lconv_loop:
	xor edx, edx
	div rcx
	add dl, '0'
	dec rdi
	mov byte ptr [rdi], dl
	test rax, rax
	jnz .Lconv_loop
.Lwrite_num:
	mov rsi, rdi
	lea rdx, [rsp + 32]
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall
	add rsp, 32
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + nl]
	mov edx, 1
	syscall
	mov eax, 60
	mov edi, 1
	syscall

skip_ws:
.Lsw_loop:
	cmp rsi, r15
	jae .Lsw_ret
	movzx eax, byte ptr [rsi]
	cmp al, 0x20
	je .Lsw_adv
	cmp al, 9
	je .Lsw_adv
	cmp al, 10
	je .Lsw_adv
	cmp al, 13
	je .Lsw_adv
.Lsw_ret:
	ret
.Lsw_adv:
	inc rsi
	jmp .Lsw_loop

parse_value:
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '{'
	je parse_object
	cmp al, '['
	je parse_array
	cmp al, '"'
	je parse_string
	cmp al, 't'
	je parse_true
	cmp al, 'f'
	je parse_false
	cmp al, 'n'
	je parse_null
	cmp al, '-'
	je parse_number
	cmp al, '0'
	jb fail
	cmp al, '9'
	ja fail
	jmp parse_number

parse_object:
	inc rsi
	call skip_ws
	cmp rsi, r15
	jae fail
	cmp byte ptr [rsi], '}'
	jne .Lpo_member
	inc rsi
	ret
.Lpo_member:
	cmp byte ptr [rsi], '"'
	jne fail
	call parse_string
	call skip_ws
	cmp rsi, r15
	jae fail
	cmp byte ptr [rsi], ':'
	jne fail
	inc rsi
	call skip_ws
	call parse_value
	call skip_ws
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '}'
	je .Lpo_end
	cmp al, ','
	jne fail
	inc rsi
	call skip_ws
	jmp .Lpo_member
.Lpo_end:
	inc rsi
	ret

parse_array:
	inc rsi
	call skip_ws
	cmp rsi, r15
	jae fail
	cmp byte ptr [rsi], ']'
	jne .Lpa_value
	inc rsi
	ret
.Lpa_value:
	call parse_value
	call skip_ws
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, ']'
	je .Lpa_end
	cmp al, ','
	jne fail
	inc rsi
	call skip_ws
	jmp .Lpa_value
.Lpa_end:
	inc rsi
	ret

parse_string:
	inc rsi
.Lps_loop:
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '"'
	je .Lps_done
	cmp al, 0x5C
	je .Lps_esc
	cmp al, 0x20
	jb fail
	inc rsi
	jmp .Lps_loop
.Lps_esc:
	inc rsi
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '"'
	je .Lps_esc_ok
	cmp al, 0x5C
	je .Lps_esc_ok
	cmp al, '/'
	je .Lps_esc_ok
	cmp al, 'b'
	je .Lps_esc_ok
	cmp al, 'f'
	je .Lps_esc_ok
	cmp al, 'n'
	je .Lps_esc_ok
	cmp al, 'r'
	je .Lps_esc_ok
	cmp al, 't'
	je .Lps_esc_ok
	cmp al, 'u'
	je .Lps_uni
	jmp fail
.Lps_esc_ok:
	inc rsi
	jmp .Lps_loop
.Lps_uni:
	inc rsi
	mov ecx, 4
.Lps_hex:
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb fail
	cmp al, '9'
	jbe .Lps_hex_ok
	cmp al, 'A'
	jb fail
	cmp al, 'F'
	jbe .Lps_hex_ok
	cmp al, 'a'
	jb fail
	cmp al, 'f'
	ja fail
.Lps_hex_ok:
	inc rsi
	dec ecx
	jnz .Lps_hex
	jmp .Lps_loop
.Lps_done:
	inc rsi
	ret

parse_number:
	cmp byte ptr [rsi], '-'
	jne .Lpn_int
	inc rsi
	cmp rsi, r15
	jae fail
.Lpn_int:
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	je .Lpn_zero
	cmp al, '1'
	jb fail
	cmp al, '9'
	ja fail
.Lpn_digits:
	inc rsi
	cmp rsi, r15
	jae .Lpn_ret
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb .Lpn_after_int
	cmp al, '9'
	jbe .Lpn_digits
	jmp .Lpn_after_int
.Lpn_zero:
	inc rsi
	cmp rsi, r15
	jae .Lpn_ret
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb .Lpn_after_int
	cmp al, '9'
	jbe fail
.Lpn_after_int:
	cmp al, '.'
	jne .Lpn_check_exp
	inc rsi
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb fail
	cmp al, '9'
	ja fail
.Lpn_frac_dig:
	inc rsi
	cmp rsi, r15
	jae .Lpn_ret
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb .Lpn_check_exp
	cmp al, '9'
	jbe .Lpn_frac_dig
.Lpn_check_exp:
	cmp al, 'e'
	je .Lpn_exp
	cmp al, 'E'
	je .Lpn_exp
.Lpn_ret:
	ret
.Lpn_exp:
	inc rsi
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
	cmp al, '+'
	je .Lpn_exp_sign
	cmp al, '-'
	je .Lpn_exp_sign
	jmp .Lpn_exp_need_dig
.Lpn_exp_sign:
	inc rsi
	cmp rsi, r15
	jae fail
	movzx eax, byte ptr [rsi]
.Lpn_exp_need_dig:
	cmp al, '0'
	jb fail
	cmp al, '9'
	ja fail
.Lpn_exp_dig:
	inc rsi
	cmp rsi, r15
	jae .Lpn_ret
	movzx eax, byte ptr [rsi]
	cmp al, '0'
	jb .Lpn_ret
	cmp al, '9'
	jbe .Lpn_exp_dig
	ret

parse_true:
	lea rdx, [rip + str_true]
	mov ecx, 4
	jmp parse_lit

parse_false:
	lea rdx, [rip + str_false]
	mov ecx, 5
	jmp parse_lit

parse_null:
	lea rdx, [rip + str_null]
	mov ecx, 4
parse_lit:
.Lpl_loop:
	cmp rsi, r15
	jae fail
	mov al, byte ptr [rdx]
	cmp al, byte ptr [rsi]
	jne fail
	inc rsi
	inc rdx
	dec ecx
	jnz .Lpl_loop
	ret

.section .rodata
msg_ok:
	.ascii "ok\n"
msg_err:
	.ascii "error at "
msg_err_len = . - msg_err
nl:
	.ascii "\n"
str_true:
	.ascii "true"
str_false:
	.ascii "false"
str_null:
	.ascii "null"

.bss
.align 16
buffer:
	.zero 65536
