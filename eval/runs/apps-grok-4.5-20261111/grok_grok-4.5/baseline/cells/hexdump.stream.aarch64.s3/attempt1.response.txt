.text
.globl _start
_start:
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0

read_loop:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop

read_done:
	adrp	x21, outbuf
	add	x21, x21, :lo12:outbuf
	mov	x26, x21
	adrp	x27, hextab
	add	x27, x27, :lo12:hextab
	mov	x22, #0

row_loop:
	cmp	x22, x20
	b.hs	print_total
	sub	x23, x20, x22
	cmp	x23, #16
	b.ls	1f
	mov	x23, #16
1:
	mov	w0, w22
	bl	emit_hex8
	mov	w0, #' '
	strb	w0, [x21], #1
	strb	w0, [x21], #1
	mov	x24, #0

hex_loop:
	cmp	x24, #16
	b.ge	hex_done
	cmp	x24, x23
	b.ge	emit_pad
	add	x0, x19, x22
	ldrb	w0, [x0, x24]
	bl	emit_hex2
	mov	w0, #' '
	strb	w0, [x21], #1
	b	hex_next

emit_pad:
	mov	w0, #' '
	strb	w0, [x21], #1
	strb	w0, [x21], #1
	strb	w0, [x21], #1

hex_next:
	cmp	x24, #7
	b.ne	2f
	mov	w0, #' '
	strb	w0, [x21], #1
2:
	add	x24, x24, #1
	b	hex_loop

hex_done:
	mov	w0, #' '
	strb	w0, [x21], #1
	mov	w0, #'|'
	strb	w0, [x21], #1
	mov	x24, #0

ascii_loop:
	cmp	x24, x23
	b.ge	ascii_done
	add	x0, x19, x22
	ldrb	w0, [x0, x24]
	cmp	w0, #0x20
	b.lt	make_dot
	cmp	w0, #0x7f
	b.lt	store_char
make_dot:
	mov	w0, #'.'
store_char:
	strb	w0, [x21], #1
	add	x24, x24, #1
	b	ascii_loop

ascii_done:
	mov	w0, #'|'
	strb	w0, [x21], #1
	mov	w0, #'\n'
	strb	w0, [x21], #1
	add	x22, x22, #16
	b	row_loop

print_total:
	mov	w0, w20
	bl	emit_hex8
	mov	w0, #'\n'
	strb	w0, [x21], #1
	mov	x1, x26
	sub	x25, x21, x26

write_more:
	cbz	x25, write_done
	mov	x0, #1
	mov	x2, x25
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	write_done
	add	x1, x1, x0
	sub	x25, x25, x0
	b	write_more

write_done:
	mov	x0, #0
	mov	x8, #93
	svc	#0

emit_hex8:
	mov	w1, w0
	mov	w2, #8
1:
	lsr	w3, w1, #28
	and	w3, w3, #0xf
	ldrb	w3, [x27, x3]
	strb	w3, [x21], #1
	lsl	w1, w1, #4
	subs	w2, w2, #1
	b.ne	1b
	ret

emit_hex2:
	lsr	w1, w0, #4
	and	w1, w1, #0xf
	ldrb	w1, [x27, x1]
	strb	w1, [x21], #1
	and	w1, w0, #0xf
	ldrb	w1, [x27, x1]
	strb	w1, [x21], #1
	ret

.section .rodata
hextab:
	.ascii	"0123456789abcdef"

.bss
.align	4
inbuf:
	.space	65536
outbuf:
	.space	524288
