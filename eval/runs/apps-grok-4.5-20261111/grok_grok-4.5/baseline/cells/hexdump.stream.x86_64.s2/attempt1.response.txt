.globl _start

.section .rodata
hexdigits:
.ascii "0123456789abcdef"

.section .bss
.align 16
inbuf:
.zero 65536
linebuf:
.zero 80

.text
put_hex8:
    push rcx
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [rip + hexdigits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    pop rcx
    ret

_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_more:
    mov eax, 65536
    sub eax, r13d
    jz read_done
    mov edx, eax
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    add r13, rax
    jmp read_more
read_done:
    lea r12, [rip + inbuf]
    xor r14, r14
print_rows:
    cmp r14, r13
    jae print_total
    mov rcx, r13
    sub rcx, r14
    cmp rcx, 16
    jbe 2f
    mov ecx, 16
2:
    lea rdi, [rip + linebuf]
    mov eax, r14d
    call put_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [r12 + r14]
    xor rbx, rbx
hex_loop:
    cmp ebx, 8
    jne 3f
    mov byte ptr [rdi], 0x20
    inc rdi
3:
    cmp ebx, ecx
    jae do_pad
    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr eax, 4
    and edx, 0xf
    mov al, byte ptr [rip + hexdigits + rax]
    mov byte ptr [rdi], al
    mov al, byte ptr [rip + hexdigits + rdx]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp next_hex
do_pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next_hex:
    inc ebx
    cmp ebx, 16
    jb hex_loop
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    xor rbx, rbx
ascii_loop:
    cmp ebx, ecx
    jae ascii_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb do_dot
    cmp al, 0x7e
    ja do_dot
    mov byte ptr [rdi], al
    jmp next_ascii
do_dot:
    mov byte ptr [rdi], '.'
next_ascii:
    inc rdi
    inc ebx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r14, 16
    jmp print_rows
print_total:
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call put_hex8
    mov byte ptr [rdi], 10
    lea rsi, [rip + linebuf]
    mov edx, 9
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
