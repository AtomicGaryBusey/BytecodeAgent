.globl _start

.section .rodata
hexdigits:
.ascii "0123456789abcdef"

.section .bss
.align 16
inbuf:
.zero 65536
.align 16
linebuf:
.zero 80

.text

put_hex2:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [hexdigits + rdx]
    mov byte ptr [rdi], dl
    and eax, 0xf
    mov al, byte ptr [hexdigits + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 2
    ret

put_hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [hexdigits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

_start:
    xor ebx, ebx
read_loop:
    cmp ebx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [inbuf + rbx]
    mov edx, 65536
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add ebx, eax
    jmp read_loop
read_done:
    mov r13d, ebx
    xor r12d, r12d

dump_loop:
    cmp r12d, r13d
    jae dump_done

    mov r14d, r13d
    sub r14d, r12d
    cmp r14d, 16
    jbe 2f
    mov r14d, 16
2:
    lea rdi, [linebuf]
    mov eax, r12d
    call put_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor ecx, ecx
hex_loop:
    cmp ecx, 16
    jae hex_done
    cmp ecx, 8
    jne 3f
    mov byte ptr [rdi], 0x20
    inc rdi
3:
    cmp ecx, r14d
    jae pad_slot
    movzx eax, byte ptr [inbuf + r12 + rcx]
    push rcx
    call put_hex2
    pop rcx
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp next_hex
pad_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next_hex:
    inc ecx
    jmp hex_loop
hex_done:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor ecx, ecx
asc_loop:
    cmp ecx, r14d
    jae asc_done
    movzx eax, byte ptr [inbuf + r12 + rcx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    mov byte ptr [rdi], al
    jmp store_asc
dot:
    mov byte ptr [rdi], '.'
store_asc:
    inc rdi
    inc ecx
    jmp asc_loop
asc_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r12d, r14d
    jmp dump_loop

dump_done:
    lea rdi, [linebuf]
    mov eax, r13d
    call put_hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
