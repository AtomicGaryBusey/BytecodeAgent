.globl _start
.text
_start:
	xor	r12, r12
	xor	r13, r13
	xor	r14, r14
	xor	r15, r15
	lea	rsi, [rip + buffer]
	mov	edx, 4096
.read:
	xor	eax, eax
	xor	edi, edi
	syscall
	test	rax, rax
	jle	.done
	add	r14, rax
	mov	rcx, rax
	mov	rdi, rsi
.Lprocess:
	movzx	eax, byte ptr [rdi]
	cmp	al, 0x20
	je	.Lws
	cmp	al, 0x09
	je	.Lws
	cmp	al, 0x0A
	je	.Lnl
	cmp	al, 0x0B
	je	.Lws
	cmp	al, 0x0C
	je	.Lws
	cmp	al, 0x0D
	je	.Lws
	test	r15, r15
	jnz	.Lnext
	inc	r13
	mov	r15, 1
	jmp	.Lnext
.Lnl:
	inc	r12
.Lws:
	xor	r15, r15
.Lnext:
	inc	rdi
	dec	rcx
	jnz	.Lprocess
	jmp	.read
.done:
	lea	rdi, [rip + outbuf + 63]
	mov	byte ptr [rdi], 0x0A
	dec	rdi
	mov	rax, r14
	call	append_dec
	mov	byte ptr [rdi], 0x20
	dec	rdi
	mov	rax, r13
	call	append_dec
	mov	byte ptr [rdi], 0x20
	dec	rdi
	mov	rax, r12
	call	append_dec
	lea	rsi, [rdi + 1]
	lea	rdx, [rip + outbuf + 64]
	sub	rdx, rsi
	mov	eax, 1
	mov	edi, 1
	syscall
	mov	eax, 60
	xor	edi, edi
	syscall
append_dec:
	push	rbx
	mov	ebx, 10
	test	rax, rax
	jnz	.L1
	mov	byte ptr [rdi], '0'
	dec	rdi
	pop	rbx
	ret
.L1:
.L2:
	xor	rdx, rdx
	div	rbx
	add	dl, '0'
	mov	byte ptr [rdi], dl
	dec	rdi
	test	rax, rax
	jnz	.L2
	pop	rbx
	ret
.bss
