.globl _start

.equ BUFSIZE, 65536

.text
_start:
    lea r12, [rip + buffer]
    mov r13, BUFSIZE
read_loop:
    test r13, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    sub r13, rax
    jmp read_loop
read_done:
    lea rbx, [rip + buffer]
    mov r14, r12

    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r14
    jne error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + msg_err]
    mov ecx, 9
    rep movsb

    lea rax, [rip + buffer]
    mov rax, rbx
    sub rax, [rip + buffer_addr]

    lea r8, [rip + numtmp + 16]
    mov r9, r8
    mov ecx, 10
    test rax, rax
    jnz conv_loop
    dec r8
    mov byte ptr [r8], '0'
    jmp conv_done
conv_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz conv_loop
conv_done:
    mov rsi, r8
    mov rcx, r9
    sub rcx, r8
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp rbx, r14
    jae skip_ws_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je skip_ws_adv
    cmp al, 0x09
    je skip_ws_adv
    cmp al, 0x0A
    je skip_ws_adv
    cmp al, 0x0D
    je skip_ws_adv
skip_ws_done:
    ret
skip_ws_adv:
    inc rbx
    jmp skip_ws

expect:
    cmp rbx, r14
    jae error
    cmp byte ptr [rbx], al
    jne error
    inc rbx
    ret

parse_value:
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe parse_number
    jmp error

parse_true:
    inc rbx
    mov al, 'r'
    call expect
    mov al, 'u'
    call expect
    mov al, 'e'
    call expect
    ret

parse_false:
    inc rbx
    mov al, 'a'
    call expect
    mov al, 'l'
    call expect
    mov al, 's'
    call expect
    mov al, 'e'
    call expect
    ret

parse_null:
    inc rbx
    mov al, 'u'
    call expect
    mov al, 'l'
    call expect
    mov al, 'l'
    call expect
    ret

parse_string:
    inc rbx
str_loop:
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je str_done
    cmp al, '\\'
    je str_escape
    cmp al, 0x20
    jb error
    inc rbx
    jmp str_loop
str_escape:
    inc rbx
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je str_esc_ok
    cmp al, '\\'
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_unicode
    jmp error
str_esc_ok:
    inc rbx
    jmp str_loop
str_unicode:
    inc rbx
    mov ecx, 4
str_hex:
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe str_hex_ok
    cmp al, 'A'
    jb error
    cmp al, 'F'
    jbe str_hex_ok
    cmp al, 'a'
    jb error
    cmp al, 'f'
    jbe str_hex_ok
    jmp error
str_hex_ok:
    inc rbx
    dec ecx
    jnz str_hex
    jmp str_loop
str_done:
    inc rbx
    ret

parse_number:
    cmp byte ptr [rbx], '-'
    jne num_int
    inc rbx
    cmp rbx, r14
    jae error
num_int:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
num_int_digits:
    inc rbx
    cmp rbx, r14
    jae num_after_int
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_after_int
    cmp al, '9'
    jbe num_int_digits
    jmp num_after_int
num_zero:
    inc rbx
num_after_int:
    cmp rbx, r14
    jae num_done
    cmp byte ptr [rbx], '.'
    jne num_check_exp
    inc rbx
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
num_frac_digits:
    inc rbx
    cmp rbx, r14
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_check_exp
    cmp al, '9'
    jbe num_frac_digits
num_check_exp:
    cmp rbx, r14
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, 'e'
    je num_exp
    cmp al, 'E'
    je num_exp
    jmp num_done
num_exp:
    inc rbx
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    je num_exp_sign
    jmp num_exp_digit
num_exp_sign:
    inc rbx
    cmp rbx, r14
    jae error
num_exp_digit:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
num_exp_digits:
    inc rbx
    cmp rbx, r14
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    jbe num_exp_digits
num_done:
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r14
    jae error
    cmp byte ptr [rbx], '}'
    je obj_done
obj_member:
    cmp rbx, r14
    jae error
    cmp byte ptr [rbx], '"'
    jne error
    call parse_string
    call skip_ws
    cmp rbx, r14
    jae error
    cmp byte ptr [rbx], ':'
    jne error
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je obj_next
    cmp al, '}'
    je obj_done
    jmp error
obj_next:
    inc rbx
    call skip_ws
    jmp obj_member
obj_done:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r14
    jae error
    cmp byte ptr [rbx], ']'
    je arr_done
arr_value:
    call parse_value
    call skip_ws
    cmp rbx, r14
    jae error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je arr_next
    cmp al, ']'
    je arr_done
    jmp error
arr_next:
    inc rbx
    call skip_ws
    cmp rbx, r14
    jae error
    jmp arr_value
arr_done:
    inc rbx
    ret

.section .rodata
buffer_addr:
    .quad buffer
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "

.bss
.align 16
buffer:
    .space BUFSIZE
outbuf:
    .space 64
numtmp:
    .space 32
