.globl _start

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.section .bss
.align 16
input:
	.zero 65536
linebuf:
	.zero 80

.text
_start:
	xor ebx, ebx
read_loop:
	mov eax, 65536
	sub eax, ebx
	jz read_done
	mov edx, eax
	lea rsi, [rip + input]
	add rsi, rbx
	xor edi, edi
	xor eax, eax
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	jmp read_loop
read_done:

	xor r12, r12
dump_loop:
	cmp r12, rbx
	jae dump_done

	mov r13, rbx
	sub r13, r12
	cmp r13, 16
	jbe 1f
	mov r13, 16
1:
	lea r15, [rip + input]
	add r15, r12
	lea r8, [rip + hexdigits]
	lea r14, [rip + linebuf]

	mov eax, r12d
	mov ecx, 8
off_hex:
	rol eax, 4
	mov edx, eax
	and edx, 0xf
	mov dl, byte ptr [r8 + rdx]
	mov byte ptr [r14], dl
	inc r14
	dec ecx
	jnz off_hex

	mov byte ptr [r14], ' '
	mov byte ptr [r14 + 1], ' '
	add r14, 2

	xor ecx, ecx
byte_hex_loop:
	cmp rcx, r13
	jae byte_empty
	movzx eax, byte ptr [r15 + rcx]
	mov edx, eax
	shr edx, 4
	mov dl, byte ptr [r8 + rdx]
	mov byte ptr [r14], dl
	mov edx, eax
	and edx, 0xf
	mov dl, byte ptr [r8 + rdx]
	mov byte ptr [r14 + 1], dl
	mov byte ptr [r14 + 2], ' '
	add r14, 3
	jmp byte_after
byte_empty:
	mov byte ptr [r14], ' '
	mov byte ptr [r14 + 1], ' '
	mov byte ptr [r14 + 2], ' '
	add r14, 3
byte_after:
	cmp ecx, 7
	jne 2f
	mov byte ptr [r14], ' '
	inc r14
2:
	inc ecx
	cmp ecx, 16
	jb byte_hex_loop

	mov byte ptr [r14], ' '
	inc r14
	mov byte ptr [r14], '|'
	inc r14

	xor ecx, ecx
ascii_loop:
	cmp rcx, r13
	jae ascii_done
	movzx eax, byte ptr [r15 + rcx]
	cmp al, 0x20
	jb is_dot
	cmp al, 0x7e
	ja is_dot
	mov byte ptr [r14], al
	jmp ascii_next
is_dot:
	mov byte ptr [r14], '.'
ascii_next:
	inc r14
	inc ecx
	jmp ascii_loop
ascii_done:

	mov byte ptr [r14], '|'
	inc r14
	mov byte ptr [r14], 10
	inc r14

	lea rsi, [rip + linebuf]
	mov rdx, r14
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall

	add r12, r13
	jmp dump_loop

dump_done:
	lea r8, [rip + hexdigits]
	lea r14, [rip + linebuf]
	mov eax, ebx
	mov ecx, 8
final_hex:
	rol eax, 4
	mov edx, eax
	and edx, 0xf
	mov dl, byte ptr [r8 + rdx]
	mov byte ptr [r14], dl
	inc r14
	dec ecx
	jnz final_hex
	mov byte ptr [r14], 10
	inc r14

	lea rsi, [rip + linebuf]
	mov rdx, r14
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall

	mov eax, 60
	xor edi, edi
	syscall
