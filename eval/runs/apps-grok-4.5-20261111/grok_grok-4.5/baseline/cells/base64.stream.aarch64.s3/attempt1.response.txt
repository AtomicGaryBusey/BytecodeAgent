.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INMAX, 49152

.text
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #INMAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    mov x23, #0
    adrp x25, alphabet
    add x25, x25, :lo12:alphabet
    mov x24, #0

encode_loop:
    sub x0, x20, x24
    cmp x0, #3
    b.lt encode_tail

    ldrb w1, [x19, x24]
    add x24, x24, #1
    ldrb w2, [x19, x24]
    add x24, x24, #1
    ldrb w3, [x19, x24]
    add x24, x24, #1

    lsr w4, w1, #2
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w1, #3
    lsl w4, w4, #4
    lsr w5, w2, #4
    orr w4, w4, w5
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w2, #15
    lsl w4, w4, #2
    lsr w5, w3, #6
    orr w4, w4, w5
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w3, #63
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    b encode_loop

encode_tail:
    cmp x0, #0
    beq encode_done
    cmp x0, #1
    beq encode_one

    ldrb w1, [x19, x24]
    add x24, x24, #1
    ldrb w2, [x19, x24]

    lsr w4, w1, #2
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w1, #3
    lsl w4, w4, #4
    lsr w5, w2, #4
    orr w4, w4, w5
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w2, #15
    lsl w4, w4, #2
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    mov w5, #'='
    strb w5, [x22, x23]
    add x23, x23, #1
    b encode_done

encode_one:
    ldrb w1, [x19, x24]

    lsr w4, w1, #2
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    and w4, w1, #3
    lsl w4, w4, #4
    ldrb w5, [x25, x4]
    strb w5, [x22, x23]
    add x23, x23, #1

    mov w5, #'='
    strb w5, [x22, x23]
    add x23, x23, #1
    strb w5, [x22, x23]
    add x23, x23, #1

encode_done:
    adrp x26, nl
    add x26, x26, :lo12:nl
    mov x24, #0

write_loop:
    cmp x24, x23
    b.hs write_done

    sub x2, x23, x24
    cmp x2, #76
    b.ls 1f
    mov x2, #76
1:
    mov x0, #1
    add x1, x22, x24
    mov x8, #SYS_write
    svc #0
    add x24, x24, x2

    mov x0, #1
    mov x1, x26
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b write_loop

write_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
.align 4

    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .byte 0x0A

.bss
.align 4

    .space 49152

    .space 66500
