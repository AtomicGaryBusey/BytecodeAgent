.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.text
_start:
    mov x19, #0
    adrp x20, buffer
    add x20, x20, :lo12:buffer

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    mov x23, #0
    mov x0, x20
    mov x1, x19

count_loop:
    cbz x1, count_done
    ldrb w2, [x0], #1
    sub x1, x1, #1
    cmp w2, #0x0A
    b.eq count_nl
    cmp w2, #0x20
    b.eq count_ws
    cmp w2, #0x09
    b.eq count_ws
    cmp w2, #0x0B
    b.eq count_ws
    cmp w2, #0x0C
    b.eq count_ws
    cmp w2, #0x0D
    b.eq count_ws
    cbnz x23, count_loop
    add x22, x22, #1
    mov x23, #1
    b count_loop

count_nl:
    add x21, x21, #1
count_ws:
    mov x23, #0
    b count_loop

count_done:
    mov x0, x21
    bl print_num
    bl print_space
    mov x0, x22
    bl print_num
    bl print_space
    mov x0, x19
    bl print_num
    bl print_newline
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

print_num:
    sub sp, sp, #32
    add x1, sp, #32
    mov x2, x0
    mov x3, #10
    cbnz x2, 1f
    mov w4, #'0'
    strb w4, [x1, #-1]!
    b 2f
1:
    udiv x4, x2, x3
    msub x5, x4, x3, x2
    add w5, w5, #'0'
    strb w5, [x1, #-1]!
    mov x2, x4
    cbnz x2, 1b
2:
    add x2, sp, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    add sp, sp, #32
    ret

print_space:
    mov x0, #1
    adrp x1, space_char
    add x1, x1, :lo12:space_char
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    ret

print_newline:
    mov x0, #1
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    ret

.section .rodata
space_char:
    .byte 0x20
newline_char:
    .byte 0x0A

.bss
.align 4
buffer:
    .space BUF_SIZE
