.globl _start

.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NEST, 1000

.bss
.align 16
input_buf:
.zero BUF_SIZE
tape:
.zero TAPE_SIZE
jump_table:
.zero 524288
bracket_stack:
.zero 8000

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, BUF_SIZE
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    xor rcx, rcx
find_bang:
    cmp rcx, r12
    je no_bang
    cmp byte ptr [r13 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r14, rcx
    lea r15, [r13 + rcx + 1]
    mov rbp, r12
    sub rbp, rcx
    dec rbp
    jmp bang_done
no_bang:
    mov r14, r12
    lea r15, [r13 + r12]
    xor rbp, rbp
bang_done:

    lea r10, [rip + jump_table]
    xor r8, r8
    lea r9, [rip + bracket_stack]
    xor rcx, rcx
validate_loop:
    cmp rcx, r14
    je validate_end
    movzx eax, byte ptr [r13 + rcx]
    cmp al, '['
    je do_open
    cmp al, ']'
    je do_close
    inc rcx
    jmp validate_loop
do_open:
    cmp r8, MAX_NEST
    jae unbalanced
    mov qword ptr [r9 + r8*8], rcx
    inc r8
    inc rcx
    jmp validate_loop
do_close:
    test r8, r8
    jz unbalanced
    dec r8
    mov rdx, qword ptr [r9 + r8*8]
    mov qword ptr [r10 + rdx*8], rcx
    mov qword ptr [r10 + rcx*8], rdx
    inc rcx
    jmp validate_loop
validate_end:
    test r8, r8
    jnz unbalanced

    lea r8, [r15 + rbp]
    lea rdi, [rip + tape]
    xor rbx, rbx
interp_loop:
    cmp rbx, r14
    jae interp_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    inc rbx
    jmp interp_loop
cmd_right:
    inc rdi
    inc rbx
    jmp interp_loop
cmd_left:
    dec rdi
    inc rbx
    jmp interp_loop
cmd_plus:
    inc byte ptr [rdi]
    inc rbx
    jmp interp_loop
cmd_minus:
    dec byte ptr [rdi]
    inc rbx
    jmp interp_loop
cmd_dot:
    push rdi
    mov rsi, rdi
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    pop rdi
    inc rbx
    jmp interp_loop
cmd_comma:
    cmp r15, r8
    jae comma_eof
    mov al, byte ptr [r15]
    mov byte ptr [rdi], al
    inc r15
    inc rbx
    jmp interp_loop
comma_eof:
    mov byte ptr [rdi], 0
    inc rbx
    jmp interp_loop
cmd_open:
    cmp byte ptr [rdi], 0
    jne open_cont
    mov rbx, qword ptr [r10 + rbx*8]
open_cont:
    inc rbx
    jmp interp_loop
cmd_close:
    cmp byte ptr [rdi], 0
    je close_cont
    mov rbx, qword ptr [r10 + rbx*8]
close_cont:
    inc rbx
    jmp interp_loop
interp_done:
    mov rax, 60
    xor rdi, rdi
    syscall
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
