.globl _start

.equ BUFSIZE, 32768
.equ MAXLINES, 1000

.text

_start:
	xor	rbx, rbx
read_loop:
	mov	rdx, BUFSIZE
	sub	rdx, rbx
	jz	read_done
	lea	rsi, [rip + buffer]
	add	rsi, rbx
	xor	eax, eax
	xor	edi, edi
	syscall
	test	rax, rax
	jle	read_done
	add	rbx, rax
	jmp	read_loop
read_done:
	lea	rsi, [rip + buffer]
	mov	rdi, rsi
	add	rdi, rbx
	xor	ecx, ecx
	lea	r8, [rip + lines]
parse_loop:
	cmp	rsi, rdi
	jae	parse_done
	mov	r9, rsi
find_nl:
	cmp	rsi, rdi
	jae	found_end
	cmp	byte ptr [rsi], 10
	je	found_nl
	inc	rsi
	jmp	find_nl
found_nl:
	mov	r10, rsi
	sub	r10, r9
	cmp	rcx, MAXLINES
	jae	parse_done
	mov	qword ptr [r8], r9
	mov	qword ptr [r8 + 8], r10
	add	r8, 16
	inc	rcx
	inc	rsi
	jmp	parse_loop
found_end:
	mov	r10, rdi
	sub	r10, r9
	cmp	rcx, MAXLINES
	jae	parse_done
	mov	qword ptr [r8], r9
	mov	qword ptr [r8 + 8], r10
	add	r8, 16
	inc	rcx
parse_done:
	mov	r15, rcx
	cmp	r15, 1
	jbe	sort_done
	xor	r12, r12
bubble_outer:
	mov	rax, r15
	dec	rax
	cmp	r12, rax
	jae	sort_done
	xor	r13, r13
	mov	r14, rax
	sub	r14, r12
bubble_inner:
	cmp	r13, r14
	jae	next_i
	lea	rdi, [rip + lines]
	mov	rax, r13
	shl	rax, 4
	add	rdi, rax
	lea	rsi, [rdi + 16]
	call	cmp_lines
	test	rax, rax
	jle	no_swap
	movdqu	xmm0, xmmword ptr [rdi]
	movdqu	xmm1, xmmword ptr [rsi]
	movdqu	xmmword ptr [rdi], xmm1
	movdqu	xmmword ptr [rsi], xmm0
no_swap:
	inc	r13
	jmp	bubble_inner
next_i:
	inc	r12
	jmp	bubble_outer
sort_done:
	xor	r12, r12
	lea	r14, [rip + lines]
print_loop:
	cmp	r12, r15
	jae	print_done
	mov	rax, r12
	shl	rax, 4
	mov	rsi, qword ptr [r14 + rax]
	mov	rdx, qword ptr [r14 + rax + 8]
	mov	eax, 1
	mov	edi, 1
	syscall
	mov	eax, 1
	mov	edi, 1
	lea	rsi, [rip + nl]
	mov	edx, 1
	syscall
	inc	r12
	jmp	print_loop
print_done:
	mov	eax, 60
	xor	edi, edi
	syscall

cmp_lines:
	push	rbx
	push	rcx
	push	rdx
	push	rsi
	push	rdi
	push	r8
	push	r9
	push	r10
	push	r11
	mov	r8, qword ptr [rdi]
	mov	r9, qword ptr [rdi + 8]
	mov	r10, qword ptr [rsi]
	mov	r11, qword ptr [rsi + 8]
	mov	rax, r9
	cmp	rax, r11
	cmova	rax, r11
	xor	edx, edx
cmp_loop:
	cmp	rdx, rax
	jae	cmp_len
	movzx	ecx, byte ptr [r8 + rdx]
	movzx	ebx, byte ptr [r10 + rdx]
	cmp	ecx, ebx
	jne	cmp_diff
	inc	rdx
	jmp	cmp_loop
cmp_diff:
	sub	ecx, ebx
	movsx	rax, ecx
	jmp	cmp_ret
cmp_len:
	mov	rax, r9
	sub	rax, r11
cmp_ret:
	pop	r11
	pop	r10
	pop	r9
	pop	r8
	pop	rdi
	pop	rsi
	pop	rdx
	pop	rcx
	pop	rbx
	ret

.section .rodata
nl:
	.byte 10

.bss
.align 8
buffer:
	.space BUFSIZE
.align 8
lines:
	.space MAXLINES * 16
