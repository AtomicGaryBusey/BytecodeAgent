.globl _start

.equ BUFSIZE, 32768
.equ MAXLINES, 1000
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, x19
    mov     x21, #BUFSIZE
read_loop:
    cbz     x21, read_done
    mov     x0, #0
    mov     x1, x20
    mov     x2, x21
    mov     x8, #SYS_READ
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    sub     x21, x21, x0
    b       read_loop
read_done:
    adrp    x24, lines
    add     x24, x24, :lo12:lines
    mov     x22, #0
    mov     x23, x19
parse_loop:
    cmp     x23, x20
    b.hs    parse_done
    mov     x25, x23
find_loop:
    cmp     x25, x20
    b.hs    last_line
    ldrb    w0, [x25]
    cmp     w0, #10
    b.eq    found_nl
    add     x25, x25, #1
    b       find_loop
found_nl:
    add     x0, x24, x22, lsl #4
    str     x23, [x0]
    sub     x1, x25, x23
    str     x1, [x0, #8]
    add     x22, x22, #1
    add     x23, x25, #1
    b       parse_loop
last_line:
    cmp     x23, x20
    b.hs    parse_done
    add     x0, x24, x22, lsl #4
    str     x23, [x0]
    sub     x1, x20, x23
    str     x1, [x0, #8]
    add     x22, x22, #1
parse_done:
    mov     x25, #1
sort_outer:
    cmp     x25, x22
    b.ge    sort_done
    add     x0, x24, x25, lsl #4
    ldp     x26, x27, [x0]
    mov     x28, x25
    sub     x28, x28, #1
sort_inner:
    cmp     x28, #0
    b.lt    sort_insert
    add     x4, x24, x28, lsl #4
    ldr     x0, [x4]
    ldr     x1, [x4, #8]
    mov     x2, x26
    mov     x3, x27
    bl      compare
    cmp     x0, #0
    b.le    sort_insert
    add     x4, x24, x28, lsl #4
    ldp     x0, x1, [x4]
    stp     x0, x1, [x4, #16]
    sub     x28, x28, #1
    b       sort_inner
sort_insert:
    add     x28, x28, #1
    add     x4, x24, x28, lsl #4
    stp     x26, x27, [x4]
    add     x25, x25, #1
    b       sort_outer
sort_done:
    mov     x25, #0
print_loop:
    cmp     x25, x22
    b.ge    do_exit
    add     x4, x24, x25, lsl #4
    ldr     x1, [x4]
    ldr     x2, [x4, #8]
    cbz     x2, do_nl
    mov     x0, #1
    mov     x8, #SYS_WRITE
    svc     #0
do_nl:
    mov     x0, #1
    adrp    x1, nl
    add     x1, x1, :lo12:nl
    mov     x2, #1
    mov     x8, #SYS_WRITE
    svc     #0
    add     x25, x25, #1
    b       print_loop
do_exit:
    mov     x0, #0
    mov     x8, #SYS_EXIT
    svc     #0

compare:
    cmp     x1, x3
    csel    x4, x1, x3, lo
    mov     x5, #0
cmp_loop:
    cmp     x5, x4
    b.hs    lengths
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    cmp     w6, w7
    b.ne    differ
    add     x5, x5, #1
    b       cmp_loop
differ:
    sub     x0, x6, x7
    ret
lengths:
    sub     x0, x1, x3
    ret

.section .rodata
nl:
    .byte   10

.bss
.align 3
input_buf:
    .space  BUFSIZE
.align 3
lines:
    .space  MAXLINES * 16
