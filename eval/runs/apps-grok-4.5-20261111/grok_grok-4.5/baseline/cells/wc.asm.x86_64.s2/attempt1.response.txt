.globl _start

.equ BUF_SIZE, 65536

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + buffer]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jle done_read
    add r14, rax
    mov rcx, rax
    mov rsi, rbx

byte_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne check_ws
    inc r12
check_ws:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb not_whitespace
    cmp al, 13
    ja not_whitespace
whitespace:
    xor r15, r15
    jmp next_byte
not_whitespace:
    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
next_byte:
    inc rsi
    dec rcx
    jnz byte_loop
    jmp read_loop

done_read:
    lea r8, [rip + outbuf + 64]
    mov rdi, r8
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call u64toa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call u64toa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call u64toa
    mov rsi, rdi
    mov rdx, r8
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

u64toa:
    mov ecx, 10
    test rax, rax
    jnz 1f
    dec rdi
    mov byte ptr [rdi], 48
    ret
1:
    xor edx, edx
    div rcx
    add edx, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.align 16
buffer:
    .zero BUF_SIZE
outbuf:
    .zero 64
