.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 4096

.globl _start

.text
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #BUFSIZE
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le done_reading
    mov x24, x0
    adrp x25, buf
    add x25, x25, :lo12:buf

process_loop:
    cbz x24, read_loop
    ldrb w26, [x25], #1
    sub x24, x24, #1
    add x21, x21, #1
    cmp w26, #10
    bne 1f
    add x19, x19, #1
1:
    cmp w26, #0x20
    b.eq is_ws
    cmp w26, #0x09
    b.lo not_ws
    cmp w26, #0x0D
    b.hi not_ws
is_ws:
    cbz x22, process_loop
    add x20, x20, #1
    mov x22, #0
    b process_loop
not_ws:
    mov x22, #1
    b process_loop

done_reading:
    cbz x22, 1f
    add x20, x20, #1
1:
    mov x0, x19
    bl print_uint
    mov w0, #32
    bl print_char
    mov x0, x20
    bl print_uint
    mov w0, #32
    bl print_char
    mov x0, x21
    bl print_uint
    mov w0, #10
    bl print_char
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

print_char:
    stp x29, x30, [sp, #-16]!
    sub sp, sp, #16
    strb w0, [sp]
    mov x0, #1
    mov x1, sp
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add sp, sp, #16
    ldp x29, x30, [sp], #16
    ret

print_uint:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    add x1, sp, #32
    mov x3, x0
    mov x4, #0
    cbnz x3, dig_loop
    sub x1, x1, #1
    mov w2, #48
    strb w2, [x1]
    mov x4, #1
    b do_write
dig_loop:
    cbz x3, do_write
    mov x5, #10
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #48
    sub x1, x1, #1
    strb w7, [x1]
    add x4, x4, #1
    mov x3, x6
    b dig_loop
do_write:
    mov x0, #1
    mov x2, x4
    mov x8, #SYS_write
    svc #0
    mov sp, x29
    ldp x29, x30, [sp], #16
    ret

.bss
.align 4
buf:
    .space BUFSIZE
