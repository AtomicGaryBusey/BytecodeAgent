
	.globl	_start

	.equ	SYS_read, 63
	.equ	SYS_write, 64
	.equ	SYS_exit, 93
	.equ	BUF_SIZE, 4096

	.text
_start:
	adrp	x23, buf
	add	x23, x23, :lo12:buf

.Lread_loop:
	mov	x0, #0
	mov	x1, x23
	mov	x2, #BUF_SIZE
	mov	x8, #SYS_read
	svc	#0
	cbz	x0, .Lread_done
	add	x21, x21, x0
	mov	x1, x23
	add	x2, x1, x0

.Lbyte_loop:
	ldrb	w3, [x1], #1
	cmp	w3, #10
	b.ne	1f
	add	x19, x19, #1
1:
	cmp	w3, #32
	b.eq	.Lwhitespace
	cmp	w3, #9
	b.lo	.Lnot_whitespace
	cmp	w3, #13
	b.hi	.Lnot_whitespace
.Lwhitespace:
	cbz	x22, .Lcontinue
	add	x20, x20, #1
	mov	x22, #0
	b	.Lcontinue
.Lnot_whitespace:
	mov	x22, #1
.Lcontinue:
	cmp	x1, x2
	b.lo	.Lbyte_loop
	b	.Lread_loop

.Lread_done:
	cbz	x22, 1f
	add	x20, x20, #1
1:
	mov	x0, x19
	bl	print_uint
	bl	print_space
	mov	x0, x20
	bl	print_uint
	bl	print_space
	mov	x0, x21
	bl	print_uint
	bl	print_newline

	mov	x0, #0
	mov	x8, #SYS_exit
	svc	#0

print_uint:
	sub	sp, sp, #32
	add	x1, sp, #32
	mov	x2, x0
	cbnz	x2, .Lconv
	mov	w3, #'0'
	strb	w3, [x1, #-1]!
	b	.Ldo_write
.Lconv:
1:
	mov	x3, #10
	udiv	x4, x2, x3
	msub	x5, x4, x3, x2
	add	w5, w5, #'0'
	strb	w5, [x1, #-1]!
	mov	x2, x4
	cbnz	x2, 1b
.Ldo_write:
	mov	x0, #1
	add	x2, sp, #32
	sub	x2, x2, x1
	mov	x8, #SYS_write
	svc	#0
	add	sp, sp, #32
	ret

print_space:
	adrp	x1, space_char
	add	x1, x1, :lo12:space_char
	mov	x0, #1
	mov	x2, #1
	mov	x8, #SYS_write
	svc	#0
	ret

print_newline:
	adrp	x1, nl_char
	add	x1, x1, :lo12:nl_char
	mov	x0, #1
	mov	x2, #1
	mov	x8, #SYS_write
	svc	#0
	ret

	.section	.rodata
space_char:
	.byte	0x20
nl_char:
	.byte	0x0a

	.bss
buf:
	.space	BUF_SIZE
