.globl _start
.text
_start:
	lea r13, [rip + input_buf]
	xor r12, r12
.read_more:
	mov eax, 0
	mov edi, 0
	lea rsi, [r13 + r12]
	mov edx, 65536
	sub rdx, r12
	jz .read_done
	syscall
	test rax, rax
	jle .read_done
	add r12, rax
	jmp .read_more
.read_done:
	xor r14, r14
.line_loop:
	cmp r14, r12
	jae .finish
	mov r15, r14
.find_nl:
	cmp r15, r12
	jae .last_line
	cmp byte ptr [r13 + r15], 0x0A
	je .got_nl
	inc r15
	jmp .find_nl
.got_nl:
	lea rsi, [r13 + r14]
	mov rdx, r15
	add rdx, r13
	call process_line
	mov r14, r15
	inc r14
	jmp .line_loop
.last_line:
	cmp r14, r12
	jae .finish
	lea rsi, [r13 + r14]
	lea rdx, [r13 + r12]
	call process_line
.finish:
	mov eax, 60
	xor edi, edi
	syscall

process_line:
	push rbx
	lea rbx, [rip + rpn_stack]
	mov r8, rbx
.skip_sep:
	cmp rsi, rdx
	jae .tokens_done
	movzx eax, byte ptr [rsi]
	cmp al, 0x20
	je .do_skip
	cmp al, 0x09
	je .do_skip
	jmp .got_token
.do_skip:
	inc rsi
	jmp .skip_sep
.got_token:
	mov rdi, rsi
.find_tok_end:
	inc rdi
	cmp rdi, rdx
	jae .tok_end
	movzx eax, byte ptr [rdi]
	cmp al, 0x20
	je .tok_end
	cmp al, 0x09
	je .tok_end
	jmp .find_tok_end
.tok_end:
	mov rcx, rdi
	sub rcx, rsi
	# try number: optional '-' then 1..18 digits
	mov r9, rsi
	xor r10, r10
	movzx eax, byte ptr [rsi]
	cmp al, 0x2D
	jne .parse_pos
	# negative literal
	cmp rcx, 2
	jb .try_op
	cmp rcx, 19
	ja .try_op
	inc r9
	dec rcx
	jmp .parse_digits
.parse_pos:
	cmp al, 0x30
	jb .try_op
	cmp al, 0x39
	ja .try_op
	cmp rcx, 1
	jb .try_op
	cmp rcx, 18
	ja .try_op
.parse_digits:
	mov r11, rcx
.digit_loop:
	movzx eax, byte ptr [r9]
	cmp al, 0x30
	jb .try_op
	cmp al, 0x39
	ja .try_op
	sub al, 0x30
	imul r10, r10, 10
	movzx eax, al
	add r10, rax
	inc r9
	dec r11
	jnz .digit_loop
	# check if we had a leading '-'
	movzx eax, byte ptr [rsi]
	cmp al, 0x2D
	jne .push_num
	neg r10
.push_num:
	mov qword ptr [r8], r10
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.try_op:
	cmp rcx, 1
	jne .invalid
	movzx eax, byte ptr [rsi]
	cmp al, 0x2B
	je .do_add
	cmp al, 0x2D
	je .do_sub
	cmp al, 0x2A
	je .do_mul
	cmp al, 0x2F
	je .do_div
	cmp al, 0x25
	je .do_mod
	jmp .invalid
.do_add:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	add rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_sub:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	sub rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mul:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	imul rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_div:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 1
	shl r9, 63
	cmp rax, r9
	jne .div_ok
	cmp rcx, -1
	je .invalid
.div_ok:
	cqo
	idiv rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mod:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 1
	shl r9, 63
	cmp rax, r9
	jne .mod_ok
	cmp rcx, -1
	je .invalid
.mod_ok:
	cqo
	idiv rcx
	mov qword ptr [r8], rdx
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.tokens_done:
	mov rcx, r8
	sub rcx, rbx
	cmp rcx, 8
	jne .invalid
	mov rax, qword ptr [r8 - 8]
	call print_number
	pop rbx
	ret
.invalid:
	lea rsi, [rip + error_msg]
	mov edx, 6
	mov edi, 1
	mov eax, 1
	syscall
	pop rbx
	ret

print_number:
	mov r9, 1
	shl r9, 63
	cmp rax, r9
	je .print_min
	lea r10, [rip + digit_buf + 24]
	xor r11, r11
	test rax, rax
	jns .abs_done
	neg rax
	mov r11, 1
.abs_done:
	test rax, rax
	jnz .not_zero
	mov byte ptr [r10], 0x30
	dec r10
	jmp .after_digits
.not_zero:
	mov ecx, 10
.dig_loop:
	xor rdx, rdx
	div rcx
	add dl, 0x30
	mov byte ptr [r10], dl
	dec r10
	test rax, rax
	jnz .dig_loop
.after_digits:
	test r11, r11
	jz .no_sign
	mov byte ptr [r10], 0x2D
	dec r10
.no_sign:
	inc r10
	mov byte ptr [rip + digit_buf + 25], 0x0A
	lea rdx, [rip + digit_buf + 26]
	sub rdx, r10
	mov rsi, r10
	mov edi, 1
	mov eax, 1
	syscall
	ret
.print_min:
	lea rsi, [rip + min_msg]
	mov edx, 21
	mov edi, 1
	mov eax, 1
	syscall
	ret

.section .rodata
error_msg:
	.ascii "error\n"
min_msg:
	.ascii "-9223372036854775808\n"

.bss
.align 8
input_buf:
	.space 65536
rpn_stack:
	.space 8192
digit_buf:
	.space 32
