.globl _start
.text
_start:
	lea r13, [rip + input_buf]
	xor r12, r12
.read_more:
	mov rax, 0
	mov rdi, 0
	lea rsi, [r13 + r12]
	mov rdx, 65536
	sub rdx, r12
	jz .read_done
	syscall
	test rax, rax
	jle .read_done
	add r12, rax
	jmp .read_more
.read_done:
	xor r14, r14
.line_loop:
	cmp r14, r12
	jae .finish
	mov r15, r14
.find_nl_loop:
	cmp r15, r12
	jae .last_line
	cmp byte ptr [r13 + r15], 0x0A
	je .got_nl
	inc r15
	jmp .find_nl_loop
.got_nl:
	lea rsi, [r13 + r14]
	lea rdx, [r13 + r15]
	call process_line
	mov r14, r15
	inc r14
	jmp .line_loop
.last_line:
	lea rsi, [r13 + r14]
	lea rdx, [r13 + r12]
	call process_line
.finish:
	mov rax, 60
	xor rdi, rdi
	syscall

process_line:
	lea r8, [rip + rpn_stack]
.skip_sep:
	cmp rsi, rdx
	jae .tokens_done
	movzx eax, byte ptr [rsi]
	cmp al, 0x20
	je .do_skip
	cmp al, 0x09
	je .do_skip
	jmp .got_token
.do_skip:
	inc rsi
	jmp .skip_sep
.got_token:
	mov rdi, rsi
.find_end:
	inc rdi
	cmp rdi, rdx
	jae .end_tok
	movzx eax, byte ptr [rdi]
	cmp al, 0x20
	je .end_tok
	cmp al, 0x09
	je .end_tok
	jmp .find_end
.end_tok:
	mov rcx, rdi
	sub rcx, rsi
	cmp rcx, 1
	jne .check_number
	movzx eax, byte ptr [rsi]
	cmp al, '+'
	je .do_add
	cmp al, '-'
	je .do_sub
	cmp al, '*'
	je .do_mul
	cmp al, '/'
	je .do_div
	cmp al, '%'
	je .do_mod
.check_number:
	movzx eax, byte ptr [rsi]
	cmp al, '-'
	je .num_neg
	cmp al, '0'
	jb .invalid
	cmp al, '9'
	ja .invalid
	cmp rcx, 18
	ja .invalid
	xor r10, r10
	mov r11, rsi
.num_pos_loop:
	movzx eax, byte ptr [r11]
	cmp al, '0'
	jb .invalid
	cmp al, '9'
	ja .invalid
	sub eax, '0'
	imul r10, r10, 10
	add r10, rax
	inc r11
	cmp r11, rdi
	jb .num_pos_loop
	mov qword ptr [r8], r10
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.num_neg:
	cmp rcx, 2
	jb .invalid
	cmp rcx, 19
	ja .invalid
	xor r10, r10
	mov r11, rsi
	inc r11
.num_neg_loop:
	movzx eax, byte ptr [r11]
	cmp al, '0'
	jb .invalid
	cmp al, '9'
	ja .invalid
	sub eax, '0'
	imul r10, r10, 10
	add r10, rax
	inc r11
	cmp r11, rdi
	jb .num_neg_loop
	neg r10
	mov qword ptr [r8], r10
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_add:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	add rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_sub:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	sub rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mul:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	imul rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_div:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 0x8000000000000000
	cmp rax, r9
	jne .div_do
	cmp rcx, -1
	je .invalid
.div_do:
	cqo
	idiv rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mod:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 0x8000000000000000
	cmp rax, r9
	jne .mod_do
	cmp rcx, -1
	je .invalid
.mod_do:
	cqo
	idiv rcx
	mov qword ptr [r8], rdx
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.tokens_done:
	lea r9, [rip + rpn_stack]
	mov rcx, r8
	sub rcx, r9
	cmp rcx, 8
	jne .invalid
	mov rax, qword ptr [r8 - 8]
	call print_number
	ret
.invalid:
	lea rsi, [rip + error_msg]
	mov edx, 6
	mov edi, 1
	mov eax, 1
	syscall
	ret

print_number:
	mov r11, 0x8000000000000000
	cmp rax, r11
	je .print_int_min
	lea r10, [rip + digit_buf + 20]
	mov r9, 0
	test rax, rax
	jns .abs_done
	neg rax
	mov r9, 1
.abs_done:
	test rax, rax
	jnz .not_zero
	mov byte ptr [r10], '0'
	dec r10
	jmp .after_digits
.not_zero:
	mov rcx, 10
.digit_loop:
	xor rdx, rdx
	div rcx
	add dl, '0'
	mov byte ptr [r10], dl
	dec r10
	test rax, rax
	jnz .digit_loop
.after_digits:
	cmp r9, 1
	jne .no_minus
	mov byte ptr [r10], '-'
	dec r10
.no_minus:
	mov byte ptr [rip + digit_buf + 21], 0x0A
	inc r10
	lea rdx, [rip + digit_buf + 22]
	sub rdx, r10
	mov rsi, r10
	mov rdi, 1
	mov rax, 1
	syscall
	ret
.print_int_min:
	lea rsi, [rip + min_msg]
	mov edx, 21
	mov edi, 1
	mov eax, 1
	syscall
	ret

.section .rodata
error_msg:
	.ascii "error\n"
min_msg:
	.ascii "-9223372036854775808\n"

.bss
.align 8
input_buf:
	.space 65536
rpn_stack:
	.space 8192
digit_buf:
	.space 32
