.globl _start
.text
_start:
	lea r13, [rip + input_buf]
	xor r12, r12
.read_loop:
	mov eax, 0
	mov edi, 0
	lea rsi, [r13 + r12]
	mov edx, 65536
	sub rdx, r12
	jz .read_done
	syscall
	test rax, rax
	jle .read_done
	add r12, rax
	jmp .read_loop
.read_done:
	xor r14, r14
.line_loop:
	cmp r14, r12
	jae .exit
	mov r15, r14
.find_nl:
	cmp r15, r12
	jae .last_line
	cmp byte ptr [r13 + r15], 0x0A
	je .got_nl
	inc r15
	jmp .find_nl
.got_nl:
	lea rsi, [r13 + r14]
	lea rdx, [r13 + r15]
	call process_line
	mov r14, r15
	inc r14
	jmp .line_loop
.last_line:
	cmp r14, r12
	jae .exit
	lea rsi, [r13 + r14]
	lea rdx, [r13 + r12]
	call process_line
.exit:
	mov eax, 60
	xor edi, edi
	syscall

process_line:
	push rbx
	push r12
	push r13
	push r14
	push r15
	mov r13, rdx
	lea rbx, [rip + rpn_stack]
	mov r8, rbx
.skip_sep:
	cmp rsi, r13
	jae .tokens_done
	movzx eax, byte ptr [rsi]
	cmp al, 0x20
	je .do_skip
	cmp al, 0x09
	je .do_skip
	jmp .got_token
.do_skip:
	inc rsi
	jmp .skip_sep
.got_token:
	mov rdi, rsi
.find_end:
	inc rdi
	cmp rdi, r13
	jae .tok_end
	movzx eax, byte ptr [rdi]
	cmp al, 0x20
	je .tok_end
	cmp al, 0x09
	je .tok_end
	jmp .find_end
.tok_end:
	mov rcx, rdi
	sub rcx, rsi
	cmp rcx, 1
	jne .try_number
	movzx eax, byte ptr [rsi]
	cmp al, 0x2B
	je .do_add
	cmp al, 0x2D
	je .do_sub
	cmp al, 0x2A
	je .do_mul
	cmp al, 0x2F
	je .do_div
	cmp al, 0x25
	je .do_mod
.try_number:
	xor r10, r10
	mov r9, rsi
	mov r11, rcx
	xor r12, r12
	movzx eax, byte ptr [rsi]
	cmp al, 0x2D
	jne .check_pos
	cmp rcx, 2
	jb .invalid
	cmp rcx, 19
	ja .invalid
	inc r9
	dec r11
	mov r12, 1
	jmp .digit_loop
.check_pos:
	cmp al, 0x30
	jb .invalid
	cmp al, 0x39
	ja .invalid
	cmp rcx, 18
	ja .invalid
.digit_loop:
	movzx eax, byte ptr [r9]
	cmp al, 0x30
	jb .invalid
	cmp al, 0x39
	ja .invalid
	sub eax, 0x30
	lea r10, [r10 + r10*4]
	add r10, r10
	add r10, rax
	inc r9
	dec r11
	jnz .digit_loop
	test r12, r12
	jz .push_num
	neg r10
.push_num:
	mov qword ptr [r8], r10
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_add:
	mov rax, r8
	sub rax, rbx
	cmp rax, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	add rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_sub:
	mov rax, r8
	sub rax, rbx
	cmp rax, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	sub rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mul:
	mov rax, r8
	sub rax, rbx
	cmp rax, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	imul rax, rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_div:
	mov rax, r8
	sub rax, rbx
	cmp rax, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 0x8000000000000000
	cmp rax, r9
	jne .div_ok
	cmp rcx, -1
	je .invalid
.div_ok:
	cqo
	idiv rcx
	mov qword ptr [r8], rax
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.do_mod:
	mov rax, r8
	sub rax, rbx
	cmp rax, 16
	jb .invalid
	sub r8, 8
	mov rcx, qword ptr [r8]
	sub r8, 8
	mov rax, qword ptr [r8]
	test rcx, rcx
	jz .invalid
	mov r9, 0x8000000000000000
	cmp rax, r9
	jne .mod_ok
	cmp rcx, -1
	je .invalid
.mod_ok:
	cqo
	idiv rcx
	mov qword ptr [r8], rdx
	add r8, 8
	mov rsi, rdi
	jmp .skip_sep
.tokens_done:
	mov rax, r8
	sub rax, rbx
	cmp rax, 8
	jne .invalid
	mov rax, qword ptr [r8 - 8]
	call print_number
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbx
	ret
.invalid:
	lea rsi, [rip + error_msg]
	mov edx, 6
	mov edi, 1
	mov eax, 1
	syscall
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbx
	ret

print_number:
	mov r9, 0x8000000000000000
	cmp rax, r9
	je .print_min
	lea r10, [rip + digit_buf + 24]
	xor r11, r11
	test rax, rax
	jns .abs_done
	neg rax
	mov r11, 1
.abs_done:
	test rax, rax
	jnz .not_zero
	mov byte ptr [r10], 0x30
	dec r10
	jmp .after_digits
.not_zero:
	mov ecx, 10
.dig_loop:
	xor rdx, rdx
	div rcx
	add dl, 0x30
	mov byte ptr [r10], dl
	dec r10
	test rax, rax
	jnz .dig_loop
.after_digits:
	test r11, r11
	jz .no_sign
	mov byte ptr [r10], 0x2D
	dec r10
.no_sign:
	inc r10
	mov byte ptr [rip + digit_buf + 25], 0x0A
	lea rdx, [rip + digit_buf + 26]
	sub rdx, r10
	mov rsi, r10
	mov edi, 1
	mov eax, 1
	syscall
	ret
.print_min:
	lea rsi, [rip + min_msg]
	mov edx, 21
	mov edi, 1
	mov eax, 1
	syscall
	ret

.section .rodata
error_msg:
	.ascii "error\n"
min_msg:
	.ascii "-9223372036854775808\n"

.bss
.align 8
input_buf:
	.space 65536
rpn_stack:
	.space 8192
digit_buf:
	.space 32
