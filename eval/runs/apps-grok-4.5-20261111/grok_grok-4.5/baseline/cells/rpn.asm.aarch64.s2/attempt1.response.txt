.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536
.equ STACK_MAX, 512

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x19
    add x27, x19, x20

line_loop:
    cmp x21, x27
    b.ge all_done
    mov x22, x21

find_nl:
    cmp x21, x27
    b.ge do_final_line
    ldrb w0, [x21], #1
    cmp w0, #10
    b.ne find_nl
    sub x23, x21, #1
    bl eval_line
    b line_loop

do_final_line:
    mov x23, x21
    bl eval_line

all_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

eval_line:
    stp x29, x30, [sp, #-16]!
    mov x25, #0
    mov x24, x22

el_parse:
el_skip:
    cmp x24, x23
    b.ge el_parse_done
    ldrb w0, [x24]
    cmp w0, #32
    b.eq el_skip_inc
    cmp w0, #9
    b.eq el_skip_inc
    b el_token
el_skip_inc:
    add x24, x24, #1
    b el_skip

el_token:
    mov x26, x24
el_tok_end:
    add x24, x24, #1
    cmp x24, x23
    b.ge el_got_tok
    ldrb w0, [x24]
    cmp w0, #32
    b.eq el_got_tok
    cmp w0, #9
    b.eq el_got_tok
    b el_tok_end

el_got_tok:
    sub x9, x24, x26
    cmp x9, #1
    b.ne el_check_num
    ldrb w0, [x26]
    cmp w0, #'+'
    b.eq el_op_add
    cmp w0, #'-'
    b.eq el_op_sub
    cmp w0, #'*'
    b.eq el_op_mul
    cmp w0, #'/'
    b.eq el_op_div
    cmp w0, #'%'
    b.eq el_op_mod

el_check_num:
    mov x10, x26
    mov x11, #0
    ldrb w0, [x10]
    cmp w0, #'-'
    b.ne el_num_digits
    add x10, x10, #1
    cmp x10, x24
    b.eq el_error
    mov x11, #1

el_num_digits:
    cmp x10, x24
    b.eq el_error
    mov x12, #0
    mov x13, #0

el_dig_loop:
    cmp x10, x24
    b.ge el_dig_done
    ldrb w0, [x10]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi el_error
    add x13, x13, #1
    cmp x13, #19
    b.ge el_error
    mov x14, #10
    mul x12, x12, x14
    add x12, x12, x1
    add x10, x10, #1
    b el_dig_loop

el_dig_done:
    cbz x11, el_push
    neg x12, x12
el_push:
    cmp x25, #STACK_MAX
    b.ge el_error
    adrp x0, rpn_stack
    add x0, x0, :lo12:rpn_stack
    str x12, [x0, x25, lsl #3]
    add x25, x25, #1
    b el_parse

el_op_add:
    bl pop2
    cbnz x0, el_error
    add x1, x1, x2
    b el_push_result

el_op_sub:
    bl pop2
    cbnz x0, el_error
    sub x1, x1, x2
    b el_push_result

el_op_mul:
    bl pop2
    cbnz x0, el_error
    mul x1, x1, x2
    b el_push_result

el_op_div:
    bl pop2
    cbnz x0, el_error
    cbz x2, el_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.ne el_div_ok
    cmp x2, #-1
    b.eq el_error
el_div_ok:
    sdiv x1, x1, x2
    b el_push_result

el_op_mod:
    bl pop2
    cbnz x0, el_error
    cbz x2, el_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.ne el_mod_ok
    cmp x2, #-1
    b.eq el_error
el_mod_ok:
    sdiv x3, x1, x2
    msub x1, x3, x2, x1
    b el_push_result

el_push_result:
    adrp x0, rpn_stack
    add x0, x0, :lo12:rpn_stack
    str x1, [x0, x25, lsl #3]
    add x25, x25, #1
    b el_parse

el_parse_done:
    cmp x25, #1
    b.ne el_error
    adrp x0, rpn_stack
    add x0, x0, :lo12:rpn_stack
    ldr x0, [x0]
    bl print_num
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b el_ret

el_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #6
    mov x8, #SYS_write
    svc #0

el_ret:
    ldp x29, x30, [sp], #16
    ret

pop2:
    cmp x25, #2
    b.lt pop2_fail
    adrp x0, rpn_stack
    add x0, x0, :lo12:rpn_stack
    sub x25, x25, #1
    ldr x2, [x0, x25, lsl #3]
    sub x25, x25, #1
    ldr x1, [x0, x25, lsl #3]
    mov x0, #0
    ret
pop2_fail:
    mov x0, #1
    ret

print_num:
    stp x29, x30, [sp, #-64]!
    add x7, sp, #48
    mov x9, x7
    mov w4, #0
    cmp x0, #0
    b.ge 1f
    mov w4, #1
    mov x3, #1
    lsl x3, x3, #63
    cmp x0, x3
    b.eq 1f
    neg x0, x0
1:
    cbnz x0, 2f
    mov w5, #'0'
    sub x7, x7, #1
    strb w5, [x7]
    b 3f
2:
    mov x2, #10
    udiv x6, x0, x2
    msub x5, x6, x2, x0
    add w5, w5, #'0'
    sub x7, x7, #1
    strb w5, [x7]
    mov x0, x6
    cbnz x0, 2b
3:
    cbz w4, 4f
    mov w5, #'-'
    sub x7, x7, #1
    strb w5, [x7]
4:
    mov x0, #1
    mov x1, x7
    sub x2, x9, x7
    mov x8, #SYS_write
    svc #0
    ldp x29, x30, [sp], #64
    ret

.section .rodata
err_msg:
    .ascii "error\n"
nl_msg:
    .ascii "\n"

.bss
.align 3
input_buf:
    .space 65536
.align 3
rpn_stack:
    .space 4096
