.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_MAX, 49152

.text
_start:
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf
	mov	x20, #0

read_loop:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #INPUT_MAX
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x8, #SYS_read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop

read_done:
	adrp	x22, output_buf
	add	x22, x22, :lo12:output_buf
	mov	x21, x22
	mov	x23, #0
	adrp	x24, alphabet
	add	x24, x24, :lo12:alphabet
	mov	x26, x19
	add	x25, x19, x20

encode_loop:
	sub	x0, x25, x26
	cmp	x0, #3
	b.lt	encode_tail
	ldrb	w9, [x26], #1
	ldrb	w10, [x26], #1
	ldrb	w11, [x26], #1
	lsr	w0, w9, #2
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w9, #3
	lsl	w0, w0, #4
	lsr	w1, w10, #4
	orr	w0, w0, w1
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w10, #15
	lsl	w0, w0, #2
	lsr	w1, w11, #6
	orr	w0, w0, w1
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w11, #63
	ldrb	w0, [x24, x0]
	bl	emit_char
	b	encode_loop

encode_tail:
	cmp	x0, #0
	b.eq	encode_finish
	cmp	x0, #1
	b.eq	encode_1
	ldrb	w9, [x26], #1
	ldrb	w10, [x26], #1
	lsr	w0, w9, #2
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w9, #3
	lsl	w0, w0, #4
	lsr	w1, w10, #4
	orr	w0, w0, w1
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w10, #15
	lsl	w0, w0, #2
	ldrb	w0, [x24, x0]
	bl	emit_char
	mov	w0, #61
	bl	emit_char
	b	encode_finish

encode_1:
	ldrb	w9, [x26], #1
	lsr	w0, w9, #2
	ldrb	w0, [x24, x0]
	bl	emit_char
	and	w0, w9, #3
	lsl	w0, w0, #4
	ldrb	w0, [x24, x0]
	bl	emit_char
	mov	w0, #61
	bl	emit_char
	mov	w0, #61
	bl	emit_char

encode_finish:
	cbz	x23, write_output
	mov	w0, #10
	strb	w0, [x21], #1

write_output:
	sub	x2, x21, x22
	cbz	x2, do_exit
	mov	x0, #1
	mov	x1, x22
	mov	x8, #SYS_write
	svc	#0

do_exit:
	mov	x0, #0
	mov	x8, #SYS_exit
	svc	#0

emit_char:
	strb	w0, [x21], #1
	add	x23, x23, #1
	cmp	x23, #76
	b.ne	1f
	mov	w1, #10
	strb	w1, [x21], #1
	mov	x23, #0
1:
	ret
