.text
.globl _start
_start:
    lea rbx, [rip + buffer]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz done_read
    lea rsi, [rbx + r12]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    xor r13, r13
process_loop:
    cmp r13, r12
    jae exit_ok
    mov r14, r13
find_nl_loop:
    cmp r14, r12
    jae no_nl_found
    cmp byte ptr [rbx + r14], 0x0A
    je nl_found
    inc r14
    jmp find_nl_loop
nl_found:
    mov r15, r14
    lea r8, [r14 + 1]
    jmp do_reverse
no_nl_found:
    mov r15, r12
    mov r8, r12
do_reverse:
    cmp r13, r15
    jae write_line
    mov rdi, r13
    lea rsi, [r15 - 1]
rev_loop:
    cmp rdi, rsi
    jae write_line
    mov al, byte ptr [rbx + rdi]
    mov dl, byte ptr [rbx + rsi]
    mov byte ptr [rbx + rdi], dl
    mov byte ptr [rbx + rsi], al
    inc rdi
    dec rsi
    jmp rev_loop
write_line:
    mov rdx, r15
    sub rdx, r13
    jz write_nl
    lea rsi, [rbx + r13]
    mov eax, 1
    mov edi, 1
    syscall
write_nl:
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    mov r13, r8
    jmp process_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 0x0A

.bss
.align 16
buffer:
    .zero 65536
