.globl _start
.text
_start:
    lea r14, [rip + input_buf]
    xor r15, r15
.read_more:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, 65536
    sub rdx, r15
    jz .read_done
    syscall
    test rax, rax
    jle .read_done
    add r15, rax
    jmp .read_more
.read_done:
    mov r13, r14
    add r13, r15
.line_loop:
    cmp r14, r13
    jae .done
    mov r12, r14
.find_nl:
    cmp r14, r13
    jae .last_line
    cmp byte ptr [r14], 10
    je .got_nl
    inc r14
    jmp .find_nl
.got_nl:
    mov r11, r14
    inc r14
    jmp .do_line
.last_line:
    mov r11, r13
.do_line:
    lea rbx, [rip + rpn_stack]
    xor r15, r15
    mov rsi, r12
.token_loop:
.skip_ws:
    cmp rsi, r11
    jae .tokens_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je .is_ws
    cmp al, 0x09
    je .is_ws
    jmp .got_token_start
.is_ws:
    inc rsi
    jmp .skip_ws
.got_token_start:
    mov rdi, rsi
.find_tok_end:
    inc rdi
    cmp rdi, r11
    jae .tok_end
    movzx eax, byte ptr [rdi]
    cmp al, 0x20
    je .tok_end
    cmp al, 0x09
    je .tok_end
    jmp .find_tok_end
.tok_end:
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, 1
    jne .check_number
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .is_op
    cmp al, '-'
    je .is_op
    cmp al, '*'
    je .is_op
    cmp al, '/'
    je .is_op
    cmp al, '%'
    je .is_op
.check_number:
    xor r9, r9
    mov r8, rsi
    cmp byte ptr [r8], '-'
    jne .parse_digits
    mov r9, 1
    inc r8
    dec rcx
.parse_digits:
    cmp rcx, 1
    jb .invalid
    cmp rcx, 18
    ja .invalid
    xor rax, rax
.parse_dig_loop:
    movzx edx, byte ptr [r8]
    cmp dl, '0'
    jb .invalid
    cmp dl, '9'
    ja .invalid
    imul rax, 10
    sub dl, '0'
    add rax, rdx
    inc r8
    dec rcx
    jnz .parse_dig_loop
    test r9, r9
    jz .push_num
    neg rax
.push_num:
    mov qword ptr [rbx + r15*8], rax
    inc r15
    mov rsi, rdi
    jmp .token_loop
.is_op:
    movzx r10d, byte ptr [rsi]
    cmp r15, 2
    jb .invalid
    dec r15
    mov rcx, qword ptr [rbx + r15*8]
    dec r15
    mov rax, qword ptr [rbx + r15*8]
    cmp r10b, '+'
    je .do_add
    cmp r10b, '-'
    je .do_sub
    cmp r10b, '*'
    je .do_mul
    cmp r10b, '/'
    je .do_div
    cmp r10b, '%'
    je .do_mod
.do_add:
    add rax, rcx
    jmp .push_res
.do_sub:
    sub rax, rcx
    jmp .push_res
.do_mul:
    imul rax, rcx
    jmp .push_res
.do_div:
    test rcx, rcx
    jz .invalid
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne .div_ok
    cmp rcx, -1
    je .invalid
.div_ok:
    cqo
    idiv rcx
    jmp .push_res
.do_mod:
    test rcx, rcx
    jz .invalid
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne .mod_ok
    cmp rcx, -1
    je .invalid
.mod_ok:
    cqo
    idiv rcx
    mov rax, rdx
.push_res:
    mov qword ptr [rbx + r15*8], rax
    inc r15
    mov rsi, rdi
    jmp .token_loop
.invalid:
    jmp .output_error
.tokens_done:
    cmp r15, 1
    jne .output_error
.output_num:
    mov rax, qword ptr [rbx]
    lea rdi, [rip + out_buf + 30]
    mov byte ptr [rdi], 10
    test rax, rax
    jns .num_pos
    mov r9, 1
    shl r9, 63
    cmp rax, r9
    je .num_min
    neg rax
    mov r10, 1
    jmp .num_digits
.num_pos:
    xor r10, r10
.num_digits:
    test rax, rax
    jnz .num_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp .num_finish
.num_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .num_loop
.num_finish:
    test r10, r10
    jz .num_write
    dec rdi
    mov byte ptr [rdi], '-'
.num_write:
    lea rdx, [rip + out_buf + 30]
    sub rdx, rdi
    inc rdx
    mov rsi, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp .line_loop
.num_min:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, 21
    syscall
    jmp .line_loop
.output_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    jmp .line_loop
.done:
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
err_msg:
    .ascii "error\n"
min_str:
    .ascii "-9223372036854775808\n"
.bss
.align 8
input_buf:
    .zero 65536
rpn_stack:
    .zero 8000
out_buf:
    .zero 32
