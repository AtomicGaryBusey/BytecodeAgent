.globl _start
.text
_start:
	lea r15, [rip + buffer]
	xor r14, r14
read_loop:
	mov eax, 0
	xor edi, edi
	lea rsi, [r15 + r14]
	mov rdx, 65536
	sub rdx, r14
	jbe done_read
	syscall
	test rax, rax
	jle done_read
	add r14, rax
	jmp read_loop
done_read:
	mov r13, r15
	lea r12, [r15 + r14]
process:
	cmp r13, r12
	jae exit_ok
	mov r11, r13
find_nl:
	cmp r11, r12
	je no_nl
	cmp byte ptr [r11], 10
	je has_nl
	inc r11
	jmp find_nl
has_nl:
	lea r10, [r11 + 1]
	mov rsi, r13
	mov rdi, r11
	dec rdi
rev1:
	cmp rsi, rdi
	jae rev1_done
	mov al, byte ptr [rsi]
	mov dl, byte ptr [rdi]
	mov byte ptr [rsi], dl
	mov byte ptr [rdi], al
	inc rsi
	dec rdi
	jmp rev1
rev1_done:
	mov eax, 1
	mov edi, 1
	mov rsi, r13
	mov rdx, r11
	sub rdx, r13
	syscall
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + nl]
	mov edx, 1
	syscall
	mov r13, r10
	jmp process
no_nl:
	mov rsi, r13
	mov rdi, r12
	dec rdi
rev2:
	cmp rsi, rdi
	jae rev2_done
	mov al, byte ptr [rsi]
	mov dl, byte ptr [rdi]
	mov byte ptr [rsi], dl
	mov byte ptr [rdi], al
	inc rsi
	dec rdi
	jmp rev2
rev2_done:
	mov eax, 1
	mov edi, 1
	mov rsi, r13
	mov rdx, r12
	sub rdx, r13
	syscall
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + nl]
	mov edx, 1
	syscall
exit_ok:
	mov eax, 60
	xor edi, edi
	syscall
.section .rodata
nl:
	.byte 10
.bss
.align 8
buffer:
	.space 65536
