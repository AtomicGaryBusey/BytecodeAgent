.globl _start

.text
_start:
	// Read all of stdin into buf
	adrp	x19, buf
	add	x19, x19, :lo12:buf
	mov	x20, #0			// bytes read so far
	mov	x21, #0x8000		// 32768 max

read_loop:
	sub	x2, x21, x20
	cbz	x2, read_done
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop

read_done:
	// Parse lines: x19=buf, x20=size
	adrp	x21, lines
	add	x21, x21, :lo12:lines
	adrp	x22, lengths
	add	x22, x22, :lo12:lengths
	mov	x23, #0			// line count
	mov	x24, #0			// pos
	mov	x25, #0			// line start

parse_loop:
	cmp	x24, x20
	b.hs	parse_tail
	ldrb	w0, [x19, x24]
	cmp	w0, #10
	b.ne	parse_next
	// Emit line [x25, x24)
	add	x0, x19, x25
	str	x0, [x21, x23, lsl #3]
	sub	x0, x24, x25
	str	w0, [x22, x23, lsl #2]
	add	x23, x23, #1
	add	x25, x24, #1
parse_next:
	add	x24, x24, #1
	b	parse_loop

parse_tail:
	cmp	x25, x20
	b.hs	parse_done
	add	x0, x19, x25
	str	x0, [x21, x23, lsl #3]
	sub	x0, x20, x25
	str	w0, [x22, x23, lsl #2]
	add	x23, x23, #1
parse_done:

	// Insertion sort on lines[0..x23)
	mov	x24, #1			// i
sort_outer:
	cmp	x24, x23
	b.hs	sort_done
	ldr	x9, [x21, x24, lsl #3]	// key_ptr
	ldr	w10, [x22, x24, lsl #2]	// key_len
	sub	x25, x24, #1		// j = i-1

sort_inner:
	tbnz	x25, #63, sort_insert	// j < 0
	ldr	x0, [x21, x25, lsl #3]
	ldr	w1, [x22, x25, lsl #2]
	mov	x2, x9
	mov	w3, w10
	bl	compare_lines
	cmp	x0, #0
	b.le	sort_insert		// lines[j] <= key
	// shift lines[j] to j+1
	add	x26, x25, #1
	ldr	x0, [x21, x25, lsl #3]
	str	x0, [x21, x26, lsl #3]
	ldr	w0, [x22, x25, lsl #2]
	str	w0, [x22, x26, lsl #2]
	sub	x25, x25, #1
	b	sort_inner

sort_insert:
	add	x26, x25, #1
	str	x9, [x21, x26, lsl #3]
	str	w10, [x22, x26, lsl #2]
	add	x24, x24, #1
	b	sort_outer
sort_done:

	// Print each line followed by '\n'
	mov	x24, #0
print_loop:
	cmp	x24, x23
	b.hs	print_done
	ldr	x1, [x21, x24, lsl #3]
	ldr	w2, [x22, x24, lsl #2]
	mov	x0, #1
	mov	x8, #64
	svc	#0
	mov	x0, #1
	adr	x1, nl
	mov	x2, #1
	mov	x8, #64
	svc	#0
	add	x24, x24, #1
	b	print_loop

print_done:
	mov	x0, #0
	mov	x8, #93
	svc	#0

// compare_lines: x0=ptr1, w1=len1, x2=ptr2, w3=len2
// returns x0 <0 / 0 / >0 ; clobbers x4-x7
compare_lines:
	mov	x4, #0
	cmp	w1, w3
	csel	w5, w1, w3, lo		// min length
cmp_loop:
	cmp	w4, w5
	b.hs	cmp_by_len
	ldrb	w6, [x0, x4]
	ldrb	w7, [x2, x4]
	cmp	w6, w7
	b.ne	cmp_diff
	add	x4, x4, #1
	b	cmp_loop
cmp_diff:
	sub	x0, x6, x7
	ret
cmp_by_len:
	sub	x0, x1, x3
	ret

.section .rodata
nl:
	.byte	10

.bss
.balign	8
buf:
	.space	32768
lines:
	.space	8000
lengths:
	.space	4000
