.equ BUFSIZE, 65536
.equ TAPESIZE, 30000
.equ MAXNEST, 1000

.globl _start
.text
_start:
	lea r13, [rip + input_buf]
	xor r12, r12
read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [r13 + r12]
	mov rdx, BUFSIZE
	sub rdx, r12
	jz read_done
	syscall
	test rax, rax
	jle read_done
	add r12, rax
	jmp read_loop
read_done:
	xor r14, r14
find_bang:
	cmp r14, r12
	jae no_bang
	cmp byte ptr [r13 + r14], 0x21
	je found_bang
	inc r14
	jmp find_bang
found_bang:
	lea r15, [r13 + r14 + 1]
	mov r8, r12
	sub r8, r14
	dec r8
	jmp bang_done
no_bang:
	xor r8, r8
	mov r15, r13
bang_done:
	lea r12, [rip + jumps]
	lea rsi, [rip + bracket_stack]
	xor ecx, ecx
	xor ebx, ebx
bracket_loop:
	cmp rbx, r14
	jae bracket_end
	movzx eax, byte ptr [r13 + rbx]
	cmp al, '['
	je do_open
	cmp al, ']'
	je do_close
	inc ebx
	jmp bracket_loop
do_open:
	cmp ecx, MAXNEST
	jae unbalanced
	mov dword ptr [rsi + rcx*4], ebx
	inc ecx
	inc ebx
	jmp bracket_loop
do_close:
	test ecx, ecx
	jz unbalanced
	dec ecx
	mov edx, dword ptr [rsi + rcx*4]
	lea eax, [rbx + 1]
	mov dword ptr [r12 + rdx*4], eax
	lea eax, [rdx + 1]
	mov dword ptr [r12 + rbx*4], eax
	inc ebx
	jmp bracket_loop
bracket_end:
	test ecx, ecx
	jnz unbalanced
	lea r9, [rip + tape]
	xor r10, r10
	xor ebx, ebx
interp_loop:
	cmp rbx, r14
	jae exit_ok
	movzx eax, byte ptr [r13 + rbx]
	cmp al, '>'
	je do_gt
	cmp al, '<'
	je do_lt
	cmp al, '+'
	je do_plus
	cmp al, '-'
	je do_minus
	cmp al, '.'
	je do_dot
	cmp al, ','
	je do_comma
	cmp al, '['
	je do_lbrack
	cmp al, ']'
	je do_rbrack
	inc ebx
	jmp interp_loop
do_gt:
	inc r9
	inc ebx
	jmp interp_loop
do_lt:
	dec r9
	inc ebx
	jmp interp_loop
do_plus:
	inc byte ptr [r9]
	inc ebx
	jmp interp_loop
do_minus:
	dec byte ptr [r9]
	inc ebx
	jmp interp_loop
do_dot:
	mov rax, 1
	mov rdi, 1
	mov rsi, r9
	mov rdx, 1
	syscall
	inc ebx
	jmp interp_loop
do_comma:
	cmp r10, r8
	jae comma_eof
	mov al, byte ptr [r15 + r10]
	mov byte ptr [r9], al
	inc r10
	jmp comma_done
comma_eof:
	mov byte ptr [r9], 0
comma_done:
	inc ebx
	jmp interp_loop
do_lbrack:
	cmp byte ptr [r9], 0
	jne lbrack_ Cont
	mov ebx, dword ptr [r12 + rbx*4]
	jmp interp_loop
lbrack_cont:
	inc ebx
	jmp interp_loop
do_rbrack:
	cmp byte ptr [r9], 0
	je rbrack_cont
	mov ebx, dword ptr [r12 + rbx*4]
	jmp interp_loop
rbrack_cont:
	inc ebx
	jmp interp_loop
unbalanced:
	mov rax, 60
	mov rdi, 1
	syscall
exit_ok:
	mov rax, 60
	xor rdi, rdi
	syscall

.section .bss
.align 16
input_buf:
	.space BUFSIZE
.align 8
tape:
	.space TAPESIZE
.align 8
jumps:
	.space 262144
.align 8
bracket_stack:
	.space 4000
