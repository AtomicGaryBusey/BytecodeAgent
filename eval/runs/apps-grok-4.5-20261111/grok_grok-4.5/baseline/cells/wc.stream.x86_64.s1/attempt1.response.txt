.globl _start

.bss
.align 16
buffer:
	.zero 65536
outbuf:
	.zero 64

.text
_start:
	lea r12, [rip + buffer]
	mov rsi, r12
	mov r13d, 65536

read_loop:
	xor eax, eax
	xor edi, edi
	mov edx, r13d
	syscall
	test rax, rax
	jle read_done
	add rsi, rax
	sub r13d, eax
	jnz read_loop

read_done:
	mov r14, rsi
	sub r14, r12
	mov r13, rsi

	xor r8, r8
	xor r9, r9
	xor ecx, ecx
	mov rsi, r12

count_loop:
	cmp rsi, r13
	jae count_done
	movzx eax, byte ptr [rsi]
	inc rsi

	cmp al, 0x0A
	jne check_ws
	inc r8

check_ws:
	cmp al, 0x20
	je is_ws
	cmp al, 0x09
	jb not_ws
	cmp al, 0x0D
	jbe is_ws

not_ws:
	test ecx, ecx
	jnz count_loop
	mov ecx, 1
	inc r9
	jmp count_loop

is_ws:
	xor ecx, ecx
	jmp count_loop

count_done:
	lea rdi, [rip + outbuf]
	mov rax, r8
	call itoa
	mov byte ptr [rdi], ' '
	inc rdi
	mov rax, r9
	call itoa
	mov byte ptr [rdi], ' '
	inc rdi
	mov rax, r14
	call itoa
	mov byte ptr [rdi], 0x0A
	inc rdi

	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall

	mov eax, 60
	xor edi, edi
	syscall

itoa:
	mov rsi, rdi
	test rax, rax
	jnz itoa_nonzero
	mov byte ptr [rdi], '0'
	inc rdi
	ret

itoa_nonzero:
itoa_divloop:
	xor rdx, rdx
	mov ecx, 10
	div rcx
	add dl, '0'
	mov byte ptr [rdi], dl
	inc rdi
	test rax, rax
	jnz itoa_divloop

	mov rcx, rdi
	dec rcx
itoa_revloop:
	cmp rsi, rcx
	jae itoa_done
	mov al, byte ptr [rsi]
	mov dl, byte ptr [rcx]
	mov byte ptr [rsi], dl
	mov byte ptr [rcx], al
	inc rsi
	dec rcx
	jmp itoa_revloop

itoa_done:
	ret
