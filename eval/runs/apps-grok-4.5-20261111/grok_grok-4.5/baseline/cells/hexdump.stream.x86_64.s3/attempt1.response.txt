.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60

.globl _start
.text
_start:
	lea rbx, [rip + input_buf]
	mov r12, rbx
	mov r13, 65536
.Lread_loop:
	xor eax, eax
	xor edi, edi
	mov rsi, r12
	mov rdx, r13
	syscall
	test rax, rax
	jle .Lread_done
	add r12, rax
	sub r13, rax
	jnz .Lread_loop
.Lread_done:
	mov r14, r12
	sub r14, rbx
	# r15 = 0 (off) already
.Lrows:
	cmp r15, r14
	jae .Lfinal
	mov rcx, r14
	sub rcx, r15
	mov r8d, 16
	cmp rcx, r8
	cmovb r8, rcx
	lea r10, [rbx + r15]
	lea rdi, [rip + line_buf]
	mov eax, r15d
	call put_hex8
	mov al, ' '
	stosb
	stosb
	xor r9d, r9d
.Lhex_loop:
	cmp r9, r8
	jae .Ldo_pad
	movzx eax, byte ptr [r10 + r9]
	call put_hex2
	mov al, ' '
	stosb
	jmp .Lhex_next
.Ldo_pad:
	mov al, ' '
	stosb
	stosb
	stosb
.Lhex_next:
	cmp r9d, 7
	jne .Lno_extra
	mov al, ' '
	stosb
.Lno_extra:
	inc r9
	cmp r9d, 16
	jb .Lhex_loop
	mov al, ' '
	stosb
	mov al, '|'
	stosb
	xor r9d, r9d
.Lasc_loop:
	cmp r9, r8
	jae .Lasc_done
	mov al, byte ptr [r10 + r9]
	cmp al, 0x20
	jb .Ldo_dot
	cmp al, 0x7e
	ja .Ldo_dot
	jmp .Lput_char
.Ldo_dot:
	mov al, '.'
.Lput_char:
	stosb
	inc r9
	jmp .Lasc_loop
.Lasc_done:
	mov al, '|'
	stosb
	mov al, 0x0a
	stosb
	lea rsi, [rip + line_buf]
	mov rdx, rdi
	sub rdx, rsi
	mov eax, SYS_write
	mov edi, 1
	syscall
	add r15, 16
	jmp .Lrows
.Lfinal:
	lea rdi, [rip + line_buf]
	mov eax, r14d
	call put_hex8
	mov al, 0x0a
	stosb
	lea rsi, [rip + line_buf]
	mov rdx, rdi
	sub rdx, rsi
	mov eax, SYS_write
	mov edi, 1
	syscall
	mov eax, SYS_exit
	xor edi, edi
	syscall

put_hex8:
	mov ecx, 8
1:
	rol eax, 4
	mov edx, eax
	and edx, 0xf
	mov dl, byte ptr [hexdigits + rdx]
	mov byte ptr [rdi], dl
	inc rdi
	dec ecx
	jnz 1b
	ret

put_hex2:
	movzx ecx, al
	mov eax, ecx
	shr eax, 4
	mov al, byte ptr [hexdigits + rax]
	mov byte ptr [rdi], al
	and ecx, 0xf
	mov al, byte ptr [hexdigits + rcx]
	mov byte ptr [rdi + 1], al
	add rdi, 2
	ret

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.bss
.align 16
input_buf:
	.space 65536
line_buf:
	.space 80
