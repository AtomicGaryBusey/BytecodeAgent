.globl _start

.section .rodata

	.ascii "ok\n"

.text

	add x19, x19, :lo12:buffer
	mov x22, #0
.Lread_loop:
	mov x0, #65536
	sub x2, x0, x22
	cbz x2, .Lread_done
	mov x0, #0
	add x1, x19, x22
	mov x8, #63
	svc #0

	mov x20, x19

	bl parse_value
	cbnz x0, .Lerror
	bl skip_ws
	cmp x20, x21

	mov x0, #1
	adrp x1, str_ok

	adrp x1, str_nl
	add x1, x1, :lo12:str_nl

	mov x8, #64

	mov x0, #1

	svc #0

skip_ws:
.Lskip_loop:
	cmp x20, x21
	b.eq .Lskip_done
	ldrb w0, [x20]
	cmp w0, #0x20
	b.eq .Lskip_adv
	cmp w0, #0x09
	b.eq .Lskip_adv
	cmp w0, #0x0A
	b.eq .Lskip_adv
	cmp w0, #0x0D
	b.eq .Lskip_adv
	b .Lskip_done
.Lskip_adv:
	add x20, x20, #1
	b .Lskip_loop
.Lskip_done:

parse_value:
	stp x29, x30, [sp, #-16]!
	cmp x20, x21
	b.eq .Lval_err
	ldrb w0, [x20]
	cmp w0, #'{'
	b.eq .Lval_obj
	cmp w0, #'['
	b.eq .Lval_arr
	cmp w0, #'"'
	b.eq .Lval_str
	cmp w0, #'t'
	b.eq .Lval_true
	cmp w0, #'f'
	b.eq .Lval_false
	cmp w0, #'n'
	b.eq .Lval_null
	cmp w0, #'-'
	b.eq .Lval_num
	cmp w0, #'0'
	b.lo .Lval_err
	cmp w0, #'9'
	b.hi .Lval_err
.Lval_num:
	bl parse_number
	b .Lval_done
.Lval_obj:
	bl parse_object
	b .Lval_done
.Lval_arr:
	bl parse_array
	b .Lval_done
.Lval_str:
	bl parse_string
	b .Lval_done
.Lval_true:
	adrp x1, lit_true
	add x1, x1, :lo12:lit_true
	mov x2, #4
	bl match_literal
	b .Lval_done
.Lval_false:
	adrp x1, lit_false
	add x1, x1, :lo12:lit_false
	mov x2, #5
	bl match_literal
	b .Lval_done
.Lval_null:
	adrp x1, lit_null
	add x1, x1, :lo12:lit_null
	mov x2, #4
	bl match_literal
	b .Lval_done
.Lval_err:
	mov x0, #1
.Lval_done:
	ldp x29, x30, [sp], #16
	ret

	ret

	stp x29, x30, [sp, #-16]!
	add x20, x20, #1
	bl skip_ws
	cmp x20, x21
	b.eq .Lobj_err
	ldrb w0, [x20]

	cmp w0, #'"'
	b.ne .Lobj_err
	bl parse_string
	cbnz x0, .Lobj_err_ret
	bl skip_ws
	cmp x20, x21
	b.eq .Lobj_err
	ldrb w0, [x20]
	cmp w0, #':'
	b.ne .Lobj_err
	add x20, x20, #1
	bl skip_ws
	bl parse_value
	cbnz x0, .Lobj_err_ret
	bl skip_ws
	cmp x20, x21
	b.eq .Lobj_err
	ldrb w0, [x20]
	cmp w0, #','
	b.eq .Lobj_comma
	cmp w0, #'}'

.Lobj_comma:
	add x20, x20, #1
	bl skip_ws
	cmp x20, x21
	b.eq .Lobj_err
	ldrb w0, [x20]

.Lobj_end:
	add x20, x20, #1
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
.Lobj_err:

.Lobj_err_ret:
	ldp x29, x30, [sp], #16
	ret

	add x20, x20, #1
.Lstr_loop:
	cmp x20, x21
	b.eq .Lstr_fail
	ldrb w0, [x20]
	cmp w0, #0x22
	b.eq .Lstr_end

	b.eq .Lstr_esc_ok
	cmp w0, #0x2F

	cmp w0, #'b'
	b.eq .Lstr_esc_ok
	cmp w0, #'f'
	b.eq .Lstr_esc_ok
	cmp w0, #'n'
	b.eq .Lstr_esc_ok
	cmp w0, #'r'
	b.eq .Lstr_esc_ok
	cmp w0, #'t'

	cmp w0, #'F'

	ldrb w5, [x1, x3]
