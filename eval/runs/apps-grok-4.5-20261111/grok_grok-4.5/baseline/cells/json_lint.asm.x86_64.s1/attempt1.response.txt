.globl _start

.section .rodata
msg_ok:
.ascii "ok\n"
msg_err:
.ascii "error at "
lit_true:
.ascii "true"
lit_false:
.ascii "false"
lit_null:
.ascii "null"

.bss
.align 16
buffer:
.zero 65536
numbuf:
.zero 32
outbuf:
.zero 64

.text
_start:
    lea r12, [rip + buffer]
    mov rbx, r12
    mov r15d, 65536
.read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r15
    syscall
    test rax, rax
    jle .read_done
    add rbx, rax
    sub r15, rax
    jnz .read_loop
.read_done:
    mov r14, rbx
    mov r13, r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jne report_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
1:
    cmp r13, r14
    jae 2f
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
2:
    ret
3:
    inc r13
    jmp 1b

parse_value:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe parse_number
    jmp report_error

parse_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    jmp parse_literal

parse_literal:
1:
    test edx, edx
    jz 2f
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, byte ptr [rsi]
    jne report_error
    inc r13
    inc rsi
    dec edx
    jmp 1b
2:
    ret

parse_string:
    inc r13
.str_loop:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '"'
    je .str_end
    cmp al, '\\'
    je .str_esc
    cmp al, 0x20
    jb report_error
    inc r13
    jmp .str_loop
.str_end:
    inc r13
    ret
.str_esc:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    inc r13
    cmp al, '"'
    je .str_loop
    cmp al, '\\'
    je .str_loop
    cmp al, '/'
    je .str_loop
    cmp al, 'b'
    je .str_loop
    cmp al, 'f'
    je .str_loop
    cmp al, 'n'
    je .str_loop
    cmp al, 'r'
    je .str_loop
    cmp al, 't'
    je .str_loop
    cmp al, 'u'
    jne .str_bad_esc
    mov ecx, 4
.str_hex:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe .str_hex_ok
    cmp al, 'A'
    jb report_error
    cmp al, 'F'
    jbe .str_hex_ok
    cmp al, 'a'
    jb report_error
    cmp al, 'f'
    ja report_error
.str_hex_ok:
    inc r13
    dec ecx
    jnz .str_hex
    jmp .str_loop
.str_bad_esc:
    dec r13
    jmp report_error

parse_number:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '-'
    jne .num_int
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
.num_int:
    cmp al, '0'
    je .num_zero
    cmp al, '1'
    jb report_error
    cmp al, '9'
    ja report_error
    inc r13
.num_digits:
    cmp r13, r14
    jae .num_after_int
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .num_after_int
    cmp al, '9'
    ja .num_after_int
    inc r13
    jmp .num_digits
.num_zero:
    inc r13
    cmp r13, r14
    jae .num_after_int
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .num_after_int
    cmp al, '9'
    ja .num_after_int
    jmp report_error
.num_after_int:
    cmp r13, r14
    jae .num_done
    movzx eax, byte ptr [r13]
    cmp al, '.'
    jne .num_check_exp
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
.num_frac:
    inc r13
    cmp r13, r14
    jae .num_after_frac
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .num_after_frac
    cmp al, '9'
    ja .num_after_frac
    jmp .num_frac
.num_after_frac:
.num_check_exp:
    cmp r13, r14
    jae .num_done
    movzx eax, byte ptr [r13]
    cmp al, 'e'
    je .num_exp
    cmp al, 'E'
    je .num_exp
    jmp .num_done
.num_exp:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '+'
    je .num_exp_sign
    cmp al, '-'
    je .num_exp_sign
    jmp .num_exp_dig
.num_exp_sign:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
.num_exp_dig:
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
.num_exp_loop:
    inc r13
    cmp r13, r14
    jae .num_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .num_done
    cmp al, '9'
    ja .num_done
    jmp .num_exp_loop
.num_done:
    ret

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    cmp byte ptr [r13], '}'
    je .obj_end
.obj_member:
    cmp byte ptr [r13], '"'
    jne report_error
    call parse_string
    call skip_ws
    cmp r13, r14
    jae report_error
    cmp byte ptr [r13], ':'
    jne report_error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '}'
    je .obj_end
    cmp al, ','
    jne report_error
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    jmp .obj_member
.obj_end:
    inc r13
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    cmp byte ptr [r13], ']'
    je .arr_end
.arr_loop:
    call parse_value
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, ']'
    je .arr_end
    cmp al, ','
    jne report_error
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    jmp .arr_loop
.arr_end:
    inc r13
    ret

report_error:
    mov rax, r13
    sub rax, r12
    lea rdi, [rip + outbuf]
    lea rsi, [rip + msg_err]
    mov ecx, 9
    rep movsb
    lea rsi, [rip + numbuf + 20]
    mov rbx, rsi
    test rax, rax
    jnz .conv
    dec rsi
    mov byte ptr [rsi], '0'
    jmp .conv_done
.conv:
    mov ecx, 10
.conv_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .conv_loop
.conv_done:
    mov rcx, rbx
    sub rcx, rsi
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
