.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.text
_start:
	mov	x19, #0
	adrp	x20, buffer
	add	x20, x20, :lo12:buffer

read_loop:
	mov	x0, #0
	add	x1, x20, x19
	mov	x2, #BUFSIZE
	sub	x2, x2, x19
	cbz	x2, read_done
	mov	x8, #SYS_read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x19, x19, x0
	b	read_loop

read_done:
	mov	x21, #0

print_rows:
	cmp	x21, x19
	b.hs	print_final

	sub	x22, x19, x21
	cmp	x22, #16
	mov	x23, #16
	csel	x22, x22, x23, lo

	adrp	x24, linebuf
	add	x24, x24, :lo12:linebuf
	mov	x25, x24

	mov	x0, x21
	mov	x1, x25
	bl	write_hex8
	mov	x25, x0

	mov	w0, #' '
	strb	w0, [x25], #1
	strb	w0, [x25], #1

	adrp	x27, hexdigits
	add	x27, x27, :lo12:hexdigits
	mov	x26, #0
hex_bytes_loop:
	cmp	x26, #16
	b.ge	hex_bytes_done

	cmp	x26, x22
	b.ge	pad_spaces

	add	x0, x20, x21
	ldrb	w0, [x0, x26]
	lsr	w1, w0, #4
	ldrb	w1, [x27, x1]
	strb	w1, [x25], #1
	and	w0, w0, #0xf
	ldrb	w0, [x27, x0]
	strb	w0, [x25], #1
	mov	w0, #' '
	strb	w0, [x25], #1
	b	after_slot

pad_spaces:
	mov	w0, #' '
	strb	w0, [x25], #1
	strb	w0, [x25], #1
	strb	w0, [x25], #1

after_slot:
	cmp	x26, #7
	b.ne	1f
	mov	w0, #' '
	strb	w0, [x25], #1
1:
	add	x26, x26, #1
	b	hex_bytes_loop

hex_bytes_done:
	mov	w0, #' '
	strb	w0, [x25], #1
	mov	w0, #'|'
	strb	w0, [x25], #1

	mov	x26, #0
ascii_loop:
	cmp	x26, x22
	b.ge	ascii_done
	add	x0, x20, x21
	ldrb	w0, [x0, x26]
	cmp	w0, #0x20
	b.lo	not_print
	cmp	w0, #0x7f
	b.hs	not_print
	b	store_ch
not_print:
	mov	w0, #'.'
store_ch:
	strb	w0, [x25], #1
	add	x26, x26, #1
	b	ascii_loop

ascii_done:
	mov	w0, #'|'
	strb	w0, [x25], #1
	mov	w0, #'\n'
	strb	w0, [x25], #1

	mov	x0, #1
	mov	x1, x24
	sub	x2, x25, x24
	mov	x8, #SYS_write
	svc	#0

	add	x21, x21, #16
	b	print_rows

print_final:
	adrp	x24, linebuf
	add	x24, x24, :lo12:linebuf
	mov	x0, x19
	mov	x1, x24
	bl	write_hex8
	mov	x25, x0
	mov	w0, #'\n'
	strb	w0, [x25], #1

	mov	x0, #1
	mov	x1, x24
	sub	x2, x25, x24
	mov	x8, #SYS_write
	svc	#0

	mov	x0, #0
	mov	x8, #SYS_exit
	svc	#0

write_hex8:
	mov	x5, x0
	adrp	x4, hexdigits
	add	x4, x4, :lo12:hexdigits
	mov	x2, #28
1:
	lsr	x3, x5, x2
	and	x3, x3, #0xf
	ldrb	w3, [x4, x3]
	strb	w3, [x1], #1
	subs	x2, x2, #4
	b.ge	1b
	mov	x0, x1
	ret

.bss
.align 4
buffer:
	.space BUFSIZE
linebuf:
	.space 80
