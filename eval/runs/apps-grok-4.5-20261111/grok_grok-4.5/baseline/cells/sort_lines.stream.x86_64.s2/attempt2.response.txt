.globl _start
.text
_start:
	lea rsi, [rip + buffer]
	xor rbx, rbx
read_loop:
	mov rax, 0
	mov rdi, 0
	mov rdx, 32768
	sub rdx, rbx
	jz read_done
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	add rsi, rax
	jmp read_loop
read_done:
	lea rsi, [rip + buffer]
	lea rdi, [rip + lines]
	xor ecx, ecx
	xor r8, r8
	xor r9, r9
parse_loop:
	cmp r9, rbx
	jae parse_end
	mov al, byte ptr [rsi + r9]
	cmp al, 0x0A
	jne parse_next
	mov rax, rsi
	add rax, r8
	mov qword ptr [rdi], rax
	mov rax, r9
	sub rax, r8
	mov qword ptr [rdi + 8], rax
	add rdi, 16
	inc ecx
	lea r8, [r9 + 1]
parse_next:
	inc r9
	jmp parse_loop
parse_end:
	cmp r8, rbx
	jae parse_no_last
	mov rax, rsi
	add rax, r8
	mov qword ptr [rdi], rax
	mov rax, rbx
	sub rax, r8
	mov qword ptr [rdi + 8], rax
	inc ecx
parse_no_last:
	mov r12, rcx
	cmp r12, 2
	jb do_print
	xor r13, r13
outer_loop:
	mov rax, r12
	dec rax
	cmp r13, rax
	jae do_print
	xor r14, r14
	mov r15, rax
	sub r15, r13
inner_loop:
	cmp r14, r15
	jae outer_next
	lea rdi, [rip + lines]
	mov rax, r14
	shl rax, 4
	add rdi, rax
	lea rsi, [rdi + 16]
	call compare_lines
	cmp rax, 0
	jle no_swap
	call swap_lines
no_swap:
	inc r14
	jmp inner_loop
outer_next:
	inc r13
	jmp outer_loop
do_print:
	lea rbx, [rip + lines]
print_loop:
	test r12, r12
	jz do_exit
	mov rax, 1
	mov rdi, 1
	mov rsi, qword ptr [rbx]
	mov rdx, qword ptr [rbx + 8]
	syscall
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + newline]
	mov rdx, 1
	syscall
	add rbx, 16
	dec r12
	jmp print_loop
do_exit:
	mov rax, 60
	xor rdi, rdi
	syscall
compare_lines:
	mov r8, qword ptr [rdi]
	mov rcx, qword ptr [rdi + 8]
	mov r9, qword ptr [rsi]
	mov rdx, qword ptr [rsi + 8]
	mov r10, rcx
	cmp r10, rdx
	cmova r10, rdx
	xor r11, r11
1:
	cmp r11, r10
	jae 2f
	mov al, byte ptr [r8 + r11]
	cmp al, byte ptr [r9 + r11]
	jne 3f
	inc r11
	jmp 1b
3:
	jb 4f
	mov rax, 1
	ret
2:
	cmp rcx, rdx
	jb 4f
	ja 5f
	xor eax, eax
	ret
4:
	mov rax, -1
	ret
5:
	mov rax, 1
	ret
swap_lines:
	mov rax, qword ptr [rdi]
	mov rcx, qword ptr [rsi]
	mov qword ptr [rdi], rcx
	mov qword ptr [rsi], rax
	mov rax, qword ptr [rdi + 8]
	mov rcx, qword ptr [rsi + 8]
	mov qword ptr [rdi + 8], rcx
	mov qword ptr [rsi + 8], rax
	ret
.section .rodata
newline:
	.byte 0x0A
.section .bss
.align 8
buffer:
	.zero 32768
.align 8
lines:
	.zero 16000
