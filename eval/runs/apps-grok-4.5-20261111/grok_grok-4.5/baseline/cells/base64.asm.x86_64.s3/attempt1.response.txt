.equ INPUT_MAX, 49152
.equ LINE_LEN, 76
.equ OUTPUT_MAX, 67000

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
.zero INPUT_MAX + 4
.align 16
output_buf:
.zero OUTPUT_MAX

.text
.globl _start
_start:
lea rsi, [rip + input_buf]
mov edx, INPUT_MAX
xor ebx, ebx
read_loop:
mov eax, 0
xor edi, edi
syscall
test rax, rax
jz read_done
js read_done
add ebx, eax
add rsi, rax
sub edx, eax
jnz read_loop
read_done:
test ebx, ebx
jz do_exit
lea rsi, [rip + input_buf]
lea rdi, [rip + output_buf]
xor r8d, r8d
xor r9d, r9d
encode_loop:
cmp r9d, ebx
jge encode_done
mov ecx, ebx
sub ecx, r9d
movzx r12d, byte ptr [rsi + r9]
xor r13d, r13d
xor r14d, r14d
cmp ecx, 1
jle .Lhave
movzx r13d, byte ptr [rsi + r9 + 1]
cmp ecx, 2
jle .Lhave
movzx r14d, byte ptr [rsi + r9 + 2]
.Lhave:
mov eax, r12d
shr eax, 2
movzx eax, byte ptr [rip + alphabet + rax]
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 1f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
1:
mov eax, r12d
and eax, 3
shl eax, 4
mov edx, r13d
shr edx, 4
or eax, edx
movzx eax, byte ptr [rip + alphabet + rax]
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 2f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
2:
cmp ecx, 1
je .Lpad2
mov eax, r13d
and eax, 0xf
shl eax, 2
mov edx, r14d
shr edx, 6
or eax, edx
movzx eax, byte ptr [rip + alphabet + rax]
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 3f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
3:
cmp ecx, 2
je .Lpad1
mov eax, r14d
and eax, 63
movzx eax, byte ptr [rip + alphabet + rax]
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 4f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
4:
jmp .Lnext
.Lpad2:
mov al, '='
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 5f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
5:
mov al, '='
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 6f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
6:
jmp .Lnext
.Lpad1:
mov al, '='
mov byte ptr [rdi], al
inc rdi
inc r8d
cmp r8d, LINE_LEN
jne 7f
mov byte ptr [rdi], 0x0a
inc rdi
xor r8d, r8d
7:
.Lnext:
add r9d, 3
jmp encode_loop
encode_done:
test r8d, r8d
jz .Lno_final_nl
mov byte ptr [rdi], 0x0a
inc rdi
.Lno_final_nl:
lea rsi, [rip + output_buf]
mov rdx, rdi
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
do_exit:
mov eax, 60
xor edi, edi
syscall
