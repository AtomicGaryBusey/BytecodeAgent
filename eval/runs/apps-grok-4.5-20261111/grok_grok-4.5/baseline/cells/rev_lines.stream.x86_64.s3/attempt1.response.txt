.globl _start
.text
_start:
	lea rsi, [rip + buffer]
	mov rdx, 65536
	xor r8, r8
read_loop:
	mov rax, 0
	xor rdi, rdi
	syscall
	test rax, rax
	jle read_done
	add r8, rax
	add rsi, rax
	sub rdx, rax
	jnz read_loop
read_done:
	lea r9, [rip + buffer]
	mov r10, r9
	lea r11, [r9 + r8]
process_loop:
	cmp r10, r11
	jae last_line
	cmp byte ptr [r10], 10
	je found_nl
	inc r10
	jmp process_loop
found_nl:
	mov rdi, r9
	mov rsi, r10
	call reverse
	mov rax, 1
	mov rdi, 1
	mov rsi, r9
	mov rdx, r10
	sub rdx, r9
	inc rdx
	syscall
	lea r9, [r10 + 1]
	lea r10, [r10 + 1]
	jmp process_loop
last_line:
	cmp r9, r11
	je done
	mov rdi, r9
	mov rsi, r11
	call reverse
	mov rax, 1
	mov rdi, 1
	mov rsi, r9
	mov rdx, r11
	sub rdx, r9
	syscall
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + newline]
	mov rdx, 1
	syscall
done:
	mov rax, 60
	xor rdi, rdi
	syscall
reverse:
	cmp rdi, rsi
	jae rev_done
	dec rsi
rev_loop:
	cmp rdi, rsi
	jae rev_done
	mov al, byte ptr [rdi]
	mov cl, byte ptr [rsi]
	mov byte ptr [rdi], cl
	mov byte ptr [rsi], al
	inc rdi
	dec rsi
	jmp rev_loop
rev_done:
	ret
.section .rodata
newline:
	.byte 10
.section .bss
.align 16
buffer:
	.zero 65536
