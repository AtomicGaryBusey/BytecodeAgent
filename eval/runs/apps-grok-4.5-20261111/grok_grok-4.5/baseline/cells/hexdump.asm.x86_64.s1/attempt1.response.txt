.globl _start
.text
_start:
        cld
        xor     ebx, ebx
read_more:
        mov     edx, 65536
        sub     edx, ebx
        jz      read_done
        lea     rsi, [rip + buffer]
        add     rsi, rbx
        xor     eax, eax
        xor     edi, edi
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_more
read_done:
        mov     r13, rbx
        xor     r12, r12
print_rows:
        cmp     r12, r13
        jae     print_total
        mov     r14, r13
        sub     r14, r12
        cmp     r14, 16
        jbe     1f
        mov     r14, 16
1:
        lea     rdi, [rip + linebuf]
        mov     rax, r12
        call    put_hex8
        mov     ax, 0x2020
        stosw
        lea     rsi, [rip + buffer]
        add     rsi, r12
        xor     r15d, r15d
hexl:
        cmp     r15, r14
        jae     padl
        movzx   eax, byte ptr [rsi + r15]
        call    put_hex2
        mov     al, 0x20
        stosb
        jmp     hexnext
padl:
        mov     word ptr [rdi], 0x2020
        mov     byte ptr [rdi + 2], 0x20
        add     rdi, 3
hexnext:
        cmp     r15d, 7
        jne     no_extra_sp
        mov     al, 0x20
        stosb
no_extra_sp:
        inc     r15
        cmp     r15d, 16
        jb      hexl
        mov     ax, 0x7c20
        stosw
        xor     r15d, r15d
ascl:
        cmp     r15, r14
        jae     ascdone
        movzx   eax, byte ptr [rsi + r15]
        cmp     al, 0x20
        jb      dodot
        cmp     al, 0x7e
        ja      dodot
        stosb
        jmp     ascnext
dodot:
        mov     al, '.'
        stosb
ascnext:
        inc     r15
        jmp     ascl
ascdone:
        mov     al, '|'
        stosb
        mov     al, 10
        stosb
        mov     rdx, rdi
        lea     rsi, [rip + linebuf]
        sub     rdx, rsi
        mov     eax, 1
        mov     edi, 1
        syscall
        add     r12, 16
        jmp     print_rows
print_total:
        lea     rdi, [rip + linebuf]
        mov     rax, r13
        call    put_hex8
        mov     al, 10
        stosb
        mov     edx, 9
        lea     rsi, [rip + linebuf]
        mov     eax, 1
        mov     edi, 1
        syscall
        mov     eax, 60
        xor     edi, edi
        syscall
put_hex8:
        mov     ecx, 8
1:
        mov     ebx, eax
        shr     ebx, 28
        movzx   ebx, byte ptr [hexdigits + rbx]
        mov     [rdi], bl
        inc     rdi
        shl     eax, 4
        dec     ecx
        jnz     1b
        ret
put_hex2:
        movzx   eax, al
        mov     ebx, eax
        shr     ebx, 4
        movzx   ebx, byte ptr [hexdigits + rbx]
        mov     [rdi], bl
        inc     rdi
        and     eax, 0xf
        movzx   eax, byte ptr [hexdigits + rax]
        mov     [rdi], al
        inc     rdi
        ret
.section .rodata
hexdigits:
        .ascii  "0123456789abcdef"
.bss
        .align  16
buffer:
        .zero   65536
linebuf:
        .zero   80
