	xor edi, edi
	lea rsi, [r13 + r12]
	mov edx, 49152

	test rax, rax
	jle read_done
	add r12, rax
	jmp read_loop

	lea rsi, [rip + inbuf]
	mov rbx, r12
	lea rdi, [rip + outbuf]
	xor r14, r14

encode_loop:

encode_tail:

	je encode_finish
	cmp rbx, 1
	je encode_one
	movzx eax, byte ptr [rsi]
	shl eax, 16
	mov r8d, eax

encode_finish:
