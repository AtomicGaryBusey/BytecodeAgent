_start:
	xor ebx, ebx
read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [rip + buffer]
	add rsi, rbx
	mov rdx, 32768
	sub rdx, rbx
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	cmp rbx, 32768
	jb read_loop
read_done:
	xor r8d, r8d
	xor ecx, ecx
	lea r14, [rip + buffer]
parse_loop:
	cmp rcx, rbx
	jae parse_done
	mov rax, r14
	add rax, rcx
	lea rdx, [rip + line_ptr]
	mov qword ptr [rdx + r8*8], rax
	mov rdx, rcx
find_nl:
	cmp rdx, rbx
	jae no_nl
	cmp byte ptr [r14 + rdx], 0x0A
	je got_nl
	inc rdx
	jmp find_nl
got_nl:
	mov rsi, rdx
	sub rsi, rcx
	lea rax, [rip + line_len]
	mov dword ptr [rax + r8*4], esi
	inc rdx
	mov rcx, rdx
	inc r8d
	cmp r8d, 1000
	jae parse_done
	jmp parse_loop
no_nl:
	mov rsi, rdx
	sub rsi, rcx
	lea rax, [rip + line_len]
	mov dword ptr [rax + r8*4], esi
	inc r8d
parse_done:
	mov r12d, r8d
	cmp r12d, 2
	jb no_need_sort
	xor ebx, ebx
sort_outer:
	cmp ebx, r12d
	jae sort_done
	mov r13d, ebx
	inc r13d
sort_inner:
	cmp r13d, r12d
	jae next_i
	mov edi, r13d
	mov esi, ebx
	call compare_lines
	test rax, rax
	jns no_swap
	lea rax, [rip + line_ptr]
	mov r8, qword ptr [rax + rbx*8]
	mov r9, qword ptr [rax + r13*8]
	mov qword ptr [rax + rbx*8], r9
	mov qword ptr [rax + r13*8], r8
	lea rax, [rip + line_len]
	mov r8d, dword ptr [rax + rbx*4]
	mov r9d, dword ptr [rax + r13*4]
	mov dword ptr [rax + rbx*4], r9d
	mov dword ptr [rax + r13*4], r8d
no_swap:
	inc r13d
	jmp sort_inner
next_i:
	inc ebx
	jmp sort_outer
sort_done:
no_need_sort:
	xor ebx, ebx
print_loop:
	cmp ebx, r12d
	jae do_exit
	lea rax, [rip + line_ptr]
	mov rsi, qword ptr [rax + rbx*8]
	lea rax, [rip + line_len]

	syscall

	mov rdi, 1

	mov edx, 1
	syscall
	inc ebx
	jmp print_loop
do_exit:
	mov rax, 60
	xor edi, edi
	syscall

compare_lines:
	lea rax, [rip + line_ptr]
	mov r8, qword ptr [rax + rdi*8]
	lea rax, [rip + line_len]
	mov r9d, dword ptr [rax + rdi*4]
	lea rax, [rip + line_ptr]
	mov r10, qword ptr [rax + rsi*8]
	lea rax, [rip + line_len]
	mov r11d, dword ptr [rax + rsi*4]
	mov ecx, r9d
	cmp ecx, r11d
	cmova ecx, r11d
	test ecx, ecx
	jz lens_cmp
	mov rdi, r8
	mov rsi, r10
	repe cmpsb
	jne differ
lens_cmp:
	mov eax, r9d
	sub eax, r11d
	movsxd rax, eax
	ret
differ:
	movzx eax, byte ptr [rdi - 1]
	movzx edx, byte ptr [rsi - 1]
	sub eax, edx
	movsxd rax, eax
	ret
