.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDIN_FILENO, 0
.equ STDOUT_FILENO, 1
.equ BUFSIZE, 65536
.equ NEWLINE, 10

.text
.globl _start
_start:
	adrp x20, input
	add x20, x20, :lo12:input
	mov x21, x20
	mov x22, BUFSIZE
1:
	cbz x22, 2f
	mov x0, STDIN_FILENO
	mov x1, x21
	mov x2, x22
	mov x8, SYS_read
	svc #0
	cbz x0, 2f
	add x21, x21, x0
	sub x22, x22, x0
	b 1b
2:
	mov x23, x20
3:
	cmp x23, x21
	b.hs 9f
	mov x24, x23
4:
	cmp x24, x21
	b.hs 5f
	ldrb w0, [x24]
	cmp w0, NEWLINE
	b.eq 6f
	add x24, x24, #1
	b 4b
5:
	mov x25, x21
	mov x26, x21
	b 7f
6:
	mov x25, x24
	add x26, x24, #1
7:
	mov x0, x23
	sub x1, x25, #1
8:
	cmp x0, x1
	b.hs 10f
	ldrb w2, [x0]
	ldrb w3, [x1]
	strb w3, [x0]
	strb w2, [x1]
	add x0, x0, #1
	sub x1, x1, #1
	b 8b
10:
	sub x2, x25, x23
	cbz x2, 11f
	mov x0, STDOUT_FILENO
	mov x1, x23
	mov x8, SYS_write
	svc #0
11:
	mov x0, STDOUT_FILENO
	adrp x1, nl
	add x1, x1, :lo12:nl
	mov x2, #1
	mov x8, SYS_write
	svc #0
	mov x23, x26
	b 3b
9:
	mov x0, #0
	mov x8, SYS_exit
	svc #0

.section .rodata
nl:
	.byte NEWLINE

.bss
.align 3
input:
	.space BUFSIZE
