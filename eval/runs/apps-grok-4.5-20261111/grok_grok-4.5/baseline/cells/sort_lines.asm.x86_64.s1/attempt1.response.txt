.globl _start

.equ BUFSIZE, 32768
.equ MAX_LINES, 1000

.text
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, BUFSIZE
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    xor r13d, r13d
    xor r14d, r14d
    xor ecx, ecx
parse_loop:
    cmp rcx, r12
    jae parse_check_last
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 10
    jne parse_cont
    cmp r13, MAX_LINES
    jae parse_skip_store
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    lea rsi, [rip + input_buf]
    add rsi, r14
    mov qword ptr [rdi], rsi
    mov rax, rcx
    sub rax, r14
    mov qword ptr [rdi + 8], rax
    inc r13
parse_skip_store:
    lea r14, [rcx + 1]
parse_cont:
    inc rcx
    jmp parse_loop
parse_check_last:
    cmp r14, r12
    jae parse_done
    cmp r13, MAX_LINES
    jae parse_done
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    lea rsi, [rip + input_buf]
    add rsi, r14
    mov qword ptr [rdi], rsi
    mov rax, r12
    sub rax, r14
    mov qword ptr [rdi + 8], rax
    inc r13
parse_done:
    test r13, r13
    jz exit_ok
    mov r15, 1
sort_outer:
    cmp r15, r13
    jae sort_done
    mov rax, r15
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    sub rsp, 16
    mov rax, qword ptr [rsi]
    mov qword ptr [rsp], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rsp + 8], rax
    mov r14, r15
    dec r14
sort_inner:
    test r14, r14
    js sort_insert
    mov rax, r14
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    lea rsi, [rsp]
    call cmp_lines
    test rax, rax
    jle sort_insert
    mov rax, r14
    inc rax
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    mov rax, r14
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    dec r14
    jmp sort_inner
sort_insert:
    mov rax, r14
    inc rax
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    mov rax, qword ptr [rsp]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [rdi + 8], rax
    add rsp, 16
    inc r15
    jmp sort_outer
sort_done:
    xor r15d, r15d
print_loop:
    cmp r15, r13
    jae print_done
    mov rax, r15
    shl rax, 4
    lea rbx, [rip + lines]
    add rbx, rax
    mov rsi, qword ptr [rbx]
    mov rdx, qword ptr [rbx + 8]
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r15
    jmp print_loop
print_done:
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov rcx, r9
    cmp rcx, r11
    cmova rcx, r11
    xor edx, edx
cmp_loop:
    cmp rdx, rcx
    jae cmp_len
    movzx eax, byte ptr [r8 + rdx]
    movzx r12d, byte ptr [r10 + rdx]
    cmp eax, r12d
    jb cmp_lt
    ja cmp_gt
    inc rdx
    jmp cmp_loop
cmp_len:
    cmp r9, r11
    jb cmp_lt
    ja cmp_gt
    xor eax, eax
    ret
cmp_lt:
    mov rax, -1
    ret
cmp_gt:
    mov rax, 1
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
input_buf:
    .space BUFSIZE
.align 8
lines:
    .space MAX_LINES * 16
