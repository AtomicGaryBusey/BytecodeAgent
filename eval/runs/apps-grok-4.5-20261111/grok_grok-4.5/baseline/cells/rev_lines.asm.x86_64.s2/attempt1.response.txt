.globl _start

.text
_start:
    xor r12, r12
    lea r13, [rip + buffer]
.read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .done_read
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.done_read:
    xor r14, r14
.process_loop:
    cmp r14, r12
    jae .exit
    mov r15, r14
.find_nl:
    cmp r14, r12
    jae .found_end
    cmp byte ptr [r13 + r14], 0x0A
    je .found_end
    inc r14
    jmp .find_nl
.found_end:
    cmp r14, r15
    je .write_line
    mov rax, r15
    mov rbx, r14
    dec rbx
.rev_loop:
    cmp rax, rbx
    jge .rev_done
    mov cl, byte ptr [r13 + rax]
    mov dl, byte ptr [r13 + rbx]
    mov byte ptr [r13 + rax], dl
    mov byte ptr [r13 + rbx], cl
    inc rax
    dec rbx
    jmp .rev_loop
.rev_done:
.write_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [r13 + r15]
    mov rdx, r14
    sub rdx, r15
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    cmp r14, r12
    jae .exit
    inc r14
    jmp .process_loop
.exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
nl:
    .byte 0x0A

.bss
.align 16
buffer:
    .zero 65536
