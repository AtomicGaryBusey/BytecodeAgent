.text
.globl _start

.equ BUFSIZE, 65536

format_hex:
    mov x2, x1
1:
    sub x2, x2, #1
    lsl x3, x2, #2
    lsr x3, x0, x3
    and x3, x3, #0xf
    ldrb w3, [x28, x3]
    strb w3, [x26], #1
    cbnz x2, 1b
    ret

_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0
    adrp x28, hexdigits
    add x28, x28, :lo12:hexdigits

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

print_loop:
    cmp x21, x20
    b.hs print_final
    sub x0, x20, x21
    mov x1, #16
    cmp x0, #16
    csel x24, x0, x1, lo
    adrp x25, linebuf
    add x25, x25, :lo12:linebuf
    mov x26, x25
    mov x0, x21
    mov x1, #8
    bl format_hex
    mov w0, #' '
    strb w0, [x26], #1
    strb w0, [x26], #1
    mov x27, #0

hex_slots:
    cmp x27, #16
    b.ge hex_slots_done
    cmp x27, x24
    b.ge do_pad
    add x3, x19, x21
    ldrb w0, [x3, x27]
    mov x1, #2
    bl format_hex
    mov w0, #' '
    strb w0, [x26], #1
    b after_slot
do_pad:
    mov w0, #' '
    strb w0, [x26], #1
    strb w0, [x26], #1
    strb w0, [x26], #1
after_slot:
    cmp x27, #7
    b.ne no_extra
    mov w0, #' '
    strb w0, [x26], #1
no_extra:
    add x27, x27, #1
    b hex_slots

hex_slots_done:
    mov w0, #' '
    strb w0, [x26], #1
    mov w0, #'|'
    strb w0, [x26], #1
    mov x27, #0

ascii_loop:
    cmp x27, x24
    b.ge ascii_done
    add x3, x19, x21
    ldrb w0, [x3, x27]
    cmp w0, #0x20
    b.lo to_dot
    cmp w0, #0x7f
    b.hs to_dot
    b store_asc
to_dot:
    mov w0, #'.'
store_asc:
    strb w0, [x26], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w0, #'|'
    strb w0, [x26], #1
    mov w0, #'\n'
    strb w0, [x26], #1
    mov x0, #1
    mov x1, x25
    sub x2, x26, x25
    mov x8, #64
    svc #0
    add x21, x21, #16
    b print_loop

print_final:
    adrp x25, linebuf
    add x25, x25, :lo12:linebuf
    mov x26, x25
    mov x0, x20
    mov x1, #8
    bl format_hex
    mov w0, #'\n'
    strb w0, [x26], #1
    mov x0, #1
    mov x1, x25
    sub x2, x26, x25
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
buffer:
    .space BUFSIZE
linebuf:
    .space 80
