.globl _start

.text
_start:
	lea	r12, [rip + buffer]
	xor	r13, r13

read_more:
	mov	eax, 0
	xor	edi, edi
	mov	rsi, r12
	mov	rdx, 65536
	sub	rdx, r13
	jz	read_done
	syscall
	test	rax, rax
	jle	read_done
	add	r12, rax
	add	r13, rax
	jmp	read_more

read_done:
	lea	r14, [rip + buffer]
	xor	r15, r15

row_loop:
	cmp	r15, r13
	jae	final_line

	mov	r8, r13
	sub	r8, r15
	cmp	r8, 16
	jbe	1f
	mov	r8d, 16
1:
	lea	rdi, [rip + linebuf]

	mov	eax, r15d
	mov	ebx, 8
offset_hex:
	rol	eax, 4
	mov	edx, eax
	and	edx, 0xf
	mov	dl, byte ptr [hexdigits + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ebx
	jnz	offset_hex

	mov	word ptr [rdi], 0x2020
	add	rdi, 2

	xor	ebx, ebx
hex_bytes:
	cmp	ebx, r8d
	jae	hex_pad

	movzx	eax, byte ptr [r14 + rbx]
	mov	edx, eax
	shr	edx, 4
	mov	dl, byte ptr [hexdigits + rdx]
	mov	[rdi], dl
	mov	edx, eax
	and	edx, 0xf
	mov	dl, byte ptr [hexdigits + rdx]
	mov	[rdi + 1], dl
	mov	byte ptr [rdi + 2], 0x20
	add	rdi, 3
	jmp	hex_after

hex_pad:
	mov	byte ptr [rdi], 0x20
	mov	byte ptr [rdi + 1], 0x20
	mov	byte ptr [rdi + 2], 0x20
	add	rdi, 3

hex_after:
	cmp	ebx, 7
	jne	hex_next
	mov	byte ptr [rdi], 0x20
	inc	rdi

hex_next:
	inc	ebx
	cmp	ebx, 16
	jb	hex_bytes

	mov	byte ptr [rdi], 0x20
	mov	byte ptr [rdi + 1], '|'
	add	rdi, 2

	xor	ebx, ebx
ascii_loop:
	cmp	ebx, r8d
	jae	ascii_done
	movzx	eax, byte ptr [r14 + rbx]
	cmp	al, 0x20
	jb	is_dot
	cmp	al, 0x7e
	ja	is_dot
	mov	[rdi], al
	jmp	ascii_next
is_dot:
	mov	byte ptr [rdi], '.'
ascii_next:
	inc	rdi
	inc	ebx
	jmp	ascii_loop

ascii_done:
	mov	byte ptr [rdi], '|'
	mov	byte ptr [rdi + 1], 10
	add	rdi, 2

	lea	rsi, [rip + linebuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1
	mov	edi, 1
	syscall

	add	r14, r8
	add	r15, r8
	jmp	row_loop

final_line:
	lea	rdi, [rip + linebuf]
	mov	eax, r15d
	mov	ebx, 8
final_hex:
	rol	eax, 4
	mov	edx, eax
	and	edx, 0xf
	mov	dl, byte ptr [hexdigits + rdx]
	mov	[rdi], dl
	inc	rdi
	dec	ebx
	jnz	final_hex
	mov	byte ptr [rdi], 10

	mov	eax, 1
	mov	edi, 1
	lea	rsi, [rip + linebuf]
	mov	edx, 9
	syscall

	mov	eax, 60
	xor	edi, edi
	syscall

.section .rodata
hexdigits:
	.ascii	"0123456789abcdef"

.bss
.align	16
buffer:
	.zero	65536
linebuf:
	.zero	80
