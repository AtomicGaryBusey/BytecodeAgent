.globl _start
.text
_start:
    lea r13, [rip + buf]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz done_read
    lea rsi, [r13 + r12]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    mov r14, r13
    lea r15, [r13 + r12]
process_loop:
    cmp r14, r15
    jae finish
    mov rbx, r14
find_loop:
    cmp rbx, r15
    jae no_newline
    cmp byte ptr [rbx], 10
    je got_newline
    inc rbx
    jmp find_loop
got_newline:
    mov rcx, rbx
    sub rcx, r14
    mov rsi, r14
    mov rdi, rbx
    dec rdi
rev_loop:
    cmp rsi, rdi
    jae do_write
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev_loop
do_write:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, rcx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    lea r14, [rbx + 1]
    jmp process_loop
no_newline:
    mov rcx, r15
    sub rcx, r14
    mov rsi, r14
    mov rdi, r15
    dec rdi
rev_loop2:
    cmp rsi, rdi
    jae write_last
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev_loop2
write_last:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, rcx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata
nl:
    .byte 10
.bss
.align 16
buf:
    .zero 65536
