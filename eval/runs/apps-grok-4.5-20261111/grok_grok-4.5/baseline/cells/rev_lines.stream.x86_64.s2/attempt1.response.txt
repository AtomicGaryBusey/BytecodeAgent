.globl _start
.text
_start:
	lea r14, [rip + buffer]
	xor rbx, rbx
read_loop:
	mov rdx, 65536
	sub rdx, rbx
	jz done_read
	mov rax, 0
	xor rdi, rdi
	lea rsi, [r14 + rbx]
	syscall
	test rax, rax
	jle done_read
	add rbx, rax
	jmp read_loop
done_read:
	xor r12, r12
process_loop:
	cmp r12, rbx
	jge done_all
	mov r13, r12
find_loop:
	cmp r13, rbx
	jge do_reverse
	cmp byte ptr [r14 + r13], 0x0a
	je do_reverse
	inc r13
	jmp find_loop
do_reverse:
	mov rcx, r12
	mov rdx, r13
	dec rdx
rev_loop:
	cmp rcx, rdx
	jge rev_done
	mov al, byte ptr [r14 + rcx]
	mov r8b, byte ptr [r14 + rdx]
	mov byte ptr [r14 + rcx], r8b
	mov byte ptr [r14 + rdx], al
	inc rcx
	dec rdx
	jmp rev_loop
rev_done:
	mov rax, 1
	mov rdi, 1
	lea rsi, [r14 + r12]
	mov rdx, r13
	sub rdx, r12
	syscall
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + nl]
	mov rdx, 1
	syscall
	cmp r13, rbx
	jge done_all
	lea r12, [r13 + 1]
	jmp process_loop
done_all:
	mov rax, 60
	xor rdi, rdi
	syscall
