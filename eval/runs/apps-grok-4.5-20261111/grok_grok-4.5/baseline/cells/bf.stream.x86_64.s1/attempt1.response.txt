.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NEST, 1000

.text
_start:
  lea rsi, [rip + input_buf]
  mov r8, rsi
  mov r9d, BUF_SIZE
read_loop:
  test r9, r9
  jz read_done
  xor eax, eax
  xor edi, edi
  mov rdx, r9
  syscall
  test rax, rax
  jle read_done
  add rsi, rax
  sub r9, rax
  jmp read_loop
read_done:
  mov r13, rsi
  mov rdi, r8
find_bang:
  cmp rdi, r13
  jae no_bang
  cmp byte ptr [rdi], '!'
  je found_bang
  inc rdi
  jmp find_bang
found_bang:
  mov rbp, rdi
  lea r12, [rdi + 1]
  jmp check_brackets
no_bang:
  mov rbp, r13
  mov r12, r13
check_brackets:
  mov rbx, r8
  xor ecx, ecx
  lea rdx, [rip + bracket_stack]
  mov rsi, rbx
br_loop:
  cmp rsi, rbp
  jae br_done
  movzx eax, byte ptr [rsi]
  cmp al, '['
  je br_open
  cmp al, ']'
  je br_close
  inc rsi
  jmp br_loop
br_open:
  cmp ecx, MAX_NEST
  jae unbalanced
  mov qword ptr [rdx + rcx*8], rsi
  inc ecx
  inc rsi
  jmp br_loop
br_close:
  test ecx, ecx
  jz unbalanced
  dec ecx
  mov rdi, qword ptr [rdx + rcx*8]
  lea r10, [rip + bracket_map]
  mov r8, rdi
  sub r8, rbx
  lea r9, [rsi + 1]
  mov qword ptr [r10 + r8*8], r9
  mov r8, rsi
  sub r8, rbx
  lea r9, [rdi + 1]
  mov qword ptr [r10 + r8*8], r9
  inc rsi
  jmp br_loop
br_done:
  test ecx, ecx
  jnz unbalanced
  lea r14, [rip + tape]
  lea r15, [rip + bracket_map]
  mov rsi, rbx
interp:
  cmp rsi, rbp
  jae done_ok
  movzx eax, byte ptr [rsi]
  inc rsi
  cmp al, '>'
  je do_right
  cmp al, '<'
  je do_left
  cmp al, '+'
  je do_plus
  cmp al, '-'
  je do_minus
  cmp al, '.'
  je do_dot
  cmp al, ','
  je do_comma
  cmp al, '['
  je do_lbrack
  cmp al, ']'
  je do_rbrack
  jmp interp
do_right:
  inc r14
  jmp interp
do_left:
  dec r14
  jmp interp
do_plus:
  inc byte ptr [r14]
  jmp interp
do_minus:
  dec byte ptr [r14]
  jmp interp
do_dot:
  mov rax, SYS_write
  mov rdi, 1
  push rsi
  mov rsi, r14
  mov rdx, 1
  syscall
  pop rsi
  jmp interp
do_comma:
  cmp r12, r13
  jae comma_eof
  mov al, byte ptr [r12]
  mov byte ptr [r14], al
  inc r12
  jmp interp
comma_eof:
  mov byte ptr [r14], 0
  jmp interp
do_lbrack:
  cmp byte ptr [r14], 0
  jne interp
  lea rax, [rsi - 1]
  sub rax, rbx
  mov rsi, qword ptr [r15 + rax*8]
  jmp interp
do_rbrack:
  cmp byte ptr [r14], 0
  je interp
  lea rax, [rsi - 1]
  sub rax, rbx
  mov rsi, qword ptr [r15 + rax*8]
  jmp interp
done_ok:
  mov rax, SYS_exit
  xor edi, edi
  syscall
unbalanced:
  mov rax, SYS_exit
  mov edi, 1
  syscall

.bss
.align 16
input_buf:
  .zero BUF_SIZE
tape:
  .zero TAPE_SIZE
bracket_map:
  .zero BUF_SIZE * 8
bracket_stack:
  .zero MAX_NEST * 8
