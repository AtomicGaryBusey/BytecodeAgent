.globl _start

.equ BUFSIZE, 32768
.equ MAXLINES, 1000

.text
_start:
	lea r12, [rip + buffer]
	xor r13, r13
.read_loop:
	mov eax, 0
	xor edi, edi
	lea rsi, [r12 + r13]
	mov edx, BUFSIZE
	sub rdx, r13
	jbe .done_read
	syscall
	test rax, rax
	jle .done_read
	add r13, rax
	jmp .read_loop
.done_read:
	xor r14, r14
	xor r15, r15
	lea rbx, [rip + line_ptrs]
	lea rbp, [rip + line_lens]
.parse_loop:
	cmp r15, r13
	jae .parse_done
	mov rcx, r15
.find_nl:
	cmp rcx, r13
	jae .no_nl
	cmp byte ptr [r12 + rcx], 0x0A
	je .found_nl
	inc rcx
	jmp .find_nl
.found_nl:
	lea rax, [r12 + r15]
	mov qword ptr [rbx + r14*8], rax
	mov rax, rcx
	sub rax, r15
	mov qword ptr [rbp + r14*8], rax
	inc r14
	lea r15, [rcx + 1]
	jmp .parse_loop
.no_nl:
	lea rax, [r12 + r15]
	mov qword ptr [rbx + r14*8], rax
	mov rax, r13
	sub rax, r15
	mov qword ptr [rbp + r14*8], rax
	inc r14
.parse_done:
	cmp r14, 2
	jb .print_all
	mov r8, r14
	dec r8
.outer:
	xor r9, r9
	mov r10, r8
.inner:
	mov rsi, qword ptr [rbx + r9*8]
	mov rdx, qword ptr [rbp + r9*8]
	mov rdi, qword ptr [rbx + r9*8 + 8]
	mov rcx, qword ptr [rbp + r9*8 + 8]
	mov r11, rdx
	cmp r11, rcx
	cmova r11, rcx
	xor eax, eax
.cmp_loop:
	cmp rax, r11
	jae .cmp_len
	movzx r15d, byte ptr [rsi + rax]
	movzx r13d, byte ptr [rdi + rax]
	cmp r15d, r13d
	jb .no_swap
	ja .do_swap
	inc rax
	jmp .cmp_loop
.cmp_len:
	cmp rdx, rcx
	jbe .no_swap
.do_swap:
	mov qword ptr [rbx + r9*8], rdi
	mov qword ptr [rbx + r9*8 + 8], rsi
	mov qword ptr [rbp + r9*8], rcx
	mov qword ptr [rbp + r9*8 + 8], rdx
.no_swap:
	inc r9
	cmp r9, r10
	jb .inner
	dec r8
	jnz .outer
.print_all:
	xor r15, r15
.print_loop:
	cmp r15, r14
	jae .exit
	mov rsi, qword ptr [rbx + r15*8]
	mov rdx, qword ptr [rbp + r15*8]
	mov eax, 1
	mov edi, 1
	syscall
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + nl]
	mov edx, 1
	syscall
	inc r15
	jmp .print_loop
.exit:
	mov eax, 60
	xor edi, edi
	syscall

.section .rodata
nl:
	.byte 0x0A

.bss
.align 8
buffer:
	.zero BUFSIZE
line_ptrs:
	.zero MAXLINES * 8
line_lens:
	.zero MAXLINES * 8
