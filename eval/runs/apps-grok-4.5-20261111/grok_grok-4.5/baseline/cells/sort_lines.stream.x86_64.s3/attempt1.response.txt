.globl _start

.equ BUFSIZE, 32768
.equ MAXLINES, 1000

.text
_start:
	lea r12, [rip + buffer]
	xor r13, r13
.read_loop:
	mov rax, 0
	mov rdi, 0
	mov rsi, r12
	add rsi, r13
	mov rdx, BUFSIZE
	sub rdx, r13
	jbe .done_read
	syscall
	cmp rax, 0
	jle .done_read
	add r13, rax
	jmp .read_loop
.done_read:
	xor r14, r14
	xor r15, r15
.parse_loop:
	cmp r15, r13
	jae .parse_done
	mov rcx, r15
.find_nl:
	cmp rcx, r13
	jae .no_nl
	cmp byte ptr [r12 + rcx], 0x0A
	je .found_nl
	inc rcx
	jmp .find_nl
.found_nl:
	lea rax, [r12 + r15]
	mov qword ptr [rip + line_ptrs + r14*8], rax
	mov eax, ecx
	sub eax, r15d
	mov dword ptr [rip + line_lens + r14*4], eax
	inc r14
	lea r15, [rcx + 1]
	jmp .parse_loop
.no_nl:
	lea rax, [r12 + r15]
	mov qword ptr [rip + line_ptrs + r14*8], rax
	mov eax, r13d
	sub eax, r15d
	mov dword ptr [rip + line_lens + r14*4], eax
	inc r14
.parse_done:
	cmp r14, 2
	jb .print_lines
	xor ebx, ebx
.outer:
	mov eax, r14d
	dec eax
	cmp ebx, eax
	jae .sorted
	mov edx, r14d
	sub edx, ebx
	dec edx
	xor ebp, ebp
.inner:
	cmp ebp, edx
	jae .next_outer
	mov rdi, qword ptr [rip + line_ptrs + rbp*8]
	mov esi, dword ptr [rip + line_lens + rbp*4]
	mov rdx, qword ptr [rip + line_ptrs + rbp*8 + 8]
	mov ecx, dword ptr [rip + line_lens + rbp*4 + 4]
	call compare_lines
	test rax, rax
	jle .no_swap
	mov rax, qword ptr [rip + line_ptrs + rbp*8]
	mov rcx, qword ptr [rip + line_ptrs + rbp*8 + 8]
	mov qword ptr [rip + line_ptrs + rbp*8], rcx
	mov qword ptr [rip + line_ptrs + rbp*8 + 8], rax
	mov eax, dword ptr [rip + line_lens + rbp*4]
	mov ecx, dword ptr [rip + line_lens + rbp*4 + 4]
	mov dword ptr [rip + line_lens + rbp*4], ecx
	mov dword ptr [rip + line_lens + rbp*4 + 4], eax
.no_swap:
	inc ebp
	jmp .inner
.next_outer:
	inc ebx
	jmp .outer
.sorted:
.print_lines:
	xor ebx, ebx
.print_loop:
	cmp ebx, r14d
	jae .exit
	mov rax, 1
	mov rdi, 1
	mov rsi, qword ptr [rip + line_ptrs + rbx*8]
	mov edx, dword ptr [rip + line_lens + rbx*4]
	syscall
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + newline]
	mov edx, 1
	syscall
	inc ebx
	jmp .print_loop
.exit:
	mov rax, 60
	xor rdi, rdi
	syscall

compare_lines:
	mov r8d, esi
	cmp r8d, ecx
	cmova r8d, ecx
	xor eax, eax
	test r8d, r8d
	jz .cmp_lens
.cmp_bytes:
	movzx r9d, byte ptr [rdi]
	movzx r10d, byte ptr [rdx]
	cmp r9d, r10d
	jne .bytes_diff
	inc rdi
	inc rdx
	dec r8d
	jnz .cmp_bytes
.cmp_lens:
	cmp esi, ecx
	je .cmp_eq
	jb .cmp_lt
	mov eax, 1
	ret
.cmp_lt:
	mov eax, -1
	ret
.cmp_eq:
	xor eax, eax
	ret
.bytes_diff:
	cmp r9d, r10d
	jb .cmp_lt
	mov eax, 1
	ret

.section .rodata
newline:
	.byte 0x0A

.bss
.align 8
buffer:
	.zero BUFSIZE
line_ptrs:
	.zero MAXLINES*8
line_lens:
	.zero MAXLINES*4
