.globl _start
.text
_start:
	xor	r15, r15
	xor	r14, r14
	xor	r13, r13
	xor	ebx, ebx
read_loop:
	mov	eax, 0
	xor	edi, edi
	lea	rsi, [rip + buffer]
	mov	edx, 4096
	syscall
	test	rax, rax
	jle	done_reading
	add	r13, rax
	mov	rcx, rax
	lea	rsi, [rip + buffer]
process_loop:
	movzx	eax, byte ptr [rsi]
	cmp	al, 0x20
	je	is_ws
	cmp	al, 0x09
	jb	not_ws
	cmp	al, 0x0D
	ja	not_ws
is_ws:
	cmp	al, 0x0A
	jne	end_word
	inc	r15
end_word:
	test	ebx, ebx
	jz	next_byte
	xor	ebx, ebx
	jmp	next_byte
not_ws:
	test	ebx, ebx
	jnz	next_byte
	inc	r14
	mov	ebx, 1
next_byte:
	inc	rsi
	dec	rcx
	jnz	process_loop
	jmp	read_loop
done_reading:
	lea	rdi, [rip + outbuf]
	mov	rax, r15
	call	to_dec
	mov	byte ptr [rdi], ' '
	inc	rdi
	mov	rax, r14
	call	to_dec
	mov	byte ptr [rdi], ' '
	inc	rdi
	mov	rax, r13
	call	to_dec
	mov	byte ptr [rdi], 0x0A
	inc	rdi
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1
	mov	edi, 1
	syscall
	mov	eax, 60
	xor	edi, edi
	syscall
to_dec:
	cmp	rax, 0
	jne	1f
	mov	byte ptr [rdi], '0'
	inc	rdi
	ret
1:
	xor	ecx, ecx
	mov	rbx, 10
2:
	xor	edx, edx
	div	rbx
	add	dl, '0'
	push	rdx
	inc	ecx
	test	rax, rax
	jnz	2b
	mov	eax, ecx
3:
	pop	rdx
	mov	byte ptr [rdi], dl
	inc	rdi
	dec	ecx
	jnz	3b
	ret
.bss
.align 16
buffer:
	.space 4096
outbuf:
	.space 64
