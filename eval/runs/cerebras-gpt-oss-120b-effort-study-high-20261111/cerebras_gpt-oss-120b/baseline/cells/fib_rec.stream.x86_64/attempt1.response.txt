fib_rec:
    push rbx
    cmp rdi, 0
    je .Lzero
    cmp rdi, 1
    je .Lone
    sub rsp, 16
    mov qword ptr [rsp+8], rdi
    sub rdi, 1
    call fib_rec
    mov qword ptr [rsp], rax
    mov rdi, qword ptr [rsp+8]
    sub rdi, 2
    call fib_rec
    add rax, qword ptr [rsp]
    add rsp, 16
    pop rbx
    ret
.Lzero:
    mov rax, 0
    pop rbx
    ret
.Lone:
    mov rax, 1
    pop rbx
    ret
