binary_search:
    test rsi, rsi
    je .Lnot_found
    xor rcx, rcx
    mov r8, rsi
    dec r8
.Lloop:
    cmp rcx, r8
    jg .Lnot_found
    mov r9, r8
    sub r9, rcx
    shr r9, 1
    add r9, rcx
    mov rax, qword ptr [rdi + r9*8]
    cmp rax, rdx
    je .Lfound
    jl .Llow
    mov r8, r9
    dec r8
    jmp .Lloop
.Llow:
    lea rcx, [r9 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnot_found:
    mov rax, -1
    ret
