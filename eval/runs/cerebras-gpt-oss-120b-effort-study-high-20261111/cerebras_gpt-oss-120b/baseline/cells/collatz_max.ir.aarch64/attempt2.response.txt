export function l $collatz_max(l %n) {
@entry
    %limit = add %n $1
    %i0 = add $0 $1
    %max0 = add $0 $0
    jmp @loop(%i0, %max0)
@loop(%i, %max)
    %cond = slt %i %limit
    jnz %cond, @step(%i, %max, %i, $0)
    jmp @exit
@step(%i, %max, %x, %steps)
    %eq_one = eq %x $1
    jnz %eq_one, @step_done(%i, %max, %steps)
    %lsb = and %x $1
    %even = eq %lsb $0
    jnz %even, @even_case(%i, %max, %x, %steps)
    jmp @odd_case(%i, %max, %x, %steps)
@even_case(%i, %max, %x, %steps)
    %x1 = shr %x $1
    %steps1 = add %steps $1
    jmp @step(%i, %max, %x1, %steps1)
@odd_case(%i, %max, %x, %steps)
    %temp = mul %x $3
    %x1 = add %temp $1
    %steps1 = add %steps $1
    jmp @step(%i, %max, %x1, %steps1)
@step_done(%i, %max, %steps)
    %gt = slt %max %steps
    jnz %gt, @max_update(%i, %steps)
    jmp @max_keep(%i, %max)
@max_update(%i, %newmax)
    jmp @after_max(%i, %newmax)
@max_keep(%i, %max)
    jmp @after_max(%i, %max)
@after_max(%i, %max)
    %i_next = add %i $1
    jmp @loop(%i_next, %max)
@exit
    ret %max
}
