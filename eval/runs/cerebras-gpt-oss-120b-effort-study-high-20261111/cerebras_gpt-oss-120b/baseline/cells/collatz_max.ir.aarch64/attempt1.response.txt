export function l $collatz_max(l %n) {
    @entry
        %i0 = addl $0, $1
        %max0 = addl $0, $0
        jmp @loop_head(%i0, %max0)

    @loop_head
        %i = phi %i0, %i_next
        %max = phi %max0, %max_next
        %n1 = addl %n, $1
        %cond = csltl %i, %n1
        jnz %cond, @init_steps
        jmp @exit

    @init_steps
        %x0 = %i
        %steps0 = addl $0, $0
        jmp @step_test(%x0, %steps0)

    @step_test
        %x = phi %x0, %x_even, %x_odd
        %steps = phi %steps0, %steps_even, %steps_odd
        %eq_one = ceql %x, $1
        jnz %eq_one, @step_done
        %lsb = andl %x, $1
        %even = ceql %lsb, $0
        jnz %even, @even_case
        jmp @odd_case

    @even_case
        %one = addl $0, $1
        %x_even = shrl %x, %one
        %steps_even = addl %steps, $1
        jmp @step_test(%x_even, %steps_even)

    @odd_case
        %temp = mull %x, $3
        %x_odd = addl %temp, $1
        %steps_odd = addl %steps, $1
        jmp @step_test(%x_odd, %steps_odd)

    @step_done
        %gt = csltl %max, %steps
        jnz %gt, @max_update
        jmp @max_keep

    @max_update
        jmp @after_max_compare(%steps)

    @max_keep
        jmp @after_max_compare(%max)

    @after_max_compare
        %max_next = phi %steps, %max
        %i_next = addl %i, $1
        jmp @loop_head(%i_next, %max_next)

    @exit
        ret %max
}
