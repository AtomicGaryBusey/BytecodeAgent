export function l $qsort_pairs(l %a, l %n) {
@entry:
    %c0 = csltl %n, 2
    jnz %c0, @ret, @init

@ret:
    ret 0

@init:
    %high = subl %n, 1
    %pivot_off = shll %high, 4
    %pivot_ptr = addl %a, %pivot_off
    %pivot_key = loadl %pivot_ptr
    %pivot_val_addr = addl %pivot_ptr, 8
    %pivot_val = loadl %pivot_val_addr
    jmp @loop

@loop:
    %i = phi [ @init, 0 ], [ @swap, %i_next_s ], [ @noswap, %i_next_n ]
    %j = phi [ @init, 0 ], [ @swap, %j_next_s ], [ @noswap, %j_next_n ]
    %c_j = csltl %j, %high
    jnz %c_j, @loop_body, @post_loop

@loop_body:
    %j_off = shll %j, 4
    %addr_j = addl %a, %j_off
    %key_j = loadl %addr_j
    %addr_j_val = addl %addr_j, 8
    %val_j = loadl %addr_j_val

    %c_key_lt = csltl %key_j, %pivot_key
    %c_key_eq = ceql %key_j, %pivot_key
    %c_val_lt = csltl %val_j, %pivot_val
    %c_val_eq = ceql %val_j, %pivot_val
    %c_val_le = orw %c_val_lt, %c_val_eq
    %c_key_eq_and_val_le = andw %c_key_eq, %c_val_le
    %c_swap = orw %c_key_lt, %c_key_eq_and_val_le

    jnz %c_swap, @swap, @noswap

@swap:
    %i_off_s = shll %i, 4
    %addr_i_s = addl %a, %i_off_s
    %key_i_s = loadl %addr_i_s
    %addr_i_val_s = addl %addr_i_s, 8
    %val_i_s = loadl %addr_i_val_s

    storel %key_i_s, %addr_j
    storel %val_i_s, %addr_j_val
    storel %key_j, %addr_i_s
    storel %val_j, %addr_i_val_s

    %i_next_s = addl %i, 1
    %j_next_s = addl %j, 1
    jmp @loop

@noswap:
    %i_next_n = %i
    %j_next_n = addl %j, 1
    jmp @loop

@post_loop:
    %i_off = shll %i, 4
    %addr_i = addl %a, %i_off
    %key_i = loadl %addr_i
    %addr_i_val = addl %addr_i, 8
    %val_i = loadl %addr_i_val

    storel %pivot_key, %addr_i
    storel %pivot_val, %addr_i_val
    storel %key_i, %pivot_ptr
    storel %val_i, %pivot_val_addr

    %c_left = csltl 0, %i
    jnz %c_left, @left_call, @right_check

@left_call:
    %r_left = call $qsort_pairs(l %a, l %i)
    jmp @right_check

@right_check:
    %p1 = addl %i, 1
    %right_sz = subl %n, %p1
    %c_right = csltl 0, %right_sz
    jnz %c_right, @right_call, @end

@right_call:
    %right_off = shll %p1, 4
    %right_base = addl %a, %right_off
    %r_right = call $qsort_pairs(l %right_base, l %right_sz)
    jmp @end

@end:
    ret 0
}
