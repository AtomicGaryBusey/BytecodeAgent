export function l $qsort_pairs(l %a, l %n) {
@entry:
    %c0 = csltl %n, 1
    jnz %c0, %ret0, %init

%ret0:
    ret 0

%init:
    %high = subl %n, 1
    %pivot_off = shll %high, 4
    %pivot_rec = addl %a, %pivot_off
    %pivot_key = loadl %pivot_rec
    %tmp1 = addl %pivot_rec, 8
    %pivot_val = loadl %tmp1
    jmp %loop

%loop:
    %i = phi [ 0, %init ], [ %i_next, %swap ], [ %i_next, %noswap ]
    %j = phi [ 0, %init ], [ %j_next, %swap ], [ %j_next, %noswap ]
    %c_j = csltl %j, %high
    jnz %c_j, %loop_body, %post_loop

%loop_body:
    %j_off = shll %j, 4
    %addr_j = addl %a, %j_off
    %key_j = loadl %addr_j
    %valj_addr = addl %addr_j, 8
    %val_j = loadl %valj_addr

    %i_off = shll %i, 4
    %addr_i = addl %a, %i_off

    %c_key_lt = csltl %key_j, %pivot_key
    %c_key_eq = ceql %key_j, %pivot_key
    %c_val_lt = csltl %val_j, %pivot_val
    %c_val_eq = ceql %val_j, %pivot_val
    %c_val_le = orw %c_val_lt, %c_val_eq
    %c_key_eq_and_val_le = andw %c_key_eq, %c_val_le
    %c_swap = orw %c_key_lt, %c_key_eq_and_val_le

    jnz %c_swap, %swap, %noswap

%swap:
    %key_i = loadl %addr_i
    %vali_addr = addl %addr_i, 8
    %val_i = loadl %vali_addr

    storel %key_i, %addr_j
    storel %val_i, %valj_addr

    storel %key_j, %addr_i
    storel %val_j, %vali_addr

    %i_next = addl %i, 1
    %j_next = addl %j, 1
    jmp %loop

%noswap:
    %i_next = %i
    %j_next = addl %j, 1
    jmp %loop

%post_loop:
    %i_off2 = shll %i, 4
    %addr_i2 = addl %a, %i_off2
    %key_i2 = loadl %addr_i2
    %vali2_addr = addl %addr_i2, 8
    %val_i2 = loadl %vali2_addr

    storel %pivot_key, %addr_i2
    storel %pivot_val, %vali2_addr

    storel %key_i2, %pivot_rec
    storel %val_i2, %tmp1

    %pivot_idx = %i

    %c_left = csltl 0, %pivot_idx
    jnz %c_left, %call_left, %after_left

%call_left:
    %r_left =l call $qsort_pairs(l %a, l %pivot_idx)
    jmp %after_left

%after_left:
    %p1 = addl %pivot_idx, 1
    %right_sz = subl %n, %p1
    %c_right = csltl 0, %right_sz
    jnz %c_right, %call_right, %end

%call_right:
    %right_off = shll %p1, 4
    %right_base = addl %a, %right_off
    %r_right =l call $qsort_pairs(l %right_base, l %right_sz)
    jmp %end

%end:
    ret 0
}
