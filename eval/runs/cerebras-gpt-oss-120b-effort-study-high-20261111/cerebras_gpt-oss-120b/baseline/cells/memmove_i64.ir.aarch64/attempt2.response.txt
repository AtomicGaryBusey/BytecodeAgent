export function l $memmove_i64(l %dst, l %src, l %n) {
@entry:
    %bytes = shl %n, 3
    %src_end = add %src, %bytes
    %dst_end = add %dst, %bytes
    %dst_lt_src = csltl %dst, %src
    %dst_ge_src_end = csgel %dst, %src_end
    %forward = or %dst_lt_src, %dst_ge_src_end
    jnz %forward, @Lforward

@Lbackward:
    %src_ptr = sub %src_end, 8
    %dst_ptr = sub %dst_end, 8
    %cnt = %n
    jmp @Lbackward_loop

@Lbackward_loop:
    %cnt_eq0 = ceql %cnt, 0
    jnz %cnt_eq0, @Lret
    %val = loadl %src_ptr
    storel %val, %dst_ptr
    %src_ptr = sub %src_ptr, 8
    %dst_ptr = sub %dst_ptr, 8
    %cnt = sub %cnt, 1
    jmp @Lbackward_loop

@Lforward:
    %src_ptr = add %src, 0
    %dst_ptr = add %dst, 0
    %cnt = %n
    jmp @Lforward_loop

@Lforward_loop:
    %cnt_eq0 = ceql %cnt, 0
    jnz %cnt_eq0, @Lret
    %val = loadl %src_ptr
    storel %val, %dst_ptr
    %src_ptr = add %src_ptr, 8
    %dst_ptr = add %dst_ptr, 8
    %cnt = sub %cnt, 1
    jmp @Lforward_loop

@Lret:
    ret 0
}
