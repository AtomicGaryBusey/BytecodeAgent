export function l $memmove_i64(l %dst, l %src, l %n) {
@entry:
    %bytes = shl %n, 3
    %src_end = add %src, %bytes
    %dst_end = add %dst, %bytes
    %dst_lt_src = csltl %dst, %src
    %dst_ge_src_end = csgel %dst, %src_end
    %forward = or %dst_lt_src, %dst_ge_src_end
    %src_back_initial = sub %src_end, 8
    %dst_back_initial = sub %dst_end, 8
    jnz %forward, @forward_loop
    jmp @backward_loop

@forward_loop:
    %src_f = phi %src, %src_next_f
    %dst_f = phi %dst, %dst_next_f
    %cnt_f = phi %n, %cnt_next_f
    %cond_f = ceql %cnt_f, 0
    jnz %cond_f, @ret
    %val_f = loadl %src_f
    storel %val_f, %dst_f
    %src_next_f = add %src_f, 8
    %dst_next_f = add %dst_f, 8
    %cnt_next_f = sub %cnt_f, 1
    jmp @forward_loop

@backward_loop:
    %src_b = phi %src_back_initial, %src_next_b
    %dst_b = phi %dst_back_initial, %dst_next_b
    %cnt_b = phi %n, %cnt_next_b
    %cond_b = ceql %cnt_b, 0
    jnz %cond_b, @ret
    %val_b = loadl %src_b
    storel %val_b, %dst_b
    %src_next_b = sub %src_b, 8
    %dst_next_b = sub %dst_b, 8
    %cnt_next_b = sub %cnt_b, 1
    jmp @backward_loop

@ret:
    ret 0
}
