export function l $bubble_sort(l %a, l %n) {
@entry:
    jmp @outer_head

@outer_head:
    %i = phi(l 0, %i_next)
    %n1 = subl %n, 1
    %c_i = csltl %i, %n1
    jnz %c_i, @inner_head
    jmp @ret

@inner_head:
    %j = phi(l 0, %j_next)
    %tmp1 = subl %n, %i
    %bound = subl %tmp1, 1
    %c_j = csltl %j, %bound
    jnz %c_j, @inner_body
    jmp @outer_inc

@inner_body:
    %j8 = mull %j, 8
    %addr = addl %a, %j8
    %addr1 = addl %addr, 8
    %val = loadl *%addr
    %val1 = loadl *%addr1
    %c_swap = csltl %val1, %val
    jnz %c_swap, @swap
    jmp @inner_inc

@swap:
    storel %val1, *%addr
    storel %val, *%addr1
    jmp @inner_inc

@inner_inc:
    %j_next = addl %j, 1
    jmp @inner_head

@outer_inc:
    %i_next = addl %i, 1
    jmp @outer_head

@ret:
    ret 0
}
