export function l $bubble_sort(p a, l n) {
@start:
    %i0 = 0
    jmp @outer(%i0)

@outer(%i):
    %n1 = subl %n, 1
    %c_i = csltl %i, %n1
    jnz %c_i, @inner(%i)
    jmp @ret

@inner(%i):
    %ni = subl %n, %i
    %bound = subl %ni, 1
    %j0 = 0
    jmp @inner_cond(%i, %j0, %bound)

@inner_cond(%i, %j, %bound):
    %c_j = csltl %j, %bound
    jnz %c_j, @inner_body(%i, %j, %bound)
    jmp @outer_next(%i)

@inner_body(%i, %j, %bound):
    %j8 = mull %j, 8
    %addr = addp %a, %j8
    %addr1 = addp %addr, 8
    %val = loadl *%addr
    %val1 = loadl *%addr1
    %c_swap = csltl %val1, %val
    jnz %c_swap, @swap(%i, %j, %bound)
    jmp @inner_inc(%i, %j, %bound)

@swap(%i, %j, %bound):
    %j8 = mull %j, 8
    %addr = addp %a, %j8
    %addr1 = addp %addr, 8
    %val = loadl *%addr
    %val1 = loadl *%addr1
    storel %val1, *%addr
    storel %val, *%addr1
    jmp @inner_inc(%i, %j, %bound)

@inner_inc(%i, %j, %bound):
    %j1 = addl %j, 1
    jmp @inner_cond(%i, %j1, %bound)

@outer_next(%i):
    %i1 = addl %i, 1
    jmp @outer(%i1)

@ret:
    ret 0
}
