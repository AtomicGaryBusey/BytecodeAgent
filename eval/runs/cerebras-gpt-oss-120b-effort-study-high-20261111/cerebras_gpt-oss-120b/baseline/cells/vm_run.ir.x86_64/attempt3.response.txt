export function l $vm_run(l code, l len, l stack, l cap) {
@entry
%pc0 = copy 0
%sp0 = copy 0
%zero = copy 0
jump @dispatch

@dispatch
%pc = phi %pc0, @entry,
           %pc_push, @push,
           %pc_add, @add,
           %pc_sub, @sub,
           %pc_mul, @mul,
           %pc_dup, @dup,
           %pc_swap, @swap,
           %pc_drop, @drop,
           %pc_over, @over,
           %pc_lt, @lt,
           %pc_jz0, @jz_zero,
           %pc_jz1, @jz_nonzero,
           %pc_jmp, @jmp,
           %pc_unknown, @unknown
%sp = phi %sp0, @entry,
           %sp_push, @push,
           %sp_add, @add,
           %sp_sub, @sub,
           %sp_mul, @mul,
           %sp_dup, @dup,
           %sp_swap, @swap,
           %sp_drop, @drop,
           %sp_over, @over,
           %sp_lt, @lt,
           %sp_jz0, @jz_zero,
           %sp_jz1, @jz_nonzero,
           %sp_jmp, @jmp,
           %sp_unknown, @unknown
%pc_lt_len = csltl %pc, %len
jnz %pc_lt_len, @fetch
jump @exit

@fetch
%pc_off = mul %pc, 8
%op_addr = add %code, %pc_off
%op = loadl %op_addr

%is_halt = ceql %op, 0
jnz %is_halt, @halt

%is_push = ceql %op, 1
jnz %is_push, @push

%is_add = ceql %op, 2
jnz %is_add, @add

%is_sub = ceql %op, 3
jnz %is_sub, @sub

%is_mul = ceql %op, 4
jnz %is_mul, @mul

%is_dup = ceql %op, 5
jnz %is_dup, @dup

%is_swap = ceql %op, 6
jnz %is_swap, @swap

%is_drop = ceql %op, 7
jnz %is_drop, @drop

%is_over = ceql %op, 8
jnz %is_over, @over

%is_lt = ceql %op, 9
jnz %is_lt, @lt

%is_jz = ceql %op, 10
jnz %is_jz, @jz

%is_jmp = ceql %op, 11
jnz %is_jmp, @jmp

jump @unknown

@halt
jump @exit

@push
%push_pc1 = add %pc, 1
%push_pc1_off = mul %push_pc1, 8
%push_operand_addr = add %code, %push_pc1_off
%push_operand = loadl %push_operand_addr
%push_sp_off = mul %sp, 8
%push_dst_addr = add %stack, %push_sp_off
storel %push_operand, %push_dst_addr
%sp_push = add %sp, 1
%pc_push = add %pc, 2
jump @dispatch

@add
%add_sp_m1 = sub %sp, 1
%add_sp_m2 = sub %sp, 2
%add_off_b = mul %add_sp_m1, 8
%add_addr_b = add %stack, %add_off_b
%add_off_a = mul %add_sp_m2, 8
%add_addr_a = add %stack, %add_off_a
%add_val_b = loadl %add_addr_b
%add_val_a = loadl %add_addr_a
%add_sum = add %add_val_a, %add_val_b
storel %add_sum, %add_addr_a
%sp_add = sub %sp, 1
%pc_add = add %pc, 1
jump @dispatch

@sub
%sub_sp_m1 = sub %sp, 1
%sub_sp_m2 = sub %sp, 2
%sub_off_b = mul %sub_sp_m1, 8
%sub_addr_b = add %stack, %sub_off_b
%sub_off_a = mul %sub_sp_m2, 8
%sub_addr_a = add %stack, %sub_off_a
%sub_val_b = loadl %sub_addr_b
%sub_val_a = loadl %sub_addr_a
%sub_diff = sub %sub_val_a, %sub_val_b
storel %sub_diff, %sub_addr_a
%sp_sub = sub %sp, 1
%pc_sub = add %pc, 1
jump @dispatch

@mul
%mul_sp_m1 = sub %sp, 1
%mul_sp_m2 = sub %sp, 2
%mul_off_b = mul %mul_sp_m1, 8
%mul_addr_b = add %stack, %mul_off_b
%mul_off_a = mul %mul_sp_m2, 8
%mul_addr_a = add %stack, %mul_off_a
%mul_val_b = loadl %mul_addr_b
%mul_val_a = loadl %mul_addr_a
%mul_prod = mul %mul_val_a, %mul_val_b
storel %mul_prod, %mul_addr_a
%sp_mul = sub %sp, 1
%pc_mul = add %pc, 1
jump @dispatch

@dup
%dup_sp_m1 = sub %sp, 1
%dup_off_top = mul %dup_sp_m1, 8
%dup_addr_top = add %stack, %dup_off_top
%dup_val_top = loadl %dup_addr_top
%dup_off_dst = mul %sp, 8
%dup_addr_dst = add %stack, %dup_off_dst
storel %dup_val_top, %dup_addr_dst
%sp_dup = add %sp, 1
%pc_dup = add %pc, 1
jump @dispatch

@swap
%swap_sp_m1 = sub %sp, 1
%swap_sp_m2 = sub %sp, 2
%swap_off_b = mul %swap_sp_m1, 8
%swap_addr_b = add %stack, %swap_off_b
%swap_off_a = mul %swap_sp_m2, 8
%swap_addr_a = add %stack, %swap_off_a
%swap_val_b = loadl %swap_addr_b
%swap_val_a = loadl %swap_addr_a
storel %swap_val_b, %swap_addr_a
storel %swap_val_a, %swap_addr_b
%sp_swap = copy %sp
%pc_swap = add %pc, 1
jump @dispatch

@drop
%sp_drop = sub %sp, 1
%pc_drop = add %pc, 1
jump @dispatch

@over
%over_sp_m2 = sub %sp, 2
%over_off_src = mul %over_sp_m2, 8
%over_addr_src = add %stack, %over_off_src
%over_val_src = loadl %over_addr_src
%over_off_dst = mul %sp, 8
%over_addr_dst = add %stack, %over_off_dst
storel %over_val_src, %over_addr_dst
%sp_over = add %sp, 1
%pc_over = add %pc, 1
jump @dispatch

@lt
%lt_sp_m1 = sub %sp, 1
%lt_sp_m2 = sub %sp, 2
%lt_off_b = mul %lt_sp_m1, 8
%lt_addr_b = add %stack, %lt_off_b
%lt_off_a = mul %lt_sp_m2, 8
%lt_addr_a = add %stack, %lt_off_a
%lt_val_b = loadl %lt_addr_b
%lt_val_a = loadl %lt_addr_a
%lt_cmp = csltl %lt_val_a, %lt_val_b
%lt_cmp_val = zextw %lt_cmp
storel %lt_cmp_val, %lt_addr_a
%sp_lt = sub %sp, 1
%pc_lt = add %pc, 1
jump @dispatch

@jz
%jz_sp_m1 = sub %sp, 1
%jz_off_v = mul %jz_sp_m1, 8
%jz_addr_v = add %stack, %jz_off_v
%jz_val_v = loadl %jz_addr_v
%jz_pc1 = add %pc, 1
%jz_pc1_off = mul %jz_pc1, 8
%jz_addr_t = add %code, %jz_pc1_off
%jz_t = loadl %jz_addr_t
%jz_is_zero = ceql %jz_val_v, 0
jnz %jz_is_zero, @jz_zero
jump @jz_nonzero

@jz_zero
%pc_jz0 = copy %jz_t
%sp_jz0 = copy %jz_sp_m1
jump @dispatch

@jz_nonzero
%pc_jz1 = add %pc, 2
%sp_jz1 = copy %jz_sp_m1
jump @dispatch

@jmp
%jmp_pc1 = add %pc, 1
%jmp_pc1_off = mul %jmp_pc1, 8
%jmp_addr_t = add %code, %jmp_pc1_off
%jmp_t = loadl %jmp_addr_t
%pc_jmp = copy %jmp_t
%sp_jmp = copy %sp
jump @dispatch

@unknown
%pc_unknown = add %pc, 1
%sp_unknown = copy %sp
jump @dispatch

@exit
jump @zero

@zero
%i = phi %sp, @exit,
           %i_next, @zero_body
%cond = csltl %i, %cap
jnz %cond, @zero_body
jump @final

@zero_body
%offset_i = mul %i, 8
%addr_i = add %stack, %offset_i
storel %zero, %addr_i
%i_next = add %i, 1
jump @zero

@final
%sp_nonzero = csltl 0, %sp
jnz %sp_nonzero, @ret_top
ret 0

@ret_top
%sp_m1 = sub %sp, 1
%offset = mul %sp_m1, 8
%addr = add %stack, %offset
%top_val = loadl %addr
ret %top_val
}
