export function l $vm_run(l *code, l len, l *stack, l cap) {
    @entry
    %pc0 = copy 0
    %sp0 = copy 0
    %zero_val = copy 0
    jump @L1

    @L1
    %pc = phi %pc0, @entry,
               %pc_push, @push,
               %pc_add, @add,
               %pc_sub, @sub,
               %pc_mul, @mul,
               %pc_dup, @dup,
               %pc_swap, @swap,
               %pc_drop, @drop,
               %pc_over, @over,
               %pc_lt, @lt,
               %pc_jz_zero, @jz_zero,
               %pc_jz_nonzero, @jz_nonzero,
               %pc_jmp, @jmp,
               %pc_unknown, @unknown
    %sp = phi %sp0, @entry,
               %sp_push, @push,
               %sp_add, @add,
               %sp_sub, @sub,
               %sp_mul, @mul,
               %sp_dup, @dup,
               %sp_swap, @swap,
               %sp_drop, @drop,
               %sp_over, @over,
               %sp_lt, @lt,
               %sp_jz0, @jz_zero,
               %sp_jz1, @jz_nonzero,
               %sp_jmp, @jmp,
               %sp_unknown, @unknown

    %pc_lt_len = csltl %pc, %len
    jnz %pc_lt_len, @fetch
    jump @exit

    @fetch
    %fetch_pc8 = mul %pc, 8
    %fetch_op_addr = add %code, %fetch_pc8
    %fetch_op = loadl %fetch_op_addr

    %fetch_is_halt = ceql %fetch_op, 0
    jnz %fetch_is_halt, @halt

    %fetch_is_push = ceql %fetch_op, 1
    jnz %fetch_is_push, @push

    %fetch_is_add = ceql %fetch_op, 2
    jnz %fetch_is_add, @add

    %fetch_is_sub = ceql %fetch_op, 3
    jnz %fetch_is_sub, @sub

    %fetch_is_mul = ceql %fetch_op, 4
    jnz %fetch_is_mul, @mul

    %fetch_is_dup = ceql %fetch_op, 5
    jnz %fetch_is_dup, @dup

    %fetch_is_swap = ceql %fetch_op, 6
    jnz %fetch_is_swap, @swap

    %fetch_is_drop = ceql %fetch_op, 7
    jnz %fetch_is_drop, @drop

    %fetch_is_over = ceql %fetch_op, 8
    jnz %fetch_is_over, @over

    %fetch_is_lt = ceql %fetch_op, 9
    jnz %fetch_is_lt, @lt

    %fetch_is_jz = ceql %fetch_op, 10
    jnz %fetch_is_jz, @jz

    %fetch_is_jmp = ceql %fetch_op, 11
    jnz %fetch_is_jmp, @jmp

    jump @unknown

    @halt
    jump @exit

    @push
    %push_pc8 = mul %pc, 8
    %push_pc1_8 = add %push_pc8, 8
    %push_v_addr = add %code, %push_pc1_8
    %push_v = loadl %push_v_addr
    %push_sp8 = mul %sp, 8
    %push_dst = add %stack, %push_sp8
    storel %push_v, %push_dst
    %sp_push = add %sp, 1
    %pc_push = add %pc, 2
    jump @L1

    @add
    %add_sp_m1 = sub %sp, 1
    %add_sp_m2 = sub %sp, 2
    %add_addr_b = add %stack, mul %add_sp_m1, 8
    %add_addr_a = add %stack, mul %add_sp_m2, 8
    %add_b = loadl %add_addr_b
    %add_a = loadl %add_addr_a
    %add_sum = add %add_a, %add_b
    storel %add_sum, %add_addr_a
    %sp_add = sub %sp, 1
    %pc_add = add %pc, 1
    jump @L1

    @sub
    %sub_sp_m1 = sub %sp, 1
    %sub_sp_m2 = sub %sp, 2
    %sub_addr_b = add %stack, mul %sub_sp_m1, 8
    %sub_addr_a = add %stack, mul %sub_sp_m2, 8
    %sub_b = loadl %sub_addr_b
    %sub_a = loadl %sub_addr_a
    %sub_diff = sub %sub_a, %sub_b
    storel %sub_diff, %sub_addr_a
    %sp_sub = sub %sp, 1
    %pc_sub = add %pc, 1
    jump @L1

    @mul
    %mul_sp_m1 = sub %sp, 1
    %mul_sp_m2 = sub %sp, 2
    %mul_addr_b = add %stack, mul %mul_sp_m1, 8
    %mul_addr_a = add %stack, mul %mul_sp_m2, 8
    %mul_b = loadl %mul_addr_b
    %mul_a = loadl %mul_addr_a
    %mul_prod = mul %mul_a, %mul_b
    storel %mul_prod, %mul_addr_a
    %sp_mul = sub %sp, 1
    %pc_mul = add %pc, 1
    jump @L1

    @dup
    %dup_sp_m1 = sub %sp, 1
    %dup_addr_top = add %stack, mul %dup_sp_m1, 8
    %dup_val = loadl %dup_addr_top
    %dup_sp8 = mul %sp, 8
    %dup_dst = add %stack, %dup_sp8
    storel %dup_val, %dup_dst
    %sp_dup = add %sp, 1
    %pc_dup = add %pc, 1
    jump @L1

    @swap
    %swap_sp_m1 = sub %sp, 1
    %swap_sp_m2 = sub %sp, 2
    %swap_addr_b = add %stack, mul %swap_sp_m1, 8
    %swap_addr_a = add %stack, mul %swap_sp_m2, 8
    %swap_b = loadl %swap_addr_b
    %swap_a = loadl %swap_addr_a
    storel %swap_b, %swap_addr_a
    storel %swap_a, %swap_addr_b
    %sp_swap = copy %sp
    %pc_swap = add %pc, 1
    jump @L1

    @drop
    %sp_drop = sub %sp, 1
    %pc_drop = add %pc, 1
    jump @L1

    @over
    %over_sp_m2 = sub %sp, 2
    %over_addr_src = add %stack, mul %over_sp_m2, 8
    %over_val = loadl %over_addr_src
    %over_sp8 = mul %sp, 8
    %over_dst = add %stack, %over_sp8
    storel %over_val, %over_dst
    %sp_over = add %sp, 1
    %pc_over = add %pc, 1
    jump @L1

    @lt
    %lt_sp_m1 = sub %sp, 1
    %lt_sp_m2 = sub %sp, 2
    %lt_addr_b = add %stack, mul %lt_sp_m1, 8
    %lt_addr_a = add %stack, mul %lt_sp_m2, 8
    %lt_b = loadl %lt_addr_b
    %lt_a = loadl %lt_addr_a
    %lt_cmp = csltl %lt_a, %lt_b
    %lt_cmp_l = zextw %lt_cmp
    storel %lt_cmp_l, %lt_addr_a
    %sp_lt = sub %sp, 1
    %pc_lt = add %pc, 1
    jump @L1

    @jz
    %jz_sp_pop = sub %sp, 1
    %jz_sp_m1 = sub %sp, 1
    %jz_addr_v = add %stack, mul %jz_sp_m1, 8
    %jz_v = loadl %jz_addr_v
    %jz_pc8 = mul %pc, 8
    %jz_pc1_8 = add %jz_pc8, 8
    %jz_t_addr = add %code, %jz_pc1_8
    %jz_t = loadl %jz_t_addr
    %jz_is_zero = ceql %jz_v, 0
    jnz %jz_is_zero, @jz_zero
    jump @jz_nonzero

    @jz_zero
    %pc_jz_zero = copy %jz_t
    %sp_jz0 = copy %jz_sp_pop
    jump @L1

    @jz_nonzero
    %pc_jz_nonzero = add %pc, 2
    %sp_jz1 = copy %jz_sp_pop
    jump @L1

    @jmp
    %jmp_pc8 = mul %pc, 8
    %jmp_pc1_8 = add %jmp_pc8, 8
    %jmp_t_addr = add %code, %jmp_pc1_8
    %jmp_t = loadl %jmp_t_addr
    %pc_jmp = copy %jmp_t
    %sp_jmp = copy %sp
    jump @L1

    @unknown
    %pc_unknown = add %pc, 1
    %sp_unknown = copy %sp
    jump @L1

    @exit
    jump @zero

    @zero
    %i = phi %sp, @exit, %i_next, @zero_body
    %zero_cond_i_lt_cap = csltl %i, %cap
    jnz %zero_cond_i_lt_cap, @zero_body
    jump @final

    @zero_body
    %i8 = mul %i, 8
    %zero_addr = add %stack, %i8
    storel %zero_val, %zero_addr
    %i_next = add %i, 1
    jump @zero

    @final
    %cond_sp_pos = csltl %zero_val, %sp
    jnz %cond_sp_pos, @ret_top
    ret %zero_val

    @ret_top
    %sp_m1_top = sub %sp, 1
    %offset_top = mul %sp_m1_top, 8
    %addr_top = add %stack, %offset_top
    %top_val = loadl %addr_top
    ret %top_val
}
