export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
  %zero = const l 0
  %one = const l 1
  %eight = const l 8
  %src_lt_dst = csltl %src, %dst
  %n8 = mul %n, %eight
  %src_end = add %src, %n8
  %dst_lt_src_end = csltl %dst, %src_end
  %overlap = and %src_lt_dst, %dst_lt_src_end
  jnz %overlap, @bwd
  jmp @fwd

@fwd
  %i_f = phi l %zero, %i_f_next
  %cond = csltl %i_f, %n
  jnz %cond, @fwd_body
  jmp @exit

@fwd_body
  %off = mul %i_f, %eight
  %src_addr = add %src, %off
  %dst_addr = add %dst, %off
  %val = loadl %src_addr
  storel %val, %dst_addr
  %i_f_next = add %i_f, %one
  jmp @fwd

@bwd
  %i_b = phi l %n, %i_b_next
  %condb = ceql %i_b, %zero
  jnz %condb, @exit
  jmp @bwd_body

@bwd_body
  %i_b_next = sub %i_b, %one
  %offb = mul %i_b_next, %eight
  %src_addrb = add %src, %offb
  %dst_addrb = add %dst, %offb
  %valb = loadl %src_addrb
  storel %valb, %dst_addrb
  jmp @bwd

@exit
  ret %zero
}
