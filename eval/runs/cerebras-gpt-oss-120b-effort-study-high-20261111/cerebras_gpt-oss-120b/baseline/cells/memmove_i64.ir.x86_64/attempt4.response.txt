export function l $memmove_i64(l %dst, l %src, l %n) {
@entry:
    %zero = iconst 0
    %one = iconst 1
    %eight = iconst 8
    %src_lt_dst = csltl %src, %dst
    %n8 = mul %n, %eight
    %src_end = add %src, %n8
    %dst_lt_src_end = csltl %dst, %src_end
    %overlap = and %src_lt_dst, %dst_lt_src_end
    jnz %overlap, @bwd_head
    jmp @fwd_head

@fwd_head:
    %i_f = phi l %zero, @entry, %i_next_f, @fwd_body
    %cond_f = csltl %i_f, %n
    jnz %cond_f, @fwd_body
    jmp @exit

@fwd_body:
    %off_f = mul %i_f, %eight
    %src_addr_f = add %src, %off_f
    %dst_addr_f = add %dst, %off_f
    %val_f = loadl %src_addr_f
    storel %val_f, %dst_addr_f
    %i_next_f = add %i_f, %one
    jmp @fwd_head

@bwd_head:
    %i_b = phi l %n, @entry, %i_next_b, @bwd_body
    %cond_b = csltl %zero, %i_b
    jnz %cond_b, @bwd_body
    jmp @exit

@bwd_body:
    %i_next_b = sub %i_b, %one
    %off_b = mul %i_next_b, %eight
    %src_addr_b = add %src, %off_b
    %dst_addr_b = add %dst, %off_b
    %val_b = loadl %src_addr_b
    storel %val_b, %dst_addr_b
    jmp @bwd_head

@exit:
    ret %zero
}
