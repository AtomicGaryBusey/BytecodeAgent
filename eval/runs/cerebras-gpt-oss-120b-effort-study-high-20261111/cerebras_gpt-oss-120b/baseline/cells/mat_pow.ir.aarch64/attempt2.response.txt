export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %zero = imm l 0
    %one = imm l 1
    %eight = imm l 8
    %nn = mull %n, %n
    %nn2 = addl %nn, %nn
    jmp @id_i_head

@id_i_head
    %i_id = phi l (@entry %zero) (@id_i_body %i_id_next)
    %i_id_lt_n = csltl %i_id, %n
    jnz %i_id_lt_n, @id_j_head
    jmp @exp_head

@id_j_head
    %j_id = phi l (@id_i_head %zero) (@id_j_body %j_id_next)
    %j_id_lt_n = csltl %j_id, %n
    jnz %j_id_lt_n, @id_j_body
    jmp @id_i_body

@id_j_body
    %i_n_id = mull %i_id, %n
    %idx_id = addl %i_n_id, %j_id
    %off_id = mull %idx_id, %eight
    %dst_id = addl %out, %off_id
    %eq_id = ceql %i_id, %j_id
    %val_id = sell %eq_id, %one, %zero
    storel %val_id, %dst_id
    %j_id_next = addl %j_id, %one
    jmp @id_j_head

@id_i_body
    %i_id_next = addl %i_id, %one
    jmp @id_i_head

@exp_head
    %cnt = phi l (@id_i_head %k) (@exp_body_finish %cnt_next)
    %cnt_gt0 = csltl %zero, %cnt
    jnz %cnt_gt0, @prod_i_head
    jmp @zero_head

@prod_i_head
    %i_prod = phi l (@exp_head %zero) (@prod_i_body %i_prod_next)
    %i_prod_lt_n = csltl %i_prod, %n
    jnz %i_prod_lt_n, @prod_j_head
    jmp @copy_head

@prod_i_body
    %i_prod_next = addl %i_prod, %one
    jmp @prod_i_head

@prod_j_head
    %j_prod = phi l (@prod_i_head %zero) (@prod_store_sum %j_prod_next)
    %j_prod_lt_n = csltl %j_prod, %n
    jnz %j_prod_lt_n, @prod_t_head
    jmp @prod_i_body

@prod_t_head
    %t = phi l (@prod_j_head %zero) (@prod_t_body %t_next)
    %sum = phi l (@prod_j_head %zero) (@prod_t_body %sum_next)
    %t_lt_n = csltl %t, %n
    jnz %t_lt_n, @prod_t_compute
    jmp @prod_store_sum

@prod_t_compute
    %i_n_prod = mull %i_prod, %n
    %idx_out = addl %i_n_prod, %t
    %off_out = mull %idx_out, %eight
    %ptr_out = addl %out, %off_out
    %val_out = loadl %ptr_out
    %t_n = mull %t, %n
    %idx_a = addl %t_n, %j_prod
    %off_a = mull %idx_a, %eight
    %ptr_a = addl %a, %off_a
    %val_a = loadl %ptr_a
    %term = mull %val_out, %val_a
    jmp @prod_t_body

@prod_t_body
    %sum_next = addl %sum, %term
    %t_next = addl %t, %one
    jmp @prod_t_head

@prod_store_sum
    %i_n_ps = mull %i_prod, %n
    %idx_tmp = addl %i_n_ps, %j_prod
    %off_tmp = mull %idx_tmp, %eight
    %ptr_tmp = addl %tmp, %off_tmp
    storel %sum, %ptr_tmp
    %j_prod_next = addl %j_prod, %one
    jmp @prod_j_head

@copy_head
    %idx = phi l (@prod_i_head %zero) (@copy_body %idx_next)
    %idx_lt_nn = csltl %idx, %nn
    jnz %idx_lt_nn, @copy_body
    jmp @exp_body_finish

@copy_body
    %off_copy = mull %idx, %eight
    %src_ptr = addl %tmp, %off_copy
    %val = loadl %src_ptr
    %dst_ptr = addl %out, %off_copy
    storel %val, %dst_ptr
    %idx_next = addl %idx, %one
    jmp @copy_head

@exp_body_finish
    %cnt_next = subl %cnt, %one
    jmp @exp_head

@zero_head
    %z = phi l (@exp_head %zero) (@zero_body %z_next)
    %z_lt_nn2 = csltl %z, %nn2
    jnz %z_lt_nn2, @zero_body
    ret %zero

@zero_body
    %off_zero = mull %z, %eight
    %ptr_zero = addl %tmp, %off_zero
    storel %zero, %ptr_zero
    %z_next = addl %z, %one
    jmp @zero_head
}
