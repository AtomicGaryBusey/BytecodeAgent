export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %zero = const 0
    %one = const 1
    %two = const 2
    %eight = const 8
    %nn = mul %n, %n
    %nn_bytes = mul %nn, %eight
    %product_ptr = add %tmp, %nn_bytes
    %nn2 = add %nn, %nn
    %idx0 = const 0
    jmp @copy_a_head

@copy_a_head
    %idx = phi l (@entry %idx0) (@copy_a_body %idx_next)
    %idx_lt = cslt %idx, %nn
    jnz %idx_lt, @copy_a_body
    jmp @init_i_head

@copy_a_body
    %off = mul %idx, %eight
    %src = add %a, %off
    %val = loadl %src
    %dst = add %tmp, %off
    storel %val, %dst
    %idx_next = add %idx, %one
    jmp @copy_a_head

@init_i_head
    %i = phi l (@copy_a_head %zero) (@init_i_body %i_next)
    %i_lt = cslt %i, %n
    jnz %i_lt, @init_j_head
    jmp @exp_head

@init_j_head
    %j = phi l (@init_i_head %zero) (@init_j_body %j_next)
    %j_lt = cslt %j, %n
    jnz %j_lt, @init_body
    jmp @init_i_body

@init_body
    %i_n = mul %i, %n
    %idx = add %i_n, %j
    %off = mul %idx, %eight
    %dst = add %out, %off
    %eq = ceq %i, %j
    %val = sel %eq, %one, %zero
    storel %val, %dst
    %j_next = add %j, %one
    jmp @init_j_head

@init_i_body
    %i_next = add %i, %one
    jmp @init_i_head

@exp_head
    %exp = phi l (@init_i_body %k) (@after_mul_cur_cur %exp_shift)
    %exp_gt0 = cslt %zero, %exp
    jnz %exp_gt0, @exp_check_bit
    jmp @zero_tmp

@exp_check_bit
    %bit = and %exp, %one
    %bit_nonzero = cslt %zero, %bit
    jnz %bit_nonzero, @mul_out_cur_i_head
    jmp @mul_cur_cur_i_head

@mul_out_cur_i_head
    %i = phi l (@exp_check_bit %zero) (@mul_out_cur_i_body %i_next)
    %i_lt = cslt %i, %n
    jnz %i_lt, @mul_out_cur_j_head
    jmp @after_mul_out_cur

@mul_out_cur_i_body
    %i_next = add %i, %one
    jmp @mul_out_cur_i_head

@mul_out_cur_j_head
    %j = phi l (@mul_out_cur_i_head %zero) (@mul_out_cur_j_body %j_next)
    %j_lt = cslt %j, %n
    jnz %j_lt, @mul_out_cur_t_head
    jmp @mul_out_cur_i_body

@mul_out_cur_j_body
    %j_next = add %j, %one
    jmp @mul_out_cur_j_head

@mul_out_cur_t_head
    %t = phi l (@mul_out_cur_j_head %zero) (@mul_out_cur_t_body %t_next)
    %sum = phi l (@mul_out_cur_j_head %zero) (@mul_out_cur_t_body %sum_next)
    %t_lt = cslt %t, %n
    jnz %t_lt, @mul_out_cur_t_compute
    jmp @mul_out_cur_t_store

@mul_out_cur_t_compute
    %i_n = mul %i, %n
    %idx_x = add %i_n, %t
    %off_x = mul %idx_x, %eight
    %ptr_x = add %out, %off_x
    %val_x = loadl %ptr_x

    %t_n = mul %t, %n
    %idx_y = add %t_n, %j
    %off_y = mul %idx_y, %eight
    %ptr_y = add %tmp, %off_y
    %val_y = loadl %ptr_y

    %term = mul %val_x, %val_y
    jmp @mul_out_cur_t_body

@mul_out_cur_t_body
    %sum_next = add %sum, %term
    %t_next = add %t, %one
    jmp @mul_out_cur_t_head

@mul_out_cur_t_store
    %i_n2 = mul %i, %n
    %idx_z = add %i_n2, %j
    %off_z = mul %idx_z, %eight
    %ptr_z = add %product_ptr, %off_z
    storel %sum, %ptr_z
    jmp @mul_out_cur_j_body

@after_mul_out_cur
    %cidx = phi l (@mul_out_cur_i_head %zero) (@copy_out_body %cidx_next)
    %cidx_lt = cslt %cidx, %nn
    jnz %cidx_lt, @copy_out_body
    jmp @mul_cur_cur_i_head

@copy_out_body
    %off = mul %cidx, %eight
    %src = add %product_ptr, %off
    %val = loadl %src
    %dst = add %out, %off
    storel %val, %dst
    %cidx_next = add %cidx, %one
    jmp @after_mul_out_cur

@mul_cur_cur_i_head
    %i = phi l (@exp_check_bit %zero) (@after_mul_out_cur %zero) (@mul_cur_cur_i_body %i_next)
    %i_lt = cslt %i, %n
    jnz %i_lt, @mul_cur_cur_j_head
    jmp @after_mul_cur_cur

@mul_cur_cur_i_body
    %i_next = add %i, %one
    jmp @mul_cur_cur_i_head

@mul_cur_cur_j_head
    %j = phi l (@mul_cur_cur_i_head %zero) (@mul_cur_cur_j_body %j_next)
    %j_lt = cslt %j, %n
    jnz %j_lt, @mul_cur_cur_t_head
    jmp @mul_cur_cur_i_body

@mul_cur_cur_j_body
    %j_next = add %j, %one
    jmp @mul_cur_cur_j_head

@mul_cur_cur_t_head
    %t = phi l (@mul_cur_cur_j_head %zero) (@mul_cur_cur_t_body %t_next)
    %sum = phi l (@mul_cur_cur_j_head %zero) (@mul_cur_cur_t_body %sum_next)
    %t_lt = cslt %t, %n
    jnz %t_lt, @mul_cur_cur_t_compute
    jmp @mul_cur_cur_t_store

@mul_cur_cur_t_compute
    %i_n = mul %i, %n
    %idx_x = add %i_n, %t
    %off_x = mul %idx_x, %eight
    %ptr_x = add %tmp, %off_x
    %val_x = loadl %ptr_x

    %t_n = mul %t, %n
    %idx_y = add %t_n, %j
    %off_y = mul %idx_y, %eight
    %ptr_y = add %tmp, %off_y
    %val_y = loadl %ptr_y

    %term = mul %val_x, %val_y
    jmp @mul_cur_cur_t_body

@mul_cur_cur_t_body
    %sum_next = add %sum, %term
    %t_next = add %t, %one
    jmp @mul_cur_cur_t_head

@mul_cur_cur_t_store
    %i_n2 = mul %i, %n
    %idx_z = add %i_n2, %j
    %off_z = mul %idx_z, %eight
    %ptr_z = add %product_ptr, %off_z
    storel %sum, %ptr_z
    jmp @mul_cur_cur_j_body

@after_mul_cur_cur
    %cidx2 = phi l (@mul_cur_cur_i_head %zero) (@copy_cur_body %cidx_next)
    %cidx2_lt = cslt %cidx2, %nn
    jnz %cidx2_lt, @copy_cur_body
    %exp_shift = shr %exp, %one
    jmp @exp_head

@copy_cur_body
    %off = mul %cidx2, %eight
    %src = add %product_ptr, %off
    %val = loadl %src
    %dst = add %tmp, %off
    storel %val, %dst
    %cidx_next = add %cidx2, %one
    jmp @after_mul_cur_cur

@zero_tmp
    %z = phi l (@exp_head %zero) (@zero_tmp_body %z_next)
    %z_lt = cslt %z, %nn2
    jnz %z_lt, @zero_tmp_body
    ret %zero

@zero_tmp_body
    %off = mul %z, %eight
    %addr = add %tmp, %off
    storel %zero, %addr
    %z_next = add %z, %one
    jmp @zero_tmp
}
