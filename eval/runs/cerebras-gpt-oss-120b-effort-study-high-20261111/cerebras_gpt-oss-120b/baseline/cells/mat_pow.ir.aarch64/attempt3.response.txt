export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %zero = subl %n, %n
    %one = addl %zero, 1
    %eight = addl %zero, 8
    %nn = mull %n, %n
    %nn2 = addl %nn, %nn
    %nn_bytes = mull %nn, %eight
    %product_ptr = addl %tmp, %nn_bytes
    jmp @copy_a_head

@copy_a_head
    %idx = phi l (@entry %zero) (@copy_a_body %idx_next)
    %idx_lt = csltl %idx, %nn
    jnz %idx_lt, @copy_a_body
    jmp @init_id_i_head

@copy_a_body
    %off = mull %idx, %eight
    %src = addl %a, %off
    %val = loadl %src
    %dst = addl %tmp, %off
    storel %val, %dst
    %idx_next = addl %idx, %one
    jmp @copy_a_head

@init_id_i_head
    %i = phi l (@copy_a_head %zero) (@init_id_i_body %i_next)
    %i_lt = csltl %i, %n
    jnz %i_lt, @init_id_j_head
    jmp @exp_head

@init_id_j_head
    %j = phi l (@init_id_i_head %zero) (@init_id_j_body %j_next)
    %j_lt = csltl %j, %n
    jnz %j_lt, @init_id_body
    jmp @init_id_i_body

@init_id_body
    %i_n = mull %i, %n
    %idx = addl %i_n, %j
    %off = mull %idx, %eight
    %dst = addl %out, %off
    %eq = ceql %i, %j
    %val = sell %eq, %one, %zero
    storel %val, %dst
    %j_next = addl %j, %one
    jmp @init_id_j_head

@init_id_i_body
    %i_next = addl %i, %one
    jmp @init_id_i_head

@exp_head
    %exp = phi l (@init_id_i_head %k) (@after_mul_cur_cur %exp_shift)
    %exp_gt0 = csltl %zero, %exp
    jnz %exp_gt0, @exp_check_bit
    jmp @zero_tmp

@exp_check_bit
    %bit = andl %exp, %one
    %bit_nonzero = csltl %zero, %bit
    jnz %bit_nonzero, @mul_out_cur_i_head
    jmp @mul_cur_cur_i_head

@mul_out_cur_i_head
    %i = phi l (@exp_check_bit %zero) (@mul_out_cur_i_body %i_next)
    %i_lt = csltl %i, %n
    jnz %i_lt, @mul_out_cur_j_head
    jmp @after_mul_out_cur

@mul_out_cur_i_body
    %i_next = addl %i, %one
    jmp @mul_out_cur_i_head

@mul_out_cur_j_head
    %j = phi l (@mul_out_cur_i_head %zero) (@mul_out_cur_j_body %j_next)
    %j_lt = csltl %j, %n
    jnz %j_lt, @mul_out_cur_t_head
    jmp @mul_out_cur_i_body

@mul_out_cur_j_body
    %j_next = addl %j, %one
    jmp @mul_out_cur_j_head

@mul_out_cur_t_head
    %t = phi l (@mul_out_cur_j_head %zero) (@mul_out_cur_t_body %t_next)
    %sum = phi l (@mul_out_cur_j_head %zero) (@mul_out_cur_t_body %sum_next)
    %t_lt = csltl %t, %n
    jnz %t_lt, @mul_out_cur_t_compute
    jmp @mul_out_cur_t_store

@mul_out_cur_t_compute
    %i_n = mull %i, %n
    %idx_x = addl %i_n, %t
    %off_x = mull %idx_x, %eight
    %ptr_x = addl %out, %off_x
    %val_x = loadl %ptr_x

    %t_n = mull %t, %n
    %idx_y = addl %t_n, %j
    %off_y = mull %idx_y, %eight
    %ptr_y = addl %tmp, %off_y
    %val_y = loadl %ptr_y

    %term = mull %val_x, %val_y
    jmp @mul_out_cur_t_body

@mul_out_cur_t_body
    %sum_next = addl %sum, %term
    %t_next = addl %t, %one
    jmp @mul_out_cur_t_head

@mul_out_cur_t_store
    %i_n2 = mull %i, %n
    %idx_z = addl %i_n2, %j
    %off_z = mull %idx_z, %eight
    %ptr_z = addl %product_ptr, %off_z
    storel %sum, %ptr_z
    jmp @mul_out_cur_j_body

@after_mul_out_cur
    %cidx = phi l (@mul_out_cur_i_head %zero) (@copy_out_body %cidx_next)
    %cidx_lt = csltl %cidx, %nn
    jnz %cidx_lt, @copy_out_body
    jmp @mul_cur_cur_i_head

@copy_out_body
    %off = mull %cidx, %eight
    %src = addl %product_ptr, %off
    %val = loadl %src
    %dst = addl %out, %off
    storel %val, %dst
    %cidx_next = addl %cidx, %one
    jmp @after_mul_out_cur

@mul_cur_cur_i_head
    %i = phi l (@after_mul_out_cur %zero) (@mul_cur_cur_i_body %i_next)
    %i_lt = csltl %i, %n
    jnz %i_lt, @mul_cur_cur_j_head
    jmp @after_mul_cur_cur

@mul_cur_cur_i_body
    %i_next = addl %i, %one
    jmp @mul_cur_cur_i_head

@mul_cur_cur_j_head
    %j = phi l (@mul_cur_cur_i_head %zero) (@mul_cur_cur_j_body %j_next)
    %j_lt = csltl %j, %n
    jnz %j_lt, @mul_cur_cur_t_head
    jmp @mul_cur_cur_i_body

@mul_cur_cur_j_body
    %j_next = addl %j, %one
    jmp @mul_cur_cur_j_head

@mul_cur_cur_t_head
    %t = phi l (@mul_cur_cur_j_head %zero) (@mul_cur_cur_t_body %t_next)
    %sum = phi l (@mul_cur_cur_j_head %zero) (@mul_cur_cur_t_body %sum_next)
    %t_lt = csltl %t, %n
    jnz %t_lt, @mul_cur_cur_t_compute
    jmp @mul_cur_cur_t_store

@mul_cur_cur_t_compute
    %i_n = mull %i, %n
    %idx_x = addl %i_n, %t
    %off_x = mull %idx_x, %eight
    %ptr_x = addl %tmp, %off_x
    %val_x = loadl %ptr_x

    %t_n = mull %t, %n
    %idx_y = addl %t_n, %j
    %off_y = mull %idx_y, %eight
    %ptr_y = addl %tmp, %off_y
    %val_y = loadl %ptr_y

    %term = mull %val_x, %val_y
    jmp @mul_cur_cur_t_body

@mul_cur_cur_t_body
    %sum_next = addl %sum, %term
    %t_next = addl %t, %one
    jmp @mul_cur_cur_t_head

@mul_cur_cur_t_store
    %i_n2 = mull %i, %n
    %idx_z = addl %i_n2, %j
    %off_z = mull %idx_z, %eight
    %ptr_z = addl %product_ptr, %off_z
    storel %sum, %ptr_z
    jmp @mul_cur_cur_j_body

@after_mul_cur_cur
    %cidx = phi l (@mul_cur_cur_i_head %zero) (@copy_cur_body %cidx_next)
    %cidx_lt = csltl %cidx, %nn
    jnz %cidx_lt, @copy_cur_body
    %exp_shift = shrl %exp, 1
    jmp @exp_head

@copy_cur_body
    %off = mull %cidx, %eight
    %src = addl %product_ptr, %off
    %val = loadl %src
    %dst = addl %tmp, %off
    storel %val, %dst
    %cidx_next = addl %cidx, %one
    jmp @after_mul_cur_cur

@zero_tmp
    %z = phi l (@exp_head %zero) (@zero_tmp_body %z_next)
    %z_lt = csltl %z, %nn2
    jnz %z_lt, @zero_tmp_body
    ret %zero

@zero_tmp_body
    %off = mull %z, %eight
    %addr = addl %tmp, %off
    storel %zero, %addr
    %z_next = addl %z, %one
    jmp @zero_tmp
}
