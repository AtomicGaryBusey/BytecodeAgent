export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %zero = const l 0
    %one = const l 1
    %eight = const l 8
    %nn = mull %n, %n
    %nn2 = addl %nn, %nn
    jmp @id_i_head

@id_i_head
    %i = phi l @entry 0 @id_i_body %i_next
    %i_lt_n = csltl %i, %n
    jz %i_lt_n, @id_done
    jmp @id_j_head

@id_i_body
    %i_next = addl %i, %one
    jmp @id_i_head

@id_j_head
    %j = phi l @id_i_head 0 @id_j_body %j_next
    %j_lt_n = csltl %j, %n
    jz %j_lt_n, @id_i_body
    %id_i_n = mull %i, %n
    %id_dest_idx = addl %id_i_n, %j
    %id_dest_off = mull %id_dest_idx, %eight
    %id_dest_ptr = addl %out, %id_dest_off
    %id_eq = ceql %i, %j
    %id_val = sell %id_eq, %one, %zero
    storel %id_val, %id_dest_ptr
    jmp @id_j_body

@id_j_body
    %j_next = addl %j, %one
    jmp @id_j_head

@id_done
    jmp @exp_head

@exp_head
    %cnt = phi l @id_done %k @exp_body_finish %cnt_next
    %cnt_gt0 = csltl %zero, %cnt
    jnz %cnt_gt0, @prod_i_head
    jmp @exp_end

@prod_i_head
    %mi = phi l @exp_head 0 @prod_i_body %mi_next
    %mi_lt_n = csltl %mi, %n
    jz %mi_lt_n, @prod_done
    jmp @prod_j_head

@prod_i_body
    %mi_next = addl %mi, %one
    jmp @prod_i_head

@prod_j_head
    %mj = phi l @prod_i_head 0 @prod_j_body %mj_next
    %mj_lt_n = csltl %mj, %n
    jz %mj_lt_n, @prod_i_body
    jmp @prod_t_head

@prod_j_body
    %mj_next = addl %mj, %one
    jmp @prod_j_head

@prod_t_head
    %mt = phi l @prod_j_head 0 @prod_t_body %mt_next
    %msum = phi l @prod_j_head 0 @prod_t_body %msum_next
    %mt_lt_n = csltl %mt, %n
    jz %mt_lt_n, @prod_store_sum
    jmp @prod_t_body

@prod_t_body
    %pt_i_n = mull %mi, %n
    %pt_out_idx = addl %pt_i_n, %mt
    %pt_out_off = mull %pt_out_idx, %eight
    %pt_out_ptr = addl %out, %pt_out_off
    %pt_out_val = loadl %pt_out_ptr
    %pt_t_n = mull %mt, %n
    %pt_a_idx = addl %pt_t_n, %mj
    %pt_a_off = mull %pt_a_idx, %eight
    %pt_a_ptr = addl %a, %pt_a_off
    %pt_a_val = loadl %pt_a_ptr
    %pt_prod = mull %pt_out_val, %pt_a_val
    %msum_next = addl %msum, %pt_prod
    %mt_next = addl %mt, %one
    jmp @prod_t_head

@prod_store_sum
    %ps_i_n = mull %mi, %n
    %ps_dest_idx = addl %ps_i_n, %mj
    %ps_dest_off = mull %ps_dest_idx, %eight
    %ps_dest_ptr = addl %tmp, %ps_dest_off
    storel %msum, %ps_dest_ptr
    jmp @prod_j_body

@prod_done
    jmp @copy_head

@copy_head
    %c_idx = phi l @prod_done 0 @copy_body %c_idx_next
    %c_lt = csltl %c_idx, %nn
    jz %c_lt, @copy_done
    %c_off = mull %c_idx, %eight
    %c_src_ptr = addl %tmp, %c_off
    %c_val = loadl %c_src_ptr
    %c_dst_ptr = addl %out, %c_off
    storel %c_val, %c_dst_ptr
    %c_idx_next = addl %c_idx, %one
    jmp @copy_body

@copy_body
    jmp @copy_head

@copy_done
    jmp @exp_body_finish

@exp_body_finish
    %cnt_next = subl %cnt, %one
    jmp @exp_head

@exp_end
    jmp @zero_head

@zero_head
    %z_idx = phi l @exp_end 0 @zero_body %z_idx_next
    %z_lt = csltl %z_idx, %nn2
    jz %z_lt, @zero_done
    %z_off = mull %z_idx, %eight
    %z_ptr = addl %tmp, %z_off
    storel %zero, %z_ptr
    %z_idx_next = addl %z_idx, %one
    jmp @zero_body

@zero_body
    jmp @zero_head

@zero_done
    ret %zero
}
