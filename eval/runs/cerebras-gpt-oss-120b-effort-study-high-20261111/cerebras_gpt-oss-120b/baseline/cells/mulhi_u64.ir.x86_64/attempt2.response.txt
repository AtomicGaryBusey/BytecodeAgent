export function l $mulhi_u64(l a, l b) {
@start
  %a0 = and a, c0xffffffff
  %a1 = shr a, c32
  %b0 = and b, c0xffffffff
  %b1 = shr b, c32
  %p0 = mul %a0, %b0
  %p1 = mul %a0, %b1
  %p2 = mul %a1, %b0
  %p3 = mul %a1, %b1
  %x0hi = shr %p0, c32
  %s1 = add %p1, %p2
  %c12 = ult %s1, %p1
  %c12z = zext %c12
  %s2 = add %s1, %x0hi
  %c0 = ult %s2, %s1
  %c0z = zext %c0
  %carry = add %c12z, %c0z
  %hi_part = shr %s2, c32
  %carry_shift = shl %carry, c32
  %tmp = add %p3, %hi_part
  %hi = add %tmp, %carry_shift
  ret %hi
}
