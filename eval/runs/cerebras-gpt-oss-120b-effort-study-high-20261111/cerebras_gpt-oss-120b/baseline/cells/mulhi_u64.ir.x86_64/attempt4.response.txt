export function l $mulhi_u64(l a, l b) {
@start:
    %a0 = and a, 4294967295
    %a1 = shr a, 32
    %b0 = and b, 4294967295
    %b1 = shr b, 32
    %p0 = mul %a0, %b0
    %p1 = mul %a0, %b1
    %p2 = mul %a1, %b0
    %p3 = mul %a1, %b1
    %s = add %p1, %p2
    %c_s = ult %s, %p1
    %c_sz = zext %c_s
    %c_s_shift = shl %c_sz, 32
    %s_low = and %s, 4294967295
    %s_high = shr %s, 32
    %t = shl %s_low, 32
    %low = add %p0, %t
    %c_low = ult %low, %p0
    %c_low_z = zext %c_low
    %hi1 = add %p3, %s_high
    %hi2 = add %hi1, %c_s_shift
    %hi3 = add %hi2, %c_low_z
    ret %hi3
}
