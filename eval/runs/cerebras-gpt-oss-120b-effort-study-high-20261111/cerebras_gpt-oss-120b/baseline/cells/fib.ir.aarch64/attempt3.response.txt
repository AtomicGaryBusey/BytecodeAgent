export function l $fib(l %n) {
@entry:
  %c1 = cnst l 1
  %c2 = cnst l 2
  %lt2 = csltl %n %c2
  jnz %lt2, @ret_n
  %a0 = cnst l 0
  %b0 = cnst l 1
  %i0 = cnst l 2
  jmp @loop(%a0, %b0, %i0)
@loop(%a, %b, %i):
  %cond = csltl %n %i
  jnz %cond, @exit(%b)
  %c = addl %a %b
  %next_a = %b
  %next_b = %c
  %next_i = addl %i %c1
  jmp @loop(%next_a, %next_b, %next_i)
@exit(%result):
  ret %result
@ret_n:
  ret %n
}
