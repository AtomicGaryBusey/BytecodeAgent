export function l $fib(l %n) {
@entry:
  %c0 = cnst l 0
  %c1 = cnst l 1
  %c2 = cnst l 2
  %lt2 = csltl %n %c2
  jnz %lt2, @ret_n
  %a = %c0
  %b = %c1
  %i = %c2
  jmp @loop(%a, %b, %i)
@loop(%a, %b, %i):
  %cmp = csltl %n %i
  jnz %cmp, @exit(%b)
  %c = addl %a %b
  %a_next = %b
  %b_next = %c
  %i_next = addl %i %c1
  jmp @loop(%a_next, %b_next, %i_next)
@exit(%result):
  ret %result
@ret_n:
  ret %n
}
