export function l $gcd(l, l) {
@entry
    %a0 = argl 0
    %b0 = argl 1
    jmp @loop
@loop
    %a_cur = phi l ( @entry %a0, @body %a_next )
    %b_cur = phi l ( @entry %b0, @body %b_next )
    %cond = ceql %b_cur, 0
    jnz %cond, @exit
@body
    %t = modl %a_cur, %b_cur
    %a_next = copyl %b_cur
    %b_next = copyl %t
    jmp @loop
@exit
    ret %a_cur
}
