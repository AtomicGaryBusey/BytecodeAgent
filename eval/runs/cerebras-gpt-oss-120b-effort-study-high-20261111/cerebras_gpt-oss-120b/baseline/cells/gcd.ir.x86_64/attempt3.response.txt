export function l $gcd(l a, l b) {
    @entry
        jmp @loop
    @loop
        %a_cur = phi l ( @entry %a, @body %a_next )
        %b_cur = phi l ( @entry %b, @body %b_next )
        %cond = ceql %b_cur, 0
        jnz %cond, @exit
    @body
        %t = modl %a_cur, %b_cur
        %a_next = copyl %b_cur
        %b_next = copyl %t
        jmp @loop
    @exit
        ret %a_cur
}
