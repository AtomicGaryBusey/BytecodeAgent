export function l $gcd(l a, l b) {
    @loop
        %a_cur = phi l a, %a_next
        %b_cur = phi l b, %b_next
        %cond = ceql %b_cur, 0
        jnz %cond, @exit
        %t = modl %a_cur, %b_cur
        %a_next = copyl %b_cur
        %b_next = copyl %t
        jmp @loop
    @exit
        ret %a_cur
}
