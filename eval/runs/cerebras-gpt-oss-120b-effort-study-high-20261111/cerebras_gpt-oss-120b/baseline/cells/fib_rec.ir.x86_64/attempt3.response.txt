export function l $fib_rec(l %n)
{
    @entry
    %c2 =l constl 2
    %cond =w csltl %n, %c2
    jnz %cond, @base
    @recurse
    %c1 =l constl 1
    %nm1 =l subl %n, %c1
    %r1 =l call $fib_rec(l %nm1)
    %nm2 =l subl %n, %c2
    %r2 =l call $fib_rec(l %nm2)
    %sum =l addl %r1, %r2
    ret %sum
    @base
    ret %n
}
