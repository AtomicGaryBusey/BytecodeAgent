export function q $fib_rec(q %n)
{
    @entry
    %c2 =q constq 2
    %cond =w csltq %n, %c2
    jnz %cond, @base

    @recurse
    %c1 =q constq 1
    %nm1 =q subq %n, %c1
    %r1 =q call $fib_rec(q %nm1)
    %nm2 =q subq %n, %c2
    %r2 =q call $fib_rec(q %nm2)
    %sum =q addq %r1, %r2
    ret %sum

    @base
    ret %n
}
