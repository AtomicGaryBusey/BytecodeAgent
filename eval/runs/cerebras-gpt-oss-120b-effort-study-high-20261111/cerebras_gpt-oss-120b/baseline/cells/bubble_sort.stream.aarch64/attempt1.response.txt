bubble_sort:
    cmp x1, #1
    ble .Lret
    sub x2, x1, #1
    mov x9, #0
.Louter:
    cmp x9, x2
    b.ge .Lret
    sub x3, x2, x9
    mov x10, #0
.Linner:
    cmp x10, x3
    b.ge .Linner_done
    lsl x4, x10, #3
    add x7, x4, #8
    ldr x5, [x0, x4]
    ldr x6, [x0, x7]
    cmp x5, x6
    b.le .Linner_next
    str x6, [x0, x4]
    str x5, [x0, x7]
.Linner_next:
    add x10, x10, #1
    b .Linner
.Linner_done:
    add x9, x9, #1
    b .Louter
.Lret:
    mov x0, #0
    ret
