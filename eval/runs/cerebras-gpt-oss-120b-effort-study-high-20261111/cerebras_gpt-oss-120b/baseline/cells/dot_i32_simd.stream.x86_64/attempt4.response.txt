dot_i32_simd:
    xor eax, eax
    pxor xmm0, xmm0
    mov rcx, rdx
    shr rcx, 2
    and rdx, 3
    test rcx, rcx
    jz .Lscalar
.Lvec_loop:
    movdqu xmm1, [rdi]
    movdqu xmm2, [rsi]
    pmovsxdq xmm3, xmm1
    pmovsxdq xmm4, xmm2
    movdqa xmm5, xmm3
    psrad xmm5, 31
    movdqa xmm6, xmm4
    psrad xmm6, 31
    movdqa xmm7, xmm3
    pmuludq xmm7, xmm4
    movdqa xmm8, xmm5
    pand xmm8, xmm4
    movdqa xmm9, xmm6
    pand xmm9, xmm3
    paddq xmm8, xmm9
    psllq xmm8, 32
    psubq xmm7, xmm8
    paddq xmm0, xmm7
    psrldq xmm1, 8
    psrldq xmm2, 8
    pmovsxdq xmm3, xmm1
    pmovsxdq xmm4, xmm2
    movdqa xmm5, xmm3
    psrad xmm5, 31
    movdqa xmm6, xmm4
    psrad xmm6, 31
    movdqa xmm7, xmm3
    pmuludq xmm7, xmm4
    movdqa xmm8, xmm5
    pand xmm8, xmm4
    movdqa xmm9, xmm6
    pand xmm9, xmm3
    paddq xmm8, xmm9
    psllq xmm8, 32
    psubq xmm7, xmm8
    paddq xmm0, xmm7
    add rdi, 16
    add rsi, 16
    dec rcx
    jne .Lvec_loop
.Lscalar:
    movq rax, xmm0
    psrldq xmm0, 8
    movq rcx, xmm0
    add rax, rcx
    test rdx, rdx
    jz .Ldone
.Lscalar_loop:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rdx
    jne .Lscalar_loop
.Ldone:
    ret
