dot_i32_simd:
    mov rcx, rdx
    shr rcx, 1
    and rdx, 1
    xorps xmm2, xmm2
    test rcx, rcx
    jz .Lscalar
.Lvec_loop:
    movq xmm0, qword ptr [rdi]
    movq xmm1, qword ptr [rsi]
    pmuldq xmm0, xmm1
    paddq xmm2, xmm0
    add rdi, 8
    add rsi, 8
    dec rcx
    jne .Lvec_loop
.Lscalar:
    pextrq rax, xmm2, 0
    pextrq rcx, xmm2, 1
    add rax, rcx
    test rdx, rdx
    je .Lreturn
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
.Lreturn:
    ret
