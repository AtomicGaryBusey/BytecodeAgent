dot_i32_simd:
    mov rcx, rdx
    shr rcx, 2
    and rdx, 3
    xorps xmm4, xmm4
    test rcx, rcx
    jz .Lscalar
.Lvec_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pmulld xmm0, xmm1
    paddq xmm4, xmm0
    psrldq xmm2, 4
    psrldq xmm3, 4
    pmulld xmm2, xmm3
    paddq xmm4, xmm2
    add rdi, 16
    add rsi, 16
    dec rcx
    jne .Lvec_loop
.Lscalar:
    movq rax, xmm4
    pextrq rcx, xmm4, 1
    add rax, rcx
    test rdx, rdx
    je .Lreturn
.Lscalar_loop:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rdx
    jne .Lscalar_loop
.Lreturn:
    ret
