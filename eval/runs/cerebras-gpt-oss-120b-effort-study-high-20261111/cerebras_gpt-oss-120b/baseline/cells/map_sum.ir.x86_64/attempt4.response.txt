export function l $map_sum(l %a, l %n, l %f) {
@entry:
    %i0 =l const 0
    %sum0 =l const 0
    %eight =l const 8
    %one =l const 1
    jmp Ltest

@Ltest:
    %i =phi l (%i0, entry) (%i1, Lbody)
    %sum =phi l (%sum0, entry) (%sum1, Lbody)
    %cond =w csltl %i, %n
    jnz %cond, Lbody, Lend

@Lbody:
    %offset =l mull %i, %eight
    %addr =l addl %a, %offset
    %val =l loadl %addr
    %r =l call %f(l %val)
    %sum1 =l addl %sum, %r
    %i1 =l addl %i, %one
    jmp Ltest

@Lend:
    ret %sum
}
