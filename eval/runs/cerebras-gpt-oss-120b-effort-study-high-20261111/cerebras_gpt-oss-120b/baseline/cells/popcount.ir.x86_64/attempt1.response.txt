export function l $popcount(l x) {
@start:
    %a = shrl %x, 1
    %b = andl %a, 0x5555555555555555
    %c = subl %x, %b
    %d = shrl %c, 2
    %e = andl %d, 0x3333333333333333
    %f = andl %c, 0x3333333333333333
    %g = addl %f, %e
    %h = shrl %g, 4
    %i = addl %g, %h
    %j = andl %i, 0x0F0F0F0F0F0F0F0F
    %k = mull %j, 0x0101010101010101
    %l = shrl %k, 56
    retl %l
}
