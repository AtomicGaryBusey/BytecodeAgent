export function l $popcount(l) {
@start:
    %cnt0 = addl 0l, 0l
    jmp @loop
@loop:
    %x = phi l (%0, @start, %x_next, @body)
    %cnt = phi l (%cnt0, @start, %cnt_next, @body)
    %nz = csnel %x, 0l
    jnz %nz, @body
    retl %cnt
@body:
    %tmp = subl %x, 1l
    %x_next = andl %x, %tmp
    %cnt_next = addl %cnt, 1l
    jmp @loop
}
