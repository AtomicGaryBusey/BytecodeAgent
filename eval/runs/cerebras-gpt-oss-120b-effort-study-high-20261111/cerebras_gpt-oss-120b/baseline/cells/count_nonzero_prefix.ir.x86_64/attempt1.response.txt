export function l $count_nonzero_prefix(l %a, l %n) {
@entry
jmp @cond

@cond
%i = phi l 0 @entry, %i_next @inc
%cmp = csltl %i, %n
jnz %cmp, @load, @exit

@load
%addr_off = shll %i, 3
%addr = addl %a, %addr_off
%val = loadl %addr
%zero_test = ceql %val, 0
jnz %zero_test, @exit, @inc

@inc
%i_next = addl %i, 1
jmp @cond

@exit
ret %i
}
