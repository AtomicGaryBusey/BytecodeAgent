export function l $collatz_max(l) {
@entry
    %limit = addl %0, $1
    %i = $0
    %max = $0
    jmp @outer_cond

@outer_cond
    %i = addl %i, $1
    %cond = csltl %i, %limit
    jnz %cond, @process_k
    jmp @done

@process_k
    %x = %i
    %steps = $0
    jmp @inner_check

@inner_check
    %eq = cseql %x, $1
    jnz %eq, @inner_done
    %odd = andl %x, $1
    %is_even = cseql %odd, $0
    jnz %is_even, @even_case
    jmp @odd_case

@odd_case
    %tmp = mull %x, $3
    %x = addl %tmp, $1
    jmp @step_inc

@even_case
    %x = shrl %x, $1
    jmp @step_inc

@step_inc
    %steps = addl %steps, $1
    jmp @inner_check

@inner_done
    %gt = csltl %max, %steps
    jnz %gt, @update_max
    jmp @outer_cond

@update_max
    %max = addl %steps, $0
    jmp @outer_cond

@done
    ret %max
}
