export function l $collatz_max(l n) {
@entry
    %limit = addl %n, $1
    %i = subl %n, %n
    %max = subl %n, %n
    jmp @outer

@outer
    %i = addl %i, $1
    %cond = csltl %i, %limit
    jnz %cond, @process_k
    jmp @done

@process_k
    %x = addl %i, $0
    %steps = subl %n, %n
    jmp @inner_check

@inner_check
    %eq = cseql %x, $1
    jnz %eq, @inner_done
    %odd = andl %x, $1
    %is_even = cseql %odd, $0
    jnz %is_even, @even_case
    jmp @odd_case

@odd_case
    %tmp = mull %x, $3
    %x = addl %tmp, $1
    jmp @step_inc

@even_case
    %x = shrl %x, $1
    jmp @step_inc

@step_inc
    %steps = addl %steps, $1
    jmp @inner_check

@inner_done
    %gt = csgtl %steps, %max
    jnz %gt, @update_max
    jmp @outer

@update_max
    %max = addl %steps, $0
    jmp @outer

@done
    ret %max
}
