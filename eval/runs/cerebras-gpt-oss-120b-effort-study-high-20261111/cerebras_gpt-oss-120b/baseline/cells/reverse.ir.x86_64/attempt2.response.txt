export function l $reverse(l, l) {
@entry
%a = arg l
%n = arg l
%zero = const l 0
%one = const l 1
%two = const l 2
%three = const l 3
%half = divl %n, %two
jmp @loop

@loop
%i = phi l (%zero @entry, %i_next @body)
%cond = csltl %i, %half
jnz %cond, @body
jmp @exit

@body
%tmp1 = subl %n, %i
%j = subl %tmp1, %one

%off_i = shll %i, %three
%addr_i = addl %a, %off_i
%off_j = shll %j, %three
%addr_j = addl %a, %off_j

%val_i = loadl %addr_i
%val_j = loadl %addr_j

storel %val_j, %addr_i
storel %val_i, %addr_j

%i_next = addl %i, %one
jmp @loop

@exit
ret %zero
}
