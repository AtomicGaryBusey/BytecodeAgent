export function l $reverse(l, l) {
@entry
%a = arg l
%n = arg l
%zero = const l 0
%one = const l 1
%two = const l 2
%eight = const l 8
%half = divl %n, %two
jmp @head

@head
%i = phi l (%zero @entry, %i_next @body)
%cond = csltl %i, %half
jnz %cond, @body
jmp @exit

@body
%sub = subl %n, %i
%j = subl %sub, %one

%ofs_i = mull %i, %eight
%addr_i = addl %a, %ofs_i
%ofs_j = mull %j, %eight
%addr_j = addl %a, %ofs_j

%val_i = loadl %addr_i
%val_j = loadl %addr_j

storel %val_j, %addr_i
storel %val_i, %addr_j

%i_next = addl %i, %one
jmp @head

@exit
ret %zero
}
