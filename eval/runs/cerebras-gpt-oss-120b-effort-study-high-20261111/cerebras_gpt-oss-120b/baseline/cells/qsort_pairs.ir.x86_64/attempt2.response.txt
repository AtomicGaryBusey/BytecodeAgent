export function l $qsort_pairs(l %a, l %n) {
@entry:
    %cond_base = w csltl %n, 2
    jnz %cond_base, @exit
    %limit = l subl %n, 1
    %off_last = l shll %limit, 4
    %addr_key_last = l addl %a, %off_last
    %addr_val_last = l addl %addr_key_last, 8
    %pkey = l loadl %addr_key_last
    %pval = l loadl %addr_val_last
    %i0 = l 0
    %s0 = l 0
    jmp @loop(%a, %n, %limit, %pkey, %pval, %i0, %s0)

@loop(l %a, l %n, l %limit, l %pkey, l %pval, l %i, l %s):
    %cond_i_lt_limit = w csltl %i, %limit
    jnz %cond_i_lt_limit, @loop_body(%a, %n, %limit, %pkey, %pval, %i, %s)
    jmp @after_loop(%a, %n, %limit, %pkey, %pval, %s)

@loop_body(l %a, l %n, l %limit, l %pkey, l %pval, l %i, l %s):
    %off_i = l shll %i, 4
    %addr_key_i = l addl %a, %off_i
    %addr_val_i = l addl %addr_key_i, 8
    %key_i = l loadl %addr_key_i
    %val_i = l loadl %addr_val_i
    %k_lt = w csltl %key_i, %pkey
    %k_eq = w ceql %key_i, %pkey
    %v_lt = w csltl %val_i, %pval
    %eq_and_v_lt = w and %k_eq, %v_lt
    %less = w or %k_lt, %eq_and_v_lt
    jnz %less, @swap_and_inc(%a, %n, %limit, %pkey, %pval, %i, %s, %key_i, %val_i, %addr_key_i, %addr_val_i)
    jmp @inc_i(%a, %n, %limit, %pkey, %pval, %i, %s)

@swap_and_inc(l %a, l %n, l %limit, l %pkey, l %pval, l %i, l %s, l %key_i, l %val_i, l %addr_key_i, l %addr_val_i):
    %off_s = l shll %s, 4
    %addr_key_s = l addl %a, %off_s
    %addr_val_s = l addl %addr_key_s, 8
    %key_s = l loadl %addr_key_s
    %val_s = l loadl %addr_val_s
    storel %key_i, %addr_key_s
    storel %val_i, %addr_val_s
    storel %key_s, %addr_key_i
    storel %val_s, %addr_val_i
    %i1 = l addl %i, 1
    %s1 = l addl %s, 1
    jmp @loop(%a, %n, %limit, %pkey, %pval, %i1, %s1)

@inc_i(l %a, l %n, l %limit, l %pkey, l %pval, l %i, l %s):
    %i1 = l addl %i, 1
    jmp @loop(%a, %n, %limit, %pkey, %pval, %i1, %s)

@after_loop(l %a, l %n, l %limit, l %pkey, l %pval, l %s):
    %off_s = l shll %s, 4
    %addr_key_s = l addl %a, %off_s
    %addr_val_s = l addl %addr_key_s, 8
    %key_s = l loadl %addr_key_s
    %val_s = l loadl %addr_val_s
    %off_last = l shll %limit, 4
    %addr_key_last = l addl %a, %off_last
    %addr_val_last = l addl %addr_key_last, 8
    storel %pkey, %addr_key_s
    storel %pval, %addr_val_s
    storel %key_s, %addr_key_last
    storel %val_s, %addr_val_last
    %cond_left = w csltl %s, 2
    jnz %cond_left, @right_test(%a, %n, %limit, %s)
    %r_left = l call $qsort_pairs(l %a, l %s)
    jmp @right_test(%a, %n, %limit, %s)

@right_test(l %a, l %n, l %limit, l %s):
    %tmp = l subl %n, %s
    %right_size = l subl %tmp, 1
    %cond_right = w csltl %right_size, 2
    jnz %cond_right, @exit
    %s1 = l addl %s, 1
    %off_s1 = l shll %s1, 4
    %ptr_right = l addl %a, %off_s1
    %r_right = l call $qsort_pairs(l %ptr_right, l %right_size)
    jmp @exit

@exit:
    retl 0
}
