qsort_pairs:
    push rbx
    push r12
    push r13
    mov rbx, rdi
    mov r12, rsi
    cmp r12, 1
    jle .Lreturn_zero
    mov rcx, r12
    dec rcx
    shl rcx, 4
    mov rdx, qword ptr [rbx + rcx]
    mov r8, qword ptr [rbx + rcx + 8]
    mov r9, -1
    xor r10, r10
    mov r11, r12
    dec r11
.Lloop:
    cmp r10, r11
    jge .Lafter_loop
    mov rsi, r10
    shl rsi, 4
    mov rax, qword ptr [rbx + rsi]
    mov rcx, qword ptr [rbx + rsi + 8]
    cmp rax, rdx
    jl .Lswap
    jg .Lnoswap
    cmp rcx, r8
    jle .Lswap
    jmp .Lnoswap
.Lswap:
    inc r9
    mov rdi, r9
    shl rdi, 4
    mov r15, qword ptr [rbx + rdi]
    mov r14, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rdi], rax
    mov qword ptr [rbx + rdi + 8], rcx
    mov qword ptr [rbx + rsi], r15
    mov qword ptr [rbx + rsi + 8], r14
.Lnoswap:
    inc r10
    jmp .Lloop
.Lafter_loop:
    inc r9
    mov r13, r9
    mov rdx, r12
    dec rdx
    shl rdx, 4
    mov rax, qword ptr [rbx + rdx]
    mov rcx, qword ptr [rbx + rdx + 8]
    mov rdi, r13
    shl rdi, 4
    mov r8, qword ptr [rbx + rdi]
    mov r14, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rdi], rax
    mov qword ptr [rbx + rdi + 8], rcx
    mov qword ptr [rbx + rdx], r8
    mov qword ptr [rbx + rdx + 8], r14
    cmp r13, 1
    jle .Lskip_left
    mov rdi, rbx
    mov rsi, r13
    call qsort_pairs
.Lskip_left:
    mov rax, r12
    sub rax, r13
    dec rax
    cmp rax, 1
    jle .Lskip_right
    mov rdx, r13
    inc rdx
    shl rdx, 4
    lea rdi, [rbx + rdx]
    mov rsi, rax
    call qsort_pairs
.Lskip_right:
    xor rax, rax
    pop r13
    pop r12
    pop rbx
    ret
.Lreturn_zero:
    xor rax, rax
    pop r13
    pop r12
    pop rbx
    ret
