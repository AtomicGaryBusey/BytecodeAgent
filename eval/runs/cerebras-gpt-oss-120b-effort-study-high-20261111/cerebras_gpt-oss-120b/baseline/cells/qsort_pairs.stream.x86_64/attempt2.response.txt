qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    cmp r12, 1
    jle .Lreturn
    mov rdx, r12
    dec rdx
    shl rdx, 4
    mov r14, qword ptr [rbx + rdx]
    mov r15, qword ptr [rbx + rdx + 8]
    mov r13, -1
    xor r8, r8
    mov r9, r12
    dec r9
.Lloop:
    cmp r8, r9
    jge .Lafter_loop
    mov r10, r8
    shl r10, 4
    mov rcx, qword ptr [rbx + r10]
    mov rdx, qword ptr [rbx + r10 + 8]
    cmp rcx, r14
    jl .Lswap
    jg .Lnoswap
    cmp rdx, r15
    jle .Lswap
    jmp .Lnoswap
.Lswap:
    inc r13
    mov rsi, r13
    shl rsi, 4
    mov rax, qword ptr [rbx + rsi]
    mov r11, qword ptr [rbx + rsi + 8]
    mov qword ptr [rbx + rsi], rcx
    mov qword ptr [rbx + rsi + 8], rdx
    mov qword ptr [rbx + r10], rax
    mov qword ptr [rbx + r10 + 8], r11
.Lnoswap:
    inc r8
    jmp .Lloop
.Lafter_loop:
    inc r13
    mov rsi, r13
    shl rsi, 4
    mov rdx, r12
    dec rdx
    shl rdx, 4
    mov rax, qword ptr [rbx + rsi]
    mov r11, qword ptr [rbx + rsi + 8]
    mov qword ptr [rbx + rsi], r14
    mov qword ptr [rbx + rsi + 8], r15
    mov qword ptr [rbx + rdx], rax
    mov qword ptr [rbx + rdx + 8], r11
    cmp r13, 0
    jle .Lskip_left
    mov rdi, rbx
    mov rsi, r13
    call qsort_pairs
.Lskip_left:
    mov rax, r12
    sub rax, r13
    dec rax
    cmp rax, 0
    jle .Lskip_right
    mov rdx, r13
    inc rdx
    shl rdx, 4
    lea rdi, [rbx + rdx]
    mov rsi, rax
    call qsort_pairs
.Lskip_right:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lreturn:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
