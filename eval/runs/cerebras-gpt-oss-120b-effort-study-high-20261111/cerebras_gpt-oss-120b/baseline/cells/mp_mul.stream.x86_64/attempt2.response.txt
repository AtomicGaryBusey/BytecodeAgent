mp_mul:
    push rbx
    push rbp
    push r12
    mov rbp, rdx
    mov rdx, rsi
    add rdx, rcx
    xor r9, r9
.Lzero_loop:
    cmp r9, rdx
    je .Lzero_done
    mov qword ptr [r8 + r9*8], 0
    inc r9
    jmp .Lzero_loop
.Lzero_done:
    xor r10, r10
.Louter_loop:
    cmp r10, rsi
    jae .Louter_done
    lea rbx, [r8 + r10*8]
    mov r12, qword ptr [rdi + r10*8]
    xor r11, r11
    xor r9, r9
.Linner_loop:
    cmp r9, rcx
    jae .Linner_done
    mov rax, qword ptr [rbp + r9*8]
    mul r12
    add rax, qword ptr [rbx + r9*8]
    adc rdx, 0
    add rax, r11
    adc rdx, 0
    mov qword ptr [rbx + r9*8], rax
    mov r11, rdx
    inc r9
    jmp .Linner_loop
.Linner_done:
    mov rax, r11
    add qword ptr [rbx + rcx*8], rax
    jnc .Louter_continue
    mov r9, rcx
.Lcarry_loop:
    inc r9
    adc qword ptr [rbx + r9*8], 0
    jnc .Louter_continue
    jmp .Lcarry_loop
.Louter_continue:
    inc r10
    jmp .Louter_loop
.Louter_done:
    mov rdx, rsi
    add rdx, rcx
    mov r9, rdx
.Lscan_loop:
    test r9, r9
    je .Lreturn_zero
    dec r9
    mov rax, qword ptr [r8 + r9*8]
    test rax, rax
    jne .Lfound_nonzero
    jmp .Lscan_loop
.Lfound_nonzero:
    inc r9
    mov rax, r9
    jmp .Lcleanup
.Lreturn_zero:
    xor rax, rax
.Lcleanup:
    pop r12
    pop rbp
    pop rbx
    ret
