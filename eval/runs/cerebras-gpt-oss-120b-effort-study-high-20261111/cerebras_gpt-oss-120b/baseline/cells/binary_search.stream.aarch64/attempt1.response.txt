binary_search:
    cbz x1, .Lreturn_minus_one
    mov x3, #0
    sub x4, x1, #1
.Lloop:
    cmp x3, x4
    b.gt .Lreturn_minus_one
    sub x5, x4, x3
    lsr x5, x5, #1
    add x6, x3, x5
    ldr x8, [x0, x6, lsl #3]
    cmp x8, x2
    b.eq .Lfound_mid
    b.lt .Lmid_lt_key
    sub x4, x6, #1
    b .Lloop
.Lmid_lt_key:
    add x3, x6, #1
    b .Lloop
.Lfound_mid:
    mov x0, x6
    ret
.Lreturn_minus_one:
    mov x0, #-1
    ret
