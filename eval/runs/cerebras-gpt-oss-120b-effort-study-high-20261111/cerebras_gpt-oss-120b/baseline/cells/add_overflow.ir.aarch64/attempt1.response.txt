export function l $add_overflow(l a, l b, l out)
{
@start
    %sum = addl %a, %b
    storel %sum, %out
    %axor = xorl %a, %sum
    %bxor = xorl %b, %sum
    %and = andl %axor, %bxor
    %zero = constl 0
    %cond = csltl %and, %zero
    jnz %cond, @overflow
    ret %zero
@overflow
    %one = constl 1
    ret %one
}
