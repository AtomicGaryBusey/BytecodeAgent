export function l $add_overflow(l %a, l %b, l %out)
{
@start
    %sum = add %a, %b
    storel %sum, %out
    %axor = xor %a, %sum
    %bxor = xor %b, %sum
    %and = and %axor, %bxor
    %zero = const 0
    %cond = cslt %and, %zero
    jnz %cond, @overflow
    ret %zero
@overflow
    %one = const 1
    ret %one
}
