export function l $bmin_empty(l a, l n) {
@entry
    %zero = constl 0
    %iszero = ceql n, %zero
    jnz %iszero, Lzero
    %min0 = loadl a
    %one = constl 1
    %sh3 = constl 3
    jmp Lcond

@Lzero
    %max = constl 9223372036854775807
    ret %max

@Lcond
    %i = phi l %one @entry, %i_next @Lbody
    %min = phi l %min0 @entry, %min_next @Lbody
    %cond = csltl %i, n
    jnz %cond, Lbody
    jmp Lexit

@Lbody
    %idx = shll %i, %sh3
    %addr = addl a, %idx
    %val = loadl %addr
    %lt = csltl %val, %min
    %min_next = sel %lt, %val, %min
    %i_next = addl %i, %one
    jmp Lcond

@Lexit
    ret %min
}
