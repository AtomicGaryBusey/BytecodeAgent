export function l $fill_down(l %a, l %n, l %start) {
@entry:
    %i0 = constl 0
    jmp @cond

@cond:
    %i = phi l @entry %i0, @body %i_next
    %cmp = csltl %i, %n
    jnz %cmp, @body, @done

@body:
    %i8 = mull %i, constl 8
    %addr = addl %a, %i8
    %val = subl %start, %i
    storel %val, %addr
    %i_next = addl %i, constl 1
    jmp @cond

@done:
    ret %n
}
