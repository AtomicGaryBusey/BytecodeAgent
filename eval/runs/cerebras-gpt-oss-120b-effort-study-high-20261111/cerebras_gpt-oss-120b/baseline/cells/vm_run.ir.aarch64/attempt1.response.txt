export function l $vm_run(l $code, l $len, l $stack, l $cap)
@entry
%pc0 = const $0
%sp0 = const $0
jmp @loop

@loop
%pc = phi %pc0, %push_pc, %add_pc, %sub_pc, %mul_pc, %dup_pc, %swap_pc, %drop_pc, %over_pc, %lt_pc, %jz0_pc, %jz1_pc, %jmp_pc, %default_pc
%sp = phi %sp0, %push_sp, %add_sp, %sub_sp, %mul_sp, %dup_sp, %swap_sp, %drop_sp, %over_sp, %lt_sp, %jz0_sp, %jz1_sp, %jmp_sp, %default_sp
%cond = csltl %pc, %len
jnz %cond, @dispatch, @after_loop

@dispatch
%op = loadl %code, %pc
%eq0 = ceql %op, const $0
jnz %eq0, @halt, @chk1

@chk1
%eq1 = ceql %op, const $1
jnz %eq1, @push, @chk2

@chk2
%eq2 = ceql %op, const $2
jnz %eq2, @add, @chk3

@chk3
%eq3 = ceql %op, const $3
jnz %eq3, @sub, @chk4

@chk4
%eq4 = ceql %op, const $4
jnz %eq4, @mul, @chk5

@chk5
%eq5 = ceql %op, const $5
jnz %eq5, @dup, @chk6

@chk6
%eq6 = ceql %op, const $6
jnz %eq6, @swap, @chk7

@chk7
%eq7 = ceql %op, const $7
jnz %eq7, @drop, @chk8

@chk8
%eq8 = ceql %op, const $8
jnz %eq8, @over, @chk9

@chk9
%eq9 = ceql %op, const $9
jnz %eq9, @lt, @chk10

@chk10
%eq10 = ceql %op, const $10
jnz %eq10, @jz, @chk11

@chk11
%eq11 = ceql %op, const $11
jnz %eq11, @jmp, @default

@halt
jmp @after_loop

@push
%push_operand_idx = addl %pc, const $1
%push_operand = loadl %code, %push_operand_idx
%push_pc = addl %pc, const $2
%push_sp = addl %sp, const $1
storel %push_operand, %stack, %sp
jmp @loop

@add
%add_sp2 = subl %sp, const $2
%add_a = loadl %stack, %add_sp2
%add_idx_b = addl %add_sp2, const $1
%add_b = loadl %stack, %add_idx_b
%add_sum = addl %add_a, %add_b
storel %add_sum, %stack, %add_sp2
%add_sp = addl %add_sp2, const $1
%add_pc = addl %pc, const $1
jmp @loop

@sub
%sub_sp2 = subl %sp, const $2
%sub_a = loadl %stack, %sub_sp2
%sub_idx_b = addl %sub_sp2, const $1
%sub_b = loadl %stack, %sub_idx_b
%sub_res = subl %sub_a, %sub_b
storel %sub_res, %stack, %sub_sp2
%sub_sp = addl %sub_sp2, const $1
%sub_pc = addl %pc, const $1
jmp @loop

@mul
%mul_sp2 = subl %sp, const $2
%mul_a = loadl %stack, %mul_sp2
%mul_idx_b = addl %mul_sp2, const $1
%mul_b = loadl %stack, %mul_idx_b
%mul_res = mull %mul_a, %mul_b
storel %mul_res, %stack, %mul_sp2
%mul_sp = addl %mul_sp2, const $1
%mul_pc = addl %pc, const $1
jmp @loop

@dup
%dup_top_idx = subl %sp, const $1
%dup_val = loadl %stack, %dup_top_idx
%dup_pc = addl %pc, const $1
%dup_sp = addl %sp, const $1
storel %dup_val, %stack, %sp
jmp @loop

@swap
%swap_idx_a = subl %sp, const $2
%swap_idx_b = subl %sp, const $1
%swap_a = loadl %stack, %swap_idx_a
%swap_b = loadl %stack, %swap_idx_b
%swap_pc = addl %pc, const $1
%swap_sp = %sp
storel %swap_b, %stack, %swap_idx_a
storel %swap_a, %stack, %swap_idx_b
jmp @loop

@drop
%drop_sp = subl %sp, const $1
%drop_pc = addl %pc, const $1
jmp @loop

@over
%over_src_idx = subl %sp, const $2
%over_val = loadl %stack, %over_src_idx
%over_pc = addl %pc, const $1
%over_sp = addl %sp, const $1
storel %over_val, %stack, %sp
jmp @loop

@lt
%lt_sp2 = subl %sp, const $2
%lt_a = loadl %stack, %lt_sp2
%lt_idx_b = addl %lt_sp2, const $1
%lt_b = loadl %stack, %lt_idx_b
%lt_cond = csltl %lt_a, %lt_b
%lt_cond_l = zextw %lt_cond
storel %lt_cond_l, %stack, %lt_sp2
%lt_sp = addl %lt_sp2, const $1
%lt_pc = addl %pc, const $1
jmp @loop

@jz
%jz_new_sp = subl %sp, const $1
%jz_val = loadl %stack, %jz_new_sp
%jz_cond = ceql %jz_val, const $0
%jz_target_idx = addl %pc, const $1
%jz_target = loadl %code, %jz_target_idx
%jz_pc_nonzero = addl %pc, const $2
jnz %jz_cond, @jz_zero, @jz_nonzero

@jz_zero
%jz0_pc = %jz_target
%jz0_sp = %jz_new_sp
jmp @loop

@jz_nonzero
%jz1_pc = %jz_pc_nonzero
%jz1_sp = %jz_new_sp
jmp @loop

@jmp
%jmp_target_idx = addl %pc, const $1
%jmp_target = loadl %code, %jmp_target_idx
%jmp_pc = %jmp_target
%jmp_sp = %sp
jmp @loop

@default
%default_pc = addl %pc, const $1
%default_sp = %sp
jmp @loop

@after_loop
jmp @zero_loop

@zero_loop
%i = phi %sp, %i_next
%cond_z = csltl %i, %cap
jnz %cond_z, @zero_body, @final

@zero_body
storel const $0, %stack, %i
%i_next = addl %i, const $1
jmp @zero_loop

@final
%sp_gt0 = csltl const $0, %sp
jnz %sp_gt0, @nonempty, @empty

@empty
ret const $0

@nonempty
%top_idx = subl %sp, const $1
%top_val = loadl %stack, %top_idx
ret %top_val
