export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    %pc0 = subl %len, %len
    %sp0 = subl %len, %len
    jmp @loop

@loop
    %pc = phi %pc0 @entry, %push_pc @push, %add_pc @add, %sub_pc @sub, %mul_pc @mul, %dup_pc @dup, %swap_pc @swap, %drop_pc @drop, %over_pc @over, %lt_pc @lt, %jz0_pc @jz_zero, %jz1_pc @jz_nonzero, %jmp_pc @jmp, %def_pc @default
    %sp = phi %sp0 @entry, %push_sp @push, %add_sp @add, %sub_sp @sub, %mul_sp @mul, %dup_sp @dup, %swap_sp @swap, %drop_sp @drop, %over_sp @over, %lt_sp @lt, %jz0_sp @jz_zero, %jz1_sp @jz_nonzero, %jmp_sp @jmp, %def_sp @default
    %cond = csltl %pc, %len
    jnz %cond, @dispatch, @after_loop

@dispatch
    %pc_off = mull %pc, $8
    %op = loadl %code, %pc_off
    %eq0 = ceql %op, $0
    jnz %eq0, @halt, @chk1

@chk1
    %eq1 = ceql %op, $1
    jnz %eq1, @push, @chk2

@chk2
    %eq2 = ceql %op, $2
    jnz %eq2, @add, @chk3

@chk3
    %eq3 = ceql %op, $3
    jnz %eq3, @sub, @chk4

@chk4
    %eq4 = ceql %op, $4
    jnz %eq4, @mul, @chk5

@chk5
    %eq5 = ceql %op, $5
    jnz %eq5, @dup, @chk6

@chk6
    %eq6 = ceql %op, $6
    jnz %eq6, @swap, @chk7

@chk7
    %eq7 = ceql %op, $7
    jnz %eq7, @drop, @chk8

@chk8
    %eq8 = ceql %op, $8
    jnz %eq8, @over, @chk9

@chk9
    %eq9 = ceql %op, $9
    jnz %eq9, @lt, @chk10

@chk10
    %eq10 = ceql %op, $10
    jnz %eq10, @jz, @chk11

@chk11
    %eq11 = ceql %op, $11
    jnz %eq11, @jmp, @default

@halt
    jmp @after_loop

@push
    %pc1 = addl %pc, $1
    %operand_off = mull %pc1, $8
    %operand = loadl %code, %operand_off
    %sp_off = mull %sp, $8
    storel %operand, %stack, %sp_off
    %push_sp = addl %sp, $1
    %push_pc = addl %pc, $2
    jmp @loop

@add
    %a_idx = subl %sp, $2
    %a_off = mull %a_idx, $8
    %a = loadl %stack, %a_off
    %b_idx = subl %sp, $1
    %b_off = mull %b_idx, $8
    %b = loadl %stack, %b_off
    %sum = addl %a, %b
    storel %sum, %stack, %a_off
    %add_sp = subl %sp, $1
    %add_pc = addl %pc, $1
    jmp @loop

@sub
    %a_idx = subl %sp, $2
    %a_off = mull %a_idx, $8
    %a = loadl %stack, %a_off
    %b_idx = subl %sp, $1
    %b_off = mull %b_idx, $8
    %b = loadl %stack, %b_off
    %diff = subl %a, %b
    storel %diff, %stack, %a_off
    %sub_sp = subl %sp, $1
    %sub_pc = addl %pc, $1
    jmp @loop

@mul
    %a_idx = subl %sp, $2
    %a_off = mull %a_idx, $8
    %a = loadl %stack, %a_off
    %b_idx = subl %sp, $1
    %b_off = mull %b_idx, $8
    %b = loadl %stack, %b_off
    %prod = mull %a, %b
    storel %prod, %stack, %a_off
    %mul_sp = subl %sp, $1
    %mul_pc = addl %pc, $1
    jmp @loop

@dup
    %top_idx = subl %sp, $1
    %top_off = mull %top_idx, $8
    %val = loadl %stack, %top_off
    %sp_off = mull %sp, $8
    storel %val, %stack, %sp_off
    %dup_sp = addl %sp, $1
    %dup_pc = addl %pc, $1
    jmp @loop

@swap
    %a_idx = subl %sp, $2
    %b_idx = subl %sp, $1
    %a_off = mull %a_idx, $8
    %b_off = mull %b_idx, $8
    %a_val = loadl %stack, %a_off
    %b_val = loadl %stack, %b_off
    storel %b_val, %stack, %a_off
    storel %a_val, %stack, %b_off
    %swap_sp = %sp
    %swap_pc = addl %pc, $1
    jmp @loop

@drop
    %drop_sp = subl %sp, $1
    %drop_pc = addl %pc, $1
    jmp @loop

@over
    %src_idx = subl %sp, $2
    %src_off = mull %src_idx, $8
    %val = loadl %stack, %src_off
    %sp_off = mull %sp, $8
    storel %val, %stack, %sp_off
    %over_sp = addl %sp, $1
    %over_pc = addl %pc, $1
    jmp @loop

@lt
    %sp2 = subl %sp, $2
    %a_off = mull %sp2, $8
    %a = loadl %stack, %a_off
    %b_idx = addl %sp2, $1
    %b_off = mull %b_idx, $8
    %b = loadl %stack, %b_off
    %cond_w = csltl %a, %b
    %cond_l = zextw %cond_w
    storel %cond_l, %stack, %a_off
    %lt_sp = addl %sp2, $1
    %lt_pc = addl %pc, $1
    jmp @loop

@jz
    %sp1 = subl %sp, $1
    %top_idx = subl %sp, $1
    %top_off = mull %top_idx, $8
    %val = loadl %stack, %top_off
    %cond = ceql %val, $0
    %pc1 = addl %pc, $1
    %t_off = mull %pc1, $8
    %t = loadl %code, %t_off
    %pc_nonzero = addl %pc, $2
    jnz %cond, @jz_zero, @jz_nonzero

@jz_zero
    %jz0_sp = %sp1
    %jz0_pc = %t
    jmp @loop

@jz_nonzero
    %jz1_sp = %sp1
    %jz1_pc = %pc_nonzero
    jmp @loop

@jmp
    %pc1 = addl %pc, $1
    %t_off = mull %pc1, $8
    %t = loadl %code, %t_off
    %jmp_sp = %sp
    %jmp_pc = %t
    jmp @loop

@default
    %def_sp = %sp
    %def_pc = addl %pc, $1
    jmp @loop

@after_loop
    jmp @zero_loop

@zero_loop
    %i = phi %sp @after_loop, %i_next @zero_body
    %cond_z = csltl %i, %cap
    jnz %cond_z, @zero_body, @final

@zero_body
    %zero_const = subl %sp, %sp
    %off = mull %i, $8
    storel %zero_const, %stack, %off
    %i_next = addl %i, $1
    jmp @zero_loop

@final
    %zero2 = subl %sp, %sp
    %nonempty_cond = csltl %zero2, %sp
    jnz %nonempty_cond, @nonempty, @empty

@empty
    ret %zero2

@nonempty
    %top_idx = subl %sp, $1
    %top_off = mull %top_idx, $8
    %top_val = loadl %stack, %top_off
    ret %top_val
}
