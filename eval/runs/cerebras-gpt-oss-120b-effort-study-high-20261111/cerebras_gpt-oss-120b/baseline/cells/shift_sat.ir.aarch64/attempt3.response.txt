export function l $shift_sat(l %x, l %n) {
@entry:
    %cond = csltl %n, 64
    jnz %cond, @normal
    jmp @big

@normal:
    %shl = shll %x, %n
    %sar = sarl %x, %n
    jmp @join, %shl, %sar

@big:
    %sar_big = sarl %x, 63
    jmp @join, 0, %sar_big

@join(l %shl, l %sar):
    %sum = addl %shl, %sar
    ret %sum
}
