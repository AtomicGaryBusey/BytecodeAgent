lcs_len:
    // Move arguments to working registers
    mov x9, x0          // x pointer
    mov x10, x1         // m
    mov x11, x2         // y pointer
    mov x12, x3         // n
    mov x13, x4         // row pointer

    // Zero row[0..n]
    mov x0, #0          // constant zero
    mov x14, #0         // index j = 0
.Lzero_loop:
    cmp x14, x12
    bgt .Lzero_done
    str x0, [x13, x14, LSL #3]
    add x14, x14, #1
    b .Lzero_loop
.Lzero_done:

    // Outer loop over i (rows)
    mov x8, x10         // outer counter = m
.Louter:
    cbz x8, .Louter_done

    // Load x[i] byte and advance x pointer
    ldrb w5, [x9], #1   // xi in w5

    // Reset diag and left to 0
    mov x4, #0          // diag
    mov x7, #0          // left

    // Reset inner pointers and indices
    mov x16, x11        // y pointer
    mov x14, #1         // j = 1

.Linner:
    cmp x14, x12
    bgt .Linner_done

    // up = row[j]
    ldr x15, [x13, x14, LSL #3]

    // Load y byte and advance pointer
    ldrb w6, [x16], #1

    // max = max(up, left)
    cmp x15, x7
    csel x2, x15, x7, gt

    // diag+1 candidate
    add x1, x4, #1

    // Choose result based on equality
    cmp w5, w6
    csel x3, x1, x2, eq

    // Store result
    str x3, [x13, x14, LSL #3]

    // Update diag and left for next iteration
    mov x4, x15
    mov x7, x3

    // Increment j
    add x14, x14, #1
    b .Linner
.Linner_done:
    // Decrement outer counter and loop
    sub x8, x8, #1
    b .Louter
.Louter_done:
    // Return row[n]
    ldr x0, [x13, x12, LSL #3]
    ret
