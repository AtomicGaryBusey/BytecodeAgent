stats_f:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    mov x20, x0          // a base pointer
    mov x21, x2          // function pointer f
    mov x22, x3          // out pointer
    mov x27, x1          // n (count)
    mov x19, #0          // i = 0
    mov x26, #0          // odd count = 0

    // first element (i = 0)
    ldr x0, [x20]        // a[0]
    blr x21
    mov x23, x0          // min = y0
    mov x24, x0          // max = y0
    mov x25, x0          // sum = y0
    and x9, x0, #1
    cbz x9, .Lskip_od0
    add x26, x26, #1
.Lskip_od0:
    add x19, x19, #1

.Lloop:
    cmp x19, x27
    bge .Lfinish

    ldr x0, [x20, x19, lsl #3]   // a[i]
    blr x21
    add x25, x25, x0             // sum += y
    cmp x0, x23
    csel x23, x0, x23, lt       // min = y if y < min
    cmp x0, x24
    csel x24, x0, x24, gt       // max = y if y > max
    and x9, x0, #1
    cbz x9, .Lskip_od
    add x26, x26, #1
.Lskip_od:
    add x19, x19, #1
    b .Lloop

.Lfinish:
    str x23, [x22]          // out[0] = min
    str x24, [x22, #8]      // out[1] = max
    str x25, [x22, #16]     // out[2] = sum
    mov x0, x26

    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
