export function l $packbits_decode(l src, l n, l out, l cap) {
    %src_end = addl src, n
    %out_end = addl out, cap
    jmp @loop src, out, 0, %src_end, %out_end

@loop(%src, %out, %written, %src_end, %out_end):
    %eof =w ceql %src, %src_end
    jnz %eof, @ret_written %written
    %h =w loadsb %src
    %src1 = addl %src, 1
    %is_nop =w ceql %h, -128
    jnz %is_nop, @loop %src1, %out, %written, %src_end, %out_end
    %is_literal =w csgel %h, 0
    jnz %is_literal, @literal %src1, %out, %written, %src_end, %out_end, %h
    jmp @repeat %src1, %out, %written, %src_end, %out_end, %h

@literal(%src, %out, %written, %src_end, %out_end, %h):
    %lit_cnt =l addl %h, 1
    %src_need_end = addl %src, %lit_cnt
    %src_ok =w csgel %src_end, %src_need_end
    jnz %src_ok, @literal_src_ok %src, %out, %written, %src_end, %out_end, %lit_cnt
    jmp @error_minus2

@literal_src_ok(%src, %out, %written, %src_end, %out_end, %lit_cnt):
    %out_need_end = addl %out, %lit_cnt
    %out_ok =w csgel %out_end, %out_need_end
    jnz %out_ok, @copy_literal_all %src, %out, %written, %src_end, %out_end, %lit_cnt
    %out_rem = subl %out_end, %out
    %out_rem_zero =w ceql %out_rem, 0
    jnz %out_rem_zero, @error_minus1
    jmp @copy_literal_partial %src, %out, %out_rem, %written

@copy_literal_all(%src_cur, %out_cur, %written, %src_end, %out_end, %cnt):
    %cnt_zero =w ceql %cnt, 0
    jnz %cnt_zero, @loop %src_cur, %out_cur, %written, %src_end, %out_end
    %byte =w loadub %src_cur
    storeb %byte, %out_cur
    %src_next = addl %src_cur, 1
    %out_next = addl %out_cur, 1
    %cnt_next = subl %cnt, 1
    %written_next = addl %written, 1
    jmp @copy_literal_all %src_next, %out_next, %written_next, %src_end, %out_end, %cnt_next

@copy_literal_partial(%src_cur, %out_cur, %cnt, %written):
    %cnt_zero =w ceql %cnt, 0
    jnz %cnt_zero, @error_minus1
    %byte =w loadub %src_cur
    storeb %byte, %out_cur
    %src_next = addl %src_cur, 1
    %out_next = addl %out_cur, 1
    %cnt_next = subl %cnt, 1
    jmp @copy_literal_partial %src_next, %out_next, %cnt_next, %written

@repeat(%src, %out, %written, %src_end, %out_end, %h):
    %src_eq =w ceql %src, %src_end
    jnz %src_eq, @error_minus2
    %rep_byte =w loadub %src
    %src_next = addl %src, 1
    %rep_cnt =l subl 1, %h
    %out_need_end = addl %out, %rep_cnt
    %out_ok =w csgel %out_end, %out_need_end
    jnz %out_ok, @repeat_write_all %src_next, %out, %written, %src_end, %out_end, %rep_byte, %rep_cnt
    %out_rem = subl %out_end, %out
    %out_rem_zero =w ceql %out_rem, 0
    jnz %out_rem_zero, @error_minus1
    jmp @repeat_write_partial %src_next, %out, %written, %src_end, %out_end, %rep_byte, %out_rem

@repeat_write_all(%src, %out_cur, %written, %src_end, %out_end, %rep_byte, %cnt):
    %cnt_zero =w ceql %cnt, 0
    jnz %cnt_zero, @loop %src, %out_cur, %written, %src_end, %out_end
    storeb %rep_byte, %out_cur
    %out_next = addl %out_cur, 1
    %cnt_next = subl %cnt, 1
    %written_next = addl %written, 1
    jmp @repeat_write_all %src, %out_next, %written_next, %src_end, %out_end, %rep_byte, %cnt_next

@repeat_write_partial(%src, %out_cur, %written, %src_end, %out_end, %rep_byte, %cnt):
    %cnt_zero =w ceql %cnt, 0
    jnz %cnt_zero, @error_minus1
    storeb %rep_byte, %out_cur
    %out_next = addl %out_cur, 1
    %cnt_next = subl %cnt, 1
    jmp @repeat_write_partial %src, %out_next, %written, %src_end, %out_end, %rep_byte, %cnt_next

@ret_written(%written):
    ret %written

@error_minus2:
    ret -2

@error_minus1:
    ret -1
}
