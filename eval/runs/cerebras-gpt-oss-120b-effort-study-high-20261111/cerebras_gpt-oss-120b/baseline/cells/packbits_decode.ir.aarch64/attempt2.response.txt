export function l $packbits_decode(l src, l n, l out, l cap) {
%src_end = addl src, n
%out_end = addl out, cap
%written = 0
jmp @loop src, out, written, src_end, out_end

@loop(%src, %out, %written, %src_end, %out_end):
%src_eq_end =w ceql %src, %src_end
jnz %src_eq_end, @ret_written %written
%h =w loadsb %src
%src1 = addl %src, 1
%is_nop =w ceql %h, -128
jnz %is_nop, @loop %src1, %out, %written, %src_end, %out_end
%is_literal =w csgel %h, 0
jnz %is_literal, @literal %src1, %out, %written, %src_end, %out_end, %h
jmp @repeat %src1, %out, %written, %src_end, %out_end, %h

@literal(%src, %out, %written, %src_end, %out_end, %h):
%h_l = sextl %h
%cnt = addl %h_l, 1
%src_need_end = addl %src, %cnt
%src_ok =w csgel %src_end, %src_need_end
jnz %src_ok, @literal_have_src %src, %out, %written, %src_end, %out_end, %cnt
ret -2

@literal_have_src(%src, %out, %written, %src_end, %out_end, %cnt):
%out_need_end = addl %out, %cnt
%out_ok =w csgel %out_end, %out_need_end
jnz %out_ok, @copy_literal_all %src, %out, %written, %src_end, %out_end, %cnt
%out_rem = subl %out_end, %out
%out_rem_zero =w ceql %out_rem, 0
jnz %out_rem_zero, @error_minus1
jmp @copy_literal_partial %src, %out, %written, %src_end, %out_end, %out_rem

@copy_literal_all(%src, %out, %written, %src_end, %out_end, %cnt):
%cnt_zero =w ceql %cnt, 0
jnz %cnt_zero, @loop %src, %out, %written, %src_end, %out_end
%byte =w loadub %src
storeb %byte, %out
%src_next = addl %src, 1
%out_next = addl %out, 1
%cnt_next = subl %cnt, 1
%written_next = addl %written, 1
jmp @copy_literal_all %src_next, %out_next, %written_next, %src_end, %out_end, %cnt_next

@copy_literal_partial(%src, %out, %written, %src_end, %out_end, %cnt):
%cnt_zero =w ceql %cnt, 0
jnz %cnt_zero, @error_minus1
%byte =w loadub %src
storeb %byte, %out
%src_next = addl %src, 1
%out_next = addl %out, 1
%cnt_next = subl %cnt, 1
%written_next = addl %written, 1
jmp @copy_literal_partial %src_next, %out_next, %written_next, %src_end, %out_end, %cnt_next

@repeat(%src, %out, %written, %src_end, %out_end, %h):
%src_eq_end =w ceql %src, %src_end
jnz %src_eq_end, @error_minus2
%rep_byte =w loadub %src
%src_next = addl %src, 1
%rep_cnt_w = subl 1, %h
%rep_cnt = sextl %rep_cnt_w
%out_need_end = addl %out, %rep_cnt
%out_ok =w csgel %out_end, %out_need_end
jnz %out_ok, @repeat_write_all %src_next, %out, %written, %src_end, %out_end, %rep_byte, %rep_cnt
%out_rem = subl %out_end, %out
%out_rem_zero =w ceql %out_rem, 0
jnz %out_rem_zero, @error_minus1
jmp @repeat_write_partial %src_next, %out, %written, %src_end, %out_end, %rep_byte, %out_rem

@repeat_write_all(%src, %out, %written, %src_end, %out_end, %rep_byte, %cnt):
%cnt_zero =w ceql %cnt, 0
jnz %cnt_zero, @loop %src, %out, %written, %src_end, %out_end
storeb %rep_byte, %out
%out_next = addl %out, 1
%cnt_next = subl %cnt, 1
%written_next = addl %written, 1
jmp @repeat_write_all %src, %out_next, %written_next, %src_end, %out_end, %rep_byte, %cnt_next

@repeat_write_partial(%src, %out, %written, %src_end, %out_end, %rep_byte, %cnt):
%cnt_zero =w ceql %cnt, 0
jnz %cnt_zero, @error_minus1
storeb %rep_byte, %out
%out_next = addl %out, 1
%cnt_next = subl %cnt, 1
%written_next = addl %written, 1
jmp @repeat_write_partial %src, %out_next, %written_next, %src_end, %out_end, %rep_byte, %cnt_next

@ret_written(%written):
ret %written

@error_minus2:
ret -2

@error_minus1:
ret -1
}
