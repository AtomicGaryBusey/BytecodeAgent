export function l $packbits_decode(l, l, l, l) {
@entry(%src, %n, %out, %cap):
    %src_end =l add %src, %n
    %out_end =l add %out, %cap
    jmp @loop %src, %out, 0, %src_end, %out_end

@loop(%src, %out, %written, %src_end, %out_end):
    %src_eq_end =w ceq %src, %src_end
    jnz %src_eq_end, @ret_written %written
    %h =w loadsb %src
    %src1 =l add %src, 1
    %is_nop =w ceq %h, -128
    jnz %is_nop, @loop %src1, %out, %written, %src_end, %out_end
    %is_literal =w csgt %h, -1
    jnz %is_literal, @literal %src1, %out, %written, %src_end, %out_end, %h
    jmp @repeat %src1, %out, %written, %src_end, %out_end, %h

@literal(%src, %out, %written, %src_end, %out_end, %h):
    %cnt_w =w add %h, 1
    %cnt_l =l sext %cnt_w
    %src_rem =l sub %src_end, %src
    %src_ok =w csge %src_rem, %cnt_l
    jnz %src_ok, @literal_src_ok %src, %out, %written, %src_end, %out_end, %cnt_l
    ret -2

@literal_src_ok(%src, %out, %written, %src_end, %out_end, %cnt):
    %out_rem =l sub %out_end, %out
    %out_ok =w csge %out_rem, %cnt
    jnz %out_ok, @copy_literal_full %src, %out, %written, %src_end, %out_end, %cnt
    %out_rem_zero =w ceq %out_rem, 0
    jnz %out_rem_zero, @error_minus1
    jmp @copy_literal_partial %src, %out, %written, %src_end, %out_end, %out_rem

@copy_literal_full(%src, %out, %written, %src_end, %out_end, %cnt):
    %cnt_zero =w ceq %cnt, 0
    jnz %cnt_zero, @loop %src, %out, %written, %src_end, %out_end
    %byte =w loadub %src
    storeb %byte, %out
    %src_next =l add %src, 1
    %out_next =l add %out, 1
    %written_next =l add %written, 1
    %cnt_next =l sub %cnt, 1
    jmp @copy_literal_full %src_next, %out_next, %written_next, %src_end, %out_end, %cnt_next

@copy_literal_partial(%src, %out, %written, %src_end, %out_end, %cnt):
    %cnt_zero =w ceq %cnt, 0
    jnz %cnt_zero, @error_minus1
    %byte =w loadub %src
    storeb %byte, %out
    %src_next =l add %src, 1
    %out_next =l add %out, 1
    %written_next =l add %written, 1
    %cnt_next =l sub %cnt, 1
    jmp @copy_literal_partial %src_next, %out_next, %written_next, %src_end, %out_end, %cnt_next

@repeat(%src, %out, %written, %src_end, %out_end, %h):
    %src_eq_end =w ceq %src, %src_end
    jnz %src_eq_end, @error_minus2
    %rep_byte =w loadub %src
    %src_next =l add %src, 1
    %rep_cnt_w =w sub 1, %h
    %rep_cnt =l sext %rep_cnt_w
    %out_rem =l sub %out_end, %out
    %out_ok =w csge %out_rem, %rep_cnt
    jnz %out_ok, @repeat_write_all %out, %written, %rep_cnt, %rep_byte, %src_next, %src_end, %out_end
    %out_rem_zero =w ceq %out_rem, 0
    jnz %out_rem_zero, @error_minus1
    jmp @repeat_write_partial %out, %written, %out_rem, %rep_byte, %src_next, %src_end, %out_end

@repeat_write_all(%out, %written, %cnt, %rep_byte, %src, %src_end, %out_end):
    %cnt_zero =w ceq %cnt, 0
    jnz %cnt_zero, @loop %src, %out, %written, %src_end, %out_end
    storeb %rep_byte, %out
    %out_next =l add %out, 1
    %written_next =l add %written, 1
    %cnt_next =l sub %cnt, 1
    jmp @repeat_write_all %out_next, %written_next, %cnt_next, %rep_byte, %src, %src_end, %out_end

@repeat_write_partial(%out, %written, %cnt, %rep_byte, %src, %src_end, %out_end):
    %cnt_zero =w ceq %cnt, 0
    jnz %cnt_zero, @error_minus1
    storeb %rep_byte, %out
    %out_next =l add %out, 1
    %written_next =l add %written, 1
    %cnt_next =l sub %cnt, 1
    jmp @repeat_write_partial %out_next, %written_next, %cnt_next, %rep_byte, %src, %src_end, %out_end

@ret_written(%written):
    ret %written

@error_minus2:
    ret -2

@error_minus1:
    ret -1
}
