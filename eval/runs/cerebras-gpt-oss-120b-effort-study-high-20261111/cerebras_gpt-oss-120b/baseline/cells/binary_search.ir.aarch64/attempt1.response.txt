export function l $binary_search(l a, l n, l key) {
@entry
    %zero = const 0
    %one = const 1
    %minus_one = const -1
    %three = const 3
    %high0 = subl %n, %one
    jmp @loop

@loop
    %low = phi %zero, %low_branch_low, %low_branch_high
    %high = phi %high0, %high_branch_low, %high_branch_high
    %cond = csgtl %low, %high
    jnz %cond, @exit
    %diff = subl %high, %low
    %half = shrl %diff, %one
    %mid = addl %low, %half
    %offset = shll %mid, %three
    %addr = addl %a, %offset
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @found
    %lt = csltl %key, %val
    jnz %lt, @branch_low
    jmp @branch_high

@branch_low
    %high_branch_low = subl %mid, %one
    %low_branch_low = copy %low
    jmp @loop

@branch_high
    %low_branch_high = addl %mid, %one
    %high_branch_high = copy %high
    jmp @loop

@found
    ret %mid

@exit
    ret %minus_one
}
