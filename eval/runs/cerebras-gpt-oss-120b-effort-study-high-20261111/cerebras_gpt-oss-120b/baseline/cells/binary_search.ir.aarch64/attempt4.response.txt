export function l $binary_search(l, l, l) {
@entry:
    %a = arg 0
    %n = arg 1
    %key = arg 2
    %zero = const 0
    %one = const 1
    %size = const 8
    %two = const 2
    %high0 = subl %n, %one
    jmp @loop, %zero, %high0

@loop:
    %low = phi %zero, %low, %mid_plus_one
    %high = phi %high0, %mid_minus_one, %high
    %cond = csltl %high, %low
    jnz %cond, @exit
    %diff = subl %high, %low
    %half = sdivl %diff, %two
    %mid = addl %low, %half
    %offset = mull %mid, %size
    %addr = addl %a, %offset
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @found
    %lt = csltl %key, %val
    jnz %lt, @branch_low
    jmp @branch_high

@branch_low:
    %mid_minus_one = subl %mid, %one
    jmp @loop, %low, %mid_minus_one

@branch_high:
    %mid_plus_one = addl %mid, %one
    jmp @loop, %mid_plus_one, %high

@found:
    ret %mid

@exit:
    %minus_one = const -1
    ret %minus_one
}
