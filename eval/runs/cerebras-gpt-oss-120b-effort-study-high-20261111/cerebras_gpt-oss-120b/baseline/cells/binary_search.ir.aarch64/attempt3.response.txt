export function l $binary_search(l a, l n, l key) {
@entry:
    %zero = const 0
    %one = const 1
    %minus_one = const -1
    %size = const 8
    %two = const 2
    %high0 = subl %n, %one
    jmp @loop

@loop:
    %low = phi %zero, %low_branch_low, %low_branch_high
    %high = phi %high0, %high_branch_low, %high_branch_high
    %cond = csltl %high, %low
    jnz %cond, @exit
    %diff = subl %high, %low
    %half = sdivl %diff, %two
    %mid = addl %low, %half
    %offset = mull %mid, %size
    %addr = addl %a, %offset
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @found
    %lt = csltl %key, %val
    jnz %lt, @branch_low
    jmp @branch_high

@branch_low:
    %mid_minus_one = subl %mid, %one
    %high_branch_low = copy %mid_minus_one
    %low_branch_low = copy %low
    jmp @loop

@branch_high:
    %mid_plus_one = addl %mid, %one
    %low_branch_high = copy %mid_plus_one
    %high_branch_high = copy %high
    jmp @loop

@found:
    ret %mid

@exit:
    ret %minus_one
}
