packbits_encode:
    test rsi, rsi
    jz .Lreturn0
    mov r8, rdx                # out_base

.Lmain_loop:
    test rsi, rsi
    jz .Ldone

    mov r11b, byte ptr [rdi]   # repeat_byte
    mov r9, 1                  # run_len = 1

.Lrun_check:
    cmp r9, 128
    jae .Lrun_done
    cmp r9, rsi
    jae .Lrun_done
    mov al, byte ptr [rdi + r9]
    cmp al, r11b
    jne .Lrun_done
    inc r9
    jmp .Lrun_check

.Lrun_done:
    cmp r9, 3
    jb .Lliteral_start

    cmp rcx, 2
    jb .Lreturn_error
    mov eax, 257
    sub eax, r9d               # header = 257 - run_len
    mov byte ptr [rdx], al
    mov byte ptr [rdx+1], r11b
    add rdx, 2
    sub rcx, 2
    add rdi, r9
    sub rsi, r9
    jmp .Lmain_loop

.Lliteral_start:
    xor r9, r9                 # literal_len = 0

.Lliteral_scan:
    cmp r9, 128
    jae .Lliteral_done
    cmp r9, rsi
    je .Lliteral_done

    mov r10, rsi
    sub r10, r9
    cmp r10, 3
    jl .Lliteral_extend

    mov al, byte ptr [rdi + r9]
    cmp al, byte ptr [rdi + r9 + 1]
    jne .Lliteral_extend
    cmp al, byte ptr [rdi + r9 + 2]
    jne .Lliteral_extend
    jmp .Lliteral_done

.Lliteral_extend:
    inc r9
    jmp .Lliteral_scan

.Lliteral_done:
    mov r10, r9                # r10 = literal_len
    inc r10                    # r10 = literal_len + 1
    cmp rcx, r10
    jb .Lreturn_error

    mov al, r9b
    dec al                     # header = literal_len - 1
    mov byte ptr [rdx], al
    inc rdx                    # advance past header

    mov r11, r9                # copy counter = literal_len
.Lcopy_literal:
    test r11, r11
    jz .Lafter_copy_literal
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec r11
    jmp .Lcopy_literal

.Lafter_copy_literal:
    sub rcx, r10
    sub rsi, r9
    jmp .Lmain_loop

.Ldone:
    mov rax, rdx
    sub rax, r8
    ret

.Lreturn_error:
    mov rax, -1
    ret

.Lreturn0:
    xor eax, eax
    ret
