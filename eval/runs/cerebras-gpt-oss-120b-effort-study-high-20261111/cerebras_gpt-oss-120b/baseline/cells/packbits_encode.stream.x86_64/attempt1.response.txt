packbits_encode:
    test rsi, rsi
    jz .Lreturn0
    mov r10, rdx            # out_base
    mov r8, rcx             # remaining capacity
.Lmain:
    test rsi, rsi
    jz .Ldone
    mov al, byte ptr [rdi]          # load current byte
    mov r9, 1                       # run_len = 1
.Lrun_loop:
    cmp r9, 128
    jae .Lrun_done
    cmp r9, rsi
    jae .Lrun_done
    mov bl, byte ptr [rdi + r9]
    cmp bl, al
    jne .Lrun_done
    inc r9
    jmp .Lrun_loop
.Lrun_done:
    cmp r9, 3
    jl .Lliteral_scan
    mov r11b, al                # preserve byte value for repeat
    mov eax, 257
    sub eax, r9d                # header = 257 - run_len
    cmp r8, 2
    jb .Lreturn_error
    mov byte ptr [rdx], al
    mov byte ptr [rdx + 1], r11b
    add rdx, 2
    sub r8, 2
    add rdi, r9
    sub rsi, r9
    jmp .Lmain
.Lliteral_scan:
    xor rcx, rcx                # literal_len = 0
.Lliteral_loop:
    cmp rcx, 128
    jae .Lliteral_done
    cmp rcx, rsi
    je .Lliteral_done
    mov rax, rsi
    sub rax, rcx
    cmp rax, 3
    jl .Linclude_current
    mov bl, byte ptr [rdi + rcx]
    mov al, byte ptr [rdi + rcx + 1]
    cmp al, bl
    jne .Linclude_current
    mov al, byte ptr [rdi + rcx + 2]
    cmp al, bl
    jne .Linclude_current
    jmp .Lliteral_done
.Linclude_current:
    inc rcx
    jmp .Lliteral_loop
.Lliteral_done:
    mov rax, rcx
    inc rax                     # packet size = L+1
    cmp r8, rax
    jb .Lreturn_error
    mov al, cl                  # header = L-1
    dec al
    mov byte ptr [rdx], al
    lea r11, [rdx + 1]          # dest pointer for literal bytes
    mov r9, rcx                 # copy count = L
.Lcopy_loop:
    test r9, r9
    jz .Lafter_copy
    mov al, byte ptr [rdi]
    mov byte ptr [r11], al
    inc rdi
    inc r11
    dec r9
    jmp .Lcopy_loop
.Lafter_copy:
    add rdx, rcx
    inc rdx
    sub r8, rcx
    dec r8
    sub rsi, rcx
    jmp .Lmain
.Ldone:
    mov rax, rdx
    sub rax, r10
    ret
.Lreturn_error:
    mov rax, -1
    ret
.Lreturn0:
    xor eax, eax
    ret
