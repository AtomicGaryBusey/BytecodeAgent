export function l $crc32(l %p, l %n) {
@entry
%i =l add 0, 0
%crc =w xor 0, -1
jmp @outer_cond

@outer_cond
%cond =w csltl %i, %n
jnz %cond, @outer_body, @after_outer

@outer_body
%addr =l add %p, %i
%byte =w loadub %addr
%crc =w xor %crc, %byte
%bitcnt =w add 0, 8
jmp @inner_cond

@inner_cond
jnz %bitcnt, @inner_body, @after_inner

@inner_body
%lsb =w and %crc, 1
jnz %lsb, @bit_set, @bit_clear

@bit_set
%crc =w shr %crc, 1
%crc =w xor %crc, -306674912
jmp @inner_cont

@bit_clear
%crc =w shr %crc, 1
jmp @inner_cont

@inner_cont
%bitcnt =w sub %bitcnt, 1
jmp @inner_cond

@after_inner
%i =l add %i, 1
jmp @outer_cond

@after_outer
%crc =w xor %crc, -1
%retval =l extuw %crc
ret %retval
}
