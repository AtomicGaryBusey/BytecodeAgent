export function l $crc32(l %p, l %n) {
@entry
%one_l =l iconst 1
%one_w =w iconst 1
%poly =w iconst 0xedb88320
%xor_val =w iconst 0xffffffff
%crc =w iconst 0xffffffff
%i =l iconst 0
jmp @outer_cond

@outer_cond
%cond =w csltl %i, %n
jnz %cond, @outer_body, @after_outer

@outer_body
%addr =l add %p, %i
%byte =w loadub %addr
%crc =w xor %crc, %byte
%bitcnt =w iconst 8
jmp @inner_cond

@inner_cond
jnz %bitcnt, @inner_body, @after_inner

@inner_body
%lsb =w and %crc, %one_w
jnz %lsb, @bit_set, @bit_clear

@bit_set
%tmp =w shr %crc, %one_w
%crc =w xor %tmp, %poly
jmp @inner_cont

@bit_clear
%crc =w shr %crc, %one_w
jmp @inner_cont

@inner_cont
%bitcnt =w sub %bitcnt, %one_w
jmp @inner_cond

@after_inner
%i =l add %i, %one_l
jmp @outer_cond

@after_outer
%crc =w xor %crc, %xor_val
%retval =l extuw %crc
ret %retval
}
