export function l $crc32(l %p, l %n) {
@entry
%poly =w const 0xEDB88320
%init =w const 0xffffffff
%xor =w const 0xffffffff
%onew =w const 1
%eight =w const 8
%c1l =l const 1
%crc =w %init
%i =l const 0
jmp @outer_cond

@outer_cond
%cond =l csltl %i, %n
jnz %cond, @outer_body, @after_outer

@outer_body
%addr =l add %p, %i
%byte =w loadub %addr
%crc =w xor %crc, %byte
%bitcnt =w %eight
jmp @inner_cond

@inner_cond
jnz %bitcnt, @inner_body, @after_inner

@inner_body
%lsb =w and %crc, %onew
jnz %lsb, @bit_set, @bit_clear

@bit_set
%tmp =w shr %crc, %onew
%crc =w xor %tmp, %poly
jmp @inner_cont

@bit_clear
%crc =w shr %crc, %onew
jmp @inner_cont

@inner_cont
%bitcnt =w sub %bitcnt, %onew
jmp @inner_cond

@after_inner
%i =l add %i, %c1l
jmp @outer_cond

@after_outer
%crc =w xor %crc, %xor
%retval =l extuw %crc
ret %retval
}
