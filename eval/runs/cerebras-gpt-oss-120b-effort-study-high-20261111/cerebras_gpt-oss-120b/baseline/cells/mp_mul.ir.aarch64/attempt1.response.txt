export function l $mp_mul(l a, l na, l b, l nb, l r) {
    %len = add %na, %nb
    jmp @zero_loop 0 %len

    @zero_loop %i %len {
        %cond = csltl %i, %len
        jnz %cond, @zero_body %i %len
        jmp @zero_done %len
    }

    @zero_body %i %len {
        %off = shl %i, 3
        %addr = add %r, %off
        storel 0, %addr
        %i_next = add %i, 1
        jmp @zero_loop %i_next %len
    }

    @zero_done %len {
        jmp @outer_loop 0 %len
    }

    @outer_loop %i %len {
        %cond = csltl %i, %na
        jnz %cond, @outer_body %i %len
        jmp @after_outer %len
    }

    @outer_body %i %len {
        %off_ai = shl %i, 3
        %aptr = add %a, %off_ai
        %ai = loadl %aptr
        jmp @inner_loop %i 0 %ai 0 %len
    }

    @inner_loop %i %j %ai %carry %len {
        %cond_j = csltl %j, %nb
        jnz %cond_j, @inner_body %i %j %ai %carry %len
        jmp @inner_done %i %carry %len
    }

    @inner_body
