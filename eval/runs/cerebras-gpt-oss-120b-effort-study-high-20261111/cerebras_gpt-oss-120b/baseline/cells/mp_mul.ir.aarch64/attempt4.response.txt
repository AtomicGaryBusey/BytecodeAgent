export function l $mp_mul(l a, l na, l b, l nb, l r) {
    %zero =l const 0
    %one =l const 1
    %mask32 =l const 4294967295

    %len =l add na, nb

    %i =l const 0
@zero_loop:
    %c_i_len =w csltl %i, %len
    jnz %c_i_len, @zero_body
    jmp @zero_done
@zero_body:
    %i8 =l shl %i, 3
    %addr =l add r, %i8
    storel %zero, %addr
    %i =l add %i, %one
    jmp @zero_loop
@zero_done:

    %i =l const 0
@outer:
    %c_i_na =w csltl %i, na
    jnz %c_i_na, @outer_body
    jmp @outer_done
@outer_body:
    %i8a =l shl %i, 3
    %aptr =l add a, %i8a
    %aval =l loadl %aptr

    %carry =l const 0

    %j =l const 0
@inner:
    %c_j_nb =w csltl %j, nb
    jnz %c_j_nb, @inner_body
    jmp @inner_done
@inner_body:
    %j8b =l shl %j, 3
    %bptr =l add b, %j8b
    %bval =l loadl %bptr

    %a0 =l and %aval, %mask32
    %a1 =l shr %aval, 32
    %b0 =l and %bval, %mask32
    %b1 =l shr %bval, 32

    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1

    %p1_low =l and %p1, %mask32
    %p2_low =l and %p2, %mask32
    %p1_low_sh =l shl %p1_low, 32
    %p2_low_sh =l shl %p2_low, 32

    %sum0 =l add %p0, %p1_low_sh
    %c0 =w csltl %sum0, %p0
    %sum1 =l add %sum0, %p2_low_sh
    %c1 =w csltl %sum1, %sum0
    %prod_low =l copy %sum1

    %c0_l =l zextw %c0
    %c1_l =l zextw %c1
    %carry_low =l add %c0_l, %c1_l

    %p1_high =l shr %p1, 32
    %p2_high =l shr %p2, 32
    %tmp_h =l add %p3, %p1_high
    %tmp_h =l add %tmp_h, %p2_high
    %prod_high =l add %tmp_h, %carry_low

    %k =l add %i, %j
    %k8 =l shl %k, 3
    %raddr =l add r, %k8
    %rval =l loadl %raddr

    %sum_a =l add %rval, %prod_low
    %c2 =w csltl %sum_a, %rval
    %sum_b =l add %sum_a, %carry
    %c3 =w csltl %sum_b, %sum_a

    storel %sum_b, %raddr

    %c2_l =l zextw %c2
    %c3_l =l zextw %c3
    %c_sum =l add %c2_l, %c3_l
    %carry =l add %prod_high, %c_sum

    %j =l add %j, %one
    jmp @inner
@inner_done:
    %carry_idx =l add %i, nb
    %ci8 =l shl %carry_idx, 3
    %caddr =l add r, %ci8
    storel %carry, %caddr

    %i =l add %i, %one
    jmp @outer
@outer_done:

    %res =l const 0
    %i =l const 0
@trim:
    %c_i_len =w csltl %i, %len
    jnz %c_i_len, @trim_body
    jmp @trim_done
@trim_body:
    %i8 =l shl %i, 3
    %taddr =l add r, %i8
    %tval =l loadl %taddr
    %is_zero =w ceql %tval, %zero
    jnz %is_zero, @trim_inc
    %res =l add %i, %one
@trim_inc:
    %i =l add %i, %one
    jmp @trim
@trim_done:
    ret %res
}
