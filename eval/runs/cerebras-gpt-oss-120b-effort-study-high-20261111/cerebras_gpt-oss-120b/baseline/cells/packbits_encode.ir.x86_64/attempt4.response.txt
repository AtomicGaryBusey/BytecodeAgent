export function l $packbits_encode(l %src, l %n, l %out, l %cap) {

@run_start(%i, %out_idx):

    jmp @run_loop(%i, %out_idx, %src_i, %run)

    %next_byte =b loadub %next_ptr
    %eq =w ceqb %src_i, %next_byte

    %cond_cont =w andw %cond_j_lt_n, %cond_len

@lit_check_run(%i, %out_idx, %j):

    %cond_jp2_lt_n =w csltl %j_plus2, %n

    %b0 =b loadub %src_j_ptr
    %src_j1_ptr =l addl %src_j_ptr, 1l
    %b1 =b loadub %src_j1_ptr
    %src_j2_ptr =l addl %src_j_ptr, 2l
    %b2 =b loadub %src_j2_ptr

    %new_end =l addl %out_idx, %need

@lit_write(%i, %out_idx, %len):

    %header =b cvtb %len_minus_one
    %out_base =l addl %out, %out_idx
    storeb %header, %out_base
    %k =l 0l
    jmp @copy_loop(%i, %out_idx, %len, %k)

    %src_addr =l addl %src_base, %k
    %b =b loadub %src_addr
    %out_base1 =l addl %out, %out_idx
    %out_base2 =l addl %out_base1, 1l
    %out_addr =l addl %out_base2, %k

    jmp @copy_loop(%i, %out_idx, %len, %k_next)

    %out_idx_next =l addl %out_idx, %len_plus_one
    %i_next =l addl %i, %len
    jmp @loop(%i_next, %out_idx_next)

@ret(%out_idx):
    ret %out_idx

@fail:

    ret %neg1
}
