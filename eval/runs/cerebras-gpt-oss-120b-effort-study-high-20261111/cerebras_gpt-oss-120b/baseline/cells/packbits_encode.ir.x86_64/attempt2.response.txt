export function l $packbits_encode(l %src, l %n, l %out, l %cap) {

@loop(%i, %out_idx):

    br %cond, @run_start(%i, %out_idx), @ret(%out_idx)

@run_start(%i, %out_idx):

    %src_i =b loadub %src_i_ptr
    %run0 =l 1
    jmp @run_loop(%i, %out_idx, %src_i, %run0)

@run_loop(%i, %out_idx, %src_i, %run):

    %cond_len =w csltl %run, 128
    %cond_idx =w csltl %next, %n

    %src_next_ptr =l addl %src, %next
    %src_next =b loadub %src_next_ptr
    %eq =w ceqb %src_i, %src_next
    %run_next =l addl %run, 1
    br %eq, @run_loop(%i, %out_idx, %src_i, %run_next), @run_done(%i, %out_idx, %src_i, %run)

    %overflow =w csltl %cap, %out_end

@repeat_write(%i, %out_idx, %src_i, %run):

    %header =b subb 1b, %run_b
    %out_ptr =l addl %out, %out_idx
    storeb %header, %out_ptr
    %out_ptr1 =l addl %out_ptr, 1

    %out_idx2 =l addl %out_idx, 2
    jmp @loop(%i2, %out_idx2)

@lit_scan(%i, %out_idx):

    jmp @lit_loop(%i, %out_idx, %j0)

@lit_loop(%i, %out_idx, %j):
    %cond1 =w csltl %j, %n
    %len_cur =l subl %j, %i
    %cond2 =w csltl %len_cur, 128
    %cond =w andw %cond1, %cond2
    br %cond, @lit_test_rem(%i, %out_idx, %j), @lit_done(%i, %out_idx, %j)

@lit_test_rem(%i, %out_idx, %j):
    %rem =l subl %n, %j
    %cond_rem =w csltl %rem, 3
    br %cond_rem, @lit_continue(%i, %out_idx, %j), @lit_check_run(%i, %out_idx, %j)

@lit_continue(%i, %out_idx, %j):

    jmp @lit_loop(%i, %out_idx, %j_next)

@lit_check_run(%i, %out_idx, %j):

    %b0 =b loadub %src_j_ptr
    %src_j1_ptr =l addl %src_j_ptr, 1
    %b1 =b loadub %src_j1_ptr
    %src_j2_ptr =l addl %src_j_ptr, 2
    %b2 =b loadub %src_j2_ptr

    %eq02 =w ceqb %b0, %b2

@lit_done(%i, %out_idx, %j):
    %len =l subl %j, %i
    %need =l addl %len, 1
    %out_end =l addl %out_idx, %need
    %overflow =w csltl %cap, %out_end
    br %overflow, @fail, @lit_write(%i, %out_idx, %len)

@lit_write(%i, %out_idx, %len):
    %len_minus_one =l subl %len, 1
    %header_b =b cvtb %len_minus_one
    %out_base =l addl %out, %out_idx
    storeb %header_b, %out_base
    %k0 =l 0
    jmp @copy_loop(%i, %out_idx, %len, %k0)

@copy_loop(%i, %out_idx, %len, %k):

@copy_body(%i, %out_idx, %len, %k):

    %src_addr =l addl %src_off, %k
    %b =b loadub %src_addr
    %out_base =l addl %out, %out_idx
    %out_off =l addl %out_base, 1
    %out_addr =l addl %out_off, %k
    storeb %b, %out_addr
    %k_next =l addl %k, 1
    jmp @copy_loop(%i, %out_idx, %len, %k_next)

@copy_done(%i, %out_idx, %len):
    %need =l addl %len, 1
    %out_idx2 =l addl %out_idx, %need
    %i2 =l addl %i, %len
    jmp @loop(%i2, %out_idx2)

@ret(%out_idx):
    ret %out_idx

@fail:
    %neg1 =l -1
    ret %neg1
}
