export function l $fill_down(l a, l n, l start)
{
@entry:
    %a = arg a
    %n = arg n
    %start = arg start
    %zero = const 0
    jmp @header

@header:
    %i = phi %zero, @entry, %i_next, @loop
    %cond = csltl %i, %n
    jnz %cond, @loop, @exit

@loop:
    %val = subl %start, %i
    %off = mull %i, 8
    %addr = addl %a, %off
    storel %val, %addr
    %i_next = addl %i, 1
    jmp @header

@exit:
    ret %n
}
