export function l $fill_down(l a, l n, l start) {
@entry:
    %zero = const 0
    %one = const 1
    %eight = const 8
    jmp @header

@header:
    %i = phi %zero, @entry, %i_next, @loop
    %cond = csltl %i, %n
    jnz %cond, @loop, @exit

@loop:
    %val = subl %start, %i
    %off = mull %i, %eight
    %addr = addl %a, %off
    storel %val, %addr
    %i_next = addl %i, %one
    jmp @header

@exit:
    ret %n
}
