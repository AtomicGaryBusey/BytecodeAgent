packbits_decode:
    cld
    lea r8, [rdi + rsi]           # src_end = src + n
    lea r9, [rdx + rcx]           # out_end = out + cap
    xor r10, r10                  # total_written = 0
.Ltop:
    cmp rdi, r8
    jae .Ldone                    # no more source bytes
    movsx r11, byte ptr [rdi]     # header signed
    inc rdi
    test r11b, r11b
    jge .Lliteral                 # h >= 0  -> literal
    cmp r11b, -128
    je .Lnoop                     # h == -128 -> no-op
    # repeat packet
    cmp rdi, r8
    jae .Lerror_src               # need repeat byte
    mov al, byte ptr [rdi]        # repeat value
    inc rdi
    neg r11d
    inc r11d                      # repeat count = 1 - h
    push rax                      # save repeat byte (in al)
    mov rax, r9
    sub rax, rdx                  # out_remaining = out_end - out_ptr
    cmp rax, r11
    jae .Lrepeat_full
    test rax, rax
    je .Loverflow_ret             # no space left
    mov rcx, rax                  # rcx = out_remaining
    pop rax                       # restore repeat byte into al
    mov rdi, rdx
    rep stosb                     # write partial repeat
    mov rdx, rdi
    jmp .Loverflow_ret
.Lrepeat_full:
    pop rax                       # restore repeat byte into al
    mov rcx, r11                  # rcx = repeat count
    mov rdi, rdx
    rep stosb                     # write full repeat
    mov rdx, rdi
    add r10, r11
    jmp .Ltop
.Lliteral:
    add r11d, 1                   # count = h + 1
    # check source remaining
    mov rax, r8
    sub rax, rdi
    cmp rax, r11
    jb .Lerror_src
    # check out remaining
    mov rax, r9
    sub rax, rdx
    cmp rax, r11
    jae .Lcopy_full
    test rax, rax
    je .Loverflow_ret
    mov rcx, rax                  # rcx = out_remaining
    mov rsi, rdi
    mov rdi, rdx
    rep movsb                     # copy partial literal
    mov rdx, rdi
    jmp .Loverflow_ret
.Lcopy_full:
    mov rcx, r11                  # rcx = count
    mov rsi, rdi
    mov rdi, rdx
    rep movsb                     # copy full literal
    mov rdx, rdi
    mov rdi, rsi
    add r10, r11
    jmp .Ltop
.Lnoop:
    jmp .Ltop
.Ldone:
    mov rax, r10
    ret
.Lerror_src:
    mov rax, -2
    ret
.Loverflow_ret:
    mov rax, -1
    ret
