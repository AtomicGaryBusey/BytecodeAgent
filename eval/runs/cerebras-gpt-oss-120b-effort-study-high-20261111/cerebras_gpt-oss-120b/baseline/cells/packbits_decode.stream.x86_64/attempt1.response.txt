packbits_decode:
    cld
    lea r8, [rdi + rsi]         # src_end = src + n
    lea r9, [rdx + rcx]         # out_end = out + cap
    xor r10, r10                # bytes_written = 0
.Lloop:
    cmp rdi, r8
    jae .Ldone
    movsx r11, byte ptr [rdi]
    inc rdi
    test r11, r11
    jge .Lliteral
    cmp r11, -128
    je .Lnoop
.Lrepeat:
    cmp rdi, r8
    jae .Lerror_src
    mov al, byte ptr [rdi]
    inc rdi
    neg r11
    inc r11                      # repeat count = 1 - h
    mov rax, r9
    sub rax, rdx                # out_remaining
    cmp rax, r11
    jae .Lwrite_repeat_full
    cmp rax, 0
    je .Lreturn_overflow
    mov rcx, rax
    mov rdi, rdx
    rep stosb
    mov rdx, rdi
    jmp .Lreturn_overflow
.Lwrite_repeat_full:
    mov rcx, r11
    mov rdi, rdx
    rep stosb
    mov rdx, rdi
    add r10, r11
    jmp .Lloop
.Lliteral:
    lea r11, [r11 + 1]          # count = h + 1
    mov rax, r8
    sub rax, rdi                # src_remaining
    cmp rax, r11
    jb .Lerror_src
    mov rax, r9
    sub rax, rdx                # out_remaining
    cmp rax, r11
    jae .Lcopy_literal_full
    cmp rax, 0
    je .Lreturn_overflow
    mov rcx, rax
    mov rsi, rdi
    mov rdi, rdx
    rep movsb
    mov rdx, rdi
    mov rdi, rsi
    jmp .Lreturn_overflow
.Lcopy_literal_full:
    mov rcx, r11
    mov rsi, rdi
    mov rdi, rdx
    rep movsb
    mov rdx, rdi
    mov rdi, rsi
    add r10, r11
    jmp .Lloop
.Lnoop:
    jmp .Lloop
.Ldone:
    mov rax, r10
    ret
.Lerror_src:
    mov rax, -2
    ret
.Lreturn_overflow:
    mov rax, -1
    ret
