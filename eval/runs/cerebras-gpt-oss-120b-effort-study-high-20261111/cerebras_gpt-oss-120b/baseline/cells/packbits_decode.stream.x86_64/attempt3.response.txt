packbits_decode:
    cld
    lea r8, [rdi + rsi]          # src_end = src + n
    lea r9, [rdx + rcx]          # out_end = out + cap
    mov r10, rdx                 # out_base = out
    mov rsi, rdi                 # src_ptr = src
    mov rdi, rdx                 # out_ptr = out
.Lloop:
    cmp rsi, r8
    jae .Ldone
    movsx rax, byte ptr [rsi]
    inc rsi
    cmp al, 0
    jge .Lliteral
    cmp al, -128
    je .Lnoop
    # repeat packet
    mov rcx, 1
    sub rcx, rax                 # rcx = 1 - header (repeat count)
    cmp rsi, r8
    jae .Lerror_src
    mov al, byte ptr [rsi]
    inc rsi
    mov rdx, r9
    sub rdx, rdi                 # out_remaining = out_end - out_ptr
    cmp rdx, rcx
    jae .Lrepeat_full
    test rdx, rdx
    je .Loverflow_ret
    mov rcx, rdx
    rep stosb
    jmp .Loverflow_ret
.Lrepeat_full:
    rep stosb
    jmp .Lloop
.Lliteral:
    movzx rcx, al
    inc rcx                      # count = header + 1
    mov rdx, r8
    sub rdx, rsi                 # src_remaining = src_end - src_ptr
    cmp rdx, rcx
    jb .Lerror_src
    mov rdx, r9
    sub rdx, rdi                 # out_remaining = out_end - out_ptr
    cmp rdx, rcx
    jae .Lcopy_full
    test rdx, rdx
    je .Loverflow_ret
    mov rcx, rdx
    rep movsb
    jmp .Loverflow_ret
.Lcopy_full:
    rep movsb
    jmp .Lloop
.Lnoop:
    jmp .Lloop
.Ldone:
    mov rax, rdi
    sub rax, r10
    ret
.Lerror_src:
    mov rax, -2
    ret
.Loverflow_ret:
    mov rax, -1
    ret
