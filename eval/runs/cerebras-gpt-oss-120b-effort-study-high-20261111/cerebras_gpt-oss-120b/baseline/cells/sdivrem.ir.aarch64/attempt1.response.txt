export function l $sdivrem(l a, l b, l rem) {
@entry:
    %b_is_zero = ceql %b, 0
    jnz %b_is_zero, @div_by_zero

    %a_is_min = ceql %a, -9223372036854775808
    %b_is_neg1 = ceql %b, -1
    %overflow = andw %a_is_min, %b_is_neg1
    jnz %overflow, @overflow_case

    %q = divl %a, %b
    %r = modl %a, %b
    storel %r, %rem
    ret %q

@div_by_zero:
    storel %a, %rem
    ret 0

@overflow_case:
    storel 0, %rem
    ret -9223372036854775808
}
