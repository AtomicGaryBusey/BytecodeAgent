export function l $sdivrem(l, l, l) {
@entry:
    %b_is_zero = ceql %1, 0
    jnz %b_is_zero, @div_by_zero

    %one = addl 0, 1
    %c63 = addw 0, 63
    %min = shll %one, %c63
    %neg_one = subl 0, %one
    %a_is_min = ceql %0, %min
    %b_is_neg1 = ceql %1, %neg_one
    %overflow = andw %a_is_min, %b_is_neg1
    jnz %overflow, @overflow_case

    %q = divl %0, %1
    %r = modl %0, %1
    storel %r, %2
    ret %q

@div_by_zero:
    storel %0, %2
    ret 0

@overflow_case:
    storel 0, %2
    ret %0
}
