export function l $sdivrem(l %a, l %b, l* %rem) {
@entry:
	%b_eq0 = ceql %b, 0
	jnz %b_eq0, @b_is_zero

	%a_eq_min = ceql %a, -9223372036854775808
	%b_eq_minus1 = ceql %b, -1
	%overflow = andw %a_eq_min, %b_eq_minus1
	jnz %overflow, @overflow

	%q = divl %a, %b
	%r = reml %a, %b
	storel %r, %rem

@b_is_zero:
