export function l $count_desc(l, l) {
@start
    %zero = copyl 0
    %two = copyl 2
    %lt2 = csltl %1, %two
    jnz %lt2, @Lzero
    jmp @Linit

@Lzero
    ret %zero

@Linit
    %cnt = copyl 0
    %i = copyl 0
    %p = copyl %0
    %n1 = subl %1, 1
    jmp @Lcheck

@Lcheck
    %c = csltl %i, %n1
    jnz %c, @Lbody
    jmp @Lend

@Lbody
    %cur = loadl %p
    %p_next = addl %p, 8
    %nxt = loadl %p_next
    %desc = csltl %nxt, %cur
    %desc_l = zextw %desc
    %cnt = addl %cnt, %desc_l
    %p = addl %p, 8
    %i = addl %i, 1
    jmp @Lcheck

@Lend
    ret %cnt
}
