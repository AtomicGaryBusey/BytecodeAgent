export function l $count_desc(l a, l n) {
@start
    %lt2 = csltl %n, 2
    jnz %lt2, @Lzero
    jmp @Linit

@Lzero
    ret 0

@Linit
    %cnt = 0
    %i = 0
    %n1 = subl %n, 1
    jmp @Lcheck

@Lcheck
    %c = csltl %i, %n1
    jnz %c, @Lbody
    jmp @Lend

@Lbody
    %i8 = shll %i, 3
    %addr = addl %a, %i8
    %cur = loadl %addr
    %addr_next = addl %addr, 8
    %nxt = loadl %addr_next
    %desc = csltl %nxt, %cur
    %desc_l = zextw %desc
    %cnt = addl %cnt, %desc_l
    %i = addl %i, 1
    jmp @Lcheck

@Lend
    ret %cnt
}
