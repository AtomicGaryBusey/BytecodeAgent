packbits_encode:
    mov x4, x0
    mov x5, x1
    mov x6, x2
    mov x7, x3
    mov x8, #0
.Louter:
    cbz x5, .Ldone
    ldrb w10, [x4]
    mov x9, #1
.Lrun_len_loop:
    cmp x9, #128
    b.ge .Lrun_len_done
    cmp x9, x5
    b.ge .Lrun_len_done
    add x11, x4, x9
    ldrb w12, [x11]
    cmp w12, w10
    b.ne .Lrun_len_done
    add x9, x9, #1
    b .Lrun_len_loop
.Lrun_len_done:
    cmp x9, #3
    b.ge .Lrepeat
    // literal packet
    mov x9, #0
.Llit_scan:
    cmp x9, #128
    b.ge .Llit_done
    subs x12, x5, x9
    b.le .Llit_done
    cmp x12, #3

    add x13, x4, x9
    ldrb w14, [x13]
    ldrb w15, [x13, #1]
    cmp w14, w15
    b.ne .Llit_include
    ldrb w0, [x13, #2]
    cmp w14, w0
    b.ne .Llit_include

.Llit_include:
    add x9, x9, #1
    b .Llit_scan
.Llit_done:
    add x14, x9, #1
    cmp x7, x14
    b.lt .Lcap_error
    sub w10, w9, #1
    strb w10, [x6]
    // copy literal bytes
    mov x15, #0
.Lcopy_lit:

    b.ge .Lcopy_done
    add x16, x4, x15
    ldrb w18, [x16]
    add x17, x6, #1
    add x17, x17, x15
    strb w18, [x17]
    add x15, x15, #1
    b .Lcopy_lit

    sub x7, x7, x14
    add x8, x8, x14
    add x4, x4, x9
    sub x5, x5, x9
    b .Louter
.Lrepeat:
    cmp x7, #2
    b.lt .Lcap_error
    mov w10, #257
    sub w10, w10, w9
    ldrb w11, [x4]
    strb w10, [x6]
    strb w11, [x6, #1]
    add x6, x6, #2
    sub x7, x7, #2
    add x8, x8, #2
    add x4, x4, x9
    sub x5, x5, x9
    b .Louter
.Lcap_error:
    mov x0, #-1
    ret
.Ldone:
    mov x0, x8
    ret
