    cmp x0, x1
    beq .Lreturn
    b.lo .Lforward
.Lbackward:

    add x1, x1, x5
    add x0, x0, x5
    mov x3, x2
.Lbwd_loop:

    sub x0, x0, #8
    ldr x4, [x1]
    str x4, [x0]
    subs x3, x3, #1
    bne .Lbwd_loop
    b .Lreturn
.Lforward:
    mov x3, x2
.Lfwd_loop:

    str x4, [x0], #8
    subs x3, x3, #1
    bne .Lfwd_loop
