export function l $madd32(l a, l b, l c)
@start
    %mask = constl 0xffffffff
    %a32 = andl a, %mask
    %b32 = andl b, %mask
    %c32 = andl c, %mask
    %prod = mull %a32, %b32
    %prod32 = andl %prod, %mask
    %sum = addl %prod32, %c32
    %sum32 = andl %sum, %mask
    %w = truncw %sum32
    %res = extsw %w
    ret %res
