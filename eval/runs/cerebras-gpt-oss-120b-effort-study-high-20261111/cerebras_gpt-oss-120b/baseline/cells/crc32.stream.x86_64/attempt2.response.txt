crc32:
    cld
    mov eax, -1
    test rsi, rsi
    jz .Lend
.Louter:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit:
    test al, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lnext
.Lshift:
    shr eax, 1
.Lnext:
    dec ecx
    jnz .Lbit
    inc rdi
    dec rsi
    test rsi, rsi
    jnz .Louter
.Lend:
    not eax
    ret
