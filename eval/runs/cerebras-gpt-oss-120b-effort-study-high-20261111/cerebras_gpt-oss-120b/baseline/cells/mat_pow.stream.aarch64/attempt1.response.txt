mat_pow:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    // x0=a, x1=n, x2=k, x3=out, x4=tmp
    mul x5, x1, x1          // nn = n*n
    lsl x6, x1, #3          // n_bytes = n*8
    lsl x7, x5, #3          // nn_bytes = nn*8
    add x8, x4, x7          // product buffer = tmp + nn_bytes

    mov x19, x3             // out pointer
    mov x20, x4             // base pointer (tmp)
    mov x21, x8             // product buffer pointer
    mov x22, x1             // n
    mov x23, x6             // n_bytes

    // copy a -> base
    mov x0, x0
    mov x1, x20
    mov x2, x5
    bl .Lcopy

    // zero out out
    mov x0, x19
    mov x1, x5
    bl .Lzero

    // set identity in out
    add x13, x22, #1        // n+1
    mov x12, #0             // i = 0
    mov x14, #1             // one
.Ldiag_loop:
    cmp x12, x22
    bge .Ldiag_done
    mul x15, x12, x13       // idx = i*(n+1)
    lsl x15, x15, #3        // offset bytes
    add x16, x19, x15
    str x14, [x16]
    add x12, x12, #1
    b .Ldiag_loop
.Ldiag_done:

    // if k == 0 skip exponentiation
    cbz x2, .Lskip_pow

.Lpow_loop:
    cbz x2, .Lpow_done

    and x9, x2, #1          // bit = k & 1
    lsr x10, x2, #1         // new_k = k >> 1

    // if bit set: out = out * base
    cbz x9, .Lskip_res_mul
    mov x0, x19
    mov x1, x20
    mov x2, x22
    mov x3, x21
    bl .Lmat_mul
    // copy product -> out
    mul x5, x22, x22        // nn
    mov x0, x21
    mov x1, x19
    mov x2, x5
    bl .Lcopy
.Lskip_res_mul:

    // base = base * base
    mov x0, x20
    mov x1, x20
    mov x2, x22
    mov x3, x21
    bl .Lmat_mul
    // copy product -> base
    mul x5, x22, x22
    mov x0, x21
    mov x1, x20
    mov x2, x5
    bl .Lcopy

    // next exponent
    mov x2, x10
    b .Lpow_loop
.Lpow_done:

.Lskip_pow:
    // zero base buffer
    mul x5, x22, x22
    mov x0, x20
    mov x1, x5
    bl .Lzero
    // zero product buffer
    mov x0, x21
    mov x1, x5
    bl .Lzero

    mov x0, #0
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

.Lcopy:
    cbz x2, .Lcopy_done
.Lcopy_loop:
    ldr x3, [x0], #8
    str x3, [x1], #8
    subs x2, x2, #1
    bne .Lcopy_loop
.Lcopy_done:
    ret

.Lzero:
    mov x2, #0
    cbz x1, .Lzero_done
.Lzero_loop:
    str x2, [x0], #8
    subs x1, x1, #1
    bne .Lzero_loop
.Lzero_done:
    ret

.Lmat_mul:
    // x0=A, x1=B, x2=n, x3=C
    lsl x4, x2, #3          // n_bytes
    mov x5, #0              // i = 0
.Lmat_mul_i_loop:
    cmp x5, x2
    bge .Lmat_mul_done

    mul x6, x5, x4          // row_offset = i*n_bytes
    add x7, x0, x6          // A_row
    add x8, x3, x6          // C_row

    mov x9, #0              // j = 0
.Lmat_mul_j_loop:
    cmp x9, x2
    bge .Lmat_mul_i_next

    mov x10, #0             // sum = 0
    lsl x11, x9, #3         // col_offset = j*8
    add x12, x1, x11        // B_col pointer

    mov x13, x7             // a_ptr
    mov x14, x12            // b_ptr
    mov x15, #0             // t = 0
.Lmat_mul_t_loop:
    cmp x15, x2
    bge .Lmat_mul_t_done

    ldr x16, [x13], #8
    ldr x17, [x14]
    mul x16, x16, x17
    add x10, x10, x16

    add x14, x14, x4        // b_ptr += n_bytes
    add x15, x15, #1
    b .Lmat_mul_t_loop
.Lmat_mul_t_done:
    add x16, x8, x11
    str x10, [x16]

    add x9, x9, #1
    b .Lmat_mul_j_loop
.Lmat_mul_i_next:
    add x5, x5, #1
    b .Lmat_mul_i_loop
.Lmat_mul_done:
    ret
