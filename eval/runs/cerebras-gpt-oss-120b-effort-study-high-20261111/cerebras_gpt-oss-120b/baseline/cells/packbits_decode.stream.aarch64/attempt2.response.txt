packbits_decode:
    // set up pointers and counters
    add x1, x0, x1               // x1 = src_end
    add x3, x2, x3               // x3 = out_end
    mov x4, #0                   // x4 = written count

.Lloop:
    cmp x0, x1
    b.hs .Ldone                  // if src >= src_end, done

    ldrb w5, [x0], #1            // x5 = header byte (post-increment src)
    sxtb x5, w5                  // sign-extend to 64-bit

    cmp x5, #0
    b.ge .Lliteral               // header >= 0 -> literal

    // negative header
    cmp x5, #-128
    beq .Lnoop                   // -128 -> no-op

    // repeat packet
    cmp x0, x1
    b.hs .Lsrc_err               // need one more source byte for repeat value

    ldrb w6, [x0], #1            // repeat value
    mov x7, #1
    sub x7, x7, x5               // x7 = count = 1 - header

    // check output capacity
    sub x8, x3, x2               // x8 = remaining out space
    cmp x8, x7
    b.lo .Lrepeat_overflow       // not enough space

    // copy repeat bytes
    mov x9, x7
.Lrepeat_copy:
    strb w6, [x2], #1
    add x4, x4, #1
    subs x9, x9, #1
    b.ne .Lrepeat_copy
    b .Lloop

.Lrepeat_overflow:
    // copy as many as possible (x8 bytes)
    cbz x8, .Lreturn_overflow
    mov x9, x8
.Lrepeat_copy_overflow:
    strb w6, [x2], #1
    subs x9, x9, #1
    b.ne .Lrepeat_copy_overflow
    b .Lreturn_overflow

.Lliteral:
    // literal packet: count = header + 1
    add x7, x5, #1               // x7 = count

    // ensure enough source bytes
    sub x8, x1, x0
    cmp x8, x7
    b.lo .Lsrc_err               // not enough source

    // check output capacity
    sub x8, x3, x2
    cmp x8, x7
    b.lo .Lliteral_overflow      // not enough space

    // copy literal bytes
    mov x9, x7
.Lcopy:
    ldrb w6, [x0], #1
    strb w6, [x2], #1
    add x4, x4, #1
    subs x9, x9, #1
    b.ne .Lcopy
    b .Lloop

.Lliteral_overflow:
    // copy as many as possible (x8 bytes)
    cbz x8, .Lreturn_overflow
    mov x9, x8
.Lcopy_overflow:
    ldrb w6, [x0], #1
    strb w6, [x2], #1
    subs x9, x9, #1
    b.ne .Lcopy_overflow
    b .Lreturn_overflow

.Lnoop:
    b .Lloop

.Lsrc_err:
    mov x0, #-2
    ret

.Lreturn_overflow:
    mov x0, #-1
    ret

.Ldone:
    mov x0, x4
    ret
