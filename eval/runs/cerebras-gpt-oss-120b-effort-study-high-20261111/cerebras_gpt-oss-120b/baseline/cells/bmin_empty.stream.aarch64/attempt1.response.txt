bmin_empty:
    cbz x1, .Lreturn_max
    ldr x2, [x0]          // load a[0] as initial min
    add x3, x0, #8        // pointer to a[1]
    sub x4, x1, #1        // count = n-1
.Lloop:
    cbz x4, .Ldone
    ldr x5, [x3], #8
    cmp x5, x2
    csel x2, x5, x2, lt
    sub x4, x4, #1
    b .Lloop
.Ldone:
    mov x0, x2
    ret
.Lreturn_max:
    mov x0, #-1
    bfc x0, #63, #1
    ret
