export function l count_desc(l %a, l %n) {
@entry:
    %cnt = constl 0
    %cond_small = csltl %n, constl 2
    jnz %cond_small @ret_zero

    %limit = subl %n, constl 1
    %i = constl 0
    jmp @loop

@loop:
    %cond = csltl %i, %limit
    jz %cond @out

    %i8 = mull %i, constl 8
    %ptr_i = addl %a, %i8
    %val_i = loadl %ptr_i
    %ptr_i1 = addl %ptr_i, constl 8
    %val_i1 = loadl %ptr_i1
    %desc = csltl %val_i1, %val_i
    %inc = zextw %desc
    %cnt = addl %cnt, %inc

    %i = addl %i, constl 1
    jmp @loop

@out:
    ret %cnt

@ret_zero:
    ret constl 0
}
