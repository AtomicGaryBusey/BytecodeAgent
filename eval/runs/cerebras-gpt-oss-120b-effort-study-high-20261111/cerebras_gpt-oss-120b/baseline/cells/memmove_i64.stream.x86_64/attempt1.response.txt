memmove_i64:
    test rdx, rdx
    je .Ldone
    cmp rdi, rsi
    jae .Lbackward
.Lforward:
    mov rax, [rsi]
    mov [rdi], rax
    add rsi, 8
    add rdi, 8
    dec rdx
    jne .Lforward
    jmp .Ldone
.Lbackward:
    dec rdx
    mov rax, [rsi + rdx*8]
    mov [rdi + rdx*8], rax
    jne .Lbackward
    jmp .Ldone
.Ldone:
    xor eax, eax
    ret
