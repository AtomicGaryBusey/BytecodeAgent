export function l $max_byte(l %p, l %n) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    jmp @loop

@loop:
    %i =l phi @entry %i0, @loop_body %i_next
    %max =l phi @entry %max0, @loop_body %max_next
    %c =w csltl %i, %n
    jnz %c, @loop_body
    jmp @exit

@loop_body:
    %addr =l add %p, %i
    %byte =l loadub %addr
    %c2 =w csltl %max, %byte
    %max_next =l sel %c2, %byte, %max
    %i_next =l add %i, const 1
    jmp @loop

@exit:
    ret %max
}
