export function l $max_byte(l %p, l %n) {
@entry:
    %max0 =l const 0
    %i0 =l const 0
    jmp @loop
@loop:
    %max =l phi @entry %max0, @inc %max1
    %i =l phi @entry %i0, @inc %i1
    %c =w csltl %i, %n
    jnz %c, @body
    jmp @exit
@body:
    %addr =l add %p, %i
    %byte =w loadub %addr
    %byte64 =l zext %byte
    %c2 =w csltl %max, %byte64
    jnz %c2, @update
    jmp @inc
@update:
    %max_new =l mov %byte64
    jmp @inc
@inc:
    %max1 =l phi @body %max, @update %max_new
    %i1 =l add %i, const 1
    jmp @loop
@exit:
    ret %max
}
