export function l $bmax(l a, l n) {
@entry:
    %max0 = loadl %a
    jmp @cond

@cond:
    %max = phi l %max0, @entry, %max1, @body
    %i = phi l 1, @entry, %i1, @body
    %c = csltl %i, %n
    jnz %c, @body
    jmp @end

@body:
    %off = mull %i, 8
    %addr = addl %a, %off
    %cur = loadl %addr
    %cond = csltl %max, %cur
    %max1 = selc %cond, %cur, %max
    %i1 = addl %i, 1
    jmp @cond

@end:
    ret %max
}
