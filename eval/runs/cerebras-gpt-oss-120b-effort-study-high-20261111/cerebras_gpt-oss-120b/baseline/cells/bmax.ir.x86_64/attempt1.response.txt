export function l $bmax(l a, l n) {
@entry:
    %m0 = loadl a
    jmp @test, %m0, 1

@test:
    %max = phi l %m0, @entry, %max2, @body
    %i = phi l 1, @entry, %i2, @body
    %c = csltl %i, n
    jnz %c, @body, @end

@body:
    %off = mull %i, 8
    %addr = addl a, %off
    %cur = loadl %addr
    %cond = csltl %max, %cur
    %max2 = sell %cond, %cur, %max
    %i2 = addl %i, 1
    jmp @test, %max2, %i2

@end:
    ret %max
}
