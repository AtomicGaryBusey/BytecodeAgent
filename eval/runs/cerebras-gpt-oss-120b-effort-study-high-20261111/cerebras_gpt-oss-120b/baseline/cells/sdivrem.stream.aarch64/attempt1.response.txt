sdivrem:
    cbz x1, .Ldiv0
    adds xzr, x1, #1
    bne .Ldo_div
    movz x9, #0x8000, lsl #48
    cmp x0, x9
    bne .Ldo_div
    mov x3, #0
    str x3, [x2]
    ret
.Ldo_div:
    sdiv x3, x0, x1
    mul x4, x3, x1
    sub x5, x0, x4
    str x5, [x2]
    mov x0, x3
    ret
.Ldiv0:

    mov x0, #0
    ret
