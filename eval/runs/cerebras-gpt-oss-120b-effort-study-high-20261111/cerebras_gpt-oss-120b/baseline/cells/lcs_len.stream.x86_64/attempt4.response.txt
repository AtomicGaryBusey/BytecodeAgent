lcs_len:
    mov r9, rcx
    lea r9, [r9+1]
    xor rax, rax
    mov r10, r8
.Lzero_loop:
    mov [r10], rax
    add r10, 8
    dec r9
    jne .Lzero_loop
    test rsi, rsi
    jz .Lreturn
.Louter_loop:
    movzx eax, byte ptr [rdi]
    inc rdi
    xor r10d, r10d
    xor r11d, r11d
    test rcx, rcx
    jz .Louter_end_iteration
    mov r9, 1
.Linner_loop:
    mov r12, [r8 + r9*8]
    mov bl, byte ptr [rdx + r9 - 1]
    cmp al, bl
    jne .Lno_match
    mov rax, r10
    inc rax
    jmp .Lstore_cur
.Lno_match:
    mov rax, r12
    cmp rax, r11
    cmovl rax, r11
.Lstore_cur:
    mov [r8 + r9*8], rax
    mov r10, r12
    mov r11, rax
    inc r9
    cmp r9, rcx
    jle .Linner_loop
.Louter_end_iteration:
    dec rsi
    jnz .Louter_loop
.Lreturn:
    mov rax, [r8 + rcx*8]
    ret
