lcs_len:
    push r12
    push r13
    push r14
    push r15
    mov r15, rcx
    mov rcx, r15
    inc rcx
    mov r14, r8
    xor rax, rax
.Lzero_loop:
    mov qword ptr [r14], rax
    add r14, 8
    dec rcx
    jne .Lzero_loop
    test rsi, rsi
    jz .Lreturn
.Louter_loop:
    mov r13b, byte ptr [rdi]
    inc rdi
    xor r12, r12
    xor r11, r11
    lea r9, [r8 + 8]
    mov r10, rdx
    mov rcx, r15
.Linner_loop:
    mov rax, qword ptr [r9]
    mov r14, rax
    movzx eax, byte ptr [r10]
    cmp al, r13b
    jne .Lnot_match
    mov rax, r11
    inc rax
    jmp .Lstore_cur
.Lnot_match:
    mov rax, r14
    cmp rax, r12
    cmovl rax, r12
.Lstore_cur:
    mov qword ptr [r9], rax
    mov r11, r14
    mov r12, rax
    add r9, 8
    inc r10
    dec rcx
    jne .Linner_loop
    dec rsi
    jnz .Louter_loop
.Lreturn:
    mov rax, r15
    shl rax, 3
    add rax, r8
    mov rax, qword ptr [rax]
    pop r15
    pop r14
    pop r13
    pop r12
    ret
