lcs_len:
    push r12
    mov r9, rcx
    xor rax, rax
    mov r10, r8
    inc rcx
.Lzero_loop:
    mov qword ptr [r10], rax
    add r10, 8
    dec rcx
    jnz .Lzero_loop
    mov rcx, r9
    test rsi, rsi
    jz .Lafter_outer
.Louter_loop:
    mov al, byte ptr [rdi]
    inc rdi
    xor r10, r10
    xor r11, r11
    mov r9, 1
.Linner_loop:
    cmp al, byte ptr [rdx + r9 - 1]
    je .Lmatch
    mov rax, qword ptr [r8 + r9*8]
    mov r12, rax
    cmp r12, r11
    cmovb r12, r11
    jmp .Lstore
.Lmatch:
    mov rax, qword ptr [r8 + r9*8]
    mov r12, r10
    inc r12
.Lstore:
    mov qword ptr [r8 + r9*8], r12
    mov r10, rax
    mov r11, r12
    inc r9
    cmp r9, rcx
    jle .Linner_loop
    dec rsi
    jnz .Louter_loop
.Lafter_outer:
    mov rax, qword ptr [r8 + rcx*8]
    pop r12
    ret
