lcs_len:
    push r12
    push r13
    push r15
    mov r9, rdx
    mov r15, rcx
    mov rcx, r15
    inc rcx
    xor rax, rax
    mov r13, r8
.Lzero_loop:
    mov [r13], rax
    add r13, 8
    dec rcx
    jnz .Lzero_loop
    test rsi, rsi
    jz .Lafter_outer
.Louter_loop:
    mov al, byte ptr [rdi]
    inc rdi
    xor r10, r10
    xor r11, r11
    mov rcx, r15
    test rcx, rcx
    jz .Lskip_inner
    lea r13, [r8 + 8]
    mov rdx, r9
.Linner_loop:
    mov r12, [r13]
    cmp al, byte ptr [rdx]
    jne .Lnot_match
    mov rax, r10
    inc rax
    jmp .Lstore_cur
.Lnot_match:
    mov rax, r12
    cmp rax, r11
    cmovl rax, r11
.Lstore_cur:
    mov [r13], rax
    mov r10, r12
    mov r11, rax
    add r13, 8
    inc rdx
    dec rcx
    jnz .Linner_loop
.Lskip_inner:
    dec rsi
    jnz .Louter_loop
.Lafter_outer:
    mov rcx, r15
    lea rax, [r8 + rcx*8]
    mov rax, [rax]
    pop r15
    pop r13
    pop r12
    ret
