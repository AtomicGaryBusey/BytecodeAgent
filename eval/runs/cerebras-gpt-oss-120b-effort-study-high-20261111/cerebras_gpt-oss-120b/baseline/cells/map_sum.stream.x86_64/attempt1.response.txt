map_sum:
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor r12, r12
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
    test r14, r14
    jz .Lend
.Lloop:
    mov rdi, qword ptr [r13]
    cld
    call r15
    add r12, rax
    add r13, 8
    dec r14
    jnz .Lloop
.Lend:
    mov rax, r12
    cld
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    ret
