export function l $crc32(l %p, l %n) {
@entry
    %crc =w add -1, 0
    %end =l add %p, %n
    jmp @loop_cond
@loop_cond
    %cond =w cult %p, %end
    jnz %cond, @loop_body
    jmp @final
@loop_body
    %byte =w loadub %p
    %crc =w xor %crc, %byte
    %cnt =w add 8, 0
    jmp @bit_loop
@bit_loop
    %lowbit =w and %crc, 1
    %crc_shr =w shr %crc, 1
    jnz %lowbit, @bit_one
    %crc =w add %crc_shr, 0
    jmp @bit_continue
@bit_one
    %crc =w xor %crc_shr, -306674912
@bit_continue
    %cnt =w sub %cnt, 1
    jnz %cnt, @bit_loop
    %p =l add %p, 1
    jmp @loop_cond
@final
    %crc =w xor %crc, -1
    %result =l extuw %crc
    ret %result
}
