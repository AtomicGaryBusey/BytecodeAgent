export function l $crc32(l %p, l %n) {
@entry
    %end =l add %p, %n
    %crc0 =w -1
    jmp @outer %p, %end, %crc0
@outer
    %cond =w cslt %0, %1
    jnz %cond, @body %0, %1, %2
    jmp @final %2
@body
    %byte =w loadub %0
    %crc1 =w xor %2, %byte
    %cnt0 =w 8
    jmp @bit_loop %crc1, %cnt0, %0, %1
@bit_loop
    %low =w and %0, 1
    %crc_shr =w shr %0, 1
    %mask =w and %low, -306674912
    %crc_next =w xor %crc_shr, %mask
    %cnt_next =w sub %1, 1
    jnz %cnt_next, @bit_loop %crc_next, %cnt_next, %2, %3
    %p_next =l add %2, 1
    jmp @outer %p_next, %3, %crc_next
@final
    %crc_out =w xor %0, -1
    %result =l extuw %crc_out
    ret %result
}
