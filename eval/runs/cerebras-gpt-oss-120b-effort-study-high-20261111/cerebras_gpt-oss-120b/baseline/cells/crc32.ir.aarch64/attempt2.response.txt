export function l $crc32(l %p, l %n) {
@entry
	%crc =w add -1, 0
	%end =l add %p, %n
	jmp @loop
@loop
	%cond =csltl %p, %end
	jnz %cond, @body
	jmp @final
@body
	%byte =w loadub %p
	%crc =w xor %crc, %byte
	%cnt =w add 8, 0
	jmp @bit_loop
@bit_loop
	%bit =w and %crc, 1
	%crc_shr =w shr %crc, 1
	jnz %bit, @bit_one
	%crc =w add %crc_shr, 0
	jmp @bit_next
@bit_one
	%crc =w xor %crc_shr, 3988292384
@bit_next
	%cnt =w sub %cnt, 1
	%zero =csltw %cnt, 1
	jnz %zero, @after_bits
	jmp @bit_loop
@after_bits
	%p =l add %p, 1
	jmp @loop
@final
	%crc =w xor %crc, -1
	%result =l extuw %crc
	ret %result
}
