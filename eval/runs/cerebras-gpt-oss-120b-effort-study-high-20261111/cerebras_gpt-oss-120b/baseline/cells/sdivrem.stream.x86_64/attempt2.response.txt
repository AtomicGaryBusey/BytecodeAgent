sdivrem:
    test rsi, rsi
    jne .Lcheck_overflow
    mov qword ptr [rdx], rdi
    xor rax, rax
    ret
.Lcheck_overflow:
    mov r8, 0x8000000000000000
    cmp rdi, r8
    jne .Ldo_divide
    cmp rsi, -1
    jne .Ldo_divide
    mov qword ptr [rdx], 0
    mov rax, r8
    ret
.Ldo_divide:
    mov rcx, rdx
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
