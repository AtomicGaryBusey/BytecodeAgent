export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
  %zero_w =w const 0
  %one_w =w const 1
  %eight_l =l const 8
  jmp @loop

@loop
  %i =w phi %zero_w, %i_next
  %cnt =w phi %zero_w, %cnt_next
  %i_l =l zext w %i
  %cond =w csltl %i_l, %n
  jnz w %cond, @body, @exit

@body
  %off =l mul %i_l, %eight_l
  %src_addr =l add %src, %off
  %src_val =l loadl %src_addr
  %tmp =l call %f(l %src_val)
  %dst_addr =l add %dst, %off
  storel %tmp, %dst_addr
  %lt =w csltl %tmp, %src_val
  %cnt_next =w add %cnt, %lt
  %i_next =w add %i, %one_w
  jmp @loop

@exit
  %cnt_l =l zext w %cnt
  ret l %cnt_l
}
