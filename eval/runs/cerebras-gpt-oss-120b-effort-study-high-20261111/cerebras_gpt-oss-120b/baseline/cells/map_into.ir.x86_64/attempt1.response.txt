export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %i0 =l const 0
    %cnt0 =l const 0
    jmp @loop

@loop
    %i =l phi %i0, %i_next
    %cnt =l phi %cnt0, %cnt_next
    %cond =w csltl %i, %n
    jnz w %cond, @body, @exit

@body
    %off =l shl %i, 3
    %src_addr =l add %src, %off
    %src_val =l loadl %src_addr
    %tmp =l call %f(l %src_val)
    %dst_addr =l add %dst, %off
    storel %tmp, %dst_addr
    %lt =w csltl %tmp, %src_val
    %lt_ext =l zext w %lt
    %cnt_next =l add %cnt, %lt_ext
    %i_next =l add %i, 1
    jmp @loop

@exit
    ret %cnt
}
