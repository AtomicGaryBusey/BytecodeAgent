export function l binary_search(l a, l n, l key) {
@entry:
    %zero = const l 0
    jmp @loop %zero %n

@loop(%low, %high):
    %cond = csltl %low, %high
    jnz %cond, @loop_body %low %high, @after %low %high

@loop_body(%low, %high):
    %sub = subl %high, %low
    %one = const l 1
    %half = shrl %sub, %one
    %mid = addl %low, %half
    %three = const l 3
    %mid_off = shll %mid, %three
    %addr = addl %a, %mid_off
    %val = loadl %addr
    %eq = ceql %val, %key
    jnz %eq, @return_mid %mid
    %lt = csltl %val, %key
    %midp1 = addl %mid, %one
    %new_low = sel %lt, %midp1, %low
    %new_high = sel %lt, %high, %mid
    jmp @loop %new_low %new_high

@return_mid(%mid):
    ret %mid

@after(%low, %high):
    %neg1 = const l -1
    ret %neg1
}
