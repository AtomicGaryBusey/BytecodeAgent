export function l $binary_search(l a, l n, l key) {
@entry:
    %c0 = const l 0
    %c1 = const l 1
    %cneg1 = const l -1
    %c3 = const l 3
    %low0 = %c0
    %high0 = %n
    jmp @loop %low0 %high0

@loop(%low, %high):
    %cond = csltl %low, %high
    jnz %cond, @loop_body %low %high, @after %low %high

@loop_body(%low, %high):
    %sub = subl %high, %low
    %half = shrl %sub, %c1
    %mid = addl %low, %half
    %mid_off = shll %mid, %c3
    %addr = addl %a, %mid_off
    %val = loadl %addr
    %lt = csltl %val, %key
    %midp1 = addl %mid, %c1
    %new_low = sel %lt, %midp1, %low
    %new_high = sel %lt, %high, %mid
    jmp @loop %new_low %new_high

@after(%low, %high):
    %cond2 = csltl %low, %n
    jnz %cond2, @check %low %high, @ret_minus_one

@check(%low, %high):
    %low_off = shll %low, %c3
    %addr2 = addl %a, %low_off
    %val2 = loadl %addr2
    %eq = ceql %val2, %key
    jnz %eq, @return_low %low %high, @ret_minus_one

@return_low(%low, %high):
    ret %low

@ret_minus_one:
    ret %cneg1
}
