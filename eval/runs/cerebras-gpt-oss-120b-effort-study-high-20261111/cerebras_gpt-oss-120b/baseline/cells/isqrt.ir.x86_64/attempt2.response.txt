export function l $isqrt(l) {
@entry
    %n = arg
    %zero = const l 0
    %hi0 = const l 4294967296
    jmp loop

@loop
    %lo = phi l %zero, entry, %lo1, if_le, %lo2, if_gt
    %hi = phi l %hi0, entry, %hi1, if_le, %hi2, if_gt
    %cond = cultl %lo, %hi
    jnz %cond, body
    jmp exit

@body
    %mid_sum = add l %lo, %hi
    %mid = shr l %mid_sum, const l 1
    %mid_sq = mul l %mid, %mid
    %le = culel %mid_sq, %n
    jnz %le, if_le
    jmp if_gt

@if_le
    %lo1 = add l %mid, const l 1
    %hi1 = add l %hi, const l 0
    jmp loop

@if_gt
    %lo2 = add l %lo, const l 0
    %hi2 = add l %mid, const l 0
    jmp loop

@exit
    %res = sub l %lo, const l 1
    ret l %res
}
