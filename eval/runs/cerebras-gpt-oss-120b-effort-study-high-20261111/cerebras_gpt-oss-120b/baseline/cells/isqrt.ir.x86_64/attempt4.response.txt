export function l $isqrt(l) {
@entry
    %n = arg
    %zero = const l 0
    %one = const l 1
    %two = const l 2
    %hi0 = const l 4294967296
    jmp loop

@loop
    %lo = phi l %zero, entry, %lo1, update_low, %lo2, update_hi
    %hi = phi l %hi0, entry, %hi1, update_low, %hi2, update_hi
    %cond = cultl %lo, %hi
    jnz %cond, body
    jmp exit

@body
    %sum = add l %lo, %hi
    %mid = udivl %sum, %two
    %mid_is_zero = ceql %mid, %zero
    jnz %mid_is_zero, update_low
    %n_div_mid = udivl %n, %mid
    %cond_le = culel %mid, %n_div_mid
    jnz %cond_le, update_low
    jmp update_hi

@update_low
    %lo1 = add l %mid, %one
    %hi1 = %hi
    jmp loop

@update_hi
    %lo2 = %lo
    %hi2 = %mid
    jmp loop

@exit
    %res = sub l %lo, %one
    ret l %res
}
