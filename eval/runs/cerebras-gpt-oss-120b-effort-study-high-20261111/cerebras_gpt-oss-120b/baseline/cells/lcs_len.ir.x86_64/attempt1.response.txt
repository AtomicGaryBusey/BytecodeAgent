export function lcs_len(x l, m l, y l, n l, row l) l
@entry
  %p0 =l 0
  %zero =l 0
  jmp @init
@init
  %p = phi l %p0 %p_next
  %cond_p = sgtl %p %n
  jnz %cond_p @init_done
  %p8 = mull %p 8
  %addr_p = addl %row %p8
  storel %addr_p %zero
  %p_next = addl %p 1
  jmp @init
@init_done
  %i0 =l 0
  jmp @outer
@outer
  %i = phi l %i0 %i_next
  %cond_i = sgtl %m %i
  jnz %cond_i @inner_pre
  jmp @exit
@inner_pre
  %j_start =l 1
  %diag_start =l 0
  jmp @inner
@inner
  %j = phi l %j_start %j_next
  %diag = phi l %diag_start %diag_next
  %cond_j = sgtl %j %n
  jnz %cond_j @inner_done
  %j8 = mull %j 8
  %addr_j = addl %row %j8
  %up =l loadl %addr_j
  %j1 = subl %j 1
  %j1_8 = mull %j1 8
  %addr_left = addl %row %j1_8
  %left =l loadl %addr_left
  %x_addr = addl %x %i
  %xi =w loadub %x_addr
  %y_addr = addl %y %j1
  %yj =w loadub %y_addr
  %eq =w ceqw %xi %yj
  %diag_plus_one =l addl %diag 1
  %gt =w sgtl %up %left
  %max =l sell %gt %up %left
  %new =l sell %eq %diag_plus_one %max
  storel %addr_j %new
  %j_next = addl %j 1
  %diag_next =l %up
  jmp @inner
@inner_done
  %i_next =l addl %i 1
  jmp @outer
@exit
  %n8 = mull %n 8
  %addr_res = addl %row %n8
  %res =l loadl %addr_res
  ret %res
