export lcs_len
function lcs_len(x l, m l, y l, n l, row l) l
lcs_len:
    %zero =l 0
    %one =l 1
    jmp row_init

row_init:
    %p = phi l %zero lcs_len %p_next row_init
    %cond_p = w sgtl %p %n
    jnz %cond_p row_init_done
    %p8 = mull %p 8
    %addr = addl %row %p8
    storel %addr %zero
    %p_next = addl %p %one
    jmp row_init

row_init_done:
    jmp outer

outer:
    %i = phi l %zero row_init_done %i_next outer_loop
    %cond_i = w sgtl %m %i
    jnz %cond_i inner
    jmp exit

inner:
    %j = phi l %one outer %j_next inner
    %diag = phi l %zero outer %diag_next inner
    %cond_j = w sgtl %j %n
    jnz %cond_j inner_done
    %j8 = mull %j 8
    %addr_j = addl %row %j8
    %up = l loadl %addr_j
    %j1 = subl %j %one
    %j1_8 = mull %j1 8
    %addr_left = addl %row %j1_8
    %left = l loadl %addr_left
    %x_addr = addl %x %i
    %xi = w loadub %x_addr
    %y_addr = addl %y %j1
    %yj = w loadub %y_addr
    %eq = w ceqw %xi %yj
    %diag_plus_one = addl %diag %one
    %gt = w sgtl %up %left
    %max = l sel %gt %up %left
    %new = l sel %eq %diag_plus_one %max
    storel %addr_j %new
    %j_next = addl %j %one
    %diag_next = %up
    jmp inner

inner_done:
    %i_next = addl %i %one
    jmp outer_loop

outer_loop:
    jmp outer

exit:
    %n8 = mull %n 8
    %addr_res = addl %row %n8
    %res = l loadl %addr_res
    ret %res
