export function l $stats_f(l a, l n, l f, l out)
{
@entry
    %c0 =l imm 0
    %c1 =l imm 1
    %c8 =l imm 8
    %c16 =l imm 16

    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %cnt0 =l and %y0, %c1
    jmp @loop

@loop
    %i =l phi %c1, %i_next
    %min =l phi %y0, %min_next
    %max =l phi %y0, %max_next
    %sum =l phi %y0, %sum_next
    %cnt =l phi %cnt0, %cnt_next
    %cond =w slt %i, %n
    jnz %cond, @body, @exit

@body
    %off =l mul %i, %c8
    %addr =l add %a, %off
    %aval =l loadl %addr
    %y =l call %f(l %aval)
    %sum_next =l add %sum, %y
    %lt =w slt %y, %min
    %min_next =l sel %lt, %y, %min
    %gt =w sgt %y, %max
    %max_next =l sel %gt, %y, %max
    %odd =l and %y, %c1
    %cnt_next =l add %cnt, %odd
    %i_next =l add %i, %c1
    jmp @loop

@exit
    storel %min, %out
    %out1 =l add %out, %c8
    storel %max, %out1
    %out2 =l add %out, %c16
    storel %sum, %out2
    ret %cnt
}
