export function l $stats_f(l a, l n, l f, l out)
{
@entry
    %c0 =l imm 0
    %c1 =l imm 1
    %c8 =l imm 8
    %c16 =l imm 16
    %cMAX =l imm 9223372036854775807
    %cMIN =l imm -9223372036854775808
    jmp @loop

@loop
    %i =l phi %c0, %i_next
    %min =l phi %cMAX, %min_next
    %max =l phi %cMIN, %max_next
    %sum =l phi %c0, %sum_next
    %cnt =l phi %c0, %cnt_next
    %cond =w slt %i, %n
    jnz %cond, @body, @exit

@body
    %a_off =l mul %i, %c8
    %a_addr =l add %a, %a_off
    %a_val =l loadl %a_addr
    %y =l call %f(l %a_val)

    %sum_next =l add %sum, %y

    %lt =w slt %y, %min
    %min_next =l sel %lt, %y, %min

    %gt =w sgt %y, %max
    %max_next =l sel %gt, %y, %max

    %low =l and %y, %c1
    %odd =w sgt %low, %c0
    %oddL =l sel %odd, %c1, %c0
    %cnt_next =l add %cnt, %oddL

    %i_next =l add %i, %c1

    jmp @loop

@exit
    storel %min, %out
    %out1 =l add %out, %c8
    storel %max, %out1
    %out2 =l add %out, %c16
    storel %sum, %out2
    ret %cnt
}
