export function l $stats_f(l a, l n, l f, l out)
{
@entry
    %aval0 =l loadl %a
    %y0 =l call %f(l %aval0)
    %cnt0 =l and %y0, imm 1
    jmp @loop

@loop
    %i =l phi imm 1, %i_next
    %sum =l phi %y0, %sum_next
    %cnt =l phi %cnt0, %cnt_next
    %min =l phi %y0, %min_next
    %max =l phi %y0, %max_next
    %cond =w csltl %i, %n
    jnz %cond, @body

@exit
    storel %min, %out
    %out1 =l add %out, imm 8
    storel %max, %out1
    %out2 =l add %out, imm 16
    storel %sum, %out2
    ret %cnt

@body
    %off =l shl %i, imm 3
    %addr =l add %a, %off
    %aval =l loadl %addr
    %y =l call %f(l %aval)
    %sum_next =l add %sum, %y
    %odd =l and %y, imm 1
    %cnt_next =l add %cnt, %odd
    %lt =w csltl %y, %min
    %min_next =l sel %lt, %y, %min
    %gt =w csgtl %y, %max
    %max_next =l sel %gt, %y, %max
    %i_next =l add %i, imm 1
    jmp @loop
}
