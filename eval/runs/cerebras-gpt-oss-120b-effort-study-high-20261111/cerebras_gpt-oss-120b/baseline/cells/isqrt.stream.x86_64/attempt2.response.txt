    mov r9, 0x100000000
.Lloop:
    cmp r8, r9
    jae .Ldone
    mov rcx, r8
    add rcx, r9
    shr rcx, 1
    mov rax, rcx
    mul rcx
    cmp rax, rdi
    jbe .Lle
    mov r9, rcx
    jmp .Lloop
.Lle:

    mov r8, rcx
    jmp .Lloop
.Ldone:

    mov rax, r8
    ret
