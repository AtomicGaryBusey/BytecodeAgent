export function l $reverse(l a, l n) {
  @entry:
    %zero = const l 0
    %one = const l 1
    %two = const l 2
    %eight = const l 8
    %half = divl n, %two
    jmp @head

  @head:
    %i = phi (entry %zero, body %i_next)
    %cond = csltl %i, %half
    jnz %cond, @body
    jmp @exit

  @body:
    %nm1 = subl n, %one
    %j = subl %nm1, %i
    %off_i = mull %i, %eight
    %off_j = mull %j, %eight
    %ptr_i = addl a, %off_i
    %ptr_j = addl a, %off_j
    %val_i = loadl %ptr_i
    %val_j = loadl %ptr_j
    storel %val_i, %ptr_j
    storel %val_j, %ptr_i
    %i_next = addl %i, %one
    jmp @head

  @exit:
    ret l %zero
}
