export function l $bubble_sort(l a, l n) {
@entry:
    %n_minus_one = subl %n 1
    jmp @outer_loop

@outer_loop:
    %i = phi l [0, @entry], [%i_next, @outer_inc]
    %outer_cond = csltl %i %n_minus_one
    jnz %outer_cond, @outer_body, @exit

@outer_body:
    %j = phi l [0, @outer_loop], [%j_next, @inner_inc]
    %limit = subl %n_minus_one %i
    %inner_cond = csltl %j %limit
    jnz %inner_cond, @inner_body, @outer_inc

@inner_body:
    %offset_j = mull %j 8
    %addr_j = addl %a %offset_j
    %addr_j1 = addl %addr_j 8
    %val_j = loadl %addr_j
    %val_j1 = loadl %addr_j1
    %swap_cond = csgtl %val_j %val_j1
    jnz %swap_cond, @swap, @inner_inc

@swap:
    storel %val_j1 %addr_j
    storel %val_j %addr_j1
    jmp @inner_inc

@inner_inc:
    %j_next = addl %j 1
    jmp @outer_body

@outer_inc:
    %i_next = addl %i 1
    jmp @outer_loop

@exit:
    retl 0
}
