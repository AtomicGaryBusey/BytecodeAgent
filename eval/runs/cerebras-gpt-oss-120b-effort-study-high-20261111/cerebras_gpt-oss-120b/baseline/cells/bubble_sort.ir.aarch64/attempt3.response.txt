export function l $bubble_sort(l %a, l %n) {
@entry:
%nm1 = subl %n 1
jmp @outer_loop

@outer_loop:
%i = phi l [0, @entry], [%i_next, @outer_inc]
%cont = csltl %i %nm1
jnz %cont, @outer_body, @exit

@outer_body:
%j = phi l [0, @outer_loop], [%j_next, @inner_inc]
%limit = subl %nm1 %i
%condj = csltl %j %limit
jnz %condj, @inner_body, @outer_inc

@inner_body:
%offj = mull %j 8
%addrj = addl %a %offj
%addrj1 = addl %addrj 8
%valj = loadl %addrj
%valj1 = loadl %addrj1
%swapc = csgtl %valj %valj1
jnz %swapc, @swap, @inner_inc

@swap:
storel %valj1 %addrj
storel %valj %addrj1
jmp @inner_inc

@inner_inc:
%j_next = addl %j 1
jmp @outer_body

@outer_inc:
%i_next = addl %i 1
jmp @outer_loop

@exit:
retl 0
}
