export function l $bubble_sort(l %a, l %n) {
@entry:
    %nm1 = subl %n 1
    jmp @outer_header

@outer_header:
    %i = phi l [0, @entry], [%i_next, @outer_inc]
    %cond_i = cslt %i %nm1
    jnz %cond_i, @inner_header, @exit

@inner_header:
    %j = phi l [0, @outer_header], [%j_next, @inner_inc]
    %limit = subl %nm1 %i
    %cond_j = cslt %j %limit
    jnz %cond_j, @inner_body, @outer_inc

@inner_body:
    %offj = mull %j 8
    %addrj = addl %a %offj
    %addrj1 = addl %addrj 8
    %valj = loadl %addrj
    %valj1 = loadl %addrj1
    %swapc = csgt %valj %valj1
    jnz %swapc, @swap, @inner_inc

@swap:
    storel %valj1 %addrj
    storel %valj %addrj1
    jmp @inner_inc

@inner_inc:
    %j_next = addl %j 1
    jmp @inner_header

@outer_inc:
    %i_next = addl %i 1
    jmp @outer_header

@exit:
    retl 0
}
