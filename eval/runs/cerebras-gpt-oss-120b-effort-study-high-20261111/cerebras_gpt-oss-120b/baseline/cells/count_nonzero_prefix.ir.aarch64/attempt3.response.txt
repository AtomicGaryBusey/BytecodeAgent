export function l $count_nonzero_prefix(l %a, l %n) {
@entry:
    %i0 = const 0
    jmp @head

@head:
    %i = phi %i0 @entry, %i_next @inc
    %c = csltl %i, %n
    jz %c, @exit, @load

@load:
    %off = shll %i, 3
    %addr = add %a, %off
    %val = loadl %addr
    %c2 = ceql %val, 0
    jnz %c2, @exit, @inc

@inc:
    %i_next = add %i, 1
    jmp @head

@exit:
    ret %i
}
