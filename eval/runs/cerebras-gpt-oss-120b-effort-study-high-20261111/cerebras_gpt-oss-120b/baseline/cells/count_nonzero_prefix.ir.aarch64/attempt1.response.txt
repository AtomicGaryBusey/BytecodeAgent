export function l $count_nonzero_prefix(l %a, l %n) {
@entry:
    %i0 = const 0
    jmp @head

@head:
    %i = phi %i0, @entry, %i_next, @inc
    %cond_lt = csltl %i, %n
    jz %cond_lt, @exit, @load

@load:
    %off = shll %i, 3
    %addr = addl %a, %off
    %val = loadl %addr
    %cond_eq = ceql %val, 0
    jnz %cond_eq, @exit, @inc

@inc:
    %i_next = addl %i, 1
    jmp @head

@exit:
    ret %i
}
