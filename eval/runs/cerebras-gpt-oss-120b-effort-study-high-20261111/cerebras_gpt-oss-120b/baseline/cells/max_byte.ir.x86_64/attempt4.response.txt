export function l $max_byte(l, l) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    %one =l const 1
    jmp @loop
@loop:
    %i =l phi %i0 @entry %i_next @body
    %max =l phi %max0 @entry %max_next @body
    %c =w csltl %i %1
    jnz %c, @body
    ret %max
@body:
    %addr =l add %0 %i
    %v =l loadub %addr
    %gt =w csltl %max %v
    %max_next =l sel %gt %v %max
    %i_next =l add %i %one
    jmp @loop
}
