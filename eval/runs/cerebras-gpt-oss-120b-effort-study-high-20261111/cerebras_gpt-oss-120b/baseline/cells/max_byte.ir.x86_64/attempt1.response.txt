export function l $max_byte(p: l, n: l) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    jmp @cond
@cond:
    %i =l phi %i0, %i_next
    %max =l phi %max0, %max_inc
    %c =w csltl %i, %n
    jnz %c, @load
    ret %max
@load:
    %addr =l add %p, %i
    %v =l loadub %addr
    %gt =w csltl %max, %v
    jnz %gt, @set_max
    jmp @no_set_max
@set_max:
    %max_set =l %v
    jmp @inc
@no_set_max:
    %max_nset =l %max
    jmp @inc
@inc:
    %max_inc =l phi %max_set, %max_nset
    %i_next =l add %i, const 1
    jmp @cond
}
