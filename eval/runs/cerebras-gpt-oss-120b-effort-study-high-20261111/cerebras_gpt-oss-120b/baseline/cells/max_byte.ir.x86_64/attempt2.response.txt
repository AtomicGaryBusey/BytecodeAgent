export function l $max_byte(p: l, n: l) {
@entry:
    %i0 =l const 0
    %max0 =l const 0
    %one =l const 1
    jmp @loop_cond

@loop_cond:
    %i =l phi %i0 @entry %i_next @loop_body
    %max =l phi %max0 @entry %max_next @loop_body
    %c =w csltl %i %n
    jnz %c, @loop_body
    ret %max

@loop_body:
    %addr =l add %p %i
    %v =l loadub %addr
    %gt =w csltl %max %v
    %max_next =l sel %gt %v %max
    %i_next =l add %i %one
    jmp @loop_cond
}
