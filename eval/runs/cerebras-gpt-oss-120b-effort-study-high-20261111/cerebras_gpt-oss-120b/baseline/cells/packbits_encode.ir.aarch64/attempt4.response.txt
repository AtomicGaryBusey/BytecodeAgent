export function l $packbits_encode(l src, l n, l out, l cap) {
@entry
    %zero =imm 0
    %one =imm 1
    %two =imm 2
    %neg_one =imm -1
    %c257 =imm 257
    %c128 =imm 128
    %c3 =imm 3
    jmp @i_loop

@i_loop
    %i =phi %zero, @entry, %i_rep_next, @repeat_write, %i_lit_next, @lit_copy_done
    %out_idx =phi %zero, @entry, %out_idx_rep_next, @repeat_write, %out_idx_lit_next, @lit_copy_done
    %cond =csltl %i, %n
    jnz %cond, @run_init, @exit

@run_init
    %src_i =add %src, %i
    %b0 =w loadub %src_i
    jmp @run_check

@run_check
    %r =phi %one, @run_init, %r_next, @run_continue
    %i_plus_r =add %i, %r
    %cond1 =csgel %i_plus_r, %n
    %cond2 =csgel %r, %c128
    %break =or %cond1, %cond2
    jnz %break, @run_done, @run_cmp

@run_cmp
    %addr_next =add %src, %i_plus_r
    %b_next =w loadub %addr_next
    %eq =ceql %b_next, %b0
    jnz %eq, @run_continue, @run_done

@run_continue
    %r_next =add %r, %one
    jmp @run_check

@run_done
    %repeat_cond =csgel %r, %c3
    jnz %repeat_cond, @repeat_pkt, @lit_check

@repeat_pkt
    %remain =sub %cap, %out_idx
    %overflow =csltl %remain, %two
    jnz %overflow, @overflow, @repeat_write

@repeat_write
    %hdr =sub %c257, %r
    %out_hdr_ptr =add %out, %out_idx
    storeb %hdr, %out_hdr_ptr
    %out_val_ptr =add %out_hdr_ptr, %one
    storeb %b0, %out_val_ptr
    %out_idx_rep_next =add %out_idx, %two
    %i_rep_next =add %i, %r
    jmp @i_loop

@lit_check
    %j =phi %i, @run_done, %j_next, @lit_continue
    %j_ge_n =csgel %j, %n
    %len =sub %j, %i
    %len_ge_128 =csgel %len, %c128
    %break1 =or %j_ge_n, %len_ge_128
    jnz %break1, @lit_done, @lit_has_three

@lit_has_three
    %j_plus_2 =add %j, %two
    %has_three =csltl %j_plus_2, %n
    jnz %has_three, @lit_check_run, @lit_continue

@lit_check_run
    %addr_j =add %src, %j
    %b_j =w loadub %addr_j
    %j1 =add %j, %one
    %addr_j1 =add %src, %j1
    %b_j1 =w loadub %addr_j1
    %j2 =add %j, %two
    %addr_j2 =add %src, %j2
    %b_j2 =w loadub %addr_j2
    %eq01 =ceql %b_j, %b_j1
    %eq02 =ceql %b_j, %b_j2
    %run_ge_3 =and %eq01, %eq02
    jnz %run_ge_3, @lit_done, @lit_continue

@lit_continue
    %j_next =add %j, %one
    jmp @lit_check

@lit_done
    %len_lit =sub %j, %i
    %need =add %len_lit, %one
    %remain =sub %cap, %out_idx
    %overflow =csltl %remain, %need
    jnz %overflow, @overflow, @lit_write_header

@lit_write_header
    %hdr =sub %len_lit, %one
    %out_hdr_ptr =add %out, %out_idx
    storeb %hdr, %out_hdr_ptr
    jmp @lit_copy

@lit_copy
    %k =phi %zero, @lit_write_header, %k_next, @lit_copy_body
    %k_lt_len =csltl %k, %len_lit
    jnz %k_lt_len, @lit_copy_body, @lit_copy_done

@lit_copy_body
    %src_off =add %i, %k
    %src_addr =add %src, %src_off
    %b =w loadub %src_addr
    %out_off =add %out_idx, %k
    %out_off1 =add %out_off, %one
    %out_addr =add %out, %out_off1
    storeb %b, %out_addr
    %k_next =add %k, %one
    jmp @lit_copy

@lit_copy_done
    %out_idx_lit_next =add %out_idx, %need
    %i_lit_next =%j
    jmp @i_loop

@overflow
    ret %neg_one

@exit
    ret %out_idx
}
