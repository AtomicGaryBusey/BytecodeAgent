export function l $bmax(l a, l n) {
@entry:
    %zero = constl 0
    %one = constl 1
    %eight = constl 8
    %max0 = loadl a
    %cnt0 = subl n, %one
    %ptr0 = addl a, %eight
    %cntZero = ceql %cnt0, %zero
    jz %cntZero, @loop
    ret %max0

@loop:
    %max = phi %max0, %max_next
    %ptr = phi %ptr0, %ptr_next
    %cnt = phi %cnt0, %cnt_next
    %val = loadl %ptr
    %gt = csltl %max, %val
    %max_next = sell %gt, %val, %max
    %ptr_next = addl %ptr, %eight
    %cnt_next = subl %cnt, %one
    %cntPositive = csltl %zero, %cnt_next
    jnz %cntPositive, @loop
    ret %max_next
}
