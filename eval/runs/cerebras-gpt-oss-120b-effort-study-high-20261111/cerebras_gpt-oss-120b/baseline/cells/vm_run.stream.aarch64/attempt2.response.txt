vm_run:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x9, x0          // code base
    mov x12, x1         // len
    mov x10, x2         // stack base
    mov x11, x3         // cap
    add x13, x9, x12, LSL #3   // code end pointer
    mov x14, x9         // pc pointer
    mov x15, #0         // stack size

.Lloop:
    cmp x14, x13
    b.ge .Lfinish
    ldr x0, [x14]       // opcode
    cmp x0, #0
    b.eq .Lhalt
    cmp x0, #1
    b.eq .Lpush
    cmp x0, #2
    b.eq .Ladd
    cmp x0, #3
    b.eq .Lsub
    cmp x0, #4
    b.eq .Lmul
    cmp x0, #5
    b.eq .Ldup
    cmp x0, #6
    b.eq .Lswap
    cmp x0, #7
    b.eq .Ldrop
    cmp x0, #8
    b.eq .Lover
    cmp x0, #9
    b.eq .Llt
    cmp x0, #10
    b.eq .Ljz
    cmp x0, #11
    b.eq .Ljmp
    add x14, x14, #8
    b .Lloop

.Lhalt:
    b .Lfinish

.Lpush:
    ldr x0, [x14, #8]               // value
    add x1, x10, x15, LSL #3        // &stack[stack_size]
    str x0, [x1]
    add x15, x15, #1
    add x14, x14, #16
    b .Lloop

.Ladd:
    sub x15, x15, #2
    add x1, x10, x15, LSL #3        // &stack[n-2]
    ldr x0, [x1]                    // a
    ldr x2, [x1, #8]                // b
    add x0, x0, x2
    str x0, [x1]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Lsub:
    sub x15, x15, #2
    add x1, x10, x15, LSL #3
    ldr x0, [x1]
    ldr x2, [x1, #8]
    sub x0, x0, x2
    str x0, [x1]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Lmul:
    sub x15, x15, #2
    add x1, x10, x15, LSL #3
    ldr x0, [x1]
    ldr x2, [x1, #8]
    mul x0, x0, x2
    str x0, [x1]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Ldup:
    sub x1, x15, #1
    add x2, x10, x1, LSL #3
    ldr x0, [x2]
    add x3, x10, x15, LSL #3
    str x0, [x3]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Lswap:
    sub x1, x15, #2
    add x2, x10, x1, LSL #3        // address of a
    ldr x0, [x2]                   // a
    ldr x3, [x2, #8]               // b
    str x3, [x2]
    str x0, [x2, #8]
    add x14, x14, #8
    b .Lloop

.Ldrop:
    sub x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Lover:
    sub x1, x15, #2
    add x2, x10, x1, LSL #3
    ldr x0, [x2]
    add x3, x10, x15, LSL #3
    str x0, [x3]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Llt:
    sub x15, x15, #2
    add x1, x10, x15, LSL #3
    ldr x0, [x1]                  // a
    ldr x2, [x1, #8]              // b
    cmp x0, x2
    cset x0, lt
    str x0, [x1]
    add x15, x15, #1
    add x14, x14, #8
    b .Lloop

.Ljz:
    ldr x4, [x14, #8]              // target
    sub x5, x15, #1
    add x6, x10, x5, LSL #3        // address of top
    ldr x7, [x6]                   // value
    sub x15, x15, #1
    cbz x7, .Ljz_target
    add x14, x14, #16
    b .Lloop
.Ljz_target:
    add x14, x9, x4, LSL #3
    b .Lloop

.Ljmp:
    ldr x4, [x14, #8]              // target
    add x14, x9, x4, LSL #3
    b .Lloop

.Lfinish:
    cbz x15, .Lempty
    sub x1, x15, #1
    add x1, x10, x1, LSL #3
    ldr x0, [x1]
    b .Lzero_fill
.Lempty:
    mov x0, #0
.Lzero_fill:
    mov x2, x15                    // i = stack_size
.Lzero_loop:
    cmp x2, x11
    b.ge .Lzero_done
    add x3, x10, x2, LSL #3
    mov x4, #0
    str x4, [x3]
    add x2, x2, #1
    b .Lzero_loop
.Lzero_done:
    ldp x29, x30, [sp], #16
    ret
