export function l $mulhi_u64(l %a, l %b)
@0:
	%a_lo = andl %a, 4294967295
	%a_hi = shrl %a, 32
	%b_lo = andl %b, 4294967295
	%b_hi = shrl %b, 32

	%p0 = mull %a_lo, %b_lo
	%p1 = mull %a_lo, %b_hi
	%p2 = mull %a_hi, %b_lo
	%p3 = mull %a_hi, %b_hi

	%mid_sum = addl %p1, %p2
	%mid_carry = cultl %mid_sum, %p1

	%mid_shift = shll %mid_sum, 32
	%low_sum = addl %p0, %mid_shift
	%low_carry = cultl %low_sum, %p0

	%mid_hi = shrl %mid_sum, 32
	%carry_mid_l = zextl %mid_carry
	%carry_mid_shift = shll %carry_mid_l, 32
	%carry_low_l = zextl %low_carry

	%high_tmp = addl %p3, %mid_hi
	%high_tmp2 = addl %high_tmp, %carry_mid_shift
	%high = addl %high_tmp2, %carry_low_l

	ret %high
