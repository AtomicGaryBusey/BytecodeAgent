export function q $mulhi_u64(q %a, q %b)
@0:
	%a_lo = andq %a, 0xffffffff
	%a_hi = shrq %a, 32
	%b_lo = andq %b, 0xffffffff
	%b_hi = shrq %b, 32

	%p0 = mulq %a_lo, %b_lo
	%p1 = mulq %a_lo, %b_hi
	%p2 = mulq %a_hi, %b_lo
	%p3 = mulq %a_hi, %b_hi

	%mid_sum = addq %p1, %p2
	%mid_carry = cultq %mid_sum, %p1

	%mid_shift = shlq %mid_sum, 32
	%low_sum = addq %p0, %mid_shift
	%low_carry = cultq %low_sum, %p0

	%mid_hi = shrq %mid_sum, 32

	%mid_carry_l = zextw %mid_carry
	%mid_carry_q = zextl %mid_carry_l
	%mid_carry_shift = shlq %mid_carry_q, 32

	%low_carry_l = zextw %low_carry
	%low_carry_q = zextl %low_carry_l

	%high_tmp = addq %p3, %mid_hi
	%high_tmp2 = addq %high_tmp, %mid_carry_shift
	%high = addq %high_tmp2, %low_carry_q

	ret %high
