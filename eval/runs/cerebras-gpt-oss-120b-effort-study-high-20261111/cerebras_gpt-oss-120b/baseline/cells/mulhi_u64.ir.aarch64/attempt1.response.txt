export function l $mulhi_u64(l a, l b)
@start
  %a_lo = and %a, 0xffffffff
  %a_hi = shr %a, 32
  %b_lo = and %b, 0xffffffff
  %b_hi = shr %b, 32
  %p0 = mul %a_lo, %b_lo
  %p1 = mul %a_lo, %b_hi
  %p2 = mul %a_hi, %b_lo
  %p3 = mul %a_hi, %b_hi
  %c0 = shr %p0, 32
  %sum12 = add %p1, %p2
  %c1_w = cultl %sum12, %p1
  %c1 = zextl %c1_w
  %sum_mid = add %sum12, %c0
  %c2_w = cultl %sum_mid, %sum12
  %c2 = zextl %c2_w
  %carries = add %c1, %c2
  %mid_hi = shr %sum_mid, 32
  %carry_sh = shl %carries, 32
  %mid_high = add %mid_hi, %carry_sh
  %result = add %p3, %mid_high
  ret %result
