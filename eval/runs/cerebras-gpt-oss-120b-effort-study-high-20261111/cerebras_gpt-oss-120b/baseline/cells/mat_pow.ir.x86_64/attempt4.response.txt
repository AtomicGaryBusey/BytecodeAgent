export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
%zero = l 0
%one = l 1
%eight = l 8
%nn = mul l %n, %n
%size_bytes = mul l %nn, %eight
%base = %tmp
%prod = add l %tmp, %size_bytes
%n8 = mul l %n, %eight
%diag_factor = add l %n8, %eight
jmp @out_zero_header

@out_zero_header
%i_zo = phi @entry %zero @out_zero_body %i_zo_next
%cond_zo = cslt %i_zo, %nn
jnz %cond_zo, @out_zero_body, @out_zero_done

@out_zero_body
%off_zo = mul l %i_zo, %eight
%addr_zo = add l %out, %off_zo
storel %zero, %addr_zo
%i_zo_next = add l %i_zo, %one
jmp @out_zero_header

@out_zero_done
jmp @diag_header

@diag_header
%i_diag = phi @out_zero_done %zero @diag_body %i_diag_next
%cond_diag = cslt %i_diag, %n
jnz %cond_diag, @diag_body, @diag_done

@diag_body
%off_diag = mul l %i_diag, %diag_factor
%addr_diag = add l %out, %off_diag
storel %one, %addr_diag
%i_diag_next = add l %i_diag, %one
jmp @diag_header

@diag_done
jmp @copy_a_header

@copy_a_header
%i_copy = phi @diag_done %zero @copy_a_body %i_copy_next
%cond_copy = cslt %i_copy, %nn
jnz %cond_copy, @copy_a_body, @copy_a_done

@copy_a_body
%off_copy = mul l %i_copy, %eight
%src_addr_copy = add l %a, %off_copy
%val_copy = loadl %src_addr_copy
%dst_addr_copy = add l %base, %off_copy
storel %val_copy, %dst_addr_copy
%i_copy_next = add l %i_copy, %one
jmp @copy_a_header

@copy_a_done
jmp @pow_header

@pow_header
%k_cur = phi @copy_a_done %k @pow_continue %k_next
%cond_pow = cne %k_cur, %zero
jnz %cond_pow, @pow_body, @pow_done

@pow_body
%k_and = and l %k_cur, %one
%bit_set = cne %k_and, %zero
jnz %bit_set, @out_mul_i_header, @skip_out_mul

@skip_out_mul
jmp @base_mul_i_header

@out_mul_i_header
%i_om = phi @pow_body %zero @out_mul_i_body %i_om_next
%cond_i_om = cslt %i_om, %n
jnz %cond_i_om, @out_mul_i_body, @out_mul_i_done

@out_mul_i_body
%j_om = phi @out_mul_i_header %zero @out_mul_t_done %j_om_next
%cond_j_om = cslt %j_om, %n
jnz %cond_j_om, @out_mul_j_body, @out_mul_j_done

@out_mul_j_body
%sum_om = phi @out_mul_i_body %zero @out_mul_t_body %sum_next
%t_om = phi @out_mul_i_body %zero @out_mul_t_body %t_next
%cond_t_om = cslt %t_om, %n
jnz %cond_t_om, @out_mul_t_body, @out_mul_t_done

@out_mul_t_body
%row_i_om = mul l %i_om, %n
%offX_idx_om = add l %row_i_om, %t_om
%offX_om = mul l %offX_idx_om, %eight
%addrX_om = add l %out, %offX_om
%valX_om = loadl %addrX_om

%row_t_om = mul l %t_om, %n
%offY_idx_om = add l %row_t_om, %j_om
%offY_om = mul l %offY_idx_om, %eight
%addrY_om = add l %base, %offY_om
%valY_om = loadl %addrY_om

%prod_val_om = mul l %valX_om, %valY_om
%sum_next = add l %sum_om, %prod_val_om
%t_next = add l %t_om, %one
jmp @out_mul_j_body

@out_mul_t_done
%row_i2_om = mul l %i_om, %n
%offZ_idx_om = add l %row_i2_om, %j_om
%offZ_om = mul l %offZ_idx_om, %eight
%addrZ_om = add l %prod, %offZ_om
storel %sum_om, %addrZ_om
%j_om_next = add l %j_om, %one
jmp @out_mul_i_body

@out_mul_j_done
%i_om_next = add l %i_om, %one
jmp @out_mul_i_header

@out_mul_i_done
jmp @copy_out_header

@copy_out_header
%i_co = phi @out_mul_i_done %zero @copy_out_body %i_co_next
%cond_co = cslt %i_co, %nn
jnz %cond_co, @copy_out_body, @copy_out_done

@copy_out_body
%off_co = mul l %i_co, %eight
%src_addr_co = add l %prod, %off_co
%val_co = loadl %src_addr_co
%dst_addr_co = add l %out, %off_co
storel %val_co, %dst_addr_co
%i_co_next = add l %i_co, %one
jmp @copy_out_header

@copy_out_done
jmp @base_mul_i_header

@base_mul_i_header
%i_bm = phi @copy_out_done %zero @base_mul_i_body %i_bm_next
%cond_i_bm = cslt %i_bm, %n
jnz %cond_i_bm, @base_mul_i_body, @base_mul_i_done

@base_mul_i_body
%j_bm = phi @base_mul_i_header %zero @base_mul_t_done %j_bm_next
%cond_j_bm = cslt %j_bm, %n
jnz %cond_j_bm, @base_mul_j_body, @base_mul_j_done

@base_mul_j_body
%sum_bm = phi @base_mul_i_body %zero @base_mul_t_body %sum_next_bm
%t_bm = phi @base_mul_i_body %zero @base_mul_t_body %t_next_bm
%cond_t_bm = cslt %t_bm, %n
jnz %cond_t_bm, @base_mul_t_body, @base_mul_t_done

@base_mul_t_body
%row_i_bm = mul l %i_bm, %n
%offX_idx_bm = add l %row_i_bm, %t_bm
%offX_bm = mul l %offX_idx_bm, %eight
%addrX_bm = add l %base, %offX_bm
%valX_bm = loadl %addrX_bm

%row_t_bm = mul l %t_bm, %n
%offY_idx_bm = add l %row_t_bm, %j_bm
%offY_bm = mul l %offY_idx_bm, %eight
%addrY_bm = add l %base, %offY_bm
%valY_bm = loadl %addrY_bm

%prod_val_bm = mul l %valX_bm, %valY_bm
%sum_next_bm = add l %sum_bm, %prod_val_bm
%t_next_bm = add l %t_bm, %one
jmp @base_mul_j_body

@base_mul_t_done
%row_i2_bm = mul l %i_bm, %n
%offZ_idx_bm = add l %row_i2_bm, %j_bm
%offZ_bm = mul l %offZ_idx_bm, %eight
%addrZ_bm = add l %prod, %offZ_bm
storel %sum_bm, %addrZ_bm
%j_bm_next = add l %j_bm, %one
jmp @base_mul_i_body

@base_mul_j_done
%i_bm_next = add l %i_bm, %one
jmp @base_mul_i_header

@base_mul_i_done
jmp @copy_base_header

@copy_base_header
%i_cb = phi @base_mul_i_done %zero @copy_base_body %i_cb_next
%cond_cb = cslt %i_cb, %nn
jnz %cond_cb, @copy_base_body, @copy_base_done

@copy_base_body
%off_cb = mul l %i_cb, %eight
%src_addr_cb = add l %prod, %off_cb
%val_cb = loadl %src_addr_cb
%dst_addr_cb = add l %base, %off_cb
storel %val_cb, %dst_addr_cb
%i_cb_next = add l %i_cb, %one
jmp @copy_base_header

@copy_base_done
jmp @pow_continue

@pow_continue
%k_next = shr l %k_cur, %one
jmp @pow_header

@pow_done
jmp @tmp_zero_base_header

@tmp_zero_base_header
%i_zb = phi @pow_done %zero @tmp_zero_base_body %i_zb_next
%cond_zb = cslt %i_zb, %nn
jnz %cond_zb, @tmp_zero_base_body, @tmp_zero_base_done

@tmp_zero_base_body
%off_zb = mul l %i_zb, %eight
%addr_zb = add l %base, %off_zb
storel %zero, %addr_zb
%i_zb_next = add l %i_zb, %one
jmp @tmp_zero_base_header

@tmp_zero_base_done
jmp @tmp_zero_prod_header

@tmp_zero_prod_header
%i_zp = phi @tmp_zero_base_done %zero @tmp_zero_prod_body %i_zp_next
%cond_zp = cslt %i_zp, %nn
jnz %cond_zp, @tmp_zero_prod_body, @tmp_zero_prod_done

@tmp_zero_prod_body
%off_zp = mul l %i_zp, %eight
%addr_zp = add l %prod, %off_zp
storel %zero, %addr_zp
%i_zp_next = add l %i_zp, %one
jmp @tmp_zero_prod_header

@tmp_zero_prod_done
jmp @return

@return
%ret = l 0
ret %ret
}
