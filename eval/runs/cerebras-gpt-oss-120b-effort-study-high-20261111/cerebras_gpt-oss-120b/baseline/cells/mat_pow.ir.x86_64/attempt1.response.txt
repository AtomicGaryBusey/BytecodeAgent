export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
%zero = const l 0
%one = const l 1
%eight = const l 8
%nn = mull %n, %n
%size_bytes = mull %nn, %eight
%base = %tmp
%prod = addl %tmp, %size_bytes
%n_plus_one = addl %n, %one
%diag_factor = mull %n_plus_one, %eight
jmp @out_zero_header

@out_zero_header
%idx0 = phi @entry %zero @out_zero_body %idx0_next
%cond0 = csltl %idx0, %nn
jnz %cond0, @out_zero_body, @out_zero_done

@out_zero_body
%off0 = mull %idx0, %eight
%addr0 = addl %out, %off0
storel %zero, %addr0
%idx0_next = addl %idx0, %one
jmp @out_zero_header

@out_zero_done
jmp @diag_header

@diag_header
%i_diag = phi @out_zero_done %zero @diag_body %i_diag_next
%cond_diag = csltl %i_diag, %n
jnz %cond_diag, @diag_body, @diag_done

@diag_body
%off_diag = mull %i_diag, %diag_factor
%addr_diag = addl %out, %off_diag
storel %one, %addr_diag
%i_diag_next = addl %i_diag, %one
jmp @diag_header

@diag_done
jmp @copy_a_header

@copy_a_header
%i_copy = phi @diag_done %zero @copy_a_body %i_copy_next
%cond_copy = csltl %i_copy, %nn
jnz %cond_copy, @copy_a_body, @copy_a_done

@copy_a_body
%off = mull %i_copy, %eight
%src_addr = addl %a, %off
%val = loadl %src_addr
%dst_addr = addl %base, %off
storel %val, %dst_addr
%i_copy_next = addl %i_copy, %one
jmp @copy_a_header

@copy_a_done
jmp @pow_loop

@pow_loop
%k_cur = phi @copy_a_done %k @after_copy_base %k_next
%cond_pow = cneql %k_cur, %zero
jnz %cond_pow, @pow_body, @pow_done

@pow_body
%k_and = andl %k_cur, %one
%bit_set = cneql %k_and, %zero
jnz %bit_set, @out_mul_start, @base_mul_start

@out_mul_start
jmp @out_mul_outer

@out_mul_outer
%idx = phi @out_mul_start %zero @out_mul_store %idx_next
%cond = csltl %idx, %nn
jnz %cond, @out_mul_body, @out_mul_done

@out_mul_body
%row = divl %idx, %n
%col = modl %idx, %n
%t = phi @out_mul_outer %zero @out_mul_inner %t_next
%sum = phi @out_mul_outer %zero @out_mul_inner %sum_next
%cond_t = csltl %t, %n
jnz %cond_t, @out_mul_inner, @out_mul_store

@out_mul_inner
%row_n = mull %row, %n
%row_n_t = addl %row_n, %t
%offX = mull %row_n_t, %eight
%addrX = addl %out, %offX
%valX = loadl %addrX
%t_n = mull %t, %n
%t_n_col = addl %t_n, %col
%offY = mull %t_n_col, %eight
%addrY = addl %base, %offY
%valY = loadl %addrY
%product = mull %valX, %valY
%sum_next = addl %sum, %product
%t_next = addl %t, %one
jmp @out_mul_body

@out_mul_store
%offZ = mull %idx, %eight
%addrZ = addl %prod, %offZ
storel %sum, %addrZ
%idx_next = addl %idx, %one
jmp @out_mul_outer

@out_mul_done
jmp @copy_out_header

@copy_out_header
%c = phi @out_mul_done %zero @copy_out_body %c_next
%cond_c = csltl %c, %nn
jnz %cond_c, @copy_out_body, @copy_out_done

@copy_out_body
%off = mull %c, %eight
%src_addr = addl %prod, %off
%val = loadl %src_addr
%dst_addr = addl %out, %off
storel %val, %dst_addr
%c_next = addl %c, %one
jmp @copy_out_header

@copy_out_done
jmp @base_mul_start

@base_mul_start
jmp @base_mul_outer

@base_mul_outer
%idx = phi @base_mul_start %zero @base_mul_store %idx_next
%cond = csltl %idx, %nn
jnz %cond, @base_mul_body, @base_mul_done

@base_mul_body
%row = divl %idx, %n
%col = modl %idx, %n
%t = phi @base_mul_outer %zero @base_mul_inner %t_next
%sum = phi @base_mul_outer %zero @base_mul_inner %sum_next
%cond_t = csltl %t, %n
jnz %cond_t, @base_mul_inner, @base_mul_store

@base_mul_inner
%row_n = mull %row, %n
%row_n_t = addl %row_n, %t
%offX = mull %row_n_t, %eight
%addrX = addl %base, %offX
%valX = loadl %addrX
%t_n = mull %t, %n
%t_n_col = addl %t_n, %col
%offY = mull %t_n_col, %eight
%addrY = addl %base, %offY
%valY = loadl %addrY
%product = mull %valX, %valY
%sum_next = addl %sum, %product
%t_next = addl %t, %one
jmp @base_mul_body

@base_mul_store
%offZ = mull %idx, %eight
%addrZ = addl %prod, %offZ
storel %sum, %addrZ
%idx_next = addl %idx, %one
jmp @base_mul_outer

@base_mul_done
jmp @copy_base_header

@copy_base_header
%c = phi @base_mul_done %zero @copy_base_body %c_next
%cond_c = csltl %c, %nn
jnz %cond_c, @copy_base_body, @copy_base_done

@copy_base_body
%off = mull %c, %eight
%src_addr = addl %prod, %off
%val = loadl %src_addr
%dst_addr = addl %base, %off
storel %val, %dst_addr
%c_next = addl %c, %one
jmp @copy_base_header

@copy_base_done
%k_next = shrl %k_cur, %one
jmp @pow_loop

@pow_done
jmp @tmp_zero_base_header

@tmp_zero_base_header
%i = phi @pow_done %zero @tmp_zero_base_body %i_next
%cond = csltl %i, %nn
jnz %cond, @tmp_zero_base_body, @tmp_zero_base_done

@tmp_zero_base_body
%off = mull %i, %eight
%addr = addl %base, %off
storel %zero, %addr
%i_next = addl %i, %one
jmp @tmp_zero_base_header

@tmp_zero_base_done
jmp @tmp_zero_prod_header

@tmp_zero_prod_header
%i = phi @tmp_zero_base_done %zero @tmp_zero_prod_body %i_next
%cond = csltl %i, %nn
jnz %cond, @tmp_zero_prod_body, @tmp_zero_prod_done

@tmp_zero_prod_body
%off = mull %i, %eight
%addr = addl %prod, %off
storel %zero, %addr
%i_next = addl %i, %one
jmp @tmp_zero_prod_header

@tmp_zero_prod_done
jmp @return

@return
%ret = const l 0
ret %ret
}
