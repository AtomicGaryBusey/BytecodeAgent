export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
%zero = l 0
%one = l 1
%eight = l 8

%nn = mul %n, %n
%size_bytes = mul %nn, %eight
%base = %tmp
%prod = add %tmp, %size_bytes
%n_plus_one = add %n, %one
%diag_factor = mul %n_plus_one, %eight

jmp @out_zero_header

@out_zero_header
%i_zo = phi @entry %zero @out_zero_body %i_zo_next
%cond_zo = cslt %i_zo, %nn
jnz %cond_zo, @out_zero_body, @out_zero_done

@out_zero_body
%off_zo = mul %i_zo, %eight
%addr_zo = add %out, %off_zo
storel %zero, %addr_zo
%i_zo_next = add %i_zo, %one
jmp @out_zero_header

@out_zero_done

jmp @diag_header

@diag_header
%i_diag = phi @out_zero_done %zero @diag_body %i_diag_next
%cond_diag = cslt %i_diag, %n
jnz %cond_diag, @diag_body, @diag_done

@diag_body
%off_diag = mul %i_diag, %diag_factor
%addr_diag = add %out, %off_diag
storel %one, %addr_diag
%i_diag_next = add %i_diag, %one
jmp @diag_header

@diag_done

jmp @copy_a_header

@copy_a_header
%i_copy = phi @diag_done %zero @copy_a_body %i_copy_next
%cond_copy = cslt %i_copy, %nn
jnz %cond_copy, @copy_a_body, @copy_a_done

@copy_a_body
%off_copy = mul %i_copy, %eight
%src_addr_copy = add %a, %off_copy
%val_copy = loadl %src_addr_copy
%dst_addr_copy = add %base, %off_copy
storel %val_copy, %dst_addr_copy
%i_copy_next = add %i_copy, %one
jmp @copy_a_header

@copy_a_done

jmp @pow_header

@pow_header
%k_cur = phi @copy_a_done %k @pow_body %k_next
%cond_pow = cne %k_cur, %zero
jnz %cond_pow, @pow_body, @pow_done

@pow_body
%k_and = and %k_cur, %one
%bit_set = cne %k_and, %zero
jnz %bit_set, @out_mul_start, @skip_out_mul

@skip_out_mul
jmp @base_mul_start

@out_mul_start
jmp @out_mul_row_header

@out_mul_row_header
%i_om = phi @out_mul_start %zero @out_mul_col_done %i_om_next
%cond_i_om = cslt %i_om, %n
jnz %cond_i_om, @out_mul_row_body, @out_mul_done

@out_mul_row_body
%j_om = phi @out_mul_row_header %zero @out_mul_inner_done %j_om_next
%cond_j_om = cslt %j_om, %n
jnz %cond_j_om, @out_mul_col_header, @out_mul_col_done

@out_mul_col_header
%t_om = phi @out_mul_row_body %zero @out_mul_inner_body %t_om_next
%sum_om = phi @out_mul_row_body %zero @out_mul_inner_body %sum_om_next
%cond_t_om = cslt %t_om, %n
jnz %cond_t_om, @out_mul_inner_body, @out_mul_inner_done

@out_mul_inner_body
%row_n_om = mul %i_om, %n
%offX_idx_om = add %row_n_om, %t_om
%offX_om = mul %offX_idx_om, %eight
%addrX_om = add %out, %offX_om
%valX_om = loadl %addrX_om

%t_n_om = mul %t_om, %n
%offY_idx_om = add %t_n_om, %j_om
%offY_om = mul %offY_idx_om, %eight
%addrY_om = add %base, %offY_om
%valY_om = loadl %addrY_om

%prod_val_om = mul %valX_om, %valY_om
%sum_om_next = add %sum_om, %prod_val_om
%t_om_next = add %t_om, %one
jmp @out_mul_col_header

@out_mul_inner_done
%row_n2_om = mul %i_om, %n
%offZ_idx_om = add %row_n2_om, %j_om
%offZ_om = mul %offZ_idx_om, %eight
%addrZ_om = add %prod, %offZ_om
storel %sum_om, %addrZ_om
%j_om_next = add %j_om, %one
jmp @out_mul_row_body

@out_mul_col_done
%i_om_next = add %i_om, %one
jmp @out_mul_row_header

@out_mul_done
jmp @copy_out_header

@copy_out_header
%c_co = phi @out_mul_done %zero @copy_out_body %c_co_next
%cond_c_co = cslt %c_co, %nn
jnz %cond_c_co, @copy_out_body, @copy_out_done

@copy_out_body
%off_co = mul %c_co, %eight
%src_addr_co = add %prod, %off_co
%val_co = loadl %src_addr_co
%dst_addr_co = add %out, %off_co
storel %val_co, %dst_addr_co
%c_co_next = add %c_co, %one
jmp @copy_out_header

@copy_out_done
jmp @base_mul_start

@base_mul_start
jmp @base_mul_row_header

@base_mul_row_header
%i_bm = phi @base_mul_start %zero @base_mul_col_done %i_bm_next
%cond_i_bm = cslt %i_bm, %n
jnz %cond_i_bm, @base_mul_row_body, @base_mul_done

@base_mul_row_body
%j_bm = phi @base_mul_row_header %zero @base_mul_inner_done %j_bm_next
%cond_j_bm = cslt %j_bm, %n
jnz %cond_j_bm, @base_mul_col_header, @base_mul_col_done

@base_mul_col_header
%t_bm = phi @base_mul_row_body %zero @base_mul_inner_body %t_bm_next
%sum_bm = phi @base_mul_row_body %zero @base_mul_inner_body %sum_bm_next
%cond_t_bm = cslt %t_bm, %n
jnz %cond_t_bm, @base_mul_inner_body, @base_mul_inner_done

@base_mul_inner_body
%row_n_bm = mul %i_bm, %n
%offX_idx_bm = add %row_n_bm, %t_bm
%offX_bm = mul %offX_idx_bm, %eight
%addrX_bm = add %base, %offX_bm
%valX_bm = loadl %addrX_bm

%t_n_bm = mul %t_bm, %n
%offY_idx_bm = add %t_n_bm, %j_bm
%offY_bm = mul %offY_idx_bm, %eight
%addrY_bm = add %base, %offY_bm
%valY_bm = loadl %addrY_bm

%prod_val_bm = mul %valX_bm, %valY_bm
%sum_bm_next = add %sum_bm, %prod_val_bm
%t_bm_next = add %t_bm, %one
jmp @base_mul_col_header

@base_mul_inner_done
%row_n2_bm = mul %i_bm, %n
%offZ_idx_bm = add %row_n2_bm, %j_bm
%offZ_bm = mul %offZ_idx_bm, %eight
%addrZ_bm = add %prod, %offZ_bm
storel %sum_bm, %addrZ_bm
%j_bm_next = add %j_bm, %one
jmp @base_mul_row_body

@base_mul_col_done
%i_bm_next = add %i_bm, %one
jmp @base_mul_row_header

@base_mul_done
jmp @copy_base_header

@copy_base_header
%d_cb = phi @base_mul_done %zero @copy_base_body %d_cb_next
%cond_d_cb = cslt %d_cb, %nn
jnz %cond_d_cb, @copy_base_body, @copy_base_done

@copy_base_body
%off_cb = mul %d_cb, %eight
%src_addr_cb = add %prod, %off_cb
%val_cb = loadl %src_addr_cb
%dst_addr_cb = add %base, %off_cb
storel %val_cb, %dst_addr_cb
%d_cb_next = add %d_cb, %one
jmp @copy_base_header

@copy_base_done
%k_next = shr %k_cur, %one
jmp @pow_header

@pow_done
jmp @tmp_zero_base_header

@tmp_zero_base_header
%i_zb = phi @pow_done %zero @tmp_zero_base_body %i_zb_next
%cond_zb = cslt %i_zb, %nn
jnz %cond_zb, @tmp_zero_base_body, @tmp_zero_base_done

@tmp_zero_base_body
%off_zb = mul %i_zb, %eight
%addr_zb = add %base, %off_zb
storel %zero, %addr_zb
%i_zb_next = add %i_zb, %one
jmp @tmp_zero_base_header

@tmp_zero_base_done
jmp @tmp_zero_prod_header

@tmp_zero_prod_header
%i_zp = phi @tmp_zero_base_done %zero @tmp_zero_prod_body %i_zp_next
%cond_zp = cslt %i_zp, %nn
jnz %cond_zp, @tmp_zero_prod_body, @tmp_zero_prod_done

@tmp_zero_prod_body
%off_zp = mul %i_zp, %eight
%addr_zp = add %prod, %off_zp
storel %zero, %addr_zp
%i_zp_next = add %i_zp, %one
jmp @tmp_zero_prod_header

@tmp_zero_prod_done
jmp @return

@return
%ret = l 0
ret %ret
}
