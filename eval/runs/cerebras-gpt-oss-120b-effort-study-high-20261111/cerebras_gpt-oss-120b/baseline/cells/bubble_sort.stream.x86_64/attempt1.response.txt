bubble_sort:
    cmp rsi, 1
    jle .Lreturn
    xor r8, r8
.Louter:
    mov rdx, rsi
    sub rdx, r8
    dec rdx
    jz .Lnext_outer
    xor rcx, rcx
.Linner:
    mov rax, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp rax, r9
    jle .Lskip_swap
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], rax
.Lskip_swap:
    inc rcx
    dec rdx
    jnz .Linner
.Lnext_outer:
    inc r8
    mov rax, rsi
    dec rax
    cmp r8, rax
    jl .Louter
.Lreturn:
    xor eax, eax
    ret
