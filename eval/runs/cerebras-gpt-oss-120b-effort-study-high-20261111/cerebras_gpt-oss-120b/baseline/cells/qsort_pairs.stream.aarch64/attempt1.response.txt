qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    cmp x1, #1
    ble .Lbase_return
    mov x19, x0
    mov x20, x1
    ldr x21, [x19]          // pivot key
    ldr x22, [x19, #8]      // pivot value
    mov x23, #0             // i = 0
    sub x24, x20, #1        // j = n-1
.Lpartition_loop:
    cmp x23, x24
    bgt .Lpartition_done
.Li_loop:
    cmp x23, x24
    bgt .Lpartition_done
    lsl x2, x23, #4
    add x3, x19, x2
    ldr x5, [x3]            // key_i
    ldr x6, [x3, #8]        // value_i
    cmp x5, x21
    blt .Li_inc
    bgt .Li_done
    cmp x6, x22
    blt .Li_inc
    b .Li_done
.Li_inc:
    add x23, x23, #1
    b .Li_loop
.Li_done:
    b .Lj_loop
.Lj_loop:
    cmp x23, x24
    bgt .Lpartition_done
    lsl x2, x24, #4
    add x3, x19, x2
    ldr x7, [x3]            // key_j
    ldr x8, [x3, #8]        // value_j
    cmp x7, x21
    blt .Lj_done
    bgt .Lj_dec
    cmp x8, x22
    bgt .Lj_dec
    b .Lj_done
.Lj_dec:
    sub x24, x24, #1
    b .Lj_loop
.Lj_done:
    b .Lswap_check
.Lswap_check:
    cmp x23, x24
    bgt .Lpartition_done
    lsl x2, x23, #4
    add x3, x19, x2         // addr_i
    lsl x9, x24, #4
    add x10, x19, x9        // addr_j
    ldr x11, [x3]           // key_i
    ldr x12, [x3, #8]       // value_i
    ldr x13, [x10]          // key_j
    ldr x14, [x10, #8]      // value_j
    str x13, [x3]
    str x14, [x3, #8]
    str x11, [x10]
    str x12, [x10, #8]
    add x23, x23, #1
    sub x24, x24, #1
    b .Lpartition_loop
.Lpartition_done:
    lsl x2, x23, #4
    add x25, x19, x2        // right pointer
    sub x26, x20, x23       // right size
    cbz x23, .Lright_rec
    mov x0, x19
    mov x1, x23
    bl qsort_pairs
.Lright_rec:
    cbz x26, .Lbase_return
    mov x0, x25
    mov x1, x26
    bl qsort_pairs
.Lbase_return:
    mov x0, #0
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
