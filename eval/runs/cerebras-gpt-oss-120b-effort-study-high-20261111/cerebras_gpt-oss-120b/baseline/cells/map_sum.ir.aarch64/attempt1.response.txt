export function l $map_sum(l %a, l %n, l %f) {
@entry:
  %i =l phi l 0, l %i_next
  %sum =l phi l 0, l %sum_next
  %cond =w csltl %i, %n
  jnz %cond, @body
  ret %sum

@body:
  %offset =l mul %i, 8
  %ptr =l add %a, %offset
  %val =l loadl %ptr
  %r =l call %f(l %val)
  %sum_next =l add %sum, %r
  %i_next =l add %i, 1
  br @entry
}
