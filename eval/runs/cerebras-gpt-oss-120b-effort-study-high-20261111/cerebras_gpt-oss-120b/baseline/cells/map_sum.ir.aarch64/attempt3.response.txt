export function l $map_sum(l %a, l %n, l %f) {
@start:
    br @loop
@loop:
    %i = phi l 0 @start, %i_next @loop
    %sum = phi l 0 @start, %sum_next @loop
    %cond =w cslt %i, %n
    jnz %cond @body
    ret %sum
@body:
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %r =l call %f(l %val)
    %sum_next =l add %sum, %r
    %i_next =l add %i, 1
    br @loop
}
