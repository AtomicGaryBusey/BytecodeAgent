export function l $count_ugt(%a:pl, %n:l, %t:l) {
    @entry
        %cnt0 = constl 0
        %i0 = constl 0
        jmp @loop(%cnt0, %i0)

    @loop(%cnt:l, %i:l)
        %cond = csltl %i, %n
        jnz %cond, @body(%cnt, %i)
        ret %cnt

    @body(%cnt:l, %i:l)
        %off = mull %i, constl 8
        %addr = addl %a, %off
        %val = loadl %addr
        %gt = cugtl %val, %t
        %inc = zextw %gt
        %cnt1 = addl %cnt, %inc
        %i1 = addl %i, constl 1
        jmp @loop(%cnt1, %i1)
}
