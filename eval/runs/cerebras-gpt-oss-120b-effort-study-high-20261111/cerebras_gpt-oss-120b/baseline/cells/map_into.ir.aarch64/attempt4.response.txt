export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry:
    %i0 =l 0
    %cnt0 =l 0
    jmp @loop(%i0, %cnt0)

@loop(%i, %cnt):
    %cond =w csltl %i %n
    jnz w %cond @body(%i, %cnt)
    ret %cnt

@body(%i, %cnt):
    %off =l mul %i 8
    %src_addr =l add %src %off
    %val =l loadl %src_addr
    %fval =l call %f(l %val)
    %dst_addr =l add %dst %off
    storel %fval, %dst_addr
    %cmp =w csltl %fval %val
    %inc =l zext %cmp
    %cnt1 =l add %cnt %inc
    %i1 =l add %i 1
    jmp @loop(%i1, %cnt1)
}
