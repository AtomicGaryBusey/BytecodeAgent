export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry:
    %zero =l 0
    jmp @loop(%zero, %zero)

@loop(l %i, l %cnt):
    %cond =w csltl %i %n
    jnz %cond @process(%i, %cnt)
    ret %cnt

@process(l %i, l %cnt):
    %off =l mul %i 8
    %src_addr =l add %src %off
    %val =l loadl %src_addr
    %fval =l call %f(l %val)
    %dst_addr =l add %dst %off
    storel %fval, %dst_addr
    %cmp =w csltl %fval %val
    %inc =l zext %cmp
    %cnt1 =l add %cnt %inc
    %i1 =l add %i 1
    jmp @loop(%i1, %cnt1)
}
