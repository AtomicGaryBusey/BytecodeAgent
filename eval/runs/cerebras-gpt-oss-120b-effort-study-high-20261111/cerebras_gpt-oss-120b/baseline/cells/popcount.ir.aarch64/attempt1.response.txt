export function l $popcount(l x) {
@entry
    %sh1 = shrl %x, 1
    %and1 = andl %sh1, 0x5555555555555555
    %x1 = subl %x, %and1
    %sh2 = shrl %x1, 2
    %and2 = andl %sh2, 0x3333333333333333
    %and3 = andl %x1, 0x3333333333333333
    %x2 = addl %and3, %and2
    %sh4 = shrl %x2, 4
    %x3 = addl %x2, %sh4
    %x4 = andl %x3, 0x0F0F0F0F0F0F0F0F
    %mul = mull %x4, 0x0101010101010101
    %result = shrl %mul, 56
    ret %result
}
