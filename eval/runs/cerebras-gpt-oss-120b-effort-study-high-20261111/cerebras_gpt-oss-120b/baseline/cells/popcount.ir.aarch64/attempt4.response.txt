export function l $popcount(l) {
@entry
    %x = copyl %0
    %count = copyl 0l
    jmp @loop %x %count

@loop(%x, %count)
    %zero = ceql %x, 0l
    jnz %zero, @exit %count
    %xm1 = subl %x, 1l
    %newx = andl %x, %xm1
    %newcount = addl %count, 1l
    jmp @loop %newx %newcount

@exit(%count)
    ret %count
}
