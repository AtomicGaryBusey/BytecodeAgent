export function l $base64_encode(l src, l n, l out) {
@entry:
    %full =l div %n, $3
    %rem =l mod %n, $3
    %n_plus_2 =l add %n, $2
    %blocks =l div %n_plus_2, $3
    %out_len =l mul %blocks, $4
    jmp @loop

@loop:
    %i =l phi(l $0, @entry, %i_next, @loop_body)
    %cond_full =w csltl %i, %full
    jnz %cond_full, @loop_body, @tail

@loop_body:
    %src_off =l mul %i, $3
    %src_ptr =l add %src, %src_off
    %b0 =l loadub %src_ptr
    %b1 =l loadub (l add %src_ptr, $1)
    %b2 =l loadub (l add %src_ptr, $2)

    %v0_sh =l shr %b0, $2
    %v0 =l and %v0_sh, $63

    %v1_shl =l shl %b0, $4
    %v1_shr =l shr %b1, $4
    %v1_or =l or %v1_shl, %v1_shr
    %v1 =l and %v1_or, $63

    %v2_shl =l shl %b1, $2
    %v2_shr =l shr %b2, $6
    %v2_or =l or %v2_shl, %v2_shr
    %v2 =l and %v2_or, $63

    %v3 =l and %b2, $63

    ; map v0 to c0_l
    %c0_A =l add %v0, $65
    %c0_a =l add %v0, $71
    %c0_digit =l add %v0, $-4
    %c0_plus =l $43
    %c0_slash =l $47
    %cond0_lt26 =w csltl %v0, $26
    %c0_tmp0 =l sel w %cond0_lt26, %c0_A, %c0_a
    %cond0_lt52 =w csltl %v0, $52
    %c0_tmp1 =l sel w %cond0_lt52, %c0_tmp0, %c0_digit
    %cond0_eq62 =w ceql %v0, $62
    %c0_tmp2 =l sel w %cond0_eq62, %c0_plus, %c0_tmp1
    %cond0_eq63 =w ceql %v0, $63
    %c0_l =l sel w %cond0_eq63, %c0_slash, %c0_tmp2

    ; map v1 to c1_l
    %c1_A =l add %v1, $65
    %c1_a =l add %v1, $71
    %c1_digit =l add %v1, $-4
    %c1_plus =l $43
    %c1_slash =l $47
    %cond1_lt26 =w csltl %v1, $26
    %c1_tmp0 =l sel w %cond1_lt26, %c1_A, %c1_a
    %cond1_lt52 =w csltl %v1, $52
    %c1_tmp1 =l sel w %cond1_lt52, %c1_tmp0, %c1_digit
    %cond1_eq62 =w ceql %v1, $62
    %c1_tmp2 =l sel w %cond1_eq62, %c1_plus, %c1_tmp1
    %cond1_eq63 =w ceql %v1, $63
    %c1_l =l sel w %cond1_eq63, %c1_slash, %c1_tmp2

    ; map v2 to c2_l
    %c2_A =l add %v2, $65
    %c2_a =l add %v2, $71
    %c2_digit =l add %v2, $-4
    %c2_plus =l $43
    %c2_slash =l $47
    %cond2_lt26 =w csltl %v2, $26
    %c2_tmp0 =l sel w %cond2_lt26, %c2_A, %c2_a
    %cond2_lt52 =w csltl %v2, $52
    %c2_tmp1 =l sel w %cond2_lt52, %c2_tmp0, %c2_digit
    %cond2_eq62 =w ceql %v2, $62
    %c2_tmp2 =l sel w %cond2_eq62, %c2_plus, %c2_tmp1
    %cond2_eq63 =w ceql %v2, $63
    %c2_l =l sel w %cond2_eq63, %c2_slash, %c2_tmp2

    ; map v3 to c3_l
    %c3_A =l add %v3, $65
    %c3_a =l add %v3, $71
    %c3_digit =l add %v3, $-4
    %c3_plus =l $43
    %c3_slash =l $47
    %cond3_lt26 =w csltl %v3, $26
    %c3_tmp0 =l sel w %cond3_lt26, %c3_A, %c3_a
    %cond3_lt52 =w csltl %v3, $52
    %c3_tmp1 =l sel w %cond3_lt52, %c3_tmp0, %c3_digit
    %cond3_eq62 =w ceql %v3, $62
    %c3_tmp2 =l sel w %cond3_eq62, %c3_plus, %c3_tmp1
    %cond3_eq63 =w ceql %v3, $63
    %c3_l =l sel w %cond3_eq63, %c3_slash, %c3_tmp2

    %dst_off =l mul %i, $4
    %dst_ptr =l add %out, %dst_off

    storeb %c0_l, %dst_ptr
    storeb %c1_l, (l add %dst_ptr, $1)
    storeb %c2_l, (l add %dst_ptr, $2)
    storeb %c3_l, (l add %dst_ptr, $3)

    %i_next =l add %i, $1
    jmp @loop

@tail:
    %rem_eq0 =w ceql %rem, $0
    jnz %rem_eq0, @return, @rem_check1

@rem_check1:
    %rem_eq1 =w ceql %rem, $1
    jnz %rem_eq1, @rem_one, @rem_two

@rem_one:
    %src_off_one =l mul %full, $3
    %src_ptr_one =l add %src, %src_off_one
    %b0_one =l loadub %src_ptr_one

    %v0_sh_one =l shr %b0_one, $2
    %v0_one =l and %v0_sh_one, $63

    %v1_sh_one =l shl %b0_one, $4
    %v1_one =l and %v1_sh_one, $63

    ; map v0_one
    %c0A_one =l add %v0_one, $65
    %c0a_one =l add %v0_one, $71
    %c0d_one =l add %v0_one, $-4
    %c0p_one =l $43
    %c0s_one =l $47
    %cond0_lt26_one =w csltl %v0_one, $26
    %c0tmp0_one =l sel w %cond0_lt26_one, %c0A_one, %c0a_one
    %cond0_lt52_one =w csltl %v0_one, $52
    %c0tmp1_one =l sel w %cond0_lt52_one, %c0tmp0_one, %c0d_one
    %cond0_eq62_one =w ceql %v0_one, $62
    %c0tmp2_one =l sel w %cond0_eq62_one, %c0p_one, %c0tmp1_one
    %cond0_eq63_one =w ceql %v0_one, $63
    %c0_one =l sel w %cond0_eq63_one, %c0s_one, %c0tmp2_one

    ; map v1_one
    %c1A_one =l add %v1_one, $65
    %c1a_one =l add %v1_one, $71
    %c1d_one =l add %v1_one, $-4
    %c1p_one =l $43
    %c1s_one =l $47
    %cond1_lt26_one =w csltl %v1_one, $26
    %c1tmp0_one =l sel w %cond1_lt26_one, %c1A_one, %c1a_one
    %cond1_lt52_one =w csltl %v1_one, $52
    %c1tmp1_one =l sel w %cond1_lt52_one, %c1tmp0_one, %c1d_one
    %cond1_eq62_one =w ceql %v1_one, $62
    %c1tmp2_one =l sel w %cond1_eq62_one, %c1p_one, %c1tmp1_one
    %cond1_eq63_one =w ceql %v1_one, $63
    %c1_one =l sel w %cond1_eq63_one, %c1s_one, %c1tmp2_one

    %dst_off_one =l mul %full, $4
    %dst_ptr_one =l add %out, %dst_off_one

    storeb %c0_one, %dst_ptr_one
    storeb %c1_one, (l add %dst_ptr_one, $1)
    storeb $61, (l add %dst_ptr_one, $2)
    storeb $61, (l add %dst_ptr_one, $3)
    jmp @return

@rem_two:
    %src_off_two =l mul %full, $3
    %src_ptr_two =l add %src, %src_off_two
    %b0_two =l loadub %src_ptr_two
    %b1_two =l loadub (l add %src_ptr_two, $1)

    %v0_sh_two =l shr %b0_two, $2
    %v0_two =l and %v0_sh_two, $63

    %v1_shl_two =l shl %b0_two, $4
    %v1_shr_two =l shr %b1_two, $4
    %v1_or_two =l or %v1_shl_two, %v1_shr_two
    %v1_two =l and %v1_or_two, $63

    %v2_sh_two =l shl %b1_two, $2
    %v2_two =l and %v2_sh_two, $63

    ; map v0_two
    %c0A_two =l add %v0_two, $65
    %c0a_two =l add %v0_two, $71
    %c0d_two =l add %v0_two, $-4
    %c0p_two =l $43
    %c0s_two =l $47
    %cond0_lt26_two =w csltl %v0_two, $26
    %c0tmp0_two =l sel w %cond0_lt26_two, %c0A_two, %c0a_two
    %cond0_lt52_two =w csltl %v0_two, $52
    %c0tmp1_two =l sel w %cond0_lt52_two, %c0tmp0_two, %c0d_two
    %cond0_eq62_two =w ceql %v0_two, $62
    %c0tmp2_two =l sel w %cond0_eq62_two, %c0p_two, %c0tmp1_two
    %cond0_eq63_two =w ceql %v0_two, $63
    %c0_two =l sel w %cond0_eq63_two, %c0s_two, %c0tmp2_two

    ; map v1_two
    %c1A_two =l add %v1_two, $65
    %c1a_two =l add %v1_two, $71
    %c1d_two =l add %v1_two, $-4
    %c1p_two =l $43
    %c1s_two =l $47
    %cond1_lt26_two =w csltl %v1_two, $26
    %c1tmp0_two =l sel w %cond1_lt26_two, %c1A_two, %c1a_two
    %cond1_lt52_two =w csltl %v1_two, $52
    %c1tmp1_two =l sel w %cond1_lt52_two, %c1tmp0_two, %c1d_two
    %cond1_eq62_two =w ceql %v1_two, $62
    %c1tmp2_two =l sel w %cond1_eq62_two, %c1p_two, %c1tmp1_two
    %cond1_eq63_two =w ceql %v1_two, $63
    %c1_two =l sel w %cond1_eq63_two, %c1s_two, %c1tmp2_two

    ; map v2_two
    %c2A_two =l add %v2_two, $65
    %c2a_two =l add %v2_two, $71
    %c2d_two =l add %v2_two, $-4
    %c2p_two =l $43
    %c2s_two =l $47
    %cond2_lt26_two =w csltl %v2_two, $26
    %c2tmp0_two =l sel w %cond2_lt26_two, %c2A_two, %c2a_two
    %cond2_lt52_two =w csltl %v2_two, $52
    %c2tmp1_two =l sel w %cond2_lt52_two, %c2tmp0_two, %c2d_two
    %cond2_eq62_two =w ceql %v2_two, $62
    %c2tmp2_two =l sel w %cond2_eq62_two, %c2p_two, %c2tmp1_two
    %cond2_eq63_two =w ceql %v2_two, $63
    %c2_two =l sel w %cond2_eq63_two, %c2s_two, %c2tmp2_two

    %dst_off_two =l mul %full, $4
    %dst_ptr_two =l add %out, %dst_off_two

    storeb %c0_two, %dst_ptr_two
    storeb %c1_two, (l add %dst_ptr_two, $1)
    storeb %c2_two, (l add %dst_ptr_two, $2)
    storeb $61, (l add %dst_ptr_two, $3)
    jmp @return

@return:
    ret %out_len
}
