export function l $base64_encode(l src, l n, l out) {

    %rem = l mod %n, $3
    %tmp = l add %n, $2
    %tmp2 = l div %tmp, $3
    %len = l mul %tmp2, $4
    %i0 = l $0
    jmp @cond

@cond:
    %i = phi(l %i0, @entry, %i_next, @loop_body)
    %cmp = w csltl %i, %full
    jnz %cmp, @loop_body, @tail

@loop_body:

    %src_ptr = l add %src, %src_off
    %b0 =l loadub %src_ptr

    %v1 = l and (or (shl %b0, $4), (shr %b1, $4)), $63
    %v2 = l and (or (shl %b1, $2), (shr %b2, $6)), $63
    %v3 = l and %b2, $63

    %dst_off = l mul %i, $4
    %dst_ptr = l add %out, %dst_off

    %c0 = l sel w (csltl %v0, $26), (add %v0, $65),
                (sel w (csltl %v0, $52), (add %v0, $71),
                    (sel w (csltl %v0, $62), (add %v0, $-4),
                        (sel w (ceql %v0, $62), $43, $47))))
    %c1 = l sel w (csltl %v1, $26), (add %v1, $65),
                (sel w (csltl %v1, $52), (add %v1, $71),
                    (sel w (csltl %v1, $62), (add %v1, $-4),
                        (sel w (ceql %v1, $62), $43, $47))))

                    (sel w (csltl %v2, $62), (add %v2, $-4),
                        (sel w (ceql %v2, $62), $43, $47))))
    %c3 = l sel w (csltl %v3, $26), (add %v3, $65),
                (sel w (csltl %v3, $52), (add %v3, $71),
                    (sel w (csltl %v3, $62), (add %v3, $-4),
                        (sel w (ceql %v3, $62), $43, $47))))

    storeb %c0, %dst_ptr
    storeb %c1, (l add %dst_ptr, $1)
    storeb %c2, (l add %dst_ptr, $2)
    storeb %c3, (l add %dst_ptr, $3)

    %i_next = l add %i, $1
    jmp @cond

@tail:

    jnz %rem_is_zero, @return, @rem_not_zero

@rem_not_zero:
    %rem_is_one = w ceql %rem, $1
    jnz %rem_is_one, @rem_one, @rem_two

@rem_one:

    %dst_ptr_tail = l add %out, %dst_off_tail
    %src_off_tail = l mul %full, $3
    %src_ptr_tail = l add %src, %src_off_tail
    %b0 =l loadub %src_ptr_tail

    %v0 = l and (shr %b0, $2), $63
    %v1 = l and (shl %b0, $4), $63

    %c0 = l sel w (csltl %v0, $26), (add %v0, $65),
                (sel w (csltl %v0, $52), (add %v0, $71),
                    (sel w (csltl %v0, $62), (add %v0, $-4),
                        (sel w (ceql %v0, $62), $43, $47))))
    %c1 = l sel w (csltl %v1, $26), (add %v1, $65),

                    (sel w (csltl %v1, $62), (add %v1, $-4),
                        (sel w (ceql %v1, $62), $43, $47))))

    storeb %c0, %dst_ptr_tail
    storeb %c1, (l add %dst_ptr_tail, $1)
    storeb $61, (l add %dst_ptr_tail, $2)
    storeb $61, (l add %dst_ptr_tail, $3)
    jmp @return

@rem_two:
    %dst_off_tail = l mul %full, $4
    %dst_ptr_tail = l add %out, %dst_off_tail
    %src_off_tail = l mul %full, $3
    %src_ptr_tail = l add %src, %src_off_tail
    %b0 =l loadub %src_ptr_tail
    %b1 =l loadub (l add %src_ptr_tail, $1)

    %v0 = l and (shr %b0, $2), $63
    %v1 = l and (or (shl %b0, $4), (shr %b1, $4)), $63
    %v2 = l and (shl %b1, $2), $63

    %c0 = l sel w (csltl %v0, $26), (add %v0, $65),
                (sel w (csltl %v0, $52), (add %v0, $71),
                    (sel w (csltl %v0, $62), (add %v0, $-4),
                        (sel w (ceql %v0, $62), $43, $47))))
