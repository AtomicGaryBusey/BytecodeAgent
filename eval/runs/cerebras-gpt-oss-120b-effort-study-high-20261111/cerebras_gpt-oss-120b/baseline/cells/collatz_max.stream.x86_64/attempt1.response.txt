collatz_max:
    xor r9, r9
    xor rcx, rcx
.Louter_loop:
    inc rcx
    cmp rcx, rdi
    ja .Ldone_outer
    mov rax, rcx
    xor r8, r8
.Linner_loop:
    cmp rax, 1
    je .Linner_done
    inc r8
    test rax, 1
    jz .Leven
    lea rdx, [rax + rax*2]
    add rdx, 1
    mov rax, rdx
    jmp .Linner_loop
.Leven:
    shr rax, 1
    jmp .Linner_loop
.Linner_done:
    cmp r8, r9
    jle .Lcontinue_outer
    mov r9, r8
.Lcontinue_outer:
    jmp .Louter_loop
.Ldone_outer:
    mov rax, r9
    ret
