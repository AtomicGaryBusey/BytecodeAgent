export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =l and %a, 4294967295
    %a1 =l shr %a, 32
    %b0 =l and %b, 4294967295
    %b1 =l shr %b, 32
    %p00 =l mul %a0, %b0
    %p10 =l mul %a1, %b0
    %p01 =l mul %a0, %b1
    %p11 =l mul %a1, %b1
    %carry0 =l shr %p00, 32
    %t =l add %p10, %carry0
    %middle =l and %t, 4294967295
    %carry1 =l shr %t, 32
    %u =l add %p01, %middle
    %carry2 =l shr %u, 32
    %h =l add %p11, %carry1
    %result =l add %h, %carry2
    ret %result
}
