export function l $qsort_pairs(l %a, l %n) {
@start
    %small =w cslel %n, 1
    jnz %small, @done, @setup
@setup
    %last =l sub %n, 1
    %poff =l mul %last, 16
    %pp =l add %a, %poff
    %ppv =l add %pp, 8
    %pk =l loadl %pp
    %pv =l loadl %ppv
    jmp @scan
@scan
    %j =l phi @setup 0, @next %jnext
    %i =l phi @setup 0, @next %inext
    %more =w csltl %j, %last
    jnz %more, @compare, @partitioned
@compare
    %off =l mul %j, 16
    %p =l add %a, %off
    %pval =l add %p, 8
    %k =l loadl %p
    %v =l loadl %pval
    %kl =w csltl %k, %pk
    %ke =w ceql %k, %pk
    %vl =w csltl %v, %pv
    %tie =w and %ke, %vl
    %less =w or %kl, %tie
    jnz %less, @swap, @skip
@swap
    %ioff =l mul %i, 16
    %ip =l add %a, %ioff
    %ipv =l add %ip, 8
    %ik =l loadl %ip
    %iv =l loadl %ipv
    storel %k, %ip
    storel %v, %ipv
    storel %ik, %p
    storel %iv, %pval
    %inc =l add %i, 1
    jmp @next
@skip
    jmp @next
@next
    %inext =l phi @swap %inc, @skip %i
    %jnext =l add %j, 1
    jmp @scan
@partitioned
    %finaloff =l mul %i, 16
    %fp =l add %a, %finaloff
    %fpv =l add %fp, 8
    %fk =l loadl %fp
    %fv =l loadl %fpv
    storel %pk, %fp
    storel %pv, %fpv
    storel %fk, %pp
    storel %fv, %ppv
    %rleft =l call $qsort_pairs(l %a, l %i)
    %rp =l add %fp, 16
    %rn =l sub %last, %i
    %rright =l call $qsort_pairs(l %rp, l %rn)
    jmp @done
@done
    ret 0
}
