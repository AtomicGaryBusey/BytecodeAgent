packbits_decode:
    cld
    xor eax, eax
.Lpacket:
    test rsi, rsi
    jz .Ldone
    movsx r8d, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp r8d, -128
    je .Lpacket
    test r8d, r8d
    js .Lrepeat
    lea r9d, [r8d + 1]
    cmp rsi, r9
    jb .Lmalformed
.Lliteral_loop:
    cmp rax, rcx
    jae .Lfull
    mov r10b, byte ptr [rdi]
    mov byte ptr [rdx + rax], r10b
    inc rdi
    dec rsi
    inc rax
    dec r9
    jnz .Lliteral_loop
    jmp .Lpacket
.Lrepeat:
    test rsi, rsi
    jz .Lmalformed
    mov r10b, byte ptr [rdi]
    inc rdi
    dec rsi
    mov r9d, 1
    sub r9d, r8d
.Lrepeat_loop:
    cmp rax, rcx
    jae .Lfull
    mov byte ptr [rdx + rax], r10b
    inc rax
    dec r9
    jnz .Lrepeat_loop
    jmp .Lpacket
.Lmalformed:
    mov rax, -2
    ret
.Lfull:
    mov rax, -1
.Ldone:
    ret
