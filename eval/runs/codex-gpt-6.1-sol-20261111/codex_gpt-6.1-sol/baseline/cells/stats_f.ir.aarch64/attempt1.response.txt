export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%min =l phi @start 9223372036854775807, @next %newmin
	%max =l phi @start -9223372036854775808, @next %newmax
	%sum =l phi @start 0, @next %newsum
	%count =l phi @start 0, @next %newcount
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%less =w csltl %y, %min
	jnz %less, @minyes, @minno
@minyes
	jmp @minjoin
@minno
	jmp @minjoin
@minjoin
	%newmin =l phi @minyes %y, @minno %min
	%greater =w csgtl %y, %max
	jnz %greater, @maxyes, @maxno
@maxyes
	jmp @next
@maxno
	jmp @next
@next
	%newmax =l phi @maxyes %y, @maxno %max
	%newsum =l add %sum, %y
	%odd =l and %y, 1
	%newcount =l add %count, %odd
	%inext =l add %i, 1
	%more =w csltl %inext, %n
	jnz %more, @loop, @done
@done
	storel %newmin, %out
	%outmax =l add %out, 8
	storel %newmax, %outmax
	%outsum =l add %out, 16
	storel %newsum, %outsum
	ret %newcount
}
