export function l $reverse(l %a, l %n) {
@start
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @swap %next
	%j =l phi @start %last, @swap %prev
	%more =w csltl %i, %j
	jnz %more, @swap, @done
@swap
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%ip =l add %a, %ioff
	%jp =l add %a, %joff
	%iv =l loadl %ip
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%next =l add %i, 1
	%prev =l sub %j, 1
	jmp @loop
@done
	ret 0
}
