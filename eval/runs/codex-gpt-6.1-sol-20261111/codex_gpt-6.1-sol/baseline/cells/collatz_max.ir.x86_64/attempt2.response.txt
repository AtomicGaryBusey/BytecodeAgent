export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @advance %next
	%best =l phi @start 0, @advance %updated
	jmp @inner
@inner
	%x =l phi @outer %k, @step %y
	%count =l phi @outer 0, @step %inc
	%finished =w ceql %x, 1
	jnz %finished, @done, @choose
@choose
	%parity =l and %x, 1
	%isodd =w cnel %parity, 0
	jnz %isodd, @odd, @even
@odd
	%triple =l mul %x, 3
	%oddval =l add %triple, 1
	jmp @step
@even
	%evenval =l shr %x, 1
	jmp @step
@step
	%y =l phi @odd %oddval, @even %evenval
	%inc =l add %count, 1
	jmp @inner
@done
	%greater =w csgtl %count, %best
	jnz %greater, @replace, @keep
@replace
	jmp @advance
@keep
	jmp @advance
@advance
	%updated =l phi @replace %count, @keep %best
	%next =l add %k, 1
	%more =w cslel %next, %n
	jnz %more, @outer, @exit
@exit
	ret %updated
}
