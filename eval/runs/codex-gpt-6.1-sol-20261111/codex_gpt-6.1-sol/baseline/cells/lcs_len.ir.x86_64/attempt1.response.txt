export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%k =l phi @start 0, @initbody %knext
	%initok =w cslel %k, %n
	jnz %initok, @initbody, @outer
@initbody
	%ko =l mul %k, 8
	%kp =l add %row, %ko
	storel 0, %kp
	%knext =l add %k, 1
	jmp @init
@outer
	%i =l phi @init 0, @outerdone %inext
	%iok =w csltl %i, %m
	jnz %iok, @outerbody, @done
@outerbody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%j =l phi @outerbody 1, @update %jnext
	%diag =l phi @outerbody 0, @update %up
	%left =l phi @outerbody 0, @update %value
	%jok =w cslel %j, %n
	jnz %jok, @innerbody, @outerdone
@innerbody
	%jo =l mul %j, 8
	%rp =l add %row, %jo
	%up =l loadl %rp
	%yi =l sub %j, 1
	%yp =l add %y, %yi
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch
@match
	%matched =l add %diag, 1
	jmp @update
@mismatch
	%greater =w csgtl %up, %left
	jnz %greater, @takeup, @takeleft
@takeup
	jmp @update
@takeleft
	jmp @update
@update
	%value =l phi @match %matched, @takeup %up, @takeleft %left
	storel %value, %rp
	%jnext =l add %j, 1
	jmp @inner
@outerdone
	%inext =l add %i, 1
	jmp @outer
@done
	%no =l mul %n, 8
	%np =l add %row, %no
	%result =l loadl %np
	ret %result
}
