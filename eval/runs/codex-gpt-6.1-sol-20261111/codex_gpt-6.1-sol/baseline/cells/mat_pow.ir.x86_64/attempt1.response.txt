export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%count =l mul %n, %n
	%stride =l add %n, 1
	%total =l mul %count, 2
	jmp @init
@init
	%ix =l phi @start 0, @init %ixnext
	%diagrem =l urem %ix, %stride
	%isdiag =w ceql %diagrem, 0
	%value =l extuw %isdiag
	%ioff =l mul %ix, 8
	%iptr =l add %out, %ioff
	storel %value, %iptr
	%ixnext =l add %ix, 1
	%imore =w cultl %ixnext, %count
	jnz %imore, @init, @powercheck
@powercheck
	%zero =w ceql %k, 0
	jnz %zero, @clearstart, @power
@power
	%iteration =l phi @powercheck 0, @copydone %itnext
	jmp @element
@element
	%p =l phi @power 0, @productdone %pnext
	%row =l udiv %p, %n
	%col =l urem %p, %n
	%rowbase =l mul %row, %n
	jmp @dot
@dot
	%t =l phi @element 0, @dot %tnext
	%sum =l phi @element 0, @dot %sumnew
	%leftindex =l add %rowbase, %t
	%leftoff =l mul %leftindex, 8
	%leftptr =l add %out, %leftoff
	%left =l loadl %leftptr
	%rightrow =l mul %t, %n
	%rightindex =l add %rightrow, %col
	%rightoff =l mul %rightindex, 8
	%rightptr =l add %a, %rightoff
	%right =l loadl %rightptr
	%term =l mul %left, %right
	%sumnew =l add %sum, %term
	%tnext =l add %t, 1
	%tmore =w cultl %tnext, %n
	jnz %tmore, @dot, @productdone
@productdone
	%poff =l mul %p, 8
	%pptr =l add %tmp, %poff
	storel %sumnew, %pptr
	%pnext =l add %p, 1
	%pmore =w cultl %pnext, %count
	jnz %pmore, @element, @copystart
@copystart
	jmp @copy
@copy
	%c =l phi @copystart 0, @copy %cnext
	%coff =l mul %c, 8
	%src =l add %tmp, %coff
	%dst =l add %out, %coff
	%cv =l loadl %src
	storel %cv, %dst
	%cnext =l add %c, 1
	%cmore =w cultl %cnext, %count
	jnz %cmore, @copy, @copydone
@copydone
	%itnext =l add %iteration, 1
	%itmore =w cultl %itnext, %k
	jnz %itmore, @power, @clearstart
@clearstart
	jmp @clear
@clear
	%z =l phi @clearstart 0, @clear %znext
	%zoff =l mul %z, 8
	%zptr =l add %tmp, %zoff
	storel 0, %zptr
	%znext =l add %z, 1
	%zmore =w cultl %znext, %total
	jnz %zmore, @clear, @done
@done
	ret 0
}
