export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty = ceql %n, 0
    jnz %empty, @done, @direction
@direction
    %backward = cugtl %dst, %src
    jnz %backward, @back_init, @forward_init
@forward_init
    jmp @forward
@forward
    %i = phi @forward_init 0, @forward %next
    %offset = shl %i, 3
    %sptr = add %src, %offset
    %dptr = add %dst, %offset
    %value = loadl %sptr
    storel %value, %dptr
    %next = add %i, 1
    %more = cultl %next, %n
    jnz %more, @forward, @done
@back_init
    jmp @back
@back
    %remaining = phi @back_init %n, @back %index
    %index = sub %remaining, 1
    %boffset = shl %index, 3
    %bsptr = add %src, %boffset
    %bdptr = add %dst, %boffset
    %bvalue = loadl %bsptr
    storel %bvalue, %bdptr
    %bmore = cnel %index, 0
    jnz %bmore, @back, @done
@done
    ret 0
}
