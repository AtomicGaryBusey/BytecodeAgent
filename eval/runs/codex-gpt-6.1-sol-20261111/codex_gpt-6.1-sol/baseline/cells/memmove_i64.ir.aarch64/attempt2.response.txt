export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w ceql %n, 0
    jnz %empty, @done, @direction
@direction
    %backward =w cugtl %dst, %src
    jnz %backward, @back_init, @forward_init
@forward_init
    jmp @forward
@forward
    %i =l phi @forward_init 0, @forward %next
    %offset =l shl %i, 3
    %sptr =l add %src, %offset
    %dptr =l add %dst, %offset
    %value =l loadl %sptr
    storel %value, %dptr
    %next =l add %i, 1
    %more =w cultl %next, %n
    jnz %more, @forward, @done
@back_init
    jmp @back
@back
    %remaining =l phi @back_init %n, @back %index
    %index =l sub %remaining, 1
    %boffset =l shl %index, 3
    %bsptr =l add %src, %boffset
    %bdptr =l add %dst, %boffset
    %bvalue =l loadl %bsptr
    storel %bvalue, %bdptr
    %bmore =w cnel %index, 0
    jnz %bmore, @back, @done
@done
    ret 0
}
