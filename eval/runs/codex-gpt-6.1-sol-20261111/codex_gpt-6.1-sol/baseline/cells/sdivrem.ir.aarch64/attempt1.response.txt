export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %zero =w ceql %b, 0
    jnz %zero, @divzero, @checkmin
@divzero
    storel %a, %rem
    ret 0
@checkmin
    %ismin =w ceql %a, -9223372036854775808
    %isnegone =w ceql %b, -1
    %overflow =w and %ismin, %isnegone
    jnz %overflow, @special, @normal
@special
    storel 0, %rem
    ret %a
@normal
    %q =l div %a, %b
    %product =l mul %q, %b
    %r =l sub %a, %product
    storel %r, %rem
    ret %q
}
