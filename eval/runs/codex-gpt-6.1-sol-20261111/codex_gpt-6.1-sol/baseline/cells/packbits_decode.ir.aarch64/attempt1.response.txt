export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet
@packet
	%i =l phi @start 0, @nop %next, @done %end
	%o =l phi @start 0, @nop %o, @done %pos
	%more =w csltl %i, %n
	jnz %more, @header, @success
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%next =l add %i, 1
	%noop =w ceql %h, -128
	jnz %noop, @nop, @classify
@nop
	jmp @packet
@classify
	%literal =w csgel %h, 0
	jnz %literal, @literal, @repeat
@literal
	%lc =l add %h, 1
	%le =l add %next, %lc
	%lok =w cslel %le, %n
	jnz %lok, @literalready, @truncated
@literalready
	jmp @copy
@repeat
	%re =l add %next, 1
	%rok =w cslel %re, %n
	jnz %rok, @repeatready, @truncated
@repeatready
	%rc =l sub 1, %h
	jmp @copy
@copy
	%count =l phi @literalready %lc, @repeatready %rc
	%end =l phi @literalready %le, @repeatready %re
	%step =l phi @literalready 1, @repeatready 0
	%base =l add %src, %next
	jmp @loop
@loop
	%left =l phi @copy %count, @write %remaining
	%p =l phi @copy %base, @write %pnext
	%pos =l phi @copy %o, @write %opnext
	%active =w csgtl %left, 0
	jnz %active, @space, @done
@space
	%room =w csltl %pos, %cap
	jnz %room, @write, @overflow
@write
	%b =w loadub %p
	%dst =l add %out, %pos
	storeb %b, %dst
	%pnext =l add %p, %step
	%opnext =l add %pos, 1
	%remaining =l sub %left, 1
	jmp @loop
@done
	jmp @packet
@success
	ret %o
@truncated
	ret -2
@overflow
	ret -1
}
