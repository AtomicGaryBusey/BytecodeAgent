export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%ntmp =l mul %nn, 2
	jmp @init

@init
	%ii =l phi @start 0, @init_next %ii_next
	%ir =l div %ii, %n
	%ic =l rem %ii, %n
	%diag =w ceql %ir, %ic
	%iv =l extuw %diag
	%ioff =l mul %ii, 8
	%ip =l add %out, %ioff
	storel %iv, %ip
	jmp @init_next

@init_next
	%ii_next =l add %ii, 1
	%imore =w csltl %ii_next, %nn
	jnz %imore, @init, @power

@power
	%step =l phi @init_next 0, @copy_done %step_next
	%needed =w csltl %step, %k
	jnz %needed, @product, @clear

@product
	%idx =l phi @power 0, @product_done %idx_next
	%row =l div %idx, %n
	%col =l rem %idx, %n
	%rowbase =l mul %row, %n
	jmp @dot

@dot
	%t =l phi @product 0, @dot %t_next
	%sum =l phi @product 0, @dot %sum_next
	%leftidx =l add %rowbase, %t
	%leftoff =l mul %leftidx, 8
	%leftptr =l add %out, %leftoff
	%left =l loadl %leftptr
	%rightrow =l mul %t, %n
	%rightidx =l add %rightrow, %col
	%rightoff =l mul %rightidx, 8
	%rightptr =l add %a, %rightoff
	%right =l loadl %rightptr
	%term =l mul %left, %right
	%sum_next =l add %sum, %term
	%t_next =l add %t, 1
	%tmore =w csltl %t_next, %n
	jnz %tmore, @dot, @product_done

@product_done
	%poff =l mul %idx, 8
	%pp =l add %tmp, %poff
	storel %sum_next, %pp
	%idx_next =l add %idx, 1
	%pmore =w csltl %idx_next, %nn
	jnz %pmore, @product, @copy

@copy
	%ci =l phi @product_done 0, @copy %ci_next
	%coff =l mul %ci, 8
	%src =l add %tmp, %coff
	%dst =l add %out, %coff
	%cv =l loadl %src
	storel %cv, %dst
	%ci_next =l add %ci, 1
	%cmore =w csltl %ci_next, %nn
	jnz %cmore, @copy, @copy_done

@copy_done
	%step_next =l add %step, 1
	jmp @power

@clear
	%zi =l phi @power 0, @clear %zi_next
	%zoff =l mul %zi, 8
	%zp =l add %tmp, %zoff
	storel 0, %zp
	%zi_next =l add %zi, 1
	%zmore =w csltl %zi_next, %ntmp
	jnz %zmore, @clear, @done

@done
	ret 0
}
