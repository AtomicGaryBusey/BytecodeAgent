export function l $qsort_pairs(l %a, l %n) {
@start
    %small =w cslel %n, 1
    jnz %small, @done, @setup
@setup
    %last =l sub %n, 1
    %poff =l mul %last, 16
    %pp =l add %a, %poff
    %pvptr =l add %pp, 8
    %pk =l loadl %pp
    %pv =l loadl %pvptr
    jmp @loop
@loop
    %i =l phi @setup 0, @advance %nexti
    %j =l phi @setup 0, @advance %nextj
    %more =w csltl %j, %last
    jnz %more, @compare, @partitioned
@compare
    %joff =l mul %j, 16
    %jp =l add %a, %joff
    %jvp =l add %jp, 8
    %jk =l loadl %jp
    %jv =l loadl %jvp
    %kl =w csltl %jk, %pk
    %ke =w ceql %jk, %pk
    %vl =w csltl %jv, %pv
    %eqlo =w and %ke, %vl
    %less =w or %kl, %eqlo
    jnz %less, @swap, @keep
@swap
    %ioff =l mul %i, 16
    %ip =l add %a, %ioff
    %ivp =l add %ip, 8
    %ik =l loadl %ip
    %iv =l loadl %ivp
    storel %jk, %ip
    storel %jv, %ivp
    storel %ik, %jp
    storel %iv, %jvp
    %inc =l add %i, 1
    jmp @advance
@keep
    jmp @advance
@advance
    %nexti =l phi @swap %inc, @keep %i
    %nextj =l add %j, 1
    jmp @loop
@partitioned
    %foff =l mul %i, 16
    %fp =l add %a, %foff
    %fvp =l add %fp, 8
    %fk =l loadl %fp
    %fv =l loadl %fvp
    storel %pk, %fp
    storel %pv, %fvp
    storel %fk, %pp
    storel %fv, %pvptr
    %rightp =l add %fp, 16
    %rightn =l sub %last, %i
    %rleft =l call $qsort_pairs(l %a, l %i)
    %rright =l call $qsort_pairs(l %rightp, l %rightn)
    jmp @done
@done
    ret 0
}
