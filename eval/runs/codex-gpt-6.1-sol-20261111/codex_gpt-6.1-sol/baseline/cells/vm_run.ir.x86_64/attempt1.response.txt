export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch
@fetch
	%off =l shl %pc, 3
	%ip =l add %code, %off
	%op =l loadl %ip
	%halt =w ceql %op, 0
	jnz %halt, @finish, @dispatch1
@dispatch1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @dispatch2
@dispatch2
	%c2 =w ceql %op, 2
	jnz %c2, @binary, @dispatch3
@dispatch3
	%c3 =w ceql %op, 3
	jnz %c3, @binary, @dispatch4
@dispatch4
	%c4 =w ceql %op, 4
	jnz %c4, @binary, @dispatch5
@dispatch5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @dispatch6
@dispatch6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @dispatch7
@dispatch7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @dispatch8
@dispatch8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @dispatch9
@dispatch9
	%c9 =w ceql %op, 9
	jnz %c9, @binary, @dispatch10
@dispatch10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @jump
@push
	%operand =l add %ip, 8
	%v =l loadl %operand
	%pc =l add %pc, 1
	jmp @append
@dup
	%di =l sub %d, 1
	%doff =l shl %di, 3
	%dp =l add %stack, %doff
	%v =l loadl %dp
	jmp @append
@over
	%oi =l sub %d, 2
	%ooff =l shl %oi, 3
	%optr =l add %stack, %ooff
	%v =l loadl %optr
	jmp @append
@append
	%woff =l shl %d, 3
	%wp =l add %stack, %woff
	storel %v, %wp
	%d =l add %d, 1
	jmp @next
@binary
	%bi =l sub %d, 1
	%boff =l shl %bi, 3
	%bp =l add %stack, %boff
	%ap =l sub %bp, 8
	%b =l loadl %bp
	%a =l loadl %ap
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @testsub
@add
	%r =l add %a, %b
	jmp @binarydone
@testsub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @testmul
@sub
	%r =l sub %a, %b
	jmp @binarydone
@testmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @lt
@mul
	%r =l mul %a, %b
	jmp @binarydone
@lt
	%less =w csltl %a, %b
	%r =l extuw %less
	jmp @binarydone
@binarydone
	storel %r, %ap
	%d =l sub %d, 1
	jmp @next
@swap
	%si =l sub %d, 1
	%soff =l shl %si, 3
	%sp =l add %stack, %soff
	%sq =l sub %sp, 8
	%sx =l loadl %sp
	%sy =l loadl %sq
	storel %sy, %sp
	storel %sx, %sq
	jmp @next
@drop
	%d =l sub %d, 1
	jmp @next
@jz
	%d =l sub %d, 1
	%zoff =l shl %d, 3
	%zp =l add %stack, %zoff
	%zv =l loadl %zp
	%zero =w ceql %zv, 0
	jnz %zero, @jump, @notjump
@notjump
	%pc =l add %pc, 2
	jmp @loop
@jump
	%jp =l add %ip, 8
	%pc =l loadl %jp
	jmp @loop
@next
	%pc =l add %pc, 1
	jmp @loop
@finish
	%i =l copy %d
	jmp @clearcheck
@clearcheck
	%more =w csltl %i, %cap
	jnz %more, @clear, @result
@clear
	%coff =l shl %i, 3
	%cp =l add %stack, %coff
	storel 0, %cp
	%i =l add %i, 1
	jmp @clearcheck
@result
	%empty =w ceql %d, 0
	jnz %empty, @emptyresult, @topresult
@emptyresult
	ret 0
@topresult
	%ti =l sub %d, 1
	%toff =l shl %ti, 3
	%tp =l add %stack, %toff
	%top =l loadl %tp
	ret %top
}
