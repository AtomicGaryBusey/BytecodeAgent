mp_mul:
push rbx
push r12
push r13
push r14
mov r9, rdx
mov r10, rcx
lea r11, [rsi + rcx]
xor eax, eax
.Lzero:
cmp rax, r11
jae .Lzero_done
mov qword ptr [r8 + rax*8], 0
inc rax
jmp .Lzero
.Lzero_done:
test rsi, rsi
jz .Lscan_start
test r10, r10
jz .Lscan_start
xor r12d, r12d
.Louter:
mov r14, qword ptr [rdi + r12*8]
lea rbx, [r8 + r12*8]
xor r13d, r13d
xor ecx, ecx
.Linner:
mov rax, r14
mul qword ptr [r9 + rcx*8]
add rax, qword ptr [rbx + rcx*8]
adc rdx, 0
add rax, r13
adc rdx, 0
mov qword ptr [rbx + rcx*8], rax
mov r13, rdx
inc rcx
cmp rcx, r10
jb .Linner
mov qword ptr [rbx + r10*8], r13
inc r12
cmp r12, rsi
jb .Louter
.Lscan_start:
mov rax, r11
.Lscan:
test rax, rax
jz .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lscan
.Ldone:
pop r14
pop r13
pop r12
pop rbx
ret
