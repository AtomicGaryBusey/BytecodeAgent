export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @repeat_done %ri, @literal_done %j
	%used =l phi @start 0, @repeat_done %ru, @literal_done %lu
	%end =w ceql %i, %n
	jnz %end, @success, @run_start
@run_start
	%ip =l add %src, %i
	%b =w loadub %ip
	jmp @run_loop
@run_loop
	%r =l phi @run_start 1, @run_more %rn
	%full =w ceql %r, 128
	jnz %full, @run_ready, @run_bound
@run_bound
	%pos =l add %i, %r
	%past =w cugel %pos, %n
	jnz %past, @run_ready, @run_load
@run_load
	%rp =l add %src, %pos
	%rb =w loadub %rp
	%same =w ceqw %rb, %b
	jnz %same, @run_more, @run_ready
@run_more
	%rn =l add %r, 1
	jmp @run_loop
@run_ready
	%repeat =w cugel %r, 3
	jnz %repeat, @repeat_check, @literal_start
@repeat_check
	%ru =l add %used, 2
	%rfits =w culel %ru, %cap
	jnz %rfits, @repeat_write, @failure
@repeat_write
	%rh =l sub 257, %r
	%rop =l add %out, %used
	storeb %rh, %rop
	%rob =l add %rop, 1
	storeb %b, %rob
	%ri =l add %i, %r
	jmp @repeat_done
@repeat_done
	jmp @outer
@literal_start
	jmp @literal_scan
@literal_scan
	%j =l phi @literal_start %i, @literal_next %jn
	%len =l sub %j, %i
	%lend =w ceql %j, %n
	%lmax =w ceql %len, 128
	%stop =w or %lend, %lmax
	jnz %stop, @literal_check, @triple_bound
@triple_bound
	%j2 =l add %j, 2
	%enough =w cultl %j2, %n
	jnz %enough, @triple_load, @literal_next
@triple_load
	%jp =l add %src, %j
	%jb =w loadub %jp
	%jp1 =l add %jp, 1
	%jb1 =w loadub %jp1
	%eq1 =w ceqw %jb, %jb1
	jnz %eq1, @triple_last, @literal_next
@triple_last
	%jp2 =l add %jp, 2
	%jb2 =w loadub %jp2
	%eq2 =w ceqw %jb, %jb2
	jnz %eq2, @literal_check, @literal_next
@literal_next
	%jn =l add %j, 1
	jmp @literal_scan
@literal_check
	%packet =l add %len, 1
	%lu =l add %used, %packet
	%lfits =w culel %lu, %cap
	jnz %lfits, @literal_header, @failure
@literal_header
	%lh =l sub %len, 1
	%lop =l add %out, %used
	storeb %lh, %lop
	%dst =l add %lop, 1
	jmp @copy
@copy
	%k =l phi @literal_header 0, @copy_byte %kn
	%copied =w ceql %k, %len
	jnz %copied, @literal_done, @copy_byte
@copy_byte
	%cp =l add %ip, %k
	%cb =w loadub %cp
	%dp =l add %dst, %k
	storeb %cb, %dp
	%kn =l add %k, 1
	jmp @copy
@literal_done
	jmp @outer
@failure
	ret -1
@success
	ret %used
}
