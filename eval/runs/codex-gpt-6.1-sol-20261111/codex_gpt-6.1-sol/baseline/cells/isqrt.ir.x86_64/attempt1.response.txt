export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%rem =l phi @start %n, @next %rem_next
	%root =l phi @start 0, @next %root_next
	%bit =l phi @start 4611686018427387904, @next %bit_next
	%trial =l add %root, %bit
	%half =l shr %root, 1
	%fits =w cugel %rem, %trial
	jnz %fits, @take, @skip
@take
	%reduced =l sub %rem, %trial
	%increased =l add %half, %bit
	jmp @next
@skip
	jmp @next
@next
	%rem_next =l phi @take %reduced, @skip %rem
	%root_next =l phi @take %increased, @skip %half
	%bit_next =l shr %bit, 2
	%more =w cnel %bit_next, 0
	jnz %more, @loop, @done
@done
	ret %root_next
}
