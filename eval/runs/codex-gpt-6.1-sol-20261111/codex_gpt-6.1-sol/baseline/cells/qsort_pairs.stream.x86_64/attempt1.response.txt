qsort_pairs:
    cmp rsi, 1
    jle .Ldone
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov rax, rsi
    dec rax
    shl rax, 4
    mov rbx, qword ptr [rdi + rax]
    mov rbp, qword ptr [rdi + rax + 8]
    xor r14d, r14d
    xor r15d, r15d
.Lpartition:
    mov rax, r13
    dec rax
    cmp r15, rax
    jge .Lplace_pivot
    mov rax, r15
    shl rax, 4
    lea r8, [r12 + rax]
    mov rcx, qword ptr [r8]
    mov rdx, qword ptr [r8 + 8]
    cmp rcx, rbx
    jl .Lswap
    jg .Lnext
    cmp rdx, rbp
    jg .Lnext
.Lswap:
    mov rax, r14
    shl rax, 4
    lea r9, [r12 + rax]
    mov r10, qword ptr [r9]
    mov r11, qword ptr [r9 + 8]
    mov qword ptr [r9], rcx
    mov qword ptr [r9 + 8], rdx
    mov qword ptr [r8], r10
    mov qword ptr [r8 + 8], r11
    inc r14
.Lnext:
    inc r15
    jmp .Lpartition
.Lplace_pivot:
    mov rax, r14
    shl rax, 4
    lea r8, [r12 + rax]
    mov rax, r13
    dec rax
    shl rax, 4
    lea r9, [r12 + rax]
    mov rcx, qword ptr [r8]
    mov rdx, qword ptr [r8 + 8]
    mov qword ptr [r9], rcx
    mov qword ptr [r9 + 8], rdx
    mov qword ptr [r8], rbx
    mov qword ptr [r8 + 8], rbp
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs
    lea rax, [r14 + 1]
    mov rsi, r13
    sub rsi, rax
    shl rax, 4
    lea rdi, [r12 + rax]
    call qsort_pairs
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Ldone:
    xor eax, eax
    ret
