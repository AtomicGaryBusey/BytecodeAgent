export function l $reverse(l %a, l %n) {
@start
	%half =l udiv %n, 2
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @swap %next
	%more =w csltl %i, %half
	jnz %more, @swap, @done
@swap
	%j =l sub %last, %i
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%p =l add %a, %ioff
	%q =l add %a, %joff
	%x =l loadl %p
	%y =l loadl %q
	storel %y, %p
	storel %x, %q
	%next =l add %i, 1
	jmp @loop
@done
	ret 0
}
