collatz_max:
    xor eax, eax
    mov esi, 1
.Louter:
    mov rcx, rsi
    xor edx, edx
.Linner:
    cmp rcx, 1
    je .Ldone
    test rcx, 1
    jz .Leven
    lea rcx, [rcx + rcx*2 + 1]
    inc rdx
    jmp .Linner
.Leven:
    shr rcx, 1
    inc rdx
    jmp .Linner
.Ldone:
    cmp rax, rdx
    cmovb rax, rdx
    inc rsi
    cmp rsi, rdi
    jbe .Louter
    ret
