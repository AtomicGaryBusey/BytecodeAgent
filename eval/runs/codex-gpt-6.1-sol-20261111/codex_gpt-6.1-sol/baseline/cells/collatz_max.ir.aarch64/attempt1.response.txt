export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @done %next
	%best =l phi @start 0, @done %updated
	jmp @inner
@inner
	%x =l phi @outer %k, @even %half, @odd %triple
	%steps =l phi @outer 0, @even %inc, @odd %inc
	%finished =w ceql %x, 1
	jnz %finished, @done, @step
@step
	%inc =l add %steps, 1
	%parity =l and %x, 1
	%isodd =w cnel %parity, 0
	jnz %isodd, @odd, @even
@even
	%half =l shr %x, 1
	jmp @inner
@odd
	%times3 =l mul %x, 3
	%triple =l add %times3, 1
	jmp @inner
@done
	%greater =w csgtl %steps, %best
	%mask =l extsw %greater
	%negmask =l neg %mask
	%delta =l sub %steps, %best
	%selected =l and %delta, %negmask
	%updated =l add %best, %selected
	%next =l add %k, 1
	%more =w cslel %next, %n
	jnz %more, @outer, @exit
@exit
	ret %updated
}
