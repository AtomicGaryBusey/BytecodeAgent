bubble_sort:
cmp x1, #1
b.le .Ldone
sub x1, x1, #1
.Louter:
mov x9, #0
mov x10, #0
.Linner:
ldr x11, [x0, x9, lsl #3]
add x12, x9, #1
ldr x13, [x0, x12, lsl #3]
cmp x11, x13
b.le .Lnext
str x13, [x0, x9, lsl #3]
str x11, [x0, x12, lsl #3]
mov x10, #1
.Lnext:
mov x9, x12
cmp x9, x1
b.lt .Linner
cbz x10, .Ldone
subs x1, x1, #1
b.ne .Louter
.Ldone:
mov x0, #0
ret
