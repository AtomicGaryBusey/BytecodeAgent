export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%k =l phi @start 0, @init %knext
	%ko =l mul %k, 8
	%kp =l add %row, %ko
	storel 0, %kp
	%knext =l add %k, 1
	%moreinit =w cslel %knext, %n
	jnz %moreinit, @init, @outertest
@outertest
	%i =l phi @init 0, @outerend %inext
	%morei =w csltl %i, %m
	jnz %morei, @outer, @done
@outer
	%xp =l add %x, %i
	%xb =w loadub %xp
	%hascols =w csgtl %n, 0
	jnz %hascols, @inner, @outerend
@inner
	%j =l phi @outer 1, @update %jnext
	%diag =l phi @outer 0, @update %up
	%left =l phi @outer 0, @update %value
	%off =l mul %j, 8
	%rp =l add %row, %off
	%up =l loadl %rp
	%yj =l sub %j, 1
	%yp =l add %y, %yj
	%yb =w loadub %yp
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @different
@match
	%matched =l add %diag, 1
	jmp @update
@different
	%greater =w csgtl %up, %left
	jnz %greater, @takeup, @takeleft
@takeup
	jmp @update
@takeleft
	jmp @update
@update
	%value =l phi @match %matched, @takeup %up, @takeleft %left
	storel %value, %rp
	%jnext =l add %j, 1
	%morej =w cslel %jnext, %n
	jnz %morej, @inner, @outerend
@outerend
	%inext =l add %i, 1
	jmp @outertest
@done
	%lastoff =l mul %n, 8
	%lastp =l add %row, %lastoff
	%result =l loadl %lastp
	ret %result
}
