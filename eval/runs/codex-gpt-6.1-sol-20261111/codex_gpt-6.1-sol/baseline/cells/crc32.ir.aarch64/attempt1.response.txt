export function l $crc32(l %p, l %n) {
@start
	%c =w copy 4294967295
	%i =l copy 0
	jmp @check
@check
	%more =w cultl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%c =w xor %c, %b
	%j =w copy 0
	jmp @bit
@bit
	%low =w and %c, 1
	%mask =w sub 0, %low
	%poly =w and %mask, 3988292384
	%c =w shr %c, 1
	%c =w xor %c, %poly
	%j =w add %j, 1
	%again =w cultw %j, 8
	jnz %again, @bit, @next
@next
	%i =l add %i, 1
	jmp @check
@done
	%c =w xor %c, 4294967295
	%result =l extuw %c
	ret %result
}
