dot_i32_simd:
    pxor xmm0, xmm0
    xor ecx, ecx
    cmp rdx, 4
    jl .Lreduce
.Lvector:
    movdqu xmm1, xmmword ptr [rdi + rcx*4]
    movdqu xmm2, xmmword ptr [rsi + rcx*4]
    movdqa xmm3, xmm1
    pmuldq xmm3, xmm2
    psrlq xmm1, 32
    psrlq xmm2, 32
    pmuldq xmm1, xmm2
    paddq xmm0, xmm3
    paddq xmm0, xmm1
    add rcx, 4
    mov r8, rdx
    sub r8, rcx
    cmp r8, 4
    jge .Lvector
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    paddq xmm0, xmm1
    movq rax, xmm0
    cmp rcx, rdx
    jge .Ldone
.Lscalar:
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    cmp rcx, rdx
    jl .Lscalar
.Ldone:
    ret
