export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero
@zero
	%z =l phi @start 0, @zero_body %zn
	%zmore =w csltl %z, %n
	jnz %zmore, @zero_body, @outer_init
@zero_body
	%zo =l shl %z, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%zn =l add %z, 1
	jmp @zero
@outer_init
	jmp @outer
@outer
	%i =l phi @outer_init 0, @outer_end %in
	%imore =w csltl %i, %na
	jnz %imore, @outer_body, @scan_init
@outer_body
	%ao =l shl %i, 3
	%ap =l add %a, %ao
	%av =l loadl %ap
	%al =l and %av, 4294967295
	%ah =l shr %av, 32
	jmp @inner
@inner
	%j =l phi @outer_body 0, @inner_body %jn
	%carry =l phi @outer_body 0, @inner_body %nextcarry
	%jmore =w csltl %j, %nb
	jnz %jmore, @inner_body, @outer_end
@inner_body
	%bo =l shl %j, 3
	%bp =l add %b, %bo
	%bv =l loadl %bp
	%bl =l and %bv, 4294967295
	%bh =l shr %bv, 32
	%p00 =l mul %al, %bl
	%p10 =l mul %ah, %bl
	%p01 =l mul %al, %bh
	%p11 =l mul %ah, %bh
	%toph =l shr %p00, 32
	%u =l add %p10, %toph
	%ul =l and %u, 4294967295
	%uh =l shr %u, 32
	%v =l add %p01, %ul
	%vh =l shr %v, 32
	%vl =l shl %v, 32
	%bottom =l and %p00, 4294967295
	%lo =l or %vl, %bottom
	%h0 =l add %p11, %uh
	%hi =l add %h0, %vh
	%idx =l add %i, %j
	%ro =l shl %idx, 3
	%rp =l add %r, %ro
	%old =l loadl %rp
	%s1 =l add %lo, %old
	%c1 =w cultl %s1, %lo
	%c1l =l extuw %c1
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%c2l =l extuw %c2
	%h1 =l add %hi, %c1l
	%nextcarry =l add %h1, %c2l
	storel %s2, %rp
	%jn =l add %j, 1
	jmp @inner
@outer_end
	%endidx =l add %i, %nb
	%endo =l shl %endidx, 3
	%endp =l add %r, %endo
	storel %carry, %endp
	%in =l add %i, 1
	jmp @outer
@scan_init
	jmp @scan
@scan
	%k =l phi @scan_init %n, @scan_zero %km
	%empty =w ceql %k, 0
	jnz %empty, @return, @scan_load
@scan_load
	%km =l sub %k, 1
	%ko =l shl %km, 3
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%iszero =w ceql %kv, 0
	jnz %iszero, @scan_zero, @return
@scan_zero
	jmp @scan
@return
	ret %k
}
