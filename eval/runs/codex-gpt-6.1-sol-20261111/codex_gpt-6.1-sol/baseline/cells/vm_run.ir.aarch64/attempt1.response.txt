export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @fetch
@fetch
	%end =w ceql %pc, %len
	jnz %end, @finish, @decode
@decode
	%off =l shl %pc, 3
	%ip =l add %code, %off
	%op =l loadl %ip
	%c0 =w ceql %op, 0
	jnz %c0, @finish, @test1
@test1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @test2
@test2
	%c2 =w ceql %op, 2
	jnz %c2, @binary, @test3
@test3
	%c3 =w ceql %op, 3
	jnz %c3, @binary, @test4
@test4
	%c4 =w ceql %op, 4
	jnz %c4, @binary, @test5
@test5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @test6
@test6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @test7
@test7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @test8
@test8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @test9
@test9
	%c9 =w ceql %op, 9
	jnz %c9, @binary, @test10
@test10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @jump
@push
	%operandp =l add %ip, 8
	%v =l loadl %operandp
	%pc =l add %pc, 2
	jmp @append
@dup
	%idx =l sub %d, 1
	%voff =l shl %idx, 3
	%vp =l add %stack, %voff
	%v =l loadl %vp
	%pc =l add %pc, 1
	jmp @append
@over
	%idx =l sub %d, 2
	%voff =l shl %idx, 3
	%vp =l add %stack, %voff
	%v =l loadl %vp
	%pc =l add %pc, 1
	jmp @append
@append
	%dstoff =l shl %d, 3
	%dst =l add %stack, %dstoff
	storel %v, %dst
	%d =l add %d, 1
	jmp @fetch
@binary
	%bi =l sub %d, 1
	%ai =l sub %d, 2
	%bo =l shl %bi, 3
	%ao =l shl %ai, 3
	%bp =l add %stack, %bo
	%ap =l add %stack, %ao
	%b =l loadl %bp
	%a =l loadl %ap
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @subtest
@add
	%r =l add %a, %b
	jmp @binarydone
@subtest
	%issub =w ceql %op, 3
	jnz %issub, @sub, @multest
@sub
	%r =l sub %a, %b
	jmp @binarydone
@multest
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @lt
@mul
	%r =l mul %a, %b
	jmp @binarydone
@lt
	%less =w csltl %a, %b
	%r =l extuw %less
	jmp @binarydone
@binarydone
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @fetch
@swap
	%si =l sub %d, 1
	%so =l shl %si, 3
	%sptr =l add %stack, %so
	%lower =l sub %sptr, 8
	%sv =l loadl %sptr
	%lv =l loadl %lower
	storel %lv, %sptr
	storel %sv, %lower
	%pc =l add %pc, 1
	jmp @fetch
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @fetch
@jz
	%d =l sub %d, 1
	%zo =l shl %d, 3
	%zp =l add %stack, %zo
	%zv =l loadl %zp
	%zero =w ceql %zv, 0
	jnz %zero, @jump, @notjump
@notjump
	%pc =l add %pc, 2
	jmp @fetch
@jump
	%targetp =l add %ip, 8
	%pc =l loadl %targetp
	jmp @fetch
@finish
	%result =l copy 0
	%empty =w ceql %d, 0
	jnz %empty, @clearstart, @getresult
@getresult
	%ri =l sub %d, 1
	%ro =l shl %ri, 3
	%rp =l add %stack, %ro
	%result =l loadl %rp
	jmp @clearstart
@clearstart
	%i =l copy %d
	jmp @clearcheck
@clearcheck
	%more =w csltl %i, %cap
	jnz %more, @clear, @return
@clear
	%co =l shl %i, 3
	%cp =l add %stack, %co
	storel 0, %cp
	%i =l add %i, 1
	jmp @clearcheck
@return
	ret %result
}
