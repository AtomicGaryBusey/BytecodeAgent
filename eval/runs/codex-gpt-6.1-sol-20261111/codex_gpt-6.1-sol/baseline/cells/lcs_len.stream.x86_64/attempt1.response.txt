lcs_len:
    xor eax, eax
.Linit:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Linit
    test rsi, rsi
    jz .Ldone
    test rcx, rcx
    jz .Ldone
    push rbx
    push r12
    xor r9d, r9d
.Louter:
    movzx ebx, byte ptr [rdi + r9]
    xor r10d, r10d
    xor r11d, r11d
    mov eax, 1
.Linner:
    mov r12, qword ptr [r8 + rax*8]
    cmp bl, byte ptr [rdx + rax - 1]
    je .Lmatch
    cmp r11, r12
    cmovl r11, r12
    jmp .Lstore
.Lmatch:
    lea r11, [r10 + 1]
.Lstore:
    mov qword ptr [r8 + rax*8], r11
    mov r10, r12
    inc rax
    cmp rax, rcx
    jle .Linner
    inc r9
    cmp r9, rsi
    jl .Louter
    pop r12
    pop rbx
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    ret
