export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @repeat_done %ri, @literal_done %j
	%used =l phi @start 0, @repeat_done %ru, @literal_done %lu
	%end =w ceql %i, %n
	jnz %end, @done, @run_start
@run_start
	%sp =l add %src, %i
	%b =w loadub %sp
	jmp @run_loop
@run_loop
	%r =l phi @run_start 1, @run_more %rn
	%rp =l add %i, %r
	%rbound =w cultl %rp, %n
	%rlimit =w cultl %r, 128
	%rok =w and %rbound, %rlimit
	jnz %rok, @run_test, @run_end
@run_test
	%rsptr =l add %src, %rp
	%rb =w loadub %rsptr
	%same =w ceqw %b, %rb
	jnz %same, @run_more, @run_end
@run_more
	%rn =l add %r, 1
	jmp @run_loop
@run_end
	%repeat =w cugel %r, 3
	jnz %repeat, @repeat_check, @literal_start
@repeat_check
	%ru =l add %used, 2
	%rfits =w culel %ru, %cap
	jnz %rfits, @repeat_write, @fail
@repeat_write
	%rh =l sub 257, %r
	%rd =l add %out, %used
	storeb %rh, %rd
	%rd1 =l add %rd, 1
	storeb %b, %rd1
	jmp @repeat_done
@repeat_done
	%ri =l add %i, %r
	jmp @outer
@literal_start
	jmp @literal_loop
@literal_loop
	%j =l phi @literal_start %i, @literal_advance %jn
	%len =l sub %j, %i
	%jbound =w cultl %j, %n
	%jlimit =w cultl %len, 128
	%jok =w and %jbound, %jlimit
	jnz %jok, @literal_probe, @literal_check
@literal_probe
	%j2 =l add %j, 2
	%has3 =w cultl %j2, %n
	jnz %has3, @literal_test, @literal_advance
@literal_test
	%p0 =l add %src, %j
	%p1 =l add %p0, 1
	%p2 =l add %p0, 2
	%b0 =w loadub %p0
	%b1 =w loadub %p1
	%b2 =w loadub %p2
	%eq1 =w ceqw %b0, %b1
	%eq2 =w ceqw %b0, %b2
	%triple =w and %eq1, %eq2
	jnz %triple, @literal_check, @literal_advance
@literal_advance
	%jn =l add %j, 1
	jmp @literal_loop
@literal_check
	%payload_end =l add %used, %len
	%lu =l add %payload_end, 1
	%lfits =w culel %lu, %cap
	jnz %lfits, @literal_write, @fail
@literal_write
	%ld =l add %out, %used
	%lh =l sub %len, 1
	storeb %lh, %ld
	%dest =l add %ld, 1
	jmp @copy
@copy
	%k =l phi @literal_write 0, @copy_byte %kn
	%copy_end =w ceql %k, %len
	jnz %copy_end, @literal_done, @copy_byte
@copy_byte
	%si =l add %i, %k
	%cp =l add %src, %si
	%cb =w loadub %cp
	%dp =l add %dest, %k
	storeb %cb, %dp
	%kn =l add %k, 1
	jmp @copy
@literal_done
	jmp @outer
@fail
	ret -1
@done
	ret %used
}
