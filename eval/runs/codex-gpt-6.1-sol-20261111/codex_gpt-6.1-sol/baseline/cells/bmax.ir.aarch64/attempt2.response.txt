export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    jmp @loop
@loop
    %i =l phi @start 1, @merge %next
    %max =l phi @start %first, @merge %updated
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w csgtl %value, %max
    jnz %greater, @replace, @keep
@replace
    jmp @merge
@keep
    jmp @merge
@merge
    %updated =l phi @replace %value, @keep %max
    %next =l add %i, 1
    jmp @loop
@done
    ret %max
}
