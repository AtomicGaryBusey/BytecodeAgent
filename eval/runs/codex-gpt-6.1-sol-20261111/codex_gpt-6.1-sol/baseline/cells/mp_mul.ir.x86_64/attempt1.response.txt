export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero_test

@zero_test
	%z =l phi @start 0, @zero_body %znext
	%zmore =w cultl %z, %n
	jnz %zmore, @zero_body, @check

@zero_body
	%zo =l shl %z, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%znext =l add %z, 1
	jmp @zero_test

@check
	%az =w ceql %na, 0
	%bz =w ceql %nb, 0
	%empty =w or %az, %bz
	jnz %empty, @trim_start, @outer

@outer
	%i =l phi @check 0, @row_end %inext
	%ao =l shl %i, 3
	%ap =l add %a, %ao
	%av =l loadl %ap
	%al =l and %av, 4294967295
	%ah =l shr %av, 32
	jmp @inner

@inner
	%j =l phi @outer 0, @inner %jnext
	%carry =l phi @outer 0, @inner %newcarry
	%bo =l shl %j, 3
	%bp =l add %b, %bo
	%bv =l loadl %bp
	%bl =l and %bv, 4294967295
	%bh =l shr %bv, 32
	%p0 =l mul %al, %bl
	%p1 =l mul %ah, %bl
	%p2 =l mul %al, %bh
	%p3 =l mul %ah, %bh
	%p0h =l shr %p0, 32
	%u =l add %p1, %p0h
	%ul =l and %u, 4294967295
	%uh =l shr %u, 32
	%v =l add %p2, %ul
	%vh =l shr %v, 32
	%vl =l shl %v, 32
	%p0l =l and %p0, 4294967295
	%lo =l or %vl, %p0l
	%h0 =l add %p3, %uh
	%hi =l add %h0, %vh
	%idx =l add %i, %j
	%ro =l shl %idx, 3
	%rp =l add %r, %ro
	%old =l loadl %rp
	%s =l add %lo, %old
	%c1 =w cultl %s, %lo
	%t =l add %s, %carry
	%c2 =w cultl %t, %s
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%h1 =l add %hi, %c1l
	%newcarry =l add %h1, %c2l
	storel %t, %rp
	%jnext =l add %j, 1
	%morej =w cultl %jnext, %nb
	jnz %morej, @inner, @row_end

@row_end
	%endidx =l add %i, %nb
	%endo =l shl %endidx, 3
	%endp =l add %r, %endo
	storel %newcarry, %endp
	%inext =l add %i, 1
	%morei =w cultl %inext, %na
	jnz %morei, @outer, @trim_start

@trim_start
	jmp @trim

@trim
	%k =l phi @trim_start %n, @trim_load %km
	%done =w ceql %k, 0
	jnz %done, @return, @trim_load

@trim_load
	%km =l sub %k, 1
	%ko =l shl %km, 3
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%iszero =w ceql %kv, 0
	jnz %iszero, @trim, @return

@return
	ret %k
}
