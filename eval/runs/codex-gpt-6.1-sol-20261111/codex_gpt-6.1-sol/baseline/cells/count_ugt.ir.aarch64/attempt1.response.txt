export function l $count_ugt(l %a, l %n, l %t) {
@start
	jnz %n, @loop, @zero
@zero
	ret 0
@loop
	%i =l phi @start 0, @loop %next
	%count =l phi @start 0, @loop %updated
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w cugtl %value, %t
	%inc =l extuw %greater
	%updated =l add %count, %inc
	%next =l add %i, 1
	%more =w csltl %next, %n
	jnz %more, @loop, @done
@done
	ret %updated
}
