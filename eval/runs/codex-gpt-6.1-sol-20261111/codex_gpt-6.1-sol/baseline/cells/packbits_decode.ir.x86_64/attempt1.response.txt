export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @packet
@packet
	%i =l phi @entry 0, @noop %after, @finish %next
	%written =l phi @entry 0, @noop %written, @finish %newwritten
	%end =w ceql %i, %n
	jnz %end, @success, @header
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%after =l add %i, 1
	%isnoop =w ceql %h, -128
	jnz %isnoop, @noop, @classify
@noop
	jmp @packet
@classify
	%literal =w csgel %h, 0
	jnz %literal, @literal, @repeat
@literal
	%lc =l add %h, 1
	%lend =l add %after, %lc
	%lbad =w csgtl %lend, %n
	jnz %lbad, @malformed, @literalready
@literalready
	jmp @prepare
@repeat
	%rc =l sub 1, %h
	%rend =l add %after, 1
	%rbad =w csgtl %rend, %n
	jnz %rbad, @malformed, @repeatready
@repeatready
	jmp @prepare
@prepare
	%count =l phi @literalready %lc, @repeatready %rc
	%next =l phi @literalready %lend, @repeatready %rend
	%step =l phi @literalready 1, @repeatready 0
	%room =l sub %cap, %written
	%overflow =w csgtl %count, %room
	jnz %overflow, @limited, @full
@limited
	jmp @copyinit
@full
	jmp @copyinit
@copyinit
	%take =l phi @limited %room, @full %count
	%base =l add %src, %after
	%dest =l add %out, %written
	%newwritten =l add %written, %take
	jmp @copy
@copy
	%k =l phi @copyinit 0, @byte %k1
	%offset =l phi @copyinit 0, @byte %offset1
	%done =w ceql %k, %take
	jnz %done, @finish, @byte
@byte
	%sp =l add %base, %offset
	%b =w loadub %sp
	%dp =l add %dest, %k
	storeb %b, %dp
	%k1 =l add %k, 1
	%offset1 =l add %offset, %step
	jmp @copy
@finish
	jnz %overflow, @capacity, @packet
@success
	ret %written
@malformed
	ret -2
@capacity
	ret -1
}
