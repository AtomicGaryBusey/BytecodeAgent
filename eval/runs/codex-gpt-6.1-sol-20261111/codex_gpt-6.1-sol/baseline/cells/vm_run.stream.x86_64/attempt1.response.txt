vm_run:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    je .Lfinish
    mov rax, qword ptr [rdi + r8*8]
    inc r8
    test rax, rax
    je .Lfinish
    cmp eax, 1
    je .Lpush
    cmp eax, 2
    je .Ladd
    cmp eax, 3
    je .Lsub
    cmp eax, 4
    je .Lmul
    cmp eax, 5
    je .Ldup
    cmp eax, 6
    je .Lswap
    cmp eax, 7
    je .Ldrop
    cmp eax, 8
    je .Lover
    cmp eax, 9
    je .Llt
    cmp eax, 10
    je .Ljz
    mov r8, qword ptr [rdi + r8*8]
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rdi + r8*8]
    inc r8
.Lstore:
    mov qword ptr [rdx + r9*8], rax
    inc r9
    jmp .Lloop
.Ladd:
    dec r9
    mov rax, qword ptr [rdx + r9*8]
    add qword ptr [rdx + r9*8 - 8], rax
    jmp .Lloop
.Lsub:
    dec r9
    mov rax, qword ptr [rdx + r9*8]
    sub qword ptr [rdx + r9*8 - 8], rax
    jmp .Lloop
.Lmul:
    dec r9
    mov rax, qword ptr [rdx + r9*8]
    imul rax, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8 - 8], rax
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [rdx + r9*8 - 8]
    jmp .Lstore
.Lswap:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 16], rax
    mov qword ptr [rdx + r9*8 - 8], r10
    jmp .Lloop
.Ldrop:
    dec r9
    jmp .Lloop
.Lover:
    mov rax, qword ptr [rdx + r9*8 - 16]
    jmp .Lstore
.Llt:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    xor eax, eax
    cmp qword ptr [rdx + r9*8 - 8], r10
    setl al
    mov qword ptr [rdx + r9*8 - 8], rax
    jmp .Lloop
.Ljz:
    dec r9
    cmp qword ptr [rdx + r9*8], 0
    jne .Lnotzero
    mov r8, qword ptr [rdi + r8*8]
    jmp .Lloop
.Lnotzero:
    inc r8
    jmp .Lloop
.Lfinish:
    xor eax, eax
    test r9, r9
    je .Lclear
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lclear:
    cmp r9, rcx
    jae .Lreturn
    mov qword ptr [rdx + r9*8], 0
    inc r9
    jmp .Lclear
.Lreturn:
    cld
    ret
