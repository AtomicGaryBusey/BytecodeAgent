export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @next %ni
	%o =l phi @start 0, @next %no
	%more =w csltl %i, %n
	jnz %more, @first, @done
@first
	%p =l add %src, %i
	%b0 =l loadub %p
	%v0 =l shl %b0, 16
	%rem =l sub %n, %i
	%has2 =w csgtl %rem, 1
	jnz %has2, @second, @one
@one
	jmp @ready
@second
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%b1s =l shl %b1, 8
	%v1 =l or %v0, %b1s
	%has3 =w csgtl %rem, 2
	jnz %has3, @third, @two
@two
	jmp @ready
@third
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @ready
@ready
	%v =l phi @one %v0, @two %v1, @third %v2
	%count =l phi @one 1, @two 2, @third 3
	jmp @char
@char
	%k =l phi @ready 0, @stored %nk
	%pad =w csgtl %k, %count
	jnz %pad, @padding, @index
@padding
	jmp @emit
@index
	%bits =l mul %k, 6
	%shift =l sub 18, %bits
	%shifted =l shr %v, %shift
	%x =l and %shifted, 63
	%upper =w csltl %x, 26
	jnz %upper, @uppercase, @testlower
@uppercase
	%cu =l add %x, 65
	jmp @emit
@testlower
	%lower =w csltl %x, 52
	jnz %lower, @lowercase, @testdigit
@lowercase
	%cl =l add %x, 71
	jmp @emit
@testdigit
	%digit =w csltl %x, 62
	jnz %digit, @number, @symbol
@number
	%cd =l sub %x, 4
	jmp @emit
@symbol
	%slash =l and %x, 1
	%delta =l mul %slash, 4
	%cs =l add %delta, 43
	jmp @emit
@emit
	%c =l phi @padding 61, @uppercase %cu, @lowercase %cl, @number %cd, @symbol %cs
	%off =l add %o, %k
	%dst =l add %out, %off
	storeb %c, %dst
	jmp @stored
@stored
	%nk =l add %k, 1
	%again =w csltl %nk, 4
	jnz %again, @char, @next
@next
	%ni =l add %i, %count
	%no =l add %o, 4
	jmp @group
@done
	ret %o
}
