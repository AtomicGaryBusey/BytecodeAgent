export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w ceql %n, 0
    jnz %empty, @done, @direction
@direction
    %backward =w cugtl %dst, %src
    jnz %backward, @backinit, @forward
@forward
    %i =l phi @direction 0, @fnext %inext
    %off =l shl %i, 3
    %sp =l add %src, %off
    %dp =l add %dst, %off
    %v =l loadl %sp
    storel %v, %dp
    %inext =l add %i, 1
    %more =w cultl %inext, %n
    jnz %more, @fnext, @done
@fnext
    jmp @forward
@backinit
    jmp @backward
@backward
    %j =l phi @backinit %n, @bnext %jprev
    %jprev =l sub %j, 1
    %boff =l shl %jprev, 3
    %bsp =l add %src, %boff
    %bdp =l add %dst, %boff
    %bv =l loadl %bsp
    storel %bv, %bdp
    %bmore =w cnel %jprev, 0
    jnz %bmore, @bnext, @done
@bnext
    jmp @backward
@done
    ret 0
}
