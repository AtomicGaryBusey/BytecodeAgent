export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%initial =l sub %n, 1
	jmp @outer
@outer
	%limit =l phi @init %initial, @next %reduced
	jmp @inner
@inner
	%i =l phi @outer 0, @advance %inc
	%offset =l mul %i, 8
	%left =l add %a, %offset
	%right =l add %left, 8
	%x =l loadl %left
	%y =l loadl %right
	%swap =w csgtl %x, %y
	jnz %swap, @exchange, @advance
@exchange
	storel %y, %left
	storel %x, %right
	jmp @advance
@advance
	%inc =l add %i, 1
	%more =w csltl %inc, %limit
	jnz %more, @inner, @next
@next
	%reduced =l sub %limit, 1
	%again =w csgtl %reduced, 0
	jnz %again, @outer, @done
@done
	ret 0
}
