export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @next %inext
	%o =l phi @start 0, @next %onext
	%more =w csltl %i, %n
	jnz %more, @first, @done
@first
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%remaining =l sub %n, %i
	%has2 =w csgtl %remaining, 1
	jnz %has2, @second, @one
@one
	jmp @emit
@second
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%v1shift =l shl %b1, 8
	%v1 =l or %v0, %v1shift
	%has3 =w csgtl %remaining, 2
	jnz %has3, @third, @two
@two
	jmp @emit
@third
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @emit
@emit
	%v =l phi @one %v0, @two %v1, @third %v2
	%valid =l phi @one 2, @two 3, @third 4
	jmp @char
@char
	%j =l phi @emit 0, @written %jnext
	%ispad =w csgel %j, %valid
	jnz %ispad, @padding, @index
@index
	%bits =l mul %j, 6
	%shift =l sub 18, %bits
	%shifted =l shr %v, %shift
	%idx =l and %shifted, 63
	%upper =w csltl %idx, 26
	jnz %upper, @uppercase, @lowercheck
@uppercase
	%cu =l add %idx, 65
	jmp @write
@lowercheck
	%lower =w csltl %idx, 52
	jnz %lower, @lowercase, @digitcheck
@lowercase
	%cl =l add %idx, 71
	jmp @write
@digitcheck
	%digit =w csltl %idx, 62
	jnz %digit, @number, @symbol
@number
	%cd =l sub %idx, 4
	jmp @write
@symbol
	%delta =l sub %idx, 62
	%scaled =l mul %delta, 4
	%cs =l add %scaled, 43
	jmp @write
@padding
	jmp @write
@write
	%c =l phi @uppercase %cu, @lowercase %cl, @number %cd, @symbol %cs, @padding 61
	%offset =l add %o, %j
	%dest =l add %out, %offset
	storeb %c, %dest
	jmp @written
@written
	%jnext =l add %j, 1
	%again =w csltl %jnext, 4
	jnz %again, @char, @next
@next
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @group
@done
	ret %o
}
