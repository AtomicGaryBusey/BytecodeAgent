export function l $max_byte(l %p, l %n) {
@start
	jnz %n, @loop, @empty
@empty
	ret 0
@loop
	%i =l phi @start 0, @next %inext
	%max =l phi @start 0, @next %updated
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w cugtl %b, %max
	jnz %greater, @replace, @keep
@replace
	jmp @next
@keep
	jmp @next
@next
	%updated =l phi @replace %b, @keep %max
	%inext =l add %i, 1
	%more =w cultl %inext, %n
	jnz %more, @loop, @done
@done
	ret %updated
}
