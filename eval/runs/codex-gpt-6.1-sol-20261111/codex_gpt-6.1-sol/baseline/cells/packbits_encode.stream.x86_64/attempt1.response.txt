packbits_encode:
    xor eax, eax
    xor r8d, r8d
.Lnext:
    cmp r8, rsi
    jae .Ldone
    mov r9d, 1
    mov r11b, byte ptr [rdi + r8]
.Lrun:
    cmp r9, 128
    jae .Lrun_done
    lea r10, [r8 + r9]
    cmp r10, rsi
    jae .Lrun_done
    cmp byte ptr [rdi + r10], r11b
    jne .Lrun_done
    inc r9
    jmp .Lrun
.Lrun_done:
    cmp r9, 3
    jb .Lliteral
    mov r10, rcx
    sub r10, rax
    cmp r10, 2
    jb .Lfail
    mov r10d, 257
    sub r10d, r9d
    mov byte ptr [rdx + rax], r10b
    mov byte ptr [rdx + rax + 1], r11b
    add rax, 2
    add r8, r9
    jmp .Lnext
.Lliteral:
    mov r9, r8
.Lliteral_scan:
    cmp r9, rsi
    jae .Lliteral_ready
    mov r10, r9
    sub r10, r8
    cmp r10, 128
    jae .Lliteral_ready
    lea r10, [r9 + 2]
    cmp r10, rsi
    jae .Lliteral_advance
    mov r11b, byte ptr [rdi + r9]
    cmp r11b, byte ptr [rdi + r9 + 1]
    jne .Lliteral_advance
    cmp r11b, byte ptr [rdi + r9 + 2]
    je .Lliteral_ready
.Lliteral_advance:
    inc r9
    jmp .Lliteral_scan
.Lliteral_ready:
    mov r10, r9
    sub r10, r8
    mov r11, rcx
    sub r11, rax
    cmp r11, r10
    jbe .Lfail
    dec r10
    mov byte ptr [rdx + rax], r10b
    inc rax
.Lcopy:
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + rax], r11b
    inc r8
    inc rax
    cmp r8, r9
    jb .Lcopy
    jmp .Lnext
.Lfail:
    mov rax, -1
.Ldone:
    ret
