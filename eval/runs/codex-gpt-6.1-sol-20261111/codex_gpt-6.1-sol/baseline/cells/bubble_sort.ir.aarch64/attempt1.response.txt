export function l $bubble_sort(l %a, l %n) {
@start
	%small = csltl %n, 2
	jnz %small, @done, @init
@init
	%initial = sub %n, 1
	jmp @outer
@outer
	%end =l phi @init %initial, @passdone %nextend
	jmp @inner
@inner
	%i =l phi @outer 0, @advance %nexti
	%offset = mul %i, 8
	%p = add %a, %offset
	%q = add %p, 8
	%left = loadl %p
	%right = loadl %q
	%swap = csgtl %left, %right
	jnz %swap, @exchange, @advance
@exchange
	storel %right, %p
	storel %left, %q
	jmp @advance
@advance
	%nexti = add %i, 1
	%more = csltl %nexti, %end
	jnz %more, @inner, @passdone
@passdone
	%nextend = sub %end, 1
	%again = csgtl %nextend, 0
	jnz %again, @outer, @done
@done
	ret 0
}
