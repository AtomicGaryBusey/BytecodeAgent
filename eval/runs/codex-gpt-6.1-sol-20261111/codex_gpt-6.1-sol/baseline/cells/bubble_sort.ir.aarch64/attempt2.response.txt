export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%initial =l sub %n, 1
	jmp @outer
@outer
	%end =l phi @init %initial, @passdone %nextend
	jmp @inner
@inner
	%i =l phi @outer 0, @advance %nexti
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%q =l add %p, 8
	%left =l loadl %p
	%right =l loadl %q
	%swap =w csgtl %left, %right
	jnz %swap, @exchange, @advance
@exchange
	storel %right, %p
	storel %left, %q
	jmp @advance
@advance
	%nexti =l add %i, 1
	%more =w csltl %nexti, %end
	jnz %more, @inner, @passdone
@passdone
	%nextend =l sub %end, 1
	%again =w csgtl %nextend, 0
	jnz %again, @outer, @done
@done
	ret 0
}
