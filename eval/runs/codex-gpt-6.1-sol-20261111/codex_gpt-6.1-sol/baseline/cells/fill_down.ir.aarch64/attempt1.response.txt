export function l $fill_down(l %a, l %n, l %start) {
@entry
    %empty = ceql %n, 0
    jnz %empty, @done, @loop
@loop
    %i = phi @entry 0, @loop %next
    %offset = mul %i, 8
    %p = add %a, %offset
    %v = sub %start, %i
    storel %v, %p
    %next = add %i, 1
    %more = csltl %next, %n
    jnz %more, @loop, @done
@done
    ret %n
}
