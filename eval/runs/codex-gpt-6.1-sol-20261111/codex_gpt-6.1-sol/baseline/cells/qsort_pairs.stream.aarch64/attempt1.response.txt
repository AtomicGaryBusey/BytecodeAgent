qsort_pairs:
    cmp x1, #1
    b.le .Lreturn
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    mov x19, x0
    mov x20, x1
    sub x9, x20, #1
    add x10, x19, x9, lsl #4
    ldp x23, x24, [x10]
    mov x21, #0
    mov x22, #0
.Lpartition:
    sub x9, x20, #1
    cmp x22, x9
    b.ge .Lpivot
    add x10, x19, x22, lsl #4
    ldp x11, x12, [x10]
    cmp x11, x23
    b.lt .Lswap
    b.gt .Lnext
    cmp x12, x24
    b.gt .Lnext
.Lswap:
    add x13, x19, x21, lsl #4
    ldp x14, x15, [x13]
    stp x11, x12, [x13]
    stp x14, x15, [x10]
    add x21, x21, #1
.Lnext:
    add x22, x22, #1
    b .Lpartition
.Lpivot:
    add x10, x19, x9, lsl #4
    add x13, x19, x21, lsl #4
    ldp x14, x15, [x13]
    stp x23, x24, [x13]
    stp x14, x15, [x10]
    mov x0, x19
    mov x1, x21
    bl qsort_pairs
    add x9, x21, #1
    add x0, x19, x9, lsl #4
    sub x1, x20, x9
    bl qsort_pairs
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x23, x24, [sp, #48]
    ldp x29, x30, [sp], #64
.Lreturn:
    mov x0, #0
    ret
