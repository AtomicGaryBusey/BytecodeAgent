.globl _start

.text
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [buf + r12]
    mov edx, 40000
    sub edx, r12d
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor ebx, ebx
    xor r14d, r14d
parse_loop:
    cmp rbx, r12
    jae parse_end
    cmp byte ptr [buf + rbx], 10
    jne parse_next
    mov dword ptr [starts + r13*4], r14d
    mov eax, ebx
    sub eax, r14d
    mov dword ptr [lens + r13*4], eax
    mov dword ptr [arrA + r13*4], r13d
    inc r13
    lea r14, [rbx + 1]
parse_next:
    inc rbx
    jmp parse_loop
parse_end:
    cmp r14, r12
    jae parse_fin
    mov dword ptr [starts + r13*4], r14d
    mov eax, r12d
    sub eax, r14d
    mov dword ptr [lens + r13*4], eax
    mov dword ptr [arrA + r13*4], r13d
    inc r13
parse_fin:
    test r13, r13
    jz do_exit

    lea r8, [arrA]
    lea r9, [arrB]
    mov r10d, 1
width_loop:
    cmp r10, r13
    jae sort_done
    xor ebx, ebx
merge_loop:
    mov rcx, rbx
    mov rdx, rbx
    add rdx, r10
    cmp rdx, r13
    cmova rdx, r13
    mov r15, rdx
    add r15, r10
    cmp r15, r13
    cmova r15, r13
    mov r14, rdx
    mov rdi, rbx
ML:
    cmp rcx, rdx
    jae RonlyChk
    cmp r14, r15
    jae takeL
    mov eax, dword ptr [r8 + rcx*4]
    mov r11d, dword ptr [r8 + r14*4]
    call cmpl
    test eax, eax
    jnz takeR
takeL:
    mov eax, dword ptr [r8 + rcx*4]
    mov dword ptr [r9 + rdi*4], eax
    inc rcx
    inc rdi
    jmp ML
RonlyChk:
    cmp r14, r15
    jae mdone
takeR:
    mov eax, dword ptr [r8 + r14*4]
    mov dword ptr [r9 + rdi*4], eax
    inc r14
    inc rdi
    jmp ML
mdone:
    lea rbx, [rbx + r10*2]
    cmp rbx, r13
    jb merge_loop
    xchg r8, r9
    add r10, r10
    jmp width_loop

sort_done:
    lea rdi, [outbuf]
    xor ebx, ebx
out_loop:
    cmp rbx, r13
    jae out_write
    mov eax, dword ptr [r8 + rbx*4]
    mov esi, dword ptr [starts + rax*4]
    mov ecx, dword ptr [lens + rax*4]
    lea rsi, [buf + rsi]
copy_loop:
    test ecx, ecx
    jz copy_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jmp copy_loop
copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp out_loop
out_write:
    lea rsi, [outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov edi, 1
    mov eax, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    xor edi, edi
    mov eax, 60
    syscall

# eax = left index a, r11d = right index b; returns eax=1 if b < a else 0
cmpl:
    push rcx
    push rdx
    push rdi
    push rsi
    push r12
    mov r12d, dword ptr [lens + rax*4]
    mov edx, dword ptr [lens + r11*4]
    mov esi, dword ptr [starts + rax*4]
    mov edi, dword ptr [starts + r11*4]
    lea rsi, [buf + rsi]
    lea rdi, [buf + rdi]
    mov ecx, r12d
    cmp edx, ecx
    cmovb ecx, edx
cmp_loop:
    test ecx, ecx
    jz cmp_tie
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jne cmp_diff
    inc rsi
    inc rdi
    dec ecx
    jmp cmp_loop
cmp_diff:
    seta al
    movzx eax, al
    jmp cmp_out
cmp_tie:
    xor eax, eax
    cmp r12d, edx
    seta al
cmp_out:
    pop r12
    pop rsi
    pop rdi
    pop rdx
    pop rcx
    ret

.bss
.balign 16
buf:
    .zero 40000
.balign 16
outbuf:
    .zero 40000
.balign 16
starts:
    .zero 4016
lens:
    .zero 4016
arrA:
    .zero 4016
arrB:
    .zero 4016
