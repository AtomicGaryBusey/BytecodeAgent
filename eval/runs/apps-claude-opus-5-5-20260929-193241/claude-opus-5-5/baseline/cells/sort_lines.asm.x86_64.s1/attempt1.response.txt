.globl _start

.text
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor ecx, ecx
    xor ebx, ebx
    lea rsi, [rip + buf]
    lea r8, [rip + starts]
    lea r9, [rip + lens]
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rsi + rcx], 10
    jne parse_next
    mov [r8 + r13*8], rbx
    mov rax, rcx
    sub rax, rbx
    mov [r9 + r13*8], rax
    inc r13
    lea rbx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rbx, r12
    jae parse_done
    mov [r8 + r13*8], rbx
    mov rax, r12
    sub rax, rbx
    mov [r9 + r13*8], rax
    inc r13
parse_done:
    test r13, r13
    jz finish
    lea r14, [rip + idxa]
    lea r15, [rip + idxb]
    xor ecx, ecx
init_loop:
    mov [r14 + rcx*4], ecx
    inc rcx
    cmp rcx, r13
    jb init_loop

    mov ebp, 1
wloop:
    cmp rbp, r13
    jae sort_done
    xor ebx, ebx
lloop:
    cmp rbx, r13
    jae endpass
    lea r9, [rbx + rbp]
    cmp r9, r13
    cmova r9, r13
    lea r10, [rbx + rbp*2]
    cmp r10, r13
    cmova r10, r13
    mov r12, rbx
    mov r11, r9
mloop:
    cmp r12, r9
    jae takeR_only
    cmp r11, r10
    jae takeL
    mov eax, [r14 + r12*4]
    mov edx, [r14 + r11*4]
    call compare
    test eax, eax
    jnz takeR
takeL:
    mov eax, [r14 + r12*4]
    lea rcx, [r12 + r11]
    sub rcx, r9
    mov [r15 + rcx*4], eax
    inc r12
    jmp mloop
takeR_only:
    cmp r11, r10
    jae mdone
takeR:
    mov eax, [r14 + r11*4]
    lea rcx, [r12 + r11]
    sub rcx, r9
    mov [r15 + rcx*4], eax
    inc r11
    jmp mloop
mdone:
    mov rbx, r10
    jmp lloop
endpass:
    xchg r14, r15
    add rbp, rbp
    jmp wloop

sort_done:
    lea rdi, [rip + outbuf]
    lea r8, [rip + starts]
    lea r9, [rip + lens]
    lea r10, [rip + buf]
    xor ebx, ebx
out_loop:
    cmp rbx, r13
    jae out_write
    mov eax, [r14 + rbx*4]
    mov rsi, [r8 + rax*8]
    add rsi, r10
    mov rcx, [r9 + rax*8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp out_loop
out_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# eax=a, edx=b ; returns eax=1 if line b < line a strictly
compare:
    mov eax, eax
    mov edx, edx
    lea r8, [rip + starts]
    mov rsi, [r8 + rax*8]
    mov rdi, [r8 + rdx*8]
    lea r8, [rip + lens]
    mov rcx, [r8 + rax*8]
    mov r8, [r8 + rdx*8]
    lea rax, [rip + buf]
    add rsi, rax
    add rdi, rax
    mov rax, rcx
    cmp r8, rax
    cmovb rax, r8
cmp_loop:
    test rax, rax
    jz cmp_len
    movzx edx, byte ptr [rsi]
    cmp dl, byte ptr [rdi]
    jne cmp_diff
    inc rsi
    inc rdi
    dec rax
    jmp cmp_loop
cmp_diff:
    seta al
    movzx eax, al
    ret
cmp_len:
    cmp rcx, r8
    seta al
    movzx eax, al
    ret

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 67584
starts:
    .zero 8256
lens:
    .zero 8256
idxa:
    .zero 4128
idxb:
    .zero 4128
