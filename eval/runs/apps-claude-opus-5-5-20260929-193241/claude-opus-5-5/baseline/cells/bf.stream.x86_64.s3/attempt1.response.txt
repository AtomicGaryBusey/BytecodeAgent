.globl _start

.equ INBUF_SIZE, 65540

.text
_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INBUF_SIZE
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je have_bang
    inc rcx
    jmp find_bang
no_bang:
    mov r9, r12
    lea r14, [rsi + r12]
    lea r15, [rsi + r12]
    jmp compile
have_bang:
    mov r9, rcx
    lea r14, [rsi + rcx + 1]
    lea r15, [rsi + r12]
compile:
    lea rbp, [rip + code]
    lea r8, [rip + jmptab]
    lea r10, [rip + bstack]
    xor ebx, ebx
    xor edx, edx
    xor ecx, ecx
comp_loop:
    cmp rcx, r9
    jae comp_end
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x2B
    je store
    cmp al, 0x2D
    je store
    cmp al, 0x3C
    je store
    cmp al, 0x3E
    je store
    cmp al, 0x2E
    je store
    cmp al, 0x2C
    je store
    cmp al, 0x5B
    je open_br
    cmp al, 0x5D
    je close_br
    jmp comp_loop
open_br:
    mov dword ptr [r10 + rdx*4], ebx
    inc rdx
    jmp store
close_br:
    test rdx, rdx
    jz fail
    dec rdx
    mov r11d, dword ptr [r10 + rdx*4]
    mov dword ptr [r8 + rbx*4], r11d
    mov dword ptr [r8 + r11*4], ebx
store:
    mov byte ptr [rbp + rbx], al
    inc rbx
    jmp comp_loop
comp_end:
    test rdx, rdx
    jnz fail
    mov byte ptr [rbp + rbx], 0
    xor ebx, ebx
    lea r13, [rip + tape]
    lea rdi, [rip + outbuf]
run:
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp finish
op_inc:
    inc byte ptr [r13]
    inc rbx
    jmp run
op_dec:
    dec byte ptr [r13]
    inc rbx
    jmp run
op_right:
    inc r13
    inc rbx
    jmp run
op_left:
    dec r13
    inc rbx
    jmp run
op_open:
    cmp byte ptr [r13], 0
    jne 1f
    mov ebx, dword ptr [r8 + rbx*4]
1:
    inc rbx
    jmp run
op_close:
    cmp byte ptr [r13], 0
    je 1f
    mov ebx, dword ptr [r8 + rbx*4]
1:
    inc rbx
    jmp run
op_out:
    mov cl, byte ptr [r13]
    mov byte ptr [rdi], cl
    inc rdi
    inc rbx
    jmp run
op_in:
    cmp r14, r15
    jae 1f
    mov cl, byte ptr [r14]
    inc r14
    mov byte ptr [r13], cl
    inc rbx
    jmp run
1:
    mov byte ptr [r13], 0
    inc rbx
    jmp run
finish:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    push rdx
    push rsi
    syscall
    pop rsi
    pop rdx
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
fail:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
inbuf:
    .zero 65552
code:
    .zero 65552
jmptab:
    .zero 262208
bstack:
    .zero 262208
tape:
    .zero 30016
outbuf:
    .zero 65552
