.globl _start

.equ INCAP, 70000

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INCAP
    sub rdx, r12
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    lea rbx, [rip + inbuf]
    lea r8, [rbx + r12]
    lea rdi, [rip + outbuf]
line_loop:
    cmp rbx, r8
    jae flush
    mov rcx, rbx
scan:
    cmp rcx, r8
    jae found
    cmp byte ptr [rcx], 10
    je found
    inc rcx
    jmp scan
found:
    mov rdx, rcx
rev:
    cmp rdx, rbx
    jbe endrev
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev
endrev:
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rcx + 1]
    jmp line_loop
flush:
    lea rsi, [rip + outbuf]
    mov r13, rdi
write_loop:
    mov rdx, r13
    sub rdx, rsi
    jle finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 70016
outbuf:
    .zero 70016
