.globl _start

.equ BUFSZ, 70000

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
msg_ok:  .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err: .ascii "error at "
msg_err_len = . - msg_err

.bss
.balign 16
buf:    .zero BUFSZ
outbuf: .zero 32

.text
_start:
    lea r13, [rip + buf]
    xor r14d, r14d
read_loop:
    mov edx, BUFSZ - 1
    sub edx, r14d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r14]
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    lea r12, [r13 + r14]
    mov byte ptr [r12], 0
    mov rbx, r13
    call skip_ws
    call value
    call skip_ws
    cmp rbx, r12
    jne err
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

err:
    mov r15, rbx
    sub r15, r13
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, msg_err_len
    syscall
    lea rdi, [rip + outbuf + 31]
    mov byte ptr [rdi], 10
    mov rax, r15
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov rsi, rdi
    lea rdx, [rip + outbuf + 32]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
    ret
1:
    inc rbx
    jmp skip_ws

value:
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je object
    cmp al, '['
    je array
    cmp al, '"'
    je string
    cmp al, 't'
    je lit_true
    cmp al, 'f'
    je lit_false
    cmp al, 'n'
    je lit_null
    cmp al, '-'
    je number
    cmp al, '0'
    jb err
    cmp al, '9'
    ja err
    jmp number

lit_true:
    lea rdi, [rip + s_true]
    jmp match
lit_false:
    lea rdi, [rip + s_false]
    jmp match
lit_null:
    lea rdi, [rip + s_null]
match:
    movzx eax, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [rbx]
    jne err
    inc rbx
    inc rdi
    jmp match
1:
    ret

number:
    movzx eax, byte ptr [rbx]
    cmp al, '-'
    jne 1f
    inc rbx
    movzx eax, byte ptr [rbx]
1:
    cmp al, '0'
    jne 2f
    inc rbx
    jmp 4f
2:
    cmp al, '1'
    jb err
    cmp al, '9'
    ja err
    inc rbx
3:
    movzx eax, byte ptr [rbx]
    sub al, '0'
    cmp al, 9
    ja 4f
    inc rbx
    jmp 3b
4:
    movzx eax, byte ptr [rbx]
    cmp al, '.'
    jne 6f
    inc rbx
    movzx eax, byte ptr [rbx]
    sub al, '0'
    cmp al, 9
    ja err
    inc rbx
5:
    movzx eax, byte ptr [rbx]
    sub al, '0'
    cmp al, 9
    ja 6f
    inc rbx
    jmp 5b
6:
    movzx eax, byte ptr [rbx]
    or al, 0x20
    cmp al, 'e'
    jne 9f
    inc rbx
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je 7f
    cmp al, '-'
    jne 8f
7:
    inc rbx
8:
    movzx eax, byte ptr [rbx]
    sub al, '0'
    cmp al, 9
    ja err
    inc rbx
81:
    movzx eax, byte ptr [rbx]
    sub al, '0'
    cmp al, 9
    ja 9f
    inc rbx
    jmp 81b
9:
    ret

string:
    inc rbx
str_loop:
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je str_end
    cmp al, '\\'
    je str_esc
    cmp al, 0x20
    jb err
    inc rbx
    jmp str_loop
str_end:
    inc rbx
    ret
str_esc:
    inc rbx
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je 1f
    cmp al, '\\'
    je 1f
    cmp al, '/'
    je 1f
    cmp al, 'b'
    je 1f
    cmp al, 'f'
    je 1f
    cmp al, 'n'
    je 1f
    cmp al, 'r'
    je 1f
    cmp al, 't'
    je 1f
    cmp al, 'u'
    jne err
    inc rbx
    mov ecx, 4
2:
    movzx eax, byte ptr [rbx]
    mov edx, eax
    sub dl, '0'
    cmp dl, 9
    jbe 3f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja err
3:
    inc rbx
    dec ecx
    jnz 2b
    jmp str_loop
1:
    inc rbx
    jmp str_loop

object:
    inc rbx
    call skip_ws
    cmp byte ptr [rbx], '}'
    jne 1f
    inc rbx
    ret
1:
    cmp byte ptr [rbx], '"'
    jne err
    call string
    call skip_ws
    cmp byte ptr [rbx], ':'
    jne err
    inc rbx
    call skip_ws
    call value
    call skip_ws
    movzx eax, byte ptr [rbx]
    cmp al, ','
    jne 2f
    inc rbx
    call skip_ws
    jmp 1b
2:
    cmp al, '}'
    jne err
    inc rbx
    ret

array:
    inc rbx
    call skip_ws
    cmp byte ptr [rbx], ']'
    jne 1f
    inc rbx
    ret
1:
    call value
    call skip_ws
    movzx eax, byte ptr [rbx]
    cmp al, ','
    jne 2f
    inc rbx
    call skip_ws
    jmp 1b
2:
    cmp al, ']'
    jne err
    inc rbx
    ret
