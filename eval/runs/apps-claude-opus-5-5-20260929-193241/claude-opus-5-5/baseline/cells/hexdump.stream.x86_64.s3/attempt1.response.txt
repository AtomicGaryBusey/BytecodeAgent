.globl _start

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65600
outbuf:
    .zero 340000

.text

# hex8: eax = value, rdi = dest; advances rdi. clobbers ecx, edx
hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret

_start:
    lea rbx, [rip + hexd]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov edx, 65600
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    lea r14, [rip + inbuf]
    xor r12d, r12d
row_loop:
    cmp r12, r13
    jae rows_done
    mov r8, r13
    sub r8, r12
    cmp r8, 16
    jbe 2f
    mov r8d, 16
2:
    mov eax, r12d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [r14 + r12]
    xor r9d, r9d
byte_loop:
    cmp r9, r8
    jae 3f
    movzx eax, byte ptr [rsi + r9]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp 4f
3:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
4:
    add rdi, 3
    cmp r9, 7
    jne 5f
    mov byte ptr [rdi], 0x20
    inc rdi
5:
    inc r9
    cmp r9, 16
    jb byte_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor r9d, r9d
char_loop:
    movzx eax, byte ptr [rsi + r9]
    cmp al, 0x20
    jb 6f
    cmp al, 0x7e
    jbe 7f
6:
    mov al, 0x2e
7:
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    cmp r9, r8
    jb char_loop
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r12, 16
    jmp row_loop
rows_done:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov r15, rdi
    sub r15, rsi
write_loop:
    test r15, r15
    jz done
    mov eax, 1
    mov edi, 1
    mov rdx, r15
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r15, rax
    jmp write_loop
done:
    mov eax, 60
    xor edi, edi
    syscall
