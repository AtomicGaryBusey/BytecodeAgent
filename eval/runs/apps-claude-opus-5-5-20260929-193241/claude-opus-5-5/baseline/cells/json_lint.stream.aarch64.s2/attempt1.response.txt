.globl _start

.equ BUFSZ, 131072

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
rloop:
    mov x0, #0
    add x1, x19, x21
    mov x2, #BUFSZ
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rdone
    add x21, x21, x0
    b rloop
rdone:
    mov x20, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne fail
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// peek: w0 = byte at pos or 256 at end
peek:
    cmp x20, x21
    b.hs 1f
    ldrb w0, [x19, x20]
    ret
1:  mov w0, #256
    ret

skip_ws:
1:  cmp x20, x21
    b.hs 2f
    ldrb w0, [x19, x20]
    cmp w0, #32
    b.eq 3f
    cmp w0, #9
    b.eq 3f
    cmp w0, #10
    b.eq 3f
    cmp w0, #13
    b.eq 3f
2:  ret
3:  add x20, x20, #1
    b 1b

digits:
1:  cmp x20, x21
    b.hs 2f
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 2f
    add x20, x20, #1
    b 1b
2:  ret

parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl peek
    cmp w0, #'{'
    b.eq pv_obj
    cmp w0, #'['
    b.eq pv_arr
    cmp w0, #'"'
    b.eq pv_str
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    cmp w0, #'-'
    b.eq pv_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls pv_num
    b fail

pv_ret:
    ldp x29, x30, [sp], #16
    ret

pv_true:
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    b pv_lit
pv_false:
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    b pv_lit
pv_null:
    adrp x1, s_null
    add x1, x1, :lo12:s_null
pv_lit:
    ldrb w2, [x1], #1
    cbz w2, pv_ret
    bl peek
    cmp w0, w2
    b.ne fail
    add x20, x20, #1
    b pv_lit

pv_str:
    bl parse_string
    b pv_ret

pv_num:
    bl peek
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    bl peek
1:  cmp w0, #'0'
    b.ne 2f
    add x20, x20, #1
    b 3f
2:  sub w1, w0, #'1'
    cmp w1, #8
    b.hi fail
    add x20, x20, #1
    bl digits
3:  bl peek
    cmp w0, #'.'
    b.ne 4f
    add x20, x20, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
    bl digits
4:  bl peek
    cmp w0, #'e'
    b.eq 5f
    cmp w0, #'E'
    b.ne pv_ret
5:  add x20, x20, #1
    bl peek
    cmp w0, #'+'
    b.eq 6f
    cmp w0, #'-'
    b.ne 7f
6:  add x20, x20, #1
7:  bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
    bl digits
    b pv_ret

pv_arr:
    add x20, x20, #1
    bl skip_ws
    bl peek
    cmp w0, #']'
    b.eq 2f
1:  bl parse_value
    bl skip_ws
    bl peek
    cmp w0, #','
    b.ne 3f
    add x20, x20, #1
    bl skip_ws
    b 1b
3:  cmp w0, #']'
    b.ne fail
2:  add x20, x20, #1
    b pv_ret

pv_obj:
    add x20, x20, #1
    bl skip_ws
    bl peek
    cmp w0, #'}'
    b.eq 2f
1:  bl peek
    cmp w0, #'"'
    b.ne fail
    bl parse_string
    bl skip_ws
    bl peek
    cmp w0, #':'
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    bl peek
    cmp w0, #','
    b.ne 3f
    add x20, x20, #1
    bl skip_ws
    b 1b
3:  cmp w0, #'}'
    b.ne fail
2:  add x20, x20, #1
    b pv_ret

parse_string:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    add x20, x20, #1
ps_loop:
    bl peek
    cmp w0, #256
    b.eq fail
    cmp w0, #'"'
    b.eq ps_end
    cmp w0, #'\\'
    b.eq ps_esc
    cmp w0, #0x20
    b.lo fail
    add x20, x20, #1
    b ps_loop
ps_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret
ps_esc:
    add x20, x20, #1
    bl peek
    cmp w0, #'"'
    b.eq ps_ok
    cmp w0, #'\\'
    b.eq ps_ok
    cmp w0, #'/'
    b.eq ps_ok
    cmp w0, #'b'
    b.eq ps_ok
    cmp w0, #'f'
    b.eq ps_ok
    cmp w0, #'n'
    b.eq ps_ok
    cmp w0, #'r'
    b.eq ps_ok
    cmp w0, #'t'
    b.eq ps_ok
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    mov w9, #4
1:  bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 2f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi fail
2:  add x20, x20, #1
    subs w9, w9, #1
    b.ne 1b
    b ps_loop
ps_ok:
    add x20, x20, #1
    b ps_loop

fail:
    mov x0, #2
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x4, numbuf
    add x4, x4, :lo12:numbuf
    add x4, x4, #32
    mov x5, x4
    mov w3, #10
    strb w3, [x4, #-1]!
    mov x0, x20
    mov x6, #10
1:  udiv x2, x0, x6
    msub x3, x2, x6, x0
    add w3, w3, #'0'
    strb w3, [x4, #-1]!
    mov x0, x2
    cbnz x0, 1b
    mov x0, #1
    mov x1, x4
    sub x2, x5, x4
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
s_ok:
    .ascii "ok\n"
s_err:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"

.bss
.balign 16
buf:
    .zero BUFSZ
numbuf:
    .zero 32
