.globl _start
.text
_start:
    mov x19, #0            // lines
    mov x20, #0            // words
    mov x21, #0            // bytes
    mov x22, #0            // in-word flag
read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    adrp x1, buf
    add x1, x1, :lo12:buf
    add x3, x1, x0
byte_loop:
    cmp x1, x3
    b.hs read_loop
    ldrb w4, [x1], #1
    cmp w4, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w4, #32
    b.eq ws
    sub w5, w4, #9
    cmp w5, #4
    b.ls ws
    cbnz x22, byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop
ws:
    mov x22, #0
    b byte_loop
done:
    adrp x10, outbuf
    add x10, x10, :lo12:outbuf
    mov x0, x19
    bl putnum
    mov w4, #32
    strb w4, [x10], #1
    mov x0, x20
    bl putnum
    mov w4, #32
    strb w4, [x10], #1
    mov x0, x21
    bl putnum
    mov w4, #10
    strb w4, [x10], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x10, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// putnum: x0 = value, x10 = output pointer (advanced)
putnum:
    adrp x11, tmp
    add x11, x11, :lo12:tmp
    add x11, x11, #32
    mov x12, x11
    mov x13, #10
2:
    udiv x14, x0, x13
    msub x15, x14, x13, x0
    add w15, w15, #48
    strb w15, [x12, #-1]!
    mov x0, x14
    cbnz x0, 2b
3:
    cmp x12, x11
    b.hs 4f
    ldrb w15, [x12], #1
    strb w15, [x10], #1
    b 3b
4:
    ret

.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 128
tmp:
    .zero 64
