.globl _start

BUFSZ = 70000

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
.balign 16
buf:
    .zero BUFSZ
numbuf:
    .zero 32

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, BUFSZ - 1
    sub rdx, r12
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    lea rbx, [rip + buf]
    mov byte ptr [rbx + r12], 0
    lea r13, [rbx + r12]
    call skip_ws
    call value
    call skip_ws
    cmp rbx, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rax, [rip + buf]
    mov r8, rbx
    sub r8, rax
    mov rax, r8
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    dec rdi
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    je 1f
    ret
1:
    inc rbx
    jmp skip_ws

value:
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je object
    cmp al, '['
    je array
    cmp al, '"'
    je string
    cmp al, 't'
    je v_true
    cmp al, 'f'
    je v_false
    cmp al, 'n'
    je v_null
    cmp al, '-'
    je number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp number

v_true:
    lea rdi, [rip + lit_true]
    jmp literal
v_false:
    lea rdi, [rip + lit_false]
    jmp literal
v_null:
    lea rdi, [rip + lit_null]
literal:
    mov al, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [rbx]
    jne error
    inc rbx
    inc rdi
    jmp literal
1:
    ret

string:
    inc rbx
s_loop:
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je s_end
    cmp al, 0x20
    jb error
    cmp al, 0x5C
    je s_esc
    inc rbx
    jmp s_loop
s_end:
    inc rbx
    ret
s_esc:
    inc rbx
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je s_esc_ok
    cmp al, 0x5C
    je s_esc_ok
    cmp al, '/'
    je s_esc_ok
    cmp al, 'b'
    je s_esc_ok
    cmp al, 'f'
    je s_esc_ok
    cmp al, 'n'
    je s_esc_ok
    cmp al, 'r'
    je s_esc_ok
    cmp al, 't'
    je s_esc_ok
    cmp al, 'u'
    jne error
    inc rbx
    mov ecx, 4
s_hex:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb 2f
    cmp al, '9'
    jbe 3f
2:
    or al, 0x20
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
3:
    inc rbx
    dec ecx
    jnz s_hex
    jmp s_loop
s_esc_ok:
    inc rbx
    jmp s_loop

number:
    movzx eax, byte ptr [rbx]
    cmp al, '-'
    jne 1f
    inc rbx
    movzx eax, byte ptr [rbx]
1:
    cmp al, '0'
    je n_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc rbx
    call digits
    jmp n_frac
n_zero:
    inc rbx
n_frac:
    movzx eax, byte ptr [rbx]
    cmp al, '.'
    jne n_exp
    inc rbx
    call req_digit
    call digits
n_exp:
    movzx eax, byte ptr [rbx]
    cmp al, 'e'
    je 1f
    cmp al, 'E'
    jne 2f
1:
    inc rbx
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je 3f
    cmp al, '-'
    jne 4f
3:
    inc rbx
4:
    call req_digit
    call digits
2:
    ret

req_digit:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc rbx
    ret

digits:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb 1f
    cmp al, '9'
    ja 1f
    inc rbx
    jmp digits
1:
    ret

array:
    inc rbx
    call skip_ws
    cmp byte ptr [rbx], ']'
    jne a_loop
    inc rbx
    ret
a_loop:
    call value
    call skip_ws
    movzx eax, byte ptr [rbx]
    cmp al, ','
    jne 1f
    inc rbx
    call skip_ws
    jmp a_loop
1:
    cmp al, ']'
    jne error
    inc rbx
    ret

object:
    inc rbx
    call skip_ws
    cmp byte ptr [rbx], '}'
    jne o_loop
    inc rbx
    ret
o_loop:
    cmp byte ptr [rbx], '"'
    jne error
    call string
    call skip_ws
    cmp byte ptr [rbx], ':'
    jne error
    inc rbx
    call skip_ws
    call value
    call skip_ws
    movzx eax, byte ptr [rbx]
    cmp al, ','
    jne 1f
    inc rbx
    call skip_ws
    jmp o_loop
1:
    cmp al, '}'
    jne error
    inc rbx
    ret
