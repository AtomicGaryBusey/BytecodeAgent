.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 64
inbuf:
    .zero 65536
outbuf:
    .zero 90000

.text
_start:
    xor r12d, r12d
readloop:
    mov edx, 65536
    sub edx, r12d
    jz done_read
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp readloop
done_read:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alpha]
    xor r8d, r8d
    mov rcx, r12
full:
    cmp rcx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit
    mov edx, eax
    and edx, 63
    call emit
    add rsi, 3
    sub rcx, 3
    jmp full
tail:
    test rcx, rcx
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 1
    je tail1
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit
    mov edx, 61
    call emitchar
    jmp finish
tail1:
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov edx, 61
    call emitchar
    mov edx, 61
    call emitchar
finish:
    test r8, r8
    jz 2f
    mov byte ptr [rdi], 10
    inc rdi
2:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
writeloop:
    test rdx, rdx
    jz doexit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle doexit
    add rsi, rax
    sub rdx, rax
    jmp writeloop
doexit:
    mov eax, 60
    xor edi, edi
    syscall

emit:
    movzx edx, byte ptr [rbx + rdx]
emitchar:
    mov byte ptr [rdi], dl
    inc rdi
    inc r8
    cmp r8, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:
    ret
