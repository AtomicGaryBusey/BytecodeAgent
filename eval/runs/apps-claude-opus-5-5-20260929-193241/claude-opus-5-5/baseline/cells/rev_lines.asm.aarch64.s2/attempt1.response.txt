.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x20, #0
    mov x22, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #0x20000
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x3, #0
    mov x4, #0
scan_loop:
    cmp x4, x20
    b.hs scan_end
    ldrb w5, [x19, x4]
    cmp w5, #10
    b.ne scan_next
    bl emit
    add x3, x4, #1
scan_next:
    add x4, x4, #1
    b scan_loop

scan_end:
    cmp x3, x20
    b.hs write_out
    mov x4, x20
    bl emit

write_out:
    mov x23, #0
write_loop:
    cmp x23, x22
    b.hs finish
    mov x0, #1
    add x1, x21, x23
    sub x2, x22, x23
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x23, x23, x0
    b write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

// emit bytes [x3, x4) reversed, then newline
emit:
    mov x6, x4
1:
    cmp x6, x3
    b.ls 2f
    sub x6, x6, #1
    ldrb w7, [x19, x6]
    strb w7, [x21, x22]
    add x22, x22, #1
    b 1b
2:
    mov w7, #10
    strb w7, [x21, x22]
    add x22, x22, #1
    ret

.bss
.balign 16
inbuf:
    .zero 0x20000
outbuf:
    .zero 0x20000
