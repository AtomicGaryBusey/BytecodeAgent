.globl _start

.equ INCAP, 65552

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65552
.balign 16
outbuf:
    .zero 340000

.text

# put_hex8: eax = value, rdi = output pointer (advanced by 8)
put_hex8:
    lea r8, [rip + hexdig]
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret

_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INCAP
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    lea r14, [rip + inbuf]
    lea r15, [rip + hexdig]
    xor ebx, ebx
row_loop:
    cmp rbx, r12
    jae rows_done
    mov r13, r12
    sub r13, rbx
    cmp r13, 16
    jbe 2f
    mov r13d, 16
2:
    mov eax, ebx
    call put_hex8
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor esi, esi
hex_loop:
    cmp rsi, r13
    jae 3f
    lea rax, [rbx + rsi]
    movzx eax, byte ptr [r14 + rax]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    jmp 4f
3:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
4:
    cmp esi, 7
    jne 5f
    mov byte ptr [rdi], ' '
    inc rdi
5:
    inc esi
    cmp esi, 16
    jb hex_loop
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor esi, esi
chr_loop:
    lea rax, [rbx + rsi]
    movzx eax, byte ptr [r14 + rax]
    cmp eax, 0x20
    jb 6f
    cmp eax, 0x7e
    jbe 7f
6:
    mov eax, '.'
7:
    mov byte ptr [rdi], al
    inc rdi
    inc rsi
    cmp rsi, r13
    jb chr_loop
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add rbx, 16
    jmp row_loop
rows_done:
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
write_loop:
    mov rdx, rbx
    sub rdx, rsi
    jle finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
