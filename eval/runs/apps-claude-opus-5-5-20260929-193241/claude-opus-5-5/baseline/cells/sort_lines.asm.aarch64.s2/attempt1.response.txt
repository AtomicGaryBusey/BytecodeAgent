.globl _start

.equ INCAP, 32776
.equ MAXL, 1024

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x20, starts
    add x20, x20, :lo12:starts
    adrp x21, lens
    add x21, x21, :lo12:lens
    mov x10, #0                 // total
rd_loop:
    mov x0, #0
    add x1, x19, x10
    mov x2, #INCAP
    sub x2, x2, x10
    cbz x2, rd_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x10, x10, x0
    b rd_loop
rd_done:
    // split lines
    mov x11, #0                 // pos
    mov x22, #0                 // n
split_loop:
    cmp x11, x10
    b.ge split_done
    mov x12, x11                // start
scan:
    cmp x11, x10
    b.ge scan_end
    ldrb w13, [x19, x11]
    cmp w13, #10
    b.eq scan_end
    add x11, x11, #1
    b scan
scan_end:
    sub x13, x11, x12
    str w12, [x20, x22, lsl #2]
    str w13, [x21, x22, lsl #2]
    add x22, x22, #1
    cmp x22, #MAXL
    b.ge split_done
    cmp x11, x10
    b.ge split_done
    add x11, x11, #1
    b split_loop
split_done:
    cbz x22, finish
    adrp x23, arrA
    add x23, x23, :lo12:arrA
    adrp x24, arrB
    add x24, x24, :lo12:arrB
    mov x0, #0
init_loop:
    cmp x0, x22
    b.ge init_done
    str w0, [x23, x0, lsl #2]
    add x0, x0, #1
    b init_loop
init_done:
    mov x25, #1                 // width
pass_loop:
    cmp x25, x22
    b.ge sort_done
    mov x26, #0                 // lo
lo_loop:
    cmp x26, x22
    b.ge pass_done
    add x13, x26, x25
    cmp x13, x22
    csel x13, x13, x22, lt      // mid
    add x14, x26, x25, lsl #1
    cmp x14, x22
    csel x14, x14, x22, lt      // hi
    mov x27, x26                // i
    mov x28, x13                // j
    mov x12, x26                // k
m_loop:
    cmp x12, x14
    b.ge m_done
    cmp x27, x13
    b.ge take_j
    cmp x28, x14
    b.ge take_i
    ldr w0, [x23, x27, lsl #2]
    ldr w1, [x23, x28, lsl #2]
    bl cmpl
    cmp w0, #0
    b.le take_i
take_j:
    ldr w0, [x23, x28, lsl #2]
    str w0, [x24, x12, lsl #2]
    add x28, x28, #1
    add x12, x12, #1
    b m_loop
take_i:
    ldr w0, [x23, x27, lsl #2]
    str w0, [x24, x12, lsl #2]
    add x27, x27, #1
    add x12, x12, #1
    b m_loop
m_done:
    add x26, x26, x25, lsl #1
    b lo_loop
pass_done:
    mov x0, x23
    mov x23, x24
    mov x24, x0
    lsl x25, x25, #1
    b pass_loop
sort_done:
    adrp x15, outbuf
    add x15, x15, :lo12:outbuf
    mov x16, x15                // out ptr
    mov x12, #0
out_loop:
    cmp x12, x22
    b.ge out_done
    ldr w0, [x23, x12, lsl #2]
    ldr w2, [x20, x0, lsl #2]
    ldr w3, [x21, x0, lsl #2]
    add x2, x19, x2
cp_loop:
    cbz w3, cp_done
    ldrb w4, [x2], #1
    strb w4, [x16], #1
    sub w3, w3, #1
    b cp_loop
cp_done:
    mov w4, #10
    strb w4, [x16], #1
    add x12, x12, #1
    b out_loop
out_done:
    mov x17, x15
wr_loop:
    cmp x17, x16
    b.ge finish
    mov x0, #1
    mov x1, x17
    sub x2, x16, x17
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x17, x17, x0
    b wr_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish

// cmpl: w0=i, w1=j -> w0 <0,0,>0 ; clobbers x0-x9
cmpl:
    ldr w2, [x20, w0, uxtw #2]
    ldr w3, [x21, w0, uxtw #2]
    ldr w4, [x20, w1, uxtw #2]
    ldr w5, [x21, w1, uxtw #2]
    cmp w3, w5
    csel w6, w3, w5, lo
    add x2, x19, x2
    add x4, x19, x4
c_loop:
    cbz w6, c_len
    ldrb w7, [x2], #1
    ldrb w9, [x4], #1
    subs w7, w7, w9
    b.ne c_diff
    sub w6, w6, #1
    b c_loop
c_diff:
    mov w0, w7
    ret
c_len:
    sub w0, w3, w5
    ret

.bss
.balign 16
inbuf:  .zero 32784
outbuf: .zero 33808
starts: .zero 4096
lens:   .zero 4096
arrA:   .zero 4096
arrB:   .zero 4096
