.globl _start

    .section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .asciz "rue"
lit_false:
    .asciz "alse"
lit_null:
    .asciz "ull"
esc_chars:
    .asciz "\"\\/bfnrt"

    .bss
    .balign 16
buf:
    .zero 65600
numbuf:
    .zero 32

    .text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
read_loop:
    movz x2, #0x20
    movk x2, #1, lsl #16
    subs x2, x2, x21
    b.ls read_done
    mov x0, #0
    add x1, x19, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop
read_done:
    strb wzr, [x19, x21]
    mov x20, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne error
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// ---------------- skip whitespace (leaf)
skip_ws:
1:
    ldrb w0, [x19, x20]
    cmp w0, #32
    b.eq 2f
    cmp w0, #9
    b.eq 2f
    cmp w0, #10
    b.eq 2f
    cmp w0, #13
    b.eq 2f
    ret
2:
    add x20, x20, #1
    b 1b

// ---------------- parse_value
parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    ldrb w0, [x19, x20]
    cmp w0, #'{'
    b.eq pv_object
    cmp w0, #'['
    b.eq pv_array
    cmp w0, #'"'
    b.eq pv_string
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    cmp w0, #'-'
    b.eq pv_number
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls pv_number
    b error

pv_ret:
    ldp x29, x30, [sp], #16
    ret

pv_string:
    bl parse_string
    b pv_ret

pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    b pv_lit
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    b pv_lit
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
pv_lit:
    add x20, x20, #1
1:
    ldrb w2, [x1], #1
    cbz w2, pv_ret
    ldrb w0, [x19, x20]
    cmp w0, w2
    b.ne error
    add x20, x20, #1
    b 1b

pv_number:
    ldrb w0, [x19, x20]
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    ldrb w0, [x19, x20]
1:
    cmp w0, #'0'
    b.ne 2f
    add x20, x20, #1
    b 4f
2:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
    add x20, x20, #1
3:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 4f
    add x20, x20, #1
    b 3b
4:
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    b.ne 6f
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    add x20, x20, #1
5:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 6f
    add x20, x20, #1
    b 5b
6:
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.eq 7f
    cmp w0, #'E'
    b.ne pv_ret
7:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    b.eq 8f
    cmp w0, #'-'
    b.ne 9f
8:
    add x20, x20, #1
    ldrb w0, [x19, x20]
9:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    add x20, x20, #1
10:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi pv_ret
    add x20, x20, #1
    b 10b

pv_array:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.ne 1f
    add x20, x20, #1
    b pv_ret
1:
    bl parse_value
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #','
    b.ne 2f
    add x20, x20, #1
    bl skip_ws
    b 1b
2:
    cmp w0, #']'
    b.ne error
    add x20, x20, #1
    b pv_ret

pv_object:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.ne 1f
    add x20, x20, #1
    b pv_ret
1:
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.ne error
    bl parse_string
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #':'
    b.ne error
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #','
    b.ne 2f
    add x20, x20, #1
    bl skip_ws
    b 1b
2:
    cmp w0, #'}'
    b.ne error
    add x20, x20, #1
    b pv_ret

// ---------------- parse_string (leaf), at opening quote
parse_string:
    add x20, x20, #1
ps_loop:
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq ps_end
    cmp w0, #'\\'
    b.eq ps_esc
    cmp w0, #0x20
    b.lo error
    add x20, x20, #1
    b ps_loop
ps_end:
    add x20, x20, #1
    ret
ps_esc:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    b.eq ps_u
    adrp x1, esc_chars
    add x1, x1, :lo12:esc_chars
1:
    ldrb w2, [x1], #1
    cbz w2, error
    cmp w0, w2
    b.ne 1b
    add x20, x20, #1
    b ps_loop
ps_u:
    add x20, x20, #1
    mov x3, #4
1:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 2f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
2:
    add x20, x20, #1
    subs x3, x3, #1
    b.ne 1b
    b ps_loop

// ---------------- error: print "error at <x20>\n", exit 1
error:
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x3, numbuf
    add x3, x3, :lo12:numbuf
    add x3, x3, #31
    mov x5, x3
    mov w4, #10
    strb w4, [x3]
    mov x10, #10
1:
    udiv x6, x20, x10
    msub x7, x6, x10, x20
    add w7, w7, #'0'
    strb w7, [x3, #-1]!
    mov x20, x6
    cbnz x20, 1b
    mov x1, x3
    add x2, x5, #1
    sub x2, x2, x3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
