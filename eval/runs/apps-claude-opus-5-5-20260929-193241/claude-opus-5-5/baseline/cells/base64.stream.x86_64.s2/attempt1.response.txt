.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 64
inbuf:
    .zero 65536
outbuf:
    .zero 70000

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov r10, r12
    lea rdi, [rip + outbuf]
    xor r8d, r8d
    lea rbx, [rip + alpha]
grp:
    cmp r10, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit_idx
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit_idx
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit_idx
    mov edx, eax
    and edx, 63
    call emit_idx
    add rsi, 3
    sub r10, 3
    jmp grp
tail:
    test r10, r10
    jz fin
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r10, 1
    je tail1
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit_idx
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit_idx
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit_idx
    mov edx, 61
    call emit_chr
    jmp fin
tail1:
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit_idx
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit_idx
    mov edx, 61
    call emit_chr
    mov edx, 61
    call emit_chr
fin:
    test r8, r8
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit_idx:
    movzx edx, byte ptr [rbx + rdx]
emit_chr:
    mov byte ptr [rdi], dl
    inc rdi
    inc r8
    cmp r8, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:
    ret
