.globl _start

.equ INCAP, 65600
.equ OUTCAP, 65536

.section .rodata
errmsg:
    .ascii "error\n"

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, estack
    add x22, x22, :lo12:estack
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x25, #0
    mov x20, x19
    movz x28, #(INCAP & 0xffff)
    movk x28, #(INCAP >> 16), lsl #16
    add x28, x19, x28
read_loop:
    sub x2, x28, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, x19

line_loop:
    cmp x21, x20
    b.hs finish
    mov x26, x21
find_nl:
    cmp x26, x20
    b.hs found_nl
    ldrb w0, [x26]
    cmp w0, #10
    b.eq found_nl
    add x26, x26, #1
    b find_nl
found_nl:
    mov x27, x26
    cmp x26, x20
    b.hs 1f
    add x27, x26, #1
1:
    mov x23, #0
    mov x9, x21

tok_loop:
    cmp x9, x26
    b.hs line_end
    ldrb w0, [x9]
    cmp w0, #32
    b.eq skip_sep
    cmp w0, #9
    b.eq skip_sep
    mov x10, x9
scan:
    add x9, x9, #1
    cmp x9, x26
    b.hs tokend
    ldrb w0, [x9]
    cmp w0, #32
    b.eq tokend
    cmp w0, #9
    b.eq tokend
    b scan
tokend:
    sub x11, x9, x10
    ldrb w0, [x10]
    cmp x11, #1
    b.ne number
    mov x5, #0
    cmp w0, #'+'
    b.eq do_op
    mov x5, #1
    cmp w0, #'-'
    b.eq do_op
    mov x5, #2
    cmp w0, #'*'
    b.eq do_op
    mov x5, #3
    cmp w0, #'/'
    b.eq do_op
    mov x5, #4
    cmp w0, #'%'
    b.eq do_op
number:
    mov x12, x10
    mov x14, #0
    cmp w0, #'-'
    b.ne 1f
    add x12, x12, #1
    mov x14, #1
1:
    sub x13, x9, x12
    cbz x13, line_error
    cmp x13, #18
    b.hi line_error
    mov x15, #0
    mov x16, #10
digit_loop:
    ldrb w0, [x12], #1
    sub w0, w0, #'0'
    cmp w0, #9
    b.hi line_error
    madd x15, x15, x16, x0
    cmp x12, x9
    b.lo digit_loop
    cbz x14, 2f
    neg x15, x15
2:
    str x15, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

skip_sep:
    add x9, x9, #1
    b tok_loop

do_op:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #2
    add x2, x22, x23, lsl #3
    ldp x3, x4, [x2]
    cmp x5, #0
    b.ne 1f
    add x3, x3, x4
    b op_store
1:
    cmp x5, #1
    b.ne 1f
    sub x3, x3, x4
    b op_store
1:
    cmp x5, #2
    b.ne 1f
    mul x3, x3, x4
    b op_store
1:
    cbz x4, line_error
    cmn x4, #1
    b.ne 2f
    mov x6, #1
    lsl x6, x6, #63
    cmp x3, x6
    b.eq line_error
2:
    sdiv x6, x3, x4
    cmp x5, #3
    b.ne 3f
    mov x3, x6
    b op_store
3:
    msub x3, x6, x4, x3
op_store:
    str x3, [x2]
    add x23, x23, #1
    b tok_loop

line_end:
    cmp x23, #1
    b.ne line_error
    ldr x0, [x22]
    cmp x0, #0
    b.ge 1f
    mov w3, #'-'
    strb w3, [x24, x25]
    add x25, x25, #1
    neg x0, x0
1:
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #32
    mov x7, x1
    mov x10, #10
2:
    udiv x2, x0, x10
    msub x3, x2, x10, x0
    add w3, w3, #'0'
    strb w3, [x1, #-1]!
    mov x0, x2
    cbnz x0, 2b
3:
    ldrb w3, [x1], #1
    strb w3, [x24, x25]
    add x25, x25, #1
    cmp x1, x7
    b.lo 3b
    mov w3, #10
    strb w3, [x24, x25]
    add x25, x25, #1
    b next_line

line_error:
    adr x1, errmsg
    mov x2, #6
1:
    ldrb w3, [x1], #1
    strb w3, [x24, x25]
    add x25, x25, #1
    subs x2, x2, #1
    b.ne 1b

next_line:
    mov x21, x27
    mov x0, #(OUTCAP - 64)
    cmp x25, x0
    b.lo line_loop
    bl flush
    b line_loop

finish:
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0

flush:
    cbz x25, 1f
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    mov x25, #0
1:
    ret

.bss
.balign 16
inbuf:
    .zero INCAP
.balign 16
estack:
    .zero 8192
.balign 16
outbuf:
    .zero OUTCAP
.balign 16
numbuf:
    .zero 32
