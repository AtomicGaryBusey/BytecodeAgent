.globl _start

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .zero 70000
.balign 16
outbuf:
    .zero 65600
.balign 16
vstack:
    .zero 16384
numtmp:
    .zero 32

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 70000
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov rsi, rbx
    lea r13, [rbx + r12]
    lea r14, [rip + outbuf]
    lea rbp, [rip + vstack]

line_loop:
    cmp rsi, r13
    jae finish
    xor r15d, r15d
tok_loop:
    cmp rsi, r13
    jae eol
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je eol
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    mov rdi, rsi
1:
    inc rsi
    cmp rsi, r13
    jae 2f
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je 2f
    cmp al, 32
    je 2f
    cmp al, 9
    je 2f
    jmp 1b
2:
    mov rcx, rsi
    sub rcx, rdi
    movzx eax, byte ptr [rdi]
    cmp rcx, 1
    jne literal
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
literal:
    xor r8d, r8d
    cmp al, '-'
    jne 3f
    inc rdi
    dec rcx
    mov r8d, 1
3:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor edx, edx
4:
    movzx eax, byte ptr [rdi]
    sub eax, '0'
    cmp eax, 9
    ja line_error
    imul rdx, rdx, 10
    add rdx, rax
    inc rdi
    dec rcx
    jnz 4b
    test r8d, r8d
    jz 5f
    neg rdx
5:
    mov qword ptr [rbp + r15*8], rdx
    inc r15
    jmp tok_loop

skip_sep:
    inc rsi
    jmp tok_loop

op_load:
    # not used
    ret

op_add:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbp + r15*8 - 16]
    mov rcx, qword ptr [rbp + r15*8 - 8]
    dec r15
    add rax, rcx
    mov qword ptr [rbp + r15*8 - 8], rax
    jmp tok_loop
op_sub:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbp + r15*8 - 16]
    mov rcx, qword ptr [rbp + r15*8 - 8]
    dec r15
    sub rax, rcx
    mov qword ptr [rbp + r15*8 - 8], rax
    jmp tok_loop
op_mul:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbp + r15*8 - 16]
    mov rcx, qword ptr [rbp + r15*8 - 8]
    dec r15
    imul rax, rcx
    mov qword ptr [rbp + r15*8 - 8], rax
    jmp tok_loop
op_div:
    xor r9d, r9d
    jmp op_divmod
op_mod:
    mov r9d, 1
op_divmod:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbp + r15*8 - 16]
    mov rcx, qword ptr [rbp + r15*8 - 8]
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne 6f
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
6:
    dec r15
    cqo
    idiv rcx
    test r9d, r9d
    jz 7f
    mov rax, rdx
7:
    mov qword ptr [rbp + r15*8 - 8], rax
    jmp tok_loop

line_error:
    cmp rsi, r13
    jae 8f
    cmp byte ptr [rsi], 10
    je 8f
    inc rsi
    jmp line_error
8:
    lea rdx, [rip + errmsg]
    mov eax, dword ptr [rdx]
    mov dword ptr [r14], eax
    movzx eax, word ptr [rdx + 4]
    mov word ptr [r14 + 4], ax
    add r14, 6
    jmp after_line

eol:
    cmp r15, 1
    jne line_error
    mov rax, qword ptr [rbp]
    test rax, rax
    jns 9f
    mov byte ptr [r14], '-'
    inc r14
    neg rax
9:
    lea rdi, [rip + numtmp + 32]
    mov ecx, 10
10:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 10b
    lea rdx, [rip + numtmp + 32]
11:
    mov al, byte ptr [rdi]
    mov byte ptr [r14], al
    inc r14
    inc rdi
    cmp rdi, rdx
    jb 11b
    mov byte ptr [r14], 10
    inc r14

after_line:
    lea rax, [rip + outbuf]
    mov rdx, r14
    sub rdx, rax
    cmp rdx, 65400
    jb 12f
    call flush
12:
    cmp rsi, r13
    jae line_loop
    inc rsi
    jmp line_loop

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    push rsi
    lea rsi, [rip + outbuf]
13:
    mov rdx, r14
    sub rdx, rsi
    jle 14f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 14f
    add rsi, rax
    jmp 13b
14:
    lea r14, [rip + outbuf]
    pop rsi
    ret
