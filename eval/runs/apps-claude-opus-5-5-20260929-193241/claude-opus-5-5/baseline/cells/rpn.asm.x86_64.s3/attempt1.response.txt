.globl _start

.equ INSZ, 70000
.equ OUTSZ, 65536

.section .rodata
dummy: .byte 0

.bss
.balign 16
inbuf:  .zero 70016
outbuf: .zero 65536
stk:    .zero 8192
tmp:    .zero 64

.text
_start:
    xor r12d, r12d
.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INSZ
    sub rdx, r12
    jle .Lreaddone
    syscall
    test rax, rax
    jle .Lreaddone
    add r12, rax
    jmp .Lread
.Lreaddone:
    lea rsi, [rip + inbuf]
    lea r12, [rsi + r12]
    lea rbx, [rip + stk]
    lea r14, [rip + outbuf]

.Lline:
    cmp rsi, r12
    jae .Lfinish
    xor r13d, r13d

.Ltok:
    cmp rsi, r12
    jae .Leol
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je .Lsep
    cmp al, 9
    jne .Lnotsep
.Lsep:
    inc rsi
    jmp .Ltok
.Lnotsep:
    cmp al, 10
    jne .Ltoken
    inc rsi
    jmp .Leol

.Ltoken:
    mov rdi, rsi
.Ltl:
    cmp rdi, r12
    jae .Ltend
    movzx ecx, byte ptr [rdi]
    cmp cl, 32
    je .Ltend
    cmp cl, 9
    je .Ltend
    cmp cl, 10
    je .Ltend
    inc rdi
    jmp .Ltl
.Ltend:
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, 1
    jne .Llit
    cmp al, '+'
    je .Lop
    cmp al, '-'
    je .Lop
    cmp al, '*'
    je .Lop
    cmp al, '/'
    je .Lop
    cmp al, '%'
    je .Lop

.Llit:
    xor edx, edx
    mov r8, rsi
    cmp al, '-'
    jne .Lnd
    mov edx, 1
    inc r8
.Lnd:
    mov r9, rdi
    sub r9, r8
    jz .Lerr
    cmp r9, 18
    ja .Lerr
    xor eax, eax
.Ldig:
    movzx ecx, byte ptr [r8]
    sub ecx, '0'
    cmp ecx, 9
    ja .Lerr
    imul rax, rax, 10
    add rax, rcx
    inc r8
    cmp r8, rdi
    jb .Ldig
    test edx, edx
    jz .Lpush
    neg rax
.Lpush:
    mov qword ptr [rbx + r13*8], rax
    inc r13
    mov rsi, rdi
    jmp .Ltok

.Lop:
    cmp r13, 2
    jb .Lerr
    mov r10d, eax
    mov rcx, qword ptr [rbx + r13*8 - 8]
    mov rax, qword ptr [rbx + r13*8 - 16]
    dec r13
    cmp r10b, '+'
    je .Ladd
    cmp r10b, '-'
    je .Lsub
    cmp r10b, '*'
    je .Lmul
    test rcx, rcx
    jz .Lerr
    cmp rcx, -1
    jne .Ldivok
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    je .Lerr
.Ldivok:
    cqo
    idiv rcx
    cmp r10b, '%'
    jne .Lstore
    mov rax, rdx
    jmp .Lstore
.Ladd:
    add rax, rcx
    jmp .Lstore
.Lsub:
    sub rax, rcx
    jmp .Lstore
.Lmul:
    imul rax, rcx
.Lstore:
    mov qword ptr [rbx + r13*8 - 8], rax
    mov rsi, rdi
    jmp .Ltok

.Lerr:
    cmp rsi, r12
    jae .Lerrout
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne .Lerr
.Lerrout:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp .Lcheck

.Leol:
    cmp r13, 1
    jne .Lerrout
    mov rax, qword ptr [rbx]
    test rax, rax
    jns .Lpos
    mov byte ptr [r14], '-'
    inc r14
    neg rax
.Lpos:
    lea r8, [rip + tmp + 32]
    mov r9, r8
    mov ecx, 10
.Lconv:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz .Lconv
.Lcopy:
    mov al, byte ptr [r8]
    mov byte ptr [r14], al
    inc r14
    inc r8
    cmp r8, r9
    jb .Lcopy
    mov byte ptr [r14], 10
    inc r14

.Lcheck:
    lea rax, [rip + outbuf + OUTSZ - 64]
    cmp r14, rax
    jb .Lline
    call flush
    jmp .Lline

.Lfinish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    push rsi
    lea rsi, [rip + outbuf]
.Lfl:
    mov rdx, r14
    sub rdx, rsi
    jle .Lfldone
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lfldone
    add rsi, rax
    jmp .Lfl
.Lfldone:
    lea r14, [rip + outbuf]
    pop rsi
    ret
