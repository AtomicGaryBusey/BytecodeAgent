.globl _start

.equ INSZ, 65600
.equ OUTSZ, 400000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INSZ
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r13, [r12 + r13]
    lea rbp, [rip + outbuf]
    lea rbx, [rip + stk]

line_loop:
    cmp r12, r13
    jae finish
    mov r15, r12
find_eol:
    cmp r15, r13
    jae found_eol
    cmp byte ptr [r15], 10
    je found_eol
    inc r15
    jmp find_eol
found_eol:
    xor r14d, r14d

tok:
    cmp r12, r15
    jae eol
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip
    cmp al, 9
    je skip
    mov rsi, r12
    mov rdi, r12
scan:
    cmp rdi, r15
    jae scan_end
    movzx eax, byte ptr [rdi]
    cmp al, 32
    je scan_end
    cmp al, 9
    je scan_end
    inc rdi
    jmp scan
skip:
    inc r12
    jmp tok

scan_end:
    mov rcx, rdi
    sub rcx, rsi
    mov r12, rdi
    cmp rcx, 1
    jne number
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je op
    cmp al, '-'
    je op
    cmp al, '*'
    je op
    cmp al, '/'
    je op
    cmp al, '%'
    je op

number:
    xor r9d, r9d
    cmp byte ptr [rsi], '-'
    jne 1f
    mov r9d, 1
    inc rsi
    dec rcx
1:
    test rcx, rcx
    jz err
    cmp rcx, 18
    ja err
    xor eax, eax
2:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja err
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz 2b
    test r9d, r9d
    jz 3f
    neg rax
3:
    cmp r14, 2000
    jae err
    mov qword ptr [rbx + r14*8], rax
    inc r14
    jmp tok

op:
    cmp r14, 2
    jb err
    mov r8, qword ptr [rbx + r14*8 - 8]
    mov rdx, rax
    mov rax, qword ptr [rbx + r14*8 - 16]
    dec r14
    cmp dl, '+'
    je op_add
    cmp dl, '-'
    je op_sub
    cmp dl, '*'
    je op_mul
    test r8, r8
    jz err
    cmp r8, -1
    jne 4f
    mov r10, 0x8000000000000000
    cmp rax, r10
    je err
4:
    mov r10b, dl
    cqo
    idiv r8
    cmp r10b, '/'
    je op_store
    mov rax, rdx
    jmp op_store
op_add:
    add rax, r8
    jmp op_store
op_sub:
    sub rax, r8
    jmp op_store
op_mul:
    imul rax, r8
op_store:
    mov qword ptr [rbx + r14*8 - 8], rax
    jmp tok

eol:
    cmp r14, 1
    jne err
    mov rax, qword ptr [rbx]
    test rax, rax
    jns 5f
    mov byte ptr [rbp], '-'
    inc rbp
    neg rax
5:
    lea rdi, [rip + tmp + 32]
    lea r9, [rip + tmp + 32]
    mov ecx, 10
6:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 6b
7:
    mov al, byte ptr [rdi]
    mov byte ptr [rbp], al
    inc rbp
    inc rdi
    cmp rdi, r9
    jb 7b
    mov byte ptr [rbp], 10
    inc rbp
    jmp nextline

err:
    mov dword ptr [rbp], 0x6f727265
    mov word ptr [rbp + 4], 0x0a72
    add rbp, 6

nextline:
    lea r12, [r15 + 1]
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rdx, rbp
    sub rdx, rsi
8:
    test rdx, rdx
    jz 9f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 9f
    add rsi, rax
    sub rdx, rax
    jmp 8b
9:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero INSZ
.balign 16
outbuf:
    .zero OUTSZ
.balign 16
stk:
    .zero 16064
tmp:
    .zero 64
