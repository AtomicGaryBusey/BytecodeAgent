.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea rbx, [rsi + rax]
byte_loop:
    cmp rsi, rbx
    jae read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp eax, 10
    jne 1f
    inc r12
1:
    cmp eax, 0x20
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 5
    jb ws
    test r15d, r15d
    jnz byte_loop
    mov r15d, 1
    inc r13
    jmp byte_loop
ws:
    xor r15d, r15d
    jmp byte_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = dest; advances rdi
utoa:
    lea r8, [rip + tmpbuf + 32]
    mov r9, r8
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    cmp r8, r9
    jae 4f
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    jmp 3b
4:
    ret

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
tmpbuf:
    .zero 32
