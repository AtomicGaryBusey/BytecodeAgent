.globl _start

.section .rodata
.balign 8
jtab:
    .quad op_add
    .quad op_move
    .quad op_out
    .quad op_in
    .quad op_jz
    .quad op_jnz
    .quad op_end

.bss
.balign 16
inbuf:
    .zero 65552
tape:
    .zero 30016
outbuf:
    .zero 65536
ops:
    .zero 524352

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65540
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
scan:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp scan
no_bang:
    lea r8, [rsi + rbx]
    mov r13, r8
    mov r10, r8
    jmp compile
found_bang:
    lea r8, [rsi + rcx]
    lea r13, [rsi + rcx + 1]
    lea r10, [rsi + rbx]
compile:
    lea r15, [rip + ops]
    xor r9d, r9d
    xor edx, edx
c_loop:
    cmp rsi, r8
    jae c_done
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x2b
    je c_plus
    cmp al, 0x2d
    je c_minus
    cmp al, 0x3e
    je c_right
    cmp al, 0x3c
    je c_left
    cmp al, 0x2e
    je c_out
    cmp al, 0x2c
    je c_in
    cmp al, 0x5b
    je c_open
    cmp al, 0x5d
    je c_close
    jmp c_loop
c_plus:
    mov ecx, 1
    xor eax, eax
    jmp c_merge
c_minus:
    mov ecx, -1
    xor eax, eax
    jmp c_merge
c_right:
    mov ecx, 1
    mov eax, 1
    jmp c_merge
c_left:
    mov ecx, -1
    mov eax, 1
c_merge:
    test r9, r9
    jz c_new
    cmp dword ptr [r15 + r9*8 - 8], eax
    jne c_new
    add dword ptr [r15 + r9*8 - 4], ecx
    jmp c_loop
c_new:
    mov dword ptr [r15 + r9*8], eax
    mov dword ptr [r15 + r9*8 + 4], ecx
    inc r9
    jmp c_loop
c_out:
    mov eax, 2
    xor ecx, ecx
    jmp c_new
c_in:
    mov eax, 3
    xor ecx, ecx
    jmp c_new
c_open:
    push r9
    inc rdx
    mov eax, 4
    xor ecx, ecx
    jmp c_new
c_close:
    test rdx, rdx
    jz fail
    pop rax
    dec rdx
    lea ecx, [rax + 1]
    mov dword ptr [r15 + r9*8], 5
    mov dword ptr [r15 + r9*8 + 4], ecx
    inc r9
    mov dword ptr [r15 + rax*8 + 4], r9d
    jmp c_loop
c_done:
    test rdx, rdx
    jnz fail
    mov dword ptr [r15 + r9*8], 6
    mov dword ptr [r15 + r9*8 + 4], 0

    lea rbx, [rip + tape]
    mov r12, r15
    xor r14d, r14d
    lea rbp, [rip + jtab]
run:
    mov eax, dword ptr [r12]
    movsxd rcx, dword ptr [r12 + 4]
    add r12, 8
    jmp qword ptr [rbp + rax*8]
op_add:
    add byte ptr [rbx], cl
    jmp run
op_move:
    add rbx, rcx
    jmp run
op_out:
    mov al, byte ptr [rbx]
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r14], al
    inc r14
    cmp r14, 65536
    jb run
    call flush
    jmp run
op_in:
    xor eax, eax
    cmp r13, r10
    jae 1f
    mov al, byte ptr [r13]
    inc r13
1:
    mov byte ptr [rbx], al
    jmp run
op_jz:
    cmp byte ptr [rbx], 0
    jne run
    lea r12, [r15 + rcx*8]
    jmp run
op_jnz:
    cmp byte ptr [rbx], 0
    je run
    lea r12, [r15 + rcx*8]
    jmp run
op_end:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    lea rsi, [rip + outbuf]
1:
    test r14, r14
    jz 2f
    mov eax, 1
    mov edi, 1
    mov rdx, r14
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub r14, rax
    jmp 1b
2:
    xor r14d, r14d
    ret

fail:
    mov eax, 60
    mov edi, 1
    syscall
