.globl _start

.section .rodata
lit_true:  .asciz "true"
lit_false: .asciz "false"
lit_null:  .asciz "null"
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "

.bss
.balign 16
buf: .zero 65680
numbuf: .zero 32

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
rd_loop:
    mov x0, #0
    add x1, x19, x21
    mov x2, #0x10000
    add x2, x2, #64
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x21, x21, x0
    b rd_loop
rd_done:
    mov x20, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne fail
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

fail:
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x4, numbuf
    add x4, x4, :lo12:numbuf
    add x4, x4, #24
    mov w5, #10
    strb w5, [x4]
    mov x6, x4
    mov x0, x20
    mov x3, #10
1:
    udiv x2, x0, x3
    msub x5, x2, x3, x0
    add w5, w5, #48
    strb w5, [x6, #-1]!
    mov x0, x2
    cbnz x0, 1b
    mov x1, x6
    add x2, x4, #1
    sub x2, x2, x6
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
1:
    ldrb w0, [x19, x20]
    cmp w0, #32
    b.eq 2f
    cmp w0, #9
    b.eq 2f
    cmp w0, #10
    b.eq 2f
    cmp w0, #13
    b.eq 2f
    ret
2:
    add x20, x20, #1
    b 1b

parse_value:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x19, x20]
    cmp w0, #'{'
    b.eq pv_obj
    cmp w0, #'['
    b.eq pv_arr
    cmp w0, #'"'
    b.eq pv_str
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    cmp w0, #'-'
    b.eq pv_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls pv_num
    b fail
pv_ret:
    ldp x29, x30, [sp], #16
    ret

pv_str:
    bl parse_string
    b pv_ret

pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    b pv_lit
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    b pv_lit
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
pv_lit:
    ldrb w2, [x1], #1
    cbz w2, pv_ret
    ldrb w0, [x19, x20]
    cmp w0, w2
    b.ne fail
    add x20, x20, #1
    b pv_lit

pv_num:
    ldrb w0, [x19, x20]
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    ldrb w0, [x19, x20]
1:
    cmp w0, #'0'
    b.eq num_zero
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi fail
    add x20, x20, #1
2:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi num_frac
    add x20, x20, #1
    b 2b
num_zero:
    add x20, x20, #1
num_frac:
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    b.ne num_exp
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
3:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi num_exp
    add x20, x20, #1
    b 3b
num_exp:
    ldrb w0, [x19, x20]
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne pv_ret
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    b.eq 4f
    cmp w0, #'-'
    b.ne 5f
4:
    add x20, x20, #1
    ldrb w0, [x19, x20]
5:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
6:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi pv_ret
    add x20, x20, #1
    b 6b

pv_obj:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.eq pv_close
obj_loop:
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.ne fail
    bl parse_string
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #':'
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #','
    b.eq obj_comma
    cmp w0, #'}'
    b.eq pv_close
    b fail
obj_comma:
    add x20, x20, #1
    bl skip_ws
    b obj_loop

pv_close:
    add x20, x20, #1
    b pv_ret

pv_arr:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.eq pv_close
arr_loop:
    bl parse_value
    bl skip_ws
    ldrb w0, [x19, x20]
    cmp w0, #','
    b.eq arr_comma
    cmp w0, #']'
    b.eq pv_close
    b fail
arr_comma:
    add x20, x20, #1
    bl skip_ws
    b arr_loop

parse_string:
    add x20, x20, #1
ps_loop:
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq ps_end
    cmp w0, #0x20
    b.lo fail
    cmp w0, #'\\'
    b.eq ps_esc
    add x20, x20, #1
    b ps_loop
ps_end:
    add x20, x20, #1
    ret
ps_esc:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq ps_esc_ok
    cmp w0, #'\\'
    b.eq ps_esc_ok
    cmp w0, #'/'
    b.eq ps_esc_ok
    cmp w0, #'b'
    b.eq ps_esc_ok
    cmp w0, #'f'
    b.eq ps_esc_ok
    cmp w0, #'n'
    b.eq ps_esc_ok
    cmp w0, #'r'
    b.eq ps_esc_ok
    cmp w0, #'t'
    b.eq ps_esc_ok
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    mov w3, #4
ps_hex:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 1f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi fail
1:
    add x20, x20, #1
    subs w3, w3, #1
    b.ne ps_hex
    b ps_loop
ps_esc_ok:
    add x20, x20, #1
    b ps_loop
