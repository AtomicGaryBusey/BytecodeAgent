.globl _start

.equ BUFSZ, 40000
.equ MAXL, 1100

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, BUFSZ
    sub edx, ebx
    jz read_done
    lea rsi, [rip + buf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    # parse lines
    xor r12d, r12d
    xor esi, esi
    xor r8d, r8d
    lea r9, [rip + buf]
    lea r10, [rip + lstart]
    lea r11, [rip + llen]
parse_loop:
    cmp rsi, rbx
    je parse_end
    cmp byte ptr [r9 + rsi], 10
    jne parse_next
    mov dword ptr [r10 + r12*4], r8d
    mov eax, esi
    sub eax, r8d
    mov dword ptr [r11 + r12*4], eax
    inc r12
    lea r8, [rsi + 1]
parse_next:
    inc rsi
    jmp parse_loop
parse_end:
    cmp r8, rbx
    jae parse_fin
    mov dword ptr [r10 + r12*4], r8d
    mov eax, ebx
    sub eax, r8d
    mov dword ptr [r11 + r12*4], eax
    inc r12
parse_fin:
    # binary insertion sort
    xor r13d, r13d
sort_outer:
    cmp r13, r12
    jae sort_done
    xor r14d, r14d
    mov r15, r13
bs_loop:
    cmp r14, r15
    jae bs_done
    lea rbp, [r14 + r15]
    shr rbp, 1
    lea rax, [rip + idx]
    mov esi, dword ptr [rax + rbp*4]
    mov edi, r13d
    call cmp_lines
    test eax, eax
    jz bs_right
    mov r15, rbp
    jmp bs_loop
bs_right:
    lea r14, [rbp + 1]
    jmp bs_loop
bs_done:
    lea rax, [rip + idx]
    mov rcx, r13
shift_loop:
    cmp rcx, r14
    jbe shift_done
    mov edx, dword ptr [rax + rcx*4 - 4]
    mov dword ptr [rax + rcx*4], edx
    dec rcx
    jmp shift_loop
shift_done:
    mov dword ptr [rax + r14*4], r13d
    inc r13
    jmp sort_outer
sort_done:
    # build output
    lea rdi, [rip + outbuf]
    lea r9, [rip + buf]
    lea r10, [rip + lstart]
    lea r11, [rip + llen]
    lea r8, [rip + idx]
    xor ecx, ecx
out_loop:
    cmp rcx, r12
    jae out_built
    mov eax, dword ptr [r8 + rcx*4]
    mov esi, dword ptr [r10 + rax*4]
    add rsi, r9
    mov edx, dword ptr [r11 + rax*4]
copy_loop:
    test rdx, rdx
    jz copy_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rdx
    jmp copy_loop
copy_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc rcx
    jmp out_loop
out_built:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jz finish
    mov rdx, rbx
    mov edi, 1
    mov eax, 1
    push rsi
    syscall
    pop rsi
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# edi = line a, esi = line b; returns eax = 1 if a < b else 0
cmp_lines:
    lea r9, [rip + buf]
    lea r10, [rip + lstart]
    lea r11, [rip + llen]
    mov r8d, dword ptr [r10 + rdi*4]
    add r8, r9
    mov ecx, dword ptr [r10 + rsi*4]
    add rcx, r9
    mov edx, dword ptr [r11 + rdi*4]
    mov esi, dword ptr [r11 + rsi*4]
    mov edi, edx
    cmp edi, esi
    cmova edi, esi
    xor eax, eax
cmp_loop:
    cmp rax, rdi
    je cmp_prefix
    movzx r9d, byte ptr [r8 + rax]
    movzx r10d, byte ptr [rcx + rax]
    cmp r9d, r10d
    jne cmp_diff
    inc rax
    jmp cmp_loop
cmp_diff:
    setb al
    movzx eax, al
    ret
cmp_prefix:
    cmp edx, esi
    setb al
    movzx eax, al
    ret

.bss
.balign 16
buf:
    .zero BUFSZ
lstart:
    .zero MAXL*4
llen:
    .zero MAXL*4
idx:
    .zero MAXL*4
outbuf:
    .zero BUFSZ + MAXL + 16
