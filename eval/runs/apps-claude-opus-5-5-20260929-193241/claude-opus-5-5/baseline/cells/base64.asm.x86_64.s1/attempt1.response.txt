.globl _start

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 90112

.text
_start:
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov rbx, r12
    lea rdi, [rip + outbuf]
    xor r8d, r8d
    lea r13, [rip + b64tab]
enc_loop:
    cmp rbx, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx
    mov r9d, 4
    call enc4
    add rsi, 3
    sub rbx, 3
    jmp enc_loop
enc_tail:
    test rbx, rbx
    jz enc_end
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov r9d, 2
    cmp rbx, 1
    je 1f
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    mov r9d, 3
1:
    call enc4
enc_end:
    test r8, r8
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi
write_out:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jz finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

enc4:
    mov r10d, 18
    xor r11d, r11d
2:
    cmp r11, r9
    jae 3f
    shrx ecx, eax, r10d
    and ecx, 63
    movzx edx, byte ptr [r13 + rcx]
    jmp 4f
3:
    mov edx, 61
4:
    call emit
    sub r10d, 6
    inc r11
    cmp r11, 4
    jb 2b
    ret

emit:
    mov byte ptr [rdi], dl
    inc rdi
    inc r8
    cmp r8, 76
    jne 5f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
5:
    ret
