.globl _start

.equ BUFSZ, 65600

.text
_start:
    lea rbx, [rip + buf]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, BUFSZ - 16
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r12, [rbx + r12]
    mov byte ptr [r12], 0
    mov rsi, rbx
    call skip_ws
    call value
    call skip_ws
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov r13, rsi
    sub r13, rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov rax, r13
    lea rdi, [rip + outbuf + 64]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov rsi, rdi
    lea rdx, [rip + outbuf + 65]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    je 1f
    ret
1:
    inc rsi
    jmp skip_ws

value:
    movzx eax, byte ptr [rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je v_true
    cmp al, 0x66
    je v_false
    cmp al, 0x6E
    je v_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb error
    cmp al, 0x39
    ja error
    jmp parse_number
v_true:
    lea rdi, [rip + s_true]
    jmp literal
v_false:
    lea rdi, [rip + s_false]
    jmp literal
v_null:
    lea rdi, [rip + s_null]
literal:
    mov cl, byte ptr [rdi]
    test cl, cl
    jz 1f
    cmp byte ptr [rsi], cl
    jne error
    inc rsi
    inc rdi
    jmp literal
1:
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x7D
    jne obj_member
    inc rsi
    ret
obj_member:
    cmp byte ptr [rsi], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], 0x3A
    jne error
    inc rsi
    call skip_ws
    call value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je 1f
    cmp al, 0x7D
    jne error
    inc rsi
    ret
1:
    inc rsi
    call skip_ws
    jmp obj_member

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x5D
    jne arr_elem
    inc rsi
    ret
arr_elem:
    call value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je 1f
    cmp al, 0x5D
    jne error
    inc rsi
    ret
1:
    inc rsi
    call skip_ws
    jmp arr_elem

parse_string:
    inc rsi
str_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb error
    inc rsi
    jmp str_loop
str_end:
    inc rsi
    ret
str_esc:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, 0x2F
    je str_esc_ok
    cmp al, 0x62
    je str_esc_ok
    cmp al, 0x66
    je str_esc_ok
    cmp al, 0x6E
    je str_esc_ok
    cmp al, 0x72
    je str_esc_ok
    cmp al, 0x74
    je str_esc_ok
    cmp al, 0x75
    jne error
    inc rsi
    mov ecx, 4
hex_loop:
    movzx eax, byte ptr [rsi]
    lea edx, [rax - 0x30]
    cmp edx, 9
    jbe hex_ok
    or eax, 0x20
    sub eax, 0x61
    cmp eax, 5
    ja error
hex_ok:
    inc rsi
    dec ecx
    jnz hex_loop
    jmp str_loop
str_esc_ok:
    inc rsi
    jmp str_loop

parse_number:
    cmp byte ptr [rsi], 0x2D
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    je num_zero
    cmp al, 0x31
    jb error
    cmp al, 0x39
    ja error
    inc rsi
    call digits
    jmp num_frac
num_zero:
    inc rsi
num_frac:
    cmp byte ptr [rsi], 0x2E
    jne num_exp
    inc rsi
    call need_digits
num_exp:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 0x65
    jne num_done
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B
    je 2f
    cmp al, 0x2D
    jne 3f
2:
    inc rsi
3:
    call need_digits
num_done:
    ret

need_digits:
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 9
    ja error
    inc rsi
digits:
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 9
    ja 1f
    inc rsi
    jmp digits
1:
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"

.bss
.balign 16
buf:
    .zero BUFSZ
outbuf:
    .zero 80
