.globl _start

.text
_start:
    mov     x19, #0                 // lines
    mov     x20, #0                 // words
    mov     x21, #0                 // in-word flag
    mov     x22, #0                 // bytes

readloop:
    mov     x0, #0
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #4096
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x22, x22, x0
    mov     x3, x1
    add     x4, x1, x0
byteloop:
    ldrb    w5, [x3], #1
    cmp     w5, #10
    cinc    x19, x19, eq
    sub     w6, w5, #9
    cmp     w6, #4
    b.ls    isws
    cmp     w5, #32
    b.eq    isws
    cbnz    x21, next
    add     x20, x20, #1
    mov     x21, #1
    b       next
isws:
    mov     x21, #0
next:
    cmp     x3, x4
    b.lo    byteloop
    b       readloop

done:
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x0, x19
    bl      putnum
    mov     w1, #32
    strb    w1, [x23], #1
    mov     x0, x20
    bl      putnum
    mov     w1, #32
    strb    w1, [x23], #1
    mov     x0, x22
    bl      putnum
    mov     w1, #10
    strb    w1, [x23], #1

    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x23, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

// x0 = value, x23 = output pointer (advanced)
putnum:
    adrp    x10, tmpend
    add     x10, x10, :lo12:tmpend
    mov     x14, x10
    mov     x11, #10
1:
    udiv    x12, x0, x11
    msub    x13, x12, x11, x0
    add     w13, w13, #48
    strb    w13, [x10, #-1]!
    mov     x0, x12
    cbnz    x0, 1b
2:
    ldrb    w13, [x10], #1
    strb    w13, [x23], #1
    cmp     x10, x14
    b.lo    2b
    ret

.bss
.balign 16
buf:
    .zero   4096
outbuf:
    .zero   128
tmp:
    .zero   32
tmpend:
    .zero   16
