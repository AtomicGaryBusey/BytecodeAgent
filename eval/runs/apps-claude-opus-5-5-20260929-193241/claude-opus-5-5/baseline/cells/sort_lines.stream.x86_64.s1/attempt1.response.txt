.globl _start

.text
_start:
    xor ebx, ebx
.Lread:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + buf]
    add rsi, rbx
    mov edx, 40000
    sub rdx, rbx
    syscall
    test rax, rax
    jle .Lparse
    add rbx, rax
    jmp .Lread

.Lparse:
    lea r12, [rip + buf]
    lea r13, [rip + ents]
    xor r14d, r14d
    xor ecx, ecx
    xor r15d, r15d
.Lploop:
    cmp rcx, rbx
    jae .Lpend
    cmp byte ptr [r12 + rcx], 10
    jne .Lpnext
    cmp r14, 1024
    jae .Lpskip
    lea rax, [r12 + r15]
    mov rdx, r14
    shl rdx, 4
    mov qword ptr [r13 + rdx], rax
    mov rax, rcx
    sub rax, r15
    mov qword ptr [r13 + rdx + 8], rax
    inc r14
.Lpskip:
    lea r15, [rcx + 1]
.Lpnext:
    inc rcx
    jmp .Lploop
.Lpend:
    cmp r15, rbx
    jae .Lsort
    cmp r14, 1024
    jae .Lsort
    lea rax, [r12 + r15]
    mov rdx, r14
    shl rdx, 4
    mov qword ptr [r13 + rdx], rax
    mov rax, rbx
    sub rax, r15
    mov qword ptr [r13 + rdx + 8], rax
    inc r14

.Lsort:
    mov esi, 1
.Louter:
    cmp rsi, r14
    jae .Loutput
    mov rax, rsi
    shl rax, 4
    mov r10, qword ptr [r13 + rax]
    mov r11, qword ptr [r13 + rax + 8]
    lea rbx, [rsi - 1]
.Linner:
    test rbx, rbx
    js .Lplace
    mov rax, rbx
    shl rax, 4
    mov r8, qword ptr [r13 + rax]
    mov r9, qword ptr [r13 + rax + 8]
    call cmpfn
    test eax, eax
    jz .Lplace
    mov rax, rbx
    shl rax, 4
    mov qword ptr [r13 + rax + 16], r8
    mov qword ptr [r13 + rax + 24], r9
    dec rbx
    jmp .Linner
.Lplace:
    mov rax, rbx
    shl rax, 4
    mov qword ptr [r13 + rax + 16], r10
    mov qword ptr [r13 + rax + 24], r11
    inc rsi
    jmp .Louter

.Loutput:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
.Loloop:
    cmp rbx, r14
    jae .Lwrite
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    cld
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp .Loloop
.Lwrite:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
.Lwloop:
    test rdx, rdx
    jle .Lexit
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rdx, rax
    jmp .Lwloop
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

# compare a=(r8,r9) with b=(r10,r11); eax=1 if a>b else 0
# clobbers rax, rcx, rdx, rdi
cmpfn:
    mov rcx, r9
    cmp rcx, r11
    cmova rcx, r11
    xor edx, edx
.Lcq:
    mov rax, rcx
    sub rax, rdx
    cmp rax, 8
    jb .Lcb
    mov rax, qword ptr [r8 + rdx]
    mov rdi, qword ptr [r10 + rdx]
    cmp rax, rdi
    jne .Lcdq
    add rdx, 8
    jmp .Lcq
.Lcdq:
    bswap rax
    bswap rdi
    cmp rax, rdi
    seta al
    movzx eax, al
    ret
.Lcb:
    cmp rdx, rcx
    jae .Lclen
    movzx eax, byte ptr [r8 + rdx]
    movzx edi, byte ptr [r10 + rdx]
    cmp eax, edi
    jne .Lcdb
    inc rdx
    jmp .Lcb
.Lcdb:
    seta al
    movzx eax, al
    ret
.Lclen:
    cmp r9, r11
    seta al
    movzx eax, al
    ret

.bss
.balign 16
buf:
    .zero 40016
.balign 16
ents:
    .zero 16448
.balign 16
outbuf:
    .zero 42048
