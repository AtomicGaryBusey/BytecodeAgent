.globl _start

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    add x21, x21, x0
    adrp x9, inbuf
    add x9, x9, :lo12:inbuf
    add x10, x9, x0
byte_loop:
    cmp x9, x10
    b.hs read_loop
    ldrb w3, [x9], #1
    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    cbnz x22, byte_loop
    add x20, x20, #1
    mov x22, #1
    b byte_loop
is_ws:
    mov x22, #0
    b byte_loop

done_read:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x19
    bl utoa
    mov w4, #32
    strb w4, [x1], #1
    mov x0, x20
    bl utoa
    mov w4, #32
    strb w4, [x1], #1
    mov x0, x21
    bl utoa
    mov w4, #10
    strb w4, [x1], #1
    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// utoa: x0 = value, x1 = dest; returns x1 advanced past digits
utoa:
    adrp x5, tmpbuf
    add x5, x5, :lo12:tmpbuf
    add x6, x5, #32
    mov x7, #10
2:
    udiv x4, x0, x7
    msub x3, x4, x7, x0
    add w3, w3, #48
    strb w3, [x6, #-1]!
    mov x0, x4
    cbnz x0, 2b
    add x5, x5, #32
3:
    cmp x6, x5
    b.hs 4f
    ldrb w3, [x6], #1
    strb w3, [x1], #1
    b 3b
4:
    ret

.bss
.balign 16
inbuf:
    .zero 4096
outbuf:
    .zero 128
tmpbuf:
    .zero 32
