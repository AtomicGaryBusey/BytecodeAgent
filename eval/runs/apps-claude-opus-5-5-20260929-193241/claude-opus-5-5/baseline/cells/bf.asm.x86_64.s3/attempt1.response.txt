.globl _start

.section .rodata
.balign 8
jtab:
    .quad op_halt
    .quad op_right
    .quad op_left
    .quad op_inc
    .quad op_dec
    .quad op_out
    .quad op_in
    .quad op_open
    .quad op_close

.bss
.balign 16
inbuf:  .zero 70016
ops:    .zero 70016
.balign 16
tgt:    .zero 280064
tape:   .zero 30016
outbuf: .zero 70016
ctab:   .zero 256

.text
_start:
    # build command table
    mov byte ptr [ctab + 62], 1
    mov byte ptr [ctab + 60], 2
    mov byte ptr [ctab + 43], 3
    mov byte ptr [ctab + 45], 4
    mov byte ptr [ctab + 46], 5
    mov byte ptr [ctab + 44], 6
    mov byte ptr [ctab + 91], 7
    mov byte ptr [ctab + 93], 8

    xor r13d, r13d
read_loop:
    mov edx, 70000
    sub rdx, r13
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [inbuf + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:

    xor ecx, ecx
find_loop:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [inbuf + rcx], 33
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    lea r14, [inbuf + rcx + 1]
    mov r9, rcx
    jmp have_split
no_bang:
    lea r14, [inbuf + r13]
    mov r9, r13
have_split:
    lea r15, [inbuf + r13]

    xor esi, esi
    xor ebx, ebx
    xor r12d, r12d
comp_loop:
    cmp rsi, r9
    jae comp_done
    movzx eax, byte ptr [inbuf + rsi]
    movzx eax, byte ptr [ctab + rax]
    test eax, eax
    jz comp_next
    mov byte ptr [ops + rbx], al
    cmp eax, 7
    jne 1f
    push rbx
    inc r12
    jmp comp_stored
1:
    cmp eax, 8
    jne comp_stored
    test r12, r12
    jz bad
    dec r12
    pop rdx
    mov dword ptr [tgt + rbx*4], edx
    mov dword ptr [tgt + rdx*4], ebx
comp_stored:
    inc rbx
comp_next:
    inc rsi
    jmp comp_loop
comp_done:
    test r12, r12
    jnz bad
    mov byte ptr [ops + rbx], 0

    xor ebx, ebx
    lea rdi, [tape]
    xor r8d, r8d
exec:
    movzx eax, byte ptr [ops + rbx]
    jmp qword ptr [jtab + rax*8]

op_right:
    inc rdi
    inc rbx
    jmp exec
op_left:
    dec rdi
    inc rbx
    jmp exec
op_inc:
    inc byte ptr [rdi]
    inc rbx
    jmp exec
op_dec:
    dec byte ptr [rdi]
    inc rbx
    jmp exec
op_out:
    mov al, byte ptr [rdi]
    mov byte ptr [outbuf + r8], al
    inc r8
    inc rbx
    jmp exec
op_in:
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rdi], al
    inc rbx
    jmp exec
1:
    mov byte ptr [rdi], 0
    inc rbx
    jmp exec
op_open:
    cmp byte ptr [rdi], 0
    jne 1f
    mov ebx, dword ptr [tgt + rbx*4]
1:
    inc rbx
    jmp exec
op_close:
    cmp byte ptr [rdi], 0
    je 1f
    mov ebx, dword ptr [tgt + rbx*4]
1:
    inc rbx
    jmp exec

op_halt:
    lea rsi, [outbuf]
    mov rdx, r8
write_loop:
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

bad:
    mov eax, 60
    mov edi, 1
    syscall
