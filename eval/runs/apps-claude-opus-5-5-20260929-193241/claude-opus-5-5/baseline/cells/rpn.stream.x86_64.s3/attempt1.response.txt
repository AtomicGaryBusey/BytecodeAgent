.globl _start

.equ INSIZE, 70000

.bss
.balign 16
inbuf:
    .zero 70016
outbuf:
    .zero 480000
stackbuf:
    .zero 16384
tmpbuf:
    .zero 64

.text
_start:
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov edx, INSIZE
    sub rdx, r15
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea r14, [rip + outbuf]
    lea r13, [rip + stackbuf]
    lea rsi, [rip + inbuf]
    lea rbx, [rsi + r15]

line_loop:
    cmp rsi, rbx
    jae finish
    mov rdi, rsi
find_nl:
    cmp rdi, rbx
    jae found_nl
    cmp byte ptr [rdi], 10
    je found_nl
    inc rdi
    jmp find_nl
found_nl:
    mov r8, rdi
    xor r12d, r12d
    mov rcx, rsi

tok_loop:
    cmp rcx, r8
    jae line_end
    movzx eax, byte ptr [rcx]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    mov rdx, rcx
tok_scan:
    cmp rcx, r8
    jae tok_end
    movzx eax, byte ptr [rcx]
    cmp al, 32
    je tok_end
    cmp al, 9
    je tok_end
    inc rcx
    jmp tok_scan
skip_sep:
    inc rcx
    jmp tok_loop

tok_end:
    mov r9, rcx
    sub r9, rdx
    cmp r9, 1
    jne literal
    movzx eax, byte ptr [rdx]
    cmp al, '+'
    je do_op
    cmp al, '-'
    je do_op
    cmp al, '*'
    je do_op
    cmp al, '/'
    je do_op
    cmp al, '%'
    je do_op

literal:
    mov r10, rdx
    xor r11d, r11d
    cmp byte ptr [r10], '-'
    jne 1f
    mov r11d, 1
    inc r10
1:
    mov r9, rcx
    sub r9, r10
    cmp r9, 1
    jb line_error
    cmp r9, 18
    ja line_error
    xor eax, eax
2:
    cmp r10, rcx
    jae 3f
    movzx edi, byte ptr [r10]
    sub edi, '0'
    cmp edi, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdi
    inc r10
    jmp 2b
3:
    test r11d, r11d
    jz 4f
    neg rax
4:
    cmp r12, 2000
    jae line_error
    mov qword ptr [r13 + r12*8], rax
    inc r12
    jmp tok_loop

do_op:
    mov r9d, eax
    cmp r12, 2
    jb line_error
    dec r12
    mov r10, qword ptr [r13 + r12*8]
    dec r12
    mov rax, qword ptr [r13 + r12*8]
    cmp r9b, '+'
    jne 1f
    add rax, r10
    jmp push_res
1:
    cmp r9b, '-'
    jne 2f
    sub rax, r10
    jmp push_res
2:
    cmp r9b, '*'
    jne 3f
    imul rax, r10
    jmp push_res
3:
    test r10, r10
    jz line_error
    cmp r10, -1
    jne 4f
    movabs r11, 0x8000000000000000
    cmp rax, r11
    je line_error
4:
    cqo
    idiv r10
    cmp r9b, '%'
    jne push_res
    mov rax, rdx
push_res:
    mov qword ptr [r13 + r12*8], rax
    inc r12
    jmp tok_loop

line_end:
    cmp r12, 1
    jne line_error
    mov rax, qword ptr [r13]
    test rax, rax
    jns 1f
    mov byte ptr [r14], '-'
    inc r14
    neg rax
1:
    lea rdi, [rip + tmpbuf + 40]
    mov r9, rdi
    mov r10d, 10
2:
    xor edx, edx
    div r10
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
3:
    cmp rdi, r9
    jae 4f
    movzx eax, byte ptr [rdi]
    mov byte ptr [r14], al
    inc rdi
    inc r14
    jmp 3b
4:
    mov byte ptr [r14], 10
    inc r14
    jmp next_line

line_error:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6

next_line:
    lea rsi, [r8 + 1]
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
