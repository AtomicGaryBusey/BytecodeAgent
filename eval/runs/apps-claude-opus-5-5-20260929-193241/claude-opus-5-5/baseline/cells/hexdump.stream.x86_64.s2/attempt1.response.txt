.globl _start

.equ INCAP, 69632

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 69632
outbuf:
    .zero 340000

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INCAP
    sub edx, r12d
    jz done_read
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    lea rbx, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r13d, r13d
    lea r15, [rip + hexd]
row:
    cmp r13, r12
    jae final
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 2f
    mov r14d, 16
2:
    mov eax, r13d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hl:
    cmp rcx, r14
    jae pad
    lea rax, [rbx + r13]
    movzx eax, byte ptr [rax + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp nextb
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
nextb:
    add rdi, 3
    cmp ecx, 7
    jne 1f
    mov byte ptr [rdi], 0x20
    inc rdi
1:
    inc ecx
    cmp ecx, 16
    jb hl
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
chl:
    lea rax, [rbx + r13]
    movzx eax, byte ptr [rax + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 3f
    mov eax, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    cmp rcx, r14
    jb chl
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r13, r14
    jmp row
final:
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wloop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    push rdx
    push rsi
    syscall
    pop rsi
    pop rdx
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp wloop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov r8d, 8
4:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec r8d
    jnz 4b
    ret
