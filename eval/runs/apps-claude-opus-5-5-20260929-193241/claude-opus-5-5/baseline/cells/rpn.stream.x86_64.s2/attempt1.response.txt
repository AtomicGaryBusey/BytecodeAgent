.globl _start

.equ INCAP, 70000
.equ OUTCAP, 440000
.equ FLUSHAT, 400000
.equ STACKMAX, 1024

.text
_start:
    xor r15d, r15d
read_loop:
    mov edx, INCAP
    sub rdx, r15
    jz done_read
    lea rsi, [rip + inbuf]
    add rsi, r15
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle done_read
    add r15, rax
    jmp read_loop
done_read:
    lea r12, [rip + inbuf]
    lea r13, [r12 + r15]
    lea rbp, [rip + outbuf]
    lea r14, [rip + vstack]

line_loop:
    cmp r12, r13
    jae finish
    xor ebx, ebx
tok_loop:
    cmp r12, r13
    jae eol
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    cmp al, 10
    je eol
    mov rsi, r12
1:
    inc r12
    cmp r12, r13
    jae 2f
    movzx eax, byte ptr [r12]
    cmp al, 32
    je 2f
    cmp al, 9
    je 2f
    cmp al, 10
    je 2f
    jmp 1b
2:
    mov rcx, r12
    sub rcx, rsi
    movzx eax, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod
literal:
    xor edx, edx
    cmp al, 45
    jne 3f
    mov edx, 1
    inc rsi
    dec rcx
3:
    test rcx, rcx
    jz line_err
    cmp rcx, 18
    ja line_err
    xor eax, eax
4:
    movzx r8d, byte ptr [rsi]
    sub r8d, 48
    cmp r8d, 9
    ja line_err
    imul rax, rax, 10
    add rax, r8
    inc rsi
    dec rcx
    jnz 4b
    test edx, edx
    jz 5f
    neg rax
5:
    cmp rbx, STACKMAX
    jae line_err
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    jmp tok_loop

skip_sep:
    inc r12
    jmp tok_loop

op_add:
    cmp rbx, 2
    jb line_err
    mov rax, qword ptr [r14 + rbx*8 - 16]
    add rax, qword ptr [r14 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r14 + rbx*8 - 8], rax
    jmp tok_loop
op_sub:
    cmp rbx, 2
    jb line_err
    mov rax, qword ptr [r14 + rbx*8 - 16]
    sub rax, qword ptr [r14 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r14 + rbx*8 - 8], rax
    jmp tok_loop
op_mul:
    cmp rbx, 2
    jb line_err
    mov rax, qword ptr [r14 + rbx*8 - 16]
    imul rax, qword ptr [r14 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r14 + rbx*8 - 8], rax
    jmp tok_loop
op_div:
    xor r10d, r10d
    jmp do_div
op_mod:
    mov r10d, 1
do_div:
    cmp rbx, 2
    jb line_err
    mov r8, qword ptr [r14 + rbx*8 - 8]
    test r8, r8
    jz line_err
    mov rax, qword ptr [r14 + rbx*8 - 16]
    cmp r8, -1
    jne 6f
    mov r9, 0x8000000000000000
    cmp rax, r9
    je line_err
6:
    cqo
    idiv r8
    test r10d, r10d
    jz 7f
    mov rax, rdx
7:
    dec rbx
    mov qword ptr [r14 + rbx*8 - 8], rax
    jmp tok_loop

eol:
    cmp rbx, 1
    jne line_err
    mov rax, qword ptr [r14]
    test rax, rax
    jns 8f
    mov byte ptr [rbp], 45
    inc rbp
    neg rax
8:
    lea rsi, [rip + numbuf + 32]
    mov r8, rsi
    mov ecx, 10
9:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 9b
10:
    mov al, byte ptr [rsi]
    mov byte ptr [rbp], al
    inc rbp
    inc rsi
    cmp rsi, r8
    jb 10b
    mov byte ptr [rbp], 10
    inc rbp
    jmp consume_nl

line_err:
    cmp r12, r13
    jae 11f
    cmp byte ptr [r12], 10
    je 11f
    inc r12
    jmp line_err
11:
    mov dword ptr [rbp], 0x6f727265
    mov word ptr [rbp + 4], 0x0a72
    add rbp, 6
consume_nl:
    cmp r12, r13
    jae next_line
    inc r12
next_line:
    lea rax, [rip + outbuf]
    mov rcx, rbp
    sub rcx, rax
    cmp rcx, FLUSHAT
    jb line_loop
    call flush
    jmp line_loop

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    lea rsi, [rip + outbuf]
12:
    mov rdx, rbp
    sub rdx, rsi
    jle 13f
    mov edi, 1
    mov eax, 1
    push rsi
    syscall
    pop rsi
    test rax, rax
    jle 13f
    add rsi, rax
    jmp 12b
13:
    lea rbp, [rip + outbuf]
    ret

.bss
.balign 16
inbuf:
    .zero INCAP
.balign 16
outbuf:
    .zero OUTCAP
.balign 16
vstack:
    .zero 8 * STACKMAX + 64
numbuf:
    .zero 48
