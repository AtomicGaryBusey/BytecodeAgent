.globl _start

.equ BUFSZ, 70000

.section .rodata
s_true: .asciz "true"
s_false: .asciz "false"
s_null: .asciz "null"
s_err: .ascii "error at "
s_ok: .ascii "ok\n"

.bss
.balign 16
buf: .zero 70016
outbuf: .zero 40
numend: .zero 24

.text
_start:
    lea rbx, [rip + buf]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, BUFSZ
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r12, [rbx + r12]
    mov byte ptr [r12], 0
    mov rsi, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rax, rsi
    sub rax, rbx
    lea r8, [rip + numend]
    mov byte ptr [r8], 10
    mov r9, r8
    mov r10d, 10
1:
    xor edx, edx
    div r10
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 1b
    sub r9, 9
    lea rdi, [rip + s_err]
    xor ecx, ecx
2:
    mov al, byte ptr [rdi + rcx]
    mov byte ptr [r9 + rcx], al
    inc ecx
    cmp ecx, 9
    jb 2b
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    lea rdx, [r8 + 1]
    sub rdx, r9
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
    ret
1:
    inc rsi
    jmp skip_ws

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    lea ecx, [rax - '0']
    cmp ecx, 9
    jbe parse_number
    lea rdi, [rip + s_true]
    cmp al, 't'
    je parse_lit
    lea rdi, [rip + s_false]
    cmp al, 'f'
    je parse_lit
    lea rdi, [rip + s_null]
    cmp al, 'n'
    je parse_lit
    jmp error

parse_lit:
    mov al, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp parse_lit
1:
    ret

parse_string:
    inc rsi
ps_loop:
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb error
    inc rsi
    jmp ps_loop
ps_end:
    inc rsi
    ret
ps_esc:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    jne error
    inc rsi
    mov edx, 4
ps_hex:
    movzx eax, byte ptr [rsi]
    lea ecx, [rax - '0']
    cmp ecx, 9
    jbe 1f
    mov ecx, eax
    or ecx, 0x20
    sub ecx, 'a'
    cmp ecx, 5
    ja error
1:
    inc rsi
    dec edx
    jnz ps_hex
    jmp ps_loop
ps_esc_ok:
    inc rsi
    jmp ps_loop

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne 1f
    inc rsi
    movzx eax, byte ptr [rsi]
1:
    cmp al, '0'
    je pn_zero
    lea ecx, [rax - '1']
    cmp ecx, 8
    ja error
    inc rsi
    call digits
    jmp pn_frac
pn_zero:
    inc rsi
pn_frac:
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne pn_exp
    inc rsi
    call digits1
pn_exp:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne pn_done
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 2f
    cmp al, '-'
    jne 3f
2:
    inc rsi
3:
    call digits1
pn_done:
    ret

digits1:
    movzx eax, byte ptr [rsi]
    lea ecx, [rax - '0']
    cmp ecx, 9
    ja error
digits:
    movzx eax, byte ptr [rsi]
    lea ecx, [rax - '0']
    cmp ecx, 9
    ja 1f
    inc rsi
    jmp digits
1:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], ']'
    je pa_end
pa_loop:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je 1f
    cmp al, ']'
    je pa_end
    jmp error
1:
    inc rsi
    call skip_ws
    jmp pa_loop
pa_end:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], '}'
    je po_end
po_loop:
    cmp byte ptr [rsi], '"'
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je 1f
    cmp al, '}'
    je po_end
    jmp error
1:
    inc rsi
    call skip_ws
    jmp po_loop
po_end:
    inc rsi
    ret
