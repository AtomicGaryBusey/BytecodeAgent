.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf
    mov x21, x26
    adrp x22, vstack
    add x22, x22, :lo12:vstack

read_loop:
    mov x3, #0x20000
    add x2, x19, x3
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

line_loop:
    cmp x19, x20
    b.hs finish
    mov x23, #0

tok_loop:
    cmp x19, x20
    b.hs end_line
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_sep
    cmp w0, #9
    b.eq skip_sep
    cmp w0, #10
    b.eq end_line
    b tok_start
skip_sep:
    add x19, x19, #1
    b tok_loop

tok_start:
    mov x24, x19
1:
    add x19, x19, #1
    cmp x19, x20
    b.hs 2f
    ldrb w0, [x19]
    cmp w0, #32
    b.eq 2f
    cmp w0, #9
    b.eq 2f
    cmp w0, #10
    b.eq 2f
    b 1b
2:
    sub x25, x19, x24
    ldrb w1, [x24]
    cmp x25, #1
    b.ne lit
    cmp w1, #'+'
    b.eq op
    cmp w1, #'-'
    b.eq op
    cmp w1, #'*'
    b.eq op
    cmp w1, #'/'
    b.eq op
    cmp w1, #'%'
    b.eq op

lit:
    mov x2, x24
    mov x6, #0
    cmp w1, #'-'
    b.ne 3f
    mov x6, #1
    add x2, x2, #1
3:
    sub x3, x19, x2
    cbz x3, error
    cmp x3, #18
    b.hi error
    mov x4, #0
    mov x5, #10
4:
    ldrb w1, [x2], #1
    sub w1, w1, #'0'
    cmp w1, #9
    b.hi error
    madd x4, x4, x5, x1
    cmp x2, x19
    b.lo 4b
    cbz x6, 5f
    neg x4, x4
5:
    str x4, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

op:
    mov w9, w1
    cmp x23, #2
    b.lo error
    sub x23, x23, #2
    ldr x0, [x22, x23, lsl #3]
    add x7, x23, #1
    ldr x1, [x22, x7, lsl #3]
    cmp w9, #'+'
    b.eq op_add
    cmp w9, #'-'
    b.eq op_sub
    cmp w9, #'*'
    b.eq op_mul
    cbz x1, error
    cmn x1, #1
    b.ne 6f
    movz x2, #0x8000, lsl #48
    cmp x0, x2
    b.eq error
6:
    sdiv x2, x0, x1
    cmp w9, #'/'
    b.eq 7f
    msub x0, x2, x1, x0
    b op_store
7:
    mov x0, x2
    b op_store
op_add:
    add x0, x0, x1
    b op_store
op_sub:
    sub x0, x0, x1
    b op_store
op_mul:
    mul x0, x0, x1
op_store:
    str x0, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

end_line:
    cmp x23, #1
    b.ne error
    ldr x0, [x22]
    bl print_num
    mov w0, #10
    strb w0, [x21], #1
    cmp x19, x20
    b.hs 8f
    add x19, x19, #1
8:
    bl maybe_flush
    b line_loop

error:
    adrp x0, errmsg
    add x0, x0, :lo12:errmsg
    mov x1, #6
9:
    ldrb w2, [x0], #1
    strb w2, [x21], #1
    subs x1, x1, #1
    b.ne 9b
10:
    cmp x19, x20
    b.hs 11f
    ldrb w0, [x19], #1
    cmp w0, #10
    b.ne 10b
11:
    bl maybe_flush
    b line_loop

finish:
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0
    mov x0, #0
    mov x8, #94
    svc #0

print_num:
    sub sp, sp, #32
    mov x1, x0
    tbz x0, #63, 1f
    mov w2, #'-'
    strb w2, [x21], #1
    neg x1, x0
1:
    add x3, sp, #32
    mov x5, #10
2:
    udiv x4, x1, x5
    msub x6, x4, x5, x1
    add w6, w6, #'0'
    strb w6, [x3, #-1]!
    mov x1, x4
    cbnz x1, 2b
    add x7, sp, #32
3:
    ldrb w6, [x3], #1
    strb w6, [x21], #1
    cmp x3, x7
    b.lo 3b
    add sp, sp, #32
    ret

maybe_flush:
    sub x0, x21, x26
    cmp x0, #0xF000
    b.hs flush
    ret

flush:
    mov x1, x26
    sub x2, x21, x26
1:
    cbz x2, 2f
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x1, x1, x0
    sub x2, x2, x0
    b 1b
2:
    mov x21, x26
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .zero 0x20000
.balign 16
outbuf:
    .zero 0x10000
.balign 16
vstack:
    .zero 16384
