.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 64
inbuf:
    .zero 65536
outbuf:
    .zero 90000

.text
_start:
    xor r12d, r12d
read_loop:
    mov eax, 65536
    sub eax, r12d
    jz done_read
    mov edx, eax
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop

done_read:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r8d, r8d
    lea rbx, [rip + alphabet]
    mov rcx, r12

triple:
    cmp rcx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov r9d, eax
    mov edx, r9d
    shr edx, 18
    and edx, 63
    call putidx
    mov edx, r9d
    shr edx, 12
    and edx, 63
    call putidx
    mov edx, r9d
    shr edx, 6
    and edx, 63
    call putidx
    mov edx, r9d
    and edx, 63
    call putidx
    add rsi, 3
    sub rcx, 3
    jmp triple

tail:
    test rcx, rcx
    jz fin
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 2
    jne 2f
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
2:
    mov r9d, eax
    mov edx, r9d
    shr edx, 18
    and edx, 63
    call putidx
    mov edx, r9d
    shr edx, 12
    and edx, 63
    call putidx
    cmp rcx, 2
    jne 3f
    mov edx, r9d
    shr edx, 6
    and edx, 63
    call putidx
    jmp 4f
3:
    mov al, 61
    call putc
4:
    mov al, 61
    call putc

fin:
    test r8d, r8d
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    push rdx
    push rsi
    syscall
    pop rsi
    pop rdx
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

putidx:
    movzx eax, byte ptr [rbx + rdx]
putc:
    mov byte ptr [rdi], al
    inc rdi
    inc r8d
    cmp r8d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:
    ret
