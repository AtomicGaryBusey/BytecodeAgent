.globl _start

.equ BUFSZ, 32784

.text
_start:
    xor r12d, r12d
rd_loop:
    mov edx, BUFSZ
    sub rdx, r12
    jz rd_done
    lea rsi, [rip + buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    lea rsi, [rip + buf]
    lea r8, [rip + starts]
    lea r9, [rip + lens]
    xor ecx, ecx
    xor edx, edx
    xor r14d, r14d
p_loop:
    cmp rcx, r12
    jae p_end
    cmp byte ptr [rsi + rcx], 10
    jne p_next
    mov dword ptr [r8 + r14*4], edx
    mov eax, ecx
    sub eax, edx
    mov dword ptr [r9 + r14*4], eax
    inc r14
    lea edx, [rcx + 1]
p_next:
    inc rcx
    jmp p_loop
p_end:
    cmp rdx, r12
    jae p_done
    mov dword ptr [r8 + r14*4], edx
    mov eax, r12d
    sub eax, edx
    mov dword ptr [r9 + r14*4], eax
    inc r14
p_done:
    lea r12, [rip + idxa]
    lea r13, [rip + idxb]
    xor ecx, ecx
i_loop:
    cmp rcx, r14
    jae i_done
    mov dword ptr [r12 + rcx*4], ecx
    inc rcx
    jmp i_loop
i_done:
    mov r15d, 1
Lw:
    cmp r15, r14
    jae sort_done
    xor r11d, r11d
Ll:
    cmp r11, r14
    jae Lpass_end
    mov rbx, r11
    lea r10, [r11 + r15]
    cmp r10, r14
    cmova r10, r14
    lea r11, [rbx + r15*2]
    cmp r11, r14
    cmova r11, r14
    mov rbp, r10
Lm:
    cmp rbx, r10
    jae Lrest_j
    cmp rbp, r11
    jae Lrest_i
    mov r8d, dword ptr [r12 + rbx*4]
    mov r9d, dword ptr [r12 + rbp*4]
    call cmp_lines
    lea rdx, [rbx + rbp]
    sub rdx, r10
    test eax, eax
    jnz Ltakej
    mov eax, dword ptr [r12 + rbx*4]
    mov dword ptr [r13 + rdx*4], eax
    inc rbx
    jmp Lm
Ltakej:
    mov eax, dword ptr [r12 + rbp*4]
    mov dword ptr [r13 + rdx*4], eax
    inc rbp
    jmp Lm
Lrest_i:
    cmp rbx, r10
    jae Lrest_j
    lea rdx, [rbx + rbp]
    sub rdx, r10
    mov eax, dword ptr [r12 + rbx*4]
    mov dword ptr [r13 + rdx*4], eax
    inc rbx
    jmp Lrest_i
Lrest_j:
    cmp rbp, r11
    jae Ll
    lea rdx, [rbx + rbp]
    sub rdx, r10
    mov eax, dword ptr [r12 + rbp*4]
    mov dword ptr [r13 + rdx*4], eax
    inc rbp
    jmp Lrest_j
Lpass_end:
    xchg r12, r13
    shl r15, 1
    jmp Lw
sort_done:
    lea rdi, [rip + outbuf]
    lea r8, [rip + starts]
    lea r9, [rip + lens]
    lea r10, [rip + buf]
    xor ebx, ebx
o_loop:
    cmp rbx, r14
    jae o_done
    mov eax, dword ptr [r12 + rbx*4]
    mov esi, dword ptr [r8 + rax*4]
    add rsi, r10
    mov ecx, dword ptr [r9 + rax*4]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp o_loop
o_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
w_loop:
    test rdx, rdx
    jz w_done
    mov edi, 1
    mov eax, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle w_done
    add rsi, rax
    sub rdx, rax
    jmp w_loop
w_done:
    mov eax, 60
    xor edi, edi
    syscall

# in: r8d = a, r9d = b ; out: eax = 1 if line a > line b else 0
cmp_lines:
    lea rax, [rip + starts]
    mov esi, dword ptr [rax + r8*4]
    mov edi, dword ptr [rax + r9*4]
    lea rax, [rip + lens]
    mov r8d, dword ptr [rax + r8*4]
    mov edx, dword ptr [rax + r9*4]
    lea rax, [rip + buf]
    add rsi, rax
    add rdi, rax
    mov ecx, r8d
    cmp ecx, edx
    cmova ecx, edx
    test ecx, ecx
    jz 2f
1:
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jne 3f
    inc rsi
    inc rdi
    dec ecx
    jnz 1b
2:
    xor eax, eax
    cmp r8d, edx
    seta al
    ret
3:
    seta al
    movzx eax, al
    ret

.bss
.balign 16
buf:
    .zero 32800
.balign 16
starts:
    .zero 4200
.balign 16
lens:
    .zero 4200
.balign 16
idxa:
    .zero 4200
.balign 16
idxb:
    .zero 4200
.balign 16
outbuf:
    .zero 34000
