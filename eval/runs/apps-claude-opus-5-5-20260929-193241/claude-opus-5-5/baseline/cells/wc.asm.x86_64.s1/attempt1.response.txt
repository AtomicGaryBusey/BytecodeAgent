.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je is_ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe is_ws
    test r15, r15
    jnz next
    inc r13
    mov r15d, 1
    jmp next
is_ws:
    xor r15d, r15d
next:
    dec rcx
    jnz byte_loop
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jnz write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

utoa:
    lea rsi, [rip + tmpbuf + 32]
    mov r8, rsi
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    cmp rsi, r8
    jb 2b
    ret

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
tmpbuf:
    .zero 64
