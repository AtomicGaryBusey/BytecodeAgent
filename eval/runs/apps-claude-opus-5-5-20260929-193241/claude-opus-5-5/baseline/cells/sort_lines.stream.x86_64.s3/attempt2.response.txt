.globl _start

.equ BUFSZ, 40960

.text
_start:
    xor r12d, r12d
rd_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [buf + r12]
    mov edx, BUFSZ
    sub rdx, r12
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    lea rsi, [rip + buf]
    lea r13, [rsi + r12]
    mov rdi, rsi
    xor ebx, ebx
p_loop:
    cmp rsi, r13
    jae p_end
    cmp byte ptr [rsi], 10
    jne p_next
    mov qword ptr [starts + rbx*8], rdi
    mov rax, rsi
    sub rax, rdi
    mov qword ptr [lens + rbx*8], rax
    mov dword ptr [idxA + rbx*4], ebx
    inc rbx
    lea rdi, [rsi + 1]
p_next:
    inc rsi
    jmp p_loop
p_end:
    cmp rdi, r13
    jae parsed
    mov qword ptr [starts + rbx*8], rdi
    mov rax, r13
    sub rax, rdi
    mov qword ptr [lens + rbx*8], rax
    mov dword ptr [idxA + rbx*4], ebx
    inc rbx
parsed:
    mov r10, rbx
    test r10, r10
    jz do_exit
    lea r8, [rip + idxA]
    lea r9, [rip + idxB]
    mov r11d, 1
w_loop:
    cmp r11, r10
    jae sorted
    xor r12d, r12d
l_loop:
    cmp r12, r10
    jae l_end
    lea r13, [r12 + r11]
    cmp r13, r10
    cmova r13, r10
    lea r14, [r12 + r11*2]
    cmp r14, r10
    cmova r14, r10
    mov r15, r12
    mov rbx, r13
    mov rbp, r12
m_loop:
    cmp r15, r13
    jae rest_r
    cmp rbx, r14
    jae rest_l
    mov esi, dword ptr [r8 + r15*4]
    mov edi, dword ptr [r8 + rbx*4]
    call line_le
    test al, al
    jz take_r
    mov eax, dword ptr [r8 + r15*4]
    mov dword ptr [r9 + rbp*4], eax
    inc r15
    inc rbp
    jmp m_loop
take_r:
    mov eax, dword ptr [r8 + rbx*4]
    mov dword ptr [r9 + rbp*4], eax
    inc rbx
    inc rbp
    jmp m_loop
rest_l:
    cmp r15, r13
    jae m_done
    mov eax, dword ptr [r8 + r15*4]
    mov dword ptr [r9 + rbp*4], eax
    inc r15
    inc rbp
    jmp rest_l
rest_r:
    cmp rbx, r14
    jae m_done
    mov eax, dword ptr [r8 + rbx*4]
    mov dword ptr [r9 + rbp*4], eax
    inc rbx
    inc rbp
    jmp rest_r
m_done:
    lea r12, [r12 + r11*2]
    jmp l_loop
l_end:
    xchg r8, r9
    shl r11, 1
    jmp w_loop
sorted:
    lea rdi, [rip + outbuf]
    xor r12d, r12d
o_loop:
    mov eax, dword ptr [r8 + r12*4]
    mov rsi, qword ptr [starts + rax*8]
    mov rcx, qword ptr [lens + rax*8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r12
    cmp r12, r10
    jb o_loop
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
w_out:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jnz w_out
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

# line_le: esi = left index, edi = right index; al = 1 if left <= right
line_le:
    mov rcx, qword ptr [lens + rsi*8]
    mov rdx, qword ptr [lens + rdi*8]
    mov rsi, qword ptr [starts + rsi*8]
    mov rdi, qword ptr [starts + rdi*8]
cl_loop:
    test rcx, rcx
    jz cl_one
    test rdx, rdx
    jz cl_zero
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jb cl_one
    ja cl_zero
    inc rsi
    inc rdi
    dec rcx
    dec rdx
    jmp cl_loop
cl_one:
    mov eax, 1
    ret
cl_zero:
    xor eax, eax
    ret

.bss
.balign 16
buf:
    .zero 40976
.balign 16
starts:
    .zero 8032
.balign 16
lens:
    .zero 8032
.balign 16
idxA:
    .zero 4032
.balign 16
idxB:
    .zero 4032
.balign 16
outbuf:
    .zero 34000
