.globl _start

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 69632
outbuf:
    .zero 340000

.text
_start:
    lea r8, [rip + hexdig]
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 69632
    sub rdx, r12
    jz read_done
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae rows_done
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    mov eax, r13d
    call hex8
    mov al, ' '
    stosb
    stosb
    lea rsi, [rip + inbuf]
    add rsi, r13
    xor ebx, ebx
hex_loop:
    cmp rbx, r14
    jae pad
    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [r8 + rax]
    stosb
    and edx, 15
    mov al, byte ptr [r8 + rdx]
    stosb
    mov al, ' '
    stosb
    jmp hex_next
pad:
    mov al, ' '
    stosb
    stosb
    stosb
hex_next:
    cmp ebx, 7
    jne 2f
    mov al, ' '
    stosb
2:
    inc ebx
    cmp ebx, 16
    jb hex_loop
    mov al, ' '
    stosb
    mov al, '|'
    stosb
    xor ebx, ebx
chr_loop:
    cmp rbx, r14
    jae chr_done
    movzx eax, byte ptr [rsi + rbx]
    lea ecx, [rax - 0x20]
    cmp ecx, 0x5E
    jbe 3f
    mov al, '.'
3:
    stosb
    inc ebx
    jmp chr_loop
chr_done:
    mov al, '|'
    stosb
    mov al, 10
    stosb
    add r13, r14
    jmp row_loop
rows_done:
    mov eax, r12d
    call hex8
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov r15, rdx
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    mov rdx, r15
    sub rdx, rax
    jmp write_loop
finish:
    xor edi, edi
    mov eax, 60
    syscall

hex8:
    mov edx, eax
    mov ecx, 8
1:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [r8 + rax]
    stosb
    dec ecx
    jnz 1b
    ret
