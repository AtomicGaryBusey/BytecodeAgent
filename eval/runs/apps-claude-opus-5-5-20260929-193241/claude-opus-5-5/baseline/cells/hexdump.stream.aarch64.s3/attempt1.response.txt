.globl _start

.equ INCAP, 65600

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65664
.balign 16
outbuf:
    .zero 340000

.text

// put_hex8: x0 = value, writes 8 hex digits at x21, advances x21
put_hex8:
    mov x9, #28
1:
    lsr x10, x0, x9
    and x10, x10, #15
    ldrb w10, [x23, x10]
    strb w10, [x21], #1
    subs x9, x9, #4
    b.ge 1b
    ret

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x23, hexdig
    add x23, x23, :lo12:hexdig
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #(INCAP & 0xffff)
    movk x2, #(INCAP >> 16), lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x22, #0              // offset
    mov w24, #' '
row_loop:
    cmp x22, x20
    b.ge rows_done
    sub x25, x20, x22
    cmp x25, #16
    mov x9, #16
    csel x25, x25, x9, lt    // n
    mov x0, x22
    bl put_hex8
    strb w24, [x21], #1
    strb w24, [x21], #1
    add x26, x19, x22        // row ptr
    mov x27, #0              // i
hex_loop:
    cmp x27, x25
    b.ge hex_blank
    ldrb w10, [x26, x27]
    lsr w11, w10, #4
    ldrb w11, [x23, x11]
    strb w11, [x21], #1
    and w10, w10, #15
    ldrb w10, [x23, x10]
    strb w10, [x21], #1
    strb w24, [x21], #1
    b hex_next
hex_blank:
    strb w24, [x21], #1
    strb w24, [x21], #1
    strb w24, [x21], #1
hex_next:
    cmp x27, #7
    b.ne 2f
    strb w24, [x21], #1
2:
    add x27, x27, #1
    cmp x27, #16
    b.lt hex_loop
    strb w24, [x21], #1
    mov w10, #'|'
    strb w10, [x21], #1
    mov x27, #0
    mov w12, #'.'
chr_loop:
    ldrb w10, [x26, x27]
    sub w11, w10, #0x20
    cmp w11, #0x5e
    csel w10, w10, w12, ls
    strb w10, [x21], #1
    add x27, x27, #1
    cmp x27, x25
    b.lt chr_loop
    mov w10, #'|'
    strb w10, [x21], #1
    mov w10, #10
    strb w10, [x21], #1
    add x22, x22, x25
    b row_loop
rows_done:
    mov x0, x20
    bl put_hex8
    mov w10, #10
    strb w10, [x21], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
write_loop:
    sub x2, x21, x1
    cbz x2, finish
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x1, x1, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish
