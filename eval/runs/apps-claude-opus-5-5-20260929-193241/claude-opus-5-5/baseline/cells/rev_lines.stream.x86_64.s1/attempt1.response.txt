.globl _start

.text

_start:
    lea rbx, [rip + inbuf]
    xor r12, r12

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r12]
    mov edx, 70000
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + outbuf]
    xor r14, r14
    xor ecx, ecx

scan_loop:
    cmp rcx, r12
    je at_end
    cmp byte ptr [rbx + rcx], 10
    jne scan_next
    call emit

scan_next:
    inc rcx
    jmp scan_loop

at_end:
    cmp r14, r12
    je do_output
    call emit

do_output:
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    push rdx
    push rsi
    syscall
    pop rsi
    pop rdx
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

# emit: reverse bytes [r14, rcx) of inbuf into out at r13, append newline, set r14 = rcx + 1
emit:
    mov rdx, rcx
1:
    cmp rdx, r14
    jbe 2f
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [r13], al
    inc r13
    jmp 1b
2:
    mov byte ptr [r13], 10
    inc r13
    lea r14, [rcx + 1]
    ret

.bss
.balign 16
inbuf:
    .zero 70016
outbuf:
    .zero 70016
