.globl _start

.equ BUFSZ, 65600
.equ OUTSZ, 65536

.text
_start:
    xor r15d, r15d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r15
    mov edx, BUFSZ
    sub edx, r15d
    jz 2f
    syscall
    test rax, rax
    jle 2f
    add r15, rax
    jmp 1b
2:
    lea r10, [rip + buf]
    xor ecx, ecx
3:
    cmp rcx, r15
    jae 4f
    cmp byte ptr [r10 + rcx], 0x21
    je 4f
    inc rcx
    jmp 3b
4:
    mov r11, rcx
    lea rax, [rcx + 1]
    cmp rax, r15
    jbe 5f
    mov rax, r15
5:
    lea r12, [r10 + rax]
    lea r13, [r10 + r15]

    lea rsi, [rip + code]
    lea r9, [rip + match]
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d
cloop:
    cmp rcx, r11
    jae cdone
    movzx eax, byte ptr [r10 + rcx]
    inc rcx
    cmp al, '+'
    je cstore
    cmp al, '-'
    je cstore
    cmp al, '<'
    je cstore
    cmp al, '>'
    je cstore
    cmp al, '.'
    je cstore
    cmp al, ','
    je cstore
    cmp al, '['
    je copen
    cmp al, ']'
    je cclose
    jmp cloop
copen:
    push rdx
    inc r8
    jmp cstore
cclose:
    test r8, r8
    jz fail
    dec r8
    pop rdi
    mov dword ptr [r9 + rdx*4], edi
    mov dword ptr [r9 + rdi*4], edx
cstore:
    mov byte ptr [rsi + rdx], al
    inc rdx
    jmp cloop
cdone:
    test r8, r8
    jnz fail
    mov byte ptr [rsi + rdx], 0

    lea rdi, [rip + tape]
    lea r14, [rip + outbuf]
    xor ebx, ebx
run:
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '+'
    je op_inc
    cmp al, '-'
    je op_dec
    cmp al, '>'
    je op_right
    cmp al, '<'
    je op_left
    cmp al, '['
    je op_open
    cmp al, ']'
    je op_close
    cmp al, '.'
    je op_out
    cmp al, ','
    je op_in
    jmp finish
op_inc:
    inc byte ptr [rdi]
    jmp next
op_dec:
    dec byte ptr [rdi]
    jmp next
op_right:
    inc rdi
    jmp next
op_left:
    dec rdi
    jmp next
op_open:
    cmp byte ptr [rdi], 0
    jne next
    mov ebx, dword ptr [r9 + rbx*4]
    jmp next
op_close:
    cmp byte ptr [rdi], 0
    je next
    mov ebx, dword ptr [r9 + rbx*4]
    jmp next
op_out:
    mov al, byte ptr [rdi]
    mov byte ptr [r14], al
    inc r14
    lea rax, [rip + outbuf + OUTSZ]
    cmp r14, rax
    jb next
    call flush
    jmp next
op_in:
    xor eax, eax
    cmp r12, r13
    jae 6f
    mov al, byte ptr [r12]
    inc r12
6:
    mov byte ptr [rdi], al
next:
    inc rbx
    jmp run

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    push rsi
    push rdi
    lea rsi, [rip + outbuf]
7:
    mov rdx, r14
    sub rdx, rsi
    jle 8f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 8f
    add rsi, rax
    jmp 7b
8:
    lea r14, [rip + outbuf]
    pop rdi
    pop rsi
    ret

.bss
.balign 64
buf:
    .zero BUFSZ
.balign 64
code:
    .zero 65600
.balign 64
match:
    .zero 262400
.balign 64
tape:
    .zero 30016
.balign 64
outbuf:
    .zero OUTSZ
