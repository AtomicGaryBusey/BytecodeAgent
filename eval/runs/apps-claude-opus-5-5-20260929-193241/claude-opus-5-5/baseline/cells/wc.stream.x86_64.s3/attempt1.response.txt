.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
readloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
bloop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15, r15
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    dec rcx
    jnz bloop
    jmp readloop

done:
    lea rdi, [rip + outbuf + 96]
    mov rbx, rdi
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call utoa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call utoa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call utoa
    mov rsi, rdi
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

utoa:
    mov r8d, 10
2:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    ret

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
