.globl _start

.equ INSZ, 65544

.text
_start:
    xor r12d, r12d
read_loop:
    mov edx, INSZ
    sub rdx, r12
    jle read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor ebx, ebx
    xor r13d, r13d
line_loop:
    cmp rbx, r12
    jae output
    mov rcx, rbx
find_end:
    cmp rcx, r12
    jae found_end
    cmp byte ptr [rsi + rcx], 10
    je found_end
    inc rcx
    jmp find_end
found_end:
    mov rdx, rcx
rev_loop:
    cmp rdx, rbx
    jbe rev_done
    dec rdx
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi + r13], al
    inc r13
    jmp rev_loop
rev_done:
    mov byte ptr [rdi + r13], 10
    inc r13
    lea rbx, [rcx + 1]
    jmp line_loop
output:
    lea r14, [rip + outbuf]
write_loop:
    test r13, r13
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r13
    syscall
    test rax, rax
    jle finish
    add r14, rax
    sub r13, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero 65552
outbuf:
    .zero 65552
