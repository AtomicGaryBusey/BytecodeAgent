.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 64
inbuf:
    .zero 65536
outbuf:
    .zero 98304

.text
_start:
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub edx, ebx
    syscall
    test rax, rax
    jle done_read
    add rbx, rax
    jmp read_loop

done_read:
    lea rsi, [rip + inbuf]
    mov rcx, rbx
    lea rdi, [rip + outbuf]
    xor r8d, r8d
    lea r9, [rip + alphabet]

g3:
    cmp rcx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3
    sub rcx, 3
    mov r10d, eax
    mov r11d, 18
    call enc
    mov r11d, 12
    call enc
    mov r11d, 6
    call enc
    xor r11d, r11d
    call enc
    jmp g3

tail:
    test rcx, rcx
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 1
    je tail1
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov r10d, eax
    mov r11d, 18
    call enc
    mov r11d, 12
    call enc
    mov r11d, 6
    call enc
    mov al, '='
    call put
    jmp finish
tail1:
    mov r10d, eax
    mov r11d, 18
    call enc
    mov r11d, 12
    call enc
    mov al, '='
    call put
    mov al, '='
    call put

finish:
    test r8, r8
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jz do_exit
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rbx, rax
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

enc:
    shrx eax, r10d, r11d
    and eax, 63
    movzx eax, byte ptr [r9 + rax]
put:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:
    ret
