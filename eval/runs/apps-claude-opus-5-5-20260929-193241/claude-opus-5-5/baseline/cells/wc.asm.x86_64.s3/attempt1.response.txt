.globl _start

.bss
.balign 16
buf:
    .zero 65536
tmp:
    .zero 32
outbuf:
    .zero 96

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rbx, rax
    xor ecx, ecx
byte_loop:
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    inc rcx
    cmp rcx, rbx
    jb byte_loop
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

utoa:
    lea r8, [rip + tmp + 24]
    mov r9, r8
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r9
    jb 2b
    ret
