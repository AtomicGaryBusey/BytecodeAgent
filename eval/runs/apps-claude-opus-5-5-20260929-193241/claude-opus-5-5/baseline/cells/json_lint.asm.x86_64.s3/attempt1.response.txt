.globl _start

.section .rodata
lit_true:  .byte 0x74,0x72,0x75,0x65,0
lit_false: .byte 0x66,0x61,0x6c,0x73,0x65,0
lit_null:  .byte 0x6e,0x75,0x6c,0x6c,0
esc_chars: .byte 0x22,0x5c,0x2f,0x62,0x66,0x6e,0x72,0x74,0
msg_ok: .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err: .ascii "error at "
msg_err_len = . - msg_err

.bss
.balign 16
buf: .zero 70000
numbuf: .zero 32

.text

_start:
    xor r14d, r14d
read_loop:
    mov edx, 70000
    sub rdx, r14
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r14
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    lea r13, [rip + buf]
    lea r12, [r13 + r14]
    mov rsi, r13
    call skip_ws
    call value
    call skip_ws
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rbx, rsi
    sub rbx, r13
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, msg_err_len
    syscall
    lea r8, [rip + numbuf + 31]
    mov byte ptr [r8], 10
    mov rdi, r8
    mov rax, rbx
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov rsi, rdi
    lea rdx, [r8 + 1]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

# eax = byte at rsi or 256 at end
peek:
    cmp rsi, r12
    jae 1f
    movzx eax, byte ptr [rsi]
    ret
1:
    mov eax, 256
    ret

skip_ws:
1:
    call peek
    cmp eax, 0x20
    je 2f
    cmp eax, 0x09
    je 2f
    cmp eax, 0x0a
    je 2f
    cmp eax, 0x0d
    je 2f
    ret
2:
    inc rsi
    jmp 1b

value:
    call peek
    cmp eax, 0x7b
    je object
    cmp eax, 0x5b
    je array
    cmp eax, 0x22
    je string
    cmp eax, 0x2d
    je number
    lea ecx, [rax - 0x30]
    cmp ecx, 9
    jbe number
    cmp eax, 0x74
    je v_true
    cmp eax, 0x66
    je v_false
    cmp eax, 0x6e
    je v_null
    jmp error
v_true:
    lea rdi, [rip + lit_true]
    jmp literal
v_false:
    lea rdi, [rip + lit_false]
    jmp literal
v_null:
    lea rdi, [rip + lit_null]
literal:
    movzx edx, byte ptr [rdi]
    test edx, edx
    jz 1f
    call peek
    cmp eax, edx
    jne error
    inc rsi
    inc rdi
    jmp literal
1:
    ret

string:
    inc rsi
s_loop:
    call peek
    cmp eax, 256
    je error
    cmp eax, 0x22
    je s_end
    cmp eax, 0x20
    jb error
    cmp eax, 0x5c
    je s_esc
    inc rsi
    jmp s_loop
s_end:
    inc rsi
    ret
s_esc:
    inc rsi
    call peek
    cmp eax, 0x75
    je s_u
    lea rdi, [rip + esc_chars]
1:
    movzx edx, byte ptr [rdi]
    test edx, edx
    jz error
    cmp eax, edx
    je 2f
    inc rdi
    jmp 1b
2:
    inc rsi
    jmp s_loop
s_u:
    inc rsi
    mov r8d, 4
3:
    call peek
    lea ecx, [rax - 0x30]
    cmp ecx, 9
    jbe 4f
    mov ecx, eax
    or ecx, 0x20
    sub ecx, 0x61
    cmp ecx, 5
    ja error
4:
    inc rsi
    dec r8d
    jnz 3b
    jmp s_loop

number:
    call peek
    cmp eax, 0x2d
    jne 1f
    inc rsi
    call peek
1:
    cmp eax, 0x30
    je n_zero
    lea ecx, [rax - 0x31]
    cmp ecx, 8
    ja error
    inc rsi
    call digits
    jmp n_frac
n_zero:
    inc rsi
n_frac:
    call peek
    cmp eax, 0x2e
    jne n_exp
    inc rsi
    call peek
    lea ecx, [rax - 0x30]
    cmp ecx, 9
    ja error
    call digits
n_exp:
    call peek
    cmp eax, 0x65
    je 2f
    cmp eax, 0x45
    je 2f
    ret
2:
    inc rsi
    call peek
    cmp eax, 0x2b
    je 3f
    cmp eax, 0x2d
    jne 4f
3:
    inc rsi
    call peek
4:
    lea ecx, [rax - 0x30]
    cmp ecx, 9
    ja error
    call digits
    ret

digits:
1:
    call peek
    lea ecx, [rax - 0x30]
    cmp ecx, 9
    ja 2f
    inc rsi
    jmp 1b
2:
    ret

array:
    inc rsi
    call skip_ws
    call peek
    cmp eax, 0x5d
    je a_end
a_loop:
    call value
    call skip_ws
    call peek
    cmp eax, 0x2c
    je a_comma
    cmp eax, 0x5d
    je a_end
    jmp error
a_comma:
    inc rsi
    call skip_ws
    jmp a_loop
a_end:
    inc rsi
    ret

object:
    inc rsi
    call skip_ws
    call peek
    cmp eax, 0x7d
    je o_end
o_loop:
    call peek
    cmp eax, 0x22
    jne error
    call string
    call skip_ws
    call peek
    cmp eax, 0x3a
    jne error
    inc rsi
    call skip_ws
    call value
    call skip_ws
    call peek
    cmp eax, 0x2c
    je o_comma
    cmp eax, 0x7d
    je o_end
    jmp error
o_comma:
    inc rsi
    call skip_ws
    jmp o_loop
o_end:
    inc rsi
    ret
