.section .rodata

    .ascii "0123456789abcdef"

    .zero 65552
.balign 16

    .zero 8192

.text

    add x23, x23, :lo12:outbuf
    mov x22, x23
    adrp x24, hexd
    add x24, x24, :lo12:hexd

    sub x25, x20, x21
    mov x9, #16
    cmp x25, #16

    mov w0, w21
    bl hex8
    mov w15, #' '
    strb w15, [x22], #1
    strb w15, [x22], #1

    b.hs pad
    add x11, x21, x10

    ldrb w13, [x24, x13]
    strb w13, [x22], #1

    strb w15, [x22], #1
    strb w15, [x22], #1

    b.ne 1f

    cmp x10, #16
    b.lo hexloop
    strb w15, [x22], #1
    mov w9, #'|'
    strb w9, [x22], #1

    strb w9, [x22], #1
    mov w9, #10
    strb w9, [x22], #1
    add x21, x21, x25
    sub x9, x22, x23
    cmp x9, #4000

    bl hex8
    mov w9, #10
    strb w9, [x22], #1

    mov x8, #93

    cmp x2, #0
    b.le 3f
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 3f
    add x1, x1, x0
    b 2b

    mov x22, x23
    ret
