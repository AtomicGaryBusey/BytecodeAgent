.globl _start

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 331776

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, hexdig
    add x22, x22, :lo12:hexdig

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x23, #0

row_loop:
    cmp x23, x20
    b.hs final
    sub x24, x20, x23
    cmp x24, #16
    b.ls 1f
    mov x24, #16
1:
    mov x0, x23
    bl put8hex
    mov w5, #' '
    strb w5, [x21], #1
    strb w5, [x21], #1
    mov x25, #0
hex_loop:
    cmp x25, x24
    b.hs 2f
    add x6, x23, x25
    ldrb w6, [x19, x6]
    lsr w7, w6, #4
    ldrb w7, [x22, x7]
    strb w7, [x21], #1
    and w7, w6, #15
    ldrb w7, [x22, x7]
    strb w7, [x21], #1
    mov w5, #' '
    strb w5, [x21], #1
    b 3f
2:
    mov w5, #' '
    strb w5, [x21], #1
    strb w5, [x21], #1
    strb w5, [x21], #1
3:
    cmp x25, #7
    b.ne 4f
    mov w5, #' '
    strb w5, [x21], #1
4:
    add x25, x25, #1
    cmp x25, #16
    b.lo hex_loop

    mov w5, #' '
    strb w5, [x21], #1
    mov w5, #'|'
    strb w5, [x21], #1
    mov x25, #0
chr_loop:
    add x6, x23, x25
    ldrb w6, [x19, x6]
    cmp w6, #0x20
    b.lo 5f
    cmp w6, #0x7e
    b.hi 5f
    b 6f
5:
    mov w6, #'.'
6:
    strb w6, [x21], #1
    add x25, x25, #1
    cmp x25, x24
    b.lo chr_loop
    mov w5, #'|'
    strb w5, [x21], #1
    mov w5, #10
    strb w5, [x21], #1
    add x23, x23, #16
    b row_loop

final:
    mov x0, x20
    bl put8hex
    mov w5, #10
    strb w5, [x21], #1

    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1
write_loop:
    cbz x2, do_exit
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b do_exit

put8hex:
    mov w3, #28
7:
    lsr x4, x0, x3
    and x4, x4, #15
    ldrb w4, [x22, x4]
    strb w4, [x21], #1
    subs w3, w3, #4
    b.ge 7b
    ret
