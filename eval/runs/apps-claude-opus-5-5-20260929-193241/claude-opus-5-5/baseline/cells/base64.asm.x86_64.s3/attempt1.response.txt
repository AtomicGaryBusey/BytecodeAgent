.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 90112

.text
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r8, [rip + inbuf]
    mov r9, rbx
    lea rdi, [rip + outbuf]
    xor r10d, r10d
    lea r11, [rip + alpha]
loop3:
    cmp r9, 3
    jb tail
    movzx eax, byte ptr [r8]
    shl eax, 16
    movzx ecx, byte ptr [r8 + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [r8 + 2]
    or eax, ecx
    add r8, 3
    sub r9, 3
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit
    mov edx, eax
    and edx, 63
    call emit
    jmp loop3
tail:
    test r9, r9
    jz finish
    movzx eax, byte ptr [r8]
    shl eax, 16
    cmp r9, 2
    jne tail1
    movzx ecx, byte ptr [r8 + 1]
    shl ecx, 8
    or eax, ecx
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 6
    and edx, 63
    call emit
    mov cl, 61
    call emitc
    jmp finish
tail1:
    mov edx, eax
    shr edx, 18
    and edx, 63
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    call emit
    mov cl, 61
    call emitc
    mov cl, 61
    call emitc
finish:
    test r10, r10
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    push rdx
    push rsi
    syscall
    pop rsi
    pop rdx
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit:
    movzx ecx, byte ptr [r11 + rdx]
emitc:
    mov byte ptr [rdi], cl
    inc rdi
    inc r10
    cmp r10, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r10d, r10d
1:
    ret
