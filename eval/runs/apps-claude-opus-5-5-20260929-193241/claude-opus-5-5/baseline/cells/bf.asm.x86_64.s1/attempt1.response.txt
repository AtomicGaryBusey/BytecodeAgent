.globl _start

.equ INSZ, 65600
.equ OUTSZ, 65536

.text
_start:
    xor ebx, ebx
rd_loop:
    mov edx, INSZ
    sub edx, ebx
    jz rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd_loop
rd_done:
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae not_found
    cmp byte ptr [inbuf + rcx], 0x21
    je found
    inc rcx
    jmp find_loop
not_found:
    mov r12, rcx
    mov r13, rbx
    jmp compile
found:
    mov r12, rcx
    lea r13, [rcx + 1]
compile:
    mov r14, rbx
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d
c_loop:
    cmp rcx, r12
    jae c_end
    movzx eax, byte ptr [inbuf + rcx]
    inc rcx
    cmp al, '['
    je c_open
    cmp al, ']'
    je c_close
    cmp al, '+'
    je c_store
    cmp al, '-'
    je c_store
    cmp al, '<'
    je c_store
    cmp al, '>'
    je c_store
    cmp al, '.'
    je c_store
    cmp al, ','
    je c_store
    jmp c_loop
c_store:
    mov byte ptr [code + rdx], al
    inc rdx
    jmp c_loop
c_open:
    mov byte ptr [code + rdx], al
    mov dword ptr [stk + r8*4], edx
    inc r8
    inc rdx
    jmp c_loop
c_close:
    test r8, r8
    jz bad
    dec r8
    mov r9d, dword ptr [stk + r8*4]
    mov byte ptr [code + rdx], al
    mov dword ptr [match + rdx*4], r9d
    mov dword ptr [match + r9*4], edx
    inc rdx
    jmp c_loop
c_end:
    test r8, r8
    jnz bad

    mov rbp, rdx
    xor r12d, r12d
    lea rdi, [rip + tape]
    xor r15d, r15d
run:
    cmp r12, rbp
    jae finish
    movzx eax, byte ptr [code + r12]
    cmp al, '+'
    je op_inc
    cmp al, '-'
    je op_dec
    cmp al, '>'
    je op_right
    cmp al, '<'
    je op_left
    cmp al, '['
    je op_open
    cmp al, ']'
    je op_close
    cmp al, '.'
    je op_out
    jmp op_in
op_inc:
    inc byte ptr [rdi]
    inc r12
    jmp run
op_dec:
    dec byte ptr [rdi]
    inc r12
    jmp run
op_right:
    inc rdi
    inc r12
    jmp run
op_left:
    dec rdi
    inc r12
    jmp run
op_open:
    cmp byte ptr [rdi], 0
    jne next
    mov r12d, dword ptr [match + r12*4]
    jmp next
op_close:
    cmp byte ptr [rdi], 0
    je next
    mov r12d, dword ptr [match + r12*4]
    jmp next
op_out:
    mov al, byte ptr [rdi]
    mov byte ptr [outbuf + r15], al
    inc r15
    cmp r15, OUTSZ
    jb next
    call flush
    jmp next
op_in:
    cmp r13, r14
    jae in_zero
    mov al, byte ptr [inbuf + r13]
    inc r13
    mov byte ptr [rdi], al
    jmp next
in_zero:
    mov byte ptr [rdi], 0
next:
    inc r12
    jmp run

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

bad:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    push rdi
    xor ebx, ebx
fl_loop:
    cmp rbx, r15
    jae fl_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r15
    sub rdx, rbx
    syscall
    test rax, rax
    jle fl_done
    add rbx, rax
    jmp fl_loop
fl_done:
    xor r15d, r15d
    pop rdi
    ret

.bss
.balign 16
inbuf:
    .zero 65616
code:
    .zero 65616
.balign 16
match:
    .zero 262400
stk:
    .zero 262400
tape:
    .zero 30016
outbuf:
    .zero 65552
