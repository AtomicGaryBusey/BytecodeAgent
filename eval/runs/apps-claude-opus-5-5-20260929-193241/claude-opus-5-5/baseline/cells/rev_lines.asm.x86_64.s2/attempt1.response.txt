.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65600
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor ecx, ecx
line_loop:
    cmp rcx, rbx
    jae flush
    mov rdx, rcx
scan:
    cmp rdx, rbx
    jae found
    cmp byte ptr [rsi + rdx], 10
    je found
    inc rdx
    jmp scan
found:
    mov r8, rdx
rev:
    cmp r8, rcx
    jbe end_rev
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev
end_rev:
    mov byte ptr [rdi], 10
    inc rdi
    lea rcx, [rdx + 1]
    jmp line_loop

flush:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65664
outbuf:
    .zero 65664
