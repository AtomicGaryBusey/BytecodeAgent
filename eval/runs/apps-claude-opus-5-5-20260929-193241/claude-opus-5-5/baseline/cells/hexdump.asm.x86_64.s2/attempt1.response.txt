.globl _start

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 69632
outbuf:
    .zero 340000

.text

# hex8: write eax as 8 hex digits at rdi, advance rdi. clobbers ecx, edx
hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rip + hexdig + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret

_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 69632
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    lea r14, [rip + inbuf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae rows_done
    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe 2f
    mov r15d, 16
2:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor ebx, ebx
hex_loop:
    cmp rbx, r15
    jae 3f
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rip + hexdig + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rip + hexdig + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    jmp 4f
3:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
4:
    cmp ebx, 7
    jne 5f
    mov byte ptr [rdi], ' '
    inc rdi
5:
    inc ebx
    cmp ebx, 16
    jb hex_loop
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor ebx, ebx
chr_loop:
    lea rsi, [r14 + r13]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb 6f
    cmp al, 0x7e
    jbe 7f
6:
    mov al, '.'
7:
    mov byte ptr [rdi], al
    inc rdi
    inc rbx
    cmp rbx, r15
    jb chr_loop
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r15
    jmp row_loop
rows_done:
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jz done
    mov edi, 1
    mov rdx, rbx
    mov eax, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rbx, rax
    jmp write_loop
done:
    xor edi, edi
    mov eax, 60
    syscall
