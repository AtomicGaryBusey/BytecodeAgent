.globl _start

.bss
.balign 16
buf:    .zero 65536
tmp:    .zero 32
outbuf: .zero 96

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rbx, [rip + buf]
    lea rbp, [rbx + rax]
byte_loop:
    movzx eax, byte ptr [rbx]
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je is_ws
    sub eax, 9
    cmp eax, 4
    jbe is_ws
    test r15, r15
    jnz next
    inc r13
    mov r15d, 1
    jmp next
is_ws:
    xor r15d, r15d
next:
    inc rbx
    cmp rbx, rbp
    jb byte_loop
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_num:
    lea r8, [rip + tmp + 24]
    mov rsi, r8
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    inc rsi
    cmp rsi, r8
    jb 2b
    ret
