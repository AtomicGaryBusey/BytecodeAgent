.globl _start
.text
_start:
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65552
    sub edx, ebx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r8, [rsi + rbx]
    lea rdi, [rip + outbuf]
outer:
    cmp rsi, r8
    jae output
    mov rdx, rsi
scan:
    cmp rdx, r8
    jae found
    cmp byte ptr [rdx], 10
    je found
    inc rdx
    jmp scan
found:
    mov rcx, rdx
rev:
    cmp rcx, rsi
    jbe endrev
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev
endrev:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rdx + 1]
    jmp outer
output:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65552
outbuf:
    .zero 65552
