.globl _start

.equ BUFCAP, 0x10000

.section .rodata
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err:
    .ascii "error at "
msg_err_len = . - msg_err

.bss
.balign 16
buf:
    .zero 65600
numbuf:
    .zero 32

.text

_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
read_loop:
    mov x2, #BUFCAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    strb wzr, [x19, x20]
    add x21, x19, x20
    mov x20, x19

    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne fail

    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #msg_ok_len
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

fail:
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #msg_err_len
    mov x8, #64
    svc #0
    sub x0, x20, x19
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #31
    mov w3, #0x0A
    strb w3, [x1]
    mov x4, #10
1:
    udiv x2, x0, x4
    msub x3, x2, x4, x0
    add w3, w3, #0x30
    strb w3, [x1, #-1]!
    mov x0, x2
    cbnz x0, 1b
    adrp x2, numbuf
    add x2, x2, :lo12:numbuf
    add x2, x2, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip whitespace (leaf)
skip_ws:
1:
    ldrb w0, [x20]
    cmp w0, #0x20
    b.eq 2f
    cmp w0, #0x09
    b.eq 2f
    cmp w0, #0x0A
    b.eq 2f
    cmp w0, #0x0D
    b.eq 2f
    ret
2:
    add x20, x20, #1
    b 1b

// parse string starting at '"' (leaf)
parse_string:
    add x20, x20, #1
ps_loop:
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq ps_end
    cmp w0, #0x20
    b.lo fail
    cmp w0, #0x5C
    b.eq ps_esc
    add x20, x20, #1
    b ps_loop
ps_esc:
    add x20, x20, #1
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq ps_ok
    cmp w0, #0x5C
    b.eq ps_ok
    cmp w0, #0x2F
    b.eq ps_ok
    cmp w0, #0x62
    b.eq ps_ok
    cmp w0, #0x66
    b.eq ps_ok
    cmp w0, #0x6E
    b.eq ps_ok
    cmp w0, #0x72
    b.eq ps_ok
    cmp w0, #0x74
    b.eq ps_ok
    cmp w0, #0x75
    b.ne fail
    add x20, x20, #1
    mov w3, #4
ps_hex:
    ldrb w0, [x20]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls ps_hok
    orr w1, w0, #0x20
    sub w1, w1, #0x61
    cmp w1, #5
    b.hi fail
ps_hok:
    add x20, x20, #1
    subs w3, w3, #1
    b.ne ps_hex
    b ps_loop
ps_ok:
    add x20, x20, #1
    b ps_loop
ps_end:
    add x20, x20, #1
    ret

// parse number (leaf)
parse_number:
    ldrb w0, [x20]
    cmp w0, #0x2D
    b.ne 1f
    add x20, x20, #1
    ldrb w0, [x20]
1:
    cmp w0, #0x30
    b.eq pn_zero
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
2:
    add x20, x20, #1
    ldrb w0, [x20]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 2b
    b pn_frac
pn_zero:
    add x20, x20, #1
    ldrb w0, [x20]
pn_frac:
    cmp w0, #0x2E
    b.ne pn_exp
    add x20, x20, #1
    ldrb w0, [x20]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
3:
    add x20, x20, #1
    ldrb w0, [x20]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 3b
pn_exp:
    orr w1, w0, #0x20
    cmp w1, #0x65
    b.ne pn_ret
    add x20, x20, #1
    ldrb w0, [x20]
    cmp w0, #0x2B
    b.eq 4f
    cmp w0, #0x2D
    b.ne 5f
4:
    add x20, x20, #1
    ldrb w0, [x20]
5:
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
6:
    add x20, x20, #1
    ldrb w0, [x20]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 6b
pn_ret:
    ret

// parse value (recursive)
parse_value:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x20]
    cmp w0, #0x7B
    b.eq pv_obj
    cmp w0, #0x5B
    b.eq pv_arr
    cmp w0, #0x22
    b.eq pv_str
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6E
    b.eq pv_null
    cmp w0, #0x2D
    b.eq pv_num
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pv_num
    b fail
pv_ret:
    ldp x29, x30, [sp], #16
    ret
pv_str:
    bl parse_string
    b pv_ret
pv_num:
    bl parse_number
    b pv_ret
pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    b pv_lit
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    b pv_lit
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
pv_lit:
    ldrb w2, [x1], #1
    cbz w2, pv_ret
    ldrb w3, [x20]
    cmp w2, w3
    b.ne fail
    add x20, x20, #1
    b pv_lit

pv_arr:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x20]
    cmp w0, #0x5D
    b.eq pv_arr_close
pv_arr_loop:
    bl parse_value
    bl skip_ws
    ldrb w0, [x20]
    cmp w0, #0x2C
    b.ne 1f
    add x20, x20, #1
    bl skip_ws
    b pv_arr_loop
1:
    cmp w0, #0x5D
    b.ne fail
pv_arr_close:
    add x20, x20, #1
    b pv_ret

pv_obj:
    add x20, x20, #1
    bl skip_ws
    ldrb w0, [x20]
    cmp w0, #0x7D
    b.eq pv_obj_close
pv_obj_loop:
    ldrb w0, [x20]
    cmp w0, #0x22
    b.ne fail
    bl parse_string
    bl skip_ws
    ldrb w0, [x20]
    cmp w0, #0x3A
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    ldrb w0, [x20]
    cmp w0, #0x2C
    b.ne 1f
    add x20, x20, #1
    bl skip_ws
    b pv_obj_loop
1:
    cmp w0, #0x7D
    b.ne fail
pv_obj_close:
    add x20, x20, #1
    b pv_ret
