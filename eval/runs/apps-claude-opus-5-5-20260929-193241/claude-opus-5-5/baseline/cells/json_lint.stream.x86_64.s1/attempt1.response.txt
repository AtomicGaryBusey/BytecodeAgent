.globl _start

.equ BUFSZ, 131072

.section .rodata
lit_true:  .asciz "true"
lit_false: .asciz "false"
lit_null:  .asciz "null"
msg_ok: .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err: .ascii "error at "
msg_err_len = . - msg_err

.bss
.balign 16
buf: .space BUFSZ + 16
outbuf: .space 64

.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
read_loop:
    mov edx, BUFSZ
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r15, [r12 + r13]
    mov byte ptr [r15], 0
    mov rsi, r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r15
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rax, rsi
    sub rax, r12
    lea rdi, [rip + outbuf]
    lea rsi, [rip + msg_err]
    mov ecx, msg_err_len
    xor edx, edx
1:
    mov r8b, byte ptr [rsi + rdx]
    mov byte ptr [rdi + rdx], r8b
    inc rdx
    cmp rdx, rcx
    jb 1b
    # convert rax to decimal into temp at outbuf+40 backwards
    lea r9, [rdi + 60]
    mov r10, r9
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 2b
    lea r11, [rdi + msg_err_len]
3:
    mov al, byte ptr [r9]
    mov byte ptr [r11], al
    inc r9
    inc r11
    cmp r9, r10
    jb 3b
    mov byte ptr [r11], 10
    inc r11
    mov rdx, r11
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    je 1f
    ret
1:
    inc rsi
    jmp skip_ws

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je 1f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 3f
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp parse_number
1:
    lea rdi, [rip + lit_true]
    jmp parse_literal
2:
    lea rdi, [rip + lit_false]
    jmp parse_literal
3:
    lea rdi, [rip + lit_null]
    jmp parse_literal

parse_literal:
    movzx eax, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp parse_literal
1:
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], '}'
    jne 1f
    inc rsi
    ret
1:
    cmp byte ptr [rsi], '"'
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:
    cmp al, '}'
    jne error
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], ']'
    jne 1f
    inc rsi
    ret
1:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:
    cmp al, ']'
    jne error
    inc rsi
    ret

parse_string:
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je 9f
    cmp al, 0x20
    jb error
    cmp al, '\\'
    je 2f
    inc rsi
    jmp 1b
2:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je 3f
    cmp al, '\\'
    je 3f
    cmp al, '/'
    je 3f
    cmp al, 'b'
    je 3f
    cmp al, 'f'
    je 3f
    cmp al, 'n'
    je 3f
    cmp al, 'r'
    je 3f
    cmp al, 't'
    je 3f
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
4:
    movzx eax, byte ptr [rsi]
    lea edx, [eax - '0']
    cmp edx, 9
    jbe 5f
    or eax, 0x20
    sub eax, 'a'
    cmp eax, 5
    ja error
5:
    inc rsi
    dec ecx
    jnz 4b
    jmp 1b
3:
    inc rsi
    jmp 1b
9:
    inc rsi
    ret

parse_number:
    cmp byte ptr [rsi], '-'
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp 4f
2:
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc rsi
    call skip_digits
4:
    cmp byte ptr [rsi], '.'
    jne 5f
    inc rsi
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja error
    inc rsi
    call skip_digits
5:
    movzx eax, byte ptr [rsi]
    or eax, 0x20
    cmp eax, 'e'
    jne 7f
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 6f
    cmp al, '-'
    jne 8f
6:
    inc rsi
8:
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja error
    inc rsi
    call skip_digits
7:
    ret

skip_digits:
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret
