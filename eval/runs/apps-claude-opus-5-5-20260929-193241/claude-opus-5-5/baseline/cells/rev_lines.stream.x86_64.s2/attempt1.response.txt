.globl _start

.text
_start:
    # read all input into inbuf
    xor r12d, r12d              # total bytes read
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536 + 16
    sub rdx, r12
    jbe 2f
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    # r12 = length
    lea rsi, [rip + inbuf]       # input base
    lea rdi, [rip + outbuf]      # output pointer
    xor ecx, ecx                 # current index
    xor r8d, r8d                 # line start
3:
    cmp rcx, r12
    jae 6f
    cmp byte ptr [rsi + rcx], 10
    je 4f
    inc rcx
    jmp 3b
4:
    # line is [r8, rcx)
    mov rdx, rcx
5:
    cmp rdx, r8
    jbe 7f
    dec rdx
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp 5b
7:
    mov byte ptr [rdi], 10
    inc rdi
    inc rcx
    mov r8, rcx
    jmp 3b
6:
    # final line if remainder non-empty
    cmp r8, r12
    jae 8f
    mov rdx, r12
9:
    cmp rdx, r8
    jbe 10f
    dec rdx
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp 9b
10:
    mov byte ptr [rdi], 10
    inc rdi
8:
    # write outbuf
    lea r13, [rip + outbuf]
    mov r14, rdi
11:
    mov rdx, r14
    sub rdx, r13
    jz 12f
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    syscall
    test rax, rax
    jle 12f
    add r13, rax
    jmp 11b
12:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65536 + 32
outbuf:
    .zero 65536 * 2 + 32
