.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea r8, [rsi + rax]
byte_loop:
    cmp rsi, r8
    jae read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 4
    jbe ws
    test r15, r15
    jnz byte_loop
    inc r13
    mov r15d, 1
    jmp byte_loop
ws:
    xor r15d, r15d
    jmp byte_loop
done:
    lea rdi, [rip + outbuf + 128]
    mov r9, rdi
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call conv
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call conv
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call conv
    mov rsi, rdi
    mov rdx, r9
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

conv:
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    ret

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
