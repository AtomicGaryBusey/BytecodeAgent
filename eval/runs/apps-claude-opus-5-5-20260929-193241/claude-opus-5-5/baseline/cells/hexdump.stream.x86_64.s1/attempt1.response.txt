.globl _start

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 70000
.balign 16
outbuf:
    .zero 340000

.text
.equ INSZ, 70000

_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INSZ
    sub edx, r12d
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r8, [rip + hexchars]
    lea rdi, [rip + outbuf]
    xor ebx, ebx
row_loop:
    cmp rbx, r12
    jae final
    mov r13, r12
    sub r13, rbx
    cmp r13, 16
    jbe 2f
    mov r13d, 16
2:
    lea r15, [rip + inbuf]
    add r15, rbx
    mov eax, ebx
    call puthex8
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2
    xor r14d, r14d
hex_loop:
    cmp r14, r13
    jae 3f
    movzx eax, byte ptr [r15 + r14]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp 4f
3:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
4:
    cmp r14, 7
    jne 5f
    mov byte ptr [rdi], 32
    inc rdi
5:
    inc r14
    cmp r14, 16
    jb hex_loop
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2
    xor r14d, r14d
chr_loop:
    movzx eax, byte ptr [r15 + r14]
    cmp eax, 0x20
    jb 6f
    cmp eax, 0x7e
    jbe 7f
6:
    mov eax, 46
7:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, r13
    jb chr_loop
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add rbx, r13
    jmp row_loop
final:
    mov eax, r12d
    call puthex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle exit_prog
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

puthex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret
