.globl _start

.equ INCAP, 65552
.equ OUTSZ, 458752
.equ VSTK, 1024

.text
_start:
    lea r14, [rip + inbuf]
    xor r15d, r15d
read_loop:
    mov edx, INCAP
    sub rdx, r15
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    mov rsi, r14
    lea rbp, [r14 + r15]
    lea r13, [rip + outbuf]
    lea rbx, [rip + vstack]

line_loop:
    cmp rsi, rbp
    jae all_done
    xor r12d, r12d
tok_loop:
    cmp rsi, rbp
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    jmp token
sep:
    inc rsi
    jmp tok_loop

token:
    mov rdi, rsi
scan:
    inc rsi
    cmp rsi, rbp
    jae tend
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je tend
    cmp al, 32
    je tend
    cmp al, 9
    je tend
    jmp scan
tend:
    mov rcx, rsi
    sub rcx, rdi
    movzx eax, byte ptr [rdi]
    cmp rcx, 1
    jne literal
    cmp al, '+'
    je do_op
    cmp al, '-'
    je do_op
    cmp al, '*'
    je do_op
    cmp al, '/'
    je do_op
    cmp al, '%'
    je do_op
literal:
    xor r8d, r8d
    cmp al, '-'
    jne 1f
    inc rdi
    dec rcx
    mov r8d, 1
1:
    test rcx, rcx
    jz error
    cmp rcx, 18
    ja error
    xor eax, eax
2:
    movzx edx, byte ptr [rdi]
    sub edx, '0'
    cmp edx, 9
    ja error
    imul rax, rax, 10
    add rax, rdx
    inc rdi
    dec rcx
    jnz 2b
    test r8d, r8d
    jz 3f
    neg rax
3:
    cmp r12, VSTK
    jae error
    mov qword ptr [rbx + r12*8], rax
    inc r12
    jmp tok_loop

do_op:
    mov edx, eax
    cmp r12, 2
    jb error
    mov rcx, qword ptr [rbx + r12*8 - 8]
    mov rax, qword ptr [rbx + r12*8 - 16]
    dec r12
    cmp dl, '+'
    jne 1f
    add rax, rcx
    jmp op_store
1:
    cmp dl, '-'
    jne 2f
    sub rax, rcx
    jmp op_store
2:
    cmp dl, '*'
    jne 3f
    imul rax, rcx
    jmp op_store
3:
    mov r9d, edx
    test rcx, rcx
    jz error
    cmp rcx, -1
    jne 4f
    mov r8, 0x8000000000000000
    cmp rax, r8
    je error
4:
    cqo
    idiv rcx
    cmp r9b, '%'
    jne op_store
    mov rax, rdx
op_store:
    mov qword ptr [rbx + r12*8 - 8], rax
    jmp tok_loop

line_end_nl:
    inc rsi
line_end:
    cmp r12, 1
    jne err_out
    mov rax, qword ptr [rbx]
    test rax, rax
    jns 5f
    mov byte ptr [r13], '-'
    inc r13
    neg rax
5:
    lea r9, [rip + tmpbuf + 32]
    mov r10, r9
    mov ecx, 10
6:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 6b
7:
    movzx eax, byte ptr [r9]
    mov byte ptr [r13], al
    inc r13
    inc r9
    cmp r9, r10
    jb 7b
    mov byte ptr [r13], 10
    inc r13
    jmp line_next

error:
    cmp rsi, rbp
    jae err_out
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
err_out:
    mov dword ptr [r13], 0x6f727265
    mov word ptr [r13 + 4], 0x0a72
    add r13, 6

line_next:
    lea rax, [rip + outbuf + OUTSZ - 128]
    cmp r13, rax
    jb line_loop
    call flush
    jmp line_loop

all_done:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    push rsi
    lea rsi, [rip + outbuf]
8:
    mov rdx, r13
    sub rdx, rsi
    jle 9f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 9f
    add rsi, rax
    jmp 8b
9:
    lea r13, [rip + outbuf]
    pop rsi
    ret

.bss
.balign 16
inbuf:
    .zero INCAP
.balign 16
vstack:
    .zero VSTK*8
.balign 16
tmpbuf:
    .zero 48
.balign 16
outbuf:
    .zero OUTSZ
