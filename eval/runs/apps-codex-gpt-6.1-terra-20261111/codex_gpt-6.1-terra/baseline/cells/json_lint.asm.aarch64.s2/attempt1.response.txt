.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    lsl x2, x2, #16
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    adrp x23, states
    add x23, x23, :lo12:states
    mov w24, #0

main_loop:
    cmp w24, #0
    b.eq expect_value
    cmp w24, #1
    b.eq document_end
    cmp w24, #2
    b.eq array_first
    cmp w24, #3
    b.eq array_after
    cmp w24, #4
    b.eq object_first
    cmp w24, #5
    b.eq object_key
    cmp w24, #6
    b.eq object_colon
    b object_after

skip_ws:
    cmp x21, x20
    b.hs skip_done
    ldrb w0, [x19, x21]
    cmp w0, #32
    b.eq skip_one
    cmp w0, #9
    b.eq skip_one
    cmp w0, #10
    b.eq skip_one
    cmp w0, #13
    b.ne skip_done
skip_one:
    add x21, x21, #1
    b skip_ws
skip_done:
    ret

document_end:
    bl skip_ws
    cmp x21, x20
    b.ne error
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

array_first:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq close_container
    mov w24, #0
    b main_loop

array_after:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x21, x21, #1
    mov w24, #0
    b main_loop

object_first:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq close_container
    b key_start

object_key:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
key_start:
    cmp w0, #'"'
    b.ne error
    mov w25, #1
    b string_start

object_colon:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #':'
    b.ne error
    add x21, x21, #1
    mov w24, #0
    b main_loop

object_after:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x21, x21, #1
    mov w24, #5
    b main_loop

close_container:
    add x21, x21, #1
    sub x22, x22, #1
    b value_complete

expect_value:
    bl skip_ws
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'{'
    b.eq open_object
    cmp w0, #'['
    b.eq open_array
    cmp w0, #'"'
    b.eq value_string
    cmp w0, #'t'
    b.eq literal_true
    cmp w0, #'f'
    b.eq literal_false
    cmp w0, #'n'
    b.eq literal_null
    cmp w0, #'-'
    b.eq number_minus
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number_int
    b error

open_array:
    mov w0, #3
    strb w0, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov w24, #2
    b main_loop

open_object:
    mov w0, #7
    strb w0, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov w24, #4
    b main_loop

value_complete:
    cbz x22, complete_document
    sub x0, x22, #1
    ldrb w24, [x23, x0]
    b main_loop
complete_document:
    mov w24, #1
    b main_loop

value_string:
    mov w25, #0
string_start:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq string_done
    cmp w0, #32
    b.lo error
    cmp w0, #92
    b.eq string_escape
    add x21, x21, #1
    b string_loop

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'u'
    b.eq string_unicode
    cmp w0, #'"'
    b.eq escape_ok
    cmp w0, #92
    b.eq escape_ok
    cmp w0, #'/'
    b.eq escape_ok
    cmp w0, #'b'
    b.eq escape_ok
    cmp w0, #'f'
    b.eq escape_ok
    cmp w0, #'n'
    b.eq escape_ok
    cmp w0, #'r'
    b.eq escape_ok
    cmp w0, #'t'
    b.ne error
escape_ok:
    add x21, x21, #1
    b string_loop

string_unicode:
    add x21, x21, #1
    mov w2, #4
unicode_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls unicode_digit
    orr w1, w0, #32
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
unicode_digit:
    add x21, x21, #1
    subs w2, w2, #1
    b.ne unicode_loop
    b string_loop

string_done:
    add x21, x21, #1
    cbz w25, value_complete
    mov w24, #6
    b main_loop

literal_true:
    adrp x2, true_text
    add x2, x2, :lo12:true_text
    mov x3, #4
    b literal_loop
literal_false:
    adrp x2, false_text
    add x2, x2, :lo12:false_text
    mov x3, #5
    b literal_loop
literal_null:
    adrp x2, null_text
    add x2, x2, :lo12:null_text
    mov x3, #4
literal_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    ldrb w1, [x2], #1
    cmp w0, w1
    b.ne error
    add x21, x21, #1
    subs x3, x3, #1
    b.ne literal_loop
    b value_complete

number_minus:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
number_int:
    add x21, x21, #1
    cmp w0, #'0'
    b.eq number_fraction
number_digits:
    cmp x21, x20
    b.eq value_complete
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi number_fraction
    add x21, x21, #1
    b number_digits

number_fraction:
    cmp x21, x20
    b.eq value_complete
    ldrb w0, [x19, x21]
    cmp w0, #'.'
    b.ne number_exponent
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_complete
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls fraction_digits

number_exponent:
    cmp w0, #'e'
    b.eq exponent_start
    cmp w0, #'E'
    b.ne value_complete
exponent_start:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq exponent_sign
    cmp w0, #'-'
    b.ne exponent_first
exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
exponent_first:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_complete
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls exponent_digits
    b value_complete

error:
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    mov x0, #1
    mov x8, #64
    svc #0
    adrp x3, output_buf
    add x3, x3, :lo12:output_buf
    add x4, x3, #32
    mov w0, #10
    strb w0, [x4, #-1]!
    mov x5, x21
    mov x6, #10
decimal_loop:
    udiv x7, x5, x6
    msub x0, x7, x6, x5
    add w0, w0, #'0'
    strb w0, [x4, #-1]!
    mov x5, x7
    cbnz x5, decimal_loop
    add x2, x3, #32
    sub x2, x2, x4
    mov x1, x4
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.bss
.balign 16
input:
    .zero 65537
states:
    .zero 1000
output_buf:
    .zero 32
