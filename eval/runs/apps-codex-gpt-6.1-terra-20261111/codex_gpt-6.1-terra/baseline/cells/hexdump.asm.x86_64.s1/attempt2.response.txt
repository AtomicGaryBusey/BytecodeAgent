.section .rodata
digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input:
.zero 65552
output:
.zero 324000

.text
.globl _start
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65552
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r13, [rip + input]
    lea r14, [rip + output]
    lea r15, [rip + digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae final_offset

    mov rax, rbx
    mov ecx, 8
offset_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14], dl
    inc r14
    dec ecx
    jnz offset_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    add r14, 2

    mov r8, r12
    sub r8, rbx
    cmp r8, 16
    jbe row_size_ready
    mov r8d, 16
row_size_ready:
    xor r9d, r9d
hex_loop:
    cmp r9, r8
    jae empty_slot
    movzx eax, byte ptr [r13 + rbx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [r15 + rax]
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14], al
    mov byte ptr [r14 + 1], dl
    inc rbx
    jmp slot_end
empty_slot:
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
slot_end:
    mov byte ptr [r14 + 2], ' '
    add r14, 3
    inc r9
    cmp r9, 8
    jne no_extra_space
    mov byte ptr [r14], ' '
    inc r14
no_extra_space:
    cmp r9, 16
    jb hex_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], '|'
    add r14, 2

    mov r10, rbx
    sub r10, r8
    xor r9d, r9d
ascii_loop:
    mov al, byte ptr [r13 + r10]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [r14], al
    inc r14
    inc r10
    inc r9
    cmp r9, r8
    jb ascii_loop

    mov byte ptr [r14], '|'
    mov byte ptr [r14 + 1], 10
    add r14, 2
    jmp row_loop

final_offset:
    mov rax, r12
    mov ecx, 8
final_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14], dl
    inc r14
    dec ecx
    jnz final_loop
    mov byte ptr [r14], 10
    inc r14

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall
