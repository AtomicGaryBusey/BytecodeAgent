.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x20, output_buffer
    add x20, x20, :lo12:output_buffer
    adrp x21, hex_digits
    add x21, x21, :lo12:hex_digits
    mov x22, #0

read_loop:
    mov x0, #0
    add x1, x19, x22
    mov x2, #16
    sub x2, x2, x22
    mov x8, #63
    svc #0
    cbz x0, end_of_input
    add x22, x22, x0
    cmp x22, #16
    b.ne read_loop
    bl print_row
    add x23, x23, x22
    mov x22, #0
    b read_loop

end_of_input:
    cbz x22, final_offset
    bl print_row
    add x23, x23, x22

final_offset:
    mov x24, x20
    bl emit_offset
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_row:
    stp x29, x30, [sp, #-16]!
    mov x24, x20
    bl emit_offset
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    mov x25, #0

hex_loop:
    cmp x25, x22
    b.hs hex_padding
    ldrb w0, [x19, x25]
    lsr w1, w0, #4
    ldrb w1, [x21, x1]
    strb w1, [x24], #1
    and w0, w0, #15
    ldrb w0, [x21, x0]
    strb w0, [x24], #1
    mov w0, #32
    strb w0, [x24], #1
    b hex_next

hex_padding:
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1

hex_next:
    add x25, x25, #1
    cmp x25, #8
    b.ne hex_no_extra
    mov w0, #32
    strb w0, [x24], #1
hex_no_extra:
    cmp x25, #16
    b.ne hex_loop
    mov w0, #32
    strb w0, [x24], #1
    mov w0, #124
    strb w0, [x24], #1
    mov x25, #0

ascii_loop:
    cmp x25, x22
    b.hs ascii_done
    ldrb w0, [x19, x25]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.ls ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    strb w0, [x24], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret

emit_offset:
    mov w2, #28
offset_loop:
    lsr x0, x23, x2
    and x0, x0, #15
    ldrb w0, [x21, x0]
    strb w0, [x24], #1
    subs w2, w2, #4
    b.ge offset_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 16
output_buffer:
    .zero 80
