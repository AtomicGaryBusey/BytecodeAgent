.text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    mov x22, x21
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits
    mov x24, #0

row_loop:
    cmp x24, x20
    b.hs final_offset

    mov x0, x24
    bl emit_offset
    mov w0, #' '
    strb w0, [x22], #1
    strb w0, [x22], #1

    sub x25, x20, x24
    cmp x25, #16
    b.ls row_length_ready
    mov x25, #16
row_length_ready:
    mov x26, #0

hex_loop:
    cmp x26, x25
    b.hs empty_slot
    add x1, x19, x24
    ldrb w2, [x1, x26]
    lsr w3, w2, #4
    ldrb w3, [x23, w3, uxtw]
    strb w3, [x22], #1
    and w3, w2, #15
    ldrb w3, [x23, w3, uxtw]
    strb w3, [x22], #1
    mov w3, #' '
    strb w3, [x22], #1
    b slot_done

empty_slot:
    mov w3, #' '
    strb w3, [x22], #1
    strb w3, [x22], #1
    strb w3, [x22], #1

slot_done:
    add x26, x26, #1
    cmp x26, #8
    b.ne no_extra_space
    mov w3, #' '
    strb w3, [x22], #1
no_extra_space:
    cmp x26, #16
    b.lo hex_loop

    mov w3, #' '
    strb w3, [x22], #1
    mov w3, #'|'
    strb w3, [x22], #1
    mov x26, #0
    add x1, x19, x24

ascii_loop:
    ldrb w2, [x1, x26]
    cmp w2, #0x20
    b.lo nonprintable
    cmp w2, #0x7e
    b.ls printable
nonprintable:
    mov w2, #'.'
printable:
    strb w2, [x22], #1
    add x26, x26, #1
    cmp x26, x25
    b.lo ascii_loop

    mov w3, #'|'
    strb w3, [x22], #1
    mov w3, #10
    strb w3, [x22], #1
    add x24, x24, x25
    b row_loop

final_offset:
    mov x0, x20
    bl emit_offset
    mov w0, #10
    strb w0, [x22], #1

    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov w1, #28
offset_loop:
    lsr x2, x0, x1
    and x2, x2, #15
    ldrb w3, [x23, x2]
    strb w3, [x22], #1
    subs w1, w1, #4
    b.pl offset_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 69632
output_buffer:
    .zero 327689
