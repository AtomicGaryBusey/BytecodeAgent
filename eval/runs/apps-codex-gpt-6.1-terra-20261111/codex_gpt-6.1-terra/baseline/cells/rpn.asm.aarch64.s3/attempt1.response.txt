.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #65536

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    cbnz x2, read_call
    adrp x1, extra_byte
    add x1, x1, :lo12:extra_byte
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, x19
    add x23, x19, x20
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

next_line:
    cmp x22, x23
    b.hs finish
    mov x26, #0

token_loop:
    cmp x22, x23
    b.hs line_end
    ldrb w0, [x22]
    cmp w0, #10
    b.eq consume_newline
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator

    mov x1, x22
find_token_end:
    cmp x22, x23
    b.hs token_found
    ldrb w0, [x22]
    cmp w0, #10
    b.eq token_found
    cmp w0, #32
    b.eq token_found
    cmp w0, #9
    b.eq token_found
    add x22, x22, #1
    b find_token_end

token_found:
    sub x2, x22, x1
    cmp x2, #1
    b.ne parse_number
    ldrb w3, [x1]
    cmp w3, #'+'
    b.eq operator
    cmp w3, #'-'
    b.eq operator
    cmp w3, #'*'
    b.eq operator
    cmp w3, #'/'
    b.eq operator
    cmp w3, #'%'
    b.eq operator

parse_number:
    mov x4, #0
    ldrb w3, [x1]
    cmp w3, #'-'
    b.ne number_digits
    mov x4, #1
    add x1, x1, #1
    sub x2, x2, #1
number_digits:
    cbz x2, line_error
    cmp x2, #18
    b.hi line_error
    mov x5, #0
    mov x6, #10
digit_loop:
    ldrb w3, [x1], #1
    sub w3, w3, #'0'
    cmp w3, #9
    b.hi line_error
    madd x5, x5, x6, x3
    subs x2, x2, #1
    b.ne digit_loop
    cbz x4, push_number
    neg x5, x5
push_number:
    str x5, [x24, x26, lsl #3]
    add x26, x26, #1
    b token_loop

operator:
    cmp x26, #2
    b.lo line_error
    sub x26, x26, #1
    ldr x5, [x24, x26, lsl #3]
    sub x26, x26, #1
    ldr x4, [x24, x26, lsl #3]
    cmp w3, #'+'
    b.eq do_add
    cmp w3, #'-'
    b.eq do_sub
    cmp w3, #'*'
    b.eq do_mul
    cbz x5, line_error
    mov x6, #1
    lsl x6, x6, #63
    cmp x4, x6
    b.ne do_division
    cmn x5, #1
    b.eq line_error
do_division:
    sdiv x6, x4, x5
    cmp w3, #'/'
    b.eq div_result
    msub x5, x6, x5, x4
    b push_number
div_result:
    mov x5, x6
    b push_number
do_add:
    add x5, x4, x5
    b push_number
do_sub:
    sub x5, x4, x5
    b push_number
do_mul:
    mul x5, x4, x5
    b push_number

skip_separator:
    add x22, x22, #1
    b token_loop

consume_newline:
    add x22, x22, #1
line_end:
    cmp x26, #1
    b.ne emit_error
    ldr x4, [x24]
    add x1, x25, #31
    mov w2, #10
    strb w2, [x1]
    mov x6, x1
    mov x7, #0
    cmp x4, #0
    b.ge magnitude_ready
    mov x7, #1
    neg x4, x4
magnitude_ready:
    mov x5, #10
format_loop:
    udiv x2, x4, x5
    msub x3, x2, x5, x4
    add w3, w3, #'0'
    strb w3, [x1, #-1]!
    mov x4, x2
    cbnz x4, format_loop
    cbz x7, write_number
    mov w3, #'-'
    strb w3, [x1, #-1]!
write_number:
    sub x2, x6, x1
    add x2, x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    b next_line

line_error:
    cmp x22, x23
    b.hs emit_error
    ldrb w0, [x22], #1
    cmp w0, #10
    b.ne line_error
emit_error:
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    mov x8, #64
    svc #0
    b next_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_text:
    .ascii "error\n"

.bss
.balign 16
input_buf:
    .zero 65536
extra_byte:
    .zero 1
.balign 16
value_stack:
    .zero 8000
output_buf:
    .zero 32
