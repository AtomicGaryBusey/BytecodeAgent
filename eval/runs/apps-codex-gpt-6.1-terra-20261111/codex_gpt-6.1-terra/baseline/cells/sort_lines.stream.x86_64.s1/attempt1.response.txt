.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea rbx, [rip + input]
    lea r13, [rip + lines]
    xor r14d, r14d
    xor r8d, r8d
    xor ecx, ecx
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rbx + rcx], 10
    jne parse_next
    lea rax, [rbx + r8]
    mov qword ptr [r13 + r14*8], rax
    mov rax, rcx
    sub rax, r8
    mov qword ptr [r13 + r14*8 + 8000], rax
    inc r14
    lea r8, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp r8, r12
    je sort_begin
    lea rax, [rbx + r8]
    mov qword ptr [r13 + r14*8], rax
    mov rax, r12
    sub rax, r8
    mov qword ptr [r13 + r14*8 + 8000], rax
    inc r14

sort_begin:
    mov r15d, 1
sort_outer:
    cmp r15, r14
    jae output_begin
    mov r8, qword ptr [r13 + r15*8]
    mov r9, qword ptr [r13 + r15*8 + 8000]
    mov r10, r15
sort_inner:
    test r10, r10
    jz sort_insert
    lea rax, [r10 - 1]
    mov rsi, qword ptr [r13 + rax*8]
    mov rdx, qword ptr [r13 + rax*8 + 8000]
    xor ecx, ecx
compare_loop:
    cmp rcx, rdx
    jae previous_ended
    cmp rcx, r9
    jae shift_previous
    movzx eax, byte ptr [rsi + rcx]
    movzx edi, byte ptr [r8 + rcx]
    cmp eax, edi
    ja shift_previous
    jb sort_insert
    inc rcx
    jmp compare_loop
previous_ended:
    jmp sort_insert
shift_previous:
    mov qword ptr [r13 + r10*8], rsi
    mov qword ptr [r13 + r10*8 + 8000], rdx
    dec r10
    jmp sort_inner
sort_insert:
    mov qword ptr [r13 + r10*8], r8
    mov qword ptr [r13 + r10*8 + 8000], r9
    inc r15
    jmp sort_outer

output_begin:
    lea rdi, [rip + output]
    xor r15d, r15d
output_loop:
    cmp r15, r14
    jae output_write
    mov rsi, qword ptr [r13 + r15*8]
    mov rcx, qword ptr [r13 + r15*8 + 8000]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r15
    jmp output_loop
output_write:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input:
    .zero 32769
.balign 16
lines:
    .zero 16000
.balign 16
output:
    .zero 32769
