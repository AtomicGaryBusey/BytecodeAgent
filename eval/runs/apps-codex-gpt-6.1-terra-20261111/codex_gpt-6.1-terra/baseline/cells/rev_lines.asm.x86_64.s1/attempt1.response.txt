.section .bss
input:
    .zero 65537
output:
    .zero 65537

.section .text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + output]
    xor r15d, r15d
    xor ebx, ebx
    xor r8d, r8d

scan:
    cmp r8, r13
    je end_input
    cmp byte ptr [r12 + r8], 10
    je newline
    inc r8
    jmp scan

newline:
    mov r9, r8
reverse_line:
    cmp r9, rbx
    je line_reversed
    dec r9
    mov al, byte ptr [r12 + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_line

line_reversed:
    mov byte ptr [r14 + r15], 10
    inc r15
    inc r8
    mov rbx, r8
    jmp scan

end_input:
    cmp rbx, r13
    je write_output
    mov r9, r13
reverse_final:
    cmp r9, rbx
    je final_reversed
    dec r9
    mov al, byte ptr [r12 + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_final

final_reversed:
    mov byte ptr [r14 + r15], 10
    inc r15

write_output:
    test r15, r15
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
