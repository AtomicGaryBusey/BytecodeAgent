.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
    .zero 49153
output:
    .zero 66400

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    lea r14, [rip + output]
    mov r15, r14
    lea rbx, [rip + alphabet]
    xor r8d, r8d
    xor r9d, r9d

encode_loop:
    mov rax, r13
    sub rax, r8
    cmp rax, 3
    jb encode_tail

    movzx eax, byte ptr [r12 + r8]
    movzx edx, byte ptr [r12 + r8 + 1]
    movzx esi, byte ptr [r12 + r8 + 2]
    add r8, 3

    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, esi

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 2], dl

    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 3], al
    add r15, 4
    add r9d, 4
    cmp r9d, 76
    jne encode_loop
    mov byte ptr [r15], 10
    inc r15
    xor r9d, r9d
    jmp encode_loop

encode_tail:
    test rax, rax
    jz finish_output

    movzx edx, byte ptr [r12 + r8]
    mov esi, edx
    shr edx, 2
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl

    and esi, 3
    shl esi, 4
    cmp rax, 1
    je tail_one

    movzx edx, byte ptr [r12 + r8 + 1]
    mov ecx, edx
    shr edx, 4
    or esi, edx
    mov sil, byte ptr [rbx + rsi]
    mov byte ptr [r15 + 1], sil

    and ecx, 15
    shl ecx, 2
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [r15 + 2], cl
    mov byte ptr [r15 + 3], '='
    jmp tail_done

tail_one:
    mov sil, byte ptr [rbx + rsi]
    mov byte ptr [r15 + 1], sil
    mov byte ptr [r15 + 2], '='
    mov byte ptr [r15 + 3], '='

tail_done:
    add r15, 4
    add r9d, 4

finish_output:
    test r9d, r9d
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    mov rdx, r15
    sub rdx, r14
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
