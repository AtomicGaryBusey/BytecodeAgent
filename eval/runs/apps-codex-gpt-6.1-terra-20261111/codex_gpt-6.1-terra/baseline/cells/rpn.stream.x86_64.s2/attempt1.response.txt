.section .rodata
error_text:
    .ascii "error\n"

.bss
input_buffer:
    .zero 65537
value_stack:
    .zero 8192
number_buffer:
    .zero 32

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz input_ready
    add r13, rax
    jmp read_more

input_ready:
    lea r14, [r12 + r13]
    mov r15, r12

next_line:
    cmp r15, r14
    jae finish
    mov rbx, r15

find_line_end:
    cmp r15, r14
    jae line_found
    cmp byte ptr [r15], 10
    je line_found
    inc r15
    jmp find_line_end

line_found:
    mov rbp, r15
    cmp r15, r14
    jae line_start
    inc r15

line_start:
    xor r13d, r13d

skip_separators:
    cmp rbx, rbp
    jae line_complete
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_start

skip_one:
    inc rbx
    jmp skip_separators

token_start:
    mov r8, rbx

find_token_end:
    cmp rbx, rbp
    jae token_found
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc rbx
    jmp find_token_end

token_found:
    mov r9, rbx
    sub r9, r8
    cmp r9, 1
    jne parse_number
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_number:
    xor r10d, r10d
    cmp byte ptr [r8], '-'
    jne number_digits
    inc r8
    dec r9
    mov r10d, 1

number_digits:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor eax, eax

digit_loop:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r8
    dec r9
    jnz digit_loop
    test r10d, r10d
    jz push_number
    neg rax

push_number:
    lea rdx, [rip + value_stack]
    mov qword ptr [rdx + r13*8], rax
    inc r13
    jmp skip_separators

operator:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + value_stack]
    mov r8, qword ptr [rdx + r13*8 - 8]
    mov r9, qword ptr [rdx + r13*8 - 16]
    movzx ecx, al
    cmp cl, '+'
    je op_add
    cmp cl, '-'
    je op_sub
    cmp cl, '*'
    je op_mul
    test r8, r8
    jz line_error
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne do_div
    cmp r8, -1
    je line_error

do_div:
    mov rax, r9
    cqo
    idiv r8
    cmp cl, '%'
    jne store_op
    mov rax, rdx
    jmp store_op

op_add:
    lea rax, [r9 + r8]
    jmp store_op

op_sub:
    mov rax, r9
    sub rax, r8
    jmp store_op

op_mul:
    mov rax, r9
    imul rax, r8

store_op:
    dec r13
    lea rdx, [rip + value_stack]
    mov qword ptr [rdx + r13*8 - 8], rax
    jmp skip_separators

line_complete:
    cmp r13, 1
    jne line_error
    lea rdx, [rip + value_stack]
    mov rax, qword ptr [rdx]
    lea rsi, [rip + number_buffer + 32]
    dec rsi
    mov byte ptr [rsi], 10
    xor r8d, r8d
    test rax, rax
    jns format_digits
    mov r8d, 1
    neg rax

format_digits:
    mov r9d, 10

format_loop:
    xor edx, edx
    div r9
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz format_loop
    test r8d, r8d
    jz write_number
    dec rsi
    mov byte ptr [rsi], '-'

write_number:
    lea rdx, [rip + number_buffer + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_text]
    mov edx, 6
    syscall
    jmp next_line

finish:
    mov eax, 60
    xor edi, edi
    syscall
