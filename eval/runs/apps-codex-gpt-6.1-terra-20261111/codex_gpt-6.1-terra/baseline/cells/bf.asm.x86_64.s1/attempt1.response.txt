.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor ebx, ebx
find_separator:
    cmp rbx, r13
    jae no_separator
    cmp byte ptr [r12 + rbx], 33
    je separator_found
    inc rbx
    jmp find_separator
separator_found:
    lea r14, [rbx + 1]
    jmp validate_setup
no_separator:
    mov r14, r13

validate_setup:
    xor ecx, ecx
    xor r8d, r8d
    lea r9, [rip + bracket_stack]
    lea r10, [rip + jumps]
validate_loop:
    cmp rcx, rbx
    jae validate_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
validate_next:
    inc rcx
    jmp validate_loop
open_bracket:
    mov dword ptr [r9 + r8*4], ecx
    inc r8
    jmp validate_next
close_bracket:
    test r8, r8
    jz invalid
    dec r8
    mov edx, dword ptr [r9 + r8*4]
    mov dword ptr [r10 + rcx*4], edx
    mov dword ptr [r10 + rdx*4], ecx
    jmp validate_next
validate_done:
    test r8, r8
    jnz invalid

    xor r15d, r15d
    lea rbp, [rip + tape]
    lea r11, [rip + output_buffer]
    xor r8d, r8d
execute_loop:
    cmp r15, rbx
    jae finished
    mov al, byte ptr [r12 + r15]
    inc r15
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 46
    je output_byte
    cmp al, 44
    je input_byte
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
    jmp execute_loop
move_right:
    inc rbp
    jmp execute_loop
move_left:
    dec rbp
    jmp execute_loop
increment:
    inc byte ptr [rbp]
    jmp execute_loop
decrement:
    dec byte ptr [rbp]
    jmp execute_loop
output_byte:
    mov al, byte ptr [rbp]
    mov byte ptr [r11 + r8], al
    inc r8
    jmp execute_loop
input_byte:
    xor eax, eax
    cmp r14, r13
    jae input_store
    mov al, byte ptr [r12 + r14]
    inc r14
input_store:
    mov byte ptr [rbp], al
    jmp execute_loop
loop_open:
    cmp byte ptr [rbp], 0
    jne execute_loop
    mov r15d, dword ptr [r10 + r15*4 - 4]
    inc r15
    jmp execute_loop
loop_close:
    cmp byte ptr [rbp], 0
    je execute_loop
    mov r15d, dword ptr [r10 + r15*4 - 4]
    inc r15
    jmp execute_loop

finished:
    test r8, r8
    jz success
    mov eax, 1
    mov edi, 1
    mov rsi, r11
    mov rdx, r8
    syscall
success:
    mov eax, 60
    xor edi, edi
    syscall
invalid:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero 65537
.balign 16
jumps:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
output_buffer:
    .zero 65536
