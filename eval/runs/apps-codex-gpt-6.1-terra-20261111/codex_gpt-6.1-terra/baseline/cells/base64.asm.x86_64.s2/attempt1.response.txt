.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
    .zero 49153
output:
    .zero 66400

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_more

encode_start:
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    lea rbx, [rip + alphabet]
    lea r14, [rsi + r13]
    xor r8d, r8d

encode_loop:
    mov rax, r14
    sub rax, rsi
    cmp rax, 3
    jb final_group

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx r9d, byte ptr [rsi + 2]
    add rsi, 3

    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl

    and eax, 3
    shl eax, 4
    mov ecx, edx
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al

    and edx, 15
    shl edx, 2
    mov ecx, r9d
    shr ecx, 6
    or edx, ecx
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl

    and r9d, 63
    mov al, byte ptr [rbx + r9]
    mov byte ptr [rdi + 3], al
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne encode_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp encode_loop

final_group:
    test rax, rax
    jz finish_line
    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl
    and eax, 3
    shl eax, 4

    lea rcx, [rsi + 1]
    cmp rcx, r14
    je one_byte

    movzx edx, byte ptr [rsi + 1]
    mov ecx, edx
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    and edx, 15
    shl edx, 2
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], '='
    jmp final_written

one_byte:
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], '='
    mov byte ptr [rdi + 3], '='

final_written:
    add rdi, 4
    add r8d, 4

finish_line:
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
