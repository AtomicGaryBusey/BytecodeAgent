.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
read_loop:
    mov x0, #0
    mov x1, x20
    movz x2, #1, lsl #16
    add x2, x2, #1
    sub x2, x2, x20
    add x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    strb wzr, [x20]
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.ne error
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

skip_ws:
    ldrb w0, [x19]
    cmp w0, #32
    b.eq ws_next
    cmp w0, #9
    b.eq ws_next
    cmp w0, #10
    b.eq ws_next
    cmp w0, #13
    b.eq ws_next
    ret
ws_next:
    add x19, x19, #1
    b skip_ws

parse_value:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x19]
    cmp w0, #123
    b.eq value_object
    cmp w0, #91
    b.eq value_array
    cmp w0, #34
    b.eq value_string
    cmp w0, #116
    b.eq value_true
    cmp w0, #102
    b.eq value_false
    cmp w0, #110
    b.eq value_null
    cmp w0, #45
    b.eq value_number
    sub w1, w0, #48
    cmp w1, #9
    b.ls value_number
    b error

value_string:
    bl parse_string
    b value_return
value_number:
    bl parse_number
    b value_return

value_true:
    adrp x1, true_text
    add x1, x1, :lo12:true_text
    mov x2, #4
    bl parse_literal
    b value_return
value_false:
    adrp x1, false_text
    add x1, x1, :lo12:false_text
    mov x2, #5
    bl parse_literal
    b value_return
value_null:
    adrp x1, null_text
    add x1, x1, :lo12:null_text
    mov x2, #4
    bl parse_literal
    b value_return

value_array:
    add x19, x19, #1
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #93
    b.eq array_close
array_item:
    bl parse_value
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #93
    b.eq array_close
    cmp w0, #44
    b.ne error
    add x19, x19, #1
    bl skip_ws
    b array_item
array_close:
    add x19, x19, #1
    b value_return

value_object:
    add x19, x19, #1
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #125
    b.eq object_close
object_member:
    ldrb w0, [x19]
    cmp w0, #34
    b.ne error
    bl parse_string
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #58
    b.ne error
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #125
    b.eq object_close
    cmp w0, #44
    b.ne error
    add x19, x19, #1
    bl skip_ws
    b object_member
object_close:
    add x19, x19, #1
value_return:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    ldrb w3, [x19]
    ldrb w4, [x1], #1
    cmp w3, w4
    b.ne error
    add x19, x19, #1
    subs x2, x2, #1
    b.ne parse_literal
    ret

parse_string:
    add x19, x19, #1
string_loop:
    ldrb w0, [x19]
    cmp w0, #34
    b.eq string_end
    cmp w0, #32
    b.lo error
    cmp w0, #92
    b.eq string_escape
    add x19, x19, #1
    b string_loop
string_end:
    add x19, x19, #1
    ret
string_escape:
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #34
    b.eq escape_simple
    cmp w0, #92
    b.eq escape_simple
    cmp w0, #47
    b.eq escape_simple
    cmp w0, #98
    b.eq escape_simple
    cmp w0, #102
    b.eq escape_simple
    cmp w0, #110
    b.eq escape_simple
    cmp w0, #114
    b.eq escape_simple
    cmp w0, #116
    b.eq escape_simple
    cmp w0, #117
    b.ne error
    add x19, x19, #1
    mov w1, #4
hex_loop:
    ldrb w0, [x19]
    sub w2, w0, #48
    cmp w2, #9
    b.ls hex_valid
    orr w0, w0, #32
    sub w2, w0, #97
    cmp w2, #5
    b.hi error
hex_valid:
    add x19, x19, #1
    subs w1, w1, #1
    b.ne hex_loop
    b string_loop
escape_simple:
    add x19, x19, #1
    b string_loop

parse_number:
    ldrb w0, [x19]
    cmp w0, #45
    b.ne number_int
    add x19, x19, #1
    ldrb w0, [x19]
number_int:
    cmp w0, #48
    b.eq number_zero
    sub w1, w0, #49
    cmp w1, #8
    b.hi error
number_digits:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls number_digits
    b number_fraction
number_zero:
    add x19, x19, #1
    ldrb w0, [x19]
number_fraction:
    cmp w0, #46
    b.ne number_exponent
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
fraction_digits:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls fraction_digits
number_exponent:
    cmp w0, #101
    b.eq exponent_start
    cmp w0, #69
    b.ne number_end
exponent_start:
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #43
    b.eq exponent_sign
    cmp w0, #45
    b.ne exponent_first
exponent_sign:
    add x19, x19, #1
    ldrb w0, [x19]
exponent_first:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
exponent_digits:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls exponent_digits
number_end:
    ret

error:
    adrp x0, input
    add x0, x0, :lo12:input
    sub x3, x19, x0
    adrp x4, output
    add x4, x4, :lo12:output
    adrp x5, error_prefix
    add x5, x5, :lo12:error_prefix
    mov x6, #9
copy_prefix:
    ldrb w7, [x5], #1
    strb w7, [x4], #1
    subs x6, x6, #1
    b.ne copy_prefix
    add x5, x4, #20
    mov x6, x5
    mov x7, #10
decimal_loop:
    udiv x9, x3, x7
    msub x10, x9, x7, x3
    add w10, w10, #48
    strb w10, [x6, #-1]!
    mov x3, x9
    cbnz x3, decimal_loop
copy_digits:
    ldrb w7, [x6], #1
    strb w7, [x4], #1
    cmp x6, x5
    b.ne copy_digits
    mov w7, #10
    strb w7, [x4], #1
    adrp x1, output
    add x1, x1, :lo12:output
    sub x2, x4, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
