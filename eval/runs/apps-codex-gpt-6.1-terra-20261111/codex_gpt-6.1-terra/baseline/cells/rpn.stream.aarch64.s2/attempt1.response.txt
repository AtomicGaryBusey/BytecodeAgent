.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    adrp x22, values
    add x22, x22, :lo12:values
    adrp x23, output
    add x23, x23, :lo12:output

line_start:
    cmp x21, x20
    b.hs program_exit
    mov x24, #0
    mov x25, #0

token_start:
    cmp x21, x20
    b.hs line_end
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq newline
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator

    mov x26, x21
token_scan:
    cmp x21, x20
    b.hs token_scanned
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq token_scanned
    cmp w0, #32
    b.eq token_scanned
    cmp w0, #9
    b.eq token_scanned
    add x21, x21, #1
    b token_scan

token_scanned:
    sub x27, x21, x26
    ldrb w0, [x19, x26]
    cmp x27, #1
    b.ne parse_literal
    cmp w0, #'+'
    b.eq operator
    cmp w0, #'-'
    b.eq operator
    cmp w0, #'*'
    b.eq operator
    cmp w0, #'/'
    b.eq operator
    cmp w0, #'%'
    b.eq operator

parse_literal:
    mov x1, x26
    add x2, x26, x27
    mov x3, #0
    ldrb w4, [x19, x1]
    cmp w4, #'-'
    b.ne literal_digits
    mov x3, #1
    add x1, x1, #1

literal_digits:
    sub x4, x2, x1
    cbz x4, line_error
    cmp x4, #18
    b.hi line_error
    mov x5, #0
    mov x6, #10
digit_loop:
    ldrb w4, [x19, x1]
    sub w4, w4, #'0'
    cmp w4, #9
    b.hi line_error
    madd x5, x5, x6, x4
    add x1, x1, #1
    cmp x1, x2
    b.lo digit_loop
    cbz x3, literal_push
    neg x5, x5
literal_push:
    str x5, [x22, x24, lsl #3]
    add x24, x24, #1
    b token_start

operator:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x2, [x22, x24, lsl #3]
    sub x24, x24, #1
    ldr x1, [x22, x24, lsl #3]
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    cbz x2, line_error
    mov x3, #0x8000000000000000
    cmp x1, x3
    b.ne do_div
    cmn x2, #1
    b.eq line_error
do_div:
    sdiv x3, x1, x2
    cmp w0, #'/'
    b.eq op_store
    msub x3, x3, x2, x1
    b op_store
op_add:
    add x3, x1, x2
    b op_store
op_sub:
    sub x3, x1, x2
    b op_store
op_mul:
    mul x3, x1, x2
op_store:
    str x3, [x22, x24, lsl #3]
    add x24, x24, #1
    b token_start

skip_separator:
    add x21, x21, #1
    b token_start

line_error:
    mov x25, #1
error_scan:
    cmp x21, x20
    b.hs line_end
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq newline
    add x21, x21, #1
    b error_scan

newline:
    add x21, x21, #1

line_end:
    cbnz x25, print_error
    cmp x24, #1
    b.ne print_error

    ldr x0, [x22]
    add x1, x23, #32
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x3, #10
    mov x4, x0
    cmp x0, #0
    b.ge format_digits
    neg x4, x0
format_digits:
    udiv x5, x4, x3
    msub x6, x5, x3, x4
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    mov x4, x5
    cbnz x4, format_digits
    cmp x0, #0
    b.ge print_number
    mov w2, #'-'
    strb w2, [x1, #-1]!
print_number:
    add x2, x23, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    b line_start

print_error:
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #6
    mov x8, #64
    svc #0
    b line_start

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
