.section .rodata
ok_msg:
.ascii "ok\n"
error_msg:
.ascii "error at "
true_word:
.ascii "true"
false_word:
.ascii "false"
null_word:
.ascii "null"

.bss
.balign 16
input:
.zero 65537
states:
.zero 1024
output:
.zero 64

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input]
    lea r14, [r13 + r12]
    lea r15, [rip + states]
    xor ebx, ebx
    mov byte ptr [r15], 0

dispatch:
    cmp r13, r14
    je at_eof
    movzx eax, byte ptr [r13]
    cmp al, 32
    je skip_ws
    cmp al, 9
    je skip_ws
    cmp al, 10
    je skip_ws
    cmp al, 13
    je skip_ws

    movzx edx, byte ptr [r15 + rbx]
    cmp edx, 0
    je parse_value
    cmp edx, 1
    je fail
    cmp edx, 2
    je array_first
    cmp edx, 3
    je parse_value
    cmp edx, 4
    je array_after
    cmp edx, 5
    je object_first
    cmp edx, 6
    je object_key
    cmp edx, 7
    je object_colon
    cmp edx, 8
    je parse_value
    jmp object_after

skip_ws:
    inc r13
    jmp dispatch

at_eof:
    cmp rbx, 0
    jne fail
    cmp byte ptr [r15], 1
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

array_first:
    cmp al, ']'
    je close_container
    jmp parse_value

array_after:
    cmp al, ']'
    je close_container
    cmp al, ','
    jne fail
    inc r13
    mov byte ptr [r15 + rbx], 3
    jmp dispatch

object_first:
    cmp al, '}'
    je close_container
object_key:
    cmp al, '"'
    jne fail
    inc r13
    call scan_string
    mov byte ptr [r15 + rbx], 7
    jmp dispatch

object_colon:
    cmp al, ':'
    jne fail
    inc r13
    mov byte ptr [r15 + rbx], 8
    jmp dispatch

object_after:
    cmp al, '}'
    je close_container
    cmp al, ','
    jne fail
    inc r13
    mov byte ptr [r15 + rbx], 6
    jmp dispatch

close_container:
    inc r13
    dec rbx
    jmp value_done

parse_value:
    cmp al, '{'
    je open_object
    cmp al, '['
    je open_array
    cmp al, '"'
    je value_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je number_minus
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp number_int

open_object:
    inc r13
    inc rbx
    mov byte ptr [r15 + rbx], 5
    jmp dispatch

open_array:
    inc r13
    inc rbx
    mov byte ptr [r15 + rbx], 2
    jmp dispatch

value_string:
    inc r13
    call scan_string
    jmp value_done

value_true:
    lea rsi, [rip + true_word]
    mov ecx, 4
    jmp literal
value_false:
    lea rsi, [rip + false_word]
    mov ecx, 5
    jmp literal
value_null:
    lea rsi, [rip + null_word]
    mov ecx, 4
literal:
    cmp r13, r14
    je fail
    mov al, byte ptr [r13]
    cmp al, byte ptr [rsi]
    jne fail
    inc r13
    inc rsi
    dec ecx
    jnz literal
    jmp value_done

number_minus:
    inc r13
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
number_int:
    cmp al, '0'
    je number_zero
number_digits:
    inc r13
    cmp r13, r14
    je value_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb number_fraction
    cmp al, '9'
    jbe number_digits
    jmp number_fraction
number_zero:
    inc r13
    cmp r13, r14
    je value_done
    movzx eax, byte ptr [r13]
number_fraction:
    cmp al, '.'
    jne number_exponent
    inc r13
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
fraction_digits:
    inc r13
    cmp r13, r14
    je value_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    jbe fraction_digits
number_exponent:
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne value_done
exponent_start:
    inc r13
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_require_digit
exponent_sign:
    inc r13
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
exponent_require_digit:
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
exponent_digits:
    inc r13
    cmp r13, r14
    je value_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb value_done
    cmp al, '9'
    jbe exponent_digits

value_done:
    movzx eax, byte ptr [r15 + rbx]
    cmp eax, 0
    je done_root
    cmp eax, 2
    je done_array
    cmp eax, 3
    je done_array
    mov byte ptr [r15 + rbx], 9
    jmp dispatch
done_root:
    mov byte ptr [r15], 1
    jmp dispatch
done_array:
    mov byte ptr [r15 + rbx], 4
    jmp dispatch

scan_string:
string_loop:
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, '"'
    je string_end
    cmp al, 32
    jb fail
    cmp al, 92
    je string_escape
    inc r13
    jmp string_loop
string_end:
    inc r13
    ret
string_escape:
    inc r13
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, 'u'
    je string_unicode
    cmp al, '"'
    je string_escape_ok
    cmp al, 92
    je string_escape_ok
    cmp al, '/'
    je string_escape_ok
    cmp al, 'b'
    je string_escape_ok
    cmp al, 'f'
    je string_escape_ok
    cmp al, 'n'
    je string_escape_ok
    cmp al, 'r'
    je string_escape_ok
    cmp al, 't'
    jne fail
string_escape_ok:
    inc r13
    jmp string_loop
string_unicode:
    inc r13
    mov ecx, 4
unicode_loop:
    cmp r13, r14
    je fail
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe unicode_digit
    or al, 32
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
unicode_digit:
    inc r13
    dec ecx
    jnz unicode_loop
    jmp string_loop

fail:
    lea rax, [rip + input]
    sub r13, rax
    lea rdi, [rip + output]
    lea rsi, [rip + error_msg]
    mov ecx, 9
    rep movsb
    lea r8, [rip + output + 63]
    mov byte ptr [r8], 10
    mov rax, r13
    mov r9d, 10
decimal_loop:
    xor edx, edx
    div r9
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_loop
    lea rcx, [rip + output + 64]
    sub rcx, r8
    mov rsi, r8
    rep movsb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
