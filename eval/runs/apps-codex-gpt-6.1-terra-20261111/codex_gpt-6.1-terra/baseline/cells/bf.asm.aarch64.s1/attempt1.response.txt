.equ INPUT_CAP, 65536
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 4000

.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #INPUT_CAP

read_loop:
    cmp x20, x21
    b.eq read_full
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_full:
    mov x0, #0
    adrp x1, read_extra
    add x1, x1, :lo12:read_extra
    mov x2, #1
    mov x8, #63
    svc #0
    cbnz x0, read_full

read_done:
    mov x22, #0
find_delimiter:
    cmp x22, x20
    b.eq delimiter_done
    ldrb w0, [x19, x22]
    cmp w0, #33
    b.eq delimiter_found
    add x22, x22, #1
    b find_delimiter

delimiter_found:
    add x23, x22, #1
    b validate_init

delimiter_done:
    mov x23, x20

validate_init:
    adrp x24, jumps
    add x24, x24, :lo12:jumps
    adrp x25, bracket_stack
    add x25, x25, :lo12:bracket_stack
    mov x26, #0
    mov x27, #0

validate_loop:
    cmp x26, x22
    b.eq validate_done
    ldrb w0, [x19, x26]
    cmp w0, #91
    b.eq validate_open
    cmp w0, #93
    b.eq validate_close
    add x26, x26, #1
    b validate_loop

validate_open:
    str w26, [x25, x27, lsl #2]
    add x27, x27, #1
    add x26, x26, #1
    b validate_loop

validate_close:
    cbz x27, invalid
    sub x27, x27, #1
    ldr w0, [x25, x27, lsl #2]
    str w26, [x24, x0, lsl #2]
    str w0, [x24, x26, lsl #2]
    add x26, x26, #1
    b validate_loop

validate_done:
    cbnz x27, invalid
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x27, output
    add x27, x27, :lo12:output
    mov x26, #0
    mov x28, #0

run_loop:
    cmp x26, x22
    b.eq finish
    ldrb w0, [x19, x26]
    add x26, x26, #1
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_inc
    cmp w0, #45
    b.eq cmd_dec
    cmp w0, #46
    b.eq cmd_out
    cmp w0, #44
    b.eq cmd_in
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    b run_loop

cmd_right:
    add x25, x25, #1
    b run_loop

cmd_left:
    sub x25, x25, #1
    b run_loop

cmd_inc:
    ldrb w0, [x25]
    add w0, w0, #1
    strb w0, [x25]
    b run_loop

cmd_dec:
    ldrb w0, [x25]
    sub w0, w0, #1
    strb w0, [x25]
    b run_loop

cmd_out:
    ldrb w0, [x25]
    strb w0, [x27, x28]
    add x28, x28, #1
    b run_loop

cmd_in:
    cmp x23, x20
    b.hs input_exhausted
    ldrb w0, [x19, x23]
    add x23, x23, #1
    strb w0, [x25]
    b run_loop

input_exhausted:
    strb wzr, [x25]
    b run_loop

cmd_open:
    ldrb w0, [x25]
    cbnz w0, run_loop
    sub x0, x26, #1
    ldr w0, [x24, x0, lsl #2]
    add x26, x0, #1
    b run_loop

cmd_close:
    ldrb w0, [x25]
    cbz w0, run_loop
    sub x0, x26, #1
    ldr w0, [x24, x0, lsl #2]
    add x26, x0, #1
    b run_loop

finish:
    cbz x28, exit_success
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero INPUT_CAP
read_extra:
    .zero 1
.balign 16
jumps:
    .zero 262144
.balign 16
bracket_stack:
    .zero STACK_SIZE
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
output:
    .zero INPUT_CAP
