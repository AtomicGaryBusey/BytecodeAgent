.globl _start

.section .bss
.balign 16
buffer:
    .zero 65536
output:
    .zero 64

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz finish_input

    add r14, rax
    lea rsi, [rip + buffer]
    lea rdi, [rsi + rax]

scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, rdi
    jne scan
    jmp read_more

finish_input:
    lea rdi, [rip + output]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

decimal:
    xor ecx, ecx
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz 1b
2:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 2b
    ret
