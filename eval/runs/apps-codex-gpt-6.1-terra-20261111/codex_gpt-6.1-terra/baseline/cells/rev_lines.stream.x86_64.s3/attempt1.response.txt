.globl _start

.bss
input:
    .zero 65537
output:
    .zero 65537

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + output]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan_loop:
    cmp r8, r13
    je input_end
    cmp byte ptr [r12 + r8], 10
    je line_end
    inc r8
    jmp scan_loop

line_end:
    mov rax, r8
reverse_line:
    cmp rax, r9
    je append_newline
    dec rax
    mov dl, byte ptr [r12 + rax]
    mov byte ptr [r14 + r10], dl
    inc r10
    jmp reverse_line

append_newline:
    mov byte ptr [r14 + r10], 10
    inc r10
    inc r8
    mov r9, r8
    jmp scan_loop

input_end:
    cmp r9, r13
    je write_output
    mov rax, r13
reverse_final:
    cmp rax, r9
    je final_newline
    dec rax
    mov dl, byte ptr [r12 + rax]
    mov byte ptr [r14 + r10], dl
    inc r10
    jmp reverse_final

final_newline:
    mov byte ptr [r14 + r10], 10
    inc r10

write_output:
    test r10, r10
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
