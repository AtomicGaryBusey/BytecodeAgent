.section .rodata
error_text:
.ascii "error\n"

.bss
.balign 16
input_buffer:
.zero 65537
value_stack:
.zero 8192
output_buffer:
.zero 32

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r13, [r12 + r13]

line_start:
    cmp r12, r13
    jae exit_program
    xor r14d, r14d

next_token:
    cmp r12, r13
    jae finish_line
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline_finish
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov r8, r12
find_token_end:
    cmp r12, r13
    jae token_found
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_found
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r12
    jmp find_token_end

skip_separator:
    inc r12
    jmp next_token

token_found:
    mov r9, r12
    sub r9, r8
    cmp r9, 1
    jne parse_literal
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je operator_add
    cmp al, '-'
    je operator_sub
    cmp al, '*'
    je operator_mul
    cmp al, '/'
    je operator_div
    cmp al, '%'
    je operator_rem

parse_literal:
    xor r10d, r10d
    cmp byte ptr [r8], '-'
    jne literal_digits
    mov r10d, 1
    inc r8
    dec r9

literal_digits:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor eax, eax

digit_loop:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r8
    dec r9
    jnz digit_loop
    test r10d, r10d
    jz push_literal
    neg rax

push_literal:
    lea rdx, [rip + value_stack]
    mov qword ptr [rdx + r14*8], rax
    inc r14
    jmp next_token

operator_add:
    cmp r14, 2
    jb line_error
    lea rdx, [rip + value_stack]
    mov rax, qword ptr [rdx + r14*8 - 8]
    add qword ptr [rdx + r14*8 - 16], rax
    dec r14
    jmp next_token

operator_sub:
    cmp r14, 2
    jb line_error
    lea rdx, [rip + value_stack]
    mov rax, qword ptr [rdx + r14*8 - 8]
    sub qword ptr [rdx + r14*8 - 16], rax
    dec r14
    jmp next_token

operator_mul:
    cmp r14, 2
    jb line_error
    lea rdx, [rip + value_stack]
    mov rax, qword ptr [rdx + r14*8 - 16]
    imul rax, qword ptr [rdx + r14*8 - 8]
    mov qword ptr [rdx + r14*8 - 16], rax
    dec r14
    jmp next_token

operator_div:
    xor r10d, r10d
    jmp divide_common

operator_rem:
    mov r10d, 1

divide_common:
    cmp r14, 2
    jb line_error
    lea r8, [rip + value_stack]
    mov r9, qword ptr [r8 + r14*8 - 8]
    test r9, r9
    jz line_error
    mov rax, qword ptr [r8 + r14*8 - 16]
    cmp r9, -1
    jne do_divide
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error

do_divide:
    cqo
    idiv r9
    test r10d, r10d
    jz store_division
    mov rax, rdx

store_division:
    mov qword ptr [r8 + r14*8 - 16], rax
    dec r14
    jmp next_token

newline_finish:
    inc r12

finish_line:
    cmp r14, 1
    jne print_error
    lea rdx, [rip + value_stack]
    mov rax, qword ptr [rdx]
    lea rsi, [rip + output_buffer + 32]
    dec rsi
    mov byte ptr [rsi], 10
    xor r8d, r8d
    test rax, rax
    jns format_digits
    mov r8d, 1
    neg rax

format_digits:
    mov r9d, 10
format_loop:
    xor edx, edx
    div r9
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz format_loop
    test r8d, r8d
    jz print_number
    dec rsi
    mov byte ptr [rsi], '-'

print_number:
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp line_start

line_error:
    cmp r12, r13
    jae print_error
    cmp byte ptr [r12], 10
    je error_newline
    inc r12
    jmp line_error

error_newline:
    inc r12

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_text]
    mov edx, 6
    syscall
    jmp line_start

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
