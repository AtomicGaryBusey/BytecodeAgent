.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
true_tail:
    .ascii "rue"
false_tail:
    .ascii "alse"
null_tail:
    .ascii "ull"

.bss
.balign 16
input:
    .zero 65537
states:
    .zero 1001
outbuf:
    .zero 32

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input]
    lea r14, [r13 + r12]
    xor r15d, r15d
    lea rbx, [rip + states]
    mov byte ptr [rbx], 0

dispatch:
    cmp r13, r14
    jae at_eof
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je skip_ws
    cmp al, 0x09
    je skip_ws
    cmp al, 0x0a
    je skip_ws
    cmp al, 0x0d
    je skip_ws

    movzx edx, byte ptr [rbx + r15]
    cmp edx, 0
    je value_start
    cmp edx, 1
    je parse_error
    cmp edx, 2
    je array_first
    cmp edx, 3
    je value_start
    cmp edx, 4
    je array_after
    cmp edx, 5
    je object_first
    cmp edx, 6
    je object_key
    cmp edx, 7
    je object_colon
    cmp edx, 8
    je value_start
    jmp object_after

skip_ws:
    inc r13
    jmp dispatch

array_first:
    cmp al, ']'
    je close_container
    jmp value_start

array_after:
    cmp al, ']'
    je close_container
    cmp al, ','
    jne parse_error
    inc r13
    mov byte ptr [rbx + r15], 3
    jmp dispatch

object_first:
    cmp al, '}'
    je close_container
object_key:
    cmp al, '"'
    jne parse_error
    inc r13
    mov byte ptr [rbx + r15], 7
    jmp string_body

object_colon:
    cmp al, ':'
    jne parse_error
    inc r13
    mov byte ptr [rbx + r15], 8
    jmp dispatch

object_after:
    cmp al, '}'
    je close_container
    cmp al, ','
    jne parse_error
    inc r13
    mov byte ptr [rbx + r15], 6
    jmp dispatch

close_container:
    inc r13
    dec r15
    jmp dispatch

value_start:
    cmp al, '{'
    je start_object
    cmp al, '['
    je start_array
    cmp al, '"'
    je start_string
    cmp al, 't'
    je start_true
    cmp al, 'f'
    je start_false
    cmp al, 'n'
    je start_null
    cmp al, '-'
    je start_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
    jmp start_number

mark_value:
    movzx edx, byte ptr [rbx + r15]
    cmp edx, 0
    je mark_top
    cmp edx, 8
    je mark_object
    mov byte ptr [rbx + r15], 4
    ret
mark_top:
    mov byte ptr [rbx + r15], 1
    ret
mark_object:
    mov byte ptr [rbx + r15], 9
    ret

start_object:
    call mark_value
    inc r13
    inc r15
    mov byte ptr [rbx + r15], 5
    jmp dispatch

start_array:
    call mark_value
    inc r13
    inc r15
    mov byte ptr [rbx + r15], 2
    jmp dispatch

start_string:
    call mark_value
    inc r13
string_body:
    cmp r13, r14
    jae parse_error
    movzx eax, byte ptr [r13]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb parse_error
    cmp al, 0x5c
    je string_escape
    inc r13
    jmp string_body

string_end:
    inc r13
    jmp dispatch

string_escape:
    inc r13
    cmp r13, r14
    jae parse_error
    movzx eax, byte ptr [r13]
    cmp al, 'u'
    je string_unicode
    cmp al, '"'
    je escape_done
    cmp al, 0x5c
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    jne parse_error
escape_done:
    inc r13
    jmp string_body

string_unicode:
    inc r13
    mov ecx, 4
unicode_loop:
    cmp r13, r14
    jae parse_error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe unicode_digit
    or al, 0x20
    cmp al, 'a'
    jb parse_error
    cmp al, 'f'
    ja parse_error
unicode_digit:
    inc r13
    dec ecx
    jnz unicode_loop
    jmp string_body

start_true:
    lea rsi, [rip + true_tail]
    mov ecx, 3
    jmp literal_start
start_false:
    lea rsi, [rip + false_tail]
    mov ecx, 4
    jmp literal_start
start_null:
    lea rsi, [rip + null_tail]
    mov ecx, 3
literal_start:
    call mark_value
    inc r13
literal_loop:
    cmp r13, r14
    jae parse_error
    mov al, byte ptr [r13]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc r13
    inc rsi
    dec ecx
    jnz literal_loop
    jmp dispatch

start_number:
    call mark_value
    cmp byte ptr [r13], '-'
    jne number_int
    inc r13
number_int:
    cmp r13, r14
    jae parse_error
    mov al, byte ptr [r13]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error
    inc r13
number_digits:
    cmp r13, r14
    jae dispatch
    mov al, byte ptr [r13]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    ja number_suffix
    inc r13
    jmp number_digits

number_zero:
    inc r13
    cmp r13, r14
    jae dispatch
    mov al, byte ptr [r13]
number_suffix:
    cmp al, '.'
    je number_fraction
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    jmp dispatch

number_fraction:
    inc r13
    cmp r13, r14
    jae parse_error
    mov al, byte ptr [r13]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
fraction_digits:
    inc r13
    cmp r13, r14
    jae dispatch
    mov al, byte ptr [r13]
    cmp al, '0'
    jb fraction_done
    cmp al, '9'
    jbe fraction_digits
fraction_done:
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    jmp dispatch

number_exponent:
    inc r13
    cmp r13, r14
    jae parse_error
    mov al, byte ptr [r13]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc r13
    cmp r13, r14
    jae parse_error
    mov al, byte ptr [r13]
exponent_first:
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
exponent_digits:
    inc r13
    cmp r13, r14
    jae dispatch
    mov al, byte ptr [r13]
    cmp al, '0'
    jb dispatch
    cmp al, '9'
    jbe exponent_digits
    jmp dispatch

at_eof:
    test r15, r15
    jnz parse_error
    cmp byte ptr [rbx], 1
    jne parse_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

parse_error:
    lea rax, [rip + input]
    sub r13, rax
    lea rsi, [rip + outbuf]
    mov byte ptr [rsi], 'e'
    mov byte ptr [rsi + 1], 'r'
    mov byte ptr [rsi + 2], 'r'
    mov byte ptr [rsi + 3], 'o'
    mov byte ptr [rsi + 4], 'r'
    mov byte ptr [rsi + 5], ' '
    mov byte ptr [rsi + 6], 'a'
    mov byte ptr [rsi + 7], 't'
    mov byte ptr [rsi + 8], ' '
    lea rdi, [rsi + 30]
    mov byte ptr [rdi], 10
    mov rax, r13
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal_loop
    lea r8, [rsi + 9]
decimal_copy:
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc rdi
    inc r8
    cmp al, 10
    jne decimal_copy
    mov rdx, r8
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
