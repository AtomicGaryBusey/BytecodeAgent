.globl _start

.bss
.balign 16
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r8, [rip + input_buffer]
    lea r9, [rip + output_buffer]
    xor r10d, r10d
    xor r13d, r13d
    xor r14d, r14d

scan:
    cmp r10, r12
    je end_input
    cmp byte ptr [r8 + r10], 10
    je newline
    inc r10
    jmp scan

newline:
    mov rbx, r10
reverse_line:
    cmp rbx, r13
    je finish_line
    dec rbx
    mov al, byte ptr [r8 + rbx]
    mov byte ptr [r9 + r14], al
    inc r14
    jmp reverse_line

finish_line:
    mov byte ptr [r9 + r14], 10
    inc r14
    inc r10
    mov r13, r10
    jmp scan

end_input:
    cmp r13, r12
    je write_output
    mov rbx, r12
reverse_final:
    cmp rbx, r13
    je finish_final
    dec rbx
    mov al, byte ptr [r8 + rbx]
    mov byte ptr [r9 + r14], al
    inc r14
    jmp reverse_final

finish_final:
    mov byte ptr [r9 + r14], 10
    inc r14

write_output:
    test r14, r14
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
