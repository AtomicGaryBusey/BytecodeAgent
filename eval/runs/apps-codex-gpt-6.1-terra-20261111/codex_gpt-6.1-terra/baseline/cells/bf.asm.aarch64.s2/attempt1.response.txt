.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #1
    lsl x21, x21, #16

read_loop:
    cmp x20, x21
    b.eq read_full
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_full:
    mov x0, #0
    adrp x1, extra_byte
    add x1, x1, :lo12:extra_byte
    mov x2, #1
    mov x8, #63
    svc #0
    cbnz x0, read_full

read_done:
    mov x22, #0
find_delimiter:
    cmp x22, x20
    b.eq delimiter_absent
    ldrb w0, [x19, x22]
    cmp w0, #33
    b.eq delimiter_found
    add x22, x22, #1
    b find_delimiter

delimiter_found:
    add x23, x22, #1
    b validate_setup

delimiter_absent:
    mov x23, x20

validate_setup:
    adrp x24, jump_table
    add x24, x24, :lo12:jump_table
    adrp x25, bracket_stack
    add x25, x25, :lo12:bracket_stack
    mov x26, #0
    mov x27, #0

validate_loop:
    cmp x27, x22
    b.eq validate_end
    ldrb w0, [x19, x27]
    cmp w0, #91
    b.eq open_bracket
    cmp w0, #93
    b.eq close_bracket
    add x27, x27, #1
    b validate_loop

open_bracket:
    str w27, [x25, x26, lsl #2]
    add x26, x26, #1
    add x27, x27, #1
    b validate_loop

close_bracket:
    cbz x26, invalid_program
    sub x26, x26, #1
    ldr w1, [x25, x26, lsl #2]
    add w2, w27, #1
    str w2, [x24, x1, lsl #2]
    add w2, w1, #1
    str w2, [x24, x27, lsl #2]
    add x27, x27, #1
    b validate_loop

validate_end:
    cbnz x26, invalid_program
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x27, #0
    mov x28, #0

execute_loop:
    cmp x27, x22
    b.hs execute_end
    ldrb w0, [x19, x27]
    cmp w0, #62
    b.eq command_right
    cmp w0, #60
    b.eq command_left
    cmp w0, #43
    b.eq command_plus
    cmp w0, #45
    b.eq command_minus
    cmp w0, #46
    b.eq command_output
    cmp w0, #44
    b.eq command_input
    cmp w0, #91
    b.eq command_open
    cmp w0, #93
    b.eq command_close

next_command:
    add x27, x27, #1
    b execute_loop

command_right:
    add x25, x25, #1
    b next_command

command_left:
    sub x25, x25, #1
    b next_command

command_plus:
    ldrb w1, [x25]
    add w1, w1, #1
    strb w1, [x25]
    b next_command

command_minus:
    ldrb w1, [x25]
    sub w1, w1, #1
    strb w1, [x25]
    b next_command

command_output:
    ldrb w1, [x25]
    strb w1, [x26, x28]
    add x28, x28, #1
    b next_command

command_input:
    cmp x23, x20
    b.hs input_empty
    ldrb w1, [x19, x23]
    add x23, x23, #1
    strb w1, [x25]
    b next_command

input_empty:
    strb wzr, [x25]
    b next_command

command_open:
    ldrb w1, [x25]
    cbnz w1, next_command
    ldr w27, [x24, x27, lsl #2]
    b execute_loop

command_close:
    ldrb w1, [x25]
    cbz w1, next_command
    ldr w27, [x24, x27, lsl #2]
    b execute_loop

execute_end:
    cbz x28, exit_success
    mov x0, #1
    mov x1, x26
    mov x2, x28
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0
