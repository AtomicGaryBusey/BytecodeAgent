.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14d, r14d
find_separator:
    cmp r14, r13
    jae no_separator
    cmp byte ptr [r12 + r14], 33
    je separator_found
    inc r14
    jmp find_separator

separator_found:
    lea r15, [r14 + 1]
    jmp validate_start

no_separator:
    mov r15, r13

validate_start:
    xor ebx, ebx
    xor ebp, ebp
    lea r8, [rip + bracket_stack]
    lea r9, [rip + jump_table]

validate_loop:
    cmp rbx, r14
    jae validate_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
validate_next:
    inc rbx
    jmp validate_loop

open_bracket:
    mov dword ptr [r8 + rbp*4], ebx
    inc rbp
    jmp validate_next

close_bracket:
    test rbp, rbp
    jz invalid_program
    dec rbp
    mov eax, dword ptr [r8 + rbp*4]
    mov dword ptr [r9 + rax*4], ebx
    mov dword ptr [r9 + rbx*4], eax
    jmp validate_next

validate_done:
    test rbp, rbp
    jnz invalid_program

    xor ebx, ebx
    lea rbp, [rip + tape]
    lea r10, [rip + output_buffer]
    xor r8d, r8d

execute_loop:
    cmp rbx, r14
    jae execution_done
    movzx eax, byte ptr [r12 + rbx]
    inc rbx
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment_cell
    cmp al, 45
    je decrement_cell
    cmp al, 46
    je output_cell
    cmp al, 44
    je input_cell
    cmp al, 91
    je loop_start
    cmp al, 93
    je loop_end
    jmp execute_loop

move_right:
    inc rbp
    jmp execute_loop

move_left:
    dec rbp
    jmp execute_loop

increment_cell:
    inc byte ptr [rbp]
    jmp execute_loop

decrement_cell:
    dec byte ptr [rbp]
    jmp execute_loop

output_cell:
    mov al, byte ptr [rbp]
    mov byte ptr [r10 + r8], al
    inc r8
    jmp execute_loop

input_cell:
    xor eax, eax
    cmp r15, r13
    jae input_store
    mov al, byte ptr [r12 + r15]
    inc r15
input_store:
    mov byte ptr [rbp], al
    jmp execute_loop

loop_start:
    cmp byte ptr [rbp], 0
    jne execute_loop
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    inc rbx
    jmp execute_loop

loop_end:
    cmp byte ptr [rbp], 0
    je execute_loop
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    inc rbx
    jmp execute_loop

execution_done:
    test r8, r8
    jz success
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    syscall

success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 16
input_buffer:
    .zero 65537
.balign 16
jump_table:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
.balign 16
output_buffer:
    .zero 65536
