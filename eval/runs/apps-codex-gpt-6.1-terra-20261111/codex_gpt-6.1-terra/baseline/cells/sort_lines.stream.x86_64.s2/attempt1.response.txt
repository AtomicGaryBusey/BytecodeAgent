.equ INPUT_CAP, 32769

.section .bss
.balign 16
input_buffer:
    .zero INPUT_CAP
.balign 16
lines:
    .zero 16000
.balign 16
scratch:
    .zero 16000
.balign 16
output_buffer:
    .zero 33768

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea r8, [rip + input_buffer]
    lea r9, [rip + lines]
    xor r10d, r10d
    xor ecx, ecx
    xor edx, edx

parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [r8 + rcx], 10
    jne parse_next
    lea rax, [r8 + rdx]
    mov qword ptr [r9 + r10], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [r9 + r10 + 8], rax
    add r10, 16
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_end:
    cmp rdx, r12
    je sort_setup
    lea rax, [r8 + rdx]
    mov qword ptr [r9 + r10], rax
    mov rax, r12
    sub rax, rdx
    mov qword ptr [r9 + r10 + 8], rax
    add r10, 16

sort_setup:
    shr r10, 4
    lea r12, [rip + lines]
    lea r13, [rip + scratch]
    mov r14d, 1

sort_pass:
    cmp r14, r10
    jae output_setup
    xor r15d, r15d

merge_group:
    cmp r15, r10
    jae pass_done
    mov r8, r15
    add r8, r14
    cmp r8, r10
    cmova r8, r10
    mov r9, r8
    add r9, r14
    cmp r9, r10
    cmova r9, r10
    mov rbx, r15
    mov rbp, r8
    mov rdx, r15

merge_loop:
    cmp rbx, r8
    jae copy_right
    cmp rbp, r9
    jae copy_left

    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov r11, qword ptr [r12 + rax + 8]
    mov rax, rbp
    shl rax, 4
    mov rdi, qword ptr [r12 + rax]
    mov rcx, qword ptr [r12 + rax + 8]
    mov rax, r11
    cmp rax, rcx
    cmova rax, rcx
    push rcx
    mov rcx, rax
    repe cmpsb
    pop rcx
    jne compare_bytes
    cmp r11, rcx
    jbe copy_left
    jmp copy_right

compare_bytes:
    jb copy_left
    jmp copy_right

copy_left:
    cmp rbx, r8
    jae group_done
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rdi, qword ptr [r12 + rax + 8]
    inc rbx
    jmp store_entry

copy_right:
    cmp rbp, r9
    jae group_done
    mov rax, rbp
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rdi, qword ptr [r12 + rax + 8]
    inc rbp

store_entry:
    mov rax, rdx
    shl rax, 4
    mov qword ptr [r13 + rax], rsi
    mov qword ptr [r13 + rax + 8], rdi
    inc rdx
    jmp merge_loop

group_done:
    mov r15, r9
    jmp merge_group

pass_done:
    xchg r12, r13
    shl r14, 1
    jmp sort_pass

output_setup:
    lea rdi, [rip + output_buffer]
    xor r8d, r8d

output_loop:
    cmp r8, r10
    jae write_output
    mov rax, r8
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rcx, qword ptr [r12 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r8
    jmp output_loop

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
