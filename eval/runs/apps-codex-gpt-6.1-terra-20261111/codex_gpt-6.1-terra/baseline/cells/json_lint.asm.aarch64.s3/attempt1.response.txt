.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    adrp x23, states
    add x23, x23, :lo12:states
    mov w24, #0

dispatch:
    cmp x21, x20
    b.eq at_eof
    ldrb w25, [x19, x21]
    cmp w24, #0
    b.eq value_ws
    cmp w24, #1
    b.eq array_first
    cmp w24, #2
    b.eq array_after
    cmp w24, #3
    b.eq object_first
    cmp w24, #4
    b.eq object_key
    cmp w24, #5
    b.eq object_colon
    cmp w24, #6
    b.eq object_after
    cmp w24, #7
    b.eq document_after
    b value_required

skip_ws:
    cmp w25, #0x20
    b.eq skip_one
    cmp w25, #9
    b.eq skip_one
    cmp w25, #10
    b.eq skip_one
    cmp w25, #13
    b.eq skip_one
    ret
skip_one:
    add x21, x21, #1
    b dispatch

value_ws:
value_required:
    bl skip_ws
    cmp w25, #'{'
    b.eq open_object
    cmp w25, #'['
    b.eq open_array
    cmp w25, #'"'
    b.eq value_string
    cmp w25, #'-'
    b.eq number_minus
    cmp w25, #'0'
    b.eq number_zero
    cmp w25, #'1'
    b.lo value_literal
    cmp w25, #'9'
    b.ls number_digits_start
value_literal:
    cmp w25, #'t'
    b.eq literal_true
    cmp w25, #'f'
    b.eq literal_false
    cmp w25, #'n'
    b.eq literal_null
    b error

open_array:
    mov w0, #2
    strb w0, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov w24, #1
    b dispatch

open_object:
    mov w0, #6
    strb w0, [x23, x22]
    add x22, x22, #1
    add x21, x21, #1
    mov w24, #3
    b dispatch

array_first:
    bl skip_ws
    cmp w25, #']'
    b.eq close_container
    mov w24, #8
    b dispatch

array_after:
    bl skip_ws
    cmp w25, #']'
    b.eq close_container
    cmp w25, #','
    b.ne error
    add x21, x21, #1
    mov w24, #8
    b dispatch

object_first:
    bl skip_ws
    cmp w25, #'}'
    b.eq close_container
    mov w24, #4
    b dispatch

object_key:
    bl skip_ws
    cmp w25, #'"'
    b.ne error
    mov w26, #1
    add x21, x21, #1
    b string_loop

object_colon:
    bl skip_ws
    cmp w25, #':'
    b.ne error
    add x21, x21, #1
    mov w24, #8
    b dispatch

object_after:
    bl skip_ws
    cmp w25, #'}'
    b.eq close_container
    cmp w25, #','
    b.ne error
    add x21, x21, #1
    mov w24, #4
    b dispatch

close_container:
    add x21, x21, #1
    sub x22, x22, #1
    b value_complete

document_after:
    bl skip_ws
    b error

value_string:
    mov w26, #0
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'"'
    b.eq string_end
    cmp w25, #0x20
    b.lo error
    cmp w25, #'\'
    b.eq string_escape
    add x21, x21, #1
    b string_loop
string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'u'
    b.eq string_unicode
    cmp w25, #'"'
    b.eq escape_ok
    cmp w25, #'\'
    b.eq escape_ok
    cmp w25, #'/'
    b.eq escape_ok
    cmp w25, #'b'
    b.eq escape_ok
    cmp w25, #'f'
    b.eq escape_ok
    cmp w25, #'n'
    b.eq escape_ok
    cmp w25, #'r'
    b.eq escape_ok
    cmp w25, #'t'
    b.ne error
escape_ok:
    add x21, x21, #1
    b string_loop
string_unicode:
    add x21, x21, #1
    mov w27, #4
unicode_loop:
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    sub w0, w25, #'0'
    cmp w0, #9
    b.ls unicode_digit
    orr w0, w25, #0x20
    sub w0, w0, #'a'
    cmp w0, #5
    b.hi error
unicode_digit:
    add x21, x21, #1
    subs w27, w27, #1
    b.ne unicode_loop
    b string_loop
string_end:
    add x21, x21, #1
    cbz w26, value_complete
    mov w24, #5
    b dispatch

number_minus:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'0'
    b.eq number_zero
    cmp w25, #'1'
    b.lo error
    cmp w25, #'9'
    b.hi error
number_digits_start:
    add x21, x21, #1
number_digits:
    cmp x21, x20
    b.eq value_complete
    ldrb w25, [x19, x21]
    sub w0, w25, #'0'
    cmp w0, #9
    b.hi number_suffix
    add x21, x21, #1
    b number_digits
number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_complete
    ldrb w25, [x19, x21]
number_suffix:
    cmp w25, #'.'
    b.eq number_fraction
    cmp w25, #'e'
    b.eq number_exponent
    cmp w25, #'E'
    b.eq number_exponent
    b value_complete
number_fraction:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    sub w0, w25, #'0'
    cmp w0, #9
    b.hi error
fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_complete
    ldrb w25, [x19, x21]
    sub w0, w25, #'0'
    cmp w0, #9
    b.ls fraction_digits
    cmp w25, #'e'
    b.eq number_exponent
    cmp w25, #'E'
    b.eq number_exponent
    b value_complete
number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'+'
    b.eq exponent_sign
    cmp w25, #'-'
    b.ne exponent_first
exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
exponent_first:
    sub w0, w25, #'0'
    cmp w0, #9
    b.hi error
exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_complete
    ldrb w25, [x19, x21]
    sub w0, w25, #'0'
    cmp w0, #9
    b.ls exponent_digits
    b value_complete

literal_true:
    adrp x27, lit_true
    add x27, x27, :lo12:lit_true
    mov x28, #4
    b literal_loop
literal_false:
    adrp x27, lit_false
    add x27, x27, :lo12:lit_false
    mov x28, #5
    b literal_loop
literal_null:
    adrp x27, lit_null
    add x27, x27, :lo12:lit_null
    mov x28, #4
literal_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    ldrb w1, [x27], #1
    cmp w0, w1
    b.ne error
    add x21, x21, #1
    subs x28, x28, #1
    b.ne literal_loop

value_complete:
    cbz x22, root_complete
    sub x0, x22, #1
    ldrb w24, [x23, x0]
    b dispatch
root_complete:
    mov w24, #7
    b dispatch

at_eof:
    cmp w24, #7
    b.ne error
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x0, #1
    mov x8, #64
    svc #0
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #31
    mov w0, #10
    strb w0, [x1]
    mov x2, #1
    mov x3, x21
    mov x4, #10
decimal_loop:
    udiv x5, x3, x4
    msub x6, x5, x4, x3
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    add x2, x2, #1
    mov x3, x5
    cbnz x3, decimal_loop
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
input:
    .zero 65537
states:
    .zero 1001
outbuf:
    .zero 32
