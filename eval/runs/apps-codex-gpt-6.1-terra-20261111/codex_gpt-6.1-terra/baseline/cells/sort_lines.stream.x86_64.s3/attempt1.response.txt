.section .bss
.balign 16
input:
    .zero 32769
output:
    .zero 32769
lines:
    .zero 16000
scratch:
    .zero 16000

.section .text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + input]
    lea r13, [rip + lines]
    xor r14d, r14d
    xor r8d, r8d
    xor r9d, r9d

parse_loop:
    cmp r8, r12
    jae parse_end
    cmp byte ptr [rbx + r8], 10
    jne parse_next
    lea rax, [rbx + r9]
    mov qword ptr [r13 + r14], rax
    mov rax, r8
    sub rax, r9
    mov qword ptr [r13 + r14 + 8], rax
    add r14, 16
    lea r9, [r8 + 1]
parse_next:
    inc r8
    jmp parse_loop

parse_end:
    cmp r9, r12
    je sort_setup
    lea rax, [rbx + r9]
    mov qword ptr [r13 + r14], rax
    mov rax, r12
    sub rax, r9
    mov qword ptr [r13 + r14 + 8], rax
    add r14, 16

sort_setup:
    mov r15, r14
    shr r15, 4
    lea r14, [rip + scratch]
    mov ebp, 1

sort_pass:
    cmp rbp, r15
    jae emit_setup
    xor r12d, r12d

merge_block:
    cmp r12, r15
    jae pass_done
    mov r8, r12
    lea r9, [r12 + rbp]
    cmp r9, r15
    cmova r9, r15
    mov r10, r9
    lea r11, [r9 + rbp]
    cmp r11, r15
    cmova r11, r15
    mov rbx, r12

merge_loop:
    cmp r8, r9
    jae copy_right
    cmp r10, r11
    jae copy_left

    mov rax, r8
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rdx, qword ptr [r13 + rax + 8]
    mov rax, r10
    shl rax, 4
    mov rdi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    mov rax, rdx
    cmp rax, rcx
    cmova rax, rcx
    xor r12d, r12d

compare_loop:
    cmp r12, rax
    jae compare_lengths
    movzx edx, byte ptr [rsi + r12]
    movzx ecx, byte ptr [rdi + r12]
    cmp edx, ecx
    jb copy_left
    ja copy_right
    inc r12
    jmp compare_loop

compare_lengths:
    mov rax, r8
    shl rax, 4
    mov rdx, qword ptr [r13 + rax + 8]
    mov rax, r10
    shl rax, 4
    cmp rdx, qword ptr [r13 + rax + 8]
    jbe copy_left

copy_right:
    mov rax, r10
    inc r10
    jmp copy_entry
copy_left:
    mov rax, r8
    inc r8
copy_entry:
    shl rax, 4
    mov rdx, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    mov rax, rbx
    shl rax, 4
    mov qword ptr [r14 + rax], rdx
    mov qword ptr [r14 + rax + 8], rcx
    inc rbx
    cmp r8, r9
    jb merge_loop
    cmp r10, r11
    jb merge_loop
    mov r12, r11
    jmp merge_block

pass_done:
    xchg r13, r14
    shl rbp, 1
    jmp sort_pass

emit_setup:
    lea rdi, [rip + output]
    xor ebx, ebx
emit_loop:
    cmp rbx, r15
    jae write_output
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp emit_loop

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
