.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x20
    cbnz x2, read_call
    adrp x1, extra
    add x1, x1, :lo12:extra
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    adrp x22, contexts
    add x22, x22, :lo12:contexts
    mov x23, #0
    mov w24, #0
    b parse_loop

// States: 0 value, 1 document end, 2 array first, 3 array next,
// 4 array after value, 5 object first, 6 object next,
// 7 object colon, 8 object value, 9 object after value.
parse_loop:
    cmp x21, x20
    b.eq at_end
    ldrb w25, [x19, x21]
    cmp w25, #0x20
    b.eq skip_space
    cmp w25, #9
    b.eq skip_space
    cmp w25, #10
    b.eq skip_space
    cmp w25, #13
    b.eq skip_space
    cmp w24, #0
    b.eq value_start
    cmp w24, #1
    b.eq error
    cmp w24, #2
    b.eq array_first
    cmp w24, #3
    b.eq value_start
    cmp w24, #4
    b.eq array_after
    cmp w24, #5
    b.eq object_first
    cmp w24, #6
    b.eq object_key
    cmp w24, #7
    b.eq object_colon
    cmp w24, #8
    b.eq value_start
    b object_after

skip_space:
    add x21, x21, #1
    b parse_loop

array_first:
    cmp w25, #']'
    b.eq close_container
    b value_start
array_after:
    cmp w25, #']'
    b.eq close_container
    cmp w25, #','
    b.ne error
    add x21, x21, #1
    mov w24, #3
    b parse_loop
object_first:
    cmp w25, #'}'
    b.eq close_container
object_key:
    cmp w25, #'"'
    b.ne error
    mov w26, #1
    b string_start
object_colon:
    cmp w25, #':'
    b.ne error
    add x21, x21, #1
    mov w24, #8
    b parse_loop
object_after:
    cmp w25, #'}'
    b.eq close_container
    cmp w25, #','
    b.ne error
    add x21, x21, #1
    mov w24, #6
    b parse_loop

value_start:
    cmp w25, #'{'
    b.eq open_object
    cmp w25, #'['
    b.eq open_array
    cmp w25, #'"'
    b.eq value_string
    cmp w25, #'t'
    b.eq literal_true
    cmp w25, #'f'
    b.eq literal_false
    cmp w25, #'n'
    b.eq literal_null
    cmp w25, #'-'
    b.eq number_minus
    cmp w25, #'0'
    b.lo error
    cmp w25, #'9'
    b.hi error
    b number_int

open_object:
    mov w26, #9
    b open_container
open_array:
    mov w26, #4
open_container:
    strb w26, [x22, x23]
    add x23, x23, #1
    add x21, x21, #1
    cmp w26, #4
    b.eq open_array_state
    mov w24, #5
    b parse_loop
open_array_state:
    mov w24, #2
    b parse_loop
close_container:
    add x21, x21, #1
    sub x23, x23, #1
    b value_done

value_string:
    mov w26, #0
string_start:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #0x20
    b.lo error
    cmp w25, #'"'
    b.eq string_end
    cmp w25, #0x5c
    b.eq string_escape
    add x21, x21, #1
    b string_loop
string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'"'
    b.eq escape_simple
    cmp w25, #0x5c
    b.eq escape_simple
    cmp w25, #'/'
    b.eq escape_simple
    cmp w25, #'b'
    b.eq escape_simple
    cmp w25, #'f'
    b.eq escape_simple
    cmp w25, #'n'
    b.eq escape_simple
    cmp w25, #'r'
    b.eq escape_simple
    cmp w25, #'t'
    b.eq escape_simple
    cmp w25, #'u'
    b.ne error
    add x21, x21, #1
    mov w27, #4
hex_loop:
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    sub w28, w25, #'0'
    cmp w28, #9
    b.ls hex_ok
    orr w28, w25, #0x20
    sub w28, w28, #'a'
    cmp w28, #5
    b.hi error
hex_ok:
    add x21, x21, #1
    subs w27, w27, #1
    b.ne hex_loop
    b string_loop
escape_simple:
    add x21, x21, #1
    b string_loop
string_end:
    add x21, x21, #1
    cbz w26, value_done
    mov w24, #7
    b parse_loop

literal_true:
    adrp x26, true_text
    add x26, x26, :lo12:true_text
    mov x27, #4
    b literal_loop
literal_false:
    adrp x26, false_text
    add x26, x26, :lo12:false_text
    mov x27, #5
    b literal_loop
literal_null:
    adrp x26, null_text
    add x26, x26, :lo12:null_text
    mov x27, #4
literal_loop:
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    ldrb w28, [x26], #1
    cmp w25, w28
    b.ne error
    add x21, x21, #1
    subs x27, x27, #1
    b.ne literal_loop
    b value_done

number_minus:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'0'
    b.lo error
    cmp w25, #'9'
    b.hi error
number_int:
    cmp w25, #'0'
    b.eq number_zero
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_done
    ldrb w25, [x19, x21]
    sub w28, w25, #'0'
    cmp w28, #9
    b.ls number_digits
    b number_fraction_check
number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_done
    ldrb w25, [x19, x21]
number_fraction_check:
    cmp w25, #'.'
    b.ne number_exp_check
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    sub w28, w25, #'0'
    cmp w28, #9
    b.hi error
number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_done
    ldrb w25, [x19, x21]
    sub w28, w25, #'0'
    cmp w28, #9
    b.ls number_fraction_digits
number_exp_check:
    cmp w25, #'e'
    b.eq number_exp
    cmp w25, #'E'
    b.ne value_done
number_exp:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
    cmp w25, #'+'
    b.eq number_exp_sign
    cmp w25, #'-'
    b.ne number_exp_first
number_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w25, [x19, x21]
number_exp_first:
    sub w28, w25, #'0'
    cmp w28, #9
    b.hi error
number_exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq value_done
    ldrb w25, [x19, x21]
    sub w28, w25, #'0'
    cmp w28, #9
    b.ls number_exp_digits
    b value_done

value_done:
    cbz x23, document_done
    sub x0, x23, #1
    ldrb w24, [x22, x0]
    b parse_loop
document_done:
    mov w24, #1
    b parse_loop

at_end:
    cmp w24, #1
    b.ne error
    adrp x1, ok_text
    add x1, x1, :lo12:ok_text
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    adrp x1, output
    add x1, x1, :lo12:output
    mov w0, #'e'
    strb w0, [x1]
    mov w0, #'r'
    strb w0, [x1, #1]
    strb w0, [x1, #2]
    mov w0, #'o'
    strb w0, [x1, #3]
    mov w0, #'r'
    strb w0, [x1, #4]
    mov w0, #' '
    strb w0, [x1, #5]
    mov w0, #'a'
    strb w0, [x1, #6]
    mov w0, #'t'
    strb w0, [x1, #7]
    mov w0, #' '
    strb w0, [x1, #8]
    add x3, x1, #30
    mov w4, #10
    mov x5, x21
decimal_loop:
    udiv x6, x5, x4
    msub x7, x6, x4, x5
    add w7, w7, #'0'
    strb w7, [x3, #-1]!
    mov x5, x6
    cbnz x5, decimal_loop
    add x2, x1, #9
decimal_copy:
    ldrb w0, [x3], #1
    strb w0, [x2], #1
    add x6, x1, #30
    cmp x3, x6
    b.ne decimal_copy
    mov w0, #10
    strb w0, [x2], #1
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_text:
    .ascii "ok\n"
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.bss
.balign 16
input:
    .zero 65536
extra:
    .zero 1
contexts:
    .zero 1001
output:
    .zero 32
