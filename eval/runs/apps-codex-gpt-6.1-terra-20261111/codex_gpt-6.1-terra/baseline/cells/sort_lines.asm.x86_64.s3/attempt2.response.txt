.section .bss
.balign 16
input:
    .zero 32769
records:
    .zero 16000
scratch:
    .zero 16000
output:
    .zero 32769

.section .text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea rbx, [rip + input]
    lea r13, [rip + records]
    xor r14d, r14d
    xor r8d, r8d
    xor r9d, r9d
parse_loop:
    cmp r8, r12
    jae parse_end
    cmp byte ptr [rbx + r8], 10
    jne parse_next
    mov rax, r14
    shl rax, 4
    lea rdx, [rbx + r9]
    mov qword ptr [r13 + rax], rdx
    mov rdx, r8
    sub rdx, r9
    mov qword ptr [r13 + rax + 8], rdx
    inc r14
    lea r9, [r8 + 1]
parse_next:
    inc r8
    jmp parse_loop
parse_end:
    cmp r9, r12
    je sort_setup
    mov rax, r14
    shl rax, 4
    lea rdx, [rbx + r9]
    mov qword ptr [r13 + rax], rdx
    sub r12, r9
    mov qword ptr [r13 + rax + 8], r12
    inc r14

sort_setup:
    lea r13, [rip + records]
    lea r15, [rip + scratch]
    mov r12d, 1
sort_pass:
    cmp r12, r14
    jae build_output
    xor r8d, r8d
merge_group:
    cmp r8, r14
    jae pass_done
    mov r9, r8
    lea r10, [r8 + r12]
    cmp r10, r14
    cmova r10, r14
    lea r11, [r10 + r12]
    cmp r11, r14
    cmova r11, r14
    mov rbx, r10
    mov rbp, r8
merge_loop:
    cmp r9, r10
    jae copy_right
    cmp rbx, r11
    jae copy_left

    mov rax, r9
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rdx, qword ptr [r13 + rax + 8]
    mov rax, rbx
    shl rax, 4
    mov rdi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    mov rax, rdx
    cmp rax, rcx
    cmova rax, rcx
    push r8
    xor r8d, r8d
compare_bytes:
    cmp r8, rax
    jae compare_lengths
    movzx r16d, byte ptr [rsi + r8]
compare_lengths:
    pop r8
    cmp rdx, rcx
    jbe choose_left
choose_right:
    mov rax, rbx
    inc rbx
    jmp store_record
choose_left:
    mov rax, r9
    inc r9
    jmp store_record
copy_right:
    cmp rbx, r11
    jae group_done
    mov rax, rbx
    inc rbx
    jmp store_record
copy_left:
    mov rax, r9
    inc r9
store_record:
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rdx, qword ptr [r13 + rax + 8]
    mov rax, rbp
    shl rax, 4
    mov qword ptr [r15 + rax], rsi
    mov qword ptr [r15 + rax + 8], rdx
    inc rbp
    jmp merge_loop
group_done:
    mov r8, r11
    jmp merge_group
pass_done:
    xchg r13, r15
    shl r12, 1
    jmp sort_pass

build_output:
    lea rdi, [rip + output]
    xor r8d, r8d
output_loop:
    cmp r8, r14
    jae write_output
    mov rax, r8
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r8
    jmp output_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
