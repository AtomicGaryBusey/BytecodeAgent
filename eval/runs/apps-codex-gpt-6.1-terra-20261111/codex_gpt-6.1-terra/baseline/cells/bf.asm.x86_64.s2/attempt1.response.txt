.text
.globl _start

_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
find_delimiter:
    cmp r13, r12
    jae no_delimiter
    cmp byte ptr [input_buffer + r13], 33
    je delimiter_found
    inc r13
    jmp find_delimiter

delimiter_found:
    lea r14, [r13 + 1]
    jmp validate_start

no_delimiter:
    mov r14, r12

validate_start:
    xor ebx, ebx
    xor r8d, r8d
validate_loop:
    cmp rbx, r13
    jae validate_done
    mov al, byte ptr [input_buffer + rbx]
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket
validate_next:
    inc rbx
    jmp validate_loop

open_bracket:
    mov dword ptr [bracket_stack + r8*4], ebx
    inc r8
    jmp validate_next

close_bracket:
    test r8, r8
    jz invalid
    dec r8
    mov eax, dword ptr [bracket_stack + r8*4]
    mov dword ptr [jump_table + rbx*4], eax
    mov dword ptr [jump_table + rax*4], ebx
    jmp validate_next

validate_done:
    test r8, r8
    jnz invalid
    xor ebx, ebx
    lea r15, [rip + tape]
    xor ebp, ebp

execute:
    cmp rbx, r13
    jae finish
    movzx eax, byte ptr [input_buffer + rbx]
    inc rbx
    cmp al, '>'
    je move_right
    cmp al, '<'
    je move_left
    cmp al, '+'
    je increment
    cmp al, '-'
    je decrement
    cmp al, '.'
    je output_byte
    cmp al, ','
    je input_byte
    cmp al, '['
    je loop_start
    cmp al, ']'
    je loop_end
    jmp execute

move_right:
    inc r15
    jmp execute

move_left:
    dec r15
    jmp execute

increment:
    inc byte ptr [r15]
    jmp execute

decrement:
    dec byte ptr [r15]
    jmp execute

output_byte:
    mov al, byte ptr [r15]
    mov byte ptr [output_buffer + rbp], al
    inc rbp
    jmp execute

input_byte:
    xor eax, eax
    cmp r14, r12
    jae input_store
    mov al, byte ptr [input_buffer + r14]
    inc r14
input_store:
    mov byte ptr [r15], al
    jmp execute

loop_start:
    cmp byte ptr [r15], 0
    jne execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute

loop_end:
    cmp byte ptr [r15], 0
    je execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute

finish:
    test rbp, rbp
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbp
    syscall
success:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero 65537
.balign 16
jump_table:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
.balign 16
output_buffer:
    .zero 65536
