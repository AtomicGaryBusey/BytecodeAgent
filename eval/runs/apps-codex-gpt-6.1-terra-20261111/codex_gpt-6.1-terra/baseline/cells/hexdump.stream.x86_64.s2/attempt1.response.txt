.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65536
output:
    .zero 327689

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jnz read_call
    # The input is at its maximum size; use an in-bounds byte for the EOF read.
    lea rsi, [rip + input]
    mov edx, 1
read_call:
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input]
    lea r14, [rip + output]
    xor r15d, r15d

row_loop:
    cmp r15, r12
    jae final_offset

    mov eax, r15d
    call print_offset
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    add r14, 2

    mov rbx, r12
    sub rbx, r15
    cmp rbx, 16
    jbe row_size_ready
    mov ebx, 16
row_size_ready:
    xor ecx, ecx
hex_loop:
    cmp rcx, rbx
    jae hex_blank
    movzx eax, byte ptr [r13 + r15]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [digits + rax]
    mov byte ptr [r14], al
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [r14 + 1], dl
    mov byte ptr [r14 + 2], ' '
    inc r15
    jmp hex_next
hex_blank:
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    mov byte ptr [r14 + 2], ' '
hex_next:
    add r14, 3
    inc ecx
    cmp ecx, 8
    jne no_middle_space
    mov byte ptr [r14], ' '
    inc r14
no_middle_space:
    cmp ecx, 16
    jb hex_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], '|'
    add r14, 2
    mov rdx, r15
    sub rdx, rbx
    xor ecx, ecx
ascii_loop:
    mov al, byte ptr [r13 + rdx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, '.'
ascii_store:
    mov byte ptr [r14], al
    inc r14
    inc rdx
    inc rcx
    cmp rcx, rbx
    jb ascii_loop
    mov byte ptr [r14], '|'
    mov byte ptr [r14 + 1], 10
    add r14, 2
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_offset
    mov byte ptr [r14], 10
    inc r14

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov ecx, 8
offset_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [r14], dl
    inc r14
    dec ecx
    jnz offset_loop
    ret
