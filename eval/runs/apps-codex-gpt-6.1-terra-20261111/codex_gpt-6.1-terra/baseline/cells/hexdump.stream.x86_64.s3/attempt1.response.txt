.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65536
output:
    .zero 324000

.text
.globl _start
_start:
    xor r12d, r12d
read_more:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jnz do_read
    # Probe for EOF using a valid one-byte buffer.
    lea rsi, [rip + output]
    mov edx, 1
do_read:
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r13, [rip + input]
    lea r14, [rip + output]
    mov rdi, r14
    xor r15d, r15d

row_loop:
    cmp r15, r12
    jae final_offset

    mov rax, r15
    call hex_offset
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2

    mov rbx, r12
    sub rbx, r15
    cmp rbx, 16
    jbe row_size_ready
    mov ebx, 16
row_size_ready:
    xor r8d, r8d
hex_slots:
    cmp r8, rbx
    jae empty_slot
    movzx eax, byte ptr [r13 + r15]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [digits + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [rdi + 1], dl
    inc r15
    jmp slot_space
empty_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
slot_space:
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    inc r8
    cmp r8, 8
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi
no_middle_space:
    cmp r8, 16
    jb hex_slots

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    mov r9, r15
    sub r9, rbx
ascii_loop:
    movzx eax, byte ptr [r13 + r9]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    cmp r9, r15
    jb ascii_loop
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    jmp row_loop

final_offset:
    mov rax, r12
    call hex_offset
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, r14
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall
    mov eax, 60
    xor edi, edi
    syscall

hex_offset:
    mov ecx, 28
offset_digit:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns offset_digit
    ret
