.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz print_result

    add r14, rax
    lea rsi, [rip + input_buffer]
    lea rdi, [rsi + rax]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, rdi
    jb scan_loop
    jmp read_loop

print_result:
    lea rdi, [rip + output_buffer]
    mov rax, r12
    call print_number
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_number
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_number
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

print_number:
    lea r8, [rip + digit_buffer + 20]
    mov r9d, 10
number_loop:
    xor edx, edx
    div r9
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz number_loop
    lea r9, [rip + digit_buffer + 20]
copy_digits:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb copy_digits
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
digit_buffer:
    .zero 20
