    ldr w1, [x25, x27, lsl #2]
    str w26, [x24, x1, lsl #2]

    add x25, x25, :lo12:tape

    ldrb w0, [x19, x26]
    add x26, x26, #1
    cmp w0, #62

    cmp w0, #60
    b.eq move_left
    cmp w0, #43
    b.eq increment
    cmp w0, #45
    b.eq decrement
    cmp w0, #46
    b.eq emit
    cmp w0, #44
    b.eq consume
    cmp w0, #91
    b.eq loop_open
