.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x20, output_buffer
    add x20, x20, :lo12:output_buffer
    adrp x21, hex_digits
    add x21, x21, :lo12:hex_digits
    mov x22, #0
    mov x23, #0

read_more:
    mov x0, #0
    add x1, x19, x23
    mov x2, #16
    sub x2, x2, x23
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x23, x23, x0
    cmp x23, #16
    b.ne read_more
    bl print_row
    add x22, x22, x23
    mov x23, #0
    b read_more

end_input:
    cbz x23, final_offset
    bl print_row
    add x22, x22, x23

final_offset:
    mov x24, x20
    mov x25, x22
    bl emit_offset
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_row:
    stp x29, x30, [sp, #-16]!
    mov x24, x20
    mov x25, x22
    bl emit_offset
    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1
    mov x26, #0

hex_loop:
    cmp x26, x23
    b.hs hex_padding
    ldrb w3, [x19, x26]
    lsr w4, w3, #4
    ldrb w4, [x21, w4, uxtw]
    strb w4, [x24], #1
    and w3, w3, #15
    ldrb w3, [x21, w3, uxtw]
    strb w3, [x24], #1
    mov w0, #' '
    strb w0, [x24], #1
    b hex_slot_done

hex_padding:
    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1

hex_slot_done:
    cmp x26, #7
    b.ne no_extra_space
    mov w0, #' '
    strb w0, [x24], #1
no_extra_space:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_loop
    mov w0, #' '
    strb w0, [x24], #1
    mov w0, #'|'
    strb w0, [x24], #1
    mov x26, #0

ascii_loop:
    cmp x26, x23
    b.hs ascii_done
    ldrb w3, [x19, x26]
    cmp w3, #0x20
    b.lo ascii_dot
    cmp w3, #0x7e
    b.ls ascii_store
ascii_dot:
    mov w3, #'.'
ascii_store:
    strb w3, [x24], #1
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w0, #'|'
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret

emit_offset:
    mov x3, #28
offset_loop:
    lsr x4, x25, x3
    and x4, x4, #15
    ldrb w4, [x21, x4]
    strb w4, [x24], #1
    subs x3, x3, #4
    b.ge offset_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 16
output_buffer:
    .zero 80
