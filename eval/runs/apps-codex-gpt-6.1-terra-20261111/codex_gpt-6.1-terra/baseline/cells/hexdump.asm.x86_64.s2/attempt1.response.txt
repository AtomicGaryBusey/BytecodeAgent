.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 65552
output_buffer:
    .zero 327700

.text
.globl _start
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65552
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r13, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    lea r15, [rip + digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae final_offset

    mov eax, ebx
    call emit_offset
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    add r14, 2

    mov rbp, r12
    sub rbp, rbx
    cmp rbp, 16
    jbe row_length_ready
    mov ebp, 16
row_length_ready:
    xor r8d, r8d
hex_loop:
    cmp r8, rbp
    jae empty_slot
    movzx eax, byte ptr [r13 + rbx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r14], al
    mov al, byte ptr [r15 + rdx]
    mov byte ptr [r14 + 1], al
    mov byte ptr [r14 + 2], ' '
    inc rbx
    jmp slot_done

empty_slot:
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    mov byte ptr [r14 + 2], ' '

slot_done:
    add r14, 3
    inc r8d
    cmp r8d, 8
    jne no_extra_space
    mov byte ptr [r14], ' '
    inc r14
no_extra_space:
    cmp r8d, 16
    jb hex_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], '|'
    add r14, 2

    mov r9, rbx
    sub r9, rbp
ascii_loop:
    mov al, byte ptr [r13 + r9]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [r14], al
    inc r14
    inc r9
    cmp r9, rbx
    jb ascii_loop

    mov byte ptr [r14], '|'
    mov byte ptr [r14 + 1], 10
    add r14, 2
    jmp row_loop

final_offset:
    mov eax, r12d
    call emit_offset
    mov byte ptr [r14], 10
    inc r14

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov ecx, 28
offset_digit:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14], dl
    inc r14
    sub ecx, 4
    jns offset_digit
    ret
