.globl _start

.bss
input:
    .zero 65537
output:
    .zero 65537

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r8, [rip + input]
    lea r9, [r8 + r12]
    lea r10, [rip + output]
    mov r13, r8
    mov r14, r8

scan_loop:
    cmp r14, r9
    je end_input
    cmp byte ptr [r14], 10
    je line_end
    inc r14
    jmp scan_loop

line_end:
    mov r15, r14
reverse_line:
    cmp r15, r13
    je append_newline
    dec r15
    mov al, byte ptr [r15]
    mov byte ptr [r10], al
    inc r10
    jmp reverse_line

append_newline:
    mov byte ptr [r10], 10
    inc r10
    inc r14
    mov r13, r14
    jmp scan_loop

end_input:
    cmp r13, r14
    je write_output
    mov r15, r14
reverse_final:
    cmp r15, r13
    je final_newline
    dec r15
    mov al, byte ptr [r15]
    mov byte ptr [r10], al
    inc r10
    jmp reverse_final

final_newline:
    mov byte ptr [r10], 10
    inc r10

write_output:
    lea rsi, [rip + output]
    mov rdx, r10
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
