.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_more:
    mov x0, #0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, finish_input

    add x21, x21, x0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    add x2, x1, x0

process_byte:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq

    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace

    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

whitespace:
    mov x22, #0

next_byte:
    cmp x1, x2
    b.ne process_byte
    b read_more

finish_input:
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    mov x24, x23

    mov x0, x19
    bl append_number
    mov w3, #32
    strb w3, [x24], #1

    mov x0, x20
    bl append_number
    mov w3, #32
    strb w3, [x24], #1

    mov x0, x21
    bl append_number
    mov w3, #10
    strb w3, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_number:
    sub sp, sp, #32
    add x4, sp, #32
    mov x5, #10
1:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x4, #-1]!
    mov x0, x6
    cbnz x0, 1b
    add x6, sp, #32
2:
    ldrb w7, [x4], #1
    strb w7, [x24], #1
    cmp x4, x6
    b.ne 2b
    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
