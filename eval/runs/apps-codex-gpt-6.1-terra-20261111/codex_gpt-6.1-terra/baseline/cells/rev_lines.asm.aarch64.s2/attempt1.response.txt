.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #65536

read_loop:
    add x1, x19, x20
    sub x2, x21, x20
    cbnz x2, read_call
    adrp x1, extra
    add x1, x1, :lo12:extra
    mov x2, #1
read_call:
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, process_input
    add x20, x20, x0
    b read_loop

process_input:
    adrp x22, output
    add x22, x22, :lo12:output
    mov x23, x22
    mov x24, x19
    add x25, x19, x20
    mov x26, x19

scan_loop:
    cmp x26, x25
    b.eq end_input
    ldrb w3, [x26]
    cmp w3, #10
    b.eq finish_line
    add x26, x26, #1
    b scan_loop

finish_line:
    mov x4, x26
reverse_line:
    cmp x4, x24
    b.eq line_reversed
    ldrb w3, [x4, #-1]!
    strb w3, [x23], #1
    b reverse_line
line_reversed:
    mov w3, #10
    strb w3, [x23], #1
    add x26, x26, #1
    mov x24, x26
    b scan_loop

end_input:
    cmp x24, x25
    b.eq write_output
    mov x4, x25
reverse_final:
    cmp x4, x24
    b.eq final_reversed
    ldrb w3, [x4, #-1]!
    strb w3, [x23], #1
    b reverse_final
final_reversed:
    mov w3, #10
    strb w3, [x23], #1

write_output:
    sub x2, x23, x22
    cbz x2, exit_program
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65536
extra:
    .zero 1
.balign 16
output:
    .zero 65537
