.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "

.bss
.balign 16
input:
    .zero 65537
states:
    .zero 1001
outbuf:
    .zero 32

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input]
    lea r14, [r13 + r12]
    lea r15, [rip + states]
    xor ebx, ebx

# State 0: document value; 1: document complete.
# Array: 2 initial, 3 value after comma, 4 delimiter.
# Object: 5 initial, 6 key after comma, 7 colon, 8 value, 9 delimiter.
dispatch:
    call skip_ws
    movzx eax, byte ptr [r15 + rbx]
    cmp eax, 0
    je need_value
    cmp eax, 1
    je document_done
    cmp eax, 2
    je array_initial
    cmp eax, 3
    je need_value
    cmp eax, 4
    je array_delimiter
    cmp eax, 5
    je object_initial
    cmp eax, 6
    je object_key
    cmp eax, 7
    je object_colon
    cmp eax, 8
    je need_value
    jmp object_delimiter

document_done:
    cmp r13, r14
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

array_initial:
    cmp r13, r14
    je error
    cmp byte ptr [r13], ']'
    je close_container
    jmp need_value

array_delimiter:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, ']'
    je close_container
    cmp al, ','
    jne error
    inc r13
    mov byte ptr [r15 + rbx], 3
    jmp dispatch

object_initial:
    cmp r13, r14
    je error
    cmp byte ptr [r13], '}'
    je close_container
object_key:
    cmp r13, r14
    je error
    cmp byte ptr [r13], '"'
    jne error
    call parse_string
    mov byte ptr [r15 + rbx], 7
    jmp dispatch

object_colon:
    cmp r13, r14
    je error
    cmp byte ptr [r13], ':'
    jne error
    inc r13
    mov byte ptr [r15 + rbx], 8
    jmp dispatch

object_delimiter:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '}'
    je close_container
    cmp al, ','
    jne error
    inc r13
    mov byte ptr [r15 + rbx], 6
    jmp dispatch

close_container:
    inc r13
    dec rbx
    jmp dispatch

need_value:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '{'
    je open_object
    cmp al, '['
    je open_array
    cmp al, '"'
    je string_value
    cmp al, 't'
    je true_value
    cmp al, 'f'
    je false_value
    cmp al, 'n'
    je null_value
    cmp al, '-'
    je number_value
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp number_value

open_object:
    call mark_value
    inc r13
    inc rbx
    mov byte ptr [r15 + rbx], 5
    jmp dispatch

open_array:
    call mark_value
    inc r13
    inc rbx
    mov byte ptr [r15 + rbx], 2
    jmp dispatch

string_value:
    call parse_string
    call mark_value
    jmp dispatch

true_value:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp literal_value
false_value:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp literal_value
null_value:
    lea rsi, [rip + lit_null]
    mov edx, 4
literal_value:
    cmp r13, r14
    je error
    mov al, byte ptr [rsi]
    cmp byte ptr [r13], al
    jne error
    inc r13
    inc rsi
    dec edx
    jnz literal_value
    call mark_value
    jmp dispatch

number_value:
    cmp byte ptr [r13], '-'
    jne number_int
    inc r13
number_int:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc r13
number_digits:
    cmp r13, r14
    je number_complete
    mov al, byte ptr [r13]
    cmp al, '0'
    jb number_fraction
    cmp al, '9'
    ja number_fraction
    inc r13
    jmp number_digits
number_zero:
    inc r13
    cmp r13, r14
    je number_complete
number_fraction:
    cmp byte ptr [r13], '.'
    jne number_exponent
    inc r13
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r13
fraction_digits:
    cmp r13, r14
    je number_complete
    mov al, byte ptr [r13]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    ja number_exponent
    inc r13
    jmp fraction_digits
number_exponent:
    mov al, byte ptr [r13]
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_complete
exponent_start:
    inc r13
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc r13
    cmp r13, r14
    je error
exponent_first:
    mov al, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r13
exponent_digits:
    cmp r13, r14
    je number_complete
    mov al, byte ptr [r13]
    cmp al, '0'
    jb number_complete
    cmp al, '9'
    ja number_complete
    inc r13
    jmp exponent_digits
number_complete:
    call mark_value
    jmp dispatch

mark_value:
    mov al, byte ptr [r15 + rbx]
    cmp al, 0
    je mark_document
    cmp al, 8
    je mark_object
    mov byte ptr [r15 + rbx], 4
    ret
mark_document:
    mov byte ptr [r15 + rbx], 1
    ret
mark_object:
    mov byte ptr [r15 + rbx], 9
    ret

skip_ws:
    cmp r13, r14
    je skip_done
    mov al, byte ptr [r13]
    cmp al, 0x20
    je skip_one
    cmp al, 0x09
    je skip_one
    cmp al, 0x0a
    je skip_one
    cmp al, 0x0d
    jne skip_done
skip_one:
    inc r13
    jmp skip_ws
skip_done:
    ret

parse_string:
    inc r13
string_loop:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb error
    cmp al, 0x5c
    je string_escape
    inc r13
    jmp string_loop
string_escape:
    inc r13
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, 'u'
    je unicode_escape
    cmp al, '"'
    je simple_escape
    cmp al, 0x5c
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    jne error
simple_escape:
    inc r13
    jmp string_loop
unicode_escape:
    inc r13
    mov edx, 4
unicode_digit:
    cmp r13, r14
    je error
    mov al, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe unicode_next
    or al, 0x20
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
unicode_next:
    inc r13
    dec edx
    jnz unicode_digit
    jmp string_loop
string_end:
    inc r13
    ret

error:
    lea rax, [rip + input]
    mov rax, r13
    lea rcx, [rip + input]
    sub rax, rcx
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ecx, 10
error_digits:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz error_digits
    mov r8, rsi
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov r9, rdx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
