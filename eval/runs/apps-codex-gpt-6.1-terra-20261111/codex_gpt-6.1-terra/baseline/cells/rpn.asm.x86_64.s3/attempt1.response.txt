.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    jnz read_more
    lea rsi, [rip + extra]
    mov edx, 1
read_more:
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    cmp r13, 65536
    jbe read_loop
read_done:
    lea r14, [rip + input]
    lea r15, [r14 + r13]
    lea rbx, [rip + output]
line_start:
    cmp r14, r15
    jae finish
    xor r8d, r8d
token_start:
    cmp r14, r15
    jae line_end
    movzx eax, byte ptr [r14]
    cmp al, 10
    je newline
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov rsi, r14
find_token_end:
    cmp r14, r15
    jae token_found
    movzx eax, byte ptr [r14]
    cmp al, 10
    je token_found
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r14
    jmp find_token_end

skip_separator:
    inc r14
    jmp token_start

token_found:
    mov rcx, r14
    sub rcx, rsi
    cmp rcx, 1
    jne parse_number
    mov al, byte ptr [rsi]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_number:
    xor r9d, r9d
    cmp byte ptr [rsi], '-'
    jne digits_start
    mov r9d, 1
    inc rsi
    dec rcx
digits_start:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
digits_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digits_loop
    test r9d, r9d
    jz push_number
    neg rax
push_number:
    lea rdx, [rip + values]
    mov qword ptr [rdx + r8*8], rax
    inc r8
    jmp token_start

operator:
    cmp r8, 2
    jb line_error
    lea rdx, [rip + values]
    mov r10, qword ptr [rdx + r8*8 - 8]
    mov rax, qword ptr [rdx + r8*8 - 16]
    dec r8
    movzx ecx, byte ptr [rsi]
    cmp cl, '+'
    je op_add
    cmp cl, '-'
    je op_sub
    cmp cl, '*'
    je op_mul
    test r10, r10
    jz line_error
    mov r11, 0x8000000000000000
    cmp rax, r11
    jne do_div
    cmp r10, -1
    je line_error
do_div:
    cqo
    idiv r10
    cmp cl, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
op_add:
    add rax, r10
    jmp store_result
op_sub:
    sub rax, r10
    jmp store_result
op_mul:
    imul rax, r10
store_result:
    lea rdx, [rip + values]
    mov qword ptr [rdx + r8*8 - 8], rax
    jmp token_start

newline:
    inc r14
line_end:
    cmp r8, 1
    jne emit_error
    lea rdx, [rip + values]
    mov rax, qword ptr [rdx]
    test rax, rax
    jns format_number
    mov byte ptr [rbx], '-'
    inc rbx
    neg rax
format_number:
    lea rsi, [rip + digits_end]
    mov ecx, 10
format_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz format_loop
copy_digits:
    mov al, byte ptr [rsi]
    mov byte ptr [rbx], al
    inc rsi
    inc rbx
    lea rdx, [rip + digits_end]
    cmp rsi, rdx
    jb copy_digits
    mov byte ptr [rbx], 10
    inc rbx
    jmp line_start

line_error:
    cmp r14, r15
    jae emit_error
error_skip:
    mov al, byte ptr [r14]
    inc r14
    cmp al, 10
    je emit_error
    cmp r14, r15
    jb error_skip
emit_error:
    mov byte ptr [rbx], 'e'
    mov byte ptr [rbx + 1], 'r'
    mov byte ptr [rbx + 2], 'r'
    mov byte ptr [rbx + 3], 'o'
    mov byte ptr [rbx + 4], 'r'
    mov byte ptr [rbx + 5], 10
    add rbx, 6
    jmp line_start

finish:
    lea rsi, [rip + output]
    mov rdx, rbx
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input:
    .zero 65536
extra:
    .zero 1
.balign 16
values:
    .zero 8192
digits:
    .zero 32
digits_end:
.balign 16
output:
    .zero 393216
