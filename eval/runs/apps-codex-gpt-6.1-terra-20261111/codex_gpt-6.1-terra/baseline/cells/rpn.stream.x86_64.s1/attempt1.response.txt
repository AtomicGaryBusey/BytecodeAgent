.section .rodata
errstr:
.ascii "error\n"

.bss
.balign 8
input:
.zero 65537
values:
.zero 8008
outbuf:
.zero 393222
numberbuf:
.zero 32

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [r12 + r13]
    lea r15, [rip + outbuf]

line_loop:
    cmp r12, r14
    jae finish
    lea rbx, [rip + values]
    xor ebp, ebp

token_loop:
    cmp r12, r14
    jae line_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline_end
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov r8, r12
find_token_end:
    cmp r12, r14
    jae token_found
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_found
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r12
    jmp find_token_end

token_found:
    mov r9, r12
    sub r9, r8
    cmp r9, 1
    jne parse_literal
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_literal:
    xor r10d, r10d
    cmp byte ptr [r8], '-'
    jne literal_digits
    mov r10d, 1
    inc r8
    dec r9
literal_digits:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor edx, edx
digit_loop:
    movzx eax, byte ptr [r8]
    sub eax, '0'
    cmp eax, 9
    ja line_error
    imul rdx, rdx, 10
    add rdx, rax
    inc r8
    dec r9
    jnz digit_loop
    test r10d, r10d
    jz push_literal
    neg rdx
push_literal:
    mov qword ptr [rbx + rbp*8], rdx
    inc rbp
    jmp token_loop

operator:
    cmp rbp, 2
    jb line_error
    mov r10b, al
    mov r8, qword ptr [rbx + rbp*8 - 8]
    mov rax, qword ptr [rbx + rbp*8 - 16]
    dec rbp
    cmp r10b, '+'
    je op_add
    cmp r10b, '-'
    je op_sub
    cmp r10b, '*'
    je op_mul
    test r8, r8
    jz line_error
    mov r9, 0x8000000000000000
    cmp rax, r9
    jne do_div
    cmp r8, -1
    je line_error
do_div:
    cqo
    idiv r8
    cmp r10b, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
op_add:
    add rax, r8
    jmp store_result
op_sub:
    sub rax, r8
    jmp store_result
op_mul:
    imul rax, r8
store_result:
    mov qword ptr [rbx + rbp*8 - 8], rax
    jmp token_loop

skip_separator:
    inc r12
    jmp token_loop

newline_end:
    inc r12
line_end:
    cmp rbp, 1
    jne emit_error
    mov rax, qword ptr [rbx]
    lea rdi, [rip + numberbuf + 32]
    mov byte ptr [rdi - 1], 10
    dec rdi
    xor r8d, r8d
    test rax, rax
    jns magnitude_ready
    mov r8d, 1
    neg rax
magnitude_ready:
    mov ecx, 10
format_digit:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz format_digit
    test r8d, r8d
    jz copy_number
    dec rdi
    mov byte ptr [rdi], '-'
copy_number:
    lea rsi, [rip + numberbuf + 32]
    sub rsi, rdi
    mov rcx, rsi
    mov rsi, rdi
    mov rdi, r15
    rep movsb
    mov r15, rdi
    jmp line_loop

line_error:
    cmp r12, r14
    jae emit_error
    cmp byte ptr [r12], 10
    je error_newline
    inc r12
    jmp line_error
error_newline:
    inc r12
emit_error:
    lea rsi, [rip + errstr]
    mov rdi, r15
    mov ecx, 6
    rep movsb
    mov r15, rdi
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
