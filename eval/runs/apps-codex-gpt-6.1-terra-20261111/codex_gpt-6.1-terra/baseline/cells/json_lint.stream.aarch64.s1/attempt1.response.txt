.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, #1
    lsl x2, x2, #16
    sub x3, x20, x19
    sub x2, x2, x3
    add x2, x2, #1
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x20
    mov x20, x19
    adrp x22, states
    add x22, x22, :lo12:states
    mov w23, #0

main_loop:
    cmp x20, x21
    b.eq at_end
    ldrb w0, [x20]
    cmp w0, #0x20
    b.eq skip_ws
    cmp w0, #9
    b.eq skip_ws
    cmp w0, #10
    b.eq skip_ws
    cmp w0, #13
    b.eq skip_ws

    cmp w23, #0
    b.eq root_value
    cmp w23, #1
    b.eq error
    cmp w23, #2
    b.eq array_initial
    cmp w23, #3
    b.eq array_value
    cmp w23, #4
    b.eq array_after
    cmp w23, #5
    b.eq object_initial
    cmp w23, #6
    b.eq object_key
    cmp w23, #7
    b.eq object_colon
    cmp w23, #8
    b.eq object_value
    b object_after

skip_ws:
    add x20, x20, #1
    b main_loop

at_end:
    cmp w23, #1
    b.ne error
    b success

root_value:
    mov w23, #1
    b value

array_initial:
    cmp w0, #']'
    b.eq close_container
array_value:
    mov w23, #4
    b value

array_after:
    cmp w0, #']'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x20, x20, #1
    mov w23, #3
    b main_loop

object_initial:
    cmp w0, #'}'
    b.eq close_container
object_key:
    cmp w0, #'"'
    b.ne error
    mov w23, #7
    b string_start

object_colon:
    cmp w0, #':'
    b.ne error
    add x20, x20, #1
    mov w23, #8
    b main_loop

object_value:
    mov w23, #9
    b value

object_after:
    cmp w0, #'}'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x20, x20, #1
    mov w23, #6
    b main_loop

close_container:
    add x20, x20, #1
    ldrb w23, [x22, #-1]!
    b main_loop

value:
    cmp w0, #'{'
    b.eq open_object
    cmp w0, #'['
    b.eq open_array
    cmp w0, #'"'
    b.eq string_start
    cmp w0, #'t'
    b.eq literal_true
    cmp w0, #'f'
    b.eq literal_false
    cmp w0, #'n'
    b.eq literal_null
    cmp w0, #'-'
    b.eq number_minus
    cmp w0, #'0'
    b.lo error
    cmp w0, #'9'
    b.hi error
    b number_integer

open_object:
    strb w23, [x22], #1
    mov w23, #5
    add x20, x20, #1
    b main_loop

open_array:
    strb w23, [x22], #1
    mov w23, #2
    add x20, x20, #1
    b main_loop

string_start:
    add x20, x20, #1
string_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq string_end
    cmp w0, #0x20
    b.lo error
    cmp w0, #0x5c
    b.eq string_escape
    add x20, x20, #1
    b string_loop

string_end:
    add x20, x20, #1
    b main_loop

string_escape:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'u'
    b.eq unicode_start
    cmp w0, #'"'
    b.eq escape_end
    cmp w0, #0x5c
    b.eq escape_end
    cmp w0, #'/'
    b.eq escape_end
    cmp w0, #'b'
    b.eq escape_end
    cmp w0, #'f'
    b.eq escape_end
    cmp w0, #'n'
    b.eq escape_end
    cmp w0, #'r'
    b.eq escape_end
    cmp w0, #'t'
    b.ne error
escape_end:
    add x20, x20, #1
    b string_loop

unicode_start:
    add x20, x20, #1
    mov w1, #4
unicode_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    sub w2, w0, #'0'
    cmp w2, #9
    b.ls unicode_digit
    orr w2, w0, #0x20
    sub w2, w2, #'a'
    cmp w2, #5
    b.hi error
unicode_digit:
    add x20, x20, #1
    subs w1, w1, #1
    b.ne unicode_loop
    b string_loop

literal_true:
    adrp x1, true_text
    add x1, x1, :lo12:true_text
    mov w2, #4
    b literal_loop
literal_false:
    adrp x1, false_text
    add x1, x1, :lo12:false_text
    mov w2, #5
    b literal_loop
literal_null:
    adrp x1, null_text
    add x1, x1, :lo12:null_text
    mov w2, #4
literal_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne error
    add x20, x20, #1
    subs w2, w2, #1
    b.ne literal_loop
    b main_loop

number_minus:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo error
    cmp w0, #'9'
    b.hi error

number_integer:
    cmp w0, #'0'
    b.eq number_zero
number_digits:
    add x20, x20, #1
    cmp x20, x21
    b.eq main_loop
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number_digits
    b number_fraction_check
number_zero:
    add x20, x20, #1
    cmp x20, x21
    b.eq main_loop
    ldrb w0, [x20]
number_fraction_check:
    cmp w0, #'.'
    b.ne number_exponent_check
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
number_fraction_digits:
    add x20, x20, #1
    cmp x20, x21
    b.eq main_loop
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number_fraction_digits

number_exponent_check:
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne main_loop
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'+'
    b.eq exponent_sign
    cmp w0, #'-'
    b.ne exponent_first
exponent_sign:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
exponent_first:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
exponent_digits:
    add x20, x20, #1
    cmp x20, x21
    b.eq main_loop
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls exponent_digits
    b main_loop

success:
    mov x0, #1
    adrp x1, ok_text
    add x1, x1, :lo12:ok_text
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    sub x4, x20, x19
    adrp x5, output_buffer
    add x5, x5, :lo12:output_buffer
    add x6, x5, #31
    mov w7, #10
    strb w7, [x6, #-1]!
    mov x9, #10
decimal_loop:
    udiv x10, x4, x9
    msub x11, x10, x9, x4
    add w11, w11, #'0'
    strb w11, [x6, #-1]!
    mov x4, x10
    cbnz x4, decimal_loop
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x6
    add x2, x5, #31
    sub x2, x2, x6
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"
ok_text:
    .ascii "ok\n"
error_text:
    .ascii "error at "

.bss
.balign 16
input:
    .space 65537
states:
    .space 1001
output_buffer:
    .space 32
