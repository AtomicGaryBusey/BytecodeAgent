.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
.zero 65536
output_buffer:
.zero 327680

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jnz read_call
    lea rsi, [rip + output_buffer]
    mov edx, 1
read_call:
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    lea r15, [rip + hex_digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae final_offset
    mov rax, rbx
    call emit_offset
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    add r14, 2

    mov r8, r12
    sub r8, rbx
    cmp r8, 16
    jbe row_size_ready
    mov r8d, 16
row_size_ready:
    xor r9d, r9d
hex_loop:
    cmp r9, r8
    jae empty_slot
    movzx eax, byte ptr [r13 + rbx]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r14], al
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14 + 1], dl
    mov byte ptr [r14 + 2], ' '
    inc rbx
    jmp slot_done
empty_slot:
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    mov byte ptr [r14 + 2], ' '
slot_done:
    add r14, 3
    inc r9d
    cmp r9d, 8
    jne no_extra_space
    mov byte ptr [r14], ' '
    inc r14
no_extra_space:
    cmp r9d, 16
    jb hex_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], '|'
    add r14, 2
    mov r10, rbx
    sub r10, r8
    xor r9d, r9d
ascii_loop:
    mov al, byte ptr [r13 + r10]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [r14], al
    inc r14
    inc r10
    inc r9
    cmp r9, r8
    jb ascii_loop
    mov byte ptr [r14], '|'
    mov byte ptr [r14 + 1], 10
    add r14, 2
    jmp row_loop

final_offset:
    mov rax, r12
    call emit_offset
    mov byte ptr [r14], 10
    inc r14
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov ecx, 28
offset_digit:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14], dl
    inc r14
    sub ecx, 4
    jns offset_digit
    ret
