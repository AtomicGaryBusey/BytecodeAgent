.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [r12 + r13]
    lea r15, [rip + output]
line_loop:
    cmp r12, r14
    jae finish
    xor ebx, ebx
    lea rbp, [rip + values]
token_loop:
    cmp r12, r14
    jae line_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je consume_newline
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov r8, r12
find_token_end:
    cmp r12, r14
    jae token_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_end
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    inc r12
    jmp find_token_end

token_end:
    mov r9, r12
    sub r9, r8
    cmp r9, 1
    jne parse_literal
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_literal:
    xor r10d, r10d
    cmp byte ptr [r8], '-'
    jne literal_digits
    mov r10d, 1
    inc r8
    dec r9
literal_digits:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r8
    dec r9
    jnz digit_loop
    test r10d, r10d
    jz push_literal
    neg rax
push_literal:
    mov qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp token_loop

operator:
    cmp rbx, 2
    jb line_error
    mov r10b, al
    sub rbx, 2
    mov rax, qword ptr [rbp + rbx*8]
    mov r8, qword ptr [rbp + rbx*8 + 8]
    cmp r10b, '+'
    je op_add
    cmp r10b, '-'
    je op_sub
    cmp r10b, '*'
    je op_mul
    test r8, r8
    jz line_error
    mov r9, 0x8000000000000000
    cmp rax, r9
    jne do_div
    cmp r8, -1
    je line_error
do_div:
    cqo
    idiv r8
    cmp r10b, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
op_add:
    add rax, r8
    jmp store_result
op_sub:
    sub rax, r8
    jmp store_result
op_mul:
    imul rax, r8
store_result:
    mov qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp token_loop

skip_separator:
    inc r12
    jmp token_loop
consume_newline:
    inc r12
line_end:
    cmp rbx, 1
    jne emit_error
    mov rax, qword ptr [rbp]
    test rax, rax
    jns format_number
    mov byte ptr [r15], '-'
    inc r15
    neg rax
format_number:
    lea rdi, [rip + digits + 20]
    mov r8d, 10
format_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz format_loop
    lea rsi, [rip + digits + 20]
copy_digits:
    cmp rdi, rsi
    je number_done
    mov al, byte ptr [rdi]
    mov byte ptr [r15], al
    inc rdi
    inc r15
    jmp copy_digits
number_done:
    mov byte ptr [r15], 10
    inc r15
    jmp line_loop

line_error:
    cmp r12, r14
    jae emit_error
error_skip:
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    je emit_error
    cmp r12, r14
    jb error_skip
emit_error:
    mov byte ptr [r15], 'e'
    mov byte ptr [r15 + 1], 'r'
    mov byte ptr [r15 + 2], 'r'
    mov byte ptr [r15 + 3], 'o'
    mov byte ptr [r15 + 4], 'r'
    mov byte ptr [r15 + 5], 10
    add r15, 6
    jmp line_loop

finish:
    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input:
    .zero 65537
output:
    .zero 393216
.balign 8
values:
    .zero 8192
digits:
    .zero 20
