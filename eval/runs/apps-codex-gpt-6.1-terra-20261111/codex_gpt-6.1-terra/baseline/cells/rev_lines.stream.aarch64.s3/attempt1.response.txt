.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output
    add x21, x21, :lo12:output
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x22, x20
    b.eq end_input
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq line_end
    add x22, x22, #1
    b scan_loop

line_end:
    mov x25, x22
reverse_line:
    cmp x25, x23
    b.eq append_newline
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x24]
    add x24, x24, #1
    b reverse_line

append_newline:
    mov w0, #10
    strb w0, [x21, x24]
    add x24, x24, #1
    cmp x22, x20
    b.eq write_output
    add x22, x22, #1
    mov x23, x22
    b scan_loop

end_input:
    cmp x23, x22
    b.ne line_end

write_output:
    cbz x24, exit_program
    mov x0, #1
    mov x1, x21
    mov x2, x24
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 65537
