.globl _start

.bss
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz process_input
    add r12, rax
    jmp read_loop

process_input:
    lea r8, [rip + input_buffer]
    lea r9, [rip + output_buffer]
    xor r10d, r10d
    xor r13d, r13d
    xor r14d, r14d

scan_loop:
    cmp r10, r12
    je end_of_input
    cmp byte ptr [r8 + r10], 10
    je line_end
    inc r10
    jmp scan_loop

line_end:
    mov r15, r10
reverse_line:
    cmp r15, r13
    je append_newline
    dec r15
    mov al, byte ptr [r8 + r15]
    mov byte ptr [r9 + r14], al
    inc r14
    jmp reverse_line

append_newline:
    mov byte ptr [r9 + r14], 10
    inc r14
    inc r10
    mov r13, r10
    jmp scan_loop

end_of_input:
    cmp r13, r12
    je write_output
    mov r15, r12

reverse_final:
    cmp r15, r13
    je final_newline
    dec r15
    mov al, byte ptr [r8 + r15]
    mov byte ptr [r9 + r14], al
    inc r14
    jmp reverse_final

final_newline:
    mov byte ptr [r9 + r14], 10
    inc r14

write_output:
    test r14, r14
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
