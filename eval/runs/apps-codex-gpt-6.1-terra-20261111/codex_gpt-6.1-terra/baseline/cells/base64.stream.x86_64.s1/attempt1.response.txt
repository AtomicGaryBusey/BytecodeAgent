.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
    .zero 49153
output:
    .zero 67000

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    lea r14, [rip + output]
    mov r15, r14
    lea rbx, [rip + alphabet]
    xor r8d, r8d
    xor r9d, r9d

group_loop:
    mov rax, r13
    sub rax, r8
    cmp rax, 3
    jb remainder

    movzx eax, byte ptr [r12 + r8]
    movzx edx, byte ptr [r12 + r8 + 1]
    movzx esi, byte ptr [r12 + r8 + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, esi
    add r8, 3

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 2], dl
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 3], al
    add r15, 4
    add r9d, 4
    cmp r9d, 76
    jne group_loop
    mov byte ptr [r15], 10
    inc r15
    xor r9d, r9d
    jmp group_loop

remainder:
    test rax, rax
    jz finish_line
    movzx eax, byte ptr [r12 + r8]
    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl
    and eax, 3
    shl eax, 4
    lea rdx, [r8 + 1]
    cmp rdx, r13
    je one_byte
    movzx edx, byte ptr [r12 + r8 + 1]
    mov esi, edx
    shr esi, 4
    or eax, esi
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 1], al
    and edx, 15
    shl edx, 2
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 2], dl
    mov byte ptr [r15 + 3], '='
    jmp remainder_done

one_byte:
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], '='
    mov byte ptr [r15 + 3], '='

remainder_done:
    add r15, 4
    add r9d, 4

finish_line:
    test r9d, r9d
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    mov rdx, r15
    sub rdx, r14
    jz done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

done:
    mov eax, 60
    xor edi, edi
    syscall
