.text
.globl _start

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    adrp x22, values
    add x22, x22, :lo12:values

line_start:
    cmp x21, x20
    b.hs finish
    mov x23, #0

token_start:
    cmp x21, x20
    b.hs line_end
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq newline_end
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator

    mov x24, x21
token_scan:
    cmp x21, x20
    b.hs token_scanned
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq token_scanned
    cmp w0, #32
    b.eq token_scanned
    cmp w0, #9
    b.eq token_scanned
    add x21, x21, #1
    b token_scan

skip_separator:
    add x21, x21, #1
    b token_start

token_scanned:
    sub x25, x21, x24
    ldrb w0, [x19, x24]
    cmp x25, #1
    b.ne parse_number
    cmp w0, #'+'
    b.eq operator_add
    cmp w0, #'-'
    b.eq operator_sub
    cmp w0, #'*'
    b.eq operator_mul
    cmp w0, #'/'
    b.eq operator_div
    cmp w0, #'%'
    b.eq operator_rem

parse_number:
    mov x26, #0
    mov x27, #0
    cmp w0, #'-'
    b.ne number_digits
    mov x27, #1
    add x24, x24, #1
    sub x25, x25, #1

number_digits:
    cbz x25, line_error
    cmp x25, #18
    b.hi line_error
    mov x28, #10

digit_loop:
    ldrb w0, [x19, x24]
    sub w0, w0, #'0'
    cmp w0, #9
    b.hi line_error
    madd x26, x26, x28, x0
    add x24, x24, #1
    subs x25, x25, #1
    b.ne digit_loop
    cbz x27, push_number
    neg x26, x26

push_number:
    str x26, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_start

operator_add:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x2, x23, #1
    ldr x0, [x22, x2, lsl #3]
    add x0, x0, x1
    str x0, [x22, x2, lsl #3]
    b token_start

operator_sub:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x2, x23, #1
    ldr x0, [x22, x2, lsl #3]
    sub x0, x0, x1
    str x0, [x22, x2, lsl #3]
    b token_start

operator_mul:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x2, x23, #1
    ldr x0, [x22, x2, lsl #3]
    mul x0, x0, x1
    str x0, [x22, x2, lsl #3]
    b token_start

operator_div:
    mov x3, #0
    b divide_common

operator_rem:
    mov x3, #1

divide_common:
    cmp x23, #2
    b.lo line_error
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x2, x23, #1
    ldr x0, [x22, x2, lsl #3]
    cbz x1, line_error
    cmn x1, #1
    b.ne divide_safe
    mov x4, #0x8000000000000000
    cmp x0, x4
    b.eq line_error
divide_safe:
    sdiv x4, x0, x1
    cbz x3, divide_store
    msub x4, x4, x1, x0
divide_store:
    str x4, [x22, x2, lsl #3]
    b token_start

newline_end:
    add x21, x21, #1

line_end:
    cmp x23, #1
    b.ne output_error
    ldr x0, [x22]
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #32
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x3, #0
    cmp x0, #0
    b.ge format_magnitude
    mov x3, #1
    neg x0, x0
format_magnitude:
    mov x4, #10
format_digit:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    mov x0, x5
    cbnz x0, format_digit
    cbz x3, format_write
    mov w6, #'-'
    strb w6, [x1, #-1]!
format_write:
    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    add x2, x2, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    b line_start

line_error:
    cmp x21, x20
    b.hs output_error
    ldrb w0, [x19, x21]
    add x21, x21, #1
    cmp w0, #10
    b.ne line_error

output_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0
    b line_start

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_msg:
    .ascii "error\n"

.bss
.balign 16
input:
    .zero 69632
.balign 16

    .zero 8008
.balign 16
