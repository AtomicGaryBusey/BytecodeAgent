.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "

.bss
.balign 16
buffer:
    .zero 65537
states:
    .zero 1002
outbuf:
    .zero 32

.text
_start:
    lea r14, [rip + buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d
    xor ebx, ebx
    lea r15, [rip + states]
    mov byte ptr [r15], 0

main_loop:
    cmp r12, r13
    jae at_eof
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x20
    je skip_ws
    cmp al, 9
    je skip_ws
    cmp al, 10
    je skip_ws
    cmp al, 13
    je skip_ws

    movzx edx, byte ptr [r15 + rbx]
    cmp edx, 0
    je root_value
    cmp edx, 1
    je error
    cmp edx, 2
    je array_first
    cmp edx, 3
    je array_value
    cmp edx, 4
    je array_delim
    cmp edx, 5
    je object_first
    cmp edx, 6
    je object_key
    cmp edx, 7
    je object_colon
    cmp edx, 8
    je object_value
    jmp object_delim

skip_ws:
    inc r12
    jmp main_loop

at_eof:
    test rbx, rbx
    jnz error
    cmp byte ptr [r15], 1
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

root_value:
    mov byte ptr [r15 + rbx], 1
    jmp parse_value

array_first:
    cmp al, ']'
    je close_container
array_value:
    mov byte ptr [r15 + rbx], 4
    jmp parse_value

array_delim:
    cmp al, ']'
    je close_container
    cmp al, ','
    jne error
    inc r12
    mov byte ptr [r15 + rbx], 3
    jmp main_loop

object_first:
    cmp al, '}'
    je close_container
object_key:
    cmp al, '"'
    jne error
    call parse_string
    mov byte ptr [r15 + rbx], 7
    jmp main_loop

object_colon:
    cmp al, ':'
    jne error
    inc r12
    mov byte ptr [r15 + rbx], 8
    jmp main_loop

object_value:
    mov byte ptr [r15 + rbx], 9
    jmp parse_value

object_delim:
    cmp al, '}'
    je close_container
    cmp al, ','
    jne error
    inc r12
    mov byte ptr [r15 + rbx], 6
    jmp main_loop

close_container:
    inc r12
    dec rbx
    jmp main_loop

parse_value:
    cmp al, '{'
    je open_object
    cmp al, '['
    je open_array
    cmp al, '"'
    je value_string
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp parse_number

open_object:
    inc r12
    inc rbx
    mov byte ptr [r15 + rbx], 5
    jmp main_loop

open_array:
    inc r12
    inc rbx
    mov byte ptr [r15 + rbx], 2
    jmp main_loop

value_string:
    call parse_string
    jmp main_loop

value_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4
parse_literal:
    cmp r12, r13
    jae error
    mov al, byte ptr [r14 + r12]
    cmp al, byte ptr [rsi]
    jne error
    inc r12
    inc rsi
    dec ecx
    jnz parse_literal
    jmp main_loop

parse_string:
    inc r12
string_loop:
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb error
    cmp al, '\\'
    je string_escape
    inc r12
    jmp string_loop

string_end:
    inc r12
    ret

string_escape:
    inc r12
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
    cmp al, 'u'
    je string_unicode
    cmp al, '"'
    je string_escape_ok
    cmp al, '\\'
    je string_escape_ok
    cmp al, '/'
    je string_escape_ok
    cmp al, 'b'
    je string_escape_ok
    cmp al, 'f'
    je string_escape_ok
    cmp al, 'n'
    je string_escape_ok
    cmp al, 'r'
    je string_escape_ok
    cmp al, 't'
    jne error
string_escape_ok:
    inc r12
    jmp string_loop

string_unicode:
    inc r12
    mov ecx, 4
unicode_loop:
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe unicode_digit
    or al, 0x20
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
unicode_digit:
    inc r12
    dec ecx
    jnz unicode_loop
    jmp string_loop

parse_number:
    cmp al, '-'
    jne number_int
    inc r12
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
number_int:
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
number_int_digits:
    inc r12
    cmp r12, r13
    jae main_loop
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_fraction_check
    cmp al, '9'
    jbe number_int_digits
    jmp number_fraction_check

number_zero:
    inc r12
    cmp r12, r13
    jae main_loop
    movzx eax, byte ptr [r14 + r12]

number_fraction_check:
    cmp al, '.'
    jne number_exp_check
    inc r12
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
number_frac_digits:
    inc r12
    cmp r12, r13
    jae main_loop
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb number_exp_check
    cmp al, '9'
    jbe number_frac_digits

number_exp_check:
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    jne main_loop
number_exp:
    inc r12
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    jne number_exp_first
number_exp_sign:
    inc r12
    cmp r12, r13
    jae error
    movzx eax, byte ptr [r14 + r12]
number_exp_first:
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
number_exp_digits:
    inc r12
    cmp r12, r13
    jae main_loop
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb main_loop
    cmp al, '9'
    jbe number_exp_digits
    jmp main_loop

error:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + error_msg]
    mov ecx, 9
    rep movsb
    lea r8, [rip + outbuf + 31]
    mov byte ptr [r8], 10
    mov rax, r12
    mov ecx, 10
error_digits:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz error_digits
error_copy:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp al, 10
    jne error_copy
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
