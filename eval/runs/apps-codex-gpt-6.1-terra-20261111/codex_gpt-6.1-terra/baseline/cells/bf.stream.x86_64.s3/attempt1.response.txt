.section .bss
.balign 16
input_buf:
    .zero 65537
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
output_buf:
    .zero 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
find_separator:
    cmp r13, r12
    jae no_separator
    cmp byte ptr [input_buf + r13], 33
    je separator_found
    inc r13
    jmp find_separator
separator_found:
    lea r14, [r13 + 1]
    jmp validate
no_separator:
    mov r14, r12

validate:
    xor ecx, ecx
    xor edx, edx
validate_loop:
    cmp rcx, r13
    jae validate_done
    movzx eax, byte ptr [input_buf + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
validate_next:
    inc rcx
    jmp validate_loop
open_bracket:
    mov dword ptr [bracket_stack + rdx*4], ecx
    inc edx
    jmp validate_next
close_bracket:
    test edx, edx
    jz invalid
    dec edx
    mov eax, dword ptr [bracket_stack + rdx*4]
    mov dword ptr [jump_table + rax*4], ecx
    mov dword ptr [jump_table + rcx*4], eax
    jmp validate_next
validate_done:
    test edx, edx
    jnz invalid

    xor ebx, ebx
    lea r15, [rip + tape]
    xor ebp, ebp
execute:
    cmp rbx, r13
    jae finish
    movzx eax, byte ptr [input_buf + rbx]
    inc rbx
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 46
    je output_byte
    cmp al, 44
    je input_byte
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
    jmp execute

move_right:
    inc r15
    jmp execute
move_left:
    dec r15
    jmp execute
increment:
    inc byte ptr [r15]
    jmp execute
decrement:
    dec byte ptr [r15]
    jmp execute
output_byte:
    mov al, byte ptr [r15]
    mov byte ptr [output_buf + rbp], al
    inc rbp
    jmp execute
input_byte:
    xor eax, eax
    cmp r14, r12
    jae input_store
    mov al, byte ptr [input_buf + r14]
    inc r14
input_store:
    mov byte ptr [r15], al
    jmp execute
loop_open:
    cmp byte ptr [r15], 0
    jne execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute
loop_close:
    cmp byte ptr [r15], 0
    je execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute

finish:
    test rbp, rbp
    jz exit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, rbp
    syscall
exit_success:
    mov eax, 60
    xor edi, edi
    syscall
invalid:
    mov eax, 60
    mov edi, 1
    syscall
