.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
    .zero 49153
output:
    .zero 66400

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_more

encode_start:
    lea r14, [rip + output]
    mov r15, r14
    lea rbx, [rip + alphabet]
    xor r8d, r8d
    mov r9, r13

encode_groups:
    cmp r9, 3
    jb encode_tail

    movzx eax, byte ptr [r12]
    movzx edx, byte ptr [r12 + 1]
    movzx ecx, byte ptr [r12 + 2]
    add r12, 3
    sub r9, 3

    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 2], dl

    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 3], al
    add r15, 4
    add r8d, 4
    cmp r8d, 76
    jne encode_groups
    mov byte ptr [r15], 10
    inc r15
    xor r8d, r8d
    jmp encode_groups

encode_tail:
    test r9, r9
    jz finish_line

    movzx eax, byte ptr [r12]
    shl eax, 16
    cmp r9, 1
    je tail_chars
    movzx edx, byte ptr [r12 + 1]
    shl edx, 8
    or eax, edx

tail_chars:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 1], dl

    mov byte ptr [r15 + 2], '='
    cmp r9, 1
    je tail_padding
    shr eax, 6
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r15 + 2], al

tail_padding:
    mov byte ptr [r15 + 3], '='
    add r15, 4
    add r8d, 4

finish_line:
    test r8d, r8d
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    mov rdx, r15
    sub rdx, r14
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
