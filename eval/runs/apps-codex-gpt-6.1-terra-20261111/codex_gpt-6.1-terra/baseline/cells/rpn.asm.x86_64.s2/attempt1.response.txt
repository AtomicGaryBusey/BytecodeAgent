.section .rodata
error_text:
.ascii "error\n"

.bss
.balign 16
input_buffer:
.zero 65536
read_buffer:
.zero 4096
value_stack:
.zero 8192
output_buffer:
.zero 32

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + read_buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz read_done
    mov rcx, rax
    lea rsi, [rip + read_buffer]
    lea rdi, [rip + input_buffer]
    add rdi, r12
    add r12, rax
    rep movsb
    jmp read_loop

read_done:
    lea r13, [rip + input_buffer]
    lea r14, [r13 + r12]

line_loop:
    cmp r13, r14
    jae exit_program
    mov r15, r13
find_line_end:
    cmp r15, r14
    jae line_found
    cmp byte ptr [r15], 10
    je line_found
    inc r15
    jmp find_line_end

line_found:
    xor r12d, r12d
token_loop:
    cmp r13, r15
    jae line_complete
    movzx eax, byte ptr [r13]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov rbx, r13
find_token_end:
    cmp r13, r15
    jae token_found
    movzx eax, byte ptr [r13]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r13
    jmp find_token_end

skip_separator:
    inc r13
    jmp token_loop

token_found:
    mov rdx, r13
    sub rdx, rbx
    cmp rdx, 1
    jne parse_number
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator

parse_number:
    xor r8d, r8d
    cmp byte ptr [rbx], '-'
    jne number_digits
    mov r8d, 1
    inc rbx
    dec rdx
number_digits:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error
    xor eax, eax
digit_loop:
    movzx ecx, byte ptr [rbx]
    sub ecx, '0'
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc rbx
    dec rdx
    jnz digit_loop
    test r8d, r8d
    jz push_number
    neg rax
push_number:
    lea rdi, [rip + value_stack]
    mov qword ptr [rdi + r12*8], rax
    inc r12
    jmp token_loop

operator:
    cmp r12, 2
    jb line_error
    lea rdi, [rip + value_stack]
    mov r8, qword ptr [rdi + r12*8 - 8]
    mov r9, qword ptr [rdi + r12*8 - 16]
    movzx ecx, byte ptr [rbx]
    cmp cl, '+'
    je do_add
    cmp cl, '-'
    je do_sub
    cmp cl, '*'
    je do_mul
    test r8, r8
    jz line_error
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne do_div
    cmp r8, -1
    je line_error
do_div:
    mov rax, r9
    cqo
    idiv r8
    cmp cl, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
do_add:
    lea rax, [r9 + r8]
    jmp store_result
do_sub:
    mov rax, r9
    sub rax, r8
    jmp store_result
do_mul:
    mov rax, r9
    imul rax, r8
store_result:
    dec r12
    mov qword ptr [rdi + r12*8 - 8], rax
    jmp token_loop

line_complete:
    cmp r12, 1
    jne line_error
    lea rdi, [rip + value_stack]
    mov rax, qword ptr [rdi]
    lea rsi, [rip + output_buffer + 32]
    dec rsi
    mov byte ptr [rsi], 10
    xor r8d, r8d
    test rax, rax
    jns format_digits
    mov r8d, 1
    neg rax
format_digits:
    mov ecx, 10
format_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz format_loop
    test r8d, r8d
    jz write_number
    dec rsi
    mov byte ptr [rsi], '-'
write_number:
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_text]
    mov edx, 6
    syscall

next_line:
    mov r13, r15
    cmp r13, r14
    jae exit_program
    inc r13
    jmp line_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
