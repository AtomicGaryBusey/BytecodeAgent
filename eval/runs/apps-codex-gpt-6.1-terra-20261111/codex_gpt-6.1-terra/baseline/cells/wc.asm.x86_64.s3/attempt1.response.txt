.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz finish_input

    add r14, rax
    lea rsi, [rip + buffer]
    lea rdi, [rsi + rax]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    sete dl
    movzx edx, dl
    add r12, rdx

    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    cmp rsi, rdi
    jb process_loop
    jmp read_loop

finish_input:
    lea rdi, [rip + output]
    mov rax, r12
    call format_number
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call format_number
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call format_number
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

format_number:
    lea rsi, [rip + digits + 20]
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + digits + 20]
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    cmp rsi, rdx
    jb 2b
    ret

.bss
.balign 16
buffer:
    .zero 65536
output:
    .zero 64
digits:
    .zero 20
