.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea rbx, [rip + input_buffer]
    lea r13, [rip + line_table]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rbx + rcx], 10
    jne parse_next
    mov qword ptr [r13 + r14*8], rdx
    inc r14
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [r13 + r14*8], rax
    inc r14
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, r12
    je sort_begin
    mov qword ptr [r13 + r14*8], rdx
    inc r14
    mov rax, r12
    sub rax, rdx
    mov qword ptr [r13 + r14*8], rax
    inc r14

sort_begin:
    shr r14, 1
    mov r15d, 1
sort_outer:
    cmp r15, r14
    jae output_begin
    mov rax, r15
    shl rax, 4
    mov r8, qword ptr [r13 + rax]
    mov r9, qword ptr [r13 + rax + 8]
    mov r10, r15
sort_inner:
    test r10, r10
    jz insert_line
    mov rax, r10
    dec rax
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rdi, qword ptr [r13 + rax + 8]
    xor ecx, ecx
compare_loop:
    cmp rcx, rdi
    jae insert_line
    cmp rcx, r9
    jae shift_previous
    lea rax, [rsi + rcx]
    movzx edx, byte ptr [rbx + rax]
    lea rax, [r8 + rcx]
    movzx eax, byte ptr [rbx + rax]
    cmp dl, al
    ja shift_previous
    jb insert_line
    inc rcx
    jmp compare_loop
shift_previous:
    mov rax, r10
    shl rax, 4
    mov rdx, qword ptr [r13 + rax - 16]
    mov qword ptr [r13 + rax], rdx
    mov rdx, qword ptr [r13 + rax - 8]
    mov qword ptr [r13 + rax + 8], rdx
    dec r10
    jmp sort_inner
insert_line:
    mov rax, r10
    shl rax, 4
    mov qword ptr [r13 + rax], r8
    mov qword ptr [r13 + rax + 8], r9
    inc r15
    jmp sort_outer

output_begin:
    lea rdi, [rip + output_buffer]
    xor r15d, r15d
output_loop:
    cmp r15, r14
    jae output_write
    mov rax, r15
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    add rsi, rbx
    mov rcx, qword ptr [r13 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r15
    jmp output_loop
output_write:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 32769
.balign 16
line_table:
    .zero 16000
.balign 16
output_buffer:
    .zero 32769
