.globl _start

.section .bss
.balign 8
input_buffer:
    .zero 65537
.balign 8
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
output_buffer:
    .zero 65536

.section .text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d

find_separator:
    cmp r13, r12
    jae no_separator
    cmp byte ptr [input_buffer + r13], 33
    je separator_found
    inc r13
    jmp find_separator

separator_found:
    lea r14, [r13 + 1]
    jmp validate

no_separator:
    mov r14, r12

validate:
    xor ecx, ecx
    xor edx, edx

validate_loop:
    cmp rcx, r13
    jae validate_done
    mov al, byte ptr [input_buffer + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    inc rcx
    jmp validate_loop

open_bracket:
    mov dword ptr [bracket_stack + rdx*4], ecx
    inc edx
    inc rcx
    jmp validate_loop

close_bracket:
    test edx, edx
    jz invalid
    dec edx
    mov eax, dword ptr [bracket_stack + rdx*4]
    mov dword ptr [jump_table + rax*4], ecx
    mov dword ptr [jump_table + rcx*4], eax
    inc rcx
    jmp validate_loop

validate_done:
    test edx, edx
    jnz invalid
    xor ebx, ebx
    lea r15, [rip + tape]
    xor ebp, ebp

execute:
    cmp rbx, r13
    jae finish
    mov al, byte ptr [input_buffer + rbx]
    inc rbx
    cmp al, 62
    je move_right
    cmp al, 60
    je move_left
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 46
    je output
    cmp al, 44
    je input
    cmp al, 91
    je loop_start
    cmp al, 93
    je loop_end
    jmp execute

move_right:
    inc r15
    jmp execute

move_left:
    dec r15
    jmp execute

increment:
    inc byte ptr [r15]
    jmp execute

decrement:
    dec byte ptr [r15]
    jmp execute

output:
    mov al, byte ptr [r15]
    mov byte ptr [output_buffer + rbp], al
    inc rbp
    jmp execute

input:
    xor eax, eax
    cmp r14, r12
    jae input_store
    mov al, byte ptr [input_buffer + r14]
    inc r14
input_store:
    mov byte ptr [r15], al
    jmp execute

loop_start:
    cmp byte ptr [r15], 0
    jne execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute

loop_end:
    cmp byte ptr [r15], 0
    je execute
    mov ebx, dword ptr [jump_table + rbx*4 - 4]
    inc rbx
    jmp execute

finish:
    test rbp, rbp
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbp
    syscall

success:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall
