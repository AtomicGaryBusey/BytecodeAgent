.text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x23, #0

read_loop:
    mov x0, #0
    mov x1, x19
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x22, x22, x0
    mov x24, x0
    mov x25, x19

byte_loop:
    ldrb w3, [x25], #1
    cmp w3, #10
    cinc x20, x20, eq
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace
    cbnz x23, next_byte
    add x21, x21, #1
    mov x23, #1
    b next_byte

whitespace:
    mov x23, #0

next_byte:
    subs x24, x24, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x27, x26

    mov x0, x20
    bl print_number
    mov w3, #32
    strb w3, [x27], #1

    mov x0, x21
    bl print_number
    mov w3, #32
    strb w3, [x27], #1

    mov x0, x22
    bl print_number
    mov w3, #10
    strb w3, [x27], #1

    mov x0, #1
    mov x1, x26
    sub x2, x27, x26
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_number:
    sub sp, sp, #32
    add x4, sp, #32
    mov x5, x4
    mov x6, #10
1:
    udiv x7, x0, x6
    msub x3, x7, x6, x0
    add w3, w3, #48
    strb w3, [x5, #-1]!
    mov x0, x7
    cbnz x0, 1b
2:
    ldrb w3, [x5], #1
    strb w3, [x27], #1
    cmp x5, x4
    b.ne 2b
    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .space 4096
output_buffer:
    .space 64
