.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    adrp x21, hex_digits
    add x21, x21, :lo12:hex_digits
    mov x22, #0
    mov x23, #0

read_more:
    mov x0, #0
    add x1, x19, x22
    mov x2, #16
    sub x2, x2, x22
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x22, x22, x0
    cmp x22, #16
    b.ne read_more
    bl print_row
    mov x22, #0
    b read_more

end_input:
    cbz x22, final_line
    bl print_row

final_line:
    mov x24, x20
    mov x25, x23
    bl write_offset
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_row:
    stp x29, x30, [sp, #-16]!
    mov x24, x20
    mov x25, x23
    bl write_offset
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    mov x26, #0

hex_loop:
    cmp x26, x22
    b.hs empty_slot
    ldrb w1, [x19, x26]
    lsr w2, w1, #4
    ldrb w2, [x21, x2]
    strb w2, [x24], #1
    and w1, w1, #15
    ldrb w1, [x21, x1]
    strb w1, [x24], #1
    mov w0, #32
    strb w0, [x24], #1
    b slot_done

empty_slot:
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1

slot_done:
    add x26, x26, #1
    cmp x26, #8
    b.ne no_extra_space
    mov w0, #32
    strb w0, [x24], #1
no_extra_space:
    cmp x26, #16
    b.ne hex_loop

    mov w0, #32
    strb w0, [x24], #1
    mov w0, #124
    strb w0, [x24], #1
    mov x26, #0

ascii_loop:
    ldrb w1, [x19, x26]
    cmp w1, #32
    b.lo nonprintable
    cmp w1, #126
    b.ls store_ascii
nonprintable:
    mov w1, #46
store_ascii:
    strb w1, [x24], #1
    add x26, x26, #1
    cmp x26, x22
    b.lo ascii_loop

    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x20
    sub x2, x24, x20
    mov x8, #64
    svc #0
    add x23, x23, x22
    ldp x29, x30, [sp], #16
    ret

write_offset:
    mov x9, #28
offset_loop:
    lsr x10, x25, x9
    and x10, x10, #15
    ldrb w10, [x21, x10]
    strb w10, [x24], #1
    subs x9, x9, #4
    b.pl offset_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 16
output_buf:
    .zero 80
