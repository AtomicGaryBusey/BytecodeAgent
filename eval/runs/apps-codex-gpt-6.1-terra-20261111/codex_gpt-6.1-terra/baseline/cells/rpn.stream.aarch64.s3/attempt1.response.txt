    ldrb w0, [x19]
    cmp w0, #10
    b.eq consume_newline
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9

    b.hs token_scanned
    ldrb w0, [x19]
    cmp w0, #10
    b.eq token_scanned
    cmp w0, #32
    b.eq token_scanned
    cmp w0, #9

    cmp x27, #1
    b.ne parse_number
    ldrb w0, [x26]
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42

    cmp w0, #47

    cmp w0, #37

    mov x2, x27

    b.lo line_error
    sub x24, x24, #1
    ldr x2, [x21, x24, lsl #3]
    sub x24, x24, #1
    ldr x1, [x21, x24, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42

    mov x3, #1
