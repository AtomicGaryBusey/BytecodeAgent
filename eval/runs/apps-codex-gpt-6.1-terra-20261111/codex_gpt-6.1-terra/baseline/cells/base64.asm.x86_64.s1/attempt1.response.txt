.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
    .zero 49153
output:
    .zero 66400

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz encode_start
    add r12, rax
    jmp read_loop

encode_start:
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    lea rbx, [rip + alphabet]
    xor r8d, r8d

group_loop:
    cmp r12, 3
    jb remainder
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx ecx, byte ptr [rsi + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl

    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], al

    add rsi, 3
    sub r12, 3
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne group_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp group_loop

remainder:
    test r12, r12
    jz finish_line
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 1
    je remainder_encode
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

remainder_encode:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], '='
    cmp r12, 1
    je remainder_pad
    shr eax, 6
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 2], al
remainder_pad:
    mov byte ptr [rdi + 3], '='
    add rdi, 4
    add r8d, 4

finish_line:
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz done
    mov eax, 1
    mov edi, 1
    syscall

done:
    mov eax, 60
    xor edi, edi
    syscall
