.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x20
    cbnz x2, read_call
    adrp x1, extra
    add x1, x1, :lo12:extra
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    adrp x22, states
    add x22, x22, :lo12:states
    mov x23, #0
    mov w24, #0

main_loop:
    bl skip_ws
    cmp w24, #0
    b.eq expect_value
    cmp w24, #1
    b.eq document_done
    cmp w24, #2
    b.eq array_first
    cmp w24, #3
    b.eq array_after
    cmp w24, #4
    b.eq array_value
    cmp w24, #5
    b.eq object_first
    cmp w24, #6
    b.eq object_key
    cmp w24, #7
    b.eq object_colon
    b object_after

document_done:
    cmp x21, x20
    b.ne error
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

array_first:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq close_container
    b expect_value
array_value:
    b expect_value
array_after:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x21, x21, #1
    mov w24, #4
    b main_loop
object_first:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq close_container
object_key:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.ne error
    bl parse_string
    mov w24, #7
    b main_loop
object_colon:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #':'
    b.ne error
    add x21, x21, #1
    mov w24, #0
    b main_loop
object_after:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq close_container
    cmp w0, #','
    b.ne error
    add x21, x21, #1
    mov w24, #6
    b main_loop

expect_value:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'['
    b.eq open_array
    cmp w0, #'{'
    b.eq open_object
    cmp w0, #'"'
    b.eq string_value
    cmp w0, #'t'
    b.eq true_value
    cmp w0, #'f'
    b.eq false_value
    cmp w0, #'n'
    b.eq null_value
    cmp w0, #'-'
    b.eq number_value
    cmp w0, #'0'
    b.lo error
    cmp w0, #'9'
    b.hi error
    b number_value

open_array:
    mov w0, #3
    strb w0, [x22, x23]
    add x23, x23, #1
    add x21, x21, #1
    mov w24, #2
    b main_loop
open_object:
    mov w0, #8
    strb w0, [x22, x23]
    add x23, x23, #1
    add x21, x21, #1
    mov w24, #5
    b main_loop
close_container:
    add x21, x21, #1
    sub x23, x23, #1
    b value_complete
string_value:
    bl parse_string
    b value_complete
true_value:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    mov x2, #4
    bl parse_literal
    b value_complete
false_value:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    mov x2, #5
    bl parse_literal
    b value_complete
null_value:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    mov x2, #4
    bl parse_literal
    b value_complete
number_value:
    bl parse_number
value_complete:
    cbz x23, root_complete
    sub x0, x23, #1
    ldrb w24, [x22, x0]
    b main_loop
root_complete:
    mov w24, #1
    b main_loop

skip_ws:
    cmp x21, x20
    b.eq skip_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq skip_one
    cmp w0, #0x09
    b.eq skip_one
    cmp w0, #0x0a
    b.eq skip_one
    cmp w0, #0x0d
    b.ne skip_done
skip_one:
    add x21, x21, #1
    b skip_ws
skip_done:
    ret

parse_literal:
    cbz x2, literal_done
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne error
    add x21, x21, #1
    sub x2, x2, #1
    b parse_literal
literal_done:
    ret

parse_string:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.lo error
    cmp w0, #'"'
    b.eq string_done
    cmp w0, #0x5c
    b.eq string_escape
    add x21, x21, #1
    b string_loop
string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq escape_simple
    cmp w0, #0x5c
    b.eq escape_simple
    cmp w0, #'/'
    b.eq escape_simple
    cmp w0, #'b'
    b.eq escape_simple
    cmp w0, #'f'
    b.eq escape_simple
    cmp w0, #'n'
    b.eq escape_simple
    cmp w0, #'r'
    b.eq escape_simple
    cmp w0, #'t'
    b.eq escape_simple
    cmp w0, #'u'
    b.ne error
    add x21, x21, #1
    mov w2, #4
hex_loop:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls hex_ok
    orr w0, w0, #0x20
    sub w1, w0, #'a'
    cmp w1, #5
    b.hi error
hex_ok:
    add x21, x21, #1
    subs w2, w2, #1
    b.ne hex_loop
    b string_loop
escape_simple:
    add x21, x21, #1
    b string_loop
string_done:
    add x21, x21, #1
    ret

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #'-'
    b.ne number_int
    add x21, x21, #1
number_int:
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.eq number_zero
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number_digits
    b number_frac_check
number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
number_frac_check:
    cmp w0, #'.'
    b.ne number_exp_check
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls frac_digits
number_exp_check:
    cmp w0, #'e'
    b.eq number_exp
    cmp w0, #'E'
    b.ne number_done
number_exp:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq exp_sign
    cmp w0, #'-'
    b.ne exp_require
exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error
    ldrb w0, [x19, x21]
exp_require:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls exp_digits
number_done:
    ret

error:
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #9
    mov x0, #1
    mov x8, #64
    svc #0
    adrp x3, outbuf
    add x3, x3, :lo12:outbuf
    add x4, x3, #32
    mov w0, #10
    strb w0, [x4, #-1]!
    mov x5, x21
    mov x6, #10
decimal_loop:
    udiv x7, x5, x6
    msub x0, x7, x6, x5
    add w0, w0, #'0'
    strb w0, [x4, #-1]!
    mov x5, x7
    cbnz x5, decimal_loop
    add x2, x3, #32
    sub x2, x2, x4
    mov x1, x4
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
input:
    .zero 65536
extra:
    .zero 1
states:
    .zero 1001
outbuf:
    .zero 32
