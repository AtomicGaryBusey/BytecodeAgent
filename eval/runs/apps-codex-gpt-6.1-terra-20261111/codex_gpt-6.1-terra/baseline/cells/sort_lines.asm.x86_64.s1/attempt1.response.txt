.section .bss
.balign 16
input_buffer:
    .zero 32769
.balign 16
line_table:
    .zero 16000
.balign 16
output_buffer:
    .zero 33768

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + input_buffer]
    lea r13, [rip + line_table]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rbx + rcx], 10
    jne parse_next
    lea rax, [rbx + rdx]
    mov qword ptr [r13 + r14], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [r13 + r14 + 8], rax
    add r14, 16
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, r12
    je sort_begin
    lea rax, [rbx + rdx]
    mov qword ptr [r13 + r14], rax
    mov rax, r12
    sub rax, rdx
    mov qword ptr [r13 + r14 + 8], rax
    add r14, 16

sort_begin:
    mov r12, 16
sort_outer:
    cmp r12, r14
    jae output_begin
    mov r15, r12
sort_inner:
    test r15, r15
    jz sort_next
    mov rsi, qword ptr [r13 + r15 - 16]
    mov rdi, qword ptr [r13 + r15]
    mov r8, qword ptr [r13 + r15 - 8]
    mov r9, qword ptr [r13 + r15 + 8]
    mov rcx, r8
    cmp rcx, r9
    cmova rcx, r9
    repe cmpsb
    jne compare_bytes
    cmp r8, r9
    jbe sort_next
    jmp swap_lines
compare_bytes:
    jb sort_next
swap_lines:
    mov rax, qword ptr [r13 + r15 - 16]
    mov rdx, qword ptr [r13 + r15 - 8]
    mov rcx, qword ptr [r13 + r15]
    mov r8, qword ptr [r13 + r15 + 8]
    mov qword ptr [r13 + r15 - 16], rcx
    mov qword ptr [r13 + r15 - 8], r8
    mov qword ptr [r13 + r15], rax
    mov qword ptr [r13 + r15 + 8], rdx
    sub r15, 16
    jmp sort_inner
sort_next:
    add r12, 16
    jmp sort_outer

output_begin:
    lea rdi, [rip + output_buffer]
    xor r12d, r12d
output_loop:
    cmp r12, r14
    jae output_done
    mov rsi, qword ptr [r13 + r12]
    mov rcx, qword ptr [r13 + r12 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r12, 16
    jmp output_loop
output_done:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
