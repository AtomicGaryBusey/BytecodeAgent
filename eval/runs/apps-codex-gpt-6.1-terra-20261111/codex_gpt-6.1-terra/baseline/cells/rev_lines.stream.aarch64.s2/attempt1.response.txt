.equ INPUT_CAPACITY, 65536
.equ OUTPUT_CAPACITY, 65537

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x20, output_buffer
    add x20, x20, :lo12:output_buffer
    mov x21, #0
    mov x22, #INPUT_CAPACITY

read_loop:
    mov x0, #0
    add x1, x19, x21
    sub x2, x22, x21
    cbnz x2, read_call
    adrp x1, read_scratch
    add x1, x1, :lo12:read_scratch
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x23, x19
    add x24, x19, x21
    mov x25, x20

scan_loop:
    cmp x23, x24
    b.eq output_done
    mov x26, x23

find_newline:
    cmp x23, x24
    b.eq reverse_line
    ldrb w0, [x23]
    cmp w0, #10
    b.eq reverse_line
    add x23, x23, #1
    b find_newline

reverse_line:
    mov x27, x23
reverse_loop:
    cmp x27, x26
    b.eq append_newline
    ldrb w0, [x27, #-1]!
    strb w0, [x25], #1
    b reverse_loop

append_newline:
    mov w0, #10
    strb w0, [x25], #1
    cmp x23, x24
    b.eq output_done
    add x23, x23, #1
    b scan_loop

output_done:
    sub x2, x25, x20
    cbz x2, exit_program
    mov x0, #1
    mov x1, x20
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buffer:
    .zero INPUT_CAPACITY
output_buffer:
    .zero OUTPUT_CAPACITY
read_scratch:
    .zero 1
