.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65536
output:
    .zero 327690

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jnz read_call
    mov edx, 1
    lea rsi, [rip + scratch]
read_call:
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input]
    lea r14, [rip + output]
    xor r15d, r15d

row_loop:
    cmp r15, r12
    jae final_offset

    mov rax, r15
    mov ecx, 28
offset_digits:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [r14], dl
    inc r14
    sub ecx, 4
    jns offset_digits

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
    add r14, 2

    mov rbx, r12
    sub rbx, r15
    cmp rbx, 16
    jbe row_size_ready
    mov ebx, 16
row_size_ready:
    xor r8d, r8d
hex_loop:
    cmp r8, rbx
    jae hex_blank
    movzx eax, byte ptr [r13 + r15]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [digits + rax]
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [r14], al
    mov byte ptr [r14 + 1], dl
    inc r15
    jmp hex_space
hex_blank:
    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], ' '
hex_space:
    mov byte ptr [r14 + 2], ' '
    add r14, 3
    inc r8d
    cmp r8d, 8
    jne no_middle_space
    mov byte ptr [r14], ' '
    inc r14
no_middle_space:
    cmp r8d, 16
    jb hex_loop

    mov byte ptr [r14], ' '
    mov byte ptr [r14 + 1], '|'
    add r14, 2
    mov r9, r15
    sub r9, rbx
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [r13 + r9]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, '.'
ascii_store:
    mov byte ptr [r14], al
    inc r14
    inc r9
    inc r8
    cmp r8, rbx
    jb ascii_loop
    mov byte ptr [r14], '|'
    mov byte ptr [r14 + 1], 10
    add r14, 2
    jmp row_loop

final_offset:
    mov rax, r12
    mov ecx, 28
final_digits:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [r14], dl
    inc r14
    sub ecx, 4
    jns final_digits
    mov byte ptr [r14], 10
    inc r14

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.bss
scratch:
    .zero 1
