export function l $crc32(l %p, l %n) {
@start
	jmp @oloop
@oloop
	%i =l phi @start 0, @odone %inext
	%crc =w phi @start -1, @odone %c8
	%cmp =w csltl %i, %n
	jnz %cmp, @obody, @done
@obody
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%c0 =w xor %crc, %b
	jmp @bloop
@bloop
	%k =w phi @obody 0, @bbody %knext
	%c =w phi @obody %c0, @bbody %cnext
	%kcmp =w csltw %k, 8
	jnz %kcmp, @bbody, @odone
@bbody
	%lsb =w and %c, 1
	%mask =w neg %lsb
	%shr =w shr %c, 1
	%pand =w and %mask, -306674912
	%cnext =w xor %shr, %pand
	%knext =w add %k, 1
	jmp @bloop
@odone
	%c8 =w copy %c
	%inext =l add %i, 1
	jmp @oloop
@done
	%flipped =w xor %crc, -1
	%result =l extuw %flipped
	ret %result
}
