mp_mul:
test rsi, rsi
jz .Lzero
test rcx, rcx
jz .Lzero
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbx, r8
xor ebp, ebp
.Louter:
mov r10, qword ptr [r12 + rbp*8]
xor r11, r11
xor ecx, ecx
.Linner:
mov rax, r10
mul qword ptr [r14 + rcx*8]
lea r9, [rbp + rcx]
test ebp, ebp
jz .Lskip_add
add rax, qword ptr [rbx + r9*8]
adc rdx, 0
.Lskip_add:
add rax, r11
adc rdx, 0
mov qword ptr [rbx + r9*8], rax
mov r11, rdx
inc rcx
cmp rcx, r15
jb .Linner
lea r9, [rbp + r15]
mov qword ptr [rbx + r9*8], r11
inc rbp
cmp rbp, r13
jb .Louter
lea rcx, [r13 + r15]
.Lsig:
test rcx, rcx
jz .Lret
cmp qword ptr [rbx + rcx*8 - 8], 0
jnz .Lret
dec rcx
jmp .Lsig
.Lret:
mov rax, rcx
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
.Lzero:
lea rdx, [rsi + rcx]
xor eax, eax
test rdx, rdx
jz .Lzero_done
xor ecx, ecx
.Lzero_loop:
mov qword ptr [r8 + rcx*8], 0
inc rcx
cmp rcx, rdx
jb .Lzero_loop
.Lzero_done:
ret
