lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbx, r8
    xor eax, eax
    xor r9, r9
.Lzero:
    mov qword ptr [rbx + r9*8], rax
    inc r9
    cmp r9, r15
    jle .Lzero
    xor r10, r10
.Louter:
    cmp r10, r13
    jge .Ldone
    movzx eax, byte ptr [r12 + r10]
    xor r11, r11
    mov r9, 1
.Linner:
    cmp r9, r15
    jg .Lnexti
    mov r8, qword ptr [rbx + r9*8]
    movzx esi, byte ptr [r14 + r9 - 1]
    cmp al, sil
    jne .Lnoteq
    lea rcx, [r11 + 1]
    mov qword ptr [rbx + r9*8], rcx
    jmp .Ladv
.Lnoteq:
    mov rcx, qword ptr [rbx + r9*8 - 8]
    cmp r8, rcx
    cmovg rcx, r8
    mov qword ptr [rbx + r9*8], rcx
.Ladv:
    mov r11, r8
    inc r9
    jmp .Linner
.Lnexti:
    inc r10
    jmp .Louter
.Ldone:
    mov rax, qword ptr [rbx + r15*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
