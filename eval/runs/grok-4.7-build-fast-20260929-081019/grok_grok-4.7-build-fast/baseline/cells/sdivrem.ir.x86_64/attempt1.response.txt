export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%one =l copy 1
	%min =l shl %one, 63
	%bz =w ceql %b, 0
	jnz %bz, @byzero, @chkmin
@byzero
	storel %a, %rem
	ret 0
@chkmin
	%amin =w ceql %a, %min
	jnz %amin, @chkneg, @dodiv
@chkneg
	%bn1 =w ceql %b, -1
	jnz %bn1, @ovf, @dodiv
@ovf
	storel 0, %rem
	ret %min
@dodiv
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
