crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
.Lbyte:
    movzx ecx, byte ptr [rdi]
    xor eax, ecx
    inc rdi
    mov ecx, 8
.Lbit:
    shr eax, 1
    jnc .Lskip
    xor eax, 0xEDB88320
.Lskip:
    dec ecx
    jnz .Lbit
    dec rsi
    jnz .Lbyte
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
