    mov x2, xzr
    mov x1, #1
.Louter:
    cmp x1, x0
    b.gt .Ldone
    mov x3, x1
    mov x4, xzr
.Linner:
    cmp x3, #1
    b.eq .Lnext
    tbnz x3, #0, .Lodd
    lsr x3, x3, #1
    b .Linc
.Lodd:
    add x3, x3, x3, lsl #1
    add x3, x3, #1
.Linc:
    add x4, x4, #1
    b .Linner
.Lnext:
    cmp x4, x2
    csel x2, x4, x2, gt
    add x1, x1, #1
    b .Louter
.Ldone:
    mov x0, x2
    ret
