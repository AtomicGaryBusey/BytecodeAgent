export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%bit0 =l and %y0, 1
	%odd0 =w cnel %bit0, 0
	%cnt0 =l extuw %odd0
	jmp @loop
@loop
	%i =l phi @start 1, @body %inext
	%sum =l phi @start %y0, @body %sumnext
	%minv =l phi @start %y0, @body %min1
	%maxv =l phi @start %y0, @body %max1
	%cnt =l phi @start %cnt0, @body %cntnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%ai =l loadl %ptr
	%yi =l call %f(l %ai)
	%sumnext =l add %sum, %yi
	%lt =w csltl %yi, %minv
	%ltl =l extuw %lt
	%mask =l sub 0, %ltl
	%xd =l xor %yi, %minv
	%md =l and %xd, %mask
	%min1 =l xor %minv, %md
	%gt =w csgtl %yi, %maxv
	%gtl =l extuw %gt
	%mask2 =l sub 0, %gtl
	%xd2 =l xor %yi, %maxv
	%md2 =l and %xd2, %mask2
	%max1 =l xor %maxv, %md2
	%bit =l and %yi, 1
	%odd =w cnel %bit, 0
	%oddl =l extuw %odd
	%cntnext =l add %cnt, %oddl
	%inext =l add %i, 1
	jmp @loop
@done
	storel %minv, %out
	%out1 =l add %out, 8
	storel %maxv, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
