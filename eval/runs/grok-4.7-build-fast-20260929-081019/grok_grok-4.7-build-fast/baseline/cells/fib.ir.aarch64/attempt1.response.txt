export function l $fib(l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @small, @prep
@small
	ret %n
@prep
	%a0 =l copy 0
	%b0 =l copy 1
	%i0 =l copy 1
	jmp @loop
@loop
	%a =l phi @prep %a0, @loop %a1
	%b =l phi @prep %b0, @loop %b1
	%i =l phi @prep %i0, @loop %i1
	%a1 =l copy %b
	%b1 =l add %a, %b
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @done
@done
	ret %b1
}
