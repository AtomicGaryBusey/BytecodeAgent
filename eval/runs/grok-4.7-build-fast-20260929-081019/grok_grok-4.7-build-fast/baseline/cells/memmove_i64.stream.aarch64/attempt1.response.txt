memmove_i64:
    cbz x2, .Lret
    cmp x0, x1
    b.hs .Lmaybe_back
.Lforward:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    b.ne .Lforward
    b .Lret
.Lmaybe_back:
    b.eq .Lret
.Lbackward:
    sub x2, x2, #1
    ldr x3, [x1, x2, lsl #3]
    str x3, [x0, x2, lsl #3]
    cbnz x2, .Lbackward
.Lret:
    mov x0, #0
    ret
