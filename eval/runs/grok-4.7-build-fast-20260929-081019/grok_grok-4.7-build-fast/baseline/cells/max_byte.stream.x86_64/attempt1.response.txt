max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Lret
    cmp rsi, 16
    jb .Lscalar
    pxor xmm0, xmm0
.Lvec:
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    cmp rsi, 16
    jae .Lvec
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
    test rsi, rsi
    jz .Lret
.Lscalar:
    movzx ecx, byte ptr [rdi]
    cmp eax, ecx
    cmovb eax, ecx
    add rdi, 1
    sub rsi, 1
    jnz .Lscalar
.Lret:
    ret
