export function l $qsort_pairs(l %a, l %n) {
@start
	%one =l copy 1
	%nle1 =w cslel %n, %one
	jnz %nle1, @ret0, @init
@ret0
	%zero =l copy 0
	ret %zero
@init
	%nm1 =l sub %n, %one
	%sixteen =l copy 16
	%eight =l copy 8
	%pivoff =l mul %nm1, %sixteen
	%pivptr =l add %a, %pivoff
	%pkey =l loadl %pivptr
	%pvalptr =l add %pivptr, %eight
	%pval =l loadl %pvalptr
	%zero2 =l copy 0
	jmp @loop
@loop
	%i =l phi @init %zero2, @notless %isame, @swap %inext
	%j =l phi @init %zero2, @notless %jpass1, @swap %jpass2
	%jdone =w csgel %j, %nm1
	jnz %jdone, @partend, @body
@body
	%joff =l mul %j, %sixteen
	%jptr =l add %a, %joff
	%jkey =l loadl %jptr
	%jvalptr =l add %jptr, %eight
	%jval =l loadl %jvalptr
	%jnext =l add %j, %one
	%klt =w csltl %jkey, %pkey
	jnz %klt, @swap, @kge
@kge
	%keq =w ceql %jkey, %pkey
	jnz %keq, @cmpv, @notless
@cmpv
	%vlt =w csltl %jval, %pval
	jnz %vlt, @swap, @notless
@notless
	%isame =l copy %i
	%jpass1 =l copy %jnext
	jmp @loop
@swap
	%ioff =l mul %i, %sixteen
	%iptr =l add %a, %ioff
	%ikey =l loadl %iptr
	%ivalptr =l add %iptr, %eight
	%ival =l loadl %ivalptr
	storel %jkey, %iptr
	storel %jval, %ivalptr
	storel %ikey, %jptr
	storel %ival, %jvalptr
	%inext =l add %i, %one
	%jpass2 =l copy %jnext
	jmp @loop
@partend
	%ioff2 =l mul %i, %sixteen
	%iptr2 =l add %a, %ioff2
	%ikey2 =l loadl %iptr2
	%ivalptr2 =l add %iptr2, %eight
	%ival2 =l loadl %ivalptr2
	storel %pkey, %iptr2
	storel %pval, %ivalptr2
	storel %ikey2, %pivptr
	storel %ival2, %pvalptr
	%hasleft =w csgtl %i, %zero2
	jnz %hasleft, @doleft, @afterleft
@doleft
	%left =l call $qsort_pairs(l %a, l %i)
	jmp @afterleft
@afterleft
	%ip1 =l add %i, %one
	%rightn =l sub %n, %ip1
	%hasright =w csgtl %rightn, %zero2
	jnz %hasright, @doright, @ret0
@doright
	%roff =l mul %ip1, %sixteen
	%rptr =l add %a, %roff
	%right =l call $qsort_pairs(l %rptr, l %rightn)
	jmp @ret0
}
