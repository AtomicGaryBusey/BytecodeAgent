export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @rep_store %i_next, @lit_done %j
	%wri =l phi @start 0, @rep_store %wri_next, @lit_done %wri_lit
	%atend =w csgel %i, %n
	jnz %atend, @ret_ok, @getb
@ret_ok
	ret %wri
@getb
	%pi =l add %src, %i
	%bi =w loadub %pi
	jmp @run1
@run1
	%c =w phi @getb 0, @run1inc %c1
	%c128 =w csgew %c, 128
	jnz %c128, @run1done, @run1n
@run1n
	%cl =l extuw %c
	%pos =l add %i, %cl
	%posn =w csgel %pos, %n
	jnz %posn, @run1done, @run1eq
@run1eq
	%pp =l add %src, %pos
	%bb =w loadub %pp
	%eq =w ceqw %bb, %bi
	jnz %eq, @run1inc, @run1done
@run1inc
	%c1 =w add %c, 1
	jmp @run1
@run1done
	%r =w phi @run1 %c, @run1n %c, @run1eq %c
	%isrep =w csgew %r, 3
	jnz %isrep, @emit_rep, @find_lit
@emit_rep
	%sum2 =l add %wri, 2
	%over =w csgtl %sum2, %cap
	jnz %over, @fail, @rep_store
@fail
	%neg =l copy -1
	ret %neg
@rep_store
	%hdrw =w sub 257, %r
	%op =l add %out, %wri
	storeb %hdrw, %op
	%op1 =l add %op, 1
	storeb %bi, %op1
	%wri_next =l add %wri, 2
	%rl =l extuw %r
	%i_next =l add %i, %rl
	jmp @loop
@find_lit
	jmp @jloop
@jloop
	%j =l phi @find_lit %i, @jinc %j1
	%jcnt =w phi @find_lit 0, @jinc %jcnt1
	%jge =w csgel %j, %n
	jnz %jge, @lit_emit, @jlen
@jlen
	%ljge =w csgew %jcnt, 128
	jnz %ljge, @lit_emit, @jgetb
@jgetb
	%pj =l add %src, %j
	%bj =w loadub %pj
	jmp @run2
@run2
	%c2 =w phi @jgetb 0, @run2inc %c2n
	%c2ge =w csgew %c2, 128
	jnz %c2ge, @run2done, @run2n
@run2n
	%c2l =l extuw %c2
	%pos2 =l add %j, %c2l
	%pos2ge =w csgel %pos2, %n
	jnz %pos2ge, @run2done, @run2eq
@run2eq
	%pp2 =l add %src, %pos2
	%bb2 =w loadub %pp2
	%eq2 =w ceqw %bb2, %bj
	jnz %eq2, @run2inc, @run2done
@run2inc
	%c2n =w add %c2, 1
	jmp @run2
@run2done
	%rj =w phi @run2 %c2, @run2n %c2, @run2eq %c2
	%rjge =w csgew %rj, 3
	jnz %rjge, @lit_emit, @jinc
@jinc
	%j1 =l add %j, 1
	%jcnt1 =w add %jcnt, 1
	jmp @jloop
@lit_emit
	%llen =l extuw %jcnt
	%pktsize =l add %llen, 1
	%neww =l add %wri, %pktsize
	%nover =w csgtl %neww, %cap
	jnz %nover, @fail, @lit_store
@lit_store
	%lhdrw =w sub %jcnt, 1
	%opb =l add %out, %wri
	storeb %lhdrw, %opb
	jmp @copyloop
@copyloop
	%k =l phi @lit_store 0, @copyone %k1
	%kdone =w csgel %k, %llen
	jnz %kdone, @lit_done, @copyone
@copyone
	%srcpos =l add %i, %k
	%sp =l add %src, %srcpos
	%byte =w loadub %sp
	%dstbase =l add %wri, 1
	%dstpos =l add %dstbase, %k
	%dp =l add %out, %dstpos
	storeb %byte, %dp
	%k1 =l add %k, 1
	jmp @copyloop
@lit_done
	%wri_lit =l add %wri, %pktsize
	jmp @loop
}
