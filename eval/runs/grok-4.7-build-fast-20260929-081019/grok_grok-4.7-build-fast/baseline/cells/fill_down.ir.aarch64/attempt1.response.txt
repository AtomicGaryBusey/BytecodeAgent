export function l $fill_down(l %a, l %n, l %start) {
@start
	%izero =l copy 0
	jmp @loop
@loop
	%i =l phi @start %izero, @body %inext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l sub %start, %i
	storel %val, %ptr
	%inext =l add %i, 1
	jmp @loop
@end
	ret %n
}
