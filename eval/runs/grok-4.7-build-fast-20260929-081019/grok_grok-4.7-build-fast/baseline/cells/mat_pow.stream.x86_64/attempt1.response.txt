mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rsi
    mov r15, rdx
    mov r13, rcx
    mov r14, r8
    mov rbx, r12
    imul rbx, r12
    lea rbp, [r14 + rbx*8]
    test r15, r15
    jz .Lidentity
    xor rcx, rcx
.Lcopy_a:
    mov rax, qword ptr [rdi + rcx*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    cmp rcx, rbx
    jb .Lcopy_a
    xor rcx, rcx
.Lzero_out:
    mov qword ptr [r13 + rcx*8], 0
    inc rcx
    cmp rcx, rbx
    jb .Lzero_out
    xor rcx, rcx
.Lset_diag:
    mov rax, rcx
    imul rax, r12
    add rax, rcx
    mov qword ptr [r13 + rax*8], 1
    inc rcx
    cmp rcx, r12
    jb .Lset_diag
.Lpow_loop:
    test r15, r15
    jz .Lcleanup
    test r15, 1
    jz .Lno_mul
    mov rdi, r13
    mov rsi, r14
    mov rdx, rbp
    call .Lmatmul
    xor rcx, rcx
.Lcopy_out:
    mov rax, qword ptr [rbp + rcx*8]
    mov qword ptr [r13 + rcx*8], rax
    inc rcx
    cmp rcx, rbx
    jb .Lcopy_out
.Lno_mul:
    shr r15, 1
    jz .Lcleanup
    mov rdi, r14
    mov rsi, r14
    mov rdx, rbp
    call .Lmatmul
    xor rcx, rcx
.Lcopy_base:
    mov rax, qword ptr [rbp + rcx*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    cmp rcx, rbx
    jb .Lcopy_base
    jmp .Lpow_loop
.Lidentity:
    xor rcx, rcx
.Lid_zero:
    mov qword ptr [r13 + rcx*8], 0
    inc rcx
    cmp rcx, rbx
    jb .Lid_zero
    xor rcx, rcx
.Lid_diag:
    mov rax, rcx
    imul rax, r12
    add rax, rcx
    mov qword ptr [r13 + rax*8], 1
    inc rcx
    cmp rcx, r12
    jb .Lid_diag
.Lcleanup:
    mov rcx, rbx
    shl rcx, 1
    xor rax, rax
.Lzero_tmp:
    test rcx, rcx
    jz .Lepi
    dec rcx
    mov qword ptr [r14 + rcx*8], rax
    jmp .Lzero_tmp
.Lepi:
    xor eax, eax
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lmatmul:
    xor r10, r10
.Lmm_i:
    cmp r10, r12
    jae .Lmm_ret
    xor r11, r11
.Lmm_j:
    cmp r11, r12
    jae .Lmm_inext
    xor rax, rax
    xor rcx, rcx
.Lmm_t:
    cmp rcx, r12
    jae .Lmm_st
    mov r8, r10
    imul r8, r12
    add r8, rcx
    mov r8, qword ptr [rdi + r8*8]
    mov r9, rcx
    imul r9, r12
    add r9, r11
    mov r9, qword ptr [rsi + r9*8]
    imul r8, r9
    add rax, r8
    inc rcx
    jmp .Lmm_t
.Lmm_st:
    mov r8, r10
    imul r8, r12
    add r8, r11
    mov qword ptr [rdx + r8*8], rax
    inc r11
    jmp .Lmm_j
.Lmm_inext:
    inc r10
    jmp .Lmm_i
.Lmm_ret:
    ret
