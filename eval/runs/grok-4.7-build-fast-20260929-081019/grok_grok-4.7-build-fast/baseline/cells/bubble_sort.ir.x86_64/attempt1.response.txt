export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer_cond
@outer_cond
	%i =l phi @start 0, @outer_inc %i_next
	%i_lt =w csltl %i, %n
	jnz %i_lt, @outer_body, @done
@outer_body
	%nm1 =l sub %n, 1
	%lim =l sub %nm1, %i
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body 0, @inner_inc %j_next
	%j_lt =w csltl %j, %lim
	jnz %j_lt, @inner_body, @outer_inc
@inner_body
	%j8 =l mul %j, 8
	%pj =l add %a, %j8
	%vj =l loadl %pj
	%pj1 =l add %pj, 8
	%vj1 =l loadl %pj1
	%gt =w csgtl %vj, %vj1
	jnz %gt, @swap, @inner_inc
@swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	%z =l copy 0
	ret %z
}
