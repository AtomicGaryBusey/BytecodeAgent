export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %ip1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%isz =w ceql %val, 0
	jnz %isz, @found, @next
@next
	%ip1 =l add %i, 1
	jmp @loop
@found
	ret %i
@done
	ret %n
}
