map_into:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
test r14, r14
jz .Ldone
.Lloop:
mov rdi, qword ptr [r13 + rbp*8]
call r15
mov qword ptr [r12 + rbp*8], rax
cmp rax, qword ptr [r13 + rbp*8]
jge .Lskip
inc rbx
.Lskip:
inc rbp
cmp rbp, r14
jl .Lloop
.Ldone:
mov rax, rbx
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
