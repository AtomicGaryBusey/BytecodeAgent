export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %inext
	%max =l phi @start 0, @cont %maxnext
	%ge =w csgel %i, %n
	jnz %ge, @done, @body
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @take, @skip
@take
	%maxt =l copy %b
	jmp @cont
@skip
	%maxs =l copy %max
	jmp @cont
@cont
	%maxnext =l phi @take %maxt, @skip %maxs
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
