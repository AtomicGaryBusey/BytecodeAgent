reverse:
    cmp x1, #1
    b.le .Ldone
    mov x2, #0
    sub x3, x1, #1
.Lloop:
    cmp x2, x3
    b.ge .Ldone
    ldr x4, [x0, x2, lsl #3]
    ldr x5, [x0, x3, lsl #3]
    str x5, [x0, x2, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x2, x2, #1
    sub x3, x3, #1
    b .Lloop
.Ldone:
    mov x0, #0
    ret
