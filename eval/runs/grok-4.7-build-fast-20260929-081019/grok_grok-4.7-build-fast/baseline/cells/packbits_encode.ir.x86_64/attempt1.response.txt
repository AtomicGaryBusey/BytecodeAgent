export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @emit_rep %i_rep, @lit_done %jf
    %w =l phi @start 0, @emit_rep %w2, @lit_done %nw
    %atend =w csgel %i, %n
    jnz %atend, @ret_ok, @run_init
@ret_ok
    ret %w
@run_init
    %p0 =l add %src, %i
    %b0 =w loadub %p0
    jmp @run
@run
    %k =l phi @run_init %i, @run_inc %k1
    %cnt =l phi @run_init 0, @run_inc %cnt1
    %klt =w csltl %k, %n
    jnz %klt, @run_lt128, @run_end
@run_lt128
    %clt =w csltl %cnt, 128
    jnz %clt, @run_cmp, @run_end
@run_cmp
    %pk =l add %src, %k
    %bk =w loadub %pk
    %eq =w ceqw %bk, %b0
    jnz %eq, @run_inc, @run_end
@run_inc
    %k1 =l add %k, 1
    %cnt1 =l add %cnt, 1
    jmp @run
@run_end
    %r =l phi @run %cnt, @run_lt128 %cnt, @run_cmp %cnt
    %rep =w csgel %r, 3
    jnz %rep, @rep_fit, @jloop
@rep_fit
    %w2 =l add %w, 2
    %rbig =w csgtl %w2, %cap
    jnz %rbig, @ret_fail, @emit_rep
@emit_rep
    %hdrl =l sub 257, %r
    %hdrw =w copy %hdrl
    %dp =l add %out, %w
    storeb %hdrw, %dp
    %dp1 =l add %dp, 1
    storeb %b0, %dp1
    %i_rep =l add %i, %r
    jmp @loop
@jloop
    %j =l phi @run_end %i, @j_inc %j1
    %jend =w csgel %j, %n
    jnz %jend, @lit_fit, @j_len
@j_len
    %jd =l sub %j, %i
    %jlong =w csgel %jd, 128
    jnz %jlong, @lit_fit, @jrun_init
@jrun_init
    %pj =l add %src, %j
    %bj =w loadub %pj
    jmp @jrun
@jrun
    %jk =l phi @jrun_init %j, @jrun_inc %jk1
    %jc =l phi @jrun_init 0, @jrun_inc %jc1
    %jklt =w csltl %jk, %n
    jnz %jklt, @jrun_lt, @jrun_end
@jrun_lt
    %jclt =w csltl %jc, 128
    jnz %jclt, @jrun_cmp, @jrun_end
@jrun_cmp
    %pjk =l add %src, %jk
    %bjk =w loadub %pjk
    %jeq =w ceqw %bjk, %bj
    jnz %jeq, @jrun_inc, @jrun_end
@jrun_inc
    %jk1 =l add %jk, 1
    %jc1 =l add %jc, 1
    jmp @jrun
@jrun_end
    %rj =l phi @jrun %jc, @jrun_lt %jc, @jrun_cmp %jc
    %rstop =w csgel %rj, 3
    jnz %rstop, @lit_fit, @j_inc
@j_inc
    %j1 =l add %j, 1
    jmp @jloop
@lit_fit
    %jf =l phi @jloop %j, @j_len %j, @jrun_end %j
    %L =l sub %jf, %i
    %psz =l add %L, 1
    %nw =l add %w, %psz
    %lbig =w csgtl %nw, %cap
    jnz %lbig, @ret_fail, @lit_hdr
@lit_hdr
    %lhdrl =l sub %L, 1
    %lhdrw =w copy %lhdrl
    %lp =l add %out, %w
    storeb %lhdrw, %lp
    jmp @cloop
@cloop
    %c =l phi @lit_hdr 0, @c_inc %c1
    %cdone =w csgel %c, %L
    jnz %cdone, @lit_done, @c_byte
@c_byte
    %si =l add %i, %c
    %sp =l add %src, %si
    %sb =w loadub %sp
    %di0 =l add %w, 1
    %di =l add %di0, %c
    %dpb =l add %out, %di
    storeb %sb, %dpb
    jmp @c_inc
@c_inc
    %c1 =l add %c, 1
    jmp @cloop
@lit_done
    jmp @loop
@ret_fail
    %m1 =l copy -1
    ret %m1
}
