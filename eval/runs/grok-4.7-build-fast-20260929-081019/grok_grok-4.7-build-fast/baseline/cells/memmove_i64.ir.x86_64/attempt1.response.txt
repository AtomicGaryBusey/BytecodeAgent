export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nzero =w ceql %n, 0
	jnz %nzero, @ret, @dir
@dir
	%before =w cultl %dst, %src
	jnz %before, @fwd, @bwd
@fwd
	%i =l phi @dir 0, @fwdbody %inext
	%fdone =w ceql %i, %n
	jnz %fdone, @ret, @fwdbody
@fwdbody
	%foff =l mul %i, 8
	%fptr =l add %src, %foff
	%dptr =l add %dst, %foff
	%val =l loadl %fptr
	storel %val, %dptr
	%inext =l add %i, 1
	jmp @fwd
@bwd
	%j =l phi @dir %n, @bwdbody %jnext
	%bdone =w ceql %j, 0
	jnz %bdone, @ret, @bwdbody
@bwdbody
	%jnext =l sub %j, 1
	%boff =l mul %jnext, 8
	%bsptr =l add %src, %boff
	%bdptr =l add %dst, %boff
	%bval =l loadl %bsptr
	storel %bval, %bdptr
	jmp @bwd
@ret
	%rv =l copy 0
	ret %rv
}
