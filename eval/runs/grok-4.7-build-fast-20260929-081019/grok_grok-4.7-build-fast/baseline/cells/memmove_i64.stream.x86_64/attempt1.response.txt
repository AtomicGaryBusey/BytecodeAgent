memmove_i64:
    xor eax, eax
    test rdx, rdx
    jz .Lret
    cmp rdi, rsi
    ja .Lbackward
    xor ecx, ecx
.Lforward:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    add rcx, 1
    cmp rcx, rdx
    jb .Lforward
    ret
.Lbackward:
    mov rcx, rdx
.Lback:
    sub rcx, 1
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    test rcx, rcx
    jnz .Lback
.Lret:
    ret
