bubble_sort:
    cmp rsi, 2
    jl .Ldone
    xor rcx, rcx
.Louter:
    mov rdx, rsi
    sub rdx, rcx
    dec rdx
    xor r8, r8
.Linner:
    cmp r8, rdx
    jge .Lnext_outer
    mov rax, qword ptr [rdi + r8*8]
    mov r9, qword ptr [rdi + r8*8 + 8]
    cmp rax, r9
    jle .Lno_swap
    mov qword ptr [rdi + r8*8], r9
    mov qword ptr [rdi + r8*8 + 8], rax
.Lno_swap:
    inc r8
    jmp .Linner
.Lnext_outer:
    inc rcx
    lea rax, [rsi - 1]
    cmp rcx, rax
    jl .Louter
.Ldone:
    xor eax, eax
    ret
