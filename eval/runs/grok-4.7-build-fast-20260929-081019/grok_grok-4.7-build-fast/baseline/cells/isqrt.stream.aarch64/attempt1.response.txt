    mov x2, #1
    lsl x2, x2, #32
    sub x2, x2, #1
.Lloop:
    cmp x1, x2
    b.hs .Ldone
    add x3, x1, x2
    add x3, x3, #1
    lsr x3, x3, #1
    mul x4, x3, x3
    cmp x4, x0
    b.hi .Ltoobig
    mov x1, x3
    b .Lloop
.Ltoobig:

    b .Lloop
.Ldone:
    mov x0, x1
    ret
