export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%mask =l copy 4294967295
	%c32 =w copy 32
	%n =l add %na, %nb
	%nis0 =w ceql %n, 0
	jnz %nis0, @ret0, @zcond
@zcond
	%zi =l phi @start 0, @zbody %zi1
	%zlt =w csltl %zi, %n
	jnz %zlt, @zbody, @emptychk
@zbody
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi1 =l add %zi, 1
	jmp @zcond
@emptychk
	%na0 =w ceql %na, 0
	jnz %na0, @siginit, @bchk
@bchk
	%nb0 =w ceql %nb, 0
	jnz %nb0, @siginit, @icond
@icond
	%i =l phi @bchk 0, @jdone %i1
	%ilt =w csltl %i, %na
	jnz %ilt, @ibody, @siginit
@ibody
	%ioff =l mul %i, 8
	%ip =l add %a, %ioff
	%ai =l loadl %ip
	%ail =l and %ai, %mask
	%aih =l shr %ai, %c32
	jmp @jcond
@jcond
	%j =l phi @ibody 0, @jbody %j1
	%carry =l phi @ibody 0, @jbody %carryout
	%jlt =w csltl %j, %nb
	jnz %jlt, @jbody, @jdone
@jbody
	%joff =l mul %j, 8
	%jp =l add %b, %joff
	%bj =l loadl %jp
	%bjl =l and %bj, %mask
	%bjh =l shr %bj, %c32
	%p0 =l mul %ail, %bjl
	%p1 =l mul %ail, %bjh
	%p2 =l mul %aih, %bjl
	%p3 =l mul %aih, %bjh
	%p0hi =l shr %p0, %c32
	%p1lo =l and %p1, %mask
	%p2lo =l and %p2, %mask
	%mid0 =l add %p0hi, %p1lo
	%mid =l add %mid0, %p2lo
	%p0lo =l and %p0, %mask
	%midlo =l and %mid, %mask
	%midsh =l shl %midlo, %c32
	%lo =l or %p0lo, %midsh
	%p1hi =l shr %p1, %c32
	%p2hi =l shr %p2, %c32
	%midhi =l shr %mid, %c32
	%hi0 =l add %p3, %p1hi
	%hi1 =l add %hi0, %p2hi
	%hi =l add %hi1, %midhi
	%idx =l add %i, %j
	%roff =l mul %idx, 8
	%rp =l add %r, %roff
	%old =l loadl %rp
	%sum1 =l add %old, %lo
	%c1w =w cultl %sum1, %old
	%c1 =l extuw %c1w
	%sum2 =l add %sum1, %carry
	%c2w =w cultl %sum2, %sum1
	%c2 =l extuw %c2w
	storel %sum2, %rp
	%ct =l add %hi, %c1
	%carryout =l add %ct, %c2
	%j1 =l add %j, 1
	jmp @jcond
@jdone
	%iend =l add %i, %nb
	%eoff =l mul %iend, 8
	%ep =l add %r, %eoff
	storel %carry, %ep
	%i1 =l add %i, 1
	jmp @icond
@siginit
	jmp @scond
@scond
	%k =l phi @siginit %n, @speek %k1
	%keq =w ceql %k, 0
	jnz %keq, @ret0, @speek
@speek
	%k1 =l sub %k, 1
	%koff =l mul %k1, 8
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @scond, @retk
@retk
	ret %k
@ret0
	ret 0
}
