packbits_decode:
lea r8, [rdi + rsi]
lea r9, [rdx + rcx]
mov r10, rdx
.Lloop:
cmp rdi, r8
jae .Ldone
movzx eax, byte ptr [rdi]
inc rdi
cmp eax, 128
je .Lloop
ja .Lrepeat
inc eax
mov rcx, r8
sub rcx, rdi
cmp rax, rcx
ja .Lsrc_err
mov rcx, r9
sub rcx, rdx
cmp rax, rcx
ja .Llit_trunc
mov rsi, rax
.Llit_copy:
mov r11b, byte ptr [rdi]
mov byte ptr [rdx], r11b
inc rdi
inc rdx
dec rsi
jnz .Llit_copy
jmp .Lloop
.Llit_trunc:
test rcx, rcx
jz .Lout_err
.Llit_trunc_copy:
mov r11b, byte ptr [rdi]
mov byte ptr [rdx], r11b
inc rdi
inc rdx
dec rcx
jnz .Llit_trunc_copy
jmp .Lout_err
.Lrepeat:
cmp rdi, r8
jae .Lsrc_err
movzx r11d, byte ptr [rdi]
inc rdi
neg al
movzx eax, al
inc eax
mov rcx, r9
sub rcx, rdx
cmp rax, rcx
ja .Lrep_trunc
.Lrep_fill:
mov byte ptr [rdx], r11b
inc rdx
dec rax
jnz .Lrep_fill
jmp .Lloop
.Lrep_trunc:
test rcx, rcx
jz .Lout_err
.Lrep_trunc_fill:
mov byte ptr [rdx], r11b
inc rdx
dec rcx
jnz .Lrep_trunc_fill
.Lout_err:
mov rax, -1
ret
.Lsrc_err:
mov rax, -2
ret
.Ldone:
mov rax, rdx
sub rax, r10
ret
