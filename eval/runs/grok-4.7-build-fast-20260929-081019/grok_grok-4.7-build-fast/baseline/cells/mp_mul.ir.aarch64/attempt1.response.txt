export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%mask =l copy 4294967295
	%zero =l copy 0
	jmp @zcond
@zcond
	%zi =l phi @start %zero, @zbody %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @icond
@zbody
	%zoff =l shl %zi, 3
	%zptr =l add %r, %zoff
	storel %zero, %zptr
	%zi1 =l add %zi, 1
	jmp @zcond
@icond
	%i =l phi @zcond %zero, @iend %i1
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @scond
@ibody
	%ioff =l shl %i, 3
	%aptr =l add %a, %ioff
	%ai =l loadl %aptr
	%jinit =l copy 0
	%cinit =l copy 0
	jmp @jcond
@jcond
	%j =l phi @ibody %jinit, @jbody %j1
	%carry =l phi @ibody %cinit, @jbody %carry1
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @iend
@jbody
	%joff =l shl %j, 3
	%bptr =l add %b, %joff
	%bj =l loadl %bptr
	%x0 =l and %ai, %mask
	%x1 =l shr %ai, 32
	%y0 =l and %bj, %mask
	%y1 =l shr %bj, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, %mask
	%p10lo =l and %p10, %mask
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%p00lo =l and %p00, %mask
	%midlo =l and %mid, %mask
	%midsh =l shl %midlo, 32
	%lo =l or %p00lo, %midsh
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%midhi =l shr %mid, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%hi =l add %hi1, %midhi
	%idx =l add %i, %j
	%idoff =l shl %idx, 3
	%rptr =l add %r, %idoff
	%rj =l loadl %rptr
	%s1 =l add %rj, %lo
	%c1w =w cultl %s1, %rj
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rptr
	%hic =l add %hi, %c1
	%carry1 =l add %hic, %c2
	%j1 =l add %j, 1
	jmp @jcond
@iend
	%inb =l add %i, %nb
	%inboff =l shl %inb, 3
	%iptr =l add %r, %inboff
	storel %carry, %iptr
	%i1 =l add %i, 1
	jmp @icond
@scond
	%k =l phi @icond %n, @sdec %k1
	%kc =w csgtl %k, 0
	jnz %kc, @stest, @sret
@stest
	%km1 =l sub %k, 1
	%koff =l shl %km1, 3
	%kptr =l add %r, %koff
	%limb =l loadl %kptr
	%lz =w ceql %limb, 0
	jnz %lz, @sdec, @sret
@sdec
	%k1 =l sub %k, 1
	jmp @scond
@sret
	ret %k
}
