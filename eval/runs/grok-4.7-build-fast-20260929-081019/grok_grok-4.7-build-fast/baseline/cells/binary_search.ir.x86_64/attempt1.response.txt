export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	jmp @loop
@loop
	%lo =l phi @start %lo0, @go_right %lo_r, @go_left %lo_l
	%hi =l phi @start %n, @go_right %hi_r, @go_left %hi_l
	%in =w csltl %lo, %hi
	jnz %in, @body, @miss
@body
	%span =l sub %hi, %lo
	%half =l udiv %span, 2
	%mid =l add %lo, %half
	%bytes =l mul %mid, 8
	%ptr =l add %a, %bytes
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @hit, @cmp
@hit
	ret %mid
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @go_right, @go_left
@go_right
	%lo_r =l add %mid, 1
	%hi_r =l copy %hi
	jmp @loop
@go_left
	%lo_l =l copy %lo
	%hi_l =l copy %mid
	jmp @loop
@miss
	%none =l copy -1
	ret %none
}
