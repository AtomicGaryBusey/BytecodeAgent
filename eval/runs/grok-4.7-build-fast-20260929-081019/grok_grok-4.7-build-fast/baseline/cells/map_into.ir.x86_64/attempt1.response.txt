export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @head
@head
	%i =l phi @start %i0, @body %i1
	%c =l phi @start %c0, @body %c1
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%sv =l loadl %sp
	%rv =l call %f(l %sv)
	storel %rv, %dp
	%lt =w csltl %rv, %sv
	%ltl =l extuw %lt
	%c1 =l add %c, %ltl
	%i1 =l add %i, 1
	jmp @head
@end
	ret %c
}
