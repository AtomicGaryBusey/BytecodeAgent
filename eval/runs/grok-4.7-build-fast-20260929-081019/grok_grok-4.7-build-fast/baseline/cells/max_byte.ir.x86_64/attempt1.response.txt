export function l $max_byte(l %p, l %n) {
@start
	%isz =w ceql %n, 0
	jnz %isz, @empty, @init
@empty
	%z =l copy 0
	ret %z
@init
	%m0 =l copy 0
	jmp @head
@head
	%m =l phi @init %m0, @next %m1
	%ptr =l phi @init %p, @next %ptr1
	%cnt =l phi @init %n, @next %cnt1
	%done =w ceql %cnt, 0
	jnz %done, @end, @body
@body
	%b =l loadub %ptr
	%ptr1 =l add %ptr, 1
	%cnt1 =l sub %cnt, 1
	%lt =w cultl %m, %b
	jnz %lt, @take, @keep
@take
	%tval =l copy %b
	jmp @next
@keep
	%kval =l copy %m
	jmp @next
@next
	%m1 =l phi @take %tval, @keep %kval
	jmp @head
@end
	ret %m
}
