export function l $reverse(l %a, l %n) {
@start
	%jm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %jm1, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @end
@body
	%iscale =l mul %i, 8
	%pi =l add %a, %iscale
	%vi =l loadl %pi
	%jscale =l mul %j, 8
	%pj =l add %a, %jscale
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@end
	%ret =l copy 0
	ret %ret
}
