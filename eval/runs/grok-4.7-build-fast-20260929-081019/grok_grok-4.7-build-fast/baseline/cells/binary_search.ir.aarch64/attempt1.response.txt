export function l $binary_search(l %a, l %n, l %key) {
@start
	%zero =l copy 0
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start %zero, @go_right %lo_r, @go_left %lo_l
	%hi =l phi @start %hi0, @go_right %hi_r, @go_left %hi_l
	%done =w csgtl %lo, %hi
	jnz %done, @miss, @body
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@found
	ret %mid
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @go_right, @go_left
@go_right
	%lo_r =l add %mid, 1
	%hi_r =l copy %hi
	jmp @loop
@go_left
	%lo_l =l copy %lo
	%hi_l =l sub %mid, 1
	jmp @loop
@miss
	%neg =l copy -1
	ret %neg
}
