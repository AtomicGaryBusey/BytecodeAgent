qsort_pairs:
    cmp x1, #1
    b.gt .Lstart
    mov x0, #0
    ret
.Lstart:
    stp x29, x30, [sp, #-48]!
    stp x19, x20, [sp, #16]
    str x21, [sp, #32]
    mov x19, x0
    mov x20, x1
.Lloop:
    sub x9, x20, #1
    lsl x10, x9, #4
    add x10, x19, x10
    ldr x11, [x10]
    ldr x12, [x10, #8]
    mov x21, xzr
    mov x10, xzr
.Lscan:
    cmp x10, x9
    b.ge .Lplace
    lsl x13, x10, #4
    add x13, x19, x13
    ldr x14, [x13]
    ldr x15, [x13, #8]
    cmp x14, x11
    b.lt .Ltake
    b.gt .Lnext
    cmp x15, x12
    b.gt .Lnext
.Ltake:
    cmp x21, x10
    b.eq .Linc_i
    lsl x0, x21, #4
    add x0, x19, x0
    ldr x1, [x0]
    ldr x2, [x0, #8]
    str x14, [x0]
    str x15, [x0, #8]
    str x1, [x13]
    str x2, [x13, #8]
.Linc_i:
    add x21, x21, #1
.Lnext:
    add x10, x10, #1
    b .Lscan
.Lplace:
    cmp x21, x9
    b.eq .Lplaced
    lsl x0, x21, #4
    add x0, x19, x0
    lsl x1, x9, #4
    add x1, x19, x1
    ldr x2, [x0]
    ldr x3, [x0, #8]
    ldr x14, [x1]
    ldr x15, [x1, #8]
    str x14, [x0]
    str x15, [x0, #8]
    str x2, [x1]
    str x3, [x1, #8]
.Lplaced:
    sub x9, x9, x21
    cmp x21, x9
    b.ge .Lrec_right
    mov x0, x19
    mov x1, x21
    bl qsort_pairs
    add x10, x21, #1
    lsl x10, x10, #4
    add x19, x19, x10
    sub x20, x20, x21
    sub x20, x20, #1
    b .Lcheck
.Lrec_right:

    cbz x9, .Ldo_right_call
    add x10, x21, #1
    lsl x10, x10, #4
    add x0, x19, x10
.Ldo_right_call:
    mov x1, x9
    bl qsort_pairs
    mov x20, x21
.Lcheck:
    cmp x20, #1
    b.gt .Lloop
    ldr x21, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #48
    mov x0, #0
    ret
