export function l $collatz_max(l %n) {
@start
	jmp @kloop
@kloop
	%k =l phi @start 1, @kinc %k1
	%max =l phi @start 0, @kinc %max1
	%kdone =w csgtl %k, %n
	jnz %kdone, @end, @collatz
@collatz
	%x =l phi @kloop %k, @step %xnext
	%steps =l phi @kloop 0, @step %steps1
	%is1 =w ceql %x, 1
	jnz %is1, @updmax, @dostep
@dostep
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @odd, @even
@even
	%xe =l shr %x, 1
	jmp @step
@odd
	%t3 =l mul %x, 3
	%xo =l add %t3, 1
	jmp @step
@step
	%xnext =l phi @even %xe, @odd %xo
	%steps1 =l add %steps, 1
	jmp @collatz
@updmax
	%gt =w csgtl %steps, %max
	jnz %gt, @newmax, @keepmax
@newmax
	jmp @kinc
@keepmax
	jmp @kinc
@kinc
	%max1 =l phi @newmax %steps, @keepmax %max
	%k1 =l add %k, 1
	jmp @kloop
@end
	ret %max
}
