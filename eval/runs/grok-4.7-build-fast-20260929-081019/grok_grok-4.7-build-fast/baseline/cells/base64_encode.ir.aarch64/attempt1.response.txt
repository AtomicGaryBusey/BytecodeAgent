export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @main_continue %inext
	%op =l phi @start %out, @main_continue %opnext
	%ip3 =l add %i, 3
	%cont =w culel %ip3, %n
	jnz %cont, @body, @rem
@body
	%sp0 =l add %src, %i
	%b0 =l loadub %sp0
	%sp1 =l add %sp0, 1
	%b1 =l loadub %sp1
	%sp2 =l add %sp0, 2
	%b2 =l loadub %sp2
	%sh0 =l shl %b0, 16
	%sh1 =l shl %b1, 8
	%hi =l or %sh0, %sh1
	%valm =l or %hi, %b2
	jmp @emit_loop
@rem
	%remn =l sub %n, %i
	%isz =w ceql %remn, 0
	jnz %isz, @finish, @remnz
@remnz
	%is1 =w ceql %remn, 1
	jnz %is1, @rem1, @rem2
@rem1
	%rp0 =l add %src, %i
	%rb0 =l loadub %rp0
	%val1 =l shl %rb0, 16
	jmp @rem_setup
@rem2
	%qp0 =l add %src, %i
	%qb0 =l loadub %qp0
	%qp1 =l add %qp0, 1
	%qb1 =l loadub %qp1
	%qsh0 =l shl %qb0, 16
	%qsh1 =l shl %qb1, 8
	%val2 =l or %qsh0, %qsh1
	jmp @rem_setup
@rem_setup
	%valr =l phi @rem1 %val1, @rem2 %val2
	%nd =l phi @rem1 2, @rem2 3
	jmp @emit_loop
@emit_loop
	%k =l phi @body 0, @rem_setup 0, @storech %knext
	%vcur =l phi @body %valm, @rem_setup %valr, @storech %vnext
	%pc =l phi @body %op, @rem_setup %op, @storech %pcnext
	%limit =l phi @body 4, @rem_setup %nd, @storech %limitk
	%after =l phi @body 0, @rem_setup 1, @storech %afterk
	%kdone =w ceql %k, %limit
	jnz %kdone, @emit_done, @emit_body
@emit_body
	%idx =l shr %vcur, 18
	%idx6 =l and %idx, 63
	%lt26 =w cultl %idx6, 26
	jnz %lt26, @up, @nup
@up
	%cup =l add %idx6, 65
	jmp @storech
@nup
	%lt52 =w cultl %idx6, 52
	jnz %lt52, @lo, @nlo
@lo
	%lo0 =l sub %idx6, 26
	%clo =l add %lo0, 97
	jmp @storech
@nlo
	%lt62 =w cultl %idx6, 62
	jnz %lt62, @dig, @ndig
@dig
	%di0 =l sub %idx6, 52
	%cd =l add %di0, 48
	jmp @storech
@ndig
	%is62 =w ceql %idx6, 62
	jnz %is62, @plus, @slash
@plus
	%cplus =l copy 43
	jmp @storech
@slash
	%csl =l copy 47
	jmp @storech
@storech
	%ch =l phi @up %cup, @lo %clo, @dig %cd, @plus %cplus, @slash %csl
	%chw =w copy %ch
	storeb %chw, %pc
	%vnext =l shl %vcur, 6
	%knext =l add %k, 1
	%pcnext =l add %pc, 1
	%limitk =l copy %limit
	%afterk =l copy %after
	jmp @emit_loop
@emit_done
	%npad =l sub 4, %limit
	%isrem =w ceql %after, 1
	jnz %isrem, @pad_loop, @main_continue
@main_continue
	%inext =l add %i, 3
	%opnext =l copy %pc
	jmp @loop
@pad_loop
	%pk =l phi @emit_done 0, @pad_body %pknext
	%ppc =l phi @emit_done %pc, @pad_body %ppcnext
	%paddone =w ceql %pk, %npad
	jnz %paddone, @finish, @pad_body
@pad_body
	%eqch =w copy 61
	storeb %eqch, %ppc
	%pknext =l add %pk, 1
	%ppcnext =l add %ppc, 1
	jmp @pad_loop
@finish
	%np2 =l add %n, 2
	%ng =l udiv %np2, 3
	%retv =l mul %ng, 4
	ret %retv
}
