export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	jmp @iinit
@iinit
	%i =l phi @start %zero, @iinc %inext
	%icmp =w csltl %i, %n
	jnz %icmp, @jinit, @ploop
@jinit
	%j =l phi @iinit %zero, @jbody %jnext
	%jcmp =w csltl %j, %n
	jnz %jcmp, @jbody, @iinc
@jbody
	%eq =w ceql %i, %j
	%val =l extuw %eq
	%in =l mul %i, %n
	%inj =l add %in, %j
	%injs =l mul %inj, %eight
	%addr =l add %out, %injs
	storel %val, %addr
	%jnext =l add %j, %one
	jmp @jinit
@iinc
	%inext =l add %i, %one
	jmp @iinit
@ploop
	%p =l phi @iinit %zero, @pinc %pnext
	%pcmp =w csltl %p, %k
	jnz %pcmp, @mi, @zi
@mi
	%mi =l phi @ploop %zero, @miinc %minext
	%micmp =w csltl %mi, %n
	jnz %micmp, @mj, @ci
@mj
	%mj =l phi @mi %zero, @mtend %mjnext
	%mjcmp =w csltl %mj, %n
	jnz %mjcmp, @mt, @miinc
@mt
	%mt =l phi @mj %zero, @mtbody %mtnext
	%sum =l phi @mj %zero, @mtbody %sumnext
	%mtcmp =w csltl %mt, %n
	jnz %mtcmp, @mtbody, @mtend
@mtbody
	%ri =l mul %mi, %n
	%oi =l add %ri, %mt
	%os =l mul %oi, %eight
	%oa =l add %out, %os
	%ov =l loadl %oa
	%ai =l mul %mt, %n
	%aj =l add %ai, %mj
	%as =l mul %aj, %eight
	%aa =l add %a, %as
	%av =l loadl %aa
	%pr =l mul %ov, %av
	%sumnext =l add %sum, %pr
	%mtnext =l add %mt, %one
	jmp @mt
@mtend
	%ti =l mul %mi, %n
	%tj =l add %ti, %mj
	%ts =l mul %tj, %eight
	%ta =l add %tmp, %ts
	storel %sum, %ta
	%mjnext =l add %mj, %one
	jmp @mj
@miinc
	%minext =l add %mi, %one
	jmp @mi
@ci
	%ci =l phi @mi %zero, @cbody %cinext
	%ccmp =w csltl %ci, %nn
	jnz %ccmp, @cbody, @pinc
@cbody
	%cs =l mul %ci, %eight
	%ca =l add %tmp, %cs
	%cv =l loadl %ca
	%da =l add %out, %cs
	storel %cv, %da
	%cinext =l add %ci, %one
	jmp @ci
@pinc
	%pnext =l add %p, %one
	jmp @ploop
@zi
	%zi =l phi @ploop %zero, @zbody %zinext
	%zcmp =w csltl %zi, %nn2
	jnz %zcmp, @zbody, @done
@zbody
	%zs =l mul %zi, %eight
	%za =l add %tmp, %zs
	storel %zero, %za
	%zinext =l add %zi, %one
	jmp @zi
@done
	ret %zero
}
