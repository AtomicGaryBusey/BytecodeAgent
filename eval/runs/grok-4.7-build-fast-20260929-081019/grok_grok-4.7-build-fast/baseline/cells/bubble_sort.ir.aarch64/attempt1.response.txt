export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l copy 0
	jmp @outercond
@outercond
	%i =l phi @start %i0, @outerinc %i1
	%nm1 =l sub %n, 1
	%oc =w csltl %i, %nm1
	jnz %oc, @innerinit, @done
@innerinit
	%j0 =l copy 0
	%limit =l sub %nm1, %i
	jmp @innercond
@innercond
	%j =l phi @innerinit %j0, @innernext %j1
	%ic =w csltl %j, %limit
	jnz %ic, @cmp, @outerinc
@cmp
	%off =l mul %j, 8
	%pj =l add %a, %off
	%pj1 =l add %pj, 8
	%vj =l loadl %pj
	%vj1 =l loadl %pj1
	%gt =w csgtl %vj, %vj1
	jnz %gt, @swap, @innernext
@swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @innernext
@innernext
	%j1 =l add %j, 1
	jmp @innercond
@outerinc
	%i1 =l add %i, 1
	jmp @outercond
@done
	%z =l copy 0
	ret %z
}
