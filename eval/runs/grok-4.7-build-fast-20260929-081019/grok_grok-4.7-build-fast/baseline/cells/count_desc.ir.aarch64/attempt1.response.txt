export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %inext
	%cnt =l phi @init 0, @body %cntnext
	%done =w ceql %i, %last
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%ai =l loadl %p
	%inext =l add %i, 1
	%off2 =l mul %inext, 8
	%p2 =l add %a, %off2
	%ai1 =l loadl %p2
	%gt =w csgtl %ai, %ai1
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	jmp @loop
@end
	ret %cnt
}
