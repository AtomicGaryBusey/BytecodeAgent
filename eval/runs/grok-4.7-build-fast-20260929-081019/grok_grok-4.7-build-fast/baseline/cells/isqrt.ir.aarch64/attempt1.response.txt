export function l $isqrt(l %n) {
@start
	%one =l copy 1
	%zero =l copy 0
	jmp @check
@check
	%lo =l phi @start %one, @merge %lo_n
	%hi =l phi @start %n, @merge %hi_n
	%res =l phi @start %zero, @merge %res_n
	%cmp =w culel %lo, %hi
	jnz %cmp, @body, @end
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%ok =w culel %mid, %quot
	jnz %ok, @setlo, @sethi
@setlo
	%lo_a =l add %mid, 1
	%hi_a =l copy %hi
	%res_a =l copy %mid
	jmp @merge
@sethi
	%lo_b =l copy %lo
	%hi_b =l sub %mid, 1
	%res_b =l copy %res
	jmp @merge
@merge
	%lo_n =l phi @setlo %lo_a, @sethi %lo_b
	%hi_n =l phi @setlo %hi_a, @sethi %hi_b
	%res_n =l phi @setlo %res_a, @sethi %res_b
	jmp @check
@end
	ret %res
}
