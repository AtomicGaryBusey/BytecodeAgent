export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%si0 =l copy 0
	%oi0 =l copy 0
	jmp @hdr
@hdr
	%si =l phi @start %si0, @nop %si_nop, @lit_done %si_lit, @rep_done %si_rep
	%oi =l phi @start %oi0, @nop %oi_nop, @lit_done %oi_lit, @rep_done %oi_rep
	%inrange =w csltl %si, %n
	jnz %inrange, @read, @done
@done
	ret %oi
@read
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%m128 =w copy -128
	%isnop =w ceqw %h, %m128
	jnz %isnop, @nop, @notnop
@nop
	%si_nop =l copy %si1
	%oi_nop =l copy %oi
	jmp @hdr
@notnop
	%zerow =w copy 0
	%ispos =w csgew %h, %zerow
	jnz %ispos, @literal, @repeat
@literal
	%onew =w copy 1
	%hp1 =w add %h, %onew
	%count =l extsw %hp1
	%rem =l sub %n, %si1
	%enough =w csgel %rem, %count
	jnz %enough, @lit_ok, @err_src
@lit_ok
	%avail =l sub %cap, %oi
	%exceeds =w csgtl %count, %avail
	jnz %exceeds, @lit_partial, @lit_full
@lit_full
	%k0 =l copy 0
	jmp @lit_loop
@lit_loop
	%k =l phi @lit_full %k0, @lit_step %k1
	%siL =l phi @lit_full %si1, @lit_step %siL1
	%oiL =l phi @lit_full %oi, @lit_step %oiL1
	%more =w csltl %k, %count
	jnz %more, @lit_step, @lit_done
@lit_step
	%sp =l add %src, %siL
	%byte =w loadub %sp
	%dp =l add %out, %oiL
	storeb %byte, %dp
	%siL1 =l add %siL, 1
	%oiL1 =l add %oiL, 1
	%k1 =l add %k, 1
	jmp @lit_loop
@lit_done
	%si_lit =l copy %siL
	%oi_lit =l copy %oiL
	jmp @hdr
@lit_partial
	%pk0 =l copy 0
	jmp @lp_loop
@lp_loop
	%pk =l phi @lit_partial %pk0, @lp_step %pk1
	%psi =l phi @lit_partial %si1, @lp_step %psi1
	%poi =l phi @lit_partial %oi, @lp_step %poi1
	%pmore =w csltl %pk, %avail
	jnz %pmore, @lp_step, @err_cap
@lp_step
	%psp =l add %src, %psi
	%pbyte =w loadub %psp
	%pdp =l add %out, %poi
	storeb %pbyte, %pdp
	%psi1 =l add %psi, 1
	%poi1 =l add %poi, 1
	%pk1 =l add %pk, 1
	jmp @lp_loop
@repeat
	%has =w csltl %si1, %n
	jnz %has, @rep_load, @err_src
@rep_load
	%rp =l add %src, %si1
	%rb =w loadub %rp
	%siR1 =l add %si1, 1
	%onew2 =w copy 1
	%cntw =w sub %onew2, %h
	%rcount =l extsw %cntw
	%ravail =l sub %cap, %oi
	%rex =w csgtl %rcount, %ravail
	jnz %rex, @rep_partial, @rep_full
@rep_full
	%rk0 =l copy 0
	jmp @rf_loop
@rf_loop
	%rk =l phi @rep_full %rk0, @rf_step %rk1
	%roi =l phi @rep_full %oi, @rf_step %roi1
	%rmore =w csltl %rk, %rcount
	jnz %rmore, @rf_step, @rep_done
@rf_step
	%rdp =l add %out, %roi
	storeb %rb, %rdp
	%roi1 =l add %roi, 1
	%rk1 =l add %rk, 1
	jmp @rf_loop
@rep_done
	%si_rep =l copy %siR1
	%oi_rep =l copy %roi
	jmp @hdr
@rep_partial
	%rpk0 =l copy 0
	jmp @rp_loop
@rp_loop
	%rpk =l phi @rep_partial %rpk0, @rp_step %rpk1
	%rpoi =l phi @rep_partial %oi, @rp_step %rpoi1
	%rpmore =w csltl %rpk, %ravail
	jnz %rpmore, @rp_step, @err_cap
@rp_step
	%rpdp =l add %out, %rpoi
	storeb %rb, %rpdp
	%rpoi1 =l add %rpoi, 1
	%rpk1 =l add %rpk, 1
	jmp @rp_loop
@err_cap
	%errm1 =l copy -1
	ret %errm1
@err_src
	%errm2 =l copy -2
	ret %errm2
}
