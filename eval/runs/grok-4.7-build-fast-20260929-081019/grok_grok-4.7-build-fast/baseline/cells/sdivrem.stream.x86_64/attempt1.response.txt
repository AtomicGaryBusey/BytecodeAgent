sdivrem:
    mov rcx, rdx
    test rsi, rsi
    jz .Lbzero
    mov rax, 0x8000000000000000
    cmp rdi, rax
    jne .Ldiv
    cmp rsi, -1
    jne .Ldiv
    xor edx, edx
    mov qword ptr [rcx], rdx
    ret
.Ldiv:
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Lbzero:
    xor eax, eax
    mov qword ptr [rcx], rdi
    ret
