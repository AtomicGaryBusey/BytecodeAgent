export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%before =w cultl %dst, %src
	jnz %before, @fwd, @bwd
@fwd
	%i =l phi @start 0, @fwdbody %inext
	%fmore =w csltl %i, %n
	jnz %fmore, @fwdbody, @done
@fwdbody
	%foff =l mul %i, 8
	%fsptr =l add %src, %foff
	%fdptr =l add %dst, %foff
	%fval =l loadl %fsptr
	storel %fval, %fdptr
	%inext =l add %i, 1
	jmp @fwd
@bwd
	%j =l phi @start %n, @bwdbody %jnext
	%bmore =w csgtl %j, 0
	jnz %bmore, @bwdbody, @done
@bwdbody
	%jnext =l sub %j, 1
	%boff =l mul %jnext, 8
	%bsptr =l add %src, %boff
	%bdptr =l add %dst, %boff
	%bval =l loadl %bsptr
	storel %bval, %bdptr
	jmp @bwd
@done
	%retv =l copy 0
	ret %retv
}
