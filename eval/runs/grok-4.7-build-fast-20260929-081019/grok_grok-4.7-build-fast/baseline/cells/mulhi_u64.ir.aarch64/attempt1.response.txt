export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%sh =w copy 32
	%alo =l and %a, %mask
	%ahi =l shr %a, %sh
	%blo =l and %b, %mask
	%bhi =l shr %b, %sh
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, %sh
	%p1lo =l and %p1, %mask
	%p2lo =l and %p2, %mask
	%p1hi =l shr %p1, %sh
	%p2hi =l shr %p2, %sh
	%mid0 =l add %p0hi, %p1lo
	%mid =l add %mid0, %p2lo
	%midhi =l shr %mid, %sh
	%hi0 =l add %p3, %p1hi
	%hi1 =l add %hi0, %p2hi
	%hi =l add %hi1, %midhi
	ret %hi
}
