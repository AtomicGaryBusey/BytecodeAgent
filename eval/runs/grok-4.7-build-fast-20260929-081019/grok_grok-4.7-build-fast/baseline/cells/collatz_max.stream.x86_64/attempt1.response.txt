collatz_max:
    mov rsi, 1
    xor r8, r8
.Lnextk:
    mov rcx, rsi
    xor rax, rax
.Lstep:
    cmp rcx, 1
    je .Ldone
    test rcx, 1
    jnz .Lodd
    shr rcx, 1
    inc rax
    jmp .Lstep
.Lodd:
    lea rcx, [rcx + rcx*2 + 1]
    inc rax
    jmp .Lstep
.Ldone:
    cmp rax, r8
    cmovg r8, rax
    inc rsi
    cmp rsi, rdi
    jle .Lnextk
    mov rax, r8
    ret
