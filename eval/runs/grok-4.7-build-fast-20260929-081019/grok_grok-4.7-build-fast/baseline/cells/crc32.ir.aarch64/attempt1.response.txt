export function l $crc32(l %p, l %n) {
@start
	%enter =w csltl 0, %n
	jnz %enter, @bytes, @end
@bytes
	%pi =l phi @start %p, @nextb %pn
	%ii =l phi @start 0, @nextb %in
	%ci =w phi @start -1, @nextb %c8
	%bv =w loadub %pi
	%c0 =w xor %ci, %bv
	jmp @bits
@bits
	%bi =w phi @bytes 0, @bstep %bn
	%cb =w phi @bytes %c0, @bstep %cn
	%bit =w and %cb, 1
	jnz %bit, @shrx, @shr
@shrx
	%t1 =w shr %cb, 1
	%cx =w xor %t1, -306674912
	jmp @bstep
@shr
	%cs =w shr %cb, 1
	jmp @bstep
@bstep
	%cn =w phi @shrx %cx, @shr %cs
	%bn =w add %bi, 1
	%bdone =w ceqw %bn, 8
	jnz %bdone, @nextb, @bits
@nextb
	%c8 =w copy %cn
	%pn =l add %pi, 1
	%in =l add %ii, 1
	%more =w csltl %in, %n
	jnz %more, @bytes, @end
@end
	%cf =w phi @start -1, @nextb %c8
	%xf =w xor %cf, -1
	%r =l extuw %xf
	ret %r
}
