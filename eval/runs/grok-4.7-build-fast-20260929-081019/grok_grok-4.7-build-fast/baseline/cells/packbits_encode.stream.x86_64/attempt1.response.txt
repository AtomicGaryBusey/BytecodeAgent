packbits_encode:
push rbx
push r12
xor r8, r8
xor r9, r9
.Lloop:
cmp r8, rsi
jge .Ldone
movzx eax, byte ptr [rdi + r8]
mov r10d, 1
lea r11, [r8 + 1]
.Lrun:
cmp r10, 128
jge .Lrun_done
cmp r11, rsi
jge .Lrun_done
cmp al, byte ptr [rdi + r11]
jne .Lrun_done
inc r10
inc r11
jmp .Lrun
.Lrun_done:
cmp r10, 3
jl .Llit_start
lea rax, [r9 + 2]
cmp rax, rcx
ja .Lfail
mov eax, 1
sub eax, r10d
mov byte ptr [rdx + r9], al
movzx eax, byte ptr [rdi + r8]
mov byte ptr [rdx + r9 + 1], al
add r9, 2
add r8, r10
jmp .Lloop
.Llit_start:
mov rbx, r8
.Llit_scan:
cmp rbx, rsi
jge .Llit_emit
mov rax, rbx
sub rax, r8
cmp rax, 128
jge .Llit_emit
movzx eax, byte ptr [rdi + rbx]
mov r10d, 1
lea r11, [rbx + 1]
.Lrun2:
cmp r10, 128
jge .Lrun2_done
cmp r11, rsi
jge .Lrun2_done
cmp al, byte ptr [rdi + r11]
jne .Lrun2_done
inc r10
inc r11
jmp .Lrun2
.Lrun2_done:
cmp r10, 3
jge .Llit_emit
inc rbx
jmp .Llit_scan
.Llit_emit:
mov r12, rbx
sub r12, r8
lea rax, [r9 + r12 + 1]
cmp rax, rcx
ja .Lfail
mov eax, r12d
dec eax
mov byte ptr [rdx + r9], al
xor r10, r10
.Lcopy:
cmp r10, r12
jge .Lcopy_done
lea r11, [r8 + r10]
movzx eax, byte ptr [rdi + r11]
lea r11, [r9 + r10 + 1]
mov byte ptr [rdx + r11], al
inc r10
jmp .Lcopy
.Lcopy_done:
lea r9, [r9 + r12 + 1]
mov r8, rbx
jmp .Lloop
.Ldone:
mov rax, r9
pop r12
pop rbx
ret
.Lfail:
mov rax, -1
pop r12
pop rbx
ret
