export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @head
@head
	%i =l phi @start 0, @body %inext
	%c =l phi @start 0, @body %cnext
	%lt =w csltl %i, %n
	jnz %lt, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gte =l extuw %gt
	%cnext =l add %c, %gte
	%inext =l add %i, 1
	jmp @head
@end
	ret %c
}
