export function l $reverse(l %a, l %n) {
@start
	%zero =l copy 0
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %zero, @body %inext
	%j =l phi @start %nm1, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @end
@body
	%offi =l mul %i, 8
	%addri =l add %a, %offi
	%vali =l loadl %addri
	%offj =l mul %j, 8
	%addrj =l add %a, %offj
	%valj =l loadl %addrj
	storel %valj, %addri
	storel %vali, %addrj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@end
	%ret =l copy 0
	ret %ret
}
