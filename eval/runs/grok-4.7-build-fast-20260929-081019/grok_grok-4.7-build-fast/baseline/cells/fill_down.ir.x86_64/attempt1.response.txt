export function l $fill_down(l %a, l %n, l %start) {
@start
	%iz =l copy 0
	jmp @loop
@loop
	%i =l phi @start %iz, @body %inext
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%diff =l sub %start, %i
	%off =l mul %i, 8
	%p =l add %a, %off
	storel %diff, %p
	%inext =l add %i, 1
	jmp @loop
@done
	ret %n
}
