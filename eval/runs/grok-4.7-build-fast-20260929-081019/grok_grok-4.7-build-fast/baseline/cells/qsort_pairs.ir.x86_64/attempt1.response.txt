export function l $qsort_pairs(l %a, l %n) {
@start
	%a0 =l copy %a
	%n0 =l copy %n
	jmp @loop
@loop
	%base =l phi @start %a0, @setR %rp, @recR %baseR2, @setL %baseL
	%cnt =l phi @start %n0, @setR %cntR, @recR %cntL2, @setL %cntL
	%small =w cslel %cnt, 1
	jnz %small, @done, @part
@part
	%nm1 =l sub %cnt, 1
	%poff =l mul %nm1, 16
	%pp =l add %base, %poff
	%pkey =l loadl %pp
	%pp8 =l add %pp, 8
	%pval =l loadl %pp8
	%iz =l copy 0
	%jz =l copy 0
	jmp @scan
@scan
	%i =l phi @part %iz, @adv %inext
	%j =l phi @part %jz, @adv %jnext
	%jdone =w csgel %j, %nm1
	jnz %jdone, @place, @body
@body
	%joff =l mul %j, 16
	%jp =l add %base, %joff
	%jkey =l loadl %jp
	%jp8 =l add %jp, 8
	%jval =l loadl %jp8
	%klt =w csltl %jkey, %pkey
	jnz %klt, @doswap, @ck
@ck
	%keq =w ceql %jkey, %pkey
	jnz %keq, @vtst, @skip
@vtst
	%vlt =w csltl %jval, %pval
	jnz %vlt, @doswap, @skip
@doswap
	%ioff =l mul %i, 16
	%ip =l add %base, %ioff
	%ikey =l loadl %ip
	%ip8 =l add %ip, 8
	%ival =l loadl %ip8
	storel %jkey, %ip
	storel %jval, %ip8
	storel %ikey, %jp
	storel %ival, %jp8
	%iplus =l add %i, 1
	jmp @adv
@skip
	%isame =l copy %i
	jmp @adv
@adv
	%inext =l phi @doswap %iplus, @skip %isame
	%jnext =l add %j, 1
	jmp @scan
@place
	%ioff2 =l mul %i, 16
	%ip2 =l add %base, %ioff2
	%ikey2 =l loadl %ip2
	%ip28 =l add %ip2, 8
	%ival2 =l loadl %ip28
	storel %pkey, %ip2
	storel %pval, %ip28
	storel %ikey2, %pp
	storel %ival2, %pp8
	%rcnt =l sub %nm1, %i
	%le =w cslel %i, %rcnt
	jnz %le, @smallL, @smallR
@smallL
	%c1 =l call $qsort_pairs(l %base, l %i)
	%needr =w csgtl %rcnt, 1
	jnz %needr, @setR, @done
@setR
	%i1 =l add %i, 1
	%off =l mul %i1, 16
	%rp =l add %base, %off
	%cntR =l copy %rcnt
	jmp @loop
@smallR
	%needr2 =w csgtl %rcnt, 1
	jnz %needr2, @recR, @noR
@recR
	%i1b =l add %i, 1
	%offb =l mul %i1b, 16
	%rpb =l add %base, %offb
	%c2 =l call $qsort_pairs(l %rpb, l %rcnt)
	%baseR2 =l copy %base
	%cntL2 =l copy %i
	jmp @loop
@noR
	%needl =w csgtl %i, 1
	jnz %needl, @setL, @done
@setL
	%baseL =l copy %base
	%cntL =l copy %i
	jmp @loop
@done
	%zero =l copy 0
	ret %zero
}
