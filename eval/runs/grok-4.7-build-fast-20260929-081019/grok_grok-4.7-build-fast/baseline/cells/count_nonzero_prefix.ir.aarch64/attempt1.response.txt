export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %ip1
	%lt =w csltl %i, %n
	jnz %lt, @check, @done
@check
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%z =w ceql %v, 0
	jnz %z, @found, @next
@next
	%ip1 =l add %i, 1
	jmp @loop
@found
	ret %i
@done
	ret %n
}
