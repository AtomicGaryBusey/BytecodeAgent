export function l $shift_sat(l %x, l %n) {
@start
	%ge =w cugel %n, 64
	jnz %ge, @big, @loop
@big
	%res =l sar %x, 63
	ret %res
@loop
	%lc =l phi @start %x, @step %ln
	%rc =l phi @start %x, @step %rn
	%ic =l phi @start 0, @step %in
	%done =w ceql %ic, %n
	jnz %done, @doneblk, @step
@step
	%ln =l shl %lc, 1
	%rn =l sar %rc, 1
	%in =l add %ic, 1
	jmp @loop
@doneblk
	%sum =l add %lc, %rc
	ret %sum
}
