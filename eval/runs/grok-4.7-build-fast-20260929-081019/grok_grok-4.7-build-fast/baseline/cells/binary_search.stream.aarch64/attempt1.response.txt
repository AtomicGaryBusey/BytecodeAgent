binary_search:
cbz x1, .Lnotfound
mov x3, #0
sub x4, x1, #1
.Lloop:
cmp x3, x4
b.gt .Lnotfound
sub x5, x4, x3
lsr x5, x5, #1
add x5, x3, x5
ldr x6, [x0, x5, lsl #3]
cmp x6, x2
b.eq .Lfound
b.lt .Lright
sub x4, x5, #1
b .Lloop
.Lright:
add x3, x5, #1
b .Lloop
.Lfound:
mov x0, x5
ret
.Lnotfound:
mov x0, #-1
ret
