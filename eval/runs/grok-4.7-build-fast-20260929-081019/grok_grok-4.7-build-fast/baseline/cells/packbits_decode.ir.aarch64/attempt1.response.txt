export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%i0 =l copy 0
	%w0 =l copy 0
	jmp @hdr
@hdr
	%i =l phi @entry %i0, @nop %i_nop, @litend %i_after_lit, @repend %i_after_rep
	%wri =l phi @entry %w0, @nop %wri_nop, @litend %wri_lit, @repend %wri_rep
	%more =w csltl %i, %n
	jnz %more, @dohdr, @success
@success
	ret %wri
@dohdr
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i1 =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @checklit
@nop
	%i_nop =l copy %i1
	%wri_nop =l copy %wri
	jmp @hdr
@checklit
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%hl =l extuw %h
	%cnt =l add %hl, 1
	%end =l add %i1, %cnt
	%srcok =w cslel %end, %n
	jnz %srcok, @litinit, @errsrc
@errsrc
	%m2 =l copy -2
	ret %m2
@litinit
	%j0 =l copy 0
	%wri_l0 =l copy %wri
	jmp @litloop
@litloop
	%j =l phi @litinit %j0, @litwrite %j2
	%wcurl =l phi @litinit %wri_l0, @litwrite %wri_l2
	%jmore =w csltl %j, %cnt
	jnz %jmore, @litstep, @litend
@litstep
	%room =w csltl %wcurl, %cap
	jnz %room, @litwrite, @errout
@litwrite
	%off =l add %i1, %j
	%sp =l add %src, %off
	%byte =w loadub %sp
	%dp =l add %out, %wcurl
	storeb %byte, %dp
	%wri_l2 =l add %wcurl, 1
	%j2 =l add %j, 1
	jmp @litloop
@litend
	%i_after_lit =l add %i1, %cnt
	%wri_lit =l copy %wcurl
	jmp @hdr
@errout
	%m1 =l copy -1
	ret %m1
@repeat
	%have =w csltl %i1, %n
	jnz %have, @repbyte, @errsrc
@repbyte
	%rp =l add %src, %i1
	%rb =w loadub %rp
	%i_after_rep0 =l add %i1, 1
	%hl2 =l extsw %h
	%rcnt =l sub 1, %hl2
	%rj0 =l copy 0
	%wri_r0 =l copy %wri
	jmp @reploop
@reploop
	%rj =l phi @repbyte %rj0, @repwrite %rj2
	%wrir =l phi @repbyte %wri_r0, @repwrite %wri_r2
	%rmore =w csltl %rj, %rcnt
	jnz %rmore, @repstep, @repend
@repstep
	%rroom =w csltl %wrir, %cap
	jnz %rroom, @repwrite, @errout
@repwrite
	%rdp =l add %out, %wrir
	storeb %rb, %rdp
	%wri_r2 =l add %wrir, 1
	%rj2 =l add %rj, 1
	jmp @reploop
@repend
	%i_after_rep =l copy %i_after_rep0
	%wri_rep =l copy %wrir
	jmp @hdr
}
