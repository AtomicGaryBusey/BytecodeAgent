qsort_pairs:
cmp rsi, 1
jg .Lwork
xor eax, eax
ret
.Lwork:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov rbp, rsi
.Lloop_part:
cmp rbp, 1
jle .Lepilogue
lea rcx, [rbp - 1]
lea rcx, [rcx + rcx]
mov r12, qword ptr [rbx + rcx*8]
mov r13, qword ptr [rbx + rcx*8 + 8]
xor r14, r14
xor r15, r15
.Lscan:
lea rax, [rbp - 1]
cmp r15, rax
jge .Lfinish_part
lea rcx, [r15 + r15]
mov rax, qword ptr [rbx + rcx*8]
mov rdx, qword ptr [rbx + rcx*8 + 8]
cmp rax, r12
jl .Ltake
jg .Lnext_j
cmp rdx, r13
jg .Lnext_j
.Ltake:
lea rsi, [r14 + r14]
lea rcx, [r15 + r15]
mov r8, qword ptr [rbx + rsi*8]
mov r9, qword ptr [rbx + rsi*8 + 8]
mov r10, qword ptr [rbx + rcx*8]
mov r11, qword ptr [rbx + rcx*8 + 8]
mov qword ptr [rbx + rsi*8], r10
mov qword ptr [rbx + rsi*8 + 8], r11
mov qword ptr [rbx + rcx*8], r8
mov qword ptr [rbx + rcx*8 + 8], r9
inc r14
.Lnext_j:
inc r15
jmp .Lscan
.Lfinish_part:
lea rsi, [r14 + r14]
lea rcx, [rbp - 1]
lea rcx, [rcx + rcx]
mov r8, qword ptr [rbx + rsi*8]
mov r9, qword ptr [rbx + rsi*8 + 8]
mov r10, qword ptr [rbx + rcx*8]
mov r11, qword ptr [rbx + rcx*8 + 8]
mov qword ptr [rbx + rsi*8], r10
mov qword ptr [rbx + rsi*8 + 8], r11
mov qword ptr [rbx + rcx*8], r8
mov qword ptr [rbx + rcx*8 + 8], r9
mov r15, rbp
sub r15, r14
dec r15
lea rax, [r14 + 1]
lea rax, [rax + rax]
lea r13, [rbx + rax*8]
cmp r14, r15
jg .Lright_smaller
cmp r14, 1
jle .Lleft_skip
mov rdi, rbx
mov rsi, r14
call qsort_pairs
.Lleft_skip:
mov rbx, r13
mov rbp, r15
jmp .Lloop_part
.Lright_smaller:
cmp r15, 1
jle .Lright_skip
mov rdi, r13
mov rsi, r15
call qsort_pairs
.Lright_skip:
mov rbp, r14
jmp .Lloop_part
.Lepilogue:
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
xor eax, eax
ret
