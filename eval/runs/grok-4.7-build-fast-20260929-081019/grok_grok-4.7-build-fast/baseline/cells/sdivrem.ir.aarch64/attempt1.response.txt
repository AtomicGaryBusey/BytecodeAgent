export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @byzero, @chkmin
@byzero
	storel %a, %rem
	ret 0
@chkmin
	%one =l copy 1
	%imin =l shl %one, 63
	%ismin =w ceql %a, %imin
	jnz %ismin, @chkneg, @div
@chkneg
	%isn1 =w ceql %b, -1
	jnz %isn1, @ovf, @div
@ovf
	storel 0, %rem
	ret %imin
@div
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
