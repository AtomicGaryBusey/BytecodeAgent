export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %i1
	%c =l phi @init 0, @body %c1
	%done =w ceql %i, %last
	jnz %done, @end, @body
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%i1 =l add %i, 1
	%off1 =l shl %i1, 3
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%gt =w csgtl %v, %v1
	%gtl =l extuw %gt
	%c1 =l add %c, %gtl
	jmp @loop
@end
	ret %c
}
