dot_i32_simd:
    pxor xmm2, xmm2
    mov rcx, rdx
    cmp rcx, 4
    jl .Lno_simd
.Lsimd_loop:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqa xmm3, xmm0
    pmuldq xmm3, xmm1
    psrldq xmm0, 4
    psrldq xmm1, 4
    pmuldq xmm0, xmm1
    paddq xmm2, xmm3
    paddq xmm2, xmm0
    add rdi, 16
    add rsi, 16
    sub rcx, 4
    cmp rcx, 4
    jge .Lsimd_loop
    movq rax, xmm2
    psrldq xmm2, 8
    movq rdx, xmm2
    add rax, rdx
    jmp .Lscalar_check
.Lno_simd:
    xor rax, rax
.Lscalar_check:
    test rcx, rcx
    jz .Ldone
.Lscalar:
    movsxd rdx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rdx, r8
    add rax, rdx
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Lscalar
.Ldone:
    ret
