export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%one =l copy 1
	jmp @head
@head
	%max =l phi @start %max0, @latch %max2
	%i =l phi @start %one, @latch %i2
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %max
	jnz %gt, @take, @skip
@take
	jmp @latch
@skip
	jmp @latch
@latch
	%max2 =l phi @take %v, @skip %max
	%i2 =l add %i, 1
	jmp @head
@end
	ret %max
}
