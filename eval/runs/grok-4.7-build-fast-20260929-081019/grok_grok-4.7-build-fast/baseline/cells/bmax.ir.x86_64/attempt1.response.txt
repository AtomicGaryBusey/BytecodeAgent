export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%m =l phi @start %m0, @next %m2
	%i =l phi @start %i0, @next %i1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%lt =w csltl %m, %v
	jnz %lt, @update, @next
@update
	jmp @next
@next
	%m2 =l phi @update %v, @body %m
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %m
}
