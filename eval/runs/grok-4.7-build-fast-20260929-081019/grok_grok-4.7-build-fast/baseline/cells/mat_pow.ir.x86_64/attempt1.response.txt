export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%two_nn =l mul %nn, l 2
	jmp @id_i
@id_i
	%i =l phi @start l 0, @id_i_next %i_next
	%i_lt =w csltl %i, %n
	jnz %i_lt, @id_j, @pow_r
@id_j
	%j =l phi @id_i l 0, @id_j_next %j_next
	%j_lt =w csltl %j, %n
	jnz %j_lt, @id_body, @id_i_next
@id_body
	%ij_eq =w ceql %i, %j
	%val =l extuw %ij_eq
	%id_row =l mul %i, %n
	%id_idx =l add %id_row, %j
	%id_byte =l mul %id_idx, l 8
	%id_ptr =l add %out, %id_byte
	storel %val, %id_ptr
	jmp @id_j_next
@id_j_next
	%j_next =l add %j, l 1
	jmp @id_j
@id_i_next
	%i_next =l add %i, l 1
	jmp @id_i
@pow_r
	%r =l phi @id_i l 0, @pow_r_next %r_next
	%r_lt =w csltl %r, %k
	jnz %r_lt, @mul_i, @zero
@mul_i
	%mi =l phi @pow_r l 0, @mul_i_next %mi_next
	%mi_lt =w csltl %mi, %n
	jnz %mi_lt, @mul_j, @copy
@mul_j
	%mj =l phi @mul_i l 0, @mul_j_next %mj_next
	%mj_lt =w csltl %mj, %n
	jnz %mj_lt, @mul_t, @mul_i_next
@mul_t
	%t =l phi @mul_j l 0, @mul_t_next %t_next
	%s =l phi @mul_j l 0, @mul_t_next %s_next
	%t_lt =w csltl %t, %n
	jnz %t_lt, @mul_body, @mul_store
@mul_body
	%m_row =l mul %mi, %n
	%o_idx =l add %m_row, %t
	%o_byte =l mul %o_idx, l 8
	%o_ptr =l add %out, %o_byte
	%ov =l loadl %o_ptr
	%a_row =l mul %t, %n
	%a_idx =l add %a_row, %mj
	%a_byte =l mul %a_idx, l 8
	%a_ptr =l add %a, %a_byte
	%av =l loadl %a_ptr
	%prod =l mul %ov, %av
	%s_next =l add %s, %prod
	jmp @mul_t_next
@mul_t_next
	%t_next =l add %t, l 1
	jmp @mul_t
@mul_store
	%ms_row =l mul %mi, %n
	%ms_idx =l add %ms_row, %mj
	%ms_byte =l mul %ms_idx, l 8
	%ms_ptr =l add %tmp, %ms_byte
	storel %s, %ms_ptr
	jmp @mul_j_next
@mul_j_next
	%mj_next =l add %mj, l 1
	jmp @mul_j
@mul_i_next
	%mi_next =l add %mi, l 1
	jmp @mul_i
@copy
	%c =l phi @mul_i l 0, @copy_body %c_next
	%c_lt =w csltl %c, %nn
	jnz %c_lt, @copy_body, @pow_r_next
@copy_body
	%c_byte =l mul %c, l 8
	%c_tmp_ptr =l add %tmp, %c_byte
	%cv =l loadl %c_tmp_ptr
	%c_out_ptr =l add %out, %c_byte
	storel %cv, %c_out_ptr
	%c_next =l add %c, l 1
	jmp @copy
@pow_r_next
	%r_next =l add %r, l 1
	jmp @pow_r
@zero
	%z =l phi @pow_r l 0, @zero_body %z_next
	%z_lt =w csltl %z, %two_nn
	jnz %z_lt, @zero_body, @done
@zero_body
	%z_byte =l mul %z, l 8
	%z_ptr =l add %tmp, %z_byte
	storel l 0, %z_ptr
	%z_next =l add %z, l 1
	jmp @zero
@done
	ret l 0
}
