export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%one =l copy 1
	%eight =l copy 8
	%sixteen =l copy 16
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%odds0 =l and %y0, %one
	%i0 =l copy 1
	jmp @check
@check
	%i =l phi @start %i0, @aftermax %inext
	%min =l phi @start %y0, @aftermax %min2
	%max =l phi @start %y0, @aftermax %max2
	%sum =l phi @start %y0, @aftermax %sum2
	%odds =l phi @start %odds0, @aftermax %odds2
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, %eight
	%ptr =l add %a, %off
	%ai =l loadl %ptr
	%y =l call %f(l %ai)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @keepmin
@setmin
	jmp @aftermin
@keepmin
	jmp @aftermin
@aftermin
	%min2 =l phi @setmin %y, @keepmin %min
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @keepmax
@setmax
	jmp @aftermax
@keepmax
	jmp @aftermax
@aftermax
	%max2 =l phi @setmax %y, @keepmax %max
	%sum2 =l add %sum, %y
	%bit =l and %y, %one
	%odds2 =l add %odds, %bit
	%inext =l add %i, %one
	jmp @check
@done
	storel %min, %out
	%out1 =l add %out, %eight
	storel %max, %out1
	%out2 =l add %out, %sixteen
	storel %sum, %out2
	ret %odds
}
