export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %countnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%da =l add %dst, %off
	%sv =l loadl %sa
	%dv =l call %f(l %sv)
	storel %dv, %da
	%lt =w csltl %dv, %sv
	%lte =l extsw %lt
	%countnext =l add %count, %lte
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
