export function l $map_sum(l %a, l %n, l %f) {
@start
	%zero =l copy 0
	%iz =w ceql %n, 0
	jnz %iz, @exit, @head
@head
	%i =l phi @start %zero, @head %i1
	%s =l phi @start %zero, @head %s1
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%r =l call %f(l %v)
	%s1 =l add %s, %r
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @head, @exit
@exit
	%sum =l phi @start %zero, @head %s1
	ret %sum
}
