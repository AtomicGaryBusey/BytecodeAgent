export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%jinit =l copy 0
	jmp @izero
@izero
	%jz =l phi @start %jinit, @izero_body %jznext
	%jzle =w cslel %jz, %n
	jnz %jzle, @izero_body, @idone
@izero_body
	%zoff =l mul %jz, 8
	%zaddr =l add %row, %zoff
	storel %zero, %zaddr
	%jznext =l add %jz, 1
	jmp @izero
@idone
	%iinit =l copy 0
	jmp @iloop
@iloop
	%i =l phi @idone %iinit, @iinc %inext
	%ilt =w csltl %i, %m
	jnz %ilt, @ibody, @done
@ibody
	%jinit2 =l copy 1
	%diaginit =l copy 0
	jmp @jloop
@jloop
	%j =l phi @ibody %jinit2, @jcont %jnext
	%diag =l phi @ibody %diaginit, @jcont %diagnew
	%jle =w cslel %j, %n
	jnz %jle, @jbody, @iinc
@jbody
	%joff =l mul %j, 8
	%jaddr =l add %row, %joff
	%up =l loadl %jaddr
	%jm1 =l sub %j, 1
	%jm1off =l mul %jm1, 8
	%laddr =l add %row, %jm1off
	%left =l loadl %laddr
	%xaddr =l add %x, %i
	%xb =w loadub %xaddr
	%yaddr =l add %y, %jm1
	%yb =w loadub %yaddr
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%diag1 =l add %diag, 1
	jmp @jcont
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @takeup, @takeleft
@takeup
	%maxu =l copy %up
	jmp @jcont
@takeleft
	%maxl =l copy %left
	jmp @jcont
@jcont
	%val =l phi @match %diag1, @takeup %maxu, @takeleft %maxl
	storel %val, %jaddr
	%diagnew =l copy %up
	%jnext =l add %j, 1
	jmp @jloop
@iinc
	%inext =l add %i, 1
	jmp @iloop
@done
	%noff =l mul %n, 8
	%naddr =l add %row, %noff
	%result =l loadl %naddr
	ret %result
}
