export function l $isqrt(l %n) {
@start
	%umax =l copy 4294967295
	%small =w cultl %n, %umax
	jnz %small, @small, @big
@small
	%lo0 =l copy 0
	%hi0 =l copy %n
	jmp @loop
@big
	%lo1 =l copy 0
	%hi1 =l copy %umax
	jmp @loop
@loop
	%lo =l phi @small %lo0, @big %lo1, @yes %lo_y, @no %lo_n
	%hi =l phi @small %hi0, @big %hi1, @yes %hi_y, @no %hi_n
	%eq =w ceql %lo, %hi
	jnz %eq, @end, @cont
@cont
	%diff =l sub %hi, %lo
	%diff1 =l add %diff, 1
	%half =l shr %diff1, 1
	%mid =l add %lo, %half
	%q =l udiv %n, %mid
	%ok =w culel %mid, %q
	jnz %ok, @yes, @no
@yes
	%lo_y =l copy %mid
	%hi_y =l copy %hi
	jmp @loop
@no
	%lo_n =l copy %lo
	%hi_n =l sub %mid, 1
	jmp @loop
@end
	ret %lo
}
