export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @head
@head
	%i =l phi @start %i0, @body %inext
	%cnt =l phi @start %c0, @body %cntnext
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%gt =w cugtl %elem, %t
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l add %i, 1
	jmp @head
@end
	ret %cnt
}
