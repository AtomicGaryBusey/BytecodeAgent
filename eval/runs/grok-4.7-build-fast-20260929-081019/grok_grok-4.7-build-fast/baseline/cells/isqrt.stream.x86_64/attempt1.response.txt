isqrt:
    xor r8, r8
    mov r9d, 0xffffffff
.Lloop:
    cmp r8, r9
    ja .Ldone
    mov rax, r8
    add rax, r9
    shr rax, 1
    mov r10, rax
    mul rax
    cmp rax, rdi
    ja .Lhigh
    lea r8, [r10 + 1]
    jmp .Lloop
.Lhigh:
    lea r9, [r10 - 1]
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
