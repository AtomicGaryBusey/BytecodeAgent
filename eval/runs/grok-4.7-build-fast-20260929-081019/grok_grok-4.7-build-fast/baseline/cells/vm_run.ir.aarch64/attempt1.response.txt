export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc0 =l copy 0
	%dp0 =l copy 0
	jmp @loop
@loop
	%pc =l phi @start %pc0, @next %pcn
	%dp =l phi @start %dp0, @next %dpn
	%end =w ceql %pc, %len
	jnz %end, @zloop, @decode
@decode
	%pc1 =l add %pc, 1
	%pc2 =l add %pc, 2
	%sp1 =l add %dp, 1
	%spm1 =l sub %dp, 1
	%pcoff =l mul %pc, 8
	%pcaddr =l add %code, %pcoff
	%op =l loadl %pcaddr
	%ishalt =w ceql %op, 0
	jnz %ishalt, @dohalt, @cpush
@cpush
	%ispush =w ceql %op, 1
	jnz %ispush, @dopush, @cadd
@cadd
	%isadd =w ceql %op, 2
	jnz %isadd, @doadd, @csub
@csub
	%issub =w ceql %op, 3
	jnz %issub, @dosub, @cmul
@cmul
	%ismul =w ceql %op, 4
	jnz %ismul, @domul, @cdup
@cdup
	%isdup =w ceql %op, 5
	jnz %isdup, @dodup, @cswap
@cswap
	%isswap =w ceql %op, 6
	jnz %isswap, @doswap, @cdrop
@cdrop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @dodrop, @cover
@cover
	%isover =w ceql %op, 8
	jnz %isover, @doover, @clt
@clt
	%islt =w ceql %op, 9
	jnz %islt, @dolt, @cjz
@cjz
	%isjz =w ceql %op, 10
	jnz %isjz, @dojz, @cjmp
@cjmp
	%isjmp =w ceql %op, 11
	jnz %isjmp, @dojmp, @doother
@dohalt
	jmp @zloop
@dopush
	%pushoff =l mul %pc1, 8
	%pushaddr =l add %code, %pushoff
	%pushv =l loadl %pushaddr
	%pushsoff =l mul %dp, 8
	%pushsaddr =l add %stack, %pushsoff
	storel %pushv, %pushsaddr
	jmp @next
@doadd
	%addbidx =l sub %dp, 1
	%addaidx =l sub %dp, 2
	%addboff =l mul %addbidx, 8
	%addaoff =l mul %addaidx, 8
	%addbaddr =l add %stack, %addboff
	%addaaddr =l add %stack, %addaoff
	%addb =l loadl %addbaddr
	%adda =l loadl %addaaddr
	%addsum =l add %adda, %addb
	storel %addsum, %addaaddr
	jmp @epdec
@dosub
	%subbidx =l sub %dp, 1
	%subaidx =l sub %dp, 2
	%subboff =l mul %subbidx, 8
	%subaoff =l mul %subaidx, 8
	%subbaddr =l add %stack, %subboff
	%subaaddr =l add %stack, %subaoff
	%subb =l loadl %subbaddr
	%suba =l loadl %subaaddr
	%subdiff =l sub %suba, %subb
	storel %subdiff, %subaaddr
	jmp @epdec
@domul
	%mulbidx =l sub %dp, 1
	%mulaidx =l sub %dp, 2
	%mulboff =l mul %mulbidx, 8
	%mulaoff =l mul %mulaidx, 8
	%mulbaddr =l add %stack, %mulboff
	%mulaaddr =l add %stack, %mulaoff
	%mulb =l loadl %mulbaddr
	%mula =l loadl %mulaaddr
	%mulprod =l mul %mula, %mulb
	storel %mulprod, %mulaaddr
	jmp @epdec
@dodup
	%dupidx =l sub %dp, 1
	%dupoff =l mul %dupidx, 8
	%dupaddr =l add %stack, %dupoff
	%dupv =l loadl %dupaddr
	%dupssoff =l mul %dp, 8
	%dupsaddr =l add %stack, %dupssoff
	storel %dupv, %dupsaddr
	jmp @epinc
@doswap
	%swbidx =l sub %dp, 1
	%swaidx =l sub %dp, 2
	%swboff =l mul %swbidx, 8
	%swaoff =l mul %swaidx, 8
	%swbaddr =l add %stack, %swboff
	%swaaddr =l add %stack, %swaoff
	%swb =l loadl %swbaddr
	%swa =l loadl %swaaddr
	storel %swb, %swaaddr
	storel %swa, %swbaddr
	jmp @epsame
@dodrop
	jmp @epdec
@doover
	%ovidx =l sub %dp, 2
	%ovoff =l mul %ovidx, 8
	%ovaddr =l add %stack, %ovoff
	%ovv =l loadl %ovaddr
	%ovssoff =l mul %dp, 8
	%ovsaddr =l add %stack, %ovssoff
	storel %ovv, %ovsaddr
	jmp @epinc
@dolt
	%ltbidx =l sub %dp, 1
	%ltaidx =l sub %dp, 2
	%ltboff =l mul %ltbidx, 8
	%ltaoff =l mul %ltaidx, 8
	%ltbaddr =l add %stack, %ltboff
	%ltaaddr =l add %stack, %ltaoff
	%ltb =l loadl %ltbaddr
	%lta =l loadl %ltaaddr
	%ltcmp =w csltl %lta, %ltb
	jnz %ltcmp, @lt1, @lt0
@lt1
	storel 1, %ltaaddr
	jmp @epdec
@lt0
	storel 0, %ltaaddr
	jmp @epdec
@dojz
	%jzidx =l sub %dp, 1
	%jzoff =l mul %jzidx, 8
	%jzaddr =l add %stack, %jzoff
	%jzv =l loadl %jzaddr
	%jzisz =w ceql %jzv, 0
	jnz %jzisz, @jzyes, @jzno
@jzyes
	%jytoff =l mul %pc1, 8
	%jytaddr =l add %code, %jytoff
	%jyt =l loadl %jytaddr
	jmp @next
@jzno
	jmp @next
@dojmp
	%jmptoff =l mul %pc1, 8
	%jmptaddr =l add %code, %jmptoff
	%jmpt =l loadl %jmptaddr
	jmp @next
@doother
	jmp @epsame
@epdec
	jmp @next
@epinc
	jmp @next
@epsame
	jmp @next
@next
	%pcn =l phi @dopush %pc2, @epdec %pc1, @epinc %pc1, @epsame %pc1, @jzyes %jyt, @jzno %pc2, @dojmp %jmpt
	%dpn =l phi @dopush %sp1, @epdec %spm1, @epinc %sp1, @epsame %dp, @jzyes %spm1, @jzno %spm1, @dojmp %dp
	jmp @loop
@zloop
	%zi =l phi @loop %dp, @dohalt %dp, @zstep %zi1
	%zcmp =w csltl %zi, %cap
	jnz %zcmp, @zstep, @doret
@zstep
	%zoff =l mul %zi, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%zi1 =l add %zi, 1
	jmp @zloop
@doret
	%isempty =w ceql %dp, 0
	jnz %isempty, @ret0, @rettop
@ret0
	ret 0
@rettop
	%topidx =l sub %dp, 1
	%topoff =l mul %topidx, 8
	%topaddr =l add %stack, %topoff
	%topv =l loadl %topaddr
	ret %topv
}
