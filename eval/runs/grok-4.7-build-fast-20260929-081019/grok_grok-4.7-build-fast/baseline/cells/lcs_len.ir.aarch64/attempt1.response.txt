export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	jmp @izero
@izero
	%j =l phi @start %zero, @izbody %j1
	%jok =w cslel %j, %n
	jnz %jok, @izbody, @iinit
@izbody
	%jsh =l mul %j, 8
	%jp =l add %row, %jsh
	storel %zero, %jp
	%j1 =l add %j, %one
	jmp @izero
@iinit
	jmp @iloop
@iloop
	%i =l phi @iinit %zero, @jend %i1
	%iok =w csltl %i, %m
	jnz %iok, @ibod, @end
@ibod
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @jloop
@jloop
	%j2 =l phi @ibod %one, @jnext %j3
	%diag =l phi @ibod %zero, @jnext %diag1
	%j2ok =w cslel %j2, %n
	jnz %j2ok, @jcmp, @jend
@jcmp
	%j2sh =l mul %j2, 8
	%rp =l add %row, %j2sh
	%up =l loadl %rp
	%jm =l sub %j2, %one
	%jmsh =l mul %jm, 8
	%lp =l add %row, %jmsh
	%left =l loadl %lp
	%yp =l add %y, %jm
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nmax
@match
	%mv =l add %diag, %one
	jmp @jstore
@nmax
	%gt =w csgtl %up, %left
	jnz %gt, @picku, @pickl
@picku
	jmp @jstore
@pickl
	jmp @jstore
@jstore
	%nv =l phi @match %mv, @picku %up, @pickl %left
	storel %nv, %rp
	jmp @jnext
@jnext
	%diag1 =l copy %up
	%j3 =l add %j2, %one
	jmp @jloop
@jend
	%i1 =l add %i, %one
	jmp @iloop
@end
	%nsh =l mul %n, 8
	%np =l add %row, %nsh
	%res =l loadl %np
	ret %res
}
