export function l $collatz_max(l %n) {
@start
	%c0 =l copy 0
	%c1 =l copy 1
	%c2 =l copy 2
	%c3 =l copy 3
	jmp @kloop
@kloop
	%k =l phi @start %c1, @knext %kn
	%maxs =l phi @start %c0, @knext %maxsn
	%kcmp =w cslel %k, %n
	jnz %kcmp, @xloop, @done
@xloop
	%x =l phi @kloop %k, @xnext %xn
	%steps =l phi @kloop %c0, @xnext %stepsn
	%is1 =w ceql %x, %c1
	jnz %is1, @updmax, @split
@split
	%par =l and %x, %c1
	%isodd =w cnel %par, %c0
	jnz %isodd, @odd, @even
@even
	%xe =l div %x, %c2
	jmp @xnext
@odd
	%t3 =l mul %x, %c3
	%xo =l add %t3, %c1
	jmp @xnext
@xnext
	%xn =l phi @even %xe, @odd %xo
	%stepsn =l add %steps, %c1
	jmp @xloop
@updmax
	%gt =w csgtl %steps, %maxs
	jnz %gt, @newmax, @keepmax
@newmax
	jmp @knext
@keepmax
	jmp @knext
@knext
	%maxsn =l phi @newmax %steps, @keepmax %maxs
	%kn =l add %k, %c1
	jmp @kloop
@done
	ret %maxs
}
