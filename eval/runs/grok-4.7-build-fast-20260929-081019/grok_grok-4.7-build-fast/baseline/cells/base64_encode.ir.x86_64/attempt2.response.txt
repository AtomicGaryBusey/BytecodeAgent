export function l $base64_encode(l %src, l %n, l %out) {
@start
	%idxs =l alloc4 4
	jmp @loop
@loop
	%i =l phi @start 0, @after %i3
	%o =l phi @start 0, @after %o4
	%left =l sub %n, %i
	%lt3 =w cultl %left, 3
	jnz %lt3, @tail, @trip
@trip
	%p0 =l add %src, %i
	%b0w =w loadub %p0
	%i1 =l add %i, 1
	%p1 =l add %src, %i1
	%b1w =w loadub %p1
	%i2 =l add %i, 2
	%p2 =l add %src, %i2
	%b2w =w loadub %p2
	%b0s =w shl %b0w, 16
	%b1s =w shl %b1w, 8
	%m01 =w or %b0s, %b1s
	%acc =w or %m01, %b2w
	%t0 =w shr %acc, 18
	%v0 =w and %t0, 63
	%t1 =w shr %acc, 12
	%v1 =w and %t1, 63
	%t2 =w shr %acc, 6
	%v2 =w and %t2, 63
	%v3 =w and %acc, 63
	storeb %v0, %idxs
	%ip1 =l add %idxs, 1
	storeb %v1, %ip1
	%ip2 =l add %idxs, 2
	storeb %v2, %ip2
	%ip3 =l add %idxs, 3
	storeb %v3, %ip3
	jmp @emit
@tail
	%z =w ceql %left, 0
	jnz %z, @done, @tailnz
@tailnz
	%isone =w ceql %left, 1
	jnz %isone, @do1, @do2
@do1
	%q0 =l add %src, %i
	%c0w =w loadub %q0
	%r0 =w shr %c0w, 2
	%low =w and %c0w, 3
	%r1 =w shl %low, 4
	storeb %r0, %idxs
	%q1p =l add %idxs, 1
	storeb %r1, %q1p
	jmp @emit
@do2
	%u0 =l add %src, %i
	%d0w =w loadub %u0
	%i1b =l add %i, 1
	%u1 =l add %src, %i1b
	%d1w =w loadub %u1
	%y0 =w shr %d0w, 2
	%lo =w and %d0w, 3
	%lo4 =w shl %lo, 4
	%hi =w shr %d1w, 4
	%y1 =w or %lo4, %hi
	%lo2 =w and %d1w, 15
	%y2 =w shl %lo2, 2
	storeb %y0, %idxs
	%z1 =l add %idxs, 1
	storeb %y1, %z1
	%z2 =l add %idxs, 2
	storeb %y2, %z2
	jmp @emit
@emit
	%k =l phi @trip 0, @do1 0, @do2 0, @estep %k1
	%cnt =l phi @trip 4, @do1 2, @do2 3, @estep %cntk
	%pads =l phi @trip 0, @do1 2, @do2 1, @estep %padsk
	%cont =w phi @trip 1, @do1 0, @do2 0, @estep %contk
	%kend =w ceql %k, %cnt
	jnz %kend, @padloop, @estep
@estep
	%ip =l add %idxs, %k
	%vw =w loadub %ip
	%n25 =w copy 25
	%ge26 =w cultw %n25, %vw
	%n51 =w copy 51
	%ge52 =w cultw %n51, %vw
	%eq62 =w ceqw %vw, 62
	%eq63 =w ceqw %vw, 63
	%c65 =w add %vw, 65
	%m6 =w mul %ge26, 6
	%c1 =w add %c65, %m6
	%m75 =w mul %ge52, 75
	%c2 =w sub %c1, %m75
	%m15 =w mul %eq62, 15
	%c3 =w sub %c2, %m15
	%m12 =w mul %eq63, 12
	%c4 =w sub %c3, %m12
	%off =l add %o, %k
	%dp =l add %out, %off
	storeb %c4, %dp
	%k1 =l add %k, 1
	%cntk =l copy %cnt
	%padsk =l copy %pads
	%contk =w copy %cont
	jmp @emit
@padloop
	%p =l phi @emit 0, @pstep %ps
	%pend =w ceql %p, %pads
	jnz %pend, @padafter, @pstep
@pstep
	%po =l add %cnt, %p
	%po2 =l add %o, %po
	%pp =l add %out, %po2
	%eqch =w copy 61
	storeb %eqch, %pp
	%ps =l add %p, 1
	jmp @padloop
@padafter
	jnz %cont, @after, @done
@after
	%i3 =l add %i, 3
	%o4 =l add %o, 4
	jmp @loop
@done
	%n2 =l add %n, 2
	%cq =l udiv %n2, 3
	%retv =l mul %cq, 4
	ret %retv
}
