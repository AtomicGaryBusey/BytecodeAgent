

export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum =l add %a, %b
    storel %sum, %out
    %b_neg =w csltl %b, 0
    jnz %b_neg, @neg, @pos
@pos
    %ov_pos =w csltl %sum, %a
    jnz %ov_pos, @ret1, @ret0
@neg
    %ov_neg =w csltl %a, %sum
    jnz %ov_neg, @ret1, @ret0
@ret1
    ret 1
@ret0
    ret 0
}
