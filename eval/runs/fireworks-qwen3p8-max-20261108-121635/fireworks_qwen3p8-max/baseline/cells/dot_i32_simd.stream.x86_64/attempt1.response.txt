

dot_i32_simd:
    xor eax, eax
    pxor xmm4, xmm4
    cmp rdx, 4
    jl .Lscalar
.Lvec:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pmuldq xmm2, xmm3
    psrldq xmm0, 4
    psrldq xmm1, 4
    pmuldq xmm0, xmm1
    paddq xmm0, xmm2
    paddq xmm4, xmm0
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jge .Lvec
    movq rax, xmm4
    psrldq xmm4, 8
    movq rcx, xmm4
    add rax, rcx
.Lscalar:
    test rdx, rdx
    jle .Ldone
.Lscalar_loop:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Lscalar_loop
.Ldone:
    ret
