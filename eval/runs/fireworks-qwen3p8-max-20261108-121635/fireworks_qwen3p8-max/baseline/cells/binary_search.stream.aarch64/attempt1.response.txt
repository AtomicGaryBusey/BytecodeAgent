

binary_search:
    cmp x1, #0
    b.le .Lnotfound
    mov x3, x0
    mov x4, #0
    sub x5, x1, #1
.Lloop:
    cmp x4, x5
    b.gt .Lnotfound
    sub x6, x5, x4
    lsr x6, x6, #1
    add x6, x4, x6
    ldr x7, [x3, x6, lsl #3]
    cmp x7, x2
    b.eq .Lfound
    b.lt .Lless
    sub x5, x6, #1
    b .Lloop
.Lless:
    add x4, x6, #1
    b .Lloop
.Lfound:
    mov x0, x6
    ret
.Lnotfound:
    mov x0, #-1
    ret
