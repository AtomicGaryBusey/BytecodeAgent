

export function l $isqrt(l %n) {
@start
  %s0 =l shl 1, 31
  jmp @loop %s0
@loop %s
  %b =l and %s, 4294967295
  %z =w ceql %b, 0
  jnz %z, @end, @body
@body
  %r =l shr %s, 32
  %c =l add %r, %b
  %q =l mul %c, %c
  %ok =w culel %q, %n
  %b1 =l shr %b, 1
  jnz %ok, @take, @skip
@take
  %h =l shl %c, 32
  %s1 =l or %h, %b1
  jmp @loop %s1
@skip
  %h0 =l shl %r, 32
  %s2 =l or %h0, %b1
  jmp @loop %s2
@end
  %res =l shr %s, 32
  ret %res
}
