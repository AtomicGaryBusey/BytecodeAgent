

export function l $isqrt(l %n) {
@start
    %zero =l sub %n, %n
    %b0 =l shl 1, 31
    jmp @head
@head
    %r =l phi @start %zero, @take %rt, @skip %rs
    %b =l phi @start %b0, @take %bt, @skip %bs
    %z =w ceql %b, %zero
    jnz %z, @done, @body
@body
    %t =l or %r, %b
    %q =l mul %t, %t
    %ok =w culel %q, %n
    %b1 =l shr %b, 1
    jnz %ok, @take, @skip
@take
    %rt =l or %t, %zero
    %bt =l or %b1, %zero
    jmp @head
@skip
    %rs =l or %r, %zero
    %bs =l or %b1, %zero
    jmp @head
@done
    ret %r
}
