

dot_i32_simd:
    mov x3, #0
    movi v0.2d, #0
    cmp x2, #4
    b.lo .Ltail
.Lvec:
    ld1 {v1.s}[0], [x0], #4
    ld1 {v1.s}[1], [x0], #4
    ld1 {v1.s}[2], [x0], #4
    ld1 {v1.s}[3], [x0], #4
    ld1 {v2.s}[0], [x1], #4
    ld1 {v2.s}[1], [x1], #4
    ld1 {v2.s}[2], [x1], #4
    ld1 {v2.s}[3], [x1], #4
    smlal v0.2d, v1.2s, v2.2s
    smlal2 v0.2d, v1.4s, v2.4s
    subs x2, x2, #4
    cmp x2, #4
    b.ge .Lvec
    umov x3, v0.d[0]
    umov x4, v0.d[1]
    add x3, x3, x4
.Ltail:
    cbz x2, .Ldone
.Ltail_loop:
    ldr w4, [x0], #4
    ldr w5, [x1], #4
    smaddl x3, w4, w5, x3
    subs x2, x2, #1
    b.ne .Ltail_loop
.Ldone:
    mov x0, x3
    ret
