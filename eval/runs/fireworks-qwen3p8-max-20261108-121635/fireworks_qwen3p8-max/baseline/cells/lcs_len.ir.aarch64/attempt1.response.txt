

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %n1 =l add %n, %one
    jmp @init_loop
@init_loop
    %k =l phi @start %zero, @init_body %k1
    %kc =w csltl %k, %n1
    jnz %kc, @init_body, @outer_loop
@init_body
    %koff =l mul %k, %eight
    %kaddr =l add %row, %koff
    storel %zero, %kaddr
    %k1 =l add %k, %one
    jmp @init_loop
@outer_loop
    %i =l phi @init_loop %zero, @outer_next %i1
    %ic =w csltl %i, %m
    jnz %ic, @outer_body, @finish
@outer_body
    %xaddr =l add %x, %i
    %xi =w loadub %xaddr
    jmp @inner_loop
@inner_loop
    %j =l phi @outer_body %one, @inner_cont %j1
    %diag =l phi @outer_body %zero, @inner_cont %nextdiag
    %jc =w csltl %j, %n1
    jnz %jc, @inner_body, @outer_next
@inner_body
    %joff =l mul %j, %eight
    %rj =l add %row, %joff
    %up =l loadl %rj
    %jm1 =l sub %j, %one
    %jm1off =l mul %jm1, %eight
    %rjm1 =l add %row, %jm1off
    %left =l loadl %rjm1
    %yaddr =l add %y, %jm1
    %yj =w loadub %yaddr
    %eq =w ceqw %xi, %yj
    jnz %eq, @match, @nomatch
@match
    %mval =l add %diag, %one
    storel %mval, %rj
    jmp @inner_cont
@nomatch
    %lt =w csltl %up, %left
    jnz %lt, @take_left, @take_up
@take_left
    storel %left, %rj
    jmp @inner_cont
@take_up
    storel %up, %rj
    jmp @inner_cont
@inner_cont
    %j1 =l add %j, %one
    %nextdiag =l copy %up
    jmp @inner_loop
@outer_next
    %i1 =l add %i, %one
    jmp @outer_loop
@finish
    %noff =l mul %n, %eight
    %naddr =l add %row, %noff
    %res =l loadl %naddr
    ret %res
}
