

export function l $binary_search(l %a, l %n, l %key) {
@start
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %eight =l copy 8
    %hi0 =l copy %n
    %neg1 =l sub %zero, %one
    jmp @loop
@loop
    %lo =l phi @start %zero, @right %lo_next, @left %lo_keep
    %hi =l phi @start %hi0, @right %hi_keep, @left %hi_next
    %cond =w csltl %lo, %hi
    jnz %cond, @body, @notfound
@body
    %diff =l sub %hi, %lo
    %half =l div %diff, %two
    %mid =l add %lo, %half
    %off =l mul %mid, %eight
    %addr =l add %a, %off
    %val =l loadl %addr
    %less =w csltl %val, %key
    jnz %less, @right, @notless
@right
    %lo_next =l add %mid, %one
    %hi_keep =l copy %hi
    jmp @loop
@notless
    %greater =w csltl %key, %val
    jnz %greater, @left, @found
@left
    %lo_keep =l copy %lo
    %hi_next =l copy %mid
    jmp @loop
@found
    ret %mid
@notfound
    ret %neg1
}
