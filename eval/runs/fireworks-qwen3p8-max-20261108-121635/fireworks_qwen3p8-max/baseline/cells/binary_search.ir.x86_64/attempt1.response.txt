

export function l $binary_search(l %a, l %n, l %key) {
@start
    jmp @loop(l 0, l %n)
@loop(l %lo, l %hi)
    %cond =w csltl %lo, %hi
    jnz %cond, @body, @notfound
@body
    %diff =l sub %hi, %lo
    %half =l div %diff, 2
    %mid =l add %lo, %half
    %off =l mul %mid, 8
    %addr =l add %a, %off
    %val =l loadl %addr
    %less =w csltl %val, %key
    jnz %less, @right, @notless
@right
    %lo1 =l add %mid, 1
    jmp @loop(l %lo1, l %hi)
@notless
    %greater =w csltl %key, %val
    jnz %greater, @left, @found
@left
    jmp @loop(l %lo, l %mid)
@found
    ret %mid
@notfound
    %neg =l sub 0, 1
    ret %neg
}
