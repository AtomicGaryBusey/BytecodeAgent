

mp_mul:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rdx
    mov r14, r8
    mov r15, rsi
    mov rbx, rcx
    cld
    lea r9, [r15 + rbx]
    xor ecx, ecx
    xor eax, eax
.Lzero_loop:
    cmp rcx, r9
    jge .Lzero_done
    mov [r14 + rcx*8], rax
    inc rcx
    jmp .Lzero_loop
.Lzero_done:
    test r15, r15
    jz .Lreturn_zero
    test rbx, rbx
    jz .Lreturn_zero
    xor r10d, r10d
.Lrow_loop:
    cmp r10, r15
    jge .Lmul_done
    mov r11, [r12 + r10*8]
    xor r8d, r8d
    xor r9d, r9d
.Lcol_loop:
    cmp r9, rbx
    jge .Lrow_done
    mov rax, r11
    mul qword ptr [r13 + r9*8]
    lea rcx, [r10 + r9]
    add rax, [r14 + rcx*8]
    adc rdx, 0
    add rax, r8
    adc rdx, 0
    mov [r14 + rcx*8], rax
    mov r8, rdx
    inc r9
    jmp .Lcol_loop
.Lrow_done:
    lea rcx, [r10 + rbx]
    mov [r14 + rcx*8], r8
    inc r10
    jmp .Lrow_loop
.Lmul_done:
    lea r9, [r15 + rbx]
    xor eax, eax
    test r9, r9
    jz .Lreturn_zero
    lea r10, [r9 - 1]
.Lscan_loop:
    cmp r10, 0
    jl .Lreturn_zero
    cmp rax, [r14 + r10*8]
    jne .Lfound
    dec r10
    jmp .Lscan_loop
.Lfound:
    lea rax, [r10 + 1]
    jmp .Lepilogue
.Lreturn_zero:
    xor eax, eax
.Lepilogue:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
