

export function l $collatz_max(l %n) {
@start
	%limit =l add %n, 1
	%k0 =l copy 1
	%max0 =l copy 0
	jmp @outer_loop
@outer_loop
	%k =l phi @start %k0, @outer_next %k_next
	%max =l phi @start %max0, @outer_next %max_next
	%cont =w csltl %k, %limit
	jnz %cont, @outer_body, @outer_done
@outer_body
	%x0 =l copy %k
	%steps0 =l copy 0
	jmp @inner_loop
@inner_loop
	%x =l phi @outer_body %x0, @inner_next %x_next
	%steps =l phi @outer_body %steps0, @inner_next %steps_next
	%done =w ceql %x, 1
	jnz %done, @inner_done, @inner_body
@inner_body
	%bit =l and %x, 1
	%even =w ceql %bit, 0
	jnz %even, @even_case, @odd_case
@even_case
	%x_even =l div %x, 2
	jmp @inner_next
@odd_case
	%t =l mul %x, 3
	%x_odd =l add %t, 1
	jmp @inner_next
@inner_next
	%x_next =l phi @even_case %x_even, @odd_case %x_odd
	%steps_next =l add %steps, 1
	jmp @inner_loop
@inner_done
	%less =w csltl %max, %steps
	jnz %less, @update_max, @keep_max
@update_max
	%max_up =l copy %steps
	jmp @outer_next
@keep_max
	%max_keep =l copy %max
	jmp @outer_next
@outer_next
	%max_next =l phi @update_max %max_up, @keep_max %max_keep
	%k_next =l add %k, 1
	jmp @outer_loop
@outer_done
	ret %max
}
