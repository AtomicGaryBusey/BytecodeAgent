

export function l $crc32(l %p, l %n) {
@start
	%zero =l copy 0
	%all =w sub 0, 1
	%poly =w sub 0, 306674912
	%enter =w csltl %zero, %n
	jnz %enter, @loop, @end
@loop
	%i =l phi @start %zero, @loop %i1
	%crc =w phi @start %all, @loop %c8
	%addr =l add %p, %i
	%byte =w loadub %addr
	%x0 =w xor %crc, %byte
	%b1 =w and %x0, 1
	%m1 =w sub 0, %b1
	%s1 =w shr %x0, 1
	%q1 =w and %m1, %poly
	%c1 =w xor %s1, %q1
	%b2 =w and %c1, 1
	%m2 =w sub 0, %b2
	%s2 =w shr %c1, 1
	%q2 =w and %m2, %poly
	%c2 =w xor %s2, %q2
	%b3 =w and %c2, 1
	%m3 =w sub 0, %b3
	%s3 =w shr %c2, 1
	%q3 =w and %m3, %poly
	%c3 =w xor %s3, %q3
	%b4 =w and %c3, 1
	%m4 =w sub 0, %b4
	%s4 =w shr %c3, 1
	%q4 =w and %m4, %poly
	%c4 =w xor %s4, %q4
	%b5 =w and %c4, 1
	%m5 =w sub 0, %b5
	%s5 =w shr %c4, 1
	%q5 =w and %m5, %poly
	%c5 =w xor %s5, %q5
	%b6 =w and %c5, 1
	%m6 =w sub 0, %b6
	%s6 =w shr %c5, 1
	%q6 =w and %m6, %poly
	%c6 =w xor %s6, %q6
	%b7 =w and %c6, 1
	%m7 =w sub 0, %b7
	%s7 =w shr %c6, 1
	%q7 =w and %m7, %poly
	%c7 =w xor %s7, %q7
	%b8 =w and %c7, 1
	%m8 =w sub 0, %b8
	%s8 =w shr %c7, 1
	%q8 =w and %m8, %poly
	%c8 =w xor %s8, %q8
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @end
@end
	%final =w phi @start %all, @loop %c8
	%inv =w xor %final, %all
	%res =l extuw %inv
	ret %res
}
