

export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %n =l add %na, %nb
    %nbytes =l mul %n, 8
    %end =l add %r, %nbytes
    jmp @zero_loop, %r, 0
@zero_loop(%zp l, %zi l)
    %zc =w cultl %zi, %n
    jnz %zc, @zero_body, @zero_done
@zero_body
    storel 0, %zp
    %zp1 =l add %zp, 8
    %zi1 =l add %zi, 1
    jmp @zero_loop, %zp1, %zi1
@zero_done
    %na0 =w ceql %na, 0
    jnz %na0, @skip_mul, @nb_check
@nb_check
    %nb0 =w ceql %nb, 0
    jnz %nb0, @skip_mul, @outer_entry
@skip_mul
    jmp @scan_loop, %n
@outer_entry
    jmp @outer_loop, 0
@outer_loop(%i l)
    %oc =w cultl %i, %na
    jnz %oc, @outer_body, @scan_entry
@outer_body
    %ioff =l mul %i, 8
    %aptr =l add %a, %ioff
    %ai =l loadl %aptr
    %rptr0 =l add %r, %ioff
    jmp @inner_loop, 0, 0, %rptr0, %b
@inner_loop(%j l, %carry l, %rp l, %bp l)
    %ic =w cultl %j, %nb
    jnz %ic, @inner_body, @inner_done
@inner_body
    %bv =l loadl %bp
    %x0 =l and %ai, 4294967295
    %x1 =l shr %ai, 32
    %y0 =l and %bv, 4294967295
    %y1 =l shr %bv, 32
    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1
    %p00lo =l and %p00, 4294967295
    %p00hi =l shr %p00, 32
    %p01lo =l and %p01, 4294967295
    %p01hi =l shr %p01, 32
    %p10lo =l and %p10, 4294967295
    %p10hi =l shr %p10, 32
    %mid0 =l add %p00hi, %p01lo
    %mid =l add %mid0, %p10lo
    %midlo =l and %mid, 4294967295
    %midhi =l shr %mid, 32
    %loshift =l mul %midlo, 4294967296
    %lo =l add %p00lo, %loshift
    %hi0 =l add %p11, %p01hi
    %hi1 =l add %hi0, %p10hi
    %hi =l add %hi1, %midhi
    %old =l loadl %rp
    %s1 =l add %old, %lo
    %c1 =w cultl %s1, %old
    %s2 =l add %s1, %carry
    %c2 =w cultl %s2, %s1
    storel %s2, %rp
    %c1l =l extuw %c1
    %c2l =l extuw %c2
    %car0 =l add %hi, %c1l
    %car1 =l add %car0, %c2l
    %j1 =l add %j, 1
    %rp1 =l add %rp, 8
    %bp1 =l add %bp, 8
    jmp @inner_loop, %j1, %car1, %rp1, %bp1
@inner_done
    jmp @prop_check, %rp, %carry
@prop_check(%cp l, %cy l)
    %cyzero =w ceql %cy, 0
    jnz %cyzero, @outer_next, @prop_nonzero
@prop_nonzero
    %cpin =w cultl %cp, %end
    jnz %cpin, @prop_body, @outer_next
@prop_body
    %pold =l loadl %cp
    %psum =l add %pold, %cy
    storel %psum, %cp
    %pc =w cultl %psum, %pold
    %pcy =l extuw %pc
    %cp1 =l add %cp, 8
    jmp @prop_check, %cp1, %pcy
@outer_next
    %i1 =l add %i, 1
    jmp @outer_loop, %i1
@scan_entry
    jmp @scan_loop, %n
@scan_loop(%k l)
    %kzero =w ceql %k, 0
    jnz %kzero, @ret_zero, @scan_body
@scan_body
    %k1 =l sub %k, 1
    %koff =l mul %k1, 8
    %kaddr =l add %r, %koff
    %kv =l loadl %kaddr
    %kvz =w ceql %kv, 0
    jnz %kvz, @scan_next, @found
@scan_next
    jmp @scan_loop, %k1
@found
    ret %k
@ret_zero
    ret 0
}
