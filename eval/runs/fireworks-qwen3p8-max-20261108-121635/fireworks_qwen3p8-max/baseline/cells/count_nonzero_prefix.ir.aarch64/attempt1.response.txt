

export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @inc %i1
    %p =l phi @start %a, @inc %p1
    %lt =w csltl %i, %n
    jnz %lt, @load, @done
@load
    %v =l loadl %p
    %iszero =w ceql %v, 0
    jnz %iszero, @done, @inc
@inc
    %i1 =l add %i, 1
    %p1 =l add %p, 8
    jmp @loop
@done
    ret %i
}
