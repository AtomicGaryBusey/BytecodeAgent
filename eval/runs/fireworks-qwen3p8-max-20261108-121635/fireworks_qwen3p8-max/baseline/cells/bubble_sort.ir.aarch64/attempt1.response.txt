

export function l $bubble_sort(l %a, l %n) {
@start
	%limit0 =l sub %n, 1
	jmp @outer
@outer
	%limit =l phi @start %limit0, @dec %limitdec
	%couter =w csltl 0, %limit
	jnz %couter, @inner, @done
@inner
	%j =l phi @outer 0, @next %jnext
	%cinner =w csltl %j, %limit
	jnz %cinner, @body, @dec
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%x =l loadl %p
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%y =l loadl %p1
	%cswap =w csltl %y, %x
	jnz %cswap, @swap, @next
@swap
	storel %y, %p
	storel %x, %p1
	jmp @next
@next
	%jnext =l add %j, 1
	jmp @inner
@dec
	%limitdec =l sub %limit, 1
	jmp @outer
@done
	ret 0
}
