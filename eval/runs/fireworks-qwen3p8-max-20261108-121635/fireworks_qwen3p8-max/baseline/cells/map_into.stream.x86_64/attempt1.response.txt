

map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rbp
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor rbx, rbx
    test r14, r14
    jz .Ldone
.Lloop:
    mov rdi, qword ptr [r13]
    mov rbp, rdi
    call r15
    mov qword ptr [r12], rax
    cmp rax, rbp
    jge .Lskip
    inc rbx
.Lskip:
    add r12, 8
    add r13, 8
    dec r14
    jnz .Lloop
.Ldone:
    mov rax, rbx
    add rsp, 8
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
