

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %written =l copy 0
    jmp @outer
@outer
    %cont =w csltl %i, %n
    jnz %cont, @run_begin, @done
@run_begin
    %src_i =l add %src, %i
    %b =w loadub %src_i
    %r =w copy 1
    %p =l add %i, 1
    jmp @run_loop
@run_loop
    %p_ok =w csltl %p, %n
    jnz %p_ok, @run_r_check, @run_done
@run_r_check
    %r_ok =w csltw %r, 128
    jnz %r_ok, @run_byte, @run_done
@run_byte
    %src_p =l add %src, %p
    %bp =w loadub %src_p
    %same =w ceqw %bp, %b
    jnz %same, @run_match, @run_done
@run_match
    %r =w add %r, 1
    %p =l add %p, 1
    jmp @run_loop
@run_done
    %is_lit =w csltw %r, 3
    jnz %is_lit, @lit_begin, @rep_check
@rep_check
    %need =l add %written, 2
    %fail =w csltl %cap, %need
    jnz %fail, @fail, @rep_write
@rep_write
    %c257 =w copy 257
    %hdr =w sub %c257, %r
    %outp =l add %out, %written
    storeb %hdr, %outp
    %outp1 =l add %outp, 1
    storeb %b, %outp1
    %written =l add %written, 2
    %i =l copy %p
    jmp @outer
@lit_begin
    %j =l copy %i
    %len_l =l copy 0
    %len_w =w copy 0
    jmp @lit_loop
@lit_loop
    %j_ok =w csltl %j, %n
    jnz %j_ok, @lit_len_check, @lit_done
@lit_len_check
    %len_ok =w csltl %len_l, 128
    jnz %len_ok, @lit_run_check, @lit_done
@lit_run_check
    %src_j =l add %src, %j
    %bj =w loadub %src_j
    %j1 =l add %j, 1
    %j1_ok =w csltl %j1, %n
    jnz %j1_ok, @lit_check2, @lit_accept
@lit_check2
    %src_j1 =l add %src, %j1
    %bj1 =w loadub %src_j1
    %same1 =w ceqw %bj1, %bj
    jnz %same1, @lit_check3, @lit_accept
@lit_check3
    %j2 =l add %j, 2
    %j2_ok =w csltl %j2, %n
    jnz %j2_ok, @lit_check_byte3, @lit_accept
@lit_check_byte3
    %src_j2 =l add %src, %j2
    %bj2 =w loadub %src_j2
    %same2 =w ceqw %bj2, %bj
    jnz %same2, @lit_stop, @lit_accept
@lit_stop
    jmp @lit_done
@lit_accept
    %j =l add %j, 1
    %len_l =l add %len_l, 1
    %len_w =w add %len_w, 1
    jmp @lit_loop
@lit_done
    %pkt_len =l add %len_l, 1
    %need =l add %written, %pkt_len
    %fail =w csltl %cap, %need
    jnz %fail, @fail, @lit_write
@lit_write
    %hdr =w sub %len_w, 1
    %outp =l add %out, %written
    storeb %hdr, %outp
    %dst =l add %outp, 1
    %srcp =l add %src, %i
    %remaining =l copy %len_l
    jmp @copy_loop
@copy_loop
    %rem_zero =w ceql %remaining, 0
    jnz %rem_zero, @copy_done, @copy_body
@copy_body
    %byte =w loadub %srcp
    storeb %byte, %dst
    %srcp =l add %srcp, 1
    %dst =l add %dst, 1
    %remaining =l sub %remaining, 1
    jmp @copy_loop
@copy_done
    %written =l add %written, %pkt_len
    %i =l copy %j
    jmp @outer
@fail
    %zero =l copy 0
    %one =l copy 1
    %neg =l sub %zero, %one
    ret %neg
@done
    ret %written
}
