

export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%eq =w ceql %dst, %src
	jnz %eq, @done, @dir
@dir
	%lt =w csltl %dst, %src
	jnz %lt, @fwd_loop, @bwd_loop
@fwd_loop
	%fi =l phi @dir 0, @fwd_body %finext
	%fc =w csltl %fi, %n
	jnz %fc, @fwd_body, @done
@fwd_body
	%foff =l mul %fi, 8
	%fda =l add %dst, %foff
	%fsa =l add %src, %foff
	%fv =l loadl %fsa
	storel %fv, %fda
	%finext =l add %fi, 1
	jmp @fwd_loop
@bwd_loop
	%bi =l phi @dir %n, @bwd_body %bim1
	%bc =w csltl 0, %bi
	jnz %bc, @bwd_body, @done
@bwd_body
	%bim1 =l sub %bi, 1
	%boff =l mul %bim1, 8
	%bda =l add %dst, %boff
	%bsa =l add %src, %boff
	%bv =l loadl %bsa
	storel %bv, %bda
	jmp @bwd_loop
@done
	ret 0
}
