

packbits_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    sub rsp, 608
    cld
    test r13, r13
    jle .Lcopy_done
    lea rdi, [rsp]
    mov rsi, r12
    mov rcx, r13
    rep movsb
.Lcopy_done:
    mov r12, rsp
    xor ebx, ebx
    xor r10d, r10d
.Lloop:
    cmp rbx, r13
    jge .Lreturn_written
    movzx eax, byte ptr [r12 + rbx]
    lea r8, [rbx + 1]
    lea r9, [rbx + 128]
    cmp r9, r13
    cmovg r9, r13
.Lrun_loop:
    cmp r8, r9
    jge .Lrun_done
    cmp byte ptr [r12 + r8], al
    jne .Lrun_done
    inc r8
    jmp .Lrun_loop
.Lrun_done:
    mov r11, r8
    sub r11, rbx
    cmp r11, 3
    jge .Lrepeat
    mov r8, rbx
.Llit_loop:
    cmp r8, r13
    jge .Llit_done
    mov r9, r8
    sub r9, rbx
    cmp r9, 128
    jge .Llit_done
    lea r9, [r8 + 2]
    cmp r9, r13
    jge .Llit_add
    movzx ecx, byte ptr [r12 + r8]
    cmp byte ptr [r12 + r8 + 1], cl
    jne .Llit_add
    cmp byte ptr [r12 + r8 + 2], cl
    je .Llit_done
.Llit_add:
    inc r8
    jmp .Llit_loop
.Llit_done:
    mov r9, r8
    sub r9, rbx
    test r9, r9
    jz .Lfail
    lea r11, [r9 + 1]
    cmp r15, r11
    jl .Lfail
    lea eax, [r9 - 1]
    mov byte ptr [r14 + r10], al
    lea rdi, [r14 + r10 + 1]
    lea rsi, [r12 + rbx]
    mov rcx, r9
    rep movsb
    sub r15, r11
    add r10, r11
    mov rbx, r8
    jmp .Lloop
.Lrepeat:
    cmp r15, 2
    jl .Lfail
    mov eax, 257
    sub eax, r11d
    mov byte ptr [r14 + r10], al
    movzx ecx, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10 + 1], cl
    sub r15, 2
    add r10, 2
    add rbx, r11
    jmp .Lloop
.Lfail:
    mov rax, -1
    add rsp, 608
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lreturn_written:
    mov rax, r10
    add rsp, 608
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
