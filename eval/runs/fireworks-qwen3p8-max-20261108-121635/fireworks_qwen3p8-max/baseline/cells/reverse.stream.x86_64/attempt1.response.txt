

reverse:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    lea rcx, [rdi + rsi*8 - 8]
.Lloop:
    cmp rdi, rcx
    jae .Ldone
    mov rdx, qword ptr [rdi]
    mov r8, qword ptr [rcx]
    mov qword ptr [rcx], rdx
    mov qword ptr [rdi], r8
    add rdi, 8
    sub rcx, 8
    jmp .Lloop
.Ldone:
    ret
