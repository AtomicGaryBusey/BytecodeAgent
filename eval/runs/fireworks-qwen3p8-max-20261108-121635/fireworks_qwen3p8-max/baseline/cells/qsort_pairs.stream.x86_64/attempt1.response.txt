

qsort_pairs:
    cmp rsi, 1
    jle .Lreturn_zero
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r15, rsi
    mov r12, qword ptr [rdi]
    mov r13, qword ptr [rdi + 8]
    mov rsi, -2
    mov rdi, r15
    shl rdi, 1
.Lpartition:
    add rsi, 2
    mov rax, qword ptr [rbx + rsi*8]
    mov rcx, qword ptr [rbx + rsi*8 + 8]
    cmp rax, r12
    jl .Lpartition
    jg .Li_stop
    cmp rcx, r13
    jl .Lpartition
.Li_stop:
.Ldec_j:
    sub rdi, 2
    mov rax, qword ptr [rbx + rdi*8]
    mov rcx, qword ptr [rbx + rdi*8 + 8]
    cmp r12, rax
    jl .Ldec_j
    jg .Lj_stop
    cmp r13, rcx
    jl .Ldec_j
.Lj_stop:
    cmp rsi, rdi
    jge .Lpartition_done
    mov rax, qword ptr [rbx + rsi*8]
    mov rcx, qword ptr [rbx + rsi*8 + 8]
    mov rdx, qword ptr [rbx + rdi*8]
    mov r8, qword ptr [rbx + rdi*8 + 8]
    mov qword ptr [rbx + rsi*8], rdx
    mov qword ptr [rbx + rsi*8 + 8], r8
    mov qword ptr [rbx + rdi*8], rax
    mov qword ptr [rbx + rdi*8 + 8], rcx
    jmp .Lpartition
.Lpartition_done:
    lea r14, [rdi + 2]
    shr r14, 1
    mov r12, r15
    sub r12, r14
    mov rdi, rbx
    mov rsi, r14
    call qsort_pairs
    lea rax, [r14 + r14]
    lea rdi, [rbx + rax*8]
    mov rsi, r12
    call qsort_pairs
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lreturn_zero:
    xor eax, eax
    ret
