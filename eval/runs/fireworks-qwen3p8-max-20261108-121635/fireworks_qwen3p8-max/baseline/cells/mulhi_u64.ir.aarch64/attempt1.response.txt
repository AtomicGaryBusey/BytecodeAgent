

export function l $mulhi_u64(l %a, l %b) {
@start
    %al =l and %a, 0xffffffff
    %ah =l shr %a, 32
    %bl =l and %b, 0xffffffff
    %bh =l shr %b, 32

    %p0 =l mul %al, %bl
    %p1 =l mul %ah, %bl
    %p2 =l mul %al, %bh
    %p3 =l mul %ah, %bh

    %p0hi =l shr %p0, 32
    %p1lo =l and %p1, 0xffffffff
    %p2lo =l and %p2, 0xffffffff
    %p1hi =l shr %p1, 32
    %p2hi =l shr %p2, 32

    %mid0 =l add %p0hi, %p1lo
    %mid =l add %mid0, %p2lo
    %carry =l shr %mid, 32

    %hi0 =l add %p3, %p1hi
    %hi1 =l add %hi0, %p2hi
    %hi =l add %hi1, %carry

    ret %hi
}
