

crc32:
    movz w6, #0xffff
    movk w6, #0xffff, lsl #16
    movz w5, #0x8320
    movk w5, #0xedb8, lsl #16
    orr w2, wzr, w6
    cbz x1, .Ldone
.Lbyte:
    ldrb w3, [x0], #1
    eor w2, w2, w3
    movz w4, #8
.Lbit:
    tst w2, #1
    b.eq .Lzero
    lsr w2, w2, #1
    eor w2, w2, w5
    b .Lnext
.Lzero:
    lsr w2, w2, #1
.Lnext:
    subs w4, w4, #1
    b.ne .Lbit
    subs x1, x1, #1
    b.ne .Lbyte
.Ldone:
    eor w0, w2, w6
    ret
