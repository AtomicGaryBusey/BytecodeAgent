

export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%nn =l mul %n, %n
	%n2 =l mul %nn, 2
	jmp @init_i_loop
@init_i_loop
	%ii =l phi @entry 0, @init_i_inc %ii_next
	%idone =w ceql %ii, %n
	jnz %idone, @after_init, @init_j_loop
@init_j_loop
	%ij =l phi @init_i_loop 0, @init_j_inc %ij_next
	%jdone =w ceql %ij, %n
	jnz %jdone, @init_i_inc, @init_j_body
@init_j_body
	%eq =w ceql %ii, %ij
	%i_n =l mul %ii, %n
	%idx =l add %i_n, %ij
	%off =l mul %idx, 8
	%addr =l add %out, %off
	jnz %eq, @init_diag, @init_zero
@init_diag
	storel 1, %addr
	jmp @init_j_inc
@init_zero
	storel 0, %addr
	jmp @init_j_inc
@init_j_inc
	%ij_next =l add %ij, 1
	jmp @init_j_loop
@init_i_inc
	%ii_next =l add %ii, 1
	jmp @init_i_loop
@after_init
	jmp @pow_loop
@pow_loop
	%e =l phi @after_init %k, @pow_cont %e_next
	%iszero =w ceql %e, 0
	jnz %iszero, @zero_loop, @mul_i_loop
@mul_i_loop
	%mi =l phi @pow_loop 0, @mul_i_inc %mi_next
	%mi_done =w ceql %mi, %n
	jnz %mi_done, @copy_loop, @mul_j_loop
@mul_j_loop
	%mj =l phi @mul_i_loop 0, @mul_j_inc %mj_next
	%mj_done =w ceql %mj, %n
	jnz %mj_done, @mul_i_inc, @mul_t_loop
@mul_t_loop
	%mt =l phi @mul_j_loop 0, @mul_t_body %mt_next
	%sum =l phi @mul_j_loop 0, @mul_t_body %sum_next
	%mt_done =w ceql %mt, %n
	jnz %mt_done, @mul_store, @mul_t_body
@mul_t_body
	%mi_n =l mul %mi, %n
	%a_idx =l add %mi_n, %mt
	%a_off =l mul %a_idx, 8
	%a_addr =l add %out, %a_off
	%av =l loadl %a_addr
	%mt_n =l mul %mt, %n
	%b_idx =l add %mt_n, %mj
	%b_off =l mul %b_idx, 8
	%b_addr =l add %a, %b_off
	%bv =l loadl %b_addr
	%prod =l mul %av, %bv
	%sum_next =l add %sum, %prod
	%mt_next =l add %mt, 1
	jmp @mul_t_loop
@mul_store
	%mi_n_s =l mul %mi, %n
	%c_idx =l add %mi_n_s, %mj
	%c_off =l mul %c_idx, 8
	%c_addr =l add %tmp, %c_off
	storel %sum, %c_addr
	jmp @mul_j_inc
@mul_j_inc
	%mj_next =l add %mj, 1
	jmp @mul_j_loop
@mul_i_inc
	%mi_next =l add %mi, 1
	jmp @mul_i_loop
@copy_loop
	%ci =l phi @mul_i_loop 0, @copy_body %ci_next
	%ci_done =w ceql %ci, %nn
	jnz %ci_done, @pow_cont, @copy_body
@copy_body
	%coff =l mul %ci, 8
	%caddr =l add %tmp, %coff
	%cv =l loadl %caddr
	%oaddr =l add %out, %coff
	storel %cv, %oaddr
	%ci_next =l add %ci, 1
	jmp @copy_loop
@pow_cont
	%e_next =l sub %e, 1
	jmp @pow_loop
@zero_loop
	%z =l phi @pow_loop 0, @zero_body %znext
	%z_done =w ceql %z, %n2
	jnz %z_done, @exit, @zero_body
@zero_body
	%zoff =l mul %z, 8
	%zaddr =l add %tmp, %zoff
	storel 0, %zaddr
	%znext =l add %z, 1
	jmp @zero_loop
@exit
	ret 0
}
