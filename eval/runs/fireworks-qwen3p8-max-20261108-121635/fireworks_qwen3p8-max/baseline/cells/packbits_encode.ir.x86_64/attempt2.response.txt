

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i0 =l copy 0
	%written0 =l copy 0
	%one_l =l copy 1
	%two_l =l copy 2
	%three_l =l copy 3
	%c128_l =l copy 128
	%one_w =w copy 1
	jmp @outer
@outer
	%i =l phi @start %i0, @rep_store %i_next_rep, @copy_done %i_next_lit
	%written =l phi @start %written0, @rep_store %w_next_rep, @copy_done %w_next_lit
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@done
	ret %written
@body
	%b_ptr =l add %src, %i
	%b =w loadub %b_ptr
	%ri_pos0 =l add %i, %one_l
	%ri_limit =l add %i, %c128_l
	%ri_cnt_l0 =l copy %one_l
	%ri_cnt_w0 =w copy %one_w
	jmp @ri_head
@ri_head
	%ri_pos =l phi @body %ri_pos0, @ri_inc %ri_pos1
	%ri_cnt_l =l phi @body %ri_cnt_l0, @ri_inc %ri_cnt_l1
	%ri_cnt_w =w phi @body %ri_cnt_w0, @ri_inc %ri_cnt_w1
	%ri_c1 =w csltl %ri_pos, %n
	jnz %ri_c1, @ri_lim, @ri_done
@ri_lim
	%ri_c2 =w csltl %ri_pos, %ri_limit
	jnz %ri_c2, @ri_byte, @ri_done
@ri_byte
	%ri_p =l add %src, %ri_pos
	%ri_x =w loadub %ri_p
	%ri_same =w ceqw %ri_x, %b
	jnz %ri_same, @ri_inc, @ri_done
@ri_inc
	%ri_pos1 =l add %ri_pos, %one_l
	%ri_cnt_l1 =l add %ri_cnt_l, %one_l
	%ri_cnt_w1 =w add %ri_cnt_w, %one_w
	jmp @ri_head
@ri_done
	%run_l =l phi @ri_head %ri_cnt_l, @ri_lim %ri_cnt_l, @ri_byte %ri_cnt_l
	%run_w =w phi @ri_head %ri_cnt_w, @ri_lim %ri_cnt_w, @ri_byte %ri_cnt_w
	%lt3 =w csltl %run_l, %three_l
	jnz %lt3, @lit_init, @rep_check
@rep_check
	%rep_need =l add %written, %two_l
	%rep_fail =w csltl %cap, %rep_need
	jnz %rep_fail, @error, @rep_store
@rep_store
	%rep_hdr =w sub %one_w, %run_w
	%rep_out =l add %out, %written
	storeb %rep_hdr, %rep_out
	%rep_out2 =l add %rep_out, %one_l
	storeb %b, %rep_out2
	%i_next_rep =l add %i, %run_l
	%w_next_rep =l add %written, %two_l
	jmp @outer
@lit_init
	%lit_start =l copy %i
	%j0 =l copy %i
	%len_l0 =l copy 0
	%len_w0 =w copy 0
	jmp @lit_head
@lit_head
	%j =l phi @lit_init %j0, @lit_inc %j1
	%len_l =l phi @lit_init %len_l0, @lit_inc %len_l1
	%len_w =w phi @lit_init %len_w0, @lit_inc %len_w1
	%lit_c1 =w csltl %j, %n
	jnz %lit_c1, @lit_len_check, @lit_done
@lit_len_check
	%lit_c2 =w csltl %len_l, %c128_l
	jnz %lit_c2, @lit_run_check, @lit_done
@lit_run_check
	%j_plus2 =l add %j, %two_l
	%lit_c3 =w csltl %j_plus2, %n
	jnz %lit_c3, @lit_run_load, @lit_inc
@lit_run_load
	%lit_p0 =l add %src, %j
	%lit_b0 =w loadub %lit_p0
	%lit_p1 =l add %lit_p0, %one_l
	%lit_b1 =w loadub %lit_p1
	%lit_same01 =w ceqw %lit_b0, %lit_b1
	jnz %lit_same01, @lit_run_third, @lit_inc
@lit_run_third
	%lit_p2 =l add %lit_p0, %two_l
	%lit_b2 =w loadub %lit_p2
	%lit_same02 =w ceqw %lit_b0, %lit_b2
	jnz %lit_same02, @lit_stop, @lit_inc
@lit_stop
	jmp @lit_done
@lit_inc
	%j1 =l add %j, %one_l
	%len_l1 =l add %len_l, %one_l
	%len_w1 =w add %len_w, %one_w
	jmp @lit_head
@lit_done
	%j_final =l phi @lit_head %j, @lit_len_check %j, @lit_stop %j
	%L_l =l phi @lit_head %len_l, @lit_len_check %len_l, @lit_stop %len_l
	%L_w =w phi @lit_head %len_w, @lit_len_check %len_w, @lit_stop %len_w
	%lit_packet_len =l add %L_l, %one_l
	%lit_need =l add %written, %lit_packet_len
	%lit_fail =w csltl %cap, %lit_need
	jnz %lit_fail, @error, @lit_store_hdr
@lit_store_hdr
	%lit_hdr =w sub %L_w, %one_w
	%lit_out =l add %out, %written
	storeb %lit_hdr, %lit_out
	%copy_dst_base =l add %written, %one_l
	%k0 =l copy 0
	jmp @copy_head
@copy_head
	%k =l phi @lit_store_hdr %k0, @copy_body %k1
	%copy_cont =w csltl %k, %L_l
	jnz %copy_cont, @copy_body, @copy_done
@copy_body
	%src_off =l add %lit_start, %k
	%src_addr =l add %src, %src_off
	%byte =w loadub %src_addr
	%dst_off =l add %copy_dst_base, %k
	%dst_addr =l add %out, %dst_off
	storeb %byte, %dst_addr
	%k1 =l add %k, %one_l
	jmp @copy_head
@copy_done
	%i_next_lit =l copy %j_final
	%w_next_lit =l add %written, %lit_packet_len
	jmp @outer
@error
	%neg1 =l copy -1
	ret %neg1
}
