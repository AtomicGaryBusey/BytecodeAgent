

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i0 =l copy 0
	%w0 =l copy 0
	%one =l copy 1
	%two =l copy 2
	%three =l copy 3
	%c128 =l copy 128
	%c257 =l copy 257
	jmp @outer_top
@outer_top
	%i =l phi @start %i0, @rep_write %i_next, @copy_done %i_next2
	%written =l phi @start %w0, @rep_write %w_next, @copy_done %w_next2
	%not_done =w csltl %i, %n
	jnz %not_done, @outer_body, @done
@outer_body
	%ri_ptr0 =l add %src, %i
	%ri_b =w loadub %ri_ptr0
	%ri_end0 =l add %i, %one
	%ri_lim =l add %i, %c128
	jmp @run_i_loop
@run_i_loop
	%ri_end =l phi @outer_body %ri_end0, @run_i_inc %ri_end1
	%ri_c1 =w csltl %ri_end, %n
	jnz %ri_c1, @run_i_lim, @run_i_done
@run_i_lim
	%ri_c2 =w csltl %ri_end, %ri_lim
	jnz %ri_c2, @run_i_byte, @run_i_done
@run_i_byte
	%ri_ptr =l add %src, %ri_end
	%ri_x =w loadub %ri_ptr
	%ri_same =w ceqw %ri_x, %ri_b
	jnz %ri_same, @run_i_inc, @run_i_done
@run_i_inc
	%ri_end1 =l add %ri_end, %one
	jmp @run_i_loop
@run_i_done
	%ri_endf =l phi @run_i_loop %ri_end, @run_i_lim %ri_end, @run_i_byte %ri_end
	%run_i =l sub %ri_endf, %i
	%run_i_lt3 =w csltl %run_i, %three
	jnz %run_i_lt3, @literal_start, @repeat
@repeat
	%rep_need =l add %written, %two
	%rep_fail =w csltl %cap, %rep_need
	jnz %rep_fail, @error, @rep_write
@rep_write
	%rep_hdrl =l sub %c257, %run_i
	%rep_hdr =w cast %rep_hdrl
	%rep_optr =l add %out, %written
	storeb %rep_hdr, %rep_optr
	%rep_bptr =l add %src, %i
	%rep_b =w loadub %rep_bptr
	%rep_optr2 =l add %rep_optr, %one
	storeb %rep_b, %rep_optr2
	%i_next =l add %i, %run_i
	%w_next =l add %written, %two
	jmp @outer_top
@literal_start
	%lit_start_pos =l copy %i
	%j0 =l copy %i
	jmp @lit_top
@lit_top
	%j =l phi @literal_start %j0, @lit_inc %j1
	%lit_c1 =w csltl %j, %n
	jnz %lit_c1, @lit_len, @lit_done
@lit_len
	%lit_len_sofar =l sub %j, %lit_start_pos
	%lit_c2 =w csltl %lit_len_sofar, %c128
	jnz %lit_c2, @runj_start, @lit_done
@runj_start
	%rj_ptr0 =l add %src, %j
	%rj_b =w loadub %rj_ptr0
	%rj_end0 =l add %j, %one
	%rj_lim =l add %j, %c128
	jmp @runj_loop
@runj_loop
	%rj_end =l phi @runj_start %rj_end0, @runj_inc %rj_end1
	%rj_c1 =w csltl %rj_end, %n
	jnz %rj_c1, @runj_lim, @runj_done
@runj_lim
	%rj_c2 =w csltl %rj_end, %rj_lim
	jnz %rj_c2, @runj_byte, @runj_done
@runj_byte
	%rj_ptr =l add %src, %rj_end
	%rj_x =w loadub %rj_ptr
	%rj_same =w ceqw %rj_x, %rj_b
	jnz %rj_same, @runj_inc, @runj_done
@runj_inc
	%rj_end1 =l add %rj_end, %one
	jmp @runj_loop
@runj_done
	%rj_endf =l phi @runj_loop %rj_end, @runj_lim %rj_end, @runj_byte %rj_end
	%run_j =l sub %rj_endf, %j
	%run_j_lt3 =w csltl %run_j, %three
	jnz %run_j_lt3, @lit_inc, @lit_done
@lit_inc
	%j1 =l add %j, %one
	jmp @lit_top
@lit_done
	%j_final =l phi @lit_top %j, @lit_len %j, @runj_done %j
	%L =l sub %j_final, %lit_start_pos
	%lit_packet_len =l add %L, %one
	%lit_need =l add %written, %lit_packet_len
	%lit_fail =w csltl %cap, %lit_need
	jnz %lit_fail, @error, @lit_write_hdr
@lit_write_hdr
	%lit_hdrl =l sub %L, %one
	%lit_hdr =w cast %lit_hdrl
	%lit_optr =l add %out, %written
	storeb %lit_hdr, %lit_optr
	%copy_dst_base =l add %written, %one
	%k0 =l copy 0
	jmp @copy_top
@copy_top
	%k =l phi @lit_write_hdr %k0, @copy_body %k1
	%copy_c =w csltl %k, %L
	jnz %copy_c, @copy_body, @copy_done
@copy_body
	%src_ptr =l add %src, %lit_start_pos
	%src_ptr2 =l add %src_ptr, %k
	%byte =w loadub %src_ptr2
	%dst_ptr =l add %out, %copy_dst_base
	%dst_ptr2 =l add %dst_ptr, %k
	storeb %byte, %dst_ptr2
	%k1 =l add %k, %one
	jmp @copy_top
@copy_done
	%w_next2 =l add %written, %lit_packet_len
	%i_next2 =l copy %j_final
	jmp @outer_top
@done
	ret %written
@error
	%err =l copy -1
	ret %err
}
