

lcs_len:
    push rbx
    mov rbx, rcx
    xor eax, eax
    lea rcx, [rbx + 1]
    mov r9, r8
.Linit:
    mov qword ptr [r9], rax
    add r9, 8
    dec rcx
    jnz .Linit
    test rsi, rsi
    jz .Lreturn
    test rbx, rbx
    jz .Lreturn
.Louter:
    movzx r9d, byte ptr [rdi]
    add rdi, 1
    xor r10d, r10d
    mov rcx, 1
.Linner:
    mov r11, qword ptr [r8 + rcx*8]
    cmp r9b, byte ptr [rdx + rcx - 1]
    je .Lmatch
    mov rax, qword ptr [r8 + rcx*8 - 8]
    cmp r11, rax
    cmovge rax, r11
    jmp .Lstore
.Lmatch:
    lea rax, [r10 + 1]
.Lstore:
    mov qword ptr [r8 + rcx*8], rax
    mov r10, r11
    inc rcx
    cmp rcx, rbx
    jle .Linner
    dec rsi
    jnz .Louter
.Lreturn:
    mov rax, qword ptr [r8 + rbx*8]
    pop rbx
    ret
