

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l sub %row, %row
	%n1 =l add %n, 1
	%yend =l add %y, %n
	%xend =l add %x, %m
	jmp @init_check %row %zero
@init_check %rp %cnt
	%c_init =w csltl %cnt, %n1
	jnz %c_init, @init_store, @init_done
@init_store
	storel 0, %rp
	%rp_next =l add %rp, 8
	%cnt_next =l add %cnt, 1
	jmp @init_check %rp_next %cnt_next
@init_done
	%row_n =l sub %rp, 8
	jmp @outer_check %x
@outer_check %xp
	%c_outer =w csltl %xp, %xend
	jnz %c_outer, @outer_body, @return
@outer_body
	%xb =w loadub %xp
	%pj_start =l add %row, 8
	jmp @inner_check %zero %y %pj_start %row
@inner_check %diag %yp %pj %pjm1
	%c_inner =w csltl %yp, %yend
	jnz %c_inner, @inner_body, @outer_next
@inner_body
	%yb =w loadub %yp
	%cur =l loadl %pj
	%left =l loadl %pjm1
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch
@match
	%new_match =l add %diag, 1
	jmp @inner_store %new_match
@mismatch
	%left_greater =w csltl %cur, %left
	jnz %left_greater, @use_left, @use_cur
@use_left
	jmp @inner_store %left
@use_cur
	jmp @inner_store %cur
@inner_store %new
	storel %new, %pj
	%yp_next =l add %yp, 1
	%pj_next =l add %pj, 8
	jmp @inner_check %cur %yp_next %pj_next %pj
@outer_next
	%xp_next =l add %xp, 1
	jmp @outer_check %xp_next
@return
	%res =l loadl %row_n
	ret %res
}
