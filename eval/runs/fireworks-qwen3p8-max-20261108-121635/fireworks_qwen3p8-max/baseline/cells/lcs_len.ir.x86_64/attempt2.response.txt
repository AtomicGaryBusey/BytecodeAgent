

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l sub %row, %row
	%one =l add %zero, 1
	%eight =l add %zero, 8
	%n1 =l add %n, %one
	%xend =l add %x, %m
	%yend =l add %y, %n
	%n8 =l mul %n, %eight
	%rown =l add %row, %n8
	jmp @init_loop
@init_loop
	%ip =l phi @start %row, @init_store %ip1
	%ic =l phi @start %zero, @init_store %ic1
	%cinit =w csltl %ic, %n1
	jnz %cinit, @init_store, @outer_loop
@init_store
	storel %zero, %ip
	%ip1 =l add %ip, %eight
	%ic1 =l add %ic, %one
	jmp @init_loop
@outer_loop
	%xp =l phi @init_loop %x, @outer_next %xp1
	%cout =w csltl %xp, %xend
	jnz %cout, @outer_body, @done
@outer_body
	%xb =w loadub %xp
	%pj0 =l add %row, %eight
	jmp @inner_loop
@inner_loop
	%diag =l phi @outer_body %zero, @inner_store %diag_next
	%yp =l phi @outer_body %y, @inner_store %yp1
	%pj =l phi @outer_body %pj0, @inner_store %pj1
	%pjm1 =l phi @outer_body %row, @inner_store %pjm1_next
	%cin =w csltl %yp, %yend
	jnz %cin, @inner_body, @outer_next
@inner_body
	%yb =w loadub %yp
	%cur =l loadl %pj
	%left =l loadl %pjm1
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch
@match
	%newm =l add %diag, %one
	jmp @inner_store
@mismatch
	%lt =w csltl %cur, %left
	jnz %lt, @use_left, @use_cur
@use_left
	%newl =l add %left, %zero
	jmp @inner_store
@use_cur
	%newc =l add %cur, %zero
	jmp @inner_store
@inner_store
	%new =l phi @match %newm, @use_left %newl, @use_cur %newc
	storel %new, %pj
	%diag_next =l add %cur, %zero
	%pjm1_next =l add %pj, %zero
	%yp1 =l add %yp, %one
	%pj1 =l add %pj, %eight
	jmp @inner_loop
@outer_next
	%xp1 =l add %xp, %one
	jmp @outer_loop
@done
	%res =l loadl %rown
	ret %res
}
