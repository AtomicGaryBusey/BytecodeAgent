

export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    %pc_addr =l alloc8 8
    %d_addr =l alloc8 8
    %zero =l copy 0
    storel %zero, %pc_addr
    storel %zero, %d_addr
    jmp @loop
@loop
    %pc =l loadl %pc_addr
    %d =l loadl %d_addr
    %done =w ceql %pc, %len
    jnz %done, @finish, @fetch
@fetch
    %pc_off =l mul %pc, 8
    %op_addr =l add %code, %pc_off
    %op =l loadl %op_addr
    %is0 =w ceql %op, 0
    jnz %is0, @finish, @chk1
@chk1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @chk2
@chk2
    %is2 =w ceql %op, 2
    jnz %is2, @add_op, @chk3
@chk3
    %is3 =w ceql %op, 3
    jnz %is3, @sub_op, @chk4
@chk4
    %is4 =w ceql %op, 4
    jnz %is4, @mul_op, @chk5
@chk5
    %is5 =w ceql %op, 5
    jnz %is5, @dup_op, @chk6
@chk6
    %is6 =w ceql %op, 6
    jnz %is6, @swap_op, @chk7
@chk7
    %is7 =w ceql %op, 7
    jnz %is7, @drop_op, @chk8
@chk8
    %is8 =w ceql %op, 8
    jnz %is8, @over_op, @chk9
@chk9
    %is9 =w ceql %op, 9
    jnz %is9, @lt_op, @chk10
@chk10
    %is10 =w ceql %op, 10
    jnz %is10, @jz_op, @chk11
@chk11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp_op, @unknown
@push
    %p_op_idx =l add %pc, 1
    %p_op_off =l mul %p_op_idx, 8
    %p_op_addr =l add %code, %p_op_off
    %p_val =l loadl %p_op_addr
    %p_slot_off =l mul %d, 8
    %p_slot_addr =l add %stack, %p_slot_off
    storel %p_val, %p_slot_addr
    %p_next_pc =l add %pc, 2
    %p_next_d =l add %d, 1
    storel %p_next_pc, %pc_addr
    storel %p_next_d, %d_addr
    jmp @loop
@add_op
    %add_b_idx =l sub %d, 1
    %add_a_idx =l sub %d, 2
    %add_b_off =l mul %add_b_idx, 8
    %add_a_off =l mul %add_a_idx, 8
    %add_b_addr =l add %stack, %add_b_off
    %add_a_addr =l add %stack, %add_a_off
    %add_b =l loadl %add_b_addr
    %add_a =l loadl %add_a_addr
    %add_res =l add %add_a, %add_b
    storel %add_res, %add_a_addr
    %add_next_pc =l add %pc, 1
    %add_next_d =l sub %d, 1
    storel %add_next_pc, %pc_addr
    storel %add_next_d, %d_addr
    jmp @loop
@sub_op
    %sub_b_idx =l sub %d, 1
    %sub_a_idx =l sub %d, 2
    %sub_b_off =l mul %sub_b_idx, 8
    %sub_a_off =l mul %sub_a_idx, 8
    %sub_b_addr =l add %stack, %sub_b_off
    %sub_a_addr =l add %stack, %sub_a_off
    %sub_b =l loadl %sub_b_addr
    %sub_a =l loadl %sub_a_addr
    %sub_res =l sub %sub_a, %sub_b
    storel %sub_res, %sub_a_addr
    %sub_next_pc =l add %pc, 1
    %sub_next_d =l sub %d, 1
    storel %sub_next_pc, %pc_addr
    storel %sub_next_d, %d_addr
    jmp @loop
@mul_op
    %mul_b_idx =l sub %d, 1
    %mul_a_idx =l sub %d, 2
    %mul_b_off =l mul %mul_b_idx, 8
    %mul_a_off =l mul %mul_a_idx, 8
    %mul_b_addr =l add %stack, %mul_b_off
    %mul_a_addr =l add %stack, %mul_a_off
    %mul_b =l loadl %mul_b_addr
    %mul_a =l loadl %mul_a_addr
    %mul_res =l mul %mul_a, %mul_b
    storel %mul_res, %mul_a_addr
    %mul_next_pc =l add %pc, 1
    %mul_next_d =l sub %d, 1
    storel %mul_next_pc, %pc_addr
    storel %mul_next_d, %d_addr
    jmp @loop
@dup_op
    %dup_idx =l sub %d, 1
    %dup_off =l mul %dup_idx, 8
    %dup_addr =l add %stack, %dup_off
    %dup_val =l loadl %dup_addr
    %dup_new_off =l mul %d, 8
    %dup_new_addr =l add %stack, %dup_new_off
    storel %dup_val, %dup_new_addr
    %dup_next_pc =l add %pc, 1
    %dup_next_d =l add %d, 1
    storel %dup_next_pc, %pc_addr
    storel %dup_next_d, %d_addr
    jmp @loop
@swap_op
    %swap_b_idx =l sub %d, 1
    %swap_a_idx =l sub %d, 2
    %swap_b_off =l mul %swap_b_idx, 8
    %swap_a_off =l mul %swap_a_idx, 8
    %swap_b_addr =l add %stack, %swap_b_off
    %swap_a_addr =l add %stack, %swap_a_off
    %swap_b =l loadl %swap_b_addr
    %swap_a =l loadl %swap_a_addr
    storel %swap_a, %swap_b_addr
    storel %swap_b, %swap_a_addr
    %swap_next_pc =l add %pc, 1
    storel %swap_next_pc, %pc_addr
    jmp @loop
@drop_op
    %drop_next_pc =l add %pc, 1
    %drop_next_d =l sub %d, 1
    storel %drop_next_pc, %pc_addr
    storel %drop_next_d, %d_addr
    jmp @loop
@over_op
    %over_idx =l sub %d, 2
    %over_off =l mul %over_idx, 8
    %over_addr =l add %stack, %over_off
    %over_val =l loadl %over_addr
    %over_new_off =l mul %d, 8
    %over_new_addr =l add %stack, %over_new_off
    storel %over_val, %over_new_addr
    %over_next_pc =l add %pc, 1
    %over_next_d =l add %d, 1
    storel %over_next_pc, %pc_addr
    storel %over_next_d, %d_addr
    jmp @loop
@lt_op
    %lt_b_idx =l sub %d, 1
    %lt_a_idx =l sub %d, 2
    %lt_b_off =l mul %lt_b_idx, 8
    %lt_a_off =l mul %lt_a_idx, 8
    %lt_b_addr =l add %stack, %lt_b_off
    %lt_a_addr =l add %stack, %lt_a_off
    %lt_b =l loadl %lt_b_addr
    %lt_a =l loadl %lt_a_addr
    %lt_cmp =w csltl %lt_a, %lt_b
    %lt_next_pc =l add %pc, 1
    %lt_next_d =l sub %d, 1
    storel %lt_next_pc, %pc_addr
    storel %lt_next_d, %d_addr
    jnz %lt_cmp, @lt_true, @lt_false
@lt_true
    %lt_one =l copy 1
    storel %lt_one, %lt_a_addr
    jmp @loop
@lt_false
    %lt_zero =l copy 0
    storel %lt_zero, %lt_a_addr
    jmp @loop
@jz_op
    %jz_target_idx =l add %pc, 1
    %jz_target_off =l mul %jz_target_idx, 8
    %jz_target_addr =l add %code, %jz_target_off
    %jz_target =l loadl %jz_target_addr
    %jz_v_idx =l sub %d, 1
    %jz_v_off =l mul %jz_v_idx, 8
    %jz_v_addr =l add %stack, %jz_v_off
    %jz_v =l loadl %jz_v_addr
    %jz_next_d =l sub %d, 1
    storel %jz_next_d, %d_addr
    %jz_cmp =w ceql %jz_v, 0
    jnz %jz_cmp, @jz_take, @jz_skip
@jz_take
    storel %jz_target, %pc_addr
    jmp @loop
@jz_skip
    %jz_skip_pc =l add %pc, 2
    storel %jz_skip_pc, %pc_addr
    jmp @loop
@jmp_op
    %jmp_target_idx =l add %pc, 1
    %jmp_target_off =l mul %jmp_target_idx, 8
    %jmp_target_addr =l add %code, %jmp_target_off
    %jmp_target =l loadl %jmp_target_addr
    storel %jmp_target, %pc_addr
    jmp @loop
@unknown
    %unknown_next_pc =l add %pc, 1
    storel %unknown_next_pc, %pc_addr
    jmp @loop
@finish
    %final_d =l loadl %d_addr
    storel %final_d, %pc_addr
    jmp @zero
@zero
    %zero_i =l loadl %pc_addr
    %zero_more =w csltl %zero_i, %cap
    jnz %zero_more, @zero_body, @after_zero
@zero_body
    %zero_off =l mul %zero_i, 8
    %zero_addr =l add %stack, %zero_off
    %zero_val =l copy 0
    storel %zero_val, %zero_addr
    %zero_next_i =l add %zero_i, 1
    storel %zero_next_i, %pc_addr
    jmp @zero
@after_zero
    %ret_d =l loadl %d_addr
    %ret_empty =w ceql %ret_d, 0
    jnz %ret_empty, @ret_zero, @ret_top
@ret_zero
    %ret_zero_val =l copy 0
    ret %ret_zero_val
@ret_top
    %top_idx =l sub %ret_d, 1
    %top_off =l mul %top_idx, 8
    %top_addr =l add %stack, %top_off
    %top_val =l loadl %top_addr
    ret %top_val
}
