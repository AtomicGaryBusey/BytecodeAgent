

export function l $count_desc(l %a, l %n) {
@start
    %limit =l sub %n, 1
    %small =w csltl %n, 2
    jnz %small, @zero, @loop
@zero
    ret 0
@loop
    %i =l phi @start 0, @cont %inext
    %cnt =l phi @start 0, @cont %cntnext
    %test =w csltl %i, %limit
    jnz %test, @body, @done
@body
    %ip1 =l add %i, 1
    %off =l mul %i, 8
    %p =l add %a, %off
    %x =l loadl %p
    %off1 =l mul %ip1, 8
    %p1 =l add %a, %off1
    %y =l loadl %p1
    %desc =w csltl %y, %x
    %cntbody =l add %cnt, 0
    jnz %desc, @inc, @cont
@inc
    %cntinc =l add %cnt, 1
    jmp @cont
@cont
    %cntnext =l phi @body %cntbody, @inc %cntinc
    %inext =l add %i, 1
    jmp @loop
@done
    ret %cnt
}
