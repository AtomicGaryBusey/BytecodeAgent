

stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    mov r12, rdi
    mov rbx, rsi
    mov r13, rdx
    mov r14, rcx
    mov rdi, qword ptr [r12]
    cld
    call r13
    mov rbp, rax
    mov qword ptr [rsp], rax
    mov qword ptr [rsp + 8], rax
    mov r15, rax
    and r15, 1
    lea r12, [r12 + 8]
    dec rbx
    jz .Lstore
.Lloop:
    mov rdi, qword ptr [r12]
    cld
    call r13
    add rbp, rax
    mov rcx, rax
    and rcx, 1
    add r15, rcx
    cmp rax, qword ptr [rsp]
    jl .Lmin
.Lnotmin:
    cmp rax, qword ptr [rsp + 8]
    jg .Lmax
.Lnotmax:
    lea r12, [r12 + 8]
    dec rbx
    jnz .Lloop
    jmp .Lstore
.Lmin:
    mov qword ptr [rsp], rax
    jmp .Lnotmin
.Lmax:
    mov qword ptr [rsp + 8], rax
    jmp .Lnotmax
.Lstore:
    mov rax, qword ptr [rsp]
    mov qword ptr [r14], rax
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [r14 + 8], rax
    mov qword ptr [r14 + 16], rbp
    mov rax, r15
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    cld
    ret
