

export function l $reverse(l %a, l %n) {
@start
    %done =w csltl %n, 2
    jnz %done, @end, @pre
@pre
    %i0 =l sub %n, %n
    %j0 =l sub %n, 1
    jmp @cond
@cond
    %i =l phi @pre %i0, @loop %inext
    %j =l phi @pre %j0, @loop %jnext
    %cont =w csltl %i, %j
    jnz %cont, @loop, @end
@loop
    %io =l mul %i, 8
    %pa =l add %a, %io
    %va =l loadl %pa
    %jo =l mul %j, 8
    %pb =l add %a, %jo
    %vb =l loadl %pb
    storel %vb, %pa
    storel %va, %pb
    %inext =l add %i, 1
    %jnext =l sub %j, 1
    jmp @cond
@end
    ret 0
}
