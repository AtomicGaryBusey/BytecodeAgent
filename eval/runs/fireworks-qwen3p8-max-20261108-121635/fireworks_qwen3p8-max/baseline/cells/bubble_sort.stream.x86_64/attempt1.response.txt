

bubble_sort:
    cmp rsi, 1
    jle .Lreturn
    lea r8, [rsi - 1]
.Louter:
    test r8, r8
    jz .Lreturn
    xor r9d, r9d
.Linner:
    cmp r9, r8
    jge .Lendinner
    mov rax, qword ptr [rdi + r9*8]
    mov rcx, qword ptr [rdi + r9*8 + 8]
    cmp rax, rcx
    jle .Lnoswap
    mov qword ptr [rdi + r9*8], rcx
    mov qword ptr [rdi + r9*8 + 8], rax
.Lnoswap:
    inc r9
    jmp .Linner
.Lendinner:
    dec r8
    jmp .Louter
.Lreturn:
    xor eax, eax
    ret
