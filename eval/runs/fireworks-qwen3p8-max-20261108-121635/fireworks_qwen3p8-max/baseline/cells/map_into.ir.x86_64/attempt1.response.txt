

export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
%i0 =l copy 0
%c0 =l copy 0
jmp @loop
@loop
%i =l phi @start %i0, @next %i1
%p =l phi @start %src, @next %p1
%q =l phi @start %dst, @next %q1
%c =l phi @start %c0, @next %c1
%cont =w csltl %i, %n
jnz %cont, @body, @end
@body
%x =l loadl %p
%y =l call %f(l %x)
storel %y, %q
%lt =w csltl %y, %x
%ckeep =l copy %c
jnz %lt, @inc, @next
@inc
%cinc =l add %c, 1
jmp @next
@next
%c1 =l phi @inc %cinc, @body %ckeep
%i1 =l add %i, 1
%p1 =l add %p, 8
%q1 =l add %q, 8
jmp @loop
@end
ret %c
}
