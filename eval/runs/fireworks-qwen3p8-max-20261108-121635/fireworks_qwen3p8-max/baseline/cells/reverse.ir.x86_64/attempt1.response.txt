

export function l $reverse(l %a, l %n) {
@start
    %j0 =l sub %n, 1
    jmp @cond
@cond
    %i =l phi @start 0, @body %inext
    %j =l phi @start %j0, @body %jnext
    %cont =w csltl %i, %j
    jnz %cont, @body, @done
@body
    %io =l mul %i, 8
    %ip =l add %a, %io
    %jo =l mul %j, 8
    %jp =l add %a, %jo
    %x =l loadl %ip
    %y =l loadl %jp
    storel %y, %ip
    storel %x, %jp
    %inext =l add %i, 1
    %jnext =l sub %j, 1
    jmp @cond
@done
    ret l 0
}
