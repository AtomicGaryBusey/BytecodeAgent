

export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l copy 0
	%nm1 =l sub %n, 1
	jmp @outer_header
@outer_header
	%i =l phi @start %i0, @outer_inc %i_next
	%outer_cond =w csltl %i, %nm1
	jnz %outer_cond, @outer_body, @done
@outer_body
	%tlimit =l sub %n, %i
	%limit =l sub %tlimit, 1
	%j0 =l copy 0
	jmp @inner_header
@inner_header
	%j =l phi @outer_body %j0, @inner_inc %j_next
	%inner_cond =w csltl %j, %limit
	jnz %inner_cond, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%p =l add %a, %off
	%p1 =l add %p, 8
	%cur =l loadl %p
	%next =l loadl %p1
	%swap_cond =w csltl %next, %cur
	jnz %swap_cond, @do_swap, @no_swap
@do_swap
	storel %next, %p
	storel %cur, %p1
	jmp @inner_inc
@no_swap
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner_header
@outer_inc
	%i_next =l add %i, 1
	jmp @outer_header
@done
	%result =l copy 0
	ret %result
}
