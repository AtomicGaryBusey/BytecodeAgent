

export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    %rem0 =l sub %n, 1
    %p0 =l add %a, 8
    jmp @loop
@loop
    %p =l phi @start %p0, @maxmerge %p1
    %rem =l phi @start %rem0, @maxmerge %rem1
    %min =l phi @start %y0, @maxmerge %min2
    %max =l phi @start %y0, @maxmerge %max1
    %sum =l phi @start %y0, @maxmerge %sum1
    %cnt =l phi @start %odd0, @maxmerge %cnt1
    %done =w ceql %rem, 0
    jnz %done, @finish, @body
@body
    %x =l loadl %p
    %y =l call %f(l %x)
    %ismin =w csltl %y, %min
    jnz %ismin, @minyes, @minno
@minyes
    %ymin =l add %y, 0
    jmp @minmerge
@minno
    %minold =l add %min, 0
    jmp @minmerge
@minmerge
    %min1 =l phi @minyes %ymin, @minno %minold
    %ismax =w csltl %max, %y
    jnz %ismax, @maxyes, @maxno
@maxyes
    %ymax =l add %y, 0
    jmp @maxmerge
@maxno
    %maxold =l add %max, 0
    jmp @maxmerge
@maxmerge
    %max1 =l phi @maxyes %ymax, @maxno %maxold
    %min2 =l add %min1, 0
    %sum1 =l add %sum, %y
    %odd =l and %y, 1
    %cnt1 =l add %cnt, %odd
    %p1 =l add %p, 8
    %rem1 =l sub %rem, 1
    jmp @loop
@finish
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %cnt
}
