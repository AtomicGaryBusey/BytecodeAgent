

export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%v0 =l loadl %a
	%y0 =l call %f(l %v0)
	%odd0 =l and %y0, 1
	%rem0 =l sub %n, 1
	%p0 =l add %a, 8
	jmp @loop(%p0, %rem0, %y0, %y0, %y0, %odd0)
@loop(%p l, %rem l, %min l, %max l, %sum l, %cnt l)
	%done =w ceql %rem, 0
	jnz %done, @finish, @body
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	jnz %ismin, @min_new, @min_old
@min_new
	jmp @min_done(%y)
@min_old
	jmp @min_done(%min)
@min_done(%min1 l)
	%ismax =w csltl %max, %y
	jnz %ismax, @max_new, @max_old
@max_new
	jmp @max_done(%y)
@max_old
	jmp @max_done(%max)
@max_done(%max1 l)
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%cnt1 =l add %cnt, %odd
	%p1 =l add %p, 8
	%rem1 =l sub %rem, 1
	jmp @loop(%p1, %rem1, %min1, %max1, %sum1, %cnt1)
@finish
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
