

export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%rem0 =l sub %n, 1
	%p0 =l add %a, 8
	jmp @loop
@loop
	%p =l phi @start, %p0, @max_merge, %p1
	%rem =l phi @start, %rem0, @max_merge, %rem1
	%min =l phi @start, %y0, @max_merge, %min1
	%max =l phi @start, %y0, @max_merge, %max1
	%sum =l phi @start, %y0, @max_merge, %sum1
	%cnt =l phi @start, %odd0, @max_merge, %cnt1
	%done =w ceql %rem, 0
	jnz %done, @finish, @body
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%is_min =w csltl %y, %min
	jnz %is_min, @min_new, @min_old
@min_new
	jmp @min_merge
@min_old
	jmp @min_merge
@min_merge
	%min1 =l phi @min_new, %y, @min_old, %min
	%is_max =w csltl %max, %y
	jnz %is_max, @max_new, @max_old
@max_new
	jmp @max_merge
@max_old
	jmp @max_merge
@max_merge
	%max1 =l phi @max_new, %y, @max_old, %max
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%cnt1 =l add %cnt, %odd
	%p1 =l add %p, 8
	%rem1 =l sub %rem, 1
	jmp @loop
@finish
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
