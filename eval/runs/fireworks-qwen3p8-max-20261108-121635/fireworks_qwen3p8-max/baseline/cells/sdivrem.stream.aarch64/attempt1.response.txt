

sdivrem:
    cbz x1, .Ldiv_zero
    movz x9, #0x8000, lsl #48
    cmp x0, x9
    b.ne .Lnormal
    movn x10, #0
    cmp x1, x10
    b.ne .Lnormal
    str xzr, [x2]
    mov x0, x9
    ret
.Ldiv_zero:
    str x0, [x2]
    movz x0, #0
    ret
.Lnormal:
    sdiv x3, x0, x1
    msub x4, x3, x1, x0
    str x4, [x2]
    mov x0, x3
    ret
