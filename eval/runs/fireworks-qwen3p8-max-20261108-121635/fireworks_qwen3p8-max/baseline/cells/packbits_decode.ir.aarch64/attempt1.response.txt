

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %posp =l alloc8 8
    %wrp =l alloc8 8
    %zero_l =l copy 0
    %zero_w =w copy 0
    %one_w =w copy 1
    %m128_w =w copy -128
    storel %zero_l, %posp
    storel %zero_l, %wrp
    jmp @main
@main
    %m_pos =l loadl %posp
    %m_cont =w csltl %m_pos, %n
    jnz %m_cont, @header, @done
@header
    %h_pos =l loadl %posp
    %h_addr =l add %src, %h_pos
    %h_val =w loadsb %h_addr
    %h_pos1 =l add %h_pos, 1
    storel %h_pos1, %posp
    %h_neg =w csltw %h_val, %zero_w
    jnz %h_neg, @negative, @literal
@negative
    %ng_noop =w ceqw %h_val, %m128_w
    jnz %ng_noop, @main, @repeat
@repeat
    %rp_pos =l loadl %posp
    %rp_need =l add %rp_pos, 1
    %rp_bad =w csltl %n, %rp_need
    jnz %rp_bad, @err2, @rep_src_ok
@rep_src_ok
    %ro_countw =w sub %one_w, %h_val
    %ro_count =l extsw %ro_countw
    %ro_wr =l loadl %wrp
    %ro_capend =l add %ro_wr, %ro_count
    %ro_ex =w csltl %cap, %ro_capend
    jnz %ro_ex, @rep_exceed, @rep_full
@rep_exceed
    %rx_pos =l loadl %posp
    %rx_addr =l add %src, %rx_pos
    %rx_val =w loadub %rx_addr
    %rx_wr =l loadl %wrp
    %rx_zero =w ceql %rx_wr, %cap
    jnz %rx_zero, @err1, @rep_fill
@rep_fill
    %rf_wr =l loadl %wrp
    %rf_done =w ceql %rf_wr, %cap
    jnz %rf_done, @err1, @rep_fill_body
@rep_fill_body
    %rfb_wr =l loadl %wrp
    %rfb_addr =l add %out, %rfb_wr
    storeb %rx_val, %rfb_addr
    %rfb_wr1 =l add %rfb_wr, 1
    storel %rfb_wr1, %wrp
    jmp @rep_fill
@rep_full
    %rfu_pos =l loadl %posp
    %rfu_addr =l add %src, %rfu_pos
    %rfu_val =w loadub %rfu_addr
    %rfu_pos1 =l add %rfu_pos, 1
    storel %rfu_pos1, %posp
    %rfu_wr =l loadl %wrp
    %rfu_limit =l add %rfu_wr, %ro_count
    jmp @rep_full_check
@rep_full_check
    %rfc_wr =l loadl %wrp
    %rfc_done =w ceql %rfc_wr, %rfu_limit
    jnz %rfc_done, @main, @rep_full_body
@rep_full_body
    %rfub_wr =l loadl %wrp
    %rfub_addr =l add %out, %rfub_wr
    storeb %rfu_val, %rfub_addr
    %rfub_wr1 =l add %rfub_wr, 1
    storel %rfub_wr1, %wrp
    jmp @rep_full_check
@literal
    %lt_countw =w add %h_val, %one_w
    %lt_count =l extsw %lt_countw
    %lt_pos =l loadl %posp
    %lt_need =l add %lt_pos, %lt_count
    %lt_bad =w csltl %n, %lt_need
    jnz %lt_bad, @err2, @lit_src_ok
@lit_src_ok
    %lo_wr =l loadl %wrp
    %lo_capend =l add %lo_wr, %lt_count
    %lo_ex =w csltl %cap, %lo_capend
    jnz %lo_ex, @lit_exceed, @lit_full
@lit_exceed
    %lx_wr =l loadl %wrp
    %lx_zero =w ceql %lx_wr, %cap
    jnz %lx_zero, @err1, @lit_fill
@lit_fill
    %lf_wr =l loadl %wrp
    %lf_done =w ceql %lf_wr, %cap
    jnz %lf_done, @err1, @lit_fill_body
@lit_fill_body
    %lfb_pos =l loadl %posp
    %lfb_saddr =l add %src, %lfb_pos
    %lfb_b =w loadub %lfb_saddr
    %lfb_wr =l loadl %wrp
    %lfb_daddr =l add %out, %lfb_wr
    storeb %lfb_b, %lfb_daddr
    %lfb_pos1 =l add %lfb_pos, 1
    %lfb_wr1 =l add %lfb_wr, 1
    storel %lfb_pos1, %posp
    storel %lfb_wr1, %wrp
    jmp @lit_fill
@lit_full
    %lfu_wr =l loadl %wrp
    %lfu_limit =l add %lfu_wr, %lt_count
    jmp @lit_full_check
@lit_full_check
    %lfc_wr =l loadl %wrp
    %lfc_done =w ceql %lfc_wr, %lfu_limit
    jnz %lfc_done, @main, @lit_full_body
@lit_full_body
    %lfub_pos =l loadl %posp
    %lfub_saddr =l add %src, %lfub_pos
    %lfub_b =w loadub %lfub_saddr
    %lfub_wr =l loadl %wrp
    %lfub_daddr =l add %out, %lfub_wr
    storeb %lfub_b, %lfub_daddr
    %lfub_pos1 =l add %lfub_pos, 1
    %lfub_wr1 =l add %lfub_wr, 1
    storel %lfub_pos1, %posp
    storel %lfub_wr1, %wrp
    jmp @lit_full_check
@done
    %d_wr =l loadl %wrp
    ret %d_wr
@err1
    %e1 =l copy -1
    ret %e1
@err2
    %e2 =l copy -2
    ret %e2
}
