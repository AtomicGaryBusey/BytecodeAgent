

crc32:
    mov eax, 0xffffffff
    mov r8d, 0xedb88320
    test rsi, rsi
    jz .Ldone
.Lbyte_loop:
    movzx ecx, byte ptr [rdi]
    xor eax, ecx
    mov edx, 8
.Lbit_loop:
    shr eax, 1
    jnc .Lno_xor
    xor eax, r8d
.Lno_xor:
    dec edx
    jnz .Lbit_loop
    inc rdi
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    not eax
    ret
