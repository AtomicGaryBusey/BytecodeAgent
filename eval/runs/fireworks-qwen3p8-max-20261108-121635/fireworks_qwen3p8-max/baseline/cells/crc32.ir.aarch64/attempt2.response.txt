

export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy -1
	%ptr0 =l copy %p
	%rem0 =l copy %n
	%poly =w copy -306674912
	jmp @byte_loop
@byte_loop
	%crc =w phi @start %crc0, @byte_body %crc8
	%ptr =l phi @start %ptr0, @byte_body %ptr1
	%rem =l phi @start %rem0, @byte_body %rem1
	%done =w ceql %rem, 0
	jnz %done, @finish, @byte_body
@byte_body
	%b =w loadub %ptr
	%x0 =w xor %crc, %b
	%lsb1 =w and %x0, 1
	%mask1 =w sub 0, %lsb1
	%poly1 =w and %mask1, %poly
	%shift1 =w shr %x0, 1
	%crc1 =w xor %shift1, %poly1
	%lsb2 =w and %crc1, 1
	%mask2 =w sub 0, %lsb2
	%poly2 =w and %mask2, %poly
	%shift2 =w shr %crc1, 1
	%crc2 =w xor %shift2, %poly2
	%lsb3 =w and %crc2, 1
	%mask3 =w sub 0, %lsb3
	%poly3 =w and %mask3, %poly
	%shift3 =w shr %crc2, 1
	%crc3 =w xor %shift3, %poly3
	%lsb4 =w and %crc3, 1
	%mask4 =w sub 0, %lsb4
	%poly4 =w and %mask4, %poly
	%shift4 =w shr %crc3, 1
	%crc4 =w xor %shift4, %poly4
	%lsb5 =w and %crc4, 1

	%crc5 =w xor %shift5, %poly5
	%lsb6 =w and %crc5, 1

	%crc6 =w xor %shift6, %poly6
	%lsb7 =w and %crc6, 1
	%mask7 =w sub 0, %lsb7

	%shift7 =w shr %crc6, 1
	%crc7 =w xor %shift7, %poly7
	%lsb8 =w and %crc7, 1
	%mask8 =w sub 0, %lsb8
	%poly8 =w and %mask8, %poly
	%shift8 =w shr %crc7, 1
	%crc8 =w xor %shift8, %poly8
	%ptr1 =l add %ptr, 1
	%rem1 =l sub %rem, 1
	jmp @byte_loop
@finish
	%final =w xor %crc, %crc0
	%res =l extuw %final
	ret %res
}
