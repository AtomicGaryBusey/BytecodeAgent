

export function l $count_desc(l %a, l %n) {
@entry
    %early =w csltl %n, 2
    jnz %early, @retzero, @preloop
@preloop
    %limit =l sub %n, 1
    %zero =l sub %n, %n
    jmp @loop
@loop
    %i =l phi @preloop %zero, @body %i1
    %cnt =l phi @preloop %zero, @body %cnt1
    %c =w csltl %i, %limit
    jnz %c, @body, @done
@body
    %off =l mul %i, 8
    %p =l add %a, %off
    %cur =l loadl %p
    %p1 =l add %p, 8
    %nxt =l loadl %p1
    %dec =w csltl %nxt, %cur
    %inc =l extuw %dec
    %cnt1 =l add %cnt, %inc
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %cnt
@retzero
    ret 0
}
