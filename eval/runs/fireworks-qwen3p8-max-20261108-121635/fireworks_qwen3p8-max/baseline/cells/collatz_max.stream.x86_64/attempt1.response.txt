

collatz_max:
    xor eax, eax
    test rdi, rdi
    jle .Ldone
    mov ecx, 1
.Lkloop:
    mov rdx, rcx
    xor esi, esi
.Lcollatz:
    cmp rdx, 1
    je .Lupdate
    test dl, 1
    jne .Lodd
    shr rdx, 1
    inc rsi
    jmp .Lcollatz
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
    inc rsi
    jmp .Lcollatz
.Lupdate:
    cmp rsi, rax
    jle .Lskip
    mov rax, rsi
.Lskip:
    inc rcx
    cmp rcx, rdi
    jle .Lkloop
.Ldone:
    ret
