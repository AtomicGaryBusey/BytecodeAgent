

export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %ntot =l add %na, %nb
  %zero =l sub %ntot, %ntot
  %zi =l add %zero, %zero
  jmp @zero_loop
@zero_loop
  %zcond =w cultl %zi, %ntot
  jnz %zcond, @zero_body, @outer_init
@zero_body
  %zoff =l shl %zi, 3
  %zaddr =l add %r, %zoff
  storel %zero, %zaddr
  %zi =l add %zi, 1
  jmp @zero_loop
@outer_init
  %oi =l add %zero, %zero
  jmp @outer_check
@outer_check
  %icond =w cultl %oi, %na
  jnz %icond, @outer_body, @count_init
@outer_body
  %aoff =l shl %oi, 3
  %aaddr =l add %a, %aoff
  %ai =l loadl %aaddr
  %jj =l add %zero, %zero
  %cc =l add %zero, %zero
  jmp @inner_check
@inner_check
  %jcond =w cultl %jj, %nb
  jnz %jcond, @inner_body, @inner_store
@inner_body
  %boff =l shl %jj, 3
  %baddr =l add %b, %boff
  %bj =l loadl %baddr
  %k =l add %oi, %jj
  %roff =l shl %k, 3
  %raddr =l add %r, %roff
  %cur =l loadl %raddr
  %xlo =l and %ai, 4294967295
  %xhi =l shr %ai, 32
  %ylo =l and %bj, 4294967295
  %yhi =l shr %bj, 32
  %p0 =l mul %xlo, %ylo
  %p1 =l mul %xlo, %yhi
  %p2 =l mul %xhi, %ylo
  %p3 =l mul %xhi, %yhi
  %mid =l add %p1, %p2
  %midcw =w cultl %mid, %p1
  %midc =l extuw %midcw
  %midlo =l and %mid, 4294967295
  %midhi =l shr %mid, 32
  %midlosh =l shl %midlo, 32
  %plo =l add %p0, %midlosh
  %locw =w cultl %plo, %p0
  %loc =l extuw %locw
  %midcsh =l shl %midc, 32
  %phi1 =l add %p3, %midhi
  %phi2 =l add %phi1, %midcsh
  %phi =l add %phi2, %loc
  %sum1 =l add %cur, %plo
  %c1w =w cultl %sum1, %cur
  %c1 =l extuw %c1w
  %sum2 =l add %sum1, %cc
  %c2w =w cultl %sum2, %sum1
  %c2 =l extuw %c2w
  storel %sum2, %raddr
  %nc1 =l add %phi, %c1
  %cc =l add %nc1, %c2
  %jj =l add %jj, 1
  jmp @inner_check
@inner_store
  %stidx =l add %oi, %nb
  %stoff =l shl %stidx, 3
  %staddr =l add %r, %stoff
  storel %cc, %staddr
  %oi =l add %oi, 1
  jmp @outer_check
@count_init
  %kk =l add %ntot, %zero
  jmp @count_loop
@count_loop
  %kzero =w ceql %kk, 0
  jnz %kzero, @count_zero, @count_body
@count_body
  %km1 =l sub %kk, 1
  %koff =l shl %km1, 3
  %kaddr =l add %r, %koff
  %kv =l loadl %kaddr
  %viszero =w ceql %kv, 0
  jnz %viszero, @count_next, @count_found
@count_found
  ret %kk
@count_next
  %kk =l sub %kk, 1
  jmp @count_loop
@count_zero
  ret %zero
}
