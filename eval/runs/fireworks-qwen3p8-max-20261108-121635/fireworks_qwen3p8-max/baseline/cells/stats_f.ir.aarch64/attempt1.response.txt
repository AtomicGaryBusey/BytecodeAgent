

export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    %rem0 =l sub %n, 1
    %p1 =l add %a, 8
    jmp @loop(l %p1, l %rem0, l %y0, l %y0, l %y0, l %odd0)
@loop(l %lp, l %lrem, l %lmn, l %lmx, l %lsm, l %lcnt)
    %done =w ceql %lrem, 0
    jnz %done, @finish, @body
@body
    %x =l loadl %lp
    %y =l call %f(l %x)
    %sm1 =l add %lsm, %y
    %bit =l and %y, 1
    %cnt1 =l add %lcnt, %bit
    %p2 =l add %lp, 8
    %rem1 =l sub %lrem, 1
    %less =w csltl %y, %lmn
    jnz %less, @min, @notmin
@min
    jmp @loop(l %p2, l %rem1, l %y, l %lmx, l %sm1, l %cnt1)
@notmin
    %greater =w csltl %lmx, %y
    jnz %greater, @max, @cont
@max
    jmp @loop(l %p2, l %rem1, l %lmn, l %y, l %sm1, l %cnt1)
@cont
    jmp @loop(l %p2, l %rem1, l %lmn, l %lmx, l %sm1, l %cnt
