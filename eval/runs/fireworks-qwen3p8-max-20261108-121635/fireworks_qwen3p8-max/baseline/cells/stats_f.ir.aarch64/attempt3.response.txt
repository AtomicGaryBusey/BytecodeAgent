

export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
  %x0 =l loadl %a
  %y0 =l call %f(l %x0)
  %two0 =l copy 2
  %zero0 =l copy 0
  %r0 =l rem %y0, %two0
  %z0 =w ceql %r0, %zero0
  jnz %z0, @first_even, @first_odd
@first_odd
  %cnt_o =l copy 1
  jmp @first_join
@first_even
  %cnt_e =l copy 0
  jmp @first_join
@first_join
  %cnt0 =l phi @first_odd %cnt_o, @first_even %cnt_e
  %min_init =l copy %y0
  %max_init =l copy %y0
  %sum_init =l copy %y0
  %i1 =l copy 1
  %eight_f =l copy 8
  %p1 =l add %a, %eight_f
  %c0 =w csltl %i1, %n
  jnz %c0, @loop, @done
@loop
  %i =l phi @first_join %i1, @next %i2
  %p =l phi @first_join %p1, @next %p2
  %min =l phi @first_join %min_init, @next %min2
  %max =l phi @first_join %max_init, @next %max2
  %sum =l phi @first_join %sum_init, @next %sum2
  %cnt =l phi @first_join %cnt0, @next %cnt3
  %x =l loadl %p
  %y =l call %f(l %x)
  %sum1 =l add %sum, %y
  %ltmin =w csltl %y, %min
  jnz %ltmin, @min_new, @min_old
@min_new
  %min_new_val =l copy %y
  jmp @min_join
@min_old
  %min_old_val =l copy %min
  jmp @min_join
@min_join
  %min1 =l phi @min_new %min_new_val, @min_old %min_old_val
  %ltmax =w csltl %max, %y
  jnz %ltmax, @max_new, @max_old
@max_new
  %max_new_val =l copy %y
  jmp @max_join
@max_old
  %max_old_val =l copy %max
  jmp @max_join
@max_join
  %max1 =l phi @max_new %max_new_val, @max_old %max_old_val
  %two_m =l copy 2
  %zero_m =l copy 0
  %r =l rem %y, %two_m
  %z =w ceql %r, %zero_m
  jnz %z, @even, @odd
@odd
  %one_o =l copy 1
  %cnt_odd_val =l add %cnt, %one_o
  jmp @odd_join
@even
  %cnt_even_val =l copy %cnt
  jmp @odd_join
@odd_join
  %cnt2 =l phi @odd %cnt_odd_val, @even %cnt_even_val
  jmp @next
@next
  %min2 =l copy %min1
  %max2 =l copy %max1
  %sum2 =l copy %sum1
  %cnt3 =l copy %cnt2
  %one_n =l copy 1
  %eight_n =l copy 8
  %i2 =l add %i, %one_n
  %p2 =l add %p, %eight_n
  %c =w csltl %i2, %n
  jnz %c, @loop, @done
@done
  %minf =l phi @first_join %min_init, @next %min2
  %maxf =l phi @first_join %max_init, @next %max2
  %sumf =l phi @first_join %sum_init, @next %sum2
  %cntf =l phi @first_join %cnt0, @next %cnt3
  %eight_d =l copy 8
  %sixteen_d =l copy 16
  storel %minf, %out
  %out1 =l add %out, %eight_d
  storel %maxf, %out1
  %out2 =l add %out, %sixteen_d
  storel %sumf, %out2
  ret %cntf
}
