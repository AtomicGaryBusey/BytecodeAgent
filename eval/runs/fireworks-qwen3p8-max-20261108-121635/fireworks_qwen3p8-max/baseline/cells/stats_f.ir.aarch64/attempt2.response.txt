

export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
  %v0 =l loadl %a
  %y0 =l call %f(l %v0)
  %odd0 =l and %y0, 1
  %i1 =l 1
  %p1 =l add %a, 8
  %c0 =w csltl %i1, %n
  jnz %c0, @loop, @done
@loop
  %i =l phi @entry %i1, @cont %inext
  %p =l phi @entry %p1, @cont %pnext
  %min =l phi @entry %y0, @cont %minc
  %max =l phi @entry %y0, @cont %max1
  %sum =l phi @entry %y0, @cont %sum1
  %cnt =l phi @entry %odd0, @cont %cnt1
  %x =l loadl %p
  %y =l call %f(l %x)
  %ltmin =w csltl %y, %min
  jnz %ltmin, @min_yes, @min_no
@min_yes
  %miny =l add %y, 0
  jmp @max_check
@min_no
  %mino =l add %min, 0
  jmp @max_check
@max_check
  %min1 =l phi @min_yes %miny, @min_no %mino
  %ltmax =w csltl %max, %y
  jnz %ltmax, @max_yes, @max_no
@max_yes
  %maxy =l add %y, 0
  jmp @cont
@max_no
  %maxo =l add %max, 0
  jmp @cont
@cont
  %max1 =l phi @max_yes %maxy, @max_no %maxo
  %minc =l add %min1, 0
  %sum1 =l add %sum, %y
  %oddy =l and %y, 1
  %cnt1 =l add %cnt, %oddy
  %inext =l add %i, 1
  %pnext =l add %p, 8
  %c =w csltl %inext, %n
  jnz %c, @loop, @done
@done
  %minf =l phi @entry %y0, @cont %minc
  %maxf =l phi @entry %y0, @cont %max1
  %sumf =l phi @entry %y0, @cont %sum1
  %cntf =l phi @entry %odd0, @cont %cnt1
  storel %minf, %out
  %out1 =l add %out, 8
  storel %maxf, %out1
  %out2 =l add %out, 16
  storel %sumf, %out2
  ret %cntf
}
