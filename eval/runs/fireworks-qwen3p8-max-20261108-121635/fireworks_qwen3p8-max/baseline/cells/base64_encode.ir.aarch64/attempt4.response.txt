

export function l $base64_encode(l %src, l %n, l %out) {
@start
    %init_ip =l add %src, 0
    %init_op =l add %out, 0
    %init_rem =l add %n, 0
    jmp @group_loop
@group_loop
    %ip =l phi @start %init_ip, @emit_done %g_ip
    %op =l phi @start %init_op, @emit_done %g_op
    %rem =l phi @start %init_rem, @emit_done %g_rem
    %done =w ceql %rem, 0
    jnz %done, @finish, @group_type
@finish
    %len =l sub %op, %out
    ret %len
@group_type
    %is_tail =w csltl %rem, 3
    jnz %is_tail, @tail_check, @full_prep
@full_prep
    %f_b0 =l loadub %ip
    %f_p1 =l add %ip, 1
    %f_b1 =l loadub %f_p1
    %f_p2 =l add %ip, 2
    %f_b2 =l loadub %f_p2
    %f_idx0 =l sar %f_b0, 2
    %f_m0 =l and %f_b0, 3
    %f_s0 =l shl %f_m0, 4
    %f_h1 =l sar %f_b1, 4
    %f_idx1 =l or %f_s0, %f_h1
    %f_m1 =l and %f_b1, 15
    %f_s1 =l shl %f_m1, 2
    %f_h2 =l sar %f_b2, 6
    %f_idx2 =l or %f_s1, %f_h2
    %f_idx3 =l and %f_b2, 63
    %f_ip =l add %ip, 0
    %f_op =l add %op, 0
    %f_rem =l add %rem, 0
    %f_consumed =l add 0, 3
    %f_state =l add 0, 0
    jmp @emit_loop
@tail_check
    %is_one =w ceql %rem, 1
    jnz %is_one, @tail1_prep, @tail2_prep
@tail1_prep
    %t1_b0 =l loadub %ip
    %t1_idx0 =l sar %t1_b0, 2
    %t1_m0 =l and %t1_b0, 3
    %t1_idx1 =l shl %t1_m0, 4
    %t1_idx2 =l add 0, 64
    %t1_idx3 =l add 0, 64
    %t1_ip =l add %ip, 0
    %t1_op =l add %op, 0
    %t1_rem =l add %rem, 0
    %t1_consumed =l add 0, 1
    %t1_state =l add 0, 0
    jmp @emit_loop
@tail2_prep
    %t2_b0 =l loadub %ip
    %t2_p1 =l add %ip, 1
    %t2_b1 =l loadub %t2_p1
    %t2_idx0 =l sar %t2_b0, 2
    %t2_m0 =l and %t2_b0, 3
    %t2_s0 =l shl %t2_m0, 4
    %t2_h1 =l sar %t2_b1, 4
    %t2_idx1 =l or %t2_s0, %t2_h1
    %t2_m1 =l and %t2_b1, 15
    %t2_idx2 =l shl %t2_m1, 2
    %t2_idx3 =l add 0, 64
    %t2_ip =l add %ip, 0
    %t2_op =l add %op, 0
    %t2_rem =l add %rem, 0
    %t2_consumed =l add 0, 2
    %t2_state =l add 0, 0
    jmp @emit_loop
@emit_loop
    %e_ip =l phi @full_prep %f_ip, @tail1_prep %t1_ip, @tail2_prep %t2_ip, @emit_next %n_ip
    %e_op =l phi @full_prep %f_op, @tail1_prep %t1_op, @tail2_prep %t2_op, @emit_next %n_op
    %e_rem =l phi @full_prep %f_rem, @tail1_prep %t1_rem, @tail2_prep %t2_rem, @emit_next %n_rem
    %e_consumed =l phi @full_prep %f_consumed, @tail1_prep %t1_consumed, @tail2_prep %t2_consumed, @emit_next %n_consumed
    %e_state =l phi @full_prep %f_state, @tail1_prep %t1_state, @tail2_prep %t2_state, @emit_next %n_state
    %e_i0 =l phi @full_prep %f_idx0, @tail1_prep %t1_idx0, @tail2_prep %t2_idx0, @emit_next %n_i0
    %e_i1 =l phi @full_prep %f_idx1, @tail1_prep %t1_idx1, @tail2_prep %t2_idx1, @emit_next %n_i1
    %e_i2 =l phi @full_prep %f_idx2, @tail1_prep %t1_idx2, @tail2_prep %t2_idx2, @emit_next %n_i2
    %e_i3 =l phi @full_prep %f_idx3, @tail1_prep %t1_idx3, @tail2_prep %t2_idx3, @emit_next %n_i3
    %e_done =w ceql %e_state, 4
    jnz %e_done, @emit_done, @emit_body
@emit_body
    %s0w =w ceql %e_state, 0
    %s0 =l extuw %s0w
    %s1w =w ceql %e_state, 1
    %s1 =l extuw %s1w
    %s2w =w ceql %e_state, 2
    %s2 =l extuw %s2w
    %s3w =w ceql %e_state, 3
    %s3 =l extuw %s3w
    %sel0 =l mul %e_i0, %s0
    %sel1 =l mul %e_i1, %s1
    %sel2 =l mul %e_i2, %s2
    %sel3 =l mul %e_i3, %s3
    %sel01 =l add %sel0, %sel1
    %sel23 =l add %sel2, %sel3
    %idx =l add %sel01, %sel23
    %is_pad =w ceql %idx, 64
    jnz %is_pad, @emit_pad, @emit_map
@emit_pad
    %padc =w add 0, 61
    storeb %padc, %e_op
    jmp @emit_next
@emit_map
    %m0 =l and %idx, 1
    %z0 =w ceql %m0, 0
    %b0 =w sub 1, %z0
    %v0 =w mul %b0, 1
    %m1 =l and %idx, 2
    %z1 =w ceql %m1, 0
    %b1 =w sub 1, %z1
    %v1 =w mul %b1, 2
    %m2 =l and %idx, 4
    %z2 =w ceql %m2, 0
    %b2 =w sub 1, %z2
    %v2 =w mul %b2, 4
    %m3 =l and %idx, 8
    %z3 =w ceql %m3, 0
    %b3 =w sub 1, %z3
    %v3 =w mul %b3, 8
    %m4 =l and %idx, 16
    %z4 =w ceql %m4, 0
    %b4 =w sub 1, %z4
    %v4 =w mul %b4, 16
    %m5 =l and %idx, 32
    %z5 =w ceql %m5, 0
    %b5 =w sub 1, %z5
    %v5 =w mul %b5, 32
    %s01w =w add %v0, %v1
    %s23w =w add %v2, %v3
    %s45w =w add %v4, %v5
    %s0123w =w add %s01w, %s23w
    %idxw =w add %s0123w, %s45w
    %lt26 =w csltl %idx, 26
    %ge26 =w sub 1, %lt26
    %lt52 =w csltl %idx, 52
    %ge52 =w sub 1, %lt52
    %lt62 =w csltl %idx, 62
    %ge62 =w sub 1, %lt62
    %eq63 =w ceql %idx, 63
    %c0 =w add %idxw, 65
    %a1 =w mul %ge26, 6
    %c1 =w add %c0, %a1
    %a2 =w mul %ge52, 75
    %c2 =w sub %c1, %a2
    %a3 =w mul %ge62, 15
    %c3 =w sub %c2, %a3
    %a4 =w mul %eq63, 3
    %c4 =w add %c3, %a4
    storeb %c4, %e_op
    jmp @emit_next
@emit_next
    %n_state =l add %e_state, 1
    %n_op =l add %e_op, 1
    %n_ip =l add %e_ip, 0
    %n_rem =l add %e_rem, 0
    %n_consumed =l add %e_consumed, 0
    %n_i0 =l add %e_i0, 0
    %n_i1 =l add %e_i1, 0
    %n_i2 =l add %e_i2, 0
    %n_i3 =l add %e_i3, 0
    jmp @emit_loop
@emit_done
    %next_ip =l add %e_ip, %e_consumed
    %next_rem =l sub %e_rem, %e_consumed
    %g_ip =l add %next_ip, 0
    %g_op =l add %e_op, 0
    %g_rem =l add %next_rem, 0
    jmp @group_loop
}
