

export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @group_loop(%src, %out, %n)
@group_loop(%g_ip, %g_op, %g_rem)
    %g_done =w ceql %g_rem, 0
    jnz %g_done, @finish, @group_sel
@finish
    %f_len =l sub %g_op, %out
    ret %f_len
@group_sel
    %g_is_tail =w csltl %g_rem, 3
    jnz %g_is_tail, @tail_group, @full_group
@full_group
    %fg_b0 =l loadub %g_ip
    %fg_p1 =l add %g_ip, 1
    %fg_b1 =l loadub %fg_p1
    %fg_p2 =l add %g_ip, 2
    %fg_b2 =l loadub %fg_p2
    %fg_i0 =l sar %fg_b0, 2
    %fg_m0 =l and %fg_b0, 3
    %fg_s0 =l shl %fg_m0, 4
    %fg_h1 =l sar %fg_b1, 4
    %fg_i1 =l or %fg_s0, %fg_h1
    %fg_m1 =l and %fg_b1, 15
    %fg_s1 =l shl %fg_m1, 2
    %fg_h2 =l sar %fg_b2, 6
    %fg_i2 =l or %fg_s1, %fg_h2
    %fg_i3 =l and %fg_b2, 63
    %fg_consumed =l add 0, 3
    %fg_state =l add 0, 0
    jmp @emit_dispatch(%g_ip, %g_op, %g_rem, %fg_consumed, %fg_i0, %fg_i1, %fg_i2, %fg_i3, %fg_state)
@tail_group
    %tg_is1 =w ceql %g_rem, 1
    jnz %tg_is1, @tail1, @tail2
@tail1
    %t1_b0 =l loadub %g_ip
    %t1_i0 =l sar %t1_b0, 2
    %t1_m0 =l and %t1_b0, 3
    %t1_i1 =l shl %t1_m0, 4
    %t1_pad =l add 0, 64
    %t1_consumed =l add 0, 1
    %t1_state =l add 0, 0
    jmp @emit_dispatch(%g_ip, %g_op, %g_rem, %t1_consumed, %t1_i0, %t1_i1, %t1_pad, %t1_pad, %t1_state)
@tail2
    %t2_b0 =l loadub %g_ip
    %t2_p1 =l add %g_ip, 1
    %t2_b1 =l loadub %t2_p1
    %t2_i0 =l sar %t2_b0, 2
    %t2_m0 =l and %t2_b0, 3
    %t2_s0 =l shl %t2_m0, 4
    %t2_h1 =l sar %t2_b1, 4
    %t2_i1 =l or %t2_s0, %t2_h1
    %t2_m1 =l and %t2_b1, 15
    %t2_i2 =l shl %t2_m1, 2
    %t2_pad =l add 0, 64
    %t2_consumed =l add 0, 2
    %t2_state =l add 0, 0
    jmp @emit_dispatch(%g_ip, %g_op, %g_rem, %t2_consumed, %t2_i0, %t2_i1, %t2_i2, %t2_pad, %t2_state)
@emit_dispatch(%ed_ip, %ed_op, %ed_rem, %ed_consumed, %ed_i0, %ed_i1, %ed_i2, %ed_i3, %ed_state)
    %ed_c0 =w ceql %ed_state, 0
    jnz %ed_c0, @emit_pick0, @emit_not0
@emit_pick0
    jmp @map(%ed_i0, %ed_state, %ed_ip, %ed_op, %ed_rem, %ed_consumed, %ed_i0, %ed_i1, %ed_i2, %ed_i3)
@emit_not0
    %ed_c1 =w ceql %ed_state, 1
    jnz %ed_c1, @emit_pick1, @emit_not1
@emit_pick1
    jmp @map(%ed_i1, %ed_state, %ed_ip, %ed_op, %ed_rem, %ed_consumed, %ed_i0, %ed_i1, %ed_i2, %ed_i3)
@emit_not1
    %ed_c2 =w ceql %ed_state, 2
    jnz %ed_c2, @emit_pick2, @emit_pick3
@emit_pick2
    jmp @map(%ed_i2, %ed_state, %ed_ip, %ed_op, %ed_rem, %ed_consumed, %ed_i0, %ed_i1, %ed_i2, %ed_i3)
@emit_pick3
    jmp @map(%ed_i3, %ed_state, %ed_ip, %ed_op, %ed_rem, %ed_consumed, %ed_i0, %ed_i1, %ed_i2, %ed_i3)
@map(%m_idx, %m_state, %m_ip, %m_op, %m_rem, %m_consumed, %m_i0, %m_i1, %m_i2, %m_i3)
    %m_is_pad =w ceql %m_idx, 64
    jnz %m_is_pad, @map_pad, @map_not_pad
@map_pad
    %m_pad_char =l add 0, 61
    jmp @map_done(%m_pad_char)
@map_not_pad
    %m_lt26 =w csltl %m_idx, 26
    jnz %m_lt26, @map_upper, @map_not_upper
@map_upper
    %m_upper_char =l add %m_idx, 65
    jmp @map_done(%m_upper_char)
@map_not_upper
    %m_lt52 =w csltl %m_idx, 52
    jnz %m_lt52, @map_lower, @map_not_lower
@map_lower
    %m_lower_off =l sub %m_idx, 26
    %m_lower_char =l add %m_lower_off, 97
    jmp @map_done(%m_lower_char)
@map_not_lower
    %m_lt62 =w csltl %m_idx, 62
    jnz %m_lt62, @map_digit, @map_not_digit
@map_digit
    %m_digit_off =l sub %m_idx, 52
    %m_digit_char =l add %m_digit_off, 48
    jmp @map_done(%m_digit_char)
@map_not_digit
    %m_eq62 =w ceql %m_idx, 62
    jnz %m_eq62, @map_plus, @map_slash
@map_plus
    %m_plus_char =l add 0, 43
    jmp @map_done(%m_plus_char)
@map_slash
    %m_slash_char =l add 0, 47
    jmp @map_done(%m_slash_char)
@map_done(%md_char)
    storeb %md_char, %m_op
    %md_next_state =l add %m_state, 1
    %md_next_op =l add %m_op, 1
    %md_more =w csltl %md_next_state, 4
    jnz %md_more, @emit_more, @group_next
@emit_more
    jmp @emit_dispatch(%m_ip, %md_next_op, %m_rem, %m_consumed, %m_i0, %m_i1, %m_i2, %m_i3, %md_next_state)
@group_next
    %gn_ip =l add %m_ip, %m_consumed
    %gn_rem =l sub %m_rem, %m_consumed
    jmp @group_loop(%gn_ip, %md_next_op, %gn_rem)
}
