

export function l $fill_down(l %a, l %n, l %start) {
@start
  %i0 =l copy 0
  %one =l copy 1
  %eight =l copy 8
  jmp @loop
@loop
  %i =l phi @start %i0, @body %inext
  %ok =w csltl %i, %n
  jnz %ok, @body, @done
@body
  %off =l mul %i, %eight
  %p =l add %a, %off
  %val =l sub %start, %i
  storel %val, %p
  %inext =l add %i, %one
  jmp @loop
@done
  %res =l copy %n
  ret %res
}
