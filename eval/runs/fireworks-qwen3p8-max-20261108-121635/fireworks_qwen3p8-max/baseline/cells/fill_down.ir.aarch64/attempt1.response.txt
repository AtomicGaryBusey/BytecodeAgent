

export function l $fill_down(l %a, l %n, l %start) {
@entry
  %zero =l copy 0
  %one =l copy 1
  %eight =l copy 8
  jmp @loop
@loop
  %i =l phi @entry %zero, @body %next
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %offset =l mul %i, %eight
  %p =l add %a, %offset
  %v =l sub %start, %i
  storel %v, %p
  %next =l add %i, %one
  jmp @loop
@exit
  ret l %n
}
