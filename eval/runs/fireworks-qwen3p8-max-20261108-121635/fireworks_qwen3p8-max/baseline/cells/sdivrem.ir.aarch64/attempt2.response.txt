

export function l $sdivrem(l %a, l %b, l %rem)
{
@start
%zero =l copy 0
%bzero =w ceql %b, %zero
jnz %bzero, @bzero, @nonzero
@bzero
storel %a, %rem
ret %zero
@nonzero
%one =l copy 1
%shift =l copy 63
%min =l shl %one, %shift
%amin =w ceql %a, %min
jnz %amin, @maybe, @normal
@maybe
%negone =l sub %zero, %one
%bneg =w ceql %b, %negone
jnz %bneg, @special, @normal
@special
storel %zero, %rem
ret %min
@normal
%q =l div %a, %b
%r =l rem %a, %b
storel %r, %rem
ret %q
}
