

binary_search:
    test rsi, rsi
    jle .Lnot_found
    xor eax, eax
    lea rcx, [rsi - 1]
.Lloop:
    cmp rax, rcx
    jg .Lnot_found
    mov r8, rcx
    sub r8, rax
    shr r8, 1
    lea r9, [rax + r8]
    cmp rdx, qword ptr [rdi + r9*8]
    je .Lfound
    jl .Lleft
    lea rax, [r9 + 1]
    jmp .Lloop
.Lleft:
    lea rcx, [r9 - 1]
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnot_found:
    mov rax, -1
    ret
