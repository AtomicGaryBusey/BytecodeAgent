

export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %zero, @next %inext
	%count =l phi @entry %zero, @next %countnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%countbody =l copy %count
	jnz %gt, @then, @next
@then
	%countthen =l add %countbody, 1
	jmp @next
@next
	%countnext =l phi @body %countbody, @then %countthen
	%inext =l add %i, 1
	jmp @loop
@end
	ret %count
}
