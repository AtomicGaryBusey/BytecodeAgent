

qsort_pairs:
    cmp x1, #1
    b.le .Lreturn_zero
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    lsr x21, x20, #1
    add x22, x19, x21, lsl #4
    ldp x24, x25, [x22]
    mov x21, x19
    mov x23, x19
    sub x22, x20, #1
    add x22, x19, x22, lsl #4
.Lloop:
    cmp x23, x22
    b.hi .Lafter_loop
    ldp x9, x10, [x23]
    cmp x9, x24
    b.lt .Lless
    b.gt .Lgreater
    cmp x10, x25
    b.lt .Lless
    b.gt .Lgreater
    add x23, x23, #16
    b .Lloop
.Lless:
    ldp x9, x10, [x23]
    ldp x11, x12, [x21]
    stp x9, x10, [x21]
    stp x11, x12, [x23]
    add x21, x21, #16
    add x23, x23, #16
    b .Lloop
.Lgreater:
    ldp x9, x10, [x23]
    ldp x11, x12, [x22]
    stp x9, x10, [x22]
    stp x11, x12, [x23]
    sub x22, x22, #16
    b .Lloop
.Lafter_loop:
    sub x26, x21, x19
    lsr x26, x26, #4
    add x25, x22, #16
    add x0, x19, x20, lsl #4
    sub x0, x0, x25
    lsr x0, x0, #4
    mov x23, x26
    mov x21, x25
    mov x22, x0
    cmp x23, #1
    b.le .Lskip_less
    mov x0, x19
    mov x1, x23
    bl qsort_pairs
.Lskip_less:
    cmp x22, #1
    b.le .Lskip_greater
    mov x0, x21
    mov x1, x22
    bl qsort_pairs
.Lskip_greater:
    mov x0, #0
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
.Lreturn_zero:
    mov x0, #0
    ret
