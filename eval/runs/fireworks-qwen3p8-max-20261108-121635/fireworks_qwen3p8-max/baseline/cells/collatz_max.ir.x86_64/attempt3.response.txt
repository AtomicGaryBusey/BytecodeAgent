

export function l $collatz_max(l %n)
{
@start
    %kptr =l alloc8 8
    %maxptr =l alloc8 8
    %xptr =l alloc8 8
    %stepsptr =l alloc8 8
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %three =l copy 3
    %limit =l add %n, %one
    storel %one, %kptr
    storel %zero, %maxptr
    jmp @loop
@loop
    %k =l loadl %kptr
    %cont =w csltl %k, %limit
    jnz %cont, @body, @exit
@body
    storel %k, %xptr
    storel %zero, %stepsptr
    jmp @inner
@inner
    %x =l loadl %xptr
    %isone =w ceql %x, %one
    jnz %isone, @update, @step
@step
    %odd =l and %x, %one
    %isodd =w ceql %odd, %one
    jnz %isodd, @odd, @even
@odd
    %t =l mul %x, %three
    %xnew =l add %t, %one
    %steps =l loadl %stepsptr
    %stepsnew =l add %steps, %one
    storel %xnew, %xptr
    storel %stepsnew, %stepsptr
    jmp @inner
@even
    %xnewe =l div %x, %two
    %stepse =l loadl %stepsptr
    %stepsnewe =l add %stepse, %one
    storel %xnewe, %xptr
    storel %stepsnewe, %stepsptr
    jmp @inner
@update
    %stepsu =l loadl %stepsptr
    %max =l loadl %maxptr
    %maxlt =w csltl %max, %stepsu
    jnz %maxlt, @take, @next
@take
    storel %stepsu, %maxptr
    jmp @next
@next
    %kcur =l loadl %kptr
    %knext =l add %kcur, %one
    storel %knext, %kptr
    jmp @loop
@exit
    %res =l loadl %maxptr
    ret %res
}
