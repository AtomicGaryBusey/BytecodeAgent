

export function l $collatz_max(l %n)
{
@start
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %three =l copy 3
    %limit =l addl %n, %one
    jmp @loop(%one, %zero)
@loop(%k l, %max l)
    %cont =w csltl %k, %limit
    jnz %cont, @body, @exit
@body
    jmp @inner(%k, %zero)
@inner(%x l, %steps l)
    %isone =w ceql %x, %one
    jnz %isone, @update, @step
@step
    %oddval =l andl %x, %one
    %isodd =w ceql %oddval, %one
    jnz %isodd, @odd, @even
@odd
    %t =l mull %x, %three
    %xnext =l addl %t, %one
    %snext =l addl %steps, %one
    jmp @inner(%xnext, %snext)
@even
    %xnext =l divl %x, %two
    %snext =l addl %steps, %one
    jmp @inner(%xnext, %snext)
@update
    %maxlt =w csltl %max, %steps
    jnz %maxlt, @take, @keep
@take
    jmp @next(%steps)
@keep
    jmp @next(%max)
@next(%newmax l)
    %knext =l addl %k, %one
    jmp @loop(%knext, %newmax)
@exit
    ret %max
}
