

export function l $qsort_pairs(l %a, l %n) {
@start
    %small =w csltl %n, 2
    jnz %small, @ret0, @init
@ret0
    ret 0
@init
    %last =l sub %n, 1
    %lastoff =l mul %last, 16
    %lastptr =l add %a, %lastoff
    %lastvalptr =l add %lastptr, 8
    %pk =l loadl %lastptr
    %pv =l loadl %lastvalptr
    %zero =l sub %n, %n
    %i_init =l sub %zero, 1
    jmp @loop
@loop
    %i =l phi @init %i_init, @loop_next %i_new
    %j =l phi @init %zero, @loop_next %j_new
    %cont =w csltl %j, %last
    jnz %cont, @body, @finish
@body
    %joff =l mul %j, 16
    %jptr =l add %a, %joff
    %jvalptr =l add %jptr, 8
    %jk =l loadl %jptr
    %jv =l loadl %jvalptr
    %ltkey =w csltl %jk, %pk
    %eqkey =w ceql %jk, %pk
    %ltval =w csltl %jv, %pv
    %eqval =w ceql %jv, %pv
    %leval =w or %ltval, %eqval
    %eqle =w and %eqkey, %leval
    %le =w or %ltkey, %eqle
    jnz %le, @take, @skip
@take
    %ip =l add %i, 1
    %ioff =l mul %ip, 16
    %iptr =l add %a, %ioff
    %ivalptr =l add %iptr, 8
    %ik =l loadl %iptr
    %iv =l loadl %ivalptr
    storel %jk, %iptr
    storel %jv, %ivalptr
    storel %ik, %jptr
    storel %iv, %jvalptr
    jmp @loop_next
@skip
    %i_keep =l add %i, %zero
    jmp @loop_next
@loop_next
    %i_new =l phi @take %ip, @skip %i_keep
    %j_new =l add %j, 1
    jmp @loop
@finish
    %p =l add %i, 1
    %poff =l mul %p, 16
    %pptr =l add %a, %poff
    %pvalptr =l add %pptr, 8
    %oldpk =l loadl %pptr
    %oldpv =l loadl %pvalptr
    storel %pk, %pptr
    storel %pv, %pvalptr
    storel %oldpk, %lastptr
    storel %oldpv, %lastvalptr
    %right_start =l add %p, 1
    %right_n =l sub %n, %right_start
    %r1 =l call $qsort_pairs(l %a, l %p)
    %has_right =w csltl 0, %right_n
    jnz %has_right, @call_right, @done
@call_right
    %right_off =l mul %right_start, 16
    %right_ptr =l add %a, %right_off
    %r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
    jmp @done
@done
    ret 0
}
