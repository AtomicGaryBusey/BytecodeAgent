

export function l $qsort_pairs(l %a, l %n) {
@entry
%isbase =w csltl %n, 2
jnz %isbase, @retzero, @start
@retzero
ret 0
@start
%mid =l div %n, 2
%moff =l mul %mid, 16
%mptr =l add %a, %moff
%pkey =l loadl %mptr
%mptrv =l add %mptr, 8
%pval =l loadl %mptrv
%jinit =l sub %n, 1
%iinit =l sub %n, %n
jump @outer(%iinit, %jinit)
@outer(%oi l, %oj l)
%ocont =w cslel %oi, %oj
jnz %ocont, @obody, @oexit
@obody
jump @scani(%oi, %oj)
@oexit
jump @done(%oi, %oj)
@scani(%si l, %sj l)
%siok =w cslel %si, %sj
jnz %siok, @sicheck, @siend
@sicheck
%sioff =l mul %si, 16
%siptr =l add %a, %sioff
%sikey =l loadl %siptr
%siptrv =l add %siptr, 8
%sival =l loadl %siptrv
%siltkey =w csltl %sikey, %pkey
%sieqkey =w ceql %sikey, %pkey
%siltval =w csltl %sival, %pval
%siand =w and %sieqkey, %siltval
%siless =w or %siltkey, %siand
jnz %siless, @siinc, @siend
@siinc
%sinext =l add %si, 1
jump @scani(%sinext, %sj)
@siend
jump @scanj(%si, %sj)
@scanj(%ti l, %tj l)
%tjok =w cslel %ti, %tj
jnz %tjok, @tjcheck, @tjend
@tjcheck
%tjoff =l mul %tj, 16
%tjptr =l add %a, %tjoff
%tjkey =l loadl %tjptr
%tjptrv =l add %tjptr, 8
%tjval =l loadl %tjptrv
%tjgtkey =w csgtl %tjkey, %pkey
%tjeqkey =w ceql %tjkey, %pkey
%tjgtval =w csgtl %tjval, %pval
%tjand =w and %tjeqkey, %tjgtval
%tjgreater =w or %tjgtkey, %tjand
jnz %tjgreater, @tjdec, @tjend
@tjdec
%tjnext =l sub %tj, 1
jump @scanj(%ti, %tjnext)
@tjend
jump @after_scans(%ti, %tj)
@after_scans(%ui l, %uj l)
%swok =w cslel %ui, %uj
jnz %swok, @swap, @noswap
@swap
%uioff =l mul %ui, 16
%uiptr =l add %a, %uioff
%ujoff =l mul %uj, 16
%ujptr =l add %a, %ujoff
%uik =l loadl %uiptr
%uivptr =l add %uiptr, 8
%uiv =l loadl %uivptr
%ujk =l loadl %ujptr
%ujvptr =l add %ujptr, 8
%ujv =l loadl %ujvptr
storel %ujk, %uiptr
storel %ujv, %uivptr
storel %uik, %ujptr
storel %uiv, %ujvptr
%uinext =l add %ui, 1
%ujnext =l sub %uj, 1
jump @outer(%uinext, %ujnext)
@noswap
jump @outer(%ui, %uj)
@done(%di l, %dj l)
%ln =l add %dj, 1
%roff =l mul %di, 16
%rptr =l add %a, %roff
%rn =l sub %n, %di
%lret =l call $qsort_pairs(l %a, l %ln)
%rret =l call $qsort_pairs(l %rptr, l %rn)
ret 0
}
