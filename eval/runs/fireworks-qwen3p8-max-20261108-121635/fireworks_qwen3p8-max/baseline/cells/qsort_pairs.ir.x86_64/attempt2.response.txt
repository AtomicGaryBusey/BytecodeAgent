

export function l $qsort_pairs(l %a, l %n) {
@entry
%entry_small =w csltl %n, 2
jnz %entry_small, @return_zero, @start
@return_zero
ret 0
@start
%mid =l div %n, 2
%midoff =l mul %mid, 16
%pivotptr =l add %a, %midoff
%pivotkey =l loadl %pivotptr
%pivotvalptr =l add %pivotptr, 8
%pivotval =l loadl %pivotvalptr
%initj =l sub %n, 1
%initi =l copy 0
jmp @outer(%initi, %initj)
@outer(%oi l, %oj l)
%outer_lt =w csltl %oi, %oj
%outer_eq =w ceql %oi, %oj
%outer_cont =w or %outer_lt, %outer_eq
jnz %outer_cont, @outer_body, @outer_done
@outer_body
jmp @scani(%oi, %oj)
@outer_done
jmp @recurse(%oi, %oj)
@scani(%si l, %sj l)
%scani_lt =w csltl %si, %sj
%scani_eq =w ceql %si, %sj
%scani_cont =w or %scani_lt, %scani_eq
jnz %scani_cont, @scani_check, @scani_end
@scani_check
%scani_off =l mul %si, 16
%scani_ptr =l add %a, %scani_off
%scani_key =l loadl %scani_ptr
%scani_valptr =l add %scani_ptr, 8
%scani_val =l loadl %scani_valptr
%scani_ltkey =w csltl %scani_key, %pivotkey
%scani_eqkey =w ceql %scani_key, %pivotkey
%scani_ltval =w csltl %scani_val, %pivotval
%scani_and =w and %scani_eqkey, %scani_ltval
%scani_less =w or %scani_ltkey, %scani_and
jnz %scani_less, @scani_inc, @scani_end
@scani_inc
%scani_next =l add %si, 1
jmp @scani(%scani_next, %sj)
@scani_end
jmp @scanj(%si, %sj)
@scanj(%ti l, %tj l)
%scanj_lt =w csltl %ti, %tj
%scanj_eq =w ceql %ti, %tj
%scanj_cont =w or %scanj_lt, %scanj_eq
jnz %scanj_cont, @scanj_check, @scanj_end
@scanj_check
%scanj_off =l mul %tj, 16
%scanj_ptr =l add %a, %scanj_off
%scanj_key =l loadl %scanj_ptr
%scanj_valptr =l add %scanj_ptr, 8
%scanj_val =l loadl %scanj_valptr
%scanj_gtkey =w csltl %pivotkey, %scanj_key
%scanj_eqkey =w ceql %scanj_key, %pivotkey
%scanj_gtval =w csltl %pivotval, %scanj_val
%scanj_and =w and %scanj_eqkey, %scanj_gtval
%scanj_greater =w or %scanj_gtkey, %scanj_and
jnz %scanj_greater, @scanj_dec, @scanj_end
@scanj_dec
%scanj_next =l sub %tj, 1
jmp @scanj(%ti, %scanj_next)
@scanj_end
jmp @swap_check(%ti, %tj)
@swap_check(%ui l, %uj l)
%swap_lt =w csltl %ui, %uj
%swap_eq =w ceql %ui, %uj
%swap_cont =w or %swap_lt, %swap_eq
jnz %swap_cont, @swap_do, @swap_skip
@swap_do
%swap_ioff =l mul %ui, 16
%swap_iptr =l add %a, %swap_ioff
%swap_joff =l mul %uj, 16
%swap_jptr =l add %a, %swap_joff
%swap_ikey =l loadl %swap_iptr
%swap_ivalptr =l add %swap_iptr, 8
%swap_ival =l loadl %swap_ivalptr
%swap_jkey =l loadl %swap_jptr
%swap_jvalptr =l add %swap_jptr, 8
%swap_jval =l loadl %swap_jvalptr
storel %swap_jkey, %swap_iptr
storel %swap_jval, %swap_ivalptr
storel %swap_ikey, %swap_jptr
storel %swap_ival, %swap_jvalptr
%swap_inext =l add %ui, 1
%swap_jnext =l sub %uj, 1
jmp @outer(%swap_inext, %swap_jnext)
@swap_skip
jmp @outer(%ui, %uj)
@recurse(%ri l, %rj l)
%left_n =l add %rj, 1
%right_off =l mul %ri, 16
%right_ptr =l add %a, %right_off
%right_n =l sub %n, %ri
%left_has =w csltl 0, %left_n
jnz %left_has, @recurse_left, @recurse_right_check
@recurse_left
%left_ret =l call $qsort_pairs(l %a, l %left_n)
jmp @recurse_right_check
@recurse_right_check
%right_has =w csltl 0, %right_n
jnz %right_has, @recurse_right, @recurse_ret
@recurse_right
%right_ret =l call $qsort_pairs(l %right_ptr, l %right_n)
jmp @recurse_ret
@recurse_ret
ret 0
}
