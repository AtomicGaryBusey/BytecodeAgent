

mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    cld
    mov rbx, rdi
    mov rbp, rcx
    mov r12, r8
    mov r13, rsi
    mov r14, rsi
    imul r14, rsi
    mov r15, rdx
    xor eax, eax
    xor ecx, ecx
.Lzero_out:
    cmp rcx, r14
    je .Lzero_out_done
    mov [rbp + rcx*8], rax
    inc rcx
    jmp .Lzero_out
.Lzero_out_done:
    xor ecx, ecx
.Ldiag:
    cmp rcx, r13
    je .Lexp_check
    mov rax, rcx
    imul rax, r13
    add rax, rcx
    mov qword ptr [rbp + rax*8], 1
    inc rcx
    jmp .Ldiag
.Lexp_check:
    test r15, r15
    jz .Lzero_tmp
.Lexp_loop:
    xor ecx, ecx
.Lmul_i:
    cmp rcx, r13
    je .Lcopy_tmp
    mov rax, rcx
    imul rax, r13
    lea r9, [rbp + rax*8]
    lea r11, [r12 + rax*8]
    xor edx, edx
.Lmul_j:
    cmp rdx, r13
    je .Lmul_i_next
    xor r8d, r8d
    xor esi, esi
    mov r10, rdx
.Lmul_t:
    cmp rsi, r13
    je .Lmul_t_done
    mov rax, [r9 + rsi*8]
    imul rax, qword ptr [rbx + r10*8]
    add r8, rax
    inc rsi
    add r10, r13
    jmp .Lmul_t
.Lmul_t_done:
    mov [r11 + rdx*8], r8
    inc rdx
    jmp .Lmul_j
.Lmul_i_next:
    inc rcx
    jmp .Lmul_i
.Lcopy_tmp:
    xor ecx, ecx
.Lcopy_loop:
    cmp rcx, r14
    je .Lcopy_done
    mov rax, [r12 + rcx*8]
    mov [rbp + rcx*8], rax
    inc rcx
    jmp .Lcopy_loop
.Lcopy_done:
    dec r15
    jnz .Lexp_loop
.Lzero_tmp:
    xor eax, eax
    xor ecx, ecx
    mov rdx, r14
    add rdx, r14
.Lzero_tmp_loop:
    cmp rcx, rdx
    je .Ldone
    mov [r12 + rcx*8], rax
    inc rcx
    jmp .Lzero_tmp_loop
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
