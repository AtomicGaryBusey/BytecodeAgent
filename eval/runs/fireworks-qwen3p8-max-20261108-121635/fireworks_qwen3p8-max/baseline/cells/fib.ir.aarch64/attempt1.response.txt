

export function l $fib(l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	jmp @loop
@loop
	%a =l phi @start %zero, @body %a_next
	%b =l phi @start %one, @body %b_next
	%i =l phi @start %zero, @body %i_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%t =l add %a, %b
	%a_next =l copy %b
	%b_next =l copy %t
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %a
}
