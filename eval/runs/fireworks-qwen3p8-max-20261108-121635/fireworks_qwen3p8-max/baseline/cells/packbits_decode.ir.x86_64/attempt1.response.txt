

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%z =l copy 0
	%one =l copy 1
	%two =l copy 2
	%c128 =l copy 128
	%m1 =l sub %z, %one
	%m2 =l sub %z, %two
	%m128 =l sub %z, %c128
	jmp @loop(%z, %z)
@loop(l %i, l %o)
	%cond =w csltl %i, %n
	jnz %cond, @packet, @done
@packet
	%hp =l add %src, %i
	%h =w loadsb %hp
	%hl =l extsw %h
	%i1 =l add %i, %one
	%isneg =w csltl %hl, %z
	jnz %isneg, @neg, @lit
@lit
	%lit_len =l add %hl, %one
	%lit_end =l add %i1, %lit_len
	%lit_under =w csltl %n, %lit_end
	jnz %lit_under, @errsrc, @lit_ok
@lit_ok
	%lit_oend =l add %o, %lit_len
	%lit_over =w csltl %cap, %lit_oend
	jnz %lit_over, @lit_ov, @lit_full
@lit_ov
	%lit_rem =l sub %cap, %o
	%lit_ov_sp =l add %src, %i1
	%lit_ov_dp =l add %out, %o
	jmp @lit_ov_loop(%z, %lit_ov_sp, %lit_ov_dp, %lit_rem)
@lit_ov_loop(l %lok, l %losp, l %lodp, l %lorem)
	%lok_done =w ceql %lok, %lorem
	jnz %lok_done, @overdone, @lit_ov_body
@lit_ov_body
	%lob =w loadub %losp
	storeb %lob, %lodp
	%lok1 =l add %lok, %one
	%losp1 =l add %losp, %one
	%lodp1 =l add %lodp, %one
	jmp @lit_ov_loop(%lok1, %losp1, %lodp1, %lorem)
@lit_full
	%lf_sp =l add %src, %i1
	%lf_dp =l add %out, %o
	jmp @lit_full_loop(%z, %lf_sp, %lf_dp)
@lit_full_loop(l %lfk, l %lfsp, l %lfdp)
	%lf_done =w ceql %lfk, %lit_len
	jnz %lf_done, @lit_full_done, @lit_full_body
@lit_full_body
	%lfb =w loadub %lfsp
	storeb %lfb, %lfdp
	%lfk1 =l add %lfk, %one
	%lfsp1 =l add %lfsp, %one
	%lfdp1 =l add %lfdp, %one
	jmp @lit_full_loop(%lfk1, %lfsp1, %lfdp1)
@lit_full_done
	%lit_i2 =l add %i1, %lit_len
	%lit_o2 =l add %o, %lit_len
	jmp @loop(%lit_i2, %lit_o2)
@neg
	%isnoop =w ceql %hl, %m128
	jnz %isnoop, @noop, @rep
@noop
	jmp @loop(%i1, %o)
@rep
	%rep_need =l add %i1, %one
	%rep_under =w csltl %n, %rep_need
	jnz %rep_under, @errsrc, @rep_ok
@rep_ok
	%rep_vp =l add %src, %i1
	%rep_val =w loadub %rep_vp
	%rep_len =l sub %one, %hl
	%rep_oend =l add %o, %rep_len
	%rep_over =w csltl %cap, %rep_oend
	jnz %rep_over, @rep_ov, @rep_full
@rep_ov
	%rep_rem =l sub %cap, %o
	%rep_ov_dp =l add %out, %o
	jmp @rep_ov_loop(%z, %rep_ov_dp, %rep_rem)
@rep_ov_loop(l %rok, l %rodp, l %rorem)
	%rok_done =w ceql %rok, %rorem
	jnz %rok_done, @overdone, @rep_ov_body
@rep_ov_body
	storeb %rep_val, %rodp
	%rok1 =l add %rok, %one
	%rodp1 =l add %rodp, %one
	jmp @rep_ov_loop(%rok1, %rodp1, %rorem)
@rep_full
	%rf_dp =l add %out, %o
	jmp @rep_full_loop(%z, %rf_dp)
@rep_full_loop(l %rfk, l %rfdp)
	%rf_done =w ceql %rfk, %rep_len
	jnz %rf_done, @rep_full_done, @rep_full_body
@rep_full_body
	storeb %rep_val, %rfdp
	%rfk1 =l add %rfk, %one
	%rfdp1 =l add %rfdp, %one
	jmp @rep_full_loop(%rfk1, %rfdp1)
@rep_full_done
	%rep_i2 =l add %i1, %one
	%rep_o2 =l add %o, %rep_len
	jmp @loop(%rep_i2, %rep_o2)
@errsrc
	ret %m2
@overdone
	ret %m1
@done
	ret %o
}
