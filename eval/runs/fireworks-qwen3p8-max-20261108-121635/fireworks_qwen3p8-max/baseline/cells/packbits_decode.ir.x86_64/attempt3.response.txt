

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%isp =l alloc8 8
	%osp =l alloc8 8
	%hslot =l alloc8 8
	%i1slot =l alloc8 8
	%lenslot =l alloc8 8
	%spslot =l alloc8 8
	%dpslot =l alloc8 8
	%cntslot =l alloc8 8
	%valslot =l alloc8 8
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%c128 =l copy 128
	%m1 =l sub %zero, %one
	%m2 =l sub %zero, %two
	%m128 =l sub %zero, %c128
	storel %zero, %isp
	storel %zero, %osp
	jmp @loop
@loop
	%i_loop =l loadl %isp
	%loop_ok =w csltl %i_loop, %n
	jnz %loop_ok, @packet, @done
@packet
	%i_pkt =l loadl %isp
	%hp =l add %src, %i_pkt
	%h =w loadsb %hp
	%hl =l extsw %h
	storel %hl, %hslot
	%i1_pkt =l add %i_pkt, %one
	storel %i1_pkt, %i1slot
	%isneg =w csltl %hl, %zero
	jnz %isneg, @neg, @lit
@lit
	%hl_lit =l loadl %hslot
	%i1_lit =l loadl %i1slot
	%lit_len =l add %hl_lit, %one
	storel %lit_len, %lenslot
	%lit_end =l add %i1_lit, %lit_len
	%lit_bad =w csltl %n, %lit_end
	jnz %lit_bad, @errsrc, @lit_ok
@lit_ok
	%len_ok =l loadl %lenslot
	%o_ok =l loadl %osp
	%oend_ok =l add %o_ok, %len_ok
	%lit_over =w csltl %cap, %oend_ok
	jnz %lit_over, @lit_ov, @lit_full
@lit_ov
	%o_ov =l loadl %osp
	%rem_ov =l sub %cap, %o_ov
	storel %rem_ov, %cntslot
	%i1_ov =l loadl %i1slot
	%sp_ov =l add %src, %i1_ov
	storel %sp_ov, %spslot
	%dp_ov =l add %out, %o_ov
	storel %dp_ov, %dpslot
	jmp @copy_ov
@copy_ov
	%c_cov =l loadl %cntslot
	%cov_done =w ceql %c_cov, %zero
	jnz %cov_done, @overdone, @copy_ov_body
@copy_ov_body
	%sp_covb =l loadl %spslot
	%dp_covb =l loadl %dpslot
	%cnt_covb =l loadl %cntslot
	%b_covb =w loadub %sp_covb
	storeb %b_covb, %dp_covb
	%sp_covb1 =l add %sp_covb, %one
	%dp_covb1 =l add %dp_covb, %one
	%cnt_covb1 =l sub %cnt_covb, %one
	storel %sp_covb1, %spslot
	storel %dp_covb1, %dpslot
	storel %cnt_covb1, %cntslot
	jmp @copy_ov
@lit_full
	%o_lf =l loadl %osp
	%len_lf =l loadl %lenslot
	%i1_lf =l loadl %i1slot
	%i2_lf =l add %i1_lf, %len_lf
	%o2_lf =l add %o_lf, %len_lf
	storel %i2_lf, %isp
	storel %o2_lf, %osp
	%sp_lf =l add %src, %i1_lf
	storel %sp_lf, %spslot
	%dp_lf =l add %out, %o_lf
	storel %dp_lf, %dpslot
	storel %len_lf, %cntslot
	jmp @copy_full
@copy_full
	%c_cf =l loadl %cntslot
	%cf_done =w ceql %c_cf, %zero
	jnz %cf_done, @loop, @copy_full_body
@copy_full_body
	%sp_cfb =l loadl %spslot
	%dp_cfb =l loadl %dpslot
	%cnt_cfb =l loadl %cntslot
	%b_cfb =w loadub %sp_cfb
	storeb %b_cfb, %dp_cfb
	%sp_cfb1 =l add %sp_cfb, %one
	%dp_cfb1 =l add %dp_cfb, %one
	%cnt_cfb1 =l sub %cnt_cfb, %one
	storel %sp_cfb1, %spslot
	storel %dp_cfb1, %dpslot
	storel %cnt_cfb1, %cntslot
	jmp @copy_full
@neg
	%hl_neg =l loadl %hslot
	%isnoop =w ceql %hl_neg, %m128
	jnz %isnoop, @noop, @rep
@noop
	%i1_noop =l loadl %i1slot
	storel %i1_noop, %isp
	jmp @loop
@rep
	%i1_rep =l loadl %i1slot
	%rep_need =l add %i1_rep, %one
	%rep_bad =w csltl %n, %rep_need
	jnz %rep_bad, @errsrc, @rep_ok
@rep_ok
	%i1_rok =l loadl %i1slot
	%vp_rok =l add %src, %i1_rok
	%val_rok =w loadub %vp_rok
	storeb %val_rok, %valslot
	%hl_rok =l loadl %hslot
	%rep_len =l sub %one, %hl_rok
	storel %rep_len, %lenslot
	%o_rok =l loadl %osp
	%oend_rok =l add %o_rok, %rep_len
	%rep_over =w csltl %cap, %oend_rok
	jnz %rep_over, @rep_ov, @rep_full
@rep_ov
	%o_rov =l loadl %osp
	%rem_rov =l sub %cap, %o_rov
	storel %rem_rov, %cntslot
	%dp_rov =l add %out, %o_rov
	storel %dp_rov, %dpslot
	jmp @fill_ov
@fill_ov
	%c_fov =l loadl %cntslot
	%fov_done =w ceql %c_fov, %zero
	jnz %fov_done, @overdone, @fill_ov_body
@fill_ov_body
	%dp_fovb =l loadl %dpslot
	%cnt_fovb =l loadl %cntslot
	%b_fovb =w loadub %valslot
	storeb %b_fovb, %dp_fovb
	%dp_fovb1 =l add %dp_fovb, %one
	%cnt_fovb1 =l sub %cnt_fovb, %one
	storel %dp_fovb1, %dpslot
	storel %cnt_fovb1, %cntslot
	jmp @fill_ov
@rep_full
	%o_rf =l loadl %osp
	%len_rf =l loadl %lenslot
	%i1_rf =l loadl %i1slot
	%i2_rf =l add %i1_rf, %one
	%o2_rf =l add %o_rf, %len_rf
	storel %i2_rf, %isp
	storel %o2_rf, %osp
	%dp_rf =l add %out, %o_rf
	storel %dp_rf, %dpslot
	storel %len_rf, %cntslot
	jmp @fill_full
@fill_full
	%c_ff =l loadl %cntslot
	%ff_done =w ceql %c_ff, %zero
	jnz %ff_done, @loop, @fill_full_body
@fill_full_body
	%dp_ffb =l loadl %dpslot
	%cnt_ffb =l loadl %cntslot
	%b_ffb =w loadub %valslot
	storeb %b_ffb, %dp_ffb
	%dp_ffb1 =l add %dp_ffb, %one
	%cnt_ffb1 =l sub %cnt_ffb, %one
	storel %dp_ffb1, %dpslot
	storel %cnt_ffb1, %cntslot
	jmp @fill_full
@errsrc
	ret %m2
@overdone
	ret %m1
@done
	%o_done =l loadl %osp
	ret %o_done
}
