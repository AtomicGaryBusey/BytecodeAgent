

export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %tmp_elems =l mul %nn, 2
    jmp @id_head
@id_head
    %id_i =l phi @start 0, @id_inc %id_i1
    %id_c =w csltl %id_i, %n
    jnz %id_c, @id_i_body, @mul_head
@id_i_body
    jmp @idj_head
@idj_head
    %id_j =l phi @id_i_body 0, @idj_inc %id_j1
    %id_jc =w csltl %id_j, %n
    jnz %id_jc, @idj_body, @id_inc
@idj_body
    %id_in =l mul %id_i, %n
    %id_idx =l add %id_in, %id_j
    %id_off =l mul %id_idx, 8
    %id_p =l add %out, %id_off
    %id_eq =w ceql %id_i, %id_j
    jnz %id_eq, @id_one, @id_zero
@id_one
    storel 1, %id_p
    jmp @idj_inc
@id_zero
    storel 0, %id_p
    jmp @idj_inc
@idj_inc
    %id_j1 =l add %id_j, 1
    jmp @idj_head
@id_inc
    %id_i1 =l add %id_i, 1
    jmp @id_head
@mul_head
    %cnt =l phi @id_head 0, @mul_inc %cnt1
    %cnt_c =w csltl %cnt, %k
    jnz %cnt_c, @mul_body, @zero_head
@mul_body
    jmp @mi_head
@mi_head
    %mi =l phi @mul_body 0, @mi_inc %mi1
    %mi_c =w csltl %mi, %n
    jnz %mi_c, @mi_body, @copy_head
@mi_body
    jmp @mj_head
@mj_head
    %mj =l phi @mi_body 0, @mj_inc %mj1
    %mj_c =w csltl %mj, %n
    jnz %mj_c, @mj_body, @mi_inc
@mj_body
    jmp @mt_head
@mt_head
    %mt =l phi @mj_body 0, @mt_body %mt1
    %sum =l phi @mj_body 0, @mt_body %sum1
    %mt_c =w csltl %mt, %n
    jnz %mt_c, @mt_body, @mt_done
@mt_body
    %mt_in =l mul %mi, %n
    %mt_idx1 =l add %mt_in, %mt
    %mt_off1 =l mul %mt_idx1, 8
    %mt_p1 =l add %out, %mt_off1
    %mt_x =l loadl %mt_p1
    %mt_tn =l mul %mt, %n
    %mt_idx2 =l add %mt_tn, %mj
    %mt_off2 =l mul %mt_idx2, 8
    %mt_p2 =l add %a, %mt_off2
    %mt_y =l loadl %mt_p2
    %mt_prod =l mul %mt_x, %mt_y
    %sum1 =l add %sum, %mt_prod
    %mt1 =l add %mt, 1
    jmp @mt_head
@mt_done
    %md_in =l mul %mi, %n
    %md_idx =l add %md_in, %mj
    %md_off =l mul %md_idx, 8
    %md_p =l add %tmp, %md_off
    storel %sum, %md_p
    jmp @mj_inc
@mj_inc
    %mj1 =l add %mj, 1
    jmp @mj_head
@mi_inc
    %mi1 =l add %mi, 1
    jmp @mi_head
@copy_head
    %cp_idx =l phi @mi_head 0, @copy_body %cp_idx1
    %cp_c =w csltl %cp_idx, %nn
    jnz %cp_c, @copy_body, @mul_inc
@copy_body
    %cp_off =l mul %cp_idx, 8
    %cp_pt =l add %tmp, %cp_off
    %cp_po =l add %out, %cp_off
    %cp_v =l loadl %cp_pt
    storel %cp_v, %cp_po
    %cp_idx1 =l add %cp_idx, 1
    jmp @copy_head
@mul_inc
    %cnt1 =l add %cnt, 1
    jmp @mul_head
@zero_head
    %z_idx =l phi @mul_head 0, @zero_body %z_idx1
    %z_c =w csltl %z_idx, %tmp_elems
    jnz %z_c, @zero_body, @end
@zero_body
    %z_off =l mul %z_idx, 8
    %z_p =l add %tmp, %z_off
    storel 0, %z_p
    %z_idx1 =l add %z_idx, 1
    jmp @zero_head
@end
    ret 0
}
