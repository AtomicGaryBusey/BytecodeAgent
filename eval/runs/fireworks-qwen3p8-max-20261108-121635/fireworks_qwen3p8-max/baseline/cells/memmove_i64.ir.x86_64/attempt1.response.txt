

export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w csltl %n, 1
    jnz %empty, @ret, @cmp
@cmp
    %eq =w ceql %dst, %src
    jnz %eq, @ret, @dir
@dir
    %fwd =w csltl %dst, %src
    jnz %fwd, @fwd_init, @bwd_init
@fwd_init
    %i0 =l copy 0
    jmp @fwd_header
@fwd_header
    %i =l phi @fwd_init %i0, @fwd_body %inext
    %cont =w csltl %i, %n
    jnz %cont, @fwd_body, @ret
@fwd_body
    %off =l mul %i, 8
    %sp =l add %src, %off
    %v =l loadl %sp
    %dp =l add %dst, %off
    storel %v, %dp
    %inext =l add %i, 1
    jmp @fwd_header
@bwd_init
    %last =l sub %n, 1
    jmp @bwd_header
@bwd_header
    %j =l phi @bwd_init %last, @bwd_body %jnext
    %done =w csltl %j, 0
    jnz %done, @ret, @bwd_body
@bwd_body
    %offb =l mul %j, 8
    %spb =l add %src, %offb
    %vb =l loadl %spb
    %dpb =l add %dst, %offb
    storel %vb, %dpb
    %jnext =l sub %j, 1
    jmp @bwd_header
@ret
    %r =l copy 0
    ret %r
}
