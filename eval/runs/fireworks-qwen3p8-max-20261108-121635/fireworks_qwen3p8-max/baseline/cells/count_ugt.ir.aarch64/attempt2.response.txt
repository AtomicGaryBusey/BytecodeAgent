

export function l $count_ugt(l %a, l %n, l %t) {
@start
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %a0 =l copy %a
    %n0 =l copy %n
    %t0 =l copy %t
    jmp @head
@head
    %i =l phi @start %zero, @next %i1
    %p =l phi @start %a0, @next %p1
    %c =l phi @start %zero, @next %c2
    %cond =w csltl %i, %n0
    jnz %cond, @body, @exit
@body
    %x =l loadl %p
    %gt =w cugtl %x, %t0
    %cbody =l copy %c
    jnz %gt, @incr, @next
@incr
    %c1 =l add %c, %one
    jmp @next
@next
    %c2 =l phi @body %cbody, @incr %c1
    %i1 =l add %i, %one
    %p1 =l add %p, %eight
    jmp @head
@exit
    ret %c
}
