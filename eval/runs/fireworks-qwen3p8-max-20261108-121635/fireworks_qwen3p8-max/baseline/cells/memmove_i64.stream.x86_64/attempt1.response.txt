

memmove_i64:
    cld
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
    mov rcx, rdx
.Lback:
    mov r8, qword ptr [rsi + rcx*8 - 8]
    mov qword ptr [rdi + rcx*8 - 8], r8
    dec rcx
    jnz .Lback
    jmp .Ldone
.Lforward:
    xor ecx, ecx
.Lfwd:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jb .Lfwd
.Ldone:
    ret
