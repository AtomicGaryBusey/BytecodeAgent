

export function l $qsort_pairs(l %a, l %n) {
@entry
    %zero =l sub %n, %n
    %one =l add %zero, 1
    %two =l add %one, 1
    %eight =l add %zero, 8
    %sixteen =l add %zero, 16
    %small =w csltl %n, %two
    jnz %small, @ret0, @part
@ret0
    ret l 0
@part
    %z =l add %zero, %zero
    %last =l sub %n, %one
    %poff =l mul %last, %sixteen
    %paddr =l add %a, %poff
    %pkey =l loadl %paddr
    %pvaddr =l add %paddr, %eight
    %pval =l loadl %pvaddr
    jmp @loop
@loop
    %i =l phi @part %z, @swap %inext_s, @noswap %inext_n
    %s =l phi @part %z, @swap %snext, @noswap %s_keep
    %cont =w csltl %i, %last
    jnz %cont, @body, @finish
@body
    %ioff =l mul %i, %sixteen
    %iaddr =l add %a, %ioff
    %ivaddr =l add %iaddr, %eight
    %ikey =l loadl %iaddr
    %ival =l loadl %ivaddr
    %klt =w csltl %ikey, %pkey
    %keq =w ceql %ikey, %pkey
    %vlt =w csltl %ival, %pval
    %eqv =w and %keq, %vlt
    %less =w or %klt, %eqv
    jnz %less, @swap, @noswap
@swap
    %soff =l mul %s, %sixteen
    %saddr =l add %a, %soff
    %svaddr =l add %saddr, %eight
    %skey =l loadl %saddr
    %sval =l loadl %svaddr
    storel %skey, %iaddr
    storel %sval, %ivaddr
    storel %ikey, %saddr
    storel %ival, %svaddr
    %inext_s =l add %i, %one
    %snext =l add %s, %one
    jmp @loop
@noswap
    %inext_n =l add %i, %one
    %s_keep =l add %s, %zero
    jmp @loop
@finish
    %sfoff =l mul %s, %sixteen
    %sfaddr =l add %a, %sfoff
    %sfvaddr =l add %sfaddr, %eight
    %sfkey =l loadl %sfaddr
    %sfval =l loadl %sfvaddr
    storel %pkey, %sfaddr
    storel %pval, %sfvaddr
    storel %sfkey, %paddr
    storel %sfval, %pvaddr
    %s1 =l add %s, %one
    %rightoff =l mul %s1, %sixteen
    %righta =l add %a, %rightoff
    %rightn =l sub %last, %s
    %left_pos =w csltl %zero, %s
    jnz %left_pos, @callleft, @checkright
@callleft
    %r1 =l call $qsort_pairs(l %a, l %s)
    jmp @checkright
@checkright
    %right_pos =w csltl %zero, %rightn
    jnz %right_pos, @callright, @ret0b
@callright
    %r2 =l call $qsort_pairs(l %righta, l %rightn)
    jmp @ret0b
@ret0b
    ret l 0
}
