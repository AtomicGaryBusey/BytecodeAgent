

isqrt:
    mov eax, 0xffffffff
    cmp rdi, rax
    cmovb rax, rdi
    xor edx, edx
.Lloop:
    cmp rdx, rax
    jae .Ldone
    lea rcx, [rdx + rax + 1]
    shr rcx, 1
    mov rsi, rcx
    imul rsi, rcx
    cmp rsi, rdi
    ja .Lhigh
    mov rdx, rcx
    jmp .Lloop
.Lhigh:
    lea rax, [rcx - 1]
    jmp .Lloop
.Ldone:
    mov rax, rdx
    ret
