

export function l $fill_down(l %a, l %n, l %start) {
@start
    %i =l copy 0
    jmp @loop
@loop
    %c =w csltl %i, %n
    jnz %c, @body, @end
@body
    %o =l mul %i, 8
    %p =l add %a, %o
    %v =l sub %start, %i
    storel %v, %p
    %i1 =l add %i, 1
    %i =l copy %i1
    jmp @loop
@end
    ret %n
}
