

export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %ni
    %s =l phi @start 0, @body %ns
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %x =l loadl %ptr
    %r =l call %f(l %x)
    %ns =l add %s, %r
    %ni =l add %i, 1
    jmp @loop
@done
    ret %s
}
