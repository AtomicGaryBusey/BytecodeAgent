

export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%p = phi @start %a, @inc %pnext
	%rem = phi @start %n, @inc %remnext
	%empty = ceql %rem, 0
	%res_empty = subl %n, %rem
	jnz %empty, @ret_empty, @body
@body
	%v = loadl %p
	%zero = ceql %v, 0
	%res_zero = subl %n, %rem
	jnz %zero, @ret_zero, @inc
@inc
	%pnext = addl %p, 8
	%remnext = subl %rem, 1
	jmp @loop
@ret_empty
	retl %res_empty
@ret_zero
	retl %res_zero
}
