

export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%zero =l sub %n, %n
	%empty =w ceql %n, %zero
	jnz %empty, @done, @loop
@loop
	%p =l phi @start %a, @next %pnext
	%i =l phi @start %zero, @next %inext
	%v =l loadl %p
	%iszero =w ceql %v, %zero
	jnz %iszero, @found, @next
@next
	%pnext =l add %p, 8
	%inext =l add %i, 1
	%cont =w csltl %inext, %n
	jnz %cont, @loop, @donei
@found
	retl %i
@donei
	retl %inext
@done
	retl %n
}
