

packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx rax, byte ptr [rdi + r8]
    inc r8
    cmp rax, -128
    je .Lloop
    test rax, rax
    js .Lrepeat
    lea r10, [rax + 1]
    mov r11, rsi
    sub r11, r8
    cmp r10, r11
    jg .Lsrc_error
    mov r11, rcx
    sub r11, r9
    cmp r10, r11
    jg .Lliteral_overflow
.Lliteral_copy:
    test r10, r10
    je .Lloop
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jmp .Lliteral_copy
.Lrepeat:
    cmp r8, rsi
    jge .Lsrc_error
    movzx r10d, byte ptr [rdi + r8]
    inc r8
    mov r11, 1
    sub r11, rax
    mov rax, rcx
    sub rax, r9
    cmp r11, rax
    jg .Lrepeat_overflow
.Lrepeat_copy:
    test r11, r11
    je .Lloop
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jmp .Lrepeat_copy
.Lliteral_overflow:
    test r11, r11
    jle .Loverflow_ret
.Lliteral_overflow_copy:
    test r11, r11
    je .Loverflow_ret
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jmp .Lliteral_overflow_copy
.Lrepeat_overflow:
    test rax, rax
    jle .Loverflow_ret
.Lrepeat_overflow_copy:
    test rax, rax
    je .Loverflow_ret
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec rax
    jmp .Lrepeat_overflow_copy
.Loverflow_ret:
    mov rax, -1
    ret
.Lsrc_error:
    mov rax, -2
    ret
.Ldone:
    mov rax, r9
    ret
