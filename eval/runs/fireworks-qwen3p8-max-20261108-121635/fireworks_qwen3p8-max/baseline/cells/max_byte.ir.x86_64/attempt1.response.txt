

export function l $max_byte(l %p, l %n) {
@start
	%zero =l copy 0
	jmp @head
@head
	%i =l phi @start %zero, @step %i1
	%m =l phi @start %zero, @step %m1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%less =w csltl %m, %b
	jnz %less, @take, @keep
@take
	jmp @step
@keep
	jmp @step
@step
	%m1 =l phi @take %b, @keep %m
	%i1 =l add %i, 1
	jmp @head
@done
	ret %m
}
