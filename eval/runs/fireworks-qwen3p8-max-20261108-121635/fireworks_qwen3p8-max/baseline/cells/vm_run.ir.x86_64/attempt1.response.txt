

export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
  %pcptr =l alloc8 16
  %spptr =l add %pcptr, 8
  storel 0, %pcptr
  storel 0, %spptr
  jmp @loop
@loop
  %pc =l loadl %pcptr
  %sp =l loadl %spptr
  %cont =w csltl %pc, %len
  jnz %cont, @fetch, @finish
@fetch
  %coff =l mul %pc, 8
  %caddr =l add %code, %coff
  %op =l loadl %caddr
  %is0 =w ceql %op, 0
  jnz %is0, @do_halt, @chk1
@chk1
  %is1 =w ceql %op, 1
  jnz %is1, @do_push, @chk2
@chk2
  %is2 =w ceql %op, 2
  jnz %is2, @do_add, @chk3
@chk3
  %is3 =w ceql %op, 3
  jnz %is3, @do_sub, @chk4
@chk4
  %is4 =w ceql %op, 4
  jnz %is4, @do_mul, @chk5
@chk5
  %is5 =w ceql %op, 5
  jnz %is5, @do_dup, @chk6
@chk6
  %is6 =w ceql %op, 6
  jnz %is6, @do_swap, @chk7
@chk7
  %is7 =w ceql %op, 7
  jnz %is7, @do_drop, @chk8
@chk8
  %is8 =w ceql %op, 8
  jnz %is8, @do_over, @chk9
@chk9
  %is9 =w ceql %op, 9
  jnz %is9, @do_lt, @chk10
@chk10
  %is10 =w ceql %op, 10
  jnz %is10, @do_jz, @chk11
@chk11
  %is11 =w ceql %op, 11
  jnz %is11, @do_jmp, @do_default
@do_halt
  jmp @finish
@do_push
  %opaddr_push =l add %caddr, 8
  %val_push =l loadl %opaddr_push
  %off_push =l mul %sp, 8
  %dst_push =l add %stack, %off_push
  storel %val_push, %dst_push
  %sp_push =l add %sp, 1
  storel %sp_push, %spptr
  %pc_push =l add %pc, 2
  storel %pc_push, %pcptr
  jmp @loop
@do_add
  %spm2_add =l sub %sp, 2
  %offa_add =l mul %spm2_add, 8
  %addra_add =l add %stack, %offa_add
  %a_add =l loadl %addra_add
  %spm1_add =l sub %sp, 1
  %offb_add =l mul %spm1_add, 8
  %addrb_add =l add %stack, %offb_add
  %b_add =l loadl %addrb_add
  %res_add =l add %a_add, %b_add
  storel %res_add, %addra_add
  %sp_add =l sub %sp, 1
  storel %sp_add, %spptr
  %pc_add =l add %pc, 1
  storel %pc_add, %pcptr
  jmp @loop
@do_sub
  %spm2_sub =l sub %sp, 2
  %offa_sub =l mul %spm2_sub, 8
  %addra_sub =l add %stack, %offa_sub
  %a_sub =l loadl %addra_sub
  %spm1_sub =l sub %sp, 1
  %offb_sub =l mul %spm1_sub, 8
  %addrb_sub =l add %stack, %offb_sub
  %b_sub =l loadl %addrb_sub
  %res_sub =l sub %a_sub, %b_sub
  storel %res_sub, %addra_sub
  %sp_sub =l sub %sp, 1
  storel %sp_sub, %spptr
  %pc_sub =l add %pc, 1
  storel %pc_sub, %pcptr
  jmp @loop
@do_mul
  %spm2_mul =l sub %sp, 2
  %offa_mul =l mul %spm2_mul, 8
  %addra_mul =l add %stack, %offa_mul
  %a_mul =l loadl %addra_mul
  %spm1_mul =l sub %sp, 1
  %offb_mul =l mul %spm1_mul, 8
  %addrb_mul =l add %stack, %offb_mul
  %b_mul =l loadl %addrb_mul
  %res_mul =l mul %a_mul, %b_mul
  storel %res_mul, %addra_mul
  %sp_mul =l sub %sp, 1
  storel %sp_mul, %spptr
  %pc_mul =l add %pc, 1
  storel %pc_mul, %pcptr
  jmp @loop
@do_dup
  %spm1_dup =l sub %sp, 1
  %off_dup =l mul %spm1_dup, 8
  %addr_dup =l add %stack, %off_dup
  %val_dup =l loadl %addr_dup
  %offnew_dup =l mul %sp, 8
  %dstnew_dup =l add %stack, %offnew_dup
  storel %val_dup, %dstnew_dup
  %sp_dup =l add %sp, 1
  storel %sp_dup, %spptr
  %pc_dup =l add %pc, 1
  storel %pc_dup, %pcptr
  jmp @loop
@do_swap
  %spm2_swap =l sub %sp, 2
  %offa_swap =l mul %spm2_swap, 8
  %addra_swap =l add %stack, %offa_swap
  %a_swap =l loadl %addra_swap
  %spm1_swap =l sub %sp, 1
  %offb_swap =l mul %spm1_swap, 8
  %addrb_swap =l add %stack, %offb_swap
  %b_swap =l loadl %addrb_swap
  storel %b_swap, %addra_swap
  storel %a_swap, %addrb_swap
  %pc_swap =l add %pc, 1
  storel %pc_swap, %pcptr
  jmp @loop
@do_drop
  %sp_drop =l sub %sp, 1
  storel %sp_drop, %spptr
  %pc_drop =l add %pc, 1
  storel %pc_drop, %pcptr
  jmp @loop
@do_over
  %spm2_over =l sub %sp, 2
  %off_over =l mul %spm2_over, 8
  %addr_over =l add %stack, %off_over
  %val_over =l loadl %addr_over
  %offnew_over =l mul %sp, 8
  %dstnew_over =l add %stack, %offnew_over
  storel %val_over, %dstnew_over
  %sp_over =l add %sp, 1
  storel %sp_over, %spptr
  %pc_over =l add %pc, 1
  storel %pc_over, %pcptr
  jmp @loop
@do_lt
  %spm2_lt =l sub %sp, 2
  %offa_lt =l mul %spm2_lt, 8
  %addra_lt =l add %stack, %offa_lt
  %a_lt =l loadl %addra_lt
  %spm1_lt =l sub %sp, 1
  %offb_lt =l mul %spm1_lt, 8
  %addrb_lt =l add %stack, %offb_lt
  %b_lt =l loadl %addrb_lt
  %cmp_lt =w csltl %a_lt, %b_lt
  jnz %cmp_lt, @lt_true, @lt_false
@lt_true
  storel 1, %addra_lt
  %sp_lt_true =l sub %sp, 1
  storel %sp_lt_true, %spptr
  %pc_lt_true =l add %pc, 1
  storel %pc_lt_true, %pcptr
  jmp @loop
@lt_false
  storel 0, %addra_lt
  %sp_lt_false =l sub %sp, 1
  storel %sp_lt_false, %spptr
  %pc_lt_false =l add %pc, 1
  storel %pc_lt_false, %pcptr
  jmp @loop
@do_jz
  %spm1_jz =l sub %sp, 1
  %off_jz =l mul %spm1_jz, 8
  %addr_jz =l add %stack, %off_jz
  %v_jz =l loadl %addr_jz
  storel %spm1_jz, %spptr
  %iszero_jz =w ceql %v_jz, 0
  jnz %iszero_jz, @jz_take, @jz_skip
@jz_take
  %opaddr_jz =l add %caddr, 8
  %t_jz =l loadl %opaddr_jz
  storel %t_jz, %pcptr
  jmp @loop
@jz_skip
  %pc_jz_skip =l add %pc, 2
  storel %pc_jz_skip, %pcptr
  jmp @loop
@do_jmp
  %opaddr_jmp =l add %caddr, 8
  %t_jmp =l loadl %opaddr_jmp
  storel %t_jmp, %pcptr
  jmp @loop
@do_default
  %pc_default =l add %pc, 1
  storel %pc_default, %pcptr
  jmp @loop
@finish
  %sp_fin =l loadl %spptr
  storel %sp_fin, %pcptr
  jmp @zero_check
@zero_check
  %idx =l loadl %pcptr
  %more =w csltl %idx, %cap
  jnz %more, @zero_body, @ret_check
@zero_body
  %off_zero =l mul %idx, 8
  %addr_zero =l add %stack, %off_zero
  storel 0, %addr_zero
  %idx_next =l add %idx, 1
  storel %idx_next, %pcptr
  jmp @zero_check
@ret_check
  %sp_ret =l loadl %spptr
  %nonempty =w csltl 0, %sp_ret
  jnz %nonempty, @ret_top, @ret_zero
@ret_top
  %spm1_ret =l sub %sp_ret, 1
  %off_ret =l mul %spm1_ret, 8
  %addr_ret =l add %stack, %off_ret
  %top =l loadl %addr_ret
  ret %top
@ret_zero
  ret 0
}
