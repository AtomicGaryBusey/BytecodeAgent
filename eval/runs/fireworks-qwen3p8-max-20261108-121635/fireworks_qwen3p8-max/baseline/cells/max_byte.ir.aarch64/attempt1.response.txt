

export function l $max_byte(l %p, l %n) {
@entry
	%p0 =l add %p, 0
	%n0 =l add %n, 0
	%empty =w csltl %n0, 1
	jnz %empty, @return0, @loop
@return0
	ret 0
@loop
	%ptr =l phi @entry %p0, @step %ptr_next
	%rem =l phi @entry %n0, @step %rem_next
	%best =l phi @entry 0, @step %best_next
	%byte =l loadub %ptr
	%less =w csltl %best, %byte
	jnz %less, @take, @keep
@take
	%best_take =l add %byte, 0
	jmp @step
@keep
	%best_keep =l add %best, 0
	jmp @step
@step
	%best_next =l phi @take %best_take, @keep %best_keep
	%ptr_next =l add %ptr, 1
	%rem_next =l sub %rem, 1
	%done =w csltl %rem_next, 1
	jnz %done, @returnmax, @loop
@returnmax
	ret %best_next
}
