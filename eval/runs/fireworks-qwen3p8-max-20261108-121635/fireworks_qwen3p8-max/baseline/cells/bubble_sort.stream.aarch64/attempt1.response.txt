

bubble_sort:
    subs    x2, x1, #1
    b.le    .Ldone
.Louter:
    mov     x3, x0
    mov     x4, x2
.Linner:
    ldr     x5, [x3]
    ldr     x6, [x3, #8]
    cmp     x5, x6
    b.le    .Lnoswap
    str     x6, [x3]
    str     x5, [x3, #8]
.Lnoswap:
    add     x3, x3, #8
    subs    x4, x4, #1
    b.ne    .Linner
    subs    x2, x2, #1
    b.gt    .Louter
.Ldone:
    mov     x0, #0
    ret
