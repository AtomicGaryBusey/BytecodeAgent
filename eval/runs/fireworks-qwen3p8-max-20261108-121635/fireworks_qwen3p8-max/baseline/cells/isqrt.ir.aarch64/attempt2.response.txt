

export function l $isqrt(l %n) {
@start
    %z =w ceql %n, 0
    jnz %z, @zero, @init
@zero
    %r0 =l add 0, 0
    ret %r0
@init
    %lo0 =l add 0, 1
    %t =l shl %lo0, 32
    %hi0 =l sub %t, 1
    %ans0 =l add 0, 0
    jmp @loop
@loop
    %lo =l phi @init %lo0, @ok %lo1, @not %lo2
    %hi =l phi @init %hi0, @ok %hi1, @not %hi2
    %ans =l phi @init %ans0, @ok %ans1, @not %ans2
    %lth =w csltl %lo, %hi
    %eqh =w ceql %lo, %hi
    %cont =w or %lth, %eqh
    jnz %cont, @body, @done
@body
    %sum =l add %lo, %hi
    %mid =l shr %sum, 1
    %q =l udiv %n, %mid
    %qneg =w csltl %q, 0
    %ltq =w csltl %mid, %q
    %eqq =w ceql %mid, %q
    %leq =w or %ltq, %eqq
    %okc =w or %qneg, %leq
    jnz %okc, @ok, @not
@ok
    %ans1 =l add %mid, 0
    %lo1 =l add %mid, 1
    %hi1 =l add %hi, 0
    jmp @loop
@not
    %ans2 =l add %ans, 0
    %lo2 =l add %lo, 0
    %hi2 =l sub %mid, 1
    jmp @loop
@done
    ret %ans
}
