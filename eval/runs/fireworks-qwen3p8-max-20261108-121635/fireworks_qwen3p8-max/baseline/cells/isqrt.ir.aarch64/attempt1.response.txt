

export function l $isqrt(l %n) {
@start
    %z =w ceql %n, 0
    jnz %z, @zero, @init
@zero
    ret l 0
@init
    jmp @loop
@loop
    %lo =l phi @init 1, @ok %lo1, @not %lo2
    %hi =l phi @init 4294967295, @ok %hi1, @not %hi2
    %ans =l phi @init 0, @ok %ans1, @not %ans2
    %lth =w csltl %lo, %hi
    %eqh =w ceql %lo, %hi
    %cont =w or %lth, %eqh
    jnz %cont, @body, @done
@body
    %sum =l add %lo, %hi
    %mid =l shr %sum, 1
    %q =l udiv %n, %mid
    %min =l shl 1, 63
    %mb =l xor %mid, %min
    %qb =l xor %q, %min
    %ltq =w csltl %mb, %qb
    %eqq =w ceql %mid, %q
    %okc =w or %ltq, %eqq
    jnz %okc, @ok, @not
@ok
    %ans1 =l add %mid, 0
    %lo1 =l add %mid, 1
    %hi1 =l add %hi, 0
    jmp @loop
@not
    %ans2 =l add %ans, 0
    %lo2 =l add %lo, 0
    %hi2 =l sub %mid, 1
    jmp @loop
@done
    ret l %ans
}
