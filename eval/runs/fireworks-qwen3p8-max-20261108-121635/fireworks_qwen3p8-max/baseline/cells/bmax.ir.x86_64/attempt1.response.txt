

export function l $bmax(l %a, l %n) {
@start
    %first = loadl %a
    %p0 = addl %a, 8
    %rem0 = subl %n, 1
    jmp @loop(%p0, %rem0, %first)
@loop(l %p, l %rem, l %max)
    %iszero = ceql %rem, 0
    jnz %iszero, @exit, @body
@body
    %v = loadl %p
    %less = csltl %max, %v
    %p1 = addl %p, 8
    %rem1 = subl %rem, 1
    jnz %less, @take, @keep
@take
    jmp @loop(%p1, %rem1, %v)
@keep
    jmp @loop(%p1, %rem1, %max)
@exit
    retl %max
}
