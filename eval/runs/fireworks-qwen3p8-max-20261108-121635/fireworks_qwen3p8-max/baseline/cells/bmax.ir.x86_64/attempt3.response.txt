

export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    %p0 =l add %a, 8
    %rem0 =l sub %n, 1
    jmp @loop
@loop
    %p =l phi @start %p0, @take %pt, @keep %pk
    %rem =l phi @start %rem0, @take %rt, @keep %rk
    %max =l phi @start %first, @take %vt, @keep %mk
    %done =w ceql %rem, 0
    jnz %done, @exit, @body
@body
    %v =l loadl %p
    %less =w csltl %max, %v
    jnz %less, @take, @keep
@take
    %pt =l add %p, 8
    %rt =l sub %rem, 1
    %vt =l copy %v
    jmp @loop
@keep
    %pk =l add %p, 8
    %rk =l sub %rem, 1
    %mk =l copy %max
    jmp @loop
@exit
    ret %max
}
