

export function l $popcount(l %x) {
@start
	%m1a =l copy 1
	%m1b =l shl %m1a, 2
	%m1c =l or %m1a, %m1b
	%m1d =l shl %m1c, 4
	%m1e =l or %m1c, %m1d
	%m1f =l shl %m1e, 8
	%m1g =l or %m1e, %m1f
	%m1h =l shl %m1g, 16
	%m1i =l or %m1g, %m1h
	%m1j =l shl %m1i, 32
	%m1 =l or %m1i, %m1j
	%m2a =l copy 3
	%m2b =l shl %m2a, 4
	%m2c =l or %m2a, %m2b
	%m2d =l shl %m2c, 8
	%m2e =l or %m2c, %m2d
	%m2f =l shl %m2e, 16
	%m2g =l or %m2e, %m2f
	%m2h =l shl %m2g, 32
	%m2 =l or %m2g, %m2h
	%m4a =l copy 15
	%m4b =l shl %m4a, 8
	%m4c =l or %m4a, %m4b
	%m4d =l shl %m4c, 16
	%m4e =l or %m4c, %m4d
	%m4f =l shl %m4e, 32
	%m4 =l or %m4e, %m4f
	%s1 =l shr %x, 1
	%hi1 =l and %s1, %m1
	%lo1 =l and %x, %m1
	%x1 =l add %lo1, %hi1
	%s2 =l shr %x1, 2
	%hi2 =l and %s2, %m2
	%lo2 =l and %x1, %m2
	%x2 =l add %lo2, %hi2
	%s4 =l shr %x2, 4
	%x3 =l add %x2, %s4
	%x4 =l and %x3, %m4
	%s8 =l shr %x4, 8
	%x5 =l add %x4, %s8
	%s16 =l shr %x5, 16
	%x6 =l add %x5, %s16
	%s32 =l shr %x6, 32
	%x7 =l add %x6, %s32
	%mask =l copy 255
	%r =l and %x7, %mask
	ret %r
}
