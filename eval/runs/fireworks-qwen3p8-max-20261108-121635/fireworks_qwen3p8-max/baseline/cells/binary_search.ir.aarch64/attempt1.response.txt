

export function l $binary_search(l %a, l %n, l %key) {
@start
    %lo0 =l copy 0
    %hi0 =l copy %n
    %ok =w csltl %lo0, %hi0
    jnz %ok, @loop, @notfound
@loop
    %lo =l phi @start %lo0, @less %lo1, @greater %lo2
    %hi =l phi @start %hi0, @less %hi1, @greater %hi2
    %diff =l sub %hi, %lo
    %half =l shr %diff, 1
    %mid =l add %lo, %half
    %off =l mul %mid, 8
    %addr =l add %a, %off
    %val =l loadl %addr
    %eq =w ceql %val, %key
    jnz %eq, @found, @noteq
@noteq
    %lt =w csltl %val, %key
    jnz %lt, @less, @greater
@less
    %lo1 =l add %mid, 1
    %hi1 =l copy %hi
    %cont1 =w csltl %lo1, %hi1
    jnz %cont1, @loop, @notfound
@greater
    %lo2 =l copy %lo
    %hi2 =l copy %mid
    %cont2 =w csltl %lo2, %hi2
    jnz %cont2, @loop, @notfound
@found
    ret %mid
@notfound
    %nf =l copy -1
    ret %nf
}
