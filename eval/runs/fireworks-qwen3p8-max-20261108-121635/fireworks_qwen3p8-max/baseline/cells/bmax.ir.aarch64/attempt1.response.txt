

export function l $bmax(l %a, l %n) {
@start
    %m0 =l loadl %a
    %c0 =l sub %n, 1
    jmp @loop
@loop
    %p =l phi @start %a, @cont %p1
    %m =l phi @start %m0, @cont %m1
    %c =l phi @start %c0, @cont %c1
    %done =w ceql %c, 0
    jnz %done, @exit, @body
@body
    %pnext =l add %p, 8
    %v =l loadl %pnext
    %less =w csltl %m, %v
    jnz %less, @take, @skip
@take
    %mt =l add %v, 0
    jmp @cont
@skip
    %ms =l add %m, 0
    jmp @cont
@cont
    %m1 =l phi @take %mt, @skip %ms
    %p1 =l add %pnext, 0
    %c1 =l sub %c, 1
    jmp @loop
@exit
    ret %m
}
