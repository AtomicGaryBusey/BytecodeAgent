

export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i0 =l copy 0
    %c0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @body %inext
    %c =l phi @start %c0, @body %cnext
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %srcptr =l add %src, %off
    %x =l loadl %srcptr
    %y =l call %f(l %x)
    %dstptr =l add %dst, %off
    storel %y, %dstptr
    %less =w csltl %y, %x
    %inc =l extsw %less
    %cnext =l add %c, %inc
    %inext =l add %i, 1
    jmp @loop
@done
    ret %c
}
