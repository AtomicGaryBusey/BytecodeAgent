

export function l $fib(l %n) {
@start
	%i0 =l copy 0
	%a0 =l copy 0
	%b0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%a =l phi @start %a0, @body %a1
	%b =l phi @start %b0, @body %b1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%sum =l addl %a, %b
	%a1 =l copy %b
	%b1 =l copy %sum
	%i1 =l addl %i, 1
	jmp @loop
@done
	ret %a
}
