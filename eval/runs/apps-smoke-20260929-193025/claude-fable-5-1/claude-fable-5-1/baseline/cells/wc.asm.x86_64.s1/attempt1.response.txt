.text
.globl _start
_start:
    xor r12d, r12d          # bytes
    xor r13d, r13d          # lines
    xor r14d, r14d          # words
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r12, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]
scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r13
1:
    cmp al, 32
    je ws
    sub al, 9
    cmp al, 4
    jbe ws
    test r15d, r15d
    jnz 2f
    inc r14
    mov r15d, 1
2:
    cmp rsi, rdx
    jb scan
    jmp read_loop
ws:
    xor r15d, r15d
    cmp rsi, rdx
    jb scan
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call fmt
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call fmt
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call fmt
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp done

# fmt: rax = value, rdi = destination; returns rdi advanced
fmt:
    sub rsp, 32
    lea rcx, [rsp + 32]
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    lea rdx, [rsp + 32]
    cmp rcx, rdx
    jb 2b
    add rsp, 32
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
