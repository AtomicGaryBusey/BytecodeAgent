.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 8192

.bss
.balign 16
inbuf:
    .zero BUF_SIZE
numbuf:
    .zero 32
outbuf:
    .zero 64

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jle finish
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + inbuf]

scan_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x0A
    jne not_nl
    inc r12
not_nl:
    cmp al, 0x20
    je mark_ws
    cmp al, 0x09
    jb mark_word
    cmp al, 0x0D
    jbe mark_ws

mark_word:
    test r15, r15
    jnz advance
    inc r13
    mov r15d, 1
    jmp advance

mark_ws:
    xor r15d, r15d

advance:
    inc rsi
    dec rcx
    jnz scan_loop
    jmp read_loop

finish:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call append_uint
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall

    mov eax, SYS_exit
    xor edi, edi
    syscall

append_uint:
    lea rsi, [rip + numbuf]
    add rsi, 32
    mov r8, rsi
    mov r9, 10
conv_loop:
    xor edx, edx
    div r9
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz conv_loop
copy_loop:
    cmp rsi, r8
    je append_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp copy_loop
append_done:
    ret
