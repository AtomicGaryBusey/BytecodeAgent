.globl _start

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
tmp:
    .zero 32

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax
    lea rsi, [rip + buf]
    lea rbx, [rsi + rax]
byte_loop:
    cmp rsi, rbx
    jae read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je is_ws
    sub eax, 9
    cmp eax, 4
    jbe is_ws
    test r15, r15
    jnz byte_loop
    inc r13
    mov r15d, 1
    jmp byte_loop
is_ws:
    xor r15d, r15d
    jmp byte_loop

done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = output pointer (advanced)
utoa:
    lea r8, [rip + tmp + 32]
    mov r9, r8
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    cmp r8, r9
    jae 4f
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    jmp 3b
4:
    ret
