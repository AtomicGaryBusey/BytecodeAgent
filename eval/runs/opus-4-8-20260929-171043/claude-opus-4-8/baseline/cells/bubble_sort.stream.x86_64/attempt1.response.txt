bubble_sort:
    cmp rsi, 1
    jle .Ldone
.Louter:
    xor rcx, rcx
    mov r8, rsi
    dec r8
.Linner:
    cmp rcx, r8
    jge .Lafter
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rax, rdx
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rcx*8 + 8], rax
.Lnoswap:
    inc rcx
    jmp .Linner
.Lafter:
    dec rsi
    cmp rsi, 1
    jg .Louter
.Ldone:
    xor rax, rax
    ret
