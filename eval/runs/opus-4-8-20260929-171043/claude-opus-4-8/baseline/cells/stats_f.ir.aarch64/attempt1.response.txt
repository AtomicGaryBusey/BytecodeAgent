export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%pi =l alloc8 8
	%psum =l alloc8 8
	%pcnt =l alloc8 8
	%pmin =l alloc8 8
	%pmax =l alloc8 8
	storel 0, %pi
	storel 0, %psum
	storel 0, %pcnt
	jmp @loop
@loop
	%i =l loadl %pi
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%sum =l loadl %psum
	%sum2 =l add %sum, %y
	storel %sum2, %psum
	%m =l and %y, 1
	%cnt =l loadl %pcnt
	%cnt2 =l add %cnt, %m
	storel %cnt2, %pcnt
	%isfirst =w ceql %i, 0
	jnz %isfirst, @init, @update
@init
	storel %y, %pmin
	storel %y, %pmax
	jmp @next
@update
	%min =l loadl %pmin
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	storel %y, %pmin
	jmp @chkmax
@chkmax
	%max =l loadl %pmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @next
@setmax
	storel %y, %pmax
	jmp @next
@next
	%i2 =l add %i, 1
	storel %i2, %pi
	jmp @loop
@done
	%fmin =l loadl %pmin
	%fmax =l loadl %pmax
	%fsum =l loadl %psum
	storel %fmin, %out
	%o1 =l add %out, 8
	storel %fmax, %o1
	%o2 =l add %out, 16
	storel %fsum, %o2
	%rc =l loadl %pcnt
	ret %rc
}
