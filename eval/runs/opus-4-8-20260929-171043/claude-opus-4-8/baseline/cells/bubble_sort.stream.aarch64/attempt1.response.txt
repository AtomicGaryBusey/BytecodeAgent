bubble_sort:
cmp x1, #2
b.lt .Ldone
mov x9, #0
.Louter:
cmp x9, x1
b.ge .Ldone
mov x10, #0
sub x11, x1, #1
.Linner:
cmp x10, x11
b.ge .Linner_end
ldr x12, [x0, x10, lsl #3]
add x13, x10, #1
ldr x14, [x0, x13, lsl #3]
cmp x12, x14
b.le .Lnoswap
str x14, [x0, x10, lsl #3]
str x12, [x0, x13, lsl #3]
.Lnoswap:
add x10, x10, #1
b .Linner
.Linner_end:
add x9, x9, #1
b .Louter
.Ldone:
mov x0, #0
ret
