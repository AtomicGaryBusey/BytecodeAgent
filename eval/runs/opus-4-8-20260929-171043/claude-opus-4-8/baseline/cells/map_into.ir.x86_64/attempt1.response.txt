export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i =l copy 0
    %cnt =l copy 0
@loop
    %cond =l csltl %i, %n
    jnz %cond, @body, @end
@body
    %off =l mul %i, 8
    %sp =l add %src, %off
    %x =l loadl %sp
    %r =l call %f(l %x)
    %dp =l add %dst, %off
    storel %r, %dp
    %less =w csltl %r, %x
    %lessl =l extuw %less
    %cnt =l add %cnt, %lessl
    %i =l add %i, 1
    jmp @loop
@end
    ret %cnt
}
