memmove_i64:
	test rdx, rdx
	jle .Ldone
	cmp rdi, rsi
	je .Ldone
	jb .Lforward
.Lbackward:
	mov rcx, rdx
.Lbloop:
	dec rcx
	mov rax, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], rax
	test rcx, rcx
	jg .Lbloop
	jmp .Ldone
.Lforward:
	xor rcx, rcx
.Lfloop:
	mov rax, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], rax
	inc rcx
	cmp rcx, rdx
	jl .Lfloop
.Ldone:
	xor rax, rax
	ret
