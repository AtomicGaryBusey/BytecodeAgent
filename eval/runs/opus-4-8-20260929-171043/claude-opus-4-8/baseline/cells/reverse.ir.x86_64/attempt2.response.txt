export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%cond =w csltl %i, %j
	jnz %cond, @body, @end
@body
	%ai =l mul %i, 8
	%pi =l add %a, %ai
	%aj =l mul %j, 8
	%pj =l add %a, %aj
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@end
	ret 0
}
