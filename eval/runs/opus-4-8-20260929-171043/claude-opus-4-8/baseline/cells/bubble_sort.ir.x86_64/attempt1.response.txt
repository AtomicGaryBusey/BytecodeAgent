export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%oc =l csltl %i, %n
	jnz %oc, @outerb, @done
@outerb
	%nm1 =l sub %n, 1
	%lim =l sub %nm1, %i
	%j =l copy 0
@inner
	%ic =l csltl %j, %lim
	jnz %ic, @innerb, @outernext
@innerb
	%p0 =l mul %j, 8
	%addr0 =l add %a, %p0
	%addr1 =l add %addr0, 8
	%v0 =l loadl %addr0
	%v1 =l loadl %addr1
	%sw =w csgtl %v0, %v1
	jnz %sw, @swap, @innernext
@swap
	storel %v1, %addr0
	storel %v0, %addr1
@innernext
	%j =l add %j, 1
	jmp @inner
@outernext
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
