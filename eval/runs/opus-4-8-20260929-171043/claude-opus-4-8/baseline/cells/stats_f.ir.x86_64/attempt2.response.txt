export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%sum =l copy 0
	%cnt =l copy 0
	%first =l loadl %a
	%min =l call %f(l %first)
	%max =l copy %min
	%sum =l add %sum, %min
	%rem =l and %min, 1
	%cnt =l add %cnt, %rem
	%i =l copy 1
@loop
	%cond =l csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum =l add %sum, %y
	%rem2 =l and %y, 1
	%cnt =l add %cnt, %rem2
	%ltmin =l csltl %y, %min
	jnz %ltmin, @setmin, @chkmax
@setmin
	%min =l copy %y
@chkmax
	%gtmax =l csgtl %y, %max
	jnz %gtmax, @setmax, @next
@setmax
	%max =l copy %y
@next
	%i =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
