export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%al =l and %a, %mask
	%ah =l shr %a, 32
	%bl =l and %b, %mask
	%bh =l shr %b, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%llhi =l shr %ll, 32
	%lhlo =l and %lh, %mask
	%hllo =l and %hl, %mask
	%cross =l add %llhi, %lhlo
	%cross2 =l add %cross, %hllo
	%crosshi =l shr %cross2, 32
	%lhhi =l shr %lh, 32
	%hlhi =l shr %hl, 32
	%h1 =l add %hh, %lhhi
	%h2 =l add %h1, %hlhi
	%high =l add %h2, %crosshi
	ret %high
}
