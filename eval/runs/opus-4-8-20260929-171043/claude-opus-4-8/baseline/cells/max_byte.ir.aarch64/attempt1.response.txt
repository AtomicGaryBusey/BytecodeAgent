export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @update, @cont
@update
	%max =l copy %b
@cont
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
