export function l $qsort_pairs(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	%c =w csltl %n, 2
	jnz %c, @ret0, @go
@go
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%pptr =l add %a, %poff
	%pk =l loadl %pptr
	%pvp =l add %pptr, 8
	%pv =l loadl %pvp
	storel -1, %ip
	storel 0, %jp
@loop
	%j =l loadl %jp
	%jlt =w csltl %j, %nm1
	jnz %jlt, @body, @afterloop
@body
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%jk =l loadl %jptr
	%jvp =l add %jptr, 8
	%jv =l loadl %jvp
	%klt =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vlt =w csltl %jv, %pv
	%kv =w and %keq, %vlt
	%less =w or %klt, %kv
	jnz %less, @doswap, @cont
@doswap
	%i0 =l loadl %ip
	%i1 =l add %i0, 1
	storel %i1, %ip
	%ioff =l mul %i1, 16
	%iptr =l add %a, %ioff
	%ak =l loadl %iptr
	%ivp =l add %iptr, 8
	%av =l loadl %ivp
	storel %jk, %iptr
	storel %jv, %ivp
	storel %ak, %jptr
	storel %av, %jvp
	jmp @cont
@cont
	%j2 =l loadl %jp
	%j3 =l add %j2, 1
	storel %j3, %jp
	jmp @loop
@afterloop
	%if =l loadl %ip
	%pi =l add %if, 1
	%pioff =l mul %pi, 16
	%piptr =l add %a, %pioff
	%ak2 =l loadl %piptr
	%pivp2 =l add %piptr, 8
	%av2 =l loadl %pivp2
	storel %pk, %piptr
	storel %pv, %pivp2
	storel %ak2, %pptr
	storel %av2, %pvp
	%lc =l copy %pi
	%r1 =l call $qsort_pairs(l %a, l %lc)
	%ro =l mul %pi, 16
	%ro2 =l add %ro, 16
	%rptr =l add %a, %ro2
	%rc =l sub %nm1, %pi
	%r2 =l call $qsort_pairs(l %rptr, l %rc)
	ret 0
@ret0
	ret 0
}
