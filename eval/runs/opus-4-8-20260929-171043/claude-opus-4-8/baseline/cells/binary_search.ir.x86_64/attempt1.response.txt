export function l $binary_search(%a :l, %n :l, %key :l) {
@start
	%lo =l copy 0
	%hi =l sub %n, 1
@loop
	%c =l csgtl %lo, %hi
	jnz %c, @notfound, @body
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =l ceql %val, %key
	jnz %eq, @found, @cont
@cont
	%lt =l csltl %val, %key
	jnz %lt, @goright, @goleft
@goright
	%lo =l add %mid, 1
	jmp @loop
@goleft
	%hi =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
