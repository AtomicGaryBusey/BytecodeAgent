export function l $binary_search(%a :l, %n :l, %key :l) {
@start
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start 0, @goright %lonew, @goleft %lo2
	%hi =l phi @start %hi0, @goright %hi2, @goleft %hinew
	%c =w csgtl %lo, %hi
	jnz %c, @notfound, @body
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cont
@cont
	%lt =w csltl %val, %key
	jnz %lt, @goright, @goleft
@goright
	%lonew =l add %mid, 1
	%hi2 =l copy %hi
	jmp @loop
@goleft
	%hinew =l sub %mid, 1
	%lo2 =l copy %lo
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
