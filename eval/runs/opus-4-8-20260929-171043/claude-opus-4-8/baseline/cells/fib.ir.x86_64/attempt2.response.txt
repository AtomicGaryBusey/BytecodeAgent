export function l $fib(l %n) {
@start
	%c =w csltl %n, 1
	jnz %c, @zero, @loop_init
@zero
	ret %n
@loop_init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 1
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%t =l add %a, %b
	%a =l copy %b
	%b =l copy %t
	%i =l add %i, 1
	jmp @loop
@done
	ret %b
}
