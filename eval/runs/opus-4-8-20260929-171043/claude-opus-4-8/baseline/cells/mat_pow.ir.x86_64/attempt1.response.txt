export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%n2 =l mul %n, %n
	%i =l copy 0
@idi
	%cond1 =w csltl %i, %n
	jnz %cond1, @idi_body, @idi_end
@idi_body
	%irow =l mul %i, %n
	%j =l copy 0
@idj
	%cond2 =w csltl %j, %n
	jnz %cond2, @idj_body, @idj_end
@idj_body
	%idx =l add %irow, %j
	%off =l mul %idx, 8
	%addr =l add %out, %off
	%eq =w ceql %i, %j
	jnz %eq, @setone, @setzero
@setone
	storel 1, %addr
	jmp @idj_next
@setzero
	storel 0, %addr
	jmp @idj_next
@idj_next
	%j =l add %j, 1
	jmp @idj
@idj_end
	%i =l add %i, 1
	jmp @idi
@idi_end
	%c =l copy 0
@ploop
	%condp =w csltl %c, %k
	jnz %condp, @pbody, @pend
@pbody
	%mi =l copy 0
@mi
	%cmi =w csltl %mi, %n
	jnz %cmi, @mi_body, @mi_end
@mi_body
	%mirow =l mul %mi, %n
	%mj =l copy 0
@mj
	%cmj =w csltl %mj, %n
	jnz %cmj, @mj_body, @mj_end
@mj_body
	%sum =l copy 0
	%mt =l copy 0
@mt
	%cmt =w csltl %mt, %n
	jnz %cmt, @mt_body, @mt_end
@mt_body
	%oidx =l add %mirow, %mt
	%ooff =l mul %oidx, 8
	%oaddr =l add %out, %ooff
	%oval =l loadl %oaddr
	%trow =l mul %mt, %n
	%aidx =l add %trow, %mj
	%aoff =l mul %aidx, 8
	%aaddr =l add %a, %aoff
	%aval =l loadl %aaddr
	%prod =l mul %oval, %aval
	%sum =l add %sum, %prod
	%mt =l add %mt, 1
	jmp @mt
@mt_end
	%tidx =l add %mirow, %mj
	%toff =l mul %tidx, 8
	%taddr =l add %tmp, %toff
	storel %sum, %taddr
	%mj =l add %mj, 1
	jmp @mj
@mj_end
	%mi =l add %mi, 1
	jmp @mi
@mi_end
	%ci =l copy 0
@cp
	%ccp =w csltl %ci, %n2
	jnz %ccp, @cp_body, @cp_end
@cp_body
	%cpoff =l mul %ci, 8
	%tsrc =l add %tmp, %cpoff
	%odst =l add %out, %cpoff
	%v =l loadl %tsrc
	storel %v, %odst
	%ci =l add %ci, 1
	jmp @cp
@cp_end
	%c =l add %c, 1
	jmp @ploop
@pend
	%n2x2 =l mul %n2, 2
	%zi =l copy 0
@zl
	%czl =w csltl %zi, %n2x2
	jnz %czl, @zl_body, @zl_end
@zl_body
	%zoff =l mul %zi, 8
	%zaddr =l add %tmp, %zoff
	storel 0, %zaddr
	%zi =l add %zi, 1
	jmp @zl
@zl_end
	ret 0
}
