export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
@loop
	%e =w ceql %pc, %len
	jnz %e, @done, @cont
@cont
	%oa =l mul %pc, 8
	%oa =l add %oa, %code
	%op =l loadl %oa
	%t =w ceql %op, 0
	jnz %t, @halt, @c1
@c1
	%t =w ceql %op, 1
	jnz %t, @push, @c2
@c2
	%t =w ceql %op, 2
	jnz %t, @add, @c3
@c3
	%t =w ceql %op, 3
	jnz %t, @sub, @c4
@c4
	%t =w ceql %op, 4
	jnz %t, @mul, @c5
@c5
	%t =w ceql %op, 5
	jnz %t, @dup, @c6
@c6
	%t =w ceql %op, 6
	jnz %t, @swap, @c7
@c7
	%t =w ceql %op, 7
	jnz %t, @drop, @c8
@c8
	%t =w ceql %op, 8
	jnz %t, @over, @c9
@c9
	%t =w ceql %op, 9
	jnz %t, @lt, @c10
@c10
	%t =w ceql %op, 10
	jnz %t, @jz, @c11
@c11
	%t =w ceql %op, 11
	jnz %t, @jmp, @other
@other
	%pc =l add %pc, 1
	jmp @loop
@halt
	jmp @done
@push
	%va =l add %pc, 1
	%va =l mul %va, 8
	%va =l add %va, %code
	%v =l loadl %va
	%da =l mul %sp, 8
	%da =l add %da, %stack
	storel %v, %da
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%bi =l sub %sp, 1
	%ba =l mul %bi, 8
	%ba =l add %ba, %stack
	%b =l loadl %ba
	%ai =l sub %sp, 2
	%aa =l mul %ai, 8
	%aa =l add %aa, %stack
	%a =l loadl %aa
	%r =l add %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@sub
	%bi =l sub %sp, 1
	%ba =l mul %bi, 8
	%ba =l add %ba, %stack
	%b =l loadl %ba
	%ai =l sub %sp, 2
	%aa =l mul %ai, 8
	%aa =l add %aa, %stack
	%a =l loadl %aa
	%r =l sub %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@mul
	%bi =l sub %sp, 1
	%ba =l mul %bi, 8
	%ba =l add %ba, %stack
	%b =l loadl %ba
	%ai =l sub %sp, 2
	%aa =l mul %ai, 8
	%aa =l add %aa, %stack
	%a =l loadl %aa
	%r =l mul %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@lt
	%bi =l sub %sp, 1
	%ba =l mul %bi, 8
	%ba =l add %ba, %stack
	%b =l loadl %ba
	%ai =l sub %sp, 2
	%aa =l mul %ai, 8
	%aa =l add %aa, %stack
	%a =l loadl %aa
	%c =w csltl %a, %b
	%r =l extuw %c
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%ti =l sub %sp, 1
	%ta =l mul %ti, 8
	%ta =l add %ta, %stack
	%tv =l loadl %ta
	%da =l mul %sp, 8
	%da =l add %da, %stack
	storel %tv, %da
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%bi =l sub %sp, 1
	%ba =l mul %bi, 8
	%ba =l add %ba, %stack
	%b =l loadl %ba
	%ai =l sub %sp, 2
	%aa =l mul %ai, 8
	%aa =l add %aa, %stack
	%a =l loadl %aa
	storel %a, %ba
	storel %b, %aa
	%pc =l add %pc, 1
	jmp @loop
@drop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%ti =l sub %sp, 2
	%ta =l mul %ti, 8
	%ta =l add %ta, %stack
	%tv =l loadl %ta
	%da =l mul %sp, 8
	%da =l add %da, %stack
	storel %tv, %da
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%vi =l sub %sp, 1
	%vaddr =l mul %vi, 8
	%vaddr =l add %vaddr, %stack
	%v =l loadl %vaddr
	%sp =l sub %sp, 1
	%z =w ceql %v, 0
	jnz %z, @jztaken, @jznot
@jztaken
	%ta =l add %pc, 1
	%ta =l mul %ta, 8
	%ta =l add %ta, %code
	%pc =l loadl %ta
	jmp @loop
@jznot
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%ta =l add %pc, 1
	%ta =l mul %ta, 8
	%ta =l add %ta, %code
	%pc =l loadl %ta
	jmp @loop
@done
	%hv =w csgtl %sp, 0
	jnz %hv, @getval, @zeroval
@getval
	%ri =l sub %sp, 1
	%ra =l mul %ri, 8
	%ra =l add %ra, %stack
	%ret =l loadl %ra
	jmp @zero
@zeroval
	%ret =l copy 0
	jmp @zero
@zero
	%i =l copy %sp
@zloop
	%zd =w csgel %i, %cap
	jnz %zd, @ret, @zbody
@zbody
	%za =l mul %i, 8
	%za =l add %za, %stack
	storel 0, %za
	%i =l add %i, 1
	jmp @zloop
@ret
	ret %ret
}
