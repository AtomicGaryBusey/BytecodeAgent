mp_mul:
	push rbx
	push r12
	push r13
	push r14
	mov r9, rdx
	mov rbx, rsi
	mov r12, rcx
	mov rax, rbx
	add rax, r12
	xor r11, r11
.Lzero:
	cmp r11, rax
	jae .Lzdone
	mov qword ptr [r8 + r11*8], 0
	inc r11
	jmp .Lzero
.Lzdone:
	test rbx, rbx
	jz .Lsig
	test r12, r12
	jz .Lsig
	xor rsi, rsi
.Louter:
	cmp rsi, rbx
	jae .Lsig
	mov rdx, qword ptr [rdi + rsi*8]
	xor r13, r13
	xor rcx, rcx
.Linner:
	cmp rcx, r12
	jae .Linnerdone
	mulx r11, rax, qword ptr [r9 + rcx*8]
	add rax, r13
	adc r11, 0
	lea r14, [rsi + rcx]
	add rax, qword ptr [r8 + r14*8]
	adc r11, 0
	mov qword ptr [r8 + r14*8], rax
	mov r13, r11
	inc rcx
	jmp .Linner
.Linnerdone:
	lea r14, [rsi + r12]
	mov qword ptr [r8 + r14*8], r13
	inc rsi
	jmp .Louter
.Lsig:
	mov rax, rbx
	add rax, r12
.Lsigloop:
	test rax, rax
	jz .Lsigdone
	cmp qword ptr [r8 + rax*8 - 8], 0
	jne .Lsigdone
	dec rax
	jmp .Lsigloop
.Lsigdone:
	pop r14
	pop r13
	pop r12
	pop rbx
	ret
