memmove_i64:
	cbz x2, .Ldone
	cmp x0, x1
	b.ls .Lfwd
	// dst > src, copy backward
	sub x3, x2, #1
.Lbwd:
	ldr x4, [x1, x3, lsl #3]
	str x4, [x0, x3, lsl #3]
	subs x3, x3, #1
	b.ge .Lbwd
	b .Ldone
.Lfwd:
	mov x3, #0
.Lfwdloop:
	ldr x4, [x1, x3, lsl #3]
	str x4, [x0, x3, lsl #3]
	add x3, x3, #1
	cmp x3, x2
	b.lt .Lfwdloop
.Ldone:
	mov x0, #0
	ret
