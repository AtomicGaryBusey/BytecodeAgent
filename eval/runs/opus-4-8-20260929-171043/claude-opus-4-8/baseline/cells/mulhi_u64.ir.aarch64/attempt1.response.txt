export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%al =l and %a, %mask
	%ah =l shr %a, 32
	%bl =l and %b, %mask
	%bh =l shr %b, 32
	%t0 =l mul %al, %bl
	%t1 =l mul %ah, %bl
	%t2 =l mul %al, %bh
	%t3 =l mul %ah, %bh
	%t0hi =l shr %t0, 32
	%t1lo =l and %t1, %mask
	%t2lo =l and %t2, %mask
	%s =l add %t0hi, %t1lo
	%s2 =l add %s, %t2lo
	%carry =l shr %s2, 32
	%t1hi =l shr %t1, 32
	%t2hi =l shr %t2, 32
	%r0 =l add %t3, %t1hi
	%r1 =l add %r0, %t2hi
	%r2 =l add %r1, %carry
	ret %r2
}
