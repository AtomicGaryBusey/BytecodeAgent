sdivrem:
    test rsi, rsi
    jz .Lzero
    mov rax, rdi
    mov r8, 0x8000000000000000
    cmp rdi, r8
    jne .Lnormal
    cmp rsi, -1
    jne .Lnormal
    mov qword ptr [rdx], 0
    ret
.Lnormal:
    cqo
    idiv rsi
    mov qword ptr [rdx], rdx
    ret
.Lzero:
    mov qword ptr [rdx], rdi
    xor eax, eax
    ret
