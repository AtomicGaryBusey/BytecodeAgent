export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%fwd =w csgel %dst, %src
	jnz %fwd, @back, @forward
@forward
	%i =l copy 0
	jmp @floop
@floop
	%fc =w csltl %i, %n
	jnz %fc, @fbody, @done
@fbody
	%fo =l mul %i, 8
	%fsp =l add %src, %fo
	%fdp =l add %dst, %fo
	%fv =l loadl %fsp
	storel %fv, %fdp
	%i2 =l add %i, 1
	%i =l copy %i2
	jmp @floop
@back
	%j =l sub %n, 1
	jmp @bloop
@bloop
	%bc =w csgel %j, 0
	jnz %bc, @bbody, @done
@bbody
	%bo =l mul %j, 8
	%bsp =l add %src, %bo
	%bdp =l add %dst, %bo
	%bv =l loadl %bsp
	storel %bv, %bdp
	%j2 =l sub %j, 1
	%j =l copy %j2
	jmp @bloop
@done
	ret 0
}
