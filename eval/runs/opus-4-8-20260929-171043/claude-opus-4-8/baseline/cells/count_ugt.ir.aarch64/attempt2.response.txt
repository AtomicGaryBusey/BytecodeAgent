export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cnt =l copy 0
	%i =l copy 0
@loop
	%cond =l csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
