    push r13
    push r14
    xor r9, r9
.zloop:
    mov qword ptr [r8 + r9*8], 0
    inc r9
    cmp r9, rcx
    jle .zloop
    xor r9, r9
.outer:
    cmp r9, rsi
    jge .done
    xor r11, r11
    movzx eax, byte ptr [rdi + r9]
    mov r10, 1
.inner:
    cmp r10, rcx
    jg .inner_done
    mov rbx, [r8 + r10*8]
    movzx r12d, byte ptr [rdx + r10 - 1]
    cmp al, r12b
    jne .notequal
    lea r13, [r11 + 1]
    mov [r8 + r10*8], r13
    jmp .after
.notequal:
    mov r13, [r8 + r10*8 - 8]
    mov r14, rbx
    cmp r14, r13
    cmovl r14, r13
    mov [r8 + r10*8], r14
.after:
    mov r11, rbx
    inc r10
    jmp .inner
.inner_done:
    inc r9
    jmp .outer
.done:
    mov rax, [r8 + rcx*8]
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
