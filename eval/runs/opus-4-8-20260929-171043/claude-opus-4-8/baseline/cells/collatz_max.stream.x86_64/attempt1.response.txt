collatz_max:
  push rbx
  push r12
  xor ebx, ebx
  mov r12, 1
.Louter:
  cmp r12, rdi
  jg .Ldone
  mov rax, r12
  xor ecx, ecx
.Linner:
  cmp rax, 1
  je .Lcheck
  test rax, 1
  jnz .Lodd
  sar rax, 1
  jmp .Lnext
.Lodd:
  lea rax, [rax + rax*2]
  add rax, 1
.Lnext:
  add rcx, 1
  jmp .Linner
.Lcheck:
  cmp rcx, rbx
  jle .Lskip
  mov rbx, rcx
.Lskip:
  add r12, 1
  jmp .Louter
.Ldone:
  mov rax, rbx
  pop r12
  pop rbx
  ret
