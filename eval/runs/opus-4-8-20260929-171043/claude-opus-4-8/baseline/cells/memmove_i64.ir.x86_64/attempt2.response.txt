export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%fwd =w csltl %dst, %src
	jnz %fwd, @forward, @backward
@forward
	%i =l copy 0
	jmp @floop
@floop
	%fc =w csltl %i, %n
	jnz %fc, @fbody, @end
@fbody
	%foff =l mul %i, 8
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fv =l loadl %fsp
	storel %fv, %fdp
	%i =l add %i, 1
	jmp @floop
@backward
	%j =l sub %n, 1
	jmp @bloop
@bloop
	%bc =w csgel %j, 0
	jnz %bc, @bbody, @end
@bbody
	%boff =l mul %j, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	%j =l sub %j, 1
	jmp @bloop
@end
	ret 0
}
