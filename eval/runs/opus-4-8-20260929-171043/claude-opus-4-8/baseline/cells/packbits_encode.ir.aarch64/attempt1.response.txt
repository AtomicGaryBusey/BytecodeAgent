export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c1 =w csgel %i, %n
	jnz %c1, @done, @body
@body
	%pi =l add %src, %i
	%b =w loadub %pi
	%r =w copy 1
	%k =l add %i, 1
@runloop
	%rc1 =w csltl %k, %n
	jnz %rc1, @runchk2, @runend
@runchk2
	%rc2 =w csltw %r, 128
	jnz %rc2, @runchk3, @runend
@runchk3
	%pk =l add %src, %k
	%bk =w loadub %pk
	%rc3 =w ceqw %bk, %b
	jnz %rc3, @runinc, @runend
@runinc
	%r =w add %r, 1
	%k =l add %k, 1
	jmp @runloop
@runend
	%isrun =w csgew %r, 3
	jnz %isrun, @dorepeat, @doliteral
@dorepeat
	%o2 =l add %o, 2
	%fit =w csgtl %o2, %cap
	jnz %fit, @fail, @repwrite
@repwrite
	%hdrw =w sub 257, %r
	%po =l add %out, %o
	storeb %hdrw, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %o2
	%rl =l extuw %r
	%i =l add %i, %rl
	jmp @loop
@doliteral
	%j =l copy %i
	%Lw =w copy 0
@litloop
	%lc1 =w csltl %j, %n
	jnz %lc1, @litchk2, @litend
@litchk2
	%ji =l sub %j, %i
	%lc2 =w csltl %ji, 128
	jnz %lc2, @litchk3, @litend
@litchk3
	%pj =l add %src, %j
	%bj =w loadub %pj
	%rr =w copy 1
	%kk =l add %j, 1
@litrunloop
	%q1 =w csltl %kk, %n
	jnz %q1, @litrun2, @litrunend
@litrun2
	%q2 =w csltw %rr, 3
	jnz %q2, @litrun3, @litrunend
@litrun3
	%pkk =l add %src, %kk
	%bkk =w loadub %pkk
	%q3 =w ceqw %bkk, %bj
	jnz %q3, @litruninc, @litrunend
@litruninc
	%rr =w add %rr, 1
	%kk =l add %kk, 1
	jmp @litrunloop
@litrunend
	%stop =w csgew %rr, 3
	jnz %stop, @litend, @litadv
@litadv
	%j =l add %j, 1
	%Lw =w add %Lw, 1
	jmp @litloop
@litend
	%Ll =l extuw %Lw
	%needl =l add %Ll, 1
	%oend =l add %o, %needl
	%fit2 =w csgtl %oend, %cap
	jnz %fit2, @fail, @litwrite
@litwrite
	%hdr2 =w sub %Lw, 1
	%po2 =l add %out, %o
	storeb %hdr2, %po2
	%ci =l copy 0
@copyloop
	%ccond =w csltl %ci, %Ll
	jnz %ccond, @copybody, @copydone
@copybody
	%sp =l add %src, %i
	%sp2 =l add %sp, %ci
	%val =w loadub %sp2
	%dp =l add %po2, 1
	%dp2 =l add %dp, %ci
	storeb %val, %dp2
	%ci =l add %ci, 1
	jmp @copyloop
@copydone
	%o =l add %o, %needl
	%i =l copy %j
	jmp @loop
@fail
	%m1 =l copy -1
	ret %m1
@done
	ret %o
}
