export function l $isqrt(l %n) {
@start
	%bitInit =l copy 4611686018427387904
	jmp @loop1
@loop1
	%bitL =l phi @start %bitInit, @shift %bitS
	%c1 =w cugtl %bitL, %n
	jnz %c1, @shift, @loop2
@shift
	%bitS =l shr %bitL, 2
	jmp @loop1
@loop2
	%bit =l phi @loop1 %bitL, @next %bitN
	%res =l phi @loop1 0, @next %resN
	%nn =l phi @loop1 %n, @next %nN
	%c2 =w cnel %bit, 0
	jnz %c2, @body, @end
@body
	%sum =l add %res, %bit
	%c3 =w cugel %nn, %sum
	jnz %c3, @yes, @no
@yes
	%nY =l sub %nn, %sum
	%rsY =l shr %res, 1
	%resY =l add %rsY, %bit
	jmp @next
@no
	%resNo =l shr %res, 1
	jmp @next
@next
	%resN =l phi @yes %resY, @no %resNo
	%nN =l phi @yes %nY, @no %nn
	%bitN =l shr %bit, 2
	jmp @loop2
@end
	ret %res
}
