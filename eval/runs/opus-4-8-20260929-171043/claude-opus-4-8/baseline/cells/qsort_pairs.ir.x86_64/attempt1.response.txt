export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w cslel %n, 1
	jnz %c, @ret0, @work
@ret0
	ret 0
@work
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%pp =l add %a, %poff
	%pk =l loadl %pp
	%pvp =l add %pp, 8
	%pv =l loadl %pvp
	%i =l copy 0
	%j =l copy 0
@loop
	%cj =w csltl %j, %nm1
	jnz %cj, @body, @after
@body
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%k =l loadl %jp
	%vp =l add %jp, 8
	%v =l loadl %vp
	%klt =w csltl %k, %pk
	%keq =w ceql %k, %pk
	%vlt =w csltl %v, %pv
	%and =w and %keq, %vlt
	%less =w or %klt, %and
	jnz %less, @swapij, @cont
@swapij
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%tk1 =l loadl %ip
	%tk2 =l loadl %jp
	storel %tk2, %ip
	storel %tk1, %jp
	%ip8 =l add %ip, 8
	%jp8 =l add %jp, 8
	%tv1 =l loadl %ip8
	%tv2 =l loadl %jp8
	storel %tv2, %ip8
	storel %tv1, %jp8
	%i =l add %i, 1
@cont
	%j =l add %j, 1
	jmp @loop
@after
	%aioff =l mul %i, 16
	%aip =l add %a, %aioff
	%sk1 =l loadl %aip
	%sk2 =l loadl %pp
	storel %sk2, %aip
	storel %sk1, %pp
	%aip8 =l add %aip, 8
	%sv1 =l loadl %aip8
	%sv2 =l loadl %pvp
	storel %sv2, %aip8
	storel %sv1, %pvp
	%lr =l call $qsort_pairs(l %a, l %i)
	%rip1 =l add %i, 1
	%roff =l mul %rip1, 16
	%rp =l add %a, %roff
	%rn =l sub %n, %rip1
	%rr =l call $qsort_pairs(l %rp, l %rn)
	ret 0
}
