export function l $collatz_max(%n) {
@start
	%max =l copy 0
	%k =l copy 1
@outer
	%cont =l csgtl %k, %n
	jnz %cont, @done, @body
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%isone =l ceql %x, 1
	jnz %isone, @endinner, @stepcont
@stepcont
	%rem =l and %x, 1
	jnz %rem, @odd, @even
@even
	%x2 =l div %x, 2
	%x =l copy %x2
	jmp @nextstep
@odd
	%m =l mul %x, 3
	%x3 =l add %m, 1
	%x =l copy %x3
	jmp @nextstep
@nextstep
	%steps =l add %steps, 1
	jmp @inner
@endinner
	%better =l csgtl %steps, %max
	jnz %better, @update, @afterupd
@update
	%max =l copy %steps
	jmp @afterupd
@afterupd
	%k =l add %k, 1
	jmp @outer
@done
	ret %max
}
