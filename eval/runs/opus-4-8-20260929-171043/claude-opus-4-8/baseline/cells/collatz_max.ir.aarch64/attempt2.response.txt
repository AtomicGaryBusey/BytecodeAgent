export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%max =l phi @start 0, @afterupd %maxnew
	%k =l phi @start 1, @afterupd %knext
	%cont =l csgtl %k, %n
	jnz %cont, @done, @body
@body
	jmp @inner
@inner
	%x =l phi @body %k, @nextstep %xnew
	%steps =l phi @body 0, @nextstep %stepsnew
	%isone =l ceql %x, 1
	jnz %isone, @endinner, @stepcont
@stepcont
	%rem =l and %x, 1
	jnz %rem, @odd, @even
@even
	%xe =l div %x, 2
	jmp @nextstep
@odd
	%m =l mul %x, 3
	%xo =l add %m, 1
	jmp @nextstep
@nextstep
	%xnew =l phi @even %xe, @odd %xo
	%stepsnew =l add %steps, 1
	jmp @inner
@endinner
	%better =l csgtl %steps, %max
	jnz %better, @update, @afterupd
@update
	jmp @afterupd
@afterupd
	%maxnew =l phi @update %steps, @endinner %max
	%knext =l add %k, 1
	jmp @outer
@done
	ret %max
}
