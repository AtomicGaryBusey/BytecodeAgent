export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %size =l mul %n, %n
    %size8 =l mul %size, 8
    %base =l copy %tmp
    %prod =l add %tmp, %size8
    %ci =l copy 0
@copyloop
    %ccond =w csltl %ci, %size
    jnz %ccond, @copybody, @copydone
@copybody
    %coff =l mul %ci, 8
    %csrc =l add %a, %coff
    %cval =l loadl %csrc
    %cdst =l add %base, %coff
    storel %cval, %cdst
    %ci =l add %ci, 1
    jmp @copyloop
@copydone
    %ii =l copy 0
@idi
    %idicond =w csltl %ii, %n
    jnz %idicond, @idibody, @ididone
@idibody
    %ij =l copy 0
@idj
    %idjcond =w csltl %ij, %n
    jnz %idjcond, @idjbody, @idjdone
@idjbody
    %idx =l mul %ii, %n
    %idx =l add %idx, %ij
    %idoff =l mul %idx, 8
    %idptr =l add %out, %idoff
    %eq =w ceql %ii, %ij
    jnz %eq, @setone, @setzero
@setone
    storel 1, %idptr
    jmp @idjnext
@setzero
    storel 0, %idptr
@idjnext
    %ij =l add %ij, 1
    jmp @idj
@idjdone
    %ii =l add %ii, 1
    jmp @idi
@ididone
    %kc =l copy 0
@kloop
    %kcond =w csltl %kc, %k
    jnz %kcond, @kbody, @kdone
@kbody
    %mi =l copy 0
@mi
    %micond =w csltl %mi, %n
    jnz %micond, @mibody, @midone
@mibody
    %mj =l copy 0
@mj
    %mjcond =w csltl %mj, %n
    jnz %mjcond, @mjbody, @mjdone
@mjbody
    %sum =l copy 0
    %mt =l copy 0
@mt
    %mtcond =w csltl %mt, %n
    jnz %mtcond, @mtbody, @mtdone
@mtbody
    %oidx =l mul %mi, %n
    %oidx =l add %oidx, %mt
    %ooff =l mul %oidx, 8
    %optr =l add %out, %ooff
    %oval =l loadl %optr
    %bidx =l mul %mt, %n
    %bidx =l add %bidx, %mj
    %boff =l mul %bidx, 8
    %bptr =l add %base, %boff
    %bval =l loadl %bptr
    %prodv =l mul %oval, %bval
    %sum =l add %sum, %prodv
    %mt =l add %mt, 1
    jmp @mt
@mtdone
    %pidx =l mul %mi, %n
    %pidx =l add %pidx, %mj
    %poff =l mul %pidx, 8
    %pptr =l add %prod, %poff
    storel %sum, %pptr
    %mj =l add %mj, 1
    jmp @mj
@mjdone
    %mi =l add %mi, 1
    jmp @mi
@midone
    %pc =l copy 0
@pcloop
    %pccond =w csltl %pc, %size
    jnz %pccond, @pcbody, @pcdone
@pcbody
    %pcoff =l mul %pc, 8
    %pcsrc =l add %prod, %pcoff
    %pcval =l loadl %pcsrc
    %pcdst =l add %out, %pcoff
    storel %pcval, %pcdst
    %pc =l add %pc, 1
    jmp @pcloop
@pcdone
    %kc =l add %kc, 1
    jmp @kloop
@kdone
    %tsize =l mul %size, 2
    %zc =l copy 0
@zloop
    %zcond =w csltl %zc, %tsize
    jnz %zcond, @zbody, @zdone
@zbody
    %zoff =l mul %zc, 8
    %zptr =l add %tmp, %zoff
    storel 0, %zptr
    %zc =l add %zc, 1
    jmp @zloop
@zdone
    ret 0
}
