export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@zloop
	%zc =w cslel %j, %n
	jnz %zc, @zbody, @zend
@zbody
	%zp =l mul %j, 8
	%za =l add %row, %zp
	storel 0, %za
	%j =l add %j, 1
	jmp @zloop
@zend
	%i =l copy 0
@iloop
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @iend
@ibody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag =l copy 0
	%jj =l copy 1
@jloop
	%jc =w cslel %jj, %n
	jnz %jc, @jbody, @jend
@jbody
	%jp =l mul %jj, 8
	%ja =l add %row, %jp
	%up =l loadl %ja
	%jm1 =l sub %jj, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%nv =l add %diag, 1
	jmp @setval
@nomatch
	%lp =l mul %jm1, 8
	%la =l add %row, %lp
	%left =l loadl %la
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft
@useup
	%nv1 =l copy %up
	jmp @setval2
@useleft
	%nv1 =l copy %left
@setval2
	%nv =l copy %nv1
@setval
	storel %nv, %ja
	%diag =l copy %up
	%jj =l add %jj, 1
	jmp @jloop
@jend
	%i =l add %i, 1
	jmp @iloop
@iend
	%rp =l mul %n, 8
	%ra =l add %row, %rp
	%res =l loadl %ra
	ret %res
}
