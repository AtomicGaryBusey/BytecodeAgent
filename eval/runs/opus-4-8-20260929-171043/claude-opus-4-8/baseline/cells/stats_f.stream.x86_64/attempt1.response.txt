stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    mov r12, rdi
    mov rbp, rsi
    mov r13, rdx
    mov r14, rcx
    xor r15, r15
    xor ebx, ebx
    mov qword ptr [rsp+16], 0
.Lloop:
    mov rdi, [r12 + rbx*8]
    call r13
    add [rsp+16], rax
    test rax, 1
    jz .Lnotodd
    inc r15
.Lnotodd:
    test rbx, rbx
    jnz .Lupdate
    mov [rsp], rax
    mov [rsp+8], rax
    jmp .Lnext
.Lupdate:
    mov rdx, [rsp]
    cmp rax, rdx
    jge .Lnotmin
    mov [rsp], rax
.Lnotmin:
    mov rdx, [rsp+8]
    cmp rax, rdx
    jle .Lnotmax
    mov [rsp+8], rax
.Lnotmax:
.Lnext:
    inc rbx
    cmp rbx, rbp
    jl .Lloop
    mov rax, [rsp]
    mov [r14], rax
    mov rax, [rsp+8]
    mov [r14+8], rax
    mov rax, [rsp+16]
    mov [r14+16], rax
    mov rax, r15
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
