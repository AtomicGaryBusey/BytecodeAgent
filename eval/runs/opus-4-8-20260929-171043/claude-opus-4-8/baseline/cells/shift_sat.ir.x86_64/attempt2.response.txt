export function l $shift_sat(%x :l, %n :l) {
@start
	%cge =w csgel %n, 64
	jnz %cge, @bigshift, @smallshift
@smallshift
	%ls1 =l shl %x, %n
	%rs1 =l sar %x, %n
	jmp @combine
@bigshift
	%neg =w csltl %x, 0
	%negl =l extsw %neg
	%rs2 =l sub 0, %negl
	jmp @combine
@combine
	%ls =l phi @smallshift %ls1, @bigshift 0
	%rs =l phi @smallshift %rs1, @bigshift %rs2
	%res =l add %ls, %rs
	ret %res
}
