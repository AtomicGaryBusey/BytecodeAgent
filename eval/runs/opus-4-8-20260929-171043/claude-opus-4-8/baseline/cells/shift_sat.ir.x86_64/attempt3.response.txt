export function l $shift_sat(%x :l, %n :l) {
@start
	%big =w csgel %n, 64
	jnz %big, @bigshift, @smallshift
@smallshift
	%ls1 =l shl %x, %n
	%rs1 =l sar %x, %n
	jmp @combine
@bigshift
	%rs2 =l sar %x, 63
	%ls2 =l copy 0
	jmp @combine
@combine
	%ls =l phi @smallshift %ls1, @bigshift %ls2
	%rs =l phi @smallshift %rs1, @bigshift %rs2
	%res =l add %ls, %rs
	ret %res
}
