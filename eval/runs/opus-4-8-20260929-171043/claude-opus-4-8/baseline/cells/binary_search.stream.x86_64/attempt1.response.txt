binary_search:
    xor rax, rax
    mov r8, rsi
.Lloop:
    cmp rax, r8
    jge .Lnotfound
    mov r9, rax
    add r9, r8
    sar r9, 1
    mov r10, qword ptr [rdi + r9*8]
    cmp r10, rdx
    je .Lfound
    jl .Lgo_right
    mov r8, r9
    jmp .Lloop
.Lgo_right:
    lea rax, [r9 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnotfound:
    mov rax, -1
    ret
