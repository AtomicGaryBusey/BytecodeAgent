export function l $base64_encode(l %src, l %n, l %out) {
@start
    %g =l div %n, 3
    %rem =l rem %n, 3
    %si =l copy 0
    %oi =l copy 0
    %gi =l copy 0
@grouploop
    %cond =w csltl %gi, %g
    jnz %cond, @groupbody, @groupdone
@groupbody
    %p0 =l add %src, %si
    %b0 =l loadub %p0
    %p1 =l add %p0, 1
    %b1 =l loadub %p1
    %p2 =l add %p0, 2
    %b2 =l loadub %p2
    %t0 =l shl %b0, 16
    %t1 =l shl %b1, 8
    %val =l or %t0, %t1
    %val2 =l or %val, %b2
    %i0 =l shr %val2, 18
    %i0m =l and %i0, 63
    %i1 =l shr %val2, 12
    %i1m =l and %i1, 63
    %i2 =l shr %val2, 6
    %i2m =l and %i2, 63
    %i3m =l and %val2, 63
    %op =l add %out, %oi
    storeb %i0m, %op
    %opa =l add %op, 1
    storeb %i1m, %opa
    %opb =l add %op, 2
    storeb %i2m, %opb
    %opc =l add %op, 3
    storeb %i3m, %opc
    %si =l add %si, 3
    %oi =l add %oi, 4
    %gi =l add %gi, 1
    jmp @grouploop
@groupdone
    %isrem1 =w ceql %rem, 1
    jnz %isrem1, @rem1, @chkrem2
@rem1
    %rp0 =l add %src, %si
    %rb0 =l loadub %rp0
    %ri0 =l shr %rb0, 2
    %rt =l shl %rb0, 4
    %ri1 =l and %rt, 63
    %rop =l add %out, %oi
    storeb %ri0, %rop
    %rop1 =l add %rop, 1
    storeb %ri1, %rop1
    %oi =l add %oi, 2
    jmp @convstart
@chkrem2
    %isrem2 =w ceql %rem, 2
    jnz %isrem2, @rem2, @convstart
@rem2
    %sp0 =l add %src, %si
    %sb0 =l loadub %sp0
    %sp1 =l add %sp0, 1
    %sb1 =l loadub %sp1
    %si0 =l shr %sb0, 2
    %sa =l shl %sb0, 4
    %sbb =l shr %sb1, 4
    %sc =l or %sa, %sbb
    %si1 =l and %sc, 63
    %sd =l shl %sb1, 2
    %si2 =l and %sd, 63
    %sop =l add %out, %oi
    storeb %si0, %sop
    %sop1 =l add %sop, 1
    storeb %si1, %sop1
    %sop2 =l add %sop, 2
    storeb %si2, %sop2
    %oi =l add %oi, 3
    jmp @convstart
@convstart
    %real =l copy %oi
    %ci =l copy 0
@convloop
    %cc =w csltl %ci, %real
    jnz %cc, @convbody, @convdone
@convbody
    %cp =l add %out, %ci
    %v =l loadub %cp
    %lt26 =w csltl %v, 26
    jnz %lt26, @c_upper, @c_chk52
@c_upper
    %chu =l add %v, 65
    %ch =l copy %chu
    jmp @c_store
@c_chk52
    %lt52 =w csltl %v, 52
    jnz %lt52, @c_lower, @c_chk62
@c_lower
    %chl =l add %v, 71
    %ch =l copy %chl
    jmp @c_store
@c_chk62
    %lt62 =w csltl %v, 62
    jnz %lt62, @c_digit, @c_chk63
@c_digit
    %chd =l sub %v, 4
    %ch =l copy %chd
    jmp @c_store
@c_chk63
    %eq62 =w ceql %v, 62
    jnz %eq62, @c_plus, @c_slash
@c_plus
    %ch =l copy 43
    jmp @c_store
@c_slash
    %ch =l copy 47
    jmp @c_store
@c_store
    storeb %ch, %cp
    %ci =l add %ci, 1
    jmp @convloop
@convdone
    %pr1 =w ceql %rem, 1
    jnz %pr1, @pad2, @chkpad2
@pad2
    %pp =l add %out, %real
    storeb 61, %pp
    %pp1 =l add %pp, 1
    storeb 61, %pp1
    jmp @finish
@chkpad2
    %pr2 =w ceql %rem, 2
    jnz %pr2, @pad1, @finish
@pad1
    %qp =l add %out, %real
    storeb 61, %qp
    jmp @finish
@finish
    %g4 =l mul %g, 4
    %remnz =w cnel %rem, 0
    %remnzl =l extuw %remnz
    %add4 =l mul %remnzl, 4
    %total =l add %g4, %add4
    ret %total
}
