export function l $base64_encode(l %src, l %n, l %out) {
@start
    %g =l div %n, 3
    %rem =l rem %n, 3
    jmp @grouphead
@grouphead
    %gi =l phi @start 0, @groupbody %gi2
    %si =l phi @start 0, @groupbody %si2
    %oi =l phi @start 0, @groupbody %oi2
    %cond =w csltl %gi, %g
    jnz %cond, @groupbody, @groupdone
@groupbody
    %p0 =l add %src, %si
    %b0 =l loadub %p0
    %p1 =l add %p0, 1
    %bb1 =l loadub %p1
    %p2 =l add %p0, 2
    %b2 =l loadub %p2
    %t0 =l shl %b0, 16
    %t1 =l shl %bb1, 8
    %v01 =l or %t0, %t1
    %val =l or %v01, %b2
    %i0 =l shr %val, 18
    %i0m =l and %i0, 63
    %i1 =l shr %val, 12
    %i1m =l and %i1, 63
    %i2 =l shr %val, 6
    %i2m =l and %i2, 63
    %i3m =l and %val, 63
    %op =l add %out, %oi
    storeb %i0m, %op
    %opa =l add %op, 1
    storeb %i1m, %opa
    %opb =l add %op, 2
    storeb %i2m, %opb
    %opc =l add %op, 3
    storeb %i3m, %opc
    %si2 =l add %si, 3
    %oi2 =l add %oi, 4
    %gi2 =l add %gi, 1
    jmp @grouphead
@groupdone
    %r1 =w ceql %rem, 1
    jnz %r1, @rem1, @g2
@rem1
    %rp0 =l add %src, %si
    %rb0 =l loadub %rp0
    %ri0 =l shr %rb0, 2
    %rt =l shl %rb0, 4
    %ri1 =l and %rt, 63
    %rop =l add %out, %oi
    storeb %ri0, %rop
    %rop1 =l add %rop, 1
    storeb %ri1, %rop1
    %oi_a =l add %oi, 2
    jmp @mapinit
@g2
    %r2 =w ceql %rem, 2
    jnz %r2, @rem2, @rem0
@rem2
    %sp0 =l add %src, %si
    %sb0 =l loadub %sp0
    %sp1 =l add %sp0, 1
    %sb1 =l loadub %sp1
    %si0 =l shr %sb0, 2
    %sa =l shl %sb0, 4
    %sbb =l shr %sb1, 4
    %sc =l or %sa, %sbb
    %si1 =l and %sc, 63
    %sd =l shl %sb1, 2
    %si2b =l and %sd, 63
    %sop =l add %out, %oi
    storeb %si0, %sop
    %sop1 =l add %sop, 1
    storeb %si1, %sop1
    %sop2 =l add %sop, 2
    storeb %si2b, %sop2
    %oi_b =l add %oi, 3
    jmp @mapinit
@rem0
    jmp @mapinit
@mapinit
    %fo =l phi @rem1 %oi_a, @rem2 %oi_b, @rem0 %oi
    jmp @maphead
@maphead
    %ci =l phi @mapinit 0, @mstore %ci2
    %cc =w csltl %ci, %fo
    jnz %cc, @mapbody, @mapdone
@mapbody
    %cp =l add %out, %ci
    %v =l loadub %cp
    %c1 =w csltl %v, 26
    jnz %c1, @upper, @m1
@upper
    %chU =l add %v, 65
    jmp @mstore
@m1
    %c2 =w csltl %v, 52
    jnz %c2, @lower, @m2
@lower
    %chL =l add %v, 71
    jmp @mstore
@m2
    %c3 =w csltl %v, 62
    jnz %c3, @digit, @m3
@digit
    %chD =l sub %v, 4
    jmp @mstore
@m3
    %c4 =w ceql %v, 62
    jnz %c4, @plus, @slash
@plus
    %chP =l copy 43
    jmp @mstore
@slash
    %chS =l copy 47
    jmp @mstore
@mstore
    %ch =l phi @upper %chU, @lower %chL, @digit %chD, @plus %chP, @slash %chS
    storeb %ch, %cp
    %ci2 =l add %ci, 1
    jmp @maphead
@mapdone
    %pp1 =w ceql %rem, 1
    jnz %pp1, @pad2, @pd2
@pad2
    %pp =l add %out, %fo
    storeb 61, %pp
    %ppa =l add %pp, 1
    storeb 61, %ppa
    jmp @finish
@pd2
    %pp2 =w ceql %rem, 2
    jnz %pp2, @pad1, @finish
@pad1
    %qp =l add %out, %fo
    storeb 61, %qp
    jmp @finish
@finish
    %g4 =l mul %g, 4
    %rnz =w cnel %rem, 0
    %rnzl =l extuw %rnz
    %a4 =l mul %rnzl, 4
    %tot =l add %g4, %a4
    ret %tot
}
