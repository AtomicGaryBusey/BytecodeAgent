packbits_decode:
  xor r8, r8
  xor r9, r9
.Lloop:
  cmp r8, rsi
  jge .Ldone
  movsx eax, byte ptr [rdi + r8]
  inc r8
  cmp eax, -128
  je .Lloop
  test eax, eax
  js .Lrepeat
  lea r10, [rax + 1]
  mov r11, r8
  add r11, r10
  cmp r11, rsi
  jg .Lerr2
.Llitcopy:
  test r10, r10
  jz .Lloop
  cmp r9, rcx
  jge .Lerr1
  mov al, byte ptr [rdi + r8]
  mov byte ptr [rdx + r9], al
  inc r8
  inc r9
  dec r10
  jmp .Llitcopy
.Lrepeat:
  mov r10d, 1
  sub r10d, eax
  mov r11, r8
  inc r11
  cmp r11, rsi
  jg .Lerr2
  movzx eax, byte ptr [rdi + r8]
  inc r8
.Lreprep:
  test r10, r10
  jz .Lloop
  cmp r9, rcx
  jge .Lerr1
  mov byte ptr [rdx + r9], al
  inc r9
  dec r10
  jmp .Lreprep
.Ldone:
  mov rax, r9
  ret
.Lerr1:
  mov rax, -1
  ret
.Lerr2:
  mov rax, -2
  ret
