    push r14
    push r15

    mov r14, rdx
    xor r15, r15
    xor rbx, rbx
.Lloop:
    cmp r15, r13
    jge .Ldone
    mov rdi, qword ptr [r12 + r15*8]
    call r14
    add rbx, rax
    inc r15
    jmp .Lloop
.Ldone:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
