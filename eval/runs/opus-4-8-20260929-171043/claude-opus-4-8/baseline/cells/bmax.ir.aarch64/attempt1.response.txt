export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%cmp =l csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @update, @next
@update
	%max =l copy %val
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
