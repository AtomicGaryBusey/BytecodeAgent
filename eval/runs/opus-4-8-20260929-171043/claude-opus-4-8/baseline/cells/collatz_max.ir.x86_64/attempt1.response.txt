export function l $collatz_max(%n) {
@start
	%best =l copy 0
	%k =l copy 1
@kloop
	%kcmp =w csgtl %k, %n
	jnz %kcmp, @done, @kbody
@kbody
	%x =l copy %k
	%steps =l copy 0
@sloop
	%s1 =w cnel %x, 1
	jnz %s1, @sbody, @sdone
@sbody
	%r =l rem %x, 2
	%iseven =w ceql %r, 0
	jnz %iseven, @even, @odd
@even
	%xe =l div %x, 2
	%x2 =l copy %xe
	jmp @scont
@odd
	%m =l mul %x, 3
	%xo =l add %m, 1
	%x3 =l copy %xo
	jmp @scont2
@scont
	%xn =l copy %x2
	jmp @supdate
@scont2
	%xn =l copy %x3
@supdate
	%x =l copy %xn
	%steps =l add %steps, 1
	jmp @sloop
@sdone
	%isbetter =w csgtl %steps, %best
	jnz %isbetter, @update, @nomax
@update
	%best =l copy %steps
@nomax
	%k =l add %k, 1
	jmp @kloop
@done
	ret %best
}
