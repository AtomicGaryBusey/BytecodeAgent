export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	jmp @zloop
@zloop
	%zi =l phi @start 0, @zbody %zi2
	%zc =w csltl %zi, %total
	jnz %zc, @zbody, @zend
@zbody
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi2 =l add %zi, 1
	jmp @zloop
@zend
	jmp @iloop
@iloop
	%i =l phi @zend 0, @jend %i2
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @iend
@ibody
	%ai_off =l mul %i, 8
	%ai_p =l add %a, %ai_off
	%ai =l loadl %ai_p
	jmp @jloop
@jloop
	%j =l phi @ibody 0, @jbody %j2
	%carry =l phi @ibody 0, @jbody %carry2
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @jend
@jbody
	%bj_off =l mul %j, 8
	%bj_p =l add %b, %bj_off
	%bj =l loadl %bj_p
	%mask =l copy 4294967295
	%xl =l and %ai, %mask
	%xh =l shr %ai, 32
	%yl =l and %bj, %mask
	%yh =l shr %bj, 32
	%ll =l mul %xl, %yl
	%lh =l mul %xl, %yh
	%hl =l mul %xh, %yl
	%hh =l mul %xh, %yh
	%ll_hi =l shr %ll, 32
	%lh_lo =l and %lh, %mask
	%hl_lo =l and %hl, %mask
	%ta =l add %ll_hi, %lh_lo
	%t =l add %ta, %hl_lo
	%ll_lo =l and %ll, %mask
	%t_lo =l and %t, %mask
	%t_lo_sh =l shl %t_lo, 32
	%lo =l or %ll_lo, %t_lo_sh
	%lh_hi =l shr %lh, 32
	%hl_hi =l shr %hl, 32
	%t_hi =l shr %t, 32
	%hia =l add %hh, %lh_hi
	%hib =l add %hia, %hl_hi
	%hi =l add %hib, %t_hi
	%ij =l add %i, %j
	%ij_off =l mul %ij, 8
	%rp =l add %r, %ij_off
	%rv =l loadl %rp
	%s1 =l add %rv, %lo
	%c1 =w cultl %s1, %rv
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	storel %s2, %rp
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%cy1 =l add %hi, %c1l
	%carry2 =l add %cy1, %c2l
	%j2 =l add %j, 1
	jmp @jloop
@jend
	%inb =l add %i, %nb
	%inb_off =l mul %inb, 8
	%inbp =l add %r, %inb_off
	storel %carry, %inbp
	%i2 =l add %i, 1
	jmp @iloop
@iend
	jmp @cloop
@cloop
	%k =l phi @iend %total, @cdec %km1
	%kpos =w csgtl %k, 0
	jnz %kpos, @cbody, @cend
@cbody
	%km1 =l sub %k, 1
	%km1o =l mul %km1, 8
	%kp =l add %r, %km1o
	%kv =l loadl %kp
	%knz =w cnel %kv, 0
	jnz %knz, @cend, @cdec
@cdec
	jmp @cloop
@cend
	ret %k
}
