export function l $isqrt(l %n) {
@start
	jmp @adjust
@adjust
	%bitA =l phi @start 4611686018427387904, @shiftadj %bitnew
	%c1 =w cugtl %bitA, %n
	jnz %c1, @shiftadj, @loop
@shiftadj
	%bitnew =l shr %bitA, 2
	jmp @adjust
@loop
	%num2 =l phi @adjust %n, @cont %numC
	%res2 =l phi @adjust 0, @cont %resC
	%bit2 =l phi @adjust %bitA, @cont %bitshift
	%cz =w cnel %bit2, 0
	jnz %cz, @body, @end
@body
	%sum =l add %res2, %bit2
	%ge =w cugel %num2, %sum
	jnz %ge, @take, @notake
@take
	%numT =l sub %num2, %sum
	%tmp =l shr %res2, 1
	%resT =l add %tmp, %bit2
	jmp @cont
@notake
	%resN =l shr %res2, 1
	jmp @cont
@cont
	%numC =l phi @take %numT, @notake %num2
	%resC =l phi @take %resT, @notake %resN
	%bitshift =l shr %bit2, 2
	jmp @loop
@end
	ret %res2
}
