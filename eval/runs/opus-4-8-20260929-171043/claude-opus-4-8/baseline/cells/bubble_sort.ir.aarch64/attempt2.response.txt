export function l $bubble_sort(l %a, l %n) {
@start
    %i =l copy 0
@iloop
    %ic =w csltl %i, %n
    jnz %ic, @ibody, @done
@ibody
    %j =l copy 0
    %nm1 =l sub %n, 1
    %jlim =l sub %nm1, %i
@jloop
    %jc =w csltl %j, %jlim
    jnz %jc, @jbody, @jdone
@jbody
    %pa =l mul %j, 8
    %pa2 =l add %a, %pa
    %v1 =l loadl %pa2
    %pb =l add %pa2, 8
    %v2 =l loadl %pb
    %sw =w csgtl %v1, %v2
    jnz %sw, @doswap, @noswap
@doswap
    storel %v2, %pa2
    storel %v1, %pb
@noswap
    %j =l add %j, 1
    jmp @jloop
@jdone
    %i =l add %i, 1
    jmp @iloop
@done
    ret 0
}
