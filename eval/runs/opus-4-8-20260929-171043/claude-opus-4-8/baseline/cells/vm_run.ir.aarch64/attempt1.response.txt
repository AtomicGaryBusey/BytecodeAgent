export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%resp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%ae =w ceql %pc, %len
	jnz %ae, @done, @cont
@cont
	%o1 =l mul %pc, 8
	%o2 =l add %code, %o1
	%op =l loadl %o2
	%e0 =w ceql %op, 0
	jnz %e0, @done, @d1
@d1
	%e1 =w ceql %op, 1
	jnz %e1, @push, @d2
@d2
	%e2 =w ceql %op, 2
	jnz %e2, @add, @d3
@d3
	%e3 =w ceql %op, 3
	jnz %e3, @sub, @d4
@d4
	%e4 =w ceql %op, 4
	jnz %e4, @mul, @d5
@d5
	%e5 =w ceql %op, 5
	jnz %e5, @dup, @d6
@d6
	%e6 =w ceql %op, 6
	jnz %e6, @swap, @d7
@d7
	%e7 =w ceql %op, 7
	jnz %e7, @drop, @d8
@d8
	%e8 =w ceql %op, 8
	jnz %e8, @over, @d9
@d9
	%e9 =w ceql %op, 9
	jnz %e9, @lt, @d10
@d10
	%e10 =w ceql %op, 10
	jnz %e10, @jz, @d11
@d11
	%e11 =w ceql %op, 11
	jnz %e11, @jmp, @other
@push
	%pp_pa =l add %pc, 1
	%pp_o =l mul %pp_pa, 8
	%pp_ad =l add %code, %pp_o
	%pp_v =l loadl %pp_ad
	%pp_d =l loadl %dp
	%pp_do =l mul %pp_d, 8
	%pp_sa =l add %stack, %pp_do
	storel %pp_v, %pp_sa
	%pp_d2 =l add %pp_d, 1
	storel %pp_d2, %dp
	%pp_pc =l add %pc, 2
	storel %pp_pc, %pcp
	jmp @loop
@add
	%ad_d =l loadl %dp
	%ad_bi =l sub %ad_d, 1
	%ad_bo =l mul %ad_bi, 8
	%ad_ba =l add %stack, %ad_bo
	%ad_b =l loadl %ad_ba
	%ad_ai =l sub %ad_d, 2
	%ad_ao =l mul %ad_ai, 8
	%ad_aa =l add %stack, %ad_ao
	%ad_a =l loadl %ad_aa
	%ad_r =l add %ad_a, %ad_b
	storel %ad_r, %ad_aa
	%ad_nd =l sub %ad_d, 1
	storel %ad_nd, %dp
	%ad_pc =l add %pc, 1
	storel %ad_pc, %pcp
	jmp @loop
@sub
	%sb_d =l loadl %dp
	%sb_bi =l sub %sb_d, 1
	%sb_bo =l mul %sb_bi, 8
	%sb_ba =l add %stack, %sb_bo
	%sb_b =l loadl %sb_ba
	%sb_ai =l sub %sb_d, 2
	%sb_ao =l mul %sb_ai, 8
	%sb_aa =l add %stack, %sb_ao
	%sb_a =l loadl %sb_aa
	%sb_r =l sub %sb_a, %sb_b
	storel %sb_r, %sb_aa
	%sb_nd =l sub %sb_d, 1
	storel %sb_nd, %dp
	%sb_pc =l add %pc, 1
	storel %sb_pc, %pcp
	jmp @loop
@mul
	%ml_d =l loadl %dp
	%ml_bi =l sub %ml_d, 1
	%ml_bo =l mul %ml_bi, 8
	%ml_ba =l add %stack, %ml_bo
	%ml_b =l loadl %ml_ba
	%ml_ai =l sub %ml_d, 2
	%ml_ao =l mul %ml_ai, 8
	%ml_aa =l add %stack, %ml_ao
	%ml_a =l loadl %ml_aa
	%ml_r =l mul %ml_a, %ml_b
	storel %ml_r, %ml_aa
	%ml_nd =l sub %ml_d, 1
	storel %ml_nd, %dp
	%ml_pc =l add %pc, 1
	storel %ml_pc, %pcp
	jmp @loop
@dup
	%du_d =l loadl %dp
	%du_ti =l sub %du_d, 1
	%du_to =l mul %du_ti, 8
	%du_ta =l add %stack, %du_to
	%du_v =l loadl %du_ta
	%du_do =l mul %du_d, 8
	%du_sa =l add %stack, %du_do
	storel %du_v, %du_sa
	%du_nd =l add %du_d, 1
	storel %du_nd, %dp
	%du_pc =l add %pc, 1
	storel %du_pc, %pcp
	jmp @loop
@swap
	%sw_d =l loadl %dp
	%sw_i1 =l sub %sw_d, 1
	%sw_o1 =l mul %sw_i1, 8
	%sw_a1 =l add %stack, %sw_o1
	%sw_v1 =l loadl %sw_a1
	%sw_i2 =l sub %sw_d, 2
	%sw_o2 =l mul %sw_i2, 8
	%sw_a2 =l add %stack, %sw_o2
	%sw_v2 =l loadl %sw_a2
	storel %sw_v2, %sw_a1
	storel %sw_v1, %sw_a2
	%sw_pc =l add %pc, 1
	storel %sw_pc, %pcp
	jmp @loop
@drop
	%dr_d =l loadl %dp
	%dr_nd =l sub %dr_d, 1
	storel %dr_nd, %dp
	%dr_pc =l add %pc, 1
	storel %dr_pc, %pcp
	jmp @loop
@over
	%ov_d =l loadl %dp
	%ov_i =l sub %ov_d, 2
	%ov_o =l mul %ov_i, 8
	%ov_a =l add %stack, %ov_o
	%ov_v =l loadl %ov_a
	%ov_do =l mul %ov_d, 8
	%ov_sa =l add %stack, %ov_do
	storel %ov_v, %ov_sa
	%ov_nd =l add %ov_d, 1
	storel %ov_nd, %dp
	%ov_pc =l add %pc, 1
	storel %ov_pc, %pcp
	jmp @loop
@lt
	%lt_d =l loadl %dp
	%lt_bi =l sub %lt_d, 1
	%lt_bo =l mul %lt_bi, 8
	%lt_ba =l add %stack, %lt_bo
	%lt_b =l loadl %lt_ba
	%lt_ai =l sub %lt_d, 2
	%lt_ao =l mul %lt_ai, 8
	%lt_aa =l add %stack, %lt_ao
	%lt_a =l loadl %lt_aa
	%lt_c =w csltl %lt_a, %lt_b
	%lt_r =l extuw %lt_c
	storel %lt_r, %lt_aa
	%lt_nd =l sub %lt_d, 1
	storel %lt_nd, %dp
	%lt_pc =l add %pc, 1
	storel %lt_pc, %pcp
	jmp @loop
@jz
	%jz_d =l loadl %dp
	%jz_vi =l sub %jz_d, 1
	%jz_vo =l mul %jz_vi, 8
	%jz_va =l add %stack, %jz_vo
	%jz_v =l loadl %jz_va
	storel %jz_vi, %dp
	%jz_ta =l add %pc, 1
	%jz_to =l mul %jz_ta, 8
	%jz_tad =l add %code, %jz_to
	%jz_t =l loadl %jz_tad
	%jz_z =w ceql %jz_v, 0
	jnz %jz_z, @jztaken, @jznot
@jztaken
	storel %jz_t, %pcp
	jmp @loop
@jznot
	%jz_pc =l add %pc, 2
	storel %jz_pc, %pcp
	jmp @loop
@jmp
	%jm_ta =l add %pc, 1
	%jm_to =l mul %jm_ta, 8
	%jm_tad =l add %code, %jm_to
	%jm_t =l loadl %jm_tad
	storel %jm_t, %pcp
	jmp @loop
@other
	%ot_pc =l add %pc, 1
	storel %ot_pc, %pcp
	jmp @loop
@done
	%dn_d =l loadl %dp
	%dn_hz =w csgtl %dn_d, 0
	jnz %dn_hz, @hastop, @notop
@hastop
	%dn_ti =l sub %dn_d, 1
	%dn_to =l mul %dn_ti, 8
	%dn_ta =l add %stack, %dn_to
	%dn_top =l loadl %dn_ta
	storel %dn_top, %resp
	jmp @zinit
@notop
	storel 0, %resp
	jmp @zinit
@zinit
	%zi_d =l loadl %dp
	storel %zi_d, %ip
	jmp @zloop
@zloop
	%z_i =l loadl %ip
	%z_c =w csltl %z_i, %cap
	jnz %z_c, @zbody, @zend
@zbody
	%z_o =l mul %z_i, 8
	%z_a =l add %stack, %z_o
	storel 0, %z_a
	%z_ni =l add %z_i, 1
	storel %z_ni, %ip
	jmp @zloop
@zend
	%rv =l loadl %resp
	ret %rv
}
