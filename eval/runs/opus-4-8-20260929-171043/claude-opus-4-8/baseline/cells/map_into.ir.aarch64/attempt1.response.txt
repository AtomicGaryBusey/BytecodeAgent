export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
@loop
	%c =l csltl %i, %n
	jnz %c, @body, @end
@body
	%so =l mul %i, 8
	%sp =l add %src, %so
	%x =l loadl %sp
	%r =l call %f(l %x)
	%dp =l add %dst, %so
	storel %r, %dp
	%lt =l csltl %r, %x
	%cnt =l add %cnt, %lt
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
