export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l sub %n, 1
@loop
	%c1 =l csgel %hi, %lo
	jnz %c1, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check
@check
	%lt =w csltl %val, %key
	jnz %lt, @goright, @goleft
@goright
	%lo =l add %mid, 1
	jmp @loop
@goleft
	%hi =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
