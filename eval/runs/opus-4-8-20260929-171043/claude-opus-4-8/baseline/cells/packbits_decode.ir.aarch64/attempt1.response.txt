export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%si =l copy 0
	%oi =l copy 0
@loop
	%c0 =w csltl %si, %n
	jnz %c0, @body, @done
@body
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si =l add %si, 1
	%hl =l extsw %h
	%isneg =w csltl %hl, 0
	jnz %isneg, @neg, @literal
@literal
	%cnt =l add %hl, 1
	%endi =l add %si, %cnt
	%c1 =w csgtl %endi, %n
	jnz %c1, @err2, @litloop
@litloop
	%d0 =w csgel %si, %endi
	jnz %d0, @loop, @litchk
@litchk
	%d1 =w csgel %oi, %cap
	jnz %d1, @err1, @litdo
@litdo
	%sp =l add %src, %si
	%bv =w loadub %sp
	%op =l add %out, %oi
	storeb %bv, %op
	%si =l add %si, 1
	%oi =l add %oi, 1
	jmp @litloop
@neg
	%c128 =w ceql %hl, -128
	jnz %c128, @loop, @repeat
@repeat
	%rcnt =l sub 1, %hl
	%rneed =l add %si, 1
	%r1 =w csgtl %rneed, %n
	jnz %r1, @err2, @rep0
@rep0
	%rsp =l add %src, %si
	%rbv =w loadub %rsp
	%si =l add %si, 1
	%endo =l add %oi, %rcnt
@reploop
	%e0 =w csgel %oi, %endo
	jnz %e0, @loop, @repchk
@repchk
	%e1 =w csgel %oi, %cap
	jnz %e1, @err1, @repdo
@repdo
	%rop =l add %out, %oi
	storeb %rbv, %rop
	%oi =l add %oi, 1
	jmp @reploop
@err2
	ret -2
@err1
	ret -1
@done
	ret %oi
}
