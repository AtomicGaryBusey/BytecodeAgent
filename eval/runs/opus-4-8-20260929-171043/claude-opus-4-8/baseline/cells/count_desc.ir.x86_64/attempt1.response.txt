export function l $count_desc(%a, %n) {
@start
	%cnt =l copy 0
	%lim =l sub %n, 1
	%i =l copy 0
@loop
	%c =l csltl %i, %lim
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%gt =l csgtl %x, %y
	%cnt =l add %cnt, %gt
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
