export function l $max_byte(l %p, l %n) {
@start
	jmp @cond
@cond
	%max =l phi @start 0, @next %max2, @upd %max3
	%i =l phi @start 0, @next %i2, @upd %i2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	%i2 =l add %i, 1
	jnz %gt, @upd, @next
@upd
	%max3 =l copy %b
	jmp @cond
@next
	%max2 =l copy %max
	jmp @cond
@end
	ret %max
}
