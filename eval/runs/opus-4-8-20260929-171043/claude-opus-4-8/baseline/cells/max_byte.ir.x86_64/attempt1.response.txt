export function l $max_byte(%p, %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @upd, @next
@upd
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @cond
@end
	ret %max
}
