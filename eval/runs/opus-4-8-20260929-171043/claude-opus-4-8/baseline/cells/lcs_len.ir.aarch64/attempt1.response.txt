export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
@initloop
	%c0 =w cslel %j0, %n
	jnz %c0, @initbody, @initdone
@initbody
	%p0 =l mul %j0, 8
	%p0a =l add %row, %p0
	storel 0, %p0a
	%j0 =l add %j0, 1
	jmp @initloop
@initdone
	%i =l copy 0
@iloop
	%ci =w csltl %i, %m
	jnz %ci, @ibody, @idone
@ibody
	%diag =l copy 0
	%xi =l add %x, %i
	%xib =w loadub %xi
	%j =l copy 1
@jloop
	%cj =w cslel %j, %n
	jnz %cj, @jbody, @jdone
@jbody
	%jp =l mul %j, 8
	%rjp =l add %row, %jp
	%up =l loadl %rjp
	%jm1 =l sub %j, 1
	%yj =l add %y, %jm1
	%yjb =w loadub %yj
	%eq =w ceqw %xib, %yjb
	jnz %eq, @match, @nomatch
@match
	%nv =l add %diag, 1
	storel %nv, %rjp
	jmp @jend
@nomatch
	%rjm1p =l mul %jm1, 8
	%rjm1 =l add %row, %rjm1p
	%left =l loadl %rjm1
	%cm =w csgtl %up, %left
	jnz %cm, @useup, @useleft
@useup
	storel %up, %rjp
	jmp @jend
@useleft
	storel %left, %rjp
	jmp @jend
@jend
	%diag =l copy %up
	%j =l add %j, 1
	jmp @jloop
@jdone
	%i =l add %i, 1
	jmp @iloop
@idone
	%np =l mul %n, 8
	%rnp =l add %row, %np
	%res =l loadl %rnp
	ret %res
}
