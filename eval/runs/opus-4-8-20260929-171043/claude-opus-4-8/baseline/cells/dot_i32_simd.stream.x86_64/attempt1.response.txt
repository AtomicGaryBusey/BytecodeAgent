dot_i32_simd:
  pxor xmm2, xmm2
  xor rax, rax
  xor rcx, rcx
.Lloop:
  lea r8, [rcx + 4]
  cmp r8, rdx
  jg .Ltail
  movdqu xmm0, [rdi + rcx*4]
  movdqu xmm1, [rsi + rcx*4]
  movdqa xmm3, xmm0
  movdqa xmm4, xmm1
  pmuldq xmm3, xmm4
  paddq xmm2, xmm3
  psrlq xmm0, 32
  psrlq xmm1, 32
  pmuldq xmm0, xmm1
  paddq xmm2, xmm0
  add rcx, 4
  jmp .Lloop
.Ltail:
  movq rax, xmm2
  psrldq xmm2, 8
  movq r9, xmm2
  add rax, r9
.Lscalar:
  cmp rcx, rdx
  jge .Ldone
  movsxd r8, dword ptr [rdi + rcx*4]
  movsxd r9, dword ptr [rsi + rcx*4]
  imul r8, r9
  add rax, r8
  add rcx, 1
  jmp .Lscalar
.Ldone:
  ret
