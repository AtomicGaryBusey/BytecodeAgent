export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %o =l copy 0
    jmp @mainloop
@mainloop
    %cmp =w csltl %i, %n
    jnz %cmp, @body, @done
@body
    %pi =l add %src, %i
    %bi =w loadub %pi
    %r =w copy 1
    %j2 =l add %i, 1
    jmp @runloop
@runloop
    %c1 =w csltw %r, 128
    jnz %c1, @runchk2, @runend
@runchk2
    %c2 =w csltl %j2, %n
    jnz %c2, @runchk3, @runend
@runchk3
    %pj =l add %src, %j2
    %bj =w loadub %pj
    %c3 =w ceqw %bj, %bi
    jnz %c3, @runinc, @runend
@runinc
    %r =w add %r, 1
    %j2 =l add %j2, 1
    jmp @runloop
@runend
    %isrep =w csgew %r, 3
    jnz %isrep, @emitrep, @emitlit
@emitrep
    %o2 =l add %o, 2
    %fit =w cslel %o2, %cap
    jnz %fit, @dorep, @fail
@dorep
    %hdr =w sub 257, %r
    %po =l add %out, %o
    storeb %hdr, %po
    %po1 =l add %po, 1
    storeb %bi, %po1
    %o =l copy %o2
    %rl =l extuw %r
    %i =l add %i, %rl
    jmp @mainloop
@emitlit
    %j =l copy %i
    %litcnt =w copy 0
    jmp @litloop
@litloop
    %lc1 =w csltl %j, %n
    jnz %lc1, @litchk2, @litend
@litchk2
    %lc2 =w csltw %litcnt, 128
    jnz %lc2, @litchk3, @litend
@litchk3
    %pjj =l add %src, %j
    %bjj =w loadub %pjj
    %rj =w copy 1
    %jj =l add %j, 1
    jmp @litrunloop
@litrunloop
    %rc1 =w csltw %rj, 3
    jnz %rc1, @litrunchk2, @litrundone
@litrunchk2
    %rc2 =w csltl %jj, %n
    jnz %rc2, @litrunchk3, @litrundone
@litrunchk3
    %pjj2 =l add %src, %jj
    %bjj2 =w loadub %pjj2
    %rc3 =w ceqw %bjj2, %bjj
    jnz %rc3, @litruninc, @litrundone
@litruninc
    %rj =w add %rj, 1
    %jj =l add %jj, 1
    jmp @litrunloop
@litrundone
    %stop =w csgew %rj, 3
    jnz %stop, @litend, @litadvance
@litadvance
    %j =l add %j, 1
    %litcnt =w add %litcnt, 1
    jmp @litloop
@litend
    %Ll =l extuw %litcnt
    %need1 =l add %o, 1
    %need2 =l add %need1, %Ll
    %fit2 =w cslel %need2, %cap
    jnz %fit2, @dolit, @fail
@dolit
    %hdrl =w sub %litcnt, 1
    %po0 =l add %out, %o
    storeb %hdrl, %po0
    %o =l add %o, 1
    %k =l copy %i
    jmp @copyloop
@copyloop
    %ck =w csltl %k, %j
    jnz %ck, @copybody, @copyend
@copybody
    %ps =l add %src, %k
    %vb =w loadub %ps
    %pd =l add %out, %o
    storeb %vb, %pd
    %o =l add %o, 1
    %k =l add %k, 1
    jmp @copyloop
@copyend
    %i =l copy %j
    jmp @mainloop
@fail
    %neg =l copy -1
    ret %neg
@done
    ret %o
}
