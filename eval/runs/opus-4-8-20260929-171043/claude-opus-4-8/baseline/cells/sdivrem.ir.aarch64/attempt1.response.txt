export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @bzero, @chkmin
@bzero
	storel %a, %rem
	ret 0
@chkmin
	%amin =w ceql %a, -9223372036854775808
	%bm1 =w ceql %b, -1
	%both =w and %amin, %bm1
	jnz %both, @overflow, @normal
@overflow
	storel 0, %rem
	ret -9223372036854775808
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
