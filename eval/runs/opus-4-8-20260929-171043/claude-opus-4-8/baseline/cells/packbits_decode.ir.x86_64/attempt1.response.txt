export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%psi =l alloc8 8
	%poi =l alloc8 8
	%pk =l alloc8 8
	storel 0, %psi
	storel 0, %poi
	jmp @loop
@loop
	%si =l loadl %psi
	%c1 =w csltl %si, %n
	jnz %c1, @body, @done
@body
	%hp =l add %src, %si
	%h =w loadsb %hp
	%sih =l add %si, 1
	storel %sih, %psi
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @notnop
@notnop
	%isneg =w csltw %h, 0
	jnz %isneg, @repeat, @literal
@repeat
	%hl =l extsw %h
	%count =l sub 1, %hl
	%need =l add %sih, 1
	%ok =w cslel %need, %n
	jnz %ok, @rep2, @err2
@err2
	ret -2
@rep2
	%bp =l add %src, %sih
	%b =w loadub %bp
	%sadv =l add %sih, 1
	storel %sadv, %psi
	storel 0, %pk
@replw
	%k =l loadl %pk
	%more =w csltl %k, %count
	jnz %more, @repwrite, @loop
@repwrite
	%oi =l loadl %poi
	%full =w csgel %oi, %cap
	jnz %full, @capfull, @repdo
@capfull
	ret -1
@repdo
	%op =l add %out, %oi
	storeb %b, %op
	%oi2 =l add %oi, 1
	storel %oi2, %poi
	%k2 =l add %k, 1
	storel %k2, %pk
	jmp @replw
@literal
	%hl2 =l extsw %h
	%count2 =l add %hl2, 1
	%need2 =l add %sih, %count2
	%ok2 =w cslel %need2, %n
	jnz %ok2, @lit2, @err2b
@err2b
	ret -2
@lit2
	%sadv2 =l add %sih, %count2
	storel %sadv2, %psi
	storel 0, %pk
@litlw
	%k3 =l loadl %pk
	%more3 =w csltl %k3, %count2
	jnz %more3, @litwrite, @loop
@litwrite
	%oi3 =l loadl %poi
	%full3 =w csgel %oi3, %cap
	jnz %full3, @capfull2, @litdo
@capfull2
	ret -1
@litdo
	%srcp =l add %src, %sih
	%srcp2 =l add %srcp, %k3
	%lb =w loadub %srcp2
	%op3 =l add %out, %oi3
	storeb %lb, %op3
	%oi4 =l add %oi3, 1
	storel %oi4, %poi
	%k4 =l add %k3, 1
	storel %k4, %pk
	jmp @litlw
@done
	%res =l loadl %poi
	ret %res
}
