export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%c =w csltl %i, %j
	jnz %c, @body, @end
@body
	%pi =l mul %i, 8
	%pi2 =l add %a, %pi
	%pj =l mul %j, 8
	%pj2 =l add %a, %pj
	%vi =l loadl %pi2
	%vj =l loadl %pj2
	storel %vj, %pi2
	storel %vi, %pj2
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@end
	ret 0
}
