export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
@loop
	%crc =w phi @start %crc0, @bitdone %crcend
	%i =l phi @start %i0, @bitdone %i1
	%pp =l phi @start %p, @bitdone %pp1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%b =w loadub %pp
	%crcb =w xor %crc, %b
	jmp @bitloop
@bitloop
	%crcj =w phi @body %crcb, @next %crcn
	%j =w phi @body 0, @next %j1
	%bcond =w csltw %j, 8
	jnz %bcond, @bit, @bitdone
@bit
	%lsb =w and %crcj, 1
	%crcsh =w shr %crcj, 1
	jnz %lsb, @xor, @next
@xor
	%crcx =w xor %crcsh, 3988292384
	jmp @next
@next
	%crcn =w phi @bit %crcsh, @xor %crcx
	%j1 =w add %j, 1
	jmp @bitloop
@bitdone
	%crcend =w copy %crcj
	%i1 =l add %i, 1
	%pp1 =l add %pp, 1
	jmp @loop
@done
	%crcf =w xor %crc, 4294967295
	%r =l extuw %crcf
	ret %r
}
