export function l $crc32(%p, %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@loop
	%cond =l csltl %i, %n
	jnz %cond, @body, @done
@body
	%b =w loadub %p
	%crc =w xor %crc, %b
	%j =w copy 0
@bitloop
	%bcond =w csltw %j, 8
	jnz %bcond, @bit, @bitdone
@bit
	%lsb =w and %crc, 1
	%crc =w shr %crc, 1
	jnz %lsb, @xor, @next
@xor
	%crc =w xor %crc, 3988292384
	jmp @next
@next
	%j =w add %j, 1
	jmp @bitloop
@bitdone
	%p =l add %p, 1
	%i =l add %i, 1
	jmp @loop
@done
	%crc =w xor %crc, 4294967295
	%r =l extuw %crc
	ret %r
}
