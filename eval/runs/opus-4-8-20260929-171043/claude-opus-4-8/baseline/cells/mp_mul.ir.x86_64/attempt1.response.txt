export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pc =l alloc8 8
	%pk =l alloc8 8
	storel 0, %pi
@zloop
	%zi =l loadl %pi
	%zcmp =w csltl %zi, %total
	jnz %zcmp, @zbody, @zend
@zbody
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi2 =l add %zi, 1
	storel %zi2, %pi
	jmp @zloop
@zend
	storel 0, %pi
@oloop
	%i =l loadl %pi
	%ocmp =w csltl %i, %na
	jnz %ocmp, @obody, @oend
@obody
	storel 0, %pc
	storel 0, %pj
	%aoff =l mul %i, 8
	%aaddr =l add %a, %aoff
	%ai =l loadl %aaddr
@iloop
	%j =l loadl %pj
	%icmp =w csltl %j, %nb
	jnz %icmp, @ibody, @iend
@ibody
	%boff =l mul %j, 8
	%baddr =l add %b, %boff
	%bj =l loadl %baddr
	%alo =l and %ai, 4294967295
	%ahi =l shr %ai, 32
	%blo =l and %bj, 4294967295
	%bhi =l shr %bj, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%mid =l add %p1, %p2
	%midc =w cultl %mid, %p1
	%midcl =l extuw %midc
	%midlo =l shl %mid, 32
	%midhi =l shr %mid, 32
	%lo =l add %p0, %midlo
	%loc =w cultl %lo, %p0
	%locl =l extuw %loc
	%hi0 =l add %p3, %midhi
	%midcsh =l shl %midcl, 32
	%hi1 =l add %hi0, %midcsh
	%hi =l add %hi1, %locl
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%raddr =l add %r, %roff
	%rval =l loadl %raddr
	%lo2 =l add %lo, %rval
	%lo2c =w cultl %lo2, %lo
	%lo2cl =l extuw %lo2c
	%hi2 =l add %hi, %lo2cl
	%carry =l loadl %pc
	%lo3 =l add %lo2, %carry
	%lo3c =w cultl %lo3, %lo2
	%lo3cl =l extuw %lo3c
	%hi3 =l add %hi2, %lo3cl
	storel %lo3, %raddr
	storel %hi3, %pc
	%j2 =l add %j, 1
	storel %j2, %pj
	jmp @iloop
@iend
	%carryf =l loadl %pc
	%inb =l add %i, %nb
	%fnoff =l mul %inb, 8
	%fnaddr =l add %r, %fnoff
	storel %carryf, %fnaddr
	%i2 =l add %i, 1
	storel %i2, %pi
	jmp @oloop
@oend
	storel 0, %pk
	storel 0, %pi
@cloop
	%ci =l loadl %pi
	%ccmp =w csltl %ci, %total
	jnz %ccmp, @cbody, @cend
@cbody
	%coff =l mul %ci, 8
	%caddr =l add %r, %coff
	%cval =l loadl %caddr
	%cnz =w cnel %cval, 0
	jnz %cnz, @cset, @cnext
@cset
	%ci1 =l add %ci, 1
	storel %ci1, %pk
	jmp @cnext
@cnext
	%ci2 =l add %ci, 1
	storel %ci2, %pi
	jmp @cloop
@cend
	%result =l loadl %pk
	ret %result
}
