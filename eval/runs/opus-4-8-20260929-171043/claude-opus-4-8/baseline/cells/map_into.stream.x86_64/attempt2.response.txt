map_into:
    push r12
    push r13
    push r14
    push r15
    push rbx
    push rbp
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor rbx, rbx
    xor rbp, rbp
.Lloop:
    cmp rbp, r14
    jge .Ldone
    mov rdi, qword ptr [r13 + rbp*8]
    call r15
    mov rdx, qword ptr [r13 + rbp*8]
    mov qword ptr [r12 + rbp*8], rax
    cmp rax, rdx
    jl .Lless
    jmp .Lnext
.Lless:
    inc rbx
.Lnext:
    inc rbp
    jmp .Lloop
.Ldone:
    mov rax, rbx
    add rsp, 8
    pop rbp
    pop rbx
    pop r15
    pop r14
    pop r13
    pop r12
    ret
