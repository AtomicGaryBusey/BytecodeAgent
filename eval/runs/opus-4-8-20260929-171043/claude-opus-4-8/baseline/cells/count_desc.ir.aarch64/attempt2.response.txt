export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%cmp =w csgtl %lim, 0
	jnz %cmp, @init, @end
@init
	%prev0 =l loadl %a
	jmp @body
@body
	%i =l phi @init 0, @cont %i2
	%prev =l phi @init %prev0, @cont %next
	%cnt =l phi @init 0, @cont %cnt2
	%off =l add %i, 1
	%p =l mul %off, 8
	%nextaddr =l add %a, %p
	%next =l loadl %nextaddr
	%gt =w csgtl %prev, %next
	%gtl =l extuw %gt
	%cnt2 =l add %cnt, %gtl
	%i2 =l add %i, 1
	%c2 =w csltl %i2, %lim
	jnz %c2, @cont, @done
@cont
	jmp @body
@done
	ret %cnt2
@end
	ret 0
}
