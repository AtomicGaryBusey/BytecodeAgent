export function l $count_desc(%a, %n) {
@start
	%cnt =l copy 0
	%lim =l sub %n, 1
	%cmp =w csgtl %lim, 0
	jnz %cmp, @loop, @end
@loop
	%i =l copy 0
	%prev =l loadl %a
@body
	%off =l add %i, 1
	%addr =l add %a, %off
	%addr8 =l mul %addr, 1
	%p =l mul %off, 8
	%nextaddr =l add %a, %p
	%next =l loadl %nextaddr
	%gt =w csgtl %prev, %next
	%cnt =l add %cnt, %gt
	%prev =l copy %next
	%i =l add %i, 1
	%c2 =w csltl %i, %lim
	jnz %c2, @body, @end
@end
	ret %cnt
}
