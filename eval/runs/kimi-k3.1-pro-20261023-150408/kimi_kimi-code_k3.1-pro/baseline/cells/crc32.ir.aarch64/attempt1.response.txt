export function l $crc32(l %p, l %n) {
@start
	%all_ones =w copy -1
	%polynomial =w copy -306674912
	jmp @loop
@loop
	%ptr =l phi @start %p, @body %ptr_next
	%remaining =l phi @start %n, @body %remaining_next
	%crc =w phi @start %all_ones, @body %c8
	%done =w ceql %remaining, 0
	jnz %done, @finish, @body
@body
	%byte =w loadub %ptr
	%mixed =w xor %crc, %byte

	%bit0 =w and %mixed, 1
	%shift0 =w shr %mixed, 1
	%mask0 =w sub 0, %bit0
	%poly0 =w and %mask0, %polynomial
	%c1 =w xor %shift0, %poly0

	%bit1 =w and %c1, 1
	%shift1 =w shr %c1, 1
	%mask1 =w sub 0, %bit1
	%poly1 =w and %mask1, %polynomial
	%c2 =w xor %shift1, %poly1

	%bit2 =w and %c2, 1
	%shift2 =w shr %c2, 1
	%mask2 =w sub 0, %bit2
	%poly2 =w and %mask2, %polynomial
	%c3 =w xor %shift2, %poly2

	%bit3 =w and %c3, 1
	%shift3 =w shr %c3, 1
	%mask3 =w sub 0, %bit3
	%poly3 =w and %mask3, %polynomial
	%c4 =w xor %shift3, %poly3

	%bit4 =w and %c4, 1
	%shift4 =w shr %c4, 1
	%mask4 =w sub 0, %bit4
	%poly4 =w and %mask4, %polynomial
	%c5 =w xor %shift4, %poly4

	%bit5 =w and %c5, 1
	%shift5 =w shr %c5, 1
	%mask5 =w sub 0, %bit5
	%poly5 =w and %mask5, %polynomial
	%c6 =w xor %shift5, %poly5

	%bit6 =w and %c6, 1
	%shift6 =w shr %c6, 1
	%mask6 =w sub 0, %bit6
	%poly6 =w and %mask6, %polynomial
	%c7 =w xor %shift6, %poly6

	%bit7 =w and %c7, 1
	%shift7 =w shr %c7, 1
	%mask7 =w sub 0, %bit7
	%poly7 =w and %mask7, %polynomial
	%c8 =w xor %shift7, %poly7

	%ptr_next =l add %ptr, 1
	%remaining_next =l sub %remaining, 1
	jmp @loop
@finish
	%final =w xor %crc, %all_ones
	%result =l extuw %final
	ret %result
}
