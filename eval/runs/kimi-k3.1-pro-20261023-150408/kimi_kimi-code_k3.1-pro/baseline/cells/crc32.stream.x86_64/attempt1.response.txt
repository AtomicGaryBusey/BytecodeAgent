crc32:
    mov eax, -1
    test rsi, rsi
    je .Lfinish
.Lnext_byte:
    movzx edx, byte ptr [rdi]
    inc rdi
    xor eax, edx
    mov ecx, 8
.Lnext_bit:
    mov edx, eax
    and edx, 1
    shr eax, 1
    test edx, edx
    je .Lno_xor
    xor eax, 0xEDB88320
.Lno_xor:
    dec ecx
    jne .Lnext_bit
    dec rsi
    jne .Lnext_byte
.Lfinish:
    not eax
    ret
