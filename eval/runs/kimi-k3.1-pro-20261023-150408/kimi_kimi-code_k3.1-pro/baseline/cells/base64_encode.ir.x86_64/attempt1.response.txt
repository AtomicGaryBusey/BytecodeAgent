export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @group
@group
    %p =l phi @start %src, @full_group %p_next
    %rem =l phi @start %n, @full_group %rem_next
    %op =l phi @start %out, @full_group %op_next
    %at_end =w ceql %rem, 0
    jnz %at_end, @finish, @load_b0
@load_b0
    %b0 =l loadub %p
    %hi =l shl %b0, 16
    %lt2 =w csltl %rem, 2
    jnz %lt2, @emit_setup, @two_or_more
@two_or_more
    %p1 =l add %p, 1
    %b1 =l loadub %p1
    %mid =l shl %b1, 8
    %himi =l or %hi, %mid
    %lt3 =w csltl %rem, 3
    jnz %lt3, @emit_setup, @three_bytes
@three_bytes
    %p2 =l add %p, 2
    %b2 =l loadub %p2
    %val3 =l or %himi, %b2
    jmp @emit_setup
@emit_setup
    %val =l phi @load_b0 %hi, @two_or_more %himi, @three_bytes %val3
    jmp @emit_loop
@emit_loop
    %bits =l phi @emit_setup %val, @char_ready %bits_next
    %step =l phi @emit_setup 0, @char_ready %step_next
    %dst =l phi @emit_setup %op, @char_ready %dst_next
    %top =l shr %bits, 18
    %idx =l and %top, 63
    %upper_test =w csltl %idx, 26
    jnz %upper_test, @char_upper, @not_upper
@char_upper
    %upper_char =l add %idx, 65
    jmp @char_ready
@not_upper
    %lower_test =w csltl %idx, 52
    jnz %lower_test, @char_lower, @not_lower
@char_lower
    %lower_char =l add %idx, 71
    jmp @char_ready
@not_lower
    %digit_test =w csltl %idx, 62
    jnz %digit_test, @char_digit, @special
@char_digit
    %digit_char =l sub %idx, 4
    jmp @char_ready
@special
    %plus_test =w ceql %idx, 62
    jnz %plus_test, @char_plus, @char_slash
@char_plus
    %plus_char =l copy 43
    jmp @char_ready
@char_slash
    %slash_char =l copy 47
    jmp @char_ready
@char_ready
    %ch =l phi @char_upper %upper_char, @char_lower %lower_char, @char_digit %digit_char, @char_plus %plus_char, @char_slash %slash_char
    storeb %ch, %dst
    %bits_next =l shl %bits, 6
    %dst_next =l add %dst, 1
    %step_next =l add %step, 1
    %more =w csltl %step_next, 4
    jnz %more, @emit_loop, @group_done
@group_done
    %is_one =w ceql %rem, 1
    jnz %is_one, @pad_one, @check_two
@check_two
    %is_two =w ceql %rem, 2
    jnz %is_two, @pad_two, @full_group
@pad_one
    %pad2 =l add %op, 2
    storeb 61, %pad2
    %pad3 =l add %op, 3
    storeb 61, %pad3
    %len1 =l sub %dst_next, %out
    ret %len1
@pad_two
    %pad3_two =l add %op, 3
    storeb 61, %pad3_two
    %len2 =l sub %dst_next, %out
    ret %len2
@full_group
    %p_next =l add %p, 3
    %rem_next =l sub %rem, 3
    %op_next =l copy %dst_next
    jmp @group
@finish
    %len =l sub %op, %out
    ret %len
}
