export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    jmp @cond
@cond
    %i =l phi @entry 0, @body %next_i
    %count =l phi @entry 0, @body %next_count
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %src_addr =l add %src, %offset
    %x =l loadl %src_addr
    %r =l call %f(l %x)
    %dst_addr =l add %dst, %offset
    storel %r, %dst_addr
    %less =w csltl %r, %x
    %increment =l extsw %less
    %next_count =l add %count, %increment
    %next_i =l add %i, 1
    jmp @cond
@done
    ret %count
}
