export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @outer_init
@outer_init
	%i0 =l sub %n, 1
	jmp @outer
@outer
	%i =l phi @outer_init %i0, @pass_done %i_next
	jmp @inner
@inner
	%j =l phi @outer 0, @inner_next %j_next
	%cond =w csltl %j, %i
	jnz %cond, @compare, @pass_done
@compare
	%offset =l mul %j, 8
	%p =l add %a, %offset
	%p1 =l add %p, 8
	%x =l loadl %p
	%y =l loadl %p1
	%swap =w csltl %y, %x
	jnz %swap, @do_swap, @no_swap
@do_swap
	storel %y, %p
	storel %x, %p1
	jmp @inner_next
@no_swap
	jmp @inner_next
@inner_next
	%j_next =l add %j, 1
	jmp @inner
@pass_done
	%i_next =l sub %i, 1
	%finished =w ceql %i_next, 0
	jnz %finished, @done, @outer
@done
	ret 0
}
