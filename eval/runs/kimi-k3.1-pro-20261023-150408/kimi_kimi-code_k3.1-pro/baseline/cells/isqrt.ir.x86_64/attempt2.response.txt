export function l $isqrt(l %n) {
@start
	%lo_ptr =l alloc8 8
	%hi_ptr =l alloc8 8
	storel 0, %lo_ptr
	storel 4294967295, %hi_ptr
	jmp @loop
@loop
	%lo =l loadl %lo_ptr
	%hi =l loadl %hi_ptr
	%finished =w ceql %lo, %hi
	jnz %finished, @done, @search
@search
	%diff =l sub %hi, %lo
	%span =l add %diff, 1
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%square =l mul %mid, %mid
	%too_big =w cugtl %square, %n
	jnz %too_big, @lower, @raise
@raise
	storel %mid, %lo_ptr
	jmp @loop
@lower
	%new_hi =l sub %mid, 1
	storel %new_hi, %hi_ptr
	jmp @loop
@done
	ret %lo
}
