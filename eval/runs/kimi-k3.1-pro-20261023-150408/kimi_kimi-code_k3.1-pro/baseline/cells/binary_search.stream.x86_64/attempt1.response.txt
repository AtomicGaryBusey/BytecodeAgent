binary_search:
xor ecx, ecx
mov r8, rsi
.Lloop:
cmp rcx, r8
jae .Lnot_found
mov r9, r8
sub r9, rcx
shr r9, 1
add r9, rcx
mov rax, qword ptr [rdi + r9*8]
cmp rax, rdx
je .Lfound
jl .Lright
mov r8, r9
jmp .Lloop
.Lright:
lea rcx, [r9 + 1]
jmp .Lloop
.Lfound:
mov rax, r9
ret
.Lnot_found:
mov rax, -1
ret
