export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%m =l phi @start 9223372036854775807, @next %m1
	%done =w ceql %i, %n
	jnz %done, @exit, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%less =w csltl %x, %m
	jnz %less, @update, @next
@update
	jmp @next
@next
	%m1 =l phi @body %m, @update %x
	%i1 =l add %i, 1
	jmp @loop
@exit
	ret %m
}
