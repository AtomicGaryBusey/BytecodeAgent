export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	jmp @zero_head
@zero_head
	%zi =l phi @entry 0, @zero_body %zinext
	%zmore =w cultl %zi, %n
	jnz %zmore, @zero_body, @after_zero
@zero_body
	%zioff =l mul %zi, 8
	%ziptr =l add %r, %zioff
	storel 0, %ziptr
	%zinext =l add %zi, 1
	jmp @zero_head
@after_zero
	%nazero =w ceql %na, 0
	jnz %nazero, @go_trim, @test_nb
@test_nb
	%nbzero =w ceql %nb, 0
	jnz %nbzero, @go_trim, @mul_begin
@go_trim
	jmp @trim_start
@mul_begin
	jmp @outer_head
@outer_head
	%i =l phi @mul_begin 0, @inner_done %inext
	%imore =w cultl %i, %na
	jnz %imore, @outer_body, @outer_done
@outer_body
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	jmp @inner_head
@inner_head
	%j =l phi @outer_body 0, @inner_body %jnext
	%carry =l phi @outer_body 0, @inner_body %h2
	%jmore =w cultl %j, %nb
	jnz %jmore, @inner_body, @inner_done
@inner_body
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bv =l loadl %bptr
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%ij =l add %i, %j
	%ijoff =l mul %ij, 8
	%rptr =l add %r, %ijoff
	%rv =l loadl %rptr
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00lo =l and %p00, 4294967295
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p01hi =l shr %p01, 32
	%p10lo =l and %p10, 4294967295
	%p10hi =l shr %p10, 32
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%midsh =l shl %mid, 32
	%plo =l add %p00lo, %midsh
	%midc =l shr %mid, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%phi =l add %hi1, %midc
	%s1 =l add %plo, %rv
	%c1 =w cultl %s1, %plo
	%c1x =l extuw %c1
	%h1 =l add %phi, %c1x
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%c2x =l extuw %c2
	%h2 =l add %h1, %c2x
	storel %s2, %rptr
	%jnext =l add %j, 1
	jmp @inner_head
@inner_done
	%endidx =l add %i, %nb
	%endoff =l mul %endidx, 8
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%inext =l add %i, 1
	jmp @outer_head
@outer_done
	jmp @trim_start
@trim_start
	jmp @trim_head
@trim_head
	%k =l phi @trim_start %n, @trim_dec %kdec
	%kzero =w ceql %k, 0
	jnz %kzero, @trim_return, @trim_check
@trim_check
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kptr =l add %r, %koff
	%top =l loadl %kptr
	%topzero =w ceql %top, 0
	jnz %topzero, @trim_dec, @trim_return
@trim_dec
	%kdec =l copy %km1
	jmp @trim_head
@trim_return
	ret %k
}
