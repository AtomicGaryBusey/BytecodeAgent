export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%pi =l add %a, %ioff
	%pj =l add %a, %joff
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
