export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%count0 =l and %y0, 1
	%p1 =l add %a, 8
	%more0 =w csltl 1, %n
	jnz %more0, @loop, @single

@loop
	%i =l phi @entry 1, @next %i1
	%p =l phi @entry %p1, @next %p2
	%min =l phi @entry %y0, @next %min1
	%max =l phi @entry %y0, @next %max1
	%sum =l phi @entry %y0, @next %sum1
	%count =l phi @entry %count0, @next %count1
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%count1 =l add %count, %odd
	%ismin =w csltl %y, %min
	jnz %ismin, @newmin, @keepmin

@newmin
	jmp @minjoin

@keepmin
	jmp @minjoin

@minjoin
	%min1 =l phi @newmin %y, @keepmin %min
	%ismax =w csltl %max, %y
	jnz %ismax, @newmax, @keepmax

@newmax
	jmp @next

@keepmax
	jmp @next

@next
	%max1 =l phi @newmax %y, @keepmax %max
	%i1 =l add %i, 1
	%p2 =l add %p, 8
	%more =w csltl %i1, %n
	jnz %more, @loop, @finish

@finish
	%out1 =l add %out, 8
	%out2 =l add %out, 16
	storel %min1, %out
	storel %max1, %out1
	storel %sum1, %out2
	ret %count1

@single
	%out1s =l add %out, 8
	%out2s =l add %out, 16
	storel %y0, %out
	storel %y0, %out1s
	storel %y0, %out2s
	ret %count0
}
