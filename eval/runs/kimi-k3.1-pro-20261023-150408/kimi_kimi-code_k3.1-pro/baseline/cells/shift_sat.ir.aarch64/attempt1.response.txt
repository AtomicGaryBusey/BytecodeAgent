export function l $shift_sat(l %x, l %n) {
@start
	jmp @loop
@loop
	%left =l phi @start %x, @body %left_next
	%right =l phi @start %x, @body %right_next
	%count =l phi @start %n, @body %count_next
	%done =w ceql %count, 0
	jnz %done, @exit, @body
@body
	%left_next =l shl %left, 1
	%right_next =l sar %right, 1
	%count_next =l sub %count, 1
	jmp @loop
@exit
	%sum =l add %left, %right
	ret %sum
}
