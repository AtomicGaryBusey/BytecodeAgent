export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%total =l add %nn, %nn
	jmp @zero_head
@zero_head
	%zi =l phi @start 0, @zero_body %zi_next
	%zc =w csltl %zi, %nn
	jnz %zc, @zero_body, @diag_start
@zero_body
	%zoff =l mul %zi, 8
	%zaddr =l add %out, %zoff
	storel 0, %zaddr
	%zi_next =l add %zi, 1
	jmp @zero_head
@diag_start
	jmp @diag_head
@diag_head
	%di =l phi @diag_start 0, @diag_body %di_next
	%dc =w csltl %di, %n
	jnz %dc, @diag_body, @exp_start
@diag_body
	%drow =l mul %di, %n
	%didx =l add %drow, %di
	%doff =l mul %didx, 8
	%daddr =l add %out, %doff
	storel 1, %daddr
	%di_next =l add %di, 1
	jmp @diag_head
@exp_start
	jmp @exp_head
@exp_head
	%e =l phi @exp_start 0, @exp_done %e_next
	%ec =w csltl %e, %k
	jnz %ec, @i_start, @cleanup_start
@i_start
	jmp @i_head
@i_head
	%i =l phi @i_start 0, @row_done %i_next
	%in =l mul %i, %n
	%ic =w csltl %i, %n
	jnz %ic, @j_head, @copy_start
@j_head
	%j =l phi @i_head 0, @element_done %j_next
	%jc =w csltl %j, %n
	jnz %jc, @t_head, @row_done
@t_head
	%t =l phi @j_head 0, @t_body %t_next
	%sum =l phi @j_head 0, @t_body %sum_next
	%tc =w csltl %t, %n
	jnz %tc, @t_body, @element_done
@t_body
	%it =l add %in, %t
	%ito =l mul %it, 8
	%xp =l add %out, %ito
	%x =l loadl %xp
	%tn =l mul %t, %n
	%tj =l add %tn, %j
	%tjo =l mul %tj, 8
	%yp =l add %a, %tjo
	%y =l loadl %yp
	%prod =l mul %x, %y
	%sum_next =l add %sum, %prod
	%t_next =l add %t, 1
	jmp @t_head
@element_done
	%pidx =l add %in, %j
	%poff =l mul %pidx, 8
	%pp =l add %tmp, %poff
	storel %sum, %pp
	%j_next =l add %j, 1
	jmp @j_head
@row_done
	%i_next =l add %i, 1
	jmp @i_head
@copy_start
	jmp @copy_head
@copy_head
	%ci =l phi @copy_start 0, @copy_body %ci_next
	%cc =w csltl %ci, %nn
	jnz %cc, @copy_body, @exp_done
@copy_body
	%coff =l mul %ci, 8
	%src =l add %tmp, %coff
	%dst =l add %out, %coff
	%v =l loadl %src
	storel %v, %dst
	%ci_next =l add %ci, 1
	jmp @copy_head
@exp_done
	%e_next =l add %e, 1
	jmp @exp_head
@cleanup_start
	jmp @cleanup_head
@cleanup_head
	%qi =l phi @cleanup_start 0, @cleanup_body %qi_next
	%qc =w csltl %qi, %total
	jnz %qc, @cleanup_body, @return
@cleanup_body
	%qoff =l mul %qi, 8
	%qp =l add %tmp, %qoff
	storel 0, %qp
	%qi_next =l add %qi, 1
	jmp @cleanup_head
@return
	ret 0
}
