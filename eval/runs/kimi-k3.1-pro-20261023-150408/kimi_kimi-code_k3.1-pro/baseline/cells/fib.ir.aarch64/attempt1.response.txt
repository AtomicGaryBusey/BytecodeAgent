export function l $fib(l %n) {
@entry
    jmp @loop
@loop
    %left =l phi @entry %n, @body %leftnext
    %a =l phi @entry 0, @body %b
    %b =l phi @entry 1, @body %sum
    %done =w ceql %left, 0
    jnz %done, @exit, @body
@body
    %sum =l add %a, %b
    %leftnext =l sub %left, 1
    jmp @loop
@exit
    ret %a
}
