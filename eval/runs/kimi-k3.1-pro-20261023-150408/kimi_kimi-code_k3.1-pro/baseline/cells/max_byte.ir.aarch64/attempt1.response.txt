export function l $max_byte(l %p, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @next %i1
	%m =l phi @entry 0, @next %m1
	%end =w ceql %i, %n
	jnz %end, @done, @body
@body
	%a =l add %p, %i
	%b =l loadub %a
	%gt =w csltl %m, %b
	jnz %gt, @take, @keep
@take
	jmp @next
@keep
	jmp @next
@next
	%m1 =l phi @take %b, @keep %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
