packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Lpacket_loop:
    cmp r8, rsi
    jae .Ldone
    mov r10d, 1
    mov al, byte ptr [rdi + r8]
.Lrun_loop:
    cmp r10, 128
    jae .Lrun_done
    lea r11, [r8 + r10]
    cmp r11, rsi
    jae .Lrun_done
    cmp byte ptr [rdi + r11], al
    jne .Lrun_done
    inc r10
    jmp .Lrun_loop
.Lrun_done:
    cmp r10, 3
    jb .Lliteral
    mov rax, rcx
    sub rax, r9
    cmp rax, 2
    jb .Lfail
    mov eax, 257
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lpacket_loop
.Lliteral:
    mov r10, r8
.Lliteral_scan:
    cmp r10, rsi
    jae .Lliteral_ready
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jae .Lliteral_ready
    lea rax, [r10 + 2]
    cmp rax, rsi
    jae .Lliteral_advance
    mov al, byte ptr [rdi + r10]
    cmp byte ptr [rdi + r10 + 1], al
    jne .Lliteral_advance
    cmp byte ptr [rdi + r10 + 2], al
    je .Lliteral_ready
.Lliteral_advance:
    inc r10
    jmp .Lliteral_scan
.Lliteral_ready:
    mov r11, r10
    sub r11, r8
    mov rax, rcx
    sub rax, r9
    lea r10, [r11 + 1]
    cmp rax, r10
    jb .Lfail
    mov rax, r11
    dec rax
    mov byte ptr [rdx + r9], al
    inc r9
.Lliteral_copy:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Lliteral_copy
    jmp .Lpacket_loop
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
