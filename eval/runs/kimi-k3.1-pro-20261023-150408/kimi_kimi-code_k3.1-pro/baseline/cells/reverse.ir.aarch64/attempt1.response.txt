export function l $reverse(l %a, l %n) {
@start
	%half =l shr %n, 1
	jmp @test
@test
	%i =l phi @start 0, @body %next
	%stop =w ceql %i, %half
	jnz %stop, @exit, @body
@body
	%loff =l mul %i, 8
	%lo =l add %a, %loff
	%j =l sub %n, %i
	%j1 =l sub %j, 1
	%hoff =l mul %j1, 8
	%hi =l add %a, %hoff
	%x =l loadl %lo
	%y =l loadl %hi
	storel %y, %lo
	storel %x, %hi
	%next =l add %i, 1
	jmp @test
@exit
	ret 0
}
