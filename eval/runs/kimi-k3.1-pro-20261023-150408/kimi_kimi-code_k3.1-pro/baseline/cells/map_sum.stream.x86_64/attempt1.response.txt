map_sum:
    push rbx
    push r12
    push r13
    push r14
    sub rsp, 8
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    xor r14d, r14d
    test r12, r12
    je .Ldone
.Lloop:
    mov rdi, qword ptr [rbx]
    cld
    call r13
    add r14, rax
    add rbx, 8
    dec r12
    jne .Lloop
.Ldone:
    cld
    mov rax, r14
    add rsp, 8
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
