export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%k0 =l copy 1
	jmp @outer
@outer
	%k =l phi @start %k0, @next %k1
	%max =l phi @start %max0, @next %max1
	%done =w csgtl %k, %n
	jnz %done, @exit, @init
@init
	%x0 =l copy %k
	%steps0 =l copy 0
	jmp @inner
@inner
	%x =l phi @init %x0, @step %x1
	%steps =l phi @init %steps0, @step %steps1
	%at_one =w ceql %x, 1
	jnz %at_one, @after, @parity
@parity
	%bit =l and %x, 1
	%is_odd =w ceql %bit, 1
	jnz %is_odd, @odd, @even
@even
	%next_even =l div %x, 2
	jmp @step
@odd
	%triple =l mul %x, 3
	%next_odd =l add %triple, 1
	jmp @step
@step
	%x1 =l phi @even %next_even, @odd %next_odd
	%steps1 =l add %steps, 1
	jmp @inner
@after
	%greater =w csgtl %steps, %max
	jnz %greater, @update, @keep
@update
	%updated =l copy %steps
	jmp @next
@keep
	%kept =l copy %max
	jmp @next
@next
	%max1 =l phi @update %updated, @keep %kept
	%k1 =l add %k, 1
	jmp @outer
@exit
	ret %max
}
