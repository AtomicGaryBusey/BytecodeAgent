export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%rslot =l alloc4 4
	%lslot =l alloc4 4
	%kslot =l alloc8 8
	storel 0, %islot
	storel 0, %oslot
	jmp @outer

@outer
	%i_outer =l loadl %islot
	%more =w csltl %i_outer, %n
	jnz %more, @packet, @done

@done
	%result =l loadl %oslot
	ret %result

@packet
	storew 1, %rslot
	jmp @run_loop

@run_loop
	%r_run =w loadw %rslot
	%can_run =w csltw %r_run, 128
	jnz %can_run, @run_bounds, @run_ready

@run_bounds
	%i_run =l loadl %islot
	%r_run_l =l extuw %r_run
	%run_index =l add %i_run, %r_run_l
	%run_in_bounds =w csltl %run_index, %n
	jnz %run_in_bounds, @run_compare, @run_ready

@run_compare
	%run_base_ptr =l add %src, %i_run
	%run_base_byte =w loadub %run_base_ptr
	%run_next_ptr =l add %src, %run_index
	%run_next_byte =w loadub %run_next_ptr
	%run_same =w ceqw %run_base_byte, %run_next_byte
	jnz %run_same, @run_grow, @run_ready

@run_grow
	%r_grown =w add %r_run, 1
	storew %r_grown, %rslot
	jmp @run_loop

@run_ready
	%r_ready =w loadw %rslot
	%literal_needed =w csltw %r_ready, 3
	jnz %literal_needed, @literal_start, @repeat_check

@repeat_check
	%o_repeat =l loadl %oslot
	%room_repeat =l sub %cap, %o_repeat
	%repeat_too_big =w csltl %room_repeat, 2
	jnz %repeat_too_big, @fail, @repeat_emit

@repeat_emit
	%i_repeat =l loadl %islot
	%repeat_header =w sub 257, %r_ready
	%repeat_out =l add %out, %o_repeat
	storeb %repeat_header, %repeat_out
	%repeat_src =l add %src, %i_repeat
	%repeat_byte =w loadub %repeat_src
	%repeat_value_out =l add %repeat_out, 1
	storeb %repeat_byte, %repeat_value_out
	%o_repeat_next =l add %o_repeat, 2
	storel %o_repeat_next, %oslot
	%r_ready_l =l extuw %r_ready
	%i_repeat_next =l add %i_repeat, %r_ready_l
	storel %i_repeat_next, %islot
	jmp @outer

@literal_start
	storew 0, %lslot
	jmp @literal_condition

@literal_condition
	%i_lit =l loadl %islot
	%len_w_cond =w loadw %lslot
	%len_l_cond =l extuw %len_w_cond
	%j_lit =l add %i_lit, %len_l_cond
	%lit_has_src =w csltl %j_lit, %n
	jnz %lit_has_src, @literal_length_check, @literal_scan_done

@literal_length_check
	%lit_has_length =w csltw %len_w_cond, 128
	jnz %lit_has_length, @literal_run_init, @literal_scan_done

@literal_run_init
	storew 1, %rslot
	jmp @literal_run_loop

@literal_run_loop
	%r_lit =w loadw %rslot
	%lit_run_can_grow =w csltw %r_lit, 128
	jnz %lit_run_can_grow, @literal_run_bounds, @literal_run_ready

@literal_run_bounds
	%r_lit_l =l extuw %r_lit
	%lit_run_index =l add %j_lit, %r_lit_l
	%lit_run_in_bounds =w csltl %lit_run_index, %n
	jnz %lit_run_in_bounds, @literal_run_compare, @literal_run_ready

@literal_run_compare
	%lit_base_ptr =l add %src, %j_lit
	%lit_base_byte =w loadub %lit_base_ptr
	%lit_next_ptr =l add %src, %lit_run_index
	%lit_next_byte =w loadub %lit_next_ptr
	%lit_same =w ceqw %lit_base_byte, %lit_next_byte
	jnz %lit_same, @literal_run_grow, @literal_run_ready

@literal_run_grow
	%r_lit_grown =w add %r_lit, 1
	storew %r_lit_grown, %rslot
	jmp @literal_run_loop

@literal_run_ready
	%r_lit_ready =w loadw %rslot
	%lit_stop_for_run =w csltw 2, %r_lit_ready
	jnz %lit_stop_for_run, @literal_scan_done, @literal_advance

@literal_advance
	%len_lit_next =w add %len_w_cond, 1
	storew %len_lit_next, %lslot
	jmp @literal_condition

@literal_scan_done
	%i_lit_final =l loadl %islot
	%len_lit_final_w =w loadw %lslot
	%len_lit_final =l extuw %len_lit_final_w
	%j_lit_final =l add %i_lit_final, %len_lit_final
	%o_lit_final =l loadl %oslot
	%lit_packet_size =l add %len_lit_final, 1
	%lit_remaining =l sub %cap, %o_lit_final
	%lit_too_big =w csltl %lit_remaining, %lit_packet_size
	jnz %lit_too_big, @fail, @literal_copy_init

@literal_copy_init
	%lit_header =w sub %len_lit_final_w, 1
	%lit_out_packet =l add %out, %o_lit_final
	storeb %lit_header, %lit_out_packet
	storel 0, %kslot
	jmp @literal_copy_condition

@literal_copy_condition
	%k_lit =l loadl %kslot
	%copy_more =w csltl %k_lit, %len_lit_final
	jnz %copy_more, @literal_copy, @literal_commit

@literal_copy
	%copy_src_index =l add %i_lit_final, %k_lit
	%copy_src_ptr =l add %src, %copy_src_index
	%copy_byte =w loadub %copy_src_ptr
	%copy_out_index =l add %o_lit_final, %k_lit
	%copy_out_data_index =l add %copy_out_index, 1
	%copy_out_ptr =l add %out, %copy_out_data_index
	storeb %copy_byte, %copy_out_ptr
	%k_lit_next =l add %k_lit, 1
	storel %k_lit_next, %kslot
	jmp @literal_copy_condition

@literal_commit
	%o_lit_next =l add %o_lit_final, %lit_packet_size
	storel %o_lit_next, %oslot
	storel %j_lit_final, %islot
	jmp @outer

@fail
	ret -1
}
