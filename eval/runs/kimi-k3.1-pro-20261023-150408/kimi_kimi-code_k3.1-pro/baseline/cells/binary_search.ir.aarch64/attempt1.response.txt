export function l $binary_search(l %a, l %n, l %key) {
@entry
	jmp @loop
@loop
	%lo =l phi @entry 0, @right %next_lo, @left %lo
	%hi =l phi @entry %n, @right %hi, @left %mid
	%nonempty =w csltl %lo, %hi
	jnz %nonempty, @body, @not_found
@body
	%range =l sub %hi, %lo
	%half =l shr %range, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%address =l add %a, %offset
	%value =l loadl %address
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @right, @left
@right
	%next_lo =l add %mid, 1
	jmp @loop
@left
	jmp @loop
@found
	ret %mid
@not_found
	%result =l copy -1
	ret %result
}
