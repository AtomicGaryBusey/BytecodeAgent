export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %posp =l alloc8 8
    %writtenp =l alloc8 8
    %remp =l alloc8 8
    %payloadpp =l alloc8 8
    %modep =l alloc4 4
    %valuep =l alloc4 4
    storel 0, %posp
    storel 0, %writtenp
    jmp @loop
@loop
    %pos =l loadl %posp
    %have_packet =w csltl %pos, %n
    jnz %have_packet, @packet, @success
@success
    %result =l loadl %writtenp
    ret %result
@packet
    %header_ptr =l add %src, %pos
    %header =w loadub %header_ptr
    %after_header =l add %pos, 1
    storel %after_header, %posp
    %is_literal =w csltw %header, 128
    jnz %is_literal, @literal, @nonliteral
@literal
    %literal_header =l extuw %header
    %literal_count =l add %literal_header, 1
    %literal_end =l add %after_header, %literal_count
    %literal_truncated =w csltl %n, %literal_end
    jnz %literal_truncated, @source_error, @literal_valid
@literal_valid
    storel %literal_end, %posp
    storel %literal_count, %remp
    %literal_payload =l add %src, %after_header
    storel %literal_payload, %payloadpp
    storew 0, %modep
    jmp @emit
@nonliteral
    %is_nop =w ceqw %header, 128
    jnz %is_nop, @loop, @repeat
@repeat
    %repeat_end =l add %after_header, 1
    %repeat_truncated =w csltl %n, %repeat_end
    jnz %repeat_truncated, @source_error, @repeat_valid
@repeat_valid
    %repeat_header =l extuw %header
    %repeat_count =l sub 257, %repeat_header
    storel %repeat_end, %posp
    storel %repeat_count, %remp
    %repeat_ptr =l add %src, %after_header
    %repeat_byte =w loadub %repeat_ptr
    storew %repeat_byte, %valuep
    storew 1, %modep
    jmp @emit
@emit
    %remaining =l loadl %remp
    %has_output =w csltl 0, %remaining
    jnz %has_output, @emit_capacity, @loop
@emit_capacity
    %written =l loadl %writtenp
    %has_space =w csltl %written, %cap
    jnz %has_space, @emit_select, @overflow
@emit_select
    %mode =w loadw %modep
    %repeat_mode =w ceqw %mode, 1
    jnz %repeat_mode, @emit_repeat, @emit_literal
@emit_literal
    %payload =l loadl %payloadpp
    %literal_byte =w loadub %payload
    %next_payload =l add %payload, 1
    storel %next_payload, %payloadpp
    %literal_out =l add %out, %written
    storeb %literal_byte, %literal_out
    jmp @emit_advance
@emit_repeat
    %repeat_value =w loadw %valuep
    %repeat_out =l add %out, %written
    storeb %repeat_value, %repeat_out
    jmp @emit_advance
@emit_advance
    %next_written =l add %written, 1
    storel %next_written, %writtenp
    %next_remaining =l sub %remaining, 1
    storel %next_remaining, %remp
    jmp @emit
@overflow
    ret -1
@source_error
    ret -2
}
