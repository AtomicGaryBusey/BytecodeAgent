export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %two_nn =l mul %nn, 2
    %zero_out_start =l copy 0
    jmp @zero_out

@zero_out
    %zero_out_index =l phi @start %zero_out_start, @zero_out_body %zero_out_index_next
    %zero_out_cond =w csltl %zero_out_index, %nn
    jnz %zero_out_cond, @zero_out_body, @diag_setup

@zero_out_body
    %zero_out_offset =l mul %zero_out_index, 8
    %zero_out_pointer =l add %out, %zero_out_offset
    storel 0, %zero_out_pointer
    %zero_out_index_next =l add %zero_out_index, 1
    jmp @zero_out

@diag_setup
    %diag_start =l copy 0
    jmp @diag

@diag
    %diag_index =l phi @diag_setup %diag_start, @diag_body %diag_index_next
    %diag_cond =w csltl %diag_index, %n
    jnz %diag_cond, @diag_body, @exp_setup

@diag_body
    %diag_row =l mul %diag_index, %n
    %diag_flat =l add %diag_row, %diag_index
    %diag_offset =l mul %diag_flat, 8
    %diag_pointer =l add %out, %diag_offset
    storel 1, %diag_pointer
    %diag_index_next =l add %diag_index, 1
    jmp @diag

@exp_setup
    %exp_start =l copy 0
    jmp @exp

@exp
    %exp_index =l phi @exp_setup %exp_start, @exp_back %exp_index_next
    %exp_cond =w csltl %exp_index, %k
    jnz %exp_cond, @mul_i_setup, @zero_tmp_setup

@mul_i_setup
    %mul_i_start =l copy 0
    jmp @mul_i

@mul_i
    %mul_i_index =l phi @mul_i_setup %mul_i_start, @mul_i_back %mul_i_next
    %mul_i_cond =w csltl %mul_i_index, %n
    jnz %mul_i_cond, @mul_j_setup, @copy_setup

@mul_j_setup
    %mul_j_start =l copy 0
    jmp @mul_j

@mul_j
    %mul_j_index =l phi @mul_j_setup %mul_j_start, @mul_t_done %mul_j_next
    %mul_j_cond =w csltl %mul_j_index, %n
    jnz %mul_j_cond, @mul_t_setup, @mul_i_back

@mul_t_setup
    %mul_t_start =l copy 0
    %mul_sum_start =l copy 0
    jmp @mul_t

@mul_t
    %mul_t_index =l phi @mul_t_setup %mul_t_start, @mul_t_body %mul_t_next
    %mul_sum =l phi @mul_t_setup %mul_sum_start, @mul_t_body %mul_sum_next
    %mul_t_cond =w csltl %mul_t_index, %n
    jnz %mul_t_cond, @mul_t_body, @mul_t_done

@mul_t_body
    %mul_x_row =l mul %mul_i_index, %n
    %mul_x_index =l add %mul_x_row, %mul_t_index
    %mul_x_offset =l mul %mul_x_index, 8
    %mul_x_pointer =l add %out, %mul_x_offset
    %mul_x_value =l loadl %mul_x_pointer
    %mul_y_row =l mul %mul_t_index, %n
    %mul_y_index =l add %mul_y_row, %mul_j_index
    %mul_y_offset =l mul %mul_y_index, 8
    %mul_y_pointer =l add %a, %mul_y_offset
    %mul_y_value =l loadl %mul_y_pointer
    %mul_product =l mul %mul_x_value, %mul_y_value
    %mul_sum_next =l add %mul_sum, %mul_product
    %mul_t_next =l add %mul_t_index, 1
    jmp @mul_t

@mul_t_done
    %mul_result_row =l mul %mul_i_index, %n
    %mul_result_index =l add %mul_result_row, %mul_j_index
    %mul_result_offset =l mul %mul_result_index, 8
    %mul_result_pointer =l add %tmp, %mul_result_offset
    storel %mul_sum, %mul_result_pointer
    %mul_j_next =l add %mul_j_index, 1
    jmp @mul_j

@mul_i_back
    %mul_i_next =l add %mul_i_index, 1
    jmp @mul_i

@copy_setup
    %copy_start =l copy 0
    jmp @copy

@copy
    %copy_index =l phi @copy_setup %copy_start, @copy_body %copy_index_next
    %copy_cond =w csltl %copy_index, %nn
    jnz %copy_cond, @copy_body, @exp_back

@copy_body
    %copy_offset =l mul %copy_index, 8
    %copy_tmp_pointer =l add %tmp, %copy_offset
    %copy_out_pointer =l add %out, %copy_offset
    %copy_value =l loadl %copy_tmp_pointer
    storel %copy_value, %copy_out_pointer
    %copy_index_next =l add %copy_index, 1
    jmp @copy

@exp_back
    %exp_index_next =l add %exp_index, 1
    jmp @exp

@zero_tmp_setup
    %zero_tmp_start =l copy 0
    jmp @zero_tmp

@zero_tmp
    %zero_tmp_index =l phi @zero_tmp_setup %zero_tmp_start, @zero_tmp_body %zero_tmp_index_next
    %zero_tmp_cond =w csltl %zero_tmp_index, %two_nn
    jnz %zero_tmp_cond, @zero_tmp_body, @done

@zero_tmp_body
    %zero_tmp_offset =l mul %zero_tmp_index, 8
    %zero_tmp_pointer =l add %tmp, %zero_tmp_offset
    storel 0, %zero_tmp_pointer
    %zero_tmp_index_next =l add %zero_tmp_index, 1
    jmp @zero_tmp

@done
    ret 0
}
