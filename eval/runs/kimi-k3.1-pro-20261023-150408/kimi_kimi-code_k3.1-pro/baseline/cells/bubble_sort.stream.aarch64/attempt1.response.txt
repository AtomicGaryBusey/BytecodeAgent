bubble_sort:
cmp x1, #2
b.lo .Ldone
sub x2, x1, #1
.Louter:
mov x3, #0
mov x4, #0
.Linner:
ldr x6, [x0, x4, lsl #3]
add x7, x4, #1
ldr x8, [x0, x7, lsl #3]
cmp x6, x8
b.le .Lno_swap
str x8, [x0, x4, lsl #3]
str x6, [x0, x7, lsl #3]
mov x3, #1
.Lno_swap:
add x4, x4, #1
cmp x4, x2
b.lo .Linner
cbz x3, .Ldone
sub x2, x2, #1
cbnz x2, .Louter
.Ldone:
mov x0, #0
ret
