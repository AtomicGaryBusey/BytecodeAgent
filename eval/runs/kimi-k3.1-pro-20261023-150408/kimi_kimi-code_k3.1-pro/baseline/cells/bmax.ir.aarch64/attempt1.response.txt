export function l $bmax(l %a, l %n) {
@entry
	%first =l loadl %a
	jmp @test
@test
	%i =l phi @entry 1, @inc %i1
	%max =l phi @entry %first, @inc %max1
	%more =w csltl %i, %n
	jnz %more, @body, @exit
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%v =l loadl %p
	%greater =w csltl %max, %v
	jnz %greater, @newmax, @inc
@newmax
	jmp @inc
@inc
	%max1 =l phi @newmax %v, @body %max
	%i1 =l add %i, 1
	jmp @test
@exit
	ret %max
}
