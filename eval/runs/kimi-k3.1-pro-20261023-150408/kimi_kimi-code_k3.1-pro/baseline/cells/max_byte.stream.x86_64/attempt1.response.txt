max_byte:
    xor eax, eax
    test rsi, rsi
    jz .Lreturn
    cmp rsi, 16
    jae .Lvector
.Lscalar_loop:
    movzx edx, byte ptr [rdi]
    cmp edx, eax
    cmova eax, edx
    inc rdi
    dec rsi
    jnz .Lscalar_loop
.Lreturn:
    ret
.Lvector:
    pxor xmm0, xmm0
.Lvector_loop:
    cmp rsi, 16
    jbe .Lvector_last
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    jmp .Lvector_loop
.Lvector_last:
    movdqu xmm1, xmmword ptr [rdi + rsi - 16]
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    pextrb eax, xmm0, 0
    ret
