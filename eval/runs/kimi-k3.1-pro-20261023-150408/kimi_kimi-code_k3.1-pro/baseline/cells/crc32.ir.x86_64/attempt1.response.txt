export function l $crc32(l %p, l %n) {
@start
	%crc_init =w copy -1
	jmp @outer
@outer
	%i =l phi @start 0, @byte_done %i_next
	%crc =w phi @start %crc_init, @byte_done %crc_after_byte
	%done =w ceql %i, %n
	jnz %done, @finish, @body
@body
	%ptr =l add %p, %i
	%byte =w loadub %ptr
	%crc_in =w xor %crc, %byte
	jmp @inner
@inner
	%j =w phi @body 0, @bit %j_next
	%c =w phi @body %crc_in, @bit %c_next
	%bits_done =w ceqw %j, 8
	jnz %bits_done, @byte_done, @bit
@bit
	%lsb =w and %c, 1
	%shifted =w shr %c, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -306674912
	%c_next =w xor %shifted, %poly
	%j_next =w add %j, 1
	jmp @inner
@byte_done
	%crc_after_byte =w copy %c
	%i_next =l add %i, 1
	jmp @outer
@finish
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
