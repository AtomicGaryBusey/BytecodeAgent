dot_i32_simd:
pxor xmm4, xmm4
cmp rdx, 4
jl .Lreduce
.Lsimd_loop:
movdqu xmm0, xmmword ptr [rdi]
movdqu xmm1, xmmword ptr [rsi]
movdqa xmm2, xmm0
movdqa xmm3, xmm1
pmovsxdq xmm0, xmm0
pmovsxdq xmm1, xmm1
pmuldq xmm0, xmm1
paddq xmm4, xmm0
psrldq xmm2, 8
psrldq xmm3, 8
pmovsxdq xmm2, xmm2
pmovsxdq xmm3, xmm3
pmuldq xmm2, xmm3
paddq xmm4, xmm2
add rdi, 16
add rsi, 16
sub rdx, 4
cmp rdx, 4
jge .Lsimd_loop
.Lreduce:
movdqa xmm0, xmm4
psrldq xmm4, 8
paddq xmm0, xmm4
movq rax, xmm0
test rdx, rdx
jz .Ldone
.Lscalar_loop:
movsxd rcx, dword ptr [rdi]
movsxd r8, dword ptr [rsi]
imul rcx, r8
add rax, rcx
add rdi, 4
add rsi, 4
dec rdx
jnz .Lscalar_loop
.Ldone:
ret
