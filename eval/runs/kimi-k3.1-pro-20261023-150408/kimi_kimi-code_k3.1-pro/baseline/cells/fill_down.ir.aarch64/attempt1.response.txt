export function l $fill_down(l %a, l %n, l %start) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %next
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @exit
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l sub %start, %i
	storel %value, %ptr
	%next =l add %i, 1
	jmp @loop
@exit
	ret %n
}
