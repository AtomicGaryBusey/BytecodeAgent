export function l $isqrt(l %n) {
@start
	%lo =l alloc8 8
	%hi =l alloc8 8
	storel 0, %lo
	storel 4294967296, %hi
	jmp @loop
@loop
	%lo0 =l loadl %lo
	%hi0 =l loadl %hi
	%diff =l sub %hi0, %lo0
	%more =w cugtl %diff, 1
	jnz %more, @body, @done
@body
	%sum =l add %lo0, %hi0
	%mid =l shr %sum, 1
	%quot =l udiv %n, %mid
	%ok =w culel %mid, %quot
	jnz %ok, @take, @skip
@take
	storel %mid, %lo
	jmp @loop
@skip
	storel %mid, %hi
	jmp @loop
@done
	%ans =l loadl %lo
	ret %ans
}
