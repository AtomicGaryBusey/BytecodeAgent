export function l $bmin_empty(l %a, l %n) {
@start
	%is_empty =w ceql %n, 0
	jnz %is_empty, @empty, @nonempty
@empty
	ret 9223372036854775807
@nonempty
	%initial_min =l loadl %a
	jmp @loop
@loop
	%i =l phi @nonempty 1, @advance %next_i
	%min =l phi @nonempty %initial_min, @advance %next_min
	%at_end =w ceql %i, %n
	jnz %at_end, @done, @body
@body
	%offset =l mul %i, 8
	%address =l add %a, %offset
	%value =l loadl %address
	%is_less =w csltl %value, %min
	jnz %is_less, @update, @keep
@update
	jmp @advance
@keep
	jmp @advance
@advance
	%next_min =l phi @update %value, @keep %min
	%next_i =l add %i, 1
	jmp @loop
@done
	ret %min
}
