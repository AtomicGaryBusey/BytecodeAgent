export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%count0 =l and %y0, 1
	%p1 =l add %a, 8
	jmp @loop
@loop
	%i =l phi @start 1, @body %inext
	%p =l phi @start %p1, @body %pnext
	%min =l phi @start %y0, @body %min1
	%max =l phi @start %y0, @body %max1
	%sum =l phi @start %y0, @body %sum1
	%count =l phi @start %count0, @body %count1
	%atend =w ceql %i, %n
	jnz %atend, @finish, @body
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%count1 =l add %count, %odd
	%lt =w csltl %y, %min
	%lt0 =l extsw %lt
	%ltmask =l sub 0, %lt0
	%mindiff =l xor %y, %min
	%minmasked =l and %mindiff, %ltmask
	%min1 =l xor %min, %minmasked
	%gt =w csltl %max, %y
	%gt0 =l extsw %gt
	%gtmask =l sub 0, %gt0
	%maxdiff =l xor %y, %max
	%maxmasked =l and %maxdiff, %gtmask
	%max1 =l xor %max, %maxmasked
	%inext =l add %i, 1
	%pnext =l add %p, 8
	jmp @loop
@finish
	%out1 =l add %out, 8
	%out2 =l add %out, 16
	storel %min, %out
	storel %max, %out1
	storel %sum, %out2
	ret %count
}
