export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%state =l alloc8 16
	%depthp =l add %state, 8
	storel 0, %state
	storel 0, %depthp
	jmp @loop

@loop
	%pc =l loadl %state
	%depth =l loadl %depthp
	%at_end =w ceql %pc, %len
	jnz %at_end, @stop, @fetch

@fetch
	%insn_off =l mul %pc, 8
	%insn_ptr =l add %code, %insn_off
	%op =l loadl %insn_ptr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @stop, @check_push

@check_push
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @check_add

@check_add
	%is_add =w ceql %op, 2
	jnz %is_add, @op_add, @check_sub

@check_sub
	%is_sub =w ceql %op, 3
	jnz %is_sub, @op_sub, @check_mul

@check_mul
	%is_mul =w ceql %op, 4
	jnz %is_mul, @op_mul, @check_dup

@check_dup
	%is_dup =w ceql %op, 5
	jnz %is_dup, @op_dup, @check_swap

@check_swap
	%is_swap =w ceql %op, 6
	jnz %is_swap, @op_swap, @check_drop

@check_drop
	%is_drop =w ceql %op, 7
	jnz %is_drop, @op_drop, @check_over

@check_over
	%is_over =w ceql %op, 8
	jnz %is_over, @op_over, @check_lt

@check_lt
	%is_lt =w ceql %op, 9
	jnz %is_lt, @op_lt, @check_jz

@check_jz
	%is_jz =w ceql %op, 10
	jnz %is_jz, @op_jz, @check_jmp

@check_jmp
	%is_jmp =w ceql %op, 11
	jnz %is_jmp, @op_jmp, @other

@push
	%push_operand_index =l add %pc, 1
	%push_operand_off =l mul %push_operand_index, 8
	%push_operand_ptr =l add %code, %push_operand_off
	%push_value =l loadl %push_operand_ptr
	%push_stack_off =l mul %depth, 8
	%push_stack_ptr =l add %stack, %push_stack_off
	storel %push_value, %push_stack_ptr
	%push_depth =l add %depth, 1
	storel %push_depth, %depthp
	%push_pc =l add %pc, 2
	storel %push_pc, %state
	jmp @loop

@op_add
	%add_index =l sub %depth, 2
	%add_off =l mul %add_index, 8
	%add_aptr =l add %stack, %add_off
	%add_bptr =l add %add_aptr, 8
	%add_a =l loadl %add_aptr
	%add_b =l loadl %add_bptr
	%add_result =l add %add_a, %add_b
	storel %add_result, %add_aptr
	%add_depth =l sub %depth, 1
	storel %add_depth, %depthp
	%add_pc =l add %pc, 1
	storel %add_pc, %state
	jmp @loop

@op_sub
	%sub_index =l sub %depth, 2
	%sub_off =l mul %sub_index, 8
	%sub_aptr =l add %stack, %sub_off
	%sub_bptr =l add %sub_aptr, 8
	%sub_a =l loadl %sub_aptr
	%sub_b =l loadl %sub_bptr
	%sub_result =l sub %sub_a, %sub_b
	storel %sub_result, %sub_aptr
	%sub_depth =l sub %depth, 1
	storel %sub_depth, %depthp
	%sub_pc =l add %pc, 1
	storel %sub_pc, %state
	jmp @loop

@op_mul
	%mul_index =l sub %depth, 2
	%mul_off =l mul %mul_index, 8
	%mul_aptr =l add %stack, %mul_off
	%mul_bptr =l add %mul_aptr, 8
	%mul_a =l loadl %mul_aptr
	%mul_b =l loadl %mul_bptr
	%mul_result =l mul %mul_a, %mul_b
	storel %mul_result, %mul_aptr
	%mul_depth =l sub %depth, 1
	storel %mul_depth, %depthp
	%mul_pc =l add %pc, 1
	storel %mul_pc, %state
	jmp @loop

@op_dup
	%dup_index =l sub %depth, 1
	%dup_off =l mul %dup_index, 8
	%dup_src =l add %stack, %dup_off
	%dup_value =l loadl %dup_src
	%dup_dst =l add %dup_src, 8
	storel %dup_value, %dup_dst
	%dup_depth =l add %depth, 1
	storel %dup_depth, %depthp
	%dup_pc =l add %pc, 1
	storel %dup_pc, %state
	jmp @loop

@op_swap
	%swap_index =l sub %depth, 2
	%swap_off =l mul %swap_index, 8
	%swap_aptr =l add %stack, %swap_off
	%swap_bptr =l add %swap_aptr, 8
	%swap_a =l loadl %swap_aptr
	%swap_b =l loadl %swap_bptr
	storel %swap_b, %swap_aptr
	storel %swap_a, %swap_bptr
	%swap_pc =l add %pc, 1
	storel %swap_pc, %state
	jmp @loop

@op_drop
	%drop_depth =l sub %depth, 1
	storel %drop_depth, %depthp
	%drop_pc =l add %pc, 1
	storel %drop_pc, %state
	jmp @loop

@op_over
	%over_index =l sub %depth, 2
	%over_off =l mul %over_index, 8
	%over_src =l add %stack, %over_off
	%over_value =l loadl %over_src
	%over_dst =l add %over_src, 16
	storel %over_value, %over_dst
	%over_depth =l add %depth, 1
	storel %over_depth, %depthp
	%over_pc =l add %pc, 1
	storel %over_pc, %state
	jmp @loop

@op_lt
	%lt_index =l sub %depth, 2
	%lt_off =l mul %lt_index, 8
	%lt_aptr =l add %stack, %lt_off
	%lt_bptr =l add %lt_aptr, 8
	%lt_a =l loadl %lt_aptr
	%lt_b =l loadl %lt_bptr
	%lt_cmp =w csltl %lt_a, %lt_b
	%lt_result =l extuw %lt_cmp
	storel %lt_result, %lt_aptr
	%lt_depth =l sub %depth, 1
	storel %lt_depth, %depthp
	%lt_pc =l add %pc, 1
	storel %lt_pc, %state
	jmp @loop

@op_jz
	%jz_index =l sub %depth, 1
	%jz_off =l mul %jz_index, 8
	%jz_ptr =l add %stack, %jz_off
	%jz_value =l loadl %jz_ptr
	%jz_depth =l sub %depth, 1
	storel %jz_depth, %depthp
	%jz_operand_index =l add %pc, 1
	%jz_operand_off =l mul %jz_operand_index, 8
	%jz_operand_ptr =l add %code, %jz_operand_off
	%jz_target =l loadl %jz_operand_ptr
	%jz_is_zero =w ceql %jz_value, 0
	jnz %jz_is_zero, @jz_taken, @jz_fallthrough

@jz_taken
	storel %jz_target, %state
	jmp @loop

@jz_fallthrough
	%jz_pc =l add %pc, 2
	storel %jz_pc, %state
	jmp @loop

@op_jmp
	%jmp_operand_index =l add %pc, 1
	%jmp_operand_off =l mul %jmp_operand_index, 8
	%jmp_operand_ptr =l add %code, %jmp_operand_off
	%jmp_target =l loadl %jmp_operand_ptr
	storel %jmp_target, %state
	jmp @loop

@other
	%other_pc =l add %pc, 1
	storel %other_pc, %state
	jmp @loop

@stop
	%final_depth =l loadl %depthp
	storel %final_depth, %state
	jmp @clear_loop

@clear_loop
	%clear_index =l loadl %state
	%clear_done =w ceql %clear_index, %cap
	jnz %clear_done, @after_clear, @clear_body

@clear_body
	%clear_off =l mul %clear_index, 8
	%clear_ptr =l add %stack, %clear_off
	storel 0, %clear_ptr
	%clear_next =l add %clear_index, 1
	storel %clear_next, %state
	jmp @clear_loop

@after_clear
	%return_depth =l loadl %depthp
	%return_empty =w ceql %return_depth, 0
	jnz %return_empty, @return_zero, @return_top

@return_top
	%return_index =l sub %return_depth, 1
	%return_off =l mul %return_index, 8
	%return_ptr =l add %stack, %return_off
	%return_value =l loadl %return_ptr
	ret %return_value

@return_zero
	ret 0
}
