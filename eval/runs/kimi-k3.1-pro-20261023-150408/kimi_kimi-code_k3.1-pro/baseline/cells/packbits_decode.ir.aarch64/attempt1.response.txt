export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet
@packet
	%index =l phi @start 0, @advance %next_index
	%written =l phi @start 0, @advance %next_written
	%source_done =w cslel %n, %index
	jnz %source_done, @success, @read_header
@success
	ret %written
@read_header
	%header_address =l add %src, %index
	%header =w loadsb %header_address
	%after_header =l add %index, 1
	%is_negative =w csltw %header, 0
	jnz %is_negative, @negative_header, @literal
@negative_header
	%is_noop =w ceqw %header, -128
	jnz %is_noop, @noop, @repeat
@noop
	jmp @advance
@literal
	%literal_length_word =w add %header, 1
	%literal_length =l extsw %literal_length_word
	%literal_end =l add %after_header, %literal_length
	%literal_source_ok =w cslel %literal_end, %n
	jnz %literal_source_ok, @literal_valid, @source_error
@literal_valid
	%literal_data =l add %src, %after_header
	jmp @literal_loop
@literal_loop
	%literal_left =l phi @literal_valid %literal_length, @literal_write %literal_left_next
	%literal_position =l phi @literal_valid %literal_data, @literal_write %literal_position_next
	%literal_written =l phi @literal_valid %written, @literal_write %literal_written_next
	%literal_output_done =w ceql %literal_left, 0
	jnz %literal_output_done, @literal_done, @literal_check_cap
@literal_check_cap
	%literal_overflow =w cslel %cap, %literal_written
	jnz %literal_overflow, @output_overflow, @literal_write
@literal_write
	%literal_byte =w loadub %literal_position
	%literal_output_address =l add %out, %literal_written
	storeb %literal_byte, %literal_output_address
	%literal_position_next =l add %literal_position, 1
	%literal_written_next =l add %literal_written, 1
	%literal_left_next =l sub %literal_left, 1
	jmp @literal_loop
@literal_done
	jmp @advance
@repeat
	%repeat_end =l add %after_header, 1
	%repeat_source_ok =w cslel %repeat_end, %n
	jnz %repeat_source_ok, @repeat_valid, @source_error
@repeat_valid
	%repeat_count_word =w sub 1, %header
	%repeat_count =l extsw %repeat_count_word
	%repeat_value_address =l add %src, %after_header
	%repeat_value =w loadub %repeat_value_address
	jmp @repeat_loop
@repeat_loop
	%repeat_left =l phi @repeat_valid %repeat_count, @repeat_write %repeat_left_next
	%repeat_written =l phi @repeat_valid %written, @repeat_write %repeat_written_next
	%repeat_output_done =w ceql %repeat_left, 0
	jnz %repeat_output_done, @repeat_done, @repeat_check_cap
@repeat_check_cap
	%repeat_overflow =w cslel %cap, %repeat_written
	jnz %repeat_overflow, @output_overflow, @repeat_write
@repeat_write
	%repeat_output_address =l add %out, %repeat_written
	storeb %repeat_value, %repeat_output_address
	%repeat_written_next =l add %repeat_written, 1
	%repeat_left_next =l sub %repeat_left, 1
	jmp @repeat_loop
@repeat_done
	jmp @advance
@advance
	%next_index =l phi @noop %after_header, @literal_done %literal_end, @repeat_done %repeat_end
	%next_written =l phi @noop %written, @literal_done %literal_written, @repeat_done %repeat_written
	jmp @packet
@source_error
	%source_error_value =l copy -2
	ret %source_error_value
@output_overflow
	%output_overflow_value =l copy -1
	ret %output_overflow_value
}
