memmove_i64:
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jbe .Lforward_start
mov rcx, rdx
.Lbackward:
mov rax, qword ptr [rsi + rcx*8 - 8]
mov qword ptr [rdi + rcx*8 - 8], rax
dec rcx
jnz .Lbackward
jmp .Ldone
.Lforward_start:
xor ecx, ecx
.Lforward:
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
inc rcx
cmp rcx, rdx
jb .Lforward
.Ldone:
xor eax, eax
ret
