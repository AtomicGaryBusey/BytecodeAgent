export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %i_next
    %count =w phi @start 0, @body %count_next
    %more =w cultl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %p =l add %a, %offset
    %v =l loadl %p
    %gt =w cugtl %v, %t
    %count_next =w add %count, %gt
    %i_next =l add %i, 1
    jmp @loop
@done
    %result =l extsw %count
    ret %result
}
