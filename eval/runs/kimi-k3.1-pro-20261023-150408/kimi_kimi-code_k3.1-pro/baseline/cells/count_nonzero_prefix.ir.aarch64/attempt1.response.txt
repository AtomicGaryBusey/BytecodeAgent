export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @next %i1
	%end =w ceql %i, %n
	jnz %end, @exit, @check
@check
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%zero =w ceql %v, 0
	jnz %zero, @exit, @next
@next
	%i1 =l add %i, 1
	jmp @loop
@exit
	ret %i
}
