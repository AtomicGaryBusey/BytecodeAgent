packbits_decode:
xor r8d, r8d
.Lpacket_loop:
test rsi, rsi
jz .Lsuccess
movsx eax, byte ptr [rdi]
inc rdi
dec rsi
cmp eax, -128
je .Lpacket_loop
test eax, eax
js .Lrepeat
mov r9d, eax
inc r9d
cmp rsi, r9
jb .Lsource_error
cmp rcx, r9
jb .Lliteral_short
sub rsi, r9
add r8, r9
sub rcx, r9
.Lliteral_loop:
mov al, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec r9
jnz .Lliteral_loop
jmp .Lpacket_loop
.Lliteral_short:
test rcx, rcx
jz .Loutput_error
.Lliteral_short_loop:
mov al, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec rcx
jnz .Lliteral_short_loop
jmp .Loutput_error
.Lrepeat:
neg eax
inc eax
mov r9d, eax
test rsi, rsi
jz .Lsource_error
mov r10b, byte ptr [rdi]
cmp rcx, r9
jb .Lrepeat_short
inc rdi
dec rsi
add r8, r9
sub rcx, r9
.Lrepeat_loop:
mov byte ptr [rdx], r10b
inc rdx
dec r9
jnz .Lrepeat_loop
jmp .Lpacket_loop
.Lrepeat_short:
test rcx, rcx
jz .Loutput_error
.Lrepeat_short_loop:
mov byte ptr [rdx], r10b
inc rdx
dec rcx
jnz .Lrepeat_short_loop
.Loutput_error:
mov rax, -1
ret
.Lsource_error:
mov rax, -2
ret
.Lsuccess:
mov rax, r8
ret
