export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%empty =w ceql %n, 0
	jnz %empty, @zero, @loop
@loop
	%left =l phi @entry %n, @loop %left_next
	%sptr =l phi @entry %src, @loop %src_next
	%dptr =l phi @entry %dst, @loop %dst_next
	%count =l phi @entry 0, @loop %count_next
	%x =l loadl %sptr
	%r =l call %f(l %x)
	storel %r, %dptr
	%less =w csltl %r, %x
	%less_l =l extsw %less
	%count_next =l add %count, %less_l
	%src_next =l add %sptr, 8
	%dst_next =l add %dptr, 8
	%left_next =l sub %left, 1
	%done =w ceql %left_next, 0
	jnz %done, @return, @loop
@return
	ret %count_next
@zero
	ret 0
}
