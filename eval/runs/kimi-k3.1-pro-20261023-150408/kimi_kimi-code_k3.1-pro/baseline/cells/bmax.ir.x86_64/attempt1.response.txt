export function l $bmax(l %a, l %n) {
@start
	%initial =l loadl %a
	jmp @loop
@loop
	%max =l phi @start %initial, @next %maxnext
	%i =l phi @start 1, @next %inext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w csltl %max, %value
	jnz %greater, @update, @next
@update
	jmp @next
@next
	%maxnext =l phi @body %max, @update %value
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
