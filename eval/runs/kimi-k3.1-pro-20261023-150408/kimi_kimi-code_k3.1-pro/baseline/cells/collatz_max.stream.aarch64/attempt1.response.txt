collatz_max:
mov x1, #1
mov x2, #0
.Lnext:
mov x3, x1
mov x4, #0
.Lsteps:
cmp x3, #1
b.eq .Ldone
tbz x3, #0, .Leven
add x3, x3, x3, lsl #1
add x3, x3, #1
lsr x3, x3, #1
add x4, x4, #2
b .Lsteps
.Leven:
lsr x3, x3, #1
add x4, x4, #1
b .Lsteps
.Ldone:
cmp x4, x2
csel x2, x4, x2, gt
add x1, x1, #1
cmp x1, x0
b.le .Lnext
mov x0, x2
ret
