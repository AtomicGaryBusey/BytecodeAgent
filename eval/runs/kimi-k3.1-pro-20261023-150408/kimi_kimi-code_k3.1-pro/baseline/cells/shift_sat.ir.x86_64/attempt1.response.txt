export function l $shift_sat(l %x, l %n) {
@start
	jmp @loop
@loop
	%left =l phi @start %x, @body %leftnext
	%right =l phi @start %x, @body %rightnext
	%count =l phi @start %n, @body %countnext
	%done =w ceql %count, 0
	jnz %done, @exit, @body
@body
	%leftnext =l shl %left, 1
	%rightnext =l sar %right, 1
	%countnext =l sub %count, 1
	jmp @loop
@exit
	%result =l add %left, %right
	ret %result
}
