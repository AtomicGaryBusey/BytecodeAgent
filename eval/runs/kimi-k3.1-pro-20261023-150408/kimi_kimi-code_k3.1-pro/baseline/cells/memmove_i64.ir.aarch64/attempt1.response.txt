export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nzero =w ceql %n, 0
	jnz %nzero, @done, @direction
@direction
	%forward =w cultl %dst, %src
	jnz %forward, @fentry, @bentry
@fentry
	jmp @fcond
@fcond
	%fi =l phi @fentry 0, @fbody %finext
	%fdone =w ceql %fi, %n
	jnz %fdone, @done, @fbody
@fbody
	%foff =l mul %fi, 8
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fvalue =l loadl %fsrc
	storel %fvalue, %fdst
	%finext =l add %fi, 1
	jmp @fcond
@bentry
	jmp @bcond
@bcond
	%bi =l phi @bentry %n, @bbody %biprev
	%bzero =w ceql %bi, 0
	jnz %bzero, @done, @bbody
@bbody
	%biprev =l sub %bi, 1
	%boff =l mul %biprev, 8
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bvalue =l loadl %bsrc
	storel %bvalue, %bdst
	jmp @bcond
@done
	ret 0
}
