export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @zero, @checkneg
@checkneg
	%bn =w ceql %b, -1
	jnz %bn, @negone, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@zero
	storel %a, %rem
	ret 0
@negone
	%qn =l sub 0, %a
	storel 0, %rem
	ret %qn
}
