export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @outer_init
@outer_init
	%i0 =l sub %n, 1
	jmp @outer
@outer
	%i =l phi @outer_init %i0, @outer_next %i1
	%outer_cond =w csltl 0, %i
	jnz %outer_cond, @inner, @done
@inner
	%j =l phi @outer 0, @inner_next %j1
	%inner_cond =w csltl %j, %i
	jnz %inner_cond, @body, @outer_next
@body
	%offset =l mul %j, 8
	%p =l add %a, %offset
	%next_offset =l add %offset, 8
	%q =l add %a, %next_offset
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @do_swap, @inner_next
@do_swap
	storel %y, %p
	storel %x, %q
	jmp @inner_next
@inner_next
	%j1 =l add %j, 1
	jmp @inner
@outer_next
	%i1 =l sub %i, 1
	jmp @outer
@done
	ret 0
}
