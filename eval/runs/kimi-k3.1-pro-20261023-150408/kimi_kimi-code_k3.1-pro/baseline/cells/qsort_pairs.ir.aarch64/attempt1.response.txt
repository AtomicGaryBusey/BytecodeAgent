export function l $qsort_pairs(l %a, l %n) {
@entry
    %islot =l alloc8 8
    %jslot =l alloc8 8
    %base =w csltl %n, 2
    jnz %base, @done, @partition
@done
    ret 0
@partition
    %last =l sub %n, 1
    %lastoff =l mul %last, 16
    %pp =l add %a, %lastoff
    %ppval =l add %pp, 8
    %pkey =l loadl %pp
    %pval =l loadl %ppval
    storel 0, %islot
    storel 0, %jslot
    jmp @loop
@loop
    %j =l loadl %jslot
    %more =w csltl %j, %last
    jnz %more, @check, @finish
@check
    %joff =l mul %j, 16
    %pj =l add %a, %joff
    %pjval =l add %pj, 8
    %jkey =l loadl %pj
    %jval =l loadl %pjval
    %keyless =w csltl %jkey, %pkey
    jnz %keyless, @take, @keyequal
@keyequal
    %keyeq =w ceql %jkey, %pkey
    jnz %keyeq, @valuecheck, @next
@valuecheck
    %valless =w csltl %jval, %pval
    jnz %valless, @take, @next
@take
    %i =l loadl %islot
    %ioff =l mul %i, 16
    %pi =l add %a, %ioff
    %pival =l add %pi, 8
    %ikey =l loadl %pi
    %ival =l loadl %pival
    storel %jkey, %pi
    storel %jval, %pival
    storel %ikey, %pj
    storel %ival, %pjval
    %inext =l add %i, 1
    storel %inext, %islot
    jmp @next
@next
    %jold =l loadl %jslot
    %jnext =l add %jold, 1
    storel %jnext, %jslot
    jmp @loop
@finish
    %ifinal =l loadl %islot
    %ifoff =l mul %ifinal, 16
    %pif =l add %a, %ifoff
    %pifval =l add %pif, 8
    %ifkey =l loadl %pif
    %ifval =l loadl %pifval
    storel %pkey, %pif
    storel %pval, %pifval
    storel %ifkey, %pp
    storel %ifval, %ppval
    %r1 =l call $qsort_pairs(l %a, l %ifinal)
    %iafter =l loadl %islot
    %rightfirst =l add %iafter, 1
    %rightoff =l mul %rightfirst, 16
    %rightptr =l add %a, %rightoff
    %rightn =l sub %n, %rightfirst
    %r2 =l call $qsort_pairs(l %rightptr, l %rightn)
    ret 0
}
