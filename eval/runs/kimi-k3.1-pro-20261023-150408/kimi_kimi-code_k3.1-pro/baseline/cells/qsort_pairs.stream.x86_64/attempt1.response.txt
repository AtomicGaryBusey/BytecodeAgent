jle .Lzero

push r13

mov r13, rsi
.Lsort_again:
cmp r13, 1
jle .Ldone
lea rax, [r13 - 1]
shl rax, 4
lea rbx, [r12 + rax]
mov rax, qword ptr [rbx]
mov qword ptr [rsp], rax
mov rax, qword ptr [rbx + 8]

.Lpartition_loop:
lea rax, [r13 - 1]
cmp r15, rax

cmp rax, qword ptr [rsp]

mov qword ptr [rdx + 8], rax

.Lnext:
inc r15
jmp .Lpartition_loop
.Lpartition_done:
mov rax, r14
shl rax, 4
lea rdx, [r12 + rax]
lea rax, [r13 - 1]

mov qword ptr [rbx + 8], rax

sub rax, r14
dec rax
cmp r14, rax

call qsort_pairs

pop r13
