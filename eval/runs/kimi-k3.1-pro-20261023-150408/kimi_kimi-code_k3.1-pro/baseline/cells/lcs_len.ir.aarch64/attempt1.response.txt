export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	%zero =l copy 0
	%nbytes =l mul %n, 8
	%rowend =l add %row, %nbytes
	jmp @init
@init
	%ip =l phi @entry %row, @init %ip1
	storel %zero, %ip
	%ip1 =l add %ip, 8
	%init_done =w ceql %ip, %rowend
	jnz %init_done, @outer, @init
@outer
	%i =l phi @init %zero, @outer_latch %i1
	%outer_done =w ceql %i, %m
	jnz %outer_done, @finish, @outer_body
@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	%nzero =w ceql %n, %zero
	jnz %nzero, @outer_latch, @inner_setup
@inner_setup
	%row1 =l add %row, 8
	jmp @inner
@inner
	%diag =l phi @inner_setup %zero, @inner_next %up
	%rjp =l phi @inner_setup %row1, @inner_next %rjp1
	%yp =l phi @inner_setup %y, @inner_next %yp1
	%up =l loadl %rjp
	%leftp =l sub %rjp, 8
	%left =l loadl %leftp
	%yb =w loadub %yp
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @mismatch
@match
	%matched =l add %diag, 1
	storel %matched, %rjp
	jmp @inner_next
@mismatch
	%left_greater =w csltl %up, %left
	jnz %left_greater, @use_left, @use_up
@use_left
	storel %left, %rjp
	jmp @inner_next
@use_up
	storel %up, %rjp
	jmp @inner_next
@inner_next
	%rjp1 =l add %rjp, 8
	%yp1 =l add %yp, 1
	%inner_done =w ceql %rjp, %rowend
	jnz %inner_done, @outer_latch, @inner
@outer_latch
	%i1 =l add %i, 1
	jmp @outer
@finish
	%result =l loadl %rowend
	ret %result
}
