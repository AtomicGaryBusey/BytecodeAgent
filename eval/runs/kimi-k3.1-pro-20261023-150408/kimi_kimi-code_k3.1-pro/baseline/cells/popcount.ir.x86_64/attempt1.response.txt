export function l $popcount(l %x) {
@start
	%shift1 =l shr %x, 1
	%mask1 =l and %shift1, 6148914691236517205
	%pairs =l sub %x, %mask1
	%low2 =l and %pairs, 3689348814741910323
	%shift2 =l shr %pairs, 2
	%high2 =l and %shift2, 3689348814741910323
	%nibbles =l add %low2, %high2
	%shift4 =l shr %nibbles, 4
	%sum4 =l add %nibbles, %shift4
	%bytes =l and %sum4, 1085102592571150095
	%product =l mul %bytes, 72340172838076673
	%count =l shr %product, 56
	ret %count
}
