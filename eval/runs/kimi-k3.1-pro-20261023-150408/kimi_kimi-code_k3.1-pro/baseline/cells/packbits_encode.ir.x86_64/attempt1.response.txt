export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%jslot =l alloc8 8
	%kslot =l alloc8 8
	%rslot =l alloc4 4
	%bslot =l alloc4 4
	%lenslot =l alloc4 4
	storel 0, %islot
	storel 0, %oslot
	jmp @outer

@outer
	%outer_i =l loadl %islot
	%outer_more =w csltl %outer_i, %n
	jnz %outer_more, @run_init, @done

@done
	%done_o =l loadl %oslot
	ret %done_o

@run_init
	%run_i =l loadl %islot
	storew 1, %rslot
	%run_src =l add %src, %run_i
	%run_first =w loadub %run_src
	storew %run_first, %bslot
	jmp @run_test

@run_test
	%run_test_r =w loadw %rslot
	%run_can_grow =w csltw %run_test_r, 128
	jnz %run_can_grow, @run_bound, @run_done

@run_bound
	%run_bound_i =l loadl %islot
	%run_bound_r =w loadw %rslot
	%run_bound_rl =l extuw %run_bound_r
	%run_next_index =l add %run_bound_i, %run_bound_rl
	%run_in_bounds =w csltl %run_next_index, %n
	jnz %run_in_bounds, @run_load, @run_done

@run_load
	%run_next_ptr =l add %src, %run_next_index
	%run_next_byte =w loadub %run_next_ptr
	%run_first_byte =w loadw %bslot
	%run_same =w ceqw %run_next_byte, %run_first_byte
	jnz %run_same, @run_inc, @run_done

@run_inc
	%run_old =w loadw %rslot
	%run_new =w add %run_old, 1
	storew %run_new, %rslot
	jmp @run_test

@run_done
	%run_final =w loadw %rslot
	%run_is_short =w csltw %run_final, 3
	jnz %run_is_short, @literal_begin, @repeat_check

@repeat_check
	%rep_o =l loadl %oslot
	%rep_room =l sub %cap, %rep_o
	%rep_no_fit =w csltl %rep_room, 2
	jnz %rep_no_fit, @fail, @repeat_write

@repeat_write
	%rep_r =w loadw %rslot
	%rep_header =w sub 257, %rep_r
	%rep_write_o =l loadl %oslot
	%rep_out =l add %out, %rep_write_o
	storeb %rep_header, %rep_out
	%rep_value_out =l add %rep_out, 1
	%rep_byte =w loadw %bslot
	storeb %rep_byte, %rep_value_out
	%rep_new_o =l add %rep_write_o, 2
	storel %rep_new_o, %oslot
	%rep_i =l loadl %islot
	%rep_rl =l extuw %rep_r
	%rep_new_i =l add %rep_i, %rep_rl
	storel %rep_new_i, %islot
	jmp @outer

@literal_begin
	%literal_i =l loadl %islot
	storel %literal_i, %jslot
	storew 0, %lenslot
	jmp @literal_test

@literal_test
	%literal_test_j =l loadl %jslot
	%literal_j_in_bounds =w csltl %literal_test_j, %n
	jnz %literal_j_in_bounds, @literal_length_test, @literal_ready

@literal_length_test
	%literal_test_len =w loadw %lenslot
	%literal_len_ok =w csltw %literal_test_len, 128
	jnz %literal_len_ok, @literal_run_init, @literal_ready

@literal_run_init
	%literal_run_j =l loadl %jslot
	storew 1, %rslot
	%literal_run_ptr =l add %src, %literal_run_j
	%literal_run_byte =w loadub %literal_run_ptr
	storew %literal_run_byte, %bslot
	jmp @literal_run_test

@literal_run_test
	%literal_run_test_r =w loadw %rslot
	%literal_run_can_grow =w csltw %literal_run_test_r, 128
	jnz %literal_run_can_grow, @literal_run_bound, @literal_run_done

@literal_run_bound
	%literal_run_bound_j =l loadl %jslot
	%literal_run_bound_r =w loadw %rslot
	%literal_run_bound_rl =l extuw %literal_run_bound_r
	%literal_run_next_index =l add %literal_run_bound_j, %literal_run_bound_rl
	%literal_run_in_bounds =w csltl %literal_run_next_index, %n
	jnz %literal_run_in_bounds, @literal_run_load, @literal_run_done

@literal_run_load
	%literal_run_next_ptr =l add %src, %literal_run_next_index
	%literal_run_next_byte =w loadub %literal_run_next_ptr
	%literal_run_first_byte =w loadw %bslot
	%literal_run_same =w ceqw %literal_run_next_byte, %literal_run_first_byte
	jnz %literal_run_same, @literal_run_inc, @literal_run_done

@literal_run_inc
	%literal_run_old =w loadw %rslot
	%literal_run_new =w add %literal_run_old, 1
	storew %literal_run_new, %rslot
	jmp @literal_run_test

@literal_run_done
	%literal_run_final =w loadw %rslot
	%literal_run_is_short =w csltw %literal_run_final, 3
	jnz %literal_run_is_short, @literal_advance, @literal_ready

@literal_advance
	%literal_old_j =l loadl %jslot
	%literal_new_j =l add %literal_old_j, 1
	storel %literal_new_j, %jslot
	%literal_old_len =w loadw %lenslot
	%literal_new_len =w add %literal_old_len, 1
	storew %literal_new_len, %lenslot
	jmp @literal_test

@literal_ready
	%literal_ready_len_w =w loadw %lenslot
	%literal_ready_len =l extuw %literal_ready_len_w
	%literal_packet_size =l add %literal_ready_len, 1
	%literal_ready_o =l loadl %oslot
	%literal_ready_room =l sub %cap, %literal_ready_o
	%literal_no_fit =w csltl %literal_ready_room, %literal_packet_size
	jnz %literal_no_fit, @fail, @literal_write

@literal_write
	%literal_write_len =w loadw %lenslot
	%literal_header =w sub %literal_write_len, 1
	%literal_write_o =l loadl %oslot
	%literal_write_out =l add %out, %literal_write_o
	storeb %literal_header, %literal_write_out
	storel 0, %kslot
	jmp @literal_copy_test

@literal_copy_test
	%copy_k =l loadl %kslot
	%copy_len_w =w loadw %lenslot
	%copy_len =l extuw %copy_len_w
	%copy_more =w csltl %copy_k, %copy_len
	jnz %copy_more, @literal_copy_body, @literal_copy_done

@literal_copy_body
	%copy_i =l loadl %islot
	%copy_src_index =l add %copy_i, %copy_k
	%copy_src_ptr =l add %src, %copy_src_index
	%copy_byte =w loadub %copy_src_ptr
	%copy_o =l loadl %oslot
	%copy_dest_start =l add %copy_o, 1
	%copy_dest_index =l add %copy_dest_start, %copy_k
	%copy_dest_ptr =l add %out, %copy_dest_index
	storeb %copy_byte, %copy_dest_ptr
	%copy_next_k =l add %copy_k, 1
	storel %copy_next_k, %kslot
	jmp @literal_copy_test

@literal_copy_done
	%copy_done_o =l loadl %oslot
	%copy_done_len_w =w loadw %lenslot
	%copy_done_len =l extuw %copy_done_len_w
	%copy_done_size =l add %copy_done_len, 1
	%copy_done_new_o =l add %copy_done_o, %copy_done_size
	storel %copy_done_new_o, %oslot
	%copy_done_j =l loadl %jslot
	storel %copy_done_j, %islot
	jmp @outer

@fail
	ret -1
}
