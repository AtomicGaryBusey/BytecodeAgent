export function l $binary_search(l %a, l %n, l %key) {
@entry
	jmp @loop

@loop
	%lo =l phi @entry 0, @update %next_lo
	%hi =l phi @entry %n, @update %next_hi
	%more =w csltl %lo, %hi
	jnz %more, @body, @missing

@body
	%span =l sub %hi, %lo
	%half =l div %span, 2
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%address =l add %a, %offset
	%value =l loadl %address
	%equal =w ceql %value, %key
	jnz %equal, @found, @unequal

@unequal
	%less =w csltl %value, %key
	jnz %less, @right, @left

@right
	%mid_plus_one =l add %mid, 1
	jmp @update

@left
	jmp @update

@update
	%next_lo =l phi @right %mid_plus_one, @left %lo
	%next_hi =l phi @right %hi, @left %mid
	jmp @loop

@found
	ret %mid

@missing
	ret -1
}
