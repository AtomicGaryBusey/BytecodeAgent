push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor r15d, r15d
test r13, r13
jz .Ldone
.Lloop:
mov rbp, qword ptr [r12]
mov rdi, rbp
call r14
mov qword ptr [rbx], rax
cmp rax, rbp
jge .Lnot_less
inc r15
.Lnot_less:
add r12, 8
add rbx, 8
dec r13
jnz .Lloop
.Ldone:
mov rax, r15
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
