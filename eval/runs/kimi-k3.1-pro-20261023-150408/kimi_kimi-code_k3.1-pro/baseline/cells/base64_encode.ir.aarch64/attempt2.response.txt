export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%p =l phi @start %src, @nextgroup %pnext
	%q =l phi @start %out, @nextgroup %qnext
	%r =l phi @start %n, @nextgroup %rnext
	%empty =w ceql %r, 0
	jnz %empty, @emptydone, @loadfirst
@emptydone
	%len =l sub %q, %out
	ret %len
@loadfirst
	%b0 =l loadub %p
	%v0 =l shl %b0, 16
	%isone =w ceql %r, 1
	jnz %isone, @emitsetup, @loadsecond
@loadsecond
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%b1shift =l shl %b1, 8
	%v01 =l or %v0, %b1shift
	%istwo =w ceql %r, 2
	jnz %istwo, @emitsetup, @loadthird
@loadthird
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%vfull =l or %v01, %b2
	jmp @emitsetup
@emitsetup
	%initial =l phi @loadfirst %v0, @loadsecond %v01, @loadthird %vfull
	%m =l phi @loadfirst 2, @loadsecond 3, @loadthird 4
	%consumed =l phi @loadfirst 1, @loadsecond 2, @loadthird 3
	jmp @emit
@emit
	%k =l phi @emitsetup 0, @chardone %knext
	%work =l phi @emitsetup %initial, @chardone %worknext
	%needchar =w csltl %k, %m
	jnz %needchar, @encode, @pad
@pad
	%padaddr =l add %q, %k
	storeb 61, %padaddr
	jmp @chardone
@encode
	%top =l shr %work, 18
	%index =l and %top, 63
	%isupper =w csltl %index, 26
	jnz %isupper, @upper, @testlower
@upper
	%upperch =l add %index, 65
	jmp @havechar
@testlower
	%islower =w csltl %index, 52
	jnz %islower, @lower, @testdigit
@lower
	%lowerch =l add %index, 71
	jmp @havechar
@testdigit
	%isdigit =w csltl %index, 62
	jnz %isdigit, @digit, @special
@digit
	%digitch =l sub %index, 4
	jmp @havechar
@special
	%isplus =w ceql %index, 62
	jnz %isplus, @plus, @slash
@plus
	jmp @havechar
@slash
	jmp @havechar
@havechar
	%ch =l phi @upper %upperch, @lower %lowerch, @digit %digitch, @plus 43, @slash 47
	%charaddr =l add %q, %k
	storeb %ch, %charaddr
	jmp @chardone
@chardone
	%knext =l add %k, 1
	%worknext =l shl %work, 6
	%done =w ceql %knext, 4
	jnz %done, @nextgroup, @emit
@nextgroup
	%pnext =l add %p, %consumed
	%qnext =l add %q, 4
	%rnext =l sub %r, %consumed
	jmp @group
}
