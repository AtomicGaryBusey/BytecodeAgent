export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bzero =w ceql %b, 0
	jnz %bzero, @zero, @checkoverflow
@zero
	storel %a, %rem
	ret 0
@checkoverflow
	%bminusone =w ceql %b, -1
	jnz %bminusone, @checkmin, @divide
@checkmin
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	jnz %amin, @overflow, @divide
@overflow
	storel 0, %rem
	ret %min
@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
