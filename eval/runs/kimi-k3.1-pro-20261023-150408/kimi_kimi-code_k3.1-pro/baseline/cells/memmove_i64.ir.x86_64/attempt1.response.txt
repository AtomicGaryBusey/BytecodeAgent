export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%zero =w ceql %n, 0
	jnz %zero, @done, @compare
@compare
	%same =w ceql %dst, %src
	jnz %same, @done, @direction
@direction
	%before =w cultl %dst, %src
	jnz %before, @forward_init, @backward_init
@forward_init
	jmp @forward_test
@forward_test
	%fi =l phi @forward_init 0, @forward_body %fi_next
	%fmore =w csltl %fi, %n
	jnz %fmore, @forward_body, @done
@forward_body
	%foffset =l mul %fi, 8
	%fsrc =l add %src, %foffset
	%fvalue =l loadl %fsrc
	%fdst =l add %dst, %foffset
	storel %fvalue, %fdst
	%fi_next =l add %fi, 1
	jmp @forward_test
@backward_init
	%bi_start =l sub %n, 1
	jmp @backward_test
@backward_test
	%bi =l phi @backward_init %bi_start, @backward_body %bi_next
	%bmore =w csgel %bi, 0
	jnz %bmore, @backward_body, @done
@backward_body
	%boffset =l mul %bi, 8
	%bsrc =l add %src, %boffset
	%bvalue =l loadl %bsrc
	%bdst =l add %dst, %boffset
	storel %bvalue, %bdst
	%bi_next =l sub %bi, 1
	jmp @backward_test
@done
	ret 0
}
