export function l $collatz_max(l %n) {
@start
	%k_slot =l alloc8 8
	%max_slot =l alloc8 8
	%x_slot =l alloc8 8
	%steps_slot =l alloc8 8
	storel 1, %k_slot
	storel 0, %max_slot
	jmp @outer
@outer
	%k =l loadl %k_slot
	storel %k, %x_slot
	storel 0, %steps_slot
	jmp @inner
@inner
	%x =l loadl %x_slot
	%at_one =w ceql %x, 1
	jnz %at_one, @finished, @step
@step
	%half =l div %x, 2
	%triple =l mul %x, 3
	%odd =l add %triple, 1
	%difference =l sub %odd, %half
	%parity =l and %x, 1
	%adjustment =l mul %parity, %difference
	%next_x =l add %half, %adjustment
	storel %next_x, %x_slot
	%steps =l loadl %steps_slot
	%next_steps =l add %steps, 1
	storel %next_steps, %steps_slot
	jmp @inner
@finished
	%final_steps =l loadl %steps_slot
	%current_max =l loadl %max_slot
	%is_larger =w csltl %current_max, %final_steps
	jnz %is_larger, @update_max, @check_k
@update_max
	storel %final_steps, %max_slot
	jmp @check_k
@check_k
	%current_k =l loadl %k_slot
	%has_more =w csltl %current_k, %n
	jnz %has_more, @advance_k, @return
@advance_k
	%next_k =l add %current_k, 1
	storel %next_k, %k_slot
	jmp @outer
@return
	%result =l loadl %max_slot
	ret %result
}
