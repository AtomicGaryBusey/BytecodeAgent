export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @setup
@setup
	%pairs =l sub %n, 1
	%span =l mul %pairs, 8
	%end =l add %a, %span
	jmp @loop
@loop
	%p =l phi @setup %a, @body %q
	%total =l phi @setup 0, @body %newtotal
	%atend =w ceql %p, %end
	jnz %atend, @done, @body
@body
	%current =l loadl %p
	%q =l add %p, 8
	%following =l loadl %q
	%isdesc =w csltl %following, %current
	%increment =l extsw %isdesc
	%newtotal =l add %total, %increment
	jmp @loop
@done
	ret %total
@zero
	ret 0
}
