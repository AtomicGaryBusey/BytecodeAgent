export function l $max_byte(l %p, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @merge %next
	%max =l phi @entry 0, @merge %curmax
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%next =l add %i, 1
	%greater =w csltl %max, %b
	jnz %greater, @update, @merge
@update
	jmp @merge
@merge
	%curmax =l phi @body %max, @update %b
	jmp @loop
@done
	ret %max
}
