mp_mul:
push rbp
push rbx
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbp, r8
lea rbx, [rsi + rcx]
mov rdi, rbp
mov rcx, rbx
xor eax, eax
cld
rep stosq
test r13, r13
jz .Lsignificant
test r15, r15
jz .Lsignificant
xor r10d, r10d
.Louter_loop:
cmp r10, r13
jae .Lsignificant
mov r11, qword ptr [r12 + r10*8]
xor r9d, r9d
xor r8d, r8d
.Linner_loop:
cmp r9, r15
jae .Linner_done
mov rax, r11
mul qword ptr [r14 + r9*8]
lea rsi, [r10 + r9]
add rax, qword ptr [rbp + rsi*8]
adc rdx, 0
add rax, r8
adc rdx, 0
mov qword ptr [rbp + rsi*8], rax
mov r8, rdx
inc r9
jmp .Linner_loop
.Linner_done:
lea rsi, [r10 + r15]
mov qword ptr [rbp + rsi*8], r8
inc r10
jmp .Louter_loop
.Lsignificant:
mov rax, rbx
.Lscan_loop:
test rax, rax
jz .Lrestore
cmp qword ptr [rbp + rax*8 - 8], 0
jne .Lrestore
dec rax
jmp .Lscan_loop
.Lrestore:
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
ret
