export function l $qsort_pairs(l %a, l %n) {
@start
	%base =w csltl %n, 2
	jnz %base, @done, @init
@done
	ret 0
@init
	%nm1 =l sub %n, 1
	%lastoff =l mul %nm1, 16
	%plast =l add %a, %lastoff
	%pivotk =l loadl %plast
	%plastv =l add %plast, 8
	%pivotv =l loadl %plastv
	jmp @loop
@loop
	%i =l phi @init 0, @next %inext
	%j =l phi @init 0, @next %jnext
	%more =w csltl %j, %nm1
	jnz %more, @body, @finish
@body
	%joff =l mul %j, 16
	%pj =l add %a, %joff
	%kj =l loadl %pj
	%pjv =l add %pj, 8
	%vj =l loadl %pjv
	%keyless =w csltl %kj, %pivotk
	jnz %keyless, @take, @eqkey
@eqkey
	%keyeq =w ceql %kj, %pivotk
	jnz %keyeq, @cmpval, @skip
@cmpval
	%valok =w cslel %vj, %pivotv
	jnz %valok, @take, @skip
@take
	%ioff =l mul %i, 16
	%pi =l add %a, %ioff
	%ki =l loadl %pi
	%piv =l add %pi, 8
	%vi =l loadl %piv
	storel %ki, %pj
	storel %vi, %pjv
	storel %kj, %pi
	storel %vj, %piv
	%iinc =l add %i, 1
	jmp @next
@skip
	jmp @next
@next
	%inext =l phi @take %iinc, @skip %i
	%jnext =l add %j, 1
	jmp @loop
@finish
	%foff =l mul %i, 16
	%pf =l add %a, %foff
	%kf =l loadl %pf
	%pfv =l add %pf, 8
	%vf =l loadl %pfv
	storel %pivotk, %pf
	storel %pivotv, %pfv
	storel %kf, %plast
	storel %vf, %plastv
	%r =l call $qsort_pairs(l %a, l %i)
	%pright =l add %pf, 16
	%nright =l sub %nm1, %i
	%r2 =l call $qsort_pairs(l %pright, l %nright)
	ret 0
}
