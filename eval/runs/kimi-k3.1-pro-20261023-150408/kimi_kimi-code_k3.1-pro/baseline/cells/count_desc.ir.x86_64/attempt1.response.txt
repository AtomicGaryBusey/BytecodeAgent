export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @setup
@setup
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @setup 0, @body %i_next
	%sum =l phi @setup 0, @body %sum_next
	%go =w csltl %i, %last
	jnz %go, @body, @done
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%p_next =l add %p, 8
	%y =l loadl %p_next
	%desc =w csltl %y, %x
	%inc =l extuw %desc
	%sum_next =l add %sum, %inc
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %sum
@zero
	ret 0
}
