export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%inbounds =w csltl %i, %n
	jnz %inbounds, @body, @done
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%iszero =w ceql %value, 0
	jnz %iszero, @done, @next
@next
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
