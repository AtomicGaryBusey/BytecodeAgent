export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @run
@run
	%pc =l phi @start 0, @push %push_pc, @add %add_pc, @sub %sub_pc, @mul %mul_pc, @dup %dup_pc, @swap %swap_pc, @drop %drop_pc, @over %over_pc, @lt_true %lt_pc, @lt_false %lt_pc, @jz_taken %jz_target, @jz_not %jz_not_pc, @jump %jump_target, @other %other_pc
	%depth =l phi @start 0, @push %push_depth, @add %add_depth, @sub %sub_depth, @mul %mul_depth, @dup %dup_depth, @swap %swap_depth, @drop %drop_depth, @over %over_depth, @lt_true %lt_depth, @lt_false %lt_depth, @jz_taken %jz_depth, @jz_not %jz_depth, @jump %jump_depth, @other %other_depth
	%is_end =w ceql %pc, %len
	jnz %is_end, @finish, @fetch
@fetch
	%pc_offset =l mul %pc, 8
	%instruction_ptr =l add %code, %pc_offset
	%op =l loadl %instruction_ptr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @halt, @check_push
@check_push
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @check_add
@check_add
	%is_add =w ceql %op, 2
	jnz %is_add, @add, @check_sub
@check_sub
	%is_sub =w ceql %op, 3
	jnz %is_sub, @sub, @check_mul
@check_mul
	%is_mul =w ceql %op, 4
	jnz %is_mul, @mul, @check_dup
@check_dup
	%is_dup =w ceql %op, 5
	jnz %is_dup, @dup, @check_swap
@check_swap
	%is_swap =w ceql %op, 6
	jnz %is_swap, @swap, @check_drop
@check_drop
	%is_drop =w ceql %op, 7
	jnz %is_drop, @drop, @check_over
@check_over
	%is_over =w ceql %op, 8
	jnz %is_over, @over, @check_lt
@check_lt
	%is_lt =w ceql %op, 9
	jnz %is_lt, @lt, @check_jz
@check_jz
	%is_jz =w ceql %op, 10
	jnz %is_jz, @jz, @check_jump
@check_jump
	%is_jump =w ceql %op, 11
	jnz %is_jump, @jump, @other
@halt
	jmp @finish
@push
	%push_operand_index =l add %pc, 1
	%push_operand_offset =l mul %push_operand_index, 8
	%push_operand_ptr =l add %code, %push_operand_offset
	%push_value =l loadl %push_operand_ptr
	%push_stack_offset =l mul %depth, 8
	%push_stack_ptr =l add %stack, %push_stack_offset
	storel %push_value, %push_stack_ptr
	%push_pc =l add %pc, 2
	%push_depth =l add %depth, 1
	jmp @run
@add
	%add_index =l sub %depth, 2
	%add_offset =l mul %add_index, 8
	%add_aptr =l add %stack, %add_offset
	%add_a =l loadl %add_aptr
	%add_bptr =l add %add_aptr, 8
	%add_b =l loadl %add_bptr
	%add_value =l add %add_a, %add_b
	storel %add_value, %add_aptr
	%add_pc =l add %pc, 1
	%add_depth =l sub %depth, 1
	jmp @run
@sub
	%sub_index =l sub %depth, 2
	%sub_offset =l mul %sub_index, 8
	%sub_aptr =l add %stack, %sub_offset
	%sub_a =l loadl %sub_aptr
	%sub_bptr =l add %sub_aptr, 8
	%sub_b =l loadl %sub_bptr
	%sub_value =l sub %sub_a, %sub_b
	storel %sub_value, %sub_aptr
	%sub_pc =l add %pc, 1
	%sub_depth =l sub %depth, 1
	jmp @run
@mul
	%mul_index =l sub %depth, 2
	%mul_offset =l mul %mul_index, 8
	%mul_aptr =l add %stack, %mul_offset
	%mul_a =l loadl %mul_aptr
	%mul_bptr =l add %mul_aptr, 8
	%mul_b =l loadl %mul_bptr
	%mul_value =l mul %mul_a, %mul_b
	storel %mul_value, %mul_aptr
	%mul_pc =l add %pc, 1
	%mul_depth =l sub %depth, 1
	jmp @run
@dup
	%dup_source_index =l sub %depth, 1
	%dup_source_offset =l mul %dup_source_index, 8
	%dup_source_ptr =l add %stack, %dup_source_offset
	%dup_value =l loadl %dup_source_ptr
	%dup_dest_offset =l mul %depth, 8
	%dup_dest_ptr =l add %stack, %dup_dest_offset
	storel %dup_value, %dup_dest_ptr
	%dup_pc =l add %pc, 1
	%dup_depth =l add %depth, 1
	jmp @run
@swap
	%swap_index =l sub %depth, 2
	%swap_offset =l mul %swap_index, 8
	%swap_aptr =l add %stack, %swap_offset
	%swap_a =l loadl %swap_aptr
	%swap_bptr =l add %swap_aptr, 8
	%swap_b =l loadl %swap_bptr
	storel %swap_b, %swap_aptr
	storel %swap_a, %swap_bptr
	%swap_pc =l add %pc, 1
	%swap_depth =l add %depth, 0
	jmp @run
@drop
	%drop_pc =l add %pc, 1
	%drop_depth =l sub %depth, 1
	jmp @run
@over
	%over_source_index =l sub %depth, 2
	%over_source_offset =l mul %over_source_index, 8
	%over_source_ptr =l add %stack, %over_source_offset
	%over_value =l loadl %over_source_ptr
	%over_dest_offset =l mul %depth, 8
	%over_dest_ptr =l add %stack, %over_dest_offset
	storel %over_value, %over_dest_ptr
	%over_pc =l add %pc, 1
	%over_depth =l add %depth, 1
	jmp @run
@lt
	%lt_index =l sub %depth, 2
	%lt_offset =l mul %lt_index, 8
	%lt_aptr =l add %stack, %lt_offset
	%lt_a =l loadl %lt_aptr
	%lt_bptr =l add %lt_aptr, 8
	%lt_b =l loadl %lt_bptr
	%lt_cmp =w csltl %lt_a, %lt_b
	%lt_pc =l add %pc, 1
	%lt_depth =l sub %depth, 1
	jnz %lt_cmp, @lt_true, @lt_false
@lt_true
	storel 1, %lt_aptr
	jmp @run
@lt_false
	storel 0, %lt_aptr
	jmp @run
@jz
	%jz_operand_index =l add %pc, 1
	%jz_operand_offset =l mul %jz_operand_index, 8
	%jz_operand_ptr =l add %code, %jz_operand_offset
	%jz_target =l loadl %jz_operand_ptr
	%jz_value_index =l sub %depth, 1
	%jz_value_offset =l mul %jz_value_index, 8
	%jz_value_ptr =l add %stack, %jz_value_offset
	%jz_value =l loadl %jz_value_ptr
	%jz_depth =l sub %depth, 1
	%jz_is_zero =w ceql %jz_value, 0
	jnz %jz_is_zero, @jz_taken, @jz_not
@jz_taken
	jmp @run
@jz_not
	%jz_not_pc =l add %pc, 2
	jmp @run
@jump
	%jump_operand_index =l add %pc, 1
	%jump_operand_offset =l mul %jump_operand_index, 8
	%jump_operand_ptr =l add %code, %jump_operand_offset
	%jump_target =l loadl %jump_operand_ptr
	%jump_depth =l add %depth, 0
	jmp @run
@other
	%other_pc =l add %pc, 1
	%other_depth =l add %depth, 0
	jmp @run
@finish
	%is_empty =w ceql %depth, 0
	jnz %is_empty, @empty, @nonempty
@empty
	jmp @cleanup
@nonempty
	%top_index =l sub %depth, 1
	%top_offset =l mul %top_index, 8
	%top_ptr =l add %stack, %top_offset
	%top_value =l loadl %top_ptr
	jmp @cleanup
@cleanup
	%result =l phi @empty 0, @nonempty %top_value
	jmp @zero_tail
@zero_tail
	%zero_index =l phi @cleanup %depth, @zero_tail_body %zero_next
	%zero_done =w ceql %zero_index, %cap
	jnz %zero_done, @done, @zero_tail_body
@zero_tail_body
	%zero_offset =l mul %zero_index, 8
	%zero_ptr =l add %stack, %zero_offset
	storel 0, %zero_ptr
	%zero_next =l add %zero_index, 1
	jmp @zero_tail
@done
	ret %result
}
