export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	jmp @init_head

@init_head
	%init_j =l phi @entry 0, @init_body %init_j_next
	%init_done =w csltl %n, %init_j
	jnz %init_done, @outer_setup, @init_body

@init_body
	%init_off =l mul %init_j, 8
	%init_addr =l add %row, %init_off
	storel 0, %init_addr
	%init_j_next =l add %init_j, 1
	jmp @init_head

@outer_setup
	jmp @outer_head

@outer_head
	%i =l phi @outer_setup 0, @outer_next %i_next
	%have_row =w csltl %i, %m
	jnz %have_row, @row_setup, @finish

@row_setup
	%x_addr =l add %x, %i
	%xi =w loadub %x_addr
	jmp @inner_head

@inner_head
	%j =l phi @row_setup 1, @inner_next %j_next
	%diag =l phi @row_setup 0, @inner_next %up
	%inner_done =w csltl %n, %j
	jnz %inner_done, @outer_next, @inner_body

@inner_body
	%j_off =l mul %j, 8
	%row_j_addr =l add %row, %j_off
	%up =l loadl %row_j_addr
	%prev_j =l sub %j, 1
	%prev_off =l mul %prev_j, 8
	%row_prev_addr =l add %row, %prev_off
	%left =l loadl %row_prev_addr
	%y_addr =l add %y, %prev_j
	%yj =w loadub %y_addr
	%equal =w ceqw %xi, %yj
	jnz %equal, @match, @mismatch

@match
	%match_value =l add %diag, 1
	storel %match_value, %row_j_addr
	jmp @inner_next

@mismatch
	%use_left =w csltl %up, %left
	jnz %use_left, @store_left, @store_up

@store_left
	storel %left, %row_j_addr
	jmp @inner_next

@store_up
	storel %up, %row_j_addr
	jmp @inner_next

@inner_next
	%j_next =l add %j, 1
	jmp @inner_head

@outer_next
	%i_next =l add %i, 1
	jmp @outer_head

@finish
	%result_off =l mul %n, 8
	%result_addr =l add %row, %result_off
	%result =l loadl %result_addr
	ret %result
}
