mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r15, rsi
mov rbp, rdx
mov r13, rcx
mov r14, r8
mov rbx, r15
imul rbx, r15
xor eax, eax
.Linit_out:
cmp rax, rbx
jae .Linit_diag_setup
mov qword ptr [r13 + rax*8], 0
inc rax
jmp .Linit_out
.Linit_diag_setup:
xor eax, eax
xor ecx, ecx
lea rdx, [r15 + 1]
.Linit_diag:
cmp rax, r15
jae .Lexp_test
mov qword ptr [r13 + rcx*8], 1
add rcx, rdx
inc rax
jmp .Linit_diag
.Lexp_test:
test rbp, rbp
jz .Lclear_setup
.Lmultiply:
xor r8d, r8d
.Lrow_loop:
cmp r8, r15
jae .Lcopy_setup
mov rsi, r8
imul rsi, r15
xor r9d, r9d
.Lcol_loop:
cmp r9, r15
jae .Lnext_row
xor r11d, r11d
xor r10d, r10d
.Ldot_loop:
cmp r10, r15
jae .Lstore_product
mov rax, rsi
add rax, r10
mov rax, qword ptr [r13 + rax*8]
mov rcx, r10
imul rcx, r15
add rcx, r9
imul rax, qword ptr [r12 + rcx*8]
add r11, rax
inc r10
jmp .Ldot_loop
.Lstore_product:
mov rax, rsi
add rax, r9
mov qword ptr [r14 + rax*8], r11
inc r9
jmp .Lcol_loop
.Lnext_row:
inc r8
jmp .Lrow_loop
.Lcopy_setup:
xor eax, eax
.Lcopy_loop:
cmp rax, rbx
jae .Lnext_power
mov rcx, qword ptr [r14 + rax*8]
mov qword ptr [r13 + rax*8], rcx
inc rax
jmp .Lcopy_loop
.Lnext_power:
dec rbp
jnz .Lmultiply
.Lclear_setup:
xor eax, eax
add rbx, rbx
.Lclear_tmp:
cmp rax, rbx
jae .Ldone
mov qword ptr [r14 + rax*8], 0
inc rax
jmp .Lclear_tmp
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
