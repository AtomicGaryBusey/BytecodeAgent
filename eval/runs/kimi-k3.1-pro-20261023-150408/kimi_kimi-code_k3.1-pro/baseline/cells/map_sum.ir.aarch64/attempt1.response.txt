export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%sum =l phi @start 0, @body %sum_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum_next =l add %sum, %r
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %sum
}
