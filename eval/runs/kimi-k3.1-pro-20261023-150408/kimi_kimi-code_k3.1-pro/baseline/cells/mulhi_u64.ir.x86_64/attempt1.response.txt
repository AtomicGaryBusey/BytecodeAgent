export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%h00 =l shr %p00, 32
	%l01 =l and %p01, 4294967295
	%l10 =l and %p10, 4294967295
	%t0 =l add %h00, %l01
	%t =l add %t0, %l10
	%h01 =l shr %p01, 32
	%h10 =l shr %p10, 32
	%r0 =l add %p11, %h01
	%r1 =l add %r0, %h10
	%carry =l shr %t, 32
	%result =l add %r1, %carry
	ret %result
}
