export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @test
@test
	%i =l phi @entry %i0, @body %i1
	%c =l phi @entry %c0, @body %c1
	%more =w csltl %i, %n
	jnz %more, @body, @exit
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%inc =l extuw %gt
	%c1 =l add %c, %inc
	%i1 =l add %i, 1
	jmp @test
@exit
	ret %c
}
