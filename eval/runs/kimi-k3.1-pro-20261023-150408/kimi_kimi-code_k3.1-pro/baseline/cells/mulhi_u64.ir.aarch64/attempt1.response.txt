export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p10 =l mul %a1, %b0
	%carry0 =l shr %p00, 32
	%t =l add %p10, %carry0
	%tlo =l and %t, 4294967295
	%thi =l shr %t, 32
	%p01 =l mul %a0, %b1
	%s =l add %p01, %tlo
	%carry1 =l shr %s, 32
	%p11 =l mul %a1, %b1
	%hi0 =l add %p11, %thi
	%hi =l add %hi0, %carry1
	ret %hi
}
