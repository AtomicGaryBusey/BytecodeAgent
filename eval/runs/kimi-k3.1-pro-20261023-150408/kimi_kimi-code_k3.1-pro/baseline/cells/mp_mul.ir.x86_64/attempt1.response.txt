export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero_test

@zero_test
	%zi =l phi @start 0, @zero_body %zi_next
	%zmore =w csltl %zi, %n
	jnz %zmore, @zero_body, @zero_done

@zero_body
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi_next =l add %zi, 1
	jmp @zero_test

@zero_done
	%na_is_zero =w ceql %na, 0
	jnz %na_is_zero, @return_zero, @check_nb

@check_nb
	%nb_is_zero =w ceql %nb, 0
	jnz %nb_is_zero, @return_zero, @mul_setup

@mul_setup
	jmp @outer_test

@outer_test
	%i =l phi @mul_setup 0, @inner_done %i_next
	%imore =w csltl %i, %na
	jnz %imore, @outer_body, @normalize_test

@outer_body
	%ai_off =l mul %i, 8
	%ai_addr =l add %a, %ai_off
	%ai =l loadl %ai_addr
	jmp @inner_test

@inner_test
	%j =l phi @outer_body 0, @inner_body %j_next
	%carry =l phi @outer_body 0, @inner_body %carry_next
	%jmore =w csltl %j, %nb
	jnz %jmore, @inner_body, @inner_done

@inner_body
	%bj_off =l mul %j, 8
	%bj_addr =l add %b, %bj_off
	%bj =l loadl %bj_addr

	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32

	%t0 =l mul %x0, %y0
	%w0 =l and %t0, 4294967295
	%k0 =l shr %t0, 32

	%p10 =l mul %x1, %y0
	%t1 =l add %p10, %k0
	%w1 =l and %t1, 4294967295
	%w2 =l shr %t1, 32

	%p01 =l mul %x0, %y1
	%t2 =l add %p01, %w1
	%k2 =l shr %t2, 32

	%t2_shift =l shl %t2, 32
	%prod_lo =l add %t2_shift, %w0
	%p11 =l mul %x1, %y1
	%hi0 =l add %p11, %w2
	%prod_hi =l add %hi0, %k2

	%ij =l add %i, %j
	%rij_off =l mul %ij, 8
	%rij_addr =l add %r, %rij_off
	%old =l loadl %rij_addr

	%sum0 =l add %prod_lo, %old
	%c0w =w cultl %sum0, %prod_lo
	%sum1 =l add %sum0, %carry
	%c1w =w cultl %sum1, %sum0
	%c0 =l extuw %c0w
	%c1 =l extuw %c1w
	%carry0 =l add %prod_hi, %c0
	%carry_next =l add %carry0, %c1

	storel %sum1, %rij_addr
	%j_next =l add %j, 1
	jmp @inner_test

@inner_done
	%end_index =l add %i, %nb
	%end_off =l mul %end_index, 8
	%end_addr =l add %r, %end_off
	storel %carry, %end_addr
	%i_next =l add %i, 1
	jmp @outer_test

@normalize_test
	%k =l phi @outer_test %n, @normalize_dec %k_next
	%k_is_zero =w ceql %k, 0
	jnz %k_is_zero, @return_k, @normalize_check

@normalize_check
	%k_minus_one =l sub %k, 1
	%koff =l mul %k_minus_one, 8
	%kaddr =l add %r, %koff
	%ktop =l loadl %kaddr
	%ktop_is_zero =w ceql %ktop, 0
	jnz %ktop_is_zero, @normalize_dec, @return_k

@normalize_dec
	%k_next =l sub %k, 1
	jmp @normalize_test

@return_k
	ret %k

@return_zero
	ret 0
}
