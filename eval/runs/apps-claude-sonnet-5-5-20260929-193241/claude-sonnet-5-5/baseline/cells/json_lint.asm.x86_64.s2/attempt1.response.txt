.globl _start
.text
.equ S_VAL, 0
.equ S_ARRF, 1
.equ S_AFTER, 2
.equ S_OBJF, 3
.equ S_OBJK, 4
.equ S_COLON, 5
.equ S_STR, 6
.equ S_ESC, 7
.equ S_HEX, 8
.equ S_LIT, 9
.equ S_MINUS, 10
.equ S_ZERO, 11
.equ S_INT, 12
.equ S_DOT, 13
.equ S_FRAC, 14
.equ S_E, 15
.equ S_ESIGN, 16
.equ S_EXP, 17

is_ws:
    cmp al, 32
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
1:
    ret

_start:
    xor r13d, r13d
    lea r15, [rip + buf]
    lea r14, [rip + stk]
1:
    lea rsi, [r15 + r13]
    mov edx, 70000
    sub rdx, r13
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    xor esi, esi
    xor ebx, ebx
    xor r12d, r12d
loop:
    cmp rsi, r13
    jae eof
    movzx eax, byte ptr [r15 + rsi]
    cmp ebx, S_VAL
    je st_val
    cmp ebx, S_ARRF
    je st_arrf
    cmp ebx, S_AFTER
    je st_after
    cmp ebx, S_OBJF
    je st_objf
    cmp ebx, S_OBJK
    je st_objk
    cmp ebx, S_COLON
    je st_colon
    cmp ebx, S_STR
    je st_str
    cmp ebx, S_ESC
    je st_esc
    cmp ebx, S_HEX
    je st_hex
    cmp ebx, S_LIT
    je st_lit
    cmp ebx, S_MINUS
    je st_minus
    cmp ebx, S_ZERO
    je st_zero
    cmp ebx, S_INT
    je st_int
    cmp ebx, S_DOT
    je st_dot
    cmp ebx, S_FRAC
    je st_frac
    cmp ebx, S_E
    je st_e
    cmp ebx, S_ESIGN
    je st_esign
    jmp st_exp

next:
    inc rsi
    jmp loop

eof:
    test r12, r12
    jnz err
    cmp ebx, S_AFTER
    je ok
    cmp ebx, S_ZERO
    je ok
    cmp ebx, S_INT
    je ok
    cmp ebx, S_FRAC
    je ok
    cmp ebx, S_EXP
    je ok
    jmp err

ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

err:
    mov byte ptr [rip + obuf + 32], 10
    lea rdi, [rip + obuf + 32]
    mov rax, rsi
    mov ecx, 10
3:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 3b
    mov r8, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    lea rdx, [rip + obuf + 33]
    sub rdx, r8
    mov rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

st_val:
    call is_ws
    je next
value_start:
    cmp al, '{'
    jne 1f
    mov byte ptr [r14 + r12], al
    inc r12
    mov ebx, S_OBJF
    jmp next
1:
    cmp al, '['
    jne 1f
    mov byte ptr [r14 + r12], al
    inc r12
    mov ebx, S_ARRF
    jmp next
1:
    cmp al, '"'
    jne 1f
    xor r8d, r8d
    mov ebx, S_STR
    jmp next
1:
    cmp al, '-'
    jne 1f
    mov ebx, S_MINUS
    jmp next
1:
    cmp al, '0'
    jne 1f
    mov ebx, S_ZERO
    jmp next
1:
    lea ecx, [rax - 49]
    cmp ecx, 8
    ja 1f
    mov ebx, S_INT
    jmp next
1:
    lea r10, [rip + lit_t + 1]
    cmp al, 't'
    je 2f
    lea r10, [rip + lit_f + 1]
    cmp al, 'f'
    je 2f
    lea r10, [rip + lit_n + 1]
    cmp al, 'n'
    jne err
2:
    mov ebx, S_LIT
    jmp next

st_arrf:
    call is_ws
    je next
    cmp al, ']'
    jne value_start
    dec r12
    mov ebx, S_AFTER
    jmp next

st_after:
    call is_ws
    je next
    test r12, r12
    jz err
    movzx ecx, byte ptr [r14 + r12 - 1]
    cmp al, ','
    je 1f
    lea edx, [rcx + 2]
    cmp al, dl
    jne err
    dec r12
    jmp next
1:
    mov ebx, S_VAL
    cmp cl, '['
    je next
    mov ebx, S_OBJK
    jmp next

st_objf:
    call is_ws
    je next
    cmp al, '}'
    jne st_objk_go
    dec r12
    mov ebx, S_AFTER
    jmp next

st_objk:
    call is_ws
    je next
st_objk_go:
    cmp al, '"'
    jne err
    mov r8d, 1
    mov ebx, S_STR
    jmp next

st_colon:
    call is_ws
    je next
    cmp al, ':'
    jne err
    mov ebx, S_VAL
    jmp next

st_str:
    cmp al, '"'
    je 1f
    cmp al, '\\'
    je 2f
    cmp al, 32
    jb err
    jmp next
1:
    mov ebx, S_AFTER
    test r8d, r8d
    jz next
    mov ebx, S_COLON
    jmp next
2:
    mov ebx, S_ESC
    jmp next

st_esc:
    cmp al, 'u'
    jne 1f
    mov ebx, S_HEX
    mov r9d, 4
    jmp next
1:
    lea rdi, [rip + escs]
    mov ecx, 8
    repne scasb
    jne err
    mov ebx, S_STR
    jmp next

st_hex:
    lea ecx, [rax - 48]
    cmp ecx, 9
    jbe 1f
    or eax, 32
    sub eax, 97
    cmp eax, 5
    ja err
1:
    dec r9d
    jnz next
    mov ebx, S_STR
    jmp next

st_lit:
    cmp al, byte ptr [r10]
    jne err
    inc r10
    cmp byte ptr [r10], 0
    jne next
    mov ebx, S_AFTER
    jmp next

st_minus:
    cmp al, '0'
    jne 1f
    mov ebx, S_ZERO
    jmp next
1:
    lea ecx, [rax - 49]
    cmp ecx, 8
    ja err
    mov ebx, S_INT
    jmp next

st_zero:
    cmp al, '.'
    je to_dot
    mov ecx, eax
    or ecx, 32
    cmp cl, 'e'
    je to_e
    mov ebx, S_AFTER
    jmp loop

st_int:
    lea ecx, [rax - 48]
    cmp ecx, 9
    jbe next
    jmp st_zero

to_dot:
    mov ebx, S_DOT
    jmp next
to_e:
    mov ebx, S_E
    jmp next

st_dot:
    lea ecx, [rax - 48]
    cmp ecx, 9
    ja err
    mov ebx, S_FRAC
    jmp next

st_frac:
    lea ecx, [rax - 48]
    cmp ecx, 9
    jbe next
    mov ecx, eax
    or ecx, 32
    cmp cl, 'e'
    je to_e
    mov ebx, S_AFTER
    jmp loop

st_e:
    cmp al, '+'
    je 1f
    cmp al, '-'
    je 1f
st_esign:
    lea ecx, [rax - 48]
    cmp ecx, 9
    ja err
    mov ebx, S_EXP
    jmp next
1:
    mov ebx, S_ESIGN
    jmp next

st_exp:
    lea ecx, [rax - 48]
    cmp ecx, 9
    jbe next
    mov ebx, S_AFTER
    jmp loop

.section .rodata
okmsg: .ascii "ok\n"
errmsg: .ascii "error at "
escs: .ascii "\"\\/bfnrt"
lit_t: .asciz "true"
lit_f: .asciz "false"
lit_n: .asciz "null"

.bss
buf: .zero 70016
stk: .zero 2048
obuf: .zero 64
