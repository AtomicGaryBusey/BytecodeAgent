.globl _start
.text
_start:
    lea r15, [rip + buf]
    lea r12, [rip + stk]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r13]
    mov edx, 65600
    sub rdx, r13
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    xor ebx, ebx
    xor r14d, r14d

value_start:
    call ws
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '{'
    je obj
    cmp al, '['
    je arr
    cmp al, '"'
    je vstr
    cmp al, '-'
    je vnum
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    jbe vnum
    lea rsi, [rip + s_true]
    mov ecx, 4
    cmp al, 't'
    je lit
    lea rsi, [rip + s_false]
    mov ecx, 5
    cmp al, 'f'
    je lit
    lea rsi, [rip + s_null]
    mov ecx, 4
    cmp al, 'n'
    je lit
    jmp fail

lit:
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r15 + rbx]
    cmp al, byte ptr [rsi]
    jne fail
    inc rbx
    inc rsi
    dec ecx
    jnz lit
    jmp after_value

vstr:
    call string
    jmp after_value

vnum:
    cmp byte ptr [r15 + rbx], '-'
    jne 1f
    inc rbx
1:
    call need_digit
    cmp al, 0
    jne 2f
    inc rbx
    jmp 3f
2:
    call digits
3:
    cmp rbx, r13
    jae after_value
    cmp byte ptr [r15 + rbx], '.'
    jne 4f
    inc rbx
    call need_digit
    call digits
4:
    cmp rbx, r13
    jae after_value
    mov al, byte ptr [r15 + rbx]
    or al, 0x20
    cmp al, 'e'
    jne after_value
    inc rbx
    cmp rbx, r13
    jae fail
    mov al, byte ptr [r15 + rbx]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rbx
6:
    call need_digit
    call digits
    jmp after_value

obj:
    mov byte ptr [r12 + r14], 1
    inc r14
    inc rbx
    call ws
    cmp rbx, r13
    jae fail
    cmp byte ptr [r15 + rbx], '}'
    je close
key:
    call ws
    cmp rbx, r13
    jae fail
    cmp byte ptr [r15 + rbx], '"'
    jne fail
    call string
    call ws
    cmp rbx, r13
    jae fail
    cmp byte ptr [r15 + rbx], ':'
    jne fail
    inc rbx
    jmp value_start

arr:
    mov byte ptr [r12 + r14], 0
    inc r14
    inc rbx
    call ws
    cmp rbx, r13
    jae fail
    cmp byte ptr [r15 + rbx], ']'
    je close
    jmp value_start

close:
    inc rbx
    dec r14
after_value:
    call ws
    test r14, r14
    jnz 1f
    cmp rbx, r13
    jb fail
    jmp ok
1:
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    mov dl, byte ptr [r12 + r14 - 1]
    cmp al, ','
    jne 2f
    inc rbx
    test dl, dl
    jz value_start
    jmp key
2:
    test dl, dl
    jnz 3f
    cmp al, ']'
    je close
    jmp fail
3:
    cmp al, '}'
    je close
    jmp fail

ws:
    cmp rbx, r13
    jae 9f
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 32
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
9:
    ret
1:
    inc rbx
    jmp ws

need_digit:
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja fail
    sub eax, '0'
    ret

digits:
    cmp rbx, r13
    jae 1f
    movzx eax, byte ptr [r15 + rbx]
    sub eax, '0'
    cmp eax, 9
    ja 1f
    inc rbx
    jmp digits
1:
    ret

string:
    inc rbx
1:
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '"'
    je 8f
    cmp al, 0x20
    jb fail
    inc rbx
    cmp al, '\\'
    jne 1b
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 'u'
    je 5f
    lea rsi, [rip + s_esc]
    mov ecx, 8
2:
    cmp al, byte ptr [rsi]
    je 3f
    inc rsi
    dec ecx
    jnz 2b
    jmp fail
3:
    inc rbx
    jmp 1b
5:
    inc rbx
    mov ecx, 4
6:
    cmp rbx, r13
    jae fail
    movzx eax, byte ptr [r15 + rbx]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    jbe 7f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja fail
7:
    inc rbx
    dec ecx
    jnz 6b
    jmp 1b
8:
    inc rbx
    ret

ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_err]
    mov edx, 9
    syscall
    lea rdi, [rip + obuf + 32]
    mov byte ptr [rdi - 1], 10
    dec rdi
    mov rax, rbx
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + obuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
s_true: .ascii "true"
s_false: .ascii "false"
s_null: .ascii "null"
s_ok: .ascii "ok\n"
s_err: .ascii "error at "
s_esc: .ascii "\"\\/bfnrt"

.bss
buf: .zero 65664
stk: .zero 2048
obuf: .zero 64
