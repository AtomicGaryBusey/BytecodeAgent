.globl _start
.text
_start:
    lea r15, [rip + inbuf]
    lea r8, [rip + digits]
    lea rbx, [rip + rowbuf]
    xor r12d, r12d
1:
    mov edx, 65536
    sub edx, r12d
    lea rsi, [r15 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    xor r13d, r13d
rows:
    cmp r13, r12
    jae final
    mov r9, r12
    sub r9, r13
    cmp r9, 16
    jbe 3f
    mov r9d, 16
3:
    mov rdi, rbx
    mov eax, r13d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [r15 + r13]
    xor r14d, r14d
bytes:
    cmp r14, r9
    jae pad
    movzx eax, byte ptr [rsi + r14]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp nxt
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
nxt:
    cmp r14, 7
    jne 4f
    mov byte ptr [rdi], 0x20
    inc rdi
4:
    inc r14
    cmp r14, 16
    jb bytes
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
chars:
    cmp rcx, r9
    jae cdone
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 5f
    mov eax, 0x2e
5:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    jmp chars
cdone:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    mov rdx, rdi
    sub rdx, rbx
    mov rsi, rbx
    mov edi, 1
    mov eax, 1
    syscall
    add r13, 16
    jmp rows
final:
    mov rdi, rbx
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rdx, rdi
    sub rdx, rbx
    mov rsi, rbx
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 231
    syscall

hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
rowbuf:
    .zero 128
