_start:
    xor r12d, r12d
1:
    mov edx, 32768
    sub rdx, r12
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    xor ecx, ecx
    xor r14d, r14d
    xor r13d, r13d
    lea rsi, [rip + inbuf]
    lea rdi, [rip + ents]
3:
    cmp rcx, r12
    jae 5f
    cmp byte ptr [rsi + rcx], 10
    jne 4f
    lea rax, [rsi + r14]
    mov [rdi], rax
    mov rax, rcx
    sub rax, r14
    mov [rdi + 8], rax
    add rdi, 16
    inc r13
    lea r14, [rcx + 1]
4:
    inc rcx
    jmp 3b
5:
    cmp r14, r12
    jae 6f
    lea rax, [rsi + r14]
    mov [rdi], rax
    mov rax, r12
    sub rax, r14
    mov [rdi + 8], rax
    inc r13
6:
    # insertion sort
    lea rbp, [rip + ents]
    mov r15d, 1
sortloop:
    cmp r15, r13
    jae sorted
    mov rax, r15
    shl rax, 4
    mov r8, [rbp + rax]
    mov r9, [rbp + rax + 8]
    mov rbx, r15
inner:
    test rbx, rbx
    jz place
    mov rax, rbx
    dec rax
    shl rax, 4
    mov r10, [rbp + rax]
    mov r11, [rbp + rax + 8]
    mov rcx, r11
    cmp rcx, r9
    jbe 7f
    mov rcx, r9
7:
    mov rsi, r10
    mov rdi, r8
8:
    test rcx, rcx
    jz lencmp
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rdi]
    cmp eax, edx
    jne diff
    inc rsi
    inc rdi
    dec rcx
    jmp 8b
diff:
    ja shift
    jmp place
lencmp:
    cmp r11, r9
    ja shift
    jmp place
shift:
    mov rax, rbx
    shl rax, 4
    mov [rbp + rax], r10
    mov [rbp + rax + 8], r11
    dec rbx
    jmp inner
place:
    mov rax, rbx
    shl rax, 4
    mov [rbp + rax], r8
    mov [rbp + rax + 8], r9
    inc r15
    jmp sortloop
sorted:
    lea rdi, [rip + outbuf]
    mov rbx, rbp
    mov r15, r13
9:
    test r15, r15
    jz 10f
    mov rsi, [rbx]
    mov rcx, [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    dec r15
    jmp 9b
10:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
11:
    test rdx, rdx
    jle 12f
    mov edi, 1
    mov eax, 1
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    add rsi, rax
    sub rdx, rax
    jmp 11b
12:
    xor edi, edi
    mov eax, 231
    syscall
