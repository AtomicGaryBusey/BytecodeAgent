.globl _start
.text
_start:
    lea rbx, [rip + buf]
    lea r10, [rip + stk]
    xor r8d, r8d
    xor r9d, r9d
1:
    mov edx, 69000
    sub rdx, r8
    lea rsi, [rbx + r8]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add r8, rax
    jmp 1b
2:
    xor esi, esi
value:
    call ws
    movzx eax, byte ptr [rbx + rsi]
    cmp al, '{'
    je 10f
    cmp al, '['
    je 11f
    cmp al, '"'
    je 12f
    cmp al, '-'
    je num
    lea ecx, [rax - '0']
    cmp cl, 9
    jbe num
    cmp al, 't'
    je 13f
    cmp al, 'f'
    je 14f
    cmp al, 'n'
    je 15f
    jmp fail
10:
    inc rsi
    mov byte ptr [r10 + r9], 1
    inc r9
    call ws
    cmp byte ptr [rbx + rsi], '}'
    je close
    jmp key
11:
    inc rsi
    mov byte ptr [r10 + r9], 2
    inc r9
    call ws
    cmp byte ptr [rbx + rsi], ']'
    je close
    jmp value
12:
    call str
    jmp after
13:
    lea rdi, [rip + lit_true]
    mov ecx, 4
    jmp lit
14:
    lea rdi, [rip + lit_false]
    mov ecx, 5
    jmp lit
15:
    lea rdi, [rip + lit_null]
    mov ecx, 4
lit:
    mov al, byte ptr [rdi]
    cmp al, byte ptr [rbx + rsi]
    jne fail
    inc rdi
    inc rsi
    dec ecx
    jnz lit
    jmp after
key:
    cmp byte ptr [rbx + rsi], '"'
    jne fail
    call str
    call ws
    cmp byte ptr [rbx + rsi], ':'
    jne fail
    inc rsi
    jmp value
close:
    inc rsi
    dec r9
after:
    call ws
    movzx eax, byte ptr [rbx + rsi]
    test r9, r9
    jnz 1f
    cmp rsi, r8
    jne fail
    jmp ok
1:
    movzx edx, byte ptr [r10 + r9 - 1]
    cmp al, ','
    je 2f
    cmp dl, 1
    jne 3f
    cmp al, '}'
    je close
    jmp fail
3:
    cmp al, ']'
    je close
    jmp fail
2:
    inc rsi
    call ws
    cmp dl, 1
    je key
    jmp value

ws:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 32
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
    ret
1:
    inc rsi
    jmp ws

str:
    inc rsi
1:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, '"'
    je 9f
    cmp al, '\\'
    je 2f
    cmp al, 32
    jb fail
    inc rsi
    jmp 1b
9:
    inc rsi
    ret
2:
    inc rsi
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 'u'
    je 4f
    lea rdi, [rip + esc]
3:
    mov dl, byte ptr [rdi]
    test dl, dl
    jz fail
    cmp al, dl
    je 5f
    inc rdi
    jmp 3b
5:
    inc rsi
    jmp 1b
4:
    inc rsi
    mov ecx, 4
6:
    movzx eax, byte ptr [rbx + rsi]
    lea edx, [rax - '0']
    cmp dl, 9
    jbe 7f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja fail
7:
    inc rsi
    dec ecx
    jnz 6b
    jmp 1b

needd:
    movzx eax, byte ptr [rbx + rsi]
    sub eax, '0'
    cmp eax, 9
    ja fail
    inc rsi
    ret
digits:
    movzx eax, byte ptr [rbx + rsi]
    sub eax, '0'
    cmp eax, 9
    ja 1f
    inc rsi
    jmp digits
1:
    ret

num:
    cmp al, '-'
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp 3f
2:
    call needd
    call digits
3:
    cmp byte ptr [rbx + rsi], '.'
    jne 4f
    inc rsi
    call needd
    call digits
4:
    movzx eax, byte ptr [rbx + rsi]
    or al, 0x20
    cmp al, 'e'
    jne after
    inc rsi
    movzx eax, byte ptr [rbx + rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rsi
6:
    call needd
    call digits
    jmp after

ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov r12, rsi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    lea rdi, [rip + outbuf + 31]
    mov byte ptr [rdi], 10
    mov rax, r12
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + outbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
okmsg: .ascii "ok\n"
errmsg: .ascii "error at "
lit_true: .ascii "true"
lit_false: .ascii "false"
lit_null: .ascii "null"
esc: .asciz "\"\\/bfnrt"

.bss
buf: .zero 70000
stk: .zero 2048
outbuf: .zero 32
