.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
readloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]
byteloop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 32
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15, r15
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
    cmp al, 10
    jne next
    inc r12
next:
    dec rcx
    jnz byteloop
    jmp readloop
done:
    lea rdi, [rip + out]
    mov rax, r12
    call itoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call itoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call itoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + out]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
itoa:
    mov r8d, 10
    xor ecx, ecx
1:
    xor edx, edx
    div r8
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz 1b
2:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz 2b
    ret
.bss
buf:

out:
