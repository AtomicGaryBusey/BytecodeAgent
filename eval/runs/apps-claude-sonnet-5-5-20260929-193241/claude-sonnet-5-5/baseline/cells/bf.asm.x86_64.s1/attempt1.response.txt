.globl _start
.text
_start:
    lea r12, [rip + buf]
    xor r10d, r10d
1:
    mov eax, 65536
    sub eax, r10d
    mov edx, eax
    lea rsi, [r12 + r10]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add r10, rax
    jmp 1b
2:
    # find '!'
    xor r8d, r8d
3:
    cmp r8, r10
    jae 4f
    cmp byte ptr [r12 + r8], 0x21
    je 5f
    inc r8
    jmp 3b
4:
    mov rdx, r10
    jmp 6f
5:
    lea rdx, [r8 + 1]
6:
    # r8 = plen, rdx = inp, r10 = inend
    lea r13, [rip + jmpt]
    lea rdi, [rip + stk]
    xor esi, esi
    xor r9d, r9d
7:
    cmp rsi, r8
    jae 10f
    movzx eax, byte ptr [r12 + rsi]
    cmp al, '['
    jne 8f
    mov dword ptr [rdi + r9*4], esi
    inc r9
    jmp 9f
8:
    cmp al, ']'
    jne 9f
    test r9, r9
    jz fail
    dec r9
    mov ecx, dword ptr [rdi + r9*4]
    mov dword ptr [r13 + rsi*4], ecx
    mov dword ptr [r13 + rcx*4], esi
9:
    inc rsi
    jmp 7b
10:
    test r9, r9
    jnz fail
    lea rbx, [rip + tape]
    lea r15, [rip + outb]
    xor ebp, ebp
    xor esi, esi
11:
    cmp rsi, r8
    jae done
    movzx eax, byte ptr [r12 + rsi]
    inc rsi
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '>'
    je op_gt
    cmp al, '<'
    je op_lt
    cmp al, '.'
    je op_dot
    cmp al, ','
    je op_comma
    cmp al, '['
    je op_open
    cmp al, ']'
    je op_close
    jmp 11b
op_plus:
    inc byte ptr [rbx]
    jmp 11b
op_minus:
    dec byte ptr [rbx]
    jmp 11b
op_gt:
    inc rbx
    jmp 11b
op_lt:
    dec rbx
    jmp 11b
op_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r15 + rbp], al
    inc rbp
    jmp 11b
op_comma:
    xor eax, eax
    cmp rdx, r10
    jae 12f
    movzx eax, byte ptr [r12 + rdx]
    inc rdx
12:
    mov byte ptr [rbx], al
    jmp 11b
op_open:
    cmp byte ptr [rbx], 0
    jne 11b
    mov ecx, dword ptr [r13 + rsi*4 - 4]
    lea rsi, [rcx + 1]
    jmp 11b
op_close:
    cmp byte ptr [rbx], 0
    je 11b
    mov ecx, dword ptr [r13 + rsi*4 - 4]
    lea rsi, [rcx + 1]
    jmp 11b
done:
    mov rsi, r15
    mov rdx, rbp
13:
    test rdx, rdx
    jz 14f
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle 14f
    add rsi, rax
    sub rdx, rax
    jmp 13b
14:
    xor edi, edi
    mov eax, 231
    syscall
fail:
    mov edi, 1
    mov eax, 231
    syscall

.bss
.balign 16
buf: .zero 65552
jmpt: .zero 262144
stk: .zero 262144
tape: .zero 30016
outb: .zero 65552
