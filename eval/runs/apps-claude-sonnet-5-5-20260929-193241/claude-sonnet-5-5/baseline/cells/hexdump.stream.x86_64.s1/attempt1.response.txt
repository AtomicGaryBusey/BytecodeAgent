.globl _start
.text
hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret

_start:
    lea r9, [rip + hextab]
    xor r12d, r12d
rdloop:
    mov edx, 65536
    sub rdx, r12
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle rdone
    add r12, rax
    jmp rdloop
rdone:
    xor ebx, ebx
rowloop:
    cmp rbx, r12
    jae fin
    mov r13, r12
    sub r13, rbx
    cmp r13, 16
    jbe 1f
    mov r13d, 16
1:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    lea rdi, [rip + line]
    mov eax, ebx
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor r14d, r14d
2:
    cmp r14d, 8
    jne 3f
    mov byte ptr [rdi], 32
    inc rdi
3:
    cmp r14, r13
    jae 4f
    movzx eax, byte ptr [rsi + r14]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp 5f
4:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 32
    add rdi, 3
5:
    inc r14
    cmp r14, 16
    jb 2b
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2
    xor r14d, r14d
6:
    movzx eax, byte ptr [rsi + r14]
    cmp al, 0x20
    jb 7f
    cmp al, 0x7e
    jbe 8f
7:
    mov al, 46
8:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, r13
    jb 6b
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    lea rsi, [rip + line]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    add rbx, 16
    jmp rowloop
fin:
    lea rdi, [rip + line]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + line]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
