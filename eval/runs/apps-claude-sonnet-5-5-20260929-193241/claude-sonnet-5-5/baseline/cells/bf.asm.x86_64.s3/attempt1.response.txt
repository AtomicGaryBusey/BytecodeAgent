.globl _start
.text
_start:
    lea r13, [rip + inbuf]
    xor r12d, r12d
1:
    mov edx, 65536
    sub edx, r12d
    jz 2f
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    # r12 = total; find '!'
    xor r8d, r8d
3:
    cmp r8, r12
    jae 4f
    cmp byte ptr [r13 + r8], 0x21
    je 4f
    inc r8
    jmp 3b
4:
    # r8 = plen
    lea r9, [r13 + r8 + 1]
    lea r10, [r13 + r12]
    cmp r8, r12
    jb 5f
    mov r9, r10
5:
    lea r14, [rip + jump]
    lea rdi, [rip + stk]
    xor ecx, ecx          # stack count
    xor esi, esi
6:
    cmp rsi, r8
    jae 9f
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x5b
    jne 7f
    mov dword ptr [rdi + rcx*4], esi
    inc ecx
    jmp 8f
7:
    cmp al, 0x5d
    jne 8f
    test ecx, ecx
    jz bad
    dec ecx
    mov edx, dword ptr [rdi + rcx*4]
    mov dword ptr [r14 + rsi*4], edx
    mov dword ptr [r14 + rdx*4], esi
8:
    inc rsi
    jmp 6b
9:
    test ecx, ecx
    jnz bad
    lea rbx, [rip + tape]
    lea r15, [rip + outbuf]
    xor r12d, r12d
    xor esi, esi
top:
    cmp rsi, r8
    jae done
    movzx eax, byte ptr [r13 + rsi]
    inc rsi
    cmp al, 0x2b
    je op_plus
    cmp al, 0x2d
    je op_minus
    cmp al, 0x3e
    je op_gt
    cmp al, 0x3c
    je op_lt
    cmp al, 0x2e
    je op_dot
    cmp al, 0x2c
    je op_comma
    cmp al, 0x5b
    je op_lb
    cmp al, 0x5d
    je op_rb
    jmp top
op_plus:
    inc byte ptr [rbx]
    jmp top
op_minus:
    dec byte ptr [rbx]
    jmp top
op_gt:
    inc rbx
    jmp top
op_lt:
    dec rbx
    jmp top
op_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r15 + r12], al
    inc r12
    jmp top
op_comma:
    xor eax, eax
    cmp r9, r10
    jae 1f
    mov al, byte ptr [r9]
    inc r9
1:
    mov byte ptr [rbx], al
    jmp top
op_lb:
    cmp byte ptr [rbx], 0
    jne top
    mov esi, dword ptr [r14 + rsi*4 - 4]
    inc esi
    jmp top
op_rb:
    cmp byte ptr [rbx], 0
    je top
    mov esi, dword ptr [r14 + rsi*4 - 4]
    inc esi
    jmp top
done:
    test r12, r12
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r12
    syscall
1:
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall
