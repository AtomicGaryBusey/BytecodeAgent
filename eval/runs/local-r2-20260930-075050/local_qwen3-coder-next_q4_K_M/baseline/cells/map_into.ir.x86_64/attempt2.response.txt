export function l $map_into(l %dst, l %src, l %n, l %f) {
%count =l copy 0
%i =l copy 0
@loop_test
%cmp =csltl %i, %n
jnz %cmp, @loop_body, @done
@loop_body
%src_ptr =l add %src, %i
%src_val =l loadl %src_ptr
%r =l call %f(l %src_val)
%dst_ptr =l add %dst, %i
storel %r, %dst_ptr
%cmp2 =csltl %r, %src_val
jnz %cmp2, @increment_count, @next
@increment_count

@next

jmp @loop_test
@done

}
