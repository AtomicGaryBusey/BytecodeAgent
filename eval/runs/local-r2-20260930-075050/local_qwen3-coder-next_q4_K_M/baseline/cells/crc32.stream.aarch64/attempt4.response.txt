crc32:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov w8, #0xFFFFFFFF      // initial crc value
    mov x9, xzr              // byte index i = 0
    cmp x1, xzr              // n == 0?
    beq .Lexit
.Lloop:
    ldrb w10, [x0, x9]       // load byte p[i]
    eor w8, w8, w10          // crc ^= byte
    mov w11, #8              // 8 bits per byte
.Linner_loop:
    mov w12, w8              // copy crc
    and w12, w12, #1         // crc & 1
    cbz w12, .Lshift_only
    lsrv w8, w8, #1          // crc >>= 1 (logical right shift)
    l
