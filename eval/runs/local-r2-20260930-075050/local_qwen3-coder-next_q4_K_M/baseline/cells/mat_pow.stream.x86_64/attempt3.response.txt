mat_pow:
    push rbp
    mov rbp, rsp
    sub rsp, 48
    push rbx
    push r12
    push r13
    push r14
    push r15

    test rdx, rdx
    jnz .Lnot_zero_k

    # Identity matrix: out[i*n + i] = 1, others 0
    mov rbx, rcx           # out pointer
    xor rcx, rcx           # i counter
.Lidentity_loop:
    cmp rcx, rsi           # i < n?
    jge .Lidentity_done
    mov rax, rcx
    imul rax, rsi          # rax = i*n
    add rax, rcx           # rax = i*n + i
    mov qword ptr [rbx + rax*8], 1
    inc rcx
    jmp .Lidentity_loop
.Lidentity_done:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    leave
    ret

.Lnot_zero_k:
    # Copy input matrix a to tmp (for base)
    mov r9, rdi            # a pointer
    mov r10, r8            # tmp pointer
    xor r11, r11           # n*n will be computed in r11
    mov r11, rsi
    imul r11, rsi          # r11 = n*n

.Lidentity_cleanup:
    # Clear tmp buffer completely (2*n*n zeros)
    xor r14, r14           # counter for clearing tmp
    lea r15, [r8 + r11*8]  # end of first n*n block in tmp
    mov rax, r15
.Lclear_loop:
    cmp r14, r15
    jge .Lcopy_done
    xor rdi, rdi
    mov qword ptr [r8 + r14*8], rdi
    inc r14
    jmp .Lclear_loop
.Lcopy_done:

    # Copy a to tmp (first half of tmp buffer)
    lea rbx, [r8]          # dest = tmp
    mov rcx, rsi           # n
    imul rcx, rsi          # rcx = n*n
    xor rdx, rdx           # index

.Lcopy_loop:
    cmp rdx, rcx
    jge .Lsetup_done
    mov rax, qword ptr [r9 + rdx*8]
    mov qword ptr [rbx + rdx*8], rax
    inc rdx
    jmp .Lcopy_loop

.Lsetup_done:
    # Initialize result matrix as identity
    xor r12, r12           # i counter for identity
.Linit_res_loop:
    cmp r12, rsi
    jge .Linit_res_done
    mov rax, r12
    imul rax, rsi          # rax = i*n
    add rax, r12           # rax = i*n + i
    mov qword ptr [rcx + rax*8], 0   # zero first
    inc r12
    jmp .Linit_res_loop
.Linit_res_done:

    # Now square-and-multiply exponentiation
    xor r13, r13           # result index counter for setting identity
    lea rbx, [rcx]         # out = result matrix

.Lset_identity:
    cmp r13, rsi
    jge .Lmat_pow_done_init
    mov rax, r13
    imul rax, rsi          # i*n
    add rax, r13           # i*n+i
    mov qword ptr [rbx + rax*8], 1
    xor rax, rax
    mov qword ptr [rcx + rax*8], 0   # clear tmp[0] just in case
    inc r13
    jmp .Lset_identity

.Lmat_pow_done_init:

    # Set result = identity; base = a (already copied to tmp)
    xor r14, r14           # i counter for identity set
.Lset_res_identity:
    cmp r14, rsi
    jge .Lres_id_done
    mov rax, r14
    imul rax, rsi
    add rax, r14
    mov qword ptr [rcx + rax*8], 1
    inc r14
    jmp .Lset_res_identity
.Lres_id_done:

    # Clear tmp for product storage (second half)
    lea rdi, [r8 + r11*8]  # start of second n*n in tmp
    xor r15, r15
.Lclear_product:
    cmp r15, r11
    jge .Lmat_loop_start
    mov qword ptr [rdi + r15*8], 0
    inc r15
    jmp .Lclear_product

.Lmat_loop_start:
    # Exponentiation by squaring: while k > 0
    test rdx, rdx
    jz .Ldone

    # Check if lowest bit of k is set
    mov r8, rdx            # k copy
    and r9, r8             # r9 = k & 1 (use r9 as scratch: k&1)
    test r9, r9
    jz .Lno_mult

    # result *= base
    call .Lmat_mul_helper
    jmp .Lsquaring_done

.Lno_mult:
    # Skip multiplication if bit not set

.Lsquaring_done:
    # base = base * base
    lea rdi, [r8]          # first half of tmp is base
    lea rsi, [r8 + r11*8]  # second half of tmp is temp for product
    call .Lmat_mul_base

    # Shift k right by 1
    shr rdx, 1
    test rdx, rdx
    jnz .Lmat_loop_start

.Ldone:
    xor rax, rax           # return 0
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    leave
    ret

# Helper: multiply matrices in tmp[0..n*n-1] and tmp[n*n..2*n*n-1], store to out
.Lmat_mul_helper:
    # Save registers used by helper
    push rbp
    mov rbp, rsp
    sub rsp, 64            # local space for result (max n=6 => 36 elements)
    push rbx
    push rsi
    push rdi

    # Copy base and temp pointers
    mov rbx, rdi           # base matrix starts at rdi
    lea rsi, [rdi + r11*8] # product target (tmp+ n*n), but we need to overwrite it
    lea rdi, [rbp - 64]    # store result in local space

    # For each row i:
    xor r12, r12           # i counter
.Lmul_i_loop:
    cmp r12, rsi
    jge .Lmul_done_rows
    mov r13, r12           # save i
    xor r14, r14           # j counter

.Lmul_j_loop:
    cmp r14, rsi
    jge .Lmul_next_i
    mov r15, r14           # save j

    # Compute dot product of row i (rbx) and column j
    xor rcx, rcx           # accumulator
    xor r8, r8             # t index

.Lmul_t_loop:
    cmp r8, rsi
    jge .Lmul_col_done
    # element at [i,t] from first matrix: rbx[i*n + t]
    mov rax, r12           # i
    imul rax, rsi          # i*n
    add rax, r8            # i*n + t
    mov r9, qword ptr [rbx + rax*8]

    # element at [t,j] from second matrix: rbx[t*n + j]
    mov r10, r8            # t
    imul r10, rsi          # t*n
    add r10, r15           # t*n + j
    mov r11, qword ptr [rbx + r10*8]

    # Multiply and accumulate
    imul r9, r11           # r9 = element1 * element2 (wraps mod 2^64)
    add rcx, r9            # acc += product

    inc r8
    jmp .Lmul_t_loop

.Lmul_col_done:
    # Store result[rcx] at position [i*n+j]
    mov rax, r13           # i
    imul rax, rsi          # i*n
    add rax, r15           # i*n + j
    mov qword ptr [rdi + rax*8], rcx

    inc r14
    jmp .Lmul_j_loop

.Lmul_next_i:
    inc r12
    jmp .Lmul_i_loop

.Lmul_done_rows:
    # Copy result from local buffer to output (tmp+ n*n)
    lea r9, [rbp - 64]     # source: local result
    lea r10, [r8 + r11*8]  # dest: tmp + n*n
    xor r12, r12           # counter

.Lcopy_back:
    cmp r12, r11
    jge .Lmul_exit
    mov rax, qword ptr [r9 + r12*8]
    mov qword ptr [r10 + r12*8], rax
    inc r12
    jmp .Lcopy_back

.Lmul_exit:
    pop rdi
    pop rsi
    pop rbx
    leave
    ret

# Helper: square base matrix (in place)
.Lmat_mul_base:
    # rdi = base pointer, rsi = n*n, tmp dest at r8+n*n
    # We will compute base * base into r8 + r11*8
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 64            # local buffer for result
    push rsi               # save n*n
    push rdi               # save base pointer

    lea r9, [rbp - 64]     # local result buffer
    mov rbx, rdi           # base matrix

    # For each row i:
    xor r12, r12           # i counter
.Lsq_i_loop:
    cmp r12, rsi
    jge .Lsq_done_rows
    mov r13, r12           # save i
    xor r14, r14           # j counter

.Lsq_j_loop:
    cmp r14, rsi
    jge .Lsq_next_i
    mov r15, r14           # save j

    # Compute dot product of row i and column j of base
    xor rcx, rcx           # accumulator
    xor r8, r8             # t index

.Lsq_t_loop:
    cmp r8, rsi
    jge .Lsq_col_done
    # [i,t]
    mov rax, r12           # i
    imul rax, rsi          # i*n
    add rax, r8            # i*n + t
    mov r9, qword ptr [rbx + rax*8]

    # [t,j]
    mov r10, r8            # t
    imul r10, rsi          # t*n
    add r10, r15           # t*n + j
    mov r11, qword ptr [rbx + r10*8]

    imul r9, r11
    add rcx, r9

    inc r8
    jmp .Lsq_t_loop

.Lsq_col_done:
    # Store to local buffer
    mov rax, r13           # i
    imul rax, rsi          # i*n
    add rax, r15           # i*n + j
    mov qword ptr [r9 + rax*8], rcx

    inc r14
    jmp .Lsq_j_loop

.Lsq_next_i:
    inc r12
    jmp .Lsq_i_loop

.Lsq_done_rows:
    # Copy result to tmp+ n*n (overwrite base with square)
    lea r9, [rbp - 64]     # source: local result
    pop rdi                # base pointer restored
    pop rsi                # n*n restored
    lea r10, [rdi + rsi*8] # dest = tmp + n*n

    xor r12, r12           # counter
.Lsq_copy_back:
    cmp r12, rsi
    jge .Lsq_exit
    mov rax, qword ptr [r9 + r12*8]
    mov qword ptr [r10 + r12*8], rax
    inc r12
    jmp .Lsq_copy_back

.Lsq_exit:
    leave
    ret
