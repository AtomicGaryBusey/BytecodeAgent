mat_pow:
    # rdi = a, rsi = n, rdx = k, rcx = out, r8 = tmp
    # callee-saved: rbx, rbp, r12, r13, r14, r15
    push rbp
    mov rbp, rsp
    sub rsp, 48        # allocate space for callee-saved and align stack (rsp was 8 mod 16, now 0 mod 16)
    push rbx
    push r12
    push r13
    push r14
    push r15

    # Check k == 0 -> identity matrix
    test rdx, rdx
    jnz .Lnot_zero_k

    # Identity matrix: out[i*n + i] = 1, others 0
    mov r9, rcx           # out pointer
    xor rax, rax          # zero register
    mov rbx, rsi          # n
    mul rbx               # rdx:rax = n*n (max 36), use rax as count
    cdq                   # rdx = sign extend of rax (0 for positive)
    xor rcx, rcx          # i counter
.Lidentity_loop:
    cmp rcx, rsi          # i < n?
    jge .Lidentity_done
    # Set diagonal element to 1
    mov r10, rcx
    mul r10, rsi           # r10d = i*n
