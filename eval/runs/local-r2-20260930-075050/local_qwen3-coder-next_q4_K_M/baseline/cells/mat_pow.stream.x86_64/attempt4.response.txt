mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 64

    test rdx, rdx
    jnz .Lnot_zero_k

.Lidentity_only:
    mov rbx, rcx
    xor rax, rax
    mov rbp, rsi
    imul rbp, rbp
    lea r12, [rbp]
    lea r13, [r12 + r12]
    xor r14, r14
.Lidentity_fill:
    cmp r14, rsi
    jge .Lidentity_done
    mov rax, r14
    imul rax, rsi
    add rax, r14
    mov qword ptr [rbx + rax*8], 1
    inc r14
    jmp .Lidentity_fill
.Lidentity_done:
    xor rax, rax
    lea rsp, [rsp + 64]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lnot_zero_k:
    mov r8, rdi
    mov r9, r8
    mov r10, rsi
    imul r10, r10
    lea r11, [r10 + r10]
    lea r12, [r8 + r10*8]

.Lcopy_to_tmp:
    cmp rbx, r10
    jge .Lcopy_done
    mov rax, qword ptr [r9 + rbx*8]
    mov qword ptr [r8 + rbx*8], rax
    inc rbx
    jmp .Lcopy_to_tmp
.Lcopy_done:

    xor rbx, rbx

.Lclear_all_tmp:
    cmp rbx, r11
    jge .Ltmp_cleared
    mov qword ptr [r8 + rbx*8], 0
    inc rbx
    jmp .Lclear_all_tmp
.Ltmp_cleared:

    lea r12, [r8]
    lea r13, [r12 + r10*8]

.Linit_result:
    cmp r14, rsi
    jge .Lresult_inited
    mov rax, r14
    imul rax, rsi
    add rax, r14
    mov qword ptr [r12 + rax*8], 0
    inc r14
    jmp .Linit_result
.Lresult_inited:

    xor r14, r14

.Lset_id_diag:
    cmp r14, rsi
    jge .Lid_set_done
    mov rax, r14
    imul rax, rsi
    add rax, r14
    mov qword ptr [r12 + rax*8], 1
    inc r14
    jmp .Lset_id_diag
.Lid_set_done:

    xor rbx, rbx

.Lmat_loop:
    test rdx, rdx
    jz .Ldone_exp

    mov rcx, rdx
    and rcx, 1
    test rcx, rcx
    jz .Lskip_mult

    lea rdi, [r12]
    lea rsi, [r12 + r10*8]
    call .Lmat_mul

.Lskip_mult:
    lea rdi, [r12]
    lea rsi, [r12 + r10*8]
    call .Lmat_square

    shr rdx, 1
    jmp .Lmat_loop

.Ldone_exp:
    mov rax, rcx
    xor rax, rax
    lea rsp, [rsp + 64]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lmat_mul:
    push rbp
    mov rbp, rsp
    sub rsp, 384

    xor r12, r12

.Lmul_i_loop:
    cmp r12, rsi
    jge .Lmul_done_rows
    mov r13, r12
    xor r14, r14

.Lmul_j_loop:
    cmp r14, rsi
    jge .Lmul_next_row
    mov r15, r14

    xor rbx, rbx
    xor rcx, rcx

.Lmul_t_loop:
    cmp rbx, rsi
    jge .Lmul_col_done
    mov rax, r12
    imul rax, rsi
    add rax, rbx
    mov r8, qword ptr [rdi + rax*8]

    mov r9, rbx
    imul r9, rsi
    add r9, r15
    mov r9, qword ptr [rsi - r10*8 + r9*8]

    im
