mat_pow:
    push rbp
    mov rbp, rsp
    sub rsp, 48
    push rbx
    push r12
    push r13
    push r14
    push r15

    test rdx, rdx
    jnz .Lnot_zero_k

    # Identity matrix: out[i*n + i] = 1, others 0
    mov rbx, rcx           # out pointer
    xor rcx, rcx           # i counter
.Lidentity_loop:
    cmp rcx, rsi           # i < n?
    jge .Lidentity_done
    mov rax, rcx
    imul rax, rsi          # rax = i*n
    add rax, rcx           # rax = i*n + i
    mov qword ptr [rbx + rax*8], 1
    inc rcx
    jmp .Lidentity_loop
.Lidentity_done:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    leave
    ret

.Lnot_zero_k:
    # Copy input matrix a to tmp (for base)
    mov r9, rdi            # a pointer
    mov r10, r8            # tmp pointer
    imul r11, rsi, rsi     # n*n
    xor
