export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
%out_w =w copy %cap
	%src_ptr =l copy %src
	%out_ptr =l copy %out
	%n_remaining =l copy %n
	%written =l copy 0

@loop_header:
	%bytes_left =l sub %n, %n_remaining
	%cmp_done =w csltl %bytes_left, %n
	jnz @done %cmp_done
	jmp @process_packet

@process_packet:
	%h_signed =b loadsb %src_ptr
	%src_ptr_next =l add %src_ptr, 1
	%n_remaining_new =l sub %n_remaining, 1
	%cmp_nego =w csltl %h_signed, 0
	jnz @literal_packet %cmp_nego

@repeat_packet:
	%h_plus1 =w add %h_signed, 1
	%count =w sub -1, %h_plus1
	%cmp_count_zero =w ceql %count, 0
	jnz @skip_repeat %cmp_count_zero
	%src_left =l sub %n_remaining_new, 1
	%cmp_enough_src =w csltl 1, %src_left
	jnz @error_src %cmp_enough_src
	%c =b loadub %src_ptr_next

@repeat_loop:
	%cmp_written_cap =w csltl %written, %out_w
	jz @exit_write_cap %cmp_written_cap
	storeb %c, %out_ptr
	%out_ptr_next =l add %out_ptr, 1
	%out_ptr =l copy %out_ptr_next
	%written =l add %written, 1
	%count =w sub %count, 1
	jnz @repeat_loop %count
	jmp @next_packet

@skip_repeat:
	jmp @next_packet

@literal_packet:
	%h_val =w and %h_signed, 255
	%lit_count =w add %h_val, 1
	%src_left =l sub %n_remaining_new, %lit_count
	%cmp_enough_src =w csltl %lit_count, %src_left
	jnz @error_src %cmp_enough_src

	live %src_ptr_next, %out_w, %written, %out_ptr, %n_remaining_new, %lit_count

@literal_loop:
	%cmp_written_cap =w csltl %written, %out_w
	jz @exit_write_cap %cmp_written_cap
	%c =b loadub %src_ptr_next
	storeb %c, %out_ptr
	%src_ptr_tmp =l add %src_ptr_next, 1
	%src_ptr_next =l copy %src_ptr_tmp
	%out_ptr_next =l add %out_ptr, 1
	%out_ptr =l copy %out_ptr_next
	%written =l add %written, 1
	%lit_count =w sub %lit_count, 1
	jnz @literal_loop %lit_count

@next_packet:
	%n_remaining =l copy %src_left
	%src_ptr =l copy %src_ptr_next
	jmp @loop_header

@error_src:
	%ret =l copy -2
	ret %ret

@exit_write_cap:
	%cmp_written_exact =w ceql %written, %out_w
	jnz @full_out %cmp_written_exact
	%ret =l copy -1
	ret %ret

@full_out:
	%ret =l copy %written
	ret %ret

@done:
	%ret =l copy %written
	ret %ret
}
