export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
%written =l copy 0
%src_ptr =l copy %src
%out_ptr =l copy %out
%n_remaining =l copy %n

@loop_header:
	%bytes_read =l sub %n, %n_remaining
	%done_check =w csltl %bytes_read, %n
	jnz @end_loop %done_check
	jmp @process_packet

@process_packet:
	%h_signed =b loadsb %src_ptr
	%next_src =l add %src_ptr, 1
	%n_after_header =l sub %n_remaining, 1
	%cmp_nego =w csltl %h_signed, 0
	jnz @literal_case %cmp_nego

@repeat_case:
	%count =w sub -1, %h_signed
	%cmp_skip =w ceql %count, 0
	jz @next_packet %cmp_skip
	%src_left =l sub %n_after_header, 1
	%enough_src =w csltl 1, %src_left
	jnz @error_src %enough_src
	%c =b loadub %next_src

@repeat_loop:
	%cmp_written =w csltl %written, %cap
	jz @cap_reached %cmp_written
	storeb %c, %out_ptr
	%out_next =l add %out_ptr, 1
	%out_ptr =l copy %out_next
	%written =l add %written, 1
	%count =w sub %count, 1
	jnz @repeat_loop %count
	jmp @next_packet

@literal_case:
	%h_val =w and %h_signed, 255
	%lit_count =w add %h_val, 1
	%needed =l mul %lit_count, 1
	%src_left =l sub %n_after_header, %lit_count
	%enough_src =w csltl %lit_count, %src_left
	jnz @error_src %enough_src

@literal_loop:
	%cmp_written =w csltl %written, %cap
	jz @cap_reached %cmp_written
	%c =b loadub %next_src
	storeb %c, %out_ptr
	%next_src_next =l add %next_src, 1
	%next_src =l copy %next_src_next
	%out_next =l add %out_ptr, 1
	%out_ptr =l copy %out_next
	%written =l add %written, 1
	%lit_count =w sub %lit_count, 1
	jnz @literal_loop %lit_count

@next_packet:
	%n_remaining =l copy %src_left
	%src_ptr =l copy %next_src
	jmp @loop_header

@end_loop:
	%ret =l copy %written
	ret %ret

@error_src:
	%ret =l copy -2
	ret %ret

@cap_reached:
	%full_cap =w ceql %written, %cap
	jz @return_minus_one %full_cap
	%ret =l copy -1
	ret %ret

@return_minus_one:
	%ret =l copy -1
	ret %ret
}
