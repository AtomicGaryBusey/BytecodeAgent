	%negtwob = neg l %twob

	%fourd = mul l %d, 4
	%ninfourd = neg l %fourd
	%fivee = mul l %e, 5
	%sixf = mul l %f, 6
	%ninsixf = neg l %sixf
	%seveng = mul l %g, 7
	%eighth = mul l %h, 8
	%negeighth = neg l %eighth

	%tmp1 = add l %a, %negtwob

}
