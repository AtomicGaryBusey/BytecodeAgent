fib_rec:
    cmp rdi, 1
    jle .Lbase
    # n >= 2: compute fib(n-1) + fib(n-2)
    push rbx                # save rbx (callee-saved)
    sub rsp, 8              # align stack to 16-byte (rsp was 8 mod 16, now 0 mod 16)
    
    mov rbx, rdi            # save original n in rbx
    dec rdi                 # prepare n-1 for first call
    call fib_rec            # compute fib(n-1), result in rax
    
    add rsp, 8              # restore stack alignment (rsp = 8 mod 16 again)
    mov r9, rax             # save fib(n-1) in r9 (caller-saved is ok temporarily)
    
    sub rsp, 8              # align stack to 16-byte for second call
    mov rdi, rbx            # restore original n
    dec rdi                 # prepare n-1
    dec rdi                 # prepare n-2
    call fib_rec            # compute fib(n-2), result in rax
    
    add rsp, 8              # restore stack alignment (rsp = 8 mod 16 again)
    add rax, r9             # add fib(n-1) and fib(n-2)
    
    pop rbx                 # restore rbx
    ret

.Lbase:
    mov rax, rdi            # return n (0 or 1) for base case
    ret
