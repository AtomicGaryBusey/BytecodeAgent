fib_rec:
    cmp rdi, 1
    jle .Lbase
    # n >= 2: naive recursion requires two calls
    push rbx                # save callee-saved rbx (rsp: 8 mod 16 -> 0 mod 16 after push)
    sub rsp, 8              # align to 16-byte (rsp: 0 mod 16 -> 8 mod 16 before call)
    
    mov rbx, rdi            # save original n in rbx
    dec rdi                 # prepare n-1 for first recursive call
    call fib_rec            # compute fib(n-1), result in rax
    
    mov r9, rax             # save fib(n-1) in caller-saved r9
    
    mov rdi, rbx            # restore original n
    dec rdi                 # prepare n-1 (n-2 total decrements)
    dec rdi                 # prepare n-2 for second recursive call
    call fib_rec            # compute fib(n-2), result in rax
    
    add rax, r9             # add fib(n-1) + fib(n-2)
    add rsp, 8              # restore stack (rsp: 8 mod 16 -> 0 mod 16 before pop)
    pop rbx                 # restore callee-saved rbx (rsp: 0 mod 16 -> 8 mod 16)
    ret

.Lbase:
    mov rax, rdi
    ret
