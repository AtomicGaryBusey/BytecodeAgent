export function l $fib(l %n) {
%0 = copy l %n
%1 = ceql l %0, 0
jnz %1, @ret0, @cont1
@ret0:
%2 = copy l 0
ret l %2
@cont1:
%3 = ceql l %0, 1
jnz %3, @ret1, @iter
@ret1:
%4 = copy l 1
ret l %4
@iter:
%a = copy l 0
%b = copy l 1
%i = copy l 2
@loop:
%5 = csltl l %i, %n
jz %5, @done
%c = addl %a, %b
%a = copy l %b
%b = copy l %c
%i = addl %i, 1
jmp @loop
@done:
ret l %b
}
