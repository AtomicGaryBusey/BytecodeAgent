export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc = l 0
	%dsp = l 0
	%tmpi = l 0
	%tmpv = l 0
	%tmpt = l 0
	%tmpa = l 0
	%tmpb = l 0
	%op = l 0
	%ret = l 0

@loop
	%pc_eq_len = w %pc == %len
	jnz %pc_eq_len, @done

	%op = loadl l [%code + %pc * 8]
	%inc_pc = l %pc + 1

	cswtl %op, %case_0, %case_1, %case_2, %case_3, %case_4, %case_5, %case_6, %case_7, %case_8, %case_9, %case_10, %case_11

@default_instr
	%pc = l %inc_pc
	jmp @loop

@case_0
@HALT
	jmp @done

@case_1
@PUSH
	%tmpv = loadl l [%code + %inc_pc * 8]
	%tmpi = l %dsp
	storel l %tmpv, [%stack + %tmpi * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc + 1
	jmp @loop

@case_2
@ADD
	%dsp = l %dsp - 1
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa + %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_3
@SUB
	%dsp = l %dsp - 1
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa - %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_4
@MUL
	%dsp = l %dsp - 1
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa * %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_5
@DUP
	%tmpi = l %dsp - 1
	%tmpv = loadl l [%stack + %tmpi * 8]
	storel l %tmpv, [%stack + %dsp * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc
	jmp @loop

@case_6
@SWAP
	%dsp = l %dsp - 1
	%tmpa = loadl l [%stack + (%dsp - 1) * 8]
	%tmpb = loadl l [%stack + %dsp * 8]
	storel l %tmpb, [%stack + (%dsp - 1) * 8]
	storel l %tmpa, [%stack + %dsp * 8]
	jmp @loop

@case_7
@DROP
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_8
@OVER
	%tmpi = l %dsp - 2
	%tmpv = loadl l [%stack + %tmpi * 8]
	storel l %tmpv, [%stack + %dsp * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc
	jmp @loop

@case_9
@LT
	%dsp = l %dsp - 1
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = w %tmpa < %tmpb
	%tmpv = l %tmpt ? 1 : 0
	storel l %tmpv, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_10
@JZ
	%dsp = l %dsp - 1
	%tmpv = loadl l [%stack + %dsp * 8]
	%is_zero = w %tmpv == 0
	%tmpt = loadl l [%code + %inc_pc * 8]
	jnz %is_zero, @jz_take_branch, @jz_skip_branch

@jz_skip_branch
	%pc = l %inc_pc + 1
	jmp @loop

@jz_take_branch
	%pc = l %tmpt
	jmp @loop

@case_11
@JMP
	%tmpt = loadl l [%code + %inc_pc * 8]
	%pc = l %tmpt
	jmp @loop

@done
	%dsp_ge_0 = w %dsp > 0
	jnz %dsp_ge_0, @has_value, @empty_stack

@empty_stack
	%ret = l 0
	jmp @cleanup

@has_value
	%tmpi = l %dsp - 1
	%ret = loadl l [%stack + %tmpi * 8]

@cleanup
	%i = l 0
@zero_loop
	%cmp = w %i >= %cap
	jnz %cmp, @end_zero
	storel l 0, [%stack + %i * 8]
	%i = l %i + 1
	jmp @zero_loop

@end_zero
	ret %ret
}
