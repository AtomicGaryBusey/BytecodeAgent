export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc = l 0
	%dsp = l 0
	%tmpi = l 0
	%tmpv = l 0
	%tmpt = l 0
	%tmpa = l 0
	%tmpb = l 0
	%op = l 0

@loop
	%pc_eq_len = w %pc == %len
	jnz %pc_eq_len, @done

	%op = loadl l [%code + %pc * 8]
	%inc_pc = l %pc + 1

	cswtl %op, @case_0, @case_1, @case_2, @case_3, @case_4, @case_5, @case_6, @case_7, @case_8, @case_9, @case_10, @case_11

@default_instr
	%pc = l %inc_pc
	jmp @loop

@case_0
@HALT
	jmp @done

@case_1
@PUSH
	%tmpv = loadl l [%code + %inc_pc * 8]
	storel l %tmpv, [%stack + %dsp * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc + 1
	jmp @loop

@case_2
@ADD
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa + %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_3
@SUB
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa - %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_4
@MUL
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = l %tmpa * %tmpb
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_5
@DUP
	%tmpi = l %dsp - 1
	%tmpv = loadl l [%stack + %tmpi * 8]
	storel l %tmpv, [%stack + %dsp * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc
	jmp @loop

@case_6
@SWAP
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	storel l %tmpb, [%stack + (%dsp - 2) * 8]
	storel l %tmpa, [%stack + (%dsp - 1) * 8]
	%pc = l %inc_pc
	jmp @loop

@case_7
@DROP
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_8
@OVER
	%tmpv = loadl l [%stack + (%dsp - 2) * 8]
	storel l %tmpv, [%stack + %dsp * 8]
	%dsp = l %dsp + 1
	%pc = l %inc_pc
	jmp @loop

@case_9
@LT
	%tmpb = loadl l [%stack + (%dsp - 1) * 8]
	%tmpa = loadl l [%stack + (%dsp - 2) * 8]
	%tmpt = w %tmpa < %tmpb ? 1 : 0
	storel l %tmpt, [%stack + (%dsp - 2) * 8]
	%dsp = l %dsp - 1
	%pc = l %inc_pc
	jmp @loop

@case_10
@JZ
	%dsp = l %dsp - 1
	%tmpv = loadl l [%stack + %dsp * 8]
	%is_zero = w %tmpv == 0
	%tmpt = loadl l [%code + %inc_pc * 8]
	jnz %is_zero, @jz_take_branch, @jz_skip_branch

@jz_skip_branch
	%pc = l %inc_pc + 1
	jmp @loop

@jz_take_branch
	%pc = l %tmpt
	jmp @loop

@case_11
@JMP
	%tmpt = loadl l [%code + %inc_pc * 8]
	%pc = l %tmpt
	jmp @loop

@done
	%ret_val = l 0
	%is_nonempty = w %dsp != 0
	jnz %is_nonempty, @stack_has_value, @return_zero

@stack_has_value
	%ret_val = loadl l [%stack + (%dsp - 1) * 8]
	jmp @return_result

@return_zero
	%ret_val = l 0

@return_result
	ret %ret_val
}
