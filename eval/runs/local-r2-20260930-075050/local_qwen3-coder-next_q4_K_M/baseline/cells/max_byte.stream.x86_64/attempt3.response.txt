max_byte:
    xor eax, eax          # result = 0
    test esi, esi
    jz .Lret_zero         # if n == 0, return 0

    movzx ebx, byte ptr [rdi]  # load first byte as initial max (zero-extend)
    xor ecx, ecx          # i = 0
    dec rsi               # n-- to get last index to check
    .Lloop:
        cmp bl, [rdi + rcx + 1]   # compare current max with next byte
        cmovb ebx, dword ptr [rdi + rcx + 1]  # update max if current > existing
        inc ecx
        cmp ecx, rsi
        j
