max_byte:
    # rdi = p, rsi = n
    xor eax, eax          # result = 0
    test esi, esi
    jz .Lret_zero         # if n == 0, return 0

    mov ebx, [rdi]        # load first byte as initial max
    xor ecx, ecx          # i = 0
    dec rsi               # n-- for loop condition
    .Lloop:
        cmp cl, [rdi + rcx]
        cmovb ebx, dword ptr [rdi + rcx]  # update max if current byte > max
        inc rcx
        cmp rcx, rsi
        jbe .Lloop

    mov eax, ebx
.Lret_zero:
    ret
