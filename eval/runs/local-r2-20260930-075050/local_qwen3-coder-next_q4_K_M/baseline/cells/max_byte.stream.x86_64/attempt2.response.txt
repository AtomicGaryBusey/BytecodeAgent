max_byte:
    # rdi = p, rsi = n
    xor eax, eax          # result = 0
    test esi, esi
    jz .Lret_zero         # if n == 0, return 0

    movzx ebx, byte ptr [rdi]  # load first byte as initial max (zero-extend)
    xor ecx, ecx          # i = 0
    dec rsi               # n-- for loop condition
    .Lloop:
        cmp bl, [rdi + rcx]   # compare current max with current byte
        cmovb ebx, dword ptr [rdi + rcx]  # update max if current > existing
        inc rcx
        cmp rcx, rsi
        jbe .Lloop

    mov eax, ebx
.Lret_zero:
    ret
