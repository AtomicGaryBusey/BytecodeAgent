isqrt:
    push rbp
    mov rbp, rsp
    sub rsp, 16                  # allocate stack frame
    mov [rbp-8], rdi             # save n
    xor rax, rax                 # result = 0
    xor rbx, rbx                 # low = 0
    mov rcx, 0x100000000         # high = 2^32 (upper bound for sqrt)
    cmp [rbp-8], rcx             # if n < 2^32, adjust high
    jb .Ladjust_high
    jmp .Lbinary_search

.Ladjust_high:
    mov rcx, [rbp-8]             # high = n+1 to ensure coverage
    inc rcx

.Lbinary_search:
    cmp rbx, rcx                 # while low < high
    jge .Ldone
    mov rdx, rbx                 # mid = (low + high) / 2
    add rdx, rcx
    shr rdx, 1
    xor rax, rax                 # compute mid * mid safely
    mov rax, rdx
    mul rdx                      # rdx:rax = mid^2
    cmp [rbp-8], rax             # compare n with low part of product
    ja .Lproduct_less            # if n > mid^2 (low part), need high check
    jb .Lproduct_greater         # if n < mid^2, definitely too big
    cmp qword ptr [rbp-8+8], rdx # compare high word (n is 64-bit, so high=0)
    ja .Lproduct_less            # n > product (impossible for n<2^64 but safe)
    je .Lproduct_equal

.Lproduct_greater:
    mov rcx, rdx                 # high = mid
    jmp .Lbinary_search

.Lproduct_less:
    mov [rbp-16], rax            # save mid*mid low part
    test rdx, rdx                # check if product exceeds 64 bits
    jnz .Lproduct_greater        # if high word !=0, product > 2^64-1, too big
    mov [rbp-8], rax             # update n? no: keep original
    mov rbx, rdx                 # low = mid + 1 (but first check mid+1)
    add rdx, 1                   # test mid+1
    xor rax, rax
    mul rdx                      # (mid+1)^2
    cmp [rbp-8], rax             # if (mid+1)^2 <= n
    ja .Lproduct_greater         # then mid is too small
    mov rax, rdx                 # return mid+1
    jmp .Ldone

.Lproduct_equal:
    mov rax, rdx                 # exact match
    jmp .Ldone

.Ldone:
    mov rax, rbx                 # result = low - 1? No: track best candidate
    mov rsp, rbp
    pop rbp
    ret
