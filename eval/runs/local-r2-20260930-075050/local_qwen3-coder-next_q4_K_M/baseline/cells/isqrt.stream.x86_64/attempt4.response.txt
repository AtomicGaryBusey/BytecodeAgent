isqrt:
    push rbx              # save callee-saved register
    push rbp
    mov rax, rdi          # copy n to rax for later use
    xor rcx, rcx          # low = 0
    mov rdx, 0x100000000  # high = 2^32 (upper bound for sqrt of uint64)
.Lloop:
    cmp rcx, rdx          # while low < high
    jge .Ldone
    mov rbx, rcx          # mid = (low + high) / 2
    add rbx, rdx
    shr rbx, 1            # rbx = mid
    # compute mid * mid safely using mul to get full product in rdx:rax
    mov rbp, rbx          # save mid in rbp (callee-saved)
    mul rbx               # rdx:rax = mid^2
    # compare with n (original n is still in rdi)
    cmp rdi, rdx          # high word comparison first
    ja .Ltoo_big          # if n < product high word, too big (since n's high word is 0 for valid uint64 inputs)
    jb .Lcheck_high       # if n > product high word, definitely not too big
    # high words equal, check low word
    cmp rdi, rax          # compare low parts
    ja .Lfound            # n > mid^2, might be ok (but need to verify mid+1 doesn't fit)
    jb .Ltoo_big          # n < mid^2 -> too big
    je .Lfound            # exact match
.Lcheck_high:
    # n >= low_part(mid^2), but need to check if (mid+1)^2 <= n
    mov rbx, rbp          # restore mid into rbx
    # compute (mid+1)^2 = mid^2 + 2*mid + 1 without overflow
    mov rax, rbp          # mid
    add rax, rax          # 2*mid
    jc .Ltoo_big          # if carry, 2*mid >= 2^64 -> (mid+1)^2 overflows -> too big
    add rax, rdi          # we'll compute in a safe way: just check if mid is maximal
    # simpler approach: if n >= mid^2 and (mid+1) would overflow product or exceed n, then mid is answer
    # let's recompute: mid^2 = rdx:rax. We want to know if (mid+1)^2 <= n.
    # But easier: check if mid+1 fits in sqrt range by testing (mid+1)*(mid+1) > n.
    mov rbx, rbp          # mid
    inc rbx               # mid + 1
    cmp rbx, rdx          # compare with high bound to avoid overflow
    ja .Lfound            # if mid+1 >= 2^32, then it's too big (since sqrt(2^64-1) < 2^32)
    mul rbx               # rdx:rax = (mid+1)^2
    cmp rdi, rdx          # high word comparison
    ja .Lfound            # n > (mid+1)^2 high word -> mid is too small? No: wait, this logic is reversed.
.Lfixed_check:
    # Let's restart with correct binary search logic:
    # We want largest r such that r*r <= n.
    # Binary search on [low, high), so when low == high we stop.
    # At each step, mid = (low+high)/2
    # If mid*mid <= n: then answer is >= mid, set low = mid + 1 (so final result is low-1)
    # Else: set high = mid
    mov rbx, rbp          # restore mid
    mul rbx               # rdx:rax = mid^2 again
    # Check if mid^2 <= n
    xor rsi, rsi          # clear for comparison
    cmp rdx, rsi          # is high part > 0?
    ja .Ltoo_big          # mid^2 >= 2^64 -> too big (since n < 2^64)
    jb .Lmid_ok           # high part < n's high part (which is 0), so definitely ok
    # high parts equal, check low part
    cmp rax, rdi          # mid^2 <= n?
    ja .Ltoo_big          # too big
.Lmid_ok:
    # mid^2 <= n: try larger
    mov rbx, rbp          # mid
    inc rbx               # mid+1 to check if next fits
    cmp rbx, 0x100000000  # can't exceed 2^32 because (2^32)^2 = 2^64 > uint64_max
    j
