isqrt:
    mov rbx, rdi                  # copy input n to rbx for safekeeping
    xor rax, rax                 # result r = 0
    xor rcx, rcx                 # low bound = 0
    mov rdx, 1                   # high bound = 1 (will grow)
    cmp rbx, 0xffffffffffffffff  # handle max value specially? not needed
    jbe .Lstart_binary_search    # if n < 2^64-1, proceed normally

.Lbinary_search:
    # Compute mid = (low + high) / 2 without overflow
    mov r8, rcx                  # low
    mov r9, rdx                  # high
    lea r10, [r8 + r9]           # low + high
    shr r10, 1                   # mid = (low + high) / 2

    # Compute mid * mid carefully to avoid overflow
    xor rax, rax                 # clear rax for multiplication result
    mov rax, r10                 # copy mid
    mul r10                      # rdx:rax = mid * mid (unsigned multiply)
                                 # since rdx:rax is 128-bit product

    # Compare product with n (rbx)
    cmp rax, rbx                 # compare low 64 bits of product to n
    ja .Lproduct_too_big         # if mid*mid > n (considering overflow)
                                 # but need to check high word too
    jna .Lproduct_ok             # if product <= n

.Lproduct_too_big:
    mov rdx, r10                 # high = mid - 1 (since we want floor(sqrt(n)))
    sub rdx, 1                   # high = mid - 1
    jmp .Lcheck_done

.Lproduct_ok:
    # Product <= n, so mid is a candidate; try to increase low
    mov rax, r10                 # current candidate in rax
    mov rcx, r10                 # low = mid + 1? No: set low = mid + 1 only after confirming better?
                                 # Standard binary search: keep best in result

    mov rax, r10                 # update result to mid (since mid*mid <= n)
    add rcx, 1                   # low = mid + 1
    cmp rcx, rdx                 # is low > high?
    jg .Ldone                    # if so, we're done

.Lcheck_done:
    cmp rcx, rdx                 # check again for loop condition
    jle .Lbinary_search          # continue while low <= high

.Ldone:
    ret

.Lstart_binary_search:
    # Initialize binary search bounds: low=0, high=min(n, 2^32)
    xor rcx, rcx                 # low = 0
    mov rdx, 0x100000000         # high = 2^32 (since sqrt(2^64-1) < 2^32)
    cmp rbx, rdx                 # if n >= 2^32, keep high=2^32
    jae .Lbinary_search          # proceed with binary search
    mov rdx, rbx                 # else high = n (but actually sqrt(n) <= n for n>=1)
    inc rdx                      # high = n+1 to cover edge cases
    jmp .Lbinary_search
