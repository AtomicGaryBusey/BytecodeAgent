isqrt:
    xor rax, rax          # result = 0
    xor rcx, rcx          # low = 0
    mov rdx, 0x100000000  # high = 2^32 (upper bound for sqrt of uint64)
.Lloop:
    cmp rcx, rdx          # while low < high
    jge .Ldone
    mov rbx, rcx          # mid = (low + high) / 2
    add rbx, rdx
    shr rbx, 1
    # compute mid * mid safely: use mul to get full product in rdx:rax
    mov rsi, rbx          # save mid
    mul rbx               # rdx:rax = mid^2
    # compare with n (in rdi)
    cmp rdi, rax          # low part
    jb .Ltoo_big          # if n < low_part(mid^2), definitely too big
    ja .Lcheck_high       # if n > low_part, maybe ok but need to check overflow
    test rdx, rdx         # high word == 0?
    jnz .Ltoo_big         # if high word !=0, product > 2^64-1, too big
    je .Lfound            # exact match (low part equal and no overflow)
.Lcheck_high:
    # n >= low_part(mid^2), but might still be too big if mid+1 also fits
    mov rax, rsi          # mid
    inc rax               # mid + 1
    mul rsi               # (mid+1)*mid (we'll use this to check if (mid+1)^2 > n)
    lea r8, [rax + rsi]   # (mid+1)*mid + mid = mid*(mid+1) + mid = mid^2 + 2*mid
                          # actually: mul gives rdx:rax = (mid+1)*mid
                          # need (mid+1)^2 = mid^2 + 2*mid + 1
    add r8, rsi           # r8 = mid^2 + 2*mid (from previous rdx:rax, we lost the original low part)
    dec r8                # r8 = mid^2 + 2*mid - 1? Not quite: redo this cleanly
    mov rax, rsi          # mid
    mul rbx               # rdx:rax = mid * mid again (rbx was saved mid)
    mov r9, rax           # low part of mid^2
    test rdx, rdx         # check overflow
    jnz .Ltoo_big         # if high word !=0, too big
    # compute (mid+1)^2 = mid^2 + 2*mid + 1 without overflow
    mov rax, rsi          # mid
    add rax, rax          # 2*mid
    jo .Ltoo_big          # if carry in signed addition, definitely overflow
    add r9, rax           # low part += 2*mid (uint64 safe: if overflow, still >= n)
    jc .Ltoo_big          # carry means sum >= 2^64, so (mid+1)^2 > uint64_max, thus > n
    inc r9                # +1 for full square
    jc .Ltoo_big          # another carry: impossible but safe
    cmp rdi, r9           # if n < (mid+1)^2 then mid is answer
    jae .Lfound           # n >= (mid+1)^2 would mean mid+1 might work, but we just checked it doesn't
.Ltoo_big:
    mov rdx, rbx          # high = mid
    jmp .Lloop
.Lfound:
    mov rax, rbx          # result = mid
.Ldone:
    ret
