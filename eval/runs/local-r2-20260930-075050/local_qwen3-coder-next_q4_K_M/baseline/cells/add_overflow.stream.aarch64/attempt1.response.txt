add_overflow:
    // Save caller-saved registers we might use, set up stack frame
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Compute sum: out = a + b (mod 2^64)
    add x8, x0, x1        // x8 = a + b (wrapped)
    str x8, [x2]          // store *out = wrapped sum

    // Detect signed overflow for addition:
    // Overflow occurs if both operands have the same sign and result has different sign
    // Check signs: a<0? b<0? (sign bit is bit 63)
    lsrv x9, x0, #63      // x9 = (a >> 63) & 1  // 1 if a negative
    ls
