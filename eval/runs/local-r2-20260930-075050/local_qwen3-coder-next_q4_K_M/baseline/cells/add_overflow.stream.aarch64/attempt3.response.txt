add_overflow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Compute sum: out = a + b (mod 2^64)
    add x8, x0, x1        // x8 = a + b (wrapped)
    str x8, [x2]          // store *out = wrapped sum

    // Detect signed overflow for addition:
    // Overflow occurs if both operands have the same sign and result has different sign
    // Extract sign bits using ubfx
    ubfx x9, x0, #63, #1  // x9 = a.sign (1 if negative)
    ubfx x10, x1, #63, #1 // x10 = b.sign (1 if negative)
    ubfx x11, x8, #63, #1 // x11 = sum.sign (1 if negative)

    // Check for overflow: (a and b same sign) AND (result different sign)
    eor x12, x9, x10      // x12 = 1 if a and b have different signs -> no overflow possible
    cbnz x12, .Lno_overflow   // if signs differ, no overflow

    // Here: both operands have same sign. Check if result has opposite sign.
    eor x0, x9, x11       // x0 = 1 if result sign differs from operand sign (overflow)
    b .Ldone

.Lno_overflow:
    mov x0, #0            // no overflow when signs differ or both zero operands

.Ldone:
    ldp x29, x30, [sp], #16
    ret
