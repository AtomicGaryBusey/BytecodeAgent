add_overflow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Compute sum: out = a + b (mod 2^64)
    add x8, x0, x1        // x8 = a + b (wrapped)
    str x8, [x2]          // store *out = wrapped sum

    // Detect signed overflow for addition:
    // Overflow occurs if both operands have the same sign and result has different sign
    // Extract sign bits using ubfx (unsigned bit field extract)
    ubfx x9, x0, #63, #1  // x9 = (a >> 63) & 1  // 1 if a negative
    ubfx x10, x1, #63, #1 // x10 = (b >> 63) & 1 // 1 if b negative
    ubfx x11, x8, #63, #1 // x11 = (sum >> 63) & 1 // 1 if sum negative

    // Check: both positive (sign=0) but result negative (sign=1)
    // or both negative (sign=1) but result positive (sign=0)
    cmp x9, #1            // a < 0?
    cset x12, eq          // x12 = 1 if a negative
    cmp x10, #1           // b < 0?
    cset x13, eq          // x13 = 1 if b negative
    cbz x12, .Lno_overflow_a_negative // skip second check if a >= 0
    cbz x13, .Lno_overflow_b_positive // if both negative and one positive -> no overflow

    // Both negative: overflow if result non-negative (sign bit == 0)
    cmp x11, #0           // result sign? 0 = positive
    cset x0, eq           // overflow = 1 if result is positive (both inputs negative but sum >=0)
    b .Ldone

.Lno_overflow_b_positive:
    // Both positive: overflow if result negative (sign bit == 1)
    cmp x11, #1           // result sign? 1 = negative
    cset x0, eq           // overflow = 1 if result is negative (both inputs positive but sum <0)
    b .Ldone

.Lno_overflow_a_negative:
    // a >= 0; check if both positive: b must be positive and result negative?
    cmp x13, #1           // b negative? (we already handled case where b is negative above)
    cset x0, ne           // overflow = 1 only if b also non-negative and result negative?
    cbz x13, .Lboth_positive
    // One positive, one negative -> no overflow possible
    mov x0, #0
    b .Ldone

.Lboth_positive:
    cmp x11, #1           // result sign? 1 = negative
    cset x0, eq           // overflow = 1 if result is negative (sum overflows to negative)
    
.Ldone:
    ldp x29, x30, [sp], #16
    ret
