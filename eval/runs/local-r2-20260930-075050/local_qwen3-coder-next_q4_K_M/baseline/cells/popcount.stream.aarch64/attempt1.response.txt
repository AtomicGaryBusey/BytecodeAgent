popcount:
        mov     x1, #0          // counter for result
        cmp     x0, #0
        b.eq    .Ldone         // early exit if zero
.Lloop:
        sub     x2, x0, #1      // x-1
        and     x0, x0, x2      // clear lowest set bit (x & (x-1))
        add     x1, x1, #1      // increment count
        cmp     x0, #0
        b.ne    .Lloop         // repeat until zero
.Ldone:
        mov     x0, x1          // move result to return register
        ret                     // return
