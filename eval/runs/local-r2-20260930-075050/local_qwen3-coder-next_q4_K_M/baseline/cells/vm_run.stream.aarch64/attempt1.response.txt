vm_run:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    str x23, [sp, #40]

    // Initialize: pc = 0, sp_ptr = stack, depth = 0
    mov x19, x0          // x19 = code pointer
    mov x20, x1          // x20 = len
    mov x21, x2          // x21 = stack pointer
    mov x22, x3          // x22 = capacity
    mov x23, #0          // x23 = depth (stack size)

.Lloop:
    cmp x23, x22         // depth >= cap? (shouldn't happen per spec, but safety)
    b.ge .Lexit

    // Check if pc == len (run off end)
    cmp x24, x20
    csel x24, x24, xzr, eq  // if pc == len, stop now
    cbz x24, .Lexit

    ldr x25, [x19, x24, lsl #3]   // x25 = instruction at pc
    add x24, x24, #1              // pc++ (tentative)

    // Decode opcode: x25 & 0xF
    and x26, x25, #15             // x26 = opcode
    cbz x26, .Lop_halt            // op 0: HALT
    cmp x26, #1
    b.eq .Lop_push                // op 1: PUSH
    cmp x26, #2
    b.eq .Lop_add                 // op 2: ADD
    cmp x26, #3
    b.eq .Lop_sub                 // op 3: SUB
    cmp x26, #4
    b.eq .Lop_mul                 // op 4: MUL
    cmp x26, #5
    b.eq .Lop_dup                 // op 5: DUP
    cmp x26, #6
    b.eq .Lop_swap                // op 6: SWAP
    cmp x26, #7
    b.eq .Lop_drop                // op 7: DROP
    cmp x26, #8
    b.eq .Lop_over                // op 8: OVER
    cmp x26, #9
    b.eq .Lop_lt                  // op 9: LT
    cmp x26, #10
    b.eq .Lop_jz                  // op 10: JZ
    cmp x26, #11
    b.eq .Lop_jmp                 // op 11: JMP
    b .Lloop_next                 // default: pc +=1

// HALT
.Lop_halt:
    b .Lexit

// PUSH v (pc+=2)
.Lop_push:
    ldr x27, [x19, x24, lsl #3]   // x27 = operand v (at pc+1)
    add x24, x24, #2              // pc += 2
    str x27, [x21, x23, lsl #3]   // stack[depth] = v
    add x23, x23, #1              // depth++
    b .Lloop_next

// ADD: pop b, a, push a+b
.Lop_add:
    cbz x23, .Lexit               // empty stack? shouldn't happen per spec
    sub x23, x23, #1              // depth--
    ldr x0, [x21, x23, lsl #3]    // b = top (now popped)
    cbz x23, .Lexit               // now empty? no, should have a
    sub x23, x23, #1              // depth--
    ldr x1, [x21, x23, lsl #3]    // a = new top
    add x1, x1, x0                // a + b
    str x1, [x21, x23, lsl #3]    // store result as new top
    add x23, x23, #1              // depth++
    b .Lloop_next

// SUB: pop b, a, push a-b
.Lop_sub:
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x0, [x21, x23, lsl #3]    // b
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x1, [x21, x23, lsl #3]    // a
    sub x1, x1, x0                // a - b
    str x1, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

// MUL: pop b, a, push a*b (64-bit wrap)
.Lop_mul:
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x0, [x21, x23, lsl #3]    // b
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x1, [x21, x23, lsl #3]    // a
    mul x1, x1, x0                // a * b (signed multiplication)
    str x1, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

// DUP: push copy of top
.Lop_dup:
    cbz x23, .Lexit               // empty stack? can't dup
    ldr x0, [x21, x23-1, lsl #3]  // x0 = stack[depth-1]
    str
