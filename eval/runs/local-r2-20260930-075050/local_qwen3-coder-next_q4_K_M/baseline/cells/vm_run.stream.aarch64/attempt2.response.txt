vm_run:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    str x23, [sp, #40]

    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, x3
    mov x23, #0

.Lloop:
    cmp x23, x22
    b.ge .Lexit

    cmp x24, x20
    beq .Lexit

    ldr x25, [x19, x24, lsl #3]
    add x24, x24, #1

    and x26, x25, #15
    cbz x26, .Lop_halt
    cmp x26, #1
    beq .Lop_push
    cmp x26, #2
    beq .Lop_add
    cmp x26, #3
    beq .Lop_sub
    cmp x26, #4
    beq .Lop_mul
    cmp x26, #5
    beq .Lop_dup
    cmp x26, #6
    beq .Lop_swap
    cmp x26, #7
    beq .Lop_drop
    cmp x26, #8
    beq .Lop_over
    cmp x26, #9
    beq .Lop_lt
    cmp x26, #10
    beq .Lop_jz
    cmp x26, #11
    beq .Lop_jmp
    b .Lloop_next

.Lop_halt:
    b .Lexit

.Lop_push:
    ldr x27, [x19, x24, lsl #3]
    add x24, x24, #2
    str x27, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

.Lop_add:

    sub x23, x23, #1
    ldr x0, [x21, x23, lsl #3]
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x1, [x21, x23, lsl #3]
    add x1, x1, x0
    str x1, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

.Lop_sub:
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x0, [x21, x23, lsl #3]
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x1, [x21, x23, lsl #3]
    sub x1, x1, x0
    str x1, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

.Lop_mul:
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x0, [x21, x23, lsl #3]
    cbz x23, .Lexit
    sub x23, x23, #1
    ldr x1, [x21, x23, lsl #3]
    mul x1, x1, x0
    str x1, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

.Lop_dup:

    sub x24, x23, #1
    ldr x0, [x21, x24, lsl #3]
    str x0, [x21, x23, lsl #3]
    add x23, x23, #1
    b .Lloop_next

.Lop_swap:

    sub x24, x23, #2
    ldr x0, [x21, x24, lsl #3]
    ldr x1, [x21, x24, lsl #3, #8]
    str
