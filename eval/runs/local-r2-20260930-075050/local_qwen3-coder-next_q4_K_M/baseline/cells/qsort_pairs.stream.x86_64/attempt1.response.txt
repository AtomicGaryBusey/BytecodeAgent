qsort_pairs:
    push rbp
    mov rbp, rsp
    sub rsp, 48
    mov qword ptr [rbp-8], rdi
    mov qword ptr [rbp-16], rsi
    cmp rsi, 1
    jle .Lbase_case
    # Choose pivot: middle element (n//2)
    mov rax, rsi
    shr rax, 1
    lea rcx, [rdi + rax*8]
    lea rcx, [rcx + rax*8]        # rcx = &a[2*(n/2)] = pivot address
    # Save pivot values
    mov r8, qword ptr [rcx]
    mov r9, qword ptr [rcx+8]
    lea rbx, [rdi]                 # lb = left index (start)
    lea r10, [rdi + rsi*16]        # ub = right end (exclusive), points past last record
    dec
