qsort_pairs:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov qword ptr [rsp+8], rdi
    mov qword ptr [rsp+16], rsi
    cmp rsi, 1
    jle .Lbase_case
    mov rax, rsi
    shr rax, 1
    lea rcx, [rdi + rax*8]
    mov rbx, qword ptr [rcx]
    mov rdx, qword ptr [rcx+8]
    xor r8, r8
    xor r9, r9
    mov r10, rdi
    add r10, 16
    mov r11, rsi
    dec r11
.Lpartition_loop:
    cmp r8, r11
    jge .Lpartition_done
    mov r12, qword ptr [r10]
    mov r13, qword ptr [r10+8]
    cmp r12, rbx
    jl .Lswap_left
    cmp r12, rbx
    jg .Lright_side
    cmp r13, rdx
    jle .Lswap_left
.Lright_side:
    sub r11, 2
    mov r14, qword ptr [r10 + r11*8]
    mov r15, qword ptr [r10 + r11*8+8]
    cmp r14, rbx
    jg .Lswap_right
    cmp r14, rbx
    jl .Lleft_side
    cmp r15, rdx
    jge .Lswap_right
.Lleft_side:
    add r8, 2
    jmp .Lpartition_loop
.Lswap_left:
    mov qword ptr [r10 + r11*8], r12
    mov qword ptr [r10 + r11*8+8], r13
    mov qword ptr [r10], rbx
    mov qword ptr [r10+8], rdx
    mov rbx, r14
    mov rdx, r15
    add r8, 2
    sub r11, 2
    jmp .Lpartition_loop
.Lswap_right:
    mov qword ptr [r10], r14
    mov qword ptr [r10+8], r15
    mov qword ptr [r10 + r11*8], rbx
    mov qword ptr [r10 + r11*8+8], rdx
    mov rbx, r12
    mov rdx, r13
    add r8, 2
    sub r11, 2
    jmp .Lpartition_loop
.Lpartition_done:
    lea rcx, [rdi + r8*8]
    mov qword ptr [rcx], rbx
    mov qword ptr [rcx+8], rdx
    mov rax, r8
    shr rax, 1
    cmp rax, 1
    jl .Lsort_right
    lea rdi, [rdi]
    call qsort_pairs
.Lsort_right:
    mov rdi, qword ptr [rsp+8]
    mov rsi, qword ptr [rsp+16]
    sub rsi, r8
    sub rsi, 2
    je .Ldone
    jle .Ldone
    lea rdi, [rdi + r8*8 + 16]
    call qsort_pairs
.Ldone:
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lbase_case:
    mov rsp, rbp
    mov rax, 0
    pop rbp
    pop rbx
    ret
