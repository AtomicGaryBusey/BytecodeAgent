map_sum:
    push rbx
    push rbp
    push r12
    sub rsp, 8                # align stack to 16-byte boundary (rsp was 8 mod 16, now 0 mod 16)
    xor rax, rax              # accumulator = 0
    test rsi, rsi             # n == 0?
    jz .Ldone                 # if n == 0, return 0
    mov rbx, rdi              # save array pointer in callee-saved reg
    mov rbp, rsi              # save n in callee-saved reg
    mov r12, rdx              # save function pointer f in callee-saved reg
    xor rdi, rdi              # index i = 0

.Lloop:
    cmp rdi, rbp              # compare i with n
    jge .Ldone                # if i >= n, exit loop
    mov rsi, qword ptr [rbx + rdi*8]  # load a[i]
    call r12                  # call f(a[i]), result in rax
    add rbx, rax              # accumulate into rbx (which holds sum so far)
    inc rdi                   # increment index
    jmp .Lloop                # continue loop

.Ldone:
    mov rax, rbx              # move accumulated sum to return register
    add rsp, 8                # restore stack pointer
    pop r12
    pop rbp
    pop rbx
    ret
