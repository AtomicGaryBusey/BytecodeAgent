export function l $count_ugt(l %a, l %n, l %t) {
l %i
w %cond
%start:
	%i = copy l 0
	jmp %loop
%loop:
	%cond = csltl %i, %n
	jnz %cond, %body, %done
%body:
	l %val
	%val = loadl l [%a + %i * 8]
	%cond = culgt %val, %t
	w %inc
	%inc = select w %cond, 1, 0
	%i = add l %i, %inc
	jmp %loop
%done:
	return %i
}
