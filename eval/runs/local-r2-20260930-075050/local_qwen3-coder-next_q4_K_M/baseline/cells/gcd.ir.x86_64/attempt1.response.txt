export function l $gcd(l %a, l %b) {
@start
	l %t0
	w %t1
	jnz %a, @check_b
	lret %b
@check_b
	jnz %b, @loop_entry
	lret %a
@loop_entry
	l %tmp_a = copy %a
	l %tmp_b = copy %b
@loop
	l %rem = modl %tmp_a, %tmp_b
	w %is_zero = ceql %rem, 0
	jnz %is_zero, @done
	l %tmp_a_new = copy %tmp_b
	l %tmp_b_new = copy %rem
	l %tmp_a = copy %tmp_a_new
	l %tmp_b = copy %tmp_b_new
	jmp @loop
@done
	lret %tmp_b
}
