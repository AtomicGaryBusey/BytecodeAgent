export function l $gcd(l %a, l %b) {
@start
	w %t0
	l %res
	jnz %a, @check_b
	lret %b
@check_b
	jnz %b, @loop_entry
	lret %a
@loop_entry
	l %tmp_a = copy %a
	l %tmp_b = copy %b
@loop
	l %rem = modl %tmp_a, %tmp_b
	w %is_zero = ceql %rem, 0
	jnz %is_zero, @done
	l %new_a = copy %tmp_b
	l %new_b = copy %rem
	l %tmp_a = copy %new_a
	l %tmp_b = copy %new_b
	jmp @loop
@done
	lret %tmp_b
}
