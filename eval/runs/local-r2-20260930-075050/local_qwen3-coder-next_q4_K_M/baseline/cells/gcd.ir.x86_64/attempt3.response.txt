export function l $gcd(l %a, l %b) {
@start
	w %a_is_zero
	w %b_is_zero
	l %result
	%a_is_zero = ceql %a, 0
	jnz %a_is_zero, @check_b
	lret %b

@check_b
	%b_is_zero = ceql %b, 0
	jnz %b_is_zero, @done_a
	jmp @loop_entry

@loop_entry
	l %tmp_a = copy %a
	l %tmp_b = copy %b
@loop
	l %rem = modl %tmp_a, %tmp_b
	w %is_zero = ceql %rem, 0
	jnz %is_zero, @done
	l %new_a = copy %tmp_b
	l %new_b = copy %rem
	l %tmp_a = copy %new_a
	l %tmp_b = copy %new_b
	jmp @loop

@done
	lret %tmp_b

@done_a
	lret %a
}
