export function l $gcd(l %a, l %b) {
@start
	w %zero_check_a
	w %zero_check_b
	%zero_check_a = ceql %a, 0
	jnz %zero_check_a, @check_zero_b
	%zero_check_b = ceql %b, 0
	jnz %zero_check_b, @return_a
	jmp @loop_start

@check_zero_b
	w %is_b_zero
	%is_b_zero = ceql %b, 0
	jnz %is_b_zero, @return_a
	lret %a

@return_a
	lret %a

@loop_start
	l %current_a = copy %a
	l %current_b = copy %b
@loop_body
	l %remainder = modl %current_a, %current_b
	w %rem_is_zero
	%rem_is_zero = ceql %remainder, 0
	jnz %rem_is_zero, @done_loop
	l %next_a = copy %current_b
	l %next_b = copy %remainder
	l %current_a = copy %next_a
	l %current_b = copy %next_b
	jmp @loop_body

@done_loop
	lret %current_b
}
