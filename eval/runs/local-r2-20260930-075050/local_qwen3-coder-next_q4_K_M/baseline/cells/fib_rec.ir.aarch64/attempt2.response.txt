export function l $fib_rec(l %n) {
    @entry
    %0 =l ceql %n, 0
    jnz %0, @base0
    %1 =l ceql %n, 1
    jnz %1, @base1
    %2 =l subl %n, 1
    %3 =l call $fib_rec(l %2)
    %4 =l subl %n, 2
    %5 =l call $fib_rec(l %4)
    %6 =l addl %3, %5
    ret %6
    @base0
    ret 0
    @base1
    ret 1
}
