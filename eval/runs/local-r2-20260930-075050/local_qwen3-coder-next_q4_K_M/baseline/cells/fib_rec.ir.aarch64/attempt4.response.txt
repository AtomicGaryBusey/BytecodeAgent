export function l $fib_rec(l %n) {
    @entry
    %is_zero =w ceql %n, 0
    jnz %is_zero, @base0
    %is_one =w ceql %n, 1
    jnz %is_one, @base1
    %n_minus_1 =l subl %n, 1
    %r1 =l call $fib_rec(l %n_minus_1)
    %n_minus_2 =l subl %n, 2
    %r2 =l call $fib_rec(l %n_minus_2)
    %result =l addl %r1, %r2
    ret %result
    @base0
    ret 0
    @base1
    ret 1
}
