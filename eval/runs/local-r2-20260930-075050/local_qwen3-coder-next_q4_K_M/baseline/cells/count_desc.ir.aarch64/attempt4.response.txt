export function l $count_desc(l %a, l %n) {
@start
	l %n_m1 = l.subl %n, 1
	w %cond = w.csltl %n, 2
	jnz %cond, @loop
	jmp @zero

@loop
	l %i = l.cltl 0, 0
	l %count = l.cltl 0, 0
	jmp @loop_test

@loop_test
	w %cond2 = w.csltl %i, %n_m1
	jnz %cond2, @body
	jmp @done_ret

@body
	l %addr_i = l.addl %a, %i
	l %val_i = loadl l %addr_i
	l %addr_next = l.addl %a, 8
	l %val_next = loadl l %addr_next
	w %is_desc = w.csgtl %val_i, %val_next
	l %inc = l.sextw %is_desc
	l %count = l.addl %count, %inc
	l %i = l.addl %i, 8
	l %next_idx = l.addl %i, -7
	jmp @loop_test

@zero
	l %zero = l.cltl 0, 0
	ret %zero

@done_ret
	ret %count
}
