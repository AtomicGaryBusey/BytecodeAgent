export function l $count_desc(l %a, l %n) {
@start
	l %i = l.csltl %n, 2
	w %exit_if_lt_2 = w.ceql %i, 0
	jnz %exit_if_lt_2, @done
	l %count = l.cltl 0, 1
	l %n_minus_1 = l.subl %n, 1
	l %i = l.cltl 0, 1
@loop
	w %cond = w.csltl %i, %n_minus_1
	jnz %cond, @body
	jmp @done
@body
	l %idx_i = l.shll %i, 3
	l %addr_i = l.addl %a, %idx_i
	l %val_i = loadl l %addr_i
	l %idx_next = l.addl %idx_i, 8
	l %addr_next = l.addl %a, %idx_next
	l %val_next = loadl l %addr_next
	w %is_desc = w.csgtl %val_i, %val_next
	l %inc = l.sextw %is_desc
	l %count = l.addl %count, %inc
	l %i = l.addl %i, 1
	jmp @loop
@done
	ret %count
}
