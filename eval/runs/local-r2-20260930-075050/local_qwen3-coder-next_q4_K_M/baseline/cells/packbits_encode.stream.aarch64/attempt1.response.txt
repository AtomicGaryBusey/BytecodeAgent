packbits_encode:
    // Save frame pointer and return address
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Initialize i = 0 (index into src)
    mov x4, xzr           // i = 0
    mov x5, xzr           // out_pos = 0 (current position in output)

    // n in x1, cap in x3, src base in x0, out base in x2

.loop_start:
    // Check if i >= n (end of input)
    cmp x4, x1
    b.ge .Ldone

    // Compute run(i): count consecutive equal bytes from src[i], max 128
    mov x6, x4            // j = i
    ldrb w7, [x0, x4]     // b0 = src[i]
    mov x8, #1            // run_length = 1

.check_run:
    add x9, x6, #1        // j+1
    cmp x9, x1            // if j+1 >= n, stop
    b.ge .Lrun_end
    ldrb w10, [x0, x9]    // src[j+1]
    cmp w7, w10           // src[j+1] == b0?
    b.ne .Lrun_end
    add x8, x8, #1        // run_length++
    mov x6, x9            // j = j+1
    cmp x8, #128          // run limited to 128
    b.lt .check_run

.Lrun_end:
    // Now x8 holds run(i) (capped at 128)

    // If run(i) >= 3, emit repeat packet
    cmp x8, #3
    b.ge .Lemit_repeat

    // Otherwise: gather literal run starting at i, while run(j) < 3 and length < 128
    mov x9, x4            // j = i (start of literal)
    add x10, x4, #1       // end candidate

.Lgather_literal:
    cmp x10, x1           // if j+1 >= n, finalize
    b.ge .Lliteral_end
    ldrb w11, [x0, x10]   // src[j]
    ldrb w12, [x0, x4]    // src[i], same as src[original i]

.compare_with_prev:
    cmp w11, w12          // check if new char equals current run's char
    b.ne .Lliteral_end

.check_next:
    add x13, x10, #1      // j+2 for next run check
    cmp x13, x1           // if out of bounds
    b.ge .Lliteral_end_inc
    ldrb w14, [x0, x13]   // src[j+1]
    cmp w11, w14          // src[j+1] == src[j]?
    beq .Lextend_run      // if equal, we need to check run length

// if not run of 2 or more starting at j
.Lliteral_end_inc:
    add x9, x9, #1        // extend literal by 1 (j+1)
    mov x6, x9            // current run pointer for next iteration
    cmp x10, x1           // j < n?
    b.ge .Lliteral_end    // done
    mov x10, x6           // j = j+1 for next loop iteration
    b .Lgather_literal

.extend_run:
    add x15, x13, #1      // j+2
    cmp x15, x1           // out of bounds?
    b.ge .Lliteral_end_inc
    ldrb w14, [x0, x15]   // src[j+2]
    cmp w11, w14          // run length at least 3?
    beq .Lextend_run      // continue if still equal

.Lliteral_end:
    // literal_len = j - i
    sub x8, x9, x4        // literal_length = j - i
    cbz x8, .loop_start   // skip zero-length literals

    // Check header + data fits in remaining capacity: (1 + literal_length) <= cap - out_pos
    add x6, x8, #1        // needed bytes = 1 + len
    sub x7, x3, x5        // remaining = cap - out_pos
    cmp x6, x7
    b.gt .Lerror          // not enough space

    // Encode literal packet: header = literal_length - 1, then the data
    mov x7, x8            // literal_len
    sub x7, x7, #1        // header = len - 1
    strb w7, [x2, x5]     // out[out_pos] = header
    add x5, x5, #1        // out_pos++

    // Copy the literal bytes
    mov x6, xzr           // offset within literal

.Lliteral_copy_loop:
    cmp x6, x8            // while offset < len
    b.ge .Lliteral_done
    ldrb w7, [x0, x4, x6] // src[i + offset]
    str
