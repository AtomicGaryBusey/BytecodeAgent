collatz_max:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    # rsp is now 8 mod 16 again (was 8 mod 16 at entry)
    mov rbp, rdi                # n in rbp for convenience
    xor rbx, rbx                # max_steps = 0
    xor r12, r12                # current k starting from 1

.Louter_loop:
    cmp r12, rbp
    jge .Ldone                  # if k > n, done
    inc r12                     # k++
    
    mov rax, r12                # start with x = k
    xor r14, r14                # steps = 0

.Linner_loop:
    cmp rax, 1
    je .Lupdate_max             # if x == 1, done with this k
    
    test rax, 1
    jnz .Lodd_case
    
    # even: x /= 2
    shr rax, 1
    inc r14
    jmp .Linner_loop

.Lodd_case:
    lea rcx, [rax + rax*2]      # 3*x = x + 2*x
    inc rcx                     # 3*x + 1
    mov rax, rcx
    inc r14
    jmp .Linner_loop

.Lupdate_max:
    cmp r14, rbx
    jbe .Louter_loop            # if current steps <= max_steps, skip
    
    mov rbx, r14                # update max_steps
    jmp .Louter_loop

.Ldone:
    mov rax, rbx                # return max_steps
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
