mat_pow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // k == 0: return identity matrix and clear tmp
    cmp x2, #0
    b.eq .Lzero_power

    // Copy input matrix a to tmp (first n*n elements)
    mov x8, x1              // n
    lsl x9, x8, #3          // n*8 (row stride in bytes)
    mul x10, x8, x8         // n*n
    lsl x10, x10, #3        // n*n*8 total bytes for matrix

    // Copy a to tmp[0..n*n-1]
    add x11, x4, x10        // tmp + n*n (end of first half)
    mov x9, x0              // src = a
.Lcopy_loop:
    ldr x12, [x9], #8
    str x12, [x4], #8       // store to tmp and advance dst
    cmp x4, x11
    b.lt .Lcopy_loop

    // Reset dst back to start of tmp for multiplication result storage
    sub x4, x4, x10         // x4 = tmp again (dst for product)

    // Initialize out to identity matrix
    mov x9, x3              // out base
    lsl x10, x8, #3         // n*8 (row stride in bytes)
    mul x12, x8, x8         // n*n
    lsl x12, x12, #3        // total bytes = n*n*8
    add x11, x9, x12        // end of out region

.Linit_out:
    str xzr, [x9], #8       // clear element
    cmp x9, x11
    b.lt .Linit_out

    // Set diagonal to 1
    mov x9, x3              // out base
    xor x10, x10, x10       // row index = 0
.L
