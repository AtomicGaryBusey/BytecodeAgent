mat_pow:
    // Save callee-saved registers and set up stack frame
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Load arguments into registers
    ldr x5, [x0, #0]        // Not used directly; just placeholder for clarity
    ldr x6, [x1, #0]
    ldr x7, [x2, #0]

    // Initialize: if k == 0, return identity matrix and clear tmp
    cmp x3, #0
    b.eq .Lzero_power

    // Copy input matrix a to tmp (first n*n elements)
    mov x9, x1              // x9 = n
    lsl x10, x9, #3         // x10 = n * 8 (size in bytes of one row)
    mul x10, x10, x9        // x10 = n*n*8 total bytes for matrix
    add x11, x4, x5         // tmp + 0 (dst) = out + 0? No: out is result, tmp is scratch.
    // Fix: use tmp as dst for copy. x4 = out; x5 = tmp; but pass order is a,n,k,out,tmp
    // So in registers: x0=a, x1=n, x2=k, x3=out, x4=tmp
    add x12, x4, x5         // Not needed

    mov x9, x0              // src = a
    mov x10, x4             // dst = tmp (for base matrix copy)
    lsl x11, x1, #3         // stride in bytes for one row: n*8
    mul x12, x1, x1         // n*n
    lsl x12, x12, #3        // total bytes = n*n*8

    // Copy a to tmp (first half)
    add x13, x4, x5, lsl #6 // tmp + n*n*sizeof(int64_t) = start of second half = tmp2
    mov x14, x0             // src
    mov x15, x4             // dst
.Lcopy_loop:
    ldr x16, [x14], #8
    str x16, [x15], #8
    sub x12, x12, #8
    cbnz x12, .Lcopy_loop

    // Now tmp holds copy of a; out will become result matrix (start as identity for k>0)
    // If k == 0 handled above. For k>=1: start with identity in out.

.Lnonzero_power:
    mov x9, x1              // n
    mul x10, x9, x9         // n*n
    lsl x10, x10, #3        // total bytes for matrix = n*n*8

    // Initialize out to identity matrix: all zeros first, then 1s on diagonal
    mov x0, x3              // ptr to out
    mov x1, #0
    add x2, x0, x10         // end of out region
.Lclear_out:
    str xzr, [x0], #8
    cmp x0, x2
    b.ne .Lclear_out

    // Now set diagonal elements to 1
    mov x9, x3              // out base
    lsl x10, x1, #3         // n*8 (stride for row)
