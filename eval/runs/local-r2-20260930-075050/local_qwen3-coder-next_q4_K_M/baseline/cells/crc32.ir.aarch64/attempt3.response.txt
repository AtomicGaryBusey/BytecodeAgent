export function l $crc32(l %p, l %n) {
@start
	%ptr =l copy %p
	%len =l copy %n
	%crc =w extuw 4294967295
	%i =l copy 0

@outer_loop
	%cmp_outer =w csltl %i, %len
	jnz %cmp_outer @outer_body
	jmp @done

@outer_body
	%byte_ptr =l add %ptr, %i
	%b =w loadub %byte_ptr
	%crc =w xor %crc, %b

	%j =w copy 8

@inner_loop
	%cmp_inner =w ceql %j, 0
	jnz %cmp_inner @inner_done
	%bit_test =w and %crc, 1
	%is_odd =w cneq %bit_test, 0
	jnz %is_odd @handle_odd
@handle_even
	%crc =w shr %crc, 1
	jmp @advance_inner
@handle_odd
	%shifted =w shr %crc, 1
	%xor_val =w extuw 3988292384
	%crc =w xor %shifted, %xor_val
	jmp @advance_inner
@advance_inner
	%j =w sub %j, 1
	jmp @inner_loop

@inner_done
	%i =l add %i, 1
	jmp @outer_loop

@done
	%xor_final =w extuw 4294967295
	%result =w xor %crc, %xor_final
	ret %result
}
