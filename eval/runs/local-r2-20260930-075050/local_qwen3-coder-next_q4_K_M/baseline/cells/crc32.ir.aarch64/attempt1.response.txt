export function l $crc32(l %p, l %n) {
@start
	%0 =l copy %p
	%1 =l copy %n
	%crc_lo =w extuw 4294967295
	%crc_hi =w zeroinit
	%i =l copy 0
@loop_outer
	%cmp_outer =w csltl %i, %1
	jnz %cmp_outer @outer_body
	jmp @done
@outer_body
	%byte_ptr =l add %0, %i
	%b =w loadub %byte_ptr
	%crc_lo =w xor %crc_lo, %b
	%j =w copy 8
@loop_inner
	%cmp_inner =w ceql %j, 0
	jnz %cmp_inner @inner_done
	%bit_test =w and %crc_lo, 1
	%is_odd =w cneq %bit_test, 0
	jnz %is_odd @handle_odd
@handle_even
	%crc_lo =w shr %crc_lo, 1
	jmp @advance_inner
@handle_odd
	%shifted =w shr %crc_lo, 1
	%xor_val =w extuw 2644025344
	%crc_lo =w xor %shifted, %xor_val
	jmp @advance_inner
@advance_inner
	%j =w sub %j, 1
	jmp @loop_inner
@inner_done
	%i =l add %i, 1
	jmp @loop_outer
@done
	%xor_final =w extuw 4294967295
	%result_lo =w xor %crc_lo, %xor_final
	%result_hi =w zeroinit
	%result =l pack %result_lo, %result_hi
	ret %result
}
