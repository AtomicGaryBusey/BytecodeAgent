export function l $count_nonzero_prefix(l %a, l %n) {
	%r = alloca l
	storel %a, %r
	%idx = copy l 0
	%bound = copy l %n
	%ptr = loadl %r
loop:
	celql %idx, %bound
	jnz @done, %1
load_elem:
	%val = loadl %ptr
	celql %val, 0
	jnz @exit, %2
inc_idx:
	addl %idx, 1, %idx
	addl %ptr, 8, %ptr
	jmp @loop
@loop:
	jmp @load_elem
@exit:
	jmp @done
@done:
	copy l %idx, %ret
	retl %ret
}
