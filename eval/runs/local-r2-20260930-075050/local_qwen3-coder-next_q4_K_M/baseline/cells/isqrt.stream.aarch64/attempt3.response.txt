isqrt:
    // Handle special cases: n == 0 => return 0
    cbz     x0, .Lreturn_zero

    // Initialize: low = 0 (x19), high = 2^32 (x20)
    mov     x19, #0          // low = 0
    mov     x20, #1
    lsl     x20, x20, #32   // high = 2^32

.Lbinsearch:
    cmp     x20, x19
    b.le    .Lfound_root

    // mid = low + (high - low) / 2
    sub     x21, x20, x19   // high - low
    asr     x21, x21, #1    // /2
    add     x21, x21, x19   // mid = low + (high-low)/2

    // Check if mid > 2^32-1 (i.e., mid == 2^32)
    asr     x22, x21, #32   // Extract high 32 bits (0 if mid < 2^32, 1 if mid == 2^32)
    cbnz    x22, .Lmid_too_large

    // Compute mid*mid safely (mid < 2^32, so no overflow in 64-bit multiply)
    umull   x22, xzr, x21, x21  // x22 = low 64 bits of mid*mid (umull is safe for 32-bit inputs)

    cmp
