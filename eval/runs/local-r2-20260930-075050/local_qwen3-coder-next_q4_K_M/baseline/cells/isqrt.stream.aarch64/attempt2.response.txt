isqrt:
    // Handle special cases: n == 0 => return 0
    cbz     x0, .Lreturn_zero
    
    // Use binary search on r in [low, high], where low=0, high=1<<32
    mov     x19, #0         // low = 0 (callee-saved)
    mov     x20, #1         // high = 1
    lsl     x20, x20, #32   // high = 2^32
    
.Lbinsearch:
    cmp     x20, x19
    b.le    .Lfound_root
    
    // mid = low + (high - low) / 2
    sub     x21, x20, x19   // high - low
    asr     x21, x21, #1    // /2
    add     x21, x19, x21   // mid = low + (high-low)/2
    
    // Check if mid > 2^32-1 => mid*mid would overflow
    // Since high <= 2^32 and low >=0, mid is at most 2^32
    asr     x22, x21, #32   // high 32 bits (0 if mid < 2^32, 1 if mid == 2^32)
    cbnz    x22, .Lmid_too_large
    
    // mid < 2^32, so compute mid*mid safely in 64-bit
    mul     x22, x21, x21   // low 64 bits of mid*mid (no overflow for mid<2^32)
    
    cmp     x22, x0         // mid*mid <= n?
    b.le    .Llow_update
    
.Lhigh_update:
    mov     x20, x21        // high = mid
    b       .Lbinsearch

.Lmid_too_large:
    // mid == 2^32 => mid*mid > 2^64-1 >= n, so too big
    b       .Lhigh_update

.Llow_update:
    add     x19, x21, #1   // low = mid + 1
    b       .Lbinsearch

.Lfound_root:
    sub     x0, x19, #1     // result = low - 1 (largest r with r*r <= n)
    ret

.Lreturn_zero:
    mov     x0, #0
    ret
