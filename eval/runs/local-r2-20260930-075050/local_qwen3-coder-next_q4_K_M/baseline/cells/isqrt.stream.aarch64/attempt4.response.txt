isqrt:
    cbz     x0, .Lreturn_zero

    mov     x19, #0          // low = 0
    mov     x20, #1
    lsl     x20, x20, #32   // high = 2^32

.Lbinsearch:
    cmp     x20, x19
    b.le    .Lfound_root

    sub     x21, x20, x19   // high - low
    asr     x21, x21, #1    // /2
    add     x21, x21, x19   // mid = low + (high-low)/2

    cmp     x21, #0x100000000
    b
