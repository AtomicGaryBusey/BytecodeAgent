isqrt:
    // Handle special cases: n == 0 => return 0
    cbz     x0, .Lreturn_zero
    
    // Newton's method starting with a reasonable initial guess
    // Initial guess: 2^32 if n >= 2^64 - 2^32, else count leading zeros to estimate
    clz     x1, x0          // x1 = number of leading zeros in n
    sub     x2, x1, #32     // adjustment for 64-bit vs 32-bit half
    neg     x2, x2          // x2 = (64 - clz(n)) / 2 ~ log2(sqrt(n))
    mov     x0, #1          // start with r = 1 << ((64 - clz(n))/2)
    lsl     x0, x0, x2      // initial guess r
    
    // Newton iterations: r' = (r + n/r) / 2
.Lnewton_loop:
    mov     x3, x0          // save current r
    udiv    x4, x0, x1      // x4 = n / r (x1 preserved from clz? no—overwrite needed)
    // Actually, we need to reload n since clz destroyed it.
    // Let's restructure: preserve original n on stack or in callee-saved reg
    
    // Better approach: use binary search over [0, 2^32) since sqrt(2^64-1)=2^32-1
    // This avoids division and overflow issues.
    // We'll do binary search on r in [low, high], where low=0, high=1<<32
    
    mov     x19, #0         // low = 0 (callee-saved)
    mov     x20, #1         // high = 1
    lsl     x20, x20, #32   // high = 2^32 (fits in 64-bit, > max sqrt)
    
.Lbinsearch:
    cmp     x20, x19
    b.le    .Lfound_root    // while low < high
    
    // mid = (low + high) / 2; avoid overflow: mid = low + (high - low)/2
    sub     x21, x20, x19   // high - low
    asr     x21, x21, #1    // (high - low) / 2
    add     x21, x19, x21   // mid = low + (high - low)/2
    
    // Compute r_mid = mid; check if mid*mid <= n
    // But mid*mid may overflow, so compute carefully:
    // If mid > 0xffffffff, then mid*mid > 2^64-1, so too big
    mov     x22, x21
    asr     x23, x21, #32   // high 32 bits of mid (will be zero if mid <= 2^32-1)
    cbnz    x23, .Lmid_too_large  // mid >= 2^32 => too large
    
    // Now mid < 2^32, so mid*mid fits in 64 bits
    mul     x23, x22, x22   // mid * mid (lower 64 bits)
    
    cmp     x23, x0         // compare mid*mid <= n?
    b.le    .Llow_update    // if mid*mid <= n, set low = mid + 1
    b       .Lhigh_update   // else high = mid
    
.Lmid_too_large:
    // mid >= 2^32 => mid*mid > 2^64-1 >= n, so too big
    b       .Lhigh_update

.Llow_update:
    add     x19, x21, #1   // low = mid + 1
    b       .Lbinsearch

.Lhigh_update:
    mov     x20, x21        // high = mid
    b       .Lbinsearch

.Lfound_root:
    // At exit of binary search: high == low, and we want the largest r with r*r <= n.
    // During search, when mid*mid <= n we set low = mid+1, so final low is first r where r*r > n.
    // So answer = low - 1.
    sub     x0, x19, #1     // result = low - 1
    ret

.Lreturn_zero:
    mov     x0, #0
    ret
