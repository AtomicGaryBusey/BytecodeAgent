export function l $fill_down(l %a, l %n, l %start) {
%@main:
	%idx = copy 0
	%count = copy %n
@loop:
	%cond = csltl %idx, %count
	jnz %cond, @body
	jmp @exit
@body:
	%offset = muli %idx, 8
	%addr = addl %a, %offset
	%val = subl %start, %idx
	storel %val, %addr
	%next = addi %idx, 1
	%idx = copy %next
	jmp @loop
@exit:
	copy %count
}
