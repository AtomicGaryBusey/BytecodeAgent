crc32:
    # Save callee-saved registers we'll use
    push rbp
    mov rbp, rsp
    sub rsp, 16                # Allocate local space (8 bytes for crc, 8 bytes padding for alignment)
    push rbx
    push rsi
    push rdi
    push r12
    push r13

    # Initialize: crc = 0xFFFFFFFF
    mov eax, -1                #crc in lower 32 bits of rax (upper 32 bits zeroed by mov to 32-bit reg)
    mov [rbp-8], eax           #save initial crc

    # Handle n == 0 case early
    test rsi, rsi
    jz .Ldone                  #if n==0, skip loop

    # Setup: rdi = p, rsi = n
    mov rbx, rdi               #rbx = pointer to data
    mov rcx, rsi               #rcx = counter (n)
    xor r12, r12               #zero r12 for inner loop counter

.Lloop_outer:
    # Load next byte
    movzx edx, byte ptr [rbx + r12]   #dl = next byte
    xor eax, edx                      #xor byte into crc (eax holds crc)

    # Process 8 bits
    mov r13, 8               #bit counter

.Lloop_inner:
    test al, 1               #check LSB of crc
    jz .Lno_poly
    shr eax, 1               #shift right
    xor eax, 0xEDB88320      #apply polynomial
    jmp .Lnext_bit
.Lno_poly:
    shr eax, 1               #just shift right
.Lnext_bit:
    dec r13                  #decrement bit counter
    jnz .Lloop_inner

    inc r12                  #move to next byte
    cmp r12, rcx           #cmp processed count with total n
    jb .Lloop_outer

.Ldone:
    xor eax, -1             #crc ^= 0xFFFFFFFF
    pop r13                 #restore callee-saved registers
    pop r12
    pop rdi
    pop rsi
    pop rbx
    leave                   #mov rsp, rbp; pop rbp
    ret
