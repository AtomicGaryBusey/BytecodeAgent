lcs_len:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #256
    mov x8, x0          // x = x0
    mov x9, x1          // m = x1
    mov x10, x2         // y = x2
    mov x11, x3         // n = x3
    mov x12, x4         // row = x4

    cbz x11, .Linit_done
    mov x13, #0
    str x13, [x12]      // row[0] = 0
    mov x14, #8
    mul x15, x11, x14   // n * 8 bytes for offset
    add x16, x12, x14   // &row[1]
    add x17, x12, x15   // end = row + n*8

.Linit_loop:
    str x13, [x16], #8
    cmp x16, x17
    b.lt .Linit_loop

.Linit_done:
    mov x5, #0          // i = 0
    cbz x9, .Lend_outer

.Louter_loop:
    ldrb w7, [x8, x5]   // xi = x[i]
    mov x13, #0         // diag = 0
    mov x6, #1          // j = 1

.Linner_loop:
    cmp x6, x11
    b.gt .Lend_inner

    ldr x14, [x12, x6, lsl #3]   // up = row[j]
    add x15, x10, x6
    sub x15, x15, #1             // y index = j-1
    ldrb w17, [x15]              // y[j-1]
    cmp w7, w17
    b.ne .Lnot_equal

    add x14, x13, #1             // diag + 1
    str x14, [x12, x6, lsl #3]
    mov x13, x14
    b .Lupdate_diag

.Lnot_equal:
    ldr x18, [x12, x6, lsl #3]   // row[j]
    ldr x17, [x12, x6, lsl #3, -8]  // row[j-1]
    cmp
