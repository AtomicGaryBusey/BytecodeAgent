shift_sat:
    # rdi = x (int64_t), rsi = n (int64_t)
    # Need to compute (x << n) + (x >> n) with wrapping addition
    # Handle n >= 64 specially: left shift becomes 0, right shift is sign extension

    mov rbx, rdi           # save original x in rbx (callee-saved)
    xor rax, rax           # prepare result = 0
    test rsi, rsi          # check if n == 0
    jz .Ldone              # if n == 0: result = (x << 0) + (x >> 0) = x + x = 2*x

    cmp rsi, 64            # compare n with 64
    jb .Lless64            # if n < 64, go to normal case
    
    # n >= 64: left shift result is 0; right shift is sign extension (x >> 63)
    xor rax, rax           # result = 0 for x << n
    mov rcx, rbx           # copy x
    sar rcx, 63            # arithmetic shift: rcx = 0 if x >= 0, -1 if x < 0
    add rax, rcx           # result += (x >> n)
    jmp .Ldone             # final result in rax

.Lless64:
    # normal case: 0 < n < 64
    mov rcx, rbx           # copy x for left shift
    shl rcx, rsi           # rcx = x << n (logical left shift)
    
    mov
