shift_sat:
    # rdi = x (int64_t), rsi = n (int64_t)
    # Compute (x << n) + (x >> n) with wrapping addition
    
    push rbx               # save callee-saved register
    mov rbx, rdi           # save original x in rbx
    
    xor rax, rax           # result = 0
    test rsi, rsi          # check if n == 0
    jz .Lzero_shift        # if n == 0: result = (x << 0) + (x >> 0) = x + x = 2*x
    
    cmp rsi, 64            # compare n with 64
    jb .Lless64            # if n < 64, go to normal case
    
    # n >= 64: left shift result is 0; right shift is sign extension (x >> 63)
    xor rax, rax           # result = 0 for x << n
    mov rcx, rbx           # copy x
    sar rcx, 63            # arithmetic shift: rcx = 0 if x >= 0, -1 if x < 0
    add rax, rcx           # result += (x >> n)
    jmp .Ldone             # final result in rax

.Lzero_shift:
    # n == 0: return x + x with wrapping
    mov rax, rbx
    add rax, rbx
    jmp .Ldone

.Lless64:
    # normal case: 0 < n < 64
    # Use shlx for left shift (logical) and sarx for right shift (arithmetic)
    xor rcx, rcx           # prepare temporary
    mov rdx, rbx           # copy x
    
    shlx rax, rbx, rsi     # rax = x << n (logical left shift by n bits)
    sarx rcx, rbx, rsi     # rcx = x >> n (arithmetic right shift by n bits)
    
    add rax, rcx           # add with wrapping: result = rax + rcx

.Ldone:
    pop rbx                # restore callee-saved register
    ret
