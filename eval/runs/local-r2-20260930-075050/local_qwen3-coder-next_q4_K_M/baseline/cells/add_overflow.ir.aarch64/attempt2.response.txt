export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum = add l %a, %b
	storel %sum, %out
	%pos_a = csltl zext l %a, 0
	%neg_b = csltl zext l %b, 0
	%pos_sum = csltl zext l %sum, 0
	%ovf_pos = and w %pos_a, %neg_b
	%ovf_neg = and w %pos_sum, %neg_b
	%ovf_w = or w %ovf_pos, %ovf_neg
	ret %ovf_w
}
