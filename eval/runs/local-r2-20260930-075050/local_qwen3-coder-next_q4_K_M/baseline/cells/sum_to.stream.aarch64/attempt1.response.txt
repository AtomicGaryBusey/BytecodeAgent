sum_to:
    // Check if n <= 0
    cmp x0, #0
    ble .Lreturn_zero
    
    // Use arithmetic series formula: sum = n * (n + 1) / 2
    add x1, x0, #1          // x1 = n + 1
    mul x0, x0, x1          // x0 = n * (n + 1)
    asr x0, x0, #1          // x0 = sum (divide by 2 using arithmetic shift)
    ret

.Lreturn_zero:
    mov x0, #0
    ret
