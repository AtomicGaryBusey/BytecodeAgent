packbits_decode:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    str x23, [sp, #40]
    eor x23, x23, x23    // written = 0 (output index)
    eor x24, x24, x24    // i = 0 (source index)

.Lloop:
    cmp x24, x1
    b.ge .Ldone

    ldrb w9, [x0, x24]
    add x24, x24, #1     // advance past header

    cmp w9, #128
    b.ge .Lrepeat_packet  // h >= 128 means negative or -128 in signed byte interpretation

    // Literal packet: copy h+1 bytes
    add x25, x9, #1      // count = h + 1

    sub x26, x1, x24     // remaining source bytes after header
    cmp x25, x26
    b.gt .Lerror_src_short

    sub x27, x3, x23     // remaining output capacity
    cmp x25, x27
    ble .Lfull_literal_copy

.Lwrite_partial_literal:
    mov x9, x27          // write only up to capacity
    add x10, x0, x24     // src_ptr = src + i
    add x11, x2, x23     // out + written
    add x12, x10, x9     // src_end for bounds check (to verify we don't read past n)
    cmp x9, x26
    b.gt .Lerror_src_short

.Lliteral_partial_copy:
    cbz x9, .Lpartial_literal_done
    add x13, x10, x9     // end src pointer for partial copy
.Lpartial_literal_loop:
    ldrb w8, [x10], #1
    strb w8, [x11], #1
    add x23, x23, #1
    cmp x10, x13
    b.lt .Lpartial_literal_loop

.Lpartial_literal_done:
    b .Lerror_cap_exceeded

.Lfull_literal_copy:
    cbz x25, .Lliteral_done
    add x9, x0, x24      // src_ptr = src + i
    add x10, x2, x23     // out + written
    mov x11, x25         // count
.Lliteral_copy_loop:
    ldrb w8, [x9], #1
    strb w8, [x10], #1
    sub x11, x11, #1
    b.ne .Lliteral_copy_loop

    add x23, x23, x25    // written += count
    add x24, x24, x25    // i += count
.Lliteral_done:
    cmp x23, x3
    b.gt .Lerror_cap_exceeded  // if exceeded, next iteration will fail cap check but we must handle here
    b .Lloop

.Lrepeat_packet:
    cmp w9, #128
    beq .Lskip_nop       // h == -128 (0x80 as signed) -> no op

    sub x25, x9, #256    // signed_h = w9 - 256 (convert unsigned to signed)
    add x25, x25, #1     // count = 1 - signed_h = 1 - (w9 - 256) = 257 - w9

.Lrepeat_prep:
    cmp x24, x1
    b.gt .Lerror_src_short
    ldrb w10, [x0, x24]
    add x24, x24, #1     // advance past data byte

    sub x26, x3, x23     // remaining output capacity
    cmp x25, x26
    ble .Lfull_repeat_copy

.Lwrite_partial_repeat:
    mov x9, x26          // write only up to capacity
    b .Lpartial_repeat_copy

.Lfull_repeat_copy:
    mov x9, x25

.Lcheck_cap_after_repeat:
    add x23, x23, x9     // advance written first (will be capped by cap in next loop check)
    cmp x23, x3
    b.gt .Lerror_cap_exceeded  // if we'd exceed capacity, write partial instead
    sub x23, x23, x9     // revert, use partial path

.Lpartial_repeat_copy:
    cbz x9, .Lrepeat_done
    add x10, x2, x23     // dst_ptr = out + written
.Lrepeat_write_loop:
    strb w10, [x10], #1
    sub
