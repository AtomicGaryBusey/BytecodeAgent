packbits_decode:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #28]
    mov x19, x0          // src
    mov x20, x1          // n
    mov x21, x2          // out
    mov x22, x3          // cap
    eor x23, x23, x23    // written = 0 (output pointer)
    eor x24, x24, x24    // i = 0 (source index)

.Lloop:
    cmp x24, x20
    b.ge .Ldone

    ldrb w9, [x19, x24]
    add x24, x24, #1     // advance past header

    cmp w9, #128
    b.ge .Lrepeat_packet  // h >= 128 (only -128 remains in signed interpretation)

    // Literal packet: copy h+1 bytes
    add x25, x9, #1      // count = h + 1
    cmp x24, x20
    b.gt .Lerror_src_short

    sub x26, x22, x23    // remaining output capacity
    cmp x25, x26
    ble .Lfull_literal_copy

.Lwrite_partial_literal:
    mov x9, x26          // write only up to capacity
    add x10, x19, x24    // src_ptr = src + i
    add x11, x21, x23    // out + written
    add x12, x10, x9     // src_end for bounds check (to verify we don't read past n)
    sub x14, x20, x24    // remaining source bytes after header
    cmp x9, x14
    b.gt .Lerror_src_short

.Lliteral_partial_copy:
    add x13, x10, x9     // end src pointer for partial copy
.Lpartial_literal_loop:
    cmp x10, x13
    b.ge .Lpartial_literal_done
    ldrb w8, [x10], #1
    strb w8, [x21, x23]
    add x23, x23, #1
    b .Lpartial_literal_loop

.Lpartial_literal_done:
    b .Lerror_cap_exceeded

.Lfull_literal_copy:
    add x9, x19, x24     // src_ptr = src + i
    mov x10, x23         // dst_ptr = written
    add x11, x21, x10    // out + written
    bl .Lmemcpy_part
    add x23, x23, x25    // written += count
    add x24, x24, x25    // i += count
    cmp x23, x22
    b.ge .Lerror_cap_exceeded
    b .Lloop

.Lrepeat_packet:
    cmp w9, #128
    beq .Lskip_nop       // h == -128 (0x80 as signed) -> no op
    neg x25, x9          // count = 1 - h (since h is negative, e.g., -1 -> 2)
    add x25, x25, #1     // actually: count = 1 - h, but h is signed; we need careful math

.Lrepeat_prep:
    cmp x24, x20
    b.gt .Lerror_src_short
    ldrb w10, [x19, x24]
    add x24, x24, #1     // advance past data byte

    sub x26, x22, x23    // remaining output capacity
    cmp x25, x26
    ble .Lfull_repeat_copy

.Lwrite_partial_repeat:
    mov x9, x26          // write only up to capacity
    b .Lpartial_repeat_copy

.Lfull_repeat_copy:
    mov x9, x25
    add x23, x23, x9     // advance written first (will be capped by cap in next loop check)

.Lcheck_cap_after_repeat:
    cmp x23, x22
    b.gt .Lerror_cap_exceeded  // this shouldn't happen with proper logic but safety first

.Lpartial_repeat_copy:
    // write byte w10 for count = x9 times into out + written (x23)
    add x10, x21, x23    // dst_ptr
    cbz x9, .Lrepeat_done
.Lrepeat_write_loop:
    strb w10, [x10], #1
    sub
