packbits_decode:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    mov x19, x0          // src
    mov x20, x1          // n
    mov x21, x2          // out
    mov x22, x3          // cap
    eor x23, x23, x23    // written = 0 (output pointer)
    eor x24, x24, x24    // i = 0 (source index)

.Lloop:
    cmp x24, x20
    b.ge .Ldone

    ldrb w9, [x19, x24]
    add x24, x24, #1     // advance past header

    cmp w9, #128
    ccmp w9, #0, #4, ge  // if h >= 0 and h <= 127 (literal packet)
    b.gt .Lrepeat_packet

    // Literal packet: copy h+1 bytes
    add x25, x9, #1      // count = h + 1
    cmp x24, x20
    ble .Lcheck_src_len_ok_literal
    b .Lerror_src_short

.Lcheck_src_len_ok_literal:
    sub x26, x22, x23    // remaining output capacity
    cmp x25, x26
    cset x9, lt          // x9 = (count <= remaining) ? 1 : 0
    cbz x9, .Lwrite_partial_literal

    // Full literal copy: count bytes from src to out
    add x9, x19, x24     // src_ptr = src + i
    mov x10, x23         // dst_ptr = written
    add x11, x21, x10    // out + written
    bl .Lmemcpy

    add x23, x23, x25    // written += count
    add x24, x24, x25    // i += count
    cmp x23, x22
    b.ge .Lerror_cap_exceeded
    b .Lloop

.Lwrite_partial_literal:
    // Write min(count, remaining) bytes to fill cap
    mov x9, x26          // write only up to capacity
    add x10, x19, x24    // src_ptr = src + i
    add x11, x21, x23    // out + written
    add x12, x10, x9     // src_end for bounds check (to verify we don't read past n)
    cmp x12, x19
    blo .Lerror_src_short  // if src pointer wraps (impossible for valid data but safe)
    sub x14, x20, x24    // remaining source bytes after header
    cmp x9, x14
    b.gt .Lerror_src_short

.Lliteral_partial_copy:
    add x13, x10, x9     // end src pointer for partial copy
.Lpartial_literal_loop:
    cmp x10, x13
    b.ge .Lpartial_literal_done
    ldrb w8, [x10], #1
    strb w8, [x21, x23]
    add x23, x23, #1
    b .Lpartial_literal_loop

.Lpartial_literal_done:
    b .Lerror_cap_exceeded

.Lrepeat_packet:
    // h is negative: repeat packet
    cmp w9, #0xc0        // compare with -64 (0xffffffc0) to detect -128 specially
    ccmp w9, #0x80, #4, ne  // if h != -128
    b
