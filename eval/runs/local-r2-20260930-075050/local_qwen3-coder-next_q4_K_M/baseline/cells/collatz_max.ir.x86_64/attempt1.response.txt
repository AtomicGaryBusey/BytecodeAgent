export function l $collatz_max(l %n) {
.entry:
	l %max_steps
	l %k
	l %current
	w %steps
	w %is_one
	w %cond

	/* Initialize max_steps = 0 */
	storel %max_steps, 0
	/* k = 1 */
	storel %k, 1
	jmp @loop_check

@loop_check:
	/* if (k > n) goto @done */
	l %k_val
	loadl %k_val, %k
	csltl w %cond, %k_val, %n
	w %not_done
	notw %not_done, %cond
	jnz @done, %not_done

	/* Compute Collatz steps for current k */
	l %curr_val
	loadl %curr_val, %k
	storel %current, %curr_val
	steps %steps, 0

@collatz_loop:
	/* if (current == 1) goto @done_current */
	l %curr_check
	loadl %curr_check, %current
	ceql w %is_one, %curr_check, 1
	jnz @done_current, %is_one

	/* steps++ */
	w %steps_inc
	addw %steps_inc, %steps, 1
	storew %steps, %steps_inc

	/* if (current is even) current /= 2 else current = 3*current + 1 */
	l %curr_even_check
	loadl %curr_even_check, %current
	/* check parity: current & 1 */
	w %parity
	andw %parity, %curr_even_check, 1
	w %is_even
	ceql %is_even, %parity, 0
	jnz @even_branch, %is_even

@odd_branch:
	l %curr_odd
	loadl %curr_odd, %current
	/* current = 3 * curr + 1 */
	l %three_times
	imul64 %three_times, %curr_odd, 3
	l %next_val
	addl %next_val, %three_times, 1
	storel %current, %next_val
	jmp @collatz_loop

@even_branch:
	l %curr_even
	loadl %curr_even, %current
	/* current /= 2 */
	l %half
	sar64 %half, %curr_even, 1 /* arithmetic shift for signed division by 2 */
	storel %current, %half
	jmp @collatz_loop

@done_current:
	/* if (steps > max_steps) max_steps = steps */
	w %max_s
	loadw %max_s, %max_steps
	csltw w %greater, %steps, %max_s
	jnz @update_max, %greater
	jmp @next_k

@update_max:
	storel %max_steps, %steps

@next_k:
	/* k++ */
	l %k_val2
	loadl %k_val2, %k
	l %k_next
	addl %k_next, %k_val2, 1
	storel %k, %k_next
	jmp @loop_check

@done:
	/* return max_steps */
	l %final_max
	loadl %final_max, %max_steps
	ret %final_max
}
