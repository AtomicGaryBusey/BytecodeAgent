export function l $collatz_max(l %n) {
@entry:
	l %max_steps
	l %k
	l %current
	w %steps

	storel %max_steps, 0
	storel %k, 1
	jmp @loop_check

@loop_check:
	l %k_val
	loadl %k_val, %k
	csltl w %cond, %k_val, %n
	w %not_done
	notw %not_done, %cond
	jnz @done, %not_done

	l %curr_val
	loadl %curr_val, %k
	storel %current, %curr_val
	storew %steps, 0

@collatz_loop:
	l %curr_check
	loadl %curr_check, %current
	ceql w %is_one, %curr_check, 1
	jnz @done_current, %is_one

	w %steps_inc
	addw %steps_inc, %steps, 1
	storew %steps, %steps_inc

	l %curr_parity
	loadl %curr_parity, %current
	andw w %parity, %curr_parity, 1
	ceql w %is_even, %parity, 0
	jnz @even_branch, %is_even

@odd_branch:
	l %curr_odd
	loadl %curr_odd, %current
	imul64 l %three_times, %curr_odd, 3
	addl l %next_val, %three_times, 1
	storel %current, %next_val
	jmp @collatz_loop

@even_branch:
	l %curr_even
	loadl %curr_even, %current
	shrw w %half, %curr_even, 1
	storel %current, %half
	jmp @collatz_loop

@done_current:
	w %max_s
	loadw %max_s, %max_steps
	csltw w %greater, %steps, %max_s
	jnz @update_max, %greater
	jmp @next_k

@update_max:
	storel %max_steps, %steps

@next_k:
	l %k_val2
	loadl %k_val2, %k
	addl l %k_next, %k_val2, 1
	storel %k, %k_next
	jmp @loop_check

@done:
	l %final_max
	loadl %final_max, %max_steps
	ret %final_max
}
