export function l $collatz_max(l %n) {
@entry:
	l %max_steps
	l %k
	storel %n, 8
	jmp @loop_check

@loop_check:
	l %k_val
	loadl %k_val, 8
	csltl w %cond, %k_val, 1
	w %done
	notw %done, %cond
	jnz @exit, %done
	storel %k_val, 0

	l %current
	l %steps_w
	storel %k_val, 16
	storew %steps_w, 0

@collatz_loop:
	l %curr_check
	loadl %curr_check, 16
	ceql w %is_one, %curr_check, 1
	jnz @done_current, %is_one

	w %steps_inc
	addw %steps_inc, %steps_w, 1
	storew %steps_inc, 0

	l %curr_parity
	loadl %curr_parity, 16
	andw w %parity, %curr_parity, 1
	ceql w %is_even, %parity, 0
	jnz @even_branch, %is_even

@odd_branch:
	l %curr_odd
	loadl %curr_odd, 16
	imul64 l %three_times, %curr_odd, 3
	addl l %next_val, %three_times, 1
	storel %next_val, 16
	jmp @collatz_loop

@even_branch:
	l %curr_even
	loadl %curr_even, 16
	shrw w %half, %curr_even, 1
	storel %half, 16
	jmp @collatz_loop

@done_current:
	w %max_s
	loadw %max_s, %max_steps
	csltw w %greater, %steps_w, %max_s
	jnz @update_max, %greater
	jmp @next_k

@update_max:
	storew %steps_w, %max_steps

@next_k:
	l %k_val2
	loadl %k_val2, 0
	addl l %k_next, %k_val2, 1
	storel %k_next, 8
	jmp @loop_check

@exit:
	l %final_max
	loadl %final_max, %max_steps
	ret %final_max
}
