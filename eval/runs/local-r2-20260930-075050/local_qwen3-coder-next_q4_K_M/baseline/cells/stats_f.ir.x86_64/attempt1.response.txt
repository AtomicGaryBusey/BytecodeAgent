export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%ymin =l copy %a
	%ymax =l copy %a
	%sum =l copy 0
	%count_odd =l copy 0
	%i =l copy 0

@loop_start
	%csltl =csltl %i, %n
	jnz %csltl, @body
	jmp @done

@body
	%idx =lmull 8, %i
	%a_idx_ptr =addl %a, %idx
	%y_val =l loadl %a_idx_ptr
	%r =l call %f(%y_val)
	%odd_test =andl %r, 1
	%is_odd =ceql %odd_test, 1
	jnz %is_odd, @increment_odd
	jmp @update_min_max

@increment_odd
	%count_odd =addl %count_odd, 1
	jmp @update_min_max

@update_min_max
	%csltl_ymin =csltl %r, %ymin
	jnz %csltl_ymin, @set_ymin
	jmp @check_ymax

@set_ymin
	%ymin =copy %r
	jmp @check_ymax

@check_ymax
	%csgtl_ymax =csgtl %r, %ymax
	jnz %csgtl_ymax, @set_ymax
	jmp @update_sum

@set_ymax
	%ymax =copy %r
	jmp @update_sum

@update_sum
	%sum =addl %sum, %r
	%i =addl %i, 1
	jmp @loop_start

@done
	%out_ptr0 =addl %out, 0
	%out_ptr1 =addl %out, 8
	%out_ptr2 =addl %out, 16
	storel %ymin, %out_ptr0
	storel %ymax, %out_ptr1
	storel %sum, %out_ptr2
	ret %count_odd
}
