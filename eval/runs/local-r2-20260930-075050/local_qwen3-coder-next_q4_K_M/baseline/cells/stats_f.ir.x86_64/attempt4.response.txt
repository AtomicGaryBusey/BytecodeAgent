export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%count_odd =l copy 0
	%sum =l copy 0
	%ymin =l loadl %a
	%ymax =l copy %ymin
	%i =l copy 0

@loop_start
	csltl %i, %n
	jz @done_loop

@body
	%idx =lmull 8, %i
	%a_ptr =addl %a, %idx
	%val =l loadl %a_ptr
	%y_val =l call %f(%val)

	%is_odd_test =andl %y_val, 1
	%is_odd =ceql %is_odd_test, 1
	jnz %is_odd, @odd_inc

	jmp @minmax_update

@odd_inc
	%count_odd =addl %count_odd, 1
	jmp @minmax_update

@minmax_update
	%new_ymin =csltl %y_val, %ymin
	jnz %new_ymin, @update_min
	jmp @check_max

@update_min
	%ymin =copy %y_val
	jmp @check_max

@check_max
	%new_ymax =csgtl %y_val, %ymax
	jnz %new_ymax, @update_max
	jmp @sum_update

@update_max
	%ymax =copy %y_val
	jmp @sum_update

@sum_update
	%sum =addl %sum, %y_val
	%i =addl %i, 1
	jmp @loop_start

@done_loop
	storel %ymin, %out
	%out1 =addl %out, 8
	storel %ymax, %out1
	%out2 =addl %out, 16
	storel %sum, %out2
	ret %count_odd
}
