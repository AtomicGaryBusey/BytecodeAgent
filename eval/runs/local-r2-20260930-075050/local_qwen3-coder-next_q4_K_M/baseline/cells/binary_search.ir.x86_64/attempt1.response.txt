export function l $binary_search(l %a, l %n, l %key) {
@start
	%0 = copy l %n
	%1 = copy l 0
	%2 = copy l -1
	jnz l %0, @loop_entry, @not_found
@loop_entry
	csltl %3 = %1, %0
	jz w %3, @not_found
	%4 = addl %1, %0
	%5 = shrl %4, 1
	%6 = copy l %a
	%7 = sxtw l %5
	%8 = mull %7, 8
	%9 = addl %6, %8
	%10 = loadl l %9
	ceql %11 = %10, %key
	jnz w %11, @found
	csltl %12 = %10, %key
	jnz w %12, @greater
	%13 = copy l %5
	%14 = subl %13, 1
	%15 = copy l %1
	%16 = copy l %14
	jmp @loop_entry
@greater
	%17 = copy l %5
	%18 = addl %17, 1
	%19 = copy l %18
	%20 = copy l %0
	jmp @loop_entry
@found
	%21 = copy l %5
	ret l %21
@not_found
	ret l %2
}
