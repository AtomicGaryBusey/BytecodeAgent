export function l $binary_search(l %a, l %n, l %key) {
@start
	%0 = copy l %n
	%1 = copy l 0
	%2 = copy l -1
	csltl %3 = %1, %0
	jz w %3, @not_found
@loop_entry
	%4 = copyl %1
	%5 = copyl %0
	csltl %6 = %4, %5
	jz w %6, @not_found
	%7 = addl %4, %5
	%8 = shrl %7, 1
	%9 = sxtw l %8
	%10 = mull %9, 8
	%11 = addl %a, %10
	%12 = loadl l %11
	ceql %13 = %12, %key
	jnz w %13, @found
	csltl %14 = %12, %key
	jnz w %14, @right_half
	%15 = copyl %8
	%16 = subl %15, 1
	%17 = copyl %1
	%18 = copyl %16
	%19 = copyl %0
	jmp @loop_entry
@right_half
	%20 = addl %8, 1
	%21 = copyl %20
	%22 = copyl %1
	%23 = copyl %5
	%1 = copyl %21
	%0 = copyl %23
	jmp @loop_entry
@found
	%24 = copyl %8
	ret l %24
@not_found
	ret l %2
}
