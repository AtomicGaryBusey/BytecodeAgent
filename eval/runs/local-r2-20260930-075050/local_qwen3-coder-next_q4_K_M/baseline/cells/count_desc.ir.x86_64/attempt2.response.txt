export function l $count_desc(l %a, l %n) {
l %i
l %count
w %cond
w %cond2
@start
    %count = 0l
    %i = 0l
    %cond2 = csltl %n, 2l
    jnz %cond2, @done
@loop
    %cond = clt l %i, sub(l %n, 1l)
    jnz %cond, @body
    jmp @done
@body
    %val1 = loadl l [%a + mul(l %i, 8l)]
    %val2 = loadl l [%a + mul(add(l %i, 1l), 8l)]
    %cond = csgtl %val1, %val2
    %count = add(l %count, zext(w %cond))
    %i = add(l %i, 1l)
    jmp @loop
@done
    ret l %count
}
