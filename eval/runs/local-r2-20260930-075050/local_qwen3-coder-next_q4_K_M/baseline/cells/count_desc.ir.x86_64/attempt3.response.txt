export function l $count_desc(l %a, l %n) {
l %i
l %count
w %cond
@start
    %count = 0l
    %i = 0l
    %cond = csltl %n, 2l
    jnz %cond, @done
@loop
    %cond = clt l %i, sub(l %n, 1l)
    jnz %cond, @body
    jmp @done
@body
    %ptr1 = add(l %a, mul(l %i, 8l))
    %val1 = loadl l [%ptr1]
    %ptr2 = add(l %a, mul(add(l %i, 1l), 8l))
    %val2 = loadl l [%ptr2]
    %cond = csgtl l %val1, l %val2
    %count = add(l %count, zext(w %cond))
    %i = add(l %i, 1l)
    jmp @loop
@done
    ret l %count
}
