export function l $count_desc(l %a, l %n) {
l %i
l %count
w %cond
w %cond2
l %val1
l %val2
@start
    %count = 0
    %i = 0
    %cond2 = csltl %n, 2
    jnz %cond2, @done
@loop
    %cond = clt l %i, sub(l %n, 1)
    jnz %cond, @body
    jmp @done
@body
    %val1 = loadl l [%a + mul(l %i, 8)]
    %val2 = loadl l [%a + mul(add(l %i, 1), 8)]
    %cond = csgtl %val1, %val2
    %count = add(l %count, %cond)
    %i = add(l %i, 1)
    jmp @loop
@done
    ret l %count
}
