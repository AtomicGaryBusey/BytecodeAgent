export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	l %i = zext w 0
	w %cond0

@loop
	%cond0 = csltl %i, %n
	jnz %cond0, @body, @end

@body
	l %val = loadl l %src + i64 %i * 8
	storel l %dst + i64 %i * 8, %val
	l %i_next = add l %i, 1
	%i = %i_next
	jmp @loop

@end
	l %ret = zext w 0
	ret l %ret
}
