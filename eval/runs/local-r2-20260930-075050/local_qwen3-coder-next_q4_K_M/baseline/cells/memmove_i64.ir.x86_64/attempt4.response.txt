export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	l %i = zext w 0
	l %zero = zext w 0

@loop
	w %cond = csltl %i, %n
	jnz %cond, @body, @end

@body
	l %idx = mul l %i, 8
	l %src_addr = add l %src, %idx
	l %val = loadl l %src_addr
	l %dst_addr = add l %dst, %idx
	storel l %dst_addr, %val
	l %i_next = add l %i, 1
	%i = %i_next
	jmp @loop

@end
	ret l %zero
}
