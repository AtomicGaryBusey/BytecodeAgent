export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	l %i = zext w 0
	l %tmp0
	l %tmp1
	w %cond0
	w %done

@loop
	%cond0 = csltl %i, %n
	jnz %cond0, @body, @end

@body
	%tmp0 = loadl l %src + i64 %i * 8
	storel l %dst + i64 %i * 8, %tmp0
	%i = add l %i, 1
	jmp @loop

@end
	l %ret = zext w 0
	ret l %ret
}
