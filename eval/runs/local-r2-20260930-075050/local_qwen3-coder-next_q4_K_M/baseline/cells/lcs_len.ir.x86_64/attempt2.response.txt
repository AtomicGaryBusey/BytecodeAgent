export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
	%0 =w ceql %m, 0
	jnz %0, @return_zero
	%1 =w ceql %n, 0
	jnz %1, @return_zero

	// Initialize row[0..n] to 0
	%i_init = l 0
@init_loop:
	%2 =w csltl %i_init, %n
	jz %2, @init_done
	%3 = l addl %row, %i_init
	storel 0, %3
	%i_init = l addl %i_init, 1
	jnz 1, @init_loop

@init_done:
	%outer_i = l 0
	%x_ptr = l %x
@outer_loop:
	%4 =w csltl %outer_i, %m
	jz %4, @outer_done
	%diag = l 0
	%inner_j = l 1
@inner_loop:
	%5 =w csltl %inner_j, %n
	jz %5, @inner_done

	// row[j] pointer and load up = row[j]
	%row_j_ptr = l addl %row, %inner_j
	%up = l loadl %row_j_ptr

	// row[j-1] pointer and load left = row[j-1]
	%row_jm1_ptr = l subl %row_j_ptr, 8
	%left = l loadl %row_jm1_ptr

	// Load x[i]
	%x_index = l addl %outer_i, 0
	%x_byte_ptr = l addl %x_ptr, %x_index
	%x_val =w loadub %x_byte_ptr

	// Load y[j-1]
	%y_index = l subl %inner_j, 1
	%y_ptr_offset = l addl %y, %y_index
	%y_val =w loadub %y_ptr_offset

	// Compare x[i] and y[j-1]
	%equal =w ceql %x_val, %y_val
	jnz %equal, @match_case

	// mismatch: new_val = max(up, left)
	%new_val = l maxil %up, %left
	storel %new_val, %row_j_ptr
	jnz 1, @diag_update

@match_case:
	%new_val = l addl %diag, 1
	storel %new_val, %row_j_ptr

@diag_update:
	%diag = l %up
	%inner_j = l addl %inner_j, 1
	jnz 1, @inner_loop

@inner_done:
	%outer_i = l addl %outer_i, 1
	jnz 1, @outer_loop

@outer_done:
	%result_ptr = l addl %row, %n
	%result = l loadl %result_ptr
	ret %result

@return_zero:
	ret 0
}
