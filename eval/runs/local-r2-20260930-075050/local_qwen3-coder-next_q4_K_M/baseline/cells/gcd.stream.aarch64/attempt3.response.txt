gcd:
    // Handle cases where either input is zero
    cbz x0, .Lreturn_b
    cbz x1, .Lreturn_a

.Lloop:
    // Compute b % a using udiv and msub (Euclidean algorithm)
    udiv x2, x1, x0     // x2 = b / a
    msub x3, x2, x0, x1 // x3 = b - (b/a)*a = b % a
    mov x1, x0          // b = a
    mov x0, x3          // a = b % a
    cbz x0, .Ldone      // if a == 0, done
    b .Lloop

.Ldone:
    mov x0, x1          // return previous value (which is gcd)
    ret

.Lreturn_a:
    ret

.Lreturn_b:
    mov x0, x1          // return x1 (b)
    ret
