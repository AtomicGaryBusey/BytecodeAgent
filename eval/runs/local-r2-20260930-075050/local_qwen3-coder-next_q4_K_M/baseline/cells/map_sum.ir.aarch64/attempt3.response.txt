export function l $map_sum(l %a, l %n, l %f) {
@start
	%count =l copy %n
	%sum =l copy 0
	%i =l copy 0
@loop_cond
	%cond =csltl %i, %count
	jnz %cond, @loop_body
	jmp @done
@loop_body
	%idx =l mul %i, 8
	%addr =l add %a, %idx
	%x =l loadl %addr
	%result =l call %f(l %x)
	%sum =l add %sum, %result
	%i =l add %i, 1
	jmp @loop_cond
@done
	ret %sum
}
