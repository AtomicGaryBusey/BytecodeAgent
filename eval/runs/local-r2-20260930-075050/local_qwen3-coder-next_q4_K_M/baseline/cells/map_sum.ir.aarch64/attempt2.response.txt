export function l $map_sum(l %a, l %n, l %f) {
@start
	%0 =l copy %n
	%sum =l copy 0
	%i =l copy 0
@loop_cond
	%cond =csltl %i, %0
	jnz %cond, @loop_body
	jmp @done
@loop_body
	%idx =l mul %i, 8
	%addr =l add %a, %idx
	%x =l loadl %addr
	%r =l call %f(l %x)
	%sum =l add %sum, %r
	%i =l add %i, 1
	jmp @loop_cond
@done
	ret %sum
}
