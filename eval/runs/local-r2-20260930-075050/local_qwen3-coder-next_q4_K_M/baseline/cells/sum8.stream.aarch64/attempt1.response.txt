sum8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Load all arguments into registers for computation
    // a in x0, b in x1, c in x2, d in x3, e in x4, f in x5, g in x6, h in x7

    // Compute terms:
    // term1 = a (x0)
    // term2 = -2*b (x8)
    // term3 = 3*c (x9)
    // term4 = -4*d (x10)
    // term5 = 5*e (x11)
    // term6 = -6*f (x12)
    // term7 = 7*g (x13)
    // term8 = -8*h (x14)

    mov x8, x1
    lsl x8, x8, #1      // 2*b
    neg x8, x8          // -2*b

    mov x9, x2
    add x9, x9, x9, lsl #1   // 3*c = c + 2*c

    mov x10, x3
    lsl x10, x10, #2         // 4*d
    neg x10, x10             // -4*d

    mov x11, x4
    add x11, x11, x11, lsl #2   // 5*e = e + 4*e

    mov x12, x5
    add x12, x12, x12, lsl #1   // 3*f
    add x12, x12, x12           // 6*f
    neg x12, x12                // -6*f

    mov x13, x6
    add x13, x13, x13, lsl #2   // 5*g
    add x13, x13, x13, lsl #1   // 7*g = 5*g + 2*g

    mov x14, x7
    lsl x14, x14, #3          // 8*h
    neg x14, x14              // -8*h

    // Sum all terms: x0 (a) + x8..x14
    add x0, x0, x8            // a + (-2*b)
    add x0, x0, x9            // + 3*c
    add x0, x0, x10           // + (-4*d)
    add x0, x0, x11           // + 5*e
    add x0, x0, x12           // + (-6*f)
    add x0, x0, x13           // + 7*g
    add x0, x0, x14           // + (-8*h)

    ldp x29, x30, [sp], #16
    ret
