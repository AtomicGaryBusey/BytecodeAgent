sum8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // term2 = -2*b: compute 2*b then negate
    add x1, x1, x1          // 2*b
    neg x1, x1              // -2*b

    // term3 = 3*c: c + 2*c
    add x2, x2, x2, lsl #1  // 3*c

    // term4 = -4*d: 4*d then negate
    lsl x3, x3, #2          // 4*d
    neg x3, x3              // -4*d

    // term5 = 5*e: e + 4*e
    add x4, x4, x4, lsl #2  // 5*e

    // term6 = -6*f: compute 6*f then negate
    add x5, x5, x5          // 2*f
    add x5, x5, x5, lsl #1  // 6*f
    neg x5, x5              // -6*f

    // term7 = 7*g: g + 6*g
    mov x9, x6              // save original g in scratch register
    add x6, x6, x6          // 2*g
    add x6, x6, x6, lsl #1  // 6*g
    add x6, x9, x6          // 7*g

    // term8 = -8*h: 8*h then negate
    lsl x7, x7, #3          // 8*h
    neg x7, x7              // -8*h

    // Accumulate all terms into x0
    add x0, x0, x1          // a + (-2*b)
    add x0, x0, x2          // + 3*c
    add x0, x0, x3          // + (-4*d)
    add x0, x0, x4          // + 5*e
    add x0, x0, x5          // + (-6*f)
    add x0, x0, x6          // + 7*g
    add x0, x0, x7          // + (-8*h)

    ldp x29, x30, [sp], #16
    ret
