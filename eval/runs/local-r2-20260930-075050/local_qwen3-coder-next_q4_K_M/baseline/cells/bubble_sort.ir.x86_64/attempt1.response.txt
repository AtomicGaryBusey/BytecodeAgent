export function l $bubble_sort(l %a, l %n) {
entry:
	%0 = slt l %n, 2
	jnz %0, exit
	%i = copy l 0
loop_i:
	%1 = slt l %i, %n
	jz %1, exit
	%j = copy l 0
loop_j:
	%2 = sub l %n, 1
	%3 = slt l %j, %2
	jz %3, next_i
	%4 = add l %a, mul l %j, 8
	%5 = loadl l %4
	%6 = add l %a, mul l %j, 8
	%7 = loadl l %6
	%8 = add l %a, mul l %j, 8
	%9 = add l %j, 1
	%10 = add l %a, mul l %9, 8
	%11 = loadl l %10
	%12 = slt l %11, %7
	jz %12, skip_swap
	%13 = copy l %7
	storel l %11, %4
	storel l %13, %6
skip_swap:
	%14 = add l %j, 1
	%j = copy l %14
	jmp loop_j
next_i:
	%15 = add l %i, 1
	%i = copy l %15
	jmp loop_i
exit:
	%ret = copy l 0
	ret l %ret
}
