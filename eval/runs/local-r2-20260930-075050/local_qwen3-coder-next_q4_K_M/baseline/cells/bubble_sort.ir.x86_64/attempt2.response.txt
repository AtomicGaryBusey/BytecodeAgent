export function l $bubble_sort(l %a, l %n) {
@entry:
	%0 = slt l %n, 2
	jnz %0, @exit
	%i = copy l 0
@loop_i:
	%1 = slt l %i, %n
	jz %1, @exit
	%j = copy l 0
@loop_j:
	%2 = sub l %n, 1
	%3 = slt l %j, %2
	jz %3, @next_i
	%4 = add l %a, mul l %j, 8
	%5 = loadl l %4
	%6 = add l %a, mul l %j, 8
	%7 = add l %j, 1
	%8 = add l %a, mul l %7, 8
	%9 = loadl l %8
	%10 = slt l %9, %5
	jnz %10, @do_swap
	jmp @skip_swap
@do_swap:
	storel l %9, %4
	storel l %5, %8
	jmp @skip_swap
@skip_swap:
	%11 = add l %j, 1
	%j = copy l %11
	jmp @loop_j
@next_i:
	%12 = add l %i, 1
	%i = copy l %12
	jmp @loop_i
@exit:
	%ret = copy l 0
	ret l %ret
}
