export function l $bubble_sort(l %a, l %n) {
@entry:
	%0 = slt l %n, 2
	jnz %0, @exit
	%i = copy l 0
@loop_i:
	%1 = sub l %n, 1
	%2 = slt l %i, %1
	jz %2, @exit
	%j = copy l 0
@loop_j:
	%3 = sub l %n, 1
	%4 = slt l %j, %3
	jz %4, @next_i
	%5 = add l %a, mul l %j, 8
	%6 = loadl l %5
	%7 = add l %j, 1
	%8 = add l %a, mul l %7, 8
	%9 = loadl l %8
	%10 = slt l %9, %6
	jnz %10, @do_swap
	jmp @skip_swap
@do_swap:
	storel l %9, %5
	storel l %6, %8
	jmp @skip_swap
@skip_swap:
	%11 = add l %j, 1
	%j = copy l %11
	jmp @loop_j
@next_i:
	%12 = add l %i, 1
	%i = copy l %12
	jmp @loop_i
@exit:
	%ret = copy l 0
	ret l %ret
}
