binary_search:
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        cbz     x1, .Lnot_found // n == 0? return -1 immediately

        mov     x4, x0          // save base pointer a in x4
        mov     x5, x1          // high = n (exclusive bound)
        sub     x6, x6, x6      // low = 0

.Lloop:
        cmp     x6, x5
        b.ge    .Lnot_found     // if low >= high, exit with -1

        // mid = low + (high - low) / 2 to avoid overflow
        sub     x7, x5, x6      // high - low
        asr     x8, x7, #1      // divide by 2
        add     x9, x6, x8      // mid = low + (high-low)/2

        // Load a[mid]
        ldr     x10, [x4, x9, lsl #3]

        // Compare key with a[mid]
        cmp     x2, x10         // note: compare key (x2) with *a[mid] (x10)
        b.eq    .Lfound         // if equal, return mid

        // if key < a[mid], search lower half: high = mid
        b.lt    .Lsearch_lower

        // else key > a[mid], search upper half: low = mid + 1
        add     x6, x9, #1      // low = mid + 1
        b       .Lloop

.Lsearch_lower:
        mov     x5, x9          // high = mid
        b       .Lloop

.Lnot_found:
        mov     x0, #0xffffffffffffffff
        b       .Lepilogue

.Lfound:
        mov     x0, x9          // return mid

.Lepilogue:
        ldp     x29, x30, [sp], #16
        ret
