binary_search:
        // Prologue: save x29/x30, set up frame pointer, allocate stack space if needed
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        // Parameters: x0 = a, x1 = n, x2 = key
        mov     x4, x0          // save base pointer in callee-saved or scratch (x4 is free)
        mov     x5, x1          // n -> x5 (low index bound remains 0 initially)
        eor     x6, x6, x6      // low = 0
        cbz     x5, .Lnot_found // if n == 0, return -1

        // Initialize high = n-1
        sub     x7, x5, #1      // high = n-1 (x7 is scratch)

.Lloop:
        // Compute mid = low + (high - low) / 2
        sub     x8, x7, x6      // high - low
        asr     x9, x8, #1      // (high - low) >> 1  (arithmetic shift for signed divide by 2)
        add     x9, x6, x9      // mid = low + (high-low)/2

        // Load a[mid] using x4 as base
        ldr     x10, [x4, x9, lsl #3]

        // Compare key with a[mid]
        cmp     x10, x2
        b.eq    .Lfound         // if equal, return mid

        // If key < a[mid], search lower half: high = mid - 1
        blt     .Lsearch_lower

        // Else (key > a[mid]), search upper half: low = mid + 1
        add     x6, x9, #1      // low = mid + 1
        b       .Lcheck_high

.Lsearch_lower:
        sub     x7, x9, #1      // high = mid - 1

.Lcheck_high:
        // Check if loop continues: while (low <= high)
        cmp     x6, x7
        ble     .Lloop          // continue if low <= high

.Lnot_found:
        mvn     x0, xzr         // return -1 (all bits set)
        b       .Lepilogue

.Lfound:
        mov     x0, x9          // return mid

.Lepilogue:
        // Epilogue: restore frame pointer and LR
        ldp     x29, x30, [sp], #16
        ret
