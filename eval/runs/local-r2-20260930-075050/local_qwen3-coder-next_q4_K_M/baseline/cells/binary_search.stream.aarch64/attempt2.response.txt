binary_search:
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        mov     x4, x0          // save base pointer a in x4
        cbz     x1, .Lnot_found // n == 0? return -1 immediately

        mov     x5, x1          // high = n-1 (will be updated)
        sub     x5, x5, #1      // high = n-1
        eor     x6, x6, x6      // low = 0

.Lloop:
        // mid = low + (high - low) / 2 to avoid overflow
        sub     x7, x5, x6      // high - low
        asr     x8, x7, #1      // divide by 2
        add     x9, x6, x8      // mid = low + (high-low)/2

        // Load a[mid]
        ldr     x10, [x4, x9, lsl #3]

        // Compare key with a[mid]
        cmp     x10, x2
        b.eq    .Lfound         // if equal, return mid

        // if key < a[mid], search lower half: high = mid - 1
        b.lt    .Lsearch_lower

        // else key > a[mid], search upper half: low = mid + 1
        add     x6, x9, #1      // low = mid + 1
        b       .Lcheck_high

.Lsearch_lower:
        sub     x5, x9, #1      // high = mid - 1

.Lcheck_high:
        // while (low <= high)
        cmp     x6, x5
        ble     .Lloop          // continue if low <= high

.Lnot_found:
        mov     x0, #0xffffffffffffffff
        b       .Lepilogue

.Lfound:
        mov     x0, x9          // return mid

.Lepilogue:
        ldp     x29, x30, [sp], #16
        ret
