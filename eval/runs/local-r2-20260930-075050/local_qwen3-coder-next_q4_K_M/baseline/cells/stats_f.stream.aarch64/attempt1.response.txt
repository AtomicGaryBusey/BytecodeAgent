stats_f:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #28]

    mov x21, x0          // a pointer in x21
    mov x22, x1          // n in x22
    mov x19, x3          // f function ptr in x19

    cbz x22, .Ldone      // if n == 0, skip loop

    ldr x0, [x21], #8    // load a[0]
    blr x19              // y0 = f(a[0])
    mov x20, x0          // min = max = sum = y0
    mov x19, x0
    add x23, x0, x0      // tmp = 2*y0 to check oddness via LSB
    eor x24, x0, x23     // x0 ^ (x0 << 1) → LSB tells if original was odd
    lsr x25, x24, #63    // count_odd = (y0 & 1)
    neg x0, x0           // x0 = -min for later min update check

.Lloop:
    subs x22, x22, #1    // n--
    b.eq .Ldone          // if no more elements, done
    ldr x0, [x21], #8    // load a[i]
    blr x19              // y = f(a[i])
    
    // update min
    cmp x0, x20
    csel x20, x0, x20, lt

    // update max
    cmp x0, x19
    csel x19, x0, x19, gt

    // update sum
    add x23, x23, x0     // sum += y[i]

    // count odd: if (y & 1) increment counter
    and x24, x0, #1
    add x25, x25, x24

    b .Lloop

.Ldone:
    str x20, [x2]        // out[0] = min
    str x19, [x2, #8]    // out[1] = max
    str x23, [x2, #16]   // out[2] = sum

    mov x0, x25          // return count_odd

    ldr x19, [sp, #8]
    ldr x20, [sp, #16]
    ldr x21, [sp, #24]
    ldr x22, [sp, #28]
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret
