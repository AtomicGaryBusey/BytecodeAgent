stats_f:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    str x23, [sp, #40]

    mov x21, x0          // a pointer in x21
    mov x22, x1          // n in x22
    mov x19, x3          // f function ptr in x19

    cbz x22, .Ldone      // if n == 0, skip loop

    ldr x0, [x21], #8    // load a[0]
    add x21, x21, #0     // adjust pointer to next element (already incremented)
    blr x19              // y0 = f(a[0])
    mov x20, x0          // min = y0
    mov x21, x0          // max = y0
    mov x23, x0          // sum = y0
    and x22, x0, #1      // count_odd = y0 & 1

.Lloop:
    subs x1, x1, #1      // n--
    b.eq .Ldone          // if no more elements, done
    ldr x0, [x21], #8    // load a[i]
    blr x19              // y = f(a[i])
    
    // update min
    cmp x0, x20
    csel x20, x0, x20, lt

    // update max
    cmp x0, x21
    csel x21, x0, x21, gt

    // update sum
    add x23, x23, x0     // sum += y[i]

    // count odd: if (y & 1) increment counter
    and x4, x0, #1
    add x22, x22, x4

    b .Lloop

.Ldone:
    str x20, [x2]        // out[0] = min
    str x21, [x2, #8]    // out[1] = max
    str x23, [x2, #16]   // out[2] = sum

    mov x0, x22          // return count_odd

    ldr x19, [sp, #8]
    ldr x20, [sp, #16]
    ldr x21, [sp, #24]
    ldr x22, [sp, #32]
    ldr x23, [sp, #40]
    add sp, sp, #48
    ldp x29, x30, [sp], #16
    ret
