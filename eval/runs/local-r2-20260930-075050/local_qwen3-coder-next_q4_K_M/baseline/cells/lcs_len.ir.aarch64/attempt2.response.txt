export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =w ceql 0, 0
	%one =w cneq 0, 0

; Initialize row[0..n] to zero
	%i_init = l 0
@store_loop
	%cnd_store =csltl %i_init, %n
	jnz %cnd_store, @store_body
	jmp @outer_i_loop_start

@store_body
	%addr_store =l add %row, %i_init
	%val_store =l mul 0, 1
	storel %val_store, %addr_store
	%i_init =l add %i_init, 1
	jmp @store_loop

@outer_i_loop_start
	%zero_cmp =csltl 0, %m
	jnz %zero_cmp, @outer_i_body
	jmp @done

@outer_i_body
	; diag = 0
	%diag = l %zero
	%i_val = l %m
	%i_dec =l sub %i_val, 1
	%x_ptr = l %x
	%x_addr =l add %x_ptr, %i_dec
	%b_xi =w loadub %x_addr

	%j_val = l 1
@inner_j_loop_start
	%cnd_inner = csltl %j_val, %n
	jnz %cnd_inner, @inner_j_body
	jmp @outer_i_next

@inner_j_body
	; up = row[j]
	%addr_up =l add %row, %j_val
	%up = l loadl %addr_up

	; load y[j-1]
	%y_addr =l add %y, %j_val
	%y_addr_dec =l sub %y_addr, 1
	%b_yj =w loadub %y_addr_dec

	; if x[i] == y[j-1]
	%eq_cmp =w ceql %b_xi, %b_yj
	jnz %eq_cmp, @match_case
	jmp @no_match_case

@match_case
	%new_val =l add %diag, 1
	storel %new_val, %addr_up
	jmp @update_diag_after_match

@update_diag_after_match
	%diag = l %up
	%j_val =l add %j_val, 1
	jmp @inner_j_loop_start

@no_match_case
	; row[j] = max(row[j], row[j-1])
	%addr_prev =l add %row, %j_val
	%prev_addr_dec =l sub %addr_prev, 8
	%cur_val = l loadl %addr_up
	%prev_val = l loadl %prev_addr_dec

	%cnd_max = csltl %cur_val, %prev_val
	jnz %cnd_max, @use_prev
	%max_val = l %cur_val
	jmp @store_after_max
@use_prev
	%max_val = l %prev_val
@store_after_max
	storel %max_val, %addr_up

	%diag = l %up
	%j_val =l add %j_val, 1
	jmp @inner_j_loop_start

@outer_i_next
	%m_dec =l sub %m, 1
	%m = l %m_dec
	jmp @outer_i_loop_start

@done
	; return row[n]
	%addr_res =l add %row, %n
	%res = l loadl %addr_res
	ret %res
}
