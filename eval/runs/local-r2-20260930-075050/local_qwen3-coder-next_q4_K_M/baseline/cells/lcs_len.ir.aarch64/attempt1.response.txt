export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =w ceql 0, 0
	%i = l %m
	%j = l %n
	%row_base = l %row
	%x_base = l %x
	%y_base = l %y

; Initialize row[0..n] to zero
	%j_init = l %n
	%idx_init = l 0
@init_loop
	%cnd_init =csltl %idx_init, %j_init
	jnz %cnd_init, @init_body
	jmp @x_loop_start

@init_body
	%row_ptr_init = l %row_base
	%row_off_init = l %idx_init
	%row_addr_init = l %row_ptr_init
	%row_addroff_init = l 8
	%row_idx_init = l %row_off_init
	%row_idx_times8 = l %row_idx_init
	%row_idx_times8 =l mul %row_idx_times8, 8
	%row_final_addr_init = l %row_addr_init
	%row_final_addr_init =l add %row_final_addr_init, %row_idx_times8
	storel %zero, %row_final_addr_init
	%idx_init = l 1
	%idx_init =l add %idx_init, %idx_init
	jmp @init_loop

@x_loop_start
	%i_cmp =csltl 0, %i
	jnz %i_cmp, @x_loop_body
	jmp @done

@x_loop_body
	; diag = 0
	%diag = l %zero
	%j_idx = l 1
@y_loop_start
	%cnd_y = csltl %j_idx, %n
	jnz %cnd_y, @y_loop_body
	%i = l 1
	%i =l sub %i, %i
	jmp @x_loop_start

@y_loop_body
	; up = row[j]
	%row_j_ptr = l %row_base
	%row_off_j = l %j_idx
	%row_off_j_times8 = l %row_off_j
	%row_off_j_times8 =l mul %row_off_j_times8, 8
	%row_addr_j = l %row_j_ptr
	%row_final_addr_j = l %row_addr_j
	%row_final_addr_j =l add %row_final_addr_j, %row_off_j_times8
	%up = l loadl %row_final_addr_j

	; load x[i] and y[j-1]
	%x_i_ptr = l %x_base
	%x_i_off = l %i
	%x_i_idx_times1 = l %x_i_off
	%x_i_addr = l %x_i_ptr
	%x_i_final_addr = l %x_i_addr
	%x_i_final_addr =l add %x_i_final_addr, %x_i_idx_times1
	%b_xi =w loadub %x_i_final_addr

	%y_jm1_ptr = l %y_base
	%y_jm1_off = l %j_idx
	%y_jm1_off =l sub %y_jm1_off, 1
	%y_jm1_idx_times1 = l %y_jm1_off
	%y_jm1_addr = l %y_jm1_ptr
	%y_jm1_final_addr = l %y_jm1_addr
	%y_jm1_final_addr =l add %y_jm1_final_addr, %y_jm1_idx_times1
	%b_yjm1 =w loadub %y_jm1_final_addr

	; if x[i] == y[j-1]
	%eq_cmp =w ceql %b_xi, %b_yjm1
	jnz %eq_cmp, @match_case
	jmp @no_match_case

@match_case
	; row[j] = diag + 1
	%diag_plus1 = l 1
	%diag_plus1 =l add %diag_plus1, %diag
	%row_j_ptr2 = l %row_base
	%row_off_j2 = l %j_idx
	%row_off_j_times8_2 = l %row_off_j2
	%row_off_j_times8_2 =l mul %row_off_j_times8_2, 8
	%row_addr_j2 = l %row_j_ptr2
	%row_final_addr_j2 = l %row_addr_j2
	%row_final_addr_j2 =l add %row_final_addr_j2, %row_off_j_times8_2
	storel %diag_plus1, %row_final_addr_j2
	jmp @update_diag

@no_match_case
	; row[j] = max(row[j], row[j-1])
	%row_j_ptr3 = l %row_base
	%row_off_j3 = l %j_idx
	%row_off_j_times8_3 = l %row_off_j3
	%row_off_j_times8_3 =l mul %row_off_j_times8_3, 8
	%row_addr_j3 = l %row_j_ptr3
	%row_final_addr_j3 = l %row_addr_j3
	%row_final_addr_j3 =l add %row_final_addr_j3, %row_off_j_times8_3
	%cur_val = l loadl %row_final_addr_j3

	%row_jm1_ptr = l %row_base
	%row_off_jm1 = l %j_idx
	%row_off_jm1 =l sub %row_off_jm1, 1
	%row_off_jm1_times8 = l %row_off_jm1
	%row_off_jm1_times8 =l mul %row_off_jm1_times8, 8
	%row_addr_jm1 = l %row_jm1_ptr
	%row_final_addr_jm1 = l %row_addr_jm1
	%row_final_addr_jm1 =l add %row_final_addr_jm1, %row_off_jm1_times8
	%prev_val = l loadl %row_final_addr_jm1

	; max(cur_val, prev_val)
	%cnd_max = csltl %cur_val, %prev_val
	jnz %cnd_max, @use_prev
	%max_val = l %cur_val
	jmp @store_max
@use_prev
	%max_val = l %prev_val
@store_max
	storel %max_val, %row_final_addr_j3

@update_diag
	; diag = up (old row[j])
	%diag = l %up

	%j_idx = l 1
	%j_idx =l add %j_idx, %j_idx
	jmp @y_loop_start

@done
	; return row[n]
	%row_n_ptr = l %row_base
	%row_off_n = l %n
	%row_off_n_times8 = l %row_off_n
	%row_off_n_times8 =l mul %row_off_n_times8, 8
	%row_addr_n = l %row_n_ptr
	%row_final_addr_n = l %row_addr_n
	%row_final_addr_n =l add %row_final_addr_n, %row_off_n_times8
	%res = l loadl %row_final_addr_n
	ret %res
}
