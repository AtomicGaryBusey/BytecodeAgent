export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =w ceql 0, 0
	%i_init = l 0

@store_loop
	%cnd_store =csltl %i_init, %n
	jnz %cnd_store, @store_body
	jmp @outer_i_loop_start

@store_body
	%addr_store =l add %row, %i_init
	storel %zero, %addr_store
	%i_init =l add %i_init, 1
	jmp @store_loop

@outer_i_loop_start
	%cnd_outer =csltl %m, 0
	jnz %cnd_outer, @outer_i_body
	jmp @done

@outer_i_body
	%diag = l %zero

@inner_loop_start
	%j_val = l 1

@inner_loop_header
	%cnd_inner = csltl %j_val, %n
	jnz %cnd_inner, @inner_body
	jmp @inner_loop_end

@inner_body
	; up = row[j]
	%addr_up =l add %row, %j_val
	%up = l loadl %addr_up

	; load y[j-1]
	%y_addr_dec =l sub %j_val, 1
	%addr_y =l add %y, %y_addr_dec
	%b_yj =w loadub %addr_y

	; load x[i] (using original m from entry)
	%x_addr_dec =l sub %m, 1
	%x_addr =l add %x, %x_addr_dec
	%b_xi =w loadub %x_addr

	; if x[i] == y[j-1]
	%eq_cmp =w ceql %b_xi, %b_yj
	jnz %eq_cmp, @match_case
	jmp @no_match_case

@match_case
	%new_val =l add %diag, 1
	storel %new_val, %addr_up
	%diag = l %up
	jmp @inner_next

@no_match_case
	; row[j-1]
	%addr_prev =l sub %addr_up, 8
	%prev_val = l loadl %addr_prev
	%cur_val = l loadl %addr_up

	%cnd_max = csltl %cur_val, %prev_val
	jnz %cnd_max, @use_prev
	%max_val = l %cur_val
	jmp @store_after_max
@use_prev
	%max_val = l %prev_val
@store_after_max
	storel %max_val, %addr_up

	%diag = l %up
	jmp @inner_next

@inner_next
	%j_val =l add %j_val, 1
	jmp @inner_loop_header

@inner_loop_end
	jmp @outer_i_update

@outer_i_update
	%m =l sub %m, 1
	jmp @outer_i_loop_start

@done
	%addr_res =l add %row, %n
	%res = l loadl %addr_res
	ret %res
}
