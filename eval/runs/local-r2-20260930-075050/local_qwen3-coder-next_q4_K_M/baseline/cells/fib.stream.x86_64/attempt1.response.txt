fib:
    cmp rdi, 1
    jle .Lbase_case
    mov rax, 0          # fib(0)
    mov rbx, 1          # fib(1)
    xor rcx, rcx        # i = 0
.Lloop:
    inc rcx             # i++
    mov rdx, rax        # temp = fib(i-2)
    add rdx, rbx        # temp = fib(i-2) + fib(i-1)
    mov rax, rbx        # fib(i-2) = old fib(i-1)
    mov rbx, rdx        # fib(i-1) = new value
    cmp rcx, rdi
    jl .Lloop
    mov rax, rbx        # return fib(n)
    ret
.Lbase_case:
    mov rax, 1
    dec rax             # if n==0: 0; if n==1: 1
    ret
