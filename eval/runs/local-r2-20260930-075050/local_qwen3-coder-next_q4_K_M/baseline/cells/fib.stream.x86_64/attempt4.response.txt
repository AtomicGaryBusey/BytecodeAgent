fib:
    cmp rdi, 1
    jle .Lbase_case
    mov rbx, rdi          # save n in callee-saved register
    xor rcx, rcx          # i = 0
    mov rax, 0            # fib(0)
    mov rdx, 1            # fib(1)
.Lloop:
    inc rcx               # i++
    cmp rcx, rbx          # compare i with n
    jge .Ldone            # if i >= n, we're done
    mov r8, rax           # temp = fib(i-2)
    add r8, rdx           # temp = fib(i-2) + fib(i-1)
    mov rax, rdx          # shift: old fib(i-1) becomes new fib(i-2)
    mov rdx, r8           # new fib(i-1) = sum
    jmp .Lloop
.Ldone:
    mov rax, rdx          # return fib(n)
    ret
.Lbase_case:
    cmp rdi, 0
    je .Lzero
    mov rax, 1
    ret
.Lzero:
    xor rax, rax
    ret
