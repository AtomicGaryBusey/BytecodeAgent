shift_sat:
    // Prologue: save frame pointer and link register, set up stack frame
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Load arguments: x in x0, n in x1
    // We'll need to preserve x (original value) for later use
    mov x2, x0          // x2 = original x

    // Handle n >= 64 separately because shifts by >=64 are implementation-defined or special
    cmp x1, #64
    b.ge .L_large_n

    // Normal case: 0 <= n < 64
    // Compute logical left shift: x << n (using lsl)
    lsl x3, x2, x1      // x3 = x << n

    // Compute arithmetic right shift: x >> n (using asr)
    asr x4, x2, x1      // x4 = x >> n

    // Add with wrapping (64-bit addition)
    adds x0, x3, x4     // x0 = (x<<n) + (x>>n), sets flags
    bvc .L_done         // if no overflow, we're done

    // Overflow occurred: result is too large; clamp to INT64_MAX or INT64_MIN
    // Since addition of two 64-bit values can only overflow by 1 bit,
    // and we want wrapping (i.e., 2's complement wrap), but the spec says "wrapping 64-bit addition",
    // so actually just use the result as-is because ARMv8 adds with truncation on overflow for integers?
    // Wait: `adds` with V flag set means signed overflow; but in C, signed overflow is undefined.
    // However, specification says *wrapping* addition — meaning 2's complement modulo behavior.

    // In AArch64, the result of integer arithmetic is *always* modulo 2^64 for unsigned,
    // but for signed operations, the hardware doesn't define wrap — however, in practice,
    // the raw add instruction (add) does modulo 2^64.
    // Since spec says "wrapping", use `add` instead of `adds`.

    // Let's correct: replace `adds` with `add`
    // But we can't change code after writing it; so rewrite logic using `add`.

    // Instead, restructure to avoid adds entirely:
    // We'll redo the normal case with add.

.L_normal_add:
    mov x0, x3          // restore x<<n into result reg (x0)
    add x0, x0, x4      // x0 = (x<<n) + (x>>n), no flags, wraps modulo 2^64
    b .L_done

.L_large_n:
    // n >= 64: left shift is always 0; right shift is sign extension to all-ones if x<0 else zero.
    // x >> n = (x < 0) ? -1 : 0, for any n>=64
    asr x3, x2, #63     // x3 = x >> 63: gives all 1s if x negative, 0 otherwise
    // Now x << n = 0 (already set by initial mov? no; we need to zero it)
    eor x4, x4, x4      // x4 = 0 (logical left shift result for n>=64)
    // So result = 0 + sign_extended = x3
    mov x0, x3
    b .L_done

.L_done:
    // Epilogue: restore frame pointer and link register, return
    mov sp, x29
    ldp x29, x30, [sp], #16
    ret
