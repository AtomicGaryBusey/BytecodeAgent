bmin_empty:
    test rsi, rsi
    jz .Lempty
    mov rax, [rdi]
    lea rcx, [rsi - 1]
    lea rdx, [rdi + 8]
.Lloop:
    cmp rcx, 0
    jb .Ldone
    mov rbx, [rdx]
    cmp rbx, rax
    cmovg rax, rbx
    add rdx, 8
    dec rcx
    jmp .Lloop
.Ldone:
    ret
.Lempty:
    mov rax, 0x7fffffffffffffff
    ret
