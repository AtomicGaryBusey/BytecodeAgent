bmin_empty:
    test rsi, rsi
    jz .Lempty
    mov rax, qword ptr [rdi]
    cmp rsi, 1
    jbe .Ldone
    lea rcx, [rsi - 2]
    lea rdx, [rdi + 8]
.Lloop:
    test rcx, rcx
    js .Ldone
    mov rbx, qword ptr [rdx]
    cmp rbx, rax
    cmovg rax, rbx
    add rdx, 8
    dec rcx
    jmp .Lloop
.Ldone:
    ret
.Lempty:
    mov rax, 0x7fffffffffffffff
    ret
