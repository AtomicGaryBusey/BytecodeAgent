export function l $bmax(l %a, l %n) {
	%max = loadl %a
	%i = subl %n, 1

@loop:
	%cond = ceql %i, -1
	jnz %cond .exit
	%off = shll %i, 3
	%ptr = addl %a, %off
	%val = loadl %ptr
	%cond = csgtl %val, %max
	jnz %cond .update
	%i = subl %i, 1
	jmp .loop

@update:
	%max = %val
	%i = subl %i, 1
	jmp .loop

@exit:
	return %max
}
