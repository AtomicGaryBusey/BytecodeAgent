export function l $bmax(l %a, l %n) {
start:
	l %max
	l %i
	w %cond
	l %val
	l %next_max

	; max = a[0]
	%max = loadl %a
	; i = 1
	%i = addl %n, -1
	%cond = setneq %i, 0
	jnz %cond .body

	; if n == 1, return max immediately
	jmp .exit

.body:
	; loop start
.loop:
	; load a[i]
	%val = loadl addl %a, shll %i, 3
	; if val > max then max = val
	%cond = csgtl %val, %max
	jnz %cond .update_max
	jmp .next

.update_max:
	%max = %val
	jmp .next

.next:
	%i = addl %i, -1
	%cond = setneq %i, 0
	jnz %cond .loop

.exit:
	return %max
}
