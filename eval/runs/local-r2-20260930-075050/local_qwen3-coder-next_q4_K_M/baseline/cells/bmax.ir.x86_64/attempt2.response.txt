export function l $bmax(l %a, l %n) {
	%max = loadl %a
	%i = addl %n, -1
	%cond = ceql %i, 0
	jnz %cond .exit

.loop:
	%off = shll %i, 3
	%ptr = addl %a, %off
	%val = loadl %ptr
	%cond = csgtl %val, %max
	jnz %cond .update
	%i = addl %i, -1
	%cond = ceql %i, 0
	jnz %cond .exit
	jmp .loop

.update:
	%max = %val
	%i = addl %i, -1
	%cond = ceql %i, 0
	jnz %cond .exit
	jmp .loop

.exit:
	return %max
}
