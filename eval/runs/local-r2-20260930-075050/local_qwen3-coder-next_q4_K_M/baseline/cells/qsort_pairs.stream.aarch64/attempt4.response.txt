qsort_pairs:
    stp x29, x30, [sp, #-64]!
    add x29, sp, #16
    str x19, [x29, #-8]
    str x20, [x29, #-16]
    str x21, [x29, #-24]
    str x22, [sp, #8]
    str x23, [sp, #16]
    cmp x1, #1
    b.le .Lexit

    // Choose pivot: median-of-three for n>=2
    mov x19, x0                // base pointer (callee-saved)
    lsl x4, x1, #4             // total_bytes = 16*n
    add x5, x0, x4             // end pointer (one past last record)
    asr x6, x1, #1             // mid_index = n/2
    lsl x7, x6, #4             // mid_byte_offset = 16 * mid_index
    add x3, x0, x7             // middle record addr

    // Load first, middle, last elements
    ldp x7, x8, [x0]           // first: key,value
    ldp x9, x10, [x3]          // middle: key,value
    sub x11, x5, #16           // last record addr
    ldp x12, x13, [x11]

    // Compare first and middle
    cmp x7, x9
    b.gt .Lfirst_mid_swap
    b.lt .Lnext_compare_mid_last
    cmp x8, x10
    b.le .Lnext_compare_mid_last
.Lfirst_mid_swap:
    stp x9, x10, [x0]
    stp x7, x8, [x3]
    ldp x7, x8, [x0]           // reload first after swap
.Lnext_compare_mid_last:
    // Now compare middle (now in x7,x8) with last (x12,x13)
    cmp x7, x12
    b.gt .Lmid_last_swap
    b.lt .Lcompare_first_mid_final
    cmp x8, x13
    b.le .Lcompare_first_mid_final
.Lmid_last_swap:
    stp x9, x10, [x11]         // put original middle to last position
    stp x12, x13, [x3]         // put original last to middle
    ldp x7, x8, [x3]           // reload new middle
.Lcompare_first_mid_final:
    // Compare first with mid again (now first may have changed)
    cmp x7, x9
    b.gt .Lfirst_mid_swap2
    b.lt .Lpivot_selected
    cmp x8, x10
    b.le .Lpivot_selected
.Lfirst_mid_swap2:
    stp x9, x10, [x0]
    stp x7, x8, [x3]
    ldp x7, x8, [x3]           // pivot is now in middle (x7,x8)
.Lpivot_selected:

    mov x20, x0                // left pointer
    sub x21, x5, #16           // right pointer (last record)

.Lpartition_loop:
    // Move left pointer while *(left) < pivot
.Ladvance_left:
    ldp x4, x5, [x20]
    cmp x4, x7
    b.lt .Lskip_left_advance
    b.gt .Lcheck_right_side1
    cmp x5, x8
    b.le .Lskip_left_advance

.Lcheck_right_side1:
    // *(left) >= pivot: stop and check right side
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x4, x5, [x21]
    cmp x4, x7
    b.gt .Ldecrement_right
    b.lt .Lfound_swap_pair
    cmp x8, x5               // pivot.value vs right.value (note: reversed args)
    b.ge .Ldecrement_right

.Lfound_swap_pair:
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x9, x10, [x20]
    ldp x11, x12, [x21]
    stp x9, x10, [x21]
    stp x11, x12, [x20]

.Lskip_left_advance:
    add x20, x20, #16
    cmp x20, x21
    b.le .Lpartition_loop
    b .Lpartition_done

.Ldecrement_right:
    sub x21, x21, #16
    ldp x4, x5, [x21]          // pre-check to avoid re-reading after swap
    cmp x4, x7
    b.gt .Lpartition_loop      // continue decrementing
    b.lt .Lcheck_left_after_right
    cmp x5, x8
    b.ge .Lpartition_loop

.Lcheck_left_after_right:
    add x21, x21, #16          // undo last decrement (we'll re-check at this pos)
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x9, x10, [x20]
    ldp x11, x12, [x21]
    stp x9, x10, [x21]
    stp x11, x12, [x20]

.Lpartition_done:
    // Partition completed. Now recurse.

    // Left part: from base (x19) to x20 inclusive
    mov x0, x19
    sub x1, x20, x0             // byte distance left side (inclusive)
    add x1, x1, #16             // include last record in left part
    asr x1, x1, #4              // count = bytes / 16

    cbz x1, .Lsort_right_only   // skip if no elements

    bl qsort_pairs              // sort left part (recursive call)

.Lsort_right_only:
    // Right part: from x20+16 to end
    add x0, x20, #16            // right base
    sub x1, x5, x0              // byte distance
    asr x1, x1, #4              // count

    cbz x1, .Lexit              // skip if no elements

    bl qsort_pairs              // sort right part (recursive call)

.Lexit:
    ldr x19, [x29, #-8]
    ldr x20, [x29, #-16]
    ldr x21, [x29, #-24]
    ldr x22, [sp, #8]
    ldr x23, [sp, #16]
    add sp, sp, #32
    ldp x29, x30, [sp], #64
    ret
