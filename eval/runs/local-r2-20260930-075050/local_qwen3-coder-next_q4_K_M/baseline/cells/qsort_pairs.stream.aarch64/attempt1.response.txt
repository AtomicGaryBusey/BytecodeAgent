qsort_pairs:
    stp x29, x30, [sp, #-32]!
    add x29, sp, #16
    sub sp, sp, #16
    str x19, [x29, #-8]
    str x20, [x29, #-16]
    str x21, [x29, #-24]
    str x22, [sp, #8]
    mov x19, x0
    cmp x1, #1
    b.le .Lexit
    mov x20, x0
    ldr x21, [x0, #0]
    ldr x22, [x0, #8]
    adrp x0, .Lpivot_addr
    add x0, x0, :lo12:.Lpivot_addr
    ldp x23, x24, [x0]
    cmp x21, x23
    b.gt .Lcompare_high
    b.lt .Lswap_pivot
    cmp x22, x24
    b.ge .Lno_swap_pivot
.Lswap_pivot:
    str x23, [x20, #0]
    str x24, [x20, #8]
    str x21, [x0]
    str x22, [x0, #8]
.Lno_swap_pivot:
    mov x0, x20
    add x23, x20, #16
    ldr x24, [x23]
    ldr x25, [x23, #8]
    cmp x24, x23
    b.gt .Lpartition_done
    b.lt .Lfind_high
.Lcompare_low:
    cmp x24, x21
    b.lt .Lnext_low
    cmp x24, x21
    b.ne .Lnext_low
    cmp x25, x22
    b.le .Lnext_low
.Lfind_high:
    add x0, x20, #16
    sub x23, x29, #8
    str x24, [x23]
    ldr x24, [x20, #0]
    cmp x24, x21
    b.gt .Lfind_high_loop
.Lfind_high_loop:
    ldr x25, [x0, #16]
    ldr x26, [x0, #24]
    add x0, x0, #16
    cmp x25, x21
    b.gt .Lfind_high_done
    b.ne .Lfind_high_loop
    cmp x26, x22
    b.gt .Lfind_high_done
    cmp x0, x23
    b.ge .Lfind_high_done
    b.lt .Lfind_high_loop
.Lfind_high_done:
    ldr x24, [x23]
    cmp x0, x23
    b.le .Lpartition_done
.Lexchange:
    ldp x25, x26, [x20]
    ldp x27, x28, [x0]
    stp x25, x26, [x0]
    stp x27, x28, [x20]
    add x20, x20, #16
    sub x0, x0, #16
    ldr x24, [x20]
    cmp x24, x21
    b.le .Lfind_high
.Lnext_low:
    add x23, x23, #16
    ldr x24, [x23]
    cmp x23, x22
    b.ge .Lpartition_done
    b.lt .Lcompare_low
.Lpartition_done:
    sub x0, x0, #16
    mov x20, x0
    mov x1, #0
    ldr x2, [x19]
    cmp x2, x20
    b.ge .Lright_empty
    ldp x0, x1, [x20]
    stp x2, x1, [x20]
    str x0, [x20, #8]
.Lright_empty:
    add x19, x19, #16
    sub x1, x20, x19
    asr x1, x1, #4
    cbz x1, .Lleft_done
    mov x0, x19
    bl qsort_pairs
.Lleft_done:
    ldr x1, [x20, #8]
    add x0, x20, #16
    sub x1, x20, x0
    asr x1, x1, #4
    cbz x1, .Lexit
    bl qsort_pairs
.Lexit:
    ldr x19, [x29, #-8]
    ldr x20, [x29, #-16]
    ldr x21, [x29, #-24]
    ldr x22, [sp, #8]
    add sp, sp, #32
    ldp x29, x30, [sp], #32
    ret
.Lpivot_addr:
    .quad 0x7fffffffffffffff, 0x7fffffffffffffff
