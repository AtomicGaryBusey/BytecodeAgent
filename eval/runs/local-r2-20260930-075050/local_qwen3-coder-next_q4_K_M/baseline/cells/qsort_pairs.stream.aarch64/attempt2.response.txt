qsort_pairs:
    stp x29, x30, [sp, #-48]!
    add x29, sp, #16
    sub sp, sp, #16
    str x19, [x29, #-8]
    str x20, [x29, #-16]
    str x21, [x29, #-24]
    str x22, [sp, #8]
    mov x19, x0
    cmp x1, #1
    b.le .Lexit
    // Choose pivot: middle element
    add x2, x0, x1, lsl #3   // a + 2*n*8 = end pointer
    sub x2, x2, #16          // last record start
    add x3, x0, x1, lsr #1
    add x3, x3, x1, lsr #1
    sub x3, x3, #1           // (n-1)/2
    add x3, x3, x0, lsl #3   // offset to middle record: 8 * ((n-1)/2)
    ldp x4, x5, [x3]
    mov x6, x0
    ldp x7, x8, [x6]         // load first element
    cmp x4, x7
    b.gt .Lswap_pivot_needed
    b.lt .Lcompare_values_first
    cmp x5, x8
    b.ge .Lno_swap_pivot
.Lswap_pivot_needed:
    stp x4, x5, [x6]
    stp x7, x8, [x3]
.Lno_swap_pivot:
    mov x20, x0              // left index pointer
    add x21, x0, x1, lsl #3  // right boundary (end of array)
    sub x21, x21, #16        // last record start
    mov x7, x4               // pivot key
    mov x8, x5               // pivot value
    // Partition loop: find i,j such that elements <= pivot on left, >= pivot on right
.Lpartition_loop:
    ldp x9, x10, [x20]
    cmp x9, x7
    b.lt .Ladvance_left
    cmp x9, x7
    b.ne .Lcheck_right_side
    cmp x10, x8
    b.le .Ladvance_left
.Lcheck_right_side:
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x9, x10, [x21]
    cmp x9, x7
    b.gt .Ldecrement_right
    cmp x9, x7
    b.ne .Lcheck_left_after_right
    cmp x10, x8
    b.ge .Ldecrement_right
.Lcheck_left_after_right:
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x9, x10, [x20]
    ldp x11, x12, [x21]
    stp x9, x10, [x21]
    stp x11, x12, [x20]
.Ladvance_left:
    add x20, x20, #16
    cmp x20, x21
    b.le .Lpartition_loop
    b .Lpartition_done
.Ldecrement_right:
    sub x21, x21, #16
    cmp x20, x21
    b.ge .Lpartition_done
    b .Lpartition_loop
.Lpartition_done:
    mov x0, x19              // base pointer
    add x1, x20, #16         // position of first right element
    sub x1, x1, x0           // size of left part in bytes
    asr x1, x1, #4           // number of records in left part
    cbz x1, .Lsort_right_only
    mov x2, x1               // save n_left
    mov x0, x19              // base pointer for left sort
    bl qsort_pairs
.Lsort_right_only:
    add x0, x20, #16         // base of right part
    sub x1, x21, x0          // size in bytes
    add x1, x1, #16
    asr x1, x1, #4           // number of records in right part
    cbz x1, .Lexit
    bl qsort_pairs
.Lexit:
    ldr x19, [x29, #-8]
    ldr x20, [x29, #-16]
    ldr x21, [x29, #-24]
    ldr x22, [sp, #8]
    add sp, sp, #32
    ldp x29, x30, [sp], #48
    ret
