qsort_pairs:
    stp x29, x30, [sp, #-64]!
    add x29, sp, #16
    sub sp, sp, #32
    str x19, [x29, #-8]
    str x20, [x29, #-16]
    str x21, [x29, #-24]
    str x22, [sp, #8]
    str x23, [sp, #16]
    mov x19, x0
    cmp x1, #1
    b.le .Lexit
    // Choose pivot: middle element
    add x2, x0, x1, lsl #4     // a + 2*n*8 = end of array (byte offset)
    sub x2, x2, #16            // last record start
    add x3, x0, x1, lsr #1
    sub x3, x3, #1
    asr x4, x1, #1
    add x3, x0, x4, lsl #4     // middle record: 8 * (n/2) * 2? wait, n records each 16 bytes -> 16*(i)
    sub x3, x3, #8             // adjust for odd: use (n-1)/2
    ldp x7, x8, [x3]           // pivot key,value
    ldp x9, x10, [x0]          // first element
    cmp x7, x9
    b.gt .Lswap_pivot_needed
    b.lt .Lno_swap_pivot
    cmp x8, x10
    b.ge .Lno_swap_pivot
.Lswap_pivot_needed:
    stp x7, x8, [x0]
    stp x9, x10, [x3]
.Lno_swap_pivot:
    mov x20, x0                // left pointer
    add x21, x0, x1, lsl #4    // right boundary (end of array)
    sub x21, x21, #16          // last record start
    ldp x7, x8, [x3]           // reload pivot (may have been swapped)
    mov x23, x3                // save pivot index offset for later if needed

.Lpartition_loop:
    ldp x4, x5, [x20]
    cmp x4, x7
    b.lt .Ladvance_left
    cmp x4, x7
    b.gt .Lcheck_right_side
    cmp x5, x8
    b.le .Ladvance_left

.Lcheck_right_side:
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x4, x5, [x21]
    cmp x4, x7
    b.gt .Ldecrement_right
    cmp x4, x7
    b.lt .Lcheck_left_after_right
    cmp x5, x8
    b.ge .Ldecrement_right

.Lcheck_left_after_right:
    cmp x20, x21
    b.ge .Lpartition_done
    ldp x9, x10, [x20]
    ldp x11, x12, [x21]
    stp x9, x10, [x21]
    stp x11, x12, [x20]

.Ladvance_left:
    add x20, x20, #16
    cmp x20, x21
    b.le .Lpartition_loop
    b .Lpartition_done

.Ldecrement_right:
    sub x21, x21, #16
    cmp x20, x21
    b.ge .Lpartition_done
    b .Lpartition_loop

.Lpartition_done:
    // Now left part is [base, x20], right part is [x20+16, end]
    mov x0, x19                // base pointer for recursion call
    add x1, x20, #16           // first index after left part (byte)
    sub x1, x1, x0             // size in bytes of left part
    asr x1, x1, #4             // number of records
    cbz x1, .Lsort_right_only

    mov x2, x1                 // save n_left
    bl qsort_pairs

.Lsort_right_only:
    add x0, x20, #16           // base of right part
    sub x1, x21, x0            // distance from left end to right start (negative if overlap)
    add x1, x1, #16            // adjust to count records in right
    asr x1, x1, #4             // number of records
    cbz x1, .Lexit

    bl qsort_pairs

.Lexit:
    ldr x19, [x29, #-8]
    ldr x20, [x29, #-16]
    ldr x21, [x29, #-24]
    ldr x22, [sp, #8]
    ldr x23, [sp, #16]
    add sp, sp, #32
    ldp x29, x30, [sp], #64
    ret
