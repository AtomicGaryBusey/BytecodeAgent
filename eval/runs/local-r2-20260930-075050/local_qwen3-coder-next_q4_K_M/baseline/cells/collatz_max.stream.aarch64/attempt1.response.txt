collatz_max:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    mov x19, x0                    // n saved in x19
    mov x20, #0                    // max_steps = 0

.Louter_loop:
    cmp x0, #1
    blt .Ldone                     // if n < 1, done (but spec says n >= 1)
    cbz x0, .Ldone                 // if n == 0, done (defensive)

    mov x21, x0                    // current k = n
    mov x22, #0                    // steps for this k

.Linner_loop:
    cmp x21, #1
    beq .Lcount_done               // reached 1, exit inner loop

    tbz x21, #0, .Leven_case     // check LSB (is even?)
    // odd case: 3*x + 1
    add x22, x22, #1              // steps++
    lsl x9, x21, #1               // x21 * 2
    add x21, x9, x21              // x21 * 3
    add x21, x21, #1              // x21 = 3*x + 1
    b .Linner_loop

.Leven_case:
    lsr x21, x21, #1              // x21 /= 2
    add x22, x22, #1              // steps++
    b .Linner_loop

.Lcount_done:
    cmp x22, x20                  // compare current steps to max
    csel x20, x22, x20, gt        // max_steps = max(max_steps, steps)

    sub x0, x0, #1                // n--
    cbnz x0, .Louter_loop          // loop for next k

.Ldone:
    mov x0, x20                   // result = max_steps
    ldr x19, [sp, #8]
    ldr x20, [sp, #16]
    ldr x21, [sp, #24]
    ldr x22, [sp, #32]
    add sp, sp, #48
    ldp x29, x30, [sp], #16
    ret
