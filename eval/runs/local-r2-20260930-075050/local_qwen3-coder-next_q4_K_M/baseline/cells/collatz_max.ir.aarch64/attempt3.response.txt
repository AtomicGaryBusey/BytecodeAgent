export function l $collatz_max(l %n) {
@start
	l %max_steps
	l %i
	l %current
	l %steps
	l %next_val
	l %is_even
%max_steps = copy 0
%i = copy 1
@loop_start
	csltl %loop_cond, %i, %n
	jnz %loop_cond, @loop_body
	jmp @done
@loop_body
	%current = copy %i
	%steps = copy 0
@collatz_loop
	ceql %is_one, %current, 1
	jnz %is_one, @collatz_done
	lshr %temp_even, %current, 1
	and %is_even, %current, 1
	ceql %even_test, %is_even, 0
	jnz %even_test, @even_case
	mul %odd_part, %current, 3
	add %next_val, %odd_part, 1
	jmp @next_step
@even_case
	%next_val = copy %temp_even
@next_step
	%current = copy %next_val
	%steps = add %steps, 1
	jmp @collatz_loop
@collatz_done
	csltl %update_max, %max_steps, %steps
	selgt %new_max, %steps, %max_steps
	%max_steps = copy %new_max
%i = add %i, 1
	jmp @loop_start
@done
	lret %max_steps
}
