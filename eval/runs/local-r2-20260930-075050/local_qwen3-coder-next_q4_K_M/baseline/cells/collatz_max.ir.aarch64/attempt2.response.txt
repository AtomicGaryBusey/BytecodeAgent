export function l $collatz_max(l %n) {
@start
	l %max_steps
	l %i
	l %current
	l %steps
%max_steps = copy 0
%i = copy 1
@loop_start
	csltl %loop_cond, %i, %n
	jnz %loop_cond, @loop_body
	jmp @done
@loop_body
	%current = copy %i
	%steps = copy 0
@collatz_loop
	ceql %is_one, %current, 1
	jnz %is_one, @collatz_done
	l %next_val
	l %temp_even
	l %is_even
	l %odd_part
	temp_even = lshr 1, %current
	is_even = and %current, 1
	ceql %even_test, is_even, 0
	jnz %even_test, @even_case
	odd_part = mul %current, 3
	next_val = add odd_part, 1
	jmp @next_step
@even_case
	next_val = copy temp_even
@next_step
	%current = copy next_val
	%steps = add %steps, 1
	jmp @collatz_loop
@collatz_done
	csltl %update_max, %max_steps, %steps
	l %new_max
	new_max = select %update_max, %steps, %max_steps
	%max_steps = copy new_max
%i = add %i, 1
	jmp @loop_start
@done
	x0 = copy %max_steps
	ret
}
