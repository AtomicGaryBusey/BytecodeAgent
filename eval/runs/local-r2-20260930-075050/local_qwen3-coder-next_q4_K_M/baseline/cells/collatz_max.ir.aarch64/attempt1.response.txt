export function l $collatz_max(l %n) {
@start
	l %max_steps
	l %i
	l %current
	l %steps
%max_steps = copy 0
%i = copy 1
@loop_start
	csltl %i_check, %i, %n
	jnz %i_check, @loop_body
	jmp @done
@loop_body
	%current = copy %i
	steps = copy 0
@collatz_loop
	ceql %check_one, %current, 1
	jnz %check_one, @collatz_done
	l %temp
	temp = lshr 1, %current
	%is_even = and %current, 1
	ceql %even_check, %is_even, 0
	mul %odd_part, %current, 3
	add %next_val, %odd_part, 1
	%current = select %even_check, temp, %next_val
	steps = add steps, 1
	jmp @collatz_loop
@collatz_done
	csltl %max_check, %max_steps, steps
	l %new_max
	new_max = select %max_check, steps, %max_steps
	%max_steps = copy new_max
%i = add %i, 1
	jmp @loop_start
@done
	x0 = copy %max_steps
	ret
}
