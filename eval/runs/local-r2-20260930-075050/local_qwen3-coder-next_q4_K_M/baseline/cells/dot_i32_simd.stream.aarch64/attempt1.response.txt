dot_i32_simd:
    // Prologue: save frame pointer, link register, and callee-saved registers we use
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48              // align stack to 16 bytes, reserve space for x19-x22 (4 regs)
    stp x19, x20, [sp]
    stp x21, x22, [sp, #16]
    stp d8, d9, [sp, #32]

    // Load arguments
    mov x19, x0                  // a (callee-saved)
    mov x20, x1                  // b (callee-saved)
    mov x21, x2                  // n (callee-saved)

    // Initialize accumulator to zero
    movi v0.2d, #0               // sum_lo: lower 64-bit chunks of partial products
    movi v1.2d, #0               // sum_hi: upper 64-bit chunks of partial products (we'll use v0/v1 for final sum as int64_t)

    // Check if n == 0
    cbz x21, .Ldone

    // Compute how many full vectors of 4 elements we can process
    mov x22, x21
    lsl x22, x22, #2             // bytes per 4 ints = 16 bytes; but we'll count by elements
    lsr x22, x22, #2             // back to n
    lsrv x22, x22, #2            // n / 4 (number of vector iterations)
    cb
