dot_i32_simd:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    stp x19, x20, [sp]
    stp x21, x22, [sp, #16]
    stp d8, d9, [sp, #32]

    mov x19, x0
    mov x20, x1
    mov x21, x2

    movi v0.2d, #0
    movi v1.2d, #0

    cbz x21, .Ldone

    // n / 4 for vector iterations
    lsr x22, x21, #2

.Lloop:
    ld1 {v2.4s}, [x19], #16
    ld1 {v3.4s}, [x20], #16
    smull v4.2d, v2.2s, v3.2s
    smull2 v5.2d, v2.4s, v3.4s
    addp v0.2d, v0.2d, v4.2d
    addp v1.2d, v1.2d, v5.2d
    subs x22, x22, #1
    b.ne .Lloop

    add v0.2d, v0.2d, v1.2d
    fmov x0, d0

    ands x21, x21, #3
    beq .Lfinish

.Lscalar_loop:
    ldr w2, [x19], #4
    ldr w3, [x20], #4
    smull x2, x2, x3
    add
