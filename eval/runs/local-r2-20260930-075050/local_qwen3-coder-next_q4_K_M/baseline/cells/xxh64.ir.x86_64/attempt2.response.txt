export function l $xxh64(l %p, l %n, l %seed) {
.Lentry:
  %0 =l copy %n
  %1 =l csltl %0, 32
  %2 =w copy %0
  jnz %2, .Lbody32, .Lless32

.Lbody32:
  %v1 =l add %seed, 7046029288634856825
  %v1 =l add %v1, 13985955806128127823
  %v2 =l add %seed, 13985955806128127823
  %v3 =l copy %seed
  %v4 =l sub %seed, 7046029288634856825
  %q =l copy %p
  %rem =l copy %n

.LstripeLoop:
  %lenLeft =l copy %rem
  %cmpStripe =l csltl %lenLeft, 32
  %wCmpStripe =w copy %cmpStripe
  jz %wCmpStripe, .LstripeDone

  %x1 =l loadl %q
  %tmp1 =l mul %x1, 13985955806128127823
  %tmp2 =l add %v1, %tmp1
  %rot1 =.Lrotl64(%tmp2, 31)
  %v1 =l mul %rot1, 7046029288634856825

  %q1 =l add %q, 8
  %x2 =l loadl %q1
  %tmp3 =l mul %x2, 13985955806128127823
  %tmp4 =l add %v2, %tmp3
  %rot2 =.Lrotl64(%tmp4, 31)
  %v2 =l mul %rot2, 7046029288634856825

  %q2 =l add %q1, 8
  %x3 =l loadl %q2
  %tmp5 =l mul %x3, 13985955806128127823
  %tmp6 =l add %v3, %tmp5
  %rot3 =.Lrotl64(%tmp6, 31)
  %v3 =l mul %rot3, 7046029288634856825

  %q3 =l add %q2, 8
  %x4 =l loadl %q3
  %tmp7 =l mul %x4, 13985955806128127823
  %tmp8 =l add %v4, %tmp7
  %rot4 =.Lrotl64(%tmp8, 31)
  %v4 =l mul %rot4, 7046029288634856825

  %q =l add %q3, 8
  %rem =l sub %lenLeft, 32
  jnz .LstripeLoop

.LstripeDone:
  %rotV1 =.Lrotl64(%v1, 1)
  %rotV2 =.Lrotl64(%v2, 7)
  %rotV3 =.Lrotl64(%v3, 12)
  %rotV4 =.Lrotl64(%v4, 18)

  %h =l add %rotV1, %rotV2
  %h =l add %h, %rotV3
  %h =l add %h, %rotV4

  %h =.Lmerge(%h, %v1)
  %h =.Lmerge(%h, %v2)
  %h =.Lmerge(%h, %v3)
  %h =.Lmerge(%h, %v4)
  jnz .LpostStripe

.Lless32:
  %h =l add %seed, 2988233467441803525

.LpostStripe:
  %h =l add %h, %n
  %q =l copy %p
  %remBytes =l copy %n

.LtailLoop:
  %lenTail =l copy %remBytes
  %cmpTail =l csltl %lenTail, 8
  %wCmpTail =w copy %cmpTail
  jz %wCmpTail, .Ltail4

  %xTail =l loadl %q
  %tmpTail =.Lround(0, %xTail)
  %h ^= %tmpTail
  %rotH =.Lrotl64(%h, 27)
  %h =l mul %rotH, 7046029288634856825
  %h =l add %h, 13985955806128127823

  %q =l add %q, 8
  %remBytes =l sub %lenTail, 8
  jnz .LtailLoop

.Ltail4:
  %len4 =l copy %remBytes
  %cmp4 =l csltl %len4, 4
  %wCmp4 =w copy %cmp4
  jz %wCmp4, .Ltail1

  %x32 =l loadw %q
  %mulP1 =l mul %x32, 7046029288634856825
  %h ^= %mulP1
  %rotH4 =.Lrotl64(%h, 23)
  %tmpH4 =l mul %rotH4, 13985955806128127823
  %h =l add %tmpH4, 16566719274737489973

  %q =l add %q, 4
  %remBytes =l sub %len4, 4

.Ltail1:
  %cmpByte =l csltl %remBytes, 1
  %wCmpByte =w copy %cmpByte
  jz %wCmpByte, .LfinalMix

.LbyteLoop:
  %c =l loadw %q
  %mulP5 =l mul %c, 2988233467441803525
  %h ^= %mulP5
  %rotHB =.Lrotl64(%h, 11)
  %h =l mul %rotHB, 7046029288634856825

  %q =l add %q, 1
  %remBytes =l sub %remBytes, 1
  %cmpByte2 =l csltl %remBytes, 1
  %wCmpByte2 =w copy %cmpByte2
  jnz .LbyteLoop

.LfinalMix:
  %h_shift1 =l shr %h, 33
  %h ^= %h_shift1
  %h_mul1 =l mul %h, 13985955806128127823
  %h_shift2 =l shr %h_mul1, 29
  %h ^= %h_shift2
  %h_mul2 =l mul %h, 16566719274737489973
  %h_shift3 =l shr %h_mul2, 32
  %h ^= %h_shift3

  ret l %h

.Lrotl64:
  %r =i %2
  %shlRes =l shl %1, %r
  %shiftAmt =l sub 64, %r
  %shrRes =l shr %1, %shiftAmt
  %res =l or %shlRes, %shrRes
  ret l %res

.Lround:
  %x =l copy %2
  %mulP2 =l mul %x, 13985955806128127823
  %addZ =l add %1, %mulP2
  %rotRes =.Lrotl64(%addZ, 31)
  %res =l mul %rotRes, 7046029288634856825
  ret l %res

.Lmerge:
  %v =l copy %2
  %mulP2_merge =l mul %v, 13985955806128127823
  %addRound =l add 0, %mulP2_merge
  %rotRound =.Lrotl64(%addRound, 31)
  %roundRes =l mul %rotRound, 7046029288634856825
  %tmpXor =l xor %1, %roundRes
  %mulP1_merge =l mul %tmpXor, 7046029288634856825
  %res =l add %mulP1_merge, 13985955806128127823
  ret l %res
}
