export function l $popcount(l %x) {
@start
	w %count
	l %val
	w %loop_cond

%count = copy 0
%val = copy %x
%loop_cond = csltl 0, %val
jnz %loop_cond, @loop_body
j @end

@loop_body
	l %t1
	w %mask
	
%mask = copy 1
%t1 = subl %val, %mask
%val = andl %val, %t1
%count = addw %count, 1
%loop_cond = csltl 0, %val
jnz %loop_cond, @loop_body
j @end

@end
%result = zextwl %count
ret %result
}
