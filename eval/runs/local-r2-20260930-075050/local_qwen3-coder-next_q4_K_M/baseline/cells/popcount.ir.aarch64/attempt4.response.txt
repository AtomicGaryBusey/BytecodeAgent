export function l $popcount(l %x) {
@start
	w %count
	l %val

%count = copy 0
%val = copy %x
j @loop_test

@loop_body
	l %t1
	l %mask
	
%mask = copy 1
%t1 = subl %val, %mask
%val = andl %val, %t1
%count = addw %count, 1

@loop_test
	w %loop_cond
%loop_cond = csltl 0, %val
jnz %loop_cond, @loop_body
j @end

@end
	l %result
%result = zextwl %count
ret %result
}
