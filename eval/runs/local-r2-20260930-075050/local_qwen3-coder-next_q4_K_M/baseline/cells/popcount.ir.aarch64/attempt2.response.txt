export function l $popcount(l %x) {
@start
	w %count
	l %val
	w %done

%count = copy 0
%val = copy %x
%done = ceql %val, 0
jnz %done, @end

@loop_body
	l %t1
	w %mask
	
%mask = copy 1
%t1 = subl %val, %mask
%val = andl %val, %t1
%count = addw %count, 1
%done = ceql %val, 0
jnz %done, @end
j @loop_body

@end
%result = zextwl %count
ret %result
}
