export function l $popcount(l %x) {
@start
	w %mask
	l %t1, %t2, %t3, %t4, %t5, %t6, %t7, %t8, %t9, %t10
	w %cond

; Brian Kernighan’s algorithm: clear the lowest set bit repeatedly
; Count iterations until x becomes zero

	w %count
	w %done
	l %val

%count = copy 0
%val = copy %x
%done = ceql %val, 0
jnz %done, @end

@loop_body
%mask = copy 1
%t1 = subl %val, %mask
%t2 = andl %val, %t1
%val = copy %t2
%count = addw %count, 1
%done = ceql %val, 0
jnz %done, @end
j @loop_body

@end
%result = zextwl %count
ret %result
}
