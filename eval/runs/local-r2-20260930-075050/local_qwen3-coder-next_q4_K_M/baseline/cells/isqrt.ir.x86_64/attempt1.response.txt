export function l $isqrt(l %n) {
@start
    %r = alloc1l
    %low = alloc1l
    %high = alloc1l
    %mid = alloc1l
    %mid_sq = alloc1l
    %mid_plus_one = alloc1l
    %mid_plus_one_sq = alloc1l

    storel %n, %r
    storel $0, %low
    storel %n, %high
    jnz @loop_start

@loop_start
    %low_val = loadl %low
    %high_val = loadl %high
    %cmp = csltl %low_val, %high_val
    jnz @loop_body, %cmp
    jmp @done

@loop_body
    %mid_val = addl %low_val, %high_val
    %mid_val = shrl %mid_val, $1
    storel %mid_val, %mid

    %m = loadl %mid
    %m_sq = mulh %m, %m
    storel %m_sq, %mid_sq

    %n_val = loadl %r
    %cmp_mid = csltl %n_val, %m_sq
    jnz @less_than_mid, %cmp_mid
    jmp @greater_equal_mid

@less_than_mid
    %high_val_new = subl %mid_val, $1
    storel %high_val_new, %high
    jmp @loop_start

@greater_equal_mid
    %m_plus_one = addl %m, $1
    storel %m_plus_one, %mid_plus_one

    %mpo = loadl %mid_plus_one
    %cmp_overflow = csltl %mpo, %m
    jnz @overflow_detected, %cmp_overflow

    %mpo_sq = mulh %mpo, %mpo
    storel %mpo_sq, %mid_plus_one_sq

    %n_val2 = loadl %r
    %cmp_mpo = csltl %n_val2, %mpo_sq
    jnz @greater_equal_mid_check, %cmp_mpo
    jmp @found_exact

@overflow_detected
    storel %mid_val, %low
    jmp @loop_start

@greater_equal_mid_check
    storel %mid_val, %low
    jmp @loop_start

@found_exact
    %m_plus_one_final = loadl %mid_plus_one
    storel $1, %r
    %result = addl %m_plus_one_final, -1
    storel %result, %r
    jmp @done

@done
    %final_result = loadl %r
    ret %final_result
}
