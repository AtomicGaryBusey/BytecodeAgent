export function l $isqrt(l %n) {
@start
    %low = $0
    %high = %n
    if %n == $0 { jmp @done }

@loop_start
    %mid = addl %low, %high
    %mid = shrl %mid, $1

    %sq_lo = mull %mid, %mid
    %cmp_eq = ceql %sq_lo, %n
    jnz @found_exact, %cmp_eq

    %cmp_lt = csltl %sq_lo, %n
    jnz @mid_sq_less_than_n, %cmp_lt

    %sq_hi = mulh %mid, %mid
    %cmp_lt_high = csltl %n, %sq_hi
    jnz @mid_too_large, %cmp_lt_high

    %is_exact = ceql %sq_hi, $0
    jnz @check_lo_partial, %is_exact

@check_lo_partial
    %cmp_lo_exact = ceql %sq_lo, %n
    jnz @found_exact, %cmp_lo_exact
    jmp @mid_too_large

@mid_sq_less_than_n
    %next_mid = addl %mid, $1
    %next_sq_lo = mull %next_mid, %next_mid
    %cmp_next_eq = ceql %next_sq_lo, %n
    jnz @found_exact, %cmp_next_eq

    %cmp_next_lt = csltl %next_sq_lo, %n
    jnz @mid_valid_but_increase, %cmp_next_lt

    %next_sq_hi = mulh %next_mid, %next_mid
    %cmp_next_lt_high = csltl %n, %next_sq_hi
    jnz @found_exact, %cmp_next_lt_high

    %is_next_exact = ceql %next_sq_hi, $0
    jnz @check_next_lo_partial, %is_next_exact

@check_next_lo_partial
    %cmp_next_lo_eq = ceql %next_sq_lo, %n
    jnz @found_exact, %cmp_next_lo_eq
    jmp @mid_valid_but_increase

@mid_too_large
    %high = subl %mid, $1
    cmpw %low, %high
    jbe @loop_start
    jmp @done

@mid_valid_but_increase
    %low = addl %mid, $1
    cmpw %low, %high
    jbe @loop_start
    jmp @done

@found_exact
    %low = %mid
    jmp @done

@done
    ret %low
}
