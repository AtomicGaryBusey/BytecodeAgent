export function l $isqrt(l %n) {
@start
    %r = $0
    %rem = %n
    %bit = $1
    %bit = shll %bit, $62
@loop
    %t = addl %r, %bit
    %t_lo = mull %t, %t
    %t_hi = mulh %t, %t
    %rem_cmp = csltl %rem, %t_hi
    jnz @skip_bit, %rem_cmp
    %rem_eq = ceql %t_hi, %rem
    jz @less_than_t_sq, %rem_eq
    %t_lo_check = mull %t, %bit
    %t_lo_check = shll %t_lo_check, $1
    %diff = subl %t_lo, %t_lo_check
    %rem_ge = csltl %diff, %rem
    jnz @skip_bit, %rem_ge
@apply_bit
    %r = %t
    %rem = subl %rem, %t_lo
@skip_bit
    %bit = shrl %bit, $2
    cmpw %bit, $0
    jnz @loop, %bit
@done
    ret %r
@less_than_t_sq
    %bit = shrl %bit, $2
    cmpw %bit, $0
    jnz @loop, %bit
    jmp @done
}
