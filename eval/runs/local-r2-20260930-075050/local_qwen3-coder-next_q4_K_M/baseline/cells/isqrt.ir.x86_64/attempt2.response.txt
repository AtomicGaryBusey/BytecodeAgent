export function l $isqrt(l %n) {
@start
    %low = alloca l
    %high = alloca l
    %mid = alloca l

    storel $0, %low
    storel %n, %high

@loop_start
    %low_val = loadl %low
    %high_val = loadl %high
    %cmp = csltl %low_val, %high_val
    jnz @loop_body, %cmp
    jmp @done

@loop_body
    %sum = addl %low_val, %high_val
    %mid_val = shrl %sum, $1
    storel %mid_val, %mid

    %m = loadl %mid
    %m_sq_hi = mulh %m, %m
    %m_sq_lo = mull %m, %m
    %cmp_m_sq = csltl %n, %m_sq_hi
    jnz @less_than_mid, %cmp_m_sq

    %cmp_equal = ceql %m_sq_hi, $0
    jnz @check_low, %cmp_equal

@check_low
    %cmp_lo = csltl %n, %m_sq_lo
    jz @found_or_higher, %cmp_lo

@less_than_mid
    %new_high = subl %mid_val, $1
    storel %new_high, %high
    jmp @loop_start

@found_or_higher
    %next_m = addl %m, $1
    %check_overflow = csltl %next_m, %m
    jnz @handle_overflow, %check_overflow

    %next_sq_hi = mulh %next_m, %next_m
    %next_sq_lo = mull %next_m, %next_m
    %cmp_next = csltl %n, %next_sq_hi
    jnz @use_current, %cmp_next

    %cmp_next_lo = ceql %next_sq_hi, $0
    jnz @check_next_low, %cmp_next_lo

@check_next_low
    %cmp_next_lo_val = csltl %n, %next_sq_lo
    jz @use_current, %cmp_next_lo_val

@handle_overflow
    storel %mid_val, %low
    jmp @loop_start

@use_current
    storel %m, %low
    storel %mid_val, %high
    jmp @loop_start

@done
    %final = loadl %low
    ret %final
}
