sum8:
    # Compute: a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h
    # Arguments: rdi=a, rsi=b, rdx=c, rcx=d, r8=e, r9=f, [rsp+8]=g, [rsp+16]=h

    mov rax, rdi          # a
    lea rbx, [rsi*2]      # 2*b
    sub rax, rbx          # a - 2*b

    lea rcx, [rdx + rdx*2] # 3*c (c + 2*c)
    add rax, rcx          # a - 2*b + 3*c

    mov rcx, rcx          # zero-extend d
    shl rcx, 2            # 4*d = d << 2
    sub rax, rcx          # a - 2*b + 3*c - 4*d

    lea rcx, [r8 + r8*4]  # 5*e = e + 4*e
    add rax, rcx          # ... + 5*e

    mov rdx, r9           # f
    shl rdx, 1            # 2*f
    lea rcx, [rdx + rdx*2]# 6*f = 2*f + 4*f
    sub rax, rcx          # ... - 6*f

    mov r8, qword ptr [rsp+8] # g (7th arg)
    lea r9, [r8 + r8*2]     # 3*g
    shl r8, 1               # 2*g
    add r8, r9              # 5*g = 2*g + 3*g; wait, need 7*g
    lea r8, [r8 + r9]       # 7*g = 5*g + 2*g

    mov rdi, qword ptr [rsp+16] # h (8th arg)
    shl rdi, 3                # 8*h = h << 3

    add rax, r8               # ... + 7*g
    sub rax, rdi              # ... - 8*h

    ret
