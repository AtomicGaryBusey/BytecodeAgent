sum8:
    # Arguments: rdi=a, rsi=b, rdx=c, rcx=d, r8=e, r9=f, [rsp+8]=g, [rsp+16]=h
    # Must preserve callee-saved registers (rbx, rbp, r12-r15)

    push rbx              # Save rbx (callee-saved)
    sub rsp, 8            # Align stack to 16-byte boundary (rsp was 8 mod 16, now 0 mod 16)

    mov rbx, rdi          # a
    lea rax, [rsi + rsi]  # 2*b = b + b
    sub rbx, rax          # a - 2*b

    lea rcx, [rdx + rdx*2]# 3*c = c + 2*c
    add rbx, rcx          # a - 2*b + 3*c

    mov rax, rcx          # reuse rcx temporarily; save 3*c in rax? No, recalc
    mov rax, rcx          # rax = 3*c (but we need to recompute 4*d)
    
    mov r12, rcx          # store 3*c in r12 (callee-saved)
    mov rax, rcx          # rax = d value
    shl rax, 2            # 4*d = d << 2
    sub rbx, rax          # a - 2*b + 3*c - 4*d

    lea rax, [r8 + r8*4]  # 5*e = e + 4*e
    add rbx, rax          # ... + 5*e

    mov rax, r9           # f
    shl rax, 1            # 2*f
    lea rcx, [rax + rax*2]# 6*f = 2*f + 4*f (rax is 2*f)
    sub rbx, rcx          # ... - 6*f

    mov rcx, qword ptr [rsp+24]   # g (7th arg: original rsp+8 + 8 for push)
    lea rdx, [rcx + rcx*2]        # 3*g
    shl rcx, 1                    # 2*g
    add rcx, rdx                  # 5*g = 2*g + 3*g
    add rcx, rdx                  # 7*g = 5*g + 2*g (rcx now holds 7*g)

    mov rdx, qword ptr [rsp+16]   # h (8th arg: original rsp+16)
    shl rdx, 3                    # 8*h = h << 3

    add rbx, rcx                  # ... + 7*g
    sub rbx, rdx                  # ... - 8*h

    mov rax, rbx                  # result to rax

    add rsp, 8                    # restore stack pointer
    pop rbx                       # restore callee-saved rbx
    ret
