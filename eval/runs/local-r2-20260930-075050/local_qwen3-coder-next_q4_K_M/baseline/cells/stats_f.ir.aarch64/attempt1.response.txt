export function l $stats_f(l %a, l %n, l %f, l %out) {
%entry:
	%min =l copy %n
	%max =l copy %n
	%sum =l copy 0
	%count =l copy 0
	%i =l copy 0

@loop:
	%csltl =w csltl %i, %n
	jnz %csltl, @body, @done

	%idx =l mul %i, 8
	%p =l add %a, %idx
	%x =l loadl %p
	%y =l call %f(%x)

	%is_min =w csltl %y, %min
	jnz %is_min, @update_min, @check_max

@update_min:
	storel %y, %out
	%min =l copy %y
	jmp @check_max

@check_max:
	%csgtl =w csgtl %y, %max
	jnz %csgtl, @update_max, @check_odd

@update_max:
	%tmp_out1 =l add %out, 8
	storel %y, %tmp_out1
	%max =l copy %y
	jmp @check_odd

@check_odd:
	%and1 =l and %y, 1
	%is_odd =w ceql %and1, 1
	jnz %is_odd, @odd_inc, @next_i

@odd_inc:
	%count_incr =l add %count, 1
	%count =l copy %count_incr
	jmp @next_i

@next_i:
	%sum_incr =l add %sum, %y
	%sum =l copy %sum_incr
	%i_incr =l add %i, 1
	%i =l copy %i_incr
	jmp @loop

@done:
	%tmp_out2 =l add %out, 16
	storel %sum, %tmp_out2
	ret %count

@body:
	%x =l loadl %p
	%y =l call %f(%x)
	jmp @check_max
}
