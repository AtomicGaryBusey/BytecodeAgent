export function l $max_byte(l %p, l %n) {
@entry
	%zero = w const 0i64
	%n_cmp = w csltl %zero, %n
	jnz %n_cmp, @loop_init, @empty

@loop_init
	%x =l loadub %p
	%p_inc = l add %p, 1i64
	%i =l const 1i64
	%max =l %x
	jmp @check

@loop
	%b =l loadub %p_inc
	%ge = w csgtl %b, %max
	jnz %ge, @update_max
	%next_i =l add %i, 1i64
	%end_cmp = w ceql %next_i, %n
	jnz %end_cmp, @done
	%p_inc = l add %p_inc, 1i64
	%i =l %next_i
	jmp @loop

@update_max
	%max =l %b
	jmp @loop

@empty
	%max =l const 0i64
	jmp @done

@check
	%end_cmp = w ceql %i, %n
	jnz %end_cmp, @done
	jmp @loop

@done
	ret %max
}
