mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    # Save arguments in callee-saved registers where convenient
    mov r12, rdi        # a
    mov r13, rsi        # na
    mov r14, rdx        # b
    mov r15, rcx        # nb
    mov rbp, r8         # r

    xor rax, rax        # will hold number of significant limbs (initially 0)
    test r13, r13
    jz .Lzero_result    # if na == 0, return 0
    test r15, r15
    jz .Lzero_result    # if nb == 0, return 0

    mov rbx, rax        # carry = 0 (rbx)
    xor rdx, rdx        # high part will be used for multiplication

    lea r8, [r13 + r15] # total limbs = na + nb
    dec r8              # last index = na+nb-1
    mov r9, r8          # i+j loop max index (for outer loop)
    xor rsi, rsi        # outer index i = 0

.Louter_loop:
    # Initialize carry and high for this row
    xor rbx, rbx        # carry = 0
    xor rdx, rdx        # hi = 0
    mov rcx, rsi        # j = 0 (inner loop index)

.Linner_loop:
    mov rax, [r12 + rcx*8]   # a[j]
    mov r10, [r14 + rsi*8]   # b[i]
    mulx rax, rdx, r10       # rdx:rax = a[j]*b[i]
    add rax, rbx             # add carry
    adc rdx, 0               # propagate carry to high part

    mov rbx, rdx             # new carry = high part

    # Add product to r[ci] where ci = i+j
    lea r11, [rsi + rcx]
    cmp r11, r8
    ja .Lskip_write          # skip if index out of range

    add [rbp + r11*8], rax   # add low part to r[i+j]
    jc .Lhandle_carry        # if carry in addition, propagate

.Lnext_inner:
    inc rcx
    cmp rcx, r13             # continue while j < na
    jb .Linner_loop
    # End of row: store final carry (it may contribute to next rows)
    # But since we're done with this row, and next row adds to different indices,
    # we only need to check if carry > 0 at end of entire computation.

    inc rsi                  # i++
    cmp rsi, r15             # continue while i < nb
    jb .Louter_loop

    # Now compute significant limbs from top down
.Lfind_sig:
    dec r8                   # start from highest limb (na+nb-1) downwards
    mov rax, [rbp + r8*8]
    test rax, rax
    jnz .Lfound_sig          # found non-zero limb at r8

    test r8, r8
    jnz .Lfind_sig           # continue downward

    xor rax, rax             # all zero -> return 0
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lfound_sig:
    inc r8                   # because r8 points to last non-zero limb
    mov rax, r8              # return na+nb limbs? no: we need count = index+1
    add rax, 1               # count of significant limbs
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lzero_result:
    xor rax, rax             # return 0
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lskip_write:
    inc rcx
    cmp rcx, r13             # continue while j < na
    jb .Linner_loop
    inc rsi                  # i++
    cmp rsi, r15             # continue while i < nb
    jb .Louter_loop
    jmp .Lfind_sig

.Lhandle_carry:
    # carry occurred when adding to r[i+j], so need to propagate to next limb
    lea r10, [rsi + rcx]     # original index i+j
    mov rax, 1               # add 1 to high part
    adc [rbp + r10*8 + 8], rax   # this may overflow again...
    jmp .Lnext_inner
