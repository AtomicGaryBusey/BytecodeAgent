mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov r12, rdi        # a pointer
    mov r13, rsi        # na
    mov r14, rdx        # b pointer
    mov r15, rcx        # nb
    mov rbp, r8         # r pointer

    xor rax, rax        # significant limbs count
    test r13, r13
    jz .Lret_zero
    test r15, r15
    jz .Lret_zero

    # Zero output array first (all na+nb limbs)
    mov rcx, r13
    add rcx, r15        # total limbs = na + nb
    xor rbx, rbx

.Lzero_r:
    cmp rbx, rcx
    jge .Linit_done
    mov qword ptr [rbp + rbx*8], 0
    inc rbx
    jmp .Lzero_r

.Linit_done:

    xor r9, r9          # i = 0 (outer loop index)

.Louter_loop:
    xor rbx, rbx        # carry = 0 for this row
    xor r10, r10        # j = 0 (inner loop index)

.Linner_loop:
    cmp r10, r13        # while j < na
    jge .Lnext_row

    mov rax, [r12 + r10*8]   # a[j]
    mov rdi, [r14 + r9*8]    # b[i]
    mulx rdx, rcx, rdi       # rdx:rcx = a[j]*b[i]

    add rcx, rbx             # add previous carry
    adc rdx, 0               # propagate to high part

    mov rbx, rdx             # new carry

    lea r8, [r9 + r10]       # index i+j
    add qword ptr [rbp + r8*8], rcx   # add low part to result
    jc .Lpropagate_carry     # if carry in addition, propagate

.Linner_inc:
    inc r10
    jmp .Linner_loop

.Lnext_row:
    # Store final carry of this row (if any)
    cmp rbx, 0
    je .Louter_inc

    # Propagate final carry to higher limbs within same row
    lea r8, [r9 + r13]      # first index beyond product of this row (i+na)

.Lcarry_prop_loop:
    add qword ptr [rbp + r8*8], rbx
    jb .Ldone_carry         # no further carry
    mov rbx, 1              # new carry = 1
    inc r8                  # next limb index
    mov rcx, r13
    add rcx, r15            # total limbs = na + nb
    cmp r8, rcx
    jae .Ldone_carry        # if beyond array size, stop propagation
    jmp .Lcarry_prop_loop

.Ldone_carry:
.Louter_inc:
    inc r9                  # i++
    cmp r9, r15             # while i < nb
    jb .Louter_loop

.Lfind_sig:
    mov rcx, r13            # na
    add rcx, r15            # total = na+nb limbs
    dec rcx                 # last index = na+nb-1

.Lsig_loop:
    test rcx, rcx
    js .Lret_zero           # all zero -> return 0 if we went below 0
    mov rax, [rbp + rcx*8]
    test rax, rax
    jnz .Lfound_sig
    dec rcx
    jmp .Lsig_loop

.Lfound_sig:
    inc rcx                 # index+1 = number of significant limbs
    mov rax, rcx
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lret_zero:
    xor rax, rax
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lpropagate_carry:
    # carry from addition to result limb => need to increment next higher limb
    lea r8, [r9 + r10]     # current index = i+j
    inc r8                 # next limb index
    mov rcx, r13
    add rcx, r15           # total limbs = na + nb
    cmp r8, rcx
    jae .Linner_inc        # skip if beyond array size

    mov rbx, 1             # carry for next limb
    add qword ptr [rbp + r8*8], rbx
    jc .Lpropagate_carry   # if that overflowed, propagate further
    jmp .Linner_inc
