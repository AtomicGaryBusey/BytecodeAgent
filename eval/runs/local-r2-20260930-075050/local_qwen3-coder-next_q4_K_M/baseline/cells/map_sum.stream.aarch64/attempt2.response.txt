map_sum:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x8, xzr            // x8 = sum (accumulator)
    cbz x1, .Ldone         // if n == 0, return 0
    mov x9, x0             // x9 = pointer to current element
    mov x10, xzr           // x10 = index counter = 0

.Lloop:
    ldr x11, [x9, x10, lsl #3]   // load a[i]
    blr x2                       // call f(a[i]), result in x0
    add x8, x8, x0               // add result to sum
    add x10, x10, #1             // increment index
    cmp x10, x1                  // compare with n
    b.lt .Lloop                  // loop if index < n

.Ldone:
    mov x0, x8                   // return sum
    ldp x29, x30, [sp], #16      // restore frame pointer and return address
    ret
