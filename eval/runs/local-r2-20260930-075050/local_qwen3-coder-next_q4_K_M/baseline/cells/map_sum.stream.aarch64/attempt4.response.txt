map_sum:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x8, xzr            // x8 = sum (accumulator)
    cbz x1, .Ldone         // if n == 0, return 0
    mov x9, x0             // x9 = pointer to current element

.Lloop:
    ldr x11, [x9], #8      // load *a and increment pointer post-indexed
    blr x2                 // call f(*a), result in x0
    add x8, x8, x0         // add result to sum
    sub x1, x1, #1         // decrement n
    cbnz x1, .Lloop        // loop if n != 0

.Ldone:
    mov x0, x8             // return sum
    ldp x29, x30, [sp], #16
    ret
