madd32:
    mov eax, edi        # low 32 bits of a (32-bit move zero-extends)
    imul eax, esi       # signed multiply by low 32 bits of b; wraps to 32 bits in eax
    add eax, edx        # add low 32 bits of c; 32-bit add with wrap
    cdq                 # sign-extend eax into edx:eax (preparing for correct sign extension)
    ret                 # return with result in rax (sign-extended from 32 to 64 bits)
