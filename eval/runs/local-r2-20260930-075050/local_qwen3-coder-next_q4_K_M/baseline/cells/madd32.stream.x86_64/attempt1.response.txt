madd32:
    mov eax, edi        # load low 32 bits of a into eax
    imul esi            # multiply by low 32 bits of b (signed), result in edx:eax
    add eax, edx        # c's low 32 bits are already in edx (overwrites) -- no, wait
    # Correction: need to extract c's low 32 bits and add them
    # Actually: imul esi overwrites eax with low 32 bits of product (since it's signed 32-bit mul), but we need full 64-bit mul then truncate?
    # Specification says "wrapping 32-bit arithmetic" so a32 * b32 is done in 32-bit, then + c32
    # So: extract low 32 bits of each argument
    mov eax, edi        # a32 = low 32 bits of a
    movsx eax, ax       # sign-extend to 64 bits (or use movzx; but result should be signed)
    mov ebx, esi        # b32 = low 32 bits of b
    cdq                 # sign-extend eax into edx:eax for 64-bit mul? No, better do 32-bit imul directly
    # Actually, simplest: use 32-bit imul which truncates to 32 bits (signed), then add c32
    movsx eax, di       # wrong; we need just low 16/32 bits
    # Simpler: use movzx to zero-extend or mov with implicit zero-extension in 32-bit op
    mov eax, edi        # a32 (high bits auto-zeroed by 32-bit move)
    imul esi            # signed multiply: eax = al * dl (low 16), but with esi as source? No:
    # Actually, `imul r32` with one operand does: dest = dest * src, truncating to 32 bits
    # So we need to load a32 in eax, then multiply by b32 (esi)
    mov eax, edi        # a32
    imul eax, esi       # eax = eax * esi, low 32 bits of product (signed), high bits ignored
    add eax, edx        # no: c not yet loaded
    # Fix: load c's low 32 bits first
    mov ecx, edx        # c32 = low 32 bits of c (argument passed in rdx)
    mov eax, edi        # a32
    imul eax, esi       # a32 * b32, result in eax (32-bit, signed wrap)
    add eax, ecx        # + c32
    cdq                 # sign-extend eax into edx:eax for 64-bit return
    ret                 # return with value in rax (eax zero/signed extend to rax)
