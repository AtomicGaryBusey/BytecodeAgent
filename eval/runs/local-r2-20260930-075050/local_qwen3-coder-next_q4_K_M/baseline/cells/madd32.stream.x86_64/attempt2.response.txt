madd32:
    mov eax, edi        # load low 32 bits of a (32-bit op zeros upper half)
    imul eax, esi       # multiply by low 32 bits of b; result in eax (low 32 bits, signed wrap)
    add eax, edx        # add low 32 bits of c (c passed in rdx, its low 32 bits are in edx)
    cdq                 # sign-extend eax into edx:eax
    ret                 # return with value in rax (sign-extended)
