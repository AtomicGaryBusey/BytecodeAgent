fib:
    cmp x0, #1
    b.le .Lbase
    mov x1, #0          // a = fib(0)
    mov x2, #1          // b = fib(1)
    mov x3, x0          // i = n
.Lloop:
    subs x3, x3, #1     // i--
    b.eq .Ldone         // if (i == 0) break
    add x4, x1, x2      // c = a + b
    mov x1, x2          // a = b
    mov x2, x4          // b = c
    b .Lloop
.Ldone:
    mov x0, x2          // return b
    ret
.Lbase:
    mov x0, x0          // fib(0)=0, fib(1)=1 (x0 unchanged)
    ret
