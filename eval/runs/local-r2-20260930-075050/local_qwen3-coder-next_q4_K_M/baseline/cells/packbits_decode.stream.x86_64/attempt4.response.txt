packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

.Lloop:
    cmp rdi, rsi
    jae .Ldone

    movzx rcx, byte ptr [rdi]
    inc rdi

    cmp cl, 0x80
    jae .Lrepeat

    # Literal packet (h in [0,127])
    lea rdx, [rcx + 1]
    cmp rdi, rsi
    ja .Lerror_no_src

.Lliteral_loop:
    test rdx, rdx
    jz .Lloop
    cmp r13, rcx
    jae .Ldone
    mov al, byte ptr [rdi]
    inc rdi
    dec rdx
    cmp r13, rcx
    jae .Lnext_packet
    mov byte ptr [r12], al
    inc r12
    jmp .Lliteral_loop

.Lrepeat:
    cmp cl, 0x80
    je .Lnop

    # h is negative: compute count = 1 - h (h signed)
    movsx rcx, byte ptr [rdi - 1]
    neg rcx
    lea rdx, [rcx + 1]

.Lrepeat_loop:
    test rdx, rdx
    jz .Lloop
    cmp r13, rcx
    jae .Ldone
    mov al, byte ptr [rdi]
    dec rdx
    cmp r13, rcx
    jae .Lnext_packet
    mov byte ptr [r12], al
    inc r12
    jmp .Lrepeat_loop

.Lnop:
    jmp .Lloop

.Lerror_no_src:
    mov rax, -2
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Ldone:
    mov rax, r12
    sub rax, rcx
    mov rcx, rcx
    cmp rcx, rsi
    ja .Lerror_cap_exceeded
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lnext_packet:
    cmp rcx, rsi
    ja .Lerror_no_src
    jmp .Lloop

.Lerror_cap_exceeded:
    mov rax, -1
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
