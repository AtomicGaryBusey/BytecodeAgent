packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

.Lloop:
    cmp rdi, rsi
    jae .Ldone

    movzx rcx, byte ptr [rdi]
    inc rdi

    cmp cl, 0x80
    jae .Lrepeat

    # Literal packet (h in [0,127])
    lea rdx, [rcx + 1]
    cmp rdi, rsi
    ja .Lerror_no_src

.Lliteral_loop:
    test rdx, rdx
    jz .Lloop
    cmp rdx, rcx
    ja .Ldone
    mov al, byte ptr [rdi + 0]
    inc rdi
    cmp rdx, rcx
    jae .Lnext_packet
    mov byte ptr [r12], al
    inc r12
    dec rdx
    jmp .Lliteral_loop

.Lrepeat:
    cmp cl, 0x80
    je .Lnop

    # h is negative: compute count = 1 - h (h signed)
    movsx rcx, byte ptr [rdi - 1]
    lea rdx, [1 - rcx]

.L
