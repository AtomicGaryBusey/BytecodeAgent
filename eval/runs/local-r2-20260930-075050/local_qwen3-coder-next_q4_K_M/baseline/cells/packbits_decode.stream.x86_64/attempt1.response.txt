packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    # At entry: src=rdi, n=rsi, out=rdx, cap=rcx
    xor rax, rax          # output_bytes_written = 0
    xor rdi, rdi          # src_index = 0 (source index)
    xor rsi, rsi          # dst_index = 0 (destination index)
    mov r8, rdx           # out pointer in r8
    mov r9, rcx           # cap in r9
    test rsi, rsi         # n == 0?
    jz .Lexit             # if no source bytes, done

.Lloop:
    cmp rdi, rsi          # src_index >= n?
    jae .Lerror_no_src    # error: need more src bytes than available
    movzx rbx, byte ptr [rdx + rdi]  # read header byte h (as unsigned)
    inc rdi               # advance past header
    cmp bl, 0x80          # signed comparison: h >= 0?
    ja .Lrepeat           # if h > 127, it's negative -> repeat packet
                        # note: x86 signed comparison uses SF != OF; here we use unsigned to branch

    # Literal packet (h in [0,127])
    test rbx, rbx         # h == 0? then copy count = 1
    jz .Lliteral_one
    lea rcx, [rbx + 1]    # copy_count = h + 1
    mov r10, rcx          # save copy_count

.Lliteral_loop:
    cmp rdi, rsi          # src_index >= n?
    jae .Lerror_no_src
    cmp r9, rsi           # cap check against remaining capacity? no, compare dst_index + count vs cap
    mov r11, rcx          # temp for loop count
.Lliteral_copy:
    cmp r9, rsi           # this is wrong; need to track dst_index and cap
