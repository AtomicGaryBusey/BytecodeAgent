memmove_i64:
    # Handle n == 0 immediately
    test rdx, rdx
    jz .Ldone
    
    # Save callee-saved registers we'll use (rbx, rbp, r12, r13)
    push rbx
    push rbp
    push r12
    push r13
    
    # rdi = dst, rsi = src, rdx = n (count of int64_t elements)
    
    # Check for overlap and direction to decide copy order
    mov rbx, rdi       # save original dst in rbx
    mov rbp, rsi       # save original src in rbp
    cmp rdi, rsi
    jge .Lcopy_forward  # if dst >= src, copy forward (no backward overlap issue)
    
    # dst < src: safe to copy forward (dst[i] before src[i], no overlap or dst starts earlier)
.Lcopy_forward:
    xor rcx, rcx       # i = 0
.Lfwd_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lfwd_loop
    jmp .Lcleanup
    
    # dst > src: need to copy backward to avoid overwriting src[0..n-1]
.Lcopy_backward:
    # We already know dst > src (rbx = dst, rbp = src)
    mov rcx, rdx       # i = n - 1
    dec rcx            # start at last element
    
.Lbwd_loop:
    mov rax, qword ptr [rsi + rcx*8]   # read from src[rcx]
    mov qword ptr [rdi + rcx*8], rax   # write to dst[rcx]
    dec rcx
    jge .Lbwd_loop

.Lcleanup:
    pop r13
    pop r12
    pop rbp
    pop rbx
    
.Ldone:
    xor eax, eax       # return 0
    ret
