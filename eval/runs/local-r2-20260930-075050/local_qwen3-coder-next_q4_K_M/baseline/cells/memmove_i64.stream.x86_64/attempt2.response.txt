memmove_i64:
    test rdx, rdx
    jz .Ldone
    
    # Save callee-saved registers we'll use (rbx, rbp, r12, r13)
    push rbx
    push rbp
    push r12
    push r13
    
    mov rbx, rdi        # original dst
    mov rbp, rsi        # original src
    cmp rdi, rsi
    jle .Lcopy_forward  # if dst <= src, copy forward (including equal case)
    
    # dst > src: need backward copy to avoid overwriting source
    lea rcx, [rsi + rdx*8 - 8]   # src_end = src + (n-1)*8
    lea rsi, [rdi + rdx*8 - 8]   # dst_end = dst + (n-1)*8
    
.Lbwd_loop:
    mov rax, qword ptr [rcx]
    mov qword ptr [rsi], rax
    sub rcx, 8
    sub rsi, 8
    dec rdx
    jnz .Lbwd_loop
    jmp .Lcleanup
    
.Lcopy_forward:
    xor rcx, rcx        # i = 0
.Lfwd_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lfwd_loop
    
.Lcleanup:
    pop r13
    pop r12
    pop rbp
    pop rbx
    
.Ldone:
    xor eax, eax        # return 0
    ret
