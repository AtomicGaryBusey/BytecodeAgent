export function l $map_sum(l %a, l %n, l %f) {
entry:
	%sum =l alloc4i 8
	storel 0, %sum
	%idx =l alloc4i 8
	storel 0, %idx
	jmp loop_check

loop_check:
	%current_idx =l loadl %idx
	%n_val =l loadl %n
	%cmp =w csltl %current_idx, %n_val
	jnz %cmp, body
	jmp done

body:
	%elem_addr =l add l %a, %current_idx
	%val =l loadl %elem_addr
	%r =l call %f(l %val)
	%old_sum =l loadl %sum
	%new_sum =l add l %old_sum, %r
	storel %new_sum, %sum
	%inc_idx =l add l %current_idx, 1
	storel %inc_idx, %idx
	jmp loop_check

done:
	%result =l loadl %sum
	ret %result
}
