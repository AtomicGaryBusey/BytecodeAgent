export function l $map_sum(l %a, l %n, l %f) {
	%sum =l alloc4i 8
	storel 0, %sum
	%i =l alloc4i 8
	storel 0, %i

	jmp @loop_check

@loop_check:
	%current_i =l loadl %i
	%n_val =l loadl %n
	%done =w ceql %current_i, %n_val
	jnz %done, @exit

	%array_addr =l add l %a, 8
	%elem_ptr =l mul l %current_i, 8
	%elem_addr =l add l %a, %elem_ptr
	%val =l loadl %elem_addr
	%f_result =l call %f(l %val)
	%old_sum =l loadl %sum
	%new_sum =l add l %old_sum, %f_result
	storel %new_sum, %sum

	%next_i =l add l %current_i, 1
	storel %next_i, %i

	jmp @loop_check

@exit:
	%result =l loadl %sum
	ret %result
}
