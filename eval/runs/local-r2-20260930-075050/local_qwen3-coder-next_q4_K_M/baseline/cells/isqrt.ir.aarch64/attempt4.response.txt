export function l $isqrt(l %n) {
@start
	l %low
	l %high
	l %mid
	w %cmp
	l %r

	%low = copy 0
	%r = copy 0
	%high = copy %n

@loop
	%cmp = csltl %low, %high
	jnz %cmp, @body
	/* Now low >= high: check if high is valid */
	l %high_sq
	%high_sq = mul %high, %high
	%cmp = culgt %high_sq, %n
	jnz %cmp, @adjust_high
	jmp @done_with_r

@adjust_high
	%high = sub %high, 1
	jmp @done_with_r

@body
	%mid = add %low, %high
	%mid = lshr %mid, 1

	l %quotient
	%cmp = ceql %mid, 0
	jnz %cmp, @mid_is_zero
	/* Avoid division by zero: mid > 0 here */
	%quotient = udiv %n, %mid
	%cmp = csltl %mid, %quotient
	jnz %cmp, @too_high
	jmp @mid_ok

@mid_is_zero
	/* mid == 0, square is 0 <= n (true for all valid n), so update r */
	%r = copy %mid
	%low = add %mid, 1
	jmp @loop

@mid_ok
	%r = copy %mid
	l %next_mid
	%next_mid = add %mid, 1
	/* Check if (mid+1)^2 <= n using udiv to avoid overflow */
	w %next_valid
	%cmp = ceql %next_mid, 0
	jnz %cmp, @skip_next_check
	l %quotient2
	%quotient2 = udiv %n, %next_mid
	%cmp = csltl %next_mid, %quotient2
	jnz %cmp, @can_increase
	jmp @done_with_r

@skip_next_check
	/* mid+1 == 0 (impossible for positive numbers), but handle safely */
	jmp @done_with_r

@too_high
	%high = sub %mid, 1
	jmp @loop

@can_increase
	%low = add %mid, 1
	jmp @loop

@done_with_r
	ret %r
}
