export function l $isqrt(l %n) {
@start
	l %low
	l %high
	l %mid
	l %mid_sq
	l %temp
	w %cmp
	l %r

	%low = copy 0
	%high = copy %n
	%r = copy 0

@loop
	%cmp = csltl %low, %high
	jnz %cmp, @body
	jmp @done

@body
	%mid = add %low, %high
	%mid = lshr %mid, 1
	%temp = mul %mid, %mid
	%cmp = culgt %temp, %n
	jnz %cmp, @too_high
	jmp @update_r

@too_high
	%high = sub %mid, 1
	jmp @loop

@update_r
	%r = copy %mid
	%low = add %mid, 1
	jmp @loop

@done
	ret %r
}
