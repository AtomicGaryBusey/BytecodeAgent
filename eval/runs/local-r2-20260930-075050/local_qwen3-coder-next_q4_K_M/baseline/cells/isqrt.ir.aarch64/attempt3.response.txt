export function l $isqrt(l %n) {
@start
	l %low
	l %high
	l %mid
	l %mid_sq_lo
	w %cmp
	l %r

	%low = copy 0
	%r = copy 0
	%high = copy %n

@loop
	%cmp = csltl %low, %high
	jnz %cmp, @body
	jmp @done

@body
	%mid = add %low, %high
	%mid = lshr %mid, 1

	/* Compute mid*mid */
	%mid_sq_lo = mul %mid, %mid

	/* Check if mid > 0 and mid > n/mid (to avoid overflow in comparison) */
	%cmp = ceql %mid, 0
	jnz %cmp, @check_mid_zero
	l %quotient
	%quotient = udiv %n, %mid
	%cmp = csltl %mid, %quotient
	jnz %cmp, @too_high
	jmp @update_r

@check_mid_zero
	/* mid == 0, square is 0 <= n (always true for n >= 0), so update r */
	%r = copy %mid
	%low = add %mid, 1
	jmp @loop

@update_r
	%r = copy %mid
	%cmp = ceql %mid_sq_lo, %n
	jnz %cmp, @exact_match
	l %next_mid
	%next_mid = add %mid, 1
	%next_mid_sq = mul %next_mid, %next_mid
	/* Check if next_mid*mid <= n by using udiv to avoid overflow */
	w %safe_check
	%safe_check = ceql %next_mid, 0
	jnz %safe_check, @skip_next_check
	l %quotient2
	%quotient2 = udiv %n, %next_mid
	%cmp = csltl %next_mid, %quotient2
	jnz %cmp, @can_increase
	jmp @done_with_r

@exact_match
	jmp @done_with_r

@can_increase
	%low = add %mid, 1
	jmp @loop

@skip_next_check
	jmp @done_with_r

@too_high
	%high = sub %mid, 1
	jmp @loop

@done_with_r
	jmp @done

@done
	ret %r
}
