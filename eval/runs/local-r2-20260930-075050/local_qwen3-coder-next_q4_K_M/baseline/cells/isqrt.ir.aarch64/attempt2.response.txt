export function l $isqrt(l %n) {
@start
	l %low
	l %high
	l %mid
	l %mid_sq_lo
	l %mid_sq_hi
	l %temp_lo
	l %temp_hi
	w %cmp
	l %r

	%low = copy 0
	%high = copy %n
	%r = copy 0

@loop
	%cmp = csltl %low, %high
	jnz %cmp, @body
	jmp @done

@body
	%mid = add %low, %high
	%mid = lshr %mid, 1

	/* Compute mid*mid using 128-bit arithmetic to avoid overflow */
	%mid_sq_lo = mul %mid, %mid
	%mid_sq_hi = umulh %mid, %mid

	/* Compare mid*mid with n: if high part > 0, then too high */
	%cmp = cmplt %mid_sq_hi, 1
	jnz %cmp, @check_low
	jmp @too_high

@check_low
	/* high part is 0, compare low parts */
	%cmp = culgt %mid_sq_lo, %n
	jnz %cmp, @too_high
	jmp @update_r

@too_high
	%high = sub %mid, 1
	jmp @loop

@update_r
	%r = copy %mid
	%low = add %mid, 1
	jmp @loop

@done
	ret %r
}
