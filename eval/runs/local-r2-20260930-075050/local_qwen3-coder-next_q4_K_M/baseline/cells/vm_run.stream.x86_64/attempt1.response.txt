vm_run:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8                    # align stack to 16-byte boundary (rsp was 8 mod 16 before)
    mov rax, 0                    # return value = 0 (default if stack empty)
    xor r12, r12                  # pc register
    xor r13, r13                  # stack depth d
    mov rbp, rdx                  # base pointer for stack array
    mov rbx, rcx                  # cap for bounds check

.Lloop:
    cmp r12, rsi                  # pc == len?
    jge .Ldone                    # if so, exit loop
    mov rax, qword ptr [rdi + r12*8]  # fetch instruction at code[pc]
    add r12, 1                    # advance pc by 1 (will adjust for ops with operands)
    cmp rax, 0
    je .Lop_halt
    cmp rax, 1
    je .Lop_push
    cmp rax, 2
    je .Lop_add
    cmp rax, 3
    je .Lop_sub
    cmp rax, 4
    je .Lop_mul
    cmp rax, 5
    je .Lop_dup
    cmp rax, 6
    je .Lop_swap
    cmp rax, 7
    je .Lop_drop
    cmp rax, 8
    je .Lop_over
    cmp rax, 9
    je .Lop_lt
    cmp rax, 10
    je .Lop_jz
    cmp rax, 11
    je .Lop_jmp
    # default: just advance pc and continue (for invalid opcodes)
    jmp .Lloop

.Lop_halt:
    jmp .Ldone

.Lop_push:
    add r12, 1                    # skip operand word
    mov rax, qword ptr [rdi + r12*8 - 8]  # load operand v (code[pc-1])
    cmp r13, rbx                  # d >= cap?
    jge .Ldone                    # stack overflow guard (shouldn't happen per spec, but safe)
    mov qword ptr [rbp + r13*8], rax
    inc r13                       # increment depth
    jmp .Lloop

.Lop_add:
    cmp r13, 0
    je .Ldone                     # no op if stack empty (per spec stack not underflowing but safe)
    dec r13                       # d--
    mov rbx, qword ptr [rbp + r13*8]  # b = stack[d]
    cmp r13, 0
    je .Ldone                     # one element left? (but spec guarantees valid ops so should have >=2)
    dec r13                       # d--
    mov rcx, qword ptr [rbp + r13*8]  # a = stack[d]
    add rbx, rcx                  # b = a + b
    mov qword ptr [rbp + r13*8], rbx
    inc r13                       # d++
    jmp .Lloop

.Lop_sub:
    cmp r13, 0
    je .Ldone
    dec r13
    mov rbx, qword ptr [rbp + r13*8]  # b
    cmp r13, 0
    je .Ldone
    dec r13
    mov rcx, qword ptr [rbp + r13*8]  # a
    sub rcx, rbx                  # a - b
    mov qword ptr [rbp + r13*8], rcx
    inc r13
    jmp .Lloop

.Lop_mul:
    cmp r13, 0
    je .Ldone
    dec r13
    mov rbx, qword ptr [rbp + r13*8]  # b
    cmp r13, 0
    je .Ldone
    dec r13
    mov rcx, qword ptr [rbp + r13*8]  # a
    imul rbx                      # rdx:rax = a * b (but we need only lower 64 bits)
    mov rax, rbx                  # low 64 bits in rbx? no: imul reg, reg gives result in same reg if 1-operand; use mul
    # correct imul usage for signed multiply:
    # Let's redo: want a * b with overflow semantics (wrapping mod 2^64)
    # Use mul for unsigned? But spec says "signed" comparison in LT, and arithmetic wraps modulo 2^64
    # For addition/subtraction/multiplication wrap, it doesn't matter if signed or unsigned; same bit pattern
    # So use imul with sign extension isn't needed; just want lower 64 bits of product
    # Use imul rbx, rcx, but that's 2-operand: rbx = rbx * rcx? Actually:
    # mul r/m64 uses rax as implicit operand and stores to rdx:rax.
    # So better:
    # mov rax, [d] ; pop a
    # mul qword ptr [stack top-8] ; multiply by b; result in rdx:rax
    # But we need 2 pops first

    # Reconstruct: stack has d entries. We need top two: stack[d-2], stack[d-1]
    # Let's reload d to compute indices:
    mov rcx, r13                  # current depth before pop
    dec rcx                       # d - 1 (index of top)
    mov rbx, qword ptr [rbp + rcx*8]  # b = stack[d-1]
    dec rcx                       # d - 2 (index of next below top)
    mov rax, qword ptr [rbp + rcx*8]  # a = stack[d-2]
    imul rbx                      # rax = a * b (signed/unsigned wrap same mod 2^64 for low 64 bits)
    mov qword ptr [rbp + rcx*8], rax
    mov r13, rcx                  # d = d - 2 + 1 = d - 1
    inc r13                       # d++
    jmp .Lloop

.Lop_dup:
    cmp r13, 0
    je .Ldone                     # dup of empty stack? spec guarantees not underflowing but for safety
    dec r13                       # temporary to access top
    mov rbx, qword ptr [rbp + r13*8]  # a = stack[d-1]
    cmp r13, rbx                  # d >= cap?
    jge .Ldone
    inc r13                       # restore depth
    mov qword ptr [rbp + r13*8], rbx
    inc r13                       # d++
    jmp .Lloop

.Lop_swap:
    cmp r13, 0
    je .Ldone
    dec r13                       # top index = d-1
    cmp r13, 0
    je .Ldone                     # need at least 2 elements
    mov rcx, qword ptr [rbp + (r13-1)*8]  # stack[d-2]
    mov rbx, qword ptr [rbp + r13*8]      # top = stack[d-1]
    mov qword ptr [rbp + (r13-1)*8], rbx
    mov qword ptr [rbp + r13*8], rcx
    jmp .Lloop

.Lop_drop:
    cmp r13, 0
    je .Ldone
    dec r13                       # d--
    mov qword ptr [rbp + r13*8], 0   # zero the dropped slot (spec says stack[d..cap-1] = 0 on return)
    jmp .Lloop

.Lop_over:
    cmp r13, 0
    je .Ldone
    dec r13                       # top index = d - 1
    cmp r13, 0
    je .Ldone                     # need at least 2 entries
    mov rbx, qword ptr [rbp + (r13-1)*8]  # entry below top: stack[d-2]
    cmp r13, rbx                  # d+1 >= cap? (d currently = current depth - 1; after push will be d+1)
    jge .Ldone
    mov qword ptr [rbp + r13*8], rbx  # push copy of stack[d-2]
    inc r13                       # d++
    jmp .Lloop

.Lop_lt:
    cmp r13, 0
    je .Ldone
    dec r13                       # top = d-1
    mov rcx, qword ptr [rbp + r13*8]      # b = stack[d-1]
    dec r13                       # now index = d-2
    mov rbx, qword ptr [rbp + r13*8]      # a = stack[d-2]
    cmp rbx, rcx                  # a < b (signed)
    setl al                       # al = 1 if true else 0
    movzx eax, al                 # zero extend to rax
    mov qword ptr [rbp + r13*8], rax
    inc r13                       # d++
    jmp .Lloop

.Lop_jz:
    add r12, 1                    # skip operand word (pc += 2 total)
    cmp r13, 0
    je .Ldone                     # underflow? spec guarantees not but be safe
    dec r13                       # pop top value
    mov rbx, qword ptr [rbp + r13*8]      # v = stack[d]
    test rbx, rbx                 # if (v == 0)
    jne .Lloop                    # if nonzero, skip target and continue normal flow
    # else: pc = operand
    mov rax, qword ptr [rdi + r12*8 - 16] # code[pc-2] because we advanced twice: pc was incremented once at top, then +=2 here means we fetch from current pc+0 (which is code[pc_prev]) but need to use operand stored after the JZ opcode
    mov r12, rax                  # set pc = target
    jmp .Lloop

.Lop_jmp:
    add r12, 1                    # skip operand word (pc += 2 total)
    mov rax, qword ptr [rdi + r12*8 - 16] # fetch operand at code[pc-2]
    mov r12, rax                  # pc = target
    jmp .Lloop

.Ldone:
    xor rbx, rbx                  # zero out stack[0..d-1]? no: spec says stack[0..d-1] remain, and d..cap-1 zeroed
    # We have r13 = final depth d; need to zero from d to cap-1
    mov rcx, r13                  # i = 0 (will compare to rbx=cap later)
    mov rdx, rbx                  # but we need cap in a register: use rbp? no, rbp holds stack base. Use r9?
    # Actually: original args: rdi=code, rsi=len, rdx=stack, rcx=cap; so cap is rcx
    mov rbx, rcx                  # save cap in rbx (callee-saved) for cleanup loop
.Lclear:
    cmp r13, rbx                  # i < cap?
    jge .Lcleanup_done
    test rbx, rbx                 # if cap == 0 skip
    je .Lcleanup_done
    mov qword ptr [rdx + r13*8], 0   # stack[i] = 0 for i >= d
    inc r13                       # i++
    jmp .Lclear

.Lcleanup_done:
    test r13, r13                 # if d > 0 get top element
    je .Lfinalize                 # stack empty -> return 0
    dec r13                       # top index = d - 1
    mov rax, qword ptr [rbp + r13*8]   # return stack[d-1]
    jmp .Lfinalize

.Lfinalize:
    add rsp, 8                    # undo alignment adjustment (rsp was aligned before function)
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
