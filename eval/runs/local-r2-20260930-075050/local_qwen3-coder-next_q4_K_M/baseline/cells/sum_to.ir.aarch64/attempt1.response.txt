export function l $sum_to(l %n) {
@start
	w %cond
	l %result
	l %i
	l %tmp
	ceql %cond, %n, 0
	jnz %done, %cond
	ceql %cond, %n, -1
	jnz %done, %cond
	movl %result, 0
	jmp %end
@loop_start
	ceql %cond, %i, %n
	jnz %end, %cond
	addl %tmp, %result, %i
	movl %result, %tmp
	addl %i, %i, 1
	jmp %loop_start
@end
	ret %result
@done
	movl %result, 0
	jmp %end
}
