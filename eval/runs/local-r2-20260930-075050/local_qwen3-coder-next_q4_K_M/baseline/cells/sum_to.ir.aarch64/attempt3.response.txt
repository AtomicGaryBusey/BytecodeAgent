export function l $sum_to(l %n) {
@start
	l %result
	w %cond
	ceql %cond, %n, 0
	jnz @zero_or_negative, %cond
	csltl %cond, %n, 0
	jnz @zero_or_negative, %cond
	addl %result, %n, 1
	mull %result, %n, %result
	s rall, %result, 2
	ret %result
@loop_start
	l %i
	movl %i, 1
	addl %result, 0, %n
@body
	subl %cond, %i, %result
	csltl %cond, %cond, 0
	jnz @end, %cond
	addl %result, %result, %i
	addl %i, %i, 1
	jmp @body
@end
	ret %result
@zero_or_negative
	l %res_zero
	movl %res_zero, 0
	ret %res_zero
}
