export function l $sum_to(l %n) {
@start
	w %is_zero_or_negative
	l %result
	l %i
	w %cond
	ceql %is_zero_or_negative, %n, 0
	jnz @done, %is_zero_or_negative
	addl %result, %n, 1
	mull %result, %n, %result
	s rall, %result, 2
	ret %result
@loop_start
	%cond = csltl %i, %n
	jnz @end, %cond
	addl %result, %result, %i
	addl %i, %i, 1
	jmp @loop_start
@end
	ret %result
@done
	movl %result, 0
	ret %result
}
