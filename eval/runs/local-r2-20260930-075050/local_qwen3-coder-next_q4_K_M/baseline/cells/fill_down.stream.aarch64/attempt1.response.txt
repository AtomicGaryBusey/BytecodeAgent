fill_down:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x4, x0          // x4 = a (base address)
    mov x5, x2          // x5 = start
    eor x6, x6, x6      // i = 0
.Lloop:
    cmp x6, x1          // compare i with n
    b.ge .Ldone         // if i >= n, exit loop
    sub x7, x5, x6      // start - i
    str x7, [x4, x6, lsl #3]  // a[i] = start - i
    add x6, x6, #1      // i++
    b .Lloop
.Ldone:
    mov x0, x1          // return n
    ldp x29, x30, [sp], #16
    ret
