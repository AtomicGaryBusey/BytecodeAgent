fib_rec:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cmp x0, #1
    ble .Lbase_case
    // n >= 2: compute fib(n-2) and fib(n-1)
    stp x19, x20, [sp, #-16]!
    sub sp, sp, #32           // allocate space for original n (16), result of fib(n-2) (16)
    str x0, [sp, #16]         // save original n at offset 16 from current sp
    mov x29, sp               // update frame pointer to new sp
    sub x0, x0, #2            // n-2 for first call
    bl fib_rec                // fib(n-2)
    mov x19, x0               // save fib(n-2) in callee-saved register
    ldr x0, [sp, #16]         // restore original n
    sub x0, x0, #1            // n-1 for second call
    bl fib_rec                // fib(n-1)
    add x0, x0, x19           // fib(n-1) + fib(n-2)
    ldp x19, x20, [sp], #16   // restore x19, x20 and pop 16 bytes
    add sp, sp, #16           // pop space for saved n (16 more bytes)
    b .Ldone

.Lbase_case:
    // n is 0 or 1; result already in x0
    ldp x29, x30, [sp], #16
    ret

.Ldone:
    ldp x29, x30, [sp], #16
    ret
