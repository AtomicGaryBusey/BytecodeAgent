fib_rec:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // if (n < 2) return n;
    cmp x0, #1
    ble .Lbase_case
    // Save callee-saved registers we'll use (x19, x20)
    stp x19, x20, [sp, #-16]!
    sub sp, sp, #32       // allocate space for original n and result of fib(n-2)
    str x0, [sp, #16]     // save original n
    sub x0, x0, #2        // prepare n-2
    bl fib_rec            // call fib_rec(n-2)
    mov x19, x0           // save fib(n-2) in x19
    ldr x0, [sp, #16]     // restore original n
    sub x0, x0, #1        // prepare n-1
    bl fib_rec            // call fib_rec(n-1)
    add x0, x0, x19       // fib(n-1) + fib(n-2)
    ldp x19, x20, [sp], #16  // restore x19, x20 and adjust sp
    b .Ldone

.Lbase_case:
    // n is 0 or 1, return n (already in x0)
    add sp, sp, #16       // unwind frame for base case without callee-saved saves

.Ldone:
    ldp x29, x30, [sp], #16
    ret
