bubble_sort:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    xor rax, rax
    test rsi, rsi
    jle .Ldone
    dec rsi
    jz .Ldone
.Louter_loop:
    mov rbx, 0
    mov r12, rsi
.Linner_loop:
    cmp rbx, r12
    jge .Louter_end
    mov r13, qword ptr [rdi + rbx*8]
    mov r14, qword ptr [rdi + rbx*8 + 8]
    cmp r13, r14
    jle .Lno_swap
    mov qword ptr [rdi + rbx*8], r14
    mov qword ptr [rdi + rbx*8 + 8], r13
    mov r15, 1
.Lno_swap:
    inc rbx
    jmp .Linner_loop
.Louter_end:
    test r15, r15
    jz .Ldone
    dec rsi
    cmp rsi, 0
    jg .Louter_loop
.Ldone:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
