bubble_sort:
    mov rax, 0
    test rsi, rsi
    jle .Ldone
    dec rsi
    jz .Ldone
.Louter_loop:
    push rbp
    mov rbp, rdi
    xor rbx, rbx
    lea r12, [rsi - 1]
.Linner_loop:
    cmp rbx, r12
    jg .Louter_end
    mov r13, qword ptr [rbp + rbx*8]
    mov r14, qword ptr [rbp + rbx*8 + 8]
    cmp r13, r14
    jle .Lno_swap
    mov qword ptr [rbp + rbx*8], r14
    mov qword ptr [rbp + rbx*8 + 8], r13
    mov rbx, 1
.Lno_swap:
    inc rbx
    jmp .Linner_loop
.Louter_end:
    pop rbp
    test rbx, rbx
    jz .Ldone
    dec rsi
    jmp .Louter_loop
.Ldone:
    ret
