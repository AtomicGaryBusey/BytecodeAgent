bubble_sort:
    cbz x1, .Ldone
    mov x2, x1          // outer loop counter = n-1
.Louter_loop:
    cmp x2, #0
    b.eq .Ldone
    add x3, x0, x2, lsl #3   // x3 points to end of unsorted portion
    sub x3, x3, #8           // x3 points to second-to-last element
    mov x4, x2               // inner loop counter = n-1
.Linner_loop:
    ldr x5, [x3]
    ldr x6, [x3, #8]
    cmp x5, x6
    b.ge .Lno_swap
    str x6, [x3]
    str x5, [x3, #8]
.Lno_swap:
    sub x3, x3, #8           // move backwards
    sub x4, x4, #1
    cbnz x4, .Linner_loop
    sub x2, x2, #1
    b .Louter_loop
.Ldone:
    mov x0, #0
    ret
