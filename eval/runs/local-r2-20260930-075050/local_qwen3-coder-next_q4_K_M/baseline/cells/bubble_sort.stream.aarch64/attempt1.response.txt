bubble_sort:
    cbz x1, .Ldone
    mov x2, x1
    sub x2, x2, #1
.Louter_loop:
    cmp x2, #0
    beq .Ldone
    mov x3, x0
    mov x4, x2
.Linner_loop:
    ldr x5, [x3]
    ldr x6, [x3, #8]
    cmp x5, x6
    b.ge .Lno_swap
    str x6, [x3]
    str x5, [x3, #8]
.Lno_swap:
    add x3, x3, #8
    sub x4, x4, #1
    cbnz x4, .Linner_loop
    sub x2, x2, #1
    b .Louter_loop
.Ldone:
    mov x0, #0
    ret
