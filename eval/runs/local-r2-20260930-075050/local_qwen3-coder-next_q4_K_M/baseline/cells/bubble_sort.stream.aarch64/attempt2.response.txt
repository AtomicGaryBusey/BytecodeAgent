bubble_sort:
    cbz x1, .Ldone
    mov x2, x0
    sub x2, x2, #8
    add x2, x2, x1, lsl #3
.Louter_loop:
    cmp x1, #1
    b.le .Ldone
    mov x3, x0
    mov x4, x1
    sub x4, x4, #1
.Linner_loop:
    ldr x5, [x3]
    ldr x6, [x3, #8]
    cmp x5, x6
    b.ge .Lno_swap
    str x6, [x3]
    str x5, [x3, #8]
.Lno_swap:
    add x3, x3, #8
    sub x4, x4, #1
    cbnz x4, .Linner_loop
    sub x1, x1, #1
    b .Louter_loop
.Ldone:
    mov x0, #0
    ret
