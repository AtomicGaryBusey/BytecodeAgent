export function l $bubble_sort(l %a, l %n) {
	%outer_count = copy 0
	j @outer_check

@outer_loop:
	%inner_limit = sub %n, %outer_count
	%j = copy 0
	j @inner_check

@inner_loop:
	%j_index = add %a, %j
	%j_offset = lshl %j, 3
	%current_addr = add %a, %j_offset
	%next_addr = add %current_addr, 8
	%current_val = loadl %current_addr
	%next_val = loadl %next_addr
	csltl %next_val, %current_val
	jnz @should_swap
	j @inner_continue

@should_swap:
	storel %next_val, %current_addr
	storel %current_val, %next_addr

@inner_continue:
	%j_next = add %j, 1
	%inner_check_val = sub %inner_limit, 1
	csltl %j_next, %inner_check_val
	jnz @inner_loop

@outer_increment:
	%outer_count_next = add %outer_count, 1
	%outer_count = copy %outer_count_next
@outer_check:
	csltl %outer_count, %n
	jnz @outer_loop

	%ret = copy 0
	copyl l 0
}
