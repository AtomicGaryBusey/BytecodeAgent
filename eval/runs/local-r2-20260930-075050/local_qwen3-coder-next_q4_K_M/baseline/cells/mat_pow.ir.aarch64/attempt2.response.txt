export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
l %size, %i, %j, %t, %idx_in, %idx_out, %val1, %val2, %acc, %base_idx, %res_idx
w %zero_w, %one_w, %k_zero, %k_odd

@entry
	%size = mul l %n, %n
	%zero_w = csltl w 0, 1
	%one_w = csltl w 1, 0

	// Handle k == 0: set out to identity matrix
	%k_zero = ceql w %k, 0
	jnz %k_zero, @init_identity

	// Initialize result as zero matrix in out
	%i = mul l 0, %n
@store_loop:
	%j = mul l 0, %n
@store_row:
	%idx_out = mul l %i, %n
	%idx_out = add l %idx_out, %j
	storel %zero_w, %out+8*%idx_out
	add %j, 1
	cmp %j, %n
	csltl w %j, %n
	jnz @store_row
	add %i, 1
	cmp %i, %n
	csltl w %i, %n
	jnz @store_loop

	// Set out[i][i] = 1 for identity
	%i = mul l 0, %n
@set_id_diag:
	%res_idx = mul l %i, %n
	%res_idx = add l %res_idx, %i
	storel %one_w, %out+8*%res_idx
	add %i, 1
	cmp %i, %n
	csltl w %i, %n
	jnz @set_id_diag

	// Copy base matrix a into first half of tmp (tmp + 0)
@store_base:
	%idx_in = sub l %size, 1
@copy_a_loop:
	%val1 = loadl l %a+8*%idx_in
	storel %val1, %tmp+8*%idx_in
	sub %idx_in, 1
	cmp %idx_in, -1
	jnz @copy_a_loop

	// Main exponentiation loop (square-and-multiply)
@exp_start:
	csltl w 0, %k
	jz @done

@mat_pow_loop:
	// Check if k is odd
	%k_odd = and l %k, 1
	csltl w 1, %k_odd
	jnz @multiply

	// Square the base: tmp[0..size-1] = tmp * tmp
	call @matrix_mult_base_sq

	// Copy squared result back to tmp (overwrite same region)
	%copy_idx = sub l %size, 1
@copy_sq_to_base:
	%val2 = loadl l %tmp+8*%copy_idx
	storel %val2, %tmp+8*%copy_idx
	sub %copy_idx, 1
	cmp %copy_idx, -1
	jnz @copy_sq_to_base

	// k = k / 2 (right shift)
	%k = srl l %k, 1
	jmp @mat_pow_loop

@multiply:
	// Multiply result (out) by base (tmp), store into tmp2 (tmp + size)
	call @matrix_mult

	// Copy product from tmp+size to out
	%copy_idx = sub l %size, 1
@copy_prod_to_out:
	%val2 = loadl l %tmp+8*%size+8*%copy_idx
	storel %val2, %out+8*%copy_idx
	sub %copy_idx, 1
	cmp %copy_idx, -1
	jnz @copy_prod_to_out

	// Square the base: tmp[0..size-1] = tmp * tmp
	call @matrix_mult_base_sq

	// Copy squared result back to tmp (overwrite same region)
	%copy_idx = sub l %size, 1
@copy_sq_to_base2:
	%val2 = loadl l %tmp+8*%copy_idx
	storel %val2, %tmp+8*%copy_idx
	sub %copy_idx, 1
	cmp %copy_idx, -1
	jnz @copy_sq_to_base2

	// k = k / 2 (right shift)
	%k = srl l %k, 1
	csltl w 0, %k
	jnz @mat_pow_loop

@done:
	// Clear tmp arrays as required: set all 2*n^2 elements of tmp to zero
	%total_elems = mul l 2, %size
	%clear_idx = sub l %total_elems, 1
@clear_tmp:
	storel %zero_w, %tmp+8*%clear_idx
	sub %clear_idx, 1
	cmp %clear_idx, -1
	jnz @clear_tmp

	lret l 0

@init_identity:
	// Initialize result as identity matrix in out (handled above at start)
	jmp @done

@matrix_mult_base_sq:
	// Square base matrix stored in tmp[0..size-1], output to tmp[size..2*size-1]
	%outer_i = mul l 0, %n
@sq_loop_i:
	%inner_j = mul l 0, %n
@sq_loop_j:
	%acc = mul l 0, 0 // zero accumulator
	%inner_t = mul l 0, %n
@sq_loop_t:
	// Load A[outer_i][inner_t]
	%idx_row_t = add l %outer_i, %inner_t
	%val1 = loadl l %tmp+8*%idx_row_t

	// Load A[inner_t][inner_j]
	%idx_col_t = mul l %inner_t, %n
	%idx_col_t = add l %idx_col_t, %inner_j
	%val2 = loadl l %tmp+8*%idx_col_t

	%prod = mull l %val1, %val2
	%acc = add l %acc, %prod

	add %inner_t, 1
	cmp %inner_t, %n
	csltl w %inner_t, %n
	jnz @sq_loop_t

	// Store result to tmp[size + outer_i * n + inner_j]
	%res_idx = mul l %outer_i, %n
	%res_idx = add l %res_idx, %inner_j
	storel %acc, %tmp+8*%size+8*%res_idx

	add %inner_j, 1
	cmp %inner_j, %n
	csltl w %inner_j, %n
	jnz @sq_loop_j

	add %outer_i, 1
	cmp %outer_i, %n
	csltl w %outer_i, %n
	jnz @sq_loop_i

	lret l 0

@matrix_mult:
	// Multiply out * tmp (base), store to tmp+size
	%outer_i = mul l 0, %n
@mult_loop_i:
	%inner_j = mul l 0, %n
@mult_loop_j:
	%acc = mul l 0, 0 // zero accumulator
	%inner_t = mul l 0, %n
@mult_loop_t:
	// Load out[outer_i][inner_t]
	%idx_row_t = add l %outer_i, %inner_t
	%val1 = loadl l %out+8*%idx_row_t

	// Load tmp[inner_t][inner_j]
	%idx_col_t = mul l %inner_t, %n
	%idx_col_t = add l %idx_col_t, %inner_j
	%val2 = loadl l %tmp+8*%idx_col_t

	%prod = mull l %val1, %val2
	%acc = add l %acc, %prod

	add %inner_t, 1
	cmp %inner_t, %n
	csltl w %inner_t, %n
	jnz @mult_loop_t

	// Store result to tmp[size + outer_i * n + inner_j]
	%res_idx = mul l %outer_i, %n
	%res_idx = add l %res_idx, %inner_j
	storel %acc, %tmp+8*%size+8*%res_idx

	add %inner_j, 1
	cmp %inner_j, %n
	csltl w %inner_j, %n
	jnz @mult_loop_j

	add %outer_i, 1
	cmp %outer_i, %n
	csltl w %outer_i, %n
	jnz @mult_loop_i

	lret l 0
}
