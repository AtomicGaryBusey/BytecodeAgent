mp_mul:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Check for zero-length operands first
    cmp x1, #0
    cbz x1, .Lret_zero
    cmp x3, #0
    cbz x3, .Lret_zero

    // Calculate total limbs = na + nb
    add x25, x1, x3

    // Save callee-saved registers we'll use: x19-x24
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x19, x0        // a pointer
    mov x20, x1        // na
    mov x21, x2        // b pointer
    mov x22, x3        // nb
    mov x23, x4        // r pointer

    // i = 0 (result limb index)
    mov x8, #0
    mov x9, xzr        // carry-in for next limb (high part)

.Louter_loop:
    cmp x8, x25
    b.ge .Lnormalize_start

    // Clear partial sum registers
    mov x10, #0        // low part of partial sum
    mov x11, #0        // high part of partial sum

    // j from max(0, i-nb+1) to min(i, na-1)
    sub x7, x8, x22       // i - nb
    add x7, x7, #1        // i - nb + 1
    csel x6, xzr, x7, gt  // lower bound = max(0, i-nb+1)
    mov x7, x8            // upper bound candidate: min(i, na-1) => j <= min(i, na-1)
    cmp x7, x20
    csel x7, x20, x7, lt  // upper = min(i+1, na) but we'll use j < upper so it's correct

.Linner_loop:
    cmp x6, x7            // j < upper?
    b.ge .Linner_done

    sub x99, x8, x6       // k = i - j
    cmp
