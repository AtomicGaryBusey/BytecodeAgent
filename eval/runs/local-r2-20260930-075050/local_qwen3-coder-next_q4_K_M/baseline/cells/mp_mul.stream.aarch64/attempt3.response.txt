mp_mul:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Check for zero-length operands first to avoid null pointer dereference
    cmp x1, #0
    cbz x1, .Lret_zero
    cmp x3, #0
    cbz x3, .Lret_zero

    // Save callee-saved registers we'll use: x19-x24
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x9, x0        // a pointer
    mov x10, x1       // na
    mov x11, x2       // b pointer
    mov x12, x3       // nb
    mov x13, x4       // r pointer

    add x25, x10, x3  // total limbs = na + nb
    mov x14, #0       // i = result limb index
    mov x21, xzr      // carry-in for next limb (high part)

.Louter_loop:
    cmp x14, x25
    b.ge .Lnormalize_start

    mov x15, #0       // partial sum low
    mov x16, #0       // partial sum high

    // Compute j range: j from max(0, i-nb+1) to min(i, na-1)
    sub x7, x14, x12      // i - nb
    add x7, x7, #1        // i - nb + 1
    csel x8, xzr, x7, gt  // lower = max(0, i-nb+1)
    mov x6, x14           // upper_bound candidate: min(i+1, na) => j < min(i+1, na)
    cmp x6, x10
    csel x6, x10, x6, lt  // upper = min(i+1, na)

.Linner_loop:
    cmp x8, x6            // j < upper?
    b.ge .Linner_done

    sub x7, x14, x8       // k = i - j
    cmp x7, x12           // k < nb?
    b.ge .Lnext_j

    ldr x0, [x9, x8, lsl #3]   // a[j]
    ldr x6, [x11, x7, lsl #3]  // b[k]

    umulh x20, x0, x6         // high = (a[j]*b[k]) >> 64
    mul x25, x0, x6           // low = a[j]*b[k]

    adds x15, x15, x25        // add to partial sum low
    adc x16, x16, x20         // add high + carry from previous

.Lnext_j:
    add x8, x8, #1
    b .Linner_loop

.Linner_done:
    adds x15, x15, x21        // add previous carry-in
    adc x16, x16, xzr         // propagate

    str x15, [x13, x14, lsl #3]  // store r[i]
    mov x21, x16              // carry for next iteration

    add x14, x14, #1
    b .Louter_loop

.Lnormalize_start:
    sub x8, x25, #1           // start from last limb index (na+nb-1)
    cbz x25, .Lret_zero       // if total limbs == 0, return 0

.Lscan:
    ldr x9, [x13, x8, lsl #3]
    cbnz x9, .Lfound_significant
    subs x8, x8, #1
    b.ge .Lscan
    mov x0, #0                // all zero limbs
    b .Ldone

.Lfound_significant:
    add x0, x8, #1            // significant limbs = index + 1

.Ldone:
    ldp x23, x24, [sp], #16   // restore callee-saved
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

.Lret_zero:
    mov x0, #0
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
