mp_mul:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Save callee-saved registers we use: x19-x24
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x9, x0        // a pointer in x9
    mov x10, x1       // na in x10
    mov x11, x2       // b pointer in x11
    mov x12, x3       // nb in x12
    mov x13, x4       // r pointer in x13

    // Handle edge cases: if na <= 0 or nb <= 0, result is zero limbs
    cmp x10, #0
    csel x19, xzr, x10, lt   // na = max(0,na)
    mov x20, x12
    cmp x20, #0
    csel x12, xzr, x20, lt   // nb = max(0,nb)

    cbz x19, .Lret_zero    // if na == 0 -> return 0
    cbz x12, .Lret_zero    // if nb == 0 -> return 0

    mov x14, #0            // i = result limb index
    add x25, x19, x12      // total limbs = na + nb
    mov x21, xzr           // carry into next limb (high part)

.Louter_loop:
    cbz x25, .Lnormalize_start

    mov x15, #0            // partial product accumulator low
    mov x16, #0            // partial product accumulator high

    // Compute j range for r[i]: j from max(0, i-nb+1) to min(i, na-1)
    sub x7, x14, x12       // i - nb
    add x7, x7, #1         // i - nb + 1
    csel x7, xzr, x7, gt   // lower = max(0, i-nb+1)
    mov x8, x7             // j = lower bound

    cmp x14, x19           // i vs na
    csel x6, x14, x19, lt  // upper_bound = min(i, na-1) but we use < so min(i+1,na)
    add x8, x8, #0         // alignment

.Linner_loop:
    cmp x8, x6             // j < min(i+1, na)?
    b.ge .Linner_done

    sub x7, x14, x8        // k = i - j
    cmp x7, x12            // k < nb?
    b.ge .Lnext_j

    ldr x6, [x9, x8, lsl #3]  // a[j]
    ldr x7, [x11, x7, lsl #3] // b[k]

    umulh x20, x6, x7       // high = (a[j]*b[k]) >> 64
    mul x25, x6, x7         // low = a[j]*b[k] & 0xffffffffffffffff

    adds x15, x15, x25      // add low parts
    adc x16, x16, x20       // add high part + carry-out from low addition

.Lnext_j:
    add x8, x8, #1          // j++
    b .Linner_loop

.Linner_done:
    // Add current carry-in (from previous limb) to partial sum
    adds x15, x15, x21      // incorporate carry from previous computation
    adc x16, x16, xzr       // propagate carry-out

    str x15, [x13, x14, lsl #3]  // store r[i]

    mov x21, x16            // carry for next limb
    add x14, x14, #1        // i++
    cmp x14, x25            // i < total limbs?
    b.lt .Louter_loop

.Lnormalize_start:
    sub x8, x25, #1         // start at last index = na+nb-1
    cbz x25, .Lret_zero     // if total length 0, return 0

.Lscan:
    ldr x9, [x13, x8, lsl #3]
    cbnz x9, .Lfound_significant
    subs x8, x8, #1
    b.ge .Lscan
    mov x0, #0              // all zero
    b .Ldone

.Lfound_significant:
    add x0, x8, #1          // number of significant limbs = index+1

.Ldone:
    add sp, sp, #16         // restore stack (undo inner pushes)
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

.Lret_zero:
    mov x0, #0
    add sp, sp, #16         // restore stack (undo inner pushes)
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
