export function l $qsort_pairs(l %a, l %n) {
%entry:
	%0 =w ceql %n, 0
	jnz %0, %exit
	%1 =l call $memcpy(l %dst, l %src, l %n)
	storel %0, (l)%0
	ret %0

@quick_sort:
	%2 =w ceql %n, 0
	jnz %2, %qs_exit
	%3 =w ceql %n, 1
	jnz %3, %qs_done_single
	%low_ptr = l %a
	%high_ptr =l addl %a, l %n << 4
	%high_ptr_dec =l subl %high_ptr, 16
	storel %low_ptr, l %i
	storel %high_ptr_dec, l %j
	basic %pivot_sel
@pivot_sel:
	%mid_idx =l shrl %n, 1
	%mid_off =lshl %mid_idx, 4
	%mid_ptr =l addl %a, %mid_off
	%pm0 = loadl l %mid_ptr + 0
	%pm1 = loadl l %mid_ptr + 8
	storel %pm0, l %pivot_key
	storel %pm1, l %pivot_val
	basic %loop_start

@loop_start:
	%li = loadl l %i
	%left_key = loadl l %li + 0
	%pkey = loadl l %pivot_key
	%cmp_lt =w csltl %left_key, %pkey
	jnz %cmp_lt, %inc_left1
	%cmp_eq =w ceql %left_key, %pkey
	jnz %cmp_eq, %check_val1
	jmp %check_right_side1

@check_val1:
	%left_val = loadl l %li + 8
	%pval = loadl l %pivot_val
	%cmp_lt2 =w csltl %left_val, %pval
	jnz %cmp_lt2, %inc_left1
	jmp %check_right_side1

@inc_left1:
	%li_next =l addl %li, 16
	storel %li_next, l %i
	basic %loop_start

@check_right_side1:
	%rj = loadl l %j
	%right_key = loadl l %rj + 0
	%pkey2 = loadl l %pivot_key
	%cmp_gt =w csgtl %right_key, %pkey2
	jnz %cmp_gt, %dec_right1
	%cmp_eq_r =w ceql %right_key, %pkey2
	jnz %cmp_eq_r, %check_val_r1
	jmp %ij_check1

@check_val_r1:
	%right_val = loadl l %rj + 8
	%pval2 = loadl l %pivot_val
	%cmp_gt2 =w csgtl %right_val, %pval2
	jnz %cmp_gt2, %dec_right1
	jmp %ij_check1

@dec_right1:
	%rj_prev =l subl %rj, 16
	storel %rj_prev, l %j
	basic %check_right_side1

@ij_check1:
	%li2 = loadl l %i
	%rj2 = loadl l %j
	%cmp_ij =w csgtl %li2, %rj2
	jnz %cmp_ij, %scan_done1
	%tmp_k0 = loadl l %li2 + 0
	%tmp_v0 = loadl l %li2 + 8
	%tmp_k1 = loadl l %rj2 + 0
	%tmp_v1 = loadl l %rj2 + 8
	storel %tmp_k0, l %rj2 + 0
	storel %tmp_v0, l %rj2 + 8
	storel %tmp_k1, l %li2 + 0
	storel %tmp_v1, l %li2 + 8
	%li3 = loadl l %i
	%rj3 = loadl l %j
	%ni =l addl %li3, 16
	storel %ni, l %i
	%nj =l subl %rj3, 16
	storel %nj, l %j
	basic %loop_start

@scan_done1:
	%ri = loadl l %a + 0
	%ri_val = loadl l %a + 8
	%j_ptr = loadl l %j
	%ji_key = loadl l %j_ptr + 0
	%ji_val = loadl l %j_ptr + 8
	storel %ji_key, l %a + 0
	storel %ji_val, l %a + 8
	storel %ri, l %j_ptr + 0
	storel %ri_val, l %j_ptr + 8
	%j_off = loadl l %j
	%left_n =l subl %j_off, %a
	%left_cnt =l shrl %left_n, 4
	basic %rec_left
@rec_left:
	%r1 =l call $qsort_pairs(l %a, l %left_cnt)
	%j_ptr2 = loadl l %j
	%right_off =l addl %j_ptr2, 16
	%right_n =l subl %n, %left_cnt
	%right_cnt =l subl %right_n, 1
	basic %rec_right
@rec_right:
	%r2 =l call $qsort_pairs(l %right_off, l %right_cnt)
	basic %qs_exit

@qs_done_single:
	basic %qs_exit
@qs_exit:
	ret %0
}
