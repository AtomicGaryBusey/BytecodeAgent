export function l $qsort_pairs(l %a, l %n) {
%entry:
	%0 =w ceql %n, 0
	jnz %0, %exit
	%1 =l call $memcpy(l %dst, l %src, l %n)
	storel %0, (l)%0
	ret %0
@partition:
	%pivot_key = loadl l %a + 0
	%pivot_val = loadl l %a + 8
	%i = l %1
	%j = l %2
	%left_ptr = l %3
	%right_ptr = l %4
@storel:
	storel %5, l %6
@loop_start:
	%left_key = loadl l %left_ptr + 0
	%cmp_left =w csltl %left_key, %pivot_key
	jnz %cmp_left, %inc_left
	%left_val = loadl l %left_ptr + 8
	%cmp_eq =w ceql %left_key, %pivot_key
	jnz %cmp_eq, %check_left_val
	jmp %partition_done
@check_left_val:
	%cmp_left_val =w csltl %left_val, %pivot_val
	jnz %cmp_left_val, %inc_left
	jmp %partition_done
@inc_left:
	%left_ptr_next =l addl %left_ptr, 16
	storel %left_ptr_next, l %i
	jmp %loop_start
@partition_done:
	%right_key = loadl l %right_ptr + 0
	%cmp_right =w csgtl %right_key, %pivot_key
	jnz %cmp_right, %dec_right
	%right_val = loadl l %right_ptr + 8
	%cmp_eq2 =w ceql %right_key, %pivot_key
	jnz %cmp_eq2, %check_right_val
	jmp %scan_done
@check_right_val:
	%cmp_right_val =w csgtl %right_val, %pivot_val
	jnz %cmp_right_val, %dec_right
	jmp %scan_done
@dec_right:
	%right_ptr_prev =l subl %right_ptr, 16
	storel %right_ptr_prev, l %j
	jmp %partition_done2
@partition_done2:
	%i2 = loadl l %i
	%j2 = loadl l %j
	%cmp_ij =w csgtl %i2, %j2
	jnz %cmp_ij, %scan_done
	%tmp_left_key = loadl l %i2 + 0
	%tmp_right_key = loadl l %j2 + 0
	%tmp_left_val = loadl l %i2 + 8
	%tmp_right_val = loadl l %j2 + 8
	storel %tmp_left_key, l %j2 + 0
	storel %tmp_left_val, l %j2 + 8
	storel %tmp_right_key, l %i2 + 0
	storel %tmp_right_val, l %i2 + 8
	%next_i =l addl %i2, 16
	storel %next_i, l %i
	%prev_j =l subl %j2, 16
	storel %prev_j, l %j
	jmp %loop_start
@scan_done:
	ret
@exit:
	ret %0
}
