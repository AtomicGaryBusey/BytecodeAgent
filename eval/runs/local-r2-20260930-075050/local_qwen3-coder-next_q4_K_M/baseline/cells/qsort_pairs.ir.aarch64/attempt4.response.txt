export function l $qsort_pairs(l %a, l %n) {
%entry:
	%0 =w ceql %n, 0
	jnz %0, %exit
	%1 =w ceql %n, 1
	jnz %1, %exit

	// Save callee-saved registers (x19-x28, x29, x30)
	// We'll use x19..x22 for local values; save them first
	storel %n, l (%sp, #-(16*5)) // 80 bytes: 4*8 + 48 padding to keep stack aligned
	storel %a, l (%sp, #8)
	stp x19, x20, l (%sp, #16)
	stp x21, x22, l (%sp, #32)
	stp x23, x24, l (%sp, #48)
	stp x25, x26, l (%sp, #64)

	// n >= 2; proceed with quicksort
	basic %qs_init

@qs_init:
	// Compute mid index = n/2
	%mid_idx =l shrl %n, 1
	// offset = mid_idx * 16
	%mid_off =lshl %mid_idx, 4
	// pivot_ptr = a + offset
	%pivot_ptr =l addl %a, %mid_off
	// load pivot key and value
	%pivot_key = loadl l %pivot_ptr + 0
	%pivot_val = loadl l %pivot_ptr + 8

	// Initialize i = a, j = a + (n-1)*2
	%n_minus_1 =l subl %n, 1
	%j_init_off =lshl %n_minus_1, 4
	%j_init =l addl %a, %j_init_off

	storel %a, l %i_ptr_loc
	storel %j_init, l %j_ptr_loc

	basic %loop_start

@loop_start:
	// while (*i < pivot) i += 2;
	%li = loadl l %i_ptr_loc
	%left_key = loadl l %li + 0
	%cmp_lt =w csltl %left_key, %pivot_key
	jnz %cmp_lt, %inc_left1

	// if (*i == pivot), check value: (*i).val < pivot.val ?
	%cmp_eq =w ceql %left_key, %pivot_key
	jnz %cmp_eq, %check_val1
	jmp %scan_right1

@check_val1:
	%left_val = loadl l %li + 8
	%cmp_lt2 =w csltl %left_val, %pivot_val
	jnz %cmp_lt2, %inc_left1
	jmp %scan_right1

@inc_left1:
	%li_next =l addl %li, 16
	storel %li_next, l %i_ptr_loc
	basic %loop_start

@scan_right1:
	// while (*j > pivot) j -= 2;
	%rj = loadl l %j_ptr_loc
	%right_key = loadl l %rj + 0
	%cmp_gt =w csgtl %right_key, %pivot_key
	jnz %cmp_gt, %dec_right1

	// if (*j == pivot), check value: (*j).val > pivot.val ?
	%cmp_eq_r =w ceql %right_key, %pivot_key
	jnz %cmp_eq_r, %check_val_r1
	jmp %ij_check1

@check_val_r1:
	%right_val = loadl l %rj + 8
	%cmp_gt2 =w csgtl %right_val, %pivot_val
	jnz %cmp_gt2, %dec_right1
	jmp %ij_check1

@dec_right1:
	%rj_prev =l subl %rj, 16
	storel %rj_prev, l %j_ptr_loc
	basic %scan_right1

@ij_check1:
	// if i <= j, swap and move both
	%li2 = loadl l %i_ptr_loc
	%rj2 = loadl l %j_ptr_loc
	%cmp_ij =w csgtl %li2, %rj2
	jnz %cmp_ij, %scan_done1

	// swap *i and *j
	%tmp_k0 = loadl l %li2 + 0
	%tmp_v0 = loadl l %li2 + 8
	%tmp_k1 = loadl l %rj2 + 0
	%tmp_v1 = loadl l %rj2 + 8

	storel %tmp_k0, l %rj2 + 0
	storel %tmp_v0, l %rj2 + 8
	storel %tmp_k1, l %li2 + 0
	storel %tmp_v1, l %li2 + 8

	// i += 2, j -= 2
	%ni =l addl %li2, 16
	%nj =l subl %rj2, 16
	storel %ni, l %i_ptr_loc
	storel %nj, l %j_ptr_loc

	basic %loop_start

@scan_done1:
	// Partition complete. Now recurse.
	// j is last index of left partition (a..j), i = j+2 is start of right.

	%ji = loadl l %j_ptr_loc
	// left part: from a to ji, count = ((ji - a)/16) + 1 = (ji-a)/16 + 1
	%left_n_bytes =l subl %ji, %a
	%left_cnt =l shrl %left_n_bytes, 4
	%left_cnt1 =l addl %left_cnt, 1

	// right part: from i to end, count = n - left_cnt1
	%right_off = loadl l %i_ptr_loc
	%right_n_total =l subl %n, %left_cnt1

	// Save registers before recursion
	stp x27, x28, l (%sp, #80)

	basic %rec_left

@rec_left:
	// sort left part: a, left_cnt1
	%r1 =l call $qsort_pairs(l %a, l %left_cnt1)

	// Restore registers
	ldp x27, x28, l (%sp, #80)
	stp x29, x30, l (%sp, #(-16))
	addl %r30 =l x30

	basic %rec_right

@rec_right:
	// sort right part: right_off, right_n_total
	if (right_n_total > 0) {
		%r2 =l call $qsort_pairs(l %right_off, l %right_n_total)
	}

	// Restore caller-saved + frame pointer + ret addr
	ldp x19, x20, l (%sp, #16)
	ldp x21, x22, l (%sp, #32)
	ldp x23, x24, l (%sp, #48)
	ldp x25, x26, l (%sp, #64)
	addl %sp =l addl %sp, 80
	ret %0

@exit:
	ret %0
}
