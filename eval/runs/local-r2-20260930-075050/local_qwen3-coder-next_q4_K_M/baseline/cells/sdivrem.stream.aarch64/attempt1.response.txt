sdivrem:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Check b == 0
    cmp x1, #0
    beq .Lzero_divisor
    // Check special case a == INT64_MIN && b == -1
    mov x8, #1
    neg x8, x8         // x8 = -1
    cmp x0, #0x8000000000000000
    ccmp x1, x8, #4, ne
    b.ne .Lnormal_case

    // Special case: INT64_MIN / -1 -> overflow, return INT64_MIN and rem = 0
    mov x0, #0x8000000000000000
    str x0, [x2]
    b .Lreturn

.Lzero_divisor:
    // b == 0: q = 0, r = a
    clz x3, x1         // set flags for rem, but we use direct store
    mov x0, #0
    str x0, [x2]
    b .Lreturn

.Lnormal_case:
    // Save caller-saved registers if needed (we don't use extra callee-saved except x29)
    // Perform division: a / b with truncation toward zero
    sdiv x3, x0, x1    // quotient
    msub x4, x3, x1, x0 // remainder = a - q * b
    str x4, [x2]
    mov x0, x3

.Lreturn:
    ldp x29, x30, [sp], #16
    ret
