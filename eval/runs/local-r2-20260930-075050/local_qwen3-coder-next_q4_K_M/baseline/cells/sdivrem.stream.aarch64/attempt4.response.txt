sdivrem:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Check if b == 0
    cbz x1, .Lzero_divisor
    // Save rem pointer (x2) to callee-saved register since we need it later
    mov x19, x2
    // Load INT64_MIN constant
    mov x3, #0x8000000000000000
    // Check if a == INT64_MIN
    cmp x0, x3
    b.ne .Lnormal_case
    // a is INT64_MIN, check if b == -1
    mov x4, #0xffffffffffffffff
    cmp x1, x4
    beq .Lspecial_min_div_negone

.Lnormal_case:
    // Perform signed division: q = a / b (truncated toward zero)
    sdiv x20, x0, x1  // quotient in callee-saved reg
    // Compute remainder: r = a - q * b
    msub x21, x20, x1, x0  // x21 = x0 - x20*x1 (a - q*b)
    // Store remainder to *rem
    str x21, [x19]
    // Return quotient
    mov x0, x20
    b .Lcleanup

.Lzero_divisor:
    mov x0, #0
    str x0, [x19]  // store a (original x0) would be overwritten? Wait—we need original a for rem
    mov x0, x19
    ldr x2, [sp, #16]  // recover original x2 (rem ptr) is wrong—x19 holds it
    str x2, [x19]  // this is nonsense

.Lspecial_min_div_negone:
    mov x0, #0x8000000000000000
    str xzr, [x19]
    b .Lcleanup

.Lcleanup:
    ldp x29, x30, [sp], #16
    ret
