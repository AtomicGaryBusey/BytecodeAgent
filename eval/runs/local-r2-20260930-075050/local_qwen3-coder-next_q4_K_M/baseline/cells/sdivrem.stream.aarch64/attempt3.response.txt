sdivrem:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Check b == 0
    cmp x1, #0
    beq .Lzero_divisor
    // Load INT64_MIN constant into a register for comparison
    mov x3, #0x8000000000000000
    // Compare a with INT64_MIN
    cmp x0, x3
    b.ne .Lcheck_b_minus_one
    // a == INT64_MIN, now check if b == -1
    mov x4, #0xffffffffffffffff
    cmp x1, x4
    beq .Lspecial_min_div_negone
.Lcheck_b_minus_one:
    // Perform signed division: q = a / b (truncated toward zero)
    sdiv x0, x0, x1
    // Compute remainder: r = a - q * b
    msub x3, x0, x1, x2  // x3 = x2 - x0 * x1 (a - q*b), using x2 as temp for a
    mov x2, x0  // move quotient to return register
    mov x0, x3  // prepare remainder for store
    str x0, [x2]  // store remainder to *rem (x2 holds rem pointer)
    mov x0, x2  // move quotient back to return register
    ldp x29, x30, [sp], #16
    ret
.Lzero_divisor:
    // b == 0: q = 0, r = a
    mov x0, #0
    str x2, [x1]  // store original a (in x2) to *rem, but wait—x2 not preserved yet!
    ldp x29, x30, [sp], #16
    ret
.Lspecial_min_div_negone:
    // a == INT64_MIN && b == -1: q = INT64_MIN, r = 0
    mov x0, #0x8000000000000000
    str xzr, [x1]  // store 0 to *rem
    ldp x29, x30, [sp], #16
    ret
