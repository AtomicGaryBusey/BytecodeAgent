base64_encode:
    // Prologue
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Initialize registers: src in x5, n in x6, out in x7, output count in x8
    mov x5, x0          // src
    mov x6, x1          // n
    mov x7, x2          // out
    mov x8, #0          // output count

    // Handle zero-length input
    cbz x6, .Ldone

.Lloop:
    // Check remaining bytes
    cmp x6, #3
    b.lt .Lpad_final

    // Load 3 bytes (b0, b1, b2)
    ldrb w9, [x5]         // b0
    ldrb w10, [x5, #1]    // b1
    ldrb w11, [x5, #2]    // b2

    // Combine into 24-bit value: (b0 << 16) | (b1 << 8) | b2
    mov x0, x9
    lsl x0, x0, #16
    orr x0, x0, x10, lsl #8
    orr x0, x0, x11

    // Extract four 6-bit indices (from MSB to LSB)
    and x1, x0, #0xfc0000
    lsrv x1, x1, #18      // index0 = bits 23..18

    and
