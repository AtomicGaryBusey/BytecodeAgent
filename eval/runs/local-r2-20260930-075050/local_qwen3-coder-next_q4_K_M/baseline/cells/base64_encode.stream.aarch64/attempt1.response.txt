base64_encode:
    // Prologue
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Initialize output pointer in x2 (out), src in x0, n in x1
    mov x5, x0          // x5 = src
    mov x6, x1          // x6 = n
    mov x7, x2          // x7 = out
    mov x28, #0         // x28 = output count (will accumulate result)

    // Handle zero-length input
    cbz x6, .Ldone

.Lloop:
    // Check remaining bytes
    cmp x6, #3
    ccmp x6, #0, #4, lt  // n == 0 ? 
    b.le .Lpad_final      // if n < 3, handle padding

    // Load 3 bytes (b0, b1, b2)
    ldrb w8, [x5]         // b0
    ldrb w9, [x5, #1]     // b1
    ldrb w10, [x5, #2]    // b2

    // Combine into 24-bit value: (b0 << 16) | (b1 << 8) | b2
    mov x0, x8
    lsl x0, x0, #16
    orr x0, x0, x9, lsl #8
    orr x0, x0, x10

    // Extract four 6-bit indices (from MSB to LSB)
    and x1, x0, #0xfc0000
    lsri x1, x1, #18      // index0 = bits 23..18

    and
