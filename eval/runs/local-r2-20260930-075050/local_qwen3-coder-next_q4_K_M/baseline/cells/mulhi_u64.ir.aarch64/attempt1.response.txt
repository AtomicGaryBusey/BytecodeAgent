export function l $mulhi_u64(l %a, l %b) {
l %al, %ah, %bl, %bh
l %t0, %t1, %t2, %t3, %t4, %t5, %t6
w %carry

; Split a into low and high 32-bit parts
%al = and l %a, 4294967295 ; a & 0xFFFFFFFF
%ah = shr u l %a, 32       ; a >> 32

; Split b into low and high 32-bit parts
%bl = and l %b, 4294967295 ; b & 0xFFFFFFFF
*bh = shr u l %b, 32       ; b >> 32

; Compute partial products
; t0 = al * bl (low 64 bits)
%t0 = mul l %al, %bl

; t1 = ah * bl + (al * bh) + carry from low multiplication
; First compute al*bl high part for initial carry
%t2 = shr u l %t0, 32 ; low product's high 32 bits
%t1 = mul l %ah, %bl
%t3 = mul l %al, %bh
%t4 = add l %t1, %t3
; Add carry from t0
%carry = csltl w %t0, %t2 ; check if low 32 bits of t0 overflowed when adding high part
%t5 = wzrv w %carry, 0    ; convert carry to int (0 or 1)
%t6 = add l %t4, %t5      ; add initial carry
%t1 = and l %t6, 4294967295 ; keep low 32 bits of middle sum

; t2 = ah*bl + al*bh high part + carry from previous addition
%t0 = shr u l %t6, 32     ; get carry from adding middle parts
%carry = csltl w %t6, %t0 ; check for overflow in high part calculation
%t5 = wzrv w %carry, 0    ; convert to 0 or 1
%t2 = shr u l %t6, 32     ; high 32 bits of middle sum

; t3 = ah * bh (high 64 bits)
%t3 = mul l %ah, %bh

; Now compute final result: 
; full product = 
;   (al*bl) +
;   ((ah*bl + al*bh) << 32) +
;   (ah*bh << 64)

; The high 64 bits consist of:
;   ah*bh 
; + carry from middle sum (which is %t2)
; + any carry from shifting the middle sum into the high part

%result = add l %t3, %t2
%result = add l %result, %t5 ; add final carry bit

ret l %result
}
