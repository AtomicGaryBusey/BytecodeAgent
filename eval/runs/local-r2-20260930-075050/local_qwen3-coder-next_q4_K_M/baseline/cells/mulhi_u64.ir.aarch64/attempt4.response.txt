export function l $mulhi_u64(l %a, l %b) {
	l %al, %ah, %bl, %bh
	l %t0, %t1, %t2, %t3, %t4, %t5, %t6
	w %c1

@mulhi_u64.1
	; Split a and b into 32-bit halves
	%al = and l %a, 4294967295
	%ah = shr u l %a, 32
	%bl = and l %b, 4294967295
	%bh = shr u l %b, 32

	; Low 64-bit product: al * bl
	%t0 = mul l %al, %bl
	%t1 = shr u l %t0, 32 ; high 32 bits of low product

	; Middle terms: ah*bl + al*bh
	%t2 = mul l %ah, %bl
	%t3 = mul l %al, %bh
	%t6 = add l %t2, %t3

	; Add carry from low product (t1)
	; Check if t1 + t6 overflows 32 bits in the high part
	%t5 = add l %t6, %t1
	;c1 = carry out of (t6 + t1) in 64-bit addition? We need to detect overflow
	csltl w %t5, %t6 ; if t5 < t6 then carry occurred
	%t2 = select w %c1, 1, 0

	; Get high 32 bits of middle sum (t5)
	%t4 = shr u l %t5, 32

	; Final adjustment: add carry from middle addition to high product
	%t6 = mul l %ah, %bh
	%t1 = add l %t4, %t2
	%result = add l %t6, %t1

	ret l %result
}
