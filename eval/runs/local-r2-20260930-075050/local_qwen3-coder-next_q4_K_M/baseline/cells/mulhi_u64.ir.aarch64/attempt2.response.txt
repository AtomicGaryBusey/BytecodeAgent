export function l $mulhi_u64(l %a, l %b) {
l %al, %ah, %bl, %bh
l %t0, %t1, %t2, %t3, %t4, %t5, %t6, %t7
w %c1, %c2

; Split a and b into 32-bit halves
%al = and l %a, 4294967295
%ah = shr u l %a, 32
%bl = and l %b, 4294967295
%bh = shr u l %b, 32

; Low 64-bit product: al * bl
%t0 = mul l %al, %bl
%t1 = shr u l %t0, 32 ; high 32 bits of low product

; Middle terms: ah*bl + al*bh
%t2 = mul l %ah, %bl
%t3 = mul l %al, %bh
%t4 = add l %t2, %t3

; Add carry from low product (t1)
%c1 = csltl w %t0, %t1 ; t0 + (t1<<32) overflowed?
%c2 = wzrv w %c1, 0
%t5 = add l %t4, %c2   ; add carry to middle sum

; Split the middle result: low and high 32 bits
%t6 = and l %t5, 4294967295 ; low 32 bits of middle sum
%t7 = shr u l %t5, 32       ; high 32 bits of middle sum

; High product: ah * bh
%t2 = mul l %ah, %bh

; Final high 64 bits = t2 + t7 + carry from shifting middle term into high part
%c1 = csltl w %t5, %t7 ; did adding to middle overflow?
%c2 = wzrv w %c1, 0
%result = add l %t2, %t7
%result = add l %result, %c2

ret l %result
}
