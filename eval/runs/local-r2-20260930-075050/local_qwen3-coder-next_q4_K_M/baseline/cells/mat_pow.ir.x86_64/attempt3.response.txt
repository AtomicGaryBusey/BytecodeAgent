export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
%init:
	%zero = constl 0
	%nval = copyl %n
	%kval = copyl %k
	%outptr = copyl %out
	%tmp1ptr = copyl %tmp
	%tmp2ptr = adjl %tmp, 288 ; tmp + n*n*8 for max n=6: 36*8=288
	%n_sq = mull %nval, %nval
	%n_sq_bytes = mull %n_sq, constl 8

@zero_all:
	%i = constl 0
	%total_bytes = mull %n_sq, constl 128 ; 2 * n*n * 8
@zero_loop:
	%c = ceql %i, %total_bytes
	jcz %c, @done_zeroing
	storel %zero, %outptr, %i
	storel %zero, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @zero_loop

@done_zeroing:
	; k == 0? return identity matrix in out
	%c = ceql %kval, %zero
	jnz %c, @identity_case

	; copy base matrix to tmp1ptr
	%i = constl 0
@copy_base_loop:
	%c = ceql %i, %n_sq_bytes
	jcz %c, @base_copied
	%val = loadl %a, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_base_loop

@base_copied:
	; result = identity matrix in out
	%j = constl 0
@store_identity:
	%c = ceql %j, %n_sq
	jcz %c, @exp_init_done
	%irow = subl %j, %mulsl(%divsl(%j, %nval), %nval)
	%icol = remul(%j, %nval)
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(%j, 8)
	%j = addl %j, constl 1
	jmp @store_identity

@exp_init_done:
@main_loop:
	%c = ceql %kval, %zero
	jcz %c, @exp_done

	; if (k & 1) result *= tmp1
	%bit = andl %kval, constl 1
	%c2 = ceql %bit, %zero
	jnz %c2, @skip_mult

	; copy out to tmp2ptr
	%i = constl 0
@copy_out_to_tmp2:
	%c3 = ceql %i, %n_sq_bytes
	jcz %c3, @copied_to_tmp2
	%val = loadl %outptr, %i
	storel %val, %tmp2ptr, %i
	%i = addl %i, constl 8
	jmp @copy_out_to_tmp2

@copied_to_tmp2:
	; multiply tmp1 * tmp2 -> tmp2 (output in tmp2)
	%prod_ptr = copyl %tmp2ptr
	call $matmul(l %tmp1ptr, l %tmp2ptr, l %nval, l %prod_ptr)

	; copy tmp2 back to out
	%i = constl 0
@copy_tmp2_to_out:
	%c4 = ceql %i, %n_sq_bytes
	jcz %c4, @copied_back
	%val = loadl %tmp2ptr, %i
	storel %val, %outptr, %i
	%i = addl %i, constl 8
	jmp @copy_tmp2_to_out

@copied_back:
	; zero tmp2 (first n*n part)
	%zero_i = constl 0
@zero_tmp2_loop:
	%c5 = ceql %zero_i, %n_sq_bytes
	jcz %c5, @tmp2_zeroed
	storel %zero, %tmp2ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @zero_tmp2_loop

@tmp2_zeroed:
@skip_mult:

	; square tmp1: tmp1 * tmp1 -> tmp2
	call $matmul(l %tmp1ptr, l %tmp1ptr, l %nval, l %tmp2ptr)

	; copy tmp2 to tmp1
	%i = constl 0
@copy_tmp2_to_tmp1:
	%c6 = ceql %i, %n_sq_bytes
	jcz %c6, @tmp1_squared
	%val = loadl %tmp2ptr, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_tmp2_to_tmp1

@tmp1_squared:
	; zero tmp2 (first n*n part)
	%zero_i = constl 0
@zero_tmp2_2_loop:
	%c7 = ceql %zero_i, %n_sq_bytes
	jcz %c7, @tmp2_zeroed_2
	storel %zero, %tmp2ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @zero_tmp2_2_loop

@tmp2_zeroed_2:
	; k >>= 1
	%kval = shrl %kval, constl 1
	jmp @main_loop

@identity_case:
	%j = constl 0
@store_identity_id:
	%c = ceql %j, %n_sq
	jcz %c, @done_id
	%irow = subl %j, %mulsl(%divsl(%j, %nval), %nval)
	%icol = remul(%j, %nval)
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(%j, 8)
	%j = addl %j, constl 1
	jmp @store_identity_id

@done_id:
@exp_done:
	; zero entire tmp region (2 * n*n elements)
	%i = constl 0
	%total_tmp_bytes = mull %n_sq, constl 128 ; 2 * n*n * 8
@final_zero_tmp:
	%c = ceql %i, %total_tmp_bytes
	jcz %c, @all_zeroed
	storel %zero, %tmp, %i
	%i = addl %i, constl 8
	jmp @final_zero_tmp

@all_zeroed:
	%ret = constl 0
	ret %ret

}
