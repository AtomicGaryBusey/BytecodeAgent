export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
%init:
	%zero = constl 0
	%nval = copyl %n
	%kval = copyl %k
	%outptr = copyl %out
	%tmp1ptr = copyl %tmp
	%n_sq = mull %nval, %nval
	%n_sq_bytes = mull %n_sq, constl 8

; Zero entire tmp region (2 * n*n elements)
%i = constl 0
%total_tmp_bytes = mull %n_sq, constl 16
@zero_tmp_loop:
	%c = ceql %i, %total_tmp_bytes
	jcz %c, @zero_tmp_done
	storel %zero, %tmp, %i
	%i = addl %i, constl 8
	jmp @zero_tmp_loop

@zero_tmp_done:

; Handle k == 0 case: identity matrix in out
%c = ceql %kval, %zero
jnz %c, @identity_case

; Copy base matrix to tmp1ptr
%i = constl 0
@copy_base_loop:
	%c = ceql %i, %n_sq_bytes
	jcz %c, @base_copied
	%val = loadl %a, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_base_loop

@base_copied:

; Initialize result as identity matrix in out
%j = constl 0
@store_identity:
	%c = ceql %j, %n_sq
	jcz %c, @exp_init_done
	%irow = divsl %j, %nval
	%icol = subl %j, mull %irow, %nval
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(%j, 8)
	%j = addl %j, constl 1
	jmp @store_identity

@exp_init_done:

; Main loop: square-and-multiply exponentiation
@main_loop:
	%c = ceql %kval, %zero
	jcz %c, @exp_done

	; if (k & 1) result *= tmp1
	%bit = andl %kval, constl 1
	%c2 = ceql %bit, %zero
	jnz %c2, @skip_mult

	; Copy out to tmp+n*n (second half of tmp)
	%tmp2ptr = adjl %tmp, %n_sq_bytes
	%i = constl 0
@copy_out_to_tmp2:
	%c3 = ceql %i, %n_sq_bytes
	jcz %c3, @copied_to_tmp2
	%val = loadl %outptr, %i
	storel %val, %tmp2ptr, %i
	%i = addl %i, constl 8
	jmp @copy_out_to_tmp2

@copied_to_tmp2:
	; Multiply tmp1 * tmp2 -> tmp2
	%prod_ptr = copyl %tmp2ptr
	; matmul(tmp1ptr, tmp2ptr, nval, prod_ptr)
	%_r = call $matmul(l %tmp1ptr, l %tmp2ptr, l %nval, l %prod_ptr)

	; Copy result back to out
	%i = constl 0
@copy_tmp2_to_out:
	%c4 = ceql %i, %n_sq_bytes
	jcz %c4, @copied_back
	%val = loadl %tmp2ptr, %i
	storel %val, %outptr, %i
	%i = addl %i, constl 8
	jmp @copy_tmp2_to_out

@copied_back:
	; Zero tmp2 (second half)
	%i = constl 0
@zero_tmp2_loop:
	%c5 = ceql %i, %n_sq_bytes
	jcz %c5, @tmp2_zeroed
	storel %zero, %tmp2ptr, %i
	%i = addl %i, constl 8
	jmp @zero_tmp2_loop

@tmp2_zeroed:
@skip_mult:

	; Square tmp1: tmp1 * tmp1 -> tmp2 (second half)
	%prod_ptr_sq = adjl %tmp, %n_sq_bytes
	call $matmul(l %tmp1ptr, l %tmp1ptr, l %nval, l %prod_ptr_sq)

	; Copy result back to tmp1 (first half)
	%i = constl 0
@copy_prod_to_tmp1:
	%c6 = ceql %i, %n_sq_bytes
	jcz %c6, @tmp1_squared
	%val = loadl %prod_ptr_sq, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_prod_to_tmp1

@tmp1_squared:
	; Zero tmp2 (second half)
	%i = constl 0
@zero_tmp2_sq_loop:
	%c7 = ceql %i, %n_sq_bytes
	jcz %c7, @tmp2_zeroed_sq
	storel %zero, %prod_ptr_sq, %i
	%i = addl %i, constl 8
	jmp @zero_tmp2_sq_loop

@tmp2_zeroed_sq:
	; k >>= 1
	%kval = shrl %kval, constl 1
	jmp @main_loop

@identity_case:
; Build identity matrix directly in out
%j = constl 0
@store_identity_id:
	%c = ceql %j, %n_sq
	jcz %c, @done_id
	%irow = divsl %j, %nval
	%icol = subl %j, mull %irow, %nval
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(%j, 8)
	%j = addl %j, constl 1
	jmp @store_identity_id

@done_id:
@exp_done:
; Zero entire tmp region before returning
%i = constl 0
@final_zero_tmp:
	%c = ceql %i, %total_tmp_bytes
	jcz %c, @all_zeroed
	storel %zero, %tmp, %i
	%i = addl %i, constl 8
	jmp @final_zero_tmp

@all_zeroed:
	%ret = constl 0
	ret %ret
}
