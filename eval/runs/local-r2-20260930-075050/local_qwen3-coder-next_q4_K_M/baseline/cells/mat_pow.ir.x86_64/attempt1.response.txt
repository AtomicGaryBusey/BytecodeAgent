export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
%init:
	%zero = constl 0
	%nval = copyl %n
	%kval = copyl %k
	%outptr = copyl %out
	%tmp1ptr = copyl %tmp
	%tmp2ptr = copyl %tmp
	%tmp2ptr = adjl %tmp2ptr, 8 ; tmp + n*n (since each element is 8 bytes)
	; zero out output and tmp arrays initially
	%i = constl 0
	%size = mull %nval, %nval
	%total = mull %size, constl 8
	%j = constl 0

@zero_loop
	%c = ceql %j, %total
	jcz %c, %done_zeroing
	storel %zero, %outptr, %j
	storel %zero, %tmp1ptr, %j
	%j = addl %j, constl 8
	jmp @zero_loop

@done_zeroing
	; k == 0? return identity matrix in out
	%c = ceql %kval, %zero
	jnz %c, @identity_case

	; copy base matrix to tmp1ptr (tmp)
	%size_bytes = mull %nval, %nval
	%size_bytes = mull %size_bytes, constl 8
	%i = constl 0

@copy_base_loop
	%c = ceql %i, %size_bytes
	jcz %c, @base_copied
	%val = loadl %a, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_base_loop

@base_copied
	; result = identity matrix (start with it for exponentiation)
	%j = constl 0

@store_identity
	%c = ceql %j, %size
	jcz %c, @exp_init_done
	%irow = subl %j, %modl(%j, %nval) ; i = j / n * n
	%icol = %modl(%j, %nval)
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(j, 8)
	%j = addl %j, constl 1
	jmp @store_identity

@exp_init_done
	; kval > 0: exponentiate using square-and-multiply
@main_loop
	;c = ceql %kval, %zero ; check for loop termination? Actually, we'll do while k > 0
	%c = ceql %kval, %zero
	jcz %c, @exp_done

	; if (k & 1) result *= tmp1
	%bit = andl %kval, constl 1
	%c2 = ceql %bit, constl 0
	jnz %c2, @skip_mult

	; copy result to tmp2 (for output of mult)
	%i = constl 0
@copy_result_to_tmp2_loop
	%c3 = ceql %i, %size_bytes
	jcz %c3, @copied_to_tmp2
	%val = loadl %outptr, %i
	storel %val, %tmp2ptr, %i
	%i = addl %i, constl 8
	jmp @copy_result_to_tmp2_loop

@copied_to_tmp2
	; tmp1 * tmp2 -> out (but out will be overwritten: use tmp2 as output? No — we need to store result somewhere safe before copying back. We'll do: mul(tmp1, tmp2, tmp1) then copy tmp1 to out)
	; So compute product = tmp1 * tmp2 -> tmp1
	call $matmul(l %tmp1ptr, l %tmp2ptr, l %nval, l %tmp2ptr) ; reuse tmp2 as output temporarily? No: need separate buffer. Use tmp2 as temp for result of multiplication (then copy back to out)
	; Actually: use tmp2 as output matrix for product = tmp1 * out -> tmp2
	call $matmul(l %tmp1ptr, l %outptr, l %nval, l %tmp2ptr)
	; copy tmp2 -> out
	%i = constl 0
@copy_tmp2_to_out_loop
	%c4 = ceql %i, %size_bytes
	jcz %c4, @copied_back
	%val = loadl %tmp2ptr, %i
	storel %val, %outptr, %i
	%i = addl %i, constl 8
	jmp @copy_tmp2_to_out_loop

@copied_back
	; zero tmp2 (as required: whole tmp area must be zero on return)
	%zero_i = constl 0
@zero_tmp2_loop
	%c5 = ceql %zero_i, %size_bytes
	jcz %c5, @tmp2_zeroed
	storel %zero, %tmp2ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @zero_tmp2_loop

@tmp2_zeroed
@skip_mult

	; square base: tmp1 = tmp1 * tmp1 (store in tmp2 first)
	call $matmul(l %tmp1ptr, l %tmp1ptr, l %nval, l %tmp2ptr)
	; copy tmp2 -> tmp1
	%i = constl 0
@copy_tmp2_to_tmp1_loop
	%c6 = ceql %i, %size_bytes
	jcz %c6, @tmp1_squared
	%val = loadl %tmp2ptr, %i
	storel %val, %tmp1ptr, %i
	%i = addl %i, constl 8
	jmp @copy_tmp2_to_tmp1_loop

@tmp1_squared
	; zero tmp2 again
	%zero_i = constl 0
@zero_tmp2_2_loop
	%c7 = ceql %zero_i, %size_bytes
	jcz %c7, @tmp2_zeroed_2
	storel %zero, %tmp2ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @zero_tmp2_2_loop

@tmp2_zeroed_2
	; k >>= 1
	%kval = shrl %kval, constl 1
	jmp @main_loop

@identity_case
	; identity matrix: diagonal 1s
	%j = constl 0

@store_identity_id
	%c = ceql %j, %size
	jcz %c, @done_id
	%irow = subl %j, %modl(%j, %nval)
	%icol = %modl(%j, %nval)
	%diag = ceql %irow, %icol
	%one_or_zero = selectl %diag, constl 1, constl 0
	storel %one_or_zero, %outptr, %mulsl(j, 8)
	%j = addl %j, constl 1
	jmp @store_identity_id

@done_id
	; zero tmp area
	%i = constl 0
	%total_tmp_bytes = mulsl(size_bytes, 2)
@zero_all_tmp_loop
	%c = ceql %i, %total_tmp_bytes
	jcz %c, @done_all_zeroing
	storel %zero, %tmp, %i
	%i = addl %i, constl 8
	jmp @zero_all_tmp_loop

@done_all_zeroing
	%ret = constl 0
	ret %ret

@exp_done
	; zero tmp1 and tmp2
	%zero_i = constl 0
@final_zero_tmp1_loop
	%c = ceql %zero_i, %size_bytes
	jcz %c, @done_tmp1_zeroing
	storel %zero, %tmp1ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @final_zero_tmp1_loop

@done_tmp1_zeroing
	%zero_i = constl 0
@final_zero_tmp2_loop
	%c = ceql %zero_i, %size_bytes
	jcz %c, @done_all_final
	storel %zero, %tmp2ptr, %zero_i
	%zero_i = addl %zero_i, constl 8
	jmp @final_zero_tmp2_loop

@done_all_final
	%ret = constl 0
	ret %ret
}
