count_ugt:
    mov x1, #0        // result counter = 0
    cbz x2, .Ldone    // if n == 0, return 0 (t is in x3, a in x0, n in x1 initially but we reuse x1 for count)
.Lloop:
    ldr x4, [x0], #8
    cmp x4, x3
    csel x5, xzr, xzr, hi  // dummy: use cset to avoid branch if possible
    b.hi .Lincrement
    b .Lnext
.Lincrement:
    add x1, x1, #1
.Lnext:
    subs x2, x2, #1
    b.ne .Lloop
.Ldone:
    mov x0, x1
    ret
