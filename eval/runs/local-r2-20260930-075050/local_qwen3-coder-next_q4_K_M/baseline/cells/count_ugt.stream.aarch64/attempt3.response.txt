count_ugt:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x0, #0            // result counter = 0
    cbz x1, .Ldone        // if n == 0, return 0
    mov x4, x0            // save original 'a' pointer in x4 (callee-saved later)
    mov x5, x1            // copy n to callee-saved register
.Lloop:
    ldr x6, [x4], #8      // load a[i] and advance pointer
    cmp x6, x2            // compare with t
    b.hi .Lincrement      // branch if a[i] > t (unsigned)
    subs x5, x5, #1       // decrement counter n
    cbz x5, .Ldone        // exit loop if n == 0
    b .Lloop              // continue
.Lincrement:
    add x0, x0, #1        // increment count
    subs x5, x5, #1       // decrement counter n
    cbnz x5, .Lloop       // continue loop if n != 0
.Ldone:
    ldp x29, x30, [sp], #16
    ret
