fill_down:
    mov rax, rsi        # save n in rax for return value
    test rsi, rsi
    jz .Ldone           # if n == 0, skip loop and return

    mov rbx, rdx        # save start in rbx (callee-saved)
    xor rcx, rcx        # i = 0
.Lloop:
    lea rdx, [rdi + rcx*8]  # address of a[i]
    mov qword ptr [rdx], rbx# store current start value at a[i]
    sub rbx, 1            # start--
    inc rcx               # i++
    cmp rcx, rax          # compare i with n
    jl .Lloop             # loop while i < n
.Ldone:
    ret                   # return n (already in rax)
