fill_down:
    mov rax, rsi        # rax = n
    test rsi, rsi
    jz .Ldone           # if n == 0, skip loop

    xor rcx, rcx        # i = 0
.Lloop:
    lea rdx, [rdi + rcx*8]  # address of a[i]
    mov qword ptr [rdx], rsi# store start - i (start in rsi, subtract i)
    sub rsi, 1            # start = start - 1 for next iteration
    inc rcx               # i++
    cmp rcx, rax          # compare i with n
    jl .Lloop             # loop if i < n
.Ldone:
    ret
