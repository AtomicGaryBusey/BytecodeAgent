export function l $reverse(l %a, l %n) {
@start
	l %i
	l %j
	l %tmp1
	l %tmp2
	w %cmp

	%cmp = ceql %n, 0
	jnz %cmp, @done

	%i = copy 0
	%j = sub l %n, 1

@loop
	%cmp = csltl %i, %j
	jnz %cmp, @swap
	jmp @done

@swap
	%tmp1 = loadl l %a + %i * 8
	%tmp2 = loadl l %a + %j * 8
	storel l %a + %i * 8, %tmp2
	storel l %a + %j * 8, %tmp1

	%i = add l %i, 1
	%j = sub l %j, 1
	jmp @loop

@done
	l %ret
	%ret = copy 0
	ret l %ret
}
