export function l $reverse(l %a, l %n) {
@start
	l %i
	l %j
	l %t0
	l %tmp
	w %cmp
	w %done

	%cmp = csltl %n, 2
	jnz %cmp, @done

	%i = copy 0
	%j = sub l %n, 1

@loop
	%cmp = csltl %i, %j
	jnz %cmp, @swap
	jmp @done

@swap
	%tmp = loadl l %a + %i * 8
	storel l %a + %j * 8, %tmp
	%t0 = loadl l %a + %i * 8
	storel l %a + %i * 8, %tmp

	%i = add l %i, 1
	%j = sub l %j, 1
	jmp @loop

@done
	l %ret
	%ret = copy 0
	ret l %ret
}
