export function l $count_nonzero_prefix(l %a, l %n) {
@start
	l %i = l.const 0
	jmp @loop
@loop
	w %cmp = csltl %i, %n
	jnz %cmp, @body
	jmp @exit
@body
	l %addr = l.add %a, %i * 8
	l %val = loadl %addr
	w %zero_test = ceql %val, 0
	jnz %zero_test, @exit
	l %i = l.add %i, 1
	jmp @loop
@exit
	ret %i
}
