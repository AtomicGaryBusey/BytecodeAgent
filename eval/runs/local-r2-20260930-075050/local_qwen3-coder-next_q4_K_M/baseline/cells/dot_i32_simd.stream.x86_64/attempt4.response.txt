dot_i32_simd:
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 32
    and rsp, -16
    lea rax, [rbp+48]
    mov qword ptr [rsp], rax      # save original return address
    xor r9d, r9d                  # loop counter i = 0
    mov r10, rdi                  # save a pointer
    mov r11, rsi                  # save b pointer
    pxor xmm0, xmm0               # clear accumulator low
    pxor xmm1, xmm1               # clear accumulator high (not used but keep in pair)
    mov rax, rdx                  # n in rax (64-bit)
    cmp rax, 4
    jl .Lscalar_tail

.Lsimd_loop:
    movdqa xmm2, oword ptr [r10 + r9*4]   # load 4 int32 from a (4*4=16 bytes)
    movdqa xmm3, oword ptr [r11 + r9*4]   # load 4 int32 from b
    pmovsxdq xmm2, xmm2                   # expand to 4x int64 in lower half
    pmovsxdq xmm3, xmm3                   # expand to 4x int64 in lower half
    pmulld xmm2, xmm3                     # multiply low: 4 int32 * int32 -> 4 int32
    vmovdqa xmm4, xmm2                    # copy result (for expansion)
    pm
