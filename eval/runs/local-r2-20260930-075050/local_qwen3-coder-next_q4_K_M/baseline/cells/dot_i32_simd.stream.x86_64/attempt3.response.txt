dot_i32_simd:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    and rsp, -16
    lea rax, [rbp+48]
    mov qword ptr [rsp], rax      # save original return address
    xor rax, rax                  # clear accumulator (will use xmm)
    xor rbx, rbx                  # byte counter for alignment handling
    xor r12, r12                  # low 64 bits of sum
    xor r13, r13                  # high 64 bits (not used directly but kept for completeness)
    mov rdx, rdi                  # a pointer in rdx
    mov rcx, rsi                  # b pointer in rcx
    mov r8, r8                    # n in r8 (32-bit)
    cmp r8d, 4
    jl .Lscalar_tail

.Lsimd_loop:
    movq xmm0, qword ptr [rdx]
    movq xmm1, qword ptr [rcx]
    pmovsxdq xmm0, xmm0           # expand 4x int32 to 4x int64
    pmovsxdq xmm1, xmm1           # expand 4x int32 to 4x int64
    pmulld xmm0, xmm1             # multiply low pairs (int32*int32 -> int32)
    pmovsxdq xmm0, xmm0           # convert 4x int32 results to 4x int64
    paddq r12, xmm0               # accumulate into r12 via xmm
    add
