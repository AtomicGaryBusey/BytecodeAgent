dot_i32_simd:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    and rsp, -16
    lea rax, [rsp+16]
    mov r12, qword ptr [rax-8]    # save original return address
    mov qword ptr [rsp], r12
    mov rax, 0
    mov rbx, 0
    mov rcx, rdi                  # a pointer
    mov rdx, rsi                  # b pointer
    mov rbp, rdx                  # save b pointer in rbp temporarily
    cmp r8d, 4                    # n >= 4?
    jl .Lscalar_tail
