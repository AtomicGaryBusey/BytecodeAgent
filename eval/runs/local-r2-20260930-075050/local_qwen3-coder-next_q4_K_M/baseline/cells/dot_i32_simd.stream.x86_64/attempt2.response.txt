dot_i32_simd:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    and rsp, -16
    lea rax, [rsp+48]
    mov r12, qword ptr [rax-8]    # save original return address
    mov qword ptr [rsp], r12
    xor rax, rax                  # accumulator low (partial)
    xor rbx, rbx                  # we'll accumulate in xmm registers and final rax
    mov rcx, rdi                  # a pointer
    mov rdx, rsi                  # b pointer
    mov r8, r8                    # n as counter (32-bit, zero-extended to 64)
    cmp r8d, 4
    jl .Lscalar_tail

.Lsimd_loop:
