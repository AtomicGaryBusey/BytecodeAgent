export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%tmp =l sub %sp, 16
	storel %tmp, (%sp)
@entry
	%0 =l slt %n, 0
	jnz %0, @error
	%1 =l slt %cap, 0
	jnz %1, @error
@main_loop
	%i =l phi 0, 0
	%out_pos =l phi 0, 0
	%n_left =l sub %n, %i
	jz %n_left, @done
@check_run
	%run_start_ptr =l add %src, %i
	%first_val =w loadub (%run_start_ptr)
	%j =l phi 0, 0
@count_run_bytes
	%curr_idx =l add %i, %j
	%run_len =l add %j, 1
	%run_check =l sub %n, %curr_idx
	%can_read =l slt 0, %run_check
	jz %can_read, @end_count_run
	%next_ptr =l add %src, %curr_idx
	%next_val =w loadub (%next_ptr)
	%diff =l sub %first_val, %next_val
	jnz %diff, @end_count_run
	%next_j =l add %j, 1
	%max_check =l slt %next_j, 128
	jz %max_check, @end_count_run
	storel %next_j, (%sp)
	jmp @count_run_bytes
@end_count_run
	%run_len_final =l add %j, 1
@check_repeat_packet
	%is_repeat =l sgt %run_len_final, 2
	jz %is_repeat, @start_literal
@emit_repeat_packet
	%header_val =w sub 257, %run_len_final
	%out_ptr =l add %out, %out_pos
	storeb %header_val, %out_ptr
	%data_ptr =l add %out_ptr, 1
	storeb %first_val, %data_ptr
	%new_out_pos =l add %out_pos, 2
	%remaining_cap =l sub %cap, %new_out_pos
	jz %remaining_cap, @end_outer
	%advance_i =l add %i, %run_len_final
	storel %advance_i, (%sp)
	storel %new_out_pos, (%sp)
	jmp @main_loop
@start_literal
	%literal_start_ptr =l add %src, %i
	%lit_j =l phi 0, 0
@extend_literal
	%lit_idx =l add %i, %lit_j
	%n_minus_lit =l sub %n, %lit_idx
	%can_read_lit =l slt 0, %n_minus_lit
	jz %can_read_lit, @end_extend_literal
	%curr_ptr =l add %src, %lit_idx
	%curr_val =w loadub (%curr_ptr)
	%run_at_lit =l sub %n, %lit_idx
	%check_len_3 =l sgt %run_at_lit, 2
	jz %check_len_3, @end_extend_literal
	%next1_ptr =l add %src, %lit_idx
	%val1 =w loadub (%next1_ptr)
	%next2_ptr =l add %src, %lit_idx
	%val2 =w loadub (%next2_ptr)
	%diff1 =l sub %curr_val, %val1
	jnz %diff1, @end_extend_literal
	%diff2 =l sub %curr_val, %val2
	jnz %diff2, @end_extend_literal
	%next_j =l add %lit_j, 1
	%check_128 =l slt %next_j, 128
	jz %check_128, @end_extend_literal
	storel %next_j, (%sp)
	jmp @extend_literal
@end_extend_literal
	%literal_len =l add %lit_j, 1
@trim_literal_if_needed
	%too_long =l sgt %literal_len, 128
	jnz %too_long, @emit_trimmed_literal
@store_full_literal
	%header_val_lit =w sub %literal_len, 1
	%out_ptr_lit =l add %out, %out_pos
	storeb %header_val_lit, %out_ptr_lit
	%src_copy_ptr =l add %src, %i
	%dst_copy_ptr =l add %out_ptr_lit, 1
@copy_literal
	%k =l phi 0, 0
@copy_loop
	%byte_to_copy =w loadub (%src_copy_ptr)
	storeb %byte_to_copy, %dst_copy_ptr
	%next_src =l add %src_copy_ptr, 1
	%next_dst =l add %dst_copy_ptr, 1
	%next_k =l add %k, 1
	%done_copy =l sub %literal_len, %next_k
	jz %done_copy, @end_copy_literal
	storel %next_src, (%sp)
	storel %next_dst, (%sp)
	storel %next_k, (%sp)
	jmp @copy_loop
@end_copy_literal
	%new_out_pos_lit =l add %out_pos, %literal_len
	%new_out_pos_lit =l add %new_out_pos_lit, 1
	%remaining_cap_lit =l sub %cap, %new_out_pos_lit
	jz %remaining_cap_lit, @end_outer
	%final_i =l add %i, %literal_len
	storel %final_i, (%sp)
	storel %new_out_pos_lit, (%sp)
	jmp @main_loop
@emit_trimmed_literal
	%trunc_len =w sub 128, 0
	storel %trunc_len, (%sp)
	storel %out_pos, (%sp)
	jmp@store_full_literal
@end_outer
	%result =l sub %out_pos, %out
	ret %result
@done
	%result_done =l sub %out_pos, %out
	ret %result_done
@error
	%minus_one =w sub 0, 1
	%result_err =w sext %minus_one
	ret %result_err
}
