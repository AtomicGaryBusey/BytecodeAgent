export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%tmp =l sub %sp, 16
	storel %tmp, (%sp)
@entry
	%0 =l slt %n, 0
	jnz %0, @error
	%1 =l slt %cap, 0
	jnz %1, @error
@main_loop
	%i =l phi l 0, %i_next
	%out_pos =l phi l 0, %out_pos_next
	%n_left =l sub %n, %i
	jz %n_left, @done
@check_run
	%run_start_ptr =l add %src, %i
	%first_val =w loadub (%run_start_ptr)
	%j =l phi l 0, %j_next
@count_run_bytes
	%curr_idx =l add %i, %j
	%run_len =l add %j, 1
	%run_check =l sub %n, %curr_idx
	%can_read =l slt 0, %run_check
	jz %can_read, @end_count_run
	%next_ptr =l add %src, %curr_idx
	%next_val =w loadub (%next_ptr)
	%diff =l sub %first_val, %next_val
	jnz %diff, @end_count_run
	%next_j =l add %j, 1
	%max_check =l slt %next_j, 128
	jz %max_check, @end_count_run
	storel %next_j, (%sp)
	jmp @count_run_bytes
@end_count_run
	%run_len_final =l add %j, 1
@check_repeat_packet
	%is_repeat =l sge %run_len_final, 3
	jz %is_repeat, @start_literal
@emit_repeat_packet
	%header_val =w sub 257, %run_len_final
	%out_ptr =l add %out, %out_pos
	storeb %header_val, %out_ptr
	%data_ptr =l add %out_ptr, 1
	storeb %first_val, %data_ptr
	%new_out_pos =l add %out_pos, 2
	%remaining_cap =l sub %cap, %new_out_pos
	jz %remaining_cap, @end_outer
	%advance_i =l add %i, %run_len_final
	%i_next =l add %advance_i, 0
	%out_pos_next =l add %new_out_pos, 0
	jmp @main_loop
@start_literal
	%literal_start_ptr =l add %src, %i
	%lit_j =l phi l 0, %lit_j_next
@extend_literal
	%lit_idx =l add %i, %lit_j
	%n_minus_lit =l sub %n, %lit_idx
	%can_read_lit =l slt 0, %n_minus_lit
	jz %can_read_lit, @end_extend_literal
	%curr_ptr =l add %src, %lit_idx
	%curr_val =w loadub (%curr_ptr)
	%run_at_lit =l sub %n, %lit_idx
	%check_len_3 =l sge %run_at_lit, 3
	jz %check_len_3, @end_extend_literal
	%next1_ptr =l add %src, %lit_idx
	%val1 =w loadub (%next1_ptr)
	%next2_ptr =l add %src, %lit_idx
	%val2 =w loadub (%next2_ptr)
	%diff1 =l sub %curr_val, %val1
	jnz %diff1, @end_extend_literal
	%diff2 =l sub %curr_val, %val2
	jnz %diff2, @end_extend_literal
	%lit_next_j =l add %lit_j, 1
	%check_128 =l slt %lit_next_j, 128
	jz %check_128, @end_extend_literal
	storel %lit_next_j, (%sp)
	jmp @extend_literal
@end_extend_literal
	%literal_len =l add %lit_j, 1
@store_full_literal
	%header_val_lit =w sub %literal_len, 1
	%out_ptr_lit =l add %out, %out_pos
	storeb %header_val_lit, %out_ptr_lit
	%src_copy_ptr =l add %src, %i
	%dst_copy_ptr =l add %out_ptr_lit, 1
@copy_literal
	%k =l phi l 0, %k_next
@copy_loop
	%byte_to_copy =w loadub (%src_copy_ptr)
	storeb %byte_to_copy, %dst_copy_ptr
	%next_src =l add %src_copy_ptr, 1
	%next_dst =l add %dst_copy_ptr, 1
	%next_k =l add %k, 1
	%done_copy =l sub %literal_len, %next_k
	jz %done_copy, @end_copy_literal
	storel %next_src, (%sp)
	storel %next_dst, (%sp)
	storel %next_k, (%sp)
	jmp @copy_loop
@end_copy_literal
	%new_out_pos_lit =l add %out_pos, %literal_len
	%new_out_pos_lit =l add %new_out_pos_lit, 1
	%remaining_cap_lit =l sub %cap, %new_out_pos_lit
	jz %remaining_cap_lit, @end_outer
	%final_i =l add %i, %literal_len
	%i_next =l add %final_i, 0
	%out_pos_next =l add %new_out_pos_lit, 0
	jmp @main_loop
@end_extend_literal
	%literal_len_final =l add %lit_j, 1
@store_full_literal2
	%header_val_lit2 =w sub %literal_len_final, 1
	%out_ptr_lit2 =l add %out, %out_pos
	storeb %header_val_lit2, %out_ptr_lit2
	%src_copy_ptr2 =l add %src, %i
	%dst_copy_ptr2 =l add %out_ptr_lit2, 1
@copy_literal2
	%k2 =l phi l 0, %k_next2
@copy_loop2
	%byte_to_copy2 =w loadub (%src_copy_ptr2)
	storeb %byte_to_copy2, %dst_copy_ptr2
	%next_src2 =l add %src_copy_ptr2, 1
	%next_dst2 =l add %dst_copy_ptr2, 1
	%next_k2 =l add %k2, 1
	%done_copy2 =l sub %literal_len_final, %next_k2
	jz %done_copy2, @end_copy_literal2
	storel %next_src2, (%sp)
	storel %next_dst2, (%sp)
	storel %next_k2, (%sp)
	jmp @copy_loop2
@end_copy_literal2
	%new_out_pos_lit2 =l add %out_pos, %literal_len_final
	%new_out_pos_lit2 =l add %new_out_pos_lit2, 1
	%remaining_cap_lit2 =l sub %cap, %new_out_pos_lit2
	jz %remaining_cap_lit2, @end_outer
	%final_i2 =l add %i, %literal_len_final
	%i_next =l add %final_i2, 0
	%out_pos_next =l add %new_out_pos_lit2, 0
	jmp @main_loop
@trim_literal_if_needed
	%trunc_len =w sub 128, 0
	storel %trunc_len, (%sp)
@store_full_literal3
	%header_val_lit3 =w sub %trunc_len, 0
	%out_ptr_lit3 =l add %out, %out_pos
	storeb %header_val_lit3, %out_ptr_lit3
	%src_copy_ptr3 =l add %src, %i
	%dst_copy_ptr3 =l add %out_ptr_lit3, 1
@copy_literal3
	%k3 =l phi l 0, %k_next3
@copy_loop3
	%byte_to_copy3 =w loadub (%src_copy_ptr3)
	storeb %byte_to_copy3, %dst_copy_ptr3
	%next_src3 =l add %src_copy_ptr3, 1
	%next_dst3 =l add %dst_copy_ptr3, 1
	%next_k3 =l add %k3, 1
	%done_copy3 =l sub %trunc_len, %next_k3
	jz %done_copy3, @end_copy_literal3
	storel %next_src3, (%sp)
	storel %next_dst3, (%sp)
	storel %next_k3, (%sp)
	jmp @copy_loop3
@end_copy_literal3
	%new_out_pos_lit3 =l add %out_pos, %trunc_len
	%new_out_pos_lit3 =l add %new_out_pos_lit3, 1
	%remaining_cap_lit3 =l sub %cap, %new_out_pos_lit3
	jz %remaining_cap_lit3, @end_outer
	%final_i3 =l add %i, %trunc_len
	%i_next =l add %final_i3, 0
	%out_pos_next =l add %new_out_pos_lit3, 0
	jmp @main_loop
@end_outer
	%result =l sub %out_pos, %out
	ret %result
@done
	%result_done =l sub %out_pos, %out
	ret %result_done
@error
	%minus_one =w sub 0, 1
	%result_err =l sext %minus_one
	ret %result_err
}
