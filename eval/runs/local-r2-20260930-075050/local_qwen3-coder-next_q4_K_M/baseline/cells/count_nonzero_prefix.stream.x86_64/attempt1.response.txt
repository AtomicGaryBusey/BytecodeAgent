count_nonzero_prefix:
    xor rcx, rcx          # rcx = 0 (index counter)
    test rsi, rsi         # check if n == 0
    jz .Ldone             # if n == 0, return 0
.Lloop:
    mov rax, qword ptr [rdi + rcx*8]  # load a[rcx]
    test rax, rax        # check if element is zero
    jz .Ldone            # if zero, stop
    inc rcx              # increment index
    cmp rcx, rsi         # compare with n
    jb .Lloop            # continue if rcx < n
.Ldone:
    mov rax, rcx         # return count (index or n)
    ret
