export function l $shift_sat(l %x, l %n) {
	l %tmp_shift_left, %tmp_shift_right, %result;
	w %cond_n_ge_64;

	@start
	%cond_n_ge_64 = csltl %n, 64
	jnz %cond_n_ge_64, @n_lt_64, @n_ge_64

	@n_lt_64
	%tmp_shift_left = lshl %x, %n
	%tmp_shift_right = ashr %x, %n
	%result = addl %tmp_shift_left, %tmp_shift_right
	jmp @done

	@n_ge_64
	lslr %tmp_shift_left, 0
	aslr %tmp_shift_right, 0
	w %is_neg;
	%is_neg = csltl %x, 0
	jnz %is_neg, @negative_x, @non_negative_x

	@non_negative_x
	storel 0, %tmp_shift_right
	jmp @combine

	@negative_x
	storel -1, %tmp_shift_right
	jmp @combine

	@combine
	l = loadl %tmp_shift_left
	r = loadl %tmp_shift_right
	%result = addl l, r
	jmp @done

	@done
	ret %result
}
