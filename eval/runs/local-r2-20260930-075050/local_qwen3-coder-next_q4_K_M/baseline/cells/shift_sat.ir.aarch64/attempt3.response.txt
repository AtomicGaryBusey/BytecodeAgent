export function l $shift_sat(l %x, l %n) {
	l %tmp_shift_left, %tmp_shift_right, %result;
	w %cond_n_ge_64;

	@start
	%cond_n_ge_64 = csltl 64, %n
	jnz %cond_n_ge_64, @n_lt_64, @n_ge_64

	@n_lt_64
	%tmp_shift_left = lshl %x, %n
	%tmp_shift_right = ashr %x, %n
	%result = addl %tmp_shift_left, %tmp_shift_right
	jmp @done

	@n_ge_64
	lslr %tmp_shift_left, 0
	csltl %tmp_shift_right, %x, 0
	notw %tmp_shift_right, %tmp_shift_right
	%result = addl %tmp_shift_left, %tmp_shift_right
	jmp @done

	@done
	ret %result
}
