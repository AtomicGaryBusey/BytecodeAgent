export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%count =l 0
	%i =l 0
@loop_cond
	%cslt =w csltl %i, %n
	jnz %cslt, @loop_body
	jmp @done
@loop_body
	%idx_off =l mul l %i, 8
	%src_ptr =l add l %src, %idx_off
	%val =l load l %src_ptr
	%r =l call %f(l %val)
	%dst_ptr =l add l %dst, %idx_off
	store l %r, %dst_ptr
	%cslt2 =w csltl %r, %val
	jnz %cslt2, @inc_count
	jmp @next
@inc_count
	%count =l addl %count, 1
@next
	%i =l addl %i, 1
	jmp @loop_cond
@done
	ret l %count
}
