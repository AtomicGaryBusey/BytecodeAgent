export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%count =w 0
	%i =l 0
@loop_cond
	%cslt =w csltl %i, %n
	jnz %cslt, @loop_body
	jmp @done
@loop_body
	%val =l loadl %src + %i * 8
	%r =l call %f(l %val)
	storel %r, %dst + %i * 8
	%cslt2 =w csltl %r, %val
	jnz %cslt2, @inc_count
	jmp @next
@inc_count
	%count =w addl %count, 1
@next
	%i =l addl %i, 1
	jmp @loop_cond
@done
	%result =l zextlw %count
	ret %result
}
