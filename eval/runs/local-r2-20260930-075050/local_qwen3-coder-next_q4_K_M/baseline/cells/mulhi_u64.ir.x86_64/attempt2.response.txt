export function l $mulhi_u64(l %a, l %b) {
	%al = andl %a, 0xffffffff
	%ah = shrl %a, 32
	%bl = andl %b, 0xffffffff
	%bh = shrl %b, 32

	%t0 = mul l %al, %bl
	%t1 = mul l %al, %bh
	%t2 = mul l %ah, %bl
	%t3 = mul l %ah, %bh

	%t4 = shrl %t1, 32
	%t5 = shrl %t2, 32

	%mid_low1 = andl %t0, 0xffffffff
	%carry1 = shrl %t0, 32

	%sum1 = addl %mid_low1, %t4
	%carry2 = shrl %sum1, 32
	%sum1_low = andl %sum1, 0xffffffff

	%sum2 = addl %sum1_low, %t5
	%carry3 = shrl %sum2, 32

	%total_carry = addl %carry1, %carry2
	%total_carry2 = addl %total_carry, %carry3

	%result = addl %t3, %total_carry2
	ret l %result
}
