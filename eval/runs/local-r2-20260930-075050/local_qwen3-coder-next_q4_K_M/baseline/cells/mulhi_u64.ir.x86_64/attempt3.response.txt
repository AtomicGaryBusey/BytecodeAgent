export function l $mulhi_u64(l %a, l %b) {
	%al = andl %a, 0xffffffff
	%ah = shrl %a, 32
	%bl = andl %b, 0xffffffff
	%dh = shrl %b, 32

	%t0 = mul l %al, %bl
	%t1 = mul l %al, %dh
	%t2 = mul l %ah, %bl
	%t3 = mul l %ah, %dh

	%t4 = shrl %t1, 32
	%t5 = shrl %t2, 32

	%mid_low = addl %t0, 0x0
	%mid_carry = shrl %t0, 32

	%sum_mid1 = addl %mid_carry, %t4
	%carry_from_1 = shrl %sum_mid1, 32
	%sum_low = andl %sum_mid1, 0xffffffff

	%sum_mid2 = addl %sum_low, %t5
	%carry_from_2 = shrl %sum_mid2, 32
	%mid_val = andl %sum_mid2, 0xffffffff

	%total_carry = addl %carry_from_1, %carry_from_2

	%result = addl %t3, %total_carry
	ret l %result
}
