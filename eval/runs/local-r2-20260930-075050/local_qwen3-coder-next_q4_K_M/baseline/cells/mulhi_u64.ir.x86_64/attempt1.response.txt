export function l $mulhi_u64(l %a, l %b) {
  %al = andl %a, 0xffffffff
  %ah = shrl %a, 32
  %bl = andl %b, 0xffffffff
  %bh = shrl %b, 32

  %t0 = mul l %al, %bl
  %t1 = mul l %al, %bh
  %t2 = mul l %ah, %bl
  %t3 = mul l %ah, %bh

  %t4 = shrl %t1, 32
  %t5 = shrl %t2, 32
  %t6 = addl %t0, %t4
  %t7 = addl %t6, %t5

  %carry = shrl %t7, 32
  %mid_low = andl %t7, 0xffffffff

  %t8 = mul l %mid_low, 1
  %mid_high = addl %t3, %carry

  %result = addl %mid_high, %t8
  ret l %result
}
