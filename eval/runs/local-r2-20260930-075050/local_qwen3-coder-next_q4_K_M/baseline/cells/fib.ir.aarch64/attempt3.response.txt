export function l $fib(l %n) {
	%0 = ceql %n, 0
	jnz %0, .return_zero
	%1 = ceql %n, 1
	jnz %1, .return_one
	%2 = copy l 0
	%3 = copy l 1
	%4 = addl %n, -1
	%5 = copy l 2

.loop:
	%6 = loadl l %2
	%7 = loadl l %3
	%8 = addl %6, %7
	%9 = copy l %7
	%10 = copy l %8
	%2 = copy l %9
	%3 = copy l %10
	%11 = copy l %5
	%12 = addl %11, 1
	%5 = copy l %12
	%13 = subl %4, 1
	%4 = copy l %13
	%14 = ceql %4, 0
	jnz %14, .done
	jmp .loop

.return_zero:
	ret l 0

.return_one:
	ret l 1

.done:
	%15 = loadl l %3
	ret l %15
}
