export function l $fib(l %n) {
	%0 = csltl %n, 2
	jnz %0, .return_n
	%1 = copy l 0
	%2 = copy l 1
	%3 = copy l %n
	br .loop

.loop:
	%4 = subl %3, 1
	%5 = addl %1, %2
	%6 = copy l %2
	%7 = copy l %5
	%3 = copy l %4
	%1 = copy l %6
	%2 = copy l %7
	%8 = ceql %3, 1
	jnz %8, .done
	jmp .loop

.return_n:
	ret %n

.done:
	ret %2
}
