export function l $fib(l %n) {
	%0 = ceql l %n, 0
	jnz w %0, .return_zero
	%1 = ceql l %n, 1
	jnz w %1, .return_one
	
	%2 = copy l 0
	%3 = copy l 1
	%4 = subl l %n, 1
	%5 = copy l 1

.loop:
	%6 = addl l %2, %3
	%2 = copy l %3
	%3 = copy l %6
	%7 = subl l %4, 1
	%4 = copy l %7
	%8 = ceql l %5, %4
	jnz w %8, .done
	jmp .loop

.return_zero:
	ret l 0

.return_one:
	ret l 1

.done:
	ret l %3
}
