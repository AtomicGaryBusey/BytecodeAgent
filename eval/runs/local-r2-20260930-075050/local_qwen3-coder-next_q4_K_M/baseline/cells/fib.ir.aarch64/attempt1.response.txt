export function l $fib(l %n) {
alloc:
	%0 = csltl %n, 2
	jnz %0, .return_n
	%1 = copy l 0
	%2 = copy l 1
	%3 = copy l 0
	%4 = copy l 1
	%5 = copy l %n
	br .loop

.loop:
	%6 = subl %5, 1
	%7 = addl %1, %2
	%8 = copy l %2
	%9 = copy l %7
	%5 = copy l %6
	%1 = copy l %8
	%2 = copy l %9
	%10 = ceql %5, 1
	jnz %10, .done
	jmp .loop

.return_n:
	%11 = copy l %n
	ret %11

.done:
	ret %2
}
