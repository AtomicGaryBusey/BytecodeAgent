export function l $sdivrem(l %a, l %b, l* %rem) {
@entry
	l %q
	w %is_zero_b
	w %is_min_a
	w %is_minus_one_b
	w %cond1
	w %cond2
	w %should_handle_special

	# Check if b == 0
	%is_zero_b = ceql %b, 0
	jnz %is_zero_b, @handle_zero_divisor

	# Check if a == INT64_MIN (0x8000000000000000) and b == -1
	l %int64_min
	%int64_min = lconst 9223372036854775808 ; 0x8000000000000000
	%is_min_a = ceql %a, %int64_min
	l %minus_one
	%minus_one = lconst 18446744073709551615 ; 0xFFFFFFFFFFFFFFFF (-1 in two's complement)
	%is_minus_one_b = ceql %b, %minus_one
	%cond1 = band %is_min_a, %is_minus_one_b
	jnz %cond1, @handle_special_div

@compute_normal:
	# Normal division: q = a / b (signed divide truncating toward zero)
	%q = sdivl %a, %b
	# Compute remainder: r = a - q * b
	l %product
	%product = lmul %q, %b
	l %r
	%r = subl %a, %product
	# Store remainder to *rem
	storel %r, %rem
	jmp @return

@handle_zero_divisor:
	%q = lconst 0
	storel %a, %rem
	jmp @return

@handle_special_div:
	%q = %int64_min ; INT64_MIN
	l %zero
	%zero = lconst 0
	storel %zero, %rem
	jmp @return

@return:
	ret %q
}
