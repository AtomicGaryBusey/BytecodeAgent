export function l $sdivrem(l %a, l %b, l* %rem) {
@entry
	l %q
	w %is_zero_b
	w %is_min_a
	w %is_minus_one_b
	w %cond1

	# Check if b == 0
	%is_zero_b = ceql %b, 0l
	jnz %is_zero_b, @handle_zero_divisor

	# Load constants
	l %int64_min_val
	%int64_min_val = lconst 9223372036854775808
	l %minus_one_val
	%minus_one_val = lconst -1l

	# Check if a == INT64_MIN and b == -1
	%is_min_a = ceql %a, %int64_min_val
	%is_minus_one_b = ceql %b, %minus_one_val
	%cond1 = band %is_min_a, %is_minus_one_b
	jnz %cond1, @handle_special_div

@compute_normal:
	# Normal division: q = a / b (signed truncate toward zero)
	%q = sdivl %a, %b
	l %product
	%product = lmul %q, %b
	l %r
	%r = subl %a, %product
	storel %r, %rem
	jmp @return

@handle_zero_divisor:
	%q = lconst 0
	storel %a, %rem
	jmp @return

@handle_special_div:
	%q = %int64_min_val
	l %zero_val
	%zero_val = lconst 0
	storel %zero_val, %rem
	jmp @return

@return:
	ret %q
}
