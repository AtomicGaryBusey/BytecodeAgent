export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%0 =w ceql %n, 0
	jnz %0, @empty_return
	%i =l copy 0
	%written =l copy 0
@main_loop
	%1 =w csltl %i, %n
	jz %1, @done
	%pos =l add %src, %i
	%base_byte =w loadub %pos
	%run_len =l copy 1
@run_count_loop
	%2 =l add %i, %run_len
	%3 =w csltl %2, %n
	jz %3, @end_run_check
	%run_pos =l add %src, %2
	%4 =w loadub %run_pos
	%5 =w ceql %4, %base_byte
	jz %5, @match_in_run
	jmp @end_run_check
@match_in_run
	%6 =w ceql %run_len, 127
	jnz %6, @end_run_check
	%run_len =l add %run_len, 1
	jmp @run_count_loop
@end_run_check
	%7 =w csltl %run_len, 3
	jnz %7, @literal_packet_start
@repeat_packet
	%header_val =l sub 257, %run_len
	%header_byte =w int2b %header_val
	%packet_size =l add 2, 0
	%8 =w csltl %written, %cap
	jz %8, @fail
	%9 =l add %written, %packet_size
	%10 =w csltl %9, %cap
	jnz %10, @fail
	%header_offset =l add %out, %written
	storeb %header_byte, %header_offset
	%data_offset =l add %header_offset, 1
	storeb %base_byte, %data_offset
	%written =l add %written, 2
	%11 =l add %i, %run_len
	%i =l copy %11
	jmp @main_loop
@literal_packet_start
	%lit_end =l copy %i
@literal_scan
	%12 =w csltl %lit_end, %n
	jz %12, @emit_literal
	%lit_pos =l add %src, %lit_end
	%lit_byte =w loadub %lit_pos
	%run_for_lit =l copy 1
@lit_run_count
	%13 =l add %lit_end, %run_for_lit
	%14 =w csltl %13, %n
	jz %14, @end_lit_run_check
	%lit_run_pos =l add %src, %13
	%15 =w loadub %lit_run_pos
	%16 =w ceql %15, %lit_byte
	jz %16, @match_in_lit_run
	jmp @end_lit_run_check
@match_in_lit_run
	%17 =w ceql %run_for_lit, 127
	jnz %17, @end_lit_run_check
	%run_for_lit =l add %run_for_lit, 1
	jmp @lit_run_count
@end_lit_run_check
	%18 =w csltl %run_for_lit, 3
	jnz %18, @extend_literal
	jmp @emit_literal
@extend_literal
	%19 =l add %lit_end, 1
	%lit_end =l copy %19
	%20 =w csltl %lit_end, %n
	jz %20, @emit_literal
	%21 =l sub %lit_end, %i
	%22 =w csltl %21, 128
	jnz %22, @emit_literal
	jmp @literal_scan
@emit_literal
	%lit_len =l sub %lit_end, %i
	%header_val2 =l sub %lit_len, 1
	%header_byte2 =w int2b %header_val2
	%packet_size =l add %lit_len, 1
	%23 =w csltl %written, %cap
	jz %23, @fail
	%24 =l add %written, %packet_size
	%25 =w csltl %24, %cap
	jnz %25, @fail
	%header_offset2 =l add %out, %written
	storeb %header_byte2, %header_offset2
	%curr_out =l add %header_offset2, 1
	%curr_src =l add %src, %i
	%lit_cnt =l copy 0
@copy_literals_loop
	%26 =w csltl %lit_cnt, %lit_len
	jz %26, @after_copy_literal
	%27 =w loadub %curr_src
	storeb %27, %curr_out
	%28 =l add %curr_out, 1
	%curr_out =l copy %28
	%29 =l add %curr_src, 1
	%curr_src =l copy %29
	%lit_cnt =l add %lit_cnt, 1
	jmp @copy_literals_loop
@after_copy_literal
	%written =l add %written, %packet_size
	%i =l copy %lit_end
	jmp @main_loop
@fail
	%ret_val =l copy -1
	ret %ret_val
@empty_return
	%30 =w csltl 0, %cap
	jz @fail
	ret 0
@done
	ret %written
}
