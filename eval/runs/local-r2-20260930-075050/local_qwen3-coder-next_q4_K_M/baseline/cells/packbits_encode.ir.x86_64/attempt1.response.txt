export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%0 =w ceql %n, 0
	jnz %0, @empty_return
	%i =l copy 0
	%j =l copy 0
	%written =l copy 0
@main_loop
	%1 =w csltl %i, %n
	jz %1, @done
	%pos =l add %src, %i
	%base_byte =w loadub %pos
	%run_len =l copy 1
	%run_pos =l add %pos, 1
@run_count_loop
	%2 =w csltl %run_pos, %src
	jz %2, @end_run
	%3 =w loadub %run_pos
	%4 =w ceql %3, %base_byte
	jz %4, @match_in_run
	jmp @end_run
@match_in_run
	%run_len =l add %run_len, 1
	%5 =w ceql %run_len, 128
	jnz %5, @end_run
	%6 =w csltl %run_pos, %src
	jz %6, @end_run
	%7 =l add %run_pos, 1
	%run_pos =l copy %7
	jmp @run_count_loop
@end_run
	%8 =w csltl %run_len, 3
	jnz %8, @literal_packet_start
@repeat_packet
	%header_val =l sub 257, %run_len
	%header_byte =w int2b %header_val
	%header_offset =l add %out, %written
	%9 =w csltl %written, %cap
	jz %9, @fail
	storeb %header_byte, %header_offset
	%10 =l add %written, 1
	%11 =w csltl %10, %cap
	jz %11, @fail
	%data_offset =l add %out, %10
	storeb %base_byte, %data_offset
	%written =l add %written, 2
	%12 =l add %i, %run_len
	%i =l copy %12
	jmp @main_loop
@literal_packet_start
	%lit_end =l copy %i
@literal_scan
	%13 =w csltl %lit_end, %n
	jz %13, @emit_literal
	%14 =l add %src, %lit_end
	%lit_byte =w loadub %14
	%run_for_lit =l copy 1
	%lit_run_pos =l add %14, 1
@lit_run_count
	%15 =w csltl %lit_run_pos, %src
	jz %15, @end_lit_run
	%16 =w loadub %lit_run_pos
	%17 =w ceql %16, %lit_byte
	jz %17, @match_in_lit_run
	jmp @end_lit_run
@match_in_lit_run
	%run_for_lit =l add %run_for_lit, 1
	%18 =w ceql %run_for_lit, 128
	jnz %18, @end_lit_run
	%19 =w csltl %lit_run_pos, %src
	jz %19, @end_lit_run
	%20 =l add %lit_run_pos, 1
	%lit_run_pos =l copy %20
	jmp @lit_run_count
@end_lit_run
	%21 =w csltl %run_for_lit, 3
	jnz %21, @extend_literal
	jmp @emit_literal
@extend_literal
	%22 =l add %lit_end, 1
	%lit_end =l copy %22
	%23 =w csltl %lit_end, %n
	jz %23, @emit_literal
	%24 =l sub %lit_end, %i
	%25 =w csltl %24, 128
	jnz %25, @emit_literal
	jmp @literal_scan
@emit_literal
	%lit_len =l sub %lit_end, %i
	%header_val2 =l sub %lit_len, 1
	%header_byte2 =w int2b %header_val2
	%packet_size =l add %lit_len, 1
	%26 =w csltl %written, %cap
	jz %26, @fail
	%27 =l add %written, %packet_size
	%28 =w csltl %27, %cap
	jnz %28, @fail
	storeb %header_byte2, %out
	%curr_out =l add %out, 1
	%curr_src =l add %src, %i
	%lit_cnt =l copy 0
@copy_literals_loop
	%29 =w csltl %lit_cnt, %lit_len
	jz %29, @after_copy_literal
	%30 =w loadub %curr_src
	storeb %30, %curr_out
	%31 =l add %curr_out, 1
	%curr_out =l copy %31
	%32 =l add %curr_src, 1
	%curr_src =l copy %32
	%lit_cnt =l add %lit_cnt, 1
	jmp @copy_literals_loop
@after_copy_literal
	%written =l add %written, %packet_size
	%i =l copy %lit_end
	jmp @main_loop
@fail
	%ret_val =l copy -1
	ret %ret_val
@empty_return
	%33 =w csltl 0, %cap
	jz @fail
	%34 =l add %written, 0
	ret %34
@done
	ret %written
}
