packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor rax, rax
    mov rbx, rdi              # src
    mov rsi, rsi              # n (keep in rsi)
    mov rdx, rdx              # out
    mov rcx, rcx              # cap (keep in rcx)
    xor r12, r12              # i = 0
    xor r13, r13              # written bytes counter
.Lloop:
    cmp r12, rsi              # if i >= n, done
    jge .Ldone
    mov r14, r12              # run start position
    mov dl, byte ptr [rbx + r12]  # b = src[i]
    xor r15, r15              # run_len = 0
.Lcount_run:
    cmp r14, rsi              # check boundary
    jge .Lrun_end
    mov al, byte ptr [rbx + r14]
    cmp al, dl                # compare with b
    jne .Lrun_end
    inc r15                   # run_len++
    inc r14
    jmp .Lcount_run
.Lrun_end:
    cmp r15, 3                # if run_len >= 3, use repeat packet
    jl .Lliteral_start
    # emit repeat packet: header = 257 - r
    mov rax, r15              # r = run_len (1..128)
    xor r14d, r14d            # clear for subtraction
    sub r14, rax              # r14 = 0 - r = -r
    add r14, 257              # r14 = 257 - r
    mov al, byte ptr [rbx + r12]  # b = src[i]
    # check space: need 2 bytes
    mov rax, r13
    add rax, 2
    cmp rax, rcx
    jg .Loverflow
    # write header and byte
    movzx eax, r14w           # zero-extend 16-bit to 32 for store
    mov byte ptr [rdx + r13], al
    inc r13
    mov byte ptr [rdx + r13], dl
    inc r13
    mov r12, r14              # i += run_len (r14 already had r)
    jmp .Lloop
.Lliteral_start:
    mov r14, r12              # literal start = i
    xor r15, r15              # literal_len = 0
.Lfind_literal_end:
    cmp r14, rsi              # if j >= n, stop
    jge .Lemit_literal
    mov al, byte ptr [rbx + r14]
    xor rcx, rcx
    mov cl, al
    xor r9d, r9d              # run count for current pos
    xor r8d, r8d              # temp for counting
.Lcheck_run:
    cmp r9d, 128              # max run is 128
    jge .Lrun_check_done
    lea rax, [rbx + r14]
    add rax, r9
    mov dl, byte ptr [rax]    # src[j + run_count]
    cmp dl, cl                # compare with current literal byte
    jne .Lrun_check_done
    inc r8d                   # run count++
    inc r9d
    jmp .Lcheck_run
.Lrun_check_done:
    cmp r8d, 3                # if run >= 3, stop literal before it
    jge .Lemit_literal
    inc r15                   # include this byte in literal
    inc r14
    jmp .Lfind_literal_end
.Lemit_literal:
    mov rax, r15
    cmp rax, 0
    je .Lloop                 # no bytes in literal, skip (shouldn't happen but safe)
    dec rax                   # header = L - 1
    # check space: need 1 + r15 bytes
    mov rbx, r13              # save original written count
    add rax, r15
    cmp rax, rcx
    jg .Loverflow_restore
    # write header
    mov byte ptr [rdx + r13], al
    inc r13
    # copy literal bytes
    xor r9, r9                # offset in literal
.Lcopy_literal:
    cmp r9, r15
    jge .Lliteral_done
    mov al, byte ptr [rbx + r9]   # rbx still holds src (restored from original rdi? no - problem)
    mov byte ptr [rdx + r13], al
    inc r13
    inc r9
    jmp .Lcopy_literal
.Lliteral_done:
    mov r12, r14              # i = j
    jmp .Lloop
.Ldone:
    mov rax, r13
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Loverflow_restore:
    # restore rbx and return -1
    mov r13, rbx              # restore written counter
    jmp .Loverflow
.Loverflow:
    mov rax, -1
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
