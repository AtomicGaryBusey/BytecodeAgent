packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor rax, rax
    mov rbx, rdi              # src
    mov rsi, rsi              # n (keep in rsi)
    mov rdx, rdx              # out
    mov rcx, rcx              # cap (keep in rcx)
    xor r12, r12              # i = 0
    xor r13, r13              # written bytes counter
.Lloop:
    cmp r12, rsi              # if i >= n, done
    jge .Ldone
    mov r14, r12              # run start position
    mov r15b, byte ptr [rbx + r12]  # b = src[i]
.Lcount_run:
    cmp r15, rsi              # prevent overflow beyond n
    jge .Lrun_end
    mov dl, byte ptr [rbx + r14 + 1]   # peek next
    inc r14
    cmp dl, r15b              # compare with b
    je .Lcount_run
.Lrun_end:
    sub r14, r12              # run_len = j - i
    cmp r14, 3                # if run_len >= 3, use repeat packet
    jl .Lliteral_start
    # emit repeat packet: header = 257 - r = -(r-1)
    mov rax, r14              # r = run_len (1..128)
    dec rax                   # r - 1
    neg rax                   # -(r-1) as signed byte
    cbw                       # sign extend al to ax
    cwd                       # sign extend ax to dx:ax (not needed, we need byte)
    # Actually we want header = 257 - r. Since r >=3 and <=128:
    # 257 - r = 0x101 - r. Compute directly.
    mov al, 0FFh              # load 255
    sub
