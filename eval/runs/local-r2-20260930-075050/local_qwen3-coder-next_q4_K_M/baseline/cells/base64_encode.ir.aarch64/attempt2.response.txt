export function l $base64_encode(l %src, l %n, l %out) {
@init
  %res =l alloc1 8
  storel %n, %res
  %i =l copy 0
  %out_ptr =l copy %out
@loop
  %nval =l loadl %res
  %cmp =csltl %i, %nval
  jnz %cmp, @body
  jump @done
@body
  %rem =w copy 3
  %rmd =w cmodu %nval, %rem
  %cmp2 =ceql %rmd, 0
  jnz %cmp2, @full_group
  switch %rmd {
    case 1: jump @final_1
    case 2: jump @final_2
  }
@full_group
  %in_ptr =l add %src, %i
  %b0 =w loadub %in_ptr
  %off1 =l add %in_ptr, 1
  %b1 =w loadub %off1
  %off2 =l add %in_ptr, 2
  %b2 =w loadub %off2
  %sh0 =l copy 16
  %v0 =l mul %b0, %sh0
  %sh1 =l copy 8
  %v1 =l mul %b1, %sh1
  %v2 =l add %v0, %v1
  %v3 =l add %v2, %b2
  jump @encode_all
@final_1
  %in_ptr =l add %src, %i
  %b0 =w loadub %in_ptr
  %b1 =w copy 0
  %b2 =w copy 0
  %sh0 =l copy 16
  %v0 =l mul %b0, %sh0
  %v1 =l mul %b1, %sh1
  %v2 =l add %v0, %v1
  %v3 =l add %v2, %b2
  jump @encode_two
@final_2
  %in_ptr =l add %src, %i
  %b0 =w loadub %in_ptr
  %off1 =l add %in_ptr, 1
  %b1 =w loadub %off1
  %b2 =w copy 0
  %sh0 =l copy 16
  %v0 =l mul %b0, %sh0
  %v1 =l mul %b1, %sh1
  %v2 =l add %v0, %v1
  %v3 =l add %v2, %b2
  jump @encode_three
@encode_all
  %shift0 =w copy 18
  %val_mask0 =l copy 262143
  %tmp_val0 =l and %v3, %val_mask0
  %val_shifted0 =l lshr %tmp_val0, %shift0
  %out_loc =l copy %out_ptr
  call @encode_char(%val_shifted0, %out_loc)
  %out_ptr1 =l add %out_ptr, 1
  %shift1 =w copy 12
  %val_mask1 =l copy 4080
  %tmp_val1 =l and %v3, %val_mask1
  %val_shifted1 =l lshr %tmp_val1, %shift1
  %out_loc1 =l copy %out_ptr1
  call @encode_char(%val_shifted1, %out_loc1)
  %out_ptr2 =l add %out_ptr1, 1
  %shift2 =w copy 6
  %val_mask2 =l copy 64512
  %tmp_val2 =l and %v3, %val_mask2
  %val_shifted2 =l lshr %tmp_val2, %shift2
  %out_loc2 =l copy %out_ptr2
  call @encode_char(%val_shifted2, %out_loc2)
  %out_ptr3 =l add %out_ptr2, 1
  %shift3 =w copy 0
  %val_mask3 =l copy 645120
  %val_shifted3 =l lshr %v3, %shift3
  %tmp_val3 =l and %val_shifted3, %val_mask3
  %out_loc3 =l copy %out_ptr3
  call @encode_char(%tmp_val3, %out_loc3)
  %out_ptr_final =l add %out_ptr3, 1
  storel %out_ptr_final, %res
  %inc =l copy 3
  %i_inc =l add %i, %inc
  storel %i_inc, %res
  jump @loop
@encode_two
  %shift0 =w copy 18
  %val_mask0 =l copy 262143
  %tmp_val0 =l and %v3, %val_mask0
  %val_shifted0 =l lshr %tmp_val0, %shift0
  %out_loc =l copy %out_ptr
  call @encode_char(%val_shifted0, %out_loc)
  %out_ptr1 =l add %out_ptr, 1
  %shift1 =w copy 12
  %val_mask1 =l copy 4080
  %tmp_val1 =l and %v3, %val_mask1
  %val_shifted1 =l lshr %tmp_val1, %shift1
  %out_loc1 =l copy %out_ptr1
  call @encode_char(%val_shifted1, %out_loc1)
  %out_ptr2 =l add %out_ptr1, 1
  storeb 61, %out_ptr2
  %out_ptr_final =l add %out_ptr2, 1
  storel %out_ptr_final, %res
  %inc =l copy 3
  %i_inc =l add %i, %inc
  storel %i_inc, %res
  jump @loop
@encode_three
  %shift0 =w copy 18
  %val_mask0 =l copy 262143
  %tmp_val0 =l and %v3, %val_mask0
  %val_shifted0 =l lshr %tmp_val0, %shift0
  %out_loc =l copy %out_ptr
  call @encode_char(%val_shifted0, %out_loc)
  %out_ptr1 =l add %out_ptr, 1
  %shift1 =w copy 12
  %val_mask1 =l copy 4080
  %tmp_val1 =l and %v3, %val_mask1
  %val_shifted1 =l lshr %tmp_val1, %shift1
  %out_loc1 =l copy %out_ptr1
  call @encode_char(%val_shifted1, %out_loc1)
  %out_ptr2 =l add %out_ptr1, 1
  %shift2 =w copy 6
  %val_mask2 =l copy 64512
  %tmp_val2 =l and %v3, %val_mask2
  %val_shifted2 =l lshr %tmp_val2, %shift2
  %out_loc2 =l copy %out_ptr2
  call @encode_char(%val_shifted2, %out_loc2)
  %out_ptr_final =l add %out_ptr2, 1
  storeb 61, %out_ptr_final
  %out_ptr_fin =l add %out_ptr_final, 1
  storel %out_ptr_fin, %res
  %inc =l copy 3
  %i_inc =l add %i, %inc
  storel %i_inc, %res
  jump @loop
@encode_char
  %val_in =l copy %0
  %out_loc =l copy %1
  %lt26 =csltl %val_in, 26
  jnz %lt26, @alpha_upper
  %lt52 =csltl %val_in, 52
  jnz %lt52, @alpha_lower
  %lt62 =csltl %val_in, 62
  jnz %lt62, @digit
  %eq62 =ceql %val_in, 62
  jnz %eq62, @sym_plus
  jump @sym_slash
@alpha_upper
  %base_upper =w copy 65
  %chr =l add %val_in, %base_upper
  storeb %chr, %out_loc
  ret
@alpha_lower
  %sub26 =l sub %val_in, 26
  %base_lower =w copy 97
  %chr2 =l add %sub26, %base_lower
  storeb %chr2, %out_loc
  ret
@digit
  %sub52 =l sub %val_in, 52
  %base_digit =w copy 48
  %chr3 =l add %sub52, %base_digit
  storeb %chr3, %out_loc
  ret
@sym_plus
  %chr4 =w copy 43
  storeb %chr4, %out_loc
  ret
@sym_slash
  %chr5 =w copy 47
  storeb %chr5, %out_loc
  ret
@done
  %final_out =l loadl %res
  ret %final_out
}
