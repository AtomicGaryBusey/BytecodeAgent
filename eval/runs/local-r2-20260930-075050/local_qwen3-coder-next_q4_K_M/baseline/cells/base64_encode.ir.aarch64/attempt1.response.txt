export function l $base64_encode(l %src, l %n, l %out) {
@init
  %res =l alloc1 8
  storel %n, %res
  %i =l copy 0
@loop
  %nval =l loadl %res
  %cmp =csltl %i, %nval
  jnz %cmp, @body
  jump @done
@body
  %rem =w copy 3
  %rmd =w cmodu %nval, %rem
  %cmp2 =ceql %rmd, 0
  jnz %cmp2, @no_pad
  %pad_cnt =l copy 4
  jump @process
@no_pad
  switch %rmd {
    case 1: jump @pad1
    case 2: jump @pad2
  }
@pad1
  %pad_cnt =l copy 2
  jump @process
@pad2
  %pad_cnt =l copy 3
  jump @process
@process
  %in_ptr =l add %src, %i
  %b0 =w loadub %in_ptr
  %off1 =l add %in_ptr, 1
  %cmp_len =csltl %off1, %src
  jnz %cmp_len, @read_b1
  %b1 =w copy 0
  jump @cont2
@read_b1
  %b1 =w loadub %off1
  %off2 =l add %in_ptr, 2
  %cmp_len2 =csltl %off2, %src
  jnz %cmp_len2, @read_b2
  %b2 =w copy 0
  jump @cont3
@read_b2
  %b2 =w loadub %off2
@cont2
@cont3
  %tmp1 =l copy 24
  %sh0 =l copy 16
  %sh1 =l copy 8
  %v0 =l mul %b0, %sh0
  %v1 =l mul %b1, %sh1
  %v2 =l add %v0, %v1
  %v3 =l add %v2, %b2
  %idx0 =w copy 6
  %shift0 =w copy 18
  %val0 =l copy 63
  %tmp_val =l and %v3, %val0
  %val_shifted =l lshr %tmp_val, %shift0
  call @encode_char(%val_shifted, %out)
  %tmp_out =l add %out, 1
  %idx1 =w copy 6
  %shift1 =w copy 12
  %val_mask =l copy 4032
  %tmp_val1 =l and %v3, %val_mask
  %val_shifted1 =l lshr %tmp_val1, %shift1
  call @encode_char(%val_shifted1, %tmp_out)
  %tmp_out1 =l add %tmp_out, 1
  %idx2 =w copy 6
  %shift2 =w copy 6
  %val_mask2 =l copy 161280
  %tmp_val2 =l and %v3, %val_mask2
  %val_shifted2 =l lshr %tmp_val2, %shift2
  call @encode_char(%val_shifted2, %tmp_out1)
  %tmp_out2 =l add %tmp_out1, 1
  %idx3 =w copy 6
  %shift3 =w copy 0
  %val_mask3 =l copy 645120
  %val_mask_shifted =l lshr %v3, %shift3
  %tmp_val3 =l and %val_mask_shifted, %val_mask3
  call @encode_char(%tmp_val3, %tmp_out2)
  %out_final =l add %tmp_out2, 1
  storel %out_final, %res
  %inc =l copy 3
  %i_inc =l add %i, %inc
  storel %i_inc, %res
  jump @check_pad
@pad_chars
  %out_ptr =l loadl %res
  storeb 61, %out_ptr
  %out_next =l add %out_ptr, 1
  storel %out_next, %res
  %tmp_cnt =l loadl %res
  %pad_cnt_sub =l sub %tmp_cnt, 1
  storel %pad_cnt_sub, %res
  %cmp4 =w copy 0
  %cmp5 =ceql %pad_cnt_sub, 0
  jnz %cmp5, @done_pad
  jump @pad_chars
@check_pad
  %tmp_cnt2 =l loadl %res
  %pad_cnt_val =l sub %tmp_cnt2, 3
  %cmp6 =csltl %pad_cnt_val, 0
  jnz %cmp6, @done
  %out_ptr2 =l loadl %res
  storeb 61, %out_ptr2
  %out_next2 =l add %out_ptr2, 1
  storel %out_next2, %res
  jump @pad_done
@done_pad
@pad_done
  jump @loop
@encode_char
  %val_in =l copy %0
  %out_loc =l copy %1
  %lt26 =csltl %val_in, 26
  jnz %lt26, @alpha_upper
  %lt52 =csltl %val_in, 52
  jnz %lt52, @alpha_lower
  %lt62 =csltl %val_in, 62
  jnz %lt62, @digit
  %eq62 =ceql %val_in, 62
  jnz %eq62, @sym_plus
  jump @sym_slash
@alpha_upper
  %base_upper =w copy 65
  %chr =l add %val_in, %base_upper
  storeb %chr, %out_loc
  ret
@alpha_lower
  %sub26 =l sub %val_in, 26
  %base_lower =w copy 97
  %chr2 =l add %sub26, %base_lower
  storeb %chr2, %out_loc
  ret
@digit
  %sub52 =l sub %val_in, 52
  %base_digit =w copy 48
  %chr3 =l add %sub52, %base_digit
  storeb %chr3, %out_loc
  ret
@sym_plus
  %chr4 =w copy 43
  storeb %chr4, %out_loc
  ret
@sym_slash
  %chr5 =w copy 47
  storeb %chr5, %out_loc
  ret
@done
  %final_out =l loadl %res
  ret %final_out
}
