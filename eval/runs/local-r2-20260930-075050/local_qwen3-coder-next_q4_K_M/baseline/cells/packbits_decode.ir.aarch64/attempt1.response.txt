export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
%body:
	%r0 =l copy %src
	%r1 =l copy %out
	%r2 =l copy %n
	%r3 =l copy %cap
	%written =l copy 0
@loop_header
	%have_src =csltl 0, %r2
	jnz %have_src, @has_header
	jmp @done
@has_header
	%h_byte =w loadsb %r0
	%h_signed =l signext %h_byte
	%remaining =l sub %r2, 1
	%r2 =l copy %remaining
	%r0 =l add %r0, 1
	%is_literal =csltl -128, %h_signed
	jnz %is_literal, @literal_check
	%is_repeat =csltl %h_signed, 0
	jnz %is_repeat, @repeat_check
	%h_is_neg128 =ceql %h_signed, -128
	jnz %h_is_neg128, @loop_header
@literal_check
	%extra =l add %h_signed, 1
	%needed =l add %extra, 0
	%fits_src =csltl %needed, %r2+1
	jnz %fits_src, @do_literal
	jmp @err_source
@repeat_check
	%n_repeats =l sub 1, %h_signed
	%fits_src =csltl 0, %r2
	jnz %fits_src, @do_repeat
	jmp @err_source
@do_literal
	%i =l copy 0
@literal_loop
	%cmp_i =csltl %i, %extra
	jnz %cmp_i, @literal_end
	%out_cap_cmp =csltl %written, %r3
	jnz %out_cap_cmp, @write_byte
	jmp @err_capacity
@write_byte
	%byte_to_write =w loadub %r0
	%ptr_out =l add %r1, %written
	storeb %byte_to_write, %ptr_out
	%written =l add %written, 1
	%r0 =l add %r0, 1
	%i =l add %i, 1
	jmp @literal_loop
@literal_end
	jmp @loop_header
@do_repeat
	%repeat_byte =w loadub %r0
	%remaining_src =l sub %r2, 1
	%r2 =l copy %remaining_src
	%r0 =l add %r0, 1
	%i =l copy 0
@repeat_loop
	%cmp_i =csltl %i, %n_repeats
	jnz %cmp_i, @repeat_end
	%out_cap_cmp =csltl %written, %r3
	jnz %out_cap_cmp, @write_repeat_byte
	jmp @err_capacity
@write_repeat_byte
	%ptr_out =l add %r1, %written
	storeb %repeat_byte, %ptr_out
	%written =l add %written, 1
	%i =l add %i, 1
	jmp @repeat_loop
@repeat_end
	jmp @loop_header
@err_source
	%ret =l copy -2
	ret %ret
@err_capacity
	%ret =l copy -1
	ret %ret
@done
	ret %written
}
