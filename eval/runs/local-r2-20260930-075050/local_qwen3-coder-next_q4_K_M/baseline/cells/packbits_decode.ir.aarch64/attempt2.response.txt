export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
%body:
	%r_src =l copy %src
	%r_out =l copy %out
	%r_n =l copy %n
	%r_cap =l copy %cap
	%written =l copy 0

@loop_start:
	%has_more_src =csltl 0, %r_n
	jnz %has_more_src, @process_header
	jmp @done

@process_header:
	%h_byte =w loadsb %r_src
	%h_val =l signext %h_byte
	%n_remaining =l sub %r_n, 1
	%r_n =l copy %n_remaining
	%ptr_next =l add %r_src, 1
	%r_src =l copy %ptr_next

	%is_literal =csltl -128, %h_val
	jnz %is_literal, @literal_case
	%is_repeat =csltl %h_val, 0
	jnz %is_repeat, @repeat_case
	%is_nop =ceql %h_val, -128
	jnz %is_nop, @loop_start

@literal_case:
	%lit_count =l add %h_val, 1
	%needed_src =l add %lit_count, 0
	%fits_source =csltl %needed_src, %r_n+1
	jnz %fits_source, @do_literal
	jmp @err_source

@repeat_case:
	%n_repeats =l sub 1, %h_val
	%has_data_byte =csltl 0, %r_n
	jnz %has_data_byte, @do_repeat
	jmp @err_source

@do_literal:
	%i =l copy 0
@literal_loop:
	%cmp_i =csltl %i, %lit_count
	jnz %cmp_i, @literal_done
	%check_cap =csltl %written, %r_cap
	jnz %check_cap, @write_literal_byte
	jmp @err_capacity

@write_literal_byte:
	%byte_val =w loadub %r_src
	%out_ptr =l add %r_out, %written
	storeb %byte_val, %out_ptr
	%written =l add %written, 1
	%src_ptr_inc =l add %r_src, 1
	%r_src =l copy %src_ptr_inc
	%i =l add %i, 1
	jmp @literal_loop

@literal_done:
	jmp @loop_start

@do_repeat:
	%rep_byte =w loadub %r_src
	%n_remaining_after =l sub %r_n, 1
	%r_n =l copy %n_remaining_after
	%src_ptr_inc2 =l add %r_src, 1
	%r_src =l copy %src_ptr_inc2

	%i =l copy 0
@repeat_loop:
	%cmp_i2 =csltl %i, %n_repeats
	jnz %cmp_i2, @repeat_done
	%check_cap2 =csltl %written, %r_cap
	jnz %check_cap2, @write_repeat_byte
	jmp @err_capacity

@write_repeat_byte:
	%out_ptr2 =l add %r_out, %written
	storeb %rep_byte, %out_ptr2
	%written =l add %written, 1
	%i =l add %i, 1
	jmp @repeat_loop

@repeat_done:
	jmp @loop_start

@err_source:
	%ret_err_src =l copy -2
	ret %ret_err_src

@err_capacity:
	%ret_err_cap =l copy -1
	ret %ret_err_cap

@done:
	ret %written
}
