export function l $qsort_pairs(l %a, l %n) {
entry:
	%0 =l ceql %n, 0
	jnz %0, @empty
	%1 =l ceql %n, 1
	jnz %1, @single
	%2 =l call $alloca(l 32)
	storel %a, %2+0
	storel %n, %2+8
	storel %rax, %2+16
	jmp @init_pivot
@init_pivot:
	%3 =l loadl %2+0
	%4 =l loadl %2+8
	%pivotsz =l shl %4, 1
	%5 =l lshl %pivotsz, 3
	%6 =l add %3, %5
	%7 =l loadl %6+0
	%8 =l loadl %6+8
	jmp @partition_start
@partition_start:
	%9 =l loadl %2+0
	%10 =l loadl %2+8
	%pivot_key =l loadl %7+0
	%pivot_val =l loadl %7+8
	%i =l add %3, 8
	%j =l sub %6, 16
	jmp @outer_loop
@outer_loop:
	%11 =l csltl %i, %j
	jz %11, @after_partition
	%12 =l loadl %i+0
	%lt_pivot_key =l csltl %12, %pivot_key
	jnz %lt_pivot_key, @i_in_place
	%i_incr:
	%i =l add %i, 16
	jmp @outer_loop
@j_decr:
	%j =l sub %j, 16
@j_check:
	%13 =l loadl %j+0
	%gt_pivot_key =l csltl %pivot_key, %13
	jnz %gt_pivot_key, @j_in_place
	%14 =l ceql %13, %pivot_key
	jz %14, @skip_val_check
	jmp @j_decr
@skip_val_check:
	%15 =l loadl %j+8
	%gt_pivot_val =csltl %pivot_val, %15
	jnz %gt_pivot_val, @j_in_place
	jmp @j_decr
@i_in_place:
	%16 =l loadl %j+0
	%lt_pivot_key2 =l csltl %16, %pivot_key
	jz %lt_pivot_key2, @i_j_swapped
	%17 =l ceql %16, %pivot_key
	jnz %17, @j_check
	%18 =l loadl %j+8
	%lt_pivot_val2 =csltl %pivot_val, %18
	jz %lt_pivot_val2, @i_j_swapped
	jmp @j_decr
@i_j_swapped:
	%19 =l loadl %i+0
	%20 =l loadl %j+0
	storel %20, %i+0
	storel %19, %j+0
	%21 =l loadl %i+8
	%22 =l loadl %j+8
	storel %22, %i+8
	storel %21, %j+8
	jmp @outer_loop
@j_in_place:
	%lt_i_key =l csltl %pivot_key, %13
	jnz %lt_i_key, @do_swap
	%eq_i_key =l ceql %13, %pivot_key
	jz %eq_i_key, @check_i_val
	jmp @j_decr
@check_i_val:
	%23 =l loadl %i+8
	%24 =l loadl %j+8
	%geq_i_val =csltl %24, %23
	jnz %geq_i_val, @do_swap
	jmp @j_decr
@do_swap:
	%25 =l loadl %i+0
	%26 =l loadl %j+0
	storel %26, %i+0
	storel %25, %j+0
	%27 =l loadl %i+8
	%28 =l loadl %j+8
	storel %28, %i+8
	storel %27, %j+8
	jmp @outer_loop
@after_partition:
	%29 =l loadl %2+0
	%30 =l loadl %2+8
	%left_size =l sub %i, %29
	%right_ptr =l add %j, 16
	%right_size =l sub %30, %left_size
	%right_size =l sub %right_size, 1
	%left_size =l shr %left_size, 1
	jmp @recurse_left
@recurse_left:
	%31 =l ceql %left_size, 0
	jnz %31, @skip_left
	%32 =l ceql %left_size, 1
	jz %32, @do_left_call
	storel %i, %2+0
	storel %left_size, %2+8
	%left_key_addr =l loadl %i+0
	%left_val_addr =l loadl %i+8
	jmp @exit_ret
@do_left_call:
	storel %3, %2+0
	storel %left_size, %2+8
	%r =l call $qsort_pairs(l %3, l %left_size)
@skip_left:
	%33 =l ceql %right_size, 0
	jnz %33, @done
	%34 =l ceql %right_size, 1
	jz %34, @exit_ret
	storel %right_ptr, %2+0
	storel %right_size, %2+8
	%r =l call $qsort_pairs(l %right_ptr, l %right_size)
@done:
	exit_ret:
	%ret_val =l add %left_size, 1
	%ret_val =l sub %ret_val, 1
	ret %ret_val
@single:
	ret 0
@empty:
	ret 0
}
