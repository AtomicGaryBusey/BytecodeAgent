export function l $popcount(l %x) {
@entry
	l %result =l 0
	l %n =l %x
@loop
	w %cond =ceql %n, 0
	jnz %cond, @done
	l %low_bit =andl %n, 1
	w %is_zero =ceql %low_bit, 0
	jnz %is_zero, @skip
	l %result =addl %result, 1
@skip
	l %shift_amount =l 1
	l %new_n =shrl %n, %shift_amount
	%n =%new_n
	jmp @loop
@done
	ret %result
}
