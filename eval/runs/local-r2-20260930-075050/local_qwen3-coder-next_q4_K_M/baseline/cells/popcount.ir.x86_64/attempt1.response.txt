export function l $popcount(l %x) {
@entry
	l %result =l 0
	l %n =l %x
@loop
	w %cond =ceql %n, 0
	jnz %cond, @done
	w %bit =csltl %n, 0
	w %is_set =ceql %bit, 0
	w %not_set =ceql %is_set, 1
	l %mask =l 1
	l %updated =subl %n, %mask
	l %new_n =andl %updated, %n
	l %result =addl %result, 1
	%n =%new_n
	jmp @loop
@done
	ret %result
}
