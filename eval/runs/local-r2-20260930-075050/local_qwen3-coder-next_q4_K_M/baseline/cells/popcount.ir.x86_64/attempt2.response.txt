export function l $popcount(l %x) {
@entry
	l %result =l 0
	l %n =l %x
@loop
	w %cond =ceql %n, 0
	jnz %cond, @done
	l %mask =l 1
	w %bit =csltl %n, 63
	w %is_set =ceql %bit, 0
	jnz %is_set, @skip
	l %updated =subl %n, %mask
	l %new_n =andl %updated, %n
	l %result =addl %result, 1
	%n =%new_n
	jmp @loop
@skip
	l %shift_amount =l 63
	l %high_bit =shrl %n, %shift_amount
	w %is_high_set =ceql %high_bit, 1
	jnz %is_high_set, @handle_high
	l %shift_amount2 =l 1
	l %new_n2 =shll %n, %shift_amount2
	%n =%new_n2
	jmp @loop
@handle_high
	l %updated2 =subl %n, %mask
	l %new_n3 =andl %updated2, %n
	l %result2 =addl %result, 1
	l %shift_amount3 =l 1
	l %new_n4 =shll %new_n3, %shift_amount3
	%n =%new_n4
	l %result =%result2
	jmp @loop
@done
	ret %result
}
