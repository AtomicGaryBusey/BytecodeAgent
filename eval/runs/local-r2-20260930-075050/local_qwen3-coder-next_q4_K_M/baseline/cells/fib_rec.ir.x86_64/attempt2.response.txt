export function l $fib_rec(l %n) {
	%0 =l ceql %n, 0
	w %0, @base0
	%1 =l ceql %n, 1
	w %1, @base1
	jmp @recursive

@base0:
	%r =l const 0
	ret %r

@base1:
	%r =l const 1
	ret %r

@recursive:
	%sub1 =l subl %n, 1
	%arg0 =l copy %sub1
	%r1 =l call $fib_rec(l %arg0)
	%sub2 =l subl %n, 2
	%arg1 =l copy %sub2
	%r2 =l call $fib_rec(l %arg1)
	%sum =l addl %r1, %r2
	ret %sum
}
