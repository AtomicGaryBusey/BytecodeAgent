export function l $fib_rec(l %n) {
@entry
	%is_zero = w ceql %n, 0
	w %is_zero, @base_case_0
	%is_one = w ceql %n, 1
	w %is_one, @base_case_1
	jmp @recursive_case

@base_case_0:
	%result = l const 0
	ret %result

@base_case_1:
	%result = l const 1
	ret %result

@recursive_case:
	%sub1 = l subl %n, 1
	%arg1 = l copy %sub1
	%ret_val1 = l call $fib_rec(l %arg1)
	%sub2 = l subl %n, 2
	%arg2 = l copy %sub2
	%ret_val2 = l call $fib_rec(l %arg2)
	%sum = l addl %ret_val1, %ret_val2
	ret %sum
}
