export function l $count_ugt(l %a, l %n, l %t) {
entry:
	l %i = 0l
	l %count = 0l
	w %loop_cond = csltl %n, 0l
	jnz %loop_cond, .exit
	jmp .loop_body

.loop_body:
	l %val = loadl %a
	w %cond = csgtul %val, %t
	w %inc = select.w %cond, 1w, 0w
	l %add = addl %count, %inc
	storel %add, %count
	l %new_i = addl %i, 1l
	l %new_a = addl %a, 8l
	l %n_cmp = subl %n, 1l
	l %i_cmp = addl %new_i, 0l
	w %loop_cond2 = csltl %new_i, %n
	jnz %loop_cond2, .loop_body

.exit:
	l %result = loadl %count
	ret %result
}
