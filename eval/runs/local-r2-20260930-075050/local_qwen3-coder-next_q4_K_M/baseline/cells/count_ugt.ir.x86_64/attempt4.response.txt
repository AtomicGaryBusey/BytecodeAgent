export function l $count_ugt(l %a, l %n, l %t) {
@entry
	l %i = 0l
	l %count = 0l
	w %done = ceql %n, 0l
	jnz %done, @exit
@loop_body:
	l %val = loadl l %a
	w %cmp = csgtul %val, %t
	w %inc = select.w %cmp, 1w, 0w
	l %count_new = addl %count, %inc
	l %i_new = addl %i, 1l
	l %a_next = addl %a, 8l
	storel %count_new, l %count
	l %a_store = copy %a_next
	storel %a_store, l %a
	l %i_store = copy %i_new
	storel %i_store, l %i
	w %loop_cond = csltl %i_new, %n
	jnz %loop_cond, @loop_body
@exit:
	l %result = loadl l %count
	ret %result
}
