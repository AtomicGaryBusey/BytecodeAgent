bmax:
    mov rax, qword ptr [rdi]        # load first element as initial max
    xor rcx, rcx                    # index = 0
.Lloop:
    inc rcx                         # increment index
    cmp rcx, rsi                    # compare index with n
    jge .Ldone                      # exit loop if index >= n
    mov rbx, qword ptr [rdi + rcx*8] # load a[index]
    cmp rbx, rax                    # compare current element with max
    cmovl rax, rbx                  # update max if current is larger
    jmp .Lloop                      # continue loop
.Ldone:
    ret                             # return max value in rax
