reverse:
    cmp rsi, 1
    jle .Ldone
    xor rcx, rcx
    lea rdx, [rsi - 1]
.Lloop:
    mov rax, qword ptr [rdi + rcx*8]
    mov rbx, qword ptr [rdi + rdx*8]
    mov qword ptr [rdi + rcx*8], rbx
    mov qword ptr [rdi + rdx*8], rax
    inc rcx
    dec rdx
    cmp rcx, rdx
    jle .Lloop
.Ldone:
    xor rax, rax
    ret
