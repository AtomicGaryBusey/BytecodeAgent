reverse:
    cmp rsi, 1
    jle .Ldone
    push rbx
    xor rcx, rcx
    lea rbx, [rsi - 1]
.Lloop:
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rbx*8]
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rbx*8], rax
    inc rcx
    dec rbx
    cmp rcx, rbx
    jle .Lloop
    pop rbx
.Ldone:
    xor rax, rax
    ret
