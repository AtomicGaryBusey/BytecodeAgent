count_ugt:
    xor rax, rax           # result counter
    test rsi, rsi          # check if n == 0
    jz .Ldone              # if n is zero, return 0
    xor rcx, rcx           # loop index i = 0

.Lloop:
    cmp rcx, rsi           # i < n?
    jge .Ldone             # exit loop when i >= n
    mov rdx, qword ptr [rdi + rcx*8]   # load a[i]
    cmp rdx, rdx           # set flags for comparison (just to check if greater)
    ja .Lgreater           # jump if a[i] > t (unsigned comparison using carry flag)
    inc rcx                # increment index
    jmp .Lloop

.Lgreater:
    inc rax                # increment count
    inc rcx                # increment index
    jmp .Lloop

.Ldone:
    ret
