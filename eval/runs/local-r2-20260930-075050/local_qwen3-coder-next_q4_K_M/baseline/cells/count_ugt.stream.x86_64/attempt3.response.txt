count_ugt:
    xor rax, rax           # result counter = 0
    test rsi, rsi          # check if n == 0
    jz .Ldone              # if n is zero, return 0

    xor rcx, rcx           # loop index i = 0
    mov rbx, rdx           # save t in rbx (callee-saved register)

.Lloop:
    cmp rcx, rsi           # i < n?
    jge .Ldone             # exit when i >= n

    mov rdx, qword ptr [rdi + rcx*8]  # load a[i]
    cmp rdx, rbx           # compare a[i] with t
    jbe .Lnext             # skip if a[i] <= t (unsigned)
    inc rax                # increment count

.Lnext:
    inc rcx                # i++
    jmp .Lloop             # continue loop

.Ldone:
    ret                    # return count
