export function l $shift_sat(...) {
%entry:
	%x = param.l
	%n = param.l
	%cmp = cgeq.l %n, 64
	jnz %cmp, %if_n_ge_64, %if_n_lt_64

%if_n_ge_64:
	%sar = lshr.s %x, 63
	%shl = shl.l %x, 0
	%add = add.l %shl, %sar
	ret.l %add

%if_n_lt_64:
	%shl = shl.l %x, %n
	%sar = lshr.s %x, %n
	%add = add.l %shl, %sar
	ret.l %add
}
