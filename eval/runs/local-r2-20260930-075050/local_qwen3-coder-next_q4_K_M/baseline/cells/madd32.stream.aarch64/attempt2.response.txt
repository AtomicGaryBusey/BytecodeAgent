madd32:
    uxtw x9, w0                      // x9 = a32 (unsigned extend low 32 of a to 64-bit)
    uxtw x10, w1                     // x10 = b32 (unsigned extend low 32 of b to 64-bit)
    smull x11, w9, w10               // x11 = a32 * b32 as signed 32-bit multiply, result in lower 64 bits
    uxtw x12, w2                     // x12 = c32 (unsigned extend low 32 of c to 64-bit)
    add x0, x11, x12                 // add c32 to product (64-bit intermediate)
    and x0, x0, #0xffffffff          // keep only low 32 bits
    sxtw x0, w0                      // sign-extend 32-bit result to int64_t
    ret
