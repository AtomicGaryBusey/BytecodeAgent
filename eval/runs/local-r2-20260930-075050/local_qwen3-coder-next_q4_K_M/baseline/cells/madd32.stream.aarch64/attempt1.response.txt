madd32:
    and x1, x0, #0xffffffff          // x1 = a32 (low 32 bits of a)
    and x2, x1, x1                   // copy a32 to x2 for multiply
    lsl x3, x1, #32                  // shift b into upper 32 bits for smlsl
    lsr x4, x1, #0                   // x4 = a32 (reuse)
    and x5, x2, #0xffffffff          // x5 = a32 (clean)
    and x6, x1, #0xffffffff          // x6 = b32 (will overwrite)
    lsl x7, x1, #0                   // dummy move to free registers
    mov x8, xzr                      // clear for multiplication
    uxtw x9, w1                      // x9 = a32 as signed 32-bit
    uxtw x10, w2                     // x10 = b32 as signed 32-bit
    smull x11, w9, w10               // x11 = a32 * b32 (64-bit result)
    sxtw x12, w3                     // c32 sign-extended to 64-bit
    add x0, x11, x12                 // add c32
    and x0, x0, #0xffffffff          // keep low 32 bits
    sxtw x0, w0                    // sign-extend to int64_t
    ret                              // return
