count_nonzero_prefix:
    cbz x1, .Ldone       // if n == 0, return 0
    mov x2, xzr          // i = 0
.Lloop:
    cmp x2, x1           // compare i with n
    b.ge .Ldone          // if i >= n, done (return i)
    ldr x3, [x0, x2, lsl #3]  // load a[i]
    cbz x3, .Ldone       // if a[i] == 0, return i
    add x2, x2, #1       // i++
    b .Lloop             // repeat
.Ldone:
    mov x0, x2           // return i (result in x0)
    ret
