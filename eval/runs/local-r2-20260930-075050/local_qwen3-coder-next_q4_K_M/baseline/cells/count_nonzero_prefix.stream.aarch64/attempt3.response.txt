count_nonzero_prefix:
    mov x1, xzr          // i = 0
    cbz x2, .Ldone       // if n == 0, return 0
.Lloop:
    cmp x1, x2           // compare i with n
    b.ge .Ldone          // if i >= n, done (return i)
    ldr x3, [x0, x1, lsl #3]  // load a[i]
    cbz x3, .Ldone       // if a[i] == 0, return i
    add x1, x1, #1       // i++
    b .Lloop             // repeat
.Ldone:
    mov x0, x1           // return i (result in x0)
    ret
