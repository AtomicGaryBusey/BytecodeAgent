export function l $reverse(l %a, l %n) {
@start
	lc.l %zero, 0
	lc.l %one, 1
	lc.l %eight, 8
	mov.l %i, %zero
	sub.l %n_minus_1, %n, %one
	csltl %has_elements, %n, %one
	jnz %has_elements, @loop_body
	jmp @done

@loop_body
	mull.l %offset_i, %i, %eight
	add.l %addr_i, %a, %offset_i
	sub.l %j, %n_minus_1, %i
	csltl %swap_check, %i, %j
	jnz %swap_check, @process_swap
	jmp @done

@process_swap
	mull.l %offset_j, %j, %eight
	add.l %addr_j, %a, %offset_j
	loadl %val_i, %addr_i
	loadl %val_j, %addr_j
	storel %val_j, %addr_i
	storel %val_i, %addr_j
	add.l %i, %i, %one
	jmp @loop_body

@done
	ret %zero
}
