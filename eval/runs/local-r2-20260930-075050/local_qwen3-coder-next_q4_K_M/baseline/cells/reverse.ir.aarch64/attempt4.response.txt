export function l $reverse(l %a, l %n) {
@start
	lc.l %zero, 0
	lc.l %one, 1
	lc.l %eight, 8
	mov.l %i, %zero
	sub.l %high, %n, %one
	csltl %check_start, %n, %one
	jnz %check_start, @done

@loop
	csltl %cond, %i, %high
	jz %cond, @done
	mull.l %off_i, %i, %eight
	add.l %addr_i, %a, %off_i
	mull.l %off_high, %high, %eight
	add.l %addr_high, %a, %off_high
	loadl %val_i, l %addr_i
	loadl %val_high, l %addr_high
	storel %val_high, l %addr_i
	storel %val_i, l %addr_high
	add.l %i, %i, %one
	sub.l %high, %high, %one
	jmp @loop

@done
	ret %zero
}
