export function l $reverse(l %a, l %n) {
@start
	lc.w %zero, 0
	lc.l %eight, 8
	mov.l %i, %zero
	lsl.l %hi, %n, 1
	lsr.l %n2, %hi, 3
	csltl %has_elements, %n, %one
	jnz %has_elements, @loop_body
	jmp @done

@loop_body
	lc.l %i8, 8
	mull.l %offset_i, %i, %eight
	add.l %addr_i, %a, %offset_i
	sub.l %j, %n2, %i
	sub.w %j_check, %j, %one
	csltl %j_valid, %zero, %j_check
	jnz %j_valid, @process_swap
	jmp @done

@process_swap
	lc.l %j8, 8
	mull.l %offset_j, %j, %eight
	sub.l %n_minus_1, %n, %one
	sub.l %actual_j, %n_minus_1, %i
	lsl.l %offset_j2, %actual_j, 3
	add.l %addr_j, %a, %offset_j2
	loadl %val_i, %addr_i
	loadl %val_j, %addr_j
	storel %val_j, %addr_i
	storel %val_i, %addr_j
	add.l %i, %i, %one
	lc.l %limit, 0
	csltl %loop_cond, %i, %n2
	jnz %loop_cond, @process_swap

@done
	mov.l %ret_val, %zero
	ret %ret_val
}
