bmax:
    mov x19, x0                // save base pointer for array
    mov x20, x1                // save n
    ldr x0, [x19]              // load first element as initial max
    sub x21, x20, #1           // counter = n - 1
    cbz x21, .Ldone            // if n == 1, done
    add x19, x19, #8           // skip first element (already loaded)
.Lloop:
    ldr x22, [x19], #8         // load next element and increment pointer
    cmp x22, x0                // compare with current max
    csel x0, x22, x0, gt       // update max if larger
    sub x21, x21, #1           // decrement counter
    cbnz x21, .Lloop           // loop until done
.Ldone:
    ret                        // return with max in x0
