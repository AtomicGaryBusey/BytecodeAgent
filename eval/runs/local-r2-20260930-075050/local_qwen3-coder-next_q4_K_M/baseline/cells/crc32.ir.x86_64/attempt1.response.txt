export function l $crc32(l %p, l %n) {
@start
	%crc_hi =l extuw 0
	%crc_lo =l extuw -1
	%i =l extuw 0
@loop_byte
	%csltl %done, %i, %n
	jnz %done, @exit
	%p_i =w loadub %p
	%crc_lo_w =w extuw %crc_lo
	%crc_lo_w =w xor %crc_lo_w, %p_i
	%tmp =l extuw %crc_lo_w
	%tmp2 =l and %tmp, 1
	%tmp3 =l ceql %tmp2, 0
	jnz %tmp3, @no_xor
	%crc_lo_w =w xor %crc_lo_w, -0x741d8f70
@no_xor
	%crc_lo_w =w shr %crc_lo_w, 1
	%tmp4 =l extuw %crc_lo_w
	%tmp5 =l and %tmp, 1
	%tmp6 =l ceql %tmp5, 0
	jnz %tmp6, @bit_done
	%crc_hi =l or %crc_hi, -0x80000000
@bit_done
	%i =l add %i, 1
	%p =l add %p, 1
	jmp @loop_byte
@exit
	%final_lo =w extuw %crc_lo
	%crc_lo_w =w xor %final_lo, -1
	%result =l extuw %crc_lo_w
	ret %result
}
