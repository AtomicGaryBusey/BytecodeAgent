export function l $crc32(l %p, l %n) {
@start
	%crc =w xor 4294967295, 4294967295
	%crc =w xor %crc, -1
	%i =l extuw 0
@loop_byte
	%done =w csltl %i, %n
	jnz %done, @exit
	%byte =w loadub %p
	%crc =w xor %crc, %byte
	%i8 =l extuw 0
@loop_bit
	%done8 =w csltl %i8, 8
	jnz %done8, @next_byte
	%lsb =w and %crc, 1
	%cond =w ceql %lsb, 1
	jnz %cond, @do_xor
	%crc =w shr %crc, 1
	jmp @bit_done
@do_xor
	%crc =w shr %crc, 1
	%crc =w xor %crc, -0x741d8f70
@bit_done
	%i8 =l add %i8, 1
	jmp @loop_bit
@next_byte
	%p =l add %p, 1
	%i =l add %i, 1
	jmp @loop_byte
@exit
	%crc_final =w xor %crc, -1
	%ret_val =l extuw %crc_final
	ret %ret_val
}
