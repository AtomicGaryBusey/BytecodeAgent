export function l $crc32(l %p, l %n) {
@start
	%crc =l extuw 4294967295
	%i =l extuw 0
@loop_byte
	%done =w csltl %i, %n
	jnz %done, @exit
	%byte =w loadub %p
	%crc_w =w extuw %crc
	%crc_w =w xor %crc_w, %byte
	%i8 =l extuw 0
@loop_bit
	%done8 =w csltl %i8, 8
	jnz %done8, @next_byte
	%lsb =w and %crc_w, 1
	%cond =w ceql %lsb, 0
	jnz %cond, @no_xor_bit
	%crc_w =w xor %crc_w, -0x741d8f70
@no_xor_bit
	%crc_w =w shr %crc_w, 1
	%i8 =l add %i8, 1
	jmp @loop_bit
@next_byte
	%crc =l extuw %crc_w
	%i =l add %i, 1
	%p =l add %p, 1
	jmp @loop_byte
@exit
	%crc_final =w extuw %crc
	%result =w xor %crc_final, -1
	%ret_val =l extuw %result
	ret %ret_val
}
