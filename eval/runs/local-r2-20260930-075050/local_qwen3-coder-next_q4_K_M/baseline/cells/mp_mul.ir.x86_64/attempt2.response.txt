export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%zero = constl 0
	%one = constl 1
	%nai = extsw %na
	%nbi = extsw %nb
	%c = csltl %nai, %zero
	%d = csltl %nbi, %zero
	%ea = orw %c, %d
	jnz %ea, @ret_zero

	%nai = zextl %na
	%nbi = zextl %nb
	%nn = addl %nai, %nbi

	%i = constl 0
@outer_loop_init
	jmp @outer_check

@outer_check
	%c = csltl %i, %nai
	jz %c, @ret_zero
	jmp @outer_body

@outer_body
	%ai = loadl %a+8*%i
	%j = constl 0
	%carry = constl 0
@inner_loop
	%bj = loadl %b+8*%j
	%sij = mulu l %ai, %bj
	%sumlo = addl %sij, %carry
	%ci = cultl %sumlo, %sij
	%carry_hi = zextl %ci
	%sumhi = addl %sumlo, %carry_hi
	%c2 = cultl %sumhi, %sumlo
	%carry_next = zextl %c2
	storel %sumhi, %r+8*%j

	%jinc = addl %j, %one
	%cj = csltl %jinc, %nbi
	jnz %cj, @inner_done
	%j = %jinc
	jmp @inner_loop

@inner_done
	storel %carry_next, %r+8*%jinc
	%k = addl %jinc, %one

	%iinc = addl %i, %one
	%ci2 = csltl %iinc, %nai
	jnz %ci2, @outer_done
	%i = %iinc
	jmp @outer_loop_init

@outer_done
	%ri = constl 0
	jmp @scan_zero

@ret_zero
	storel %zero, %r+8*%nn
	retl %zero

@scan_zero
	%rtmp = loadl %r+8*%ri
	%isnz = ceql %rtmp, %zero
	jnz %isnz, @next limb
	retl %ri

@next limb
	%riinc = addl %ri, %one
	%cmax = csltl %riinc, %nn
	jnz %cmax, @ret_zero
	%ri = %riinc
	jmp @scan_zero
}
