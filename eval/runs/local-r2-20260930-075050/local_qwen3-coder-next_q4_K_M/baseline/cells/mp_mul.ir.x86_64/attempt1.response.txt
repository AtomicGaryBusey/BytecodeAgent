export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
	block:
	%zero = constl 0
	%one = constl 1
	%nai = extsw %na
	%nbi = extsw %nb
	%c = csltl %nai, %zero
	%d = csltl %nbi, %zero
	%ea = orw %c, %d
	jnz %ea, .ret_zero
	%nai = zextl %na
	%nbi = zextl %nb
	%nn = addl %nai, %nbi
	storel %zero, %r+8*%nn
	%ri = constl 0
	jmp .outer_loop_init

.outer_loop:
	%ai = loadl %a+8*%i
	%j = constl 0
	jmp .inner_loop

.inner_loop:
	%bj = loadl %b+8*%j
	%sij = mulu l %ai, %bj
	%sum = addl %sij, %carry
	%ci = cultl %sum, %sij
	%carry1 = addw %ci, %carry_next
	%carry_next = zextl %carry1
	storel %sum, %r+8*%j
	%jinc = addl %j, %one
	%cj = csltl %jinc, %nbi
	jnz %cj, .inner_done
	%j = %jinc
	jmp .inner_loop

.inner_done:
	%sum = %carry_next
	storel %sum, %r+8*%j
	%iinc = addl %i, %one
	%ci2 = csltl %iinc, %nai
	jnz %ci2, .outer_done
	%i = %iinc
	jmp .outer_loop

.outer_loop_init:
	%i = constl 0
	jmp .outer_check

.outer_check:
	%c = csltl %i, %nai
	jz %c, .ret_zero
	jmp .outer_loop

.ret_zero:
	storel %zero, %r+8*%nn
	retl %one
}
