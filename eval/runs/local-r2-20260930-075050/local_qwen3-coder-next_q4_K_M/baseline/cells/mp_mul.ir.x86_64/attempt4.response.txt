export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%zero = 0
	%one = 1
	%nai = zextl %na
	%nbi = zextl %nb

	%c = csltl %nai, %zero
	%d = csltl %nbi, %zero
	%ea = or %c, %d
	jnz %ea, @ret_zero

	%nn = addl %nai, %nbi
	jnz %nn, @main_init
	retl %zero

@main_init
	%i = const 0
@outer_loop
	%c = csltl %i, %nai
	jz %c, @ret_significant
	%ai = load l %a+8*%i
	%j = const 0
	%carry_lo = const 0
	%carry_hi = const 0
@inner_loop
	%c = csltl %j, %nbi
	jz %c, @inner_done
	%bj = load l %b+8*%j
	%prod = mullu %ai, %bj
	%prod_lo = zextl %prod
	%prod_hi = srl %prod, 32
	%high_part = srl %prod, 64
	%s1 = addl %prod_lo, %carry_lo
	%c1 = ult %s1, %prod_lo
	%s2 = addl %s1, %carry_hi
	%c2 = ult %s2, %s1
	%total_carry = addl %c1, %c2
	%s3 = addl %s2, %total_carry
	%rj = load l %r+8*%j
	%sum = addl %s3, %rj
	storel %sum, %r+8*%j

	%carry_lo_next = addl %prod_hi, %high_part
	%carry_lo = addl %carry_lo_next, %c2
	%carry_lo = addl %carry_lo, %total_carry

	%jinc = addl %j, %one
	%i_unchanged = %i
	j = %jinc
	carry_lo = %carry_lo
	carry_hi = %carry_hi
	jmp @inner_loop

@inner_done
	%c = csltl %j, %nn
	jz %c, @next_i
	%sum = addl %carry_lo, %carry_hi
	storel %sum, %r+8*%j

	%i_unchanged = %i
	jmp @next_i

@next_i
	%j = const 0
	%carry_lo = const 0
	%carry_hi = const 0
@clear_high
	%c = csltl %j, %nn
	jz %c, @outer_inc
	storel %zero, %r+8*%j

	%jinc = addl %j, %one
	j = %jinc
	jmp @clear_high

@outer_inc
	%iinc = addl %i, %one
	i = %iinc
	jmp @outer_loop

@ret_significant
	%k = subl %nn, %one
@scan_zero
	%c = csltl %k, %zero
	jnz %c, @return_k
	%ri = load l %r+8*%k
	%is_zero = ceql %ri, %zero
	jnz %is_zero, @scan_zero
	%kinc = addl %k, %one
	retl %kinc

@return_k
	retl %one

@ret_zero
	storel %zero, %r
	retl %zero
}
