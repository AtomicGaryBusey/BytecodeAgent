export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%zero = const 0
	%one = const 1
	%nai = sextl %na
	%nbi = sextl %nb
	%c = slt %nai, %zero
	%d = slt %nbi, %zero
	%ea = or %c, %d
	jnz %ea, @ret_zero

	%nai = zextl %na
	%nbi = zextl %nb
	%nn = addl %nai, %nbi

	%i = const 0
@outer_loop_init
	jmp @outer_check

@outer_check
	%c = slt %i, %nai
	jz %c, @ret_zero
	jmp @outer_body

@outer_body
	%ai = load l %a+8*%i
	%j = const 0
	%carry_lo = const 0
	%carry_hi = const 0
@inner_loop
	%bj = load l %b+8*%j
	%t0 = mulu l %ai, %bj
	%prod_lo = zextl %t0
	%prod_hi = sextl %t0
	%s1 = addl %prod_lo, %carry_lo
	%c1 = ult %s1, %prod_lo
	%carry_lo_next = zextl %c1
	%s2 = addl %s1, %carry_lo_next
	%s3 = addl %s2, %carry_hi
	%c2 = ult %s3, %s2
	%carry_hi_next = zextl %c2

	store l %s3, %r+8*%j

	%jinc = addl %j, %one
	%cj = slt %jinc, %nbi
	jnz %cj, @inner_done
	%j = %jinc
	%carry_lo = %s1
	%carry_hi = %carry_lo_next
	jmp @inner_loop

@inner_done
	storel %carry_lo_next, %r+8*%jinc
	%k = addl %jinc, %one

	%iinc = addl %i, %one
	%ci2 = slt %iinc, %nai
	jnz %ci2, @outer_done
	%i = %iinc
	jmp @outer_loop_init

@outer_done
	storel %zero, %r+8*%nn
	retl %zero

@ret_zero
	storel %zero, %r
	retl %zero
}
