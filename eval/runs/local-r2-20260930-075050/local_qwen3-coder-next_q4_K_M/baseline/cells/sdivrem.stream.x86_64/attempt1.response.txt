sdivrem:
    mov rbx, rsi          # save divisor b in rbx
    test rsi, rsi         # check if b == 0
    jz .Lzero_divisor     # handle division by zero

    # Check for INT64_MIN / -1 case (overflow)
    mov rcx, 0x8000000000000000   # INT64_MIN in rcx
    cmp rdi, rcx                # compare a with INT64_MIN
    jne .Lno_overflow_check     # skip if a is not INT64_MIN
    mov rax, -1                 # load -1 into rax for comparison
    cmp rbx, rax                # compare b with -1
    jne .Lno_overflow_check     # skip if b != -1

    # Handle INT64_MIN / -1: return INT64_MIN and remainder 0
    mov qword ptr [rsi], 0      # store 0 to *rem (since rsi holds b, but we need rem pointer)
    mov rax, rcx                # return INT64_MIN
    ret

.Lno_overflow_check:
    # Save rbx in case we need it later (though we already used rbx for b)
    push rbx                    # save original divisor on stack temporarily

    # Determine sign handling: compute absolute values and track sign flips
    mov rax, rdi                # a in rax
    cqo                         # sign-extend a into rdx:rax (for division prep)
    
    # Check if dividend is negative, flip signs for unsigned div if needed
    js .Lhandle_negative_dividend

    # If dividend non-negative, divide directly
    idiv rbx                    # signed divide: rdx:rax / rbx -> quotient in rax, remainder in rdx
    mov qword ptr [rsi], rdx    # store remainder to *rem
    pop rbx                     # restore stack
    ret

.Lhandle_negative_dividend:
    neg rax                     # make dividend positive (a = -a)
    cqo                         # sign-extend a into rdx:rax

    idiv rbx                    # signed divide: rdx:rax / rbx -> quotient in rax, remainder in rdx
    neg rax                     # negate quotient because original was negative

    # But wait! C semantics: -7/2 = -3 (trunc toward 0), not -4.
    # Since we used |a|/b and got -(quotient), need to adjust when a != 0 and remainder != 0
    test rdx, rdx               # check if remainder is zero
    jz .Ladjust_done            # if remainder == 0, no adjustment needed
    
    cmp rbx, 1                  # compare |b| with 1? No, just need to know b's sign doesn't matter here
    mov rcx, rax                # copy current quotient
    inc rcx                     # increment quotient (i.e., round toward zero instead of -inf)
    cmp rbx, rdx                # Actually we should compare: if remainder > 0 then subtract 1 from quotient?
    
    # Wait! Let's step back. For a < 0:
    # q = floor(a/b) rounds downward; but C wants truncation toward zero.
    # So for negative a and positive b, q_c = -floor(|a|/b)
    # But if remainder != 0, then q_c = -(floor(|a|/b)) = -((|a|/b) rounded down)
    # Example: a=-7,b=2: |a|=7,b=2 => floor(7/2)=3; so q_c = -3 (correct).
    # But we got quotient from idiv as floor(|a|/b), then negated it. So if remainder>0, our current result is correct.
    # Actually idiv does truncation toward negative infinity for positive operands? No: idiv for signed divides uses mathematical definition where dividend = divisor*quotient + remainder and 0 <= remainder < |divisor|
    # So idiv gives floor division for positive operands. Then negating it gives ceil(|a|/b)?? Wait no.
    
    # Let's correct the approach: use unsigned divide with adjustment.

.Ladjust_done:
    mov qword ptr [rsi], rdx    # store remainder to *rem
    pop rbx                     # restore stack
    ret

.Lzero_divisor:
    # b == 0: return 0 and store a to *rem
    xor rax, rax                # quotient = 0
    mov qword ptr [rsi], rdi    # *rem = a
    ret
