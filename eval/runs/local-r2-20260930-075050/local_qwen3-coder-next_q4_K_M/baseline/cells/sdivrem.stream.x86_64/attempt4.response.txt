sdivrem:
    push rbp
    mov rbp, rsp
    sub rsp, 16                 # make room for local storage (8-byte remainder + alignment)
    mov qword ptr [rbp-8], rsi  # save b in stack slot (callee-saved register not used)
    test esi, esi             # check if b == 0 (use 32-bit to clear upper bits)
    jz .Lzero_divisor

    # Check for INT64_MIN / -1 case
    mov rax, rdi              # a
    mov rcx, 0x8000000000000000  # INT64_MIN
    cmp rax, rcx
    jne .Lno_overflow_check
    mov rdx, qword ptr [rbp-8]  # reload b
    cmp rdx, -1
    jne .Lno_overflow_check

    # Handle INT64_MIN / -1: return INT64_MIN, remainder 0
    mov qword ptr [rdx], 0      # *rem = 0 (rdx is rem pointer from ABI)
    mov rax, rcx                # return INT64_MIN
    leave
    ret

.Lno_overflow_check:
    mov rax, rdi              # a in rax
    cdq                       # sign-extend eax into edx (32-bit), but we need 64-bit
    cqo                       # sign-extend rax into rdx:rax
    mov rbx, qword ptr [rbp-8] # load b into rbx for division
    idiv rbx                 # signed divide: rdx:rax / rbx -> q in rax, r in rdx

    # Store remainder to *rem (rdi = a, rsi = b not needed anymore; rdx has rem pointer)
    mov rcx, rdx             # save remainder
    mov qword ptr [rdx], rcx # store remainder to *rem
    ret

.Lzero_divisor:
    xor eax, eax            # quotient = 0 (clear upper bits too)
    mov qword ptr [rdx], rdi # *rem = a
    leave
    ret
