sdivrem:
    mov rbx, rsi              # save divisor b in rbx (callee-saved)
    test rsi, rsi             # check if b == 0
    jz .Lzero_divisor         # handle division by zero

    # Check for INT64_MIN / -1 case (overflow)
    mov rcx, 0x8000000000000000   # INT64_MIN in rcx
    cmp rdi, rcx                # compare a with INT64_MIN
    jne .Lno_overflow_check     # skip if a is not INT64_MIN
    mov rax, -1                 # load -1 into rax for comparison
    cmp rbx, rax                # compare b with -1
    jne .Lno_overflow_check     # skip if b != -1

    # Handle INT64_MIN / -1: return INT64_MIN and remainder 0
    mov qword ptr [rdx], 0      # store 0 to *rem (rdx has rem pointer from ABI)
    mov rax, rcx                # return INT64_MIN
    ret

.Lno_overflow_check:
    push rbx                    # save divisor on stack (callee-saved, also helps alignment)
    sub rsp, 8                  # make room for saved rem pointer; now 16-byte aligned before calls
    mov [rsp + 8], rdx          # save rem pointer (at [rsp+8] after sub rsp,8)

    mov rax, rdi                # a in rax
    cqo                         # sign-extend a into rdx:rax

    idiv rbx                    # signed divide: rdx:rax / rbx -> quotient in rax, remainder in rdx

    mov r8, rdx                 # save remainder in r8 (callee-saved register)
    mov rax, [rsp]              # get divisor back
    mov rdx, [rsp + 8]          # restore rem pointer to rdx
    mov qword ptr [rdx], r8     # store remainder to *rem

    add rsp, 16                 # restore stack (8 for push + 8 for sub)
    ret

.Lzero_divisor:
    xor rax, rax                # quotient = 0
    mov qword ptr [rdx], rdi    # *rem = a (rdx has rem pointer from ABI)
    ret
