export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%is_na_zero =w ceql %na, 0
	jnz %is_na_zero, @return_zero
	%is_nb_zero =w ceql %nb, 0
	jnz %is_nb_zero, @return_zero

	%total_limbs =l add %na, %nb
	%i =l const 0

@outer_loop
	%i_check =w sltul %i, %na
	jz %i_check, @cleanup_and_return

	%j =l const 0

@inner_loop
	%j_check =w sltul %j, %nb
	jz %j_check, @next_i

	%a_idx =l add %a, %i
	%b_idx =l add %b, %j
	%r_idx =l add %r, %i
	%j_shl =l add %i, %j

	%a_val =l loadl (%a_idx)
	%b_val =l loadl (%b_idx)

	%ah =l shr %a_val, 32
	%al =l and %a_val, 4294967295
	%bh =l shr %b_val, 32
	%bl =l and %b_val, 4294967295

	%t0 =l mul %al, %bl
	%t1 =l mul %ah, %bl
	%t2 =l mul %al, %bh
	%th =l add %t1, %t2
	%th_lo =l and %th, 4294967295
	%th_hi =l shr %th, 32

	%mid_carry =l shl %th_lo, 32
	%prod_low =l add %t0, %mid_carry
	%carry_out =w cultl %prod_low, %t0
	%high_part =l add %th_hi, %carry_out

	%dest =l add %r, %j_shl
	%orig_val =l loadl (%dest)

	%new_lo =l add %orig_val, %prod_low
	%lo_carry =w cultl %new_lo, %orig_val

	storel %new_lo, (%dest)
	%next_dest =l add %dest, 8
	%orig_high =l loadl (%next_dest)

	%new_high =l add %orig_high, %high_part
	%final_high_carry =w cultl %new_high, %orig_high
	%total_high =l add %new_high, %lo_carry

	storel %total_high, (%next_dest)

	%j_inc =l add %j, 1
	%j =l copy %j_inc
	jnz 1, @inner_loop

@next_i
	%i_inc =l add %i, 1
	%i =l copy %i_inc
	jnz 1, @outer_loop

@cleanup_and_return
	%k =l copy %total_limbs
	%last_nonzero_found =l sub %total_limbs, 1
	%found_nonzero =w const 0

@find_last_nonzero
	%k_check =w sltul %k, 1
	jnz %k_check, @return_result

	%k_dec =l add %k, -1
	%chk_idx =l add %r, %k_dec
	%val =l loadl (%chk_idx)
	%is_zero =w ceql %val, 0
	jnz %is_zero, @next_k

	%last_nonzero_found =l copy %k_dec
	%found_nonzero =w const 1
	jnz 1, @return_result

@next_k
	%k =l copy %k_dec
	jnz 1, @find_last_nonzero

@return_result
	%ret_val =l add %last_nonzero_found, 1
	%is_zero_product =w ceql %found_nonzero, 0
	jnz %is_zero_product, @return_zero_impl
	return %ret_val

@return_zero_impl
@return_zero
	return 0
}
