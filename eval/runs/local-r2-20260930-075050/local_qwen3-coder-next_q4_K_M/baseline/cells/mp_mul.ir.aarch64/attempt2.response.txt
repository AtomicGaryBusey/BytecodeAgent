export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {

	jnz %0, @zero_result

	jnz %1, @zero_result
	%2 =l mul %na, %nb
	%3 =w ceql %2, 0
	jnz %3, @zero_result
	storel 0, (%r)
	%4 =l add %na, %nb

	jnz %5, @finish_zero
@store_loops

@outer_i_loop

@store_inner_j

	%9 =l add %i, 1
	storel 0, (%r, %9)
@inner_k_loop
@store_k_val
	storel 0, (%r, 8)
	csltl 1, 1
@k_done
	csltl 1, 1
	%10 =l add %i, 1
	storel 0, (%r, %10)
	jnz 0, @finish_zero
@zero_result
	return 0
@finish_zero

	%12 =w sltul %11, 1
	jnz %12, @return_empty

	%14 =l loadl (%13)
	%15 =w ceql %14, 0
	jnz %15, @return_count

@return_empty
	return 0

	csltl 1, 1
	return 0
@loop_end
	return 0
@return_count
	return 0
}
