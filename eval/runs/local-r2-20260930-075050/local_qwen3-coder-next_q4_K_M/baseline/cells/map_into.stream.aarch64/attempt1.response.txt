map_into:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Save callee-saved registers we use: x19-x28, but only need a few here
    // We'll use x19 for f pointer (passed in x4), x20 for src base, x21 for dst base,
    // x22 for n (counter), x23 for count of where dst[i] < src[i]
    mov x19, x4          // function pointer
    mov x20, x1          // src pointer
    mov x21, x0          // dst pointer
    cbz x2, .Lreturn_zero   // n == 0? return 0
    mov x22, x2          // copy n to counter register
    mov x23, xzr         // initialize count of dst[i] < src[i]

.Lloop:
    ldr x9, [x20], #8    // load *src, post-increment
    mov x0, x9           // arg for f: src[i]
    blr x19              // call f(src[i]), result in x0
    str x0, [x21], #8    // store dst[i] = result, post-increment

    // compare result with original src value (x9)
    cmp x0, x9           // dst[i] vs src[i]
    csel xzr, xzr, xzr, lt  # no-op but to preserve flags? Actually use cset
    b
