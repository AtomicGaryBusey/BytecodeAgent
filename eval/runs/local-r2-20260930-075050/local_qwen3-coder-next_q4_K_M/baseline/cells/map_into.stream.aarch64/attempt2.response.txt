map_into:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Save callee-saved registers we use: x19-x23 (x24 not needed)
    str x19, [sp, #-8]!
    str x20, [sp, #-8]!
    str x21, [sp, #-8]!
    str x22, [sp, #-8]!
    str x23, [sp, #-8]!

    mov x19, x4          // function pointer f
    mov x20, x1          // src base pointer
    mov x21, x0          // dst base pointer
    cbz x2, .Lreturn_zero   // n == 0? return 0
    mov x22, x2          // counter = n
    mov x23, xzr         // count of dst[i] < src[i]

.Lloop:
    ldr x9, [x20], #8    // load *src, post-increment src pointer
    mov x0, x9           // argument for f: src[i]
    blr x19              // call f(src[i]), result in x0

    str x0, [x21], #8    // store dst[i] = result, post-increment dst pointer

    cmp x0, x9           // compare dst[i] with original src[i]
    cset x10, lt         // x10 = 1 if dst[i] < src[i], else 0
    add x23, x23, x10    // increment count if condition true

    sub x22, x22, #1     // decrement counter
    cbnz x22, .Lloop     // continue loop if n > 0

.Lreturn_zero:
    mov x0, x23          // result = count of elements where dst[i] < src[i]

    // Restore callee-saved registers and return
    ldr x23, [sp], #8
    ldr x22, [sp], #8
    ldr x21, [sp], #8
    ldr x20, [sp], #8
    ldr x19, [sp], #8
    ldp x29, x30, [sp], #16
    ret
