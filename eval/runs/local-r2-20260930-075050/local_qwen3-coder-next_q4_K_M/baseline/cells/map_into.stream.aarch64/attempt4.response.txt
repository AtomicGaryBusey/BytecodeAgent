map_into:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #48
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]
    str x22, [sp, #32]
    str x23, [sp, #40]

    mov x19, x4          // function pointer f
    mov x20, x1          // src base pointer
    mov x21, x0          // dst base pointer
    cbz x2, .Lreturn_zero   // n == 0? return 0
    mov x22, x2           // counter = n
    mov x23, xzr          // count of dst[i] < src[i]

.Lloop:
    ldr x9, [x20], #8     // load *src, post-increment src pointer
    mov x0, x9            // argument for f: src[i]
    blr x19               // call f(src[i]), result in x0

    str x0, [x21], #8     // store dst[i] = f(src[i]), increment dst pointer
    cmp x0, x9            // compare dst[i] with original src[i]
    b.lt .Lless           // branch if dst[i] < src[i]
    b .Lnext

.Lless:
    add x23, x23, #1      // increment count of dst[i] < src[i]

.Lnext:
    sub x22, x22, #1      // decrement counter
    cbnz x22, .Lloop

.Lreturn_zero:
    mov x0, x23           // return count (0 if n == 0)
    ldr x19, [sp, #8]
    ldr x20, [sp, #16]
    ldr x21, [sp, #24]
    ldr x22, [sp, #32]
    ldr x23, [sp, #40]
    add sp, sp, #48
    ldp x29, x30, [sp], #16
    ret
