stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8                  # align stack to 16 bytes for function calls

    mov r10, rdi                # r10 = a (input array)
    mov rsi, rsi                # rsi = n (number of elements)
    mov r12, rdx                # r12 = f (function pointer)
    mov r13, rcx                # r13 = out (output array)

    xor rax, rax                # count_odd = 0
    test rsi, rsi
    jz .Ldone                   # if n == 0, skip processing

    mov rbx, qword ptr [r10]    # load a[0]
    call r12                    # compute y[0] = f(a[0])
    mov r14, rax                # min_val = y[0]
    mov r15, rax                # max_val = y[0]
    mov rcx, rax                # sum = y[0]

    test al, 1
    setnz dl                    # dl = (y[0] & 1) != 0 ? 1 : 0
    add rax, rdx                # accumulate odd count (dl is 0 or 1)
    mov r9d, eax                # save current odd count in r9d

    dec rsi
    jz .Ldone                   # if n == 1, skip loop

.Lloop:
    mov rbx, qword ptr [r10 + 8]   # load a[i]
    call r12                       # y[i] = f(a[i])

    test al, 1
    setnz dl
    add rax, rdx                   # accumulate odd count

    # update min: if y < min then min = y
    cmp r14, rax
    cmovg r14, rax

    # update max: if y > max then max = y
    cmp r15, rax
    cmovl r15, rax

    # update sum
    add rcx, rax                   # sum += y[i]

    dec rsi
    jnz .Lloop

.Ldone:
    mov qword ptr [r13 + 0], r14   # out[0] = min
    mov qword ptr [r13 + 8], r15   # out[1] = max
    mov qword ptr [r13 + 16], rcx  # out[2] = sum

    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
