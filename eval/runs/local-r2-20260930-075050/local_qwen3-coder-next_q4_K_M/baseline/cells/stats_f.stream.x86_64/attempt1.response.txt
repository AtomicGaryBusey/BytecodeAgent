stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8                  # align stack to 16 bytes for function calls

    mov r10, rdi                # r10 = a (input array)
    mov rsi, rsi                # rsi = n (number of elements)
    mov r12, rdx                # r12 = f (function pointer)
    mov r13, rcx                # r13 = out (output array)

    xor rax, rax                # count_odd = 0
    mov rbx, qword ptr [r10]    # load y[0] = f(a[0])
    call r12                    # compute y[0]
    mov r14, rax                # min_val = y[0]
    mov r15, rax                # max_val = y[0]
    mov rcx, rax                # sum = y[0]

    test al, 1
    setnz dl
    shr dl, 7                   # dl = (y[0] & 1) != 0 ? 1 : 0
    add rax, rdx

    dec rsi
    jz .Ldone
    lea rdi, [r10 + 8]
    xor rbx, rbx                # reset loop counter for array indexing

.Lloop:
    mov rax, qword ptr [rdi + rbx*8]   # load a[i]
    call r12                           # y[i] = f(a[i])
    
    # update odd count
    test al, 1
    setnz dh
    shr dh, 7                   # dh = (y[i] & 1) != 0 ? 1 : 0
    add rax, rdx                # accumulate odd count
    
    # update min
    cmp r14, rax
    cmovg r14, rax              # if min > y[i], then min = y[i]
    
    # update max
    cmp r15, rax
    cmovl r15, rax              # if max < y[i], then max = y[i]
    
    # update sum
    add rcx, rax                # sum += y[i]

    inc rbx
    dec rsi
    jnz .Lloop

.Ldone:
    mov qword ptr [r13 + 0], r14   # out[0] = min
    mov qword ptr [r13 + 8], r15   # out[1] = max
    mov qword ptr [r13 + 16], rcx  # out[2] = sum

    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
