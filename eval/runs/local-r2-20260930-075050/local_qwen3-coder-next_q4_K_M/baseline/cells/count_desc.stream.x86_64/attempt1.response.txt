count_desc:
    mov rax, 0                # initialize count to 0
    cmp rsi, 1               # compare n with 1
    jle .Lend              # if n < 2, return 0

    xor rcx, rcx           # i = 0
.Lloop:
    mov rdx, qword ptr [rdi + rcx*8]     # load a[i]
    mov rbx, qword ptr [rdi + (rcx+1)*8] # load a[i+1]
    cmp rdx, rbx           # compare a[i] and a[i+1]
    jle .Lnext             # skip if not descending
    inc rax                # increment count
.Lnext:
    inc rcx                # i++
    cmp rcx, rsi           # compare i with n
    jl .Lloop              # continue while i < n-1 (i.e. i+1 < n)

.Lend:
    ret
