count_desc:
    cmp rsi, 1               # compare n with 1
    jle .Lend              # if n < 2, return 0

    xor rax, rax           # initialize count to 0
    xor rcx, rcx           # i = 0
.Lloop:
    mov rdx, qword ptr [rdi + rcx*8]     # load a[i]
    lea rbx, [rdi + rcx*8 + 8]          # compute address of a[i+1]
    cmp rdx, qword ptr [rbx]            # compare a[i] and a[i+1]
    jle .Lnext             # skip if not descending
    inc rax                # increment count
.Lnext:
    inc rcx                # i++
    cmp rcx, rsi           # compare i with n
    jl .Lloop              # continue while i < n-1 (since last valid i is n-2)

.Lend:
    ret
