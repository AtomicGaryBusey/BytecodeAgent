count_desc:
    mov rax, 0                # initialize count to 0
    cmp rsi, 1               # compare n with 1
    jle .Lend              # if n < 2, return 0

    xor rcx, rcx           # i = 0
.Lloop:
    mov rdx, qword ptr [rdi + rcx*8]     # load a[i]
    lea rbx, [rdi + (rcx+1)*8]           # compute address of a[i+1]
    cmp rbx, rdi                         # check if address >= rdi + n*8
    jae .Lend                            # if out of bounds, exit loop
    mov rbx, qword ptr [rbx]             # load a[i+1]
    cmp rdx, rbx           # compare a[i] and a[i+1]
    jle .Lnext             # skip if not descending
    inc rax                # increment count
.Lnext:
    inc rcx                # i++
    cmp rcx, rsi           # compare i with n-1 (since we need i+1 < n)
    jl .Lloop              # continue while i < n-1

.Lend:
    ret
