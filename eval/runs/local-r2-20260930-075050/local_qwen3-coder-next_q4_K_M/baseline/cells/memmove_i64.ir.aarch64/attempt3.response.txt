export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	w %is_zero = ceql %n, 0
	jnz %is_zero, @done

	%check_overlap = subl %dst, %src
	w %forward = csltl %check_overlap, 0
	jnz %forward, @copy_backward

@copy_forward:
	l %current_src = %src
	l %current_dst = %dst
	w %i = ceql %n, 0
	jnz %i, @done

@loop_forward:
	l %val = loadl l [%current_src]
	storel l [%current_dst], %val
	%current_src = addl %current_src, 8
	%current_dst = addl %current_dst, 8
	%n = subl %n, 1
	w %check = ceql %n, 0
	jnz %check, @done
	j @loop_forward

@copy_backward:
	l %src_end = %src
	l %dst_end = %dst
	l %offset = mull %n, 8
	%src_end = subl %src_end, %offset
	%dst_end = subl %dst_end, %offset

@loop_backward:
	l %val = loadl l [%src_end]
	storel l [%dst_end], %val
	%src_end = addl %src_end, 8
	%dst_end = addl %dst_end, 8
	%n = subl %n, 1
	w %check = ceql %n, 0
	jnz %check, @done
	j @loop_backward

@done:
	l %ret = 0
	ret %ret
}
