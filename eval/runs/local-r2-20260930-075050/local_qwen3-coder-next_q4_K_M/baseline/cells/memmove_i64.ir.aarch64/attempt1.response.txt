export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	w %i = ceql %n, 0
	jnz %i, @done
	@loop_header
	w %i2 = ceql %n, 0
	jnz %i2, @done
	l %val = loadl l [%src]
	storel l %dst, %val
	l %src_next = addl %src, 8
	l %dst_next = addl %dst, 8
	l %n_dec = subl %n, 1
	%src = %src_next
	%dst = %dst_next
	%n = %n_dec
	j @loop_header
@done
	l %ret = 0
	ret %ret
}
