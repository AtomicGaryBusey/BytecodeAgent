export function l $xxh64(l %p, l %n, l %seed) {
@entry
%q =l add %p, 0
%norig =l copy %n
%seed_copy =l copy %seed
%v1 =l copy %seed
%v2 =l copy %seed
%v3 =l copy %seed
%v4 =l copy %seed
%c32 =l copy %n
%c8 =l copy 0
%c4 =l copy 0
%c1 =l copy 0
%h =l copy %seed

%check_head:  
csltl %cmp_head, %norig, 32
jnz %loop_body, %cmp_head
@body_head

%P1 =l copy -7046029288634856825
%P2 =l copy -4542638004300531121
%v1_init =l add %seed, %P1
%v1_init =l add %v1_init, %P2
%v2_init =l add %seed, %P2
%v3_init =l copy %seed
%n_neg =l sub 0, %P1
%v4_init =l add %seed, %n_neg

%v1 =l copy %v1_init
%v2 =l copy %v2_init
%v3 =l copy %v3_init
%v4 =l copy %v4_init

%loop_start:
csltl %cmp_loop, %c32, 32
jnz %after_loop, %cmp_loop

% stripe processing
%qstripe =l add %p, 0
%c32_stripe =l copy %c32
@loop_body_stripe
%inner_check:
csltl %inner_cmp, %c32_stripe, 32
jnz %after_stripe_loop, %inner_cmp

%qstripe_addr =l add %qstripe, 0
%val1 =l loadl %qstripe_addr
%val2 =l loadl 8(%qstripe)
%val3 =l loadl 16(%qstripe)
%val4 =l loadl 24(%qstripe)

%r1 =l mul %val1, %P2
%v1_tmp =l add %v1, %r1
%rot1 =l rotl %v1_tmp, 31
%v1_new =l mul %rot1, %P1

%r2 =l mul %val2, %P2
%v2_tmp =l add %v2, %r2
%rot2 =l rotl %v2_tmp, 31
%v2_new =l mul %rot2, %P1

%r3 =l mul %val3, %P2
%v3_tmp =l add %v3, %r3
%rot3 =l rotl %v3_tmp, 31
%v3_new =l mul %rot3, %P1

%r4 =l mul %val4, %P2
%v4_tmp =l add %v4, %r4
%rot4 =l rotl %v4_tmp, 31
%v4_new =l mul %rot4, %P1

%v1 =l copy %v1_new
%v2 =l copy %v2_new
%v3 =l copy %v3_new
%v4 =l copy %v4_new

%qstripe =l add %qstripe, 32
%c32_stripe =l sub %c32_stripe, 32
jnz @loop_body_stripe, 1
@after_stripe_loop

%h_v1 =l rotl %v1, 1
%h_v2 =l rotl %v2, 7
%h_v3 =l rotl %v3, 12
%h_v4 =l rotl %v4, 18
%h_mid =l add %h_v1, %h_v2
%h_mid =l add %h_mid, %h_v3
%h_mid =l add %h_mid, %h_v4

%P4 =l copy -8765619404506079357
%tmp =l copy 0
%tmp =l mul %tmp, %P2
%tmp =l rotl %tmp, 31
%tmp =l mul %tmp, %P1
%h_mid =l xor %h_mid, %tmp
%h_mid =l mul %h_mid, %P1
%h_mid =l add %h_mid, %P4

%tmp2 =l copy 0
%tmp2 =l mul %tmp2, %P2
%tmp2 =l rotl %tmp2, 31
%tmp2 =l mul %tmp2, %P1
%h_mid =l xor %h_mid, %v1
%h_mid =l mul %h_mid, %P1
%h_mid =l add %h_mid, %P4

%tmp3 =l copy 0
%tmp3 =l mul %tmp3, %P2
%tmp3 =l rotl %tmp3, 31
%tmp3 =l mul %tmp3, %P1
%h_mid =l xor %h_mid, %v2
%h_mid =l mul %h_mid, %P1
%h_mid =l add %h_mid, %P4

%tmp4 =l copy 0
%tmp4 =l mul %tmp4, %P2
%tmp4 =l rotl %tmp4, 31
%tmp4 =l mul %tmp4, %P1
%h_mid =l xor %h_mid, %v3
%h_mid =l mul %h_mid, %P1
%h_mid =l add %h_mid, %P4

%tmp5 =l copy 0
%tmp5 =l mul %tmp5, %P2
%tmp5 =l rotl %tmp5, 31
%tmp5 =l mul %tmp5, %P1
%h =l xor %h_mid, %v4
%h =l mul %h, %P1
%h =l add %h, %P4

jnz @after_head, 1

@loop_body:
%P5 =l copy -8273865409945709371
%h =l add %h, %P5

@after_head:

%rem_bytes =l copy %c32
%q_rem =l add %p, 0

@loop8:
csltl %cmp8, %rem_bytes, 8
jnz @after_loop8, %cmp8

%val8 =l loadl %q_rem
%tmp8_1 =l mul %val8, %P2
%tmp8_2 =l rotl %h, 31
%tmp8_3 =l add %tmp8_2, %tmp8_1
%tmp8_4 =l rotl %tmp8_3, 31
%tmp8_5 =l mul %tmp8_4, %P1
%h =l xor %h, %tmp8_5

%h_rot =l rotl %h, 27
%h =l mul %h_rot, %P1
%h =l add %h, %P4

%q_rem =l add %q_rem, 8
%rem_bytes =l sub %rem_bytes, 8
jnz @loop8, 1
@after_loop8:

@c4_check:
csltl %cmp4, %rem_bytes, 4
jnz @after_loop4, %cmp4

%val4 =l loadw %q_rem
%val4_64 =l copy %val4
%tmp4_1 =l mul %val4_64, %P1
%h =l xor %h, %tmp4_1
%h_rot4 =l rotl %h, 23
%h =l mul %h_rot4, %P2
%h =l add %h, %P3

%q_rem =l add %q_rem, 4
%rem_bytes =l sub %rem_bytes, 4
jnz @after_loop4, 0
@loop4:
; skipped due to above jump logic fix
%val4 =l loadw %q_rem
%val4_64 =l copy %val4
%tmp4_1 =l mul %val4_64, %P1
%h =l xor %h, %tmp4_1
%h_rot4 =l rotl %h, 23
%h =l mul %h_rot4, %P2
%h =l add %h, %P3

%q_rem =l add %q_rem, 4
%rem_bytes =l sub %rem_bytes, 4
jnz @loop4, 1
@after_loop4:

@c1_check:
csltl %cmp1, %rem_bytes, 1
jnz @after_loop1, %cmp1

@loop1:
csltl %byte_cmp, %rem_bytes, 1
jnz @end_loop1, %byte_cmp

%val1 =l loadb %q_rem
%val1_64 =l copy %val1
%tmp1_1 =l mul %val1_64, %P5
%h =l xor %h, %tmp1_1
%h_rot1 =l rotl %h, 11
%h =l mul %h_rot1, %P1

%q_rem =l add %q_rem, 1
%rem_bytes =l sub %rem_bytes, 1
jnz @loop1, 1
@end_loop1:
@after_loop1:

@after_stripe_and_tail:

%h ^= h >> 33
%tmp_h1 =l shr %h, 33
%h =l xor %h, %tmp_h1

%h *= P2
%h =l mul %h, %P2

%tmp_h2 =l shr %h, 29
%h =l xor %h, %tmp_h2

%h *= P3
%h =l mul %h, %P3

%tmp_h3 =l shr %h, 32
%h =l xor %h, %tmp_h3

ret %h
@after_loop:

%P5 =l copy -8273865409945709371
%h =l add %h, %P5
jnz @after_head, 1

@loop_body:
; fallback for when head is <32 bytes (unreachable due to earlier jump)
jnz @error_block, 0
@end_error:

@error_block:
ret 0
}
