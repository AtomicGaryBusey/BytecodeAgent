max_byte:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Handle n == 0 case: return 0 immediately
    cbz x1, .Lreturn_zero
    // Initialize max with first byte and advance pointer/count
    ldrb w3, [x0]
    add x0, x0, #1
    sub x1, x1, #1
    cbz x1, .Ldone
    mov x4, x3        // x4 holds current max
.Lloop:
    ldrb w5, [x0]
    add x0, x0, #1
    cmp x5, x4
    csel x4, x5, x4, hi  // if higher, update max
    sub x1, x1, #1
    cbnz x1, .Lloop
.Ldone:
    mov x0, x4
    ldp x29, x30, [sp], #16
    ret
.Lreturn_zero:
    mov x0, xzr
