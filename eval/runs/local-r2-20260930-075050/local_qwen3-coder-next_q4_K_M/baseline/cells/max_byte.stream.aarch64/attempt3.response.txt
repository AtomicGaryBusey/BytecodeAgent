max_byte:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Handle n == 0 case: return 0 immediately
    cbz x1, .Lreturn_zero
    // Initialize max with first byte and advance pointer/count
    ldrb w2, [x0]
    add x0, x0, #1
    sub x1, x1, #1
    cbz x1, .Ldone
    mov x3, x2        // x3 holds current max
.Lloop:
    ldrb w4, [x0]
    add x0, x0, #1
    cmp w4, w3
    csel x3, x4, x3, hi  // if higher, update max
    sub x1, x1, #1
    cbnz x1, .Lloop
.Ldone:
    mov x0, x3
    ldp x29, x30, [sp], #16
    ret
.Lreturn_zero:
    mov x0, xzr
