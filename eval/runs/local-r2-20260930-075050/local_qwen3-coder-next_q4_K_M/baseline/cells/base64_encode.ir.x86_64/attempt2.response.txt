export function l $base64_encode(l %src, l %n, l %out) {
entry:
	%zero =l const 0
	%one =l const 1
	%three =l const 3
	%six =l const 6
	%eight =l const 8
	%twentyfour =l const 24
	%sixtythree =l const 63
	%baseA =l const 65
	%basea =l const 97
	%basedigit =l const 48
	%sixtwotwenty =l const 61
	%pluschar =l const 43
	%slashchar =l const 47
	%eqchar =l const 61

	%len =l copy %n
	jnz %len, @process_data
	%result =l copy %zero
	ret %result

@process_data:
	%total_out =l const 0
	%src_p =l copy %src
	%out_p =l copy %out
	%remaining =l copy %n

@loop_main:
	csltl %remaining, %three, %has_full_group
	jz %has_full_group, @final_bytes
	%b0 =l loadub %src_p
	%off1 =l add %src_p, %one
	%b1 =l loadub %off1
	%off2 =l add %off1, %one
	%b2 =l loadub %off2
	%off3 =l add %off2, %one
	%src_p =l copy %off3

	%val24 =l shl %b0, %twentyfour
	%val24 =l shr %val24, %sixteen
	%val24 =l or %val24, %b1
	%val24 =l shl %val24, %eight
	%val24 =l shr %val24, %twentyfour
	%val24 =l or %val24, %b2

	%idx0 =l shr %val24, %eighteen
	%val24_1 =l shl %val24, %six
	%idx1 =l shr %val24_1, %twentyfour
	%val24_2 =l shl %val24, %twelve
	%idx2 =l shr %val24_2, %twentyfour
	%val24_3 =l shl %val24, %eighteen
	%idx3 =l shr %val24_3, %twentyfour

	storeb %idx0, %out_p
	%oo1 =l add %out_p, %one
	storeb %idx1, %oo1
	%oo2 =l add %oo1, %one
	storeb %idx2, %oo2
	%oo3 =l add %oo2, %one
	storeb %idx3, %oo3

	%out_p =l copy %oo3
	%increment =l const 4
	%total_out =l add %total_out, %increment
	%remaining =l sub %remaining, %three
	jnz %remaining, @loop_main
	jmp @done

@final_bytes:
	csltl %remaining, %one, %no_more
	jnz %no_more, @exit_final
	csltl %remaining, %two, %exactly_one
	jz %exactly_one, @exactly_two

@exactly_two:
	%b0_2 =l loadub %src_p
	%off1_2 =l add %src_p, %one
	%b1_2 =l loadub %off1_2

	%val16 =l shl %b0_2, %sixteen
	%val16 =l shr %val16, %ten
	%val16 =l or %val16, %b1_2
	%val16 =l shl %val16, %eight
	%val16 =l shr %val16, %sixteen

	%idx0_2 =l shr %val16, %twelve
	%val16_1 =l shl %val16, %six
	%idx1_2 =l shr %val16_1, %sixteen
	%idx2_pad =l const 64

	storeb %idx0_2, %out_p
	%oo1_2 =l add %out_p, %one
	storeb %idx1_2, %oo1_2
	%oo2_2 =l add %oo1_2, %one
	storeb %eqchar, %oo2_2
	%oo3_2 =l add %oo2_2, %one
	storeb %eqchar, %oo3_2

	%out_p =l copy %oo3_2
	%increment_2 =l const 4
	%total_out =l add %total_out, %increment_2
	jmp @exit_final

@exactly_one:
	%b0_1 =l loadub %src_p

	%val8 =l shl %b0_1, %eighteen
	%val8 =l shr %val8, %twelve
	%idx0_1 =l copy %val8

	storeb %idx0_1, %out_p
	%oo1_1 =l add %out_p, %one
	storeb %eqchar, %oo1_1
	%oo2_1 =l add %oo1_1, %one
	storeb %eqchar, %oo2_1

	%out_p =l copy %oo2_1
	%increment_1 =l const 3
	%total_out =l add %total_out, %increment_1
	jmp @exit_final

@done:
	%result =l copy %total_out
	ret %result

@exit_final:
	%result =l copy %total_out
	ret %result
}
