export function l $base64_encode(l %src, l %n, l %out) {
@entry:
	%zero =l const 0
	%one =l const 1
	%three =l const 3
	%six =l const 6
	%eight =l const 8
	%twentyfour =l const 24
	%sixtythree =l const 63
	%baseA =l const 65
	%basea =l const 97
	%basedigit =l const 48
	%sixtwotwenty =l const 61
	%pluschar =l const 43
	%slashchar =l const 47

	%len =l copy %n
	jnz %len, @process_data
	%result =l copy %zero
	ret %result

@process_data:
	%total_out =l const 0
	%src_p =l copy %src
	%out_p =l copy %out
	%remaining =l copy %n

@loop_main:
	csltl %remaining, %three, %has_full_group
	jz %has_full_group, @final_bytes
	%b0 =l loadub %src_p
	%off1 =l add %src_p, %one
	%b1 =l loadub %off1
	%off2 =l add %off1, %one
	%b2 =l loadub %off2
	%off3 =l add %off2, %one
	%src_p =l copy %off3

	%val24 =l const 0
	%val24 =l or %val24, %b0
	%val24 =l shl %val24, %eight
	%val24 =l or %val24, %b1
	%val24 =l shl %val24, %eight
	%val24 =l or %val24, %b2

	%idx0 =l copy %val24
	%idx0 =l shr %idx0, %eighteen
	%val24_1 =l copy %val24
	%val24_1 =l shl %val24_1, %six
	%idx1 =l copy %val24_1
	%idx1 =l shr %idx1, %twentyfour
	%val24_2 =l copy %val24
	%val24_2 =l shl %val24_2, %twelve
	%idx2 =l copy %val24_2
	%idx2 =l shr %idx2, %twentyfour
	%val24_3 =l copy %val24
	%val24_3 =l shl %val24_3, %eighteen
	%idx3 =l copy %val24_3
	%idx3 =l shr %idx3, %twentyfour

	%ch0 =l call @map_idx_to_char(l %idx0)
	storeb %ch0, %out_p
	%oo1 =l add %out_p, %one
	%ch1 =l call @map_idx_to_char(l %idx1)
	storeb %ch1, %oo1
	%oo2 =l add %oo1, %one
	%ch2 =l call @map_idx_to_char(l %idx2)
	storeb %ch2, %oo2
	%oo3 =l add %oo2, %one
	%ch3 =l call @map_idx_to_char(l %idx3)
	storeb %ch3, %oo3

	%out_p =l copy %oo3
	%increment =l const 4
	%total_out =l add %total_out, %increment
	%remaining =l sub %remaining, %three
	jnz %remaining, @loop_main
	jmp @done

@final_bytes:
	csltl %remaining, %one, %no_more
	jnz %no_more, @exit_final
	csltl %remaining, %two, %exactly_one
	jz %exactly_one, @exactly_two

@exactly_two:
	%b0_2 =l loadub %src_p
	%off1_2 =l add %src_p, %one
	%b1_2 =l loadub %off1_2

	%val24 =l const 0
	%val24 =l or %val24, %b0_2
	%val24 =l shl %val24, %eight
	%val24 =l or %val24, %b1_2
	%val24 =l shl %val24, %eight

	%idx0 =l copy %val24
	%idx0 =l shr %idx0, %eighteen
	%val24_1 =l copy %val24
	%val24_1 =l shl %val24_1, %six
	%idx1 =l copy %val24_1
	%idx1 =l shr %idx1, %twentyfour

	%ch0 =l call @map_idx_to_char(l %idx0)
	storeb %ch0, %out_p
	%oo1 =l add %out_p, %one
	%ch1 =l call @map_idx_to_char(l %idx1)
	storeb %ch1, %oo1
	%oo2 =l add %oo1, %one
	storeb %pluschar, %oo2
	%oo3 =l add %oo2, %one
	storeb %pluschar, %oo3

	%out_p =l copy %oo3
	%increment_2 =l const 4
	%total_out =l add %total_out, %increment_2
	jmp @exit_final

@exactly_one:
	%b0_1 =l loadub %src_p

	%val24 =l const 0
	%val24 =l or %val24, %b0_1
	%val24 =l shl %val24, %eight

	%idx0 =l copy %val24
	%idx0 =l shr %idx0, %eighteen

	%ch0 =l call @map_idx_to_char(l %idx0)
	storeb %ch0, %out_p
	%oo1 =l add %out_p, %one
	storeb %pluschar, %oo1
	%oo2 =l add %oo1, %one
	storeb %pluschar, %oo2

	%out_p =l copy %oo2
	%increment_1 =l const 3
	%total_out =l add %total_out, %increment_1
	jmp @exit_final

@done:
	%result =l copy %total_out
	ret %result

@exit_final:
	%result =l copy %total_out
	ret %result
}

export function w @map_idx_to_char(l %idx) {
@map_start:
	%zero =l const 0
	%twosix =l const 26
	%fourtwotwo =l const 52

	csltl %idx, %twosix, %is_upper
	jnz %is_upper, @upper

	csltl %idx, %fourtwotwo, %is_lower
	jnz %is_lower, @lower

	csltl %idx, %sixtythree, %is_digit
	jnz %is_digit, @digit
	%result =l const 47
	ret %result

@upper:
	%result =l add %baseA, %idx
	ret %result

@lower:
	%off =l sub %idx, %twosix
	%result =l add %basea, %off
	ret %result

@digit:
	%off =l sub %idx, %fourtwotwo
	%result =l add %basedigit, %off
	ret %result
}
