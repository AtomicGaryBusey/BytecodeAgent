export function l $bmax(l %a, l %n) {
entry:
	l %idx
	l %current_max
	w %done
	w %cond
	l %val

	%idx = copy 0
	%current_max = loadl %a
loop_start:
	%cond = csltl %idx, %n
	jnz %cond, loop_body
	jmp loop_end
loop_body:
	%val = loadl l add %a, mul %idx, 8
	%done = csgt %val, %current_max
	jnz %done, update_max
	%idx = add %idx, 1
	jmp loop_start
update_max:
	%current_max = copy %val
	%idx = add %idx, 1
	jmp loop_start
loop_end:
	ret %current_max
}
