export function l $bmax(l %a, l %n) {
@entry
	l %idx
	l %current_max
	w %cond

	%idx = copy 0
	%current_max = loadl l add %a, 0
@loop_start
	%cond = csltl %idx, %n
	jnz %cond, @loop_body
	jmp @loop_end
@loop_body
	l %val
	w %max_cond
	%val = loadl l add %a, mul %idx, 8
	%max_cond = csgt %val, %current_max
	jnz %max_cond, @update_max
	%idx = add %idx, 1
	jmp @loop_start
@update_max
	%current_max = copy %val
	%idx = add %idx, 1
	jmp @loop_start
@loop_end
	ret %current_max
}
