binary_search:
    # Entry: rdi = a, rsi = n, rdx = key
    # Return index in rax or -1 if not found
    
    push rbx              # save callee-saved register rbx (high bound)
    
    xor rax, rax          # low = 0 (use rax as low)
    mov rbx, rsi          # high = n (exclusive upper bound), rbx is saved
    
    test rsi, rsi         # if n == 0
    jz .Lnot_found        # then not found
    
.Lloop:
    cmp rax, rbx          # while low < high
    jge .Lnot_found       # exit if low >= high
    
    mov rcx, rax          # copy low to rcx for mid calculation
    add rcx, rbx          # rcx = low + high
    shr rcx, 1            # rcx = (low + high) / 2 using unsigned shift (same as sar for non-negative)
    
    mov r9, qword ptr [rdi + rcx*8]   # load a[mid]
    cmp r9, rdx           # compare with key
    je .Lfound            # if equal, found
    
    jl .Llow              # if a[mid] < key, search upper half
    jmp .Lhigh            # else (a[mid] > key), search lower half
    
.Llow:
    inc rcx               # mid + 1
    mov rax, rcx          # low = mid + 1
    jmp .Lloop
    
.Lhigh:
    mov rbx, rcx          # high = mid
    jmp .Lloop
    
.Lfound:
    mov rax, rcx          # return mid (rcx holds the index)
    pop rbx               # restore callee-saved register
    ret
    
.Lnot_found:
    xor rax, rax          # rax = 0
    dec rax               # rax = -1
    pop rbx               # restore callee-saved register
    ret
