binary_search:
    # Setup: rdi = a, rsi = n, rdx = key
    # Return index or -1 in rax
    xor rax, rax          # low index (initially 0)
    mov rbx, rsi          # high index (n), will be exclusive upper bound
    test rsi, rsi         # check if n == 0
    jz .Lnot_found        # if n == 0, not found
    
.Lloop:
    cmp rax, rbx          # while low < high
    jge .Lnot_found       # if low >= high, not found
    
    mov rcx, rbx          # compute mid = (low + high) / 2
    sub rcx, rax          # rcx = high - low
    sar rcx, 1            # rcx = (high - low) / 2
    add rcx, rax          # rcx = low + (high - low)/2
    mov r9, rcx           # store mid in r9 for later use
    
    mov r10, qword ptr [rdi + rcx*8]   # load a[mid]
    cmp r10, rdx          # compare with key
    je .Lfound            # if equal, found
    
    jl .Llow              # if a[mid] < key, search upper half
    jmp .Lhigh            # else (a[mid] > key), search lower half
    
.Llow:
    mov rax, rcx          # low = mid + 1
    inc rax               # low = mid + 1
    jmp .Lloop
    
.Lhigh:
    mov rbx, rcx          # high = mid
    jmp .Lloop
    
.Lfound:
    mov rax, r9           # return mid (r9 holds mid)
    ret
    
.Lnot_found:
    mov rax, -1           # return -1
    ret
