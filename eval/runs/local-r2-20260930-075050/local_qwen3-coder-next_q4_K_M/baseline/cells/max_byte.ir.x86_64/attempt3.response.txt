export function l $max_byte(l %p, l %n) {
@start
	%is_empty =w ceql %n, 0
	jnz %is_empty, @empty
@init_max
	%byte0 =l loadub %p
	%max =l copy %byte0
	%p_cur =l addl %p, 1
	%remaining =l subl %n, 1
@loop_top
	%done_check =w ceql %remaining, 0
	jnz %done_check, @exit_loop
	%cur_byte =l loadub %p_cur
	%is_greater =w csltl %max, %cur_byte
	jz %is_greater, @no_update
	%max =l copy %cur_byte
@no_update
	%p_cur =l addl %p_cur, 1
	%remaining =l subl %remaining, 1
	jmp @loop_top
@exit_loop
	ret %max
@empty
	%zero =l zext w 0 to l
	ret %zero
}
