export function l $max_byte(l %p, l %n) {
@start
	%0 =w ceql %n, 0
	jnz %0, @empty
@loop_init
	%max =l loadub %p
	%p_inc =l addl %p, 1
	%i =l subl %n, 1
@loop_top
	%cond =w ceql %i, 0
	jnz %cond, @done
	%byte =l loadub %p_inc
	%cond2 =w csltl %max, %byte
	jz %cond2, @advance
	%max =l copy %byte
@advance
	%p_inc =l addl %p_inc, 1
	%i =l subl %i, 1
	jmp @loop_top
@done
	%result =l copy %max
	ret %result
@empty
	%zero =l zext w 0 to l
	ret %zero
}
