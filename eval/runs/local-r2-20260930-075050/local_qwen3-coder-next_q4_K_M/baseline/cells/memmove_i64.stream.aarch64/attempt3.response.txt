memmove_i64:
        cbz     x2, .Ldone

        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        lsl     x4, x2, #3              // n * 8
        add     x5, x1, x4              // src_end = src + n*8
        cmp     x0, x5                  // dst vs src_end
        b.ge    .Lforward_copy          // if dst >= src_end: safe forward copy

        add     x6, x0, x4              // dst_end = dst + n*8
        cmp     x6, x1                  // dst_end vs src
        b.le    .Lbackward_copy         // if dst_end <= src: safe backward copy

        // Full overlap with dst before src and dst after src (i.e., src inside dst region)
        // Must copy forward to avoid overwriting source before reading
        b       .Lforward_copy

.Lbackward_copy:
        lsl     x7, x2, #3              // total bytes = n*8
        add     x0, x0, x7              // dst_end = dst + n*8
        add     x1, x1, x7              // src_end = src + n*8
        neg     x7, x7                  // -n*8

.Lbackward_loop:
        ldr     x3, [x1, x7]
        str     x3, [x0, x7]
        add     x7, x7, #8
        cbnz    x7, .Lbackward_loop
        b       .Lcleanup

.Lforward_copy:
        neg     x7, x2                  // -n
        lsl     x7, x7, #3              // -n*8

.Lforward_loop:
        ldr     x3, [x1, x7]
        str     x3, [x0, x7]
        add     x7, x7, #8
        cbnz    x7, .Lforward_loop

.Lcleanup:
        mov     sp, x29
        ldp     x29, x30, [sp], #16

.Ldone:
        mov     x0, #0
        ret
