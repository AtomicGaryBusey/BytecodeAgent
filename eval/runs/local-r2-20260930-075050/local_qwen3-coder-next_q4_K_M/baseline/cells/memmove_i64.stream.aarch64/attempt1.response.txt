memmove_i64:
        // Check if n is zero
        cbz     x2, .Ldone

        // Save callee-saved registers (x19-x28, x29, x30) and allocate stack frame
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        // Check if dst >= src + n * 8 or dst <= src (no overlap or non-overlapping with dst before src)
        // Then forward copy is safe
        lsl     x4, x2, #3              // n * 8 bytes
        add     x5, x1, x4              // src_end = src + n*8
        cmp     x0, x5
        b.ge    .Lforward

        // Check if dst + n*8 <= src (no overlap or non-overlapping with dst after src)
        add     x6, x0, x4              // dst_end = dst + n*8
        cmp     x6, x1
        b.le    .Lbackward

        // Overlapping case: must use backward copy
        lsl     x2, x2, #3              // n * 8 bytes total
        add     x0, x0, x2              // dst_end = dst + n*8
        add     x1, x1, x2              // src_end = src + n*8
        neg     x2, x2                  // -n*8 for index

.Lbackward_loop:
        ldr     x3, [x1, x2]
        str     x3, [x0, x2]
        add     x2, x2, #8
        cbnz    x2, .Lbackward_loop
        b       .Lcleanup

.Lforward:
        sub     x2, x2, #1              // n-1 for loop condition (n > 0 here)
        lsl     x3, x2, #3              // (n-1)*8
        add     x4, x0, x3              // dst_end = dst + (n-1)*8
        add     x5, x1, x3              // src_end = src + (n-1)*8

.Lforward_loop:
        ldr     x6, [x5]
        str     x6, [x4]
        subs    x2, x2, #1
        bmi     .Lcleanup
        sub     x4, x4, #8
        sub     x5, x5, #8
        b       .Lforward_loop

.Lcleanup:
        // Restore callee-saved registers and return
        mov     sp, x29                 // restore sp
        ldp     x29, x30, [sp], #16

.Ldone:
        // Return 0
        mov     x0, #0
        ret
