        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        lsl     x4, x2, #3              // n * 8
        add     x5, x1, x4              // src_end = src + n*8
        cmp     x0, x5
        b.ge    .Lforward_copy

        cmp     x6, x1
        b.le    .Lbackward_copy

        add     x0, x0, x2              // dst_end = dst + n*8
        add     x1, x1, x2              // src_end = src + n*8

        ldr     x3, [x1, x7]
        str     x3, [x0, x7]
        add     x7, x7, #8
        cbnz    x7, .Lbackward_loop

        sub     x2, x2, #1              // n-1
        lsl     x3, x2, #3              // (n-1)*8
        add     x4, x0, x3              // dst_start + (n-1)*8
        add     x5, x1, x3              // src_start + (n-1)*8

.Lforward_loop:
        ldr     x6, [x5]
        str     x6, [x4]
        subs    x2, x2, #1
        bmi     .Lcleanup
        sub     x4, x4, #8
        sub     x5, x5, #8
        b       .Lforward_loop

        mov     sp, x29
        ldp     x29, x30, [sp], #16

        ret
