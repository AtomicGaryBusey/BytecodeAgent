memmove_i64:
        cbz     x2, .Ldone

        // Save frame pointer and return address
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp

        // Calculate total bytes
        lsl     x7, x2, #3              // n * 8

        // Check if dst < src: copy forward (dst before src)
        cmp     x0, x1
        b.lo    .Lforward_copy

        // If dst > src: check if we need backward copy
        add     x4, x0, x7              // dst_end = dst + n*8
        cmp     x4, x1                  // dst_end vs src
        b.le    .Lbackward_copy         // dst_end <= src: disjoint or dst after src but no overlap

        // Otherwise (dst >= src and dst_end > src): dst starts at or before src but they overlap,
        // so backward copy is safe (dst is after src)
        b       .Lbackward_copy

.Lforward_copy:
        lsl     x4, x2, #3              // n * 8
        neg     x7, x4                  // -n*8
.Lfwd_loop:
        ldr     x5, [x1, x4]
        str     x5, [x0, x4]
        add     x4, x4, #8
        cmp     x4, x7                  // compare with negative (so loop until i == n)
        b.lt    .Lfwd_loop
        b       .Lcleanup

.Lbackward_copy:
        lsl     x4, x2, #3              // n * 8
        add     x0, x0, x4              // dst_end = dst + n*8
        add     x1, x1, x4              // src_end = src + n*8
        neg     x7, x4                  // -n*8
.Lbwd_loop:
        ldr     x5, [x1, x7]
        str     x5, [x0, x7]
        add     x7, x7, #8
        cbnz    x7, .Lbwd_loop

.Lcleanup:
        mov     sp, x29
        ldp     x29, x30, [sp], #16

.Ldone:
        mov     x0, #0
        ret
