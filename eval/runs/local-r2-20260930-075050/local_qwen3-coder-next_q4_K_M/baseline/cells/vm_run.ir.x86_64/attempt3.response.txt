export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc = copy l 0
	%dsp = copy l 0
	jmp @loop

@loop
	%pc_lt_len = csltl l %pc, %len
	jz w %pc_lt_len @end
	%inst = loadl l %code + %pc * 8
	%op = and l %inst, 15
	%pc_next = add l %pc, 1
	jmp @switch

@switch
	switch %op [
		0: @case_halt
		1: @case_push
		2: @case_add
		3: @case_sub
		4: @case_mul
		5: @case_dup
		6: @case_swap
		7: @case_drop
		8: @case_over
		9: @case_lt
		10: @case_jz
		11: @case_jmp
	]: @default_case

@default_case
	%pc = copy l %pc_next
	jnz w 1 @loop

@case_halt
	jmp @end

@case_push
	%v = loadl l %code + %pc_next * 8
	storel l %stack + %dsp * 8, %v
	%dsp_inc = add l %dsp, 1
	%dsp = copy l %dsp_inc
	%pc = add l %pc, 2
	jnz w 1 @loop

@case_add
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%b = loadl l %stack + %dsp_dec1 * 8
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%dsp_dec2 = sub l %dsp, 2
	%a = loadl l %stack + %dsp_dec2 * 8
	%sum = add l %a, %b
	storel l %stack + %dsp_dec2 * 8, %sum
	%dsp = copy l %dsp_dec2
	jnz w 1 @loop

@case_sub
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%b = loadl l %stack + %dsp_dec1 * 8
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%dsp_dec2 = sub l %dsp, 2
	%a = loadl l %stack + %dsp_dec2 * 8
	%diff = sub l %a, %b
	storel l %stack + %dsp_dec2 * 8, %diff
	%dsp = copy l %dsp_dec2
	jnz w 1 @loop

@case_mul
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%b = loadl l %stack + %dsp_dec1 * 8
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%dsp_dec2 = sub l %dsp, 2
	%a = loadl l %stack + %dsp_dec2 * 8
	%prod = mul l %a, %b
	storel l %stack + %dsp_dec2 * 8, %prod
	%dsp = copy l %dsp_dec2
	jnz w 1 @loop

@case_dup
	%cmp = ceql l %dsp, 0
	jnz w %cmp @underflow
	%dsp_dec1 = sub l %dsp, 1
	%top = loadl l %stack + %dsp_dec1 * 8
	storel l %stack + %dsp * 8, %top
	%dsp_inc = add l %dsp, 1
	%dsp = copy l %dsp_inc
	jnz w 1 @loop

@case_swap
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%idx0 = sub l %dsp, 2
	%top_idx = sub l %dsp, 1
	%bottom = loadl l %stack + %idx0 * 8
	%top_val = loadl l %stack + %top_idx * 8
	storel l %stack + %idx0 * 8, %top_val
	storel l %stack + %top_idx * 8, %bottom
	jnz w 1 @loop

@case_drop
	%cmp = ceql l %dsp, 0
	jnz w %cmp @underflow
	%dsp_dec1 = sub l %dsp, 1
	%dsp = copy l %dsp_dec1
	storel l %stack + %dsp * 8, 0
	jnz w 1 @loop

@case_over
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%idx0 = sub l %dsp, 2
	%bottom_val = loadl l %stack + %idx0 * 8
	storel l %stack + %dsp * 8, %bottom_val
	%dsp_inc = add l %dsp, 1
	%dsp = copy l %dsp_inc
	jnz w 1 @loop

@case_lt
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%b = loadl l %stack + %dsp_dec1 * 8
	%cmp2 = ceql l %dsp_dec1, 0
	jnz w %cmp2 @underflow
	%dsp_dec2 = sub l %dsp, 2
	%a = loadl l %stack + %dsp_dec2 * 8
	%cmp_val = csltl l %a, %b
	%one = copy l 1
	%res = select w %cmp_val, %one, 0
	storel l %stack + %dsp_dec2 * 8, %res
	%dsp = copy l %dsp_dec2
	jnz w 1 @loop

@case_jz
	%cmp1 = ceql l %dsp, 0
	jnz w %cmp1 @underflow
	%dsp_dec1 = sub l %dsp, 1
	%v = loadl l %stack + %dsp_dec1 * 8
	%zero = copy l 0
	%is_zero = ceql l %v, %zero
	jnz w %is_zero @jz_branch_taken
	%pc_next2 = add l %pc, 2
	%pc = copy l %pc_next2
	jnz w 1 @loop

@jz_branch_taken
	%target_idx = add l %pc, 1
	%target = loadl l %code + %target_idx * 8
	%pc = copy l %target
	jnz w 1 @loop

@case_jmp
	%target_idx = add l %pc, 1
	%target = loadl l %code + %target_idx * 8
	%pc = copy l %target
	jnz w 1 @loop

@end
	%dsp_zero = ceql l %dsp, 0
	jnz w %dsp_zero @return_zero
	%dsp_dec1 = sub l %dsp, 1
	%result = loadl l %stack + %dsp_dec1 * 8
	jmp @finalize_stack

@underflow
	ret l 0

@return_zero
	%result = copy l 0
	jmp @finalize_stack

@finalize_stack
	%zero = copy l 0
	%i = copy l 0
@clear_loop
	%cmp_i_cap = csltl l %i, %cap
	jz w %cmp_i_cap @clear_done
	storel l %stack + %i * 8, %zero
	%i_inc = add l %i, 1
	%i = copy l %i_inc
	jnz w 1 @clear_loop

@clear_done
	ret l %result
}
