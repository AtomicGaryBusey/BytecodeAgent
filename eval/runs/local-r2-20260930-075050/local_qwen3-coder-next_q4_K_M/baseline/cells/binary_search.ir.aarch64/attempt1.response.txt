export function l $binary_search(l %a, l %n, l %key) {
    %low = w 0
    %high = l %n
    %mid = l 0
    %val = l 0

@loop
    %cmp = csltl %low, %high
    jnz %cmp, @body, @not_found

@body
    %mid_val = addl %low, %high
    %mid_shifted = asrl %mid_val, 1
    storel %mid_shifted, %mid
    
    %idx = loadl %mid
    %ptr = addl %a, %idx
    %val = loadl %ptr
    
    %eq = ceql %val, %key
    jnz %eq, @found

    %less = csltl %val, %key
    jnz %less, @right_half, @left_half

@left_half
    %mid_plus1 = addl %idx, 1
    storel %mid_plus1, %low
    jmp @loop

@right_half
    %mid_minus1 = subl %idx, 1
    storel %mid_minus1, %high
    jmp @loop

@found
    ret %idx

@not_found
    %minus_one = l -1
    ret %minus_one
}
