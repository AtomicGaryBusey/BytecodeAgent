export function l $binary_search(l %a, l %n, l %key) {
    %low = l 0
    %high = subl %n, l 1

@loop
    %cmp = csltl %low, %high
    jnz %cmp, @body, @check_last

@body
    %mid_sum = addl %low, %high
    %mid_val = asrl %mid_sum, l 1
    %idx = %mid_val
    
    %ptr = addl %a, %idx
    %val = loadl %ptr
    
    %eq = ceql %val, %key
    jnz %eq, @found

    %less = csltl %val, %key
    jnz %less, @right_half, @left_half

@left_half
    %mid_plus1 = addl %idx, l 1
    %low = %mid_plus1
    jmp @loop

@right_half
    %mid_minus1 = subl %idx, l 1
    %high = %mid_minus1
    jmp @loop

@check_last
    %eq0 = ceql %low, %high
    jnz %eq0, @last_check, @not_found

@last_check
    %ptr_last = addl %a, %low
    %val_last = loadl %ptr_last
    %eq_last = ceql %val_last, %key
    jnz %eq_last, @found, @not_found

@found
    ret %idx

@not_found
    %minus_one = l -1
    ret %minus_one
}
