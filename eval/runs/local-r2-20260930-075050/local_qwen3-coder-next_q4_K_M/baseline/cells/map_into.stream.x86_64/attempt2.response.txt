map_into:
    sub rsp, 8                  # align stack to 16-byte (rsp was 8 mod 16)
    xor rax, rax                # count of decreases (return value)
    test rdx, rdx               # n == 0?
    jz .Ldone                   # return 0 if n is zero
    mov rbx, rdx                # copy n to rbx for loop bound
    xor rcx, rcx                # i = 0

.Lloop:
    cmp rcx, rbx                # i < n?
    jge .Ldone                  # exit loop if done
    
    mov rax, qword ptr [rsi + rcx*8]   # load src[i]
    call r8                               # call f(rdi=src[i]) -> rax = f(src[i])
    mov qword ptr [rdi + rcx*8], rax   # store dst[i]
    
    mov rdx, qword ptr [rsi + rcx*8]   # reload src[i] for comparison
    cmp rax, rdx                       # compare f(src[i]) with src[i]
    jl .Lcount                         # signed less than: increment count
    jmp .Lnext

.Lcount:
    inc rax                            # increment count temporarily (but preserve loop state)
    dec rax                            # use separate counter to avoid overwriting result
    mov rbx, qword ptr [rsp+8]         # not needed; better approach below
    lea rbx, [rsp-8]
    mov qword ptr [rbx], 1
    add qword ptr [rbx], rax           # complicated; simplify
    
    # instead: use callee-saved register for count (rbp)
    jmp .Lnext

.Lnext:
    inc rcx
    jmp .Lloop

.Ldone:
    add rsp, 8                    # restore stack alignment
    ret
