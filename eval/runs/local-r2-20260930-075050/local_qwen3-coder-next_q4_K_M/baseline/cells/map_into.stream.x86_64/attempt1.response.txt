map_into:
    push rbp
    mov rbp, rsp
    sub rsp, 16          # space for locals: count (rbx) and tmp (r12)
    mov rbx, 0           # count of decreases
    mov r12, rcx         # n in r12 (callee-saved)
    test r12, r12
    jz .Ldone
    xor rcx, rcx         # i = 0

.Lloop:
    cmp rcx, r12
    jge .Ldone
    mov rax, qword ptr [rsi + rcx*8]   # load src[i]
    call r13                           # call f (r13 holds f)
    mov qword ptr [rdi + rcx*8], rax   # store dst[i]
    mov r14, qword ptr [rsi + rcx*8]   # reload src[i]
    cmp rax, r14
    jge .Lnext
    inc rbx                            # count this decrease

.Lnext:
    inc rcx
    jmp .Lloop

.Ldone:
    mov rax, rbx         # return count
    leave
    ret
