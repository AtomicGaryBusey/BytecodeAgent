map_into:
    sub rsp, 8                  # align stack to 16-byte (rsp was 8 mod 8 at entry, now 0 mod 16)
    xor rax, rax                # count of decreases (will be return value)
    test rdx, rdx               # n == 0?
    jz .Ldone                   # return 0 if n is zero
    mov rbx, rdi                # save dst pointer in callee-saved register (rbx)
    mov rsi, rsi                # save src pointer (rsi is caller-saved, but we'll use it directly)
    xor rcx, rcx                # i = 0 (index counter)
    xor r12, r12                # count of decreases (callee-saved register)

.Lloop:
    cmp rcx, rdx                # i < n?
    jge .Ldone                  # exit loop if done

    mov rax, qword ptr [rsi + rcx*8]   # load src[i]
    call r10                           # call f(rdi=rax) -> rax = f(src[i])
    mov qword ptr [rbx + rcx*8], rax   # store dst[i]

    cmp rax, qword ptr [rsi + rcx*8]   # compare f(src[i]) with src[i]
    jl .Lcount                         # signed less than: increment count

.Lnext:
    inc rcx                            # i++
    jmp .Lloop

.Lcount:
    inc r12                            # increment decrease counter
    jmp .Lnext

.Ldone:
    mov rax, r12                     # return count of decreases
    add rsp, 8                       # restore stack alignment
    ret
