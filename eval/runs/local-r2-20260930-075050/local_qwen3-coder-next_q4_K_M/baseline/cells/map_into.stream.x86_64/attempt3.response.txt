map_into:
    sub rsp, 8                  # align stack to 16-byte (rsp was 8 mod 16)
    xor rax, rax                # count of decreases (return value), use as counter later via rbp
    test rdx, rdx               # n == 0?
    jz .Ldone                   # return 0 if n is zero
    mov rbp, rdi                # save dst pointer in callee-saved register
    mov rbx, rsi                # save src pointer in callee-saved register
    mov rcx, rdx                # i = n (countdown loop)
    xor r12, r12                # count of decreases (callee-saved)

.Lloop:
    cmp rcx, 0                  # check remaining count
    jz .Ldone                   # exit loop if done
    
    mov rax, qword ptr [rbx + r13*8]   # load src[i] where r13 = i (will set up below)
    call r8                            # call f(rdi=rax) -> rax = f(src[i])
    mov qword ptr [rbp + r13*8], rax   # store dst[i]
    
    mov rdx, qword ptr [rbx + r13*8]   # reload src[i] for comparison
    cmp rax, rdx                       # compare f(src[i]) with src[i]
    jl .Lcount                         # signed less than: increment count
    
.Lnext:
    inc r13                            # i++
    dec rcx                            # decrement remaining count
    jmp .Lloop

.Lcount:
    inc r12                            # increment decrease counter
    jmp .Lnext

.Ldone:
    mov rax, r12                     # return count of decreases
    add rsp, 8                       # restore stack alignment
    ret
