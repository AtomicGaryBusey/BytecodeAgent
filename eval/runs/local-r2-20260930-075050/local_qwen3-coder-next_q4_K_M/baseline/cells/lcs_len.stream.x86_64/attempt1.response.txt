lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor eax, eax
    xor ebx, ebx
    # Initialize row[0..n] = 0
.Linit_loop:
    cmp rbx, rdx
    jge .Linit_done
    mov qword ptr [rcx + rbx*8], rax
    inc rbx
    jmp .Linit_loop
.Linit_done:
    xor r13, r13
    # Outer loop over i (0..m-1)
.Louter_loop:
    cmp r13, rsi
    jge .Louter_done
    xor rbx, rbx        # diag = 0
    xor r14, r14        # j counter = 0
    movzx rax, byte ptr [rdi + r13]
    mov r12, rax        # x[i] in r12
.Linner_loop:
    cmp r14, rdx
    jge .Louter_continue
    # up = row[j]
    mov r15, qword ptr [rcx + r14*8 + 8]
    # Check if x[i] == y[j-1]
    movzx eax, byte ptr [rdx + r14 - 8] # y[j-1] where j = r14+1 => index = r14
    cmp al, r12b
    jne .Lnot_equal
    lea rbx, [rbx + 1]
    jmp .Lassign
.Lnot_equal:
    # row[j] = max(row[j], row[j-1]) but we need row[j] first (current), then row[j-1]
    mov rax, qword ptr [rcx + r14*8 + 8]
    cmp rax, rbx
    cmovl rax, rbx      # if row[j] < diag+1? no: max of row[j] (current) and row[j-1]=rbx
    mov qword ptr [rcx + r14*8 + 8], rax
.Lassign:
    # diag = up for next iteration
    mov rbx, r15
    inc r14
    jmp .Linner_loop
.Louter_continue:
    xor rbx, rbx        # reset diag for next i
    inc r13
    jmp .Louter_loop
.Louter_done:
    mov rax, qword ptr [rcx + rdx*8]
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
