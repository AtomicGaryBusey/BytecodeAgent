lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    # Initialize row[0..n] = 0
    xor rax, rax
    xor rbx, rbx
.Linit_loop:
    cmp rbx, rdx
    jge .Linit_done
    mov qword ptr [rcx + rbx*8], rax
    inc rbx
    jmp .Linit_loop

.Linit_done:
    # Outer loop: for i in 0..m-1
    xor r13, r13
.Louter_loop:
    cmp r13, rsi
    jge .Louter_done

    xor rbx, rbx        # diag = 0
    xor r14, r14        # j = 0 (will go from 1 to n)

    # Load x[i]
    movzx eax, byte ptr [rdi + r13]
    mov r12d, eax       # x[i] in r12d (zero-extended)

.Linner_loop:
    inc r14             # j = 1..n
    cmp r14, rdx
    jg .Linner_done

    # up = row[j]
    mov rax, qword ptr [rcx + r14*8]

    # Store up for later use in diag update
    push rax

    # Check if x[i] == y[j-1]
    movzx eax, byte ptr [rdx + r14 - 1]  # Load y[j-1] - but rdx is n! Use r9 instead.
    # Fix: we need y pointer in another register. Pass y in r8 (5th arg) so rdx=n, r8=y
    movzx eax, byte ptr [r8 + r14 - 1]
    cmp r12d, eax
    jne .Lno_match

    # Match: row[j] = diag + 1 = rbx + 1
    lea rax, [rbx + 1]
    mov qword ptr [rcx + r14*8], rax
    pop rbx             # restore old up as new diag (since we had stored it)
    jmp .Linner_loop

.Lno_match:
    # No match: row[j] = max(row[j], row[j-1])
    # row[j] is already in the saved value on stack (rax before push was original row[j])
    pop rbp             # rbp holds old row[j] (up)
    mov rbx, qword ptr [rcx + r14*8 - 8]  # row[j-1]
    cmp rbp, rbx
    cmovl rbx, rbp      # rbx = max(row[j], row[j-1])
    mov qword ptr [rcx + r14*8], rbx

    # Now diag becomes original row[j] (which was up)
    pop rax             # This is wrong; need to restructure stack
    push rbx            # Push new row[j]
    mov rbx, rax        # diag = up

.Linner_done:
    inc r13
    jmp .Louter_loop

.Louter_done:
    # Return row[n]
    mov rax, qword ptr [rcx + rdx*8]

    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
