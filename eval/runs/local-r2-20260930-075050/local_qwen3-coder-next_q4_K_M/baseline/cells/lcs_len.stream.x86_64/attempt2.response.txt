lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    # Initialize row[0..n] = 0
    xor rax, rax
    xor rbx, rbx
.Linit_loop:
    cmp rbx, rdx
    jge .Linit_done
    mov qword ptr [rcx + rbx*8], rax
    inc rbx
    jmp .Linit_loop

.Linit_done:
    # Outer loop: for i in 0..m-1
    xor r13, r13
.Louter_loop:
    cmp r13, rsi
    jge .Louter_done

    xor rbx, rbx        # diag = 0
    xor r14, r14        # j = 0 (will go from 1 to n)
    
    # Load x[i]
    movzx eax, byte ptr [rdi + r13]
    movzx r12b, al      # x[i] in r12b

.L
