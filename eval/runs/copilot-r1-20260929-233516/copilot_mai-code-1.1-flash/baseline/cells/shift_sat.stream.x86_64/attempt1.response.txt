shift_sat:
    mov rcx, rsi
    cmp rcx, 64
    jae .Lleft_zero
    mov rax, rdi
    shl rax, cl
    mov r8, rax
    jmp .Lleft_done
.Lleft_zero:
    xor r8, r8
.Lleft_done:
    cmp rcx, 64
    jae .Lright_special
    mov rax, rdi
    sar rax, cl
    jmp .Lsum
.Lright_special:
    mov rax, rdi
    test rax, rax
    js .Lright_minus_one
    xor rax, rax
    jmp .Lsum
.Lright_minus_one:
    mov rax, -1
.Lsum:
    add r8, rax
    mov rax, r8
    ret
