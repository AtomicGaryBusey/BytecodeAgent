  %c1 =l add %k1a, 97
  jmp @map1_done
@map1_d
  %k1d =l sub %k1, 52
  %c1 =l add %k1d, 48
  jmp @map1_done
@map1_plus
  %c1 =l copy 43
  jmp @map1_done
@map1_slash
  %c1 =l copy 47
@map1_done

  %cmp2_a =w csltl %k2, 26
  jnz %cmp2_a, @map2_A, @map2_n1
@map2_n1
  %cmp2_b =w csltl %k2, 52
  jnz %cmp2_b, @map2_a, @map2_n2
@map2_n2

  jnz %cmp2_c, @map2_d, @map2_n3

  jnz %cmp2_d, @map2_plus, @map2_slash

  jnz %cmp3_b, @map3_a, @map3_n2
@map3_n2

  jmp @map1b_done

@map1b_done

  %out1a =l add %outp, 1
  %out1b =l add %outp, 2
  %out1c =l add %outp, 3
  storeb %c1a0, %outp
  storeb %c1a1, %out1a
  storeb %pad1, %out1b
  storeb %pad1, %out1c

  %total1 =l add %written, 4
  ret %total1

@tail2
  %b2a0p =l add %src, %i
  %b2a0 =l loadub %b2a0p
  %b2a1p =l add %b2a0p, 1
  %b2a1 =l loadub %b2a1p

  %v2a =l shl %b2a0, 16
  %v2b =l shl %b2a1, 8
  %val2a =l or %v2a, %v2b

  %q2a0 =l shr %val2a, 18
  %q2a1 =l shr %val2a, 12
  %q2a2 =l shr %val2a, 6
  %k2a0 =l and %q2a0, 63
  %k2a1 =l and %q2a1, 63
  %k2a2 =l and %q2a2, 63

  %cmp2a_a =w csltl %k2a0, 26
  jnz %cmp2a_a, @map2a_A, @map2a_n1
@map2a_n1
  %cmp2a_b =w csltl %k2a0, 52
  jnz %cmp2a_b, @map2a_a, @map2a_n2
@map2a_n2
  %cmp2a_c =w csltl %k2a0, 62
  jnz %cmp2a_c, @map2a_d, @map2a_n3

  %cmp2a_d =w ceql %k2a0, 62

  %c2a2 =l add %k2a2a, 97
  jmp @map2c_done
@map2c_d
  %k2a2d =l sub %k2a2, 52
  %c2a2 =l add %k2a2d, 48
  jmp @map2c_done
@map2c_plus
  %c2a2 =l copy 43
  jmp @map2c_done
@map2c_slash
  %c2a2 =l copy 47
@map2c_done

  %pad2 =l copy 61
  %out2a =l add %outp, 1
  %out2b =l add %outp, 2
  %out2c =l add %outp, 3
  storeb %c2a0, %outp
  storeb %c2a1, %out2a

  ret %total2

@return_written
