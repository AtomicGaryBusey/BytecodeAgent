export function l $map_sum(l %a, l %n, l %f) {
@entry
  %sum0 =l add $0, $0
  %ptr0 =l add %a, $0
  %count0 =l add %n, $0
  jmp @loop

@loop
  %sum =l phi @entry %sum0, @body %sum1
  %ptr =l phi @entry %ptr0, @body %ptr1
  %count =l phi @entry %count0, @body %count1
  %zero =w ceql %count, $0
  jnz %zero, @done, @body

@body
  %x =l loadl %ptr
  %v =l call %f(l %x)
  %sum1 =l add %sum, %v
  %ptr1 =l add %ptr, $8
  %count1 =l sub %count, $1
  jmp @loop

@done
  ret %sum
}
