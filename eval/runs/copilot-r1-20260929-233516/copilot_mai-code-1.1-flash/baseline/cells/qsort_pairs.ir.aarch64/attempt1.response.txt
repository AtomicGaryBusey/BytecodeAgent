export function l $qsort_pairs(l %a, l %n) {
@entry
  %nlt2 =w csltl %n, 2
  jnz %nlt2, @done

  %pivot_key =l loadl %a
  %pivot_val_ptr =l addl %a, 8
  %pivot_val =l loadl %pivot_val_ptr

  %lt =l 0
  %i =l 0
  %gt =l subl %n, 1
  jmp @partition_loop

@partition_loop
  %gt_lt_i =w csltl %gt, %i
  jnz %gt_lt_i, @after_partition

  %i_off1 =l addl %i, %i
  %i_off2 =l addl %i_off1, %i_off1
  %i_off3 =l addl %i_off2, %i_off2
  %i_off =l addl %i_off3, %i_off3
  %i_ptr =l addl %a, %i_off
  %cur_key =l loadl %i_ptr
  %cur_val_ptr =l addl %i_ptr, 8
  %cur_val =l loadl %cur_val_ptr

  %ltk =w csltl %cur_key, %pivot_key
  jnz %ltk, @lt_case

  %eqk =w ceql %cur_key, %pivot_key
  jnz %eqk, @check_equal_key

  jmp @gt_case

@check_equal_key
  %ltv =w csltl %cur_val, %pivot_val
  jnz %ltv, @lt_case

  %gtv =w csltl %pivot_val, %cur_val
  jnz %gtv, @gt_case

  %i =l addl %i, 1
  jmp @partition_loop

@lt_case
  %lt_off1 =l addl %lt, %lt
  %lt_off2 =l addl %lt_off1, %lt_off1
  %lt_off3 =l addl %lt_off2, %lt_off2
  %lt_off =l addl %lt_off3, %lt_off3
  %lt_ptr =l addl %a, %lt_off
  %same_lt_i =w ceql %lt, %i
  jnz %same_lt_i, @after_lt_swap

  %lt_key =l loadl %lt_ptr
  %lt_val_ptr =l addl %lt_ptr, 8
  %lt_val =l loadl %lt_val_ptr
  %cur_key2 =l loadl %i_ptr
  %cur_val_ptr2 =l addl %i_ptr, 8
  %cur_val2 =l loadl %cur_val_ptr2

  storel %cur_key2, %lt_ptr
  storel %cur_val2, %lt_val_ptr
  storel %lt_key, %i_ptr
  storel %lt_val, %cur_val_ptr2

@after_lt_swap
  %lt =l addl %lt, 1
  %i =l addl %i, 1
  jmp @partition_loop

@gt_case
  %gt_off1 =l addl %gt, %gt
  %gt_off2 =l addl %gt_off1, %gt_off1
  %gt_off3 =l addl %gt_off2, %gt_off2
  %gt_off =l addl %gt_off3, %gt_off3
  %gt_ptr =l addl %a, %gt_off
  %same_gt_i =w ceql %gt, %i
  jnz %same_gt_i, @after_gt_swap

  %gt_key =l loadl %gt_ptr
  %gt_val_ptr =l addl %gt_ptr, 8
  %gt_val =l loadl %gt_val_ptr
  %cur_key3 =l loadl %i_ptr
  %cur_val_ptr3 =l addl %i_ptr, 8
  %cur_val3 =l loadl %cur_val_ptr3

  storel %cur_key3, %gt_ptr
  storel %cur_val3, %gt_val_ptr
  storel %gt_key, %i_ptr
  storel %gt_val, %cur_val_ptr3

@after_gt_swap
  %gt =l subl %gt, 1
  jmp @partition_loop

@after_partition
  %left_len =l %lt
  %right_start =l addl %gt, 1
  %right_off1 =l addl %right_start, %right_start
  %right_off2 =l addl %right_off1, %right_off1
  %right_off3 =l addl %right_off2, %right_off2
  %right_off =l addl %right_off3, %right_off3
  %right_ptr =l addl %a, %right_off
  %right_len =l subl %n, %right_start

  %left_rc =l call $qsort_pairs(l %a, l %left_len)
  %right_rc =l call $qsort_pairs(l %right_ptr, l %right_len)
  ret l 0

@done
  ret l 0
}
