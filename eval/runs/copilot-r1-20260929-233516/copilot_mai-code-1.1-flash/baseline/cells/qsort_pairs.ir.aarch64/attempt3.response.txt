export function l $qsort_pairs(l %a, l %n) {
@entry
  %n_lt_2 =w csltl %n, 2
  jnz %n_lt_2, @done

  %last_index =l subl %n, 1
  %last_off =l mull %last_index, 16
  %last_ptr =l addl %a, %last_off

  %pivot_key =l loadl %last_ptr
  %pivot_val_ptr =l addl %last_ptr, 8
  %pivot_val =l loadl %pivot_val_ptr

  %i =l 0
  %j =l 0
  jmp @loop

@loop
  %j_lt_last =w csltl %j, %last_index
  jnz %j_lt_last, @cmp
  jmp @pivot_move

@cmp
  %j_off =l mull %j, 16
  %j_ptr =l addl %a, %j_off
  %j_key =l loadl %j_ptr
  %j_val_ptr =l addl %j_ptr, 8
  %j_val =l loadl %j_val_ptr

  %j_key_lt =w csltl %j_key, %pivot_key
  jnz %j_key_lt, @swap_i

  %j_key_gt =w csltl %pivot_key, %j_key
  jnz %j_key_gt, @j_inc

  %j_val_lt =w csltl %j_val, %pivot_val
  jnz %j_val_lt, @swap_i

  jmp @j_inc

@swap_i
  %i_off =l mull %i, 16
  %i_ptr =l addl %a, %i_off
  %i_key =l loadl %i_ptr
  %i_val_ptr =l addl %i_ptr, 8
  %i_val =l loadl %i_val_ptr

  %j_key_swap =l loadl %j_ptr
  %j_val_ptr_swap =l addl %j_ptr, 8
  %j_val_swap =l loadl %j_val_ptr_swap

  storel %j_key_swap, %i_ptr
  storel %j_val_swap, %i_val_ptr
  storel %i_key, %j_ptr
  storel %i_val, %j_val_ptr_swap

  %i =l addl %i, 1
  jmp @j_inc

@j_inc
  %j =l addl %j, 1
  jmp @loop

@pivot_move
  %i_after_loop_off =l mull %i, 16
  %i_after_loop_ptr =l addl %a, %i_after_loop_off

  %i_after_loop_key =l loadl %i_after_loop_ptr
  %i_after_loop_val_ptr =l addl %i_after_loop_ptr, 8
  %i_after_loop_val =l loadl %i_after_loop_val_ptr

  %last_key =l loadl %last_ptr
  %last_val_ptr2 =l addl %last_ptr, 8
  %last_val =l loadl %last_val_ptr2

  storel %last_key, %i_after_loop_ptr
  storel %last_val, %i_after_loop_val_ptr
  storel %i_after_loop_key, %last_ptr
  storel %i_after_loop_val, %last_val_ptr2

  %left_len =l %i
  %left_gt_1 =w csltl 1, %left_len
  jnz %left_gt_1, @left_recurse
  jmp @check_right

@left_recurse
  %left_rc =l call $qsort_pairs(l %a, l %left_len)
  jmp @check_right

@check_right
  %right_start =l addl %i, 1
  %right_len =l subl %n, %right_start
  %right_gt_1 =w csltl 1, %right_len
  jnz %right_gt_1, @right_recurse
  jmp @done

@right_recurse
  %right_off =l mull %right_start, 16
  %right_ptr =l addl %a, %right_off
  %right_rc =l call $qsort_pairs(l %right_ptr, l %right_len)
  jmp @done

@done
  ret l 0
}
