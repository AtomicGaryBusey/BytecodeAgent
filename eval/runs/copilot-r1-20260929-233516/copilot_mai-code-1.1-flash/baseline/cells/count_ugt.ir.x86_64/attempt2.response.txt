export function l $count_ugt(l %a, l %n, l %t) {
@start
  %count = copy 0
  %idx = copy 0
  %ptr = copy %a
@loop
  %cond = csltl %idx, %n
  jnz %cond, @body, @done
@body
  %val = loadl %ptr
  %gt = cultl %t, %val
  jnz %gt, @countup, @step
@countup
  %count = addl %count, 1
@step
  %idx = addl %idx, 1
  %ptr = addl %ptr, 8
  jmp @loop
@done
  ret %count
}
