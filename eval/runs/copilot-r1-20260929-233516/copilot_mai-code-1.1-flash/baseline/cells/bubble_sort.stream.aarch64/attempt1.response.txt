bubble_sort:
    cmp x1, #0
    b.le .Ldone
    sub x2, x1, #1
    cmp x2, #0
    b.le .Ldone
.LOuter:
    mov x3, #0
.LInner:
    cmp x3, x2
    b.hs .LafterInner
    ldr x4, [x0, x3, lsl #3]
    add x5, x3, #1
    ldr x6, [x0, x5, lsl #3]
    cmp x4, x6
    b.le .LNoSwap
    str x6, [x0, x3, lsl #3]
    str x4, [x0, x5, lsl #3]
.LNoSwap:
    add x3, x3, #1
    b .LInner
.LafterInner:
    sub x2, x2, #1
    cmp x2, #0
    b.gt .LOuter
.Ldone:
    mov x0, #0
    ret
