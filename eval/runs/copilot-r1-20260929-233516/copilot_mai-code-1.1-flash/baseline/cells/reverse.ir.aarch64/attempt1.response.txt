export function l $reverse(l %a, l %n) {
@entry
    %zero = constl 0
    %one = constl 1
    %eight = constl 8
    %i = constl 0
    %j = subl %n, %one
    %cond = csltl %i, %j
    jnz %cond, @loop(%i, %j)
    ret %zero

@loop(%i, %j)
    %i8 = mull %i, %eight
    %j8 = mull %j, %eight
    %pi = addl %a, %i8
    %pj = addl %a, %j8
    %vi = loadl %pi
    %vj = loadl %pj
    storel %vj, %pi
    storel %vi, %pj
    %i1 = addl %i, %one
    %j1 = subl %j, %one
    %cond1 = csltl %i1, %j1
    jnz %cond1, @loop(%i1, %j1)
    ret %zero
}
