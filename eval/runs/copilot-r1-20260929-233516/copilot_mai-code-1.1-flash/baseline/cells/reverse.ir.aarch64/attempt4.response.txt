export function l $reverse(l %a, l %n) {
@entry
    %j = sub %n, 1
    %cond = csltl 0, %j
    jnz %cond, @loop(0, %j)
    ret 0

@loop(%i, %j)
    %i8 = mul %i, 8
    %j8 = mul %j, 8
    %pi = add %a, %i8
    %pj = add %a, %j8
    %vi = loadl %pi
    %vj = loadl %pj
    storel %vj, %pi
    storel %vi, %pj

    %j1 = sub %j, 1
    %cond1 = csltl %i1, %j1
    jnz %cond1, @loop(%i1, %j1)

}
