export function l $reverse(l %a, l %n) {
@entry
    %i = subl %n, %n
    %j = subl %n, 1
    %cond = csltl %i, %j
    jnz %cond, @loop(%i, %j)
    ret 0

@loop(%i, %j)
    %i8 = mull %i, 8
    %j8 = mull %j, 8
    %pi = addl %a, %i8
    %pj = addl %a, %j8
    %vi = loadl %pi
    %vj = loadl %pj
    storel %vj, %pi
    storel %vi, %pj
    %i1 = addl %i, 1
    %j1 = subl %j, 1
    %cond1 = csltl %i1, %j1
    jnz %cond1, @loop(%i1, %j1)
    ret 0
}
