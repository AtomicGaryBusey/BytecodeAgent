export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %i0 =l const 0
    %count0 =l const 0
    jmp @loop
@loop
    %i =l phi @entry %i0, @continue %i1
    %count =l phi @entry %count0, @continue %count1
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %srcaddr =l add %src, %off
    %srcv =l loadl %srcaddr
    %res =l call %f(l %srcv)
    %dstaddr =l add %dst, %off
    storel %res, %dstaddr
    %lt =w csltl %res, %srcv
    jnz %lt, @inc, @continue
@inc
    %count_inc =l add %count, 1
    jmp @continue
@continue
    %count1 =l phi @body %count, @inc %count_inc
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %count
}
