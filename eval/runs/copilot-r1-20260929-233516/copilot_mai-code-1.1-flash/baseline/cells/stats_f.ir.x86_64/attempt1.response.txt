export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %min0 =l addl %y0, 0
    %max0 =l addl %y0, 0
    %sum0 =l addl %y0, 0
    %one =l addl 0, 1
    %mask0 =l andl %y0, %one
    %isodd0 =w ceql %mask0, %one
    jnz %isodd0, @firstodd
    %odd_init =l addl 0, 0
    jmp @after_first
@firstodd
    %odd_first =l addl 0, 1
    jmp @after_first
@after_first
    %odd_start =l phi %odd_init, @entry, %odd_first, @firstodd
    %p0 =l addl %a, 8
    %i0 =l addl 0, 1
    jmp @loop
@loop
    %min =l phi %min0, @after_first, %min_next, @after_odd
    %max =l phi %max0, @after_first, %max_next, @after_odd
    %sum =l phi %sum0, @after_first, %sum_next, @after_odd
    %odd =l phi %odd_start, @after_first, %odd_next, @after_odd
    %p =l phi %p0, @after_first, %p_next, @after_odd
    %i =l phi %i0, @after_first, %i_next, @after_odd
    %cond =w csltl %i, %n
    jnz %cond, @body
    jmp @store
@body
    %x1 =l loadl %p
    %y1 =l call %f(l %x1)
    %sum_body =l addl %sum, %y1
    %lt1 =w csltl %y1, %min
    jnz %lt1, @min_update
    jmp @after_lt
@min_update
    %min_lt =l addl %y1, 0
    jmp @after_min
@after_lt
    %min_no =l addl %min, 0
    jmp @after_min
@after_min
    %min_after0 =l phi %min_no, @after_lt, %min_lt, @min_update
    %gt1 =w csltl %max, %y1
    jnz %gt1, @max_update
    jmp @after_gt
@max_update
    %max_gt =l addl %y1, 0
    jmp @after_max
@after_gt
    %max_no =l addl %max, 0
    jmp @after_max
@after_max
    %max_after0 =l phi %max_no, @after_gt, %max_gt, @max_update
    %mask1 =l andl %y1, %one
    %isodd1 =w ceql %mask1, %one
    jnz %isodd1, @odd_inc
    jmp @odd_no
@odd_inc
    %odd_add =l addl %odd, 1
    jmp @after_odd
@odd_no
    jmp @after_odd
@after_odd
    %odd_after0 =l phi %odd, @odd_no, %odd_add, @odd_inc
    %min_next =l addl %min_after0, 0
    %max_next =l addl %max_after0, 0
    %sum_next =l addl %sum_body, 0
    %odd_next =l addl %odd_after0, 0
    %p_next =l addl %p, 8
    %i_next =l addl %i, 1
    jmp @loop
@store
    %out1 =l addl %out, 8
    %out2 =l addl %out, 16
    storel %min, %out
    storel %max, %out1
    storel %sum, %out2
    ret %odd
}
