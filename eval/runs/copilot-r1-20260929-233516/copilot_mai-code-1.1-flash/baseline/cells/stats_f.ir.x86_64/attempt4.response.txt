export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %min0 =l copy %y0
    %max0 =l copy %y0
    %sum0 =l copy %y0
    %p_init =l add %a, 8
    %i_init =l copy 1
    %mask0 =l and %y0, 1
    %isodd0 =w ceql %mask0, 1
    jnz %isodd0, @odd_init
    %odd_init_val =l copy 0
    jmp @loop_setup
@odd_init
    %odd_init_val =l copy 1
@loop_setup
    %min =l phi l %min0, @entry, %min0, @odd_init, %min_next, @after_odd
    %max =l phi l %max0, @entry, %max0, @odd_init, %max_next, @after_odd
    %sum =l phi l %sum0, @entry, %sum0, @odd_init, %sum_next, @after_odd
    %odd =l phi l 0, @entry, %odd_init_val, @odd_init, %odd_next, @after_odd
    %p =l phi l %p_init, @entry, %p_init, @odd_init, %p_next, @after_odd
    %i =l phi l %i_init, @entry, %i_init, @odd_init, %i_next, @after_odd
    %cond =w csltl %i, %n
    jnz %cond, @body
    jmp @store
@body
    %x =l loadl %p
    %y =l call %f(l %x)
    %sum_body =l add %sum, %y
    %lt =w csltl %y, %min
    jnz %lt, @min_update
    %min_after =l copy %min
    jmp @after_min
@min_update
    %min_after =l copy %y
@after_min
    %gt =w csltl %max, %y
    jnz %gt, @max_update
    %max_after =l copy %max
    jmp @after_max
@max_update
    %max_after =l copy %y
@after_max
    %mask =l and %y, 1
    %isodd =w ceql %mask, 1
    jnz %isodd, @odd_add
    %odd_after =l copy %odd
    jmp @after_odd
@odd_add
    %odd_after =l add %odd, 1
@after_odd
    %min_next =l copy %min_after
    %max_next =l copy %max_after
    %sum_next =l copy %sum_body
    %odd_next =l copy %odd_after
    %p_next =l add %p, 8
    %i_next =l add %i, 1
    jmp @loop_setup
@store
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
