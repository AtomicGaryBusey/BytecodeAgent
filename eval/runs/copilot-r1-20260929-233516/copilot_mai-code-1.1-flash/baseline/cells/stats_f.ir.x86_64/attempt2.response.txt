export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %min0 =l copy %y0
    %max0 =l copy %y0
    %sum0 =l copy %y0
    %odd0 =l copy 0
    %mask0 =l and %y0, 1
    %isodd0 =w ceql %mask0, 1
    jnz %isodd0, @odd_init_yes
    jmp @after_init_odd
@odd_init_yes
    %odd_first =l copy 1
    jmp @after_init_odd
@after_init_odd
    %odd_init =l phi %odd0, @entry, %odd_first, @odd_init_yes
    %p0 =l add %a, 8
    %i0 =l copy 1
    jmp @loop
@loop
    %min =l phi %min0, @after_init_odd, %min_next, @after_odd
    %max =l phi %max0, @after_init_odd, %max_next, @after_odd
    %sum =l phi %sum0, @after_init_odd, %sum_next, @after_odd
    %odd =l phi %odd_init, @after_init_odd, %odd_next, @after_odd
    %p =l phi %p0, @after_init_odd, %p_next, @after_odd
    %i =l phi %i0, @after_init_odd, %i_next, @after_odd
    %cond =w csltl %i, %n
    jnz %cond, @body
    jmp @store
@body
    %x =l loadl %p
    %y =l call %f(l %x)
    %sum_body =l add %sum, %y
    %lt =w csltl %y, %min
    jnz %lt, @min_update
    jmp @after_min
@min_update
    %min_lt =l copy %y
    jmp @after_min2
@after_min
    %min_no =l copy %min
    jmp @after_min2
@after_min2
    %min_after =l phi %min_no, @after_min, %min_lt, @min_update
    %gt =w csltl %max, %y
    jnz %gt, @max_update
    jmp @after_max
@max_update
    %max_gt =l copy %y
    jmp @after_max2
@after_max
    %max_no =l copy %max
    jmp @after_max2
@after_max2
    %max_after =l phi %max_no, @after_max, %max_gt, @max_update
    %mask =l and %y, 1
    %isodd =w ceql %mask, 1
    jnz %isodd, @oddbody_yes
    jmp @oddbody_no
@oddbody_yes
    %odd_add =l add %odd, 1
    jmp @after_odd
@oddbody_no
    %odd_same =l copy %odd
    jmp @after_odd
@after_odd
    %odd_after =l phi %odd_same, @oddbody_no, %odd_add, @oddbody_yes
    %min_next =l copy %min_after
    %max_next =l copy %max_after
    %sum_next =l copy %sum_body
    %odd_next =l copy %odd_after
    %p_next =l add %p, 8
    %i_next =l add %i, 1
    jmp @loop
@store
    %out1 =l add %out, 8
    %out2 =l add %out, 16
    storel %min, %out
    storel %max, %out1
    storel %sum, %out2
    ret %odd
}
