export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %min0 =l copy %y0
    %max0 =l copy %y0
    %sum0 =l copy %y0
    %odd0 =l copy 0
    %mask0 =l and %y0, 1
    %isodd0 =w ceql %mask0, 1
    jnz %isodd0, @odd_init_yes
    jmp @loop_setup
@odd_init_yes
    %odd1 =l copy 1
    jmp @loop_setup
@loop_setup
    %odd_init =l phi %odd0, @entry, %odd1, @odd_init_yes
    %min_setup =l copy %min0
    %max_setup =l copy %max0
    %sum_setup =l copy %sum0
    %p0 =l add %a, 8
    %i0 =l copy 1
    jmp @loop
@loop
    %min =l phi %min_setup, @loop_setup, %min_next, @after_odd
    %max =l phi %max_setup, @loop_setup, %max_next, @after_odd
    %sum =l phi %sum_setup, @loop_setup, %sum_next, @after_odd
    %odd =l phi %odd_init, @loop_setup, %odd_next, @after_odd
    %p =l phi %p0, @loop_setup, %p_next, @after_odd
    %i =l phi %i0, @loop_setup, %i_next, @after_odd
    %cond =w csltl %i, %n
    jnz %cond, @body
    jmp @store
@body
    %x =l loadl %p
    %y =l call %f(l %x)
    %sum_body =l add %sum, %y
    %lt =w csltl %y, %min
    jnz %lt, @min_update
    %min_after =l copy %min
    jmp @after_min
@min_update
    %min_after =l copy %y
@after_min
    %gt =w csltl %max, %y
    jnz %gt, @max_update
    %max_after =l copy %max
    jmp @after_max
@max_update
    %max_after =l copy %y
@after_max
    %mask =l and %y, 1
    %isodd =w ceql %mask, 1
    jnz %isodd, @odd_add
    %odd_after =l copy %odd
    jmp @after_odd
@odd_add
    %odd_after =l add %odd, 1
@after_odd
    %min_next =l copy %min_after
    %max_next =l copy %max_after
    %sum_next =l copy %sum_body
    %odd_next =l copy %odd_after
    %p_next =l add %p, 8
    %i_next =l add %i, 1
    jmp @loop
@store
    %out1 =l add %out, 8
    %out2 =l add %out, 16
    storel %min, %out
    storel %max, %out1
    storel %sum, %out2
    ret %odd
}
