fib_rec:
    sub rsp, 16
    cmp rdi, 1
    jle .Lbase
    mov qword ptr [rsp], rdi
    lea rdi, [rdi - 1]
    call fib_rec
    mov qword ptr [rsp + 8], rax
    mov rdi, qword ptr [rsp]
    lea rdi, [rdi - 2]
    call fib_rec
    add rax, qword ptr [rsp + 8]
    add rsp, 16
    ret
.Lbase:
    mov rax, rdi
    add rsp, 16
    ret
