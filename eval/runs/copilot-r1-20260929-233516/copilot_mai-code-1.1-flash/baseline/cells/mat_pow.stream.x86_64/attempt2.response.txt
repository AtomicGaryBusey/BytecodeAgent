mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    mov r13, rsi
    imul r13, rsi
    mov r14, rcx
    mov r15, r8
    mov rbx, rdx

    xor r9, r9
.Lzero_out:
    cmp r9, r13
    jae .Ldiag_init
    mov qword ptr [r14 + r9*8], 0
    inc r9
    jmp .Lzero_out
.Ldiag_init:
    xor r9, r9
.Ldiag_loop:
    cmp r9, r12
    jae .Lcopy_base
    mov rax, r9
    imul rax, r12
    add rax, r9
    mov qword ptr [r14 + rax*8], 1
    inc r9
    jmp .Ldiag_loop
.Lcopy_base:
    xor r9, r9
.Lcopy_base_loop:
    cmp r9, r13
    jae .Lpow_loop
    mov rax, qword ptr [rdi + r9*8]
    mov qword ptr [r15 + r9*8], rax
    inc r9
    jmp .Lcopy_base_loop

.Lpow_loop:
    test rbx, rbx
    jz .Lzero_tmp
    test rbx, 1
    jz .Lsquare_base

    mov rdi, r14
    mov rsi, r15
    lea rdx, [r15 + r13*8]
    push rdx
    mov rcx, r12
    xor r8, r8
.Lmul_out_i:
    cmp r8, rcx
    jae .Lmul_out_done
    xor r9, r9
.Lmul_out_j:
    cmp r9, rcx
    jae .Lmul_out_i_inc
    xor r10, r10
    mov rax, 0
.Lmul_out_t:
    cmp r10, rcx
    jae .Lmul_out_store
    mov r11, r8
    imul r11, rcx
    add r11, r10
    mov r11, qword ptr [rdi + r11*8]
    mov rdx, r10
    imul rdx, rcx
    add rdx, r9
    imul r11, qword ptr [rsi + rdx*8]
    add rax, r11
    inc r10
    jmp .Lmul_out_t
.Lmul_out_store:
    mov r11, r8
    imul r11, rcx
    add r11, r9
    mov rdx, qword ptr [rsp]
    mov qword ptr [rdx + r11*8], rax
    inc r9
    jmp .Lmul_out_j
.Lmul_out_i_inc:
    inc r8
    jmp .Lmul_out_i
.Lmul_out_done:
    pop rdx
    lea rdx, [r15 + r13*8]
    xor r8, r8
.Lcopy_out:
    cmp r8, r13
    jae .Lafter_out_copy
    mov rax, qword ptr [rdx + r8*8]
    mov qword ptr [r14 + r8*8], rax
    inc r8
    jmp .Lcopy_out
.Lafter_out_copy:
    shr rbx, 1
    jmp .Lpow_loop

.Lsquare_base:
    mov rdi, r15
    mov rsi, r15
    lea rdx, [r15 + r13*8]
    push rdx
    mov rcx, r12
    xor r8, r8
.Lmul_sq_i:
    cmp r8, rcx
    jae .Lmul_sq_done
    xor r9, r9
.Lmul_sq_j:
    cmp r9, rcx
    jae .Lmul_sq_i_inc
    xor r10, r10
    mov rax, 0
.Lmul_sq_t:
    cmp r10, rcx
    jae .Lmul_sq_store
    mov r11, r8
    imul r11, rcx
    add r11, r10
    mov r11, qword ptr [rdi + r11*8]
    mov rdx, r10
    imul rdx, rcx
    add rdx, r9
    imul r11, qword ptr [rsi + rdx*8]
    add rax, r11
    inc r10
    jmp .Lmul_sq_t
.Lmul_sq_store:
    mov r11, r8
    imul r11, rcx
    add r11, r9
    mov rdx, qword ptr [rsp]
    mov qword ptr [rdx + r11*8], rax
    inc r9
    jmp .Lmul_sq_j
.Lmul_sq_i_inc:
    inc r8
    jmp .Lmul_sq_i
.Lmul_sq_done:
    pop rdx
    lea rdx, [r15 + r13*8]
    xor r8, r8
.Lcopy_sq:
    cmp r8, r13
    jae .Lafter_sq_copy
    mov rax, qword ptr [rdx + r8*8]
    mov qword ptr [r15 + r8*8], rax
    inc r8
    jmp .Lcopy_sq
.Lafter_sq_copy:
    shr rbx, 1
    jmp .Lpow_loop

.Lzero_tmp:
    xor r9, r9
    lea r10, [r13 + r13]
.Lzero_tmp_loop:
    cmp r9, r10
    jae .Lreturn
    mov qword ptr [r15 + r9*8], 0
    inc r9
    jmp .Lzero_tmp_loop
.Lreturn:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
