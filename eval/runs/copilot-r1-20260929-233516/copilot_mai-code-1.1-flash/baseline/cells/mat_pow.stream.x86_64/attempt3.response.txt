mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    mov r13, rsi
    imul r13, rsi
    mov r14, rcx
    mov r15, r8
    mov rbx, rdx

    xor r8, r8
.Lzero_out:
    cmp r8, r13
    jae .Lident
    mov qword ptr [r14 + r8*8], 0
    inc r8
    jmp .Lzero_out
.Lident:
    xor r8, r8
.Lident_loop:
    cmp r8, r12
    jae .Lcheck_k
    mov rax, r8
    imul rax, r12
    add rax, r8
    mov qword ptr [r14 + rax*8], 1
    inc r8
    jmp .Lident_loop
.Lcheck_k:
    test rbx, rbx
    jz .Lzero_tmp

    xor r8, r8
.Lcopy_a:
    cmp r8, r13
    jae .Lpow_loop
    mov rax, qword ptr [rdi + r8*8]
    mov qword ptr [r15 + r8*8], rax
    inc r8
    jmp .Lcopy_a

.Lpow_loop:
    test rbx, rbx
    jz .Lzero_tmp
    test rbx, 1
    jz .Lsquare_base

    mov rdi, r14
    mov rsi, r15
    mov rbp, r15
    lea rbp, [rbp + r13*8]
    xor r8, r8
.Lmul_out_i:
    cmp r8, r12
    jae .Lmul_out_done
    xor r9, r9
.Lmul_out_j:
    cmp r9, r12
    jae .Lmul_out_i_inc
    xor r10, r10
    xor rax, rax
.Lmul_out_t:
    cmp r10, r12
    jae .Lmul_out_store
    mov rcx, r8
    imul rcx, r12
    add rcx, r10
    mov rcx, qword ptr [rdi + rcx*8]
    mov rdx, r10
    imul rdx, r12
    add rdx, r9
    imul rcx, qword ptr [rsi + rdx*8]
    add rax, rcx
    inc r10
    jmp .Lmul_out_t
.Lmul_out_store:
    mov rcx, r8
    imul rcx, r12
    add rcx, r9
    mov qword ptr [rbp + rcx*8], rax
    inc r9
    jmp .Lmul_out_j
.Lmul_out_i_inc:
    inc r8
    jmp .Lmul_out_i
.Lmul_out_done:
    xor r8, r8
.Lcopy_result:
    cmp r8, r13
    jae .Lsquare_base
    mov rax, qword ptr [rbp + r8*8]
    mov qword ptr [r14 + r8*8], rax
    inc r8
    jmp .Lcopy_result

.Lsquare_base:
    mov rdi, r15
    mov rsi, r15
    mov rbp, r15
    lea rbp, [rbp + r13*8]
    xor r8, r8
.Lsq_i:
    cmp r8, r12
    jae .Lsq_done
    xor r9, r9
.Lsq_j:
    cmp r9, r12
    jae .Lsq_i_inc
    xor r10, r10
    xor rax, rax
.Lsq_t:
    cmp r10, r12
    jae .Lsq_store
    mov rcx, r8
    imul rcx, r12
    add rcx, r10
    mov rcx, qword ptr [rdi + rcx*8]
    mov rdx, r10
    imul rdx, r12
    add rdx, r9
    imul rcx, qword ptr [rsi + rdx*8]
    add rax, rcx
    inc r10
    jmp .Lsq_t
.Lsq_store:
    mov rcx, r8
    imul rcx, r12
    add rcx, r9
    mov qword ptr [rbp + rcx*8], rax
    inc r9
    jmp .Lsq_j
.Lsq_i_inc:
    inc r8
    jmp .Lsq_i
.Lsq_done:
    xor r8, r8
.Lcopy_base_back:
    cmp r8, r13
    jae .Lafter_square
    mov rax, qword ptr [rbp + r8*8]
    mov qword ptr [r15 + r8*8], rax
    inc r8
    jmp .Lcopy_base_back
.Lafter_square:
    shr rbx, 1
    jmp .Lpow_loop

.Lzero_tmp:
    xor r8, r8
    mov r9, r13
    shl r9, 1
.Lzero_tmp_loop:
    cmp r8, r9
    jae .Lreturn
    mov qword ptr [r15 + r8*8], 0
    inc r8
    jmp .Lzero_tmp_loop
.Lreturn:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
