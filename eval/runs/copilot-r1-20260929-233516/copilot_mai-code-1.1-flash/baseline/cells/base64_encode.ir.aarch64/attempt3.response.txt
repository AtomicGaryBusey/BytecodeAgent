    %q0 =l add %out, 0

    jmp @loop

@loop
    %p =l phi @entry %p0, @full_done %p_after
    %q =l phi @entry %q0, @full_done %q_after
    %chars =l phi @entry %chars0, @full_done %chars_after
    %nleft =l phi @entry %nleft0, @full_done %nleft_after

    jnz %need_full, @tail

    %b0 =l loadub %p
    %p1 =l add %p, 1
    %b1 =l loadub %p1
    %p2 =l add %p, 2
    %b2 =l loadub %p2
    %v0 =l shl %b0, 16
    %v1 =l shl %b1, 8
    %v2 =l or %v0, %v1
    %v3 =l or %v2, %b2

    %idx0 =l and %idx0_raw, 63
    %idx1_raw =l shr %v3, 12
    %idx1 =l and %idx1_raw, 63
    %idx2_raw =l shr %v3, 6

    %idx3 =l and %v3, 63

    %lt52_0 =w csltl %idx0, 52
    jnz %lt52_0, @full_idx0_a

    jnz %eq62_0, @full_idx0_plus

    jmp @full_idx0_done

    %char0_A =l add %idx0, 65
    storeb %char0_A, %q
    jmp @full_idx0_done
@full_idx0_a

    storeb %char0_a, %q
    jmp @full_idx0_done

    %char0_num =l sub %idx0, 4
    storeb %char0_num, %q
    jmp @full_idx0_done
@full_idx0_plus
    storeb 43, %q
@full_idx0_done
    %q1 =l add %q, 1

    %lt26_1 =w csltl %idx1, 26
    jnz %lt26_1, @full_idx1_A
    %lt52_1 =w csltl %idx1, 52
    jnz %lt52_1, @full_idx1_a
    %lt62_1 =w csltl %idx1, 62
    jnz %lt62_1, @full_idx1_num
    %eq62_1 =w ceql %idx1, 62
    jnz %eq62_1, @full_idx1_plus
    storeb 47, %q1
    jmp @full_idx1_done
@full_idx1_A
    %char1_A =l add %idx1, 65
    storeb %char1_A, %q1
    jmp @full_idx1_done
@full_idx1_a
    %char1_a =l add %idx1, 71
    storeb %char1_a, %q1
    jmp @full_idx1_done
@full_idx1_num
    %char1_num =l sub %idx1, 4
    storeb %char1_num, %q1
    jmp @full_idx1_done
@full_idx1_plus
    storeb 43, %q1
@full_idx1_done
    %q2 =l add %q1, 1

    %lt26_2 =w csltl %idx2, 26
    jnz %lt26_2, @full_idx2_A
    %lt52_2 =w csltl %idx2, 52
    jnz %lt52_2, @full_idx2_a
    %lt62_2 =w csltl %idx2, 62
    jnz %lt62_2, @full_idx2_num
    %eq62_2 =w ceql %idx2, 62
    jnz %eq62_2, @full_idx2_plus
    storeb 47, %q2
    jmp @full_idx2_done
@full_idx2_A
    %char2_A =l add %idx2, 65
    storeb %char2_A, %q2
    jmp @full_idx2_done
@full_idx2_a
    %char2_a =l add %idx2, 71
    storeb %char2_a, %q2
    jmp @full_idx2_done
@full_idx2_num
    %char2_num =l sub %idx2, 4
    storeb %char2_num, %q2
    jmp @full_idx2_done
@full_idx2_plus
    storeb 43, %q2
@full_idx2_done
    %q3 =l add %q2, 1

    %lt26_3 =w csltl %idx3, 26
    jnz %lt26_3, @full_idx3_A
    %lt52_3 =w csltl %idx3, 52
    jnz %lt52_3, @full_idx3_a
    %lt62_3 =w csltl %idx3, 62
    jnz %lt62_3, @full_idx3_num
    %eq62_3 =w ceql %idx3, 62
    jnz %eq62_3, @full_idx3_plus
    storeb 47, %q3
    jmp @full_idx3_done
@full_idx3_A
    %char3_A =l add %idx3, 65
    storeb %char3_A, %q3
    jmp @full_idx3_done
@full_idx3_a
    %char3_a =l add %idx3, 71
    storeb %char3_a, %q3
    jmp @full_idx3_done
@full_idx3_num
    %char3_num =l sub %idx3, 4
    storeb %char3_num, %q3
    jmp @full_idx3_done
@full_idx3_plus
    storeb 43, %q3
@full_idx3_done
    %q4 =l add %q3, 1
    jmp @full_done

@full_done
    %p_after =l add %p, 3
    %q_after =l add %q4, 0
    %chars_after =l add %chars, 4
    %nleft_after =l sub %nleft, 3
    jmp @loop

@tail
    %nleft_is_zero =w ceql %nleft, 0
    jnz %nleft_is_zero, @tail_zero
    %nleft_is_one =w ceql %nleft, 1
    jnz %nleft_is_one, @one
    %nleft_is_two =w ceql %nleft, 2
    jnz %nleft_is_two, @two
    jmp @done

@tail_zero
    jmp @done

@one
    %b0_one =l loadub %p
    %v0_one =l shl %b0_one, 16
    %idx0_one_raw =l shr %v0_one, 18
    %idx0_one =l and %idx0_one_raw, 63
    %idx1_one_raw =l shr %v0_one, 12
    %idx1_one =l and %idx1_one_raw, 63

    %lt26_0_one =w csltl %idx0_one, 26
    jnz %lt26_0_one, @one_idx0_A
    %lt52_0_one =w csltl %idx0_one, 52
    jnz %lt52_0_one, @one_idx0_a
    %lt62_0_one =w csltl %idx0_one, 62
    jnz %lt62_0_one, @one_idx0_num

    jmp @one_idx0_done

    %char0_one_A =l add %idx0_one, 65

    jmp @one_idx0_done
@one_idx0_a
    %char0_one_a =l add %idx0_one, 71
    storeb %char0_one_a, %q
    jmp @one_idx0_done
@one_idx0_num
    %char0_one_num =l sub %idx0_one, 4
    storeb %char0_one_num, %q
    jmp @one_idx0_done
@one_idx0_plus
    storeb 43, %q
@one_idx0_done
    %q_one_1 =l add %q, 1

    %lt26_1_one =w csltl %idx1_one, 26
    jnz %lt26_1_one, @one_idx1_A
    %lt52_1_one =w csltl %idx1_one, 52
    jnz %lt52_1_one, @one_idx1_a
    %lt62_1_one =w csltl %idx1_one, 62
    jnz %lt62_1_one, @one_idx1_num

    storeb %char1_one_num, %q_one_1
    jmp @one_idx1_done
@one_idx1_plus
    storeb 43, %q_one_1
@one_idx1_done
    %q_one_2 =l add %q_one_1, 1
    storeb 61, %q_one_2
    %q_one_3 =l add %q_one_2, 1
    storeb 61, %q_one_3
    %chars_one_done =l add %chars, 4
    jmp @done

@two
    %b0_two =l loadub %p
    %p_two_1 =l add %p, 1
    %b1_two =l loadub %p_two_1
    %v0_two =l shl %b0_two, 16
    %v1_two =l shl %b1_two, 8
    %v2_two =l or %v0_two, %v1_two

    %idx0_two_raw =l shr %v2_two, 18
    %idx0_two =l and %idx0_two_raw, 63
    %idx1_two_raw =l shr %v2_two, 12
    %idx1_two =l and %idx1_two_raw, 63
    %idx2_two_raw =l shr %v2_two, 6
    %idx2_two =l and %idx2_two_raw, 63

    %lt26_0_two =w csltl %idx0_two, 26
    jnz %lt26_0_two, @two_idx0_A
    %lt52_0_two =w csltl %idx0_two, 52
    jnz %lt52_0_two, @two_idx0_a
    %lt62_0_two =w csltl %idx0_two, 62
    jnz %lt62_0_two, @two_idx0_num
    %eq62_0_two =w ceql %idx0_two, 62
    jnz %eq62_0_two, @two_idx0_plus
    storeb 47, %q
    jmp @two_idx0_done
@two_idx0_A
    %char0_two_A =l add %idx0_two, 65
    storeb %char0_two_A, %q
    jmp @two_idx0_done
@two_idx0_a
    %char0_two_a =l add %idx0_two, 71
    storeb %char0_two_a, %q
    jmp @two_idx0_done
@two_idx0_num
    %char0_two_num =l sub %idx0_two, 4
    storeb %char0_two_num, %q
    jmp @two_idx0_done
@two_idx0_plus
    storeb 43, %q
@two_idx0_done
    %q_two_1 =l add %q, 1

    %lt26_1_two =w csltl %idx1_two, 26
    jnz %lt26_1_two, @two_idx1_A
    %lt52_1_two =w csltl %idx1_two, 52
    jnz %lt52_1_two, @two_idx1_a
    %lt62_1_two =w csltl %idx1_two, 62
    jnz %lt62_1_two, @two_idx1_num
    %eq62_1_two =w ceql %idx1_two, 62
    jnz %eq62_1_two, @two_idx1_plus
    storeb 47, %q_two_1
    jmp @two_idx1_done
@two_idx1_A
    %char1_two_A =l add %idx1_two, 65
    storeb %char1_two_A, %q_two_1
    jmp @two_idx1_done
@two_idx1_a
    %char1_two_a =l add %idx1_two, 71
    storeb %char1_two_a, %q_two_1
    jmp @two_idx1_done
@two_idx1_num
    %char1_two_num =l sub %idx1_two, 4
    storeb %char1_two_num, %q_two_1
    jmp @two_idx1_done
@two_idx1_plus
    storeb 43, %q_two_1
@two_idx1_done
    %q_two_2 =l add %q_two_1, 1

    %lt26_2_two =w csltl %idx2_two, 26
    jnz %lt26_2_two, @two_idx2_A
    %lt52_2_two =w csltl %idx2_two, 52
    jnz %lt52_2_two, @two_idx2_a
    %lt62_2_two =w csltl %idx2_two, 62
    jnz %lt62_2_two, @two_idx2_num
    %eq62_2_two =w ceql %idx2_two, 62
    jnz %eq62_2_two, @two_idx2_plus
    storeb 47, %q_two_2
    jmp @two_idx2_done
@two_idx2_A
    %char2_two_A =l add %idx2_two, 65
    storeb %char2_two_A, %q_two_2
    jmp @two_idx2_done
@two_idx2_a
    %char2_two_a =l add %idx2_two, 71
    storeb %char2_two_a, %q_two_2
    jmp @two_idx2_done
@two_idx2_num
    %char2_two_num =l sub %idx2_two, 4
    storeb %char2_two_num, %q_two_2
    jmp @two_idx2_done
@two_idx2_plus
    storeb 43, %q_two_2
@two_idx2_done
    %q_two_3 =l add %q_two_2, 1
    storeb 61, %q_two_3
    %chars_two_done =l add %chars, 4
    jmp @done

@done
    %chars_ret =l phi @tail_zero %chars, @one %chars_one_done, @two %chars_two_done, @tail %chars
    ret %chars_ret
}
