lcs_len:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rcx
    mov r13, r8
    mov rcx, rbx
    inc rcx
    xor eax, eax
.zero_loop:
    mov qword ptr [r13], rax
    add r13, 8
    dec rcx
    jnz .zero_loop

    test rsi, rsi
    jz .done

.outer:
    movzx r15d, byte ptr [rdi]
    inc rdi
    mov r9, 0
    mov r10, rdx
    lea r11, [r8 + 8]
    mov r12, rbx
    test r12, r12
    jz .after_inner
.inner:
    mov rax, qword ptr [r11]
    mov r14, qword ptr [r11 - 8]
    movzx ecx, byte ptr [r10]
    cmp r15b, cl
    jne .notmatch
    lea r13, [r9 + 1]
    mov qword ptr [r11], r13
    mov r9, rax
    jmp .advance
.notmatch:
    mov r13, rax
    cmp rax, r14
    cmovl rax, r14
    mov qword ptr [r11], rax
    mov r9, r13
.advance:
    add r11, 8
    inc r10
    dec r12
    jnz .inner
.after_inner:
    dec rsi
    jnz .outer

.done:
    mov rax, qword ptr [r8 + rbx*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
