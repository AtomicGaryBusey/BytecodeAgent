lcs_len:
    push rbx
    push r14
    push r15
    mov rbx, rcx
    mov r15, r8
    mov rcx, rbx
    inc rcx
    xor eax, eax
.zero_loop:
    mov qword ptr [r15], rax
    add r15, 8
    dec rcx
    jnz .zero_loop
    xor r9d, r9d
    test rsi, rsi
    jz .done
.outer:
    movzx eax, byte ptr [rdi]
    inc rdi
    test rbx, rbx
    jz .next_outer
    lea r10, [r8 + 8]
    lea r11, [rdx]
    mov r15, rbx
.inner:
    mov r12, qword ptr [r10]
    mov r13, qword ptr [r10 - 8]
    movzx ecx, byte ptr [r11]
    cmp al, cl
    jne .notmatch
    lea r13, [r9 + 1]
    mov qword ptr [r10], r13
    mov r9, r12
    jmp .advance
.notmatch:
    mov r14, r12
    cmp r12, r13
    cmovl r12, r13
    mov qword ptr [r10], r12
    mov r9, r14
.advance:
    add r10, 8
    add r11, 1
    dec r15
    jnz .inner
.next_outer:
    dec rsi
    jnz .outer
.done:
    mov rax, qword ptr [r8 + rbx*8]
    pop r15
    pop r14
    pop rbx
    ret
