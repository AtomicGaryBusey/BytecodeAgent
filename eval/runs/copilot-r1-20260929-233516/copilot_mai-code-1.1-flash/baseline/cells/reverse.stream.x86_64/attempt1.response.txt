reverse:
    cmp rsi, 1
    jle .Ldone
    lea rdx, [rsi - 1]
    xor rcx, rcx
.Lloop:
    cmp rcx, rdx
    jge .Ldone
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rdx*8]
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rdx*8], r8
    inc rcx
    dec rdx
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
