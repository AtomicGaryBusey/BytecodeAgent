export function l $fib_rec(l %n) {
@entry
    %is_zero =w ceql %n, 0
    jnz %is_zero, @base0, @check_one
@check_one
    %is_one =w ceql %n, 1
    jnz %is_one, @base1, @recur
@base0
    ret 0
@base1
    ret 1
@recur
    %n_minus_1 =l sub %n, 1
    %n_minus_2 =l sub %n, 2
    %fib_n_minus_1 =l call $fib_rec(l %n_minus_1)
    %fib_n_minus_2 =l call $fib_rec(l %n_minus_2)
    %sum =l add %fib_n_minus_1, %fib_n_minus_2
    ret %sum
}
