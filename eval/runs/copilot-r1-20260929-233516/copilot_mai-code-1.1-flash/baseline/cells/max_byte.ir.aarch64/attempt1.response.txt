export function l $max_byte(p: l, n: l) {
@entry
  %zero =l const 0
  %n_is_zero =w ceql %n, %zero
  jnz %n_is_zero, @ret_zero, @loop_init

@ret_zero
  ret %zero

@loop_init
  %i0 =l const 0
  %max0 =l const 0
  br @loop

@loop
  %i =l phi @loop_init -> %i0, @advance -> %i1
  %max =l phi @loop_init -> %max0, @advance -> %max1
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %addr =l add %p, %i
  %b =l loadub %addr
  %is_new_max =w csltl %max, %b
  jnz %is_new_max, @update, @skip

@update
  %max_update =l add %b, %zero
  br @advance

@skip
  %max_skip =l add %max, %zero
  br @advance

@advance
  %max1 =l phi @update -> %max_update, @skip -> %max_skip
  %i1 =l add %i, 1
  br @loop

@done
  ret %max
}
