export function l $bubble_sort(l %a, l %n) {
@entry
  %n_lt_1 =w csltl %n, 1
  jnz %n_lt_1, @done
  %n_eq_1 =w ceql %n, 1
  jnz %n_eq_1, @done
  %i_init =l copy 0
  jmp @outer_loop

@outer_loop
  %i =l phi @entry %i_init, @outer_inc %i_next
  %i_lt_n =w csltl %i, %n
  jnz %i_lt_n, @outer_body
  jmp @done

@outer_body
  %limit =l sub %n, 1
  %limit_i =l sub %limit, %i
  %j_init =l copy 0
  jmp @inner_loop

@inner_loop
  %j =l phi @outer_body %j_init, @inner_inc %j_next
  %j_lt_limit =w csltl %j, %limit_i
  jnz %j_lt_limit, @inner_body
  jmp @outer_inc

@inner_body
  %j_off =l mul %j, 8
  %j_ptr =l add %a, %j_off
  %a_j =l loadl %j_ptr
  %j1_off =l add %j_off, 8
  %j1_ptr =l add %a, %j1_off
  %a_j1 =l loadl %j1_ptr
  %swap =w csltl %a_j1, %a_j
  jnz %swap, @swap_block
  jmp @inner_inc

@swap_block
  storel %a_j1, %j_ptr
  storel %a_j, %j1_ptr
  jmp @inner_inc

@inner_inc
  %j_next =l add %j, 1
  jmp @inner_loop

@outer_inc
  %i_next =l add %i, 1
  jmp @outer_loop

@done
  ret 0
}
