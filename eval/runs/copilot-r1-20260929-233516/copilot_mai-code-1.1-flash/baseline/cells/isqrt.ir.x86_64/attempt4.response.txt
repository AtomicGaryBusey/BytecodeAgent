export function l $isqrt(l %n) {
@entry
    %lo = copy 0
    %hi = copy 4294967296
    jmp @loop

@loop
    %lo_cur = phi l [%lo, @entry], [%lo_take, @take_lo], [%lo_take_hi, @take_hi]
    %hi_cur = phi l [%hi, @entry], [%hi_take, @take_lo], [%hi_take_hi, @take_hi]
    %delta = sub %hi_cur, %lo_cur
    %done = culel %delta, 1
    jnz %done, @done, @step

@step
    %sum = add %lo_cur, %hi_cur
    %mid = udiv %sum, 2
    %sq = mul %mid, %mid
    %good = culel %sq, %n
    jnz %good, @take_lo, @take_hi

@take_lo
    %lo_take = copy %mid
    %hi_take = copy %hi_cur
    jmp @loop

@take_hi
    %lo_take_hi = copy %lo_cur
    %hi_take_hi = copy %mid
    jmp @loop

@done
    ret %lo_cur
}
