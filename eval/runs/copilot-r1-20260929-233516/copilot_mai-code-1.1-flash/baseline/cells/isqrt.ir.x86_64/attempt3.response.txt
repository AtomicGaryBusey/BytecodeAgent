export function l $isqrt(l %n) {
@entry
    %lo_init = copy 0
    %hi_init = copy 4294967296
    jmp @loop

@loop
    %lo = phi l [%lo_init, @entry], [%lo_take, @take_lo], [%lo_take_hi, @take_hi]
    %hi = phi l [%hi_init, @entry], [%hi_take, @take_lo], [%hi_take_hi, @take_hi]
    %delta = sub %hi, %lo
    %done = cslel %delta, 1
    jnz %done, @done, @step

@step
    %sum = add %lo, %hi
    %mid = udiv %sum, 2
    %prod = mul %mid, %mid
    %le = culel %prod, %n
    jnz %le, @take_lo, @take_hi

@take_lo
    %lo_take = copy %mid
    %hi_take = copy %hi
    jmp @loop

@take_hi
    %lo_take_hi = copy %lo
    %hi_take_hi = copy %mid
    jmp @loop

@done
    ret %lo
}
