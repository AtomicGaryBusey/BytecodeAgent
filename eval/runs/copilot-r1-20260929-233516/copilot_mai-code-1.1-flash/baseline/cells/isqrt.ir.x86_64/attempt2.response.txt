export function l $isqrt(l %n) {
@entry
    %lo_init = addl 0, 0
    %hi_init = addl 0, 4294967296
    jmp @loop

@loop
    %lo = phi l [%lo_init, @entry], [%lo_take, @take_lo], [%lo_take_hi, @take_hi]
    %hi = phi l [%hi_init, @entry], [%hi_take, @take_lo], [%hi_take_hi, @take_hi]
    %delta = subl %hi, %lo
    %done = cslel %delta, 1
    jnz %done, @done, @step

@step
    %sum = addl %lo, %hi
    %mid = udivl %sum, 2
    %prod = mull %mid, %mid
    %le = culel %prod, %n
    jnz %le, @take_lo, @take_hi

@take_lo
    %lo_take = addl %mid, 0
    %hi_take = addl %hi, 0
    jmp @loop

@take_hi
    %lo_take_hi = addl %lo, 0
    %hi_take_hi = addl %mid, 0
    jmp @loop

@done
    ret %lo
}
