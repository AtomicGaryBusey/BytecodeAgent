export function l $isqrt(l %n) {
@entry
    %p = alloc8 16
    %lo_ptr = addl %p, 0
    %hi_ptr = addl %p, 8
    storel 0, %lo_ptr
    storel 4294967296, %hi_ptr
    jmp @loop
@loop
    %lo = loadl %lo_ptr
    %hi = loadl %hi_ptr
    %delta = subl %hi, %lo
    %done = cslel %delta, 1
    jnz %done, @done, @step
@step
    %sum = addl %lo, %hi
    %mid = udivl %sum, 2
    %prod = mull %mid, %mid
    %le = culel %prod, %n
    jnz %le, @take_lo, @take_hi
@take_lo
    storel %mid, %lo_ptr
    jmp @loop
@take_hi
    storel %mid, %hi_ptr
    jmp @loop
@done
    %ret = loadl %lo_ptr
    ret %ret
}
