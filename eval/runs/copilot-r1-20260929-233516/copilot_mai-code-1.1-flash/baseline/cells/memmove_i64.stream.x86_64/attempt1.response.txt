memmove_i64:
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
.Lbackward:
    mov rcx, rdx
    dec rcx
.Lbackward_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    cmp rcx, 0
    je .Ldone
    dec rcx
    jmp .Lbackward_loop
.Lforward:
    xor rcx, rcx
.Lforward_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jb .Lforward_loop
.Ldone:
    xor eax, eax
    ret
