count_desc:
    xor rax, rax
    cmp rsi, 2
    jl .Ldone
    mov rcx, 0
    lea r10, [rsi - 1]
.Lloop:
    cmp rcx, r10
    jge .Ldone
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lskip
    inc rax
.Lskip:
    inc rcx
    jmp .Lloop
.Ldone:
    ret
