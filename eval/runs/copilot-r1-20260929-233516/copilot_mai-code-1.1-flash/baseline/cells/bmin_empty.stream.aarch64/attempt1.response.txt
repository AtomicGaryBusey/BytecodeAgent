bmin_empty:
    cbz x1, .Lzero
    ldr x3, [x0]
    mov x2, #1
.Lloop:
    cmp x2, x1
    bhs .Ldone
    ldr x9, [x0, x2, lsl #3]
    cmp x9, x3
    bge .Lskip
    mov x3, x9
.Lskip:
    add x2, x2, #1
    b .Lloop
.Ldone:
    mov x0, x3
    ret
.Lzero:
    movz x0, #0xFFFF
    movk x0, #0xFFFF, lsl #16
    movk x0, #0xFFFF, lsl #32
    movk x0, #0x7FFF, lsl #48
    ret
