export function l $count_desc(l %a, l %n) {
@entry
  %lt2 = csltl %n, $2
  jnz %lt2, @ret0
  %limit = sub l %n, $1
  jmp @loop

@loop
  %i = phi l [$0, @entry], [%i1, @step]
  %p = phi l [%a, @entry], [%p1, @step]
  %count = phi l [$0, @entry], [%count1, @step]
  %cond = csltl %i, %limit
  jnz %cond, @body
  ret %count

@body
  %cur = loadl %p
  %p2 = add l %p, $8
  %nxt = loadl %p2
  %gt = csgtl %cur, %nxt
  jnz %gt, @inc
  jmp @step

@inc
  %count_inc = add l %count, $1
  jmp @step

@step
  %count1 = phi l [%count, @body], [%count_inc, @inc]
  %i1 = add l %i, $1
  %p1 = add l %p, $8
  jmp @loop

@ret0
  ret $0
}
