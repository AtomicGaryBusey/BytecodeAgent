export function l $count_desc(l %a, l %n) {
@entry
  %lt2 = csltl %n, $2
  jnz %lt2, @ret0
  %limit = sub l %n, $1
  %i0 = add l $0, $0
  %p0 = add l %a, $0
  %count0 = add l $0, $0
  jmp @loop

@loop
  %i = phi l [%i0, @entry], [%i1b, @body], [%i1c, @inc]
  %p = phi l [%p0, @entry], [%p1b, @body], [%p1c, @inc]
  %count = phi l [%count0, @entry], [%count1b, @body], [%count1c, @inc]
  %cond = csltl %i, %limit
  jnz %cond, @body
  ret %count

@body
  %cur = loadl %p
  %nxtp = add l %p, $8
  %nxt = loadl %nxtp
  %gt = csgtl %cur, %nxt
  jnz %gt, @inc
  %count1b = add l %count, $0
  %i1b = add l %i, $1
  %p1b = add l %p, $8
  jmp @loop

@inc
  %count1c = add l %count, $1
  %i1c = add l %i, $1
  %p1c = add l %p, $8
  jmp @loop

@ret0
  ret $0
}
