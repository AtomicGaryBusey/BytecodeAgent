export function l $count_desc(l %a, l %n) {
@entry
  %lt2 = csltl %n, $2
  jnz %lt2, @ret0
  %limit = sub l %n, $1
  %i0 = add l $0, $0
  %count0 = add l $0, $0
  jmp @loop

@loop
  %i = phi l [%i0, @entry], [%i1, @after]
  %count = phi l [%count0, @entry], [%count1, @after]
  %cond = csltl %i, %limit
  jnz %cond, @body
  ret %count

@body
  %i8 = mul l %i, $8
  %p = add l %a, %i8
  %cur = loadl %p
  %nxtp = add l %p, $8
  %nxt = loadl %nxtp
  %gt = csgtl %cur, %nxt
  jnz %gt, @inc
  %i_skip = add l %i, $1
  %count_skip = add l %count, $0
  jmp @after

@inc
  %count_inc = add l %count, $1
  %i_inc = add l %i, $1
  jmp @after

@after
  %i1 = phi l [%i_inc, @inc], [%i_skip, @body]
  %count1 = phi l [%count_inc, @inc], [%count_skip, @body]
  jmp @loop

@ret0
  ret $0
}
