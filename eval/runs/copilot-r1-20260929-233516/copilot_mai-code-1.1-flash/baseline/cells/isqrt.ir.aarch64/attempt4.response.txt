export function l $isqrt(l %n) {
@entry
  %zero =l copy 0
  %one =l copy 1
  %res0 =l copy %zero
  %bit0 =l shl %one, 62
  jmp @loop
@loop
  %res =l phi @entry %res0, @bitshrink %res1, @update_yes %res2, @update_no %res3
  %bit =l phi @entry %bit0, @bitshrink %bit1, @update_yes %bit2, @update_no %bit3
  %nval =l phi @entry %n, @bitshrink %n1, @update_yes %n2, @update_no %n3
  %cond_bit =w cugt %bit, %nval
  jnz %cond_bit, @bitshrink, @check_zero
@bitshrink
  %res1 =l copy %res
  %bit1 =l shr %bit, 2
  %n1 =l copy %nval
  jmp @loop
@check_zero
  %cond_zero =w ceql %bit, 0
  jnz %cond_zero, @done, @check_sum
@check_sum
  %sum =l add %res, %bit
  %eq =w ceql %nval, %sum
  %gt =w cugt %nval, %sum
  %ge =w or %eq, %gt
  jnz %ge, @update_yes, @update_no
@update_yes
  %n_sub =l sub %nval, %sum
  %res_half =l shr %res, 1
  %res_new =l add %res_half, %bit
  %res2 =l copy %res_new
  %bit2 =l shr %bit, 2
  %n2 =l copy %n_sub
  jmp @loop
@update_no
  %res_half =l shr %res, 1
  %res3 =l copy %res_half
  %bit3 =l shr %bit, 2
  %n3 =l copy %nval
  jmp @loop
@done
  ret %res
}
