export function l $isqrt(l %n) {
@entry
  %res0 =l const 0
  %bit0 =l const 4611686018427387904
  jmp @loop
@loop
  %res =l phi @entry %res0, @shift %res1, @update_yes %res2, @update_no %res3
  %bit =l phi @entry %bit0, @shift %bit1, @update_yes %bit2, @update_no %bit3
  %nval =l phi @entry %n, @shift %n1, @update_yes %n2, @update_no %n3
  %cond1 =w cugtl %bit %nval
  jnz %cond1, @shift, @check
@shift
  %bit_shift =l shrl %bit 2
  %res1 =l copy %res
  %bit1 =l copy %bit_shift
  %n1 =l copy %nval
  jmp @loop
@check
  %cond2 =w ceql %bit 0
  jnz %cond2, @done, @body
@body
  %sum =l addl %res %bit
  %eq =w ceql %nval %sum
  %gt =w cugtl %nval %sum
  %cond3 =w orw %eq %gt
  jnz %cond3, @update_yes, @update_no
@update_yes
  %n_sub =l subl %nval %sum
  %res_half =l shrl %res 1
  %res_new =l addl %res_half %bit
  %res2 =l copy %res_new
  %bit2 =l shrl %bit 2
  %n2 =l copy %n_sub
  jmp @loop
@update_no
  %res_half =l shrl %res 1
  %res3 =l copy %res_half
  %bit3 =l shrl %bit 2
  %n3 =l copy %nval
  jmp @loop
@done
  ret %res
}
