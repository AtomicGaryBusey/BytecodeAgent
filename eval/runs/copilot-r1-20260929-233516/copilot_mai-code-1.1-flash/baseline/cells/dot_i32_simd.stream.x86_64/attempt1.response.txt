dot_i32_simd:
    xor rax, rax
    mov r8, rdi
    mov r9, rsi
    mov rcx, rdx
    shr rcx, 2
    jz .Ltail
.Lmain:
    movdqu xmm0, xmmword ptr [r8]
    movdqu xmm1, xmmword ptr [r9]
    movdqu xmm2, xmm0
    movdqu xmm3, xmm1
    pshufd xmm2, xmm2, 0xDD
    pshufd xmm3, xmm3, 0xDD
    pmuldq xmm0, xmm1
    pmuldq xmm2, xmm3
    paddq xmm0, xmm2
    movq r10, xmm0
    psrldq xmm0, 8
    movq r11, xmm0
    add r10, r11
    add rax, r10
    add r8, 16
    add r9, 16
    dec rcx
    jnz .Lmain
.Ltail:
    mov rcx, rdx
    and rcx, 3
    jz .Ldone
.Ltail_loop:
    movsxd r10, dword ptr [r8]
    movsxd r11, dword ptr [r9]
    imul r10, r10, r11
    add rax, r10
    add r8, 4
    add r9, 4
    dec rcx
    jnz .Ltail_loop
.Ldone:
    ret
