dot_i32_simd:
    cld
    xor rax, rax
    test rdx, rdx
    jz .Ldone
    mov rcx, rdx
    shr rcx, 1
    jz .Ltail
.Lmain:
    movdqu xmm2, xmmword ptr [rdi]
    movdqu xmm3, xmmword ptr [rsi]
    pmovsxdq xmm0, xmm2
    pmovsxdq xmm1, xmm3
    pmuldq xmm0, xmm1
    movq r8, xmm0
    psrldq xmm0, 8
    movq r9, xmm0
    add rax, r8
    add rax, r9
    add rdi, 8
    add rsi, 8
    dec rcx
    jnz .Lmain
.Ltail:
    mov rcx, rdx
    and rcx, 1
    jz .Ldone
.Ltail_loop:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Ltail_loop
.Ldone:
    ret
