dot_i32_simd:
    cld
    xor rax, rax
    mov rcx, rdx
    shr rcx, 2
    jz .Ltail
.Lmain:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqu xmm2, xmm0
    movdqu xmm3, xmm1
    pshufd xmm2, xmm2, 0x4E
    pshufd xmm3, xmm3, 0x4E
    pmuldq xmm0, xmm1
    pmuldq xmm2, xmm3
    paddq xmm0, xmm2
    movq r10, xmm0
    psrldq xmm0, 8
    movq r11, xmm0
    add r10, r11
    add rax, r10
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lmain
.Ltail:
    mov rcx, rdx
    and rcx, 3
    jz .Ldone
.Ltail_loop:
    movsxd r10, dword ptr [rdi]
    movsxd r11, dword ptr [rsi]
    imul r10, r11
    add rax, r10
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Ltail_loop
.Ldone:
    ret
