export function l $max_byte(l %p, l %n) {
@entry
  %is_zero =w ceql %n, 0
  jnz %is_zero, @empty, @loop
@empty
  ret 0
@loop
  %max =l phi @entry 0, @tail %max2
  %pcur =l phi @entry %p, @tail %pcur1
  %ncur =l phi @entry %n, @tail %ncur1
  %nzero =w ceql %ncur, 0
  jnz %nzero, @done, @body
@body
  %byte =l loadub %pcur
  %gt =w csltl %max, %byte
  jnz %gt, @update, @skip
@update
  %max1 =l copy %byte
  jmp @tail
@skip
  %max1 =l copy %max
  jmp @tail
@tail
  %max2 =l phi @update %max1, @skip %max1
  %pcur1 =l add %pcur, 1
  %ncur1 =l sub %ncur, 1
  jmp @loop
@done
  ret %max
}
