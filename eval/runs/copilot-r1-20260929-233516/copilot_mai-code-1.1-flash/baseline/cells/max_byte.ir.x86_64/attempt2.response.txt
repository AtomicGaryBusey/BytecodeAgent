export function l $max_byte(l %p, l %n) {
@entry
  %is_zero =w ceql %n, 0
  jnz %is_zero, @empty, @loop
@loop
  %max =l phi @entry 0, @tail %max_after
  %pcur =l phi @entry %p, @tail %pcur_after
  %ncur =l phi @entry %n, @tail %ncur_after
  %n_zero =w ceql %ncur, 0
  jnz %n_zero, @done, @body
@body
  %byte =l loadub %pcur
  %gt =w csltl %max, %byte
  jnz %gt, @update, @skip
@update
  %max_update =l copy %byte
  jmp @tail
@skip
  %max_skip =l copy %max
  jmp @tail
@tail
  %max_after =l phi @update %max_update, @skip %max_skip
  %pcur_after =l add %pcur, 1
  %ncur_after =l sub %ncur, 1
  jmp @loop
@done
  ret %max
@empty
  ret 0
}
