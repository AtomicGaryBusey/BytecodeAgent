bmin_empty:
    cmp rsi, 0
    jne .Lnonempty
    mov rax, 0x7fffffffffffffff
    ret
.Lnonempty:
    mov rax, qword ptr [rdi]
    mov rcx, 1
.Lloop:
    cmp rcx, rsi
    jge .Ldone
    mov r8, qword ptr [rdi + rcx*8]
    cmp r8, rax
    jge .Lskip
    mov rax, r8
.Lskip:
    inc rcx
    jmp .Lloop
.Ldone:
    ret
