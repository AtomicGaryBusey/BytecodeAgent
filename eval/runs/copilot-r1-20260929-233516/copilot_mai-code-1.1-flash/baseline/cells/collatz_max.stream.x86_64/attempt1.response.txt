collatz_max:
    cld
    push r15
    push r14
    push r13
    push r12
    push rbx
    mov r12, rdi
    mov r13, 1
    mov r15, 0
.Lloop_k:
    cmp r13, r12
    jg .Ldone
    mov r14, r13
    xor rbx, rbx
.Lloop_steps:
    cmp r14, 1
    je .Lstep_done
    test r14, 1
    jz .Leven
    lea r14, [r14 + r14*2]
    add r14, 1
    jmp .Lafter
.Leven:
    shr r14, 1
.Lafter:
    inc rbx
    jmp .Lloop_steps
.Lstep_done:
    cmp rbx, r15
    jle .Lskip_update
    mov r15, rbx
.Lskip_update:
    inc r13
    jmp .Lloop_k
.Ldone:
    mov rax, r15
    pop rbx
    pop r12
    pop r13
    pop r14
    pop r15
    ret
