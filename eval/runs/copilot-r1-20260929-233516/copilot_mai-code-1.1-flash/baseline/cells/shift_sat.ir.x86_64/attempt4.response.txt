export function l $shift_sat(l %x, l %n) {
@start
    %zero = copy 0
    %one = copy 1
    %sixtyfour = copy 64
    %n_lt_64 = csltl %n, %sixtyfour
    jnz %n_lt_64, @lt64, @ge64

@lt64
    %left = shl %x, %n
    %right = sar %x, %n
    %sum = add %left, %right
    ret %sum

@ge64
    %neg = csltl %x, %zero
    jnz %neg, @neg, @nonneg

@neg
    %negone = sub %zero, %one
    ret %negone

@nonneg
    ret %zero
}
