export function l $shift_sat(l %x, l %n) {
@start
    %n_lt_64 = csltl %n, 64
    jnz %n_lt_64, @lt64, @ge64

@lt64
    %left = shl %x, %n
    %right = sar %x, %n
    %sum = add %left, %right
    ret %sum

@ge64
    %neg = csltl %x, 0
    jnz %neg, @neg, @nonneg

@neg
    %negone = sub 0, 1
    ret %negone

@nonneg
    ret 0
}
