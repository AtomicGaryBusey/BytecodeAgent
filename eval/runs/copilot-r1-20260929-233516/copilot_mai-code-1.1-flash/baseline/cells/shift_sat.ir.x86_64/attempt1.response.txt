export function l $shift_sat(l %x, l %n) {
@start
    %n_lt_64 = csltl %n, 64
    jnz %n_lt_64, @lt64, @ge64

@lt64
    %left = shl %x, %n
    %right = sar %x, %n
    %sum = add %left, %right
    ret %sum

@ge64
    %neg = csltl %x, 0
    jnz %neg, @neg, @nonneg

@neg
    %left_neg = add 0, 0
    %right_neg = add -1, 0
    %sum_neg = add %left_neg, %right_neg
    ret %sum_neg

@nonneg
    %left_nonneg = add 0, 0
    %right_nonneg = add 0, 0
    %sum_nonneg = add %left_nonneg, %right_nonneg
    ret %sum_nonneg
}
