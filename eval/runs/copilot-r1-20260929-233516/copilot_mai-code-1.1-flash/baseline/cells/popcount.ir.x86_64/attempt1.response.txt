export function l $popcount(l %x) {
@entry
  %t0 = shrl %x, 1
  %t1 = andl %t0, 0x5555555555555555
  %x1 = subl %x, %t1
  %t2 = shrl %x1, 2
  %t3 = andl %t2, 0x3333333333333333
  %t4 = andl %x1, 0x3333333333333333
  %x2 = addl %t4, %t3
  %t5 = shrl %x2, 4
  %t6 = addl %x2, %t5
  %x3 = andl %t6, 0x0f0f0f0f0f0f0f0f
  %t7 = shrl %x3, 8
  %x4 = addl %x3, %t7
  %t8 = shrl %x4, 16
  %x5 = addl %x4, %t8
  %t9 = shrl %x5, 32
  %x6 = addl %x5, %t9
  %count = andl %x6, 0xff
  ret %count
}
