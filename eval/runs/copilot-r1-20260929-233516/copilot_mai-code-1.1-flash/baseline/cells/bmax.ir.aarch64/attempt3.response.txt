export function l $bmax(l %a, l %n) {
@entry
%best = loadl %a
%idx = add 0, 0
@loop
%cond = csltl %idx, %n
jnz %cond, @body
j @done
@body
%off = mul %idx, 8
%ptr = add %a, %off
%val = loadl %ptr
%is_greater = csltl %best, %val
jnz %is_greater, @update
j @next
@update
%best = add %val, 0
@next
%idx = add %idx, 1
j @loop
@done
ret %best
}
