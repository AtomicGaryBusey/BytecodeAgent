export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %len =l add %na, %nb
    %zero =l copy 0
    %i =l copy 0
    jmp @zloop

@zloop
    %zcond =w csltl %i, %len
    jnz %zcond, @zbody
    jmp @mul

@zbody
    %zoff =l shl %i, 3
    %zptr =l add %r, %zoff
    storel %zero, %zptr, 0
    %i =l add %i, 1
    jmp @zloop

@mul
    %i =l copy 0
    jmp @iloop

@iloop
    %icond =w csltl %i, %na
    jnz %icond, @ibody
    jmp @count

@ibody
    %j =l copy 0
    jmp @jloop

@jloop
    %jcond =w csltl %j, %nb
    jnz %jcond, @jbody
    jmp @jdone

@jbody
    %k =l add %i, %j
    %koff =l shl %k, 3
    %rptr =l add %r, %koff
    %old =l loadl %rptr, 0

    %aoff =l shl %i, 3
    %aptr =l add %a, %aoff
    %x =l loadl %aptr, 0

    %boff =l shl %j, 3
    %bptr =l add %b, %boff
    %y =l loadl %bptr, 0

    %x0 =l and %x, 4294967295
    %x1 =l shr %x, 32
    %y0 =l and %y, 4294967295
    %y1 =l shr %y, 32

    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1

    %m =l add %p01, %p10
    %mcarryw =w cultl %m, %p01
    %mcarry =l cast %mcarryw
    %mlow32 =l and %m, 4294967295
    %mshift =l shl %mlow32, 32
    %t =l add %p00, %mshift
    %tcarryw =w cultl %t, %p00
    %tcarry =l cast %tcarryw
    %mhi32 =l shr %m, 32
    %p11m =l add %p11, %mhi32
    %p_hi =l add %p11m, %mcarry
    %p_hi =l add %p_hi, %tcarry

    %sum0 =l add %old, %t
    %c0 =w cultl %sum0, %old
    %sum1 =l add %sum0, 0
    %c1 =w cultl %sum1, %sum0
    storel %sum1, %rptr, 0

    %t0 =l add %p_hi, %c0
    %c0h =w cultl %t0, %p_hi
    %t1 =l add %t0, %c1
    %c1h =w cultl %t1, %t0
    %carry_lo =l copy %t1
    %carry_hi =l add %c0h, %c1h

    %k1 =l add %k, 1
    %k1off =l shl %k1, 3
    %r1ptr =l add %r, %k1off
    %old1 =l loadl %r1ptr, 0
    %sum2 =l add %old1, %carry_lo
    %c2 =w cultl %sum2, %old1
    storel %sum2, %r1ptr, 0
    %carry =l add %carry_hi, %c2

    %k2 =l add %k1, 1
    jmp @carryprop

@carryprop
    %carry_zero =w ceql %carry, 0
    jnz %carry_zero, @aftercarry
    %k2off =l shl %k2, 3
    %r2ptr =l add %r, %k2off
    %old2 =l loadl %r2ptr, 0
    %sum3 =l add %old2, %carry
    %c3 =w cultl %sum3, %old2
    storel %sum3, %r2ptr, 0
    %carry =l cast %c3
    %k2 =l add %k2, 1
    jmp @carryprop

@aftercarry
    %j =l add %j, 1
    jmp @jloop

@jdone
    %i =l add %i, 1
    jmp @iloop

@count
    %pos =l add %len, -1
    jmp @cloop

@cloop
    %neg =w csltl %pos, 0
    jnz %neg, @czero
    %off =l shl %pos, 3
    %ptr =l add %r, %off
    %v =l loadl %ptr, 0
    %vzero =w ceql %v, 0
    jnz %vzero, @cskip
    %cnt =l add %pos, 1
    ret %cnt

@cskip
    %pos =l add %pos, -1
    jmp @cloop

@czero
    %cnt =l copy 0
    ret %cnt
}
