export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %len =l add %na, %nb
    %zero =l copy 0
    %i =l copy 0
    jmp @zloop

@zloop
    %zcond =w csltl %i, %len
    jnz %zcond, @zbody
    jmp @mul

@zbody
    %zoff =l shl %i, 3
    %zptr =l add %r, %zoff
    storel %zero, %zptr, 0
    %i =l add %i, 1
    jmp @zloop

@mul
    %i =l copy 0
    jmp @iloop

@iloop
    %icond =w csltl %i, %na
    jnz %icond, @ibegin
    jmp @count

@ibegin
    %j =l copy 0
    %carry_lo =l copy 0
    %carry_hi =l copy 0
    jmp @jloop

@jloop
    %jcond =w csltl %j, %nb
    jnz %jcond, @jbody
    jmp @jdone

@jbody
    %idx =l add %i, %j
    %off =l shl %idx, 3
    %ptr =l add %r, %off
    %old =l loadl %ptr, 0

    %aoff =l shl %i, 3
    %aptr =l add %a, %aoff
    %x =l loadl %aptr, 0

    %boff =l shl %j, 3
    %bptr =l add %b, %boff
    %y =l loadl %bptr, 0

    %x0 =l and %x, 4294967295
    %x1 =l shr %x, 32
    %y0 =l and %y, 4294967295
    %y1 =l shr %y, 32

    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1

    %m =l add %p01, %p10
    %mcarryw =w cultl %m, %p01
    %mcarry =l cast %mcarryw
    %mlow32 =l and %m, 4294967295
    %mshift =l shl %mlow32, 32
    %t =l add %p00, %mshift
    %tcarryw =w cultl %t, %p00
    %tcarry =l cast %tcarryw
    %mhi32 =l shr %m, 32
    %p_hi =l add %p11, %mhi32
    %p_hi =l add %p_hi, %mcarry
    %p_hi =l add %p_hi, %tcarry

    %sum0 =l add %old, %t
    %c0 =w cultl %sum0, %old
    %sum1 =l add %sum0, %carry_lo
    %c1 =w cultl %sum1, %sum0
    storel %sum1, %ptr, 0

    %h0 =l add %p_hi, %carry_hi
    %c00 =w cultl %h0, %p_hi
    %h1 =l add %h0, %c0
    %c01 =w cultl %h1, %h0
    %h2 =l add %h1, %c1
    %c02 =w cultl %h2, %h1
    %carry_lo =l copy %h2

    %carry_hi_0 =l cast %c00
    %carry_hi_1 =l cast %c01
    %carry_hi_2 =l cast %c02
    %carry_hi =l add %carry_hi_0, %carry_hi_1
    %carry_hi =l add %carry_hi, %carry_hi_2

    %j =l add %j, 1
    jmp @jloop

@jdone
    %k =l add %i, %nb
    %koff =l shl %k, 3
    %kptr =l add %r, %koff
    %oldk =l loadl %kptr, 0
    %sumk =l add %oldk, %carry_lo
    %ck =w cultl %sumk, %oldk
    storel %sumk, %kptr, 0
    %carry =l add %carry_hi, %ck
    %k =l add %k, 1
    jmp @prop

@prop
    %carry_zero =w ceql %carry, 0
    jnz %carry_zero, @nexti
    %koff2 =l shl %k, 3
    %kptr2 =l add %r, %koff2
    %oldp =l loadl %kptr2, 0
    %sump =l add %oldp, %carry
    %cp =w cultl %sump, %oldp
    storel %sump, %kptr2, 0
    %carry =l cast %cp
    %k =l add %k, 1
    jmp @prop

@nexti
    %i =l add %i, 1
    jmp @iloop

@count
    %len2 =l add %na, %nb
    %pos =l add %len2, -1
    jmp @cloop

@cloop
    %neg =w csltl %pos, 0
    jnz %neg, @czero
    %off =l shl %pos, 3
    %ptr =l add %r, %off
    %v =l loadl %ptr, 0
    %vzero =w ceql %v, 0
    jnz %vzero, @cskip
    %cnt =l add %pos, 1
    ret %cnt

@cskip
    %pos =l add %pos, -1
    jmp @cloop

@czero
    ret 0
}
