export function l $count_desc(l %a, l %n) {
@entry
    %lt2 = csltl %n, 2
    jnz %lt2, @ret0, @start
@start
    %n1 = subl %n, 1
    %shift = shll %n1, 3
    %limit = addl %a, %shift
    jmp @loop
@loop
    %p = phi l [ %a, @start ], [ %pnext, @after ]
    %c = phi l [ 0, @start ], [ %cnext, @after ]
    %cond = csltl %p, %limit
    jnz %cond, @body, @exit
@body
    %v1 = loadl %p
    %p1 = addl %p, 8
    %v2 = loadl %p1
    %gt = csgtl %v1, %v2
    jnz %gt, @inc, @keep
@inc
    %cinc = addl %c, 1
    jmp @after
@keep
    %ckeep = copyl %c
    jmp @after
@after
    %pnext = addl %p, 8
    %cnext = phi l [ %cinc, @inc ], [ %ckeep, @keep ]
    jmp @loop
@exit
    ret %c
@ret0
    ret 0
}
