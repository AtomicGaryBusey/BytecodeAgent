export function l $count_desc(l %a, l %n) {
@entry
    %two = copyl 2
    %lt2 = csltl %n, %two
    jnz %lt2, @ret0, @loop
@loop
    %i = phi l [ 0, @entry ], [ %i2, @inc ], [ %i2, @skip ]
    %p = phi l [ %a, @entry ], [ %p2, @inc ], [ %p2, @skip ]
    %c = phi l [ 0, @entry ], [ %cinc, @inc ], [ %cskip, @skip ]
    %one = copyl 1
    %limit = subl %n, %one
    %cond = csltl %i, %limit
    jnz %cond, @body, @done
@body
    %v1 = loadl %p
    %p1 = addl %p, 8
    %v2 = loadl %p1
    %gt = csgtl %v1, %v2
    jnz %gt, @inc, @skip
@inc
    %cinc = addl %c, 1
    %i2 = addl %i, 1
    %p2 = addl %p, 8
    jmp @loop
@skip
    %cskip = copyl %c
    %i2 = addl %i, 1
    %p2 = addl %p, 8
    jmp @loop
@done
    ret %c
@ret0
    ret 0
}
