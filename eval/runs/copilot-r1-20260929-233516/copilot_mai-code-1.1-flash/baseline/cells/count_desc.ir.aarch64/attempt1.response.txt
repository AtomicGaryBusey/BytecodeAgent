export function l $count_desc(l %a, l %n) {
@entry
    %lt2 = csltl %n, 2
    jnz %lt2, @ret0, @after_entry
@after_entry
    %n1 = subl %n, 1
    %shift = shll %n1, 3
    %limit = addl %a, %shift
    jmp @loop
@loop
    %p = phi l [ %a, @after_entry ], [ %p_next, @after_cmp ]
    %c = phi l [ 0, @after_entry ], [ %c_next, @after_cmp ]
    %cond = csltl %p, %limit
    jnz %cond, @body, @exit
@body
    %v1 = loadl %p
    %p1 = addl %p, 8
    %v2 = loadl %p1
    %gt = csgtl %v1, %v2
    jnz %gt, @inc, @keep
@inc
    %c_inc = addl %c, 1
    jmp @after_cmp
@keep
    %c_keep = copyl %c
    jmp @after_cmp
@after_cmp
    %c_next = phi l [ %c_inc, @inc ], [ %c_keep, @keep ]
    %p_next = addl %p, 8
    jmp @loop
@exit
    ret %c
@ret0
    ret 0
}
