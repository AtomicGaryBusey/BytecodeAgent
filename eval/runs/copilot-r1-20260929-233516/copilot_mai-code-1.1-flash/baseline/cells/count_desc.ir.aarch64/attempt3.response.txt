export function l $count_desc(l %a, l %n) {
@entry
    %lt2 = csltl %n, 2
    jnz %lt2, @ret0, @loop
@loop
    %i = phi l [ 0, @entry ], [ %i2, @after ]
    %p = phi l [ %a, @entry ], [ %p2, @after ]
    %c = phi l [ 0, @entry ], [ %c2, @after ]
    %limit = subl %n, 1
    %cond = csltl %i, %limit
    jnz %cond, @body, @done
@body
    %v1 = loadl %p
    %p1 = addl %p, 8
    %v2 = loadl %p1
    %gt = csgtl %v1, %v2
    jnz %gt, @inc, @skip
@inc
    %cinc = addl %c, 1
    jmp @after
@skip
    %cskip = copyl %c
    jmp @after
@after
    %i2 = addl %i, 1
    %p2 = addl %p, 8
    %c2 = phi l [ %cinc, @inc ], [ %cskip, @skip ]
    jmp @loop
@done
    ret %c
@ret0
    ret 0
}
