export function l $shift_sat(l %x, l %n) {
@entry
%zero = copy 0
%minus1 = copy -1
%sixtyfour = copy 64

%nlt64 = csltl %n, %sixtyfour
jnz %nlt64, @small

%isneg = csltl %x, %zero
jnz %isneg, @negbig

%left = copy %zero
%right = copy %zero
jmp @join

@negbig
%left = copy %zero
%right = copy %minus1
jmp @join

@small
%left = shll %x, %n
%right = sarl %x, %n

@join
%sum = addl %left, %right
ret l %sum
}
