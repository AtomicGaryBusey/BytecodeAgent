export function l $shift_sat(l %x, l %n) {
@entry
%nlt64 = csltl %n, 64
jnz %nlt64, @small
%isneg = csltl %x, 0
jnz %isneg, @negbig
%left = add l 0, 0
%right = add l 0, 0
jmp @join
@negbig
%left = add l 0, 0
%right = sub l 0, 1
jmp @join
@small
%left = shl l %x, %n
%right = sar l %x, %n
jmp @join
@join
%sum = add l %left, %right
ret l %sum
}
