export function l $shift_sat(l %x, l %n) {
@entry
%zero = copyl 0
%minus1 = copyl -1
%sixtyfour = copyl 64

%nlt64 = csltl %n, %sixtyfour
jnz %nlt64, @small

%isneg = csltl %x, %zero
jnz %isneg, @negbig

%left = copyl %zero
%right = copyl %zero
jmp @join

@negbig
%left = copyl %zero
%right = copyl %minus1
jmp @join

@small
%left = shll %x, %n
%right = sarl %x, %n

@join
%sum = addl %left, %right
ret l %sum
}
