export function l $shift_sat(l %x, l %n) {
@entry
%nlt64 = csltl %n, 64
jnz %nlt64, @small
%isneg = csltl %x, 0
jnz %isneg, @negbig
%left = addl 0, 0
%right = addl 0, 0
jmp @join
@negbig
%left = addl 0, 0
%right = subl 0, 1
jmp @join
@small
%left = shll %x, %n
%right = sarl %x, %n
@join
%sum = addl %left, %right
ret l %sum
}
