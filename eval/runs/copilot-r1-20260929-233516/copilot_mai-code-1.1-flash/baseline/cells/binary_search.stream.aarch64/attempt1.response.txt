binary_search:
    cmp x1, #0
    b.le .Lnotfound
    mov x9, xzr
    mov x10, x1
    mov x12, x0
.Lloop:
    cmp x9, x10
    b.hs .Ldone
    add x11, x9, x10
    lsr x11, x11, #1
    ldr x13, [x12, x11, lsl #3]
    cmp x13, x2
    b.lt .Llower
    mov x10, x11
    b .Lloop
.Llower:
    add x9, x11, #1
    b .Lloop
.Ldone:
    cmp x9, x1
    b.hs .Lnotfound
    ldr x13, [x12, x9, lsl #3]
    cmp x13, x2
    b.ne .Lnotfound
    mov x0, x9
    ret
.Lnotfound:
    mov x0, #-1
    ret
