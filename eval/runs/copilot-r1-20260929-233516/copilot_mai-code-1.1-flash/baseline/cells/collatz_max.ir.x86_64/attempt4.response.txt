export function l $collatz_max(l %n) {
@entry
  %best0 =l copy 0
  %k0 =l copy 1
  jmp @outer_cond

@outer_cond
  %best =l phi @entry %best0, @after_best %best_next
  %k =l phi @entry %k0, @after_best %k_next
  %k_gt_n =w csgtl %k, %n
  jnz %k_gt_n, @done, @body

@body
  %x0 =l copy %k
  %steps0 =l copy 0
  jmp @inner_cond

@inner_cond
  %x =l phi @body %x0, @inner_next %x_next
  %steps =l phi @body %steps0, @inner_next %steps_next
  %x_is_one =w ceql %x, 1
  jnz %x_is_one, @inner_done, @inner_body

@inner_body
  %x_and_1 =l and %x, 1
  %is_even =w ceql %x_and_1, 0
  jnz %is_even, @even_case, @odd_case

@even_case
  %x_even =l div %x, 2
  %steps_even =l add %steps, 1
  jmp @inner_next

@odd_case
  %x_mul =l mul %x, 3
  %x_odd =l add %x_mul, 1
  %steps_odd =l add %steps, 1
  jmp @inner_next

@inner_next
  %x_next =l phi @even_case %x_even, @odd_case %x_odd
  %steps_next =l phi @even_case %steps_even, @odd_case %steps_odd
  jmp @inner_cond

@inner_done
  %steps_gt_best =w csgtl %steps, %best
  jnz %steps_gt_best, @best_update, @best_skip

@best_update
  %best_update_value =l copy %steps
  jmp @after_best

@best_skip
  %best_skip_value =l copy %best
  jmp @after_best

@after_best
  %best_next =l phi @best_update %best_update_value, @best_skip %best_skip_value
  %k_next =l add %k, 1
  jmp @outer_cond

@done
  ret %best
}
