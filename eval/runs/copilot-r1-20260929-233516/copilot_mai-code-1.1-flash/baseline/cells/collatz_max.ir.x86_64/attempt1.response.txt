export function l $collatz_max(l %n) {
@entry
  %best0 =l copy 0
  %k0 =l copy 1
  jmp @outer_cond

@outer_cond
  %best =l phi @entry %best0, @after_best %best_next
  %k =l phi @entry %k0, @after_best %k_next
  %k_gt_n =w csgtl %k, %n
  jnz %k_gt_n, @done, @body

@body
  %x0 =l copy %k
  %steps0 =l copy 0
  jmp @inner_cond

@inner_cond
  %x1 =l phi @body %x0, @inner_next %x2
  %steps1 =l phi @body %steps0, @inner_next %steps2
  %x_is_one =w ceql %x1, 1
  jnz %x_is_one, @inner_done, @inner_body

@inner_body
  %low =l andl %x1, 1
  %is_even =w ceql %low, 0
  jnz %is_even, @even_case, @odd_case

@even_case
  %x_even =l divl %x1, 2
  %steps_even =l addl %steps1, 1
  jmp @inner_next

@odd_case
  %x_mul =l mull %x1, 3
  %x_odd =l addl %x_mul, 1
  %steps_odd =l addl %steps1, 1
  jmp @inner_next

@inner_next
  %x2 =l phi @even_case %x_even, @odd_case %x_odd
  %steps2 =l phi @even_case %steps_even, @odd_case %steps_odd
  jmp @inner_cond

@inner_done
  %cmp_best =w csgtl %steps1, %best
  jnz %cmp_best, @best_update, @skip_update

@best_update
  %best_update_value =l copy %steps1
  jmp @after_best

@skip_update
  %best_skip_value =l copy %best
  jmp @after_best

@after_best
  %best_next =l phi @best_update %best_update_value, @skip_update %best_skip_value
  %k_next =l addl %k, 1
  jmp @outer_cond

@done
  ret %best
}
