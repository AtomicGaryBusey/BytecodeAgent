export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    %pc0 = copy $0
    %sp0 = copy $0
    jmp @loop

@loop
    %pc = phi l [%pc0, @entry], [%pc_push, @case_push], [%pc_add, @case_add], [%pc_sub, @case_sub], [%pc_mul, @case_mul], [%pc_dup, @case_dup], [%pc_swap, @case_swap], [%pc_drop, @case_drop], [%pc_over, @case_over], [%pc_lt_true, @case_lt_true], [%pc_lt_false, @case_lt_false], [%pc_jz_taken, @case_jz_taken], [%pc_jz_else, @case_jz_else], [%pc_jmp, @case_jmp], [%pc_default, @case_default]
    %sp = phi l [%sp0, @entry], [%sp_push, @case_push], [%sp_add, @case_add], [%sp_sub, @case_sub], [%sp_mul, @case_mul], [%sp_dup, @case_dup], [%sp_swap, @case_swap], [%sp_drop, @case_drop], [%sp_over, @case_over], [%sp_lt_true, @case_lt_true], [%sp_lt_false, @case_lt_false], [%sp_jz_taken, @case_jz_taken], [%sp_jz_else, @case_jz_else], [%sp_jmp, @case_jmp], [%sp_default, @case_default]
    %at_end = ceql %pc, %len
    jnz %at_end, @finish
    %opaddr = addl %code, mull %pc, $8
    %op = loadl %opaddr
    %is_halt = ceql %op, $0
    jnz %is_halt, @finish
    %is_push = ceql %op, $1
    jnz %is_push, @case_push
    %is_add = ceql %op, $2
    jnz %is_add, @case_add
    %is_sub = ceql %op, $3
    jnz %is_sub, @case_sub
    %is_mul = ceql %op, $4
    jnz %is_mul, @case_mul
    %is_dup = ceql %op, $5
    jnz %is_dup, @case_dup
    %is_swap = ceql %op, $6
    jnz %is_swap, @case_swap
    %is_drop = ceql %op, $7
    jnz %is_drop, @case_drop
    %is_over = ceql %op, $8
    jnz %is_over, @case_over
    %is_lt = ceql %op, $9
    jnz %is_lt, @case_lt
    %is_jz = ceql %op, $10
    jnz %is_jz, @case_jz
    %is_jmp = ceql %op, $11
    jnz %is_jmp, @case_jmp
    jmp @case_default

@case_push
    %pc1_push = addl %pc, $1
    %argaddr_push = addl %code, mull %pc1_push, $8
    %arg_push = loadl %argaddr_push
    %dst_push = addl %stack, mull %sp, $8
    storel %arg_push, %dst_push
    %sp_push = addl %sp, $1
    %pc_push = addl %pc, $2
    jmp @loop

@case_add
    %bidx_add = subl %sp, $1
    %baddr_add = addl %stack, mull %bidx_add, $8
    %b_add = loadl %baddr_add
    %aidx_add = subl %bidx_add, $1
    %aaddr_add = addl %stack, mull %aidx_add, $8
    %a_add = loadl %aaddr_add
    %sum_add = addl %a_add, %b_add
    storel %sum_add, %aaddr_add
    %sp_add = subl %sp, $1
    %pc_add = addl %pc, $1
    jmp @loop

@case_sub
    %bidx_sub = subl %sp, $1
    %baddr_sub = addl %stack, mull %bidx_sub, $8
    %b_sub = loadl %baddr_sub
    %aidx_sub = subl %bidx_sub, $1
    %aaddr_sub = addl %stack, mull %aidx_sub, $8
    %a_sub = loadl %aaddr_sub
    %diff_sub = subl %a_sub, %b_sub
    storel %diff_sub, %aaddr_sub
    %sp_sub = subl %sp, $1
    %pc_sub = addl %pc, $1
    jmp @loop

@case_mul
    %bidx_mul = subl %sp, $1
    %baddr_mul = addl %stack, mull %bidx_mul, $8
    %b_mul = loadl %baddr_mul
    %aidx_mul = subl %bidx_mul, $1
    %aaddr_mul = addl %stack, mull %aidx_mul, $8
    %a_mul = loadl %aaddr_mul
    %prod_mul = mull %a_mul, %b_mul
    storel %prod_mul, %aaddr_mul
    %sp_mul = subl %sp, $1
    %pc_mul = addl %pc, $1
    jmp @loop

@case_dup
    %topidx_dup = subl %sp, $1
    %topaddr_dup = addl %stack, mull %topidx_dup, $8
    %top_dup = loadl %topaddr_dup
    %dst_dup = addl %stack, mull %sp, $8
    storel %top_dup, %dst_dup
    %sp_dup = addl %sp, $1
    %pc_dup = addl %pc, $1
    jmp @loop

@case_swap
    %bidx_swap = subl %sp, $1
    %baddr_swap = addl %stack, mull %bidx_swap, $8
    %b_swap = loadl %baddr_swap
    %aidx_swap = subl %bidx_swap, $1
    %aaddr_swap = addl %stack, mull %aidx_swap, $8
    %a_swap = loadl %aaddr_swap
    storel %b_swap, %aaddr_swap
    storel %a_swap, %baddr_swap
    %sp_swap = copy %sp
    %pc_swap = addl %pc, $1
    jmp @loop

@case_drop
    %sp_drop = subl %sp, $1
    %pc_drop = addl %pc, $1
    jmp @loop

@case_over
    %belowidx_over = subl %sp, $2
    %belowaddr_over = addl %stack, mull %belowidx_over, $8
    %value_over = loadl %belowaddr_over
    %dst_over = addl %stack, mull %sp, $8
    storel %value_over, %dst_over
    %sp_over = addl %sp, $1
    %pc_over = addl %pc, $1
    jmp @loop

@case_lt
    %bidx_lt = subl %sp, $1
    %baddr_lt = addl %stack, mull %bidx_lt, $8
    %b_lt = loadl %baddr_lt
    %aidx_lt = subl %bidx_lt, $1
    %aaddr_lt = addl %stack, mull %aidx_lt, $8
    %a_lt = loadl %aaddr_lt
    %lt_cmp = csltl %a_lt, %b_lt
    jnz %lt_cmp, @case_lt_true
    storel $0, %aaddr_lt
    %sp_lt_false = subl %sp, $1
    %pc_lt_false = addl %pc, $1
    jmp @loop

@case_lt_true
    storel $1, %aaddr_lt
    %sp_lt_true = subl %sp, $1
    %pc_lt_true = addl %pc, $1
    jmp @loop

@case_jz
    %topidx_jz = subl %sp, $1
    %vaddr_jz = addl %stack, mull %topidx_jz, $8
    %v_jz = loadl %vaddr_jz
    %sp_jz_pop = subl %sp, $1
    %pc1_jz = addl %pc, $1
    %targetaddr_jz = addl %code, mull %pc1_jz, $8
    %target_jz = loadl %targetaddr_jz
    %zero_jz = ceql %v_jz, $0
    jnz %zero_jz, @case_jz_taken
    %sp_jz_else = copy %sp_jz_pop
    %pc_jz_else = addl %pc, $2
    jmp @loop

@case_jz_taken
    %sp_jz_taken = copy %sp_jz_pop
    %pc_jz_taken = copy %target_jz
    jmp @loop

@case_jmp
    %pc1_jmp = addl %pc, $1
    %targetaddr_jmp = addl %code, mull %pc1_jmp, $8
    %target_jmp = loadl %targetaddr_jmp
    %sp_jmp = copy %sp
    %pc_jmp = copy %target_jmp
    jmp @loop

@case_default
    %sp_default = copy %sp
    %pc_default = addl %pc, $1
    jmp @loop

@finish
    %empty = ceql %sp, $0
    jnz %empty, @empty_stack
    %topidx = subl %sp, $1
    %topaddr = addl %stack, mull %topidx, $8
    %res = loadl %topaddr
    %i0_finish = copy $0
    jmp @clear_loop

@empty_stack
    %res = copy $0
    %i0_empty = copy $0
    jmp @clear_loop

@clear_loop
    %i = phi l [%i0_finish, @finish], [%i0_empty, @empty_stack], [%i_next, @clear_loop]
    %done = ceql %i, %cap
    jnz %done, @done
    %keep = csltl %i, %sp
    jnz %keep, @keep
    %addr = addl %stack, mull %i, $8
    storel $0, %addr
    %i_next = addl %i, $1
    jmp @clear_loop

@keep
    %i_next = addl %i, $1
    jmp @clear_loop

@done
    ret %res
}
