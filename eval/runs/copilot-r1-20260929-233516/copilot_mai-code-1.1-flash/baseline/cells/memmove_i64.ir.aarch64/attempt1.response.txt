export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %zero = const l 0
    %nzero = ceql %n, %zero
    jnz %nzero, @done
    %same = ceql %dst, %src
    jnz %same, @done
    %lt = csltl %dst, %src
    jnz %lt, @fwd_loop
    %n1 = subl %n, 1
    jmp @bwd_loop
@fwd_loop
    %fi = phi l [ %zero, @entry ] [ %fi1, @fwd_loop ]
    %fcond = ceql %fi, %n
    jnz %fcond, @done
    %foff = shll %fi, 3
    %fdst = addl %dst, %foff
    %fsrc = addl %src, %foff
    %fval = loadl %fsrc
    storel %fval, %fdst
    %fi1 = addl %fi, 1
    jmp @fwd_loop
@bwd_loop
    %bi = phi l [ %n1, @entry ] [ %bi1, @bwd_loop ]
    %bcond = csltl %bi, %zero
    jnz %bcond, @done
    %boff = shll %bi, 3
    %bdst = addl %dst, %boff
    %bsrc = addl %src, %boff
    %bval = loadl %bsrc
    storel %bval, %bdst
    %bi1 = subl %bi, 1
    jmp @bwd_loop
@done
    ret %zero
}
