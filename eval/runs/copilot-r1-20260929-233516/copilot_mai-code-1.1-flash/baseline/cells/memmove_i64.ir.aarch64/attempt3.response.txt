export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %nzero = ceql %n, 0
    jnz %nzero, @done
    %same = ceql %dst, %src
    jnz %same, @done
    %lt = csltl %dst, %src
    jnz %lt, @fwd
    jmp @bwd

@fwd
    %fi = phi l [ 0, @entry ] [ %fi1, @fwd ]
    %fcond = ceql %fi, %n
    jnz %fcond, @done
    %foff = shll %fi, 3
    %fdst = addl %dst, %foff
    %fsrc = addl %src, %foff
    %fval = loadl %fsrc
    storel %fval, %fdst
    %fi1 = addl %fi, 1
    jmp @fwd

@bwd
    %bi = phi l [ %n, @entry ] [ %bi1, @bwd ]
    %bcond = ceql %bi, 0
    jnz %bcond, @done
    %bi1 = subl %bi, 1
    %boff = shll %bi1, 3
    %bdst = addl %dst, %boff
    %bsrc = addl %src, %boff
    %bval = loadl %bsrc
    storel %bval, %bdst
    jmp @bwd

@done
    ret 0
}
