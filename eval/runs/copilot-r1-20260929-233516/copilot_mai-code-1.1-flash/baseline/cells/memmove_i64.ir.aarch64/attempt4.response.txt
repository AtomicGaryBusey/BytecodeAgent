export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %nzero = ceql %n, 0
    jnz %nzero, @done
    %same = ceql %dst, %src
    jnz %same, @done
    %lt = cultl %dst, %src
    jnz %lt, @forward
    jmp @backward

@forward
    %i = phi l [ 0, @entry ] [ %i1, @forward ]
    %donef = ceql %i, %n
    jnz %donef, @done
    %off = shll %i, 3
    %d = addl %dst, %off
    %s = addl %src, %off
    %v = loadl %s
    storel %v, %d
    %i1 = addl %i, 1
    jmp @forward

@backward
    %i = phi l [ %n, @entry ] [ %i1, @backward ]
    %doneb = ceql %i, 0
    jnz %doneb, @done
    %i1 = subl %i, 1
    %off = shll %i1, 3
    %d = addl %dst, %off
    %s = addl %src, %off
    %v = loadl %s
    storel %v, %d
    jmp @backward

@done
    ret 0
}
