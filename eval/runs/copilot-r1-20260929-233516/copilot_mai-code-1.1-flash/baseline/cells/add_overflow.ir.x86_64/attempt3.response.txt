export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum = addl %a, %b
    storel %sum, %out
    %ax = xorl %a, %sum
    %bx = xorl %b, %sum
    %mask = andl %ax, %bx
    %sign = andl %mask, 0x8000000000000000
    %ovf = csltl %sign, 0
    jnz %ovf, @overflow
    ret 0
@overflow
    ret 1
}
