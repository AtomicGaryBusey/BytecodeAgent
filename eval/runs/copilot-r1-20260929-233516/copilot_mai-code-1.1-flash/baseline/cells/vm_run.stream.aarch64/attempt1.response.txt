vm_run:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    mov x19, x0
    mov x20, x2
    mov x21, x3
    mov x22, x1
    mov x23, #0
    mov x24, #0
    b .Lcheck

.Lloop:
    cmp x23, x22
    b.hs .Ldone
    ldr x9, [x19, x23, lsl #3]
    cmp x9, #0
    b.eq .Lhalt
    cmp x9, #1
    b.eq .LPUSH
    cmp x9, #2
    b.eq .LADD
    cmp x9, #3
    b.eq .LSUB
    cmp x9, #4
    b.eq .LMUL
    cmp x9, #5
    b.eq .LDUP
    cmp x9, #6
    b.eq .LSWAP
    cmp x9, #7
    b.eq .LDROP
    cmp x9, #8
    b.eq .LOVER
    cmp x9, #9
    b.eq .LLT
    cmp x9, #10
    b.eq .LJZ
    cmp x9, #11
    b.eq .LJMP
    add x23, x23, #1
    b .Lcheck

.Lhalt:
    b .Ldone

.LPUSH:
    add x10, x23, #1
    ldr x9, [x19, x10, lsl #3]
    add x10, x20, x24, lsl #3
    str x9, [x10]
    add x24, x24, #1
    add x23, x23, #2
    b .Lcheck

.LADD:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x10, x10, #1
    ldr x11, [x20, x10, lsl #3]
    add x9, x11, x9
    str x9, [x20, x10, lsl #3]
    sub x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LSUB:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x10, x10, #1
    ldr x11, [x20, x10, lsl #3]
    sub x9, x11, x9
    str x9, [x20, x10, lsl #3]
    sub x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LMUL:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x10, x10, #1
    ldr x11, [x20, x10, lsl #3]
    mul x9, x11, x9
    str x9, [x20, x10, lsl #3]
    sub x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LDUP:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    add x10, x20, x24, lsl #3
    str x9, [x10]
    add x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LSWAP:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x11, x10, #1
    ldr x12, [x20, x11, lsl #3]
    str x9, [x20, x11, lsl #3]
    str x12, [x20, x10, lsl #3]
    add x23, x23, #1
    b .Lcheck

.LDROP:
    sub x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LOVER:
    sub x10, x24, #2
    ldr x9, [x20, x10, lsl #3]
    add x10, x20, x24, lsl #3
    str x9, [x10]
    add x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LLT:
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x11, x10, #1
    ldr x12, [x20, x11, lsl #3]
    cmp x12, x9
    cset x9, lt
    str x9, [x20, x11, lsl #3]
    sub x24, x24, #1
    add x23, x23, #1
    b .Lcheck

.LJZ:
    add x10, x23, #1
    ldr x11, [x19, x10, lsl #3]
    sub x10, x24, #1
    ldr x9, [x20, x10, lsl #3]
    sub x24, x24, #1
    cbz x9, .LJZ_true
    add x23, x23, #2
    b .Lcheck
.LJZ_true:
    mov x23, x11
    b .Lcheck

.LJMP:
    add x10, x23, #1
    ldr x11, [x19, x10, lsl #3]
    mov x23, x11
    b .Lcheck

.Lcheck:
    cmp x23, x22
    b.hs .Ldone
    b .Lloop

.Ldone:
    cmp x24, x21
    b.hs .Ltop
    mov x10, x24
.Lzero:
    cmp x10, x21
    b.hs .Ltop
    str xzr, [x20, x10, lsl #3]
    add x10, x10, #1
    b .Lzero

.Ltop:
    cmp x24, #0
    b.eq .Lret_zero
    sub x10, x24, #1
    ldr x0, [x20, x10, lsl #3]
    b .Lexit
.Lret_zero:
    mov x0, xzr
.Lexit:
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
