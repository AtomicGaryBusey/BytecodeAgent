export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    %i =l add 0, 0
    %outpos =l add 0, 0
    jmp @outer

@outer
    %i_lt_n =w csltl %i, %n
    jnz %i_lt_n, @check_run
    ret %outpos

@check_run
    %src_i =l add %src, %i
    %b0 =w loadub %src_i
    %run =l add 1, 0

@run_loop
    %next =l add %i, %run
    %next_lt_n =w csltl %next, %n
    jnz %next_lt_n, @run_check_limit
    jmp @after_run

@run_check_limit
    %run_lt_128 =w csltl %run, 128
    jnz %run_lt_128, @run_check_same
    jmp @after_run

@run_check_same
    %ptr =l add %src, %next
    %b =w loadub %ptr
    %same =w ceql %b0, %b
    jnz %same, @run_inc
    jmp @after_run

@run_inc
    %run =l add %run, 1
    jmp @run_loop

@after_run
    %run_ge3 =w csltl 2, %run
    jnz %run_ge3, @repeat
    jmp @literal

@repeat
    %need =l add %outpos, 2
    %cap_lt_need =w csltl %cap, %need
    jnz %cap_lt_need, @fail
    %header =l sub 257, %run
    %dst =l add %out, %outpos
    storeb %header, %dst
    %outpos =l add %outpos, 1
    %dst =l add %out, %outpos
    storeb %b0, %dst
    %outpos =l add %outpos, 1
    %i =l add %i, %run
    jmp @outer

@literal
    %j =l add %i, 0

@literal_search
    %j_lt_n =w csltl %j, %n
    jnz %j_lt_n, @literal_j_ok
    jmp @literal_done

@literal_j_ok
    %j_minus_i =l sub %j, %i
    %j_minus_i_lt_128 =w csltl %j_minus_i, 128
    jnz %j_minus_i_lt_128, @literal_check_run
    jmp @literal_done

@literal_check_run
    %jsrc =l add %src, %j
    %bj =w loadub %jsrc
    %runj =l add 1, 0
    %k =l add %j, 1

@literal_run_loop
    %k_lt_n =w csltl %k, %n
    jnz %k_lt_n, @literal_run_check_limit
    jmp @literal_run_end

@literal_run_check_limit
    %runj_lt_128 =w csltl %runj, 128
    jnz %runj_lt_128, @literal_run_check_same
    jmp @literal_run_end

@literal_run_check_same
    %kptr =l add %src, %k
    %kv =w loadub %kptr
    %samej =w ceql %bj, %kv
    jnz %samej, @literal_run_inc
    jmp @literal_run_end

@literal_run_inc
    %runj =l add %runj, 1
    %k =l add %k, 1
    jmp @literal_run_loop

@literal_run_end
    %runj_ge3 =w csltl 2, %runj
    jnz %runj_ge3, @literal_done
    %j =l add %j, 1
    jmp @literal_search

@literal_done
    %litlen =l sub %j, %i
    %need =l add %outpos, %litlen
    %need =l add %need, 1
    %cap_lt_need =w csltl %cap, %need
    jnz %cap_lt_need, @fail
    %hdr =l sub %litlen, 1
    %dst =l add %out, %outpos
    storeb %hdr, %dst
    %outpos =l add %outpos, 1
    %k =l add %i, 0

@copy_literal
    %k_lt_j =w csltl %k, %j
    jnz %k_lt_j, @copy_literal_body
    jmp @after_copy_literal

@copy_literal_body
    %srcptr =l add %src, %k
    %byte =w loadub %srcptr
    %dst =l add %out, %outpos
    storeb %byte, %dst
    %outpos =l add %outpos, 1
    %k =l add %k, 1
    jmp @copy_literal

@after_copy_literal
    %i =l add %j, 0
    jmp @outer

@fail
    %negone =l sub 0, 1
    ret %negone
}
