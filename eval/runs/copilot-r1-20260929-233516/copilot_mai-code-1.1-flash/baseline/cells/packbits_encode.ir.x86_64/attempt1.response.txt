export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    %i =l add 0, 0
    %outpos =l add 0, 0
    jmp @outer

@outer
    %i_lt_n =w csltl %i, %n
    jz %i_lt_n, @done
    %src_ptr =l add %src, %i
    %b0 =w loadub %src_ptr
    %run =l add 1, 0
@run_loop
    %next =l add %i, %run
    %next_lt_n =w csltl %next, %n
    jz %next_lt_n, @after_run
    %run_lt_128 =w csltl %run, 128
    jz %run_lt_128, @after_run
    %ptr =l add %src, %next
    %b =w loadub %ptr
    %same =w ceql %b0, %b
    jz %same, @after_run
    %run =l add %run, 1
    jmp @run_loop
@after_run
    %r3 =l sub %run, 3
    %rlt0 =w csltl %r3, 0
    %run_ge3 =w xor %rlt0, 1
    jnz %run_ge3, @repeat
    jmp @literal

@repeat
    %packet_end =l add %outpos, 2
    %cap_lt_packet =w csltl %cap, %packet_end
    jnz %cap_lt_packet, @fail
    %header =l sub 257, %run
    %dst =l add %out, %outpos
    storeb %header, %dst
    %outpos =l add %outpos, 1
    %dst =l add %out, %outpos
    storeb %b0, %dst
    %outpos =l add %outpos, 1
    %i =l add %i, %run
    jmp @outer

@literal
    %j =l add %i, 0
@literal_search
    %j_lt_n =w csltl %j, %n
    jz %j_lt_n, @literal_done
    %j_minus_i =l sub %j, %i
    %j_lt_128 =w csltl %j_minus_i, 128
    jz %j_lt_128, @literal_done
    %jsrc =l add %src, %j
    %bj =w loadub %jsrc
    %runj =l add 1, 0
    %k =l add %j, 1
@literal_run
    %k_lt_n =w csltl %k, %n
    jz %k_lt_n, @literal_run_end
    %runj_lt_128 =w csltl %runj, 128
    jz %runj_lt_128, @literal_run_end
    %kptr =l add %src, %k
    %kv =w loadub %kptr
    %samej =w ceql %bj, %kv
    jz %samej, @literal_run_end
    %runj =l add %runj, 1
    %k =l add %k, 1
    jmp @literal_run
@literal_run_end
    %r3j =l sub %runj, 3
    %rlt0j =w csltl %r3j, 0
    %runj_ge3 =w xor %rlt0j, 1
    jnz %runj_ge3, @literal_done
    %j =l add %j, 1
    jmp @literal_search
@literal_done
    %litlen =l sub %j, %i
    %packet_end =l add %outpos, %litlen
    %packet_end =l add %packet_end, 1
    %cap_lt_packet =w csltl %cap, %packet_end
    jnz %cap_lt_packet, @fail
    %hdr =l sub %litlen, 1
    %dst =l add %out, %outpos
    storeb %hdr, %dst
    %outpos =l add %outpos, 1
    %k =l add %i, 0
@copy_literal
    %k_lt_j =w csltl %k, %j
    jz %k_lt_j, @after_copy_literal
    %srcptr =l add %src, %k
    %byte =w loadub %srcptr
    %dst =l add %out, %outpos
    storeb %byte, %dst
    %outpos =l add %outpos, 1
    %k =l add %k, 1
    jmp @copy_literal
@after_copy_literal
    %i =l add %j, 0
    jmp @outer

@fail
    %negone =l sub 0, 1
    ret %negone

@done
    ret %outpos
}
