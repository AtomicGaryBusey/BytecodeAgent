export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
  %v0 =l loadl %a
  %y0 =l call %f(l %v0)
  %sum =l copy %y0
  %min =l copy %y0
  %max =l copy %y0
  %odd =l copy 0
  %mask0 =l and %y0, 1
  %isodd0 =w ceql %mask0, 1
  jnz %isodd0, @odd0, @startloop
@odd0
  %odd =l add %odd, 1
@startloop
  %i =l copy 1
  %p =l add %a, 8
  jmp @loop
@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %x =l loadl %p
  %y =l call %f(l %x)
  %sum =l add %sum, %y
  %lt =w csltl %y, %min
  jnz %lt, @newmin, @checkmax
@newmin
  %min =l copy %y
  jmp @checkmax
@checkmax
  %gt =w csgtl %y, %max
  jnz %gt, @newmax, @aftermax
@newmax
  %max =l copy %y
@aftermax
  %mask =l and %y, 1
  %isodd =w ceql %mask, 1
  jnz %isodd, @oddinc, @afterodd
@oddinc
  %odd =l add %odd, 1
  jmp @afterodd
@afterodd
  %i =l add %i, 1
  %p =l add %p, 8
  jmp @loop
@done
  %out0 =l add %out, 0
  storel %min, %out0
  %out1 =l add %out, 8
  storel %max, %out1
  %out2 =l add %out, 16
  storel %sum, %out2
  ret %odd
}
