isqrt:
    cld
    mov r10, rdi
    xor rsi, rsi
    mov rcx, 0xffffffff
    xor r9, r9
.Lloop:
    mov r8, rsi
    add r8, rcx
    shr r8, 1
    mov rax, r8
    mul r8
    cmp rax, r10
    jbe .Lok
    mov rcx, r8
    sub rcx, 1
    cmp rsi, rcx
    jbe .Lloop
    mov rax, r9
    ret
.Lok:
    mov r9, r8
    lea rsi, [r8 + 1]
    cmp rsi, rcx
    jbe .Lloop
    mov rax, r9
    ret
