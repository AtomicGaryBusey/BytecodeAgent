export function l $crc32(l %p, l %n) {
@entry
    %crc =w copy 4294967295
    %i =l copy 0
    %zero =w copy 0
    %one =w copy 1
    %poly =w copy 3988292384
    %allones =w copy 4294967295
@loop
    %done =w ceql %i, %n
    jnz %done, @finalize
    %bytep =l add %p, %i
    %b =w loadub %bytep
    %crc =w xor %crc, %b
    %bits =w copy 8
@bitloop
    %bitdone =w ceql %bits, %zero
    jnz %bitdone, @afterbits
    %mask =w and %crc, %one
    jnz %mask, @odd
    %crc =w shr %crc, 1
    jmp @nextbit
@odd
    %tmp =w shr %crc, 1
    %crc =w xor %tmp, %poly
@nextbit
    %bits =w sub %bits, %one
    jmp @bitloop
@afterbits
    %i =l add %i, 1
    jmp @loop
@finalize
    %crc =w xor %crc, %allones
    %res =l extuw %crc
    ret %res
}
