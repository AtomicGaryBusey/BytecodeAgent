export function l $crc32(l %p, l %n) {
@entry
    %crc =w copy 4294967295
    %i =l copy 0
@loop
    %done =w ceql %i, %n
    jnz %done, @final
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %bits =w copy 8
@bitloop
    %bitdone =w ceql %bits, 0
    jnz %bitdone, @afterbits
    %mask =w and %crc, 1
    jnz %mask, @odd
    %crc =w shr %crc, 1
    jmp @nextbit
@odd
    %tmp =w shr %crc, 1
    %crc =w xor %tmp, 3988292384
@nextbit
    %bits =w sub %bits, 1
    jmp @bitloop
@afterbits
    %i =l add %i, 1
    jmp @loop
@final
    %crc =w xor %crc, 4294967295
    %res =l extuw %crc
    ret %res
}
