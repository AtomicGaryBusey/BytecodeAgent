export function l $bmax(l %a, l %n) {
@start
  %mx = loadl %a
  %i = copy 1
  jmp @loop

@loop
  %c = csltl %i, %n
  jnz %c, @body
  ret %mx

@body
  %off = mull %i, 8
  %p = addl %a, %off
  %v = loadl %p
  %lt = csltl %mx, %v
  jnz %lt, @update
  jmp @next

@update
  %mx = copy %v
  jmp @next

@next
  %i = addl %i, 1
  jmp @loop
}
