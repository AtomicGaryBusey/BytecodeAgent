export function l $bmax(l %a, l %n) {
@start
  %mx = loadl %a
  %zero = subl %n, %n
  %i = addl %zero, 1
  jmp @loop

@loop
  %cond = csltl %i, %n
  jnz %cond, @body
  ret %mx

@body
  %off = mull %i, 8
  %p = addl %a, %off
  %v = loadl %p
  %lt = csltl %mx, %v
  jnz %lt, @update
  jmp @next

@update
  %mx = addl %v, 0
  jmp @next

@next
  %i = addl %i, 1
  jmp @loop
}
