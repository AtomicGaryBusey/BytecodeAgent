export function l $bmax(l %a, l %n) {
@start
  %mx = loadl %a
  %i = copy 1
  jmp @loop

@loop
  %cond = csltl %i, %n
  jnz %cond, @body
  jmp @done

@body
  %off = mul %i, 8
  %p = add %a, %off
  %v = loadl %p
  %lt = csltl %mx, %v
  jnz %lt, @update
  jmp @next

@update
  %mx = copy %v
  jmp @next

@next
  %i = add %i, 1
  jmp @loop

@done
  ret %mx
}
