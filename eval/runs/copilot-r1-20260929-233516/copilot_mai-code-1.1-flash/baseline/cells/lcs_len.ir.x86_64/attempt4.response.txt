export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
  %zero =l add 0, 0
  %one =l add 1, 0
  %eight =l add 8, 0
  %j =l add 0, 0
@init_loop
  %j_gt_n =w csltl %n, %j
  jnz %j_gt_n, @init_done
  %off =l mul %j, %eight
  %ptr =l add %row, %off
  storel %zero, %ptr
  %j =l add %j, %one
  jmp @init_loop
@init_done
  %i =l add 0, 0
@outer
  %i_lt_m =w csltl %i, %m
  jnz %i_lt_m, @outer_body
  jmp @done
@outer_body
  %xptr =l add %x, %i
  %xb =w loadub %xptr
  %diag =l add 0, 0
  %j =l add 1, 0
@inner
  %j_gt_n =w csltl %n, %j
  jnz %j_gt_n, @outer_inc
  %up_off =l mul %j, %eight
  %up_ptr =l add %row, %up_off
  %up =l loadl %up_ptr
  %prev_j =l sub %j, %one
  %prev_off =l mul %prev_j, %eight
  %prev_ptr =l add %row, %prev_off
  %prev =l loadl %prev_ptr
  %yp =l add %y, %prev_j
  %yb =w loadub %yp
  %eq =w ceqw %xb, %yb
  jnz %eq, @eq_case
  %prev_lt_up =w csltl %prev, %up
  jnz %prev_lt_up, @pick_up
  %new =l add %prev, 0
  jmp @store
@pick_up
  %new =l add %up, 0
  jmp @store
@eq_case
  %new =l add %diag, %one
@store
  storel %new, %up_ptr
  %diag =l add %up, 0
  %j =l add %j, %one
  jmp @inner
@outer_inc
  %i =l add %i, %one
  jmp @outer
@done
  %res_off =l mul %n, %eight
  %res_ptr =l add %row, %res_off
  %res =l loadl %res_ptr
  ret %res
}
