export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
  %zero =l const 0
  %one =l const 1
  %eight =l const 8
  %jz =l const 0
@zero_loop
  %off_z =l mul %jz, %eight
  %addr_z =l add %row, %off_z
  storel %zero, %addr_z
  %j_eq_n_z =w ceql %jz, %n
  jnz %j_eq_n_z, @zero_done
  %jz =l add %jz, %one
  jmp @zero_loop
@zero_done
  %ii =l const 0
@outer_cond
  %i_eq_m =w ceql %ii, %m
  jnz %i_eq_m, @done
  %n_eq_0 =w ceql %n, %zero
  jnz %n_eq_0, @skip_inner
  %xptr =l add %x, %ii
  %xb =w loadub %xptr
  %diag =l const 0
  %jj =l const 1
@inner_loop
  %joff =l mul %jj, %eight
  %rowj =l add %row, %joff
  %upv =l loadl %rowj
  %prev_idx =l sub %jj, %one
  %prev_off =l mul %prev_idx, %eight
  %prev_addr =l add %row, %prev_off
  %prevv =l loadl %prev_addr
  %yptr =l add %y, %prev_idx
  %yb =w loadub %yptr
  %eqxy =w ceqw %xb, %yb
  jnz %eqxy, @eq_case
  %ltcmp =w csltl %upv, %prevv
  jnz %ltcmp, @choose_prev
  %newv =l add %upv, %zero
  jmp @store_it
@choose_prev
  %newv =l add %prevv, %zero
  jmp @store_it
@eq_case
  %newv =l add %diag, %one
@store_it
  storel %newv, %rowj
  %diag =l add %upv, %zero
  %j_eq_n =w ceql %jj, %n
  jnz %j_eq_n, @skip_inner
  %jj =l add %jj, %one
  jmp @inner_loop
@skip_inner
  %ii =l add %ii, %one
  jmp @outer_cond
@done
  %n_off =l mul %n, %eight
  %res_addr =l add %row, %n_off
  %res =l loadl %res_addr
  ret %res
}
