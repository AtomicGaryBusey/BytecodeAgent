export function l $map_sum(l %a, l %n, l %f) {
@start
  %sum =l add 0, 0
  %i =l add 0, 0
  %p =l add %a, 0
  jmp @loop

@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @end

@body
  %x =l loadl %p
  %v =l call %f(l %x)
  %sum =l add %sum, %v
  %i =l add %i, 1
  %p =l add %p, 8
  jmp @loop

@end
  ret %sum
}
