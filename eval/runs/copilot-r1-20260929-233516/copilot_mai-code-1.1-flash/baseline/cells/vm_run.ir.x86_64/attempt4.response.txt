export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
  %i = copy 0
@zero_loop
  %done_zero = ceql %i, %cap
  jnz %done_zero, @zero_done
  %off_zero = mull %i, 8
  %ptr_zero = addl %stack, %off_zero
  storel 0, %ptr_zero
  %i = addl %i, 1
  jmp @zero_loop
@zero_done
  %sp = copy 0
  %pc = copy 0
@loop
  %done = ceql %pc, %len
  jnz %done, @finish
  %op_off = mull %pc, 8
  %op_ptr = addl %code, %op_off
  %op = loadl %op_ptr

  %is0 = ceql %op, 0
  jnz %is0, @op_halt
  %is1 = ceql %op, 1
  jnz %is1, @op_push
  %is2 = ceql %op, 2
  jnz %is2, @op_add
  %is3 = ceql %op, 3
  jnz %is3, @op_sub
  %is4 = ceql %op, 4
  jnz %is4, @op_mul
  %is5 = ceql %op, 5
  jnz %is5, @op_dup
  %is6 = ceql %op, 6
  jnz %is6, @op_swap
  %is7 = ceql %op, 7
  jnz %is7, @op_drop
  %is8 = ceql %op, 8
  jnz %is8, @op_over
  %is9 = ceql %op, 9
  jnz %is9, @op_lt
  %is10 = ceql %op, 10
  jnz %is10, @op_jz
  %is11 = ceql %op, 11
  jnz %is11, @op_jmp

  %pc = addl %pc, 1
  jmp @loop

@op_halt
  jmp @finish

@op_push
  %pc1 = addl %pc, 1
  %arg_off = mull %pc1, 8
  %arg_ptr = addl %code, %arg_off
  %val = loadl %arg_ptr
  %slot_off = mull %sp, 8
  %slot_ptr = addl %stack, %slot_off
  storel %val, %slot_ptr
  %sp = addl %sp, 1
  %pc = addl %pc, 2
  jmp @loop

@op_add
  %spm1 = subl %sp, 1
  %spm2 = subl %sp, 2
  %off_b = mull %spm1, 8
  %ptr_b = addl %stack, %off_b
  %b = loadl %ptr_b
  %off_a = mull %spm2, 8
  %ptr_a = addl %stack, %off_a
  %a = loadl %ptr_a
  %sum = addl %a, %b
  storel %sum, %ptr_a
  storel 0, %ptr_b
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_sub
  %spm1s = subl %sp, 1
  %spm2s = subl %sp, 2
  %off_b2 = mull %spm1s, 8
  %ptr_b2 = addl %stack, %off_b2
  %b2 = loadl %ptr_b2
  %off_a2 = mull %spm2s, 8
  %ptr_a2 = addl %stack, %off_a2
  %a2 = loadl %ptr_a2
  %diff = subl %a2, %b2
  storel %diff, %ptr_a2
  storel 0, %ptr_b2
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_mul
  %spm1m = subl %sp, 1
  %spm2m = subl %sp, 2
  %off_b3 = mull %spm1m, 8
  %ptr_b3 = addl %stack, %off_b3
  %b3 = loadl %ptr_b3
  %off_a3 = mull %spm2m, 8
  %ptr_a3 = addl %stack, %off_a3
  %a3 = loadl %ptr_a3
  %prod = mull %a3, %b3
  storel %prod, %ptr_a3
  storel 0, %ptr_b3
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_dup
  %top_index = subl %sp, 1
  %top_off = mull %top_index, 8
  %top_ptr = addl %stack, %top_off
  %top_val = loadl %top_ptr
  %slot_off2 = mull %sp, 8
  %slot_ptr2 = addl %stack, %slot_off2
  storel %top_val, %slot_ptr2
  %sp = addl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_swap
  %spm1x = subl %sp, 1
  %spm2x = subl %sp, 2
  %off_x = mull %spm2x, 8
  %ptr_x = addl %stack, %off_x
  %a4 = loadl %ptr_x
  %off_y = mull %spm1x, 8
  %ptr_y = addl %stack, %off_y
  %b4 = loadl %ptr_y
  storel %b4, %ptr_x
  storel %a4, %ptr_y
  %pc = addl %pc, 1
  jmp @loop

@op_drop
  %spm1d = subl %sp, 1
  %off_drop = mull %spm1d, 8
  %ptr_drop = addl %stack, %off_drop
  storel 0, %ptr_drop
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_over
  %over_index = subl %sp, 2
  %over_off = mull %over_index, 8
  %over_ptr = addl %stack, %over_off
  %ov = loadl %over_ptr
  %slot_off3 = mull %sp, 8
  %slot_ptr3 = addl %stack, %slot_off3
  storel %ov, %slot_ptr3
  %sp = addl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_lt
  %ltm1 = subl %sp, 1
  %ltm2 = subl %sp, 2
  %off_b4 = mull %ltm1, 8
  %ptr_b4 = addl %stack, %off_b4
  %b5 = loadl %ptr_b4
  %off_a4 = mull %ltm2, 8
  %ptr_a4 = addl %stack, %off_a4
  %a5 = loadl %ptr_a4
  %ltv = csltl %a5, %b5
  %ltw = ceql %ltv, 1
  jnz %ltw, @lt_true
  storel 0, %ptr_a4
  jmp @lt_done
@lt_true
  storel 1, %ptr_a4
@lt_done
  storel 0, %ptr_b4
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop

@op_jz
  %jz_top = subl %sp, 1
  %jz_off = mull %jz_top, 8
  %jz_ptr = addl %stack, %jz_off
  %jzv = loadl %jz_ptr
  storel 0, %jz_ptr
  %sp = subl %sp, 1

  %jzcmp = ceql %jzv, 0
  jnz %jzcmp, @op_jz_take
  %pc = addl %pc, 2
  jmp @loop
@op_jz_take
  %jz_targ_pc = addl %pc, 1
  %jz_targ_off = mull %jz_targ_pc, 8
  %jz_targ_ptr = addl %code, %jz_targ_off
  %jz_targ = loadl %jz_targ_ptr
  %pc = copy %jz_targ
  jmp @loop

@op_jmp
  %jmp_targ_pc = addl %pc, 1
  %jmp_targ_off = mull %jmp_targ_pc, 8
  %jmp_targ_ptr = addl %code, %jmp_targ_off
  %jmp_targ = loadl %jmp_targ_ptr
  %pc = copy %jmp_targ
  jmp @loop

@finish
  %empty = ceql %sp, 0
  jnz %empty, @return_zero
  %last_index = subl %sp, 1
  %last_off = mull %last_index, 8
  %last_ptr = addl %stack, %last_off
  %res = loadl %last_ptr
  jmp @clear_tail
@return_zero
  %res = copy 0
@clear_tail
  %clear_i = copy %sp
@clear_loop
  %clear_done = ceql %clear_i, %cap
  jnz %clear_done, @return
  %clear_off = mull %clear_i, 8
  %clear_ptr = addl %stack, %clear_off
  storel 0, %clear_ptr
  %clear_i = addl %clear_i, 1
  jmp @clear_loop
@return
  ret %res
}
