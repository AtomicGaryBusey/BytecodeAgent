export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
  %sp = addl 0, 0
  %i = addl 0, 0
  jmp @zero_loop
@zero_loop
  %done_zero = ceql %i, %cap
  jnz %done_zero, @zero_done
  %off_zero = mull %i, 8
  %ptr_zero = addl %stack, %off_zero
  storel 0, %ptr_zero
  %i = addl %i, 1
  jmp @zero_loop
@zero_done
  %pc = addl 0, 0
  jmp @loop
@loop
  %end_loop = ceql %pc, %len
  jnz %end_loop, @finish
  %off_pc = mull %pc, 8
  %iptr = addl %code, %off_pc
  %op = loadl %iptr
  %is0 = ceql %op, 0
  jnz %is0, @op_halt
  %is1 = ceql %op, 1
  jnz %is1, @op_push
  %is2 = ceql %op, 2
  jnz %is2, @op_add
  %is3 = ceql %op, 3
  jnz %is3, @op_sub
  %is4 = ceql %op, 4
  jnz %is4, @op_mul
  %is5 = ceql %op, 5
  jnz %is5, @op_dup
  %is6 = ceql %op, 6
  jnz %is6, @op_swap
  %is7 = ceql %op, 7
  jnz %is7, @op_drop
  %is8 = ceql %op, 8
  jnz %is8, @op_over
  %is9 = ceql %op, 9
  jnz %is9, @op_lt
  %is10 = ceql %op, 10
  jnz %is10, @op_jz
  %is11 = ceql %op, 11
  jnz %is11, @op_jmp
  jmp @op_default
@op_halt
  jmp @finish
@op_push
  %pc1 = addl %pc, 1
  %off_arg = mull %pc1, 8
  %arg_ptr = addl %code, %off_arg
  %val = loadl %arg_ptr
  %slot_off = mull %sp, 8
  %slot_ptr = addl %stack, %slot_off
  storel %val, %slot_ptr
  %sp = addl %sp, 1
  %pc = addl %pc, 2
  jmp @loop
@op_add
  %spm1 = subl %sp, 1
  %off_b = mull %spm1, 8
  %ptr_b = addl %stack, %off_b
  %b = loadl %ptr_b
  %spm2 = subl %sp, 2
  %off_a = mull %spm2, 8
  %ptr_a = addl %stack, %off_a
  %a = loadl %ptr_a
  %sum = addl %a, %b
  storel %sum, %ptr_a
  storel 0, %ptr_b
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_sub
  %spm1b = subl %sp, 1
  %off_b2 = mull %spm1b, 8
  %ptr_b2 = addl %stack, %off_b2
  %b2 = loadl %ptr_b2
  %spm2b = subl %sp, 2
  %off_a2 = mull %spm2b, 8
  %ptr_a2 = addl %stack, %off_a2
  %a2 = loadl %ptr_a2
  %diff = subl %a2, %b2
  storel %diff, %ptr_a2
  storel 0, %ptr_b2
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_mul
  %spm1c = subl %sp, 1
  %off_b3 = mull %spm1c, 8
  %ptr_b3 = addl %stack, %off_b3
  %b3 = loadl %ptr_b3
  %spm2c = subl %sp, 2
  %off_a3 = mull %spm2c, 8
  %ptr_a3 = addl %stack, %off_a3
  %a3 = loadl %ptr_a3
  %prod = mull %a3, %b3
  storel %prod, %ptr_a3
  storel 0, %ptr_b3
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_dup
  %topi = subl %sp, 1
  %topoff = mull %topi, 8
  %topptr = addl %stack, %topoff
  %topv = loadl %topptr
  %slot_off2 = mull %sp, 8
  %slot_ptr2 = addl %stack, %slot_off2
  storel %topv, %slot_ptr2
  %sp = addl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_swap
  %spm1d = subl %sp, 1
  %spm2d = subl %sp, 2
  %off_x = mull %spm2d, 8
  %ptr_x = addl %stack, %off_x
  %a4 = loadl %ptr_x
  %off_y = mull %spm1d, 8
  %ptr_y = addl %stack, %off_y
  %b4 = loadl %ptr_y
  storel %b4, %ptr_x
  storel %a4, %ptr_y
  %pc = addl %pc, 1
  jmp @loop
@op_drop
  %spm1e = subl %sp, 1
  %off_drop = mull %spm1e, 8
  %ptr_drop = addl %stack, %off_drop
  storel 0, %ptr_drop
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_over
  %overi = subl %sp, 2
  %overoff = mull %overi, 8
  %overptr = addl %stack, %overoff
  %ov = loadl %overptr
  %slot_off3 = mull %sp, 8
  %slot_ptr3 = addl %stack, %slot_off3
  storel %ov, %slot_ptr3
  %sp = addl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_lt
  %ltm1 = subl %sp, 1
  %ltm2 = subl %sp, 2
  %off_b4 = mull %ltm1, 8
  %ptr_b4 = addl %stack, %off_b4
  %b5 = loadl %ptr_b4
  %off_a4 = mull %ltm2, 8
  %ptr_a4 = addl %stack, %off_a4
  %a5 = loadl %ptr_a4
  %ltv = csltl %a5, %b5
  jnz %ltv, @lt_true
  storel 0, %ptr_a4
  jmp @lt_done
@lt_true
  storel 1, %ptr_a4
@lt_done
  storel 0, %ptr_b4
  %sp = subl %sp, 1
  %pc = addl %pc, 1
  jmp @loop
@op_jz
  %jz1 = subl %sp, 1
  %jzoff = mull %jz1, 8
  %jzptr = addl %stack, %jzoff
  %jzv = loadl %jzptr
  storel 0, %jzptr
  %sp = %jz1
  %jzcmp = ceql %jzv, 0
  jnz %jzcmp, @op_jz_take
  %pc = addl %pc, 2
  jmp @loop
@op_jz_take
  %jz_targ_pc = addl %pc, 1
  %jz_targ_off = mull %jz_targ_pc, 8
  %jz_targ_ptr = addl %code, %jz_targ_off
  %jz_targ = loadl %jz_targ_ptr
  %pc = %jz_targ
  jmp @loop
@op_jmp
  %jmp_targ_pc = addl %pc, 1
  %jmp_targ_off = mull %jmp_targ_pc, 8
  %jmp_targ_ptr = addl %code, %jmp_targ_off
  %jmp_targ = loadl %jmp_targ_ptr
  %pc = %jmp_targ
  jmp @loop
@op_default
  %pc = addl %pc, 1
  jmp @loop
@finish
  %empty = ceql %sp, 0
  jnz %empty, @return_zero
  %topidx = subl %sp, 1
  %topoff = mull %topidx, 8
  %topptr = addl %stack, %topoff
  %res = loadl %topptr
  ret l %res
@return_zero
  ret l 0
}
