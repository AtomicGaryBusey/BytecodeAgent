export function l $xxh64(l %p, l %n, l %seed) {
@entry
    %P1 =l copy -7046029288634856825
    %P2 =l copy -4417276706812531889
    %P3 =l copy 1609587929392839161
    %P4 =l copy -8796714831421723037
    %P5 =l copy 2870177450012600261

    %n_ge_32 =w cugtl %n, 31
    jnz %n_ge_32, @big, @small

@big
    %v1_init0 =l add %seed, %P1
    %v1_init =l add %v1_init0, %P2
    %v2_init =l add %seed, %P2
    %v3_init =l copy %seed
    %v4_init =l sub %seed, %P1
    %q_big =l copy %p
    %rem_big =l copy %n
    jmp @stripe_check

@stripe_check
    %v1_cur =l phi @big %v1_init, @stripe_body %v1_after
    %v2_cur =l phi @big %v2_init, @stripe_body %v2_after
    %v3_cur =l phi @big %v3_init, @stripe_body %v3_after
    %v4_cur =l phi @big %v4_init, @stripe_body %v4_after
    %q_stripe =l phi @big %q_big, @stripe_body %q_after_stripe
    %rem_stripe =l phi @big %rem_big, @stripe_body %rem_after_stripe
    %stripe_cond =w cugtl %rem_stripe, 31
    jnz %stripe_cond, @stripe_body, @after_stripes

@stripe_body
    %x1 =l loadl %q_stripe
    %x1m =l mul %x1, %P2
    %s1 =l add %v1_cur, %x1m
    %r1a =l shl %s1, 31
    %r1b =l shr %s1, 33
    %r1rot =l or %r1a, %r1b
    %v1n =l mul %r1rot, %P1
    %v1_after =l copy %v1n

    %x2p =l add %q_stripe, 8
    %x2 =l loadl %x2p
    %x2m =l mul %x2, %P2
    %s2 =l add %v2_cur, %x2m
    %r2a =l shl %s2, 31
    %r2b =l shr %s2, 33
    %r2rot =l or %r2a, %r2b
    %v2n =l mul %r2rot, %P1
    %v2_after =l copy %v2n

    %x3p =l add %q_stripe, 16
    %x3 =l loadl %x3p
    %x3m =l mul %x3, %P2
    %s3 =l add %v3_cur, %x3m
    %r3a =l shl %s3, 31
    %r3b =l shr %s3, 33
    %r3rot =l or %r3a, %r3b
    %v3n =l mul %r3rot, %P1
    %v3_after =l copy %v3n

    %x4p =l add %q_stripe, 24
    %x4 =l loadl %x4p
    %x4m =l mul %x4, %P2
    %s4 =l add %v4_cur, %x4m
    %r4a =l shl %s4, 31
    %r4b =l shr %s4, 33
    %r4rot =l or %r4a, %r4b
    %v4n =l mul %r4rot, %P1
    %v4_after =l copy %v4n

    %q_after_stripe =l add %q_stripe, 32
    %rem_after_stripe =l sub %rem_stripe, 32
    jmp @stripe_check

@after_stripes
    %v1_end =l phi @stripe_check %v1_cur
    %v2_end =l phi @stripe_check %v2_cur
    %v3_end =l phi @stripe_check %v3_cur
    %v4_end =l phi @stripe_check %v4_cur
    %q_after_stripes =l phi @stripe_check %q_stripe
    %rem_after_stripes =l phi @stripe_check %rem_stripe

    %hrot1a =l shl %v1_end, 1
    %hrot1b =l shr %v1_end, 63
    %hrot1 =l or %hrot1a, %hrot1b

    %hrot2a =l shl %v2_end, 7
    %hrot2b =l shr %v2_end, 57
    %hrot2 =l or %hrot2a, %hrot2b

    %hrot3a =l shl %v3_end, 12
    %hrot3b =l shr %v3_end, 52
    %hrot3 =l or %hrot3a, %hrot3b

    %hrot4a =l shl %v4_end, 18
    %hrot4b =l shr %v4_end, 46
    %hrot4 =l or %hrot4a, %hrot4b

    %hbase =l add %hrot1, %hrot2
    %hbase2 =l add %hbase, %hrot3
    %hbase3 =l add %hbase2, %hrot4

    %m1tmp =l mul %v1_end, %P2
    %m1r1a =l shl %m1tmp, 31
    %m1r1b =l shr %m1tmp, 33
    %m1r1 =l or %m1r1a, %m1r1b
    %m1r =l mul %m1r1, %P1
    %m1x =l xor %hbase3, %m1r
    %m1y =l mul %m1x, %P1
    %h1 =l add %m1y, %P4

    %m2tmp =l mul %v2_end, %P2
    %m2r1a =l shl %m2tmp, 31
    %m2r1b =l shr %m2tmp, 33
    %m2r1 =l or %m2r1a, %m2r1b
    %m2r =l mul %m2r1, %P1
    %m2x =l xor %h1, %m2r
    %m2y =l mul %m2x, %P1
    %h2 =l add %m2y, %P4

    %m3tmp =l mul %v3_end, %P2
    %m3r1a =l shl %m3tmp, 31
    %m3r1b =l shr %m3tmp, 33
    %m3r1 =l or %m3r1a, %m3r1b
    %m3r =l mul %m3r1, %P1
    %m3x =l xor %h2, %m3r
    %m3y =l mul %m3x, %P1
    %h3 =l add %m3y, %P4

    %m4tmp =l mul %v4_end, %P2
    %m4r1a =l shl %m4tmp, 31
    %m4r1b =l shr %m4tmp, 33
    %m4r1 =l or %m4r1a, %m4r1b
    %m4r =l mul %m4r1, %P1
    %m4x =l xor %h3, %m4r
    %m4y =l mul %m4x, %P1
    %h_after_stripes =l add %m4y, %P4
    jmp @common_after_hash

@small
    %h_small =l add %seed, %P5
    jmp @common_after_hash

@common_after_hash
    %h_input =l phi @small %h_small, @after_stripes %h_after_stripes
    %q_common =l phi @small %p, @after_stripes %q_after_stripes
    %rem_common =l phi @small %n, @after_stripes %rem_after_stripes
    %h_add_n =l add %h_input, %n
    %h_cur =l copy %h_add_n

@rem8_check
    %q8 =l phi @common_after_hash %q_common, @rem8_body %q8_after
    %rem8 =l phi @common_after_hash %rem_common, @rem8_body %rem8_after
    %h8 =l phi @common_after_hash %h_cur, @rem8_body %h8_after
    %rem8_cond =w cugtl %rem8, 7
    jnz %rem8_cond, @rem8_body, @rem4_check

@rem8_body
    %r8v =l loadl %q8
    %r8m =l mul %r8v, %P2
    %r8ra =l shl %r8m, 31
    %r8rb =l shr %r8m, 33
    %r8rot =l or %r8ra, %r8rb
    %r8 =l mul %r8rot, %P1
    %h8x =l xor %h8, %r8
    %h8r1a =l shl %h8x, 27
    %h8r1b =l shr %h8x, 37
    %h8rot =l or %h8r1a, %h8r1b
    %h8m =l mul %h8rot, %P1
    %h8p =l add %h8m, %P4
    %q8_after =l add %q8, 8
    %rem8_after =l sub %rem8, 8
    %h8_after =l copy %h8p
    jmp @rem8_check

@rem4_check
    %q4 =l phi @rem8_check %q8, @rem4_body %q4_after
    %rem4 =l phi @rem8_check %rem8, @rem4_body %rem4_after
    %h4cur =l phi @rem8_check %h8, @rem4_body %h4_after
    %rem4_cond =w cugtl %rem4, 3
    jnz %rem4_cond, @rem4_body, @rem_byte_check

@rem4_body
    %r4v =l loaduw %q4
    %r4m =l mul %r4v, %P1
    %h4x =l xor %h4cur, %r4m
    %h4r1a =l shl %h4x, 23
    %h4r1b =l shr %h4x, 41
    %h4rot =l or %h4r1a, %h4r1b
    %h4m =l mul %h4rot, %P2
    %h4p =l add %h4m, %P3
    %q4_after =l add %q4, 4
    %rem4_after =l sub %rem4, 4
    %h4_after =l copy %h4p
    jmp @rem4_check

@rem_byte_check
    %qbyte =l phi @rem4_check %q4, @rem_byte_body %qbyte_after
    %rembyte =l phi @rem4_check %rem4, @rem_byte_body %rembyte_after
    %hbyte =l phi @rem4_check %h4cur, @rem_byte_body %hbyte_after
    %byte_cond =w cugtl %rembyte, 0
    jnz %byte_cond, @rem_byte_body, @final_mix

@rem_byte_body
    %cb =l loadub %qbyte
    %cbm =l mul %cb, %P5
    %hbx =l xor %hbyte, %cbm
    %hbr1a =l shl %hbx, 11
    %hbr1b =l shr %hbx, 53
    %hbrot =l or %hbr1a, %hbr1b
    %hbm =l mul %hbrot, %P1
    %qbyte_after =l add %qbyte, 1
    %rembyte_after =l sub %rembyte, 1
    %hbyte_after =l copy %hbm
    jmp @rem_byte_check

@final_mix
    %hfinal_m =l phi @rem_byte_check %hbyte
    %h1x =l shr %hfinal_m, 33
    %h1a =l xor %hfinal_m, %h1x
    %h1b =l mul %h1a, %P2
    %h2x =l shr %h1b, 29
    %h2a =l xor %h1b, %h2x
    %h2b =l mul %h2a, %P3
    %h3x =l shr %h2b, 32
    %h3r =l xor %h2b, %h3x
    ret %h3r
}
