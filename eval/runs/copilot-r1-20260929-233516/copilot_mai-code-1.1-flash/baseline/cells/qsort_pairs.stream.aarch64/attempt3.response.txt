qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    b.le .Ldone

    sub x10, x20, #1
    lsl x11, x10, #4
    add x12, x19, x11
    ldr x13, [x12]
    ldr x14, [x12, #8]
    mov x15, #0
    mov x16, #0

.Lloop:
    cmp x16, x20
    b.ge .Lpartition
    cmp x16, x10
    b.ge .Lpartition
    add x17, x19, x16, lsl #4
    ldr x8, [x17]
    ldr x9, [x17, #8]
    cmp x8, x13
    b.lt .Lless
    b.gt .Lnext
    cmp x9, x14
    b.lt .Lless
    b.ge .Lnext

.Lless:
    cmp x15, x16
    beq .Lskip_swap
    add x3, x19, x15, lsl #4
    ldr x4, [x3]
    ldr x5, [x3, #8]
    str x8, [x3]
    str x9, [x3, #8]
    str x4, [x17]
    str x5, [x17, #8]

.Lskip_swap:
    add x15, x15, #1

.Lnext:
    add x16, x16, #1
    b .Lloop

.Lpartition:
    add x17, x19, x15, lsl #4
    ldr x8, [x17]
    ldr x9, [x17, #8]
    str x13, [x17]
    str x14, [x17, #8]
    str x8, [x12]
    str x9, [x12, #8]

    cmp x15, #0
    ble .Lskip_left
    sub sp, sp, #16
    str x15, [sp]
    stp x29, x30, [sp, #-16]!
    mov x0, x19
    mov x1, x15
    bl qsort_pairs
    ldp x29, x30, [sp], #16
    ldr x15, [sp], #16

.Lskip_left:
    sub x10, x20, x15
    sub x10, x10, #1
    cmp x10, #0
    ble .Ldone
    add x0, x19, x15, lsl #4
    add x0, x0, #16
    mov x1, x10
    stp x29, x30, [sp, #-16]!
    bl qsort_pairs
    ldp x29, x30, [sp], #16

.Ldone:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
