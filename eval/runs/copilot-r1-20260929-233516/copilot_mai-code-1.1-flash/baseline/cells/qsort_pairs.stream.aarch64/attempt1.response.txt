qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    b.le .Ldone

    sub x9, x20, #1
    lsl x9, x9, #1
    add x10, x19, x9, lsl #3
    ldr x11, [x10]
    ldr x12, [x10, #8]
    mov x13, #0
    mov x14, #0
.Lloop:
    sub x15, x20, #1
    cmp x14, x15
    b.ge .Lpartdone
    add x0, x19, x14, lsl #4
    ldr x15, [x0]
    ldr x1, [x0, #8]
    cmp x15, x11
    b.lt .Lswap
    b.gt .Lnext
    cmp x1, x12
    b.lt .Lswap
    b.ge .Lnext
.Lswap:
    add x0, x19, x13, lsl #4
    ldr x2, [x0]
    ldr x3, [x0, #8]
    add x4, x19, x14, lsl #4
    ldr x5, [x4]
    ldr x6, [x4, #8]
    str x5, [x0]
    str x6, [x0, #8]
    str x2, [x4]
    str x3, [x4, #8]
    add x13, x13, #1
.Lnext:
    add x14, x14, #1
    b .Lloop
.Lpartdone:
    sub x9, x20, #1
    lsl x9, x9, #4
    add x10, x19, x9
    add x0, x19, x13, lsl #4
    ldr x2, [x0]
    ldr x3, [x0, #8]
    ldr x5, [x10]
    ldr x6, [x10, #8]
    str x5, [x0]
    str x6, [x0, #8]
    str x2, [x10]
    str x3, [x10, #8]

    str x13, [sp, #-16]!
    cmp x13, #0
    b.le .Lskip_left
    mov x0, x19
    mov x1, x13
    bl qsort_pairs
.Lskip_left:
    ldr x13, [sp], #16

    sub x14, x20, x13
    sub x14, x14, #1
    cmp x14, #0
    ble .Ldone

    add x15, x13, #1
    add x0, x19, x15, lsl #4
    mov x1, x14
    bl qsort_pairs

.Ldone:
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
