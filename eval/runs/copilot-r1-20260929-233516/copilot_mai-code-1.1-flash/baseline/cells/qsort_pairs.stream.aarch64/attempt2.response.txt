qsort_pairs:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    cmp x20, #1
    b.le .Ldone

    sub x21, x20, #1
    lsl x22, x21, #4
    add x9, x19, x22
    ldr x10, [x9]
    ldr x11, [x9, #8]
    mov x12, #0
    mov x13, #0

.Lloop:
    cmp x13, x21
    b.ge .Lpart
    add x14, x19, x13, lsl #4
    ldr x15, [x14]
    ldr x16, [x14, #8]
    cmp x15, x10
    b.lt .Lswap
    b.gt .Lnext
    cmp x16, x11
    b.ge .Lnext

.Lswap:
    add x17, x19, x12, lsl #4
    ldr x2, [x17]
    ldr x3, [x17, #8]
    str x15, [x17]
    str x16, [x17, #8]
    str x2, [x14]
    str x3, [x14, #8]
    add x12, x12, #1

.Lnext:
    add x13, x13, #1
    b .Lloop

.Lpart:
    add x14, x19, x12, lsl #4
    add x15, x19, x22
    ldr x2, [x14]
    ldr x3, [x14, #8]
    ldr x16, [x15]
    ldr x17, [x15, #8]
    str x16, [x14]
    str x17, [x14, #8]
    str x2, [x15]
    str x3, [x15, #8]

    cmp x12, #0
    ble .Lskip_left
    mov x0, x19
    mov x1, x12
    bl qsort_pairs

.Lskip_left:
    sub x21, x20, x12
    sub x21, x21, #1
    cmp x21, #0
    ble .Ldone
    add x23, x12, #1
    lsl x23, x23, #4
    add x0, x19, x23
    mov x1, x21
    bl qsort_pairs

.Ldone:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
