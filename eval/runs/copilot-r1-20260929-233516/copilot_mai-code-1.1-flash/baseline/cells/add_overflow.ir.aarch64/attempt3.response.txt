export function l $add_overflow(l %a, l %b, l %out) {
@start
  %sum = add %a, %b
  storel %sum, %out
  %a_neg = csltl %a, 0
  %b_neg = csltl %b, 0
  %same_sign = ceqw %a_neg, %b_neg
  %sum_neg = csltl %sum, 0
  %sign_diff = xorw %a_neg, %sum_neg
  %ovf = andw %same_sign, %sign_diff
  %ret = castl %ovf
  ret %ret
}
