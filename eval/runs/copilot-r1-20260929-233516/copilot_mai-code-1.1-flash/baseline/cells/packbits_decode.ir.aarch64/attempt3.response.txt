export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    %i =l copy 0
    %o =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @packet
    ret %o

@packet
    %p =l add %src, %i
    %h =l loadsb %p
    %neg =w csltl %h, 0
    jnz %neg, @neg

    %len =l add %h, 1
    %need_lit =l add %i, 1
    %need_lit =l add %need_lit, %len
    %nshort_lit =w csltl %n, %need_lit
    jnz %nshort_lit, @err2

    %total_lit =l add %o, %len
    %oflow_lit =w csltl %cap, %total_lit
    jnz %oflow_lit, @lit_overflow

    %dptr_lit =l add %out, %o
    %sptr_lit =l add %src, %i
    %datap_lit =l add %sptr_lit, 1
    %k_lit =l copy 0
@lit_copy_loop
    %klt_lit =w csltl %k_lit, %len
    jnz %klt_lit, @lit_copy_body
    %i =l add %i, 1
    %i =l add %i, %len
    %o =l add %o, %len
    jmp @loop

@lit_copy_body
    %spos_lit =l add %datap_lit, %k_lit
    %byte_lit =w loadub %spos_lit
    %dpos_lit =l add %dptr_lit, %k_lit
    storeb %byte_lit, %dpos_lit
    %k_lit =l add %k_lit, 1
    jmp @lit_copy_loop

@neg
    %eqm128 =w ceql %h, -128
    jnz %eqm128, @noop

    %count_rep =l sub 1, %h
    %need_rep =l add %i, 2
    %nshort_rep =w csltl %n, %need_rep
    jnz %nshort_rep, @err2

    %vptr_rep =l add %src, %i
    %vptr_rep =l add %vptr_rep, 1
    %val_rep =w loadub %vptr_rep

    %total_rep =l add %o, %count_rep
    %oflow_rep =w csltl %cap, %total_rep
    jnz %oflow_rep, @rep_overflow

    %dptr_rep =l add %out, %o
    %k_rep =l copy 0
@rep_copy_loop
    %klt_rep =w csltl %k_rep, %count_rep
    jnz %klt_rep, @rep_copy_body
    %i =l add %i, 2
    %o =l add %o, %count_rep
    jmp @loop

@rep_copy_body
    %dpos_rep =l add %dptr_rep, %k_rep
    storeb %val_rep, %dpos_rep
    %k_rep =l add %k_rep, 1
    jmp @rep_copy_loop

@noop
    %i =l add %i, 1
    jmp @loop

@err2
    %ret_err2 =l copy -2
    ret %ret_err2

@lit_overflow
    %rem_lit =l sub %cap, %o
    %copylen_lit =l copy %rem_lit
    %dptr_lit_ov =l add %out, %o
    %sptr_lit_ov =l add %src, %i
    %datap_lit_ov =l add %sptr_lit_ov, 1
    %k_lit_ov =l copy 0
@lit_ov_loop
    %klt_lit_ov =w csltl %k_lit_ov, %copylen_lit
    jnz %klt_lit_ov, @lit_ov_body
    %ret_lit_ov =l copy -1
    ret %ret_lit_ov

@lit_ov_body
    %spos_lit_ov =l add %datap_lit_ov, %k_lit_ov
    %byte_lit_ov =w loadub %spos_lit_ov
    %dpos_lit_ov =l add %dptr_lit_ov, %k_lit_ov
    storeb %byte_lit_ov, %dpos_lit_ov
    %k_lit_ov =l add %k_lit_ov, 1
    jmp @lit_ov_loop

@rep_overflow
    %rem_rep =l sub %cap, %o
    %copylen_rep =l copy %rem_rep
    %dptr_rep_ov =l add %out, %o
    %k_rep_ov =l copy 0
@rep_ov_loop
    %klt_rep_ov =w csltl %k_rep_ov, %copylen_rep
    jnz %klt_rep_ov, @rep_ov_body
    %ret_rep_ov =l copy -1
    ret %ret_rep_ov

@rep_ov_body
    %dpos_rep_ov =l add %dptr_rep_ov, %k_rep_ov
    storeb %val_rep, %dpos_rep_ov
    %k_rep_ov =l add %k_rep_ov, 1
    jmp @rep_ov_loop
}
