export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    %i =l copy 0
    %o =l copy 0
    jmp @loop

@loop
    %i_lt_n =w csltl %i, %n
    jnz %i_lt_n, @header
    %ret_ok =l copy %o
    ret %ret_ok

@header
    %p =l add %src, %i
    %h =l loadsb %p
    %h_neg =w csltl %h, 0
    jnz %h_neg, @neg_branch

    %lit_len =l add %h, 1
    %need_lit_base =l add %i, 1
    %need_lit =l add %need_lit_base, %lit_len
    %n_short_lit =w csltl %n, %need_lit
    jnz %n_short_lit, @err2

    %total_lit =l add %o, %lit_len
    %oflow_lit =w csltl %cap, %total_lit
    jnz %oflow_lit, @lit_overflow

    %dptr_lit =l add %out, %o
    %src_lit =l add %src, %i
    %data_lit =l add %src_lit, 1
    %k_lit =l copy 0
@lit_copy_loop
    %k_lit_lt_len =w csltl %k_lit, %lit_len
    jnz %k_lit_lt_len, @lit_copy_body
    %i =l add %i, 1
    %i =l add %i, %lit_len
    %o =l add %o, %lit_len
    jmp @loop

@lit_copy_body
    %spos_lit =l add %data_lit, %k_lit
    %byte_lit =w loadub %spos_lit
    %dpos_lit =l add %dptr_lit, %k_lit
    storeb %byte_lit, %dpos_lit
    %k_lit =l add %k_lit, 1
    jmp @lit_copy_loop

@neg_branch
    %eq_neg128 =w ceql %h, -128
    jnz %eq_neg128, @noop

    %count_rep =l sub 1, %h
    %need_rep =l add %i, 2
    %n_short_rep =w csltl %n, %need_rep
    jnz %n_short_rep, @err2

    %src_rep =l add %src, %i
    %rep_val_ptr =l add %src_rep, 1
    %val_rep =w loadub %rep_val_ptr

    %total_rep =l add %o, %count_rep
    %oflow_rep =w csltl %cap, %total_rep
    jnz %oflow_rep, @rep_overflow

    %dptr_rep =l add %out, %o
    %k_rep =l copy 0
@rep_copy_loop
    %k_rep_lt_count =w csltl %k_rep, %count_rep
    jnz %k_rep_lt_count, @rep_copy_body
    %i =l add %i, 2
    %o =l add %o, %count_rep
    jmp @loop

@rep_copy_body
    %dpos_rep =l add %dptr_rep, %k_rep
    storeb %val_rep, %dpos_rep
    %k_rep =l add %k_rep, 1
    jmp @rep_copy_loop

@noop
    %i =l add %i, 1
    jmp @loop

@err2
    %ret_err2 =l copy -2
    ret %ret_err2

@lit_overflow
    %rem_lit =l sub %cap, %o
    %copy_lit =l copy %rem_lit
    %dptr_lit_ov =l add %out, %o
    %src_lit_ov =l add %src, %i
    %data_lit_ov =l add %src_lit_ov, 1
    %k_lit_ov =l copy 0
@lit_ov_loop
    %k_lit_ov_lt_copy =w csltl %k_lit_ov, %copy_lit
    jnz %k_lit_ov_lt_copy, @lit_ov_body
    %ret_lit_ov =l copy -1
    ret %ret_lit_ov

@lit_ov_body
    %spos_lit_ov =l add %data_lit_ov, %k_lit_ov
    %byte_lit_ov =w loadub %spos_lit_ov
    %dpos_lit_ov =l add %dptr_lit_ov, %k_lit_ov
    storeb %byte_lit_ov, %dpos_lit_ov
    %k_lit_ov =l add %k_lit_ov, 1
    jmp @lit_ov_loop

@rep_overflow
    %rem_rep =l sub %cap, %o
    %copy_rep =l copy %rem_rep
    %dptr_rep_ov =l add %out, %o
    %k_rep_ov =l copy 0
@rep_ov_loop
    %k_rep_ov_lt_copy =w csltl %k_rep_ov, %copy_rep
    jnz %k_rep_ov_lt_copy, @rep_ov_body
    %ret_rep_ov =l copy -1
    ret %ret_rep_ov

@rep_ov_body
    %dpos_rep_ov =l add %dptr_rep_ov, %k_rep_ov
    storeb %val_rep, %dpos_rep_ov
    %k_rep_ov =l add %k_rep_ov, 1
    jmp @rep_ov_loop
}
