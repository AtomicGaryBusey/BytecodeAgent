export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    %i =l copy 0
    %o =l copy 0
    jmp @loop
@loop
    %cond =w csltl %i, %n
    jnz %cond, @packet
    ret %o
@packet
    %p =l add %src, %i
    %h =l loadsb %p
    %neg =w csltl %h, 0
    jnz %neg, @neg
    %len =l add %h, 1
    %need =l add %i, 1
    %need =l add %need, %len
    %nshort =w csltl %n, %need
    jnz %nshort, @err2
    %total =l add %o, %len
    %oflow =w csltl %cap, %total
    jnz %oflow, @lit_overflow
    %dptr =l add %out, %o
    %sptr =l add %src, %i
    %datap =l add %sptr, 1
    %k =l copy 0
@lit_copy_loop
    %klt =w csltl %k, %len
    jnz %klt, @lit_copy_body
    %i =l add %i, 1
    %i =l add %i, %len
    %o =l add %o, %len
    jmp @loop
@lit_copy_body
    %spos =l add %datap, %k
    %byte =w loadub %spos
    %dpos =l add %dptr, %k
    storeb %byte, %dpos
    %k =l add %k, 1
    jmp @lit_copy_loop
@neg
    %eqm128 =w ceql %h, -128
    jnz %eqm128, @noop
    %count =l sub 1, %h
    %need =l add %i, 2
    %nshort =w csltl %n, %need
    jnz %nshort, @err2
    %vptr =l add %src, %i
    %vptr =l add %vptr, 1
    %val =w loadub %vptr
    %total =l add %o, %count
    %oflow =w csltl %cap, %total
    jnz %oflow, @rep_overflow
    %dptr =l add %out, %o
    %k =l copy 0
@rep_copy_loop
    %klt =w csltl %k, %count
    jnz %klt, @rep_copy_body
    %i =l add %i, 2
    %o =l add %o, %count
    jmp @loop
@rep_copy_body
    %dpos =l add %dptr, %k
    storeb %val, %dpos
    %k =l add %k, 1
    jmp @rep_copy_loop
@noop
    %i =l add %i, 1
    jmp @loop
@err2
    %ret =l copy -2
    ret %ret
@lit_overflow
    %rem =l sub %cap, %o
    %copylen =l copy %rem
    %dptr =l add %out, %o
    %sptr =l add %src, %i
    %datap =l add %sptr, 1
    %k =l copy 0
@lit_ov_loop
    %klt =w csltl %k, %copylen
    jnz %klt, @lit_ov_body
    %ret =l copy -1
    ret %ret
@lit_ov_body
    %spos =l add %datap, %k
    %byte =w loadub %spos
    %dpos =l add %dptr, %k
    storeb %byte, %dpos
    %k =l add %k, 1
    jmp @lit_ov_loop
@rep_overflow
    %rem =l sub %cap, %o
    %copylen =l copy %rem
    %dptr =l add %out, %o
    %k =l copy 0
@rep_ov_loop
    %klt =w csltl %k, %copylen
    jnz %klt, @rep_ov_body
    %ret =l copy -1
    ret %ret
@rep_ov_body
    %dpos =l add %dptr, %k
    storeb %val, %dpos
    %k =l add %k, 1
    jmp @rep_ov_loop
}
