export function l $binary_search(l %a, l %n, l %key) {
@entry
    %low0 = copy 0
    %high0 = copy %n
    jmp @check

@check
    %low = phi l [%low0 @entry, %low1 @lower, %low2 @higher]
    %high = phi l [%high0 @entry, %high1 @lower, %high2 @higher]
    %cond = csltl %low %high
    jnz %cond @loop
    %ret = copy -1
    ret %ret

@loop
    %diff = subl %high %low
    %half = sdivl %diff 2
    %mid = addl %low %half
    %off = shll %mid 3
    %ptr = addl %a %off
    %val = loadl %ptr
    %eq = ceql %val %key
    jnz %eq @found
    %lt = csltl %val %key
    jnz %lt @lower
    jmp @higher

@lower
    %low1 = addl %mid 1
    %high1 = copy %high
    jmp @check

@higher
    %low2 = copy %low
    %high2 = copy %mid
    jmp @check

@found
    ret %mid
}
