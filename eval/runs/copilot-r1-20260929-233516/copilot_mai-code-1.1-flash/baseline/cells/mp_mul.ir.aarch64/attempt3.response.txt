export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %na0 =w ceql %na, 0
    jnz %na0, @retzero, @checknb
@checknb
    %nb0 =w ceql %nb, 0
    jnz %nb0, @retzero, @zsetup
@zsetup
    %len =l add %na, %nb
    %i =l add 0, 0
@zloop
    %zdone =w ceql %i, %len
    jnz %zdone, @afterzero, @zbody
@zbody
    %i8 =l shl %i, 3
    %ptr =l add %r, %i8
    storel %ptr, 0
    %i =l add %i, 1
    jmp @zloop
@afterzero
    %i =l add 0, 0
@outerloop
    %i_lt =w csltl %i, %na
    jnz %i_lt, @outerbody, @scan
@outerbody
    %j =l add 0, 0
@inner
    %j_lt =w csltl %j, %nb
    jnz %j_lt, @innerbody, @outernext
@innerbody
    %aidx =l add %i, 0
    %a8 =l shl %aidx, 3
    %aptr =l add %a, %a8
    %x =l loadl %aptr

    %bidx =l add %j, 0
    %b8 =l shl %bidx, 3
    %bptr =l add %b, %b8
    %y =l loadl %bptr

    %x0 =l and %x, 4294967295
    %x1 =l shr %x, 32
    %y0 =l and %y, 4294967295
    %y1 =l shr %y, 32

    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1

    %v =l add %p01, %p10
    %vlow =l and %v, 4294967295
    %vhi =l shr %v, 32
    %mask =l shl %vlow, 32
    %prodlo =l add %p00, %mask
    %carry1w =w cultl %prodlo, %p00
    %carry1 =l add %carry1w, 0

    %prodhi =l add %p11, %vhi
    %prodhi =l add %prodhi, %carry1

    %k =l add %i, %j
    %k8 =l shl %k, 3
    %rptr0 =l add %r, %k8
    %r0 =l loadl %rptr0
    %sum0 =l add %r0, %prodlo
    %c0 =w cultl %sum0, %r0
    storel %rptr0, %sum0
    %carry =w add %c0, 0

    %pos =l add %k, 1
    %pos8 =l shl %pos, 3
    %rptr1 =l add %r, %pos8
    %r1 =l loadl %rptr1
    %t0 =l add %r1, %prodhi
    %c1 =w cultl %t0, %r1
    %carryl =l add %carry, 0
    %t1 =l add %t0, %carryl
    %c2 =w cultl %t1, %t0
    %carry =w or %c1, %c2
    storel %rptr1, %t1
    %pos =l add %pos, 1

@carryloop
    %pos_lt =w csltl %pos, %len
    jnz %pos_lt, @carrycheck, @carrydone
@carrycheck
    %carry_zero =w ceql %carry, 0
    jnz %carry_zero, @carrydone, @carryprop
@carryprop
    %pos8b =l shl %pos, 3
    %ptr =l add %r, %pos8b
    %vcur =l loadl %ptr
    %carryl =l add %carry, 0
    %sum =l add %vcur, %carryl
    %cc =w cultl %sum, %vcur
    storel %ptr, %sum
    %carry =w add %cc, 0
    %pos =l add %pos, 1
    jmp @carryloop
@carrydone

    %j =l add %j, 1
    jmp @inner
@outernext
    %i =l add %i, 1
    jmp @outerloop
@scan
    %k =l add %len, 0
@scanloop
    %done =w ceql %k, 0
    jnz %done, @scanzero, @scanbody
@scanbody
    %k1 =l sub %k, 1
    %k1_8 =l shl %k1, 3
    %scanp =l add %r, %k1_8
    %v =l loadl %scanp
    %vz =w ceql %v, 0
    jnz %vz, @scannext, @scanret
@scannext
    %k =l add %k1, 0
    jmp @scanloop
@scanret
    %res =l add %k1, 1
    ret %res
@scanzero
    ret 0
@retzero
    ret 0
}
