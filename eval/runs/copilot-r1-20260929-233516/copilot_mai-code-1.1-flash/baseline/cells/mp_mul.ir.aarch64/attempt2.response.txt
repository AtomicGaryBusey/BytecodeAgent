export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %na0 =w ceql %na, 0
    jnz %na0, @retzero, @checknb
@checknb
    %nb0 =w ceql %nb, 0
    jnz %nb0, @retzero, @zsetup
@zsetup
    %len =l add %na, %nb
    %i =l add 0, 0
@zloop
    %zdone =w ceql %i, %len
    jnz %zdone, @afterzero, @zbody
@zbody
    %i8 =l shl %i, 3
    %ptr =l add %r, %i8
    storel %ptr, 0
    %i =l add %i, 1
    jmp @zloop
@afterzero
    %i =l add 0, 0
@outerloop
    %icond =w csltl %i, %na
    jnz %icond, @outerbody, @scan
@outerbody
    %i8 =l shl %i, 3
    %aptr =l add %a, %i8
    %x =l loadl %aptr
    %j =l add 0, 0
@inner
    %jcond =w csltl %j, %nb
    jnz %jcond, @innerbody, @outernext
@innerbody
    %carry =w and 0, 0
    %j8 =l shl %j, 3
    %bptr =l add %b, %j8
    %y =l loadl %bptr
    %k =l add %i, %j
    %k8 =l shl %k, 3
    %x0 =l and %x, 4294967295
    %x1 =l shr %x, 32
    %y0 =l and %y, 4294967295
    %y1 =l shr %y, 32
    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1
    %p00l =l and %p00, 4294967295
    %p00h =l shr %p00, 32
    %p01l =l and %p01, 4294967295
    %p10l =l and %p10, 4294967295
    %sum1 =l add %p00h, %p01l
    %sum1b =l add %sum1, %p10l
    %bit1 =l and %sum1b, 4294967295
    %carry1 =l shr %sum1b, 32
    %p01h =l shr %p01, 32
    %p10h =l shr %p10, 32
    %p11l =l and %p11, 4294967295
    %sum2 =l add %p01h, %p10h
    %sum2b =l add %sum2, %p11l
    %sum2c =l add %sum2b, %carry1
    %bit2 =l and %sum2c, 4294967295
    %carry2 =l shr %sum2c, 32
    %p11h =l shr %p11, 32
    %bit3 =l add %p11h, %carry2
    %bit1s =l shl %bit1, 32
    %prodlo =l add %p00l, %bit1s
    %bit3s =l shl %bit3, 32
    %prodhi =l add %bit2, %bit3s
    %rptr0 =l add %r, %k8
    %r0 =l loadl %rptr0
    %s0 =l add %r0, %prodlo
    %c0 =w cultl %s0, %r0
    %s1 =l add %s0, %carry
    %c1 =w cultl %s1, %s0
    %carry =w or %c0, %c1
    storel %rptr0, %s1
    %rptr1 =l add %rptr0, 8
    %r1 =l loadl %rptr1
    %t0 =l add %r1, %prodhi
    %c2 =w cultl %t0, %r1
    %t1 =l add %t0, %carry
    %c3 =w cultl %t1, %t0
    %carry =w or %c2, %c3
    storel %rptr1, %t1
    %j =l add %j, 1
    jmp @inner
@outernext
    %i =l add %i, 1
    jmp @outerloop
@scan
    %k =l add %len, 0
@scanloop
    %done =w ceql %k, 0
    jnz %done, @scanzero, @scanbody
@scanbody
    %k1 =l sub %k, 1
    %scan8 =l shl %k1, 3
    %scanp =l add %r, %scan8
    %v =l loadl %scanp
    %vz =w ceql %v, 0
    jnz %vz, @scannext, @scanret
@scannext
    %k =l add %k1, 0
    jmp @scanloop
@scanret
    %res =l add %k1, 1
    ret %res
@scanzero
    ret 0
@retzero
    ret 0
}
