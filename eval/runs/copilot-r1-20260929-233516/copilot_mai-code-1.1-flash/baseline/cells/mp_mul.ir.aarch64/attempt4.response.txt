export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %na0 =w ceql %na, 0
    jnz %na0, @retzero, @checknb
@checknb
    %nb0 =w ceql %nb, 0
    jnz %nb0, @retzero, @zinit
@zinit
    %len =l add %na, %nb
    %i =l add 0, 0
@zloop
    %zd =w ceql %i, %len
    jnz %zd, @mulstart, @zbody
@zbody
    %off =l shl %i, 3
    %p =l add %r, %off
    storel %p, 0
    %i =l add %i, 1
    jmp @zloop
@mulstart
    %i =l add 0, 0
@outer
    %i_lt =w csltl %i, %na
    jnz %i_lt, @outerbody, @scan
@outerbody
    %j =l add 0, 0
@inner
    %j_lt =w csltl %j, %nb
    jnz %j_lt, @innerbody, @outernext
@innerbody
    %aoff =l shl %i, 3
    %ap =l add %a, %aoff
    %aval =l loadl %ap
    %boff =l shl %j, 3
    %bp =l add %b, %boff
    %bval =l loadl %bp

    %a0 =l and %aval, 4294967295
    %a1 =l shr %aval, 32
    %b0 =l and %bval, 4294967295
    %b1 =l shr %bval, 32

    %p00 =l mul %a0, %b0
    %p01 =l mul %a0, %b1
    %p10 =l mul %a1, %b0
    %p11 =l mul %a1, %b1

    %s12 =l add %p01, %p10
    %s12lo =l and %s12, 4294967295
    %s12hi =l shr %s12, 32
    %mask =l shl %s12lo, 32
    %prodlo =l add %p00, %mask
    %c0 =w cultl %prodlo, %p00

    %prodhi =l add %p11, %s12hi
    %c0z =w ceql %c0, 0
    jnz %c0z, @prodhi_ok, @prodhi_inc
@prodhi_inc
    %prodhi =l add %prodhi, 1
@prodhi_ok

    %k =l add %i, %j
    %k8 =l shl %k, 3
    %r0p =l add %r, %k8
    %r0 =l loadl %r0p
    %sum0 =l add %r0, %prodlo
    %c1 =w cultl %sum0, %r0
    storel %r0p, %sum0

    %carry =w add %c1, 0
    %pos =l add %k, 1
    %pos8 =l shl %pos, 3
    %r1p =l add %r, %pos8
    %r1 =l loadl %r1p
    %t0 =l add %r1, %prodhi
    %c2 =w cultl %t0, %r1

    %carry0 =w ceql %carry, 0
    jnz %carry0, @no_carry_add, @do_carry_add
@do_carry_add
    %t1 =l add %t0, 1
    %c3 =w cultl %t1, %t0
    %carry =w or %c2, %c3
    storel %r1p, %t1
    %pos =l add %k, 2
    jmp @carryloop
@no_carry_add
    %carry =w add %c2, 0
    storel %r1p, %t0
    %pos =l add %k, 2
@carryloop
    %pos_lt =w csltl %pos, %len
    jnz %pos_lt, @carrycheck, @carrydone
@carrycheck
    %carryz =w ceql %carry, 0
    jnz %carryz, @carrydone, @carryprop
@carryprop
    %pos8b =l shl %pos, 3
    %curp =l add %r, %pos8b
    %cur =l loadl %curp
    %nxt =l add %cur, 1
    %cc =w cultl %nxt, %cur
    storel %curp, %nxt
    %carry =w add %cc, 0
    %pos =l add %pos, 1
    jmp @carryloop
@carrydone

    %j =l add %j, 1
    jmp @inner
@outernext
    %i =l add %i, 1
    jmp @outer
@scan
    %k =l add %len, 0
@scanloop
    %k0 =w ceql %k, 0
    jnz %k0, @retzero, @scanbody
@scanbody
    %k1 =l sub %k, 1
    %k1_8 =l shl %k1, 3
    %scanp =l add %r, %k1_8
    %v =l loadl %scanp
    %vz =w ceql %v, 0
    jnz %vz, @scannext, @scanret
@scannext
    %k =l add %k1, 0
    jmp @scanloop
@scanret
    %res =l add %k1, 1
    ret %res
@retzero
    ret 0
}
