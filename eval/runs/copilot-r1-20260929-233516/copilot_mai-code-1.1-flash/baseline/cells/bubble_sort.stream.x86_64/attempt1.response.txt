bubble_sort:
    push r12
    push r13
    push r14
    push r15
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    cmp rsi, 1
    jle .Ldone
    mov r12, rsi
    xor r13d, r13d
.Louter:
    mov r14, r12
    dec r14
    cmp r13, r14
    jae .Ldone
    xor r15d, r15d
.Linner:
    mov rcx, r12
    sub rcx, r13
    dec rcx
    cmp r15, rcx
    jae .Lnext_outer
    lea r10, qword ptr [rdi + r15*8]
    lea r11, qword ptr [r10 + 8]
    mov rax, qword ptr [r10]
    mov rdx, qword ptr [r11]
    cmp rax, rdx
    jle .Lno_swap
    mov qword ptr [r10], rdx
    mov qword ptr [r11], rax
.Lno_swap:
    inc r15
    jmp .Linner
.Lnext_outer:
    inc r13
    jmp .Louter
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    ret
