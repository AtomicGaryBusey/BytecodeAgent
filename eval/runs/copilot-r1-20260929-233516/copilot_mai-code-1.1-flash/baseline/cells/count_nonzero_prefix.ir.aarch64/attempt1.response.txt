export function l $count_nonzero_prefix(l %a, l %n) {
@entry
  %i0 =l copy 0
  jmp @loop

@loop
  %i =l phi @entry %i0, @next %i1
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %idx =l shll %i, 3
  %ptr =l addl %a, %idx
  %val =l loadl %ptr
  %eqz =w ceql %val, 0
  jnz %eqz, @done, @next

@next
  %i1 =l addl %i, 1
  jmp @loop

@done
  ret %i
}
