export function l $count_nonzero_prefix(l %a, l %n) {
@entry
  %i =l copy 0
  %p =l copy %a
  jmp @loop

@loop
  %i2 =l phi @entry %i, @next %i3
  %p2 =l phi @entry %p, @next %p3
  %cond =w csltl %i2, %n
  jnz %cond, @body, @done

@body
  %v =l loadl %p2
  %eqz =w ceql %v, 0
  jnz %eqz, @done, @next

@next
  %i3 =l add %i2, 1
  %p3 =l add %p2, 8
  jmp @loop

@done
  ret %i2
}
