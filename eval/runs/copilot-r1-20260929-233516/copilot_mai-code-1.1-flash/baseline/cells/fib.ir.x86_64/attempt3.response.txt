export function l $fib(l %n) {
@entry
  %zero = copyl 0
  %one = copyl 1
  %iszero = ceql %n, %zero
  jnz %iszero, @ret0, @init

@init
  %a0 = copyl 0
  %b0 = copyl 1
  %i0 = copyl 0
  jmp @loop

@loop
  %a = phi l [ %a0, @init ], [ %a1, @body ]
  %b = phi l [ %b0, @init ], [ %b1, @body ]
  %i = phi l [ %i0, @init ], [ %i1, @body ]
  %cond = csltl %i, %n
  jnz %cond, @body, @done

@body
  %sum = addl %a, %b
  %a1 = copyl %b
  %b1 = copyl %sum
  %i1 = addl %i, %one
  jmp @loop

@done
  ret %a

@ret0
  ret %zero
}
