export function l $fib(l %n) {
@entry
  %zero = copy l 0
  %iszero = ceql l %n, %zero
  jnz %iszero, @ret0, @init

@init
  %a0 = copy l 0
  %b0 = copy l 1
  %i0 = copy l 0
  jmp @loop

@loop
  %a = phi l [ %a0, @init ], [ %a1, @body ]
  %b = phi l [ %b0, @init ], [ %b1, @body ]
  %i = phi l [ %i0, @init ], [ %i1, @body ]
  %cond = csltl l %i, %n
  jnz %cond, @body, @done

@body
  %sum = addl %a, %b
  %a1 = copy l %b
  %b1 = copy l %sum
  %i1 = addl %i, 1
  jmp @loop

@done
  ret l %a

@ret0
  ret l %zero
}
