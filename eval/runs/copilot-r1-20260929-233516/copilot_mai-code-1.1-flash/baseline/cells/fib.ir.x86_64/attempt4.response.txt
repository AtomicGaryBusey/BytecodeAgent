export function l $fib(l %n) {
@entry
  %zero = copy 0
  %one = copy 1
  %a0 = copy %zero
  %b0 = copy %one
  %i0 = copy %zero
  jmp @loop

@loop
  %a = phi l [ %a0, @entry ], [ %a1, @body ]
  %b = phi l [ %b0, @entry ], [ %b1, @body ]
  %i = phi l [ %i0, @entry ], [ %i1, @body ]
  %cond = csltl %i, %n
  jnz %cond, @body, @done

@body
  %sum = addl %a, %b
  %a1 = copy %b
  %b1 = copy %sum
  %i1 = addl %i, %one
  jmp @loop

@done
  ret %a
}
