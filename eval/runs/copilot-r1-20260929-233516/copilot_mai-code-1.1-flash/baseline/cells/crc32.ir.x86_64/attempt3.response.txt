export function l $crc32(l %p, l %n) {
@entry
    %crc =w xor 0, 4294967295
    %i =l xor 0, 0
@loop
    %done =w ceql %i, %n
    jnz %done, @end
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc =w xor %crc, %byte
    %k =w xor 0, 0
@bitloop
    %bit =w and %crc, 1
    %isodd =w ceql %bit, 1
    jnz %isodd, @odd
    %crc =w shr %crc, 1
    jmp @nextbit
@odd
    %shr =w shr %crc, 1
    %crc =w xor %shr, 3988292384
@nextbit
    %k =w add %k, 1
    %done8 =w ceql %k, 8
    jnz %done8, @afterbits
    jmp @bitloop
@afterbits
    %i =l add %i, 1
    jmp @loop
@end
    %final =w xor %crc, 4294967295
    %ret =l extuw %final
    ret %ret
}
