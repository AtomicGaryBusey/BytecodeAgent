export function l $crc32(l %p, l %n) {
@entry
    %crc =w const 0xFFFFFFFF
    %i =l const 0
@loop
    %done =w ceql %i, %n
    jnz %done, @end
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc =w xor %crc, %byte
    %k =w const 0
@bitloop
    %bit =w and %crc, 1
    %isodd =w ceql %bit, 1
    jnz %isodd, @odd
    %shr =w shr %crc, 1
    %crc =w add %shr, 0
    jmp @nextbit
@odd
    %shr =w shr %crc, 1
    %x =w xor %shr, 0xEDB88320
    %crc =w add %x, 0
@nextbit
    %k =w add %k, 1
    %done8 =w ceql %k, 8
    jnz %done8, @afterbits
    jmp @bitloop
@afterbits
    %i =l add %i, 1
    jmp @loop
@end
    %final =w xor %crc, 0xFFFFFFFF
    %ret =l extuw %final
    ret %ret
}
