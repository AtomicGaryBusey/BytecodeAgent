export function l $qsort_pairs(l %a, l %n) {
@entry
    %two =l copy 2
    %nlt2 =w csltl %n, %two
    jnz %nlt2, @done

    %n1 =l subl %n, 1
    %pivot_off =l mull %n1, 16
    %pivot_ptr =l addl %a, %pivot_off
    %pivot_key =l loadl %pivot_ptr
    %pivot_val_ptr =l addl %pivot_ptr, 8
    %pivot_val =l loadl %pivot_val_ptr

    %i =l copy 0
    %j =l copy 0
@loop
    %j_eq_n1 =w ceql %j, %n1
    jnz %j_eq_n1, @partition_done

    %j_off =l mull %j, 16
    %j_ptr =l addl %a, %j_off
    %j_key =l loadl %j_ptr
    %j_val_ptr =l addl %j_ptr, 8
    %j_val =l loadl %j_val_ptr

    %ltk =w csltl %j_key, %pivot_key
    %eqk =w ceql %j_key, %pivot_key
    %ltv =w csltl %j_val, %pivot_val
    %eqv_lt =w andw %eqk, %ltv
    %lt_pivot =w orw %ltk, %eqv_lt

    jnz %lt_pivot, @swap_left
@continue
    %j =l addl %j, 1
    jmp @loop

@swap_left
    %i_off =l mull %i, 16
    %i_ptr =l addl %a, %i_off

    %tmp_i_key =l loadl %i_ptr
    %tmp_i_val_ptr =l addl %i_ptr, 8
    %tmp_i_val =l loadl %tmp_i_val_ptr

    %tmp_j_key =l loadl %j_ptr
    %tmp_j_val_ptr =l addl %j_ptr, 8
    %tmp_j_val =l loadl %tmp_j_val_ptr

    storel %tmp_j_key, %i_ptr
    storel %tmp_j_val, %tmp_i_val_ptr
    storel %tmp_i_key, %j_ptr
    storel %tmp_i_val, %tmp_j_val_ptr

    %i =l addl %i, 1
    jmp @continue

@partition_done
    %i_off2 =l mull %i, 16
    %i_ptr2 =l addl %a, %i_off2

    %tmp_i_key2 =l loadl %i_ptr2
    %tmp_i_val_ptr2 =l addl %i_ptr2, 8
    %tmp_i_val2 =l loadl %tmp_i_val_ptr2

    %tmp_p_key =l loadl %pivot_ptr
    %tmp_p_val_ptr =l addl %pivot_ptr, 8
    %tmp_p_val =l loadl %tmp_p_val_ptr

    storel %tmp_p_key, %i_ptr2
    storel %tmp_p_val, %tmp_i_val_ptr2
    storel %tmp_i_key2, %pivot_ptr
    storel %tmp_i_val2, %tmp_p_val_ptr

    %left_n =l copy %i
    %right_base =l addl %i, 1
    %right_base_off =l mull %right_base, 16
    %right_ptr =l addl %a, %right_base_off

    %right_n =l subl %n, %i
    %right_n =l subl %right_n, 1

    %left_result =l call $qsort_pairs(l %a, l %left_n)
    %right_result =l call $qsort_pairs(l %right_ptr, l %right_n)

@done
    %ret0 =l copy 0
    ret %ret0
}
