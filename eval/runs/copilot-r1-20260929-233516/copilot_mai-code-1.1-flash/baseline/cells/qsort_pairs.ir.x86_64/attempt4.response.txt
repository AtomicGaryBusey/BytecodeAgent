export function l $qsort_pairs(l %a, l %n) {
@entry
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %sixteen =l copy 16

    %n_lt_two =w csltl %n, %two
    jnz %n_lt_two, @done

    %n_minus_one =l subl %n, %one
    %pivot_off =l mull %n_minus_one, %sixteen
    %pivot_ptr =l addl %a, %pivot_off
    %pivot_key =l loadl %pivot_ptr
    %pivot_val_ptr =l addl %pivot_ptr, 8
    %pivot_val =l loadl %pivot_val_ptr

    %i0 =l copy %zero
    %j0 =l copy %zero
    jmp @loop

@loop
    %i =l phi %i0, @entry %i_after_cmp, @cmp %i_after_swap, @swap
    %j =l phi %j0, @entry %j_after_cmp, @cmp %j_after_swap, @swap

    %j_lt_n_minus_one =w csltl %j, %n_minus_one
    jnz %j_lt_n_minus_one, @cmp
    jmp @partition

@cmp
    %j_off =l mull %j, %sixteen
    %j_ptr =l addl %a, %j_off
    %j_key =l loadl %j_ptr
    %j_val_ptr =l addl %j_ptr, 8
    %j_val =l loadl %j_val_ptr

    %lt_key =w csltl %j_key, %pivot_key
    %eq_key =w ceql %j_key, %pivot_key
    %lt_val =w csltl %j_val, %pivot_val
    %eq_and_lt =w andw %eq_key, %lt_val
    %less =w orw %lt_key, %eq_and_lt

    jnz %less, @swap

    %i_after_cmp =l copy %i
    %j_after_cmp =l addl %j, %one
    jmp @loop

@swap
    %i_off =l mull %i, %sixteen
    %i_ptr =l addl %a, %i_off
    %i_key =l loadl %i_ptr
    %i_val_ptr =l addl %i_ptr, 8
    %i_val =l loadl %i_val_ptr

    %j_key_2 =l loadl %j_ptr
    %j_val_ptr_2 =l addl %j_ptr, 8
    %j_val_2 =l loadl %j_val_ptr_2

    storel %j_key_2, %i_ptr
    storel %j_val_2, %i_val_ptr
    storel %i_key, %j_ptr
    storel %i_val, %j_val_ptr_2

    %i_after_swap =l addl %i, %one
    %j_after_swap =l addl %j, %one
    jmp @loop

@partition
    %i_off_part =l mull %i, %sixteen
    %i_ptr_part =l addl %a, %i_off_part
    %last_off =l mull %n_minus_one, %sixteen
    %last_ptr =l addl %a, %last_off

    %part_i_key =l loadl %i_ptr_part
    %part_i_val_ptr =l addl %i_ptr_part, 8
    %part_i_val =l loadl %part_i_val_ptr
    %part_last_key =l loadl %last_ptr
    %part_last_val_ptr =l addl %last_ptr, 8
    %part_last_val =l loadl %part_last_val_ptr

    storel %part_last_key, %i_ptr_part
    storel %part_last_val, %part_i_val_ptr
    storel %part_i_key, %last_ptr
    storel %part_i_val, %part_last_val_ptr

    %left_n =l copy %i
    %right_start =l addl %i, %one
    %right_off =l mull %right_start, %sixteen
    %right_ptr =l addl %a, %right_off
    %right_n =l subl %n, %right_start

    %left_ret =l call $qsort_pairs(l %a, l %left_n)
    %right_ret =l call $qsort_pairs(l %right_ptr, l %right_n)

@done
    %ret0 =l copy 0
    ret %ret0
}
