export function l $qsort_pairs(l %a, l %n) {
@entry
    %two =l copy 2
    %nlt2 =w csltl %n, %two
    jnz %nlt2, @done

    %n1 =l subl %n, 1
    %pivot_off =l mull %n1, 16
    %pivot_ptr =l addl %a, %pivot_off
    %pivot_key =l loadl %pivot_ptr
    %pivot_val_ptr =l addl %pivot_ptr, 8
    %pivot_val =l loadl %pivot_val_ptr

    %i =l copy 0
    %j =l copy 0

@loop
    %j_lt_n1 =w csltl %j, %n1
    jnz %j_lt_n1, @body
    jmp @partition

@body
    %j_off =l mull %j, 16
    %j_ptr =l addl %a, %j_off
    %j_key =l loadl %j_ptr
    %j_val_ptr =l addl %j_ptr, 8
    %j_val =l loadl %j_val_ptr

    %lt_key =w csltl %j_key, %pivot_key
    %eq_key =w ceql %j_key, %pivot_key
    %lt_val =w csltl %j_val, %pivot_val
    %eq_and_lt =w andw %eq_key, %lt_val
    %less =w orw %lt_key, %eq_and_lt

    jnz %less, @swap
@next
    %j_after =l addl %j, 1
    %j =l copy %j_after
    jmp @loop

@swap
    %i_off_swap =l mull %i, 16
    %i_ptr_swap =l addl %a, %i_off_swap

    %tmp_i_key_swap =l loadl %i_ptr_swap
    %tmp_i_val_ptr_swap =l addl %i_ptr_swap, 8
    %tmp_i_val_swap =l loadl %tmp_i_val_ptr_swap

    %tmp_j_key_swap =l loadl %j_ptr
    %tmp_j_val_ptr_swap =l addl %j_ptr, 8
    %tmp_j_val_swap =l loadl %tmp_j_val_ptr_swap

    storel %tmp_j_key_swap, %i_ptr_swap
    storel %tmp_j_val_swap, %tmp_i_val_ptr_swap
    storel %tmp_i_key_swap, %j_ptr
    storel %tmp_i_val_swap, %tmp_j_val_ptr_swap

    %i_after_swap =l addl %i, 1
    %i =l copy %i_after_swap
    jmp @next

@partition
    %i_off_part =l mull %i, 16
    %i_ptr_part =l addl %a, %i_off_part

    %tmp_i_key_part =l loadl %i_ptr_part
    %tmp_i_val_ptr_part =l addl %i_ptr_part, 8
    %tmp_i_val_part =l loadl %tmp_i_val_ptr_part

    %tmp_p_key_part =l loadl %pivot_ptr
    %tmp_p_val_ptr_part =l addl %pivot_ptr, 8
    %tmp_p_val_part =l loadl %tmp_p_val_ptr_part

    storel %tmp_p_key_part, %i_ptr_part
    storel %tmp_p_val_part, %tmp_i_val_ptr_part
    storel %tmp_i_key_part, %pivot_ptr
    storel %tmp_i_val_part, %tmp_p_val_ptr_part

    %left_n =l addl %i, 0
    %right_start =l addl %i, 1
    %right_off =l mull %right_start, 16
    %right_ptr =l addl %a, %right_off
    %right_n =l subl %n, %right_start

    %left_ret =l call $qsort_pairs(l %a, l %left_n)
    %right_ret =l call $qsort_pairs(l %right_ptr, l %right_n)

@done
    %ret0 =l copy 0
    ret %ret0
}
