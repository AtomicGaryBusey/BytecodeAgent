export function l $collatz_max(l %n) {
@entry:
  jmp @outer

@outer:
  %k =l phi @entry 1, @continue_outer %k_next
  %max =l phi @entry 0, @continue_outer %max_after
  %cond =w csltl %k, %n
  jnz %cond, @body_outer, @done

@body_outer:
  %x_init =l copy %k
  %steps_init =l copy 0
  jmp @inner

@inner:
  %x =l phi @body_outer %x_init, @inner_after %x_next
  %steps =l phi @body_outer %steps_init, @inner_after %steps_next
  %is_one =w ceql %x, 1
  jnz %is_one, @after_inner, @step

@step:
  %rem =l reml %x, 2
  %is_even =w ceql %rem, 0
  jnz %is_even, @even_case, @odd_case

@even_case:
  %x_even =l divl %x, 2
  %steps_even =l addl %steps, 1
  jmp @inner_after

@odd_case:
  %x2 =l addl %x, %x
  %x3 =l addl %x2, %x
  %x4 =l addl %x3, 1
  %steps_odd =l addl %steps, 1
  jmp @inner_after

@inner_after:
  %x_next =l phi @even_case %x_even, @odd_case %x4
  %steps_next =l phi @even_case %steps_even, @odd_case %steps_odd
  jmp @inner

@after_inner:
  %is_bigger =w csltl %max, %steps
  jnz %is_bigger, @update_max, @continue_outer

@update_max:
  %max_new =l copy %steps
  jmp @continue_outer

@continue_outer:
  %max_after =l phi @after_inner %max, @update_max %max_new
  %k_next =l addl %k, 1
  jmp @outer

@done:
  ret %max
}
