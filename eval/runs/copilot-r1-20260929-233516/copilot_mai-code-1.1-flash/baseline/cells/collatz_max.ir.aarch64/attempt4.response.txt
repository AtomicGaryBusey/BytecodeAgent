export function l $collatz_max(l %n) {
@entry:
  %k_init =l copy 1
  %max_init =l copy 0
  jmp @outer

@outer:
  %k =l phi @entry %k_init, @outer_continue %k_next
  %max =l phi @entry %max_init, @outer_continue %max_next
  %done =w csltl %k, %n
  jnz %done, @body, @exit

@body:
  %x =l copy %k
  %steps =l copy 0
  jmp @inner

@inner:
  %x_cur =l phi @body %x, @inner_after %x_next
  %steps_cur =l phi @body %steps, @inner_after %steps_next
  %is_one =w ceql %x_cur, 1
  jnz %is_one, @inner_done, @step

@step:
  %rem =l reml %x_cur, 2
  %is_even =w ceql %rem, 0
  jnz %is_even, @even_case, @odd_case

@even_case:
  %x_even =l divl %x_cur, 2
  %steps_even =l addl %steps_cur, 1
  jmp @inner_after

@odd_case:
  %x2 =l addl %x_cur, %x_cur
  %x3 =l addl %x2, %x_cur
  %x_odd =l addl %x3, 1
  %steps_odd =l addl %steps_cur, 1
  jmp @inner_after

@inner_after:
  %x_next =l phi @even_case %x_even, @odd_case %x_odd
  %steps_next =l phi @even_case %steps_even, @odd_case %steps_odd
  jmp @inner

@inner_done:
  %is_bigger =w csltl %max, %steps_cur
  jnz %is_bigger, @update_max, @outer_continue

@update_max:
  %max_updated =l copy %steps_cur
  jmp @outer_continue

@outer_continue:
  %max_next =l phi @inner_done %max, @update_max %max_updated
  %k_next =l addl %k, 1
  jmp @outer

@exit:
  ret %max
}
