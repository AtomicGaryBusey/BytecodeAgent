vm_run:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Ldone
    mov r10, qword ptr [rdi + r8*8]
    cmp r10, 0
    je .Lhalt
    cmp r10, 1
    je .Lpush
    cmp r10, 2
    je .Ladd
    cmp r10, 3
    je .Lsub
    cmp r10, 4
    je .Lmul
    cmp r10, 5
    je .Ldup
    cmp r10, 6
    je .Lswap
    cmp r10, 7
    je .Ldrop
    cmp r10, 8
    je .Lover
    cmp r10, 9
    je .Llt
    cmp r10, 10
    je .Ljz
    cmp r10, 11
    je .Ljmp
    inc r8
    jmp .Lloop
.Lhalt:
    jmp .Ldone
.Lpush:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    add r11, r10
    mov qword ptr [rdx + r9*8 - 16], r11
    dec r9
    inc r8
    jmp .Lloop
.Lsub:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    sub r11, r10
    mov qword ptr [rdx + r9*8 - 16], r11
    dec r9
    inc r8
    jmp .Lloop
.Lmul:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    imul r11, r10
    mov qword ptr [rdx + r9*8 - 16], r11
    dec r9
    inc r8
    jmp .Lloop
.Ldup:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 16], r10
    mov qword ptr [rdx + r9*8 - 8], r11
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    xor rax, rax
    cmp r11, r10
    setl al
    mov qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Ljz:
    mov r10, qword ptr [rdx + r9*8 - 8]
    dec r9
    mov r11, qword ptr [rdi + r8*8 + 8]
    test r10, r10
    jnz .Ljz_not_zero
    mov r8, r11
    jmp .Lloop
.Ljz_not_zero:
    add r8, 2
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ldone:
    mov r11, rcx
    sub r11, r9
    mov rcx, r11
    xor rax, rax
    mov rdi, rdx
    lea rdi, [rdi + r9*8]
    rep stosq
    cmp r9, 0
    je .Lret_zero
    mov rax, qword ptr [rdx + r9*8 - 8]
    ret
.Lret_zero:
    xor rax, rax
    ret
