export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
  %i0 =l copy 0
  %pos0 =l copy 0
  jmp @loop

@loop
  %i =l phi @entry %i0, @lit_done %i_after_lit, @rep_done %i_after_rep, @noop_done %i_after_noop
  %pos =l phi @entry %pos0, @lit_done %pos_after_lit, @rep_done %pos_after_rep, @noop_done %pos_after_noop
  %more =w csltl %i, %n
  jnz %more, @packet
  ret %pos

@packet
  %src_i =l add %src, %i
  %h32 =w loadsb %src_i
  %h =l extsw %h32
  %hneg =w csltl %h, 0
  jnz %hneg, @neg_header

  %len =l add %h, 1
  %packet_end =l add %i, 1
  %packet_end2 =l add %packet_end, %len
  %src_short =w csgtl %packet_end2, %n
  jnz %src_short, @err2
  %limit =l add %pos, %len
  %overflow =w csgtl %limit, %cap
  jnz %overflow, @lit_overflow

  %lit_src0 =l add %src, %i
  %lit_src =l add %lit_src0, 1
  %lit_dst =l add %out, %pos
  %lit_count =l copy %len
  jmp @lit_copy

@lit_copy
  %lc =l phi @packet %lit_count, @lit_copy_next %lc_next
  %sp =l phi @packet %lit_src, @lit_copy_next %sp_next
  %dp =l phi @packet %lit_dst, @lit_copy_next %dp_next
  %lc_zero =w ceql %lc, 0
  jnz %lc_zero, @lit_done
  %lb =w loadub %sp
  storeb %lb, %dp
  jmp @lit_copy_next

@lit_copy_next
  %sp_next =l add %sp, 1
  %dp_next =l add %dp, 1
  %lc_next =l sub %lc, 1
  jmp @lit_copy

@lit_done
  %i_step =l add %i, 1
  %i_after_lit =l add %i_step, %len
  %pos_after_lit =l add %pos, %len
  jmp @loop

@neg_header
  %h128 =w ceql %h, -128
  jnz %h128, @noop_done

  %count =l sub 1, %h
  %repeat_end =l add %i, 2
  %repeat_short =w csgtl %repeat_end, %n
  jnz %repeat_short, @err2
  %repeat_limit =l add %pos, %count
  %repeat_overflow =w csgtl %repeat_limit, %cap
  jnz %repeat_overflow, @rep_overflow

  %rep_src0 =l add %src, %i
  %rep_src =l add %rep_src0, 1
  %rep_val =w loadub %rep_src
  %rep_dst =l add %out, %pos
  %rep_count =l copy %count
  jmp @rep_fill

@rep_fill
  %rc =l phi @neg_header %rep_count, @rep_fill_next %rc_next
  %rp =l phi @neg_header %rep_dst, @rep_fill_next %rp_next
  %rc_zero =w ceql %rc, 0
  jnz %rc_zero, @rep_done
  storeb %rep_val, %rp
  jmp @rep_fill_next

@rep_fill_next
  %rp_next =l add %rp, 1
  %rc_next =l sub %rc, 1
  jmp @rep_fill

@rep_done
  %i_after_rep =l add %i, 2
  %pos_after_rep =l add %pos, %count
  jmp @loop

@noop_done
  %i_after_noop =l add %i, 1
  %pos_after_noop =l copy %pos
  jmp @loop

@lit_overflow
  %lit_pos =l phi @packet %pos
  %lit_i =l phi @packet %i
  %lit_rem =l sub %cap, %lit_pos
  %lit_ov_count =l copy %lit_rem
  %lit_ov_zero =w ceql %lit_ov_count, 0
  jnz %lit_ov_zero, @ret_m1
  %lit_ov_src0 =l add %src, %lit_i
  %lit_ov_src =l add %lit_ov_src0, 1
  %lit_ov_dst =l add %out, %lit_pos
  jmp @lit_ov_copy

@lit_ov_copy
  %lovc =l phi @lit_overflow %lit_ov_count, @lit_ov_copy_next %lovc_next
  %lovs =l phi @lit_overflow %lit_ov_src, @lit_ov_copy_next %lovs_next
  %lovd =l phi @lit_overflow %lit_ov_dst, @lit_ov_copy_next %lovd_next
  %lov_zero =w ceql %lovc, 0
  jnz %lov_zero, @ret_m1
  %lov_byte =w loadub %lovs
  storeb %lov_byte, %lovd
  jmp @lit_ov_copy_next

@lit_ov_copy_next
  %lovs_next =l add %lovs, 1
  %lovd_next =l add %lovd, 1
  %lovc_next =l sub %lovc, 1
  jmp @lit_ov_copy

@rep_overflow
  %rep_pos =l phi @neg_header %pos
  %rep_byte =w phi @neg_header %rep_val
  %rep_rem =l sub %cap, %rep_pos
  %rep_ov_count =l copy %rep_rem
  %rep_ov_zero =w ceql %rep_ov_count, 0
  jnz %rep_ov_zero, @ret_m1
  %rep_ov_dst =l add %out, %rep_pos
  jmp @rep_ov_fill

@rep_ov_fill
  %rovc =l phi @rep_overflow %rep_ov_count, @rep_ov_fill_next %rovc_next
  %rovd =l phi @rep_overflow %rep_ov_dst, @rep_ov_fill_next %rovd_next
  %rov_zero =w ceql %rovc, 0
  jnz %rov_zero, @ret_m1
  storeb %rep_byte, %rovd
  jmp @rep_ov_fill_next

@rep_ov_fill_next
  %rovd_next =l add %rovd, 1
  %rovc_next =l sub %rovc, 1
  jmp @rep_ov_fill

@err2
  ret -2

@ret_m1
  ret -1
}
