export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
  %i0 =l copy 0
  %pos0 =l copy 0
  jmp @loop

@loop
  %i =l phi @entry %i0, @lit_done %i_after_lit_full, @rep_done %i_after_rep, @noop_done %i_after_noop
  %pos =l phi @entry %pos0, @lit_done %pos_after_lit, @rep_done %pos_after_rep, @noop_done %pos_after_noop
  %cont =w csltl %i, %n
  jnz %cont, @packet
  ret %pos

@packet
  %hp =l add %src, %i
  %h32 =w loadsb %hp
  %h =l extsw %h32
  %isneg =w csltl %h, 0
  jnz %isneg, @neg_header

  %len =l add %h, 1
  %need =l add %i, 1
  %need_total =l add %need, %len
  %src_short =w csgtl %need_total, %n
  jnz %src_short, @err2

  %need_out =l add %pos, %len
  %lit_over =w csgtl %need_out, %cap
  jnz %lit_over, @lit_overflow

  %lit_src =l add %hp, 1
  %lit_dst =l add %out, %pos
  %lit_count =l copy %len
  jmp @lit_copy

@lit_copy
  %lc =l phi @packet %lit_count, @lit_copy_next %lc_next
  %sp =l phi @packet %lit_src, @lit_copy_next %sp_next
  %dp =l phi @packet %lit_dst, @lit_copy_next %dp_next
  %lc_zero =w ceql %lc, 0
  jnz %lc_zero, @lit_done
  %lb =w loadub %sp
  storeb %lb, %dp
  jmp @lit_copy_next

@lit_copy_next
  %sp_next =l add %sp, 1
  %dp_next =l add %dp, 1
  %lc_next =l sub %lc, 1
  jmp @lit_copy

@lit_done
  %i_after_lit =l add %i, 1
  %i_after_lit_full =l add %i_after_lit, %len
  %pos_after_lit =l add %pos, %len
  jmp @loop

@neg_header
  %is_noop =w ceql %h, -128
  jnz %is_noop, @noop_done

  %count =l sub 1, %h
  %rep_need =l add %i, 2
  %rep_short =w csgtl %rep_need, %n
  jnz %rep_short, @err2

  %rep_need_out =l add %pos, %count
  %rep_over =w csgtl %rep_need_out, %cap
  jnz %rep_over, @rep_overflow

  %rep_src =l add %hp, 1
  %rep_byte =w loadub %rep_src
  %rep_dst =l add %out, %pos
  %rep_count =l copy %count
  jmp @rep_fill

@rep_fill
  %rc =l phi @neg_header %rep_count, @rep_fill_next %rc_next
  %rd =l phi @neg_header %rep_dst, @rep_fill_next %rd_next
  %rc_zero =w ceql %rc, 0
  jnz %rc_zero, @rep_done
  storeb %rep_byte, %rd
  jmp @rep_fill_next

@rep_fill_next
  %rd_next =l add %rd, 1
  %rc_next =l sub %rc, 1
  jmp @rep_fill

@rep_done
  %i_after_rep =l add %i, 2
  %pos_after_rep =l add %pos, %count
  jmp @loop

@noop_done
  %i_after_noop =l add %i, 1
  %pos_after_noop =l copy %pos
  jmp @loop

@lit_overflow
  %room =l sub %cap, %pos
  %len_gt_room =w csgtl %len, %room
  jnz %len_gt_room, @lit_overflow_clip
  %copylen0 =l copy %len
  %ov_sp0 =l add %hp, 1
  %ov_dp0 =l add %out, %pos
  jmp @ov_loop

@lit_overflow_clip
  %copylen1 =l copy %room
  %ov_sp1 =l add %hp, 1
  %ov_dp1 =l add %out, %pos
  jmp @ov_loop

@ov_loop
  %copylen =l phi @lit_overflow %copylen0, @lit_overflow_clip %copylen1, @ov_next %copylen_next
  %ov_sp =l phi @lit_overflow %ov_sp0, @lit_overflow_clip %ov_sp1, @ov_next %ov_sp_next
  %ov_dp =l phi @lit_overflow %ov_dp0, @lit_overflow_clip %ov_dp1, @ov_next %ov_dp_next
  %ov_zero =w ceql %copylen, 0
  jnz %ov_zero, @ret_m1
  %ov_byte =w loadub %ov_sp
  storeb %ov_byte, %ov_dp
  jmp @ov_next

@ov_next
  %ov_sp_next =l add %ov_sp, 1
  %ov_dp_next =l add %ov_dp, 1
  %copylen_next =l sub %copylen, 1
  jmp @ov_loop

@rep_overflow
  %room_rep =l sub %cap, %pos
  %count_gt_room =w csgtl %count, %room_rep
  jnz %count_gt_room, @rep_overflow_clip
  %rep_copy0 =l copy %count
  %rep_dst0 =l add %out, %pos
  jmp @rep_ov_loop

@rep_overflow_clip
  %rep_copy1 =l copy %room_rep
  %rep_dst1 =l add %out, %pos
  jmp @rep_ov_loop

@rep_ov_loop
  %rep_copy =l phi @rep_overflow %rep_copy0, @rep_overflow_clip %rep_copy1, @rep_ov_next %rep_copy_next
  %rep_dst =l phi @rep_overflow %rep_dst0, @rep_overflow_clip %rep_dst1, @rep_ov_next %rep_dst_next
  %rep_zero =w ceql %rep_copy, 0
  jnz %rep_zero, @ret_m1
  storeb %rep_byte, %rep_dst
  jmp @rep_ov_next

@rep_ov_next
  %rep_dst_next =l add %rep_dst, 1
  %rep_copy_next =l sub %rep_copy, 1
  jmp @rep_ov_loop

@err2
  %m2 =l copy -2
  ret %m2

@ret_m1
  %m1 =l copy -1
  ret %m1
}
