packbits_decode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r8, rsi
    mov r9, rdi
    mov r10, rdx
    mov r11, rcx
    xor r12, r12
    cmp r8, 0
    jle .Ldone
.Lloop:
    cmp r8, 0
    jle .Ldone
    movsx rax, byte ptr [r9]
    inc r9
    dec r8
    cmp rax, 0
    jge .Lliteral
    cmp rax, -128
    je .Lloop
    cmp r8, 0
    jle .Lerr_src
    movzx ecx, byte ptr [r9]
    inc r9
    dec r8
    mov rbx, rax
    neg rbx
    add rbx, 1
    mov rdx, r11
    sub rdx, r12
    cmp rbx, rdx
    jle .Lrepeat_full
    mov rdx, r11
    sub rdx, r12
    mov r13, r10
    add r13, r12
.Lrepeat_partial_fill:
    cmp rdx, 0
    je .Lcap_overflow
    mov byte ptr [r13], cl
    inc r13
    dec rdx
    jmp .Lrepeat_partial_fill
.Lrepeat_full:
    mov rdx, rbx
    mov r13, r10
    add r13, r12
.Lrepeat_full_fill:
    cmp rdx, 0
    je .Lrepeat_done
    mov byte ptr [r13], cl
    inc r13
    dec rdx
    jmp .Lrepeat_full_fill
.Lrepeat_done:
    add r12, rbx
    jmp .Lloop
.Lliteral:
    mov rbx, rax
    inc rbx
    cmp r8, rbx
    jl .Lerr_src
    mov rdx, r11
    sub rdx, r12
    cmp rbx, rdx
    jle .Lliteral_full
    mov rdx, r11
    sub rdx, r12
    mov r13, r10
    add r13, r12
.Lliteral_partial_fill:
    cmp rdx, 0
    je .Lcap_overflow
    mov r14b, byte ptr [r9]
    mov byte ptr [r13], r14b
    inc r9
    inc r13
    dec rdx
    jmp .Lliteral_partial_fill
.Lliteral_full:
    mov rdx, rbx
    mov r13, r10
    add r13, r12
.Lliteral_full_fill:
    cmp rdx, 0
    je .Lliteral_after_write
    mov r14b, byte ptr [r9]
    mov byte ptr [r13], r14b
    inc r9
    inc r13
    dec rdx
    jmp .Lliteral_full_fill
.Lliteral_after_write:
    add r12, rbx
    jmp .Lloop
.Lcap_overflow:
    mov r12, r11
    mov rax, -1
    jmp .Lret
.Lerr_src:
    mov rax, -2
    jmp .Lret
.Ldone:
    mov rax, r12
.Lret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
