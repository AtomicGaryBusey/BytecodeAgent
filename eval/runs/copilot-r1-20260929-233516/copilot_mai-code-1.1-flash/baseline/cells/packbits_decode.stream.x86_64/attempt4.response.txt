packbits_decode:
    push r12
    push r13
    push r14
    xor r12, r12
    xor r13, r13
.Lloop:
    cmp r13, rsi
    jge .Ldone
    movsx rax, byte ptr [rdi + r13]
    cmp eax, -128
    je .Lnoop
    cmp eax, 0
    jge .Lliteral
    mov r9, 1
    sub r9, rax
    mov r8, rsi
    sub r8, r13
    cmp r8, 2
    jl .Lerr_src
    mov r8, rcx
    sub r8, r12
    cmp r8, 0
    jle .Lreturn_neg1
    cmp r9, r8
    jle .Lrep_write_all
    mov r9, r8
    mov r14b, byte ptr [rdi + r13 + 1]
    lea r8, [rdx + r12]
.Lrep_partial_loop:
    mov byte ptr [r8], r14b
    inc r8
    dec r9
    jnz .Lrep_partial_loop
    mov rax, -1
    jmp .Lreturn
.Lrep_write_all:
    mov r14b, byte ptr [rdi + r13 + 1]
    lea r8, [rdx + r12]
    mov r9, 1
    sub r9, rax
.Lrep_full_loop:
    mov byte ptr [r8], r14b
    inc r8
    dec r9
    jnz .Lrep_full_loop
    mov r10, 1
    sub r10, rax
    add r12, r10
    add r13, 2
    jmp .Lloop
.Lliteral:
    mov r9, rax
    inc r9
    mov r8, rsi
    sub r8, r13
    lea r10, [r9 + 1]
    cmp r8, r10
    jl .Lerr_src
    mov r8, rcx
    sub r8, r12
    cmp r8, 0
    jle .Lreturn_neg1
    cmp r9, r8
    jle .Llit_write_all
    mov r9, r8
    lea r10, [rdi + r13 + 1]
    lea r11, [rdx + r12]
.Llit_partial_loop:
    mov r14b, byte ptr [r10]
    mov byte ptr [r11], r14b
    inc r10
    inc r11
    dec r9
    jnz .Llit_partial_loop
    mov rax, -1
    jmp .Lreturn
.Llit_write_all:
    lea r10, [rdi + r13 + 1]
    lea r11, [rdx + r12]
    mov r9, rax
    inc r9
.Llit_full_loop:
    mov r14b, byte ptr [r10]
    mov byte ptr [r11], r14b
    inc r10
    inc r11
    dec r9
    jnz .Llit_full_loop
    mov r10, rax
    inc r10
    add r12, r10
    add r13, r10
    inc r13
    jmp .Lloop
.Lnoop:
    inc r13
    jmp .Lloop
.Lerr_src:
    mov rax, -2
    jmp .Lreturn
.Lreturn_neg1:
    mov rax, -1
.Lreturn:
    pop r14
    pop r13
    pop r12
    ret
.Ldone:
    mov rax, r12
    pop r14
    pop r13
    pop r12
    ret
