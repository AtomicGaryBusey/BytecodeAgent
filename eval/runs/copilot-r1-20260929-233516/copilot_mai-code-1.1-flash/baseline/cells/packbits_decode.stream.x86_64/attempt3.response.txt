packbits_decode:
    xor r8, r8
.Lloop:
    cmp rsi, 0
    jle .Ldone
    movsx rax, byte ptr [rdi]
    cmp rax, 0
    jl .Lrepeat
    mov r9, rax
    inc r9
    mov r10, rsi
    dec r10
    cmp r10, r9
    jl .Lerr_src
    mov r10, rcx
    sub r10, r8
    cmp r10, r9
    jl .Llit_partial
    lea r11, [rdx + r8]
    lea r12, [rdi + 1]
    mov r13, r9
.Llit_write_all:
    mov r14b, byte ptr [r12]
    mov byte ptr [r11], r14b
    inc r12
    inc r11
    dec r13
    jnz .Llit_write_all
    add r8, r9
    lea rdi, [rdi + 1 + r9]
    sub rsi, r9
    dec rsi
    jmp .Lloop
.Llit_partial:
    cmp r10, 0
    jle .Lreturn_neg1
    lea r11, [rdx + r8]
    lea r12, [rdi + 1]
    mov r13, r10
.Llit_write_partial:
    mov r14b, byte ptr [r12]
    mov byte ptr [r11], r14b
    inc r12
    inc r11
    dec r13
    jnz .Llit_write_partial
    mov rax, -1
    ret
.Lrepeat:
    cmp rax, -128
    je .Lnoop
    mov r9, rsi
    dec r9
    cmp r9, 1
    jl .Lerr_src
    mov r9, 1
    sub r9, rax
    mov r10, rcx
    sub r10, r8
    cmp r10, r9
    jge .Lrep_all
    cmp r10, 0
    jle .Lreturn_neg1
    lea r11, [rdx + r8]
    mov r12b, byte ptr [rdi + 1]
    mov r13, r10
.Lrep_write_partial:
    mov byte ptr [r11], r12b
    inc r11
    dec r13
    jnz .Lrep_write_partial
    mov rax, -1
    ret
.Lrep_all:
    lea r11, [rdx + r8]
    mov r12b, byte ptr [rdi + 1]
    mov r13, r9
.Lrep_write_all:
    mov byte ptr [r11], r12b
    inc r11
    dec r13
    jnz .Lrep_write_all
    add r8, r9
    add rdi, 2
    sub rsi, 2
    jmp .Lloop
.Lnoop:
    inc rdi
    dec rsi
    jmp .Lloop
.Lerr_src:
    mov rax, -2
    ret
.Lreturn_neg1:
    mov rax, -1
    ret
.Ldone:
    mov rax, r8
    ret
