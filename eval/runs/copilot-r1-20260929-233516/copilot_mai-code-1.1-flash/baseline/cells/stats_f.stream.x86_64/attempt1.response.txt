stats_f:
    push r12
    push r13
    push r14
    push r15
    push rbp
    sub rsp, 32
    mov r12, rcx
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
    mov rbp, rsi
    xor eax, eax
    mov qword ptr [rsp], rax
    mov qword ptr [rsp + 8], rax
    mov qword ptr [rsp + 16], rax
    mov qword ptr [rsp + 24], rax
    cld
.Lloop:
    cmp r14, 0
    je .Ldone
    mov rdi, qword ptr [r13]
    cld
    call r15
    test rax, 1
    jz .Lskipodd
    mov rcx, qword ptr [rsp + 24]
    inc rcx
    mov qword ptr [rsp + 24], rcx
.Lskipodd:
    cmp r14, rbp
    jne .Lnotfirst
    mov qword ptr [rsp], rax
    mov qword ptr [rsp + 8], rax
    mov qword ptr [rsp + 16], rax
    jmp .Ladvance
.Lnotfirst:
    cmp rax, qword ptr [rsp]
    jl .Lsetmin
    jmp .Lcheckmax
.Lsetmin:
    mov qword ptr [rsp], rax
.Lcheckmax:
    cmp rax, qword ptr [rsp + 8]
    jle .Lsum
    mov qword ptr [rsp + 8], rax
.Lsum:
    mov rcx, qword ptr [rsp + 16]
    add rcx, rax
    mov qword ptr [rsp + 16], rcx
.Ladvance:
    add r13, 8
    dec r14
    jmp .Lloop
.Ldone:
    mov rax, qword ptr [rsp + 24]
    mov rcx, qword ptr [rsp]
    mov qword ptr [r12], rcx
    mov rcx, qword ptr [rsp + 8]
    mov qword ptr [r12 + 8], rcx
    mov rcx, qword ptr [rsp + 16]
    mov qword ptr [r12 + 16], rcx
    add rsp, 32
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    cld
    ret
