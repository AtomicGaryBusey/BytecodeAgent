qsort_pairs:
    push rbp
    push r12
    push r13
    mov rbp, rdi
    cmp rsi, 1
    jle .Ldone
    mov r8, rsi
    shr r8, 1
    mov r9, rsi
    dec r9
    lea r10, [rbp + r8*16]
    lea r11, [rbp + r9*16]
    mov rax, qword ptr [r10]
    mov rcx, qword ptr [r10+8]
    mov rdx, qword ptr [r11]
    mov r8, qword ptr [r11+8]
    mov qword ptr [r10], rdx
    mov qword ptr [r10+8], r8
    mov qword ptr [r11], rax
    mov qword ptr [r11+8], rcx
    mov r12, 0
    mov r13, 0
    mov r8, rsi
    dec r8
.partition_loop:
    lea r11, [rbp + r8*16]
    cmp r13, r8
    jge .Lpartition_done
    lea r9, [rbp + r13*16]
    mov r10, qword ptr [r9]
    cmp r10, qword ptr [r11]
    jl .Lswap
    jg .Lcontinue
    mov r10, qword ptr [r9+8]
    cmp r10, qword ptr [r11+8]
    jl .Lswap
.Lcontinue:
    inc r13
    jmp .partition_loop
.Lswap:
    lea r9, [rbp + r12*16]
    lea r10, [rbp + r13*16]
    mov rax, qword ptr [r9]
    mov rdx, qword ptr [r9+8]
    mov rcx, qword ptr [r10]
    mov r8, qword ptr [r10+8]
    mov qword ptr [r9], rcx
    mov qword ptr [r9+8], r8
    mov qword ptr [r10], rax
    mov qword ptr [r10+8], rdx
    inc r12
    inc r13
    jmp .partition_loop
.Lpartition_done:
    lea r9, [rbp + r12*16]
    mov r10, rsi
    dec r10
    lea r10, [rbp + r10*16]
    mov rax, qword ptr [r9]
    mov rdx, qword ptr [r9+8]
    mov rcx, qword ptr [r10]
    mov r8, qword ptr [r10+8]
    mov qword ptr [r9], rcx
    mov qword ptr [r9+8], r8
    mov qword ptr [r10], rax
    mov qword ptr [r10+8], rdx
    mov rdx, rsi
    sub rdx, r12
    dec rdx
    cmp r12, 1
    jle .Lskip_left
    lea r8, [rbp + r12*16 + 16]
    push r8
    push rdx
    mov rdi, rbp
    mov rsi, r12
    call qsort_pairs
    pop rdx
    pop r8
.Lskip_left:
    cmp rdx, 1
    jle .Ldone
    mov rdi, r8
    mov rsi, rdx
    call qsort_pairs
.Ldone:
    xor eax, eax
    pop r13
    pop r12
    pop rbp
    ret
