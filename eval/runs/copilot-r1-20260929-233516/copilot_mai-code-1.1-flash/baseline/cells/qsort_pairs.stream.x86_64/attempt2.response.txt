qsort_pairs:
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    mov r13, rdi
    cmp r12, 1
    jle .Ldone
    mov r8, 0
    mov r9, 1
    mov r10, qword ptr [r13]
    mov r11, qword ptr [r13+8]
.Lloop:
    cmp r9, r12
    jge .Lpart_done
    mov r14, r9
    shl r14, 4
    lea r14, [r13 + r14]
    mov rax, qword ptr [r14]
    cmp rax, r10
    jl .Lis_less
    jg .Lnext
    mov rax, qword ptr [r14+8]
    cmp rax, r11
    jl .Lis_less
    jg .Lnext
    jmp .Lnext
.Lis_less:
    inc r8
    mov r15, r8
    shl r15, 4
    lea r15, [r13 + r15]
    mov rax, qword ptr [r15]
    mov rdx, qword ptr [r15+8]
    mov rcx, qword ptr [r14]
    mov rsi, qword ptr [r14+8]
    mov qword ptr [r15], rcx
    mov qword ptr [r15+8], rsi
    mov qword ptr [r14], rax
    mov qword ptr [r14+8], rdx
.Lnext:
    inc r9
    jmp .Lloop
.Lpart_done:
    mov r14, r8
    shl r14, 4
    lea r14, [r13 + r14]
    mov rax, qword ptr [r13]
    mov rdx, qword ptr [r13+8]
    mov rcx, qword ptr [r14]
    mov rsi, qword ptr [r14+8]
    mov qword ptr [r13], rcx
    mov qword ptr [r13+8], rsi
    mov qword ptr [r14], rax
    mov qword ptr [r14+8], rdx
    mov r14, r8
    inc r14
    shl r14, 4
    lea r14, [r13 + r14]
    mov r15, r12
    sub r15, r8
    dec r15
    mov rsi, r8
    cmp rsi, 1
    jle .Lskip_left
    mov rdi, r13
    call qsort_pairs
.Lskip_left:
    cmp r15, 1
    jle .Ldone
    mov rdi, r14
    mov rsi, r15
    call qsort_pairs
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    ret
