export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    %j =l copy 0
@zero_loop
    %j_le_n =w csletl %j, %n
    jnz %j_le_n, @zero_body
    jmp @after_zero
@zero_body
    %off =l mul %j, 8
    %p =l add %row, %off
    %zero =l copy 0
    storel %zero, %p
    %j =l add %j, 1
    jmp @zero_loop
@after_zero
    %i =l copy 0
@outer_loop
    %i_lt_m =w csltl %i, %m
    jnz %i_lt_m, @inner_init
    jmp @done
@inner_init
    %diag =l copy 0
    %j =l copy 1
@inner_loop
    %j_le_n =w csletl %j, %n
    jnz %j_le_n, @inner_body
    jmp @outer_next
@inner_body
    %joff =l mul %j, 8
    %rowj =l add %row, %joff
    %up =l loadl %rowj
    %j_minus_1 =l sub %j, 1
    %j1off =l mul %j_minus_1, 8
    %rowjm1 =l add %row, %j1off
    %prev =l loadl %rowjm1
    %xp =l add %x, %i
    %xb =w loadub %xp
    %yp =l add %y, %j_minus_1
    %yb =w loadub %yp
    %eq =w ceqw %xb, %yb
    jnz %eq, @take_diag
    %gt =w csgtl %up, %prev
    jnz %gt, @store_up
    storel %prev, %rowj
    jmp @after_store
@store_up
    storel %up, %rowj
    jmp @after_store
@take_diag
    %diag_plus_1 =l add %diag, 1
    storel %diag_plus_1, %rowj
@after_store
    %diag =l copy %up
    %j =l add %j, 1
    jmp @inner_loop
@outer_next
    %i =l add %i, 1
    jmp @outer_loop
@done
    %n_off =l mul %n, 8
    %res_p =l add %row, %n_off
    %res =l loadl %res_p
    ret %res
}
