export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %j =l copy 0
@zero_loop
    %j_le_n =w cslel %j, %n
    jnz %j_le_n, @zero_body
    jmp @after_zero
@zero_body
    %off =l mul %j, %eight
    %p =l add %row, %off
    storel %zero, %p
    %j =l add %j, %one
    jmp @zero_loop
@after_zero
    %i =l copy 0
@outer_loop
    %i_lt_m =w csltl %i, %m
    jnz %i_lt_m, @inner_init
    jmp @done
@inner_init
    %diag =l copy 0
    %j =l copy 1
@inner_loop
    %j_le_n =w cslel %j, %n
    jnz %j_le_n, @inner_body
    jmp @outer_next
@inner_body
    %rowj_off =l mul %j, %eight
    %rowj_ptr =l add %row, %rowj_off
    %up =l loadl %rowj_ptr
    %j_minus_1 =l sub %j, %one
    %prev_off =l mul %j_minus_1, %eight
    %prev_ptr =l add %row, %prev_off
    %prev =l loadl %prev_ptr
    %xp =l add %x, %i
    %xb =w loadub %xp
    %yp =l add %y, %j_minus_1
    %yb =w loadub %yp
    %eq =w ceqw %xb, %yb
    jnz %eq, @take_diag
    %up_gt_prev =w csgtl %up, %prev
    jnz %up_gt_prev, @store_up
    storel %prev, %rowj_ptr
    jmp @after_update
@store_up
    storel %up, %rowj_ptr
    jmp @after_update
@take_diag
    %diag_plus_1 =l add %diag, %one
    storel %diag_plus_1, %rowj_ptr
@after_update
    %diag =l copy %up
    %j =l add %j, %one
    jmp @inner_loop
@outer_next
    %i =l add %i, %one
    jmp @outer_loop
@done
    %res_off =l mul %n, %eight
    %res_ptr =l add %row, %res_off
    %res =l loadl %res_ptr
    ret %res
}
