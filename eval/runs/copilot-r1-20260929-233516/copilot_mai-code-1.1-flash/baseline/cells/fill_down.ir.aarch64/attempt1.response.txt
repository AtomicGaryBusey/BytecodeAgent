export function l $fill_down(l %a, l %n, l %start) {
@entry
  %i = copyl 0
  jmp @loop

@loop
  %cond = csltl %i, %n
  jnz %cond, @body
  jmp @done

@body
  %v = subl %start, %i
  %off = shll %i, 3
  %p = addl %a, %off
  storel %v, %p
  %next = addl %i, 1
  %i = copyl %next
  jmp @loop

@done
  ret l %n
}
