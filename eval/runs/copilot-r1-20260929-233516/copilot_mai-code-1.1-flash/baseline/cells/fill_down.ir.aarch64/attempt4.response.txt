export function l $fill_down(l %a, l %n, l %start) {
@entry
  jmp @loop

@loop
  %i = phi l [0, @entry], [%next, @body]
  %cond = csltl %i, %n
  jnz %cond, @body
  jmp @done

@body
  %v = subl %start, %i
  %off = shll %i, 3
  %p = addl %a, %off
  storel %v, %p
  %next = addl %i, 1
  jmp @loop

@done
  ret l %n
}
