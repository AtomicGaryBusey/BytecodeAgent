export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %dst_lt_src = cultl %dst, %src
    jnz %dst_lt_src, @copy_forward, @check_equal

@check_equal
    %src_lt_dst = cultl %src, %dst
    jnz %src_lt_dst, @prepare_backward, @done

@copy_forward
    %i = phi l [0, @start], [%i2, @fwd_body]
    %p = phi l [%src, @start], [%p2, @fwd_body]
    %q = phi l [%dst, @start], [%q2, @fwd_body]
    %cond = csltl %i, %n
    jnz %cond, @fwd_body, @done

@fwd_body
    %v = loadl %p, 0
    storel %q, %v, 0
    %i2 = addl %i, 1
    %p2 = addl %p, 8
    %q2 = addl %q, 8
    jmp @copy_forward

@prepare_backward
    %n8 = mull %n, 8
    %src_end = addl %src, %n8
    %dst_end = addl %dst, %n8
    jmp @copy_backward

@copy_backward
    %i_back = phi l [%n, @prepare_backward], [%i_back2, @bwd_body]
    %p_back = phi l [%src_end, @prepare_backward], [%p_back2, @bwd_body]
    %q_back = phi l [%dst_end, @prepare_backward], [%q_back2, @bwd_body]
    %cond_back = csltl 0, %i_back
    jnz %cond_back, @bwd_body, @done

@bwd_body
    %p_back2 = subl %p_back, 8
    %q_back2 = subl %q_back, 8
    %v_back = loadl %p_back2, 0
    storel %q_back2, %v_back, 0
    %i_back2 = subl %i_back, 1
    jmp @copy_backward

@done
    ret 0
}
