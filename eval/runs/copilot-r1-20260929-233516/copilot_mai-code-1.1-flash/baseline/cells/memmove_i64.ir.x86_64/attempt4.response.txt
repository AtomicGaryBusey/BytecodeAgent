export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %dst_lt_src = cultl %dst, %src
    jnz %dst_lt_src, @copy_forward, @check_reverse

@check_reverse
    %src_lt_dst = cultl %src, %dst
    jnz %src_lt_dst, @copy_backward, @done

@copy_forward
    %i = phi l [0, @entry], [%i2, @fwd_body]
    %p = phi l [%src, @entry], [%p2, @fwd_body]
    %q = phi l [%dst, @entry], [%q2, @fwd_body]
    %cond = csltl %i, %n
    jnz %cond, @fwd_body, @done

@fwd_body
    %v = loadl %p, 0
    storel %q, %v, 0
    %i2 = addl %i, 1
    %p2 = addl %p, 8
    %q2 = addl %q, 8
    jmp @copy_forward

@copy_backward
    %i_back = phi l [%n, @check_reverse], [%i_back2, @bwd_body]
    %p_back = phi l [%src, @check_reverse], [%p_back2, @bwd_body]
    %q_back = phi l [%dst, @check_reverse], [%q_back2, @bwd_body]
    %cond_back = csgtl %i_back, 0
    jnz %cond_back, @bwd_body, @done

@bwd_body
    %p_back2 = subl %p_back, 8
    %q_back2 = subl %q_back, 8
    %v_back = loadl %p_back2, 0
    storel %q_back2, %v_back, 0
    %i_back2 = subl %i_back, 1
    jmp @copy_backward

@done
    ret 0
}
