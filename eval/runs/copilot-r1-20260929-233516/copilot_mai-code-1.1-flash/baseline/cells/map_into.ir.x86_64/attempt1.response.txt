export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %count0 =l copy 0
    %srcp0 =l copy %src
    %dstp0 =l copy %dst
    %nrem0 =l copy %n
    %is_zero =w ceql %nrem0, 0
    jnz %is_zero, @done, @loop

@loop
    %count =l phi @entry %count0 @advance %count_after
    %srcp =l phi @entry %srcp0 @advance %srcp_after
    %dstp =l phi @entry %dstp0 @advance %dstp_after
    %nrem =l phi @entry %nrem0 @advance %nrem_after
    %srcv =l loadl %srcp
    %val =l call %f(l %srcv)
    storel %val, %dstp
    %lt =w csltl %val, %srcv
    jnz %lt, @inc, @next

@inc
    %count_inc =l addl %count, 1
    jmp @advance

@next
    %count_inc_n =l copy %count
    jmp @advance

@advance
    %count_after =l phi @inc %count_inc @next %count_inc_n
    %srcp_after =l addl %srcp, 8
    %dstp_after =l addl %dstp, 8
    %nrem_after =l subl %nrem, 1
    %done_test =w ceql %nrem_after, 0
    jnz %done_test, @done, @loop

@done
    %count_done =l phi @entry %count0 @advance %count_after
    ret %count_done
}
