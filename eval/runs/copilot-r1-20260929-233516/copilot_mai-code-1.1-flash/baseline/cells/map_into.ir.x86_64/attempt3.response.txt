export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %count0 =l copy 0
    %is_zero =w ceql %n, 0
    jnz %is_zero, @done, @loop

@loop
    %count =l phi @entry %count0 @continue %count1
    %srcp =l phi @entry %src @continue %srcp1
    %dstp =l phi @entry %dst @continue %dstp1
    %nrem =l phi @entry %n @continue %nrem1
    %srcv =l loadl %srcp
    %val =l call %f(l %srcv)
    storel %val, %dstp
    %lt =w csltl %val, %srcv
    jnz %lt, @inccount, @next

@inccount
    %count_inc =l addl %count, 1
    jmp @continue

@next
    %count_inc =l copy %count
    jmp @continue

@continue
    %count1 =l phi @inccount %count_inc @next %count_inc
    %srcp1 =l addl %srcp, 8
    %dstp1 =l addl %dstp, 8
    %nrem1 =l subl %nrem, 1
    %done_test =w ceql %nrem1, 0
    jnz %done_test, @done, @loop

@done
    %count_done =l phi @entry %count0 @continue %count1
    ret %count_done
}
