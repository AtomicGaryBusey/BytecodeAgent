export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %count0 =l copy 0
    %srcp0 =l copy %src
    %dstp0 =l copy %dst
    %nrem0 =l copy %n
    %is_zero =w ceql %nrem0, 0
    jnz %is_zero, @done, @loop

@loop
    %count =l phi @entry %count0 @advance %count_inc
    %srcp =l phi @entry %srcp0 @advance %srcp1
    %dstp =l phi @entry %dstp0 @advance %dstp1
    %nrem =l phi @entry %nrem0 @advance %nrem1
    %srcv =l loadl %srcp
    %val =l call %f(l %srcv)
    storel %val, %dstp
    %lt =w csltl %val, %srcv
    jnz %lt, @inccount, @next

@inccount
    %count_inc_a =l addl %count, 1
    jmp @advance

@next
    %count_inc_b =l copy %count
    jmp @advance

@advance
    %count_inc =l phi @inccount %count_inc_a @next %count_inc_b
    %srcp1 =l addl %srcp, 8
    %dstp1 =l addl %dstp, 8
    %nrem1 =l subl %nrem, 1
    %done_test =w ceql %nrem1, 0
    jnz %done_test, @done, @loop

@done
    %count_done =l phi @entry %count0 @advance %count_inc
    ret %count_done
}
