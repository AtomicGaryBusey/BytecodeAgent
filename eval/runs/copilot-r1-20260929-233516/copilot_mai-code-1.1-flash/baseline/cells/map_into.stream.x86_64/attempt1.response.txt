map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rcx
    mov r13, rsi
    mov r14, rdi
    mov rbp, rdx
    xor r15, r15
    xor rbx, rbx
.Lloop:
    cmp rbx, rbp
    jge .Ldone
    mov rdi, qword ptr [r13 + rbx*8]
    call r12
    mov qword ptr [r14 + rbx*8], rax
    cmp rax, qword ptr [r13 + rbx*8]
    jl .Linc
.Lnext:
    inc rbx
    jmp .Lloop
.Linc:
    inc r15
    jmp .Lnext
.Ldone:
    mov rax, r15
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
