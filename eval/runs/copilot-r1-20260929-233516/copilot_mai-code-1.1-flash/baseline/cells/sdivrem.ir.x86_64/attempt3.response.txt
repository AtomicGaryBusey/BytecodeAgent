export function l $sdivrem(l %a, l %b, l %rem) {
@entry
%zero = copy 0
%neg1 = copy -1
%int64_min = copy -9223372036854775808
%is_zero_b = ceql %b, %zero
jnz %is_zero_b, @b_zero
%is_min_a = ceql %a, %int64_min
%is_minus_one_b = ceql %b, %neg1
%overflow = andw %is_min_a, %is_minus_one_b
jnz %overflow, @a_min_b_minus_one
%q = div %a, %b
%r = rem %a, %b
storel %r, %rem
ret %q

@b_zero
storel %a, %rem
ret %zero

@a_min_b_minus_one
storel %zero, %rem
ret %int64_min
}
