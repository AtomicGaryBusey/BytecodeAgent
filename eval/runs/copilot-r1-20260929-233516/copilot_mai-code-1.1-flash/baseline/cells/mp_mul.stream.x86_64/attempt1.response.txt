mp_mul:
    push rbx
    push rbp
    push r13
    push r15

    mov r13, rdx
    mov r15, rsi
    add r15, rcx

    test r15, r15
    jz .Lret0

    xor rax, rax
.Lclear:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, r15
    jb .Lclear

    test rsi, rsi
    jz .Lret0
    test rcx, rcx
    jz .Lret0

    xor rbp, rbp
.Li:
    cmp rbp, rsi
    jae .Lscan
    xor rbx, rbx
    xor r9, r9
.Lj:
    cmp rbx, rcx
    jae .Lendj
    mov rax, qword ptr [rdi + rbp*8]
    mul qword ptr [r13 + rbx*8]
    lea r11, [r8 + rbp*8]
    lea r11, [r11 + rbx*8]
    mov r10, qword ptr [r11]
    add rax, r10
    adc rdx, 0
    add rax, r9
    adc rdx, 0
    mov qword ptr [r11], rax
    mov r9, rdx
    inc rbx
    jmp .Lj
.Lendj:
    mov r10, rbp
    add r10, rcx
    lea r11, [r8 + r10*8]
    mov rax, qword ptr [r11]
    add rax, r9
    mov qword ptr [r11], rax
    mov r9, 0
    jnc .Lnexti
    mov r9, 1
.Lprop:
    inc r10
    cmp r10, r15
    jae .Lnexti
    mov rax, qword ptr [r8 + r10*8]
    add rax, r9
    mov qword ptr [r8 + r10*8], rax
    mov r9, 0
    jnc .Lnexti
    mov r9, 1
    jmp .Lprop
.Lnexti:
    inc rbp
    jmp .Li

.Lscan:
    mov rax, r15
    dec rax
.Lscan_loop:
    cmp qword ptr [r8 + rax*8], 0
    jne .Lfound
    test rax, rax
    jz .Lret0
    dec rax
    jmp .Lscan_loop
.Lfound:
    lea rax, [rax + 1]
    jmp .Ldone

.Lret0:
    xor eax, eax
.Ldone:
    pop r15
    pop r13
    pop rbp
    pop rbx
    ret
