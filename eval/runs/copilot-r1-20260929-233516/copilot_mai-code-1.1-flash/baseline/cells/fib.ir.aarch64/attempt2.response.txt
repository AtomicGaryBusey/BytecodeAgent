export function l $fib(l %n) {
@entry
    %a0 = copy 0
    %b0 = copy 1
    %i0 = copy 0
    jmp @loop
@loop
    %i = phi l [%i0, @entry], [%i2, @loop]
    %a = phi l [%a0, @entry], [%a2, @loop]
    %b = phi l [%b0, @entry], [%b2, @loop]
    %done = ceql %i, %n
    jnz %done, @done
    %sum = addl %a, %b
    %a2 = copy %b
    %b2 = copy %sum
    %i2 = addl %i, 1
    jmp @loop
@done
    ret %a
}
