export function l $fib(l %n) {
@entry
    jmp @loop
@loop
    %i = phi l [0, @entry], [%i2, @loop]
    %a = phi l [0, @entry], [%a2, @loop]
    %b = phi l [1, @entry], [%b2, @loop]
    %done = ceql %i, %n
    jnz %done, @done
    %sum = addl %a, %b
    %a2 = copy %b
    %b2 = copy %sum
    %i2 = addl %i, 1
    jmp @loop
@done
    ret %a
}
