export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
  %i =l add 0, 0
  %written =l add 0, 0
  jmp @loop

@loop
  %i_lt_n =w csltl %i, %n
  jnz %i_lt_n, @body
  ret %written

@body
  %run =w add 1, 0
  %j =l add %i, 1

@runloop
  %j_lt_n =w csltl %j, %n
  jnz %j_lt_n, @runcheck
  jmp @runend

@runcheck
  %run_lt_128 =w csltl %run, 128
  jnz %run_lt_128, @runeq
  jmp @runend

@runeq
  %sip =l add %src, %i
  %bi =w loadub %sip
  %sjp =l add %src, %j
  %bj =w loadub %sjp
  %eq =w ceql %bi, %bj
  jnz %eq, @runinc
  jmp @runend

@runinc
  %run =w add %run, 1
  %j =l add %j, 1
  jmp @runloop

@runend
  %run_lt_3 =w csltl %run, 3
  jnz %run_lt_3, @literal
  jmp @repeat

@repeat
  %packetlen =l add 2, 0
  %need =l add %written, %packetlen
  %cap_lt_need =w csltl %cap, %need
  jnz %cap_lt_need, @fail
  %outp =l add %out, %written
  %header =w sub 257, %run
  %hptr =l add %outp, 0
  storeb %header, %hptr
  %datap =l add %hptr, 1
  %srcp =l add %src, %i
  %b =w loadub %srcp
  storeb %b, %datap
  %written =l add %written, 2
  %i =l add %i, %run
  jmp @loop

@literal
  %j =l add %i, 0

@lscan
  %j_lt_n =w csltl %j, %n
  jnz %j_lt_n, @lcheck
  jmp @lscan_done

@lcheck
  %delta =l sub %j, %i
  %delta_lt_128 =w csltl %delta, 128
  jnz %delta_lt_128, @lruncheck
  jmp @lscan_done

@lruncheck
  %runj =w add 1, 0
  %srcj =l add %src, %j
  %cj =w loadub %srcj
  %j1 =l add %j, 1

@lrunloop
  %j1_lt_n =w csltl %j1, %n
  jnz %j1_lt_n, @lruncheck2
  jmp @lrunend

@lruncheck2
  %runj_lt_128 =w csltl %runj, 128
  jnz %runj_lt_128, @lruncmp
  jmp @lrunend

@lruncmp
  %srcj1 =l add %src, %j1
  %cj1 =w loadub %srcj1
  %eqj =w ceql %cj, %cj1
  jnz %eqj, @lruninc
  jmp @lrunend

@lruninc
  %runj =w add %runj, 1
  %j1 =l add %j1, 1
  jmp @lrunloop

@lrunend
  %runj_lt_3 =w csltl %runj, 3
  jnz %runj_lt_3, @linclude
  jmp @lscan_done

@linclude
  %j =l add %j, 1
  jmp @lscan

@lscan_done
  %litlen =l sub %j, %i
  %packetlen =l add %litlen, 1
  %need =l add %written, %packetlen
  %cap_lt_need =w csltl %cap, %need
  jnz %cap_lt_need, @fail
  %outp =l add %out, %written
  %header =w sub %litlen, 1
  %hptr =l add %outp, 0
  storeb %header, %hptr
  %k =l add 0, 0

@litcopy
  %k_lt_len =w csltl %k, %litlen
  jnz %k_lt_len, @litcopy_body
  jmp @literal_done

@litcopy_body
  %srcp =l add %src, %i
  %srcbytep =l add %srcp, %k
  %srcbyte =w loadub %srcbytep
  %dstp =l add %hptr, 1
  %dstp2 =l add %dstp, %k
  storeb %srcbyte, %dstp2
  %k =l add %k, 1
  jmp @litcopy

@literal_done
  %written =l add %written, %packetlen
  %i =l add %j, 0
  jmp @loop

@fail
  ret -1
}
