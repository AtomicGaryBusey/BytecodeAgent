export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
  %i =l add 0, 0
  %written =l add 0, 0
  %outp =l add %out, 0
  jmp @loop

@loop
  %i_lt_n =w csltl %i, %n
  jnz %i_lt_n, @body
  ret %written

@body
  %srcp =l add %src, %i
  %b0 =w loadub %srcp
  %run =w add 1, 0
  %jrun =l add %i, 1

@runloop
  %jrun_lt_n =w csltl %jrun, %n
  jnz %jrun_lt_n, @runcheck
  jmp @runend

@runcheck
  %run_lt_128 =w csltl %run, 128
  jnz %run_lt_128, @runeq
  jmp @runend

@runeq
  %cp =l add %src, %jrun
  %b1 =w loadub %cp
  %eq =w ceql %b0, %b1
  jnz %eq, @runinc
  jmp @runend

@runinc
  %run =w add %run, 1
  %jrun =l add %jrun, 1
  jmp @runloop

@runend
  %run_lt_3 =w csltl %run, 3
  jnz %run_lt_3, @literal
  jmp @repeat

@repeat
  %need =l add %written, 2
  %cap_lt_need =w csltl %cap, %need
  jnz %cap_lt_need, @fail
  %header =w sub 257, %run
  %hdrp =l add %outp, 0
  storeb %header, %hdrp
  %datap =l add %hdrp, 1
  storeb %b0, %datap
  %outp =l add %outp, 2
  %written =l add %written, 2
  %i =l add %i, %run
  jmp @loop

@literal
  %j =l add %i, 0
  jmp @lscan

@lscan
  %j_lt_n =w csltl %j, %n
  jnz %j_lt_n, @lcheck
  jmp @lscan_done

@lcheck
  %delta =l sub %j, %i
  %delta_lt_128 =w csltl %delta, 128
  jnz %delta_lt_128, @lcheck_run
  jmp @lscan_done

@lcheck_run
  %runj =w add 1, 0
  %jp =l add %src, %j
  %cj =w loadub %jp
  %j1 =l add %j, 1

@lrunloop
  %j1_lt_n =w csltl %j1, %n
  jnz %j1_lt_n, @lruncheck
  jmp @lrunend

@lruncheck
  %runj_lt_128 =w csltl %runj, 128
  jnz %runj_lt_128, @lruncmp
  jmp @lrunend

@lruncmp
  %cp2 =l add %src, %j1
  %cj2 =w loadub %cp2
  %eq2 =w ceql %cj, %cj2
  jnz %eq2, @lruninc
  jmp @lrunend

@lruninc
  %runj =w add %runj, 1
  %j1 =l add %j1, 1
  jmp @lrunloop

@lrunend
  %runj_lt_3 =w csltl %runj, 3
  jnz %runj_lt_3, @linclude
  jmp @lscan_done

@linclude
  %j =l add %j, 1
  jmp @lscan

@lscan_done
  %litlen =l sub %j, %i
  %litpacket =l add %litlen, 1
  %need2 =l add %written, %litpacket
  %cap_lt_need2 =w csltl %cap, %need2
  jnz %cap_lt_need2, @fail
  %header2 =w sub %litlen, 1
  %hptr2 =l add %outp, 0
  storeb %header2, %hptr2
  %k =l add 0, 0
  jmp @litcopy

@litcopy
  %k_lt_len =w csltl %k, %litlen
  jnz %k_lt_len, @litcopy_body
  jmp @literal_done

@litcopy_body
  %srck =l add %src, %i
  %srck2 =l add %srck, %k
  %srcbyte =w loadub %srck2
  %dstp =l add %hptr2, 1
  %dstp2 =l add %dstp, %k
  storeb %srcbyte, %dstp2
  %k =l add %k, 1
  jmp @litcopy

@literal_done
  %outp =l add %hptr2, %litpacket
  %written =l add %written, %litpacket
  %i =l add %j, 0
  jmp @loop

@fail
  ret -1
}
