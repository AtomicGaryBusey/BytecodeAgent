export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
  %i =l add 0, 0
  %written =l add 0, 0
  %outp =l add %out, 0
  jmp @loop

@loop
  %i_lt_n =w csltl %i, %n
  jnz %i_lt_n, @body
  jmp @done

@body
  %srcp =l add %src, %i
  %b =w loadub %srcp
  %run =w add 1, 0
  %jrun =l add %i, 1

@runloop
  %jrun_lt_n =w csltl %jrun, %n
  jnz %jrun_lt_n, @runcheck
  jmp @runend

@runcheck
  %run_lt_128 =w csltl %run, 128
  jnz %run_lt_128, @runeq
  jmp @runend

@runeq
  %p2 =l add %src, %jrun
  %b2 =w loadub %p2
  %eq =w ceql %b, %b2
  jnz %eq, @runinc
  jmp @runend

@runinc
  %run =w add %run, 1
  %jrun =l add %jrun, 1
  jmp @runloop

@runend
  %run_lt_3 =w csltl %run, 3
  jnz %run_lt_3, @literal
  jmp @repeat

@repeat
  %limit =l add %written, 2
  %too_big =w csltl %cap, %limit
  jnz %too_big, @fail
  %header =w sub 257, %run
  %hp =l add %outp, 0
  storeb %header, %hp
  %dp =l add %hp, 1
  storeb %b, %dp
  %outp =l add %outp, 2
  %written =l add %written, 2
  %i =l add %i, %run
  jmp @loop

@literal
  %j =l add %i, 0
  jmp @lscan

@lscan
  %j_lt_n =w csltl %j, %n
  jnz %j_lt_n, @lcheck
  jmp @lscan_done

@lcheck
  %delta =l sub %j, %i
  %delta_lt_128 =w csltl %delta, 128
  jnz %delta_lt_128, @lcheck_run
  jmp @lscan_done

@lcheck_run
  %jr =w add 1, 0
  %p3 =l add %src, %j
  %c =w loadub %p3
  %qj =l add %j, 1

@lrunloop
  %qj_lt_n =w csltl %qj, %n
  jnz %qj_lt_n, @lruncheck
  jmp @lrunend

@lruncheck
  %jr_lt_128 =w csltl %jr, 128
  jnz %jr_lt_128, @lruncmp
  jmp @lrunend

@lruncmp
  %p4 =l add %src, %qj
  %c2 =w loadub %p4
  %eq2 =w ceql %c, %c2
  jnz %eq2, @lruninc
  jmp @lrunend

@lruninc
  %jr =w add %jr, 1
  %qj =l add %qj, 1
  jmp @lrunloop

@lrunend
  %jr_lt_3 =w csltl %jr, 3
  jnz %jr_lt_3, @linclude
  jmp @lscan_done

@linclude
  %j =l add %j, 1
  jmp @lscan

@lscan_done
  %litlen =l sub %j, %i
  %litpacket =l add %litlen, 1
  %limit2 =l add %written, %litpacket
  %too_big2 =w csltl %cap, %limit2
  jnz %too_big2, @fail
  %header2 =w sub %litlen, 1
  %hptr =l add %outp, 0
  storeb %header2, %hptr
  %k =l add 0, 0
  jmp @litcopy

@litcopy
  %k_lt_len =w csltl %k, %litlen
  jnz %k_lt_len, @litcopy_body
  jmp @literal_done

@litcopy_body
  %srck =l add %src, %i
  %srck2 =l add %srck, %k
  %srcbyte =w loadub %srck2
  %dstp =l add %hptr, 1
  %dstp2 =l add %dstp, %k
  storeb %srcbyte, %dstp2
  %k =l add %k, 1
  jmp @litcopy

@literal_done
  %outp =l add %hptr, %litpacket
  %written =l add %written, %litpacket
  %i =l add %j, 0
  jmp @loop

@done
  ret %written

@fail
  ret -1
}
