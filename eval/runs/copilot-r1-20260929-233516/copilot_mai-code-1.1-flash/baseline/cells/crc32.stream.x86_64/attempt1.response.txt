    mov r8d, eax
    test rsi, rsi
    jz .Ldone
    mov r10, rdi
.Lloop:
    movzx edx, byte ptr [r10]
    xor r8d, edx
    mov ecx, 8
.Lbit:
    test r8d, 1
    jz .Lno
    shr r8d, 1
    xor r8d, 0xEDB88320
    jmp .Lcont
.Lno:
    shr r8d, 1
.Lcont:
    dec ecx
    jnz .Lbit
    inc r10
    dec rsi
    jnz .Lloop
.Ldone:
