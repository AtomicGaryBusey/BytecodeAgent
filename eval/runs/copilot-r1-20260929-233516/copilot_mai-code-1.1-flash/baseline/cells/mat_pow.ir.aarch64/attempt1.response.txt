export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %nn = mull %n, %n
    %i = copy $0
    jmp @init_i

@init_i
    %icond = ceql %i, %n
    jnz %icond, @after_init
    %j = copy $0
    jmp @init_j

@init_j
    %jcond = ceql %j, %n
    jnz %jcond, @next_init_i
    %idx = addl %i, %j
    %r = mull %i, %n
    %row_idx = addl %r, %j
    %off = mull %row_idx, $8
    %ptr = addl %out, %off
    %isdiag = ceql %i, %j
    jnz %isdiag, @set_one
    storel $0, %ptr
    jmp @after_set

@set_one
    storel $1, %ptr

@after_set
    %j = addl %j, $1
    jmp @init_j

@next_init_i
    %i = addl %i, $1
    jmp @init_i

@after_init
    jmp @matpow_loop

@matpow_loop
    %kzero = ceql %k, $0
    jnz %kzero, @zero_tmp
    %i = copy $0
    jmp @mul_i

@mul_i
    %i_end = ceql %i, %n
    jnz %i_end, @copy_out
    %j = copy $0
    jmp @mul_j

@mul_j
    %j_end = ceql %j, %n
    jnz %j_end, @next_mul_i
    %t = copy $0
    %sum = copy $0
    jmp @mul_t

@mul_t
    %t_end = ceql %t, %n
    jnz %t_end, @store_mul_cell
    %left_row = mull %i, %n
    %left_idx = addl %left_row, %t
    %left_off = mull %left_idx, $8
    %left_ptr = addl %out, %left_off
    %left_val = loadl %left_ptr
    %right_row = mull %t, %n
    %right_idx = addl %right_row, %j
    %right_off = mull %right_idx, $8
    %right_ptr = addl %a, %right_off
    %right_val = loadl %right_ptr
    %prod = mull %left_val, %right_val
    %sum = addl %sum, %prod
    %t = addl %t, $1
    jmp @mul_t

@store_mul_cell
    %dst_row = mull %i, %n
    %dst_idx = addl %dst_row, %j
    %dst_off = mull %dst_idx, $8
    %dst_ptr = addl %tmp, %dst_off
    storel %sum, %dst_ptr
    %j = addl %j, $1
    jmp @mul_j

@next_mul_i
    %i = addl %i, $1
    jmp @mul_i

@copy_out
    %c = copy $0
    jmp @copy_loop

@copy_loop
    %c_end = ceql %c, %nn
    jnz %c_end, @after_copy
    %c_off = mull %c, $8
    %src_ptr = addl %tmp, %c_off
    %dst_ptr = addl %out, %c_off
    %val = loadl %src_ptr
    storel %val, %dst_ptr
    %c = addl %c, $1
    jmp @copy_loop

@after_copy
    %k = subl %k, $1
    jmp @matpow_loop

@zero_tmp
    %zero_n = mull %n, %n
    %zero_count = addl %zero_n, %zero_n
    %z = copy $0
    jmp @zero_loop

@zero_loop
    %z_end = ceql %z, %zero_count
    jnz %z_end, @done
    %z_off = mull %z, $8
    %z_ptr = addl %tmp, %z_off
    storel $0, %z_ptr
    %z = addl %z, $1
    jmp @zero_loop

@done
    ret l $0
}
