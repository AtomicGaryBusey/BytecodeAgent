export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %nn = mull %n, %n
    %zero_count = addl %nn, %nn
    %z = copy $0
    jmp @zero_tmp_all

@zero_tmp_all
    %z_end = ceql %z, %zero_count
    jnz %z_end, @init_out
    %zoff = mull %z, $8
    %zptr = addl %tmp, %zoff
    storel $0, %zptr
    %z = addl %z, $1
    jmp @zero_tmp_all

@init_out
    %i = copy $0
    jmp @init_i

@init_i
    %i_end = ceql %i, %n
    jnz %i_end, @check_k
    %j = copy $0
    jmp @init_j

@init_j
    %j_end = ceql %j, %n
    jnz %j_end, @next_i
    %row = mull %i, %n
    %idx = addl %row, %j
    %off = mull %idx, $8
    %ptr = addl %out, %off
    %isdiag = ceql %i, %j
    jnz %isdiag, @set_one
    storel $0, %ptr
    jmp @after_set

@set_one
    storel $1, %ptr
    jmp @after_set

@after_set
    %j = addl %j, $1
    jmp @init_j

@next_i
    %i = addl %i, $1
    jmp @init_i

@check_k
    %kzero = ceql %k, $0
    jnz %kzero, @final_zero
    %p = copy $0
    jmp @pow_loop

@pow_loop
    %p_end = ceql %p, %k
    jnz %p_end, @final_zero
    %i2 = copy $0
    jmp @mul_i

@mul_i
    %i2_end = ceql %i2, %n
    jnz %i2_end, @copy_back
    %j2 = copy $0
    jmp @mul_j

@mul_j
    %j2_end = ceql %j2, %n
    jnz %j2_end, @next_i2
    %sum = copy $0
    %t = copy $0
    jmp @mul_t

@mul_t
    %t_end = ceql %t, %n
    jnz %t_end, @store_cell
    %left_row = mull %i2, %n
    %left_idx = addl %left_row, %t
    %left_off = mull %left_idx, $8
    %left_ptr = addl %out, %left_off
    %left_val = loadl %left_ptr
    %right_row = mull %t, %n
    %right_idx = addl %right_row, %j2
    %right_off = mull %right_idx, $8
    %right_ptr = addl %a, %right_off
    %right_val = loadl %right_ptr
    %prod = mull %left_val, %right_val
    %sum = addl %sum, %prod
    %t = addl %t, $1
    jmp @mul_t

@store_cell
    %dst_row = mull %i2, %n
    %dst_idx = addl %dst_row, %j2
    %dst_off = mull %dst_idx, $8
    %dst_ptr = addl %tmp, %dst_off
    storel %sum, %dst_ptr
    %j2 = addl %j2, $1
    jmp @mul_j

@next_i2
    %i2 = addl %i2, $1
    jmp @mul_i

@copy_back
    %c = copy $0
    jmp @copy_loop

@copy_loop
    %c_end = ceql %c, %nn
    jnz %c_end, @advance_p
    %c8 = mull %c, $8
    %src_ptr = addl %tmp, %c8
    %dst_ptr = addl %out, %c8
    %val = loadl %src_ptr
    storel %val, %dst_ptr
    %c = addl %c, $1
    jmp @copy_loop

@advance_p
    %p = addl %p, $1
    jmp @pow_loop

@final_zero
    %z2 = copy $0
    jmp @clear_tmp

@clear_tmp
    %z2_end = ceql %z2, %zero_count
    jnz %z2_end, @done
    %z2off = mull %z2, $8
    %z2ptr = addl %tmp, %z2off
    storel $0, %z2ptr
    %z2 = addl %z2, $1
    jmp @clear_tmp

@done
    ret $0
}
