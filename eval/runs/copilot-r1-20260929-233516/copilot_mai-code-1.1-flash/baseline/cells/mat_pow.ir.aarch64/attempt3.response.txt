export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
    %nn = mul %n, %n
    %tmpcount = add %nn, %nn
    jmp @zero_tmp

@zero_tmp
    %z = phi @entry $0, @zero_tmp %z1
    %z_done = ceql %z, %tmpcount
    jnz %z_done, @init_row
    %zoff = mul %z, $8
    %zptr = add %tmp, %zoff
    storel $0, %zptr
    %z1 = add %z, $1
    jmp @zero_tmp

@init_row
    %i = phi @zero_tmp $0, @next_row %i1
    %i_done = ceql %i, %n
    jnz %i_done, @check_k
    jmp @init_col

@init_col
    %j = phi @init_row $0, @after_set %j1
    %j_done = ceql %j, %n
    jnz %j_done, @next_row
    %row = mul %i, %n
    %idx = add %row, %j
    %off = mul %idx, $8
    %ptr = add %out, %off
    %diag = ceql %i, %j
    jnz %diag, @set_one
    storel $0, %ptr
    jmp @after_set

@set_one
    storel $1, %ptr
    jmp @after_set

@after_set
    %j1 = add %j, $1
    jmp @init_col

@next_row
    %i1 = add %i, $1
    jmp @init_row

@check_k
    %kzero = ceql %k, $0
    jnz %kzero, @empty_k
    %p0 = copy $0
    jmp @pow_loop

@empty_k
    %q0 = copy $0
    jmp @clear_tmp

@pow_loop
    %p = phi @check_k %p0, @advance_p %p1
    %p_done = ceql %p, %k
    jnz %p_done, @after_pow
    jmp @mul_i

@mul_i
    %ii = phi @pow_loop $0, @next_i2 %ii1
    %ii_done = ceql %ii, %n
    jnz %ii_done, @copy_back
    jmp @mul_j

@mul_j
    %jj = phi @mul_i $0, @store_sum %jj1
    %jj_done = ceql %jj, %n
    jnz %jj_done, @next_i2
    jmp @mul_t

@mul_t
    %tt = phi @mul_j $0, @mul_t %tt1
    %sum = phi @mul_j $0, @mul_t %sum1
    %tt_done = ceql %tt, %n
    jnz %tt_done, @store_sum
    %left_row = mul %ii, %n
    %left_idx = add %left_row, %tt
    %left_off = mul %left_idx, $8
    %left_ptr = add %out, %left_off
    %left_val = loadl %left_ptr
    %right_row = mul %tt, %n
    %right_idx = add %right_row, %jj
    %right_off = mul %right_idx, $8
    %right_ptr = add %a, %right_off
    %right_val = loadl %right_ptr
    %prod = mul %left_val, %right_val
    %sum1 = add %sum, %prod
    %tt1 = add %tt, $1
    jmp @mul_t

@store_sum
    %dst_row = mul %ii, %n
    %dst_idx = add %dst_row, %jj
    %dst_off = mul %dst_idx, $8
    %dst_ptr = add %tmp, %dst_off
    storel %sum, %dst_ptr
    %jj1 = add %jj, $1
    jmp @mul_j

@next_i2
    %ii1 = add %ii, $1
    jmp @mul_i

@copy_back
    %c0 = copy $0
    jmp @copy_loop

@copy_loop
    %c = phi @copy_back %c0, @copy_loop %c1
    %c_done = ceql %c, %nn
    jnz %c_done, @advance_p
    %c8 = mul %c, $8
    %src_ptr = add %tmp, %c8
    %dst_ptr = add %out, %c8
    %val = loadl %src_ptr
    storel %val, %dst_ptr
    %c1 = add %c, $1
    jmp @copy_loop

@advance_p
    %p1 = add %p, $1
    jmp @pow_loop

@after_pow
    %q1 = copy $0
    jmp @clear_tmp

@clear_tmp
    %q = phi @empty_k %q0, @after_pow %q1, @clear_tmp %qnext
    %q_done = ceql %q, %tmpcount
    jnz %q_done, @done
    %qoff = mul %q, $8
    %qptr = add %tmp, %qoff
    storel $0, %qptr
    %qnext = add %q, $1
    jmp @clear_tmp

@done
    ret $0
}
