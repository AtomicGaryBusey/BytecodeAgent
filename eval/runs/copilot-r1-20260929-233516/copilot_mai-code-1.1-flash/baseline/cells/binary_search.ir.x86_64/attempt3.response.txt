export function l $binary_search(l %a, l %n, l %key) {
@entry
    %zero = copy 0
    %n_neg = csltl %n, %zero
    jnz %n_neg, @not_found, @check_zero

@check_zero
    %n_zero = ceql %n, %zero
    jnz %n_zero, @not_found, @init

@init
    %one = copy 1
    %low = copy 0
    %high = subl %n, %one
    jmp @loop

@loop
    %high_lt_low = csltl %high, %low
    jnz %high_lt_low, @not_found, @body

@body
    %sum = addl %low, %high
    %two = copy 2
    %mid = divl %sum, %two
    %eight = copy 8
    %mid8 = mull %mid, %eight
    %ptr = addl %a, %mid8
    %val = loadl %ptr
    %eq = ceql %val, %key
    jnz %eq, @found, @check_lt

@check_lt
    %lt = csltl %val, %key
    jnz %lt, @low_update, @high_update

@low_update
    %low = addl %mid, %one
    jmp @loop

@high_update
    %high = subl %mid, %one
    jmp @loop

@found
    ret %mid

@not_found
    %neg_one = copy -1
    ret %neg_one
}
