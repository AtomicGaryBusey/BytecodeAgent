export function l $binary_search(l %a, l %n, l %key) {
@entry
    %zero = copy 0
    %n_lt_zero = csltl %n, %zero
    jnz %n_lt_zero, @not_found, @check_zero

@check_zero
    %n_eq_zero = ceql %n, %zero
    jnz %n_eq_zero, @not_found, @init

@init
    %one = copy 1
    %low = copy 0
    %high = subl %n, %one
    jmp @loop

@loop
    %high_lt_low = csltl %high, %low
    jnz %high_lt_low, @not_found, @body

@body
    %sum = addl %low, %high
    %two = copy 2
    %mid = divl %sum, %two
    %eight = copy 8
    %mid8 = mull %mid, %eight
    %ptr = addl %a, %mid8
    %val = loadl %ptr
    %eq = ceql %val, %key
    jnz %eq, @found, @check_lt

@check_lt
    %lt = csltl %val, %key
    jnz %lt, @low_update, @check_gt

@check_gt
    %gt = csltl %key, %val
    jnz %gt, @high_update, @found

@low_update
    %mid_plus_one = addl %mid, %one
    %low = copy %mid_plus_one
    jmp @loop

@high_update
    %mid_minus_one = subl %mid, %one
    %high = copy %mid_minus_one
    jmp @loop

@found
    ret %mid

@not_found
    %neg_one = subl %zero, %one
    ret %neg_one
}
