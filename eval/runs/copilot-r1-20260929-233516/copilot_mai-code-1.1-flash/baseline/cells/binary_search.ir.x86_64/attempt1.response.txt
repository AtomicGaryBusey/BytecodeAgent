export function l $binary_search(l %a, l %n, l %key) {
@entry
    %is_zero = ceql %n, 0
    jnz %is_zero, @not_found, @init

@init
    %low = copy 0
    %high = subl %n, 1
    jmp @loop

@loop
    %high_less_low = csltl %high, %low
    jnz %high_less_low, @not_found, @body

@body
    %sum = addl %low, %high
    %mid = divl %sum, 2
    %mid8 = mull %mid, 8
    %ptr = addl %a, %mid8
    %val = loadl %ptr
    %eq = ceql %val, %key
    jnz %eq, @found, @check_lt

@check_lt
    %lt = csltl %val, %key
    jnz %lt, @low_update, @high_update

@low_update
    %low = addl %mid, 1
    jmp @loop

@high_update
    %high = subl %mid, 1
    jmp @loop

@found
    ret %mid

@not_found
    %neg_one = copy -1
    ret %neg_one
}
