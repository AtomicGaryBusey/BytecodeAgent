binary_search:
    cld
    test rsi, rsi
    jle .Lnotfound
    lea rcx, [rsi - 1]
    xor rsi, rsi
.Lloop:
    cmp rsi, rcx
    jg .Lnotfound
    mov rax, rcx
    sub rax, rsi
    shr rax, 1
    add rax, rsi
    mov r8, qword ptr [rdi + rax*8]
    cmp r8, rdx
    je .Lfound
    jl .Lmove_right
    lea rcx, [rax - 1]
    jmp .Lloop
.Lmove_right:
    lea rsi, [rax + 1]
    jmp .Lloop
.Lfound:
    ret
.Lnotfound:
    mov rax, -1
    ret
