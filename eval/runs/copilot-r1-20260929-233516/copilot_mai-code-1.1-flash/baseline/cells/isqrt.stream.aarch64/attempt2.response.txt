isqrt:
    mov x1, xzr
    mov x2, #1
    lsl x2, x2, #62
.Lbit_adjust:
    cmp x2, x0
    b.le .Lloop
    lsr x2, x2, #2
    b .Lbit_adjust
.Lloop:
    cbz x2, .Ldone
    add x3, x1, x2
    cmp x0, x3
    b.lo .Lelse
    sub x0, x0, x3
    lsr x1, x1, #1
    add x1, x1, x2
    b .Ladvance
.Lelse:
    lsr x1, x1, #1
.Ladvance:
    lsr x2, x2, #2
    b .Lloop
.Ldone:
    mov x0, x1
    ret
