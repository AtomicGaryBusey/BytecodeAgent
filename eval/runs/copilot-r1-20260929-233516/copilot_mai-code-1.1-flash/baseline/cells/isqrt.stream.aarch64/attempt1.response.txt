isqrt:
    mov w2, #65535
    orr w2, w2, w2, lsl #16
    mov x1, xzr
.Lloop:
    cmp x1, x2
    b.hs .Ldone
    add x3, x1, x2
    add x3, x3, #1
    lsr x3, x3, #1
    mul x4, x3, x3
    cmp x4, x0
    b.gt .Lhi_update
    mov x1, x3
    b .Lloop
.Lhi_update:
    sub x2, x3, #1
    b .Lloop
.Ldone:
    mov x0, x1
    ret
