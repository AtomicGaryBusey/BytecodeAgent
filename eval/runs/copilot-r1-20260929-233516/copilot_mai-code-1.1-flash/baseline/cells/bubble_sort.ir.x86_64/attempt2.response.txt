export function l $bubble_sort(l %a, l %n) {
@entry
  %i0 = copy 0
  jmp @outer_loop

@outer_loop
  %i = phi l [ %i0, @entry ], [ %i1, @outer_inc ]
  %i_lt_n = csltl %i, %n
  jnz %i_lt_n, @inner_init, @done

@inner_init
  %j0 = copy 0
  %n_minus_i = subl %n, %i
  %limit = subl %n_minus_i, 1
  jmp @inner_loop

@inner_loop
  %j = phi l [ %j0, @inner_init ], [ %j_inc, @inner_inc ]
  %j_lt_limit = csltl %j, %limit
  jnz %j_lt_limit, @compare, @outer_inc

@compare
  %j_mul_8 = mull %j, 8
  %j1 = addl %j, 1
  %j1_mul_8 = mull %j1, 8
  %p0 = addl %a, %j_mul_8
  %p1 = addl %a, %j1_mul_8
  %v0 = loadl %p0
  %v1 = loadl %p1
  %should_swap = csltl %v1, %v0
  jnz %should_swap, @swap, @inner_inc

@swap
  storel %p0, %v1
  storel %p1, %v0
  jmp @inner_inc

@inner_inc
  %j_inc = addl %j, 1
  jmp @inner_loop

@outer_inc
  %i1 = addl %i, 1
  jmp @outer_loop

@done
  ret 0
}
