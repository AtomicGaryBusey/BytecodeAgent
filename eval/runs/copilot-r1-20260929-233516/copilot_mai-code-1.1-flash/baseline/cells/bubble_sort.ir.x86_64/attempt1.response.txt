export function l $bubble_sort(l %a, l %n) {
@entry
  %i = copy 0
  jmp @outer_loop

@outer_loop
  %cond_i = csltl %i, %n
  jnz %cond_i, @inner_init, @done

@inner_init
  %j = copy 0
  %n_minus_i = subl %n, %i
  %limit = subl %n_minus_i, 1
  jmp @inner_loop

@inner_loop
  %cond_j = csltl %j, %limit
  jnz %cond_j, @compare, @outer_inc

@compare
  %j8 = mull %j, 8
  %j1_8 = addl %j8, 8
  %p0 = addl %a, %j8
  %p1 = addl %a, %j1_8
  %v0 = loadl %p0
  %v1 = loadl %p1
  %swap = csltl %v1, %v0
  jnz %swap, @swap, @inner_inc

@swap
  storel %p0, %v1
  storel %p1, %v0
  jmp @inner_inc

@inner_inc
  %j = addl %j, 1
  jmp @inner_loop

@outer_inc
  %i = addl %i, 1
  jmp @outer_loop

@done
  ret 0
}
