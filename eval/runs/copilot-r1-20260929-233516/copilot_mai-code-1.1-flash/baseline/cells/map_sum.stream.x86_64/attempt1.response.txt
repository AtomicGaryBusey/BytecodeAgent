map_sum:
    cld
    push r12
    push r13
    push r14
    push r15
    mov r12, rdx
    mov r13, rdi
    mov r14, rsi
    xor r15, r15
    test r14, r14
    jz .Ldone
.Lloop:
    sub rsp, 8
    mov rdi, qword ptr [r13]
    call r12
    add rsp, 8
    add r15, rax
    add r13, 8
    dec r14
    jnz .Lloop
.Ldone:
    mov rax, r15
    pop r15
    pop r14
    pop r13
    pop r12
    ret
