export function l $count_nonzero_prefix(l %a, l %n) {
@entry
  %i = copy 0
  %ptr = copy %a
  br @loop

@loop
  %cond = csltl %i, %n
  jnz %cond, @check, @done

@check
  %v = loadl %ptr
  %is_zero = ceql %v, 0
  jnz %is_zero, @done, @next

@next
  %i = addl %i, 1
  %ptr = addl %ptr, 8
  br @loop

@done
  ret %i
}
