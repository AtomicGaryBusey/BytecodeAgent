packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r9, rdx
    mov r12, rsi
    mov r13, rcx
    mov rbp, rdx
    xor r14, r14
.Lloop:
    cmp r14, r12
    jge .Ldone
    movzx r10d, byte ptr [rbx + r14]
    mov r11d, 1
    mov r15, r14
    inc r15
    cmp r15, r12
    jge .Lrun_done
    mov dl, byte ptr [rbx + r15]
    cmp dl, r10b
    jne .Lrun_done
    mov r11d, 2
    inc r15
    cmp r15, r12
    jge .Lrun_done
    mov dl, byte ptr [rbx + r15]
    cmp dl, r10b
    jne .Lrun_done
    mov r11d, 3
.Lrun_loop:
    cmp r11d, 128
    jae .Lrun_done
    inc r15
    cmp r15, r12
    jge .Lrun_done
    mov dl, byte ptr [rbx + r15]
    cmp dl, r10b
    jne .Lrun_done
    inc r11d
    jmp .Lrun_loop
.Lrun_done:
    cmp r11d, 3
    jb .Lliteral
.repeat:
    cmp r13, 2
    jb .Lfail
    mov rax, 257
    sub rax, r11
    mov byte ptr [rbp], al
    inc rbp
    mov byte ptr [rbp], r10b
    inc rbp
    sub r13, 2
    add r14, r11
    jmp .Lloop
.Lliteral:
    mov r15, r14
.Llit_top:
    cmp r15, r12
    jae .Llit_end
    mov rax, r15
    sub rax, r14
    cmp rax, 128
    jae .Llit_end
    mov r11d, 1
    mov rcx, r15
    inc rcx
    cmp rcx, r12
    jge .Llit_run_done
    mov dl, byte ptr [rbx + rcx]
    mov sil, byte ptr [rbx + r15]
    cmp dl, sil
    jne .Llit_run_done
    mov r11d, 2
    inc rcx
    cmp rcx, r12
    jge .Llit_run_done
    mov dl, byte ptr [rbx + rcx]
    cmp dl, sil
    jne .Llit_run_done
    mov r11d, 3
.Llit_run_loop:
    cmp r11d, 128
    jae .Llit_run_done
    inc rcx
    cmp rcx, r12
    jge .Llit_run_done
    mov dl, byte ptr [rbx + rcx]
    cmp dl, sil
    jne .Llit_run_done
    inc r11d
    jmp .Llit_run_loop
.Llit_run_done:
    cmp r11d, 3
    jb .Llit_advance
    jmp .Llit_end
.Llit_advance:
    inc r15
    jmp .Llit_top
.Llit_end:
    mov r11, r15
    sub r11, r14
    mov rax, r11
    inc rax
    cmp r13, rax
    jb .Lfail
    mov rax, r11
    dec rax
    mov byte ptr [rbp], al
    inc rbp
    mov rsi, rbx
    add rsi, r14
    mov rdi, rbp
    mov rcx, r11
    rep movsb
    add rbp, r11
    mov rax, r11
    inc rax
    sub r13, rax
    mov rax, r15
    mov r14, rax
    jmp .Lloop
.Ldone:
    mov rax, rbp
    sub rax, r9
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lfail:
    mov rax, -1
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
