export function l $count_ugt(l %a, l %n, l %t) {
@entry
  %i =l copy 0
  %count =l copy 0
  %p =l copy %a
  %is_zero =w ceql %n, 0
  jnz %is_zero, @done, @loop

@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @done

@body
  %v =l loadl %p
  %gt =w cugtl %v, %t
  jnz %gt, @inc, @skip

@inc
  %count =l add %count, 1

@skip
  %p =l add %p, 8
  %i =l add %i, 1
  jmp @loop

@done
  ret %count
}
