export function l $count_ugt(l %a, l %n, l %t) {
@entry
  %i =l copy 0
  %count =l copy 0
  %ptr =l copy %a
  %is_zero =w ceql %n, 0
  jnz %is_zero, @done, @loop
@loop
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %v =l loadl %ptr
  %gt =w cugtl %v, %t
  jnz %gt, @inc, @skip
@inc
  %count =l addl %count, 1
@skip
  %ptr =l addl %ptr, 8
  %i =l addl %i, 1
  jmp @loop
@done
  ret %count
}
