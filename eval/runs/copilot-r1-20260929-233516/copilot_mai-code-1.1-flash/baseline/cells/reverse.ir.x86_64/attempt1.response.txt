export function l $reverse(l %a, l %n) {
@start
  %iszero = ceql %n, 0
  jnz %iszero, @done
  %i = copy 0
  %j = subl %n, 1
@loop
  %cond = csltl %i, %j
  jnz %cond, @swap
  jmp @done
@swap
  %i8 = mull %i, 8
  %j8 = mull %j, 8
  %pi = addl %a, %i8
  %pj = addl %a, %j8
  %vi = loadl %pi
  %vj = loadl %pj
  storel %vj, %pi
  storel %vi, %pj
  %i1 = addl %i, 1
  %j1 = subl %j, 1
  %i = copy %i1
  %j = copy %j1
  jmp @loop
@done
  ret 0
}
