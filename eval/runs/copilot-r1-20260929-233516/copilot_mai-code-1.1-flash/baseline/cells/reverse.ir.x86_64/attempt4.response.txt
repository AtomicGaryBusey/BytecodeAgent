export function l $reverse(l %a, l %n) {
@start
  %zero = copy 0
  %one = copy 1
  %i = copy %zero
  %j = subl %n, %one
@loop
  %cond = csltl %i, %j
  jnz %cond, @swap
  jmp @done
@swap
  %i8 = mull %i, 8
  %j8 = mull %j, 8
  %pi = addl %a, %i8
  %pj = addl %a, %j8
  %vi = loadl %pi
  %vj = loadl %pj
  storel %vj, %pi
  storel %vi, %pj
  %i1 = addl %i, %one
  %j1 = subl %j, %one
  %i = copy %i1
  %j = copy %j1
  jmp @loop
@done
  ret %zero
}
