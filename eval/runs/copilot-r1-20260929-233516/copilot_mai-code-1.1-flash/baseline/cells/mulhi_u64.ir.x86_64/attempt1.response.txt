export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 = andl %a, 4294967295
    %a1 = shrl %a, 32
    %b0 = andl %b, 4294967295
    %b1 = shrl %b, 32

    %p00 = mull %a0, %b0
    %p01 = mull %a0, %b1
    %p10 = mull %a1, %b0
    %p11 = mull %a1, %b1

    %s = addl %p01, %p10
    %low32_s = andl %s, 4294967295

    %t0 = andl %p01, %p10
    %t1 = xorl %p01, %p10
    %t2 = andl %t1, %s
    %carry_s_raw = orl %t0, %t2
    %carry_s = shrl %carry_s_raw, 63

    %lo_shift = shll %low32_s, 32
    %low_sum = addl %p00, %lo_shift

    %u0 = andl %p00, %lo_shift
    %u1 = xorl %p00, %lo_shift
    %u2 = andl %u1, %low_sum
    %carry_lo_raw = orl %u0, %u2
    %carry_lo = shrl %carry_lo_raw, 63

    %s_hi = shrl %s, 32
    %carry_s_shift = shll %carry_s, 32

    %h1 = addl %p11, %s_hi
    %h2 = addl %h1, %carry_s_shift
    %h3 = addl %h2, %carry_lo

    ret %h3
}
