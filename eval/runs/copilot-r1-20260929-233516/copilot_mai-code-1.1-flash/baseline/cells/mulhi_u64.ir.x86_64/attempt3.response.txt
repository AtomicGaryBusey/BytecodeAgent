export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 = andl %a, 4294967295
    %a1 = shrl %a, 32
    %b0 = andl %b, 4294967295
    %b1 = shrl %b, 32

    %p00 = mull %a0, %b0
    %p01 = mull %a0, %b1
    %p10 = mull %a1, %b0
    %p11 = mull %a1, %b1

    %p01_lo = andl %p01, 4294967295
    %p01_hi = shrl %p01, 32
    %p10_lo = andl %p10, 4294967295
    %p10_hi = shrl %p10, 32

    %low32_sum = addl %p01_lo, %p10_lo
    %mid_lo = andl %low32_sum, 4294967295
    %low32_carry = shrl %low32_sum, 32

    %mid_hi = addl %p01_hi, %p10_hi
    %mid_hi2 = addl %mid_hi, %low32_carry

    %p00_hi = shrl %p00, 32
    %sum_hi = addl %p00_hi, %mid_lo
    %carry = shrl %sum_hi, 32

    %result = addl %p11, %mid_hi2
    %result2 = addl %result, %carry
    ret %result2
}
