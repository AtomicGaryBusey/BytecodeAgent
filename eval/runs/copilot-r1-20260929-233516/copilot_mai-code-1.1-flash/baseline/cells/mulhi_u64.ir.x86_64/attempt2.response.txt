export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 = and %a, 4294967295
    %a1 = shr %a, 32
    %b0 = and %b, 4294967295
    %b1 = shr %b, 32

    %p00 = mul %a0, %b0
    %p01 = mul %a0, %b1
    %p10 = mul %a1, %b0
    %p11 = mul %a1, %b1

    %mid = add %p01, %p10
    %mid_lo = and %mid, 4294967295
    %mid_hi = shr %mid, 32

    %lo_hi = shr %p00, 32
    %sum_hi = add %lo_hi, %mid_lo
    %carry_lo = shr %sum_hi, 32

    %h = add %p11, %mid_hi
    %h2 = add %h, %carry_lo
    ret %h2
}
