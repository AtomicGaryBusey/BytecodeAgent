export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
  %n_sq = mul %n, %n
  %i = 0
@init_i
  %i_lt = csltl %i, %n
  jnz %i_lt, @init_i_body
  jmp @copy_base
@init_i_body
  %j = 0
@init_j
  %j_lt = csltl %j, %n
  jnz %j_lt, @init_j_body
  jmp @init_i_next
@init_j_body
  %idx = mul %i, %n
  %idx = add %idx, %j
  %idx = mul %idx, 8
  %ptr = add %out, %idx
  %diag = ceql %i, %j
  jnz %diag, @set_one
  storel 0, %ptr
  jmp @after_set
@set_one
  storel 1, %ptr
@after_set
  %j = add %j, 1
  jmp @init_j
@init_i_next
  %i = add %i, 1
  jmp @init_i

@copy_base
  %i = 0
@copy_base_i
  %i_lt = csltl %i, %n
  jnz %i_lt, @copy_base_i_body
  jmp @pow_loop
@copy_base_i_body
  %j = 0
@copy_base_j
  %j_lt = csltl %j, %n
  jnz %j_lt, @copy_base_j_body
  jmp @copy_base_i_next
@copy_base_j_body
  %idx = mul %i, %n
  %idx = add %idx, %j
  %idx = mul %idx, 8
  %src = add %a, %idx
  %dst = add %tmp, %idx
  %val = loadl %src
  storel %val, %dst
  %j = add %j, 1
  jmp @copy_base_j
@copy_base_i_next
  %i = add %i, 1
  jmp @copy_base_i

@pow_loop
  %k_pos = csltl 0, %k
  jnz %k_pos, @pow_body
  jmp @zero_tmp
@pow_body
  %prod_off = mul %n_sq, 8
  %prod_base = add %tmp, %prod_off
  %i = 0
@mul_i
  %i_lt = csltl %i, %n
  jnz %i_lt, @mul_i_body
  jmp @copy_result_back
@mul_i_body
  %j = 0
@mul_j
  %j_lt = csltl %j, %n
  jnz %j_lt, @mul_j_body
  jmp @mul_i_inc
@mul_j_body
  %sum = 0
  %t = 0
@mul_t
  %t_lt = csltl %t, %n
  jnz %t_lt, @mul_t_body
  jmp @mul_store
@mul_t_body
  %lhs_idx = mul %i, %n
  %lhs_idx = add %lhs_idx, %t
  %lhs_idx = mul %lhs_idx, 8
  %lhs_ptr = add %out, %lhs_idx
  %lhs = loadl %lhs_ptr
  %rhs_idx = mul %t, %n
  %rhs_idx = add %rhs_idx, %j
  %rhs_idx = mul %rhs_idx, 8
  %rhs_ptr = add %tmp, %rhs_idx
  %rhs = loadl %rhs_ptr
  %prod = mul %lhs, %rhs
  %sum = add %sum, %prod
  %t = add %t, 1
  jmp @mul_t
@mul_store
  %dst_idx = mul %i, %n
  %dst_idx = add %dst_idx, %j
  %dst_idx = mul %dst_idx, 8
  %dst_ptr = add %prod_base, %dst_idx
  storel %sum, %dst_ptr
  %j = add %j, 1
  jmp @mul_j
@mul_i_inc
  %i = add %i, 1
  jmp @mul_i
@copy_result_back
  %i = 0
@copy_back_i
  %i_lt = csltl %i, %n
  jnz %i_lt, @copy_back_i_body
  jmp @dec_k
@copy_back_i_body
  %j = 0
@copy_back_j
  %j_lt = csltl %j, %n
  jnz %j_lt, @copy_back_j_body
  jmp @copy_back_i_inc
@copy_back_j_body
  %src_idx = mul %i, %n
  %src_idx = add %src_idx, %j
  %src_idx = mul %src_idx, 8
  %src_ptr = add %prod_base, %src_idx
  %val = loadl %src_ptr
  %dst_ptr = add %out, %src_idx
  storel %val, %dst_ptr
  %j = add %j, 1
  jmp @copy_back_j
@copy_back_i_inc
  %i = add %i, 1
  jmp @copy_back_i
@dec_k
  %k = sub %k, 1
  jmp @pow_loop

@zero_tmp
  %total = mul %n_sq, 2
  %p = 0
@zero_loop
  %p_lt = csltl %p, %total
  jnz %p_lt, @zero_body
  jmp @return
@zero_body
  %off = mul %p, 8
  %ptr = add %tmp, %off
  storel 0, %ptr
  %p = add %p, 1
  jmp @zero_loop
@return
  ret 0
}
