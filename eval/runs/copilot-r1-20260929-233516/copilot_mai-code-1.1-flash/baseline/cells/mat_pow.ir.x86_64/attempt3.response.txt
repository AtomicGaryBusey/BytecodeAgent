export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
  %n_sq = mul %n, %n
  %i = 0
@init_i
  %cond_i = csltl %i, %n
  jnz %cond_i, @init_i_body
  jmp @copy_base
@init_i_body
  %j = 0
@init_j
  %cond_j = csltl %j, %n
  jnz %cond_j, @init_j_body
  jmp @init_i_next
@init_j_body
  %idx = mul %i, %n
  %idx = add %idx, %j
  %idx = mul %idx, 8
  %ptr = add %out, %idx
  %diag = ceql %i, %j
  jnz %diag, @init_one
  storel 0, %ptr
  jmp @init_j_next
@init_one
  storel 1, %ptr
@init_j_next
  %j = add %j, 1
  jmp @init_j
@init_i_next
  %i = add %i, 1
  jmp @init_i
@copy_base
  %prod_off = mul %n_sq, 8
  %prod_base = add %tmp, %prod_off
  %bi = 0
@copy_base_i
  %bic = csltl %bi, %n
  jnz %bic, @copy_base_i_body
  jmp @pow_loop
@copy_base_i_body
  %bj = 0
@copy_base_j
  %bjc = csltl %bj, %n
  jnz %bjc, @copy_base_j_body
  jmp @copy_base_i_next
@copy_base_j_body
  %idx = mul %bi, %n
  %idx = add %idx, %bj
  %idx = mul %idx, 8
  %src = add %a, %idx
  %dst = add %tmp, %idx
  %val = loadl %src
  storel %val, %dst
  %bj = add %bj, 1
  jmp @copy_base_j
@copy_base_i_next
  %bi = add %bi, 1
  jmp @copy_base_i
@pow_loop
  %k_pos = csltl 0, %k
  jnz %k_pos, @pow_body
  jmp @zero_tmp
@pow_body
  %bit = and %k, 1
  %odd = ceql %bit, 1
  jnz %odd, @mul_result
  jmp @square_base
@mul_result
  %ii = 0
@mr_i
  %ii_lt = csltl %ii, %n
  jnz %ii_lt, @mr_i_body
  jmp @copy_result_back
@mr_i_body
  %jj = 0
@mr_j
  %jj_lt = csltl %jj, %n
  jnz %jj_lt, @mr_j_body
  jmp @mr_i_inc
@mr_j_body
  %sum = 0
  %tt = 0
@mr_t
  %tt_lt = csltl %tt, %n
  jnz %tt_lt, @mr_t_body
  jmp @mr_store
@mr_t_body
  %lhs_i = mul %ii, %n
  %lhs_i = add %lhs_i, %tt
  %lhs_i = mul %lhs_i, 8
  %lhs_p = add %out, %lhs_i
  %lhs = loadl %lhs_p
  %rhs_i = mul %tt, %n
  %rhs_i = add %rhs_i, %jj
  %rhs_i = mul %rhs_i, 8
  %rhs_p = add %tmp, %rhs_i
  %rhs = loadl %rhs_p
  %prod = mul %lhs, %rhs
  %sum = add %sum, %prod
  %tt = add %tt, 1
  jmp @mr_t
@mr_store
  %dst_idx = mul %ii, %n
  %dst_idx = add %dst_idx, %jj
  %dst_idx = mul %dst_idx, 8
  %dst_p = add %prod_base, %dst_idx
  storel %sum, %dst_p
  %jj = add %jj, 1
  jmp @mr_j
@mr_i_inc
  %ii = add %ii, 1
  jmp @mr_i
@copy_result_back
  %ci = 0
@cr_i
  %ci_lt = csltl %ci, %n
  jnz %ci_lt, @cr_i_body
  jmp @square_base
@cr_i_body
  %cj = 0
@cr_j
  %cj_lt = csltl %cj, %n
  jnz %cj_lt, @cr_j_body
  jmp @cr_i_inc
@cr_j_body
  %src_idx = mul %ci, %n
  %src_idx = add %src_idx, %cj
  %src_idx = mul %src_idx, 8
  %src_p = add %prod_base, %src_idx
  %val = loadl %src_p
  %dst_p = add %out, %src_idx
  storel %val, %dst_p
  %cj = add %cj, 1
  jmp @cr_j
@cr_i_inc
  %ci = add %ci, 1
  jmp @cr_i
@square_base
  %ii = 0
@sb_i
  %ii_lt = csltl %ii, %n
  jnz %ii_lt, @sb_i_body
  jmp @copy_base_back
@sb_i_body
  %jj = 0
@sb_j
  %jj_lt = csltl %jj, %n
  jnz %jj_lt, @sb_j_body
  jmp @sb_i_inc
@sb_j_body
  %sum = 0
  %tt = 0
@sb_t
  %tt_lt = csltl %tt, %n
  jnz %tt_lt, @sb_t_body
  jmp @sb_store
@sb_t_body
  %lhs_i = mul %ii, %n
  %lhs_i = add %lhs_i, %tt
  %lhs_i = mul %lhs_i, 8
  %lhs_p = add %tmp, %lhs_i
  %lhs = loadl %lhs_p
  %rhs_i = mul %tt, %n
  %rhs_i = add %rhs_i, %jj
  %rhs_i = mul %rhs_i, 8
  %rhs_p = add %tmp, %rhs_i
  %rhs = loadl %rhs_p
  %prod = mul %lhs, %rhs
  %sum = add %sum, %prod
  %tt = add %tt, 1
  jmp @sb_t
@sb_store
  %dst_idx = mul %ii, %n
  %dst_idx = add %dst_idx, %jj
  %dst_idx = mul %dst_idx, 8
  %dst_p = add %prod_base, %dst_idx
  storel %sum, %dst_p
  %jj = add %jj, 1
  jmp @sb_j
@sb_i_inc
  %ii = add %ii, 1
  jmp @sb_i
@copy_base_back
  %ci = 0
@cbb_i
  %ci_lt = csltl %ci, %n
  jnz %ci_lt, @cbb_i_body
  jmp @shift_k
@cbb_i_body
  %cj = 0
@cbb_j
  %cj_lt = csltl %cj, %n
  jnz %cj_lt, @cbb_j_body
  jmp @cbb_i_inc
@cbb_j_body
  %src_idx = mul %ci, %n
  %src_idx = add %src_idx, %cj
  %src_idx = mul %src_idx, 8
  %src_p = add %prod_base, %src_idx
  %val = loadl %src_p
  %dst_p = add %tmp, %src_idx
  storel %val, %dst_p
  %cj = add %cj, 1
  jmp @cbb_j
@cbb_i_inc
  %ci = add %ci, 1
  jmp @cbb_i
@shift_k
  %k = shr %k, 1
  jmp @pow_loop
@zero_tmp
  %total = add %n_sq, %n_sq
  %p = 0
@zero_loop
  %cond_p = csltl %p, %total
  jnz %cond_p, @zero_body
  jmp @return
@zero_body
  %zidx = mul %p, 8
  %zptr = add %tmp, %zidx
  storel 0, %zptr
  %p = add %p, 1
  jmp @zero_loop
@return
  ret 0
}
