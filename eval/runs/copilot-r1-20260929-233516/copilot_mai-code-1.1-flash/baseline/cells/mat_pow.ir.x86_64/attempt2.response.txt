export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
  %i = addl 0, 0
@init_i_loop
  %cond_i = csltl %i, %n
  jnz %cond_i, @init_i_body
  jmp @init_done
@init_i_body
  %j = addl 0, 0
@init_j_loop
  %cond_j = csltl %j, %n
  jnz %cond_j, @init_j_body
  jmp @init_i_inc
@init_j_body
  %idx = mull %i, %n
  %idx = addl %idx, %j
  %idx = mull %idx, 8
  %ptr = addl %out, %idx
  %is_diag = ceql %i, %j
  jnz %is_diag, @init_diag
  storel 0, %ptr
  jmp @init_j_next
@init_diag
  storel 1, %ptr
@init_j_next
  %j = addl %j, 1
  jmp @init_j_loop
@init_i_inc
  %i = addl %i, 1
  jmp @init_i_loop
@init_done
  %rem = addl %k, 0
@pow_loop
  %keep = csltl 0, %rem
  jnz %keep, @mul_step
  jmp @zero_tmp
@mul_step
  %ii = addl 0, 0
@mul_i_loop
  %cond_ii = csltl %ii, %n
  jnz %cond_ii, @mul_i_body
  jmp @copy_back
@mul_i_body
  %jj = addl 0, 0
@mul_j_loop
  %cond_jj = csltl %jj, %n
  jnz %cond_jj, @mul_j_body
  jmp @mul_i_inc
@mul_j_body
  %sum = addl 0, 0
  %tt = addl 0, 0
@mul_t_loop
  %cond_tt = csltl %tt, %n
  jnz %cond_tt, @mul_t_body
  jmp @store_sum
@mul_t_body
  %lhs_idx = mull %ii, %n
  %lhs_idx = addl %lhs_idx, %tt
  %lhs_idx = mull %lhs_idx, 8
  %lhs_ptr = addl %out, %lhs_idx
  %lhs = loadl %lhs_ptr

  %rhs_idx = mull %tt, %n
  %rhs_idx = addl %rhs_idx, %jj
  %rhs_idx = mull %rhs_idx, 8
  %rhs_ptr = addl %a, %rhs_idx
  %rhs = loadl %rhs_ptr

  %prod = mull %lhs, %rhs
  %sum = addl %sum, %prod
  %tt = addl %tt, 1
  jmp @mul_t_loop
@store_sum
  %dst_idx = mull %ii, %n
  %dst_idx = addl %dst_idx, %jj
  %dst_idx = mull %dst_idx, 8
  %dst_ptr = addl %tmp, %dst_idx
  storel %sum, %dst_ptr
  %jj = addl %jj, 1
  jmp @mul_j_loop
@mul_i_inc
  %ii = addl %ii, 1
  jmp @mul_i_loop
@copy_back
  %ci = addl 0, 0
@copy_i_loop
  %cond_ci = csltl %ci, %n
  jnz %cond_ci, @copy_i_body
  jmp @after_copy
@copy_i_body
  %cj = addl 0, 0
@copy_j_loop
  %cond_cj = csltl %cj, %n
  jnz %cond_cj, @copy_j_body
  jmp @copy_i_inc
@copy_j_body
  %src_idx = mull %ci, %n
  %src_idx = addl %src_idx, %cj
  %src_idx = mull %src_idx, 8
  %src_ptr = addl %tmp, %src_idx
  %val = loadl %src_ptr
  %dst_ptr = addl %out, %src_idx
  storel %val, %dst_ptr
  %cj = addl %cj, 1
  jmp @copy_j_loop
@copy_i_inc
  %ci = addl %ci, 1
  jmp @copy_i_loop
@after_copy
  %rem = subl %rem, 1
  jmp @pow_loop
@zero_tmp
  %total = mull %n, %n
  %total = addl %total, %total
  %p = addl 0, 0
@zero_loop
  %cond_p = csltl %p, %total
  jnz %cond_p, @zero_body
  jmp @return
@zero_body
  %zidx = mull %p, 8
  %zptr = addl %tmp, %zidx
  storel 0, %zptr
  %p = addl %p, 1
  jmp @zero_loop
@return
  ret 0
}
