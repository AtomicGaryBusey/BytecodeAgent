export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
  %i = copy 0
@id_i_loop
  %c_i = csltl %i, %n
  jnz %c_i, @id_i_body
  jmp @id_done
@id_i_body
  %j = copy 0
@id_j_loop
  %c_j = csltl %j, %n
  jnz %c_j, @id_j_body
  jmp @id_i_inc
@id_j_body
  %val = copy 0
  %d = ceql %i, %j
  jnz %d, @id_diag
  jmp @id_store
@id_diag
  %val = copy 1
@id_store
  %idx = mull %i, %n
  %idx = addl %idx, %j
  %idx = mull %idx, 8
  %ptr = addl %out, %idx
  storel %val, %ptr
  %j = addl %j, 1
  jmp @id_j_loop
@id_i_inc
  %i = addl %i, 1
  jmp @id_i_loop
@id_done
  %m = copy 1
@pow_loop
  %keep = csleql %m, %k
  jnz %keep, @mul_step
  jmp @zero_tmp
@mul_step
  %ii = copy 0
@mul_i_loop
  %c_ii = csltl %ii, %n
  jnz %c_ii, @mul_i_body
  jmp @copy_back
@mul_i_body
  %jj = copy 0
@mul_j_loop
  %c_jj = csltl %jj, %n
  jnz %c_jj, @mul_j_body
  jmp @mul_i_inc
@mul_j_body
  %sum = copy 0
  %tt = copy 0
@mul_t_loop
  %c_tt = csltl %tt, %n
  jnz %c_tt, @mul_t_body
  jmp @store_cell
@mul_t_body
  %li = mull %ii, %n
  %li = addl %li, %tt
  %li = mull %li, 8
  %lptr = addl %out, %li
  %lhs = loadl %lptr
  %ri = mull %tt, %n
  %ri = addl %ri, %jj
  %ri = mull %ri, 8
  %rptr = addl %a, %ri
  %rhs = loadl %rptr
  %prod = mull %lhs, %rhs
  %sum = addl %sum, %prod
  %tt = addl %tt, 1
  jmp @mul_t_loop
@store_cell
  %dst_idx = mull %ii, %n
  %dst_idx = addl %dst_idx, %jj
  %dst_idx = mull %dst_idx, 8
  %dptr = addl %tmp, %dst_idx
  storel %sum, %dptr
  %jj = addl %jj, 1
  jmp @mul_j_loop
@mul_i_inc
  %ii = addl %ii, 1
  jmp @mul_i_loop
@copy_back
  %ci = copy 0
@copy_i_loop
  %c_ci = csltl %ci, %n
  jnz %c_ci, @copy_i_body
  jmp @after_copy
@copy_i_body
  %cj = copy 0
@copy_j_loop
  %c_cj = csltl %cj, %n
  jnz %c_cj, @copy_j_body
  jmp @copy_i_inc
@copy_j_body
  %sidx = mull %ci, %n
  %sidx = addl %sidx, %cj
  %sidx = mull %sidx, 8
  %sptr = addl %tmp, %sidx
  %v = loadl %sptr
  %dptr2 = addl %out, %sidx
  storel %v, %dptr2
  %cj = addl %cj, 1
  jmp @copy_j_loop
@copy_i_inc
  %ci = addl %ci, 1
  jmp @copy_i_loop
@after_copy
  %m = addl %m, 1
  jmp @pow_loop
@zero_tmp
  %total = mull %n, %n
  %total = addl %total, %total
  %p = copy 0
@zero_loop
  %c_p = csltl %p, %total
  jnz %c_p, @zero_body
  jmp @return
@zero_body

  %zptr = addl %tmp, %zidx
  storel 0, %zptr
  %p = addl %p, 1
  jmp @zero_loop
@return
  ret 0
}
