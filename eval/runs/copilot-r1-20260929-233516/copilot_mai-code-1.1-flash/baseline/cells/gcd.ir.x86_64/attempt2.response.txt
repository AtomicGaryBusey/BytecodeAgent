export function l $gcd(l %a, l %b) {
@entry
  %base = alloc8
  %bptr = alloc8
  storel %a, %base
  storel %b, %bptr
@loop
  %bcur = loadl %bptr
  %is_zero = ceql %bcur, 0
  jnz %is_zero, @done, @step
@step
  %acur = loadl %base
  %rem = remul %acur, %bcur
  storel %bcur, %base
  storel %rem, %bptr
  jmp @loop
@done
  %ret = loadl %base
  ret %ret
}
