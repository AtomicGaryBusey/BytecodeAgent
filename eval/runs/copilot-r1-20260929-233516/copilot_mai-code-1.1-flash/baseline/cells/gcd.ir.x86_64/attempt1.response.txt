export function l $gcd(l %a, l %b) {
@start
  %base = alloc8 16
  %pb = addl %base, 8
  storel %a, %base
  storel %b, %pb
@loop
  %bcur = loadl %pb
  %is_zero = ceql %bcur, 0
  jnz %is_zero, @done, @step
@step
  %acur = loadl %base
  %bcur2 = loadl %pb
  %a_lt_b = csltl %acur, %bcur2
  jnz %a_lt_b, @swap, @subtract
@subtract
  %newa = subl %acur, %bcur2
  storel %newa, %base
  jmp @loop
@swap
  storel %bcur2, %base
  storel %acur, %pb
  jmp @loop
@done
  %ret = loadl %base
  ret %ret
}
