memmove_i64:
    cmp x2, #0
    b.le .Ldone
    cmp x0, x1
    b.eq .Ldone
    cmp x0, x1
    b.lo .Lforward
    mov x3, x2
.Lbackward:
    cbz x3, .Ldone
    sub x3, x3, #1
    add x4, x1, x3, lsl #3
    add x5, x0, x3, lsl #3
    ldr x6, [x4]
    str x6, [x5]
    b .Lbackward
.Lforward:
    mov x3, #0
.Lforward_loop:
    cmp x3, x2
    b.ge .Ldone
    add x4, x1, x3, lsl #3
    add x5, x0, x3, lsl #3
    ldr x6, [x4]
    str x6, [x5]
    add x3, x3, #1
    b .Lforward_loop
.Ldone:
    mov x0, #0
    ret
