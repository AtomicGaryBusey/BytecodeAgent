stats_f:
push rbx
push rbp
push r12
push r13
push r14
sub rsp, 32
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov rbx, rcx
mov rdi, qword ptr [r12]
call r14
mov qword ptr [rsp], rax
mov qword ptr [rsp + 8], rax
mov qword ptr [rsp + 16], rax
xor ebp, ebp
test rax, 1
jz .Lfirst_even
inc rbp
.Lfirst_even:
add r12, 8
dec r13
jz .Ldone
.Lloop:
mov rdi, qword ptr [r12]
call r14
cmp rax, qword ptr [rsp]
jge .Lmin_done
mov qword ptr [rsp], rax
.Lmin_done:
cmp rax, qword ptr [rsp + 8]
jle .Lmax_done
mov qword ptr [rsp + 8], rax
.Lmax_done:
add qword ptr [rsp + 16], rax
test rax, 1
jz .Leven
inc rbp
.Leven:
add r12, 8
dec r13
jnz .Lloop
.Ldone:
mov rax, qword ptr [rsp]
mov qword ptr [rbx], rax
mov rax, qword ptr [rsp + 8]
mov qword ptr [rbx + 8], rax
mov rax, qword ptr [rsp + 16]
mov qword ptr [rbx + 16], rax
mov rax, rbp
add rsp, 32
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
