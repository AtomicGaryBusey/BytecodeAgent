export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	%limit =l sub %n, 1
	jnz %small, @zero, @loop
@loop
	%i =l phi @entry 0, @body %nexti
	%count =l phi @entry 0, @body %nextcount
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%left =l loadl %ptr
	%rightptr =l add %ptr, 8
	%right =l loadl %rightptr
	%descending =w csgtl %left, %right
	%increment =l extuw %descending
	%nextcount =l add %count, %increment
	%nexti =l add %i, 1
	jmp @loop
@done
	ret %count
@zero
	ret 0
}
