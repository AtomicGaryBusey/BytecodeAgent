export function l $reverse(l %a, l %n) {
@start
	%last = sub %n, 1
	%lastoff = mul %last, 8
	%gt = csgtl %n, 1
	jnz %gt, @loop, @done
@loop
	%i = phi @start 0, @body %inext
	%j = phi @start %lastoff, @body %jnext
	%cond = csltl %i, %j
	jnz %cond, @body, @done
@body
	%left = add %a, %i
	%right = add %a, %j
	%x = loadl %left
	%y = loadl %right
	storel %y, %left
	storel %x, %right
	%inext = add %i, 8
	%jnext = sub %j, 8
	jmp @loop
@done
	ret 0
}
