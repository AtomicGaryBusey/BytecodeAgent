export function l $reverse(l %a, l %n) {
@start
	%last =l sub %n, 1
	%lastoff =l mul %last, 8
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %lastoff, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%left =l add %a, %i
	%right =l add %a, %j
	%x =l loadl %left
	%y =l loadl %right
	storel %y, %left
	storel %x, %right
	%inext =l add %i, 8
	%jnext =l sub %j, 8
	jmp @loop
@done
	ret 0
}
