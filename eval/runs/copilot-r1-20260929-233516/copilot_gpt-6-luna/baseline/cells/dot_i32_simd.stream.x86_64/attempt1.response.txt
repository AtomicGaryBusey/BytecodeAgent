dot_i32_simd:
    mov rcx, rdx
    pxor xmm0, xmm0
.Lvector_loop:
    cmp rcx, 4
    jb .Lscalar_setup
    movdqu xmm1, xmmword ptr [rdi]
    movdqu xmm2, xmmword ptr [rsi]
    movdqa xmm3, xmm1
    pmuldq xmm1, xmm2
    pshufd xmm3, xmm3, 0xb1
    pshufd xmm4, xmm2, 0xb1
    pmuldq xmm3, xmm4
    paddq xmm1, xmm3
    paddq xmm0, xmm1
    add rdi, 16
    add rsi, 16
    sub rcx, 4
    jmp .Lvector_loop
.Lscalar_setup:
    xor eax, eax
.Lscalar_loop:
    test rcx, rcx
    jz .Lreduce
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rcx
    jmp .Lscalar_loop
.Lreduce:
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq r8, xmm0
    add rax, r8
    ret
