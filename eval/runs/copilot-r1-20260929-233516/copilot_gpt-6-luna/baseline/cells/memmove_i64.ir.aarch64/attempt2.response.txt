export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %bytes =l mul %n, 8
    %end =l add %src, %bytes
    %before =w cultl %dst, %src
    jnz %before, @forward, @checkafter
@checkafter
    %after =w cugel %dst, %end
    jnz %after, @forward, @backward
@forward
    jmp @floop
@floop
    %fi =l phi @forward 0, @fbody %finext
    %fcheck =w csltl %fi, %n
    jnz %fcheck, @fbody, @done
@fbody
    %foff =l mul %fi, 8
    %fsp =l add %src, %foff
    %fv =l loadl %fsp
    %fdp =l add %dst, %foff
    storel %fv, %fdp
    %finext =l add %fi, 1
    jmp @floop
@backward
    jmp @bloop
@bloop
    %bi =l phi @backward %n, @bbody %binext
    %bcheck =w csgtl %bi, 0
    jnz %bcheck, @bbody, @done
@bbody
    %binext =l sub %bi, 1
    %boff =l mul %binext, 8
    %bsp =l add %src, %boff
    %bv =l loadl %bsp
    %bdp =l add %dst, %boff
    storel %bv, %bdp
    jmp @bloop
@done
    ret 0
}
