export function l $crc32(l %p, l %n) {
@entry
    jmp @outer
@outer
    %i =l phi @entry 0, @outerinc %inext
    %ptr =l phi @entry %p, @outerinc %ptrnext
    %crc =w phi @entry -1, @outerinc %crcafter
    %more =w csltl %i, %n
    jnz %more, @load, @finish
@load
    %byte =w loadub %ptr
    %crcbyte =w xor %crc, %byte
    jmp @inner
@inner
    %bitidx =l phi @load 0, @innerjoin %bitnext
    %bitcrc =w phi @load %crcbyte, @innerjoin %crcnext
    %bitsleft =w csltl %bitidx, 8
    jnz %bitsleft, @bit, @outerinc
@bit
    %low =w and %bitcrc, 1
    jnz %low, @poly, @plain
@poly
    %shiftpoly =w shr %bitcrc, 1
    %xorpoly =w xor %shiftpoly, -306674912
    jmp @innerjoin
@plain
    %shiftplain =w shr %bitcrc, 1
    jmp @innerjoin
@innerjoin
    %crcnext =w phi @poly %xorpoly, @plain %shiftplain
    %bitnext =l add %bitidx, 1
    jmp @inner
@outerinc
    %inext =l add %i, 1
    %ptrnext =l add %ptr, 1
    %crcafter =w copy %bitcrc
    jmp @outer
@finish
    %finalcrc =w xor %crc, -1
    %result =l extuw %finalcrc
    ret %result
}
