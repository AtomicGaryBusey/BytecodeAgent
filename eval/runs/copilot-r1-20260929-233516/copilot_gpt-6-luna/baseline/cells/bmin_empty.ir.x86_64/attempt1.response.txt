export function l $bmin_empty(l %a, l %n) {
@entry
    %empty =w ceql %n, 0
    jnz %empty, @zero, @loop
@zero
    jmp @finish
@loop
    %i =l phi @entry 0, @step %inext
    %best =l phi @entry 9223372036854775807, @step %newbest
    %more =w csltl %i, %n
    jnz %more, @body, @finish
@body
    %off =l mul %i, 8
    %p =l add %a, %off
    %v =l loadl %p
    %less =w csltl %v, %best
    jnz %less, @update, @same
@update
    jmp @step
@same
    jmp @step
@step
    %newbest =l phi @update %v, @same %best
    %inext =l add %i, 1
    jmp @loop
@finish
    %result =l phi @zero 9223372036854775807, @loop %best
    ret %result
}
