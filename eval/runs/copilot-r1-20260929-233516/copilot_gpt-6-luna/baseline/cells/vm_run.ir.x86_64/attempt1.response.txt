export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pcptr =l alloc8 8
    %dptr =l alloc8 8
    %rvptr =l alloc8 8
    %idxptr =l alloc8 8
    storel 0, %pcptr
    storel 0, %dptr
    jmp @loop

@loop
    %pc =l loadl %pcptr
    %d =l loadl %dptr
    %end =w ceql %pc, %len
    jnz %end, @finish, @fetch

@fetch
    %codeoff =l mul %pc, 8
    %codeaddr =l add %code, %codeoff
    %op =l loadl %codeaddr
    %is0 =w ceql %op, 0
    jnz %is0, @halt, @check1

@check1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @check2

@check2
    %is2 =w ceql %op, 2
    jnz %is2, @add, @check3

@check3
    %is3 =w ceql %op, 3
    jnz %is3, @sub, @check4

@check4
    %is4 =w ceql %op, 4
    jnz %is4, @mul, @check5

@check5
    %is5 =w ceql %op, 5
    jnz %is5, @dup, @check6

@check6
    %is6 =w ceql %op, 6
    jnz %is6, @swap, @check7

@check7
    %is7 =w ceql %op, 7
    jnz %is7, @drop, @check8

@check8
    %is8 =w ceql %op, 8
    jnz %is8, @over, @check9

@check9
    %is9 =w ceql %op, 9
    jnz %is9, @lt, @check10

@check10
    %is10 =w ceql %op, 10
    jnz %is10, @jz, @check11

@check11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp, @other

@halt
    jmp @finish

@push
    %pushpc =l add %pc, 1
    %pushoff =l mul %pushpc, 8
    %pushaddr =l add %code, %pushoff
    %pushval =l loadl %pushaddr
    %pushoffstack =l mul %d, 8
    %pushslot =l add %stack, %pushoffstack
    storel %pushval, %pushslot
    %nextpushpc =l add %pc, 2
    %nextpushd =l add %d, 1
    storel %nextpushpc, %pcptr
    storel %nextpushd, %dptr
    jmp @loop

@add
    %addidxb =l sub %d, 1
    %addidxa =l sub %d, 2
    %addoffb =l mul %addidxb, 8
    %addoffa =l mul %addidxa, 8
    %addaddrb =l add %stack, %addoffb
    %addaddra =l add %stack, %addoffa
    %addb =l loadl %addaddrb
    %adda =l loadl %addaddra
    %addres =l add %adda, %addb
    storel %addres, %addaddra
    %addpc =l add %pc, 1
    %addd =l sub %d, 1
    storel %addpc, %pcptr
    storel %addd, %dptr
    jmp @loop

@sub
    %subidxb =l sub %d, 1
    %subidxa =l sub %d, 2
    %suboffb =l mul %subidxb, 8
    %suboffa =l mul %subidxa, 8
    %subaddrb =l add %stack, %suboffb
    %subaddra =l add %stack, %suboffa
    %subb =l loadl %subaddrb
    %suba =l loadl %subaddra
    %subres =l sub %suba, %subb
    storel %subres, %subaddra
    %subpc =l add %pc, 1
    %subd =l sub %d, 1
    storel %subpc, %pcptr
    storel %subd, %dptr
    jmp @loop

@mul
    %mulidxb =l sub %d, 1
    %mulidxa =l sub %d, 2
    %muloffb =l mul %mulidxb, 8
    %muloffa =l mul %mulidxa, 8
    %muladdrb =l add %stack, %muloffb
    %muladdra =l add %stack, %muloffa
    %mulb =l loadl %muladdrb
    %mula =l loadl %muladdra
    %mulres =l mul %mula, %mulb
    storel %mulres, %muladdra
    %mulpc =l add %pc, 1
    %muld =l sub %d, 1
    storel %mulpc, %pcptr
    storel %muld, %dptr
    jmp @loop

@dup
    %dupidx =l sub %d, 1
    %dupoff =l mul %dupidx, 8
    %dupsrc =l add %stack, %dupoff
    %dupval =l loadl %dupsrc
    %dupoutoff =l mul %d, 8
    %dupout =l add %stack, %dupoutoff
    storel %dupval, %dupout
    %duppc =l add %pc, 1
    %dupd =l add %d, 1
    storel %duppc, %pcptr
    storel %dupd, %dptr
    jmp @loop

@swap
    %swapidxa =l sub %d, 1
    %swapidxb =l sub %d, 2
    %swapoffa =l mul %swapidxa, 8
    %swapoffb =l mul %swapidxb, 8
    %swapaddra =l add %stack, %swapoffa
    %swapaddrb =l add %stack, %swapoffb
    %swapa =l loadl %swapaddra
    %swapb =l loadl %swapaddrb
    storel %swapb, %swapaddra
    storel %swapa, %swapaddrb
    %swappc =l add %pc, 1
    storel %swappc, %pcptr
    jmp @loop

@drop
    %droppc =l add %pc, 1
    %dropd =l sub %d, 1
    storel %droppc, %pcptr
    storel %dropd, %dptr
    jmp @loop

@over
    %overidx =l sub %d, 2
    %overoff =l mul %overidx, 8
    %oversrc =l add %stack, %overoff
    %overval =l loadl %oversrc
    %overoutoff =l mul %d, 8
    %overout =l add %stack, %overoutoff
    storel %overval, %overout
    %overpc =l add %pc, 1
    %overd =l add %d, 1
    storel %overpc, %pcptr
    storel %overd, %dptr
    jmp @loop

@lt
    %ltidxb =l sub %d, 1
    %ltidxa =l sub %d, 2
    %ltoffb =l mul %ltidxb, 8
    %ltoffa =l mul %ltidxa, 8
    %ltaddrb =l add %stack, %ltoffb
    %ltaddra =l add %stack, %ltoffa
    %ltb =l loadl %ltaddrb
    %lta =l loadl %ltaddra
    %ltcmp =w csltl %lta, %ltb
    %ltres =l extuw %ltcmp
    storel %ltres, %ltaddra
    %ltpc =l add %pc, 1
    %ltd =l sub %d, 1
    storel %ltpc, %pcptr
    storel %ltd, %dptr
    jmp @loop

@jz
    %jzidx =l sub %d, 1
    %jzoff =l mul %jzidx, 8
    %jzaddr =l add %stack, %jzoff
    %jzval =l loadl %jzaddr
    %jzoperandpc =l add %pc, 1
    %jzoperandoff =l mul %jzoperandpc, 8
    %jzoperandaddr =l add %code, %jzoperandoff
    %jztarget =l loadl %jzoperandaddr
    %jzfall =l add %pc, 2
    %jzdepth =l sub %d, 1
    storel %jzdepth, %dptr
    %jziszero =w ceql %jzval, 0
    jnz %jziszero, @jztaken, @jznot

@jztaken
    storel %jztarget, %pcptr
    jmp @loop

@jznot
    storel %jzfall, %pcptr
    jmp @loop

@jmp
    %jmppcoperand =l add %pc, 1
    %jmppcoff =l mul %jmppcoperand, 8
    %jmppcaddr =l add %code, %jmppcoff
    %jmptarget =l loadl %jmppcaddr
    storel %jmptarget, %pcptr
    jmp @loop

@other
    %otherpc =l add %pc, 1
    storel %otherpc, %pcptr
    jmp @loop

@finish
    %finald =l loadl %dptr
    %isempty =w ceql %finald, 0
    jnz %isempty, @emptyresult, @nonemptyresult

@emptyresult
    storel 0, %rvptr
    jmp @clearstart

@nonemptyresult
    %topidx =l sub %finald, 1
    %topoff =l mul %topidx, 8
    %topaddr =l add %stack, %topoff
    %topval =l loadl %topaddr
    storel %topval, %rvptr
    jmp @clearstart

@clearstart
    storel %finald, %idxptr
    jmp @clearloop

@clearloop
    %clearidx =l loadl %idxptr
    %clearcond =w csltl %clearidx, %cap
    jnz %clearcond, @clearone, @return

@clearone
    %clearoff =l mul %clearidx, 8
    %clearaddr =l add %stack, %clearoff
    storel 0, %clearaddr
    %clearidxnext =l add %clearidx, 1
    storel %clearidxnext, %idxptr
    jmp @clearloop

@return
    %retval =l loadl %rvptr
    ret %retval
}
