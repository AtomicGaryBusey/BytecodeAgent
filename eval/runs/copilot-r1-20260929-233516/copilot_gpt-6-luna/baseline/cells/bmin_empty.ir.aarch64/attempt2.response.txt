export function l $bmin_empty(l %a, l %n) {
@entry
    %zero =l copy 0
    %max =l copy 9223372036854775807
    jmp @loop
@loop
    %i =l phi @entry %zero, @step %next
    %best =l phi @entry %max, @step %newbest
    %more =w csltl %i, %n
    jnz %more, @body, @exit
@body
    %offset =l shl %i, 3
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %smaller =w csltl %value, %best
    jnz %smaller, @take, @keep
@take
    jmp @step
@keep
    jmp @step
@step
    %newbest =l phi @take %value, @keep %best
    %next =l add %i, 1
    jmp @loop
@exit
    ret %best
}
