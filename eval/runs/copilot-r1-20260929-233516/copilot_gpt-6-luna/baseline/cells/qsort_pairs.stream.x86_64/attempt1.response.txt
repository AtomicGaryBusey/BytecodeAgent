qsort_pairs:
cld
test rsi, rsi
cmp rsi, 1
jle .Lsmall
push rbx
push r12
push r13
push r14
push r15
sub rsp, 16
mov rbx, rdi
xor r12, r12
lea r13, [rsi - 1]
shl r13, 1
lea rax, [r12 + r13]
shr rax, 1
and rax, -2
mov rdx, qword ptr [rbx + rax*8]
mov qword ptr [rsp], rdx
mov rdx, qword ptr [rbx + rax*8 + 8]
mov qword ptr [rsp + 8], rdx
mov r14, r12
mov r15, r13
.Lpartition:
.Lscan_i:
mov rax, qword ptr [rbx + r14*8]
cmp rax, qword ptr [rsp]
jl .Linc_i
jg .Lscan_j
mov rax, qword ptr [rbx + r14*8 + 8]
cmp rax, qword ptr [rsp + 8]
jl .Linc_i
jmp .Lscan_j
.Linc_i:
add r14, 2
jmp .Lscan_i
.Lscan_j:
mov rax, qword ptr [rbx + r15*8]
cmp rax, qword ptr [rsp]
jg .Ldec_j
jl .Lcheck
mov rax, qword ptr [rbx + r15*8 + 8]
cmp rax, qword ptr [rsp + 8]
jg .Ldec_j
.Lcheck:
cmp r14, r15
jge .Lpartdone
mov rax, qword ptr [rbx + r14*8]
mov rdx, qword ptr [rbx + r14*8 + 8]
mov r10, qword ptr [rbx + r15*8]
mov r11, qword ptr [rbx + r15*8 + 8]
mov qword ptr [rbx + r14*8], r10
mov qword ptr [rbx + r14*8 + 8], r11
mov qword ptr [rbx + r15*8], rax
mov qword ptr [rbx + r15*8 + 8], rdx
add r14, 2
sub r15, 2
jmp .Lpartition
.Ldec_j:
sub r15, 2
jmp .Lscan_j
.Lpartdone:
mov rsi, r15
sub rsi, r12
shr rsi, 1
inc rsi
cmp rsi, 1
jle .Lright
lea rdi, [rbx + r12*8]
call qsort_pairs
.Lright:
lea rax, [r15 + 2]
cmp rax, r13
jg .Ldone
mov rsi, r13
sub rsi, rax
shr rsi, 1
inc rsi
cmp rsi, 1
jle .Ldone
lea rdi, [rbx + rax*8]
call qsort_pairs
.Ldone:
add rsp, 16
pop r15
pop r14
pop r13
pop r12
pop rbx
xor eax, eax
ret
.Lsmall:
xor eax, eax
ret
