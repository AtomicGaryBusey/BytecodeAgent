export function l $qsort_pairs(l %a, l %n) {
@entry
    %small =cslel %n, 1
    jnz %small, @done, @setup
@setup
    %last =l sub %n, 1
    %lastoff =l mul %last, 16
    %pivotptr =l add %a, %lastoff
    %pk =l loadl %pivotptr
    %pivotvptr =l add %pivotptr, 8
    %pv =l loadl %pivotvptr
    jmp @loop
@loop
    %i =l phi @setup 0, @advance %inext
    %j =l phi @setup 0, @advance %jnext
    %more =csltl %j, %last
    jnz %more, @iter, @finish
@iter
    %joff =l mul %j, 16
    %jptr =l add %a, %joff
    %jk =l loadl %jptr
    %jvptr =l add %jptr, 8
    %jv =l loadl %jvptr
    %keyless =csltl %jk, %pk
    %keyequal =ceql %jk, %pk
    %valess =csltl %jv, %pv
    %tieless =and %keyequal, %valess
    %less =or %keyless, %tieless
    jnz %less, @doswap, @noswap
@doswap
    %ioff =l mul %i, 16
    %iptr =l add %a, %ioff
    %ik =l loadl %iptr
    %ivptr =l add %iptr, 8
    %iv =l loadl %ivptr
    storel %jk, %iptr
    storel %jv, %ivptr
    storel %ik, %jptr
    storel %iv, %jvptr
    %inextswap =l add %i, 1
    jmp @advance
@noswap
    jmp @advance
@advance
    %inext =l phi @doswap %inextswap, @noswap %i
    %jnext =l add %j, 1
    jmp @loop
@finish
    %splitoff =l mul %i, 16
    %splitptr =l add %a, %splitoff
    %splitvptr =l add %splitptr, 8
    %splitk =l loadl %splitptr
    %splitv =l loadl %splitvptr
    %endk =l loadl %pivotptr
    %endv =l loadl %pivotvptr
    storel %endk, %splitptr
    storel %endv, %splitvptr
    storel %splitk, %pivotptr
    storel %splitv, %pivotvptr
    %leftret =l call $qsort_pairs(l %a, l %i)
    %rightstart =l add %i, 1
    %rightoff =l mul %rightstart, 16
    %rightptr =l add %a, %rightoff
    %rightn =l sub %n, %rightstart
    %rightret =l call $qsort_pairs(l %rightptr, l %rightn)
    jmp @done
@done
    ret 0
}
