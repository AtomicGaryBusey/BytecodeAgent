export function l $binary_search(l %a, l %n, l %key) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%lo =l phi @start %zero, @advance %lo_next, @retreat %lo
	%hi =l phi @start %n, @advance %hi, @retreat %mid
	%more =w csltl %lo, %hi
	jnz %more, @search, @miss
@search
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%offset =l shl %mid, 3
	%address =l add %a, %offset
	%value =l loadl %address
	%found =w ceql %value, %key
	jnz %found, @hit, @compare
@hit
	ret %mid
@compare
	%less =w csltl %value, %key
	jnz %less, @advance, @retreat
@advance
	%lo_next =l add %mid, 1
	jmp @loop
@retreat
	jmp @loop
@miss
	%minus_one =l copy -1
	ret %minus_one
}
