export function l $isqrt(l %n) {
@entry
    %zero =l copy 0
    %limit =l copy 4294967296
    jmp @loop
@loop
    %lo =l phi @entry %zero, @take %mid, @skip %lo
    %hi =l phi @entry %limit, @take %hi, @skip %mid
    %diff =l sub %hi, %lo
    %more =w cugtl %diff, 1
    jnz %more, @search, @done
@search
    %sum =l add %lo, %hi
    %mid =l shr %sum, 1
    %square =l mul %mid, %mid
    %fits =w culel %square, %n
    jnz %fits, @take, @skip
@take
    jmp @loop
@skip
    jmp @loop
@done
    ret %lo
}
