export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%idxslot =l alloc8 8
	%rowslot =l alloc8 8
	%colslot =l alloc8 8
	%tslot =l alloc8 8
	%iterslot =l alloc8 8
	%n2 =l mul %n, %n
	%zero =l copy 0
	storel %zero, %idxslot
	jmp @init_loop

@init_loop
	%idx0 =l loadl %idxslot
	%init_cond =w csltl %idx0, %n2
	jnz %init_cond, @init_body, @diag_start

@init_body
	%byteoff0 =l mul %idx0, 8
	%outptr0 =l add %out, %byteoff0
	storel %zero, %outptr0
	%aptr0 =l add %a, %byteoff0
	%aval0 =l loadl %aptr0
	%tmpptr0 =l add %tmp, %byteoff0
	storel %aval0, %tmpptr0
	%idx1 =l add %idx0, 1
	storel %idx1, %idxslot
	jmp @init_loop

@diag_start
	storel %zero, %rowslot
	jmp @diag_loop

@diag_loop
	%diag_i =l loadl %rowslot
	%diag_cond =w csltl %diag_i, %n
	jnz %diag_cond, @diag_body, @iter_start

@diag_body
	%diag_rowoff =l mul %diag_i, %n
	%diag_index =l add %diag_rowoff, %diag_i
	%diag_byteoff =l mul %diag_index, 8
	%diag_ptr =l add %out, %diag_byteoff
	storel 1, %diag_ptr
	%diag_next =l add %diag_i, 1
	storel %diag_next, %rowslot
	jmp @diag_loop

@iter_start
	storel %zero, %iterslot
	jmp @iter_loop

@iter_loop
	%iter =l loadl %iterslot
	%iter_cond =w csltl %iter, %k
	jnz %iter_cond, @row_start, @clear_start

@row_start
	storel %zero, %rowslot
	jmp @row_loop

@row_loop
	%row =l loadl %rowslot
	%row_cond =w csltl %row, %n
	jnz %row_cond, @col_start, @copy_start

@col_start
	storel %zero, %colslot
	jmp @col_loop

@col_loop
	%col =l loadl %colslot
	%col_cond =w csltl %col, %n
	jnz %col_cond, @sum_start, @row_next

@sum_start
	storel %zero, %tslot
	storel %zero, %idxslot
	jmp @sum_loop

@sum_loop
	%t =l loadl %tslot
	%sum_cond =w csltl %t, %n
	jnz %sum_cond, @sum_body, @sum_done

@sum_body
	%row_base =l mul %row, %n
	%out_index =l add %row_base, %t
	%out_byteoff =l mul %out_index, 8
	%out_elem_ptr =l add %out, %out_byteoff
	%out_elem =l loadl %out_elem_ptr
	%t_base =l mul %t, %n
	%base_index =l add %t_base, %col
	%base_byteoff =l mul %base_index, 8
	%base_elem_ptr =l add %tmp, %base_byteoff
	%base_elem =l loadl %base_elem_ptr
	%product =l mul %out_elem, %base_elem
	%sum_old =l loadl %idxslot
	%sum_new =l add %sum_old, %product
	storel %sum_new, %idxslot
	%t_next =l add %t, 1
	storel %t_next, %tslot
	jmp @sum_loop

@sum_done
	%result_sum =l loadl %idxslot
	%result_rowbase =l mul %row, %n
	%result_index =l add %result_rowbase, %col
	%result_tmp_index =l add %n2, %result_index
	%result_tmp_off =l mul %result_tmp_index, 8
	%result_ptr =l add %tmp, %result_tmp_off
	storel %result_sum, %result_ptr
	jmp @col_next

@col_next
	%col_now =l loadl %colslot
	%col_next_value =l add %col_now, 1
	storel %col_next_value, %colslot
	jmp @col_loop

@row_next
	%row_now =l loadl %rowslot
	%row_next_value =l add %row_now, 1
	storel %row_next_value, %rowslot
	jmp @row_loop

@copy_start
	storel %zero, %idxslot
	jmp @copy_loop

@copy_loop
	%copy_idx =l loadl %idxslot
	%copy_cond =w csltl %copy_idx, %n2
	jnz %copy_cond, @copy_body, @iter_next

@copy_body
	%copy_tmp_idx =l add %n2, %copy_idx
	%copy_tmp_off =l mul %copy_tmp_idx, 8
	%copy_tmp_ptr =l add %tmp, %copy_tmp_off
	%copy_value =l loadl %copy_tmp_ptr
	%copy_out_off =l mul %copy_idx, 8
	%copy_out_ptr =l add %out, %copy_out_off
	storel %copy_value, %copy_out_ptr
	%copy_next =l add %copy_idx, 1
	storel %copy_next, %idxslot
	jmp @copy_loop

@iter_next
	%iter_now =l loadl %iterslot
	%iter_next_value =l add %iter_now, 1
	storel %iter_next_value, %iterslot
	jmp @iter_loop

@clear_start
	storel %zero, %idxslot
	%clear_count =l mul %n2, 2
	jmp @clear_loop

@clear_loop
	%clear_idx =l loadl %idxslot
	%clear_cond =w csltl %clear_idx, %clear_count
	jnz %clear_cond, @clear_body, @done

@clear_body
	%clear_off =l mul %clear_idx, 8
	%clear_ptr =l add %tmp, %clear_off
	storel %zero, %clear_ptr
	%clear_next =l add %clear_idx, 1
	storel %clear_next, %idxslot
	jmp @clear_loop

@done
	ret 0
}
