export function l $crc32(l %p, l %n) {
@start
	jmp @byte
@byte
	%i =l phi @start 0, @finishbyte %inext
	%crc =w phi @start -1, @finishbyte %c
	%morebytes =w csltl %i, %n
	jnz %morebytes, @process, @done
@process
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crcbyte =w xor %crc, %b
	jmp @bit
@bit
	%k =l phi @process 0, @step %knext
	%c =w phi @process %crcbyte, @step %cnext
	%morebits =w csltl %k, 8
	jnz %morebits, @bitbody, @finishbyte
@bitbody
	%low =w and %c, 1
	jnz %low, @odd, @even
@odd
	%shiftodd =w shr %c, 1
	%codd =w xor %shiftodd, -306674912
	jmp @step
@even
	%ceven =w shr %c, 1
	jmp @step
@step
	%cnext =w phi @odd %codd, @even %ceven
	%knext =l add %k, 1
	jmp @bit
@finishbyte
	%inext =l add %i, 1
	jmp @byte
@done
	%result =w xor %crc, -1
	%ret =l extuw %result
	ret %ret
}
