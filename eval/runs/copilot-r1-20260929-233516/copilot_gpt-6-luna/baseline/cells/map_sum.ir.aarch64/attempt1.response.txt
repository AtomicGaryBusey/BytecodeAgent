export function l $map_sum(l %a, l %n, l %f) {
@start
	%idx =l alloc8 8
	%sum =l alloc8 8
	%len =l alloc8 8
	%fun =l alloc8 8
	storel 0, %idx
	storel 0, %sum
	storel %n, %len
	storel %f, %fun
	jmp @test
@test
	%i =l loadl %idx
	%bound =l loadl %len
	%more =w csltl %i, %bound
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%value =l loadl %addr
	%func =l loadl %fun
	%mapped =l call %func(l %value)
	%old =l loadl %sum
	%new =l add %old, %mapped
	storel %new, %sum
	%next =l add %i, 1
	storel %next, %idx
	jmp @test
@end
	%result =l loadl %sum
	ret %result
}
