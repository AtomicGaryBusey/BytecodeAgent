export function l $bmax(l %a, l %n) {
@start
	%best0 =l loadl %a
	%p0 =l add %a, 8
	jmp @test
@test
	%i =l phi @start 1, @step %inext
	%p =l phi @start %p0, @step %pnext
	%best =l phi @start %best0, @step %bestnext
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%x =l loadl %p
	%gt =w csgtl %x, %best
	jnz %gt, @take, @step
@take
	jmp @step
@step
	%bestnext =l phi @body %best, @take %x
	%inext =l add %i, 1
	%pnext =l add %p, 8
	jmp @test
@end
	ret %best
}
