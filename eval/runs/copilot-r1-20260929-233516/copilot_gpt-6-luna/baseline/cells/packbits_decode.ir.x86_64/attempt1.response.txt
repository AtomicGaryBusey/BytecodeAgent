export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @read
@read
	%si =l phi @entry 0, @next %nsi
	%oi =l phi @entry 0, @next %noi
	%has =w csltl %si, %n
	jnz %has, @header, @done
@done
	ret %oi
@header
	%hp =l add %src, %si
	%h =w loadub %hp
	%ps =l add %si, 1
	%noop =w ceqw %h, 128
	jnz %noop, @no_op, @classify
@no_op
	%noopsi =l copy %ps
	jmp @next
@classify
	%literal =w cslew %h, 127
	jnz %literal, @literal, @repeat
@literal
	%h64 =l extuw %h
	%lcnt =l add %h64, 1
	%lend =l add %ps, %lcnt
	%lbad =w csgtl %lend, %n
	jnz %lbad, @source_error, @literal_space
@literal_space
	%lavail =l sub %cap, %oi
	%lover =w csgtl %lcnt, %lavail
	jnz %lover, @literal_overflow, @literal_copy_start
@literal_copy_start
	jmp @literal_copy
@literal_copy
	%lk =l phi @literal_copy_start 0, @literal_copy_body %lk1
	%lmore =w csltl %lk, %lcnt
	jnz %lmore, @literal_copy_body, @literal_copy_done
@literal_copy_body
	%lsrcidx =l add %ps, %lk
	%lsrcptr =l add %src, %lsrcidx
	%lv =w loadub %lsrcptr
	%ldstidx =l add %oi, %lk
	%ldstptr =l add %out, %ldstidx
	storeb %lv, %ldstptr
	%lk1 =l add %lk, 1
	jmp @literal_copy
@literal_copy_done
	%lit_si =l add %ps, %lcnt
	%lit_oi =l add %oi, %lcnt
	jmp @next
@literal_overflow
	jmp @literal_truncate
@literal_truncate
	%ltk =l phi @literal_overflow 0, @literal_truncate_body %ltk1
	%ltmore =w csltl %ltk, %lavail
	jnz %ltmore, @literal_truncate_body, @overflow_done
@literal_truncate_body
	%ltsrcidx =l add %ps, %ltk
	%ltsrcptr =l add %src, %ltsrcidx
	%ltv =w loadub %ltsrcptr
	%ltdstidx =l add %oi, %ltk
	%ltdstptr =l add %out, %ltdstidx
	storeb %ltv, %ltdstptr
	%ltk1 =l add %ltk, 1
	jmp @literal_truncate
@repeat
	%rh64 =l extuw %h
	%rcnt =l sub 257, %rh64
	%rend =l add %ps, 1
	%rbad =w csgtl %rend, %n
	jnz %rbad, @source_error, @repeat_ready
@repeat_ready
	%rptr =l add %src, %ps
	%rv =w loadub %rptr
	%ravail =l sub %cap, %oi
	%rover =w csgtl %rcnt, %ravail
	jnz %rover, @repeat_overflow, @repeat_fill_start
@repeat_fill_start
	jmp @repeat_fill
@repeat_fill
	%rk =l phi @repeat_fill_start 0, @repeat_fill_body %rk1
	%rmore =w csltl %rk, %rcnt
	jnz %rmore, @repeat_fill_body, @repeat_fill_done
@repeat_fill_body
	%rdstidx =l add %oi, %rk
	%rdstptr =l add %out, %rdstidx
	storeb %rv, %rdstptr
	%rk1 =l add %rk, 1
	jmp @repeat_fill
@repeat_fill_done
	%rep_si =l add %ps, 1
	%rep_oi =l add %oi, %rcnt
	jmp @next
@repeat_overflow
	jmp @repeat_truncate
@repeat_truncate
	%rtk =l phi @repeat_overflow 0, @repeat_truncate_body %rtk1
	%rtmore =w csltl %rtk, %ravail
	jnz %rtmore, @repeat_truncate_body, @overflow_done
@repeat_truncate_body
	%rtdstidx =l add %oi, %rtk
	%rtdstptr =l add %out, %rtdstidx
	storeb %rv, %rtdstptr
	%rtk1 =l add %rtk, 1
	jmp @repeat_truncate
@overflow_done
	ret -1
@source_error
	ret -2
@next
	%nsi =l phi @no_op %noopsi, @literal_copy_done %lit_si, @repeat_fill_done %rep_si
	%noi =l phi @no_op %oi, @literal_copy_done %lit_oi, @repeat_fill_done %rep_oi
	jmp @read
}
