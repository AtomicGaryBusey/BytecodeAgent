export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %i_slot =l alloc8 8
    %j_slot =l alloc8 8
    %carry_slot =l alloc8 8
    %sig_slot =l alloc8 8
    %total =l add %na, %nb
    storel 0, %i_slot
    jmp @init

@init
    %init_i =l loadl %i_slot
    %init_cond =w csltl %init_i, %total
    jnz %init_cond, @init_body, @mul_start

@init_body
    %init_off =l shl %init_i, 3
    %init_ptr =l add %r, %init_off
    storel 0, %init_ptr
    %init_next =l add %init_i, 1
    storel %init_next, %i_slot
    jmp @init

@mul_start
    storel 0, %i_slot
    jmp @outer

@outer
    %outer_i =l loadl %i_slot
    %outer_cond =w csltl %outer_i, %na
    jnz %outer_cond, @outer_body, @scan_start

@outer_body
    storel 0, %j_slot
    storel 0, %carry_slot
    jmp @inner

@inner
    %inner_j =l loadl %j_slot
    %inner_cond =w csltl %inner_j, %nb
    jnz %inner_cond, @inner_body, @outer_finish

@inner_body
    %inner_i =l loadl %i_slot
    %a_off =l shl %inner_i, 3
    %a_ptr =l add %a, %a_off
    %x =l loadl %a_ptr
    %b_off =l shl %inner_j, 3
    %b_ptr =l add %b, %b_off
    %y =l loadl %b_ptr

    %x0 =l and %x, 4294967295
    %x1 =l shr %x, 32
    %y0 =l and %y, 4294967295
    %y1 =l shr %y, 32

    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1

    %p00_hi =l shr %p00, 32
    %p01_lo =l and %p01, 4294967295
    %p10_lo =l and %p10, 4294967295
    %mid_a =l add %p00_hi, %p01_lo
    %mid =l add %mid_a, %p10_lo

    %p00_lo =l and %p00, 4294967295
    %mid_shift =l shl %mid, 32
    %lo =l or %p00_lo, %mid_shift

    %p01_hi =l shr %p01, 32
    %p10_hi =l shr %p10, 32
    %mid_hi =l shr %mid, 32
    %hi_a =l add %p11, %p01_hi
    %hi_b =l add %hi_a, %p10_hi
    %hi =l add %hi_b, %mid_hi

    %out_index =l add %inner_i, %inner_j
    %out_off =l shl %out_index, 3
    %out_ptr =l add %r, %out_off
    %old =l loadl %out_ptr
    %sum_a =l add %old, %lo
    %carry_a =w cultl %sum_a, %old
    %carry_in =l loadl %carry_slot
    %sum_b =l add %sum_a, %carry_in
    %carry_b =w cultl %sum_b, %sum_a
    storel %sum_b, %out_ptr

    %carry_a_l =l extuw %carry_a
    %carry_b_l =l extuw %carry_b
    %carry_sum =l add %carry_a_l, %carry_b_l
    %next_carry =l add %hi, %carry_sum
    storel %next_carry, %carry_slot

    %next_j =l add %inner_j, 1
    storel %next_j, %j_slot
    jmp @inner

@outer_finish
    %finish_i =l loadl %i_slot
    %finish_index =l add %finish_i, %nb
    %finish_off =l shl %finish_index, 3
    %finish_ptr =l add %r, %finish_off
    %finish_carry =l loadl %carry_slot
    storel %finish_carry, %finish_ptr
    %next_i =l add %finish_i, 1
    storel %next_i, %i_slot
    jmp @outer

@scan_start
    storel %total, %sig_slot
    jmp @scan

@scan
    %sig =l loadl %sig_slot
    %scan_cond =w csgtl %sig, 0
    jnz %scan_cond, @scan_body, @done

@scan_body
    %top_index =l sub %sig, 1
    %top_off =l shl %top_index, 3
    %top_ptr =l add %r, %top_off
    %top_value =l loadl %top_ptr
    %top_nonzero =w cnel %top_value, 0
    jnz %top_nonzero, @done, @trim

@trim
    %trim_sig =l loadl %sig_slot
    %trim_next =l sub %trim_sig, 1
    storel %trim_next, %sig_slot
    jmp @scan

@done
    %result =l loadl %sig_slot
    ret %result
}
