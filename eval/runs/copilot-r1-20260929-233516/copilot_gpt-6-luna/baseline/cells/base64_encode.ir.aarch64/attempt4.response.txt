export function l $base64_encode(l %src, l %n, l %out) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%count =l alloc8 8
	%jp =l alloc8 8
	%idxs =l alloc8 32
	storel 0, %ip
	storel 0, %op
	jmp @group

@group
	%i =l loadl %ip
	%remain =l sub %n, %i
	%has =w csltl 0, %remain
	jnz %has, @choose, @done

@choose
	%full =w csltl 2, %remain
	jnz %full, @fullgroup, @partial

@fullgroup
	%p0 =l add %src, %i
	%p1 =l add %p0, 1
	%p2 =l add %p0, 2
	%b0 =l loadub %p0
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%x0 =l shr %b0, 2
	%x1a =l and %b0, 3
	%x1b =l shl %x1a, 4
	%x1c =l shr %b1, 4
	%x1 =l or %x1b, %x1c
	%x2a =l and %b1, 15
	%x2b =l shl %x2a, 2
	%x2c =l shr %b2, 6
	%x2 =l or %x2b, %x2c
	%x3 =l and %b2, 63
	%q1 =l add %idxs, 8
	%q2 =l add %idxs, 16
	%q3 =l add %idxs, 24
	storel %x0, %idxs
	storel %x1, %q1
	storel %x2, %q2
	storel %x3, %q3
	storel 4, %count
	jmp @emitstart

@partial
	%one =w ceql %remain, 1
	jnz %one, @onebyte, @twobytes

@onebyte
	%p0t =l add %src, %i
	%b0t =l loadub %p0t
	%x0t =l shr %b0t, 2
	%x1at =l and %b0t, 3
	%x1t =l shl %x1at, 4
	%q1t =l add %idxs, 8
	%q2t =l add %idxs, 16
	%q3t =l add %idxs, 24
	storel %x0t, %idxs
	storel %x1t, %q1t
	storel 0, %q2t
	storel 0, %q3t
	storel 2, %count
	jmp @emitstart

@twobytes
	%p0u =l add %src, %i
	%p1u =l add %p0u, 1
	%b0u =l loadub %p0u
	%b1u =l loadub %p1u
	%x0u =l shr %b0u, 2
	%x1au =l and %b0u, 3
	%x1bu =l shl %x1au, 4
	%x1cu =l shr %b1u, 4
	%x1u =l or %x1bu, %x1cu
	%x2au =l and %b1u, 15
	%x2u =l shl %x2au, 2
	%q1u =l add %idxs, 8
	%q2u =l add %idxs, 16
	%q3u =l add %idxs, 24
	storel %x0u, %idxs
	storel %x1u, %q1u
	storel %x2u, %q2u
	storel 0, %q3u
	storel 3, %count
	jmp @emitstart

@emitstart
	storel 0, %jp
	jmp @emitloop

@emitloop
	%j =l loadl %jp
	%more =w csltl %j, 4
	jnz %more, @emittest, @advance

@emittest
	%valid =l loadl %count
	%isvalid =w csltl %j, %valid
	jnz %isvalid, @mapstart, @padding

@padding
	%padop =l loadl %op
	%paddst =l add %out, %padop
	storeb 61, %paddst
	%padnext =l add %padop, 1
	storel %padnext, %op
	%padj =l add %j, 1
	storel %padj, %jp
	jmp @emitloop

@mapstart
	%slot =l shl %j, 3
	%slotptr =l add %idxs, %slot
	%ixmap =l loadl %slotptr
	%mapop =l loadl %op
	%mapdst =l add %out, %mapop
	%less26 =w csltl %ixmap, 26
	jnz %less26, @mapA, @mapcheck52

@mapA
	%charA =l add %ixmap, 65
	storeb %charA, %mapdst
	jmp @mapped

@mapcheck52
	%less52 =w csltl %ixmap, 52
	jnz %less52, @mapB, @mapcheck62

@mapB
	%charB =l add %ixmap, 71
	storeb %charB, %mapdst
	jmp @mapped

@mapcheck62
	%less62 =w csltl %ixmap, 62
	jnz %less62, @mapC, @mapcheckplus

@mapC
	%charC =l sub %ixmap, 4
	storeb %charC, %mapdst
	jmp @mapped

@mapcheckplus
	%is62 =w ceql %ixmap, 62
	jnz %is62, @mapplus, @mapslash

@mapplus
	storeb 43, %mapdst
	jmp @mapped

@mapslash
	storeb 47, %mapdst
	jmp @mapped

@mapped
	%mapnext =l add %mapop, 1
	storel %mapnext, %op
	%mapj =l add %j, 1
	storel %mapj, %jp
	jmp @emitloop

@advance
	%iold =l loadl %ip
	%ni =l add %iold, 3
	storel %ni, %ip
	jmp @group

@done
	%result =l loadl %op
	ret %result
}
