export function l $base64_encode(l %src, l %n, l %out) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%cp =l alloc8 8
	%jp =l alloc8 8
	%idxs =l alloc8 32
	storel 0, %ip
	storel 0, %op
	jmp @group

@group
	%i =l loadl %ip
	%remain =l sub %n, %i
	%has =w csltl 0, %remain
	jnz %has, @choose, @done

@choose
	%full =w csltl 2, %remain
	jnz %full, @fullgroup, @partial

@fullgroup
	%p0 =l add %src, %i
	%p1 =l add %p0, 1
	%p2 =l add %p0, 2
	%b0 =l loadub %p0
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%x0 =l shr %b0, 2
	%x1a =l and %b0, 3
	%x1b =l shl %x1a, 4
	%x1c =l shr %b1, 4
	%x1 =l or %x1b, %x1c
	%x2a =l and %b1, 15
	%x2b =l shl %x2a, 2
	%x2c =l shr %b2, 6
	%x2 =l or %x2b, %x2c
	%x3 =l and %b2, 63
	%q1 =l add %idxs, 8
	%q2 =l add %idxs, 16
	%q3 =l add %idxs, 24
	storel %x0, %idxs
	storel %x1, %q1
	storel %x2, %q2
	storel %x3, %q3
	storel 3, %cp
	jmp @emitstart

@partial
	%one =w ceql %remain, 1
	jnz %one, @onebyte, @twobytes

@onebyte
	%p0t =l add %src, %i
	%b0t =l loadub %p0t
	%x0t =l shr %b0t, 2
	%x1at =l and %b0t, 3
	%x1t =l shl %x1at, 4
	%q1t =l add %idxs, 8
	%q2t =l add %idxs, 16
	%q3t =l add %idxs, 24
	storel %x0t, %idxs
	storel %x1t, %q1t
	storel 0, %q2t
	storel 0, %q3t
	storel 1, %cp
	jmp @emitstart

@twobytes
	%p0u =l add %src, %i
	%p1u =l add %p0u, 1
	%b0u =l loadub %p0u
	%b1u =l loadub %p1u
	%x0u =l shr %b0u, 2
	%x1au =l and %b0u, 3
	%x1bu =l shl %x1au, 4
	%x1cu =l shr %b1u, 4
	%x1u =l or %x1bu, %x1cu
	%x2au =l and %b1u, 15
	%x2u =l shl %x2au, 2
	%q1u =l add %idxs, 8
	%q2u =l add %idxs, 16
	%q3u =l add %idxs, 24
	storel %x0u, %idxs
	storel %x1u, %q1u
	storel %x2u, %q2u
	storel 0, %q3u
	storel 2, %cp
	jmp @emitstart

@emitstart
	storel 0, %jp
	jmp @emitloop

@emitloop
	%j =l loadl %jp
	%more =w csltl %j, 4
	jnz %more, @emitone, @advance

@emitone
	%iptr =l add %idxs, %j
	%ix =l loadl %iptr
	%a =l add %ix, 65
	%b =l add %ix, 71
	%c =l sub %ix, 4
	%d =l sub %ix, 62
	%e =l shl %d, 2
	%f =l add %e, 43
	%is26 =w csltl %ix, 26
	%is52 =w csltl %ix, 52
	%is62 =w csltl %ix, 62
	%g =l selectl %is62, %c, %f
	%h =l selectl %is52, %b, %g
	%m =l selectl %is26, %a, %h
	%o =l loadl %op
	%dst =l add %out, %o
	storeb %m, %dst
	%no =l add %o, 1
	storel %no, %op
	%nj =l add %j, 1
	storel %nj, %jp
	jmp @emitloop

@advance
	%iold =l loadl %ip
	%consumed =l loadl %cp
	%ni =l add %iold, %consumed
	storel %ni, %ip
	jmp @group

@done
	%result =l loadl %op
	ret %result
}
