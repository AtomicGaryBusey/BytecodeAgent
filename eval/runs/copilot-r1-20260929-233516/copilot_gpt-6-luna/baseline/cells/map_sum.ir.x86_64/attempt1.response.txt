export function l $map_sum(l %a, l %n, l %f) {
@entry
    %zero =l copy 0
    jmp @loop
@loop
    %i =l phi @entry %zero, @body %next
    %sum =l phi @entry %zero, @body %sum2
    %test =w csltl %i, %n
    jnz %test, @body, @done
@body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %x =l loadl %addr
    %value =l call %f(l %x)
    %sum2 =l add %sum, %value
    %next =l add %i, 1
    jmp @loop
@done
    ret %sum
}
