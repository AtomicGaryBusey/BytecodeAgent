export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @loop

@loop
	%i =l phi @start 0, @body %nexti
	%count =l phi @start 0, @body %nextcount
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%offset =l shl %i, 3
	%srcaddr =l add %src, %offset
	%x =l loadl %srcaddr
	%result =l call %f(l %x)
	%dstaddr =l add %dst, %offset
	storel %result, %dstaddr
	%less =w csltl %result, %x
	%lessl =l extuw %less
	%nextcount =l add %count, %lessl
	%nexti =l add %i, 1
	jmp @loop

@done
	ret %count

@zero
	ret 0
}
