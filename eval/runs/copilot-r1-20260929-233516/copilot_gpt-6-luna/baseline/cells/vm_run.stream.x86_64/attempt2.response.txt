vm_run:
    cld
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov rbp, rcx
    xor rbx, rbx
    xor r15, r15
    xor r8, r8
.Lclear:
    cmp r8, rbp
    jae .Lloop
    mov qword ptr [r14 + r8*8], 0
    inc r8
    jmp .Lclear
.Lloop:
    cmp rbx, r13
    jae .Ldone
    mov rax, qword ptr [r12 + rbx*8]
    cmp rax, 0
    je .Ldone
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbx
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [r12 + rbx*8 + 8]
    mov qword ptr [r14 + r15*8], rax
    inc r15
    add rbx, 2
    jmp .Lloop
.Ladd:
    lea rcx, [r15 - 2]
    mov rax, qword ptr [r14 + rcx*8]
    add rax, qword ptr [r14 + rcx*8 + 8]
    mov qword ptr [r14 + rcx*8], rax
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    inc rbx
    jmp .Lloop
.Lsub:
    lea rcx, [r15 - 2]
    mov rax, qword ptr [r14 + rcx*8]
    sub rax, qword ptr [r14 + rcx*8 + 8]
    mov qword ptr [r14 + rcx*8], rax
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    inc rbx
    jmp .Lloop
.Lmul:
    lea rcx, [r15 - 2]
    mov rax, qword ptr [r14 + rcx*8]
    imul rax, qword ptr [r14 + rcx*8 + 8]
    mov qword ptr [r14 + rcx*8], rax
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    inc rbx
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [r14 + r15*8 - 8]
    mov qword ptr [r14 + r15*8], rax
    inc r15
    inc rbx
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [r14 + r15*8 - 8]
    mov rcx, qword ptr [r14 + r15*8 - 16]
    mov qword ptr [r14 + r15*8 - 8], rcx
    mov qword ptr [r14 + r15*8 - 16], rax
    inc rbx
    jmp .Lloop
.Ldrop:
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    inc rbx
    jmp .Lloop
.Lover:
    mov rax, qword ptr [r14 + r15*8 - 16]
    mov qword ptr [r14 + r15*8], rax
    inc r15
    inc rbx
    jmp .Lloop
.Llt:
    lea rcx, [r15 - 2]
    mov rax, qword ptr [r14 + rcx*8]
    cmp rax, qword ptr [r14 + rcx*8 + 8]
    setl al
    movzx rax, al
    mov qword ptr [r14 + rcx*8], rax
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    inc rbx
    jmp .Lloop
.Ljz:
    mov rax, qword ptr [r14 + r15*8 - 8]
    mov qword ptr [r14 + r15*8 - 8], 0
    dec r15
    test rax, rax
    jz .Ljz_taken
    add rbx, 2
    jmp .Lloop
.Ljz_taken:
    mov rbx, qword ptr [r12 + rbx*8 + 8]
    jmp .Lloop
.Ljmp:
    mov rbx, qword ptr [r12 + rbx*8 + 8]
    jmp .Lloop
.Ldone:
    xor eax, eax
    test r15, r15
    jz .Lrestore
    mov rax, qword ptr [r14 + r15*8 - 8]
.Lrestore:
    pop r15
    pop r14
    pop r13
    pop rbp
    pop rbx
    ret
