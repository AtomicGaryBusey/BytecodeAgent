packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
.Louter:
cmp rbx, r13
jae .Ldone
movzx r8d, byte ptr [r12 + rbx]
mov r10d, 1
lea r11, [rbx + 1]
.Lrun_scan:
cmp r10, 128
jae .Lrun_done
cmp r11, r13
jae .Lrun_done
movzx eax, byte ptr [r12 + r11]
cmp al, r8b
jne .Lrun_done
inc r10
inc r11
jmp .Lrun_scan
.Lrun_done:
cmp r10, 3
jb .Lliteral
mov rax, r15
sub rax, rbp
cmp rax, 2
jb .Lfail
mov eax, 257
sub eax, r10d
mov byte ptr [r14 + rbp], al
mov byte ptr [r14 + rbp + 1], r8b
add rbp, 2
add rbx, r10
jmp .Louter
.Lliteral:
xor r10d, r10d
mov r11, rbx
.Lliteral_scan:
cmp r10, 128
jae .Lliteral_done
cmp r11, r13
jae .Lliteral_done
mov rax, r13
sub rax, r11
cmp rax, 3
jb .Lliteral_consume
mov al, byte ptr [r12 + r11]
cmp al, byte ptr [r12 + r11 + 1]
jne .Lliteral_consume
cmp al, byte ptr [r12 + r11 + 2]
je .Lliteral_done
.Lliteral_consume:
inc r10
inc r11
jmp .Lliteral_scan
.Lliteral_done:
mov rax, r15
sub rax, rbp
lea rcx, [r10 + 1]
cmp rax, rcx
jb .Lfail
lea eax, [r10 - 1]
mov byte ptr [r14 + rbp], al
xor ecx, ecx
.Lliteral_copy:
cmp rcx, r10
jae .Lliteral_copied
lea rdx, [rbx + rcx]
mov al, byte ptr [r12 + rdx]
lea rdx, [rbp + rcx + 1]
mov byte ptr [r14 + rdx], al
inc rcx
jmp .Lliteral_copy
.Lliteral_copied:
add rbp, r10
inc rbp
add rbx, r10
jmp .Louter
.Lfail:
mov rax, -1
jmp .Lreturn
.Ldone:
mov rax, rbp
.Lreturn:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
