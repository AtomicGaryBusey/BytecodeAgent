export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %value =l sub %start, %i
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    storel %value, %ptr
    %next =l add %i, 1
    jmp @loop
@done
    ret %n
}
