export function l $collatz_max(l %n) {
@start
    %pk =l alloc8 8
    %px =l alloc8 8
    %psteps =l alloc8 8
    %pmax =l alloc8 8
    storel 1, %pk
    storel 0, %pmax
    jmp @outer

@outer
    %k =l loadl %pk
    %more =w cslel %k, %n
    jnz %more, @begin, @exit

@begin
    storel %k, %px
    storel 0, %psteps
    jmp @inner

@inner
    %x =l loadl %px
    %notone =w cnel %x, 1
    jnz %notone, @check, @finish

@check
    %bit =l and %x, 1
    %odd =w cnel %bit, 0
    jnz %odd, @oddstep, @evenstep

@oddstep
    %threex =l mul %x, 3
    %oddnext =l add %threex, 1
    storel %oddnext, %px
    %oddsteps =l loadl %psteps
    %oddstepsnext =l add %oddsteps, 1
    storel %oddstepsnext, %psteps
    jmp @inner

@evenstep
    %evennext =l div %x, 2
    storel %evennext, %px
    %evensteps =l loadl %psteps
    %evenstepsnext =l add %evensteps, 1
    storel %evenstepsnext, %psteps
    jmp @inner

@finish
    %steps =l loadl %psteps
    %oldmax =l loadl %pmax
    %newrecord =w csgtl %steps, %oldmax
    jnz %newrecord, @update, @advance

@update
    storel %steps, %pmax
    jmp @advance

@advance
    %nextk =l add %k, 1
    storel %nextk, %pk
    jmp @outer

@exit
    %result =l loadl %pmax
    ret %result
}
