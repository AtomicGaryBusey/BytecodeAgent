binary_search:
    xor rcx, rcx
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jae .Lnot_found
    mov r9, r8
    sub r9, rcx
    shr r9, 1
    add r9, rcx
    mov r10, qword ptr [rdi + r9*8]
    cmp r10, rdx
    jl .Lless
    jg .Lgreater
    mov rax, r9
    ret
.Lless:
    lea rcx, [r9 + 1]
    jmp .Lloop
.Lgreater:
    mov r8, r9
    jmp .Lloop
.Lnot_found:
    mov rax, -1
    ret
