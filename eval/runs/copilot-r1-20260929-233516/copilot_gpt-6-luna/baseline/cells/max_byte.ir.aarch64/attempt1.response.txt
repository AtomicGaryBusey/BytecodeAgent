export function l $max_byte(l %p, l %n) {
@entry
	jmp @check
@check
	%i =l phi @entry 0, @step %next
	%m =l phi @entry 0, @step %oldmax
	%cond =w csltl %i, %n
	jnz %cond, @body, @finish
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w csgtl %b, %m
	jnz %greater, @update, @keep
@update
	jmp @step
@keep
	jmp @step
@step
	%oldmax =l phi @update %b, @keep %m
	%next =l add %i, 1
	jmp @check
@finish
	ret %m
}
