export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop
@loop
    %i = phi @start 0, @body %i1
    %count = phi @start 0, @body %count1
    %more = csltl %i, %n
    jnz %more, @body, @done
@body
    %offset = mull %i, 8
    %ptr = add %a, %offset
    %value = loadl %ptr
    %greater = cultl %t, %value
    %increment = extuw %greater
    %count1 = add %count, %increment
    %i1 = add %i, 1
    jmp @loop
@done
    ret %count
}
