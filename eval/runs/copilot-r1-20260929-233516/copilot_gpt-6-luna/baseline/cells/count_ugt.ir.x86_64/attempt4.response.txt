export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %i1
    %count =l phi @start 0, @body %count1
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w cugtl %value, %t
    %increment =l extuw %greater
    %count1 =l add %count, %increment
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %count
}
