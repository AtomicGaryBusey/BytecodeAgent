export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    jmp @zero

@zero
    %zi =l phi @entry 0, @zero_body %zinext
    %zmore =w csltl %zi, %cap
    jnz %zmore, @zero_body, @init

@zero_body
    %zoff =l shl %zi, 3
    %zaddr =l add %stack, %zoff
    storel 0, %zaddr
    %zinext =l add %zi, 1
    jmp @zero

@init
    jmp @loop

@loop
    %pc =l phi @init 0, @next %npc
    %d =l phi @init 0, @next %nd
    %atend =w ceql %pc, %len
    jnz %atend, @finish, @fetch

@fetch
    %coff =l shl %pc, 3
    %caddr =l add %code, %coff
    %ins =l loadl %caddr
    %is0 =w ceql %ins, 0
    jnz %is0, @halt, @dispatch1

@dispatch1
    %is1 =w ceql %ins, 1
    jnz %is1, @push, @dispatch2

@dispatch2
    %is2 =w ceql %ins, 2
    jnz %is2, @add, @dispatch3

@dispatch3
    %is3 =w ceql %ins, 3
    jnz %is3, @sub, @dispatch4

@dispatch4
    %is4 =w ceql %ins, 4
    jnz %is4, @mul, @dispatch5

@dispatch5
    %is5 =w ceql %ins, 5
    jnz %is5, @dup, @dispatch6

@dispatch6
    %is6 =w ceql %ins, 6
    jnz %is6, @swap, @dispatch7

@dispatch7
    %is7 =w ceql %ins, 7
    jnz %is7, @drop, @dispatch8

@dispatch8
    %is8 =w ceql %ins, 8
    jnz %is8, @over, @dispatch9

@dispatch9
    %is9 =w ceql %ins, 9
    jnz %is9, @lt, @dispatch10

@dispatch10
    %is10 =w ceql %ins, 10
    jnz %is10, @jz, @dispatch11

@dispatch11
    %is11 =w ceql %ins, 11
    jnz %is11, @jmp, @other

@halt
    jmp @finish

@push
    %poperand =l add %pc, 1
    %poff =l shl %poperand, 3
    %paddr =l add %code, %poff
    %pvalue =l loadl %paddr
    %pstackoff =l shl %d, 3
    %pstackaddr =l add %stack, %pstackoff
    storel %pvalue, %pstackaddr
    %pnextpc =l add %pc, 2
    %pnextd =l add %d, 1
    jmp @next

@add
    %ai =l sub %d, 1
    %bi =l sub %d, 2
    %aoff =l shl %ai, 3
    %boff =l shl %bi, 3
    %aaddr =l add %stack, %boff
    %baddr =l add %stack, %aoff
    %av =l loadl %aaddr
    %bv =l loadl %baddr
    %sum =l add %av, %bv
    storel %sum, %aaddr
    storel 0, %baddr
    %anextpc =l add %pc, 1
    %anextd =l sub %d, 1
    jmp @next

@sub
    %si =l sub %d, 1
    %sbi =l sub %d, 2
    %soff =l shl %si, 3
    %sboff =l shl %sbi, 3
    %saddr =l add %stack, %sboff
    %sbaddr =l add %stack, %soff
    %sav =l loadl %saddr
    %sbv =l loadl %sbaddr
    %sdiff =l sub %sav, %sbv
    storel %sdiff, %saddr
    storel 0, %sbaddr
    %snextpc =l add %pc, 1
    %snextd =l sub %d, 1
    jmp @next

@mul
    %mi =l sub %d, 1
    %mbi =l sub %d, 2
    %moff =l shl %mi, 3
    %mboff =l shl %mbi, 3
    %maddr =l add %stack, %mboff
    %mbaddr =l add %stack, %moff
    %mav =l loadl %maddr
    %mbv =l loadl %mbaddr
    %product =l mul %mav, %mbv
    storel %product, %maddr
    storel 0, %mbaddr
    %mnextpc =l add %pc, 1
    %mnextd =l sub %d, 1
    jmp @next

@dup
    %di =l sub %d, 1
    %doff =l shl %di, 3
    %daddr =l add %stack, %doff
    %dvalue =l loadl %daddr
    %dstackoff =l shl %d, 3
    %dstackaddr =l add %stack, %dstackoff
    storel %dvalue, %dstackaddr
    %dnextpc =l add %pc, 1
    %dnextd =l add %d, 1
    jmp @next

@swap
    %swi =l sub %d, 1
    %swbi =l sub %d, 2
    %swoff =l shl %swi, 3
    %swboff =l shl %swbi, 3
    %swaddr =l add %stack, %swoff
    %swbaddr =l add %stack, %swboff
    %swv =l loadl %swaddr
    %swbv =l loadl %swbaddr
    storel %swv, %swbaddr
    storel %swbv, %swaddr
    %swnextpc =l add %pc, 1
    %swnextd =l copy %d
    jmp @next

@drop
    %dropi =l sub %d, 1
    %dropoff =l shl %dropi, 3
    %dropaddr =l add %stack, %dropoff
    storel 0, %dropaddr
    %dropnextpc =l add %pc, 1
    %dropnextd =l sub %d, 1
    jmp @next

@over
    %oi =l sub %d, 2
    %ooff =l shl %oi, 3
    %oaddr =l add %stack, %ooff
    %ovalue =l loadl %oaddr
    %ostackoff =l shl %d, 3
    %ostackaddr =l add %stack, %ostackoff
    storel %ovalue, %ostackaddr
    %onextpc =l add %pc, 1
    %onextd =l add %d, 1
    jmp @next

@lt
    %li =l sub %d, 1
    %lbi =l sub %d, 2
    %loff =l shl %li, 3
    %lboff =l shl %lbi, 3
    %laddr =l add %stack, %lboff
    %lbaddr =l add %stack, %loff
    %lav =l loadl %laddr
    %lbv =l loadl %lbaddr
    %less =w csltl %lav, %lbv
    %lresult =l extuw %less
    storel %lresult, %laddr
    storel 0, %lbaddr
    %lnextpc =l add %pc, 1
    %lnextd =l sub %d, 1
    jmp @next

@jz
    %ji =l sub %d, 1
    %joff =l shl %ji, 3
    %jaddr =l add %stack, %joff
    %jvalue =l loadl %jaddr
    storel 0, %jaddr
    %joperand =l add %pc, 1
    %jcoff =l shl %joperand, 3
    %jcaddr =l add %code, %jcoff
    %jtarget =l loadl %jcaddr
    %jzero =w ceql %jvalue, 0
    jnz %jzero, @jz_taken, @jz_skip

@jz_taken
    %jztd =l sub %d, 1
    jmp @next

@jz_skip
    %jznextpc =l add %pc, 2
    %jzsd =l sub %d, 1
    jmp @next

@jmp
    %jmpi =l add %pc, 1
    %jmpoff =l shl %jmpi, 3
    %jmpaddr =l add %code, %jmpoff
    %jmptarget =l loadl %jmpaddr
    %jmpnextd =l copy %d
    jmp @next

@other
    %othernextpc =l add %pc, 1
    %othernextd =l copy %d
    jmp @next

@next
    %npc =l phi @push %pnextpc, @add %anextpc, @sub %snextpc, @mul %mnextpc, @dup %dnextpc, @swap %swnextpc, @drop %dropnextpc, @over %onextpc, @lt %lnextpc, @jz_taken %jtarget, @jz_skip %jznextpc, @jmp %jmptarget, @other %othernextpc
    %nd =l phi @push %pnextd, @add %anextd, @sub %snextd, @mul %mnextd, @dup %dnextd, @swap %swnextd, @drop %dropnextd, @over %onextd, @lt %lnextd, @jz_taken %jztd, @jz_skip %jzsd, @jmp %jmpnextd, @other %othernextd
    jmp @loop

@finish
    %hasvalue =w csltl 0, %d
    jnz %hasvalue, @finish_load, @finish_empty

@finish_load
    %findex =l sub %d, 1
    %foff =l shl %findex, 3
    %faddr =l add %stack, %foff
    %fvalue =l loadl %faddr
    ret %fvalue

@finish_empty
    ret 0
}
