export function l $popcount(l %x) {
@start
    %px =l alloc8 8
    %pc =l alloc8 8
    storel %x, %px
    storel 0, %pc
    jmp @loop
@loop
    %v =l loadl %px
    %zero =w ceql %v, 0
    jnz %zero, @done, @body
@body
    %vm1 =l sub %v, 1
    %vnext =l and %v, %vm1
    storel %vnext, %px
    %count =l loadl %pc
    %countnext =l add %count, 1
    storel %countnext, %pc
    jmp @loop
@done
    %result =l loadl %pc
    ret %result
}
