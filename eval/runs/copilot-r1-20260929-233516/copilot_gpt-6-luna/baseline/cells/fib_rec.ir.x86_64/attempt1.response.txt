export function l $fib_rec(l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @checkone
@zero
	ret 0
@checkone
	%isone =w ceql %n, 1
	jnz %isone, @one, @recur
@one
	ret 1
@recur
	%n1 =l sub %n, 1
	%a =l call $fib_rec(l %n1)
	%n2 =l sub %n, 2
	%b =l call $fib_rec(l %n2)
	%sum =l add %a, %b
	ret %sum
}
