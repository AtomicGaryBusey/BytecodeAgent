export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %ip =l alloc8 8
    %minp =l alloc8 8
    %maxp =l alloc8 8
    %sump =l alloc8 8
    %countp =l alloc8 8
    storel 0, %ip
    storel 0, %sump
    storel 0, %countp
    jmp @loop

@loop
    %i =l loadl %ip
    %more =w csltl %i, %n
    jnz %more, @body, @done

@body
    %off =l shl %i, 3
    %elem =l add %a, %off
    %x =l loadl %elem
    %y =l call %f(l %x)
    %isfirst =w ceql %i, 0
    jnz %isfirst, @first, @checkmin

@first
    storel %y, %minp
    storel %y, %maxp
    jmp @accumulate

@checkmin
    %curmin =l loadl %minp
    %less =w csltl %y, %curmin
    jnz %less, @setmin, @checkmax

@setmin
    storel %y, %minp
    jmp @checkmax

@checkmax
    %curmax =l loadl %maxp
    %greater =w csltl %curmax, %y
    jnz %greater, @setmax, @accumulate

@setmax
    storel %y, %maxp
    jmp @accumulate

@accumulate
    %sum =l loadl %sump
    %newsum =l add %sum, %y
    storel %newsum, %sump
    %bits =l and %y, 1
    %odd =w cnel %bits, 0
    %count =l loadl %countp
    %oddlong =l extuw %odd
    %newcount =l add %count, %oddlong
    storel %newcount, %countp
    %nexti =l add %i, 1
    storel %nexti, %ip
    jmp @loop

@done
    %minval =l loadl %minp
    %maxval =l loadl %maxp
    %sumval =l loadl %sump
    %countval =l loadl %countp
    %out1 =l add %out, 8
    %out2 =l add %out, 16
    storel %minval, %out
    storel %maxval, %out1
    storel %sumval, %out2
    ret %countval
}
