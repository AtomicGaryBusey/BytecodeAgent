export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i_slot =l alloc8 8
    %o_slot =l alloc8 8
    %p_slot =l alloc8 8
    %r_slot =l alloc8 8
    %b_slot =l alloc8 8
    %j_slot =l alloc8 8
    %q_slot =l alloc8 8
    %rc_slot =l alloc8 8
    %len_slot =l alloc8 8
    %d_slot =l alloc8 8
    %end_slot =l alloc8 8
    storel 0, %i_slot
    storel 0, %o_slot
    jmp @outer

@outer
    %i0 =l loadl %i_slot
    %more =w csltl %i0, %n
    jnz %more, @measure_start, @done

@measure_start
    %p0 =l loadl %i_slot
    storel %p0, %p_slot
    %src_i =l add %src, %p0
    %b0 =w loadub %src_i
    storew %b0, %b_slot
    storew 0, %r_slot
    jmp @measure_check

@measure_check
    %p1 =l loadl %p_slot
    %r1 =w loadw %r_slot
    %in_bounds =w csltl %p1, %n
    jnz %in_bounds, @measure_limit, @measure_done

@measure_limit
    %under_limit =w csltw %r1, 128
    jnz %under_limit, @measure_read, @measure_done

@measure_read
    %src_p =l add %src, %p1
    %b1 =w loadub %src_p
    %b2 =w loadw %b_slot
    %same =w ceqw %b1, %b2
    jnz %same, @measure_match, @measure_done

@measure_match
    %p2 =l add %p1, 1
    %r2 =w add %r1, 1
    storel %p2, %p_slot
    storew %r2, %r_slot
    jmp @measure_check

@measure_done
    %r3 =w loadw %r_slot
    %is_repeat =w csgew %r3, 3
    jnz %is_repeat, @repeat_packet, @literal_start

@repeat_packet
    %r4 =w loadw %r_slot
    %r4l =l extuw %r4
    %packet_size =l add %r4l, 2
    %o0 =l loadl %o_slot
    %end0 =l add %o0, %packet_size
    %fits0 =w cslel %end0, %cap
    jnz %fits0, @repeat_write, @overflow

@repeat_write
    storel %end0, %end_slot
    %hdr0 =w sub 257, %r4
    %out0 =l add %out, %o0
    storeb %hdr0, %out0
    %out1 =l add %out0, 1
    %b3 =w loadw %b_slot
    storeb %b3, %out1
    %i1 =l loadl %i_slot
    %i2 =l add %i1, %r4l
    storel %i2, %i_slot
    storel %end0, %o_slot
    jmp @outer

@literal_start
    %i3 =l loadl %i_slot
    storel %i3, %j_slot
    storew 0, %len_slot
    jmp @literal_check

@literal_check
    %j0 =l loadl %j_slot
    %j_in_bounds =w csltl %j0, %n
    jnz %j_in_bounds, @literal_size_check, @literal_done

@literal_size_check
    %i4 =l loadl %i_slot
    %literal_size =l sub %j0, %i4
    %literal_room =w csltl %literal_size, 128
    jnz %literal_room, @literal_run_start, @literal_done

@literal_run_start
    %j1 =l loadl %j_slot
    storel %j1, %q_slot
    storew 0, %rc_slot
    %src_j =l add %src, %j1
    %bj =w loadub %src_j
    storew %bj, %b_slot
    jmp @literal_run_check

@literal_run_check
    %q0 =l loadl %q_slot
    %rc0 =w loadw %rc_slot
    %q_in_bounds =w csltl %q0, %n
    jnz %q_in_bounds, @literal_run_limit, @literal_run_finished

@literal_run_limit
    %rc_under_limit =w csltw %rc0, 3
    jnz %rc_under_limit, @literal_run_read, @literal_run_finished

@literal_run_read
    %src_q =l add %src, %q0
    %bq =w loadub %src_q
    %bj2 =w loadw %b_slot
    %q_same =w ceqw %bq, %bj2
    jnz %q_same, @literal_run_match, @literal_run_finished

@literal_run_match
    %q1 =l add %q0, 1
    %rc1 =w add %rc0, 1
    storel %q1, %q_slot
    storew %rc1, %rc_slot
    jmp @literal_run_check

@literal_run_finished
    %rc2 =w loadw %rc_slot
    %found_run =w csgew %rc2, 3
    jnz %found_run, @literal_done, @literal_advance

@literal_advance
    %j2 =l loadl %j_slot
    %j3 =l add %j2, 1
    storel %j3, %j_slot
    %len0 =w loadw %len_slot
    %len1 =w add %len0, 1
    storew %len1, %len_slot
    jmp @literal_check

@literal_done
    %len2 =w loadw %len_slot
    %len2l =l extuw %len2
    %literal_packet_size =l add %len2l, 1
    %o1 =l loadl %o_slot
    %end1 =l add %o1, %literal_packet_size
    %fits1 =w cslel %end1, %cap
    jnz %fits1, @literal_write_start, @overflow

@literal_write_start
    storel %end1, %end_slot
    %hdr1 =w sub %len2, 1
    %out2 =l add %out, %o1
    storeb %hdr1, %out2
    %i5 =l loadl %i_slot
    storel %i5, %p_slot
    %d0 =l add %o1, 1
    storel %d0, %d_slot
    jmp @literal_copy_check

@literal_copy_check
    %p3 =l loadl %p_slot
    %j4 =l loadl %j_slot
    %copy_more =w csltl %p3, %j4
    jnz %copy_more, @literal_copy, @literal_copy_done

@literal_copy
    %src_p2 =l add %src, %p3
    %copy_byte =w loadub %src_p2
    %d1 =l loadl %d_slot
    %out_p =l add %out, %d1
    storeb %copy_byte, %out_p
    %p4 =l add %p3, 1
    %d2 =l add %d1, 1
    storel %p4, %p_slot
    storel %d2, %d_slot
    jmp @literal_copy_check

@literal_copy_done
    %j5 =l loadl %j_slot
    storel %j5, %i_slot
    %end2 =l loadl %end_slot
    storel %end2, %o_slot
    jmp @outer

@overflow
    ret -1

@done
    %result =l loadl %o_slot
    ret %result
}
