export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	storel 0, %row
	jmp @init

@init
	%k =l phi @entry 1, @initstep %knext
	%ic =w cslel %k, %n
	jnz %ic, @initbody, @initdone

@initbody
	%koff =l shl %k, 3
	%kp =l add %row, %koff
	storel 0, %kp
	jmp @initstep

@initstep
	%knext =l add %k, 1
	jmp @init

@initdone
	jmp @outer

@outer
	%i =l phi @initdone 0, @outerstep %inext
	%oc =w csltl %i, %m
	jnz %oc, @outerbody, @done

@outerbody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outerbody 1, @innerstep %jnext
	%diag =l phi @outerbody 0, @innerstep %up
	%jc =w cslel %j, %n
	jnz %jc, @innerbody, @innerdone

@innerbody
	%joff =l sub %j, 1
	%yp =l add %y, %joff
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	%roff =l shl %j, 3
	%rp =l add %row, %roff
	%up =l loadl %rp
	jnz %eq, @match, @mismatch

@match
	%inc =l add %diag, 1
	jmp @update

@mismatch
	%leftp =l sub %rp, 8
	%left =l loadl %leftp
	%lt =w csltl %up, %left
	jnz %lt, @chooseleft, @chooseup

@chooseleft
	jmp @updateleft

@chooseup
	jmp @updateup

@updateleft
	jmp @update

@updateup
	jmp @update

@update
	%new =l phi @match %inc, @updateleft %left, @updateup %up
	storel %new, %rp
	jmp @innerstep

@innerstep
	%jnext =l add %j, 1
	jmp @inner

@innerdone
	jmp @outerstep

@outerstep
	%inext =l add %i, 1
	jmp @outer

@done
	%endoff =l shl %n, 3
	%endp =l add %row, %endoff
	%result =l loadl %endp
	ret %result
}
