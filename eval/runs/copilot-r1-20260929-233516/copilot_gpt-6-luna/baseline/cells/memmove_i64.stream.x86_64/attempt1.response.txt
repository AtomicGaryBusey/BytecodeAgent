memmove_i64:
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    ja .Lback
    xor ecx, ecx
.Lforward:
    cmp rcx, rdx
    jae .Ldone
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    jmp .Lforward
.Lback:
    mov rcx, rdx
.Lbackloop:
    dec rcx
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    test rcx, rcx
    jnz .Lbackloop
.Ldone:
    xor eax, eax
    ret
