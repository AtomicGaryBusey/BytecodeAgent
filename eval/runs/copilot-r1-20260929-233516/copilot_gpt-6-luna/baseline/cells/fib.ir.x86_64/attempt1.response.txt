export function l $fib(l %n) {
@start
    %zero =l copy 0
    %one =l copy 1
    jmp @loop
@loop
    %i =l phi @start %zero, @body %inext
    %a =l phi @start %zero, @body %bcur
    %bcur =l phi @start %one, @body %sum
    %cond =w csltl %i, %n
    jnz %cond, @body, @exit
@body
    %sum =l add %a, %bcur
    %inext =l add %i, %one
    jmp @loop
@exit
    ret %a
}
