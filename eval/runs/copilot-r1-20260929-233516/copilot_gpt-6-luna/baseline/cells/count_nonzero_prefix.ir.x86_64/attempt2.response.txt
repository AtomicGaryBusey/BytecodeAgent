export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @next %inc
	%p =l phi @entry %a, @next %pnext
	%more =w csltl %i, %n
	jnz %more, @check, @done
@check
	%v =l loadl %p
	%zero =w ceql %v, 0
	jnz %zero, @done, @next
@next
	%inc =l add %i, 1
	%pnext =l add %p, 8
	jmp @loop
@done
	ret %i
}
