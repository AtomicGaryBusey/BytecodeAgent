export function l $qsort_pairs(l %a, l %n) {
@start
	%i_slot =l alloc8 8
	%j_slot =l alloc8 8
	%pk_slot =l alloc8 8
	%pv_slot =l alloc8 8
	%left_slot =l alloc8 8
	%right_slot =l alloc8 8
	%small =w cslel %n, 1
	jnz %small, @ret, @init
@init
	%mid =l shr %n, 1
	%mid_off =l shl %mid, 4
	%mid_ptr =l add %a, %mid_off
	%pivot_key =l loadl %mid_ptr
	%mid_val_ptr =l add %mid_ptr, 8
	%pivot_val =l loadl %mid_val_ptr
	storel %pivot_key, %pk_slot
	storel %pivot_val, %pv_slot
	storel 0, %i_slot
	%last =l sub %n, 1
	storel %last, %j_slot
	jmp @outer
@outer
	%oi =l loadl %i_slot
	%oj =l loadl %j_slot
	%outer_cond =w cslel %oi, %oj
	jnz %outer_cond, @leftscan, @partitioned
@leftscan
	%li =l loadl %i_slot
	%lj =l loadl %j_slot
	%left_cond =w cslel %li, %lj
	jnz %left_cond, @leftcmp, @partitioned
@leftcmp
	%left_off =l shl %li, 4
	%left_ptr =l add %a, %left_off
	%left_key =l loadl %left_ptr
	%left_val_ptr =l add %left_ptr, 8
	%left_val =l loadl %left_val_ptr
	%pivot_key_l =l loadl %pk_slot
	%pivot_val_l =l loadl %pv_slot
	%left_key_less =w csltl %left_key, %pivot_key_l
	%left_key_eq =w ceql %left_key, %pivot_key_l
	%left_val_less =w csltl %left_val, %pivot_val_l
	%left_equal_less =w and %left_key_eq, %left_val_less
	%left_less =w or %left_key_less, %left_equal_less
	jnz %left_less, @leftinc, @rightscan
@leftinc
	%li_next =l add %li, 1
	storel %li_next, %i_slot
	jmp @leftscan
@rightscan
	%ri =l loadl %i_slot
	%rj =l loadl %j_slot
	%right_cond =w cslel %ri, %rj
	jnz %right_cond, @rightcmp, @partitioned
@rightcmp
	%right_off =l shl %rj, 4
	%right_ptr =l add %a, %right_off
	%right_key =l loadl %right_ptr
	%right_val_ptr =l add %right_ptr, 8
	%right_val =l loadl %right_val_ptr
	%pivot_key_r =l loadl %pk_slot
	%pivot_val_r =l loadl %pv_slot
	%right_key_greater =w csgtl %right_key, %pivot_key_r
	%right_key_eq =w ceql %right_key, %pivot_key_r
	%right_val_greater =w csgtl %right_val, %pivot_val_r
	%right_equal_greater =w and %right_key_eq, %right_val_greater
	%right_greater =w or %right_key_greater, %right_equal_greater
	jnz %right_greater, @rightdec, @swap
@rightdec
	%rj_next =l sub %rj, 1
	storel %rj_next, %j_slot
	jmp @rightscan
@swap
	storel %right_key, %left_ptr
	storel %right_val, %left_val_ptr
	storel %left_key, %right_ptr
	storel %left_val, %right_val_ptr
	%swap_i =l add %ri, 1
	%swap_j =l sub %rj, 1
	storel %swap_i, %i_slot
	storel %swap_j, %j_slot
	jmp @outer
@partitioned
	%part_j =l loadl %j_slot
	%left_count =l add %part_j, 1
	storel %left_count, %left_slot
	%left_big =w csgtl %left_count, 1
	jnz %left_big, @recleft, @rightcheck
@recleft
	%left_arg =l loadl %left_slot
	%left_result =l call $qsort_pairs(l %a, l %left_arg)
	jmp @rightcheck
@rightcheck
	%part_i =l loadl %i_slot
	%right_count =l sub %n, %part_i
	storel %right_count, %right_slot
	%right_big =w csgtl %right_count, 1
	jnz %right_big, @recright, @ret
@recright
	%right_arg =l loadl %right_slot
	%right_offset =l shl %part_i, 4
	%right_base =l add %a, %right_offset
	%right_result =l call $qsort_pairs(l %right_base, l %right_arg)
	jmp @ret
@ret
	ret 0
}
