export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %i_next
    %count =l phi @entry 0, @body %count_next
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %off =l shl %i, 3
    %addr =l add %a, %off
    %value =l loadl %addr
    %greater =w cugtl %value, %t
    %increment =l extuw %greater
    %count_next =l add %count, %increment
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %count
}
