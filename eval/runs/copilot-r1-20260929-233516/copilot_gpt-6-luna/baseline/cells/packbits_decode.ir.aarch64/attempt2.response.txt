export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %posp =l alloc8 8
    %writtenp =l alloc8 8
    %countp =l alloc8 8
    %kindp =l alloc8 8
    %srcposp =l alloc8 8
    %valuep =l alloc8 8
    %amountp =l alloc8 8
    %overflowp =l alloc8 8
    %ip =l alloc8 8
    storel 0, %posp
    storel 0, %writtenp
    jmp @packet

@packet
    %pos =l loadl %posp
    %haspacket =w csltl %pos, %n
    jnz %haspacket, @header, @done

@header
    %hp =l add %src, %pos
    %h =w loadsb %hp
    %hlong =l extsw %h
    %posafter =l add %pos, 1
    storel %posafter, %posp
    %noop =w ceql %hlong, -128
    jnz %noop, @packet, @classify

@classify
    %isrepeat =w csltl %hlong, 0
    jnz %isrepeat, @repeat, @literal

@literal
    %litcount =l add %hlong, 1
    %litavail =l sub %n, %posafter
    %litsourceok =w cslel %litcount, %litavail
    jnz %litsourceok, @literal_setup, @badsource

@literal_setup
    %litnext =l add %posafter, %litcount
    storel %litnext, %posp
    storel %litcount, %countp
    storel 0, %kindp
    storel %posafter, %srcposp
    jmp @prepare

@repeat
    %repcount =l sub 1, %hlong
    %repavail =w csltl %posafter, %n
    jnz %repavail, @repeat_read, @badsource

@repeat_read
    %repp =l add %src, %posafter
    %repvalue =w loadub %repp
    storew %repvalue, %valuep
    storel %repcount, %countp
    storel 1, %kindp
    storel 0, %srcposp
    %posnext =l add %posafter, 1
    storel %posnext, %posp
    jmp @prepare

@prepare
    %count =l loadl %countp
    %written =l loadl %writtenp
    %remaining =l sub %cap, %written
    %fits =w cslel %count, %remaining
    jnz %fits, @prepare_full, @prepare_partial

@prepare_full
    storel %count, %amountp
    storel 0, %overflowp
    jmp @write_init

@prepare_partial
    storel %remaining, %amountp
    storel 1, %overflowp
    jmp @write_init

@write_init
    storel 0, %ip
    jmp @write_loop

@write_loop
    %i =l loadl %ip
    %amount =l loadl %amountp
    %moreout =w csltl %i, %amount
    jnz %moreout, @write_byte, @write_done

@write_byte
    %kind =l loadl %kindp
    %isliteral =w ceql %kind, 0
    jnz %isliteral, @write_literal, @write_repeat

@write_literal
    %srcpos =l loadl %srcposp
    %readpos =l add %srcpos, %i
    %readptr =l add %src, %readpos
    %byte =w loadub %readptr
    jmp @store_byte

@write_repeat
    %byte =w loadw %valuep
    jmp @store_byte

@store_byte
    %writtennow =l loadl %writtenp
    %outpos =l add %writtennow, %i
    %outptr =l add %out, %outpos
    storeb %byte, %outptr
    %inext =l add %i, 1
    storel %inext, %ip
    jmp @write_loop

@write_done
    %overflow =w loadw %overflowp
    jnz %overflow, @full_output, @packet_done

@packet_done
    %oldwritten =l loadl %writtenp
    %packetcount =l loadl %countp
    %newwritten =l add %oldwritten, %packetcount
    storel %newwritten, %writtenp
    jmp @packet

@full_output
    ret -1

@badsource
    ret -2

@done
    %result =l loadl %writtenp
    ret %result
}
