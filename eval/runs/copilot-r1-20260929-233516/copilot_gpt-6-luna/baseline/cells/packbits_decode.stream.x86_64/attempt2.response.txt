packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movzx r10d, byte ptr [rdi + r8]
    inc r8
    cmp r10d, 128
    je .Lloop
    cmp r10d, 127
    jbe .Lliteral
    mov r11d, 257
    sub r11d, r10d
    mov rax, rsi
    sub rax, r8
    cmp rax, 1
    jb .Ltruncated
    movzx r10d, byte ptr [rdi + r8]
    inc r8
    mov rax, rcx
    sub rax, r9
    cmp r11, rax
    ja .Lrepeat_overflow
.Lrepeat_loop:
    test r11, r11
    jz .Lloop
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jmp .Lrepeat_loop
.Lrepeat_overflow:
    mov r11, rax
.Lrepeat_overflow_loop:
    test r11, r11
    jz .Loverflow
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jmp .Lrepeat_overflow_loop
.Lliteral:
    lea r11, [r10 + 1]
    mov rax, rsi
    sub rax, r8
    cmp rax, r11
    jb .Ltruncated
    mov rax, rcx
    sub rax, r9
    cmp r11, rax
    ja .Lliteral_overflow
.Lliteral_loop:
    test r11, r11
    jz .Lloop
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jmp .Lliteral_loop
.Lliteral_overflow:
    mov r11, rax
.Lliteral_overflow_loop:
    test r11, r11
    jz .Loverflow
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jmp .Lliteral_overflow_loop
.Ltruncated:
    mov rax, -2
    ret
.Loverflow:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
