export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inc
	%within =w csltl %i, %n
	jnz %within, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%nonzero =w cnel %value, 0
	jnz %nonzero, @next, @done
@next
	%inc =l add %i, 1
	jmp @loop
@done
	ret %i
}
