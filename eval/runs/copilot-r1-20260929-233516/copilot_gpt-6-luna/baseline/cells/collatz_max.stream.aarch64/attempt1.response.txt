collatz_max:
mov x1, #1
mov x2, #0
.Louter:
cmp x1, x0
b.gt .Ldone
mov x3, x1
mov x4, #0
.Linner:
cmp x3, #1
b.eq .Lfinished
tbz x3, #0, .Leven
add x3, x3, x3, lsl #1
add x3, x3, #1
add x4, x4, #1
b .Linner
.Leven:
lsr x3, x3, #1
add x4, x4, #1
b .Linner
.Lfinished:
cmp x4, x2
csel x2, x4, x2, gt
add x1, x1, #1
b .Louter
.Ldone:
mov x0, x2
ret
