export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
    %zero =l copy 0
    %backward =w csgtl %dst, %src
    jnz %backward, @back, @forward
@forward
    jmp @floop
@floop
    %fi =l phi @forward %zero, @fbody %finext
    %fcond =w csltl %fi, %n
    jnz %fcond, @fbody, @done
@fbody
    %foff =l mul %fi, 8
    %fsp =l add %src, %foff
    %fdp =l add %dst, %foff
    %fv =l loadl %fsp
    storel %fv, %fdp
    %finext =l add %fi, 1
    jmp @floop
@back
    jmp @bloop
@bloop
    %bi =l phi @back %n, @bbody %bprev
    %bcond =w csgtl %bi, %zero
    jnz %bcond, @bbody, @done
@bbody
    %bprev =l sub %bi, 1
    %boff =l mul %bprev, 8
    %bsp =l add %src, %boff
    %bdp =l add %dst, %boff
    %bv =l loadl %bsp
    storel %bv, %bdp
    jmp @bloop
@done
    ret %zero
}
