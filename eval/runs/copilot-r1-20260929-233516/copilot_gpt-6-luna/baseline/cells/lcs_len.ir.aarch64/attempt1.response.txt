export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	jmp @init

@init
	%ii =l phi @start %zero, @initbody %inext
	%initcond =w cslel %ii, %n
	jnz %initcond, @initbody, @outerentry

@initbody
	%ioff =l mul %ii, 8
	%iptr =l add %row, %ioff
	storel %zero, %iptr
	%inext =l add %ii, %one
	jmp @init

@outerentry
	jmp @outer

@outer
	%oi =l phi @outerentry %zero, @outerstep %onext
	%outercond =w csltl %oi, %m
	jnz %outercond, @outerbody, @done

@outerbody
	%xp =l add %x, %oi
	%xc =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outerbody %one, @write %jnext
	%diag =l phi @outerbody %zero, @write %up
	%innercond =w cslel %j, %n
	jnz %innercond, @innerbody, @outerstep

@innerbody
	%joff =l mul %j, 8
	%jptr =l add %row, %joff
	%up =l loadl %jptr
	%prevj =l sub %j, %one
	%prevoff =l mul %prevj, 8
	%prevptr =l add %row, %prevoff
	%left =l loadl %prevptr
	%yp =l add %y, %prevj
	%yc =w loadub %yp
	%equal =w ceqw %xc, %yc
	jnz %equal, @matched, @unmatched

@matched
	%dplus =l add %diag, %one
	jmp @write

@unmatched
	%leftgreater =w csltl %up, %left
	jnz %leftgreater, @takeleft, @takeup

@takeleft
	jmp @write

@takeup
	jmp @write

@write
	%newval =l phi @matched %dplus, @takeleft %left, @takeup %up
	storel %newval, %jptr
	%jnext =l add %j, %one
	jmp @inner

@outerstep
	%onext =l add %oi, %one
	jmp @outer

@done
	%resultoff =l mul %n, 8
	%resultptr =l add %row, %resultoff
	%result =l loadl %resultptr
	ret %result
}
