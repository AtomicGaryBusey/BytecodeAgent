memmove_i64:
cmp x0, x1
b.ls .Lforward
sub x3, x0, x1
lsl x4, x2, #3
cmp x3, x4
b.hs .Lforward
add x3, x1, x4
add x4, x0, x4
.Lbackward_loop:
cbz x2, .Ldone
ldr x5, [x3, #-8]!
str x5, [x4, #-8]!
subs x2, x2, #1
b.ne .Lbackward_loop
b .Ldone
.Lforward:
mov x3, x1
mov x4, x0
.Lforward_loop:
cbz x2, .Ldone
ldr x5, [x3], #8
str x5, [x4], #8
subs x2, x2, #1
b.ne .Lforward_loop
.Ldone:
mov x0, #0
ret
