export function l $bubble_sort(l %a, l %n) {
@entry
    jmp @outer

@outer
    %i =l phi @entry 0, @pass_done %i_next
    %outer_cond =w csltl %i, %n
    jnz %outer_cond, @setup, @done

@setup
    %n_minus_one =l sub %n, 1
    %limit =l sub %n_minus_one, %i
    jmp @inner

@inner
    %j =l phi @setup 0, @advance %j_next
    %inner_cond =w csltl %j, %limit
    jnz %inner_cond, @compare, @pass_done

@compare
    %offset =l mul %j, 8
    %p =l add %a, %offset
    %p_next =l add %p, 8
    %left =l loadl %p
    %right =l loadl %p_next
    %out_of_order =w csltl %right, %left
    jnz %out_of_order, @swap, @advance

@swap
    storel %right, %p
    storel %left, %p_next
    jmp @advance

@advance
    %j_next =l add %j, 1
    jmp @inner

@pass_done
    %i_next =l add %i, 1
    jmp @outer

@done
    ret 0
}
