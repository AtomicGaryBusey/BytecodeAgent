sdivrem:
mov rcx, rdx
test rsi, rsi
jz .Lzero
cmp rsi, -1
jne .Ldivide
mov rax, 0x8000000000000000
cmp rdi, rax
je .Loverflow
.Ldivide:
mov rax, rdi
cqo
idiv rsi
mov qword ptr [rcx], rdx
ret
.Lzero:
mov qword ptr [rcx], rdi
xor eax, eax
ret
.Loverflow:
mov qword ptr [rcx], 0
mov rax, 0x8000000000000000
ret
