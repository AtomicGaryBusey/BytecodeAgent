export function l $collatz_max(l %n) {
@entry
    %one =l copy 1
    %zero =l copy 0
    jmp @outer
@outer
    %i =l phi @entry %one, @next %i_next
    %best =l phi @entry %zero, @next %best_next
    %in_range =w cslel %i, %n
    jnz %in_range, @collatz, @done
@collatz
    %x =l phi @outer %i, @even %half, @odd %new_x
    %steps =l phi @outer %zero, @even %even_steps, @odd %odd_steps
    %is_one =w ceql %x, %one
    jnz %is_one, @collatz_done, @check_even
@check_even
    %parity =l andl %x, 1
    %is_even =w ceql %parity, %zero
    jnz %is_even, @even, @odd
@even
    %half =l shrl %x, 1
    %even_steps =l addl %steps, 1
    jmp @collatz
@odd
    %triple =l mull %x, 3
    %new_x =l addl %triple, 1
    %odd_steps =l addl %steps, 1
    jmp @collatz
@collatz_done
    %improved =w csltl %best, %steps
    jnz %improved, @use_steps, @keep_best
@use_steps
    jmp @best_merge
@keep_best
    jmp @best_merge
@best_merge
    %best_next =l phi @use_steps %steps, @keep_best %best
    %i_next =l addl %i, 1
    jmp @next
@next
    jmp @outer
@done
    ret %best
}
