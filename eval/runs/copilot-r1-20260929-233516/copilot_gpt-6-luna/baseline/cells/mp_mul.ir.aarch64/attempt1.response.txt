export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	jmp @zero

@zero
	%z =l phi @entry 0, @zero_body %zn
	%zcond =w csltl %z, %n
	jnz %zcond, @zero_body, @zero_done

@zero_body
	%zoff =l shl %z, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zn =l add %z, 1
	jmp @zero

@zero_done
	jmp @outer

@outer
	%i =l phi @zero_done 0, @outerinc %inext
	%icond =w csltl %i, %na
	jnz %icond, @inner_init, @scan_init

@inner_init
	jmp @inner

@inner
	%j =l phi @inner_init 0, @jinc %jnext
	%jcond =w csltl %j, %nb
	jnz %jcond, @mul, @outerinc

@mul
	%aoff =l shl %i, 3
	%aptr =l add %a, %aoff
	%aval =l loadl %aptr
	%boff =l shl %j, 3
	%bptr =l add %b, %boff
	%bval =l loadl %bptr

	%mask =l and %aval, 4294967295
	%ahi =l shr %aval, 32
	%bmask =l and %bval, 4294967295
	%bhi =l shr %bval, 32

	%p00 =l mul %mask, %bmask
	%p01 =l mul %mask, %bhi
	%p10 =l mul %ahi, %bmask
	%p11 =l mul %ahi, %bhi

	%p00lo =l and %p00, 4294967295
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p01hi =l shr %p01, 32
	%p10lo =l and %p10, 4294967295
	%p10hi =l shr %p10, 32
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%midlo =l and %mid, 4294967295
	%midshift =l shl %midlo, 32
	%lo =l or %p00lo, %midshift
	%midhi =l shr %mid, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%hi =l add %hi1, %midhi

	%k =l add %i, %j
	%roff =l shl %k, 3
	%rptr =l add %r, %roff
	%old =l loadl %rptr
	%sum =l add %old, %lo
	%c1 =w cultl %sum, %old
	storel %sum, %rptr

	%k1 =l add %k, 1
	%roff1 =l shl %k1, 3
	%rptr1 =l add %r, %roff1
	%old1 =l loadl %rptr1
	%sum1 =l add %old1, %hi
	%c2 =w cultl %sum1, %old1
	%c1l =l extuw %c1
	%sum2 =l add %sum1, %c1l
	%c3 =w cultl %sum2, %sum1
	%carry =w or %c2, %c3
	storel %sum2, %rptr1
	jnz %carry, @carry_start, @jinc

@carry_start
	%pstart =l add %k, 2
	jmp @prop

@prop
	%pidx =l phi @carry_start %pstart, @prop_more %pnext
	%poff =l shl %pidx, 3
	%pptr =l add %r, %poff
	%pold =l loadl %pptr
	%psum =l add %pold, 1
	%pcarry =w cultl %psum, %pold
	storel %psum, %pptr
	jnz %pcarry, @prop_more, @jinc

@prop_more
	%pnext =l add %pidx, 1
	jmp @prop

@jinc
	%jnext =l add %j, 1
	jmp @inner

@outerinc
	%inext =l add %i, 1
	jmp @outer

@scan_init
	jmp @scan

@scan
	%s =l phi @scan_init %n, @scan_zero %sprev
	%scond =w csltl 0, %s
	jnz %scond, @scan_check, @scan_empty

@scan_check
	%sidx =l sub %s, 1
	%soff =l shl %sidx, 3
	%sptr =l add %r, %soff
	%sval =l loadl %sptr
	%szero =w ceql %sval, 0
	jnz %szero, @scan_zero, @scan_return

@scan_zero
	%sprev =l sub %s, 1
	jmp @scan

@scan_return
	ret %s

@scan_empty
	ret 0
}
