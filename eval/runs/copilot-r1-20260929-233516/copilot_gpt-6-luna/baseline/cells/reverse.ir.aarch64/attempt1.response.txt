export function l $reverse(l %a, l %n) {
@entry
	%positive =w csltl 0, %n
	jnz %positive, @init, @done
@init
	%last =l sub %n, 1
	jmp @test
@test
	%i =l phi @init 0, @body %inext
	%j =l phi @init %last, @body %jnext
	%more =w csltl %i, %j
	jnz %more, @body, @done
@body
	%ioff =l mul %i, 8
	%iptr =l add %a, %ioff
	%joff =l mul %j, 8
	%jptr =l add %a, %joff
	%ival =l loadl %iptr
	%jval =l loadl %jptr
	storel %jval, %iptr
	storel %ival, %jptr
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @test
@done
	ret 0
}
