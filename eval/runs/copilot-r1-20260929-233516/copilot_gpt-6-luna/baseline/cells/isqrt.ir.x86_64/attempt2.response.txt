export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %mid, @upper %lo
	%hi =l phi @start 4294967296, @lower %hi, @upper %mid
	%gap =l sub %hi, %lo
	%done =w ceql %gap, 1
	jnz %done, @exit, @compute
@compute
	%half =l shr %gap, 1
	%mid =l add %lo, %half
	%square =l mul %mid, %mid
	%fits =w culel %square, %n
	jnz %fits, @lower, @upper
@lower
	jmp @loop
@upper
	jmp @loop
@exit
	ret %lo
}
