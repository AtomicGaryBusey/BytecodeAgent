collatz_max:
    mov rsi, 1
    xor r8d, r8d
.Louter:
    mov rdx, rsi
    xor ecx, ecx
.Linner:
    cmp rdx, 1
    je .Ldone_inner
    test rdx, 1
    jz .Leven
    lea rdx, [rdx + rdx*2 + 1]
    jmp .Lstep
.Leven:
    shr rdx, 1
.Lstep:
    inc rcx
    jmp .Linner
.Ldone_inner:
    cmp rcx, r8
    cmovg r8, rcx
    inc rsi
    cmp rsi, rdi
    jle .Louter
    mov rax, r8
    ret
