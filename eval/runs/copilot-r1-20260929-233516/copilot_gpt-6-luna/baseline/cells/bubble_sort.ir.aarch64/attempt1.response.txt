export function l $bubble_sort(l %a, l %n) {
@start
    %outer =l alloc8 8
    %inner =l alloc8 8
    %bound =l alloc8 8
    storel 0, %outer
    jmp @outer_check
@outer_check
    %i =l loadl %outer
    %i_next =l add %i, 1
    %outer_cond =w csltl %i_next, %n
    jnz %outer_cond, @inner_init, @done
@inner_init
    storel 0, %inner
    %bound_value =l sub %n, %i_next
    storel %bound_value, %bound
    jmp @inner_check
@inner_check
    %j =l loadl %inner
    %inner_bound =l loadl %bound
    %inner_cond =w csltl %j, %inner_bound
    jnz %inner_cond, @compare, @outer_advance
@compare
    %offset =l mul %j, 8
    %ptr =l add %a, %offset
    %next_ptr =l add %ptr, 8
    %left =l loadl %ptr
    %right =l loadl %next_ptr
    %should_swap =w csgtl %left, %right
    jnz %should_swap, @swap, @inner_advance
@swap
    storel %right, %ptr
    storel %left, %next_ptr
    jmp @inner_advance
@inner_advance
    %j_current =l loadl %inner
    %j_next =l add %j_current, 1
    storel %j_next, %inner
    jmp @inner_check
@outer_advance
    %i_current =l loadl %outer
    %i_incremented =l add %i_current, 1
    storel %i_incremented, %outer
    jmp @outer_check
@done
    ret 0
}
