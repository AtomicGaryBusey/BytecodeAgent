export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @zero, @loop
@zero
	ret 0
@loop
	%i =l phi @entry 0, @body %next
	%count =l phi @entry 0, @body %countnext
	%limit =l sub %n, 1
	%more =w csltl %i, %limit
	jnz %more, @body, @exit
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%pnext =l add %p, 8
	%x =l loadl %p
	%y =l loadl %pnext
	%desc =w csltl %y, %x
	%desc64 =l extuw %desc
	%countnext =l add %count, %desc64
	%next =l add %i, 1
	jmp @loop
@exit
	ret %count
}
