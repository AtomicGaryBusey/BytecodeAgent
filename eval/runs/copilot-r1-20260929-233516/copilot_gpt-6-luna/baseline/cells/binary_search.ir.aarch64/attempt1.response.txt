export function l $binary_search(l %a, l %n, l %key) {
@entry
    %zero =l copy 0
    %minusone =l copy -1
    jmp @loop
@loop
    %lo =l phi @entry %zero, @advance %nextlo, @shrink %lo
    %hi =l phi @entry %n, @advance %hi, @shrink %mid
    %more =w csltl %lo, %hi
    jnz %more, @search, @miss
@search
    %diff =l sub %hi, %lo
    %half =l shr %diff, 1
    %mid =l add %lo, %half
    %offset =l shl %mid, 3
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %equal =w ceql %value, %key
    jnz %equal, @found, @compare
@compare
    %less =w csltl %value, %key
    jnz %less, @advance, @shrink
@advance
    %nextlo =l add %mid, 1
    jmp @loop
@shrink
    jmp @loop
@found
    ret %mid
@miss
    ret %minusone
}
