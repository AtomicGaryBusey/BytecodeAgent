export function l $sdivrem(l %a, l %b, l %rem) {
@entry
	%bz = ceql %b, 0
	jnz %bz, @zero, @check

@check
	%amin = ceql %a, 0x8000000000000000
	%bneg = ceql %b, -1
	%ov = and %amin, %bneg
	jnz %ov, @overflow, @normal

@normal
	%q = div %a, %b
	%prod = mul %q, %b
	%r = sub %a, %prod
	storel %r, %rem
	ret %q

@zero
	storel %a, %rem
	ret 0

@overflow
	storel 0, %rem
	ret %a
}
