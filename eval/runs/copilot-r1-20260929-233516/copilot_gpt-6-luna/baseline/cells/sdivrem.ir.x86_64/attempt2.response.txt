export function l $sdivrem(l %a, l %b, l %rem) {
@entry
	%bz =w ceql %b, 0
	jnz %bz, @zero, @check

@check
	%amin =w ceql %a, -9223372036854775808
	%bneg =w ceql %b, -1
	%ov =w and %amin, %bneg
	jnz %ov, @overflow, @normal

@normal
	%q =l div %a, %b
	%prod =l mul %q, %b
	%r =l sub %a, %prod
	storel %r, %rem
	ret %q

@zero
	storel %a, %rem
	ret 0

@overflow
	storel 0, %rem
	ret %a
}
