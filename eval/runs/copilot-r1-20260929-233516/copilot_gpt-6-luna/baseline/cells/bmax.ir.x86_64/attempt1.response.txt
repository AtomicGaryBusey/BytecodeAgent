export function l $bmax(l %a, l %n) {
@entry
	%first =l loadl %a
	jmp @test
@test
	%i =l phi @entry 1, @advance %next
	%best =l phi @entry %first, @advance %new
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%greater =w csltl %best, %x
	jnz %greater, @replace, @keep
@replace
	jmp @advance
@keep
	jmp @advance
@advance
	%new =l phi @replace %x, @keep %best
	%next =l add %i, 1
	jmp @test
@done
	ret %best
}
