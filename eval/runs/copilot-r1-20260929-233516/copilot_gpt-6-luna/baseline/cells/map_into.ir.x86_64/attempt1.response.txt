export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%idxp =l alloc8 8
	%countp =l alloc8 8
	storel 0, %idxp
	storel 0, %countp
	jmp @check
@check
	%i =l loadl %idxp
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%srcaddr =l add %src, %off
	%dstaddr =l add %dst, %off
	%x =l loadl %srcaddr
	%y =l call %f(l %x)
	storel %y, %dstaddr
	%less =w csltl %y, %x
	jnz %less, @increment, @step
@increment
	%count =l loadl %countp
	%count1 =l add %count, 1
	storel %count1, %countp
	jmp @step
@step
	%nexti =l add %i, 1
	storel %nexti, %idxp
	jmp @check
@done
	%result =l loadl %countp
	ret %result
}
