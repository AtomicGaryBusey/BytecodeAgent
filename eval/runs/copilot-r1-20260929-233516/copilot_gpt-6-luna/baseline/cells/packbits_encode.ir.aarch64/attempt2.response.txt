export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @main

@main
	%i =l phi @entry 0, @after_repeat %nexti, @after_lit %lit_j
	%o =l phi @entry 0, @after_repeat %newo, @after_lit %lit_newo
	%more =w csltl %i, %n
	jnz %more, @have, @done

@done
	ret %o

@have
	%bp =l add %src, %i
	%b =w loadub %bp
	jmp @run_check

@run_check
	%rc =w phi @have 1, @run_inc %rcnext
	%under =w csltw %rc, 128
	jnz %under, @run_bound, @run_finish

@run_bound
	%rc_l =l extuw %rc
	%pos =l add %i, %rc_l
	%in_range =w csltl %pos, %n
	jnz %in_range, @run_load, @run_finish

@run_load
	%rp =l add %src, %pos
	%rb =w loadub %rp
	%same =w ceqw %rb, %b
	jnz %same, @run_inc, @run_finish

@run_inc
	%rcnext =w add %rc, 1
	jmp @run_check

@run_finish
	%finalrc =w phi @run_check %rc, @run_bound %rc, @run_load %rc
	%is_literal =w csltw %finalrc, 3
	jnz %is_literal, @lit_start, @repeat_check

@repeat_check
	%repeat_end =l add %o, 2
	%repeat_fits =w cslel %repeat_end, %cap
	jnz %repeat_fits, @repeat_write, @error

@repeat_write
	%header =w sub 257, %finalrc
	%hp =l add %out, %o
	storeb %header, %hp
	%vp =l add %hp, 1
	storeb %b, %vp
	jmp @after_repeat

@after_repeat
	%finalrc_l =l extuw %finalrc
	%nexti =l add %i, %finalrc_l
	%newo =l add %o, 2
	jmp @main

@lit_start
	jmp @lit_check

@lit_check
	%lj =l phi @lit_start %i, @lit_include %ljnext
	%ll =w phi @lit_start 0, @lit_include %llnext
	%below_limit =w csltw %ll, 128
	jnz %below_limit, @lit_check_input, @lit_finish

@lit_check_input
	%input_left =w csltl %lj, %n
	jnz %input_left, @lit_test, @lit_finish

@lit_test
	%third =l add %lj, 2
	%has_three =w csltl %third, %n
	jnz %has_three, @lit_triplet, @lit_include

@lit_triplet
	%t0p =l add %src, %lj
	%t1p =l add %t0p, 1
	%t2p =l add %t0p, 2
	%t0 =w loadub %t0p
	%t1 =w loadub %t1p
	%t2 =w loadub %t2p
	%eq01 =w ceqw %t0, %t1
	%eq12 =w ceqw %t1, %t2
	%is_run =w and %eq01, %eq12
	jnz %is_run, @lit_finish, @lit_include

@lit_include
	%ljnext =l add %lj, 1
	%llnext =w add %ll, 1
	jmp @lit_check

@lit_finish
	%lit_j =l phi @lit_check %lj, @lit_check_input %lj, @lit_triplet %lj
	%lit_len =w phi @lit_check %ll, @lit_check_input %ll, @lit_triplet %ll
	%lit_len_l =l extuw %lit_len
	%lit_size =l add %lit_len_l, 1
	%lit_end =l add %o, %lit_size
	%lit_fits =w cslel %lit_end, %cap
	jnz %lit_fits, @lit_write, @error

@lit_write
	%lit_header =w sub %lit_len, 1
	%lit_hp =l add %out, %o
	storeb %lit_header, %lit_hp
	jmp @lit_copy_check

@lit_copy_check
	%k =l phi @lit_write 0, @lit_copy_body %knext
	%lit_len_l_copy =l extuw %lit_len
	%copy_more =w csltl %k, %lit_len_l_copy
	jnz %copy_more, @lit_copy_body, @lit_copy_done

@lit_copy_body
	%src_index =l add %i, %k
	%srcp =l add %src, %src_index
	%byte =w loadub %srcp
	%out_index =l add %o, 1
	%out_index_k =l add %out_index, %k
	%outp =l add %out, %out_index_k
	storeb %byte, %outp
	%knext =l add %k, 1
	jmp @lit_copy_check

@lit_copy_done
	jmp @after_lit

@after_lit
	%lit_newo =l add %o, %lit_size
	jmp @main

@error
	ret -1
}
