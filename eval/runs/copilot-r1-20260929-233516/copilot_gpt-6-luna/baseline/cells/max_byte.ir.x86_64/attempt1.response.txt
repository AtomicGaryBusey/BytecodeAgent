export function l $max_byte(l %p, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @join %next
    %m =l phi @start 0, @join %newmax
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csgtl %b, %m
    jnz %gt, @update, @keep
@update
    jmp @join
@keep
    jmp @join
@join
    %newmax =l phi @update %b, @keep %m
    %next =l add %i, 1
    jmp @loop
@done
    ret %m
}
