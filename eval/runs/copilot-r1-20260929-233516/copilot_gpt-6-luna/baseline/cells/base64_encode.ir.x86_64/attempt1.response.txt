export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%np2 =l add %n, 2
	%groups =l udiv %np2, 3
	%len =l mul %groups, 4
	jmp @loop
@loop
	%j =l phi @entry 0, @next %jn
	%more =w csltl %j, %len
	jnz %more, @body, @done
@body
	%group =l shr %j, 2
	%base =l mul %group, 3
	%remain =l sub %n, %base
	%op =l add %out, %j
	%rem =l and %j, 3
	%is0 =w ceql %rem, 0
	jnz %is0, @case0, @check1
@check1
	%is1 =w ceql %rem, 1
	jnz %is1, @case1, @check2
@check2
	%is2 =w ceql %rem, 2
	jnz %is2, @case2, @case3
@case0
	%p0 =l add %src, %base
	%b0 =l loadub %p0
	%v0 =l shr %b0, 2
	jmp @map
@case1
	%p1 =l add %src, %base
	%b1a =l loadub %p1
	%short1 =w csltl %remain, 2
	jnz %short1, @case1one, @case1two
@case1one
	%lo1 =l and %b1a, 3
	%v1one =l shl %lo1, 4
	jmp @map
@case1two
	%p1b =l add %p1, 1
	%b1b =l loadub %p1b
	%hi1 =l shr %b1a, 2
	%lo1b =l and %b1b, 15
	%lo1c =l shl %lo1b, 4
	%v1two =l or %hi1, %lo1c
	jmp @map
@case2
	%short2 =w csltl %remain, 2
	jnz %short2, @pad2, @case2valid
@case2valid
	%p2a =l add %src, %base
	%b2a =l loadub %p2a
	%p2b =l add %p2a, 1
	%b2b =l loadub %p2b
	%lo2 =l and %b2a, 3
	%hi2 =l shl %lo2, 4
	%lo2b =l shr %b2b, 4
	%v2 =l or %hi2, %lo2b
	jmp @map
@pad2
	storeb 61, %op
	jmp @next
@case3
	%short3 =w csltl %remain, 3
	jnz %short3, @pad3, @case3valid
@case3valid
	%p3 =l add %src, %base
	%p3b =l add %p3, 2
	%b3 =l loadub %p3b
	%v3 =l and %b3, 63
	jmp @map
@pad3
	storeb 61, %op
	jmp @next
@map
	%val =l phi @case0 %v0, @case1one %v1one, @case1two %v1two, @case2valid %v2, @case3valid %v3
	%lt26 =w csltl %val, 26
	jnz %lt26, @mapA, @mapCheck52
@mapA
	%charA =l add %val, 65
	storeb %charA, %op
	jmp @next
@mapCheck52
	%lt52 =w csltl %val, 52
	jnz %lt52, @mapB, @mapCheck62
@mapB
	%charB =l add %val, 71
	storeb %charB, %op
	jmp @next
@mapCheck62
	%lt62 =w csltl %val, 62
	jnz %lt62, @mapC, @mapCheckPlus
@mapC
	%charC =l sub %val, 4
	storeb %charC, %op
	jmp @next
@mapCheckPlus
	%eq62 =w ceql %val, 62
	jnz %eq62, @mapPlus, @mapSlash
@mapPlus
	storeb 43, %op
	jmp @next
@mapSlash
	storeb 47, %op
	jmp @next
@next
	%jn =l add %j, 1
	jmp @loop
@done
	ret %len
}
