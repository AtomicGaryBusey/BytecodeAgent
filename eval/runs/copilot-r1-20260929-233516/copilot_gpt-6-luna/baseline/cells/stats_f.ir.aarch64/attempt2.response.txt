export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    jmp @loop
@loop
    %i =l phi @entry 1, @step %inext
    %mn =l phi @entry %y0, @step %mn2
    %mx =l phi @entry %y0, @step %mx2
    %sum =l phi @entry %y0, @step %sum2
    %odd =l phi @entry %odd0, @step %odd2
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %off =l shl %i, 3
    %addr =l add %a, %off
    %x =l loadl %addr
    %y =l call %f(l %x)
    %lt =w csltl %y, %mn
    jnz %lt, @setmin, @keepmin
@setmin
    jmp @minjoin
@keepmin
    jmp @minjoin
@minjoin
    %mn2 =l phi @setmin %y, @keepmin %mn
    %gt =w csgtl %y, %mx
    jnz %gt, @setmax, @keepmax
@setmax
    jmp @maxjoin
@keepmax
    jmp @maxjoin
@maxjoin
    %mx2 =l phi @setmax %y, @keepmax %mx
    %sum2 =l add %sum, %y
    %parity =l and %y, 1
    %odd2 =l add %odd, %parity
    jmp @step
@step
    %inext =l add %i, 1
    jmp @loop
@done
    storel %mn, %out
    %out1 =l add %out, 8
    storel %mx, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
