export function l $mulhi_u64(l %a, l %b) {
@start
    %mask =l copy 4294967295
    %a0 =l and %a, %mask
    %a1 =l shr %a, 32
    %b0 =l and %b, %mask
    %b1 =l shr %b, 32
    %p00 =l mul %a0, %b0
    %p00hi =l shr %p00, 32
    %p10 =l mul %a1, %b0
    %t =l add %p10, %p00hi
    %tlo =l and %t, %mask
    %thi =l shr %t, 32
    %p01 =l mul %a0, %b1
    %u =l add %tlo, %p01
    %carry =l shr %u, 32
    %p11 =l mul %a1, %b1
    %v =l add %p11, %thi
    %hi =l add %v, %carry
    ret %hi
}
