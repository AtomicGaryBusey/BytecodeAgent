export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %idx =l alloc8 8
    %i =l alloc8 8
    %j =l alloc8 8
    %t =l alloc8 8
    %exp =l alloc8 8
    %count =l alloc8 8
    %count2 =l alloc8 8
    %rowbytes =l alloc8 8
    %aptr =l alloc8 8
    %baseptr =l alloc8 8
    %outptr =l alloc8 8
    %rowptr =l alloc8 8
    %lhs =l alloc8 8
    %rhs =l alloc8 8
    %sum =l alloc8 8
    %prodptr =l alloc8 8
    %readptr =l alloc8 8
    %writeptr =l alloc8 8
    %off =l alloc8 8
    %v =l alloc8 8

    %cnt =l mul %n, %n
    %cb =l mul %n, 8
    %cnt2 =l mul %cnt, 2
    storel %cnt, %count
    storel %cnt2, %count2
    storel %cb, %rowbytes
    storel 0, %idx
    storel %a, %aptr
    storel %tmp, %baseptr

@base_copy
    %bi =l loadl %idx
    %bc =l loadl %count
    %bmore =w csltl %bi, %bc
    jnz %bmore, @base_copy_body, @init_start

@base_copy_body
    %bap =l loadl %aptr
    %bbp =l loadl %baseptr
    %bv =l loadl %bap
    storel %bv, %bbp
    %bapn =l add %bap, 8
    %bbpn =l add %bbp, 8
    %bin =l add %bi, 1
    storel %bapn, %aptr
    storel %bbpn, %baseptr
    storel %bin, %idx
    jmp @base_copy

@init_start
    storel 0, %i
    storel %out, %outptr
    jmp @init_i

@init_i
    %ii =l loadl %i
    %nin =w csltl %ii, %n
    jnz %nin, @init_row, @pow_start

@init_row
    storel 0, %j
    jmp @init_j

@init_j
    %jj =l loadl %j
    %jn =w csltl %jj, %n
    jnz %jn, @init_cell, @init_row_next

@init_cell
    %ieqj =w ceql %ii, %jj
    jnz %ieqj, @init_one, @init_zero

@init_one
    %op1 =l loadl %outptr
    storel 1, %op1
    jmp @init_advance

@init_zero
    %op0 =l loadl %outptr
    storel 0, %op0
    jmp @init_advance

@init_advance
    %op2 =l loadl %outptr
    %op3 =l add %op2, 8
    storel %op3, %outptr
    %jj1 =l add %jj, 1
    storel %jj1, %j
    jmp @init_j

@init_row_next
    %ii1 =l add %ii, 1
    storel %ii1, %i
    jmp @init_i

@pow_start
    storel %k, %exp
    jmp @pow_check

@pow_check
    %ex =l loadl %exp
    %positive =w csltl 0, %ex
    jnz %positive, @pow_setup, @cleanup_start

@pow_setup
    storel 0, %i
    %tmpoff =l mul %cnt, 8
    %pp =l add %tmp, %tmpoff
    storel %pp, %prodptr
    jmp @pow_i

@pow_i
    %pi =l loadl %i
    %pimore =w csltl %pi, %n
    jnz %pimore, @pow_row_setup, @copy_start

@pow_row_setup
    %rb =l loadl %rowbytes
    %roff =l mul %pi, %rb
    %rp =l add %out, %roff
    storel %rp, %rowptr
    storel 0, %j
    jmp @pow_j

@pow_j
    %pj =l loadl %j
    %pjmore =w csltl %pj, %n
    jnz %pjmore, @pow_cell_setup, @pow_row_next

@pow_cell_setup
    %pr =l loadl %rowptr
    storel %pr, %lhs
    %joff =l mul %pj, 8
    %bp =l loadl %tmp
    %rpj =l add %bp, %joff
    storel %rpj, %rhs
    storel 0, %t
    storel 0, %sum
    jmp @pow_t

@pow_t
    %pt =l loadl %t
    %ptmore =w csltl %pt, %n
    jnz %ptmore, @pow_multiply, @pow_store_cell

@pow_multiply
    %lp =l loadl %lhs
    %rp2 =l loadl %rhs
    %lv =l loadl %lp
    %rv =l loadl %rp2
    %product =l mul %lv, %rv
    %s =l loadl %sum
    %s1 =l add %s, %product
    storel %s1, %sum
    %lp1 =l add %lp, 8
    %rb2 =l loadl %rowbytes
    %rp3 =l add %rp2, %rb2
    %pt1 =l add %pt, 1
    storel %lp1, %lhs
    storel %rp3, %rhs
    storel %pt1, %t
    jmp @pow_t

@pow_store_cell
    %pptr =l loadl %prodptr
    %total =l loadl %sum
    storel %total, %pptr
    %pptr1 =l add %pptr, 8
    storel %pptr1, %prodptr
    %pj1 =l add %pj, 1
    storel %pj1, %j
    jmp @pow_j

@pow_row_next
    %pi1 =l add %pi, 1
    storel %pi1, %i
    jmp @pow_i

@copy_start
    %tmpoff2 =l mul %cnt, 8
    %src =l add %tmp, %tmpoff2
    storel %src, %readptr
    storel %out, %writeptr
    storel 0, %idx
    jmp @copy_loop

@copy_loop
    %ci =l loadl %idx
    %cc =l loadl %count
    %cmore =w csltl %ci, %cc
    jnz %cmore, @copy_body, @pow_finish

@copy_body
    %cr =l loadl %readptr
    %cw =l loadl %writeptr
    %cv =l loadl %cr
    storel %cv, %cw
    %cr1 =l add %cr, 8
    %cw1 =l add %cw, 8
    %ci1 =l add %ci, 1
    storel %cr1, %readptr
    storel %cw1, %writeptr
    storel %ci1, %idx
    jmp @copy_loop

@pow_finish
    %ex2 =l loadl %exp
    %ex3 =l sub %ex2, 1
    storel %ex3, %exp
    jmp @pow_check

@cleanup_start
    storel 0, %idx
    storel %tmp, %baseptr
    jmp @cleanup_loop

@cleanup_loop
    %zi =l loadl %idx
    %zc =l loadl %count2
    %zmore =w csltl %zi, %zc
    jnz %zmore, @cleanup_body, @done

@cleanup_body
    %zp =l loadl %baseptr
    storel 0, %zp
    %zp1 =l add %zp, 8
    %zi1 =l add %zi, 1
    storel %zp1, %baseptr
    storel %zi1, %idx
    jmp @cleanup_loop

@done
    ret 0
}
