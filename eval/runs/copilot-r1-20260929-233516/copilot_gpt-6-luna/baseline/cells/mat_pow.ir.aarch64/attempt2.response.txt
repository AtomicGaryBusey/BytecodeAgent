export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %islot =l alloc8 8
    %jslot =l alloc8 8
    %tslot =l alloc8 8
    %expslot =l alloc8 8
    %sumslot =l alloc8 8
    %idxslot =l alloc8 8
    %countslot =l alloc8 8

    %count =l mul %n, %n
    %count2 =l mul %count, 2
    storel %count, %countslot
    storel 0, %islot
    jmp @init_i

@init_i
    %init_i =l loadl %islot
    %init_i_more =w csltl %init_i, %n
    jnz %init_i_more, @init_j_start, @init_done

@init_j_start
    storel 0, %jslot
    jmp @init_j

@init_j
    %init_j =l loadl %jslot
    %init_j_more =w csltl %init_j, %n
    jnz %init_j_more, @init_cell, @init_i_next

@init_cell
    %init_equal =w ceql %init_i, %init_j
    jnz %init_equal, @init_one, @init_zero

@init_one
    %init_one_row =l mul %init_i, %n
    %init_one_index =l add %init_one_row, %init_j
    %init_one_offset =l mul %init_one_index, 8
    %init_one_ptr =l add %out, %init_one_offset
    storel 1, %init_one_ptr
    jmp @init_j_next

@init_zero
    %init_zero_row =l mul %init_i, %n
    %init_zero_index =l add %init_zero_row, %init_j
    %init_zero_offset =l mul %init_zero_index, 8
    %init_zero_ptr =l add %out, %init_zero_offset
    storel 0, %init_zero_ptr
    jmp @init_j_next

@init_j_next
    %init_j_next_value =l add %init_j, 1
    storel %init_j_next_value, %jslot
    jmp @init_j

@init_i_next
    %init_i_next_value =l add %init_i, 1
    storel %init_i_next_value, %islot
    jmp @init_i

@init_done
    storel %k, %expslot
    jmp @exp_check

@exp_check
    %exp_value =l loadl %expslot
    %exp_positive =w csltl 0, %exp_value
    jnz %exp_positive, @mult_i_start, @cleanup_start

@mult_i_start
    storel 0, %islot
    jmp @mult_i

@mult_i
    %mult_i_value =l loadl %islot
    %mult_i_more =w csltl %mult_i_value, %n
    jnz %mult_i_more, @mult_j_start, @copy_start

@mult_j_start
    storel 0, %jslot
    jmp @mult_j

@mult_j
    %mult_j_value =l loadl %jslot
    %mult_j_more =w csltl %mult_j_value, %n
    jnz %mult_j_more, @mult_t_start, @mult_i_next

@mult_t_start
    storel 0, %tslot
    storel 0, %sumslot
    jmp @mult_t

@mult_t
    %mult_t_value =l loadl %tslot
    %mult_t_more =w csltl %mult_t_value, %n
    jnz %mult_t_more, @mult_product, @mult_store

@mult_product
    %left_row =l mul %mult_i_value, %n
    %left_index =l add %left_row, %mult_t_value
    %left_offset =l mul %left_index, 8
    %left_ptr =l add %out, %left_offset
    %left_value =l loadl %left_ptr

    %right_row =l mul %mult_t_value, %n
    %right_index =l add %right_row, %mult_j_value
    %right_offset =l mul %right_index, 8
    %right_ptr =l add %a, %right_offset
    %right_value =l loadl %right_ptr

    %product =l mul %left_value, %right_value
    %old_sum =l loadl %sumslot
    %new_sum =l add %old_sum, %product
    storel %new_sum, %sumslot

    %next_t =l add %mult_t_value, 1
    storel %next_t, %tslot
    jmp @mult_t

@mult_store
    %store_row =l mul %mult_i_value, %n
    %store_index =l add %store_row, %mult_j_value
    %store_offset =l mul %store_index, 8
    %store_ptr =l add %tmp, %store_offset
    %store_sum =l loadl %sumslot
    storel %store_sum, %store_ptr

    %next_j =l add %mult_j_value, 1
    storel %next_j, %jslot
    jmp @mult_j

@mult_i_next
    %next_i =l add %mult_i_value, 1
    storel %next_i, %islot
    jmp @mult_i

@copy_start
    storel 0, %idxslot
    jmp @copy_loop

@copy_loop
    %copy_index =l loadl %idxslot
    %copy_more =w csltl %copy_index, %count
    jnz %copy_more, @copy_body, @copy_done

@copy_body
    %copy_offset =l mul %copy_index, 8
    %copy_src =l add %tmp, %copy_offset
    %copy_dst =l add %out, %copy_offset
    %copy_value =l loadl %copy_src
    storel %copy_value, %copy_dst
    %copy_next =l add %copy_index, 1
    storel %copy_next, %idxslot
    jmp @copy_loop

@copy_done
    %old_exp =l loadl %expslot
    %new_exp =l sub %old_exp, 1
    storel %new_exp, %expslot
    jmp @exp_check

@cleanup_start
    storel 0, %idxslot
    jmp @cleanup_loop

@cleanup_loop
    %cleanup_index =l loadl %idxslot
    %cleanup_more =w csltl %cleanup_index, %count2
    jnz %cleanup_more, @cleanup_body, @done

@cleanup_body
    %cleanup_offset =l mul %cleanup_index, 8
    %cleanup_ptr =l add %tmp, %cleanup_offset
    storel 0, %cleanup_ptr
    %cleanup_next =l add %cleanup_index, 1
    storel %cleanup_next, %idxslot
    jmp @cleanup_loop

@done
    ret 0
}
