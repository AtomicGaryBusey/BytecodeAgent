export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, %mask
	%p10lo =l and %p10, %mask
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%carry =l shr %mid, 32
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%hi =l add %hi1, %carry
	ret %hi
}
