export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%sip =l alloc8 8
	%oip =l alloc8 8
	%jp =l alloc8 8
	%cntp =l alloc8 8
	storel 0, %sip
	storel 0, %oip
	jmp @loop

@loop
	%si =l loadl %sip
	%more =w csltl %si, %n
	jnz %more, @have, @done

@have
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%neg =w csltw %h, 0
	jnz %neg, @negative, @literal

@literal
	%h64l =l extsw %h
	%lenl =l add %h64l, 1
	%needl =l add %si1, %lenl
	%badl =w csgtl %needl, %n
	jnz %badl, @err_src, @lit_src_ok

@lit_src_ok
	%oil =l loadl %oip
	%outneedl =l add %oil, %lenl
	%overl =w csgtl %outneedl, %cap
	jnz %overl, @lit_trunc, @lit_full

@lit_full
	storel 0, %jp
	storel %lenl, %cntp
	jmp @lit_full_check

@lit_full_check
	%jlf =l loadl %jp
	%cntlf =l loadl %cntp
	%cl =w csltl %jlf, %cntlf
	jnz %cl, @lit_full_copy, @lit_full_done

@lit_full_copy
	%silfc =l add %si1, %jlf
	%splfc =l add %src, %silfc
	%blfc =w loadub %splfc
	%oilfc =l add %oil, %jlf
	%oplfc =l add %out, %oilfc
	storeb %blfc, %oplfc
	%jlf1 =l add %jlf, 1
	storel %jlf1, %jp
	jmp @lit_full_check

@lit_full_done
	%si_l_done =l add %si1, %lenl
	%oi_l_done =l add %oil, %lenl
	storel %si_l_done, %sip
	storel %oi_l_done, %oip
	jmp @loop

@lit_trunc
	%avail_l =l sub %cap, %oil
	storel 0, %jp
	storel %avail_l, %cntp
	jmp @lit_trunc_check

@lit_trunc_check
	%jlt =l loadl %jp
	%cntlt =l loadl %cntp
	%clt =w csltl %jlt, %cntlt
	jnz %clt, @lit_trunc_copy, @err_cap

@lit_trunc_copy
	%siltc =l add %si1, %jlt
	%spltc =l add %src, %siltc
	%bltc =w loadub %spltc
	%oiltc =l add %oil, %jlt
	%opltc =l add %out, %oiltc
	storeb %bltc, %opltc
	%jlt1 =l add %jlt, 1
	storel %jlt1, %jp
	jmp @lit_trunc_check

@negative
	%noop =w ceqw %h, -128
	jnz %noop, @noop, @repeat

@noop
	storel %si1, %sip
	jmp @loop

@repeat
	%needr =l add %si1, 1
	%badr =w csgtl %needr, %n
	jnz %badr, @err_src, @rep_src_ok

@rep_src_ok
	%vptr =l add %src, %si1
	%val =w loadub %vptr
	%h64r =l extsw %h
	%lenr =l sub 1, %h64r
	%oir =l loadl %oip
	%outneedr =l add %oir, %lenr
	%overr =w csgtl %outneedr, %cap
	jnz %overr, @rep_trunc, @rep_full

@rep_full
	storel 0, %jp
	storel %lenr, %cntp
	jmp @rep_full_check

@rep_full_check
	%jrf =l loadl %jp
	%cntrf =l loadl %cntp
	%crf =w csltl %jrf, %cntrf
	jnz %crf, @rep_full_copy, @rep_full_done

@rep_full_copy
	%oirfc =l add %oir, %jrf
	%oprfc =l add %out, %oirfc
	storeb %val, %oprfc
	%jrf1 =l add %jrf, 1
	storel %jrf1, %jp
	jmp @rep_full_check

@rep_full_done
	%oi_r_done =l add %oir, %lenr
	storel %needr, %sip
	storel %oi_r_done, %oip
	jmp @loop

@rep_trunc
	%avail_r =l sub %cap, %oir
	storel 0, %jp
	storel %avail_r, %cntp
	jmp @rep_trunc_check

@rep_trunc_check
	%jrt =l loadl %jp
	%cntrt =l loadl %cntp
	%crt =w csltl %jrt, %cntrt
	jnz %crt, @rep_trunc_copy, @err_cap

@rep_trunc_copy
	%oirtc =l add %oir, %jrt
	%oprtc =l add %out, %oirtc
	storeb %val, %oprtc
	%jrt1 =l add %jrt, 1
	storel %jrt1, %jp
	jmp @rep_trunc_check

@done
	%res =l loadl %oip
	ret %res

@err_src
	ret -2

@err_cap
	ret -1
}
