	%m0a3 =l mul %m0e3, -15
	%m0d3 =l add %m0d2, %m0a3

	%m1c1 =w csltl 25, %i1
	%m1e1 =l extuw %m1c1

	%m3e1 =l extuw %m3c1
	%m3a1 =l mul %m3e1, 6
	%m3d1 =l add %m3b, %m3a1
	%m3c2 =w csltl 51, %i3

	%opn =l add %op, 4
	%remn =l sub %rem, 3
	%more =w csltl 0, %remn
	jnz %more, @afterfull, @donefull

@afterfull
	jmp @loop

@tail
