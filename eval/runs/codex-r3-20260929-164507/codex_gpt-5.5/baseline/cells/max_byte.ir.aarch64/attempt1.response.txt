export function l $max_byte(l %p, l %n) {
@start
	%z =w ceql %n, 0
	jnz %z, @ret0, @init
@init
	jmp @loop
@loop
	%i =l phi @init 0, @cont %inext
	%m =l phi @init 0, @cont %mnext
	%q =l add %p, %i
	%b =l loadub %q
	%gt =w csltl %m, %b
	jnz %gt, @useb, @usem
@useb
	jmp @after
@usem
	jmp @after
@after
	%mnext =l phi @useb %b, @usem %m
	%inext =l add %i, 1
	%done =w ceql %inext, %n
	jnz %done, @ret, @cont
@cont
	jmp @loop
@ret
	ret %mnext
@ret0
	ret 0
}
