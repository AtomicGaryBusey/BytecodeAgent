export function l $max_byte(l %p, l %n) {
@start
	%z =w ceql %n, 0
	jnz %z, @ret0, @loop

@loop
	%i =l phi @start 0, @next %i2
	%m =l phi @start 0, @next %m2
	%a =l add %p, %i
	%b =l loadub %a
	%c =w csltl %m, %b
	jnz %c, @set, @keep

@set
	jmp @next

@keep
	jmp @next

@next
	%m2 =l phi @set %b, @keep %m
	%i2 =l add %i, 1
	%d =w csltl %i2, %n
	jnz %d, @loop, @done

@done
	ret %m2

@ret0
	ret 0
}
