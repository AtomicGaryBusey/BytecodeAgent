export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	jmp @loop
@loop
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%io =l shl %i, 3
	%jo =l shl %j, 3
	%pi =l add %a, %io
	%pj =l add %a, %jo
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	%i =l copy %i1
	%j =l copy %j1
	jmp @loop
@done
	ret 0
}
