export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret, @init

@ret
	ret 0

@init
	jmp @outer

@outer
	%i =l phi @init 1, @place %inext
	%more =w csltl %i, %n
	jnz %more, @take, @done

@take
	%ioff =l shl %i, 4
	%ip =l add %a, %ioff
	%key =l loadl %ip
	%ipv =l add %ip, 8
	%val =l loadl %ipv
	jmp @while

@while
	%j =l phi @take %i, @shift %jm1
	%jpos =w csltl 0, %j
	jnz %jpos, @check, @place

@check
	%jm1 =l sub %j, 1
	%poff =l shl %jm1, 4
	%pp =l add %a, %poff
	%pk =l loadl %pp
	%ppv =l add %pp, 8
	%pv =l loadl %ppv
	%klt =w csltl %key, %pk
	%keq =w ceql %key, %pk
	%vlt =w csltl %val, %pv
	%eqvlt =w and %keq, %vlt
	%need =w or %klt, %eqvlt
	jnz %need, @shift, @place

@shift
	%doff =l shl %j, 4
	%dp =l add %a, %doff
	%dpv =l add %dp, 8
	storel %pk, %dp
	storel %pv, %dpv
	jmp @while

@place
	%pj =l phi @while %j, @check %j
	%ploff =l shl %pj, 4
	%plp =l add %a, %ploff
	%plpv =l add %plp, 8
	storel %key, %plp
	storel %val, %plpv
	%inext =l add %i, 1
	jmp @outer

@done
	%r =l call $qsort_pairs(l %a, l 0)
	ret 0
}
