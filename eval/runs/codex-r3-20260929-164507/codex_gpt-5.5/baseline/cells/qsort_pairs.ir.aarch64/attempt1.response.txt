export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret, @sort

@ret
	ret 0

@sort
	%mid =l shr %n, 1
	%moff =l shl %mid, 4
	%mp =l add %a, %moff
	%pk =l loadl %mp
	%mpv =l add %mp, 8
	%pv =l loadl %mpv
	jmp @scan_i

@scan_i
	%i0 =l phi @sort -1, @swap %isj, @scan_i_less %ikeep
	%j0 =l phi @sort %n, @swap %j1, @scan_i_less %jkeep
	%i1 =l add %i0, 1
	%ioff =l shl %i1, 4
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ipv =l add %ip, 8
	%iv =l loadl %ipv
	%ilk =w csltl %ik, %pk
	%ieq =w ceql %ik, %pk
	%ilv =w csltl %iv, %pv
	%il2 =w and %ieq, %ilv
	%iless =w or %ilk, %il2
	jnz %iless, @scan_i_less, @scan_j

@scan_i_less
	%ikeep =l add %i1, 0
	%jkeep =l add %j0, 0
	jmp @scan_i

@scan_j
	%isj =l phi @scan_i %i1, @scan_j_more %ikeep2
	%jsj =l phi @scan_i %j0, @scan_j_more %jkeep2
	%j1 =l sub %jsj, 1
	%joff =l shl %j1, 4
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jpv =l add %jp, 8
	%jv =l loadl %jpv
	%pglk =w csltl %pk, %jk
	%pgeq =w ceql %pk, %jk
	%pglv =w csltl %pv, %jv
	%pg2 =w and %pgeq, %pglv
	%pgreater =w or %pglk, %pg2
	jnz %pgreater, @scan_j_more, @check

@scan_j_more
	%ikeep2 =l add %isj, 0
	%jkeep2 =l add %j1, 0
	jmp @scan_j

@check
	%ltij =w csltl %isj, %j1
	jnz %ltij, @swap, @break

@swap
	%si =l shl %isj, 4
	%sp =l add %a, %si
	%spv =l add %sp, 8
	%sj =l shl %j1, 4
	%tp =l add %a, %sj
	%tpv =l add %tp, 8
	%ak =l loadl %sp
	%av =l loadl %spv
	%bk =l loadl %tp
	%bv =l loadl %tpv
	storel %bk, %sp
	storel %bv, %spv
	storel %ak, %tp
	storel %av, %tpv
	jmp @scan_i

@break
	%left =l add %j1, 1
	%r1 =l call $qsort_pairs(l %a, l %left)
	%loff =l shl %left, 4
	%ra =l add %a, %loff
	%right =l sub %n, %left
	%r2 =l call $qsort_pairs(l %ra, l %right)
	ret 0
}
