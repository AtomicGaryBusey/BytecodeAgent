export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zloop

@zloop
	%zi =l phi @start 0, @zbody %zi2
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @zdone

@zbody
	%zoff =l shl %zi, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zi2 =l add %zi, 1
	jmp @zloop

@zdone
	%az =w ceql %na, 0
	jnz %az, @ret0, @checknb

@checknb
	%bz =w ceql %nb, 0
	jnz %bz, @ret0, @outer

@ret0
	ret 0

@outer
	%i =l phi @checknb 0, @afterinner %i2
	%oc =w csltl %i, %na
	jnz %oc, @outerbody, @trim

@outerbody
	%aioff =l shl %i, 3
	%aiptr =l add %a, %aioff
	%ai =l loadl %aiptr
	jmp @inner

@inner
	%j =l phi @outerbody 0, @innerbody %j2
	%carry =l phi @outerbody 0, @innerbody %carry2
	%ic =w csltl %j, %nb
	jnz %ic, @innerbody, @afterinner

@innerbody
	%bjoff =l shl %j, 3
	%bjptr =l add %b, %bjoff
	%bj =l loadl %bjptr

	%a0 =l and %ai, 4294967295
	%a1 =l shr %ai, 32
	%b0 =l and %bj, 4294967295
	%b1 =l shr %bj, 32

	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1

	%p0h =l shr %p0, 32
	%p0l =l and %p0, 4294967295
	%p1l =l and %p1, 4294967295
	%p2l =l and %p2, 4294967295
	%t0 =l add %p0h, %p1l
	%t1 =l add %t0, %p2l
	%t1l =l shl %t1, 32
	%lo =l or %p0l, %t1l

	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%th =l shr %t1, 32
	%h0 =l add %p3, %p1h
	%h1 =l add %h0, %p2h
	%hi =l add %h1, %th

	%idx =l add %i, %j
	%roff =l shl %idx, 3
	%rptr =l add %r, %roff
	%old =l loadl %rptr

	%s1 =l add %lo, %old
	%c1w =w cultl %s1, %lo
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	storel %s2, %rptr

	%c1 =l extuw %c1w
	%c2 =l extuw %c2w
	%hc =l add %hi, %c1
	%carry2 =l add %hc, %c2
	%j2 =l add %j, 1
	jmp @inner

@afterinner
	%endidx =l add %i, %nb
	%endoff =l shl %endidx, 3
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%i2 =l add %i, 1
	jmp @outer

@trim
	%k =l phi @outer %n, @trimbody %km1
	%kg =w csgtl %k, 0
	jnz %kg, @trimcheck, @retk

@trimcheck
	%km =l sub %k, 1
	%koff =l shl %km, 3
	%kptr =l add %r, %koff
	%kv =l loadl %kptr
	%kz =w ceql %kv, 0
	jnz %kz, @trimbody, @retk

@trimbody
	%km1 =l sub %k, 1
	jmp @trim

@retk
	ret %k
}
