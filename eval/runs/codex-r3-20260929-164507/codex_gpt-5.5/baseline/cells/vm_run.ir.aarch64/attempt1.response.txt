export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop

@loop
	%pc =l phi @start 0, @def %npc_def, @op1 %npc1, @op2 %npc2, @op3 %npc3, @op4 %npc4, @op5 %npc5, @op6 %npc6, @op7 %npc7, @op8 %npc8, @op9 %npc9, @jz_zero %t10, @jz_nz %npc10, @op11 %t11
	%d =l phi @start 0, @def %d, @op1 %d1n, @op2 %d2n, @op3 %d3n, @op4 %d4n, @op5 %d5n, @op6 %d, @op7 %d7n, @op8 %d8n, @op9 %d9n, @jz_zero %d10n, @jz_nz %d10n, @op11 %d
	%atend =w ceql %pc, %len
	jnz %atend, @retprep, @fetch

@fetch
	%pcoff =l shl %pc, 3
	%ip =l add %code, %pcoff
	%op =l loadl %ip
	%c0 =w ceql %op, 0
	jnz %c0, @op0, @chk1

@chk1
	%c1 =w ceql %op, 1
	jnz %c1, @op1, @chk2

@chk2
	%c2 =w ceql %op, 2
	jnz %c2, @op2, @chk3

@chk3
	%c3 =w ceql %op, 3
	jnz %c3, @op3, @chk4

@chk4
	%c4 =w ceql %op, 4
	jnz %c4, @op4, @chk5

@chk5
	%c5 =w ceql %op, 5
	jnz %c5, @op5, @chk6

@chk6
	%c6 =w ceql %op, 6
	jnz %c6, @op6, @chk7

@chk7
	%c7 =w ceql %op, 7
	jnz %c7, @op7, @chk8

@chk8
	%c8 =w ceql %op, 8
	jnz %c8, @op8, @chk9

@chk9
	%c9 =w ceql %op, 9
	jnz %c9, @op9, @chk10

@chk10
	%c10 =w ceql %op, 10
	jnz %c10, @op10, @chk11

@chk11
	%c11 =w ceql %op, 11
	jnz %c11, @op11, @def

@def
	%npc_def =l add %pc, 1
	jmp @loop

@op0
	jmp @retprep

@op1
	%opa1 =l add %ip, 8
	%v1 =l loadl %opa1
	%soff1 =l shl %d, 3
	%sa1 =l add %stack, %soff1
	storel %v1, %sa1
	%d1n =l add %d, 1
	%npc1 =l add %pc, 2
	jmp @loop

@op2
	%db2 =l sub %d, 1
	%ob2 =l shl %db2, 3
	%ab2 =l add %stack, %ob2
	%b2 =l loadl %ab2
	%da2 =l sub %d, 2
	%oa2 =l shl %da2, 3
	%aa2 =l add %stack, %oa2
	%a2 =l loadl %aa2
	%r2 =l add %a2, %b2
	storel %r2, %aa2
	%d2n =l sub %d, 1
	%npc2 =l add %pc, 1
	jmp @loop

@op3
	%db3 =l sub %d, 1
	%ob3 =l shl %db3, 3
	%ab3 =l add %stack, %ob3
	%b3 =l loadl %ab3
	%da3 =l sub %d, 2
	%oa3 =l shl %da3, 3
	%aa3 =l add %stack, %oa3
	%a3 =l loadl %aa3
	%r3 =l sub %a3, %b3
	storel %r3, %aa3
	%d3n =l sub %d, 1
	%npc3 =l add %pc, 1
	jmp @loop

@op4
	%db4 =l sub %d, 1
	%ob4 =l shl %db4, 3
	%ab4 =l add %stack, %ob4
	%b4 =l loadl %ab4
	%da4 =l sub %d, 2
	%oa4 =l shl %da4, 3
	%aa4 =l add %stack, %oa4
	%a4 =l loadl %aa4
	%r4 =l mul %a4, %b4
	storel %r4, %aa4
	%d4n =l sub %d, 1
	%npc4 =l add %pc, 1
	jmp @loop

@op5
	%dt5 =l sub %d, 1
	%ot5 =l shl %dt5, 3
	%at5 =l add %stack, %ot5
	%v5 =l loadl %at5
	%od5 =l shl %d, 3
	%ad5 =l add %stack, %od5
	storel %v5, %ad5
	%d5n =l add %d, 1
	%npc5 =l add %pc, 1
	jmp @loop

@op6
	%dt6 =l sub %d, 1
	%ot6 =l shl %dt6, 3
	%at6 =l add %stack, %ot6
	%top6 =l loadl %at6
	%dn6 =l sub %d, 2
	%on6 =l shl %dn6, 3
	%an6 =l add %stack, %on6
	%next6 =l loadl %an6
	storel %top6, %an6
	storel %next6, %at6
	%npc6 =l add %pc, 1
	jmp @loop

@op7
	%d7n =l sub %d, 1
	%npc7 =l add %pc, 1
	jmp @loop

@op8
	%dn8 =l sub %d, 2
	%on8 =l shl %dn8, 3
	%an8 =l add %stack, %on8
	%v8 =l loadl %an8
	%od8 =l shl %d, 3
	%ad8 =l add %stack, %od8
	storel %v8, %ad8
	%d8n =l add %d, 1
	%npc8 =l add %pc, 1
	jmp @loop

@op9
	%db9 =l sub %d, 1
	%ob9 =l shl %db9, 3
	%ab9 =l add %stack, %ob9
	%b9 =l loadl %ab9
	%da9 =l sub %d, 2
	%oa9 =l shl %da9, 3
	%aa9 =l add %stack, %oa9
	%a9 =l loadl %aa9
	%cw9 =w csltl %a9, %b9
	%r9 =l extsw %cw9
	storel %r9, %aa9
	%d9n =l sub %d, 1
	%npc9 =l add %pc, 1
	jmp @loop

@op10
	%opa10 =l add %ip, 8
	%t10 =l loadl %opa10
	%d10n =l sub %d, 1
	%ov10 =l shl %d10n, 3
	%av10 =l add %stack, %ov10
	%v10 =l loadl %av10
	%z10 =w ceql %v10, 0
	jnz %z10, @jz_zero, @jz_nz

@jz_zero
	jmp @loop

@jz_nz
	%npc10 =l add %pc, 2
	jmp @loop

@op11
	%opa11 =l add %ip, 8
	%t11 =l loadl %opa11
	jmp @loop

@retprep
	%fd =l phi @loop %d, @op0 %d
	%has =w cnel %fd, 0
	jnz %has, @rv_load, @rv_zero

@rv_zero
	jmp @zero_loop

@rv_load
	%rdi =l sub %fd, 1
	%rof =l shl %rdi, 3
	%rad =l add %stack, %rof
	%top =l loadl %rad
	jmp @zero_loop

@zero_loop
	%i =l phi @rv_zero %fd, @rv_load %fd, @zero_body %in
	%rv =l phi @rv_zero 0, @rv_load %top, @zero_body %rv
	%more =w csltl %i, %cap
	jnz %more, @zero_body, @done

@zero_body
	%zof =l shl %i, 3
	%zad =l add %stack, %zof
	storel 0, %zad
	%in =l add %i, 1
	jmp @zero_loop

@done
	ret %rv
}
