export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init

@init
	%j0 =l phi @start 0, @init_body %j1
	%initc =w cslel %j0, %n
	jnz %initc, @init_body, @outer

@init_body
	%ioff =l shl %j0, 3
	%ip =l add %row, %ioff
	storel 0, %ip
	%j1 =l add %j0, 1
	jmp @init

@outer
	%i =l phi @init 0, @outer_next %i1
	%outerc =w csltl %i, %m
	jnz %outerc, @outer_body, @done

@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outer_body 1, @inner_next %jnext
	%diag =l phi @outer_body 0, @inner_next %up
	%innerc =w cslel %j, %n
	jnz %innerc, @inner_body, @outer_next

@inner_body
	%offj =l shl %j, 3
	%pj =l add %row, %offj
	%up =l loadl %pj
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch

@match
	%newm =l add %diag, 1
	storel %newm, %pj
	jmp @inner_next

@nomatch
	%offp =l shl %jm1, 3
	%pp =l add %row, %offp
	%left =l loadl %pp
	%lt =w csltl %up, %left
	jnz %lt, @store_left, @store_up

@store_left
	storel %left, %pj
	jmp @inner_next

@store_up
	storel %up, %pj
	jmp @inner_next

@inner_next
	%jnext =l add %j, 1
	jmp @inner

@outer_next
	%i1 =l add %i, 1
	jmp @outer

@done
	%roff =l shl %n, 3
	%rp =l add %row, %roff
	%res =l loadl %rp
	ret %res
}
