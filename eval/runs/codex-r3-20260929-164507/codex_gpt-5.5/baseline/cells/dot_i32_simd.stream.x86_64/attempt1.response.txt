dot_i32_simd:
pxor xmm0, xmm0
cmp rdx, 4
jl .Lscalar_prep
.Lvec_loop:
pmovsxdq xmm1, qword ptr [rdi]
pmovsxdq xmm2, qword ptr [rsi]
pmuldq xmm1, xmm2
paddq xmm0, xmm1
pmovsxdq xmm1, qword ptr [rdi + 8]
pmovsxdq xmm2, qword ptr [rsi + 8]
pmuldq xmm1, xmm2
paddq xmm0, xmm1
add rdi, 16
add rsi, 16
sub rdx, 4
cmp rdx, 4
jge .Lvec_loop
.Lscalar_prep:
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
test rdx, rdx
je .Ldone
.Lscalar_loop:
movsxd rcx, dword ptr [rdi]
movsxd r8, dword ptr [rsi]
imul rcx, r8
add rax, rcx
add rdi, 4
add rsi, 4
dec rdx
jne .Lscalar_loop
.Ldone:
ret
