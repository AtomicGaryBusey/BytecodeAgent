export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret0, @init

@ret0
	ret 0

@init
	%prev0 =l loadl %a
	jmp @loop

@loop
	%i =l phi @init 1, @cont %i2
	%cnt =l phi @init 0, @cont %cnt2
	%prev =l phi @init %prev0, @cont %curr
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%curr =l loadl %ptr
	%gt =w csgtl %prev, %curr
	jnz %gt, @inc, @noinc

@inc
	%cnti =l add %cnt, 1
	jmp @cont

@noinc
	jmp @cont

@cont
	%cnt2 =l phi @inc %cnti, @noinc %cnt
	%i2 =l add %i, 1
	%done =w ceql %i2, %n
	jnz %done, @ret, @loop

@ret
	ret %cnt2
}
