map_into:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor ebp, ebp
test r13, r13
jle .Ldone
.Lloop:
mov r15, qword ptr [r12]
mov rdi, r15
call r14
mov qword ptr [rbx], rax
cmp rax, r15
jge .Lno_count
inc rbp
.Lno_count:
add r12, 8
add rbx, 8
dec r13
jne .Lloop
.Ldone:
mov rax, rbp
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
