export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop

@loop
	%pc =l phi @start 0, @op_default %pc_def, @op_push %pc_push, @op_add %pc_add, @op_sub %pc_sub, @op_mul %pc_mul, @op_dup %pc_dup, @op_swap %pc_swap, @op_drop %pc_drop, @op_over %pc_over, @op_lt %pc_lt, @op_jz_z %pc_jz_z, @op_jz_nz %pc_jz_nz, @op_jmp %pc_jmp
	%sp =l phi @start 0, @op_default %sp_def, @op_push %sp_push, @op_add %sp_add, @op_sub %sp_sub, @op_mul %sp_mul, @op_dup %sp_dup, @op_swap %sp_swap, @op_drop %sp_drop, @op_over %sp_over, @op_lt %sp_lt, @op_jz_z %sp_jz, @op_jz_nz %sp_jz, @op_jmp %sp_jmp
	%atend =w ceql %pc, %len
	jnz %atend, @end, @fetch

@fetch
	%pcoff =l shl %pc, 3
	%ip =l add %code, %pcoff
	%op =l loadl %ip
	%is0 =w ceql %op, 0
	jnz %is0, @end, @chk1

@chk1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @chk2

@chk2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @chk3

@chk3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @chk4

@chk4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @chk5

@chk5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @chk6

@chk6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @chk7

@chk7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @chk8

@chk8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @chk9

@chk9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @chk10

@chk10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @chk11

@chk11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_default

@op_default
	%pc_def =l add %pc, 1
	%sp_def =l add %sp, 0
	jmp @loop

@op_push
	%ap1 =l add %pc, 1
	%ao1 =l shl %ap1, 3
	%avp =l add %code, %ao1
	%vpush =l loadl %avp
	%so_push =l shl %sp, 3
	%sd_push =l add %stack, %so_push
	storel %vpush, %sd_push
	%pc_push =l add %pc, 2
	%sp_push =l add %sp, 1
	jmp @loop

@op_add
	%spm1a =l sub %sp, 1
	%spm2a =l sub %sp, 2
	%o1a =l shl %spm1a, 3
	%o2a =l shl %spm2a, 3
	%p1a =l add %stack, %o1a
	%p2a =l add %stack, %o2a
	%ba =l loadl %p1a
	%aa =l loadl %p2a
	%ra =l add %aa, %ba
	storel %ra, %p2a
	%pc_add =l add %pc, 1
	%sp_add =l sub %sp, 1
	jmp @loop

@op_sub
	%spm1s =l sub %sp, 1
	%spm2s =l sub %sp, 2
	%o1s =l shl %spm1s, 3
	%o2s =l shl %spm2s, 3
	%p1s =l add %stack, %o1s
	%p2s =l add %stack, %o2s
	%bs =l loadl %p1s
	%as =l loadl %p2s
	%rs =l sub %as, %bs
	storel %rs, %p2s
	%pc_sub =l add %pc, 1
	%sp_sub =l sub %sp, 1
	jmp @loop

@op_mul
	%spm1m =l sub %sp, 1
	%spm2m =l sub %sp, 2
	%o1m =l shl %spm1m, 3
	%o2m =l shl %spm2m, 3
	%p1m =l add %stack, %o1m
	%p2m =l add %stack, %o2m
	%bm =l loadl %p1m
	%am =l loadl %p2m
	%rm =l mul %am, %bm
	storel %rm, %p2m
	%pc_mul =l add %pc, 1
	%sp_mul =l sub %sp, 1
	jmp @loop

@op_dup
	%spm1d =l sub %sp, 1
	%o1d =l shl %spm1d, 3
	%odd =l shl %sp, 3
	%p1d =l add %stack, %o1d
	%pdd =l add %stack, %odd
	%vd =l loadl %p1d
	storel %vd, %pdd
	%pc_dup =l add %pc, 1
	%sp_dup =l add %sp, 1
	jmp @loop

@op_swap
	%spm1w =l sub %sp, 1
	%spm2w =l sub %sp, 2
	%o1w =l shl %spm1w, 3
	%o2w =l shl %spm2w, 3
	%p1w =l add %stack, %o1w
	%p2w =l add %stack, %o2w
	%v1w =l loadl %p1w
	%v2w =l loadl %p2w
	storel %v2w, %p1w
	storel %v1w, %p2w
	%pc_swap =l add %pc, 1
	%sp_swap =l add %sp, 0
	jmp @loop

@op_drop
	%pc_drop =l add %pc, 1
	%sp_drop =l sub %sp, 1
	jmp @loop

@op_over
	%spm2o =l sub %sp, 2
	%o2o =l shl %spm2o, 3
	%odo =l shl %sp, 3
	%p2o =l add %stack, %o2o
	%pdo =l add %stack, %odo
	%vo =l loadl %p2o
	storel %vo, %pdo
	%pc_over =l add %pc, 1
	%sp_over =l add %sp, 1
	jmp @loop

@op_lt
	%spm1l =l sub %sp, 1
	%spm2l =l sub %sp, 2
	%o1l =l shl %spm1l, 3
	%o2l =l shl %spm2l, 3
	%p1l =l add %stack, %o1l
	%p2l =l add %stack, %o2l
	%bl =l loadl %p1l
	%al =l loadl %p2l
	%cl =w csltl %al, %bl
	%rl =l extuw %cl
	storel %rl, %p2l
	%pc_lt =l add %pc, 1
	%sp_lt =l sub %sp, 1
	jmp @loop

@op_jz
	%spm1j =l sub %sp, 1
	%o1j =l shl %spm1j, 3
	%p1j =l add %stack, %o1j
	%vj =l loadl %p1j
	%sp_jz =l sub %sp, 1
	%apj =l add %pc, 1
	%aoj =l shl %apj, 3
	%ptj =l add %code, %aoj
	%tj =l loadl %ptj
	%cz =w ceql %vj, 0
	jnz %cz, @op_jz_z, @op_jz_nz

@op_jz_z
	%pc_jz_z =l add %tj, 0
	jmp @loop

@op_jz_nz
	%pc_jz_nz =l add %pc, 2
	jmp @loop

@op_jmp
	%apm =l add %pc, 1
	%aom =l shl %apm, 3
	%ptm =l add %code, %aom
	%pc_jmp =l loadl %ptm
	%sp_jmp =l add %sp, 0
	jmp @loop

@end
	jmp @zero_loop

@zero_loop
	%zi =l phi @end %sp, @zero_body %zin
	%zc =w csltl %zi, %cap
	jnz %zc, @zero_body, @retblk

@zero_body
	%zo =l shl %zi, 3
	%zp =l add %stack, %zo
	storel 0, %zp
	%zin =l add %zi, 1
	jmp @zero_loop

@retblk
	%empty =w ceql %sp, 0
	jnz %empty, @retzero, @rettop

@retzero
	ret 0

@rettop
	%rtidx =l sub %sp, 1
	%rto =l shl %rtidx, 3
	%rtp =l add %stack, %rto
	%rv =l loadl %rtp
	ret %rv
}
