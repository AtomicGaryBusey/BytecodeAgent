export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i1
	%cnt =l phi @start 0, @cont %cnt1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ps =l add %src, %off
	%x =l loadl %ps
	%y =l call %f(l %x)
	%pd =l add %dst, %off
	storel %y, %pd
	%less =w csltl %y, %x
	jnz %less, @inc, @noinc
@inc
	%cnti =l add %cnt, 1
	jmp @cont
@noinc
	jmp @cont
@cont
	%cnt1 =l phi @inc %cnti, @noinc %cnt
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
