mp_mul:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r13, rsi
mov rbp, rdx
mov r14, rcx
mov r12, r8
mov r15, r13
add r15, r14
xor ecx, ecx
.Lzero_loop:
cmp rcx, r15
jge .Lzero_done
mov qword ptr [r12 + rcx*8], 0
inc rcx
jmp .Lzero_loop
.Lzero_done:
test r13, r13
je .Lreturn_zero
test r14, r14
je .Lreturn_zero
xor esi, esi
.Louter_loop:
cmp rsi, r13
jge .Lscan_start
mov rdx, qword ptr [rbx + rsi*8]
xor r9d, r9d
xor r10d, r10d
xor ecx, ecx
.Linner_loop:
cmp rcx, r14
jge .Linner_done
mulx rax, r8, qword ptr [rbp + rcx*8]
lea rdi, [rsi + rcx]
xor r11d, r11d
add rax, qword ptr [r12 + rdi*8]
adc r11, 0
add rax, r9
adc r11, 0
add r11, r10
mov qword ptr [r12 + rdi*8], rax
mov r9, r8
xor r10d, r10d
add r9, r11
adc r10, 0
inc rcx
jmp .Linner_loop
.Linner_done:
lea rdi, [rsi + r14]
.Lcarry_loop:
mov rax, r9
or rax, r10
je .Lnext_outer
xor r11d, r11d
add qword ptr [r12 + rdi*8], r9
adc r11, 0
add r11, r10
mov r9, r11
xor r10d, r10d
inc rdi
jmp .Lcarry_loop
.Lnext_outer:
inc rsi
jmp .Louter_loop
.Lscan_start:
mov rax, r15
.Lscan_loop:
test rax, rax
je .Lreturn
cmp qword ptr [r12 + rax*8 - 8], 0
jne .Lreturn
dec rax
jmp .Lscan_loop
.Lreturn_zero:
xor eax, eax
.Lreturn:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
