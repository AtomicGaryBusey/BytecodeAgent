export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop

@loop
	%sp =l phi @start %src, @full %spn
	%op =l phi @start %out, @full %opn
	%rem =l phi @start %n, @full %remn
	%has3 =w csgel %rem, 3
	jnz %has3, @full, @tail

@full
	%p1 =l add %sp, 1
	%p2 =l add %sp, 2
	%b0 =l loadub %sp
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%i0 =l shr %b0, 2
	%a10 =l and %b0, 3
	%a11 =l shl %a10, 4
	%a12 =l shr %b1, 4
	%i1 =l or %a11, %a12
	%a20 =l and %b1, 15
	%a21 =l shl %a20, 2
	%a22 =l shr %b2, 6
	%i2 =l or %a21, %a22
	%i3 =l and %b2, 63

	%g026 =w csgel %i0, 26
	%g052 =w csgel %i0, 52
	%g062 =w csgel %i0, 62
	%e026 =l extuw %g026
	%e052 =l extuw %g052
	%e062 =l extuw %g062
	%m026 =l mul %e026, 6
	%m052 =l mul %e052, 75
	%tv0 =l mul %i0, 3
	%tv1 =l sub %tv0, 201
	%m062 =l mul %e062, %tv1
	%c00 =l add %i0, 65
	%c01 =l add %c00, %m026
	%c02 =l sub %c01, %m052
	%c0 =l add %c02, %m062

	%g126 =w csgel %i1, 26
	%g152 =w csgel %i1, 52
	%g162 =w csgel %i1, 62
	%e126 =l extuw %g126
	%e152 =l extuw %g152
	%e162 =l extuw %g162
	%m126 =l mul %e126, 6
	%m152 =l mul %e152, 75
	%tv10 =l mul %i1, 3
	%tv11 =l sub %tv10, 201
	%m162 =l mul %e162, %tv11
	%c10 =l add %i1, 65
	%c11 =l add %c10, %m126
	%c12 =l sub %c11, %m152
	%c1 =l add %c12, %m162

	%g226 =w csgel %i2, 26
	%g252 =w csgel %i2, 52
	%g262 =w csgel %i2, 62
	%e226 =l extuw %g226
	%e252 =l extuw %g252
	%e262 =l extuw %g262
	%m226 =l mul %e226, 6
	%m252 =l mul %e252, 75
	%tv20 =l mul %i2, 3
	%tv21 =l sub %tv20, 201
	%m262 =l mul %e262, %tv21
	%c20 =l add %i2, 65
	%c21 =l add %c20, %m226
	%c22 =l sub %c21, %m252
	%c2 =l add %c22, %m262

	%g326 =w csgel %i3, 26
	%g352 =w csgel %i3, 52
	%g362 =w csgel %i3, 62
	%e326 =l extuw %g326
	%e352 =l extuw %g352
	%e362 =l extuw %g362
	%m326 =l mul %e326, 6
	%m352 =l mul %e352, 75
	%tv30 =l mul %i3, 3
	%tv31 =l sub %tv30, 201
	%m362 =l mul %e362, %tv31
	%c30 =l add %i3, 65
	%c31 =l add %c30, %m326
	%c32 =l sub %c31, %m352
	%c3 =l add %c32, %m362

	storeb %c0, %op
	%o1 =l add %op, 1
	storeb %c1, %o1
	%o2 =l add %op, 2
	storeb %c2, %o2
	%o3 =l add %op, 3
	storeb %c3, %o3

	%spn =l add %sp, 3
	%opn =l add %op, 4
	%remn =l sub %rem, 3
	jmp @loop

@tail
	%is0 =w ceql %rem, 0
	jnz %is0, @ret, @tailnz

@tailnz
	%b0t =l loadub %sp
	%ti0 =l shr %b0t, 2

	%tg026 =w csgel %ti0, 26
	%tg052 =w csgel %ti0, 52
	%tg062 =w csgel %ti0, 62
	%te026 =l extuw %tg026
	%te052 =l extuw %tg052
	%te062 =l extuw %tg062
	%tm026 =l mul %te026, 6
	%tm052 =l mul %te052, 75
	%ttv0 =l mul %ti0, 3
	%ttv1 =l sub %ttv0, 201
	%tm062 =l mul %te062, %ttv1
	%tc00 =l add %ti0, 65
	%tc01 =l add %tc00, %tm026
	%tc02 =l sub %tc01, %tm052
	%tc0 =l add %tc02, %tm062
	storeb %tc0, %op

	%is1 =w ceql %rem, 1
	jnz %is1, @tail1, @tail2

@tail1
	%ti1 =l shl %a1b, 4
	%a1b =l and %b0t, 3
	%tg126 =w csgel %ti1, 26
	%tg152 =w csgel %ti1, 52
	%tg162 =w csgel %ti1, 62
	%te126 =l extuw %tg126
	%te152 =l extuw %tg152
	%te162 =l extuw %tg162
	%tm126 =l mul %te126, 6
	%tm152 =l mul %te152, 75
	%ttv10 =l mul %ti1, 3
	%ttv11 =l sub %ttv10, 201
	%tm162 =l mul %te162, %ttv11
	%tc10 =l add %ti1, 65
	%tc11 =l add %tc10, %tm126
	%tc12 =l sub %tc11, %tm152
	%tc1 =l add %tc12, %tm162
	%to1 =l add %op, 1
	storeb %tc1, %to1
	%to2 =l add %op, 2
	storeb 61, %to2
	%to3 =l add %op, 3
	storeb 61, %to3
	%op4a =l add %op, 4
	%ra =l sub %op4a, %out
	ret %ra

@tail2
	%sp1t =l add %sp, 1
	%b1t =l loadub %sp1t
	%u10 =l and %b0t, 3
	%u11 =l shl %u10, 4
	%u12 =l shr %b1t, 4
	%ui1 =l or %u11, %u12
	%u20 =l and %b1t, 15
	%ui2 =l shl %u20, 2

	%ug126 =w csgel %ui1, 26
	%ug152 =w csgel %ui1, 52
	%ug162 =w csgel %ui1, 62
	%ue126 =l extuw %ug126
	%ue152 =l extuw %ug152
	%ue162 =l extuw %ug162
	%um126 =l mul %ue126, 6
	%um152 =l mul %ue152, 75
	%utv10 =l mul %ui1, 3
	%utv11 =l sub %utv10, 201
	%um162 =l mul %ue162, %utv11
	%uc10 =l add %ui1, 65
	%uc11 =l add %uc10, %um126
	%uc12 =l sub %uc11, %um152
	%uc1 =l add %uc12, %um162

	%ug226 =w csgel %ui2, 26
	%ug252 =w csgel %ui2, 52
	%ug262 =w csgel %ui2, 62
	%ue226 =l extuw %ug226
	%ue252 =l extuw %ug252
	%ue262 =l extuw %ug262
	%um226 =l mul %ue226, 6
	%um252 =l mul %ue252, 75
	%utv20 =l mul %ui2, 3
	%utv21 =l sub %utv20, 201
	%um262 =l mul %ue262, %utv21
	%uc20 =l add %ui2, 65
	%uc21 =l add %uc20, %um226
	%uc22 =l sub %uc21, %um252
	%uc2 =l add %uc22, %um262

	%uo1 =l add %op, 1
	storeb %uc1, %uo1
	%uo2 =l add %op, 2
	storeb %uc2, %uo2
	%uo3 =l add %op, 3
	storeb 61, %uo3
	%op4b =l add %op, 4
	%rb =l sub %op4b, %out
	ret %rb

@ret
	%r =l sub %op, %out
	ret %r
}
