export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%cnt0 =l and %y0, 1
	jmp @loop

@loop
	%i =l phi @start 1, @inc %i2
	%min =l phi @start %y0, @inc %min2
	%max =l phi @start %y0, @inc %max2
	%sum =l phi @start %y0, @inc %sum2
	%cnt =l phi @start %cnt0, @inc %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @done

@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @keepmin

@setmin
	jmp @minjoin

@keepmin
	jmp @minjoin

@minjoin
	%min2 =l phi @setmin %y, @keepmin %min
	%gt =w csltl %max, %y
	jnz %gt, @setmax, @keepmax

@setmax
	jmp @maxjoin

@keepmax
	jmp @maxjoin

@maxjoin
	%max2 =l phi @setmax %y, @keepmax %max
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%cnt2 =l add %cnt, %odd
	%i2 =l add %i, 1
	jmp @inc

@inc
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
