export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%sposp =l alloc8 8
	%dposp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %sposp
	storel 0, %dposp
	jmp @loop

@loop
	%spos =l loadl %sposp
	%more =w csltl %spos, %n
	jnz %more, @packet, @done

@done
	%dpos =l loadl %dposp
	ret %dpos

@packet
	%hptr =l add %src, %spos
	%h =w loadsb %hptr
	%spos1 =l add %spos, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @notnop

@nop
	storel %spos1, %sposp
	jmp @loop

@notnop
	%isneg =w csltw %h, 0
	jnz %isneg, @repeat, @literal

@literal
	%lenw =w add %h, 1
	%len =l extsw %lenw
	%srcend =l add %spos1, %len
	%srcok =w cslel %srcend, %n
	jnz %srcok, @lit_src_ok, @err_src

@lit_src_ok
	%dpos0 =l loadl %dposp
	%newd =l add %dpos0, %len
	%fit =w cslel %newd, %cap
	jnz %fit, @lit_full_setup, @lit_part_setup

@lit_full_setup
	storel 0, %ip
	jmp @lit_full_loop

@lit_full_loop
	%i =l loadl %ip
	%c =w csltl %i, %len
	jnz %c, @lit_full_body, @lit_full_done

@lit_full_body
	%so0 =l add %spos1, %i
	%sp0 =l add %src, %so0
	%ch =w loadub %sp0
	%do0 =l add %dpos0, %i
	%dp0 =l add %out, %do0
	storeb %ch, %dp0
	%i1 =l add %i, 1
	storel %i1, %ip
	jmp @lit_full_loop

@lit_full_done
	storel %srcend, %sposp
	storel %newd, %dposp
	jmp @loop

@lit_part_setup
	%part =l sub %cap, %dpos0
	storel 0, %ip
	jmp @lit_part_loop

@lit_part_loop
	%i2 =l loadl %ip
	%c2 =w csltl %i2, %part
	jnz %c2, @lit_part_body, @err_cap

@lit_part_body
	%so1 =l add %spos1, %i2
	%sp1 =l add %src, %so1
	%ch1 =w loadub %sp1
	%do1 =l add %dpos0, %i2
	%dp1 =l add %out, %do1
	storeb %ch1, %dp1
	%i3 =l add %i2, 1
	storel %i3, %ip
	jmp @lit_part_loop

@repeat
	%srcend2 =l add %spos1, 1
	%srcok2 =w cslel %srcend2, %n
	jnz %srcok2, @rep_src_ok, @err_src

@rep_src_ok
	%lenrw =w sub 1, %h
	%lenr =l extsw %lenrw
	%vptr =l add %src, %spos1
	%val =w loadub %vptr
	%dposr =l loadl %dposp
	%newdr =l add %dposr, %lenr
	%fitr =w cslel %newdr, %cap
	jnz %fitr, @rep_full_setup, @rep_part_setup

@rep_full_setup
	storel 0, %ip
	jmp @rep_full_loop

@rep_full_loop
	%ri =l loadl %ip
	%rc =w csltl %ri, %lenr
	jnz %rc, @rep_full_body, @rep_full_done

@rep_full_body
	%rdo =l add %dposr, %ri
	%rdp =l add %out, %rdo
	storeb %val, %rdp
	%ri1 =l add %ri, 1
	storel %ri1, %ip
	jmp @rep_full_loop

@rep_full_done
	storel %srcend2, %sposp
	storel %newdr, %dposp
	jmp @loop

@rep_part_setup
	%rpart =l sub %cap, %dposr
	storel 0, %ip
	jmp @rep_part_loop

@rep_part_loop
	%rpi =l loadl %ip
	%rpc =w csltl %rpi, %rpart
	jnz %rpc, @rep_part_body, @err_cap

@rep_part_body
	%rpdo =l add %dposr, %rpi
	%rpdp =l add %out, %rpdo
	storeb %val, %rpdp
	%rpi1 =l add %rpi, 1
	storel %rpi1, %ip
	jmp @rep_part_loop

@err_src
	ret -2

@err_cap
	ret -1
}
