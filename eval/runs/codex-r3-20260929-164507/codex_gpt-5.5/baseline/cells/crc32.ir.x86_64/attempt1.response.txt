export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @byte_done %i_next
	%crc =w phi @start 0xffffffff, @byte_done %c
	%more =w csltl %i, %n
	jnz %more, @byte, @final
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b
	jmp @bitloop
@bitloop
	%j =w phi @byte 0, @merge %j_next
	%c =w phi @byte %cx, @merge %c_next
	%bdone =w ceqw %j, 8
	jnz %bdone, @byte_done, @testbit
@testbit
	%bit =w and %c, 1
	jnz %bit, @odd, @even
@odd
	%sh1 =w shr %c, 1
	%co =w xor %sh1, 0xedb88320
	jmp @merge
@even
	%ce =w shr %c, 1
	jmp @merge
@merge
	%c_next =w phi @odd %co, @even %ce
	%j_next =w add %j, 1
	jmp @bitloop
@byte_done
	%i_next =l add %i, 1
	jmp @outer
@final
	%rw =w xor %crc, 0xffffffff
	%r =l extuw %rw
	ret %r
}
