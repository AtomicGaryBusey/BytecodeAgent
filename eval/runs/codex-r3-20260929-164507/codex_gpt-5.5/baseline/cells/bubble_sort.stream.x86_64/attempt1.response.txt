bubble_sort:
cmp rsi, 1
jle .Ldone
mov r8, rsi
dec r8
.Louter:
xor rcx, rcx
.Linner:
cmp rcx, r8
jge .Lnext_pass
mov rax, qword ptr [rdi + rcx*8]
mov rdx, qword ptr [rdi + rcx*8 + 8]
cmp rax, rdx
jle .Lno_swap
mov qword ptr [rdi + rcx*8], rdx
mov qword ptr [rdi + rcx*8 + 8], rax
.Lno_swap:
inc rcx
jmp .Linner
.Lnext_pass:
dec r8
jg .Louter
.Ldone:
xor eax, eax
ret
