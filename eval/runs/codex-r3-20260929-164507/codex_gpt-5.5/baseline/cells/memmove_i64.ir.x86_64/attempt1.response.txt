export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%dir = culel %dst, %src
	jnz %dir, @fwd, @bwd
@fwd
	%i =l copy 0
	jmp @fwd_cond
@fwd_cond
	%fc = csltl %i, %n
	jnz %fc, @fwd_body, @done
@fwd_body
	%fo =l mul %i, 8
	%fs =l add %src, %fo
	%fd =l add %dst, %fo
	%v =l loadl %fs
	storel %v, %fd
	%in =l add %i, 1
	%i =l copy %in
	jmp @fwd_cond
@bwd
	%j =l copy %n
	jmp @bwd_cond
@bwd_cond
	%bc = csgtl %j, 0
	jnz %bc, @bwd_body, @done
@bwd_body
	%jn =l sub %j, 1
	%bo =l mul %jn, 8
	%bs =l add %src, %bo
	%bd =l add %dst, %bo
	%w =l loadl %bs
	storel %w, %bd
	%j =l copy %jn
	jmp @bwd_cond
@done
	ret 0
}
