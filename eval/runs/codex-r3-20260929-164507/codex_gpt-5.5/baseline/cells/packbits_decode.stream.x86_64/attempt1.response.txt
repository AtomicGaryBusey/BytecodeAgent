packbits_decode:
lea r8, [rdi + rsi]
mov r11, rdx
lea r9, [rdx + rcx]
mov r10, rdi
.Lloop:
cmp r10, r8
jae .Ldone
movsx eax, byte ptr [r10]
inc r10
cmp eax, 0
jl .Lnegative
mov ecx, eax
inc ecx
lea rsi, [r10 + rcx]
cmp rsi, r8
ja .Lsrc_error
mov rax, r9
sub rax, rdx
cmp rcx, rax
ja .Lliteral_trunc
mov rsi, r10
mov rdi, rdx
rep movsb
mov rdx, rdi
mov r10, rsi
jmp .Lloop
.Lliteral_trunc:
mov rcx, rax
mov rsi, r10
mov rdi, rdx
rep movsb
mov rax, -1
ret
.Lnegative:
cmp eax, -128
je .Lloop
cmp r10, r8
jae .Lsrc_error
mov esi, 1
sub esi, eax
movzx eax, byte ptr [r10]
inc r10
mov rcx, r9
sub rcx, rdx
cmp rsi, rcx
ja .Lrepeat_trunc
mov rcx, rsi
mov rdi, rdx
rep stosb
mov rdx, rdi
jmp .Lloop
.Lrepeat_trunc:
mov rdi, rdx
rep stosb
mov rax, -1
ret
.Lsrc_error:
mov rax, -2
ret
.Ldone:
mov rax, rdx
sub rax, r11
ret
