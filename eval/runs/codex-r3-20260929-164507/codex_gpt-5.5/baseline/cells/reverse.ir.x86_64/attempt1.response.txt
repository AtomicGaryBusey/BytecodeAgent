export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop

@loop
	%i =l phi @start 0, @next %i1
	%j =l phi @start %j0, @next %j1
	%c =w csltl %i, %j
	jnz %c, @body, @done

@body
	%io =l shl %i, 3
	%jo =l shl %j, 3
	%pi =l add %a, %io
	%pj =l add %a, %jo
	%xi =l loadl %pi
	%xj =l loadl %pj
	storel %xj, %pi
	storel %xi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @next

@next
	jmp @loop

@done
	ret 0
}
