lcs_len:
push rbx
xor r10d, r10d
.Lzero:
mov qword ptr [r8 + r10*8], 0
cmp r10, rcx
je .Lzero_done
inc r10
jmp .Lzero
.Lzero_done:
lea rsi, [rdi + rsi]
.Louter:
cmp rdi, rsi
jae .Ldone
movzx ebx, byte ptr [rdi]
xor r11d, r11d
mov r10, 1
test rcx, rcx
jle .Linner_done
.Linner:
mov rax, qword ptr [r8 + r10*8]
cmp bl, byte ptr [rdx + r10 - 1]
je .Lmatch
mov r9, rax
cmp rax, qword ptr [r8 + r10*8 - 8]
cmovl rax, qword ptr [r8 + r10*8 - 8]
mov qword ptr [r8 + r10*8], rax
mov r11, r9
jmp .Lnext
.Lmatch:
mov r9, rax
lea rax, [r11 + 1]
mov qword ptr [r8 + r10*8], rax
mov r11, r9
.Lnext:
inc r10
cmp r10, rcx
jle .Linner
.Linner_done:
inc rdi
jmp .Louter
.Ldone:
mov rax, qword ptr [r8 + rcx*8]
pop rbx
ret
