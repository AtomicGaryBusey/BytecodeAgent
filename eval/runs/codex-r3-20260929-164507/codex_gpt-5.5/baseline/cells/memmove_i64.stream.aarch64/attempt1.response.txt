memmove_i64:

cmp x0, x1
b.ls .Lforward
sub x2, x2, #1
.Lbackward:
ldr x3, [x1, x2, lsl #3]
str x3, [x0, x2, lsl #3]
cbz x2, .Ldone
sub x2, x2, #1
b .Lbackward
.Lforward:
mov x3, #0
.Lforward_loop:
ldr x4, [x1, x3, lsl #3]
str x4, [x0, x3, lsl #3]
add x3, x3, #1
cmp x3, x2
b.lo .Lforward_loop
.Ldone:

ret
