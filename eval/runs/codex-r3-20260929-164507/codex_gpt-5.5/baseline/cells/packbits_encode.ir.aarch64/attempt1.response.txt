export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%pw =l alloc8 8
	%pr =l alloc8 8
	%pk =l alloc8 8
	%pj =l alloc8 8
	%pc =l alloc8 8
	storel 0, %pi
	storel 0, %pw
	jmp @main

@main
	%m_i =l loadl %pi
	%m_cmp =w csltl %m_i, %n
	jnz %m_cmp, @run_init, @done

@run_init
	%ri_addr =l add %src, %m_i
	%ri_b =w loadub %ri_addr
	storel 1, %pr
	%ri_k =l add %m_i, 1
	storel %ri_k, %pk
	jmp @run_loop

@run_loop
	%rl_k =l loadl %pk
	%rl_c1 =w csltl %rl_k, %n
	jnz %rl_c1, @run_check_r, @run_end

@run_check_r
	%rl_r =l loadl %pr
	%rl_c2 =w csltl %rl_r, 128
	jnz %rl_c2, @run_check_b, @run_end

@run_check_b
	%rb_addr =l add %src, %rl_k
	%rb_b =w loadub %rb_addr
	%rb_eq =w ceqw %rb_b, %ri_b
	jnz %rb_eq, @run_inc, @run_end

@run_inc
	%rinc_r0 =l loadl %pr
	%rinc_r1 =l add %rinc_r0, 1
	storel %rinc_r1, %pr
	%rinc_k0 =l loadl %pk
	%rinc_k1 =l add %rinc_k0, 1
	storel %rinc_k1, %pk
	jmp @run_loop

@run_end
	%re_r =l loadl %pr
	%re_isrep =w csltl 2, %re_r
	jnz %re_isrep, @repeat, @literal_init

@repeat
	%rep_w =l loadl %pw
	%rep_rem =l sub %cap, %rep_w
	%rep_fit =w csltl 1, %rep_rem
	jnz %rep_fit, @repeat_write, @fail

@repeat_write
	%rw_p0 =l add %out, %rep_w
	%rw_hdr_l =l sub 257, %re_r
	%rw_hdr =w copy %rw_hdr_l
	storeb %rw_hdr, %rw_p0
	%rw_p1 =l add %rw_p0, 1
	storeb %ri_b, %rw_p1
	%rw_w1 =l add %rep_w, 2
	storel %rw_w1, %pw
	%rw_i1 =l add %m_i, %re_r
	storel %rw_i1, %pi
	jmp @main

@literal_init
	storel %m_i, %pj
	jmp @lit_loop

@lit_loop
	%ll_j =l loadl %pj
	%ll_c1 =w csltl %ll_j, %n
	jnz %ll_c1, @lit_check_len, @lit_end

@lit_check_len
	%ll_len =l sub %ll_j, %m_i
	%ll_c2 =w csltl %ll_len, 128
	jnz %ll_c2, @lit_run_init, @lit_end

@lit_run_init
	%lri_addr =l add %src, %ll_j
	%lri_b =w loadub %lri_addr
	storel 1, %pr
	%lri_k =l add %ll_j, 1
	storel %lri_k, %pk
	jmp @lit_run_loop

@lit_run_loop
	%lrl_k =l loadl %pk
	%lrl_c1 =w csltl %lrl_k, %n
	jnz %lrl_c1, @lit_run_check_r, @lit_run_end

@lit_run_check_r
	%lrl_r =l loadl %pr
	%lrl_c2 =w csltl %lrl_r, 128
	jnz %lrl_c2, @lit_run_check_b, @lit_run_end

@lit_run_check_b
	%lrb_addr =l add %src, %lrl_k
	%lrb_b =w loadub %lrb_addr
	%lrb_eq =w ceqw %lrb_b, %lri_b
	jnz %lrb_eq, @lit_run_inc, @lit_run_end

@lit_run_inc
	%lrinc_r0 =l loadl %pr
	%lrinc_r1 =l add %lrinc_r0, 1
	storel %lrinc_r1, %pr
	%lrinc_k0 =l loadl %pk
	%lrinc_k1 =l add %lrinc_k0, 1
	storel %lrinc_k1, %pk
	jmp @lit_run_loop

@lit_run_end
	%lre_r =l loadl %pr
	%lre_stop =w csltl 2, %lre_r
	jnz %lre_stop, @lit_end, @lit_advance

@lit_advance
	%la_j0 =l loadl %pj
	%la_j1 =l add %la_j0, 1
	storel %la_j1, %pj
	jmp @lit_loop

@lit_end
	%le_j =l loadl %pj
	%le_len =l sub %le_j, %m_i
	%le_w =l loadl %pw
	%le_rem =l sub %cap, %le_w
	%le_fit =w csltl %le_len, %le_rem
	jnz %le_fit, @lit_write_header, @fail

@lit_write_header
	%lwh_p =l add %out, %le_w
	%lwh_hdr_l =l sub %le_len, 1
	%lwh_hdr =w copy %lwh_hdr_l
	storeb %lwh_hdr, %lwh_p
	storel 0, %pc
	jmp @lit_copy_loop

@lit_copy_loop
	%lcl_c =l loadl %pc
	%lcl_more =w csltl %lcl_c, %le_len
	jnz %lcl_more, @lit_copy_one, @lit_copy_done

@lit_copy_one
	%lco_sp0 =l add %src, %m_i
	%lco_sp =l add %lco_sp0, %lcl_c
	%lco_b =w loadub %lco_sp
	%lco_dp0 =l add %out, %le_w
	%lco_dp1 =l add %lco_dp0, 1
	%lco_dp =l add %lco_dp1, %lcl_c
	storeb %lco_b, %lco_dp
	%lco_c1 =l add %lcl_c, 1
	storel %lco_c1, %pc
	jmp @lit_copy_loop

@lit_copy_done
	%lcd_add =l add %le_len, 1
	%lcd_w1 =l add %le_w, %lcd_add
	storel %lcd_w1, %pw
	storel %le_j, %pi
	jmp @main

@done
	%d_w =l loadl %pw
	ret %d_w

@fail
	ret -1
}
