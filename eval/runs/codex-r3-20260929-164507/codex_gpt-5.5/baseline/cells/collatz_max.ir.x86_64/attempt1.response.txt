export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
	jump @outer_cond

@outer_cond
	%ok =w cslel %k, %n
	jnz %ok, @outer_body, @done

@outer_body
	%x =l copy %k
	%steps =l copy 0
	jump @inner_cond

@inner_cond
	%notone =w cnel %x, 1
	jnz %notone, @inner_body, @inner_done

@inner_body
	%bit =l and %x, 1
	%even =w ceql %bit, 0
	jnz %even, @even, @odd

@even
	%xe =l div %x, 2
	%stepse =l add %steps, 1
	%x =l copy %xe
	%steps =l copy %stepse
	jump @inner_cond

@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	%stepso =l add %steps, 1
	%x =l copy %xo
	%steps =l copy %stepso
	jump @inner_cond

@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @setmax, @nextk

@setmax
	%max =l copy %steps
	jump @nextk

@nextk
	%kn =l add %k, 1
	%k =l copy %kn
	jump @outer_cond

@done
	ret %max
}
