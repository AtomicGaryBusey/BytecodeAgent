export function l $collatz_max(l %n) {
@start
	jmp @outer_cond

@outer_cond
	%k =l phi @start 1, @nextk %kn
	%max =l phi @start 0, @nextk %maxnext
	%ok =w cslel %k, %n
	jnz %ok, @outer_body, @done

@outer_body
	jmp @inner_cond

@inner_cond
	%x =l phi @outer_body %k, @even %xe, @odd %xo
	%steps =l phi @outer_body 0, @even %stepse, @odd %stepso
	%notone =w cnel %x, 1
	jnz %notone, @inner_body, @inner_done

@inner_body
	%bit =l and %x, 1
	%iseven =w ceql %bit, 0
	jnz %iseven, @even, @odd

@even
	%xe =l div %x, 2
	%stepse =l add %steps, 1
	jmp @inner_cond

@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	%stepso =l add %steps, 1
	jmp @inner_cond

@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @setmax, @keepmax

@setmax
	jmp @nextk

@keepmax
	jmp @nextk

@nextk
	%maxnext =l phi @setmax %steps, @keepmax %max
	%kn =l add %k, 1
	jmp @outer_cond

@done
	ret %max
}
