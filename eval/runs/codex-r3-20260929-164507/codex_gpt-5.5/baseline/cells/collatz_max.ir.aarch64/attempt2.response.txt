export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%k0 =l copy 1
	jmp @outer
@outer
	%max =l phi @start %max0, @after %max2
	%k =l phi @start %k0, @after %k2
	%cont =w cslel %k, %n
	jnz %cont, @init, @ret
@init
	%x0 =l copy %k
	%s0 =l copy 0
	jmp @inner
@inner
	%x =l phi @init %x0, @inc %xnext
	%s =l phi @init %s0, @inc %snext
	%done =w ceql %x, 1
	jnz %done, @done_inner, @parity
@parity
	%bit =l and %x, 1
	%iseven =w ceql %bit, 0
	jnz %iseven, @even, @odd
@even
	%xe =l sar %x, 1
	jmp @inc
@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	jmp @inc
@inc
	%xnext =l phi @even %xe, @odd %xo
	%snext =l add %s, 1
	jmp @inner
@done_inner
	%gt =w csgtl %s, %max
	jnz %gt, @updatemax, @keepmax
@updatemax
	%maxu =l copy %s
	jmp @after
@keepmax
	%maxk =l copy %max
	jmp @after
@after
	%max2 =l phi @updatemax %maxu, @keepmax %maxk
	%k2 =l add %k, 1
	jmp @outer
@ret
	ret %max
}
