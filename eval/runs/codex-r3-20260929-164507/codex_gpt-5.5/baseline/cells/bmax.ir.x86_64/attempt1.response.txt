export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @set %iset, @next %inext
	%max =l phi @start %max0, @set %val, @next %max
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%cmp =w csltl %max, %val
	jnz %cmp, @set, @next
@set
	%iset =l add %i, 1
	jmp @loop
@next
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
