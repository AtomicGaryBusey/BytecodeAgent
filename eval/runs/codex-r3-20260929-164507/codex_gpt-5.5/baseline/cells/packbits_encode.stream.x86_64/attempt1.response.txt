packbits_encode:
push rbx
lea rsi, [rdi + rsi]
lea rcx, [rdx + rcx]
mov r8, rdx
.Lmain:
cmp rdi, rsi
jae .Lsuccess
movzx r9d, byte ptr [rdi]
mov r10, 1
.Lrun:
cmp r10, 128
jae .Lrun_done
lea r11, [rdi + r10]
cmp r11, rsi
jae .Lrun_done
cmp byte ptr [r11], r9b
jne .Lrun_done
inc r10
jmp .Lrun
.Lrun_done:
cmp r10, 3
jb .Lliteral
lea r11, [rdx + 2]
cmp r11, rcx
ja .Lfail
mov eax, 1
sub eax, r10d
mov byte ptr [rdx], al
mov byte ptr [rdx + 1], r9b
add rdi, r10
mov rdx, r11
jmp .Lmain
.Lliteral:
mov r9, rdi
mov r11, rdi
.Llit_scan:
cmp r11, rsi
jae .Llit_done
mov rax, r11
sub rax, r9
cmp rax, 128
jae .Llit_done
movzx eax, byte ptr [r11]
mov r10, 1
.Llit_run:
cmp r10, 128
jae .Llit_run_done
lea rbx, [r11 + r10]
cmp rbx, rsi
jae .Llit_run_done
cmp byte ptr [rbx], al
jne .Llit_run_done
inc r10
jmp .Llit_run
.Llit_run_done:
cmp r10, 3
jae .Llit_done
inc r11
jmp .Llit_scan
.Llit_done:
mov r10, r11
sub r10, r9
lea rbx, [r10 + 1]
lea rax, [rdx + rbx]
cmp rax, rcx
ja .Lfail
mov rbx, rax
mov eax, r10d
dec eax
mov byte ptr [rdx], al
mov rax, r9
lea r10, [rdx + 1]
.Lcopy:
cmp rax, r11
jae .Lcopy_done
mov r9b, byte ptr [rax]
mov byte ptr [r10], r9b
inc rax
inc r10
jmp .Lcopy
.Lcopy_done:
mov rdi, r11
mov rdx, rbx
jmp .Lmain
.Lsuccess:
mov rax, rdx
sub rax, r8
pop rbx
ret
.Lfail:
mov rax, -1
pop rbx
ret
