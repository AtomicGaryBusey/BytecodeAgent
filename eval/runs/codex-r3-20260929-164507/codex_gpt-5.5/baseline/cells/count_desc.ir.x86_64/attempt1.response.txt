export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret0, @init
@ret0
	ret 0
@init
	%limit =l sub %n, 1
	%i0 =l copy 0
	%cnt0 =l copy 0
	jmp @loop
@loop
	%i =l phi @init %i0, @inc %ii, @noinc %in
	%cnt =l phi @init %cnt0, @inc %cnti, @noinc %cnt
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%gt =w csgtl %x, %y
	jnz %gt, @inc, @noinc
@inc
	%cnti =l add %cnt, 1
	%ii =l add %i, 1
	jmp @loop
@noinc
	%in =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
