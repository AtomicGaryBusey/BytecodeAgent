mov qword ptr [rsi + 8], r11

.Lnext:
inc r8
jmp .Lpart_loop

mov rax, rbp
shl rax, 4
lea rdx, [rbx + rax]
mov rax, r12
dec rax
shl rax, 4
lea rsi, [rbx + rax]
