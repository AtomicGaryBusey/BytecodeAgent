export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %i1
	%m =l phi @start %m0, @next %m1
	%c =w csltl %i, %n
	jnz %c, @body, @ret
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csltl %m, %v
	jnz %gt, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%m1 =l phi @upd %v, @keep %m
	%i1 =l add %i, 1
	jmp @loop
@ret
	ret %m
}
