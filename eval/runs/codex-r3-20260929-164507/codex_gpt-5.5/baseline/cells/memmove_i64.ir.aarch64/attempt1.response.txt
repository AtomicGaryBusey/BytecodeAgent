export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%forward =w culel %dst, %src
	jnz %forward, @f_loop, @b_loop

@f_loop
	%fi =l phi @start 0, @f_body %fi_next
	%fc =w csltl %fi, %n
	jnz %fc, @f_body, @ret

@f_body
	%foff =l shl %fi, 3
	%fsp =l add %src, %foff
	%fv =l loadl %fsp
	%fdp =l add %dst, %foff
	storel %fv, %fdp
	%fi_next =l add %fi, 1
	jmp @f_loop

@b_loop
	%bi =l phi @start %n, @b_body %bi_next
	%bc =w csgtl %bi, 0
	jnz %bc, @b_body, @ret

@b_body
	%bi_next =l sub %bi, 1
	%boff =l shl %bi_next, 3
	%bsp =l add %src, %boff
	%bv =l loadl %bsp
	%bdp =l add %dst, %boff
	storel %bv, %bdp
	jmp @b_loop

@ret
	ret 0
}
