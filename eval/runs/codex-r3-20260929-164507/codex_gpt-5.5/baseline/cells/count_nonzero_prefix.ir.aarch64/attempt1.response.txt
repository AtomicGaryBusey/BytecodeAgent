export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
	jmp @loop
@loop
	%more =w csltl %i, %n
	jnz %more, @load, @done
@load
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%zero =w ceql %v, 0
	jnz %zero, @done, @next
@next
	%i2 =l add %i, 1
	%i =l copy %i2
	jmp @loop
@done
	ret %i
}
