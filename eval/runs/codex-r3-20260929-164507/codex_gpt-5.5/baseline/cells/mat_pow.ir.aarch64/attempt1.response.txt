export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%pk =l alloc8 8
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%psum =l alloc8 8
	%m =l mul %n, %n
	%b =l mul %m, 8
	%prod =l add %tmp, %b
	storel %k, %pk
	storel 0, %pi
	jmp @copy_a

@copy_a
	%ca_i =l loadl %pi
	%ca_c =w csltl %ca_i, %m
	jnz %ca_c, @copy_a_body, @out_zero_init

@copy_a_body
	%ca_o =l mul %ca_i, 8
	%ca_ap =l add %a, %ca_o
	%ca_tp =l add %tmp, %ca_o
	%ca_v =l loadl %ca_ap
	storel %ca_v, %ca_tp
	%ca_i1 =l add %ca_i, 1
	storel %ca_i1, %pi
	jmp @copy_a

@out_zero_init
	storel 0, %pi
	jmp @out_zero

@out_zero
	%oz_i =l loadl %pi
	%oz_c =w csltl %oz_i, %m
	jnz %oz_c, @out_zero_body, @diag_init

@out_zero_body
	%oz_o =l mul %oz_i, 8
	%oz_p =l add %out, %oz_o
	storel 0, %oz_p
	%oz_i1 =l add %oz_i, 1
	storel %oz_i1, %pi
	jmp @out_zero

@diag_init
	storel 0, %pi
	jmp @diag_loop

@diag_loop
	%d_i =l loadl %pi
	%d_c =w csltl %d_i, %n
	jnz %d_c, @diag_body, @pow_loop

@diag_body
	%d_in =l mul %d_i, %n
	%d_idx =l add %d_in, %d_i
	%d_o =l mul %d_idx, 8
	%d_p =l add %out, %d_o
	storel 1, %d_p
	%d_i1 =l add %d_i, 1
	storel %d_i1, %pi
	jmp @diag_loop

@pow_loop
	%pl_k =l loadl %pk
	%pl_c =w ceql %pl_k, 0
	jnz %pl_c, @final_zero_init, @pow_body

@pow_body
	%pb_bit =l and %pl_k, 1
	%pb_c =w ceql %pb_bit, 0
	jnz %pb_c, @skip_mul, @mul_init

@mul_init
	storel 0, %pi
	jmp @mul_i

@mul_i
	%mi_i =l loadl %pi
	%mi_c =w csltl %mi_i, %n
	jnz %mi_c, @mul_j_init, @mul_copy_init

@mul_j_init
	storel 0, %pj
	jmp @mul_j

@mul_j
	%mj_j =l loadl %pj
	%mj_c =w csltl %mj_j, %n
	jnz %mj_c, @mul_t_init, @mul_i_next

@mul_t_init
	storel 0, %psum
	storel 0, %pt
	jmp @mul_t

@mul_t
	%mt_t =l loadl %pt
	%mt_c =w csltl %mt_t, %n
	jnz %mt_c, @mul_t_body, @mul_store

@mul_t_body
	%mt_i =l loadl %pi
	%mt_j =l loadl %pj
	%mt_in =l mul %mt_i, %n
	%mt_xi =l add %mt_in, %mt_t
	%mt_xo =l mul %mt_xi, 8
	%mt_xp =l add %out, %mt_xo
	%mt_tn =l mul %mt_t, %n
	%mt_yi =l add %mt_tn, %mt_j
	%mt_yo =l mul %mt_yi, 8
	%mt_yp =l add %tmp, %mt_yo
	%mt_x =l loadl %mt_xp
	%mt_y =l loadl %mt_yp
	%mt_p =l mul %mt_x, %mt_y
	%mt_s =l loadl %psum
	%mt_s1 =l add %mt_s, %mt_p
	storel %mt_s1, %psum
	%mt_t1 =l add %mt_t, 1
	storel %mt_t1, %pt
	jmp @mul_t

@mul_store
	%ms_i =l loadl %pi
	%ms_j =l loadl %pj
	%ms_in =l mul %ms_i, %n
	%ms_idx =l add %ms_in, %ms_j
	%ms_o =l mul %ms_idx, 8
	%ms_p =l add %prod, %ms_o
	%ms_s =l loadl %psum
	storel %ms_s, %ms_p
	%ms_j1 =l add %ms_j, 1
	storel %ms_j1, %pj
	jmp @mul_j

@mul_i_next
	%min_i =l loadl %pi
	%min_i1 =l add %min_i, 1
	storel %min_i1, %pi
	jmp @mul_i

@mul_copy_init
	storel 0, %pi
	jmp @mul_copy

@mul_copy
	%mc_i =l loadl %pi
	%mc_c =w csltl %mc_i, %m
	jnz %mc_c, @mul_copy_body, @skip_mul

@mul_copy_body
	%mc_o =l mul %mc_i, 8
	%mc_sp =l add %prod, %mc_o
	%mc_dp =l add %out, %mc_o
	%mc_v =l loadl %mc_sp
	storel %mc_v, %mc_dp
	%mc_i1 =l add %mc_i, 1
	storel %mc_i1, %pi
	jmp @mul_copy

@skip_mul
	storel 0, %pi
	jmp @sq_i

@sq_i
	%si_i =l loadl %pi
	%si_c =w csltl %si_i, %n
	jnz %si_c, @sq_j_init, @sq_copy_init

@sq_j_init
	storel 0, %pj
	jmp @sq_j

@sq_j
	%sj_j =l loadl %pj
	%sj_c =w csltl %sj_j, %n
	jnz %sj_c, @sq_t_init, @sq_i_next

@sq_t_init
	storel 0, %psum
	storel 0, %pt
	jmp @sq_t

@sq_t
	%st_t =l loadl %pt
	%st_c =w csltl %st_t, %n
	jnz %st_c, @sq_t_body, @sq_store

@sq_t_body
	%st_i =l loadl %pi
	%st_j =l loadl %pj
	%st_in =l mul %st_i, %n
	%st_xi =l add %st_in, %st_t
	%st_xo =l mul %st_xi, 8
	%st_xp =l add %tmp, %st_xo
	%st_tn =l mul %st_t, %n
	%st_yi =l add %st_tn, %st_j
	%st_yo =l mul %st_yi, 8
	%st_yp =l add %tmp, %st_yo
	%st_x =l loadl %st_xp
	%st_y =l loadl %st_yp
	%st_p =l mul %st_x, %st_y
	%st_s =l loadl %psum
	%st_s1 =l add %st_s, %st_p
	storel %st_s1, %psum
	%st_t1 =l add %st_t, 1
	storel %st_t1, %pt
	jmp @sq_t

@sq_store
	%ss_i =l loadl %pi
	%ss_j =l loadl %pj
	%ss_in =l mul %ss_i, %n
	%ss_idx =l add %ss_in, %ss_j
	%ss_o =l mul %ss_idx, 8
	%ss_p =l add %prod, %ss_o
	%ss_s =l loadl %psum
	storel %ss_s, %ss_p
	%ss_j1 =l add %ss_j, 1
	storel %ss_j1, %pj
	jmp @sq_j

@sq_i_next
	%sin_i =l loadl %pi
	%sin_i1 =l add %sin_i, 1
	storel %sin_i1, %pi
	jmp @sq_i

@sq_copy_init
	storel 0, %pi
	jmp @sq_copy

@sq_copy
	%sc_i =l loadl %pi
	%sc_c =w csltl %sc_i, %m
	jnz %sc_c, @sq_copy_body, @shift_k

@sq_copy_body
	%sc_o =l mul %sc_i, 8
	%sc_sp =l add %prod, %sc_o
	%sc_dp =l add %tmp, %sc_o
	%sc_v =l loadl %sc_sp
	storel %sc_v, %sc_dp
	%sc_i1 =l add %sc_i, 1
	storel %sc_i1, %pi
	jmp @sq_copy

@shift_k
	%sk_k =l loadl %pk
	%sk_k1 =l shr %sk_k, 1
	storel %sk_k1, %pk
	jmp @pow_loop

@final_zero_init
	storel 0, %pi
	%twom =l add %m, %m
	jmp @final_zero

@final_zero
	%fz_i =l loadl %pi
	%fz_c =w csltl %fz_i, %twom
	jnz %fz_c, @final_zero_body, @ret

@final_zero_body
	%fz_o =l mul %fz_i, 8
	%fz_p =l add %tmp, %fz_o
	storel 0, %fz_p
	%fz_i1 =l add %fz_i, 1
	storel %fz_i1, %pi
	jmp @final_zero

@ret
	ret 0
}
