export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init

@init
	%k =l phi @start 0, @init_body %k1
	%kc =w cslel %k, %n
	jnz %kc, @init_body, @outer

@init_body
	%koff =l mul %k, 8
	%kptr =l add %row, %koff
	storel 0, %kptr
	%k1 =l add %k, 1
	jmp @init

@outer
	%i =l phi @init 0, @inner_done %i1
	%oc =w csltl %i, %m
	jnz %oc, @outer_body, @done

@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outer_body 1, @inner_next %j1
	%diag =l phi @outer_body 0, @inner_next %up
	%ic =w cslel %j, %n
	jnz %ic, @inner_body, @inner_done

@inner_body
	%joff =l mul %j, 8
	%rj =l add %row, %joff
	%up =l loadl %rj
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch

@match
	%mval =l add %diag, 1
	storel %mval, %rj
	jmp @inner_next

@nomatch
	%loff =l mul %jm1, 8
	%lp =l add %row, %loff
	%left =l loadl %lp
	%lg =w csltl %up, %left
	jnz %lg, @use_left, @use_up

@use_left
	storel %left, %rj
	jmp @inner_next

@use_up
	storel %up, %rj
	jmp @inner_next

@inner_next
	%j1 =l add %j, 1
	jmp @inner

@inner_done
	%i1 =l add %i, 1
	jmp @outer

@done
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%res =l loadl %nptr
	ret %res
}
