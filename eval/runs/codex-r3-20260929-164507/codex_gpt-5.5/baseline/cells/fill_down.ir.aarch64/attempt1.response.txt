export function l $fill_down(l %a, l %n, l %start) {
@entry
	%ip =l alloc8 8
	%z =l copy 0
	storel %z, %ip
	jmp @loop
@loop
	%i =l loadl %ip
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l sub %start, %i
	storel %v, %p
	%ni =l add %i, 1
	storel %ni, %ip
	jmp @loop
@done
	ret %n
}
