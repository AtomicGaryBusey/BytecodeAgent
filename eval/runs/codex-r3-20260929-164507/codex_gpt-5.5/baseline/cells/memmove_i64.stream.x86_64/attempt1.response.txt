memmove_i64:
test rdx, rdx
je .Ldone
cmp rdi, rsi
ja .Lbackward
xor rcx, rcx
.Lforward_loop:
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
inc rcx
cmp rcx, rdx
jb .Lforward_loop
.Ldone:
xor eax, eax
ret
.Lbackward:
mov rcx, rdx
.Lbackward_loop:
dec rcx
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
test rcx, rcx
jne .Lbackward_loop
xor eax, eax
ret
