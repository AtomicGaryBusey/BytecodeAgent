export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%inc =l extuw %gt
	%cnt1 =l add %cnt, %inc
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
