export function l $bubble_sort(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	%small =w csltl %n, 2
	jnz %small, @ret, @init
@init
	storel 0, %ip
	jmp @outer_cond
@outer_cond
	%i =l loadl %ip
	%nm1 =l sub %n, 1
	%oc =w csltl %i, %nm1
	jnz %oc, @outer_body, @ret
@outer_body
	storel 0, %jp
	%limit =l sub %nm1, %i
	jmp @inner_cond
@inner_cond
	%j =l loadl %jp
	%ic =w csltl %j, %limit
	jnz %ic, @inner_body, @outer_inc
@inner_body
	%off =l shl %j, 3
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%need =w csltl %v1, %v0
	jnz %need, @swap, @inner_inc
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_inc
@inner_inc
	%j2 =l add %j, 1
	storel %j2, %jp
	jmp @inner_cond
@outer_inc
	%i2 =l add %i, 1
	storel %i2, %ip
	jmp @outer_cond
@ret
	ret 0
}
