export function l $isqrt(l %n) {
@start
	jmp @init

@init
	%bit0 =l phi @start 4611686018427387904, @init_shift %bit_s
	%too_big =w cugtl %bit0, %n
	jnz %too_big, @init_shift, @loop

@init_shift
	%bit_s =l shr %bit0, 2
	jmp @init

@loop
	%ncur =l phi @init %n, @then %n_then, @else %n_else
	%res =l phi @init 0, @then %res_then, @else %res_else
	%bit =l phi @init %bit0, @then %bit_then, @else %bit_else
	%done =w ceql %bit, 0
	jnz %done, @return, @check

@check
	%sum =l add %res, %bit
	%ge =w cugel %ncur, %sum
	jnz %ge, @then, @else

@then
	%n_then =l sub %ncur, %sum
	%res_h1 =l shr %res, 1
	%res_then =l add %res_h1, %bit
	%bit_then =l shr %bit, 2
	jmp @loop

@else
	%n_else =l add %ncur, 0
	%res_else =l shr %res, 1
	%bit_else =l shr %bit, 2
	jmp @loop

@return
	ret %res
}
