export function l $isqrt(l %n) {
@start
	%res0 =l copy 0
	%bit0 =l copy 4611686018427387904
	jmp @shrink

@shrink
	%c0 =w cugtl %bit0, %n
	jnz %c0, @shrink_body, @loop

@shrink_body
	%bit1 =l shr %bit0, 2
	%bit0 =l copy %bit1
	jmp @shrink

@loop
	%c1 =w cnel %bit0, 0
	jnz %c1, @loop_body, @done

@loop_body
	%sum =l add %res0, %bit0
	%c2 =w cugel %n, %sum
	jnz %c2, @take, @skip

@take
	%n1 =l sub %n, %sum
	%half1 =l shr %res0, 1
	%res1 =l add %half1, %bit0
	%bit2 =l shr %bit0, 2
	%n =l copy %n1
	%res0 =l copy %res1
	%bit0 =l copy %bit2
	jmp @loop

@skip
	%res2 =l shr %res0, 1
	%bit3 =l shr %bit0, 2
	%res0 =l copy %res2
	%bit0 =l copy %bit3
	jmp @loop

@done
	ret %res0
}
