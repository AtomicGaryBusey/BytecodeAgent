export function l $bubble_sort(l %a, l %n) {
@start
	%lp =l alloc8 8
	%jp =l alloc8 8
	%c0 =w csltl %n, 2
	jnz %c0, @ret, @init

@init
	%limit0 =l sub %n, 1
	storel %limit0, %lp
	jmp @outer

@outer
	%limit1 =l loadl %lp
	%c1 =w csltl 0, %limit1
	jnz %c1, @inner_init, @ret

@inner_init
	storel 0, %jp
	jmp @inner

@inner
	%j0 =l loadl %jp
	%limit2 =l loadl %lp
	%c2 =w csltl %j0, %limit2
	jnz %c2, @body, @after_inner

@body
	%off0 =l mul %j0, 8
	%p0 =l add %a, %off0
	%j1 =l add %j0, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%c3 =w csltl %v1, %v0
	jnz %c3, @swap, @incj

@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @incj

@incj
	%j2 =l loadl %jp
	%j3 =l add %j2, 1
	storel %j3, %jp
	jmp @inner

@after_inner
	%limit3 =l loadl %lp
	%limit4 =l sub %limit3, 1
	storel %limit4, %lp
	jmp @outer

@ret
	ret 0
}
