export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i1
	%lt =w csltl %i, %n
	jnz %lt, @check, @done
@check
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, 0
	jnz %eq, @done, @cont
@cont
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %i
}
