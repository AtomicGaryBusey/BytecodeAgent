export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret, @sort

@sort
	%mid =l div %n, 2
	%last =l sub %n, 1
	%moff =l mul %mid, 16
	%loff =l mul %last, 16
	%mp =l add %a, %moff
	%lp =l add %a, %loff
	%mpv =l add %mp, 8
	%lpv =l add %lp, 8
	%mk =l loadl %mp
	%mv =l loadl %mpv
	%lk =l loadl %lp
	%lv =l loadl %lpv
	storel %lk, %mp
	storel %lv, %mpv
	storel %mk, %lp
	storel %mv, %lpv
	%pk =l %mk
	%pv =l %mv
	jmp @loop

@loop
	%i =l phi @sort 0, @next %inext
	%j =l phi @sort 0, @next %jnext
	%more =w csltl %j, %last
	jnz %more, @body, @finish

@body
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jpv =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jpv
	%klt =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vlt =w csltl %jv, %pv
	%ev =w and %keq, %vlt
	%less =w or %klt, %ev
	jnz %less, @doswap, @noswap

@doswap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ipv =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ipv
	storel %jk, %ip
	storel %jv, %ipv
	storel %ik, %jp
	storel %iv, %jpv
	%iadd =l add %i, 1
	jmp @next

@noswap
	jmp @next

@next
	%inext =l phi @doswap %iadd, @noswap %i
	%jnext =l add %j, 1
	jmp @loop

@finish
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ipv2 =l add %ip2, 8
	%ik2 =l loadl %ip2
	%iv2 =l loadl %ipv2
	storel %pk, %ip2
	storel %pv, %ipv2
	storel %ik2, %lp
	storel %iv2, %lpv
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%ra =l add %a, %roff
	%rn0 =l sub %n, %i
	%rn =l sub %rn0, 1
	%r2 =l call $qsort_pairs(l %ra, l %rn)
	ret 0

@ret
	ret 0
}
