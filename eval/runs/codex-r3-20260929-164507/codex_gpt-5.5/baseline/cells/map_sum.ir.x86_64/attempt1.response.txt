export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%sum =l phi @start 0, @body %sum1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum1 =l add %sum, %r
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %sum
}
