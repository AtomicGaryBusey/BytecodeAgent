export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %lo_next, @greater %lo
	%hi =l phi @start %n, @less %hi, @greater %mid
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @noteq
@noteq
	%lt =w csltl %val, %key
	jnz %lt, @less, @greater
@less
	%lo_next =l add %mid, 1
	jmp @loop
@greater
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
