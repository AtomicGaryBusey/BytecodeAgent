export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1
	%p0h =l shr %p0, 32
	%p1l =l and %p1, %mask
	%p2l =l and %p2, %mask
	%s1 =l add %p0h, %p1l
	%s2 =l add %s1, %p2l
	%carry =l shr %s2, 32
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%h1 =l add %p3, %p1h
	%h2 =l add %h1, %p2h
	%h3 =l add %h2, %carry
	ret %h3
}
