export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%sum =l add %na, %nb
	jmp @zero_loop

@zero_loop
	%zi =l phi @start 0, @zero_body %zi_next
	%zc =w csltl %zi, %sum
	jnz %zc, @zero_body, @outer_head

@zero_body
	%zoff =l shl %zi, 3
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi_next =l add %zi, 1
	jmp @zero_loop

@outer_head
	%i =l phi @zero_loop 0, @outer_next %i_next
	%oc =w csltl %i, %na
	jnz %oc, @outer_body, @trim_head

@outer_body
	%aoff =l shl %i, 3
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	jmp @inner_head

@inner_head
	%j =l phi @outer_body 0, @inner_body %j_next
	%carry =l phi @outer_body 0, @inner_body %carry_next
	%ic =w csltl %j, %nb
	jnz %ic, @inner_body, @after_inner

@inner_body
	%boff =l shl %j, 3
	%bp =l add %b, %boff
	%bj =l loadl %bp

	%ax0 =l and %ai, 4294967295
	%ax1 =l shr %ai, 32
	%by0 =l and %bj, 4294967295
	%by1 =l shr %bj, 32

	%p0 =l mul %ax0, %by0
	%p1 =l mul %ax0, %by1
	%p2 =l mul %ax1, %by0
	%p3 =l mul %ax1, %by1

	%p0hi =l shr %p0, 32
	%p0lo =l and %p0, 4294967295
	%p1lo =l and %p1, 4294967295
	%p2lo =l and %p2, 4294967295
	%t0 =l add %p0hi, %p1lo
	%t1 =l add %t0, %p2lo
	%tlo =l shl %t1, 32
	%plo =l or %p0lo, %tlo

	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%thi =l shr %t1, 32
	%h0 =l add %p3, %p1hi
	%h1 =l add %h0, %p2hi
	%phi =l add %h1, %thi

	%ij =l add %i, %j
	%roff =l shl %ij, 3
	%rp =l add %r, %roff
	%rv =l loadl %rp

	%s1 =l add %rv, %plo
	%c1w =w cultl %s1, %rv
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rp

	%cn0 =l add %phi, %c1
	%carry_next =l add %cn0, %c2
	%j_next =l add %j, 1
	jmp @inner_head

@after_inner
	%hiidx =l add %i, %nb
	%hioff =l shl %hiidx, 3
	%hip =l add %r, %hioff
	storel %carry, %hip
	jmp @outer_next

@outer_next
	%i_next =l add %i, 1
	jmp @outer_head

@trim_head
	%k =l phi @outer_head %sum, @trim_dec %k_dec
	%kg =w csgtl %k, 0
	jnz %kg, @trim_load, @ret

@trim_load
	%km1 =l sub %k, 1
	%koff =l shl %km1, 3
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @trim_dec, @ret

@trim_dec
	%k_dec =l sub %k, 1
	jmp @trim_head

@ret
	ret %k
}
