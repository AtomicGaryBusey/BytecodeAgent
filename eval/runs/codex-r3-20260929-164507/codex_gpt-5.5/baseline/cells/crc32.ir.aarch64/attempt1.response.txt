export function l $crc32(l %p, l %n) {
@start
	%more0 =w csgtl %n, 0
	jnz %more0, @loop, @done

@loop
	%ptr =l phi @start %p, @loop %pnext
	%cnt =l phi @start %n, @loop %cntnext
	%crc =w phi @start -1, @loop %c8
	%b =w loadub %ptr
	%c0 =w xor %crc, %b

	%l0 =w and %c0, 1
	%m0 =w sub 0, %l0
	%s0 =w shr %c0, 1
	%p0 =w and %m0, -306674912
	%c1 =w xor %s0, %p0

	%l1 =w and %c1, 1
	%m1 =w sub 0, %l1
	%s1 =w shr %c1, 1
	%p1 =w and %m1, -306674912
	%c2 =w xor %s1, %p1

	%l2 =w and %c2, 1
	%m2 =w sub 0, %l2
	%s2 =w shr %c2, 1
	%p2 =w and %m2, -306674912
	%c3 =w xor %s2, %p2

	%l3 =w and %c3, 1
	%m3 =w sub 0, %l3
	%s3 =w shr %c3, 1
	%p3 =w and %m3, -306674912
	%c4 =w xor %s3, %p3

	%l4 =w and %c4, 1
	%m4 =w sub 0, %l4
	%s4 =w shr %c4, 1
	%p4 =w and %m4, -306674912
	%c5 =w xor %s4, %p4

	%l5 =w and %c5, 1
	%m5 =w sub 0, %l5
	%s5 =w shr %c5, 1
	%p5 =w and %m5, -306674912
	%c6 =w xor %s5, %p5

	%l6 =w and %c6, 1
	%m6 =w sub 0, %l6
	%s6 =w shr %c6, 1
	%p6 =w and %m6, -306674912
	%c7 =w xor %s6, %p6

	%l7 =w and %c7, 1
	%m7 =w sub 0, %l7
	%s7 =w shr %c7, 1
	%p7 =w and %m7, -306674912
	%c8 =w xor %s7, %p7

	%pnext =l add %ptr, 1
	%cntnext =l sub %cnt, 1
	%more =w csgtl %cntnext, 0
	jnz %more, @loop, @done

@done
	%crcf =w phi @start -1, @loop %c8
	%outw =w xor %crcf, -1
	%out =l extuw %outw
	ret %out
}
