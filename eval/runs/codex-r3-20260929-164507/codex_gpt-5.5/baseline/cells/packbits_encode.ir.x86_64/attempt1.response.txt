export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main_loop(0, 0)

@main_loop(%i l, %op l)
	%more =w csltl %i, %n
	jnz %more, @main_body, @ret_ok

@ret_ok
	ret %op

@main_body
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	jmp @run_loop(1)

@run_loop(%r l)
	%ir =l add %i, %r
	%rn =w csltl %ir, %n
	jnz %rn, @run_check_len, @run_done(%r)

@run_check_len
	%rlim =w csltl %r, 128
	jnz %rlim, @run_load, @run_done(%r)

@run_load
	%pr =l add %src, %ir
	%br =w loadub %pr
	%eqr =w ceqw %br, %b0
	jnz %eqr, @run_inc, @run_done(%r)

@run_inc
	%r1 =l add %r, 1
	jmp @run_loop(%r1)

@run_done(%rf l)
	%isrep =w csltl 2, %rf
	jnz %isrep, @repeat_packet, @literal_start

@repeat_packet
	%needr =l add %op, 2
	%badr =w csltl %cap, %needr
	jnz %badr, @fail, @repeat_write

@repeat_write
	%hdrrep =l sub 257, %rf
	%outrep0 =l add %out, %op
	storeb %hdrrep, %outrep0
	%outrep1 =l add %outrep0, 1
	storeb %b0, %outrep1
	%inextrep =l add %i, %rf
	jmp @main_loop(%inextrep, %needr)

@literal_start
	jmp @lit_loop(%i)

@lit_loop(%j l)
	%ljn =w csltl %j, %n
	jnz %ljn, @lit_check_len, @lit_done(%j)

@lit_check_len
	%llen0 =l sub %j, %i
	%llim =w csltl %llen0, 128
	jnz %llim, @lit_run_start, @lit_done(%j)

@lit_run_start
	%lp0 =l add %src, %j
	%lb0 =w loadub %lp0
	jmp @lit_run_loop(1)

@lit_run_loop(%lr l)
	%ljr =l add %j, %lr
	%lrn =w csltl %ljr, %n
	jnz %lrn, @lit_run_check_len, @lit_run_done(%lr)

@lit_run_check_len
	%lrlim =w csltl %lr, 128
	jnz %lrlim, @lit_run_load, @lit_run_done(%lr)

@lit_run_load
	%lpr =l add %src, %ljr
	%lbr =w loadub %lpr
	%leqr =w ceqw %lbr, %lb0
	jnz %leqr, @lit_run_inc, @lit_run_done(%lr)

@lit_run_inc
	%lr1 =l add %lr, 1
	jmp @lit_run_loop(%lr1)

@lit_run_done(%lrf l)
	%stoplit =w csltl 2, %lrf
	jnz %stoplit, @lit_done(%j), @lit_advance

@lit_advance
	%j1 =l add %j, 1
	jmp @lit_loop(%j1)

@lit_done(%jd l)
	%L =l sub %jd, %i
	%needl0 =l add %op, 1
	%needl =l add %needl0, %L
	%badl =w csltl %cap, %needl
	jnz %badl, @fail, @literal_write

@literal_write
	%hdrlit =l sub %L, 1
	%outlit0 =l add %out, %op
	storeb %hdrlit, %outlit0
	jmp @copy_loop(0)

@copy_loop(%k l)
	%kc =w csltl %k, %L
	jnz %kc, @copy_body, @literal_finish

@copy_body
	%si0 =l add %src, %i
	%sik =l add %si0, %k
	%cb =w loadub %sik
	%do0 =l add %op, 1
	%dok =l add %do0, %k
	%dpk =l add %out, %dok
	storeb %cb, %dpk
	%k1 =l add %k, 1
	jmp @copy_loop(%k1)

@literal_finish
	jmp @main_loop(%jd, %needl)

@fail
	ret -1
}
