export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @checkov

@checkov
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	jnz %amin, @checkm1, @normal

@checkm1
	%bm1 =w ceql %b, -1
	jnz %bm1, @overflow, @normal

@divzero
	storel %a, %rem
	ret 0

@overflow
	storel 0, %rem
	ret %min

@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
