export function l $fib(l %n) {
@start
	%z =w ceql %n, 0
	jnz %z, @ret0, @loop

@ret0
	ret 0

@loop
	%i =l phi @start 0, @loop %i1
	%a =l phi @start 0, @loop %b
	%b =l phi @start 1, @loop %s
	%c =w csltl %i, %n
	jnz %c, @body, @done

@body
	%s =l add %a, %b
	%i1 =l add %i, 1
	jmp @loop

@done
	ret %a
}
