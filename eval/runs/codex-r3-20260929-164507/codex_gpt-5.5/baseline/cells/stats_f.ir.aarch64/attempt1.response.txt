export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    %more0 =w csltl 1, %n
    jnz %more0, @loop, @done

@loop
    %i =l phi @start 1, @max_done %inext
    %min =l phi @start %y0, @max_done %minn
    %max =l phi @start %y0, @max_done %maxn
    %sum =l phi @start %y0, @max_done %sumn
    %odd =l phi @start %odd0, @max_done %oddn
    %off =l shl %i, 3
    %p =l add %a, %off
    %x =l loadl %p
    %y =l call %f(l %x)
    %sumn =l add %sum, %y
    %b =l and %y, 1
    %oddn =l add %odd, %b
    %cmin =w csltl %y, %min
    jnz %cmin, @setmin, @keepmin

@setmin
    jmp @min_done

@keepmin
    jmp @min_done

@min_done
    %minn =l phi @setmin %y, @keepmin %min
    %cmax =w csltl %max, %y
    jnz %cmax, @setmax, @keepmax

@setmax
    jmp @max_done

@keepmax
    jmp @max_done

@max_done
    %maxn =l phi @setmax %y, @keepmax %max
    %inext =l add %i, 1
    %more =w csltl %inext, %n
    jnz %more, @loop, @done

@done
    %minf =l phi @start %y0, @max_done %minn
    %maxf =l phi @start %y0, @max_done %maxn
    %sumf =l phi @start %y0, @max_done %sumn
    %oddf =l phi @start %odd0, @max_done %oddn
    storel %minf, %out
    %out1 =l add %out, 8
    storel %maxf, %out1
    %out2 =l add %out, 16
    storel %sumf, %out2
    ret %oddf
}
