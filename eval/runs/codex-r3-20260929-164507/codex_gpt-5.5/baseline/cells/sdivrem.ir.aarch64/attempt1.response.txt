export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%zb =w ceql %b, 0
	jnz %zb, @divzero, @checkov
@checkov
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	%bneg1 =w ceql %b, -1
	%ov =w and %amin, %bneg1
	jnz %ov, @overflow, @normal
@divzero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret %a
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
