export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%twonn =l add %nn, %nn
	jmp @init_i

@init_i
	%ii =l phi @start 0, @init_i_next %ii_next
	%ic =w csltl %ii, %n
	jnz %ic, @init_j_start, @pow_loop

@init_j_start
	jmp @init_j

@init_j
	%ij =l phi @init_j_start 0, @init_j_next %ij_next
	%jc =w csltl %ij, %n
	jnz %jc, @init_val, @init_i_next

@init_val
	%eq =w ceql %ii, %ij
	jnz %eq, @init_one, @init_zero

@init_one
	%idx1 =l mul %ii, %n
	%idx2 =l add %idx1, %ij
	%off1 =l mul %idx2, 8
	%ptr1 =l add %out, %off1
	storel 1, %ptr1
	jmp @init_j_next

@init_zero
	%idx3 =l mul %ii, %n
	%idx4 =l add %idx3, %ij
	%off2 =l mul %idx4, 8
	%ptr2 =l add %out, %off2
	storel 0, %ptr2
	jmp @init_j_next

@init_j_next
	%ij_next =l add %ij, 1
	jmp @init_j

@init_i_next
	%ii_next =l add %ii, 1
	jmp @init_i

@pow_loop
	%p =l phi @init_i 0, @copy_done %p_next
	%pc =w csltl %p, %k
	jnz %pc, @prod_i_start, @zero_loop

@prod_i_start
	jmp @prod_i

@prod_i
	%pi =l phi @prod_i_start 0, @prod_i_next %pi_next
	%pic =w csltl %pi, %n
	jnz %pic, @prod_j_start, @copy_loop

@prod_j_start
	jmp @prod_j

@prod_j
	%pj =l phi @prod_j_start 0, @prod_j_next %pj_next
	%pjc =w csltl %pj, %n
	jnz %pjc, @prod_t_start, @prod_i_next

@prod_t_start
	jmp @prod_t

@prod_t
	%pt =l phi @prod_t_start 0, @prod_t_body %pt_next
	%sum =l phi @prod_t_start 0, @prod_t_body %sum_next
	%ptc =w csltl %pt, %n
	jnz %ptc, @prod_t_body, @prod_store

@prod_t_body
	%oi1 =l mul %pi, %n
	%oi2 =l add %oi1, %pt
	%oo1 =l mul %oi2, 8
	%op1 =l add %out, %oo1
	%ov =l loadl %op1
	%ai1 =l mul %pt, %n
	%ai2 =l add %ai1, %pj
	%ao1 =l mul %ai2, 8
	%ap1 =l add %a, %ao1
	%av =l loadl %ap1
	%pr =l mul %ov, %av
	%sum_next =l add %sum, %pr
	%pt_next =l add %pt, 1
	jmp @prod_t

@prod_store
	%ti1 =l mul %pi, %n
	%ti2 =l add %ti1, %pj
	%to1 =l mul %ti2, 8
	%tp1 =l add %tmp, %to1
	storel %sum, %tp1
	jmp @prod_j_next

@prod_j_next
	%pj_next =l add %pj, 1
	jmp @prod_j

@prod_i_next
	%pi_next =l add %pi, 1
	jmp @prod_i

@copy_loop
	%ci =l phi @prod_i 0, @copy_next %ci_next
	%cic =w csltl %ci, %nn
	jnz %cic, @copy_body, @copy_done

@copy_body
	%co =l mul %ci, 8
	%cp1 =l add %tmp, %co
	%cv =l loadl %cp1
	%cp2 =l add %out, %co
	storel %cv, %cp2
	jmp @copy_next

@copy_next
	%ci_next =l add %ci, 1
	jmp @copy_loop

@copy_done
	%p_next =l add %p, 1
	jmp @pow_loop

@zero_loop
	%zi =l phi @pow_loop 0, @zero_next %zi_next
	%zic =w csltl %zi, %twonn
	jnz %zic, @zero_body, @ret

@zero_body
	%zo =l mul %zi, 8
	%zp =l add %tmp, %zo
	storel 0, %zp
	jmp @zero_next

@zero_next
	%zi_next =l add %zi, 1
	jmp @zero_loop

@ret
	ret 0
}
