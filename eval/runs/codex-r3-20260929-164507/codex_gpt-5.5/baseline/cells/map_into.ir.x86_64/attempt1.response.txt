export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i1
	%cnt =l phi @start 0, @cont %cnt1
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	%dp =l add %dst, %off
	storel %y, %dp
	%lt =w csltl %y, %x
	jnz %lt, @inc, @noinc
@inc
	%cnt_inc =l add %cnt, 1
	jmp @cont
@noinc
	jmp @cont
@cont
	%cnt1 =l phi @inc %cnt_inc, @noinc %cnt
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
