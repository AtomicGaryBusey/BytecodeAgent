export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %iszero =w ceql %b, 0
    jnz %iszero, @divzero, @checkmin

@checkmin
    %ismin =w ceql %a, -9223372036854775808
    jnz %ismin, @checkminusone, @normal

@checkminusone
    %isminusone =w ceql %b, -1
    jnz %isminusone, @overflow, @normal

@divzero
    storel %a, %rem
    ret 0

@overflow
    storel 0, %rem
    ret %a

@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
