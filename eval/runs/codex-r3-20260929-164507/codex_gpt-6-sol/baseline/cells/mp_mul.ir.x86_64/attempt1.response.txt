export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %total =l add %na, %nb
    %mask =l copy 4294967295
    %z =l copy 0
    jmp @zero_test

@zero_test
    %more_zero =w csltl %z, %total
    jnz %more_zero, @zero_body, @outer_init

@zero_body
    %zoff =l shl %z, 3
    %zptr =l add %r, %zoff
    storel 0, %zptr
    %znext =l add %z, 1
    %z =l copy %znext
    jmp @zero_test

@outer_init
    %i =l copy 0
    jmp @outer_test

@outer_test
    %more_outer =w csltl %i, %na
    jnz %more_outer, @outer_body, @trim_init

@outer_body
    %ioff =l shl %i, 3
    %aptr =l add %a, %ioff
    %av =l loadl %aptr
    %al =l and %av, %mask
    %ah =l shr %av, 32
    %carry =l copy 0
    %j =l copy 0
    jmp @inner_test

@inner_test
    %more_inner =w csltl %j, %nb
    jnz %more_inner, @inner_body, @inner_done

@inner_body
    %joff =l shl %j, 3
    %bptr =l add %b, %joff
    %bv =l loadl %bptr
    %bl =l and %bv, %mask
    %bh =l shr %bv, 32

    %p0 =l mul %al, %bl
    %p1 =l mul %ah, %bl
    %p2 =l mul %al, %bh
    %p3 =l mul %ah, %bh

    %p0hi =l shr %p0, 32
    %p1lo =l and %p1, %mask
    %p2lo =l and %p2, %mask
    %mid0 =l add %p0hi, %p1lo
    %mid =l add %mid0, %p2lo
    %midlo =l and %mid, %mask
    %midshift =l shl %midlo, 32
    %p0lo =l and %p0, %mask
    %lo =l add %p0lo, %midshift

    %p1hi =l shr %p1, 32
    %p2hi =l shr %p2, 32
    %midhi =l shr %mid, 32
    %hi0 =l add %p3, %p1hi
    %hi1 =l add %hi0, %p2hi
    %hi =l add %hi1, %midhi

    %idx =l add %i, %j
    %roff =l shl %idx, 3
    %rptr =l add %r, %roff
    %old =l loadl %rptr
    %sum0 =l add %old, %lo
    %c0 =w cultl %sum0, %old
    %sum =l add %sum0, %carry
    %c1 =w cultl %sum, %sum0
    storel %sum, %rptr

    %c0l =l extuw %c0
    %c1l =l extuw %c1
    %next0 =l add %hi, %c0l
    %nextcarry =l add %next0, %c1l
    %carry =l copy %nextcarry
    %jnext =l add %j, 1
    %j =l copy %jnext
    jmp @inner_test

@inner_done
    %endidx =l add %i, %nb
    %endoff =l shl %endidx, 3
    %endptr =l add %r, %endoff
    storel %carry, %endptr
    %inext =l add %i, 1
    %i =l copy %inext
    jmp @outer_test

@trim_init
    %k =l copy %total
    jmp @trim_test

@trim_test
    %has_limb =w csltl 0, %k
    jnz %has_limb, @trim_body, @return

@trim_body
    %last =l sub %k, 1
    %lastoff =l shl %last, 3
    %lastptr =l add %r, %lastoff
    %value =l loadl %lastptr
    %is_zero =w ceql %value, 0
    jnz %is_zero, @trim_zero, @return

@trim_zero
    %k =l copy %last
    jmp @trim_test

@return
    ret %k
}
