export function l $count_nonzero_prefix(l %a, l %n) {
@entry
    jmp @test
@test
    %i =l phi @entry 0, @step %next
    %in_bounds =w csltl %i, %n
    jnz %in_bounds, @read, @done
@read
    %offset =l shl %i, 3
    %p =l add %a, %offset
    %value =l loadl %p
    %is_zero =w ceql %value, 0
    jnz %is_zero, @done, @step
@step
    %next =l add %i, 1
    jmp @test
@done
    ret %i
}
