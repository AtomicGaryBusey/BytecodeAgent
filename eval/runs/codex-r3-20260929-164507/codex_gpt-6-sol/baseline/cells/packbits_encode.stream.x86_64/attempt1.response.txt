xor r8d, r8d
xor r9d, r9d
.Lnext:
cmp r8, rsi
jae .Ldone
lea r11, [r8+2]
cmp r11, rsi
jae .Lliteral
mov al, byte ptr [rdi+r8]
cmp al, byte ptr [rdi+r8+1]
jne .Lliteral
cmp al, byte ptr [rdi+r8+2]
jne .Lliteral
mov r10d, 3
.Lcount_run:
cmp r10, 128
jae .Lemit_repeat
lea r11, [r8+r10]
cmp r11, rsi
jae .Lemit_repeat
cmp al, byte ptr [rdi+r11]
jne .Lemit_repeat
inc r10
jmp .Lcount_run
.Lemit_repeat:
lea r11, [r9+2]
cmp r11, rcx
ja .Lfail
mov r11d, 1
sub r11, r10
mov byte ptr [rdx+r9], r11b
mov byte ptr [rdx+r9+1], al
add r9, 2
add r8, r10
jmp .Lnext
.Lliteral:
mov r10, r8
.Lscan_literal:
cmp r10, rsi
jae .Lemit_literal
mov r11, r10
sub r11, r8
cmp r11, 128
jae .Lemit_literal
lea r11, [r10+2]
cmp r11, rsi
jae .Lextend_literal
mov al, byte ptr [rdi+r10]
cmp al, byte ptr [rdi+r10+1]
jne .Lextend_literal
cmp al, byte ptr [rdi+r10+2]
je .Lemit_literal
.Lextend_literal:
inc r10
jmp .Lscan_literal
.Lemit_literal:
mov r11, r10
sub r11, r8
lea rax, [r9+r11+1]
cmp rax, rcx
ja .Lfail
dec r11
mov byte ptr [rdx+r9], r11b
inc r9
.Lcopy_literal:

jae .Lnext
mov al, byte ptr [rdi+r8]
mov byte ptr [rdx+r9], al
inc r8
inc r9
jmp .Lcopy_literal
.Lfail:
mov rax, -1
ret
.Ldone:

ret
