max_byte:
    pxor xmm0, xmm0
.Lvector:
    cmp rsi, 16
    jb .Lreduce
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    jmp .Lvector
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    and eax, 255
.Ltail:
    test rsi, rsi
    jz .Ldone
    movzx edx, byte ptr [rdi]
    cmp eax, edx
    cmovb eax, edx
    inc rdi
    dec rsi
    jmp .Ltail
.Ldone:
    ret
