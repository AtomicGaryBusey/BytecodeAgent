export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bzero =w ceql %b, 0
    jnz %bzero, @zero, @check

@zero
    storel %a, %rem
    ret 0

@check
    %amin =w ceql %a, -9223372036854775808
    %bnegone =w ceql %b, -1
    %overflow =w and %amin, %bnegone
    jnz %overflow, @special, @divide

@special
    storel 0, %rem
    ret %a

@divide
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
