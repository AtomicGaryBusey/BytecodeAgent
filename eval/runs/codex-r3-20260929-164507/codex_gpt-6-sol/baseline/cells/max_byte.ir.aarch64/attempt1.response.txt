export function l $max_byte(l %p, l %n) {
@start
    jmp @check
@check
    %cur =l phi @start %p, @next %cur_next
    %remaining =l phi @start %n, @next %remaining_next
    %max =l phi @start 0, @next %max_next
    %has_byte =w csgtl %remaining, 0
    jnz %has_byte, @body, @done
@body
    %b =l loadub %cur
    %greater =w csgtl %b, %max
    jnz %greater, @higher, @lower
@higher
    jmp @next
@lower
    jmp @next
@next
    %max_next =l phi @higher %b, @lower %max
    %cur_next =l add %cur, 1
    %remaining_next =l sub %remaining, 1
    jmp @check
@done
    ret %max
}
