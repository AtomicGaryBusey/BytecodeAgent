export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%z =l copy 0
	jmp @zero_test

@zero_test
	%more_z =w csltl %z, %n
	jnz %more_z, @zero_body, @outer_init

@zero_body
	%zoff =l shl %z, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%z =l add %z, 1
	jmp @zero_test

@outer_init
	%i =l copy 0
	jmp @outer_test

@outer_test
	%more_i =w csltl %i, %na
	jnz %more_i, @outer_body, @trim_init

@outer_body
	%ioff =l shl %i, 3
	%aptr =l add %a, %ioff
	%av =l loadl %aptr
	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%carry =l copy 0
	%j =l copy 0
	jmp @inner_test

@inner_test
	%more_j =w csltl %j, %nb
	jnz %more_j, @inner_body, @outer_finish

@inner_body
	%joff =l shl %j, 3
	%bptr =l add %b, %joff
	%bv =l loadl %bptr
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%m01 =l shl %p01, 32
	%m10 =l shl %p10, 32
	%lo1 =l add %p00, %m01
	%c1 =w cultl %lo1, %p00
	%lo =l add %lo1, %m10
	%c2 =w cultl %lo, %lo1

	%h01 =l shr %p01, 32
	%h10 =l shr %p10, 32
	%hi1 =l add %p11, %h01
	%hi2 =l add %hi1, %h10
	%cc1 =l extuw %c1
	%cc2 =l extuw %c2
	%hi3 =l add %hi2, %cc1
	%hi =l add %hi3, %cc2

	%idx =l add %i, %j
	%roff =l shl %idx, 3
	%rptr =l add %r, %roff
	%old =l loadl %rptr
	%s1 =l add %lo, %old
	%c3 =w cultl %s1, %lo
	%s2 =l add %s1, %carry
	%c4 =w cultl %s2, %s1
	storel %s2, %rptr

	%cc3 =l extuw %c3
	%cc4 =l extuw %c4
	%next1 =l add %hi, %cc3
	%carry =l add %next1, %cc4
	%j =l add %j, 1
	jmp @inner_test

@outer_finish
	%endidx =l add %i, %nb
	%endoff =l shl %endidx, 3
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%i =l add %i, 1
	jmp @outer_test

@trim_init
	%k =l copy %n
	jmp @trim_test

@trim_test
	%nonzero_k =w cnel %k, 0
	jnz %nonzero_k, @trim_body, @done

@trim_body
	%last =l sub %k, 1
	%lastoff =l shl %last, 3
	%lastptr =l add %r, %lastoff
	%limb =l loadl %lastptr
	%nonzero_limb =w cnel %limb, 0
	jnz %nonzero_limb, @done, @trim_dec

@trim_dec
	%k =l copy %last
	jmp @trim_test

@done
	ret %k
}
