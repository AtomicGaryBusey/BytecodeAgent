export function l $map_sum(l %a, l %n, l %f) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %next
    %p =l phi @entry %a, @body %pnext
    %sum =l phi @entry 0, @body %sum_next
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %x =l loadl %p
    %r =l call %f(l %x)
    %sum_next =l add %sum, %r
    %pnext =l add %p, 8
    %next =l add %i, 1
    jmp @loop
@done
    ret %sum
}
