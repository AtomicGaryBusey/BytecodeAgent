export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init

@init
	%mid =l shr %n, 1
	%mid_off =l shl %mid, 4
	%mid_ptr =l add %a, %mid_off
	%pivot_key =l loadl %mid_ptr
	%mid_val_ptr =l add %mid_ptr, 8
	%pivot_val =l loadl %mid_val_ptr
	jmp @loop

@loop
	%lt =l phi @init 0, @less %lt_next, @greater %lt, @equal %lt
	%i =l phi @init 0, @less %i_next, @greater %i, @equal %i_next_equal
	%gt =l phi @init %n, @less %gt, @greater %gt_next, @equal %gt
	%more =w csltl %i, %gt
	jnz %more, @compare, @partitioned

@compare
	%i_off =l shl %i, 4
	%i_ptr =l add %a, %i_off
	%key =l loadl %i_ptr
	%i_val_ptr =l add %i_ptr, 8
	%val =l loadl %i_val_ptr
	%key_less =w csltl %key, %pivot_key
	jnz %key_less, @less, @check_key_greater

@check_key_greater
	%key_greater =w csgtl %key, %pivot_key
	jnz %key_greater, @greater, @check_val_less

@check_val_less
	%val_less =w csltl %val, %pivot_val
	jnz %val_less, @less, @check_val_greater

@check_val_greater
	%val_greater =w csgtl %val, %pivot_val
	jnz %val_greater, @greater, @equal

@less
	%lt_off =l shl %lt, 4
	%lt_ptr =l add %a, %lt_off
	%lt_key =l loadl %lt_ptr
	%lt_val_ptr =l add %lt_ptr, 8
	%lt_val =l loadl %lt_val_ptr
	storel %key, %lt_ptr
	storel %val, %lt_val_ptr
	storel %lt_key, %i_ptr
	storel %lt_val, %i_val_ptr
	%lt_next =l add %lt, 1
	%i_next =l add %i, 1
	jmp @loop

@greater
	%gt_next =l sub %gt, 1
	%gt_off =l shl %gt_next, 4
	%gt_ptr =l add %a, %gt_off
	%gt_key =l loadl %gt_ptr
	%gt_val_ptr =l add %gt_ptr, 8
	%gt_val =l loadl %gt_val_ptr
	storel %key, %gt_ptr
	storel %val, %gt_val_ptr
	storel %gt_key, %i_ptr
	storel %gt_val, %i_val_ptr
	jmp @loop

@equal
	%i_next_equal =l add %i, 1
	jmp @loop

@partitioned
	%left_large =w csgel %lt, 2
	jnz %left_large, @sort_left, @check_right

@sort_left
	%left_result =l call $qsort_pairs(l %a, l %lt)
	jmp @check_right

@check_right
	%right_size =l sub %n, %gt
	%right_large =w csgel %right_size, 2
	jnz %right_large, @sort_right, @done

@sort_right
	%right_off =l shl %gt, 4
	%right_ptr =l add %a, %right_off
	%right_result =l call $qsort_pairs(l %right_ptr, l %right_size)
	jmp @done

@done
	ret 0
}
