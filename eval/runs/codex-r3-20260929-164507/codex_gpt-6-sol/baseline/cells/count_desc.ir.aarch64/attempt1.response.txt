export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %nexti
	%count =l phi @init 0, @body %nextcount
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%nextaddr =l add %addr, 8
	%current =l loadl %addr
	%following =l loadl %nextaddr
	%desc =w csgtl %current, %following
	%increment =l extuw %desc
	%nextcount =l add %count, %increment
	%nexti =l add %i, 1
	jmp @loop
@done
	ret %count
}
