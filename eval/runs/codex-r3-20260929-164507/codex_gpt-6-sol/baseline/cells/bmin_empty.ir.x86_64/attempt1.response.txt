export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @advance %next
	%min =l phi @start 9223372036854775807, @advance %newmin
	%more =w csltl %i, %n
	jnz %more, @read, @end
@read
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%smaller =w csltl %value, %min
	jnz %smaller, @take, @keep
@take
	jmp @advance
@keep
	jmp @advance
@advance
	%newmin =l phi @take %value, @keep %min
	%next =l add %i, 1
	jmp @loop
@end
	ret %min
}
