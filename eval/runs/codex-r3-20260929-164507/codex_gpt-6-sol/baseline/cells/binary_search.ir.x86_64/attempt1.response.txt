export function l $binary_search(l %a, l %n, l %key) {
@start
    jmp @loop

@loop
    %lo =l phi @start 0, @smaller %nextlo, @larger %lo
    %hi =l phi @start %n, @smaller %hi, @larger %mid
    %nonempty =w csltl %lo, %hi
    jnz %nonempty, @probe, @notfound

@probe
    %span =l sub %hi, %lo
    %half =l div %span, 2
    %mid =l add %lo, %half
    %offset =l mul %mid, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %less =w csltl %value, %key
    jnz %less, @smaller, @checkgreater

@checkgreater
    %greater =w csgtl %value, %key
    jnz %greater, @larger, @found

@smaller
    %nextlo =l add %mid, 1
    jmp @loop

@larger
    jmp @loop

@found
    ret %mid

@notfound
    ret -1
}
