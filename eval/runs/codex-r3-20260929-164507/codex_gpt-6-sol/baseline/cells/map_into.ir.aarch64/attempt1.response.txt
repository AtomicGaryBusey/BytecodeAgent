export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %next
    %count =l phi @start 0, @body %countnext
    %more =w csltl %i, %n
    jnz %more, @body, @end
@body
    %off =l mul %i, 8
    %srcaddr =l add %src, %off
    %x =l loadl %srcaddr
    %r =l call %f(l %x)
    %dstaddr =l add %dst, %off
    storel %r, %dstaddr
    %less =w csltl %r, %x
    %increment =l extuw %less
    %countnext =l add %count, %increment
    %next =l add %i, 1
    jmp @loop
@end
    ret %count
}
