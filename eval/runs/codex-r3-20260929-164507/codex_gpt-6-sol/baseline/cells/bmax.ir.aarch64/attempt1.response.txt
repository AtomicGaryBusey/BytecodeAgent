export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    jmp @loop

@loop
    %i =l phi @start 1, @step %nexti
    %max =l phi @start %first, @step %nextmax
    %more =w csltl %i, %n
    jnz %more, @body, @done

@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w csgtl %value, %max
    jnz %greater, @usevalue, @usemax

@usevalue
    jmp @step

@usemax
    jmp @step

@step
    %nextmax =l phi @usevalue %value, @usemax %max
    %nexti =l add %i, 1
    jmp @loop

@done
    ret %max
}
