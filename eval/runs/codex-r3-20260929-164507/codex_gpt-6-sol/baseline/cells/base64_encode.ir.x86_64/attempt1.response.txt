export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @group

@group
    %i =l phi @start 0, @next %nexti
    %base =l phi @start 0, @next %nextbase
    %more =w csltl %i, %n
    jnz %more, @loadfirst, @done

@loadfirst
    %remaining =l sub %n, %i
    %p0 =l add %src, %i
    %b0 =l loadub %p0
    %v0 =l shl %b0, 16
    %has2 =w csgel %remaining, 2
    jnz %has2, @loadsecond, @testthird

@loadsecond
    %p1 =l add %p0, 1
    %b1 =l loadub %p1
    %b1shift =l shl %b1, 8
    %v1 =l or %v0, %b1shift
    jmp @testthird

@testthird
    %v2 =l phi @loadfirst %v0, @loadsecond %v1
    %has3 =w csgel %remaining, 3
    jnz %has3, @loadthird, @digits

@loadthird
    %p2 =l add %p0, 2
    %b2 =l loadub %p2
    %v3 =l or %v2, %b2
    jmp @digits

@digits
    %value =l phi @testthird %v2, @loadthird %v3
    jmp @digit

@digit
    %j =l phi @digits 0, @written %nextj
    %isdata =w cslel %j, %remaining
    jnz %isdata, @index, @padding

@index
    %sixj =l mul %j, 6
    %shift =l sub 18, %sixj
    %shifted =l shr %value, %shift
    %idx =l and %shifted, 63
    %lt26 =w csltl %idx, 26
    jnz %lt26, @upper, @test52

@upper
    %upperchar =l add %idx, 65
    jmp @emit

@test52
    %lt52 =w csltl %idx, 52
    jnz %lt52, @lower, @test62

@lower
    %lowerchar =l add %idx, 71
    jmp @emit

@test62
    %lt62 =w csltl %idx, 62
    jnz %lt62, @number, @symbol

@number
    %numberchar =l sub %idx, 4
    jmp @emit

@symbol
    %isplus =w ceql %idx, 62
    jnz %isplus, @plus, @slash

@plus
    jmp @emit

@slash
    jmp @emit

@padding
    jmp @emit

@emit
    %char =l phi @upper %upperchar, @lower %lowerchar, @number %numberchar, @plus 43, @slash 47, @padding 61
    %position =l add %base, %j
    %dest =l add %out, %position
    storeb %char, %dest
    jmp @written

@written
    %nextj =l add %j, 1
    %again =w csltl %nextj, 4
    jnz %again, @digit, @next

@next
    %nexti =l add %i, 3
    %nextbase =l add %base, 4
    jmp @group

@done
    ret %base
}
