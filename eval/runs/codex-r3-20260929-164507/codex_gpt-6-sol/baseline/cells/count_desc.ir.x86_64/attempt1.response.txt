export function l $count_desc(l %a, l %n) {
@start
    %short =w csltl %n, 2
    jnz %short, @zero, @init

@zero
    ret 0

@init
    %limit =l sub %n, 1
    jmp @loop

@loop
    %i =l phi @init 0, @body %next_i
    %count =l phi @init 0, @body %next_count
    %more =w csltl %i, %limit
    jnz %more, @body, @done

@body
    %offset =l shl %i, 3
    %ptr =l add %a, %offset
    %current =l loadl %ptr
    %next_ptr =l add %ptr, 8
    %following =l loadl %next_ptr
    %desc =w csltl %following, %current
    %increment =l extuw %desc
    %next_count =l add %count, %increment
    %next_i =l add %i, 1
    jmp @loop

@done
    ret %count
}
