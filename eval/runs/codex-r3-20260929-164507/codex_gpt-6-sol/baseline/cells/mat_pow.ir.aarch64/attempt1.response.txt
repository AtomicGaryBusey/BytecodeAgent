export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%limit =l mul %nn, 2
	%i =l copy 0
@init_i
	%ic =w csltl %i, %n
	jnz %ic, @init_j_start, @power_start
@init_j_start
	%j =l copy 0
@init_j
	%jc =w csltl %j, %n
	jnz %jc, @init_element, @init_next_i
@init_element
	%idx =l mul %i, %n
	%idx =l add %idx, %j
	%off =l mul %idx, 8
	%ptr =l add %out, %off
	%equal =w ceql %i, %j
	%value =l extuw %equal
	storel %value, %ptr
	%j =l add %j, 1
	jmp @init_j
@init_next_i
	%i =l add %i, 1
	jmp @init_i
@power_start
	%step =l copy 0
@power_test
	%more =w csltl %step, %k
	jnz %more, @mul_i_start, @clear_start
@mul_i_start
	%i =l copy 0
@mul_i
	%ic2 =w csltl %i, %n
	jnz %ic2, @mul_j_start, @copy_start
@mul_j_start
	%j =l copy 0
@mul_j
	%jc2 =w csltl %j, %n
	jnz %jc2, @mul_t_start, @mul_next_i
@mul_t_start
	%t =l copy 0
	%sum =l copy 0
@mul_t
	%tc =w csltl %t, %n
	jnz %tc, @mul_term, @mul_store
@mul_term
	%left_idx =l mul %i, %n
	%left_idx =l add %left_idx, %t
	%left_off =l mul %left_idx, 8
	%left_ptr =l add %out, %left_off
	%left =l loadl %left_ptr
	%right_idx =l mul %t, %n
	%right_idx =l add %right_idx, %j
	%right_off =l mul %right_idx, 8
	%right_ptr =l add %a, %right_off
	%right =l loadl %right_ptr
	%product =l mul %left, %right
	%sum =l add %sum, %product
	%t =l add %t, 1
	jmp @mul_t
@mul_store
	%dst_idx =l mul %i, %n
	%dst_idx =l add %dst_idx, %j
	%dst_off =l mul %dst_idx, 8
	%dst_ptr =l add %tmp, %dst_off
	storel %sum, %dst_ptr
	%j =l add %j, 1
	jmp @mul_j
@mul_next_i
	%i =l add %i, 1
	jmp @mul_i
@copy_start
	%idx_copy =l copy 0
@copy_test
	%copy_more =w csltl %idx_copy, %nn
	jnz %copy_more, @copy_element, @power_next
@copy_element
	%copy_off =l mul %idx_copy, 8
	%copy_src =l add %tmp, %copy_off
	%copy_dst =l add %out, %copy_off
	%copy_value =l loadl %copy_src
	storel %copy_value, %copy_dst
	%idx_copy =l add %idx_copy, 1
	jmp @copy_test
@power_next
	%step =l add %step, 1
	jmp @power_test
@clear_start
	%clear_idx =l copy 0
@clear_test
	%clear_more =w csltl %clear_idx, %limit
	jnz %clear_more, @clear_element, @done
@clear_element
	%clear_off =l mul %clear_idx, 8
	%clear_ptr =l add %tmp, %clear_off
	storel 0, %clear_ptr
	%clear_idx =l add %clear_idx, 1
	jmp @clear_test
@done
	ret 0
}
