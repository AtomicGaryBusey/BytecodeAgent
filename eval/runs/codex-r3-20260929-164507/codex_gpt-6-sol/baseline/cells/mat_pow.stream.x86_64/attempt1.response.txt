push rbp
push r12
push r13
push r14
push r15

cld
mov rbp, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
mov r15, r8
mov rbx, r12
imul rbx, r12
lea r8, [r15 + rbx*8]
xor rcx, rcx
.Linit:
mov rax, qword ptr [rbp + rcx*8]
mov qword ptr [r15 + rcx*8], rax
mov qword ptr [r14 + rcx*8], 0
inc rcx
cmp rcx, rbx
jb .Linit
xor rcx, rcx
.Lidentity:
mov rax, rcx
imul rax, r12
add rax, rcx
mov qword ptr [r14 + rax*8], 1
inc rcx
cmp rcx, r12
jb .Lidentity
test r13, r13
jz .Lclear
.Lpower:
xor r9, r9
.Lrow:
mov rdx, r9
imul rdx, r12
xor r10, r10
.Lcol:
xor rdi, rdi
xor r11, r11
.Ldot:
lea rsi, [rdx + r11]
mov rax, qword ptr [r14 + rsi*8]
mov rsi, r11
imul rsi, r12
add rsi, r10
imul rax, qword ptr [r15 + rsi*8]
add rdi, rax
inc r11
cmp r11, r12
jb .Ldot
lea rsi, [rdx + r10]
mov qword ptr [r8 + rsi*8], rdi
inc r10
cmp r10, r12
jb .Lcol
inc r9
cmp r9, r12
jb .Lrow
xor rcx, rcx
.Lcopy:
mov rax, qword ptr [r8 + rcx*8]
mov qword ptr [r14 + rcx*8], rax
inc rcx
cmp rcx, rbx
jb .Lcopy
dec r13
jnz .Lpower
.Lclear:
lea rdx, [rbx + rbx]
xor rcx, rcx
.Lzero:
mov qword ptr [r15 + rcx*8], 0
inc rcx
cmp rcx, rdx
jb .Lzero
xor eax, eax
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
