export function l $bubble_sort(l %a, l %n) {
@entry
    jmp @outer
@outer
    %i =l phi @entry 0, @advance %nexti
    %more =w csltl %i, %n
    jnz %more, @inner_init, @done
@inner_init
    %limit =l sub %n, %i
    jmp @inner
@inner
    %j =l phi @inner_init 0, @step %nextj
    %j1 =l add %j, 1
    %has_pair =w csltl %j1, %limit
    jnz %has_pair, @compare, @advance
@compare
    %offset =l shl %j, 3
    %p =l add %a, %offset
    %q =l add %p, 8
    %x =l loadl %p
    %y =l loadl %q
    %swap =w csltl %y, %x
    jnz %swap, @swap, @step
@swap
    storel %y, %p
    storel %x, %q
    jmp @step
@step
    %nextj =l add %j, 1
    jmp @inner
@advance
    %nexti =l add %i, 1
    jmp @outer
@done
    ret 0
}
