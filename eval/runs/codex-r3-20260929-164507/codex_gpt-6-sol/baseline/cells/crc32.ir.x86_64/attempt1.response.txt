export function l $crc32(l %p, l %n) {
@entry
    jmp @outer
@outer
    %i =l phi @entry 0, @advance %inext
    %crc =w phi @entry -1, @advance %cnext
    %more =w csltl %i, %n
    jnz %more, @load, @done
@load
    %ptr =l add %p, %i
    %byte =w loadub %ptr
    %init =w xor %crc, %byte
    jmp @bits
@bits
    %j =w phi @load 0, @step %jnext
    %c =w phi @load %init, @step %updated
    %odd =w and %c, 1
    %shift =w shr %c, 1
    jnz %odd, @poly, @plain
@poly
    %polyc =w xor %shift, -306674912
    jmp @step
@plain
    jmp @step
@step
    %updated =w phi @poly %polyc, @plain %shift
    %jnext =w add %j, 1
    %remaining =w csltw %jnext, 8
    jnz %remaining, @bits, @advance
@advance
    %cnext =w copy %updated
    %inext =l add %i, 1
    jmp @outer
@done
    %final =w xor %crc, -1
    %result =l extuw %final
    ret %result
}
