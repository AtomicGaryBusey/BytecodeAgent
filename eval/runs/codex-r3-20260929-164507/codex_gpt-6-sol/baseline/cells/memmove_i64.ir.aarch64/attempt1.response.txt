export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @direction
@direction
	%forward =w cultl %dst, %src
	jnz %forward, @forward_loop, @backward_init
@forward_loop
	%i =l phi @direction 0, @forward_loop %next
	%offset =l shl %i, 3
	%source =l add %src, %offset
	%target =l add %dst, %offset
	%value =l loadl %source
	storel %value, %target
	%next =l add %i, 1
	%more =w csltl %next, %n
	jnz %more, @forward_loop, @done
@backward_init
	%last =l sub %n, 1
	jmp @backward_loop
@backward_loop
	%j =l phi @backward_init %last, @backward_loop %prev
	%back_offset =l shl %j, 3
	%back_source =l add %src, %back_offset
	%back_target =l add %dst, %back_offset
	%back_value =l loadl %back_source
	storel %back_value, %back_target
	%at_zero =w ceql %j, 0
	%prev =l sub %j, 1
	jnz %at_zero, @done, @backward_loop
@done
	ret 0
}
