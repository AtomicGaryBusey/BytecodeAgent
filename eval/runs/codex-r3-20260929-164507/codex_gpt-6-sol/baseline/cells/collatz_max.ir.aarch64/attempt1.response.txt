export function l $collatz_max(l %n) {
@start
    jmp @outer
@outer
    %k =l phi @start 1, @next %knext
    %best =l phi @start 0, @next %bestnext
    %valid =w cslel %k, %n
    jnz %valid, @innerentry, @return
@innerentry
    jmp @inner
@inner
    %x =l phi @innerentry %k, @step %xnext
    %count =l phi @innerentry 0, @step %countnext
    %done =w ceql %x, 1
    jnz %done, @finish, @process
@process
    %bit =l and %x, 1
    %odd =w cnel %bit, 0
    jnz %odd, @odd, @even
@odd
    %oddnext =l mul %x, 3
    %oddnext1 =l add %oddnext, 1
    jmp @step
@even
    %evennext =l div %x, 2
    jmp @step
@step
    %xnext =l phi @odd %oddnext1, @even %evennext
    %countnext =l add %count, 1
    jmp @inner
@finish
    %greater =w csgtl %count, %best
    jnz %greater, @improve, @retain
@improve
    jmp @next
@retain
    jmp @next
@next
    %bestnext =l phi @improve %count, @retain %best
    %knext =l add %k, 1
    jmp @outer
@return
    ret %best
}
