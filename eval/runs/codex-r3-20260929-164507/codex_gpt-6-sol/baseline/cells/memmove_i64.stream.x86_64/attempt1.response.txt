memmove_i64:
test rdx, rdx
jz .Ldone
cmp rdi, rsi
ja .Lbackward
xor rcx, rcx
.Lforward_loop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
cmp rcx, rdx
jb .Lforward_loop
jmp .Ldone
.Lbackward:
dec rdx
mov r8, qword ptr [rsi + rdx*8]
mov qword ptr [rdi + rdx*8], r8
test rdx, rdx
jnz .Lbackward
.Ldone:
xor eax, eax
ret
