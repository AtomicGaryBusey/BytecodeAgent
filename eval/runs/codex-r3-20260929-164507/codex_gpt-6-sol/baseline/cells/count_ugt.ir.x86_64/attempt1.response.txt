export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @check
@check
    %i =l phi @start 0, @body %next
    %count =l phi @start 0, @body %newcount
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l shl %i, 3
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w cugtl %value, %t
    %increment =l extuw %greater
    %newcount =l add %count, %increment
    %next =l add %i, 1
    jmp @check
@done
    ret %count
}
