base64_encode:
mov r11, rdx
test rsi, rsi
je .Ldone
.Lgroup:
movzx r8d, byte ptr [rdi]
shl r8d, 16
mov r9d, 2
cmp rsi, 1
je .Lencode
movzx eax, byte ptr [rdi + 1]
shl eax, 8
or r8d, eax
mov r9d, 3
cmp rsi, 2
je .Lencode
movzx eax, byte ptr [rdi + 2]
or r8d, eax
mov r9d, 4
.Lencode:
xor r10d, r10d
.Lchar:
cmp r10d, r9d
jae .Lpadding
lea ecx, [r10 + r10*2]
add ecx, ecx
mov eax, 18
sub eax, ecx
mov ecx, eax
mov eax, r8d
shr eax, cl
and eax, 63
cmp eax, 26
jb .Lupper
cmp eax, 52
jb .Llower
cmp eax, 62
jb .Ldigit
je .Lplus
mov al, '/'
jmp .Lstore
.Lupper:
add eax, 'A'
jmp .Lstore
.Llower:
add eax, 'a' - 26
jmp .Lstore
.Ldigit:
sub eax, 4
jmp .Lstore
.Lplus:
mov al, '+'
jmp .Lstore
.Lpadding:
mov al, '='
.Lstore:
mov byte ptr [rdx + r10], al
inc r10d
cmp r10d, 4
jb .Lchar
add rdx, 4
cmp rsi, 3
jbe .Ldone
add rdi, 3
sub rsi, 3
jmp .Lgroup
.Ldone:
mov rax, rdx
sub rax, r11
ret
