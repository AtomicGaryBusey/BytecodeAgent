export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @outer

@outer
    %in =l phi @start 0, @next %nextin
    %o =l phi @start 0, @next %nextout
    %rem =l sub %n, %in
    %empty =w ceql %rem, 0
    jnz %empty, @done, @choose

@choose
    %short =w csltl %rem, 3
    jnz %short, @partial, @full

@partial
    jmp @ready

@full
    jmp @ready

@ready
    %k =l phi @partial %rem, @full 3
    jmp @read

@read
    %r =l phi @ready 0, @readbyte %rnext
    %acc =l phi @ready 0, @readbyte %accnext
    %more =w csltl %r, %k
    jnz %more, @readbyte, @prepared

@readbyte
    %pos =l add %in, %r
    %p =l add %src, %pos
    %b =l loadub %p
    %shifted =l shl %acc, 8
    %accnext =l or %shifted, %b
    %rnext =l add %r, 1
    jmp @read

@prepared
    %missing =l sub 3, %k
    %padshift =l mul %missing, 8
    %bits =l shl %acc, %padshift
    %valid =l add %k, 1
    jmp @outloop

@outloop
    %j =l phi @prepared 0, @write %nextj
    %finished =w ceql %j, 4
    jnz %finished, @next, @outputchar

@outputchar
    %ispad =w csgel %j, %valid
    jnz %ispad, @pad, @index

@index
    %six =l mul %j, 6
    %amount =l sub 18, %six
    %value =l shr %bits, %amount
    %idx =l and %value, 63
    %isupper =w csltl %idx, 26
    jnz %isupper, @upper, @checklower

@upper
    %upperchar =l add %idx, 65
    jmp @mapped

@checklower
    %islower =w csltl %idx, 52
    jnz %islower, @lower, @checkdigit

@lower
    %lowerchar =l add %idx, 71
    jmp @mapped

@checkdigit
    %isdigit =w csltl %idx, 62
    jnz %isdigit, @digit, @special

@digit
    %digitchar =l sub %idx, 4
    jmp @mapped

@special
    %fouridx =l mul %idx, 4
    %specialchar =l sub %fouridx, 205
    jmp @mapped

@mapped
    %ch =l phi @upper %upperchar, @lower %lowerchar, @digit %digitchar, @special %specialchar
    jmp @write

@pad
    jmp @write

@write
    %c =l phi @pad 61, @mapped %ch
    %outptr =l add %out, %o
    %dst =l add %outptr, %j
    storeb %c, %dst
    %nextj =l add %j, 1
    jmp @outloop

@next
    %nextin =l add %in, %k
    %nextout =l add %o, 4
    jmp @outer

@done
    ret %o
}
