export function l $bubble_sort(l %a, l %n) {
@start
    %i =l copy 0
    jmp @outer
@outer
    %more =w csltl %i, %n
    jnz %more, @inner_init, @done
@inner_init
    %limit =l sub %n, 1
    %limit =l sub %limit, %i
    %j =l copy 0
    jmp @inner
@inner
    %has_pair =w csltl %j, %limit
    jnz %has_pair, @compare, @next_outer
@compare
    %offset =l shl %j, 3
    %p =l add %a, %offset
    %q =l add %p, 8
    %x =l loadl %p
    %y =l loadl %q
    %swap =w csltl %y, %x
    jnz %swap, @do_swap, @next_inner
@do_swap
    storel %y, %p
    storel %x, %q
    jmp @next_inner
@next_inner
    %j =l add %j, 1
    jmp @inner
@next_outer
    %i =l add %i, 1
    jmp @outer
@done
    ret 0
}
