export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @check
@check
    %i =l phi @start 0, @next %i_next
    %in_bounds =w csltl %i, %n
    jnz %in_bounds, @load, @done
@load
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %value =l loadl %addr
    %nonzero =w cnel %value, 0
    jnz %nonzero, @next, @done
@next
    %i_next =l add %i, 1
    jmp @check
@done
    ret %i
}
