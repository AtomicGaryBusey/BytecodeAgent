qsort_pairs:
cld
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
.Lsort:
cmp r12, 1
jle .Ldone
mov rcx, r12
shr rcx, 1
add rcx, rcx
mov r13, qword ptr [rbx + rcx*8]
mov r14, qword ptr [rbx + rcx*8 + 8]
xor r15d, r15d
xor ebp, ebp
mov r10, r12
.Lpartition:
cmp rbp, r10
jge .Lpartition_done
lea rcx, [rbp + rbp]
lea r11, [rbx + rcx*8]
mov rax, qword ptr [r11]
mov rdx, qword ptr [r11 + 8]
cmp rax, r13
jl .Lless
jg .Lgreater
cmp rdx, r14
jl .Lless
jg .Lgreater
inc rbp
jmp .Lpartition
.Lless:
lea rcx, [r15 + r15]
lea rcx, [rbx + rcx*8]
mov r8, qword ptr [rcx]
mov r9, qword ptr [rcx + 8]
mov qword ptr [rcx], rax
mov qword ptr [rcx + 8], rdx
mov qword ptr [r11], r8
mov qword ptr [r11 + 8], r9
inc r15
inc rbp
jmp .Lpartition
.Lgreater:
dec r10
lea rcx, [r10 + r10]
lea rcx, [rbx + rcx*8]
mov r8, qword ptr [rcx]
mov r9, qword ptr [rcx + 8]
mov qword ptr [rcx], rax
mov qword ptr [rcx + 8], rdx
mov qword ptr [r11], r8
mov qword ptr [r11 + 8], r9
jmp .Lpartition
.Lpartition_done:
mov rsi, r12
sub rsi, r10
cmp r15, rsi
jg .Lright_first
cmp r15, 1
jle .Lskip_left
mov qword ptr [rsp], r10
mov rdi, rbx
mov rsi, r15
call qsort_pairs
mov r10, qword ptr [rsp]
.Lskip_left:
sub r12, r10
lea rcx, [r10 + r10]
lea rbx, [rbx + rcx*8]
jmp .Lsort
.Lright_first:
cmp rsi, 1
jle .Lskip_right
lea rcx, [r10 + r10]
lea rdi, [rbx + rcx*8]
call qsort_pairs
.Lskip_right:
mov r12, r15
jmp .Lsort
.Ldone:
xor eax, eax
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
