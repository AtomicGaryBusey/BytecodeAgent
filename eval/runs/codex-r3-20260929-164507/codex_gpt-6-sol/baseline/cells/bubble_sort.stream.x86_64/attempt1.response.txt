bubble_sort:
cmp rsi, 1
jle .Ldone
mov r8, rsi
dec r8
.Louter:
xor edx, edx
.Linner:
mov rcx, qword ptr [rdi + rdx*8]
mov r9, qword ptr [rdi + rdx*8 + 8]
cmp rcx, r9
jle .Lnext
mov qword ptr [rdi + rdx*8], r9
mov qword ptr [rdi + rdx*8 + 8], rcx
.Lnext:
inc rdx
cmp rdx, r8
jb .Linner
dec r8
jnz .Louter
.Ldone:
xor eax, eax
ret
