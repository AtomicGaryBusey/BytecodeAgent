export function l $binary_search(l %a, l %n, l %key) {
@start
    jmp @loop

@loop
    %lo =l phi @start 0, @less %nextlo, @greater %lo
    %hi =l phi @start %n, @less %hi, @greater %mid
    %has_more =w csltl %lo, %hi
    jnz %has_more, @probe, @miss

@probe
    %sum =l add %lo, %hi
    %mid =l div %sum, 2
    %offset =l mul %mid, 8
    %addr =l add %a, %offset
    %value =l loadl %addr
    %equal =w ceql %value, %key
    jnz %equal, @found, @compare

@compare
    %below =w csltl %value, %key
    jnz %below, @less, @greater

@less
    %nextlo =l add %mid, 1
    jmp @loop

@greater
    jmp @loop

@found
    ret %mid

@miss
    ret -1
}
