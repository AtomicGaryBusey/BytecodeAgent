export function l $popcount(l %x) {
@start
    jmp @loop
@loop
    %n =l phi @start %x, @body %next
    %count =l phi @start 0, @body %inc
    %nonzero =w cnel %n, 0
    jnz %nonzero, @body, @done
@body
    %minus =l sub %n, 1
    %next =l and %n, %minus
    %inc =l add %count, 1
    jmp @loop
@done
    ret %count
}
