export function l $collatz_max(l %n) {
@start
    jmp @outer
@outer
    %k =l phi @start 1, @advance %knext
    %best =l phi @start 0, @advance %newbest
    %past =w csgtl %k, %n
    jnz %past, @done, @begin
@begin
    jmp @inner
@inner
    %x =l phi @begin %k, @update %xnext
    %steps =l phi @begin 0, @update %stepsnext
    %atone =w ceql %x, 1
    jnz %atone, @finish, @branch
@branch
    %bit =l and %x, 1
    %odd =w cnel %bit, 0
    jnz %odd, @oddblock, @evenblock
@oddblock
    %triple =l mul %x, 3
    %oddnext =l add %triple, 1
    jmp @update
@evenblock
    %evennext =l div %x, 2
    jmp @update
@update
    %xnext =l phi @oddblock %oddnext, @evenblock %evennext
    %stepsnext =l add %steps, 1
    jmp @inner
@finish
    %bigger =w csgtl %steps, %best
    jnz %bigger, @replace, @keep
@replace
    jmp @advance
@keep
    jmp @advance
@advance
    %newbest =l phi @replace %steps, @keep %best
    %knext =l add %k, 1
    jmp @outer
@done
    ret %best
}
