export function l $count_ugt(l %a, l %n, l %t) {
@start
    %has_elements =w csgtl %n, 0
    jnz %has_elements, @loop, @empty

@loop
    %i =l phi @start 0, @loop %next
    %count =l phi @start 0, @loop %updated
    %offset =l shl %i, 3
    %addr =l add %a, %offset
    %value =l loadl %addr
    %greater =w cugtl %value, %t
    %increment =l extsw %greater
    %updated =l add %count, %increment
    %next =l add %i, 1
    %more =w csltl %next, %n
    jnz %more, @loop, @done

@empty
    ret 0

@done
    ret %updated
}
