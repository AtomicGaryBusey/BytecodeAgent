export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l copy 0
	jmp @head
@head
	%i =l phi @start %zero, @noop %header_next, @lit_end %lit_next, @rep_end %rep_next
	%o =l phi @start %zero, @noop %o, @lit_end %lit_o_end, @rep_end %rep_o_end
	%more =w csltl %i, %n
	jnz %more, @header, @done
@header
	%header_ptr =l add %src, %i
	%h =w loadsb %header_ptr
	%header_next =l add %i, 1
	%noop_test =w ceqw %h, -128
	jnz %noop_test, @noop, @classify
@noop
	jmp @head
@classify
	%repeat_test =w csltw %h, 0
	jnz %repeat_test, @rep_check, @lit_check
@lit_check
	%h64 =l extsw %h
	%lit_count =l add %h64, 1
	%remaining =l sub %n, %header_next
	%lit_bad =w csltl %remaining, %lit_count
	jnz %lit_bad, @error_source, @lit_capacity
@lit_capacity
	%lit_next =l add %header_next, %lit_count
	%available =l sub %cap, %o
	%lit_clipped =w csltl %available, %lit_count
	jnz %lit_clipped, @lit_short, @lit_full
@lit_short
	jmp @lit_begin
@lit_full
	jmp @lit_begin
@lit_begin
	%lit_goal =l phi @lit_short %available, @lit_full %lit_count
	jmp @lit_loop
@lit_loop
	%lj =l phi @lit_begin %zero, @lit_copy %lj_next
	%lit_more =w csltl %lj, %lit_goal
	jnz %lit_more, @lit_copy, @lit_copied
@lit_copy
	%lit_src_offset =l add %header_next, %lj
	%lit_src_ptr =l add %src, %lit_src_offset
	%lit_byte =w loadub %lit_src_ptr
	%lit_out_offset =l add %o, %lj
	%lit_out_ptr =l add %out, %lit_out_offset
	storeb %lit_byte, %lit_out_ptr
	%lj_next =l add %lj, 1
	jmp @lit_loop
@lit_copied
	jnz %lit_clipped, @error_capacity, @lit_end
@lit_end
	%lit_o_end =l add %o, %lit_count
	jmp @head
@rep_check
	%rep_have_byte =w csltl %header_next, %n
	jnz %rep_have_byte, @rep_capacity, @error_source
@rep_capacity
	%rep_src_ptr =l add %src, %header_next
	%rep_byte =w loadub %rep_src_ptr
	%rep_next =l add %header_next, 1
	%rep_h64 =l extsw %h
	%rep_count =l sub 1, %rep_h64
	%rep_available =l sub %cap, %o
	%rep_clipped =w csltl %rep_available, %rep_count
	jnz %rep_clipped, @rep_short, @rep_full
@rep_short
	jmp @rep_begin
@rep_full
	jmp @rep_begin
@rep_begin
	%rep_goal =l phi @rep_short %rep_available, @rep_full %rep_count
	jmp @rep_loop
@rep_loop
	%rj =l phi @rep_begin %zero, @rep_copy %rj_next
	%rep_more =w csltl %rj, %rep_goal
	jnz %rep_more, @rep_copy, @rep_copied
@rep_copy
	%rep_out_offset =l add %o, %rj
	%rep_out_ptr =l add %out, %rep_out_offset
	storeb %rep_byte, %rep_out_ptr
	%rj_next =l add %rj, 1
	jmp @rep_loop
@rep_copied
	jnz %rep_clipped, @error_capacity, @rep_end
@rep_end
	%rep_o_end =l add %o, %rep_count
	jmp @head
@done
	ret %o
@error_source
	ret -2
@error_capacity
	ret -1
}
