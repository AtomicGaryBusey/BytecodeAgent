export function l $fill_down(l %a, l %n, l %start) {
@entry
    %i =l copy 0
    jmp @loop
@loop
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %p =l add %a, %offset
    %v =l sub %start, %i
    storel %v, %p
    %i =l add %i, 1
    jmp @loop
@done
    ret %n
}
