export function l $popcount(l %x) {
@start
    %shift1 =l shr %x, 1
    %pairs =l and %shift1, 0x5555555555555555
    %a =l sub %x, %pairs

    %shift2 =l shr %a, 2
    %lo =l and %a, 0x3333333333333333
    %hi =l and %shift2, 0x3333333333333333
    %b =l add %lo, %hi

    %shift4 =l shr %b, 4
    %sum =l add %b, %shift4
    %c =l and %sum, 0x0f0f0f0f0f0f0f0f

    %product =l mul %c, 0x0101010101010101
    %result =l shr %product, 56
    ret %result
}
