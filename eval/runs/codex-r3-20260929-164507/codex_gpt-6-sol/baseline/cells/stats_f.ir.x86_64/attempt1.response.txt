export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    jmp @loop

@loop
    %i =l phi @start 1, @step %next
    %min =l phi @start %y0, @step %newmin
    %max =l phi @start %y0, @step %newmax
    %sum =l phi @start %y0, @step %newsum
    %odd =l phi @start %odd0, @step %newodd
    %more =w csltl %i, %n
    jnz %more, @body, @done

@body
    %offset =l mul %i, 8
    %p =l add %a, %offset
    %x =l loadl %p
    %y =l call %f(l %x)
    %ismin =w csltl %y, %min
    jnz %ismin, @minyes, @minjoin

@minyes
    jmp @minjoin

@minjoin
    %newmin =l phi @body %min, @minyes %y
    %ismax =w csltl %max, %y
    jnz %ismax, @maxyes, @maxjoin

@maxyes
    jmp @maxjoin

@maxjoin
    %newmax =l phi @minjoin %max, @maxyes %y
    %newsum =l add %sum, %y
    %isodd =l and %y, 1
    %newodd =l add %odd, %isodd
    %next =l add %i, 1
    jmp @step

@step
    jmp @loop

@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
