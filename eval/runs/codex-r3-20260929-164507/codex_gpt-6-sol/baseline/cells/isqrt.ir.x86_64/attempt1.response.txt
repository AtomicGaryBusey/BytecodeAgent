export function l $isqrt(l %n) {
@start
	jmp @loop

@loop
	%rem =l phi @start 0, @next %remnext
	%root =l phi @start 0, @next %rootnext
	%stream =l phi @start %n, @next %streamnext
	%count =l phi @start 32, @next %countnext

	%bits =l shr %stream, 62
	%streamnext =l shl %stream, 2
	%scaled =l shl %rem, 2
	%remcand =l or %scaled, %bits
	%trialbase =l shl %root, 2
	%trial =l add %trialbase, 1
	%rootshift =l shl %root, 1
	%small =w cultl %remcand, %trial
	jnz %small, @skip, @take

@take
	%remtake =l sub %remcand, %trial
	%roottake =l add %rootshift, 1
	jmp @next

@skip
	jmp @next

@next
	%remnext =l phi @take %remtake, @skip %remcand
	%rootnext =l phi @take %roottake, @skip %rootshift
	%countnext =l sub %count, 1
	%more =w cnel %countnext, 0
	jnz %more, @loop, @done

@done
	ret %rootnext
}
