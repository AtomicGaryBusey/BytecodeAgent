reverse:
    xor eax, eax
    cmp rsi, 1
    jle .Ldone
    dec rsi
    xor ecx, ecx
.Lloop:
    cmp rcx, rsi
    jae .Ldone
    mov rdx, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rsi*8]
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rsi*8], rdx
    inc rcx
    dec rsi
    jmp .Lloop
.Ldone:
    ret
