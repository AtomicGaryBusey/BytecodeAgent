export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @done, @init

@init
	%lt_slot =l alloc8 8
	%i_slot =l alloc8 8
	%gt_slot =l alloc8 8
	storel 0, %lt_slot
	storel 0, %i_slot
	storel %n, %gt_slot

	%half =l shr %n, 1
	%pivot_off =l shl %half, 4
	%pivot_ptr =l add %a, %pivot_off
	%pk =l loadl %pivot_ptr
	%pivot_val_ptr =l add %pivot_ptr, 8
	%pv =l loadl %pivot_val_ptr
	jmp @loop

@loop
	%i =l loadl %i_slot
	%gt =l loadl %gt_slot
	%more =w csltl %i, %gt
	jnz %more, @compare, @partitioned

@compare
	%i_off =l shl %i, 4
	%i_ptr =l add %a, %i_off
	%i_val_ptr =l add %i_ptr, 8
	%ik =l loadl %i_ptr
	%iv =l loadl %i_val_ptr
	%key_less =w csltl %ik, %pk
	jnz %key_less, @move_left, @check_key_greater

@check_key_greater
	%key_greater =w csgtl %ik, %pk
	jnz %key_greater, @move_right, @compare_value

@compare_value
	%value_less =w csltl %iv, %pv
	jnz %value_less, @move_left, @check_value_greater

@check_value_greater
	%value_greater =w csgtl %iv, %pv
	jnz %value_greater, @move_right, @advance

@move_left
	%lt =l loadl %lt_slot
	%lt_off =l shl %lt, 4
	%lt_ptr =l add %a, %lt_off
	%lt_val_ptr =l add %lt_ptr, 8
	%lk =l loadl %lt_ptr
	%lv =l loadl %lt_val_ptr
	storel %lk, %i_ptr
	storel %lv, %i_val_ptr
	storel %ik, %lt_ptr
	storel %iv, %lt_val_ptr
	%next_lt =l add %lt, 1
	storel %next_lt, %lt_slot
	jmp @advance

@move_right
	%next_gt =l sub %gt, 1
	storel %next_gt, %gt_slot
	%gt_off =l shl %next_gt, 4
	%gt_ptr =l add %a, %gt_off
	%gt_val_ptr =l add %gt_ptr, 8
	%gk =l loadl %gt_ptr
	%gv =l loadl %gt_val_ptr
	storel %gk, %i_ptr
	storel %gv, %i_val_ptr
	storel %ik, %gt_ptr
	storel %iv, %gt_val_ptr
	jmp @loop

@advance
	%next_i =l add %i, 1
	storel %next_i, %i_slot
	jmp @loop

@partitioned
	%left_n =l loadl %lt_slot
	%sort_left =w csgtl %left_n, 1
	jnz %sort_left, @call_left, @check_right

@call_left
	%left_result =l call $qsort_pairs(l %a, l %left_n)
	jmp @check_right

@check_right
	%right_start =l loadl %gt_slot
	%right_n =l sub %n, %right_start
	%sort_right =w csgtl %right_n, 1
	jnz %sort_right, @call_right, @done

@call_right
	%right_off =l shl %right_start, 4
	%right_ptr =l add %a, %right_off
	%right_result =l call $qsort_pairs(l %right_ptr, l %right_n)
	jmp @done

@done
	ret 0
}
