export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	%nplus =l add %n, %one
	jmp @initcheck
@initcheck
	%z =l phi @start %zero, @initbody %znext
	%initmore =w csltl %z, %nplus
	jnz %initmore, @initbody, @outercheck
@initbody
	%zoff =l shl %z, 3
	%zptr =l add %row, %zoff
	storel %zero, %zptr
	%znext =l add %z, %one
	jmp @initcheck
@outercheck
	%i =l phi @initcheck %zero, @afterinner %inext
	%morei =w csltl %i, %m
	jnz %morei, @outerbody, @done
@outerbody
	%xptr =l add %x, %i
	%xb =w loadub %xptr
	jmp @innercheck
@innercheck
	%j =l phi @outerbody %one, @store %jnext
	%diag =l phi @outerbody %zero, @store %up
	%morej =w csltl %j, %nplus
	jnz %morej, @innerbody, @afterinner
@innerbody
	%joff =l shl %j, 3
	%rptr =l add %row, %joff
	%up =l loadl %rptr
	%ym1 =l sub %j, %one
	%yptr =l add %y, %ym1
	%yb =w loadub %yptr
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @nomatch
@match
	%matched =l add %diag, %one
	jmp @store
@nomatch
	%leftptr =l sub %rptr, 8
	%left =l loadl %leftptr
	%upgreater =w csgtl %up, %left
	jnz %upgreater, @useup, @useleft
@useup
	jmp @store
@useleft
	jmp @store
@store
	%value =l phi @match %matched, @useup %up, @useleft %left
	storel %value, %rptr
	%jnext =l add %j, %one
	jmp @innercheck
@afterinner
	%inext =l add %i, %one
	jmp @outercheck
@done
	%noff =l shl %n, 3
	%lastptr =l add %row, %noff
	%result =l loadl %lastptr
	ret %result
}
