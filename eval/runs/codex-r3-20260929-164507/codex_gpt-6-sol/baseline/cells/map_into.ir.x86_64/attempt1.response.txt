export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @body %next_i
    %count =l phi @entry 0, @body %next_count
    %more =w csltl %i, %n
    jnz %more, @body, @done

@body
    %offset =l shl %i, 3
    %src_ptr =l add %src, %offset
    %dst_ptr =l add %dst, %offset
    %x =l loadl %src_ptr
    %r =l call %f(l %x)
    storel %r, %dst_ptr
    %less =w csltl %r, %x
    %increment =l extuw %less
    %next_count =l add %count, %increment
    %next_i =l add %i, 1
    jmp @loop

@done
    ret %count
}
