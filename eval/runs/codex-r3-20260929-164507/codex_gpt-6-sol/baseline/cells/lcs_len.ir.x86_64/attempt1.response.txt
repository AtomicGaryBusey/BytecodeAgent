export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %limit =l add %n, 1
    jmp @initcheck

@initcheck
    %z =l phi @start 0, @initbody %znext
    %moreinit =w csltl %z, %limit
    jnz %moreinit, @initbody, @initdone

@initbody
    %zoff =l shl %z, 3
    %zptr =l add %row, %zoff
    storel 0, %zptr
    %znext =l add %z, 1
    jmp @initcheck

@initdone
    jmp @outercheck

@outercheck
    %i =l phi @initdone 0, @outeradvance %inext
    %moreouter =w csltl %i, %m
    jnz %moreouter, @outerbody, @done

@outerbody
    %xptr =l add %x, %i
    %xb =w loadub %xptr
    jmp @innercheck

@innercheck
    %j =l phi @outerbody 1, @innerbody %jnext
    %diag =l phi @outerbody 0, @innerbody %up
    %moreinner =w csltl %j, %limit
    jnz %moreinner, @compare, @outeradvance

@compare
    %joff =l shl %j, 3
    %curptr =l add %row, %joff
    %up =l loadl %curptr
    %jm1 =l sub %j, 1
    %yptr =l add %y, %jm1
    %yb =w loadub %yptr
    %equal =w ceqw %xb, %yb
    jnz %equal, @match, @mismatch

@match
    %matched =l add %diag, 1
    jmp @innerbody

@mismatch
    %leftptr =l sub %curptr, 8
    %left =l loadl %leftptr
    %upgreater =w csgtl %up, %left
    jnz %upgreater, @takeup, @takeleft

@takeup
    jmp @innerbody

@takeleft
    jmp @innerbody

@innerbody
    %value =l phi @match %matched, @takeup %up, @takeleft %left
    storel %value, %curptr
    %jnext =l add %j, 1
    jmp @innercheck

@outeradvance
    %inext =l add %i, 1
    jmp @outercheck

@done
    %endoff =l shl %n, 3
    %endptr =l add %row, %endoff
    %result =l loadl %endptr
    ret %result
}
