lcs_len:
push rbx
cld
mov r9, rdi
mov r10, rdx
mov r11, rcx
mov rdi, r8
inc rcx
xor eax, eax
rep stosq
test rsi, rsi
jz .Ldone
test r11, r11
jz .Ldone
.Louter:
movzx edi, byte ptr [r9]
xor eax, eax
mov ecx, 1
.Linner:
mov rdx, qword ptr [r8 + rcx*8]
cmp dil, byte ptr [r10 + rcx - 1]
jne .Lno_match
inc rax
mov qword ptr [r8 + rcx*8], rax
mov rax, rdx
jmp .Ladvance
.Lno_match:
mov rbx, rdx
cmp rdx, qword ptr [r8 + rcx*8 - 8]
cmovl rdx, qword ptr [r8 + rcx*8 - 8]
mov qword ptr [r8 + rcx*8], rdx
mov rax, rbx
.Ladvance:
inc rcx
cmp rcx, r11
jle .Linner
inc r9
dec rsi
jnz .Louter
.Ldone:
mov rax, qword ptr [r8 + r11*8]
pop rbx
ret
