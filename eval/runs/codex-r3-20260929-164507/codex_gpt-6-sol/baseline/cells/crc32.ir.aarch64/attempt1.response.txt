export function l $crc32(l %p, l %n) {
@entry
    jmp @outer

@outer
    %i =l phi @entry 0, @next %inext
    %crc =w phi @entry -1, @next %crcnext
    %morebytes =w csltl %i, %n
    jnz %morebytes, @byte, @done

@byte
    %addr =l add %p, %i
    %b =w loadub %addr
    %initial =w xor %crc, %b
    jmp @bits

@bits
    %k =w phi @byte 0, @step %knext
    %v =w phi @byte %initial, @step %vnext
    %morebits =w csltw %k, 8
    jnz %morebits, @step, @next

@step
    %lsb =w and %v, 1
    %shifted =w shr %v, 1
    %mask =w sub 0, %lsb
    %poly =w and %mask, -306674912
    %vnext =w xor %shifted, %poly
    %knext =w add %k, 1
    jmp @bits

@next
    %crcnext =w copy %v
    %inext =l add %i, 1
    jmp @outer

@done
    %final =w xor %crc, -1
    %result =l extuw %final
    ret %result
}
