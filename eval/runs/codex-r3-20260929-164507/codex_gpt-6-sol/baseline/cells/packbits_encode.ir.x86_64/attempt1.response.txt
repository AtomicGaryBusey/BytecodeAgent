export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i0 =l copy 0
    %o0 =l copy 0
    jmp @loop

@loop
    %i =l phi @start %i0, @rep_done %repnext, @lit_done %j
    %o =l phi @start %o0, @rep_done %rep_o, @lit_done %lit_o
    %end =w csgel %i, %n
    jnz %end, @success, @scan_init

@success
    ret %o

@scan_init
    %src_i =l add %src, %i
    %base =w loadub %src_i
    %left =l sub %n, %i
    %run0 =l copy 1
    jmp @scan_loop

@scan_loop
    %run =l phi @scan_init %run0, @scan_inc %run_next
    %under128 =w csltl %run, 128
    %underleft =w csltl %run, %left
    %scan_more =w and %under128, %underleft
    jnz %scan_more, @scan_cmp, @scan_done

@scan_cmp
    %scan_pos =l add %src_i, %run
    %scan_byte =w loadub %scan_pos
    %same =w ceqw %scan_byte, %base
    jnz %same, @scan_inc, @scan_done

@scan_inc
    %run_next =l add %run, 1
    jmp @scan_loop

@scan_done
    %short =w csltl %run, 3
    jnz %short, @lit_init, @rep_check

@rep_check
    %rep_avail =l sub %cap, %o
    %rep_fits =w csgel %rep_avail, 2
    jnz %rep_fits, @rep_write, @failure

@rep_write
    %rep_header =l sub 257, %run
    %out_o =l add %out, %o
    storeb %rep_header, %out_o
    %rep_byte_pos =l add %out_o, 1
    storeb %base, %rep_byte_pos
    jmp @rep_done

@rep_done
    %repnext =l add %i, %run
    %rep_o =l add %o, 2
    jmp @loop

@lit_init
    %j0 =l add %i, 1
    jmp @lit_loop

@lit_loop
    %j =l phi @lit_init %j0, @lit_inc %jnext
    %lit_len =l sub %j, %i
    %before_end =w csltl %j, %n
    %below128 =w csltl %lit_len, 128
    %lit_more =w and %before_end, %below128
    jnz %lit_more, @lit_run_init, @lit_emit

@lit_run_init
    %src_j =l add %src, %j
    %lit_base =w loadub %src_j
    %lit_left =l sub %n, %j
    %lr0 =l copy 1
    jmp @lit_run_loop

@lit_run_loop
    %lr =l phi @lit_run_init %lr0, @lit_run_inc %lrnext
    %lr_under128 =w csltl %lr, 128
    %lr_underleft =w csltl %lr, %lit_left
    %lr_more =w and %lr_under128, %lr_underleft
    jnz %lr_more, @lit_run_cmp, @lit_run_end

@lit_run_cmp
    %lr_pos =l add %src_j, %lr
    %lr_byte =w loadub %lr_pos
    %lr_same =w ceqw %lr_byte, %lit_base
    jnz %lr_same, @lit_run_inc, @lit_run_end

@lit_run_inc
    %lrnext =l add %lr, 1
    jmp @lit_run_loop

@lit_run_end
    %lr_short =w csltl %lr, 3
    jnz %lr_short, @lit_inc, @lit_emit

@lit_inc
    %jnext =l add %j, 1
    jmp @lit_loop

@lit_emit
    %len =l sub %j, %i
    %need =l add %len, 1
    %avail =l sub %cap, %o
    %fits =w cslel %need, %avail
    jnz %fits, @lit_write, @failure

@lit_write
    %header =l sub %len, 1
    %lit_out =l add %out, %o
    storeb %header, %lit_out
    %k0 =l copy 0
    jmp @copy_loop

@copy_loop
    %k =l phi @lit_write %k0, @copy_step %knext
    %copy_more =w csltl %k, %len
    jnz %copy_more, @copy_step, @lit_done

@copy_step
    %src_offset =l add %i, %k
    %copy_src =l add %src, %src_offset
    %copy_byte =w loadub %copy_src
    %out_offset =l add %o, 1
    %out_offset_k =l add %out_offset, %k
    %copy_out =l add %out, %out_offset_k
    storeb %copy_byte, %copy_out
    %knext =l add %k, 1
    jmp @copy_loop

@lit_done
    %lit_o =l add %o, %need
    jmp @loop

@failure
    ret -1
}
