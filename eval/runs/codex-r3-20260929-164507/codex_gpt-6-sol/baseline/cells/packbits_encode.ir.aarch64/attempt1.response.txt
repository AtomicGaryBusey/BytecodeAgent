export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    %ip =l alloc8 8
    %op =l alloc8 8
    %rp =l alloc8 8
    %jp =l alloc8 8
    %kp =l alloc8 8
    storel 0, %ip
    storel 0, %op
    jmp @outer

@outer
    %i =l loadl %ip
    %done =w csgel %i, %n
    jnz %done, @success, @runinit

@runinit
    %basep =l add %src, %i
    %base =w loadub %basep
    storel 1, %rp
    jmp @runcheck

@runcheck
    %r =l loadl %rp
    %at =l add %i, %r
    %pastend =w csgel %at, %n
    jnz %pastend, @rundone, @runlimit

@runlimit
    %fullrun =w csgel %r, 128
    jnz %fullrun, @rundone, @runcompare

@runcompare
    %nextp =l add %src, %at
    %next =w loadub %nextp
    %equal =w ceqw %next, %base
    jnz %equal, @runextend, @rundone

@runextend
    %rnext =l add %r, 1
    storel %rnext, %rp
    jmp @runcheck

@rundone
    %rr =l loadl %rp
    %repeat =w csgel %rr, 3
    jnz %repeat, @repeatpacket, @literalinit

@repeatpacket
    %ro =l loadl %op
    %rend =l add %ro, 2
    %rno =w csgtl %rend, %cap
    jnz %rno, @failure, @repeatwrite

@repeatwrite
    %rval =l loadl %rp
    %r32 =w copy %rval
    %rhead =w sub 257, %r32
    %rout =l add %out, %ro
    storeb %rhead, %rout
    %rbyteout =l add %rout, 1
    storeb %base, %rbyteout
    storel %rend, %op
    %ri =l loadl %ip
    %rinext =l add %ri, %rval
    storel %rinext, %ip
    jmp @outer

@literalinit
    %li =l loadl %ip
    storel %li, %jp
    jmp @literalcheck

@literalcheck
    %j =l loadl %jp
    %lend =w csgel %j, %n
    jnz %lend, @literalpacket, @literallimit

@literallimit
    %start =l loadl %ip
    %distance =l sub %j, %start
    %lfull =w csgel %distance, 128
    jnz %lfull, @literalpacket, @literalruninit

@literalruninit
    %lbasep =l add %src, %j
    %lbase =w loadub %lbasep
    storel 1, %rp
    jmp @literalruncheck

@literalruncheck
    %lr =l loadl %rp
    %lat =l add %j, %lr
    %lpastend =w csgel %lat, %n
    jnz %lpastend, @literalrundone, @literalrunlimit

@literalrunlimit
    %lmax =w csgel %lr, 128
    jnz %lmax, @literalrundone, @literalruncompare

@literalruncompare
    %lnextp =l add %src, %lat
    %lnext =w loadub %lnextp
    %lequal =w ceqw %lnext, %lbase
    jnz %lequal, @literalrunextend, @literalrundone

@literalrunextend
    %lrnext =l add %lr, 1
    storel %lrnext, %rp
    jmp @literalruncheck

@literalrundone
    %lrun =l loadl %rp
    %lrepeat =w csgel %lrun, 3
    jnz %lrepeat, @literalpacket, @literaladvance

@literaladvance
    %jnext =l add %j, 1
    storel %jnext, %jp
    jmp @literalcheck

@literalpacket
    %litstart =l loadl %ip
    %litend =l loadl %jp
    %len =l sub %litend, %litstart
    %lo =l loadl %op
    %needed =l add %len, 1
    %loend =l add %lo, %needed
    %lno =w csgtl %loend, %cap
    jnz %lno, @failure, @literalwrite

@literalwrite
    %len32 =w copy %len
    %lhead =w sub %len32, 1
    %lout =l add %out, %lo
    storeb %lhead, %lout
    storel 0, %kp
    jmp @copycheck

@copycheck
    %k =l loadl %kp
    %copied =w csgel %k, %len
    jnz %copied, @literalfinish, @copybyte

@copybyte
    %sourceindex =l add %litstart, %k
    %sourcep =l add %src, %sourceindex
    %byte =w loadub %sourcep
    %destindex =l add %lo, %k
    %destoffset =l add %destindex, 1
    %destp =l add %out, %destoffset
    storeb %byte, %destp
    %knext =l add %k, 1
    storel %knext, %kp
    jmp @copycheck

@literalfinish
    storel %litend, %ip
    storel %loend, %op
    jmp @outer

@success
    %result =l loadl %op
    ret %result

@failure
    ret -1
}
