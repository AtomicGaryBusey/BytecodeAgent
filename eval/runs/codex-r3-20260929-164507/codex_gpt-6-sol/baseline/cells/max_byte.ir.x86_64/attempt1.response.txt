export function l $max_byte(l %p, l %n) {
@start
    %has =w csgtl %n, 0
    jnz %has, @loop, @empty

@loop
    %i =l phi @start 0, @advance %next_i
    %max =l phi @start 0, @advance %next_max
    %addr =l add %p, %i
    %byte =l loadub %addr
    %greater =w cugtl %byte, %max
    jnz %greater, @larger, @same

@larger
    jmp @advance

@same
    jmp @advance

@advance
    %next_max =l phi @larger %byte, @same %max
    %next_i =l add %i, 1
    %more =w csltl %next_i, %n
    jnz %more, @loop, @done

@done
    ret %next_max

@empty
    ret 0
}
