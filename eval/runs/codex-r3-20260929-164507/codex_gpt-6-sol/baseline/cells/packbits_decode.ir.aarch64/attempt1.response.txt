export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %is =l alloc8 8
    %os =l alloc8 8
    %lefts =l alloc8 8
    storel 0, %is
    storel 0, %os
    jmp @packet

@packet
    %i =l loadl %is
    %end =w ceql %i, %n
    jnz %end, @done, @header

@header
    %hp =l add %src, %i
    %h =w loadsb %hp
    %next =l add %i, 1
    storel %next, %is
    %noop =w ceqw %h, -128
    jnz %noop, @packet, @classify

@classify
    %negative =w csltw %h, 0
    jnz %negative, @repeat, @literal

@literal
    %hs =l extsw %h
    %count =l add %hs, 1
    %li =l loadl %is
    %remaining =l sub %n, %li
    %short =w csltl %remaining, %count
    jnz %short, @source_error, @literal_ready

@literal_ready
    storel %count, %lefts
    jmp @literal_loop

@literal_loop
    %ll =l loadl %lefts
    %ldone =w ceql %ll, 0
    jnz %ldone, @packet, @literal_space

@literal_space
    %lo =l loadl %os
    %lfull =w ceql %lo, %cap
    jnz %lfull, @overflow, @literal_write

@literal_write
    %ls =l loadl %is
    %lp =l add %src, %ls
    %lb =w loadub %lp
    %lq =l add %out, %lo
    storeb %lb, %lq
    %lsnext =l add %ls, 1
    %lonext =l add %lo, 1
    %llnext =l sub %ll, 1
    storel %lsnext, %is
    storel %lonext, %os
    storel %llnext, %lefts
    jmp @literal_loop

@repeat
    %ri =l loadl %is
    %missing =w ceql %ri, %n
    jnz %missing, @source_error, @repeat_ready

@repeat_ready
    %rp =l add %src, %ri
    %rb =w loadub %rp
    %rinext =l add %ri, 1
    storel %rinext, %is
    %rhs =l extsw %h
    %rcount =l sub 1, %rhs
    storel %rcount, %lefts
    jmp @repeat_loop

@repeat_loop
    %rl =l loadl %lefts
    %rdone =w ceql %rl, 0
    jnz %rdone, @packet, @repeat_space

@repeat_space
    %ro =l loadl %os
    %rfull =w ceql %ro, %cap
    jnz %rfull, @overflow, @repeat_write

@repeat_write
    %rq =l add %out, %ro
    storeb %rb, %rq
    %ronext =l add %ro, 1
    %rlnext =l sub %rl, 1
    storel %ronext, %os
    storel %rlnext, %lefts
    jmp @repeat_loop

@done
    %result =l loadl %os
    ret %result

@source_error
    ret -2

@overflow
    ret -1
}
