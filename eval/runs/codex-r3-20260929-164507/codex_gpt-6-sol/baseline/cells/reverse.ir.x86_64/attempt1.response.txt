export function l $reverse(l %a, l %n) {
@start
    %last =l sub %n, 1
    jmp @loop
@loop
    %i =l phi @start 0, @swap %nexti
    %j =l phi @start %last, @swap %nextj
    %more =w csltl %i, %j
    jnz %more, @swap, @done
@swap
    %ioff =l shl %i, 3
    %joff =l shl %j, 3
    %iptr =l add %a, %ioff
    %jptr =l add %a, %joff
    %iv =l loadl %iptr
    %jv =l loadl %jptr
    storel %jv, %iptr
    storel %iv, %jptr
    %nexti =l add %i, 1
    %nextj =l sub %j, 1
    jmp @loop
@done
    ret 0
}
