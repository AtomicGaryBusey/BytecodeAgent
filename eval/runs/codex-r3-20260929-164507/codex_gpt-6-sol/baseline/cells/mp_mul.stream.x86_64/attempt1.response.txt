mp_mul:
push rbx
push r12
mov rbx, rdx
lea r10, [rsi + rcx]
xor r11d, r11d
.Lzero:
cmp r11, r10
jae .Louter_init
mov qword ptr [r8 + r11*8], 0
inc r11
jmp .Lzero
.Louter_init:
xor r10d, r10d
.Louter:
cmp r10, rsi
jae .Ltrim_init
xor r9d, r9d
xor r11d, r11d
.Linner:
cmp r11, rcx
jae .Lend_inner
mov rax, qword ptr [rdi + r10*8]
mul qword ptr [rbx + r11*8]
lea r12, [r10 + r11]
add rax, qword ptr [r8 + r12*8]
adc rdx, 0
add rax, r9
adc rdx, 0
mov qword ptr [r8 + r12*8], rax
mov r9, rdx
inc r11
jmp .Linner
.Lend_inner:
lea r12, [r10 + rcx]
mov qword ptr [r8 + r12*8], r9
inc r10
jmp .Louter
.Ltrim_init:
lea rax, [rsi + rcx]
test rax, rax
jz .Ldone
.Ltrim:
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jne .Ltrim
.Ldone:
pop r12
pop rbx
ret
