export function l $reverse(l %a, l %n) {
@start
	%last =l sub %n, 1
@loop
	%i =l phi @start 0, @swap %next
	%j =l phi @start %last, @swap %prev
	%more =w csltl %i, %j
	jnz %more, @swap, @done
@swap
	%loff =l shl %i, 3
	%roff =l shl %j, 3
	%left =l add %a, %loff
	%right =l add %a, %roff
	%lv =l loadl %left
	%rv =l loadl %right
	storel %rv, %left
	storel %lv, %right
	%next =l add %i, 1
	%prev =l sub %j, 1
	jmp @loop
@done
	ret 0
}
