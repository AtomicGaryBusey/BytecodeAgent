export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop

@loop
	%i =l phi @entry 1, @advance %next
	%min =l phi @entry %y0, @advance %minnext
	%max =l phi @entry %y0, @advance %maxnext
	%sum =l phi @entry %y0, @advance %sumnext
	%odd =l phi @entry %odd0, @advance %oddnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%less =w csltl %y, %min
	jnz %less, @take_min, @keep_min

@take_min
	jmp @min_join

@keep_min
	jmp @min_join

@min_join
	%minnext =l phi @take_min %y, @keep_min %min
	%greater =w csgtl %y, %max
	jnz %greater, @take_max, @keep_max

@take_max
	jmp @max_join

@keep_max
	jmp @max_join

@max_join
	%maxnext =l phi @take_max %y, @keep_max %max
	jmp @advance

@advance
	%sumnext =l add %sum, %y
	%bit =l and %y, 1
	%oddnext =l add %odd, %bit
	%next =l add %i, 1
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
