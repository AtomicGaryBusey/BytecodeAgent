export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    jmp @loop
@loop
    %i =l phi @start 1, @body %next
    %max =l phi @start %first, @body %updated
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w csltl %max, %value
    %updated =l sel %greater, %value, %max
    %next =l add %i, 1
    jmp @loop
@done
    ret %max
}
