export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%lo00 =l shr %p00, 32
	%lo01 =l and %p01, 4294967295
	%lo10 =l and %p10, 4294967295
	%mid0 =l add %lo00, %lo01
	%mid =l add %mid0, %lo10
	%carry =l shr %mid, 32

	%hi01 =l shr %p01, 32
	%hi10 =l shr %p10, 32
	%sum0 =l add %p11, %hi01
	%sum1 =l add %sum0, %hi10
	%result =l add %sum1, %carry
	ret %result
}
