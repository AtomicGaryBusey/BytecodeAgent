export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @direction

@direction
	%forward =w cultl %dst, %src
	jnz %forward, @floop, @bloop

@floop
	%fi =l phi @direction 0, @fbody %fi_next
	%fmore =w csltl %fi, %n
	jnz %fmore, @fbody, @done

@fbody
	%foffset =l mul %fi, 8
	%fsource =l add %src, %foffset
	%ftarget =l add %dst, %foffset
	%fvalue =l loadl %fsource
	storel %fvalue, %ftarget
	%fi_next =l add %fi, 1
	jmp @floop

@bloop
	%bi =l phi @direction %n, @bbody %bi_next
	%bmore =w csgtl %bi, 0
	jnz %bmore, @bbody, @done

@bbody
	%bi_next =l sub %bi, 1
	%boffset =l mul %bi_next, 8
	%bsource =l add %src, %boffset
	%btarget =l add %dst, %boffset
	%bvalue =l loadl %bsource
	storel %bvalue, %btarget
	jmp @bloop

@done
	ret 0
}
