export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    %pcslot =l alloc8 8
    %dslot =l alloc8 8
    %resultslot =l alloc8 8
    %idxslot =l alloc8 8
    storel 0, %pcslot
    storel 0, %dslot
    storel 0, %resultslot
    jmp @loop

@loop
    %pc =l loadl %pcslot
    %atend =w ceql %pc, %len
    jnz %atend, @finish, @fetch

@fetch
    %codeoff =l shl %pc, 3
    %codeaddr =l add %code, %codeoff
    %op =l loadl %codeaddr
    %is0 =w ceql %op, 0
    jnz %is0, @finish, @test1

@test1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @test2
@test2
    %is2 =w ceql %op, 2
    jnz %is2, @binary, @test3
@test3
    %is3 =w ceql %op, 3
    jnz %is3, @binary, @test4
@test4
    %is4 =w ceql %op, 4
    jnz %is4, @binary, @test5
@test5
    %is5 =w ceql %op, 5
    jnz %is5, @dup, @test6
@test6
    %is6 =w ceql %op, 6
    jnz %is6, @swap, @test7
@test7
    %is7 =w ceql %op, 7
    jnz %is7, @drop, @test8
@test8
    %is8 =w ceql %op, 8
    jnz %is8, @over, @test9
@test9
    %is9 =w ceql %op, 9
    jnz %is9, @binary, @test10
@test10
    %is10 =w ceql %op, 10
    jnz %is10, @jz, @test11
@test11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp, @step1

@push
    %pushnext =l add %pc, 1
    %pushoff =l shl %pushnext, 3
    %pushaddr =l add %code, %pushoff
    %pushval =l loadl %pushaddr
    %pushd =l loadl %dslot
    %pushstackoff =l shl %pushd, 3
    %pushstackaddr =l add %stack, %pushstackoff
    storel %pushval, %pushstackaddr
    %pushnewd =l add %pushd, 1
    storel %pushnewd, %dslot
    jmp @step2

@binary
    %bind =l loadl %dslot
    %bindex =l sub %bind, 1
    %aindex =l sub %bind, 2
    %boff =l shl %bindex, 3
    %aoff =l shl %aindex, 3
    %baddr =l add %stack, %boff
    %aaddr =l add %stack, %aoff
    %b =l loadl %baddr
    %a =l loadl %aaddr
    %bisadd =w ceql %op, 2
    jnz %bisadd, @doadd, @checksub

@checksub
    %bissub =w ceql %op, 3
    jnz %bissub, @dosub, @checkmul
@checkmul
    %bismul =w ceql %op, 4
    jnz %bismul, @domul, @dolt
@doadd
    %addval =l add %a, %b
    storel %addval, %aaddr
    jmp @bindone
@dosub
    %subval =l sub %a, %b
    storel %subval, %aaddr
    jmp @bindone
@domul
    %mulval =l mul %a, %b
    storel %mulval, %aaddr
    jmp @bindone
@dolt
    %ltw =w csltl %a, %b
    %ltval =l extuw %ltw
    storel %ltval, %aaddr
    jmp @bindone
@bindone
    storel %bindex, %dslot
    jmp @step1

@dup
    %dupd =l loadl %dslot
    %duptop =l sub %dupd, 1
    %duptopoff =l shl %duptop, 3
    %duptopaddr =l add %stack, %duptopoff
    %dupval =l loadl %duptopaddr
    %dupdestoff =l shl %dupd, 3
    %dupdest =l add %stack, %dupdestoff
    storel %dupval, %dupdest
    %dupnewd =l add %dupd, 1
    storel %dupnewd, %dslot
    jmp @step1

@swap
    %swapd =l loadl %dslot
    %swapbi =l sub %swapd, 1
    %swapai =l sub %swapd, 2
    %swapboff =l shl %swapbi, 3
    %swapaoff =l shl %swapai, 3
    %swapba =l add %stack, %swapboff
    %swapaa =l add %stack, %swapaoff
    %swapb =l loadl %swapba
    %swapa =l loadl %swapaa
    storel %swapa, %swapba
    storel %swapb, %swapaa
    jmp @step1

@drop
    %dropd =l loadl %dslot
    %dropnewd =l sub %dropd, 1
    storel %dropnewd, %dslot
    jmp @step1

@over
    %overd =l loadl %dslot
    %oversrcidx =l sub %overd, 2
    %oversrcoff =l shl %oversrcidx, 3
    %oversrc =l add %stack, %oversrcoff
    %overval =l loadl %oversrc
    %overdestoff =l shl %overd, 3
    %overdest =l add %stack, %overdestoff
    storel %overval, %overdest
    %overnewd =l add %overd, 1
    storel %overnewd, %dslot
    jmp @step1

@jz
    %jzd =l loadl %dslot
    %jznewd =l sub %jzd, 1
    storel %jznewd, %dslot
    %jzoff =l shl %jznewd, 3
    %jzaddr =l add %stack, %jzoff
    %jzval =l loadl %jzaddr
    %jzzero =w ceql %jzval, 0
    jnz %jzzero, @jztaken, @step2

@jztaken
    %jzoperandpc =l add %pc, 1
    %jzoperandoff =l shl %jzoperandpc, 3
    %jzoperandaddr =l add %code, %jzoperandoff
    %jztarget =l loadl %jzoperandaddr
    storel %jztarget, %pcslot
    jmp @loop

@jmp
    %jmpoperandpc =l add %pc, 1
    %jmpoperandoff =l shl %jmpoperandpc, 3
    %jmpoperandaddr =l add %code, %jmpoperandoff
    %jmptarget =l loadl %jmpoperandaddr
    storel %jmptarget, %pcslot
    jmp @loop

@step1
    %next1 =l add %pc, 1
    storel %next1, %pcslot
    jmp @loop

@step2
    %next2 =l add %pc, 2
    storel %next2, %pcslot
    jmp @loop

@finish
    %finald =l loadl %dslot
    %nonempty =w cnel %finald, 0
    jnz %nonempty, @getresult, @clearinit

@getresult
    %topidx =l sub %finald, 1
    %topoff =l shl %topidx, 3
    %topaddr =l add %stack, %topoff
    %result =l loadl %topaddr
    storel %result, %resultslot
    jmp @clearinit

@clearinit
    storel %finald, %idxslot
    jmp @clearloop

@clearloop
    %idx =l loadl %idxslot
    %more =w csltl %idx, %cap
    jnz %more, @clearbody, @return

@clearbody
    %clearoff =l shl %idx, 3
    %clearaddr =l add %stack, %clearoff
    storel 0, %clearaddr
    %nextidx =l add %idx, 1
    storel %nextidx, %idxslot
    jmp @clearloop

@return
    %answer =l loadl %resultslot
    ret %answer
}
