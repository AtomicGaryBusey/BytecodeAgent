export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%count =l mul %n, %n
	%scratch_count =l mul %count, 2
	jmp @init_loop

@init_loop
	%ii =l phi @start 0, @init_next %ii_next
	%init_more =w csltl %ii, %count
	jnz %init_more, @init_body, @power_check

@init_body
	%init_row =l div %ii, %n
	%init_col =l rem %ii, %n
	%init_diag =w ceql %init_row, %init_col
	%init_offset =l mul %ii, 8
	%init_ptr =l add %out, %init_offset
	jnz %init_diag, @init_one, @init_zero

@init_one
	storel 1, %init_ptr
	jmp @init_next

@init_zero
	storel 0, %init_ptr
	jmp @init_next

@init_next
	%ii_next =l add %ii, 1
	jmp @init_loop

@power_check
	%step =l phi @init_loop 0, @power_next %step_next
	%power_more =w csltl %step, %k
	jnz %power_more, @product_loop, @zero_loop

@product_loop
	%idx =l phi @power_check 0, @product_next %idx_next
	%product_more =w csltl %idx, %count
	jnz %product_more, @product_body, @copy_loop

@product_body
	%row =l div %idx, %n
	%col =l rem %idx, %n
	%row_base =l mul %row, %n
	jmp @sum_loop

@sum_loop
	%t =l phi @product_body 0, @sum_body %t_next
	%acc =l phi @product_body 0, @sum_body %acc_next
	%sum_more =w csltl %t, %n
	jnz %sum_more, @sum_body, @sum_done

@sum_body
	%left_idx =l add %row_base, %t
	%left_offset =l mul %left_idx, 8
	%left_ptr =l add %out, %left_offset
	%left =l loadl %left_ptr
	%right_row =l mul %t, %n
	%right_idx =l add %right_row, %col
	%right_offset =l mul %right_idx, 8
	%right_ptr =l add %a, %right_offset
	%right =l loadl %right_ptr
	%term =l mul %left, %right
	%acc_next =l add %acc, %term
	%t_next =l add %t, 1
	jmp @sum_loop

@sum_done
	%product_offset =l mul %idx, 8
	%product_ptr =l add %tmp, %product_offset
	storel %acc, %product_ptr
	jmp @product_next

@product_next
	%idx_next =l add %idx, 1
	jmp @product_loop

@copy_loop
	%copy_idx =l phi @product_loop 0, @copy_body %copy_idx_next
	%copy_more =w csltl %copy_idx, %count
	jnz %copy_more, @copy_body, @power_next

@copy_body
	%copy_offset =l mul %copy_idx, 8
	%copy_src =l add %tmp, %copy_offset
	%copy_dst =l add %out, %copy_offset
	%copy_value =l loadl %copy_src
	storel %copy_value, %copy_dst
	%copy_idx_next =l add %copy_idx, 1
	jmp @copy_loop

@power_next
	%step_next =l add %step, 1
	jmp @power_check

@zero_loop
	%zero_idx =l phi @power_check 0, @zero_body %zero_idx_next
	%zero_more =w csltl %zero_idx, %scratch_count
	jnz %zero_more, @zero_body, @done

@zero_body
	%zero_offset =l mul %zero_idx, 8
	%zero_ptr =l add %tmp, %zero_offset
	storel 0, %zero_ptr
	%zero_idx_next =l add %zero_idx, 1
	jmp @zero_loop

@done
	ret 0
}
