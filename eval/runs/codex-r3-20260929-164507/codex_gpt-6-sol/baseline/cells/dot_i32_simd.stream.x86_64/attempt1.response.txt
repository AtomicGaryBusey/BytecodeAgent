dot_i32_simd:
xor eax, eax
xor ecx, ecx
pxor xmm0, xmm0
mov r8, rdx
and r8, -4
test r8, r8
jz .Lscalar
.Lvector:
movdqu xmm1, xmmword ptr [rdi + rcx*4]
movdqu xmm2, xmmword ptr [rsi + rcx*4]
movdqa xmm3, xmm1
movdqa xmm4, xmm2
pmuldq xmm1, xmm2
psrldq xmm3, 4
psrldq xmm4, 4
pmuldq xmm3, xmm4
paddq xmm0, xmm1
paddq xmm0, xmm3
add rcx, 4
cmp rcx, r8
jb .Lvector
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
.Lscalar:
cmp rcx, rdx
jae .Ldone
.Ltail:
movsxd r8, dword ptr [rdi + rcx*4]
movsxd r9, dword ptr [rsi + rcx*4]
imul r8, r9
add rax, r8
inc rcx
cmp rcx, rdx
jb .Ltail
.Ldone:
ret
