export function l $isqrt(l %n) {
@start
    jmp @loop

@loop
    %rem =l phi @start %n, @next %remnext
    %root =l phi @start 0, @next %rootnext
    %bit =l phi @start 4611686018427387904, @next %bitnext
    %candidate =l add %root, %bit
    %fits =w cugel %rem, %candidate
    jnz %fits, @take, @skip

@take
    %takenrem =l sub %rem, %candidate
    %halfroot =l shr %root, 1
    %takenroot =l add %halfroot, %bit
    jmp @next

@skip
    %skippedroot =l shr %root, 1
    jmp @next

@next
    %remnext =l phi @take %takenrem, @skip %rem
    %rootnext =l phi @take %takenroot, @skip %skippedroot
    %bitnext =l shr %bit, 2
    %more =w cnel %bitnext, 0
    jnz %more, @loop, @done

@done
    ret %rootnext
}
