export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcslot =l alloc8 8
	%dslot =l alloc8 8
	%rslot =l alloc8 8
	storel 0, %pcslot
	storel 0, %dslot
	storel 0, %rslot
	jmp @loop

@loop
	%pc =l loadl %pcslot
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch

@fetch
	%coff =l mul %pc, 8
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @check1

@check1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @check2
@check2
	%is2 =w ceql %op, 2
	jnz %is2, @binary, @check3
@check3
	%is3 =w ceql %op, 3
	jnz %is3, @binary, @check4
@check4
	%is4 =w ceql %op, 4
	jnz %is4, @binary, @check5
@check5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @check6
@check6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @check7
@check7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @check8
@check8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @check9
@check9
	%is9 =w ceql %op, 9
	jnz %is9, @binary, @check10
@check10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @check11
@check11
	%is11 =w ceql %op, 11
	jnz %is11, @jmpop, @step1

@push
	%pnext =l add %caddr, 8
	%value =l loadl %pnext
	%pd =l loadl %dslot
	%poff =l mul %pd, 8
	%pdest =l add %stack, %poff
	storel %value, %pdest
	%pdnew =l add %pd, 1
	storel %pdnew, %dslot
	jmp @step2

@binary
	%bd =l loadl %dslot
	%bi =l sub %bd, 1
	%ai =l sub %bd, 2
	%boff =l mul %bi, 8
	%aoff =l mul %ai, 8
	%baddr =l add %stack, %boff
	%aaddr =l add %stack, %aoff
	%b =l loadl %baddr
	%a =l loadl %aaddr
	%bisadd =w ceql %op, 2
	jnz %bisadd, @addop, @binsubcheck

@binsubcheck
	%bissub =w ceql %op, 3
	jnz %bissub, @subop, @binmulcheck
@binmulcheck
	%bismul =w ceql %op, 4
	jnz %bismul, @mulop, @ltop
@addop
	%sum =l add %a, %b
	jmp @binwrite
@subop
	%diff =l sub %a, %b
	jmp @binwrite
@mulop
	%product =l mul %a, %b
	jmp @binwrite
@ltop
	%lt =w csltl %a, %b
	%ltval =l extuw %lt
	jmp @binwrite
@binwrite
	%result =l phi @addop %sum, @subop %diff, @mulop %product, @ltop %ltval
	storel %result, %aaddr
	storel %bi, %dslot
	jmp @step1

@dup
	%dd =l loadl %dslot
	%di =l sub %dd, 1
	%doff =l mul %di, 8
	%dsr =l add %stack, %doff
	%dval =l loadl %dsr
	%ddoff =l mul %dd, 8
	%ddst =l add %stack, %ddoff
	storel %dval, %ddst
	%ddnew =l add %dd, 1
	storel %ddnew, %dslot
	jmp @step1

@swap
	%sd =l loadl %dslot
	%si =l sub %sd, 1
	%sj =l sub %sd, 2
	%sioff =l mul %si, 8
	%sjoff =l mul %sj, 8
	%siaddr =l add %stack, %sioff
	%sjaddr =l add %stack, %sjoff
	%siv =l loadl %siaddr
	%sjv =l loadl %sjaddr
	storel %sjv, %siaddr
	storel %siv, %sjaddr
	jmp @step1

@drop
	%drd =l loadl %dslot
	%drnew =l sub %drd, 1
	storel %drnew, %dslot
	jmp @step1

@over
	%od =l loadl %dslot
	%oi =l sub %od, 2
	%oioff =l mul %oi, 8
	%osrc =l add %stack, %oioff
	%oval =l loadl %osrc
	%odoff =l mul %od, 8
	%odst =l add %stack, %odoff
	storel %oval, %odst
	%odnew =l add %od, 1
	storel %odnew, %dslot
	jmp @step1

@jz
	%ztaddr =l add %caddr, 8
	%target =l loadl %ztaddr
	%zd =l loadl %dslot
	%zdnew =l sub %zd, 1
	%zoff =l mul %zdnew, 8
	%zaddr =l add %stack, %zoff
	%zval =l loadl %zaddr
	storel %zdnew, %dslot
	%zero =w ceql %zval, 0
	jnz %zero, @jztaken, @step2
@jztaken
	storel %target, %pcslot
	jmp @loop

@jmpop
	%jtaddr =l add %caddr, 8
	%jtarget =l loadl %jtaddr
	storel %jtarget, %pcslot
	jmp @loop

@step1
	%pc1 =l add %pc, 1
	storel %pc1, %pcslot
	jmp @loop

@step2
	%pc2 =l add %pc, 2
	storel %pc2, %pcslot
	jmp @loop

@finish
	%finald =l loadl %dslot
	%has =w cnel %finald, 0
	jnz %has, @top, @fillinit
@top
	%topi =l sub %finald, 1
	%topoff =l mul %topi, 8
	%topaddr =l add %stack, %topoff
	%topval =l loadl %topaddr
	storel %topval, %rslot
	jmp @fillinit

@fillinit
	jmp @fillloop
@fillloop
	%i =l phi @fillinit %finald, @fillbody %next
	%more =w csltl %i, %cap
	jnz %more, @fillbody, @return
@fillbody
	%off =l mul %i, 8
	%addr =l add %stack, %off
	storel 0, %addr
	%next =l add %i, 1
	jmp @fillloop

@return
	%answer =l loadl %rslot
	ret %answer
}
