export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @none
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%mid =l add %lo, %h
	%off =l shl %mid, 3
	%p =l add %a, %off
	%v =l loadl %p
	%e =w ceql %v, %key
	jnz %e, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%lo1 =l add %mid, 1
	%lo =l copy %lo1
	jmp @loop
@left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@none
	ret -1
}
