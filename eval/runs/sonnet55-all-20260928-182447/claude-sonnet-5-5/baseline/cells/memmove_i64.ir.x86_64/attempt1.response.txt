export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @fwd, @chk
@chk
	%eq =w ceql %dst, %src
	jnz %eq, @done, @bwd
@fwd
	%i =l copy 0
	jmp @fh
@fh
	%i1 =l phi @fwd %i, @fb %i2
	%c =w csltl %i1, %n
	jnz %c, @fb, @done
@fb
	%o =l shl %i1, 3
	%s =l add %src, %o
	%d =l add %dst, %o
	%v =l loadl %s
	storel %v, %d
	%i2 =l add %i1, 1
	jmp @fh
@bwd
	jmp @bh
@bh
	%j1 =l phi @bwd %n, @bb %j2
	%c2 =w csltl 0, %j1
	jnz %c2, @bb, @done
@bb
	%j2 =l sub %j1, 1
	%o2 =l shl %j2, 3
	%s2 =l add %src, %o2
	%d2 =l add %dst, %o2
	%v2 =l loadl %s2
	storel %v2, %d2
	jmp @bh
@done
	ret 0
}
