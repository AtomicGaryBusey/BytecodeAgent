isqrt:
    mov x1, #0
    mov x2, #1
    lsl x2, x2, #62
.Lshrink:
    cbz x2, .Ldone
    cmp x2, x0
    b.ls .Lloop
    lsr x2, x2, #2
    b .Lshrink
.Lloop:
    cbz x2, .Ldone
    add x3, x1, x2
    cmp x0, x3
    b.lo .Lelse
    sub x0, x0, x3
    lsr x1, x1, #1
    add x1, x1, x2
    b .Lnext
.Lelse:
    lsr x1, x1, #1
.Lnext:
    lsr x2, x2, #2
    b .Lloop
.Ldone:
    mov x0, x1
    ret
