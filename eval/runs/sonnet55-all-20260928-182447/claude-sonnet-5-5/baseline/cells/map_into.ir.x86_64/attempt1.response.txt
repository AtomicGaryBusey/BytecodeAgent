export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%c =l phi @start 0, @body %c2
	%d =w csltl %i, %n
	jnz %d, @body, @done
@body
	%off =l shl %i, 3
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%lt =w csltl %r, %x
	%lt2 =l extsw %lt
	%c2 =l add %c, %lt2
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %c
}
