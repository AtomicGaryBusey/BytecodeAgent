binary_search:
    xor rax, rax
    mov r8, rsi
.Lloop:
    cmp rax, r8
    jge .Lnotfound
    mov rcx, r8
    sub rcx, rax
    shr rcx, 1
    add rcx, rax
    mov r9, qword ptr [rdi + rcx*8]
    cmp r9, rdx
    je .Lfound
    jl .Lgreater
    mov r8, rcx
    jmp .Lloop
.Lgreater:
    lea rax, [rcx + 1]
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lnotfound:
    mov rax, -1
    ret
