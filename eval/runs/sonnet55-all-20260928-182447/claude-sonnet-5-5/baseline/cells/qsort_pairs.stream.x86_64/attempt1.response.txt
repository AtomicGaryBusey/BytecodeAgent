qsort_pairs:
    push rbx
    push r12
    push r13
    mov rbx, rdi
    mov r12, rsi
    cmp rsi, 2
    jl .Lret
    mov rax, r12
    shl rax, 4
    mov r8, qword ptr [rbx + rax - 16]
    mov r9, qword ptr [rbx + rax - 8]
    xor r13d, r13d
    xor ecx, ecx
.Lloop:
    lea rax, [r12 - 1]
    cmp rcx, rax
    jge .Ldone
    mov rdx, rcx
    shl rdx, 4
    mov rsi, qword ptr [rbx + rdx]
    mov rdi, qword ptr [rbx + rdx + 8]
    cmp rsi, r8
    jl .Lless
    jg .Lnext
    cmp rdi, r9
    jge .Lnext
.Lless:
    mov rax, r13
    shl rax, 4
    mov r10, qword ptr [rbx + rax]
    mov r11, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rax], rsi
    mov qword ptr [rbx + rax + 8], rdi
    mov qword ptr [rbx + rdx], r10
    mov qword ptr [rbx + rdx + 8], r11
    inc r13
.Lnext:
    inc rcx
    jmp .Lloop
.Ldone:
    mov rax, r13
    shl rax, 4
    lea rdx, [r12 - 1]
    shl rdx, 4
    mov r10, qword ptr [rbx + rax]
    mov r11, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rax], r8
    mov qword ptr [rbx + rax + 8], r9
    mov qword ptr [rbx + rdx], r10
    mov qword ptr [rbx + rdx + 8], r11
    mov rdi, rbx
    mov rsi, r13
    call qsort_pairs
    lea rdi, [r13 + 1]
    shl rdi, 4
    add rdi, rbx
    mov rsi, r12
    sub rsi, r13
    dec rsi
    call qsort_pairs
.Lret:
    xor eax, eax
    pop r13
    pop r12
    pop rbx
    ret
