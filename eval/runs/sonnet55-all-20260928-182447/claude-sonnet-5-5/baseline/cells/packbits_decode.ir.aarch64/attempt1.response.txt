export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @nop %i1, @litok %il, @repok %ir
	%o =l phi @start 0, @nop %o, @litok %ol, @repok %orr
	%c =w csltl %i, %n
	jnz %c, @hdr, @done
@done
	ret %o
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	%isn =w ceqw %h, -128
	jnz %isn, @nop, @h2
@nop
	jmp @loop
@h2
	%neg =w csltw %h, 0
	jnz %neg, @rep, @lit
@err2
	ret -2
@lit
	%hl =l extsw %h
	%cnt =l add %hl, 1
	%end =l add %i1, %cnt
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @litchk
@litchk
	%room =l sub %cap, %o
	%ov =w csgtl %cnt, %room
	jnz %ov, @litov, @litcopy
@litov
	jmp @lovl
@lovl
	%k =l phi @litov 0, @lovb %k2
	%kc =w csltl %k, %room
	jnz %kc, @lovb, @lovd
@lovb
	%sp =l add %src, %i1
	%sp2 =l add %sp, %k
	%bv =w loadub %sp2
	%dp =l add %out, %o
	%dp2 =l add %dp, %k
	storeb %bv, %dp2
	%k2 =l add %k, 1
	jmp @lovl
@lovd
	ret -1
@litcopy
	jmp @lcl
@lcl
	%q =l phi @litcopy 0, @lcb %q2
	%qc =w csltl %q, %cnt
	jnz %qc, @lcb, @litok
@lcb
	%sa =l add %src, %i1
	%sb =l add %sa, %q
	%bw =w loadub %sb
	%da =l add %out, %o
	%db =l add %da, %q
	storeb %bw, %db
	%q2 =l add %q, 1
	jmp @lcl
@litok
	%il =l add %i1, %cnt
	%ol =l add %o, %cnt
	jmp @loop
@rep
	%rh =l extsw %h
	%rcnt =l sub 1, %rh
	%rend =l add %i1, 1
	%rbad =w csgtl %rend, %n
	jnz %rbad, @err2, @repchk
@repchk
	%vp =l add %src, %i1
	%val =w loadub %vp
	%rroom =l sub %cap, %o
	%rov =w csgtl %rcnt, %rroom
	jnz %rov, @rovs, @rcs
@rovs
	jmp @rovl
@rovl
	%a =l phi @rovs 0, @rovb %a2
	%ac =w csltl %a, %rroom
	jnz %ac, @rovb, @rovd
@rovb
	%ra =l add %out, %o
	%rb =l add %ra, %a
	storeb %val, %rb
	%a2 =l add %a, 1
	jmp @rovl
@rovd
	ret -1
@rcs
	jmp @rcl
@rcl
	%b =l phi @rcs 0, @rcb %b2
	%bc =w csltl %b, %rcnt
	jnz %bc, @rcb, @repok
@rcb
	%ca =l add %out, %o
	%cb =l add %ca, %b
	storeb %val, %cb
	%b2 =l add %b, 1
	jmp @rcl
@repok
	%ir =l add %i1, 1
	%orr =l add %o, %rcnt
	jmp @loop
}
