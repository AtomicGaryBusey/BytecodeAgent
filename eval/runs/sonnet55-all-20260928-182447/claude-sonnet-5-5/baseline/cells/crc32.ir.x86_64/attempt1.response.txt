export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @outer
@outer
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 8
	jmp @inner
@inner
	%bit =w and %crc, 1
	%neg =w sub 0, %bit
	%mask =w and %neg, 3988292384
	%sh =w shr %crc, 1
	%crc =w xor %sh, %mask
	%k =w sub %k, 1
	jnz %k, @inner, @next
@next
	%i =l add %i, 1
	jmp @outer
@done
	%r =w xor %crc, 4294967295
	%res =l extuw %r
	ret %res
}
