mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    mov r9, rsi
    imul r9, rsi
    xor r10, r10
.Lzo:
    mov qword ptr [rcx + r10*8], 0
    inc r10
    cmp r10, r9
    jl .Lzo
    xor r10, r10
    lea r11, [rsi + 1]
.Ldiag:
    mov qword ptr [rcx + r10*8], 1
    add r10, r11
    cmp r10, r9
    jl .Ldiag
.Lkloop:
    test rdx, rdx
    jle .Lkdone
    xor r10, r10
.Lrow:
    xor r11, r11
.Lcol:
    xor eax, eax
    xor ebx, ebx
    xor r12, r12
.Linner:
    mov rbp, qword ptr [rcx + r10*8]
    lea r13, [r10 + rbx]
    mov rbp, qword ptr [rcx + r13*8]
    lea r13, [r12 + r11]
    imul rbp, qword ptr [rdi + r13*8]
    add rax, rbp
    add r12, rsi
    inc rbx
    cmp rbx, rsi
    jl .Linner
    lea r13, [r10 + r11]
    mov qword ptr [r8 + r13*8], rax
    inc r11
    cmp r11, rsi
    jl .Lcol
    add r10, rsi
    cmp r10, r9
    jl .Lrow
    xor r10, r10
.Lcopy:
    mov rax, qword ptr [r8 + r10*8]
    mov qword ptr [rcx + r10*8], rax
    inc r10
    cmp r10, r9
    jl .Lcopy
    dec rdx
    jmp .Lkloop
.Lkdone:
    lea r9, [r9 + r9]
    xor r10, r10
.Lzt:
    mov qword ptr [r8 + r10*8], 0
    inc r10
    cmp r10, r9
    jl .Lzt
    xor eax, eax
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
