sdivrem:
    mov r8, rdx
    test rsi, rsi
    jz .Lzero
    cmp rsi, -1
    je .Lneg
    mov rcx, rsi
    mov rax, rdi
    cqo
    idiv rcx
    mov qword ptr [r8], rdx
    ret
.Lneg:
    mov rax, rdi
    neg rax
    mov qword ptr [r8], 0
    ret
.Lzero:
    mov qword ptr [r8], rdi
    xor eax, eax
    ret
