export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @rep2 %i2, @lcd %j
	%o =l phi @start 0, @rep2 %o2, @lcd %o3
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@body
	%p =l add %src, %i
	%b =w loadub %p
	jmp @rl
@rl
	%k =l phi @body 1, @rlc %k1
	%c1 =w csltl %k, 128
	jnz %c1, @rl2, @rd
@rl2
	%ik =l add %i, %k
	%c2 =w csltl %ik, %n
	jnz %c2, @rl3, @rd
@rl3
	%q =l add %src, %ik
	%bb =w loadub %q
	%c3 =w ceqw %bb, %b
	jnz %c3, @rlc, @rd
@rlc
	%k1 =l add %k, 1
	jmp @rl
@rd
	%c4 =w csltl %k, 3
	jnz %c4, @lit, @rep
@rep
	%o2 =l add %o, 2
	%c5 =w cslel %o2, %cap
	jnz %c5, @rep2, @fail
@rep2
	%hdr =l sub 257, %k
	%po =l add %out, %o
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%i2 =l add %i, %k
	jmp @loop
@lit
	jmp @lj
@lj
	%j =l phi @lit %i, @ljc %j1
	%c6 =w csltl %j, %n
	jnz %c6, @lj1, @le
@lj1
	%d =l sub %j, %i
	%c7 =w csltl %d, 128
	jnz %c7, @lj2, @le
@lj2
	%jb =l add %j, 2
	%c8 =w csltl %jb, %n
	jnz %c8, @lj3, @ljc
@lj3
	%pj =l add %src, %j
	%a =w loadub %pj
	%pj1 =l add %pj, 1
	%b1 =w loadub %pj1
	%c9 =w ceqw %a, %b1
	jnz %c9, @lj4, @ljc
@lj4
	%pj2 =l add %pj, 2
	%b2 =w loadub %pj2
	%c10 =w ceqw %a, %b2
	jnz %c10, @le, @ljc
@ljc
	%j1 =l add %j, 1
	jmp @lj
@le
	%L =l sub %j, %i
	%t =l add %o, 1
	%o3 =l add %t, %L
	%c11 =w cslel %o3, %cap
	jnz %c11, @lw, @fail
@lw
	%h2 =l sub %L, 1
	%po_ =l add %out, %o
	storeb %h2, %po_
	jmp @cp
@cp
	%m =l phi @lw 0, @cpb %m1
	%c12 =w csltl %m, %L
	jnz %c12, @cpb, @lcd
@cpb
	%sa =l add %i, %m
	%sp =l add %src, %sa
	%v =w loadub %sp
	%da =l add %t, %m
	%dp =l add %out, %o
	%dp2 =l add %dp, 1
	%dp3 =l add %dp2, %m
	storeb %v, %dp3
	%m1 =l add %m, 1
	jmp @cp
@lcd
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
