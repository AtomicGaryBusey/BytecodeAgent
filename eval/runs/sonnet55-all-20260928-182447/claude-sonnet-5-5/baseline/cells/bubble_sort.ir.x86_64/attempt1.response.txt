export function l $bubble_sort(l %a, l %n) {
@start
	%lim0 =l sub %n, 1
	jmp @outer
@outer
	%lim =l phi @start %lim0, @outer_end %lim2
	%c =w csgtl %lim, 0
	jnz %c, @inner_init, @done
@inner_init
	jmp @inner
@inner
	%j =l phi @inner_init 0, @inner_next %j2
	%cj =w csltl %j, %lim
	jnz %cj, @body, @outer_end
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%g =w csgtl %x, %y
	jnz %g, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner_next
@inner_next
	%j2 =l add %j, 1
	jmp @inner
@outer_end
	%lim2 =l sub %lim, 1
	jmp @outer
@done
	ret 0
}
