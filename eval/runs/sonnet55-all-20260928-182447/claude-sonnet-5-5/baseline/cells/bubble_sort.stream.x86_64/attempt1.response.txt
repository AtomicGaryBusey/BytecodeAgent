bubble_sort:
    mov rcx, rsi
.Louter:
    cmp rcx, 1
    jle .Ldone
    xor rdx, rdx
    lea r8, [rcx - 1]
.Linner:
    cmp rdx, r8
    jge .Lnext
    mov rax, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r9
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
    inc rdx
    jmp .Linner
.Lnext:
    dec rcx
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
