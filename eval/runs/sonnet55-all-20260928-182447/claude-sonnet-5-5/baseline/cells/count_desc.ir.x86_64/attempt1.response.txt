export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @init
@init
	%m =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @cont %i2
	%cnt =l phi @init 0, @cont %cnt2
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%cnt2 =l add %cnt, %gl
	%i2 =l add %i, 1
	%d =w csltl %i2, %m
	jnz %d, @cont, @done2
@cont
	jmp @loop
@done2
	ret %cnt2
@done
	ret 0
}
