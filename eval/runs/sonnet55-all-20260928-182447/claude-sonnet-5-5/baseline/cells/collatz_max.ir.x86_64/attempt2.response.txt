export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %k2
	%best =l phi @start 0, @next %best2
	%c =w csgtl %k, %n
	jnz %c, @done, @loop
@loop
	%x =l phi @outer %k, @step %x3
	%s =l phi @outer 0, @step %s2
	%e =w ceql %x, 1
	jnz %e, @fin, @step
@step
	%b =l and %x, 1
	%h =l shr %x, 1
	%t =l mul %x, 3
	%t1 =l add %t, 1
	%d =l sub %t1, %h
	%m =l mul %b, %d
	%x3 =l add %h, %m
	%s2 =l add %s, 1
	jmp @loop
@fin
	%gt =w csgtl %s, %best
	jnz %gt, @upd, @next
@upd
	jmp @next
@next
	%best2 =l phi @fin %best, @upd %s
	%k2 =l add %k, 1
	jmp @outer
@done
	ret %best
}
