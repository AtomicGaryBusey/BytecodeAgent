export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done0, @init
@init
	%m =l sub %n, 1
	%z =l copy 0
	jmp @loop
@loop
	%i =l phi @init %z, @body %i2
	%cnt =l phi @init %z, @body %cnt2
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%cnt2 =l add %cnt, %gl
	%i2 =l add %i, 1
	%d =w csltl %i2, %m
	jnz %d, @body, @done
@body
	jmp @loop
@done
	ret %cnt2
@done0
	ret 0
}
