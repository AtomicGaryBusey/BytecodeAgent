export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
	jmp @initcond
@initcond
	%k =l phi @start 0, @initbody %k2
	%c =w cslel %k, %n
	jnz %c, @initbody, @outer
@initbody
	%off =l mul %k, 8
	%p =l add %row, %off
	storel 0, %p
	%k2 =l add %k, 1
	jmp @initcond
@outer
	%i =l phi @initcond 0, @outerend %i2
	%c1 =w csltl %i, %m
	jnz %c1, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%j =l phi @obody 1, @innerend %j2
	%diag =l phi @obody 0, @innerend %up
	%left =l phi @obody 0, @innerend %new
	%c2 =w cslel %j, %n
	jnz %c2, @ibody, @outerend
@ibody
	%o =l mul %j, 8
	%rp =l add %row, %o
	%up =l loadl %rp
	%jm =l sub %j, 1
	%yp =l add %y, %jm
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%d1 =l add %diag, 1
	jmp @innerend2
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft
@useup
	jmp @innerend2
@useleft
	jmp @innerend2
@innerend2
	%new =l phi @match %d1, @useup %up, @useleft %left
	storel %new, %rp
	jmp @innerend
@innerend
	%j2 =l add %j, 1
	jmp @inner
@outerend
	%i2 =l add %i, 1
	jmp @outer
@done
	%ro =l mul %n, 8
	%rq =l add %row, %ro
	%r =l loadl %rq
	ret %r
}
