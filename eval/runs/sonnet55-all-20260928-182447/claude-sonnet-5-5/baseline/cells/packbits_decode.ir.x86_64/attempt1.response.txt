export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @head
@head
	%i =l phi @start 0, @nop %i1, @litdone %ie, @repdone %ie2
	%o =l phi @start 0, @nop %o, @litdone %ol, @repdone %or
	%c =w csgel %i, %n
	jnz %c, @ret, @hdr
@ret
	ret %o
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	%isn =w ceqw %h, -128
	jnz %isn, @nop, @dec
@nop
	jmp @head
@dec
	%neg =w csltw %h, 0
	jnz %neg, @rep, @lit
@lit
	%hl =l extsw %h
	%cnt =l add %hl, 1
	%ie =l add %i1, %cnt
	%bad =w csgtl %ie, %n
	jnz %bad, @err2, @lcopy0
@lcopy0
	jmp @lloop
@lloop
	%k =l phi @lcopy0 0, @lbody %k2
	%ol =l phi @lcopy0 %o, @lbody %ol2
	%d =w csgel %k, %cnt
	jnz %d, @litdone, @lchk
@lchk
	%f =w csgel %ol, %cap
	jnz %f, @err1, @lbody
@lbody
	%s =l add %src, %i1
	%s2 =l add %s, %k
	%b =w loadub %s2
	%q =l add %out, %ol
	storeb %b, %q
	%k2 =l add %k, 1
	%ol2 =l add %ol, 1
	jmp @lloop
@litdone
	jmp @head
@rep
	%hr =l extsw %h
	%cntr =l sub 1, %hr
	%ie2 =l add %i1, 1
	%badr =w csgtl %ie2, %n
	jnz %badr, @err2, @rload
@rload
	%sp =l add %src, %i1
	%rb =w loadub %sp
	jmp @rloop
@rloop
	%rk =l phi @rload 0, @rbody %rk2
	%or =l phi @rload %o, @rbody %or2
	%rd =w csgel %rk, %cntr
	jnz %rd, @repdone, @rchk
@rchk
	%rf =w csgel %or, %cap
	jnz %rf, @err1, @rbody
@rbody
	%rq =l add %out, %or
	storeb %rb, %rq
	%rk2 =l add %rk, 1
	%or2 =l add %or, 1
	jmp @rloop
@repdone
	jmp @head
@err1
	ret -1
@err2
	ret -2
}
