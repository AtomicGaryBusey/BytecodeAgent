export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%m =l phi @start 0, @body %m2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%a =l add %p, %i
	%b =l loadub %a
	%g =w csgtl %b, %m
	%d =l sub %b, %m
	%g2 =l extuw %g
	%t =l mul %d, %g2
	%m2 =l add %m, %t
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %m
}
