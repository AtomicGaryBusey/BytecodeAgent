export function l $memmove_i64(l %dst, l %src, l %n)
{
@start
	%f =w culel %dst, %src
	jnz %f, @fwd, @bwd
@fwd
	jmp @fh
@fh
	%i =l phi @fwd 0, @fb %i2
	%c =w csltl %i, %n
	jnz %c, @fb, @end
@fb
	%o =l shl %i, 3
	%s =l add %src, %o
	%d =l add %dst, %o
	%v =l loadl %s
	storel %v, %d
	%i2 =l add %i, 1
	jmp @fh
@bwd
	jmp @bh
@bh
	%j =l phi @bwd %n, @bb %j2
	%c2 =w csgtl %j, 0
	jnz %c2, @bb, @end
@bb
	%j2 =l sub %j, 1
	%o2 =l shl %j2, 3
	%s2 =l add %src, %o2
	%d2 =l add %dst, %o2
	%v2 =l loadl %s2
	storel %v2, %d2
	jmp @bh
@end
	ret 0
}
