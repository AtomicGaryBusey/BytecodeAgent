export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%lo1 =l phi @start %lo, @less %lo2, @greater %lo1
	%hi1 =l phi @start %hi, @less %hi1, @greater %mid
	%c =w csltl %lo1, %hi1
	jnz %c, @body, @notfound
@body
	%d =l sub %hi1, %lo1
	%h =l shr %d, 1
	%mid =l add %lo1, %h
	%off =l shl %mid, 3
	%p =l add %a, %off
	%v =l loadl %p
	%e =w ceql %v, %key
	jnz %e, @found, @cmp
@cmp
	%l =w csltl %v, %key
	jnz %l, @less, @greater
@less
	%lo2 =l add %mid, 1
	jmp @loop
@greater
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
