export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @ret, @body
@ret
	ret 0
@body
	%si =l alloc8 8
	%sj =l alloc8 8
	storel 0, %si
	storel 0, %sj
	%m =l sub %n, 1
	%po =l shl %m, 4
	%pp =l add %a, %po
	%pk =l loadl %pp
	%pv1 =l add %pp, 8
	%pv =l loadl %pv1
	jmp @loop
@loop
	%j =l loadl %sj
	%d =w csltl %j, %m
	jnz %d, @chk, @done
@chk
	%jo =l shl %j, 4
	%jp =l add %a, %jo
	%jk =l loadl %jp
	%jp8 =l add %jp, 8
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @eqc
@eqc
	%eq =w ceql %jk, %pk
	%lv =w csltl %jv, %pv
	%both =w and %eq, %lv
	jnz %both, @swap, @next
@swap
	%i =l loadl %si
	%io =l shl %i, 4
	%ip =l add %a, %io
	%ik =l loadl %ip
	%ip8 =l add %ip, 8
	%iv =l loadl %ip8
	storel %ik, %jp
	storel %iv, %jp8
	storel %jk, %ip
	storel %jv, %ip8
	%i3 =l add %i, 1
	storel %i3, %si
	jmp @next
@next
	%j0 =l loadl %sj
	%j2 =l add %j0, 1
	storel %j2, %sj
	jmp @loop
@done
	%fi =l loadl %si
	%so =l shl %fi, 4
	%spp =l add %a, %so
	%sk =l loadl %spp
	%sp8 =l add %spp, 8
	%sv =l loadl %sp8
	storel %sk, %pp
	storel %sv, %pv1
	storel %pk, %spp
	storel %pv, %sp8
	%r1 =l call $qsort_pairs(l %a, l %fi)
	%i1 =l add %fi, 1
	%no =l shl %i1, 4
	%na =l add %a, %no
	%nn =l sub %m, %fi
	%r2 =l call $qsort_pairs(l %na, l %nn)
	ret 0
}
