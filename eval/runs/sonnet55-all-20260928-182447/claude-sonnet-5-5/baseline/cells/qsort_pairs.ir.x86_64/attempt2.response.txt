export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @ret, @body
@ret
	ret 0
@body
	%m =l sub %n, 1
	%po =l shl %m, 4
	%pp =l add %a, %po
	%pk =l loadl %pp
	%pv1 =l add %pp, 8
	%pv =l loadl %pv1
	jmp @loop
@loop
	%i =l phi @body, 0, @next, %i2
	%j =l phi @body, 0, @next, %j2
	%d =w csltl %j, %m
	jnz %d, @chk, @done
@chk
	%jo =l shl %j, 4
	%jp =l add %a, %jo
	%jk =l loadl %jp
	%jp8 =l add %jp, 8
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @eqc
@eqc
	%eq =w ceql %jk, %pk
	%lv =w csltl %jv, %pv
	%both =w and %eq, %lv
	jnz %both, @swap, @skip
@swap
	%io =l shl %i, 4
	%ip =l add %a, %io
	%ik =l loadl %ip
	%ip8 =l add %ip, 8
	%iv =l loadl %ip8
	storel %ik, %jp
	storel %iv, %jp8
	storel %jk, %ip
	storel %jv, %ip8
	%i3 =l add %i, 1
	jmp @next
@skip
	jmp @next
@next
	%i2 =l phi @swap, %i3, @skip, %i
	%j2 =l add %j, 1
	jmp @loop
@done
	%so =l shl %i, 4
	%sp =l add %a, %so
	%sk =l loadl %sp
	%sp8 =l add %sp, 8
	%sv =l loadl %sp8
	storel %sk, %pp
	storel %sv, %pv1
	storel %pk, %sp
	storel %pv, %sp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%i1 =l add %i, 1
	%no =l shl %i1, 4
	%na =l add %a, %no
	%nn =l sub %m, %i
	%r2 =l call $qsort_pairs(l %na, l %nn)
	ret 0
}
