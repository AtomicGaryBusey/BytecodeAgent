packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx rax, byte ptr [rdi + r8]
    inc r8
    cmp rax, -128
    je .Lloop
    test rax, rax
    js .Lrep
    lea r10, [rax + 1]
    mov r11, r8
    add r11, r10
    cmp r11, rsi
    jg .Lerr2
    mov rax, rcx
    sub rax, r9
    cmp r10, rax
    jle .Llok
    mov r10, rax
    mov eax, 1
    jmp .Llgo
.Llok:
    xor eax, eax
.Llgo:
    test r10, r10
    jz .Llafter
.Lcopy:
    movzx r11d, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    dec r10
    jnz .Lcopy
.Llafter:
    test eax, eax
    jnz .Lerr1
    jmp .Lloop
.Lrep:
    cmp r8, rsi
    jge .Lerr2
    mov r10d, 1
    sub r10, rax
    movzx r11d, byte ptr [rdi + r8]
    inc r8
    mov rax, rcx
    sub rax, r9
    cmp r10, rax
    jle .Lrok
    mov r10, rax
    mov eax, 1
    jmp .Lrgo
.Lrok:
    xor eax, eax
.Lrgo:
    test r10, r10
    jz .Lrafter
.Lfill:
    mov byte ptr [rdx + r9], r11b
    inc r9
    dec r10
    jnz .Lfill
.Lrafter:
    test eax, eax
    jnz .Lerr1
    jmp .Lloop
.Lerr2:
    mov rax, -2
    ret
.Lerr1:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
