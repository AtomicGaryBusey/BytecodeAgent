export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	jmp @zl
@zl
	%z =l phi @start 0, @zb %z2
	%c =w csltl %z, %nn
	jnz %c, @zb, @id0
@zb
	%off =l shl %z, 3
	%p =l add %out, %off
	storel 0, %p
	%z2 =l add %z, 1
	jmp @zl
@id0
	%np1 =l add %n, 1
	jmp @idl
@idl
	%i0 =l phi @id0 0, @idb %i02
	%c0 =w csltl %i0, %n
	jnz %c0, @idb, @kl
@idb

	%off0 =l shl %idx, 3
	%p0 =l add %out, %off0
	storel 1, %p0
	%i02 =l add %i0, 1
	jmp @idl
@kl
	%kk =l phi @idl 0, @kn %kk2
	%ck =w csltl %kk, %k
	jnz %ck, @il, @zt
@il
	%i =l phi @kl 0, @inext %i2
	%ci =w csltl %i, %n
	jnz %ci, @jl, @cl
@jl
	%j =l phi @il 0, @tend %j2
	%cj =w csltl %j, %n
	jnz %cj, @tl, @inext
@tl
	%t =l phi @jl 0, @tb %t2
	%s =l phi @jl 0, @tb %s2
	%ct =w csltl %t, %n
	jnz %ct, @tb, @tend
@tb
	%rowi =l mul %i, %n
	%x =l add %rowi, %t
	%xo =l shl %x, 3
	%xp =l add %out, %xo
	%xv =l loadl %xp
	%y =l mul %t, %n
	%y2 =l add %y, %j
	%yo =l shl %y2, 3
	%yp =l add %a, %yo
	%yv =l loadl %yp
	%m =l mul %xv, %yv
	%s2 =l add %s, %m
	%t2 =l add %t, 1
	jmp @tl
@tend
	%r =l mul %i, %n
	%r2 =l add %r, %j
	%ro =l shl %r2, 3
	%rp =l add %tmp, %ro
	storel %s, %rp
	%j2 =l add %j, 1
	jmp @jl
@inext
	%i2 =l add %i, 1
	jmp @il
@cl
	%q =l phi @il 0, @cb %q2
	%cq =w csltl %q, %nn
	jnz %cq, @cb, @kn
@cb
	%qo =l shl %q, 3
	%qs =l add %tmp, %qo
	%qd =l add %out, %qo
	%qv =l loadl %qs
	storel %qv, %qd
	%q2 =l add %q, 1
	jmp @cl
@kn
	%kk2 =l add %kk, 1
	jmp @kl
@zt
	%lim =l shl %nn, 1
	jmp @ztl
@ztl
	%w =l phi @zt 0, @ztb %w2
	%cw =w csltl %w, %lim
	jnz %cw, @ztb, @done
@ztb
	%wo =l shl %w, 3
	%wp =l add %tmp, %wo
	storel 0, %wp
	%w2 =l add %w, 1
	jmp @ztl
@done
	ret 0
}
