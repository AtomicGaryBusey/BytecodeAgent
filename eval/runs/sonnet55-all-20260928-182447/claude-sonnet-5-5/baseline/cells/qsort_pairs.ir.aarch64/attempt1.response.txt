export function l $qsort_pairs(l %a, l %n) {
@start
	%c0 =w csltl %n, 2
	jnz %c0, @ret, @body
@ret
	ret 0
@body
	%n1 =l sub %n, 1
	%off =l shl %n1, 4
	%pp =l add %a, %off
	%pk =l loadl %pp
	%pp8 =l add %pp, 8
	%pv =l loadl %pp8
	jmp @loop
@loop
	%i =l phi @body 0, @next %i2
	%j =l phi @body 0, @next %j2
	%c =w csltl %j, %n1
	jnz %c, @chk, @done
@chk
	%jo =l shl %j, 4
	%jp =l add %a, %jo
	%jk =l loadl %jp
	%jp8 =l add %jp, 8
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @eqc
@eqc
	%eq =w ceql %jk, %pk
	%vl =w csltl %jv, %pv
	%both =w and %eq, %vl
	jnz %both, @swap, @skip
@swap
	%io =l shl %i, 4
	%ip =l add %a, %io
	%ip8 =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%i1 =l add %i, 1
	jmp @next
@skip
	jmp @next
@next
	%i2 =l phi @swap %i1, @skip %i
	%j2 =l add %j, 1
	jmp @loop
@done
	%do =l shl %i, 4
	%dp =l add %a, %do
	%dp8 =l add %dp, 8
	%dk =l loadl %dp
	%dv =l loadl %dp8
	storel %pk, %dp
	storel %pv, %dp8
	storel %dk, %pp
	storel %dv, %pp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%i1b =l add %i, 1
	%o2 =l shl %i1b, 4
	%a2 =l add %a, %o2
	%n2 =l sub %n1, %i
	%r2 =l call $qsort_pairs(l %a2, l %n2)
	ret 0
}
