export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @hpush %pc1, @hbin %pc2, @hdup %pc3, @hswap %pc4, @hdrop %pc5, @hover %pc6, @hlt %pc7, @hjz %pc8, @hjmp %t
	%d =l phi @start 0, @hpush %d1, @hbin %d2, @hdup %d3, @hswap %d, @hdrop %d5, @hover %d6, @hlt %d7, @hjz %d8, @hjmp %d
	%c =w ceql %pc, %len
	jnz %c, @end, @fetch
@fetch
	%po =l shl %pc, 3
	%pa0 =l add %code, %po
	%op =l loadl %pa0
	%pn =l add %pc, 1
	%pn2 =l add %pc, 2
	%off =l shl %d, 3
	%pb =l add %stack, %off
	%pb1 =l sub %pb, 8
	%pa1 =l sub %pb, 16
	%o1 =l add %po, 8
	%opa =l add %code, %o1
	%arg =l loadl %opa
	%e0 =w ceql %op, 0
	jnz %e0, @hhalt, @c1
@c1
	%e1 =w ceql %op, 1
	jnz %e1, @hpush, @c2
@c2
	%e2 =w ceql %op, 2
	jnz %e2, @hadd, @c3
@c3
	%e3 =w ceql %op, 3
	jnz %e3, @hsub, @c4
@c4
	%e4 =w ceql %op, 4
	jnz %e4, @hmul, @c5
@c5
	%e5 =w ceql %op, 5
	jnz %e5, @hdup, @c6
@c6
	%e6 =w ceql %op, 6
	jnz %e6, @hswap, @c7
@c7
	%e7 =w ceql %op, 7
	jnz %e7, @hdrop, @c8
@c8
	%e8 =w ceql %op, 8
	jnz %e8, @hover, @c9
@c9
	%e9 =w ceql %op, 9
	jnz %e9, @hlt, @c10
@c10
	%e10 =w ceql %op, 10
	jnz %e10, @hjz, @hjmp
@hpush
	storel %arg, %pb
	%d1 =l add %d, 1
	%pc1 =l add %pc, 2
	jmp @loop
@hadd
	%a =l loadl %pa1
	%b =l loadl %pb1
	%r =l add %a, %b
	storel %r, %pa1
	jmp @hbin
@hsub
	%a2 =l loadl %pa1
	%b2 =l loadl %pb1
	%r2 =l sub %a2, %b2
	storel %r2, %pa1
	jmp @hbin
@hmul
	%a3 =l loadl %pa1
	%b3 =l loadl %pb1
	%r3 =l mul %a3, %b3
	storel %r3, %pa1
	jmp @hbin
@hbin
	%d2 =l sub %d, 1
	%pc2 =l add %pc, 1
	jmp @loop
@hdup
	%x =l loadl %pb1
	storel %x, %pb
	%d3 =l add %d, 1
	%pc3 =l add %pc, 1
	jmp @loop
@hswap
	%sa =l loadl %pa1
	%sb =l loadl %pb1
	storel %sb, %pa1
	storel %sa, %pb1
	%pc4 =l add %pc, 1
	jmp @loop
@hdrop
	%d5 =l sub %d, 1
	%pc5 =l add %pc, 1
	jmp @loop
@hover
	%y =l loadl %pa1
	storel %y, %pb
	%d6 =l add %d, 1
	%pc6 =l add %pc, 1
	jmp @loop
@hlt
	%la =l loadl %pa1
	%lb =l loadl %pb1
	%lw =w csltl %la, %lb
	%lr =l extuw %lw
	storel %lr, %pa1
	%d7 =l sub %d, 1
	%pc7 =l add %pc, 1
	jmp @loop
@hjz
	%v =l loadl %pb1
	%zw =w ceql %v, 0
	%z =l extuw %zw
	%df =l sub %arg, %pn2
	%mm =l mul %z, %df
	%pc8 =l add %pn2, %mm
	%d8 =l sub %d, 1
	jmp @loop
@hjmp
	%t =l copy %arg
	jmp @loop
@hhalt
	jmp @end
@end
	%dd =l phi @loop %d, @hhalt %d
	jmp @z
@z
	%i =l phi @end %dd, @zb %i1
	%zc =w csltl %i, %cap
	jnz %zc, @zb, @fin
@zb
	%zo =l shl %i, 3
	%zp =l add %stack, %zo
	storel 0, %zp
	%i1 =l add %i, 1
	jmp @z
@fin
	%ne =w ceql %dd, 0
	jnz %ne, @empty, @ret
@empty
	ret 0
@ret
	%ro =l shl %dd, 3
	%rp =l add %stack, %ro
	%rp1 =l sub %rp, 8
	%rv =l loadl %rp1
	ret %rv
}
