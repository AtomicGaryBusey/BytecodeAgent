export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l loadl %a
	%r0 =l call %f(l %y0)
	%odd0 =l and %r0, 1
	%i1 =l copy 1
	jmp @loop
@loop
	%i =l phi @start, %i1, @body, %i2
	%min =l phi @start, %r0, @body, %min2
	%max =l phi @start, %r0, @body, %max2
	%sum =l phi @start, %r0, @body, %sum2
	%cnt =l phi @start, %odd0, @body, %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%min2 =l select %lt, %y, %min
	%gt =w csgtl %y, %max
	%max2 =l select %gt, %y, %max
	%sum2 =l add %sum, %y
	%o =l and %y, 1
	%cnt2 =l add %cnt, %o
	%i2 =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%p1 =l add %out, 8
	storel %max, %p1
	%p2 =l add %out, 16
	storel %sum, %p2
	ret %cnt
}
