isqrt:
    xor eax, eax
    mov rcx, 1
    shl rcx, 62
.Lskip:
    cmp rcx, rdi
    jbe .Lloop
    shr rcx, 2
    jnz .Lskip
    ret
.Lloop:
    lea rdx, [rax + rcx]
    cmp rdi, rdx
    jb .Lno
    sub rdi, rdx
    shr rax, 1
    add rax, rcx
    jmp .Lnext
.Lno:
    shr rax, 1
.Lnext:
    shr rcx, 2
    jnz .Lloop
    ret
