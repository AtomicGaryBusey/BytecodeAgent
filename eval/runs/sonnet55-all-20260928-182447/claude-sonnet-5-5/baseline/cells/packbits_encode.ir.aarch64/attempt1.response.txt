export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @lend %i2
	%o =l phi @start 0, @lend %o2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@done
	ret %o
@body
	%pi =l add %src, %i
	%b =w loadub %pi
	jmp @r1
@r1
	%k =l phi @body 1, @r1c %k2
	%p =l add %i, %k
	%c1 =w csltl %p, %n
	jnz %c1, @r2, @rend
@r2
	%c2 =w csltl %k, 128
	jnz %c2, @r3, @rend
@r3
	%q =l add %src, %p
	%x =w loadub %q
	%c3 =w ceqw %x, %b
	jnz %c3, @r1c, @rend
@r1c
	%k2 =l add %k, 1
	jmp @r1
@rend
	%r =l phi @r1 %k, @r2 %k, @r3 %k
	%c4 =w csgel %r, 3
	jnz %c4, @rep, @lit0
@rep
	%t =l add %o, 2
	%f =w csgtl %t, %cap
	jnz %f, @fail, @rep2
@fail
	ret -1
@rep2
	%h =l sub 257, %r
	%po =l add %out, %o
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%ia =l add %i, %r
	jmp @lend
@lit0
	%ip1 =l add %i, 1
	jmp @l1
@l1
	%j =l phi @lit0 %ip1, @l1c %j2
	%d0 =w csltl %j, %n
	jnz %d0, @l2, @lfin
@l2
	%d =l sub %j, %i
	%d1 =w csltl %d, 128
	jnz %d1, @l3, @lfin
@l3
	%jn =l add %j, 2
	%d2 =w csltl %jn, %n
	jnz %d2, @l4, @l1c
@l4
	%pj =l add %src, %j
	%x0 =w loadub %pj
	%pj1 =l add %pj, 1
	%y0 =w loadub %pj1
	%pj2 =l add %pj, 2
	%z0 =w loadub %pj2
	%e1 =w ceqw %x0, %y0
	%e2 =w ceqw %x0, %z0
	%e =w and %e1, %e2
	jnz %e, @lfin, @l1c
@l1c
	%j2 =l add %j, 1
	jmp @l1
@lfin
	%jf =l phi @l1 %j, @l2 %j, @l4 %j
	%L =l sub %jf, %i
	%t1 =l add %o, 1
	%t2 =l add %t1, %L
	%f2 =w csgtl %t2, %cap
	jnz %f2, @fail, @lw
@lw
	%hl =l sub %L, 1
	%po2 =l add %out, %o
	storeb %hl, %po2
	%dst =l add %po2, 1
	%sp =l add %src, %i
	jmp @cp
@cp
	%m =l phi @lw 0, @cpb %m2
	%cc =w csltl %m, %L
	jnz %cc, @cpb, @lend
@cpb
	%sa =l add %sp, %m
	%v =w loadub %sa
	%da =l add %dst, %m
	storeb %v, %da
	%m2 =l add %m, 1
	jmp @cp
@lend
	%i2 =l phi @rep2 %ia, @cp %jf
	%o2 =l phi @rep2 %t, @cp %t2
	jmp @loop
}
