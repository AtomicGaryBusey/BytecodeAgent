packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Lmain:
    cmp r8, rsi
    jge .Ldone
    movzx r11d, byte ptr [rdi + r8]
    mov r10d, 1
.Lrun:
    cmp r10, 128
    jge .Lrund
    lea rax, [r8 + r10]
    cmp rax, rsi
    jge .Lrund
    cmp r11b, byte ptr [rdi + rax]
    jne .Lrund
    inc r10
    jmp .Lrun
.Lrund:
    cmp r10, 3
    jl .Llit
    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lfail
    mov eax, 1
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r11b
    add r9, 2
    add r8, r10
    jmp .Lmain
.Llit:
    mov r10, r8
.Lljl:
    cmp r10, rsi
    jge .Lld
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jge .Lld
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Llnext
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llnext
    cmp al, byte ptr [rdi + r10 + 2]
    je .Lld
.Llnext:
    inc r10
    jmp .Lljl
.Lld:
    mov rax, r10
    sub rax, r8
    lea r11, [r9 + rax + 1]
    cmp r11, rcx
    jg .Lfail
    lea r11, [rax - 1]
    mov byte ptr [rdx + r9], r11b
    inc r9
.Lcopy:
    cmp r8, r10
    jge .Lmain
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    jmp .Lcopy
.Lfail:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
