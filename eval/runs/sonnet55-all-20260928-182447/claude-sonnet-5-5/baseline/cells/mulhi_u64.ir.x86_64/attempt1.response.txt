export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%t =l shr %ll, 32
	%m1 =l and %lh, 4294967295
	%m2 =l and %hl, 4294967295
	%s =l add %t, %m1
	%s2 =l add %s, %m2
	%carry =l shr %s2, 32
	%lhh =l shr %lh, 32
	%hlh =l shr %hl, 32
	%r =l add %hh, %lhh
	%r2 =l add %r, %hlh
	%r3 =l add %r2, %carry
	ret %r3
}
