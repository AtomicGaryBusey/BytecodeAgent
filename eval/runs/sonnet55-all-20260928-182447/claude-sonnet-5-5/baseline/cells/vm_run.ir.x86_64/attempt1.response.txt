export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%c =w ceql %pc, %len
	jnz %c, @end, @fetch
@fetch
	%pco =l shl %pc, 3
	%ca =l add %code, %pco
	%op =l loadl %ca
	%sp =l loadl %spp
	%spo =l shl %sp, 3
	%top1 =l add %stack, %spo
	%top =l sub %top1, 8
	%sec =l sub %top, 8
	%t0 =w ceql %op, 0
	jnz %t0, @end, @d1
@d1
	%t1 =w ceql %op, 1
	jnz %t1, @push, @d2
@d2
	%t2 =w ceql %op, 2
	jnz %t2, @add, @d3
@d3
	%t3 =w ceql %op, 3
	jnz %t3, @sub, @d4
@d4
	%t4 =w ceql %op, 4
	jnz %t4, @mul, @d5
@d5
	%t5 =w ceql %op, 5
	jnz %t5, @dup, @d6
@d6
	%t6 =w ceql %op, 6
	jnz %t6, @swap, @d7
@d7
	%t7 =w ceql %op, 7
	jnz %t7, @drop, @d8
@d8
	%t8 =w ceql %op, 8
	jnz %t8, @over, @d9
@d9
	%t9 =w ceql %op, 9
	jnz %t9, @lt, @d10
@d10
	%t10 =w ceql %op, 10
	jnz %t10, @jz, @jmp
@push
	%va =l add %ca, 8
	%v =l loadl %va
	storel %v, %top1
	%spa =l add %sp, 1
	storel %spa, %spp
	%pca =l add %pc, 2
	storel %pca, %pcp
	jmp @loop
@add
	%b1 =l loadl %top
	%a1 =l loadl %sec
	%r1 =l add %a1, %b1
	storel %r1, %sec
	jmp @pop1
@sub
	%b2 =l loadl %top
	%a2 =l loadl %sec
	%r2 =l sub %a2, %b2
	storel %r2, %sec
	jmp @pop1
@mul
	%b3 =l loadl %top
	%a3 =l loadl %sec
	%r3 =l mul %a3, %b3
	storel %r3, %sec
	jmp @pop1
@lt
	%b4 =l loadl %top
	%a4 =l loadl %sec
	%r4 =w csltl %a4, %b4
	%r4l =l extuw %r4
	storel %r4l, %sec
	jmp @pop1
@drop
	jmp @pop1
@pop1
	%spm =l sub %sp, 1
	storel %spm, %spp
	jmp @inc1
@dup
	%v5 =l loadl %top
	storel %v5, %top1
	jmp @push1
@over
	%v6 =l loadl %sec
	storel %v6, %top1
	jmp @push1
@push1
	%spq =l add %sp, 1
	storel %spq, %spp
	jmp @inc1
@swap
	%b7 =l loadl %top
	%a7 =l loadl %sec
	storel %a7, %top
	storel %b7, %sec
	jmp @inc1
@inc1
	%pcq =l add %pc, 1
	storel %pcq, %pcp
	jmp @loop
@jz
	%vz =l loadl %top
	%spz =l sub %sp, 1
	storel %spz, %spp
	%ta =l add %ca, 8
	%tg =l loadl %ta
	%pcz =l add %pc, 2
	%isz =w ceql %vz, 0
	jnz %isz, @jzt, @jzf
@jzt
	storel %tg, %pcp
	jmp @loop
@jzf
	storel %pcz, %pcp
	jmp @loop
@jmp
	%ta2 =l add %ca, 8
	%tg2 =l loadl %ta2
	storel %tg2, %pcp
	jmp @loop
@end
	%spe =l loadl %spp
	jmp @z
@z
	%i =l phi @end %spe, @zb %i2
	%cz =w csltl %i, %cap
	jnz %cz, @zb, @done
@zb
	%o =l shl %i, 3
	%a =l add %stack, %o
	storel 0, %a
	%i2 =l add %i, 1
	jmp @z
@done
	%e =w ceql %spe, 0
	jnz %e, @ret0, @rt
@rt
	%o2 =l shl %spe, 3
	%a5 =l add %stack, %o2
	%a6 =l sub %a5, 8
	%rv =l loadl %a6
	ret %rv
@ret0
	ret 0
}
