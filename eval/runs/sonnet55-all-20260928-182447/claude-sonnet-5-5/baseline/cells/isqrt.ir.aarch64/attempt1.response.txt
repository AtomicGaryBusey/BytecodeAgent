export function l $isqrt(l %n) {
@start
	jmp @l1
@l1
	%bit =l phi @start 4611686018427387904, @dec %bit2
	%c =w cugtl %bit, %n
	jnz %c, @dec, @l2
@dec
	%bit2 =l shr %bit, 2
	jmp @l1
@l2
	%r =l phi @l1 0, @cont %rr
	%b =l phi @l1 %bit, @cont %b3
	%m =l phi @l1 %n, @cont %nn
	%z =w ceql %b, 0
	jnz %z, @end, @body
@body
	%t =l add %r, %b
	%ge =w cugel %m, %t
	jnz %ge, @yes, @no
@yes
	%n2 =l sub %m, %t
	%r1 =l shr %r, 1
	%r2 =l add %r1, %b
	jmp @cont
@no
	%r3 =l shr %r, 1
	jmp @cont
@cont
	%rr =l phi @yes %r2, @no %r3
	%nn =l phi @yes %n2, @no %m
	%b3 =l shr %b, 2
	jmp @l2
@end
	ret %r
}
