export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%llh =l shr %ll, 32
	%t =l add %hl, %llh
	%tl =l and %t, 4294967295
	%th =l shr %t, 32
	%u =l add %tl, %lh
	%uh =l shr %u, 32
	%r =l add %hh, %th
	%r2 =l add %r, %uh
	ret %r2
}
