export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zloop
@zloop
	%z =l phi @start 0, @zbody %z2
	%zc =w csltl %z, %n
	jnz %zc, @zbody, @oloop
@zbody
	%zo =l mul %z, 8
	%zp =l add %r, %zo
	storel 0, %zp
	%z2 =l add %z, 1
	jmp @zloop
@oloop
	%i =l phi @zloop 0, @oend %i2
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @count
@ibody
	%io =l mul %i, 8
	%ip =l add %a, %io
	%x =l loadl %ip
	%x0 =l and %x, 4294967295
	%x1 =l shr %x, 32
	jmp @iloop
@iloop
	%j =l phi @ibody 0, @jbody %j2
	%carry =l phi @ibody 0, @jbody %carry2
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @oend
@jbody
	%jo =l mul %j, 8
	%bp =l add %b, %jo
	%y =l loadl %bp
	%y0 =l and %y, 4294967295
	%y1 =l shr %y, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%m1 =l shr %p00, 32
	%m2 =l and %p01, 4294967295
	%m3 =l and %p10, 4294967295
	%m4 =l add %m1, %m2
	%mid =l add %m4, %m3
	%h1 =l shr %p01, 32
	%h2 =l shr %p10, 32
	%h3 =l shr %mid, 32
	%h4 =l add %p11, %h1
	%h5 =l add %h4, %h2
	%hi =l add %h5, %h3
	%lo =l mul %x, %y
	%ij =l add %i, %j
	%ijo =l mul %ij, 8
	%rp =l add %r, %ijo
	%rv =l loadl %rp
	%t =l add %lo, %rv
	%c1 =l cultl %t, %lo
	%t2 =l add %t, %carry
	%c2 =l cultl %t2, %t
	storel %t2, %rp
	%hc =l add %hi, %c1
	%carry2 =l add %hc, %c2
	%j2 =l add %j, 1
	jmp @iloop
@oend
	%ij2 =l add %i, %nb
	%o2 =l mul %ij2, 8
	%rp2 =l add %r, %o2
	storel %carry, %rp2
	%i2 =l add %i, 1
	jmp @oloop
@count
	%k =l phi @oloop %n, @cdec %k2
	%kc =w csgtl %k, 0
	jnz %kc, @cchk, @done
@cchk
	%km =l sub %k, 1
	%ko =l mul %km, 8
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @cdec, @done
@cdec
	%k2 =l sub %k, 1
	jmp @count
@done
	ret %k
}
