dot_i32_simd:
    xor eax, eax
    pxor xmm0, xmm0
    lea rcx, [rdx - 3]
.Lvec:
    cmp rax, rcx
    jge .Lvecend
    movdqu xmm1, xmmword ptr [rdi + rax*4]
    movdqu xmm2, xmmword ptr [rsi + rax*4]
    movdqa xmm3, xmm1
    pmuldq xmm3, xmm2
    psrlq xmm1, 32
    psrlq xmm2, 32
    pmuldq xmm1, xmm2
    paddq xmm0, xmm3
    paddq xmm0, xmm1
    add rax, 4
    jmp .Lvec
.Lvecend:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    paddq xmm0, xmm1
    movq r8, xmm0
.Lscalar:
    cmp rax, rdx
    jge .Ldone
    movsxd r9, dword ptr [rdi + rax*4]
    movsxd r10, dword ptr [rsi + rax*4]
    imul r9, r10
    add r8, r9
    inc rax
    jmp .Lscalar
.Ldone:
    mov rax, r8
    ret
