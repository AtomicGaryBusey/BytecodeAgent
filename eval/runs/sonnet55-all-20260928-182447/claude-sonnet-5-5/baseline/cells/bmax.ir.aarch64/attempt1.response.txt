export function l $bmax(l %a, l %n) {
@start
	%m =l loadl %a
	%i =l copy 1
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%g =w csgtl %v, %m
	jnz %g, @upd, @next
@upd
	%m =l copy %v
	jmp @next
@next
	%i =l add %i, 1
	jmp @cond
@end
	ret %m
}
