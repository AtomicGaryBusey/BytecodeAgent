export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%T =l add %tmp, %nn8
	jmp @il
@il
	%ii =l phi @start 0, @ib %ii2
	%c0 =w csltl %ii, %nn
	jnz %c0, @ib, @cs
@ib
	%r =l div %ii, %n
	%cc =l rem %ii, %n
	%e =w ceql %r, %cc
	%ev =l extuw %e
	%o =l mul %ii, 8
	%p =l add %out, %o
	storel %ev, %p
	%pa =l add %a, %o
	%av =l loadl %pa
	%pb =l add %tmp, %o
	storel %av, %pb
	%ii2 =l add %ii, 1
	jmp @il
@cs
	jmp @kl
@kl
	%k1 =l phi @cs %k, @next %k2
	%kz =w ceql %k1, 0
	jnz %kz, @fin, @kb
@kb
	%bit =l and %k1, 1
	%bw =w cnel %bit, 0
	jnz %bw, @dor, @sq
@dor
	jmp @mul
@sq
	jmp @mul
@mul
	%X =l phi @dor %out, @sq %tmp
	%Y =l phi @dor %tmp, @sq %tmp
	%ph =l phi @dor 0, @sq 1
	jmp @ml
@ml
	%idx =l phi @mul 0, @mlend %idx2
	%c =w csltl %idx, %nn
	jnz %c, @mb, @cp
@mb
	%i =l div %idx, %n
	%j =l rem %idx, %n
	%ibase =l mul %i, %n
	jmp @tl
@tl
	%t =l phi @mb 0, @tb %t2
	%acc =l phi @mb 0, @tb %acc2
	%ct =w csltl %t, %n
	jnz %ct, @tb, @mlend
@tb
	%xi =l add %ibase, %t
	%xo =l mul %xi, 8
	%xa =l add %X, %xo
	%x =l loadl %xa
	%yi =l mul %t, %n
	%yj =l add %yi, %j
	%yo =l mul %yj, 8
	%ya =l add %Y, %yo
	%y =l loadl %ya
	%m =l mul %x, %y
	%acc2 =l add %acc, %m
	%t2 =l add %t, 1
	jmp @tl
@mlend
	%ti =l add %ibase, %j
	%to =l mul %ti, 8
	%tp =l add %T, %to
	storel %acc, %tp
	%idx2 =l add %idx, 1
	jmp @ml
@cp
	jmp @cl
@cl
	%q =l phi @cp 0, @cbody %q2
	%cq =w csltl %q, %nn
	jnz %cq, @cbody, @cend
@cbody
	%qo =l mul %q, 8
	%qs =l add %T, %qo
	%qv =l loadl %qs
	%qd =l add %X, %qo
	storel %qv, %qd
	%q2 =l add %q, 1
	jmp @cl
@cend
	jnz %ph, @next, @sq
@next
	%k2 =l shr %k1, 1
	jmp @kl
@fin
	%n2 =l mul %nn, 2
	jmp @zl
@zl
	%z =l phi @fin 0, @zb %z2
	%cz =w csltl %z, %n2
	jnz %cz, @zb, @done
@zb
	%zo =l mul %z, 8
	%zp =l add %tmp, %zo
	storel 0, %zp
	%z2 =l add %z, 1
	jmp @zl
@done
	ret 0
}
