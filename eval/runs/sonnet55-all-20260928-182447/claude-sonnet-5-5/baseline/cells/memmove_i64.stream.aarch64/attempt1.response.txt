memmove_i64:
    mov x0, x0
    cbz x2, .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.hi .Lback
.Lfwd:
    ldr x9, [x1], #8
    str x9, [x0], #8
    subs x2, x2, #1
    b.ne .Lfwd
    b .Ldone
.Lback:
    sub x2, x2, #1
.Lbloop:
    ldr x9, [x1, x2, lsl #3]
    str x9, [x0, x2, lsl #3]
    subs x2, x2, #1
    b.hs .Lbloop
.Ldone:
    mov x0, #0
    ret
