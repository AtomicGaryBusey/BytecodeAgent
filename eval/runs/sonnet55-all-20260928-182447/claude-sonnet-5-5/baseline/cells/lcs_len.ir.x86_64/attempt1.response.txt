export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
	jmp @initcond
@initcond
	%c =w cslel %j, %n
	jnz %c, @initbody, @outer0
@initbody
	%o =l mul %j, 8
	%p =l add %row, %o
	storel 0, %p
	%j =l add %j, 1
	jmp @initcond
@outer0
	%i =l copy 0
	jmp @ocond
@ocond
	%c2 =w csltl %i, %m
	jnz %c2, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag =l copy 0
	%k =l copy 1
	jmp @icond
@icond
	%c3 =w cslel %k, %n
	jnz %c3, @ibody, @onext
@ibody
	%o2 =l mul %k, 8
	%p2 =l add %row, %o2
	%up =l loadl %p2
	%pl =l sub %p2, 8
	%left =l loadl %pl
	%yo =l add %y, %k
	%yp =l sub %yo, 1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @store
@nomatch
	%g =w csgtl %up, %left
	jnz %g, @useup, @useleft
@useup
	%v2 =l copy %up
	jmp @store2
@useleft
	%v2 =l copy %left
	jmp @store2
@store2
	storel %v2, %p2
	jmp @inext
@store
	storel %v1, %p2
	jmp @inext
@inext
	%diag =l copy %up
	%k =l add %k, 1
	jmp @icond
@onext
	%i =l add %i, 1
	jmp @ocond
@done
	%on =l mul %n, 8
	%pn =l add %row, %on
	%r =l loadl %pn
	ret %r
}
