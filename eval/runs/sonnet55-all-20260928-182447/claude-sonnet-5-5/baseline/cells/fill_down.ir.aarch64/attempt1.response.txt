export function l $fill_down(l %a, l %n, l %start) {
@entry
	%c =w csltl 0, %n
	jnz %c, @loop, @done
@loop
	%i =l phi @entry 0, @loop %i2
	%v =l phi @entry %start, @loop %v2
	%p =l phi @entry %a, @loop %p2
	storel %v, %p
	%i2 =l add %i, 1
	%v2 =l sub %v, 1
	%p2 =l add %p, 8
	%d =w csltl %i2, %n
	jnz %d, @loop, @done
@done
	ret %n
}
