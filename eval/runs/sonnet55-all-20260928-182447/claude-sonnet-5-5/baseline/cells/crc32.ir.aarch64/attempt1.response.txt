export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 0
	jmp @bit
@bit
	%lsb =w and %crc, 1
	%m =w sub 0, %lsb
	%x =w and %m, 3988292384
	%s =w shr %crc, 1
	%crc =w xor %s, %x
	%k =w add %k, 1
	%kc =w csltw %k, 8
	jnz %kc, @bit, @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	%r =w xor %crc, 4294967295
	%res =l extuw %r
	ret %res
}
