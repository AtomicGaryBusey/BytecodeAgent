export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l loadl %a
	%r0 =l call %f(l %y0)
	jmp @loop.head

@loop.head
	%i =l phi @start, 1, @loop.next, %i2
	%min =l phi @start, %r0, @loop.next, %min2
	%max =l phi @start, %r0, @loop.next, %max2
	%sum =l phi @start, %r0, @loop.next, %sum2
	%cnt =l phi @start, %c0, @loop.next, %cnt2
	%c =w csltl %i, %n
	jnz %c, @loop.body, @done

@loop.body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%lt =w csltl %r, %min
	%min2 =l select %lt, %r, %min
	%gt =w csgtl %r, %max
	%max2 =l select %gt, %r, %max
	%sum2 =l add %sum, %r
	%bit =l and %r, 1
	%cnt2 =l add %cnt, %bit
	%i2 =l add %i, 1
	jmp @loop.next

@loop.next
	jmp @loop.head

@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
