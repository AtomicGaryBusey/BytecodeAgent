export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l loadl %a
	%r0 =l call %f(l %y0)
	%c0 =l and %r0, 1
	jmp @loop.head

@loop.head
	%i =l phi @start 1, @nomax %i2
	%min =l phi @start %r0, @nomax %min2
	%max =l phi @start %r0, @nomax %max2
	%sum =l phi @start %r0, @nomax %sum2
	%cnt =l phi @start %c0, @nomax %cnt2
	%c =w csltl %i, %n
	jnz %c, @loop.body, @done

@loop.body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%lt =w csltl %r, %min
	jnz %lt, @setmin, @nomin

@setmin
	jmp @nomin

@nomin
	%min2 =l phi @loop.body %min, @setmin %r
	%gt =w csgtl %r, %max
	jnz %gt, @setmax, @nomax

@setmax
	jmp @nomax

@nomax
	%max2 =l phi @nomin %max, @setmax %r
	%sum2 =l add %sum, %r
	%bit =l and %r, 1
	%cnt2 =l add %cnt, %bit
	%i2 =l add %i, 1
	jmp @loop.head

@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
