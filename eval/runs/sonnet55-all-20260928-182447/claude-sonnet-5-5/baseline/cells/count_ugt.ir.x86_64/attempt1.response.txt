export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%c =l phi @start 0, @body %c2
	%d =w csltl %i, %n
	jnz %d, @body, @end
@body
	%o =l shl %i, 3
	%p =l add %a, %o
	%v =l loadl %p
	%g =w cugtl %v, %t
	%gl =l extuw %g
	%c2 =l add %c, %gl
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %c
}
