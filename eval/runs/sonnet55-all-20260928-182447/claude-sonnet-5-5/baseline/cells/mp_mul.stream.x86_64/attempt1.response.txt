mp_mul:
    push rbx
    push r12
    mov r9, rdx
    lea r10, [rsi + rcx]
    xor eax, eax
    xor r11d, r11d
.Lz:
    cmp r11, r10
    jae .Lzd
    mov qword ptr [r8 + r11*8], rax
    inc r11
    jmp .Lz
.Lzd:
    test rsi, rsi
    jz .Lret0
    test rcx, rcx
    jz .Lret0
    xor r11d, r11d
.Louter:
    mov rbx, qword ptr [rdi + r11*8]
    xor r10d, r10d
    xor r12d, r12d
.Linner:
    mov rax, qword ptr [r9 + r12*8]
    mul rbx
    add rax, r10
    adc rdx, 0
    add rax, qword ptr [r8 + r12*8]
    adc rdx, 0
    mov qword ptr [r8 + r12*8], rax
    mov r10, rdx
    inc r12
    cmp r12, rcx
    jb .Linner
    mov qword ptr [r8 + rcx*8], r10
    add r8, 8
    inc r11
    cmp r11, rsi
    jb .Louter
    mov rax, rsi
    shl rax, 3
    sub r8, rax
    lea rax, [rsi + rcx]
.Lc:
    test rax, rax
    jz .Ldone
    cmp qword ptr [r8 + rax*8 - 8], 0
    jne .Ldone
    dec rax
    jmp .Lc
.Lret0:
    xor eax, eax
.Ldone:
    pop r12
    pop rbx
    ret
