export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @next %i2
	%o =l phi @start 0, @next %o2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%c1 =w csgtl %rem, 1
	jnz %c1, @l1, @s1
@l1
	%p1 =l add %p0, 1
	%x1 =l loadub %p1
	jmp @m1
@s1
	jmp @m1
@m1
	%b1 =l phi @l1 %x1, @s1 0
	%c2 =w csgtl %rem, 2
	jnz %c2, @l2, @s2
@l2
	%p2 =l add %p0, 2
	%x2 =l loadub %p2
	jmp @m2
@s2
	jmp @m2
@m2
	%b2 =l phi @l2 %x2, @s2 0
	%s0 =l shl %b0, 16
	%s1v =l shl %b1, 8
	%va =l or %s0, %s1v
	%val =l or %va, %b2
	jmp @inner
@inner
	%j =l phi @m2 0, @store %j2
	%t =l mul %j, 6
	%sh =l sub 18, %t
	%sv =l shr %val, %sh
	%v =l and %sv, 63
	%a =l add %v, 65
	%g26 =w csgel %v, 26
	%e26 =l extuw %g26
	%m26 =l mul %e26, 6
	%a1 =l add %a, %m26
	%g52 =w csgel %v, 52
	%e52 =l extuw %g52
	%m52 =l mul %e52, 75
	%a2 =l sub %a1, %m52
	%g62 =w csgel %v, 62
	%e62 =l extuw %g62
	%m62 =l mul %e62, 15
	%a3 =l sub %a2, %m62
	%g63 =w ceql %v, 63
	%e63 =l extuw %g63
	%m63 =l mul %e63, 3
	%ch0 =l add %a3, %m63
	%pc =w csgtl %j, %rem
	jnz %pc, @pad, @enc
@enc
	jmp @store
@pad
	jmp @store
@store
	%ch =l phi @enc %ch0, @pad 61
	%q =l add %o, %j
	%q2 =l add %out, %q
	storeb %ch, %q2
	%j2 =l add %j, 1
	%cj =w csltl %j2, 4
	jnz %cj, @inner, @next
@next
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @outer
@done
	ret %o
}
