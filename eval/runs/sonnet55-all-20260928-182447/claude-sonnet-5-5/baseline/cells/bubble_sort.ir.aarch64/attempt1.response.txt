export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l sub %n, 1
	jmp @outer
@outer
	%i =l phi @start %i0, @inner_end %i2
	%c =w csgtl %i, 0
	jnz %c, @inner_init, @done
@inner_init
	jmp @inner
@inner
	%j =l phi @inner_init 0, @next %j2
	%c2 =w csltl %j, %i
	jnz %c2, @body, @inner_end
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%g =w csgtl %x, %y
	jnz %g, @swap, @next
@swap
	storel %y, %p
	storel %x, %q
	jmp @next
@next
	%j2 =l add %j, 1
	jmp @inner
@inner_end
	%i2 =l sub %i, 1
	jmp @outer
@done
	ret 0
}
