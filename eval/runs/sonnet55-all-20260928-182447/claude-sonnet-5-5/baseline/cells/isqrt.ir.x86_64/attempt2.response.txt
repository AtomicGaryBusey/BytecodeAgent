export function l $isqrt(l %n0) {
@start
	jmp @loop
@loop
	%n =l phi @start %n0, @next %n2
	%res =l phi @start 0, @next %res2
	%bit =l phi @start 4611686018427387904, @next %bit2
	%t =l add %res, %bit
	%c =w cultl %n, %t
	jnz %c, @no, @yes
@yes
	%ny =l sub %n, %t
	%r1 =l shr %res, 1
	%ry =l add %r1, %bit
	jmp @next
@no
	%rn =l shr %res, 1
	jmp @next
@next
	%n2 =l phi @yes %ny, @no %n
	%res2 =l phi @yes %ry, @no %rn
	%bit2 =l shr %bit, 2
	%z =w cnel %bit2, 0
	jnz %z, @loop, @end
@end
	ret %res2
}
