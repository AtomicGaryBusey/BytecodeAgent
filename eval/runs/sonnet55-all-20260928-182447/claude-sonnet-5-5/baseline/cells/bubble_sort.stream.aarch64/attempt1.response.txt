bubble_sort:
    cmp x1, #2
    b.lt .Ldone
    sub x9, x1, #1
.Louter:
    mov x10, #0
    mov x13, #0
.Linner:
    cmp x10, x9
    b.ge .Lendinner
    ldr x11, [x0, x10, lsl #3]
    add x14, x10, #1
    ldr x12, [x0, x14, lsl #3]
    cmp x11, x12
    b.le .Lnoswap
    str x12, [x0, x10, lsl #3]
    str x11, [x0, x14, lsl #3]
    mov x13, #1
.Lnoswap:
    mov x10, x14
    b .Linner
.Lendinner:
    cbz x13, .Ldone
    sub x9, x9, #1
    cbnz x9, .Louter
.Ldone:
    mov x0, #0
    ret
