export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %k2
	%best =l phi @start 0, @next %best2
	%c =w csgtl %k, %n
	jnz %c, @done, @inner_init
@inner_init
	jmp @loop
@loop
	%x =l phi @inner_init %k, @step %x2
	%s =l phi @inner_init 0, @step %s2
	%e =w ceql %x, 1
	jnz %e, @fin, @step
@step
	%b =l and %x, 1
	%h =l shr %x, 1
	%t =l mul %x, 3
	%t1 =l add %t, 1
	%o =w cnel %b, 0
	%x2 =l phi @odd %t1, @even %h
	%s2 =l add %s, 1
	jmp @loop
@odd
	jmp @loop
@even
	jmp @loop
@fin
	%gt =w csgtl %s, %best
	jnz %gt, @upd, @keep
@upd
	jmp @next2
@keep
	jmp @next2
@next2
	%best2 =l phi @upd %s, @keep %best
	%k2 =l add %k, 1
	jmp @next
@next
	jmp @outer
@done
	ret %best
}
