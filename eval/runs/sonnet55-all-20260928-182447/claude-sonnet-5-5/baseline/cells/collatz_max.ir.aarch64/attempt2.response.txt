export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %k2
	%best =l phi @start 0, @next %best2
	%c =w csgtl %k, %n
	jnz %c, @done, @loop
@loop
	%x =l phi @outer %k, @cont %x2
	%s =l phi @outer 0, @cont %s2
	%e =w ceql %x, 1
	jnz %e, @fin, @step
@step
	%b =l and %x, 1
	%o =w cnel %b, 0
	jnz %o, @odd, @even
@odd
	%t =l mul %x, 3
	%t1 =l add %t, 1
	jmp @cont
@even
	%h =l shr %x, 1
	jmp @cont
@cont
	%x2 =l phi @odd %t1, @even %h
	%s2 =l add %s, 1
	jmp @loop
@fin
	%gt =w csgtl %s, %best
	jnz %gt, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%best2 =l phi @upd %s, @keep %best
	%k2 =l add %k, 1
	jmp @outer
@done
	ret %best
}
