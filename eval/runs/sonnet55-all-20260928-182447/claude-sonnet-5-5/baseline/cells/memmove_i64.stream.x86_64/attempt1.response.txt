memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    jbe .Lfwd
    lea rcx, [rsi + rdx*8]
    cmp rdi, rcx
    jb .Lbwd
.Lfwd:
    xor ecx, ecx
.Lf1:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jl .Lf1
    ret
.Lbwd:
    mov r8, qword ptr [rsi + rdx*8 - 8]
    mov qword ptr [rdi + rdx*8 - 8], r8
    dec rdx
    jnz .Lbwd
.Ldone:
    ret
