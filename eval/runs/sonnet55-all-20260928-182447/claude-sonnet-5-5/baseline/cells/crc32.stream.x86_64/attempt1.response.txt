    mov r8d, 0xEDB88320
    xor rcx, rcx
.Lbyte:
    cmp rcx, rsi
    jge .Ldone
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r9d, 8
.Lbit:

    shr eax, 1
    test edx, 1
    jz .Lskip
    xor eax, r8d
.Lskip:

    jnz .Lbit
    inc rcx
    jmp .Lbyte
.Ldone:

    ret
