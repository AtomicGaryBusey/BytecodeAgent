export function l $bubble_sort(l %a, l %n) {
@start
  %pa =l alloc8 8
  storel %a, %pa
  %pn =l alloc8 8
  storel %n, %pn
  %pi =l alloc8 8
  %pj =l alloc8 8
  %pt =l alloc8 8
  storel 0, %pi
  jmp @outer_cond
@outer_cond
  %i_oc =l loadl %pi
  %n_oc =l loadl %pn
  %nm1_oc =l sub %n_oc, 1
  %c_oc =w csltl %i_oc, %nm1_oc
  jnz %c_oc, @inner_init, @done
@inner_init
  storel 0, %pj
  jmp @inner_cond
@inner_cond
  %j_ic =l loadl %pj
  %i_ic =l loadl %pi
  %n_ic =l loadl %pn
  %lim_ic =l sub %n_ic, %i_ic
  %lim1_ic =l sub %lim_ic, 1
  %c_ic =w csltl %j_ic, %lim1_ic
  jnz %c_ic, @inner_body, @outer_inc
@inner_body
  %j_ib =l loadl %pj
  %a_ib =l loadl %pa
  %off_ib =l mul %j_ib, 8
  %ptr_ib =l add %a_ib, %off_ib
  %vj_ib =l loadl %ptr_ib
  %j1_ib =l add %j_ib, 1
  %off1_ib =l mul %j1_ib, 8
  %ptr1_ib =l add %a_ib, %off1_ib
  %vj1_ib =l loadl %ptr1_ib
  %c_ib =w csltl %vj1_ib, %vj_ib
  jnz %c_ib, @swap, @inner_inc
@swap
  %j_sw =l loadl %pj
  %a_sw =l loadl %pa
  %off_sw =l mul %j_sw, 8
  %ptr_sw =l add %a_sw, %off_sw
  %vj_sw =l loadl %ptr_sw
  %j1_sw =l add %j_sw, 1
  %off1_sw =l mul %j1_sw, 8
  %ptr1_sw =l add %a_sw, %off1_sw
  %vj1_sw =l loadl %ptr1_sw
  storel %vj_sw, %pt
  storel %vj1_sw, %ptr_sw
  %tmp_sw =l loadl %pt
  storel %tmp_sw, %ptr1_sw
  jmp @inner_inc
@inner_inc
  %j_ii =l loadl %pj
  %j1_ii =l add %j_ii, 1
  storel %j1_ii, %pj
  jmp @inner_cond
@outer_inc
  %i_oi =l loadl %pi
  %i1_oi =l add %i_oi, 1
  storel %i1_oi, %pi
  jmp @outer_cond
@done
  ret 0
}
