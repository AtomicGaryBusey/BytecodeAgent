export function l $count_nonzero_prefix(l %a, l %n) {
@start
  %i0 =l copy 0
  %p0 =l copy %a
  jmp @loop(%i0, %p0)
@loop(%i =l, %p =l)
  %c =w csltl %i, %n
  jnz %c, @check, @done
@check
  %v =l loadl %p
  %z =w ceql %v, 0
  jnz %z, @done, @next
@next
  %i1 =l add %i, 1
  %p1 =l add %p, 8
  jmp @loop(%i1, %p1)
@done
  ret %i
}
