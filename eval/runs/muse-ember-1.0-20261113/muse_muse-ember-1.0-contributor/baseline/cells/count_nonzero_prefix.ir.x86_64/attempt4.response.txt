export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @incr %i1
	%done =w ceql %i, %n
	jnz %done, @ret, @load
@load
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @ret, @incr
@incr
	%i1 =l add %i, 1
	jmp @loop
@ret
	ret %i
}
