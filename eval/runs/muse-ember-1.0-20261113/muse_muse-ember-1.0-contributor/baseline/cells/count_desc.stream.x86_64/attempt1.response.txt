count_desc:
    cmp rsi, 2
    jl .Lret0
    xor eax, eax
    xor ecx, ecx
    mov rdx, rsi
    dec rdx
.Lloop:
    cmp rcx, rdx
    jge .Lret
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, rcx
    inc r9
    mov r10, qword ptr [rdi + r9*8]
    cmp r8, r10
    jle .Lnext
    inc rax
.Lnext:
    inc rcx
    jmp .Lloop
.Lret0:
    xor eax, eax
.Lret:
    ret
