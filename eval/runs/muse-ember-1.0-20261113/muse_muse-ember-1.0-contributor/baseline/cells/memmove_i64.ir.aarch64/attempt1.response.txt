export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
  %lt =w csltl %dst, %src
  jnz %lt, @forward_init, @entry2
@entry2
  %eq =w ceql %dst, %src
  jnz %eq, @forward_init, @backward_init
@forward_init
  jmp @forward_loop
@forward_loop
  %dptr =l phi @forward_init %dst, @forward_body %dnext
  %sptr =l phi @forward_init %src, @forward_body %snext
  %rem =l phi @forward_init %n, @forward_body %rnext
  %cond =w cnel %rem, 0
  jnz %cond, @forward_body, @done
@forward_body
  %v =l loadl %sptr
  storel %v, %dptr
  %dnext =l add %dptr, 8
  %snext =l add %sptr, 8
  %rnext =l sub %rem, 1
  jmp @forward_loop
@backward_init
  %nbytes =l mul %n, 8
  %dend =l add %dst, %nbytes
  %send =l add %src, %nbytes
  jmp @backward_loop
@backward_loop
  %dptr2 =l phi @backward_init %dend, @backward_body %dnext2
  %sptr2 =l phi @backward_init %send, @backward_body %snext2
  %rem2 =l phi @backward_init %n, @backward_body %rnext2
  %cond2 =w cnel %rem2, 0
  jnz %cond2, @backward_body, @done
@backward_body
  %dnext2 =l sub %dptr2, 8
  %snext2 =l sub %sptr2, 8
  %v2 =l loadl %snext2
  storel %v2, %dnext2
  %rnext2 =l sub %rem2, 1
  jmp @backward_loop
@done
  ret 0
}
