export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %a0 =l loadl %a
  %y0 =l call %f(l %a0)
  storel %y0, %out
  %out1 =l add %out, 8
  storel %y0, %out1
  %out2 =l add %out, 16
  storel %y0, %out2
  %aptr0 =l add %a, 8
  %odd0 =l and %y0, 1
  %isodd0 =w cnel %odd0, 0
  jnz %isodd0, @cnt1, @cnt0
@cnt1
  jmp @loop
@cnt0
  jmp @loop
@loop
  %i =l phi @cnt1 1, @cnt0 1, @next %i_next
  %aptr =l phi @cnt1 %aptr0, @cnt0 %aptr0, @next %aptr_next
  %cnt =l phi @cnt1 1, @cnt0 0, @next %cnt_next
  %more =w csltl %i, %n
  jnz %more, @body, @done
@body
  %av =l loadl %aptr
  %y =l call %f(l %av)
  %min =l loadl %out
  %lt =w csltl %y, %min
  jnz %lt, @setmin, @chkmax
@setmin
  storel %y, %out
  jmp @chkmax
@chkmax
  %max =l loadl %out1
  %gt =w csgtl %y, %max
  jnz %gt, @setmax, @chksum
@setmax
  storel %y, %out1
  jmp @chksum
@chksum
  %sum =l loadl %out2
  %sum_next =l add %sum, %y
  storel %sum_next, %out2
  %i_next =l add %i, 1
  %aptr_next =l add %aptr, 8
  %odd =l and %y, 1
  %isodd =w cnel %odd, 0
  jnz %isodd, @cnt_inc, @cnt_keep
@cnt_inc
  %cnt_inc_v =l add %cnt, 1
  jmp @next
@cnt_keep
  jmp @next
@next
  %cnt_next =l phi @cnt_inc %cnt_inc_v, @cnt_keep %cnt
  jmp @loop
@done
  ret %cnt
}
