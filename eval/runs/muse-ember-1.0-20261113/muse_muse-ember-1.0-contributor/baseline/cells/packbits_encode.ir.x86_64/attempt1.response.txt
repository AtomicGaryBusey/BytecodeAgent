export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %m1 =l copy -1
  %c257 =w copy 257
  jmp @loop
@loop
  %i =l phi @start 0, @repeat_store %i_rep, @lit_copy_done %i_lit
  %written =l phi @start 0, @repeat_store %written_rep, @lit_copy_done %written_lit
  %c0 =w csltl %i, %n
  jnz %c0, @compute_run, @done
@done
  ret %written
@fail
  ret %m1
@compute_run
  %p_i =l add %src, %i
  %first =w loadub %p_i
  %j0 =l add %i, 1
  jmp @run_loop
@run_loop
  %r =w phi @compute_run 1, @run_body %r2
  %j =l phi @compute_run %j0, @run_body %j2
  %cR =w csltw %r, 128
  jnz %cR, @run_check_n, @run_done
@run_check_n
  %cN =w csltl %j, %n
  jnz %cN, @run_check_byte, @run_done
@run_check_byte
  %p_j =l add %src, %j
  %b_j =w loadub %p_j
  %cE =w ceqw %b_j, %first
  jnz %cE, @run_body, @run_done
@run_body
  %r2 =w add %r, 1
  %j2 =l add %j, 1
  jmp @run_loop
@run_done
  %c3 =w csltw %r, 3
  jnz %c3, @literal_init, @repeat
@repeat
  %needR =l add %written, 2
  %cFailR =w csltl %cap, %needR
  jnz %cFailR, @fail, @repeat_store
@repeat_store
  %hdrR =w sub %c257, %r
  %po =l add %out, %written
  storeb %hdrR, %po
  %po2 =l add %po, 1
  storeb %first, %po2
  %written_rep =l add %written, 2
  %i_rep =l copy %j
  jmp @loop
@literal_init
  %j0l =l copy %i
  jmp @lit_loop
@lit_loop
  %j_lit =l phi @literal_init %j0l, @lit_advance %j_adv
  %lenL =w phi @literal_init 0, @lit_advance %len_adv
  %len_l =l phi @literal_init 0, @lit_advance %len_l_adv
  %cLen =w csltw %lenL, 128
  jnz %cLen, @lit_check_n, @lit_emit
@lit_check_n
  %cNl =w csltl %j_lit, %n
  jnz %cNl, @lit_run_init, @lit_emit
@lit_run_init
  %p_jl =l add %src, %j_lit
  %first_j =w loadub %p_jl
  %j_run0 =l add %j_lit, 1
  jmp @lit_run_loop
@lit_run_loop
  %r_j =w phi @lit_run_init 1, @lit_run_body %r_j2
  %j_run =l phi @lit_run_init %j_run0, @lit_run_body %j_run2
  %cRj =w csltw %r_j, 128
  jnz %cRj, @lit_run_check_n, @lit_run_done
@lit_run_check_n
  %cNj =w csltl %j_run, %n
  jnz %cNj, @lit_run_check_byte, @lit_run_done
@lit_run_check_byte
  %p_jr =l add %src, %j_run
  %b_jr =w loadub %p_jr
  %cEj =w ceqw %b_jr, %first_j
  jnz %cEj, @lit_run_body, @lit_run_done
@lit_run_body
  %r_j2 =w add %r_j, 1
  %j_run2 =l add %j_run, 1
  jmp @lit_run_loop
@lit_run_done
  %c3j =w csltw %r_j, 3
  jnz %c3j, @lit_advance, @lit_emit
@lit_advance
  %j_adv =l add %j_lit, 1
  %len_adv =w add %lenL, 1
  %len_l_adv =l add %len_l, 1
  jmp @lit_loop
@lit_emit
  %need0L =l add %written, %len_l
  %needL =l add %need0L, 1
  %cFailL =w csltl %cap, %needL
  jnz %cFailL, @fail, @lit_emit_store
@lit_emit_store
  %hdrL =w sub %lenL, 1
  %poL =l add %out, %written
  storeb %hdrL, %poL
  jmp @lit_copy_loop
@lit_copy_loop
  %k =l phi @lit_emit_store 0, @lit_copy_body %k2
  %cK =w csltl %k, %len_l
  jnz %cK, @lit_copy_body, @lit_copy_done
@lit_copy_body
  %src_off =l add %i, %k
  %src_ptr =l add %src, %src_off
  %b_c =w loadub %src_ptr
  %dst_off0 =l add %written, 1
  %dst_off =l add %dst_off0, %k
  %dst_ptr =l add %out, %dst_off
  storeb %b_c, %dst_ptr
  %k2 =l add %k, 1
  jmp @lit_copy_loop
@lit_copy_done
  %i_lit =l copy %j_lit
  %written_lit =l copy %needL
  jmp @loop
}
