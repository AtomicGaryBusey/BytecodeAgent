export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1
	%p0hi =l shr %p0, 32
	%p1hi =l shr %p1, 32
	%p1lo =l and %p1, 4294967295
	%p2hi =l shr %p2, 32
	%p2lo =l and %p2, 4294967295
	%low1 =l add %p1lo, %p2lo
	%low2 =l add %low1, %p0hi
	%carry =l shr %low2, 32
	%mid1 =l add %p1hi, %p2hi
	%mid2 =l add %mid1, %carry
	%res =l add %p3, %mid2
	ret %res
}
