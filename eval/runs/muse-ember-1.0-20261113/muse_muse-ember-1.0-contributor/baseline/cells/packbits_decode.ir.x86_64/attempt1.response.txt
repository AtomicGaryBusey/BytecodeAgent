export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  jmp @loop
@loop
  %i =l phi @start 0, @noop_done %i_noop, @lit_done %i_lit, @rep_done %i_rep
  %o =l phi @start 0, @noop_done %o_noop, @lit_done %o_lit, @rep_done %o_rep
  %loop_cond =w csltl %i, %n
  jnz %loop_cond, @body, @finished
@finished
  ret %o
@body
  %hdr_ptr =l add %src, %i
  %h =l loadsb %hdr_ptr
  %is_noop =w ceql %h, -128
  jnz %is_noop, @noop_done, @not_noop
@noop_done
  %i_noop =l add %i, 1
  %o_noop =l copy %o
  jmp @loop
@not_noop
  %is_neg =w csltl %h, 0
  jnz %is_neg, @rep_start, @lit_start
@lit_start
  %len =l add %h, 1
  %need_lit =l add %i, %len
  %need_lit2 =l add %need_lit, 1
  %src_err_lit =w csltl %n, %need_lit2
  jnz %src_err_lit, @err_src, @lit_cap
@err_src
  %r_src =l copy -2
  ret %r_src
@lit_cap
  %tot_lit =l add %o, %len
  %over_lit =w csltl %cap, %tot_lit
  jnz %over_lit, @lit_overflow, @lit_copy
@lit_copy
  %k_lit =l phi @lit_cap 0, @lit_copy_body %k_lit_next
  %k_lit_cond =w csltl %k_lit, %len
  jnz %k_lit_cond, @lit_copy_body, @lit_done
@lit_copy_body
  %src_base_lit =l add %src, %i
  %src_off_lit =l add %src_base_lit, 1
  %src_ptr_lit =l add %src_off_lit, %k_lit
  %b_lit =w loadub %src_ptr_lit
  %out_base_lit =l add %out, %o
  %out_ptr_lit =l add %out_base_lit, %k_lit
  storeb %b_lit, %out_ptr_lit
  %k_lit_next =l add %k_lit, 1
  jmp @lit_copy
@lit_done
  %i_lit =l add %i, %need_lit2
  %o_lit =l add %o, %len
  jmp @loop
@lit_overflow
  %k_lit_p =l phi @lit_cap 0, @lit_part_body %k_lit_p_next
  %rem_lit =l sub %cap, %o
  %k_lit_p_cond =w csltl %k_lit_p, %rem_lit
  jnz %k_lit_p_cond, @lit_part_body, @overflow_ret
@lit_part_body
  %src_base_lp =l add %src, %i
  %src_off_lp =l add %src_base_lp, 1
  %src_ptr_lp =l add %src_off_lp, %k_lit_p
  %b_lp =w loadub %src_ptr_lp
  %out_base_lp =l add %out, %o
  %out_ptr_lp =l add %out_base_lp, %k_lit_p
  storeb %b_lp, %out_ptr_lp
  %k_lit_p_next =l add %k_lit_p, 1
  jmp @lit_overflow
@overflow_ret
  %r_over =l copy -1
  ret %r_over
@rep_start
  %zero_rep =l copy 0
  %neg_h =l sub %zero_rep, %h
  %cnt_rep =l add %neg_h, 1
  %need_rep =l add %i, 2
  %src_err_rep =w csltl %n, %need_rep
  jnz %src_err_rep, @err_src, @rep_cap
@rep_cap
  %rep_ptr =l add %src, %i
  %rep_off =l add %rep_ptr, 1
  %rep_byte =w loadub %rep_off
  %tot_rep =l add %o, %cnt_rep
  %over_rep =w csltl %cap, %tot_rep
  jnz %over_rep, @rep_overflow, @rep_copy
@rep_copy
  %k_rep =l phi @rep_cap 0, @rep_copy_body %k_rep_next
  %k_rep_cond =w csltl %k_rep, %cnt_rep
  jnz %k_rep_cond, @rep_copy_body, @rep_done
@rep_copy_body
  %out_base_rep =l add %out, %o
  %out_ptr_rep =l add %out_base_rep, %k_rep
  storeb %rep_byte, %out_ptr_rep
  %k_rep_next =l add %k_rep, 1
  jmp @rep_copy
@rep_done
  %i_rep =l add %i, 2
  %o_rep =l add %o, %cnt_rep
  jmp @loop
@rep_overflow
  %k_rep_p =l phi @rep_cap 0, @rep_part_body %k_rep_p_next
  %rem_rep =l sub %cap, %o
  %k_rep_p_cond =w csltl %k_rep_p, %rem_rep
  jnz %k_rep_p_cond, @rep_part_body, @overflow_ret
@rep_part_body
  %out_base_rp =l add %out, %o
  %out_ptr_rp =l add %out_base_rp, %k_rep_p
  storeb %rep_byte, %out_ptr_rp
  %k_rep_p_next =l add %k_rep_p, 1
  jmp @rep_overflow
}
