lcs_len:
mov r9, r8
mov r10, rcx
xor r11d, r11d
xor eax, eax
.Lzero:
cmp r11, r10
ja .Lzero_done
mov qword ptr [r9 + r11*8], rax
inc r11
jmp .Lzero
.Lzero_done:
test rsi, rsi
jle .Lret
.Louter:
xor eax, eax
movzx r8d, byte ptr [rdi]
inc rdi
xor ecx, ecx
.Linner:
cmp rcx, r10
jae .Linner_done
mov r11, qword ptr [r9 + rcx*8 + 8]
cmp r8b, byte ptr [rdx + rcx]
je .Lmatch
mov rax, r11
cmp r11, qword ptr [r9 + rcx*8]
jge .Lnext
mov r11, qword ptr [r9 + rcx*8]
mov qword ptr [r9 + rcx*8 + 8], r11
jmp .Lnext
.Lmatch:
lea r11, [rax + 1]
mov rax, qword ptr [r9 + rcx*8 + 8]
mov qword ptr [r9 + rcx*8 + 8], r11
.Lnext:
inc rcx
jmp .Linner
.Linner_done:
dec rsi
jnz .Louter
.Lret:
mov rax, qword ptr [r9 + r10*8]
ret
