export function l $reverse(l %a, l %n) {
@start
  %j0 =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @start 0, @swap %inext
  %j =l phi @start %j0, @swap %jnext
  %c =w csltl %i, %j
  jnz %c, @swap, @exit
@swap
  %oi =l mul %i, 8
  %oj =l mul %j, 8
  %pi =l add %a, %oi
  %pj =l add %a, %oj
  %vi =l loadl %pi
  %vj =l loadl %pj
  storel %vj, %pi
  storel %vi, %pj
  %inext =l add %i, 1
  %jnext =l sub %j, 1
  jmp @loop
@exit
  ret 0
}
