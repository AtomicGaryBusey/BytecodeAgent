export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  jmp @loop
@loop
  %i =l phi @start 0, @join %i_next
  %count =l phi @start 0, @join %count_next
  %cont =w csltl %i, %n
  jnz %cont, @body, @done
@body
  %off =l mul %i, 8
  %sptr =l add %src, %off
  %v =l loadl %sptr
  %r =l call %f(l %v)
  %dptr =l add %dst, %off
  storel %r, %dptr
  %less =w csltl %r, %v
  jnz %less, @inc, @noinc
@inc
  %count_inc =l add %count, 1
  jmp @join
@noinc
  %count_no =l copy %count
  jmp @join
@join
  %count_next =l phi @inc %count_inc, @noinc %count_no
  %i_next =l add %i, 1
  jmp @loop
@done
  ret %count
}
