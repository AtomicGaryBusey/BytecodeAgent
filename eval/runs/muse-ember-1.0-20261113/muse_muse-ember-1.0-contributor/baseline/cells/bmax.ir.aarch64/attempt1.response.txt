export function l $bmax(l %a, l %n) {
@entry
  %max0 =l loadl %a
  %i0 =l copy 1
  %one =l copy 1
  %eight =l copy 8
  jmp @loop
@loop
  %max =l phi @entry %max0, @cont %maxn
  %i =l phi @entry %i0, @cont %in
  %done =w csgel %i, %n
  jnz %done, @exit, @body
@body
  %off =l shl %i, %eight
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %gt =w csgtl %val, %max
  jnz %gt, @upd, @cont
@upd
  %maxu =l copy %val
  jmp @cont
@cont
  %maxn =l phi @body %max, @upd %maxu
  %in =l add %i, %one
  jmp @loop
@exit
  ret %max
}
