export function l $bmax(l %a, l %n) {
@entry
  %max0 =l loadl %a
  %p0 =l add %a, 8
  %off =l shl %n, 3
  %end =l add %a, %off
  jmp @loop
@loop
  %max =l phi @entry %max0, @upd %maxu, @skip %maxs
  %p =l phi @entry %p0, @upd %pnu, @skip %pns
  %done =w ceql %p, %end
  jnz %done, @exit, @body
@body
  %val =l loadl %p
  %gt =w csgtl %val, %max
  jnz %gt, @upd, @skip
@upd
  %maxu =l copy %val
  %pnu =l add %p, 8
  jmp @loop
@skip
  %maxs =l copy %max
  %pns =l add %p, 8
  jmp @loop
@exit
  ret %max
}
