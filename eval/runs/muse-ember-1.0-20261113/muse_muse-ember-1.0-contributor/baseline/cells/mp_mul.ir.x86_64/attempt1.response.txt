export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %zero =l and %r, 0
  %total =l add %na, %nb
  jmp @zero_head
@zero_head
  %zi =l phi @start 0, @zero_body %zi_next
  %zcond =w csltl %zi, %total
  jnz %zcond, @zero_body, @outer_init
@zero_body
  %zoff =l mul %zi, 8
  %zptr =l add %r, %zoff
  storel %zero, %zptr
  %zi_next =l add %zi, 1
  jmp @zero_head
@outer_init
  jmp @outer_head
@outer_head
  %oi =l phi @outer_init 0, @inner_exit %oi_next
  %ocond =w csltl %oi, %na
  jnz %ocond, @outer_body, @scan_init
@outer_body
  %aoff =l mul %oi, 8
  %aptr =l add %a, %aoff
  %ai0 =l loadl %aptr
  jmp @inner_head
@inner_head
  %ij =l phi @outer_body 0, @inner_body %ij_next
  %carry =l phi @outer_body 0, @inner_body %carry_next
  %icond =w csltl %ij, %nb
  jnz %icond, @inner_body, @inner_exit
@inner_body
  %idx =l add %oi, %ij
  %idxoff =l mul %idx, 8
  %rptr =l add %r, %idxoff
  %rold =l loadl %rptr
  %boff =l mul %ij, 8
  %bptr =l add %b, %boff
  %bj =l loadl %bptr
  %alo =l and %ai0, 4294967295
  %ahi =l shr %ai0, 32
  %blo =l and %bj, 4294967295
  %bhi =l shr %bj, 32
  %p1 =l mul %alo, %blo
  %p2 =l mul %alo, %bhi
  %p3 =l mul %ahi, %blo
  %p4 =l mul %ahi, %bhi
  %p1lo =l and %p1, 4294967295
  %p1hi =l shr %p1, 32
  %p2lo =l and %p2, 4294967295
  %p2hi =l shr %p2, 32
  %p3lo =l and %p3, 4294967295
  %p3hi =l shr %p3, 32
  %p4lo =l and %p4, 4294967295
  %p4hi =l shr %p4, 32
  %cylo =l and %carry, 4294967295
  %cyhi =l shr %carry, 32
  %rolo =l and %rold, 4294967295
  %rohi =l shr %rold, 32
  %s0a =l add %p1lo, %cylo
  %s0 =l add %s0a, %rolo
  %l0 =l and %s0, 4294967295
  %c0 =l shr %s0, 32
  %s1a =l add %p1hi, %p2lo
  %s1b =l add %s1a, %p3lo
  %s1c =l add %s1b, %cyhi
  %s1d =l add %s1c, %rohi
  %s1 =l add %s1d, %c0
  %l1 =l and %s1, 4294967295
  %c1 =l shr %s1, 32
  %s2a =l add %p4lo, %p2hi
  %s2b =l add %s2a, %p3hi
  %s2 =l add %s2b, %c1
  %l2 =l and %s2, 4294967295
  %c2 =l shr %s2, 32
  %s3 =l add %p4hi, %c2
  %l3 =l and %s3, 4294967295
  %l1sh =l shl %l1, 32
  %newr =l add %l0, %l1sh
  storel %newr, %rptr
  %l3sh =l shl %l3, 32
  %carry_next =l add %l2, %l3sh
  %ij_next =l add %ij, 1
  jmp @inner_head
@inner_exit
  %eidx =l add %oi, %nb
  %eoff =l mul %eidx, 8
  %eptr =l add %r, %eoff
  storel %carry, %eptr
  %oi_next =l add %oi, 1
  jmp @outer_head
@scan_init
  jmp @scan_head
@scan_head
  %k =l phi @scan_init %total, @scan_body %k_next
  %kis0 =w ceql %k, %zero
  jnz %kis0, @done, @scan_check
@scan_check
  %km1 =l sub %k, 1
  %koff =l mul %km1, 8
  %kptr =l add %r, %koff
  %kval =l loadl %kptr
  %kvis0 =w ceql %kval, %zero
  jnz %kvis0, @scan_body, @done
@scan_body
  %k_next =l sub %k, 1
  jmp @scan_head
@done
  %retv =l phi @scan_head %k, @scan_check %k
  ret %retv
}
