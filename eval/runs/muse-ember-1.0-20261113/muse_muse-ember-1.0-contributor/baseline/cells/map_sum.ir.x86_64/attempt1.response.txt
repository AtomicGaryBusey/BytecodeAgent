export function l $map_sum(l %a, l %n, l %f) {
@start
  %zero =l sub %n, %n
  jmp @loop
@loop
  %sum =l phi @start %zero, @body %sum2
  %i =l phi @start %zero, @body %i2
  %cond =w csltl %i, %n
  jnz %cond, @body, @exit
@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %r =l call %f(l %val)
  %sum2 =l add %sum, %r
  %i2 =l add %i, 1
  jmp @loop
@exit
  ret %sum
}
