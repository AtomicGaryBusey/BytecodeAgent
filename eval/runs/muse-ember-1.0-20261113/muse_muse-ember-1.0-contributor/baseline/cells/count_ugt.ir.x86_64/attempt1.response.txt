export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mull %i, 8
	%ptr =l addl %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnext =l addl %cnt, %gtl
	%inext =l addl %i, 1
	jmp @loop
@end
	%res =l phi @loop %cnt
	ret %res
}
