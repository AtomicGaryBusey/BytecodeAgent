export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cnext
	%ptr =l phi @start %a, @body %pnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnext =l add %cnt, %gtl
	%inext =l add %i, 1
	%pnext =l add %ptr, 8
	jmp @loop
@end
	%res =l phi @loop %cnt
	ret %res
}
