export function l $sum_to(l %n) {
@start
  %zero =l copy 0
  %one =l copy 1
  %sp =l alloc8 1
  %ip =l alloc8 1
  %c =w csltl %n, %one
  jnz %c, @ret0
@init
  storel %zero, %sp
  storel %one, %ip
  jmp @loop
@loop
  %i =l loadl %ip
  %c2 =w csltl %n, %i
  jnz %c2, @done
@body
  %s =l loadl %sp
  %s2 =l add %s, %i
  storel %s2, %sp
  %i2 =l add %i, %one
  storel %i2, %ip
  jmp @loop
@ret0
  ret %zero
@done
  %r =l loadl %sp
  ret %r
}
