export function l $sum_to(l %n) {
@start
  %zero =l copy 0
  %one =l copy 1
  %sp = alloc8 1
  %ip = alloc8 1
  %c = csltl %n, %one
  jnz %c, @ret0
@init
  storel %zero, %sp
  storel %one, %ip
  jmp @loop
@loop
  %i = loadl %ip
  %c2 = csltl %n, %i
  jnz %c2, @done
@body
  %s = loadl %sp
  %s2 =l add %s, %i
  storel %s2, %sp
  %i2 =l add %i, %one
  storel %i2, %ip
  jmp @loop
@ret0
  ret %zero
@done
  %r = loadl %sp
  ret %r
}
