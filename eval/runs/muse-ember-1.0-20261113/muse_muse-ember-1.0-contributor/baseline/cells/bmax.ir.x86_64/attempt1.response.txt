export function l $bmax(l %a, l %n) {
@start
  %ip =l alloc8 1
  %mp =l alloc8 1
  %first =l loadl %a
  storel %first, %mp
  storel 1, %ip
  jmp @loop
@loop
  %i =l loadl %ip
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %i2 =l loadl %ip
  %off =l mul %i2, 8
  %p =l add %a, %off
  %v =l loadl %p
  %max =l loadl %mp
  %gt =w csltl %max, %v
  jnz %gt, @update, @next
@update
  %i3 =l loadl %ip
  %off2 =l mul %i3, 8
  %p2 =l add %a, %off2
  %v2 =l loadl %p2
  storel %v2, %mp
  jmp @next
@next
  %i4 =l loadl %ip
  %i5 =l add %i4, 1
  storel %i5, %ip
  jmp @loop
@exit
  %res =l loadl %mp
  ret %res
}
