export function l $binary_search(l %a, l %n, l %key) {
@start
  jmp @loop
@loop
  %lo =l phi @start 0, @right %lo_r, @left %lo_l
  %hi =l phi @start %n, @right %hi_r, @left %hi_l
  %c =w csltl %lo, %hi
  jnz %c, @body, @done
@body
  %diff =l sub %hi, %lo
  %half =l div %diff, 2
  %mid =l add %lo, %half
  %off =l mul %mid, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %eq =w ceql %val, %key
  jnz %eq, @found, @cmp
@cmp
  %lt =w csltl %val, %key
  jnz %lt, @right, @left
@right
  %lo_r =l add %mid, 1
  %hi_r =l copy %hi
  jmp @loop
@left
  %hi_l =l copy %mid
  %lo_l =l copy %lo
  jmp @loop
@found
  ret %mid
@done
  ret -1
}
