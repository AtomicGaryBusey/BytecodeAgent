export function l $fib(l %n) {
@entry
  jmp @loop
@loop
  %i =l phi @entry 0, @body %i2
  %a =l phi @entry 0, @body %a2
  %b =l phi @entry 1, @body %b2
  %cond =w csltl %i, %n
  jnz %cond, @body, @exit
@body
  %s =l add %a, %b
  %a2 =l add %b, 0
  %b2 =l add %s, 0
  %i2 =l add %i, 1
  jmp @loop
@exit
  ret %a
}
