export function l $fib(l %n) {
@entry
  jmp @loop
@loop
  %i = phi @entry 0, @body %i2
  %a = phi @entry 0, @body %a2
  %b = phi @entry 1, @body %b2
  %cond =w csltl %i, %n
  jnz %cond, @body, @exit
@body
  %s =l add %a, %b
  %a2 =l copy %b
  %b2 =l copy %s
  %i2 =l add %i, 1
  jmp @loop
@exit
  ret %a
}
