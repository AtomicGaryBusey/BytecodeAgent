packbits_decode:
xor r8d, r8d
xor r9d, r9d
Lloop:
cmp r8, rsi
jge Ldone
movsx rax, byte ptr [rdi + r8]
inc r8
cmp rax, -128
je Lloop
test rax, rax
js Lrepeat
mov r10, rax
inc r10
mov rax, r8
add rax, r10
cmp rax, rsi
jg Lerr2
mov rax, r9
add rax, r10
cmp rax, rcx
jg Lcaplit
Llitcopy:
test r10, r10
jz Lloop
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r10
jmp Llitcopy
Lcaplit:
mov rax, rcx
sub rax, r9
Lcaplitloop:
test rax, rax
jz Lerr1
mov r10b, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], r10b
inc r8
inc r9
dec rax
jmp Lcaplitloop
Lrepeat:
mov r10, 1
sub r10, rax
mov rax, r8
inc rax
cmp rax, rsi
jg Lerr2
mov rax, r9
add rax, r10
cmp rax, rcx
jg Lcaprep
mov al, byte ptr [rdi + r8]
inc r8
Lrepfill:
test r10, r10
jz Lloop
mov byte ptr [rdx + r9], al
inc r9
dec r10
jmp Lrepfill
Lcaprep:
mov r10b, byte ptr [rdi + r8]
mov rax, rcx
sub rax, r9
Lcapreploop:
test rax, rax
jz Lerr1
mov byte ptr [rdx + r9], r10b
inc r9
dec rax
jmp Lcapreploop
Lerr1:
mov rax, -1
ret
Lerr2:
mov rax, -2
ret
Ldone:
mov rax, r9
ret
