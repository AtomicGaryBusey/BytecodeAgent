export function l $collatz_max(l %n) {
@entry
  jmp @outer
@outer
  %max =l phi @entry 0, @outer_next %max_next
  %k =l phi @entry 1, @outer_next %k_next
  %c =w cslel %k, %n
  jnz %c, @outer_body, @exit
@outer_body
  %x0 =l copy %k
  jmp @inner
@inner
  %x =l phi @outer_body %x0, @even %x_even, @odd %x_odd
  %steps =l phi @outer_body 0, @even %steps_even, @odd %steps_odd
  %c1 =w ceql %x, 1
  jnz %c1, @inner_done, @check_even
@check_even
  %r =l and %x, 1
  %c2 =w ceql %r, 0
  jnz %c2, @even, @odd
@even
  %x_even =l div %x, 2
  %steps_even =l add %steps, 1
  jmp @inner
@odd
  %t =l mul %x, 3
  %x_odd =l add %t, 1
  %steps_odd =l add %steps, 1
  jmp @inner
@inner_done
  %c3 =w csgtl %steps, %max
  jnz %c3, @take_steps, @take_max
@take_steps
  %max_steps =l copy %steps
  jmp @outer_next
@take_max
  %max_keep =l copy %max
  jmp @outer_next
@outer_next
  %max_next =l phi @take_steps %max_steps, @take_max %max_keep
  %k_next =l add %k, 1
  jmp @outer
@exit
  ret %max
}
