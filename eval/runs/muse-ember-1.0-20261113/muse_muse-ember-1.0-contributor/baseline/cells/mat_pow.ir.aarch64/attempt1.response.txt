export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %slot_n =l alloc8 8
  %slot_k =l alloc8 8
  %slot_nn =l alloc8 8
  %slot_a =l alloc8 8
  %slot_out =l alloc8 8
  %slot_tmp =l alloc8 8
  %slot_base =l alloc8 8
  %slot_prod =l alloc8 8
  %slot_i =l alloc8 8
  %slot_j =l alloc8 8
  %slot_t =l alloc8 8
  %slot_iter =l alloc8 8
  %slot_idx =l alloc8 8
  %slot_sum =l alloc8 8
  storel %n, %slot_n
  storel %k, %slot_k
  storel %a, %slot_a
  storel %out, %slot_out
  storel %tmp, %slot_tmp
  %s_nn =l mul %n, %n
  storel %s_nn, %slot_nn
  storel %tmp, %slot_base
  %s_tmpbytes =l mul %s_nn, 8
  %s_prodptr =l add %tmp, %s_tmpbytes
  storel %s_prodptr, %slot_prod
  storel 0, %slot_idx
  jmp @cb_head
@cb_head
  %cb_idx =l loadl %slot_idx
  %cb_nn =l loadl %slot_nn
  %cb_c =w csltl %cb_idx, %cb_nn
  jnz %cb_c, @cb_body, @cb_done
@cb_body
  %cb_aptr =l loadl %slot_a
  %cb_idxb =l loadl %slot_idx
  %cb_off =l mul %cb_idxb, 8
  %cb_sptr =l add %cb_aptr, %cb_off
  %cb_val =l loadl %cb_sptr
  %cb_bptr =l loadl %slot_base
  %cb_dptr =l add %cb_bptr, %cb_off
  storel %cb_val, %cb_dptr
  %cb_idxn =l add %cb_idxb, 1
  storel %cb_idxn, %slot_idx
  jmp @cb_head
@cb_done
  %cb_k =l loadl %slot_k
  %cb_c0 =w ceql %cb_k, 0
  jnz %cb_c0, @zo_init, @co_init
@zo_init
  storel 0, %slot_idx
  jmp @zo_head
@zo_head
  %zo_idx =l loadl %slot_idx
  %zo_nn =l loadl %slot_nn
  %zo_c =w csltl %zo_idx, %zo_nn
  jnz %zo_c, @zo_body, @zo_done
@zo_body
  %zo_optr =l loadl %slot_out
  %zo_idxb =l loadl %slot_idx
  %zo_off =l mul %zo_idxb, 8
  %zo_dptr =l add %zo_optr, %zo_off
  storel 0, %zo_dptr
  %zo_idxn =l add %zo_idxb, 1
  storel %zo_idxn, %slot_idx
  jmp @zo_head
@zo_done
  storel 0, %slot_i
  jmp @diag_head
@diag_head
  %dg_i =l loadl %slot_i
  %dg_n =l loadl %slot_n
  %dg_c =w csltl %dg_i, %dg_n
  jnz %dg_c, @diag_body, @diag_done
@diag_body
  %dg_i2 =l loadl %slot_i
  %dg_n2 =l loadl %slot_n
  %dg_pos =l mul %dg_i2, %dg_n2
  %dg_pos2 =l add %dg_pos, %dg_i2
  %dg_off =l mul %dg_pos2, 8
  %dg_optr =l loadl %slot_out
  %dg_dptr =l add %dg_optr, %dg_off
  storel 1, %dg_dptr
  %dg_inext =l add %dg_i2, 1
  storel %dg_inext, %slot_i
  jmp @diag_head
@diag_done
  jmp @cl_init
@co_init
  storel 0, %slot_idx
  jmp @co_head
@co_head
  %co_idx =l loadl %slot_idx
  %co_nn =l loadl %slot_nn
  %co_c =w csltl %co_idx, %co_nn
  jnz %co_c, @co_body, @co_done
@co_body
  %co_aptr =l loadl %slot_a
  %co_idxb =l loadl %slot_idx
  %co_off =l mul %co_idxb, 8
  %co_sptr =l add %co_aptr, %co_off
  %co_val =l loadl %co_sptr
  %co_optr =l loadl %slot_out
  %co_dptr =l add %co_optr, %co_off
  storel %co_val, %co_dptr
  %co_idxn =l add %co_idxb, 1
  storel %co_idxn, %slot_idx
  jmp @co_head
@co_done
  %co_k =l loadl %slot_k
  %co_c1 =w ceql %co_k, 1
  jnz %co_c1, @cl_init, @pow_init
@pow_init
  storel 1, %slot_iter
  jmp @pow_head
@pow_head
  %pw_iter =l loadl %slot_iter
  %pw_k =l loadl %slot_k
  %pw_c =w csltl %pw_iter, %pw_k
  jnz %pw_c, @zp_init, @cl_init
@zp_init
  storel 0, %slot_idx
  jmp @zp_head
@zp_head
  %zp_idx =l loadl %slot_idx
  %zp_nn =l loadl %slot_nn
  %zp_c =w csltl %zp_idx, %zp_nn
  jnz %zp_c, @zp_body, @zp_done
@zp_body
  %zp_pptr =l loadl %slot_prod
  %zp_idxb =l loadl %slot_idx
  %zp_off =l mul %zp_idxb, 8
  %zp_dptr =l add %zp_pptr, %zp_off
  storel 0, %zp_dptr
  %zp_idxn =l add %zp_idxb, 1
  storel %zp_idxn, %slot_idx
  jmp @zp_head
@zp_done
  storel 0, %slot_i
  jmp @i_head
@i_head
  %i_i =l loadl %slot_i
  %i_n =l loadl %slot_n
  %i_c =w csltl %i_i, %i_n
  jnz %i_c, @j_init, @i_done
@j_init
  storel 0, %slot_j
  jmp @j_head
@j_head
  %j_j =l loadl %slot_j
  %j_n =l loadl %slot_n
  %j_c =w csltl %j_j, %j_n
  jnz %j_c, @t_init, @j_done
@t_init
  storel 0, %slot_t
  storel 0, %slot_sum
  jmp @t_head
@t_head
  %t_t =l loadl %slot_t
  %t_n =l loadl %slot_n
  %t_c =w csltl %t_t, %t_n
  jnz %t_c, @t_body, @t_done
@t_body
  %tb_i =l loadl %slot_i
  %tb_n =l loadl %slot_n
  %tb_t =l loadl %slot_t
  %tb_j =l loadl %slot_j
  %tb_poso =l mul %tb_i, %tb_n
  %tb_poso2 =l add %tb_poso, %tb_t
  %tb_offo =l mul %tb_poso2, 8
  %tb_optr =l loadl %slot_out
  %tb_sptr =l add %tb_optr, %tb_offo
  %tb_aval =l loadl %tb_sptr
  %tb_posb =l mul %tb_t, %tb_n
  %tb_posb2 =l add %tb_posb, %tb_j
  %tb_offb =l mul %tb_posb2, 8
  %tb_bptr =l loadl %slot_base
  %tb_dptr =l add %tb_bptr, %tb_offb
  %tb_bval =l loadl %tb_dptr
  %tb_prod =l mul %tb_aval, %tb_bval
  %tb_sumcur =l loadl %slot_sum
  %tb_sumnew =l add %tb_sumcur, %tb_prod
  storel %tb_sumnew, %slot_sum
  %tb_tnext =l add %tb_t, 1
  storel %tb_tnext, %slot_t
  jmp @t_head
@t_done
  %td_i =l loadl %slot_i
  %td_j =l loadl %slot_j
  %td_n =l loadl %slot_n
  %td_pos =l mul %td_i, %td_n
  %td_pos2 =l add %td_pos, %td_j
  %td_off =l mul %td_pos2, 8
  %td_pptr =l loadl %slot_prod
  %td_dptr =l add %td_pptr, %td_off
  %td_sum =l loadl %slot_sum
  storel %td_sum, %td_dptr
  %td_jnext =l add %td_j, 1
  storel %td_jnext, %slot_j
  jmp @j_head
@j_done
  %jd_i =l loadl %slot_i
  %jd_inext =l add %jd_i, 1
  storel %jd_inext, %slot_i
  jmp @i_head
@i_done
  storel 0, %slot_idx
  jmp @cp_head
@cp_head
  %cp_idx =l loadl %slot_idx
  %cp_nn =l loadl %slot_nn
  %cp_c =w csltl %cp_idx, %cp_nn
  jnz %cp_c, @cp_body, @cp_done
@cp_body
  %cp_pptr =l loadl %slot_prod
  %cp_idxb =l loadl %slot_idx
  %cp_off =l mul %cp_idxb, 8
  %cp_sptr =l add %cp_pptr, %cp_off
  %cp_val =l loadl %cp_sptr
  %cp_optr =l loadl %slot_out
  %cp_dptr =l add %cp_optr, %cp_off
  storel %cp_val, %cp_dptr
  %cp_idxn =l add %cp_idxb, 1
  storel %cp_idxn, %slot_idx
  jmp @cp_head
@cp_done
  %cpd_iter =l loadl %slot_iter
  %cpd_inext =l add %cpd_iter, 1
  storel %cpd_inext, %slot_iter
  jmp @pow_head
@cl_init
  storel 0, %slot_idx
  jmp @cl_head
@cl_head
  %cl_idx =l loadl %slot_idx
  %cl_nn =l loadl %slot_nn
  %cl_lim =l mul %cl_nn, 2
  %cl_c =w csltl %cl_idx, %cl_lim
  jnz %cl_c, @cl_body, @cl_done
@cl_body
  %cl_tptr =l loadl %slot_tmp
  %cl_idxb =l loadl %slot_idx
  %cl_off =l mul %cl_idxb, 8
  %cl_dptr =l add %cl_tptr, %cl_off
  storel 0, %cl_dptr
  %cl_idxn =l add %cl_idxb, 1
  storel %cl_idxn, %slot_idx
  jmp @cl_head
@cl_done
  ret 0
}
