export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %np1 =l add %n, 1
  jmp @init
@init
  %j0 =l phi @start 0, @init_body %j0n
  %c0 =w csltl %j0, %np1
  jnz %c0, @init_body, @outer
@init_body
  %o0 =l mul %j0, 8
  %p0 =l add %row, %o0
  storel 0, %p0
  %j0n =l add %j0, 1
  jmp @init
@outer
  %i0 =l phi @init 0, @outer_inc %i0n
  %co =w csltl %i0, %m
  jnz %co, @inner, @done
@inner
  %j1 =l phi @outer 1, @inner_merge %j1n
  %d0 =l phi @outer 0, @inner_merge %up
  %c1 =w csltl %j1, %np1
  jnz %c1, @inner_body, @outer_inc
@inner_body
  %o1 =l mul %j1, 8
  %p1 =l add %row, %o1
  %up =l loadl %p1
  %j1m1 =l sub %j1, 1
  %o2 =l mul %j1m1, 8
  %p2 =l add %row, %o2
  %left =l loadl %p2
  %xp =l add %x, %i0
  %xb =w loadub %xp
  %yp =l add %y, %j1m1
  %yb =w loadub %yp
  %eq =w ceqw %xb, %yb
  %j1n =l add %j1, 1
  jnz %eq, @inner_then, @inner_max
@inner_then
  %nt =l add %d0, 1
  storel %nt, %p1
  jmp @inner_merge
@inner_max
  %lt =w csltl %up, %left
  jnz %lt, @inner_store_left, @inner_store_up
@inner_store_left
  storel %left, %p1
  jmp @inner_merge
@inner_store_up
  storel %up, %p1
  jmp @inner_merge
@inner_merge
  jmp @inner
@outer_inc
  %i0n =l add %i0, 1
  jmp @outer
@done
  %on =l mul %n, 8
  %pn =l add %row, %on
  %res =l loadl %pn
  ret %res
}
