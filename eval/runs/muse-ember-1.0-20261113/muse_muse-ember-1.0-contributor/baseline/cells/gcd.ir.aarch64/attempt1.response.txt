export function l $gcd(l %a, l %b) {
@start
	%pa =l alloc8
	%pb =l alloc8
	storel %a, %pa
	storel %b, %pb
	jmp @loop
@loop
	%b1 =l loadl %pb
	%c =w ceql %b1, 0
	jnz %c, @done, @body
@body
	%a1 =l loadl %pa
	%r =l rem %a1, %b1
	storel %b1, %pa
	storel %r, %pb
	jmp @loop
@done
	%ret =l loadl %pa
	ret %ret
}
