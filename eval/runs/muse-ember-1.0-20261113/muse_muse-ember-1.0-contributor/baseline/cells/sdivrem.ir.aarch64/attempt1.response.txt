export function l $sdivrem(l %a, l %b, l %remp) {
@start
  %iszero =w ceql %b, 0
  jnz %iszero, @divzero, @checkneg1
@divzero
  storel %a, %remp
  %qzero =l copy 0
  ret %qzero
@checkneg1
  %isneg1 =w ceql %b, -1
  jnz %isneg1, @checkzeroa, @normal
@checkzeroa
  %iszeroa =w ceql %a, 0
  jnz %iszeroa, @normal, @checkneg
@checkneg
  %nega =l sub 0, %a
  %ismin =w ceql %nega, %a
  jnz %ismin, @overflow, @normal
@overflow
  %rzero =l copy 0
  storel %rzero, %remp
  ret %a
@normal
  %q =l div %a, %b
  %r =l rem %a, %b
  storel %r, %remp
  ret %q
}
