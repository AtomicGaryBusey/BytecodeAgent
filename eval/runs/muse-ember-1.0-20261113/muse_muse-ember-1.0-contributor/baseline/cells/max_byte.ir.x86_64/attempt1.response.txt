export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @merge %i1
	%max =l phi @start 0, @merge %max1
	%c =w csltl %i, %n
	jnz %c, @body, @exit
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%cmp =w csltl %max, %b
	jnz %cmp, @update, @merge
@update
	jmp @merge
@merge
	%max1 =l phi @body %max, @update %b
	%i1 =l add %i, 1
	jmp @loop
@exit
	ret %max
}
