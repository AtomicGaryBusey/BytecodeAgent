export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo0 =l phi @start 0, @updatelo %lo1, @updatehi %lo2
	%hi0 =l phi @start 4294967296, @updatelo %hi1, @updatehi %hi2
	%diff =l sub %hi0, %lo0
	%c0 =w ceql %diff, 1
	jnz %c0, @exit
	%half =l div %diff, 2
	%mid =l add %lo0, %half
	%sq =l mul %mid, %mid
	%c1 =w culel %sq, %n
	jnz %c1, @updatelo
	jmp @updatehi
@updatelo
	%lo1 =l copy %mid
	%hi1 =l copy %hi0
	jmp @loop
@updatehi
	%lo2 =l copy %lo0
	%hi2 =l copy %mid
	jmp @loop
@exit
	ret %lo0
}
