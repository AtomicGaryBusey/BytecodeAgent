export function l $isqrt(l %n) {
@start
	jmp @loop(0, 4294967296)
@loop(%lo0:l, %hi0:l)
	%diff0 =l sub %hi0, %lo0
	%c0 =w ceql %diff0, 1
	jnz %c0, @exit(%lo0)
	%sum0 =l add %lo0, %hi0
	%mid0 =l shr %sum0, 1
	%sq0 =l mul %mid0, %mid0
	%c1 =w culel %sq0, %n
	jnz %c1, @setlo(%lo0, %hi0, %mid0)
	jmp @loop(%lo0, %mid0)
@setlo(%lo1:l, %hi1:l, %mid1:l)
	jmp @loop(%mid1, %hi1)
@exit(%lo2:l)
	ret %lo2
}
