export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %pos0 =l copy 0
    %opos0 =l copy 0
    jmp @loop
@loop
    %pos =phi @start %pos0, @lit_done %pos_lit, @rep_done %pos_rep, @nop %pos_nop
    %opos =phi @start %opos0, @lit_done %opos_lit, @rep_done %opos_rep, @nop %opos_nop
    %c_end =w csgel %pos, %n
    jnz %c_end, @done, @header
@done
    ret %opos
@header
    %hp =l add %src, %pos
    %h =w loadsb %hp
    %h_neg =w csltw %h, 0
    jnz %h_neg, @neg, @lit_check
@neg
    %c_nop =w ceqw %h, -128
    jnz %c_nop, @nop, @rep_check
@nop
    %pos_nop =l add %pos, 1
    %opos_nop =l copy %opos
    jmp @loop
@lit_check
    %hl =l extsw %h
    %lneed =l add %hl, 2
    %lrem_src =l sub %n, %pos
    %lc_src =w csltl %lrem_src, %lneed
    jnz %lc_src, @err_m2, @lit_cap
@lit_cap
    %llen =l add %hl, 1
    %lnew =l add %opos, %llen
    %lc_over =w csgtl %lnew, %cap
    %lpos0 =l add %pos, 1
    %lopos0 =l copy %opos
    %lrem0 =l copy %llen
    jnz %lc_over, @lit_trunc, @lit_loop
@lit_loop
    %lpos =phi @lit_cap %lpos0, @lit_body %lpos1
    %lopos =phi @lit_cap %lopos0, @lit_body %lopos1
    %lrem =phi @lit_cap %lrem0, @lit_body %lrem1
    %lc_done =w ceql %lrem, 0
    jnz %lc_done, @lit_done, @lit_body
@lit_body
    %lbsp =l add %src, %lpos
    %lbb =w loadub %lbsp
    %lbop =l add %out, %lopos
    storeb %lbb, %lbop
    %lpos1 =l add %lpos, 1
    %lopos1 =l add %lopos, 1
    %lrem1 =l sub %lrem, 1
    jmp @lit_loop
@lit_done
    %pos_lit =l copy %lpos
    %opos_lit =l copy %lopos
    jmp @loop
@lit_trunc
    %ltpos0 =l add %pos, 1
    %ltopos0 =l copy %opos
    %ltrem0 =l sub %cap, %opos
    jmp @lit_tloop
@lit_tloop
    %ltpos =phi @lit_trunc %ltpos0, @lit_tbody %ltpos1
    %ltopos =phi @lit_trunc %ltopos0, @lit_tbody %ltopos1
    %ltrem =phi @lit_trunc %ltrem0, @lit_tbody %ltrem1
    %ltc_done =w ceql %ltrem, 0
    jnz %ltc_done, @ret_m1, @lit_tbody
@lit_tbody
    %ltbsp =l add %src, %ltpos
    %ltbb =w loadub %ltbsp
    %ltbop =l add %out, %ltopos
    storeb %ltbb, %ltbop
    %ltpos1 =l add %ltpos, 1
    %ltopos1 =l add %ltopos, 1
    %ltrem1 =l sub %ltrem, 1
    jmp @lit_tloop
@rep_check
    %rrem_src =l sub %n, %pos
    %rc_src =w csltl %rrem_src, 2
    jnz %rc_src, @err_m2, @rep_cap
@rep_cap
    %rhl =l extsw %h
    %rone =l copy 1
    %rcnt =l sub %rone, %rhl
    %rnew =l add %opos, %rcnt
    %rc_over =w csgtl %rnew, %cap
    %rop0 =l copy %opos
    %rrem0 =l copy %rcnt
    jnz %rc_over, @rep_trunc, @rep_loop
@rep_loop
    %rop =phi @rep_cap %rop0, @rep_body %rop1
    %rrem =phi @rep_cap %rrem0, @rep_body %rrem1
    %rc_done =w ceql %rrem, 0
    jnz %rc_done, @rep_done, @rep_body
@rep_body
    %rb_idx =l add %pos, 1
    %rb_ptr =l add %src, %rb_idx
    %rb_b =w loadub %rb_ptr
    %rb_op =l add %out, %rop
    storeb %rb_b, %rb_op
    %rop1 =l add %rop, 1
    %rrem1 =l sub %rrem, 1
    jmp @rep_loop
@rep_done
    %pos_rep =l add %pos, 2
    %opos_rep =l copy %rop
    jmp @loop
@rep_trunc
    %rtop0 =l copy %opos
    %rtrem0 =l sub %cap, %opos
    jmp @rep_tloop
@rep_tloop
    %rtop =phi @rep_trunc %rtop0, @rep_tbody %rtop1
    %rtrem =phi @rep_trunc %rtrem0, @rep_tbody %rtrem1
    %rtc_done =w ceql %rtrem, 0
    jnz %rtc_done, @ret_m1, @rep_tbody
@rep_tbody
    %rt_idx =l add %pos, 1
    %rt_ptr =l add %src, %rt_idx
    %rt_b =w loadub %rt_ptr
    %rt_op =l add %out, %rtop
    storeb %rt_b, %rt_op
    %rtop1 =l add %rtop, 1
    %rtrem1 =l sub %rtrem, 1
    jmp @rep_tloop
@err_m2
    ret -2
@ret_m1
    ret -1
}
