collatz_max:
	cmp rdi, 1
	jl .Lret0
	xor r8d, r8d
	mov r9, 1
.Louter:
	cmp r9, rdi
	jg .Ldone
	mov rax, r9
	xor ecx, ecx
.Linner:
	cmp rax, 1
	je .Lcounted
	test rax, 1
	jne .Lodd
	shr rax, 1
	jmp .Linc
.Lodd:
	lea rdx, [rax+rax*2]
	inc rdx
	mov rax, rdx
.Linc:
	inc rcx
	jmp .Linner
.Lcounted:
	cmp rcx, r8
	jle .Lnext
	mov r8, rcx
.Lnext:
	inc r9
	jmp .Louter
.Ldone:
	mov rax, r8
	ret
.Lret0:
	xor eax, eax
	ret
