export function l $bubble_sort(l %a, l %n) {
@start
  %one =l 1
  %eight =l 8
  %small =w cslel %n, %one
  jnz %small, @done, @outer_init
@outer_init
  %i =l 0
  jmp @outer_cond
@outer_cond
  %ocond =w csltl %i, %n
  jnz %ocond, @inner_init, @done
@inner_init
  %j =l 0
  %ip1 =l add %i, %one
  %limit =l sub %n, %ip1
  jmp @inner_cond
@inner_cond
  %icond =w csltl %j, %limit
  jnz %icond, @inner_body, @outer_step
@inner_body
  %off1 =l mul %j, %eight
  %pa =l add %a, %off1
  %va =l loadl %pa
  %jp1 =l add %j, %one
  %off2 =l mul %jp1, %eight
  %pb =l add %a, %off2
  %vb =l loadl %pb
  %gt =w csgtl %va, %vb
  jnz %gt, @do_swap, @inner_step
@do_swap
  storel %vb, %pa
  storel %va, %pb
  jmp @inner_step
@inner_step
  %j =l add %j, %one
  jmp @inner_cond
@outer_step
  %i =l add %i, %one
  jmp @outer_cond
@done
  %r =l 0
  ret %r
}
