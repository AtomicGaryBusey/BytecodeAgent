export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %v0 =l loadl %a
  %y0 =l call %f(l %v0)
  %t0 =l and %y0, 1
  %o0 =w cnel %t0, 0
  %ct0 =l extuw %o0
  %i0 =l copy 1
  %p0 =l add %a, 8
  jmp @loop
@loop
  %i =l phi @start %i0, @body %i2
  %p =l phi @start %p0, @body %p2
  %mn =l phi @start %y0, @body %mn2
  %mx =l phi @start %y0, @body %mx2
  %sm =l phi @start %y0, @body %sm2
  %ct =l phi @start %ct0, @body %ct2
  %cont =w csltl %i, %n
  jnz %cont, @body, @finish
@body
  %el =l loadl %p
  %y =l call %f(l %el)
  %sm2 =l add %sm, %y
  %t =l and %y, 1
  %o =w cnel %t, 0
  %e =l extuw %o
  %ct2 =l add %ct, %e
  %lt =w csltl %y, %mn
  %ltl =l extuw %lt
  %dmn =l sub %y, %mn
  %prmn =l mul %dmn, %ltl
  %mn2 =l add %mn, %prmn
  %gt =w csltl %mx, %y
  %gtl =l extuw %gt
  %dmx =l sub %y, %mx
  %prmx =l mul %dmx, %gtl
  %mx2 =l add %mx, %prmx
  %i2 =l add %i, 1
  %p2 =l add %p, 8
  jmp @loop
@finish
  storel %mn, %out
  %q1 =l add %out, 8
  storel %mx, %q1
  %q2 =l add %out, 16
  storel %sm, %q2
  ret %ct
}
