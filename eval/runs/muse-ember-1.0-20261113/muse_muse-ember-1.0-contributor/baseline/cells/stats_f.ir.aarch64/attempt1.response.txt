export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %v0 =l loadl %a
  %y0 =l call %f(l %v0)
  %t0 =l andl %y0, 1
  %o0 =w cnel %t0, 0
  %c0 =l extuw %o0
  jmp @loop(1, %y0, %y0, %y0, %c0)
@loop(%i1 l, %mn1 l, %mx1 l, %sm1 l, %ct1 l)
  %d1 =w cslel %n, %i1
  jnz %d1, @finish(%mn1, %mx1, %sm1, %ct1), @body(%i1, %mn1, %mx1, %sm1, %ct1)
@body(%i2 l, %mn2 l, %mx2 l, %sm2 l, %ct2 l)
  %ni2 =l addl %i2, 1
  %off2 =l mull %i2, 8
  %ad2 =l addl %a, %off2
  %el2 =l loadl %ad2
  %y2 =l call %f(l %el2)
  %s22 =l addl %sm2, %y2
  %t2 =l andl %y2, 1
  %o2 =w cnel %t2, 0
  %e2 =l extuw %o2
  %c22 =l addl %ct2, %e2
  %lt2 =w csltl %y2, %mn2
  jnz %lt2, @next(%ni2, %y2, %mx2, %s22, %c22), @chkmax(%ni2, %mn2, %mx2, %s22, %c22, %y2)
@chkmax(%ni4 l, %mn4 l, %mx4 l, %sm4 l, %ct4 l, %y4 l)
  %gt4 =w csltl %mx4, %y4
  jnz %gt4, @next(%ni4, %mn4, %y4, %sm4, %ct4), @next(%ni4, %mn4, %mx4, %sm4, %ct4)
@next(%ni5 l, %mn5 l, %mx5 l, %sm5 l, %ct5 l)
  jmp @loop(%ni5, %mn5, %mx5, %sm5, %ct5)
@finish(%mn6 l, %mx6 l, %sm6 l, %ct6 l)
  storel %mn6, %out
  %p1 =l addl %out, 8
  storel %mx6, %p1
  %p2 =l addl %out, 16
  storel %sm6, %p2
  ret %ct6
}
