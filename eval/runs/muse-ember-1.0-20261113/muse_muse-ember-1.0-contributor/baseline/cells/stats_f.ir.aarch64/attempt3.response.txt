export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %v0 =l loadl %a
  %y0 =l call %f(l %v0)
  %t0 =l and %y0, 1
  %o0 =w cnel %t0, 0
  %c0 =l extuw %o0
  %p0 =l add %a, 8
  %i0 =l copy 1
  jmp @loop(%i0, %p0, %y0, %y0, %y0, %c0)
@loop %i1 l, %p1 l, %mn1 l, %mx1 l, %sm1 l, %ct1 l
  %d1 =w cslel %n, %i1
  jnz %d1, @finish(%mn1, %mx1, %sm1, %ct1), @body(%i1, %p1, %mn1, %mx1, %sm1, %ct1)
@body %i2 l, %p2 l, %mn2 l, %mx2 l, %sm2 l, %ct2 l

  %np2 =l add %p2, 8
  %el2 =l loadl %p2
  %y2 =l call %f(l %el2)
  %s22 =l add %sm2, %y2
  %t2 =l and %y2, 1
  %o2 =w cnel %t2, 0
  %e2 =l extuw %o2

@chkmax %i4 l, %p4 l, %mn4 l, %mx4 l, %sm4 l, %ct4 l, %y4 l

  jnz %gt4, @next(%i4, %p4, %mn4, %y4, %sm4, %ct4), @next(%i4, %p4, %mn4, %mx4, %sm4, %ct4)
@next %i5 l, %p5 l, %mn5 l, %mx5 l, %sm5 l, %ct5 l
  jmp @loop(%i5, %p5, %mn5, %mx5, %sm5, %ct5)
@finish %mn6 l, %mx6 l, %sm6 l, %ct6 l
  storel %mn6, %out
  %q1 =l add %out, 8
  storel %mx6, %q1
  %q2 =l add %out, 16
  storel %sm6, %q2
  ret %ct6
}
