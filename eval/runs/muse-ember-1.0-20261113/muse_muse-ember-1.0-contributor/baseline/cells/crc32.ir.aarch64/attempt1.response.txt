export function l $crc32(l %p, l %n) {
@entry
  %crc0 =w copy -1
  %i0 =l copy 0
  %poly0 =w copy 237
  %poly1 =w shl %poly0, 8
  %poly2 =w or %poly1, 184
  %poly3 =w shl %poly2, 8
  %poly4 =w or %poly3, 131
  %poly5 =w shl %poly4, 8
  %poly =w or %poly5, 32
  jmp @loop
@loop
  %crc =w phi @entry %crc0, @next %crc_loop
  %i =l phi @entry %i0, @next %i_loop
  %done =w ceql %i, %n
  jnz %done, @final, @body
@final
  %res =w xor %crc, -1
  %rexl =l extuw %res
  ret %rexl
@body
  %addr =l add %p, %i
  %b =w loadub %addr
  %crc1 =w xor %crc, %b
  %j0 =l copy 0
  jmp @inner
@inner
  %crc_in =w phi @body %crc1, @merge %crc_after
  %j =l phi @body %j0, @merge %j_next
  %idone =w ceql %j, 8
  jnz %idone, @next, @innerbody
@next
  %crc_loop =w copy %crc_in
  %i_loop =l add %i, 1
  jmp @loop
@innerbody
  %cond =w and %crc_in, 1
  jnz %cond, @xorcase, @shiftcase
@xorcase
  %shr1 =w shr %crc_in, 1
  %crc_x =w xor %shr1, %poly
  jmp @merge
@shiftcase
  %crc_s =w shr %crc_in, 1
  jmp @merge
@merge
  %crc_after =w phi @xorcase %crc_x, @shiftcase %crc_s
  %j_next =l add %j, 1
  jmp @inner
}
