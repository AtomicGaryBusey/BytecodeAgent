export function l $bmin_empty(l %a, l %n) {
@start
  %zero =l 0
  %one =l 1
  %eight =l 8
  %max =l 9223372036854775807
  %empty =w ceql %n, %zero
  jnz %empty, @exit, @loop
@loop
  %i =l phi @start %zero, @body %i_next
  %min =l phi @start %max, @body %min_next
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %off =l mul %i, %eight
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %less =w csltl %val, %min
  %min_next =l sel %less, %val, %min
  %i_next =l add %i, %one
  jmp @loop
@exit
  %res =l phi @start %max, @loop %min
  ret %res
}
