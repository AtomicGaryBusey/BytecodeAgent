export function l $bmin_empty(l %a, l %n) {
@start
  %empty =w ceql %n, 0
  jnz %empty, @exit, @loop
@loop
  %i =l phi @start 0, @body %i_next
  %min =l phi @start 9223372036854775807, @body %min_next
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %less =w csltl %val, %min
  %min_next =l sel %less, %val, %min
  %i_next =l add %i, 1
  jmp @loop
@exit
  %res =l phi @start 9223372036854775807, @loop %min
  ret %res
}
