memmove_i64:
 cld
 test rdx, rdx
 je .Ldone
 cmp rdi, rsi
 jbe .Lforward
 mov rcx, rdx
 dec rcx
.Lback:
 mov rax, qword ptr [rsi + rcx*8]
 mov qword ptr [rdi + rcx*8], rax
 dec rcx
 jns .Lback
 jmp .Ldone
.Lforward:
 xor ecx, ecx
.Lfwd:
 mov rax, qword ptr [rsi + rcx*8]
 mov qword ptr [rdi + rcx*8], rax
 inc rcx
 cmp rcx, rdx
 jb .Lfwd
.Ldone:
 xor eax, eax
 ret
