qsort_pairs:
  cld
  cmp rsi, 2
  jl .Ldone
  push rbx
  push r12
  push r13
  push r14
  push r15
  sub rsp, 16
  mov rbx, rdi
  mov r12, rsi
  mov rax, r12
  shr rax, 1
  mov rcx, rax
  shl rcx, 4
  add rcx, rbx
  mov rdx, r12
  dec rdx
  shl rdx, 4
  add rdx, rbx
  mov r8, qword ptr [rcx]
  mov r9, qword ptr [rdx]
  mov qword ptr [rcx], r9
  mov qword ptr [rdx], r8
  mov r8, qword ptr [rcx+8]
  mov r9, qword ptr [rdx+8]
  mov qword ptr [rcx+8], r9
  mov qword ptr [rdx+8], r8
  mov r13, qword ptr [rdx]
  mov r14, qword ptr [rdx+8]
  mov qword ptr [rsp], 0
  mov qword ptr [rsp+8], 0
.Lloop:
  mov rax, qword ptr [rsp]
  mov rcx, r12
  dec rcx
  cmp rax, rcx
  jge .Lpartition_done
  mov rax, qword ptr [rsp]
  shl rax, 4
  add rax, rbx
  mov rcx, qword ptr [rax]
  cmp rcx, r13
  jl .Lless
  jg .Lnext
  mov rcx, qword ptr [rax+8]
  cmp rcx, r14
  jl .Lless
  jg .Lnext
  jmp .Lnext
.Lless:
  mov rcx, qword ptr [rsp+8]
  shl rcx, 4
  add rcx, rbx
  mov rdx, qword ptr [rax]
  mov r8, qword ptr [rcx]
  mov qword ptr [rax], r8
  mov qword ptr [rcx], rdx
  mov rdx, qword ptr [rax+8]
  mov r8, qword ptr [rcx+8]
  mov qword ptr [rax+8], r8
  mov qword ptr [rcx+8], rdx
  mov rax, qword ptr [rsp+8]
  inc rax
  mov qword ptr [rsp+8], rax
.Lnext:
  mov rax, qword ptr [rsp]
  inc rax
  mov qword ptr [rsp], rax
  jmp .Lloop
.Lpartition_done:
  mov rax, qword ptr [rsp+8]
  shl rax, 4
  add rax, rbx
  mov rcx, r12
  dec rcx
  shl rcx, 4
  add rcx, rbx
  mov rdx, qword ptr [rax]
  mov r8, qword ptr [rcx]
  mov qword ptr [rax], r8
  mov qword ptr [rcx], rdx
  mov rdx, qword ptr [rax+8]
  mov r8, qword ptr [rcx+8]
  mov qword ptr [rax+8], r8
  mov qword ptr [rcx+8], rdx
  mov r15, qword ptr [rsp+8]
  mov rdi, rbx
  mov rsi, r15
  call qsort_pairs
  mov rax, r15
  inc rax
  shl rax, 4
  add rax, rbx
  mov rdi, rax
  mov rsi, r12
  sub rsi, r15
  dec rsi
  call qsort_pairs
  add rsp, 16
  xor eax, eax
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbx
  ret
.Ldone:
  xor eax, eax
  ret
