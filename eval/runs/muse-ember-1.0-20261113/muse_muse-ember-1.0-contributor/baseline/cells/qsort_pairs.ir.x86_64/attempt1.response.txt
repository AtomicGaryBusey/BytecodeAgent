export function l $qsort_pairs(l %a, l %n) {
@start
  %zr =l sub %n, %n
  %c0 =w csltl %n, 2
  jnz %c0, @done, @head
@head
  %pa =l phi @start %a, @pdone %na
  %pn =l phi @start %n, @pdone %nn
  %c1 =w csltl %pn, 2
  jnz %c1, @done, @body
@done
  ret %zr
@body
  %m =l div %pn, 2
  %offm =l mul %m, 16
  %pmid =l add %pa, %offm
  %lastv =l sub %pn, 1
  %offl =l mul %lastv, 16
  %plast =l add %pa, %offl
  %pk =l loadl %pmid
  %pmidv =l add %pmid, 8
  %pv =l loadl %pmidv
  %lk =l loadl %plast
  %plastv =l add %plast, 8
  %lv =l loadl %plastv
  storel %lk, %pmid
  storel %lv, %pmidv
  storel %pk, %plast
  storel %pv, %plastv
  jmp @ploop
@ploop
  %i =l phi @body 0, @pswap %inext, @pnext %inext2
  %j =l phi @body 0, @pswap %jnext, @pnext %jnext2
  %cend =w csltl %j, %lastv
  jnz %cend, @pbody, @pdone
@pbody
  %offj =l mul %j, 16
  %pj =l add %pa, %offj
  %kj =l loadl %pj
  %pjv =l add %pj, 8
  %vj =l loadl %pjv
  %ckl =w csltl %kj, %pk
  jnz %ckl, @pswap, @pck
@pck
  %ckg =w csgtl %kj, %pk
  jnz %ckg, @pnext, @pvck
@pvck
  %cvl =w csltl %vj, %pv
  jnz %cvl, @pswap, @pvcg
@pvcg
  %cvg =w csgtl %vj, %pv
  jnz %cvg, @pnext, @pswap
@pswap
  %offi =l mul %i, 16
  %pi =l add %pa, %offi
  %piv =l add %pi, 8
  %ik =l loadl %pi
  %iv =l loadl %piv
  storel %kj, %pi
  storel %vj, %piv
  storel %ik, %pj
  storel %iv, %pjv
  %inext =l add %i, 1
  %jnext =l add %j, 1
  jmp @ploop
@pnext
  %inext2 =l add %i, 0
  %jnext2 =l add %j, 1
  jmp @ploop
@pdone
  %offi2 =l mul %i, 16
  %pi2 =l add %pa, %offi2
  %pi2v =l add %pi2, 8
  %fk =l loadl %pi2
  %fv =l loadl %pi2v
  %plastv2 =l add %plast, 8
  %evk =l loadl %plast
  %evv =l loadl %plastv2
  storel %evk, %pi2
  storel %evv, %pi2v
  storel %fk, %plast
  storel %fv, %plastv2
  %rc =l call $qsort_pairs(l %pa, l %i)
  %plus1 =l add %i, 1
  %offr =l mul %plus1, 16
  %na =l add %pa, %offr
  %nn =l sub %pn, %plus1
  jmp @head
}
