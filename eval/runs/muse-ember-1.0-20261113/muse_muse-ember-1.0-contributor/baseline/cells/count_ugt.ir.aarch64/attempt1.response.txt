export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %inext
	%cnt =l phi @start 0, @cont %cntnext
	%p =l phi @start %a, @cont %pnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @exit
@body
	%v =l loadl %p
	%gt =w cugtl %v, %t
	jnz %gt, @inc, @ninc
@inc
	%cnt1 =l add %cnt, 1
	jmp @cont
@ninc
	jmp @cont
@cont
	%cntnext =l phi @inc %cnt1, @ninc %cnt
	%inext =l add %i, 1
	%pnext =l add %p, 8
	jmp @loop
@exit
	ret %cnt
}
