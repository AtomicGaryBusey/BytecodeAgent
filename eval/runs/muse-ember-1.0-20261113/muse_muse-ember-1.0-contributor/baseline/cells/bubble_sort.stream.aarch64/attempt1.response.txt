bubble_sort:
    cmp x1, #2
    b.lt .Lret
    sub x2, x1, #1
.Louter:
    mov x3, #0
.Linner:
    cmp x3, x2
    b.ge .Lnext
    ldr x5, [x0, x3, lsl #3]
    add x6, x3, #1
    ldr x7, [x0, x6, lsl #3]
    cmp x5, x7
    b.le .Lskip
    str x7, [x0, x3, lsl #3]
    str x5, [x0, x6, lsl #3]
.Lskip:
    add x3, x3, #1
    b .Linner
.Lnext:
    subs x2, x2, #1
    b.gt .Louter
.Lret:
    mov x0, #0
    ret
