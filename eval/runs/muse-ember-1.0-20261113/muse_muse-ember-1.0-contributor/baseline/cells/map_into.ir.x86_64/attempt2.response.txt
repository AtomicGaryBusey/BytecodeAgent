export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
  jmp @loop
@loop
  %i =l phi @start 0, @cont %i2
  %cnt =l phi @start 0, @cont %cnt2
  %ps =l phi @start %src, @cont %ps2
  %pd =l phi @start %dst, @cont %pd2
  %lt =w csltl %i, %n
  jnz %lt, @body, @exit
@exit
  ret %cnt
@body
  %v =l loadl %ps
  %r =l call %f(l %v)
  storel %r, %pd
  %lt2 =w csltl %r, %v
  jnz %lt2, @inc, @cont
@inc
  %cntInc =l add %cnt, 1
  jmp @cont
@cont
  %cnt2 =l phi @body %cnt, @inc %cntInc
  %i2 =l add %i, 1
  %ps2 =l add %ps, 8
  %pd2 =l add %pd, 8
  jmp @loop
}
