export function l $count_desc(l %a, l %n) {
@start
    %limit = subl %n, 1
    %zero = subl %n, %n
    jmp @loop(%zero, %zero, %a, %limit)
@loop(l %i, l %cnt, l %cur, l %lim)
    %cond = csltl %i, %lim
    jnz %cond, @body, @done
@body
    %next = addl %cur, 8
    %v1 = loadl %cur
    %v2 = loadl %next
    %gt = csgtl %v1, %v2
    %gtl = extuwl %gt
    %cnt2 = addl %cnt, %gtl
    %i2 = addl %i, 1
    jmp @loop(%i2, %cnt2, %next, %lim)
@done
    ret %cnt
}
