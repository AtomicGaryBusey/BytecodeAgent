export function l $count_desc(l %a, l %n) {
@start
    %one =l copy 1
    %eight =l copy 8
    %zero =l copy 0
    %limit =l sub %n, %one
    jmp @loop(%zero, %zero, %a)
@loop(l %i, l %cnt, l %cur)
    %cond =w csltl %i, %limit
    jnz %cond, @body, @done
@body
    %next =l add %cur, %eight
    %v1 =l loadl %cur
    %v2 =l loadl %next
    %gt =w csgtl %v1, %v2
    jnz %gt, @inc, @skip
@inc
    %cnt2 =l add %cnt, %one
    jmp @next(%cnt2)
@skip
    jmp @next(%cnt)
@next(l %cnt3)
    %i2 =l add %i, %one
    jmp @loop(%i2, %cnt3, %next)
@done
    ret %cnt
}
