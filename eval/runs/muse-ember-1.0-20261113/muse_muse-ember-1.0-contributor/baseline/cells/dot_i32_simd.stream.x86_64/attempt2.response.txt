dot_i32_simd:
cld
test rdx, rdx
jle .Lreturn_zero
pxor xmm4, xmm4
xor rcx, rcx
mov r8, rdx
and r8, -4
cmp rcx, r8
jae .Lreduce
.Lvec_loop:
movdqu xmm0, xmmword ptr [rdi + rcx*4]
movdqu xmm1, xmmword ptr [rsi + rcx*4]
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm4, xmm2
pshufd xmm0, xmm0, 238
pshufd xmm1, xmm1, 238
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm4, xmm2
add rcx, 4
cmp rcx, r8
jb .Lvec_loop
.Lreduce:
movdqa xmm5, xmm4
psrldq xmm5, 8
paddq xmm4, xmm5
movq rax, xmm4
xor r10, r10
.Lscalar_loop:
cmp rcx, rdx
jae .Lreturn
movsxd r9, dword ptr [rdi + rcx*4]
movsxd r11, dword ptr [rsi + rcx*4]
imul r9, r11
add r10, r9
inc rcx
jmp .Lscalar_loop
.Lreturn:
add rax, r10
ret
.Lreturn_zero:
xor eax, eax
ret
