dot_i32_simd:
cld
test rdx, rdx
jle .Lreturn_zero
pxor xmm4, xmm4
xor rcx, rcx
mov r8, rdx
and r8, -4
cmp rcx, r8
jae .Lreduce
.Lvec_loop:
movdqu xmm0, xmmword ptr [rdi + rcx*4]
movdqu xmm1, xmmword ptr [rsi + rcx*4]
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm4, xmm2
pshufd xmm0, xmm0, 238
pshufd xmm1, xmm1, 238
pmovsxdq xmm2, xmm0
pmovsxdq xmm3, xmm1
pmuldq xmm2, xmm3
paddq xmm4, xmm2
add rcx, 4
cmp rcx, r8
jb .Lvec_loop
.Lreduce:
psrldq xmm5, xmm4, 8
paddq xmm4, xmm5
movq rax, xmm4
.Lscalar_loop:
cmp rcx, rdx
jae .Ldone
mov r8d, dword ptr [rdi + rcx*4]
movsxd r8, r8d
mov r9d, dword ptr [rsi + rcx*4]
movsxd r9, r9d
imul r8, r9
add rax, r8
add rcx, 1
jmp .Lscalar_loop
.Ldone:
ret
.Lreturn_zero:
xor rax, rax
ret
