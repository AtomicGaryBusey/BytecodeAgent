max_byte:
    test rsi, rsi
    jle .Lret0
    cmp rsi, 16
    jb .Lscalar
    pxor xmm0, xmm0
    xor ecx, ecx
    lea rdx, [rsi-16]
.Lvloop:
    cmp rcx, rdx
    ja .Lvdone
    movdqu xmm1, xmmword ptr [rdi+rcx]
    pmaxub xmm0, xmm1
    add rcx, 16
    jmp .Lvloop
.Lvdone:
    movdqu xmm1, xmmword ptr [rdi+rdx]
    pmaxub xmm0, xmm1
    pshufd xmm1, xmm0, 78
    pmaxub xmm0, xmm1
    pshufd xmm1, xmm0, 177
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrlw xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrld xmm1, 16
    pmaxub xmm0, xmm1
    movd eax, xmm0
    and eax, 255
    ret
.Lscalar:
    xor eax, eax
    xor ecx, ecx
.Lsloop:
    cmp rcx, rsi
    jae .Lsdone
    movzx edx, byte ptr [rdi+rcx]
    cmp edx, eax
    jbe .Lsnext
    mov eax, edx
.Lsnext:
    inc rcx
    jmp .Lsloop
.Lsdone:
    ret
.Lret0:
    xor eax, eax
    ret
