export function l $reverse(l %a, l %n) {
@start
  %j0 =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @start 0, @body %i2
  %j =l phi @start %j0, @body %j2
  %c =w csltl %i, %j
  jnz %c, @body, @done
@body
  %offi =l shl %i, 3
  %offj =l shl %j, 3
  %pa =l add %a, %offi
  %pj =l add %a, %offj
  %vi =l loadl %pa
  %vj =l loadl %pj
  storel %vj, %pa
  storel %vi, %pj
  %i2 =l add %i, 1
  %j2 =l sub %j, 1
  jmp @loop
@done
  ret 0
}
