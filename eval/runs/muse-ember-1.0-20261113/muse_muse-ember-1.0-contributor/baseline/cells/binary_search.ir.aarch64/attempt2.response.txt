export function l $binary_search(l %a, l %n, l %key) {
@start
  %lo0 =l copy 0
  %hi0 =l copy %n
  jmp @loop
@loop
  %lo =l phi @start %lo0, @latch %lo_next
  %hi =l phi @start %hi0, @latch %hi_next
  %cond =w csltl %lo, %hi
  jnz %cond, @body, @notfound
@body
  %diff =l sub %hi, %lo
  %half =l div %diff, 2
  %mid =l add %lo, %half
  %off =l mul %mid, 8
  %ptr =l add %a, %off
  %val =l loadl %ptr
  %eq =w ceql %val, %key
  jnz %eq, @found, @check
@check
  %lt =w csltl %val, %key
  jnz %lt, @right, @left
@right
  %lo_r =l add %mid, 1
  %hi_r =l copy %hi
  jmp @latch
@left
  %lo_l =l copy %lo
  %hi_l =l copy %mid
  jmp @latch
@latch
  %lo_next =l phi @right %lo_r, @left %lo_l
  %hi_next =l phi @right %hi_r, @left %hi_l
  jmp @loop
@found
  ret %mid
@notfound
  %neg =l copy -1
  ret %neg
}
