export function l $sum8(a l, b l, c l, d l, e l, f l, g l, h l) {
@start
  %t1 =l mull %b, 2
  %t2 =l mull %c, 3
  %t3 =l mull %d, 4
  %t4 =l mull %e, 5
  %t5 =l mull %f, 6
  %t6 =l mull %g, 7
  %t7 =l mull %h, 8
  %r0 =l subl %a, %t1
  %r1 =l addl %r0, %t2
  %r2 =l subl %r1, %t3
  %r3 =l addl %r2, %t4
  %r4 =l subl %r3, %t5
  %r5 =l addl %r4, %t6
  %r6 =l subl %r5, %t7
  ret %r6
}
