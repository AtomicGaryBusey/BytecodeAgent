export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
  %pcslot =l alloc8 1
  %dslot =l alloc8 1
  %islot =l alloc8 1
  %zero =l sub %len, %len
  %one =l add %zero, 1
  storel %zero, %pcslot
  storel %zero, %dslot
  jmp @loop
@loop
  %pc =l loadl %pcslot
  %d =l loadl %dslot
  %c_loop_lt =w csltl %pc, %len
  jnz %c_loop_lt, @dispatch, @exit
@dispatch
  %off_loop =l mul %pc, 8
  %caddr_loop =l add %code, %off_loop
  %op_loop =l loadl %caddr_loop
  %c0_chk =w ceql %op_loop, 0
  jnz %c0_chk, @h_halt, @chk1
@chk1
  %c1_chk =w ceql %op_loop, 1
  jnz %c1_chk, @h_push, @chk2
@chk2
  %c2_chk =w ceql %op_loop, 2
  jnz %c2_chk, @h_add, @chk3
@chk3
  %c3_chk =w ceql %op_loop, 3
  jnz %c3_chk, @h_sub, @chk4
@chk4
  %c4_chk =w ceql %op_loop, 4
  jnz %c4_chk, @h_mul, @chk5
@chk5
  %c5_chk =w ceql %op_loop, 5
  jnz %c5_chk, @h_dup, @chk6
@chk6
  %c6_chk =w ceql %op_loop, 6
  jnz %c6_chk, @h_swap, @chk7
@chk7
  %c7_chk =w ceql %op_loop, 7
  jnz %c7_chk, @h_drop, @chk8
@chk8
  %c8_chk =w ceql %op_loop, 8
  jnz %c8_chk, @h_over, @chk9
@chk9
  %c9_chk =w ceql %op_loop, 9
  jnz %c9_chk, @h_lt, @chk10
@chk10
  %c10_chk =w ceql %op_loop, 10
  jnz %c10_chk, @h_jz, @chk11
@chk11
  %c11_chk =w ceql %op_loop, 11
  jnz %c11_chk, @h_jmp, @h_unknown
@h_halt
  jmp @exit
@h_push
  %pc1_push =l add %pc, 1
  %off1_push =l mul %pc1_push, 8
  %a1_push =l add %code, %off1_push
  %v_push =l loadl %a1_push
  %doff_push =l mul %d, 8
  %saddr_push =l add %stack, %doff_push
  storel %v_push, %saddr_push
  %d1_push =l add %d, 1
  storel %d1_push, %dslot
  %pc2_push =l add %pc, 2
  storel %pc2_push, %pcslot
  jmp @loop
@h_add
  %dm1_add =l sub %d, 1
  %offb_add =l mul %dm1_add, 8
  %addrb_add =l add %stack, %offb_add
  %b_add =l loadl %addrb_add
  %dm2_add =l sub %d, 2
  %offa_add =l mul %dm2_add, 8
  %addra_add =l add %stack, %offa_add
  %a_add =l loadl %addra_add
  %res_add =l add %a_add, %b_add
  storel %res_add, %addra_add
  storel %dm1_add, %dslot
  %pc1_add =l add %pc, 1
  storel %pc1_add, %pcslot
  jmp @loop
@h_sub
  %dm1_sub =l sub %d, 1
  %offb_sub =l mul %dm1_sub, 8
  %addrb_sub =l add %stack, %offb_sub
  %b_sub =l loadl %addrb_sub
  %dm2_sub =l sub %d, 2
  %offa_sub =l mul %dm2_sub, 8
  %addra_sub =l add %stack, %offa_sub
  %a_sub =l loadl %addra_sub
  %res_sub =l sub %a_sub, %b_sub
  storel %res_sub, %addra_sub
  storel %dm1_sub, %dslot
  %pc1_sub =l add %pc, 1
  storel %pc1_sub, %pcslot
  jmp @loop
@h_mul
  %dm1_mul =l sub %d, 1
  %offb_mul =l mul %dm1_mul, 8
  %addrb_mul =l add %stack, %offb_mul
  %b_mul =l loadl %addrb_mul
  %dm2_mul =l sub %d, 2
  %offa_mul =l mul %dm2_mul, 8
  %addra_mul =l add %stack, %offa_mul
  %a_mul =l loadl %addra_mul
  %res_mul =l mul %a_mul, %b_mul
  storel %res_mul, %addra_mul
  storel %dm1_mul, %dslot
  %pc1_mul =l add %pc, 1
  storel %pc1_mul, %pcslot
  jmp @loop
@h_dup
  %dm1_dup =l sub %d, 1
  %off_dup =l mul %dm1_dup, 8
  %addr_dup =l add %stack, %off_dup
  %v_dup =l loadl %addr_dup
  %doff_dup =l mul %d, 8
  %saddr_dup =l add %stack, %doff_dup
  storel %v_dup, %saddr_dup
  %d1_dup =l add %d, 1
  storel %d1_dup, %dslot
  %pc1_dup =l add %pc, 1
  storel %pc1_dup, %pcslot
  jmp @loop
@h_swap
  %dm1_swap =l sub %d, 1
  %offb_swap =l mul %dm1_swap, 8
  %addrb_swap =l add %stack, %offb_swap
  %b_swap =l loadl %addrb_swap
  %dm2_swap =l sub %d, 2
  %offa_swap =l mul %dm2_swap, 8
  %addra_swap =l add %stack, %offa_swap
  %a_swap =l loadl %addra_swap
  storel %b_swap, %addra_swap
  storel %a_swap, %addrb_swap
  %pc1_swap =l add %pc, 1
  storel %pc1_swap, %pcslot
  jmp @loop
@h_drop
  %d1_drop =l sub %d, 1
  storel %d1_drop, %dslot
  %pc1_drop =l add %pc, 1
  storel %pc1_drop, %pcslot
  jmp @loop
@h_over
  %dm2_over =l sub %d, 2
  %off_over =l mul %dm2_over, 8
  %addr_over =l add %stack, %off_over
  %v_over =l loadl %addr_over
  %doff_over =l mul %d, 8
  %saddr_over =l add %stack, %doff_over
  storel %v_over, %saddr_over
  %d1_over =l add %d, 1
  storel %d1_over, %dslot
  %pc1_over =l add %pc, 1
  storel %pc1_over, %pcslot
  jmp @loop
@h_lt
  %dm1_lt =l sub %d, 1
  %offb_lt =l mul %dm1_lt, 8
  %addrb_lt =l add %stack, %offb_lt
  %b_lt =l loadl %addrb_lt
  %dm2_lt =l sub %d, 2
  %offa_lt =l mul %dm2_lt, 8
  %addra_lt =l add %stack, %offa_lt
  %a_lt =l loadl %addra_lt
  %c_lt_cond =w csltl %a_lt, %b_lt
  jnz %c_lt_cond, @h_lt_true, @h_lt_false
@h_lt_true
  storel %one, %addra_lt
  storel %dm1_lt, %dslot
  %pc1_ltt =l add %pc, 1
  storel %pc1_ltt, %pcslot
  jmp @loop
@h_lt_false
  storel %zero, %addra_lt
  storel %dm1_lt, %dslot
  %pc1_ltf =l add %pc, 1
  storel %pc1_ltf, %pcslot
  jmp @loop
@h_jz
  %dm1_jz =l sub %d, 1
  %off_jz =l mul %dm1_jz, 8
  %addr_jz =l add %stack, %off_jz
  %v_jz =l loadl %addr_jz
  %pc1_jz =l add %pc, 1
  %off1_jz =l mul %pc1_jz, 8
  %a1_jz =l add %code, %off1_jz
  %t_jz =l loadl %a1_jz
  storel %dm1_jz, %dslot
  %c_jz =w ceql %v_jz, 0
  jnz %c_jz, @h_jz_taken, @h_jz_not
@h_jz_taken
  storel %t_jz, %pcslot
  jmp @loop
@h_jz_not
  %pc2_jz =l add %pc, 2
  storel %pc2_jz, %pcslot
  jmp @loop
@h_jmp
  %pc1_jmp =l add %pc, 1
  %off_jmp =l mul %pc1_jmp, 8
  %a1_jmp =l add %code, %off_jmp
  %t_jmp =l loadl %a1_jmp
  storel %t_jmp, %pcslot
  jmp @loop
@h_unknown
  %pc1_unk =l add %pc, 1
  storel %pc1_unk, %pcslot
  jmp @loop
@exit
  %d_exit =l loadl %dslot
  storel %d_exit, %islot
  jmp @zloop
@zloop
  %i_z =l loadl %islot
  %c_z =w csltl %i_z, %cap
  jnz %c_z, @zbody, @zdone
@zbody
  %off_z =l mul %i_z, 8
  %addr_z =l add %stack, %off_z
  storel %zero, %addr_z
  %i1_z =l add %i_z, 1
  storel %i1_z, %islot
  jmp @zloop
@zdone
  %d_done =l loadl %dslot
  %c_empty =w ceql %d_done, 0
  jnz %c_empty, @ret_zero, @ret_top
@ret_zero
  ret %zero
@ret_top
  %dm1_ret =l sub %d_done, 1
  %off_ret =l mul %dm1_ret, 8
  %addr_ret =l add %stack, %off_ret
  %top_ret =l loadl %addr_ret
  ret %top_ret
}
