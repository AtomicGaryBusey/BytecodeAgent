export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %zero =l sub %m, %m
  %one =l add %zero, 1
  jmp @init_head
@init_head
  %j =l phi @start %zero, @init_body %j_next
  %init_cmp =w cslel %j, %n
  jnz %init_cmp, @init_body, @outer_head
@init_body
  %init_off =l mul %j, 8
  %init_ptr =l add %row, %init_off
  storel %zero, %init_ptr
  %j_next =l add %j, %one
  jmp @init_head
@outer_head
  %i =l phi @init_head %zero, @outer_next %i_next
  %outer_cmp =w csltl %i, %m
  jnz %outer_cmp, @inner_head, @done
@inner_head
  %j_in =l phi @outer_head %one, @inner_next %j_next_in
  %diag =l phi @outer_head %zero, @inner_next %up
  %inner_cmp =w cslel %j_in, %n
  jnz %inner_cmp, @inner_body, @outer_next
@inner_body
  %off_j =l mul %j_in, 8
  %ptr_j =l add %row, %off_j
  %up =l loadl %ptr_j
  %j_prev =l sub %j_in, 1
  %off_prev =l mul %j_prev, 8
  %ptr_prev =l add %row, %off_prev
  %left =l loadl %ptr_prev
  %xptr =l add %x, %i
  %xi =w loadub %xptr
  %yptr =l add %y, %j_prev
  %yi =w loadub %yptr
  %eq_cmp =w ceqw %xi, %yi
  %j_next_in =l add %j_in, %one
  jnz %eq_cmp, @inner_eq, @inner_ne
@inner_eq
  %newval =l add %diag, 1
  storel %newval, %ptr_j
  jmp @inner_next
@inner_ne
  %max_cmp =w csltl %left, %up
  jnz %max_cmp, @inner_ne_up, @inner_ne_left
@inner_ne_up
  storel %up, %ptr_j
  jmp @inner_next
@inner_ne_left
  storel %left, %ptr_j
  jmp @inner_next
@inner_next
  jmp @inner_head
@outer_next
  %i_next =l add %i, %one
  jmp @outer_head
@done
  %off_n =l mul %n, 8
  %ptr_n =l add %row, %off_n
  %res =l loadl %ptr_n
  ret %res
}
