export function l $isqrt(l %n) {
@start
	%lo0 =l copy 0
	%hi0 =l copy 4294967295
	jmp @loop
@loop
	%lo =l phi @start %lo0, @yes %lo_yes, @no %lo_no
	%hi =l phi @start %hi0, @yes %hi_yes, @no %hi_no
	%cond =w cultl %lo, %hi
	jnz %cond, @body, @exit
@body
	%diff =l sub %hi, %lo
	%sum =l add %diff, 1
	%half =l udiv %sum, 2
	%mid =l add %lo, %half
	%prod =l mul %mid, %mid
	%feas =w culel %prod, %n
	jnz %feas, @yes, @no
@yes
	%lo_yes =l copy %mid
	%hi_yes =l copy %hi
	jmp @loop
@no
	%lo_no =l copy %lo
	%midm1 =l sub %mid, 1
	%hi_no =l copy %midm1
	jmp @loop
@exit
	%res =l phi @loop %lo
	ret %res
}
