export function l $fib(l %n) {
@entry
  %a0 =l copy 0
  %b0 =l copy 1
  %i0 =l copy 1
  %c0 =w ceql %n, 0
  jnz %c0, @ret0, @loop
@ret0
  %z =l copy 0
  ret %z
@loop
  %i =l phi @entry %i0, @loop %inext
  %a =l phi @entry %a0, @loop %b
  %b =l phi @entry %b0, @loop %t
  %t =l addl %a, %b
  %inext =l addl %i, 1
  %c1 =w csltl %i, %n
  jnz %c1, @loop, @done
@done
  ret %b
}
