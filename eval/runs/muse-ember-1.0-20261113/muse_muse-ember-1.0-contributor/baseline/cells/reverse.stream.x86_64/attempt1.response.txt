reverse:
    test rsi, rsi
    jle .Ldone
    mov rax, rsi
    dec rax
    xor rcx, rcx
.Lloop:
    cmp rcx, rax
    jge .Ldone
    mov rdx, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rax*8]
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rax*8], rdx
    inc rcx
    dec rax
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
