export function l $collatz_max(l %n) {
@start
  jmp @outer_cond
@outer_cond
  %k =l phi @start 1, @outer_next %k_next
  %max =l phi @start 0, @outer_next %max_next
  %done =w csgtl %k, %n
  jnz %done, @finish, @inner_init
@finish
  ret %max
@inner_init
  jmp @inner_cond
@inner_cond
  %x =l phi @inner_init %k, @inner_next %x_next
  %steps =l phi @inner_init 0, @inner_next %steps_next
  %is_one =w ceql %x, 1
  jnz %is_one, @inner_done, @inner_body
@inner_body
  %r =l and %x, 1
  %is_even =w ceql %r, 0
  jnz %is_even, @even_case, @odd_case
@even_case
  %x_even =l div %x, 2
  jmp @inner_next
@odd_case
  %tmp =l mul %x, 3
  %x_odd =l add %tmp, 1
  jmp @inner_next
@inner_next
  %x_next =l phi @even_case %x_even, @odd_case %x_odd
  %steps_next =l add %steps, 1
  jmp @inner_cond
@inner_done
  %is_bigger =w csgtl %steps, %max
  jnz %is_bigger, @update_max, @outer_next
@update_max
  jmp @outer_next
@outer_next
  %max_next =l phi @inner_done %max, @update_max %steps
  %k_next =l add %k, 1
  jmp @outer_cond
}
