export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%iszero =w ceql %n, %zero
	jnz %iszero, @done, @eqcheck
@eqcheck
	%same =w ceql %dst, %src
	jnz %same, @done, @dircheck
@dircheck
	%fwd =w cultl %dst, %src
	jnz %fwd, @fwdinit, @bwdinit
@fwdinit
	%i0 =l copy 0
	jmp @fwdloop %i0
@fwdloop
	%fi =l phi @fwdinit %i0, @fwdbody %fnext
	%done1 =w ceql %fi, %n
	jnz %done1, @done, @fwdbody
@fwdbody
	%eight =l copy 8
	%foff =l mul %fi, %eight
	%fs =l add %src, %foff
	%fd =l add %dst, %foff
	%fv =l loadl %fs
	storel %fv, %fd
	%one =l copy 1
	%fnext =l add %fi, %one
	jmp @fwdloop %fnext
@bwdinit
	%oneb =l copy 1
	%blast =l sub %n, %oneb
	jmp @bwdloop %blast
@bwdloop
	%bi =l phi @bwdinit %blast, @bwdbody %bnext
	%bneg =l copy -1
	%done2 =w ceql %bi, %bneg
	jnz %done2, @done, @bwdbody
@bwdbody
	%beight =l copy 8
	%boff =l mul %bi, %beight
	%bs =l add %src, %boff
	%bd =l add %dst, %boff
	%bv =l loadl %bs
	storel %bv, %bd
	%bone =l copy 1
	%bnext =l sub %bi, %bone
	jmp @bwdloop %bnext
@done
	%ret0 =l copy 0
	ret %ret0
}
