export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %z0 =l subl %n, %n
  %c257 =l addl %z0, 257
  %m1 =l subl %z0, 1
  jmp @outer
@outer
  %i =l phi @start 0, @rep_adv %i_rep, @lit_adv %i_lit
  %written =l phi @start 0, @rep_adv %written_rep, @lit_adv %written_lit
  %outer_lt =w csltl %i, %n
  jnz %outer_lt, @run_init
  jmp @exit
@run_init
  %p0 =l addl %src, %i
  %b0 =w loadub %p0
  jmp @run_head
@run_head
  %idx =l phi @run_init %i, @run_body %idx_rb
  %cnt =l phi @run_init 0, @run_body %cnt_rb
  %run_c1 =w csltl %cnt, 128
  jnz %run_c1, @run_chk_end
  jmp @run_done
@run_chk_end
  %run_c2 =w csltl %idx, %n
  jnz %run_c2, @run_cmp
  jmp @run_done
@run_cmp
  %pr =l addl %src, %idx
  %br =w loadub %pr
  %run_eq =w ceqw %br, %b0
  jnz %run_eq, @run_body
  jmp @run_done
@run_body
  %cnt_rb =l addl %cnt, 1
  %idx_rb =l addl %idx, 1
  jmp @run_head
@run_done
  %is_rep =w csgtl %cnt, 2
  jnz %is_rep, @rep_chk
  jmp @lit_init
@rep_chk
  %rep_need =l addl %written, 2
  %rep_over =w csgtl %rep_need, %cap
  jnz %rep_over, @fail
  jmp @rep_adv
@rep_adv
  %rep_pout =l addl %out, %written
  %rep_hdr =l subl %c257, %cnt
  storeb %rep_hdr, %rep_pout
  %rep_pout2 =l addl %rep_pout, 1
  storeb %b0, %rep_pout2
  %written_rep =l addl %written, 2
  %i_rep =l addl %i, %cnt
  jmp @outer
@lit_init
  jmp @lit_head
@lit_head
  %j =l phi @lit_init %i, @lit_body %j_lb
  %lit_c1 =w csltl %j, %n
  jnz %lit_c1, @lit_chk_len
  jmp @lit_done
@lit_chk_len
  %lit_len =l subl %j, %i
  %lit_c2 =w csltl %lit_len, 128
  jnz %lit_c2, @lit_chk_run
  jmp @lit_done
@lit_chk_run
  %lit_j2 =l addl %j, 2
  %lit_c3 =w csltl %lit_j2, %n
  jnz %lit_c3, @lit_cmp1
  jmp @lit_body
@lit_cmp1
  %lit_p1 =l addl %src, %j
  %lit_b1 =w loadub %lit_p1
  %lit_j1 =l addl %j, 1
  %lit_p2 =l addl %src, %lit_j1
  %lit_b2 =w loadub %lit_p2
  %lit_eq1 =w ceqw %lit_b1, %lit_b2
  jnz %lit_eq1, @lit_cmp2
  jmp @lit_body
@lit_cmp2
  %lit_j2b =l addl %j, 2
  %lit_p3 =l addl %src, %lit_j2b
  %lit_b3 =w loadub %lit_p3
  %lit_eq2 =w ceqw %lit_b2, %lit_b3
  jnz %lit_eq2, @lit_done
  jmp @lit_body
@lit_body
  %j_lb =l addl %j, 1
  jmp @lit_head
@lit_done
  %L =l subl %j, %i
  %lit_need0 =l addl %written, %L
  %lit_need =l addl %lit_need0, 1
  %lit_over =w csgtl %lit_need, %cap
  jnz %lit_over, @fail
  jmp @lit_write_hdr
@lit_write_hdr
  %lit_pout =l addl %out, %written
  %lit_hdr =l subl %L, 1
  storeb %lit_hdr, %lit_pout
  jmp @lit_copy_init
@lit_copy_init
  jmp @lit_copy_head
@lit_copy_head
  %k =l phi @lit_copy_init 0, @lit_copy_body %k_cb
  %copy_c =w csltl %k, %L
  jnz %copy_c, @lit_copy_body
  jmp @lit_adv
@lit_copy_body
  %copy_soff =l addl %i, %k
  %copy_ps =l addl %src, %copy_soff
  %copy_b =w loadub %copy_ps
  %copy_doff0 =l addl %written, %k
  %copy_doff =l addl %copy_doff0, 1
  %copy_pd =l addl %out, %copy_doff
  storeb %copy_b, %copy_pd
  %k_cb =l addl %k, 1
  jmp @lit_copy_head
@lit_adv
  %written_lit0 =l addl %written, %L
  %written_lit =l addl %written_lit0, 1
  %i_lit =l addl %i, %L
  jmp @outer
@fail
  ret %m1
@exit
  ret %written
}
