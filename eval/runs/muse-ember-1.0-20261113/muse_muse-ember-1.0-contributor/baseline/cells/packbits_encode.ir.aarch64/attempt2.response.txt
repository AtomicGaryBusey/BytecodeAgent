export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %z0 =l sub %n, %n
  %c257 =l add %z0, 257
  %m1 =l sub %z0, 1
  jmp @outer(0, 0)
@outer %i_o, %w_o
  %outer_lt =w csltl %i_o, %n
  jnz %outer_lt, @run_init(%i_o, %w_o), @exit(%w_o)
@exit %w_e
  ret %w_e
@run_init %i_ri, %w_ri
  %p0 =l add %src, %i_ri
  %b0 =w loadub %p0
  jmp @run_head(%i_ri, 0, %i_ri, %w_ri, %b0)
@run_head %idx_rh, %cnt_rh, %i_rh, %w_rh, %b0_rh
  %run_c1 =w csltl %cnt_rh, 128
  jnz %run_c1, @run_chk_end(%idx_rh, %cnt_rh, %i_rh, %w_rh, %b0_rh), @run_done(%cnt_rh, %i_rh, %w_rh, %b0_rh)
@run_chk_end %idx_rce, %cnt_rce, %i_rce, %w_rce, %b0_rce
  %run_c2 =w csltl %idx_rce, %n
  jnz %run_c2, @run_cmp(%idx_rce, %cnt_rce, %i_rce, %w_rce, %b0_rce), @run_done(%cnt_rce, %i_rce, %w_rce, %b0_rce)
@run_cmp %idx_rc, %cnt_rc, %i_rc, %w_rc, %b0_rc
  %pr =l add %src, %idx_rc
  %br =w loadub %pr
  %run_eq =w ceqw %br, %b0_rc
  jnz %run_eq, @run_body(%idx_rc, %cnt_rc, %i_rc, %w_rc, %b0_rc), @run_done(%cnt_rc, %i_rc, %w_rc, %b0_rc)
@run_body %idx_rb, %cnt_rb, %i_rb, %w_rb, %b0_rb
  %cnt_next =l add %cnt_rb, 1
  %idx_next =l add %idx_rb, 1
  jmp @run_head(%idx_next, %cnt_next, %i_rb, %w_rb, %b0_rb)
@run_done %cnt_rd, %i_rd, %w_rd, %b0_rd
  %is_rep =w csgtl %cnt_rd, 2
  jnz %is_rep, @rep_chk(%cnt_rd, %i_rd, %w_rd, %b0_rd), @lit_init(%i_rd, %w_rd)
@rep_chk %cnt_rcp, %i_rcp, %w_rcp, %b0_rcp
  %rep_need =l add %w_rcp, 2
  %rep_over =w csgtl %rep_need, %cap
  jnz %rep_over, @fail, @rep_write(%cnt_rcp, %i_rcp, %w_rcp, %b0_rcp)
@rep_write %cnt_rw, %i_rw, %w_rw, %b0_rw
  %rep_pout =l add %out, %w_rw
  %rep_hdr_l =l sub %c257, %cnt_rw
  %rep_hdr =w trunc %rep_hdr_l
  storeb %rep_hdr, %rep_pout
  %rep_pout2 =l add %rep_pout, 1
  storeb %b0_rw, %rep_pout2
  %written_rep =l add %w_rw, 2
  %i_rep =l add %i_rw, %cnt_rw
  jmp @outer(%i_rep, %written_rep)
@lit_init %i_li, %w_li
  jmp @lit_head(%i_li, %i_li, %w_li)
@lit_head %j_lh, %i_lh, %w_lh
  %lit_c1 =w csltl %j_lh, %n
  jnz %lit_c1, @lit_chk_len(%j_lh, %i_lh, %w_lh), @lit_done(%j_lh, %i_lh, %w_lh)
@lit_chk_len %j_lcl, %i_lcl, %w_lcl
  %lit_len =l sub %j_lcl, %i_lcl
  %lit_c2 =w csltl %lit_len, 128
  jnz %lit_c2, @lit_chk_run(%j_lcl, %i_lcl, %w_lcl), @lit_done(%j_lcl, %i_lcl, %w_lcl)
@lit_chk_run %j_lcr, %i_lcr, %w_lcr
  %lit_j2 =l add %j_lcr, 2
  %lit_c3 =w csltl %lit_j2, %n
  jnz %lit_c3, @lit_cmp1(%j_lcr, %i_lcr, %w_lcr), @lit_body(%j_lcr, %i_lcr, %w_lcr)
@lit_cmp1 %j_lc1, %i_lc1, %w_lc1
  %lit_p1 =l add %src, %j_lc1
  %lit_b1 =w loadub %lit_p1
  %lit_j1 =l add %j_lc1, 1
  %lit_p2 =l add %src, %lit_j1
  %lit_b2 =w loadub %lit_p2
  %lit_eq1 =w ceqw %lit_b1, %lit_b2
  jnz %lit_eq1, @lit_cmp2(%j_lc1, %i_lc1, %w_lc1, %lit_b2), @lit_body(%j_lc1, %i_lc1, %w_lc1)
@lit_cmp2 %j_lc2, %i_lc2, %w_lc2, %lit_b2_lc2
  %lit_j2b =l add %j_lc2, 2
  %lit_p3 =l add %src, %lit_j2b
  %lit_b3 =w loadub %lit_p3
  %lit_eq2 =w ceqw %lit_b2_lc2, %lit_b3
  jnz %lit_eq2, @lit_done(%j_lc2, %i_lc2, %w_lc2), @lit_body(%j_lc2, %i_lc2, %w_lc2)
@lit_body %j_lb, %i_lb, %w_lb
  %j_next =l add %j_lb, 1
  jmp @lit_head(%j_next, %i_lb, %w_lb)
@lit_done %j_ld, %i_ld, %w_ld
  %L_ld =l sub %j_ld, %i_ld
  %lit_need0 =l add %w_ld, %L_ld
  %lit_need =l add %lit_need0, 1
  %lit_over =w csgtl %lit_need, %cap
  jnz %lit_over, @fail, @lit_write_hdr(%L_ld, %i_ld, %w_ld)
@lit_write_hdr %L_lwh, %i_lwh, %w_lwh
  %lit_pout =l add %out, %w_lwh
  %lit_hdr_l =l sub %L_lwh, 1
  %lit_hdr =w trunc %lit_hdr_l
  storeb %lit_hdr, %lit_pout
  jmp @lit_copy_head(0, %L_lwh, %i_lwh, %w_lwh)
@lit_copy_head %k_lch, %L_lch, %i_lch, %w_lch
  %copy_c =w csltl %k_lch, %L_lch
  jnz %copy_c, @lit_copy_body(%k_lch, %L_lch, %i_lch, %w_lch), @lit_adv(%L_lch, %i_lch, %w_lch)
@lit_copy_body %k_lcb, %L_lcb, %i_lcb, %w_lcb
  %copy_soff =l add %i_lcb, %k_lcb
  %copy_ps =l add %src, %copy_soff
  %copy_b =w loadub %copy_ps
  %copy_doff0 =l add %w_lcb, %k_lcb
  %copy_doff =l add %copy_doff0, 1
  %copy_pd =l add %out, %copy_doff
  storeb %copy_b, %copy_pd
  %k_next =l add %k_lcb, 1
  jmp @lit_copy_head(%k_next, %L_lcb, %i_lcb, %w_lcb)
@lit_adv %L_la, %i_la, %w_la
  %written_lit0 =l add %w_la, %L_la
  %written_lit =l add %written_lit0, 1
  %i_lit =l add %i_la, %L_la
  jmp @outer(%i_lit, %written_lit)
@fail
  ret %m1
}
