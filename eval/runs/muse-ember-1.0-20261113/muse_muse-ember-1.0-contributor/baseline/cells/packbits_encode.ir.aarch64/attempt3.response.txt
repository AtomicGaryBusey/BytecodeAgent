export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %p_i =l alloc8 8
  %p_w =l alloc8 8
  %p_cnt =l alloc8 8
  %p_idx =l alloc8 8
  %p_j =l alloc8 8
  %p_k =l alloc8 8
  %zero =l sub %n, %n
  %one =l add %zero, 1
  %c2 =l add %zero, 2
  %c128 =l add %zero, 128
  %c257 =l add %zero, 257
  %m1 =l sub %zero, %one
  storel %zero, %p_i
  storel %zero, %p_w
  jmp @outer
@outer
  %i =l loadl %p_i
  %w =l loadl %p_w
  %outer_lt =w csltl %i, %n
  jnz %outer_lt, @run_init, @exit
@exit
  ret %w
@run_init
  %p_b0 =l add %src, %i
  %b0 =w loadub %p_b0
  storel %zero, %p_cnt
  storel %i, %p_idx
  jmp @run_head
@run_head
  %cnt =l loadl %p_cnt
  %idx =l loadl %p_idx
  %run_c1 =w csltl %cnt, %c128
  jnz %run_c1, @run_chk_end, @run_done
@run_chk_end
  %run_c2 =w csltl %idx, %n
  jnz %run_c2, @run_cmp, @run_done
@run_cmp
  %pr =l add %src, %idx
  %br =w loadub %pr
  %run_eq =w ceqw %br, %b0
  jnz %run_eq, @run_body, @run_done
@run_body
  %cnt_next =l add %cnt, %one
  %idx_next =l add %idx, %one
  storel %cnt_next, %p_cnt
  storel %idx_next, %p_idx
  jmp @run_head
@run_done
  %is_rep =w csgtl %cnt, %c2
  jnz %is_rep, @rep_chk, @lit_init
@rep_chk
  %rep_need =l add %w, %c2
  %rep_over =w csgtl %rep_need, %cap
  jnz %rep_over, @fail, @rep_write
@rep_write
  %rep_pout =l add %out, %w
  %rep_hdr_l =l sub %c257, %cnt
  %rep_hdr =w trunc %rep_hdr_l
  storeb %rep_hdr, %rep_pout
  %rep_pout2 =l add %rep_pout, %one
  storeb %b0, %rep_pout2
  %written_rep =l add %w, %c2
  %i_rep =l add %i, %cnt
  storel %written_rep, %p_w
  storel %i_rep, %p_i
  jmp @outer
@fail
  ret %m1
@lit_init
  storel %i, %p_j
  jmp @lit_head
@lit_head
  %j =l loadl %p_j
  %lit_c1 =w csltl %j, %n
  jnz %lit_c1, @lit_chk_len, @lit_done
@lit_chk_len
  %lit_len =l sub %j, %i
  %lit_c2 =w csltl %lit_len, %c128
  jnz %lit_c2, @lit_chk_run, @lit_done
@lit_chk_run
  %lit_j2 =l add %j, %c2
  %lit_c3 =w csltl %lit_j2, %n
  jnz %lit_c3, @lit_cmp1, @lit_body
@lit_cmp1
  %lit_p1 =l add %src, %j
  %lit_b1 =w loadub %lit_p1
  %lit_j1 =l add %j, %one
  %lit_p2 =l add %src, %lit_j1
  %lit_b2 =w loadub %lit_p2
  %lit_eq1 =w ceqw %lit_b1, %lit_b2
  jnz %lit_eq1, @lit_cmp2, @lit_body
@lit_cmp2
  %lit_j2b =l add %j, %c2
  %lit_p3 =l add %src, %lit_j2b
  %lit_b3 =w loadub %lit_p3
  %lit_eq2 =w ceqw %lit_b2, %lit_b3
  jnz %lit_eq2, @lit_done, @lit_body
@lit_body
  %j_next =l add %j, %one
  storel %j_next, %p_j
  jmp @lit_head
@lit_done
  %L =l sub %j, %i
  %lit_need0 =l add %w, %L
  %lit_need =l add %lit_need0, %one
  %lit_over =w csgtl %lit_need, %cap
  jnz %lit_over, @fail, @lit_write_hdr
@lit_write_hdr
  %lit_pout =l add %out, %w
  %lit_hdr_l =l sub %L, %one
  %lit_hdr =w trunc %lit_hdr_l
  storeb %lit_hdr, %lit_pout
  storel %zero, %p_k
  jmp @lit_copy_head
@lit_copy_head
  %k =l loadl %p_k
  %copy_c =w csltl %k, %L
  jnz %copy_c, @lit_copy_body, @lit_adv
@lit_copy_body
  %copy_soff =l add %i, %k
  %copy_ps =l add %src, %copy_soff
  %copy_b =w loadub %copy_ps
  %copy_doff0 =l add %w, %k
  %copy_doff =l add %copy_doff0, %one
  %copy_pd =l add %out, %copy_doff
  storeb %copy_b, %copy_pd
  %k_next =l add %k, %one
  storel %k_next, %p_k
  jmp @lit_copy_head
@lit_adv
  %written_lit0 =l add %w, %L
  %written_lit =l add %written_lit0, %one
  %i_lit =l add %i, %L
  storel %written_lit, %p_w
  storel %i_lit, %p_i
  jmp @outer
}
