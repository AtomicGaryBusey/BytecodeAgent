export function l $qsort_pairs(l %a, l %n) {
@start
  %c0 =w cslel %n, 1
  jnz %c0, @done, @cont
@done
  ret 0
@cont
  %p =l div %n, 2
  %last =l sub %n, 1
  %poff =l mul %p, 16
  %paddr =l add %a, %poff
  %paddrb =l add %paddr, 8
  %pk0 =l loadl %paddr
  %pv0 =l loadl %paddrb
  %loff0 =l mul %last, 16
  %laddr0 =l add %a, %loff0
  %laddr0b =l add %laddr0, 8
  %lk0 =l loadl %laddr0
  %lv0 =l loadl %laddr0b
  storel %lk0, %paddr
  storel %lv0, %paddrb
  storel %pk0, %laddr0
  storel %pv0, %laddr0b
  %pk =l loadl %laddr0
  %pv =l loadl %laddr0b
  jmp @loop
@loop
  %i =l phi @cont 0, @swapbody %inext, @next %inext2
  %j =l phi @cont 0, @swapbody %jnext, @next %jnext2
  %c1 =w csltl %j, %last
  jnz %c1, @body, @exit
@body
  %joff =l mul %j, 16
  %jaddr =l add %a, %joff
  %jaddrb =l add %jaddr, 8
  %kj =l loadl %jaddr
  %vj =l loadl %jaddrb
  %clt =w csltl %kj, %pk
  jnz %clt, @swapbody, @chkeq
@chkeq
  %ceq =w ceql %kj, %pk
  jnz %ceq, @chkval, @next
@chkval
  %cle =w cslel %vj, %pv
  jnz %cle, @swapbody, @next
@swapbody
  %inext =l add %i, 1
  %jnext =l add %j, 1
  %ioff =l mul %i, 16
  %iaddr =l add %a, %ioff
  %iaddrb =l add %iaddr, 8
  %ik =l loadl %iaddr
  %iv =l loadl %iaddrb
  %jk =l loadl %jaddr
  %jv =l loadl %jaddrb
  storel %jk, %iaddr
  storel %jv, %iaddrb
  storel %ik, %jaddr
  storel %iv, %jaddrb
  jmp @loop
@next
  %inext2 =l add %i, 0
  %jnext2 =l add %j, 1
  jmp @loop
@exit
  %ioff2 =l mul %i, 16
  %iaddr2 =l add %a, %ioff2
  %iaddrb2 =l add %iaddr2, 8
  %ik2 =l loadl %iaddr2
  %iv2 =l loadl %iaddrb2
  %loff2 =l mul %last, 16
  %laddr2 =l add %a, %loff2
  %laddrb2 =l add %laddr2, 8
  storel %pk, %iaddr2
  storel %pv, %iaddrb2
  storel %ik2, %laddr2
  storel %iv2, %laddrb2
  %rleft =l call $qsort_pairs(l %a, l %i)
  %iplus =l add %i, 1
  %roff =l mul %iplus, 16
  %raddr =l add %a, %roff
  %rsz =l sub %n, %iplus
  %rright =l call $qsort_pairs(l %raddr, l %rsz)
  ret 0
}
