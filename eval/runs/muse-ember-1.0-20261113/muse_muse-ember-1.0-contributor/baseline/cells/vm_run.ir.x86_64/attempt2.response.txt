export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
  jmp @loop
@loop
  %pc =l phi @start 0, @push %pc_push, @add %pc_add, @sub %pc_sub, @mul %pc_mul, @dup %pc_dup, @swap %pc_swap, @drop %pc_drop, @over %pc_over, @lt_true %pc_lt_true, @lt_false %pc_lt_false, @jz_taken %pc_jz_taken, @jz_not %pc_jz_not, @jump %pc_jump, @other %pc_other
  %sp =l phi @start 0, @push %sp_push, @add %spm1_add, @sub %spm1_sub, @mul %spm1_mul, @dup %sp_dup, @swap %sp_swap, @drop %spm1_drop, @over %sp_over, @lt_true %sp_lt_true, @lt_false %sp_lt_false, @jz_taken %sp_jz_taken, @jz_not %sp_jz_not, @jump %sp_jump, @other %sp_other
  %cneg =w csltl %pc, 0
  jnz %cneg, @done, @chk_end
@chk_end
  %clt =w csltl %pc, %len
  jnz %clt, @fetch, @done
@fetch
  %off_fetch =l mul %pc, 8
  %ptr_fetch =l add %code, %off_fetch
  %op =l loadl %ptr_fetch
  %e0 =w ceql %op, 0
  jnz %e0, @halt, @chk1
@chk1
  %e1 =w ceql %op, 1
  jnz %e1, @push, @chk2
@chk2
  %e2 =w ceql %op, 2
  jnz %e2, @add, @chk3
@chk3
  %e3 =w ceql %op, 3
  jnz %e3, @sub, @chk4
@chk4
  %e4 =w ceql %op, 4
  jnz %e4, @mul, @chk5
@chk5
  %e5 =w ceql %op, 5
  jnz %e5, @dup, @chk6
@chk6
  %e6 =w ceql %op, 6
  jnz %e6, @swap, @chk7
@chk7
  %e7 =w ceql %op, 7
  jnz %e7, @drop, @chk8
@chk8
  %e8 =w ceql %op, 8
  jnz %e8, @over, @chk9
@chk9
  %e9 =w ceql %op, 9
  jnz %e9, @lt, @chk10
@chk10
  %e10 =w ceql %op, 10
  jnz %e10, @jz, @chk11
@chk11
  %e11 =w ceql %op, 11
  jnz %e11, @jump, @other
@halt
  jmp @done
@push
  %pc1_push =l add %pc, 1
  %off_push =l mul %pc1_push, 8
  %ptr_push =l add %code, %off_push
  %v_push =l loadl %ptr_push
  %soff_push =l mul %sp, 8
  %sptr_push =l add %stack, %soff_push
  storel %v_push, %sptr_push
  %sp_push =l add %sp, 1
  %pc_push =l add %pc, 2
  jmp @loop
@add
  %spm1_add =l sub %sp, 1
  %offm1_add =l mul %spm1_add, 8
  %pm1_add =l add %stack, %offm1_add
  %b_add =l loadl %pm1_add
  %spm2_add =l sub %sp, 2
  %offm2_add =l mul %spm2_add, 8
  %pm2_add =l add %stack, %offm2_add
  %a_add =l loadl %pm2_add
  %r_add =l add %a_add, %b_add
  storel %r_add, %pm2_add
  %pc_add =l add %pc, 1
  jmp @loop
@sub
  %spm1_sub =l sub %sp, 1
  %offm1_sub =l mul %spm1_sub, 8
  %pm1_sub =l add %stack, %offm1_sub
  %b_sub =l loadl %pm1_sub
  %spm2_sub =l sub %sp, 2
  %offm2_sub =l mul %spm2_sub, 8
  %pm2_sub =l add %stack, %offm2_sub
  %a_sub =l loadl %pm2_sub
  %r_sub =l sub %a_sub, %b_sub
  storel %r_sub, %pm2_sub
  %pc_sub =l add %pc, 1
  jmp @loop
@mul
  %spm1_mul =l sub %sp, 1
  %offm1_mul =l mul %spm1_mul, 8
  %pm1_mul =l add %stack, %offm1_mul
  %b_mul =l loadl %pm1_mul
  %spm2_mul =l sub %sp, 2
  %offm2_mul =l mul %spm2_mul, 8
  %pm2_mul =l add %stack, %offm2_mul
  %a_mul =l loadl %pm2_mul
  %r_mul =l mul %a_mul, %b_mul
  storel %r_mul, %pm2_mul
  %pc_mul =l add %pc, 1
  jmp @loop
@dup
  %spm1_dup =l sub %sp, 1
  %off_dup =l mul %spm1_dup, 8
  %p_dup =l add %stack, %off_dup
  %v_dup =l loadl %p_dup
  %off2_dup =l mul %sp, 8
  %p2_dup =l add %stack, %off2_dup
  storel %v_dup, %p2_dup
  %sp_dup =l add %sp, 1
  %pc_dup =l add %pc, 1
  jmp @loop
@swap
  %i1_swap =l sub %sp, 1
  %off1_swap =l mul %i1_swap, 8
  %p1_swap =l add %stack, %off1_swap
  %v1_swap =l loadl %p1_swap
  %i2_swap =l sub %sp, 2
  %off2_swap =l mul %i2_swap, 8
  %p2_swap =l add %stack, %off2_swap
  %v2_swap =l loadl %p2_swap
  storel %v1_swap, %p2_swap
  storel %v2_swap, %p1_swap
  %sp_swap =l add %sp, 0
  %pc_swap =l add %pc, 1
  jmp @loop
@drop
  %spm1_drop =l sub %sp, 1
  %pc_drop =l add %pc, 1
  jmp @loop
@over
  %i_over =l sub %sp, 2
  %off_over =l mul %i_over, 8
  %p_over =l add %stack, %off_over
  %v_over =l loadl %p_over
  %off2_over =l mul %sp, 8
  %p2_over =l add %stack, %off2_over
  storel %v_over, %p2_over
  %sp_over =l add %sp, 1
  %pc_over =l add %pc, 1
  jmp @loop
@lt
  %spm1_lt =l sub %sp, 1
  %offm1_lt =l mul %spm1_lt, 8
  %pm1_lt =l add %stack, %offm1_lt
  %b_lt =l loadl %pm1_lt
  %spm2_lt =l sub %sp, 2
  %offm2_lt =l mul %spm2_lt, 8
  %pm2_lt =l add %stack, %offm2_lt
  %a_lt =l loadl %pm2_lt
  %cmp_lt =w csltl %a_lt, %b_lt
  jnz %cmp_lt, @lt_true, @lt_false
@lt_true
  storel 1, %pm2_lt
  %sp_lt_true =l sub %sp, 1
  %pc_lt_true =l add %pc, 1
  jmp @loop
@lt_false
  storel 0, %pm2_lt
  %sp_lt_false =l sub %sp, 1
  %pc_lt_false =l add %pc, 1
  jmp @loop
@jz
  %spm1_jz =l sub %sp, 1
  %off_jz =l mul %spm1_jz, 8
  %p_jz =l add %stack, %off_jz
  %v_jz =l loadl %p_jz
  %pc1_jz =l add %pc, 1
  %off2_jz =l mul %pc1_jz, 8
  %ptr_jz =l add %code, %off2_jz
  %t_jz =l loadl %ptr_jz
  %is0_jz =w ceql %v_jz, 0
  jnz %is0_jz, @jz_taken, @jz_not
@jz_taken
  %sp_jz_taken =l sub %sp, 1
  %pc_jz_taken =l add %t_jz, 0
  jmp @loop
@jz_not
  %sp_jz_not =l sub %sp, 1
  %pc_jz_not =l add %pc, 2
  jmp @loop
@jump
  %pc1_jump =l add %pc, 1
  %off_jump =l mul %pc1_jump, 8
  %ptr_jump =l add %code, %off_jump
  %t_jump =l loadl %ptr_jump
  %sp_jump =l add %sp, 0
  %pc_jump =l add %t_jump, 0
  jmp @loop
@other
  %sp_other =l add %sp, 0
  %pc_other =l add %pc, 1
  jmp @loop
@done
  %i_init =l add %sp, 0
  jmp @clear_loop
@clear_loop
  %i =l phi @done %i_init, @clear_body %i_next
  %c_clear =w csltl %i, %cap
  jnz %c_clear, @clear_body, @ret_check
@clear_body
  %off_clear =l mul %i, 8
  %ptr_clear =l add %stack, %off_clear
  storel 0, %ptr_clear
  %i_next =l add %i, 1
  jmp @clear_loop
@ret_check
  %is0_ret =w csltl %sp, 1
  jnz %is0_ret, @ret_zero, @ret_top
@ret_top
  %spm1_top =l sub %sp, 1
  %off_top =l mul %spm1_top, 8
  %ptr_top =l add %stack, %off_top
  %val_top =l loadl %ptr_top
  ret %val_top
@ret_zero
  %z_ret =l sub %sp, %sp
  ret %z_ret
}
