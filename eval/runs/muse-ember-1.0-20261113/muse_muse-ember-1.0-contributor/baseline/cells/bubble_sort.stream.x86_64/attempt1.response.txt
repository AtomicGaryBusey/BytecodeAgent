bubble_sort:
cmp rsi, 2
jl .Lret
mov r8, rsi
dec r8
.Louter:
test r8, r8
jz .Lret
xor r9, r9
.Linner:
cmp r9, r8
jge .Linner_done
mov rax, qword ptr [rdi + r9*8]
mov rdx, qword ptr [rdi + r9*8 + 8]
cmp rax, rdx
jle .Lno_swap
mov qword ptr [rdi + r9*8], rdx
mov qword ptr [rdi + r9*8 + 8], rax
.Lno_swap:
inc r9
jmp .Linner
.Linner_done:
dec r8
jmp .Louter
.Lret:
xor eax, eax
ret
