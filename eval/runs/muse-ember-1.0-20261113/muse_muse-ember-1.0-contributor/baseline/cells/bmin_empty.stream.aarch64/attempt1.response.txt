bmin_empty:
 cbz x1, .Lempty
 ldr x2, [x0]
 mov x3, #1
.Lloop:
 cmp x3, x1
 b.ge .Ldone
 ldr x4, [x0, x3, lsl #3]
 cmp x4, x2
 b.ge .Lskip
 mov x2, x4
.Lskip:
 add x3, x3, #1
 b .Lloop
.Ldone:
 mov x0, x2
 ret
.Lempty:
 movz x0, #0xffff
 movk x0, #0xffff, lsl #16
 movk x0, #0xffff, lsl #32
 movk x0, #0x7fff, lsl #48
 ret
