export function l $map_sum(l %a, l %n, l %f) {
@entry
  jmp @loop
@loop
  %i =l phi @entry 0, @body %i1
  %s =l phi @entry 0, @body %s1
  %c =w ceql %i, %n
  jnz %c, @exit, @body
@body
  %off =l mul %i, 8
  %p =l add %a, %off
  %v =l loadl %p
  %r =l call %f(l %v)
  %s1 =l add %s, %r
  %i1 =l add %i, 1
  jmp @loop
@exit
  %res =l phi @loop %s
  ret %res
}
