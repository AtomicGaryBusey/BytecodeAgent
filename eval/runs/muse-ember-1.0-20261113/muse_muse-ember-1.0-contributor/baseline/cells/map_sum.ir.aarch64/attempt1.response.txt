export function l $map_sum(l %a, l %n, l %f) {
@entry
  jmp @loop(%a, %n, %f, 0, 0)
@loop(%a1, %n1, %f1, %i, %s)
  %c =w ceql %i, %n1
  jnz %c, @exit, @body
@body
  %off =l mull %i, 8
  %p =l addl %a1, %off
  %v =l loadl %p
  %r =l call %f1(l %v)
  %s1 =l addl %s, %r
  %i1 =l addl %i, 1
  jmp @loop(%a1, %n1, %f1, %i1, %s1)
@exit
  ret %s
}
