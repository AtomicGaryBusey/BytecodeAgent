export function l $max_byte(l %p, l %n) {
@start
	%zero =l 0
	%iszero =w ceql %n, %zero
	jnz %iszero, @done, @loop
@loop
	%i =l phi @start %zero, @update %inext, @incr %inext
	%max =l phi @start %zero, @update %b, @incr %max
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%one =l 1
	%inext =l add %i, %one
	%lt =w csltl %max, %b
	jnz %lt, @update, @incr
@update
	jmp @loop
@incr
	jmp @loop
@done
	%result =l phi @start %zero, @loop %max
	ret %result
}
