export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%base =l copy %tmp
	%nnbytes =l mul %nn, 8
	%spare =l add %tmp, %nnbytes
	%i =l copy 0
	jmp @copy_loop
@copy_loop
	%c =w csltl %i, %nn
	jnz %c, @copy_body
	jmp @ident_init
@copy_body
	%off =l mul %i, 8
	%s =l add %a, %off
	%v =l loadl %s
	%d =l add %base, %off
	storel %v, %d
	%i =l add %i, 1
	jmp @copy_loop
@ident_init
	%i =l copy 0
	jmp @ident_loop
@ident_loop
	%c =w csltl %i, %nn
	jnz %c, @ident_body
	jmp @main_loop
@ident_body
	%off =l mul %i, 8
	%row =l div %i, %n
	%rn =l mul %row, %n
	%col =l sub %i, %rn
	%eq =w ceql %row, %col
	%d =l add %out, %off
	jnz %eq, @ident_one
	jmp @ident_zero
@ident_one
	storel 1, %d
	jmp @ident_next
@ident_zero
	storel 0, %d
	jmp @ident_next
@ident_next
	%i =l add %i, 1
	jmp @ident_loop
@main_loop
	%nz =w csltl 0, %k
	jnz %nz, @loop_body
	jmp @zero_init
@loop_body
	%odd =l and %k, 1
	%o =w csltl 0, %odd
	jnz %o, @mulres_init
	jmp @shift_k
@mulres_init
	%i =l copy 0
	jmp @mulres_iloop
@mulres_iloop
	%c =w csltl %i, %n
	jnz %c, @mulres_ibody
	jmp @copyout_init
@mulres_ibody
	%j =l copy 0
	jmp @mulres_jloop
@mulres_jloop
	%c =w csltl %j, %n
	jnz %c, @mulres_jbody
	jmp @mulres_inext
@mulres_jbody
	%sum =l copy 0
	%t =l copy 0
	jmp @mulres_tloop
@mulres_tloop
	%c =w csltl %t, %n
	jnz %c, @mulres_tbody
	jmp @mulres_jdone
@mulres_tbody
	%in =l mul %i, %n
	%idx1 =l add %in, %t
	%off1 =l mul %idx1, 8
	%p1 =l add %out, %off1
	%v1 =l loadl %p1
	%tn =l mul %t, %n
	%idx2 =l add %tn, %j
	%off2 =l mul %idx2, 8
	%p2 =l add %base, %off2
	%v2 =l loadl %p2
	%prod =l mul %v1, %v2
	%sum =l add %sum, %prod
	%t =l add %t, 1
	jmp @mulres_tloop
@mulres_jdone
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, 8
	%dst =l add %spare, %off
	storel %sum, %dst
	%j =l add %j, 1
	jmp @mulres_jloop
@mulres_inext
	%i =l add %i, 1
	jmp @mulres_iloop
@copyout_init
	%p =l copy 0
	jmp @copyout_loop
@copyout_loop
	%c =w csltl %p, %nn
	jnz %c, @copyout_body
	jmp @shift_k
@copyout_body
	%off =l mul %p, 8
	%s =l add %spare, %off
	%v =l loadl %s
	%d =l add %out, %off
	storel %v, %d
	%p =l add %p, 1
	jmp @copyout_loop
@shift_k
	%k =l shr %k, 1
	jmp @square_check
@square_check
	%nz =w csltl 0, %k
	jnz %nz, @mulsq_init
	jmp @main_loop
@mulsq_init
	%i =l copy 0
	jmp @mulsq_iloop
@mulsq_iloop
	%c =w csltl %i, %n
	jnz %c, @mulsq_ibody
	jmp @swap_done
@mulsq_ibody
	%j =l copy 0
	jmp @mulsq_jloop
@mulsq_jloop
	%c =w csltl %j, %n
	jnz %c, @mulsq_jbody
	jmp @mulsq_inext
@mulsq_jbody
	%sum =l copy 0
	%t =l copy 0
	jmp @mulsq_tloop
@mulsq_tloop
	%c =w csltl %t, %n
	jnz %c, @mulsq_tbody
	jmp @mulsq_jdone
@mulsq_tbody
	%in =l mul %i, %n
	%idx1 =l add %in, %t
	%off1 =l mul %idx1, 8
	%p1 =l add %base, %off1
	%v1 =l loadl %p1
	%tn =l mul %t, %n
	%idx2 =l add %tn, %j
	%off2 =l mul %idx2, 8
	%p2 =l add %base, %off2
	%v2 =l loadl %p2
	%prod =l mul %v1, %v2
	%sum =l add %sum, %prod
	%t =l add %t, 1
	jmp @mulsq_tloop
@mulsq_jdone
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, 8
	%dst =l add %spare, %off
	storel %sum, %dst
	%j =l add %j, 1
	jmp @mulsq_jloop
@mulsq_inext
	%i =l add %i, 1
	jmp @mulsq_iloop
@swap_done
	%tb =l copy %base
	%base =l copy %spare
	%spare =l copy %tb
	jmp @main_loop
@zero_init
	%total =l add %nn, %nn
	%p =l copy 0
	jmp @zero_loop
@zero_loop
	%c =w csltl %p, %total
	jnz %c, @zero_body
	jmp @done
@zero_body
	%off =l mul %p, 8
	%d =l add %tmp, %off
	storel 0, %d
	%p =l add %p, 1
	jmp @zero_loop
@done
	%r =l copy 0
	ret %r
}
