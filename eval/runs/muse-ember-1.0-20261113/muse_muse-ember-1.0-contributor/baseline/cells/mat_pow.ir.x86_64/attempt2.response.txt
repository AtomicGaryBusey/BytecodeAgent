export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nnbytes =l mul %nn, 8
	%spare =l add %tmp, %nnbytes
	%pc0 =l copy 0
	jmp @copy_head
@copy_head
	%pc =l phi @start %pc0, @copy_body %pc1
	%cc =w csltl %pc, %nn
	jnz %cc, @copy_body, @ident_start
@copy_body
	%offc =l mul %pc, 8
	%sc =l add %a, %offc
	%vc =l loadl %sc
	%dc =l add %tmp, %offc
	storel %vc, %dc
	%pc1 =l add %pc, 1
	jmp @copy_head
@ident_start
	%ii0 =l copy 0
	jmp @ident_ihead
@ident_ihead
	%ii =l phi @ident_start %ii0, @ident_inext %ii1
	%cic =w csltl %ii, %n
	jnz %cic, @ident_jstart, @main_start
@ident_jstart
	%ij0 =l copy 0
	jmp @ident_jhead
@ident_jhead
	%ij =l phi @ident_jstart %ij0, @ident_join %ij1
	%eqij =w ceql %ii, %ij
	%rowoff =l mul %ii, %n
	%idxij =l add %rowoff, %ij
	%offij =l mul %idxij, 8
	%dij =l add %out, %offij
	jnz %eqij, @ident_one, @ident_zero
@ident_one
	storel 1, %dij
	jmp @ident_join
@ident_zero
	storel 0, %dij
	jmp @ident_join
@ident_join
	%ij1 =l add %ij, 1
	jmp @ident_jhead
@ident_inext
	%ii1 =l add %ii, 1
	jmp @ident_ihead
@main_start
	%iter0 =l copy 0
	jmp @main_head
@main_head
	%iter =l phi @main_start %iter0, @main_next %iter1
	%cm =w csltl %iter, %k
	jnz %cm, @mul_start, @zero_start
@mul_start
	%mi0 =l copy 0
	jmp @mul_ihead
@mul_ihead
	%mi =l phi @mul_start %mi0, @mul_inext %mi1
	%cmi =w csltl %mi, %n
	jnz %cmi, @mul_jstart, @copyout_start
@mul_jstart
	%mj0 =l copy 0
	jmp @mul_jhead
@mul_jhead
	%mj =l phi @mul_jstart %mj0, @mul_jdone %mj1
	%cmj =w csltl %mj, %n
	jnz %cmj, @mul_tstart, @mul_inext
@mul_tstart
	%mt0 =l copy 0
	%msum0 =l copy 0
	jmp @mul_thead
@mul_thead
	%mt =l phi @mul_tstart %mt0, @mul_tbody %mt1
	%msum =l phi @mul_tstart %msum0, @mul_tbody %msum1
	%cmt =w csltl %mt, %n
	jnz %cmt, @mul_tbody, @mul_jdone
@mul_tbody
	%irow =l mul %mi, %n
	%idx1 =l add %irow, %mt
	%off1 =l mul %idx1, 8
	%p1 =l add %out, %off1
	%v1 =l loadl %p1
	%trow =l mul %mt, %n
	%idx2 =l add %trow, %mj
	%off2 =l mul %idx2, 8
	%p2 =l add %tmp, %off2
	%v2 =l loadl %p2
	%prod =l mul %v1, %v2
	%msum1 =l add %msum, %prod
	%mt1 =l add %mt, 1
	jmp @mul_thead
@mul_jdone
	%irow2 =l mul %mi, %n
	%idxd =l add %irow2, %mj
	%offd =l mul %idxd, 8
	%pd =l add %spare, %offd
	storel %msum, %pd
	%mj1 =l add %mj, 1
	jmp @mul_jhead
@mul_inext
	%mi1 =l add %mi, 1
	jmp @mul_ihead
@copyout_start
	%po0 =l copy 0
	jmp @copyout_head
@copyout_head
	%po =l phi @copyout_start %po0, @copyout_body %po1
	%cpo =w csltl %po, %nn
	jnz %cpo, @copyout_body, @main_next
@copyout_body
	%offo =l mul %po, 8
	%so =l add %spare, %offo
	%vo =l loadl %so
	%do =l add %out, %offo
	storel %vo, %do
	%po1 =l add %po, 1
	jmp @copyout_head
@main_next
	%iter1 =l add %iter, 1
	jmp @main_head
@zero_start
	%pz0 =l copy 0
	jmp @zero_head
@zero_head
	%pz =l phi @zero_start %pz0, @zero_body %pz1
	%total =l add %nn, %nn
	%cz =w csltl %pz, %total
	jnz %cz, @zero_body, @done
@zero_body
	%offz =l mul %pz, 8
	%dz =l add %tmp, %offz
	storel 0, %dz
	%pz1 =l add %pz, 1
	jmp @zero_head
@done
	%r0 =l copy 0
	ret %r0
}
