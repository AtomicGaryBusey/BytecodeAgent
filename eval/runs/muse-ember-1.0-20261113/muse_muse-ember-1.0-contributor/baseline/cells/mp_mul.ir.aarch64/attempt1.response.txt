export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %n =l add %na, %nb
  %zero =l sub %na, %na
  jmp @zero_head
@zero_head
  %zi = phi @start 0, @zero_body %zi_next
  %zp = phi @start %r, @zero_body %zp_next
  %zlt =w csltl %zi, %n
  jnz %zlt, @zero_body, @outer_head
@zero_body
  storel %zero, %zp
  %zi_next =l add %zi, 1
  %zp_next =l add %zp, 8
  jmp @zero_head
@outer_head
  %i = phi @zero_head 0, @outer_body %i_next
  %ilt =w csltl %i, %na
  jnz %ilt, @inner_setup, @scan_head
@inner_setup
  %offi =l mul %i, 8
  %pk0 =l add %r, %offi
  %aaddr =l add %a, %offi
  %ai =l loadl %aaddr
  jmp @inner_head
@inner_head
  %j = phi @inner_setup 0, @inner_body %j_next
  %carry = phi @inner_setup %zero, @inner_body %carry_next
  %pk = phi @inner_setup %pk0, @inner_body %pk_next
  %jlt =w csltl %j, %nb
  jnz %jlt, @inner_body, @outer_body
@inner_body
  %offj =l mul %j, 8
  %baddr =l add %b, %offj
  %bj =l loadl %baddr
  %ailo =l and %ai, 4294967295
  %aihi =l shr %ai, 32
  %bjlo =l and %bj, 4294967295
  %bjhi =l shr %bj, 32
  %p0 =l mul %ailo, %bjlo
  %p1 =l mul %ailo, %bjhi
  %p2 =l mul %aihi, %bjlo
  %p3 =l mul %aihi, %bjhi
  %p1s =l shl %p1, 32
  %p2s =l shl %p2, 32
  %t1 =l add %p0, %p1s
  %ca =w cultl %t1, %p0
  %t2 =l add %t1, %p2s
  %cb =w cultl %t2, %t1
  %p1h =l shr %p1, 32
  %p2h =l shr %p2, 32
  %cal =l extuwl %ca
  %cbl =l extuwl %cb
  %h1 =l add %p3, %p1h
  %h2 =l add %h1, %p2h
  %h3 =l add %h2, %cal
  %prodhi =l add %h3, %cbl
  %rk =l loadl %pk
  %s1 =l add %t2, %rk
  %cs1 =w cultl %s1, %t2
  %s2 =l add %s1, %carry
  %cs2 =w cultl %s2, %s1
  storel %s2, %pk
  %cs1l =l extuwl %cs1
  %cs2l =l extuwl %cs2
  %cn1 =l add %prodhi, %cs1l
  %carry_next =l add %cn1, %cs2l
  %j_next =l add %j, 1
  %pk_next =l add %pk, 8
  jmp @inner_head
@outer_body
  storel %carry, %pk
  %i_next =l add %i, 1
  jmp @outer_head
@scan_head
  %k = phi @outer_head 0, @scan_body %k_next
  %res = phi @outer_head %zero, @scan_body %res_next
  %ps = phi @outer_head %r, @scan_body %ps_next
  %klt =w csltl %k, %n
  jnz %klt, @scan_body, @done
@scan_body
  %rv =l loadl %ps
  %neq =w cnel %rv, %zero
  %ind =l extuwl %neq
  %k_next =l add %k, 1
  %diff =l sub %k_next, %res
  %corr =l mul %ind, %diff
  %res_next =l add %res, %corr
  %ps_next =l add %ps, 8
  jmp @scan_head
@done
  ret %res
}
