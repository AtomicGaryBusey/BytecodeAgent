export function l $count_desc(l %a, l %n) {
@start
  %zero =l copy 0
  %one =l copy 1
  jmp @loop
@loop
  %i =l phi @start %zero, @body %inext
  %cnt =l phi @start %zero, @body %cntnext
  %lim =l sub %n, %one
  %cond =w csltl %i, %lim
  jnz %cond, @body, @exit
@body
  %off =l mul %i, 8
  %p1 =l add %a, %off
  %x =l loadl %p1
  %inext =l add %i, %one
  %off2 =l mul %inext, 8
  %p2 =l add %a, %off2
  %y =l loadl %p2
  %gt =w csgtl %x, %y
  %gtl =l extuw %gt
  %cntnext =l add %cnt, %gtl
  jmp @loop
@exit
  ret %cnt
}
