packbits_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    mov r11, rcx
    xor ebx, ebx
    xor r12d, r12d
.Lmain:
    cmp rbx, r9
    jge .Ldone
    movzx eax, byte ptr [r8 + rbx]
    mov r13, 1
.Lrun:
    cmp r13, 128
    jge .LrunDone
    mov rcx, rbx
    add rcx, r13
    cmp rcx, r9
    jge .LrunDone
    cmp byte ptr [r8 + rcx], al
    jne .LrunDone
    inc r13
    jmp .Lrun
.LrunDone:
    cmp r13, 3
    jge .Lrepeat
    mov r14, rbx
.Llit:
    cmp r14, r9
    jge .LlitDone
    mov rax, r14
    sub rax, rbx
    cmp rax, 128
    jge .LlitDone
    movzx r15d, byte ptr [r8 + r14]
    mov r13, 1
.LlitRun:
    cmp r13, 128
    jge .LlitRunDone
    mov rcx, r14
    add rcx, r13
    cmp rcx, r9
    jge .LlitRunDone
    cmp byte ptr [r8 + rcx], r15b
    jne .LlitRunDone
    inc r13
    jmp .LlitRun
.LlitRunDone:
    cmp r13, 3
    jge .LlitDone
    inc r14
    jmp .Llit
.LlitDone:
    mov r13, r14
    sub r13, rbx
    mov rax, r12
    add rax, r13
    inc rax
    cmp rax, r11
    jg .Lerror
    mov rdi, r8
    add rdi, rbx
    mov rsi, r10
    add rsi, r12
    mov edx, r13d
    sub dl, 1
    mov byte ptr [rsi], dl
    inc rsi
.Lcopy:
    test r13, r13
    je .LcopyDone
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    inc rdi
    inc rsi
    dec r13
    jmp .Lcopy
.LcopyDone:
    mov r12, rax
    mov rbx, r14
    jmp .Lmain
.Lrepeat:
    mov rax, r12
    add rax, 2
    cmp rax, r11
    jg .Lerror
    mov rcx, r10
    add rcx, r12
    mov edx, 257
    sub rdx, r13
    mov byte ptr [rcx], dl
    movzx edx, byte ptr [r8 + rbx]
    mov byte ptr [rcx + 1], dl
    add r12, 2
    add rbx, r13
    jmp .Lmain
.Lerror:
    mov rax, -1
    jmp .Lexit
.Ldone:
    mov rax, r12
.Lexit:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
