export function l $crc32(l %p, l %n) {
@start
  jmp @loop
@loop
  %i =l phi @start 0, @next %i_next
  %crc =w phi @start -1, @next %crc_next
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %ptr =l add %p, %i
  %byte =w loadub %ptr
  %c0 =w xor %crc, %byte
  jmp @it0
@it0
  %bit0 =w and %c0, 1
  jnz %bit0, @it0_if, @it0_else
@it0_if
  %shr0_if =w shr %c0, 1
  %c1_if =w xor %shr0_if, -306659912
  jmp @it1
@it0_else
  %c1_else =w shr %c0, 1
  jmp @it1
@it1
  %c1 =w phi @it0_if %c1_if, @it0_else %c1_else
  %bit1 =w and %c1, 1
  jnz %bit1, @it1_if, @it1_else
@it1_if
  %shr1_if =w shr %c1, 1
  %c2_if =w xor %shr1_if, -306659912
  jmp @it2
@it1_else
  %c2_else =w shr %c1, 1
  jmp @it2
@it2
  %c2 =w phi @it1_if %c2_if, @it1_else %c2_else
  %bit2 =w and %c2, 1
  jnz %bit2, @it2_if, @it2_else
@it2_if
  %shr2_if =w shr %c2, 1
  %c3_if =w xor %shr2_if, -306659912
  jmp @it3
@it2_else
  %c3_else =w shr %c2, 1
  jmp @it3
@it3
  %c3 =w phi @it2_if %c3_if, @it2_else %c3_else
  %bit3 =w and %c3, 1
  jnz %bit3, @it3_if, @it3_else
@it3_if
  %shr3_if =w shr %c3, 1
  %c4_if =w xor %shr3_if, -306659912
  jmp @it4
@it3_else
  %c4_else =w shr %c3, 1
  jmp @it4
@it4
  %c4 =w phi @it3_if %c4_if, @it3_else %c4_else
  %bit4 =w and %c4, 1
  jnz %bit4, @it4_if, @it4_else
@it4_if
  %shr4_if =w shr %c4, 1
  %c5_if =w xor %shr4_if, -306659912
  jmp @it5
@it4_else
  %c5_else =w shr %c4, 1
  jmp @it5
@it5
  %c5 =w phi @it4_if %c5_if, @it4_else %c5_else
  %bit5 =w and %c5, 1
  jnz %bit5, @it5_if, @it5_else
@it5_if
  %shr5_if =w shr %c5, 1
  %c6_if =w xor %shr5_if, -306659912
  jmp @it6
@it5_else
  %c6_else =w shr %c5, 1
  jmp @it6
@it6
  %c6 =w phi @it5_if %c6_if, @it5_else %c6_else
  %bit6 =w and %c6, 1
  jnz %bit6, @it6_if, @it6_else
@it6_if
  %shr6_if =w shr %c6, 1
  %c7_if =w xor %shr6_if, -306659912
  jmp @it7
@it6_else
  %c7_else =w shr %c6, 1
  jmp @it7
@it7
  %c7 =w phi @it6_if %c7_if, @it6_else %c7_else
  %bit7 =w and %c7, 1
  jnz %bit7, @it7_if, @it7_else
@it7_if
  %shr7_if =w shr %c7, 1
  %crc_next_if =w xor %shr7_if, -306659912
  jmp @next
@it7_else
  %crc_next_else =w shr %c7, 1
  jmp @next
@next
  %crc_next =w phi @it7_if %crc_next_if, @it7_else %crc_next_else
  %i_next =l add %i, 1
  jmp @loop
@exit
  %res =w xor %crc, -1
  %rl =l extuw %res
  ret %rl
}
