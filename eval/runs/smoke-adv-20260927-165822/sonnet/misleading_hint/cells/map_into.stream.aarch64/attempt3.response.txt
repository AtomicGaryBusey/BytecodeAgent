map_into:
    sub sp, sp, #48
    stp x29, x30, [sp]
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]

    mov x19, x0        // dst
    mov x20, x1        // src
    mov x21, x2        // n
    mov x22, x3        // f
    mov x9, xzr         // count
    mov x10, xzr        // i

.Lloop:
    cmp x10, x21
    b.ge .Ldone

    stp x9, x10, [sp, #-16]!   // save count, i across call
    ldr x0, [x20, x10, lsl #3]   // src[i]
    blr x22
    ldp x9, x10, [sp], #16

    str x0, [x19, x10, lsl #3]  // dst[i] = f(src[i])

    ldr x11, [x20, x10, lsl #3]  // reload src[i]
    cmp x0, x11
    b.ge .Lskip
    add x9, x9, #1
.Lskip:
    add x10, x10, #1
    b .Lloop

.Ldone:
    mov x0, x9

    ldp x29, x30, [sp]
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    add sp, sp, #48
    ret
