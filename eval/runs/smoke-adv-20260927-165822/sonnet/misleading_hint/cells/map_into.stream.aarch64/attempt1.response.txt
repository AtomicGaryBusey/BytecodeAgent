map_into:
    sub sp, sp, #64
    stp x29, x30, [sp]
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    str x23, [sp, #40]

    mov x19, x0        // dst
    mov x20, x1        // src
    mov x21, x2        // n
    mov x18, x3        // f (spare register, survives calls)
    mov x22, xzr       // count
    mov x23, xzr       // i

.Lloop:
    cmp x23, x21
    b.ge .Ldone

    ldr x0, [x20, x23, lsl #3]   // src[i]
    blr x18
    str x0, [x19, x23, lsl #3]  // dst[i] = f(src[i])

    ldr x9, [x20, x23, lsl #3]  // reload src[i]
    cmp x0, x9
    b.ge .Lskip
    add x22, x22, #1
.Lskip:
    add x23, x23, #1
    b .Lloop

.Ldone:
    mov x0, x22

    ldp x29, x30, [sp]
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldr x23, [sp, #40]
    add sp, sp, #64
    ret
