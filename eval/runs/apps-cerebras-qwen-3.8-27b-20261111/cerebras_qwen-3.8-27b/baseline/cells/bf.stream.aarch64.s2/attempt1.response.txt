

.equ FD_IN, 0
.equ FD_OUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ CH_BANG, 0x21
.equ CH_GT, 0x3e
.equ CH_LT, 0x3c
.equ CH_PLUS, 0x2b
.equ CH_MINUS, 0x2d
.equ CH_DOT, 0x2e
.equ CH_COMMA, 0x2c
.equ CH_OPEN, 0x5b
.equ CH_CLOSE, 0x5d

.bss
.balign 8
input:
    .space 65536
tape:
    .space 30000
match_open:
    .space 262144
match_close:
    .space 262144
brack_stack:
    .space 16384

.text
.globl _start
_start:
    adrp x16, input
    add x16, x16, :lo12:input
    mov x20, x16
    mov x19, #0
    mov x16, #0x10000
read_loop:
    cmp x19, x16
    b.ge read_done
    add x1, x20, x19
    sub x2, x16, x19
    mov x0, FD_IN
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done
    cmp x0, #0
    b.lt invalid_exit
    add x19, x19, x0
    b read_loop
read_done:
    adrp x16, brack_stack
    add x16, x16, :lo12:brack_stack
    mov x29, x16
    adrp x16, match_open
    add x16, x16, :lo12:match_open
    mov x27, x16
    adrp x16, match_close
    add x16, x16, :lo12:match_close
    mov x28, x16
    mov x21, #0
    mov x23, #0
scan_loop:
    cmp x21, x19
    b.ge scan_done
    ldrb w0, [x20, x21]
    cmp w0, CH_BANG
    b.eq bang_found
    cmp w0, CH_OPEN
    beq scan_open
    cmp w0, CH_CLOSE
    beq scan_close
    add x21, x21, #1
    b scan_loop
scan_open:
    str w21, [x29, x23, lsl #2]
    add x23, x23, #1
    add x21, x21, #1
    b scan_loop
scan_close:
    cbz x23, invalid_exit
    sub x23, x23, #1
    ldr w1, [x29, x23, lsl #2]
    add w2, w21, #1
    str w2, [x27, x1, lsl #2]
    add w2, w1, #1
    str w2, [x28, x21, lsl #2]
    add x21, x21, #1
    b scan_loop
bang_found:
    mov x24, x21
    add x25, x20, x21
    add x25, x25, #1
    cmp x23, #0
    b.ne invalid_exit
    b run_setup
scan_done:
    mov x24, x19
    add x25, x20, x19
    cmp x23, #0
    b.ne invalid_exit
    b run_setup
run_setup:
    mov x18, #0
    mov x21, #0
    add x26, x20, x19
    adrp x16, tape
    add x16, x16, :lo12:tape
    mov x22, x16
    b exec_loop
exec_loop:
    cmp x21, x24
    b.ge end_prog
    ldrb w0, [x20, x21]
    cmp w0, CH_GT
    beq cmd_gt
    cmp w0, CH_LT
    beq cmd_lt
    cmp w0, CH_PLUS
    beq cmd_plus
    cmp w0, CH_MINUS
    beq cmd_minus
    cmp w0, CH_DOT
    beq cmd_dot
    cmp w0, CH_COMMA
    beq cmd_comma
    cmp w0, CH_OPEN
    beq cmd_open
    cmp w0, CH_CLOSE
    beq cmd_close
    b next_cmd
cmd_gt:
    add x18, x18, #1
    b next_cmd
cmd_lt:
    sub x18, x18, #1
    b next_cmd
cmd_plus:
    ldrb w0, [x22, x18]
    add w0, w0, #1
    strb w0, [x22, x18]
    b next_cmd
cmd_minus:
    ldrb w0, [x22, x18]
    sub w0, w0, #1
    strb w0, [x22, x18]
    b next_cmd
cmd_dot:
    add x1, x22, x18
    mov x0, FD_OUT
    mov x2, #1
    mov x8, SYS_WRITE
    svc #0
    b next_cmd
cmd_comma:
    cmp x25, x26
    b.ge comma_eof
    ldrb w0, [x25]
    add x25, x25, #1
    strb w0, [x22, x18]
    b next_cmd
comma_eof:
    mov w0, #0
    strb w0, [x22, x18]
    b next_cmd
cmd_open:
    ldrb w0, [x22, x18]
    cbnz w0, next_cmd
    ldr w0, [x27, x21, lsl #2]
    mov x21, x0
    b exec_loop
cmd_close:
    ldrb w0, [x22, x18]
    cbz w0, next_cmd
    ldr w0, [x28, x21, lsl #2]
    mov x21, x0
    b exec_loop
next_cmd:
    add x21, x21, #1
    b exec_loop
end_prog:
    mov x0, #0
    mov x8, SYS_EXIT
    svc #0
invalid_exit:
    mov x0, #1
    mov x8, SYS_EXIT
    svc #0
