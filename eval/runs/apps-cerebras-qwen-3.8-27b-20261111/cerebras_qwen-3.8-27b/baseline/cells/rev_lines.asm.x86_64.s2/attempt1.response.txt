

.text
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ IN_FD, 0
.equ OUT_FD, 1
.equ INLEN, 65537
.equ OUTLEN, 65537

_start:
    lea rsi, [rip + inbuf]
    lea r12, [rip + inbuf + INLEN]
    mov rdi, IN_FD
    xor rbx, rbx

read_loop:
    mov rdx, r12
    sub rdx, rsi
    test rdx, rdx
    jz input_done

    mov rax, SYS_READ
    syscall

    test rax, rax
    jz input_done

    add rsi, rax
    add rbx, rax
    jmp read_loop

input_done:
    lea rdi, [rip + inbuf]
    lea rsi, [rip + outbuf]
    xor r13d, r13d

next_line:
    cmp r13, rbx
    je clean_exit

    mov rcx, r13

find_end:
    cmp rcx, rbx
    je no_newline

    movzx eax, byte ptr [rdi + rcx]
    cmp al, 10
    je have_newline

    inc rcx
    jmp find_end

no_newline:
have_newline:
    mov r8, rcx
    dec r8

reverse_loop:
    cmp r8, r13
    jl write_newline

    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rsi], al
    inc rsi
    dec r8
    jmp reverse_loop

write_newline:
    mov byte ptr [rsi], 10
    inc rsi

    cmp rcx, rbx
    je clean_exit

    lea r13, [rcx + 1]
    jmp next_line

clean_exit:
    mov rdx, rsi
    lea rsi, [rip + outbuf]
    sub rdx, rsi

    test rdx, rdx
    jz exit_now

    mov rdi, OUT_FD
    mov rax, SYS_WRITE
    syscall

exit_now:
    mov rax, SYS_EXIT
    xor edi, edi
    syscall

.bss
.align 8
inbuf:
.zero INLEN
outbuf:
.zero OUTLEN
