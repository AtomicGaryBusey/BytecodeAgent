

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT_GROUP, 231
.equ BUF_SIZE, 65536
.equ FLAG_WS, 1
.equ FLAG_NL, 2
.equ FLAG_WSNL, 3

.bss
.align 16
flag_table:
.zero 256
buffer:
.zero BUF_SIZE
num_buf:
.zero 16
out_buffer:
.zero 32

.text
.globl _start
_start:
    lea r8, [rip + flag_table]
    mov byte ptr [r8 + 9], FLAG_WS
    mov byte ptr [r8 + 10], FLAG_WSNL
    mov byte ptr [r8 + 11], FLAG_WS
    mov byte ptr [r8 + 12], FLAG_WS
    mov byte ptr [r8 + 13], FLAG_WS
    mov byte ptr [r8 + 32], FLAG_WS

    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor r10, r10
    lea r12, [rip + buffer]

read_loop:
    mov eax, SYS_READ
    xor edi, edi
    mov rsi, r12
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz input_done
    add r15, rax
    mov rcx, rax
    xor esi, esi

process_loop:
    movzx eax, byte ptr [r12 + rsi]
    movzx r9, byte ptr [r8 + rax]
    test r9, r9
    jz not_whitespace

    xor r10, r10
    test r9, FLAG_NL
    jz process_next
    inc r13

process_next:
    inc rsi
    dec rcx
    jnz process_loop
    jmp read_loop

not_whitespace:
    test r10, r10
    jnz notws_next
    inc r14
    mov r10, 1

notws_next:
    inc rsi
    dec rcx
    jnz process_loop
    jmp read_loop

input_done:
    lea r10, [rip + out_buffer]
    mov rbx, r10
    mov r8, 10

    mov rdi, r13
    mov rsi, rbx
    call print_uint
    mov rbx, rax
    mov byte ptr [rbx], 32
    inc rbx

    mov rdi, r14
    mov rsi, rbx
    call print_uint
    mov rbx, rax
    mov byte ptr [rbx], 32
    inc rbx

    mov rdi, r15
    mov rsi, rbx
    call print_uint
    mov rbx, rax
    mov byte ptr [rbx], 10
    inc rbx

    sub rbx, r10
    mov rdx, rbx
    mov rsi, r10
    mov eax, SYS_WRITE
    mov edi, 1
    syscall

    mov eax, SYS_EXIT_GROUP
    xor edi, edi
    syscall

print_uint:
    lea rcx, [rip + num_buf + 15]
    lea r11, [rip + num_buf + 15]
print_div_loop:
    xor rdx, rdx
    div r8
    mov rbx, rax
    mov al, dl
    add al, 48
    mov byte ptr [rcx], al
    dec rcx
    test rbx, rbx
    jnz print_div_loop

    sub r11, rcx
print_copy_loop:
    inc rcx
    movzx eax, byte ptr [rcx]
    mov byte ptr [rsi], al
    inc rsi
    dec r11
    jnz print_copy_loop

    mov rax, rsi
    ret
