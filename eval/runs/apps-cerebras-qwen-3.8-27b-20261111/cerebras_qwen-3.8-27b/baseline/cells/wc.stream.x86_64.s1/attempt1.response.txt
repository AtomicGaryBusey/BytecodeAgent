

.equ SYSCALL_READ, 0
.equ SYSCALL_WRITE, 1
.equ SYSCALL_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ EXIT_OK, 0
.equ BUF_SIZE, 65536
.equ NUMBUF_SIZE, 32
.equ OUTBUF_SIZE, 32
.equ C0, 48
.equ SPACE, 32
.equ NEWLINE, 10
.equ WS_TAB, 9
.equ WS_VTAB, 11
.equ WS_FF, 12
.equ WS_CR, 13

.globl _start

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov edi, STDIN
    lea rsi, [rip + inbuf]
    mov edx, BUF_SIZE
    mov eax, SYSCALL_READ
    syscall
    test rax, rax
    jz input_done
    mov rcx, rax
    lea rsi, [rip + inbuf]

process_byte:
    movzx eax, byte ptr [rsi]
    inc r14
    cmp al, NEWLINE
    je newline_seen
    cmp al, WS_TAB
    je whitespace_seen
    cmp al, WS_VTAB
    je whitespace_seen
    cmp al, WS_FF
    je whitespace_seen
    cmp al, WS_CR
    je whitespace_seen
    cmp al, SPACE
    je whitespace_seen
    test r15b, r15b
    jne nonws_done
    inc r13
    mov r15b, 1
nonws_done:
    jmp next_byte
newline_seen:
    inc r12
    xor r15b, r15b
    jmp next_byte
whitespace_seen:
    xor r15b, r15b
next_byte:
    inc rsi
    dec rcx
    jnz process_byte
    jmp read_loop

input_done:
    lea rsi, [rip + outbuf]
    mov rdi, r12
    call emit_decimal
    mov al, SPACE
    mov byte ptr [rsi], al
    inc rsi
    mov rdi, r13
    call emit_decimal
    mov al, SPACE
    mov byte ptr [rsi], al
    inc rsi
    mov rdi, r14
    call emit_decimal
    mov al, NEWLINE
    mov byte ptr [rsi], al
    inc rsi
    mov r10, rsi
    lea rsi, [rip + outbuf]
    sub r10, rsi
    mov rdx, r10
    mov edi, STDOUT
    mov eax, SYSCALL_WRITE
    syscall
    mov edi, EXIT_OK
    mov eax, SYSCALL_EXIT
    syscall

emit_decimal:
    lea r8, [rip + numbuf]
    mov rax, rdi
    xor r10, r10
    test rax, rax
    jz decimal_zero
decimal_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, C0
    mov byte ptr [r8 + r10], dl
    inc r10
    test rax, rax
    jnz decimal_loop
    jmp decimal_copy
decimal_zero:
    mov dl, C0
    mov byte ptr [r8], dl
    mov r10, 1
decimal_copy:
    dec r10
decimal_copy_loop:
    movzx eax, byte ptr [r8 + r10]
    mov byte ptr [rsi], al
    inc rsi
    dec r10
    jns decimal_copy_loop
    ret

.bss
.align 16
inbuf:
.space BUF_SIZE
numbuf:
.space NUMBUF_SIZE
outbuf:
.space OUTBUF_SIZE
