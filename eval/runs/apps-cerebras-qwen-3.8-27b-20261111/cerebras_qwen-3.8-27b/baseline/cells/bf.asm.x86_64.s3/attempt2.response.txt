

.globl _start

.equ INBUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ OUTBUF_SIZE, 65536
.equ MATCH_BYTES, 262144
.equ BSTACK_BYTES, 4096

.equ BANG, 33
.equ OP_GT, 62
.equ OP_LT, 60
.equ OP_PLUS, 43
.equ OP_MINUS, 45
.equ OP_DOT, 46
.equ OP_COMMA, 44
.equ OP_LB, 91
.equ OP_RB, 93

.bss
.align 16
inbuf: .space INBUF_SIZE
.align 16
tape: .space TAPE_SIZE
.align 16
outbuf: .space OUTBUF_SIZE
.align 16
match: .space MATCH_BYTES
.align 16
bstack: .space BSTACK_BYTES

.text

_start:
    xor ebx, ebx

read_next:
    cmp ebx, INBUF_SIZE
    je read_done

    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, rbx

    mov edx, INBUF_SIZE
    sub edx, ebx

    mov eax, 0
    syscall

    test rax, rax
    jz read_done

    add rbx, rax
    jmp read_next

read_done:
    xor esi, esi

find_bang:
    cmp esi, ebx
    jge no_bang

    movzx eax, byte ptr [inbuf + rsi]
    cmp al, BANG
    je bang_found

    inc esi
    jmp find_bang

no_bang:
    mov r8d, ebx
    mov r9d, ebx
    xor r10d, r10d
    jmp validate_start

bang_found:
    mov r8d, esi
    mov r9d, esi
    inc r9d

    mov r10d, ebx
    sub r10d, r8d
    dec r10d

    jmp validate_start

validate_start:
    xor esi, esi
    xor r12d, r12d

validate_loop:
    cmp esi, r8d
    jge validate_ok_check

    movzx eax, byte ptr [inbuf + rsi]

    cmp al, OP_LB
    je v_open

    cmp al, OP_RB
    je v_close

    inc esi
    jmp validate_loop

v_open:
    mov dword ptr [bstack + r12*4], esi
    inc r12d
    inc esi
    jmp validate_loop

v_close:
    cmp r12d, 0
    je exit1

    dec r12d
    mov r13d, dword ptr [bstack + r12*4]

    lea edx, [esi + 1]
    mov dword ptr [match + r13*4], edx

    lea edx, [r13 + 1]
    mov dword ptr [match + esi*4], edx

    inc esi
    jmp validate_loop

validate_ok_check:
    cmp r12d, 0
    jne exit1

execute_start:
    xor ebx, ebx
    xor esi, esi
    xor r11d, r11d
    xor r12d, r12d

execute_loop:
    cmp esi, r8d
    jge execute_done

    movzx eax, byte ptr [inbuf + rsi]

    cmp al, OP_GT
    je cmd_gt

    cmp al, OP_LT
    je cmd_lt

    cmp al, OP_PLUS
    je cmd_plus

    cmp al, OP_MINUS
    je cmd_minus

    cmp al, OP_DOT
    je cmd_dot

    cmp al, OP_COMMA
    je cmd_comma

    cmp al, OP_LB
    je cmd_lbracket

    cmp al, OP_RB
    je cmd_rbracket

    inc esi
    jmp execute_loop

cmd_gt:
    inc ebx
    inc esi
    jmp execute_loop

cmd_lt:
    dec ebx
    inc esi
    jmp execute_loop

cmd_plus:
    inc byte ptr [tape + rbx]
    inc esi
    jmp execute_loop

cmd_minus:
    dec byte ptr [tape + rbx]
    inc esi
    jmp execute_loop

cmd_dot:
    movzx eax, byte ptr [tape + rbx]
    mov byte ptr [outbuf + r12], al
    inc r12d
    inc esi
    jmp execute_loop

cmd_comma:
    cmp r11d, r10d
    jge comma_zero

    mov r13d, r9d
    add r13d, r11d

    movzx eax, byte ptr [inbuf + r13]
    mov byte ptr [tape + rbx], al

    inc r11d
    inc esi
    jmp execute_loop

comma_zero:
    mov byte ptr [tape + rbx], 0
    inc esi
    jmp execute_loop

cmd_lbracket:
    movzx eax, byte ptr [tape + rbx]
    test al, al
    jz jump_lbracket

    inc esi
    jmp execute_loop

jump_lbracket:
    mov r13d, dword ptr [match + rsi*4]
    mov esi, r13d
    jmp execute_loop

cmd_rbracket:
    movzx eax, byte ptr [tape + rbx]
    test al, al
    jz no_jump_rbracket

    mov r13d, dword ptr [match + rsi*4]
    mov esi, r13d
    jmp execute_loop

no_jump_rbracket:
    inc esi
    jmp execute_loop

execute_done:
    cmp r12d, 0
    je exit0

    mov edi, 1
    lea rsi, [rip + outbuf]
    mov edx, r12d
    mov eax, 1
    syscall

exit0:
    mov eax, 60
    xor edi, edi
    syscall

exit1:
    mov eax, 60
    mov edi, 1
    syscall
