

.equ SYSCALL_READ, 63
.equ SYSCALL_WRITE, 64
.equ SYSCALL_EXIT, 93
.equ BUF_SIZE, 0x10001

.text
.globl _start

_start:
    adrp x10, ibuf
    add x10, x10, :lo12:ibuf
    adrp x11, obuf
    add x11, x11, :lo12:obuf

    movz x19, #1, lsl #16
    mov x12, xzr

read_loop:
    cmp x12, x19
    b.ge read_done
    add x1, x10, x12
    sub x2, x19, x12
    mov x0, xzr
    mov x8, #SYSCALL_READ
    svc #0
    cmp x0, xzr
    b.le read_done
    add x12, x12, x0
    b read_loop

read_done:
    mov x13, xzr
    mov x14, xzr
    mov x15, xzr

scan_loop:
    cmp x15, x12
    b.ge scan_done
    ldrb w17, [x10, x15]
    cmp w17, #10
    b.ne scan_step

    sub x16, x15, x14
    cbz x16, append_newline
    mov x17, x15
    sub x17, x17, #1
copy_reverse:
    ldrb w18, [x10, x17]
    strb w18, [x11, x13]
    add x13, x13, #1
    sub x17, x17, #1
    sub x16, x16, #1
    cmp x16, xzr
    b.ne copy_reverse

append_newline:
    mov w17, #10
    strb w17, [x11, x13]
    add x13, x13, #1
    add x14, x15, #1

scan_step:
    add x15, x15, #1
    b scan_loop

scan_done:
    sub x16, x12, x14
    cbz x16, write_output
    mov x17, x12
    sub x17, x17, #1
copy_final:
    ldrb w18, [x10, x17]
    strb w18, [x11, x13]
    add x13, x13, #1
    sub x17, x17, #1
    sub x16, x16, #1
    cmp x16, xzr
    b.ne copy_final

    mov w17, #10
    strb w17, [x11, x13]
    add x13, x13, #1

write_output:
    cmp x13, xzr
    beq exit_zero
    mov x14, x11
write_loop:
    mov x0, #1
    mov x1, x14
    mov x2, x13
    mov x8, #SYSCALL_WRITE
    svc #0
    cmp x0, xzr
    b.le exit_zero
    add x14, x14, x0
    sub x13, x13, x0
    cbz x13, exit_zero
    b write_loop

exit_zero:
    mov x0, xzr
    mov x8, #SYSCALL_EXIT
    svc #0

.bss
.balign 16
ibuf:
.space BUF_SIZE
.balign 16
obuf:
.space BUF_SIZE
