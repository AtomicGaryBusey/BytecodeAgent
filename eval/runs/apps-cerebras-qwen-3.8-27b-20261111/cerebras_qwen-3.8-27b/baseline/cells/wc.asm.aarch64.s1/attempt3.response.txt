

.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536
.globl _start
_start:
    adrp x9, buf
    add x9, x9, :lo12:buf
    mov x10, xzr
    mov x11, xzr
    mov x12, xzr
    mov x13, xzr
read_loop:
    mov x0, #0
    mov x1, x9
    mov x2, BUF_SIZE
    mov x8, SYS_READ
    svc #0
    mov x3, x0
    cbz x3, read_done
    mov x4, x3
    mov x5, x9
bytes_loop:
    ldrb w7, [x5], #1
    sub x4, x4, #1
    add x12, x12, #1
    cmp w7, #10
    b.eq is_nl
    cmp w7, #9
    b.eq is_ws
    cmp w7, #11
    b.eq is_ws
    cmp w7, #12
    b.eq is_ws
    cmp w7, #13
    b.eq is_ws
    cmp w7, #32
    b.eq is_ws
    cbz x13, start_word
    b next_byte
is_ws:
    mov x13, #0
next_byte:
    cbnz x4, bytes_loop
    b read_loop
is_nl:
    add x10, x10, #1
    mov x13, #0
    b next_byte
start_word:
    add x11, x11, #1
    mov x13, #1
    b next_byte
read_done:
    mov x19, x10
    mov x20, x11
    mov x21, x12
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x22, x24
    mov x0, x19
    mov x1, x22
    bl write_uint
    mov x22, x1
    mov w23, #32
    strb w23, [x22], #1
    mov x0, x20
    mov x1, x22
    bl write_uint
    mov x22, x1
    mov w23, #32
    strb w23, [x22], #1
    mov x0, x21
    mov x1, x22
    bl write_uint
    mov x22, x1
    mov w23, #10
    strb w23, [x22], #1
    mov x0, #1
    mov x1, x24
    sub x2, x22, x24
    mov x8, SYS_WRITE
    svc #0
    mov x8, SYS_EXIT
    mov x0, #0
    svc #0
halt_loop:
    b halt_loop
write_uint:
    cbz x0, wu_zero
    mov x5, xzr
    mov x6, x1
    add x6, x6, #8
wu_loop:
    mov x7, xzr
    mov x8, x0
wu_div10:
    cmp x8, #10
    b.lt wu_div10_done
    sub x8, x8, #10
    add x7, x7, #1
    b wu_div10
wu_div10_done:
    add x8, x8, #48
    add x9, x6, x5
    strb w8, [x9]
    add x5, x5, #1
    mov x0, x7
    cbnz x0, wu_loop
    mov x7, x5
wu_out:
    subs x7, x7, #1
    add x9, x6, x7
    ldrb w8, [x9]
    strb w8, [x1], #1
    cbnz x7, wu_out
    ret
wu_zero:
    mov w8, #48
    strb w8, [x1], #1
    ret
.bss
.balign 16
buf:
.space BUF_SIZE
.balign 16
outbuf:
.space 32
