

.equ BUF_SZ, 0x10000

.bss
.balign 16
input_buf:
.space BUF_SZ

.section .rodata
.balign 1
newline:
.byte 10

.text
.globl _start
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x4, #0
    mov x2, #BUF_SZ
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, process_lines
    add x4, x4, x0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne read_loop
process_lines:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    adrp x11, newline
    add x11, x11, :lo12:newline
    mov x12, x1
    add x3, x1, x4
process_loop:
    cmp x12, x3
    b.ge done
    mov x5, x12
scan_loop:
    ldrb w0, [x12], #1
    cmp w0, #10
    b.eq newline_found
    cmp x12, x3
    b.lt scan_loop
    mov x6, x3
    b reverse_line
newline_found:
    sub x6, x12, #1
reverse_line:
    mov x9, x5
    mov x10, x6
reverse_loop:
    cmp x9, x10
    b.ge write_output
    ldrb w1, [x9]
    ldrb w2, [x10], #-1
    strb w1, [x10]
    strb w2, [x9], #1
    b reverse_loop
write_output:
    sub x2, x6, x5
    cmp x2, #0
    b.eq write_newline
    mov x0, #1
    mov x1, x5
    mov x8, #64
    svc #0
write_newline:
    mov x0, #1
    mov x1, x11
    mov x2, #1
    mov x8, #64
    svc #0
    b process_loop
done:
    mov x0, #0
    mov x8, #94
    svc #0
