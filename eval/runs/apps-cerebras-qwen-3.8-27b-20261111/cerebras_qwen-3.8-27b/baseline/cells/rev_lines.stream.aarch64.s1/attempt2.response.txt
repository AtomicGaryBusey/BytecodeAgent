

.equ BUF_SIZE, 0x10001

.bss
.balign 16
input_buf:
.space BUF_SIZE

.text
.globl _start
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x4, #0
    movz x2, #1, lsl #16
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, process_lines
    add x4, x4, x0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne read_loop
process_lines:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x12, x1
    add x3, x1, x4
process_loop:
    cmp x12, x3
    b.ge done
    mov x5, x12
scan_loop:
    ldrb w0, [x12], #1
    cmp w0, #10
    b.eq newline_found
    cmp x12, x3
    b.lt scan_loop
    mov x6, x3
    mov w0, #10
    strb w0, [x6]
    b reverse_line
newline_found:
    sub x6, x12, #1
reverse_line:
    sub x2, x6, x5
    cmp x2, #2
    b.lt write_output
    mov x9, x5
    sub x10, x6, #1
reverse_loop:
    ldrb w1, [x9]
    ldrb w2, [x10]
    strb w1, [x10]
    strb w2, [x9]
    add x9, x9, #1
    sub x10, x10, #1
    cmp x9, x10
    b.lt reverse_loop
    b write_output
write_output:
    sub x2, x6, x5
    add x2, x2, #1
    mov x1, x5
    mov x0, #1
    mov x8, #64
write_loop:
    svc #0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne write_loop
    b process_loop
done:
    mov x0, #0
    mov x8, #94
    svc #0
