

.section .rodata
.p2align 3
hex_tab:
.ascii "0123456789abcdef"

.section .bss
.p2align 4
in_buf:
.zero 65536
.p2align 3
out_buf:
.zero 80

.section .text
.globl _start
_start:
  mov x19, #0
  mov x20, #0x10
  orr x20, x20, x20, lsl #12

  adrp x21, in_buf
  add x21, x21, :lo12:in_buf
  adrp x22, out_buf
  add x22, x22, :lo12:out_buf
  adrp x27, hex_tab
  add x27, x27, :lo12:hex_tab

  mov w28, #32
  mov w29, #46

read_loop:
  mov x0, #0
  add x1, x21, x19
  mov x2, x20
  mov x8, #63
  svc #0
  cbz x0, read_done
  cmp x0, #1
  b.lt read_done
  add x19, x19, x0
  sub x20, x20, x0
  cbz x20, read_done
  b read_loop

read_done:
  mov x20, x19
  mov x19, #0

row_loop:
  cmp x19, x20
  b.ge final_offset

  mov x23, x22
  add x24, x21, x19

  mov w25, w20
  sub w25, w25, w19
  cmp w25, #16
  b.lo row_ready
  mov w25, #16
row_ready:

  lsr w3, w19, #28
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #24
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #20
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #16
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #12
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #8
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w19, #4
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  and w3, w19, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1

  strb w28, [x23], #1
  strb w28, [x23], #1

  mov w26, #0
present_loop:
  cmp w26, w25
  b.ge present_done

  ldrb w7, [x24], #1
  lsr w3, w7, #4
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  and w3, w7, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  strb w28, [x23], #1

  cmp w26, #7
  b.ne no_extra
  strb w28, [x23], #1
no_extra:
  add w26, w26, #1
  b present_loop

present_done:
  mov w26, w25
missing_loop:
  cmp w26, #16
  b.ge missing_done

  strb w28, [x23], #1
  strb w28, [x23], #1
  strb w28, [x23], #1

  cmp w26, #7
  b.ne no_missing_extra
  strb w28, [x23], #1
no_missing_extra:
  add w26, w26, #1
  b missing_loop

missing_done:
  strb w28, [x23], #1
  mov w3, #124
  strb w3, [x23], #1

  add x24, x21, x19
  mov w26, #0
ascii_loop:
  cmp w26, w25
  b.ge ascii_done

  ldrb w7, [x24], #1
  cmp w7, #0x20
  csel w4, w29, w7, hs
  cmp w7, #0x7f
  csel w5, w4, w29, lo
  strb w5, [x23], #1

  add w26, w26, #1
  b ascii_loop

ascii_done:
  mov w3, #124
  strb w3, [x23], #1
  mov w3, #10
  strb w3, [x23], #1

  sub x26, x23, x22
  mov x0, #1
  mov x1, x22
  mov x2, x26

write_row_loop:
  mov x8, #64
  svc #0
  cmp x0, #1
  b.lt write_row_done
  add x1, x1, x0
  sub x2, x2, x0
  cbz x2, write_row_done
  b write_row_loop

write_row_done:
  add x19, x19, x25
  b row_loop

final_offset:
  mov x23, x22

  lsr w3, w20, #28
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #24
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #20
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #16
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #12
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #8
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  lsr w3, w20, #4
  and w3, w3, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1
  and w3, w20, #15
  ldrb w4, [x27, w3, uxtb]
  strb w4, [x23], #1

  mov w3, #10
  strb w3, [x23], #1

  sub x26, x23, x22
  mov x0, #1
  mov x1, x22
  mov x2, x26

write_final_loop:
  mov x8, #64
  svc #0
  cmp x0, #1
  b.lt write_final_done
  add x1, x1, x0
  sub x2, x2, x0
  cbz x2, write_final_done
  b write_final_loop

write_final_done:
  mov x0, #0
  mov x8, #94
  svc #0
