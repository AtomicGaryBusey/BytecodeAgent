

.text
.globl _start

INBUFMAX  = 49152
CHUNK     = 8192

FLUSHLIM  = 83520

.equ SYSCALL_READ, 0
.equ SYSCALL_WRITE, 1
.equ SYSCALL_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ EXIT_OK, 0

_start:
    cld

read_loop:
    cmp r12d, INBUFMAX
    jae read_done
    mov edi, STDIN
    lea rsi, [rip + inbuf + r12]
    mov edx, CHUNK
