

.equ input_max, 49152
.equ out_max, 66560
.equ line_max, 76

.section .bss
.align 16
input:
.zero input_max
output:
.zero out_max

.section .rodata
.align 16
table:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.byte 61

.section .text
emit_char:
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [r14 + rdx], al
    inc rdx
    inc r8
    cmp r8d, line_max
    jne emit_char_done
    mov byte ptr [r14 + rdx], 10
    inc rdx
    xor r8d, r8d
emit_char_done:
    ret

.globl _start
_start:
    lea r13, [input]
    lea r14, [output]
    lea r15, [table]
    xor rbx, rbx
read_next:
    cmp rbx, input_max
    jge input_done
    mov rsi, r13
    add rsi, rbx
    mov rdx, input_max
    sub rdx, rbx
    mov eax, 0
    xor edi, edi
    syscall
    test rax, rax
    jz input_done
    js exit0
    add rbx, rax
    jmp read_next
input_done:
    test rbx, rbx
    jz exit0
    xor rcx, rcx
    xor rdx, rdx
    xor r8d, r8d
group_loop:
    cmp rcx, rbx
    jae remainder
    mov r10, rbx
    sub r10, rcx
    cmp r10d, 3
    jb remainder
    movzx r9d, byte ptr [r13 + rcx]
    movzx r11d, byte ptr [r13 + rcx + 1]
    movzx r10d, byte ptr [r13 + rcx + 2]
    mov eax, r9d
    shr eax, 2
    call emit_char
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov r12d, r11d
    shr r12d, 4
    or eax, r12d
    call emit_char
    mov eax, r11d
    and eax, 15
    shl eax, 2
    mov r12d, r10d
    shr r12d, 6
    or eax, r12d
    call emit_char
    mov eax, r10d
    and eax, 63
    call emit_char
    add rcx, 3
    jmp group_loop
remainder:
    cmp rcx, rbx
    je final
    mov r10, rbx
    sub r10, rcx
    cmp r10d, 2
    jb one_byte
    movzx r9d, byte ptr [r13 + rcx]
    movzx r10d, byte ptr [r13 + rcx + 1]
    mov eax, r9d
    shr eax, 2
    call emit_char
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov r12d, r10d
    shr r12d, 4
    or eax, r12d
    call emit_char
    mov eax, r10d
    and eax, 15
    shl eax, 2
    call emit_char
    mov eax, 64
    call emit_char
    jmp final
one_byte:
    movzx r9d, byte ptr [r13 + rcx]
    mov eax, r9d
    shr eax, 2
    call emit_char
    mov eax, r9d
    and eax, 3
    shl eax, 4
    call emit_char
    mov eax, 64
    call emit_char
    mov eax, 64
    call emit_char
final:
    test r8d, r8d
    jz write_output
    mov byte ptr [r14 + rdx], 10
    inc rdx
    xor r8d, r8d
write_output:
    test rdx, rdx
    jz exit0
    mov rsi, r14
write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jz write_done
    js write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
exit0:
    mov rax, 60
    xor edi, edi
    syscall
