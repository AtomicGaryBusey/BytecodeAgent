

.equ IN_BUF_SIZE, 49152

.section .rodata
.align 16
base64_table:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
in_buf:
.space IN_BUF_SIZE
out_buf:
.space 77

.section .text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, IN_BUF_SIZE
    syscall
    test rax, rax
    jz finish
    js finish
    mov r8, rax
    xor r9, r9
    jmp process_loop

process_loop:
    movzx eax, byte ptr [rip + in_buf + r9]
    inc r9
    dec r8
    cmp r12d, 1
    je state_one
    cmp r12d, 2
    je state_two
    mov r13d, eax
    mov r12d, 1
    jmp process_check

state_one:
    mov r14d, eax
    mov r12d, 2
    jmp process_check

state_two:
    mov r10d, eax
    xor r12d, r12d

    mov eax, r13d
    shr al, 2
    call emit_b64

    mov eax, r13d
    and al, 3
    shl al, 4
    mov edx, r14d
    shr dl, 4
    or al, dl
    call emit_b64

    mov eax, r14d
    and al, 15
    shl al, 2
    mov edx, r10d
    shr dl, 6
    or al, dl
    call emit_b64

    mov eax, r10d
    and al, 63
    call emit_b64

    jmp process_check

process_check:
    test r8, r8
    jnz process_loop
    jmp read_loop

finish:
    test r12d, r12d
    jz flush_final
    cmp r12d, 1
    je final_one

    mov eax, r13d
    shr al, 2
    call emit_b64

    mov eax, r13d
    and al, 3
    shl al, 4
    mov edx, r14d
    shr dl, 4
    or al, dl
    call emit_b64

    mov eax, r14d
    and al, 15
    shl al, 2
    call emit_b64

    call emit_equal
    jmp flush_final

final_one:
    mov eax, r13d
    shr al, 2
    call emit_b64

    mov eax, r13d
    and al, 3
    shl al, 4
    call emit_b64

    call emit_equal
    call emit_equal
    jmp flush_final

flush_final:
    test r15d, r15d
    jz do_exit
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + r15], 10
    inc r15d
    mov edi, 1
    mov edx, r15d
    mov eax, 1
    syscall

do_exit:
    mov edi, 0
    mov eax, 60
    syscall

emit_b64:
    movzx eax, byte ptr [rip + base64_table + rax]
    jmp emit_out

emit_equal:
    mov al, 61
    jmp emit_out

emit_out:
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + r15], al
    inc r15d
    cmp r15d, 76
    jne emit_out_done
    mov byte ptr [rsi + 76], 10
    mov edi, 1
    mov edx, 77
    mov eax, 1
    syscall
    xor r15d, r15d

emit_out_done:
    ret
