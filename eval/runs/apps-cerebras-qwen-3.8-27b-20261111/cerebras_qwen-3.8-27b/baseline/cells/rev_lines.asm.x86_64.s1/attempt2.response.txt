

.equ MAX_IN, 65536
.equ MAX_OUT, 65537
.equ NL, 10

.section .bss
.align 16
inbuf:
.space MAX_IN
outbuf:
.space MAX_OUT

.section .text
.globl _start
_start:
    xor r14d, r14d
read_loop:
    cmp r14, MAX_IN
    jae read_done
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r14
    mov rdx, MAX_IN
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r14, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [rip + outbuf]
    xor rbx, rbx
    xor r15d, r15d
proc_loop:
    cmp r15, r14
    jge proc_done
    mov r8, r15
find_loop:
    cmp r8, r14
    jge no_newline
    cmp byte ptr [r12 + r8], NL
    jne find_next
    jmp emit_line
find_next:
    inc r8
    jmp find_loop
no_newline:
emit_line:
    mov r9, r15
    mov r10, r8
    cmp r10, r9
    je line_no_bytes
    dec r10
rev_loop:
    movzx eax, byte ptr [r12 + r10]
    mov byte ptr [r13 + rbx], al
    inc rbx
    dec r10
    cmp r10, r9
    jge rev_loop
line_no_bytes:
    mov byte ptr [r13 + rbx], NL
    inc rbx
    cmp r8, r14
    je proc_done
    lea r15, [r8 + 1]
    jmp proc_loop
proc_done:
    test rbx, rbx
    jz exit_zero
    mov rax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, rbx
    syscall
exit_zero:
    mov rax, 60
    xor edi, edi
    syscall
