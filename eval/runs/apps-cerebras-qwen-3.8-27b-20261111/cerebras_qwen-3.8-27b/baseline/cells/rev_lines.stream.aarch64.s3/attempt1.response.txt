

.equ IN_SZ, 0x10000
.equ OUT_SZ, 0x10001
.text
.globl _start
_start:
 adrp x4, input
 add x4, x4, :lo12:input
 adrp x5, output
 add x5, x5, :lo12:output
 movz x9, #0
 movz x10, #0x1, lsl #16
read_loop:
 cmp x9, x10
 b.eq process
 sub x2, x10, x9
 add x1, x4, x9
 movz x0, #0
 movz x8, #63
 svc #0
 cmp x0, #0
 b.eq process
 b.lt error_exit
 add x9, x9, x0
 b read_loop
process:
 movz x10, #0
 movz x14, #0
outer:
 cmp x10, x9
 b.eq finish
 mov x11, x10
scan:
 cmp x11, x9
 b.eq no_newline
 ldrb w12, [x4, x11]
 cmp w12, #10
 b.eq found
 add x11, x11, #1
 b scan
no_newline:
 mov x12, x9
 b reverse
found:
 mov x12, x11
reverse:
 sub x13, x12, x10
 cbz x13, no_reverse
 sub x16, x12, #1
rev:
 ldrb w17, [x4, x16]
 strb w17, [x5, x14]
 add x14, x14, #1
 subs x13, x13, #1
 b.ne rev
no_reverse:
 movz w18, #10
 strb w18, [x5, x14]
 add x14, x14, #1
 cmp x12, x9
 b.eq finish
 add x10, x12, #1
 b outer
finish:
 cbz x14, exit0
 movz x0, #1
 mov x1, x5
 mov x2, x14
 movz x8, #64
 svc #0
exit0:
 movz x0, #0
 movz x8, #93
 svc #0
error_exit:
 movz x0, #1
 movz x8, #93
 svc #0
.bss
.balign 16
input:
.zero IN_SZ
.balign 16
output:
.zero OUT_SZ
