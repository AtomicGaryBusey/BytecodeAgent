

.globl _start
.equ INPUT_SIZE, 65536

.text
_start:
    lea rdi, [rip + input]
    xor ebx, ebx
read_loop:
    mov eax, 0
    mov rsi, rdi
    mov edx, INPUT_SIZE
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    add rdi, rax
    cmp ebx, INPUT_SIZE
    jge read_done
    jmp read_loop
read_done:
    xor r15d, r15d
row_loop:
    cmp r15, rbx
    jae final
    mov rax, rbx
    sub rax, r15
    cmp rax, 16
    jae row_full
    mov r11d, eax
    jmp init_line
row_full:
    mov r11d, 16
init_line:
    lea r14, [rip + line_buffer]
    mov esi, r15d
    mov r12d, 7
offset_loop:
    mov eax, esi
    and eax, 15
    mov al, byte ptr [rip + hex_digits + rax]
