

.equ IN_SIZE, 65536
.equ OUT_SIZE, 65537

.text
.globl _start
_start:
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    adrp x4, output_buf
    add x4, x4, :lo12:output_buf
    mov x9, #0

read_loop:
    mov x0, #0
    add x1, x3, x9
    mov x2, #65536
    sub x2, x2, x9
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x9, x9, x0
    cmp x9, #65536
    b.ne read_loop

read_done:
    mov x5, x4
    mov x10, #0

process:
    cmp x10, x9
    b.ge done
    mov x7, x10

find_loop:
    cmp x7, x9
    b.eq final_line
    ldrb w13, [x3, x7]
    cmp w13, #10
    b.eq line
    add x7, x7, #1
    b find_loop

line:
    mov x6, x10
    sub x8, x7, x10
    add x10, x7, #1
    b reverse

final_line:
    mov x6, x10
    sub x8, x9, x10
    cmp x8, #0
    b.eq done
    mov x10, x9
    b reverse

reverse:
    cmp x8, #0
    b.eq newline
    add x12, x6, x8

reverse_loop:
    sub x12, x12, #1
    ldrb w13, [x3, x12]
    strb w13, [x5], #1
    cmp x12, x6
    b.gt reverse_loop

newline:
    mov w13, #10
    strb w13, [x5], #1
    b process

done:
    sub x9, x5, x4
    cbz x9, do_exit
    mov x0, #1
    mov x1, x4
    mov x2, x9

write_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 4096
input_buf:
    .zero IN_SIZE
output_buf:
    .zero OUT_SIZE
