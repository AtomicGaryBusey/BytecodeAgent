

.globl _start
.equ in_size, 49152
.equ out_size, 66560

.text
_start:
    lea r10, [rip + inbuf]
    lea r15, [rip + outbuf]
    xor r14d, r14d

read_loop:
    cmp r14d, in_size
    jge encode_start
    xor eax, eax
    xor edi, edi
    mov rsi, r10
    add rsi, r14
    mov edx, in_size
    sub edx, r14d
    syscall
    cmp rax, 0
    jle encode_start
    add r14d, eax
    jmp read_loop

encode_start:
    xor r13d, r13d
    xor r11d, r11d
    xor r12d, r12d

enc_loop:
    cmp r12d, r14d
    jge enc_end
    mov rsi, r10
    add rsi, r12
    movzx eax, byte ptr [rsi]
    mov r8d, eax
    shl r8d, 16
    mov r9d, r14d
    sub r9d, r12d
    dec r9d
    test r9d, r9d
    jz emit_group
    movzx eax, byte ptr [rsi + 1]
    shl eax, 8
    or r8d, eax
    cmp r9d, 1
    jbe emit_group
    movzx eax, byte ptr [rsi + 2]
    or r8d, eax

emit_group:
    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    call _emit_b64
    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    call _emit_b64
    test r9d, r9d
    jz emit_pad2
    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    call _emit_b64
    cmp r9d, 1
    je emit_pad1
    mov ecx, r8d
    and ecx, 63
    call _emit_b64
    jmp next_group

emit_pad1:
    mov al, 61
    call _emit_ascii
    jmp next_group

emit_pad2:
    mov al, 61
    call _emit_ascii
    mov al, 61
    call _emit_ascii

next_group:
    add r12d, 3
    cmp r12d, r14d
    jb enc_loop
    jmp enc_end

enc_end:
    test r11d, r11d
    jz write_output
    mov byte ptr [r15 + r13], 10
    inc r13d

write_output:
    test r13d, r13d
    jz exit0
    xor r12d, r12d

write_loop:
    mov rax, 1
    mov edi, 1
    mov rsi, r15
    add rsi, r12
    mov edx, r13d
    sub edx, r12d
    syscall
    cmp rax, 0
    jle exit0
    add r12d, eax
    cmp r12d, r13d
    jb write_loop

exit0:
    mov rax, 60
    xor edi, edi
    syscall

_emit_b64:
    lea rax, [rip + b64_table]
    mov al, byte ptr [rax + rcx]
    jmp _emit_ascii

_emit_ascii:
    mov byte ptr [r15 + r13], al
    inc r13d
    inc r11d
    cmp r11d, 76
    jne ret_ascii
    mov byte ptr [r15 + r13], 10
    inc r13d
    xor r11d, r11d

ret_ascii:
    ret

.section .rodata
b64_table:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
.zero in_size
outbuf:
.zero out_size
