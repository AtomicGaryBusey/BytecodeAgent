

.equ IN_BYTES, 65536
.equ OUT_BYTES, 65536
.equ MATCH_BYTES, 262144
.equ OPEN_BYTES, 262144
.equ TAPE_BYTES, 30000

.globl _start

.text
.balign 16
_start:
    mov x26, #0x10000
    adrp x3, in_buf
    add x3, x3, :lo12:in_buf
    mov x1, x3
    mov x4, xzr

read_loop:
    cmp x4, x26
    b.hs read_done
    sub x2, x26, x4
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x4, x4, x0
    add x1, x1, x0
    b read_loop

read_done:
    mov x5, xzr

find_sep:
    cmp x5, x4
    b.ge no_sep
    ldrb w14, [x3, x5]
    cmp w14, #0x21
    beq sep_found
    add x5, x5, #1
    b find_sep

sep_found:
    mov x7, x5
    add x20, x5, #1
    b after_sep

no_sep:
    mov x7, x4
    mov x20, x4

after_sep:
    mov x21, x4

    adrp x9, open_stack
    add x9, x9, :lo12:open_stack
    adrp x10, match
    add x10, x10, :lo12:match
    mov x5, xzr
    mov x6, xzr

check_b:
    cmp x5, x7
    b.ge check_end
    ldrb w14, [x3, x5]
    cmp w14, #0x5b
    beq push_b
    cmp w14, #0x5d
    beq pop_b
    b next_b

push_b:
    str w5, [x9, x6, lsl #2]
    add x6, x6, #1

next_b:
    add x5, x5, #1
    b check_b

pop_b:
    cbz x6, exit1
    sub x6, x6, #1
    ldr w14, [x9, x6, lsl #2]
    add w15, w5, #1
    str w15, [x10, x14, lsl #2]
    str w14, [x10, x5, lsl #2]
    b next_b

check_end:
    cbnz x6, exit1

    adrp x11, tape
    add x11, x11, :lo12:tape
    adrp x12, out_buf
    add x12, x12, :lo12:out_buf
    mov x5, xzr
    mov x6, xzr
    mov x13, xzr

exec_loop:
    cmp x5, x7
    b.ge exec_done
    ldrb w14, [x3, x5]
    add x5, x5, #1
    cmp w14, #0x3e
    beq cmd_gt
    cmp w14, #0x3c
    beq cmd_lt
    cmp w14, #0x2b
    beq cmd_plus
    cmp w14, #0x2d
    beq cmd_minus
    cmp w14, #0x2e
    beq cmd_dot
    cmp w14, #0x2c
    beq cmd_comma
    cmp w14, #0x5b
    beq cmd_open
    cmp w14, #0x5d
    beq cmd_close
    b exec_loop

cmd_gt:
    add x6, x6, #1
    b exec_loop

cmd_lt:
    sub x6, x6, #1
    b exec_loop

cmd_plus:
    ldrb w15, [x11, x6]
    add w15, w15, #1
    and w15, w15, #0xff
    strb w15, [x11, x6]
    b exec_loop

cmd_minus:
    ldrb w15, [x11, x6]
    sub w15, w15, #1
    and w15, w15, #0xff
    strb w15, [x11, x6]
    b exec_loop

cmd_dot:
    ldrb w14, [x11, x6]
    strb w14, [x12, x13]
    add x13, x13, #1
    cmp x13, x26
    b.eq dot_flush
    b exec_loop

dot_flush:
    mov x0, #1
    mov x1, x12
    mov x2, x13
    mov x8, #64
    svc #0
    mov x13, xzr
    b exec_loop

cmd_comma:
    cmp x20, x21
    b.ge comma_zero
    ldrb w15, [x3, x20]
    add x20, x20, #1
    strb w15, [x11, x6]
    b exec_loop

comma_zero:
    mov w15, #0
    strb w15, [x11, x6]
    b exec_loop

cmd_open:
    ldrb w15, [x11, x6]
    cbnz w15, exec_loop
    sub x16, x5, #1
    ldr w15, [x10, x16, lsl #2]
    mov x5, x15
    b exec_loop

cmd_close:
    ldrb w15, [x11, x6]
    cbz w15, exec_loop
    sub x16, x5, #1
    ldr w15, [x10, x16, lsl #2]
    mov x5, x15
    b exec_loop

exec_done:
    cmp x13, #0
    beq exit0
    mov x0, #1
    mov x1, x12
    mov x2, x13
    mov x8, #64
    svc #0

exit0:
    mov x0, #0
    mov x8, #94
    svc #0

exit1:
    mov x0, #1
    mov x8, #94
    svc #0

.bss
.balign 4
in_buf:
.zero IN_BYTES
out_buf:
.zero OUT_BYTES
match:
.zero MATCH_BYTES
open_stack:
.zero OPEN_BYTES
tape:
.zero TAPE_BYTES
