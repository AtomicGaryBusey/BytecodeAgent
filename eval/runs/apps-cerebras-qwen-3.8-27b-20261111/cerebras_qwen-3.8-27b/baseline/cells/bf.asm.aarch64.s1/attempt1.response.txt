.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 262144
.equ STACK_SIZE, 262144

.equ C_EXCLAIM, 33
.equ C_PLUS, 43
.equ C_MINUS, 45
.equ C_DOT, 46
.equ C_COMMA, 44
.equ C_LBRACKET, 91
.equ C_RBRACKET, 93
.equ C_GT, 62
.equ C_LT, 60

.text
.globl _start

 adrp x1, stdin_buf
 add x1, x1, :lo12:stdin_buf

 mov x8, #SYS_READ

 mov x24, #0

 b.ge no_sep
 ldrb w25, [x19, x24]
 cmp w25, #C_EXCLAIM
 beq sep_found
 add x24, x24, #1
 b find_loop

 add x21, x19, x24
 add x21, x21, #1

 add x22, x19, x23

 add x3, x3, :lo12:jump_table
 adrp x4, open_stack
 add x4, x4, :lo12:open_stack

 mov x6, #0
val_loop:
 cmp x6, x20
 b.ge val_end
 ldrb w7, [x19, x6]
 cmp w7, #C_LBRACKET
 beq val_open
 cmp w7, #C_RBRACKET

 b val_next

 add x5, x5, #1
 b val_next
val_close:

 sub x5, x5, #1
 ldr w7, [x4, x5, lsl #2]
 str w7, [x3, x6, lsl #2]
 add w8, w6, #1
 str w8, [x3, x7, lsl #2]

val_next:
 add x6, x6, #1
 b val_loop
val_end:
 cbnz x5, invalid
 adrp x26, tape
 add x26, x26, :lo12:tape
 adrp x29, outbuf
 add x29, x29, :lo12:outbuf
 mov x4, x29
 add x5, x29, #OUT_SIZE
 mov x25, #0
 mov x24, #0
bf_loop:
 cmp x24, x20
 b.ge bf_end
 ldrb w28, [x19, x24]
 cmp w28, #C_LBRACKET
 beq do_open
 cmp w28, #C_RBRACKET
 beq do_close
 cmp w28, #C_GT
 beq do_gt
 cmp w28, #C_LT
 beq do_lt
 cmp w28, #C_PLUS
 beq do_plus
 cmp w28, #C_MINUS
 beq do_minus
 cmp w28, #C_DOT
 beq do_dot
 cmp w28, #C_COMMA

 cbz w7, open_jump

 ldr w7, [x3, x24, lsl #2]
 mov x24, w7

 cbz w7, adv
 ldr w7, [x3, x24, lsl #2]
 mov x24, w7

do_gt:

 b adv
do_lt:

 b adv
do_plus:

do_minus:
 ldrb w7, [x26]
 sub w7, w7, #1

 strb w7, [x4]
 add x4, x4, #1

 mov x2, x25

bf_end:
