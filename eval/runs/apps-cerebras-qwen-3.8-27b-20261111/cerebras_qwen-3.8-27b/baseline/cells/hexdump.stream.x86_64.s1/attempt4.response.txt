

.text
.globl _start
_start:
    xor ebx, ebx
    lea rsi, [rip + input_buf]
read_loop:
    mov r9d, 65536
    sub r9d, ebx
    cmp r9d, 0
    jz read_done
    mov eax, 0
    xor edi, edi
    mov edx, r9d
    syscall
    cmp rax, 0
    jle read_done
    add rsi, rax
    add ebx, eax
    jmp read_loop
read_done:
    lea r12, [rip + input_buf]
    lea r13, [rip + hex_table]
    xor r8d, r8d
lines_loop:
    cmp r8d, ebx
    jge dump_done
    mov r9d, ebx
    sub r9d, r8d
    cmp r9d, 16
    jbe line_len_set
    mov r9d, 16
line_len_set:
    lea r14, [r12 + r8]
    lea r15, [rip + line_buf]
    lea r10, [r15 + 7]
    mov eax, r8d
    mov ecx, 8
offset_hex:
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10], dl
    dec r10
    shr eax, 4
    dec ecx
    jnz offset_hex
    mov byte ptr [r15 + 8], 32
    mov byte ptr [r15 + 9], 32
    lea r10, [r15 + 10]
    xor r11d, r11d
slot_loop:
    cmp r11d, r9d
    jb slot_have
    mov byte ptr [r10], 32
    mov byte ptr [r10 + 1], 32
    mov byte ptr [r10 + 2], 32
    add r10, 3
    jmp slot_continue
slot_have:
    movzx eax, byte ptr [r14 + r11]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10], dl
    and al, 15
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], al
    mov byte ptr [r10 + 2], 32
    add r10, 3
slot_continue:
    cmp r11d, 7
    je slot_extra
    inc r11d
    cmp r11d, 16
    jb slot_loop
    jmp slots_done
slot_extra:
    mov byte ptr [r10], 32
    inc r10
    inc r11d
    cmp r11d, 16
    jb slot_loop
slots_done:
    mov byte ptr [r10], 32
    inc r10
    mov byte ptr [r10], 124
    inc r10
    xor r11d, r11d
char_loop:
    cmp r11d, r9d
    jae chars_done
    movzx eax, byte ptr [r14 + r11]
    cmp al, 32
    jb char_dot
    cmp al, 126
    jbe char_put
char_dot:
    mov al, 46
char_put:
    mov byte ptr [r10], al
    inc r10
    inc r11d
    jmp char_loop
chars_done:
    mov byte ptr [r10], 124
    inc r10
    mov byte ptr [r10], 10
    inc r10
    mov rdi, 1
    mov rsi, r15
    mov rdx, r10
    sub rdx, r15
write_loop:
    mov eax, 1
    syscall
    cmp rax, rdx
    je write_done
    cmp rax, 0
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    add r8d, 16
    jmp lines_loop
dump_done:
    lea r15, [rip + line_buf]
    lea r10, [r15 + 7]
    mov eax, ebx
    mov ecx, 8
final_hex:
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10], dl
    dec r10
    shr eax, 4
    dec ecx
    jnz final_hex
    mov byte ptr [r15 + 8], 10
    lea r10, [r15 + 9]
    mov rdi, 1
    mov rsi, r15
    mov rdx, r10
    sub rdx, r15
final_write_loop:
    mov eax, 1
    syscall
    cmp rax, rdx
    je final_done
    cmp rax, 0
    jle final_done
    add rsi, rax
    sub rdx, rax
    jmp final_write_loop
final_done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
.align 16
hex_table:
.ascii "0123456789abcdef"

.bss
.align 16
input_buf:
.space 65536
.align 16
line_buf:
.space 128
