

.text
.globl _start
_start:
    xor ebx, ebx
    lea rsi, [rip + input_buf]
read_loop:
    mov r9d, 65536
    sub r9d, ebx
    cmp r9d, 0
    jz read_done
    mov eax, 0
    mov rdi, 0
    mov rdx, r9
    syscall
    cmp rax, 0
    jle read_done
    add rsi, rax
    add ebx, eax
    jmp read_loop
read_done:
    lea r12, [rip + input_buf]
    lea r13, [rip + hex_table]
    xor r8d, r8d
lines_loop:
    cmp r8d, ebx
    jae dump_done
    mov r11d, ebx
    sub r11d, r8d
    cmp r11d, 16
    jbe set_line_len
    mov r11d, 16
set_line_len:
    lea r14, [r12 + r8]
    lea r15, [rip + line_buf]
    lea r10, [r15 + 7]
    mov ecx, 8
    mov eax, r8d
Loff_hex:
    and eax, 15
    mov dl, byte ptr [r13 + rax]
    mov byte ptr [r10], dl
    dec r10
    shr eax, 4
    dec ecx
    jnz Loff_hex
    lea r10, [r15 + 8]
    mov byte ptr [r10], 32
    mov byte ptr [r10 + 1], 32
    add r10, 2
    xor r9d, r9d
Lslot_loop:
    cmp r9d, r11d
    jb Lhave_byte
    mov byte ptr [r10], 32
    mov byte ptr [r10 + 1], 32
    mov byte ptr [r10 + 2], 32
    add r10, 3
    jmp Lslot_continue
Lhave_byte:
    movzx eax, byte ptr [r14 + r9]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r10], dl
    and al, 15
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10 + 1], al
    mov byte ptr [r10 + 2], 32
    add r10, 3
Lslot_continue:
    cmp r9d, 7
    je Lextra_space
    add r9d, 1
    cmp r9d, 16
    jb Lslot_loop
    jmp Lafter_slots
Lextra_space:
    mov byte ptr [r10], 32
    add r10, 1
    add r9d, 1
    cmp r9d, 16
    jb Lslot_loop
Lafter_slots:
    mov byte ptr [r10], 32
    add r10, 1
    mov byte ptr [r10], 124
    add r10, 1
    xor r9d, r9d
Lchar_loop:
    cmp r9d, r11d
    jae Lchars_done
    movzx eax, byte ptr [r14 + r9]
    cmp al, 32
    jb Ldot
    cmp al, 126
    jbe Lprint_char
Ldot:
    mov al, 46
Lprint_char:
    mov byte ptr [r10], al
    add r10, 1
    add r9d, 1
    jmp Lchar_loop
Lchars_done:
    mov byte ptr [r10], 124
    add r10, 1
    mov byte ptr [r10], 10
    add r10, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, r10
    sub rdx, r15
Lwrite_line:
    mov eax, 1
    syscall
    cmp rax, rdx
    je Lwrite_line_done
    cmp rax, 0
    jle Lwrite_line_done
    add rsi, rax
    sub rdx, rax
    jmp Lwrite_line
Lwrite_line_done:
    add r8d, 16
    jmp lines_loop
dump_done:
    lea r15, [rip + line_buf]
    lea r10, [r15 + 7]
    mov ecx, 8
    mov eax, ebx
Lfinal_hex:
    and eax, 15
    mov dl, byte ptr [r13 + rax]
    mov byte ptr [r10], dl
    dec r10
    shr eax, 4
    dec ecx
    jnz Lfinal_hex
    mov byte ptr [r15 + 8], 10
    mov rdi, 1
    mov rsi, r15
    mov edx, 9
Lwrite_final:
    mov eax, 1
    syscall
    cmp rax, rdx
    je Lwrite_final_done
    cmp rax, 0
    jle Lwrite_final_done
    add rsi, rax
    sub rdx, rax
    jmp Lwrite_final
Lwrite_final_done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
.align 16
hex_table:
.ascii "0123456789abcdef"

.bss
.align 16
input_buf:
.space 65536
.align 16
line_buf:
.space 128
