

.globl _start
.equ MAXLEN, 65536

.text
_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, MAXLEN
    sub rdx, r12
    syscall
    cmp rax, 1
    jl dump
    add r12, rax
    cmp r12, MAXLEN
    jne read_loop
    jmp dump
dump:
    xor r13, r13
rows:
    cmp r13, r12
    jge final
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe set_n
    mov rax, 16
set_n:
    mov r14, rax
    lea r10, [rip + linebuf]
    lea r11, [rip + hexdigit]
    mov rax, r13
    mov cl, 28
    mov ebx, 8
row_conv:
    mov rdx, rax
    shr rdx, cl
    and rdx, 15
    movzx eax, byte ptr [r11 + rdx]
    mov [r10], al
    inc r10
    sub cl, 4
    dec ebx
    jnz row_conv
    mov byte ptr [r10], 32
    mov byte ptr [r10+1], 32
    add r10, 2
    lea r9, [rip + buffer]
    add r9, r13
    xor rcx, rcx
hex_loop:
    cmp rcx, r14
    jge no_byte
    movzx eax, byte ptr [r9 + rcx]
    movzx edx, al
    shr eax, 4
    movzx eax, byte ptr [r11 + rax]
    mov [r10], al
    and edx, 15
    movzx eax, byte ptr [r11 + edx]
    mov [r10+1],
