

.globl _start

.text

_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r14d, r14d
    lea rsi, [rip + inbuf]
    mov r13d, 65536

read_loop:
    xor eax, eax
    xor edi, edi
    mov edx, r13d
    syscall
    cmp rax, 0
    jle done_read
    mov r12d, eax
    add r10d, r12d
    mov rcx, r12
byte_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je ws
    cmp al, 9
    je ws
    cmp al, 11
    je ws
    cmp al, 12
    je ws
    cmp al, 13
    je ws
    cmp al, 32
    je ws
    jmp nonws
ws:
    cmp al, 10
    jne no_newline
    inc r8d
no_newline:
    xor r14d, r14d
    jmp next
nonws:
    test r14d, r14d
    jnz next
    inc r9d
    mov r14d, 1
next:
    inc rsi
    dec rcx
    jnz byte_loop
    sub r13d, r12d
    test r13d, r13d
    jz done_read
    jmp read_loop

done_read:
    mov qword ptr [rip + cnt_lines], r8
    mov qword ptr [rip + cnt_words], r9
    mov qword ptr [rip + cnt_bytes], r10
    mov byte ptr [rip + outpos], 0
    mov rdi, qword ptr [rip + cnt_lines]
    call print_uint
    lea r11, [rip + outbuf]
    movzx ecx, byte ptr [rip + outpos]
    mov byte ptr [r11 + rcx], 32
    inc byte ptr [rip + outpos]
    mov rdi, qword ptr [rip + cnt_words]
    call print_uint
    lea r11, [rip + outbuf]
    movzx ecx, byte ptr [rip + outpos]
    mov byte ptr [r11 + rcx], 32
    inc byte ptr [rip + outpos]
    mov rdi, qword ptr [rip + cnt_bytes]
    call print_uint
    lea r11, [rip + outbuf]
    movzx ecx, byte ptr [rip + outpos]
    mov byte ptr [r11 + rcx], 10
    inc byte ptr [rip + outpos]
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    movzx edx, byte ptr [rip + outpos]
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_uint:
    mov rax, rdi
    xor ebx, ebx
    lea r12, [rip + digbuf]
div_loop:
    xor edx, edx
    mov rcx, 10
    div rcx
    mov r8, rax
    mov al, dl
    add al, 48
    mov byte ptr [r12 + rbx], al
    inc rbx
    test r8, r8
    jnz div_loop
    mov r10d, ebx
    lea r11, [rip + outbuf]
out_loop:
    dec r10d
    movzx eax, byte ptr [r12 + r10]
    movzx edx, byte ptr [rip + outpos]
    mov byte ptr [r11 + rdx], al
    inc byte ptr [rip + outpos]
    test r10d, r10d
    jnz out_loop
    ret

.bss
.balign 8
cnt_lines: .zero 8
cnt_words: .zero 8
cnt_bytes: .zero 8
inbuf: .zero 65536
outbuf: .zero 64
digbuf: .zero 32
outpos: .zero 1
