

.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    syscall

    test rax, rax
    jle print_numbers

    add r14d, eax
    mov rcx, rax
    lea rdi, [rip + buf]

count_loop:
    movzx eax, byte ptr [rdi]
    inc rdi
    dec rcx

    cmp al, 10
    je newline
    cmp al, 9
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    cmp al, 32
    je whitespace

    test r15, r15
    jz enter_word
    jmp next_byte

enter_word:
    inc r13d
    mov r15, 1
    jmp next_byte

whitespace:
    xor r15, r15
    jmp next_byte

newline:
    inc r12d
    xor r15, r15
    jmp next_byte

next_byte:
    test rcx, rcx
    jnz count_loop
    jmp read_loop

print_numbers:
    lea rsi, [rip + outbuf]

    mov rdi, r12
    call append_dec
    mov byte ptr [rax], 32
    inc rax

    mov rdi, r13
    mov rsi, rax
    call append_dec
    mov byte ptr [rax], 32
    inc rax

    mov rdi, r14
    mov rsi, rax
    call append_dec
    mov byte ptr [rax], 10
    inc rax

    lea rsi, [rip + outbuf]
    mov rdx, rax
    sub rdx, rsi
    mov edi, 1

write_loop:
    mov eax, 1
    syscall
    test rax, rax
    jz write_done
    js write_done
    sub rdx, rax
    jz write_done
    add rsi, rax
    jmp write_loop

write_done:
    xor edi, edi
    mov eax, 60
    syscall

append_dec:
    mov rax, rdi
    test rax, rax
    jnz append_nonzero

    mov byte ptr [rsi], 48
    inc rsi
    mov rax, rsi
    ret

append_nonzero:
    lea r8, [rip + numbuf]
    xor rcx, rcx
    mov r10, 10

append_div:
    xor rdx, rdx
    div r10
    add dl, 48
    mov byte ptr [r8 + rcx], dl
    inc rcx
    test rax, rax
    jnz append_div

    mov rdx, rcx
    xor rcx, rcx

append_copy:
    dec rdx
    movzx eax, byte ptr [r8 + rdx]
    mov byte ptr [rsi + rcx], al
    inc rcx
    jnz append_copy

    add rsi, rcx
    mov rax, rsi
    ret

.bss
    .align 16
buf:
    .space 4096
    .align 16
numbuf:
    .space 32
    .align 16
outbuf:
    .space 64
