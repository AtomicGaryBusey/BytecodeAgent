

.globl _start
.equ IN_SIZE, 49152
.equ B64_SIZE, 65536
.equ OUT_SIZE, 66400
.equ LINE_LEN, 76

.text
_start:
    lea r14, [rip + input]
    lea r13, [rip + output]
    lea r9, [rip + b64buf]
    lea r15, [rip + table]
    xor r8, r8
read_loop:
    lea rsi, [r14 + r8]
    mov rdx, IN_SIZE
    sub rdx, r8
    test rdx, rdx
    jz drain
    mov eax, 0
    syscall
    test rax, rax
    jle read_done
    add r8, rax
    cmp r8, IN_SIZE
    jae drain
    jmp read_loop
drain:
    lea rsi, [rip + discard]
    mov rdx, 1
    mov eax, 0
    syscall
    test rax, rax
    jg drain
read_done:
    xor r10, r10
    xor r11, r11
encode_loop:
    cmp r10, r8
    jge encode_done
    mov rax, r8
    sub rax, r10
    cmp rax, 3
    jb encode_tail
    movzx eax, byte ptr [r14 + r10]
    movzx edx, byte ptr [r14 + r10 + 1]
    movzx ecx, byte ptr [r14 + r10 + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx
    mov ebx, eax
    mov eax, ebx
    shr eax, 18
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    shr eax, 12
    and eax, 63
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    shr eax, 6
    and eax, 63
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    and eax, 63
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    add r10, 3
    jmp encode_loop
encode_tail:
    cmp rax, 2
    je encode_two
    movzx eax, byte ptr [r14 + r10]
    mov ebx, eax
    mov eax, ebx
    shr eax, 2
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov byte ptr [r9 + r11], 61
    inc r11
    mov byte ptr [r9 + r11], 61
    inc r11
    add r10, 1
    jmp encode_loop
encode_two:
    movzx eax, byte ptr [r14 + r10]
    movzx edx, byte ptr [r14 + r10 + 1]
    shl eax, 8
    or eax, edx
    mov ebx, eax
    mov eax, ebx
    shr eax, 10
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    shr eax, 4
    and eax, 63
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov eax, ebx
    shr eax, 2
    and eax, 63
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r9 + r11], al
    inc r11
    mov byte ptr [r9 + r11], 61
    inc r11
    add r10, 2
    jmp encode_loop
encode_done:
    mov r8, r11
    xor r10, r10
    xor r11, r11
    xor r12, r12
transform_loop:
    cmp r10, r8
    jge transform_finish
    movzx eax, byte ptr [r9 + r10]
    mov byte ptr [r13 + r11], al
    inc r10
    inc r11
    inc r12
    cmp r12, LINE_LEN
    jne transform_loop
    mov byte ptr [r13 + r11], 10
    inc r11
    xor r12, r12
    jmp transform_loop
transform_finish:
    test r12, r12
    jz write_output
    mov byte ptr [r13 + r11], 10
    inc r11
write_output:
    test r11, r11
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r11
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
.align 8
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 8
input:
    .zero IN_SIZE
.align 8
b64buf:
    .zero B64_SIZE
.align 8
discard:
    .zero 1
.align 8
output:
    .zero OUT_SIZE
