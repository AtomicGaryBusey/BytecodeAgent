

.equ BUF_SIZE, 65537
.equ READ_SYSCALL, 63
.equ WRITE_SYSCALL, 64
.equ EXIT_SYSCALL, 93
.equ NEWLINE, 10

.text
.globl _start

_start:
    adrp x10, buf
    add  x10, x10, :lo12:buf

    mov  x11, xzr
    movz x12, #1, lsl #16

read_loop:
    cmp  x11, x12
    b.eq read_done

    sub  x2, x12, x11
    mov  x8, #READ_SYSCALL
    mov  x0, xzr
    add  x1, x10, x11
    svc  #0

    cbz  x0, read_done
    add  x11, x11, x0
    b    read_loop

read_done:
    mov  x9, xzr

line_loop:
    cmp  x9, x11
    b.lo line_body
    b    final_check

line_body:
    ldrb w14, [x10, x9]
    cmp  w14, #NEWLINE
    b.eq empty_line

    add  x15, x9, #1

find_nl:
    cmp  x15, x11
    b.cs no_nl
    ldrb w14, [x10, x15]
    cmp  w14, #NEWLINE
    b.eq found_nl
    add  x15, x15, #1
    b    find_nl

found_nl:
    mov  x14, x15
    b    reverse_line

no_nl:
    mov  x14, x11

reverse_line:
    mov  x15, x9
    sub  x16, x14, #1

swap_loop:
    cmp  x15, x16
    b.cs swap_done

    ldrb w17, [x10, x15]
    ldrb w18, [x10, x16]
    strb w17, [x10, x16]
    strb w18, [x10, x15]

    add  x15, x15, #1
    sub  x16, x16, #1
    b    swap_loop

swap_done:
    add  x9, x14, #1
    b    line_loop

empty_line:
    add  x9, x9, #1
    b    line_loop

final_check:
    cbz  x11, exit

    sub  x12, x11, #1
    ldrb w14, [x10, x12]
    cmp  w14, #NEWLINE
    b.ne add_final_nl

    mov  x14, x11
    b    write_out

add_final_nl:
    mov  w14, #NEWLINE
    strb w14, [x10, x11]
    add  x14, x11, #1

write_out:
    mov  x8, #WRITE_SYSCALL
    mov  x0, #1
    mov  x1, x10
    mov  x2, x14
    svc  #0

exit:
    mov  x8, #EXIT_SYSCALL
    mov  x0, xzr
    svc  #0

.bss
buf:
    .space BUF_SIZE
