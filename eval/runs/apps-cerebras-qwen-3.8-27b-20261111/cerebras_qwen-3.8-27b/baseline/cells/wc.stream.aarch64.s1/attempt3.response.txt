

.text
.globl _start
.balign 16

_start:
    mov x10, #0
    mov x11, #0
    mov x12, #0
    mov x13, #0
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf

read_loop:
    mov x0, #0
    mov x1, x19
    mov w2, #8192
    mov w8, #63
    svc #0
    cmp x0, #0
    beq done_read
    b.lt error_exit
    mov x3, x19
    mov x4, x0

byte_loop:
    ldrb w5, [x3], #1
    add x11, x11, #1
    subs x4, x4, #1
    cmp w5, #10
    b.eq newline
    cmp w5, #9
    b.eq whitespace
    cmp w5, #11
    b.eq whitespace
    cmp w5, #12
    b.eq whitespace
    cmp w5, #13
    b.eq whitespace
    cmp w5, #32
    b.eq whitespace
    cbz x13, word_start
    b byte_next

newline:
    add x10, x10, #1

whitespace:
    mov x13, #0
    b byte_next

word_start:
    mov x13, #1
    add x12, x12, #1
    b byte_next

byte_next:
    cbnz x4, byte_loop
    b read_loop

done_read:
    mov x0, x10
    bl print_uint
    mov w0, #32
    bl write_char
    mov x0, x12
    bl print_uint
    mov w0, #32
    bl write_char
    mov x0, x11
    bl print_uint
    mov w0, #10
    bl write_char
    mov x0, #0
    mov w8, #94
    svc #0

error_exit:
    mov x0, #1
    mov w8, #94
    svc #0

print_uint:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x26, #0
    cbz x0, pu_zero

pu_next:
    mov x25, x0
    mov x22, #0

pu_divide:
    cmp x25, #10
    b.lt pu_store
    sub x25, x25, #10
    add x22, x22, #1
    b pu_divide

pu_store:
    add w23, w25, #48
    strb w23, [x24, x26]
    add x26, x26, #1
    mov x0, x22
    cbnz x0, pu_next
    b pu_write

pu_zero:
    mov w23, #48
    strb w23, [x24]
    mov x26, #1

pu_write:
    sub x27, x26, #1

pu_write_loop:
    ldrb w23, [x24, x27]
    mov x0, #1
    add x1, x24, x27
    mov w2, #1
    mov w8, #64
    svc #0
    cbz x27, pu_done
    sub x27, x27, #1
    b pu_write_loop

pu_done:
    ret

write_char:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    strb w0, [x24]
    mov x0, #1
    mov x1, x24
    mov w2, #1
    mov w8, #64
    svc #0
    ret

.bss
.balign 16
inbuf:
    .space 8192
outbuf:
    .space 16
