

.text
.globl _start

_start:
    mov w19, #0
    mov w20, #0
    mov w21, #0
    mov w22, #0
    adrp x9, readbuf
    add x9, x9, :lo12:readbuf
    mov x10, #65536
read_loop:
    mov x0, #0
    mov x1, x9
    mov x2, x10
    mov x8, #63
    svc #0
    cmp x0, #1
    b.lt read_done
    mov x1, x9
    mov x2, x0
byte_loop:
    ldrb w3, [x1], #1
    add w21, w21, #1
    cmp w3, #10
    b.ne not_newline
    add w19, w19, #1
    mov w22, #0
    b end_byte
not_newline:
    cmp w3, #9
    b.eq other_ws
    cmp w3, #11
    b.eq other_ws
    cmp w3, #12
    b.eq other_ws
    cmp w3, #13
    b.eq other_ws
    cmp w3, #32
    b.eq other_ws
    cbnz w22, end_byte
    add w20, w20, #1
    mov w22, #1
    b end_byte
other_ws:
    mov w22, #0
end_byte:
    subs x2, x2, #1
    b.ne byte_loop
    b read_loop
read_done:
    mov x0, x19
    bl print_uint
    adrp x0, space_char
    add x0, x0, :lo12:space_char
    bl write_char
    mov x0, x20
    bl print_uint
    adrp x0, space_char
    add x0, x0, :lo12:space_char
    bl write_char
    mov x0, x21
    bl print_uint
    adrp x0, newline_char
    add x0, x0, :lo12:newline_char
    bl write_char
    mov x0, #0
    mov x8, #94
    svc #0

print_uint:
    adrp x1, digits_end
    add x1, x1, :lo12:digits_end
    mov x17, x1
    mov w3, #10
    cbz x0, print_zero
collect:
    udiv w4, w0, w3
    mul w4, w4, w3
    sub w4, w0, w4
    add w4, w4, #48
    subs x1, x1, #1
    strb w4, [x1]
    udiv w0, w0, w3
    cbnz w0, collect
print_digit:
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x1, x1, #1
    cmp x1, x17
    b.lo print_digit
    ret
print_zero:
    adrp x1, zero_char
    add x1, x1, :lo12:zero_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret

write_char:
    mov x1, x0
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
space_char:
    .byte 32
newline_char:
    .byte 10
zero_char:
    .byte 48

.bss
.align 4
readbuf:
.space 65536
.align 4
digits:
.space 16
digits_end:
