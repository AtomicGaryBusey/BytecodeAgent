finish_padding:

rem2:

do_write:

write_loop:

    test rax, rax
    jle exit_prog
    add rsi, rax
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_prog

    mov byte ptr [rsi], al
    inc rsi
    inc r13
    cmp r13, 76
    jne out_char_done
    mov byte ptr [rsi], 10
    inc rsi
    xor r13, r13
out_char_done:
