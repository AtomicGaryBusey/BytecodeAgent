

.text
.globl _start

.equ INPUT_MAX, 49152

_start:
    xor r12, r12
read_loop:
    mov rdx, INPUT_MAX
    sub rdx, r12
    test rdx, rdx
    jz read_done
    lea rsi, [rip + input]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov rax, r12
    xor rdx, rdx
    mov rcx, 3
    div rcx
    mov r14, rax
    mov r15, rdx
    lea rdi, [rip + input]
    lea rsi, [rip + output]
    mov r12, rsi
    lea rbx, [rip + b64tab]
    xor r13, r13
    jmp encode_loop
encode_loop:
    test r14, r14
    jz remainder
    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    movzx r10d, byte ptr [rdi + 2]
    add rdi, 3
    dec r14
    mov eax, r8d
    shr eax, 2
    call put_idx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    call put_idx
    mov eax, r9d
    and eax, 0xf
    shl eax, 2
    mov ecx, r10d
    shr ecx, 6
    or eax, ecx
    call put_idx
    mov eax, r10d
    and eax, 0x3f
    call put_idx
    jmp encode_loop
remainder:
    test r15, r15
    jz finish_encode
    cmp r15, 1
    jne remainder_two
rem_one:
    movzx r8d, byte ptr [rdi]
    add rdi, 1
    mov eax, r8d
    shr eax, 2
    call put_idx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    call put_idx
    mov eax, 61
    call emit
    mov eax, 61
    call emit
    jmp finish_encode
remainder_two:
    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdi + 1]
    add rdi, 2
    mov eax, r8d
    shr eax, 2
    call put_idx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    call put_idx
    mov eax, r9d
    and eax, 0xf
    shl eax, 2
    call put_idx
    mov eax, 61
    call emit
    jmp finish_encode
finish_encode:
    test r13, r13
    jz write_output
    mov eax, 10
    mov byte ptr [rsi], al
    inc rsi
write_output:
    mov rax, rsi
    sub rax, r12
    test rax, rax
    jz exit_prog
    mov rsi, r12
    mov r14, rax
write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, r14
    syscall
    cmp rax, 0
    jle exit_prog
    add rsi, rax
    sub r14, rax
    test r14, r14
    jnz write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
    jmp $
put_idx:
    movzx eax, byte ptr [rbx + rax]
    call emit
    ret
emit:
    mov byte ptr [rsi], al
    inc rsi
    inc r13
    cmp r13, 76
    jne emit_done
    mov eax, 10
    mov byte ptr [rsi], al
    inc rsi
    xor r13, r13
emit_done:
    ret

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
    .zero 49152
output:
    .zero 70000
