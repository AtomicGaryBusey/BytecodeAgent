

.globl _start

.equ IN_SIZE, 65536
.equ OUT_SIZE, 65538

.section .bss
.align 16
inbuf:
.space IN_SIZE
outbuf:
.space OUT_SIZE

.section .text
_start:
    mov r13d, IN_SIZE
    xor r12d, r12d

read_loop:
    test r13, r13
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rax, 0
    xor edi, edi
    mov rdx, r13
    syscall
    cmp rax, 0
    je read_done
    jle read_error
    add r12, rax
    sub r13, rax
    jmp read_loop

read_done:
    xor r13d, r13d
    xor r14d, r14d
    lea rdi, [rip + inbuf]
    lea rsi, [rip + outbuf]

line_loop:
    cmp r14, r12
    jae line_done
    mov r8, r14

find_newline:
    mov al, byte ptr [rdi + r8]
    cmp al, 0x0a
    je newline_found
    inc r8
    cmp r8, r12
    jb find_newline
    jmp newline_found

newline_found:
    mov r9, r8
    sub r9, r14
    test r9, r9
    jz put_newline
    lea r15, [r14 + r9]

reverse_loop:
    dec r15
    mov al, byte ptr [rdi + r15]
    mov byte ptr [rsi + r13], al
    inc r13
    cmp r15, r14
    jne reverse_loop

put_newline:
    mov byte ptr [rsi + r13], 0x0a
    inc r13
    lea r14, [r8 + 1]
    jmp line_loop

line_done:
    mov rax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall
    jmp exit_success

read_error:
    mov edi, 1
    mov eax, 60
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall
