

.globl _start
.equ buf_cap, 65536
.equ tape_size, 30000
.equ table_size, 262144
.equ stack_size, 262144

.section .text
_start:
    xor r8, r8
read_loop:
    cmp r8, buf_cap
    je read_done
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, r8
    mov rdx, buf_cap
    sub rdx, r8
    syscall
    test rax, rax
    js fail_exit
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    lea rdi, [rip + inbuf]
    xor esi, esi
scan_loop:
    cmp rsi, r8
    jge not_found
    movzx eax, byte ptr [rdi + rsi]
    cmp al, 33
    je found
    inc rsi
    jmp scan_loop
found:
    mov r9, rsi
    mov r14, rsi
    inc r14
    mov r15, r8
    sub r15, r14
    jmp bracket_build
not_found:
    mov r9, r8
    xor r14, r14
    xor r15, r15
bracket_build:
    xor ebx, ebx
    xor ecx, ecx
build_loop:
    cmp ecx, r9
    jge build_done
    movzx eax,
