

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_OUTPUT, (MAX_INPUT + 1)
.equ MAX_LINES, 1000

.section .bss
.align 16
input:
    .zero MAX_INPUT
line_offs:
    .zero (MAX_LINES * 8)
line_lens:
    .zero (MAX_LINES * 8)
.align 16
out_buf:
    .zero MAX_OUTPUT

.section .text
.align 16
.globl _start
_start:
    lea rsi, [rip + input]
    mov rdx, MAX_INPUT
    xor r13d, r13d
read_loop:
    mov eax, SYS_read
    mov edi, FD_STDIN
    syscall
    cmp rax, 0
    je read_done
    add r13d, eax
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jne read_loop
read_done:
    xor r12d, r12d
    xor r15d, r15d
parse_loop:
    cmp r15d, r13d
    jge parse_done
    lea rsi, [rip + input + r15]
    lea
