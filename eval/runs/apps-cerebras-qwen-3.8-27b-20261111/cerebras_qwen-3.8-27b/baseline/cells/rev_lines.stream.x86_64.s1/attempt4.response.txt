

.set IN_MAX, 65536
.set OUT_MAX, 65537

.text
.globl _start

_start:
  lea r12, [rip + in_buf]
  xor r13, r13

read_loop:
  cmp r13, IN_MAX
  jge read_done
  xor eax, eax
  xor edi, edi
  lea rsi, [r12 + r13]
  mov edx, IN_MAX
  sub edx, r13d
  syscall
  cmp rax, 0
  jle read_done
  add r13, rax
  jmp read_loop

read_done:
  lea r14, [rip + out_buf]
  xor r15, r15
  xor rbx, rbx

line_loop:
  cmp rbx, r13
  jge done_lines
  mov rcx, rbx

scan_loop:
  cmp rcx, r13
  je line_end_eof
  mov al, byte ptr [r12 + rcx]
  cmp al, 10
  je line_end_nl
  inc rcx
  jmp scan_loop

line_end_nl:
  mov r8, rcx
  jmp emit_line

line_end_eof:
  mov r8, r13

emit_line:
  mov r10, r8
  sub r10, rbx
  test r10, r10
  jz append_newline
  mov rcx, r8
  dec rcx
  mov rdi, r15

reverse_loop:
  mov al, byte ptr [r12 + rcx]
  mov byte ptr [r14 + rdi], al
  inc rdi
  dec rcx
  dec r10
  jnz reverse_loop
  mov r15, rdi

append_newline:
  mov byte ptr [r14 + r15], 10
  inc r15
  cmp r8, r13
  je done_lines
  lea rbx, [r8 + 1]
  jmp line_loop

done_lines:
  test r15, r15
  jz exit0
  mov rsi, r14
  mov rdx, r15

write_loop:
  mov eax, 1
  mov edi, 1
  syscall
  mov r8, rax
  test r8, r8
  jle write_done
  cmp r8, rdx
  je write_done
  add rsi, r8
  sub rdx, r8
  jmp write_loop

write_done:
exit0:
  mov eax, 60
  xor edi, edi
  syscall

.bss
.align 16
in_buf:
.space IN_MAX
out_buf:
.space OUT_MAX
