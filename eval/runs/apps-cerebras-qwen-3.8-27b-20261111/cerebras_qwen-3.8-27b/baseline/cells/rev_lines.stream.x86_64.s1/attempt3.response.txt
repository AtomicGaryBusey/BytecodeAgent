

.equ BUF_SIZE, 65537
.equ OUT_SIZE, BUF_SIZE + 1

.bss
.align 16
buf:
.space BUF_SIZE
outbuf:
.space OUT_SIZE

.text
.globl _start
_start:
  lea r12, [rip + buf]
  xor r13, r13

read_loop:
  cmp r13, BUF_SIZE
  jae read_done
  mov rdi, 0
  lea rsi, [r12 + r13]
  mov rdx, BUF_SIZE
  sub rdx, r13
  syscall
  cmp rax, 0
  jle read_done
  add r13, rax
  jmp read_loop

read_done:
  lea r12, [rip + buf]
  lea r14, [rip + outbuf]
  xor rbx, rbx
  xor r15, r15

line_loop:
  cmp rbx, r13
  jge done_lines
  mov rax, rbx

scan_loop:
  cmp rax, r13
  je found_end
  mov al, byte ptr [r12 + rax]
  cmp al, 10
  je found_nl
  inc rax
  jmp scan_loop

found_nl:
  mov r8, rax
  jmp make_line

found_end:
  mov r8, r13

make_line:
  mov r10, r8
  sub r10, rbx
  test r10, r10
  jz put_newline
  mov rcx, r8
  dec rcx
  mov rsi, r15

reverse_loop:
  mov al, byte ptr [r12 + rcx]
  mov byte ptr [r14 + rsi], al
  inc rsi
  dec rcx
  dec r10
  jnz reverse_loop
  mov r15, rsi

put_newline:
  mov byte ptr [r14 + r15], 10
  inc r15
  cmp r8, r13
  je done_lines
  lea rbx, [r8 + 1]
  jmp line_loop

done_lines:
  test r15, r15
  jz exit0
  mov rdi, 1
  mov rsi, r14
  mov rdx, r15
  syscall

exit0:
  mov rdi, 60
  xor rsi, rsi
  syscall
