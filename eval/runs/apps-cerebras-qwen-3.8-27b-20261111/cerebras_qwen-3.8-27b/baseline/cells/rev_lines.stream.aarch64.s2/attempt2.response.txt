

.text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    adrp x9, inbuf
    add x9, x9, :lo12:inbuf
    adrp x11, outbuf
    add x11, x11, :lo12:outbuf
    mov x12, #2
    lsl x12, x12, #16
    mov x10, #0
    mov x13, #0

read_loop:
    cmp x10, x12
    b.eq read_done
    sub x2, x12, x10
    add x1, x9, x10
    mov x0, #0
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done
    cmp x0, x2
    b.hi read_done
    add x10, x10, x0
    b read_loop

read_done:
    mov x3, x9
    add x4, x9, x10

line_loop:
    cmp x3, x4
    b.eq finish
    mov x5, x3

find_nl:
    cmp x5, x4
    b.eq make_line
    ldrb w6, [x5]
    cmp w6, #10
    b.eq make_line
    add x5, x5, #1
    b find_nl

make_line:
    sub x6, x5, x3
    add x8, x11, x13
    mov x7, x5
    cbz x6, store_nl
    sub x7, x7, #1

reverse:
    ldrb w9, [x7]
    strb w9, [x8], #1
    sub x7, x7, #1
    sub x6, x6, #1
    cbnz x6, reverse

store_nl:
    mov w9, #10
    strb w9, [x8]
    add x8, x8, #1
    sub x13, x8, x11

    cmp x5, x4
    b.eq finish
    add x3, x5, #1
    b line_loop

finish:
    cbz x13, exit_prog
    mov x14, x13
    mov x15, x11

write_final:
    mov x0, #1
    mov x1, x15
    mov x2, x14
    mov x8, SYS_WRITE
    svc #0
    cmp x0, x14
    b.cs exit_prog
    cbz x0, exit_prog
    add x15, x15, x0
    sub x14, x14, x0
    b write_final

exit_prog:
    mov x0, #0
    mov x8, SYS_EXIT
    svc #0

.bss
.balign 16
inbuf:
    .space 131072
outbuf:
    .space 131073
