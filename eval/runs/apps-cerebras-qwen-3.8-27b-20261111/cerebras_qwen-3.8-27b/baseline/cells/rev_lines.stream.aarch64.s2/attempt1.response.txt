

.text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ NEWLINE, 10

_start:
    adrp x9, inbuf
    add x9, x9, :lo12:inbuf
    mov x10, #0
    movz x11, #1, lsl #16
read_loop:
    cmp x10, x11
    b.eq read_done
    sub x2, x11, x10
    add x1, x9, x10
    mov x0, #0
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done
    cmp x0, x2
    b.hi read_done
    add x10, x10, x0
    b read_loop
read_done:
    mov x3, x9
    add x4, x3, x10
    adrp x7, outbuf
    add x7, x7, :lo12:outbuf
line_loop:
    cmp x3, x4
    b.eq finish
    mov x2, x3
find_newline:
    ldrb w5, [x3]
    cmp w5, NEWLINE
    b.eq line_found
    add x3, x3, #1
    cmp x3, x4
    b.ne find_newline
line_found:
    mov x6, x3
    sub x6, x6, x2
    mov x5, x7
    mov x8, x3
    sub x8, x8, #1
    cbz x6, store_newline
fill_reverse:
    ldrb w9, [x8]
    strb w9, [x5], #1
    sub x8, x8, #1
    sub x6, x6, #1
    cbnz x6, fill_reverse
store_newline:
    mov w9, NEWLINE
    strb w9, [x5]
    add x6, x6, #1
write_loop:
    mov x0, #1
    mov x1, x5
    mov x2, x6
    mov x8, SYS_WRITE
    svc #0
    cbz x0, write_done
    cmp x0, x6
    b.cs write_done
    add x5, x5, x0
    sub x6, x6, x0
    b write_loop
write_done:
    cmp x3, x4
    b.eq finish
    add x3, x3, #1
    b line_loop
finish:
    mov x0, #0
    mov x8, SYS_EXIT
    svc #0

.bss
inbuf:
    .space 65536
outbuf:
    .space 65537
