

.equ BUF_SIZE, 65536
.globl _start
.text
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r15, r15
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    mov rdx, BUF_SIZE
    syscall
    mov rcx, rax
    test rax, rax
    jz finished_read
    lea rsi, [rip + inbuf]
process_loop:
    movzx eax, byte ptr [rsi]
    inc r10
    cmp al, 10
    je newline
    cmp al, 9
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    cmp al, 32
    je whitespace
    test r15, r15
    jnz next_byte
    inc r9
    mov r15, 1
next_byte:
    inc rsi
    dec rcx
    jnz process_loop
    jmp read_loop
newline:
    inc r8
    mov r15, 0
    jmp next_byte
whitespace:
    mov r15, 0
    jmp next_byte
finished_read:
    mov qword ptr [counts], r8
    mov qword ptr [counts + 8], r9
    mov qword ptr [counts + 16], r10
    lea r12, [rip + outbuf]
    mov rbx, 10
    xor r13, r13
numbers_loop:
    mov rax, qword ptr [counts + r13]
    xor rcx, rcx
convert_loop:
    xor rdx, rdx
    div rbx
    mov r14, rax
    movzx eax, dl
    add al, 48
    mov byte ptr [digitbuf + rcx], al
    mov rax, r14
    inc rcx
    test rax, rax
    jnz convert_loop
    lea rsi, [rip + digitbuf]
    add rsi, rcx
    dec rsi
copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [r12], al
    inc r12
    dec rsi
    dec rcx
    test rcx, rcx
    jnz copy_loop
    cmp r13, 16
    je finished_numbers
    mov byte ptr [r12], 32
    inc r12
    add r13, 8
    jmp numbers_loop
finished_numbers:
    mov byte ptr [r12], 10
    inc r12
    lea rbx, [rip + outbuf]
    mov rdx, r12
    sub rdx, rbx
    mov rsi, rbx
    mov rdi, 1
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
.bss
.align 16
inbuf:
    .zero 65536
digitbuf:
    .space 32
outbuf:
    .space 32
counts:
    .space 24
