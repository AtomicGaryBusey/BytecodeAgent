

.text
.globl _start
.equ SYSCALL_READ, 0
.equ SYSCALL_WRITE, 1
.equ SYSCALL_EXIT, 60
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 65537
.equ NEWLINE, 0x0A

_start:
    xor rbx, rbx
    lea rsi, [rip + inbuf]
    mov rdx, IN_BUF_SIZE
read_loop:
    mov rax, SYSCALL_READ
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov rcx, rbx
    xor rbx, rbx
    lea r8, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r10, r10
process_loop:
    cmp rbx, rcx
    jae finish
    cmp byte ptr [r8 + rbx], NEWLINE
    je empty_line
    mov rdx, rbx
find_newline:
    cmp rdx, rcx
    jae final_line
    cmp byte ptr [r8 + rdx], NEWLINE
    je line_found
    inc rdx
    jmp find_newline
empty_line:
    mov byte ptr [rdi], NEWLINE
    inc rdi
    inc r10
    inc rbx
    jmp process_loop
line_found:
    mov r9, rdx
    sub r9, rbx
    mov rsi, rdx
    dec rsi
reverse_line:
    movzx eax, byte ptr [r8 + rsi]
    mov byte ptr [rdi], al
    inc rdi
    inc r10
    dec rsi
    dec r9
    jnz reverse_line
    mov byte ptr [rdi], NEWLINE
    inc rdi
    inc r10
    mov rbx, rdx
    inc rbx
    jmp process_loop
final_line:
    mov r9, rcx
    sub r9, rbx
    mov rsi, rcx
    dec rsi
reverse_final:
    movzx eax, byte ptr [r8 + rsi]
    mov byte ptr [rdi], al
    inc rdi
    inc r10
    dec rsi
    dec r9
    jnz reverse_final
    mov byte ptr [rdi], NEWLINE
    inc rdi
    inc r10
    mov rbx, rcx
finish:
    test r10, r10
    jz exit_0
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r10
write_loop:
    mov rax, SYSCALL_WRITE
    syscall
    test rax, rax
    jle write_done
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_loop
write_done:
exit_0:
    xor edi, edi
    mov rax, SYSCALL_EXIT
    syscall

.bss
inbuf:
.zero IN_BUF_SIZE
outbuf:
.zero OUT_BUF_SIZE
