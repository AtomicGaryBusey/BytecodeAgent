

.equ STDIN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_ARRAY_SIZE, 65536
.equ JUMP_TABLE_SIZE, 262144
.equ BRACKET_STACK_SIZE, 262144

.equ BANG, 33
.equ OPEN_BRACKET, 91
.equ CLOSE_BRACKET, 93
.equ GT, 62
.equ LT, 60
.equ PLUS, 43
.equ MINUS, 45
.equ DOT, 46
.equ COMMA, 44

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN_FD, 0
.equ STDOUT_FD, 1

.globl _start

.text
_start:
    xor r12, r12
read_loop:
    cmp r12, STDIN_BUF_SIZE
    jae read_done
    lea rsi, [rip + stdin_buffer]
    add rsi, r12
    mov rdx, STDIN_BUF_SIZE
    sub rdx, r12
    mov eax, SYS_READ
    mov edi, STDIN_FD
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rcx, rcx
find_sep:
    cmp rcx, r12
    jae no_sep
    cmp byte ptr [rip + stdin_buffer + rcx], BANG
    je sep_found
    inc rcx
    jmp find_sep
no_sep:
    mov r9, r12
    mov r10, r12
    jmp bracket_scan
sep_found:
    mov r9, rcx
    lea r10, [rcx + 1]
bracket_scan:
    xor rcx, rcx
    xor rdx, rdx
bracket_scan_top:
    cmp rcx, r9
    je bracket_scan_done
    movzx eax, byte ptr [rip + stdin_buffer + rcx]
    cmp al, OPEN_BRACKET
    je bracket_open
    cmp al, CLOSE_BRACKET
    je bracket_close
    jmp bracket_scan_next
bracket_open:
    cmp rdx, BRACKET_ARRAY_SIZE
    jae invalid_exit
    mov [rip + bracket_stack + rdx*4], ecx
    inc rdx
    jmp bracket_scan_next
bracket_close:
    test rdx, rdx
    jz invalid_exit
    dec rdx
    mov ebx, [rip + bracket_stack + rdx*4]
    lea r8d, [rbx + 1]
    mov [rip + jump_table + rcx*4], r8d
    lea r8d, [ecx + 1]
    mov [rip + jump_table + rbx*4], r8d
bracket_scan_next:
    inc rcx
    jmp bracket_scan_top
bracket_scan_done:
    test rdx, rdx
    jnz invalid_exit

    mov r8, r10
    xor r10, r10
    xor r14, r14
    xor r15, r15
exec_top:
    cmp r15, r9
    jae exec_done
    movzx eax, byte ptr [rip + stdin_buffer + r15]
    cmp al, GT
    je cmd_gt
    cmp al, LT
    je cmd_lt
    cmp al, PLUS
    je cmd_plus
    cmp al, MINUS
    je cmd_minus
    cmp al, DOT
    je cmd_dot
    cmp al, COMMA
    je cmd_comma
    cmp al, OPEN_BRACKET
    je cmd_open
    cmp al, CLOSE_BRACKET
    je cmd_close
    jmp exec_next
cmd_gt:
    inc r14
    jmp exec_next
cmd_lt:
    dec r14
    jmp exec_next
cmd_plus:
    inc byte ptr [rip + tape + r14]
    jmp exec_next
cmd_minus:
    dec byte ptr [rip + tape + r14]
    jmp exec_next
cmd_dot:
    movzx eax, byte ptr [rip + tape + r14]
    mov [rip + out_buf + r10], al
    inc r10
    cmp r10, OUT_BUF_SIZE
    jne exec_next
    mov eax, SYS_WRITE
    mov edi, STDOUT_FD
    lea rsi, [rip + out_buf]
    mov rdx, r10
    syscall
    xor r10, r10
    jmp exec_next
cmd_comma:
    cmp r8, r12
    jae cmd_comma_zero
    movzx eax, byte ptr [rip + stdin_buffer + r8]
    inc r8
    mov [rip + tape + r14], al
    jmp exec_next
cmd_comma_zero:
    mov byte ptr [rip + tape + r14], 0
    jmp exec_next
cmd_open:
    movzx eax, byte ptr [rip + tape + r14]
    test al, al
    jz cmd_open_zero
    jmp exec_next
cmd_open_zero:
    mov eax, [rip + jump_table + r15*4]
    mov r15, rax
    jmp exec_top
cmd_close:
    movzx eax, byte ptr [rip + tape + r14]
    test al, al
    jz exec_next
    mov eax, [rip + jump_table + r15*4]
    mov r15, rax
    jmp exec_top
exec_next:
    inc r15
    jmp exec_top
exec_done:
    test r10, r10
    jz exit_success
    mov eax, SYS_WRITE
    mov edi, STDOUT_FD
    lea rsi, [rip + out_buf]
    mov rdx, r10
    syscall
    xor r10, r10
exit_success:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
    jmp exit_success
invalid_exit:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
    jmp invalid_exit

.bss
.balign 16
stdin_buffer:
.zero STDIN_BUF_SIZE
jump_table:
.zero JUMP_TABLE_SIZE
bracket_stack:
.zero BRACKET_STACK_SIZE
tape:
.zero TAPE_SIZE
out_buf:
.zero OUT_BUF_SIZE
