

.text
.globl _start
_start:
    lea r12, [rip + in_buf]
    lea r13, [rip + out_buf]
    xor rbx, rbx

read_loop:
    cmp rbx, 65536
    jae read_done
    xor rdi, rdi
    lea rsi, [r12 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    xor rax, rax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r14, [r12 + rbx]
    mov r15, r12
    mov r8, r13

process_loop:
    mov r11, r15
    mov r10, r11

find_newline:
    cmp r10, r14
    je no_newline
    movzx eax, byte ptr [r10]
    cmp al, 0x0a
    je found_newline
    inc r10
    jmp find_newline

found_newline:
    mov r9, r10
    lea r15, [r10 + 1]

reverse_line:
    cmp r9, r11
    jbe reverse_line_done
    dec r9
    movzx eax, byte ptr [r9]
    mov byte ptr [r8], al
    inc r8
    jmp reverse_line

reverse_line_done:
    mov byte ptr [r8], 0x0a
    inc r8
    jmp process_loop

no_newline:
    cmp r11, r14
    jae finished
    mov r9, r14
    jmp final_reverse

final_reverse:
    cmp r9, r11
    jbe final_reverse_done
    dec r9
    movzx eax, byte ptr [r9]
    mov byte ptr [r8], al
    inc r8
    jmp final_reverse

final_reverse_done:
    mov byte ptr [r8], 0x0a
    inc r8

finished:
    mov rsi, r13
    mov rdx, r8
    sub rdx, r13
    test rdx, rdx
    je exit_zero

write_loop:
    mov rdi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle exit_zero
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jne write_loop

exit_zero:
    xor rdi, rdi
    mov eax, 231
    syscall

.bss
in_buf:
.space 65536
out_buf:
.space 65537
