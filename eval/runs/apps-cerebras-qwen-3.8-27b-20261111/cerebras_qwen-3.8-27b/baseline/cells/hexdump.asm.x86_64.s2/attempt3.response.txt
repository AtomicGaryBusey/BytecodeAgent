

.text
.globl _start
.equ MAXIN, 65536
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ B_SPACE, 32
.equ B_PIPE, 124
.equ B_DOT, 46
.equ B_NL, 10

_start:
    lea r8, [rip + hex]
    lea r13, [rip + outbuf]
    xor r9, r9
    jmp read_loop

read_loop:
    cmp r9, MAXIN
    je read_done
    mov rdx, MAXIN
    sub rdx, r9
    lea rsi, [rip + inbuf]
    add rsi, r9
    xor edi, edi
    mov eax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop

read_done:
    xor r10, r10
row_loop:
    cmp r10, r9
    jae print_final
    mov r11, r9
    sub r11, r10
    mov r14, 16
    cmp r11, 16
    ja row_n_set
    mov r14, r11
row_n_set:
    lea rsi, [rip + outbuf]
    xor r15, r15
    mov r11, rsi
    lea r12, [rsi + 7]
    mov rax, r10
    mov ecx, 8
row_off_loop:
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r12], dl
    dec r12
    shr eax, 4
    dec ecx
    jnz row_off_loop
    lea rsi, [r11 + 8]
    add r15, 8
    mov byte ptr [rsi], B_SPACE
    mov byte ptr [rsi + 1], B_SPACE
    add rsi, 2
    add r15, 2
    lea rbx, [rip + inbuf]
    add rbx, r10
    xor rdx, rdx
row_hex_loop:
    cmp rdx, r14
    jb row_hex_byte
    mov byte ptr [rsi], B_SPACE
    mov byte ptr [rsi + 1], B_SPACE
    mov byte ptr [rsi + 2], B_SPACE
    add rsi, 3
    add r15, 3
    jmp row_hex_after
row_hex_byte:
    movzx eax, byte ptr [rbx + rdx]
    mov edx, eax
    shr edx, 4
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov byte ptr [rsi], B_SPACE
    inc rsi
    add r15, 3
row_hex_after:
    cmp rdx, 7
    je row_hex_extra
    inc rdx
    cmp rdx, 16
    jne row_hex_loop
    jmp row_ascii
row_hex_extra:
    mov byte ptr [rsi], B_SPACE
    inc rsi
    inc r15
    inc rdx
    cmp rdx, 16
    jne row_hex_loop
row_ascii:
    mov byte ptr [rsi], B_SPACE
    mov byte ptr [rsi + 1], B_PIPE
    add rsi, 2
    add r15, 2
    xor rdx, rdx
row_ascii_loop:
    cmp rdx, r14
    jae row_ascii_close
    movzx eax, byte ptr [rbx + rdx]
    cmp al, 0x20
    jb row_ascii_dot
    cmp al, 0x7e
    ja row_ascii_dot
    mov byte ptr [rsi], al
    inc rsi
    inc r15
    inc rdx
    cmp rdx, r14
    jne row_ascii_loop
    jmp row_ascii_close
row_ascii_dot:
    mov al, B_DOT
    mov byte ptr [rsi], al
    inc rsi
    inc r15
    inc rdx
    cmp rdx, r14
    jne row_ascii_loop
row_ascii_close:
    mov byte ptr [rsi], B_PIPE
    mov byte ptr [rsi + 1], B_NL
    add rsi, 2
    add r15, 2
    mov edi, 1
    mov rsi, r13
    mov rdx, r15
    mov eax, SYS_WRITE
    syscall
    add r10, 16
    jmp row_loop

print_final:
    lea rsi, [rip + outbuf]
    xor r15, r15
    mov r11, rsi
    lea r12, [rsi + 7]
    mov rax, r
