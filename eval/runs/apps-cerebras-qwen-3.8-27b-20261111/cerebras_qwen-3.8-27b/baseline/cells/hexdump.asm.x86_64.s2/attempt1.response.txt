

.section .rodata
.align 4
hexdigits:
.ascii "0123456789abcdef"

.bss
.align 16
inbuf:
.zero 65536
obuf:
.zero 128

.text
.globl _start
_start:
    lea r15, [rip + hexdigits]
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov rdx, 65536
    xor r12, r12
read_loop:
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_error
    add r12, rax
    add rsi, rax
    sub rdx, rax
    jnz read_loop
    jmp read_done
read_error:
    mov edi, 1
    mov eax, 60
    syscall
read_done:
    xor r13, r13
row_loop:
    cmp r13, r12
    jae final_line
    mov r8, r12
    sub r8, r13
    cmp r8, 16
    jbe set_n
    mov r8, 16
set_n:
    mov r9, r13
    lea rsi, [rip + inbuf]
    add rsi, r13
    lea rdi, [rip + obuf]
    mov r14, rsi

    mov r10, r9
    shr r10, 28
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 24
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 20
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 16
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 12
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 8
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 4
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

    xor r11, r11
field_loop:
    cmp r11, r8
    jb field_present
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    jmp slot_done
field_present:
    movzx eax, byte ptr [rsi]
    mov r10, rax
    shr r10, 4
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi
    mov r10, rax
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    inc rsi
slot_done:
    cmp r11, 7
    je extra_space
    jmp field_next
extra_space:
    mov byte ptr [rdi], ' '
    inc rdi
field_next:
    inc r11
    cmp r11, 16
    jb field_loop

    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    mov rsi, r14
    mov r11, r8
ascii_loop:
    test r11, r11
    jz ascii_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov sil, al
    jmp put_char
ascii_dot:
    mov sil, '.'
put_char:
    mov byte ptr [rdi], sil
    inc rdi
    inc rsi
    dec r11
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + obuf]
    mov rcx, rdi
    sub rcx, rax
    lea rsi, [rip + obuf]
    mov rdx, rcx
    mov edi, 1
    mov eax, 1
    syscall
    add r13, r8
    jmp row_loop
final_line:
    lea rdi, [rip + obuf]
    mov r9, r12

    mov r10, r9
    shr r10, 28
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 24
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 20
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 16
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 12
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 8
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    shr r10, 4
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov r10, r9
    and r10, 15
    mov sil, byte ptr [r15 + r10]
    mov byte ptr [rdi], sil
    inc rdi

    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + obuf]
    mov rcx, rdi
    sub rcx, rax
    lea rsi, [rip + obuf]
    mov rdx, rcx
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
