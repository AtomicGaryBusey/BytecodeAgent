

.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.section .bss
inbuf:
.space 65536
inbuf_end:
outbuf:
.space 512

.section .text
.globl _start
_start:
    lea r15, [rip + hex_digits]
    xor r13, r13
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r13
    lea rdx, [rip + inbuf_end]
    sub rdx, rsi
    test rdx, rdx
    jz read_done
    mov eax, 0
    xor edi, edi
    syscall
    cmp rax, 1
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rbx, [rip + outbuf]
    xor r12, r12
row_loop:
    mov rax, r13
    sub rax, r12
    test rax, rax
    jz row_done
    cmp rax, 16
    jle n_set
    mov rax, 16
n_set:
    mov r14, rax
    lea r10, [rip + inbuf]
    add r10, r12
    xor rcx, rcx
    mov rax, r12
    shr eax, 28
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 24
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 20
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 16
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 12
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 8
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    shr eax, 4
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r12
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    xor r8, r8
hex_loop:
    cmp r8, r14
    jb hex_byte
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    jmp hex_after
hex_byte:
    movzx eax, byte ptr [r10 + r8]
    mov edx, eax
    shr edx, 4
    mov al, byte ptr [r15 + rdx]
    mov byte ptr [rbx + rcx], al
    inc rcx
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
hex_after:
    cmp r8, 7
    jne hex_next
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
hex_next:
    inc r8
    cmp r8, 16
    jb hex_loop
    mov al, 32
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 124
    mov byte ptr [rbx + rcx], al
    inc rcx
    xor r8, r8
ascii_loop:
    cmp r8, r14
    jae ascii_done
    movzx eax, byte ptr [r10 + r8]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, 46
ascii_store:
    mov byte ptr [rbx + rcx], al
    inc rcx
    inc r8
    jmp ascii_loop
ascii_done:
    mov al, 124
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 10
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rsi, rbx
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    add r12, r14
    jmp row_loop
row_done:
    lea rbx, [rip + outbuf]
    xor rcx, rcx
    mov rax, r13
    shr eax, 28
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 24
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 20
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 16
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 12
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 8
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    shr eax, 4
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rax, r13
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov al, 10
    mov byte ptr [rbx + rcx], al
    inc rcx
    mov rsi, rbx
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
