

.text
.globl _start
_start:
    xor r12, r12
    lea r14, [rip + input]
read_loop:
    cmp r12, 65536
    je read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r14 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    js read_done
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r15, [rip + line]
    lea r8, [rip + hex_table]
    xor r10, r10
row_loop:
    cmp r10, r12
    jge rows_done
    mov r9, r12
    sub r9, r10
    cmp r9, 16
    jle build_row
    mov r9, 16
build_row:
    xor r13, r13
    mov rax, r10
    call print_hex8
    mov al, ' '
    call put_byte
    mov al, ' '
    call put_byte
    xor rcx, rcx
slot_loop:
    cmp rcx, r9
    jae slot_missing
    movzx eax, byte ptr [r14 + r10 + rcx]
    mov r11, rax
    shr eax, 4
    mov al, byte ptr [r8 + rax]
    call put_byte
    mov rax, r11
    and eax, 0xf
    mov al, byte ptr [r8 + rax]
    call put_byte
    mov al, ' '
    call put_byte
    jmp slot_done
slot_missing:
    mov al, ' '
    call put_byte
    mov al, ' '
    call put_byte
    mov al, ' '
    call put_byte
slot_done:
    cmp rcx, 7
    jne slot_next
    mov al, ' '
    call put_byte
slot_next:
    inc rcx
    cmp rcx, 16
    jl slot_loop
    mov al, ' '
    call put_byte
    mov al, '|'
    call put_byte
    xor rcx, rcx
ascii_loop:
    cmp rcx, r9
    jae ascii_done
    movzx eax, byte ptr [r14 + r10 + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7f
    jae ascii_dot
    call put_byte
    jmp ascii_next
ascii_dot:
    mov al, '.'
    call put_byte
ascii_next:
    inc rcx
    cmp rc
