

.equ BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 128

.section .rodata
.balign 4
ten_dword:
    .long 10

.section .bss
.balign 8
io_buf:
    .zero BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE
count_array:
    .zero 24

.section .text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r12, r12

read_again:
    xor edi, edi
    lea rsi, [rip + io_buf]
    mov rdx, BUF_SIZE
    mov rax, 0
    syscall
    test rax, rax
    jz done_read
    mov rcx, rax

proc_byte:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je newline
    cmp al, 9
    je ws_word_end
    cmp al, 11
    je ws_word_end
    cmp al, 12
    je ws_word_end
    cmp al, 13
    je ws_word_end
    cmp al, 32
    je ws_word_end
    test r12, r12
    jne inc_byte
    inc r9
    mov r12b, 1
    jmp inc_byte

newline:
    inc r8
    jmp ws_word_end

ws_word_end:
    test r12, r12
    jz inc_byte
    mov r12b, 0
    jmp inc_byte

inc_byte:
    inc r10
    inc rsi
    dec rcx
    jnz proc_byte
    jmp read_again

done_read:
    lea r12, [rip + count_array]
    mov qword ptr [r12], r8
    mov qword ptr [r12 + 8], r9
    mov qword ptr [r12 + 16], r10
    lea r13, [rip + out_buf]
    mov r15d, OUT_BUF_SIZE
    dec r15d
    mov byte ptr [r13 + r15], 10
    mov r14d, 2

out_loop:
    mov rcx, qword ptr [r12 + r14*8]
    test rcx, rcx
    jz zero_count

digit_loop:
    mov eax, ecx
    xor edx, edx
    div dword ptr [rip + ten_dword]
    lea r11d, [rdx + 48]
    dec r15d
    mov byte ptr [r13 + r15], r11b
    mov rcx, rax
    test rcx, rcx
    jnz digit_loop
    jmp out_after_count

zero_count:
    dec r15d
    mov byte ptr [r13 + r15], 48

out_after_count:
    test r14, r14
    jz write_out
    dec r15d
    mov byte ptr [r13 + r15], 32
    dec r14
    jmp out_loop

write_out:
    mov rdx, OUT_BUF_SIZE
    sub rdx, r15
    lea rsi, [r13 + r15]
    mov edi, 1
    mov rax, 1
    syscall
    xor edi, edi
    mov rax, 60
    syscall
