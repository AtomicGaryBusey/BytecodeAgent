

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN_FD, 0
.equ STDOUT_FD, 1
.equ READ_MAX, 65536
.equ C_SPACE, 32
.equ C_NEWLINE, 10
.equ C_TAB, 9
.equ C_VTAB, 11
.equ C_FF, 12
.equ C_CR, 13
.equ C_ZERO, 48
.equ TEN, 10

.text
.globl _start

_start:
    adrp x23, whitespace
    add x23, x23, :lo12:whitespace
    adrp x24, inbuf
    add x24, x24, :lo12:inbuf

    mov w19, #0
    mov w20, #0
    mov w21, #0
    mov w22, #0

    mov w25, #1
    strb w25, [x23, #C_TAB]
    strb w25, [x23, #C_NEWLINE]
    strb w25, [x23, #C_VTAB]
    strb w25, [x23, #C_FF]
    strb w25, [x23, #C_CR]
    strb w25, [x23, #C_SPACE]

    movz x2, #1, lsl #16

read_loop:
    mov x8, #SYS_READ
    mov x0, #STDIN_FD
    mov x1, x24
    svc #0
    cbz x0, output_ready

    add x21, x21, x0
    mov x3, x0
    mov x1, x24

process_loop:
    ldrb w4, [x1]
    add x1, x1, #1
    ldrb w5, [x23, x4]
    cbz w5, process_nonwhite

    cmp w4, #C_NEWLINE
    cset w6, eq
    add w19, w19, w6
    mov w22, #0
    b process_end

process_nonwhite:
    cmp w22, #0
    cset w6, eq
    add w20, w20, w6
    mov w22, #1

process_end:
    subs x3, x3, #1
    b.ne process_loop
    b read_loop

output_ready:
    mov w0, w19
    bl print_uint
    mov w0, #C_SPACE
    bl print_char
    mov w0, w20
    bl print_uint
    mov w0, #C_SPACE
    bl print_char
    mov w0, w21
    bl print_uint
    mov w0, #C_NEWLINE
    bl print_char

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

print_uint:
    adrp x14, numbuf
    add x14, x14, :lo12:numbuf
    cbz w0, pu_zero

    mov x9, xzr
    mov w10, #TEN
    mov w11, #C_ZERO

pu_digit:
    udiv w12, w0, w10
    msub w13, w12, w10, w0
    add w13, w13, w11
    strb w13, [x14, x9]
    add x9, x9, #1
    mov w0, w12
    cbnz w0, pu_digit

    mov x1, x14
    mov x10, x9
    subs x9, x9, #1
    add x1, x1, x9

pu_write:
    ldrb w13, [x1]
    mov x8, #SYS_WRITE
    mov x0, #STDOUT_FD
    mov x2, #1
    svc #0
    subs x1, x1, #1
    subs x10, x10, #1
    cbnz x10, pu_write
    ret

pu_zero:
    mov w13, #C_ZERO
    strb w13, [x14]
    mov x8, #SYS_WRITE
    mov x0, #STDOUT_FD
    mov x1, x14
    mov x2, #1
    svc #0
    ret

print_char:
    adrp x14, charbuf
    add x14, x14, :lo12:charbuf
    strb w0, [x14]
    mov x8, #SYS_WRITE
    mov x0, #STDOUT_FD
    mov x2, #1
    svc #0
    ret

.bss
.balign 16
whitespace:
    .zero 256
inbuf:
    .zero READ_MAX
numbuf:
    .zero 16
charbuf:
    .zero 1
