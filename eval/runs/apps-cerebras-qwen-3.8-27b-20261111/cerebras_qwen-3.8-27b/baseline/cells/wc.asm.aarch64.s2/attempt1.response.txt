

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN_FD, 0
.equ STDOUT_FD, 1
.equ READ_MAX, 65536
.equ C_SPACE, 32
.equ C_NEWLINE, 10
.equ C_ZERO, 48
.equ TEN, 10

.text
.globl _start

_start:
    adrp x23, whitespace
    add x23, x23, :lo12:whitespace
    adrp x24, inbuf
    add x24, x24, :lo12:inbuf

    mov w19, #0
    mov w20, #0
    mov w21, #0
    mov w22, #0

    mov w25, #1
    strb w25, [x23, #9]
    strb w25, [x23, #10]
    strb w25, [x23, #11]
    strb w25, [x23, #12]
    strb w25, [x23, #13]
    strb w25, [x23, #32]

    movz x2, #1, lsl #16

read_loop:
    mov x8, #SYS_READ
    mov x0, #STDIN_FD
    mov x1, x24
    svc #0
    cbz x0, output_ready

    add x21, x21, x0
    mov x3, x0
    mov x1, x24

process_loop:
    ldrb w4, [x1]
    add x1, x1, #1
    ldrb w5, [x23, x4]
    cbz w5, process_nonwhite

    cmp w4, #C_NEWLINE
    cset w6, eq
    add w19, w19, w6
    mov w22, #0
    b process_end

process_nonwhite:
    cmp w22, #0
    cset w6, eq
    add w20, w20, w6
    mov w22, #1

process_end:
    subs x3, x3, #1
    b.ne process_loop
    b read_loop

output_ready:
    mov w0, w19
    bl print_uint
    mov w0, #C_SPACE
    bl print_char
    mov w0, w20
    bl print_uint
    mov w0, #C_SPACE
    bl print_char
    mov w0, w21
    bl print_uint
    mov w0, #C_NEWLINE
    bl print_char

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

print_uint:
    cbz w0, pu_zero
    mov x3, xzr
    mov w4, #TEN
    mov w5, #C_ZERO
pu_digit:
    udiv w6, w0, w4
    msub w7, w6, w4, w0
    add w7, w7, w5
    strb w7, [numbuf, x3]
    add x3, x3, #1
    mov w0, w6
    cbnz w0, pu_digit

    adrp x9, numbuf
    add x9, x9, :lo12:numbuf
    mov x4, x3
    subs x3, x3, #1
    add x9, x9, x3
    b pu_write

pu_zero:
    adrp x9, numbuf
    add x9, x9, :lo12:numbuf
    mov w7, #C_ZERO
    strb w7, [x9]
    mov x4, #1

pu_write:
    ldrb w7, [x9]
    mov x8, #SYS_WRITE
    mov x0, #STDOUT_FD
    mov x1, x9
    mov x2, #1
    svc #0
    subs x4, x4, #1
    subs x9, x9, #1
    cbnz x4, pu_write
    ret

print_char:
    adrp x1, charbuf
    add x1, x1, :lo12:charbuf
    strb w0, [x1]
    mov x8, #SYS_WRITE
    mov x0, #STDOUT_FD
    mov x2, #1
    svc #0
    ret

.bss
.balign 16
whitespace:
    .zero 256
inbuf:
    .zero READ_MAX
numbuf:
    .zero 16
charbuf:
    .zero 1
