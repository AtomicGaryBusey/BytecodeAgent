

.globl _start
.equ read_len, 4096

.bss
.balign 16
inbuf:
    .space read_len
.balign 16
outbuf:
    .space 64
.balign 16
tmp_digits:
    .space 16

.text
_start:
    mov x20, xzr
    mov w21, #0
    mov w22, #0
    mov w23, #0
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
read_loop:
    mov x1, x19
    mov x0, xzr
    mov w2, #read_len
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le finish
    mov x24, x19
    mov w25, w0
char_loop:
    ldrb w26, [x24], #1
    add w22, w22, #1
    cmp w26, #10
    b.eq char_newline
    cmp w26, #9
    b.eq char_ws
    cmp w26, #32
    b.eq char_ws
    cmp w26, #11
    b.eq char_ws
    cmp w26, #12
    b.eq char_ws
    cmp w26, #13
    b.eq char_ws
    cbz w23, word_start
    sub w25, w25, #1
    b.ne char_loop
char_newline:
    add w20, w20, #1
    mov w23, #0
    sub w25, w25, #1
    b.ne char_loop
char_ws:
    mov w23, #0
    sub w25, w25, #1
    b.ne char_loop
word_start:
    add w21, w21, #1
    mov w23, #1
    sub w25, w25, #1
    b.ne char_loop
finish:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x20
    bl toa
    mov w0, #32
    strb w0, [x1], #1
    mov x0, x21
    bl toa
    mov w0, #32
    strb w0, [x1], #1
    mov x0, x22
    bl toa
    mov w0, #10
    strb w0, [x1], #1
    adrp x3, outbuf
    add x3, x3, :lo12:outbuf
    sub x2, x1, x3
    mov x1, x3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, xzr
    mov x8, #93
    svc #0
toa:
    adrp x8, tmp_digits
    add x8, x8, :lo12:tmp_digits
    mov x2, x8
    mov w3, w0
    cmp w3, #0
    b.eq toa_zero
    mov w7, #10
toa_digit_loop:
    udiv w5, w3, w7
    msub w6, w5, w7, w3
    add w6, w6, #48
    strb w6, [x2], #1
    mov w3, w5
    cbnz w3, toa_digit_loop
toa_copy_loop:
    sub x2, x2, #1
    ldrb w6, [x2]
    strb w6, [x1], #1
    cmp x2, x8
    b.ne toa_copy_loop
    ret
toa_zero:
    mov w6, #48
    strb w6, [x1], #1
    ret
