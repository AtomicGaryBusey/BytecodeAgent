

.globl _start
.equ BUF_LEN, 1024
.equ OUT_LEN, 64
.equ FRAME, 0x500
.text
.balign 16
_start:
    sub sp, sp, #FRAME
    add x19, sp, #0
    add x24, x19, #BUF_LEN
    add x25, x24, #OUT_LEN
    movz x20, #0
    movz x21, #0
    movz x22, #0
    movz x23, #0
read_loop:
    movz x0, #0
    add x1, x19, #0
    movz w2, #BUF_LEN
    movz x8, #63
    svc #0
    cmp x0, #0
    b.le finish
    add x27, x19, #0
    mov x26, x0
char_loop:
    ldrb w28, [x27], #1
    add w22, w22, #1
    cmp w28, #10
    b.eq char_newline
    cmp w28, #9
    b.eq char_ws
    cmp w28, #32
    b.eq char_ws
    cmp w28, #11
    b.eq char_ws
    cmp w28, #12
    b.eq char_ws
    cmp w28, #13
    b.eq char_ws
    cbz w23, char_word_start
    b char_next
char_newline:
    add w20, w20, #1
    movz w23, #0
    b char_next
char_ws:
    movz w23, #0
    b char_next
char_word_start:
    add w21, w21, #1
    movz w23, #1
char_next:
    sub x26, x26, #1
    cbnz x26, char_loop
    b read_loop
finish:
    mov x0, x20
    add x1, x24, #0
    bl to_dec
    movz w0, #32
    strb w0, [x1], #1
    mov x0, x21
    bl to_dec
    movz w0, #32
    strb w0, [x1], #1
    mov x0, x22
    bl to_dec
    movz w0, #10
    strb w0, [x1], #1
    sub x2, x1, x24
    movz x0, #1
    movz x8, #64
    svc #0
    movz x0, #0
    movz x8, #93
    svc #0
to_dec:
    add x8, x25, #0
    mov w3, w0
    cmp w3, #0
    b.eq to_dec_zero
    add x2, x8, #0
    movz w7, #10
to_dec_digit:
    udiv w5, w3, w7
    msub w6, w5, w7, w3
    add w6, w6, #48
    strb w6, [x2], #1
    mov w3, w5
    cbnz w3, to_dec_digit
to_dec_copy:
    sub x2, x2, #1
    ldrb w6, [x2]
    strb w6, [x1], #1
    cmp x2, x8
    b.ne to_dec_copy
    ret
to_dec_zero:
    movz w6, #48
    strb w6, [x1], #1
    ret
