

.equ MAXIN, 49152
.equ LINELEN, 76
.equ EQ, 61

.section .text
.globl _start
_start:
    xor r12, r12
read_loop:
    cmp r12, MAXIN
    je read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, MAXIN
    sub edx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit0
    xor r14, r14
    xor r15, r15

encode_loop:
    cmp r14, r12
    jae encode_done
    lea rbx, [rip + inbuf]
    add rbx, r14

    movzx eax, byte ptr [rbx]
    mov r8, rax
    shr eax, 2
    movzx eax, byte ptr [b64tab + rax]
    call emit_char

    lea rcx, [r14 + 1]
    cmp rcx, r12
    je encode_one

    movzx eax, byte ptr [rbx + 1]
    mov r9, rax
    mov eax, r9d
    shr eax, 4
    mov ebx, r8d
    and ebx, 3
    shl ebx, 4
    add eax, ebx
    movzx eax, byte ptr [b64tab + rax]
    call emit_char

    lea rcx, [r14 + 2]
    cmp rcx, r12
    je encode_two

    movzx eax, byte ptr [rbx + 2]
    mov r10, rax
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov ebx, r10d
    shr ebx, 6
    add eax, ebx
    movzx eax, byte ptr [b64tab + rax]
    call emit_char

    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [b64tab + rax]
    call emit_char

    add r14, 3
    jmp encode_loop

encode_one:
    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [b64tab + rax]
    call emit_char
    mov eax, EQ
    call emit_char
    mov eax, EQ
    call emit_char
    add r14, 1
    jmp encode_loop

encode_two:
    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [b64tab + rax]
    call emit_char
    mov eax, EQ
    call emit_char
    add r14, 2
    jmp encode_loop

encode_done:
    test r15, r15
    jz exit0
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov edx, r15
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

exit0:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov r13d, eax
    cmp r15, LINELEN
    jne emit_store
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov edx, LINELEN
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    xor r15, r15

emit_store:
    mov eax, r13d
    mov byte ptr [linebuf + r15], al
    inc r15
    ret

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline:
    .byte 10

.bss
linebuf:
    .zero LINELEN
inbuf:
    .zero MAXIN
