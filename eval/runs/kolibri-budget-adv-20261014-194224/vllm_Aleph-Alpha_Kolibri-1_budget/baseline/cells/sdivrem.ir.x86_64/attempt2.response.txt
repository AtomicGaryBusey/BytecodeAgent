

export function l $sdivrem(l %a, l %b, l %rem)
@entry:
    ceql %cmp1, %b, 0
    jne @notzero
    subll %q, %a, %a
    storel %a, %rem
    jmp @return
@notzero:
    ceql %cmp2, %b, -1
    jne @normal
    ceql %cmp3, %a, -0x8000000000000000
    jne @normal
    movl %q, %a
    subll %r, %a, %a
    storel %r, %rem
    jmp @return
@normal:
    divl %q, %a, %b
    reml %r, %a, %b
    storel %r, %rem
    jmp @return
@return:
    ret
