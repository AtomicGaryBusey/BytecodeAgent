

export function l $count_desc(l a, l n) {
  @entry:
    mov l %count, 0
    mov l %i, 0
  @loop:
    addl %i1, %i, 1
    csltl %cmp_gt, %n, %i1
    jnz %cmp_gt, @done
    ceql %cmp_eq, %i1, %n
    jnz %cmp_eq, @done
    addl %addr1, a, %i
    addl %addr2, a, %i1
    loadl %val1, %addr1
    loadl %val2, %addr2
    csltl %cmp2, %val2, %val1
    addl %count, %count, %cmp2
    mov l %i, %i1
    jmp @loop
  @done:
    retl %count
}
