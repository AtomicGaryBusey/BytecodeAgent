

export function l $count_ugt(l a, l n, l t)
@start:
  mov l 0 %i
  mov l 0 %c
  mov l a %ptr
  jmp @check
@check:
  csltl %i n -> %lt
  jnz %lt @body
  ret %c
@body:
  loadl %ptr %val
  cgtu %val t -> %gt
  jnz %gt @inc
  jmp @inc
@inc:
  addl 8 %ptr %ptr
  addl 1 %i %i
  addl 1 %c %c
  jmp @check
