

export function l $count_ugt(l %a, l %n, l %t)
@start:
  mov l 0 %i
  mov l 0 %c
  mov l %a %ptr
  jmp @check
@check:
  csltu %i %n -> %ltn
  jnz %ltn @body
  ret %c
@body:
  loadl %ptr %val
  cgtu %val %t -> %gt
  jnz %gt @inc_c
  jmp @inc_ptr
@inc_c:
  addl 1 %c %c
@inc_ptr:
  addl 8 %ptr %ptr
  addl 1 %i %i
  jmp @check
