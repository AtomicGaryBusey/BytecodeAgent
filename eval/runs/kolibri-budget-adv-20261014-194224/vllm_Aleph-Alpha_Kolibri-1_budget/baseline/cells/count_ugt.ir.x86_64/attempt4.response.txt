

export function l $count_ugt(l %a, l %n, l %t)
{
  {
    @check:
      movl 0 %i
      movl 0 %c
      movl %a %ptr
      csltu %i %n %ltn
      jnz %ltn @body
      retl %c
  }
  {
    @body:
      loadl %ptr %val
      cgtu %val %t %gt
      jnz %gt @inc_c
      jmp @inc_ptr
  }
  {
    @inc_c:
      addl 1 %c %c
      jmp @inc_ptr
  }
  {
    @inc_ptr:
      addl 8 %ptr %ptr
      addl 1 %i %i
      jmp @check
  }
}
