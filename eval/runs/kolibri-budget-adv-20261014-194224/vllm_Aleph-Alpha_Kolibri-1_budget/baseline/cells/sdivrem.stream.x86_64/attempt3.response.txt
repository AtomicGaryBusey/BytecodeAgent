

sdivrem:
push rdx
test rsi, rsi
jne .check_minus_one
mov rax, 0
pop r8
mov qword ptr [r8], rdi
ret
.check_minus_one:
xor r8, r8
dec r8
cmp rsi, r8
jne .do_division
test rdi, rdi
jge .do_division
mov rax, rdi
sub rax, 1
and rax, rdi
test rax, rax
je .overflow
jmp .do_division
.overflow:
mov rax, rdi
pop r8
mov qword ptr [r8], 0
ret
.do_division:
mov rax, rdi
cdq
idiv rsi
pop r8
mov qword ptr [r8], rdx
ret
