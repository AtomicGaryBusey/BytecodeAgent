

export function l $bmin_empty(l %a, l %n)
{
  @entry
    xorl %zero, %zero
    csltl %zero, %n
    jnz @nonzero
  @zero_case
    addl $1, %zero
    shll $63, %zero
    subtrl $1, %zero
    movl %zero, %rax
    retl
  @nonzero
    xorl %zero, %zero
    loadl %a, %min
    addl $8, %a
    subtrl $1, %n
    jmp @loop
  @loop
    xorl %zero, %zero
    csltl %zero, %n
    jnz @loop_body
    jmp @done
  @loop_body
    loadl %a, %cur
    movl %cur, %cmp
    csltl %cmp, %min
    jnz @update_min
    jmp @next
  @update_min
    movl %cur, %min
  @next
    addl $8, %a
    subtrl $1, %n
    jmp @loop
  @done
    movl %min, %rax
    retl
}
