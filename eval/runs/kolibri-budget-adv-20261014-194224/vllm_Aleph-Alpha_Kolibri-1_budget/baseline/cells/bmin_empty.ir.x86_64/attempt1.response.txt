

export function l $bmin_empty(l a, l n)
@entry:
    xorl %zero, %zero
    ceql %n, %zero
    jnz @zero_case
    loadl %a, %min
    addl $8, %a
    subtrl $1, %n
    csltl %zero, %n
    jnz @loop
    jmp @end
@loop:
    loadl %a, %temp
    csltl %temp, %min
    jnz @update
    jmp @next
@update:
    movl %temp, %min
@next:
    addl $8, %a
    subtrl $1, %n
    csltl %zero, %n
    jnz @loop
    jmp @end
@zero_case:
    xorl %zero, %zero
    addl $1, %zero
    shll $63, %zero
    subtrl $1, %zero
    movl %zero, %rax
    retl
@end:
    movl %min, %rax
    retl
