

export function l $bmin_empty(l %a, l %n)
{
  @entry:
    xorl %zero, %zero
    csltl %zero, %n
    jnz @nonzero
  @zero_case:
    addl $1, %zero
    shll $63, %zero
    subtrl $1, %zero
    movl %zero, %rax
    retl
  @nonzero:
    loadl %a, %min
    addl $8, %a
    subtrl $1, %n
    csltl %zero, %n
    jnz @loop
    jmp @done
  @loop:
    loadl %a, %cur
    csltl %cur, %min
    jnz @update_min
    jmp @next
  @update_min:
    movl %cur, %min
  @next:
    addl $8, %a
    subtrl $1, %n
    csltl %zero, %n
    jnz @loop
    jmp @done
  @done:
    movl %min, %rax
    retl
}
