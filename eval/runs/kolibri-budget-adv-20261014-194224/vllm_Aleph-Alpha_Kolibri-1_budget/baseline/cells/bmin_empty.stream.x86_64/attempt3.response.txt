

bmin_empty:
mov rax, 0x7fffffffffffffff
test rsi, rsi
jz .Ldone
mov rdx, qword ptr [rdi]
cmp rdx, rax
jge .Lafter_first
mov rax, rdx
.Lafter_first:
lea r8, [rdi + 8]
mov rcx, rsi
dec rcx
.Lloop:
test rcx, rcx
jz .Ldone
mov rdx, qword ptr [r8]
cmp rdx, rax
jge .Lnext
mov rax, rdx
.Lnext:
add r8, 8
dec rcx
jmp .Lloop
.Ldone:
ret
