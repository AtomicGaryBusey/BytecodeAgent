

map_into:
push rbx
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
mov r15, 0
test r13, r13
jz .Ldone
.Lloop:
mov rdi, qword ptr [r12]
call r14
mov qword ptr [rbx], rax
cmp rax, qword ptr [r12]
jge .Lskip_inc
inc r15
.Lskip_inc:
add rbx, 8
add r12, 8
dec r13
test r13, r13
jnz .Lloop
.Ldone:
mov rax, r15
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
