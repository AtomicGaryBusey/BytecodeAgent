

export function l $stats_f(l %a, l %n, l %f, l %out)
{
@start:
  %ptr =l loadl %a
  %y =l call %f(l %ptr)
  %min =l %y
  %max =l %y
  %sum =l %y
  %count =l andl %y, l 1
  %ptr =l addl %ptr, l 8
  %n =l subl %n, l 1
  j @loop
@loop:
  %eq =l ceql %n, l 0
  jnz %eq, @end
  %y =l call %f(l %ptr)
  %less_min =l csltl %y, %min
  jnz %less_min, @min_upd
  j @max_check
@min_upd:
  %min =l %y
  j @max_check
@max_check:
  %less_max =l csltl %max, %y
  jnz %less_max, @max_upd
  j @update_sum
@max_upd:
  %max =l %y
  j @update_sum
@update_sum:
  %sum =l addl %sum, %y
  %odd =l andl %y, l 1
  %count =l addl %count, %odd
  %ptr =l addl %ptr, l 8
  %n =l subl %n, l 1
  j @loop
@end:
  %p0 =l addl %out, l 0
  %p1 =l addl %out, l 8
  %p2 =l addl %out, l 16
  storel %min, %p0
  storel %max, %p1
  storel %sum, %p2
  %count
}
