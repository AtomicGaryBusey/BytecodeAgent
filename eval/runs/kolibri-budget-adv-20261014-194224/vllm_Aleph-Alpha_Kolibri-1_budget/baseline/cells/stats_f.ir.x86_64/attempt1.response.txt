

export function l $stats_f(l %a, l %n, l %f, l %out)
@start:
  %ptr = l %a
  %first =l loadl %ptr
  %y =l call %f(l %first)
  %min =l %y
  %max =l %y
  %sum =l %y
  %odd =l andl %y, l 1
  %count =l %odd
  storel %min, %out
  %out1 =l addl %out, l 8
  storel %max, %out1
  %out2 =l addl %out, l 16
  storel %sum, %out2
  %count
