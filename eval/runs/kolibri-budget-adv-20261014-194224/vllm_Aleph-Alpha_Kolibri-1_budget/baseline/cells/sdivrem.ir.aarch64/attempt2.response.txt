

export function l $sdivrem(int64_t a, int64_t b, int64_t *rem) {
    %eq_zero = ceql %b, 0
    jnz %eq_zero, divide
    %q = 0
    %r = %a
    storel %r, %rem
    ret %q
divide:
    %eq_min = ceql %a, -9223372036854775808
    %eq_neg1 = ceql %b, -1
    %check = andl %eq_min, %eq_neg1
    jnz %check, check
    %q = divl %a, %b
    %r = reml %a, %b
    storel %r, %rem
    ret %q
check:
    %q = -9223372036854775808
    %r = 0
    storel %r, %rem
    ret %q
}
