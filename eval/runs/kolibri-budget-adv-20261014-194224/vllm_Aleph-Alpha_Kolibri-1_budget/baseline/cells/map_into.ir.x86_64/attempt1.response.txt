

export function l $map_into(l dst, l src, l n, l f)
local l %i
local l %count
local l %src_ptr
local l %dst_ptr
local l %src_val
@entry:
  %i =l subl %n, %n
  %count =l subl %n, %n
  %src_ptr =l %src
  %dst_ptr =l %dst
  jmp @check
@check:
  %less =w csltl %i, %n
  jnz @loop_body
  jmp @done
@loop_body:
  %src_val =l loadl %src_ptr
  %result =l call %f(l %src_val)
  storel %result, %dst_ptr
  %less =w csltl %result, %src_val
  jnz @inc_count
  jmp @advance
@inc_count:
  %count =l addl %count, 1
  jmp @advance
@advance:
  %src_ptr =l addl %src_ptr, 8
  %dst_ptr =l addl %dst_ptr, 8
  %i =l addl %i, 1
  jmp @check
@done:
  retl %count
