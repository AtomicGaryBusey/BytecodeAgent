

export function l $map_into(l p1, l p2, l p3, l p4)
local l %i
local l %count
local l %src_ptr
local l %dst_ptr
local l %src_val
local l %result
local w %less
@entry:
  %i =l subl %p3, %p3
  %count =l subl %p3, %p3
  %src_ptr =l %p2
  %dst_ptr =l %p1
  jmp @check
@check:
  %less =w csltl %i, %p3
  jnz @loop_body
  jmp @done
@loop_body:
  %src_val =l loadl %src_ptr
  %result =l call %p4(l %src_val)
  storel %result, %dst_ptr
  %less =w csltl %result, %src_val
  jnz @inc_count
  jmp @advance
@inc_count:
  %count =l addl %count, 1
  jmp @advance
@advance:
  %src_ptr =l addl %src_ptr, 8
  %dst_ptr =l addl %dst_ptr, 8
  %i =l addl %i, 1
  jmp @check
@done:
  retl %count
