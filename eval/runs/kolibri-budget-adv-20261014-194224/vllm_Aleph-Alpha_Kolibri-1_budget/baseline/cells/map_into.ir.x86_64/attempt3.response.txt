

export function l $map_into(l a, l b, l c, l d)
local l %i
local l %count
local l %src_ptr
local l %dst_ptr
local l %src_val
local l %result
local w %less
@entry:
  %i =l subl %c, %c
  %count =l subl %c, %c
  %src_ptr =l %b
  %dst_ptr =l %a
  jmp @loop_check
@loop_check:
  %less =w csltl %i, %c
  jnz @loop_body
  jmp @done
@loop_body:
  %src_val =l loadl %src_ptr
  %result =l call %d(l %src_val)
  storel %result, %dst_ptr
  %less =w csltl %result, %src_val
  jnz @inc_count
  jmp @advance
@inc_count:
  %count =l addl %count, 1
  jmp @advance
@advance:
  %src_ptr =l addl %src_ptr, 8
  %dst_ptr =l addl %dst_ptr, 8
  %i =l addl %i, 1
  jmp @loop_check
@done:
  retl %count
