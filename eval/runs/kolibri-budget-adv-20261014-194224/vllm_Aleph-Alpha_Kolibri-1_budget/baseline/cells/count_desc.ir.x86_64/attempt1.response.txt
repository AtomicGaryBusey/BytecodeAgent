

export function l $count_desc(l a, l n) -> l
@start:
    %i = 0
    %result = 0
    %cmp = csltl n 2
    jnz %cmp @end
@loop:
    %next_idx = addl %i 1
    %cmp2 = csltl %next_idx n
    jnz %cmp2 @end
    %curr = loadl a[%i]
    %next_val = loadl a[%next_idx]
    %greater = csltl %next_val %curr
    jnz %greater @inc_count
    jmp @inc_i
@inc_count:
    %result = addl %result 1
    jmp @inc_i
@inc_i:
    %i = addl %i 1
    jmp @loop
@end:
    %result
