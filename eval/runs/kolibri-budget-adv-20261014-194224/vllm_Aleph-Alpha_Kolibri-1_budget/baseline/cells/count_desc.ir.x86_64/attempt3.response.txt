

export function $count_desc(l %a, l %n) -> l
@start:
    %i = l 0
    %result = l 0
    %cmp = csltl %n l 2
    jnz %cmp @end
@loop:
    %next_idx = addl %i l 1
    %cmp2 = csltl %next_idx %n
    jnz %cmp2 @end
    %curr = loadl %a[%i]
    %next_val = loadl %a[%next_idx]
    %greater = csltl %next_val %curr
    jnz %greater @inc_count
    jmp @inc_i
@inc_count:
    %result = addl %result l 1
    jmp @inc_i
@inc_i:
    %i = addl %i l 1
    jmp @loop
@end:
    retl %result
