

export function l $count_desc(l %a, l %n)
@start:
    %i = 0
    %result = 0
    %one = 1
    jmp @loop
@loop:
    %next_idx = addl %i %one
    %cmp = csltl %next_idx %n
    jnz %cmp @body
    jmp @end
@body:
    %curr = loadl %a[%i]
    %next_val = loadl %a[%next_idx]
    %greater = csltl %next_val %curr
    jnz %greater @inc_count
    jmp @inc_i
@inc_count:
    %result = addl %result %one
    jmp @inc_i
@inc_i:
    %i = addl %i %one
    jmp @loop
@end:
    retl %result
