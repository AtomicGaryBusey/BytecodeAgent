

export function l $stats_f(l $a, l $n, l $f, l $out)
  %ptr = $a;
  %val = loadl %ptr;
  %y = call %f(l %val);
  %min = %y;
  %max = %y;
  %sum = %y;
  %cnt = l 0;
  %odd = bandl %y, l 1;
  jnz @inc_cnt;
  jmp @store;
@inc_cnt:
  %cnt = l 1;
@store:
  %out0 = %out;
  %out1 = addl %out, l 8;
  %out2 = addl %out, l 16;
  storel %min, %out0;
  storel %max, %out1;
  storel %sum, %out2;
  retl %cnt;
