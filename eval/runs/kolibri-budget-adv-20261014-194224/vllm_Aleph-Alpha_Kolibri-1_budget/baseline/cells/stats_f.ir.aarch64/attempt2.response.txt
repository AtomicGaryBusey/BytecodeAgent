

export function l $stats_f(l a, l n, l f, l out)
  %ptr = a;
  %i = l 0;
  %val = loadl %ptr;
  %y = call %f(l %val);
  %min = %y;
  %max = %y;
  %sum = %y;
  %cnt = l 0;
  %odd = bandl %y, l 1;
  jnz @inc_cnt_first;
  jmp @after_first;
@inc_cnt_first:
  %cnt = l 1;
@after_first:
  %ptr = addl %ptr, l 8;
  %i = addl %i, l 1;
  %cmp = csltl %i, %n;
  jnz @loop;
  jmp @store;
@loop:
  %val = loadl %ptr;
  %y = call %f(l %val);
  %lt_min = csltl %y, %min;
  jnz @min_less;
  jmp @min_done;
@min_less:
  %min = %y;
@min_done:
  %gt_max = csltl %max, %y;
  jnz @max_greater;
  jmp @max_done;
@max_greater:
  %max = %y;
@max_done:
  %sum = addl %sum, %y;
  %odd = bandl %y, l 1;
  jnz @cnt_inc;
  jmp @after_cnt;
@cnt_inc:
  %cnt = addl %cnt, l 1;
@after_cnt:
  %ptr = addl %ptr, l 8;
  %i = addl %i, l 1;
  %cmp = csltl %i, %n;
  jnz @loop;
  jmp @store;
@store:
  %out0 = out;
  %out1 = addl out, l 8;
  %out2 = addl out, l 16;
  storel %min, %out0;
  storel %max, %out1;
  storel %sum, %out2;
  retl %cnt;
