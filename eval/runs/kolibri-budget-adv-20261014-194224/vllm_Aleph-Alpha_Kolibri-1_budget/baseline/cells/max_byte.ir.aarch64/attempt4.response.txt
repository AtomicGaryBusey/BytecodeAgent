

export function l $max_byte(l %p, l %n) {
@start
%zero = l 0
%one = l 1
%max = %zero
%i = %zero
%lt = csltl %i %n
jnz %lt @body
jmp @exit
@body
%addr = addl %p %i
%byte =l loadub %addr
%lt_rev = csltl %max %byte
jnz %lt_rev @update_max
jmp @incr
@update_max
%max = %byte
@incr
%i = addl %i %one
jmp @start
@exit
retl %max
}
