

export function l $max_byte(l %p, l %n) {
    %zero = l #0
    %one = l #1
    %max = %zero
    %i = %zero
@cond:
    %lt = csltl %i %n
    jnz @body
    jmp @exit
@body:
    %addr = addl %p %i
    %byte = l loadub %addr
    %lt_rev = csltl %max %byte
    jnz @update_max
@skip_update:
    jmp @incr
@update_max:
    %max = %byte
@incr:
    %i = addl %i %one
    jmp @cond
@exit:
    retl %max
}
