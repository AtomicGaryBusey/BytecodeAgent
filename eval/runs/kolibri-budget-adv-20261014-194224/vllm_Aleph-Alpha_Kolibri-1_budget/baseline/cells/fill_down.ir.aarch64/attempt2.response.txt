

export function l $fill_down(l %arg0, l %arg1, l %arg2) {
    @fill_down:
        %a = %arg0
        %n = %arg1
        %start = %arg2
        %zero = 0
        %one = 1
        %eight = 8
        %counter = %n
        %val = %start
        %ptr = %a
        %cmp = csltl %zero, %counter
        jnz %cmp, @loop
        retl %n
    @loop:
        storel %val, %ptr
        %val = subll %val, %one
        %ptr = addl %ptr, %eight
        %counter = subll %counter, %one
        %cmp = csltl %zero, %counter
        jnz %cmp, @loop
        retl %n
}
