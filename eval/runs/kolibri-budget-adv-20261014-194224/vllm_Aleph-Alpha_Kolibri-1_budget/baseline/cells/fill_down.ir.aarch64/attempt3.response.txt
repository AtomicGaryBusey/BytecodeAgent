

export function l $fill_down(l %arg0, l %arg1, l %arg2) {
    @fill_down
        %a = %arg0
        %n = %arg1
        %start = %arg2
        %counter = %n
        %val = %start
        %ptr = %a
        %cmp = csltl #0, %counter
        jnz %cmp, @loop
        retl %n
    @loop
        storel %val, %ptr
        %val = subll %val, #1
        %ptr = addl %ptr, #8
        %counter = subll %counter, #1
        %cmp = csltl #0, %counter
        jnz %cmp, @loop
        retl %n
}
