

export function l $count_ugt(l %a, l %n, l %t)
{
    @start
    {
        %count = 0
        %p = %a
        %zero = 0
        %cmp = cultl %zero %n
        jnz @loop
        jmp @done
    }
    @loop
    {
        %elem = loadl [%p]
        %cmp2 = cultl %t %elem
        jnz @count_inc
        jmp @decr
    }
    @count_inc
    {
        %count = addl %count 1
        jmp @decr
    }
    @decr
    {
        %n = subtl %n 1
        %p = addl %p 8
        jmp @start
    }
    @done
    {
        ret %count
    }
}
