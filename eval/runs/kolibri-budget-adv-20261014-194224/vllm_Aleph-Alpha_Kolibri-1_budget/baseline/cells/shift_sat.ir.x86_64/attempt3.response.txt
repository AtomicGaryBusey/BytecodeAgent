

function i64 $shift_sat(i64 $x, i64 $n) {
@entry:
  %lt64 = cslti64 $n, 64i64
  jnz %lt64 @less64
  %left = 0i64
  %neg = cslti64 $x, 0i64
  jnz %neg @neg_right
  %right = 0i64
  jmp @done
@less64:
  %left = shlli64 $x, $n
  %right = sarlli64 $x, $n
  jmp @done
@neg_right:
  %right = -1i64
@done:
  %sum = addi64 %left, %right
  reti64 %sum
}
