

export function l $shift_sat(l $x, l $n) {
@entry:
  %lt64 = csltl $n, 64l
  jnz %lt64 @less64
  %left = 0l
  %neg = csltl $x, 0l
  jnz %neg @neg_right
  %right = 0l
  jmp @done
@less64:
  %left = shlli64 $x, $n
  %right = sarlli64 $x, $n
  jmp @done
@neg_right:
  %right = -1l
@done:
  %sum = addi64 %left, %right
  reti64 %sum
}
