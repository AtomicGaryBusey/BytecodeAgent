base64_encode:
	mov r11, rdx            # save start of out
	xor r10d, r10d          # pad count (0 = main phase)
.Lmain:
	cmp rsi, 3
	jb .Ltail
	movzx eax, byte ptr [rdi]
	shl eax, 8
	mov al, byte ptr [rdi + 1]
	shl eax, 8
	mov al, byte ptr [rdi + 2]
	add rdi, 3
	sub rsi, 3
	mov ecx, 4              # emit 4 chars
.Lchars:
	mov r8d, eax
	shr r8d, 18
	and r8d, 63             # v = top 6 bits
	shl eax, 6
	lea r9d, [r8 + 65]      # 'A' + v
	cmp r8d, 26
	jb .Lstore
	lea r9d, [r8 + 71]      # 'a' + v - 26
	cmp r8d, 52
	jb .Lstore
	lea r9d, [r8 - 4]       # '0' + v - 52
	cmp r8d, 62
	jb .Lstore
	mov r9d, 43             # '+'
	je .Lstore
	mov r9d, 47             # '/'
.Lstore:
	mov byte ptr [rdx], r9b
	inc rdx
	dec ecx
	jnz .Lchars
	test r10, r10
	jz .Lmain
.Lpad:
	mov byte ptr [rdx], 61  # '='
	inc rdx
	dec r10
	jnz .Lpad
	jmp .Ldone
.Ltail:
	test rsi, rsi
	jz .Ldone
	movzx eax, byte ptr [rdi]
	shl eax, 16
	cmp rsi, 2
	jb .Lone
	movzx ecx, byte ptr [rdi + 1]
	shl ecx, 8
	or eax, ecx
	mov ecx, 3              # 3 chars + 1 pad
	mov r10d, 1
	jmp .Lchars
.Lone:
	mov ecx, 2              # 2 chars + 2 pads
	mov r10d, 2
	jmp .Lchars
.Ldone:
	mov rax, rdx
	sub rax, r11            # chars written
	ret
