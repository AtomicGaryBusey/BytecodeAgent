binary_search:
	xor ecx, ecx                      # lo = 0
	mov rax, rsi                      # hi = n (half-open)
.Lloop:
	cmp rcx, rax
	jge .Lnotfound
	lea r8, [rcx + rax]
	shr r8, 1                         # mid = (lo + hi) / 2 (indices <= 200, no overflow)
	mov r9, qword ptr [rdi + r8*8]
	cmp r9, rdx
	je .Lfound
	jl .Lgoright
	mov rax, r8                       # hi = mid
	jmp .Lloop
.Lgoright:
	lea rcx, [r8 + 1]                 # lo = mid + 1
	jmp .Lloop
.Lfound:
	mov rax, r8
	ret
.Lnotfound:
	mov rax, -1
	ret
