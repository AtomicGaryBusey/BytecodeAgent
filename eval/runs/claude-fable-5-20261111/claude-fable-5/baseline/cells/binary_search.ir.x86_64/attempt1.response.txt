export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l sub %n, 1
@loop
	%cont =w cslel %lo, %hi
	jnz %cont, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @goright, @goleft
@goright
	%lo =l add %mid, 1
	jmp @loop
@goleft
	%hi =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
