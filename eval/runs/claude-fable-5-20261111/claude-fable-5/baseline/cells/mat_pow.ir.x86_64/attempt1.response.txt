export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%pc =l alloc8 8
	%ps =l alloc8 8
	%nn =l mul %n, %n
	%nnb =l mul %nn, 8
	%prod =l add %tmp, %nnb
	storel 0, %pi
@cpcond
	%i =l loadl %pi
	%cc =w csltl %i, %nn
	jnz %cc, @cpbody, @zinit
@cpbody
	%off =l mul %i, 8
	%src =l add %a, %off
	%v =l loadl %src
	%dst =l add %tmp, %off
	storel %v, %dst
	%i2 =l add %i, 1
	storel %i2, %pi
	jmp @cpcond
@zinit
	storel 0, %pi
@zcond
	%zi =l loadl %pi
	%zc =w csltl %zi, %nn
	jnz %zc, @zbody, @dinit
@zbody
	%zoff =l mul %zi, 8
	%zp =l add %out, %zoff
	storel 0, %zp
	%zi2 =l add %zi, 1
	storel %zi2, %pi
	jmp @zcond
@dinit
	storel 0, %pi
@dcond
	%di =l loadl %pi
	%dc =w csltl %di, %n
	jnz %dc, @dbody, @powinit
@dbody
	%n1 =l add %n, 1
	%didx =l mul %di, %n1
	%doff =l mul %didx, 8
	%dp =l add %out, %doff
	storel 1, %dp
	%di2 =l add %di, 1
	storel %di2, %pi
	jmp @dcond
@powinit
	storel 0, %pc
@powcond
	%cnt =l loadl %pc
	%pcc =w csltl %cnt, %k
	jnz %pcc, @mulinit, @cleaninit
@mulinit
	storel 0, %pi
@micond
	%mi =l loadl %pi
	%mic =w csltl %mi, %n
	jnz %mic, @mjinit, @cpprodinit
@mjinit
	storel 0, %pj
@mjcond
	%mj =l loadl %pj
	%mjc =w csltl %mj, %n
	jnz %mjc, @mtinit, @miinc
@mtinit
	storel 0, %pt
	storel 0, %ps
@mtcond
	%mt =l loadl %pt
	%mtc =w csltl %mt, %n
	jnz %mtc, @mtbody, @mtdone
@mtbody
	%mia =l loadl %pi
	%mja =l loadl %pj
	%row =l mul %mia, %n
	%xidx =l add %row, %mt
	%xoff =l mul %xidx, 8
	%xp =l add %out, %xoff
	%xv =l loadl %xp
	%trow =l mul %mt, %n
	%yidx =l add %trow, %mja
	%yoff =l mul %yidx, 8
	%yp =l add %tmp, %yoff
	%yv =l loadl %yp
	%pr =l mul %xv, %yv
	%s =l loadl %ps
	%s2 =l add %s, %pr
	storel %s2, %ps
	%mt2 =l add %mt, 1
	storel %mt2, %pt
	jmp @mtcond
@mtdone
	%fi =l loadl %pi
	%fj =l loadl %pj
	%frow =l mul %fi, %n
	%fidx =l add %frow, %fj
	%foff =l mul %fidx, 8
	%fp =l add %prod, %foff
	%fs =l loadl %ps
	storel %fs, %fp
	%fj2 =l add %fj, 1
	storel %fj2, %pj
	jmp @mjcond
@miinc
	%ii =l loadl %pi
	%ii2 =l add %ii, 1
	storel %ii2, %pi
	jmp @micond
@cpprodinit
	storel 0, %pi
@cp2cond
	%ci =l loadl %pi
	%cic =w csltl %ci, %nn
	jnz %cic, @cp2body, @powinc
@cp2body
	%coff =l mul %ci, 8
	%csrc =l add %prod, %coff
	%cv =l loadl %csrc
	%cdst =l add %out, %coff
	storel %cv, %cdst
	%ci2 =l add %ci, 1
	storel %ci2, %pi
	jmp @cp2cond
@powinc
	%cnt2 =l loadl %pc
	%cnt3 =l add %cnt2, 1
	storel %cnt3, %pc
	jmp @powcond
@cleaninit
	storel 0, %pi
	%nn2 =l mul %nn, 2
@clcond
	%li =l loadl %pi
	%lc =w csltl %li, %nn2
	jnz %lc, @clbody, @done
@clbody
	%loff =l mul %li, 8
	%lp =l add %tmp, %loff
	storel 0, %lp
	%li2 =l add %li, 1
	storel %li2, %pi
	jmp @clcond
@done
	ret 0
}
