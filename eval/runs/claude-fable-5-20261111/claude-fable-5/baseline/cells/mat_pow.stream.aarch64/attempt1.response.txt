mat_pow:
    // x0=a, x1=n, x2=k, x3=out, x4=tmp
    mul x5, x1, x1              // nn = n*n
    // out = zero
    mov x6, xzr
.Lzero_out:
    str xzr, [x3, x6, lsl #3]
    add x6, x6, #1
    cmp x6, x5
    b.lt .Lzero_out
    // out diagonal = 1
    mov x6, xzr                 // i
    mov x7, #1
.Ldiag:
    madd x9, x6, x1, x6         // i*n+i
    str x7, [x3, x9, lsl #3]
    add x6, x6, #1
    cmp x6, x1
    b.lt .Ldiag
.Lkloop:
    cbz x2, .Lfinish
    // tmp = out * a
    mov x6, xzr                 // i
.Li:
    mov x7, xzr                 // j
.Lj:
    mov x9, xzr                 // acc
    mov x10, xzr                // t
    mul x11, x6, x1             // i*n
.Lt:
    add x12, x11, x10           // i*n+t
    ldr x12, [x3, x12, lsl #3]  // out[i][t]
    madd x13, x10, x1, x7       // t*n+j
    ldr x13, [x0, x13, lsl #3]  // a[t][j]
    madd x9, x12, x13, x9
    add x10, x10, #1
    cmp x10, x1
    b.lt .Lt
    add x12, x11, x7            // i*n+j
    str x9, [x4, x12, lsl #3]
    add x7, x7, #1
    cmp x7, x1
    b.lt .Lj
    add x6, x6, #1
    cmp x6, x1
    b.lt .Li
    // out = tmp
    mov x6, xzr
.Lcopy:
    ldr x9, [x4, x6, lsl #3]
    str x9, [x3, x6, lsl #3]
    add x6, x6, #1
    cmp x6, x5
    b.lt .Lcopy
    sub x2, x2, #1
    b .Lkloop
.Lfinish:
    // zero all 2*nn elements of tmp
    lsl x6, x5, #1
    mov x7, xzr
.Lztmp:
    str xzr, [x4, x7, lsl #3]
    add x7, x7, #1
    cmp x7, x6
    b.lt .Lztmp
    mov x0, #0
    ret
