export function l $fib(l %n) {
@start
	jnz %n, @loop, @ret0
@loop
	%a =l phi @start 0, @body %b
	%b =l phi @start 1, @body %c
	%i =l phi @start %n, @body %i2
	%c =l add %a, %b
	%i2 =l sub %i, 1
	%cond =w csgtl %i2, 0
	jnz %cond, @body, @done
@body
	jmp @loop
@done
	jmp @end
@ret0
	jmp @end
@end
	%r =l phi @done %b, @ret0 0
	ret %r
}
