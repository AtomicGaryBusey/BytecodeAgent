export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp =w cultl %dst, %src
	jnz %cmp, @fwd, @bwd
@fwd
	%i =l copy 0
@floop
	%c =w csltl %i, %n
	jnz %c, @fbody, @done
@fbody
	%off =l mul %i, 8
	%s =l add %src, %off
	%d =l add %dst, %off
	%v =l loadl %s
	storel %v, %d
	%i =l add %i, 1
	jmp @floop
@bwd
	%j =l copy %n
@bloop
	%c2 =w csgtl %j, 0
	jnz %c2, @bbody, @done
@bbody
	%j =l sub %j, 1
	%off2 =l mul %j, 8
	%s2 =l add %src, %off2
	%d2 =l add %dst, %off2
	%v2 =l loadl %s2
	storel %v2, %d2
	jmp @bloop
@done
	ret 0
}
