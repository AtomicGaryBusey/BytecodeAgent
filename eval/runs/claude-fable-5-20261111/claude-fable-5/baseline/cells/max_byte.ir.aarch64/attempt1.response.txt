export function l $max_byte(l %p, l %n) {
@start
	%max0 =l copy 0
	%i0 =l copy 0
@loop
	%max =l phi @start %max0, @next %max2
	%i =l phi @start %i0, @next %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csltl %max, %b
	jnz %gt, @upd, @next
@upd
	jmp @next
@next
	%max2 =l phi @body %max, @upd %b
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %max
}
