qsort_pairs:
    cmp x1, #2
    b.lt .Lret0
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
.Lloop:
    mov x19, x0                 // base of current part
    mov x20, x1                 // count
    sub x21, x1, #1
    add x21, x0, x21, lsl #4    // pivot ptr = a + (n-1)*16
    ldp x9, x10, [x21]          // pivot key, value
    sub x22, x0, #16            // i-ptr (last <= pivot slot)
    mov x23, x0                 // j-ptr
.Lpart:
    cmp x23, x21
    b.eq .Lpdone
    ldp x11, x12, [x23]
    cmp x11, x9
    b.lt .Lle
    b.gt .Lgt
    cmp x12, x10
    b.gt .Lgt
.Lle:
    add x22, x22, #16
    ldp x13, x14, [x22]
    stp x11, x12, [x22]
    stp x13, x14, [x23]
.Lgt:
    add x23, x23, #16
    b .Lpart
.Lpdone:
    add x22, x22, #16           // pivot destination
    ldp x13, x14, [x22]
    stp x9, x10, [x22]
    stp x13, x14, [x21]
    sub x24, x22, x19
    lsr x24, x24, #4            // left part size
    mov x0, x19
    mov x1, x24
    cmp x1, #2
    b.lt .Lskipl
    bl qsort_pairs
.Lskipl:
    sub x1, x20, x24
    sub x1, x1, #1              // right part size
    add x0, x22, #16            // right part base
    cmp x1, #2
    b.ge .Lloop
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x23, x24, [sp, #48]
    ldp x29, x30, [sp], #64
.Lret0:
    mov x0, #0
    ret
