export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shiftcheck
	%c =w cugtl %bit, %n
	jnz %c, @shiftdo, @loop
@shiftdo
	%bit =l shr %bit, 2
	jmp @shiftcheck
@loop
	%cz =w ceql %bit, 0
	jnz %cz, @done, @body
@body
	%t =l add %res, %bit
	%ge =w cugel %x, %t
	jnz %ge, @yes, @no
@yes
	%x =l sub %x, %t
	%h =l shr %res, 1
	%res =l add %h, %bit
	jmp @next
@no
	%res =l shr %res, 1
	jmp @next
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
