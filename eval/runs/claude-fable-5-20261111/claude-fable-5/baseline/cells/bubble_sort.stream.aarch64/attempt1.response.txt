bubble_sort:
    cmp     x1, #2
    b.lt    .Ldone
    sub     x9, x1, #1          // i = n-1 (inner pass length)
.Louter:
    mov     x10, #0             // j = 0
    mov     x11, x0             // p = &a[0]
.Linner:
    ldr     x12, [x11]          // a[j]
    ldr     x13, [x11, #8]      // a[j+1]
    cmp     x12, x13
    b.le    .Lnoswap
    str     x13, [x11]
    str     x12, [x11, #8]
.Lnoswap:
    add     x11, x11, #8
    add     x10, x10, #1
    cmp     x10, x9
    b.lt    .Linner
    subs    x9, x9, #1
    b.ne    .Louter
.Ldone:
    mov     x0, #0
    ret
