mp_mul:
	// x0=a, x1=na, x2=b, x3=nb, x4=r
	add x5, x1, x3          // total = na+nb
	// zero r[0..total-1]
	mov x6, #0
.Lzero:
	cmp x6, x5
	b.ge .Lzdone
	str xzr, [x4, x6, lsl #3]
	add x6, x6, #1
	b .Lzero
.Lzdone:
	mov x6, #0              // i = 0
.Louter:
	cmp x6, x1
	b.ge .Lmdone
	ldr x7, [x0, x6, lsl #3] // a[i]
	mov x8, #0              // carry
	mov x9, #0              // j = 0
.Linner:
	cmp x9, x3
	b.ge .Lindone
	ldr x10, [x2, x9, lsl #3] // b[j]
	mul x11, x7, x10          // lo
	umulh x12, x7, x10        // hi
	add x13, x6, x9           // i+j
	ldr x14, [x4, x13, lsl #3]
	adds x11, x11, x14
	adc x12, x12, xzr
	adds x11, x11, x8
	adc x8, x12, xzr          // carry = hi + carries
	str x11, [x4, x13, lsl #3]
	add x9, x9, #1
	b .Linner
.Lindone:
	add x13, x6, x3
	str x8, [x4, x13, lsl #3] // r[i+nb] = carry
	add x6, x6, #1
	b .Louter
.Lmdone:
	// count significant limbs: x5 = na+nb
.Lcount:
	cbz x5, .Lret
	sub x6, x5, #1
	ldr x7, [x4, x6, lsl #3]
	cbnz x7, .Lret
	mov x5, x6
	b .Lcount
.Lret:
	mov x0, x5
	ret
