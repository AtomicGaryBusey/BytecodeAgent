export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%i =l copy 0
@idi
	%ci =w csltl %i, %n
	jnz %ci, @idib, @pow0
@idib
	%j =l copy 0
@idj
	%cj =w csltl %j, %n
	jnz %cj, @idjb, @idinc
@idjb
	%e =w ceql %i, %j
	%v =l extuw %e
	%off =l mul %i, %n
	%off =l add %off, %j
	%off =l mul %off, 8
	%p =l add %out, %off
	storel %v, %p
	%j =l add %j, 1
	jmp @idj
@idinc
	%i =l add %i, 1
	jmp @idi
@pow0
	%c =l copy 0
@powloop
	%cc =w csltl %c, %k
	jnz %cc, @mul0, @zero0
@mul0
	%i =l copy 0
@muli
	%ci2 =w csltl %i, %n
	jnz %ci2, @mulib, @copy0
@mulib
	%j =l copy 0
@mulj
	%cj2 =w csltl %j, %n
	jnz %cj2, @muljb, @muliinc
@muljb
	%s =l copy 0
	%t =l copy 0
@mult
	%ct =w csltl %t, %n
	jnz %ct, @multb, @multdone
@multb
	%o1 =l mul %i, %n
	%o1 =l add %o1, %t
	%o1 =l mul %o1, 8
	%p1 =l add %out, %o1
	%x =l loadl %p1
	%o2 =l mul %t, %n
	%o2 =l add %o2, %j
	%o2 =l mul %o2, 8
	%p2 =l add %a, %o2
	%y =l loadl %p2
	%xy =l mul %x, %y
	%s =l add %s, %xy
	%t =l add %t, 1
	jmp @mult
@multdone
	%o3 =l mul %i, %n
	%o3 =l add %o3, %j
	%o3 =l mul %o3, 8
	%p3 =l add %tmp, %o3
	storel %s, %p3
	%j =l add %j, 1
	jmp @mulj
@muliinc
	%i =l add %i, 1
	jmp @muli
@copy0
	%idx =l copy 0
@copyloop
	%cidx =w csltl %idx, %nn
	jnz %cidx, @copyb, @copydone
@copyb
	%o4 =l mul %idx, 8
	%ps =l add %tmp, %o4
	%v2 =l loadl %ps
	%pd =l add %out, %o4
	storel %v2, %pd
	%idx =l add %idx, 1
	jmp @copyloop
@copydone
	%c =l add %c, 1
	jmp @powloop
@zero0
	%nn2 =l mul %nn, 2
	%z =l copy 0
@zeroloop
	%cz =w csltl %z, %nn2
	jnz %cz, @zerob, @done
@zerob
	%o5 =l mul %z, 8
	%pz =l add %tmp, %o5
	storel 0, %pz
	%z =l add %z, 1
	jmp @zeroloop
@done
	ret 0
}
