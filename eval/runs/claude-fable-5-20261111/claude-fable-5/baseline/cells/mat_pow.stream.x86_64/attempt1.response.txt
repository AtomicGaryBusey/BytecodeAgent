mat_pow:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	# rdi=a, rsi=n, rdx=k, rcx=out, r8=tmp
	mov r10, rdx              # k
	mov r11, rsi
	imul r11, rsi             # n*n
	# out = identity
	xor eax, eax
.Lzero_out:
	mov qword ptr [rcx + rax*8], 0
	inc rax
	cmp rax, r11
	jb .Lzero_out
	lea rbx, [rsi + 1]        # n+1 (diagonal stride)
	xor eax, eax
.Ldiag:
	mov qword ptr [rcx + rax*8], 1
	add rax, rbx
	cmp rax, r11
	jb .Ldiag
.Lpow_loop:
	test r10, r10
	jz .Lfinish
	# tmp[0..nn) = out * a
	xor r12d, r12d            # i
.Lrow:
	mov rbx, r12
	imul rbx, rsi             # i*n
	lea rbp, [rcx + rbx*8]    # &out[i*n]
	xor r13d, r13d            # j
.Lcol:
	xor eax, eax              # sum
	xor r14d, r14d            # t
	lea r15, [rdi + r13*8]    # &a[0*n + j]
.Ldot:
	mov rdx, qword ptr [rbp + r14*8]
	imul rdx, qword ptr [r15]
	add rax, rdx
	lea r15, [r15 + rsi*8]
	inc r14
	cmp r14, rsi
	jb .Ldot
	lea rdx, [rbx + r13]
	mov qword ptr [r8 + rdx*8], rax
	inc r13
	cmp r13, rsi
	jb .Lcol
	inc r12
	cmp r12, rsi
	jb .Lrow
	# out = tmp
	xor eax, eax
.Lcopy:
	mov rdx, qword ptr [r8 + rax*8]
	mov qword ptr [rcx + rax*8], rdx
	inc rax
	cmp rax, r11
	jb .Lcopy
	dec r10
	jmp .Lpow_loop
.Lfinish:
	# zero all 2*n*n tmp elements
	lea rdx, [r11 + r11]
	xor eax, eax
.Lzero_tmp:
	mov qword ptr [r8 + rax*8], 0
	inc rax
	cmp rax, rdx
	jb .Lzero_tmp
	xor eax, eax
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
