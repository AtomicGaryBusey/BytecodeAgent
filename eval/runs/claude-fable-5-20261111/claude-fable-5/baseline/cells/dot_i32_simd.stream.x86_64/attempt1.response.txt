dot_i32_simd:
	pxor xmm4, xmm4                       # 64-bit accumulator (2 lanes)
	xor r8, r8                            # index i = 0
	mov rcx, rdx
	and rcx, -4                           # n rounded down to multiple of 4
	test rcx, rcx
	jz .Lreduce
.Lvloop:
	movdqu xmm0, xmmword ptr [rdi + r8*4] # a[i..i+3]
	movdqu xmm1, xmmword ptr [rsi + r8*4] # b[i..i+3]
	movdqa xmm2, xmm0
	pmuldq xmm2, xmm1                     # signed 64-bit products of lanes 0 and 2
	paddq xmm4, xmm2
	pshufd xmm0, xmm0, 0xF5               # move lanes 1,3 into even positions
	pshufd xmm1, xmm1, 0xF5
	pmuldq xmm0, xmm1                     # products of lanes 1 and 3
	paddq xmm4, xmm0
	add r8, 4
	cmp r8, rcx
	jb .Lvloop
.Lreduce:
	movdqa xmm5, xmm4
	psrldq xmm5, 8
	paddq xmm4, xmm5
	movq rax, xmm4                        # horizontal sum of accumulator
	cmp r8, rdx
	jae .Ldone
.Lsloop:
	movsxd r9, dword ptr [rdi + r8*4]
	movsxd r10, dword ptr [rsi + r8*4]
	imul r9, r10
	add rax, r9
	add r8, 1
	cmp r8, rdx
	jb .Lsloop
.Ldone:
	ret
