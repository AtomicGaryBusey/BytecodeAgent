export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%lo =l mul %a0, %b0
	%loh =l shr %lo, 32
	%t0 =l mul %a1, %b0
	%t =l add %t0, %loh
	%tl =l and %t, 4294967295
	%th =l shr %t, 32
	%m0 =l mul %a0, %b1
	%m =l add %m0, %tl
	%mh =l shr %m, 32
	%h0 =l mul %a1, %b1
	%h1 =l add %h0, %th
	%h =l add %h1, %mh
	ret %h
}
