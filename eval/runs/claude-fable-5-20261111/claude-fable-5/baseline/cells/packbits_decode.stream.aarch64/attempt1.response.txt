packbits_decode:
	// x0=src, x1=n, x2=out, x3=cap
	mov	x4, xzr			// i = source index
	mov	x5, xzr			// o = output index
.Lloop:
	cmp	x4, x1
	b.ge	.Ldone			// no more source: done
	ldrsb	x6, [x0, x4]		// h (sign-extended header)
	add	x4, x4, #1
	cmn	x6, #128
	b.eq	.Lloop			// h == -128: no-op
	tbnz	x6, #63, .Lrepeat	// negative: repeat packet
	// literal packet: count = h + 1
	add	x7, x6, #1		// count
	add	x8, x4, x7
	cmp	x8, x1
	b.gt	.Lerr2			// not enough source bytes
	sub	x9, x3, x5		// space = cap - o
	cmp	x7, x9
	csel	x10, x7, x9, le		// m = min(count, space)
.Lcopy:
	cbz	x10, .Lcopyend
	ldrb	w11, [x0, x4]
	add	x4, x4, #1
	strb	w11, [x2, x5]
	add	x5, x5, #1
	sub	x10, x10, #1
	b	.Lcopy
.Lcopyend:
	cmp	x7, x9
	b.gt	.Lerr1			// output truncated
	b	.Lloop
.Lrepeat:
	cmp	x4, x1
	b.ge	.Lerr2			// missing repeat byte
	ldrb	w11, [x0, x4]
	add	x4, x4, #1
	mov	x7, #1
	sub	x7, x7, x6		// count = 1 - h
	sub	x9, x3, x5		// space = cap - o
	cmp	x7, x9
	csel	x10, x7, x9, le		// m = min(count, space)
.Lfill:
	cbz	x10, .Lfillend
	strb	w11, [x2, x5]
	add	x5, x5, #1
	sub	x10, x10, #1
	b	.Lfill
.Lfillend:
	cmp	x7, x9
	b.gt	.Lerr1			// output truncated
	b	.Lloop
.Ldone:
	mov	x0, x5
	ret
.Lerr1:
	mov	x0, #-1
	ret
.Lerr2:
	mov	x0, #-2
	ret
