memmove_i64:
	test rdx, rdx
	jz .Ldone
	cmp rdi, rsi
	jb .Lforward
	je .Ldone
	# backward copy
	lea rcx, [rdx - 1]
.Lback:
	mov rax, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], rax
	sub rcx, 1
	jns .Lback
	jmp .Ldone
.Lforward:
	xor ecx, ecx
.Lfwd:
	mov rax, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], rax
	add rcx, 1
	cmp rcx, rdx
	jb .Lfwd
.Ldone:
	xor eax, eax
	ret
