collatz_max:
        mov x8, x0              // n
        mov x9, #1              // k
        mov x0, #0              // max steps
.Lk_loop:
        mov x10, x9             // x = k
        mov x11, #0             // steps
.Lstep:
        cmp x10, #1
        b.eq .Ldone_k
        tbnz x10, #0, .Lodd
        lsr x10, x10, #1
        b .Lnext
.Lodd:
        add x12, x10, x10, lsl #1
        add x10, x12, #1
.Lnext:
        add x11, x11, #1
        b .Lstep
.Ldone_k:
        cmp x11, x0
        csel x0, x11, x0, gt
        add x9, x9, #1
        cmp x9, x8
        b.le .Lk_loop
        ret
