mp_mul:
    push rbx
    mov r9, rdx              # b pointer (rdx needed by mul)
    lea r10, [rsi + rcx]     # n = na + nb
    xor eax, eax
    xor r11d, r11d
.Lzero:                      # zero r[0..n-1]
    cmp r11, r10
    jge .Lzdone
    mov qword ptr [r8 + r11*8], rax
    inc r11
    jmp .Lzero
.Lzdone:
    mov r10, r8              # row pointer = r + i*8
.Louter:
    test rsi, rsi
    jz .Louterdone
    xor ebx, ebx             # carry = 0
    xor r11d, r11d           # j = 0
.Linner:
    cmp r11, rcx
    jge .Linnerdone
    mov rax, qword ptr [rdi]         # a[i]
    mul qword ptr [r9 + r11*8]       # rdx:rax = a[i]*b[j]
    add rax, rbx                     # + carry
    adc rdx, 0
    add qword ptr [r10 + r11*8], rax # r[i+j] += lo
    adc rdx, 0
    mov rbx, rdx                     # new carry
    inc r11
    jmp .Linner
.Linnerdone:
    mov qword ptr [r10 + rcx*8], rbx # r[i+nb] = carry
    add rdi, 8
    add r10, 8
    dec rsi
    jmp .Louter
.Louterdone:
    mov rax, r10             # recompute n = (r10 - r8)/8 + nb
    sub rax, r8
    shr rax, 3
    add rax, rcx
.Lscan:                      # find significant limb count
    test rax, rax
    jz .Ldone
    cmp qword ptr [r8 + rax*8 - 8], 0
    jnz .Ldone
    dec rax
    jmp .Lscan
.Ldone:
    pop rbx
    ret
