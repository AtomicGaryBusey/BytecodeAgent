lcs_len:
	push rbx
	push rbp
	# rdi=x, rsi=m, rdx=y, rcx=n, r8=row
	# zero row[0..n]
	xor r9d, r9d
.Lzero:
	mov qword ptr [r8 + r9*8], 0
	inc r9
	cmp r9, rcx
	jle .Lzero
	xor r9d, r9d                      # i = 0
.Louter:
	cmp r9, rsi
	jge .Ldone
	movzx eax, byte ptr [rdi + r9]    # al = x[i]
	xor r11d, r11d                    # diag = 0
	xor ebx, ebx                      # left = row[0] = 0
	mov r10d, 1                       # j = 1
	cmp r10, rcx
	jg .Lnext_i
.Linner:
	mov rbp, qword ptr [r8 + r10*8]   # up = row[j]
	cmp al, byte ptr [rdx + r10 - 1]  # x[i] == y[j-1] ?
	jne .Lnomatch
	lea rbx, [r11 + 1]                # new = diag + 1
	jmp .Lstore
.Lnomatch:
	cmp rbx, rbp
	cmovl rbx, rbp                    # new = max(left, up)
.Lstore:
	mov qword ptr [r8 + r10*8], rbx   # row[j] = new (also next left)
	mov r11, rbp                      # diag = up
	inc r10
	cmp r10, rcx
	jle .Linner
.Lnext_i:
	inc r9
	jmp .Louter
.Ldone:
	mov rax, qword ptr [r8 + rcx*8]   # result = row[n]
	pop rbp
	pop rbx
	ret
