export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%c1 =w csltl %i, %n
	jnz %c1, @hdr, @done
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @chk
@chk
	%neg =w csltw %h, 0
	jnz %neg, @rep, @lit
@lit
	%lenw =w add %h, 1
	%k =l extsw %lenw
	%need =l add %i, %k
	%ok =w cslel %need, %n
	jnz %ok, @litloop, @err2
@litloop
	%more =w csltl 0, %k
	jnz %more, @litbody, @loop
@litbody
	%full =w ceql %o, %cap
	jnz %full, @err1, @litcopy
@litcopy
	%sp =l add %src, %i
	%b =w loadub %sp
	%dp =l add %out, %o
	storeb %b, %dp
	%i =l add %i, 1
	%o =l add %o, 1
	%k =l sub %k, 1
	jmp @litloop
@rep
	%ok2 =w csltl %i, %n
	jnz %ok2, @repread, @err2
@repread
	%sp2 =l add %src, %i
	%b2 =w loadub %sp2
	%i =l add %i, 1
	%cntw =w sub 1, %h
	%k2 =l extsw %cntw
	jmp @reploop
@reploop
	%more2 =w csltl 0, %k2
	jnz %more2, @repbody, @loop
@repbody
	%full2 =w ceql %o, %cap
	jnz %full2, @err1, @repwrite
@repwrite
	%dp2 =l add %out, %o
	storeb %b2, %dp2
	%o =l add %o, 1
	%k2 =l sub %k2, 1
	jmp @reploop
@err2
	ret -2
@err1
	ret -1
@done
	ret %o
}
