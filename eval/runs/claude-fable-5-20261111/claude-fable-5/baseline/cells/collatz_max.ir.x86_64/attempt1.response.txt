export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@kloop
	%c =w csgtl %k, %n
	jnz %c, @done, @kbody
@kbody
	%x =l copy %k
	%steps =l copy 0
@inner
	%isone =w ceql %x, 1
	jnz %isone, @innerdone, @step
@step
	%r =l rem %x, 2
	%even =w ceql %r, 0
	jnz %even, @even, @odd
@even
	%x =l div %x, 2
	jmp @stepend
@odd
	%t =l mul %x, 3
	%x =l add %t, 1
	jmp @stepend
@stepend
	%steps =l add %steps, 1
	jmp @inner
@innerdone
	%gt =w csgtl %steps, %max
	jnz %gt, @update, @next
@update
	%max =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @kloop
@done
	ret %max
}
