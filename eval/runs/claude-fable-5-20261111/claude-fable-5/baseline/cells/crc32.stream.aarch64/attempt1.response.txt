        mov     w9, #-1                 // crc = 0xFFFFFFFF
        movz    w10, #0x8320
        movk    w10, #0xEDB8, lsl #16   // poly = 0xEDB88320
        cbz     x1, .Ldone
.Lbyte:
        ldrb    w11, [x0], #1
        eor     w9, w9, w11             // crc ^= byte
        mov     w12, #8
.Lbit:
        lsr     w13, w9, #1
        tst     w9, #1
        eor     w14, w13, w10
        csel    w9, w14, w13, ne        // crc = (crc>>1) ^ (lsb ? poly : 0)
        subs    w12, w12, #1
        b.ne    .Lbit
        subs    x1, x1, #1
        b.ne    .Lbyte
.Ldone:
        mvn     w0, w9                  // return crc ^ 0xFFFFFFFF, upper 32 bits zero
        ret
