export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%c1 =w csltl %i, %n
	jnz %c1, @body, @done
@body
	%pi =l add %src, %i
	%b =w loadub %pi
	%r =l copy 1
	jmp @runloop
@runloop
	%ir =l add %i, %r
	%c2 =w csltl %ir, %n
	jnz %c2, @run2, @runend
@run2
	%c3 =w csltl %r, 128
	jnz %c3, @run3, @runend
@run3
	%p2 =l add %src, %ir
	%b2 =w loadub %p2
	%c4 =w ceqw %b2, %b
	jnz %c4, @run4, @runend
@run4
	%r =l add %r, 1
	jmp @runloop
@runend
	%c5 =w csltl %r, 3
	jnz %c5, @literal, @repeat
@repeat
	%o2 =l add %o, 2
	%c6 =w csgtl %o2, %cap
	jnz %c6, @fail, @rep2
@rep2
	%hdr =l sub 257, %r
	%po =l add %out, %o
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %o2
	%i =l add %i, %r
	jmp @loop
@literal
	%j =l copy %i
	jmp @litloop
@litloop
	%cj =w csltl %j, %n
	jnz %cj, @lit1, @litend
@lit1
	%len =l sub %j, %i
	%cl =w csltl %len, 128
	jnz %cl, @lit2, @litend
@lit2
	%j2 =l add %j, 2
	%cjn =w csltl %j2, %n
	jnz %cjn, @lit3, @litadv
@lit3
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %pj, 1
	%bj1 =w loadub %pj1
	%ce1 =w ceqw %bj, %bj1
	jnz %ce1, @lit4, @litadv
@lit4
	%pj2 =l add %pj, 2
	%bj2 =w loadub %pj2
	%ce2 =w ceqw %bj, %bj2
	jnz %ce2, @litend, @litadv
@litadv
	%j =l add %j, 1
	jmp @litloop
@litend
	%L =l sub %j, %i
	%need =l add %L, 1
	%onew =l add %o, %need
	%cf =w csgtl %onew, %cap
	jnz %cf, @fail, @litemit
@litemit
	%hdr2 =l sub %L, 1
	%po2 =l add %out, %o
	storeb %hdr2, %po2
	%k =l copy 0
	jmp @copy
@copy
	%cc =w csltl %k, %L
	jnz %cc, @copy1, @copyend
@copy1
	%ps =l add %src, %i
	%ps2 =l add %ps, %k
	%cb =w loadub %ps2
	%pd =l add %po2, 1
	%pd2 =l add %pd, %k
	storeb %cb, %pd2
	%k =l add %k, 1
	jmp @copy
@copyend
	%o =l copy %onew
	%i =l copy %j
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
