export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%v =l loadl %a
	%y =l call %f(l %v)
	%min =l copy %y
	%max =l copy %y
	%sum =l copy %y
	%cnt =l and %y, 1
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%y =l call %f(l %v)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @accum
@setmax
	%max =l copy %y
@accum
	%sum =l add %sum, %y
	%odd =l and %y, 1
	%cnt =l add %cnt, %odd
	%i =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%p1 =l add %out, 8
	storel %max, %p1
	%p2 =l add %out, 16
	storel %sum, %p2
	ret %cnt
}
