export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
@loop
	%max =l phi @start %max0, @cont %max2
	%i =l phi @start %i0, @cont %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %max
	jnz %gt, @upd, @cont
@upd
	jmp @cont
@cont
	%max2 =l phi @body %max, @upd %v
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %max
}
