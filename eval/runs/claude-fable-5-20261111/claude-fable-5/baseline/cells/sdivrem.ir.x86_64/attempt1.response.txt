@start
	%bz =w ceql %b, 0
	jnz %bz, @bzero, @chk1
@bzero
	storel %a, %rem
	ret 0
@chk1
	%bm1 =w ceql %b, -1
	jnz %bm1, @chk2, @normal
@chk2
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	jnz %amin, @ovf, @normal

@normal

	%r =l rem %a, %b
