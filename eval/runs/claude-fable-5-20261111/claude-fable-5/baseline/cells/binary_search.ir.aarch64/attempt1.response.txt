export function l $binary_search(l %a, l %n, l %key) {
@start
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start 0, @lt %lo2, @gt %lo
	%hi =l phi @start %hi0, @lt %hi, @gt %hi2
	%done =w csgtl %lo, %hi
	jnz %done, @notfound, @body
@body
	%d =l sub %hi, %lo
	%half =l shr %d, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%below =w csltl %v, %key
	jnz %below, @lt, @gt
@lt
	%lo2 =l add %mid, 1
	jmp @loop
@gt
	%hi2 =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
