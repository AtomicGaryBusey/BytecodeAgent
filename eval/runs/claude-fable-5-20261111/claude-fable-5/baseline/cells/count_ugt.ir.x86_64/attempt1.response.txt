export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%c =l phi @start 0, @body %c1
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%c1 =l add %c, %gtl
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %c
}
