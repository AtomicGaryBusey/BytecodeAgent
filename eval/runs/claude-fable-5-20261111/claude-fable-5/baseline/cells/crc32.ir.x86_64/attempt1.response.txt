export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @bitdone %i2
	%crc =w phi @start %crc0, @bitdone %cn
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b
	%j0 =w copy 0
	jmp @bitloop
@bitloop
	%j =w phi @body %j0, @bitnext %j2
	%cc =w phi @body %cx, @bitnext %cn0
	%bit =w and %cc, 1
	%sh =w shr %cc, 1
	jnz %bit, @dox, @nox
@dox
	%cp =w xor %sh, 3988292384
	jmp @bitnext
@nox
	jmp @bitnext
@bitnext
	%cn0 =w phi @dox %cp, @nox %sh
	%j2 =w add %j, 1
	%again =w csltw %j2, 8
	jnz %again, @bitloop, @bitdone
@bitdone
	%cn =w copy %cn0
	%i2 =l add %i, 1
	jmp @loop
@end
	%rx =w xor %crc, 4294967295
	%res =l extuw %rx
	ret %res
}
