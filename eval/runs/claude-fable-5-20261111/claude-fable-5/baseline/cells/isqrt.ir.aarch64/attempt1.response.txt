export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%c =l copy 0
	%d =l copy 4611686018427387904
@dloop
	%big =w cugtl %d, %n
	jnz %big, @dshift, @loop
@dshift
	%d =l shr %d, 2
	jmp @dloop
@loop
	%done =w ceql %d, 0
	jnz %done, @end, @body
@body
	%cd =l add %c, %d
	%ge =w cugel %x, %cd
	jnz %ge, @yes, @no
@yes
	%x =l sub %x, %cd
	%ch =l shr %c, 1
	%c =l add %ch, %d
	jmp @next
@no
	%c =l shr %c, 1
	jmp @next
@next
	%d =l shr %d, 2
	jmp @loop
@end
	ret %c
}
