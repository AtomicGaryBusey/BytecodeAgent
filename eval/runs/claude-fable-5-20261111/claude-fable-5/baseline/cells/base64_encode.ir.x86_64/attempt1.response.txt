export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%rem =l sub %n, %i
	%more =w csgtl %rem, 0
	jnz %more, @group, @done
@group
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%val =l shl %b0, 16
	%m =l copy 1
	%i1 =l add %i, 1
	%h1 =w csltl %i1, %n
	jnz %h1, @get1, @have
@get1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	%t1 =l shl %b1, 8
	%val =l or %val, %t1
	%m =l copy 2
	%i2 =l add %i, 2
	%h2 =w csltl %i2, %n
	jnz %h2, @get2, @have
@get2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	%val =l or %val, %b2
	%m =l copy 3
@have
	%i =l add %i, %m
	%chars =l add %m, 1
	%j =l copy 0
@jloop
	%jd =w csltl %j, 4
	jnz %jd, @jbody, @loop
@jbody
	%real =w csltl %j, %chars
	jnz %real, @map, @pad
@map
	%sh6 =l mul %j, 6
	%sh =l sub 18, %sh6
	%vv =l shr %val, %sh
	%v =l and %vv, 63
	%lt26 =w csltl %v, 26
	jnz %lt26, @upper, @ck52
@upper
	%c =w add %v, 65
	jmp @emit
@ck52
	%lt52 =w csltl %v, 52
	jnz %lt52, @lower, @ck62
@lower
	%c =w add %v, 71
	jmp @emit
@ck62
	%lt62 =w csltl %v, 62
	jnz %lt62, @digit, @ck63
@digit
	%c =w sub %v, 4
	jmp @emit
@ck63
	%is62 =w ceql %v, 62
	jnz %is62, @plus, @slash
@plus
	%c =w copy 43
	jmp @emit
@slash
	%c =w copy 47
	jmp @emit
@pad
	%c =w copy 61
@emit
	%po =l add %out, %o
	storeb %c, %po
	%o =l add %o, 1
	%j =l add %j, 1
	jmp @jloop
@done
	ret %o
}
