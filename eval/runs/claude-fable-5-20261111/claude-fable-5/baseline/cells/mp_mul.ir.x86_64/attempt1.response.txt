export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%i0 =l copy 0
	jmp @zloop
@zloop
	%i =l phi @start %i0, @zbody %i1
	%zc =w csltl %i, %n
	jnz %zc, @zbody, @outer_init
@zbody
	%zoff =l mul %i, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%i1 =l add %i, 1
	jmp @zloop
@outer_init
	%ii0 =l copy 0
	jmp @outer
@outer
	%ii =l phi @outer_init %ii0, @outer_next %ii1
	%oc =w csltl %ii, %na
	jnz %oc, @oload, @scan_init
@oload
	%aoff =l mul %ii, 8
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	%carry0 =l copy 0
	%jj0 =l copy 0
	jmp @inner
@inner
	%jj =l phi @oload %jj0, @ibody %jj1
	%carry =l phi @oload %carry0, @ibody %carry1
	%icn =w csltl %jj, %nb
	jnz %icn, @ibody, @inner_done
@ibody
	%boff =l mul %jj, 8
	%bp =l add %b, %boff
	%bj =l loadl %bp
	%al =l and %ai, 4294967295
	%ah =l shr %ai, 32
	%bl =l and %bj, 4294967295
	%bh =l shr %bj, 32
	%p0 =l mul %al, %bl
	%p1 =l mul %al, %bh
	%p2 =l mul %ah, %bl
	%p3 =l mul %ah, %bh
	%mid =l add %p1, %p2
	%cmw =w cultl %mid, %p1
	%cml =l extuw %cmw
	%cmh =l shl %cml, 32
	%midlo =l shl %mid, 32
	%midhi0 =l shr %mid, 32
	%midhi =l add %midhi0, %cmh
	%lo =l add %p0, %midlo
	%c0w =w cultl %lo, %p0
	%c0 =l extuw %c0w
	%hi0 =l add %p3, %midhi
	%hi =l add %hi0, %c0
	%k =l add %ii, %jj
	%koff =l mul %k, 8
	%rp =l add %r, %koff
	%rv =l loadl %rp
	%t =l add %rv, %lo
	%c1w =w cultl %t, %rv
	%c1 =l extuw %c1w
	%t2 =l add %t, %carry
	%c2w =w cultl %t2, %t
	%c2 =l extuw %c2w
	storel %t2, %rp
	%ch =l add %hi, %c1
	%carry1 =l add %ch, %c2
	%jj1 =l add %jj, 1
	jmp @inner
@inner_done
	%k2 =l add %ii, %nb
	%k2off =l mul %k2, 8
	%rp2 =l add %r, %k2off
	storel %carry, %rp2
	jmp @outer_next
@outer_next
	%ii1 =l add %ii, 1
	jmp @outer
@scan_init
	%s0 =l copy %n
	jmp @scan
@scan
	%s =l phi @scan_init %s0, @sdec %s1
	%snz =w cnel %s, 0
	jnz %snz, @scheck, @done
@scheck
	%sm1 =l sub %s, 1
	%soff =l mul %sm1, 8
	%sp =l add %r, %soff
	%sv =l loadl %sp
	%svz =w ceql %sv, 0
	jnz %svz, @sdec, @done
@sdec
	%s1 =l copy %sm1
	jmp @scan
@done
	ret %s
}
