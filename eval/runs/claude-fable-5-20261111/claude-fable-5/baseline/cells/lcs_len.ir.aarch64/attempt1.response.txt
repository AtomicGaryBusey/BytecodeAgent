export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@init
	%c0 =w cslel %j, %n
	jnz %c0, @initbody, @outerinit
@initbody
	%off =l mul %j, 8
	%p =l add %row, %off
	storel 0, %p
	%j =l add %j, 1
	jmp @init
@outerinit
	%i =l copy 0
@outer
	%c1 =w csltl %i, %m
	jnz %c1, @outerbody, @done
@outerbody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%diag =l copy 0
	%jj =l copy 1
@inner
	%c2 =w cslel %jj, %n
	jnz %c2, @innerbody, @outernext
@innerbody
	%off2 =l mul %jj, 8
	%pj =l add %row, %off2
	%up =l loadl %pj
	%jm1 =l sub %jj, 1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%newv =l add %diag, 1
	jmp @store
@nomatch
	%pjm1 =l sub %pj, 8
	%left =l loadl %pjm1
	%lt =w csltl %up, %left
	jnz %lt, @useleft, @useup
@useleft
	%newv =l copy %left
	jmp @store
@useup
	%newv =l copy %up
	jmp @store
@store
	storel %newv, %pj
	%diag =l copy %up
	%jj =l add %jj, 1
	jmp @inner
@outernext
	%i =l add %i, 1
	jmp @outer
@done
	%roff =l mul %n, 8
	%rp =l add %row, %roff
	%res =l loadl %rp
	ret %res
}
