sdivrem:
	mov r8, rdx                # save rem pointer (idiv clobbers rdx)
	test rsi, rsi
	je .Lbzero
	cmp rsi, -1
	jne .Ldodiv
	movabs rax, -9223372036854775808
	cmp rdi, rax
	je .Lovf
.Ldodiv:
	mov rax, rdi
	cqo
	idiv rsi
	mov qword ptr [r8], rdx
	ret
.Lbzero:
	mov qword ptr [r8], rdi    # r = a
	xor eax, eax               # q = 0
	ret
.Lovf:
	mov qword ptr [r8], 0      # r = 0
	mov rax, rdi               # q = INT64_MIN
	ret
