memmove_i64:

	cmp	x0, x1
	b.lo	.Lfwd
	b.eq	.Ldone
	// backward copy
	mov	x9, x2

	ldr	x10, [x1, x9, lsl #3]
	str	x10, [x0, x9, lsl #3]

.Lfloop:

	str	x10, [x0, x9, lsl #3]

	cmp	x9, x2

.Ldone:
