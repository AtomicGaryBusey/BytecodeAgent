export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csgtl %n, 1
	jnz %c, @body, @done
@body
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%pp =l add %a, %poff
	%pk =l loadl %pp
	%ppv =l add %pp, 8
	%pv =l loadl %ppv
	%i =l copy 0
	%j =l copy 0
	jmp @loop
@loop
	%jc =w csltl %j, %nm1
	jnz %jc, @lbody, @after
@lbody
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jpv =l add %jp, 8
	%jv =l loadl %jpv
	%lt =w csltl %jk, %pk
	jnz %lt, @doswap, @chk
@chk
	%eq =w ceql %jk, %pk
	jnz %eq, @chk2, @next
@chk2
	%lt2 =w csltl %jv, %pv
	jnz %lt2, @doswap, @next
@doswap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ipv =l add %ip, 8
	%iv =l loadl %ipv
	storel %jk, %ip
	storel %jv, %ipv
	storel %ik, %jp
	storel %iv, %jpv
	%i =l add %i, 1
	jmp @next
@next
	%j =l add %j, 1
	jmp @loop
@after
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ipv2 =l add %ip2, 8
	%iv2 =l loadl %ipv2
	storel %pk, %ip2
	storel %pv, %ipv2
	storel %ik2, %pp
	storel %iv2, %ppv
	%r1 =l call $qsort_pairs(l %a, l %i)
	%i1 =l add %i, 1
	%roff =l mul %i1, 16
	%rp =l add %a, %roff
	%rn =l sub %n, %i1
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	jmp @done
@done
	ret 0
}
