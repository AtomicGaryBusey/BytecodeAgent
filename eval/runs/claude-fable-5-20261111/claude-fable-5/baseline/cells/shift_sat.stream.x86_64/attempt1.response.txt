shift_sat:
        cmp     rsi, 63
        ja      .Lbig
        shlx    rax, rdi, rsi          # x << n (n < 64)
        sarx    rdi, rdi, rsi          # x >> n arithmetic
        add     rax, rdi
        ret
.Lbig:
        sar     rdi, 63                # n >= 64: shl gives 0, sar gives sign mask
        mov     rax, rdi
        ret
