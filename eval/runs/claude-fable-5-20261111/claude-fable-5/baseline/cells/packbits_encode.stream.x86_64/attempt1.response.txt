packbits_encode:
	xor r8, r8                      # i = 0
	xor r9, r9                      # out position = 0
.Lmain:
	cmp r8, rsi
	jae .Ldone
	mov al, byte ptr [rdi + r8]     # byte at i
	lea r10, [r8 + 1]               # j = i + 1
.Lrun:
	cmp r10, rsi
	jae .Lrundone
	mov r11, r10
	sub r11, r8
	cmp r11, 128
	jae .Lrundone
	cmp al, byte ptr [rdi + r10]
	jne .Lrundone
	inc r10
	jmp .Lrun
.Lrundone:
	sub r10, r8                     # r = run(i)
	cmp r10, 3
	jb .Lliteral
	lea r11, [r9 + 2]               # repeat packet needs 2 bytes
	cmp r11, rcx
	ja .Lfail
	mov eax, 1
	sub eax, r10d                   # header = 257 - r (mod 256)
	mov byte ptr [rdx + r9], al
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9 + 1], al
	add r9, 2
	add r8, r10                     # i += r
	jmp .Lmain
.Lliteral:
	mov r10, r8                     # j = i
.Llit_scan:
	cmp r10, rsi                    # j < n ?
	jae .Llit_end
	mov r11, r10
	sub r11, r8
	cmp r11, 128                    # j - i < 128 ?
	jae .Llit_end
	lea r11, [r10 + 2]              # run(j) >= 3 iff j+2 < n and 3 equal bytes
	cmp r11, rsi
	jae .Llit_take
	mov al, byte ptr [rdi + r10]
	cmp al, byte ptr [rdi + r10 + 1]
	jne .Llit_take
	cmp al, byte ptr [rdi + r10 + 2]
	jne .Llit_take
	jmp .Llit_end                   # run(j) >= 3, stop literal
.Llit_take:
	inc r10
	jmp .Llit_scan
.Llit_end:
	mov r11, r10
	sub r11, r8                     # L = j - i (>= 1 here)
	lea rax, [r9 + 1]
	add rax, r11                    # need 1 + L bytes of room
	cmp rax, rcx
	ja .Lfail
	lea rax, [r11 - 1]
	mov byte ptr [rdx + r9], al     # header = L - 1
	inc r9
.Lcopy:
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9], al
	inc r8
	inc r9
	cmp r8, r10
	jb .Lcopy
	jmp .Lmain
.Lfail:
	mov rax, -1
	ret
.Ldone:
	mov rax, r9
	ret
