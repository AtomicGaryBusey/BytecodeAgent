export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%m =l sub %n, 1
@loop
	%c =w csltl %i, %m
	jnz %c, @body, @done
@body
	%p0 =l mul %i, 8
	%p1 =l add %a, %p0
	%x =l loadl %p1
	%p2 =l add %p1, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%cnt =l add %cnt, %ge
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
