export function l $mulhi_u64(l %a, l %b) {
@start
	%aL =l and %a, 4294967295
	%aH =l shr %a, 32
	%bL =l and %b, 4294967295
	%bH =l shr %b, 32
	%ll =l mul %aL, %bL
	%hl =l mul %aH, %bL
	%lh =l mul %aL, %bH
	%hh =l mul %aH, %bH
	%llh =l shr %ll, 32
	%t =l add %hl, %llh
	%tL =l and %t, 4294967295
	%tH =l shr %t, 32
	%u =l add %lh, %tL
	%uH =l shr %u, 32
	%r1 =l add %hh, %tH
	%r =l add %r1, %uH
	ret %r
}
