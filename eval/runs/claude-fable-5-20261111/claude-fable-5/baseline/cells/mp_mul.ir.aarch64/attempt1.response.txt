export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zcond
@zcond
	%zi =l phi @start 0, @zbody %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @ocond
@zbody
	%zp =l mul %zi, 8
	%zaddr =l add %r, %zp
	storel 0, %zaddr
	%zi1 =l add %zi, 1
	jmp @zcond
@ocond
	%i =l phi @zcond 0, @oend %i1
	%ic =w csltl %i, %na
	jnz %ic, @obody, @scond
@obody
	%ap =l mul %i, 8
	%aaddr =l add %a, %ap
	%av =l loadl %aaddr
	jmp @icond
@icond
	%j =l phi @obody 0, @ibody %j1
	%carry =l phi @obody 0, @ibody %carry1
	%jc =w csltl %j, %nb
	jnz %jc, @ibody, @oend
@ibody
	%bp =l mul %j, 8
	%baddr =l add %b, %bp
	%bv =l loadl %baddr
	%al =l and %av, 4294967295
	%ah =l shr %av, 32
	%bl =l and %bv, 4294967295
	%bh =l shr %bv, 32
	%ll =l mul %al, %bl
	%m1 =l mul %al, %bh
	%m2 =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%llh =l shr %ll, 32
	%m1l =l and %m1, 4294967295
	%m2l =l and %m2, 4294967295
	%t0 =l add %llh, %m1l
	%t =l add %t0, %m2l
	%lohi =l shl %t, 32
	%lll =l and %ll, 4294967295
	%lo =l or %lohi, %lll
	%m1h =l shr %m1, 32
	%m2h =l shr %m2, 32
	%th =l shr %t, 32
	%h0 =l add %hh, %m1h
	%h1 =l add %h0, %m2h
	%hi =l add %h1, %th
	%idx =l add %i, %j
	%rp =l mul %idx, 8
	%raddr =l add %r, %rp
	%rv =l loadl %raddr
	%s =l add %rv, %lo
	%c1 =w cultl %s, %lo
	%s2 =l add %s, %carry
	%c2 =w cultl %s2, %carry
	storel %s2, %raddr
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%ca =l add %hi, %c1l
	%carry1 =l add %ca, %c2l
	%j1 =l add %j, 1
	jmp @icond
@oend
	%hidx =l add %i, %nb
	%hp =l mul %hidx, 8
	%haddr =l add %r, %hp
	storel %carry, %haddr
	%i1 =l add %i, 1
	jmp @ocond
@scond
	%k =l phi @ocond %n, @sbody %k1
	%kz =w ceql %k, 0
	jnz %kz, @ret, @scheck
@scheck
	%k1 =l sub %k, 1
	%kp =l mul %k1, 8
	%kaddr =l add %r, %kp
	%kv =l loadl %kaddr
	%rz =w ceql %kv, 0
	jnz %rz, @sbody, @ret
@sbody
	jmp @scond
@ret
	ret %k
}
