bubble_sort:
	cmp rsi, 2
	jl .Ldone
	lea rcx, [rsi - 1]        # rcx = outer limit (passes)
.Louter:
	xor rdx, rdx              # rdx = inner index
	xor r8d, r8d              # swapped flag
.Linner:
	mov rax, qword ptr [rdi + rdx*8]
	mov r9, qword ptr [rdi + rdx*8 + 8]
	cmp rax, r9
	jle .Lnoswap
	mov qword ptr [rdi + rdx*8], r9
	mov qword ptr [rdi + rdx*8 + 8], rax
	mov r8d, 1
.Lnoswap:
	inc rdx
	cmp rdx, rcx
	jl .Linner
	test r8d, r8d
	je .Ldone
	dec rcx
	jnz .Louter
.Ldone:
	xor eax, eax
	ret
