export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@body
	%pi =l add %src, %i
	%b =w loadub %pi
	%j =l add %i, 1
	jmp @runloop
@runloop
	%c1 =w csltl %j, %n
	jnz %c1, @runchk1, @runend
@runchk1
	%d =l sub %j, %i
	%c2 =w csltl %d, 128
	jnz %c2, @runchk2, @runend
@runchk2
	%pj =l add %src, %j
	%bj =w loadub %pj
	%c3 =w ceqw %bj, %b
	jnz %c3, @runinc, @runend
@runinc
	%j =l add %j, 1
	jmp @runloop
@runend
	%run =l sub %j, %i
	%c4 =w csltl %run, 3
	jnz %c4, @literal, @repeat
@repeat
	%o2 =l add %o, 2
	%c5 =w csgtl %o2, %cap
	jnz %c5, @fail, @repok
@repok
	%hdr =w sub 257, %run
	%po =l add %out, %o
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %o2
	%i =l add %i, %run
	jmp @loop
@literal
	%k =l copy %i
	jmp @litscan
@litscan
	%c6 =w csltl %k, %n
	jnz %c6, @litchk1, @litend
@litchk1
	%dk =l sub %k, %i
	%c7 =w csltl %dk, 128
	jnz %c7, @litchk2, @litend
@litchk2
	%k3 =l add %k, 3
	%c8 =w cslel %k3, %n
	jnz %c8, @litchk3, @litinc
@litchk3
	%pk =l add %src, %k
	%bk =w loadub %pk
	%pk1 =l add %pk, 1
	%bk1 =w loadub %pk1
	%c9 =w ceqw %bk, %bk1
	jnz %c9, @litchk4, @litinc
@litchk4
	%pk2 =l add %pk, 2
	%bk2 =w loadub %pk2
	%c10 =w ceqw %bk, %bk2
	jnz %c10, @litend, @litinc
@litinc
	%k =l add %k, 1
	jmp @litscan
@litend
	%L =l sub %k, %i
	%need =l add %L, 1
	%onew =l add %o, %need
	%c11 =w csgtl %onew, %cap
	jnz %c11, @fail, @litok
@litok
	%hdr2 =w sub %L, 1
	%po2 =l add %out, %o
	storeb %hdr2, %po2
	%s =l copy %i
	%dptr =l add %po2, 1
	jmp @cploop
@cploop
	%c12 =w csltl %s, %k
	jnz %c12, @cpbody, @cpdone
@cpbody
	%ps =l add %src, %s
	%bs =w loadub %ps
	storeb %bs, %dptr
	%s =l add %s, 1
	%dptr =l add %dptr, 1
	jmp @cploop
@cpdone
	%o =l copy %onew
	%i =l copy %k
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
