export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%atend =w ceql %pc, %len
	jnz %atend, @done, @fetch
@fetch
	%ioff =l mul %pc, 8
	%ia =l add %code, %ioff
	%op =l loadl %ia
	%c0 =w ceql %op, 0
	jnz %c0, @done, @n1
@n1
	%c1 =w ceql %op, 1
	jnz %c1, @opush, @n2
@n2
	%c2 =w ceql %op, 2
	jnz %c2, @oadd, @n3
@n3
	%c3 =w ceql %op, 3
	jnz %c3, @osub, @n4
@n4
	%c4 =w ceql %op, 4
	jnz %c4, @omul, @n5
@n5
	%c5 =w ceql %op, 5
	jnz %c5, @odup, @n6
@n6
	%c6 =w ceql %op, 6
	jnz %c6, @oswap, @n7
@n7
	%c7 =w ceql %op, 7
	jnz %c7, @odrop, @n8
@n8
	%c8 =w ceql %op, 8
	jnz %c8, @oover, @n9
@n9
	%c9 =w ceql %op, 9
	jnz %c9, @olt, @n10
@n10
	%c10 =w ceql %op, 10
	jnz %c10, @ojz, @n11
@n11
	%c11 =w ceql %op, 11
	jnz %c11, @ojmp, @odef
@odef
	%pc =l add %pc, 1
	jmp @loop
@opush
	%poa =l add %ia, 8
	%pv =l loadl %poa
	%pdo =l mul %d, 8
	%pda =l add %stack, %pdo
	storel %pv, %pda
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@oadd
	%ad1 =l sub %d, 1
	%abo =l mul %ad1, 8
	%aba =l add %stack, %abo
	%ab =l loadl %aba
	%ad2 =l sub %d, 2
	%aao =l mul %ad2, 8
	%aaa =l add %stack, %aao
	%aa =l loadl %aaa
	%ar =l add %aa, %ab
	storel %ar, %aaa
	%d =l copy %ad1
	%pc =l add %pc, 1
	jmp @loop
@osub
	%sd1 =l sub %d, 1
	%sbo =l mul %sd1, 8
	%sba =l add %stack, %sbo
	%sb =l loadl %sba
	%sd2 =l sub %d, 2
	%sao =l mul %sd2, 8
	%saa =l add %stack, %sao
	%sa =l loadl %saa
	%sr =l sub %sa, %sb
	storel %sr, %saa
	%d =l copy %sd1
	%pc =l add %pc, 1
	jmp @loop
@omul
	%md1 =l sub %d, 1
	%mbo =l mul %md1, 8
	%mba =l add %stack, %mbo
	%mb =l loadl %mba
	%md2 =l sub %d, 2
	%mao =l mul %md2, 8
	%maa =l add %stack, %mao
	%ma =l loadl %maa
	%mr =l mul %ma, %mb
	storel %mr, %maa
	%d =l copy %md1
	%pc =l add %pc, 1
	jmp @loop
@odup
	%dd1 =l sub %d, 1
	%dto =l mul %dd1, 8
	%dta =l add %stack, %dto
	%dt =l loadl %dta
	%dno =l mul %d, 8
	%dna =l add %stack, %dno
	storel %dt, %dna
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@oswap
	%wd1 =l sub %d, 1
	%wbo =l mul %wd1, 8
	%wba =l add %stack, %wbo
	%wb =l loadl %wba
	%wd2 =l sub %d, 2
	%wao =l mul %wd2, 8
	%waa =l add %stack, %wao
	%wa =l loadl %waa
	storel %wa, %wba
	storel %wb, %waa
	%pc =l add %pc, 1
	jmp @loop
@odrop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@oover
	%vd2 =l sub %d, 2
	%vto =l mul %vd2, 8
	%vta =l add %stack, %vto
	%vt =l loadl %vta
	%vno =l mul %d, 8
	%vna =l add %stack, %vno
	storel %vt, %vna
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@olt
	%ld1 =l sub %d, 1
	%lbo =l mul %ld1, 8
	%lba =l add %stack, %lbo
	%lb =l loadl %lba
	%ld2 =l sub %d, 2
	%lao =l mul %ld2, 8
	%laa =l add %stack, %lao
	%la =l loadl %laa
	%lw =w csltl %la, %lb
	%lr =l extsw %lw
	storel %lr, %laa
	%d =l copy %ld1
	%pc =l add %pc, 1
	jmp @loop
@ojz
	%zd1 =l sub %d, 1
	%zvo =l mul %zd1, 8
	%zva =l add %stack, %zvo
	%zv =l loadl %zva
	%d =l copy %zd1
	%zz =w ceql %zv, 0
	jnz %zz, @ojztake, @ojzskip
@ojztake
	%zta =l add %ia, 8
	%pc =l loadl %zta
	jmp @loop
@ojzskip
	%pc =l add %pc, 2
	jmp @loop
@ojmp
	%jta =l add %ia, 8
	%pc =l loadl %jta
	jmp @loop
@done
	%i =l copy %d
	jmp @zl
@zl
	%zc =w csltl %i, %cap
	jnz %zc, @zb, @zend
@zb
	%zo =l mul %i, 8
	%za =l add %stack, %zo
	storel 0, %za
	%i =l add %i, 1
	jmp @zl
@zend
	%de =w ceql %d, 0
	jnz %de, @retz, @rett
@rett
	%ro =l sub %d, 1
	%ro8 =l mul %ro, 8
	%ra =l add %stack, %ro8
	%res =l loadl %ra
	ret %res
@retz
	ret 0
}
