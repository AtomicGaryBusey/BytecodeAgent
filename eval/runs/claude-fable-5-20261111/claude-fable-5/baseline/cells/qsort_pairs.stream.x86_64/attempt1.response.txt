qsort_pairs:
	cmp rsi, 2
	jl .Lsmall
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	sub rsp, 8
	mov rbx, rdi              # a
	mov rbp, rsi              # n
	lea rax, [rbp - 1]
	shl rax, 4
	mov r14, qword ptr [rbx + rax]       # pivot key (last record)
	mov r15, qword ptr [rbx + rax + 8]   # pivot value
	xor r12d, r12d            # i = store index
	xor r13d, r13d            # j = scan index
	lea rcx, [rbp - 1]
.Lploop:
	cmp r13, rcx
	jge .Lpdone
	mov rdx, r13
	shl rdx, 4
	mov r8, qword ptr [rbx + rdx]
	mov r9, qword ptr [rbx + rdx + 8]
	cmp r8, r14
	jl .Lswap
	jg .Lnoswap
	cmp r9, r15
	jge .Lnoswap
.Lswap:
	mov rax, r12
	shl rax, 4
	mov r10, qword ptr [rbx + rax]
	mov r11, qword ptr [rbx + rax + 8]
	mov qword ptr [rbx + rax], r8
	mov qword ptr [rbx + rax + 8], r9
	mov qword ptr [rbx + rdx], r10
	mov qword ptr [rbx + rdx + 8], r11
	inc r12
.Lnoswap:
	inc r13
	jmp .Lploop
.Lpdone:
	mov rax, r12              # swap record i with pivot record n-1
	shl rax, 4
	mov rdx, rcx
	shl rdx, 4
	mov r8, qword ptr [rbx + rax]
	mov r9, qword ptr [rbx + rax + 8]
	mov r10, qword ptr [rbx + rdx]
	mov r11, qword ptr [rbx + rdx + 8]
	mov qword ptr [rbx + rax], r10
	mov qword ptr [rbx + rax + 8], r11
	mov qword ptr [rbx + rdx], r8
	mov qword ptr [rbx + rdx + 8], r9
	mov rdi, rbx              # left part: a, i records
	mov rsi, r12
	call qsort_pairs
	lea rdi, [r12 + 1]        # right part: a + (i+1)*16, n-i-1 records
	shl rdi, 4
	add rdi, rbx
	mov rsi, rbp
	sub rsi, r12
	dec rsi
	call qsort_pairs
	xor eax, eax
	add rsp, 8
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
.Lsmall:
	xor eax, eax
	ret
