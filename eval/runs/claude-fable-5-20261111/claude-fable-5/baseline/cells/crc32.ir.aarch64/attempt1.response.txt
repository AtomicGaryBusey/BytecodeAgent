export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @loop
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%a =l add %p, %i
	%b =w loadub %a
	%crc =w xor %crc, %b
	%j =w copy 0
	jmp @bitloop
@bitloop
	%bit =w and %crc, 1
	%sh =w shr %crc, 1
	jnz %bit, @poly, @nopoly
@poly
	%crc =w xor %sh, 3988292384
	jmp @bitnext
@nopoly
	%crc =w copy %sh
	jmp @bitnext
@bitnext
	%j =w add %j, 1
	%c2 =w csltw %j, 8
	jnz %c2, @bitloop, @bytedone
@bytedone
	%i =l add %i, 1
	jmp @loop
@done
	%crc =w xor %crc, 4294967295
	%r =l extuw %crc
	ret %r
}
