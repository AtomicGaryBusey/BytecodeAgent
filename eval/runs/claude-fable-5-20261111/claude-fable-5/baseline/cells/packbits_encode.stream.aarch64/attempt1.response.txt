packbits_encode:
	mov x4, #0              // i = 0
	mov x5, #0              // w = bytes written
.Lloop:
	cmp x4, x1
	b.ge .Ldone
	ldrb w6, [x0, x4]       // b = src[i]
	add x7, x4, #1          // k = i+1
	add x8, x4, #128
	cmp x8, x1
	csel x8, x8, x1, lt     // limit = min(i+128, n)
.Lrun1:
	cmp x7, x8
	b.ge .Lrun1_end
	ldrb w9, [x0, x7]
	cmp w9, w6
	b.ne .Lrun1_end
	add x7, x7, #1
	b .Lrun1
.Lrun1_end:
	sub x9, x7, x4          // r = run(i)
	cmp x9, #3
	b.lt .Lliteral
	add x10, x5, #2         // repeat packet needs 2 bytes
	cmp x10, x3
	b.gt .Lfail
	mov x11, #1
	sub x11, x11, x9        // header = 1 - r (== 257 - r mod 256)
	strb w11, [x2, x5]
	add x12, x5, #1
	strb w6, [x2, x12]
	add x5, x5, #2
	add x4, x4, x9
	b .Lloop
.Lliteral:
	mov x7, x4              // j = i
	add x8, x4, #128        // j - i < 128 bound
.Llitscan:
	cmp x7, x1
	b.ge .Llit_end
	cmp x7, x8
	b.ge .Llit_end
	add x9, x7, #2
	cmp x9, x1
	b.ge .Llit_adv          // fewer than 3 bytes left: run(j) < 3
	ldrb w10, [x0, x7]
	add x11, x7, #1
	ldrb w12, [x0, x11]
	cmp w10, w12
	b.ne .Llit_adv
	ldrb w12, [x0, x9]
	cmp w10, w12
	b.eq .Llit_end          // run(j) >= 3: stop literal
.Llit_adv:
	add x7, x7, #1
	b .Llitscan
.Llit_end:
	sub x9, x7, x4          // L = j - i
	add x10, x5, #1
	add x10, x10, x9        // need 1 + L bytes
	cmp x10, x3
	b.gt .Lfail
	sub x11, x9, #1
	strb w11, [x2, x5]      // header = L - 1
	add x5, x5, #1
.Lcopy:
	cbz x9, .Lloop
	ldrb w12, [x0, x4]
	strb w12, [x2, x5]
	add x4, x4, #1
	add x5, x5, #1
	sub x9, x9, #1
	b .Lcopy
.Lfail:
	mov x0, #-1
	ret
.Ldone:
	mov x0, x5
	ret
