isqrt:
	mov	x1, #0			// r = 0
	mov	x2, #1
	lsl	x2, x2, #62		// bit = 1 << 62
.Lfind:
	cmp	x2, x0			// shrink bit until bit <= n
	b.ls	.Lloop
	lsr	x2, x2, #2
	b	.Lfind
.Lloop:
	cbz	x2, .Ldone
	add	x3, x1, x2		// t = r + bit (never overflows)
	cmp	x0, x3
	b.lo	.Lskip
	sub	x0, x0, x3		// n -= r + bit
	add	x1, x2, x1, lsr #1	// r = (r >> 1) + bit
	b	.Lnext
.Lskip:
	lsr	x1, x1, #1		// r >>= 1
.Lnext:
	lsr	x2, x2, #2		// bit >>= 2
	b	.Lloop
.Ldone:
	mov	x0, x1
	ret
