export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy 1
@kloop
	%cond =w cslel %k, %n
	jnz %cond, @body, @done
@body
	%x =l copy %k
	%steps =l copy 0
@cloop
	%isone =w ceql %x, 1
	jnz %isone, @cdone, @step
@step
	%r =l rem %x, 2
	%odd =w cnel %r, 0
	jnz %odd, @oddcase, @evencase
@oddcase
	%t =l mul %x, 3
	%x =l add %t, 1
	jmp @stepdone
@evencase
	%x =l div %x, 2
	jmp @stepdone
@stepdone
	%steps =l add %steps, 1
	jmp @cloop
@cdone
	%gt =w csgtl %steps, %best
	jnz %gt, @newbest, @next
@newbest
	%best =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @kloop
@done
	ret %best
}
