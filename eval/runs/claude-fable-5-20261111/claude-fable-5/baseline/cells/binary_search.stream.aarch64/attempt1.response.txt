binary_search:
	mov x3, #0              // lo = 0
	mov x4, x1              // hi = n
.Lloop:
	cmp x3, x4
	b.ge .Lnotfound
	add x5, x3, x4
	lsr x5, x5, #1          // mid = (lo+hi)/2 (no overflow, n<=200)
	ldr x6, [x0, x5, lsl #3]
	cmp x6, x2
	b.eq .Lfound
	b.lt .Llower
	mov x4, x5              // hi = mid
	b .Lloop
.Llower:
	add x3, x5, #1          // lo = mid+1
	b .Lloop
.Lfound:
	mov x0, x5
	ret
.Lnotfound:
	mov x0, #-1
	ret
