export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @chk
@chk
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	%bneg1 =w ceql %b, -1
	%both =w and %amin, %bneg1
	jnz %both, @overflow, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@divzero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret %a
}
