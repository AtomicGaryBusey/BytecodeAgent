export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%m =l sub %n, 1
@loop
	%c =w csltl %i, %m
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%cnt =l add %cnt, %ge
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
