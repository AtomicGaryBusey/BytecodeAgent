isqrt:
	mov rdx, rdi            # rdx = n (remaining)
	xor eax, eax            # rax = r = 0
	mov rcx, 1
	shl rcx, 62             # rcx = bit = 1 << 62
.Lfind:
	cmp rcx, rdx
	jbe .Lloop              # while bit > n: bit >>= 2
	shr rcx, 2
	jnz .Lfind
	ret                     # n == 0 -> return 0
.Lloop:
	test rcx, rcx
	jz .Ldone
	lea rsi, [rax + rcx]    # rsi = r + bit (cannot overflow: r < 2^63, bit <= 2^62)
	shr rax, 1              # r >>= 1
	cmp rdx, rsi
	jb .Lskip
	sub rdx, rsi            # n -= r + bit
	add rax, rcx            # r += bit
.Lskip:
	shr rcx, 2              # bit >>= 2
	jmp .Lloop
.Ldone:
	ret
