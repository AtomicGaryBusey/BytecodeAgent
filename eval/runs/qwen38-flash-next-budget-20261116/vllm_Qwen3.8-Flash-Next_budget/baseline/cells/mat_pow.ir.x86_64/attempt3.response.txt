

export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %zero = subl %n, %n
  %one = addl %zero, 1
  %nn = mull %n, %n
  %n8 = mull %n, 8
  %twice_nn = mull %nn, 2
  jmp @id_i_hdr
@id_i_hdr
  %i_id = phi.l @start %zero, @id_i_next %i_id_next
  %cmp_i_id = csltl %i_id, %n
  jnz %cmp_i_id, @id_i_body, @id_done
@id_i_body
  jmp @id_j_hdr
@id_j_hdr
  %j_id = phi.l @id_i_body %zero, @id_j_tail %j_id_next
  %cmp_j_id = csltl %j_id, %n
  jnz %cmp_j_id, @id_j_body, @id_i_next
@id_j_body
  %id_row_base = mull %i_id, %n
  %id_elem = addl %id_row_base, %j_id
  %id_off = mull %id_elem, 8
  %id_ptr = addl %tmp, %id_off
  %id_eq = ceql %i_id, %j_id
  jnz %id_eq, @id_one, @id_zero
@id_one
  storel %one, %id_ptr
  jmp @id_j_tail
@id_zero
  storel %zero, %id_ptr
  jmp @id_j_tail
@id_j_tail
  %j_id_next = addl %j_id, 1
  jmp @id_j_hdr
@id_i_next
  %i_id_next = addl %i_id, 1
  jmp @id_i_hdr
@id_done
  jmp @e_hdr
@e_hdr
  %e = phi.l @id_done %zero, @copy_done %e_next
  %cmp_e = csltl %e, %k
  jnz %cmp_e, @e_body, @e_done
@e_body
  jmp @mul_i_hdr
@mul_i_hdr
  %i_mul = phi.l @e_body %zero, @mul_i_next %i_mul_next
  %cmp_i_mul = csltl %i_mul, %n
  jnz %cmp_i_mul, @mul_i_body, @mul_done
@mul_i_body
  %i_row_off = mull %i_mul, %n8
  %x_row = addl %tmp, %i_row_off
  %z_row = addl %out, %i_row_off
  jmp @mul_j_hdr
@mul_j_hdr
  %j_mul = phi.l @mul_i_body %zero, @mul_j_tail %j_mul_next
  %cmp_j_mul = csltl %j_mul, %n
  jnz %cmp_j_mul, @mul_j_body, @mul_i_next
@mul_j_body
  %j_off = mull %j_mul, 8
  %mul_z_ptr = addl %z_row, %j_off
  jmp @mul_t_hdr
@mul_t_hdr
  %t = phi.l @mul_j_body %zero, @mul_t_body %t_next
  %sum = phi.l @mul_j_body %zero, @mul_t_body %sum_next
  %cmp_t = csltl %t, %n
  jnz %cmp_t, @mul_t_body, @mul_t_done
@mul_t_body
  %t_x_off = mull %t, 8
  %x_ptr = addl %x_row, %t_x_off
  %x_val = loadl %x_ptr
  %t_y_off = mull %t, %n8
  %y_row_ptr = addl %a, %t_y_off
  %y_ptr = addl %y_row_ptr, %j_off
  %y_val = loadl %y_ptr
  %prod = mull %x_val, %y_val
  %sum_next = addl %sum, %prod
  %t_next = addl %t, 1
  jmp @mul_t_hdr
@mul_t_done
  storel %sum, %mul_z_ptr
  jmp @mul_j_tail
@mul_j_tail
  %j_mul_next = addl %j_mul, 1
  jmp @mul_j_hdr
@mul_i_next
  %i_mul_next = addl %i_mul, 1
  jmp @mul_i_hdr
@mul_done
  jmp @copy_hdr
@copy_hdr
  %c = phi.l @mul_done %zero, @copy_body %c_next
  %cmp_c = csltl %c, %nn
  jnz %cmp_c, @copy_body, @copy_done
@copy_body
  %c_off = mull %c, 8
  %copy_src = addl %out, %c_off
  %copy_val = loadl %copy_src
  %copy_dst = addl %tmp, %c_off
  storel %copy_val, %copy_dst
  %c_next = addl %c, 1
  jmp @copy_hdr
@copy_done
  %e_next = addl %e, 1
  jmp @e_hdr
@e_done
  jmp @final_copy_hdr
@final_copy_hdr
  %fc = phi.l @e_done %zero, @final_copy_body %fc_next
  %cmp_fc = csltl %fc, %nn
  jnz %cmp_fc, @final_copy_body, @final_copy_done
@final_copy_body
  %fc_off = mull %fc, 8
  %final_src = addl %tmp, %fc_off
  %final_val = loadl %final_src
  %final_dst = addl %out, %fc_off
  storel %final_val, %final_dst
  %fc_next = addl %fc, 1
  jmp @final_copy_hdr
@final_copy_done
  jmp @zero_hdr
@zero_hdr
  %z = phi.l @final_copy_done %zero, @zero_body %z_next
  %cmp_z = csltl %z, %twice_nn
  jnz %cmp_z, @zero_body, @zero_done
@zero_body
  %zero_off = mull %z, 8
  %zero_ptr = addl %tmp, %zero_off
  storel %zero, %zero_ptr
  %z_next = addl %z, 1
  jmp @zero_hdr
@zero_done
  retl %zero
}
