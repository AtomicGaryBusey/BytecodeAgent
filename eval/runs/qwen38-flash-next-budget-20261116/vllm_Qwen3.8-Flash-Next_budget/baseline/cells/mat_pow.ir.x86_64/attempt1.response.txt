

export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %zero = sub.l %n, %n
  %one = add.l %zero, 1
  %nn = mul.l %n, %n
  %n8 = mul.l %n, 8
  %twice_nn = mul.l %nn, 2
  jmp @id_i_hdr
@id_i_hdr
  %i_id = phi.l %zero @start, %i_id_next @id_i_next
  %cmp_i_id = csltl %i_id, %n
  jnz %cmp_i_id, @id_i_body, @id_done
@id_i_body
  jmp @id_j_hdr
@id_j_hdr
  %j_id = phi.l %zero @id_i_body, %j_id_next @id_j_tail
  %cmp_j_id = csltl %j_id, %n
  jnz %cmp_j_id, @id_j_body, @id_i_next
@id_j_body
  %id_idx_1 = mul.l %i_id, %n
  %id_idx_2 = add.l %id_idx_1, %j_id
  %id_off = mul.l %id_idx_2, 8
  %id_ptr = add.l %tmp, %id_off
  %id_eq = ceql %i_id, %j_id
  jnz %id_eq, @id_one, @id_zero
@id_one
  storel %one, %id_ptr
  jmp @id_j_tail
@id_zero
  storel %zero, %id_ptr
  jmp @id_j_tail
@id_j_tail
  %j_id_next = add.l %j_id, 1
  jmp @id_j_hdr
@id_i_next
  %i_id_next = add.l %i_id, 1
  jmp @id_i_hdr
@id_done
  jmp @e_hdr
@e_hdr
  %e = phi.l %zero @id_done, %e_next @copy_done
  %cmp_e = csltl %e, %k
  jnz %cmp_e, @e_body, @e_done
@e_body
  jmp @mul_i_hdr
@mul_i_hdr
  %i_mul = phi.l %zero @e_body, %i_mul_next @mul_i_next
  %cmp_i_mul = csltl %i_mul, %n
  jnz %cmp_i_mul, @mul_i_body, @mul_done
@mul_i_body
  %i_row_off = mul.l %i_mul, %n8
  %x_row = add.l %tmp, %i_row_off
  %z_row = add.l %out, %i_row_off
  jmp @mul_j_hdr
@mul_j_hdr
  %j_mul = phi.l %zero @mul_i_body, %j_mul_next @mul_j_tail
  %cmp_j_mul = csltl %j_mul, %n
  jnz %cmp_j_mul, @mul_j_body, @mul_i_next
@mul_j_body
  %j_off = mul.l %j_mul, 8
  %z_ptr = add.l %z_row, %j_off
  jmp @mul_t_hdr
@mul_t_hdr
  %t = phi.l %zero @mul_j_body, %t_next @mul_t_body
  %sum = phi.l %zero @mul_j_body, %sum_next @mul_t_body
  %cmp_t = csltl %t, %n
  jnz %cmp_t, @mul_t_body, @mul_t_done
@mul_t_body
  %t_off_x = mul.l %t, 8
  %x_ptr = add.l %x_row, %t_off_x
  %x_val = loadl %x_ptr
  %t_off_y = mul.l %t, %n8
  %y_ptr_1 = add.l %a, %t_off_y
  %y_ptr_2 = add.l %y_ptr_1, %j_off
  %y_val = loadl %y_ptr_2
  %prod = mul.l %x_val, %y_val
  %sum_next = add.l %sum, %prod
  %t_next = add.l %t, 1
  jmp @mul_t_hdr
@mul_t_done
  storel %sum, %z_ptr
  jmp @mul_j_tail
@mul_j_tail
  %j_mul_next = add.l %j_mul, 1
  jmp @mul_j_hdr
@mul_i_next
  %i_mul_next = add.l %i_mul, 1
  jmp @mul_i_hdr
@mul_done
  jmp @copy_hdr
@copy_hdr
  %c = phi.l %zero @mul_done, %c_next @copy_body
  %cmp_c = csltl %c, %nn
  jnz %cmp_c, @copy_body, @copy_done
@copy_body
  %c_off = mul.l %c, 8
  %copy_src = add.l %out, %c_off
  %copy_val = loadl %copy_src
  %copy_dst = add.l %tmp, %c_off
  storel %copy_val, %copy_dst
  %c_next = add.l %c, 1
  jmp @copy_hdr
@copy_done
  %e_next = add.l %e, 1
  jmp @e_hdr
@e_done
  jmp @final_copy_hdr
@final_copy_hdr
  %fc = phi.l %zero @e_done, %fc_next @final_copy_body
  %cmp_fc = csltl %fc, %nn
  jnz %cmp_fc, @final_copy_body, @final_copy_done
@final_copy_body
  %fc_off = mul.l %fc, 8
  %final_src = add.l %tmp, %fc_off
  %final_val = loadl %final_src
  %final_dst = add.l %out, %fc_off
  storel %final_val, %final_dst
  %fc_next = add.l %fc, 1
  jmp @final_copy_hdr
@final_copy_done
  jmp @zero_hdr
@zero_hdr
  %z = phi.l %zero @final_copy_done, %z_next @zero_body
  %cmp_z = csltl %z, %twice_nn
  jnz %cmp_z, @zero_body, @zero_done
@zero_body
  %z_off = mul.l %z, 8
  %z_ptr = add.l %tmp, %z_off
  storel %zero, %z_ptr
  %z_next = add.l %z, 1
  jmp @zero_hdr
@zero_done
  ret %zero
}
