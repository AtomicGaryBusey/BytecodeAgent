

memmove_i64:
cld
test rdx, rdx
jle .Lret
cmp rdi, rsi
jae .Lback
.Lfor:
mov rcx, qword ptr [rsi]
mov qword ptr [rdi], rcx
add rsi, 8
add rdi, 8
sub rdx, 1
jne .Lfor
.Lret:
xor eax, eax
ret
.Lback:
lea rsi, [rsi + rdx*8]
lea rdi, [rdi + rdx*8]
sub rsi, 8
sub rdi, 8
.Lb:
mov rcx, qword ptr [rsi]
mov qword ptr [rdi], rcx
sub rsi, 8
sub rdi, 8
sub rdx, 1
jne .Lb
jmp .Lret
