

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
%src_end =l add %src, %n
%out_end =l add %out, %cap
%p =l add %src, $0
%dst =l add %out, $0
jmp @loop
@loop
%end_now =w ceql %p, %src_end
jnz %end_now, @done, @packet
@packet
%hb =w loadub %p
%is_lit =w csltw %hb, $128
jnz %is_lit, @literal, @not_lit
@not_lit
%is_nop =w ceqw %hb, $128
jnz %is_nop, @nop, @repeat
@literal
%len =w add %hb, $1
%body =l add %p, $1
%scan =l add %body, $0
%scan_left =w add %len, $0
jmp @lit_scan
@lit_scan
%scan_done =w ceqw %scan_left, $0
jnz %scan_done, @lit_ok, @lit_scan_chk
@lit_scan_chk
%scan_at_end =w ceql %scan, %src_end
jnz %scan_at_end, @err2, @lit_scan_step
@lit_scan_step
%scan =l add %scan, $1
%scan_left =w sub %scan_left, $1
jmp @lit_scan
@lit_ok
%copy_src =l add %body, $0
%copy_left =w add %len, $0
jmp @lit_copy
@lit_copy
%copy_done =w ceqw %copy_left, $0
jnz %copy_done, @lit_next, @lit_copy_chk
@lit_copy_chk
%dst_full =w ceql %dst, %out_end
jnz %dst_full, @ret_m1, @lit_copy_one
@lit_copy_one
%ch =w loadub %copy_src
storeb %ch, %dst
%copy_src =l add %copy_src, $1
%dst =l add %dst, $1
%copy_left =w sub %copy_left, $1
jmp @lit_copy
@lit_next
%p =l add %scan, $0
jmp @loop
@nop
%p =l add %p, $1
jmp @loop
@repeat
%rep_body =l add %p, $1
%rep_at_end =w ceql %rep_body, %src_end
jnz %rep_at_end, @err2, @rep_ok
@rep_ok
%rep_val =w loadub %rep_body
%rep_count =w sub $257, %hb
%rep_next =l add %p, $2
jmp @rep_copy
@rep_copy
%rep_done =w ceqw %rep_count, $0
jnz %rep_done, @rep_next_block, @rep_copy_chk
@rep_copy_chk
%dst_full2 =w ceql %dst, %out_end
jnz %dst_full2, @ret_m1, @rep_copy_one
@rep_copy_one
storeb %rep_val, %dst
%dst =l add %dst, $1
%rep_count =w sub %rep_count, $1
jmp @rep_copy
@rep_next_block
%p =l add %rep_next, $0
jmp @loop
@done
%written =l sub %dst, %out
ret %written
@ret_m1
ret $-1
@err2
ret $-2
}
