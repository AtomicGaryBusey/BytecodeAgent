

dot_i32_simd:
xor rax, rax
movsxd r10, edx
test r10, r10
jle .Ldot_done
mov r8, rdi
mov r9, rsi
pxor xmm7, xmm7
.Ldot_vec:
cmp r10, 4
jl .Ldot_tail
movdqu xmm0, xmmword ptr [r8]
movdqu xmm1, xmmword ptr [r9]
pmovsxdq xmm2, xmm0
pshufd xmm3, xmm0, 0x4e
pmovsxdq xmm3, xmm3
pmovsxdq xmm4, xmm1
pshufd xmm5, xmm1, 0x4e
pmovsxdq xmm5, xmm5
pmuldq xmm2, xmm4
pmuldq xmm3, xmm5
paddq xmm7, xmm2
paddq xmm7, xmm3
add r8, 16
add r9, 16
sub r10, 4
jmp .Ldot_vec
.Ldot_tail:
pextrq rax, xmm7, 0
pextrq r11, xmm7, 1
add rax, r11
test r10, r10
jz .Ldot_done
.Ldot_scalar:
movsxd r11, dword ptr [r8]
movsxd rcx, dword ptr [r9]
imul r11, rcx
add rax, r11
add r8, 4
add r9, 4
dec r10
jne .Ldot_scalar
.Ldot_done:
cld
ret
