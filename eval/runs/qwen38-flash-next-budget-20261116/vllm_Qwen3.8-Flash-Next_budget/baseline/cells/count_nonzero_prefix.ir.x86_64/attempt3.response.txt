

export function l $count_nonzero_prefix(l %a, l %n) {
@start
%i0 =l copy 0
%p0 =l copy %a
%zero =l copy 0
%one =l copy 1
%eight =l copy 8
jmp @loop
@loop
%p =l phi @start %p0, @inc %pinc
%i =l phi @start %i0, @inc %iinc
%c =w csltl %i %n
jnz %c @body @done
@body
%x =l loadl %p
%z =w ceql %x %zero
jnz %z @done @inc
@inc
%pinc =l add %p %eight
%iinc =l add %i %one
jmp @loop
@done
ret %i
}
