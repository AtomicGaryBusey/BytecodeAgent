

export function l $count_nonzero_prefix(l %a, l %n) {
@start
%i0 =l 0
%p0 =l add %a 0
jmp @loop
@loop
%p =l phi @start %p0, @inc %pinc
%i =l phi @start %i0, @inc %iinc
%c =w csltl %i %n
jnz %c @body @done
@body
%x =l loadl %p
%z =w ceql %x 0
jnz %z @done @inc
@inc
%pinc =l add %p 8
%iinc =l add %i 1
jmp @loop
@done
ret %i
}
