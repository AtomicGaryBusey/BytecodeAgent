

export function l $max_byte(l %p, l %n) {
@start
    %i0 =l copy 0
    %m0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @update %i_upd
    %m =l phi @start %m0, @update %m_upd
    %cont =w csltl %i, %n
    jnz %cont, @body, @end
@end
    ret %m
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csgtl %b, %m
    %i_next =l add %i, 1
    jnz %gt, @take, @keep
@take
    %b_take =l copy %b
    %i_take =l copy %i_next
    jmp @update
@keep
    %m_keep =l copy %m
    %i_keep =l copy %i_next
    jmp @update
@update
    %m_upd =l phi @take %b_take, @keep %m_keep
    %i_upd =l phi @take %i_take, @keep %i_keep
    jmp @loop
}
