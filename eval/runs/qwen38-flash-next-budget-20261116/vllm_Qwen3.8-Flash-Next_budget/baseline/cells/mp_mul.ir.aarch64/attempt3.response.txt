

export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@L0:
  %zero =l copy 0
  %cnt =l add %na, %nb
  %mask =l copy 4294967295
  %zi0 =l copy 0
  j @Lzero_cond
@Lzero_cond:
  %zi =l phi %zi0 @L0, %zi_next @Lzero_body
  %zc =w cultl %zi, %cnt
  jnz %zc, @Lzero_body, @Lzero_done
@Lzero_body:
  %zofs =l mul %zi, 8
  %zptr =l add %r, %zofs
  storel %zero, %zptr, 0
  %zi_next =l add %zi, 1
  j @Lzero_cond
@Lzero_done:
  %na0 =w ceql %na, 0
  jnz %na0, @Lret0, @Lnb_zero_check
@Lnb_zero_check:
  %nb0 =w ceql %nb, 0
  jnz %nb0, @Lret0, @Lmul_start
@Lmul_start:
  %oi0 =l copy 0
  j @Lo_cond
@Lo_cond:
  %oi =l phi %oi0 @Lmul_start, %oi_next @Lo_next
  %oc =w cultl %oi, %na
  jnz %oc, @Lo_body, @Lscan_start
@Lo_body:
  %ij0 =l copy 0
  %cr0 =l copy 0
  j @Li_cond
@Li_cond:
  %ij =l phi %ij0 @Lo_body, %ij_next @Li_body
  %cr =l phi %cr0 @Lo_body, %cr_next @Li_body
  %ic =w cultl %ij, %nb
  jnz %ic, @Li_body, @Li_done
@Li_body:
  %aofs =l mul %oi, 8
  %aptr =l add %a, %aofs
  %x =l loadl %aptr, 0
  %bofs =l mul %ij, 8
  %bptr =l add %b, %bofs
  %y =l loadl %bptr, 0
  %xh =l shr %x, 32
  %xl =l and %x, %mask
  %yh =l shr %y, 32
  %yl =l and %y, %mask
  %p0 =l mul %xl, %yl
  %p1 =l mul %xh, %yl
  %p2 =l mul %xl, %yh
  %p3 =l mul %xh, %yh
  %p12 =l add %p1, %p2
  %c12w =w cultl %p12, %p1
  %c12l =l extuw %c12w
  %p12lo =l and %p12, %mask
  %p0hi =l shr %p0, 32
  %p0lo =l and %p0, %mask
  %mid =l add %p0hi, %p12lo
  %midc =l shr %mid, 32
  %midlo =l and %mid, %mask
  %midsh =l shl %midlo, 32
  %plo =l or %p0lo, %midsh
  %p12hi =l shr %p12, 32
  %c12sh =l shl %c12l, 32
  %phi0 =l add %p3, %p12hi
  %phi1 =l add %phi0, %c12sh
  %phi2 =l add %phi1, %midc
  %k =l add %oi, %ij
  %kofs =l mul %k, 8
  %kptr =l add %r, %kofs
  %rv =l loadl %kptr, 0
  %s1 =l add %plo, %rv
  %c1w =w cultl %s1, %plo
  %c1l =l extuw %c1w
  %s2 =l add %s1, %cr
  %c2w =w cultl %s2, %s1
  %c2l =l extuw %c2w
  storel %s2, %kptr, 0
  %ch1 =l add %phi2, %c1l
  %cr_next =l add %ch1, %c2l
  %ij_next =l add %ij, 1
  j @Li_cond
@Li_done:
  %pcz =w ceql %cr, 0
  jnz %pcz, @Lo_next, @Lprop_start
@Lprop_start:
  %pp0 =l add %oi, %nb
  %pc0 =l copy %cr
  j @Lprop_cond
@Lprop_cond:
  %pp =l phi %pp0 @Lprop_start, %pp_next @Lprop_body
  %pc =l phi %pc0 @Lprop_start, %pc_next @Lprop_body
  %pcz2 =w ceql %pc, 0
  jnz %pcz2, @Lo_next, @Lprop_bound
@Lprop_bound:
  %ppl =w cultl %pp, %cnt
  jnz %ppl, @Lprop_body, @Lo_next
@Lprop_body:
  %pofs =l mul %pp, 8
  %pptr =l add %r, %pofs
  %pval =l loadl %pptr, 0
  %psum =l add %pval, %pc
  %pcw =w cultl %psum, %pval
  %pc_next =l extuw %pcw
  storel %psum, %pptr, 0
  %pp_next =l add %pp, 1
  j @Lprop_cond
@Lo_next:
  %oi_next =l add %oi, 1
  j @Lo_cond
@Lscan_start:
  %cntz =w ceql %cnt, 0
  jnz %cntz, @Lret0, @Lscan_pre
@Lscan_pre:
  %si0 =l sub %cnt, 1
  j @Lscan_cond
@Lscan_cond:
  %si =l phi %si0 @Lscan_pre, %si_next @Lscan_zero
  %sdone =w ceql %si, -1
  jnz %sdone, @Lret0, @Lscan_body
@Lscan_body:
  %sofs =l mul %si, 8
  %sptr =l add %r, %sofs
  %limb =l loadl %sptr, 0
  %lz =w ceql %limb, 0
  jnz %lz, @Lscan_zero, @Lscan_found
@Lscan_zero:
  %si_next =l sub %si, 1
  j @Lscan_cond
@Lscan_found:
  %ret =l add %si, 1
  ret %ret
@Lret0:
  ret %zero
}
