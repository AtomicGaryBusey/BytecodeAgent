

export function l $fib(l %n) {
@start:
%zero = sub %n, %n
%one = add %zero, 1
jmp @loop
@loop:
%i = phi @start %zero, @body %i1
%a = phi @start %zero, @body %a1
%b = phi @start %one, @body %b1
%cmp = csltl %i, %n
jnz %cmp, @body, @done
@body:
%t = sub %a, %a
%b1 = add %a, %b
%a1 = add %b, %t
%i1 = add %i, 1
jmp @loop
@done:
ret %a
}
