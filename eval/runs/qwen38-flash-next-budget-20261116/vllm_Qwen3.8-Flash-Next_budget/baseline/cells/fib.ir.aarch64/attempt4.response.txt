

export function l $fib(l %n) {
@start:
%zero = subl %n, %n
%one = addl %zero, 1
jmp @loop
@loop:
%i = phi @start %zero, @body %i1
%a = phi @start %zero, @body %a1
%b = phi @start %one, @body %b1
%cmp = csltl %i, %n
jnz %cmp, @body, @done
@body:
%b1 = addl %a, %b
%a1 = addl %b, %zero
%i1 = addl %i, 1
jmp @loop
@done:
ret %a
}
