

export function l $count_nonzero_prefix(l %a, l %n) {
@start:
	%i0 =l 0
	%p0 =l %a
	jmp @loop
@loop:
	%i =l phi @start %i0, @advance %i1
	%p =l phi @start %p0, @advance %p1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body:
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @found, @advance
@found:
	ret %i
@advance:
	%i1 =l add %i, 1
	%p1 =l add %p, 8
	jmp @loop
@done:
	ret %i
}
