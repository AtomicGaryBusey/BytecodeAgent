

export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 = and %a, $0xffffffff
    %a1 = shr %a, $32
    %b0 = and %b, $0xffffffff
    %b1 = shr %b, $32
    %p0 = mul %a0, %b0
    %p1 = mul %a0, %b1
    %p2 = mul %a1, %b0
    %p3 = mul %a1, %b1
    %p0h = shr %p0, $32
    %p1h = shr %p1, $32
    %p2h = shr %p2, $32
    %p1l = and %p1, $0xffffffff
    %p2l = and %p2, $0xffffffff
    %c0 = add %p0h, %p1l
    %c1 = add %c0, %p2l
    %carry = shr %c1, $32
    %h0 = add %p3, %p1h
    %h1 = add %h0, %p2h
    %h2 = add %h1, %carry
    ret %h2
}
