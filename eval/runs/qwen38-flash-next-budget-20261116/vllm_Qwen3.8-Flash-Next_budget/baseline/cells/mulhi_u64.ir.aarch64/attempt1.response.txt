

export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =andl %a, $0xffffffff
    %a1 =shrl %a, $32
    %b0 =andl %b, $0xffffffff
    %b1 =shrl %b, $32
    %p0 =mull %a0, %b0
    %p1 =mull %a0, %b1
    %p2 =mull %a1, %b0
    %p3 =mull %a1, %b1
    %p0h =shrl %p0, $32
    %p1h =shrl %p1, $32
    %p2h =shrl %p2, $32
    %p1l =andl %p1, $0xffffffff
    %p2l =andl %p2, $0xffffffff
    %c0 =addl %p0h, %p1l
    %c1 =addl %c0, %p2l
    %carry =shrl %c1, $32
    %h0 =addl %p3, %p1h
    %h1 =addl %h0, %p2h
    %h2 =addl %h1, %carry
    ret %h2
}
