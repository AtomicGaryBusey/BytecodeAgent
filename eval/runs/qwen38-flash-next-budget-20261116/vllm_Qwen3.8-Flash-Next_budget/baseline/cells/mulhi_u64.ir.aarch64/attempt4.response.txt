

export function l $mulhi_u64(l %a, l %b) {
@start
%m1 =l shl $1, $32
%mask =l sub %m1, $1
%ash =l shr %a, $32
%a1 =l and %ash, %mask
%a0 =l and %a, %mask
%bsh =l shr %b, $32
%b1 =l and %bsh, %mask
%b0 =l and %b, %mask
%p0 =l mul %a0, %b0
%p1 =l mul %a0, %b1
%p2 =l mul %a1, %b0
%p3 =l mul %a1, %b1
%p0sh =l shr %p0, $32
%p0h =l and %p0sh, %mask
%p1sh =l shr %p1, $32
%p1h =l and %p1sh, %mask
%p2sh =l shr %p2, $32
%p2h =l and %p2sh, %mask
%p1l =l and %p1, %mask
%p2l =l and %p2, %mask
%c0 =l add %p0h, %p1l
%c1 =l add %c0, %p2l
%csh =l shr %c1, $32
%carry =l and %csh, %mask
%h0 =l add %p3, %p1h
%h1 =l add %h0, %p2h
%hi =l add %h1, %carry
ret %hi
}
