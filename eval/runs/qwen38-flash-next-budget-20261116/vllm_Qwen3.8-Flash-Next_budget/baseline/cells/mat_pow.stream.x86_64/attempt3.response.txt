

mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
cld
mov rbx, rcx
mov r12, r8
mov r13, rsi
mov r14, rsi
imul r14, rsi
lea rbp, [r12 + r14*8]
mov r15, rdx
test r15, r15
jz .Lno_copy
mov rsi, rdi
mov rdi, r12
mov rcx, r14
rep movsq
.Lno_copy:
xor eax, eax
mov rdi, rbx
mov rcx, r14
rep stosq
lea r11, [r13 + 1]
shl r11, 3
mov rdi, rbx
xor r9d, r9d
.Ldiag:
cmp r9, r13
jae .Ldiag_done
mov qword ptr [rdi], 1
add rdi, r11
inc r9
jmp .Ldiag
.Ldiag_done:
test r15, r15
jz .Lzero_tmp
.Lmul:
xor r8d, r8d
.Lmul_i:
cmp r8, r13
jae .Lmul_done
xor r9d, r9d
.Lmul_j:
cmp r9, r13
jae .Lmul_i_next
xor eax, eax
xor r10d, r10d
.Lmul_t:
cmp r10, r13
jae .Lmul_store
mov r11, r8
imul r11, r13
add r11, r10
shl r11, 3
mov rdi, qword ptr [rbx + r11]
mov r11, r10
imul r11, r13
add r11, r9
shl r11, 3
mov rsi, qword ptr [r12 + r11]
imul rdi, rsi
add rax, rdi
inc r10
jmp .Lmul_t
.Lmul_store:
mov r11, r8
imul r11, r13
add r11, r9
shl r11, 3
mov qword ptr [rbp + r11], rax
inc r9
jmp .Lmul_j
.Lmul_i_next:
inc r8
jmp .Lmul_i
.Lmul_done:
mov rdi, rbx
mov rsi, rbp
mov rcx, r14
rep movsq
dec r15
jnz .Lmul
.Lzero_tmp:
xor eax, eax
mov rdi, r12
lea rcx, [r14 + r14]
rep stosq
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
