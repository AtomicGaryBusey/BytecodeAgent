

export function l $reverse(l %a, l %n) {
@start:
  %i0 =addl 0 0
  %j0 =subl %n 1
  jmp @loop
@loop:
  %i =phi l @start %i0 @body %i2
  %j =phi l @start %j0 @body %j2
  %c =csltl %i %j
  jnz %c @body @exit
@body:
  %i8 =mull %i 8
  %ai =addl %a %i8
  %ti =loadl %ai
  %j8 =mull %j 8
  %aj =addl %a %j8
  %tj =loadl %aj
  storel %tj %ai
  storel %ti %aj
  %i2 =addl %i 1
  %j2 =subl %j 1
  jmp @loop
@exit:
  %zero =addl 0 0
  retl %zero
}
