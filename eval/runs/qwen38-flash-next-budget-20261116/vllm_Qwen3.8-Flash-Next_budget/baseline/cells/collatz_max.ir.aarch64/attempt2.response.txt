

export function l $collatz_max(l %n) {
@start
  %limit =l add %n, 1
  %k_init =l copy 1
  %max_init =l copy 0
  %zero =l copy 0
  jmp @outer
@outer
  %k =l phi @start %k_init, @outer_back %k_next
  %max =l phi @start %max_init, @outer_back %max_next
  %outer_cond =w csltl %k, %limit
  jnz %outer_cond, @body, @done
@body
  %k_b =l copy %k
  %zero_b =l copy %zero
  jmp @inner
@inner
  %x =l phi @body %k_b, @inner_back %x_next
  %steps =l phi @body %zero_b, @inner_back %steps_next
  %is_one =w ceql %x, 1
  jnz %is_one, @inner_done, @inner_step
@inner_done
  jmp @check_max
@inner_step
  %bit =l and %x, 1
  %is_even =w ceql %bit, 0
  jnz %is_even, @even, @odd
@even
  %x_even =l div %x, 2
  jmp @inner_back
@odd
  %x_mul =l mul %x, 3
  %x_odd =l add %x_mul, 1
  jmp @inner_back
@inner_back
  %x_next =l phi @even %x_even, @odd %x_odd
  %steps_next =l add %steps, 1
  jmp @inner
@check_max
  %max_less =w csltl %max, %steps
  jnz %max_less, @use_steps, @keep_max
@use_steps
  %max_s =l copy %steps
  jmp @outer_back
@keep_max
  %max_c =l copy %max
  jmp @outer_back
@outer_back
  %max_next =l phi @use_steps %max_s, @keep_max %max_c
  %k_next =l add %k, 1
  jmp @outer
@done
  ret %max
}
