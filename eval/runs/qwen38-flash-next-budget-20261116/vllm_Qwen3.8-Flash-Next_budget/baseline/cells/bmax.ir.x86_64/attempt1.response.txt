

export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%one =l const 1
	%eight =l const 8
	jmp @loop
@loop
	%i =l phi @start, %one, @merge, %i1
	%max =l phi @start, %max0, @merge, %max2
	%more =w csltl %i, %n
	jnz %more, @body
	ret %max
@body
	%off =l mul %i, %eight
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csltl %max, %v
	jnz %gt, @take
	jmp @keep
@take
	%mt =l copy %v
	jmp @merge
@keep
	%mk =l copy %max
	jmp @merge
@merge
	%max2 =l phi @take, %mt, @keep, %mk
	%i1 =l add %i, %one
	jmp @loop
}
