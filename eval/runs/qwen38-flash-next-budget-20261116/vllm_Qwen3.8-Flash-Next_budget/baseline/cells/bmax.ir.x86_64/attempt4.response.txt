

export function l $bmax(l %a, l %n) {
@start
  %m0 =l loadl %a
  %p0 =l add %a, 8
  %i0 =l sub %n, 1
  jmp @loop
@loop
  %m =l phi @start %m0, @greater %mg, @notgreater %mn
  %p =l phi @start %p0, @greater %pg, @notgreater %pn
  %i =l phi @start %i0, @greater %ig, @notgreater %in
  %done =w ceql %i, 0
  jnz %done, @end
  %v =l loadl %p
  %less =w csltl %m, %v
  jnz %less, @greater
  jmp @notgreater
@greater
  %mg =l add %v, 0
  %pg =l add %p, 8
  %ig =l sub %i, 1
  jmp @loop
@notgreater
  %mn =l add %m, 0
  %pn =l add %p, 8
  %in =l sub %i, 1
  jmp @loop
@end
  ret %m
}
