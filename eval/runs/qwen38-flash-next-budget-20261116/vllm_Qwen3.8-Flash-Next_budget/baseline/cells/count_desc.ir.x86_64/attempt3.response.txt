

export function l $count_desc(l %a, l %n) {
@start
  %limit =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @start 0, @body %i_next
  %cnt =l phi @start 0, @body %cnt_next
  %cont =w csltl %i, %limit
  jnz %cont, @body, @exit
@exit
  ret %cnt
@body
  %i_next =l add %i, 1
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %x =l loadl %ptr
  %ptr1 =l add %ptr, 8
  %y =l loadl %ptr1
  %desc =w csltl %y, %x
  %inc =l extuw %desc
  %cnt_next =l add %cnt, %inc
  jmp @loop
}
