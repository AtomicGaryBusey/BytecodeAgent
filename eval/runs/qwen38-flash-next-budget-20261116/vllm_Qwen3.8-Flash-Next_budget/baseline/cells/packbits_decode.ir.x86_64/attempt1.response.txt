

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %pos0 =l sub %n, %n
  %wpos0 =l sub %cap, %cap
  jmp @loop_head

@loop_head
  %pos =l phi @start %pos0, @lit_full_done %lit_pos_next, @rep_full_done %rep_pos_next, @nop %nop_pos_next
  %wpos =l phi @start %wpos0, @lit_full_done %lit_wpos_next, @rep_full_done %rep_wpos_next, @nop %nop_wpos_next
  %has =w csltl %pos, %n
  jnz %has, @body, @done

@body
  %hdrp =l add %src, %pos
  %hb =w loadsb %hdrp
  %h =l extsw %hb
  %pos_h =l add %pos, 1
  %is_lit =w csgtl %h, -1
  jnz %is_lit, @lit, @neg

@neg
  %is_nop =w ceql %h, -128
  jnz %is_nop, @nop, @repeat

@lit
  %lit_len =l add %h, 1
  %lit_src_end =l add %pos_h, %lit_len
  %lit_bad_src =w csgtl %lit_src_end, %n
  jnz %lit_bad_src, @ret_minus2, @lit_cap

@lit_cap
  %lit_out_end =l add %wpos, %lit_len
  %lit_bad_cap =w csgtl %lit_out_end, %cap
  jnz %lit_bad_cap, @lit_partial, @lit_full

@lit_partial
  %lit_part =l sub %cap, %wpos
  %lit_has_part =w csgtl %lit_part, 0
  %lit_pi0 =l sub %lit_part, %lit_part
  jnz %lit_has_part, @lit_partial_head, @ret_minus1

@lit_partial_head
  %lit_pi =l phi @lit_partial %lit_pi0, @lit_partial_body %lit_pi_next
  %lit_pi_more =w csltl %lit_pi, %lit_part
  jnz %lit_pi_more, @lit_partial_body, @ret_minus1

@lit_partial_body
  %lit_psp0 =l add %src, %pos_h
  %lit_psp1 =l add %lit_psp0, %lit_pi
  %lit_pb =w loadub %lit_psp1
  %lit_pdp0 =l add %out, %wpos
  %lit_pdp1 =l add %lit_pdp0, %lit_pi
  storeb %lit_pb, %lit_pdp1
  %lit_pi_next =l add %lit_pi, 1
  jmp @lit_partial_head

@lit_full
  %lit_fi0 =l sub %lit_len, %lit_len
  jmp @lit_full_head

@lit_full_head
  %lit_fi =l phi @lit_full %lit_fi0, @lit_full_body %lit_fi_next
  %lit_fi_more =w csltl %lit_fi, %lit_len
  jnz %lit_fi_more, @lit_full_body, @lit_full_done

@lit_full_body
  %lit_fsp0 =l add %src, %pos_h
  %lit_fsp1 =l add %lit_fsp0, %lit_fi
  %lit_fb =w loadub %lit_fsp1
  %lit_fdp0 =l add %out, %wpos
  %lit_fdp1 =l add %lit_fdp0, %lit_fi
  storeb %lit_fb, %lit_fdp1
  %lit_fi_next =l add %lit_fi, 1
  jmp @lit_full_head

@lit_full_done
  %lit_pos_next =l add %pos_h, %lit_len
  %lit_wpos_next =l add %wpos, %lit_len
  jmp @loop_head

@repeat
  %rep_src_ok =w csltl %pos_h, %n
  jnz %rep_src_ok, @rep_ok, @ret_minus2

@rep_ok
  %rep_valp =l add %src, %pos_h
  %rep_val =w loadub %rep_valp
  %rep_len =l sub 1, %h
  %rep_out_end =l add %wpos, %rep_len
  %rep_bad_cap =w csgtl %rep_out_end, %cap
  jnz %rep_bad_cap, @rep_partial, @rep_full

@rep_partial
  %rep_part =l sub %cap, %wpos
  %rep_has_part =w csgtl %rep_part, 0
  %rep_pi0 =l sub %rep_part, %rep_part
  jnz %rep_has_part, @rep_partial_head, @ret_minus1

@rep_partial_head
  %rep_pi =l phi @rep_partial %rep_pi0, @rep_partial_body %rep_pi_next
  %rep_pi_more =w csltl %rep_pi, %rep_part
  jnz %rep_pi_more, @rep_partial_body, @ret_minus1

@rep_partial_body
  %rep_pdp0 =l add %out, %wpos
  %rep_pdp1 =l add %rep_pdp0, %rep_pi
  storeb %rep_val, %rep_pdp1
  %rep_pi_next =l add %rep_pi, 1
  jmp @rep_partial_head

@rep_full
  %rep_fi0 =l sub %rep_len, %rep_len
  jmp @rep_full_head

@rep_full_head
  %rep_fi =l phi @rep_full %rep_fi0, @rep_full_body %rep_fi_next
  %rep_fi_more =w csltl %rep_fi, %rep_len
  jnz %rep_fi_more, @rep_full_body, @rep_full_done

@rep_full_body
  %rep_fdp0 =l add %out, %wpos
  %rep_fdp1 =l add %rep_fdp0, %rep_fi
  storeb %rep_val, %rep_fdp1
  %rep_fi_next =l add %rep_fi, 1
  jmp @rep_full_head

@rep_full_done
  %rep_pos_next =l add %pos_h, 1
  %rep_wpos_next =l add %wpos, %rep_len
  jmp @loop_head

@nop
  %nop_pos_next =l add %pos_h, 0
  %nop_wpos_next =l add %wpos, 0
  jmp @loop_head

@done
  ret %wpos

@ret_minus2
  ret -2

@ret_minus1
  ret -1
}
