

export function l $vm_run(l $code, l $len, l $stack, l $cap) {
@start
%zero = addl 0 0
%one = addl %zero 1
%eight = mull %one 8
%sixteen = addl %eight %eight
%lenbytes = mull $len %eight
%limit = addl $code %lenbytes
%capbytes = mull $cap %eight
%spend = addl $stack %capbytes
%ip0 = addl $code %zero
%tp0 = addl $stack %zero
jmp @loop
@loop
%ip = phi @start %ip0 @cont %ip_c
%tp = phi @start %tp0 @cont %tp_c
%hitlimit = ceql %ip %limit
jnz %hitlimit @exit @exec
@exec
%opval = loadl %ip
%cmp0 = ceql %opval 0
jnz %cmp0 @halt_blk @op1
@op1
%cmp1 = ceql %opval 1
jnz %cmp1 @push_blk @op2
@op2
%cmp2 = ceql %opval 2
jnz %cmp2 @add_blk @op3
@op3
%cmp3 = ceql %opval 3
jnz %cmp3 @sub_blk @op4
@op4
%cmp4 = ceql %opval 4
jnz %cmp4 @mul_blk @op5
@op5
%cmp5 = ceql %opval 5
jnz %cmp5 @dup_blk @op6
@op6
%cmp6 = ceql %opval 6
jnz %cmp6 @swap_blk @op7
@op7
%cmp7 = ceql %opval 7
jnz %cmp7 @drop_blk @op8
@op8
%cmp8 = ceql %opval 8
jnz %cmp8 @over_blk @op9
@op9
%cmp9 = ceql %opval 9
jnz %cmp9 @lt_blk @op10
@op10
%cmp10 = ceql %opval 10
jnz %cmp10 @jz_blk @op11
@op11
%cmp11 = ceql %opval 11
jnz %cmp11 @jmp_blk @unk_blk
@halt_blk
%tp_halt = addl %tp %zero
jmp @exit
@push_blk
%p_push_op = addl %ip %eight
%v_push = loadl %p_push_op
storel %v_push %tp
%ip_push = addl %ip %sixteen
%tp_push = addl %tp %eight
jmp @cont
@add_blk
%p_a_add = subl %tp %sixteen
%p_b_add = subl %tp %eight
%a_add = loadl %p_a_add
%b_add = loadl %p_b_add
%r_add = addl %a_add %b_add
storel %r_add %p_a_add
%ip_add = addl %ip %eight
%tp_add = addl %p_b_add %zero
jmp @cont
@sub_blk
%p_a_sub = subl %tp %sixteen
%p_b_sub = subl %tp %eight
%a_sub = loadl %p_a_sub
%b_sub = loadl %p_b_sub
%r_sub = subl %a_sub %b_sub
storel %r_sub %p_a_sub
%ip_sub = addl %ip %eight
%tp_sub = addl %p_b_sub %zero
jmp @cont
@mul_blk
%p_a_mul = subl %tp %sixteen
%p_b_mul = subl %tp %eight
%a_mul = loadl %p_a_mul
%b_mul = loadl %p_b_mul
%r_mul = mull %a_mul %b_mul
storel %r_mul %p_a_mul
%ip_mul = addl %ip %eight
%tp_mul = addl %p_b_mul %zero
jmp @cont
@dup_blk
%p_top_dup = subl %tp %eight
%v_dup = loadl %p_top_dup
storel %v_dup %tp
%ip_dup = addl %ip %eight
%tp_dup = addl %tp %eight
jmp @cont
@swap_blk
%p_a_swap = subl %tp %sixteen
%p_b_swap = subl %tp %eight
%v_a_swap = loadl %p_a_swap
%v_b_swap = loadl %p_b_swap
storel %v_b_swap %p_a_swap
storel %v_a_swap %p_b_swap
%ip_swap = addl %ip %eight
%tp_swap = addl %tp %zero
jmp @cont
@drop_blk
%ip_drop = addl %ip %eight
%tp_drop = subl %tp %eight
jmp @cont
@over_blk
%p_below_over = subl %tp %sixteen
%v_over = loadl %p_below_over
storel %v_over %tp
%ip_over = addl %ip %eight
%tp_over = addl %tp %eight
jmp @cont
@lt_blk
%p_a_lt = subl %tp %sixteen
%p_b_lt = subl %tp %eight
%a_lt = loadl %p_a_lt
%b_lt = loadl %p_b_lt
%c_lt = csltl %a_lt %b_lt
jnz %c_lt @lt_true @lt_false
@lt_true
storel %one %p_a_lt
%ip_lttrue = addl %ip %eight
%tp_lttrue = addl %p_b_lt %zero
jmp @cont
@lt_false
storel %zero %p_a_lt
%ip_ltfalse = addl %ip %eight
%tp_ltfalse = addl %p_b_lt %zero
jmp @cont
@jz_blk
%p_v_jz = subl %tp %eight
%v_jz = loadl %p_v_jz
%p_t_jz = addl %ip %eight
%cmpz = ceql %v_jz %zero
%tp_jz_after = subl %tp %eight
%ip_jz_false = addl %ip %sixteen
jnz %cmpz @jz_true @jz_false
@jz_true
%t_jz = loadl %p_t_jz
%tbytes_jz = mull %t_jz %eight
%ip_jztrue = addl $code %tbytes_jz
%tp_jztrue = addl %tp_jz_after %zero
jmp @cont
@jz_false
%ip_jzfalse = addl %ip_jz_false %zero
%tp_jzfalse = addl %tp_jz_after %zero
jmp @cont
@jmp_blk
%p_t_jmp = addl %ip %eight
%t_jmp = loadl %p_t_jmp
%tbytes_jmp = mull %t_jmp %eight
%ip_jmp = addl $code %tbytes_jmp
%tp_jmp = addl %tp %zero
jmp @cont
@unk_blk
%ip_unk = addl %ip %eight
%tp_unk = addl %tp %zero
jmp @cont
@cont
%ip_c = phi @push_blk %ip_push @add_blk %ip_add @sub_blk %ip_sub @mul_blk %ip_mul @dup_blk %ip_dup @swap_blk %ip_swap @drop_blk %ip_drop @over_blk %ip_over @lt_true %ip_lttrue @lt_false %ip_ltfalse @jz_true %ip_jztrue @jz_false %ip_jzfalse @jmp_blk %ip_jmp @unk_blk %ip_unk
%tp_c = phi @push_blk %tp_push @add_blk %tp_add @sub_blk %tp_sub @mul_blk %tp_mul @dup_blk %tp_dup @swap_blk %tp_swap @drop_blk %tp_drop @over_blk %tp_over @lt_true %tp_lttrue @lt_false %tp_ltfalse @jz_true %tp_jztrue @jz_false %tp_jzfalse @jmp_blk %tp_jmp @unk_blk %tp_unk
jmp @loop
@exit
%tp_out = phi @loop %tp @halt_blk %tp_halt
jmp @zeroloop
@zeroloop
%zptr = phi @exit %tp_out @zbody %znext
%zdone = ceql %zptr %spend
jnz %zdone @zero_done @zbody
@zbody
storel %zero %zptr
%znext = addl %zptr %eight
jmp @zeroloop
@zero_done
%empty = ceql %tp_out $stack
jnz %empty @ret0 @retval
@ret0
ret %zero
@retval
%p_ret = subl %tp_out %eight
%v_ret = loadl %p_ret
ret %v_ret
}
