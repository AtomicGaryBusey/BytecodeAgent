

export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
%zero = addl 0 0
%one = addl %zero 1
%two = addl %one %one
%four = addl %two %two
%eight = addl %four %four
%sixteen = addl %eight %eight
%lenb = mull %len %eight
%limit = addl %code %lenb
%capb = mull %cap %eight
%spend = addl %stack %capb
%ip0 = addl %code %zero
%tp0 = addl %stack %zero
jmp @loop
@loop
%ip = phi @start %ip0 @cont %ipc
%tp = phi @start %tp0 @cont %tpc
%hit = ceql %ip %limit
jnz %hit @exit @exec
@exec
%op = loadl %ip
%c0 = ceql %op %zero
jnz %c0 @halt @op1
@op1
%c1 = ceql %op %one
jnz %c1 @push @op2
@op2
%c2 = ceql %op 2
jnz %c2 @add @op3
@op3
%c3 = ceql %op 3
jnz %c3 @sub @op4
@op4
%c4 = ceql %op 4
jnz %c4 @mul @op5
@op5
%c5 = ceql %op 5
jnz %c5 @dup @op6
@op6
%c6 = ceql %op 6
jnz %c6 @swap @op7
@op7
%c7 = ceql %op 7
jnz %c7 @drop @op8
@op8
%c8 = ceql %op 8
jnz %c8 @over @op9
@op9
%c9 = ceql %op 9
jnz %c9 @lt @op10
@op10
%c10 = ceql %op 10
jnz %c10 @jz @op11
@op11
%c11 = ceql %op 11
jnz %c11 @jmp @unk
@halt
%tph = addl %tp %zero
jmp @exit
@push
%pp = addl %ip %eight
%vp = loadl %pp
storel %vp %tp
%ipp = addl %ip %sixteen
%tpp = addl %tp %eight
jmp @cont
@add
%pa = subl %tp %sixteen
%pb = subl %tp %eight
%a = loadl %pa
%b = loadl %pb
%ra = addl %a %b
storel %ra %pa
%ipad = addl %ip %eight
%tpad = addl %pb %zero
jmp @cont
@sub
%pa1 = subl %tp %sixteen
%pb1 = subl %tp %eight
%a1 = loadl %pa1
%b1 = loadl %pb1
%rs = subl %a1 %b1
storel %rs %pa1
%ipsub = addl %ip %eight
%tpsub = addl %pb1 %zero
jmp @cont
@mul
%pa2 = subl %tp %sixteen
%pb2 = subl %tp %eight
%a2 = loadl %pa2
%b2 = loadl %pb2
%rm = mull %a2 %b2
storel %rm %pa2
%ipmul = addl %ip %eight
%tpmul = addl %pb2 %zero
jmp @cont
@dup
%pt = subl %tp %eight
%vd = loadl %pt
storel %vd %tp
%ipdup = addl %ip %eight
%tpdup = addl %tp %eight
jmp @cont
@swap
%pa3 = subl %tp %sixteen
%pb3 = subl %tp %eight
%a3 = loadl %pa3
%b3 = loadl %pb3
storel %b3 %pa3
storel %a3 %pb3
%ipsw = addl %ip %eight
%tpsw = addl %tp %zero
jmp @cont
@drop
%ipdr = addl %ip %eight
%tpdr = subl %tp %eight
jmp @cont
@over
%pbol = subl %tp %sixteen
%vo = loadl %pbol
storel %vo %tp
%ipov = addl %ip %eight
%tpov = addl %tp %eight
jmp @cont
@lt
%pa4 = subl %tp %sixteen
%pb4 = subl %tp %eight
%a4 = loadl %pa4
%b4 = loadl %pb4
%cl = csltl %a4 %b4
jnz %cl @ltt @ltf
@ltt
storel %one %pa4
%iprt = addl %ip %eight
%tprr = addl %pb4 %zero
jmp @cont
@ltf
storel %zero %pa4
%iprf = addl %ip %eight
%tprrf = addl %pb4 %zero
jmp @cont
@jz
%pj = subl %tp %eight
%vj = loadl %pj
%ptj = addl %ip %eight
%tj = loadl %ptj
%tz = ceql %vj %zero
%tpj = subl %tp %eight
%ipjf = addl %ip %sixteen
jnz %tz @jzt @jzf
@jzt
%tbj = mull %tj %eight
%ipjz = addl %code %tbj
%tpjz = addl %tpj %zero
jmp @cont
@jzf
%ipjzfalse = addl %ipjf %zero
%tpjzfalse = addl %tpj %zero
jmp @cont
@jmp
%ptjm = addl %ip %eight
%tjm = loadl %ptjm
%tbjm = mull %tjm %eight
%ipjmp = addl %code %tbjm
%tpjmp = addl %tp %zero
jmp @cont
@unk
%ipunk = addl %ip %eight
%tpunk = addl %tp %zero
jmp @cont
@cont
%ipc = phi @push %ipp @add %ipad @sub %ipsub @mul %ipmul @dup %ipdup @swap %ipsw @drop %ipdr @over %ipov @ltt %iprt @ltf %iprf @jzt %ipjz @jzf %ipjzfalse @jmp %ipjmp @unk %ipunk
%tpc = phi @push %tpp @add %tpad @sub %tpsub @mul %tpmul @dup %tpdup @swap %tpsw @drop %tpdr @over %tpov @ltt %tprr @ltf %tprrf @jzt %tpjz @jzf %tpjzfalse @jmp %tpjmp @unk %tpunk
jmp @loop
@exit
%tpout = phi @loop %tp @halt %tph
jmp @zloop
@zloop
%zp = phi @exit %tpout @zbody %znext
%zdone = ceql %zp %spend
jnz %zdone @done @zbody
@zbody
storel %zero %zp
%znext = addl %zp %eight
jmp @zloop
@done
%empty = ceql %tpout %stack
jnz %empty @ret0 @retv
@ret0
ret %zero
@retv
%pret = subl %tpout %eight
%vret = loadl %pret
ret %vret
}
