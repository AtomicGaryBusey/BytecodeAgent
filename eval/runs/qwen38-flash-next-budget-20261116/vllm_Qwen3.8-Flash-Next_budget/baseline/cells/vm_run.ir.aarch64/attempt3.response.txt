

export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
%zero =l add 0, 0
%one =l add %zero, 1
%lenb =l mul %len, 8
%limit =l add %code, %lenb
%capb =l mul %cap, 8
%spend =l add %stack, %capb
jmp @loop
@loop
%ip =l phi @start %code, @cont %ipc
%tp =l phi @start %stack, @cont %tpc
%hit =w ceql %ip, %limit
jnz %hit @exit @exec
@exec
%op =l loadl %ip
%c0 =w ceql %op, 0
jnz %c0 @halt @op1
@op1
%c1 =w ceql %op, 1
jnz %c1 @push @op2
@op2
%c2 =w ceql %op, 2
jnz %c2 @add @op3
@op3
%c3 =w ceql %op, 3
jnz %c3 @sub @op4
@op4
%c4 =w ceql %op, 4
jnz %c4 @mul @op5
@op5
%c5 =w ceql %op, 5
jnz %c5 @dup @op6
@op6
%c6 =w ceql %op, 6
jnz %c6 @swap @op7
@op7
%c7 =w ceql %op, 7
jnz %c7 @drop @op8
@op8
%c8 =w ceql %op, 8
jnz %c8 @over @op9
@op9
%c9 =w ceql %op, 9
jnz %c9 @lt @op10
@op10
%c10 =w ceql %op, 10
jnz %c10 @jz @op11
@op11
%c11 =w ceql %op, 11
jnz %c11 @jmp @unk
@halt
%tph =l add %tp, 0
jmp @exit
@push
%pp =l add %ip, 8
%vp =l loadl %pp
storel %vp, %tp
%ipp =l add %ip, 16
%tpp =l add %tp, 8
jmp @cont
@add
%pa =l sub %tp, 16
%pb =l sub %tp, 8
%a =l loadl %pa
%b =l loadl %pb
%ra =l add %a, %b
storel %ra, %pa
%ipad =l add %ip, 8
%tpad =l sub %tp, 8
jmp @cont
@sub
%pa1 =l sub %tp, 16
%pb1 =l sub %tp, 8
%a1 =l loadl %pa1
%b1 =l loadl %pb1
%rs =l sub %a1, %b1
storel %rs, %pa1
%ipsub =l add %ip, 8
%tpsub =l sub %tp, 8
jmp @cont
@mul
%pa2 =l sub %tp, 16
%pb2 =l sub %tp, 8
%a2 =l loadl %pa2
%b2 =l loadl %pb2
%rm =l mul %a2, %b2
storel %rm, %pa2
%ipmul =l add %ip, 8
%tpmul =l sub %tp, 8
jmp @cont
@dup
%pt =l sub %tp, 8
%vd =l loadl %pt
storel %vd, %tp
%ipdup =l add %ip, 8
%tpdup =l add %tp, 8
jmp @cont
@swap
%pa3 =l sub %tp, 16
%pb3 =l sub %tp, 8
%a3 =l loadl %pa3
%b3 =l loadl %pb3
storel %b3, %pa3
storel %a3, %pb3
%ipsw =l add %ip, 8
%tpsw =l add %tp, 0
jmp @cont
@drop
%ipdr =l add %ip, 8
%tpdr =l sub %tp, 8
jmp @cont
@over
%pbol =l sub %tp, 16
%vo =l loadl %pbol
storel %vo, %tp
%ipov =l add %ip, 8
%tpov =l add %tp, 8
jmp @cont
@lt
%pa4 =l sub %tp, 16
%pb4 =l sub %tp, 8
%a4 =l loadl %pa4
%b4 =l loadl %pb4
%cl =w csltl %a4, %b4
jnz %cl @ltt @ltf
@ltt
storel %one, %pa4
%iprt =l add %ip, 8
%tprr =l sub %tp, 8
jmp @cont
@ltf
storel %zero, %pa4
%iprf =l add %ip, 8
%tprrf =l sub %tp, 8
jmp @cont
@jz
%pj =l sub %tp, 8
%vj =l loadl %pj
%ptj =l add %ip, 8
%tj =l loadl %ptj
%tz =w ceql %vj, 0
jnz %tz @jzt @jzf
@jzt
%tbj =l mul %tj, 8
%ipjz =l add %code, %tbj
%tpjz =l sub %tp, 8
jmp @cont
@jzf
%ipjzf =l add %ip, 16
%tpjzf =l sub %tp, 8
jmp @cont
@jmp
%ptjm =l add %ip, 8
%tjm =l loadl %ptjm
%tbjm =l mul %tjm, 8
%ipjmp =l add %code, %tbjm
%tpjmp =l add %tp, 0
jmp @cont
@unk
%ipunk =l add %ip, 8
%tpunk =l add %tp, 0
jmp @cont
@cont
%ipc =l phi @push %ipp, @add %ipad, @sub %ipsub, @mul %ipmul, @dup %ipdup, @swap %ipsw, @drop %ipdr, @over %ipov, @ltt %iprt, @ltf %iprf, @jzt %ipjz, @jzf %ipjzf, @jmp %ipjmp, @unk %ipunk
%tpc =l phi @push %tpp, @add %tpad, @sub %tpsub, @mul %tpmul, @dup %tpdup, @swap %tpsw, @drop %tpdr, @over %tpov, @ltt %tprr, @ltf %tprrf, @jzt %tpjz, @jzf %tpjzf, @jmp %tpjmp, @unk %tpunk
jmp @loop
@exit
%tpout =l phi @loop %tp, @halt %tph
jmp @zloop
@zloop
%zp =l phi @exit %tpout, @zbody %znext
%zdone =w ceql %zp, %spend
jnz %zdone @done @zbody
@zbody
storel %zero, %zp
%znext =l add %zp, 8
jmp @zloop
@done
%empty =w ceql %tpout, %stack
jnz %empty @ret0 @retv
@ret0
ret %zero
@retv
%pret =l sub %tpout, 8
%vret =l loadl %pret
ret %vret
}
