

export function l $memmove_i64(l %dst, l %src, l %n) {
@start
%dgt =w csltl %src, %dst
jnz %dgt, @backward, @forward
@forward
%i0 =l $0
jmp @fcond
@fcond
%i =l phi @forward %i0, @fbody %iinc
%cf =w csltl %i, %n
jnz %cf, @fbody, @done
@fbody
%off =l mul %i, $8
%srcp =l add %src, %off
%dstp =l add %dst, %off
%val =l loadl %srcp
storel %val, %dstp
%iinc =l add %i, $1
jmp @fcond
@backward
%i0b =l sub %n, $1
jmp @bcond
@bcond
%ib =l phi @backward %i0b, @bbody %ibinc
%lt =w csltl %ib, $0
jnz %lt, @done, @bbody
@bbody
%offb =l mul %ib, $8
%srcpb =l add %src, %offb
%dstpb =l add %dst, %offb
%valb =l loadl %srcpb
storel %valb, %dstpb
%ibinc =l sub %ib, $1
jmp @bcond
@done
%ret =l $0
ret %ret
}
