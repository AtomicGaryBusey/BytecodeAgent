

export function l $bubble_sort(l %a, l %n) {
@start:
  %zero = subl %n, %n
  %c1 = cslel %n, 1
  jnz %c1, @early, @setup
@early:
  ret %zero
@setup:
  %i = subl %n, 1
@outer:
  %j = subl %i, %i
  %pj = addl %a, %zero
  %pj1 = addl %pj, 8
@inner:
  %c2 = csltl %j, %i
  jnz %c2, @body, @endinner
@body:
  %vj = loadl %pj
  %vp = loadl %pj1
  %c3 = csltl %vp, %vj
  jnz %c3, @swap, @noswap
@swap:
  storel %vp, %pj
  storel %vj, %pj1
  jmp @noswap
@noswap:
  %j = addl %j, 1
  %pj = addl %pj, 8
  %pj1 = addl %pj1, 8
  jmp @inner
@endinner:
  %i = subl %i, 1
  %c4 = csgtl %i, 0
  jnz %c4, @outer, @done
@done:
  ret %zero
}
