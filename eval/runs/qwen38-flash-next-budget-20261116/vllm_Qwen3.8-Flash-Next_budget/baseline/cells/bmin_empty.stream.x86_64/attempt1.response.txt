

bmin_empty:
 cld
 test rsi, rsi
 jle .Lbmin_zero
 mov rax, qword ptr [rdi]
 dec rsi
 jz .Lbmin_done
 lea rdi, qword ptr [rdi + 8]
.Lbmin_loop:
 mov rdx, qword ptr [rdi]
 cmp rdx, rax
 cmovl rax, rdx
 add rdi, 8
 dec rsi
 jnz .Lbmin_loop
.Lbmin_done:
 ret
.Lbmin_zero:
 mov ecx, 0x7fffffff
 shl rcx, 32
 mov edx, 0xffffffff
 or rcx, rdx
 mov rax, rcx
 ret
