

map_sum:
    cld
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdx
    mov r12, rdi
    mov r13, rsi
    xor r14, r14
.Lloop:
    test r13, r13
    jle .Ldone
    mov rdi, qword ptr [r12]
    add r12, 8
    dec r13
    cld
    call rbx
    add r14, rax
    jmp .Lloop
.Ldone:
    cld
    mov rax, r14
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
