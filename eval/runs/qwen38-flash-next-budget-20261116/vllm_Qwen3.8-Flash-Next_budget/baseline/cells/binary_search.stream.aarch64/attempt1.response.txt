

binary_search:
	mov x3, #0
	mov x4, x1
.Lloop:
	cmp x3, x4
	b.ge .Ldone
	sub x5, x4, x3
	asr x5, x5, #1
	add x5, x3, x5
	ldr x6, [x0, x5, lsl #3]
	cmp x6, x2
	b.lt .Lright
	mov x4, x5
	b .Lloop
.Lright:
	add x3, x5, #1
	b .Lloop
.Ldone:
	cmp x3, x1
	b.ge .Lnotfound
	ldr x5, [x0, x3, lsl #3]
	cmp x5, x2
	b.ne .Lnotfound
	mov x0, x3
	ret
.Lnotfound:
	mov x0, #-1
	ret
