

export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0 @join %i2
	%c =l phi @start 0 @join %c2
	%cmp =w csltl %i %n
	jnz %cmp @body @done
@body
	%off =l mul %i 8
	%sp =l add %src %off
	%x =l loadl %sp
	%dp =l add %dst %off
	%r =l call %f(l %x)
	storel %r, %dp
	%lt =w csltl %r %x
	jnz %lt @yes @no
@yes
	jmp @join
@no
	jmp @join
@join
	%add =l phi @yes 1 @no 0
	%c2 =l add %c %add
	%i2 =l add %i 1
	jmp @loop
@done
	ret %c
}
