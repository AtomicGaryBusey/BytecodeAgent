

export function l $bmax(l $a, l $n) {
@start
	%max0 =l loadl $a
	%p0 =l addl $a, 8
	%cnt0 =l subl $n, 1
	jmp @loop
@loop
	%max =l phi @start %max0, @update %max2
	%p =l phi @start %p0, @update %p2
	%cnt =l phi @start %cnt0, @update %cnt2
	%done =w ceql %cnt, 0
	jnz %done, @end
	%v =l loadl %p
	%less =w csltl %v, %max
	jnz %less, @no
	jmp @yes
@yes
	%maxY =l addl %v, 0
	jmp @update
@no
	%maxN =l addl %max, 0
	jmp @update
@update
	%max2 =l phi @yes %maxY, @no %maxN
	%p2 =l addl %p, 8
	%cnt2 =l subl %cnt, 1
	jmp @loop
@end
	ret %max
}
