

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
%zero =l add 0, 0
%limit =l add %n, 1
%zj =l add 0, 0
@zero_head
%zc =w csltl %zj, %limit
jnz %zc, @zero_body, @zero_done
@zero_body
%zoff =l mul %zj, 8
%zaddr =l add %row, %zoff
storel %zero, %zaddr
%zj =l add %zj, 1
jmp @zero_head
@zero_done
%nzero =w csltl %n, 1
jnz %nzero, @finish, @mzero_check
@mzero_check
%mzero =w csltl %m, 1
jnz %mzero, @finish, @outer_start
@outer_start
%i =l add 0, 0
@outer_head
%oc =w csltl %i, %m
jnz %oc, @outer_body, @finish
@outer_body
%px =l add %x, %i
%xi =w loadub %px
%diag =l add 0, 0
%j =l add 0, 1
@inner_head
%ic =w csltl %j, %limit
jnz %ic, @inner_body, @outer_next
@inner_body
%jm1 =l sub %j, 1
%py =l add %y, %jm1
%yj =w loadub %py
%j8 =l mul %j, 8
%addrj =l add %row, %j8
%up =l loadl %addrj
%jm18 =l mul %jm1, 8
%addrjm1 =l add %row, %jm18
%left =l loadl %addrjm1
%eq =w ceqw %xi, %yj
jnz %eq, @eq_path, @neq_path
@eq_path
%eqv =l add %diag, 1
storel %eqv, %addrj
jmp @after_update
@neq_path
%uplt =w csltl %up, %left
jnz %uplt, @val_left, @val_up
@val_left
storel %left, %addrj
jmp @after_update
@val_up
storel %up, %addrj
jmp @after_update
@after_update
%diag =l add %up, 0
%j =l add %j, 1
jmp @inner_head
@outer_next
%i =l add %i, 1
jmp @outer_head
@finish
%n8 =l mul %n, 8
%addrn =l add %row, %n8
%res =l loadl %addrn
ret %res
}
