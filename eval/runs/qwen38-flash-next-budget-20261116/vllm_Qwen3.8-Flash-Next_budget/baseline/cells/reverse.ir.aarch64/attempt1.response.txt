

export function l $reverse(l %a, l %n) {
@start:
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %j0 =l sub %n, %one
    jmp @loop
@loop:
    %i =l phi @start %zero, @body %i1
    %j =l phi @start %j0, @body %j1
    %c =w csltl %i, %j
    jnz %c, @body, @done
@body:
    %oi =l mul %i, %eight
    %oj =l mul %j, %eight
    %pi =l add %a, %oi
    %pj =l add %a, %oj
    %x =l loadl %pi
    %y =l loadl %pj
    storel %y, %pi
    storel %x, %pj
    %i1 =l add %i, %one
    %j1 =l sub %j, %one
    jmp @loop
@done:
    ret %zero
}
