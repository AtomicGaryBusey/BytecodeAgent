

export function l $qsort_pairs(l %a, l %n) {
@entry:
  %z0 =l subl %n, %n
  %c0 =w csltl %n, 2
  jnz %c0 @ret0 @setup
@ret0:
  ret %z0
@setup:
  %gt0 =l subl %n, 1
  %lt0 =l subl %gt0, %gt0
  %i0 =l subl %gt0, %gt0
  %g0 =l addl %gt0, %gt0
  %g1 =l addl %g0, %g0
  %g2 =l addl %g1, %g1
  %g3 =l addl %g2, %g2
  %p0 =l addl %a, %g3
  %p8 =l addl %p0, 8
  %pk =l loadl %p0
  %pv =l loadl %p8
  jmp @loop
@loop:
  %lt =l phi @setup %lt0 @swap_lt %lt1 @advance_eq %lt2 @swap_gt %lt3
  %i =l phi @setup %i0 @swap_lt %i1 @advance_eq %i2 @swap_gt %i3
  %gt =l phi @setup %gt0 @swap_lt %gt1 @advance_eq %gt2 @swap_gt %gt3
  %ex =w csgtl %i, %gt
  jnz %ex @after @body
@body:
  %b0 =l addl %i, %i
  %b1 =l addl %b0, %b0
  %b2 =l addl %b1, %b1
  %b3 =l addl %b2, %b2
  %ip =l addl %a, %b3
  %ip8 =l addl %ip, 8
  %ik =l loadl %ip
  %iv =l loadl %ip8
  %klt =w csltl %ik, %pk
  jnz %klt @swap_lt @check_key
@check_key:
  %kgt =w csgtl %ik, %pk
  jnz %kgt @swap_gt @check_val
@check_val:
  %vlt =w csltl %iv, %pv
  jnz %vlt @swap_lt @check_vgt
@check_vgt:
  %vgt =w csgtl %iv, %pv
  jnz %vgt @swap_gt @advance_eq
@advance_eq:
  %lt2 =l copyl %lt
  %i2 =l addl %i, 1
  %gt2 =l copyl %gt
  jmp @loop
@swap_lt:
  %s0 =l addl %lt, %lt
  %s1 =l addl %s0, %s0
  %s2 =l addl %s1, %s1
  %s3 =l addl %s2, %s2
  %lp =l addl %a, %s3
  %lp8 =l addl %lp, 8
  %lk =l loadl %lp
  %lv =l loadl %lp8
  storel %ik, %lp
  storel %iv, %lp8
  storel %lk, %ip
  storel %lv, %ip8
  %lt1 =l addl %lt, 1
  %i1 =l addl %i, 1
  %gt1 =l copyl %gt
  jmp @loop
@swap_gt:
  %u0 =l addl %gt, %gt
  %u1 =l addl %u0, %u0
  %u2 =l addl %u1, %u1
  %u3 =l addl %u2, %u2
  %gp =l addl %a, %u3
  %gp8 =l addl %gp, 8
  %gk =l loadl %gp
  %gv =l loadl %gp8
  storel %ik, %gp
  storel %iv, %gp8
  storel %gk, %ip
  storel %gv, %ip8
  %lt3 =l copyl %lt
  %i3 =l copyl %i
  %gt3 =l subl %gt, 1
  jmp @loop
@after:
  %r0 =l addl %gt, 1
  %r1 =l addl %r0, %r0
  %r2 =l addl %r1, %r1
  %r3 =l addl %r2, %r2
  %r4 =l addl %r3, %r3
  %rp =l addl %a, %r4
  %rl =l subl %n, %r0
  %res1 =l call $qsort_pairs(l %a, l %lt)
  %res2 =l call $qsort_pairs(l %rp, l %rl)
  %sum =l addl %res1, %res2
  %z1 =l subl %sum, %sum
  ret %z1
}
