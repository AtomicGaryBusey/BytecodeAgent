

export function l $qsort_pairs(l %a, l %n) {
@entry:
  %cmp =w csltl l %n, l 2
  jnz %cmp @ret @setup
@ret:
  ret l 0
@setup:
  %z0 =l sub l %n, l %n
  %gt0 =l sub l %n, l 1
  %poff0 =l shl l %gt0, l 4
  %paddr =l add l %a, l %poff0
  %pk =l loadl %paddr
  %pvaddr =l add l %paddr, l 8
  %pv =l loadl %pvaddr
  jmp @loop
@loop:
  %lt =l phi @setup %z0 @swap_lt %lt_n1 @advance_eq %lt_keep_a @swap_gt %lt_keep_g
  %i =l phi @setup %z0 @swap_lt %i_n1 @advance_eq %i_n2 @swap_gt %i_keep_g
  %gt =l phi @setup %gt0 @swap_lt %gt_keep_s @advance_eq %gt_keep_a @swap_gt %gt_n1
  %out =w csgtl l %i, l %gt
  jnz %out @after @body
@body:
  %i_off =l shl l %i, l 4
  %ip =l add l %a, l %i_off
  %ik =l loadl %ip
  %ip8 =l add l %ip, l 8
  %iv =l loadl %ip8
  %k_lt =w csltl l %ik, l %pk
  jnz %k_lt @swap_lt @check_key
@check_key:
  %k_gt =w csgtl l %ik, l %pk
  jnz %k_gt @swap_gt @check_val
@check_val:
  %v_lt =w csltl l %iv, l %pv
  jnz %v_lt @swap_lt @check_v_gt
@check_v_gt:
  %v_gt =w csgtl l %iv, l %pv
  jnz %v_gt @swap_gt @advance_eq
@advance_eq:
  %lt_keep_a =l add l %lt, l 0
  %i_n2 =l add l %i, l 1
  %gt_keep_a =l add l %gt, l 0
  jmp @loop
@swap_lt:
  %lt_off =l shl l %lt, l 4
  %lp =l add l %a, l %lt_off
  %lp8 =l add l %lp, l 8
  %lk =l loadl %lp
  %lv =l loadl %lp8
  storel %ik, %lp
  storel %iv, %lp8
  storel %lk, %ip
  storel %lv, %ip8
  %lt_n1 =l add l %lt, l 1
  %i_n1 =l add l %i, l 1
  %gt_keep_s =l add l %gt, l 0
  jmp @loop
@swap_gt:
  %gt_off =l shl l %gt, l 4
  %gp =l add l %a, l %gt_off
  %gp8 =l add l %gp, l 8
  %gk =l loadl %gp
  %gv =l loadl %gp8
  storel %ik, %gp
  storel %iv, %gp8
  storel %gk, %ip
  storel %gv, %ip8
  %lt_keep_g =l add l %lt, l 0
  %i_keep_g =l add l %i, l 0
  %gt_n1 =l sub l %gt, l 1
  jmp @loop
@after:
  %r_idx =l add l %gt, l 1
  %r_cnt =l sub l %n, l %r_idx
  %r_off =l shl l %r_idx, l 4
  %r_ptr =l add l %a, l %r_off
  %res1 =l call $qsort_pairs(l %a, l %lt)
  %res2 =l call $qsort_pairs(l %r_ptr, l %r_cnt)
  ret l 0
}
