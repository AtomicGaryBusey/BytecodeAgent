

export function l $qsort_pairs(l %a, l %n) {
@entry:
  %zero =l subl %n, %n
  %cmp =w csltl %n, 2
  jnz %cmp @small @setup
@small:
  ret %zero
@setup:
  %gt0 =l subl %n, 1
  %g0 =l addl %gt0, %gt0
  %g1 =l addl %g0, %g0
  %g2 =l addl %g1, %g1
  %g3 =l addl %g2, %g2
  %paddr =l addl %a, %g3
  %paddr8 =l addl %paddr, 8
  %pk =l loadl %paddr
  %pv =l loadl %paddr8
  jmp @loop
@loop:
  %lt =l phi @setup %zero @swap_lt %lt_n1 @advance_eq %lt_ka @swap_gt %lt_kg
  %i =l phi @setup %zero @swap_lt %i_n1 @advance_eq %i_n2 @swap_gt %i_kg
  %gt =l phi @setup %gt0 @swap_lt %gt_ks @advance_eq %gt_ka @swap_gt %gt_n1
  %exit =w csgtl %i, %gt
  jnz %exit @after @body
@body:
  %i0 =l addl %i, %i
  %i1 =l addl %i0, %i0
  %i2 =l addl %i1, %i1
  %i3 =l addl %i2, %i2
  %ip =l addl %a, %i3
  %ip8 =l addl %ip, 8
  %ik =l loadl %ip
  %iv =l loadl %ip8
  %k_lt =w csltl %ik, %pk
  jnz %k_lt @swap_lt @check_key
@check_key:
  %k_gt =w csgtl %ik, %pk
  jnz %k_gt @swap_gt @check_val
@check_val:
  %v_lt =w csltl %iv, %pv
  jnz %v_lt @swap_lt @check_vgt
@check_vgt:
  %v_gt =w csgtl %iv, %pv
  jnz %v_gt @swap_gt @advance_eq
@advance_eq:
  %lt_ka =l addl %lt, %zero
  %i_n2 =l addl %i, 1
  %gt_ka =l addl %gt, %zero
  jmp @loop
@swap_lt:
  %l0 =l addl %lt, %lt
  %l1 =l addl %l0, %l0
  %l2 =l addl %l1, %l1
  %l3 =l addl %l2, %l2
  %lp =l addl %a, %l3
  %lp8 =l addl %lp, 8
  %lk =l loadl %lp
  %lv =l loadl %lp8
  storel %ik, %lp
  storel %iv, %lp8
  storel %lk, %ip
  storel %lv, %ip8
  %lt_n1 =l addl %lt, 1
  %i_n1 =l addl %i, 1
  %gt_ks =l addl %gt, %zero
  jmp @loop
@swap_gt:
  %j0 =l addl %gt, %gt
  %j1 =l addl %j0, %j0
  %j2 =l addl %j1, %j1
  %j3 =l addl %j2, %j2
  %gp =l addl %a, %j3
  %gp8 =l addl %gp, 8
  %gk =l loadl %gp
  %gv =l loadl %gp8
  storel %ik, %gp
  storel %iv, %gp8
  storel %gk, %ip
  storel %gv, %ip8
  %lt_kg =l addl %lt, %zero
  %i_kg =l addl %i, %zero
  %gt_n1 =l subl %gt, 1
  jmp @loop
@after:
  %r_idx =l addl %gt, 1
  %r0 =l addl %r_idx, %r_idx
  %r1 =l addl %r0, %r0
  %r2 =l addl %r1, %r1
  %r3 =l addl %r2, %r2
  %r_ptr =l addl %a, %r3
  %r_len =l subl %n, %r_idx
  %res1 =l call $qsort_pairs(l %a, l %lt)
  %res2 =l call $qsort_pairs(l %r_ptr, l %r_len)
  ret %zero
}
