

fib_rec:
    push rbx
    mov rbx, rdi
    cmp rbx, 1
    jle .Lret_n
    sub rsp, 16
    mov rdi, rbx
    dec rdi
    call fib_rec
    mov qword ptr [rsp], rax
    mov rdi, rbx
    sub rdi, 2
    call fib_rec
    add rax, qword ptr [rsp]
    add rsp, 16
    pop rbx
    ret
.Lret_n:
    mov rax, rbx
    pop rbx
    ret
