

mp_mul:
cld
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbx, r8
mov rbp, rsi
add rbp, rcx
xor eax, eax
xor ecx, ecx
mov rdi, rbx
test rbp, rbp
jz .Lmp_zero_done
.Lmp_zero_loop:
mov [rdi + rcx*8], rax
inc rcx
cmp rcx, rbp
jb .Lmp_zero_loop
.Lmp_zero_done:
xor esi, esi
.Lmp_i_loop:
cmp rsi, r13
jae .Lmp_scan
xor r10d, r10d
xor ecx, ecx
.Lmp_j_loop:
cmp rcx, r15
jae .Lmp_carry_add
mov rax, [r12 + rsi*8]
mul qword ptr [r14 + rcx*8]
mov rdi, rsi
add rdi, rcx
mov r9, [rbx + rdi*8]
xor r11d, r11d
add r9, rax
adc r11, 0
add r9, r10
adc r11, 0
add r11, rdx
adc r11, 0
mov [rbx + rdi*8], r9
mov r10, r11
inc rcx
jmp .Lmp_j_loop
.Lmp_carry_add:
test r10, r10
jz .Lmp_next_i
mov rdi, rsi
add rdi, r15
.Lmp_carry_loop:
mov r9, [rbx + rdi*8]
add r9, r10
mov [rbx + rdi*8], r9
setc al
movzx eax, al
mov r10, rax
inc rdi
test r10, r10
jnz .Lmp_carry_loop
.Lmp_next_i:
inc rsi
jmp .Lmp_i_loop
.Lmp_scan:
test rbp, rbp
jz .Lmp_ret_zero
mov rsi, rbp
dec rsi
.Lmp_scan_loop:
mov rax, [rbx + rsi*8]
test rax, rax
jnz .Lmp_scan_found
dec rsi
jns .Lmp_scan_loop
xor eax, eax
jmp .Lmp_pop_ret
.Lmp_ret_zero:
xor eax, eax
jmp .Lmp_pop_ret
.Lmp_scan_found:
lea rax, [rsi + 1]
.Lmp_pop_ret:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
