

export function l $binary_search(l %a, l %n, l %key) {
@start
%lo0 =l copy 0
%hi0 =l copy %n
jmp @loop
@loop
%lo =l phi @start %lo0, @left %lo_left, @right %lo_right
%hi =l phi @start %hi0, @left %hi_left, @right %hi_right
%cond =w csltl %lo, %hi
jnz %cond, @search, @fail
@search
%sum =l add %lo, %hi
%mid =l shr %sum, 1
%off =l mul %mid, 8
%ptr =l add %a, %off
%val =l loadl %ptr
%eq =w ceql %val, %key
jnz %eq, @found, @ne
@ne
%less =w csltl %key, %val
jnz %less, @right, @left
@left
%lo_left =l add %mid, 1
%hi_left =l copy %hi
jmp @loop
@right
%lo_right =l copy %lo
%hi_right =l copy %mid
jmp @loop
@found
ret %mid
@fail
%notfound =l copy -1
ret %notfound
}
