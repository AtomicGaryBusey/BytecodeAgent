

export function l $binary_search(l %a, l %n, l %key) {
@start
%lo0 =l copy 0
%hi0 =l copy %n
jmp @loop
@loop
%lo =l phi @start %lo0, @left %lo_left, @right %lo_right
%hi =l phi @start %hi0, @left %hi_left, @right %hi_right
%cmp =w csltl %lo, %hi
jnz %cmp, @search, @fail
@search
%diff =l subl %hi, %lo
%half =l shrl %diff, 1
%mid =l addl %lo, %half
%off =l shll %mid, 3
%ptr =l addl %a, %off
%val =l loadl %ptr
%eq =w ceql %val, %key
jnz %eq, @found, @ne
@ne
%less =w csltl %key, %val
jnz %less, @right, @left
@left
%lo_left =l addl %mid, 1
%hi_left =l copy %hi
jmp @loop
@right
%lo_right =l copy %lo
%hi_right =l copy %mid
jmp @loop
@found
ret l %mid
@fail
ret l -1
}
