

export function l $sdivrem(l %a, l %b, l %rem) {
@start:
  %zero =l sub %a, %a
  %iszero =w ceql %b, %zero
  jnz %iszero, @divzero, @checkneg
@divzero:
  storel %a, %rem
  ret %zero
@checkneg:
  %one =l add %zero, 1
  %negone =l sub %zero, %one
  %isneg =w ceql %b, %negone
  jnz %isneg, @checkmin, @normal
@checkmin:
  %nega =l sub %zero, %a
  %ismin =w ceql %a, %nega
  jnz %ismin, @special, @normal
@special:
  storel %zero, %rem
  ret %a
@normal:
  %q =l div %a, %b
  %r =l rem %a, %b
  storel %r, %rem
  ret %q
}
