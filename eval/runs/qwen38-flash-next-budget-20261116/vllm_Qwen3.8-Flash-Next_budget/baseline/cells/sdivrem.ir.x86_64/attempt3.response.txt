

export function l $sdivrem(l %a, l %b, l %rem) {
@start:
  %zero =l sub %a, %a
  %iszero =w ceql %b, %zero
  jnz %iszero, @divzero, @checknegone
@divzero:
  storel %a, %rem
  ret %zero
@checknegone:
  %one =l add %zero, 1
  %negone =l sub %zero, %one
  %isnegone =w ceql %b, %negone
  jnz %isnegone, @negone, @normal
@negone:
  %qnegone =l sub %zero, %a
  storel %zero, %rem
  ret %qnegone
@normal:
  %qnormal =l div %a, %b
  %rnormal =l rem %a, %b
  storel %rnormal, %rem
  ret %qnormal
}
