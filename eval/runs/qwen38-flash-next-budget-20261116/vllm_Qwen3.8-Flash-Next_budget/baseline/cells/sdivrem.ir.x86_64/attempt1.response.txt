

export function l $sdivrem(l %a, l %b, l %rem) {
@start:
	%zero =l sub %a, %a
	%iszero =w ceql %b, %zero
	jnz %iszero, @divzero, @checkneg
@divzero:
	storel %a, %rem
	ret l %zero
@checkneg:
	%isneg =w ceql %b, -1
	jnz %isneg, @checkmin, @normal
@checkmin:
	%nega =l sub %zero, %a
	%ismin =w ceql %a, %nega
	jnz %ismin, @special, @normal
@special:
	storel %zero, %rem
	ret l %a
@normal:
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret l %q
}
