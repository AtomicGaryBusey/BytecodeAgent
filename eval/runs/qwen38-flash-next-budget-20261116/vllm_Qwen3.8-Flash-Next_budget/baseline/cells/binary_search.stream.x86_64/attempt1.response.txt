

binary_search:
cld
mov r10, rdx
mov r8, 0
mov r9, rsi
.Lloop:
cmp r8, r9
jae .Lnot_found
mov r11, r9
sub r11, r8
shr r11, 1
add r11, r8
mov rax, qword ptr [rdi + r11*8]
cmp rax, r10
jl .Lright
je .Lreturn
mov r9, r11
jmp .Lloop
.Lright:
lea r8, [r11 + 1]
jmp .Lloop
.Lreturn:
mov rax, r11
ret
.Lnot_found:
mov rax, -1
ret
