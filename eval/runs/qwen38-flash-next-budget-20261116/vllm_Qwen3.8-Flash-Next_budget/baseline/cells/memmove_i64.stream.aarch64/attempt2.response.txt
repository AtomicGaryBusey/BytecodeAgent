

memmove_i64:
    cbz x2, .Lret
    cmp x0, x1
    beq .Lret
    b.hi .Lbackward
    mov x3, x0
    mov x4, x1
    mov x5, x2
.Lforward:
    ldr x6, [x4]
    str x6, [x3]
    add x3, x3, #8
    add x4, x4, #8
    subs x5, x5, #1
    b.ne .Lforward
    b .Lret
.Lbackward:
    add x3, x0, x2, lsl #3
    add x4, x1, x2, lsl #3
    sub x3, x3, #8
    sub x4, x4, #8
.Lbackwardloop:
    ldr x6, [x4]
    str x6, [x3]
    sub x3, x3, #8
    sub x4, x4, #8
    subs x2, x2, #1
    b.ne .Lbackwardloop
.Lret:
    mov x0, #0
    ret
