

memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.lo .Lforward
    b.hi .Lbackward
.Ldone:
    mov x0, #0
    ret
.Lforward:
    mov x3, x0
    mov x4, x1
    mov x5, x2
.Lforwardloop:
    subs x5, x5, #1
    ldr x6, [x4, x5, lsl #3]
    str x6, [x3, x5, lsl #3]
    b.ne .Lforwardloop
    b .Ldone
.Lbackward:
    add x3, x0, x2, lsl #3
    add x4, x1, x2, lsl #3
.Lbackwardloop:
    subs x2, x2, #1
    ldr x6, [x4, x2, lsl #3]
    str x6, [x3, x2, lsl #3]
    b.ne .Lbackwardloop
    b .Ldone
