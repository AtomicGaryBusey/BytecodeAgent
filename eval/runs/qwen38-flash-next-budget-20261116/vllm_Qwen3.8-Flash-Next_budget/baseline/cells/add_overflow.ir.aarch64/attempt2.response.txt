

export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l addl %a, %b
	storel %sum, %out
	%aneg =w csltl %a, $0
	jnz %aneg, @a_neg, @a_pos
@a_pos
	%bneg =w csltl %b, $0
	jnz %bneg, @no_overflow, @check_sum_neg
@a_neg
	%bpos =w csltl %b, $0
	jnz %bpos, @check_sum_pos, @no_overflow
@check_sum_neg
	%sneg =w csltl %sum, $0
	jnz %sneg, @overflow, @no_overflow
@check_sum_pos
	%sneg2 =w csltl %sum, $0
	jnz %sneg2, @no_overflow, @overflow
@overflow
	ret $1
@no_overflow
	ret $0
}
