

isqrt:

    mov x2, xzr
    mov x3, #1
    lsl x3, x3, #32
.Lloop:
    cmp x2, x3
    b.hs .Ldone
    add x4, x2, x3
    lsr x4, x4, #1
    mul x5, x4, x4
    cmp x5, x1
    b.hi .Lelse
    add x2, x4, #1
    b .Lloop
.Lelse:
    mov x3, x4
    b .Lloop
.Ldone:
    sub x0, x2, #1
    ret
