

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start:
  %i =l copy 0
  %written =l copy 0
  jmp @outer
@outer:
  %outer_cond =w csltl %i, %n
  jnz %outer_cond, @packet, @done
@done:
  ret %written
@packet:
  %p0 =l add %src, %i
  %bi =w loadub %p0
  %ip128 =l add %i, 128
  %lim_lt_n =w csltl %ip128, %n
  jnz %lim_lt_n, @lim_i128, @lim_n
@lim_i128:
  %endidx =l copy %ip128
  jmp @lim_done
@lim_n:
  %endidx =l copy %n
  jmp @lim_done
@lim_done:
  %endp =l add %src, %endidx
  %p =l add %p0, 1
  %run =w copy 1
  jmp @run_loop
@run_loop:
  %run_lt_128 =w csltw %run, 128
  jnz %run_lt_128, @run_bound, @run_done
@run_bound:
  %p_lt_end =w csltl %p, %endp
  jnz %p_lt_end, @run_eq, @run_done
@run_eq:
  %b =w loadub %p
  %eq =w ceqw %b, %bi
  jnz %eq, @run_inc, @run_done
@run_inc:
  %p =l add %p, 1
  %run =w add %run, 1
  jmp @run_loop
@run_done:
  %run_lt3 =w csltw %run, 3
  jnz %run_lt3, @literal, @repeat
@repeat:
  %cap1 =l add %cap, 1
  %rem2 =l add %written, 2
  %fits2 =w csltl %rem2, %cap1
  jnz %fits2, @write_repeat, @fail
@write_repeat:
  %outp =l add %out, %written
  %outp1 =l add %outp, 1
  %hdr =w sub 257, %run
  storeb %hdr, %outp
  storeb %bi, %outp1
  %written =l add %written, 2
  %i =l sub %p, %src
  jmp @outer
@literal:
  %j =l copy %i
  %litlen =w copy 0
  jmp @lit_loop
@lit_loop:
  %j_lt_n =w csltl %j, %n
  jnz %j_lt_n, @lit_len, @lit_end
@lit_len:
  %lit_lt_128 =w csltw %litlen, 128
  jnz %lit_lt_128, @lit_check, @lit_end
@lit_check:
  %jp2 =l add %j, 2
  %has2 =w csltl %jp2, %n
  jnz %has2, @lit_eq01, @lit_adv
@lit_eq01:
  %p0 =l add %src, %j
  %p1 =l add %p0, 1
  %b0 =w loadub %p0
  %b1 =w loadub %p1
  %eq01 =w ceqw %b0, %b1
  jnz %eq01, @lit_eq12, @lit_adv
@lit_eq12:
  %p2 =l add %p1, 1
  %b2 =w loadub %p2
  %eq12 =w ceqw %b1, %b2
  jnz %eq12, @lit_stop, @lit_adv
@lit_adv:
  %j =l add %j, 1
  %litlen =w add %litlen, 1
  jmp @lit_loop
@lit_stop:
  jmp @lit_end
@lit_end:
  %len =l sub %j, %i
  %pkt =l add %len, 1
  %cap1 =l add %cap, 1
  %rem =l add %written, %pkt
  %fits =w csltl %rem, %cap1
  jnz %fits, @write_lit, @fail
@write_lit:
  %outp =l add %out, %written
  %hdr =w sub %litlen, 1
  storeb %hdr, %outp
  %dsrc =l add %src, %i
  %ddst =l add %outp, 1
  %dend =l add %dsrc, %len
  jmp @copy_loop
@copy_loop:
  %copy_cond =w csltl %dsrc, %dend
  jnz %copy_cond, @copy_body, @copy_done
@copy_body:
  %b =w loadub %dsrc
  storeb %b, %ddst
  %dsrc =l add %dsrc, 1
  %ddst =l add %ddst, 1
  jmp @copy_loop
@copy_done:
  %written =l add %written, %pkt
  %i =l copy %j
  jmp @outer
@fail:
  %m1 =l copy -1
  ret %m1
}
