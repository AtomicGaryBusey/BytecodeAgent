

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start:
%i =l copy 0
%written =l copy 0
jmp @outer
@outer:
%outercond =w csltl %i, %n
jnz %outercond, @packet, @done
@done:
ret %written
@packet:
%p0 =l add %src, %i
%bi =w loadub %p0
%ip128 =l add %i, 128
%limlt =w csltl %ip128, %n
jnz %limlt, @lim128, @limn
@lim128:
%endidx =l copy %ip128
jmp @limdone
@limn:
%endidx =l copy %n
jmp @limdone
@limdone:
%endp =l add %src, %endidx
%p =l add %p0, 1
%run =w copy 1
jmp @runloop
@runloop:
%runlt128 =w csltw %run, 128
jnz %runlt128, @runbound, @rundone
@runbound:
%pltend =w csltl %p, %endp
jnz %pltend, @runeq, @rundone
@runeq:
%b =w loadub %p
%eq =w ceqw %b, %bi
jnz %eq, @runinc, @rundone
@runinc:
%p =l add %p, 1
%run =w add %run, 1
jmp @runloop
@rundone:
%runlt3 =w csltw %run, 3
jnz %runlt3, @literal, @cap2
@cap2:
%cap1r =l add %cap, 1
%rem2 =l add %written, 2
%fits2 =w csltl %rem2, %cap1r
jnz %fits2, @write2, @fail
@write2:
%outp2 =l add %out, %written
%outp2b =l add %outp2, 1
%hdr2 =w sub 257, %run
storeb %hdr2, %outp2
storeb %bi, %outp2b
%written =l add %written, 2
%i =l sub %p, %src
jmp @outer
@literal:
%j =l copy %i
%litlen =w copy 0
jmp @litloop
@litloop:
%jlt =w csltl %j, %n
jnz %jlt, @litcheck, @litend
@litcheck:
%lilt128 =w csltw %litlen, 128
jnz %lilt128, @litcheckrun, @litend
@litcheckrun:
%jp2 =l add %j, 2
%has2 =w csltl %jp2, %n
jnz %has2, @liteq01, @litadv
@liteq01:
%lp0 =l add %src, %j
%lp1 =l add %lp0, 1
%lb0 =w loadub %lp0
%lb1 =w loadub %lp1
%eq01 =w ceqw %lb0, %lb1
jnz %eq01, @liteq12, @litadv
@liteq12:
%lp2 =l add %lp1, 1
%lb2 =w loadub %lp2
%eq12 =w ceqw %lb1, %lb2
jnz %eq12, @litstop, @litadv
@litadv:
%j =l add %j, 1
%litlen =w add %litlen, 1
jmp @litloop
@litstop:
jmp @litend
@litend:
%len =l sub %j, %i
%pkt =l add %len, 1
%cap1l =l add %cap, 1
%rem =l add %written, %pkt
%fits =w csltl %rem, %cap1l
jnz %fits, @write1, @fail
@write1:
%outp1 =l add %out, %written
%hdr1 =w sub %litlen, 1
storeb %hdr1, %outp1
%dsrc =l add %src, %i
%ddst =l add %outp1, 1
%dend =l add %dsrc, %len
jmp @copyloop
@copyloop:
%ccond =w csltl %dsrc, %dend
jnz %ccond, @copybody, @copydone
@copybody:
%cb =w loadub %dsrc
storeb %cb, %ddst
%dsrc =l add %dsrc, 1
%ddst =l add %ddst, 1
jmp @copyloop
@copydone:
%written =l add %written, %pkt
%i =l copy %j
jmp @outer
@fail:
%m1 =l copy -1
ret %m1
}
