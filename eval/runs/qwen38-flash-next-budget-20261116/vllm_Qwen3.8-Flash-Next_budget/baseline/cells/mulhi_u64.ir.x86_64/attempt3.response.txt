

export function l $mulhi_u64(l %a, l %b) {
@start
	%alo = and %a, 4294967295
	%ahi = shr %a, 32
	%blo = and %b, 4294967295
	%bhi = shr %b, 32

	%p0 = mul %alo, %blo
	%p1 = mul %alo, %bhi
	%p2 = mul %ahi, %blo
	%p3 = mul %ahi, %bhi

	%shift0 = shr %p0, 32
	%low1 = and %p1, 4294967295
	%low2 = and %p2, 4294967295
	%sum1 = add %shift0, %low1
	%sum2 = add %sum1, %low2
	%carry = shr %sum2, 32

	%high1 = shr %p1, 32
	%high2 = shr %p2, 32
	%acc0 = add %p3, %high1
	%acc1 = add %acc0, %high2
	%acc2 = add %acc1, %carry

	ret %acc2
}
