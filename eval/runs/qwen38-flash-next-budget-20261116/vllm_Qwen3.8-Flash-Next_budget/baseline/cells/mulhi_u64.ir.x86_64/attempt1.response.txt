

export function l $mulhi_u64(l %a, l %b) {
@start
	%alo = and %a, 4294967295
	%ahi = shr %a, 32
	%blo = and %b, 4294967295
	%bhi = shr %b, 32

	%p0 = mul %alo, %blo
	%p1 = mul %alo, %bhi
	%p2 = mul %ahi, %blo
	%p3 = mul %ahi, %bhi

	%s0 = shr %p0, 32
	%l1 = and %p1, 4294967295
	%l2 = and %p2, 4294967295
	%s1 = add %s0, %l1
	%s2 = add %s1, %l2
	%c = shr %s2, 32

	%h1 = shr %p1, 32
	%h2 = shr %p2, 32
	%r1 = add %p3, %h1
	%r2 = add %r1, %h2
	%r3 = add %r2, %c

	ret %r3
}
