@start
	%alo = andl %a, 4294967295
	%ahi = shrl %a, 32
	%blo = andl %b, 4294967295
	%bhi = shrl %b, 32

	%p0 = mull %alo, %blo
	%p1 = mull %alo, %bhi
	%p2 = mull %ahi, %blo
	%p3 = mull %ahi, %bhi

	%shift0 = shrl %p0, 32
	%low1 = andl %p1, 4294967295
	%low2 = andl %p2, 4294967295
	%sum1 = addl %shift0, %low1
	%sum2 = addl %sum1, %low2
	%carry = shrl %sum2, 32

	%high1 = shrl %p1, 32
	%high2 = shrl %p2, 32
	%acc0 = addl %p3, %high1
	%acc1 = addl %acc0, %high2
	%acc2 = addl %acc1, %carry

	ret %acc2
}
