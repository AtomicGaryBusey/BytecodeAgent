

export function l $mulhi_u64(l %a, l %b) {
@start
	%alo = and.l %a, 4294967295
	%ahi = shr.l %a, 32
	%blo = and.l %b, 4294967295
	%bhi = shr.l %b, 32

	%p0 = mul.l %alo, %blo
	%p1 = mul.l %alo, %bhi
	%p2 = mul.l %ahi, %blo
	%p3 = mul.l %ahi, %bhi

	%shift0 = shr.l %p0, 32
	%low1 = and.l %p1, 4294967295
	%low2 = and.l %p2, 4294967295
	%sum1 = add.l %shift0, %low1
	%sum2 = add.l %sum1, %low2
	%carry = shr.l %sum2, 32

	%high1 = shr.l %p1, 32
	%high2 = shr.l %p2, 32
	%acc0 = add.l %p3, %high1
	%acc1 = add.l %acc0, %high2
	%acc2 = add.l %acc1, %carry

	ret %acc2
}
