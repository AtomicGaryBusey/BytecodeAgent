

export function l $fill_down(l %a, l %n, l %start) {
@entry
jmp @loop
@loop
%i =l phi @entry 0, @body %i1
%p =l phi @entry %a, @body %p1
%v =l phi @entry %start, @body %v1
%c =w csltl %i, %n
jnz %c, @body, @done
@body
storel %v, %p
%i1 =l add %i, 1
%p1 =l add %p, 8
%v1 =l sub %v, 1
jmp @loop
@done
ret %n
}
