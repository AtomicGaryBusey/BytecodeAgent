

bubble_sort:
    cmp x1, #1
    ble .Ldone
.Louter:
    mov x9, xzr
    sub x10, x1, #1
    mov x11, xzr
.Linner:
    cmp x11, x10
    b.ge .Lend_inner
    add x12, x0, x11, lsl #3
    ldr x13, [x12]
    ldr x14, [x12, #8]
    cmp x13, x14
    b.le .Lno_swap
    str x14, [x12]
    str x13, [x12, #8]
    mov x9, #1
.Lno_swap:
    add x11, x11, #1
    b .Linner
.Lend_inner:
    cbz x9, .Ldone
    sub x1, x1, #1
    b .Louter
.Ldone:
    mov x0, #0
    ret
