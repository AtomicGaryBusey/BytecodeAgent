

bubble_sort:
    cmp rsi, 2
    jb .Lret
.Louter:
    mov rdx, rsi
    dec rdx
    xor rcx, rcx
.Linner:
    cmp rcx, rdx
    jge .Ldone_pass
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lno_swap
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], r8
.Lno_swap:
    inc rcx
    jmp .Linner
.Ldone_pass:
    dec rsi
    cmp rsi, 2
    jae .Louter
.Lret:
    xor eax, eax
    ret
