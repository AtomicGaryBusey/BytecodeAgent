

qsort_pairs:
cld

push r13
push r14
push r15

jle .Lret0
mov rax, r12
shr rax, 1
add rax, rax
lea rcx, [rbx + rax*8]
mov r8, qword ptr [rcx]
mov r9, qword ptr [rcx + 8]
xor r13, r13
xor r15, r15
mov r14, r12
dec r14
.Lpart:
cmp r15, r14
jg .Lpart_done
mov rdx, r15
add rdx, rdx
lea rdx, [rbx + rdx*8]
mov rax, qword ptr [rdx]
mov rdi, rax
cmp rdi, r8
jl .Llt
jg .Lgt
mov rax, qword ptr [rdx + 8]
cmp rax, r9
jl .Llt
jg .Lgt
inc r15
jmp .Lpart
.Llt:
mov rcx, qword ptr [rdx + 8]
mov rsi, r13
add rsi, rsi
lea rsi, [rbx + rsi*8]
mov r10, qword ptr [rsi]
mov r11, qword ptr [rsi + 8]

mov qword ptr [rsi + 8], rcx
mov qword ptr [rdx], r10
mov qword ptr [rdx + 8], r11

inc r15
jmp .Lpart
.Lgt:
mov rcx, qword ptr [rdx + 8]
mov rsi, r14
add rsi, rsi
lea rsi, [rbx + rsi*8]
mov r10, qword ptr [rsi]
mov r11, qword ptr [rsi + 8]
mov qword ptr [rsi], rdi
mov qword ptr [rsi + 8], rcx
mov qword ptr [rdx], r10
mov qword ptr [rdx + 8], r11
dec r14
jmp .Lpart
.Lpart_done:
cmp r13, 1
jle .Lright
mov rdi, rbx
mov rsi, r13
call qsort_pairs
.Lright:
mov rdx, r14
add rdx, 1

mov rsi, r12
sub rsi, r14
sub rsi, 1

jle .Lret0
call qsort_pairs
.Lret0:

pop r15
pop r14
pop r13
pop r12
pop rbx
