

export function l $bubble_sort(l %a, l %n) {
@start:
	%zero = subl %n, %n
	%i0 = addl %zero, %zero
	jmp @outer_cond
@outer_cond:
	%i = phi @start %i0, @outer_after_inner %i_next
	%ip1 = addl %i, 1
	%c1 = csltl %ip1, %n
	jnz %c1 @outer_body @outer_end
@outer_body:
	%j0 = addl %zero, %zero
	jmp @inner_cond
@inner_cond:
	%j = phi @outer_body %j0, @inner_after %j_next
	%jp1 = addl %j, 1
	%nmi = subl %n, %i
	%c2 = csltl %jp1, %nmi
	jnz %c2 @inner_body @outer_after_inner
@inner_body:
	%off1 = slll %j, 3
	%p1 = addl %a, %off1
	%v1 = loadl %p1
	%off2 = slll %jp1, 3
	%p2 = addl %a, %off2
	%v2 = loadl %p2
	%c3 = csltl %v2, %v1
	jnz %c3 @swap @inner_after
@swap:
	storel %v2, %p1
	storel %v1, %p2
	jmp @inner_after
@inner_after:
	%j_next = addl %j, 1
	jmp @inner_cond
@outer_after_inner:
	%i_next = addl %i, 1
	jmp @outer_cond
@outer_end:
	ret %zero
}
