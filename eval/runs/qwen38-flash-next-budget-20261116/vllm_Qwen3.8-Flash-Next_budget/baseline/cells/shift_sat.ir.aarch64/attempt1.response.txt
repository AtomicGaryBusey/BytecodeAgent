

export function l $shift_sat(l %x, l %n) {
@start
  %cmp =w csltl %n, 64
  jnz %cmp, @small, @big
@small
  %left_small =l shl %x, %n
  %right_small =l asr %x, %n
  %res_small =l add %left_small, %right_small
  ret %res_small
@big
  %zero_big =l xor %x, %x
  %right_big =l asr %x, 63
  %res_big =l add %zero_big, %right_big
  ret %res_big
}
