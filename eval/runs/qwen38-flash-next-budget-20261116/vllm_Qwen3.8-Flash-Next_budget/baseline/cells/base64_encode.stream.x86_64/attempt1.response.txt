

base64_encode:
    test rsi, rsi
    jle .Lret0
    push rbx
    push r12
    push r13
    push r14
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rdx
.Lgroup3:
    cmp r12, 3
    jl .Lfinal
    movzx eax, byte ptr [rbx]
    movzx edx, byte ptr [rbx + 1]
    movzx ecx, byte ptr [rbx + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx
    mov r10d, eax
    xor r11d, r11d
    mov r8d, 4
    mov r9d, 18
    jmp .Lemit_chars
.Lfinal:
    test r12, r12
    je .Lret
    movzx eax, byte ptr [rbx]
    xor edx, edx
    cmp r12, 2
    jl .Lcombine_final
    movzx edx, byte ptr [rbx + 1]
.Lcombine_final:
    mov ecx, edx
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    mov r10d, eax
    mov r11d, 1
    mov r8d, r12d
    add r8d, 1
    mov r9d, 18
.Lemit_chars:
    mov edx, r10d
    mov ecx, r9d
    shr edx, cl
    and edx, 63
    mov eax, edx
    add eax, 65
    mov ecx, edx
    add ecx, 71
    cmp edx, 26
    cmovae eax, ecx
    mov ecx, edx
    sub ecx, 4
    cmp edx, 52
    cmovae eax, ecx
    mov ecx, 43
    cmp edx, 62
    cmovae eax, ecx
    mov ecx, 47
    cmp edx, 63
    cmove eax, ecx
    mov byte ptr [r13], al
    inc r13
    sub r9d, 6
    dec r8d
    jnz .Lemit_chars
    test r11d, r11d
    jne .Lfinal_pad
    add rbx, 3
    sub r12, 3
    jmp .Lgroup3
.Lfinal_pad:
    test r12, r12
    je .Lret
    cmp r12, 2
    je .Lpad1
    mov byte ptr [r13], 61
    inc r13
    mov byte ptr [r13], 61
    inc r13
    jmp .Lret
.Lpad1:
    mov byte ptr [r13], 61
    inc r13
.Lret:
    mov rax, r13
    sub rax, r14
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lret0:
    xor eax, eax
    ret
