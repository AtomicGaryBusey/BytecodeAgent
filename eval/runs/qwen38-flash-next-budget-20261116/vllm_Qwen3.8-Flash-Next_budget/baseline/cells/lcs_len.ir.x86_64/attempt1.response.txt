

export function l $lcs_len(l %x, l %m, l %n, l %row) {
@start:
  %z =l add 0, 0
  storel %z, %row
  %j1 =l add %z, 1
  jmp @init_head

@init_head:
  %ij =l phi %j1 @start, %ij_next @init_body
  %i_start =l add %z, 0
  %lt = csltl %n, %ij
  jnz %lt, @outer_head, @init_body

@init_body:
  %ij8 =l mul %ij, 8
  %rp =l add %row, %ij8
  storel %z, %rp
  %ij_next =l add %ij, 1
  jmp @init_head

@outer_head:
  %i =l phi %i_start @init_head, %i_next @outer_tail
  %cont_i = csltl %i, %m
  jnz %cont_i, @outer_body, @done

@outer_body:
  %xp =l add %x, %i
  %xb =w loadub %xp
  %j_start =l add %z, 1
  %diag_init =l add %z, 0
  jmp @inner_head

@inner_head:
  %j =l phi %j_start @outer_body, %j_next @inner_tail
  %diag =l phi %diag_init @outer_body, %diag_next @inner_tail
  %gt_j = csltl %n, %j
  jnz %gt_j, @outer_tail, @inner_body

@inner_body:
  %j8 =l mul %j, 8
  %rp_j =l add %row, %j8
  %up =l loadl %rp_j
  %jm =l sub %j, 1
  %yp =l add %y, %jm
  %yb =w loadub %yp
  %eqb = ceqw %xb, %yb
  jnz %eqb, @eq_block, @neq_block

@eq_block:
  %new_eq =l add %diag, 1
  %d_eq =l add %up, 0
  jmp @row_store

@neq_block:
  %jm8 =l mul %jm, 8
  %rp_jm1 =l add %row, %jm8
  %left =l loadl %rp_jm1
  %gt_left = csltl %left, %up
  jnz %gt_left, @keep_block, @take_block

@keep_block:
  %new_keep =l add %up, 0
  %d_keep =l add %up, 0
  jmp @row_store

@take_block:
  %new_take =l add %left, 0
  %d_take =l add %up, 0
  jmp @row_store

@row_store:
  %newrow =l phi %new_eq @eq_block, %new_keep @keep_block, %new_take @take_block
  %diag2 =l phi %d_eq @eq_block, %d_keep @keep_block, %d_take @take_block
  storel %newrow, %rp_j
  jmp @inner_tail

@inner_tail:
  %j_next =l add %j, 1
  %diag_next =l add %diag2, 0
  jmp @inner_head

@outer_tail:
  %i_next =l add %i, 1
  jmp @outer_head

@done:
  %n_neg = csltl %n, 0
  jnz %n_neg, @done_zero, @done_n

@done_zero:
  %res0 =l loadl %row
  jmp @ret

@done_n:
  %n8 =l mul %n, 8
  %rp_n =l add %row, %n8
  %resn =l loadl %rp_n
  jmp @ret

@ret:
  %res =l phi %res0 @done_zero, %resn @done_n
  ret %res
}
