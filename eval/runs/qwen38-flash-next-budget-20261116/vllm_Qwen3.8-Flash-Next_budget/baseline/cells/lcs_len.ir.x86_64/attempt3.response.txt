

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start:
%zero =l add 0, 0
%one =l add 0, 1
storel %zero, %row
jmp @init_head
@init_head:
%jinit =l phi %one @start, %jinit_next @init_body
%done_init =w csltl %n, %jinit
jnz %done_init, @init_done, @init_body
@init_body:
%jinit8 =l mul %jinit, 8
%rp_init =l add %row, %jinit8
storel %zero, %rp_init
%jinit_next =l add %jinit, 1
jmp @init_head
@init_done:
%i1 =l add %one, 0
jmp @outer_head
@outer_head:
%i =l phi %i1 @init_done, %i_next @outer_tail
%done_outer =w csltl %m, %i
jnz %done_outer, @done, @outer_body
@outer_body:
%im =l sub %i, 1
%xp =l add %x, %im
%xb =w loadub %xp
%j1 =l add %one, 0
%diag0 =l add %zero, 0
jmp @inner_head
@inner_head:
%j =l phi %j1 @outer_body, %j_next @inner_tail
%diag =l phi %diag0 @outer_body, %diag_next @inner_tail
%done_inner =w csltl %n, %j
jnz %done_inner, @outer_tail, @inner_body
@inner_body:
%j8 =l mul %j, 8
%rp_j =l add %row, %j8
%up =l loadl %rp_j
%jm =l sub %j, 1
%yp =l add %y, %jm
%yb =w loadub %yp
%eqb =w ceqw %xb, %yb
jnz %eqb, @eq_block, @neq_block
@eq_block:
%new_eq =l add %diag, 1
%diag_eq =l add %up, 0
jmp @row_store
@neq_block:
%jm8 =l mul %jm, 8
%rp_jm =l add %row, %jm8
%left =l loadl %rp_jm
%left_less =w csltl %left, %up
jnz %left_less, @keep_block, @take_block
@keep_block:
%new_keep =l add %up, 0
%diag_keep =l add %up, 0
jmp @row_store
@take_block:
%new_take =l add %left, 0
%diag_take =l add %up, 0
jmp @row_store
@row_store:
%newrow =l phi %new_eq @eq_block, %new_keep @keep_block, %new_take @take_block
%diag_phi =l phi %diag_eq @eq_block, %diag_keep @keep_block, %diag_take @take_block
storel %newrow, %rp_j
jmp @inner_tail
@inner_tail:
%j_next =l add %j, 1
%diag_next =l add %diag_phi, 0
jmp @inner_head
@outer_tail:
%i_next =l add %i, 1
jmp @outer_head
@done:
%n8 =l mul %n, 8
%rp_n =l add %row, %n8
%res =l loadl %rp_n
ret %res
}
