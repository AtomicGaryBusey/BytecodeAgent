

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start:
%zero =l add 0, 0
storel %zero, %row
%jinit0 =l add 0, 1
jmp @init_head
@init_head:
%jinit =l phi %jinit0 @start, %jinit_next @init_body
%done_init =w csltl %n, %jinit
jnz %done_init, @init_done, @init_body
@init_body:
%jinit8 =l mul %jinit, 8
%rp_init =l add %row, %jinit8
storel %zero, %rp_init
%jinit_next =l add %jinit, 1
jmp @init_head
@init_done:
%i0 =l add 0, 1
jmp @outer_head
@outer_head:
%i =l phi %i0 @init_done, %i_next @outer_tail
%done_outer =w csltl %m, %i
jnz %done_outer, @finish, @outer_body
@outer_body:
%im =l sub %i, 1
%xbp =l add %x, %im
%xb =w loadub %xbp
%diag0 =l add %zero, 0
%j0 =l add 0, 1
jmp @inner_head
@inner_head:
%j =l phi %j0 @outer_body, %j_next_eq @eq, %j_next_keep @keep, %j_next_take @take
%diag =l phi %diag0 @outer_body, %diag_next_eq @eq, %diag_next_keep @keep, %diag_next_take @take
%done_inner =w csltl %n, %j
jnz %done_inner, @outer_tail, @inner_body
@inner_body:
%j8 =l mul %j, 8
%rp_j =l add %row, %j8
%up =l loadl %rp_j
%jm =l sub %j, 1
%yp =l add %y, %jm
%yb =w loadub %yp
%eqb =w ceqw %xb, %yb
jnz %eqb, @eq, @neq
@eq:
%new_eq =l add %diag, 1
storel %new_eq, %rp_j
%j_next_eq =l add %j, 1
%diag_next_eq =l add %up, 0
jmp @inner_head
@neq:
%jm8 =l mul %jm, 8
%rp_jm =l add %row, %jm8
%left =l loadl %rp_jm
%less =w csltl %left, %up
jnz %less, @keep, @take
@keep:
storel %up, %rp_j
%j_next_keep =l add %j, 1
%diag_next_keep =l add %up, 0
jmp @inner_head
@take:
storel %left, %rp_j
%j_next_take =l add %j, 1
%diag_next_take =l add %up, 0
jmp @inner_head
@outer_tail:
%i_next =l add %i, 1
jmp @outer_head
@finish:
%n8 =l mul %n, 8
%rp_n =l add %row, %n8
%res =l loadl %rp_n
ret %res
}
