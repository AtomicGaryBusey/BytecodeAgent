

isqrt:
    cld
    xor r8, r8
    xor r10, r10
    xor r9d, r9d
    not r9d
.Lloop:
    cmp r8, r9
    ja .Ldone
    mov rcx, r8
    add rcx, r9
    shr rcx, 1
    mov rax, rcx
    mul rcx
    test rdx, rdx
    jne .Lgreater
    cmp rax, rdi
    ja .Lgreater
    mov r10, rcx
    mov r8, rcx
    inc r8
    jmp .Lloop
.Lgreater:
    sub rcx, 1
    jb .Ldone
    mov r9, rcx
    jmp .Lloop
.Ldone:
    mov rax, r10
    ret
