

isqrt:
    cld
    xor r8, r8
    xor rsi, rsi
    xor r9d, r9d
    not r9d
.Lloop:
    cmp rsi, r9
    ja .Ldone
    mov rcx, rsi
    add rcx, r9
    shr rcx, 1
    mov rax, rcx
    mul rcx
    test rdx, rdx
    jne .Lgreater
    cmp rax, rdi
    ja .Lgreater
    mov r8, rcx
    inc rsi
    jmp .Lloop
.Lgreater:
    dec r9
    jmp .Lloop
.Ldone:
    mov rax, r8
    ret
