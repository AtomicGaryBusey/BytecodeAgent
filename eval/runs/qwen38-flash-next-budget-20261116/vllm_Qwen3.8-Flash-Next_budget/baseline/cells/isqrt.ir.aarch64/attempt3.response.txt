

export function l $isqrt(l %.n) {
@start
    %.zero = addl 0, 0
    %.one = addl 0, 1
    %.thirtytwo = addl 0, 32
    %.wzero = addw 0, 0
    %.high0 = lsll %.one, %.thirtytwo
    jmp @loop

@loop
    %.low = phi %.zero, @start, %.low_lmid, @lowmid, %.low_hmid, @highmid
    %.high = phi %.high0, @start, %.high_lmid, @lowmid, %.high_hmid, @highmid
    %.diff = subl %.high, %.low
    %.stop = ceql %.diff, %.one
    jnz %.stop, @end, @body

@body
    %.half = lsrl %.diff, %.one
    %.mid = addl %.low, %.half
    %.q = divul %.n, %.mid
    %.qhigh = lsrl %.q, %.thirtytwo
    %.small = ceql %.qhigh, %.zero
    jnz %.small, @qsmall, @qbig

@qbig
    jmp @lowmid

@qsmall
    %.less = csltl %.q, %.mid
    %.ok = ceqw %.less, %.wzero
    jnz %.ok, @lowmid, @highmid

@lowmid
    %.low_lmid = addl %.mid, %.zero
    %.high_lmid = addl %.high, %.zero
    jmp @loop

@highmid
    %.low_hmid = addl %.low, %.zero
    %.high_hmid = addl %.mid, %.zero
    jmp @loop

@end
    ret %.low
}
