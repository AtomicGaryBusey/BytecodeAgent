

export function l $isqrt(l %.n) {
@start
    %.zero = sub %.n, %.n
    %.iszero = ceql %.n, %.zero
    jnz %.iszero, @zero, @init
@zero
    ret %.zero
@init
    %.one = udiv %.n, %.n
    %.two = add %.one, %.one
    %.x_init = add %.zero, 4294967296
    jmp @loop
@loop
    %.x = phi %.x_init, @init, %.x_upd, @update
    %.q = udiv %.n, %.x
    %.sum = add %.x, %.q
    %.y = udiv %.sum, %.two
    %.lt = csltl %.y, %.x
    jnz %.lt, @update, @done
@update
    %.x_upd = add %.y, %.zero
    jmp @loop
@done
    ret %.x
}
