

export function l $isqrt(l %n) {
@start
    %one =l copy $1
    %s32 =l copy $32
    %hi0 =l shll %one, %s32
    %lo0 =l copy $0
    %s63 =l copy $63
    %min =l shll %one, %s63
    jmp @loop
@loop
    %lo =l phi @start %lo0, @ok %lo_ok, @bad %lo_bad
    %hi =l phi @start %hi0, @ok %hi_ok, @bad %hi_bad
    %lt =w csltl %lo, %hi
    jnz %lt, @body, @done
@body
    %diff =l subl %hi, %lo
    %half =l divl %diff, $2
    %mid =l addl %lo, %half
    %prod =l mull %mid, %mid
    %pc =l bxorl %prod, %min
    %nc =l bxorl %n, %min
    %ltu =w csltl %pc, %nc
    jnz %ltu, @ok, @eq
@eq
    %eqv =w ceql %prod, %n
    jnz %eqv, @ok, @bad
@ok
    %lo_ok =l addl %mid, $1
    %hi_ok =l copy %hi
    jmp @loop
@bad
    %lo_bad =l copy %lo
    %hi_bad =l copy %mid
    jmp @loop
@done
    %ans =l subl %lo, $1
    ret %ans
}
