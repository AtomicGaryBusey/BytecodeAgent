

export function l $isqrt(l %n) {
@start
    %maxpos =l copy 9223372036854775807
    %one =l copy 1
    %two =l copy 2

    %ntemp =l add %n, %maxpos
    %nneg =l add %ntemp, %one

    %lo0 =l copy 0
    %hi0 =l copy 4294967295

    jmp @loop
@loop
    %lo =l phi @start %lo0, @ok %look, @bad %lobad
    %hi =l phi @start %hi0, @ok %hiook, @bad %hibad

    %done =w csltl %hi, %lo
    jnz %done, @done, @body
@body
    %diff =l sub %hi, %lo
    %half =l div %diff, %two
    %mid =l add %lo, %half

    %prod =l mul %mid, %mid
    %ptemp =l add %prod, %maxpos
    %pneg =l add %ptemp, %one

    %badw =w csltl %nneg, %pneg
    jnz %badw, @bad, @ok
@ok
    %look =l add %mid, %one
    %hiook =l copy %hi
    jmp @loop
@bad
    %lobad =l copy %lo
    %hibad =l sub %mid, %one
    jmp @loop
@done
    ret %hi
}
