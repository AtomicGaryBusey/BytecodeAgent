

export function l $isqrt(l $n) {
@start:
    %lo =l copy $0
    %hi =l copy $4294967295
    %n_neg =w csltl %n, $0
@loop:
    %diff =l subl %hi, %lo
    %bad =w csltl %diff, $0
    jnz %bad, @done, @body
@body:
    %half =l divl %diff, $2
    %mid =l addl %lo, %half
    %prod =l mull %mid, %mid
    %prod_neg =w csltl %prod, $0
    jnz %prod_neg, @prod_high, @prod_low
@prod_low:
    jnz %n_neg, @square_ok, @signed_compare
@prod_high:
    jnz %n_neg, @signed_compare, @square_bad
@signed_compare:
    %lt =w csltl %prod, %n
    jnz %lt, @square_ok, @signed_eq
@signed_eq:
    %eq =w ceql %prod, %n
    jnz %eq, @square_ok, @square_bad
@square_ok:
    %lo =l addl %mid, $1
    jmp @loop
@square_bad:
    %hi =l subl %mid, $1
    jmp @loop
@done:
    ret %hi
}
