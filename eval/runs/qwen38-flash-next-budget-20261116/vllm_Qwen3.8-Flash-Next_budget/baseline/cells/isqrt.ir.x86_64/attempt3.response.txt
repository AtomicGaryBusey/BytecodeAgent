

export function l $isqrt(l %n) {
@start
    %lo0 =l copy 0
    %one =l copy 1
    %two =l addl %one, %one
    %hi0 =l shll %one, 32
    %min =l shll %one, 63
    jmp @loop
@loop
    %lo =l phi @start %lo0, @ok %look, @bad %lobad
    %hi =l phi @start %hi0, @ok %hiook, @bad %hibad
    %lt =w csltl %lo, %hi
    jnz %lt, @body, @done
@body
    %diff =l subl %hi, %lo
    %half =l divl %diff, %two
    %mid =l addl %lo, %half
    %prod =l mull %mid, %mid
    %pc =l bxorl %prod, %min
    %nc =l bxorl %n, %min
    %nl =w csltl %nc, %pc
    jnz %nl, @bad, @ok
@ok
    %look =l addl %mid, %one
    %hiook =l copy %hi
    jmp @loop
@bad
    %lobad =l copy %lo
    %hibad =l copy %mid
    jmp @loop
@done
    %ans =l subl %lo, %one
    ret %ans
}
