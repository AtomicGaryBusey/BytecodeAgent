

crc32:
    cld
    xor edx, edx
    test rsi, rsi
    jle .Lret
    mov edx, -1
.Lbyte:
    movzx eax, byte ptr [rdi]
    inc rdi
    xor edx, eax
    mov r8d, 8
.Lbit:
    test dl, 1
    je .Lbit_zero
    shr edx, 1
    xor edx, 0xEDB88320
    jmp .Lbit_next
.Lbit_zero:
    shr edx, 1
.Lbit_next:
    dec r8d
    jne .Lbit
    dec rsi
    jne .Lbyte
    xor edx, -1
.Lret:
    mov eax, edx
    ret
