.Lmp_i:

  b.ge .Lmp_scan
  ldr x5, [x9, x15, lsl #3]

.Lmp_j:

  b.ge .Lmp_i_next
  ldr x7, [x11, x16, lsl #3]

  umulh x7, x5, x7

  ldr x1, [x13, x0, lsl #3]

  str x1, [x13, x0, lsl #3]
  adc x7, x7, xzr
  cbz x7, .Lmp_j_next

  ldr x1, [x13, x0, lsl #3]
  adds x1, x1, x7
  str x1, [x13, x0, lsl #3]
  adc x2, xzr, xzr

  b.ge .Lmp_j_next
  ldr x1, [x13, x0, lsl #3]
  adds x1, x1, x2
  str x1, [x13, x0, lsl #3]
  adc x2, xzr, xzr
  cbz x2, .Lmp_j_next

.Lmp_j_next:

  b .Lmp_j
.Lmp_i_next:

  b .Lmp_i

.Lmp_scan:
  mov x0, x14
  subs x0, x0, #1
  b.mi .Lmp_ret0
.Lmp_scan_loop:
  ldr x1, [x13, x0, lsl #3]
  cbnz x1, .Lmp_ret_significant
  subs x0, x0, #1
  b.mi .Lmp_ret0
  b .Lmp_scan_loop
.Lmp_ret_significant:

  ret
.Lmp_ret0:
