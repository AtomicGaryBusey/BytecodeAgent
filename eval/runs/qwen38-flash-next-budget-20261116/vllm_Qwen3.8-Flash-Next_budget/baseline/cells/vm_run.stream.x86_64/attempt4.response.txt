

vm_run:
cld
mov r11, rcx
xor r8, r8
xor r9, r9
.Lloop:
cmp r9, rsi
jge .Ldone
lea r10, [rdi + r9*8]
mov rax, qword ptr [r10]
cmp rax, 11
jg .Lother
cmp rax, 0
je .Ldone
cmp rax, 1
je .Lpush
cmp rax, 2
je .Ladd
cmp rax, 3
je .Lsub
cmp rax, 4
je .Lmul
cmp rax, 5
je .Ldup
cmp rax, 6
je .Lswap
cmp rax, 7
je .Ldrop
cmp rax, 8
je .Lover
cmp rax, 9
je .Llt
cmp rax, 10
je .Ljz
cmp rax, 11
je .Ljmp
.Lother:
inc r9
jmp .Lloop
.Lpush:
mov rax, qword ptr [r10 + 8]
mov qword ptr [rdx + r8*8], rax
inc r8
add r9, 2
jmp .Lloop
.Ladd:
mov rax, qword ptr [rdx + r8*8 - 16]
mov rcx, qword ptr [rdx + r8*8 - 8]
add rax, rcx
mov qword ptr [rdx + r8*8 - 16], rax
sub r8, 2
inc r9
jmp .Lloop
.Lsub:
mov rax, qword ptr [rdx + r8*8 - 16]
mov rcx, qword ptr [rdx + r8*8 - 8]
sub rax, rcx
mov qword ptr [rdx + r8*8 - 16], rax
sub r8, 2
inc r9
jmp .Lloop
.Lmul:
mov rax, qword ptr [rdx + r8*8 - 16]
mov rcx, qword ptr [rdx + r8*8 - 8]
imul rax, rcx
mov qword ptr [rdx + r8*8 - 16], rax
sub r8, 2
inc r9
jmp .Lloop
.Ldup:
mov rax, qword ptr [rdx + r8*8 - 8]
mov qword ptr [rdx + r8*8], rax
inc r8
inc r9
jmp .Lloop
.Lswap:
mov rax, qword ptr [rdx + r8*8 - 8]
mov rcx, qword ptr [rdx + r8*8 - 16]
mov qword ptr [rdx + r8*8 - 16], rax
mov qword ptr [rdx + r8*8 - 8], rcx
inc r9
jmp .Lloop
.Ldrop:
dec r8
inc r9
jmp .Lloop
.Lover:
mov rax, qword ptr [rdx + r8*8 - 16]
mov qword ptr [rdx + r8*8], rax
inc r8
inc r9
jmp .Lloop
.Llt:
mov rax, qword ptr [rdx + r8*8 - 16]
mov rcx, qword ptr [rdx + r8*8 - 8]
cmp rax, rcx
mov rax, 0
setl al
mov qword ptr [rdx + r8*8 - 16], rax
sub r8, 2
inc r9
jmp .Lloop
.Ljz:
mov rax, qword ptr [rdx + r8*8 - 8]
dec r8
mov r10, qword ptr [r10 + 8]
test rax, rax
jz .Ljz_true
add r9, 2
jmp .Lloop
.Ljz_true:
mov r9, r10
jmp .Lloop
.Ljmp:
mov r9, qword ptr [r10 + 8]
jmp .Lloop
.Ldone:
mov r10, r8
.Lclear:
cmp r10, r11
jge .Ltop
mov qword ptr [rdx + r10*8], 0
inc r10
jmp .Lclear
.Ltop:
test r8, r8
jz .Lempty
mov rax, qword ptr [rdx + r8*8 - 8]
ret
.Lempty:
xor rax, rax
ret
