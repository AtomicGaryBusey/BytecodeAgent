

export function l $gcd(l %a0, l %b0) {
@start:
	%a_init = copy %a0
	%b_init = copy %b0
	jmp @loop
@loop:
	%a = phi @start, %a_init, @body, %a_new
	%b = phi @start, %b_init, @body, %b_new
	%bz = ceql %b, 0
	jnz %bz, @done, @body
@body:
	%r = reml %a, %b
	%a_new = copy %b
	%b_new = copy %r
	jmp @loop
@done:
	ret %a
}
