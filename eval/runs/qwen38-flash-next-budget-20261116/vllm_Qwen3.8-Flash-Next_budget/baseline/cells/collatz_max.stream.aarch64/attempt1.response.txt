

collatz_max:
mov x1, x0
mov x0, #0
mov x2, #1
.Louter:
cmp x2, x1
b.gt .Ldone
mov x3, x2
mov x4, #0
.Linner:
cmp x3, #1
b.eq .Linner_done
tbnz x3, #0, .Lodd
lsr x3, x3, #1
add x4, x4, #1
b .Linner
.Lodd:
add x3, x3, x3, lsl #1
add x3, x3, #1
add x4, x4, #1
b .Linner
.Linner_done:
cmp x4, x0
csel x0, x4, x0, gt
add x2, x2, #1
b .Louter
.Ldone:
ret
