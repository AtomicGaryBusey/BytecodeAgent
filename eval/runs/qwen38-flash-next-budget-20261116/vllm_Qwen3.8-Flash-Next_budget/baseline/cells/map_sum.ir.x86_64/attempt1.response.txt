

export function l $map_sum(l %a, l %n, l %f) {
@start
    %zero =l copy 0
    jmp @loop
@loop
    %sum =l phi @start %zero, @body %sum2
    %i =l phi @start %zero, @body %i2
    %p =l phi @start %a, @body %p2
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %x =l loadl %p
    %y =l call %f(l %x)
    %sum2 =l add %sum, %y
    %p2 =l add %p, 8
    %i2 =l add %i, 1
    jmp @loop
@done
    %res =l copy %sum
    ret %res
}
