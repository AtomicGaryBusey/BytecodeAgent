

export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry:
  %x0 =l loadl %a
  %y0 =l call %f(l %x0)
  %min0 =l copy %y0
  %max0 =l copy %y0
  %sum0 =l copy %y0
  %odd0 =l and %y0, 1
  %cnt0 =l copy %odd0
  %i0 =l copy 1
  jmp @loop

@loop:
  %i =l phi @entry %i0, @after_max %i_next
  %min =l phi @entry %min0, @after_max %min_in
  %max =l phi @entry %max0, @after_max %max_next
  %sum =l phi @entry %sum0, @after_max %sum_in
  %cnt =l phi @entry %cnt0, @after_max %cnt_in
  %cond =w csltl %i, %n
  jnz %cond, @body, @exit

@body:
  %off =l mul %i, 8
  %ap =l add %a, %off
  %x =l loadl %ap
  %y =l call %f(l %x)
  %sum_next =l add %sum, %y
  %odd =l and %y, 1
  %cnt_next =l add %cnt, %odd
  %y_lt_min =w csltl %y, %min
  jnz %y_lt_min, @set_min_y, @keep_min

@set_min_y:
  %min_t =l copy %y
  jmp @after_min

@keep_min:
  %min_f =l copy %min
  jmp @after_min

@after_min:
  %min_next =l phi @set_min_y %min_t, @keep_min %min_f
  %max_lt_y =w csltl %max, %y
  jnz %max_lt_y, @set_max_y, @keep_max

@set_max_y:
  %max_t =l copy %y
  jmp @after_max

@keep_max:
  %max_f =l copy %max
  jmp @after_max

@after_max:
  %max_next =l phi @set_max_y %max_t, @keep_max %max_f
  %min_in =l copy %min_next
  %sum_in =l copy %sum_next
  %cnt_in =l copy %cnt_next
  %i_next =l add %i, 1
  jmp @loop

@exit:
  storel %min, %out
  %op1 =l add %out, 8
  storel %max, %op1
  %op2 =l add %out, 16
  storel %sum, %op2
  ret %cnt
}
