

export function l $crc32(l %p, l %n) {
@start:
%zero =w copy 0
%one =w copy 1
%allones =w sub %zero, %one
%init_crc =w copy %allones
%poly0 =w copy 237
%poly1 =w shl %poly0, 8
%poly2 =w copy 184
%poly3 =w or %poly1, %poly2
%poly4 =w shl %poly3, 8
%poly5 =w copy 131
%poly6 =w or %poly4, %poly5
%poly7 =w shl %poly6, 8
%poly8 =w copy 32
%poly =w or %poly7, %poly8
%zero_l =l copy 0
%eight_l =l copy 8
%init_cur =l copy %p
jmp @outer
@outer:
%crc =w phi @start %init_crc, @inner_done %back_crc
%cur =l phi @start %init_cur, @inner_done %back_cur
%i =l phi @start %zero_l, @inner_done %back_i
%done =w ceql %i, %n
jnz %done, @done
@outer_body:
%byte =w loadub %cur
%crc_byte =w xor %crc, %byte
%j0 =l copy 0
jmp @inner
@inner:
%j =l phi @outer_body %j0, @inner_body %j_next
%crc_in =w phi @outer_body %crc_byte, @inner_body %crc_next
%itest =w ceql %j, %eight_l
jnz %itest, @inner_done
@inner_body:
%lsb =w and %crc_in, 1
%mask =w sub %zero, %lsb
%sh =w shr %crc_in, 1
%pol =w and %mask, %poly
%crc_next =w xor %sh, %pol
%j_next =l add %j, 1
jmp @inner
@inner_done:
%back_crc =w copy %crc_in
%back_cur =l add %cur, 1
%back_i =l add %i, 1
jmp @outer
@done:
%final =w xor %crc, %allones
%res =l extuw %final
ret %res
}
