

export function l $crc32(l %p, l %n) {
@start:
%zero =w copy 0
%one =w copy 1
%allones =w sub %zero, %one
%crc_init =w copy %allones
%poly_a =w copy 237
%poly_b =w shl %poly_a, 8
%poly_c =w copy 184
%poly_d =w or %poly_b, %poly_c
%poly_e =w shl %poly_d, 8
%poly_f =w copy 131
%poly_g =w or %poly_e, %poly_f
%poly_h =w shl %poly_g, 8
%poly_i =w copy 32
%poly =w or %poly_h, %poly_i
%zero_l =l copy 0
%eight_l =l copy 8
jmp @outer
@outer:
%crc =w phi @start %crc_init, @inner_done %crc_out
%cur =l phi @start %p, @inner_done %cur_out
%i =l phi @start %zero_l, @inner_done %i_out
%done =w ceql %i, %n
jnz %done, @done, @outer_body
@outer_body:
%byte =w loadub %cur
%crc_x =w xor %crc, %byte
%j_start =l copy 0
jmp @inner
@inner:
%j =l phi @outer_body %j_start, @inner_body %j_inc
%cin =w phi @outer_body %crc_x, @inner_body %cnext
%bdone =w ceql %j, %eight_l
jnz %bdone, @inner_done, @inner_body
@inner_body:
%lsb =w and %cin, 1
%mask =w sub %zero, %lsb
%sh =w shr %cin, 1
%pm =w and %mask, %poly
%cnext =w xor %sh, %pm
%j_inc =l add %j, 1
jmp @inner
@inner_done:
%crc_out =w copy %cin
%cur_out =l add %cur, 1
%i_out =l add %i, 1
jmp @outer
@done:
%final =w xor %crc, %allones
%res =l extuw %final
ret %res
}
