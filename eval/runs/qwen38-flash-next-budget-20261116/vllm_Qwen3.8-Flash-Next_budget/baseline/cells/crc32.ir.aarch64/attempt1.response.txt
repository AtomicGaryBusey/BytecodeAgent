

export function l $crc32(l %p, l %n) {
@start:
	%zero =w copy 0
	%one =w copy 1
	%eight =w copy 8
	%allones =w sub %zero, %one
	%crc =w copy %allones
	%i =l copy 0
	%cur =l copy %p
	%poly0 =w copy 237
	%poly1 =w shl %poly0, 8
	%poly2 =w copy 184
	%poly3 =w or %poly1, %poly2
	%poly4 =w shl %poly3, 8
	%poly5 =w copy 131
	%poly6 =w or %poly4, %poly5
	%poly7 =w shl %poly6, 8
	%poly8 =w copy 32
	%poly =w or %poly7, %poly8
	jmp @outer
@outer:
	%odone =w ceql %i, %n
	jnz %odone, @done
	%byte =w loadub %cur
	%crc =w xor %crc, %byte
	%j =w copy 0
	jmp @inner
@inner:
	%itest =w ceqw %j, %eight
	jnz %itest, @innerdone
	%lsb =w and %crc, 1
	%mask =w sub %zero, %lsb
	%sh =w shr %crc, 1
	%pol =w and %mask, %poly
	%crc =w xor %sh, %pol
	%j =w add %j, 1
	jmp @inner
@innerdone:
	%cur =l add %cur, 1
	%i =l add %i, 1
	jmp @outer
@done:
	%crc =w xor %crc, %allones
	%res =l extuw %crc
	ret %res
}
