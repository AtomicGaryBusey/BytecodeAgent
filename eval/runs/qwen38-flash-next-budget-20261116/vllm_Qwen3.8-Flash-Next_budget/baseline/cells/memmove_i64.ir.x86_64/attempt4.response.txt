

export function l $memmove_i64(l %dst, l %src, l %n) {
@start
%is_zero =w ceql %n, 0
jnz %is_zero, @ret0, @check
@check
%z =l copy 0
%nm1 =l sub %n, 1
%dst_lt_src =w csltl %dst, %src
jnz %dst_lt_src, @bwd_test, @fwd_test
@fwd_test
%fi =l phi @check %z, @fwd_body %fi_next
%f_go =w csltl %fi, %n
jnz %f_go, @fwd_body, @ret0
@fwd_body
%f_idx =l mul %fi, 8
%f_src =l add %src, %f_idx
%f_dst =l add %dst, %f_idx
%f_val =l loadl %f_src
storel %f_val, %f_dst
%fi_next =l add %fi, 1
jmp @fwd_test
@bwd_test
%bi =l phi @check %nm1, @bwd_body %bi_next
%b_done =w csltl %bi, 0
jnz %b_done, @ret0, @bwd_body
@bwd_body
%b_idx =l mul %bi, 8
%b_src =l add %src, %b_idx
%b_dst =l add %dst, %b_idx
%b_val =l loadl %b_src
storel %b_val, %b_dst
%bi_next =l sub %bi, 1
jmp @bwd_test
@ret0
%zero =l copy 0
ret %zero
}
