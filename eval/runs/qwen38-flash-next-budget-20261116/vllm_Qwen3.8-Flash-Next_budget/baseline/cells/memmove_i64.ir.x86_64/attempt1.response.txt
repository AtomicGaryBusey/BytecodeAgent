

export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %iszero = ceql %n, $0
  jnz %iszero, @done
  %isbefore = csltl %dst, %src
  jnz %isbefore, @fwd_init
  jmp @bwd_init
@fwd_init
  %fsrc_init = addl %src, $0
  %fdst_init = addl %dst, $0
  %fcnt_init = addl %n, $0
  jmp @fwd_test
@fwd_test
  %fsrc = phi @fwd_init %fsrc_init, @fwd_body %fsrc_next
  %fdst = phi @fwd_init %fdst_init, @fwd_body %fdst_next
  %fcnt = phi @fwd_init %fcnt_init, @fwd_body %fcnt_next
  %fc = csgtl %fcnt, $0
  jnz %fc, @fwd_body
  jmp @done
@fwd_body
  %fval = loadl @%fsrc
  storel %fval, @%fdst
  %fsrc_next = addl %fsrc, $8
  %fdst_next = addl %fdst, $8
  %fcnt_next = subl %fcnt, $1
  jmp @fwd_test
@bwd_init
  %bcnt_init = addl %n, $0
  %b2 = addl %bcnt_init, %bcnt_init
  %b4 = addl %b2, %b2
  %bbytes = addl %b4, %b4
  %boff = subl %bbytes, $8
  %psrc_init = addl %src, %boff
  %pdst_init = addl %dst, %boff
  jmp @bwd_test
@bwd_test
  %bsrc = phi @bwd_init %psrc_init, @bwd_body %bsrc_next
  %bdst = phi @bwd_init %pdst_init, @bwd_body %bdst_next
  %bcnt = phi @bwd_init %bcnt_init, @bwd_body %bcnt_next
  %bc = csgtl %bcnt, $0
  jnz %bc, @bwd_body
  jmp @done
@bwd_body
  %bval = loadl @%bsrc
  storel %bval, @%bdst
  %bsrc_next = subl %bsrc, $8
  %bdst_next = subl %bdst, $8
  %bcnt_next = subl %bcnt, $1
  jmp @bwd_test
@done
  %zero = subl %n, %n
  retl %zero
}
