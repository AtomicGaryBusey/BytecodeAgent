

export function l $base64_encode(l %src, l %n, l %out) {
@start:
  %sl_pin =l alloc 8
  %sl_pout =l alloc 8
  %sl_rem =l alloc 8
  %sl_v =l alloc 8
  %sl_nb =l alloc 8
  %sl_chars =l alloc 8
  %sl_j =l alloc 8
  %sl_idx =l alloc 8
  storel %src, %sl_pin
  storel %out, %sl_pout
  storel %n, %sl_rem
  jump @Lloop

@Lloop:
  %loop_rem =l loadl %sl_rem
  %loop_eq0 =w ceql %loop_rem, 0
  jnz %loop_eq0, @Ldone, @Lhas

@Lhas:
  %has_rem =l loadl %sl_rem
  %has_lt3 =w csltl %has_rem, 3
  jnz %has_lt3, @Lsmall, @Lfull

@Lfull:
  %full_pin =l loadl %sl_pin
  %full_b0 =l loadub %full_pin
  %full_pin1 =l add %full_pin, 1
  %full_b1 =l loadub %full_pin1
  %full_pin2 =l add %full_pin1, 1
  %full_b2 =l loadub %full_pin2
  %full_s16 =l shl %full_b0, 16
  %full_s8 =l shl %full_b1, 8
  %full_or1 =l or %full_s16, %full_s8
  %full_v =l or %full_or1, %full_b2
  storel %full_v, %sl_v
  storel 3, %sl_nb
  storel 4, %sl_chars
  jump @Lgroup

@Lsmall:
  %small_rem =l loadl %sl_rem
  %small_eq1 =w ceql %small_rem, 1
  jnz %small_eq1, @Lone, @Ltwo

@Lone:
  %one_pin =l loadl %sl_pin
  %one_b0 =l loadub %one_pin
  %one_s16 =l shl %one_b0, 16
  storel %one_s16, %sl_v
  storel 1, %sl_nb
  storel 2, %sl_chars
  jump @Lgroup

@Ltwo:
  %two_pin =l loadl %sl_pin
  %two_b0 =l loadub %two_pin
  %two_pin1 =l add %two_pin, 1
  %two_b1 =l loadub %two_pin1
  %two_s16 =l shl %two_b0, 16
  %two_s8 =l shl %two_b1, 8
  %two_v =l or %two_s16, %two_s8
  storel %two_v, %sl_v
  storel 2, %sl_nb
  storel 3, %sl_chars
  jump @Lgroup

@Lgroup:
  storel 0, %sl_j
  jump @Lchar

@Lchar:
  %char_j =l loadl %sl_j
  %char_chars =l loadl %sl_chars
  %char_do =w csltl %char_j, %char_chars
  jnz %char_do, @Lextr, @Lpad

@Lextr:
  %extr_j =l loadl %sl_j
  %extr_eq0 =w ceql %extr_j, 0
  jnz %extr_eq0, @Lset0, @Lchk1

@Lset0:
  %set0_v =l loadl %sl_v
  %set0_idx =l shr %set0_v, 18
  storel %set0_idx, %sl_idx
  jump @Lmap

@Lchk1:
  %chk1_j =l loadl %sl_j
  %chk1_eq1 =w ceql %chk1_j, 1
  jnz %chk1_eq1, @Lset1, @Lchk2

@Lset1:
  %set1_v =l loadl %sl_v
  %set1_sh =l shr %set1_v, 12
  %set1_idx =l and %set1_sh, 63
  storel %set1_idx, %sl_idx
  jump @Lmap

@Lchk2:
  %chk2_j =l loadl %sl_j
  %chk2_eq2 =w ceql %chk2_j, 2
  jnz %chk2_eq2, @Lset2, @Lset3

@Lset2:
  %set2_v =l loadl %sl_v
  %set2_sh =l shr %set2_v, 6
  %set2_idx =l and %set2_sh, 63
  storel %set2_idx, %sl_idx
  jump @Lmap

@Lset3:
  %set3_v =l loadl %sl_v
  %set3_idx =l and %set3_v, 63
  storel %set3_idx, %sl_idx
  jump @Lmap

@Lmap:
  %map_idx =l loadl %sl_idx
  %map_lt26 =w csltl %map_idx, 26
  jnz %map_lt26, @LmapA, @LnotA

@LmapA:
  %mapA_idx =l loadl %sl_idx
  %mapA_ch =l add %mapA_idx, 65
  %mapA_pout =l loadl %sl_pout
  storeb %mapA_ch, %mapA_pout
  jump @Lcharinc

@LnotA:
  %notA_idx =l loadl %sl_idx
  %notA_lt52 =w csltl %notA_idx, 52
  jnz %notA_lt52, @LmapB, @LnotB

@LmapB:
  %mapB_idx =l loadl %sl_idx
  %mapB_sub =l sub %mapB_idx, 26
  %mapB_ch =l add %mapB_sub, 97
  %mapB_pout =l loadl %sl_pout
  storeb %mapB_ch, %mapB_pout
  jump @Lcharinc

@LnotB:
  %notB_idx =l loadl %sl_idx
  %notB_lt62 =w csltl %notB_idx, 62
  jnz %notB_lt62, @LmapC, @LnotC

@LmapC:
  %mapC_idx =l loadl %sl_idx
  %mapC_sub =l sub %mapC_idx, 52
  %mapC_ch =l add %mapC_sub, 48
  %mapC_pout =l loadl %sl_pout
  storeb %mapC_ch, %mapC_pout
  jump @Lcharinc

@LnotC:
  %notC_idx =l loadl %sl_idx
  %notC_eq62 =w ceql %notC_idx, 62
  jnz %notC_eq62, @LmapD, @LmapE

@LmapD:
  %mapD_pout =l loadl %sl_pout
  storeb 43, %mapD_pout
  jump @Lcharinc

@LmapE:
  %mapE_pout =l loadl %sl_pout
  storeb 47, %mapE_pout
  jump @Lcharinc

@Lcharinc:
  %inc_pout =l loadl %sl_pout
  %inc_pout1 =l add %inc_pout, 1
  storel %inc_pout1, %sl_pout
  %inc_j =l loadl %sl_j
  %inc_j1 =l add %inc_j, 1
  storel %inc_j1, %sl_j
  jump @Lchar

@Lpad:
  %pad_j =l loadl %sl_j
  %pad_eq4 =w ceql %pad_j, 4
  jnz %pad_eq4, @Lnext, @Lpad1

@Lpad1:
  %pad1_pout =l loadl %sl_pout
  storeb 61, %pad1_pout
  %pad1_pout1 =l add %pad1_pout, 1
  storel %pad1_pout1, %sl_pout
  %pad1_j =l loadl %sl_j
  %pad1_j1 =l add %pad1_j, 1
  storel %pad1_j1, %sl_j
  jump @Lpad

@Lnext:
  %next_pin =l loadl %sl_pin
  %next_nb =l loadl %sl_nb
  %next_pin1 =l add %next_pin, %next_nb
  storel %next_pin1, %sl_pin
  %next_rem =l loadl %sl_rem
  %next_rem1 =l sub %next_rem, %next_nb
  storel %next_rem1, %sl_rem
  jump @Lloop

@Ldone:
  %done_pout =l loadl %sl_pout
  %done_res =l sub %done_pout, %out
  retl %done_res
}
