

export function l $base64_encode(l %src, l %n, l %out) {
@start:
%p_in =l alloc 8
%p_outp =l alloc 8
%p_rem =l alloc 8
%p_val =l alloc 8
%p_nbytes =l alloc 8
%p_chars =l alloc 8
%p_i =l alloc 8
%p_idx =l alloc 8
%p_len =l alloc 8
storel %src, %p_in
storel %out, %p_outp
storel %n, %p_rem
storel 0, %p_len
jmp @loop
@loop:
%loop_rem =l loadl %p_rem
%loop_zero =w ceql %loop_rem, 0
jnz %loop_zero, @done, @more
@more:
%more_rem =l loadl %p_rem
%more_lt3 =w csltl %more_rem, 3
jnz %more_lt3, @tail, @triplet
@triplet:
%tr_in =l loadl %p_in
%tr_b0 =l loadub %tr_in
%tr_in1 =l add %tr_in, 1
%tr_b1 =l loadub %tr_in1
%tr_in2 =l add %tr_in1, 1
%tr_b2 =l loadub %tr_in2
%tr_sh16 =l shl %tr_b0, 16
%tr_sh8 =l shl %tr_b1, 8
%tr_or1 =l or %tr_sh16, %tr_sh8
%tr_val =l or %tr_or1, %tr_b2
storel %tr_val, %p_val
storel 3, %p_nbytes
storel 4, %p_chars
jmp @emit
@tail:
%tl_rem =l loadl %p_rem
%tl_eq1 =w ceql %tl_rem, 1
jnz %tl_eq1, @one, @two
@one:
%one_in =l loadl %p_in
%one_b0 =l loadub %one_in
%one_sh16 =l shl %one_b0, 16
storel %one_sh16, %p_val
storel 1, %p_nbytes
storel 2, %p_chars
jmp @emit
@two:
%tw_in =l loadl %p_in
%tw_b0 =l loadub %tw_in
%tw_in1 =l add %tw_in, 1
%tw_b1 =l loadub %tw_in1
%tw_sh16 =l shl %tw_b0, 16
%tw_sh8 =l shl %tw_b1, 8
%tw_or =l or %tw_sh16, %tw_sh8
storel %tw_or, %p_val
storel 2, %p_nbytes
storel 3, %p_chars
jmp @emit
@emit:
storel 0, %p_i
jmp @emit_loop
@emit_loop:
%el_i =l loadl %p_i
%el_chars =l loadl %p_chars
%el_more =w csltl %el_i, %el_chars
jnz %el_more, @encode, @pad
@encode:
%en_i =l loadl %p_i
%en_mul =l mul %en_i, 6
%en_shift =l sub 18, %en_mul
%en_val =l loadl %p_val
%en_shr =l shr %en_val, %en_shift
%en_idx =l and %en_shr, 63
storel %en_idx, %p_idx
jmp @map
@map:
%mp_idx =l loadl %p_idx
%mp_lt26 =w csltl %mp_idx, 26
jnz %mp_lt26, @letters, @not_letters
@letters:
%lt_idx =l loadl %p_idx
%lt_add =l add %lt_idx, 65
%lt_w =w trunc %lt_add
%lt_ptr =l loadl %p_outp
storeb %lt_w, %lt_ptr
jmp @inc
@not_letters:
%nl_idx =l loadl %p_idx
%nl_lt52 =w csltl %nl_idx, 52
jnz %nl_lt52, @low, @not_low
@low:
%lo_idx =l loadl %p_idx
%lo_sub =l sub %lo_idx, 26
%lo_add =l add %lo_sub, 97
%lo_w =w trunc %lo_add
%lo_ptr =l loadl %p_outp
storeb %lo_w, %lo_ptr
jmp @inc
@not_low:
%nd_idx =l loadl %p_idx
%nd_lt62 =w csltl %nd_idx, 62
jnz %nd_lt62, @digits, @not_digits
@digits:
%dg_idx =l loadl %p_idx
%dg_sub =l sub %dg_idx, 52
%dg_add =l add %dg_sub, 48
%dg_w =w trunc %dg_add
%dg_ptr =l loadl %p_outp
storeb %dg_w, %dg_ptr
jmp @inc
@not_digits:
%nd2_idx =l loadl %p_idx
%nd_eq62 =w ceql %nd2_idx, 62
jnz %nd_eq62, @plus, @slash
@plus:
%pl_const =w add 0, 43
%pl_ptr =l loadl %p_outp
storeb %pl_const, %pl_ptr
jmp @inc
@slash:
%sl_const =w add 0, 47
%sl_ptr =l loadl %p_outp
storeb %sl_const, %sl_ptr
jmp @inc
@inc:
%ic_ptr =l loadl %p_outp
%ic_ptr1 =l add %ic_ptr, 1
storel %ic_ptr1, %p_outp
%ic_i =l loadl %p_i
%ic_i1 =l add %ic_i, 1
storel %ic_i1, %p_i
%ic_len =l loadl %p_len
%ic_len1 =l add %ic_len, 1
storel %ic_len1, %p_len
jmp @emit_loop
@pad:
%pd_i =l loadl %p_i
%pd_eq4 =w ceql %pd_i, 4
jnz %pd_eq4, @next, @write_pad
@write_pad:
%wp_const =w add 0, 61
%wp_ptr =l loadl %p_outp
storeb %wp_const, %wp_ptr
%wp_ptr1 =l add %wp_ptr, 1
storel %wp_ptr1, %p_outp
%wp_i =l loadl %p_i
%wp_i1 =l add %wp_i, 1
storel %wp_i1, %p_i
%wp_len =l loadl %p_len
%wp_len1 =l add %wp_len, 1
storel %wp_len1, %p_len
jmp @pad
@next:
%nx_in =l loadl %p_in
%nx_nb =l loadl %p_nbytes
%nx_in1 =l add %nx_in, %nx_nb
storel %nx_in1, %p_in
%nx_rem =l loadl %p_rem
%nx_rem1 =l sub %nx_rem, %nx_nb
storel %nx_rem1, %p_rem
jmp @loop
@done:
%dn_len =l loadl %p_len
retl %dn_len
}
