

export function l $crc32(l %p, l %n) {
@start
  %minus1 =w immw -1
  %poly =w immw -306674912
  %idx =l sub %n, %n
  %crc =w immw -1
  jmp @byte_loop
@byte_loop
  %have =w csltl %idx, %n
  jnz %have, @body, @final
@body
  %addr =l add %p, %idx
  %b =w loadub %addr
  %crc =w xor %crc, %b
  %bits =l sub %n, %n
  jmp @bit_loop
@bit_loop
  %lsb =w and %crc, 1
  jnz %lsb, @crc_one, @crc_zero
@crc_one
  %sh =w shr %crc, 1
  %crc =w xor %sh, %poly
  jmp @bit_end
@crc_zero
  %crc =w shr %crc, 1
  jmp @bit_end
@bit_end
  %bits =l add %bits, 1
  %go =w csltl %bits, 8
  jnz %go, @bit_loop, @next
@next
  %idx =l add %idx, 1
  jmp @byte_loop
@final
  %crc =w xor %crc, %minus1
  %res =l extuw %crc
  ret %res
}
