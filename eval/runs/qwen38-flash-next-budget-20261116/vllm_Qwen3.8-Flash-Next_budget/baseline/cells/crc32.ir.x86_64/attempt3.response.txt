

export function l $crc32(l %p, l %n) {
@start
  %i0 =l sub %n, %n
  %cond =w csltl %i0, %n
  %zr =w sub %cond, %cond
  %crc0 =w xor %zr, -1
  %poly =w xor %zr, -306674912
  jmp @loop
@loop
  %i =l phi @start %i0, @bytes %i2
  %crc =w phi @start %crc0, @bytes %c8
  %have =w csltl %i, %n
  jnz %have, @body, @final
@body
  %addr =l add %p, %i
  %b =w loadub %addr
  %c0 =w xor %crc, %b
  jmp @bytes
@bytes
  %lsb1 =w and %c0, 1
  %m1 =w sub %zr, %lsb1
  %s1 =w shr %c0, 1
  %k1 =w and %m1, %poly
  %c1 =w xor %s1, %k1
  %lsb2 =w and %c1, 1
  %m2 =w sub %zr, %lsb2
  %s2 =w shr %c1, 1
  %k2 =w and %m2, %poly
  %c2 =w xor %s2, %k2
  %lsb3 =w and %c2, 1
  %m3 =w sub %zr, %lsb3
  %s3 =w shr %c2, 1
  %k3 =w and %m3, %poly
  %c3 =w xor %s3, %k3
  %lsb4 =w and %c3, 1
  %m4 =w sub %zr, %lsb4
  %s4 =w shr %c3, 1
  %k4 =w and %m4, %poly
  %c4 =w xor %s4, %k4
  %lsb5 =w and %c4, 1
  %m5 =w sub %zr, %lsb5
  %s5 =w shr %c4, 1
  %k5 =w and %m5, %poly
  %c5 =w xor %s5, %k5
  %lsb6 =w and %c5, 1
  %m6 =w sub %zr, %lsb6
  %s6 =w shr %c5, 1
  %k6 =w and %m6, %poly
  %c6 =w xor %s6, %k6
  %lsb7 =w and %c6, 1
  %m7 =w sub %zr, %lsb7
  %s7 =w shr %c6, 1
  %k7 =w and %m7, %poly
  %c7 =w xor %s7, %k7
  %lsb8 =w and %c7, 1
  %m8 =w sub %zr, %lsb8
  %s8 =w shr %c7, 1
  %k8 =w and %m8, %poly
  %c8 =w xor %s8, %k8
  %i2 =l add %i, 1
  jmp @loop
@final
  %cf =w xor %crc, -1
  %res =l extuw %cf
  ret %res
}
