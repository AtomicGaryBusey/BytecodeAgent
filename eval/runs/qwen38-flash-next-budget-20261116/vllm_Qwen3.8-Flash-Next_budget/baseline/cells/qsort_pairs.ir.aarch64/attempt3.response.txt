

export function l $qsort_pairs(l %a, l %n) {
@start:
  %zero =l 0
  %one =l 1
  %two =l 2
  %c0 =w csltl %n, %two
  jnz %c0, @ret0, @prep
@ret0:
  %rv =l 0
  ret l %rv
@prep:
  %sixteen =l 16
  %nm1 =l subl %n, %one
  %poff =l mull %nm1, %sixteen
  %pptr =l addl %a, %poff
  %pk =l loadl %pptr, 0
  %pv =l loadl %pptr, 8
  %i0 =l addl %a, %zero
  %j0 =l addl %a, %zero
  %left0 =l addl %zero, %zero
  jmp @loop
@loop:
  %i =l phi @prep %i0, @true_cmp %i_t, @false %i_f
  %j =l phi @prep %j0, @true_cmp %j_t, @false %j_f
  %left =l phi @prep %left0, @true_cmp %left_t, @false %left_f
  %cond =w csltl %j, %pptr
  jnz %cond, @body, @done
@body:
  %kj =l loadl %j, 0
  %vj =l loadl %j, 8
  %klt =w csltl %kj, %pk
  jnz %klt, @true_cmp, @key_eq
@key_eq:
  %keq =w ceql %kj, %pk
  jnz %keq, @val_lt, @false
@val_lt:
  %vlt =w csltl %vj, %pv
  jnz %vlt, @true_cmp, @val_eq
@val_eq:
  %veq =w ceql %vj, %pv
  jnz %veq, @true_cmp, @false
@true_cmp:
  %ti =l loadl %i, 0
  %tv =l loadl %i, 8
  %tj =l loadl %j, 0
  %tvj =l loadl %j, 8
  storel %tj, %i, 0
  storel %tvj, %i, 8
  storel %ti, %j, 0
  storel %tv, %j, 8
  %i_t =l addl %i, %sixteen
  %j_t =l addl %j, %sixteen
  %left_t =l addl %left, %one
  jmp @loop
@false:
  %i_f =l addl %i, %zero
  %j_f =l addl %j, %sixteen
  %left_f =l addl %left, %zero
  jmp @loop
@done:
  %ki =l loadl %i, 0
  %vi =l loadl %i, 8
  %kp =l loadl %pptr, 0
  %vp =l loadl %pptr, 8
  storel %kp, %i, 0
  storel %vp, %i, 8
  storel %ki, %pptr, 0
  storel %vi, %pptr, 8
  %r0 =l subl %n, %left
  %right =l subl %r0, %one
  %rp =l addl %i, %sixteen
  %rcall_left =l call $qsort_pairs(l %a, l %left)
  %rlt =w csltl %right, %two
  jnz %rlt, @ret0, @do_right
@do_right:
  %rcall_right =l call $qsort_pairs(l %rp, l %right)
  jmp @ret0
}
