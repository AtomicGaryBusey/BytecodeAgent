

export function l $qsort_pairs(l %a, l %n) {
@start
  %small = w csltl %n, 2
  jnz %small, @ret_zero

  %last = l addl %a, 0
  %m = l subl %n, 1
@last_loop
  %m0 = w ceql %m, 0
  jnz %m0, @last_done
  %last = l addl %last, 16
  %m = l subl %m, 1
  j @last_loop
@last_done
  %pk = l loadl %last
  %lastv = l addl %last, 8
  %pv = l loadl %lastv

  %iptr = l addl %a, 0
  %jptr = l addl %a, 0
  %rem = l subl %n, 1
  %ipos = l addl 0, 0
@part_loop
  %rem0 = w ceql %rem, 0
  jnz %rem0, @part_done
  %jk = l loadl %jptr
  %jvptr = l addl %jptr, 8
  %jv = l loadl %jvptr
  %ltkey = w csltl %jk, %pk
  jnz %ltkey, @do_swap
  %eqkey = w ceql %jk, %pk
  jnz %eqkey, @check_val
  j @skip_swap
@check_val
  %ltval = w csltl %jv, %pv
  jnz %ltval, @do_swap
  %eqval = w ceql %jv, %pv
  jnz %eqval, @do_swap
  j @skip_swap
@do_swap
  %sk = l loadl %iptr
  %svptr = l addl %iptr, 8
  %sv = l loadl %svptr
  storel %jk, %iptr
  storel %jv, %svptr
  storel %sk, %jptr
  storel %sv, %jvptr
  %ipos = l addl %ipos, 1
  %iptr = l addl %iptr, 16
@skip_swap
  %jptr = l addl %jptr, 16
  %rem = l subl %rem, 1
  j @part_loop
@part_done
  %iptrv = l addl %iptr, 8
  %lk = l loadl %iptr
  %lv = l loadl %iptrv
  storel %pk, %iptr
  storel %pv, %iptrv
  storel %lk, %jptr
  %jptrv = l addl %jptr, 8
  storel %lv, %jptrv

  %right_pos = l addl %ipos, 1
  %right_n = l subl %n, %right_pos

  %left_zero = w ceql %ipos, 0
  jnz %left_zero, @after_left
  %r1 =l call $qsort_pairs(l %a, l %ipos)
@after_left

  %right_zero = w ceql %right_n, 0
  jnz %right_zero, @ret_zero
  %right_ptr = l addl %iptr, 16
  %r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
@ret_zero
  ret 0
}
