

collatz_max:
  cld
  xor eax, eax
  test rdi, rdi
  jle .Ldone
  mov r11, rdi
  mov rdi, 1
.Louter:
  mov rdx, rdi
  xor r8d, r8d
.Linner:
  cmp rdx, 1
  je .Lupdate
  test rdx, 1
  jne .Lodd
  shr rdx, 1
  inc r8
  jmp .Linner
.Lodd:
  imul rdx, rdx, 3
  inc rdx
  inc r8
  jmp .Linner
.Lupdate:
  cmp r8, rax
  cmovg rax, r8
  inc rdi
  dec r11
  jne .Louter
.Ldone:
  ret
