

export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
%i0 =l copy 0
%cnt0 =l copy 0
%eight =l copy 8
%one =l copy 1
jmp @loop
@loop
%i =l phi %i0 @start, %i1 @cont
%cnt =l phi %cnt0 @start, %cnt1 @cont
%cond =w csltl %i, %n
jnz %cond @body @end
@body
%i8 =l mul %i, %eight
%sp =l add %src, %i8
%dp =l add %dst, %i8
%x =l loadl %sp
%y =l call %f(l %x)
storel %y, %dp
%less =w csltl %y, %x
jnz %less @inc @skip
@inc
%cnt_inc =l add %cnt, %one
jmp @cont
@skip
%cnt_skip =l copy %cnt
jmp @cont
@cont
%cnt1 =l phi %cnt_inc @inc, %cnt_skip @skip
%i1 =l add %i, 1
jmp @loop
@end
ret %cnt
}
