

count_desc:
	cmp x1, #2
	b.lt .Lzero
	mov x3, xzr
	mov x4, xzr
	mov x6, x1
	sub x6, x6, #1
.Lloop:
	cmp x3, x6
	b.ge .Ldone
	ldr x5, [x0, x3, lsl #3]
	add x7, x3, #1
	ldr x8, [x0, x7, lsl #3]
	cmp x5, x8
	b.le .Lskip
	add x4, x4, #1
.Lskip:
	add x3, x3, #1
	b .Lloop
.Ldone:
	mov x0, x4
	ret
.Lzero:
	mov x0, xzr
	ret
