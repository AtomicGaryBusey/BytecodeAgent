

crc32:
    movz    w2, #0xffff
    movk    w2, #0xffff, lsl #16
    movz    w5, #0x8320
    movk    w5, #0xEDB8, lsl #16
    cbz     w1, .Lfin
.Lbyte:
    ldrb    w3, [x0]
    add     x0, x0, #1
    eor     w2, w2, w3
    movz    w4, #8
.Lbit:
    tbz     w2, #0, .Lzero
    lsr     w2, w2, #1
    eor     w2, w2, w5
    b       .Lbitnext
.Lzero:
    lsr     w2, w2, #1
.Lbitnext:
    subs    w4, w4, #1
    b.ne    .Lbit
    subs    w1, w1, #1
    b.ne    .Lbyte
.Lfin:
    movz    w0, #0xffff
    movk    w0, #0xffff, lsl #16
    eor     w0, w0, w2
    ret
