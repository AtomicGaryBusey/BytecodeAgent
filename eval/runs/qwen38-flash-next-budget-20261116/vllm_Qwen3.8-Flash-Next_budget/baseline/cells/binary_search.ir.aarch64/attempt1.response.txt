

export function l $binary_search(l %a, l %n, l %key) {
@start
        %zero =l sub %n, %n
        %hi0 =l sub %n, %zero
        jmp @loop
@loop
        %lo =l phi @start %zero, @update %lo_next
        %hi =l phi @start %hi0, @update %hi_next
        %cond =w csltl %lo, %hi
        jnz %cond, @body, @end
@body
        %diff =l sub %hi, %lo
        %half =l shr %diff, 1
        %mid =l add %lo, %half
        %off =l mul %mid, 8
        %p =l add %a, %off
        %v =l loadl %p
        %vlt =w csltl %v, %key
        jnz %vlt, @left, @ge
@ge
        %veq =w ceql %v, %key
        jnz %veq, @found, @right
@found
        ret %mid
@left
        %lo_left =l add %mid, 1
        %hi_left =l add %hi, %zero
        jmp @update
@right
        %lo_right =l sub %lo, %zero
        %hi_right =l add %mid, %zero
        jmp @update
@update
        %lo_next =l phi @left %lo_left, @right %lo_right
        %hi_next =l phi @left %hi_left, @right %hi_right
        jmp @loop
@end
        %notfound =l sub %zero, 1
        ret %notfound
}
