

packbits_decode:
cld
mov r8, rdx
add rsi, rdi
add rcx, rdx
.Ldecode_loop:
cmp rdi, rsi
jnb .Ldone_success
movsx rax, byte ptr [rdi]
inc rdi
test rax, rax
jns .Lliteral
cmp rax, -128
je .Ldecode_loop
mov r9, 1
sub r9, rax
cmp rdi, rsi
jae .Lerr_truncated
movzx r11d, byte ptr [rdi]
mov r10, rcx
sub r10, rdx
cmp r10, r9
jb .Lrepeat_partial
mov r10, r9
.Lrepeat_full_fill:
test r10, r10
je .Lrepeat_full_done
mov byte ptr [rdx], r11b
inc rdx
dec r10
jmp .Lrepeat_full_fill
.Lrepeat_full_done:
inc rdi
jmp .Ldecode_loop
.Lrepeat_partial:
test r10, r10
je .Lrepeat_partial_done
mov byte ptr [rdx], r11b
inc rdx
dec r10
jmp .Lrepeat_partial
.Lrepeat_partial_done:
mov rax, -1
ret
.Lliteral:
mov r9, rax
inc r9
mov r10, rdi
add r10, r9
cmp r10, rsi
ja .Lerr_truncated
mov r10, rcx
sub r10, rdx
cmp r10, r9
jb .Lliteral_partial
mov r10, r9
.Lliteral_full_copy:
test r10, r10
je .Lliteral_full_done
movzx eax, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec r10
jmp .Lliteral_full_copy
.Lliteral_full_done:
jmp .Ldecode_loop
.Lliteral_partial:
test r10, r10
je .Lliteral_partial_done
movzx eax, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec r10
jmp .Lliteral_partial
.Lliteral_partial_done:
mov rax, -1
ret
.Ldone_success:
mov rax, rdx
sub rax, r8
ret
.Lerr_truncated:
mov rax, -2
ret
