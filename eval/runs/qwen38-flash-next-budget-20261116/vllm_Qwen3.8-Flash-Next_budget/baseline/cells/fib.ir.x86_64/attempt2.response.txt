

export function l $fib(l %n) {
@start
  %lt2 =w csltl %n, $2
  jnz %lt2, @small, @loop
@small
  ret %n
@loop
  %a =l phi @start $0, @cont %a1
  %b =l phi @start $1, @cont %b1
  %i =l phi @start $2, @cont %i1
  %next =l add %a, %b
  %inext =l add %i, $1
  %done =w csltl %n, %inext
  jnz %done, @done, @cont
@cont
  %a1 =l add %b, $0
  %b1 =l add %next, $0
  %i1 =l add %inext, $0
  jmp @loop
@done
  ret %next
}
