

export function l $fib(l %.n) {
@start
%.i0 =l copy 0
%.a0 =l copy 0
%.b0 =l copy 1
@loop
%.i =l phi @start %.i0, @body %.i_next
%.a =l phi @start %.a0, @body %.a_next
%.b =l phi @start %.b0, @body %.b_next
%.cmp =w csltl %.i, %.n
jnz %.cmp, @body, @exit
@body
%.c =l add %.a, %.b
%.a_next =l copy %.b
%.b_next =l copy %.c
%.i_next =l add %.i, 1
jmp @loop
@exit
ret %.a
}
