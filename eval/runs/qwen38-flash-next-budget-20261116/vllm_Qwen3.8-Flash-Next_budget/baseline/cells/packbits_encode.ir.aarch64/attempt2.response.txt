

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    %zero_i =l add $0, $0
    %zero_o =l add $0, $0
    jmp @main
@main
    %i =l phi %zero_i @entry, %i_rep @repeat_ok, %i_lit @copy_end
    %o =l phi %zero_o @entry, %o_rep @repeat_ok, %o_lit @copy_end
    %has =w csltl %i, %n
    jnz %has, @work, @done
@work
    %p_i =l add %src, %i
    %b =w loadub %p_i
    %r_init_j =l add %i, $1
    %r_limit =l add %i, $128
    jmp @run_scan
@run_scan
    %rj =l phi %r_init_j @work, %rj_next @run_inc
    %rj_lt_n =w csltl %rj, %n
    jnz %rj_lt_n, @run_scan_1, @run_end
@run_scan_1
    %rj_lt_limit =w csltl %rj, %r_limit
    jnz %rj_lt_limit, @run_scan_2, @run_end
@run_scan_2
    %r_pj =l add %src, %rj
    %r_bj =w loadub %r_pj
    %r_eq =w ceqw %r_bj, %b
    jnz %r_eq, @run_inc, @run_end
@run_inc
    %rj_next =l add %rj, $1
    jmp @run_scan
@run_end
    %run =l sub %rj, %i
    %run_lt3 =w csltl %run, $3
    jnz %run_lt3, @lit, @repeat
@repeat
    %r_need =l add %o, $2
    %r_bad =w csltl %cap, %r_need
    jnz %r_bad, @fail, @repeat_ok
@repeat_ok
    %r_hdr =l sub $257, %run
    %r_hdr_w =w trunc %r_hdr
    %r_out0 =l add %out, %o
    storeb %r_hdr_w, %r_out0
    %r_out1 =l add %r_out0, $1
    storeb %b, %r_out1
    %i_rep =l add %i, %run
    %o_rep =l add %o, $2
    jmp @main
@lit
    %l_init_j =l add %i, $1
    jmp @lit_scan
@lit_scan
    %lj =l phi %l_init_j @lit, %lj_next @lit_inc
    %l_len =l sub %lj, %i
    %l_lt_len =w csltl %l_len, $128
    jnz %l_lt_len, @lit_j, @lit_end
@lit_j
    %l_j_lt_n =w csltl %lj, %n
    jnz %l_j_lt_n, @lit_3, @lit_end
@lit_3
    %lj3 =l add %lj, $2
    %l_j3_lt_n =w csltl %lj3, %n
    jnz %l_j3_lt_n, @lit_load3, @lit_inc
@lit_load3
    %l_pj0 =l add %src, %lj
    %l_a =w loadub %l_pj0
    %l_pj1 =l add %l_pj0, $1
    %l_b1 =w loadub %l_pj1
    %l_eq1 =w ceqw %l_a, %l_b1
    jnz %l_eq1, @lit_eq2, @lit_inc
@lit_eq2
    %l_pj2 =l add %l_pj1, $1
    %l_b2 =w loadub %l_pj2
    %l_eq2 =w ceqw %l_a, %l_b2
    jnz %l_eq2, @lit_stop, @lit_inc
@lit_stop
    jmp @lit_end
@lit_inc
    %lj_next =l add %lj, $1
    jmp @lit_scan
@lit_end
    %lit_len =l sub %lj, %i
    %need_lit =l add %lit_len, $1
    %lit_need =l add %o, %need_lit
    %lit_bad =w csltl %cap, %lit_need
    jnz %lit_bad, @fail, @lit_ok
@lit_ok
    %lit_hdr =l sub %lit_len, $1
    %lit_hdr_w =w trunc %lit_hdr
    %lit_out0 =l add %out, %o
    storeb %lit_hdr_w, %lit_out0
    %copy_src =l add %src, %i
    %copy_stop =l add %copy_src, %lit_len
    %copy_dst =l add %lit_out0, $1
    jmp @copy
@copy
    %cp_src =l phi %copy_src @lit_ok, %cp_src_next @copy_body
    %cp_dst =l phi %copy_dst @lit_ok, %cp_dst_next @copy_body
    %cp_lt =w csltl %cp_src, %copy_stop
    jnz %cp_lt, @copy_body, @copy_end
@copy_body
    %cp_b =w loadub %cp_src
    storeb %cp_b, %cp_dst
    %cp_src_next =l add %cp_src, $1
    %cp_dst_next =l add %cp_dst, $1
    jmp @copy
@copy_end
    %i_lit =l add %i, %lit_len
    %o_lit =l add %o, %lit_need
    jmp @main
@done
    ret %o
@fail
    %fail =l sub $0, $1
    ret %fail
}
