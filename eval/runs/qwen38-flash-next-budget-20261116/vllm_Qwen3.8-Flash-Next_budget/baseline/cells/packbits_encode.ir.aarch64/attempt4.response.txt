

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
%wzero =l const 0
%izero =l const 0
jmp @main_head
@main_head
%written =l phi @start %wzero, @rep_write %wr_r, @copy_done %wr_next_lit
%i =l phi @start %izero, @rep_write %i_r, @copy_done %i_lit
%more =w csltl %i, %n
jnz %more, @main_body, @main_done
@main_done
ret %written
@main_body
%ip =l add %src, %i
%first =w loadub %ip
%one =l const 1
jmp @run_head
@run_head
%cnt =l phi @main_body %one, @run_inc %cnt_next
%cnt_lt =w csltl %cnt, 128
%cnt_h =l add %cnt, 0
jnz %cnt_lt, @run_idx, @run_end
@run_idx
%off =l add %i, %cnt
%off_lt =w csltl %off, %n
%cnt_i =l add %cnt, 0
jnz %off_lt, @run_load, @run_end
@run_load
%bp =l add %src, %off
%b =w loadub %bp
%same =w ceqw %b, %first
%cnt_l =l add %cnt, 0
jnz %same, @run_inc, @run_end
@run_inc
%cnt_next =l add %cnt, 1
jmp @run_head
@run_end
%run =l phi @run_head %cnt_h, @run_idx %cnt_i, @run_load %cnt_l
%lt_three =w csltl %run, 3
jnz %lt_three, @emit_literal, @emit_repeat
@emit_repeat
%rem =l sub %cap, %written
%bad =w csltl %rem, 2
jnz %bad, @ret_neg, @rep_write
@ret_neg
%neg =l const -1
ret %neg
@rep_write
%hp =l add %out, %written
%h =l sub 257, %run
storeb %h, %hp
%vp =l add %hp, 1
storeb %first, %vp
%wr_r =l add %written, 2
%i_r =l add %i, %run
jmp @main_head
@emit_literal
%jstart =l add %i, 1
jmp @lit_scan_head
@lit_scan_head
%j =l phi @emit_literal %jstart, @lit_scan_inc %jnext
%d =l sub %j, %i
%d_lt =w csltl %d, 128
%jh =l add %j, 0
jnz %d_lt, @lit_jcheck, @lit_scan_end
@lit_jcheck
%j_lt =w csltl %j, %n
%ji =l add %j, 0
jnz %j_lt, @lit_three, @lit_scan_end
@lit_three
%jplus2 =l add %j, 2
%jplus2_lt =w csltl %jplus2, %n
%jt =l add %j, 0
jnz %jplus2_lt, @lit_read, @lit_scan_end
@lit_read
%pa =l add %src, %j
%poffb =l add %j, 1
%pb =l add %src, %poffb
%pc =l add %src, %jplus2
%ba =w loadub %pa
%bb =w loadub %pb
%bc =w loadub %pc
%eqa =w ceqw %bb, %ba
jnz %eqa, @lit_eqtwo, @lit_scan_inc
@lit_eqtwo
%eqb =w ceqw %bc, %ba
%je =l add %j, 0
jnz %eqb, @lit_scan_end, @lit_scan_inc
@lit_scan_inc
%jnext =l add %j, 1
jmp @lit_scan_head
@lit_scan_end
%jend =l phi @lit_scan_head %jh, @lit_jcheck %ji, @lit_three %jt, @lit_eqtwo %je
%len =l sub %jend, %i
%size =l add %len, 1
%rem_lit =l sub %cap, %written
%bad_lit =w csltl %rem_lit, %size
jnz %bad_lit, @ret_neg, @lit_write
@lit_write
%hp_lit =l add %out, %written
%lit_head =l sub %len, 1
storeb %lit_head, %hp_lit
%dst =l add %hp_lit, 1
%srcptr =l add %src, %i
%kzero =l const 0
jmp @copy_head
@copy_head
%k =l phi @lit_write %kzero, @copy_inc %k_next
%k_lt =w csltl %k, %len
jnz %k_lt, @copy_body, @copy_done
@copy_body
%sp =l add %srcptr, %k
%dp =l add %dst, %k
%vb =w loadub %sp
storeb %vb, %dp
jmp @copy_inc
@copy_inc
%k_next =l add %k, 1
jmp @copy_head
@copy_done
%wr_lit =l add %written, %len
%wr_next_lit =l add %wr_lit, 1
%i_lit =l add %i, %len
jmp @main_head
}
