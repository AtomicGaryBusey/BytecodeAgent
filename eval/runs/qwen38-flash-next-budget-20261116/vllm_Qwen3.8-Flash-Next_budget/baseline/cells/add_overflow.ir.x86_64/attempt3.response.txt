@start
	%sum = addl %a, %b
	storel %sum, %out
	%bneg = csltl %b, $0
	%sumneg = csltl %sum, $0
	%aneg = csltl %a, $0
	jnz %aneg, @aneg
	jmp @apos
@apos
	jnz %bneg, @ret0
	jnz %sumneg, @ret1
	jmp @ret0
@aneg
	jnz %bneg, @bothneg
	jmp @ret0
@bothneg
	jnz %sumneg, @ret0
	jmp @ret1
@ret0
	ret $0
@ret1
	ret $1
}
