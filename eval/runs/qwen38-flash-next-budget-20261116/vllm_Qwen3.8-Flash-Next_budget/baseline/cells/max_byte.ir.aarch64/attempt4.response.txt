

export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @skip %i2
	%max =l phi @start 0, @skip %max2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%cmp =w csltl %max, %b
	jnz %cmp, @set, @skip
@set
	jmp @skip
@skip
	%max2 =l phi @body %max, @set %b
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %max
}
