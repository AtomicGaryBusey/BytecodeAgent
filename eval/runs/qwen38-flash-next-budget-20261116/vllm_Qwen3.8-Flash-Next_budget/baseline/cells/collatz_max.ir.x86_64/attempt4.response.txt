

export function l $collatz_max(l %.n) {
@start:
  cld
  %.n_less_1 =w csltl %.n, 1
  jnz %.n_less_1, @zero, @prep
@zero:
  ret 0
@prep:
  %.i0 =l 1
  %.m0 =l 0
  %.nplus =l addl %.n, 1
  jmp @outer
@outer:
  %.i =l phi %.i0 @prep, %.i_next_out @next
  %.m =l phi %.m0 @prep, %.m_next_out @next
  %.x0 =l %.i
  %.c0 =l 0
  jmp @inner
@inner:
  %.x =l phi %.x0 @outer, %.xe @even, %.xo @odd
  %.cnt =l phi %.c0 @outer, %.ce @even, %.co @odd
  %.is1 =w ceql %.x, 1
  jnz %.is1, @done, @step
@step:
  %.mod =l modsl %.x, 2
  %.is_even =w ceql %.mod, 0
  jnz %.is_even, @even, @odd
@even:
  %.xe =l divsl %.x, 2
  %.ce =l addl %.cnt, 1
  jmp @inner
@odd:
  %.t =l mull %.x, 3
  %.xo =l addl %.t, 1
  %.co =l addl %.cnt, 1
  jmp @inner
@done:
  %.lt =w csltl %.m, %.cnt
  %.i_next =l addl %.i, 1
  %.m_direct =l %.m
  jnz %.lt, @update, @next
@update:
  %.m_new =l %.cnt
  jmp @next
@next:
  %.m_next =l phi %.m_direct @done, %.m_new @update
  %.i_next_out =l %.i_next
  %.m_next_out =l %.m_next
  %.cont =w csltl %.i_next_out, %.nplus
  jnz %.cont, @outer, @end
@end:
  ret %.m_next
}
