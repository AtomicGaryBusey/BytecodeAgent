

export function l $collatz_max(l %n) {
@start
%zero =l copy 0
%small =w csltl %n, 1
jnz %small @ret0 @init
@ret0
ret %zero
@init
%i0 =l copy %zero
%m0 =l copy %zero
jmp @loop
@loop
%i =l phi %i0 @init, %i2 @after
%m =l phi %m0 @init, %m3 @after
%i1 =l add %i, 1
jmp @prep
@prep
%x0 =l copy %i1
%s0 =l copy 0
jmp @inner
@inner
%x =l phi %x0 @prep, %x2 @cont
%s =l phi %s0 @prep, %s2 @cont
%is1 =w ceql %x, 1
jnz %is1 @done @body
@body
%bit =l and %x, 1
%iseven =w ceql %bit, 0
jnz %iseven @even @odd
@even
%xeven =l sar %x, 1
jmp @cont
@odd
%u =l mul %x, 3
%xodd =l add %u, 1
jmp @cont
@cont
%x2 =l phi %xeven @even, %xodd @odd
%s2 =l add %s, 1
jmp @inner
@done
%g =w csltl %m, %s
jnz %g @take @keep
@take
%mt =l copy %s
jmp @after
@keep
%mk =l copy %m
jmp @after
@after
%m3 =l phi %mt @take, %mk @keep
%i2 =l add %i1, 1
%d =w ceql %i1, %n
jnz %d @ret @loop
@ret
ret %m3
}
