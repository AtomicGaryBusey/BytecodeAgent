

export function l $map_sum(l %a, l %n, l %f) {
@start:
%i0 =l copy 0
%sum0 =l copy 0
%p0 =l copy %a
%fptr =l copy %f
jmp @loop
@loop:
%i =l phi @start %i0, @body %i1
%sum =l phi @start %sum0, @body %sum1
%p =l phi @start %p0, @body %p1
%c =w csltl %i, %n
jnz %c, @body, @exit
@body:
%x =l loadl %p
%r =l call %fptr(l %x)
%sum1 =l add %sum, %r
%p1 =l add %p, 8
%i1 =l add %i, 1
jmp @loop
@exit:
ret %sum
}
