

export function l $fill_down(l a, l n, l start) {
@entry
  %i = movl 0
  %off = movl 0
  %one = movl 1
  %eight = movl 8
  j @loop
@loop
  %c = csltl %i, %n
  jnz %c, @body
  ret %n
@body
  %addr = addl %a, %off
  %v = subl %start, %i
  storel %v, %addr
  %i = addl %i, %one
  %off = addl %off, %eight
  j @loop
}
