

binary_search:
    xor r9, r9
.Lloop:
    cmp r9, rsi
    jge .Lnot_found
    mov rax, rsi
    sub rax, r9
    shr rax, 1
    add rax, r9
    cmp qword ptr [rdi + rax * 8], rdx
    je .Lfound
    jg .Lhigh
    inc r9
    jmp .Lloop
.Lhigh:
    mov rsi, rax
    jmp .Lloop
.Lfound:
    ret
.Lnot_found:
    mov rax, -1
    ret
