

packbits_encode:
cld
mov r9, rcx
xor r8, r8
.Ltop:
cmp r8, rsi
jge .Lend
mov rax, 1
.Lrun:
cmp rax, 128
jge .Lrun_done
lea r10, [r8 + rax]
cmp r10, rsi
jae .Lrun_done
movzx r11, byte ptr [rdi + r10]
cmp r11b, byte ptr [rdi + r8]
jne .Lrun_done
inc rax
jmp .Lrun
.Lrun_done:
cmp rax, 3
jl .Llit_start
cmp r9, 2
jb .Lfail
sub r9, 2
movzx r10, byte ptr [rdi + r8]
lea r11, [r8 + rax]
mov r8, r11
mov r11, 257
sub r11, rax
mov byte ptr [rdx], r11b
mov byte ptr [rdx + 1], r10b
add rdx, 2
jmp .Ltop
.Llit_start:
xor rax, rax
.Llit:
cmp rax, 128
jge .Llit_done
lea r10, [r8 + rax]
cmp r10, rsi
jge .Llit_done
lea r11, [r10 + 2]
cmp r11, rsi
jge .Llit_inc
movzx r11, byte ptr [rdi + r10]
cmp r11b, byte ptr [rdi + r10 + 1]
jne .Llit_inc
movzx r11, byte ptr [rdi + r10 + 1]
cmp r11b, byte ptr [rdi + r10 + 2]
jne .Llit_inc
jmp .Llit_done
.Llit_inc:
inc rax
jmp .Llit
.Llit_done:
lea r10, [rax + 1]
cmp r9, r10
jb .Lfail
sub r9, r10
lea r10, [rax - 1]
mov byte ptr [rdx], r10b
lea r11, [rdi + r8]
add rdx, 1
lea r8, [r8 + rax]
.Lcopy:
movzx r10, byte ptr [r11]
mov byte ptr [rdx], r10b
inc r11
inc rdx
dec rax
jnz .Lcopy
jmp .Ltop
.Lfail:
mov rax, -1
ret
.Lend:
mov rax, rcx
sub rax, r9
ret
