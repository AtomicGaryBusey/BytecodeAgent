

export function l $isqrt (l %n)
@entry
    %rem = movl 0
    %root = movl 0
    %sh = movl 62
    br @loop
@loop
    %t0 = shrl %n, %sh
    %t1 = shrl %t0, 2
    %t2 = shll %t1, 2
    %pair = subl %t0, %t2
    %r0 = shll %rem, 2
    %rem = addl %r0, %pair
    %root = shll %root, 1
    %d0 = shll %root, 1
    %d = addl %d0, 1
    %lt = csltl %rem, %d
    jnz %lt @post
@update
    %rem = subl %rem, %d
    %root = addl %root, 1
    br @post
@post
    %zero = ceql %sh, 0
    jnz %zero @ret
@notzero
    %sh = subl %sh, 2
    br @loop
@ret
    retl %root
end
