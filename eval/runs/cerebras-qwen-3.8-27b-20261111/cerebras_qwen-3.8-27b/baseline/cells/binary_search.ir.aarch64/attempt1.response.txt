

export function l $binary_search(l a, l n, l k)
@entry
    %neg = subsl 0 1
    %low = addsl 0 0
    %high = addsl n 0
    j @loop
@loop
    %cond = csltl %low %high
    jnz %cond @body
    ret %neg
@body
    %diff = subsl %high %low
    %half = divsl %diff 2
    %mid = addsl %low %half
    %off1 = addsl %mid %mid
    %off2 = addsl %off1 %off1
    %off3 = addsl %off2 %off2
    %ptr = addsl a %off3
    %val = loadl %ptr
    %eq = ceql %val k
    jnz %eq @found
    %lt = csltl %val k
    jnz %lt @right
    %high = addsl %mid 0
    j @loop
@right
    %low = addsl %mid 1
    j @loop
@found
    ret %mid
