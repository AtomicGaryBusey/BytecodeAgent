

export function l $binary_search(l ptr, l len, l key)
@entry
    %minus = subsl 0 1
    %lo = addsl 0 0
    %hi = addsl len 0
    jumps @loop
@loop
    %go = csltl %lo %hi
    jnz %go @body
    ret %minus
@body
    %diff = subsl %hi %lo
    %half = divsl %diff 2
    %mid = addsl %lo %half
    %off1 = addsl %mid %mid
    %off2 = addsl %off1 %off1
    %off3 = addsl %off2 %off2
    %addr = addsl ptr %off3
    %v = loadl %addr
    %eq = ceql %v key
    jnz %eq @found
    %lt = csltl %v key
    jnz %lt @right
    %hi = addsl %mid 0
    jumps @loop
@right
    %lo = addsl %mid 1
    jumps @loop
@found
    ret %mid
