

export function l $binary_search(l a, l n, l key)
@entry
    %lo = addsl 0 0
    %hi = addsl n 0
    jumps @loop
@loop
    %go = csltl %lo %hi
    jnz %go @body
@no
    %r = subsl 0 1
    ret %r
@body
    %diff = subsl %hi %lo
    %half = divsl %diff 2
    %mid = addsl %lo %half
    %off = addsl %mid %mid
    %off2 = addsl %off %off
    %addr = addsl a %off2
    %value = loadl %addr
    %eq = ceql %value key
    jnz %eq @found
@after
    %lt = csltl %value key
    jnz %lt @right
@left
    %hi = addsl %mid 0
    jumps @loop
@right
    %lo = addsl %mid 1
    jumps @loop
@found
    ret %mid
