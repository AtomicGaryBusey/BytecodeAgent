

export function l $binary_search(l, l, l)
@entry
    addsl %lo 0 0
    addsl %hi a1 0
    jumps @loop
@loop
    csltl %go %lo %hi
    jnz %go @body
@no
    subsl %r 0 1
    ret %r
@body
    subsl %diff %hi %lo
    divsl %half %diff 2
    addsl %mid %lo %half
    addsl %twom %mid %mid
    addsl %fourm %twom %twom
    addsl %eightm %fourm %fourm
    addsl %addr a0 %eightm
    loadl %value %addr
    ceql %eq %value a2
    jnz %eq @found
@after
    csltl %lt %value a2
    jnz %lt @right
@left
    addsl %hi %mid 0
    jumps @loop
@right
    addsl %lo %mid 1
    jumps @loop
@found
    ret %mid
