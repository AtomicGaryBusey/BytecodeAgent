

export function l $map_sum(l %a, l %n, l %f)
@entry
    %p =l +l %a, 0
    %k =l +l %n, 0
    %sum =l +l 0, 0
    %c =w ceql %k, 0
    jnz %c, @done
    j @loop
@loop
    %v =l loadl %p, 0
    %r =l call %f(l %v)
    %sum =l +l %sum, %r
    %k =l -l %k, 1
    %p =l +l %p, 8
    %c =w ceql %k, 0
    jnz %c, @done
    j @loop
@done
    lret %sum
end
