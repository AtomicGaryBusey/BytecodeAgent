

export function l $map_sum(l %a, l %n, l %f) {
@entry
    %zero =l addl 0, 0
    %p =l addl %a, %zero
    %k =l addl %n, %zero
    %sum =l addl %zero, %zero
    %c =w ceql %k, %zero
    jnz %c, @done
    j @loop
@loop
    %v =l loadl %p
    %r =l call %f(l %v)
    %sum =l addl %sum, %r
    %k =l subl %k, 1
    %p =l addl %p, 8
    %c =w ceql %k, %zero
    jnz %c, @done
    j @loop
@done
    lret %sum
}
