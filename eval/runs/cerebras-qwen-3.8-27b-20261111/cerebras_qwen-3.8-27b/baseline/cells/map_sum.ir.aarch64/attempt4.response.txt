

export function l $map_sum(l %a, l %n, l %f) {
@entry
    %zero =l move 0
    %one =l move 1
    %eight =l move 8
    %p =l move %a
    %k =l move %n
    %sum =l move %zero
    %c =w ceql %k, %zero
    jnz %c, @done
    j @loop
@loop
    %v =l loadl %p
    %r =l call %f(l %v)
    %sum =l add %sum, %r
    %k =l sub %k, %one
    %p =l add %p, %eight
    %c =w ceql %k, %zero
    jnz %c, @done
    j @loop
@done
    lret %sum
}
