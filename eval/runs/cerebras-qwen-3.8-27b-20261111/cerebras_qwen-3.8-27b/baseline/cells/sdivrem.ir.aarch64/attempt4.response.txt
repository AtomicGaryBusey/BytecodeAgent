

export function l $sdivrem(l a, l b, l rp) {
@entry
    %bzero = ceql %b, 0
    jnz %bzero, @zero
    %bminus = ceql %b, -1
    jnz %bminus, @overflow
    jump @normal
@zero
    storel %a, %rp
    retl 0
@overflow
    %amin = ceql %a, 0x8000000000000000
    jnz %amin, @min
    jump @normal
@min
    storel 0, %rp
    retl 0x8000000000000000
@normal
    %q = divl %a, %b
    %r = reml %a, %b
    storel %r, %rp
    retl %q
}
