

export function l $sdivrem(l a, l b, l rem) {
@entry:
    ceql %bzero, %b, 0
    jnz %bzero, @bzero
    movl %min, 1
    shll %min, %min, 63
    ceql %amin, %a, %min
    jnz %amin, @aismin
    br @normal
@aismin:
    ceql %bnegone, %b, -1
    jnz %bnegone, @overflow
    br @normal
@normal:
    divl %q, %a, %b
    reml %r, %a, %b
    storel %r, %rem
    movl %ret, %q
    ret
@bzero:
    storel %a, %rem
    movl %ret, 0
    ret
@overflow:
    movl %min2, 1
    shll %min2, %min2, 63
    movl %rz, 0
    storel %rz, %rem
    movl %ret, %min2
    ret
}
