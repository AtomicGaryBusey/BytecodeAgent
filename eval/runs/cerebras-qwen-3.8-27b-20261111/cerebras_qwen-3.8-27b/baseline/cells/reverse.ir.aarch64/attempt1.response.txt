@entry:
  %left = movl %a0
  %c = csltl %a1, 2
  jnz %c @done
  %n1 = subl %a1, 1
  %o1 = addl %n1, %n1
  %o2 = addl %o1, %o1
  %o3 = addl %o2, %o2
  %right = addl %left, %o3
  jmp @loop
@loop:
  %t = loadl %left
  %u = loadl %right
  storel %t, %right
  storel %u, %left
  %left = addl %left, 8
  %right = subl %right, 8
  %c = csltl %left, %right
  jnz %c @loop
@done:
  movl 0
