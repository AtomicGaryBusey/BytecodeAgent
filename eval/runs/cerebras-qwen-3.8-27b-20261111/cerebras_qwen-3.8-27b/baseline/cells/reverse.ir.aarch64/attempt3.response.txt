

export function l $reverse(l %a, l %n)
@entry:
  %less = csltl %n, 2
  jnz %less @done
  %off = addl %n, %n
  %off = addl %off, %off
  %off = addl %off, %off
  %right = addl %a, %off
  %right = subl %right, 8
  %left = addl %a, 0
  jmp @loop
@loop:
  %tl = loadl %left
  %tr = loadl %right
  storel %right, %tl
  storel %left, %tr
  %left = addl %left, 8
  %right = subl %right, 8
  %more = csltl %left, %right
  jnz %more @loop
@done:
  %zero = subl %n, %n
  retl %zero
