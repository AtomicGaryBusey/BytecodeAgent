

export function l $reverse(l a, l n)
@entry:
  %z = subl n, n
  %c = csltl n, 2
  jnz %c @done
@body:
  %half = subl n, 1
  %o1 = addl %half, %half
  %o2 = addl %o1, %o1
  %o3 = addl %o2, %o2
  %left = addl a, %z
  %right = addl a, %o3
  jmp @loop
@loop:
  %tl = loadl %left
  %tr = loadl %right
  storel %right, %tl
  storel %left, %tr
  %left = addl %left, 8
  %right = subl %right, 8
  %c = csltl %left, %right
  jnz %c @loop
@exit:
  jmp @done
@done:
  ret %z
