

export function l $bmax(l %a, l %n)
@entry
    %max = loadl %a
    %count = subl %n, $1
    %z = ceql %count, $0
    jnz %z @done
    %p = addl %a, $8
    br @loop
@loop
    %x = loadl %p
    %gt = csltl %max, %x
    jnz %gt @update
    br @next
@update
    %max = addl %x, $0
    br @next
@next
    %p = addl %p, $8
    %count = subl %count, $1
    %z = ceql %count, $0
    jnz %z @done
    br @loop
@done
    retl %max
