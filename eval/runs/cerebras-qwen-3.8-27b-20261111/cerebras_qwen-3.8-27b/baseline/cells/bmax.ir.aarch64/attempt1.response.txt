

export function l $bmax(l %a, l %n)
{
@entry
    %max = loadl %a
    %i = $1l
    jmp @loop
@loop
    %lt = csltl %i, %n
    jnz %lt @body
    jmp @end
@body
    %off = *l %i, $8l
    %p = +l %a, %off
    %v = loadl %p
    %larger = csltl %max, %v
    jnz %larger @update
    jmp @next
@update
    %max = %v
    jmp @next
@next
    %i = +l %i, $1l
    jmp @loop
@end
    retl %max
}
