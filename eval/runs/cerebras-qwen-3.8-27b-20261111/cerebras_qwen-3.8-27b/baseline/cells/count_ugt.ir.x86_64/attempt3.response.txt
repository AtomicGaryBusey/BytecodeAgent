

export function l $count_ugt(l a, l n, l t) {
@entry
    %cnt = constl 0
    %i = constl 0
    %p = a
    jmp @loop
@loop
    %cont = csltl %i, n
    jnz %cont, @body
    retl %cnt
@body
    %v = loadl %p
    %gt = cugtl %v, t
    jnz %gt, @inc
    jmp @step
@inc
    %cnt = addl %cnt, constl 1
    jmp @step
@step
    %i = addl %i, constl 1
    %p = addl %p, constl 8
    jmp @loop
}
