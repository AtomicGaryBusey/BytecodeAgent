

export function l $count_ugt(a, n, t) {
@entry
    %p = a
    %cnt = constl 0
    %rem = n
@loop
    %more = csltl constl 0, %rem
    jnz %more, @body
    jmp @exit
@body
    %v = loadl %p
    %gt = cugtl %v, t
    jnz %gt, @inc
    jmp @next
@inc
    %cnt = addl %cnt, constl 1
    jmp @next
@next
    %p = addl %p, constl 8
    %rem = subl %rem, constl 1
    jmp @loop
@exit
    retl %cnt
}
