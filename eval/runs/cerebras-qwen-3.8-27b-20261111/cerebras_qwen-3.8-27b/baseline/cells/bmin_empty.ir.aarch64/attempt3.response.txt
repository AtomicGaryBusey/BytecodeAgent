@entry
    %zero = ceql cnt, 0
    jnz %zero @out
    %base = addl ptr, 0
    %min = loadl [%base]
    %idx = addl 1, 0
    %addr = addl %base, 8
    j @loop
@loop
    %done = ceql %idx, cnt
    jnz %done @end
    %val = loadl [%addr]
    %lt = csltl %val, %min
    jnz %lt @upd
    j @next
@upd

    j @next
@next
    %idx = addl %idx, 1
    %addr = addl %addr, 8
    j @loop
@end

@out

}
