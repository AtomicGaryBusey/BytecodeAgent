

export function l $bmin_empty (l a, l n) {
@entry
    %empty = ceql n, 0
    jnz %empty @empty_ret
    %min = loadl [a]
    %i = addl 1, 0
    %p = addl a, 8
    j @loop
@loop
    %done = ceql %i, n
    jnz %done @finish
    %v = loadl [%p]
    %less = csltl %v, %min
    jnz %less @update
    j @advance
@update
    %min = addl %v, 0
    j @advance
@advance
    %i = addl %i, 1
    %p = addl %p, 8
    j @loop
@finish
    retl %min
@empty_ret
    retl 0x7fffffffffffffff
}
