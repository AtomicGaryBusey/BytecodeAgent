

export function l $bubble_sort(a, n) -> (r) {
@entry:
    %zero = l 0
    %one = l 1
    %two = l 2
    %eight = l 8
    %ce = w csltl n, %two
    jnz %ce, @done
    %i = l 0
    j @outer_test
@outer_test:
    %os1 = l addl %i, %one
    %os2 = l addl %os1, %one
    %co = w csltl %os2, n
    jnz %co, @outer_body
    retl %zero
@outer_body:
    %j = l 0
    %p = l addl a, %zero
    j @inner_test
@inner_test:
    %is1 = l addl %j, %i
    %is2 = l addl %is1, %one
    %ci = w csltl %is2, n
    jnz %ci, @inner_body
    %i = l addl %i, %one
    j @outer_test
@inner_body:
    %pn = l addl %p, %eight
    %x = l loadl (%p, 0)
    %y = l loadl (%pn, 0)
    %cs = w csltl %y, %x
    jnz %cs, @do_swap
    j @after_swap
@do_swap:
    storel (%p, 0), %y
    storel (%pn, 0), %x
    j @after_swap
@after_swap:
    %p = l addl %p, %eight
    %j = l addl %j, %one
    j @inner_test
@done:
    retl %zero
}
