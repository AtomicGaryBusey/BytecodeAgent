

export function l $bubble_sort(%a, %n) -> (%r) {
@entry:
    %zero = l 0
    %one = l 1
    %eight = l 8
    %r = l 0
    %i = l 0
    j @outer
@outer:
    %ti = l addl %i, %one
    %co = w csltl %ti, %n
    jnz %co, @outer_body
    retl %r
@outer_body:
    %j = l 0
    %p = l addl %a, %zero
    %q = l addl %p, %eight
    j @inner
@inner:
    %t = l addl %j, %i
    %t1 = l addl %t, %one
    %ci = w csltl %t1, %n
    jnz %ci, @inner_body
    %i = l addl %i, %one
    j @outer
@inner_body:
    %x = l loadl (%p, 0)
    %y = l loadl (%q, 0)
    %cmp = w csltl %y, %x
    jnz %cmp, @swap
    j @step
@swap:
    storel (%p, 0), %y
    storel (%q, 0), %x
    j @step
@step:
    %p = l addl %p, %eight
    %q = l addl %q, %eight
    %j = l addl %j, %one
    j @inner
}
