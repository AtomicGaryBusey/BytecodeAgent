

export function l $map_sum(l %a, l %n, l %f) {
@entry
 %zero =l const 0
 %one =l const 1
 %eight =l const 8
 %i =l const 0
 %sum =l const 0
 %p =l add %a %zero
@check
 %c =w csltl %i %n
 jnz @body %c
@exit
 ret %sum
@body
 %x =l loadl %p
 %y =l call %f(l %x)
 %sum =l add %sum %y
 %i =l add %i %one
 %p =l add %p %eight
 jmp @check
}
