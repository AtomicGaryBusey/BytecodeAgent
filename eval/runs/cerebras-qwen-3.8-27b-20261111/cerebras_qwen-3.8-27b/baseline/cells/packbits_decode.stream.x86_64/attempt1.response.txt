

packbits_decode:
cld
lea r8, [rdi + rsi]
mov r9, rcx
.Lloop:
cmp rdi, r8
je .Ldone
movsx rax, byte ptr [rdi]
inc rdi
cmp rax, 0
jl .Lneg
lea r10, [rax + 1]
mov r11, r10
jmp .Lcheck
.Lneg:
cmp rax, -128
je .Lnoop
neg rax
inc rax
mov r10, rax
mov r11, 1
jmp .Lcheck
.Lnoop:
mov r10, 0
mov r11, 0
.Lcheck:
mov rsi, r8
sub rsi, rdi
cmp rsi, r11
jb .Lsrc_err
cmp r10, rcx
jbe .Lok
cmp r10, r11
je .Lpart_lit
.Lpart_rep:
test rcx, rcx
jz .Lret_neg1
mov al, byte ptr [rdi]
.Lpart_rep_loop:
mov byte ptr [rdx], al
inc rdx
dec rcx
jne .Lpart_rep_loop
jmp .Lret_neg1
.Lpart_lit:
test rcx, rcx
jz .Lret_neg1
.Lpart_lit_loop:
mov al, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec rcx
jne .Lpart_lit_loop
.Lret_neg1:
mov rax, -1
ret
.Lok:
cmp r11, 0
je .Lnext
cmp r10, r11
je .Llit_full
.Lrep_full:
mov al, byte ptr [rdi]
mov r11, r10
.Lrep_full_loop:
mov byte ptr [rdx], al
inc rdx
dec r11
jne .Lrep_full_loop
lea rdi, [rdi + 1]
sub rcx, r10
jmp .Lnext
.Llit_full:
mov r11, r10
.Llit_full_loop:
mov al, byte ptr [rdi]
mov byte ptr [rdx], al
inc rdi
inc rdx
dec r11
jne .Llit_full_loop
sub rcx, r10
.Lnext:
jmp .Lloop
.Ldone:
mov rax, r9
sub rax, rcx
ret
.Lsrc_err:
mov rax, -2
ret
