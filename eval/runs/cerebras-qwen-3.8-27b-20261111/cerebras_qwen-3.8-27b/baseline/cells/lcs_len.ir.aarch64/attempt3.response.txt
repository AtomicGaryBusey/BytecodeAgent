

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
%zero = addl 0, 0
%one = addl 1, 0
%eight = addl 8, 0
%n1 = addl %n, %one
%cnt = addl %n1, 0
%p = addl %row, 0
j @init
@init
storel %zero, %p
%p = addl %p, %eight
%cnt = subl %cnt, %one
%cinit = csltl %zero, %cnt
jnz %cinit, @init
@after_init
%xptr = addl %x, 0
%i = addl 0, 0
j @outer
@outer
%couter = csltl %i, %m
jnz %couter, @inner_body
j @finish
@inner_body
%xi =w loadub %xptr
%diag = addl 0, 0
%j = addl 1, 0
%rowj = addl %row, %eight
%rowm1 = addl %row, 0
%yj = addl %y, 0
j @inner
@inner
%cinner = csltl %j, %n1
jnz %cinner, @inner_exec
j @inner_exit
@inner_exec
%up = loadl %rowj
%left = loadl %rowm1
%yi =w loadub %yj
%match = ceqw %xi, %yi
jnz %match, @match
%cmplu = csltl %left, %up
jnz %cmplu, @pick_up
storel %left, %rowj
j @step
@pick_up
storel %up, %rowj
j @step
@match
%diagp = addl %diag, %one
storel %diagp, %rowj
j @step
@step
%diag = addl %up, 0
%j = addl %j, %one
%rowj = addl %rowj, %eight
%rowm1 = addl %rowm1, %eight
%yj = addl %yj, %one
j @inner
@inner_exit
%xptr = addl %xptr, %one
%i = addl %i, %one
j @outer
@finish
%last = subl %p, %eight
%res = loadl %last
ret %res
}
