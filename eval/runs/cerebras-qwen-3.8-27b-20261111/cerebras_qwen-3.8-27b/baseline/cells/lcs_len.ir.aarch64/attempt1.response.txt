

export function l $lcs_len(l x, l m, l y, l n, l row)
  @entry
  %zero = addl 0, 0
  %eight = addl 8, 0
  %one = addl 1, 0
  %npl = addl %n, 1
  %cnt = addl %npl, 0
  %p = addl %row, 0
  %row_end = addl %row, 0
  j @zero_row
  @zero_row
  storel %p, %zero
  %row_end = addl %p, 0
  %p = addl %p, %eight
  %cnt = subl %cnt, %one
  %cinit = csltl %zero, %cnt
  jnz %cinit, @zero_row
  @zero_row_done
  %xptr = addl %x, 0
  %i = addl 0, 0
  j @outer_check
  @outer_check
  %couter = csltl %i, %m
  jnz %couter, @outer_body
  @outer_exit
  j @done
  @outer_body
  %xi =w loadub %xptr
  %diag = addl 0, 0
  %j = addl 1, 0
  %rowcur = addl %row, %eight
  %rowprev = addl %row, 0
  %yptr = addl %y, 0
  j @inner_check
  @inner_check
  %cinner = csltl %j, %npl
  jnz %cinner, @inner_body
  @inner_exit
  j @inner_end
  @inner_body
  %up = loadl %rowcur
  %left = loadl %rowprev
  %yj =w loadub %yptr
  %nextdiag = addl %up, 0
  %cm = ceql %xi, %yj
  jnz %cm, @match
  @mismatch
  %cmax = csltl %up, %left
  jnz %cmax, @left_max
  @up_max
  %max = addl %up, 0
  j @store_done
  @left_max
  %max = addl %left, 0
  j @store_done
  @store_done
  storel %rowcur, %max
  j @inner_next
  @match
  %diagp = addl %diag, %one
  storel %rowcur, %diagp
  j @inner_next
  @inner_next
  %diag = addl %nextdiag, 0
  %j = addl %j, %one
  %rowcur = addl %rowcur, %eight
  %rowprev = addl %rowprev, %eight
  %yptr = addl %yptr, %one
  j @inner_check
  @inner_end
  %i = addl %i, %one
  %xptr = addl %xptr, %one
  j @outer_check
  @done
  %result = loadl %row_end
  ret %result
