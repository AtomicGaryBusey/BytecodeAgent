

export function l $count_nonzero_prefix(l, l)
@start:
    addl %i = #0, #0
    csltl %go = %i, %1
    jnz %go, @loop
@empty:
    retl %i
@loop:
    mull %off = %i, #8
    addl %addr = %0, %off
    loadl %x = %addr
    ceql %zero = %x, #0
    jnz %zero, @done
@next:
    addl %i = %i, #1
    csltl %more = %i, %1
    jnz %more, @loop
@done:
    retl %i
