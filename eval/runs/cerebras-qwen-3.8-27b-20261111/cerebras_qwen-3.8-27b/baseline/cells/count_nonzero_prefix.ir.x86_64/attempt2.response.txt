

export function l $count_nonzero_prefix(l a, l n)
@start
    addl %i = #0, #0
    csltl %go = %i, n
    jnz %go, @body
    retl %i
@body
    mull %off = %i, #8
    addl %addr = a, %off
    loadl %x = %addr
    ceql %zero = %x, #0
    jnz %zero, @exit
    addl %i = %i, #1
    csltl %more = %i, n
    jnz %more, @body
    retl %i
@exit
    retl %i
