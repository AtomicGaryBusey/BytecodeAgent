

export function l $count_nonzero_prefix(l a, l n)
@start:
    movl %i = #0l
    movl %p = a
    csltl %more = %i, n
    jnz %more, @loop
@zero:
    retl %i
@loop:
    loadl %x = %p
    ceql %zero = %x, #0l
    jnz %zero, @done
@next:
    addl %i = %i, #1l
    addl %p = %p, #8l
    csltl %more = %i, n
    jnz %more, @loop
@done:
    retl %i
