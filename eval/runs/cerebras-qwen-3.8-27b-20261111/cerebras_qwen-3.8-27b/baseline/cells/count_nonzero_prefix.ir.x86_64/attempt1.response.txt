

export function l $count_nonzero_prefix(a, n) l, l
@start
    addl %i = l #0, l #0
    csltl %go_init = %i, n
    jnz %go_init, @body
@exit_init
    retl %i
@body
    mull %off = %i, l #8
    addl %addr = a, %off
    loadl %x = %addr
    ceql %is_zero = %x, l #0
    jnz %is_zero, @exit
@incr
    addl %i = %i, l #1
    csltl %go_next = %i, n
    jnz %go_next, @body
@exit
    retl %i
