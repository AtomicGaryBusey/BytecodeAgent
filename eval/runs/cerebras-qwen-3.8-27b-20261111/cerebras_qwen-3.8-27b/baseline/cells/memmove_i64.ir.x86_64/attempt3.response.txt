

export function l $memmove_i64(l %dst, l %src, l %cnt) {
@entry
        ceql %zero, %cnt, 0
        jnz @zero_ret, %zero
        csltl %fwd, %dst, %src
        jnz @fwd_setup, %fwd
        jmp @back_setup
@zero_ret
        movl %ret, 0
        retl %ret
@fwd_setup
        movl %sptr, %src
        movl %dptr, %dst
        movl %i, 0
        jmp @fwd_loop
@fwd_loop
        ceql %done, %i, %cnt
        jnz @zero_ret, %done
        loadl %val, %sptr
        storel %dptr, %val
        addl %sptr, %sptr, 8
        addl %dptr, %dptr, 8
        addl %i, %i, 1
        jmp @fwd_loop
@back_setup
        movl %sptr, %src
        movl %dptr, %dst
        movl %i, %cnt
        jmp @back_advance
@back_advance
        ceql %zero, %i, 0
        jnz @back_loop, %zero
        addl %sptr, %sptr, 8
        addl %dptr, %dptr, 8
        subl %i, %i, 1
        jmp @back_advance
@back_loop
        ceql %done, %i, %cnt
        jnz @zero_ret, %done
        subl %sptr, %sptr, 8
        subl %dptr, %dptr, 8
        loadl %val, %sptr
        storel %dptr, %val
        addl %i, %i, 1
        jmp @back_loop
}
