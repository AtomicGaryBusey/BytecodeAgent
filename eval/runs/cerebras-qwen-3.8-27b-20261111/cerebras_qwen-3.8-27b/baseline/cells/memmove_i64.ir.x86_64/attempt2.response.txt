

export function l $memmove_i64(l %dst, l %src, l %cnt) {
@entry
        ceql %cond, %cnt, 0
        jnz @ret, %cond
        csltl %cond, %dst, %src
        jnz @fwd, %cond
        jmp @bset
@bset
        movl %off, 0
        movl %i, 0
        jmp @bsetup
@bsetup
        ceql %cond, %i, %cnt
        jnz @back, %cond
        addl %off, %off, 8
        addl %i, %i, 1
        jmp @bsetup
@back
        subl %off, %off, 8
        addl %sptr, %src, %off
        addl %dptr, %dst, %off
        jmp @bloop
@bloop
        ceql %cond, %i, 0
        jnz @ret, %cond
        loadl %val, %sptr
        storel %dptr, %val
        subl %sptr, %sptr, 8
        subl %dptr, %dptr, 8
        subl %i, %i, 1
        jmp @bloop
@fwd
        movl %sptr, %src
        movl %dptr, %dst
        movl %i, 0
        jmp @floop
@floop
        ceql %cond, %i, %cnt
        jnz @ret, %cond
        loadl %val, %sptr
        storel %dptr, %val
        addl %sptr, %sptr, 8
        addl %dptr, %dptr, 8
        addl %i, %i, 1
        jmp @floop
@ret
        movl %r, 0
        retl %r
}
