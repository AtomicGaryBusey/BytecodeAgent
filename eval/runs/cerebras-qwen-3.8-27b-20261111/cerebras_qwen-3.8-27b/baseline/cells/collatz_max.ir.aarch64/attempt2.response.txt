

export function l $collatz_max(l %n) {
@.entry
  %limit = addl %n, 1
  %k = addl 0, 1
  %max = addl 0, 0
  j @.outer
@.outer
  %outc = csltl %k, %limit
  jnz @.inner, %outc
  retl %max
@.inner
  %x = addl %k, 0
  %steps = addl 0, 0
  j @.collatz
@.collatz
  %done = ceql %x, 1
  jnz @.done, %done
  %lsb = andl %x, 1
  %even = ceql %lsb, 0
  jnz @.even, %even
  %triple = mull %x, 3
  %x = addl %triple, 1
  j @.step
@.even
  %x = divl %x, 2
  j @.step
@.step
  %steps = addl %steps, 1
  j @.collatz
@.done
  %upd = csltl %max, %steps
  jnz @.update, %upd
  j @.next
@.update
  %max = addl %steps, 0
  j @.next
@.next
  %k = addl %k, 1
  j @.outer
}
