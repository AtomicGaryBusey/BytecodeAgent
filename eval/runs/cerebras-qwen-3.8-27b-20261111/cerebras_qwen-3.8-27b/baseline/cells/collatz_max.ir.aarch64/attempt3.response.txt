

export function l $collatz_max(l %n) {
@.entry
  %limit = add %n, 1
  %k = add 1, 0
  %max = add 0, 0
  j @.outer
@.outer
  %outer_c = csltl %k, %limit
  jnz @.inner, %outer_c
@.ret
  retl %max
@.inner
  %x = add %k, 0
  %steps = add 0, 0
  j @.collatz
@.collatz
  %done = ceql %x, 1
  jnz @.done, %done
@.not_done
  %half = shr %x, 1
  %reconst = add %half, %half
  %even = ceql %x, %reconst
  jnz @.even, %even
@.odd
  %twice = add %x, %x
  %three = add %twice, %x
  %x = add %three, 1
  %steps = add %steps, 1
  j @.collatz
@.even
  %x = add %half, 0
  %steps = add %steps, 1
  j @.collatz
@.done
  %upd = csltl %max, %steps
  jnz @.update, %upd
@.no_update
  %k = add %k, 1
  j @.outer
@.update
  %max = add %steps, 0
  %k = add %k, 1
  j @.outer
}
