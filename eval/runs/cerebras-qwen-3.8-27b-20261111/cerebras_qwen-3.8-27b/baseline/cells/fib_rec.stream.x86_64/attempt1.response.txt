

fib_rec:
    cmp rdi, 1
    jle .Lbase
    push rbp
    sub rsp, 16
    mov qword ptr [rsp + 8], rdi
    dec rdi
    call fib_rec
    mov qword ptr [rsp], rax
    mov rdi, qword ptr [rsp + 8]
    dec rdi
    call fib_rec
    add rax, qword ptr [rsp]
    add rsp, 16
    pop rbp
    ret
.Lbase:
    mov rax, rdi
    ret
