

qsort_pairs:
    sub sp, sp, #80
    stp x29, x30, [sp]
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    str x25, [sp, #64]
    mov x19, x0
    mov x20, x1
    cbz x20, .Lret
    cmp x20, #2
    b.lt .Lret
    ldr x21, [x19]
    ldr x22, [x19, #8]
    mov x0, #1
    mov x23, #0
.Lloop:
    cmp x0, x20
    b.ge .Lafter
    add x10, x19, x0, lsl #4
    ldr x11, [x10]
    ldr x12, [x10, #8]
    cmp x11, x21
    b.lt .Lswap
    b.ne .Lnext
    cmp x12, x22
    b.lt .Lswap
.Lnext:
    add x0, x0, #1
    b .Lloop
.Lswap:
    add x23, x23, #1
    add x13, x19, x23, lsl #4
    ldr x14, [x13]
    ldr x15, [x13, #8]
    str x11, [x13]
    str x12, [x13, #8]
    str x14, [x10]
    str x15, [x10, #8]
    add x0, x0, #1
    b .Lloop
.Lafter:
    add x13, x19, x23, lsl #4
    ldr x14, [x13]
    ldr x15, [x13, #8]
    ldr x11, [x19]
    ldr x12, [x19, #8]
    str x14, [x13]
    str x15, [x13, #8]
    str x11, [x19]
    str x12, [x19, #8]
    mov x10, x23
    add x10, x10, #1
    lsl x10, x10, #4
    add x24, x19, x10
    sub x25, x20, x23
    sub x25, x25, #1
    mov x0, x19
    mov x1, x23
    bl qsort_pairs
    mov x0, x24
    mov x1, x25
    bl qsort_pairs
.Lret:
    ldp x29, x30, [sp]
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x23, x24, [sp, #48]
    ldr x25, [sp, #64]
    add sp, sp, #80
    mov x0, #0
    ret
