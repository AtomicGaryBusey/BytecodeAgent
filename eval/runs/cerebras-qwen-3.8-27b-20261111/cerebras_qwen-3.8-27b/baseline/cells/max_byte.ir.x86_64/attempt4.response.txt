

export function l $max_byte(l %p, l %n)
@entry:
    %is_zero =w ceql %n, 0l
    jnz %is_zero, @zero
@first:
    %max =l loadub %p
    %ptr =l addl %p, 1l
    %rem =l subl %n, 1l
    jmp @loop
@loop:
    %has_more =w csltl 0l, %rem
    jnz %has_more, @body
@done:
    retl %max
@body:
    %b =l loadub %ptr
    %should_update =w csltl %max, %b
    jnz %should_update, @update
@next:
    %ptr =l addl %ptr, 1l
    %rem =l subl %rem, 1l
    jmp @loop
@update:
    %max =l loadub %ptr
    jmp @next
@zero:
    retl 0l
