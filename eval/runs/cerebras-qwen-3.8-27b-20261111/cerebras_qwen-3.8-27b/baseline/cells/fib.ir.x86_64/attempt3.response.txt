

export function l $fib(l %n)
{
@entry
  %zero = addl 0, 0
  %one = addl 0, 1
  %i = addl %zero, %zero
  %a = addl %zero, %zero
  %b = addl %one, %zero
  %c = csltl %i, %n
  jnz %c, @loop
  retl %a
@loop
  %s = addl %a, %b
  %a = addl %b, %zero
  %b = addl %s, %zero
  %i = addl %i, %one
  %c = csltl %i, %n
  jnz %c, @loop
  retl %a
}
