

collatz_max:
    mov x1, #0
    mov x2, #1
    cmp x2, x0
    b.gt .Lend
.Lloop:
    mov x3, x2
    mov x4, #0
.Linner:
    cmp x3, #1
    beq .Lupdate
    tbz x3, #0, .Leven
    add x3, x3, x3, lsl #1
    add x3, x3, #1
    add x4, x4, #1
    b .Linner
.Leven:
    lsr x3, x3, #1
    add x4, x4, #1
    b .Linner
.Lupdate:
    cmp x4, x1
    csel x1, x4, x1, gt
    add x2, x2, #1
    cmp x2, x0
    b.le .Lloop
.Lend:
    mov x0, x1
    ret
