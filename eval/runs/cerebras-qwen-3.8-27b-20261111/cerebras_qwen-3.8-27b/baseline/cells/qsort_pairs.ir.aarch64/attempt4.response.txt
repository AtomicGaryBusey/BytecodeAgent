

export function l $qsort_pairs(l %a, l %n)
{
@entry
%small =w csltl %n, 2
jnz %small, @ret
%cnt =l subl %n, 1
%last =l addl %a, 0
j @findlast
@findlast
%c =w csltl %cnt, 1
jnz %c, @found
%last =l addl %last, 16
%cnt =l subl %cnt, 1
j @findlast
@found
%lastv =l addl %last, 8
%pk =l loadl %last
%pv =l loadl %lastv
%low =l addl %a, 0
%j =l addl %a, 0
%lowcnt =l addl 0, 0
j @ploop
@ploop
%eq =w ceql %j, %last
jnz %eq, @done
%jv =l addl %j, 8
%k =l loadl %j
%v =l loadl %jv
%kg =w csltl %pk, %k
jnz %kg, @skip
%ke =w ceql %pk, %k
jnz %ke, @checkv
j @move
@checkv
%vg =w csltl %pv, %v
jnz %vg, @skip
j @move
@skip
j @next
@move
%lov =l addl %low, 8
%lk =l loadl %low
%lv =l loadl %lov
storel %k, %low
storel %v, %lov
storel %lk, %j
storel %lv, %jv
%low =l addl %low, 16
%lowcnt =l addl %lowcnt, 1
j @next
@next
%j =l addl %j, 16
j @ploop
@done
%lov =l addl %low, 8
%lk =l loadl %low
%lv =l loadl %lov
%lk2 =l loadl %last
%lv2 =l loadl %lastv
storel %lk2, %low
storel %lv2, %lov
storel %lk, %last
storel %lv, %lastv
%left =l addl %lowcnt, 0
%mid =l addl %left, 1
%right =l subl %n, %mid
j @right
@right
%cr =w csltl %right, 2
jnz %cr, @left
%rptr =l addl %low, 16
%rc =l call $qsort_pairs(l %rptr, l %right)
j @left
@left
%cl =w csltl %left, 2
jnz %cl, @ret
%lc =l call $qsort_pairs(l %a, l %left)
j @ret
@ret
%zero =l addl 0, 0
retl %zero
}
