

bubble_sort:
  cmp x1, #2
  b.lt .Ldone
  mov x9, x0
  add x14, x0, x1, lsl #3
  sub x14, x14, #8
.Louter:
  mov x15, x9
  mov x13, #0
.Linner:
  cmp x15, x14
  b.hs .Lnext_pass
  add x2, x15, #8
  ldr x3, [x15]
  ldr x4, [x2]
  cmp x3, x4
  b.le .Lno_swap
  str x4, [x15]
  str x3, [x2]
  mov x13, #1
.Lno_swap:
  add x15, x15, #8
  b .Linner
.Lnext_pass:
  cbz x13, .Ldone
  sub x14, x14, #8
  cmp x14, x9
  b.ls .Ldone
  b .Louter
.Ldone:
  mov x0, #0
  ret
