

function l $fib_rec(l @n)
@entry:
    %c0 =w ceql l @n, l 0
    jnz %c0, @zero
@gt0:
    %c1 =w ceql l @n, l 1
    jnz %c1, @one
@rec:
    %n1 =l subl l @n, l 1
    %r1 =l call $fib_rec(l %n1)
    %n2 =l subl l @n, l 2
    %r2 =l call $fib_rec(l %n2)
    %res =l addl l %r1, l %r2
    retl %res
@zero:
    %z =l movl l 0
    retl %z
@one:
    %o =l movl l 1
    retl %o
end
