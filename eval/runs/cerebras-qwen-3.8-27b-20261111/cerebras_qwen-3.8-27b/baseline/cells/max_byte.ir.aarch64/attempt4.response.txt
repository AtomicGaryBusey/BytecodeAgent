

export function l $max_byte(l %p, l %n) {
@begin:
 %max =l addl 0, 0
 %i =l addl 0, 0
 b @loop
@loop:
 %cont =w csltl %i, %n
 jnz %cont, @body
 b @done
@body:
 %addr =l addl %p, %i
 %b =l loadub %addr
 %upd =w csltl %max, %b
 jnz %upd, @update
 b @next
@update:
 %max =l addl %b, 0
 b @next
@next:
 %i =l addl %i, 1
 b @loop
@done:
 retl %max
}
