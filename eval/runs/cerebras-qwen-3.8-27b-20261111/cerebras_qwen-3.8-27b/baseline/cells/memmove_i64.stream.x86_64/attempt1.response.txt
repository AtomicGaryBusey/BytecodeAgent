

memmove_i64:
    cld
    test rdx, rdx
    jz .Lret
    cmp rsi, rdi
    jb .Lbackward
.Lforward:
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    add rsi, 8
    add rdi, 8
    dec rdx
    jnz .Lforward
    xor eax, eax
    ret
.Lbackward:
    dec rdx
.Lbackward_loop:
    mov rax, qword ptr [rsi + rdx*8]
    mov qword ptr [rdi + rdx*8], rax
    dec rdx
    jns .Lbackward_loop
    xor eax, eax
    ret
.Lret:
    xor eax, eax
    ret
