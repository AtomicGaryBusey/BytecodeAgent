

bubble_sort:
    xor eax, eax
    mov r10, rsi
    dec r10
    cmp r10, 0
    jle .Ldone
.Louter:
    mov r11, 0
.Linner:
    cmp r11, r10
    jge .Lnextpass
    mov r8, qword ptr [rdi + r11*8]
    mov r9, qword ptr [rdi + r11*8 + 8]
    cmp r8, r9
    jle .Lnoswap
    mov qword ptr [rdi + r11*8], r9
    mov qword ptr [rdi + r11*8 + 8], r8
.Lnoswap:
    inc r11
    jmp .Linner
.Lnextpass:
    dec r10
    cmp r10, 0
    jg .Louter
.Ldone:
    ret
