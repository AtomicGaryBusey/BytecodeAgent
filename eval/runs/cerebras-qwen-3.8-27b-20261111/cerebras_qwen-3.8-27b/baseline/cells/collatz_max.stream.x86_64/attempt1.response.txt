

collatz_max:
cld
mov rax, 0
mov rcx, 1
.Louter:
cmp rcx, rdi
ja .Ldone
mov rdx, rcx
mov r8, 0
.Linner:
cmp rdx, 1
je .Lfinish
test rdx, 1
jz .Leven
imul rdx, rdx, 3
add rdx, 1
jmp .Lcount
.Leven:
shr rdx, 1
.Lcount:
inc r8
jmp .Linner
.Lfinish:
cmp r8, rax
jbe .Lnext
mov rax, r8
.Lnext:
inc rcx
jmp .Louter
.Ldone:
ret
