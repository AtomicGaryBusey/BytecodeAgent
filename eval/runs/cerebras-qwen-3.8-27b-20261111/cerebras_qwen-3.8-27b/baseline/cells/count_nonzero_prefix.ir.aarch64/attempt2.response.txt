

export function l $count_nonzero_prefix(l %a, l %n)
@L0:
    l %i = constl 0
    l %zero = constl 0
    l %one = constl 1
    l %eight = constl 8
    l %p = addl %a, %zero
    jump @L1
@L1:
    l %c = csltl %i, %n
    jnz %c, @L3
@L2:
    retl %i
@L3:
    l %v = loadl %p
    l %z = ceql %v, %zero
    jnz %z, @L2
@L4:
    l %p = addl %p, %eight
    l %i = addl %i, %one
    jump @L1
