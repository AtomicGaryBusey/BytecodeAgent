

export function l $count_nonzero_prefix(l %a, l %n)
{
@L0
    l %zero = 0
    l %one = 1
    l %eight = 8
    l %i = 0
    l %p = addl %a, %zero
    jump @L1
@L1
    w %lt = csltl %i, %n
    jnz %lt, @L3
@L2
    retl %i
@L3
    l %v = loadl %p
    w %eq = ceql %v, %zero
    jnz %eq, @L2
@L4
    l %p = addl %p, %eight
    l %i = addl %i, %one
    jump @L1
}
