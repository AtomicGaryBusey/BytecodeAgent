@entry
    subl %zero, %x, %x
    addl %sixtyfour, %zero, 64
    subl %minusone, %zero, 1
    csltl %lt64, %n, %sixtyfour
    jnz %lt64, @small
    csltl %neg, %x, %zero
    jnz %neg, @neg
    retl %zero
@neg
    retl %minusone
@small
    shll %left, %x, %n
    shrsl %right, %x, %n
    addl %sum, %left, %right
    retl %sum
}
