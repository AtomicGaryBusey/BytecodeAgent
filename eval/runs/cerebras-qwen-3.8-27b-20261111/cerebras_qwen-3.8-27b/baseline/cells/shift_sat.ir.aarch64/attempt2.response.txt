

export function l $shift_sat(l %x, l %n)
@entry
    movl %zero, 0
    movl %minusone, -1
    movl %sixtyfour, 64
    csltl %lt, %n, %sixtyfour
    jnz %lt, @small
    csltl %neg, %x, %zero
    jnz %neg, @neg
    retl %zero
@neg
    retl %minusone
@small
    shll %lhs, %x, %n
    shrsl %rhs, %x, %n
    addl %r, %lhs, %rhs
    retl %r
