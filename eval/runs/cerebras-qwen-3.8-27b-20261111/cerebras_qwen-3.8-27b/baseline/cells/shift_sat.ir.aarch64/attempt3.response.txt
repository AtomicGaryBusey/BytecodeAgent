

export function l $shift_sat(l %x, l %n) {
@entry
    movl %zero, 0
    movl %minusone, -1
    movl %sixtyfour, 64
    csltl %less64, %n, %sixtyfour
    jnz %less64, @small
@large
    csltl %negative, %x, %zero
    jnz %negative, @neg
@nonneg
    retl %zero
@neg
    retl %minusone
@small
    shll %lhs, %x, %n
    shrsl %rhs, %x, %n
    addl %res, %lhs, %rhs
    retl %res
}
