

export function l $packbits_decode(l src, l n, l out, l cap)
{
 @entry
  %si =l addl %0, %0
  %oi =l addl %0, %0
  jmp @loop
 @loop
  %avail =l subl %n, %si
  %done =w ceql %avail, %0
  jnz %done, @done
 @header
  %hp =l addl %src, %si
  %hw =w loadub %hp
  %hl =l zextl %hw
  %lt =w csltl %hl, %128
  jnz %lt, @literal_check
 @header_not_lt
  %eq =w ceql %hl, %128
  jnz %eq, @noop
 @repeat_check
  %rcnt =l subl %257, %hl
  %rok =w cslel %2, %avail
  jnz %rok, @repeat_out
 @repeat_err
  jmp @err2
 @literal_check
  %lcnt =l addl %hl, %1
  %need =l addl %lcnt, %1
  %lok =w cslel %need, %avail
  jnz %lok, @literal_out
 @literal_err
  jmp @err2
 @literal_out
  %lnewoi =l addl %oi, %lcnt
  %lfits =w cslel %lnewoi, %cap
  jnz %lfits, @literal_full
 @literal_partial
  %pcnt =l subl %cap, %oi
  %sptr =l addl %hp, %1
  %optr =l addl %out, %oi
  %send =l addl %optr, %pcnt
  jmp @literal_partial_loop
 @literal_partial_loop
  %pd =w ceql %optr, %send
  jnz %pd, @retm1
 @literal_partial_body
  %c =w loadub %sptr
  storeb %c, %optr
  %sptr =l addl %sptr, %1
  %optr =l addl %optr, %1
  jmp @literal_partial_loop
 @literal_full
  %sptr =l addl %hp, %1
  %optr =l addl %out, %oi
  %send =l addl %sptr, %lcnt
  jmp @literal_full_loop
 @literal_full_loop
  %fd =w ceql %sptr, %send
  jnz %fd, @literal_done
 @literal_full_body
  %c =w loadub %sptr
  storeb %c, %optr
  %sptr =l addl %sptr, %1
  %optr =l addl %optr, %1
  jmp @literal_full_loop
 @literal_done
  %oi =l addl %oi, %lcnt
  %si =l addl %si, %need
  jmp @loop
 @noop
  %si =l addl %si, %1
  jmp @loop
 @repeat_out
  %rnewoi =l addl %oi, %rcnt
  %rfits =w cslel %rnewoi, %cap
  jnz %rfits, @repeat_full
 @repeat_partial
  %pcnt =l subl %cap, %oi
  %vp =l addl %hp, %1
  %val =w loadub %vp
  %optr =l addl %out, %oi
  %send =l addl %optr, %pcnt
  jmp @repeat_partial_loop
 @repeat_partial_loop
  %rd =w ceql %optr, %send
  jnz %rd, @retm1
 @repeat_partial_body
  storeb %val, %optr
  %optr =l addl %optr, %1
  jmp @repeat_partial_loop
 @repeat_full
  %vp =l addl %hp, %1
  %val =w loadub %vp
  %optr =l addl %out, %oi
  %send =l addl %optr, %rcnt
  jmp @repeat_full_loop
 @repeat_full_loop
  %rd =w ceql %optr, %send
  jnz %rd, @repeat_done
 @repeat_full_body
  storeb %val, %optr
  %optr =l addl %optr, %1
  jmp @repeat_full_loop
 @repeat_done
  %oi =l addl %oi, %rcnt
  %si =l addl %si, %2
  jmp @loop
 @done
  retl %oi
 @retm1
  %m =l subl %0, %1
  retl %m
 @err2
  %m =l subl %0, %2
  retl %m
}
