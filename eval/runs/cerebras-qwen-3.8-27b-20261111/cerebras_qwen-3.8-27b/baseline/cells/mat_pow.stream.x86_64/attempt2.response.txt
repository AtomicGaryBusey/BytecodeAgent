

mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
push rsi
mov rbx, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov r12, r8
mov r9, 0
.Linit_row:
cmp r9, r13
jae .Linit_done
mov rax, r9
imul rax, r13
lea rdi, [rbx + rax*8]
lea rsi, [r12 + rax*8]
lea rcx, [r15 + rax*8]
mov r10, 0
.Linit_col:
cmp r10, r13
jae .Linit_row_done
mov rax, qword ptr [rdi + r10*8]
mov qword ptr [rsi + r10*8], rax
cmp r9, r10
je .Linit_one
xor rax, rax
jmp .Linit_store
.Linit_one:
mov rax, 1
.Linit_store:
mov qword ptr [rcx + r10*8], rax
inc r10
jmp .Linit_col
.Linit_row_done:
inc r9
jmp .Linit_row
.Linit_done:
.Lpow:
test r14, r14
jz .Lzero
test r14, 1
jz .Lshift
mov rdi, r15
mov rsi, r12
mov rax, r13
imul rax, r13
lea rcx, [r12 + rax*8]
mov r9, 0
.Lmul_odd:
cmp r9, r13
jae .Lmul_odd_done
mov rax, r9
imul rax, r13
lea rdx, [rdi + rax*8]
lea r8, [rcx + rax*8]
mov r10, 0
.Lmul_odd_j:
cmp r10, r13
jae .Lmul_odd_next_i
xor rbx, rbx
lea rbp, [rsi + r10*8]
mov r11, 0
.Lmul_odd_t:
cmp r11, r13
jae .Lmul_odd_t_end
mov rax, qword ptr [rdx + r11*8]
imul rax, qword ptr [rbp]
add rbx, rax
inc r11
lea rbp, [rbp + r13*8]
jmp .Lmul_odd_t
.Lmul_odd_t_end:
mov qword ptr [r8], rbx
inc r10
lea r8, [r8 + 8]
jmp .Lmul_odd_j
.Lmul_odd_next_i:
inc r9
jmp .Lmul_odd
.Lmul_odd_done:
mov rsi, rcx
mov rdi, r15
mov rax, r13
imul rax, r13
mov r9, rax
xor r10, r10
.Lcopy_odd:
cmp r10, r9
jae .Lcopy_odd_done
mov rax, qword ptr [rsi + r10*8]
mov qword ptr [rdi + r10*8], rax
inc r10
jmp .Lcopy_odd
.Lcopy_odd_done:
.Lshift:
shr r14, 1
test r14, r14
jz .Lzero
mov rdi, r12
mov rsi, r12
mov rax, r13
imul rax, r13
lea rcx, [r12 + rax*8]
mov r9, 0
.Lmul_sq:
cmp r9, r13
jae .Lmul_sq_done
mov rax, r9
imul rax, r13
lea rdx, [rdi + rax*8]
lea r8, [rcx + rax*8]
mov r10, 0
.Lmul_sq_j:
cmp r10, r13
jae .Lmul_sq_next_i
xor rbx, rbx
lea rbp, [rsi + r10*8]
mov r11, 0
.Lmul_sq_t:

jae .Lmul_sq_t_end

imul rax, qword ptr [rbp]

lea rbp, [rbp + r13*8]

.Lmul_sq_t_end:
mov qword ptr [r8], rbx
inc r10
lea r8, [r8 + 8]
jmp .Lmul_sq_j
.Lmul_sq_next_i:
inc r9
jmp .Lmul_sq
.Lmul_sq_done:
mov rsi, rcx
mov rdi, r12
mov rax, r13
imul rax, r13
mov r9, rax
xor r10, r10
.Lcopy_sq:
cmp r10, r9
jae .Lcopy_sq_done
mov rax, qword ptr [rsi + r10*8]
mov qword ptr [rdi + r10*8], rax
inc r10
jmp .Lcopy_sq
.Lcopy_sq_done:
jmp .Lpow
.Lzero:
mov rax, r13
imul rax, r13
shl rax, 1
mov r9, rax
xor r10, r10
lea rdi, [r12]
.Lzero_loop:
cmp r10, r9
jae .Lzero_done
xor rbx, rbx
mov qword ptr [rdi + r10*8], rbx
inc r10
jmp .Lzero_loop
.Lzero_done:
cld
xor eax, eax
pop rsi
pop r15
pop r14
