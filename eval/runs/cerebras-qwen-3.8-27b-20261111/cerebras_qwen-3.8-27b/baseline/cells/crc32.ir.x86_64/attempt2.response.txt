

export function l $crc32(l ptr, l len) {
@entry
  %all =w sub 0, 1
  %poly =w xor 0xEDB88320, 0
  %crc =w xor %all, 0
  %idx =l add 0, 0
  %addr =l add %ptr, 0
  %lt =w csltl %idx, %len
  jnz %lt, @body
  b @done
@body
  %byte =w loadub %addr
  %addr =l add %addr, 1
  %crc =w xor %crc, %byte
  %cnt =l add 0, 0
  b @inner
@inner
  %lo =w and %crc, 1
  %crc =w shr %crc, 1
  %adj =w mul %lo, %poly
  %crc =w xor %crc, %adj
  %cnt =l add %cnt, 1
  %cont =w csltl %cnt, 8
  jnz %cont, @inner
  b @next
@next
  %idx =l add %idx, 1
  %lt =w csltl %idx, %len
  jnz %lt, @body
  b @done
@done
  %res =w xor %crc, %all
  %ret =l extuw %res
  ret %ret
}
