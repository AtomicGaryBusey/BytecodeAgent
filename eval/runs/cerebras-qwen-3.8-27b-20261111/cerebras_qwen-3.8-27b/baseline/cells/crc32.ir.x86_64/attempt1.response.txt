

export function l $crc32(l p, l n) {
@start
  %mask =w sub 0, 1
  %poly =w xor 0xEDB88320, 0
  %i =l add 0, 0
  %c =w xor %mask, 0
  b @outer
@outer
  %lt =w csltl %i, %n
  jnz %lt, @process
  b @done
@process
  %b =w loadub %p
  %p =l add %p, 1
  %c =w xor %c, %b
  %k =l add 0, 0
  b @inner
@inner
  %shr =w shr %c, 1
  %lo =w and %c, 1
  %adj =w mul %lo, %poly
  %c =w xor %shr, %adj
  %k =l add %k, 1
  %cont =w csltl %k, 8
  jnz %cont, @inner
  b @outer_next
@outer_next
  %i =l add %i, 1
  b @outer
@done
  %r =w xor %c, %mask
  %ret =l extuw %r
  ret %ret
}
