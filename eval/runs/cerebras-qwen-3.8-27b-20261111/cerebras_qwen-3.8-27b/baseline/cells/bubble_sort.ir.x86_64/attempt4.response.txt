

export function l $bubble_sort(l, l)
{
@entry
    movl %a, %0
    movl %n, %1
    csltl %empty, %n, 1
    jnz @done, %empty
    jmp @start
@start
    subl %nm1, %n, 1
    movl %i, 0
    jmp @outer
@outer
    csltl %go, %i, %nm1
    jnz @init, %go
    jmp @done
@init
    movl %j, 0
    movl %pa, %a
    movl %pb, %a
    addl %pb, %pb, 8
    jmp @inner
@inner
    csltl %gin, %j, %nm1
    jnz @cmp, %gin
    addl %i, %i, 1
    jmp @outer
@cmp
    loadl %x, %pa
    loadl %y, %pb
    csltl %sw, %y, %x
    jnz @swap, %sw
    jmp @adv
@swap
    storel %pa, %y
    storel %pb, %x
    jmp @adv
@adv
    addl %pa, %pa, 8
    addl %pb, %pb, 8
    addl %j, %j, 1
    jmp @inner
@done
    movl %z, 0
    retl %z
}
