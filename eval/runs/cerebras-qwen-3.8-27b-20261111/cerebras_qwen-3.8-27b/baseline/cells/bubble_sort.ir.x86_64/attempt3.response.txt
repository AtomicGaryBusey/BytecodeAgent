

export function l $bubble_sort(l, l)
{
@entry
    movl %a, %0
    movl %n, %1
    csltl %lt2, %n, 2
    jnz @done
    movl %i, 0
@outer
    movl %nm1, %n
    subl %nm1, 1
    csltl %out, %i, %nm1
    jz @done
    movl %jmax, %nm1
    subl %jmax, %i
    movl %j, 0
    movl %p, %a
    jmp @inner
@inner
    csltl %in, %j, %jmax
    jz @next
    movl %pnext, %p
    addl %pnext, 8
    loadl %x, %p
    loadl %y, %pnext
    csltl %sw, %y, %x
    jz @cont
    storel %p, %y
    storel %pnext, %x
    jmp @cont
@cont
    addl %p, 8
    addl %j, 1
    jmp @inner
@next
    addl %i, 1
    jmp @outer
@done
    movl %rax, 0
    ret
}
