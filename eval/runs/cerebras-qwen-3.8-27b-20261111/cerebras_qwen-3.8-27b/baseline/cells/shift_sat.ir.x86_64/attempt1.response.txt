

export function l $shift_sat(l x, l n)
@entry:
    %lt = csltl(n, 64)
    jnz %lt, @small
@large:
    %neg = csltl(x, 0)
    jnz %neg, @neg
@nonneg:
    returnl 0
@neg:
    %negone = subl(0, 1)
    returnl %negone
@small:
    %ls = shll(x, n)
    %rs = ashrl(x, n)
    %sum = addl(%ls, %rs)
    returnl %sum
