

dot_i32_simd:
    xor rax, rax
    pxor xmm0, xmm0
    cmp rdx, 4
    jb .Ltail
.Lloop:
    movdqu xmm1, oword ptr [rdi]
    pmovsxdq xmm1, xmm1
    movdqu xmm2, oword ptr [rsi]
    pmovsxdq xmm2, xmm2
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jae .Lloop
.Ltail:
    test rdx, rdx
    jz .Lsum
.Ltail_loop:
    movsxd rcx, dword ptr [rdi]
    movsxd r11, dword ptr [rsi]
    imul rcx, r11
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Ltail_loop
.Lsum:
    pextrq rcx, xmm0, 0
    add rax, rcx
    pextrq rcx, xmm0, 1
    add rax, rcx
    pextrq rcx, xmm0, 2
    add rax, rcx
    pextrq rcx, xmm0, 3
    add rax, rcx
    ret
