

binary_search:
    cmp x1, xzr
    b.le .Lnotfound
    mov x3, xzr
    mov x4, x1
.Lloop:
    cmp x4, x3
    b.ls .Lnotfound
    sub x5, x4, x3
    lsr x5, x5, #1
    add x5, x5, x3
    ldr x6, [x0, x5, lsl #3]
    cmp x6, x2
    b.eq .Lfound
    b.lt .Llow
    mov x4, x5
    b .Lloop
.Llow:
    add x3, x5, #1
    b .Lloop
.Lfound:
    mov x0, x5
    ret
.Lnotfound:
    mov x0, #0
    movn x0, x0
    ret
