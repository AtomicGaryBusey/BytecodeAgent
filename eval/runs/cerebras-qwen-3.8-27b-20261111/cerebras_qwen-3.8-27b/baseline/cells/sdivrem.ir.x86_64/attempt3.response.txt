

export function l $sdivrem(aa l, bb l, rp l) l {
@entry
  movl %a, aa
  movl %b, bb
  movl %p, rp
  movl %zero, 0
  movl %mone, 0xffffffffffffffff
  movl %min, 0x8000000000000000
  ceql %bzero, %b, %zero
  jnz @zero_div, %bzero
  ceql %amin, %a, %min
  jnz @maybe_min, %amin
  jmp @div
@maybe_min
  ceql %bmone, %b, %mone
  jnz @min_neg_one, %bmone
  jmp @div
@zero_div
  storel %a, %p
  retl %zero
@min_neg_one
  storel %zero, %p
  retl %min
@div
  divl %q, %a, %b
  mull %prod, %q, %b
  subl %r, %a, %prod
  storel %r, %p
  retl %q
}
