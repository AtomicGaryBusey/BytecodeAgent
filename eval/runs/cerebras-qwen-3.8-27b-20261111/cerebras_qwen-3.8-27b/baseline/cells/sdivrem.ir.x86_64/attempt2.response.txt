

export function l $sdivrem(l aa, l bb, l rp) l {
@entry
  movl %a, aa
  movl %b, bb
  movl %rem, rp
  movl %zero, 0
  movl %one, 1
  subl %neg1, 0, %one
  movl %max, 0x7fffffffffffffff
  subl %min, 0, %max
  ceql %c0, %b, %zero
  jnz @bzero, %c0
  ceql %cmin, %a, %min
  jnz @maybe_min, %cmin
  jmp @div
@maybe_min
  ceql %cn1, %b, %neg1
  jnz @mincase, %cn1
  jmp @div
@bzero
  storel %a, %rem
  retl %zero
@mincase
  storel %zero, %rem
  retl %min
@div
  divl %q, %a, %b
  reml %r, %a, %b
  storel %r, %rem
  retl %q
}
