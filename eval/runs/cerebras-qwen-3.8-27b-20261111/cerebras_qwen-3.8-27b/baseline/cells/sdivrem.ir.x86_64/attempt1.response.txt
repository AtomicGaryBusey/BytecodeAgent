

export function l $sdivrem(l x, l y, l p) l {
@entry
  movl %a, x
  movl %b, y
  movl %rem, p
  movl %zero, 0
  movl %min, 0x8000000000000000
  movl %neg1, -1
  ceql %c0, %b, 0
  jnz @zero, %c0
@nz
  ceql %cmin, %a, %min
  jz @div, %cmin
@mincheck
  ceql %cneg, %b, %neg1
  jnz @mincase, %cneg
@elasediv
  jmp @div
@zero
  storel %a, %rem
  retl %zero
@mincase
  storel %zero, %rem
  retl %min
@div
  divl %q, %a, %b
  reml %r, %a, %b
  storel %r, %rem
  retl %q
}
