

export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    %one =l const 1
    %eight =l const 8
    %count =w const 0
    %i =l const 0
    jmp @loop
@loop
    %more =w csltl %i, %n
    jnz %more, @body
    jmp @done
@body
    %off =l mul %i, %eight
    %saddr =l add %src, %off
    %sval =l loadl %saddr
    %r =l call %f(l %sval)
    %daddr =l add %dst, %off
    storel %r, %daddr
    %lt =w csltl %r, %sval
    %count =w add %count, %lt
    %i =l add %i, %one
    jmp @loop
@done
    %ret =l zero %count
    returnl %ret
}
