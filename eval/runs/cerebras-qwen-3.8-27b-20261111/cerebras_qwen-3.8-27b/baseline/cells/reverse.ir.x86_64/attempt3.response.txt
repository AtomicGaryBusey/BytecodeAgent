

export function l $reverse(l p0, l n0) {
@start
    %lzero = l 0
    %wzero = ceql n0, %lzero
    jnz %wzero, @done
@setup
    %lone = l 1
    %leight = l 8
    %lnlast = subl n0, %lone
    %loffset = mull %lnlast, %leight
    %lright = addl p0, %loffset
    %lleft = p0
    j @loop
@loop
    %wlt = csltl %lleft, %lright
    jnz %wlt, @swap
@done
    retl %lzero
@swap
    %leight = l 8
    %laval = loadl %lleft
    %lbval = loadl %lright
    storel %lright, %laval
    storel %lleft, %lbval
    %lleft = addl %lleft, %leight
    %lright = subl %lright, %leight
    j @loop
}
