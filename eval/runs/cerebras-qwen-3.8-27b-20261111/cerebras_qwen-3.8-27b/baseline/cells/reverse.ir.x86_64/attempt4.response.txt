

export function l $reverse(l, l) {
@start
    %zero = l 0
    %iseq = ceql %1, %zero
    jnz %iseq, @done
@setup
    %one = l 1
    %eight = l 8
    %last = subl %1, %one
    %offset = mull %last, %eight
    %right = addl %0, %offset
    %left = addl %0, %zero
    j @loop
@loop
    %less = csltl %left, %right
    jnz %less, @swap
@done
    retl %zero
@swap
    %lv = loadl %left
    %rv = loadl %right
    storel %right, %lv
    storel %left, %rv
    %left = addl %left, %eight
    %right = subl %right, %eight
    j @loop
}
