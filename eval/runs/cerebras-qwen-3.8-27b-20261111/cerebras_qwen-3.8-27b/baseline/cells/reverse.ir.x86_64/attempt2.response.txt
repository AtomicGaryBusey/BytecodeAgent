

export function l $reverse(l a, l n) {
	@start
	%lzero = 0
	%lone = 1
	%leight = 8
	%wzero = ceql n, %lzero
	jnz %wzero, @done
	%lnm = subl n, %lone
	%loff = mull %lnm, %leight
	%lright = addl a, %loff
	%lleft = a
	@loop
	%wless = csltl %lleft, %lright
	jnz %wless, @done
	%lva = loadl %lleft
	%lvb = loadl %lright
	storel %lright, %lva
	storel %lleft, %lvb
	%lleft = addl %lleft, %leight
	%lright = subl %lright, %leight
	j @loop
	@done
	retl %lzero
}
