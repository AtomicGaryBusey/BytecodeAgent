

export function l $collatz_max(l) {
@entry
    movl %zero, 0
    movl %one, 1
    movl %two, 2
    movl %three, 3
    movl %max, 0
    movl %i, 1
    movl %n, %0
    addl %n1, %n, %one
    jump @outer
@outer
    csltl %go, %i, %n1
    jnz @body, %go
    jump @end
@body
    movl %x, %i
    movl %steps, 0
    jump @inner
@inner
    ceql %done, %x, %one
    jnz @innerend, %done
    andl %parity, %x, %one
    ceql %even, %parity, %zero
    jnz @halve, %even
    mull %t, %x, %three
    addl %x, %t, %one
    addl %steps, %steps, %one
    jump @inner
@halve
    divul %t, %x, %two
    movl %x, %t
    addl %steps, %steps, %one
    jump @inner
@innerend
    csltl %inc, %max, %steps
    jnz @updatemax, %inc
    jump @updatei
@updatemax
    movl %max, %steps
    jump @updatei
@updatei
    addl %i, %i, %one
    jump @outer
@end
    retl %max
}
