

export function l $collatz_max(l) {
@entry
    movl %c0, 0
    movl %c1, 1
    movl %c2, 2
    movl %c3, 3
    movl %n, %0
    addl %nplus, %n, %c1
    movl %max, %c0
    movl %i, %c1
    jump @outer
@outer
    csltl %go, %i, %nplus
    jnz @body, %go
    retl %max
@body
    movl %x, %i
    movl %steps, %c0
    jump @inner
@inner
    ceql %done, %x, %c1
    jnz @innerend, %done
    andl %parity, %x, %c1
    ceql %even, %parity, %c0
    jnz @halve, %even
    mull %t, %x, %c3
    addl %x, %t, %c1
    addl %steps, %steps, %c1
    jump @inner
@halve
    divul %t, %x, %c2
    movl %x, %t
    addl %steps, %steps, %c1
    jump @inner
@innerend
    csltl %less, %max, %steps
    jnz @update, %less
    jump @next
@update
    movl %max, %steps
    jump @next
@next
    addl %i, %i, %c1
    jump @outer
}
