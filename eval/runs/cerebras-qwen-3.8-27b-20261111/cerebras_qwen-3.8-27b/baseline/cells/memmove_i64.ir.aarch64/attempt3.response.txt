

export function l $memmove_i64(d0 l, s0 l, n0 l) {
@entry:
    %zero = l 0
    %one = l 1
    %eight = l 8
    %less = csltl d0, s0
    jnz %less, @fwd
    j @bwd

@fwd:
    %fs = addl s0, %zero
    %fd = addl d0, %zero
    %fi = addl %zero, %zero
    j @fwdloop

@fwdloop:
    %flt = csltl %fi, n0
    jnz %flt, @fwdbody
    j @exit

@fwdbody:
    %fv = loadl [%fs]
    storel [%fd], %fv
    %fi = addl %fi, %one
    %fs = addl %fs, %eight
    %fd = addl %fd, %eight
    j @fwdloop

@bwd:
    %bwd_nz = csltl %zero, n0
    jnz %bwd_nz, @bwdinit
    j @exit

@bwdinit:
    %off = mull n0, %eight
    %last = subl %off, %eight
    %bs = addl s0, %last
    %bd = addl d0, %last
    %bc = addl n0, %zero
    j @bwdloop

@bwdloop:
    %bv = loadl [%bs]
    storel [%bd], %bv
    %bc = subl %bc, %one
    %bs = subl %bs, %eight
    %bd = subl %bd, %eight
    %bcnz = csltl %zero, %bc
    jnz %bcnz, @bwdloop
    j @exit

@exit:
    ret %zero
}
