

export function l $memmove_i64(l dst, l src, l n) {
@entry:
    %zero = l 0
    %one = l 1
    %eight = l 8
    %less = csltl dst, src
    jnz %less, @fwd
    j @bwd

@fwd:
    %srcp = addl src, %zero
    %dstp = addl dst, %zero
    %i = addl %zero, %zero

@fwdloop:
    %fwd_lt = csltl %i, n
    jnz %fwd_lt, @fwdbody
    j @exit

@fwdbody:
    %val = loadl [%srcp]
    storel [%dstp], %val
    %i = addl %i, %one
    %srcp = addl %srcp, %eight
    %dstp = addl %dstp, %eight
    j @fwdloop

@bwd:
    %bwd_nz = csltl %zero, n
    jnz %bwd_nz, @bwdinit
    j @exit

@bwdinit:
    %off = mull n, %eight
    %offm = subl %off, %eight
    %srcp = addl src, %offm
    %dstp = addl dst, %offm
    %count = addl n, %zero

@bwdloop:
    %val = loadl [%srcp]
    storel [%dstp], %val
    %count = subl %count, %one
    %srcp = subl %srcp, %eight
    %dstp = subl %dstp, %eight
    %bwd_count_nz = csltl %zero, %count
    jnz %bwd_count_nz, @bwdloop
    j @exit

@exit:
    ret %zero
}
