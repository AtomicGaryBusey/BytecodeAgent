

export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%zero =l subl %dst, %dst
	%one =l addl %zero, 1
	%eight =l addl %zero, 8
	%count =l addl %zero, %zero
	%dp =l addl %dst, %zero
	%sr =l addl %src, %zero
	%rem =l addl %n, %zero
	j @test
@test
	%nz =w csltl %zero, %rem
	jnz %nz @body
@done
	ret %count
@body
	%val =l loadl %sr
	%res =l call %f(l %val)
	storel %dp, %res
	%less =w csltl %res, %val
	jnz %less @incr
@next
	%dp =l addl %dp, %eight
	%sr =l addl %sr, %eight
	%rem =l subl %rem, %one
	j @test
@incr
	%count =l addl %count, %one
	j @next
}
