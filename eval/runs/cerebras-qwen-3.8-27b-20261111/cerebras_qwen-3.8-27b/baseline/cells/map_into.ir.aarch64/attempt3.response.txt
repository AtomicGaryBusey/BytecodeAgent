

export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%zero =l 0
	%one =l 1
	%eight =l 8
	%dp =l add %dst, %zero
	%srcp =l add %src, %zero
	%rem =l add %n, %zero
	%count =l 0
	j @test
@test
	%nz =w csltl %zero, %rem
	jnz %nz @body
@done
	ret %count
@body
	%val =l loadl %srcp
	%res =l call %f(l %val)
	storel %res, %dp
	%less =w csltl %res, %val
	jnz %less @incr
@next
	%dp =l add %dp, %eight
	%srcp =l add %srcp, %eight
	%rem =l sub %rem, %one
	j @test
@incr
	%count =l add %count, %one
	j @next
}
