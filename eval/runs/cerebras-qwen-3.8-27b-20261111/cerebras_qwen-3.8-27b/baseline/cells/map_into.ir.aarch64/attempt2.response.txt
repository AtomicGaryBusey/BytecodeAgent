

export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%dp =l addl %dst, 0
	%sdp =l addl %src, 0
	%rem =l addl %n, 0
	%count =l 0
	%nzero =w csltl 0, %rem
	jnz %nzero @body
	ret %count
@body
	%val =l loadl %sdp
	%res =l call %f(l %val)
	storel %res, %dp
	%less =w csltl %res, %val
	jnz %less @incr
	j @next
@incr
	%count =l addl %count, 1
	j @next
@next
	%dp =l addl %dp, 8
	%sdp =l addl %sdp, 8
	%rem =l subl %rem, 1
	%nzero2 =w csltl 0, %rem
	jnz %nzero2 @body
	ret %count
}
