

export function l $count_desc(l a, l n) as
@entry
movl %p, a
movl %cnt, 0
movl %len, n
@loop
csltl %exit, %len, 2
jnz %exit, @done
loadl %x, %p
addl %q, %p, 8
loadl %y, %q
csltl %desc, %y, %x
jnz %desc, @inc
@next
addl %p, %p, 8
subl %len, %len, 1
ceql %t, 0, 0
jnz %t, @loop
@inc
addl %cnt, %cnt, 1
addl %p, %p, 8
subl %len, %len, 1
ceql %t2, 0, 0
jnz %t2, @loop
@done
retl %cnt
