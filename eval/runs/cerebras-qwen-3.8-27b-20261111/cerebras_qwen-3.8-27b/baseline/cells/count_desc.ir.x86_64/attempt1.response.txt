

export function l $count_desc(l a, l n) {
@entry
    %end = subl %n, #1
    %lt2 = csltl %n, #2
    jnz %lt2 @exit
@init
    %i = #0
    %count = #0
@loop
    %off = mull %i, #8
    %p = addl %a, %off
    %x = loadl (%p)
    %p2 = addl %p, #8
    %y = loadl (%p2)
    %gt = csltl %y, %x
    jnz %gt @inc
@noinc
    %i = addl %i, #1
    %cont = csltl %i, %end
    jnz %cont @loop
@exit
    retl %count
@inc
    %count = addl %count, #1
    %i = addl %i, #1
    %cont = csltl %i, %end
    jnz %cont @loop
@exit2
    retl %count
}
