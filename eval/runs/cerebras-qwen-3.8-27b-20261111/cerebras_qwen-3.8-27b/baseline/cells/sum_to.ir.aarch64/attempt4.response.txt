

package aarch64
export function l $sum_to(l)
    s0 l
@entry:
    movl %n, s0
    csltl %t, %n, l:0
    jnz %t, @zero
@not_negative:
    ceql %t, %n, l:0
    jnz %t, @zero
@init:
    movl %sum, l:0
    jmp @loop
@loop:
    addl %sum, %sum, %n
    subl %n, %n, l:1
    csltl %t, %n, l:0
    jnz %t, @loop
@done:
    movl r, %sum
    retl
@zero:
    movl r, l:0
    retl
