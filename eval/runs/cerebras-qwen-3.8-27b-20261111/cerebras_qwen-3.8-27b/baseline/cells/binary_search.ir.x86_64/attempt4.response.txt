

export function l $binary_search(l %a, l %n, l %key) {
@entry
        l %lo = addl 0, 0
        l %hi = addl %n, 0
@loop
        w %have = csltl %lo, %hi
        jnz %have, @body
        j @done
@body
        l %diff = subl %hi, %lo
        l %half = shrl %diff, 1
        l %mid = addl %lo, %half
        l %off0 = addl %mid, %mid
        l %off1 = addl %off0, %off0
        l %off2 = addl %off1, %off1
        l %p = addl %a, %off2
        l %v = loadl %p
        w %less = csltl %v, %key
        jnz %less, @up
        l %hi = addl %mid, 0
        j @loop
@up
        l %lo = addl %mid, 1
        j @loop
@done
        w %out = ceql %lo, %n
        jnz %out, @notfound
        l %findoff0 = addl %lo, %lo
        l %findoff1 = addl %findoff0, %findoff0
        l %findoff2 = addl %findoff1, %findoff1
        l %fp = addl %a, %findoff2
        l %fv = loadl %fp
        w %found = ceql %fv, %key
        jnz %found, @retfound
        j @notfound
@notfound
        l %minus = subl 0, 1
        retl %minus
@retfound
        retl %lo
}
