

export function l $binary_search(l %a, l %n, l %key)
@entry
    l %lo = addl 0, 0
    l %hi = addl %n, 0
@loop
    w %have = csltl %lo, %hi
    jnz %have, @body
    j @done
@body
    l %diff = subl %hi, %lo
    l %half = shrl %diff, 1
    l %mid = addl %lo, %half
    l %o1 = addl %mid, %mid
    l %o2 = addl %o1, %o1
    l %o3 = addl %o2, %o2
    l %p = addl %a, %o3
    l %v = loadl %p
    w %less = csltl %v, %key
    jnz %less, @up
    l %hi = addl %mid, 0
    j @loop
@up
    l %lo = addl %mid, 1
    j @loop
@done
    w %out = ceql %lo, %n
    jnz %out, @notfound
    l %fo1 = addl %lo, %lo
    l %fo2 = addl %fo1, %fo1
    l %fo3 = addl %fo2, %fo2
    l %fp = addl %a, %fo3
    l %fv = loadl %fp
    w %found = ceql %fv, %key
    jnz %found, @retfound
    j @notfound
@notfound
    l %minus = subl 0, 1
    retl %minus
@retfound
    retl %lo
