

reverse:
    cmp rsi, 1
    jle .Ldone
    mov rax, rdi
    mov rdx, rsi
    dec rdx
    lea rcx, [rdi + rdx*8]
.Lloop:
    cmp rax, rcx
    jae .Ldone
    mov rdx, qword ptr [rax]
    mov r10, qword ptr [rcx]
    mov qword ptr [rax], r10
    mov qword ptr [rcx], rdx
    add rax, 8
    sub rcx, 8
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
