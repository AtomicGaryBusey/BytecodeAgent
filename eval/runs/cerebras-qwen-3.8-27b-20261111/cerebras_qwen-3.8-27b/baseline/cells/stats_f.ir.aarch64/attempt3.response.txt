

export function l $stats_f (l, l, l, l)
@start
  %a =l arg0
  %n =l arg1
  %f =l arg2
  %out =l arg3
  %p =l addl %a, 8
  %x =l loadl %a
  %y =l call %f (l %x)
  %min =l addl %y, 0
  %max =l addl %y, 0
  %sum =l addl %y, 0
  %cnt =l andl %y, 1
  %i =l addl 1, 0
  jmp @loop
@loop
  %cmp =w csltl %i, %n
  jnz %cmp -> @body
  jmp @exit
@body
  %x =l loadl %p
  %y =l call %f (l %x)
  %lt =w csltl %y, %min
  jnz %lt -> @updmin
  jmp @aftermin
@updmin
  %min =l addl %y, 0
  jmp @aftermin
@aftermin
  %gt =w csltl %max, %y
  jnz %gt -> @updmax
  jmp @aftermax
@updmax
  %max =l addl %y, 0
  jmp @aftermax
@aftermax
  %sum =l addl %sum, %y
  %odd =l andl %y, 1
  %cnt =l addl %cnt, %odd
  %i =l addl %i, 1
  %p =l addl %p, 8
  jmp @loop
@exit
  storel %min, %out
  %o1 =l addl %out, 8
  storel %max, %o1
  %o2 =l addl %o1, 8
  storel %sum, %o2
  lret (l %cnt)
