

export function l $stats_f (l ptr, l nn, l fn, l dst)
@start
  %p =l addl %ptr, 8
  %x =l loadl %ptr
  %y =l call %fn (l %x)
  %min =l addl %y, 0
  %max =l addl %y, 0
  %sum =l addl %y, 0
  %cnt =l andl %y, 1
  %i =l addl 1, 0
  jmp @loop
@loop
  %m =w csltl %i, %nn
  jnz %m -> @body
  jmp @exit
@body
  %x =l loadl %p
  %y =l call %fn (l %x)
  %lt =w csltl %y, %min
  jnz %lt -> @updmin
  jmp @aftermin
@updmin
  %min =l addl %y, 0
  jmp @aftermin
@aftermin
  %gt =w csltl %max, %y
  jnz %gt -> @updmax
  jmp @aftermax
@updmax
  %max =l addl %y, 0
  jmp @aftermax
@aftermax
  %sum =l addl %sum, %y
  %odd =l andl %y, 1
  %cnt =l addl %cnt, %odd
  %i =l addl %i, 1
  %p =l addl %p, 8
  jmp @loop
@exit
  storel %min, %dst
  %d1 =l addl %dst, 8
  storel %max, %d1
  %d2 =l addl %d1, 8
  storel %sum, %d2
  lret (l %cnt)
