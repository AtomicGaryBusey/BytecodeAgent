

export function l $stats_f(l a, l n, l f, l out)
@start
  %p =l movl %a
  %x =l loadl %p
  %y =l call %f (l %x)
  %min =l movl %y
  %max =l movl %y
  %sum =l movl %y
  %count =l andl %y, 1
  %i =l movl 1
  %p =l addl %p, 8
  jmp @loop
@loop
  %cont =w csltl %i, %n
  jnz %cont -> @body
  jmp @exit
@body
  %x =l loadl %p
  %y =l call %f (l %x)
  %lt =w csltl %y, %min
  jnz %lt -> @updmin
  jmp @aftermin
@updmin
  %min =l movl %y
  jmp @aftermin
@aftermin
  %gt =w csltl %max, %y
  jnz %gt -> @updmax
  jmp @aftermax
@updmax
  %max =l movl %y
  jmp @aftermax
@aftermax
  %sum =l addl %sum, %y
  %odd =l andl %y, 1
  %oddw =w csltl 0, %odd
  jnz %oddw -> @incr
  jmp @afterincr
@incr
  %count =l addl %count, 1
  jmp @afterincr
@afterincr
  %i =l addl %i, 1
  %p =l addl %p, 8
  jmp @loop
@exit
  storel %min, %out
  %out1 =l addl %out, 8
  storel %max, %out1
  %out2 =l addl %out1, 8
  storel %sum, %out2
  lret (l %count)
