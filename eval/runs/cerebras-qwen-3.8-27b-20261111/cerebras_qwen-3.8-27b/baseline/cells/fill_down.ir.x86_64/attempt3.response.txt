

export function l $fill_down(l %a, l %n, l %start)
{
	local l %i
	local l %p
	local l %v
	local w %c
	@entry:
	%i = 0
	%p = %a
	jmp @loop
	@loop:
	@c = csltl %i, %n
	jnz %c @body
	jmp @done
	@body:
	%v = subl %start, %i
	storel %v, %p
	%i = addl %i, 1
	%p = addl %p, 8
	jmp @loop
	@done:
	return %n
}
