

export function l $fill_down(l, l, l)
    @entry
    %n = .arg1
    %i = 0
    %addr = .arg0
    %val = .arg2
    @loop
    %c = ceql %i, %n
    jnz %c, @exit
    @body
    storel %val, %addr
    %val = subl %val, 1
    %addr = addl %addr, 8
    %i = addl %i, 1
    j @loop
    @exit
    ret %n
end
