

export function l $packbits_encode(l src, l n, l out, l cap) {
@entry
  %zero =l $0
  %one =l $1
  %two =l $2
  %three =l $3
  %maxrun =l $128
  %repeat_const =l $257
  %negone =l subrl %zero, %one
  %i =l addl %zero, %zero
  %o =l addl %zero, %zero
  %zn =w ceql n, %zero
  jnz %zn, @done
@main
  %imn =w csltl %i, n
  jnz %imn, @body
  jmp @done
@body
  %p_i =l addl src, %i
  %b_i =l loadub %p_i
  %run =l addl %one, %zero
  %k =l addl %i, %one
@run_loop
  %klt =w csltl %k, n
  jnz %klt, @run_len
  jmp @run_done
@run_len
  %rlt =w csltl %run, %maxrun
  jnz %rlt, @run_load
  jmp @run_done
@run_load
  %p_k =l addl src, %k
  %b_k =l loadub %p_k
  %eq =w ceql %b_k, %b_i
  jnz %eq, @run_inc
  jmp @run_done
@run_inc
  %run =l addl %run, %one
  %k =l addl %k, %one
  jmp @run_loop
@run_done
  %lt3 =w csltl %run, %three
  jnz %lt3, @literal_start
  %need =l addl %run, %one
  %newo =l addl %o, %need
  %over =w csltl cap, %newo
  jnz %over, @fail
  %hdr =l subrl %repeat_const, %run
  %dst0 =l addl out, %o
  %dst1 =l addl %dst0, %one
  storeb %hdr, %dst0
  storeb %b_i, %dst1
  %o =l addl %newo, %zero
  %i =l addl %i, %run
  jmp @main
@literal_start
  %j =l addl %i, %zero
@lit_loop
  %jlt =w csltl %j, n
  jnz %jlt, @lit_len
  jmp @literal_end
@lit_len
  %len =l subrl %j, %i
  %lenlt =w csltl %len, %maxrun
  jnz %lenlt, @lit_run3
  jmp @literal_end
@lit_run3
  %pj =l addl src, %j
  %bj =l loadub %pj
  %pj1 =l addl %j, %one
  %has1 =w csltl %pj1, n
  jnz %has1, @lit_c1
  jmp @lit_inc
@lit_c1
  %p1 =l addl src, %pj1
  %b1 =l loadub %p1
  %eq1 =w ceql %b1, %bj
  jnz %eq1, @lit_c2
  jmp @lit_inc
@lit_c2
  %pj2 =l addl %j, %two
  %has2 =w csltl %pj2, n
  jnz %has2, @lit_c2b
  jmp @lit_inc
@lit_c2b
  %p2 =l addl src, %pj2
  %b2 =l loadub %p2
  %eq2 =w ceql %b2, %bj
  jnz %eq2, @literal_end
  jmp @lit_inc
@lit_inc
  %j =l addl %j, %one
  jmp @lit_loop
@literal_end
  %llen =l subrl %j, %i
  %needl =l addl %llen, %one
  %newol =l addl %o, %needl
  %overl =w csltl cap, %newol
  jnz %overl, @fail
  %hdrl =l subrl %llen, %one
  %dstl =l addl out, %o
  %dst =l addl %dstl, %one
  storeb %hdrl, %dstl
  %srcp =l addl src, %i
  %left =l addl %llen, %zero
@copy_loop
  %ldone =w ceql %left, %zero
  jnz %ldone, @copy_done
  %cb =l loadub %srcp
  storeb %cb, %dst
  %srcp =l addl %srcp, %one
  %dst =l addl %dst, %one
  %left =l subrl %left, %one
  jmp @copy_loop
@copy_done
  %i =l addl %j, %zero
  %o =l addl %newol, %zero
  jmp @main
@fail
  retl %negone
@done
  retl %o
}
