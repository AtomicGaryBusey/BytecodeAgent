

export function l $gcd(l a, l b) {
@entry:
        %a = addl a, 0
        %b = addl b, 0
        jmp @loop
@loop:
        %t = ceql %b, 0
        jnz %t, @done
        %q = divl %a, %b
        %r = mull %q, %b
        %na = addl %b, 0
        %nb = subl %a, %r
        %a = addl %na, 0
        %b = addl %nb, 0
        jmp @loop
@done:
        ret %a
}
