

memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.cs .Lbwd
.Lfwd:
    ldr x5, [x1]
    str x5, [x0]
    add x0, x0, #8
    add x1, x1, #8
    subs x2, x2, #1
    b.ne .Lfwd
    b .Ldone
.Lbwd:
    add x3, x1, x2, lsl #3
    add x4, x0, x2, lsl #3
.Lbwdloop:
    sub x3, x3, #8
    sub x4, x4, #8
    ldr x5, [x3]
    str x5, [x4]
    subs x2, x2, #1
    b.ne .Lbwdloop
    b .Ldone
.Ldone:
    mov x0, #0
    ret
