

export function l $count_desc(l a, l n) =
	@entry
	%c = csltl n, #2
	jnz %c, @ret0
	@init
	%p = a
	%q = addl a, #8
	%count = #0
	%rem = subl n, #1
	j @loop
	@loop
	%cend = ceql %rem, #0
	jnz %cend, @retcount
	@body
	%x = loadl [%p]
	%y = loadl [%q]
	%gt = csltl %y, %x
	jnz %gt, @incr
	@step
	%p = %q
	%q = addl %q, #8
	%rem = subl %rem, #1
	j @loop
	@incr
	%count = addl %count, #1
	j @step
	@retcount
	ret %count
	@ret0
	ret #0
