

export function l $count_desc (l a, l n)
@start
	l
	%small = csltl n, 2l
	jnz %small, @zero
	%ptr = add a, 0l
	%ptr_next = add a, 8l
	%pairs = sub n, 1l
	%count = add 0l, 0l
	j @loop
@loop
	l
	%done = ceql %pairs, 0l
	jnz %done, @end
	%cur = loadl [%ptr]
	%next = loadl [%ptr_next]
	%gt = csltl %next, %cur
	jnz %gt, @inc
	j @next
@inc
	l
	%count = add %count, 1l
	j @next
@next
	l
	%ptr = add %ptr_next, 0l
	%ptr_next = add %ptr_next, 8l
	%pairs = sub %pairs, 1l
	j @loop
@end
	l
	ret %count
@zero
	l
	%zero = add 0l, 0l
	ret %zero
end
