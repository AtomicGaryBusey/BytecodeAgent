@entry
  %all =w const 0xffffffff
  %crc =w const 0xffffffff
  %poly =w const 0xEDB88320
  %ptr =l add %p, 0
  %end =l add %ptr, %n
  goto @outer
@outer
  %eq =w ceql %ptr, %end
  jnz %eq, @done
@body

  %crc =w xor %crc, %adj
  %bit =w and %crc, 1
