

export function l $crc32(l, l)
@entry:
%p =l add %r0, 0
%n =l add %r1, 0
%c =w xor 0xffffffff, 0
%i =l add 0, 0
jmp @loop
@loop:
%done =w ceql %i, %n
jnz %done, @exit
@byte:
%b =w loadub %p
%c =w xor %c, %b
%bit =l add 7, 0
jmp @bitloop
@bitloop:
%lsb =w and %c, 1
jnz %lsb, @bitone
@bitzero:
%c =w shr %c, 1
jmp @bitnext
@bitone:
%c =w shr %c, 1
%c =w xor %c, 0xedb88320
jmp @bitnext
@bitnext:
%donebit =w ceql %bit, 0
jnz %donebit, @nextbyte
@bitdec:
%bit =l sub %bit, 1
jmp @bitloop
@nextbyte:
%p =l add %p, 1
%i =l add %i, 1
jmp @loop
@exit:
%c =w xor %c, 0xffffffff
%r0 =l extuw %c
ret
end
