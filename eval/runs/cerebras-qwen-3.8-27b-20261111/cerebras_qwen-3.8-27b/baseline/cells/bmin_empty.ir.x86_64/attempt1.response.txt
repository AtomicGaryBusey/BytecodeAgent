

export function $bmin_empty(l a, l n) l {
^entry:
    %nonempty = csltl %n, 1
    jnz %nonempty, ^empty
    j ^init
^init:
    %m = loadl 0(l, %a)
    %p = addl %a, 8
    %rem = subl %n, 1
    j ^loop
^loop:
    %finished = ceql %rem, 0
    jnz %finished, ^done
    %v = loadl 0(l, %p)
    %p = addl %p, 8
    %rem = subl %rem, 1
    %less = csltl %v, %m
    jnz %less, ^upd
    j ^loop
^upd:
    %m = movl %v
    j ^loop
^done:
    sret %m
^empty:
    %max = movl 9223372036854775807
    sret %max
}
