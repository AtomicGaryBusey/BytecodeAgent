.section .text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #65536
    b.lo read_loop

read_done:
    mov x21, #0
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne error
    b success

skip_ws:
1:
    cmp x21, x20
    b.hs 2f
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq 3f
    cmp w0, #0x09
    b.eq 3f
    cmp w0, #0x0a
    b.eq 3f
    cmp w0, #0x0d
    b.ne 2f
3:
    add x21, x21, #1
    b 1b
2:
    ret

parse_value:
    str x30, [sp, #-16]!
    bl skip_ws
    cmp x21, x20
    b.hs parse_value_error

    ldrb w0, [x19, x21]
    cmp w0, #0x7b
    b.eq parse_value_object
    cmp w0, #0x5b
    b.eq parse_value_array
    cmp w0, #0x22
    b.eq parse_value_string
    cmp w0, #0x74
    b.eq parse_value_true
    cmp w0, #0x66
    b.eq parse_value_false
    cmp w0, #0x6e
    b.eq parse_value_null
    cmp w0, #0x2d
    b.eq parse_value_number
    cmp w0, #0x30
    b.lo parse_value_error
    cmp w0, #0x39
    b.hi parse_value_error

parse_value_number:
    ldr x30, [sp], #16
    b parse_number

parse_value_object:
    ldr x30, [sp], #16
    b parse_object

parse_value_array:
    ldr x30, [sp], #16
    b parse_array

parse_value_string:
    ldr x30, [sp], #16
    b parse_string

parse_value_true:
    ldr x30, [sp], #16
    b parse_true

parse_value_false:
    ldr x30, [sp], #16
    b parse_false

parse_value_null:
    ldr x30, [sp], #16
    b parse_null

parse_value_error:
    add sp, sp, #16
    b error

parse_string:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne error
    add x21, x21, #1

parse_string_loop:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq parse_string_done
    cmp w0, #0x20
    b.lo error
    cmp w0, #0x5c
    b.eq parse_string_escape
    add x21, x21, #1
    b parse_string_loop

parse_string_done:
    add x21, x21, #1
    ret

parse_string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq parse_string_escape_one
    cmp w0, #0x5c
    b.eq parse_string_escape_one
    cmp w0, #0x2f
    b.eq parse_string_escape_one
    cmp w0, #0x62
    b.eq parse_string_escape_one
    cmp w0, #0x66
    b.eq parse_string_escape_one
    cmp w0, #0x6e
    b.eq parse_string_escape_one
    cmp w0, #0x72
    b.eq parse_string_escape_one
    cmp w0, #0x74
    b.eq parse_string_escape_one
    cmp w0, #0x75
    b.eq parse_string_unicode
    b error

parse_string_escape_one:
    add x21, x21, #1
    b parse_string_loop

parse_string_unicode:
    add x21, x21, #1
    mov w1, #4

parse_string_unicode_loop:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo parse_string_unicode_upper
    cmp w0, #0x39
    b.ls parse_string_unicode_digit

parse_string_unicode_upper:
    cmp w0, #0x41
    b.lo parse_string_unicode_lower
    cmp w0, #0x46
    b.ls parse_string_unicode_digit

parse_string_unicode_lower:
    cmp w0, #0x61
    b.lo error
    cmp w0, #0x66
    b.hi error

parse_string_unicode_digit:
    add x21, x21, #1
    subs w1, w1, #1
    b.ne parse_string_unicode_loop
    b parse_string_loop

parse_true:
    add x21, x21, #1
    mov w1, #0x72
    bl match_byte
    mov w1, #0x75
    bl match_byte
    mov w1, #0x65
    b match_byte

parse_false:
    add x21, x21, #1
    mov w1, #0x61
    bl match_byte
    mov w1, #0x6c
    bl match_byte
    mov w1, #0x73
    bl match_byte
    mov w1, #0x65
    b match_byte

parse_null:
    add x21, x21, #1
    mov w1, #0x75
    bl match_byte
    mov w1, #0x6c
    bl match_byte
    mov w1, #0x6c
    b match_byte

match_byte:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, w1
    b.ne error
    add x21, x21, #1
    ret

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #0x2d
    b.ne parse_number_integer
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]

parse_number_integer:
    cmp w0, #0x30
    b.eq parse_number_zero
    cmp w0, #0x31
    b.lo error
    cmp w0, #0x39
    b.hi error

parse_number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo parse_number_after_integer
    cmp w0, #0x39
    b.ls parse_number_digits
    b parse_number_after_integer

parse_number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_done
    ldrb w0, [x19, x21]

parse_number_after_integer:
    cmp w0, #0x2e
    b.eq parse_number_fraction
    cmp w0, #0x65
    b.eq parse_number_exponent
    cmp w0, #0x45
    b.eq parse_number_exponent
    ret

parse_number_fraction:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo error
    cmp w0, #0x39
    b.hi error

parse_number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo parse_number_after_fraction
    cmp w0, #0x39
    b.ls parse_number_fraction_digits

parse_number_after_fraction:
    cmp w0, #0x65
    b.eq parse_number_exponent
    cmp w0, #0x45
    b.eq parse_number_exponent
    ret

parse_number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x2b
    b.eq parse_number_exponent_sign
    cmp w0, #0x2d
    b.ne parse_number_exponent_digit

parse_number_exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]

parse_number_exponent_digit:
    cmp w0, #0x30
    b.lo error
    cmp w0, #0x39
    b.hi error

parse_number_exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo parse_number_done
    cmp w0, #0x39
    b.ls parse_number_exponent_digits

parse_number_done:
    ret

parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x5d
    b.eq parse_array_close

parse_array_value:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x5d
    b.eq parse_array_close
    cmp w0, #0x2c
    b.ne parse_container_error
    add x21, x21, #1
    bl skip_ws
    b parse_array_value

parse_array_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x7d
    b.eq parse_object_close

parse_object_member:
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne parse_container_error
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x3a
    b.ne parse_container_error
    add x21, x21, #1
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs parse_container_error
    ldrb w0, [x19, x21]
    cmp w0, #0x7d
    b.eq parse_object_close
    cmp w0, #0x2c
    b.ne parse_container_error
    add x21, x21, #1
    bl skip_ws
    b parse_object_member

parse_object_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

parse_container_error:
    add sp, sp, #16
    b error

success:
    mov x0, #1
    adrp x1, ok_message
    add x1, x1, :lo12:ok_message
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x1, number_buffer
    add x1, x1, :lo12:number_buffer
    add x1, x1, #31
    mov w2, #0x0a
    strb w2, [x1]
    add x3, x1, #1
    mov x4, x21
    mov x5, #10

decimal_loop:
    udiv x6, x4, x5
    msub x7, x6, x5, x4
    add w7, w7, #0x30
    sub x1, x1, #1
    strb w7, [x1]
    mov x4, x6
    cbnz x4, decimal_loop

    sub x2, x3, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "

.bss
.balign 16
input_buffer:
    .zero 65536
number_buffer:
    .zero 32
