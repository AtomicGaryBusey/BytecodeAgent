.section .text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    cmp x20, #65536
    b.lo read_loop

read_done:
    mov x21, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne error
    b success

skip_ws:
1:
    cmp x21, x20
    b.hs 2f
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq 3f
    cmp w0, #0x09
    b.eq 3f
    cmp w0, #0x0a
    b.eq 3f
    cmp w0, #0x0d
    b.ne 2f
3:
    add x21, x21, #1
    b 1b
2:
    ret

parse_value:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x7b
    b.eq parse_object
    cmp w0, #0x5b
    b.eq parse_array
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #0x74
    b.eq parse_true
    cmp w0, #0x66
    b.eq parse_false
    cmp w0, #0x6e
    b.eq parse_null
    cmp w0, #0x2d
    b.eq parse_number
    cmp w0, #0x30
    b.lo error
    cmp w0, #0x39
    b.hi error
    b parse_number

parse_string:
    add x21, x21, #1

string_loop:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq string_done
    cmp w0, #0x20
    b.lo error
    cmp w0, #0x5c
    b.eq string_escape
    add x21, x21, #1
    b string_loop

string_done:
    add x21, x21, #1
    ret

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq string_escape_done
    cmp w0, #0x5c
    b.eq string_escape_done
    cmp w0, #0x2f
    b.eq string_escape_done
    cmp w0, #0x62
    b.eq string_escape_done
    cmp w0, #0x66
    b.eq string_escape_done
    cmp w0, #0x6e
    b.eq string_escape_done
    cmp w0, #0x72
    b.eq string_escape_done
    cmp w0, #0x74
    b.eq string_escape_done
    cmp w0, #0x75
    b.eq string_unicode
    b error

string_escape_done:
    add x21, x21, #1
    b string_loop

string_unicode:
    add x21, x21, #1
    mov w1, #4

unicode_loop:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo unicode_upper
    cmp w0, #0x39
    b.ls unicode_valid

unicode_upper:
    cmp w0, #0x41
    b.lo unicode_lower
    cmp w0, #0x46
    b.ls unicode_valid

unicode_lower:
    cmp w0, #0x61
    b.lo error
    cmp w0, #0x66
    b.hi error

unicode_valid:
    add x21, x21, #1
    subs w1, w1, #1
    b.ne unicode_loop
    b string_loop

parse_true:
    adrp x1, true_tail
    add x1, x1, :lo12:true_tail
    mov x2, #4
    b parse_literal

parse_false:
    adrp x1, false_tail
    add x1, x1, :lo12:false_tail
    mov x2, #5
    b parse_literal

parse_null:
    adrp x1, null_tail
    add x1, x1, :lo12:null_tail
    mov x2, #4

parse_literal:
1:
    cbz x2, 2f
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne error
    add x21, x21, #1
    sub x2, x2, #1
    b 1b
2:
    ret

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #0x2d
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]

number_integer:
    cmp w0, #0x30
    b.eq number_zero
    cmp w0, #0x31
    b.lo error
    cmp w0, #0x39
    b.hi error

number_integer_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo number_after_integer
    cmp w0, #0x39
    b.ls number_integer_digits
    b number_after_integer

number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo number_after_integer
    cmp w0, #0x39
    b.ls error

number_after_integer:
    cmp w0, #0x2e
    b.eq number_fraction
    cmp w0, #0x65
    b.eq number_exponent
    cmp w0, #0x45
    b.eq number_exponent
    ret

number_fraction:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo error
    cmp w0, #0x39
    b.hi error

number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo number_after_fraction
    cmp w0, #0x39
    b.ls number_fraction_digits

number_after_fraction:
    cmp w0, #0x65
    b.eq number_exponent
    cmp w0, #0x45
    b.eq number_exponent
    ret

number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x2b
    b.eq number_exponent_sign
    cmp w0, #0x2d
    b.ne number_exponent_first

number_exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]

number_exponent_first:
    cmp w0, #0x30
    b.lo error
    cmp w0, #0x39
    b.hi error

number_exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs number_done
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo number_done
    cmp w0, #0x39
    b.ls number_exponent_digits

number_done:
    ret

parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x5d
    b.eq array_close

array_value:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x5d
    b.eq array_close
    cmp w0, #0x2c
    b.ne error
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x5d
    b.eq error
    b array_value

array_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x7d
    b.eq object_close

object_member:
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne error
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x3a
    b.ne error
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x7d
    b.eq object_close
    cmp w0, #0x2c
    b.ne error
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs error
    ldrb w0, [x19, x21]
    cmp w0, #0x7d
    b.eq error
    b object_member

object_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

success:
    mov x0, #1
    adrp x1, ok_message
    add x1, x1, :lo12:ok_message
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x1, number_buffer
    add x1, x1, :lo12:number_buffer
    add x1, x1, #31
    mov w2, #0x0a
    strb w2, [x1]
    add x3, x1, #1
    mov x4, x21
    mov x5, #10

decimal_loop:
    udiv x6, x4, x5
    msub x7, x6, x5, x4
    add w7, w7, #0x30
    sub x1, x1, #1
    strb w7, [x1]
    mov x4, x6
    cbnz x4, decimal_loop

    sub x2, x3, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
true_tail:
    .ascii "true"
false_tail:
    .ascii "false"
null_tail:
    .ascii "null"

.bss
.balign 16
input_buffer:
    .zero 65536
number_buffer:
    .zero 32
