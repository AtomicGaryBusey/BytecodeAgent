.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 66500

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
.zero INPUT_SIZE
.align 16
output_buffer:
.zero OUTPUT_SIZE

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    cmp r13, INPUT_SIZE
    je encode_start
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    jle encode_start
    add r12, rax
    add r13, rax
    jmp read_loop

encode_start:
    test r13, r13
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea r14, [rsi + r13]
    lea rdi, [rip + output_buffer]
    lea r15, [rip + output_buffer]
    lea r8, [rip + alphabet]
    xor r10d, r10d

encode_full_loop:
    mov rax, r14
    sub rax, rsi
    cmp rax, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    add rsi, 3

    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    and ecx, 3
    shl ecx, 4
    mov r9d, ebx
    shr r9d, 4
    or ecx, r9d
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 1], cl

    mov ecx, ebx
    and ecx, 15
    shl ecx, 2
    mov r9d, edx
    shr r9d, 6
    or ecx, r9d
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 2], cl

    mov ecx, edx
    and ecx, 63
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 3], cl

    add rdi, 4
    add r10d, 4
    cmp r10d, 76
    jne encode_full_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r10d, r10d
    jmp encode_full_loop

encode_tail:
    test rax, rax
    jz finish_lines

    mov r9d, eax
    movzx eax, byte ptr [rsi]
    xor ebx, ebx
    cmp r9d, 1
    je tail_loaded
    movzx ebx, byte ptr [rsi + 1]

tail_loaded:
    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    and ecx, 3
    shl ecx, 4
    mov edx, ebx
    shr edx, 4
    or ecx, edx
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 1], cl

    cmp r9d, 1
    je tail_one_byte

    mov ecx, ebx
    and ecx, 15
    shl ecx, 2
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 2], cl
    mov byte ptr [rdi + 3], 61
    jmp tail_done

tail_one_byte:
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

tail_done:
    add rdi, 4
    add r10d, 4
    cmp r10d, 76
    jne finish_lines
    mov byte ptr [rdi], 10
    inc rdi
    xor r10d, r10d

finish_lines:
    test r10d, r10d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    mov rdx, rdi
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
