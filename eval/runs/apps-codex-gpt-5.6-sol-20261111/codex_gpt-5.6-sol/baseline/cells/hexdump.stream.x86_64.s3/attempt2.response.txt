.globl _start

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    mov edx, 65536
    sub rdx, r13
    lea rsi, [r12 + r13]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r15, [rip + output_buffer]
    mov rdi, r15
    lea r8, [rip + hex_digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r13
    jae final_offset

    mov eax, ebx
    call emit_hex8

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2

    mov r14, r13
    sub r14, rbx
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16

row_size_ready:
    lea r10, [r12 + rbx]
    xor ecx, ecx

hex_slot_loop:
    cmp rcx, r14
    jae empty_hex_slot

    movzx eax, byte ptr [r10 + rcx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp hex_slot_done

empty_hex_slot:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3

hex_slot_done:
    cmp ecx, 7
    jne no_middle_space
    mov byte ptr [rdi], 32
    inc rdi

no_middle_space:
    inc rcx
    cmp ecx, 16
    jb hex_slot_loop

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    xor ecx, ecx

ascii_loop:
    cmp rcx, r14
    jae ascii_done

    mov al, byte ptr [r10 + rcx]
    cmp al, 32
    jb ascii_nonprintable
    cmp al, 126
    jbe ascii_store

ascii_nonprintable:
    mov al, 46

ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add rbx, r14
    jmp row_loop

final_offset:
    mov eax, r13d
    call emit_hex8
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, r15
    mov rsi, r15
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

emit_hex8:
    mov ecx, 28

emit_hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns emit_hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero 65537
.align 16
output_buffer:
    .zero 323600
