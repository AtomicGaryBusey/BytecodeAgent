.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.globl _start

.text
_start:
    xor ebx, ebx

read_loop:
    cmp rbx, MAX_INPUT
    jae parse_input

    mov edx, MAX_INPUT
    sub rdx, rbx
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle parse_input
    add rbx, rax
    jmp read_loop

parse_input:
    xor r12d, r12d
    xor r13d, r13d
    xor r15d, r15d
    lea rdi, [rip + line_table]
    lea rsi, [rip + input_buffer]

parse_loop:
    cmp r12, rbx
    jae parse_remainder
    cmp byte ptr [rsi + r12], 10
    jne parse_next

    mov qword ptr [rdi], r13
    mov rax, r12
    sub rax, r13
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r15
    lea r13, [r12 + 1]

parse_next:
    inc r12
    jmp parse_loop

parse_remainder:
    cmp r13, rbx
    jae sort_begin
    mov qword ptr [rdi], r13
    mov rax, rbx
    sub rax, r13
    mov qword ptr [rdi + 8], rax
    inc r15

sort_begin:
    cmp r15, 1
    jbe output_begin

    lea r14, [rip + line_table]
    mov r12d, 1

sort_outer:
    cmp r12, r15
    jae output_begin

    mov r13, r12
    shl r13, 4
    mov r8, qword ptr [r14 + r13]
    mov r9, qword ptr [r14 + r13 + 8]

sort_inner:
    test r13, r13
    jz sort_place

    mov rdx, qword ptr [r14 + r13 - 8]
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz compare_lengths

    lea rsi, [rip + input_buffer]
    add rsi, r8
    lea rdi, [rip + input_buffer]
    add rdi, qword ptr [r14 + r13 - 16]
    repe cmpsb
    jne compare_difference

compare_lengths:
    cmp r9, rdx
    jb shift_previous
    jmp sort_place

compare_difference:
    jb shift_previous
    jmp sort_place

shift_previous:
    mov rax, qword ptr [r14 + r13 - 16]
    mov rdx, qword ptr [r14 + r13 - 8]
    mov qword ptr [r14 + r13], rax
    mov qword ptr [r14 + r13 + 8], rdx
    sub r13, 16
    jmp sort_inner

sort_place:
    mov qword ptr [r14 + r13], r8
    mov qword ptr [r14 + r13 + 8], r9
    inc r12
    jmp sort_outer

output_begin:
    xor r12d, r12d
    lea r14, [rip + line_table]

output_loop:
    cmp r12, r15
    jae exit_success

    mov r13, r12
    shl r13, 4
    mov rdx, qword ptr [r14 + r13 + 8]
    test rdx, rdx
    jz output_newline

    lea rsi, [rip + input_buffer]
    add rsi, qword ptr [r14 + r13]
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    syscall

    inc r12
    jmp output_loop

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero 32768
.align 16
line_table:
    .zero 16000
