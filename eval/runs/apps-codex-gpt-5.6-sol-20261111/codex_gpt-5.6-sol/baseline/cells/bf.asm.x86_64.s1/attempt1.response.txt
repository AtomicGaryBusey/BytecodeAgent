.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    cmp r13, 65536
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor ecx, ecx

find_delimiter:
    cmp rcx, r13
    je no_delimiter
    cmp byte ptr [r12 + rcx], 0x21
    je delimiter_found
    inc rcx
    jmp find_delimiter

delimiter_found:
    mov r14, rcx
    lea r8, [r12 + rcx + 1]
    lea r9, [r12 + r13]
    jmp validate

no_delimiter:
    mov r14, r13
    lea r8, [r12 + r13]
    mov r9, r8

validate:
    lea r10, [rip + jump_table]
    lea rsi, [rip + bracket_stack]
    xor ecx, ecx
    xor edx, edx

validate_loop:
    cmp rcx, r14
    je validate_done
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '['
    je validate_open
    cmp al, ']'
    je validate_close
    inc rcx
    jmp validate_loop

validate_open:
    mov dword ptr [rsi + rdx*4], ecx
    inc edx
    inc rcx
    jmp validate_loop

validate_close:
    test edx, edx
    jz invalid_program
    dec edx
    mov eax, dword ptr [rsi + rdx*4]
    mov dword ptr [r10 + rcx*4], eax
    mov dword ptr [r10 + rax*4], ecx
    inc rcx
    jmp validate_loop

validate_done:
    test edx, edx
    jnz invalid_program

    lea r15, [rip + tape]
    lea r13, [rip + output_buffer]
    xor ebx, ebx
    xor ebp, ebp

execute_loop:
    cmp rbx, r14
    je execution_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '>'
    je command_right
    cmp al, '<'
    je command_left
    cmp al, '+'
    je command_add
    cmp al, '-'
    je command_sub
    cmp al, '.'
    je command_output
    cmp al, ','
    je command_input
    cmp al, '['
    je command_open
    cmp al, ']'
    je command_close
    inc rbx
    jmp execute_loop

command_right:
    inc r15
    inc rbx
    jmp execute_loop

command_left:
    dec r15
    inc rbx
    jmp execute_loop

command_add:
    inc byte ptr [r15]
    inc rbx
    jmp execute_loop

command_sub:
    dec byte ptr [r15]
    inc rbx
    jmp execute_loop

command_output:
    mov al, byte ptr [r15]
    mov byte ptr [r13 + rbp], al
    inc rbp
    inc rbx
    jmp execute_loop

command_input:
    cmp r8, r9
    jae input_exhausted
    mov al, byte ptr [r8]
    inc r8
    mov byte ptr [r15], al
    inc rbx
    jmp execute_loop

input_exhausted:
    mov byte ptr [r15], 0
    inc rbx
    jmp execute_loop

command_open:
    cmp byte ptr [r15], 0
    jne advance_instruction
    mov ebx, dword ptr [r10 + rbx*4]
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r15], 0
    je advance_instruction
    mov ebx, dword ptr [r10 + rbx*4]
    inc rbx
    jmp execute_loop

advance_instruction:
    inc rbx
    jmp execute_loop

execution_done:
    test rbp, rbp
    jz successful_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, rbp
    syscall

successful_exit:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
jump_table:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
.align 16
tape:
    .zero 30000
.align 16
output_buffer:
    .zero 65536
