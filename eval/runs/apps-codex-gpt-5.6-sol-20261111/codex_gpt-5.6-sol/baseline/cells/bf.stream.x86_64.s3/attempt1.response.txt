.equ INPUT_CAP, 65536
.equ OUTPUT_CAP, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_STACK_SIZE, 4000
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_CAP
    je read_zero
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_CAP
    sub rdx, r12
    jmp do_read

read_zero:
    lea rsi, [rip + input_buffer]
    xor edx, edx

do_read:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r10, [rip + input_buffer]
    xor r8d, r8d

find_delimiter:
    cmp r8, r12
    jae no_delimiter
    cmp byte ptr [r10 + r8], 0x21
    je delimiter_found
    inc r8
    jmp find_delimiter

delimiter_found:
    mov r13, r8
    lea r15, [r10 + r8 + 1]
    lea rbp, [r10 + r12]
    jmp validate_program

no_delimiter:
    mov r13, r12
    lea r15, [r10 + r12]
    mov rbp, r15

validate_program:
    xor r8d, r8d
    xor r9d, r9d
    lea r11, [rip + bracket_map]
    lea rdx, [rip + bracket_stack]

validate_loop:
    cmp r8, r13
    jae validate_end
    movzx eax, byte ptr [r10 + r8]
    cmp al, '['
    je validate_open
    cmp al, ']'
    je validate_close
    inc r8
    jmp validate_loop

validate_open:
    mov dword ptr [rdx + r9*4], r8d
    inc r9
    inc r8
    jmp validate_loop

validate_close:
    test r9, r9
    jz invalid_program
    dec r9
    mov eax, dword ptr [rdx + r9*4]
    mov dword ptr [r11 + r8*4], eax
    mov dword ptr [r11 + rax*4], r8d
    inc r8
    jmp validate_loop

validate_end:
    test r9, r9
    jnz invalid_program

    mov r12, r13
    xor r13d, r13d
    lea r14, [rip + tape]
    xor ebx, ebx
    lea r11, [rip + bracket_map]

execute_loop:
    cmp r13, r12
    jae execution_done
    movzx eax, byte ptr [r10 + r13]

    cmp al, '>'
    je command_right
    cmp al, '<'
    je command_left
    cmp al, '+'
    je command_increment
    cmp al, '-'
    je command_decrement
    cmp al, '.'
    je command_output
    cmp al, ','
    je command_input
    cmp al, '['
    je command_open
    cmp al, ']'
    je command_close

    inc r13
    jmp execute_loop

command_right:
    inc r14
    inc r13
    jmp execute_loop

command_left:
    dec r14
    inc r13
    jmp execute_loop

command_increment:
    inc byte ptr [r14]
    inc r13
    jmp execute_loop

command_decrement:
    dec byte ptr [r14]
    inc r13
    jmp execute_loop

command_output:
    mov al, byte ptr [r14]
    lea rdx, [rip + output_buffer]
    mov byte ptr [rdx + rbx], al
    inc rbx
    inc r13
    jmp execute_loop

command_input:
    cmp r15, rbp
    jae input_exhausted
    mov al, byte ptr [r15]
    mov byte ptr [r14], al
    inc r15
    inc r13
    jmp execute_loop

input_exhausted:
    mov byte ptr [r14], 0
    inc r13
    jmp execute_loop

command_open:
    cmp byte ptr [r14], 0
    jne open_continue
    mov eax, dword ptr [r11 + r13*4]
    lea r13, [rax + 1]
    jmp execute_loop

open_continue:
    inc r13
    jmp execute_loop

command_close:
    cmp byte ptr [r14], 0
    je close_continue
    mov eax, dword ptr [r11 + r13*4]
    lea r13, [rax + 1]
    jmp execute_loop

close_continue:
    inc r13
    jmp execute_loop

execution_done:
    test rbx, rbx
    jz exit_success
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    syscall

exit_success:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

invalid_program:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero INPUT_CAP
output_buffer:
    .zero OUTPUT_CAP
tape:
    .zero TAPE_SIZE
.balign 16
bracket_map:
    .zero INPUT_CAP * 4
bracket_stack:
    .zero BRACKET_STACK_SIZE
