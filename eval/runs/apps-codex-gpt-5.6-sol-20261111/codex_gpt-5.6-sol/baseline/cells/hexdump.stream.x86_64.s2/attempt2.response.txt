.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 323600

.section .text
.globl _start
_start:
    lea r14, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12d, INPUT_SIZE
    je read_done

    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, INPUT_SIZE
    sub edx, r12d
    syscall

    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r15, [rip + output_buffer]
    mov rdi, r15
    lea r10, [rip + hex_digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae final_offset

    mov r13, r12
    sub r13, rbx
    cmp r13, 16
    jbe row_size_ready
    mov r13d, 16

row_size_ready:
    lea r11, [r14 + rbx]

    mov edx, ebx
    mov ecx, 28

row_offset_loop:
    mov eax, edx
    shr eax, cl
    and eax, 15
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub ecx, 4
    jns row_offset_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor r8d, r8d

hex_slots_loop:
    cmp r8, r13
    jae empty_hex_slot

    movzx r9d, byte ptr [r11 + r8]
    mov eax, r9d
    shr eax, 4
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al

    mov eax, r9d
    and eax, 15
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    jmp hex_slot_done

empty_hex_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '

hex_slot_done:
    add rdi, 3
    inc r8

    cmp r8d, 8
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi

no_middle_space:
    cmp r8d, 16
    jb hex_slots_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor r8d, r8d

ascii_loop:
    cmp r8, r13
    jae ascii_done

    movzx eax, byte ptr [r11 + r8]
    cmp al, 0x20
    jb ascii_nonprintable
    cmp al, 0x7e
    jbe ascii_store

ascii_nonprintable:
    mov al, '.'

ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add rbx, r13
    jmp row_loop

final_offset:
    mov edx, r12d
    mov ecx, 28

final_offset_loop:
    mov eax, edx
    shr eax, cl
    and eax, 15
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub ecx, 4
    jns final_offset_loop

    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE

.align 16
output_buffer:
    .zero OUTPUT_SIZE
