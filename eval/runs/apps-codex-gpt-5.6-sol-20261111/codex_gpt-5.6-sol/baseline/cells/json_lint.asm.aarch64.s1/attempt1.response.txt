.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, x19

read_loop:
    sub x2, x19, x20
    add x2, x2, #1
    lsl x2, x2, #16
    lsr x2, x2, #16
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x19
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.ne parse_error

parse_ok:
    mov x0, #1
    adrp x1, ok_message
    add x1, x1, :lo12:ok_message
    mov x2, #3
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

parse_error:
    mov x0, #1
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    sub x3, x21, x19
    adrp x4, number_buffer
    add x4, x4, :lo12:number_buffer
    add x5, x4, #32
    mov w6, #10
    strb w6, [x5, #-1]!
    cbnz x3, decimal_loop
    mov w6, #'0'
    strb w6, [x5, #-1]!
    b decimal_done

decimal_loop:
    mov x6, #10
    udiv x7, x3, x6
    msub x8, x7, x6, x3
    add w8, w8, #'0'
    strb w8, [x5, #-1]!
    mov x3, x7
    cbnz x3, decimal_loop

decimal_done:
    mov x0, #1
    mov x1, x5
    add x6, x4, #32
    sub x2, x6, x5
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

parse_ws:
    cmp x21, x20
    b.eq 2f
1:
    ldrb w0, [x21]
    cmp w0, #0x20
    b.eq 3f
    cmp w0, #0x09
    b.eq 3f
    cmp w0, #0x0a
    b.eq 3f
    cmp w0, #0x0d
    b.ne 2f
3:
    add x21, x21, #1
    cmp x21, x20
    b.ne 1b
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'"'
    b.eq value_string
    cmp w0, #'{'
    b.eq value_object
    cmp w0, #'['
    b.eq value_array
    cmp w0, #'t'
    b.eq value_true
    cmp w0, #'f'
    b.eq value_false
    cmp w0, #'n'
    b.eq value_null
    cmp w0, #'-'
    b.eq value_number
    cmp w0, #'0'
    b.lo parse_error
    cmp w0, #'9'
    b.hi parse_error

value_number:
    bl parse_number
    b value_done

value_string:
    bl parse_string
    b value_done

value_object:
    bl parse_object
    b value_done

value_array:
    bl parse_array
    b value_done

value_true:
    adrp x0, literal_true
    add x0, x0, :lo12:literal_true
    mov x1, #4
    bl parse_literal
    b value_done

value_false:
    adrp x0, literal_false
    add x0, x0, :lo12:literal_false
    mov x1, #5
    bl parse_literal
    b value_done

value_null:
    adrp x0, literal_null
    add x0, x0, :lo12:literal_null
    mov x1, #4
    bl parse_literal

value_done:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
1:
    cbz x1, 2f
    cmp x21, x20
    b.eq parse_error
    ldrb w2, [x21]
    ldrb w3, [x0], #1
    cmp w2, w3
    b.ne parse_error
    add x21, x21, #1
    sub x1, x1, #1
    b 1b
2:
    ret

parse_string:
    add x21, x21, #1
1:
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'"'
    b.eq 5f
    cmp w0, #0x20
    b.lo parse_error
    cmp w0, #0x5c
    b.eq 2f
    add x21, x21, #1
    b 1b

2:
    add x21, x21, #1
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'"'
    b.eq 4f
    cmp w0, #0x5c
    b.eq 4f
    cmp w0, #'/'
    b.eq 4f
    cmp w0, #'b'
    b.eq 4f
    cmp w0, #'f'
    b.eq 4f
    cmp w0, #'n'
    b.eq 4f
    cmp w0, #'r'
    b.eq 4f
    cmp w0, #'t'
    b.eq 4f
    cmp w0, #'u'
    b.ne parse_error
    add x21, x21, #1
    mov x1, #4
3:
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo parse_error
    cmp w0, #'9'
    b.ls 6f
    cmp w0, #'A'
    b.lo parse_error
    cmp w0, #'F'
    b.ls 6f
    cmp w0, #'a'
    b.lo parse_error
    cmp w0, #'f'
    b.hi parse_error
6:
    add x21, x21, #1
    subs x1, x1, #1
    b.ne 3b
    b 1b
4:
    add x21, x21, #1
    b 1b
5:
    add x21, x21, #1
    ret

parse_number:
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'-'
    b.ne 1f
    add x21, x21, #1
    cmp x21, x20
    b.eq parse_error
1:
    ldrb w0, [x21]
    cmp w0, #'0'
    b.eq 3f
    cmp w0, #'1'
    b.lo parse_error
    cmp w0, #'9'
    b.hi parse_error
2:
    add x21, x21, #1
    cmp x21, x20
    b.eq 4f
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo 4f
    cmp w0, #'9'
    b.ls 2b
    b 4f
3:
    add x21, x21, #1
4:
    cmp x21, x20
    b.eq 8f
    ldrb w0, [x21]
    cmp w0, #'.'
    b.ne 6f
    add x21, x21, #1
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo parse_error
    cmp w0, #'9'
    b.hi parse_error
5:
    add x21, x21, #1
    cmp x21, x20
    b.eq 8f
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo 6f
    cmp w0, #'9'
    b.ls 5b
6:
    cmp x21, x20
    b.eq 8f
    ldrb w0, [x21]
    cmp w0, #'e'
    b.eq 7f
    cmp w0, #'E'
    b.ne 8f
7:
    add x21, x21, #1
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'+'
    b.eq 9f
    cmp w0, #'-'
    b.ne 10f
9:
    add x21, x21, #1
    cmp x21, x20
    b.eq parse_error
10:
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo parse_error
    cmp w0, #'9'
    b.hi parse_error
11:
    add x21, x21, #1
    cmp x21, x20
    b.eq 8f
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lo 8f
    cmp w0, #'9'
    b.ls 11b
8:
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #']'
    b.eq 3f
1:
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #']'
    b.eq 3f
    cmp w0, #','
    b.ne parse_error
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    b 1b
3:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'}'
    b.eq 4f
1:
    cmp w0, #'"'
    b.ne parse_error
    bl parse_string
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #':'
    b.ne parse_error
    add x21, x21, #1
    bl parse_ws
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    cmp w0, #'}'
    b.eq 4f
    cmp w0, #','
    b.ne parse_error
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.eq parse_error
    ldrb w0, [x21]
    b 1b
4:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.balign 16
input_buffer:
    .space INPUT_SIZE
number_buffer:
    .space 32
