.globl _start

.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 66400

.text
_start:
    xor ebx, ebx

read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, INPUT_SIZE
    sub rdx, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r10, [rip + alphabet]
    mov r8, rbx
    xor r9d, r9d

encode_full:
    cmp r8, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl

    and eax, 63
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [rdi + 3], al

    add rsi, 3
    sub r8, 3
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne encode_full
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_full

encode_tail:
    test r8, r8
    jz finish_lines
    cmp r8, 1
    je encode_one

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]

    mov edx, eax
    shr edx, 2
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    and eax, 3
    shl eax, 4
    mov edx, ecx
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], al

    and ecx, 15
    shl ecx, 2
    movzx ecx, byte ptr [r10 + rcx]
    mov byte ptr [rdi + 2], cl
    mov byte ptr [rdi + 3], 61
    jmp tail_written

encode_one:
    movzx eax, byte ptr [rsi]

    mov edx, eax
    shr edx, 2
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

tail_written:
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne finish_lines
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d

finish_lines:
    test r9d, r9d
    jz prepare_write
    mov byte ptr [rdi], 10
    inc rdi

prepare_write:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_success
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
output_buffer:
    .zero OUTPUT_SIZE
