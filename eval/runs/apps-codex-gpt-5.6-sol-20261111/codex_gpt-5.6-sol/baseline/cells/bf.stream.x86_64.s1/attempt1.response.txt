.globl _start

.text
_start:
    lea r8, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ebx, ebx

find_separator:
    cmp rbx, r12
    jae no_separator
    cmp byte ptr [r8 + rbx], 0x21
    je separator_found
    inc rbx
    jmp find_separator

separator_found:
    lea r15, [r8 + rbx + 1]
    lea rbp, [r8 + r12]
    mov r12, rbx
    jmp validate_start

no_separator:
    lea r15, [r8 + r12]
    mov rbp, r15

validate_start:
    lea r10, [rip + jump_table]
    lea rsi, [rip + bracket_stack]
    xor ebx, ebx
    xor edx, edx

validate_loop:
    cmp rbx, r12
    jae validate_end
    movzx eax, byte ptr [r8 + rbx]
    cmp al, '['
    je validate_open
    cmp al, ']'
    je validate_close

validate_next:
    inc rbx
    jmp validate_loop

validate_open:
    mov qword ptr [rsi + rdx*8], rbx
    inc rdx
    jmp validate_next

validate_close:
    test rdx, rdx
    jz invalid_program
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov qword ptr [r10 + rbx*8], rax
    mov qword ptr [r10 + rax*8], rbx
    jmp validate_next

validate_end:
    test rdx, rdx
    jnz invalid_program

    lea r14, [rip + tape]
    lea r9, [rip + output_buffer]
    xor r13d, r13d
    xor ebx, ebx

execute_loop:
    cmp rbx, r12
    jae execution_done
    movzx eax, byte ptr [r8 + rbx]

    cmp al, '>'
    je command_right
    cmp al, '<'
    je command_left
    cmp al, '+'
    je command_plus
    cmp al, '-'
    je command_minus
    cmp al, '.'
    je command_output
    cmp al, ','
    je command_input
    cmp al, '['
    je command_open
    cmp al, ']'
    je command_close

execute_next:
    inc rbx
    jmp execute_loop

command_right:
    inc r14
    jmp execute_next

command_left:
    dec r14
    jmp execute_next

command_plus:
    inc byte ptr [r14]
    jmp execute_next

command_minus:
    dec byte ptr [r14]
    jmp execute_next

command_output:
    mov al, byte ptr [r14]
    mov byte ptr [r9 + r13], al
    inc r13
    jmp execute_next

command_input:
    cmp r15, rbp
    jae input_exhausted
    mov al, byte ptr [r15]
    inc r15
    mov byte ptr [r14], al
    jmp execute_next

input_exhausted:
    mov byte ptr [r14], 0
    jmp execute_next

command_open:
    cmp byte ptr [r14], 0
    jne execute_next
    mov rbx, qword ptr [r10 + rbx*8]
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r14], 0
    je execute_next
    mov rbx, qword ptr [r10 + rbx*8]
    inc rbx
    jmp execute_loop

execution_done:
    test r13, r13
    jz successful_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r13
    syscall

successful_exit:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
jump_table:
    .zero 524288
.align 16
bracket_stack:
    .zero 8000
.align 16
tape:
    .zero 30000
.align 16
output_buffer:
    .zero 65536
