.equ INPUT_MAX, 65536
.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60

.text
.globl _start
_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_MAX
    jae process_input

    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_MAX
    sub rdx, r12
    syscall

    test rax, rax
    jz process_input
    add r12, rax
    jmp read_loop

process_input:
    xor r8d, r8d
    xor r9d, r9d

scan_loop:
    cmp r9, r12
    jae finish_remainder
    cmp byte ptr [rbx + r9], 10
    je reverse_line
    inc r9
    jmp scan_loop

reverse_line:
    cmp r9, r8
    jbe line_done
    mov rax, r8
    mov rdx, r9
    dec rdx

reverse_line_loop:
    cmp rax, rdx
    jae line_done
    mov sil, byte ptr [rbx + rax]
    mov dil, byte ptr [rbx + rdx]
    mov byte ptr [rbx + rax], dil
    mov byte ptr [rbx + rdx], sil
    inc rax
    dec rdx
    jmp reverse_line_loop

line_done:
    inc r9
    mov r8, r9
    jmp scan_loop

finish_remainder:
    cmp r8, r12
    jae write_output

    mov rax, r8
    mov rdx, r12
    dec rdx

reverse_remainder_loop:
    cmp rax, rdx
    jae append_newline
    mov sil, byte ptr [rbx + rax]
    mov dil, byte ptr [rbx + rdx]
    mov byte ptr [rbx + rax], dil
    mov byte ptr [rbx + rdx], sil
    inc rax
    dec rdx
    jmp reverse_remainder_loop

append_newline:
    mov byte ptr [rbx + r12], 10
    inc r12

write_output:
    xor r14d, r14d

write_loop:
    cmp r14, r12
    jae exit_success

    mov eax, SYS_write
    mov edi, 1
    lea rsi, [rbx + r14]
    mov rdx, r12
    sub rdx, r14
    syscall

    test rax, rax
    jle exit_success
    add r14, rax
    jmp write_loop

exit_success:
    mov eax, SYS_exit
    xor edi, edi
    syscall

.bss
.balign 16
buffer:
    .zero 65537
