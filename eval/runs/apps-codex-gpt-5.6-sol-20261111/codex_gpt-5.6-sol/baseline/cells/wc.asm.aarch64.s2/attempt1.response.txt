.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    mov x2, #1
    lsl x2, x2, #16
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    b.le input_done

    add x21, x21, x0
    mov x2, x0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer

process_loop:
    ldrb w3, [x1], #1

    cmp w3, #10
    b.ne not_newline
    add x19, x19, #1

not_newline:
    cmp w3, #32
    b.eq whitespace

    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace

    cbnz x22, byte_done
    add x20, x20, #1
    mov x22, #1
    b byte_done

whitespace:
    mov x22, #0

byte_done:
    subs x2, x2, #1
    b.ne process_loop
    b read_loop

input_done:
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    mov x24, x23

    mov x0, x19
    mov x1, x24
    bl append_decimal
    mov x24, x0

    mov w2, #32
    strb w2, [x24], #1

    mov x0, x20
    mov x1, x24
    bl append_decimal
    mov x24, x0

    mov w2, #32
    strb w2, [x24], #1

    mov x0, x21
    mov x1, x24
    bl append_decimal
    mov x24, x0

    mov w2, #10
    strb w2, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

append_decimal:
    sub sp, sp, #32
    mov x2, sp
    mov x3, #0
    mov x5, #10

decimal_generate:
    udiv x4, x0, x5
    msub x6, x4, x5, x0
    add w6, w6, #48
    strb w6, [x2], #1
    add x3, x3, #1
    mov x0, x4
    cbnz x0, decimal_generate

decimal_copy:
    sub x2, x2, #1
    ldrb w6, [x2]
    strb w6, [x1], #1
    subs x3, x3, #1
    b.ne decimal_copy

    mov x0, x1
    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .space INPUT_SIZE
output_buffer:
    .space 64
