.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x22, x19

read_loop:
    sub x2, x19, x22
    mov x3, #1
    lsl x3, x3, #16
    add x2, x2, x3
    cbz x2, read_done
    mov x0, #0
    mov x1, x22
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    ble read_done
    add x22, x22, x0
    b read_loop

read_done:
    mov x20, x22
    mov x0, x19
    bl skip_ws
    cmp x0, x20
    b.eq incomplete_error
    bl parse_value
    cbz x0, report_error
    bl skip_ws
    cmp x0, x20
    b.ne trailing_error

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

incomplete_error:
    mov x21, x20
    b report_error

trailing_error:
    mov x21, x0

report_error:
    sub x23, x21, x19
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov w0, #'e'
    strb w0, [x24, #0]
    mov w0, #'r'
    strb w0, [x24, #1]
    strb w0, [x24, #2]
    mov w0, #'o'
    strb w0, [x24, #3]
    mov w0, #'r'
    strb w0, [x24, #4]
    mov w0, #' '
    strb w0, [x24, #5]
    mov w0, #'a'
    strb w0, [x24, #6]
    mov w0, #'t'
    strb w0, [x24, #7]
    mov w0, #' '
    strb w0, [x24, #8]

    adrp x25, digit_buf
    add x25, x25, :lo12:digit_buf
    mov x26, #0
    cbnz x23, decimal_loop
    mov w0, #'0'
    strb w0, [x25]
    mov x26, #1
    b decimal_copy

decimal_loop:
    mov x1, #10
    udiv x2, x23, x1
    msub x3, x2, x1, x23
    add w3, w3, #'0'
    strb w3, [x25, x26]
    add x26, x26, #1
    mov x23, x2
    cbnz x23, decimal_loop

decimal_copy:
    add x27, x24, #9
    mov x28, x26
decimal_copy_loop:
    sub x28, x28, #1
    ldrb w0, [x25, x28]
    strb w0, [x27], #1
    cbnz x28, decimal_copy_loop
    mov w0, #10
    strb w0, [x27], #1
    sub x2, x27, x24
    mov x0, #1
    mov x1, x24
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

skip_ws:
1:
    cmp x0, x20
    b.eq 2f
    ldrb w1, [x0]
    cmp w1, #0x20
    b.eq 3f
    cmp w1, #0x09
    b.eq 3f
    cmp w1, #0x0a
    b.eq 3f
    cmp w1, #0x0d
    b.ne 2f
3:
    add x0, x0, #1
    b 1b
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x0, x20
    b.eq value_fail
    ldrb w1, [x0]
    cmp w1, #'"'
    b.eq value_string
    cmp w1, #'{'
    b.eq value_object
    cmp w1, #'['
    b.eq value_array
    cmp w1, #'-'
    b.eq value_number
    cmp w1, #'0'
    b.lo value_literals
    cmp w1, #'9'
    b.ls value_number

value_literals:
    cmp w1, #'t'
    b.eq value_true
    cmp w1, #'f'
    b.eq value_false
    cmp w1, #'n'
    b.eq value_null
    b value_fail

value_string:
    bl parse_string
    ldp x29, x30, [sp], #16
    ret
value_object:
    bl parse_object
    ldp x29, x30, [sp], #16
    ret
value_array:
    bl parse_array
    ldp x29, x30, [sp], #16
    ret
value_number:
    bl parse_number
    ldp x29, x30, [sp], #16
    ret

value_true:
    add x0, x0, #1
    mov w2, #'r'
    bl match_byte
    cbz x0, value_return
    mov w2, #'u'
    bl match_byte
    cbz x0, value_return
    mov w2, #'e'
    bl match_byte
    b value_return

value_false:
    add x0, x0, #1
    mov w2, #'a'
    bl match_byte
    cbz x0, value_return
    mov w2, #'l'
    bl match_byte
    cbz x0, value_return
    mov w2, #'s'
    bl match_byte
    cbz x0, value_return
    mov w2, #'e'
    bl match_byte
    b value_return

value_null:
    add x0, x0, #1
    mov w2, #'u'
    bl match_byte
    cbz x0, value_return
    mov w2, #'l'
    bl match_byte
    cbz x0, value_return
    mov w2, #'l'
    bl match_byte

value_return:
    ldp x29, x30, [sp], #16
    ret

value_fail:
    mov x21, x0
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret

match_byte:
    cmp x0, x20
    b.eq match_fail
    ldrb w1, [x0]
    cmp w1, w2
    b.ne match_fail
    add x0, x0, #1
    ret
match_fail:
    mov x21, x0
    mov x0, #0
    ret

parse_string:
    add x0, x0, #1
string_loop:
    cmp x0, x20
    b.eq string_fail
    ldrb w1, [x0]
    cmp w1, #'"'
    b.eq string_done
    cmp w1, #0x20
    b.lo string_fail
    cmp w1, #0x5c
    b.eq string_escape
    add x0, x0, #1
    b string_loop

string_done:
    add x0, x0, #1
    ret

string_escape:
    add x0, x0, #1
    cmp x0, x20
    b.eq string_fail
    ldrb w1, [x0]
    cmp w1, #'"'
    b.eq string_escape_done
    cmp w1, #0x5c
    b.eq string_escape_done
    cmp w1, #'/'
    b.eq string_escape_done
    cmp w1, #'b'
    b.eq string_escape_done
    cmp w1, #'f'
    b.eq string_escape_done
    cmp w1, #'n'
    b.eq string_escape_done
    cmp w1, #'r'
    b.eq string_escape_done
    cmp w1, #'t'
    b.eq string_escape_done
    cmp w1, #'u'
    b.ne string_fail
    add x0, x0, #1
    mov x2, #4
string_hex_loop:
    cmp x0, x20
    b.eq string_fail
    ldrb w1, [x0]
    cmp w1, #'0'
    b.lo string_fail
    cmp w1, #'9'
    b.ls string_hex_ok
    cmp w1, #'A'
    b.lo string_fail
    cmp w1, #'F'
    b.ls string_hex_ok
    cmp w1, #'a'
    b.lo string_fail
    cmp w1, #'f'
    b.hi string_fail
string_hex_ok:
    add x0, x0, #1
    subs x2, x2, #1
    b.ne string_hex_loop
    b string_loop

string_escape_done:
    add x0, x0, #1
    b string_loop

string_fail:
    mov x21, x0
    mov x0, #0
    ret

parse_number:
    ldrb w1, [x0]
    cmp w1, #'-'
    b.ne number_int
    add x0, x0, #1
    cmp x0, x20
    b.eq number_fail

number_int:
    ldrb w1, [x0]
    cmp w1, #'0'
    b.eq number_zero
    cmp w1, #'1'
    b.lo number_fail
    cmp w1, #'9'
    b.hi number_fail
number_digits:
    add x0, x0, #1
    cmp x0, x20
    b.eq number_done
    ldrb w1, [x0]
    cmp w1, #'0'
    b.lo number_after_int
    cmp w1, #'9'
    b.ls number_digits
    b number_after_int

number_zero:
    add x0, x0, #1

number_after_int:
    cmp x0, x20
    b.eq number_done
    ldrb w1, [x0]
    cmp w1, #'.'
    b.ne number_exp_check
    add x0, x0, #1
    cmp x0, x20
    b.eq number_fail
    ldrb w1, [x0]
    cmp w1, #'0'
    b.lo number_fail
    cmp w1, #'9'
    b.hi number_fail
number_frac_digits:
    add x0, x0, #1
    cmp x0, x20
    b.eq number_done
    ldrb w1, [x0]
    cmp w1, #'0'
    b.lo number_exp_check
    cmp w1, #'9'
    b.ls number_frac_digits

number_exp_check:
    cmp x0, x20
    b.eq number_done
    ldrb w1, [x0]
    cmp w1, #'e'
    b.eq number_exp
    cmp w1, #'E'
    b.ne number_done

number_exp:
    add x0, x0, #1
    cmp x0, x20
    b.eq number_fail
    ldrb w1, [x0]
    cmp w1, #'+'
    b.eq number_exp_sign
    cmp w1, #'-'
    b.ne number_exp_digit
number_exp_sign:
    add x0, x0, #1
    cmp x0, x20
    b.eq number_fail
    ldrb w1, [x0]
number_exp_digit:
    cmp w1, #'0'
    b.lo number_fail
    cmp w1, #'9'
    b.hi number_fail
number_exp_digits:
    add x0, x0, #1
    cmp x0, x20
    b.eq number_done
    ldrb w1, [x0]
    cmp w1, #'0'
    b.lo number_done
    cmp w1, #'9'
    b.ls number_exp_digits

number_done:
    ret
number_fail:
    mov x21, x0
    mov x0, #0
    ret

parse_array:
    stp x29, x30, [sp, #-32]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.eq array_fail
    ldrb w1, [x0]
    cmp w1, #']'
    b.eq array_empty

array_value:
    bl parse_value
    cbz x0, array_return
    bl skip_ws
    cmp x0, x20
    b.eq array_fail
    ldrb w1, [x0]
    cmp w1, #']'
    b.eq array_close
    cmp w1, #','
    b.ne array_fail
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.eq array_fail
    b array_value

array_empty:
array_close:
    add x0, x0, #1
array_return:
    ldp x29, x30, [sp], #32
    ret
array_fail:
    mov x21, x0
    mov x0, #0
    b array_return

parse_object:
    stp x29, x30, [sp, #-32]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.eq object_fail
    ldrb w1, [x0]
    cmp w1, #'}'
    b.eq object_empty

object_member:
    cmp x0, x20
    b.eq object_fail
    ldrb w1, [x0]
    cmp w1, #'"'
    b.ne object_fail
    bl parse_string
    cbz x0, object_return
    bl skip_ws
    cmp x0, x20
    b.eq object_fail
    ldrb w1, [x0]
    cmp w1, #':'
    b.ne object_fail
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.eq object_fail
    bl parse_value
    cbz x0, object_return
    bl skip_ws
    cmp x0, x20
    b.eq object_fail
    ldrb w1, [x0]
    cmp w1, #'}'
    b.eq object_close
    cmp w1, #','
    b.ne object_fail
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.eq object_fail
    b object_member

object_empty:
object_close:
    add x0, x0, #1
object_return:
    ldp x29, x30, [sp], #32
    ret
object_fail:
    mov x21, x0
    mov x0, #0
    b object_return

.section .rodata
ok_msg:
    .ascii "ok\n"

.bss
.balign 16
input_buf:
    .space INPUT_SIZE
output_buf:
    .space 64
digit_buf:
    .space 32
