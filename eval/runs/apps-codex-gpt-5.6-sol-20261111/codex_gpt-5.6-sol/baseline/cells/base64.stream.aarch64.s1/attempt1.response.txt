.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 66400
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x9, #INPUT_MAX

read_loop:
    cmp x20, x9
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    cmp x0, #0
    b.lt read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    mov x21, x22
    mov w23, #0
    adrp x24, base64_table
    add x24, x24, :lo12:base64_table
    mov x25, x19
    mov x26, x20

encode_loop:
    cmp x26, #3
    b.lt encode_tail

    ldrb w4, [x25]
    ldrb w5, [x25, #1]
    ldrb w6, [x25, #2]
    add x25, x25, #3
    sub x26, x26, #3

    lsr w0, w4, #2
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w4, #3
    lsl w0, w0, #4
    orr w0, w0, w5, lsr #4
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w5, #15
    lsl w0, w0, #2
    orr w0, w0, w6, lsr #6
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w6, #63
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    b encode_loop

encode_tail:
    cbz x26, finish_lines
    cmp x26, #1
    b.eq encode_one

    ldrb w4, [x25]
    ldrb w5, [x25, #1]

    lsr w0, w4, #2
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w4, #3
    lsl w0, w0, #4
    orr w0, w0, w5, lsr #4
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w5, #15
    lsl w0, w0, #2
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    mov w0, #61
    bl emit_char
    b finish_lines

encode_one:
    ldrb w4, [x25]

    lsr w0, w4, #2
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    and w0, w4, #3
    lsl w0, w0, #4
    ldrb w0, [x24, w0, uxtw]
    bl emit_char

    mov w0, #61
    bl emit_char
    mov w0, #61
    bl emit_char

finish_lines:
    cbz w23, write_output
    mov w0, #10
    strb w0, [x21], #1
    mov w23, #0

write_output:
    sub x2, x21, x22
    cbz x2, exit_success
    mov x0, #1
    mov x1, x22
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

emit_char:
    strb w0, [x21], #1
    add w23, w23, #1
    cmp w23, #76
    b.ne 1f
    mov w0, #10
    strb w0, [x21], #1
    mov w23, #0
1:
    ret

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .space INPUT_MAX
.balign 16
output_buffer:
