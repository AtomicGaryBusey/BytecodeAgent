.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65537

.text
.globl _start
_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0

read_loop:
    movz x2, #1, lsl #16
    add x2, x2, #1
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0

scan_loop:
    cmp x22, x20
    b.eq final_line
    ldrb w9, [x19, x22]
    cmp w9, #10
    b.eq terminated_line
    add x22, x22, #1
    b scan_loop

terminated_line:
    sub x11, x22, x21
    cmp x11, #1
    b.ls write_terminated

    add x9, x19, x21
    sub x10, x22, #1
    add x10, x19, x10

reverse_terminated:
    cmp x9, x10
    b.hs write_terminated
    ldrb w12, [x9]
    ldrb w13, [x10]
    strb w13, [x9]
    strb w12, [x10]
    add x9, x9, #1
    sub x10, x10, #1
    b reverse_terminated

write_terminated:
    mov x0, #1
    add x1, x19, x21
    add x2, x11, #1
    mov x8, #SYS_write
    svc #0
    add x22, x22, #1
    mov x21, x22
    b scan_loop

final_line:
    cmp x21, x20
    b.eq exit_success

    sub x11, x20, x21
    cmp x11, #1
    b.ls write_final

    add x9, x19, x21
    sub x10, x20, #1
    add x10, x19, x10

reverse_final:
    cmp x9, x10
    b.hs write_final
    ldrb w12, [x9]
    ldrb w13, [x10]
    strb w13, [x9]
    strb w12, [x10]
    add x9, x9, #1
    sub x10, x10, #1
    b reverse_final

write_final:
    mov w9, #10
    strb w9, [x19, x20]
    mov x0, #1
    add x1, x19, x21
    add x2, x11, #1
    mov x8, #SYS_write
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
buffer:
    .space BUF_SIZE
