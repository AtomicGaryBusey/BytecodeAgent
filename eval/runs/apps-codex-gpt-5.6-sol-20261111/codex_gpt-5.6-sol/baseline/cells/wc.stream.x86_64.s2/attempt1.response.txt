.text
.globl _start
_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz input_done

    add r10, rax
    lea rsi, [rip + input_buffer]
    mov rcx, rax

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne classify
    inc r8

classify:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb non_whitespace
    cmp al, 13
    jbe whitespace

non_whitespace:
    test r12d, r12d
    jnz next_byte
    inc r9
    mov r12d, 1
    jmp next_byte

whitespace:
    xor r12d, r12d

next_byte:
    dec rcx
    jnz scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r8
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r9
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r10
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    lea rsi, [rip + digit_buffer + 32]
    mov ebx, 10

decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop

    lea rcx, [rip + digit_buffer + 32]
    sub rcx, rsi
    rep movsb
    ret

.bss
.align 16
input_buffer:
    .zero 4096
digit_buffer:
    .zero 32
output_buffer:
    .zero 64
