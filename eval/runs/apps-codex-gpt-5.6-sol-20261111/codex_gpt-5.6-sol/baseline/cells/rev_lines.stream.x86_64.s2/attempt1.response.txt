.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 65537

.text
.globl _start
_start:
    lea r8, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    je process_input

    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall

    test rax, rax
    jle process_input
    add r12, rax
    jmp read_loop

process_input:
    lea r9, [rip + output_buffer]
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d

scan_loop:
    cmp rbx, r12
    je final_line

    cmp byte ptr [r8 + rbx], 10
    je complete_line

    inc rbx
    jmp scan_loop

complete_line:
    mov rax, rbx

reverse_line:
    cmp rax, r14
    je append_newline
    dec rax
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [r9 + r13], dl
    inc r13
    jmp reverse_line

append_newline:
    mov byte ptr [r9 + r13], 10
    inc r13
    inc rbx
    mov r14, rbx
    jmp scan_loop

final_line:
    cmp r14, r12
    je write_output

    mov rax, r12

reverse_final:
    cmp rax, r14
    je append_final_newline
    dec rax
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [r9 + r13], dl
    inc r13
    jmp reverse_final

append_final_newline:
    mov byte ptr [r9 + r13], 10
    inc r13

write_output:
    xor r15d, r15d

write_loop:
    cmp r15, r13
    je exit_success

    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + r15]
    mov rdx, r13
    sub rdx, r15
    syscall

    test rax, rax
    jle exit_success
    add r15, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero OUTPUT_SIZE
