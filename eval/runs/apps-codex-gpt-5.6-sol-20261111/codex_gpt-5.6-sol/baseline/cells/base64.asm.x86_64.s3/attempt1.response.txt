.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 66400

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
.zero INPUT_SIZE
output_buffer:
.zero OUTPUT_SIZE

.text
.globl _start
_start:
    xor ebx, ebx

read_loop:
    cmp ebx, INPUT_SIZE
    je read_done

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, INPUT_SIZE
    sub edx, ebx
    syscall

    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r8, [rip + alphabet]
    mov rcx, rbx
    xor r12d, r12d

encode_groups:
    cmp rcx, 3
    jb encode_tail

    movzx r13d, byte ptr [rsi]
    shl r13d, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or r13d, edx
    movzx edx, byte ptr [rsi + 2]
    or r13d, edx

    mov eax, r13d
    shr eax, 18
    call emit_index

    mov eax, r13d
    shr eax, 12
    and eax, 63
    call emit_index

    mov eax, r13d
    shr eax, 6
    and eax, 63
    call emit_index

    mov eax, r13d
    and eax, 63
    call emit_index

    add rsi, 3
    sub rcx, 3
    jmp encode_groups

encode_tail:
    cmp rcx, 1
    je encode_one
    cmp rcx, 2
    je encode_two
    jmp finish_output

encode_one:
    movzx r13d, byte ptr [rsi]

    mov eax, r13d
    shr eax, 2
    call emit_index

    mov eax, r13d
    and eax, 3
    shl eax, 4
    call emit_index

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
    jmp finish_output

encode_two:
    movzx r13d, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]

    mov eax, r13d
    shr eax, 2
    call emit_index

    and r13d, 3
    shl r13d, 4
    mov eax, edx
    shr eax, 4
    or eax, r13d
    call emit_index

    mov eax, edx
    and eax, 15
    shl eax, 2
    call emit_index

    mov al, '='
    call emit_char

finish_output:
    test r12d, r12d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

emit_index:
    movzx eax, byte ptr [r8 + rax]

emit_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r12d
    cmp r12d, 76
    jne emit_done
    mov byte ptr [rdi], 10
    inc rdi
    xor r12d, r12d

emit_done:
    ret
