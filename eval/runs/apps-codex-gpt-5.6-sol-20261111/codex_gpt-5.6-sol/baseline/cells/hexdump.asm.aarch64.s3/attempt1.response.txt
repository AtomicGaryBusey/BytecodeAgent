.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, x19
    mov x24, x20
    mov x25, #0
    adrp x27, hex_digits
    add x27, x27, :lo12:hex_digits

row_loop:
    cbz x24, final_offset

    cmp x24, #16
    mov x26, #16
    csel x26, x24, x26, lo

    mov w0, w25
    mov x1, x22
    bl emit_hex8
    mov x22, x1

    mov w2, #32
    strb w2, [x22], #1
    strb w2, [x22], #1

    mov x28, #0

hex_slot_loop:
    cmp x28, x26
    b.hs empty_hex_slot

    ldrb w5, [x23, x28]
    lsr w6, w5, #4
    and w6, w6, #15
    ldrb w7, [x27, x6]
    strb w7, [x22], #1
    and w6, w5, #15
    ldrb w7, [x27, x6]
    strb w7, [x22], #1
    mov w7, #32
    strb w7, [x22], #1
    b hex_slot_done

empty_hex_slot:
    mov w7, #32
    strb w7, [x22], #1
    strb w7, [x22], #1
    strb w7, [x22], #1

hex_slot_done:
    cmp x28, #7
    b.ne no_middle_space
    mov w7, #32
    strb w7, [x22], #1

no_middle_space:
    add x28, x28, #1
    cmp x28, #16
    b.lo hex_slot_loop

    mov w7, #32
    strb w7, [x22], #1
    mov w7, #124
    strb w7, [x22], #1

    mov x28, #0

ascii_loop:
    cmp x28, x26
    b.hs ascii_done
    ldrb w5, [x23, x28]
    cmp w5, #32
    b.lo ascii_dot
    cmp w5, #126
    b.hi ascii_dot
    strb w5, [x22], #1
    b ascii_next

ascii_dot:
    mov w5, #46
    strb w5, [x22], #1

ascii_next:
    add x28, x28, #1
    b ascii_loop

ascii_done:
    mov w7, #124
    strb w7, [x22], #1
    mov w7, #10
    strb w7, [x22], #1

    add x23, x23, x26
    add x25, x25, x26
    sub x24, x24, x26
    b row_loop

final_offset:
    mov w0, w20
    mov x1, x22
    bl emit_hex8
    mov x22, x1
    mov w2, #10
    strb w2, [x22], #1

    sub x20, x22, x21
    mov x19, x21

write_loop:
    cbz x20, exit_success
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #64
    svc #0
    add x19, x19, x0
    sub x20, x20, x0
    b write_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

emit_hex8:
    mov w2, #28

emit_hex8_loop:
    lsr w3, w0, w2
    and w3, w3, #15
    ldrb w4, [x27, x3]
    strb w4, [x1], #1
    subs w2, w2, #4
    b.pl emit_hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 323600
