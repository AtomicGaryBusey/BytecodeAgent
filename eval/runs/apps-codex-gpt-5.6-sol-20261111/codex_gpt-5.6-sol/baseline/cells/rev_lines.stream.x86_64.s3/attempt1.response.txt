.equ CAP, 65536
.equ OUTCAP, 65537

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, CAP
    je process_input
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz process_input
    js exit_success
    add r12, rax
    jmp read_loop

process_input:
    lea r13, [rip + output_buffer]
    mov r14, r13
    xor r8d, r8d
    xor r9d, r9d

scan_loop:
    cmp r8, r12
    je final_line
    cmp byte ptr [rbx + r8], 10
    je complete_line
    inc r8
    jmp scan_loop

complete_line:
    mov r10, r8

reverse_line:
    cmp r10, r9
    je append_newline
    dec r10
    mov al, byte ptr [rbx + r10]
    mov byte ptr [r14], al
    inc r14
    jmp reverse_line

append_newline:
    mov byte ptr [r14], 10
    inc r14
    inc r8
    mov r9, r8
    jmp scan_loop

final_line:
    cmp r9, r12
    je write_output
    mov r10, r12

reverse_final:
    cmp r10, r9
    je append_final_newline
    dec r10
    mov al, byte ptr [rbx + r10]
    mov byte ptr [r14], al
    inc r14
    jmp reverse_final

append_final_newline:
    mov byte ptr [r14], 10
    inc r14

write_output:
    mov rdx, r14
    sub rdx, r13
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero CAP
output_buffer:
    .zero OUTCAP
