.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0
    movz x28, #1, lsl #16

read_loop:
    cmp x19, x28
    b.eq read_done
    mov x0, #0
    add x1, x20, x19
    sub x2, x28, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf

row_loop:
    cmp x21, x19
    b.hs final_offset

    sub x27, x19, x21
    mov x22, #16
    cmp x27, #16
    csel x22, x27, x22, lo

    mov x24, x23
    mov x26, #28

offset_loop:
    lsr x27, x21, x26
    and x27, x27, #15
    ldrb w27, [x25, x27]
    strb w27, [x24], #1
    subs x26, x26, #4
    b.ge offset_loop

    mov w27, #32
    strb w27, [x24], #1
    strb w27, [x24], #1

    mov x26, #0

hex_slot_loop:
    cmp x26, x22
    b.hs empty_slot

    add x27, x20, x21
    ldrb w27, [x27, x26]
    lsr w0, w27, #4
    and x0, x0, #15
    ldrb w0, [x25, x0]
    strb w0, [x24], #1
    and x0, x27, #15
    ldrb w0, [x25, x0]
    strb w0, [x24], #1
    mov w0, #32
    strb w0, [x24], #1
    b slot_done

empty_slot:
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1

slot_done:
    cmp x26, #7
    b.ne no_middle_space
    mov w0, #32
    strb w0, [x24], #1

no_middle_space:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_slot_loop

    mov w0, #32
    strb w0, [x24], #1
    mov w0, #124
    strb w0, [x24], #1

    mov x26, #0

ascii_loop:
    cmp x26, x22
    b.hs ascii_done
    add x27, x20, x21
    ldrb w0, [x27, x26]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.hi ascii_dot
    b ascii_store

ascii_dot:
    mov w0, #46

ascii_store:
    strb w0, [x24], #1
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0

    add x21, x21, x22
    b row_loop

final_offset:
    mov x24, x23
    mov x26, #28

final_hex_loop:
    lsr x27, x19, x26
    and x27, x27, #15
    ldrb w27, [x25, x27]
    strb w27, [x24], #1
    subs x26, x26, #4
    b.ge final_hex_loop

    mov w0, #10
    strb w0, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 80
