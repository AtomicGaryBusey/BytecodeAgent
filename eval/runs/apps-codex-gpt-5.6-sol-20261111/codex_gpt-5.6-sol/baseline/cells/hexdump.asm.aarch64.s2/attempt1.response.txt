.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #INPUT_SIZE
    cmp x20, x2
    b.eq read_done
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x24, hex_digits
    add x24, x24, :lo12:hex_digits
    adrp x29, line_buffer
    add x29, x29, :lo12:line_buffer
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset

    sub x22, x20, x21
    cmp x22, #16
    b.ls row_size_ready
    mov x22, #16

row_size_ready:
    mov x23, x29
    mov x26, #28
    mov x28, #8

offset_hex_loop:
    lsr x27, x21, x26
    and x27, x27, #15
    ldrb w27, [x24, x27]
    strb w27, [x23], #1
    sub x26, x26, #4
    subs x28, x28, #1
    b.ne offset_hex_loop

    mov w27, #32
    strb w27, [x23], #1
    strb w27, [x23], #1

    mov x25, #0

slot_loop:
    cmp x25, x22
    b.hs empty_slot

    add x26, x19, x21
    ldrb w27, [x26, x25]
    lsr w28, w27, #4
    ldrb w28, [x24, x28]
    strb w28, [x23], #1
    and w28, w27, #15
    ldrb w28, [x24, x28]
    strb w28, [x23], #1
    mov w28, #32
    strb w28, [x23], #1
    b slot_done

empty_slot:
    mov w28, #32
    strb w28, [x23], #1
    strb w28, [x23], #1
    strb w28, [x23], #1

slot_done:
    cmp x25, #7
    b.ne no_middle_space
    mov w28, #32
    strb w28, [x23], #1

no_middle_space:
    add x25, x25, #1
    cmp x25, #16
    b.ne slot_loop

    mov w27, #32
    strb w27, [x23], #1
    mov w27, #124
    strb w27, [x23], #1

    mov x25, #0

ascii_loop:
    cmp x25, x22
    b.hs ascii_done
    add x26, x19, x21
    ldrb w27, [x26, x25]
    cmp w27, #0x20
    b.lo ascii_dot
    cmp w27, #0x7e
    b.hi ascii_dot
    b ascii_store

ascii_dot:
    mov w27, #46

ascii_store:
    strb w27, [x23], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w27, #124
    strb w27, [x23], #1
    mov w27, #10
    strb w27, [x23], #1

    mov x0, #1
    mov x1, x29
    sub x2, x23, x29
    mov x8, #SYS_write
    svc #0

    add x21, x21, x22
    b row_loop

final_offset:
    mov x23, x29
    mov x26, #28
    mov x28, #8

final_hex_loop:
    lsr x27, x20, x26
    and x27, x27, #15
    ldrb w27, [x24, x27]
    strb w27, [x23], #1
    sub x26, x26, #4
    subs x28, x28, #1
    b.ne final_hex_loop

    mov w27, #10
    strb w27, [x23], #1

    mov x0, #1
    mov x1, x29
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
.balign 16
line_buffer:
    .zero 128
