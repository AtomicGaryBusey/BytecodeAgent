.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 323600

.section .text
.globl _start
_start:
    lea r14, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    je format_input
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle format_input
    add r12, rax
    jmp read_loop

format_input:
    mov r13, r12
    lea r15, [rip + output_buffer]
    mov rdi, r15
    xor r12d, r12d

row_loop:
    cmp r12, r13
    jae final_offset

    mov eax, r12d
    call emit_hex8
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2

    mov rbp, r13
    sub rbp, r12
    cmp rbp, 16
    jbe row_count_ready
    mov ebp, 16

row_count_ready:
    xor ebx, ebx

slot_loop:
    cmp ebx, ebp
    jae empty_slot

    movzx eax, byte ptr [r14 + r12]
    add r12, 1
    mov edx, eax
    shr eax, 4
    lea r10, [rip + hex_digits]
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp slot_done

empty_slot:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3

slot_done:
    inc ebx
    cmp ebx, 8
    jne no_middle_space
    mov byte ptr [rdi], 32
    inc rdi

no_middle_space:
    cmp ebx, 16
    jb slot_loop

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    mov rax, r12
    sub rax, rbp
    xor ebx, ebx

ascii_loop:
    cmp ebx, ebp
    jae ascii_done
    movzx edx, byte ptr [r14 + rax]
    cmp edx, 32
    jb ascii_dot
    cmp edx, 126
    ja ascii_dot
    mov byte ptr [rdi], dl
    jmp ascii_stored

ascii_dot:
    mov byte ptr [rdi], 46

ascii_stored:
    inc rdi
    inc rax
    inc ebx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    jmp row_loop

final_offset:
    mov eax, r13d
    call emit_hex8
    mov byte ptr [rdi], 10
    inc rdi

    mov r12, r15
    mov r13, rdi
    sub r13, r15

write_loop:
    test r13, r13
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle exit_success
    add r12, rax
    sub r13, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov r8d, eax
    lea r10, [rip + hex_digits]
    mov edx, 28

hex8_loop:
    mov eax, r8d
    mov ecx, edx
    shr eax, cl
    and eax, 15
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub edx, 4
    jns hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
.balign 16
output_buffer:
    .zero OUTPUT_SIZE
