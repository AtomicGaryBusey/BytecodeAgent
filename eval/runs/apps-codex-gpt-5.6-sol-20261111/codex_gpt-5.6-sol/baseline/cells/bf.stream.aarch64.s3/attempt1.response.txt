.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x9, #0

find_delimiter:
    cmp x9, x20
    b.hs no_delimiter
    ldrb w10, [x19, x9]
    cmp w10, #0x21
    b.eq delimiter_found
    add x9, x9, #1
    b find_delimiter

delimiter_found:
    mov x21, x9
    add x22, x19, x9
    add x22, x22, #1
    add x23, x19, x20
    b validate_setup

no_delimiter:
    mov x21, x20
    add x22, x19, x20
    mov x23, x22

validate_setup:
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x10, bracket_stack
    add x10, x10, :lo12:bracket_stack
    mov x11, #0
    mov x12, #0

validate_loop:
    cmp x12, x21
    b.hs validate_end
    ldrb w13, [x19, x12]
    cmp w13, #0x5b
    b.eq validate_open
    cmp w13, #0x5d
    b.eq validate_close
    add x12, x12, #1
    b validate_loop

validate_open:
    str w12, [x10, x11, lsl #2]
    add x11, x11, #1
    add x12, x12, #1
    b validate_loop

validate_close:
    cbz x11, invalid_program
    sub x11, x11, #1
    ldr w14, [x10, x11, lsl #2]
    str w12, [x26, x14, lsl #2]
    str w14, [x26, x12, lsl #2]
    add x12, x12, #1
    b validate_loop

validate_end:
    cbnz x11, invalid_program

    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x27, output_buffer
    add x27, x27, :lo12:output_buffer
    mov x28, #0
    mov x25, #0

execute_loop:
    cmp x25, x21
    b.hs execution_done
    ldrb w9, [x19, x25]

    cmp w9, #0x3e
    b.eq command_right
    cmp w9, #0x3c
    b.eq command_left
    cmp w9, #0x2b
    b.eq command_increment
    cmp w9, #0x2d
    b.eq command_decrement
    cmp w9, #0x2e
    b.eq command_output
    cmp w9, #0x2c
    b.eq command_input
    cmp w9, #0x5b
    b.eq command_open
    cmp w9, #0x5d
    b.eq command_close

next_command:
    add x25, x25, #1
    b execute_loop

command_right:
    add x24, x24, #1
    b next_command

command_left:
    sub x24, x24, #1
    b next_command

command_increment:
    ldrb w10, [x24]
    add w10, w10, #1
    strb w10, [x24]
    b next_command

command_decrement:
    ldrb w10, [x24]
    sub w10, w10, #1
    strb w10, [x24]
    b next_command

command_output:
    ldrb w10, [x24]
    strb w10, [x27, x28]
    add x28, x28, #1
    b next_command

command_input:
    cmp x22, x23
    b.hs input_exhausted
    ldrb w10, [x22], #1
    strb w10, [x24]
    b next_command

input_exhausted:
    strb wzr, [x24]
    b next_command

command_open:
    ldrb w10, [x24]
    cbnz w10, next_command
    ldr w14, [x26, x25, lsl #2]
    add x25, x14, #1
    b execute_loop

command_close:
    ldrb w10, [x24]
    cbz w10, next_command
    ldr w14, [x26, x25, lsl #2]
    add x25, x14, #1
    b execute_loop

execution_done:
    cbz x28, successful_exit
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0

successful_exit:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
input_buffer:
    .zero 65536
.balign 16
jump_table:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
.balign 16
output_buffer:
    .zero 65536
