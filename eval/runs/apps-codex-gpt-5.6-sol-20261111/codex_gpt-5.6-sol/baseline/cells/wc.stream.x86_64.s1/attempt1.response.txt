.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    lea rsi, [rip + input_buffer]
    mov rbx, rax

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb non_whitespace
    cmp al, 13
    jbe whitespace

non_whitespace:
    test r15d, r15d
    jne scan_next
    inc r13
    mov r15d, 1
    jmp scan_next

whitespace:
    xor r15d, r15d

scan_next:
    dec rbx
    jne scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_uint
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_uint:
    test rax, rax
    jne convert_digits
    mov byte ptr [rdi], 48
    inc rdi
    ret

convert_digits:
    mov ebx, 10
    xor r8d, r8d

division_loop:
    xor edx, edx
    div rbx
    add edx, 48
    push rdx
    inc r8
    test rax, rax
    jne division_loop

copy_digits:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r8
    jne copy_digits
    ret

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
