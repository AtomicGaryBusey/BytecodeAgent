.text
.globl _start
_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    cmp x20, x21
    b.eq process_input
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, process_input
    add x20, x20, x0
    b read_loop

process_input:
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x23, x20
    b.eq end_of_input
    ldrb w0, [x19, x23]
    cmp w0, #10
    b.eq found_newline
    add x23, x23, #1
    b scan_loop

found_newline:
    mov x24, x22
    sub x25, x23, #1

reverse_line:
    cmp x24, x25
    b.hs line_reversed
    ldrb w0, [x19, x24]
    ldrb w1, [x19, x25]
    strb w1, [x19, x24]
    strb w0, [x19, x25]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_line

line_reversed:
    add x23, x23, #1
    mov x22, x23
    b scan_loop

end_of_input:
    cmp x22, x20
    b.eq write_output
    mov x24, x22
    sub x25, x20, #1

reverse_final:
    cmp x24, x25
    b.hs append_newline
    ldrb w0, [x19, x24]
    ldrb w1, [x19, x25]
    strb w1, [x19, x24]
    strb w0, [x19, x25]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_final

append_newline:
    mov w0, #10
    strb w0, [x19, x20]
    add x20, x20, #1

write_output:
    cbz x20, exit_success
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
buffer:
    .zero 65537
