.globl _start

.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 324000

.text
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12d, INPUT_SIZE
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub edx, r12d
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output_buffer]
    lea r9, [rip + hex_digits]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16

row_size_ready:
    mov eax, r13d
    call emit_hex8

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2

    lea rsi, [rbx + r13]
    xor r15d, r15d

slot_loop:
    cmp r15, r14
    jae empty_slot

    movzx edx, byte ptr [rsi + r15]
    mov ecx, edx
    shr ecx, 4
    mov al, byte ptr [r9 + rcx]
    mov byte ptr [rdi], al
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    jmp slot_done

empty_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '

slot_done:
    add rdi, 3
    cmp r15d, 7
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi

no_middle_space:
    inc r15
    cmp r15d, 16
    jb slot_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2

    xor r15d, r15d

ascii_loop:
    cmp r15, r14
    jae ascii_done
    movzx eax, byte ptr [rsi + r15]
    cmp eax, 0x20
    jb nonprintable
    cmp eax, 0x7e
    ja nonprintable
    mov byte ptr [rdi], al
    jmp ascii_stored

nonprintable:
    mov byte ptr [rdi], '.'

ascii_stored:
    inc rdi
    inc r15
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call emit_hex8
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov ecx, 28

hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
output_buffer:
    .zero OUTPUT_SIZE
