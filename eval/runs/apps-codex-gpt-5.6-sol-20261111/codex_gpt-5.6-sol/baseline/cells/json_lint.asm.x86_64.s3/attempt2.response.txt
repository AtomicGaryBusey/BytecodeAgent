.equ INPUT_SIZE, 65536
.equ OK_MESSAGE_LEN, 3
.equ ERROR_PREFIX_LEN, 9

.section .text
.globl _start

_start:
    lea rbx, [rip + input_buffer]
    xor r15d, r15d

read_loop:
    cmp r15, INPUT_SIZE
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r15]
    mov edx, INPUT_SIZE
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop

read_done:
    lea r12, [rip + input_buffer]
    lea r13, [r12 + r15]
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne parse_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, OK_MESSAGE_LEN
    syscall

    mov eax, 60
    xor edi, edi
    syscall

parse_value:
    cmp r12, r13
    je parse_error

    movzx eax, byte ptr [r12]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe parse_number
    jmp parse_error

parse_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4

parse_literal:
    test ecx, ecx
    jz literal_done
    cmp r12, r13
    je parse_error
    mov al, byte ptr [r12]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc r12
    inc rsi
    dec ecx
    jmp parse_literal

literal_done:
    ret

parse_string:
    inc r12

string_loop:
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_done
    cmp al, 0x20
    jb parse_error
    cmp al, 0x5c
    je string_escape
    inc r12
    jmp string_loop

string_done:
    inc r12
    ret

string_escape:
    inc r12
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je simple_escape
    cmp al, 0x5c
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    je unicode_escape
    jmp parse_error

simple_escape:
    inc r12
    jmp string_loop

unicode_escape:
    inc r12
    mov ecx, 4

unicode_loop:
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb unicode_lower
    cmp al, '9'
    jbe unicode_digit_ok

unicode_lower:
    cmp al, 'a'
    jb unicode_upper
    cmp al, 'f'
    jbe unicode_digit_ok

unicode_upper:
    cmp al, 'A'
    jb parse_error
    cmp al, 'F'
    ja parse_error

unicode_digit_ok:
    inc r12
    dec ecx
    jnz unicode_loop
    jmp string_loop

parse_number:
    movzx eax, byte ptr [r12]
    cmp al, '-'
    jne number_integer
    inc r12
    cmp r12, r13
    je parse_error

number_integer:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error
    inc r12

number_digits:
    cmp r12, r13
    je number_after_integer
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_after_integer
    cmp al, '9'
    ja number_after_integer
    inc r12
    jmp number_digits

number_zero:
    inc r12

number_after_integer:
    cmp r12, r13
    je number_done
    cmp byte ptr [r12], '.'
    jne number_exponent
    inc r12
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

fraction_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_exponent
    cmp al, '9'
    jbe fraction_digits

number_exponent:
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_done

exponent_start:
    inc r12
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    je exponent_sign
    jmp exponent_first_digit

exponent_sign:
    inc r12
    cmp r12, r13
    je parse_error

exponent_first_digit:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

exponent_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits

number_done:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], ']'
    je array_done

array_value:
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne parse_error
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], ']'
    je parse_error
    jmp array_value

array_done:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], '}'
    je object_done

object_member:
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], ':'
    jne parse_error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_error
    movzx eax, byte ptr [r12]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne parse_error
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_error
    cmp byte ptr [r12], '}'
    je parse_error
    jmp object_member

object_done:
    inc r12
    ret

skip_ws:
    cmp r12, r13
    je skip_ws_done
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    jne skip_ws_done

skip_ws_advance:
    inc r12
    jmp skip_ws

skip_ws_done:
    ret

parse_error:
    lea rax, [rip + input_buffer]
    mov r8, r12
    sub r8, rax

    lea r10, [rip + output_buffer + 31]
    mov byte ptr [r10], 0x0a
    mov r9, 1
    mov rax, r8
    test rax, rax
    jnz decimal_loop

    dec r10
    mov byte ptr [r10], '0'
    inc r9
    jmp decimal_done

decimal_loop:
    xor edx, edx
    mov r11d, 10
    div r11
    add dl, '0'
    dec r10
    mov byte ptr [r10], dl
    inc r9
    test rax, rax
    jnz decimal_loop

decimal_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, ERROR_PREFIX_LEN
    syscall

    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r9
    syscall

    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok_message:
    .ascii "ok\n"

error_prefix:
    .ascii "error at "

literal_true:
    .ascii "true"

literal_false:
    .ascii "false"

literal_null:
    .ascii "null"

.bss
.balign 16
input_buffer:
    .zero 65536

output_buffer:
    .zero 32
