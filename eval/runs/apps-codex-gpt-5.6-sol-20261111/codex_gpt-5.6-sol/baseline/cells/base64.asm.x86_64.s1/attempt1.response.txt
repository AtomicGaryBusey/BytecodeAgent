.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 66560

.section .text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_MAX
    jae read_done
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_MAX
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buffer]
    lea rbp, [rip + output_buffer]
    lea rbx, [rip + alphabet]
    xor r13d, r13d

    mov rax, r12
    xor edx, edx
    mov ecx, 3
    div rcx
    mov r14, rax
    mov r15, rdx

encode_groups:
    test r14, r14
    jz encode_tail

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]
    add rsi, 3

    mov edx, r8d
    shr edx, 2
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp], al

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov eax, r9d
    shr eax, 4
    or edx, eax
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 1], al

    mov edx, r9d
    and edx, 15
    shl edx, 2
    mov eax, r10d
    shr eax, 6
    or edx, eax
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 2], al

    mov edx, r10d
    and edx, 63
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 3], al

    add rbp, 4
    add r13d, 4
    cmp r13d, 76
    jne group_next
    mov byte ptr [rbp], 10
    inc rbp
    xor r13d, r13d

group_next:
    dec r14
    jmp encode_groups

encode_tail:
    cmp r15d, 1
    je tail_one
    cmp r15d, 2
    je tail_two
    jmp finish_lines

tail_one:
    movzx r8d, byte ptr [rsi]

    mov edx, r8d
    shr edx, 2
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp], al

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 1], al

    mov byte ptr [rbp + 2], 61
    mov byte ptr [rbp + 3], 61
    add rbp, 4
    add r13d, 4
    cmp r13d, 76
    jne finish_lines
    mov byte ptr [rbp], 10
    inc rbp
    xor r13d, r13d
    jmp finish_lines

tail_two:
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]

    mov edx, r8d
    shr edx, 2
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp], al

    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov eax, r9d
    shr eax, 4
    or edx, eax
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 1], al

    mov edx, r9d
    and edx, 15
    shl edx, 2
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rbp + 2], al

    mov byte ptr [rbp + 3], 61
    add rbp, 4
    add r13d, 4
    cmp r13d, 76
    jne finish_lines
    mov byte ptr [rbp], 10
    inc rbp
    xor r13d, r13d

finish_lines:
    test r13d, r13d
    jz write_output
    mov byte ptr [rbp], 10
    inc rbp

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rbp
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero INPUT_MAX
.align 16
output_buffer:
    .zero OUTPUT_MAX
