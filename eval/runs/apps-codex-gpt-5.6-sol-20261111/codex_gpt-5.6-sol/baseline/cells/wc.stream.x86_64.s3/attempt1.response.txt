.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    mov rbp, rax
    lea rsi, [rip + input_buffer]
    xor ebx, ebx

scan_loop:
    movzx eax, byte ptr [rsi + rbx]

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je is_whitespace
    cmp al, 9
    jb is_word_byte
    cmp al, 13
    jbe is_whitespace

is_word_byte:
    test r15, r15
    jne scan_next
    inc r13
    mov r15d, 1
    jmp scan_next

is_whitespace:
    xor r15d, r15d

scan_next:
    inc rbx
    cmp rbx, rbp
    jb scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    test rax, rax
    jne decimal_convert
    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_convert:
    xor ecx, ecx
    mov ebx, 10

decimal_divide:
    xor edx, edx
    div rbx
    add edx, 48
    push rdx
    inc rcx
    test rax, rax
    jne decimal_divide

decimal_emit:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec rcx
    jne decimal_emit
    ret

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
