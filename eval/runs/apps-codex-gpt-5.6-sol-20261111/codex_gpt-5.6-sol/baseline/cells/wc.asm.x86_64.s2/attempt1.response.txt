.globl _start

.text
_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz input_done

    add r10, rax
    mov rbx, rax
    lea rsi, [rip + input_buffer]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_whitespace
    inc r8

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb non_whitespace
    cmp al, 13
    jbe whitespace

non_whitespace:
    test r12, r12
    jne scan_next
    inc r9
    mov r12d, 1
    jmp scan_next

whitespace:
    xor r12d, r12d

scan_next:
    dec rbx
    jne scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]
    mov r14, rdi

    mov rax, r8
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r9
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r10
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, r14
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    lea rsi, [rip + decimal_buffer + 32]
    xor ecx, ecx
    test rax, rax
    jne decimal_loop

    dec rsi
    mov byte ptr [rsi], 48
    mov ecx, 1
    jmp decimal_copy

decimal_loop:
    xor edx, edx
    mov ebx, 10
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc rcx
    test rax, rax
    jne decimal_loop

decimal_copy:
    rep movsb
    ret

.bss
.align 16
input_buffer:
    .zero 65536
decimal_buffer:
    .zero 32
output_buffer:
    .zero 64
