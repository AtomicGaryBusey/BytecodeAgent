.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x3, #0
    mov x6, x20
    mov x21, x20

find_delimiter:
    cmp x3, x20
    b.hs delimiter_done
    ldrb w4, [x19, x3]
    cmp w4, #0x21
    b.eq delimiter_found
    add x3, x3, #1
    b find_delimiter

delimiter_found:
    mov x6, x3
    add x21, x3, #1

delimiter_done:
    adrp x18, program_buffer
    add x18, x18, :lo12:program_buffer
    adrp x24, match_table
    add x24, x24, :lo12:match_table
    adrp x23, bracket_stack
    add x23, x23, :lo12:bracket_stack

    mov x3, #0
    mov x4, #0
    mov x7, #0

compile_loop:
    cmp x3, x6
    b.hs compile_done
    ldrb w5, [x19, x3]
    add x3, x3, #1

    cmp w5, #0x3e
    b.eq compile_simple
    cmp w5, #0x3c
    b.eq compile_simple
    cmp w5, #0x2b
    b.eq compile_simple
    cmp w5, #0x2d
    b.eq compile_simple
    cmp w5, #0x2e
    b.eq compile_simple
    cmp w5, #0x2c
    b.eq compile_simple
    cmp w5, #0x5b
    b.eq compile_open
    cmp w5, #0x5d
    b.eq compile_close
    b compile_loop

compile_simple:
    strb w5, [x18, x7]
    add x7, x7, #1
    b compile_loop

compile_open:
    strb w5, [x18, x7]
    str w7, [x23, x4, lsl #2]
    add x4, x4, #1
    add x7, x7, #1
    b compile_loop

compile_close:
    cbz x4, bracket_error
    strb w5, [x18, x7]
    sub x4, x4, #1
    ldr w9, [x23, x4, lsl #2]
    str w7, [x24, x9, lsl #2]
    str w9, [x24, x7, lsl #2]
    add x7, x7, #1
    b compile_loop

compile_done:
    cbnz x4, bracket_error
    mov x22, x7

    adrp x25, tape
    add x25, x25, :lo12:tape
    mov x26, x25

    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    mov x27, #0
    mov x28, #0

execute_loop:
    cmp x27, x22
    b.hs execution_done

    ldrb w5, [x18, x27]
    add x27, x27, #1

    cmp w5, #0x3e
    b.eq command_right
    cmp w5, #0x3c
    b.eq command_left
    cmp w5, #0x2b
    b.eq command_increment
    cmp w5, #0x2d
    b.eq command_decrement
    cmp w5, #0x2e
    b.eq command_output
    cmp w5, #0x2c
    b.eq command_input
    cmp w5, #0x5b
    b.eq command_open
    b command_close

command_right:
    add x26, x26, #1
    b execute_loop

command_left:
    sub x26, x26, #1
    b execute_loop

command_increment:
    ldrb w9, [x26]
    add w9, w9, #1
    strb w9, [x26]
    b execute_loop

command_decrement:
    ldrb w9, [x26]
    sub w9, w9, #1
    strb w9, [x26]
    b execute_loop

command_output:
    ldrb w9, [x26]
    strb w9, [x23, x28]
    add x28, x28, #1
    b execute_loop

command_input:
    cmp x21, x20
    b.hs input_exhausted
    ldrb w9, [x19, x21]
    add x21, x21, #1
    strb w9, [x26]
    b execute_loop

input_exhausted:
    strb wzr, [x26]
    b execute_loop

command_open:
    ldrb w9, [x26]
    cbnz w9, execute_loop
    sub x10, x27, #1
    ldr w27, [x24, x10, lsl #2]
    add x27, x27, #1
    b execute_loop

command_close:
    ldrb w9, [x26]
    cbz w9, execute_loop
    sub x10, x27, #1
    ldr w27, [x24, x10, lsl #2]
    add x27, x27, #1
    b execute_loop

execution_done:
    cbz x28, exit_success
    mov x0, #1
    mov x1, x23
    mov x2, x28
    mov x8, #SYS_write
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
input_buffer:
    .zero 65536
program_buffer:
    .zero 65536
output_buffer:
    .zero 65536
tape:
    .zero 30000
.balign 16
match_table:
    .zero 262144
bracket_stack:
    .zero 4000
