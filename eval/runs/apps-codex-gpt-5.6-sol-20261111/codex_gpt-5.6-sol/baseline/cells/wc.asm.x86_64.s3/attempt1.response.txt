.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz input_done
    js read_loop

    add r14, rax
    lea rsi, [rip + input_buffer]
    mov rcx, rax

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je is_whitespace
    cmp al, 9
    jb is_word_byte
    cmp al, 13
    jbe is_whitespace

is_word_byte:
    test r15d, r15d
    jnz scan_next
    inc r13
    mov r15d, 1
    jmp scan_next

is_whitespace:
    xor r15d, r15d

scan_next:
    dec rcx
    jnz scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]
    mov r8, rdi

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, r8
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    test rax, rax
    jnz decimal_loop_start
    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_loop_start:
    xor ecx, ecx
    mov ebx, 10

decimal_loop:
    xor edx, edx
    div rbx
    add edx, 48
    push rdx
    inc rcx
    test rax, rax
    jnz decimal_loop

decimal_copy:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec rcx
    jnz decimal_copy
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
