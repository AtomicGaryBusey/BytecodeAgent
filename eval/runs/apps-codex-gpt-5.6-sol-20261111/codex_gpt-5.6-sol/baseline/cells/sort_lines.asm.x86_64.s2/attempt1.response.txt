.equ INPUT_MAX, 32768
.equ LINE_MAX, 1000
.equ DESC_SIZE, 16
.equ OUTPUT_MAX, 33768

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_MAX
    sub rdx, r12
    jnz read_call
    mov edx, 1

read_call:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + input_buffer]
    lea rbp, [rip + line_table]

parse_loop:
    cmp r13, r12
    jae parse_remainder
    cmp byte ptr [rbx + r13], 10
    jne parse_next

    mov rax, r15
    shl rax, 4
    lea rdx, [rbp + rax]
    lea rcx, [rbx + r14]
    mov qword ptr [rdx], rcx
    mov rcx, r13
    sub rcx, r14
    mov qword ptr [rdx + 8], rcx
    inc r15
    lea r14, [r13 + 1]

parse_next:
    inc r13
    jmp parse_loop

parse_remainder:
    cmp r14, r12
    jae sort_begin

    mov rax, r15
    shl rax, 4
    lea rdx, [rbp + rax]
    lea rcx, [rbx + r14]
    mov qword ptr [rdx], rcx
    mov rcx, r12
    sub rcx, r14
    mov qword ptr [rdx + 8], rcx
    inc r15

sort_begin:
    mov r13d, 1

sort_outer:
    cmp r13, r15
    jae output_build

    mov rax, r13
    shl rax, 4
    lea r10, [rbp + rax]
    mov r14, qword ptr [r10]
    mov r12, qword ptr [r10 + 8]
    mov rbx, r13

sort_inner:
    test rbx, rbx
    jz sort_place

    mov rax, rbx
    dec rax
    shl rax, 4
    lea r10, [rbp + rax]

    mov rdi, qword ptr [r10]
    mov rsi, qword ptr [r10 + 8]
    mov rdx, r14
    mov rcx, r12
    call compare_lines
    test eax, eax
    jle sort_place

    mov rax, rbx
    shl rax, 4
    lea r11, [rbp + rax]
    mov rax, qword ptr [r10]
    mov qword ptr [r11], rax
    mov rax, qword ptr [r10 + 8]
    mov qword ptr [r11 + 8], rax
    dec rbx
    jmp sort_inner

sort_place:
    mov rax, rbx
    shl rax, 4
    lea r10, [rbp + rax]
    mov qword ptr [r10], r14
    mov qword ptr [r10 + 8], r12
    inc r13
    jmp sort_outer

output_build:
    xor ebx, ebx
    xor r13d, r13d
    lea r8, [rip + output_buffer]

output_loop:
    cmp r13, r15
    jae output_write

    mov rax, r13
    shl rax, 4
    lea r10, [rbp + rax]
    mov rsi, qword ptr [r10]
    mov rdx, qword ptr [r10 + 8]
    lea rdi, [r8 + rbx]
    mov rcx, rdx
    rep movsb
    add rbx, rdx
    mov byte ptr [r8 + rbx], 10
    inc rbx
    inc r13
    jmp output_loop

output_write:
    test rbx, rbx
    jz program_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    syscall

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    xor r8d, r8d
    mov r9, rsi
    cmp r9, rcx
    cmova r9, rcx

compare_loop:
    cmp r8, r9
    jae compare_prefix
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp eax, r10d
    jne compare_difference
    inc r8
    jmp compare_loop

compare_difference:
    sub eax, r10d
    ret

compare_prefix:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

.bss
.align 16
input_buffer:
    .zero 32769
.align 16
line_table:
    .zero 16000
.align 16
output_buffer:
    .zero 33768
