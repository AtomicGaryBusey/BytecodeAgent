.text
.globl _start
_start:
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    mov x19, #0
    mov x24, #65536

read_loop:
    sub x2, x24, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    cmp x19, x24
    b.lo read_loop

    mov x0, #0
    mov x1, x20
    mov x2, #0
    mov x8, #63
    svc #0

read_done:
    adrp x22, line_buffer
    add x22, x22, :lo12:line_buffer
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits
    mov x21, #0

row_loop:
    cmp x21, x19
    b.hs final_offset

    sub x26, x19, x21
    mov x27, #16
    cmp x26, x27
    csel x26, x26, x27, lo

    mov x23, x22
    mov w27, #28

offset_digits:
    lsr w28, w21, w27
    and w28, w28, #15
    ldrb w28, [x25, x28]
    strb w28, [x23], #1
    subs w27, w27, #4
    b.ge offset_digits

    mov w28, #32
    strb w28, [x23], #1
    strb w28, [x23], #1

    mov x27, #0

hex_slots:
    cmp x27, x26
    b.hs blank_slot

    add x28, x21, x27
    ldrb w9, [x20, x28]
    lsr w10, w9, #4
    and w10, w10, #15
    ldrb w10, [x25, x10]
    strb w10, [x23], #1
    and w10, w9, #15
    ldrb w10, [x25, x10]
    strb w10, [x23], #1
    mov w10, #32
    strb w10, [x23], #1
    b slot_done

blank_slot:
    mov w10, #32
    strb w10, [x23], #1
    strb w10, [x23], #1
    strb w10, [x23], #1

slot_done:
    cmp x27, #7
    b.ne no_middle_space
    mov w10, #32
    strb w10, [x23], #1

no_middle_space:
    add x27, x27, #1
    cmp x27, #16
    b.lo hex_slots

    mov w10, #32
    strb w10, [x23], #1
    mov w10, #124
    strb w10, [x23], #1

    mov x27, #0

ascii_loop:
    cmp x27, x26
    b.hs ascii_done
    add x28, x21, x27
    ldrb w9, [x20, x28]
    cmp w9, #32
    b.lo nonprintable
    cmp w9, #126
    b.hi nonprintable
    b store_ascii

nonprintable:
    mov w9, #46

store_ascii:
    strb w9, [x23], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w9, #124
    strb w9, [x23], #1
    mov w9, #10
    strb w9, [x23], #1

    mov x0, #1
    mov x1, x22
    sub x2, x23, x22
    mov x8, #64
    svc #0

    add x21, x21, x26
    b row_loop

final_offset:
    mov x23, x22
    mov w27, #28

final_digits:
    lsr w28, w19, w27
    and w28, w28, #15
    ldrb w28, [x25, x28]
    strb w28, [x23], #1
    subs w27, w27, #4
    b.ge final_digits

    mov w9, #10
    strb w9, [x23], #1

    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 65536
.balign 16
line_buffer:
    .zero 128
