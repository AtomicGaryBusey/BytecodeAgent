.equ BUF_SIZE, 65536

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    cmp r12, BUF_SIZE
    je read_done

    lea rsi, [rip + buffer]
    add rsi, r12
    mov edx, BUF_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall

    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r8, [rip + buffer]
    lea r13, [r8 + r12]
    mov r9, r8

scan_loop:
    cmp r9, r13
    je final_line
    cmp byte ptr [r9], 10
    je reverse_terminated
    inc r9
    jmp scan_loop

reverse_terminated:
    mov r10, r8
    lea r11, [r9 - 1]

reverse_terminated_loop:
    cmp r10, r11
    jae write_terminated
    mov al, byte ptr [r10]
    mov dl, byte ptr [r11]
    mov byte ptr [r10], dl
    mov byte ptr [r11], al
    inc r10
    dec r11
    jmp reverse_terminated_loop

write_terminated:
    mov rsi, r8
    lea rdx, [r9 + 1]
    sub rdx, r8
    mov edi, 1
    mov eax, 1
    syscall

    lea r8, [r9 + 1]
    mov r9, r8
    jmp scan_loop

final_line:
    cmp r8, r13
    je exit_ok

    mov r10, r8
    lea r11, [r13 - 1]

reverse_final_loop:
    cmp r10, r11
    jae write_final
    mov al, byte ptr [r10]
    mov dl, byte ptr [r11]
    mov byte ptr [r10], dl
    mov byte ptr [r11], al
    inc r10
    dec r11
    jmp reverse_final_loop

write_final:
    mov rsi, r8
    mov rdx, r13
    sub rdx, r8
    mov edi, 1
    mov eax, 1
    syscall

    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
buffer:
    .zero BUF_SIZE
