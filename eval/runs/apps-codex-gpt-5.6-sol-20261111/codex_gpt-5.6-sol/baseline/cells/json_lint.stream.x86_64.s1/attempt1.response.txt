.section .rodata
lit_true:
.ascii "true"
lit_false:
.ascii "false"
lit_null:
.ascii "null"
msg_ok:
.ascii "ok\n"
msg_ok_len = . - msg_ok
msg_error:
.ascii "error at "

.bss
.balign 16
buffer:
.space 65536
outbuf:
.space 64

.text
.globl _start
_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rsi, rbx
    lea r15, [rbx + r12]
    call skip_ws
    cmp rsi, r15
    je parse_error
    call parse_value
    call skip_ws
    cmp rsi, r15
    jne parse_error

parse_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

parse_error:
    mov rax, rsi
    sub rax, rbx
    mov r13, rax

    lea rdi, [rip + outbuf]
    lea rsi, [rip + msg_error]
    mov ecx, 9
copy_error_prefix:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jne copy_error_prefix

    mov rax, r13
    test rax, rax
    jne convert_offset
    mov byte ptr [rdi], '0'
    inc rdi
    jmp append_newline

convert_offset:
    xor ecx, ecx
    mov r8d, 10
convert_loop:
    xor edx, edx
    div r8
    add edx, '0'
    push rdx
    inc ecx
    test rax, rax
    jne convert_loop

emit_digits:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jne emit_digits

append_newline:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
skip_ws_loop:
    cmp rsi, r15
    je skip_ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_advance:
    inc rsi
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe parse_number
    jmp parse_error

parse_true:
    lea rdi, [rip + lit_true]
    mov ecx, 4
    jmp match_literal

parse_false:
    lea rdi, [rip + lit_false]
    mov ecx, 5
    jmp match_literal

parse_null:
    lea rdi, [rip + lit_null]
    mov ecx, 4

match_literal:
match_literal_loop:
    test ecx, ecx
    jz match_literal_done
    cmp rsi, r15
    je parse_error
    mov al, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jne parse_error
    inc rsi
    inc rdi
    dec ecx
    jmp match_literal_loop
match_literal_done:
    ret

parse_string:
    inc rsi
string_loop:
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je string_done
    cmp al, 0x20
    jb parse_error
    cmp al, '\\'
    je string_escape
    inc rsi
    jmp string_loop

string_escape:
    inc rsi
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je simple_escape
    cmp al, '\\'
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    je unicode_escape
    jmp parse_error

simple_escape:
    inc rsi
    jmp string_loop

unicode_escape:
    inc rsi
    mov ecx, 4
unicode_loop:
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb unicode_error
    cmp al, '9'
    jbe unicode_digit
    cmp al, 'A'
    jb unicode_error
    cmp al, 'F'
    jbe unicode_digit
    cmp al, 'a'
    jb unicode_error
    cmp al, 'f'
    ja unicode_error
unicode_digit:
    inc rsi
    dec ecx
    jne unicode_loop
    jmp string_loop
unicode_error:
    jmp parse_error

string_done:
    inc rsi
    ret

parse_number:
    mov al, byte ptr [rsi]
    cmp al, '-'
    jne number_integer
    inc rsi
    cmp rsi, r15
    je parse_error

number_integer:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error

number_int_digits:
    inc rsi
    cmp rsi, r15
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_after_int
    cmp al, '9'
    jbe number_int_digits
    jmp number_after_int

number_zero:
    inc rsi

number_after_int:
    cmp rsi, r15
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    je number_fraction
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
number_done:
    ret

number_fraction:
    inc rsi
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

number_frac_digits:
    inc rsi
    cmp rsi, r15
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_after_fraction
    cmp al, '9'
    jbe number_frac_digits

number_after_fraction:
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    ret

number_exponent:
    inc rsi
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    je number_exp_sign
    jmp number_exp_first

number_exp_sign:
    inc rsi
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]

number_exp_first:
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

number_exp_digits:
    inc rsi
    cmp rsi, r15
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe number_exp_digits
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r15
    je parse_error
    cmp byte ptr [rsi], ']'
    je array_empty_done

array_value:
    call parse_value
    call skip_ws
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne parse_error
    inc rsi
    call skip_ws
    jmp array_value

array_empty_done:
array_done:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r15
    je parse_error
    cmp byte ptr [rsi], '}'
    je object_empty_done

object_member:
    cmp rsi, r15
    je parse_error
    cmp byte ptr [rsi], '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp rsi, r15
    je parse_error
    cmp byte ptr [rsi], ':'
    jne parse_error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r15
    je parse_error
    movzx eax, byte ptr [rsi]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne parse_error
    inc rsi
    call skip_ws
    jmp object_member

object_empty_done:
object_done:
    inc rsi
    ret
