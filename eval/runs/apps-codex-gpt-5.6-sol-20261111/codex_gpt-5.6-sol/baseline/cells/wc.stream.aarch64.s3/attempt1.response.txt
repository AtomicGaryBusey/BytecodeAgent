.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 4096
.equ DECIMAL_DIVISOR, 10000

.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

1:
    mov x0, #0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    mov x2, #INPUT_SIZE
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le 5f

    add x21, x21, x0
    mov x3, x1
    add x4, x1, x0

2:
    ldrb w5, [x3], #1

    cmp w5, #10
    b.ne 3f
    add x19, x19, #1

3:
    cmp w5, #32
    b.eq 4f
    sub w6, w5, #9
    cmp w6, #4
    b.ls 4f

    cbnz x22, 6f
    add x20, x20, #1
    mov x22, #1
    b 6f

4:
    mov x22, #0

6:
    cmp x3, x4
    b.lo 2b
    b 1b

5:
    adrp x25, output_buffer
    add x25, x25, :lo12:output_buffer
    mov x1, x25

    mov x0, x19
    bl append_decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl append_decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl append_decimal
    mov w2, #10
    strb w2, [x1], #1

    sub x2, x1, x25
    mov x0, #1
    mov x1, x25
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

append_decimal:
    mov x2, #DECIMAL_DIVISOR
    mov w3, #0
    mov x5, #10

7:
    udiv x4, x0, x2
    msub x0, x4, x2, x0
    cbnz x4, 8f
    cbnz w3, 8f
    cmp x2, #1
    b.ne 9f

8:
    add w6, w4, #48
    strb w6, [x1], #1
    mov w3, #1

9:
    cmp x2, #1
    b.eq 10f
    udiv x2, x2, x5
    b 7b

10:
    ret

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
.balign 16
output_buffer:
    .zero 64
