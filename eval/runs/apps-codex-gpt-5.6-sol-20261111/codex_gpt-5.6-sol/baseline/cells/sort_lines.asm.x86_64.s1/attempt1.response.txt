.equ INPUT_MAX, 32768
.equ LINE_MAX, 1000

.globl _start

.text
_start:
    xor ebx, ebx
    lea r8, [rip + input_buf]

read_loop:
    cmp ebx, INPUT_MAX
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + rbx]
    mov edx, INPUT_MAX
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r12, rbx
    xor ebp, ebp
    xor edx, edx
    xor ecx, ecx
    lea rbx, [rip + lines]

parse_loop:
    cmp rcx, r12
    jae parse_remainder
    cmp byte ptr [r8 + rcx], 10
    jne parse_next

    mov rax, rbp
    shl rax, 4
    lea rsi, [r8 + rdx]
    mov qword ptr [rbx + rax], rsi
    mov rdi, rcx
    sub rdi, rdx
    mov qword ptr [rbx + rax + 8], rdi
    inc rbp
    lea rdx, [rcx + 1]

parse_next:
    inc rcx
    jmp parse_loop

parse_remainder:
    cmp rdx, r12
    jae sort_begin
    mov rax, rbp
    shl rax, 4
    lea rsi, [r8 + rdx]
    mov qword ptr [rbx + rax], rsi
    mov rdi, r12
    sub rdi, rdx
    mov qword ptr [rbx + rax + 8], rdi
    inc rbp

sort_begin:
    mov r8d, 1

sort_outer:
    cmp r8, rbp
    jae output_begin

    mov rax, r8
    shl rax, 4
    mov r12, qword ptr [rbx + rax]
    mov r13, qword ptr [rbx + rax + 8]
    mov r9, r8

sort_compare:
    test r9, r9
    jz sort_place

    mov rax, r9
    dec rax
    shl rax, 4
    mov r14, qword ptr [rbx + rax]
    mov r15, qword ptr [rbx + rax + 8]

    mov rcx, r13
    cmp rcx, r15
    cmova rcx, r15
    xor edi, edi

compare_bytes:
    cmp rdi, rcx
    jae compare_prefix
    movzx eax, byte ptr [r12 + rdi]
    movzx edx, byte ptr [r14 + rdi]
    cmp eax, edx
    jb sort_shift
    ja sort_place
    inc rdi
    jmp compare_bytes

compare_prefix:
    cmp r13, r15
    jb sort_shift
    jmp sort_place

sort_shift:
    mov rax, r9
    shl rax, 4
    mov qword ptr [rbx + rax], r14
    mov qword ptr [rbx + rax + 8], r15
    dec r9
    jmp sort_compare

sort_place:
    mov rax, r9
    shl rax, 4
    mov qword ptr [rbx + rax], r12
    mov qword ptr [rbx + rax + 8], r13
    inc r8
    jmp sort_outer

output_begin:
    xor ebx, ebx
    lea r12, [rip + lines]

output_loop:
    cmp rbx, rbp
    jae exit_success

    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rdx, qword ptr [r12 + rax + 8]
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    inc rbx
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buf:
    .zero 32768
.align 16
lines:
    .zero 16000
