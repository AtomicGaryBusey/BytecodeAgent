.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.section .text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + lines]
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d

parse_loop:
    cmp rcx, r12
    jae parse_remainder
    cmp byte ptr [rsi + rcx], 10
    jne parse_next

    mov r9, r8
    shl r9, 4
    lea rax, [rsi + rdx]
    mov qword ptr [rdi + r9], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [rdi + r9 + 8], rax
    inc r8
    lea rdx, [rcx + 1]

parse_next:
    inc rcx
    jmp parse_loop

parse_remainder:
    cmp rdx, r12
    jae parse_done
    mov r9, r8
    shl r9, 4
    lea rax, [rsi + rdx]
    mov qword ptr [rdi + r9], rax
    mov rax, r12
    sub rax, rdx
    mov qword ptr [rdi + r9 + 8], rax
    inc r8

parse_done:
    mov qword ptr [rip + line_count], r8
    cmp r8, 1
    jbe output_lines

    mov r13, 1

merge_pass:
    mov rax, qword ptr [rip + line_count]
    cmp r13, rax
    jae output_lines
    xor r12d, r12d

merge_run:
    mov rax, qword ptr [rip + line_count]
    cmp r12, rax
    jae copy_back

    lea r14, [r12 + r13]
    cmp r14, rax
    cmova r14, rax

    lea r15, [r14 + r13]
    cmp r15, rax
    cmova r15, rax

    mov rbx, r12
    mov rbp, r14
    mov r10, r12

merge_items:
    cmp rbx, r14
    jae take_right
    cmp rbp, r15
    jae take_left

    lea rax, [rip + lines]
    mov r9, rbx
    shl r9, 4
    mov rdi, qword ptr [rax + r9]
    mov rsi, qword ptr [rax + r9 + 8]

    mov r9, rbp
    shl r9, 4
    mov rdx, qword ptr [rax + r9]
    mov rcx, qword ptr [rax + r9 + 8]
    call compare_le
    test eax, eax
    jnz take_left

take_right:
    cmp rbp, r15
    jae run_done
    lea rax, [rip + lines]
    mov r9, rbp
    shl r9, 4
    mov rdi, qword ptr [rax + r9]
    mov rsi, qword ptr [rax + r9 + 8]
    inc rbp
    jmp store_item

take_left:
    cmp rbx, r14
    jae run_done
    lea rax, [rip + lines]
    mov r9, rbx
    shl r9, 4
    mov rdi, qword ptr [rax + r9]
    mov rsi, qword ptr [rax + r9 + 8]
    inc rbx

store_item:
    lea rax, [rip + temp_lines]
    mov r9, r10
    shl r9, 4
    mov qword ptr [rax + r9], rdi
    mov qword ptr [rax + r9 + 8], rsi
    inc r10
    jmp merge_items

run_done:
    mov r12, r15
    jmp merge_run

copy_back:
    lea rsi, [rip + temp_lines]
    lea rdi, [rip + lines]
    mov rcx, qword ptr [rip + line_count]
    shl rcx, 1
    rep movsq
    shl r13, 1
    jmp merge_pass

compare_le:
    mov rax, rsi
    cmp rax, rcx
    cmova rax, rcx
    xor r8d, r8d

compare_loop:
    cmp r8, rax
    jae compare_prefix
    movzx r9d, byte ptr [rdi + r8]
    movzx r11d, byte ptr [rdx + r8]
    cmp r9d, r11d
    jb compare_true
    ja compare_false
    inc r8
    jmp compare_loop

compare_prefix:
    xor eax, eax
    cmp rsi, rcx
    setbe al
    ret

compare_true:
    mov eax, 1
    ret

compare_false:
    xor eax, eax
    ret

output_lines:
    xor ebx, ebx

output_loop:
    cmp rbx, qword ptr [rip + line_count]
    jae exit_success

    lea rax, [rip + lines]
    mov rcx, rbx
    shl rcx, 4
    mov r13, qword ptr [rax + rcx]
    mov r14, qword ptr [rax + rcx + 8]

    test r14, r14
    jz output_newline
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r14
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
lines:
    .zero 16000
.align 16
temp_lines:
    .zero 16000
.align 8
line_count:
    .zero 8
