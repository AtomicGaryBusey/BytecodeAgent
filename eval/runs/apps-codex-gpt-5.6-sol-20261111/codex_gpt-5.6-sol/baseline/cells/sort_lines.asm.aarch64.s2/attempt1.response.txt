.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_CAP, 32769

.text
.globl _start
_start:
    adrp x18, input_buffer
    add x18, x18, :lo12:input_buffer
    mov x19, #0

read_loop:
    mov x3, #INPUT_CAP
    sub x2, x3, x19
    add x1, x18, x19
    mov x0, #0
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x20, line_table
    add x20, x20, :lo12:line_table
    mov x21, #0
    mov x22, x18
    mov x23, x18
    add x24, x18, x19

parse_loop:
    cmp x23, x24
    b.eq parse_end
    ldrb w0, [x23]
    cmp w0, #10
    b.eq parse_newline
    add x23, x23, #1
    b parse_loop

parse_newline:
    sub x1, x23, x22
    add x2, x20, x21, lsl #4
    stp x22, x1, [x2]
    add x21, x21, #1
    add x23, x23, #1
    mov x22, x23
    b parse_loop

parse_end:
    cmp x22, x24
    b.eq sort_begin
    sub x1, x24, x22
    add x2, x20, x21, lsl #4
    stp x22, x1, [x2]
    add x21, x21, #1

sort_begin:
    mov x10, #1

sort_outer:
    cmp x10, x21
    b.hs output_begin
    add x9, x20, x10, lsl #4
    ldp x11, x12, [x9]
    mov x13, x10

sort_inner:
    cbz x13, sort_insert
    sub x14, x13, #1
    add x9, x20, x14, lsl #4
    ldp x15, x16, [x9]
    mov x17, #0

compare_loop:
    cmp x17, x16
    b.eq previous_ended
    cmp x17, x12
    b.eq shift_previous
    ldrb w4, [x15, x17]
    ldrb w5, [x11, x17]
    cmp w4, w5
    b.hi shift_previous
    b.lo sort_insert
    add x17, x17, #1
    b compare_loop

previous_ended:
    b sort_insert

shift_previous:
    add x9, x20, x13, lsl #4
    stp x15, x16, [x9]
    sub x13, x13, #1
    b sort_inner

sort_insert:
    add x9, x20, x13, lsl #4
    stp x11, x12, [x9]
    add x10, x10, #1
    b sort_outer

output_begin:
    mov x10, #0

output_loop:
    cmp x10, x21
    b.hs exit_success
    add x9, x20, x10, lsl #4
    ldp x11, x12, [x9]
    cbz x12, output_newline

    mov x0, #1
    mov x1, x11
    mov x2, x12
    mov x8, #SYS_WRITE
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0

    add x10, x10, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero 32769
.balign 16
line_table:
    .zero 16000
