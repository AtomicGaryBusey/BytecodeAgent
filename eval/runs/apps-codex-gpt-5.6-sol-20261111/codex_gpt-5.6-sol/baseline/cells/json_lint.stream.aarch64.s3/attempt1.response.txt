.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.gt read_more
    b parse_start

read_more:
    add x20, x20, x0
    b read_loop

parse_start:
    mov x21, #0
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne error_out

ok_out:
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error_out:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #9
    mov x8, #64
    svc #0

    adrp x9, output_buf
    add x9, x9, :lo12:output_buf
    add x10, x9, #31
    mov w11, #10
    strb w11, [x10]
    mov x12, x10
    mov x13, x21
    cbnz x13, decimal_loop
    mov w14, #'0'
    sub x12, x12, #1
    strb w14, [x12]
    b decimal_done

decimal_loop:
    mov x15, #10
    udiv x16, x13, x15
    msub x17, x16, x15, x13
    add w17, w17, #'0'
    sub x12, x12, #1
    strb w17, [x12]
    mov x13, x16
    cbnz x13, decimal_loop

decimal_done:
    mov x0, #1
    mov x1, x12
    add x2, x10, #1
    sub x2, x2, x12
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x21, x20
    b.eq skip_ws_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq skip_ws_advance
    cmp w0, #0x09
    b.eq skip_ws_advance
    cmp w0, #0x0a
    b.eq skip_ws_advance
    cmp w0, #0x0d
    b.eq skip_ws_advance
skip_ws_done:
    ret
skip_ws_advance:
    add x21, x21, #1
    b skip_ws

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq value_string
    cmp w0, #'{'
    b.eq value_object
    cmp w0, #'['
    b.eq value_array
    cmp w0, #'t'
    b.eq value_true
    cmp w0, #'f'
    b.eq value_false
    cmp w0, #'n'
    b.eq value_null
    cmp w0, #'-'
    b.eq value_number
    cmp w0, #'0'
    b.lo error_out
    cmp w0, #'9'
    b.hi error_out
value_number:
    bl parse_number
    b value_done
value_string:
    bl parse_string
    b value_done
value_object:
    bl parse_object
    b value_done
value_array:
    bl parse_array
    b value_done
value_true:
    adrp x0, true_text
    add x0, x0, :lo12:true_text
    mov x1, #4
    bl parse_literal
    b value_done
value_false:
    adrp x0, false_text
    add x0, x0, :lo12:false_text
    mov x1, #5
    bl parse_literal
    b value_done
value_null:
    adrp x0, null_text
    add x0, x0, :lo12:null_text
    mov x1, #4
    bl parse_literal
value_done:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    mov x2, #0
literal_loop:
    cmp x2, x1
    b.eq literal_done
    cmp x21, x20
    b.eq error_out
    ldrb w3, [x19, x21]
    ldrb w4, [x0, x2]
    cmp w3, w4
    b.ne error_out
    add x21, x21, #1
    add x2, x2, #1
    b literal_loop
literal_done:
    ret

parse_string:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq string_close
    cmp w0, #0x20
    b.lo error_out
    cmp w0, #0x5c
    b.eq string_escape
    add x21, x21, #1
    b string_loop

string_close:
    add x21, x21, #1
    ret

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq simple_escape
    cmp w0, #0x5c
    b.eq simple_escape
    cmp w0, #'/'
    b.eq simple_escape
    cmp w0, #'b'
    b.eq simple_escape
    cmp w0, #'f'
    b.eq simple_escape
    cmp w0, #'n'
    b.eq simple_escape
    cmp w0, #'r'
    b.eq simple_escape
    cmp w0, #'t'
    b.eq simple_escape
    cmp w0, #'u'
    b.ne error_out
    add x21, x21, #1
    mov x1, #4
unicode_loop:
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo unicode_alpha
    cmp w0, #'9'
    b.ls unicode_good
unicode_alpha:
    cmp w0, #'a'
    b.lo unicode_upper
    cmp w0, #'f'
    b.ls unicode_good
unicode_upper:
    cmp w0, #'A'
    b.lo error_out
    cmp w0, #'F'
    b.hi error_out
unicode_good:
    add x21, x21, #1
    subs x1, x1, #1
    b.ne unicode_loop
    b string_loop

simple_escape:
    add x21, x21, #1
    b string_loop

parse_number:
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'-'
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.eq error_out

number_integer:
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.eq number_zero
    cmp w0, #'1'
    b.lo error_out
    cmp w0, #'9'
    b.hi error_out
number_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo number_after_int
    cmp w0, #'9'
    b.ls number_digits
    b number_after_int

number_zero:
    add x21, x21, #1

number_after_int:
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #'.'
    b.ne number_check_exp
    add x21, x21, #1
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo error_out
    cmp w0, #'9'
    b.hi error_out
number_frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo number_check_exp
    cmp w0, #'9'
    b.ls number_frac_digits

number_check_exp:
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.eq number_exp
    cmp w0, #'E'
    b.ne number_done
number_exp:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq number_exp_sign
    cmp w0, #'-'
    b.ne number_exp_digit
number_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
number_exp_digit:
    cmp w0, #'0'
    b.lo error_out
    cmp w0, #'9'
    b.hi error_out
number_exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo number_done
    cmp w0, #'9'
    b.ls number_exp_digits
number_done:
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq array_close

array_value:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq array_close
    cmp w0, #','
    b.ne error_out
    add x21, x21, #1
    bl skip_ws
    b array_value

array_close:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq object_close

object_member:
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.ne error_out
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #':'
    b.ne error_out
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.eq error_out
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq object_close
    cmp w0, #','
    b.ne error_out
    add x21, x21, #1
    bl skip_ws
    b object_member

object_close:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 32
