.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 324000

.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
.space INPUT_SIZE
.align 16
output_buffer:
.space OUTPUT_SIZE

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    cmp r12d, INPUT_SIZE
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbp, [rip + output_buffer]
    mov rdi, rbp
    lea r10, [rip + hex_digits]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_count_ready
    mov r14d, 16

row_count_ready:
    mov eax, r13d
    call emit_hex8

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor r15d, r15d

hex_slot_loop:
    cmp r15, r14
    jae empty_hex_slot

    lea r11, [r13 + r15]
    movzx edx, byte ptr [rbx + r11]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    jmp hex_slot_done

empty_hex_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3

hex_slot_done:
    cmp r15d, 7
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi

no_middle_space:
    inc r15
    cmp r15d, 16
    jb hex_slot_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor r15d, r15d

ascii_loop:
    cmp r15, r14
    jae ascii_done
    lea r11, [r13 + r15]
    movzx eax, byte ptr [rbx + r11]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_stored

ascii_dot:
    mov byte ptr [rdi], '.'

ascii_stored:
    inc rdi
    inc r15
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call emit_hex8
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, rbp
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov r8d, eax
    mov ecx, 8

emit_hex8_loop:
    rol r8d, 4
    mov edx, r8d
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jne emit_hex8_loop
    ret
