.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x3, #1, lsl #16

read_loop:
    cmp x20, x3
    b.eq dump_start
    add x1, x19, x20
    sub x2, x3, x20
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le dump_start
    add x20, x20, x0
    b read_loop

dump_start:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    adrp x22, hex_digits
    add x22, x22, :lo12:hex_digits
    mov x23, #0

row_loop:
    cmp x23, x20
    b.hs final_offset

    sub x24, x20, x23
    cmp x24, #16
    b.ls row_size_ready
    mov x24, #16

row_size_ready:
    mov x25, x21
    mov w26, #28

row_offset_loop:
    lsrv w27, w23, w26
    and w27, w27, #15
    ldrb w27, [x22, w27, uxtw]
    strb w27, [x25], #1
    subs w26, w26, #4
    b.ge row_offset_loop

    mov w27, #' '
    strb w27, [x25], #1
    strb w27, [x25], #1
    mov x26, #0

hex_slot_loop:
    cmp x26, x24
    b.hs empty_hex_slot

    add x27, x23, x26
    ldrb w28, [x19, x27]
    lsr w29, w28, #4
    ldrb w29, [x22, w29, uxtw]
    strb w29, [x25], #1
    and w29, w28, #15
    ldrb w29, [x22, w29, uxtw]
    strb w29, [x25], #1
    mov w29, #' '
    strb w29, [x25], #1
    b hex_slot_done

empty_hex_slot:
    mov w29, #' '
    strb w29, [x25], #1
    strb w29, [x25], #1
    strb w29, [x25], #1

hex_slot_done:
    cmp x26, #7
    b.ne no_middle_space
    mov w29, #' '
    strb w29, [x25], #1

no_middle_space:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_slot_loop

    mov w27, #' '
    strb w27, [x25], #1
    mov w27, #'|'
    strb w27, [x25], #1
    mov x26, #0

ascii_loop:
    cmp x26, x24
    b.hs ascii_done
    add x27, x23, x26
    ldrb w28, [x19, x27]
    cmp w28, #0x20
    b.lo ascii_dot
    cmp w28, #0x7e
    b.hi ascii_dot
    strb w28, [x25], #1
    b ascii_next

ascii_dot:
    mov w28, #'.'
    strb w28, [x25], #1

ascii_next:
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w27, #'|'
    strb w27, [x25], #1
    mov w27, #'\n'
    strb w27, [x25], #1

    mov x0, #1
    mov x1, x21
    sub x2, x25, x21
    mov x8, #SYS_write
    svc #0

    add x23, x23, x24
    b row_loop

final_offset:
    mov x25, x21
    mov w26, #28

final_offset_loop:
    lsrv w27, w20, w26
    and w27, w27, #15
    ldrb w27, [x22, w27, uxtw]
    strb w27, [x25], #1
    subs w26, w26, #4
    b.ge final_offset_loop

    mov w27, #'\n'
    strb w27, [x25], #1

    mov x0, #1
    mov x1, x21
    sub x2, x25, x21
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE
.balign 16
output_buf:
    .zero 128
