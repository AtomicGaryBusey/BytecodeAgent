.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ OUTPUT_SIZE, 65536
.equ BRACKET_DEPTH, 1000
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x24, x20
    mov x10, #0

find_delimiter:
    cmp x10, x24
    b.hs no_delimiter
    ldrb w11, [x19, x10]
    cmp w11, #0x21
    b.eq delimiter_found
    add x10, x10, #1
    b find_delimiter

delimiter_found:
    mov x20, x10
    add x23, x10, #1
    b validate_setup

no_delimiter:
    mov x20, x24
    mov x23, x24

validate_setup:
    adrp x27, jump_map
    add x27, x27, :lo12:jump_map
    adrp x28, bracket_stack
    add x28, x28, :lo12:bracket_stack
    mov x9, #0
    mov x10, #0

validate_loop:
    cmp x10, x20
    b.hs validate_end
    ldrb w11, [x19, x10]
    cmp w11, #0x5b
    b.eq validate_open
    cmp w11, #0x5d
    b.eq validate_close
    add x10, x10, #1
    b validate_loop

validate_open:
    cmp x9, #BRACKET_DEPTH
    b.hs invalid_program
    str w10, [x28, x9, lsl #2]
    add x9, x9, #1
    add x10, x10, #1
    b validate_loop

validate_close:
    cbz x9, invalid_program
    sub x9, x9, #1
    ldr w11, [x28, x9, lsl #2]
    str w10, [x27, x11, lsl #2]
    str w11, [x27, x10, lsl #2]
    add x10, x10, #1
    b validate_loop

validate_end:
    cbnz x9, invalid_program

    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    mov x21, #0
    mov x26, #0

execute_loop:
    cmp x21, x20
    b.hs execution_done
    mov x10, x21
    ldrb w11, [x19, x21]
    add x21, x21, #1

    cmp w11, #0x3e
    b.eq command_right
    cmp w11, #0x3c
    b.eq command_left
    cmp w11, #0x2b
    b.eq command_add
    cmp w11, #0x2d
    b.eq command_sub
    cmp w11, #0x2e
    b.eq command_output
    cmp w11, #0x2c
    b.eq command_input
    cmp w11, #0x5b
    b.eq command_open
    cmp w11, #0x5d
    b.eq command_close
    b execute_loop

command_right:
    add x22, x22, #1
    b execute_loop

command_left:
    sub x22, x22, #1
    b execute_loop

command_add:
    ldrb w12, [x22]
    add w12, w12, #1
    strb w12, [x22]
    b execute_loop

command_sub:
    ldrb w12, [x22]
    sub w12, w12, #1
    strb w12, [x22]
    b execute_loop

command_output:
    ldrb w12, [x22]
    strb w12, [x25, x26]
    add x26, x26, #1
    b execute_loop

command_input:
    cmp x23, x24
    b.hs input_exhausted
    ldrb w12, [x19, x23]
    add x23, x23, #1
    strb w12, [x22]
    b execute_loop

input_exhausted:
    strb wzr, [x22]
    b execute_loop

command_open:
    ldrb w12, [x22]
    cbnz w12, execute_loop
    ldr w21, [x27, x10, lsl #2]
    add x21, x21, #1
    b execute_loop

command_close:
    ldrb w12, [x22]
    cbz w12, execute_loop
    ldr w21, [x27, x10, lsl #2]
    add x21, x21, #1
    b execute_loop

execution_done:
    cbz x26, exit_success
    mov x0, #1
    mov x1, x25
    mov x2, x26
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE
.balign 16
jump_map:
    .zero 262144
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
output_buf:
    .zero OUTPUT_SIZE
.balign 16
bracket_stack:
    .zero 4000
