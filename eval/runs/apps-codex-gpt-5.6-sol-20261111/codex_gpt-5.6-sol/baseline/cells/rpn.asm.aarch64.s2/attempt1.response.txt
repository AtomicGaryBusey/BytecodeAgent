.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x19
    add x22, x19, x20
    mov x23, x19

scan_lines:
    cmp x21, x22
    b.eq final_line
    ldrb w0, [x21]
    cmp w0, #10
    b.eq process_line
    add x21, x21, #1
    b scan_lines

process_line:
    mov x0, x23
    mov x1, x21
    bl eval_line
    cbnz x0, print_error
    mov x0, x1
    bl print_number
    b line_done

print_error:
    bl write_error

line_done:
    add x21, x21, #1
    mov x23, x21
    b scan_lines

final_line:
    cmp x23, x22
    b.eq exit_ok
    mov x0, x23
    mov x1, x22
    bl eval_line
    cbnz x0, final_error
    mov x0, x1
    bl print_number
    b exit_ok

final_error:
    bl write_error

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

eval_line:
    mov x2, x0
    mov x3, x1
    adrp x10, value_stack
    add x10, x10, :lo12:value_stack
    mov x11, #0

eval_skip_separators:
    cmp x2, x3
    b.eq eval_finish
    ldrb w4, [x2]
    cmp w4, #32
    b.eq eval_skip_one
    cmp w4, #9
    b.eq eval_skip_one
    b eval_token_start

eval_skip_one:
    add x2, x2, #1
    b eval_skip_separators

eval_token_start:
    mov x12, x2

eval_find_token_end:
    cmp x2, x3
    b.eq eval_token_ready
    ldrb w4, [x2]
    cmp w4, #32
    b.eq eval_token_ready
    cmp w4, #9
    b.eq eval_token_ready
    add x2, x2, #1
    b eval_find_token_end

eval_token_ready:
    mov x13, x2
    sub x5, x13, x12
    cmp x5, #1
    b.ne eval_literal

    ldrb w16, [x12]
    cmp w16, #43
    b.eq eval_operator
    cmp w16, #45
    b.eq eval_operator
    cmp w16, #42
    b.eq eval_operator
    cmp w16, #47
    b.eq eval_operator
    cmp w16, #37
    b.eq eval_operator

eval_literal:
    mov x9, x12
    mov w17, #0
    ldrb w4, [x9]
    cmp w4, #45
    b.ne eval_literal_digits
    mov w17, #1
    add x9, x9, #1

eval_literal_digits:
    sub x15, x13, x9
    cbz x15, eval_error
    cmp x15, #18
    b.hi eval_error
    mov x14, #0

eval_digit_loop:
    cmp x9, x13
    b.eq eval_literal_done
    ldrb w4, [x9], #1
    sub w4, w4, #48
    cmp w4, #9
    b.hi eval_error
    mov x6, #10
    mul x14, x14, x6
    add x14, x14, x4
    b eval_digit_loop

eval_literal_done:
    cbz w17, eval_push_literal
    neg x14, x14

eval_push_literal:
    str x14, [x10, x11, lsl #3]
    add x11, x11, #1
    b eval_skip_separators

eval_operator:
    cmp x11, #2
    b.lo eval_error
    sub x11, x11, #1
    ldr x14, [x10, x11, lsl #3]
    sub x11, x11, #1
    ldr x15, [x10, x11, lsl #3]

    cmp w16, #43
    b.eq eval_add
    cmp w16, #45
    b.eq eval_sub
    cmp w16, #42
    b.eq eval_mul
    cmp w16, #47
    b.eq eval_div
    b eval_mod

eval_add:
    add x15, x15, x14
    b eval_push_result

eval_sub:
    sub x15, x15, x14
    b eval_push_result

eval_mul:
    mul x15, x15, x14
    b eval_push_result

eval_div:
    cbz x14, eval_error
    movz x6, #0x8000, lsl #48
    cmp x15, x6
    b.ne eval_do_div
    cmn x14, #1
    b.eq eval_error

eval_do_div:
    sdiv x15, x15, x14
    b eval_push_result

eval_mod:
    cbz x14, eval_error
    movz x6, #0x8000, lsl #48
    cmp x15, x6
    b.ne eval_do_mod
    cmn x14, #1
    b.eq eval_error

eval_do_mod:
    sdiv x6, x15, x14
    msub x15, x6, x14, x15

eval_push_result:
    str x15, [x10, x11, lsl #3]
    add x11, x11, #1
    b eval_skip_separators

eval_finish:
    cmp x11, #1
    b.ne eval_error
    ldr x1, [x10]
    mov x0, #0
    ret

eval_error:
    mov x0, #1
    ret

print_number:
    mov x1, x0
    adrp x9, number_buf
    add x9, x9, :lo12:number_buf
    add x10, x9, #31
    mov w11, #10
    strb w11, [x10]
    mov x12, x10
    mov w13, #0

    tbz x1, #63, number_magnitude
    mov w13, #1
    neg x1, x1

number_magnitude:
    cbnz x1, number_digits
    sub x12, x12, #1
    mov w14, #48
    strb w14, [x12]
    b number_sign

number_digits:
    mov x15, #10

number_digit_loop:
    udiv x16, x1, x15
    msub x17, x16, x15, x1
    add w17, w17, #48
    sub x12, x12, #1
    strb w17, [x12]
    mov x1, x16
    cbnz x1, number_digit_loop

number_sign:
    cbz w13, number_write
    sub x12, x12, #1
    mov w14, #45
    strb w14, [x12]

number_write:
    mov x0, #1
    mov x1, x12
    add x2, x10, #1
    sub x2, x2, x12
    mov x8, #64
    svc #0
    ret

write_error:
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    mov x8, #64
    svc #0
    ret
