.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x20, x19, x20
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    mov x26, x23
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack

    mov x21, x19
    mov x25, x19

scan_lines:
    cmp x25, x20
    b.hs scan_finished
    ldrb w0, [x25], #1
    cmp w0, #10
    b.ne scan_lines
    sub x1, x25, #1
    mov x0, x21
    bl process_line
    mov x21, x25
    b scan_lines

scan_finished:
    cmp x21, x20
    b.eq write_output
    mov x0, x21
    mov x1, x20
    bl process_line

write_output:
    sub x27, x23, x26
    mov x28, x26

write_loop:
    cbz x27, exit_ok
    mov x0, #1
    mov x1, x28
    mov x2, x27
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    ble exit_ok
    add x28, x28, x0
    sub x27, x27, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

process_line:
    mov x2, x0
    mov x3, #0

token_loop:
skip_separators:
    cmp x2, x1
    b.hs line_complete
    ldrb w4, [x2]
    cmp w4, #32
    b.eq skip_one
    cmp w4, #9
    b.ne token_start
skip_one:
    add x2, x2, #1
    b skip_separators

token_start:
    mov x4, x2
    mov x5, x2

token_scan:
    cmp x5, x1
    b.hs token_ready
    ldrb w6, [x5]
    cmp w6, #32
    b.eq token_ready
    cmp w6, #9
    b.eq token_ready
    add x5, x5, #1
    b token_scan

token_ready:
    sub x6, x5, x4
    mov x2, x5
    cmp x6, #1
    b.ne parse_literal

    ldrb w7, [x4]
    cmp w7, #43
    b.eq op_add
    cmp w7, #45
    b.eq op_sub
    cmp w7, #42
    b.eq op_mul
    cmp w7, #47
    b.eq op_div
    cmp w7, #37
    b.eq op_mod
    b parse_literal

op_add:
    cmp x3, #2
    b.lo line_error
    sub x3, x3, #1
    ldr x10, [x24, x3, lsl #3]
    sub x11, x3, #1
    ldr x9, [x24, x11, lsl #3]
    add x9, x9, x10
    str x9, [x24, x11, lsl #3]
    b token_loop

op_sub:
    cmp x3, #2
    b.lo line_error
    sub x3, x3, #1
    ldr x10, [x24, x3, lsl #3]
    sub x11, x3, #1
    ldr x9, [x24, x11, lsl #3]
    sub x9, x9, x10
    str x9, [x24, x11, lsl #3]
    b token_loop

op_mul:
    cmp x3, #2
    b.lo line_error
    sub x3, x3, #1
    ldr x10, [x24, x3, lsl #3]
    sub x11, x3, #1
    ldr x9, [x24, x11, lsl #3]
    mul x9, x9, x10
    str x9, [x24, x11, lsl #3]
    b token_loop

op_div:
    cmp x3, #2
    b.lo line_error
    sub x3, x3, #1
    ldr x10, [x24, x3, lsl #3]
    cbz x10, line_error
    sub x11, x3, #1
    ldr x9, [x24, x11, lsl #3]
    movz x12, #0x8000, lsl #48
    cmp x9, x12
    b.ne do_div
    cmn x10, #1
    b.eq line_error
do_div:
    sdiv x9, x9, x10
    str x9, [x24, x11, lsl #3]
    b token_loop

op_mod:
    cmp x3, #2
    b.lo line_error
    sub x3, x3, #1
    ldr x10, [x24, x3, lsl #3]
    cbz x10, line_error
    sub x11, x3, #1
    ldr x9, [x24, x11, lsl #3]
    movz x12, #0x8000, lsl #48
    cmp x9, x12
    b.ne do_mod
    cmn x10, #1
    b.eq line_error
do_mod:
    sdiv x12, x9, x10
    msub x9, x12, x10, x9
    str x9, [x24, x11, lsl #3]
    b token_loop

parse_literal:
    mov x7, #0
    mov x8, x4
    mov x9, x6
    ldrb w10, [x8]
    cmp w10, #45
    b.ne literal_digits
    mov x7, #1
    add x8, x8, #1
    sub x9, x9, #1
    cbz x9, line_error

literal_digits:
    cmp x9, #18
    b.hi line_error
    mov x10, #0
    mov x11, #0

digit_loop:
    cmp x11, x9
    b.hs literal_done
    ldrb w12, [x8, x11]
    sub w12, w12, #48
    cmp w12, #9
    b.hi line_error
    mov x13, #10
    mul x10, x10, x13
    add x10, x10, x12
    add x11, x11, #1
    b digit_loop

literal_done:
    cbz x7, literal_push
    neg x10, x10

literal_push:
    str x10, [x24, x3, lsl #3]
    add x3, x3, #1
    b token_loop

line_complete:
    cmp x3, #1
    b.ne line_error
    ldr x9, [x24]
    adrp x10, format_buf
    add x10, x10, :lo12:format_buf
    mov x11, #0

    cmp x9, #0
    b.ge magnitude_ready
    mov w12, #45
    strb w12, [x23], #1
    neg x9, x9

magnitude_ready:
    cbnz x9, convert_digits
    mov w12, #48
    strb w12, [x23], #1
    b result_newline

convert_digits:
    mov x12, #10

convert_loop:
    udiv x13, x9, x12
    msub x14, x13, x12, x9
    add w14, w14, #48
    strb w14, [x10, x11]
    add x11, x11, #1
    mov x9, x13
    cbnz x9, convert_loop

copy_digits:
    cbz x11, result_newline
    sub x11, x11, #1
    ldrb w12, [x10, x11]
    strb w12, [x23], #1
    b copy_digits

result_newline:
    mov w12, #10
    strb w12, [x23], #1
    ret

line_error:
    adrp x10, error_text
    add x10, x10, :lo12:error_text
    mov x11, #6

error_copy:
    ldrb w12, [x10], #1
    strb w12, [x23], #1
    subs x11, x11, #1
    b.ne error_copy
    ret

.section .rodata
error_text:
    .ascii "error\n"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 400000
.balign 16
value_stack:
    .zero 4096
.balign 16
format_buf:
    .zero 32
