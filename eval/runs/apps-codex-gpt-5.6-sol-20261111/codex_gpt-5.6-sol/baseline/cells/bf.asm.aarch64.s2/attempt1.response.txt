.equ INPUT_MAX, 65536
.equ TAPE_SIZE, 30000
.equ OUTPUT_MAX, 65536
.equ BRACKET_DEPTH, 1000
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x24, #0
    mov x28, #INPUT_MAX

read_loop:
    cmp x24, x28
    b.eq read_done
    add x1, x19, x24
    sub x2, x28, x24
    mov x0, #0
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x24, x24, x0
    b read_loop

read_done:
    mov x9, #0

find_delimiter:
    cmp x9, x24
    b.eq no_delimiter
    ldrb w10, [x19, x9]
    cmp w10, #0x21
    b.eq delimiter_found
    add x9, x9, #1
    b find_delimiter

delimiter_found:
    mov x21, x9
    add x23, x9, #1
    b validate_setup

no_delimiter:
    mov x21, x24
    mov x23, x24

validate_setup:
    adrp x27, bracket_map
    add x27, x27, :lo12:bracket_map
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x9, #0
    mov x10, #0

validate_loop:
    cmp x9, x21
    b.eq validate_end
    ldrb w12, [x19, x9]
    cmp w12, #0x5b
    b.eq validate_open
    cmp w12, #0x5d
    b.eq validate_close
    add x9, x9, #1
    b validate_loop

validate_open:
    mov x12, #BRACKET_DEPTH
    cmp x10, x12
    b.hs invalid_program
    str w9, [x11, x10, lsl #2]
    add x10, x10, #1
    add x9, x9, #1
    b validate_loop

validate_close:
    cbz x10, invalid_program
    sub x10, x10, #1
    ldr w12, [x11, x10, lsl #2]
    uxtw x13, w12
    str w9, [x27, x13, lsl #2]
    str w12, [x27, x9, lsl #2]
    add x9, x9, #1
    b validate_loop

validate_end:
    cbnz x10, invalid_program

    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    mov x20, #0
    mov x26, #0

execute_loop:
    cmp x20, x21
    b.eq execute_done
    ldrb w9, [x19, x20]

    cmp w9, #0x3e
    b.eq command_right
    cmp w9, #0x3c
    b.eq command_left
    cmp w9, #0x2b
    b.eq command_plus
    cmp w9, #0x2d
    b.eq command_minus
    cmp w9, #0x2e
    b.eq command_output
    cmp w9, #0x2c
    b.eq command_input
    cmp w9, #0x5b
    b.eq command_open
    cmp w9, #0x5d
    b.eq command_close

next_command:
    add x20, x20, #1
    b execute_loop

command_right:
    add x22, x22, #1
    b next_command

command_left:
    sub x22, x22, #1
    b next_command

command_plus:
    ldrb w10, [x22]
    add w10, w10, #1
    strb w10, [x22]
    b next_command

command_minus:
    ldrb w10, [x22]
    sub w10, w10, #1
    strb w10, [x22]
    b next_command

command_output:
    ldrb w10, [x22]
    strb w10, [x25, x26]
    add x26, x26, #1
    b next_command

command_input:
    cmp x23, x24
    b.hs input_exhausted
    ldrb w10, [x19, x23]
    add x23, x23, #1
    strb w10, [x22]
    b next_command

input_exhausted:
    strb wzr, [x22]
    b next_command

command_open:
    ldrb w10, [x22]
    cbnz w10, next_command
    ldr w10, [x27, x20, lsl #2]
    uxtw x20, w10
    add x20, x20, #1
    b execute_loop

command_close:
    ldrb w10, [x22]
    cbz w10, next_command
    ldr w10, [x27, x20, lsl #2]
    uxtw x20, w10
    add x20, x20, #1
    b execute_loop

execute_done:
    cbz x26, exit_success
    mov x0, #1
    mov x1, x25
    mov x2, x26
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
output_buf:
    .zero OUTPUT_MAX
.balign 16
bracket_map:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
