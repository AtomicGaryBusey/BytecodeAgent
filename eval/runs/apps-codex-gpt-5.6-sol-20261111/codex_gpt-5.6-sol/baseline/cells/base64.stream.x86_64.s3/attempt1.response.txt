.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
.space 49152
output_buffer:
.space 70000

.text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    add r12, rax
    jmp read_loop

encode_start:
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea rbx, [rip + alphabet]
    mov r8, r12
    xor r9d, r9d

encode_groups:
    cmp r8, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi], r10b

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], r10b

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], r10b

    and eax, 63
    movzx r10d, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], r10b

    add rsi, 3
    sub r8, 3
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne encode_groups

    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_groups

encode_tail:
    test r8, r8
    jz finish_lines
    cmp r8, 1
    jne encode_two

    movzx eax, byte ptr [rsi]

    mov edx, eax
    shr edx, 2
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi], r10b

    and eax, 3
    shl eax, 4
    movzx r10d, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], r10b

    mov byte ptr [rdi + 2], '='
    mov byte ptr [rdi + 3], '='
    add rdi, 4
    add r9d, 4
    jmp tail_line_check

encode_two:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    movzx edx, byte ptr [rsi + 1]
    or eax, edx

    mov edx, eax
    shr edx, 10
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi], r10b

    mov edx, eax
    shr edx, 4
    and edx, 63
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], r10b

    and eax, 15
    shl eax, 2
    movzx r10d, byte ptr [rbx + rax]
    mov byte ptr [rdi + 2], r10b

    mov byte ptr [rdi + 3], '='
    add rdi, 4
    add r9d, 4

tail_line_check:
    cmp r9d, 76
    jne finish_lines
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d

finish_lines:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
