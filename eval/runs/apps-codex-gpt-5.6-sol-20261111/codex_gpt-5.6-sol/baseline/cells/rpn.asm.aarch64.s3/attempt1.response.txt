.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    mov x21, #INPUT_SIZE

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop

read_done:
    mov x22, x19
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    mov x24, x23

line_scan:
    cmp x22, x20
    b.hs input_finished
    mov x25, x22

find_newline:
    cmp x25, x20
    b.hs final_line
    ldrb w0, [x25]
    cmp w0, #10
    b.eq newline_found
    add x25, x25, #1
    b find_newline

newline_found:
    mov x0, x22
    mov x1, x25
    bl process_line
    add x22, x25, #1
    b line_scan

final_line:
    mov x0, x22
    mov x1, x20
    bl process_line

input_finished:
    mov x25, x23

write_loop:
    cmp x25, x24
    b.hs exit_ok
    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x25, x25, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x5, x0
    mov x6, x1
    mov x4, #0
    adrp x3, calc_stack
    add x3, x3, :lo12:calc_stack

token_loop:
skip_separators:
    cmp x5, x6
    b.hs line_complete
    ldrb w7, [x5]
    cmp w7, #32
    b.eq skip_one
    cmp w7, #9
    b.ne token_start

skip_one:
    add x5, x5, #1
    b skip_separators

token_start:
    mov x8, x5

token_end_scan:
    cmp x5, x6
    b.hs token_ready
    ldrb w7, [x5]
    cmp w7, #32
    b.eq token_ready
    cmp w7, #9
    b.eq token_ready
    add x5, x5, #1
    b token_end_scan

token_ready:
    sub x10, x5, x8
    cmp x10, #1
    b.ne parse_literal

    ldrb w9, [x8]
    cmp w9, #43
    b.eq op_add
    cmp w9, #45
    b.eq op_sub
    cmp w9, #42
    b.eq op_mul
    cmp w9, #47
    b.eq op_div
    cmp w9, #37
    b.eq op_mod
    b parse_literal

parse_literal:
    mov x11, x8
    mov x12, #0
    mov x14, #0
    ldrb w9, [x11]
    cmp w9, #45
    b.ne literal_digits
    mov x14, #1
    add x11, x11, #1
    sub x15, x5, x11
    cbz x15, line_error
    cmp x15, #18
    b.hi line_error
    b digit_loop

literal_digits:
    cmp x10, #18
    b.hi line_error

digit_loop:
    cmp x11, x5
    b.hs literal_done
    ldrb w9, [x11], #1
    sub w9, w9, #48
    cmp w9, #9
    b.hi line_error
    mov x15, #10
    mul x12, x12, x15
    add x12, x12, x9
    b digit_loop

literal_done:
    cbz x14, literal_push
    neg x12, x12

literal_push:
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

op_add:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #1
    ldr x13, [x3, x4, lsl #3]
    sub x4, x4, #1
    ldr x12, [x3, x4, lsl #3]
    add x12, x12, x13
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

op_sub:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #1
    ldr x13, [x3, x4, lsl #3]
    sub x4, x4, #1
    ldr x12, [x3, x4, lsl #3]
    sub x12, x12, x13
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

op_mul:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #1
    ldr x13, [x3, x4, lsl #3]
    sub x4, x4, #1
    ldr x12, [x3, x4, lsl #3]
    mul x12, x12, x13
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

op_div:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #1
    ldr x13, [x3, x4, lsl #3]
    sub x4, x4, #1
    ldr x12, [x3, x4, lsl #3]
    cbz x13, line_error
    mov x14, #1
    lsl x14, x14, #63
    cmp x12, x14
    b.ne divide_ok
    cmn x13, #1
    b.eq line_error

divide_ok:
    sdiv x12, x12, x13
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

op_mod:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #1
    ldr x13, [x3, x4, lsl #3]
    sub x4, x4, #1
    ldr x12, [x3, x4, lsl #3]
    cbz x13, line_error
    mov x14, #1
    lsl x14, x14, #63
    cmp x12, x14
    b.ne modulo_ok
    cmn x13, #1
    b.eq line_error

modulo_ok:
    sdiv x14, x12, x13
    msub x12, x14, x13, x12
    str x12, [x3, x4, lsl #3]
    add x4, x4, #1
    b token_loop

line_complete:
    cmp x4, #1
    b.ne line_error
    ldr x12, [x3]
    adrp x15, number_buf
    add x15, x15, :lo12:number_buf
    add x16, x15, #32
    mov x17, x16
    mov x14, #0
    cmp x12, #0
    b.ge magnitude_ready
    mov x14, #1
    neg x12, x12

magnitude_ready:
    cbnz x12, conversion_loop
    sub x17, x17, #1
    mov w9, #48
    strb w9, [x17]
    b conversion_done

conversion_loop:
    mov x10, #10
    udiv x11, x12, x10
    msub x9, x11, x10, x12
    add w9, w9, #48
    sub x17, x17, #1
    strb w9, [x17]
    mov x12, x11
    cbnz x12, conversion_loop

conversion_done:
    cbz x14, copy_number
    sub x17, x17, #1
    mov w9, #45
    strb w9, [x17]

copy_number:
    cmp x17, x16
    b.hs number_newline
    ldrb w9, [x17], #1
    strb w9, [x24], #1
    b copy_number

number_newline:
    mov w9, #10
    strb w9, [x24], #1
    ldp x29, x30, [sp], #16
    ret

line_error:
    mov w9, #101
    strb w9, [x24], #1
    mov w9, #114
    strb w9, [x24], #1
    strb w9, [x24], #1
    mov w9, #111
    strb w9, [x24], #1
    mov w9, #114
    strb w9, [x24], #1
    mov w9, #10
    strb w9, [x24], #1
    ldp x29, x30, [sp], #16
    ret

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 400000
.balign 16
calc_stack:
    .zero 8192
.balign 16
number_buf:
    .zero 32
