.equ CAP, 65536

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov edx, CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
    xor r14d, r14d

scan_loop:
    cmp r14, r12
    jae scan_done
    cmp byte ptr [buffer + r14], 10
    jne scan_next

    mov r8, r13
    mov r9, r14
    cmp r8, r9
    jae line_reversed
    dec r9

reverse_line:
    cmp r8, r9
    jae line_reversed
    mov al, byte ptr [buffer + r8]
    mov dl, byte ptr [buffer + r9]
    mov byte ptr [buffer + r8], dl
    mov byte ptr [buffer + r9], al
    inc r8
    dec r9
    jmp reverse_line

line_reversed:
    lea r13, [r14 + 1]

scan_next:
    inc r14
    jmp scan_loop

scan_done:
    cmp r13, r12
    jae output

    mov r8, r13
    mov r9, r12
    dec r9

reverse_final:
    cmp r8, r9
    jae final_reversed
    mov al, byte ptr [buffer + r8]
    mov dl, byte ptr [buffer + r9]
    mov byte ptr [buffer + r8], dl
    mov byte ptr [buffer + r9], al
    inc r8
    dec r9
    jmp reverse_final

final_reversed:
    mov byte ptr [buffer + r12], 10
    inc r12

output:
    test r12, r12
    jz exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + buffer]
    mov rdx, r12
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
buffer:
    .zero 65537
