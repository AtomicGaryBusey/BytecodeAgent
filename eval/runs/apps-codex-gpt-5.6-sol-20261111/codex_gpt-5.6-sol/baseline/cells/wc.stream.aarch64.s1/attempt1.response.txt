.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    mov x2, #INPUT_SIZE
    mov x8, #SYS_read
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x9, x0
    adrp x10, input_buffer
    add x10, x10, :lo12:input_buffer

process_loop:
    ldrb w11, [x10], #1

    cmp w11, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w11, #32
    b.eq whitespace
    cmp w11, #9
    b.lo word_byte
    cmp w11, #13
    b.ls whitespace

word_byte:
    cbnz x22, process_next
    add x20, x20, #1
    mov x22, #1
    b process_next

whitespace:
    mov x22, #0

process_next:
    subs x9, x9, #1
    b.ne process_loop
    b read_loop

input_done:
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer

    mov x0, x19
    mov x1, x23
    bl append_decimal
    mov w2, #32
    strb w2, [x0], #1

    mov x1, x0
    mov x0, x20
    bl append_decimal
    mov w2, #32
    strb w2, [x0], #1

    mov x1, x0
    mov x0, x21
    bl append_decimal
    mov w2, #10
    strb w2, [x0], #1

    sub x2, x0, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

append_decimal:
    sub sp, sp, #32
    mov x3, #0
    mov x4, #10

decimal_generate:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #48
    strb w6, [sp, x3]
    add x3, x3, #1
    mov x0, x5
    cbnz x0, decimal_generate

decimal_copy:
    sub x3, x3, #1
    ldrb w6, [sp, x3]
    strb w6, [x1], #1
    cbnz x3, decimal_copy

    mov x0, x1
    add sp, sp, #32
    ret

.bss
.p2align 4
input_buffer:
    .space 65536
output_buffer:
    .space 64
