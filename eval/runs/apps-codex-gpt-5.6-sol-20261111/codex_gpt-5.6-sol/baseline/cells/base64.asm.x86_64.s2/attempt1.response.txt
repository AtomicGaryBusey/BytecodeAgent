.globl _start

.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 66500

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    cmp r13, INPUT_SIZE
    je read_probe
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INPUT_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

read_probe:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + read_scratch]
    mov edx, 1
    syscall
    test rax, rax
    jz encode_start
    jmp read_probe

encode_start:
    lea rbx, [rip + alphabet]
    lea r14, [rip + output_buffer]
    xor r15d, r15d
    lea r12, [rip + input_buffer]

encode_triples:
    cmp r13, 3
    jb encode_tail

    movzx ebp, byte ptr [r12]
    shl ebp, 16
    movzx eax, byte ptr [r12 + 1]
    shl eax, 8
    or ebp, eax
    movzx eax, byte ptr [r12 + 2]
    or ebp, eax

    mov eax, ebp
    shr eax, 18
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, ebp
    shr eax, 12
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, ebp
    shr eax, 6
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, ebp
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call emit_char

    add r12, 3
    sub r13, 3
    jmp encode_triples

encode_tail:
    cmp r13, 1
    je encode_one
    cmp r13, 2
    je encode_two
    jmp finish_line

encode_one:
    movzx ebp, byte ptr [r12]

    mov eax, ebp
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, ebp
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
    jmp finish_line

encode_two:
    movzx ebp, byte ptr [r12]
    movzx r10d, byte ptr [r12 + 1]

    mov eax, ebp
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, ebp
    and eax, 3
    shl eax, 4
    mov edx, r10d
    shr edx, 4
    or eax, edx
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rbx + rax]
    call emit_char

    mov al, '='
    call emit_char

finish_line:
    test r15, r15
    jz write_output
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d

write_output:
    lea r8, [rip + output_buffer]
    mov r9, r14
    sub r9, r8

write_loop:
    test r9, r9
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    add r8, rax
    sub r9, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne emit_return
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
emit_return:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
read_scratch:
    .zero 1
.align 16
output_buffer:
    .zero OUTPUT_SIZE
