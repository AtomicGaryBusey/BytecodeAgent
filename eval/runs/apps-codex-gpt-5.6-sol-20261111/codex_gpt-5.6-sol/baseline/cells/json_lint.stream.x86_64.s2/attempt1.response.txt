.globl _start

.section .text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14d, r14d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r14, r13
    jne parse_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    syscall

    mov eax, 60
    xor edi, edi
    syscall

parse_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, error_msg_len
    syscall

    lea rsi, [rip + output_buffer + 32]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov r8d, 1
    test rax, rax
    jne decimal_loop

    dec rsi
    mov byte ptr [rsi], '0'
    inc r8
    jmp decimal_done

decimal_loop:
    mov r9d, 10
decimal_digit:
    xor edx, edx
    div r9
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc r8
    test rax, rax
    jne decimal_digit

decimal_done:
    mov eax, 1
    mov edi, 1
    mov rdx, r8
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r14, r13
    jae skip_ws_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je skip_ws_take
    cmp al, 0x09
    je skip_ws_take
    cmp al, 0x0a
    je skip_ws_take
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_take:
    inc r14
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe parse_number
    jmp parse_error

parse_true:
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'r'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'u'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'e'
    jne parse_error
    inc r14
    ret

parse_false:
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'a'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'l'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 's'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'e'
    jne parse_error
    inc r14
    ret

parse_null:
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'u'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'l'
    jne parse_error
    inc r14
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 'l'
    jne parse_error
    inc r14
    ret

parse_string:
    inc r14
string_loop:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb parse_error
    cmp al, '\'
    je string_escape
    inc r14
    jmp string_loop

string_end:
    inc r14
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je simple_escape
    cmp al, '\'
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    jne parse_error

    inc r14
    mov ecx, 4
unicode_loop:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe unicode_good
    cmp al, 'A'
    jb parse_error
    cmp al, 'F'
    jbe unicode_good
    cmp al, 'a'
    jb parse_error
    cmp al, 'f'
    ja parse_error
unicode_good:
    inc r14
    dec ecx
    jne unicode_loop
    jmp string_loop

simple_escape:
    inc r14
    jmp string_loop

parse_number:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '-'
    jne number_integer
    inc r14
    cmp r14, r13
    jae parse_error

number_integer:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error

number_digits:
    inc r14
    cmp r14, r13
    jae number_after_integer
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_after_integer
    cmp al, '9'
    jbe number_digits
    jmp number_after_integer

number_zero:
    inc r14

number_after_integer:
    cmp r14, r13
    jae number_done
    cmp byte ptr [r12 + r14], '.'
    jne number_check_exp
    inc r14
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

fraction_digits:
    inc r14
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_check_exp
    cmp al, '9'
    jbe fraction_digits

number_check_exp:
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 'e'
    je exponent_start
    cmp al, 'E'
    jne number_done

exponent_start:
    inc r14
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_require_digit

exponent_sign:
    inc r14
    cmp r14, r13
    jae parse_error

exponent_require_digit:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

exponent_digits:
    inc r14
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits

number_done:
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], ']'
    je array_end

array_value:
    call parse_value
    call skip_ws
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, ']'
    je array_end
    cmp al, ','
    jne parse_error
    inc r14
    call skip_ws
    jmp array_value

array_end:
    inc r14
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], '}'
    je object_end

object_member:
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], ':'
    jne parse_error
    inc r14
    call skip_ws
    call parse_value
    call skip_ws
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, '}'
    je object_end
    cmp al, ','
    jne parse_error
    inc r14
    call skip_ws
    jmp object_member

object_end:
    inc r14
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg

error_msg:
    .ascii "error at "
error_msg_len = . - error_msg

.bss
.balign 16
input_buffer:
    .zero 65537
output_buffer:
    .zero 33
