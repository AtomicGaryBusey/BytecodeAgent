.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 32768

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x1, x19
    add x1, x1, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, parse_input
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble parse_input
    add x20, x20, x0
    b read_loop

parse_input:
    adrp x25, line_table
    add x25, x25, :lo12:line_table
    mov x21, #0
    mov x22, #0
    mov x24, #0

parse_loop:
    cmp x21, x20
    b.hs parse_remainder
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.ne parse_next

    add x1, x19, x22
    sub x2, x21, x22
    add x3, x25, x24, lsl #4
    stp x1, x2, [x3]
    add x24, x24, #1
    add x22, x21, #1

parse_next:
    add x21, x21, #1
    b parse_loop

parse_remainder:
    cmp x22, x20
    b.hs sort_begin
    add x1, x19, x22
    sub x2, x20, x22
    add x3, x25, x24, lsl #4
    stp x1, x2, [x3]
    add x24, x24, #1

sort_begin:
    mov x26, #1

sort_outer:
    cmp x26, x24
    b.hs output_begin

    add x7, x25, x26, lsl #4
    ldp x5, x6, [x7]
    mov x27, x26

sort_inner:
    cbz x27, sort_place
    sub x8, x27, #1
    add x7, x25, x8, lsl #4
    ldp x9, x10, [x7]

    cmp x10, x6
    csel x11, x10, x6, lo
    mov x12, #0

compare_loop:
    cmp x12, x11
    b.hs compare_lengths
    ldrb w13, [x9, x12]
    ldrb w14, [x5, x12]
    cmp w13, w14
    b.hi sort_shift
    b.lo sort_place
    add x12, x12, #1
    b compare_loop

compare_lengths:
    cmp x10, x6
    b.ls sort_place

sort_shift:
    add x7, x25, x27, lsl #4
    stp x9, x10, [x7]
    sub x27, x27, #1
    b sort_inner

sort_place:
    add x7, x25, x27, lsl #4
    stp x5, x6, [x7]
    add x26, x26, #1
    b sort_outer

output_begin:
    mov x26, #0

output_loop:
    cmp x26, x24
    b.hs program_exit
    add x7, x25, x26, lsl #4
    ldp x1, x2, [x7]
    cbz x2, output_newline

    mov x0, #1
    mov x8, #SYS_write
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    add x26, x26, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .space 32768
.balign 16
line_table:
    .space 16000
