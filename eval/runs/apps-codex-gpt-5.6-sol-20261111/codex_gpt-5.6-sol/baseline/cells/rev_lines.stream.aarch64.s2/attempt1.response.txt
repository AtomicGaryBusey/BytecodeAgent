.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0

scan_loop:
    cmp x22, x20
    b.hs scan_done
    ldrb w9, [x19, x22]
    cmp w9, #10
    b.eq complete_line
    add x22, x22, #1
    b scan_loop

complete_line:
    sub x25, x22, x21
    add x23, x19, x21
    add x24, x19, x22

reverse_complete:
    cmp x25, #1
    b.ls write_complete
    sub x24, x24, #1
    ldrb w9, [x23]
    ldrb w10, [x24]
    strb w10, [x23], #1
    strb w9, [x24]
    sub x25, x25, #2
    b reverse_complete

write_complete:
    mov x0, #1
    add x1, x19, x21
    sub x2, x22, x21
    add x2, x2, #1
    mov x8, #SYS_write
    svc #0
    add x22, x22, #1
    mov x21, x22
    b scan_loop

scan_done:
    cmp x21, x20
    b.hs exit_success

    sub x25, x20, x21
    add x23, x19, x21
    add x24, x19, x20

reverse_final:
    cmp x25, #1
    b.ls write_final
    sub x24, x24, #1
    ldrb w9, [x23]
    ldrb w10, [x24]
    strb w10, [x23], #1
    strb w9, [x24]
    sub x25, x25, #2
    b reverse_final

write_final:
    mov x0, #1
    add x1, x19, x21
    sub x2, x20, x21
    mov x8, #SYS_write
    svc #0

    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_write
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
