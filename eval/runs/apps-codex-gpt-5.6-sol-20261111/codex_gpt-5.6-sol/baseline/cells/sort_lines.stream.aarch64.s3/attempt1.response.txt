.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 32768

.text
.globl _start
_start:
    adrp x9, input_buffer
    add x9, x9, :lo12:input_buffer
    mov x10, #0

read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x10
    cbz x2, read_full
    mov x0, #0
    add x1, x9, x10
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x10, x10, x0
    b read_loop

read_full:
    mov x0, #0
    add x1, x9, x10
    mov x2, #0
    mov x8, #SYS_read
    svc #0

read_done:
    adrp x11, line_entries
    add x11, x11, :lo12:line_entries
    mov x12, #0
    mov x13, #0
    mov x14, #0

parse_loop:
    cmp x14, x10
    b.hs parse_remainder
    ldrb w15, [x9, x14]
    cmp w15, #10
    b.ne parse_next

    add x16, x9, x13
    sub x17, x14, x13
    add x18, x11, x12, lsl #4
    str x16, [x18]
    str x17, [x18, #8]
    add x12, x12, #1
    add x13, x14, #1

parse_next:
    add x14, x14, #1
    b parse_loop

parse_remainder:
    cmp x13, x10
    b.hs sort_begin
    add x16, x9, x13
    sub x17, x10, x13
    add x18, x11, x12, lsl #4
    str x16, [x18]
    str x17, [x18, #8]
    add x12, x12, #1

sort_begin:
    mov x19, #1

sort_outer:
    cmp x19, x12
    b.hs output_begin

    add x0, x11, x19, lsl #4
    ldr x20, [x0]
    ldr x21, [x0, #8]
    mov x22, x19

sort_inner:
    cbz x22, sort_insert

    sub x23, x22, #1
    add x0, x11, x23, lsl #4
    ldr x24, [x0]
    ldr x25, [x0, #8]

    cmp x25, x21
    csel x26, x25, x21, ls
    mov x27, #0

compare_loop:
    cmp x27, x26
    b.hs compare_prefix
    ldrb w1, [x24, x27]
    ldrb w2, [x20, x27]
    cmp w1, w2
    b.lo sort_insert
    b.hi sort_shift
    add x27, x27, #1
    b compare_loop

compare_prefix:
    cmp x25, x21
    b.ls sort_insert

sort_shift:
    add x3, x11, x22, lsl #4
    str x24, [x3]
    str x25, [x3, #8]
    sub x22, x22, #1
    b sort_inner

sort_insert:
    add x0, x11, x22, lsl #4
    str x20, [x0]
    str x21, [x0, #8]
    add x19, x19, #1
    b sort_outer

output_begin:
    mov x19, #0

output_loop:
    cmp x19, x12
    b.hs program_exit

    add x3, x11, x19, lsl #4
    ldr x1, [x3]
    ldr x2, [x3, #8]
    cbz x2, output_newline

    mov x0, #1
    mov x8, #SYS_write
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    add x19, x19, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .space 32768
.balign 16
line_entries:
    .space 16000
