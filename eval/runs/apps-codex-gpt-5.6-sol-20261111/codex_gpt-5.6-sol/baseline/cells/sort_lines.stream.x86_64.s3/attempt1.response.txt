.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000
.equ DESC_SIZE, 16

.globl _start

.text

_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbp, [rip + descriptors]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

parse_loop:
    cmp r14, r12
    jae parse_remainder
    cmp byte ptr [rbx + r14], 10
    jne parse_next

    mov rax, r13
    shl rax, 4
    lea rcx, [rbx + r15]
    mov qword ptr [rbp + rax], rcx
    mov rdx, r14
    sub rdx, r15
    mov qword ptr [rbp + rax + 8], rdx
    inc r13
    lea r15, [r14 + 1]

parse_next:
    inc r14
    jmp parse_loop

parse_remainder:
    cmp r15, r12
    jae sort_begin
    mov rax, r13
    shl rax, 4
    lea rcx, [rbx + r15]
    mov qword ptr [rbp + rax], rcx
    mov rdx, r12
    sub rdx, r15
    mov qword ptr [rbp + rax + 8], rdx
    inc r13

sort_begin:
    mov r14d, 1

sort_outer:
    cmp r14, r13
    jae output_begin

    mov rax, r14
    shl rax, 4
    mov r8, qword ptr [rbp + rax]
    mov r9, qword ptr [rbp + rax + 8]
    mov r15, r14

sort_inner:
    test r15, r15
    jz sort_place

    mov r10, r15
    shl r10, 4
    mov rax, qword ptr [rbp + r10 - DESC_SIZE]
    mov rdx, qword ptr [rbp + r10 - DESC_SIZE + 8]

    mov rsi, r8
    mov rdi, rax
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    repe cmpsb
    jne sort_different

    cmp r9, rdx
    jb sort_shift
    jmp sort_place

sort_different:
    movzx ecx, byte ptr [rsi - 1]
    movzx edi, byte ptr [rdi - 1]
    cmp ecx, edi
    jb sort_shift
    jmp sort_place

sort_shift:
    mov qword ptr [rbp + r10], rax
    mov qword ptr [rbp + r10 + 8], rdx
    dec r15
    jmp sort_inner

sort_place:
    mov r10, r15
    shl r10, 4
    mov qword ptr [rbp + r10], r8
    mov qword ptr [rbp + r10 + 8], r9
    inc r14
    jmp sort_outer

output_begin:
    xor r14d, r14d

output_loop:
    cmp r14, r13
    jae exit_success

    mov rax, r14
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rdx, qword ptr [rbp + rax + 8]
    call write_all

    lea rsi, [rip + newline]
    mov edx, 1
    call write_all

    inc r14
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done
    mov r8, rdx

write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, r8
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub r8, rax
    jne write_loop

write_done:
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
descriptors:
    .zero MAX_LINES * DESC_SIZE
