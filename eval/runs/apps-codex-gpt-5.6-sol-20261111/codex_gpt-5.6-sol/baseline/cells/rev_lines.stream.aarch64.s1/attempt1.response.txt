.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 65537

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x21, #0
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x21, x20
    b.eq scan_done
    ldrb w24, [x19, x21]
    cmp w24, #10
    b.eq line_found
    add x21, x21, #1
    b scan_loop

line_found:
    mov x24, x21

reverse_line:
    cmp x24, x22
    b.eq append_newline
    sub x24, x24, #1
    ldrb w25, [x19, x24]
    strb w25, [x26, x23]
    add x23, x23, #1
    b reverse_line

append_newline:
    mov w25, #10
    strb w25, [x26, x23]
    add x23, x23, #1
    add x21, x21, #1
    mov x22, x21
    b scan_loop

scan_done:
    cmp x22, x20
    b.eq write_output
    mov x24, x20

reverse_final:
    cmp x24, x22
    b.eq append_final_newline
    sub x24, x24, #1
    ldrb w25, [x19, x24]
    strb w25, [x26, x23]
    add x23, x23, #1
    b reverse_final

append_final_newline:
    mov w25, #10
    strb w25, [x26, x23]
    add x23, x23, #1

write_output:
    cbz x23, exit_success
    mov x0, #1
    mov x1, x26
    mov x2, x23
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero OUTPUT_SIZE
