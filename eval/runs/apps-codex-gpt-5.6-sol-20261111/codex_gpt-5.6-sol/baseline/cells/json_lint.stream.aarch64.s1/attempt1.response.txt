.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x19
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x0, #0
    bl skip_ws
    cmp x0, x19
    b.eq error_at
    bl parse_value
    bl skip_ws
    cmp x0, x19
    b.ne error_at

valid:
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

skip_ws:
    cmp x0, x19
    b.hs 2f
1:
    ldrb w1, [x20, x0]
    cmp w1, #0x20
    b.eq 3f
    cmp w1, #0x09
    b.eq 3f
    cmp w1, #0x0a
    b.eq 3f
    cmp w1, #0x0d
    b.ne 2f
3:
    add x0, x0, #1
    cmp x0, x19
    b.lo 1b
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x22
    b.eq pv_string
    cmp w1, #0x7b
    b.eq pv_object
    cmp w1, #0x5b
    b.eq pv_array
    cmp w1, #0x74
    b.eq pv_true
    cmp w1, #0x66
    b.eq pv_false
    cmp w1, #0x6e
    b.eq pv_null
    cmp w1, #0x2d
    b.eq pv_number
    cmp w1, #0x30
    b.lo error_at
    cmp w1, #0x39
    b.hi error_at
pv_number:
    bl parse_number
    ldp x29, x30, [sp], #16
    ret
pv_string:
    bl parse_string
    ldp x29, x30, [sp], #16
    ret
pv_object:
    bl parse_object
    ldp x29, x30, [sp], #16
    ret
pv_array:
    bl parse_array
    ldp x29, x30, [sp], #16
    ret
pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    mov x2, #4
    bl parse_literal
    ldp x29, x30, [sp], #16
    ret
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    mov x2, #5
    bl parse_literal
    ldp x29, x30, [sp], #16
    ret
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    mov x2, #4
    bl parse_literal
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    mov x3, #0
1:
    cmp x3, x2
    b.eq 3f
    cmp x0, x19
    b.hs error_at
    ldrb w4, [x20, x0]
    ldrb w5, [x1, x3]
    cmp w4, w5
    b.ne error_at
    add x0, x0, #1
    add x3, x3, #1
    b 1b
3:
    ret

parse_string:
    add x0, x0, #1
1:
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x22
    b.eq 8f
    cmp w1, #0x20
    b.lo error_at
    cmp w1, #0x5c
    b.eq 3f
    add x0, x0, #1
    b 1b
3:
    add x0, x0, #1
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x22
    b.eq 7f
    cmp w1, #0x5c
    b.eq 7f
    cmp w1, #0x2f
    b.eq 7f
    cmp w1, #0x62
    b.eq 7f
    cmp w1, #0x66
    b.eq 7f
    cmp w1, #0x6e
    b.eq 7f
    cmp w1, #0x72
    b.eq 7f
    cmp w1, #0x74
    b.eq 7f
    cmp w1, #0x75
    b.ne error_at
    add x0, x0, #1
    mov x2, #4
4:
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo error_at
    cmp w1, #0x39
    b.ls 6f
    cmp w1, #0x41
    b.lo error_at
    cmp w1, #0x46
    b.ls 6f
    cmp w1, #0x61
    b.lo error_at
    cmp w1, #0x66
    b.hi error_at
6:
    add x0, x0, #1
    subs x2, x2, #1
    b.ne 4b
    b 1b
7:
    add x0, x0, #1
    b 1b
8:
    add x0, x0, #1
    ret

parse_number:
    ldrb w1, [x20, x0]
    cmp w1, #0x2d
    b.ne 1f
    add x0, x0, #1
    cmp x0, x19
    b.hs error_at
1:
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.eq 3f
    cmp w1, #0x31
    b.lo error_at
    cmp w1, #0x39
    b.hi error_at
2:
    add x0, x0, #1
    cmp x0, x19
    b.hs 4f
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo 4f
    cmp w1, #0x39
    b.ls 2b
    b 4f
3:
    add x0, x0, #1
4:
    cmp x0, x19
    b.hs 9f
    ldrb w1, [x20, x0]
    cmp w1, #0x2e
    b.ne 6f
    add x0, x0, #1
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo error_at
    cmp w1, #0x39
    b.hi error_at
5:
    add x0, x0, #1
    cmp x0, x19
    b.hs 9f
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo 6f
    cmp w1, #0x39
    b.ls 5b
6:
    cmp x0, x19
    b.hs 9f
    ldrb w1, [x20, x0]
    cmp w1, #0x65
    b.eq 7f
    cmp w1, #0x45
    b.ne 9f
7:
    add x0, x0, #1
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x2b
    b.eq 8f
    cmp w1, #0x2d
    b.ne 10f
8:
    add x0, x0, #1
    cmp x0, x19
    b.hs error_at
10:
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo error_at
    cmp w1, #0x39
    b.hi error_at
11:
    add x0, x0, #1
    cmp x0, x19
    b.hs 9f
    ldrb w1, [x20, x0]
    cmp w1, #0x30
    b.lo 9f
    cmp w1, #0x39
    b.ls 11b
9:
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x5d
    b.eq 4f
1:
    bl parse_value
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x5d
    b.eq 4f
    cmp w1, #0x2c
    b.ne error_at
    add x0, x0, #1
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x5d
    b.eq error_at
    b 1b
4:
    add x0, x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x7d
    b.eq 5f
1:
    cmp w1, #0x22
    b.ne error_at
    bl parse_string
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x3a
    b.ne error_at
    add x0, x0, #1
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    bl parse_value
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x7d
    b.eq 5f
    cmp w1, #0x2c
    b.ne error_at
    add x0, x0, #1
    bl skip_ws
    cmp x0, x19
    b.hs error_at
    ldrb w1, [x20, x0]
    cmp w1, #0x7d
    b.eq error_at
    b 1b
5:
    add x0, x0, #1
    ldp x29, x30, [sp], #16
    ret

error_at:
    mov x21, x0
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    adrp x4, output_buf
    add x4, x4, :lo12:output_buf
    add x5, x4, #31
    mov w6, #0x0a
    strb w6, [x5]
    mov x7, x5
    mov x3, x21
    cbnz x3, 2f
    sub x7, x7, #1
    mov w6, #0x30
    strb w6, [x7]
    b 3f
2:
    mov x9, #10
1:
    udiv x10, x3, x9
    msub x11, x10, x9, x3
    add w11, w11, #0x30
    sub x7, x7, #1
    strb w11, [x7]
    mov x3, x10
    cbnz x3, 1b
3:
    mov x0, #1
    mov x1, x7
    add x2, x5, #1
    sub x2, x2, x7
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero 32
