.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_CAP, 65536

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x9, #INPUT_CAP
    sub x2, x9, x20
    cbz x2, read_probe
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_probe:
    mov x0, #0
    adrp x1, probe_byte
    add x1, x1, :lo12:probe_byte
    mov x2, #1
    mov x8, #SYS_read
    svc #0
    cbnz x0, read_probe
    b read_done

read_done:
    add x20, x19, x20
    mov x21, x19
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x23, value_stack
    add x23, x23, :lo12:value_stack

next_line:
    cmp x21, x20
    b.hs output_all

    mov x25, x21

find_line_end:
    cmp x25, x20
    b.hs line_end_found
    ldrb w9, [x25]
    cmp w9, #10
    b.eq line_end_found
    add x25, x25, #1
    b find_line_end

line_end_found:
    mov x26, x21
    mov x24, #0
    mov w28, #0

token_loop:
    cmp x26, x25
    b.hs line_complete

skip_separators:
    cmp x26, x25
    b.hs line_complete
    ldrb w9, [x26]
    cmp w9, #32
    b.eq skip_one
    cmp w9, #9
    b.ne token_start
skip_one:
    add x26, x26, #1
    b skip_separators

token_start:
    mov x27, x26

scan_token:
    cmp x26, x25
    b.hs token_scanned
    ldrb w9, [x26]
    cmp w9, #32
    b.eq token_scanned
    cmp w9, #9
    b.eq token_scanned
    add x26, x26, #1
    b scan_token

token_scanned:
    sub x10, x26, x27
    cmp x10, #1
    b.ne parse_literal

    ldrb w11, [x27]
    cmp w11, #43
    b.eq operator_add
    cmp w11, #45
    b.eq operator_sub
    cmp w11, #42
    b.eq operator_mul
    cmp w11, #47
    b.eq operator_div
    cmp w11, #37
    b.eq operator_mod
    b parse_literal

operator_add:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x11, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x12, [x23, x24, lsl #3]
    add x12, x12, x11
    str x12, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

operator_sub:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x11, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x12, [x23, x24, lsl #3]
    sub x12, x12, x11
    str x12, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

operator_mul:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x11, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x12, [x23, x24, lsl #3]
    mul x12, x12, x11
    str x12, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

operator_div:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x11, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x12, [x23, x24, lsl #3]
    cbz x11, line_error
    mov x13, #1
    lsl x13, x13, #63
    cmp x12, x13
    b.ne do_div
    cmn x11, #1
    b.eq line_error
do_div:
    sdiv x12, x12, x11
    str x12, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

operator_mod:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x11, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x12, [x23, x24, lsl #3]
    cbz x11, line_error
    mov x13, #1
    lsl x13, x13, #63
    cmp x12, x13
    b.ne do_mod
    cmn x11, #1
    b.eq line_error
do_mod:
    sdiv x13, x12, x11
    msub x12, x13, x11, x12
    str x12, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

parse_literal:
    mov x11, x27
    mov x12, x10
    mov w13, #0
    ldrb w14, [x11]
    cmp w14, #45
    b.ne literal_digits
    mov w13, #1
    add x11, x11, #1
    sub x12, x12, #1

literal_digits:
    cbz x12, line_error
    cmp x12, #18
    b.hi line_error
    mov x14, #0
    mov x15, #0

literal_loop:
    cmp x15, x12
    b.hs literal_done
    ldrb w16, [x11, x15]
    sub w16, w16, #48
    cmp w16, #9
    b.hi line_error
    mov x17, #10
    madd x14, x14, x17, x16
    add x15, x15, #1
    b literal_loop

literal_done:
    cbz w13, literal_push
    neg x14, x14

literal_push:
    str x14, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

line_complete:
    cmp x24, #1
    b.ne line_error
    ldr x9, [x23]
    b emit_number

line_error:
    mov w9, #101
    strb w9, [x22], #1
    mov w9, #114
    strb w9, [x22], #1
    strb w9, [x22], #1
    mov w9, #111
    strb w9, [x22], #1
    mov w9, #114
    strb w9, [x22], #1
    mov w9, #10
    strb w9, [x22], #1
    b advance_line

emit_number:
    mov x10, x9
    mov w11, #0
    tbz x9, #63, number_abs_ready
    mov w11, #1
    neg x10, x10

number_abs_ready:
    adrp x12, number_tmp
    add x12, x12, :lo12:number_tmp
    mov x13, #0
    mov x14, #10

digit_generate:
    udiv x15, x10, x14
    msub x16, x15, x14, x10
    add w16, w16, #48
    strb w16, [x12, x13]
    add x13, x13, #1
    mov x10, x15
    cbnz x10, digit_generate

    cbz w11, copy_digits
    mov w16, #45
    strb w16, [x22], #1

copy_digits:
    cbz x13, number_newline
    sub x13, x13, #1
    ldrb w16, [x12, x13]
    strb w16, [x22], #1
    b copy_digits

number_newline:
    mov w16, #10
    strb w16, [x22], #1

advance_line:
    cmp x25, x20
    b.hs output_all
    add x21, x25, #1
    b next_line

output_all:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x22, x1
    cbz x2, exit_ok
    mov x0, #1
    mov x8, #SYS_write
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
input_buf:
    .space 65536
probe_byte:
    .space 1

.balign 16
value_stack:
    .space 8192

.balign 16
number_tmp:
    .space 32

.balign 16
output_buf:
    .space 400000
