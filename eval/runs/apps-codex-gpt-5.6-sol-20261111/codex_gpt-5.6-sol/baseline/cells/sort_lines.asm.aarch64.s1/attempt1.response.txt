.equ BUF_CAP, 32768
.equ MAX_LINES, 1000
.equ REC_SIZE, 16

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x3, #BUF_CAP
    subs x2, x3, x20
    b.eq read_full_check
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_full_check:
    mov x0, #0
    adrp x1, read_scratch
    add x1, x1, :lo12:read_scratch
    mov x2, #1
    mov x8, #63
    svc #0

read_done:
    adrp x23, line_records
    add x23, x23, :lo12:line_records
    mov x21, #0
    mov x4, #0
    mov x5, #0

parse_loop:
    cmp x5, x20
    b.hs parse_remainder
    ldrb w6, [x19, x5]
    cmp w6, #10
    b.ne parse_next

    add x7, x23, x21, lsl #4
    add x9, x19, x4
    sub x10, x5, x4
    stp x9, x10, [x7]
    add x21, x21, #1
    add x4, x5, #1

parse_next:
    add x5, x5, #1
    b parse_loop

parse_remainder:
    cmp x4, x20
    b.hs sort_begin
    add x7, x23, x21, lsl #4
    add x9, x19, x4
    sub x10, x20, x4
    stp x9, x10, [x7]
    add x21, x21, #1

sort_begin:
    mov x22, #1

sort_outer:
    cmp x22, x21
    b.hs output_begin

    add x7, x23, x22, lsl #4
    ldp x10, x11, [x7]
    mov x24, x22

sort_inner:
    cbz x24, sort_insert

    sub x25, x24, #1
    add x7, x23, x25, lsl #4
    ldp x12, x13, [x7]
    mov x14, #0

compare_loop:
    cmp x14, x11
    b.eq compare_key_end
    cmp x14, x13
    b.eq compare_no_shift

    ldrb w15, [x10, x14]
    ldrb w16, [x12, x14]
    cmp w15, w16
    b.lo compare_shift
    b.hi compare_no_shift
    add x14, x14, #1
    b compare_loop

compare_key_end:
    cmp x14, x13
    b.lo compare_shift
    b compare_no_shift

compare_shift:
    add x7, x23, x24, lsl #4
    stp x12, x13, [x7]
    sub x24, x24, #1
    b sort_inner

compare_no_shift:
sort_insert:
    add x7, x23, x24, lsl #4
    stp x10, x11, [x7]
    add x22, x22, #1
    b sort_outer

output_begin:
    mov x22, #0

output_loop:
    cmp x22, x21
    b.hs exit_success

    add x7, x23, x22, lsl #4
    ldp x24, x25, [x7]
    cbz x25, output_newline

    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    mov x8, #64
    svc #0

    add x22, x22, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline_byte:
    .byte 10

.bss
.balign 16
input_buffer:
    .space 32768
read_scratch:
    .space 1
.balign 16
line_records:
    .space 16000
