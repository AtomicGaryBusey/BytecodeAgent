.globl _start

.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ OUTPUT_SIZE, 65536
.equ BRACKET_DEPTH, 1000

.text
_start:
    xor ebx, ebx

read_loop:
    cmp ebx, INPUT_SIZE
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, INPUT_SIZE
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r15, [rip + input_buffer]
    add r15, rbx
    mov r12, rbx
    mov r14, r15
    xor eax, eax
    lea rsi, [rip + input_buffer]

find_delimiter:
    cmp rax, rbx
    jae delimiter_done
    cmp byte ptr [rsi + rax], 0x21
    je delimiter_found
    inc rax
    jmp find_delimiter

delimiter_found:
    mov r12, rax
    lea r14, [rsi + rax + 1]

delimiter_done:
    lea r9, [rip + bracket_stack]
    lea r10, [rip + bracket_map]
    xor edx, edx
    xor eax, eax

validate_loop:
    cmp rax, r12
    jae validate_end
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, '['
    je validate_open
    cmp cl, ']'
    je validate_close

validate_next:
    inc rax
    jmp validate_loop

validate_open:
    cmp edx, BRACKET_DEPTH
    jae invalid_program
    mov qword ptr [r9 + rdx*8], rax
    inc rdx
    jmp validate_next

validate_close:
    test rdx, rdx
    jz invalid_program
    dec rdx
    mov rcx, qword ptr [r9 + rdx*8]
    mov qword ptr [r10 + rcx*8], rax
    mov qword ptr [r10 + rax*8], rcx
    jmp validate_next

validate_end:
    test rdx, rdx
    jnz invalid_program

    lea r13, [rip + tape]
    xor ebx, ebx
    xor ebp, ebp
    lea r10, [rip + bracket_map]

execute_loop:
    cmp rbx, r12
    jae execute_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]

    cmp al, '>'
    je command_right
    cmp al, '<'
    je command_left
    cmp al, '+'
    je command_plus
    cmp al, '-'
    je command_minus
    cmp al, '.'
    je command_output
    cmp al, ','
    je command_input
    cmp al, '['
    je command_open
    cmp al, ']'
    je command_close

command_next:
    inc rbx
    jmp execute_loop

command_right:
    inc r13
    jmp command_next

command_left:
    dec r13
    jmp command_next

command_plus:
    inc byte ptr [r13]
    jmp command_next

command_minus:
    dec byte ptr [r13]
    jmp command_next

command_output:
    movzx eax, byte ptr [r13]
    lea rsi, [rip + output_buffer]
    mov byte ptr [rsi + rbp], al
    inc rbp
    cmp rbp, OUTPUT_SIZE
    jne command_next

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov edx, OUTPUT_SIZE
    syscall
    xor ebp, ebp
    jmp command_next

command_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [r13], al
    jmp command_next

input_exhausted:
    mov byte ptr [r13], 0
    jmp command_next

command_open:
    cmp byte ptr [r13], 0
    jne command_next
    mov rbx, qword ptr [r10 + rbx*8]
    jmp command_next

command_close:
    cmp byte ptr [r13], 0
    je command_next
    mov rbx, qword ptr [r10 + rbx*8]
    jmp command_next

execute_done:
    test rbp, rbp
    jz successful_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rbp
    syscall

successful_exit:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
tape:
    .zero TAPE_SIZE
output_buffer:
    .zero OUTPUT_SIZE
.balign 8
bracket_stack:
    .zero 8000
bracket_map:
    .zero 524288
