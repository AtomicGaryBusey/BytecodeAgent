.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_MAX, 1000

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x9, #0

find_delimiter:
    cmp x9, x20
    b.hs no_delimiter
    ldrb w10, [x19, x9]
    cmp w10, #0x21
    b.eq delimiter_found
    add x9, x9, #1
    b find_delimiter

delimiter_found:
    mov x23, x9
    add x24, x19, x9
    add x24, x24, #1
    add x25, x19, x20
    b validate_setup

no_delimiter:
    mov x23, x20
    add x24, x19, x20
    mov x25, x24

validate_setup:
    adrp x26, bracket_stack
    add x26, x26, :lo12:bracket_stack
    adrp x28, bracket_map
    add x28, x28, :lo12:bracket_map
    mov x27, #0
    mov x9, #0

validate_loop:
    cmp x9, x23
    b.hs validate_end
    ldrb w10, [x19, x9]
    cmp w10, #0x5b
    b.eq validate_open
    cmp w10, #0x5d
    b.eq validate_close

validate_next:
    add x9, x9, #1
    b validate_loop

validate_open:
    cmp x27, #BRACKET_MAX
    b.hs exit_error
    str w9, [x26, x27, lsl #2]
    add x27, x27, #1
    b validate_next

validate_close:
    cbz x27, exit_error
    sub x27, x27, #1
    ldr w11, [x26, x27, lsl #2]
    str w9, [x28, x11, lsl #2]
    str w11, [x28, x9, lsl #2]
    b validate_next

validate_end:
    cbnz x27, exit_error

    adrp x21, tape
    add x21, x21, :lo12:tape
    adrp x15, output_buffer
    add x15, x15, :lo12:output_buffer
    mov x16, #0
    mov x22, #0

execute_loop:
    cmp x22, x23
    b.hs execution_done
    mov x9, x22
    ldrb w10, [x19, x22]
    add x22, x22, #1

    cmp w10, #0x3e
    b.eq command_right
    cmp w10, #0x3c
    b.eq command_left
    cmp w10, #0x2b
    b.eq command_increment
    cmp w10, #0x2d
    b.eq command_decrement
    cmp w10, #0x2e
    b.eq command_output
    cmp w10, #0x2c
    b.eq command_input
    cmp w10, #0x5b
    b.eq command_open
    cmp w10, #0x5d
    b.eq command_close
    b execute_loop

command_right:
    add x21, x21, #1
    b execute_loop

command_left:
    sub x21, x21, #1
    b execute_loop

command_increment:
    ldrb w11, [x21]
    add w11, w11, #1
    strb w11, [x21]
    b execute_loop

command_decrement:
    ldrb w11, [x21]
    sub w11, w11, #1
    strb w11, [x21]
    b execute_loop

command_output:
    ldrb w11, [x21]
    strb w11, [x15, x16]
    add x16, x16, #1
    b execute_loop

command_input:
    cmp x24, x25
    b.hs input_exhausted
    ldrb w11, [x24], #1
    strb w11, [x21]
    b execute_loop

input_exhausted:
    strb wzr, [x21]
    b execute_loop

command_open:
    ldrb w11, [x21]
    cbnz w11, execute_loop
    ldr w22, [x28, x9, lsl #2]
    add x22, x22, #1
    b execute_loop

command_close:
    ldrb w11, [x21]
    cbz w11, execute_loop
    ldr w22, [x28, x9, lsl #2]
    add x22, x22, #1
    b execute_loop

execution_done:
    cbz x16, exit_success
    mov x0, #1
    mov x1, x15
    mov x2, x16
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

exit_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buffer:
    .zero INPUT_MAX
output_buffer:
    .zero INPUT_MAX
tape:
    .zero TAPE_SIZE
.balign 16
bracket_stack:
    .zero 4000
.balign 16
bracket_map:
    .zero 262144
