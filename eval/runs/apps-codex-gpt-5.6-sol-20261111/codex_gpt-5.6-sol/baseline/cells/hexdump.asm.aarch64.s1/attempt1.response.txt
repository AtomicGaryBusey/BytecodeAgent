.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 323600

.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
.zero INPUT_SIZE
.balign 16
output_buffer:
.zero OUTPUT_SIZE

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x9, #INPUT_SIZE
    cmp x20, x9
    b.hs format_input

    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le format_input
    add x20, x20, x0
    b read_loop

format_input:
    adrp x27, hex_digits
    add x27, x27, :lo12:hex_digits
    adrp x28, output_buffer
    add x28, x28, :lo12:output_buffer
    mov x21, x28
    mov x22, #0

row_loop:
    cmp x22, x20
    b.hs final_offset

    sub x23, x20, x22
    cmp x23, #16
    b.ls row_count_ready
    mov x23, #16

row_count_ready:
    mov x0, x22
    mov x1, x21
    bl emit_hex8
    mov x21, x1

    mov w9, #' '
    strb w9, [x21], #1
    strb w9, [x21], #1

    mov x24, #0

slot_loop:
    cmp x24, #16
    b.hs slots_done
    cmp x24, x23
    b.hs empty_slot

    add x25, x22, x24
    ldrb w26, [x19, x25]

    lsr w10, w26, #4
    ldrb w11, [x27, x10]
    strb w11, [x21], #1
    and w10, w26, #15
    ldrb w11, [x27, x10]
    strb w11, [x21], #1
    mov w11, #' '
    strb w11, [x21], #1
    b slot_complete

empty_slot:
    mov w11, #' '
    strb w11, [x21], #1
    strb w11, [x21], #1
    strb w11, [x21], #1

slot_complete:
    cmp x24, #7
    b.ne no_middle_space
    mov w11, #' '
    strb w11, [x21], #1

no_middle_space:
    add x24, x24, #1
    b slot_loop

slots_done:
    mov w9, #' '
    strb w9, [x21], #1
    mov w9, #'|'
    strb w9, [x21], #1

    mov x24, #0

ascii_loop:
    cmp x24, x23
    b.hs ascii_done
    add x25, x22, x24
    ldrb w26, [x19, x25]
    cmp w26, #0x20
    b.lo ascii_dot
    cmp w26, #0x7e
    b.hi ascii_dot
    strb w26, [x21], #1
    b ascii_next

ascii_dot:
    mov w9, #'.'
    strb w9, [x21], #1

ascii_next:
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w9, #'|'
    strb w9, [x21], #1
    mov w9, #'\n'
    strb w9, [x21], #1
    add x22, x22, #16
    b row_loop

final_offset:
    mov x0, x20
    mov x1, x21
    bl emit_hex8
    mov x21, x1
    mov w9, #'\n'
    strb w9, [x21], #1

    mov x0, #1
    mov x1, x28
    sub x2, x21, x28
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

emit_hex8:
    mov x2, #28
    mov x3, #8

emit_hex8_loop:
    lsr x4, x0, x2
    and x4, x4, #15
    ldrb w5, [x27, x4]
    strb w5, [x1], #1
    sub x2, x2, #4
    subs x3, x3, #1
    b.ne emit_hex8_loop
    ret
