.text
.globl _start

_start:
    xor r12d, r12d
    lea rbx, [rip + input_buffer]

read_loop:
    cmp r12d, 65536
    je format_input
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    js read_loop
    jz format_input
    add r12, rax
    jmp read_loop

format_input:
    lea r8, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r15, [rip + hex_digits]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r10, r12
    sub r10, r13
    cmp r10, 16
    jbe row_size_ready
    mov r10d, 16

row_size_ready:
    lea r11, [r8 + r13]
    mov eax, r13d
    call emit_hex8
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor r9d, r9d

hex_slot_loop:
    cmp r9, r10
    jae empty_hex_slot

    movzx ebx, byte ptr [r11 + r9]
    mov eax, ebx
    shr eax, 4
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi], al
    mov eax, ebx
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    jmp hex_slot_done

empty_hex_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '

hex_slot_done:
    add rdi, 3
    inc r9
    cmp r9d, 8
    jne no_middle_space
    mov byte ptr [rdi], ' '
    inc rdi

no_middle_space:
    cmp r9d, 16
    jne hex_slot_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor r9d, r9d

ascii_loop:
    cmp r9, r10
    jae ascii_done
    movzx eax, byte ptr [r11 + r9]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_stored

ascii_dot:
    mov byte ptr [rdi], '.'

ascii_stored:
    inc rdi
    inc r9
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r10
    jmp row_loop

final_offset:
    mov eax, r12d
    call emit_hex8
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov ecx, 28

emit_hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns emit_hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 323593
