.equ INPUT_CAP, 65537

.section .text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r13]
    mov edx, INPUT_CAP
    sub rdx, r13
    syscall
    test rax, rax
    jz parse_input
    add r13, rax
    jmp read_loop

parse_input:
    xor r12d, r12d
    call skip_ws
    call parse_value
    jc report_error
    call skip_ws
    cmp r12, r13
    jne report_error

report_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, ok_message_len
    syscall
    xor edi, edi
    mov eax, 60
    syscall

report_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, error_prefix_len
    syscall

    lea r8, [rip + number_buffer + 31]
    mov byte ptr [r8], 10
    mov rax, r12
    mov r9d, 1
    test rax, rax
    jnz convert_number
    dec r8
    mov byte ptr [r8], '0'
    inc r9d
    jmp write_number

convert_number:
    mov ecx, 10
convert_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    inc r9
    test rax, rax
    jnz convert_loop

write_number:
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall

    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp r12, r13
    jae skip_ws_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_advance:
    inc r12
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_failure
    cmp al, '9'
    jbe parse_number
parse_failure:
    stc
    ret

parse_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4

parse_literal:
literal_loop:
    test ecx, ecx
    jz literal_success
    cmp r12, r13
    jae parse_failure
    mov al, byte ptr [rbx + r12]
    cmp al, byte ptr [rsi]
    jne parse_failure
    inc r12
    inc rsi
    dec ecx
    jmp literal_loop
literal_success:
    clc
    ret

parse_string:
    inc r12
string_loop:
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '"'
    je string_end
    cmp al, 0x20
    jb parse_failure
    cmp al, '\'
    je string_escape
    inc r12
    jmp string_loop

string_end:
    inc r12
    clc
    ret

string_escape:
    inc r12
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '"'
    je simple_escape
    cmp al, '\'
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    je unicode_escape
    jmp parse_failure

simple_escape:
    inc r12
    jmp string_loop

unicode_escape:
    inc r12
    mov ecx, 4
unicode_loop:
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb unicode_failure
    cmp al, '9'
    jbe unicode_digit
    cmp al, 'A'
    jb unicode_failure
    cmp al, 'F'
    jbe unicode_digit
    cmp al, 'a'
    jb unicode_failure
    cmp al, 'f'
    ja unicode_failure
unicode_digit:
    inc r12
    dec ecx
    jnz unicode_loop
    jmp string_loop
unicode_failure:
    jmp parse_failure

parse_number:
    cmp byte ptr [rbx + r12], '-'
    jne number_integer
    inc r12
    cmp r12, r13
    jae parse_failure

number_integer:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_failure
    cmp al, '9'
    ja parse_failure
    inc r12

number_integer_digits:
    cmp r12, r13
    jae number_success
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_after_integer
    cmp al, '9'
    ja number_after_integer
    inc r12
    jmp number_integer_digits

number_zero:
    inc r12

number_after_integer:
    cmp r12, r13
    jae number_success
    cmp byte ptr [rbx + r12], '.'
    je number_fraction
    mov al, byte ptr [rbx + r12]
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
number_success:
    clc
    ret

number_fraction:
    inc r12
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb parse_failure
    cmp al, '9'
    ja parse_failure

fraction_digits:
    inc r12
    cmp r12, r13
    jae number_success
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb fraction_done
    cmp al, '9'
    jbe fraction_digits
fraction_done:
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    clc
    ret

number_exponent:
    inc r12
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first_digit
exponent_sign:
    inc r12
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]

exponent_first_digit:
    cmp al, '0'
    jb parse_failure
    cmp al, '9'
    ja parse_failure

exponent_digits:
    inc r12
    cmp r12, r13
    jae number_success
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_success
    cmp al, '9'
    jbe exponent_digits
    clc
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae parse_failure
    cmp byte ptr [rbx + r12], ']'
    je array_empty_end

array_value:
    call parse_value
    jc parse_failure
    call skip_ws
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, ']'
    je array_end
    cmp al, ','
    jne parse_failure
    inc r12
    call skip_ws
    call parse_value
    jc parse_failure
    call skip_ws
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, ']'
    je array_end
    cmp al, ','
    jne parse_failure
    jmp array_comma_continue

array_comma_continue:
    inc r12
    call skip_ws
    jmp array_value

array_empty_end:
array_end:
    inc r12
    clc
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae parse_failure
    cmp byte ptr [rbx + r12], '}'
    je object_end

object_member:
    cmp r12, r13
    jae parse_failure
    cmp byte ptr [rbx + r12], '"'
    jne parse_failure
    call parse_string
    jc parse_failure
    call skip_ws
    cmp r12, r13
    jae parse_failure
    cmp byte ptr [rbx + r12], ':'
    jne parse_failure
    inc r12
    call skip_ws
    call parse_value
    jc parse_failure
    call skip_ws
    cmp r12, r13
    jae parse_failure
    movzx eax, byte ptr [rbx + r12]
    cmp al, '}'
    je object_end
    cmp al, ','
    jne parse_failure
    inc r12
    call skip_ws
    jmp object_member

object_end:
    inc r12
    clc
    ret

.section .rodata
ok_message:
    .ascii "ok\n"
ok_message_len = . - ok_message

error_prefix:
    .ascii "error at "
error_prefix_len = . - error_prefix

literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.balign 16
input_buffer:
    .zero INPUT_CAP
number_buffer:
    .zero 32
