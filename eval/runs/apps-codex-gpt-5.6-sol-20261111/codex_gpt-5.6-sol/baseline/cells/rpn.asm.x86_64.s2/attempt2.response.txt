.section .text
.globl _start

.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 400000
.equ STACK_SIZE, 8192

_start:
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    jae read_done
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    js read_loop
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + input_buffer]
    add r13, r12
    lea r12, [rip + input_buffer]
    mov r15, r12
    xor r14d, r14d

find_line:
    cmp r15, r13
    jae final_remainder
    cmp byte ptr [r15], 10
    je complete_line
    inc r15
    jmp find_line

complete_line:
    mov rdi, r12
    mov rsi, r15
    call evaluate_line
    lea r12, [r15 + 1]
    mov r15, r12
    jmp find_line

final_remainder:
    cmp r12, r13
    jae flush_output
    mov rdi, r12
    mov rsi, r13
    call evaluate_line

flush_output:
    xor r12d, r12d

write_loop:
    cmp r12, r14
    jae exit_ok
    lea rsi, [rip + output_buffer]
    add rsi, r12
    mov rdx, r14
    sub rdx, r12
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle write_loop
    add r12, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

evaluate_line:
    mov r11, rsi
    mov rsi, rdi
    xor ebx, ebx
    lea r8, [rip + value_stack]

token_skip:
    cmp rsi, r11
    jae line_finished
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    jmp token_start

skip_separator:
    inc rsi
    jmp token_skip

token_start:
    mov rdi, rsi

token_scan:
    cmp rsi, r11
    jae token_found
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc rsi
    jmp token_scan

token_found:
    mov rcx, rsi
    sub rcx, rdi
    cmp rcx, 1
    jne parse_literal

    movzx eax, byte ptr [rdi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

parse_literal:
    mov rax, rdi
    mov r9, rcx
    xor r10d, r10d

    cmp byte ptr [rax], '-'
    jne literal_length
    mov r10d, 1
    inc rax
    dec r9

literal_length:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error

    xor edi, edi
    xor ecx, ecx

digit_loop:
    cmp rcx, r9
    jae literal_done
    movzx edx, byte ptr [rax + rcx]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rdi, rdi, 10
    add rdi, rdx
    inc rcx
    jmp digit_loop

literal_done:
    test r10d, r10d
    jz literal_push
    neg rdi

literal_push:
    mov qword ptr [r8 + rbx*8], rdi
    inc rbx
    jmp token_skip

op_add:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    add rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_sub:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    sub rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_mul:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    imul rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_div:
    cmp rbx, 2
    jb line_error
    mov r9, qword ptr [r8 + rbx*8 - 8]
    test r9, r9
    jz line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne divide_value
    cmp r9, -1
    je line_error

divide_value:
    cqo
    idiv r9
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_mod:
    cmp rbx, 2
    jb line_error
    mov r9, qword ptr [r8 + rbx*8 - 8]
    test r9, r9
    jz line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne modulo_value
    cmp r9, -1
    je line_error

modulo_value:
    cqo
    idiv r9
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rdx
    jmp token_skip

line_finished:
    cmp rbx, 1
    jne line_error

    mov rax, qword ptr [r8]
    lea rdi, [rip + number_buffer + 32]
    xor r10d, r10d
    test rax, rax
    jns magnitude_ready
    mov r10d, 1
    neg rax

magnitude_ready:
    mov ecx, 10

convert_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz convert_loop

    lea r8, [rip + output_buffer]
    add r8, r14
    test r10d, r10d
    jz copy_number
    mov byte ptr [r8], '-'
    inc r8
    inc r14

copy_number:
    lea rcx, [rip + number_buffer + 32]
    sub rcx, rdi

copy_loop:
    test rcx, rcx
    jz append_newline
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc rdi
    inc r8
    inc r14
    dec rcx
    jmp copy_loop

append_newline:
    mov byte ptr [r8], 10
    inc r14
    ret

line_error:
    lea rdi, [rip + output_buffer]
    add rdi, r14
    mov byte ptr [rdi], 'e'
    mov byte ptr [rdi + 1], 'r'
    mov byte ptr [rdi + 2], 'r'
    mov byte ptr [rdi + 3], 'o'
    mov byte ptr [rdi + 4], 'r'
    mov byte ptr [rdi + 5], 10
    add r14, 6
    ret

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
value_stack:
    .zero STACK_SIZE
.align 16
number_buffer:
    .zero 32
.align 16
output_buffer:
    .zero OUTPUT_SIZE
