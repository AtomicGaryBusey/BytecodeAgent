.section .text
.globl _start

.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 400000

_start:
    xor r12d, r12d

read_more:
    cmp r12, INPUT_SIZE
    jae read_done
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    js read_more
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r12, [rip + input_buffer]
    lea r13, [r12 + r12]
    lea rax, [rip + input_buffer]
    add rax, r12
    mov r13, rax
    lea r12, [rip + input_buffer]
    xor r14d, r14d
    mov rbx, r12

find_line:
    cmp rbx, r13
    jae final_remainder
    mov al, byte ptr [rbx]
    cmp al, 10
    je complete_line
    inc rbx
    jmp find_line

complete_line:
    mov rsi, r12
    mov r15, rbx
    call evaluate_line
    lea r12, [rbx + 1]
    mov rbx, r12
    jmp find_line

final_remainder:
    cmp r12, r13
    jae flush_output
    mov rsi, r12
    mov r15, r13
    call evaluate_line

flush_output:
    test r14, r14
    jz exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

evaluate_line:
    xor ebx, ebx
    lea r8, [rip + value_stack]

token_skip:
    cmp rsi, r15
    jae line_finished
    mov al, byte ptr [rsi]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_start

skip_one:
    inc rsi
    jmp token_skip

token_start:
    mov rdi, rsi

token_scan:
    cmp rsi, r15
    jae token_found
    mov al, byte ptr [rsi]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc rsi
    jmp token_scan

token_found:
    mov rcx, rsi
    sub rcx, rdi
    cmp rcx, 1
    jne parse_literal

    mov al, byte ptr [rdi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

parse_literal:
    xor edx, edx
    mov rax, rdi
    mov rcx, rsi
    sub rcx, rdi
    mov r10, rcx
    test rcx, rcx
    jz line_error
    cmp byte ptr [rax], '-'
    jne literal_digits
    mov edx, 1
    inc rax
    dec r10

literal_digits:
    test r10, r10
    jz line_error
    cmp r10, 18
    ja line_error
    xor r9d, r9d
    xor ecx, ecx

digit_loop:
    cmp rcx, r10
    jae literal_done
    movzx edi, byte ptr [rax + rcx]
    sub edi, '0'
    cmp edi, 9
    ja line_error
    imul r9, r9, 10
    add r9, rdi
    inc rcx
    jmp digit_loop

literal_done:
    test edx, edx
    jz literal_push
    neg r9

literal_push:
    mov qword ptr [r8 + rbx*8], r9
    inc rbx
    jmp token_skip

op_add:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    add rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_sub:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    sub rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_mul:
    cmp rbx, 2
    jb line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    imul rax, qword ptr [r8 + rbx*8 - 8]
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_div:
    cmp rbx, 2
    jb line_error
    mov r9, qword ptr [r8 + rbx*8 - 8]
    test r9, r9
    jz line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne do_div
    cmp r9, -1
    je line_error

do_div:
    cqo
    idiv r9
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rax
    jmp token_skip

op_mod:
    cmp rbx, 2
    jb line_error
    mov r9, qword ptr [r8 + rbx*8 - 8]
    test r9, r9
    jz line_error
    mov rax, qword ptr [r8 + rbx*8 - 16]
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne do_mod
    cmp r9, -1
    je line_error

do_mod:
    cqo
    idiv r9
    dec rbx
    mov qword ptr [r8 + rbx*8 - 8], rdx
    jmp token_skip

line_finished:
    cmp rbx, 1
    jne line_error
    mov rax, qword ptr [r8]
    lea rdi, [rip + number_buffer + 32]
    xor r10d, r10d
    test rax, rax
    jns magnitude_ready
    mov r10d, 1
    neg rax

magnitude_ready:
    mov ecx, 10

convert_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz convert_loop

    lea r8, [rip + output_buffer]
    add r8, r14
    test r10d, r10d
    jz copy_number
    mov byte ptr [r8], '-'
    inc r8
    inc r14

copy_number:
    lea rcx, [rip + number_buffer + 32]
    sub rcx, rdi

copy_number_loop:
    test rcx, rcx
    jz number_newline
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc rdi
    inc r8
    inc r14
    dec rcx
    jmp copy_number_loop

number_newline:
    mov byte ptr [r8], 10
    inc r14
    ret

line_error:
    lea rdi, [rip + output_buffer]
    add rdi, r14
    mov byte ptr [rdi], 'e'
    mov byte ptr [rdi + 1], 'r'
    mov byte ptr [rdi + 2], 'r'
    mov byte ptr [rdi + 3], 'o'
    mov byte ptr [rdi + 4], 'r'
    mov byte ptr [rdi + 5], 10
    add r14, 6
    ret

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
value_stack:
    .zero 8192
.align 16
number_buffer:
    .zero 32
.align 16
output_buffer:
    .zero OUTPUT_SIZE
