.equ INPUT_CAP, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_DEPTH, 1000
.equ OUTPUT_CAP, 65536

.section .text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r15d, r15d

read_loop:
    cmp r15, INPUT_CAP
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r15]
    mov edx, INPUT_CAP
    sub rdx, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop

read_done:
    mov r13, r15
    mov r14, r15
    xor r9d, r9d

find_delimiter:
    cmp r9, r15
    jae validate_start
    cmp byte ptr [r12 + r9], 0x21
    je delimiter_found
    inc r9
    jmp find_delimiter

delimiter_found:
    mov r13, r9
    lea r14, [r9 + 1]

validate_start:
    lea rdi, [rip + bracket_stack]
    lea r11, [rip + jump_table]
    xor r8d, r8d
    xor r9d, r9d

validate_loop:
    cmp r9, r13
    jae validate_finish
    movzx eax, byte ptr [r12 + r9]
    cmp al, '['
    je validate_open
    cmp al, ']'
    je validate_close
    inc r9
    jmp validate_loop

validate_open:
    cmp r8, BRACKET_DEPTH
    jae malformed
    mov qword ptr [rdi + r8*8], r9
    inc r8
    inc r9
    jmp validate_loop

validate_close:
    test r8, r8
    jz malformed
    dec r8
    mov rax, qword ptr [rdi + r8*8]
    mov qword ptr [r11 + rax*8], r9
    mov qword ptr [r11 + r9*8], rax
    inc r9
    jmp validate_loop

validate_finish:
    test r8, r8
    jnz malformed

    lea rbp, [rip + tape]
    lea r8, [rip + output_buffer]
    xor ebx, ebx
    xor r10d, r10d

execute_loop:
    cmp rbx, r13
    jae execute_done
    movzx eax, byte ptr [r12 + rbx]

    cmp al, '>'
    je command_right
    cmp al, '<'
    je command_left
    cmp al, '+'
    je command_inc
    cmp al, '-'
    je command_dec
    cmp al, '.'
    je command_output
    cmp al, ','
    je command_input
    cmp al, '['
    je command_open
    cmp al, ']'
    je command_close

    inc rbx
    jmp execute_loop

command_right:
    inc rbp
    inc rbx
    jmp execute_loop

command_left:
    dec rbp
    inc rbx
    jmp execute_loop

command_inc:
    inc byte ptr [rbp]
    inc rbx
    jmp execute_loop

command_dec:
    dec byte ptr [rbp]
    inc rbx
    jmp execute_loop

command_output:
    mov al, byte ptr [rbp]
    mov byte ptr [r8 + r10], al
    inc r10
    inc rbx
    jmp execute_loop

command_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r12 + r14]
    mov byte ptr [rbp], al
    inc r14
    inc rbx
    jmp execute_loop

input_exhausted:
    mov byte ptr [rbp], 0
    inc rbx
    jmp execute_loop

command_open:
    cmp byte ptr [rbp], 0
    jne open_continue
    mov rbx, qword ptr [r11 + rbx*8]
open_continue:
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [rbp], 0
    je close_continue
    mov rbx, qword ptr [r11 + rbx*8]
close_continue:
    inc rbx
    jmp execute_loop

execute_done:
    xor ebx, ebx

write_loop:
    cmp rbx, r10
    jae success
    mov eax, 1
    mov edi, 1
    lea rsi, [r8 + rbx]
    mov rdx, r10
    sub rdx, rbx
    syscall
    test rax, rax
    jle success
    add rbx, rax
    jmp write_loop

success:
    mov eax, 60
    xor edi, edi
    syscall

malformed:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero INPUT_CAP
.balign 16
jump_table:
    .zero 524288
.balign 16
bracket_stack:
    .zero 8000
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
output_buffer:
    .zero OUTPUT_CAP
