.section .text
.globl _start

_start:
    lea r12, [rip + input_buffer]
    mov r13, r12

read_loop:
    mov rdx, 65536
    sub rdx, r13
    add rdx, r12
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    mov rbx, r12
    call skip_ws
    cmp rbx, r13
    je parse_error
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne parse_error

parse_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall

    mov eax, 60
    xor edi, edi
    syscall

parse_error:
    mov r14, rbx
    sub r14, r12

    lea r15, [rip + output_buffer + 31]
    mov byte ptr [r15], 10
    mov rcx, r15
    mov rax, r14
    mov r8, 10

    test rax, rax
    jne decimal_loop
    dec rcx
    mov byte ptr [rcx], '0'
    jmp decimal_done

decimal_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jne decimal_loop

decimal_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, 9
    syscall

    mov rdx, r15
    inc rdx
    sub rdx, rcx
    mov eax, 1
    mov edi, 1
    mov rsi, rcx
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp rbx, r13
    je skip_ws_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    je skip_ws_advance
skip_ws_done:
    ret
skip_ws_advance:
    inc rbx
    jmp skip_ws

parse_value:
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_words
    cmp al, '9'
    jbe parse_number

parse_value_words:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp parse_error

parse_true:
    lea rsi, [rip + true_word]
    mov ecx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + false_word]
    mov ecx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + null_word]
    mov ecx, 4

parse_literal:
    test ecx, ecx
    jz parse_literal_done
    cmp rbx, r13
    je parse_error
    mov al, byte ptr [rbx]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc rbx
    inc rsi
    dec ecx
    jmp parse_literal
parse_literal_done:
    ret

parse_string:
    inc rbx
string_loop:
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je string_done
    cmp al, 0x20
    jb parse_error
    cmp al, '\'
    je string_escape
    inc rbx
    jmp string_loop

string_done:
    inc rbx
    ret

string_escape:
    inc rbx
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je simple_escape
    cmp al, '\'
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    je unicode_escape
    jmp parse_error

simple_escape:
    inc rbx
    jmp string_loop

unicode_escape:
    inc rbx
    mov ecx, 4
unicode_loop:
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb unicode_upper
    cmp al, '9'
    jbe unicode_digit
unicode_upper:
    cmp al, 'A'
    jb unicode_lower
    cmp al, 'F'
    jbe unicode_digit
unicode_lower:
    cmp al, 'a'
    jb parse_error
    cmp al, 'f'
    ja parse_error
unicode_digit:
    inc rbx
    dec ecx
    jne unicode_loop
    jmp string_loop

parse_number:
    cmp byte ptr [rbx], '-'
    jne number_integer
    inc rbx
    cmp rbx, r13
    je parse_error

number_integer:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error
    inc rbx

number_digits:
    cmp rbx, r13
    je number_after_integer
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb number_after_integer
    cmp al, '9'
    ja number_after_integer
    inc rbx
    jmp number_digits

number_zero:
    inc rbx

number_after_integer:
    cmp rbx, r13
    je number_done
    cmp byte ptr [rbx], '.'
    jne number_exponent_check
    inc rbx
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

fraction_digits:
    inc rbx
    cmp rbx, r13
    je number_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb number_exponent_check
    cmp al, '9'
    jbe fraction_digits

number_exponent_check:
    cmp rbx, r13
    je number_done
    movzx eax, byte ptr [rbx]
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    jne number_done

number_exponent:
    inc rbx
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first_digit

exponent_sign:
    inc rbx
    cmp rbx, r13
    je parse_error

exponent_first_digit:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error

exponent_digits:
    inc rbx
    cmp rbx, r13
    je number_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits

number_done:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    je parse_error
    cmp byte ptr [rbx], ']'
    je array_empty_done

array_value:
    call parse_value
    call skip_ws
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne parse_error
    inc rbx
    call skip_ws
    cmp rbx, r13
    je parse_error
    cmp byte ptr [rbx], ']'
    je parse_error
    jmp array_value

array_empty_done:
array_done:
    inc rbx
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    je parse_error
    cmp byte ptr [rbx], '}'
    je object_empty_done

object_member:
    cmp byte ptr [rbx], '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp rbx, r13
    je parse_error
    cmp byte ptr [rbx], ':'
    jne parse_error
    inc rbx
    call skip_ws
    cmp rbx, r13
    je parse_error
    call parse_value
    call skip_ws
    cmp rbx, r13
    je parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne parse_error
    inc rbx
    call skip_ws
    cmp rbx, r13
    je parse_error
    cmp byte ptr [rbx], '}'
    je parse_error
    jmp object_member

object_empty_done:
object_done:
    inc rbx
    ret

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
true_word:
    .ascii "true"
false_word:
    .ascii "false"
null_word:
    .ascii "null"

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 32
