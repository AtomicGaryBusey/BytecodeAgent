.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x1, x19
    add x1, x1, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf

    mov x26, x19
    mov x27, x19
    add x28, x19, x20

scan_lines:
    cmp x26, x28
    b.hs scan_eof
    ldrb w9, [x26], #1
    cmp w9, #10
    b.ne scan_lines

    mov x0, x27
    sub x1, x26, #1
    bl process_line
    mov x27, x26
    b scan_lines

scan_eof:
    cmp x27, x28
    b.eq output_all
    mov x0, x27
    mov x1, x28
    bl process_line

output_all:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x23, x1
    cbz x2, exit_ok
    mov x0, #1
    mov x8, #SYS_write
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

process_line:
    adrp x4, value_stack
    add x4, x4, :lo12:value_stack
    mov x2, x0
    mov x3, #0

token_skip:
    cmp x2, x1
    b.hs line_finish
    ldrb w7, [x2]
    cmp w7, #32
    b.eq skip_one
    cmp w7, #9
    b.eq skip_one
    b token_start

skip_one:
    add x2, x2, #1
    b token_skip

token_start:
    mov x5, x2

token_scan:
    cmp x2, x1
    b.hs token_end
    ldrb w7, [x2]
    cmp w7, #32
    b.eq token_end
    cmp w7, #9
    b.eq token_end
    add x2, x2, #1
    b token_scan

token_end:
    sub x6, x2, x5
    cmp x6, #1
    b.ne parse_literal

    ldrb w7, [x5]
    cmp w7, #43
    b.eq apply_operator
    cmp w7, #45
    b.eq apply_operator
    cmp w7, #42
    b.eq apply_operator
    cmp w7, #47
    b.eq apply_operator
    cmp w7, #37
    b.eq apply_operator

parse_literal:
    mov x10, x5
    mov x8, #0
    ldrb w7, [x10]
    cmp w7, #45
    b.ne literal_digits
    mov x8, #1
    add x10, x10, #1
    sub x11, x2, x10
    cbz x11, line_error
    b literal_length

literal_digits:
    mov x11, x6

literal_length:
    cmp x11, #18
    b.hi line_error
    cbz x11, line_error
    mov x9, #0

literal_loop:
    ldrb w7, [x10], #1
    cmp w7, #48
    b.lo line_error
    cmp w7, #57
    b.hi line_error
    sub w7, w7, #48
    mov x12, #10
    mul x9, x9, x12
    add x9, x9, x7
    subs x11, x11, #1
    b.ne literal_loop

    cbz x8, literal_push
    neg x9, x9

literal_push:
    str x9, [x4, x3, lsl #3]
    add x3, x3, #1
    b token_skip

apply_operator:
    cmp x3, #2
    b.lo line_error

    sub x10, x3, #1
    ldr x11, [x4, x10, lsl #3]
    sub x10, x10, #1
    ldr x12, [x4, x10, lsl #3]

    cmp w7, #43
    b.eq op_add
    cmp w7, #45
    b.eq op_sub
    cmp w7, #42
    b.eq op_mul
    cmp w7, #47
    b.eq op_div
    b op_mod

op_add:
    add x13, x12, x11
    b op_store

op_sub:
    sub x13, x12, x11
    b op_store

op_mul:
    mul x13, x12, x11
    b op_store

op_div:
    cbz x11, line_error
    movz x14, #0x8000, lsl #48
    cmp x12, x14
    b.ne do_div
    cmn x11, #1
    b.eq line_error
do_div:
    sdiv x13, x12, x11
    b op_store

op_mod:
    cbz x11, line_error
    movz x14, #0x8000, lsl #48
    cmp x12, x14
    b.ne do_mod
    cmn x11, #1
    b.eq line_error
do_mod:
    sdiv x13, x12, x11
    msub x13, x13, x11, x12

op_store:
    str x13, [x4, x10, lsl #3]
    sub x3, x3, #1
    b token_skip

line_finish:
    cmp x3, #1
    b.ne line_error
    ldr x9, [x4]
    adrp x10, number_buf
    add x10, x10, :lo12:number_buf
    add x11, x10, #32
    mov x12, x11
    mov x13, x9
    mov x14, #0

    tbz x9, #63, magnitude_ready
    mov x14, #1
    neg x13, x9

magnitude_ready:
    cbnz x13, number_loop
    sub x12, x12, #1
    mov w15, #48
    strb w15, [x12]
    b number_sign

number_loop:
    mov x15, #10
    udiv x16, x13, x15
    msub x17, x16, x15, x13
    add w17, w17, #48
    sub x12, x12, #1
    strb w17, [x12]
    mov x13, x16
    cbnz x13, number_loop

number_sign:
    cbz x14, number_copy
    sub x12, x12, #1
    mov w15, #45
    strb w15, [x12]

number_copy:
    cmp x12, x11
    b.hs number_newline
    ldrb w15, [x12], #1
    strb w15, [x23], #1
    b number_copy

number_newline:
    mov w15, #10
    strb w15, [x23], #1
    ret

line_error:
    movz w9, #0x7265
    movk w9, #0x6f72, lsl #16
    str w9, [x23], #4
    movz w9, #0x0a72
    strh w9, [x23], #2
    ret

.bss
.balign 16
input_buf:
    .space 65536
.balign 16
value_stack:
    .space 8192
.balign 16
number_buf:
    .space 32
.balign 16
output_buf:
    .space 400000
