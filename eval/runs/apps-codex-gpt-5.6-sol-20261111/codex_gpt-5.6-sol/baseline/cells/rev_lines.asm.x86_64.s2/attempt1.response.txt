.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 65537

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output_buffer]
    xor r8d, r8d
    xor ecx, ecx
    xor r13d, r13d

scan_loop:
    cmp rcx, r12
    jae scan_done
    cmp byte ptr [rbx + rcx], 10
    je complete_line
    inc rcx
    jmp scan_loop

complete_line:
    mov rdx, rcx

reverse_line:
    cmp rdx, r8
    jbe append_newline
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rdi + r13], al
    inc r13
    jmp reverse_line

append_newline:
    mov byte ptr [rdi + r13], 10
    inc r13
    inc rcx
    mov r8, rcx
    jmp scan_loop

scan_done:
    cmp r8, r12
    jae write_output
    mov rdx, r12

reverse_final:
    cmp rdx, r8
    jbe append_final_newline
    dec rdx
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rdi + r13], al
    inc r13
    jmp reverse_final

append_final_newline:
    mov byte ptr [rdi + r13], 10
    inc r13

write_output:
    xor r12d, r12d

write_loop:
    cmp r12, r13
    jae exit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    add rsi, r12
    mov rdx, r13
    sub rdx, r12
    syscall
    add r12, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero OUTPUT_SIZE
