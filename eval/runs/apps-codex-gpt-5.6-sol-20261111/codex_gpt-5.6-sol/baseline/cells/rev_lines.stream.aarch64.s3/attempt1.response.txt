.equ INPUT_CAP, 65536
.equ OUTPUT_CAP, 65537
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #INPUT_CAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x23, x20
    b.hs scan_done
    ldrb w9, [x19, x23]
    cmp w9, #10
    b.eq line_found
    add x23, x23, #1
    b scan_loop

line_found:
    mov x25, x23

reverse_line:
    cmp x25, x24
    b.ls append_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x21, x22]
    add x22, x22, #1
    b reverse_line

append_newline:
    mov w9, #10
    strb w9, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    mov x24, x23
    b scan_loop

scan_done:
    cmp x24, x20
    b.hs write_output
    mov x25, x20

reverse_final:
    cmp x25, x24
    b.ls append_final_newline
    sub x25, x25, #1
    ldrb w9, [x19, x25]
    strb w9, [x21, x22]
    add x22, x22, #1
    b reverse_final

append_final_newline:
    mov w9, #10
    strb w9, [x21, x22]
    add x22, x22, #1

write_output:
    cbz x22, exit_success
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #SYS_WRITE
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
input_buffer:
    .zero INPUT_CAP
.balign 16
output_buffer:
    .zero OUTPUT_CAP
