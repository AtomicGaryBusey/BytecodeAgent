.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_MAX, 32768

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, parse_input
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, parse_input
    add x20, x20, x0
    b read_loop

parse_input:
    adrp x22, line_table
    add x22, x22, :lo12:line_table
    mov x23, #0
    mov x24, #0
    mov x25, #0

parse_loop:
    cmp x25, x20
    b.hs parse_remainder
    ldrb w9, [x19, x25]
    cmp w9, #10
    b.ne parse_next

    add x9, x19, x24
    sub x10, x25, x24
    add x11, x22, x23, lsl #4
    stp x9, x10, [x11]
    add x23, x23, #1
    add x24, x25, #1

parse_next:
    add x25, x25, #1
    b parse_loop

parse_remainder:
    cmp x24, x20
    b.hs sort_begin
    add x9, x19, x24
    sub x10, x20, x24
    add x11, x22, x23, lsl #4
    stp x9, x10, [x11]
    add x23, x23, #1

sort_begin:
    mov x24, #1

sort_outer:
    cmp x24, x23
    b.hs output_begin
    add x12, x22, x24, lsl #4
    ldp x20, x21, [x12]
    mov x25, x24

sort_compare:
    cbz x25, sort_insert
    sub x26, x25, #1
    add x12, x22, x26, lsl #4
    ldp x9, x10, [x12]

    cmp x10, x21
    csel x11, x10, x21, ls
    mov x13, #0

compare_bytes:
    cmp x13, x11
    b.hs compare_lengths
    ldrb w14, [x9, x13]
    ldrb w15, [x20, x13]
    cmp w14, w15
    b.lo sort_insert
    b.hi sort_shift
    add x13, x13, #1
    b compare_bytes

compare_lengths:
    cmp x10, x21
    b.ls sort_insert

sort_shift:
    add x12, x22, x25, lsl #4
    stp x9, x10, [x12]
    sub x25, x25, #1
    b sort_compare

sort_insert:
    add x12, x22, x25, lsl #4
    stp x20, x21, [x12]
    add x24, x24, #1
    b sort_outer

output_begin:
    mov x24, #0

output_loop:
    cmp x24, x23
    b.hs exit_success
    add x12, x22, x24, lsl #4
    ldp x20, x21, [x12]
    cbz x21, output_newline

    mov x0, #1
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_write
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    add x24, x24, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero 32769
.balign 16
line_table:
    .zero 16000
