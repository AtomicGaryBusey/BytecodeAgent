.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_MAX, 65536
.equ OUTPUT_MAX, 65537

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x24, #INPUT_MAX

read_loop:
    cmp x20, x24
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x24, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x18, output_buf
    add x18, x18, :lo12:output_buf
    mov x21, #0
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x21, x20
    b.eq scan_done
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.ne scan_next

    mov x25, x21

reverse_line:
    cmp x25, x22
    b.eq append_newline
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x18, x23]
    add x23, x23, #1
    b reverse_line

append_newline:
    mov w0, #10
    strb w0, [x18, x23]
    add x23, x23, #1
    add x22, x21, #1

scan_next:
    add x21, x21, #1
    b scan_loop

scan_done:
    cmp x22, x20
    b.eq write_output
    mov x25, x20

reverse_final:
    cmp x25, x22
    b.eq append_final_newline
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x18, x23]
    add x23, x23, #1
    b reverse_final

append_final_newline:
    mov w0, #10
    strb w0, [x18, x23]
    add x23, x23, #1

write_output:
    cbz x23, exit_success
    mov x0, #1
    mov x1, x18
    mov x2, x23
    mov x8, #SYS_write
    svc #0

exit_success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
output_buf:
    .zero OUTPUT_MAX
