.section .rodata
ok_msg:
.ascii "ok\n"
ok_len = . - ok_msg
err_msg:
.ascii "error at "
err_len = . - err_msg

.bss
.align 16
input_buf:
.space 65537
output_buf:
.space 32

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r14d, r14d

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r14]
    mov edx, 65537
    sub rdx, r14
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop

read_done:
    lea r12, [rip + input_buf]
    lea r13, [r12 + r14]
    call skip_ws
    cmp r12, r13
    je parse_fail
    call parse_value
    call skip_ws
    cmp r12, r13
    jne parse_fail

parse_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

parse_fail:
    lea rax, [rip + input_buf]
    sub r12, rax
    mov r15, r12

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, err_len
    syscall

    lea rsi, [rip + output_buf + 31]
    mov byte ptr [rsi], 10
    mov rcx, 1
    mov rax, r15
    mov r8d, 10

decimal_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc rcx
    test rax, rax
    jne decimal_loop

    mov eax, 1
    mov edi, 1
    mov rdx, rcx
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
skip_ws_loop:
    cmp r12, r13
    je skip_ws_done
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je skip_ws_take
    cmp al, 0x09
    je skip_ws_take
    cmp al, 0x0a
    je skip_ws_take
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_take:
    inc r12
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    jbe parse_number
    jmp parse_fail

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4

parse_literal:
parse_literal_loop:
    test ecx, ecx
    je parse_literal_done
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, byte ptr [rsi]
    jne parse_fail
    inc r12
    inc rsi
    dec ecx
    jmp parse_literal_loop
parse_literal_done:
    ret

parse_string:
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '"'
    jne parse_fail
    inc r12

string_loop:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_done
    cmp al, 0x20
    jb parse_fail
    cmp al, '\\'
    je string_escape
    inc r12
    jmp string_loop

string_escape:
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_escape_one
    cmp al, '\\'
    je string_escape_one
    cmp al, '/'
    je string_escape_one
    cmp al, 'b'
    je string_escape_one
    cmp al, 'f'
    je string_escape_one
    cmp al, 'n'
    je string_escape_one
    cmp al, 'r'
    je string_escape_one
    cmp al, 't'
    je string_escape_one
    cmp al, 'u'
    je string_unicode
    jmp parse_fail

string_escape_one:
    inc r12
    jmp string_loop

string_unicode:
    inc r12
    mov ecx, 4
string_hex_loop:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    jbe string_hex_ok
    cmp al, 'A'
    jb parse_fail
    cmp al, 'F'
    jbe string_hex_ok
    cmp al, 'a'
    jb parse_fail
    cmp al, 'f'
    ja parse_fail
string_hex_ok:
    inc r12
    dec ecx
    jne string_hex_loop
    jmp string_loop

string_done:
    inc r12
    ret

parse_number:
    cmp byte ptr [r12], '-'
    jne number_integer
    inc r12
    cmp r12, r13
    je parse_fail

number_integer:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_fail
    cmp al, '9'
    ja parse_fail

number_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_after_integer
    cmp al, '9'
    jbe number_digits
    jmp number_after_integer

number_zero:
    inc r12

number_after_integer:
    cmp r12, r13
    je number_done
    cmp byte ptr [r12], '.'
    jne number_check_exp
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail

number_fraction_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_check_exp
    cmp al, '9'
    jbe number_fraction_digits

number_check_exp:
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    jne number_done

number_exp:
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    je number_exp_sign
    jmp number_exp_first

number_exp_sign:
    inc r12
    cmp r12, r13
    je parse_fail

number_exp_first:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail

number_exp_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe number_exp_digits

number_done:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je array_empty

array_value:
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    jmp array_value

array_empty:
    inc r12
    ret

array_done:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je object_empty

object_member:
    call parse_string
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ':'
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    jmp object_member

object_empty:
    inc r12
    ret

object_done:
    inc r12
    ret

.section .rodata
lit_true:
.ascii "true"
lit_false:
.ascii "false"
lit_null:
.ascii "null"
