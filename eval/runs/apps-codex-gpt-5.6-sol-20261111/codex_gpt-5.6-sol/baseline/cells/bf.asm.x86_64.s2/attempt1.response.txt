.text
.globl _start
_start:
    xor r8d, r8d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [input + r8]
    mov edx, 65537
    sub rdx, r8
    syscall
    test rax, rax
    jle read_done
    add r8, rax
    jmp read_loop

read_done:
    mov r12, r8
    lea r15, [input + r8]
    mov r14, r15
    xor ebx, ebx

find_separator:
    cmp rbx, r8
    jae validate_start
    cmp byte ptr [input + rbx], 0x21
    je separator_found
    inc rbx
    jmp find_separator

separator_found:
    mov r12, rbx
    lea r14, [input + rbx + 1]

validate_start:
    xor ebx, ebx
    xor r9d, r9d

validate_loop:
    cmp rbx, r12
    jae validate_done
    movzx eax, byte ptr [input + rbx]
    cmp al, 0x5b
    je validate_open
    cmp al, 0x5d
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    cmp r9, 1000
    jae error_exit
    mov dword ptr [brstack + r9*4], ebx
    inc r9
    inc rbx
    jmp validate_loop

validate_close:
    test r9, r9
    jz error_exit
    dec r9
    mov eax, dword ptr [brstack + r9*4]
    mov dword ptr [brmap + rbx*4], eax
    mov dword ptr [brmap + rax*4], ebx
    inc rbx
    jmp validate_loop

validate_done:
    test r9, r9
    jnz error_exit

    xor ebx, ebx
    lea r13, [rip + tape]
    xor ebp, ebp

execute_loop:
    cmp rbx, r12
    jae execute_done
    movzx eax, byte ptr [input + rbx]

    cmp al, 0x3e
    je command_right
    cmp al, 0x3c
    je command_left
    cmp al, 0x2b
    je command_plus
    cmp al, 0x2d
    je command_minus
    cmp al, 0x2e
    je command_output
    cmp al, 0x2c
    je command_input
    cmp al, 0x5b
    je command_open
    cmp al, 0x5d
    je command_close

    inc rbx
    jmp execute_loop

command_right:
    inc r13
    inc rbx
    jmp execute_loop

command_left:
    dec r13
    inc rbx
    jmp execute_loop

command_plus:
    inc byte ptr [r13]
    inc rbx
    jmp execute_loop

command_minus:
    dec byte ptr [r13]
    inc rbx
    jmp execute_loop

command_output:
    mov al, byte ptr [r13]
    mov byte ptr [output + rbp], al
    inc rbp
    inc rbx
    jmp execute_loop

command_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    mov byte ptr [r13], al
    inc r14
    inc rbx
    jmp execute_loop

input_exhausted:
    mov byte ptr [r13], 0
    inc rbx
    jmp execute_loop

command_open:
    cmp byte ptr [r13], 0
    jne command_advance
    mov ebx, dword ptr [brmap + rbx*4]
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r13], 0
    je command_advance
    mov ebx, dword ptr [brmap + rbx*4]
    inc rbx
    jmp execute_loop

command_advance:
    inc rbx
    jmp execute_loop

execute_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, rbp
    syscall

    mov eax, 60
    xor edi, edi
    syscall

error_exit:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input:
    .zero 65537
.balign 16
brmap:
    .zero 262144
.balign 16
brstack:
    .zero 4000
.balign 16
tape:
    .zero 30000
.balign 16
output:
    .zero 65536
