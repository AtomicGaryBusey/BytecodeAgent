.equ INPUT_SIZE, 65536
.equ VALUE_STACK_SIZE, 4096
.equ OUTPUT_SIZE, 400000

.section .text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r15d, r15d

read_loop:
    mov edx, INPUT_SIZE
    sub rdx, r15
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r15]
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop

read_done:
    lea r13, [r12 + r15]
    lea r14, [rip + output_buffer]

line_loop:
    cmp r12, r13
    jae output_all

    mov rsi, r12
    mov rdi, r12

find_line_end:
    cmp rdi, r13
    jae final_line
    cmp byte ptr [rdi], 10
    je newline_line
    inc rdi
    jmp find_line_end

newline_line:
    call eval_line
    lea r12, [rdi + 1]
    jmp line_loop

final_line:
    call eval_line
    mov r12, r13
    jmp output_all

output_all:
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    mov r8, rsi
    mov r9, rdi
    xor r10d, r10d
    lea rbp, [rip + value_stack]

token_next:
    cmp r8, r9
    jae line_finished

skip_separators:
    cmp r8, r9
    jae line_finished
    movzx eax, byte ptr [r8]
    cmp al, 32
    je skip_one
    cmp al, 9
    jne token_start

skip_one:
    inc r8
    jmp skip_separators

token_start:
    mov r11, r8

find_token_end:
    cmp r11, r9
    jae token_found
    movzx eax, byte ptr [r11]
    cmp al, 32
    je token_found
    cmp al, 9
    je token_found
    inc r11
    jmp find_token_end

token_found:
    mov rcx, r11
    sub rcx, r8
    cmp rcx, 1
    jne parse_literal

    movzx eax, byte ptr [r8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

parse_literal:
    mov rcx, r8
    xor ebx, ebx
    cmp byte ptr [rcx], '-'
    jne literal_digits
    mov ebx, 1
    inc rcx

literal_digits:
    cmp rcx, r11
    jae line_error

    mov rdx, r11
    sub rdx, rcx
    cmp rdx, 18
    ja line_error

    xor eax, eax

literal_loop:
    cmp rcx, r11
    jae literal_done
    movzx edx, byte ptr [rcx]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    jmp literal_loop

literal_done:
    test ebx, ebx
    jz literal_push
    neg rax

literal_push:
    mov qword ptr [rbp + r10*8], rax
    inc r10
    mov r8, r11
    jmp token_next

op_add:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [rbp + r10*8 - 16]
    add rax, qword ptr [rbp + r10*8 - 8]
    dec r10
    mov qword ptr [rbp + r10*8 - 8], rax
    mov r8, r11
    jmp token_next

op_sub:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [rbp + r10*8 - 16]
    sub rax, qword ptr [rbp + r10*8 - 8]
    dec r10
    mov qword ptr [rbp + r10*8 - 8], rax
    mov r8, r11
    jmp token_next

op_mul:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [rbp + r10*8 - 16]
    imul rax, qword ptr [rbp + r10*8 - 8]
    dec r10
    mov qword ptr [rbp + r10*8 - 8], rax
    mov r8, r11
    jmp token_next

op_div:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [rbp + r10*8 - 8]
    test rbx, rbx
    jz line_error
    mov rax, qword ptr [rbp + r10*8 - 16]
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne division_safe
    cmp rbx, -1
    je line_error

division_safe:
    cqo
    idiv rbx
    dec r10
    mov qword ptr [rbp + r10*8 - 8], rax
    mov r8, r11
    jmp token_next

op_mod:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [rbp + r10*8 - 8]
    test rbx, rbx
    jz line_error
    mov rax, qword ptr [rbp + r10*8 - 16]
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne modulo_safe
    cmp rbx, -1
    je line_error

modulo_safe:
    cqo
    idiv rbx
    dec r10
    mov qword ptr [rbp + r10*8 - 8], rdx
    mov r8, r11
    jmp token_next

line_finished:
    cmp r10, 1
    jne line_error

    mov rax, qword ptr [rbp]
    test rax, rax
    jns magnitude_ready
    mov byte ptr [r14], '-'
    inc r14
    neg rax

magnitude_ready:
    mov rsi, r14
    mov ecx, 10

digit_output_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [r14], dl
    inc r14
    test rax, rax
    jnz digit_output_loop

    mov rax, rsi
    lea rdx, [r14 - 1]

reverse_digits:
    cmp rax, rdx
    jae number_done
    mov cl, byte ptr [rax]
    mov bl, byte ptr [rdx]
    mov byte ptr [rax], bl
    mov byte ptr [rdx], cl
    inc rax
    dec rdx
    jmp reverse_digits

number_done:
    mov byte ptr [r14], 10
    inc r14
    ret

line_error:
    mov byte ptr [r14], 'e'
    mov byte ptr [r14 + 1], 'r'
    mov byte ptr [r14 + 2], 'r'
    mov byte ptr [r14 + 3], 'o'
    mov byte ptr [r14 + 4], 'r'
    mov byte ptr [r14 + 5], 10
    add r14, 6
    ret

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE
.align 16
value_stack:
    .zero VALUE_STACK_SIZE
.align 16
output_buffer:
    .zero OUTPUT_SIZE
