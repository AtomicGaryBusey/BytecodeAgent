.equ INPUT_MAX, 32768
.equ LINE_MAX, 1000
.equ DESC_SIZE, 16

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    mov edx, INPUT_MAX
    sub rdx, r13
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + line_table]
    xor r15d, r15d
    xor ebx, ebx
    xor ecx, ecx

parse_loop:
    cmp rcx, r13
    jae parse_remainder
    cmp byte ptr [r12 + rcx], 10
    jne parse_next

    mov rax, r15
    shl rax, 4
    lea rdx, [r12 + rbx]
    mov qword ptr [r14 + rax], rdx
    mov rdx, rcx
    sub rdx, rbx
    mov qword ptr [r14 + rax + 8], rdx
    inc r15
    lea rbx, [rcx + 1]

parse_next:
    inc rcx
    jmp parse_loop

parse_remainder:
    cmp rbx, r13
    jae parse_done
    mov rax, r15
    shl rax, 4
    lea rdx, [r12 + rbx]
    mov qword ptr [r14 + rax], rdx
    mov rdx, r13
    sub rdx, rbx
    mov qword ptr [r14 + rax + 8], rdx
    inc r15

parse_done:
    cmp r15, 1
    jbe output_begin

    mov rbx, r15
    shr rbx, 1
    test rbx, rbx
    jz heap_extract
    dec rbx

heap_build_loop:
    mov rdi, rbx
    mov rsi, r15
    call sift_down
    test rbx, rbx
    jz heap_extract
    dec rbx
    jmp heap_build_loop

heap_extract:
    mov rbx, r15
    dec rbx

heap_extract_loop:
    test rbx, rbx
    jz output_begin

    mov rax, rbx
    shl rax, 4
    lea rdx, [r14 + rax]

    mov r8, qword ptr [r14]
    mov r9, qword ptr [r14 + 8]
    mov r10, qword ptr [rdx]
    mov r11, qword ptr [rdx + 8]
    mov qword ptr [r14], r10
    mov qword ptr [r14 + 8], r11
    mov qword ptr [rdx], r8
    mov qword ptr [rdx + 8], r9

    xor edi, edi
    mov rsi, rbx
    call sift_down
    dec rbx
    jmp heap_extract_loop

output_begin:
    xor r12d, r12d

output_loop:
    cmp r12, r15
    jae exit_success

    mov rax, r12
    shl rax, 4
    lea rbx, [r14 + rax]
    mov rsi, qword ptr [rbx]
    mov rdx, qword ptr [rbx + 8]
    test rdx, rdx
    jz output_newline

    mov eax, 1
    mov edi, 1
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    inc r12
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

sift_down:
    push rbx
    push r12
    push r13
    push r14

    mov r12, rdi
    mov r13, rsi
    lea rbx, [rip + line_table]

sift_loop:
    lea r14, [r12 + r12 + 1]
    cmp r14, r13
    jae sift_done

    lea rax, [r14 + 1]
    cmp rax, r13
    jae sift_child_selected

    mov rdi, r14
    shl rdi, 4
    add rdi, rbx
    mov rsi, rax
    shl rsi, 4
    add rsi, rbx
    call compare_lines
    test eax, eax
    jge sift_child_selected
    inc r14

sift_child_selected:
    mov rdi, r12
    shl rdi, 4
    add rdi, rbx
    mov rsi, r14
    shl rsi, 4
    add rsi, rbx
    call compare_lines
    test eax, eax
    jge sift_done

    mov rax, r12
    shl rax, 4
    lea rdi, [rbx + rax]
    mov rax, r14
    shl rax, 4
    lea rsi, [rbx + rax]

    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    mov qword ptr [rsi], r8
    mov qword ptr [rsi + 8], r9

    mov r12, r14
    jmp sift_loop

sift_done:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

compare_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]

    mov rdx, r9
    cmp rdx, r11
    cmova rdx, r11
    xor ecx, ecx

compare_loop:
    cmp rcx, rdx
    jae compare_lengths
    movzx eax, byte ptr [r8 + rcx]
    movzx esi, byte ptr [r10 + rcx]
    cmp eax, esi
    jb compare_less
    ja compare_greater
    inc rcx
    jmp compare_loop

compare_lengths:
    cmp r9, r11
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero INPUT_MAX
.balign 16
line_table:
    .zero LINE_MAX * DESC_SIZE
