.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 32768

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x3, #INPUT_SIZE
    sub x2, x3, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, line_table
    add x21, x21, :lo12:line_table
    mov x22, #0
    mov x23, x19
    mov x24, x19
    add x25, x19, x20

parse_loop:
    cmp x24, x25
    b.eq parse_done
    ldrb w0, [x24]
    cmp w0, #10
    b.ne parse_next
    sub x26, x24, x23
    add x27, x21, x22, lsl #4
    stp x23, x26, [x27]
    add x22, x22, #1
    add x23, x24, #1

parse_next:
    add x24, x24, #1
    b parse_loop

parse_done:
    cmp x23, x25
    b.eq build_heap
    sub x26, x25, x23
    add x27, x21, x22, lsl #4
    stp x23, x26, [x27]
    add x22, x22, #1

build_heap:
    lsr x23, x22, #1

build_loop:
    cbz x23, extract_setup
    sub x23, x23, #1
    mov x0, x21
    mov x1, x23
    mov x2, x22
    bl sift_down
    b build_loop

extract_setup:
    mov x23, x22

extract_loop:
    cmp x23, #1
    b.ls output_setup
    sub x23, x23, #1
    ldp x24, x25, [x21]
    add x26, x21, x23, lsl #4
    ldp x27, x28, [x26]
    stp x27, x28, [x21]
    stp x24, x25, [x26]
    mov x0, x21
    mov x1, #0
    mov x2, x23
    bl sift_down
    b extract_loop

output_setup:
    mov x23, #0
    adrp x24, newline
    add x24, x24, :lo12:newline

output_loop:
    cmp x23, x22
    b.eq program_exit
    add x25, x21, x23, lsl #4
    ldp x26, x27, [x25]
    cbz x27, output_newline
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #SYS_write
    svc #0

output_newline:
    mov x0, #1
    mov x1, x24
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x23, x23, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

sift_down:
    stp x29, x30, [sp, #-80]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    mov x29, sp
    mov x19, x0
    mov x20, x1
    mov x21, x2
    add x9, x19, x20, lsl #4
    ldp x22, x23, [x9]

sift_loop:
    lsl x24, x20, #1
    add x24, x24, #1
    cmp x24, x21
    b.hs sift_place

    add x9, x19, x24, lsl #4
    ldp x25, x26, [x9]
    add x27, x24, #1
    cmp x27, x21
    b.hs sift_child_ready

    add x9, x19, x27, lsl #4
    ldp x28, x9, [x9]
    mov x0, x25
    mov x1, x26
    mov x2, x28
    mov x3, x9
    bl compare_lines
    cmp w0, #0
    b.ge sift_child_ready
    mov x24, x27
    mov x25, x28
    mov x26, x9

sift_child_ready:
    mov x0, x25
    mov x1, x26
    mov x2, x22
    mov x3, x23
    bl compare_lines
    cmp w0, #0
    b.le sift_place
    add x9, x19, x20, lsl #4
    stp x25, x26, [x9]
    mov x20, x24
    b sift_loop

sift_place:
    add x9, x19, x20, lsl #4
    stp x22, x23, [x9]
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x23, x24, [sp, #48]
    ldp x25, x26, [sp, #64]
    ldp x29, x30, [sp], #80
    ret

compare_lines:
    mov x10, #0

compare_loop:
    cmp x10, x1
    b.eq compare_prefix
    cmp x10, x3
    b.eq compare_prefix
    ldrb w11, [x0, x10]
    ldrb w12, [x2, x10]
    cmp w11, w12
    b.lo compare_less
    b.hi compare_greater
    add x10, x10, #1
    b compare_loop

compare_prefix:
    cmp x1, x3
    b.lo compare_less
    b.hi compare_greater
    mov w0, #0
    ret

compare_less:
    mov w0, #-1
    ret

compare_greater:
    mov w0, #1
    ret

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buf:
    .zero 32768
.balign 16
line_table:
    .zero 16000
