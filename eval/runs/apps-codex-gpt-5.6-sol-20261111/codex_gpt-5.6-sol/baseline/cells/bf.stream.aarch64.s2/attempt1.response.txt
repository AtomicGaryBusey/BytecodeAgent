.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #1
    lsl x21, x21, #16

read_loop:
    cmp x20, x21
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x23, x19, x20
    mov x9, x19

find_delimiter:
    cmp x9, x23
    b.eq no_delimiter
    ldrb w10, [x9]
    cmp w10, #0x21
    b.eq delimiter_found
    add x9, x9, #1
    b find_delimiter

delimiter_found:
    mov x21, x9
    add x22, x9, #1
    b validate_setup

no_delimiter:
    mov x21, x23
    mov x22, x23

validate_setup:
    adrp x24, jump_map
    add x24, x24, :lo12:jump_map
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x12, #0
    mov x9, x19

validate_loop:
    cmp x9, x21
    b.eq validate_end
    ldrb w10, [x9]
    cmp w10, #0x5b
    b.eq validate_open
    cmp w10, #0x5d
    b.eq validate_close
    add x9, x9, #1
    b validate_loop

validate_open:
    cmp x12, #1000
    b.hs invalid_program
    str x9, [x11, x12, lsl #3]
    add x12, x12, #1
    add x9, x9, #1
    b validate_loop

validate_close:
    cbz x12, invalid_program
    sub x12, x12, #1
    ldr x13, [x11, x12, lsl #3]
    sub x14, x9, x19
    str x13, [x24, x14, lsl #3]
    sub x14, x13, x19
    str x9, [x24, x14, lsl #3]
    add x9, x9, #1
    b validate_loop

validate_end:
    cbnz x12, invalid_program

    adrp x25, tape
    add x25, x25, :lo12:tape
    mov x26, x25
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, #0
    mov x9, x19

execute_loop:
    cmp x9, x21
    b.eq execute_done
    ldrb w10, [x9], #1

    cmp w10, #0x3e
    b.eq command_right
    cmp w10, #0x3c
    b.eq command_left
    cmp w10, #0x2b
    b.eq command_plus
    cmp w10, #0x2d
    b.eq command_minus
    cmp w10, #0x2e
    b.eq command_output
    cmp w10, #0x2c
    b.eq command_input
    cmp w10, #0x5b
    b.eq command_open
    cmp w10, #0x5d
    b.eq command_close
    b execute_loop

command_right:
    add x26, x26, #1
    b execute_loop

command_left:
    sub x26, x26, #1
    b execute_loop

command_plus:
    ldrb w11, [x26]
    add w11, w11, #1
    strb w11, [x26]
    b execute_loop

command_minus:
    ldrb w11, [x26]
    sub w11, w11, #1
    strb w11, [x26]
    b execute_loop

command_output:
    ldrb w11, [x26]
    strb w11, [x27, x28]
    add x28, x28, #1
    b execute_loop

command_input:
    cmp x22, x23
    b.hs input_exhausted
    ldrb w11, [x22], #1
    strb w11, [x26]
    b execute_loop

input_exhausted:
    strb wzr, [x26]
    b execute_loop

command_open:
    ldrb w11, [x26]
    cbnz w11, execute_loop
    sub x14, x9, x19
    sub x14, x14, #1
    ldr x9, [x24, x14, lsl #3]
    add x9, x9, #1
    b execute_loop

command_close:
    ldrb w11, [x26]
    cbz w11, execute_loop
    sub x14, x9, x19
    sub x14, x14, #1
    ldr x9, [x24, x14, lsl #3]
    add x9, x9, #1
    b execute_loop

execute_done:
    cbz x28, exit_success
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
jump_map:
    .zero 524288
.balign 16
bracket_stack:
    .zero 8000
.balign 16
tape:
    .zero 30000
.balign 16
output_buf:
    .zero 65536
