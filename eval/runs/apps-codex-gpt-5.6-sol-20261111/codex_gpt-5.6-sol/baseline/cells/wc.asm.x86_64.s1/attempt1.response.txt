.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    lea rbx, [rip + input_buffer]
    lea rbp, [rbx + rax]

scan_loop:
    cmp rbx, rbp
    je read_loop

    movzx eax, byte ptr [rbx]
    inc rbx

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb non_whitespace
    cmp al, 13
    jbe whitespace

non_whitespace:
    test r15d, r15d
    jne scan_loop
    inc r13
    mov r15d, 1
    jmp scan_loop

whitespace:
    xor r15d, r15d
    jmp scan_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    mov rsi, rdi
    mov r8d, 10

decimal_loop:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jne decimal_loop

    mov r9, rsi
    lea r10, [rdi - 1]

reverse_loop:
    cmp r9, r10
    jae reverse_done
    mov al, byte ptr [r9]
    mov dl, byte ptr [r10]
    mov byte ptr [r9], dl
    mov byte ptr [r10], al
    inc r9
    dec r10
    jmp reverse_loop

reverse_done:
    ret

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
