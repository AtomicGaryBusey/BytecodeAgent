.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFFER_SIZE, 65536

.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0

    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #BUFFER_SIZE
    mov x8, #SYS_read
    svc #0

    cmp x0, #0
    b.le input_done

    add x21, x21, x0
    mov x3, x23
    mov x4, x0

scan_loop:
    ldrb w5, [x3], #1

    cmp w5, #10
    b.eq newline_byte

    cmp w5, #32
    b.eq whitespace_byte

    cmp w5, #9
    b.lo word_byte
    cmp w5, #13
    b.ls whitespace_byte

word_byte:
    cbnz w22, scan_next
    add x20, x20, #1
    mov w22, #1
    b scan_next

newline_byte:
    add x19, x19, #1

whitespace_byte:
    mov w22, #0

scan_next:
    subs x4, x4, #1
    b.ne scan_loop
    b read_loop

input_done:
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x1, x24

    mov x0, x19
    bl append_uint

    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl append_uint

    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl append_uint

    mov w2, #10
    strb w2, [x1], #1

    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

append_uint:
    sub sp, sp, #32
    add x5, sp, #32
    mov x3, #10

1:
    udiv x2, x0, x3
    msub x4, x2, x3, x0
    add x4, x4, #48
    sub x5, x5, #1
    strb w4, [x5]
    mov x0, x2
    cbnz x0, 1b

    add x6, sp, #32

2:
    ldrb w4, [x5], #1
    strb w4, [x1], #1
    cmp x5, x6
    b.ne 2b

    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .zero BUFFER_SIZE
output_buffer:
    .zero 64
