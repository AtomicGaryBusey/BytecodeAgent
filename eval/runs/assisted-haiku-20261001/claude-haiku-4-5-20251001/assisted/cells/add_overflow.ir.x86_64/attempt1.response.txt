export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	
	%xor_ab =l xor %a, %b
	%same_sign =w csgtl %xor_ab, -1
	jnz %same_sign, @check_diff, @no_overflow
	
@check_diff
	%xor_a_sum =l xor %a, %sum
	%diff_sign =w csltl %xor_a_sum, 0
	jnz %diff_sign, @overflow, @no_overflow
	
@overflow
	%one =l copy 1
	ret %one
	
@no_overflow
	%zero =l copy 0
	ret %zero
}
