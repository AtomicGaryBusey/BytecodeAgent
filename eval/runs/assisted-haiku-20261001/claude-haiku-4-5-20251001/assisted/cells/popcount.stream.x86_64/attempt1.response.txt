```asm
popcount:
    xor eax, eax        # count = 0
    test rdi, rdi       # test rdi
    jz .Ldone           # if rdi == 0, done
.Lloop:
    mov rcx, rdi        # rcx = rdi
    dec rcx              # rcx = rdi - 1
    and rdi, rcx        # rdi &= rcx
    lea rax, [rax + 1]  # count++ (lea doesn't affect flags)
    jnz .Lloop          # if rdi != 0, loop
.Ldone:
    ret
```
