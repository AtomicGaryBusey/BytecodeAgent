export function l $reverse(l %a, l %n) {
@start
	%zero =l copy 0
	%left =l copy 0
	%right =l sub %n, 1
@loop
	%cond =w csltl %left, %right
	jnz %cond, @body, @done
@body
	%left_off =l mul %left, 8
	%left_addr =l add %a, %left_off
	%left_val =l loadl %left_addr
	
	%right_off =l mul %right, 8
	%right_addr =l add %a, %right_off
	%right_val =l loadl %right_addr
	
	storel %right_val, %left_addr
	storel %left_val, %right_addr
	
	%left =l add %left, 1
	%right =l sub %right, 1
	
	jmp @loop
@done
	ret %zero
}
