export function l $fib_rec(l %n) {
@start
	%eq0 =w ceql %n, 0
	jnz %eq0, @return_zero, @check_one

@return_zero
	ret 0

@check_one
	%eq1 =w ceql %n, 1
	jnz %eq1, @return_one, @compute

@return_one
	ret 1

@compute
	%n_minus_1 =l sub %n, 1
	%n_minus_2 =l sub %n, 2
	%fib_n_minus_1 =l call $fib_rec(l %n_minus_1)
	%fib_n_minus_2 =l call $fib_rec(l %n_minus_2)
	%result =l add %fib_n_minus_1, %fib_n_minus_2
	ret %result
}
