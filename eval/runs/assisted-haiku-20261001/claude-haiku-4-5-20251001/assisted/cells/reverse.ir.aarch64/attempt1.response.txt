export function l $reverse(l %a, l %n) {
@start
	%left =l copy 0
	%right =l sub %n, 1
@loop
	%cond =w csltl %left, %right
	jnz %cond, @body, @done
@body
	%off_left =l mul %left, 8
	%addr_left =l add %a, %off_left
	%val_left =l loadl %addr_left
	
	%off_right =l mul %right, 8
	%addr_right =l add %a, %off_right
	%val_right =l loadl %addr_right
	
	storel %val_right, %addr_left
	storel %val_left, %addr_right
	
	%left =l add %left, 1
	%right =l sub %right, 1
	jmp @loop
@done
	ret 0
}
