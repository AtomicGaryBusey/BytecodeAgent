export function l $fib_rec(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @zero, @check_one
@zero
	ret 0
@check_one
	%is_one =w ceql %n, 1
	jnz %is_one, @one, @recurse
@one
	ret 1
@recurse
	%n1 =l sub %n, 1
	%n2 =l sub %n, 2
	%r1 =l call $fib_rec(l %n1)
	%r2 =l call $fib_rec(l %n2)
	%result =l add %r1, %r2
	ret %result
}
