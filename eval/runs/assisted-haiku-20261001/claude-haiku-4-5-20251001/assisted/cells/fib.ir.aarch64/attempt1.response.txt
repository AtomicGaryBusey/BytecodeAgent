export function l $fib(l %n) {
@start
	%cmp0 =w ceql %n, 0
	jnz %cmp0, @ret0, @check1
@ret0
	ret 0
@check1
	%cmp1 =w ceql %n, 1
	jnz %cmp1, @ret1, @loop_init
@ret1
	ret 1
@loop_init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 2
@loop
	%less =w cslel %i, %n
	jnz %less, @body, @done
@body
	%temp =l add %a, %b
	%a =l copy %b
	%b =l copy %temp
	%i =l add %i, 1
	jmp @loop
@done
	ret %b
}
