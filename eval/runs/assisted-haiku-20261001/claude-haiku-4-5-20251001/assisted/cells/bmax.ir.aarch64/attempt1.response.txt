export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%elem =l loadl %addr
	%cmp =w csltl %max, %elem
	%i =l add %i, 1
	jnz %cmp, @update, @loop
@update
	%max =l copy %elem
	jmp @loop
@done
	ret %max
}
