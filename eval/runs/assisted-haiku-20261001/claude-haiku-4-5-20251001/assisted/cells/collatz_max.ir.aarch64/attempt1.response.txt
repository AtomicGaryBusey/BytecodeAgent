@start
	%max_steps =l copy 0
	%k =l copy 1
	jmp @k_loop
@k_loop

	jnz %k_le_n, @k_body, @k_done
@k_body
	%x =l copy %k
	%steps =l copy 0
	jmp @x_loop
@x_loop
	%x_eq_1 =w ceql %x, 1
	jnz %x_eq_1, @x_done, @x_body
@x_body
	%x_mod_2 =l rem %x, 2
	%x_is_even =w ceql %x_mod_2, 0
	jnz %x_is_even, @x_even_case, @x_odd_case
@x_even_case

	%steps =l add %steps, 1
	jmp @x_loop
@x_odd_case

	%x =l add %tmp1, 1
	%steps =l add %steps, 1
	jmp @x_loop
@x_done
	%steps_gt_max =w csgtl %steps, %max_steps
	jnz %steps_gt_max, @update_max, @next_k
@update_max
	%max_steps =l copy %steps
	jmp @next_k
@next_k
	%k =l add %k, 1
	jmp @k_loop
@k_done
	ret %max_steps
}
