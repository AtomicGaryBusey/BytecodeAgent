export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%i =l copy 0
@outer_loop
	%outer_cond =w csltl %i, %n
	jnz %outer_cond, @outer_body, @done
@outer_body
	%j =l copy 0
	%inner_max =l sub %n, %i
	%inner_max =l sub %inner_max, %one
@inner_loop
	%inner_cond =w csltl %j, %inner_max
	jnz %inner_cond, @inner_body, @inner_done
@inner_body
	%off_j =l mul %j, %eight
	%addr_j =l add %a, %off_j
	%val_j =l loadl %addr_j
	%j_plus_one =l add %j, %one
	%off_j_plus_one =l mul %j_plus_one, %eight
	%addr_j_plus_one =l add %a, %off_j_plus_one
	%val_j_plus_one =l loadl %addr_j_plus_one
	%should_swap =w csgtl %val_j, %val_j_plus_one
	jnz %should_swap, @swap, @continue
@swap
	storel %val_j_plus_one, %addr_j
	storel %val_j, %addr_j_plus_one
@continue
	%j =l add %j, %one
	jmp @inner_loop
@inner_done
	%i =l add %i, %one
	jmp @outer_loop
@done
	ret %zero
}
