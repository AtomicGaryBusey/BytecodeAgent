memmove_i64:
    test rdx, rdx
    jz .Ldone
    
    cmp rdi, rsi
    jl .Lcopy_forward
    je .Ldone
    
    lea rax, [rsi + rdx*8]
    cmp rax, rdi
    jle .Lcopy_forward
    
    mov rax, rdx
    dec rax
    
.Lcopy_backward_loop:
    cmp rax, 0
    jl .Ldone
    
    mov rcx, [rsi + rax*8]
    mov [rdi + rax*8], rcx
    
    dec rax
    jmp .Lcopy_backward_loop
    
.Lcopy_forward:
    xor rax, rax
    
.Lcopy_forward_loop:
    cmp rax, rdx
    jge .Ldone
    
    mov rcx, [rsi + rax*8]
    mov [rdi + rax*8], rcx
    
    inc rax
    jmp .Lcopy_forward_loop
    
.Ldone:
    xor eax, eax
    ret
