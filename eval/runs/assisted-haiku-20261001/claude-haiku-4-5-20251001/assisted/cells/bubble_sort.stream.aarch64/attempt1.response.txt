bubble_sort:
    cmp x1, #1
    b.le .Lreturn_zero
    
    mov x2, #0
.Louter_loop:
    cmp x2, x1
    b.ge .Lreturn_zero
    
    mov x3, #0
.Linner_loop:
    sub x4, x1, x2
    sub x4, x4, #1
    cmp x3, x4
    b.ge .Linner_loop_end
    
    ldr x5, [x0, x3, lsl #3]
    add x4, x3, #1
    ldr x6, [x0, x4, lsl #3]
    
    cmp x5, x6
    b.le .Lno_swap
    
    str x6, [x0, x3, lsl #3]
    str x5, [x0, x4, lsl #3]
    
.Lno_swap:
    add x3, x3, #1
    b .Linner_loop
    
.Linner_loop_end:
    add x2, x2, #1
    b .Louter_loop
    
.Lreturn_zero:
    mov x0, #0
    ret
