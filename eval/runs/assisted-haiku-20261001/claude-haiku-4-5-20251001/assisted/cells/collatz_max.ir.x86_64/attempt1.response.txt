export function l $collatz_max(l %n) {
@start
	%max_steps =l copy 0
	%k =l copy 1
@k_loop
	%k_gt_n =w csgtl %k, %n
	jnz %k_gt_n, @done, @k_body
@k_body
	%x =l copy %k
	%steps =l copy 0
@x_loop
	%x_eq_1 =w ceql %x, 1
	jnz %x_eq_1, @x_done, @x_check
@x_check
	%rem =l rem %x, 2
	%is_even =w ceql %rem, 0
	jnz %is_even, @x_even, @x_odd
@x_even
	%x =l udiv %x, 2
	%steps =l add %steps, 1
	jmp @x_loop
@x_odd
	%three_x =l mul %x, 3
	%x =l add %three_x, 1
	%steps =l add %steps, 1
	jmp @x_loop
@x_done
	%steps_gt_max =w csgtl %steps, %max_steps
	jnz %steps_gt_max, @update_max, @k_next
@update_max
	%max_steps =l copy %steps
	jmp @k_next
@k_next
	%k =l add %k, 1
	jmp @k_loop
@done
	ret %max_steps
}
