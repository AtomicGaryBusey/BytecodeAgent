export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%cond_o =w csltl %i, %n
	jnz %cond_o, @outer_body, @done
@outer_body
	%j =l copy 0
	%ni =l sub %n, %i
	%limit =l sub %ni, 1
@inner
	%cond_i =w csltl %j, %limit
	jnz %cond_i, @inner_body, @next_i
@inner_body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	
	%swap =w csgtl %v0, %v1
	jnz %swap, @do_swap, @no_swap
@do_swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @no_swap
@no_swap
	%j =l add %j, 1
	jmp @inner
@next_i
	%i =l add %i, 1
	jmp @outer
@done
	%r =l copy 0
	ret %r
}
