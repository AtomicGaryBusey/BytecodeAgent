export function l $sum_to(l %n) {
@start
	%cond =w csltl %n, 1
	jnz %cond, @return_zero, @loop_start
@return_zero
	%zero =l copy 0
	ret %zero
@loop_start
	%sum =l copy 0
	%i =l copy 1
	jmp @loop
@loop
	%continue =w cslel %i, %n
	jnz %continue, @loop_body, @done
@loop_body
	%sum =l add %sum, %i
	%i =l add %i, 1
	jmp @loop
@done
	ret %sum
}
