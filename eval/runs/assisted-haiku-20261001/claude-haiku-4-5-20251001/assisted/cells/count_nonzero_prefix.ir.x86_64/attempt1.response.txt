export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @check, @done
@check
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%val =l loadl %addr
	%is_zero =w ceql %val, 0
	jnz %is_zero, @found_zero, @next
@found_zero
	ret %i
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %n
}
