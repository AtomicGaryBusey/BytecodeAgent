binary_search:
    mov x3, #0
    mov x4, x1
.Lloop:
    cmp x3, x4
    b.ge .Lfail
    add x5, x3, x4
    asr x5, x5, #1
    ldr x6, [x0, x5, lsl #3]
    cmp x6, x2
    b.eq .Lfound
    b.lt .Lright
    mov x4, x5
    b .Lloop
.Lright:
    add x3, x5, #1
    b .Lloop
.Lfound:
    mov x0, x5
    ret
.Lfail:
    movn x0, #0
    ret
