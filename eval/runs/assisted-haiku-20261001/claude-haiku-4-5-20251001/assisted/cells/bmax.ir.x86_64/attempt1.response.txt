export function l $bmax(l %a, l %n) {
@start
	%v0 =l loadl %a
	%max =l copy %v0
	%i =l copy 1
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%is_greater =w csltl %max, %v
	jnz %is_greater, @update, @next
@update
	%max =l copy %v
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
