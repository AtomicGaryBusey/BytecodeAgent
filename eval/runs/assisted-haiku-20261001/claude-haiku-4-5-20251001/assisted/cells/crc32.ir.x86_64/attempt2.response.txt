export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@byte_loop
	%continue =w csltl %i, %n
	jnz %continue, @process_byte, @finalize
@process_byte
	%addr =l add %p, %i
	%byte =w loadub %addr
	%crc =w xor %crc, %byte
	%j =w copy 0
@bit_loop
	%bit_continue =w csltw %j, 8
	jnz %bit_continue, @process_bit, @next_byte
@process_bit
	%lsb =w and %crc, 1
	%is_set =w cnew %lsb, 0
	jnz %is_set, @bit_set, @bit_clear
@bit_set
	%crc_shifted =w shr %crc, 1
	%crc =w xor %crc_shifted, 3988292384
	jmp @bit_end
@bit_clear
	%crc =w shr %crc, 1
	jmp @bit_end
@bit_end
	%j =w add %j, 1
	jmp @bit_loop
@next_byte
	%i =l add %i, 1
	jmp @byte_loop
@finalize
	%result =w xor %crc, 4294967295
	%res =l extuw %result
	ret %res
}
