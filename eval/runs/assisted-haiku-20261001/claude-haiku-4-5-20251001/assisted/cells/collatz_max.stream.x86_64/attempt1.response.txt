    xor rax, rax
    cmp rdi, 1
    jl .Ldone
    mov rsi, 1
.Louter:
    cmp rsi, rdi
    jg .Ldone
    mov rcx, rsi
    xor rdx, rdx
.Linner:
    cmp rcx, 1
    je .Lnext_k
    test rcx, 1
    jz .Leven
    imul rcx, rcx, 3
    add rcx, 1
    jmp .Lcount
.Leven:
    shr rcx, 1
.Lcount:
    inc rdx
    jmp .Linner
.Lnext_k:
    cmp rdx, rax
    jle .Lskip
    mov rax, rdx
.Lskip:
    inc rsi
    jmp .Louter
.Ldone:
    ret
