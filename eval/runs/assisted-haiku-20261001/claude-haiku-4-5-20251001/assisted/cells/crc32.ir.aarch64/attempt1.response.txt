export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@byte_loop
	%more =w csltl %i, %n
	jnz %more, @byte_body, @byte_done
@byte_body
	%addr =l add %p, %i
	%byte =w loadub %addr
	%crc =w xor %crc, %byte
	
	%j =w copy 0
@bit_loop
	%bit_more =w csltw %j, 8
	jnz %bit_more, @bit_body, @bit_done
@bit_body
	%lsb =w and %crc, 1
	jnz %lsb, @bit_set, @bit_clear
@bit_set
	%crc_shifted =w shr %crc, 1
	%crc =w xor %crc_shifted, 3988292384
	%j =w add %j, 1
	jmp @bit_loop
@bit_clear
	%crc =w shr %crc, 1
	%j =w add %j, 1
	jmp @bit_loop
@bit_done
	%i =l add %i, 1
	jmp @byte_loop
@byte_done
	%result =w xor %crc, 4294967295
	%result_l =l extuw %result
	ret %result_l
}
