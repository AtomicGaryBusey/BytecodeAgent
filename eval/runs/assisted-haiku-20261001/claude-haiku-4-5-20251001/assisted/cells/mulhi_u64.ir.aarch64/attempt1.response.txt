export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	
	%a_l =l and %a, %mask
	%a_h =l shr %a, 32
	%b_l =l and %b, %mask
	%b_h =l shr %b, 32
	
	%c0 =l mul %a_l, %b_l
	%c1 =l mul %a_h, %b_l
	%c2 =l mul %a_l, %b_h
	%c3 =l mul %a_h, %b_h
	
	%c0_h =l shr %c0, 32
	%c1_h =l shr %c1, 32
	%c1_l =l and %c1, %mask
	%c2_h =l shr %c2, 32
	%c2_l =l and %c2, %mask
	
	%sum_carry =l add %c0_h, %c1_l
	%sum_carry =l add %sum_carry, %c2_l
	%carry =l shr %sum_carry, 32
	
	%result =l add %c3, %carry
	%result =l add %result, %c1_h

	ret %result
}
