export function l $binary_search(l %a, l %n, l %key) {
@start
	%left =l copy 0
	%right =l copy %n
@loop
	%cond =w csltl %left, %right
	jnz %cond, @body, @notfound
@body
	%sum =l add %left, %right
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%eq =w ceql %val, %key
	jnz %eq, @found, @notequal
@found
	ret %mid
@notequal
	%lt =w csltl %val, %key
	jnz %lt, @moveleft, @moveright
@moveleft
	%left =l add %mid, 1
	jmp @loop
@moveright
	%right =l copy %mid
	jmp @loop
@notfound
	ret -1
}
