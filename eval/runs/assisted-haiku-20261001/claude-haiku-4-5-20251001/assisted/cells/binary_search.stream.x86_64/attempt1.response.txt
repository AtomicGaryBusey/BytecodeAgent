binary_search:
    xor eax, eax
    mov rcx, rsi
    dec rcx
.loop:
    cmp rax, rcx
    jg .not_found
    mov r8, rax
    add r8, rcx
    shr r8, 1
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .found
    jl .search_right
    mov rcx, r8
    dec rcx
    jmp .loop
.search_right:
    mov rax, r8
    inc rax
    jmp .loop
.found:
    mov rax, r8
    ret
.not_found:
    mov rax, -1
    ret
