export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp =w cugtl %dst, %src
	jnz %cmp, @backward, @forward

@forward
	%i =l copy 0
@forward_loop
	%cond =w csltl %i, %n
	jnz %cond, @forward_body, @end

@forward_body
	%off =l mul %i, 8
	%src_addr =l add %src, %off
	%dst_addr =l add %dst, %off
	%val =l loadl %src_addr
	storel %val, %dst_addr
	%i =l add %i, 1
	jmp @forward_loop

@backward
	%i =l sub %n, 1
@backward_loop
	%cond =w csgtl %i, -1
	jnz %cond, @backward_body, @end

@backward_body
	%off =l mul %i, 8
	%src_addr =l add %src, %off
	%dst_addr =l add %dst, %off
	%val =l loadl %src_addr
	storel %val, %dst_addr
	%i =l sub %i, 1
	jmp @backward_loop

@end
	ret 0
}
