export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cond_n =w csltl %n, 1
	jnz %cond_n, @done, @check_dir

@check_dir
	%cond =w csgtl %dst, %src
	jnz %cond, @backward, @forward

@forward
	%i =l copy 0
	jmp @forward_loop

@forward_loop
	%cond =w csltl %i, %n
	jnz %cond, @forward_body, @done

@forward_body
	%off =l mul %i, 8
	%src_addr =l add %src, %off
	%val =l loadl %src_addr
	%dst_addr =l add %dst, %off
	storel %val, %dst_addr
	%i =l add %i, 1
	jmp @forward_loop

@backward
	%i =l sub %n, 1
	jmp @backward_loop

@backward_loop
	%cond =w csltl %i, 0
	jnz %cond, @done, @backward_body

@backward_body
	%off =l mul %i, 8
	%src_addr =l add %src, %off
	%val =l loadl %src_addr
	%dst_addr =l add %dst, %off
	storel %val, %dst_addr
	%i =l sub %i, 1
	jmp @backward_loop

@done
	%zero =l copy 0
	ret %zero
}
