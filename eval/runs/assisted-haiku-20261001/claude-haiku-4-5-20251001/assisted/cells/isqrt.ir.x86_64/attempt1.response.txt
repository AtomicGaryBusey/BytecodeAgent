export function l $isqrt(l %n) {
@start
	%left =l copy 0
	%right =l copy 4294967295
	
@loop
	%cond =w cultl %left, %right
	jnz %cond, @search, @done
	
@search
	%sum =l add %left, %right
	%sum1 =l add %sum, 1
	%mid =l shr %sum1, 1
	
	%product =l mul %mid, %mid
	%cmp =w culel %product, %n
	jnz %cmp, @update_left, @update_right
	
@update_left
	%left =l copy %mid
	jmp @loop
	
@update_right
	%right =l sub %mid, 1
	jmp @loop
	
@done
	ret %left
}
