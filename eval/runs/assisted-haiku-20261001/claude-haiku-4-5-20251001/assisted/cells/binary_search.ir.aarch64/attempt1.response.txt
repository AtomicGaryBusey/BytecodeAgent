export function l $binary_search(l %a, l %n, l %key) {
@start
	%low =l copy 0
	%high =l copy %n
@loop
	%cmp =w csltl %low, %high
	jnz %cmp, @search, @notfound
@search
	%sum =l add %low, %high
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%eq =w ceql %val, %key
	jnz %eq, @found, @notequal
@found
	ret %mid
@notequal
	%lt =w csltl %val, %key
	jnz %lt, @searchhigh, @searchlow
@searchhigh
	%low =l add %mid, 1
	jmp @loop
@searchlow
	%high =l copy %mid
	jmp @loop
@notfound
	ret -1
}
