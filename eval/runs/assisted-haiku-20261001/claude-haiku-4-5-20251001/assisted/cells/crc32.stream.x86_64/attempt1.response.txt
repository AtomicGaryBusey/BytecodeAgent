crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
    xor rcx, rcx
.Lbyte_loop:
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit_loop:
    test eax, 1
    jz .Lno_xor
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lbit_continue
.Lno_xor:
    shr eax, 1
.Lbit_continue:
    dec r8d
    jnz .Lbit_loop
    inc rcx
    cmp rcx, rsi
    jb .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
