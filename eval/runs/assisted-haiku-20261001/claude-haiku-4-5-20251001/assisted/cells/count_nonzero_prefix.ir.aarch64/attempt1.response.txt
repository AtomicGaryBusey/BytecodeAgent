export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%count =l copy 0
@loop
	%cond =w csltl %count, %n
	jnz %cond, @body, @done
@body
	%off =l mul %count, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%is_zero =w ceql %val, 0
	jnz %is_zero, @done, @next
@next
	%count =l add %count, 1
	jmp @loop
@done
	ret %count
}
