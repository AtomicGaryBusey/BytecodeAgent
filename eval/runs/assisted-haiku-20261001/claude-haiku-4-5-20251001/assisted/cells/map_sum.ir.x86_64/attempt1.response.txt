export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%result =l call %f(l %elem)
	%sum =l add %sum, %result
	%i =l add %i, 1
	jmp @loop
@done
	ret %sum
}
