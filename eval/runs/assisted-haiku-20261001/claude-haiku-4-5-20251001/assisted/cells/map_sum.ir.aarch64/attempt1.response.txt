export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%val =l loadl %addr
	%res =l call %f(l %val)
	%sum =l add %sum, %res
	%i =l add %i, 1
	jmp @loop
@done
	ret %sum
}
