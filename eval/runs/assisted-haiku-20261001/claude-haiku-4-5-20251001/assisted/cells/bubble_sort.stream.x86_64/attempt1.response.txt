    cmp rsi, 1
    jle .return_zero
    xor r8, r8
.outer_loop:
    cmp r8, rsi
    jge .return_zero
    xor r9, r9
.inner_loop:
    mov rax, rsi
    sub rax, r8
    dec rax
    cmp r9, rax
    jge .inner_done
    mov r10, qword ptr [rdi + r9*8]
    mov r11, qword ptr [rdi + r9*8 + 8]
    cmp r10, r11
    jle .no_swap
    mov qword ptr [rdi + r9*8], r11
    mov qword ptr [rdi + r9*8 + 8], r10
.no_swap:
    inc r9
    jmp .inner_loop
.inner_done:
    inc r8
    jmp .outer_loop
.return_zero:
    xor eax, eax
    ret
