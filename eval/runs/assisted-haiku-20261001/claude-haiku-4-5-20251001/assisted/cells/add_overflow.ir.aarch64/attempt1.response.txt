export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum =l add %a, %b
    storel %sum, %out
    
    %a_neg =w csltl %a, 0
    %b_neg =w csltl %b, 0
    %sum_neg =w csltl %sum, 0
    
    %same_ab =w ceqw %a_neg, %b_neg
    %diff_sign =w cnew %a_neg, %sum_neg
    
    %overflow_w =w and %same_ab, %diff_sign
    %overflow =l extuw %overflow_w
    
    ret %overflow
}
