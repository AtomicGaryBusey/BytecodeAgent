export function l $isqrt(l %n) {
@start
	%low =l copy 0
	%high =l copy 4294967295
@loop
	%cmp =w cultl %low, %high
	jnz %cmp, @body, @done
@body
	%sum =l add %low, %high
	%sum_plus_1 =l add %sum, 1
	%mid =l udiv %sum_plus_1, 2
	%mid_sq =l mul %mid, %mid
	%le =w culel %mid_sq, %n
	jnz %le, @set_low, @dec_high
@set_low
	%low =l copy %mid
	jmp @loop
@dec_high
	%high =l sub %mid, 1
	jmp @loop
@done
	ret %low
}
