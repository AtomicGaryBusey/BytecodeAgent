export function l $crc32($p,$n)
@0
%crc =w 0xffffffff
%poly =w 0xEDB88320
%cnt =w $n
%addr =l $p
%cnt_l =l extuw %cnt
%cnt_zero =w ceql %cnt_l, 0
jnz @7
@1
%byte =w loadub %addr
%crc =w xor %crc, %byte
%cnt8 =w 8
@2
%bit =w and %crc, 1
%crc =w shr %crc, 1
jnz @3
@4
jmp @5
@3
%crc =w xor %crc, %poly
jmp @5
@5
%cnt8 =w sub %cnt8, 1
jnz @2
@6
%cnt =w sub %cnt, 1
%addr =l add %addr, 1
jnz @1
@7
%crc =w xor %crc, 0xffffffff
%res =l extuw %crc
ret %res
