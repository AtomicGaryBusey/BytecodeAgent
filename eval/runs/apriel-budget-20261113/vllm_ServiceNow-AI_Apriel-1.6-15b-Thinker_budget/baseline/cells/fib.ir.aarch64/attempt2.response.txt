export function l $fib(l)
    %n = %0
    %eq0 = ceql %n, 0
    jnz @ret0
    %eq1 = ceql %n, 1
    jnz @ret1
    %a = subl %n, %n
    %b = addl %a, 1
    %counter = subl %n, 1
@loop:
    %c = addl %a, %b
    %a = %b
    %b = %c
    %counter = subl %counter, 1
    %zero = ceql %counter, 0
    jnz @ret
    br @loop
@ret0:
    ret %n
@ret1:
    ret %n
@ret:
    ret %b
