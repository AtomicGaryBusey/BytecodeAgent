export function l $fib_rec(l %x) {
    @label L0:
    %0 =l ceql %x 0
    jnz Lbase0
    %0 =l ceql %x 1
    jnz Lbase1
    %a =l subl %x 1
    %b =l subl %x 2
    %r1 =l call $fib_rec(l %a)
    %r2 =l call $fib_rec(l %b)
    %0 =l addl %r1 %r2
    ret
    Lbase0:
    %0 =l 0
    ret
    Lbase1:
    %0 =l 1
    ret
}
