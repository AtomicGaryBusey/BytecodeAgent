bmin_empty:
    test rsi, rsi
    jz .Lzero
    mov rax, qword ptr [rdi]
    mov rdx, rsi
    mov rcx, 1
.Lloop:
    cmp rcx, rdx
    jae .Ldone
    mov r8, qword ptr [rdi + rcx*8]
    cmp r8, rax
    jge .Lnext
    mov rax, r8
.Lnext:
    inc rcx
    jmp .Lloop
.Ldone:
    ret
.Lzero:
    mov rax, 0x7FFFFFFFFFFFFFFF
    ret
