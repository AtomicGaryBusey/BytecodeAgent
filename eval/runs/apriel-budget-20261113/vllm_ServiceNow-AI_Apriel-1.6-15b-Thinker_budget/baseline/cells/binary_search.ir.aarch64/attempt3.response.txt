export function l $binary_search($a,$n,$key)
    @Lstart
    subl %zero,$n,$n
    ceql %one,$n,$n
    subl %neg_one,%zero,%one
    ceql %n_eq_0,$n,%zero
    jnz %n_eq_0,@Lend
    movl %low,%zero
    movl %high,$n
    @Lloop
    csltl %lt,%low,%high
    jnz %lt,@Lmid
    j @Lend
    @Lmid
    addl %sum,%low,%high
    srl %mid,%sum,%one
    addl %addr,$a,%mid
    loadl %val,%addr
    ceql %eq,%val,$key
    jnz %eq,@Lfound
    csltl %lt2,%val,$key
    jnz %lt2,@Llow
    movl %high,%mid
    j @Lloop
    @Llow
    addl %newlow,%mid,%one
    movl %low,%newlow
    j @Lloop
    @Lfound
    movl $binary_search,%mid
    ret $binary_search
    @Lend
    movl $binary_search,%neg_one
    ret $binary_search
