export function l $binary_search(l $a, l $n, l $key)
    @Lstart
    ceql $n, 0, %n_eq_0
    jnz %n_eq_0, @Lend
    movl %high, $n
    subl %low, %high, %high
    @Lloop
    csltl %lt, %low, %high
    jnz %lt, @Lmid
    j @Lend
    @Lmid
    subl %diff, %high, %low
    srl %diff, %diff, 1
    addl %mid, %low, %diff
    addl %addr, $a, %mid
    loadl %val, %addr
    ceql %eq, %val, $key
    jnz %eq, @Lfound
    csltl %lt, %val, $key
    jnz %lt, @Llow
    movl %high, %mid
    j @Lloop
    @Llow
    addl %low, %mid, 1
    j @Lloop
    @Lfound
    movl $binary_search, %mid
    ret $binary_search
    @Lend
    subl %neg_one, 0, 1
    movl $binary_search, %neg_one
    ret $binary_search
