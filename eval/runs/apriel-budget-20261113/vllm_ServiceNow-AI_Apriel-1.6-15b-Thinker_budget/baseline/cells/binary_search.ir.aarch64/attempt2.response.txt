export function l $binary_search($a, $n, $key)
    @Lstart
    ceql $n, 0, %n_eq_0
    jnz %n_eq_0, @Lend
    movl %low, 0
    movl %high, $n
    @Lloop
    csltl %lt, %low, %high
    jnz %lt, @Lmid
    j @Lend
    @Lmid
    addl %sum, %low, %high
    srl %mid, %sum, 1
    addl %addr, $a, %mid
    loadl %val, %addr
    ceql %eq, %val, $key
    jnz %eq, @Lfound
    csltl %lt, %val, $key
    jnz %lt, @Llow
    movl %mid, %high
    j @Lloop
    @Llow
    addl %newlow, %mid, 1
    movl %newlow, %low
    j @Lloop
    @Lfound
    movl %mid, $binary_search
    ret $binary_search
    @Lend
    subl %neg_one, 0, 1
    movl %neg_one, $binary_search
    ret $binary_search
