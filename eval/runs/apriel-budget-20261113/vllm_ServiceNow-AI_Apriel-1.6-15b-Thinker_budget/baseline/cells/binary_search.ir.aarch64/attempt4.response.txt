export function l $binary_search($a,$n,$key)
    @Lstart
    subl %zero,$n,$n
    ceql %one,$n,$n
    subl %neg_one,%zero,%one
    ceql %n_eq_0,$n,%zero
    jnz %n_eq_0,@Lend
    movl %low,%zero
    movl %high,$n
    @Lloop
    csltl %lt,%low,%high
    jnz %lt,@Lmid
    j @Lend
    @Lmid
    addl %sum,%low,%high
    srl %mid,%sum,%one
    addl %addr,$a,%mid
    loadl %val,%addr
    ceql %eq,%val,$key
    jnz %eq,@Lfound
    csltl %lt2,%val,$key
    jnz %lt2,@Llow
    movl %mid,%high
    j @Lloop
    @Llow
    addl %low,%mid,%one
    j @Lloop
    @Lfound
    movl %mid,$binary_search
    ret $binary_search
    @Lend
    movl %neg_one,$binary_search
    ret $binary_search
