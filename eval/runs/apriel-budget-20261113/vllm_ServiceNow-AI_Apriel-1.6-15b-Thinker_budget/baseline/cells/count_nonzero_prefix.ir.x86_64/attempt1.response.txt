export function l $count_nonzero_prefix(l,l)
    @label .Linit:
        subl %1, %1, %i
        jmp .Lloop
    @label .Lloop:
        csltl %lt, %i, %1
        jnz %lt, .Lload
        mov %i, %rax
        ret
    @label .Lload:
        sall 3, %i, %offset
        addl %0, %offset, %addr
        loadl %val, %addr
        subl %i, %i, %zero
        ceql %wzero, %val, %zero
        jnz %wzero, .Lexit
        ceql %one, %i, %i
        addl %i, %one, %i
        jmp .Lloop
    @label .Lexit:
        mov %i, %rax
        ret
