export function l $count_nonzero_prefix(l,l)
    @label .Lstart:
        movl %z, %i
        jmp .Lloop
    @label .Lloop:
        csltl %w0, %i, %1
        jnz %w0, .Lload
        mov %i, %rax
        ret
    @label .Lload:
        sall %offset, %i, 3
        addl %addr, %0, %offset
        loadl %val, %addr
        ceql %w1, %val, %z
        jnz %w1, .Lexit
        addl %i, %i, %one
        jmp .Lloop
    @label .Lexit:
        mov %i, %rax
        ret
