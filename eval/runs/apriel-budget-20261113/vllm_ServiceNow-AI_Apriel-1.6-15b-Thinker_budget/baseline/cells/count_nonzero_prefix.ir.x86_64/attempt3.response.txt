export function l $count_nonzero_prefix(l,l)
    @label .Linit:
        movl %z, %i
        jmp .Lloop
    @label .Lloop:
        csltl %lt, %i, %1
        jnz %lt, .Lload
        mov %i, %rax
        ret
    @label .Lload:
        sall %offset, %i, 3
        addl %addr, %0, %offset
        loadl %val, %addr
        ceql %wzero, %val, %z
        jnz %wzero, .Lexit
        addl %i, %i, %one
        jmp .Lloop
    @label .Lexit:
        mov %i, %rax
        ret
