stats_f:
push rbx
push r12
push r13
push r14
push r15
push rbp
push rcx
mov rbx, rdi
mov r12, rdx
mov r13, rsi
xor r9, r9
xor r8, r8
xor rbp, rbp
.Lloop:
cmp r9, r13
je .Ldone
mov rdi, qword ptr [rbx + r9*8]
call r12
add rbp, rax
test rax, 1
jz .Lodd_even
inc r8
.Lodd_even:
cmp r9, 0
je .Lfirst
cmp rax, r14
jl .Lset_min
cmp rax, r15
jg .Lset_max
jmp .Lnext
.Lset_min:
mov r14, rax
jmp .Lnext
.Lset_max:
mov r15, rax
.Lnext:
inc r9
jmp .Lloop
.Lfirst:
mov r14, rax
mov r15, rax
inc r9
jmp .Lloop
.Ldone:
pop rcx
mov [rcx], r14
mov [rcx+8], r15
mov [rcx+16], rbp
mov rax, r8
pop rbp
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
