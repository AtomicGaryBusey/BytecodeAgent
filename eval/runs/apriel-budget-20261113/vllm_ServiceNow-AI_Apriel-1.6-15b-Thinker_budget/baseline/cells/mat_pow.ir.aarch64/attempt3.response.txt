export function l $mat_pow(l a, l n, l k, l out, l tmp) {
@label entry:
    mull %nn, n, n
    %idx = 0
@label zero_out_loop:
    csltl %w_idx, %idx, %nn
    jnz %w_idx, zero_out_body
    @label zero_out_done:
    %i = 0
@label diag_loop:
    csltl %w_i, %i, n
    jnz %w_i, diag_body
    @label diag_done:
    addl %tmp_len, %nn, %nn
    @label check_loop:
    csltl %w_k, 0, k
    jnz %w_k, do_mul
    @label after_loop:
    %idx = 0
@label zero_tmp_loop:
    csltl %w_idx, %idx, %tmp_len
    jnz %w_idx, zero_tmp_body
    @label ret:
    ret l 0
@label zero_tmp_body:
    addl %addr_tmp, tmp, %idx
    storel 0, %addr_tmp
    addl %idx, %idx, 1
    jmp zero_tmp_loop
@label do_mul:
    %i = 0
@label mul_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, mul_i_body
    @label mul_i_done:
    %idx = 0
@label copy_loop:
    csltl %w_idx, %idx, %nn
    jnz %w_idx, copy_body
    @label copy_done:
    subl k, k, 1
    jmp check_loop
@label copy_body:
    addl %addr_tmp, tmp, %idx
    addl %addr_out, out, %idx
    loadl %val, %addr_tmp
    storel %val, %addr_out
    addl %idx, %idx, 1
    jmp copy_loop
@label mul_i_body:
    mull %i_n, %i, %nn
    addl %addr_out_row, out, %i_n
    addl %addr_tmp_row, tmp, %i_n
    %j = 0
@label mul_j_loop:
    csltl %w_j, %j, n
    jnz %w_j, mul_j_body
    @label mul_j_done:
    addl %i, %i, 1
    jmp mul_i_loop
@label mul_j_body:
    %sum = 0
    %t = 0
@label mul_t_loop:
    csltl %w_t, %t, n
    jnz %w_t, mul_t_body
    @label mul_t_done:
    addl %addr_tmp, %addr_tmp_row, %j
    storel %sum, %addr_tmp
    addl %j, %j, 1
    jmp mul_j_loop
@label mul_t_body:
    addl %addr_out_elem, %addr_out_row, %t
    loadl %left, %addr_out_elem
    mull %t_n, %t, %nn
    addl %addr_a_row, a, %t_n
    addl %addr_a_elem, %addr_a_row, %j
    loadl %right, %addr_a_elem
    mull %prod, %left, %right
    addl %sum, %sum, %prod
    addl %t, %t, 1
    jmp mul_t_loop
@label zero_out_body:
    addl %addr_out, out, %idx
    storel 0, %addr_out
    addl %idx, %idx, 1
    jmp zero_out_loop
@label diag_body:
    mull %i_n, %i, %nn
    addl %addr_out, out, %i_n
    addl %addr_out, %addr_out, %i
    storel 1, %addr_out
    addl %i, %i, 1
    jmp diag_loop
}
