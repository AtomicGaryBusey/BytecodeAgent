export function l $mat_pow(l a, l n, l k, l out, l tmp) {
@label entry:
    mull %nn, n, n
    mull %tmp_len, %nn, 2
    %idx = 0
@label zero_out_loop:
    csltl %w_idx, %idx, %nn
    jnz %w_idx, zero_out_body
    @label zero_out_done:
    %i = 0
@label diag_loop:
    csltl %w_i, %i, n
    jnz %w_i, diag_body
    @label diag_done:
    %cnt = k
    @label loop_start:
    csltl %w_cnt, 0, %cnt
    jnz %w_cnt, loop_body
    @label after_loop:
    %idx = 0
@label zero_tmp_loop:
    csltl %w_idx, %idx, %tmp_len
    jnz %w_idx, zero_tmp_body
    @label zero_tmp_done:
    ret l 0
    @label zero_tmp_body:
    addl %addr_tmp, tmp, %idx
    storel 0, %addr_tmp
    addl %idx, %idx, 1
    jmp zero_tmp_loop
    @label zero_out_body:
    addl %addr_out, out, %idx
    storel 0, %addr_out
    addl %idx, %idx, 1
    jmp zero_out_loop
    @label diag_body:
    mull %i_nn, %i, %nn
    addl %off, %i_nn, %i
    storel 1, %off
    addl %i, %i, 1
    jmp diag_loop
    @label loop_body:
    %i = 0
    @label mul_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, mul_i_body
    @label mul_i_done:
    %idx = 0
    @label copy_loop:
    csltl %w_idx, %idx, %nn
    jnz %w_idx, copy_body
    @label copy_done:
    subl %cnt, %cnt, 1
    jmp loop_start
    @label copy_body:
    addl %addr_tmp, tmp, %idx
    loadl %val, %addr_tmp
    addl %addr_out, out, %idx
    storel %val, %addr_out
    addl %idx, %idx, 1
    jmp copy_loop
    @label mul_i_body:
    mull %i_nn, %i, %nn
    addl %addr_out_i, out, %i_nn
    addl %addr_tmp_i, tmp, %i_nn
    %j = 0
    @label mul_j_loop:
    csltl %w_j, %j, n
    jnz %w_j, mul_j_body
    @label mul_j_done:
    addl %i, %i, 1
    jmp mul_i_loop
    @label mul_j_body:
    %sum = 0
    %t = 0
    @label mul_t_loop:
    csltl %w_t, %t, n
    jnz %w_t, mul_t_body
    @label mul_t_done:
    addl %addr_tmp_elem, %addr_tmp_i, %j
    storel %sum, %addr_tmp_elem
    addl %j, %j, 1
    jmp mul_j_loop
    @label mul_t_body:
    addl %addr_a_elem, %addr_out_i, %t
    loadl %left, %addr_a_elem
    mull %t_nn, %t, %nn
    addl %addr_b_elem, a, %t_nn
    addl %addr_b_elem, %addr_b_elem, %j
    loadl %right, %addr_b_elem
    mull %prod, %left, %right
    addl %sum, %sum, %prod
    addl %t, %t, 1
    jmp mul_t_loop
}
