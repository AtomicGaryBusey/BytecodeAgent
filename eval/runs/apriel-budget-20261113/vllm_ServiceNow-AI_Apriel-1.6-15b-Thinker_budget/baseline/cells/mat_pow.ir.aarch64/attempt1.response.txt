export function l $mat_pow(l a, l n, l k, l out, l tmp) {
@label entry:
    mull %nn, n, n
    addl %tmp_copy, tmp, %nn
    %i = 0
@label zero_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, zero_i_body
    @label zero_i_done:
    %i = 0
@label diag_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, diag_i_body
    @label init_done:
    @label loop_start:
    csltl %w_k, 0, k
    jnz %w_k, do_mul
    @label after_loop:
    addl %tmp_len, %nn, %nn
    %idx = 0
@label zero_tmp_loop:
    csltl %w_idx, %idx, %tmp_len
    jnz %w_idx, zero_tmp_body
    @label ret_label:
    ret l 0
@label do_mul:
    %i = 0
@label copy_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, copy_i_body
    @label copy_done:
    %i = 0
@label mul_i_loop:
    csltl %w_i, %i, n
    jnz %w_i, mul_i_body
    @label decrement_k:
    subl k, k, 1
    jmp loop_start
@label zero_i_body:
    %j = 0
@label zero_j_loop:
    csltl %w_j, %j, n
    jnz %w_j, zero_j_body
    addl %i, %i, 1
    jmp zero_i_loop
@label zero_j_body:
    mull %i_n, %i, n
    addl %addr_out, out, %i_n
    addl %addr_out, %addr_out, %j
    storel 0, %addr_out
    addl %j, %j, 1
    jmp zero_j_loop
@label diag_i_body:
    mull %i_n, %i, n
    addl %addr_out, out, %i_n
    addl %addr_out, %addr_out, %i
    storel 1, %addr_out
    addl %i, %i, 1
    jmp diag_i_loop
@label copy_i_body:
    %j = 0
@label copy_j_loop:
    csltl %w_j, %j, n
    jnz %w_j, copy_j_body
    addl %i, %i, 1
    jmp copy_i_loop
@label copy_j_body:
    mull %i_n, %i, n
    addl %addr_out, out, %i_n
    addl %addr_out, %addr_out, %j
    loadl %val, %addr_out
    addl %addr_tmp, %tmp_copy, %i_n
    addl %addr_tmp, %addr_tmp, %j
    storel %val, %addr_tmp
    addl %j, %j, 1
    jmp copy_j_loop
@label mul_i_body:
    %j = 0
    %sum = 0
@label mul_j_loop:
    csltl %w_j, %j, n
    jnz %w_j, mul_j_body
    addl %i, %i, 1
    jmp mul_i_loop
@label mul_j_body:
    mull %i_n, %i, n
    addl %addr_left_base, %tmp_copy, %i_n
    addl %addr_out_base, out, %i_n
    %t = 0
@label mul_t_loop:
    csltl %w_t, %t, n
    jnz %w_t, mul_t_body
    addl %addr_out, %addr_out_base, %j
    storel %sum, %addr_out
    addl %j, %j, 1
    jmp mul_j_loop
@label mul_t_body:
    mull %t_n, %t, n
    addl %addr_left, %addr_left_base, %t
    loadl %left_val, %addr_left
    addl %addr_right, a, %t_n
    addl %addr_right, %addr_right, %j
    loadl %right_val, %addr_right
    mull %prod, %left_val, %right_val
    addl %sum, %sum, %prod
    addl %t, %t, 1
    jmp mul_t_loop
@label zero_tmp_body:
    addl %addr_tmp, tmp, %idx
    storel 0, %addr_tmp
    addl %idx, %idx, 1
    jmp zero_tmp_loop
}
