export function l $bubble_sort(l arr, l len) @L0:
    movl arr, %ptr
    movl len, %n
    subl %n, 1, %i
@Louter:
    csltl %i, 1, %w0
    jnz @Ldone
    movl 0, %j
    movl %ptr, %addr
@Linner:
    csltl %j, %i, %w0
    jnz @Linner_body
    subl %i, 1, %i
    jmp @Louter
@Linner_body:
    loadl %addr, %val1
    addl %addr, 8, %addr2
    loadl %addr2, %val2
    csltl %val2, %val1, %w0
    jnz @Lswap
    addl %j, 1, %j
    addl %addr, 8, %addr
    jmp @Linner
@Lswap:
    storel %val2, %addr
    storel %val1, %addr2
    addl %j, 1, %j
    addl %addr, 8, %addr
    jmp @Linner
@Ldone:
    movl 0, %0
    ret
