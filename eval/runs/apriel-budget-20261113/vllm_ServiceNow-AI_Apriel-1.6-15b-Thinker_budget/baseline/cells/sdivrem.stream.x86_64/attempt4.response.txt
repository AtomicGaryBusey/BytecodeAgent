sdivrem:
    test rsi, rsi
    jne .LnotZeroDiv
    xor rax, rax
    mov r8, rdx
    mov qword ptr [r8], rdi
    ret
.LnotZeroDiv:
    cmp rdi, 0x8000000000000000
    jne .LnormalDiv
    cmp rsi, -1
    jne .LnormalDiv
    mov rax, 0x8000000000000000
    mov r8, rdx
    xor rcx, rcx
    mov qword ptr [r8], rcx
    ret
.LnormalDiv:
    mov r8, rdx
    mov rax, rdi
    idiv rsi
    mov qword ptr [r8], rdx
    ret
