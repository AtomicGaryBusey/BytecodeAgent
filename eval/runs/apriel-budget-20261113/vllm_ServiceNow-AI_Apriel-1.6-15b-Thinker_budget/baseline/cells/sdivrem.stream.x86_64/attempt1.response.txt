sdivrem:
    cmp rsi, 0
    je .LbZero
    cmp rdi, 0x8000000000000000
    je .LaMinusInf
    jmp .LdoDiv
.LaMinusInf:
    cmp rsi, -1
    jne .LdoDiv
    mov rax, 0x8000000000000000
    mov qword ptr [rdx], 0
    ret
.LbZero:
    xor rax, rax
    mov qword ptr [rdx], rdi
    ret
.LdoDiv:
    mov rcx, rdx
    mov rax, rdi
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
