sdivrem:
    cmp rsi, 0
    je .LzeroDiv
    cmp rsi, -1
    jne .LnormalDiv
    cmp rdi, -0x8000000000000000
    je .LminDivMinusOne
    jmp .LnormalDiv
.LminDivMinusOne:
    mov rax, rdi
    mov rcx, rdx
    mov qword ptr [rcx], 0
    ret
.LzeroDiv:
    xor rax, rax
    mov rcx, rdx
    mov qword ptr [rcx], rdi
    ret
.LnormalDiv:
    mov rcx, rdx
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
