sdivrem:
    mov rcx, rdx
    test rsi, rsi
    jne .LnotZeroDiv
    xor rax, rax
    mov qword ptr [rcx], rdi
    ret
.LnotZeroDiv:
    cmp rdi, 0x8000000000000000
    jne .LdoDiv
    cmp rsi, 0xffffffffffffffff
    jne .LdoDiv
    mov rax, 0x8000000000000000
    xor rdx, rdx
    mov qword ptr [rcx], rdx
    ret
.LdoDiv:
    mov rax, rdi
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
