export function l $lcs_len(l, l, l, l, l)
{
    @0:
        %zero =w mov $0, %zero
        %addr0 =w mov %row, %addr0
        storel %zero, %addr0
        %k =w mov %n, %k
    @init_loop:
        csltl $0, %k, %gt
        jnz %gt, @store_zero
        %i =w mov %m, %i
    @outer_loop:
        csltl $0, %i, %gt_i
        jnz %gt_i, @outer_body
        %row_idx_n =w mull %n, $8, %row_idx_n
        %addr_n =w addl %row, %row_idx_n, %addr_n
        %res =w loadl %addr_n
        mov %res, %0
        ret
    @outer_body:
        %diag =w mov $0, %diag
        %j =w mov $1, %j
    @inner_loop:
        %nplus1 =w addl %n, $1, %nplus1
        csltl %j, %nplus1, %le
        jnz %le, @inner_body
        %i =w subl %i, $1, %i
        %dummy =w mov $1, %dummy
        jnz %dummy, @outer_loop
    @inner_body:
        %x_addr =w addl %x, %i, %x_addr
        %xv =w loadub %x_addr
        %y_idx =w subl %j, $1, %y_idx
        %y_addr =w addl %y, %y_idx, %y_addr
        %yv =w loadub %y_addr
        %eq =w ceql %xv, %yv, %eq
        %row_idx_j =w mull %j, $8, %row_idx_j
        %addr_j =w addl %row, %row_idx_j, %addr_j
        %up =w loadl %addr_j
        %row_idx_jm1 =w subl %row_idx_j, $8, %row_idx_jm1
        %addr_jm1 =w addl %row, %row_idx_jm1, %addr_jm1
        %left =w loadl %addr_jm1
        %new =w mov $0, %new
        jnz %eq, @eq_true
        %gt =w csltl %left, %up, %gt
        jnz %gt, @gt_true
        %new =w mov %left, %new
        %dummy2 =w mov $1, %dummy2
        jnz %dummy2, @store_new
    @gt_true:
        %new =w mov %up, %new
    @store_new:
        storel %new, %addr_j
        %diag =w mov %up, %diag
        %j =w addl %j, $1, %j
        %dummy3 =w mov $1, %dummy3
        jnz %dummy3, @inner_loop
    @eq_true:
        %new =w addl %diag, $1, %new
        storel %new, %addr_j
        %diag =w mov %up, %diag
        %j =w addl %j, $1, %j
        %dummy4 =w mov $1, %dummy4
        jnz %dummy4, @inner_loop
    @store_zero:
        %row_idx_k =w mull %k, $8, %row_idx_k
        %addr_k =w addl %row, %row_idx_k, %addr_k
        storel %zero, %addr_k
        %k =w subl %k, $1, %k
        %dummy5 =w mov $1, %dummy5
        jnz %dummy5, @init_loop
}
