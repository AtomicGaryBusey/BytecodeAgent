export function l $lcs_len(l, l, l, l, l)
{
    @init:
        %zero =w mov $0, %zero
        %k =w mov $0, %k
        %nplus1 =w addl %n, $1, %nplus1
    @init_body:
        %row_idx_k =w mull %k, $8, %row_idx_k
        %addr_k =w addl %row, %row_idx_k, %addr_k
        storel %zero, %addr_k
        %k =w addl %k, $1, %k
        csltl %k, %nplus1, %lt_k
        jnz %lt_k, @init_body
    @outer_loop:
        %i =w mov $0, %i
        csltl %i, %m, %lt_i
        jnz %lt_i, @outer_body
        %row_idx_n =w mull %n, $8, %row_idx_n
        %addr_n =w addl %row, %row_idx_n, %addr_n
        %res =w loadl %addr_n
        mov %res, %0
        ret
    @outer_body:
        %diag =w mov $0, %diag
        %j =w mov $1, %j
    @inner_check:
        csltl %j, %nplus1, %lt_j
        jnz %lt_j, @inner_body
        %dummy_i =w mov $1, %dummy_i
        jnz %dummy_i, @outer_increment
    @inner_body:
        %x_addr =w addl %x, %i, %x_addr
        %xv =w loadub %x_addr
        %y_idx =w subl %j, $1, %y_idx
        %y_addr =w addl %y, %y_idx, %y_addr
        %yv =w loadub %y_addr
        %eq =w ceql %xv, %yv, %eq
        %row_idx_j =w mull %j, $8, %row_idx_j
        %addr_j =w addl %row, %row_idx_j, %addr_j
        %up =w loadl %addr_j
        %row_idx_jm1 =w subl %row_idx_j, $8, %row_idx_jm1
        %addr_jm1 =w addl %row, %row_idx_jm1, %addr_jm1
        %left =w loadl %addr_jm1
        jnz %eq, @eq_branch
        %gt =w csltl %left, %up, %gt
        jnz %gt, @gt_branch
        %new =w mov %left, %new
        %dummy =w mov $1, %dummy
        jnz %dummy, @store_new
    @gt_branch:
        %new =w mov %up, %new
        %dummy2 =w mov $1, %dummy2
        jnz %dummy2, @store_new
    @eq_branch:
        %new =w addl %diag, $1, %new
        %dummy3 =w mov $1, %dummy3
        jnz %dummy3, @store_new
    @store_new:
        storel %new, %addr_j
        %diag =w mov %up, %diag
        %j =w addl %j, $1, %j
        %dummy4 =w mov $1, %dummy4
        jnz %dummy4, @inner_check
    @outer_increment:
        %i =w addl %i, $1, %i
        %dummy5 =w mov $1, %dummy5
        jnz %dummy5, @outer_loop
}
