    cmp x1, #-1
    beq .Lint_min_check
    b .Lnormal
.Lb_zero:
    str x0, [x2]
    mov x0, #0
    ret
.Lint_min_check:
    cmp x0, #0x8000000000000000
    beq .Lint_min_div_minus_one
    b .Lnormal
.Lint_min_div_minus_one:

    str x9, [x2]
    ret
.Lnormal:

    sdiv x0, x9, x1
    smull x10, x11, x0, x1
