sdivrem:
    cbz x1, .Lb_zero
    cmp x1, -1
    beq .Lint_min_check
    b .Lnormal
.Lint_min_check:
    cmp x0, #0x8000000000000000
    beq .Lint_min_div_minus_one
    b .Lnormal
.Lnormal:
    mov x9, x0
    sdiv x0, x9, x1
    srem x9, x9, x1
    str x9, [x2, #0]
    ret
.Lb_zero:
    str x0, [x2, #0]
    mov x0, #0
    ret
.Lint_min_div_minus_one:
    mov x9, #0
    str x9, [x2, #0]
    ret
