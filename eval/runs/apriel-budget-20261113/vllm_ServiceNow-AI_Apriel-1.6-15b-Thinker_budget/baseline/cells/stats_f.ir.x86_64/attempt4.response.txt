export function l $stats_f(%a,%n,%f,%out) {
    @L0:
        movl %n, %cnt
        movl %a, %ptr_a
        movl %out, %out_ptr
        loadl %val, (%ptr_a)
        %y =l call %f(l %val)
        movl %y, %min
        movl %y, %max
        movl %y, %sum
        andl %y, l1, %odd_tmp
        csgtl %w_odd0, %odd_tmp, l0
        movl %w_odd0, %odd_cnt
        addl %ptr_a, l8, %ptr_a
        subl %cnt, l1, %cnt
        @Lloop:
        csgtl %w, %cnt, l0
        jnz @Lbody
        storel %min, (%out_ptr)
        addl %out_ptr, l8, %out1
        storel %max, (%out1)
        addl %out_ptr, l16, %out2
        storel %sum, (%out2)
        ret %odd_cnt
    @Lbody:
        loadl %val, (%ptr_a)
        %y =l call %f(l %val)
        addl %sum, %y, %sum
        csltl %w_min, %y, %min
        jnz @Lmin
        jmp @Lpostmin
    @Lmin:
        movl %y, %min
    @Lpostmin:
        csgtl %w_max, %y, %max
        jnz @Lmax
        jmp @Lpostmax
    @Lmax:
        movl %y, %max
    @Lpostmax:
        andl %y, l1, %odd_tmp2
        csgtl %w_odd2, %odd_tmp2, l0
        jnz @Lodd
        jmp @Lpostodd
    @Lodd:
        addl %odd_cnt, l1, %odd_cnt
    @Lpostodd:
        addl %ptr_a, l8, %ptr_a
        subl %cnt, l1, %cnt
        jmp @Lloop
}
