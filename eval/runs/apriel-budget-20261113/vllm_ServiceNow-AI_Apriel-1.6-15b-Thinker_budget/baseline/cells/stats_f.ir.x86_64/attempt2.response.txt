export function l $stats_f(%a,%n,%f,%out) {
    @L0:
        addl %n, 0, %cnt
        addl %a, 0, %ptr
        loadl %val, (%ptr)
        %y =l call %f(l %val)
        movl %y, %min
        movl %y, %max
        movl %y, %sum
        andl %y, 1, %odd
        addl %ptr, 8, %ptr
        subl %cnt, 1, %cnt
    @L1:
        csgtl %w, %cnt, 0
        jnz @L2
        jmp @Lend
    @L2:
        loadl %val, (%ptr)
        %y =l call %f(l %val)
        addl %sum, %y, %sum
        csltl %w_min, %y, %min
        jnz @L3
        jmp @L4
    @L3:
        movl %y, %min
    @L4:
        csgtl %w_max, %y, %max
        jnz @L5
        jmp @L6
    @L5:
        movl %y, %max
    @L6:
        andl %y, 1, %is_odd
        csgtl %w_odd, %is_odd, 0
        jnz @L7
        jmp @L8
    @L7:
        addl %odd, 1, %odd
    @L8:
        addl %ptr, 8, %ptr
        subl %cnt, 1, %cnt
        jmp @L1
    @Lend:
        storel %min, (%out)
        addl %out, 8, %out8
        storel %max, (%out8)
        addl %out, 16, %out16
        storel %sum, (%out16)
        ret %odd
}
