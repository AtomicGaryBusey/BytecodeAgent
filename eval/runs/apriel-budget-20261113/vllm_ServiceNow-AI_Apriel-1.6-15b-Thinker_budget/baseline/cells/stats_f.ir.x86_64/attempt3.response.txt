export function l $stats_f(%a,%n,%f,%out) {
    @L0:
        movl %n, %cnt
        movl %a, %ptr
        loadl %val, (%ptr)
        %y =l call %f(l %val)
        movl %y, %min
        movl %y, %max
        movl %y, %sum
        andl %y, l1, %odd
        addl %ptr, l8, %ptr
        subl %cnt, l1, %cnt
    @L1:
        csgtl %w, %cnt, l0
        jnz @L2
        storel %min, (%out)
        addl %out, l8, %out8
        storel %max, (%out8)
        addl %out, l16, %out16
        storel %sum, (%out16)
        ret %odd
    @L2:
        loadl %val, (%ptr)
        %y =l call %f(l %val)
        addl %sum, %y, %sum
        csltl %w_min, %y, %min
        jnz @L3
        jmp @L4
    @L3:
        movl %y, %min
    @L4:
        csgtl %w_max, %y, %max
        jnz @L5
        jmp @L6
    @L5:
        movl %y, %max
    @L6:
        andl %y, l1, %is_odd
        csgtl %w_odd, %is_odd, l0
        jnz @L7
        jmp @L8
    @L7:
        addl %odd, l1, %odd
    @L8:
        addl %ptr, l8, %ptr
        subl %cnt, l1, %cnt
        jmp @L1
}
