export function l $qsort_pairs(l %p, l %m)
@label $L0
%zero =l l 0
%one =l l 1
%two =l l 2
%sixteen =l l 16
%minus_one =l l -1

%c_base =w csltl %m %two
jnz $Lbase

%pivot_idx =l subl %m %one
%addr_pivot =addl %p imull %pivot_idx %sixteen
%pivot_key =loadl %addr_pivot
%pivot_val =loadl %addr_pivot+8

%i =l l 0
%j =l subl %pivot_idx %one

@label $Lloop
%c_neg =w csltl %j %zero
jnz $Lafterpartition

%addr_j =addl %p imull %j %sixteen
%key_j =loadl %addr_j
%val_j =loadl %addr_j+8

%c_key =w csltl %key_j %pivot_key
jnz $Lswap

%c_eq_key =w ceql %key_j %pivot_key
jnz $Lcheck_val
jmp $Lafter_j

$Lcheck_val:
%c_val =w csltl %val_j %pivot_val
jnz $Lswap
jmp $Lafter_j

$Lswap:
%addr_i =addl %p imull %i %sixteen
%tmp_key =loadl %addr_i
%tmp_val =loadl %addr_i+8
storel %key_j %addr_i
storel %val_j %addr_i+8
storel %tmp_key %addr_j
storel %tmp_val %addr_j+8
%i =l addl %i %one

$Lafter_j:
%j =l addl %j %minus_one
jnz $Lloop

$Lafterpartition:
%addr_i =addl %p imull %i %sixteen
%tmp_key =loadl %addr_i
%tmp_val =loadl %addr_i+8
storel %pivot_key %addr_i
storel %pivot_val %addr_i+8
storel %tmp_key %addr_pivot
storel %tmp_val %addr_pivot+8

%left_len =l %i
%temp =l subl %m %i
%right_len =l subl %temp %one

%r_left =l call $qsort_pairs(l %p, l %left_len)
%r_right =l call $qsort_pairs(l addl %p imull %i %sixteen, l %right_len)

@label $Lreturn
ret %zero

@label $Lbase
ret %zero
