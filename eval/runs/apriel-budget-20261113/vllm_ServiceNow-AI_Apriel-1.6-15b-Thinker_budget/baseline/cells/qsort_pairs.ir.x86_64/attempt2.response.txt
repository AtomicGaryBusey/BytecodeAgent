export function l $qsort_pairs(l %p, l %m)
{
@label $L0
    %zero =l l 0
    %one =l l 1
    %sixteen =l l 16
    %c1 =w csltl %m %one
    jnz $Lbase
    %c2 =w ceql %m %one
    jnz $Lbase
    %j =l l 1
    %limit =l subl %m %one
    %min_idx =l l 0
    %min_key =l loadl %p
    %min_val =l loadl %p+8
@label $Lcond
    %c =w csltl %j %limit
    jnz $Lbody
    @label $Lafterloop
    %off_min =l imull %min_idx %sixteen
    %addr_min =l addl %p %off_min
    %key0 =l loadl %p
    %val0 =l loadl %p+8
    %key_min =l loadl %addr_min
    %val_min =l loadl %addr_min+8
    storel %key0 %addr_min
    storel %val0 %addr_min+8
    storel %key_min %p
    storel %val_min %p+8
    %newp =l addl %p %sixteen
    %r =l call $qsort_pairs(l %newp, l subl %m %one)
    ret %r
    @label $Lbase
    ret %zero
    @label $Lbody
    %off =l imull %j %sixteen
    %addr =l addl %p %off
    %key =l loadl %addr
    %val =l loadl %addr+8
    %c1_body =w csltl %key %min_key
    jnz $Lupd
    %c2_body =w ceql %key %min_key
    jnz $Lcheckval
    jmp $Lnext
    @label $Lcheckval
    %c3_body =w csltl %val %min_val
    jnz $Lupd
    jmp $Lnext
    @label $Lupd
    %min_key =l %key
    %min_val =l %val
    %min_idx =l %j
    @label $Lnext
    %j =l addl %j %one
    jmp $Lcond
}
