export function l $count_desc(l $a, l $n)
    @L0:
    %tmp = csltl $n 2
    jnz %tmp @Lreturn0
    %i = 0
    %cnt = 0
    %nm1 = subl $n 1
    %ptr = $a
    @Lcheck:
    %lt = csltl %i %nm1
    jnz %lt @Lbody
    @Lreturn:
    retl %cnt
    @Lbody:
    %curr = loadl %ptr
    %ptr8 = addl %ptr 8
    %next = loadl %ptr8
    %gt = csltl %next %curr
    %cnt = addl %cnt %gt
    %i = addl %i 1
    %ptr = addl %ptr 8
    jmp @Lcheck
    @Lreturn0:
    retl 0
end
