isqrt:
    mov r9, rdi
    xor rsi, rsi
    mov rdx, 0xFFFFFFFF
.Lloop:
    cmp rsi, rdx
    jg .Ldone
    lea rcx, [rsi + rdx]
    shr rcx, 1
    mov rax, rcx
    mul rcx
    cmp rax, r9
    jle .Llow
    dec rcx
    mov rdx, rcx
    jmp .Lloop
.Llow:
    inc rcx
    mov rsi, rcx
    jmp .Lloop
.Ldone:
    mov rax, rdx
    ret
