isqrt:
    mov r9, rdi          # n
    xor rsi, rsi         # low = 0
    mov rdx, 0xFFFFFFFF  # high = 2^32-1
    xor r8, r8           # ans = 0
.Lloop:
    cmp rsi, rdx
    jg .Ldone
    lea rcx, [rsi + rdx] # mid = (low+high)/2
    shr rcx, 1
    mov rax, rcx
    mul rcx
    cmp rax, r9
    jle .Llow
    dec rcx
    mov rdx, rcx
    jmp .Lloop
.Llow:
    mov r8, rcx
    inc rcx
    mov rsi, rcx
    jmp .Lloop
.Ldone:
    mov rax, r8
    ret
