export function l $max_byte(%p, %n)
@L0:
    %max =l 0
    %i =l 0
@Lloop:
    %c =w csltl %i %n
    jnz @Lbody
    @Lend:
        ret %max
@Lbody:
    %addr =l addl %p, %i
    %b =l loadub %addr
    %c2 =w csltl %max %b
    jnz @Lupdate
    @Lnext:
    %i =l addl %i, 1
    jmp @Lloop
@Lupdate:
    %max =l %b
    jmp @Lnext
