export function l $vm_run(l code l len l stack l cap)
@start:
    %pc = 0
    %sp = 0
    %zero = 0
@loop:
    %lt = csltl %pc %len
    jnz %lt @end
    %opcode = loadl [%code + %pc]
    %eq0 = ceql %opcode 0
    jnz %eq0 @halt
    %eq1 = ceql %opcode 1
    jnz %eq1 @push
    %eq2 = ceql %opcode 2
    jnz %eq2 @add
    %eq3 = ceql %opcode 3
    jnz %eq3 @sub
    %eq4 = ceql %opcode 4
    jnz %eq4 @mul
    %eq5 = ceql %opcode 5
    jnz %eq5 @dup
    %eq6 = ceql %opcode 6
    jnz %eq6 @swap
    %eq7 = ceql %opcode 7
    jnz %eq7 @drop
    %eq8 = ceql %opcode 8
    jnz %eq8 @over
    %eq9 = ceql %opcode 9
    jnz %eq9 @lt_op
    %eq10 = ceql %opcode 10
    jnz %eq10 @jz
    %eq11 = ceql %opcode 11
    jnz %eq11 @jmp
    %pc = addl %pc 1
    jmp @loop
@halt:
    jmp @end
@push:
    %v = loadl [%code + %pc + 1]
    storel %v [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 2
    jmp @loop
@add:
    %sp_minus1 = subl %sp 1
    %b = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %sp_minus1 = subl %sp 1
    %a = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %sum = addl %a %b
    storel %sum [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@sub:
    %sp_minus1 = subl %sp 1
    %b = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %sp_minus1 = subl %sp 1
    %a = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %diff = subl %a %b
    storel %diff [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@mul:
    %sp_minus1 = subl %sp 1
    %b = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %sp_minus1 = subl %sp 1
    %a = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %prod = mull %a %b
    storel %prod [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@dup:
    %sp_minus1 = subl %sp 1
    %top = loadl [%stack + %sp_minus1]
    storel %top [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@swap:
    %sp_minus2 = subl %sp 2
    %second = loadl [%stack + %sp_minus2]
    %sp_minus1 = subl %sp 1
    %top = loadl [%stack + %sp_minus1]
    storel %second [%stack + %sp_minus1]
    storel %top [%stack + %sp_minus2]
    %pc = addl %pc 1
    jmp @loop
@drop:
    %sp = subl %sp 1
    %pc = addl %pc 1
    jmp @loop
@over:
    %sp_minus2 = subl %sp 2
    %second = loadl [%stack + %sp_minus2]
    storel %second [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@lt_op:
    %sp_minus1 = subl %sp 1
    %b = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %sp_minus1 = subl %sp 1
    %a = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %c = csltl %a %b
    storel %c [%stack + %sp]
    %sp = addl %sp 1
    %pc = addl %pc 1
    jmp @loop
@jz:
    %sp_minus1 = subl %sp 1
    %v = loadl [%stack + %sp_minus1]
    %sp = subl %sp 1
    %t = loadl [%code + %pc + 1]
    %eq0 = ceql %v 0
    jnz %eq0 @jz_zero
    %pc = addl %pc 2
    jmp @loop
@jz_zero:
    %pc = %t
    jmp @loop
@jmp:
    %t = loadl [%code + %pc + 1]
    %pc = %t
    jmp @loop
@end:
    %i = cap
@zero_loop:
    %lt = csltl %sp %i
    jnz %lt @store_zero
    jmp @after_zero
@store_zero:
    %addr = subl %i 1
    storel %zero [%stack + %addr]
    %i = subl %i 1
    %pos = csltl 0 %i
    jnz %pos @zero_loop
@after_zero:
    %eq_sp_zero = ceql %sp %zero
    jnz %eq_sp_zero @result_zero
    %addr_top = subl %sp 1
    %result = loadl [%stack + %addr_top]
    ret %result
@result_zero:
    %result = %zero
    ret %result
