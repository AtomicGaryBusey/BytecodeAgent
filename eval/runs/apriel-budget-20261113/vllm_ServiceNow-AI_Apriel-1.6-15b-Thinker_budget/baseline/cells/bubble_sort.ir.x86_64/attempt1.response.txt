export function l $bubble_sort(l a, l n) {
    @label entry:
    %i = 0
    %max_i = subl n, 1
    @label outer_loop:
    %tmp_ip1 = addl %i, 1
    %bound = subl n, %tmp_ip1
    %j = 0
    @label inner_loop:
    %j_next = addl %j, 1
    %addr_j = addl a, shll %j, 3
    %tmp1 = loadl %addr_j
    %addr_j1 = addl a, shll %j_next, 3
    %tmp2 = loadl %addr_j1
    %c = csltl %tmp2, %tmp1
    jnz %c, do_swap
    jmp after_swap
    @label do_swap:
    storel %addr_j, %tmp2
    storel %addr_j1, %tmp1
    jmp after_swap
    @label after_swap:
    %j = %j_next
    %c2 = csltl %j, %bound
    jnz %c2, inner_loop
    %i = addl %i, 1
    %c3 = csltl %i, %max_i
    jnz %c3, outer_loop
    ret l 0
}
