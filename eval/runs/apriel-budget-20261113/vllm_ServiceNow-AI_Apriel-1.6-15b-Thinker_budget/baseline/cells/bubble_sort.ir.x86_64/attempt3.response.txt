export function l $bubble_sort(l a, l n) {
    %ptr = a
    %nn = n
    %i = 0
    %max_i = subl %nn, 1
    @label outer_loop:
    %i1 = addl %i, 1
    %bound = subl %nn, %i1
    %j = 0
    @label inner_loop:
    %j1 = addl %j, 1
    %addr_j = addl %ptr, shll %j, 3
    %tmp1 = loadl %addr_j
    %addr_j1 = addl %ptr, shll %j1, 3
    %tmp2 = loadl %addr_j1
    %c = csltl %tmp2, %tmp1
    jnz %c, do_swap
    jmp after_swap
    @label do_swap:
    storel %addr_j, %tmp2
    storel %addr_j1, %tmp1
    jmp after_swap
    @label after_swap:
    %j = %j1
    %c2 = csltl %j, %bound
    jnz %c2, inner_loop
    %i = %i1
    %c3 = csltl %i, %max_i
    jnz %c3, outer_loop
    ret l 0
}
