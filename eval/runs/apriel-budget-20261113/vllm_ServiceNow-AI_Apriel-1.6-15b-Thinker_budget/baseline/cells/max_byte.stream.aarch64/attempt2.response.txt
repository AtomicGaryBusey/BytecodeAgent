max_byte:
    cbz x1, .Lzero
    ldrb w2, [x0]
    add x0, x0, #1
    sub x1, x1, #1
.Lloop:
    cbnz x1, .Lnext
    b .Lresult
.Lnext:
    ldrb w3, [x0]
    cmp w3, w2
    b.le .Lskip
    mov w2, w3
.Lskip:
    add x0, x0, #1
    sub x1, x1, #1
    b .Lloop
.Lresult:
    mov x0, w2
    ret
.Lzero:
    mov x0, #0
    ret
