crc32:
    mov rax, 0xFFFFFFFF
    mov rcx, rdi
    mov rdx, rsi
.Louter:
    test rcx, rcx
    jz .Ldone
    movzx edx, byte ptr [rdx]
    xor eax, edx
    mov r8d, 8
.Linner:
    test al, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Ldecr
.Lshift:
    shr eax, 1
.Ldecr:
    dec r8d
    jnz .Linner
    inc rdx
    dec rcx
    jnz .Louter
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
