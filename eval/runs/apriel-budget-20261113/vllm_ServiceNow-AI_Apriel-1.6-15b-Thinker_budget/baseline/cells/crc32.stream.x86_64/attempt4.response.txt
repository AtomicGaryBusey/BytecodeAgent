crc32:
    mov eax, 0xFFFFFFFF
    mov rcx, rsi
.Louter:
    test rcx, rcx
    jz .Ldone
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov r8d, 8
.Linner:
    test eax, 1
    jz .Lshift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Ldecr
.Lshift:
    shr eax, 1
.Ldecr:
    dec r8d
    jnz .Linner
    inc rdi
    dec rcx
    jmp .Louter
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
