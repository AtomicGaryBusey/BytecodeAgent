bmin_empty:
	cbz x1, .Lempty
	rdr x2, [x0]
	sub x1, x1, #1
	mov x3, #1
.Lloop:
	cbnz x1, .Lhas_more
	mov x0, x2
	ret
.Lhas_more:
	ldr x4, [x0, x3, LSL #3]
	cmp x4, x2
	b.lt .Lset_min
	add x3, x3, #1
	sub x1, x1, #1
	b .Lloop
.Lset_min:
	mov x2, x4
	add x3, x3, #1
	sub x1, x1, #1
	b .Lloop
.Lempty:
	mov x0, #0x7FFFFFFFFFFFFFFF
	ret
