export function l $packbits_encode(l %0, l %1, l %2, l %3) -> l {
@start:
%i =w 0
%out =w 0
@loop_start:
%cond_i_lt_n =w csltl %i, %1
jnz @packet_start
@end:
ret %out
@packet_start:
%src_i_addr =w addl %0, %i
%cur =w loadub %src_i_addr
%run_len =w 1
@run_loop:
%cond_lt128 =w csltl %run_len, 128
%nextk =w addl %i, %run_len
%cond_nextk =w csltl %nextk, %1
jnz @run_check_eq
jmp @run_end
@run_check_eq:
%next_addr =w addl %0, %nextk
%next_byte =w loadub %next_addr
%cond_eq =w ceql %next_byte, %cur
jnz @run_increment
jmp @run_end
@run_increment:
%run_len =w addl %run_len, 1
jmp @run_loop
@run_end:
%run_lt3 =w csltl %run_len, 3
jnz @literal_packet
@repeat_packet:
%remaining =w subl %3, %out
%cond_not_enough =w csltl %remaining, 2
jnz @error
%header =w subl 257, %run_len
%dst_addr =w addl %2, %out
storeb %header, %dst_addr
%repeat_byte =w %cur
%dst_addr2 =w addl %dst_addr, 1
storeb %repeat_byte, %dst_addr2
%out =w addl %out, 2
%i =w addl %i, %run_len
jmp @loop_start
@literal_packet:
%j =w %i
%L =w 0
@lit_scan:
%cond_j_lt_n =w csltl %j, %1
jnz @lit_check_len
jmp @lit_done
@lit_check_len:
%cond_L_lt_128 =w csltl %L, 128
jnz @lit_check_run
jmp @lit_done
@lit_check_run:
%src_j_addr =w addl %0, %j
%cur_j =w loadub %src_j_addr
%run_len_j =w 1
@runj_loop:
%cond_lt128_j =w csltl %run_len_j, 128
%nextk2 =w addl %j, %run_len_j
%cond_nextk2 =w csltl %nextk2, %1
jnz @runj_check_eq
jmp @runj_done
@runj_check_eq:
%next_addr2 =w addl %0, %nextk2
%next_byte2 =w loadub %next_addr2
%cond_eq2 =w ceql %next_byte2, %cur_j
jnz @runj_increment
jmp @runj_done
@runj_increment:
%run_len_j =w addl %run_len_j, 1
jmp @runj_loop
@runj_done:
%run_lt3_j =w csltl %run_len_j, 3
jnz @lit_increment
jmp @lit_done
@lit_increment:
%L =w addl %L, 1
%j =w addl %j, 1
jmp @lit_scan
@lit_done:
%needed =w addl %L, 1
%remaining =w subl %3, %out
%cond_not_enough =w csltl %remaining, %needed
jnz @error
%header_lit =w subl %L, 1
%dst_addr =w addl %2, %out
storeb %header_lit, %dst_addr
%copy_src =w %i
%copy_dst =w addl %dst_addr, 1
@copy_loop:
%src_addr =w addl %0, %copy_src
%byte =w loadub %src_addr
%dst_addr =w addl %2, %copy_dst
storeb %byte, %dst_addr
%copy_src =w addl %copy_src, 1
%copy_dst =w addl %copy_dst, 1
%cond_copy =w csltl %copy_src, %j
jnz @copy_loop
%out =w addl %out, %needed
%i =w addl %i, %L
jmp @loop_start
@error:
%err =w -1
ret %err
}}
