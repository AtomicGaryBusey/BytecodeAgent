export function l $packbits_encode(l %0, l %1, l %2, l %3) -> l {
@start:
%i =w 0
%out =w 0
@loop_start:
%cond_i_lt_n =w csltl %i, %1
jnz @process
jmp @done
@process:
%src =w %0
%cap =w %3
%out_ptr =w %2
%cur_addr =w addl %src, %i
%cur =w loadub %cur_addr
%run_len =w 1
@run_loop:
%cond_lt128 =w csltl %run_len, 128
jnz @run_check_next
jmp @run_done
@run_check_next:
%tmp_next_idx =w %run_len
%next_idx =w addl %i, %tmp_next_idx
%cond_next_lt_n =w csltl %next_idx, %1
jnz @run_check_eq
jmp @run_done
@run_check_eq:
%tmp_next_addr =w %next_idx
%next_addr =w addl %src, %tmp_next_addr
%next_byte =w loadub %next_addr
%cond_eq =w ceql %next_byte, %cur
jnz @run_increment
jmp @run_done
@run_increment:
%run_len =w addl %run_len, 1
jmp @run_loop
@run_done:
%run_lt3 =w csltl %run_len, 3
jnz @literal_packet
@repeat_packet:
%remaining =w subl %cap, %out
%cond_not_enough =w csltl %remaining, 2
jnz @error
%header =w subl 257, %run_len
%dst_addr =w addl %out_ptr, %out
storeb %header, %dst_addr
%dst_addr2 =w addl %dst_addr, 1
storeb %cur, %dst_addr2
%out =w addl %out, 2
%i =w addl %i, %run_len
jmp @loop_start
@literal_packet:
%j =w %i
%L =w 0
@lit_scan_start:
%cond_j_lt_n =w csltl %j, %1
jnz @lit_check_L
jmp @lit_done
@lit_check_L:
%cond_L_lt_128 =w csltl %L, 128
jnz @lit_check_run
jmp @lit_done
@lit_check_run:
%cur_j_addr =w addl %src, %j
%cur_j =w loadub %cur_j_addr
%run_len_j =w 1
@runj_loop:
%cond_lt128_j =w csltl %run_len_j, 128
jnz @runj_check_next
jmp @runj_done
@runj_check_next:
%tmp_next_idx_j =w %run_len_j
%next_idx_j =w addl %j, %tmp_next_idx_j
%cond_nextj_lt_n =w csltl %next_idx_j, %1
jnz @runj_check_eq
jmp @runj_done
@runj_check_eq:
%tmp_next_addr_j =w %next_idx_j
%next_addr_j =w addl %src, %tmp_next_addr_j
%next_byte_j =w loadub %next_addr_j
%cond_eq_j =w ceql %next_byte_j, %cur_j
jnz @runj_increment
jmp @runj_done
@runj_increment:
%run_len_j =w addl %run_len_j, 1
jmp @runj_loop
@runj_done:
%run_lt3_j =w csltl %run_len_j, 3
jnz @lit_increment
jmp @lit_done
@lit_increment:
%L =w addl %L, 1
%j =w addl %j, 1
jmp @lit_scan_start
@lit_done:
%needed =w addl %L, 1
%remaining =w subl %cap, %out
%cond_not_enough =w csltl %remaining, %needed
jnz @error
%header_lit =w subl %L, 1
%dst_addr =w addl %out_ptr, %out
storeb %header_lit, %dst_addr
%k =w 0
%dst_cur =w addl %dst_addr, 1
@copy_loop:
%cond_k_lt_L =w csltl %k, %L
jnz @copy_body
jmp @copy_done
@copy_body:
%src_idx =w addl %i, %k
%src_addr =w addl %src, %src_idx
%byte =w loadub %src_addr
storeb %byte, %dst_cur
%dst_cur =w addl %dst_cur, 1
%k =w addl %k, 1
jmp @copy_loop
@copy_done:
%out =w addl %out, %needed
%i =w addl %i, %L
jmp @loop_start
@error:
%err =w -1
ret %err
@done:
ret %out
}}
