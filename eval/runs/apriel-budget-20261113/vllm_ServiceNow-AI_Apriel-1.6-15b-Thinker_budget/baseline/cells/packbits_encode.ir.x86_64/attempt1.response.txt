export function l $packbits_encode(l %0, l %1, l %2, l %3) -> l {
@main_loop:
%i =w 0
%out =w 0
@check_i_n:
%cond_i_lt_n =w csltl %i, %1
jnz @packet_start
@end:
ret %out
@packet_start:
%src_i_addr =w addl %0, %i
%cur =w loadub %src_i_addr
%run_len =w 1
%k =w %i
@run_loop:
%nextk =w addl %k, 1
%cond_nextk =w csltl %nextk, %1
jnz @run_check2
jmp @run_done
@run_check2:
%next_addr =w addl %0, %nextk
%next_byte =w loadub %next_addr
%cond_eq =w ceql %next_byte, %cur
jnz @run_check3
jmp @run_done
@run_check3:
%cond_run_len =w csltl %run_len, 128
jnz @run_increment
@run_done:
%run_lt3 =w csltl %run_len, 3
jnz @literal_packet
@repeat_packet:
%remaining =w subl %3, %out
%cond_not_enough =w csltl %remaining, 2
jnz @error
%header =w subl 257, %run_len
%dst_addr =w addl %2, %out
storeb %header, %dst_addr
%repeat_byte =w loadub %src_i_addr
%dst_addr2 =w addl %dst_addr, 1
storeb %repeat_byte, %dst_addr2
%out =w addl %out, 2
%i =w addl %i, %run_len
jmp @main_loop
@literal_packet:
%j =w %i
%L =w 0
@lit_loop:
%cond1 =w csltl %j, %1
jnz @lit_check_len
jmp @lit_done
@lit_check_len:
%cond2 =w csltl %L, 128
jnz @lit_check_run
jmp @lit_done
@lit_check_run:
%runj =w 1
%k2 =w %j
%cur2_addr =w addl %0, %j
%cur2 =w loadub %cur2_addr
@runj_loop:
%nextk2 =w addl %k2, 1
%cond1j =w csltl %nextk2, %1
jnz @runj_check2
jmp @runj_done
@runj_check2:
%next_addr2 =w addl %0, %nextk2
%next_byte2 =w loadub %next_addr2
%cond2j =w ceql %next_byte2, %cur2
jnz @runj_check3
jmp @runj_done
@runj_check3:
%cond3j =w csltl %runj, 128
jnz @runj_increment
@runj_done:
%runj_lt3 =w csltl %runj, 3
jnz @lit_increment
jmp @lit_done
@runj_increment:
%runj =w addl %runj, 1
%k2 =w %nextk2
jmp @runj_loop
@lit_increment:
%j =w addl %j, 1
%L =w addl %L, 1
jmp @lit_loop
@lit_done:
%needed =w addl %L, 1
%remaining =w subl %3, %out
%cond_not_enough =w csltl %remaining, %needed
jnz @error
%header_lit =w subl 1, %L
%dst_addr =w addl %2, %out
storeb %header_lit, %dst_addr
%copy_src =w %i
%copy_dst =w %out
@copy_loop:
%addr_src =w addl %0, %copy_src
%b =w loadub %addr_src
%addr_dst =w addl %2, %copy_dst
storeb %b, %addr_dst
%copy_src =w addl %copy_src, 1
%copy_dst =w addl %copy_dst, 1
%cond_copy =w csltl %copy_src, %j
jnz @copy_loop
%out =w addl %out, %needed
%i =w addl %i, %L
jmp @main_loop
@error:
%err =w -1
ret %err
}}
