export function l $sdivrem(a, b, rem) {
    @label .Lstart
    %is_zero = ceql b, 0
    jnz %is_zero, .Lzero
    %is_min = ceql a, -9223372036854775808
    jnz %is_min, .Lcheck_minus_one
    %q = divl a, b
    %r = reml a, b
    storel %r, rem
    ret %q
    @label .Lcheck_minus_one
    %is_minus_one = ceql b, -1
    jnz %is_minus_one, .Lspecial
    %q = divl a, b
    %r = reml a, b
    storel %r, rem
    ret %q
    @label .Lspecial
    %q = -9223372036854775808
    %r = 0
    storel %r, rem
    ret %q
    @label .Lzero
    %q = 0
    %r = a
    storel %r, rem
    ret %q
}
