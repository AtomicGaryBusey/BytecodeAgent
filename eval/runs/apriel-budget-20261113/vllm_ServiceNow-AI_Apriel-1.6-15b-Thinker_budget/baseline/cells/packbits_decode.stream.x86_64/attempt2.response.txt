packbits_decode:
	xor r8, r8
	xor r9, r9
.Lstart:
	cmp r8, rsi
	jae .Ldone
	movsx rax, byte ptr [rdi + r8]
	inc r8
	test rax, 0x80
	jz .Lliteral
	cmp rax, -128
	je .Lnoop
	cmp r8, rsi
	jae .Lsrc_error
	movzx r11, byte ptr [rdi + r8]
	inc r8
	neg rax
	inc rax
	mov r10, rax
.Lrepeat_copy:
	cmp r9, rcx
	jae .Loverflow
	mov byte ptr [rdx + r9], r11b
	inc r9
	dec r10
	jnz .Lrepeat_copy
	cmp r9, rcx
	je .Loverflow
	jmp .Lstart
.Lliteral:
	inc rax
	mov r10, rsi
	sub r10, r8
	cmp rax, r10
	ja .Lsrc_error
.Lliteral_copy:
	cmp r9, rcx
	jae .Loverflow
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9], al
	inc r8
	inc r9
	dec rax
	jnz .Lliteral_copy
	cmp r9, rcx
	je .Loverflow
	jmp .Lstart
.Lnoop:
	jmp .Lstart
.Lsrc_error:
	mov rax, -2
	ret
.Loverflow:
	mov rax, -1
	ret
.Ldone:
	mov rax, r9
	ret
