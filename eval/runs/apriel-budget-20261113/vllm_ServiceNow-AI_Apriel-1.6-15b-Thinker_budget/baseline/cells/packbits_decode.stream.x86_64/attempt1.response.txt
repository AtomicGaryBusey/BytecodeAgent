packbits_decode:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx rax, byte ptr [rdi + r8]
    inc r8
    test rax, 0x80
    jz .Lliteral
    cmp rax, -128
    je .Lnoop
    cmp r8, rsi
    jae .Lsrc_error
    movzx r10, byte ptr [rdi + r8]
    inc r8
    neg rax
    inc rax
    mov r11, rcx
    sub r11, r9
    cmp rax, r11
    jle .Lrepeat_all
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jnz .Lrepeat_overflow_loop
    mov rax, -1
    ret
.Lrepeat_overflow_loop:
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jnz .Lrepeat_overflow_loop
    mov rax, -1
    ret
.Lrepeat_all:
    mov r11, rax
.Lrepeat_all_loop:
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jnz .Lrepeat_all_loop
    jmp .Lloop
.Lliteral:
    mov r10, rax
    inc r10
    lea r11, [r8 + r10]
    cmp r11, rsi
    ja .Lsrc_error
    mov r11, rcx
    sub r11, r9
    cmp r10, r11
    jle .Lliteral_all
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Lliteral_overflow_loop
    mov rax, -1
    ret
.Lliteral_overflow_loop:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Lliteral_overflow_loop
    mov rax, -1
    ret
.Lliteral_all:
    mov r11, r10
.Lliteral_all_loop:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Lliteral_all_loop
    jmp .Lloop
.Lnoop:
    jmp .Lloop
.Lsrc_error:
    mov rax, -2
    ret
.Ldone:
    mov rax, r9
    ret
