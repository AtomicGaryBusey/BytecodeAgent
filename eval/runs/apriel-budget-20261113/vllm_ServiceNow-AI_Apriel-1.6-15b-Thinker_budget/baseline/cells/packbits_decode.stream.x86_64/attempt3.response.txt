packbits_decode:
	xor r8, r8
	xor r9, r9
.Lloop:
	cmp r8, rsi
	jae .Ldone
	movsx r10, byte ptr [rdi + r8]
	inc r8
	test r10, 0x80
	jz .Lliteral
	cmp r10, -128
	je .Lnoop
	cmp r8, rsi
	jae .Lsrc_error
	movzx r11, byte ptr [rdi + r8]
	inc r8
	neg r10
	inc r10
	mov rax, rcx
	sub rax, r9
	cmp r10, rax
	ja .Lrepeat_overflow
.Lrepeat_copy:
	mov byte ptr [rdx + r9], r11b
	inc r9
	dec r10
	jnz .Lrepeat_copy
	jmp .Lloop
.Lrepeat_overflow:
	test rax, rax
	jz .Lrepeat_done
	mov r10, rax
.Lrepeat_partial:
	mov byte ptr [rdx + r9], r11b
	inc r9
	dec r10
	jnz .Lrepeat_partial
.Lrepeat_done:
	mov rax, -1
	ret
.Lliteral:
	inc r10
	mov rax, rsi
	sub rax, r8
	cmp r10, rax
	ja .Lsrc_error
	mov rax, rcx
	sub rax, r9
	cmp r10, rax
	ja .Lliteral_overflow
.Lliteral_copy:
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9], al
	inc r8
	inc r9
	dec r10
	jnz .Lliteral_copy
	jmp .Lloop
.Lliteral_overflow:
	test rax, rax
	jz .Lliteral_done
	mov r10, rax
.Lliteral_partial:
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9], al
	inc r8
	inc r9
	dec r10
	jnz .Lliteral_partial
.Lliteral_done:
	mov rax, -1
	ret
.Lnoop:
	jmp .Lloop
.Lsrc_error:
	mov rax, -2
	ret
.Ldone:
	mov rax, r9
	ret
