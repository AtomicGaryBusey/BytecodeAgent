export function l $packbits_decode(l src, l n, l out, l cap) {
  %src_idx =w 0
  %out_idx =w 0
@Lloop:
  %has_src =w csltl %src_idx, %n
  jnz Lprocess %has_src
  ret %out_idx
@Lprocess:
  %addr =w addl src, %src_idx
  %h =w loadsb %addr
  %is_nop =w ceql %h, -128
  jnz Lnop %is_nop
  %is_neg =w csltl %h, 0
  jnz Lrepeat %is_neg
  %len =w addl %h, 1
  %need =w addl %src_idx, %len
  %src_ok =w csltl %need, %n
  jnz Lliteral_check %src_ok
  %err =w -2
  ret %err
@Lliteral_check:
  %out_need =w addl %out_idx, %len
  %overflow =w csltl %cap, %out_need
  jnz Lliteral_overflow %overflow
  %src_ptr =w addl src, %src_idx
  %src_ptr =w addl %src_ptr, 1
  %out_ptr =w addl out, %out_idx
  %cnt =w %len
@Lliteral_loop:
  %b =w loadub %src_ptr
  storeb %b %out_ptr
  %src_ptr =w addl %src_ptr, 1
  %out_ptr =w addl %out_ptr, 1
  %cnt =w subl %cnt, 1
  jnz Lliteral_loop %cnt
  %src_idx =w addl %src_idx, %len
  %src_idx =w addl %src_idx, 1
  %out_idx =w addl %out_idx, %len
  jmp Lloop
@Lliteral_overflow:
  %write_len =w subl %cap, %out_idx
  %src_ptr =w addl src, %src_idx
  %src_ptr =w addl %src_ptr, 1
  %out_ptr =w addl out, %out_idx
  %cnt =w %write_len
@Lliteral_overwrite_loop:
  %b =w loadub %src_ptr
  storeb %b %out_ptr
  %src_ptr =w addl %src_ptr, 1
  %out_ptr =w addl %out_ptr, 1
  %cnt =w subl %cnt, 1
  jnz Lliteral_overwrite_loop %cnt
  %err =w -1
  ret %err
@Lnop:
  %src_idx =w addl %src_idx, 1
  jmp Lloop
@Lrepeat:
  %need =w addl %src_idx, 1
  %src_ok =w csltl %need, %n
  jnz Lrepeat_copy_ok %src_ok
  %err =w -2
  ret %err
@Lrepeat_copy_ok:
  %len =w subl 1, %h
  %out_need =w addl %out_idx, %len
  %overflow =w csltl %cap, %out_need
  jnz Lrepeat_overflow %overflow
  %addr_val =w addl src, %src_idx
  %addr_val =w addl %addr_val, 1
  %val =w loadub %addr_val
  %out_ptr =w addl out, %out_idx
  %cnt =w %len
@Lrepeat_loop:
  storeb %val %out_ptr
  %out_ptr =w addl %out_ptr, 1
  %cnt =w subl %cnt, 1
  jnz Lrepeat_loop %cnt
  %src_idx =w addl %src_idx, 2
  %out_idx =w addl %out_idx, %len
  jmp Lloop
@Lrepeat_overflow:
  %write_len =w subl %cap, %out_idx
  %out_ptr =w addl out, %out_idx
  %cnt =w %write_len
@Lrepeat_overwrite_loop:
  storeb %val %out_ptr
  %out_ptr =w addl %out_ptr, 1
  %cnt =w subl %cnt, 1
  jnz Lrepeat_overwrite_loop %cnt
  %err =w -1
  ret %err
}
