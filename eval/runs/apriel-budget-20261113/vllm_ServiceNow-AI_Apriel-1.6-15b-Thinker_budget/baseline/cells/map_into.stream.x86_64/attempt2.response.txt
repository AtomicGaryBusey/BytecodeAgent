map_into:
push rbx
push r10
push r12
push r13
push r14
sub rsp, 16
mov QWORD PTR [rsp], 0
xor rbx, rbx
mov r10, rdi
mov r14, rsi
mov r13, rdx
mov r12, rcx
.Lloop:
cmp rbx, r13
jae .LEnd
mov rax, QWORD PTR [r14 + rbx*8]
mov QWORD PTR [rsp+8], rax
mov rdi, rax
call r12
mov QWORD PTR [r10 + rbx*8], rax
mov rdx, QWORD PTR [rsp+8]
cmp rax, rdx
jl .Linc
jmp .Linc_next
.Linc:
inc QWORD PTR [rsp]
.Linc_next:
inc rbx
jmp .Lloop
.LEnd:
mov rax, QWORD PTR [rsp]
add rsp, 16
pop r14
pop r13
pop r12
pop r10
pop rbx
ret
