export function l $max_byte(l l) {
@Entry:
    %max =l 0
    %i =l 0
    %cnt =l %n
    %one =l 1
    jmp Lloop
@Lloop:
    csltl %i %cnt
    jnz Lbody
    %0 =l %max
    ret
@Lbody:
    %addr =l add %p, %i
    %b =l loadub %addr
    csltl %max %b
    jnz Lsetmax
    %i =l add %i, %one
    jmp Lloop
@Lsetmax:
    %max =l %b
    %i =l add %i, %one
    jmp Lloop
}
