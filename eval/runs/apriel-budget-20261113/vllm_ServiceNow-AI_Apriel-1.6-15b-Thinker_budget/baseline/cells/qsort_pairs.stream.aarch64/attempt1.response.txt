qsort_pairs:
    stp x29, x30, [sp, #-16]
    stp x19, x20, [sp, #-16]
    stp x21, x22, [sp, #-16]
    mov x29, sp
    cmp x1, #2
    b le .Lbase_exit
    sub x2, x1, #1
    lsl x3, x2, #1
    add x3, x3, x0
    ldp x21, x22, [x3]
    mov x19, #0
    mov x20, #0
.Lpartition_loop:
    cmp x20, x2
    b.lt .Lpartition_body
    b .Lpost_partition
.Lpartition_body:
    lsl x4, x20, #1
    add x4, x4, x0
    ldp x5, x6, [x4]
    cmp x5, x21
    b.lt .Lswap_needed
    cmp x5, x21
    beq .Lcompare_values
    b .Lincrement_k
.Lcompare_values:
    cmp x6, x22
    b.lt .Lswap_needed
.Lincrement_k:
    add x20, x20, #1
    b .Lpartition_loop
.Lswap_needed:
    lsl x23, x19, #1
    add x23, x23, x0
    ldp x24, x25, [x23]
    stp x5, x6, [x23]
    stp x24, x25, [x4]
    add x19, x19, #1
    add x20, x20, #1
    b .Lpartition_loop
.Lpost_partition:
    cmp x19, x2
    beq .Lpivot_in_place
    lsl x23, x19, #1
    add x23, x23, x0
    ldp x5, x6, [x23]
    stp x21, x22, [x23]
    stp x5, x6, [x3]
.Lpivot_in_place:
    stp x1, x2, [sp, #-16]
    cmp x19, #0
    beq .Lskip_left
    stp x19, x20, [sp, #-16]
    stp x21, x22, [sp, #-16]
    mov x0, x0
    mov x1, x19
    bl qsort_pairs
    stp x21, x22, [sp, #+16]
    stp x19, x20, [sp, #+16]
.Lskip_left:
    sub x24, x2, x19
    cmp x24, #0
    beq .Lskip_right
    stp x19, x20, [sp, #-16]
    stp x21, x22, [sp, #-16]
    add x25, x19, #1
    lsl x25, x25, #1
    add x0, x0, x25
    mov x1, x24
    bl qsort_pairs
    stp x21, x22, [sp, #+16]
    stp x19, x20, [sp, #+16]
.Lskip_right:
    stp x1, x2, [sp, #+16]
    mov x0, #0
    stp x21, x22, [sp, #+16]
    stp x19, x20, [sp, #+16]
    stp x29, x30, [sp, #+16]
    ret
.Lbase_exit:
    mov x0, #0
    stp x21, x22, [sp, #+16]
    stp x19, x20, [sp, #+16]
    stp x29, x30, [sp, #+16]
    ret
