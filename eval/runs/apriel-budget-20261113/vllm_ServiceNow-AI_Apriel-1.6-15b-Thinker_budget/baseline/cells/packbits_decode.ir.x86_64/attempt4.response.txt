export function l $packbits_decode(%l, %l, %l, %l) {
    @start:
    %zero =w 0
    %one =w 1
    %src_idx =w %zero
    %out_idx =w %zero
    %src_ptr = %0
    %src_len = %1
    %out_ptr = %2
    %cap = %3
    @loop:
    %src_ok =w csltl %src_idx, %src_len
    jnz @read_header
    ret %out_idx
    @read_header:
    %h =w loadsb %src_ptr+%src_idx
    %src_idx =w add %src_idx, %one
    %is_noop =w ceql %h, -128
    jnz @skip_packet
    %is_neg =w csltl %h, %zero
    jnz @repeat_packet
    @literal_header:
    %cnt =w add %h, %one
    %tmp =w add %src_idx, %cnt
    %src_err =w csltl %src_len, %tmp
    jnz @literal_error
    @literal_copy:
    %cnt_gt0 =w csltl %zero, %cnt
    jnz @literal_body
    jmp @loop
    @literal_body:
    %out_ok =w csltl %out_idx, %cap
    jnz @literal_store
    ret -1
    @literal_store:
    %b =w loadub %src_ptr+%src_idx
    storeb %b, %out_ptr+%out_idx
    %src_idx =w add %src_idx, %one
    %out_idx =w add %out_idx, %one
    %cnt =w sub %cnt, %one
    jnz @literal_copy
    @skip_packet:
    jmp @loop
    @repeat_packet:
    %neg_h =w sub %zero, %h
    %repeat_cnt =w add %neg_h, %one
    %src_ok2 =w csltl %src_idx, %src_len
    jnz @repeat_value_ok
    ret -2
    @repeat_value_ok:
    %val =w loadub %src_ptr+%src_idx
    %src_idx =w add %src_idx, %one
    @repeat_copy:
    %repeat_cnt_gt0 =w csltl %zero, %repeat_cnt
    jnz @repeat_body
    jmp @loop
    @repeat_body:
    %out_ok =w csltl %out_idx, %cap
    jnz @repeat_store
    ret -1
    @repeat_store:
    storeb %val, %out_ptr+%out_idx
    %out_idx =w add %out_idx, %one
    %repeat_cnt =w sub %repeat_cnt, %one
    jnz @repeat_copy
    @literal_error:
    ret -2
}
