export function l $packbits_decode(%l, %l, %l, %l) {
    @start:
    %zero =w 0
    %one =w 1
    %src_idx =w %zero
    %out_idx =w %zero
    %src_ptr = %0
    %out_ptr = %2
    %src_len = %1
    %cap = %3
    @loop:
    %src_cond =w csltl %src_idx, %src_len
    jnz @read_header
    ret %out_idx
    @read_header:
    %h =w loadsb %src_ptr+%src_idx
    %src_idx =w addl %src_idx, %one
    %is_noop =w ceql %h, -128
    jnz @skip_packet
    %is_neg =w csltl %h, %zero
    jnz @repeat_packet
    %cnt =w addl %h, %one
    %tmp =w addl %src_idx, %cnt
    %src_err =w csltl %src_len, %tmp
    jnz @literal_error
    @literal_copy:
    %cond_copy =w csltl %zero, %cnt
    jnz @literal_copy_body
    jmp @loop
    @literal_copy_body:
    %out_cond =w csltl %out_idx, %cap
    jnz @literal_copy_store
    ret -1
    @literal_copy_store:
    %b =w loadub %src_ptr+%src_idx
    storeb %b, %out_ptr+%out_idx
    %src_idx =w addl %src_idx, %one
    %out_idx =w addl %out_idx, %one
    %cnt =w subl %cnt, %one
    jmp @literal_copy
    @skip_packet:
    jmp @loop
    @repeat_packet:
    %neg_h =w subl %zero, %h
    %repeat_cnt =w addl %neg_h, %one
    %src_cond2 =w csltl %src_idx, %src_len
    jnz @repeat_value_ok
    ret -2
    @repeat_value_ok:
    %val =w loadub %src_ptr+%src_idx
    %src_idx =w addl %src_idx, %one
    @repeat_copy:
    %cond_copy =w csltl %zero, %repeat_cnt
    jnz @repeat_copy_body
    jmp @loop
    @repeat_copy_body:
    %out_cond =w csltl %out_idx, %cap
    jnz @repeat_copy_store
    ret -1
    @repeat_copy_store:
    storeb %val, %out_ptr+%out_idx
    %out_idx =w addl %out_idx, %one
    %repeat_cnt =w subl %repeat_cnt, %one
    jmp @repeat_copy
    @literal_error:
    ret -2
}
