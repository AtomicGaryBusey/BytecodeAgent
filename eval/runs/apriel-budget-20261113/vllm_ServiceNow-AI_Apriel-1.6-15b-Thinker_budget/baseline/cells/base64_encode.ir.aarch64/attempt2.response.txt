export function l $base64_encode(l src, l n, l out)
@start:
%tmp =l add %n, 2
%blocks =l divl %tmp, 3
%out_len =l mul %blocks, 4
%full =l divl %n, 3
%rem =l sub %n, mul %full, 3
%zero =l sub %n, %n
%c_eq =l add %zero, 61
%i =l zero
@full_loop:
%tmp =l sub %i, %full
%lt =w csltl %tmp, 0
jnz %lt, @process_full_block
@after_full_loop:
%eq0 =w ceql %rem, 0
jnz %eq0, @end_encode
%eq1 =w ceql %rem, 1
jnz %eq1, @handle_one
; rem == 2
@handle_two:
%src_idx =l add %src, mul %full, 3
%out_idx =l add %out, mul %full, 4
%b0 =l loadub %src_idx
%b1 =l loadub add %src_idx, 1
%b2 =l %zero
%tmp1 =l shl %b0, 16
%tmp2 =l shl %b1, 8
%v24 =l or %tmp1, %tmp2
%v24 =l or %v24, %b2
%v0 =l shr %v24, 18
%v1 =l shr %v24, 12
%v1 =l and %v1, 63
%v2 =l shr %v24, 6
%v2 =l and %v2, 63
; map v0
%lt26 =w csltl %v0, 26
jnz %lt26, @v0_two_case0
%lt52 =w csltl %v0, 52
jnz %lt52, @v0_two_case1
%lt62 =w csltl %v0, 62
jnz %lt62, @v0_two_case2
%eq62 =w ceql %v0, 62
jnz %eq62, @v0_two_case3
%eq63 =w ceql %v0, 63
jnz %eq63, @v0_two_case4
@v0_two_case0:
%c0 =l add %v0, 65
jmp @store_v0_two
@v0_two_case1:
%c0 =l add %v0, 71
jmp @store_v0_two
@v0_two_case2:
%c0 =l sub %v0, 4
jmp @store_v0_two
@v0_two_case3:
%c0 =l sub %v0, 19
jmp @store_v0_two
@v0_two_case4:
%c0 =l sub %v0, 16
jmp @store_v0_two
@store_v0_two:
storeb %c0, add %out_idx, 0
; map v1
%lt26 =w csltl %v1, 26
jnz %lt26, @v1_two_case0
%lt52 =w csltl %v1, 52
jnz %lt52, @v1_two_case1
%lt62 =w csltl %v1, 62
jnz %lt62, @v1_two_case2
%eq62 =w ceql %v1, 62
jnz %eq62, @v1_two_case3
%eq63 =w ceql %v1, 63
jnz %eq63, @v1_two_case4
@v1_two_case0:
%c1 =l add %v1, 65
jmp @store_v1_two
@v1_two_case1:
%c1 =l add %v1, 71
jmp @store_v1_two
@v1_two_case2:
%c1 =l sub %v1, 4
jmp @store_v1_two
@v1_two_case3:
%c1 =l sub %v1, 19
jmp @store_v1_two
@v1_two_case4:
%c1 =l sub %v1, 16
jmp @store_v1_two
@store_v1_two:
storeb %c1, add %out_idx, 1
; map v2
%lt26 =w csltl %v2, 26
jnz %lt26, @v2_two_case0
%lt52 =w csltl %v2, 52
jnz %lt52, @v2_two_case1
%lt62 =w csltl %v2, 62
jnz %lt62, @v2_two_case2
%eq62 =w ceql %v2, 62
jnz %eq62, @v2_two_case3
%eq63 =w ceql %v2, 63
jnz %eq63, @v2_two_case4
@v2_two_case0:
%c2 =l add %v2, 65
jmp @store_v2_two
@v2_two_case1:
%c2 =l add %v2, 71
jmp @store_v2_two
@v2_two_case2:
%c2 =l sub %v2, 4
jmp @store_v2_two
@v2_two_case3:
%c2 =l sub %v2, 19
jmp @store_v2_two
@v2_two_case4:
%c2 =l sub %v2, 16
jmp @store_v2_two
@store_v2_two:
storeb %c2, add %out_idx, 2
storeb %c_eq, add %out_idx, 3
jmp @end_encode
@handle_one:
%src_idx =l add %src, mul %full, 3
%out_idx =l add %out, mul %full, 4
%b0 =l loadub %src_idx
%b1 =l %zero
%b2 =l %zero
%tmp1 =l shl %b0, 16
%tmp2 =l shl %b1, 8
%v24 =l or %tmp1, %tmp2
%v24 =l or %v24, %b2
%v0 =l shr %v24, 18
%v1 =l shr %v24, 12
%v1 =l and %v1, 63
; map v0
%lt26 =w csltl %v0, 26
jnz %lt26, @v0_one_case0
%lt52 =w csltl %v0, 52
jnz %lt52, @v0_one_case1
%lt62 =w csltl %v0, 62
jnz %lt62, @v0_one_case2
%eq62 =w ceql %v0, 62
jnz %eq62, @v0_one_case3
%eq63 =w ceql %v0, 63
jnz %eq63, @v0_one_case4
@v0_one_case0:
%c0 =l add %v0, 65
jmp @store_v0_one
@v0_one_case1:
%c0 =l add %v0, 71
jmp @store_v0_one
@v0_one_case2:
%c0 =l sub %v0, 4
jmp @store_v0_one
@v0_one_case3:
%c0 =l sub %v0, 19
jmp @store_v0_one
@v0_one_case4:
%c0 =l sub %v0, 16
jmp @store_v0_one
@store_v0_one:
storeb %c0, add %out_idx, 0
; map v1
%lt26 =w csltl %v1, 26
jnz %lt26, @v1_one_case0
%lt52 =w csltl %v1, 52
jnz %lt52, @v1_one_case1
%lt62 =w csltl %v1, 62
jnz %lt62, @v1_one_case2
%eq62 =w ceql %v1, 62
jnz %eq62, @v1_one_case3
%eq63 =w ceql %v1, 63
jnz %eq63, @v1_one_case4
@v1_one_case0:
%c1 =l add %v1, 65
jmp @store_v1_one
@v1_one_case1:
%c1 =l add %v1, 71
jmp @store_v1_one
@v1_one_case2:
%c1 =l sub %v1, 4
jmp @store_v1_one
@v1_one_case3:
%c1 =l sub %v1, 19
jmp @store_v1_one
@v1_one_case4:
%c1 =l sub %v1, 16
jmp @store_v1_one
@store_v1_one:
storeb %c1, add %out_idx, 1
storeb %c_eq, add %out_idx, 2
storeb %c_eq, add %out_idx, 3
jmp @end_encode
@process_full_block:
%src_base =l add %src, mul %i, 3
%out_base =l add %out, mul %i, 4
%b0 =l loadub %src_base
%b1 =l loadub add %src_base, 1
%b2 =l loadub add %src_base, 2
%tmp1 =l shl %b0, 16
%tmp2 =l shl %b1, 8
%v24 =l or %tmp1, %tmp2
%v24 =l or %v24, %b2
%v0 =l shr %v24, 18
%v1 =l shr %v24, 12
%v1 =l and %v1, 63
%v2 =l shr %v24, 6
%v2 =l and %v2, 63
%v3 =l and %v24, 63
; map v0
%lt26 =w csltl %v0, 26
jnz %lt26, @v0_case0
%lt52 =w csltl %v0, 52
jnz %lt52, @v0_case1
%lt62 =w csltl %v0, 62
jnz %lt62, @v0_case2
%eq62 =w ceql %v0, 62
jnz %eq62, @v0_case3
%eq63 =w ceql %v0, 63
jnz %eq63, @v0_case4
@v0_case0:
%c0 =l add %v0, 65
jmp @store_v0
@v0_case1:
%c0 =l add %v0, 71
jmp @store_v0
@v0_case2:
%c0 =l sub %v0, 4
jmp @store_v0
@v0_case3:
%c0 =l sub %v0, 19
jmp @store_v0
@v0_case4:
%c0 =l sub %v0, 16
jmp @store_v0
@store_v0:
storeb %c0, add %out_base, 0
; map v1
%lt26 =w csltl %v1, 26
jnz %lt26, @v1_case0
%lt52 =w csltl %v1, 52
jnz %lt52, @v1_case1
%lt62 =w csltl %v1, 62
jnz %lt62, @v1_case2
%eq62 =w ceql %v1, 62
jnz %eq62, @v1_case3
%eq63 =w ceql %v1, 63
jnz %eq63, @v1_case4
@v1_case0:
%c1 =l add %v1, 65
jmp @store_v1
@v1_case1:
%c1 =l add %v1, 71
jmp @store_v1
@v1_case2:
%c1 =l sub %v1, 4
jmp @store_v1
@v1_case3:
%c1 =l sub %v1, 19
jmp @store_v1
@v1_case4:
%c1 =l sub %v1, 16
jmp @store_v1
@store_v1:
storeb %c1, add %out_base, 1
; map v2
%lt26 =w csltl %v2, 26
jnz %lt26, @v2_case0
%lt52 =w csltl %v2, 52
jnz %lt52, @v2_case1
%lt62 =w csltl %v2, 62
jnz %lt62, @v2_case2
%eq62 =w ceql %v2, 62
jnz %eq62, @v2_case3
%eq63 =w ceql %v2, 63
jnz %eq63, @v2_case4
@v2_case0:
%c2 =l add %v2, 65
jmp @store_v2
@v2_case1:
%c2 =l add %v2, 71
jmp @store_v2
@v2_case2:
%c2 =l sub %v2, 4
jmp @store_v2
@v2_case3:
%c2 =l sub %v2, 19
jmp @store_v2
@v2_case4:
%c2 =l sub %v2, 16
jmp @store_v2
@store_v2:
storeb %c2, add %out_base, 2
; map v3
%lt26 =w csltl %v3, 26
jnz %lt26, @v3_case0
%lt52 =w csltl %v3, 52
jnz %lt52, @v3_case1
%lt62 =w csltl %v3, 62
jnz %lt62, @v3_case2
%eq62 =w ceql %v3, 62
jnz %eq62, @v3_case3
%eq63 =w ceql %v3, 63
jnz %eq63, @v3_case4
@v3_case0:
%c3 =l add %v3, 65
jmp @store_v3
@v3_case1:
%c3 =l add %v3, 71
jmp @store_v3
@v3_case2:
%c3 =l sub %v3, 4
jmp @store_v3
@v3_case3:
%c3 =l sub %v3, 19
jmp @store_v3
@v3_case4:
%c3 =l sub %v3, 16
jmp @store_v3
@store_v3:
storeb %c3, add %out_base, 3
%done =l add %i, 1
%i =l %done
jmp @full_loop
@end_encode:
ret %out_len
