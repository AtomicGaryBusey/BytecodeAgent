@handle_one:

%lt26 =w csltl %v0, 26

%lt52 =w csltl %v0, 52
jnz %lt52, @v0_one_B
%lt62 =w csltl %v0, 62

%eq63 =w ceql %v0, 63
jnz %eq63, @v0_one_E

@v0_one_B:

@v0_one_C:

jmp @store_v0_one
@v0_one_D:
%c0 =l sub %v0, 19
jmp @store_v0_one
@v0_one_E:
%c0 =l sub %v0, 16
jmp @store_v0_one
@store_v0_one:
%addr0_out =l %out_base
storeb %c0, %addr0_out
%lt26 =w csltl %v1, 26
jnz %lt26, @v1_one_A
%lt52 =w csltl %v1, 52

@store_v1_one:

storeb %c1, %addr1_out
%addr2_out =l add %out_base, 2
storeb %pad_eq, %addr2_out
%addr3_out =l add %out_base, 3
storeb %pad_eq, %addr3_out

@handle_two:

%lt62 =w csltl %v0, 62

@store_v0_full:

%lt26 =w csltl %v1, 26
jnz %lt26, @v1_full_A
