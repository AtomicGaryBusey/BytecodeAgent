export function l $count_nonzero_prefix(l a, l n)
@init:
  mov 0, %i
  mov 1, %inc
  j @loop
@loop:
  csltl %i, %n, %lt
  jnz %lt, @check_val
  ret %i
@check_val:
  addl %addr, %a, %i
  loadl %addr, %val
  ceql %val, 0, %eq
  csltl %eq, 1, %neq
  jnz %neq, @inc
  ret %i
@inc:
  addl %i, %i, %inc
  j @loop
