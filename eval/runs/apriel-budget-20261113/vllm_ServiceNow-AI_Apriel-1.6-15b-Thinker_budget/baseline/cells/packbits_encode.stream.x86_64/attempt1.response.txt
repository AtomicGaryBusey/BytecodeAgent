packbits_encode:
    sub rsp, 8
    mov [rsp], rcx
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r9, rsi
    jge .Ldone
    movzx r11, byte ptr [rdi + r9]
    mov r10, 1
.run_len:
    cmp r10, 128
    jge .run_len_done
    lea rax, [r9 + r10]
    cmp rax, rsi
    jge .run_len_done
    mov al, [rdi + rax]
    cmp al, r11b
    jne .run_len_done
    inc r10
    jmp .run_len
.run_len_done:
    cmp r10, 3
    jl .literal_packet
    mov rax, r8
    add rax, 2
    cmp rax, [rsp]
    jge .fail
    mov rax, 257
    sub rax, r10
    mov [rdx + r8], al
    mov [rdx + r8 + 1], [rdi + r9]
    lea r8, [r8 + 2]
    lea r9, [r9 + r10]
    jmp .Lloop
.literal_packet:
    xor r11, r11
    mov r10, r9
.find_literal:
    cmp r10, rsi
    jge .find_done
    cmp r11, 128
    jge .find_done
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .find_continue
    mov al, [rdi + r10]
    mov bl, [rdi + r10 + 1]
    cmp al, bl
    jne .find_continue
    mov bl, [rdi + rax]
    cmp al, bl
    jne .find_continue
    jmp .find_done
.find_continue:
    inc r10
    inc r11
    jmp .find_literal
.find_done:
    mov rax, r8
    add rax, 1
    add rax, r11
    cmp rax, [rsp]
    jge .fail
    mov rax, r11
    dec rax
    mov [rdx + r8], al
    push rsi
    push rdx
    mov rcx, r11
    lea rsi, [rdi + r9]
    lea rdi, [rdx + r8 + 1]
    rep movsb
    pop rdx
    pop rsi
    mov rcx, [rsp]
    lea r8, [r8 + 1]
    add r8, r11
    mov r9, r10
    jmp .Lloop
.fail:
    mov rax, -1
    mov rcx, [rsp]
    add rsp, 8
    ret
.Ldone:
    mov rax, r8
    mov rcx, [rsp]
    add rsp, 8
    ret
