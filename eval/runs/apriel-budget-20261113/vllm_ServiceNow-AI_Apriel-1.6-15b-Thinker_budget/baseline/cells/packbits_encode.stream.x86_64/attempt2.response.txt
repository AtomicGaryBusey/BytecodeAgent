packbits_encode:
    xor r8, r8
    xor r9, r9
.Louter:
    cmp r8, rsi
    jge .Ldone
    movzx r11, byte ptr [rdi + r8]
    mov r10, 1
.run_loop:
    cmp r10, 128
    jge .run_done
    mov rax, r8
    add rax, r10
    cmp rax, rsi
    jge .run_done
    mov al, [rdi + rax]
    cmp al, r11b
    jne .run_done
    inc r10
    jmp .run_loop
.run_done:
    cmp r10, 3
    jl .literal_packet
    mov rax, r9
    add rax, 2
    cmp rax, rcx
    jg .fail
    mov rax, 257
    sub rax, r10
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r11b
    add r9, 2
    lea r8, [r8 + r10]
    jmp .Louter
.literal_packet:
    xor r10, r10
    push rdx
.LiteralFindLoop:
    cmp r10, 128
    jge .LiteralDoneFind
    mov rax, r8
    add rax, r10
    cmp rax, rsi
    jge .LiteralDoneFind
    mov rdx, rax
    add rdx, 2
    cmp rdx, rsi
    jge .LiteralAppend
    movzx r11, byte ptr [rdi + rax]
    movzx rdx, byte ptr [rdi + rax + 1]
    cmp r11, rdx
    jne .LiteralAppend
    movzx rdx, byte ptr [rdi + rax + 2]
    cmp r11, rdx
    jne .LiteralAppend
    jmp .LiteralDoneFind
.LiteralAppend:
    inc r10
    jmp .LiteralFindLoop
.LiteralDoneFind:
    pop rdx
    mov rax, r9
    add rax, r10
    add rax, 1
    cmp rax, rcx
    jg .fail
    mov al, r10b
    dec al
    mov byte ptr [rdx + r9], al
    add r9, 1
.LiteralCopyLoop:
    test r10, r10
    jz .LiteralCopyDone
    mov al, [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    add r9, 1
    dec r10
    jmp .LiteralCopyLoop
.LiteralCopyDone:
    jmp .Louter
.Ldone:
    mov rax, r9
    ret
.fail:
    mov rax, -1
    ret
