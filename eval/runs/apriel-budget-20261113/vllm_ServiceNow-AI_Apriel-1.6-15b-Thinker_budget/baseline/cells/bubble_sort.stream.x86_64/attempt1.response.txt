bubble_sort:
    mov rdx, rsi
    dec rdx
    xor rcx, rcx
.Louter:
    cmp rcx, rdx
    jge .Ldone
    mov r10, rsi
    sub r10, rcx
    dec r10
    xor r8, r8
.Linner:
    mov rax, qword ptr [rdi + r8*8]
    mov r9, qword ptr [rdi + r8*8 + 8]
    cmp rax, r9
    jle Jnext
    mov qword ptr [rdi + r8*8], r9
    mov qword ptr [rdi + r8*8 + 8], rax
Jnext:
    inc r8
    cmp r8, r10
    jl .Linner
    inc rcx
    jmp .Louter
.Ldone:
    xor rax, rax
    ret
