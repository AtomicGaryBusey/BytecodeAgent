export function l $bmax(l a, l n) {
    @label .Lstart:
    %max: = loadl a
    %p: = a
    %cnt: = subl n, 1
    @label .Lloop:
    %c0: = ceql %cnt, 0
    jnz .Lend
    %p: = addl %p, 8
    %val: = loadl %p
    %c1: = csgtl %val, %max
    jnz .Lupd
    jmp .Ldecr
    @label .Lupd:
    %max: = %val
    @label .Ldecr:
    %cnt: = subl %cnt, 1
    jmp .Lloop
    @label .Lend:
    ret %max
}
