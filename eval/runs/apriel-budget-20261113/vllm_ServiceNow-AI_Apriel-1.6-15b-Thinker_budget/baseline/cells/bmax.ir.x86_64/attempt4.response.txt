export function l $bmax(l,l) {
    @label Lstart:
    %a: = rdi
    %n: = rsi
    %max: = loadl [%a]
    %p: = %a
    %cnt: = subl %n, 1
    @label Lloop:
    %c0: = ceql %cnt, 0
    jnz Lend
    %p: = addl %p, 8
    %val: = loadl [%p]
    %c1: = csltl %max, %val
    jnz Lupd
    jmp Ldecr
    @label Lupd:
    %max: = %val
    @label Ldecr:
    %cnt: = subl %cnt, 1
    jmp Lloop
    @label Lend:
    %rax: = %max
    ret
}
