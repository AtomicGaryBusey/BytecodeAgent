export function l $packbits_encode (l src, l n, l out, l cap) {
    @start:
        %i =l constint 0
        %outpos =l constint 0
    @check_i:
        %i_lt_n =w csltl %i %n
        jnz @process_packet
        ret %outpos
    @process_packet:
        %src_addr_i =l add src %i
        %cur =w loadub %src_addr_i
        %run =l constint 1
        %j =l add %i constint 1
    @run_loop:
        %run_lt_128 =w csltl %run constint 128
        jnz @run_next_check
        %run_len =l mov %run
        j @after_run_len
    @run_next_check:
        %j_lt_n =w csltl %j %n
        jnz @load_next_for_run
        %run_len =l mov %run
        j @after_run_len
    @load_next_for_run:
        %src_addr_j =l add src %j
        %next =w loadub %src_addr_j
        %eq =w ceql %cur %next
        jnz @run_inc
        %run_len =l mov %run
        j @after_run_len
    @run_inc:
        %run =l add %run constint 1
        %j =l add %j constint 1
        j @run_loop
    @after_run_len:
        %run_lt_3 =w csltl %run_len constint 3
        jnz @literal_packet
        %remaining =l sub cap %outpos
        %rem_lt_2 =w csltl %remaining constint 2
        jnz @insufficient
        %hdr =l sub constint 257 %run_len
        %out_hdr_addr =l add out %outpos
        storeb %hdr, %out_hdr_addr
        %next_out_addr =l add %out_hdr_addr constint 1
        storeb %cur, %next_out_addr
        %outpos =l add %outpos constint 2
        %i =l add %i %run_len
        j @check_i
    @literal_packet:
        %j_lit =l mov %i
    @literal_loop:
        %lit_len =l sub %j_lit %i
        %lit_len_lt_128 =w csltl %lit_len constint 128
        jnz @run_ge3_check_lit
        %j_lit =l mov %j_lit
        j @literal_emit
    @run_ge3_check_lit:
        %jplus2 =l add %j_lit constint 2
        %have_three =w csltl %jplus2 %n
        jnz @check_three_equal
        %src_addr_jlit =l add src %j_lit
        %byte0 =w loadub %src_addr_jlit
        %out_addr =l add out %outpos
        storeb %byte0, %out_addr
        %outpos =l add %outpos constint 1
        %j_lit =l add %j_lit constint 1
        j @literal_loop
    @check_three_equal:
        %src_addr_j1 =l add src %j_lit constint 1
        %byte1 =w loadub %src_addr_j1
        %src_addr_j2 =l add src %j_lit constint 2
        %byte2 =w loadub %src_addr_j2
        %e1 =w ceql %byte0 %byte1
        jnz @e1_equal
        %out_addr =l add out %outpos
        storeb %byte0, %out_addr
        %outpos =l add %outpos constint 1
        %j_lit =l add %j_lit constint 1
        j @literal_loop
    @e1_equal:
        %e2 =w ceql %byte1 %byte2
        jnz @break_before_j
        %out_addr =l add out %outpos
        storeb %byte0, %out_addr
        %outpos =l add %outpos constint 1
        %j_lit =l add %j_lit constint 1
        j @literal_loop
    @break_before_j:
        %j_lit =l mov %j_lit
        j @literal_emit
    @literal_emit:
        %L =l sub %j_lit %i
        %space_needed =l add %L constint 1
        %remaining =l sub cap %outpos
        %rem_lt_space =w csltl %remaining %space_needed
        jnz @insufficient
        %hdr_lit =l sub %L constint 1
        %out_hdr_addr =l add out %outpos
        storeb %hdr_lit, %out_hdr_addr
        %outpos =l add %outpos constint 1
        %k =l mov %i
    @copy_literal_body:
        %src_addr_k =l add src %k
        %bk =w loadub %src_addr_k
        %out_addr_cur =l add out %outpos
        storeb %bk, %out_addr_cur
        %outpos =l add %outpos constint 1
        %k =l add %k constint 1
        %k_lt_j =w csltl %k %j_lit
        jnz @copy_literal_body
        %i =l mov %j_lit
        j @check_i
    @insufficient:
        %neg1 =l constint -1
        ret %neg1
}
