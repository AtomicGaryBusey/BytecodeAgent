export function l $packbits_encode (l src, l n, l out, l cap)
    @start:
        %src =l src
        %n =l n
        %out =l out
        %cap =l cap
        %i =l constint 0
        %outpos =l constint 0
    @check_i:
        %i_lt_n =w csltl %i %n
        jnz @process_packet
        ret %outpos
    @process_packet:
        %src_addr_i =l add %src %i
        %cur =w loadub %src_addr_i
        %run =l constint 1
        %j =l add %i constint 1
    @run_loop:
        %run_lt_128 =w csltl %run constint 128
        jnz @run_next_check
        %run_len =l mov %run
        j @after_run_len
    @run_next_check:
        %j_lt_n =w csltl %j %n
        jnz @load_next_for_run
        %run_len =l mov %run
        j @after_run_len
    @load_next_for_run:
        %src_addr_j =l add %src %j
        %next =w loadub %src_addr_j
        %eq =w ceql %cur %next
        jnz @run_inc
        %run_len =l mov %run
        j @after_run_len
    @run_inc:
        %run =l add %run constint 1
        %j =l add %j constint 1
        j @run_loop
    @after_run_len:
        %run_lt_3 =w csltl %run_len constint 3
        jnz @literal_packet
        %remaining =l sub %cap %outpos
        %rem_lt_2 =w csltl %remaining constint 2
        jnz @insufficient
        %hdr =l sub constint 257 %run_len
        %out_hdr_addr =l add %out %outpos
        storeb %hdr, %out_hdr_addr
        %next_out_addr =l add %out_hdr_addr constint 1
        storeb %cur, %next_out_addr
        %outpos =l add %outpos constint 2
        %i =l add %i %run_len
        j @check_i
    @literal_packet:
        %j =l %i
    @scan_literal:
        %len =l sub %j %i
        %len_lt_128 =w csltl %len constint 128
        jnz @maybe_run_at_j
        %j =l %j
        %j_end =l %j
        j @emit_literal
    @maybe_run_at_j:
        %j_lt_n =w csltl %j %n
        jnz @run_check_at_j
        %j =l %j
        %j_end =l %j
        j @emit_literal
    @run_check_at_j:
        %src_j_addr =l add %src %j
        %byte_j =w loadub %src_j_addr
        %run_len =l constint 1
        %j2 =l add %j constint 1
    @run_loop_j:
        %run_len_lt_128 =w csltl %run_len constint 128
        jnz @run_next_check_j
        %run_len =l %run_len
        j @run_end_check
    @run_next_check_j:
        %j2_lt_n =w csltl %j2 %n
        jnz @load_next_for_run_j
        %run_len =l %run_len
        j @run_end_check
    @load_next_for_run_j:
        %src_j2_addr =l add %src %j2
        %next =w loadub %src_j2_addr
        %eq =w ceql %byte_j %next
        jnz @run_inc_j
        %run_len =l %run_len
        j @run_end_check
    @run_inc_j:
        %run_len =l add %run_len constint 1
        %j2 =l add %j2 constint 1
        j @run_loop_j
    @run_end_check:
        %run_lt_3 =w csltl %run_len constint 3
        jnz @continue_literal
        %j =l %j
        %j_end =l %j
        j @emit_literal
    @continue_literal:
        %j =l add %j constint 1
        %j_end =l %j
        j @scan_literal
    @emit_literal:
        %L =l sub %j %i
        %space_needed =l add %L constint 1
        %remaining =l sub %cap %outpos
        %rem_lt_space =w csltl %remaining %space_needed
        jnz @insufficient
        %hdr_lit =l sub %L constint 1
        %out_hdr_addr =l add %out %outpos
        storeb %hdr_lit, %out_hdr_addr
        %outpos =l add %outpos constint 1
        %k =l %i
    @copy_literal_body:
        %src_addr_k =l add %src %k
        %bk =w loadub %src_addr_k
        %out_addr_cur =l add %out %outpos
        storeb %bk, %out_addr_cur
        %outpos =l add %outpos constint 1
        %k =l add %k constint 1
        %k_lt_j =w csltl %k %j
        jnz @copy_literal_body
        %i =l %j
        j @check_i
    @insufficient:
        %neg1 =l constint -1
        ret %neg1
