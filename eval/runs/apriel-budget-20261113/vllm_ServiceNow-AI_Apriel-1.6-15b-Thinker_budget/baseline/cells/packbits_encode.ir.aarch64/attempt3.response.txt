export function l $packbits_encode (l src, l n, l out, l cap)
    @start:
        %src =l src
        %n =l n
        %out =l out
        %cap =l cap
        %i =l constint 0
        %outpos =l constint 0

    @loop:
        %i_lt_n =w csltl %i %n
        jnz @process_one
        ret %outpos

    @process_one:
        %src_i_addr =l add %src %i
        %cur =w loadub %src_i_addr
        %run =l constint 1
        %j =l add %i constint 1

    @run_loop:
        %run_lt_128 =w csltl %run constint 128
        jnz @run_next_check
        %run_len =l %run
        j @run_done

    @run_next_check:
        %j_lt_n =w csltl %j %n
        jnz @load_next_for_run
        %run_len =l %run
        j @run_done

    @load_next_for_run:
        %src_j_addr =l add %src %j
        %next =w loadub %src_j_addr
        %eq =w ceql %cur %next
        jnz @run_inc
        %run_len =l %run
        j @run_done

    @run_inc:
        %run =l add %run constint 1
        %j =l add %j constint 1
        j @run_loop

    @run_done:
        %run_len_lt_3 =w csltl %run_len constint 3
        jnz @literal_packet
        %remaining =l sub %cap %outpos
        %remaining_lt_2 =w csltl %remaining constint 2
        jnz @insufficient
        %hdr =l sub constint 257 %run_len
        %out_hdr_addr =l add %out %outpos
        storeb %hdr, %out_hdr_addr
        %out_byte_addr =l add %out_hdr_addr constint 1
        storeb %cur, %out_byte_addr
        %outpos =l add %outpos constint 2
        %i =l add %i %run_len
        j @loop

    @literal_packet:
        %j =l %i

    @literal_loop:
        %j_eq_n =w ceql %j %n
        jnz @j_eq_n_end
        %src_j_addr =l add %src %j
        %byte_j =w loadub %src_j_addr
        %run_j =l constint 1
        %j2 =l add %j constint 1

    @runj_loop:
        %run_j_lt_128 =w csltl %run_j constint 128
        jnz @runj_next_check
        %run_len_j =l %run_j
        j @runj_done

    @runj_next_check:
        %j2_lt_n =w csltl %j2 %n
        jnz @load_next_for_runj
        %run_len_j =l %run_j
        j @runj_done

    @load_next_for_runj:
        %src_j2_addr =l add %src %j2
        %next_j =w loadub %src_j2_addr
        %eq_j =w ceql %byte_j %next_j
        jnz @runj_inc
        %run_len_j =l %run_j
        j @runj_done

    @runj_inc:
        %run_j =l add %run_j constint 1
        %j2 =l add %j2 constint 1
        j @runj_loop

    @runj_done:
        %run_len_j_lt_3 =w csltl %run_len_j constint 3
        jnz @continue_literal
        %j_end =l %j
        j @emit_literal

    @continue_literal:
        %len =l sub %j %i
        %len_lt_128 =w csltl %len constint 128
        jnz @increment_j
        %j_end =l %j
        j @emit_literal

    @increment_j:
        %j =l add %j constint 1
        j @literal_loop

    @j_eq_n_end:
        %j_end =l %j
        j @emit_literal

    @emit_literal:
        %L =l sub %j_end %i
        %space_needed =l add %L constint 1
        %remaining =l sub %cap %outpos
        %remaining_lt_space =w csltl %remaining %space_needed
        jnz @insufficient
        %hdr_lit =l sub %L constint 1
        %out_hdr_addr =l add %out %outpos
        storeb %hdr_lit, %out_hdr_addr
        %k =l %i
    @copy_loop:
        %src_k_addr =l add %src %k
        %bk =w loadub %src_k_addr
        %out_addr_cur =l add %out %outpos
        storeb %bk, %out_addr_cur
        %outpos =l add %outpos constint 1
        %k =l add %k constint 1
        %k_lt_j_end =w csltl %k %j_end
        jnz @copy_loop
        %i =l %j_end
        j @loop

    @insufficient:
        %neg1 =l constint -1
        ret %neg1
