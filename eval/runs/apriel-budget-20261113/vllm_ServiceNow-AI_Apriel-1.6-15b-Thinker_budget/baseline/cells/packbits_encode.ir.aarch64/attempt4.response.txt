export function l $packbits_encode (l src, l n, l out, l cap)
    @start:
        %src =l src
        %n =l n
        %out =l out
        %cap =l cap
        %i =l constint 0
        %outpos =l constint 0

    @top:
        %i_lt_n =w csltl %i %n
        jnz @process
        ret %outpos

    @process:
        %src_i_addr =add %src %i
        %cur =w loadub %src_i_addr

        %run_len =l constint 1
        %j =add %i constint 1

    @repeat_run_loop:
        %run_len_lt_128 =w csltl %run_len constint 128
        jnz @repeat_run_next
        %run_len_cap =l %run_len
        j @repeat_checked

    @repeat_run_next:
        %j_lt_n =w csltl %j %n
        jnz @repeat_load_next
        %run_len_cap =l %run_len
        j @repeat_checked

    @repeat_load_next:
        %src_j_addr =add %src %j
        %next =w loadub %src_j_addr
        %eq =w ceql %cur %next
        jnz @repeat_inc
        %run_len_cap =l %run_len
        j @repeat_checked

    @repeat_inc:
        %run_len =add %run_len constint 1
        %j =add %j constint 1
        j @repeat_run_loop

    @repeat_checked:
        %run_len_lt_3 =w csltl %run_len_cap constint 3
        jnz @literal_packet

        %remaining =sub %cap %outpos
        %remaining_lt_2 =w csltl %remaining constint 2
        jnz @insufficient
        %hdr =sub constint 257 %run_len_cap
        %out_hdr_addr =add %out %outpos
        storeb %hdr, %out_hdr_addr
        %out_byte_addr =add %out_hdr_addr constint 1
        storeb %cur, %out_byte_addr
        %outpos =add %outpos constint 2
        %i =add %i %run_len_cap
        j @top

    @literal_packet:
        %j = %i
        %lit_len =l constint 0

    @find_j_end:
        %j_eq_n =w ceql %j %n
        jnz @found_end
        %lit_len_lt_128 =w csltl %lit_len constint 128
        jnz @check_run
        j @found_end

    @check_run:
        %src_j_addr =add %src %j
        %byte_j =w loadub %src_j_addr
        %j1 =add %j constint 1
        %src_j1_addr =add %src %j1
        %byte_j1 =w loadub %src_j1_addr
        %j2 =add %j1 constint 1
        %j2_lt_n =w csltl %j2 %n
        %eq1 =w ceql %byte_j %byte_j1
        jnz @run_ge3_candidate
        %j =add %j constint 1
        %lit_len =add %lit_len constint 1
        j @find_j_end

    @run_ge3_candidate:
        %j2_lt_n =w csltl %j2 %n
        jnz @load_byte_j2
        %j =add %j constint 1
        %lit_len =add %lit_len constint 1
        j @find_j_end

    @load_byte_j2:
        %src_j2_addr =add %src %j2
        %byte_j2 =w loadub %src_j2_addr
        %eq2 =w ceql %byte_j1 %byte_j2
        jnz @run_ge3_true
        %j =add %j constint 1
        %lit_len =add %lit_len constint 1
        j @find_j_end

    @run_ge3_true:
        j @found_end

    @found_end:
        %L = %lit_len
        %space_needed =add %L constint 1
        %remaining =sub %cap %outpos
        %remaining_lt_space =w csltl %remaining %space_needed
        jnz @insufficient
        %hdr_lit =sub %L constint 1
        %out_hdr_addr =add %out %outpos
        storeb %hdr_lit, %out_hdr_addr
        %j_end =add %i %L
        %k = %i
    @copy_loop:
        %k_eq_j_end =w ceql %k %j_end
        jnz @copy_done
        %src_k_addr =add %src %k
        %bk =w loadub %src_k_addr
        %out_addr_cur =add %out %outpos
        storeb %bk, %out_addr_cur
        %outpos =add %outpos constint 1
        %k =add %k constint 1
        j @copy_loop
    @copy_done:
        %i = %j_end
        j @top

    @insufficient:
        %neg1 =l constint -1
        ret %neg1
