export function l $bmax(l, l)
@label L_start
%max = loadl x0
%i = 1
%addr = x0
%addr = addl %addr, 8
@label L_loop
%lt = csltl %i x1
jnz %lt L_body
ret %max
@label L_body
%cur = loadl %addr
%gt = csltl %max %cur
jnz %gt L_update
jmp L_increment
@label L_update
%max = %cur
@label L_increment
%addr = addl %addr, 8
%i = addl %i, 1
jmp L_loop
