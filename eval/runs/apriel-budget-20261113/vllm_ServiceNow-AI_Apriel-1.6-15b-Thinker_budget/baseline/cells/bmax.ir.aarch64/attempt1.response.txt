export function l $bmax(l a, l n)
@label L_start
%max = loadl a
%i = 1
%addr = a
%addr = addl %addr, 8
@label L_loop
%lt = csltl %i %n
jnz %lt L_body
ret %max
@label L_body
%cur = loadl %addr
%gt = csltl %max %cur
jnz %gt L_update
jmp L_increment
@label L_update
%max = %cur
@label L_increment
%addr = addl %addr, 8
%i = addl %i, 1
jmp L_loop
