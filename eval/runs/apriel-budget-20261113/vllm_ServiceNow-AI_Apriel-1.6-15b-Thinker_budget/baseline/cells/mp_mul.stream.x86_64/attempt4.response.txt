mp_mul:
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rax, r13
    add rax, r15
    mov r9, rax
    xor r10, r10
.Lzero_loop:
    cmp r10, r9
    jge .Lzero_done
    mov qword ptr [r8 + r10*8], 0
    inc r10
    jmp .Lzero_loop
.Lzero_done:
    xor r10, r10
.Louter:
    cmp r10, r13
    jge .Ldone_mul
    xor r11, r11
.Linner:
    cmp r11, r15
    jge .Linner_done
    mov rax, [r12 + r10*8]
    mov rdx, [r14 + r11*8]
    mulx rdx, rax, rbx
    mov rcx, r10
    add rcx, r11
    lea rsi, [r8 + rcx*8]
    add [rsi], rax
    jnc .Lno_carry_low
    lea rdx, [rsi + 8]
    inc [rdx]
.Lno_carry_low:
    lea rdx, [rsi + 8]
    add [rdx], rbx
    inc r11
    jmp .Linner
.Linner_done:
    inc r10
    jmp .Louter
.Ldone_mul:
    xor rdx, rdx
    xor r10, r10
.Lcarry_loop:
    cmp r10, r9
    jge .Lcarry_done
    lea rsi, [r8 + r10*8]
    mov rax, [rsi]
    add rax, rdx
    mov [rsi], rax
    setc rdx
    inc r10
    jmp .Lcarry_loop
.Lcarry_done:
    test r9, r9
    jle .Lzero_rcnt
    mov rax, r9
    dec rax
.Lscan_top:
    cmp rax, -1
    jl .Lzero_result
    lea rsi, [r8 + rax*8]
    mov rdx, qword ptr [rsi]
    test rdx, rdx
    jne .Lfound
    dec rax
    jmp .Lscan_top
.Lzero_rcnt:
    xor rax, rax
    jmp .Lret
.Lzero_result:
    xor rax, rax
    jmp .Lret
.Lfound:
    inc rax
.Lret:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    ret
