mp_mul:
    push r12
    push r13
    push r14
    push r15
    mov r13, rdi
    mov r12, rdx
    mov r14, rcx
    test rsi, rsi
    jne .Lboth_nonzero
    xor r9, r9
    mov rdx, r14
.Lzero_loop:
    cmp r9, rdx
    jge .Lzero_done
    mov qword ptr [r8 + r9*8], 0
    inc r9
    jmp .Lzero_loop
.Lzero_done:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    ret
.Lboth_nonzero:
    mov rdx, rsi
    add rdx, r14
    xor r9, r9
.Lzero_full:
    cmp r9, rdx
    jge .Louter
    mov qword ptr [r8 + r9*8], 0
    inc r9
    jmp .Lzero_full
.Louter:
    xor r10, r10
.Louter_loop:
    cmp r10, rsi
    jge .Lscan
    xor r15, r15
    xor r11, r11
.Linner_loop:
    cmp r11, r14
    jge .Linner_done
    lea rdx, [r8 + r10*8 + r11*8]
    mov rcx, qword ptr [rdx]
    lea rsi, [r13 + r10*8]
    mov rsi, qword ptr [rsi]
    lea r9, [r12 + r11*8]
    mov rdx, qword ptr [r9]
    mulx rdx, rsi
    add rdx, rcx
    add rdx, r15
    adc rax, 0
    mov r15, rax
    lea rcx, [r8 + r10*8 + r11*8]
    mov qword ptr [rcx], rdx
    inc r11
    jmp .Linner_loop
.Linner_done:
    lea rcx, [r8 + r10*8 + r14*8]
    mov qword ptr [rcx], r15
    xor r15, r15
    inc r10
    jmp .Louter_loop
.Lscan:
    lea rdx, [rsi + r14]
    dec rdx
    mov rax, rdx
    test rax, rax
    jle .Lreturn0
.Lscan_loop:
    lea rcx, [r8 + rax*8]
    mov rcx, qword ptr [rcx]
    test rcx, rcx
    jnz .Lfound
    dec rax
    test rax, rax
    jge .Lscan_loop
    xor rax, rax
    jmp .Lreturn
.Lfound:
    inc rax
.Lreturn:
    pop r15
    pop r14
    pop r13
    pop r12
    ret
.Lreturn0:
    xor rax, rax
    jmp .Lreturn
