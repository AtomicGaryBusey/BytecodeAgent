mp_mul:
    push rbx
    push r12
    push r13
    push r14
    mov r9, rdx
    lea r11, [rsi + rcx]      ; total length
    test r11, r11
    jz zero_skip
    xor rax, rax
    lea rdx, [r11 - 1]
zero_loop:
    mov qword ptr [r8 + rdx*8], rax
    dec rdx
    jne zero_loop
zero_skip:
    mov r14, rcx               ; save nb
    xor r12, r12
outer_loop:
    cmp r12, rsi
    jge outer_end
    mov rax, qword ptr [rdi + r12*8]
    xor r13, r13
inner_loop:
    cmp r13, r14
    jge outer_next_i
    mov rdx, qword ptr [r9 + r13*8]
    mulx rdx
    lea r11, [r12 + r13]       ; idx
    lea r10, [r8 + r11*8]
    mov rcx, qword ptr [r10]
    add rcx, rax
    mov qword ptr [r10], rcx
    lea r10, [r10 + 8]
    mov rcx, qword ptr [r10]
    adc rdx, rcx
    mov qword ptr [r10], rdx
carry_loop:
    setc al
    test al, al
    jz carry_done
    lea r10, [r10 + 8]
    inc qword ptr [r10]
    jmp carry_loop
carry_done:
    inc r13
    cmp r13, r14
    jl inner_loop
outer_next_i:
    inc r12
    jmp outer_loop
outer_end:
    lea rdx, [rsi + r14]       ; total
    test rdx, rdx
    jz all_zero
    dec rdx
    mov r11, rdx
    jmp scan_loop
all_zero:
    xor rax, rax
    jmp restore
scan_loop:
    mov rax, qword ptr [r8 + r11*8]
    test rax, rax
    jne found_nonzero
    dec r11
    jge scan_loop
    xor rax, rax
    jmp restore
found_nonzero:
    lea rdx, [r11 + 1]
    mov rax, rdx
restore:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
