binary_search:
	cmp x1, #0
	beq .Ldone
	mov x3, #0
	sub x4, x1, #1
.Lsearch:
	cmp x3, x4
	b.gt .Ldone
	add x5, x3, x4
	lsr x5, x5, #1
	ldr x6, [x0, x5, lsl #3]
	cmp x6, x2
	b.eq .Lfound
	b.lt .Llt
	b .Lgt
.Lfound:
	mov x0, x5
	ret
.Llt:
	add x3, x5, #1
	b .Lsearch
.Lgt:
	sub x4, x5, #1
	b .Lsearch
.Ldone:
	mov x0, #0xFFFFFFFFFFFFFFFF
	ret
