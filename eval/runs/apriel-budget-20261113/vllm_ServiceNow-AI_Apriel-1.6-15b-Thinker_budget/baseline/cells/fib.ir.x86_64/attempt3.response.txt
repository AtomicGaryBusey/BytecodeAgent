export function l $fib(l)
@start:
    movl %0, %n
    ceql %n, 0, %zero_eq
    jnz %zero_eq, @ret0
    movl 0, %a_val
    movl 1, %b_val
@loop:
    csltl %n, 2, %lt2
    jnz %lt2, @ret_end
    addl %a_val, %b_val, %tmp
    movl %b_val, %a_val
    movl %tmp, %b_val
    subl %n, 1, %n
    jmp @loop
@ret_end:
    movl %b_val, %rax
    ret
@ret0:
    movl %0, %rax
    ret
