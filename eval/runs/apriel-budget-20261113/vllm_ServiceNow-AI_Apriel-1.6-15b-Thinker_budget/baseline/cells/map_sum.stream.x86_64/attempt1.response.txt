map_sum:
    sub rsp, 40
    mov [rsp], r12
    mov [rsp+8], r13
    mov [rsp+16], r14
    mov [rsp+24], r15
    mov [rsp+32], rbx
    mov r12, rdx
    mov r14, rdi
    xor r13, r13
    mov rbx, rsi
    xor r15, r15
    test rbx, rbx
    jz .Lend
.Lloop:
    mov rdi, qword ptr [r14 + r15*8]
    call r12
    add r13, rax
    inc r15
    dec rbx
    jne .Lloop
.Lend:
    mov rax, r13
    mov r12, [rsp]
    mov r13, [rsp+8]
    mov r14, [rsp+16]
    mov r15, [rsp+24]
    mov rbx, [rsp+32]
    add rsp, 40
    ret
