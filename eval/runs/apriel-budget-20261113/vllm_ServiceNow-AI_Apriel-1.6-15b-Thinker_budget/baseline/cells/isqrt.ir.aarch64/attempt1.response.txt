    @entry:
        movl %t0, 0
        movl %t1, $0
        movl %t2, 4294967295
        j @loop
    @loop:
        csltl %t3, %t0, %t2
        jnz @mid_calc, %t3
        movl $0, %t0
        ret
    @mid_calc:
        subl %t4, %t2, %t0
        addl %t4, %t4, 1
        shr %t5, %t4, 1
        addl %t6, %t0, %t5
        divl %t7, %t1, %t6
        csltl %t8, %t7, %t6
        jnz @set_high, %t8
        movl %t0, %t6
        j @loop
    @set_high:
        subl %t2, %t6, 1
        j @loop
}
