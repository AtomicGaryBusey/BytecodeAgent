        csltl %t1, %t0, 2
        jnz @return_small, %t1
        movl %t2, 1
        shr %t3, %t0, 1
        addl %t4, %t3, 1
    @loop:
        addl %t5, %t2, %t4
        addl %t5, %t5, 1
        shr %t6, %t5, 1
        divl %t7, %t0, %t6
        csltl %t8, %t7, %t6
        jnz @set_high, %t8
        movl %t6, %t2
        csltl %t9, %t2, %t4
        jnz @loop, %t9
        movl %t2, $2

    @set_high:

        movl %t10, %t4
        j @loop

        ret
