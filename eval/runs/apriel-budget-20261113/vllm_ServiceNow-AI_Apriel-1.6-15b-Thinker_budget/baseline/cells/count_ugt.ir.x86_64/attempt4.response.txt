export function l $count_ugt(l, l, l) @entry:
    %cnt = constl 0
    %i = constl 0
    %c1 = constl 1
    %c3 = constl 3
@loop:
    %cond = csltl %i, %1
    jnz %cond, @body
    ret %cnt
@body:
    %off = shl %i, %c3
    %addr = addl %0, %off
    %val = loadl %addr
    %less = csltl %val, %2
    %eq = ceql %val, %2
    %or = orl %less, %eq
    %gt = subl %c1, %or
    %cnt = addl %cnt, %gt
    %i = addl %i, %c1
    jmp @loop
