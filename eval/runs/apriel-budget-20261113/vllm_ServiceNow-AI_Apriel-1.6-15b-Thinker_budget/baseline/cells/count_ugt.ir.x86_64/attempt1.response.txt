export function l $count_ugt(l, l, l) @entry:
    %cnt = constl 0
    %i = constl 0
    @check:
    %cond = csltl %i, %1
    jnz @body
    ret %cnt
    @body:
    %i8 = shiftl %i, 3
    %addr = addl %0, %i8
    %val = loadl %addr
    %less = csltl %val, %2
    %eq = ceql %val, %2
    %or = orl %less, %eq
    %gt = subl constl 1, %or
    %cnt = addl %cnt, %gt
    %i = addl %i, constl 1
    jmp @check
