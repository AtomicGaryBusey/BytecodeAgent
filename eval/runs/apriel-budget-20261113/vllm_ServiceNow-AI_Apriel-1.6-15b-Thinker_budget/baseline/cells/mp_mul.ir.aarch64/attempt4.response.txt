export function l $mp_mul(l a, l na, l b, l nb, l r)
@entry:
    %a =l a
    %na =l na
    %b =l b
    %nb =l nb
    %r =l r
    %len =l add %na, %nb
    %i =l 0
@zero_loop:
    storel 0, add %r, shl %i, 3
    %i =l add %i, 1
    %lt_i =w csltl %i, %len
    jnz @zero_loop, %lt_i
    %i =l 0
@outer_i_check:
    %lt_i =w csltl %i, %na
    jnz @inner_start, %lt_i
    %res =l %len
    jmp @calc_start
@inner_start:
    %carry =l 0
    %j =l 0
@inner_loop:
    %lt_j =w csltl %j, %nb
    jnz @process, %lt_j
    jmp @inner_done
@process:
    %addr_a =l add %a, shl %i, 3
    loadl %ai, %addr_a
    %addr_b =l add %b, shl %j, 3
    loadl %bj, %addr_b
    %a_lo =l and %ai, 0xffffffff
    %a_hi =l shr %ai, 32
    %b_lo =l and %bj, 0xffffffff
    %b_hi =l shr %bj, 32
    %c0 =l mul %a_lo, %b_lo
    %c1 =l mul %a_lo, %b_hi
    %c2 =l mul %a_hi, %b_lo
    %c3 =l mul %a_hi, %b_hi
    %mid =l add %c1, %c2
    %mid_shl =l shl %mid, 32
    %low =l add %c0, %mid_shl
    %hi =l add %c3, shr %mid, 32
    %k =l add %i, %j
    %addr_rk =l add %r, shl %k, 3
    loadl %rk, %addr_rk
    %sum_low0 =l add %rk, %low
    %carry0 =w cultl %sum_low0, %rk
    %sum_low1 =l add %sum_low0, %carry
    %carry1 =w cultl %sum_low1, %sum_low0
    %new_carry =l add %hi, %carry1
    storel %sum_low1, %addr_rk
    %carry =l %new_carry
    %j =l add %j, 1
    jmp @inner_loop
@inner_done:
    %idx =l add %i, %nb
    %addr =l add %r, shl %idx, 3
    loadl %rx, %addr
    %sum =l add %rx, %carry
    %carry =w cultl %sum, %rx
    storel %sum, %addr
    %k =l %idx
@carry_prop:
    %addr =l add %r, shl %k, 3
    loadl %ry, %addr
    %sum =l add %ry, %carry
    %carry =w cultl %sum, %ry
    storel %sum, %addr
    %k =l add %k, 1
    %gt0 =w csgeql %carry, 1
    jnz @carry_prop, %gt0
    %i =l add %i, 1
    jmp @outer_i_check
@calc_start:
    %eq_res0 =w ceql %res, 0
    jnz @return0, %eq_res0
    %idx =l sub %res, 1
    %addr =l add %r, shl %idx, 3
    loadl %rx, %addr
    %eq_val0 =w ceql %rx, 0
    jnz @calc_start, %eq_val0
    ret %res
@return0:
    ret 0
