export function l $mp_mul(l a, l na, l b, l nb, l r)
@entry:
  %len =l add na, nb
  %idx =l 0
@zero_loop:
  %off_idx =l shl %idx, 3
  %addr_idx =l add r, %off_idx
  storel 0, %addr_idx
  %idx =l add %idx, 1
  %lt_idx =w csltl %idx, %len
  jnz @zero_loop, %lt_idx
  %i =l 0
@outer_loop:
  %lt_i =w csltl %i, na
  jnz @inner_start, %lt_i
  jmp @find_len
@inner_start:
  %j =l 0
@inner_loop_check:
  %lt_j =w csltl %j, nb
  jnz @process_product, %lt_j
  %i =l add %i, 1
  %lt_i =w csltl %i, na
  jnz @outer_loop, %lt_i
  jmp @find_len
@process_product:
  %off_i =l shl %i, 3
  %addr_i =l add a, %off_i
  loadl %ai, %addr_i
  %off_j =l shl %j, 3
  %addr_j =l add b, %off_j
  loadl %bj, %addr_j
  %a_lo =l and %ai, 0ffffffff
  %a_hi =l shr %ai, 32
  %b_lo =l and %bj, 0ffffffff
  %b_hi =l shr %bj, 32
  %c0 =l mul %a_lo, %b_lo
  %c1 =l mul %a_lo, %b_hi
  %c2 =l mul %a_hi, %b_lo
  %c3 =l mul %a_hi, %b_hi
  %sum_c12 =l add %c1, %c2
  %shift_c12 =l shl %sum_c12, 32
  %low =l add %c0, %shift_c12
  %carry0 =w cultl %low, %c0
  %hi =l add %c3, %carry0
  %k =l add %i, %j
  %off_k =l shl %k, 3
  %addr_k =l add r, %off_k
  loadl %r_k, %addr_k
  %sum_low =l add %r_k, %low
  %carry_low =w cultl %sum_low, %r_k
  storel %sum_low, %addr_k
  %k1 =l add %k, 1
  %off_k1 =l shl %k1, 3
  %addr_k1 =l add r, %off_k1
  loadl %r_k1, %addr_k1
  %sum_mid =l add %r_k1, %hi
  %carry_mid1 =w cultl %sum_mid, %r_k1
  %sum_mid2 =l add %sum_mid, %carry_low
  %carry_mid2 =w cultl %sum_mid2, %carry_low
  storel %sum_mid2, %addr_k1
  %total_carry =l add %carry_mid1, %carry_mid2
  %idx_car =l add %k1, 1
  %carry =l %total_carry
@prop_carry:
  %off_idxc =l shl %idx_car, 3
  %addr_idxc =l add r, %off_idxc
  loadl %r_idxc, %addr_idxc
  %sum_carry =l add %r_idxc, %carry
  %new_carry =w cultl %sum_carry, %r_idxc
  storel %sum_carry, %addr_idxc
  %idx_car =l add %idx_car, 1
  jnz @prop_carry, %new_carry
  %j =l add %j, 1
  %lt_j =w csltl %j, nb
  jnz @inner_loop_check, %lt_j
  %i =l add %i, 1
  %lt_i =w csltl %i, na
  jnz @outer_loop, %lt_i
  jmp @find_len
@find_len:
  %len =l add na, nb
  %idx =l add %len, -1
@find_len_loop:
  %off_idx =l shl %idx, 3
  %addr_idx =l add r, %off_idx
  loadl %tmp_len, %addr_idx
  %is_zero =w csltl %tmp_len, 1
  jnz @dec_len, %is_zero
  %ret =l add %idx, 1
  ret %ret
@dec_len:
  %idx =l add %idx, -1
  %is_minus_one =w ceql %idx, -1
  jnz @return_zero, %is_minus_one
  jmp @find_len_loop
@return_zero:
  ret 0
