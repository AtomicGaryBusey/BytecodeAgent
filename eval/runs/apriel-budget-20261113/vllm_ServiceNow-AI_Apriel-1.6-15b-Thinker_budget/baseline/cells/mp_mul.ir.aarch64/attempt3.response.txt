export function l $mp_mul(l a, l na, l nb, l b, l r)
@entry:
    %len =l add na nb
    %i0 =l 0
@zero_loop:
    storel 0, add r, shl %i0 3
    %i0 =l add %i0, 1
    %lt_i0 =w csltl %i0, %len
    jnz @zero_loop, %lt_i0
    %i =l 0
@outer_i:
    %lt_i =w csltl %i, na
    jnz @inner_j, %lt_i
    %i =l add %i, 1
    jmp @outer_i
@inner_j:
    %j =l 0
@inner_loop:
    %lt_j =w csltl %j, nb
    jnz @process, %lt_j
    %j =l add %j, 1
    jmp @inner_loop
@process:
    %off_a =l shl %i, 3
    %addr_a =l add a, %off_a
    loadl %ai, %addr_a
    %off_b =l shl %j, 3
    %addr_b =l add b, %off_b
    loadl %bj, %addr_b
    %a_lo =l and %ai, 0xffffffff
    %a_hi =l shr %ai, 32
    %b_lo =l and %bj, 0xffffffff
    %b_hi =l shr %bj, 32
    %c0 =l mul %a_lo, %b_lo
    %c1 =l mul %a_lo, %b_hi
    %c2 =l mul %a_hi, %b_lo
    %c3 =l mul %a_hi, %b_hi
    %mid =l add %c1, %c2
    %mid_shl =l shl %mid, 32
    %low =l add %c0, %mid_shl
    %carry0 =w cultl %low, %c0
    %hi =l add %c3, shr %mid, 32
    %hi =l add %hi, %carry0
    %k =l add %i, %j
    %addr_rk =l add r, shl %k, 3
    loadl %rk, %addr_rk
    %sum_low =l add %rk, %low
    %carry_low =w cultl %sum_low, %rk
    storel %sum_low, %addr_rk
    %k1 =l add %k, 1
    %addr_rk1 =l add r, shl %k1, 3
    loadl %rk1, %addr_rk1
    %sum_mid =l add %rk1, %hi
    %carry_mid =w cultl %sum_mid, %rk1
    storel %sum_mid, %addr_rk1
    %carry = %carry_mid
@propagate:
    %idx =l %k1
    %addr_idx =l add r, shl %idx, 3
    loadl %rx, %addr_idx
    %sum =l add %rx, %carry
    %new_carry =w cultl %sum, %rx
    storel %sum, %addr_idx
    %carry = %new_carry
    %idx =l add %idx, 1
    %lt_carry =w csgeql %carry, 2 // assume zero carry exits loop
    jnz @propagate, %lt_carry
    %j =l add %j, 1
    jmp @inner_loop
@done_i:
    ret %len
