export function l $mp_mul(l a, l na, l b, l nb, l r)
@entry:
    %len =l add %na, %nb
    %i0 =l 0
@zero_loop:
    %off0 =l shl %i0, 3
    %addr0 =l add %r, %off0
    storel 0, %addr0
    %i0 =l add %i0, 1
    %lt_i0 =w csltl %i0, %len
    jnz @zero_loop, %lt_i0
    %i =l 0
@outer_check:
    %lt_i =w csltl %i, %na
    jnz @inner_start, %lt_i
    jmp @find_len
@inner_start:
    %j =l 0
@inner_loop:
    %lt_j =w csltl %j, %nb
    jnz @process_product, %lt_j
    %i =l add %i, 1
    jmp @outer_check
@process_product:
    %off_i =l shl %i, 3
    %addr_i =l add %a, %off_i
    loadl %ai, %addr_i
    %off_j =l shl %j, 3
    %addr_j =l add %b, %off_j
    loadl %bj, %addr_j
    %a_lo =l and %ai, 0xffffffff
    %a_hi =l shr %ai, 32
    %b_lo =l and %bj, 0xffffffff
    %b_hi =l shr %bj, 32
    %c0 =l mul %a_lo, %b_lo
    %c1 =l mul %a_lo, %b_hi
    %c2 =l mul %a_hi, %b_lo
    %c3 =l mul %a_hi, %b_hi
    %sum_c12 =l add %c1, %c2
    %shift_c12 =l shl %sum_c12, 32
    %low =l add %c0, %shift_c12
    %carry0 =w cultl %low, %c0
    %hi =l add %c3, shr %sum_c12, 32
    %hi =l add %hi, %carry0
    %k =l add %i, %j
    %off_k =l shl %k, 3
    %addr_k =l add %r, %off_k
    loadl %r_k, %addr_k
    %sum_low =l add %r_k, %low
    %carry_low =w cultl %sum_low, %r_k
    storel %sum_low, %addr_k
    %k1 =l add %k, 1
    %off_k1 =l shl %k1, 3
    %addr_k1 =l add %r, %off_k1
    loadl %r_k1, %addr_k1
    %sum_mid =l add %r_k1, %hi
    %carry_mid0 =w cultl %sum_mid, %r_k1
    %sum_mid2 =l add %sum_mid, %carry_low
    %carry_mid1 =w cultl %sum_mid2, %sum_mid
    storel %sum_mid2, %addr_k1
    %total_carry =l add %carry_mid0, %carry_mid1
    %carry_out =w cultl %total_carry, %carry_mid0
    %idx =l add %k1, 1
    %carry = %carry_out
@prop_carry:
    %off_idx =l shl %idx, 3
    %addr_idx =l add %r, %off_idx
    loadl %r_idx, %addr_idx
    %sum =l add %r_idx, %carry
    %new_carry =w cultl %sum, %r_idx
    storel %sum, %addr_idx
    %carry = %new_carry
    %idx =l add %idx, 1
    jnz @prop_carry, %carry
    %j =l add %j, 1
    jmp @inner_loop
@find_len:
    %idx_len =l add %len, -1
@find_len_loop:
    %off_len =l shl %idx_len, 3
    %addr_len =l add %r, %off_len
    loadl %tmp_len, %addr_len
    %is_zero =w csltl %tmp_len, 1
    jnz @dec_len, %is_zero
    %ret =l add %idx_len, 1
    ret %ret
@dec_len:
    %idx_len =l add %idx_len, -1
    %is_minus_one =w ceql %idx_len, -1
    jnz @return_zero, %is_minus_one
    jmp @find_len_loop
@return_zero:
    ret 0
