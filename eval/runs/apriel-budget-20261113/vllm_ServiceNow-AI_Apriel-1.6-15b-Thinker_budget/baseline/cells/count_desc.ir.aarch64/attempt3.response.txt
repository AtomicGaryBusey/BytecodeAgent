export function l $count_desc(l %0, l %1) {
    @label .Lcheck
    sub(l) %cnt, %1, %1
    sub(l) %max_i, %1, 1
    sub(l) %i, %max_i, %max_i
    csltl %cond, %i, %max_i
    jnz .Lbody, %cond
    ret(l) %cnt
    @label .Lbody
    mul(l) %offset, %i, 8
    add(l) %addr, %0, %offset
    loadl %cur, %addr
    add(l) %addr_next, %addr, 8
    loadl %next, %addr_next
    csltl %cmp, %next, %cur
    jnz .Linc_cnt, %cmp
    @label .Lskip_cnt
    j .Linc_i
    @label .Linc_cnt
    add(l) %cnt, %cnt, 1
    @label .Linc_i
    add(l) %i, %i, 1
    j .Lcheck
}
