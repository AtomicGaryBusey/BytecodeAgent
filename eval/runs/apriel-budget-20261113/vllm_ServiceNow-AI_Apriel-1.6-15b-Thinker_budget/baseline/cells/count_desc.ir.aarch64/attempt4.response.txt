export function l $count_desc(l %0, l %1) {
    @label .Lstart
    sub(l) %cnt, %1, %1
    sub(l) %limit, %1, 1
    sub(l) %i, %cnt, %cnt
    csltl %cond, %i, %limit
    jnz .Lbody, %cond
    ret(l) %cnt
    @label .Lbody
    mul(l) %offset, %i, 8
    add(l) %addr, %0, %offset
    loadl %cur, %addr
    add(l) %addr_next, %addr, 8
    loadl %next, %addr_next
    csltl %cmp, %next, %cur
    jnz .Lhas_desc, %cmp
    @label .Lno_desc
    j .Linc_i
    @label .Lhas_desc
    add(l) %cnt, %cnt, 1
    @label .Linc_i
    add(l) %i, %i, 1
    j .Lstart
}
