export function l $map_into(l dst, l src, l n, l f) {
  @entry:
  sub rsp, 8
  xorl %i, %i
  xorl %c, %c
  @loop:
  csltl %i, %n
  jnz @body
  jmp @exit
  @body:
  imull %i, 8, %addr_src
  addl %src, %addr_src
  loadl %addr_src, %src_val
  %dst_val =l call %f(l %src_val)
  imull %i, 8, %addr_dst
  addl %dst, %addr_dst
  storel %dst_val, %addr_dst
  csltl %dst_val, %src_val
  jnz @inc_c
  jmp @inc_i
  @inc_c:
  addl 1, %c
  jmp @inc_i
  @inc_i:
  addl 1, %i
  jmp @loop
  @exit:
  add rsp, 8
  movl %c, %rax
  ret
}
