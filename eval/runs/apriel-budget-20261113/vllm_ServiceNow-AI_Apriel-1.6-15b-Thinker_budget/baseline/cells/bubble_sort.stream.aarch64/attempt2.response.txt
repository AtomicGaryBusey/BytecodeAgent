bubble_sort:
cmp x1, #1
b.le .Ldone
sub x2, x1, #1
mov x3, #0
.Louter:
cmp x3, x2
b.ge .Louter_end
sub x4, x2, x3
mov x5, #0
.Linner:
cmp x5, x4
b.ge .Linner_end
ldr x6, [x0, x5, lsl #3]
add x7, x5, #1
ldr x8, [x0, x7, lsl #3]
cmp x6, x8
b.le .Lnext_j
str x8, [x0, x5, lsl #3]
str x6, [x0, x7, lsl #3]
.Lnext_j:
add x5, x5, #1
b .Linner
.Linner_end:
add x3, x3, #1
b .Louter
.Louter_end:
mov x0, #0
ret
.Ldone:
mov x0, #0
ret
