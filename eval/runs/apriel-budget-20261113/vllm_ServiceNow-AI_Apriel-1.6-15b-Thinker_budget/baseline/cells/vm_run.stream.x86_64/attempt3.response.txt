.Ljmp:

	mov r9, rcx
	jmp .Lloop
.Ldone:

	sub r11, r8
	test r11, r11
	jle .Lfinalize
	mov r10, r8
	lea r9, [rdx + r10*8]

.Lzero_loop:

	jz .Lfinalize
	mov qword ptr [r9], rax
	add r9, 8
	dec r11
	jmp .Lzero_loop
