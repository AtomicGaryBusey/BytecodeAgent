mat_pow:
    // compute row stride (bytes per row) = n * 8
    lsl x6, x1, #3
    // compute n*n (element count)
    mul x7, x1, x1
    // compute buffer pointer = tmp + (n*n)*8
    lsl x8, x7, #3
    add x5, x4, x8

    // copy a -> tmp (base)
    mov x9, x0          // src = a
    mov x10, x4         // dst = base
    mov x12, #0         // i = 0
.Lcopy_a_i:
    cmp x12, x1
    b.ge .Lcopy_a_i_done
    mul x18, x12, x6    // offset_i (bytes)
    add x15, x9, x18    // src_row
    add x16, x10, x18   // dst_row
    mov x13, #0         // j = 0
.Lcopy_a_j:
    cmp x13, x1
    b.ge .Lcopy_a_j_done
    lsl x18, x13, #3
    add x18, x15, x18   // src_addr
    ldr x0, [x18]       // load element (x0)
    add x18, x16, x18   // dst_addr
    str x0, [x18]       // store to tmp
    add x13, x13, #1
    b .Lcopy_a_j
.Lcopy_a_j_done:
    add x12, x12, #1
    b .Lcopy_a_i
.Lcopy_a_i_done:

    // init identity matrix in out
    mov x12, #0
.Linit_i:
    cmp x12, x1
    b.ge .Linit_done
    mul x18, x12, x6    // offset_i
    lsl x19, x12, #3    // i*8
    add x18, x18, x19   // total offset
    add x18, x3, x18    // address
    mov x0, #1
    str x0, [x18]
    add x12, x12, #1
    b .Linit_i
.Linit_done:

    // exponentiation by squaring
    // while k != 0
    // Lloop:
    //   if (k & 1) result = result * base
    //   base = base * base
    //   k >>= 1
    //   if (k == 0) break
    // end
    // After loop, zero tmp (both halves) and return 0

    // Louter:
    mov x11, x2         // copy k to x11 for testing (not strictly needed)
.Louter_loop:
    cbnz x2, .Lhas_bit   // if k != 0 continue loop (we test after body)
    // if k == 0, zero tmp and return
    // zero tmp (2 * n*n elements)
    mov x9, #0
    mov x10, x4         // start = tmp (base)
    mov x12, #0
    mov x13, x7
    add x13, x13, x7    // count = 2 * n*n
.Lzero_tmp:
    cmp x12, x13
    b.ge .Lzero_done
    mul x18, x12, #8
    add x18, x10, x18
    str x9, [x18]
    add x12, x12, #1
    b .Lzero_tmp
.Lzero_done:
    mov x0, #0
    ret

.Lhas_bit:
    // multiply result * base -> buffer, then copy buffer -> result
    // set up multiplication: src1 = out (x3), src2 = base (x4), dest = buffer (x5)
    mov x9, x3          // src1
    mov x10, x4         // src2
    mov x11, x5         // dest

    // matrix multiplication routine
    mov x12, #0         // i
.Lmat_i:
    cmp x12, x1
    b.ge .Lmat_i_done
    mul x18, x12, x6    // offset_i
    add x15, x9, x18    // src1_row
    add x16, x11, x18   // dest_row

    mov x13, #0         // j
.Lmat_j:
    cmp x13, x1
    b.ge .Lmat_j_done
    mov x17, #0         // sum = 0

    mov x14, #0         // t
.Lmat_t:
    cmp x14, x1
    b.ge .Lmat_t_done
    // src1_addr = src1_row + t*8
    add x18, x15, x14, LSL #3
    ldr x0, [x18]               // load A(i,t)
    // src2_addr = src2 + t*row_stride + j*8
    mul x18, x14, x6             // t*row_stride
    add x18, x18, x13, LSL #3    // + j*8
    add x18, x10, x18            // src2 base + offset
    ldr x7, [x18]               // load B(t,j)
    mul x8, x0, x7              // product
    add x17, x17, x8            // sum
    add x14, x14, #1
    b .Lmat_t
.Lmat_t_done:
    // dest_addr = dest_row + j*8
    lsl x18, x13, #3
    add x18, x16, x18
    str x17, [x18]
.Lmat_j_done:
    add x13, x13, #1
    b .Lmat_j
.Lmat_i_done:

    // copy buffer -> result (out)
    mov x9, x5          // src = buffer
    mov x10, x3         // dst = result
    mov x12, #0
.Lcopy_res_i:
    cmp x12, x1
    b.ge .Lcopy_res_i_done
    mul x18, x12, x6
    add x15, x9, x18
    add x16, x10, x18
    mov x13, #0
.Lcopy_res_j:
    cmp x13, x1
    b.ge .Lcopy_res_j_done
    lsl x18, x13, #3
    add x18, x15, x18
    ldr x0, [x18]
    add x18, x16, x18
    str x0, [x18]
    add x13, x13, #1
    b .Lcopy_res_j
.Lcopy_res_j_done:
    add x12, x12, #1
    b .Lcopy_res_i
.Lcopy_res_i_done:

    // now base = base * base -> buffer, then copy buffer -> base
    // set up multiplication: src1 = base (x4), src2 = base (x4), dest = buffer (x5)
    mov x9, x4          // src1
    mov x10, x4         // src2
    mov x11, x5         // dest

    // matrix multiplication routine (same as above)
    mov x12, #0
.Lmat_i2:
    cmp x12, x1
    b.ge .Lmat_i2_done
    mul x18, x12, x6
    add x15, x9, x18
    add x16, x11, x18

    mov x13, #0
.Lmat_j2:
    cmp x13, x1
    b.ge .Lmat_j2_done
    mov x17, #0

    mov x14, #0
.Lmat_t2:
    cmp x14, x1
    b.ge .Lmat_t2_done
    add x18, x15, x14, LSL #3
    ldr x0, [x18]
    mul x18, x14, x6
    add x18, x18, x13, LSL #3
    add x18, x10, x18
    ldr x7, [x18]
    mul x8, x0, x7
    add x17, x17, x8
    add x14, x14, #1
    b .Lmat_t2
.Lmat_t2_done:
    lsl x18, x13, #3
    add x18, x16, x18
    str x17, [x18]
.Lmat_j2_done:
    add x13, x13, #1
    b .Lmat_j2
.Lmat_i2_done:

    // copy buffer -> base
    mov x9, x5          // src = buffer
    mov x10, x4         // dst = base
    mov x12, #0
.Lcopy_base_i:
    cmp x12, x1
    b.ge .Lcopy_base_i_done
    mul x18, x12, x6
    add x15, x9, x18
    add x16, x10, x18
    mov x13, #0
.Lcopy_base_j:
    cmp x13, x1
    b.ge .Lcopy_base_j_done
    lsl x18, x13, #3
    add x18, x15, x18
    ldr x0, [x18]
    add x18, x16, x18
    str x0, [x18]
    add x13, x13, #1
    b .Lcopy_base_j
.Lcopy_base_j_done:
    add x12, x12, #1
    b .Lcopy_base_i
.Lcopy_base_i_done:

    // shift k right by 1
    lsr x2, x2, #1
    b .Louter_loop
