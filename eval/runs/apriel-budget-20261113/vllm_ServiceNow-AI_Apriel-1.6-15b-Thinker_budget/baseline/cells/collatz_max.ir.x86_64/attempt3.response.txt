export function l $collatz_max(l) {
    @label Lstart
    %max = 0
    %k = 1
    %remaining = %0
    @label Louter
    %steps = 0
    %x = %k
    @label Lcalc_loop
    %gt = csltl 1 %x
    jnz Lcalc_body %gt
    jmp Lcalc_done
    @label Lcalc_body
    %mask = andl %x 1
    %odd = csltl 0 %mask
    jnz Lodd %odd
    %x = shr %x 1
    jmp Lcalc_step
    @label Lodd
    %tmp = add %x %x
    %tmp2 = add %tmp %x
    %x = add %tmp2 1
    @label Lcalc_step
    %steps = add %steps 1
    jmp Lcalc_loop
    @label Lcalc_done
    %gt2 = csltl %max %steps
    jnz Lmax_update %gt2
    jmp Lnext_k
    @label Lmax_update
    %max = %steps
    jmp Lnext_k
    @label Lnext_k
    %k = add %k 1
    %remaining = sub %remaining 1
    %cond = csltl 0 %remaining
    jnz Louter %cond
    ret %max
}
