export function l $collatz_max(l) {
    @label Linit
    %n = %0
    %max = 0
    %remaining = %n
    %k = 1
    @label Louter
    %steps = 0
    %x = %k
    @label Linner
    %gt = csltl 1 %x
    jnz Lcalc_body %gt
    @label Linner_done
    %gt2 = csltl %max %steps
    jnz Lmax_update %gt2
    jmp Lnext_k
    @label Lmax_update
    %max = %steps
    jmp Lnext_k
    @label Lnext_k
    %k = add %k 1
    %remaining = sub %remaining 1
    %cond = csltl 0 %remaining
    jnz Louter %cond
    ret %max
    @label Lcalc_body
    %mask = andl %x 1
    %odd = csltl 0 %mask
    jnz Lodd %odd
    %x = shr %x 1
    jmp Lafter_odd
    @label Lodd
    %tmp1 = add %x %x
    %tmp2 = add %tmp1 %x
    %x = add %tmp2 1
    @label Lafter_odd
    %steps = add %steps 1
    jmp Linner
}
