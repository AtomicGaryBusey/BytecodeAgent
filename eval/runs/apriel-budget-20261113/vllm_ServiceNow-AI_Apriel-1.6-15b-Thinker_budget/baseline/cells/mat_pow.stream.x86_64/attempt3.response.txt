mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rcx
    mov r14, r8
    mov rax, rsi
    imul rax, rax
    imul rax, rax, 8
    mov r15, r8
    add r15, rax
    mov rax, rsi
    imul rax, rax
    mov rbx, rax
    imul rbx, rbx, 2
    mov rdi, r14
    mov rcx, rbx
    xor rax, rax
    rep stosq
    mov rax, rsi
    imul rax, rax
    mov rdi, r13
    mov rcx, rax
    xor rax, rax
    rep stosq
    mov r9, rsi
    mov r10, rdx
    xor r8, r8
.Lsetdiag:
    cmp r8, rsi
    jge .Lsetdiag_done
    mov rax, r8
    imul rax, rsi
    imul rax, rax, 8
    mov rdx, r13
    add rdx, rax
    mov rax, r8
    imul rax, 8
    add rdx, rax
    mov qword ptr [rdx], 1
    inc r8
    jmp .Lsetdiag
.Lsetdiag_done:
.Lexp_loop:
    test r10, r10
    jle .Lexp_done
    test r10, 1
    jz .Lskip_mul
    mov rdi, r13
    mov rsi, r12
    mov rdx, r15
    xor r8, r8
.Li_loop:
    cmp r8, r9
    jge .Lmat_mul_done_i
    mov rax, r8
    imul rax, r9
    imul rax, rax, 8
    lea r10, [rdi + rax]
    mov rax, r8
    imul rax, r9
    imul rax, rax, 8
    lea r11, [rdx + rax]
    xor r12, r12
.Lj_loop:
    cmp r12, r9
    jge .Lj_done_i
    mov rax, r12
    imul rax, 8
    mov r15, rax
    mov r14, r11
    add r14, r15
    xor rbx, rbx
    xor r13, r13
.Lt_loop:
    cmp r13, r9
    jge .Lt_done
    mov rcx, r10
    mov rax, r13
    imul rax, 8
    lea rcx, [rcx + rax]
    mov rax, [rcx]
    mov rdx, r13
    mov rax, r9
    imul rax, rdx
    imul rax, rax, 8
    lea rdx, [rsi + rax + r15]
    mov rcx, [rdx]
    imul rax, rcx
    add rbx, rax
    inc r13
    jmp .Lt_loop
.Lt_done:
    mov [r14], rbx
    inc r12
    jmp .Lj_loop
.Lj_done_i:
    inc r8
    jmp .Li_loop
.Lmat_mul_done_i:
    mov rdi, r13
    mov rsi, r15
    mov rax, r9
    imul rax, rax
    mov rcx, rax
    rep movsq
    jmp .Lafter_mul
.Lskip_mul:
.Lafter_mul:
    mov rdi, r12
    mov rsi, r12
    mov rdx, r14
    xor r8, r8
.Li2_loop:
    cmp r8, r9
    jge .Lmat_sq_done
    mov rax, r8
    imul rax, r9
    imul rax, rax, 8
    lea r10, [rdi + rax]
    mov rax, r8
    imul rax, r9
    imul rax, rax, 8
    lea r11, [rdx + rax]
    xor r12, r12
.Lj2_loop:
    cmp r12, r9
    jge .Lj2_done_i
    mov rax, r12
    imul rax, 8
    mov r15, rax
    mov r14, r11
    add r14, r15
    xor rbx, rbx
    xor r13, r13
.Lt2_loop:
    cmp r13, r9
    jge .Lt2_done
    mov rcx, r10
    mov rax, r13
    imul rax, 8
    lea rcx, [rcx + rax]
    mov rax, [rcx]
    mov rdx, r13
    mov rax, r9
    imul rax, rdx
    imul rax, rax, 8
    lea rdx, [rsi + rax + r15]
    mov rcx, [rdx]
    imul rax, rcx
    add rbx, rax
    inc r13
    jmp .Lt2_loop
.Lt2_done:
    mov [r14], rbx
    inc r12
    jmp .Lj2_loop
.Lj2_done_i:
    inc r8
    jmp .Li2_loop
.Lmat_sq_done:
    mov r12, r14
    shr r10, 1
    jmp .Lexp_loop
.Lexp_done:
    mov rax, r9
    imul rax, rax
    mov rbx, rax
    imul rbx, rbx, 2
    mov rdi, r14
    mov rcx, rbx
    xor rax, rax
    rep stosq
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
