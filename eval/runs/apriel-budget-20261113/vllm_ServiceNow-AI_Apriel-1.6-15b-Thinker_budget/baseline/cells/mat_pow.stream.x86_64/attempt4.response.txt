mat_pow:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r13, rdi                # a
    mov r9, rsi                 # n
    mov rax, r9
    imul rax, r9                # n2
    mov r11, rax                # n2 (count)
    mov r10, r11                # temp
    shl r10, 3                  # offset2 = n2*8
    mov r12, rdx                # k
    mov rdx, r8                 # scratch_start = tmp
    mov r15, rcx                # save out pointer
    mov rax, r11
    mov rbx, rax
    shl rbx, 1                  # 2*n2
    mov rcx, rbx                # count for stosq
    xor rax, rax
    mov rdi, rdx
    rep stosq
    mov rcx, r15                # restore out pointer
    mov r14, rdx
    add r14, r10                # product_ptr = scratch_start + offset2
    mov r15, rcx                # save out pointer
    xor rax, rax
    mov rcx, r11                # count = n2
    mov rdi, r15                # dest = out
    rep stosq
    mov rcx, r15                # restore out pointer
    xor r15, r15                # i = 0
.Lsetdiag:
    cmp r15, r9
    jge .Lsetdiag_done
    mov rax, r15
    imul rax, r9
    shl rax, 3
    mov rbx, r15
    shl rbx, 3
    add rax, rbx
    mov qword ptr [rcx + rax], 1
    inc r15
    jmp .Lsetdiag
.Lsetdiag_done:
    mov rdi, r13                # src = a
    mov rdx, r8                # dest = base_ptr
    mov rbx, r11                # count = n2
.Lcopy_a_loop:
    test rbx, rbx
    jz .Lcopy_a_done
    mov rax, qword ptr [rdi]
    mov qword ptr [rdx], rax
    add rdi, 8
    add rdx, 8
    dec rbx
    jmp .Lcopy_a_loop
.Lcopy_a_done:
.Lexp_loop:
    test r12, r12
    jle .Lexp_done
    test r12, 1
    jz .Lskip_mul
    mov r15, r12                # save k
    mov r12, r8                # save base_ptr
    mov rdi, r14                # dest = product_ptr
    mov rsi, rcx                # left = result (out)
    mov rdx, r8                # right = base
    xor r8, r8                  # i = 0
.Lmat_mul_i:
    cmp r8, r9
    jge .Lmat_mul_i_end
    mov rax, r8
    imul rax, r9
    mov rbx, rax                # i_n
    xor r10, r10                # j = 0
.Lmat_mul_j:
    cmp r10, r9
    jge .Lmat_mul_j_end
    xor rax, rax                # sum = 0
    xor r11, r11                # t = 0
.Lmat_mul_t:
    cmp r11, r9
    jge .Lmat_mul_t_end
    mov rcx, qword ptr [rsi + rbx*8 + r11*8]
    mov r12, r11
    imul r12, r9
    mov r13, qword ptr [rdx + r12*8 + r10*8]
    imul rcx, r13
    add rax, rcx
    inc r11
    jmp .Lmat_mul_t
.Lmat_mul_t_end:
    mov qword ptr [rdi + rbx*8 + r10*8], rax
    inc r10
    jmp .Lmat_mul_j
.Lmat_mul_j_end:
    inc r8
    jmp .Lmat_mul_i
.Lmat_mul_i_end:
    mov r8, r12                # restore base_ptr
    mov r12, r15                # restore k
    mov r15, rcx                # save out pointer
    mov rdi, r14                # src = product_ptr
    mov rsi, r15                # dest = out
    mov rbx, r11                # count = n2
    xor rax, rax
.Lcopy_prod_out:
    test rbx, rbx
    jz .Lcopy_prod_done
    mov rax, qword ptr [rdi]
    mov qword ptr [rsi], rax
    add rdi, 8
    add rsi, 8
    dec rbx
    jmp .Lcopy_prod_out
.Lcopy_prod_done:
    mov r15, rcx                # save out pointer
    mov rcx, r11                # count = n2
    mov rdi, r14
    xor rax, rax
    rep stosq
    mov rcx, r15                # restore out pointer
    jmp .Lsquare_base
.Lskip_mul:
.Lsquare_base:
    mov r15, r12                # save k
    mov r12, r8                # save base_ptr
    mov rdi, r14                # dest = product_ptr
    mov rsi, r8                # left = base
    mov rdx, r8                # right = base
    xor r8, r8                  # i = 0
.Lmat_mul_i2:
    cmp r8, r9
    jge .Lmat_mul_i2_end
    mov rax, r8
    imul rax, r9
    mov rbx, rax                # i_n
    xor r10, r10                # j = 0
.Lmat_mul_j2:
    cmp r10, r9
    jge .Lmat_mul_j2_end
    xor rax, rax                # sum = 0
    xor r11, r11                # t = 0
.Lmat_mul_t2:
    cmp r11, r9
    jge .Lmat_mul_t2_end
    mov rcx, qword ptr [rsi + rbx*8 + r11*8]
    mov r12, r11
    imul r12, r9
    mov r13, qword ptr [rdx + r12*8 + r10*8]
    imul rcx, r13
    add rax, rcx
    inc r11
    jmp .Lmat_mul_t2
.Lmat_mul_t2_end:
    mov qword ptr [rdi + rbx*8 + r10*8], rax
    inc r10
    jmp .Lmat_mul_j2
.Lmat_mul_j2_end:
    inc r8
    jmp .Lmat_mul_i2
.Lmat_mul_i2_end:
    mov r8, r12                # restore base_ptr
    mov r12, r15                # restore k
    mov rax, r8                # old base_ptr
    mov r8, r14                # new base_ptr = product_ptr
    mov r14, rax               # product_ptr = old base_ptr
    shr r12, 1
    jmp .Lexp_loop
.Lexp_done:
    mov rax, r11
    mov rbx, rax
    shl rbx, 1
    mov rcx, rbx
    xor rax, rax
    mov rdi, rdx
    rep stosq
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
