    mov r13, rax
    mov rax, rsi

.Zeromatrix:
    mov qword ptr [rbx], rax
    add rbx, 8
    dec rdx
    jne .Zeromatrix
    xor rdx, rdx
.Diag:
    cmp rdx, rsi
    jge .Afterdiag
    mov rax, rdx
    imul rax, rsi
    imul rax, rax, 8
    lea rcx, [r12 + rax]
    mov rbx, rdx
    shl rbx, 3
    add rcx, rbx
    mov rax, 1
    mov qword ptr [rcx], rax
    inc rdx
    jmp .Diag
.Afterdiag:
    mov rdx, r14
.Lexp:
    test rdx, rdx
    jle .Ldone
    push rdx
    xor rdx, rdx
.Lmul_i:
    cmp rdx, rsi
    jge .Lmul_done_i
    mov rax, rdx
    imul rax, rsi
    imul rax, rax, 8
    mov r11, rax
    xor r9, r9
.Lmul_j:
    cmp r9, rsi
    jge .Lmul_done_j
    xor rbx, rbx
    xor r10, r10
.Lmul_t:
    cmp r10, rsi
    jge .Lmul_done_t
    mov r13, r10
    shl r13, 3
    lea r13, [r12 + r11 + r13]
    mov rax, qword ptr [r13]
    mov r14, r10
    imul r14, r14, rsi
    imul r14, r14, 8
    mov r15, r9
    shl r15, 3
    lea r14, [rdi + r14 + r15]

    imul rcx, rax
    add rbx, rcx
    inc r10
    jmp .Lmul_t
.Lmul_done_t:
    mov r15, r9
    shl r15, 3
    lea r15, [r8 + r11 + r15]
    mov qword ptr [r15], rbx
    inc r9
    jmp .Lmul_j
.Lmul_done_j:
    inc rdx
    jmp .Lmul_i
.Lmul_done_i:
    pop rdx
    mov rax, rsi
    imul rax, rax
    mov r9, rax
    xor rax, rax
    mov rcx, r12
    mov rdx, r8
.Lcopy_loop:
    mov rax, qword ptr [rdx]
    mov qword ptr [rcx], rax
    add rdx, 8
    add rcx, 8
    dec r9
    jne .Lcopy_loop
    dec rdx
    jmp .Lexp
.Ldone:
    xor rax, rax
    mov rdx, r8
    mov rax, rsi
    imul rax, rax
    mov r9, rax
.Lzero_tmp1:

    dec r9
    jne .Lzero_tmp1

    imul rax, rax, 8

    imul rax, rax
    mov r9, rax
.Lzero_tmp2:
