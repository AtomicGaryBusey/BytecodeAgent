export function l $stats_f(%a, %n, %f, %out)
@L0:
    %ptr =l %a
    %cnt =l %n
    loadl %arg, %ptr
    %y =l call %f(l %arg)
    %min =l %y
    %max =l %y
    %sum =l %y
    %lb =l and %y, %i1
    %odd =l %lb
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop:
    %zero =w ceql %cnt, %i0
    jnz %zero, @Lstore
    loadl %arg, %ptr
    %y =l call %f(l %arg)
    %lb =l and %y, %i1
    %odd =l add %odd, %lb
    %sum =l add %sum, %y
    %cmp_min =w csltl %y, %min
    jnz %cmp_min, @Lupd_min
    %cmp_max =w csltl %max, %y
    jnz %cmp_max, @Lupd_max
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
    @Lupd_min:
    %min =l %y
    %cmp_max =w csltl %max, %y
    jnz %cmp_max, @Lupd_max
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
    @Lupd_max:
    %max =l %y
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
    @Lstore:
    %out0 =l %out
    %out4 =l add %out, %i4
    %out8 =l add %out, %i8
    storel %min, %out0
    storel %max, %out4
    storel %sum, %out8
    ret %odd
