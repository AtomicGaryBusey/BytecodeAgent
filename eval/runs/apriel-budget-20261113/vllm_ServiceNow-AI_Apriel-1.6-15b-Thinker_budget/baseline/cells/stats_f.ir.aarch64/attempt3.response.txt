export function l $stats_f(%a, %n, %f, %out)
@L0:
    %ptr =l %a
    %cnt =l %n
    loadl %arg, %ptr
    %y =l call %f(l %arg)
    %min =l %y
    %max =l %y
    %sum =l %y
    %odd =l and %y, %i1
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
@Lloop:
    %cnt_w =w cgtl %cnt, %i0
    jnz %cnt_w, @Lbody
    @Lstore:
    %out0 =l %out
    %out4 =l add %out, %i4
    %out8 =l add %out, %i8
    storel %min, %out0
    storel %max, %out4
    storel %sum, %out8
    ret %odd
    @Lbody:
    loadl %arg, %ptr
    %y =l call %f(l %arg)
    %cmp_min =w csltl %y, %min
    jnz %cmp_min, @Lupd_min
    @Lno_min:
    %cmp_max =w csltl %max, %y
    jnz %cmp_max, @Lupd_max
    @Lsum0:
    %sum =l add %sum, %y
    %odd =l add %odd, and %y, %i1
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
    @Lupd_min:
    %min =l %y
    %cmp_max =w csltl %max, %y
    jnz %cmp_max, @Lupd_max
    @Lsum1:
    %sum =l add %sum, %y
    %odd =l add %odd, and %y, %i1
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
    @Lupd_max:
    %max =l %y
    %sum =l add %sum, %y
    %odd =l add %odd, and %y, %i1
    %cnt =l sub %cnt, %i1
    %ptr =l add %ptr, %i8
    @Lloop
