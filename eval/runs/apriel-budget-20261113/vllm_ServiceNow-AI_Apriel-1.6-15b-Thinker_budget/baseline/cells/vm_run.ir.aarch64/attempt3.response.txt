export function l $vm_run(l prog, l len, l stack, l cap) {
    @start:
    %pc = 0;
    %sp = 0;
    %prog = prog;
    %len = len;
    %stack = stack;
    %cap = cap;
    @loop:
    ceql %t0, %pc, %len;
    jnz @end;
    loadl %instr, [%prog + %pc*8];
    ceql %t0, %instr, 0;
    jnz @end;
    ceql %t0, %instr, 1;
    jnz @push;
    ceql %t0, %instr, 2;
    jnz @add;
    ceql %t0, %instr, 3;
    jnz @sub;
    ceql %t0, %instr, 4;
    jnz @mul;
    ceql %t0, %instr, 5;
    jnz @dup;
    ceql %t0, %instr, 6;
    jnz @swap;
    ceql %t0, %instr, 7;
    jnz @drop;
    ceql %t0, %instr, 8;
    jnz @over;
    ceql %t0, %instr, 9;
    jnz @lt;
    ceql %t0, %instr, 10;
    jnz @jz;
    ceql %t0, %instr, 11;
    jnz @jmp;
    %pc = addl %pc, 1;
    jmp @loop;
    @push:
    %tmp = addl %pc, 1;
    %oper = loadl [%prog + %tmp*8];
    storel %oper, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 2;
    jmp @loop;
    @add:
    %sp = subl %sp, 1;
    %b = loadl [%stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [%stack + %sp*8];
    %res = addl %a, %b;
    storel %res, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @sub:
    %sp = subl %sp, 1;
    %b = loadl [%stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [%stack + %sp*8];
    %res = subl %a, %b;
    storel %res, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @mul:
    %sp = subl %sp, 1;
    %b = loadl [%stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [%stack + %sp*8];
    %res = mull %a, %b;
    storel %res, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @dup:
    %tmp = subl %sp, 1;
    %a = loadl [%stack + %tmp*8];
    storel %a, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @swap:
    %tmp1 = subl %sp, 1;
    %a = loadl [%stack + %tmp1*8];
    %tmp2 = addl %tmp1, 1;
    %b = loadl [%stack + %tmp2*8];
    storel %b, [%stack + %tmp1*8];
    storel %a, [%stack + %tmp2*8];
    %pc = addl %pc, 1;
    jmp @loop;
    @drop:
    %sp = subl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @over:
    %tmp1 = subl %sp, 2;
    %a = loadl [%stack + %tmp1*8];
    storel %a, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @lt:
    %sp = subl %sp, 1;
    %b = loadl [%stack + %sp*8];
    %tmp = subl %sp, 1;
    %a = loadl [%stack + %tmp*8];
    %sp = %tmp;
    %res = csltl %a, %b;
    storel %res, [%stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @loop;
    @jz:
    %tmp = addl %pc, 1;
    %oper = loadl [%prog + %tmp*8];
    %sp = subl %sp, 1;
    %v = loadl [%stack + %sp*8];
    ceql %t0, %v, 0;
    jnz @jz_zero;
    %pc = addl %pc, 2;
    jmp @loop;
    @jz_zero:
    %pc = %oper;
    jmp @loop;
    @jmp:
    %tmp = addl %pc, 1;
    %oper = loadl [%prog + %tmp*8];
    %pc = %oper;
    jmp @loop;
    @end:
    %i = %sp;
    @zero_loop:
    ceql %t0, %i, %cap;
    jnz @zero_done;
    @zero_body:
    storel 0, [%stack + %i*8];
    %i = addl %i, 1;
    jmp @zero_loop;
    @zero_done:
    ceql %t0, %sp, 0;
    jnz @has_top;
    %result = 0;
    jmp @ret_end;
    @has_top:
    %tmp = subl %sp, 1;
    %result = loadl [%stack + %tmp*8];
    @ret_end:
    ret %result;
}}
