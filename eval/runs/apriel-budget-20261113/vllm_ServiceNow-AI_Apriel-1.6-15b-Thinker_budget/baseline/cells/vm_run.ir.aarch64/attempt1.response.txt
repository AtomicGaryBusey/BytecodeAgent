export function l $vm_run(l code, l len, l stack, l cap) {
    @start:
    %pc = 0;
    %sp = 0;
    %tmp = subl %len, %pc;
    jnz @execute;
    jmp @end;
    @execute:
    loadl %instr, [code + %pc*8];
    ceql %t0, %instr, 0;
    jnz @halt;
    ceql %t0, %instr, 1;
    jnz @push;
    ceql %t0, %instr, 2;
    jnz @add;
    ceql %t0, %instr, 3;
    jnz @sub;
    ceql %t0, %instr, 4;
    jnz @mul;
    ceql %t0, %instr, 5;
    jnz @dup;
    ceql %t0, %instr, 6;
    jnz @swap;
    ceql %t0, %instr, 7;
    jnz @drop;
    ceql %t0, %instr, 8;
    jnz @over;
    ceql %t0, %instr, 9;
    jnz @lt;
    ceql %t0, %instr, 10;
    jnz @jz;
    ceql %t0, %instr, 11;
    jnz @jmp;
    %pc = addl %pc, 1;
    jmp @start;
    @halt:
    jmp @end;
    @push:
    %tmp1 = addl %pc, 1;
    %val = loadl [code + %tmp1*8];
    storel %val, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 2;
    jmp @start;
    @add:
    %sp = subl %sp, 1;
    %b = loadl [stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [stack + %sp*8];
    %res = addl %a, %b;
    storel %res, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @sub:
    %sp = subl %sp, 1;
    %b = loadl [stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [stack + %sp*8];
    %res = subl %a, %b;
    storel %res, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @mul:
    %sp = subl %sp, 1;
    %b = loadl [stack + %sp*8];
    %sp = subl %sp, 1;
    %a = loadl [stack + %sp*8];
    %res = mull %a, %b;
    storel %res, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @dup:
    %tmp1 = subl %sp, 1;
    %a = loadl [stack + %tmp1*8];
    storel %a, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @swap:
    %tmp1 = subl %sp, 1;
    %a = loadl [stack + %tmp1*8];
    %tmp2 = addl %tmp1, 1;
    %b = loadl [stack + %tmp2*8];
    storel %b, [stack + %tmp1*8];
    storel %a, [stack + %tmp2*8];
    %pc = addl %pc, 1;
    jmp @start;
    @drop:
    %sp = subl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @over:
    %tmp1 = subl %sp, 2;
    %a = loadl [stack + %tmp1*8];
    storel %a, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @lt:
    %tmp1 = subl %sp, 1;
    %b = loadl [stack + %tmp1*8];
    %tmp2 = subl %tmp1, 1;
    %a = loadl [stack + %tmp2*8];
    %sp = %tmp2;
    %res = csltl %a, %b;
    storel %res, [stack + %sp*8];
    %sp = addl %sp, 1;
    %pc = addl %pc, 1;
    jmp @start;
    @jz:
    %tmp1 = addl %pc, 1;
    %oper = loadl [code + %tmp1*8];
    %sp = subl %sp, 1;
    %v = loadl [stack + %sp*8];
    ceql %t0, %v, 0;
    jnz @jz_zero;
    %pc = addl %pc, 2;
    jmp @start;
    @jz_zero:
    %pc = %oper;
    jmp @start;
    @jmp:
    %tmp1 = addl %pc, 1;
    %oper = loadl [code + %tmp1*8];
    %pc = %oper;
    jmp @start;
    @end:
    ceql %zero_flag, %sp, 0;
    jnz @result_zero;
    %tmp2 = subl %sp, 1;
    %result = loadl [stack + %tmp2*8];
    %i = %sp;
    jmp @zero_loop;
    @result_zero:
    %result = 0;
    %i = %sp;
    jmp @zero_loop;
    @zero_loop:
    ceql %t0, %i, %cap;
    jnz @zero_done;
    storel 0, [stack + %i*8];
    %i = addl %i, 1;
    jmp @zero_loop;
    @zero_done:
    ret %result;
}}
