dot_i32_simd:
    stp x19, x20, [sp, #-16]!
    mov x29, sp
    mov x3, x2                // n
    asr x4, x3, #2            // n/4
    mov x5, #0                // sum low
    mov x6, #0                // sum high (unused)
.Lsimd_loop:
    cbz x4, .Lrem
    // load 4 int32 from each vector
    ld1.ivp {v6.4s}, [x0], #16
    ld1.ivp {v7.4s}, [x1], #16
    // multiply‑accumulate (unsigned 32×32 → 64‑bit)
    vmlaq d8, d8, v6, v7
    // add partial result to sum (low part only, overflow wraps)
    add x5, x5, x8
    subs x4, x4, #1
    b .Lsimd_loop
.Lrem:
    // scalar tail loop for remaining <4 elements
    mov x4, x3                // remaining = original n
    and x4, x4, #3            // n & 3
    mov x7, #0                // tail sum
.Ltail_loop:
    cbnz x4, .Ltail_body
    add x5, x5, x7
    mov x0, x5                // return low 64 bits
    ldp x19, x20, [sp], #16
    ret
.Ltail_body:
    // load a[i] and b[i] (32‑bit, sign‑extend)
    ldr w10, [x0], #4
    ldr w11, [x1], #4
    uxtw x10, x10
    uxtw x11, x11
    // signed 64‑bit multiply (wrap on overflow)
    mul x12, x10, x11
    add x7, x7, x12
    subs x4, x4, #1
    b .Ltail_loop
