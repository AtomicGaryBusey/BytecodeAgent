        movl %a0, %t0
        movl $4294967295, %t1
        movl $0, %t2
        movl $0, %t3
        jmp @loop
    @loop:
        cgtl %t4, %t2, %t1
        jnz @end
        addl %t2, %t1, %t5
        shrr %t5, 1, %t6
        ceql %t7, %t6, $0
        jnz @mid_zero
        idivl %t8, %t0, %t6
        cgtl %t9, %t6, %t8
        jnz @else

        addl %t6, $1, %t2
        jmp @loop
    @else:
        subl $1, %t6, %t1
        jmp @loop
    @mid_zero:
        movl $1, %t2
        jmp @loop
    @end:
        movl %t3, %rax
}
