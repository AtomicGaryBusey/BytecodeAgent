export function l $isqrt(l) {
    @0:
        movl %a0, %t0
        movl $4294967295, %t1
        movl $0, %t2
        jmp @loop
    @loop:
        csltl %t3, %t2, %t1
        jz @end
        addl %t2, %t1, %t4
        addl $1, %t4, %t4
        shrr %t4, 1, %t5
        idivl %t5, %t0, %t6
        cgtl %t7, %t5, %t6
        jnz @set_hi
        movl %t5, %t2
        jmp @loop
    @set_hi:
        subl $1, %t5, %t1
        jmp @loop
    @end:
        movl %t2, %rax
}
