export function l $mulhi_u64(l %a, l %b)
{
    @entry
    %a_lo = and %a, i 0xffffffff
    %a_hi = shr %a, 32
    %b_lo = and %b, i 0xffffffff
    %b_hi = shr %b, 32
    %p0 = mul %a_lo, %b_lo
    %p1 = mul %a_lo, %b_hi
    %p2 = mul %a_hi, %b_lo
    %p3 = mul %a_hi, %b_hi
    %c0 = shr %p0, 32
    %c1_lo = and %p1, i 0xffffffff
    %c2_lo = and %p2, i 0xffffffff
    %tmp1 = add %c0, %c1_lo
    %tmp2 = add %tmp1, %c2_lo
    %carry32 = shr %tmp2, 32
    %p1_hi = shr %p1, 32
    %p2_hi = shr %p2, 32
    %p3_lo = and %p3, i 0xffffffff
    %p3_hi = shr %p3, 32
    %tmp3 = add %p1_hi, %p2_hi
    %tmp4 = add %tmp3, %p3_lo
    %tmp5 = add %tmp4, %carry32
    %hi_low = and %tmp5, i 0xffffffff
    %hi_carry = shr %tmp5, 32
    %hi_high = add %p3_hi, %hi_carry
    %tmp6 = mul %hi_high, i 0x100000000
    %hi = add %tmp6, %hi_low
    ret %hi
}
