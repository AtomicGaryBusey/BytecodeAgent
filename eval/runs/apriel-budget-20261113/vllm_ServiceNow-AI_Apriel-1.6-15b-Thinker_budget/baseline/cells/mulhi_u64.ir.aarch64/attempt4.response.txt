export function l $mulhi_u64(l %a, l %b)
{
    @entry
    %a_lo = and %a, l 0xffffffff
    %a_hi = shr %a, 32
    %b_lo = and %b, l 0xffffffff
    %b_hi = shr %b, 32
    %p0 = mul %a_lo, %b_lo
    %p1 = mul %a_lo, %b_hi
    %p2 = mul %a_hi, %b_lo
    %p3 = mul %a_hi, %b_hi
    %c0 = shr %p0, 32
    %p1_lo = and %p1, l 0xffffffff
    %p2_lo = and %p2, l 0xffffffff
    %tmp1 = add %c0, %p1_lo
    %tmp2 = add %tmp1, %p2_lo
    %carry32 = shr %tmp2, 32
    %p1_hi = shr %p1, 32
    %p2_hi = shr %p2, 32
    %tmp3 = add %p1_hi, %p2_hi
    %tmp4 = add %tmp3, %carry32
    %hi = add %p3, %tmp4
    ret %hi
}
