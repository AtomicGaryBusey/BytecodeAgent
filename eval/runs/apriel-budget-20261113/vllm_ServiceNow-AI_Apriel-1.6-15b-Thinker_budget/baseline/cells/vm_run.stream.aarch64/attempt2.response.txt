    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    stp x29, x30, [sp, #-16]!
    mov x19, #0
    mov x20, #0
.Lrun:
    cmp x19, x1
    bge .Lfinal
    ldr x23, [x0, x19, lsl #3]
    cmp x23, #0
    beq .Lfinal
    cmp x23, #1
    beq .Lpush
    cmp x23, #2
    beq .Ladd
    cmp x23, #3
    beq .Lsub
    cmp x23, #4
    beq .Lmul
    cmp x23, #5
    beq .Ldup
    cmp x23, #6
    beq .Lswap
    cmp x23, #7
    beq .Ldrop
    cmp x23, #8
    beq .Lover
    cmp x23, #9
    beq .Llt
    cmp x23, #10
    beq .Ljz
    cmp x23, #11
    beq .Ljmp
    add x19, x19, #1
    b .Lrun
.Lpush:
    add x25, x19, #1
    ldr x22, [x0, x25, lsl #3]
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2
    str x22, [x21]
    add x20, x20, #1
    add x19, x19, #2
    b .Lrun
.Ladd:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]    // b (top)
    sub x26, x26, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]    // a (second)
    mov x20, x26
    add x24, x23, x22
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2
    str x24, [x21]
    add x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Lsub:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]    // b (top)
    sub x26, x26, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]    // a (second)
    mov x20, x26
    sub x24, x23, x22
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2
    str x24, [x21]
    add x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Lmul:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]    // b (top)
    sub x26, x26, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]    // a (second)
    mov x20, x26
    mul x24, x23, x22
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2
    str x24, [x21]
    add x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Ldup:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]    // top
    mov x23, x20
    lsl x23, x23, #3
    add x23, x23, x2
    str x22, [x23]
    add x20, x20, #1
    add x19, x19, #2
    b .Lrun
.Lswap:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]    // b (top)
    sub x26, x26, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]    // a (second)
    mov x21, x26
    lsl x21, x21, #3
    add x21, x21, x2
    str x22, [x21]
    add x21, x26, #1
    lsl x21, x21, #3
    add x21, x21, x2
    str x23, [x21]
    add x19, x19, #1
    b .Lrun
.Ldrop:
    sub x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Lover:
    sub x26, x20, #1
    sub x26, x26, #1    // x26 = depth-2
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]      // a (below top)
    mov x23, x20
    lsl x23, x23, #3
    add x23, x23, x2
    str x22, [x23]
    add x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Llt:
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x22, [x21]      // b (top)
    sub x26, x26, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]      // a (second)
    mov x20, x26
    mov x24, #1
    mov x25, #0
    cmp x23, x22
    csel x24, x24, x25, MI
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2
    str x24, [x21]
    add x20, x20, #1
    add x19, x19, #1
    b .Lrun
.Ljz:
    add x25, x19, #1
    ldr x22, [x0, x25, lsl #3]   // target
    sub x26, x20, #1
    lsl x21, x26, #3
    add x21, x21, x2
    ldr x23, [x21]   // v (pop)
    mov x20, x26
    cmp x23, #0
    beq .Ljz_zero
    add x19, x19, #2
    b .Lrun
.Ljz_zero:
    mov x19, x22
    b .Lrun
.Ljmp:
    add x25, x19, #1
    ldr x22, [x0, x25, lsl #3]   // target
    mov x19, x22
    b .Lrun
.Lfinal:
    cmp x20, x3
    beq .Lreturn
    sub x25, x3, x20         // remaining slots
    mov x21, x20
    lsl x21, x21, #3
    add x21, x21, x2         // pointer to first remaining slot
    str xzr, [x21, #0]
    add x21, x21, #8
    sub x25, x25, #1
    cmp x25, #0
    beq .Lreturn
    b .Lzero_loop
.Lzero_loop:
    str xzr, [x21, #0]
    add x21, x21, #8
    sub x25, x25, #1
    cmp x25, #0
    beq .Lreturn
    b .Lzero_loop
.Lreturn:
    cmp x20, #0
    beq .Lzero_ret
    sub x22, x20, #1
    lsl x22, x22, #3
    add x22, x22, x2
    ldr x0, [x22]
    b .lend
.Lzero_ret:
    mov x0, #0
lend:
    ldp x29, x30, [sp], #16
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
