export function l $lcs_len(x, m, y, n, row)
@init:
    %tmp_i = 0
@zero_loop:
    %gt = csgtl %tmp_i, %n
    jnz @zero_done
    %addr = addl %row, %tmp_i
    storel %addr, 0
    %tmp_i = addl %tmp_i, 1
    jmp @zero_loop
@zero_done:
    %outer_i = 0
    %row_prev = 0
@outer_loop_start:
    %lt = csltl %outer_i, %m
    jnz @outer_body
    @outer_end:
@outer_body:
    %x_addr = addl %x, %outer_i
    %byte_x =w loadub %x_addr
    %diag = 0
    %row_prev = 0
    %row_ptr = addl %row, 8
    %y_cur = y
    %inner_j = 1
@inner_loop_start:
    %gt = csgtl %inner_j, %n
    jnz @inner_end
    %byte_y =w loadub %y_cur
    %up = loadl %row_ptr
    %eq = ceql %byte_x, %byte_y
    jnz @equal_case
    %new_val = %up
    %gt2 = csgtl %up, %row_prev
    jnz @skip_max_assign
    %new_val = %row_prev
    @skip_max_assign:
@common_body:
    storel %row_ptr, %new_val
    %diag = %up
    %row_prev = %new_val
    %row_ptr = addl %row_ptr, 8
    %y_cur = addl %y_cur, 1
    %inner_j = addl %inner_j, 1
    jmp @inner_loop_start
@equal_case:
    %new_val = addl %diag, 1
    jmp @common_body
@inner_end:
    %outer_i = addl %outer_i, 1
    jmp @outer_loop_start
@outer_end:
    ret %row_prev
