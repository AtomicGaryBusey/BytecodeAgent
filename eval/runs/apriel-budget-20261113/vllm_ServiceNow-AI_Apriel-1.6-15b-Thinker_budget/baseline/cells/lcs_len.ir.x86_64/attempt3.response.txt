export function l $lcs_len(l x, l m, l y, l n, l row)
@zero_start:
    %i = 0
@zero_loop:
    %c = csltl %i, %n
    jz @zero_done
    %addr = addl %row, %i
    storel %addr, 0
    %i = addl %i, 1
    jmp @zero_loop
@zero_done:
    %i = 0
    %diag = 0
    %prev = 0
outer_loop:
    %c = csltl %i, %m
    jz @outer_done
    %x_addr = addl %x, %i
    %bx =w loadub %x_addr
    %diag = 0
    %prev = 0
    %j = 1
    %row_addr = addl %row, %j
    %y_addr = addl %y, %j
    %y_addr = subl %y_addr, 1
inner_loop:
    %c = csltl %n, %j
    jnz @inner_done
    %by =w loadub %y_addr
    %up = loadl %row_addr
    %eq = ceql %bx, %by
    jnz @eq_case
    %gt = csltl %prev, %up
    jnz @max_up
    %new_val = %prev
    jmp @common
@max_up:
    %new_val = %up
    jmp @common
@eq_case:
    %new_val = addl %diag, 1
    jmp @common
@common:
    storel %row_addr, %new_val
    %diag = %up
    %prev = %new_val
    %row_addr = addl %row_addr, 8
    %y_addr = addl %y_addr, 1
    %j = addl %j, 1
    jmp @inner_loop
@inner_done:
    %i = addl %i, 1
    jmp @outer_loop
@outer_done:
    %res = loadl addl %row, %n
    ret %res
