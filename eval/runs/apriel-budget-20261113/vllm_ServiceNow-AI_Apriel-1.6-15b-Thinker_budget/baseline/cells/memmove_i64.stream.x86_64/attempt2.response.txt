memmove_i64:
	mov r8, rdi
	mov r9, rsi
	mov rcx, rdx
	test rcx, rcx
	jz .Ldone
	mov rdx, rcx
	shl rdx, 3
	mov r10, r9
	add r10, rdx
	mov r11, r8
	add r11, rdx
	sub r10, 8
	sub r11, 8
.Lloop:
	test rcx, rcx
	jz .Ldone
	mov rax, qword ptr [r10]
	mov qword ptr [r11], rax
	sub r10, 8
	sub r11, 8
	dec rcx
	jmp .Lloop
.Ldone:
	xor rax, rax
	ret
