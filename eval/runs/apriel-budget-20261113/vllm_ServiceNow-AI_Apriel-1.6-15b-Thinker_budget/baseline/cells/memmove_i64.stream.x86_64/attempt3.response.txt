memmove_i64:
	mov rcx, rdx
	test rcx, rcx
	jz .Ldone
	mov r8, rdx
	shl r8, 3
	cmp rdi, rsi
	jl .forward_copy
	mov r9, rsi
	add r9, r8
	sub r9, 8
	mov r10, rdi
	add r10, r8
	sub r10, 8
.backward_loop:
	mov rax, qword ptr [r9]
	mov qword ptr [r10], rax
	sub r9, 8
	sub r10, 8
	dec rcx
	jnz .backward_loop
	jmp .Ldone
.forward_copy:
	mov r9, rsi
	mov r10, rdi
.forward_loop:
	mov rax, qword ptr [r9]
	mov qword ptr [r10], rax
	add r9, 8
	add r10, 8
	dec rcx
	jnz .forward_loop
.Ldone:
	xor rax, rax
	ret
