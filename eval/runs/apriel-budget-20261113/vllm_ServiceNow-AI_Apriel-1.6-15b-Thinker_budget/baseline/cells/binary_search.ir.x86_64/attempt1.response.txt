export function l $binary_search(l, l, l)
@label Lstart:
    %lo = 0
    %hi = n
    %result = -1
    jmp Lloop
@label Lloop:
    %lt = csltl %lo, %hi
    jnz %lt, Lmid
    jmp Lreturn
@label Lmid:
    %sum = add %lo, %hi
    %mid = shr %sum, 1
    %val = loadl %val, a + %mid
    %eq = ceql %val, key
    jnz %eq, Lfound
    %lt2 = csltl %val, key
    jnz %lt2, Lset_lo
    %hi = %mid
    jmp Lloop
@label Lset_lo:
    %newlo = add %mid, 1
    %lo = %newlo
    jmp Lloop
@label Lfound:
    %result = %mid
    jmp Lreturn
@label Lreturn:
    ret l %result
