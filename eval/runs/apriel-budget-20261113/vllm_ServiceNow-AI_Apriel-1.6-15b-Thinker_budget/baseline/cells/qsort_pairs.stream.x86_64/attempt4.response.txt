qsort_pairs:
    push rbx
    push r12
    push r13
    cmp rsi, 1
    jle .Lbase_done
    mov rdx, rsi
    dec rdx
    lea r10, [rdi + rdx*16]
    mov r8, QWORD PTR [r10]
    mov r9, QWORD PTR [r10+8]
    xor rbx, rbx
    mov rcx, 1
.Lloop:
    cmp rcx, rdx
    jle .Lpartition_done
    lea r10, [rdi + rcx*16]
    mov rax, QWORD PTR [r10]
    cmp rax, r8
    jl .Lswap
    je
        mov rax, QWORD PTR [r10+8]
        cmp rax, r9
        jl .Lswap
    inc rcx
    jmp .Lloop
.Lswap:
    lea r11, [rdi + rbx*16]
    mov r12, QWORD PTR [r11]
    mov r13, QWORD PTR [r11+8]
    mov QWORD PTR [r11], QWORD PTR [r10]
    mov QWORD PTR [r11+8], QWORD PTR [r10+8]
    mov QWORD PTR [r10], r12
    mov QWORD PTR [r10+8], r13
    inc rbx
    inc rcx
    jmp .Lloop
.Lpartition_done:
    lea r11, [rdi + rbx*16]
    lea r10, [rdi + rdx*16]
    mov r12, QWORD PTR [r11]
    mov r13, QWORD PTR [r11+8]
    mov QWORD PTR [r11], QWORD PTR [r10]
    mov QWORD PTR [r11+8], QWORD PTR [r10+8]
    mov QWORD PTR [r10], r12
    mov QWORD PTR [r10+8], r13
    mov r13, rdi
    mov r12, rdx
    cmp rbx, 0
    jle .Lright_call
    mov rsi, rbx
    call qsort_pairs
.Lright_call:
    mov rsi, r12
    sub rsi, rbx
    cmp rsi, 0
    jle .Lreturn
    lea rdi, [r13 + rbx*16 + 16]
    call qsort_pairs
.Lreturn:
    pop r13
    pop r12
    pop rbx
    xor eax, eax
    ret
.Lbase_done:
    pop r13
    pop r12
    pop rbx
    xor eax, eax
    ret
