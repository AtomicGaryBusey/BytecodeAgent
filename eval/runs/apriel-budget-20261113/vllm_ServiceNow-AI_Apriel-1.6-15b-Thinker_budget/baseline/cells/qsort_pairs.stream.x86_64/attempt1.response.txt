qsort_pairs:
    push rbx
    push r12
    push r13
    sub rsp, 8
    mov r12, rsi
    cmp rsi, 1
    jle .Lbasecase
    dec rsi
    shl rsi, 1
    mov r10, rdi
    add r10, rsi
    mov rax, [r10]
    mov rdx, [r10+8]
    mov rsi, r12
    xor r8, r8
    xor rcx, rcx
    mov rbx, r12
    dec rbx
.Lloop:
    cmp rcx, rbx
    jge .Lafter
    lea r11, [rdi + rcx*8]
    cmp [r11], rax
    jl .Lswap
    je 
    cmp [r11+8], rdx
    jl .Lswap
    jmp .Lnextk
.Lswap:
    lea rax, [rdi + r8*8]
    xchg [rax], [r11]
    xchg [rax+8], [r11+8]
    inc r8
    jmp .Lnextk
.Lnextk:
    inc rcx
    jmp .Lloop
.Lafter:
    lea rax, [rdi + r8*8]
    xchg [rax], [r10]
    xchg [rax+8], [r10+8]
    mov r13, r8
    test r8, r8
    jz .Lskip_left
    mov rsi, r8
    call qsort_pairs
.Lskip_left:
    mov rdx, r12
    mov rax, r13
    sub rdx, rax
    dec rdx
    test rdx, rdx
    jz .Lend
    mov rax, rdi
    mov rbx, r13
    inc rbx
    shl rbx, 1
    add rax, rbx
    mov rdi, rax
    mov rsi, rdx
    call qsort_pairs
.Lend:
    add rsp, 8
    pop r13
    pop r12
    pop rbx
    xor rax, rax
    ret
.Lbasecase:
    add rsp, 8
    pop r13
    pop r12
    pop rbx
    xor rax, rax
    ret
