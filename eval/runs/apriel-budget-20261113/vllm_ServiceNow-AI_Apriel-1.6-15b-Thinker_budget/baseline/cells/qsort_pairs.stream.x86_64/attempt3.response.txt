qsort_pairs:
    sub rsp, 8
    push rbx
    push r12
    push r13
    mov rbx, rsi          ; n
    cmp rbx, 1
    jle .Lbase
    mov rdx, rbx          ; n
    dec rdx               ; hi = n-1
    shl rdx, 1            ; hi2 = hi*2
    mov r12, rdx          ; hi2 saved
    lea rax, [rdi + rdx*8]    ; address of pivot record
    mov r8, qword ptr [rax]   ; pivot key
    mov r9, qword ptr [rax+8]  ; pivot value
    leaq r10, -2           ; i2 = -2 (i = -1)
    xor r11, r11           ; j2 = 0
.Ljloop:
    cmp r11, r12
    jge .Lpostpartition
    lea rcx, [rdi + r11*8]      ; address of record j
    mov rax, [rcx]              ; key_j
    mov rdx, [rcx+8]            ; val_j
    cmp rax, r8
    jl .Lswap
    je .Lvalcmp
.Lvalcmp:
    cmp rdx, r9
    jl .Lswap
    add r11, 2                 ; j2 += 2
    jmp .Ljloop
.Lswap:
    leaq r10, [r10+2]           ; i2 += 2
    lea rcx, [rdi + r10*8]      ; address i+1
    lea r13, [rdi + r11*8]      ; address j
    movdqu xmm0, [rcx]
    movdqu xmm1, [r13]
    movdqu xmm0, [r13]
    movdqu xmm1, [rcx]
    add r11, 2                 ; j2 += 2
    jmp .Ljloop
.Lpostpartition:
    leaq r10, [r10+2]           ; p2 = (i+1)*2
    lea rcx, [rdi + r10*8]      ; address p
    mov rdx, rbx                ; n
    dec rdx                     ; hi = n-1
    shl rdx, 1                  ; hi2
    lea rdx, [rdi + rdx*8]      ; address hi
    movdqu xmm0, [rcx]
    movdqu xmm1, [rdx]
    movdqu xmm0, [rdx]
    movdqu xmm1, [rcx]
    mov r12, r10                ; p2
    shr r12, 1                 ; p = p2/2
    mov rsi, r12
    call qsort_pairs
    mov rdx, rbx                ; n
    sub rdx, r12                ; n - p
    dec rdx                     ; n - p - 1
    mov rsi, rdx
    inc r12                     ; p+1
    shl r12, 1                  ; (p+1)*2
    lea rdi, [rdi + r12*8]
    call qsort_pairs
    jmp .Lexit
.Lbase:
    xor eax, eax
    jmp .Lexit
.Lexit:
    add rsp, 8
    pop r13
    pop r12
    pop rbx
    ret
