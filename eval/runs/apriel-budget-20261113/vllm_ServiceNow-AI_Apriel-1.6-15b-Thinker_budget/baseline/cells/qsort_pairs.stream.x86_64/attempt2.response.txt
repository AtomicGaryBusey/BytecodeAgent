qsort_pairs:
    push rbx
    push r12
    sub rsp, 8
    mov r12, rsi
    cmp rsi, 1
    jle .Lbase
    mov rdx, rsi
    dec rdx
    mov rax, rdx
    shl rax, 1
    lea rax, [rdi + rax*8]
    mov r8, qword ptr [rax]
    mov r9, qword ptr [rax+8]
    xor r10, r10
    xor r11, r11
.Ljloop:
    cmp r11, rdx
    jge .Lpartition_done
    mov rax, r11
    shl rax, 1
    lea rax, [rdi + rax*8]
    mov r12, qword ptr [rax]
    mov rcx, qword ptr [rax+8]
    cmp r12, r8
    jl .Lswap
    je .Lvalcmp
.Lvalcmp:
    cmp rcx, r9
    jl .Lswap
    inc r11
    jmp .Ljloop
.Lswap:
    mov rcx, r10
    shl rcx, 1
    lea rcx, [rdi + rcx*8]
    movdqa xmm0, [rax]
    movdqa xmm1, [rcx]
    movdqa [rcx], xmm0
    movdqa [rax], xmm1
    inc r10
    inc r11
    jmp .Ljloop
.Lpartition_done:
    mov rax, r10
    shl rax, 1
    lea rax, [rdi + rax*8]
    mov rcx, rdx
    shl rcx, 1
    lea rcx, [rdi + rcx*8]
    movdqa xmm0, [rax]
    movdqa xmm1, [rcx]
    movdqa [rax], xmm1
    movdqa [rcx], xmm0
    cmp r10, 0
    jle .Lright_recurse
    mov rsi, r10
    call qsort_pairs
.Lright_recurse:
    mov rdx, r12
    sub rdx, r10
    dec rdx
    cmp rdx, 0
    jle .Lreturn
    mov rax, r10
    inc rax
    shl rax, 1
    lea rax, [rdi + rax*8]
    mov rsi, rdx
    mov rdi, rax
    call qsort_pairs
.Lreturn:
    xor eax, eax
    add rsp, 8
    pop r12
    pop rbx
    ret
.Lbase:
    xor eax, eax
    add rsp, 8
    pop r12
    pop rbx
    ret
