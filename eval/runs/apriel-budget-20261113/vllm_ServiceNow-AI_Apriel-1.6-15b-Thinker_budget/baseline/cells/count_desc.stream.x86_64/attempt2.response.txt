count_desc:
    cmp rsi, 1
    jle .Lreturn0
    xor rdx, rdx
    mov r8, qword ptr [rdi]
    mov r9, 1
.Lloop:
    cmp r9, rsi
    jge .Lreturn
    mov r10, qword ptr [rdi + r9*8]
    cmp r8, r10
    jg .Linc
    mov r8, r10
    inc r9
    jmp .Lloop
.Linc:
    inc rdx
    mov r8, r10
    inc r9
    jmp .Lloop
.Lreturn:
    mov rax, rdx
    ret
.Lreturn0:
    xor rax, rax
    ret
