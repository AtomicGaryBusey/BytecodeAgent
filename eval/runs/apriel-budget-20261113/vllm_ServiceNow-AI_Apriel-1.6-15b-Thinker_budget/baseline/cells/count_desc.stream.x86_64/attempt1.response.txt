count_desc:
    cmp rsi, 1
    jle .Ldone
    mov rax, qword ptr [rdi]
    xor rdx, rdx
    xor r8, r8
    mov rcx, rsi
    dec rcx
.Lloop:
    mov r9, qword ptr [rdi + r8*8 + 8]
    cmp rax, r9
    jg .Linc
    mov rax, r9
    jmp .Ldec
.Linc:
    inc rdx
.Ldec:
    inc r8
    dec rcx
    jnz .Lloop
    mov rax, rdx
.Ldone:
    ret
