gcd:
    push rbx
    test rdi, rdi
    jz .Lret_rsi
    test rsi, rsi
    jz .Lret_rdi
    xor rbx, rbx
.Lcommon_loop:
    test rdi, 1
    jnz .Lcommon_done
    test rsi, 1
    jnz .Lcommon_done
    shr rdi, 1
    shr rsi, 1
    inc rbx
    jmp .Lcommon_loop
.Lcommon_done:
.Lmain:
    test rsi, rsi
    jz .Lfinal
    .Lmake_b_odd:
        test rsi, 1
        jnz .Lmake_b_odd_done
        shr rsi, 1
        jmp .Lmake_b_odd
    .Lmake_b_odd_done:
    cmp rdi, rsi
    jle .Lno_swap
    xor rdi, rsi
    xor rsi, rdi
    xor rdi, rsi
.Lno_swap:
    sub rsi, rdi
    .Lremove_factors:
        test rsi, 1
        jnz .Lremove_factors_done
        shr rsi, 1
        jmp .Lremove_factors
    .Lremove_factors_done:
    jmp .Lmain
.Lfinal:
    mov cl, bl
    shl rdi, cl
    mov rax, rdi
    pop rbx
    ret
.Lret_rsi:
    mov rax, rsi
    pop rbx
    ret
.Lret_rdi:
    mov rax, rdi
    pop rbx
    ret
