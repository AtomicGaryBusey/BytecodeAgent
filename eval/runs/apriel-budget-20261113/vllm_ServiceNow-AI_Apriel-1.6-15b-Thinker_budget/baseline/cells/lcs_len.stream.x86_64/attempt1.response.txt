lcs_len:
	push	rbx
	push	r12
	push	r13
	push	r14
	push	r15
	mov	r9, rsi
	xor	rax, rax
	mov	[ r8], rax
	mov	r12, rcx
ZeroRow:
	mov	[ r8 + r12*8], rax
	dec	r12
	jne	ZeroRow
Outer:
	test	rsi, rsi
	jle	.Ldone
	xor	rbx, rbx
	mov	r13, r9
	sub	r13, rsi
	movzx	r14, byte [rdi + r13]
	mov	r15, 1
Inner:
	movzx	r11, byte [rdx + r15*1 - 1]
	mov	r12, qword [r8 + r15*8]
	mov	r13, qword [r8 + r15*8 - 8]
	cmp	r14, r11
	je	.CaseEqual
	mov	rax, r12
	cmp	rax, r13
	jge	.StoreNew
	mov	rax, r13
.StoreNew:
	mov	[ r8 + r15*8], rax
	mov	rbx, r12
	inc	r15
	cmp	r15, rcx
	jle	Inner
.CaseEqual:
	lea	rax, [rbx + 1]
	mov	[ r8 + r15*8], rax
	mov	rbx, r12
	inc	r15
	cmp	r15, rcx
	jle	Inner
	dec	rsi
	jmp	Outer
.Ldone:
	mov	rax, qword [r8 + rcx*8]
	pop	r15
	pop	r14
	pop	r13
	pop	r12
	pop	rbx
	ret
