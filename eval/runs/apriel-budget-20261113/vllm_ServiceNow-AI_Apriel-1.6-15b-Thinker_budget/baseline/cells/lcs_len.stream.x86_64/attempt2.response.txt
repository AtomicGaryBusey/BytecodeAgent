lcs_len:
	push	rbx
	push	r12
	push	r13
	push	r14
	push	r15
	mov	r12, rcx
	xor	rax, rax
	xor	r11, r11
zero_row:
	mov	[ r8 + r11*8], rax
	inc	r11
	cmp	r11, r12
	jne	zero_row
	xor	rbx, rbx
.outer_loop:
	cmp	rbx, rsi
	jge	.Lend
	movzx	r14, byte [rdi + rbx]
	xor	r13, r13
	mov	r9, rdx
	mov	rcx, 1
	cmp	rcx, r12
	jg	.Lskip_inner
inner_loop:
	movzx	rdx, byte [r9 + rcx - 1]
	mov	r15, [r8 + rcx*8]
	mov	r10, [r8 + rcx*8 - 8]
	cmp	r14, rdx
	je	.Lmatch_body
	cmp	r15, r10
	jge	.Lnew_up
	mov	r11, r10
	jmp	.Lstore_body
.Lnew_up:
	mov	r11, r15
.Lstore_body:
	mov	[ r8 + rcx*8], r11
	mov	r13, r15
	jmp	.Lcommon
.Lmatch_body:
	lea	r11, [r13 + 1]
	mov	[ r8 + rcx*8], r11
	mov	r13, r15
.Lcommon:
	inc	rcx
	cmp	rcx, r12
	jle	inner_loop
	jmp	.Lskip_inner_end
.Lskip_inner:
.Lskip_inner_end:
	inc	rbx
	jmp	.outer_loop
.Lend:
	mov	rax, [r8 + r12*8]
	pop	r15
	pop	r14
	pop	r13
	pop	r12
	pop	rbx
	ret
