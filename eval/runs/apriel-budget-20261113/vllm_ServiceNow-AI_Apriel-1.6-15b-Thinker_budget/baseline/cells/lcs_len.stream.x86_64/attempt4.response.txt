zero_row:

	inc	r9
	cmp	r9, rcx
	jle	zero_row
	xor	rbx, rbx
.outer_loop:
	cmp	rbx, rsi
	jge	.done_outer
	movzx	r15, byte ptr [rdi + rbx]
	xor	r13, r13
	mov	r12, 1

	jg	.inner_done

	movzx	r11, byte ptr [rdx + r12 - 1]

	mov	r13, r14

	inc	r10
	mov	qword ptr [r8 + r12*8], r10
	mov	r13, r14
.inner_next:
	inc	r12
	jmp	.inner_loop
.inner_done:
	inc	rbx
	jmp	.outer_loop

	pop	r14
	pop	r13
