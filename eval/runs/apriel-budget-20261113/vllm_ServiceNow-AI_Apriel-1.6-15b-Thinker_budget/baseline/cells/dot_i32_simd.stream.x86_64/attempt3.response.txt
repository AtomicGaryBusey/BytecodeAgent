dot_i32_simd:
    xor rax, rax
.Lloop:
    test rdx, rdx
    jle .Ldone
    cmp rdx, 4
    jl .Lscalar
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pmovsxdq xmm0, xmm0
    pmovsxdq xmm1, xmm1
    pmuldq xmm0, xmm1
    movq rcx, xmm0
    add rax, rcx
    pslldq xmm0, 8
    movq rcx, xmm0
    add rax, rcx
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    jmp .Lloop
.Lscalar:
    test rdx, rdx
    jle .Ldone
    movsxd rcx, dword ptr [rdi]
    movsxd rdx, dword ptr [rsi]
    imul rcx, rdx
    add rax, rcx
    add rdi, 4
    add rsi, 4
    sub rdx, 1
    jmp .Lscalar
.Ldone:
    ret
