dot_i32_simd:
    sub rsp, 64
    xor r8, r8
.Ltop:
    test rdx, rdx
    jle .Ldone
    cmp rdx, 8
    jl .Lscalar
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm4, xmm2, xmm3
    mov [rsp], xmm4
    xor rcx, rcx
.Lvec_loop:
    mov rax, qword ptr [rsp + rcx*8]
    add r8, rax
    inc rcx
    cmp rcx, 8
    jl .Lvec_loop
    lea rdi, [rdi+32]
    lea rsi, [rsi+32]
    sub rdx, 8
    jmp .Ltop
.Lscalar:
    test rdx, rdx
    jle .Ldone
    mov eax, [rdi]
    mov r9, [rsi]
    imul rax, r9
    add r8, rax
    lea rdi, [rdi+4]
    lea rsi, [rsi+4]
    dec rdx
    jmp .Lscalar
.Ldone:
    mov rax, r8
    add rsp, 64
    ret
