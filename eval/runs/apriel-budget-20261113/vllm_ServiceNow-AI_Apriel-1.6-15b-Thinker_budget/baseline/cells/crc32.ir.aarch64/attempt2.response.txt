export function l $crc32(p,n)
@label entry:
    %zero =w const 0
    %one =w const 1
    %poly =w const 0xEDB88320
    %init =w const 0xffffffff
    %crc =w const 0xffffffff
    %ptr =l p
    %cnt =l n
@label outer_loop_start:
    %cnt_eq =w ceql %cnt, %zero
    jnz %cnt_eq, @label after_bytes
    %byte =w loadub %ptr
    %crc =w xor %crc, %byte
    %i =w const 8
    @label bit_loop_start:
        %lsb =w and %crc, %one
        %crc =w shr %crc, 1
        jnz %lsb, @label bit_xor_body
        @label after_shift_no_xor:
            %i =w sub %i, #1
            %i_eq =w ceql %i, %zero
            jnz %i_eq, @label bit_loop_done
            j @label bit_loop_start
    @label bit_xor_body:
        %crc =w xor %crc, %poly
        %i =w sub %i, #1
        %i_eq =w ceql %i, %zero
        jnz %i_eq, @label bit_loop_done
        j @label bit_loop_start
    @label bit_loop_done:
        %ptr =l add %ptr, #1
        %cnt =l sub %cnt, #1
        j @label outer_loop_start
@label after_bytes:
    %final =w xor %crc, %init
    %result =l extuw %final
    ret %result
