export function l $crc32(%p,%n)
    %zero =w const 0
    %one =w const 1
    %poly =w const 0xEDB88320
    %init =w const 0xffffffff
    %crc =w const 0xffffffff
@label outer_loop_start:
    %cnt_eq =w ceql %n, %zero
    jnz %cnt_eq, @label after_bytes
    %byte =w loadub %p
    %crc =w xor %crc, %byte
    %i =w const 8
    @label bit_loop_start:
        %lsb =w and %crc, %one
        %crc =w shr %crc, 1
        jnz %lsb, @label bit_xor_body
        %i =w sub %i, #1
        %i_eq =w ceql %i, %zero
        jnz %i_eq, @label bit_loop_done
        j @label bit_loop_start
    @label bit_xor_body:
        %crc =w xor %crc, %poly
        %i =w sub %i, #1
        %i_eq =w ceql %i, %zero
        jnz %i_eq, @label bit_loop_done
        j @label bit_loop_start
    @label bit_loop_done:
    %p =l add %p, #1
    %n =l sub %n, #1
    j @label outer_loop_start
@label after_bytes:
    %final =w xor %crc, %init
    %result =l extuw %final
    ret %result
