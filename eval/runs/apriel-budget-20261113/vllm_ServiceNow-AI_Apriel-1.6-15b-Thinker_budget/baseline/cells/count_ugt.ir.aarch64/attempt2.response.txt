export function l $count_ugt(l a, l n, l t) {
    @start:
      mov a, %ptr
      mov $0, %cnt
      mov n, %len
    @loop:
      ceql %len, $0, %zero
      jnz %zero, @end
      loadl %ptr, %val
      cgtl %val, t, %gt
      jnz %gt, @inc_cnt
      @skip:
      addl $8, %ptr, %ptr
      sub $1, %len, %len
      jmp @loop
    @inc_cnt:
      addl $1, %cnt, %cnt
      addl $8, %ptr, %ptr
      sub $1, %len, %len
      jmp @loop
    @end:
      mov %cnt, l0
      ret
}
