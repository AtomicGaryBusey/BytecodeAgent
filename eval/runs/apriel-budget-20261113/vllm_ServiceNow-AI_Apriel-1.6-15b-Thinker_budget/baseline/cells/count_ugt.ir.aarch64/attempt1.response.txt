export function l $count_ugt(l l0, l l1, l l2) {
    @start:
      mov l0, %ptr
      mov $0, %cnt
    @loop:
      ceql l1, $0, %zero
      jnz %zero, @end
      loadl %ptr, %val
      cgtl %val, l2, %gt
      jnz %gt, @inc_cnt
      @skip:
      addl $8, %ptr, %ptr
      sub $1, l1, l1
      jmp @loop
    @inc_cnt:
      addl $1, %cnt, %cnt
      addl $8, %ptr, %ptr
      sub $1, l1, l1
      jmp @loop
    @end:
      mov %cnt, l0
      ret
}
