export function l $map_sum(l a, l n, l f) {
    @Lstart:
        sub rsp, 24
        %addr_sum =l rsp
        %addr_i =l addl rsp, 8
        %sum0 =l 0
        storel %addr_sum, %sum0
        %i0 =l 0
        storel %addr_i, %i0
    @Lcond:
        %i_load =l loadl %addr_i
        %c =l csltl %i_load %n
        jnz %c @Lbody
        %sum_ret =l loadl %addr_sum
        add rsp, 24
        retl %sum_ret
    @Lbody:
        %sum_load =l loadl %addr_sum
        %i_load =l loadl %addr_i
        %addr =l addl %a, mull %i_load, 8
        %val =l loadl %addr
        %res =l call %f(l %val)
        %sum_new =l addl %sum_load, %res
        storel %addr_sum, %sum_new
        %i_after =l loadl %addr_i
        %i_new =l addl %i_after, 1
        storel %addr_i, %i_new
        jmp @Lcond
}
