export function l $map_sum(l *a, l n, l (*f)(l)) {
    @Lstart:
        %sum =l 0
        %i =l 0
        @Lcond:
            %c =l csltl %i %n
            jnz %c @Lbody
            retl %sum
        @Lbody:
            %tmp =l mull %i, 8
            %addr =l addl %a, %tmp
            %val =l loadl %addr
            %res =l call %f(l %val)
            %sum =l addl %sum, %res
            %i =l addl %i, 1
            jmp @Lcond
}
