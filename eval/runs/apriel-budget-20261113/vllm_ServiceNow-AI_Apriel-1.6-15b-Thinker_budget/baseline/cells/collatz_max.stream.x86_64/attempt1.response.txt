collatz_max:
xor rsi, rsi
inc rsi
xor r8, r8
.Louter:
mov rax, rsi
xor rdx, rdx
.Linner:
cmp rax, 1
je .Linner_done
test rax, 1
jz .Leven
.Leodd:
lea rax, [rax + rax*2 + 1]
inc rdx
jmp .Linner
.Leven:
shr rax, 1
inc rdx
jmp .Linner
.Linner_done:
cmp rdx, r8
jle .Lafter_max
mov r8, rdx
.Lafter_max:
inc rsi
cmp rsi, rdi
jle .Louter
mov rax, r8
ret
