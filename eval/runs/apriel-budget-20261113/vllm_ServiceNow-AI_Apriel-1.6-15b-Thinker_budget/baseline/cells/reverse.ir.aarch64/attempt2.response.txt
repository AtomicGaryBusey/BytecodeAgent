export function l $reverse(l a, l n) {
    @label L0
    %i = 0
    %j = %n - 1
    @label Lloop
    csltl %lt, %i, %j
    jnz %lt, Lswap
    @label Ldone
    ret 0
    @label Lswap
    %tmp1 = loadl %a, %i
    %tmp2 = loadl %a, %j
    storel %a, %i, %tmp2
    storel %a, %j, %tmp1
    addl %i, %i, 1
    subl %j, %j, 1
    j Lloop
}
