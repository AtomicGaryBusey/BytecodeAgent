binary_search:
    mov rcx, 0
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jge .Lend
    lea r9, [rcx + r8]
    shr r9, 1
    mov rax, [rdi + r9*8]
    cmp rax, rdx
    je .Lfound
    jl .Lleft
    jg .Lright
.Lleft:
    lea rcx, [r9 + 1]
    jmp .Lloop
.Lright:
    mov r8, r9
    jmp .Lloop
.Lfound:
    mov rax, r9
    jmp .Lret
.Lend:
    xor eax, eax
    dec rax
    jmp .Lret
.Lret:
    ret
