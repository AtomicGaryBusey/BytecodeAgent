export function l $map_into(%dst l %src l %n l %f l) @L0
    %cnt =l 0
    %i =l 0
    %one =l 1
    %three =l 3
@Lloop
    %cond = w csltl %i %n
    jnz @Lbody %cond
    jmp @Lexit
@Lbody
    %src_off =l shl %i %three
    %src_addr =l add %src %src_off
    %src_val =l loadl %src_addr
    %res =l call %f(l %src_val)
    %dst_off =l shl %i %three
    %dst_addr =l add %dst %dst_off
    storel %dst_addr %res
    %cmp = w csltl %res %src_val
    jnz @Linc %cmp
    jmp @Lnext
@Linc
    %cnt =l add %cnt %one
@Lnext
    %i =l add %i %one
    jmp @Lloop
@Lexit
    ret %cnt
