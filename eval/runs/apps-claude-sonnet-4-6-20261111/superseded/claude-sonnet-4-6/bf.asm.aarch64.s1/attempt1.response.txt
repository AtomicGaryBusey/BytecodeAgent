.globl _start

.equ SYS_READ,    63
.equ SYS_WRITE,   64
.equ SYS_EXIT,    93
.equ INPUT_MAX,   65536
.equ BRACKET_MAX, 1000
.equ OUTPUT_MAX,  65536

.text

_start:
    adrp x20, input_buf
    add  x20, x20, :lo12:input_buf
    mov  x19, #0

read_loop:
    cmp  x19, #INPUT_MAX
    b.ge read_done
    mov  x0, #0
    add  x1, x20, x19
    mov  x2, #INPUT_MAX
    sub  x2, x2, x19
    mov  x8, #SYS_READ
    svc  #0
    cmp  x0, #1
    b.lt read_done
    add  x19, x19, x0
    b    read_loop

read_done:
    mov  x0, #0

find_sep:
    cmp  x0, x19
    b.ge no_sep
    ldrb w1, [x20, x0]
    cmp  w1, #0x21
    b.eq got_sep
    add  x0, x0, #1
    b    find_sep

got_sep:
    mov  x21, x0
    add  x22, x0, #1
    subs x23, x19, x22
    b.pl check_brk
    mov  x23, #0
    b    check_brk

no_sep:
    mov  x21, x19
    mov  x22, x19
    mov  x23, #0

check_brk:
    adrp x24, bracket_stack
    add  x24, x24, :lo12:bracket_stack
    adrp x25, jump_table
    add  x25, x25, :lo12:jump_table
    mov  x0, #0
    mov  x1, #0

brk_scan:
    cmp  x0, x21
    b.ge brk_ok
    ldrb w2, [x20, x0]
    cmp  w2, #0x5B
    b.eq brk_push
    cmp  w2, #0x5D
    b.eq brk_pop
    add  x0, x0, #1
    b    brk_scan

brk_push:
    cmp  x1, #BRACKET_MAX
    b.ge do_exit1
    str  w0, [x24, x1, lsl #2]
    add  x1, x1, #1
    add  x0, x0, #1
    b    brk_scan

brk_pop:
    cbz  x1, do_exit1
    sub  x1, x1, #1
    ldr  w3, [x24, x1, lsl #2]
    add  w4, w0, #1
    str  w4, [x25, x3, lsl #2]
    add  w4, w3, #1
    str  w4, [x25, x0, lsl #2]
    add  x0, x0, #1
    b    brk_scan

brk_ok:
    cbnz x1, do_exit1
    b    exec_start

do_exit1:
    mov  x0, #1
    mov  x8, #SYS_EXIT
    svc  #0

exec_start:
    adrp x27, tape
    add  x27, x27, :lo12:tape
    adrp x28, output_buf
    add  x28, x28, :lo12:output_buf
    mov  x0, #0
    mov  x1, #0
    mov  x26, #0
    mov  x19, #0

exec_loop:
    cmp  x0, x21
    b.ge exec_end
    ldrb w2, [x20, x0]
    cmp  w2, #0x3E
    b.eq op_right
    cmp  w2, #0x3C
    b.eq op_left
    cmp  w2, #0x2B
    b.eq op_plus
    cmp  w2, #0x2D
    b.eq op_minus
    cmp  w2, #0x2E
    b.eq op_dot
    cmp  w2, #0x2C
    b.eq op_comma
    cmp  w2, #0x5B
    b.eq op_lbr
    cmp  w2, #0x5D
    b.eq op_rbr
    add  x0, x0, #1
    b    exec_loop

op_right:
    add  x1, x1, #1
    add  x0, x0, #1
    b    exec_loop

op_left:
    sub  x1, x1, #1
    add  x0, x0, #1
    b    exec_loop

op_plus:
    ldrb w2, [x27, x1]
    add  w2, w2, #1
    and  w2, w2, #0xFF
    strb w2, [x27, x1]
    add  x0, x0, #1
    b    exec_loop

op_minus:
    ldrb w2, [x27, x1]
    sub  w2, w2, #1
    and  w2, w2, #0xFF
    strb w2, [x27, x1]
    add  x0, x0, #1
    b    exec_loop

op_dot:
    ldrb w2, [x27, x1]
    strb w2, [x28, x19]
    add  x19, x19, #1
    cmp  x19, #OUTPUT_MAX
    b.lt op_dot_done
    bl   flush_output
op_dot_done:
    add  x0, x0, #1
    b    exec_loop

op_comma:
    cmp  x26, x23
    b.ge op_comma_eof
    add  x3, x22, x26
    ldrb w2, [x20, x3]
    strb w2, [x27, x1]
    add  x26, x26, #1
    add  x0, x0, #1
    b    exec_loop
op_comma_eof:
    strb wzr, [x27, x1]
    add  x0, x0, #1
    b    exec_loop

op_lbr:
    ldrb w2, [x27, x1]
    cbnz w2, op_lbr_cont
    ldr  w0, [x25, x0, lsl #2]
    b    exec_loop
op_lbr_cont:
    add  x0, x0, #1
    b    exec_loop

op_rbr:
    ldrb w2, [x27, x1]
    cbz  w2, op_rbr_cont
    ldr  w0, [x25, x0, lsl #2]
    b    exec_loop
op_rbr_cont:
    add  x0, x0, #1
    b    exec_loop

exec_end:
    cbz  x19, do_exit0
    bl   flush_output
do_exit0:
    mov  x0, #0
    mov  x8, #SYS_EXIT
    svc  #0

flush_output:
    stp  x0, x30, [sp, #-32]!
    stp  x1, x2,  [sp, #16]
    mov  x0, #1
    adrp x1, output_buf
    add  x1, x1, :lo12:output_buf
    mov  x2, x19
    mov  x8, #SYS_WRITE
    svc  #0
    mov  x19, #0
    ldp  x1, x2,  [sp, #16]
    ldp  x0, x30, [sp], #32
    ret

.bss
.align 4
input_buf:
    .zero 65536
.align 4
tape:
    .zero 30000
.align 4
jump_table:
    .zero 262144
.align 4
bracket_stack:
    .zero 4000
.align 4
output_buf:
    .zero 65536
