.globl _start

.bss
.align 8
input_buf:
    .zero 65536
.align 4
tape:
    .zero 30000
.align 4
jump_table:
    .zero 262144
.align 8
bracket_stack:
    .zero 8000
.align 8
output_buf:
    .zero 65536
.align 8
total_read_var:
    .zero 8
prog_len_var:
    .zero 8
input_start_var:
    .zero 8

.text

_start:
    xor r12, r12

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov qword ptr [rip + total_read_var], r12
    lea rsi, [rip + input_buf]
    xor rcx, rcx

find_bang:
    cmp rcx, r12
    jge no_bang
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov qword ptr [rip + prog_len_var], rcx
    lea rax, [rcx + 1]
    mov qword ptr [rip + input_start_var], rax
    jmp build_jump_table

no_bang:
    mov qword ptr [rip + prog_len_var], r12
    mov qword ptr [rip + input_start_var], r12

build_jump_table:
    lea rsi, [rip + input_buf]
    mov rcx, qword ptr [rip + prog_len_var]
    lea r8, [rip + bracket_stack]
    xor r9, r9
    xor r10, r10
    lea r11, [rip + jump_table]

btloop:
    cmp r10, rcx
    jge bt_done
    movzx eax, byte ptr [rsi + r10]
    cmp al, 0x5b
    je bt_open
    cmp al, 0x5d
    je bt_close
    inc r10
    jmp btloop

bt_open:
    cmp r9, 1000
    jge bracket_error
    mov qword ptr [r8 + r9*8], r10
    inc r9
    inc r10
    jmp btloop

bt_close:
    test r9, r9
    jz bracket_error
    dec r9
    mov rax, qword ptr [r8 + r9*8]
    lea rbx, [r10 + 1]
    mov dword ptr [r11 + rax*4], ebx
    lea rbx, [rax + 1]
    mov dword ptr [r11 + r10*4], ebx
    inc r10
    jmp btloop

bt_done:
    test r9, r9
    jnz bracket_error
    jmp run_program

bracket_error:
    mov eax, 60
    mov edi, 1
    syscall

run_program:
    xor r12, r12
    xor r13, r13
    mov r14, qword ptr [rip + input_start_var]
    xor r15, r15
    mov rbx, qword ptr [rip + prog_len_var]
    lea rbp, [rip + tape]
    lea r8, [rip + input_buf]

exec_loop:
    cmp r12, rbx
    jge exec_done
    movzx eax, byte ptr [r8 + r12]
    cmp al, 0x3e
    je cmd_gt
    cmp al, 0x3c
    je cmd_lt
    cmp al, 0x2b
    je cmd_plus
    cmp al, 0x2d
    je cmd_minus
    cmp al, 0x2e
    je cmd_dot
    cmp al, 0x2c
    je cmd_comma
    cmp al, 0x5b
    je cmd_lbrace
    cmp al, 0x5d
    je cmd_rbrace
    inc r12
    jmp exec_loop

cmd_gt:
    inc r13
    inc r12
    jmp exec_loop

cmd_lt:
    dec r13
    inc r12
    jmp exec_loop

cmd_plus:
    inc byte ptr [rbp + r13]
    inc r12
    jmp exec_loop

cmd_minus:
    dec byte ptr [rbp + r13]
    inc r12
    jmp exec_loop

cmd_dot:
    movzx eax, byte ptr [rbp + r13]
    lea rdi, [rip + output_buf]
    mov byte ptr [rdi + r15], al
    inc r15
    cmp r15, 65536
    jl cmd_dot_done
    call flush_output
cmd_dot_done:
    inc r12
    jmp exec_loop

cmd_comma:
    mov rax, qword ptr [rip + total_read_var]
    cmp r14, rax
    jge comma_eof
    movzx eax, byte ptr [r8 + r14]
    inc r14
    mov byte ptr [rbp + r13], al
    inc r12
    jmp exec_loop
comma_eof:
    mov byte ptr [rbp + r13], 0
    inc r12
    jmp exec_loop

cmd_lbrace:
    movzx eax, byte ptr [rbp + r13]
    test eax, eax
    jnz lbrace_nonzero
    lea rsi, [rip + jump_table]
    mov r12d, dword ptr [rsi + r12*4]
    jmp exec_loop
lbrace_nonzero:
    inc r12
    jmp exec_loop

cmd_rbrace:
    movzx eax, byte ptr [rbp + r13]
    test eax, eax
    jz rbrace_zero
    lea rsi, [rip + jump_table]
    mov r12d, dword ptr [rsi + r12*4]
    jmp exec_loop
rbrace_zero:
    inc r12
    jmp exec_loop

exec_done:
    call flush_output
    mov eax, 60
    xor edi, edi
    syscall

flush_output:
    test r15, r15
    jz flush_ret
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r15
    xor r15, r15
    syscall
flush_ret:
    ret
