.globl _start

.bss
.align 8
input_buf:
    .zero 32768
.align 8
line_ptrs:
    .zero 8000
.align 8
line_lens:
    .zero 8000

.section .rodata
newline_byte:
    .byte 0x0A

.text

_start:
    xor r12d, r12d

read_loop:
    mov rdx, 32768
    sub rdx, r12
    jz read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz do_exit

    lea rbx, [rip + input_buf]
    xor r13d, r13d
    xor r14d, r14d

parse_loop:
    lea rax, [rbx + r13]
    lea rcx, [rip + line_ptrs]
    mov qword ptr [rcx + r14*8], rax
    mov rcx, r13

scan_for_nl:
    cmp rcx, r12
    jge partial_line
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 0x0A
    jne scan_for_nl

    mov rax, rcx
    dec rax
    sub rax, r13
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + r14*8], rax
    inc r14
    mov r13, rcx
    cmp r13, r12
    jl parse_loop
    jmp do_sort

partial_line:
    mov rax, r12
    sub rax, r13
    test rax, rax
    jz do_sort
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + r14*8], rax
    inc r14

do_sort:
    cmp r14, 1
    jle print_all
    mov r15, 1

outer_loop:
    cmp r15, r14
    jge print_all

    lea rax, [rip + line_ptrs]
    mov r8, qword ptr [rax + r15*8]
    lea rax, [rip + line_lens]
    mov r9, qword ptr [rax + r15*8]

    mov rbx, r15
    dec rbx

inner_loop:
    test rbx, rbx
    js do_insert

    lea rax, [rip + line_ptrs]
    mov rdi, qword ptr [rax + rbx*8]
    lea rax, [rip + line_lens]
    mov rsi, qword ptr [rax + rbx*8]

    call cmp_lines

    test eax, eax
    jle do_insert

    lea rax, [rip + line_ptrs]
    mov rcx, qword ptr [rax + rbx*8]
    mov qword ptr [rax + rbx*8 + 8], rcx
    lea rax, [rip + line_lens]
    mov rcx, qword ptr [rax + rbx*8]
    mov qword ptr [rax + rbx*8 + 8], rcx

    dec rbx
    jmp inner_loop

do_insert:
    inc rbx
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + rbx*8], r8
    lea rax, [rip + line_lens]
    mov qword ptr [rax + rbx*8], r9

    inc r15
    jmp outer_loop

cmp_lines:
    xor ecx, ecx
cmp_loop:
    cmp rcx, rsi
    jge cmp1_end
    cmp rcx, r9
    jge cmp2_end
    movzx eax, byte ptr [rdi + rcx]
    movzx edx, byte ptr [r8 + rcx]
    sub eax, edx
    jnz cmp_ret
    inc rcx
    jmp cmp_loop
cmp1_end:
    cmp rcx, r9
    jge cmp_equal
    mov eax, -1
    ret
cmp2_end:
    mov eax, 1
    ret
cmp_equal:
    xor eax, eax
cmp_ret:
    ret

print_all:
    xor r15d, r15d

print_loop:
    cmp r15, r14
    jge do_exit

    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + r15*8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + r15*8]

    test rdx, rdx
    jz write_nl

    mov rax, 1
    mov rdi, 1
    syscall

write_nl:
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall

    inc r15
    jmp print_loop

do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
