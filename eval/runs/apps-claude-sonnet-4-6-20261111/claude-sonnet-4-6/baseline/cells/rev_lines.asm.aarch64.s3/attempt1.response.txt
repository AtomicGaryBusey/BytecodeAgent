.globl _start

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    svc #0
    cbz x0, process_start
    add x20, x20, x0
    cmp x20, #65536
    b.lt read_loop

process_start:
    mov x21, #0

process_lines:
    cmp x21, x20
    b.ge all_done
    mov x22, x21

scan_loop:
    cmp x21, x20
    b.ge found_end
    ldrb w0, [x19, x21]
    cmp w0, #0x0A
    b.eq found_newline
    add x21, x21, #1
    b scan_loop

found_newline:
    sub x0, x21, x22
    add x1, x19, x22
    bl reverse_and_write
    add x21, x21, #1
    b process_lines

found_end:
    cmp x22, x20
    b.ge all_done
    sub x0, x20, x22
    add x1, x19, x22
    bl reverse_and_write

all_done:
    mov x8, #93
    mov x0, #0
    svc #0


reverse_and_write:
    stp x29, x30, [sp, #-32]!
    stp x19, x20, [sp, #16]
    mov x19, x0
    mov x20, x1
    cbz x19, rw_newline
    mov x2, x20
    add x3, x20, x19
    sub x3, x3, #1

rw_reverse:
    cmp x2, x3
    b.ge rw_write
    ldrb w4, [x2]
    ldrb w5, [x3]
    strb w5, [x2]
    strb w4, [x3]
    add x2, x2, #1
    sub x3, x3, #1
    b rw_reverse

rw_write:
    mov x8, #64
    mov x0, #1
    mov x1, x20
    mov x2, x19
    svc #0

rw_newline:
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret


.section .rodata
newline_char:
    .byte 0x0A

.bss
.align 4
input_buf:
    .zero 65536
