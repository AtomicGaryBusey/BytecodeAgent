.globl _start

.bss
    .align 16
input_buf:
    .zero 65536

    .align 16
tape:
    .zero 30000

    .align 4
jump_table:
    .zero 262144

    .align 4
bracket_stack:
    .zero 4000

    .align 16
output_buf:
    .zero 65536

.text

_start:
    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buf]
    xor rcx, rcx

find_bang:
    cmp rcx, r12
    jge no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov r13, rcx
    lea r14, [rsi + rcx + 1]
    mov r15, r12
    sub r15, rcx
    dec r15
    jmp bracket_setup

no_bang:
    mov r13, r12
    lea r14, [rsi + r12]
    xor r15, r15

bracket_setup:
    xor rbx, rbx
    xor rcx, rcx
    lea rdi, [rip + jump_table]
    lea r8, [rip + bracket_stack]

bracket_check:
    cmp rcx, r13
    jge bracket_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je do_open
    cmp al, 0x5d
    je do_close
    inc rcx
    jmp bracket_check

do_open:
    mov dword ptr [r8 + rbx*4], ecx
    inc rbx
    inc rcx
    jmp bracket_check

do_close:
    test rbx, rbx
    jz bracket_error
    dec rbx
    mov eax, dword ptr [r8 + rbx*4]
    lea rdx, [rcx + 1]
    mov dword ptr [rdi + rax*4], edx
    lea rdx, [rax + 1]
    mov dword ptr [rdi + rcx*4], edx
    inc rcx
    jmp bracket_check

bracket_done:
    test rbx, rbx
    jnz bracket_error
    jmp execute

bracket_error:
    mov eax, 60
    mov edi, 1
    syscall

execute:
    xor rbx, rbx
    xor r12, r12
    lea r8, [rip + input_buf]
    lea r9, [rip + tape]
    xor r10, r10

exec_loop:
    cmp rbx, r13
    jge exec_done
    movzx eax, byte ptr [r8 + rbx]
    cmp al, 0x3e
    je cmd_gt
    cmp al, 0x3c
    je cmd_lt
    cmp al, 0x2b
    je cmd_plus
    cmp al, 0x2d
    je cmd_minus
    cmp al, 0x2e
    je cmd_dot
    cmp al, 0x2c
    je cmd_comma
    cmp al, 0x5b
    je cmd_lbracket
    cmp al, 0x5d
    je cmd_rbracket
    inc rbx
    jmp exec_loop

cmd_gt:
    inc r12
    inc rbx
    jmp exec_loop

cmd_lt:
    dec r12
    inc rbx
    jmp exec_loop

cmd_plus:
    inc byte ptr [r9 + r12]
    inc rbx
    jmp exec_loop

cmd_minus:
    dec byte ptr [r9 + r12]
    inc rbx
    jmp exec_loop

cmd_dot:
    movzx eax, byte ptr [r9 + r12]
    lea rdx, [rip + output_buf]
    mov byte ptr [rdx + r10], al
    inc r10
    cmp r10, 65536
    jl dot_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, 65536
    syscall
    xor r10, r10
dot_done:
    inc rbx
    jmp exec_loop

cmd_comma:
    test r15, r15
    jz comma_zero
    movzx eax, byte ptr [r14]
    mov byte ptr [r9 + r12], al
    inc r14
    dec r15
    inc rbx
    jmp exec_loop
comma_zero:
    mov byte ptr [r9 + r12], 0
    inc rbx
    jmp exec_loop

cmd_lbracket:
    movzx eax, byte ptr [r9 + r12]
    test al, al
    jnz lbracket_nonzero
    lea rax, [rip + jump_table]
    mov ebx, dword ptr [rax + rbx*4]
    jmp exec_loop
lbracket_nonzero:
    inc rbx
    jmp exec_loop

cmd_rbracket:
    movzx eax, byte ptr [r9 + r12]
    test al, al
    jz rbracket_zero
    lea rax, [rip + jump_table]
    mov ebx, dword ptr [rax + rbx*4]
    jmp exec_loop
rbracket_zero:
    inc rbx
    jmp exec_loop

exec_done:
    test r10, r10
    jz exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r10
    syscall
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
