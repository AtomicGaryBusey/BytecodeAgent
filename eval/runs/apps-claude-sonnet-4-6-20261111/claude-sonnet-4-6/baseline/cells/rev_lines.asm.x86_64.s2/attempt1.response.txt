.globl _start

.bss
.align 16
input_buf:
    .zero 65536
output_buf:
    .zero 65538

.text
_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13

process_lines:
    cmp r13, r12
    jge all_done
    mov r14, r13

find_newline:
    cmp r13, r12
    jge found_end
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r13]
    cmp cl, 0x0A
    je found_newline
    inc r13
    jmp find_newline

found_newline:
    mov r15, r13
    inc r13
    jmp output_line

found_end:
    mov r15, r12

output_line:
    mov rax, r15
    sub rax, r14
    lea rdi, [rip + output_buf]
    test rax, rax
    jz just_newline
    lea rsi, [rip + input_buf]
    add rsi, r15
    dec rsi
    xor rbx, rbx
    mov rcx, rax

reverse_loop:
    movzx edx, byte ptr [rsi]
    mov byte ptr [rdi + rbx], dl
    dec rsi
    inc rbx
    dec rcx
    jnz reverse_loop

    mov byte ptr [rdi + rax], 0x0A
    inc rax
    jmp write_line

just_newline:
    mov byte ptr [rdi], 0x0A
    mov rax, 1

write_line:
    mov rdx, rax
    mov rsi, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp process_lines

all_done:
    mov rax, 60
    xor rdi, rdi
    syscall
