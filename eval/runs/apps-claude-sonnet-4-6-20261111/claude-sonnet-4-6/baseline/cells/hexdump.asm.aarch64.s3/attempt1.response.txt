.text
.globl _start

_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0

read_loop:
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    svc     #0
    cbz     x0, read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    mov     x21, #0
    cbz     x20, print_final

row_loop:
    cmp     x21, x20
    bge     print_final

    sub     x22, x20, x21
    cmp     x22, #16
    mov     x0, #16
    csel    x22, x22, x0, lo

    adrp    x24, out_buf
    add     x24, x24, :lo12:out_buf
    mov     x25, x24

    mov     x0, x21
    mov     x1, x25
    bl      write_hex8
    add     x25, x25, #8

    mov     w0, #0x20
    strb    w0, [x25], #1
    strb    w0, [x25], #1

    mov     x26, #0

hex_loop:
    cmp     x26, #16
    bge     hex_loop_done
    cmp     x26, #8
    bne     1f
    mov     w0, #0x20
    strb    w0, [x25], #1
1:
    cmp     x26, x22
    blt     2f
    mov     w0, #0x20
    strb    w0, [x25], #1
    strb    w0, [x25], #1
    strb    w0, [x25], #1
    b       3f
2:
    add     x0, x19, x21
    add     x0, x0, x26
    ldrb    w0, [x0]
    mov     x1, x25
    bl      write_hex2
    add     x25, x25, #2
    mov     w0, #0x20
    strb    w0, [x25], #1
3:
    add     x26, x26, #1
    b       hex_loop

hex_loop_done:
    mov     w0, #0x20
    strb    w0, [x25], #1
    mov     w0, #0x7c
    strb    w0, [x25], #1

    mov     x26, #0

ascii_loop:
    cmp     x26, x22
    bge     ascii_done
    add     x0, x19, x21
    add     x0, x0, x26
    ldrb    w0, [x0]
    cmp     w0, #0x20
    blo     1f
    cmp     w0, #0x7e
    bhi     1f
    b       2f
1:
    mov     w0, #0x2e
2:
    strb    w0, [x25], #1
    add     x26, x26, #1
    b       ascii_loop

ascii_done:
    mov     w0, #0x7c
    strb    w0, [x25], #1
    mov     w0, #0x0a
    strb    w0, [x25], #1

    mov     x8, #64
    mov     x0, #1
    mov     x1, x24
    sub     x2, x25, x24
    svc     #0

    add     x21, x21, x22
    b       row_loop

print_final:
    adrp    x24, out_buf
    add     x24, x24, :lo12:out_buf
    mov     x25, x24

    mov     x0, x20
    mov     x1, x25
    bl      write_hex8
    add     x25, x25, #8

    mov     w0, #0x0a
    strb    w0, [x25], #1

    mov     x8, #64
    mov     x0, #1
    mov     x1, x24
    sub     x2, x25, x24
    svc     #0

    mov     x8, #93
    mov     x0, #0
    svc     #0

write_hex8:
    adrp    x2, hex_chars
    add     x2, x2, :lo12:hex_chars
    mov     x3, x0
    mov     x4, #8
1:
    sub     x5, x4, #1
    lsl     x5, x5, #2
    lsr     x0, x3, x5
    and     x0, x0, #0xf
    ldrb    w0, [x2, x0]
    strb    w0, [x1], #1
    subs    x4, x4, #1
    bne     1b
    ret

write_hex2:
    adrp    x2, hex_chars
    add     x2, x2, :lo12:hex_chars
    lsr     x3, x0, #4
    and     x3, x3, #0xf
    ldrb    w3, [x2, x3]
    strb    w3, [x1], #1
    and     x3, x0, #0xf
    ldrb    w3, [x2, x3]
    strb    w3, [x1], #1
    ret

.section .rodata
hex_chars:
    .ascii  "0123456789abcdef"

.bss
.balign 4
input_buf:
    .zero   65536
out_buf:
    .zero   256
