.globl _start

.bss
.align 4
input_buf:
    .zero 65536
.align 4
output_buf:
    .zero 65538

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_all:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, done_reading
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    svc #0
    cbz x0, done_reading
    add x20, x20, x0
    b read_all

done_reading:
    mov x21, #0

next_line:
    cmp x21, x20
    bge done

    mov x22, x21

scan:
    cmp x22, x20
    bge no_newline
    ldrb w0, [x19, x22]
    cmp w0, #0x0a
    beq got_newline
    add x22, x22, #1
    b scan

got_newline:
    mov x23, x21
    add x21, x22, #1
    b print_line

no_newline:
    mov x23, x21
    mov x22, x20
    mov x21, x20

print_line:
    sub x24, x22, x23
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    cbz x24, empty_line

    sub x26, x22, #1
    mov x27, #0

rev_copy:
    ldrb w0, [x19, x26]
    strb w0, [x25, x27]
    add x27, x27, #1
    cmp x26, x23
    beq append_nl
    sub x26, x26, #1
    b rev_copy

empty_line:
    mov x27, #0

append_nl:
    mov w0, #0x0a
    strb w0, [x25, x27]
    add x27, x27, #1

    mov x8, #64
    mov x0, #1
    mov x1, x25
    mov x2, x27
    svc #0

    b next_line

done:
    mov x8, #93
    mov x0, #0
    svc #0
