.globl _start

.bss
.align 16
buffer: .zero 65536
out_buf: .zero 32

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall

    test rax, rax
    jle print_output

    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

byte_loop:
    test rcx, rcx
    jz read_loop

    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx

    cmp al, 0x0A
    je do_newline

    cmp al, 0x20
    je do_ws
    cmp al, 0x09
    jb do_word
    cmp al, 0x0D
    jbe do_ws

do_word:
    test r15, r15
    jnz byte_loop
    inc r13
    mov r15, 1
    jmp byte_loop

do_newline:
    inc r12

do_ws:
    mov r15, 0
    jmp byte_loop

print_output:
    lea rdi, [rip + out_buf]

    mov rax, r12
    call fmt_uint

    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, r13
    call fmt_uint

    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, r14
    call fmt_uint

    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + out_buf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    syscall

    mov eax, 60
    xor edi, edi
    syscall

fmt_uint:
    push rbx
    push r9

    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 0x30
    inc rdi
    pop r9
    pop rbx
    ret

1:
    xor rbx, rbx
    mov r9, 10

2:
    test rax, rax
    jz 3f
    xor edx, edx
    div r9
    push rdx
    inc rbx
    jmp 2b

3:
    test rbx, rbx
    jz 4f
    pop rax
    add al, 0x30
    mov byte ptr [rdi], al
    inc rdi
    dec rbx
    jmp 3b

4:
    pop r9
    pop rbx
    ret
