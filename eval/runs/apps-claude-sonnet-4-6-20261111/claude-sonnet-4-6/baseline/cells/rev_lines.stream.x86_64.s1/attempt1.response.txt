.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    jmp read_loop

read_done:
    lea r13, [rip + input_buf]
    mov r14, r13
    lea r15, [rip + input_buf]
    add r15, r12

process_loop:
    cmp r13, r15
    jge check_final
    movzx eax, byte ptr [r13]
    cmp al, 0x0A
    je emit_line
    inc r13
    jmp process_loop

emit_line:
    mov rbx, r13
    dec rbx
    mov rcx, r14

reverse_loop:
    cmp rcx, rbx
    jge do_write
    movzx eax, byte ptr [rcx]
    movzx edx, byte ptr [rbx]
    mov byte ptr [rcx], dl
    mov byte ptr [rbx], al
    inc rcx
    dec rbx
    jmp reverse_loop

do_write:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r13
    sub rdx, r14
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall
    inc r13
    mov r14, r13
    jmp process_loop

check_final:
    cmp r14, r15
    jge exit_prog
    mov rbx, r15
    dec rbx
    mov rcx, r14

reverse_final_loop:
    cmp rcx, rbx
    jge write_final
    movzx eax, byte ptr [rcx]
    movzx edx, byte ptr [rbx]
    mov byte ptr [rcx], dl
    mov byte ptr [rbx], al
    inc rcx
    dec rbx
    jmp reverse_final_loop

write_final:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    sub rdx, r14
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline_byte:
    .byte 0x0A
