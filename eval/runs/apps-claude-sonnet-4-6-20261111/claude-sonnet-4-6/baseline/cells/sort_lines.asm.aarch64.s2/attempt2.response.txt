.globl _start

.section .rodata
newline_char:
    .byte 0x0A

.bss
.align 3
input_buf:
    .zero 32768
.align 3
line_ptrs:
    .zero 8000
.align 2
line_lens:
    .zero 4000

.text

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x21, x19
    mov x2, #32768
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, all_done

    adrp x22, line_ptrs
    add x22, x22, :lo12:line_ptrs
    adrp x23, line_lens
    add x23, x23, :lo12:line_lens
    mov x20, #0
    mov x24, #0
    mov x25, x21

parse_loop:
    cmp x24, x19
    b.ge parse_eof
    ldrb w0, [x21, x24]
    add x24, x24, #1
    cmp w0, #0x0A
    b.ne parse_loop
    str x25, [x22, x20, lsl #3]
    sub x0, x21, x25
    add x0, x0, x24
    sub x0, x0, #1
    str w0, [x23, x20, lsl #2]
    add x20, x20, #1
    add x25, x21, x24
    b parse_loop

parse_eof:
    add x0, x21, x19
    cmp x25, x0
    b.ge parse_done
    str x25, [x22, x20, lsl #3]
    sub x0, x0, x25
    str w0, [x23, x20, lsl #2]
    add x20, x20, #1

parse_done:
    cmp x20, #1
    b.le output_phase
    mov x24, #1

isort_outer:
    cmp x24, x20
    b.ge output_phase
    ldr x25, [x22, x24, lsl #3]
    ldr w26, [x23, x24, lsl #2]
    sub x27, x24, #1

isort_inner:
    cmp x27, #0
    b.lt isort_place
    ldr x0, [x22, x27, lsl #3]
    ldr w1, [x23, x27, lsl #2]
    cmp w1, w26
    csel w2, w1, w26, lo
    mov x3, #0

cmp_loop:
    cmp x3, x2
    b.ge cmp_lens
    ldrb w4, [x0, x3]
    ldrb w5, [x25, x3]
    cmp w4, w5
    b.hi isort_shift
    b.lo isort_place
    add x3, x3, #1
    b cmp_loop

cmp_lens:
    cmp w1, w26
    b.hi isort_shift
    b isort_place

isort_shift:
    add x28, x27, #1
    str x0, [x22, x28, lsl #3]
    ldr w0, [x23, x27, lsl #2]
    str w0, [x23, x28, lsl #2]
    sub x27, x27, #1
    b isort_inner

isort_place:
    add x28, x27, #1
    str x25, [x22, x28, lsl #3]
    str w26, [x23, x28, lsl #2]
    add x24, x24, #1
    b isort_outer

output_phase:
    mov x24, #0

output_loop:
    cmp x24, x20
    b.ge all_done
    ldr x1, [x22, x24, lsl #3]
    ldr w2, [x23, x24, lsl #2]
    cbz w2, output_nl
    mov x0, #1
    mov x8, #64
    svc #0

output_nl:
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    add x24, x24, #1
    b output_loop

all_done:
    mov x0, #0
    mov x8, #93
    svc #0
