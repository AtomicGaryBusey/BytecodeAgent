.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .zero 65536
.align 4
line_buf:
    .zero 128

.text
.globl _start

write_hex8:
    adrp x2, hex_chars
    add x2, x2, :lo12:hex_chars
    mov x3, #28
write_hex8_loop:
    lsr x4, x0, x3
    and x4, x4, #0xf
    ldrb w5, [x2, x4]
    strb w5, [x1], #1
    cbz x3, write_hex8_done
    sub x3, x3, #4
    b write_hex8_loop
write_hex8_done:
    ret

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    cmp x19, #65536
    blo read_loop

read_done:
    mov x21, #0
    cbz x19, print_final

row_loop:
    sub x22, x19, x21
    cmp x22, #16
    bls row_got_size
    mov x22, #16

row_got_size:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf

    mov x0, x21
    mov x1, x23
    bl write_hex8
    add x23, x23, #8

    mov w0, #0x20
    strb w0, [x23], #1
    strb w0, [x23], #1

    adrp x25, hex_chars
    add x25, x25, :lo12:hex_chars

    mov x24, #0

hex_loop:
    cmp x24, #16
    bge hex_done

    cmp x24, #8
    bne hex_no_extra
    mov w2, #0x20
    strb w2, [x23], #1
hex_no_extra:

    cmp x24, x22
    bge hex_pad

    add x0, x20, x21
    add x0, x0, x24
    ldrb w0, [x0]

    lsr w1, w0, #4
    ldrb w2, [x25, x1]
    strb w2, [x23], #1

    and w1, w0, #0xf
    ldrb w2, [x25, x1]
    strb w2, [x23], #1

    mov w2, #0x20
    strb w2, [x23], #1
    b hex_next

hex_pad:
    mov w2, #0x20
    strb w2, [x23], #1
    strb w2, [x23], #1
    strb w2, [x23], #1

hex_next:
    add x24, x24, #1
    b hex_loop

hex_done:
    mov w0, #0x20
    strb w0, [x23], #1
    mov w0, #0x7c
    strb w0, [x23], #1

    mov x24, #0

ascii_loop:
    cmp x24, x22
    bge ascii_done

    add x0, x20, x21
    add x0, x0, x24
    ldrb w0, [x0]

    cmp w0, #0x20
    blt ascii_dot
    cmp w0, #0x7e
    bgt ascii_dot
    strb w0, [x23], #1
    b ascii_next

ascii_dot:
    mov w0, #0x2e
    strb w0, [x23], #1

ascii_next:
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w0, #0x7c
    strb w0, [x23], #1
    mov w0, #0x0a
    strb w0, [x23], #1

    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    sub x2, x23, x1
    mov x0, #1
    mov x8, #64
    svc #0

    add x21, x21, x22
    cmp x21, x19
    blt row_loop

print_final:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf

    mov x0, x19
    mov x1, x23
    bl write_hex8
    add x23, x23, #8

    mov w0, #0x0a
    strb w0, [x23], #1

    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    sub x2, x23, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
