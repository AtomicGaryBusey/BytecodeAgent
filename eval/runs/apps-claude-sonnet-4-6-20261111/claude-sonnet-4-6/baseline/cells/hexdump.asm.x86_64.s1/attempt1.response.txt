.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
line_buf:
    .zero 128

.text

_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 65536
    jl read_loop

read_done:
    xor r13d, r13d

row_loop:
    cmp r13d, r12d
    jge final_line

    mov r9d, r12d
    sub r9d, r13d
    cmp r9d, 16
    jle 1f
    mov r9d, 16
1:
    lea rbx, [rip + line_buf]
    mov r10d, r13d
    call emit_hex8

    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx

    xor r14d, r14d

hex_loop:
    cmp r14d, 16
    jge hex_loop_done

    cmp r14d, 8
    jne 2f
    mov byte ptr [rbx], 0x20
    inc rbx
2:
    cmp r14d, r9d
    jge 3f

    lea rax, [rip + input_buf]
    add rax, r13
    movzx eax, byte ptr [rax + r14]
    lea rdx, [rip + hex_chars]
    mov ecx, eax
    shr ecx, 4
    movzx ecx, byte ptr [rdx + rcx]
    mov byte ptr [rbx], cl
    inc rbx
    and eax, 0xf
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [rbx], al
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    jmp 4f
3:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
4:
    inc r14d
    jmp hex_loop

hex_loop_done:
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x7c
    inc rbx

    xor r14d, r14d

ascii_loop:
    cmp r14d, r9d
    jge ascii_done

    lea rax, [rip + input_buf]
    add rax, r13
    movzx eax, byte ptr [rax + r14]
    cmp eax, 0x20
    jb 5f
    cmp eax, 0x7e
    ja 5f
    mov byte ptr [rbx], al
    jmp 6f
5:
    mov byte ptr [rbx], 0x2e
6:
    inc rbx
    inc r14d
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx], 0x7c
    inc rbx
    mov byte ptr [rbx], 0x0a
    inc rbx

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov rdx, rbx
    sub rdx, rsi
    syscall

    add r13d, r9d
    jmp row_loop

final_line:
    lea rbx, [rip + line_buf]
    mov r10d, r12d
    call emit_hex8
    mov byte ptr [rbx], 0x0a
    inc rbx

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov rdx, rbx
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    lea rdx, [rip + hex_chars]
    mov eax, r10d
    mov ecx, 8
1:
    mov r8d, eax
    shr r8d, 28
    movzx r8d, byte ptr [rdx + r8]
    mov byte ptr [rbx], r8b
    inc rbx
    rol eax, 4
    dec ecx
    jnz 1b
    ret
