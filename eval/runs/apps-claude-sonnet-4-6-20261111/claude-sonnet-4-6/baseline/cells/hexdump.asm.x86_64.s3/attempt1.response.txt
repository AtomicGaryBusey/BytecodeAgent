.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.section .bss
input_buf:
    .zero 65536
line_buf:
    .zero 80

.text

# Convert eax to 8 lowercase hex digits at [rdi], advance rdi by 8.
# r8 must point to hex_chars. Clobbers eax, ecx, edx.
write_hex8:
    mov ecx, 28
wh8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jge wh8_loop
    ret

# Convert al to 2 lowercase hex digits at [rdi], advance rdi by 2.
# r8 must point to hex_chars. Clobbers eax, edx.
write_hex2:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    ret

_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d

read_all:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub edx, r13d
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r13d, eax
    jmp read_all

read_done:
    lea r8, [rip + hex_chars]
    xor r14d, r14d

row_loop:
    cmp r14d, r13d
    jge final_line

    mov ebx, r13d
    sub ebx, r14d
    cmp ebx, 16
    jle row_n_set
    mov ebx, 16
row_n_set:

    lea r15, [r12 + r14]
    lea rdi, [rip + line_buf]

    mov eax, r14d
    call write_hex8

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    add rdi, 2

    xor ecx, ecx

hex_loop:
    cmp ecx, 16
    jge hex_done

    cmp ecx, 8
    jne hex_no_extra
    mov byte ptr [rdi], ' '
    inc rdi
hex_no_extra:

    cmp ecx, ebx
    jge hex_pad

    movzx eax, byte ptr [r15 + rcx]
    call write_hex2
    mov byte ptr [rdi], ' '
    inc rdi
    jmp hex_next

hex_pad:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3

hex_next:
    inc ecx
    jmp hex_loop

hex_done:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], '|'
    add rdi, 2

    xor ecx, ecx

ascii_loop:
    cmp ecx, ebx
    jge ascii_done

    movzx eax, byte ptr [r15 + rcx]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7e
    ja ascii_dot
    jmp ascii_char
ascii_dot:
    mov al, '.'
ascii_char:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi+1], 10
    add rdi, 2

    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r14d, ebx
    jmp row_loop

final_line:
    lea rdi, [rip + line_buf]

    mov eax, r13d
    call write_hex8

    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
