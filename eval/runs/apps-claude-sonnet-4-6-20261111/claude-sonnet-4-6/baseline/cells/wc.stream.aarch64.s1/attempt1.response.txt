.globl _start

.bss
.align 4
read_buf:
    .zero 4096

.align 4
out_buf:
    .zero 64

.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, read_buf
    add x1, x1, :lo12:read_buf
    mov x2, #4096
    svc #0

    cbz x0, read_done

    add x21, x21, x0

    adrp x23, read_buf
    add x23, x23, :lo12:read_buf
    mov x24, x0

process_loop:
    cbz x24, read_loop
    sub x24, x24, #1
    ldrb w0, [x23], #1

    cmp w0, #0x0A
    b.eq byte_newline

    cmp w0, #0x20
    b.eq byte_white

    cmp w0, #0x09
    b.lt byte_nonwhite
    cmp w0, #0x0D
    b.le byte_white

byte_nonwhite:
    cbnz x22, process_loop
    mov x22, #1
    add x20, x20, #1
    b process_loop

byte_newline:
    add x19, x19, #1

byte_white:
    mov x22, #0
    b process_loop

read_done:
    adrp x25, out_buf
    add x25, x25, :lo12:out_buf
    mov x0, x25

    mov x1, x19
    bl write_num

    mov w2, #0x20
    strb w2, [x0], #1

    mov x1, x20
    bl write_num

    mov w2, #0x20
    strb w2, [x0], #1

    mov x1, x21
    bl write_num

    mov w2, #0x0A
    strb w2, [x0], #1

    sub x2, x0, x25

    mov x8, #64
    mov x0, #1
    mov x1, x25
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0

write_num:
    stp x29, x30, [sp, #-48]!
    mov x29, sp

    cbnz x1, 1f
    mov w2, #0x30
    strb w2, [x0], #1
    ldp x29, x30, [sp], #48
    ret

1:
    add x3, sp, #16
    mov x4, #0

2:
    cbz x1, 3f
    mov x5, #10
    udiv x2, x1, x5
    msub x1, x2, x5, x1
    add x1, x1, #0x30
    strb w1, [x3, x4]
    add x4, x4, #1
    mov x1, x2
    b 2b

3:
    mov x5, x4
4:
    cbz x5, 5f
    sub x5, x5, #1
    ldrb w2, [x3, x5]
    strb w2, [x0], #1
    b 4b

5:
    ldp x29, x30, [sp], #48
    ret
