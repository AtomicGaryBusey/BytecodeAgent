.globl _start

.section .bss
.align 4
buf:    .zero 4096

.text

// num_to_str: convert unsigned integer in x1 to decimal at [x0]
// x0 is updated to point past the last written character
// clobbers x2, x3, x4, x5; saves x29, x30
num_to_str:
    stp x29, x30, [sp, #-48]!
    mov x29, sp

    cmp x1, #0
    bne nts_nonzero
    mov w2, #0x30
    strb w2, [x0], #1
    ldp x29, x30, [sp], #48
    ret

nts_nonzero:
    add x4, sp, #16
    mov x3, #0

nts_digit_loop:
    cbz x1, nts_reverse
    mov x2, #10
    udiv x5, x1, x2
    msub x2, x5, x2, x1
    add w2, w2, #0x30
    strb w2, [x4, x3]
    add x3, x3, #1
    mov x1, x5
    b nts_digit_loop

nts_reverse:
    sub x3, x3, #1
nts_rev_loop:
    cmp x3, #0
    blt nts_rev_done
    ldrb w2, [x4, x3]
    strb w2, [x0], #1
    sub x3, x3, #1
    b nts_rev_loop

nts_rev_done:
    ldp x29, x30, [sp], #48
    ret

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #4096
    svc #0

    cbz x0, done_reading

    mov x23, x0
    add x21, x21, x23

    adrp x24, buf
    add x24, x24, :lo12:buf
    mov x25, #0

process_loop:
    cmp x25, x23
    beq read_loop

    ldrb w26, [x24, x25]
    add x25, x25, #1

    cmp w26, #0x0A
    bne check_whitespace
    add x19, x19, #1

check_whitespace:
    cmp w26, #0x20
    beq set_not_in_word
    cmp w26, #0x09
    blt is_word_char
    cmp w26, #0x0D
    ble set_not_in_word
    b is_word_char

set_not_in_word:
    mov x22, #0
    b process_loop

is_word_char:
    cmp x22, #0
    bne process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop

done_reading:
    sub sp, sp, #64
    mov x27, sp

    mov x0, x27
    mov x1, x19
    bl num_to_str

    mov w2, #0x20
    strb w2, [x0], #1

    mov x1, x20
    bl num_to_str

    mov w2, #0x20
    strb w2, [x0], #1

    mov x1, x21
    bl num_to_str

    mov w2, #0x0A
    strb w2, [x0], #1

    sub x2, x0, x27
    mov x8, #64
    mov x0, #1
    mov x1, x27
    svc #0

    add sp, sp, #64

    mov x8, #93
    mov x0, #0
    svc #0
