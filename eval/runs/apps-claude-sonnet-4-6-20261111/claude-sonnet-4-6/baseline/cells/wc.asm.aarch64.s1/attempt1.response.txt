.globl _start

.bss
.balign 8
buf:
    .zero 4096
outbuf:
    .zero 64
tmpbuf:
    .zero 32

.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, done_reading
    mov x3, x0
    adrp x4, buf
    add x4, x4, :lo12:buf

process_loop:
    cbz x3, read_loop
    ldrb w5, [x4], #1
    sub x3, x3, #1
    add x21, x21, #1
    cmp w5, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    cmp w5, #0x20
    b.eq 2f
    cmp w5, #0x09
    b.lo 3f
    cmp w5, #0x0D
    b.ls 2f
3:
    cbnz x22, process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop
2:
    mov x22, #0
    b process_loop

done_reading:
    adrp x0, outbuf
    add x0, x0, :lo12:outbuf
    mov x23, x0
    mov x1, x19
    bl num_to_str
    mov w2, #0x20
    strb w2, [x0], #1
    mov x1, x20
    bl num_to_str
    mov w2, #0x20
    strb w2, [x0], #1
    mov x1, x21
    bl num_to_str
    mov w2, #0x0A
    strb w2, [x0], #1
    sub x2, x0, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

num_to_str:
    cbnz x1, 1f
    mov w2, #0x30
    strb w2, [x0], #1
    ret
1:
    adrp x3, tmpbuf
    add x3, x3, :lo12:tmpbuf
    mov x4, x3
2:
    cbz x1, 3f
    mov x5, #10
    udiv x2, x1, x5
    msub x6, x2, x5, x1
    add w6, w6, #0x30
    strb w6, [x3], #1
    mov x1, x2
    b 2b
3:
    sub x3, x3, #1
4:
    ldrb w5, [x3], #-1
    strb w5, [x0], #1
    cmp x3, x4
    b.hs 4b
    ret
