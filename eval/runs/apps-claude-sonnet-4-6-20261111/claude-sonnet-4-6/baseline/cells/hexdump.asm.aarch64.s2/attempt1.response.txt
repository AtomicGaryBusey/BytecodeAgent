    .section .rodata
hex_chars:
    .ascii "0123456789abcdef"

    .bss
    .align 4
input_buf:
    .zero 65536
    .align 4
out_buf:
    .zero 128

    .text
    .globl _start

_start:
    mov x19, #0

.Lread_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, .Lread_done
    svc #0
    cbz x0, .Lread_done
    add x19, x19, x0
    b .Lread_loop

.Lread_done:
    mov x20, #0

.Lrow_loop:
    cmp x20, x19
    bge .Lprint_final

    sub x21, x19, x20
    cmp x21, #16
    mov x0, #16
    csel x21, x21, x0, lo

    adrp x22, out_buf
    add x22, x22, :lo12:out_buf
    mov x28, x22

    adrp x23, hex_chars
    add x23, x23, :lo12:hex_chars

    adrp x27, input_buf
    add x27, x27, :lo12:input_buf
    add x27, x27, x20

    mov x24, #28
.Loffhex:
    lsr x25, x20, x24
    and x25, x25, #0xf
    ldrb w25, [x23, x25]
    strb w25, [x22], #1
    cbz x24, .Loffhex_done
    sub x24, x24, #4
    b .Loffhex
.Loffhex_done:

    mov w25, #0x20
    strb w25, [x22], #1
    strb w25, [x22], #1

    mov x26, #0
.Lhex_loop:
    cmp x26, x21
    bge .Lhex_absent
    ldrb w24, [x27, x26]
    lsr x25, x24, #4
    ldrb w25, [x23, x25]
    strb w25, [x22], #1
    and x25, x24, #0xf
    ldrb w25, [x23, x25]
    strb w25, [x22], #1
    mov w25, #0x20
    strb w25, [x22], #1
    b .Lhex_next
.Lhex_absent:
    mov w25, #0x20
    strb w25, [x22], #1
    strb w25, [x22], #1
    strb w25, [x22], #1
.Lhex_next:
    cmp x26, #7
    bne .Lhex_no_extra
    mov w25, #0x20
    strb w25, [x22], #1
.Lhex_no_extra:
    add x26, x26, #1
    cmp x26, #16
    blt .Lhex_loop

    mov w25, #0x20
    strb w25, [x22], #1
    mov w25, #0x7c
    strb w25, [x22], #1

    mov x26, #0
.Lascii_loop:
    cmp x26, x21
    bge .Lascii_done
    ldrb w24, [x27, x26]
    cmp w24, #0x20
    blt .Lascii_dot
    cmp w24, #0x7e
    bgt .Lascii_dot
    b .Lascii_write
.Lascii_dot:
    mov w24, #0x2e
.Lascii_write:
    strb w24, [x22], #1
    add x26, x26, #1
    b .Lascii_loop
.Lascii_done:

    mov w25, #0x7c
    strb w25, [x22], #1
    mov w25, #0x0a
    strb w25, [x22], #1

    sub x2, x22, x28
    mov x8, #64
    mov x0, #1
    mov x1, x28
    svc #0

    add x20, x20, x21
    b .Lrow_loop

.Lprint_final:
    adrp x22, out_buf
    add x22, x22, :lo12:out_buf
    mov x28, x22

    adrp x23, hex_chars
    add x23, x23, :lo12:hex_chars

    mov x24, #28
.Lfinhex:
    lsr x25, x19, x24
    and x25, x25, #0xf
    ldrb w25, [x23, x25]
    strb w25, [x22], #1
    cbz x24, .Lfinhex_done
    sub x24, x24, #4
    b .Lfinhex
.Lfinhex_done:

    mov w25, #0x0a
    strb w25, [x22], #1

    sub x2, x22, x28
    mov x8, #64
    mov x0, #1
    mov x1, x28
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
