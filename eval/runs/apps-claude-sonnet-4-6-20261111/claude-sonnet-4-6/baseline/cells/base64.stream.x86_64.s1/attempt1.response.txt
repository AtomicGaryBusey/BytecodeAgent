.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
in_buf:
    .zero 49152
.align 16
out_buf:
    .zero 70000

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 49152
    sub edx, r12d
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 49152
    jl read_loop

read_done:
    test r12d, r12d
    jz do_exit

    lea rsi, [rip + in_buf]
    lea rdi, [rip + out_buf]
    lea rbx, [rip + b64_table]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

encode_loop:
    mov eax, r12d
    sub eax, r15d
    cmp eax, 3
    jl encode_tail

    movzx eax, byte ptr [rsi + r15]
    movzx ecx, byte ptr [rsi + r15 + 1]
    movzx edx, byte ptr [rsi + r15 + 2]
    add r15d, 3

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi + r13], r8b
    inc r13d

    and eax, 3
    shl eax, 4
    mov r9d, ecx
    shr r9d, 4
    or eax, r9d
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi + r13], al
    inc r13d

    and ecx, 0xf
    shl ecx, 2
    mov r9d, edx
    shr r9d, 6
    or ecx, r9d
    movzx ecx, byte ptr [rbx + rcx]
    mov byte ptr [rdi + r13], cl
    inc r13d

    and edx, 0x3f
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + r13], dl
    inc r13d

    inc r14d
    cmp r14d, 19
    jl encode_loop
    mov byte ptr [rdi + r13], 0x0a
    inc r13d
    xor r14d, r14d
    jmp encode_loop

encode_tail:
    test eax, eax
    jz encode_done
    cmp eax, 1
    je encode_one

    movzx eax, byte ptr [rsi + r15]
    movzx ecx, byte ptr [rsi + r15 + 1]

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi + r13], r8b
    inc r13d

    and eax, 3
    shl eax, 4
    mov r9d, ecx
    shr r9d, 4
    or eax, r9d
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi + r13], al
    inc r13d

    and ecx, 0xf
    shl ecx, 2
    movzx ecx, byte ptr [rbx + rcx]
    mov byte ptr [rdi + r13], cl
    inc r13d

    mov byte ptr [rdi + r13], 0x3d
    inc r13d
    inc r14d
    jmp encode_done

encode_one:
    movzx eax, byte ptr [rsi + r15]

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi + r13], r8b
    inc r13d

    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi + r13], al
    inc r13d

    mov byte ptr [rdi + r13], 0x3d
    inc r13d
    mov byte ptr [rdi + r13], 0x3d
    inc r13d
    inc r14d

encode_done:
    test r14d, r14d
    jz write_out
    mov byte ptr [rdi + r13], 0x0a
    inc r13d

write_out:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov edx, r13d
    syscall

do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
