.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536
.align 16
output_buf:
    .zero 65538

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle start_scan
    syscall
    test rax, rax
    jle start_scan
    add r12d, eax
    jmp read_more

start_scan:
    xor r13d, r13d

next_line:
    cmp r13d, r12d
    jge all_done

    mov r14d, r13d
    lea rbx, [rip + input_buf]

find_newline:
    cmp r14d, r12d
    jge do_emit
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x0A
    je do_emit
    inc r14d
    jmp find_newline

do_emit:
    lea rdi, [rip + output_buf]
    xor r15d, r15d

    cmp r14d, r13d
    je do_newline

    mov ecx, r14d
    dec ecx

copy_loop:
    movzx edx, byte ptr [rbx + rcx]
    mov byte ptr [rdi + r15], dl
    inc r15d
    cmp ecx, r13d
    je do_newline
    dec ecx
    jmp copy_loop

do_newline:
    mov byte ptr [rdi + r15], 0x0A
    inc r15d

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov edx, r15d
    syscall

    cmp r14d, r12d
    jge all_done
    mov r13d, r14d
    inc r13d
    jmp next_line

all_done:
    xor edi, edi
    mov eax, 60
    syscall
