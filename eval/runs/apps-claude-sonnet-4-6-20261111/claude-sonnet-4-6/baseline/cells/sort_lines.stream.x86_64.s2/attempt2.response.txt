.globl _start

.section .rodata
newline_byte:
    .byte 0x0A

.bss
.align 16
input_buf:
    .zero 32768
.align 8
line_ptrs:
    .zero 8000
line_lens:
    .zero 8000

.text

# compare_lines(rdi=ptr1, rsi=len1, rdx=ptr2, rcx=len2)
# returns rax: negative if ptr1<ptr2, 0 if equal, positive if ptr1>ptr2
compare_lines:
    push r8
    push r9
    push r10
    mov r8, rsi
    cmp r8, rcx
    jbe .Lcl_min_ok
    mov r8, rcx
.Lcl_min_ok:
    xor r9, r9
.Lcl_byte_loop:
    cmp r9, r8
    jge .Lcl_prefix
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    sub eax, r10d
    jne .Lcl_diff
    inc r9
    jmp .Lcl_byte_loop
.Lcl_prefix:
    mov rax, rsi
    sub rax, rcx
    jmp .Lcl_ret
.Lcl_diff:
    movsx rax, eax
.Lcl_ret:
    pop r10
    pop r9
    pop r8
    ret

_start:
    # Phase 1: read all input
    xor r12, r12
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:

    # Phase 2: parse lines
    # r12 = total bytes, r13 = pos, r14 = line count, r15 = line start
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + input_buf]

.Lparse_loop:
    cmp r13, r12
    jge .Lparse_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x0A
    jne .Lparse_not_nl
    lea rdi, [rip + line_ptrs]
    lea rax, [rbx + r15]
    mov qword ptr [rdi + r14*8], rax
    lea rdi, [rip + line_lens]
    mov rax, r13
    sub rax, r15
    mov qword ptr [rdi + r14*8], rax
    inc r14
    inc r13
    mov r15, r13
    jmp .Lparse_loop
.Lparse_not_nl:
    inc r13
    jmp .Lparse_loop

.Lparse_done:
    cmp r13, r15
    je .Lno_final_line
    lea rdi, [rip + line_ptrs]
    lea rax, [rbx + r15]
    mov qword ptr [rdi + r14*8], rax
    lea rdi, [rip + line_lens]
    mov rax, r13
    sub rax, r15
    mov qword ptr [rdi + r14*8], rax
    inc r14
.Lno_final_line:

    # Phase 3: insertion sort
    mov r12, r14
    mov r13, 1

.Lsort_outer:
    cmp r13, r12
    jge .Lsort_done
    lea r11, [rip + line_ptrs]
    mov r14, qword ptr [r11 + r13*8]
    lea r11, [rip + line_lens]
    mov r15, qword ptr [r11 + r13*8]
    mov rbx, r13
    dec rbx

.Lsort_inner:
    test rbx, rbx
    js .Lsort_insert
    lea r11, [rip + line_ptrs]
    mov rdi, qword ptr [r11 + rbx*8]
    lea r11, [rip + line_lens]
    mov rsi, qword ptr [r11 + rbx*8]
    mov rdx, r14
    mov rcx, r15
    call compare_lines
    test rax, rax
    jle .Lsort_insert
    lea r11, [rip + line_ptrs]
    mov rax, qword ptr [r11 + rbx*8]
    mov qword ptr [r11 + rbx*8 + 8], rax
    lea r11, [rip + line_lens]
    mov rax, qword ptr [r11 + rbx*8]
    mov qword ptr [r11 + rbx*8 + 8], rax
    dec rbx
    jmp .Lsort_inner

.Lsort_insert:
    lea r11, [rip + line_ptrs]
    mov qword ptr [r11 + rbx*8 + 8], r14
    lea r11, [rip + line_lens]
    mov qword ptr [r11 + rbx*8 + 8], r15
    inc r13
    jmp .Lsort_outer

.Lsort_done:
    xor r13, r13

.Loutput_loop:
    cmp r13, r12
    jge .Loutput_done
    lea r11, [rip + line_ptrs]
    mov rsi, qword ptr [r11 + r13*8]
    lea r11, [rip + line_lens]
    mov rdx, qword ptr [r11 + r13*8]
    test rdx, rdx
    jz .Loutput_just_nl
    mov eax, 1
    mov edi, 1
    syscall
.Loutput_just_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall
    inc r13
    jmp .Loutput_loop

.Loutput_done:
    mov eax, 60
    xor edi, edi
    syscall
