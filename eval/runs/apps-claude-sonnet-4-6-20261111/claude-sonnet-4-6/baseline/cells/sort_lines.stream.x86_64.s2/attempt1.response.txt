.globl _start

.section .rodata
newline_char:
    .byte 0x0A

.bss
.align 16
input_buf:
    .zero 32768
.align 8
line_ptrs:
    .zero 8000
line_lens:
    .zero 8000

.text

# compare_lines(rdi=ptr1, rsi=len1, rdx=ptr2, rcx=len2) -> rax
# negative if line1 < line2, 0 if equal, positive if line1 > line2
compare_lines:
    mov r8, rsi
    cmp r8, rcx
    jbe .Lcl_minok
    mov r8, rcx
.Lcl_minok:
    xor r9, r9
.Lcl_loop:
    cmp r9, r8
    jge .Lcl_prefix
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    cmp eax, r10d
    ja .Lcl_gt
    jb .Lcl_lt
    inc r9
    jmp .Lcl_loop
.Lcl_prefix:
    mov rax, rsi
    sub rax, rcx
    ret
.Lcl_gt:
    mov eax, 1
    ret
.Lcl_lt:
    mov rax, -1
    ret

_start:
    # Phase 1: read all input
    xor r12, r12
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:

    # Phase 2: parse lines
    lea rbx, [rip + input_buf]
    xor r13, r13
    xor r14, r14
    xor r15, r15

.Lparse_loop:
    cmp r13, r12
    jge .Lparse_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x0A
    jne .Lparse_inc
    lea r11, [rip + line_ptrs]
    lea rax, [rbx + r15]
    mov qword ptr [r11 + r14*8], rax
    lea r11, [rip + line_lens]
    mov rax, r13
    sub rax, r15
    mov qword ptr [r11 + r14*8], rax
    inc r14
    inc r13
    mov r15, r13
    jmp .Lparse_loop
.Lparse_inc:
    inc r13
    jmp .Lparse_loop

.Lparse_done:
    cmp r13, r15
    je .Lparse_no_final
    lea r11, [rip + line_ptrs]
    lea rax, [rbx + r15]
    mov qword ptr [r11 + r14*8], rax
    lea r11, [rip + line_lens]
    mov rax, r13
    sub rax, r15
    mov qword ptr [r11 + r14*8], rax
    inc r14
.Lparse_no_final:

    # Phase 3: insertion sort
    mov r12, r14
    mov r13, 1

.Lsort_outer:
    cmp r13, r12
    jge .Lsort_done
    lea r11, [rip + line_ptrs]
    mov r14, qword ptr [r11 + r13*8]
    lea r11, [rip + line_lens]
    mov r15, qword ptr [r11 + r13*8]
    mov rbx, r13
    dec rbx

.Lsort_inner:
    test rbx, rbx
    js .Lsort_insert
    lea r11, [rip + line_ptrs]
    mov rdi, qword ptr [r11 + rbx*8]
    lea r11, [rip + line_lens]
    mov rsi, qword ptr [r11 + rbx*8]
    mov rdx, r14
    mov rcx, r15
    call compare_lines
    test rax, rax
    jle .Lsort_insert
