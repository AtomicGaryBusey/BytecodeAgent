.globl _start

.section .rodata
newline_char:
    .byte 0x0A

.bss
.align 3
input_buf:
    .zero 32768
.align 3
line_ptrs:
    .zero 8000
line_lens:
    .zero 4000

.text
_start:
    mov x19, #0

read_loop:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #32768
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    mov x0, #0
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, done

    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, x20
    add x23, x20, x19
    mov x22, #0

parse_loop:
    cmp x20, x23
    bge parse_end
    ldrb w0, [x20], #1
    cmp w0, #0x0A
    bne parse_loop

    adrp x1, line_ptrs
    add x1, x1, :lo12:line_ptrs
    str x21, [x1, x22, lsl #3]
    sub x0, x20, x21
    sub x0, x0, #1
    adrp x1, line_lens
    add x1, x1, :lo12:line_lens
    str w0, [x1, x22, lsl #2]
    mov x21, x20
    add x22, x22, #1
    cmp x22, #1000
    blt parse_loop
    b sort_start

parse_end:
    cmp x21, x23
    bge sort_start
    adrp x1, line_ptrs
    add x1, x1, :lo12:line_ptrs
    str x21, [x1, x22, lsl #3]
    sub x0, x23, x21
    adrp x1, line_lens
    add x1, x1, :lo12:line_lens
    str w0, [x1, x22, lsl #2]
    add x22, x22, #1

sort_start:
    cbz x22, done
    mov x24, #1

sort_outer:
    cmp x24, x22
    bge output_start
    adrp x0, line_ptrs
    add x0, x0, :lo12:line_ptrs
    ldr x25, [x0, x24, lsl #3]
    adrp x0, line_lens
    add x0, x0, :lo12:line_lens
    ldr w26, [x0, x24, lsl #2]
    sub x27, x24, #1

sort_inner:
    tst x27, x27
    bmi insert_key
    adrp x0, line_ptrs
    add x0, x0, :lo12:line_ptrs
    ldr x1, [x0, x27, lsl #3]
    adrp x0, line_lens
    add x0, x0, :lo12:line_lens
    ldr w2, [x0, x27, lsl #2]
    cmp w2, w26
    csel w3, w2, w26, le
    mov w4, #0

byte_cmp:
    cmp w4, w3
    bge byte_cmp_done
    ldrb w5, [x1, x4]
    ldrb w6, [x25, x4]
    cmp w5, w6
    bhi do_shift
    blo insert_key
    add w4, w4, #1
    b byte_cmp

byte_cmp_done:
    cmp w2, w26
    bhi do_shift

insert_key:
    add x28, x27, #1
    adrp x0, line_ptrs
    add x0, x0, :lo12:line_ptrs
    str x25, [x0, x28, lsl #3]
    adrp x0, line_lens
    add x0, x0, :lo12:line_lens
    str w26, [x0, x28, lsl #2]
    add x24, x24, #1
    b sort_outer

do_shift:
    adrp x0, line_ptrs
    add x0, x0, :lo12:line_ptrs
    add x28, x27, #1
    ldr x1, [x0, x27, lsl #3]
    str x1, [x0, x28, lsl #3]
    adrp x0, line_lens
    add x0, x0, :lo12:line_lens
    ldr w1, [x0, x27, lsl #2]
    str w1, [x0, x28, lsl #2]
    sub x27, x27, #1
    b sort_inner

output_start:
    mov x24, #0

output_loop:
    cmp x24, x22
    bge done
    adrp x0, line_ptrs
    add x0, x0, :lo12:line_ptrs
    ldr x1, [x0, x24, lsl #3]
    adrp x0, line_lens
    add x0, x0, :lo12:line_lens
    ldr w2, [x0, x24, lsl #2]
    cbz w2, write_nl
    mov x8, #64
    mov x0, #1
    svc #0

write_nl:
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x2, #1
    mov x8, #64
    mov x0, #1
    svc #0
    add x24, x24, #1
    b output_loop

done:
    mov x8, #93
    mov x0, #0
    svc #0
