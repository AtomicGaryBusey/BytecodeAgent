.globl _start

.bss
.align 8
input_buf:
    .zero 65536
.align 8
tape:
    .zero 30000
.align 4
jump_table:
    .zero 262144
.align 4
bracket_stack:
    .zero 4000
.align 8
output_buf:
    .zero 65536

.text

_start:
    xor r12d, r12d

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buf]
    xor ecx, ecx

find_bang:
    cmp rcx, r12
    jge no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov r13, rcx
    lea r14, [rsi + rcx + 1]
    lea r15, [rsi + r12]
    jmp do_check

no_bang:
    mov r13, r12
    lea r14, [rsi + r12]
    mov r15, r14

do_check:
    lea rsi, [rip + input_buf]
    lea rdi, [rip + jump_table]
    lea r8, [rip + bracket_stack]
    xor ebx, ebx
    xor ecx, ecx

bracket_scan:
    cmp rcx, r13
    jge bracket_ok
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je b_open
    cmp al, 0x5d
    je b_close
    inc rcx
    jmp bracket_scan

b_open:
    cmp ebx, 1000
    jge error_exit
    mov dword ptr [r8 + rbx*4], ecx
    inc ebx
    inc rcx
    jmp bracket_scan

b_close:
    test ebx, ebx
    jz error_exit
    dec ebx
    mov eax, dword ptr [r8 + rbx*4]
    lea edx, [rcx + 1]
    mov dword ptr [rdi + rax*4], edx
    lea edx, [rax + 1]
    mov dword ptr [rdi + rcx*4], edx
    inc rcx
    jmp bracket_scan

bracket_ok:
    test ebx, ebx
    jnz error_exit

    lea rsi, [rip + input_buf]
    lea rdi, [rip + jump_table]
    lea rbx, [rip + tape]
    lea r8, [rip + output_buf]
    xor r10d, r10d
    xor r9d, r9d
    xor r11d, r11d

exec_loop:
    cmp r10, r13
    jge exec_done
    movzx eax, byte ptr [rsi + r10]
    cmp al, 0x3e
    je e_gt
    cmp al, 0x3c
    je e_lt
    cmp al, 0x2b
    je e_plus
    cmp al, 0x2d
    je e_minus
    cmp al, 0x2e
    je e_dot
    cmp al, 0x2c
    je e_comma
    cmp al, 0x5b
    je e_lbr
    cmp al, 0x5d
    je e_rbr
    inc r10
    jmp exec_loop

e_gt:
    inc r9
    inc r10
    jmp exec_loop

e_lt:
    dec r9
    inc r10
    jmp exec_loop

e_plus:
    add byte ptr [rbx + r9], 1
    inc r10
    jmp exec_loop

e_minus:
    sub byte ptr [rbx + r9], 1
    inc r10
    jmp exec_loop

e_dot:
    movzx eax, byte ptr [rbx + r9]
    mov byte ptr [r8 + r11], al
    inc r11
    inc r10
    jmp exec_loop

e_comma:
    cmp r14, r15
    jge e_comma_eof
    movzx eax, byte ptr [r14]
    inc r14
    mov byte ptr [rbx + r9], al
    inc r10
    jmp exec_loop

e_comma_eof:
    mov byte ptr [rbx + r9], 0
    inc r10
    jmp exec_loop

e_lbr:
    movzx eax, byte ptr [rbx + r9]
    test al, al
    jnz e_lbr_nz
    mov r10d, dword ptr [rdi + r10*4]
    jmp exec_loop

e_lbr_nz:
    inc r10
    jmp exec_loop

e_rbr:
    movzx eax, byte ptr [rbx + r9]
    test al, al
    jz e_rbr_z
    mov r10d, dword ptr [rdi + r10*4]
    jmp exec_loop

e_rbr_z:
    inc r10
    jmp exec_loop

exec_done:
    test r11d, r11d
    jz do_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r11
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

error_exit:
    mov eax, 60
    mov edi, 1
    syscall
