.globl _start

.section .bss
.align 4
buf:    .zero 65536
outbuf: .zero 64

.text

// format_number:
//   x0 = output buffer pointer (updated past written bytes)
//   x1 = unsigned number to format
// clobbers: x2-x7, x29, x30 (saved/restored)
format_number:
    stp x29, x30, [sp, #-32]!
    mov x29, sp

    cbnz x1, fn_nonzero
    mov w2, #0x30
    strb w2, [x0], #1
    ldp x29, x30, [sp], #32
    ret

fn_nonzero:
    add x3, sp, #16
    mov x4, #0

fn_loop:
    cbz x1, fn_done
    mov x5, #10
    udiv x6, x1, x5
    msub x7, x6, x5, x1
    add x7, x7, #0x30
    strb w7, [x3, x4]
    add x4, x4, #1
    mov x1, x6
    b fn_loop

fn_done:
    sub x5, x4, #1
fn_copy:
    ldrb w6, [x3, x5]
    strb w6, [x0], #1
    subs x5, x5, #1
    b.ge fn_copy

    ldp x29, x30, [sp], #32
    ret

_start:
    mov x19, #0    // lines
    mov x20, #0    // words
    mov x21, #0    // bytes
    mov x22, #0    // in_word

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    svc #0

    cbz x0, done_reading

    adrp x3, buf
    add x3, x3, :lo12:buf
    mov x4, x0

process_loop:
    cbz x4, read_loop
    ldrb w5, [x3], #1
    sub x4, x4, #1
    add x21, x21, #1

    cmp w5, #0x20
    b.eq ws_nonl
    cmp w5, #0x09
    b.lt not_ws
    cmp w5, #0x0D
    b.gt not_ws
    cmp w5, #0x0A
    b.eq is_newline
    b ws_nonl

is_newline:
    add x19, x19, #1
ws_nonl:
    mov x22, #0
    b process_loop

not_ws:
    cbnz x22, process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop

done_reading:
    adrp x0, outbuf
    add x0, x0, :lo12:outbuf

    mov x1, x19
    bl format_number

    mov w1, #0x20
    strb w1, [x0], #1

    mov x1, x20
    bl format_number

    mov w1, #0x20
    strb w1, [x0], #1

    mov x1, x21
    bl format_number

    mov w1, #0x0A
    strb w1, [x0], #1

    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    sub x2, x0, x2

    mov x8, #64
    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
