.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
line_buf:
    .zero 80

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 65536
    jl read_loop

read_done:
    xor r13d, r13d

row_loop:
    cmp r13d, r12d
    jge final_offset

    mov r14d, r12d
    sub r14d, r13d
    cmp r14d, 16
    jle 1f
    mov r14d, 16
1:
    lea rbp, [rip + input_buf]
    add rbp, r13
    lea rbx, [rip + line_buf]
    lea rdi, [rip + hex_table]
    mov eax, r13d
    mov ecx, 28

offset_digit_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xf
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    sub ecx, 4
    jge offset_digit_loop

    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    add rbx, 2
    xor r15d, r15d

hex_byte_loop:
    cmp r15d, 16
    jge hex_byte_done
    cmp r15d, 8
    jne 2f
    mov byte ptr [rbx], ' '
    inc rbx
2:
    cmp r15d, r14d
    jge 3f
    movzx eax, byte ptr [rbp + r15]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [rbx], dl
    and eax, 0xf
    movzx eax, byte ptr [rdi + rax]
    mov byte ptr [rbx + 1], al
    mov byte ptr [rbx + 2], ' '
    add rbx, 3
    jmp 4f
3:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    mov byte ptr [rbx + 2], ' '
    add rbx, 3
4:
    inc r15d
    jmp hex_byte_loop

hex_byte_done:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], '|'
    add rbx, 2
    xor r15d, r15d

ascii_loop:
    cmp r15d, r14d
    jge ascii_done
    movzx eax, byte ptr [rbp + r15]
    cmp al, 0x20
    jb 5f
    cmp al, 0x7e
    jbe 6f
5:
    mov al, '.'
6:
    mov byte ptr [rbx], al
    inc rbx
    inc r15d
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx], '|'
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2
    lea rsi, [rip + line_buf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r13d, r14d
    jmp row_loop

final_offset:
    lea rbx, [rip + line_buf]
    lea rdi, [rip + hex_table]
    mov eax, r12d
    mov ecx, 28

final_digit_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xf
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    sub ecx, 4
    jge final_digit_loop

    mov byte ptr [rbx], 0x0a
    inc rbx
    lea rsi, [rip + line_buf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
