.globl _start

.section .bss
.align 4
input_buf:
    .zero 65536
tape:
    .zero 30000
.align 2
bracket_table:
    .zero 262144
output_buf:
    .zero 65536
bracket_stack:
    .zero 4000

.text
_start:
    adrp x9, input_buf
    add  x9, x9, :lo12:input_buf
    mov  x10, #0

read_loop:
    mov  x0, #0
    add  x1, x9, x10
    mov  x2, #0x10000
    sub  x2, x2, x10
    cbz  x2, read_done
    mov  x8, #63
    svc  #0
    cbz  x0, read_done
    add  x10, x10, x0
    b    read_loop

read_done:
    mov  x11, #0

find_excl_loop:
    cmp  x11, x10
    beq  no_excl
    ldrb w12, [x9, x11]
    cmp  w12, #0x21
    beq  found_excl
    add  x11, x11, #1
    b    find_excl_loop

found_excl:
    mov  x13, x11
    add  x14, x9, x11
    add  x14, x14, #1
    sub  x15, x10, x11
    sub  x15, x15, #1
    b    do_bracket_check

no_excl:
    mov  x13, x10
    add  x14, x9, x10
    mov  x15, #0

do_bracket_check:
    adrp x16, bracket_table
    add  x16, x16, :lo12:bracket_table
    adrp x17, bracket_stack
    add  x17, x17, :lo12:bracket_stack
    mov  x18, #0
    mov  x11, #0

bscan_loop:
    cmp  x11, x13
    bge  bscan_done
    ldrb w12, [x9, x11]
    cmp  w12, #0x5b
    beq  bracket_open
    cmp  w12, #0x5d
    beq  bracket_close
    add  x11, x11, #1
    b    bscan_loop

bracket_open:
    cmp  x18, #1000
    bge  bracket_error
    str  w11, [x17, x18, lsl #2]
    add  x18, x18, #1
    add  x11, x11, #1
    b    bscan_loop

bracket_close:
    cbz  x18, bracket_error
    sub  x18, x18, #1
    ldr  w12, [x17, x18, lsl #2]
    add  w0, w11, #1
    str  w0, [x16, x12, lsl #2]
    add  w0, w12, #1
    str  w0, [x16, x11, lsl #2]
    add  x11, x11, #1
    b    bscan_loop

bscan_done:
    cbnz x18, bracket_error
    b    execute

bracket_error:
    mov  x0, #1
    mov  x8, #93
    svc  #0

execute:
    mov  x19, x9
    mov  x20, x13
    mov  x21, x14
    mov  x22, x15
    mov  x23, x16

    adrp x24, tape
    add  x24, x24, :lo12:tape
    adrp x25, output_buf
    add  x25, x25, :lo12:output_buf

    mov  x26, #0
    mov  x27, #0
    mov  x28, x24

exec_loop:
    cmp  x27, x20
    bge  exec_done

    ldrb w0, [x19, x27]
    add  x27, x27, #1

    cmp  w0, #0x3e
    beq  cmd_gt
    cmp  w0, #0x3c
    beq  cmd_lt
    cmp  w0, #0x2b
    beq  cmd_plus
    cmp  w0, #0x2d
    beq  cmd_minus
    cmp  w0, #0x2e
    beq  cmd_dot
    cmp  w0, #0x2c
    beq  cmd_comma
    cmp  w0, #0x5b
    beq  cmd_lbr
    cmp  w0, #0x5d
    beq  cmd_rbr
    b    exec_loop

cmd_gt:
    add  x28, x28, #1
    b    exec_loop

cmd_lt:
    sub  x28, x28, #1
    b    exec_loop

cmd_plus:
    ldrb w0, [x28]
    add  w0, w0, #1
    and  w0, w0, #0xff
    strb w0, [x28]
    b    exec_loop

cmd_minus:
    ldrb w0, [x28]
    sub  w0, w0, #1
    and  w0, w0, #0xff
    strb w0, [x28]
    b    exec_loop

cmd_dot:
    ldrb w0, [x28]
    strb w0, [x25, x26]
    add  x26, x26, #1
    b    exec_loop

cmd_comma:
    cbz  x22, cmd_comma_eof
    ldrb w0, [x21], #1
    sub  x22, x22, #1
    strb w0, [x28]
    b    exec_loop

cmd_comma_eof:
    strb wzr, [x28]
    b    exec_loop

cmd_lbr:
    ldrb w0, [x28]
    cbnz w0, exec_loop
    sub  x1, x27, #1
    ldr  w27, [x23, x1, lsl #2]
    b    exec_loop

cmd_rbr:
    ldrb w0, [x28]
    cbz  w0, exec_loop
    sub  x1, x27, #1
    ldr  w27, [x23, x1, lsl #2]
    b    exec_loop

exec_done:
    cbz  x26, do_exit0
    mov  x0, #1
    mov  x1, x25
    mov  x2, x26
    mov  x8, #64
    svc  #0

do_exit0:
    mov  x0, #0
    mov  x8, #93
    svc  #0
