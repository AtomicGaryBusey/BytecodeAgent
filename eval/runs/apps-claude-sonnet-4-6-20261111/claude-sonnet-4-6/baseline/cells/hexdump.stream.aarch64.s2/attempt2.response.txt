.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .zero 65536
.align 4
output_buf:
    .zero 393216

.text

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    adrp x28, hex_table
    add x28, x28, :lo12:hex_table
    mov x23, #0

row_loop:
    cmp x23, x19
    bge final_line

    sub x24, x19, x23
    cmp x24, #16
    ble skip_clamp
    mov x24, #16
skip_clamp:

    mov x26, x23
    mov x27, #28
offset_loop:
    lsr x0, x26, x27
    and x0, x0, #15
    ldrb w1, [x28, x0]
    strb w1, [x22], #1
    subs x27, x27, #4
    bge offset_loop

    mov w0, #32
    strb w0, [x22], #1
    strb w0, [x22], #1

    mov x25, #0
hex_loop:
    cmp x25, #16
    bge hex_done
    cmp x25, #8
    bne no_mid_space
    mov w0, #32
    strb w0, [x22], #1
no_mid_space:
    cmp x25, x24
    bge hex_blank
    add x26, x20, x23
    ldrb w0, [x26, x25]
    lsr w1, w0, #4
    ldrb w2, [x28, x1]
    strb w2, [x22], #1
    and w1, w0, #15
    ldrb w2, [x28, x1]
    strb w2, [x22], #1
    mov w0, #32
    strb w0, [x22], #1
    b hex_next
hex_blank:
    mov w0, #32
    strb w0, [x22], #1
    strb w0, [x22], #1
    strb w0, [x22], #1
hex_next:
    add x25, x25, #1
    b hex_loop
hex_done:

    mov w0, #32
    strb w0, [x22], #1
    mov w0, #0x7c
    strb w0, [x22], #1

    mov x25, #0
ascii_loop:
    cmp x25, x24
    bge ascii_done
    add x26, x20, x23
    ldrb w0, [x26, x25]
    cmp w0, #32
    blt ascii_dot
    cmp w0, #0x7e
    bgt ascii_dot
    strb w0, [x22], #1
    b ascii_next
ascii_dot:
    mov w0, #46
    strb w0, [x22], #1
ascii_next:
    add x25, x25, #1
    b ascii_loop
ascii_done:
    mov w0, #0x7c
    strb w0, [x22], #1
    mov w0, #10
    strb w0, [x22], #1

    add x23, x23, x24
    b row_loop

final_line:
    mov x26, x19
    mov x27, #28
final_loop:
    lsr x0, x26, x27
    and x0, x0, #15
    ldrb w1, [x28, x0]
    strb w1, [x22], #1
    subs x27, x27, #4
    bge final_loop
    mov w0, #10
    strb w0, [x22], #1

    mov x8, #64
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
