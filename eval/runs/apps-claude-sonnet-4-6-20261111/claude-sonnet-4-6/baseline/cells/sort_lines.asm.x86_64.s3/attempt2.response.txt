.globl _start

.section .rodata
nl_char:
    .byte 10

.section .bss
.align 8
input_buf:
    .zero 32768
.align 8
line_ptr:
    .zero 8008
.align 4
line_len:
    .zero 4004

.text

_start:
    xor r15d, r15d

_read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov edx, 32768
    sub edx, r15d
    jle _done_reading
    syscall
    test rax, rax
    jz _done_reading
    add r15d, eax
    jmp _read_loop

_done_reading:
    xor r13d, r13d
    xor r14d, r14d
    xor r12d, r12d

_parse_loop:
    cmp r13d, r15d
    jge _parse_done
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + r13]
    cmp al, 10
    je _found_nl
    inc r13d
    jmp _parse_loop

_found_nl:
    lea rax, [rip + input_buf]
    add rax, r14
    lea rbx, [rip + line_ptr]
    mov qword ptr [rbx + r12*8], rax
    lea rbx, [rip + line_len]
    mov ecx, r13d
    sub ecx, r14d
    mov dword ptr [rbx + r12*4], ecx
    inc r12d
    inc r13d
    mov r14d, r13d
    jmp _parse_loop

_parse_done:
    cmp r14d, r15d
    jge _no_trail
    lea rax, [rip + input_buf]
    add rax, r14
    lea rbx, [rip + line_ptr]
    mov qword ptr [rbx + r12*8], rax
    lea rbx, [rip + line_len]
    mov ecx, r15d
    sub ecx, r14d
    mov dword ptr [rbx + r12*4], ecx
    inc r12d

_no_trail:
    cmp r12d, 2
    jl _sort_done
    mov r8d, 1

_isort_outer:
    cmp r8d, r12d
    jge _sort_done
    lea rbx, [rip + line_ptr]
    mov r9, qword ptr [rbx + r8*8]
    lea rbx, [rip + line_len]
    mov r10d, dword ptr [rbx + r8*4]
    mov eax, r8d
    dec eax
    movsxd r11, eax

_isort_inner:
    test r11, r11
    js _isort_insert
    lea rbx, [rip + line_ptr]
    mov rdi, qword ptr [rbx + r11*8]
    lea rbx, [rip + line_len]
    mov esi, dword ptr [rbx + r11*4]
    mov rdx, r9
    mov ecx, r10d
    push r8
    push r9
    push r10
    push r11
    push r12
    push r13
    push r14
    push r15
    call _compare
    pop r15
    pop r14
    pop r13
    pop r12
    pop r11
    pop r10
    pop r9
    pop r8
    test eax, eax
    jle _isort_insert
    lea rbx, [rip + line_ptr]
    mov rax, qword ptr [rbx + r11*8]
    lea rcx, [r11 + 1]
    mov qword ptr [rbx + rcx*8], rax
    lea rbx, [rip + line_len]
    mov eax, dword ptr [rbx + r11*4]
    mov dword ptr [rbx + rcx*4], eax
    dec r11
    jmp _isort_inner

_isort_insert:
    lea rcx, [r11 + 1]
    lea rbx, [rip + line_ptr]
    mov qword ptr [rbx + rcx*8], r9
    lea rbx, [rip + line_len]
    mov dword ptr [rbx + rcx*4], r10d
    inc r8d
    jmp _isort_outer

_sort_done:
    xor r8d, r8d

_output_loop:
    cmp r8d, r12d
    jge _output_done
    lea rbx, [rip + line_ptr]
    mov rsi, qword ptr [rbx + r8*8]
    lea rbx, [rip + line_len]
    mov edx, dword ptr [rbx + r8*4]
    test edx, edx
    jz _write_nl
    mov eax, 1
    mov edi, 1
    syscall

_write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_char]
    mov edx, 1
    syscall
    inc r8d
    jmp _output_loop

_output_done:
    mov eax, 60
    xor edi, edi
    syscall

_compare:
    push r8
    push r9
    mov eax, esi
    cmp eax, ecx
    jbe _cmp_min_ok
    mov eax, ecx
_cmp_min_ok:
    test eax, eax
    jz _cmp_check_len

_cmp_byte_loop:
    movzx r8d, byte ptr [rdi]
    movzx r9d, byte ptr [rdx]
    cmp r8b, r9b
    jne _cmp_diff
    inc rdi
    inc rdx
    dec eax
    jnz _cmp_byte_loop

_cmp_check_len:
    cmp esi, ecx
    jl _cmp_lt
    jg _cmp_gt
    xor eax, eax
    pop r9
    pop r8
    ret

_cmp_diff:
    jb _cmp_lt

_cmp_gt:
    mov eax, 1
    pop r9
    pop r8
    ret

_cmp_lt:
    mov eax, -1
    pop r9
    pop r8
    ret
