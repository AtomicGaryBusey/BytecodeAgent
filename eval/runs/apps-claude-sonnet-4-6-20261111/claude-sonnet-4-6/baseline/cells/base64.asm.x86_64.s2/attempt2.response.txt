.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
input_buf:
    .zero 49152
    .align 16
output_buf:
    .zero 70000

.text
_start:
    lea r13, [rip + input_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    add rsi, r12
    mov edx, 49152
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 49152
    jb read_loop

read_done:
    lea rdi, [rip + b64_table]
    lea r14, [rip + output_buf]
    xor r15d, r15d
    xor ecx, ecx
    xor ebx, ebx

encode_loop:
    mov eax, r12d
    sub eax, ecx
    cmp eax, 3
    jl handle_remainder

    movzx r8d, byte ptr [r13 + rcx]
    movzx eax, byte ptr [r13 + rcx + 1]
    movzx edx, byte ptr [r13 + rcx + 2]
    shl r8d, 16
    shl eax, 8
    or r8d, eax
    or r8d, edx

    mov edx, r8d
    shr edx, 18
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    shr edx, 12
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    shr edx, 6
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    add ebx, 4
    cmp ebx, 76
    jl skip_nl_full
    mov byte ptr [r14 + r15], 0x0a
    inc r15
    xor ebx, ebx
skip_nl_full:
    add ecx, 3
    jmp encode_loop

handle_remainder:
    test eax, eax
    jz encode_done

    cmp eax, 2
    jge two_rem

    movzx r8d, byte ptr [r13 + rcx]
    shl r8d, 16

    mov edx, r8d
    shr edx, 18
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    shr edx, 12
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov byte ptr [r14 + r15], 0x3d
    inc r15
    mov byte ptr [r14 + r15], 0x3d
    inc r15
    jmp check_nl_rem

two_rem:
    movzx r8d, byte ptr [r13 + rcx]
    movzx eax, byte ptr [r13 + rcx + 1]
    shl r8d, 16
    shl eax, 8
    or r8d, eax

    mov edx, r8d
    shr edx, 18
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    shr edx, 12
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov edx, r8d
    shr edx, 6
    and edx, 0x3f
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov byte ptr [r14 + r15], 0x3d
    inc r15

check_nl_rem:
    add ebx, 4
    cmp ebx, 76
    jl no_nl_rem
    mov byte ptr [r14 + r15], 0x0a
    inc r15
    xor ebx, ebx
no_nl_rem:

encode_done:
    test ebx, ebx
    jz no_final_nl
    mov byte ptr [r14 + r15], 0x0a
    inc r15
no_final_nl:

    test r15, r15
    jz do_exit

    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
