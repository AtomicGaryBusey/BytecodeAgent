.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
line_buf:
    .zero 128

.text

write_hex8:
    lea rdi, [rip + hex_chars]
    mov ecx, 28
.Lw8loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    movzx edx, byte ptr [rdi + rdx]
    mov byte ptr [r15], dl
    inc r15
    sub ecx, 4
    jge .Lw8loop
    ret

_start:
    xor r12d, r12d

.Lread_loop:
    mov rdx, 65536
    sub rdx, r12
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jz .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    xor r13d, r13d

.Lprocess:
    cmp r13, r12
    jge .Lfinal

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jle .Ln_ok
    mov r14d, 16
.Ln_ok:
    lea r15, [rip + line_buf]

    mov rax, r13
    call write_hex8

    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], ' '
    inc r15

    xor ebx, ebx

.Lhex_loop:
    cmp rbx, 16
    jge .Lhex_done

    cmp rbx, 8
    jne .Lno_extra
    mov byte ptr [r15], ' '
    inc r15
.Lno_extra:
    cmp rbx, r14
    jge .Lhex_pad

    lea rdi, [rip + input_buf]
    add rdi, r13
    add rdi, rbx
    movzx eax, byte ptr [rdi]
    lea rdi, [rip + hex_chars]
    mov ecx, eax
    shr ecx, 4
    movzx ecx, byte ptr [rdi + rcx]
    mov byte ptr [r15], cl
    inc r15
    and eax, 15
    movzx eax, byte ptr [rdi + rax]
    mov byte ptr [r15], al
    inc r15
    mov byte ptr [r15], ' '
    inc r15
    jmp .Lhex_next

.Lhex_pad:
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], ' '
    inc r15

.Lhex_next:
    inc rbx
    jmp .Lhex_loop

.Lhex_done:
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], '|'
    inc r15

    xor ebx, ebx

.Lascii_loop:
    cmp rbx, r14
    jge .Lascii_done

    lea rdi, [rip + input_buf]
    add rdi, r13
    add rdi, rbx
    movzx eax, byte ptr [rdi]

    cmp al, 0x20
    jb .Ldot
    cmp al, 0x7e
    jbe .Lprint_char
.Ldot:
    mov al, '.'
.Lprint_char:
    mov byte ptr [r15], al
    inc r15
    inc rbx
    jmp .Lascii_loop

.Lascii_done:
    mov byte ptr [r15], '|'
    inc r15
    mov byte ptr [r15], 0x0a
    inc r15

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi
    syscall

    add r13, r14
    jmp .Lprocess

.Lfinal:
    lea r15, [rip + line_buf]
    mov rax, r12
    call write_hex8
    mov byte ptr [r15], 0x0a
    inc r15

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi
    syscall

    mov eax, 60
    xor edi, edi
    syscall
