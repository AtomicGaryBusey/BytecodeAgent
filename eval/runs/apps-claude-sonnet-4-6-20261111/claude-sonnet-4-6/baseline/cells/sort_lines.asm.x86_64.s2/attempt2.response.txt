.globl _start

.bss
.align 16
input_buf:
    .zero 32768

.align 8
line_starts:
    .zero 8000

.align 8
line_lengths:
    .zero 8000

.section .rodata
newline_char:
    .byte 0x0A

.text

compare_lines:
    # rdi=s1, rsi=len1, rdx=s2, rcx=len2 -> rax (neg/0/pos)
    # clobbers: rax, rcx, rdx, rsi, rdi
    # preserves: rbx, r12, r13, r14, r15, rbp
    mov rax, rsi
    cmp rax, rcx
    jle cl_min_ok
    mov rax, rcx
cl_min_ok:
    push rsi
    push rcx
    mov rsi, rdx
    mov rcx, rax
    test rcx, rcx
    jz cl_len_cmp
    repe cmpsb
    jne cl_diff
cl_len_cmp:
    pop rcx
    pop rax
    sub rax, rcx
    ret
cl_diff:
    add rsp, 16
    movzx eax, byte ptr [rdi - 1]
    movzx ecx, byte ptr [rsi - 1]
    sub eax, ecx
    ret

_start:
    xor r12, r12
    lea r13, [rip + input_buf]

read_loop:
    mov rdx, 32768
    sub rdx, r12
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r15, [rip + line_starts]
    lea rbp, [rip + line_lengths]
    xor r14, r14
    xor rbx, rbx

parse_loop:
    cmp rbx, r12
    jge parse_done
    mov rcx, rbx

find_newline:
    cmp rcx, r12
    jge end_of_input
    movzx eax, byte ptr [r13 + rcx]
    cmp al, 0x0A
    je found_newline
    inc rcx
    jmp find_newline

found_newline:
    lea rax, [r13 + rbx]
    mov qword ptr [r15 + r14*8], rax
    mov rax, rcx
    sub rax, rbx
    mov qword ptr [rbp + r14*8], rax
    inc r14
    lea rbx, [rcx + 1]
    jmp parse_loop

end_of_input:
    cmp rcx, rbx
    je parse_done
    lea rax, [r13 + rbx]
    mov qword ptr [r15 + r14*8], rax
    mov rax, rcx
    sub rax, rbx
    mov qword ptr [rbp + r14*8], rax
    inc r14

parse_done:
    cmp r14, 1
    jle print_lines
    mov rbx, 1

isort_outer:
    cmp rbx, r14
    jge print_lines
    sub rsp, 16
    mov rax, qword ptr [r15 + rbx*8]
    mov qword ptr [rsp + 8], rax
    mov rax, qword ptr [rbp + rbx*8]
    mov qword ptr [rsp], rax
    mov r12, rbx
    dec r12

isort_inner:
    test r12, r12
    js isort_j_neg
    mov rdi, qword ptr [r15 + r12*8]
    mov rsi, qword ptr [rbp + r12*8]
    mov rdx, qword ptr [rsp + 8]
    mov rcx, qword ptr [rsp]
    call compare_lines
    test rax, rax
    jle isort_found
    mov rax, qword ptr [r15 + r12*8]
    mov qword ptr [r15 + r12*8 + 8], rax
    mov rax, qword ptr [rbp + r12*8]
    mov qword ptr [rbp + r12*8 + 8], rax
    dec r12
    jmp isort_inner

isort_found:
    inc r12
    jmp isort_place

isort_j_neg:
    inc r12

isort_place:
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [r15 + r12*8], rax
    mov rax, qword ptr [rsp]
    mov qword ptr [rbp + r12*8], rax
    add rsp, 16
    inc rbx
    jmp isort_outer

print_lines:
    xor rbx, rbx

print_loop:
    cmp rbx, r14
    jge exit_ok
    mov rax, 1
    mov rdi, 1
    mov rsi, qword ptr [r15 + rbx*8]
    mov rdx, qword ptr [rbp + rbx*8]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    inc rbx
    jmp print_loop

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
