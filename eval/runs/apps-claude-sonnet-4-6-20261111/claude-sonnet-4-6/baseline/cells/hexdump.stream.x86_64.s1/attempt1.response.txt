.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
output_buf:
    .zero 128

.text

_start:
    lea r15, [rip + input_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 65536
    jl read_loop

read_done:
    xor r13d, r13d

process_row:
    cmp r13d, r12d
    jge print_final_offset

    mov r14d, r12d
    sub r14d, r13d
    cmp r14d, 16
    jle row_n_set
    mov r14d, 16
row_n_set:

    lea rdi, [rip + output_buf]
    lea r9, [r15 + r13]

    mov eax, r13d
    call write_hex8

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi

    xor r10d, r10d

hex_loop:
    cmp r10d, 16
    jge hex_loop_done

    cmp r10d, 8
    jne hex_no_mid_space
    mov byte ptr [rdi], 0x20
    inc rdi
hex_no_mid_space:

    cmp r10d, r14d
    jge hex_pad_pos

    movzx eax, byte ptr [r9 + r10]
    call write_hex2
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_loop_next

hex_pad_pos:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

hex_loop_next:
    inc r10d
    jmp hex_loop

hex_loop_done:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi

    xor r10d, r10d

ascii_loop:
    cmp r10d, r14d
    jge ascii_loop_done

    movzx eax, byte ptr [r9 + r10]
    cmp al, 0x20
    jb ascii_is_dot
    cmp al, 0x7e
    ja ascii_is_dot
    mov byte ptr [rdi], al
    jmp ascii_loop_next
ascii_is_dot:
    mov byte ptr [rdi], 0x2e
ascii_loop_next:
    inc rdi
    inc r10d
    jmp ascii_loop

ascii_loop_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r13d, r14d
    jmp process_row

print_final_offset:
    lea rdi, [rip + output_buf]
    mov eax, r12d
    call write_hex8
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

write_hex8:
    push rbx
    lea rbx, [rip + hex_chars]
    mov ecx, 28
write_hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xf
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    test ecx, ecx
    jz write_hex8_done
    sub ecx, 4
    jmp write_hex8_loop
write_hex8_done:
    pop rbx
    ret

write_hex2:
    push rbx
    lea rbx, [rip + hex_chars]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 0xf
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    inc rdi
    pop rbx
    ret
