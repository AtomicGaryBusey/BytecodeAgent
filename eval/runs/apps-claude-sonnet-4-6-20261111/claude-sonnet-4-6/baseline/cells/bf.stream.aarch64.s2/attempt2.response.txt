.globl _start

.bss
.align 4
input_buf:
    .zero 65536
.align 4
tape:
    .zero 30000
.align 4
output_buf:
    .zero 65536
.align 4
jump_table:
    .zero 262144
.align 4
stack_buf:
    .zero 4000

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

.Lread_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, .Lread_done
    mov x8, #63
    svc #0
    cmp x0, #0
    ble .Lread_done
    add x20, x20, x0
    b .Lread_loop

.Lread_done:
    mov x21, #0

.Lfind_bang:
    cmp x21, x20
    bge .Lno_bang
    ldrb w0, [x19, x21]
    cmp w0, #0x21
    beq .Lfound_bang
    add x21, x21, #1
    b .Lfind_bang

.Lfound_bang:
    add x22, x21, #1
    sub x23, x20, x22
    b .Ldo_check

.Lno_bang:
    mov x21, x20
    mov x22, x20
    mov x23, #0

.Ldo_check:
    adrp x24, jump_table
    add x24, x24, :lo12:jump_table
    adrp x25, stack_buf
    add x25, x25, :lo12:stack_buf
    mov x26, #0
    mov x27, #0

.Lcheck_loop:
    cmp x27, x21
    bge .Lcheck_done
    ldrb w0, [x19, x27]
    cmp w0, #0x5B
    beq .Lcheck_open
    cmp w0, #0x5D
    beq .Lcheck_close
    add x27, x27, #1
    b .Lcheck_loop

.Lcheck_open:
    cmp x26, #1000
    bge .Lunbalanced
    str w27, [x25, x26, lsl #2]
    add x26, x26, #1
    add x27, x27, #1
    b .Lcheck_loop

.Lcheck_close:
    cbz x26, .Lunbalanced
    sub x26, x26, #1
    ldr w1, [x25, x26, lsl #2]
    add w2, w27, #1
    str w2, [x24, x1, lsl #2]
    add w2, w1, #1
    str w2, [x24, x27, lsl #2]
    add x27, x27, #1
    b .Lcheck_loop

.Lcheck_done:
    cbnz x26, .Lunbalanced
    b .Lrun

.Lunbalanced:
    mov x0, #1
    mov x8, #93
    svc #0

.Lrun:
    mov x28, x24
    add x24, x19, x22
    add x25, x24, x23
    adrp x23, tape
    add x23, x23, :lo12:tape
    mov x20, x21
    mov x21, #0
    mov x22, #0
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x27, #0

.Linterp_loop:
    cmp x21, x20
    bge .Linterp_done
    ldrb w0, [x19, x21]
    cmp w0, #0x3E
    beq .Lcmd_right
    cmp w0, #0x3C
    beq .Lcmd_left
    cmp w0, #0x2B
    beq .Lcmd_inc
    cmp w0, #0x2D
    beq .Lcmd_dec
    cmp w0, #0x2E
    beq .Lcmd_out
    cmp w0, #0x2C
    beq .Lcmd_in
    cmp w0, #0x5B
    beq .Lcmd_lbracket
    cmp w0, #0x5D
    beq .Lcmd_rbracket
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_right:
    add x22, x22, #1
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_left:
    sub x22, x22, #1
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_inc:
    ldrb w0, [x23, x22]
    add w0, w0, #1
    and w0, w0, #0xFF
    strb w0, [x23, x22]
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_dec:
    ldrb w0, [x23, x22]
    sub w0, w0, #1
    and w0, w0, #0xFF
    strb w0, [x23, x22]
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_out:
    ldrb w0, [x23, x22]
    strb w0, [x26, x27]
    add x27, x27, #1
    cmp x27, #65536
    blt .Lcmd_out_done
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0
    mov x27, #0
.Lcmd_out_done:
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_in:
    cmp x24, x25
    bhs .Lcmd_in_eof
    ldrb w0, [x24], #1
    strb w0, [x23, x22]
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_in_eof:
    strb wzr, [x23, x22]
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_lbracket:
    ldrb w0, [x23, x22]
    cbnz w0, .Lcmd_lbracket_nz
    ldr w1, [x28, x21, lsl #2]
    mov x21, x1
    b .Linterp_loop
.Lcmd_lbracket_nz:
    add x21, x21, #1
    b .Linterp_loop

.Lcmd_rbracket:
    ldrb w0, [x23, x22]
    cbz w0, .Lcmd_rbracket_z
    ldr w1, [x28, x21, lsl #2]
    mov x21, x1
    b .Linterp_loop
.Lcmd_rbracket_z:
    add x21, x21, #1
    b .Linterp_loop

.Linterp_done:
    cbz x27, .Lexit_ok
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0

.Lexit_ok:
    mov x0, #0
    mov x8, #93
    svc #0
