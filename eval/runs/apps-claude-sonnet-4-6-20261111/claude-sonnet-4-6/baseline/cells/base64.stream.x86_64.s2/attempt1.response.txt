.globl _start

.section .rodata
.align 16
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
    .zero 49152
.align 16
output_buf:
    .zero 70000

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 49152
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    cmp r12d, 49152
    jl read_loop

read_done:
    test r12d, r12d
    jz do_exit

    lea r14, [rip + output_buf]
    xor r15d, r15d
    xor ebx, ebx
    xor r10d, r10d
    lea rdx, [rip + b64_table]

encode_loop:
    cmp ebx, r12d
    jge encode_done

    movzx r8d, byte ptr [r13 + rbx]
    inc ebx
    cmp ebx, r12d
    jge one_byte

    movzx r9d, byte ptr [r13 + rbx]
    inc ebx
    cmp ebx, r12d
    jge two_bytes

    movzx r11d, byte ptr [r13 + rbx]
    inc ebx

    # c0 = b0 >> 2
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    mov eax, r9d
    and eax, 0xF
    shl eax, 2
    mov ecx, r11d
    shr ecx, 6
    or eax, ecx
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c3 = b2 & 0x3F
    mov eax, r11d
    and eax, 0x3F
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    add r10d, 4
    cmp r10d, 76
    jl encode_loop
    mov byte ptr [r14 + r15], 0x0A
    inc r15d
    xor r10d, r10d
    jmp encode_loop

one_byte:
    # c0 = b0 >> 2
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c1 = (b0 & 3) << 4
    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    mov byte ptr [r14 + r15], 0x3D
    inc r15d
    mov byte ptr [r14 + r15], 0x3D
    inc r15d
    add r10d, 4
    jmp encode_done

two_bytes:
    # c0 = b0 >> 2
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    # c2 = (b1 & 0xF) << 2
    mov eax, r9d
    and eax, 0xF
    shl eax, 2
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r14 + r15], al
    inc r15d

    mov byte ptr [r14 + r15], 0x3D
    inc r15d
    add r10d, 4
    jmp encode_done

encode_done:
    test r10d, r10d
    jz do_write
    mov byte ptr [r14 + r15], 0x0A
    inc r15d

do_write:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, r15d
    syscall

do_exit:
    xor edi, edi
    mov eax, 60
    syscall
