.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .zero 65536
.align 4
out_buf:
    .zero 128

.text

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x28, hex_chars
    add x28, x28, :lo12:hex_chars
    adrp x27, out_buf
    add x27, x27, :lo12:out_buf
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    cmp x0, #0
    blt read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x19
    bge final_offset

    sub x22, x19, x21
    cmp x22, #16
    ble row_process
    mov x22, #16
row_process:
    add x23, x20, x21
    mov x25, x27

    // Print 8-digit hex offset
    mov x0, x21
    mov x1, #28
off_hex_loop:
    lsr x2, x0, x1
    and x2, x2, #0xf
    ldrb w3, [x28, x2]
    strb w3, [x25], #1
    subs x1, x1, #4
    bge off_hex_loop

    // Two spaces
    mov w3, #0x20
    strb w3, [x25], #1
    strb w3, [x25], #1

    // Hex bytes section
    mov x26, #0
hex_loop:
    cmp x26, #16
    bge hex_section_done

    // Extra space after position 7 (before position 8)
    cmp x26, #8
    bne hex_check_valid
    mov w3, #0x20
    strb w3, [x25], #1
hex_check_valid:
    cmp x26, x22
    bge hex_three_spaces

    // Valid byte: print "XX "
    ldrb w0, [x23, x26]
    lsr x2, x0, #4
    ldrb w3, [x28, x2]
    strb w3, [x25], #1
    and x2, x0, #0xf
    ldrb w3, [x28, x2]
    strb w3, [x25], #1
    mov w3, #0x20
    strb w3, [x25], #1
    b hex_loop_next

hex_three_spaces:
    mov w3, #0x20
    strb w3, [x25], #1
    strb w3, [x25], #1
    strb w3, [x25], #1

hex_loop_next:
    add x26, x26, #1
    b hex_loop

hex_section_done:
    // " |"
    mov w3, #0x20
    strb w3, [x25], #1
    mov w3, #0x7c
    strb w3, [x25], #1

    // ASCII section
    mov x26, #0
ascii_char_loop:
    cmp x26, x22
    bge ascii_section_done
    ldrb w0, [x23, x26]
    cmp w0, #0x20
    blt print_dot
    cmp w0, #0x7e
    bgt print_dot
    strb w0, [x25], #1
    b ascii_char_next
print_dot:
    mov w0, #0x2e
    strb w0, [x25], #1
ascii_char_next:
    add x26, x26, #1
    b ascii_char_loop

ascii_section_done:
    // "|\n"
    mov w3, #0x7c
    strb w3, [x25], #1
    mov w3, #0x0a
    strb w3, [x25], #1

    // Write line to stdout
    mov x0, #1
    mov x1, x27
    sub x2, x25, x27
    mov x8, #64
    svc #0

    add x21, x21, x22
    b row_loop

final_offset:
    mov x25, x27
    mov x0, x19
    mov x1, #28
final_hex_loop:
    lsr x2, x0, x1
    and x2, x2, #0xf
    ldrb w3, [x28, x2]
    strb w3, [x25], #1
    subs x1, x1, #4
    bge final_hex_loop

    mov w3, #0x0a
    strb w3, [x25], #1

    mov x0, #1
    mov x1, x27
    sub x2, x25, x27
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
