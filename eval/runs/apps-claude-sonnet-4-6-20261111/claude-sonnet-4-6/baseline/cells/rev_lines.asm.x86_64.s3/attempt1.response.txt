.globl _start

.bss
.align 16
input_buf:
    .zero 65536
output_buf:
    .zero 65538

.text
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    jle done_reading
    syscall
    test rax, rax
    jle done_reading
    add r12, rax
    jmp read_loop

done_reading:
    lea r13, [rip + output_buf]
    xor r14d, r14d

process_lines:
    cmp r14, r12
    jge write_output

    mov r9, r14
scan_newline:
    cmp r9, r12
    jge found_line
    movzx eax, byte ptr [rbx + r9]
    cmp al, 0x0A
    je found_line
    inc r9
    jmp scan_newline

found_line:
    cmp r14, r9
    jge after_reverse

    mov r10, r9
    dec r10
reverse_loop:
    movzx eax, byte ptr [rbx + r10]
    mov byte ptr [r13], al
    inc r13
    cmp r10, r14
    jle after_reverse
    dec r10
    jmp reverse_loop

after_reverse:
    mov byte ptr [r13], 0x0A
    inc r13

    mov r14, r9
    cmp r14, r12
    jge write_output
    inc r14
    jmp process_lines

write_output:
    lea rsi, [rip + output_buf]
    mov rdx, r13
    sub rdx, rsi
    test rdx, rdx
    jz do_exit
    mov rax, 1
    mov rdi, 1
    syscall

do_exit:
    mov rax, 60
    xor edi, edi
    syscall
