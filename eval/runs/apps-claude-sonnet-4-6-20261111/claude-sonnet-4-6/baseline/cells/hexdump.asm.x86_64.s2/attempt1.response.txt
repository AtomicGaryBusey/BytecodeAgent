.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
out_buf:
    .zero 256

.text

_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    test eax, eax
    jle read_done
    add r12d, eax
    cmp r12d, 65536
    jb read_loop

read_done:
    xor r13d, r13d

row_loop:
    cmp r13d, r12d
    jge print_final

    mov eax, r12d
    sub eax, r13d
    cmp eax, 16
    jbe row_got_n
    mov eax, 16
row_got_n:
    mov r15d, eax
    lea r14, [rip + input_buf]
    add r14, r13
    call format_write_row
    add r13d, 16
    jmp row_loop

print_final:
    lea r8, [rip + out_buf]
    mov r9d, r12d
    mov ecx, 28
pf_loop:
    mov eax, r9d
    shr eax, cl
    and eax, 0xf
    lea r10, [rip + hex_chars]
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [r8], al
    inc r8
    sub ecx, 4
    jns pf_loop
    mov byte ptr [r8], 0x0a
    inc r8
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    sub r8, rsi
    mov rdx, r8
    syscall
    mov eax, 60
    xor edi, edi
    syscall


format_write_row:
    lea r8, [rip + out_buf]
    mov r9d, r13d
    mov ecx, 28
fw_off_loop:
    mov eax, r9d
    shr eax, cl
    and eax, 0xf
    lea r10, [rip + hex_chars]
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [r8], al
    inc r8
    sub ecx, 4
    jns fw_off_loop
    mov byte ptr [r8], 0x20
    mov byte ptr [r8+1], 0x20
    add r8, 2
    xor ecx, ecx
fw_hex_loop:
    cmp ecx, 16
    jge fw_hex_done
    cmp ecx, 8
    jne fw_no_extra
    mov byte ptr [r8], 0x20
    inc r8
fw_no_extra:
    cmp ecx, r15d
    jge fw_empty
    movzx eax, byte ptr [r14 + rcx]
    mov r10d, eax
    shr r10d, 4
    and eax, 0xf
    lea r11, [rip + hex_chars]
    movzx r10d, byte ptr [r11 + r10]
    mov byte ptr [r8], r10b
    inc r8
    movzx eax, byte ptr [r11 + rax]
    mov byte ptr [r8], al
    inc r8
    mov byte ptr [r8], 0x20
    inc r8
    jmp fw_hex_next
fw_empty:
    mov byte ptr [r8], 0x20
    mov byte ptr [r8+1], 0x20
    mov byte ptr [r8+2], 0x20
    add r8, 3
fw_hex_next:
    inc ecx
    jmp fw_hex_loop
fw_hex_done:
    mov byte ptr [r8], 0x20
    mov byte ptr [r8+1], 0x7c
    add r8, 2
    xor ecx, ecx
fw_ascii_loop:
    cmp ecx, r15d
    jge fw_ascii_done
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 0x20
    jb fw_ascii_dot
    cmp al, 0x7e
    ja fw_ascii_dot
    mov byte ptr [r8], al
    jmp fw_ascii_next
fw_ascii_dot:
    mov byte ptr [r8], 0x2e
fw_ascii_next:
    inc r8
    inc ecx
    jmp fw_ascii_loop
fw_ascii_done:
    mov byte ptr [r8], 0x7c
    mov byte ptr [r8+1], 0x0a
    add r8, 2
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    sub r8, rsi
    mov rdx, r8
    syscall
    ret
