.globl _start

.section .bss
.align 4
input_buf:
    .zero 65536
output_buf:
    .zero 65537

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, done_reading
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble done_reading
    add x20, x20, x0
    b read_loop

done_reading:
    mov x21, #0

process_lines:
    cmp x21, x20
    bge done_output
    mov x22, x21

find_nl:
    cmp x22, x20
    bge no_newline
    ldrb w0, [x19, x22]
    cmp w0, #0x0a
    beq found_nl
    add x22, x22, #1
    b find_nl

found_nl:
    sub x23, x22, x21
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x25, #0
    cbz x23, emit_nl
    add x26, x21, x23
    sub x26, x26, #1

rev_loop:
    ldrb w0, [x19, x26]
    strb w0, [x24, x25]
    add x25, x25, #1
    sub x26, x26, #1
    cmp x25, x23
    blt rev_loop

emit_nl:
    mov w0, #0x0a
    strb w0, [x24, x25]
    add x25, x25, #1
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    add x21, x22, #1
    b process_lines

no_newline:
    sub x23, x20, x21
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x25, #0
    add x26, x21, x23
    sub x26, x26, #1

rev_loop2:
    ldrb w0, [x19, x26]
    strb w0, [x24, x25]
    add x25, x25, #1
    sub x26, x26, #1
    cmp x25, x23
    blt rev_loop2

    mov w0, #0x0a
    strb w0, [x24, x25]
    add x25, x25, #1
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0

done_output:
    mov x0, #0
    mov x8, #93
    svc #0
