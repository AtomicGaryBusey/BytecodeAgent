.globl _start

.bss
.align 16
buffer: .zero 65536
outbuf: .zero 64

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    mov rcx, rax
    lea rsi, [rip + buffer]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    inc r14
    cmp al, 0x0A
    je is_newline
    cmp al, 0x09
    jb not_whitespace
    cmp al, 0x0D
    jbe is_whitespace
    cmp al, 0x20
    je is_whitespace
    jmp not_whitespace

is_newline:
    inc r12

is_whitespace:
    xor r15, r15
    dec rcx
    jnz process_loop
    jmp read_loop

not_whitespace:
    test r15, r15
    jnz in_word
    inc r13
    mov r15d, 1
in_word:
    dec rcx
    jnz process_loop
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call num_to_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call num_to_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call num_to_dec
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall
    mov eax, 60
    xor edi, edi
    syscall

num_to_dec:
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
1:
    xor ecx, ecx
    mov r9d, 10
2:
    xor edx, edx
    div r9
    add dl, 0x30
    push rdx
    inc rcx
    test rax, rax
    jnz 2b
3:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz 3b
    ret
