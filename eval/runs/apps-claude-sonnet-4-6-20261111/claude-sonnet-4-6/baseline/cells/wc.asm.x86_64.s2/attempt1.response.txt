.globl _start

.bss
.align 16
buffer:
    .zero 65536

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall

    test rax, rax
    jle print_results

    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

scan_loop:
    test rcx, rcx
    jz read_loop

    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx

    cmp al, 0x0A
    jne not_newline
    inc r12
    xor r15, r15
    jmp scan_loop

not_newline:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jle is_ws

is_word:
    test r15, r15
    jnz scan_loop
    inc r13
    mov r15, 1
    jmp scan_loop

is_ws:
    xor r15, r15
    jmp scan_loop

print_results:
    sub rsp, 64
    mov rbx, rsp
    mov rdi, rsp

    mov rax, r12
    call itoa

    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, r13
    call itoa

    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, r14
    call itoa

    mov byte ptr [rdi], 0x0A
    inc rdi

    mov rdx, rdi
    sub rdx, rbx
    mov rdi, 1
    mov rsi, rbx
    mov rax, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

itoa:
    push rbp
    mov rbp, rsp
    sub rsp, 32

    test rax, rax
    jnz 1f

    mov byte ptr [rdi], 0x30
    inc rdi
    mov rsp, rbp
    pop rbp
    ret

1:
    lea r8, [rbp - 32]
    xor r9, r9
    mov r10, 10

2:
    xor rdx, rdx
    div r10
    mov byte ptr [r8 + r9], dl
    inc r9
    test rax, rax
    jnz 2b

3:
    dec r9
    movzx eax, byte ptr [r8 + r9]
    add al, 0x30
    mov byte ptr [rdi], al
    inc rdi
    test r9, r9
    jnz 3b

    mov rsp, rbp
    pop rbp
    ret
