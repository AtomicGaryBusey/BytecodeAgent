.globl _start

.section .rodata
nl:
    .byte 0x0A

.section .bss
.align 8
inbuf:
    .zero 65536

.text

# reverse bytes [rdi..rsi] inclusive
# trashes: rdi, rsi, eax, ecx
reverse_range:
1:
    cmp rdi, rsi
    jge 2f
    movzx eax, byte ptr [rdi]
    movzx ecx, byte ptr [rsi]
    mov byte ptr [rdi], cl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp 1b
2:
    ret

_start:
    xor r12, r12
    lea r13, [rip + inbuf]

read_all:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz after_read
    syscall
    test rax, rax
    jle after_read
    add r12, rax
    jmp read_all

after_read:
    test r12, r12
    jz exit_prog

    lea r14, [r13 + r12]
    mov r15, r13

next_line:
    cmp r15, r14
    jge exit_prog

    mov rbx, r15
scan_nl:
    cmp rbx, r14
    jge last_line
    cmp byte ptr [rbx], 0x0A
    je nl_found
    inc rbx
    jmp scan_nl

nl_found:
    # line bytes: [r15, rbx), newline at [rbx]
    mov rdi, r15
    mov rsi, rbx
    dec rsi
    call reverse_range
    # write line + existing newline = [r15, rbx+1)
    mov rax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, rbx
    sub rdx, r15
    inc rdx
    syscall
    lea r15, [rbx + 1]
    jmp next_line

last_line:
    # line bytes: [r15, r14), no terminating newline
    mov rdi, r15
    mov rsi, r14
    dec rsi
    call reverse_range
    # write line bytes
    mov rax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r14
    sub rdx, r15
    syscall
    # write newline
    mov rax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
