.globl _start

.equ SYS_READ,  63
.equ SYS_WRITE, 64
.equ SYS_EXIT,  93

.section .bss
.align 4
input_buf:
    .zero 32768
.align 3
line_ptrs:
    .zero 8000
.align 3
line_lens:
    .zero 8000

.section .rodata
newline_ch:
    .byte 0x0A

.text
_start:
    // Phase 1: read all stdin into input_buf
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, #0
    mov  x9,  #32768

1:
    sub  x2, x9, x20
    cbz  x2, 2f
    mov  x0, #0
    add  x1, x19, x20
    mov  x8, #SYS_READ
    svc  #0
    cbz  x0, 2f
    add  x20, x20, x0
    b    1b

2:
    // Phase 2: parse lines
    adrp x21, line_ptrs
    add  x21, x21, :lo12:line_ptrs
    adrp x22, line_lens
    add  x22, x22, :lo12:line_lens
    mov  x23, #0
    mov  x24, x19
    mov  x25, x19
    add  x26, x19, x20
    cbz  x20, 5f

3:
    cmp  x24, x26
    bge  4f
    ldrb w0, [x24], #1
    cmp  w0, #0x0A
    bne  3b
    sub  x0, x24, x25
    sub  x0, x0, #1
    str  x25, [x21, x23, lsl #3]
    str  x0,  [x22, x23, lsl #3]
    add  x23, x23, #1
    mov  x25, x24
    b    3b

4:
    cmp  x25, x26
    beq  5f
    sub  x0, x26, x25
    str  x25, [x21, x23, lsl #3]
    str  x0,  [x22, x23, lsl #3]
    add  x23, x23, #1

5:
    // Phase 3: insertion sort
    cmp  x23, #1
    ble  8f
    mov  x24, #1

6:
    cmp  x24, x23
    bge  8f
    ldr  x25, [x21, x24, lsl #3]
    ldr  x26, [x22, x24, lsl #3]
    sub  x27, x24, #1

7:
    cmp  x27, #0
    blt  10f
    ldr  x0, [x21, x27, lsl #3]
    ldr  x1, [x22, x27, lsl #3]
    mov  x2, x25
    mov  x3, x26
    bl   compare_lines
    cmp  x0, #0
    ble  10f
    add  x28, x27, #1
    ldr  x9,  [x21, x27, lsl #3]
    str  x9,  [x21, x28, lsl #3]
    ldr  x9,  [x22, x27, lsl #3]
    str  x9,  [x22, x28, lsl #3]
    sub  x27, x27, #1
    b    7b

10:
    add  x28, x27, #1
    str  x25, [x21, x28, lsl #3]
    str  x26, [x22, x28, lsl #3]
    add  x24, x24, #1
    b    6b

8:
    // Phase 4: print sorted lines
    adrp x25, newline_ch
    add  x25, x25, :lo12:newline_ch
    mov  x24, #0

9:
    cmp  x24, x23
    bge  11f
    ldr  x1, [x21, x24, lsl #3]
    ldr  x2, [x22, x24, lsl #3]
    cbz  x2, 12f
    mov  x0, #1
    mov  x8, #SYS_WRITE
    svc  #0

12:
    mov  x0, #1
    mov  x1, x25
    mov  x2, #1
    mov  x8, #SYS_WRITE
    svc  #0
    add  x24, x24, #1
    b    9b

11:
    mov  x0, #0
    mov  x8, #SYS_EXIT
    svc  #0

compare_lines:
    // x0=ptr1, x1=len1, x2=ptr2, x3=len2
    // returns x0: negative/zero/positive
    cmp  x1, x3
    csel x9, x1, x3, lo
    mov  x10, #0

20:
    cmp  x10, x9
    bge  21f
    ldrb w11, [x0, x10]
    ldrb w12, [x2, x10]
    cmp  w11, w12
    bne  22f
    add  x10, x10, #1
    b    20b

21:
    sub  x0, x1, x3
    ret

22:
    sub  x0, x11, x12
    ret
