.globl _start

.text

compare_lines:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0
cl_loop:
    cmp x5, x4
    bge cl_prefix
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    bne cl_diff
    add x5, x5, #1
    b cl_loop
cl_diff:
    sub w0, w6, w7
    sxtw x0, w0
    ret
cl_prefix:
    sub x0, x1, x3
    ret

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, line_starts
    add x21, x21, :lo12:line_starts
    adrp x22, line_lens
    add x22, x22, :lo12:line_lens
    mov x20, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    mov x2, #32768
    sub x2, x2, x20
    cbz x2, done_reading
    svc #0
    cbz x0, done_reading
    add x20, x20, x0
    b read_loop

done_reading:
    mov x23, #0
    mov x24, #0
    mov x25, x19
    cbz x20, do_sort

parse_loop:
    cmp x23, x20
    bge end_parse
    ldrb w0, [x19, x23]
    add x23, x23, #1
    cmp w0, #0x0a
    bne parse_loop
    str x25, [x21, x24, lsl #3]
    add x0, x19, x23
    sub x0, x0, #1
    sub x0, x0, x25
    str x0, [x22, x24, lsl #3]
    add x24, x24, #1
    add x25, x19, x23
    cmp x24, #1000
    blt parse_loop

end_parse:
    add x0, x19, x20
    cmp x25, x0
    bge do_sort
    str x25, [x21, x24, lsl #3]
    sub x0, x0, x25
    str x0, [x22, x24, lsl #3]
    add x24, x24, #1

do_sort:
    cmp x24, #2
    blt do_output
    sub x26, x24, #1

bubble_outer:
    cbz x26, do_output
    mov x27, #0

bubble_inner:
    cmp x27, x26
    bge bubble_dec
    add x28, x27, #1
    ldr x0, [x21, x27, lsl #3]
    ldr x1, [x22, x27, lsl #3]
    ldr x2, [x21, x28, lsl #3]
    ldr x3, [x22, x28, lsl #3]
    bl compare_lines
    cmp x0, #0
    ble no_swap
    ldr x0, [x21, x27, lsl #3]
    ldr x1, [x21, x28, lsl #3]
    str x1, [x21, x27, lsl #3]
    str x0, [x21, x28, lsl #3]
    ldr x0, [x22, x27, lsl #3]
    ldr x1, [x22, x28, lsl #3]
    str x1, [x22, x27, lsl #3]
    str x0, [x22, x28, lsl #3]
no_swap:
    add x27, x27, #1
    b bubble_inner

bubble_dec:
    sub x26, x26, #1
    b bubble_outer

do_output:
    mov x27, #0

out_loop:
    cmp x27, x24
    bge exit_prog
    ldr x25, [x21, x27, lsl #3]
    ldr x26, [x22, x27, lsl #3]
    cbz x26, out_newline
    mov x8, #64
    mov x0, #1
    mov x1, x25
    mov x2, x26
    svc #0
out_newline:
    adrp x1, nl_byte
    add x1, x1, :lo12:nl_byte
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    add x27, x27, #1
    b out_loop

exit_prog:
    mov x8, #93
    mov x0, #0
    svc #0

.section .rodata
nl_byte:
    .byte 0x0a

.section .bss
.align 4
input_buf:
    .zero 32768
.align 3
line_starts:
    .zero 8000
.align 3
line_lens:
    .zero 8000
