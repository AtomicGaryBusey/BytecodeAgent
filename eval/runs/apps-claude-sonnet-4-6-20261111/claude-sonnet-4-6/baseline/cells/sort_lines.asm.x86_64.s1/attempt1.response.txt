.globl _start

.section .rodata
newline_byte: .byte 0x0A

.bss
.align 16
input_buf: .zero 32768
.align 8
line_offsets: .zero 8000
.align 8
line_lengths: .zero 8000
.align 8
line_count_val: .zero 8

.text

# compare_lines(rdi=off_a, rsi=len_a, rdx=off_b, rcx=len_b)
# Returns rax: -1 (a<b), 0 (a==b), 1 (a>b)
# Saves/restores r8, r9, r10; clobbers rax only
compare_lines:
    push r8
    push r9
    push r10
    lea r8, [rip + input_buf]
    mov r9, rsi
    cmp r9, rcx
    jbe cmp_got_min
    mov r9, rcx
cmp_got_min:
    xor r10, r10
cmp_next:
    cmp r10, r9
    jge cmp_by_len
    movzx eax, byte ptr [r8 + rdi + r10]
    cmp al, byte ptr [r8 + rdx + r10]
    jb cmp_ret_neg
    ja cmp_ret_pos
    inc r10
    jmp cmp_next
cmp_by_len:
    cmp rsi, rcx
    jb cmp_ret_neg
    ja cmp_ret_pos
    xor eax, eax
    jmp cmp_finish
cmp_ret_neg:
    mov eax, -1
    jmp cmp_finish
cmp_ret_pos:
    mov eax, 1
cmp_finish:
    pop r10
    pop r9
    pop r8
    ret

_start:
    # Phase 1: read all stdin into input_buf
    xor r12, r12
read_more:
    mov rdx, 32768
    sub rdx, r12
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more
read_done:

    # Phase 2: parse lines
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + input_buf]
scan_loop:
    cmp r13, r12
    jge scan_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x0A
    jne scan_skip
    lea rax, [rip + line_offsets]
    mov [rax + r15*8], r14
    lea rax, [rip + line_lengths]
    mov rdi, r13
    sub rdi, r14
    mov [rax + r15*8], rdi
    inc r15
    lea r14, [r13 + 1]
scan_skip:
    inc r13
    jmp scan_loop
scan_done:
    cmp r14, r12
    jae save_count
    lea rax, [rip + line_offsets]
    mov [rax + r15*8], r14
    lea rax, [rip + line_lengths]
    mov rdi, r12
    sub rdi, r14
    mov [rax + r15*8], rdi
    inc r15
save_count:
    lea rax, [rip + line_count_val]
    mov [rax], r15

    # Phase 3: insertion sort
    cmp r15, 1
    jle do_output
    mov r12, 1
isort_outer:
    cmp r12, r15
    jge do_output
    lea rax, [rip + line_offsets]
    mov r13, [rax + r12*8]
    lea rax, [rip + line_lengths]
    mov r14, [rax + r12*8]
    mov rbx, r12
    dec rbx
isort_inner:
    test rbx, rbx
    js isort_place
    lea rax, [rip + line_offsets]
    mov rdi, [rax + rbx*8]
    lea rax, [rip + line_lengths]
    mov rsi, [rax + rbx*8]
    mov rdx, r13
    mov rcx, r14
    call compare_lines
    test eax, eax
    jle isort_place
    lea rax, [rip + line_offsets]
    mov [rax + rbx*8 + 8], rdi
    lea rax, [rip + line_lengths]
    mov [rax + rbx*8 + 8], rsi
    dec rbx
    jmp isort_inner
isort_place:
    lea rax, [rip + line_offsets]
    mov [rax + rbx*8 + 8], r13
    lea rax, [rip + line_lengths]
    mov [rax + rbx*8 + 8], r14
    inc r12
    jmp isort_outer

    # Phase 4: output
do_output:
    xor r12, r12
out_loop:
    cmp r12, r15
    jge prog_done
    lea rax, [rip + line_lengths]
    mov rdx, [rax + r12*8]
    test rdx, rdx
    jz out_nl
    lea rax, [rip + line_offsets]
    mov rsi, [rax + r12*8]
    lea rax, [rip + input_buf]
    add rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall
out_nl:
    lea rsi, [rip + newline_byte]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r12
    jmp out_loop
prog_done:
    mov rax, 60
    xor rdi, rdi
    syscall
