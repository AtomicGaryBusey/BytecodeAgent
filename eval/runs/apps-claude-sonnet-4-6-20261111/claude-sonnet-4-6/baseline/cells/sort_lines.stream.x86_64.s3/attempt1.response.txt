.globl _start

.bss
.align 8
input_buf:
    .zero 32768
.align 8
line_ptrs:
    .zero 8008
.align 8
line_lens:
    .zero 8008
line_count:
    .zero 8

.section .rodata
nl_byte:
    .byte 0x0a

.text

# str_compare(rdi=ptr1, rsi=len1, rdx=ptr2, rcx=len2) -> rax
# negative if s1 < s2, 0 if equal, positive if s1 > s2
# clobbers r8, r9, r10
str_compare:
    xor r10d, r10d
    mov r8, rsi
    cmp r8, rcx
    jle scmp_loop
    mov r8, rcx
scmp_loop:
    cmp r10, r8
    jge scmp_check_len
    movzx eax, byte ptr [rdi + r10]
    movzx r9d, byte ptr [rdx + r10]
    cmp eax, r9d
    ja scmp_gt
    jb scmp_lt
    inc r10
    jmp scmp_loop
scmp_check_len:
    mov rax, rsi
    sub rax, rcx
    ret
scmp_gt:
    mov eax, 1
    ret
scmp_lt:
    mov rax, -1
    ret

_start:
    xor r12, r12

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12d, 32768
    jl read_more

read_done:
    test r12, r12
    jz sort_begin

    lea rbx, [rip + input_buf]
    xor r13, r13
    xor r14, r14

next_line:
    cmp r13, r12
    jge done_parse

    lea rax, [rip + line_ptrs]
    lea rcx, [rbx + r13]
    mov qword ptr [rax + r14*8], rcx

    mov r15, r13

scan_next:
    cmp r15, r12
    jge line_at_eof
    movzx ecx, byte ptr [rbx + r15]
    cmp cl, 0x0a
    je line_at_nl
    inc r15
    jmp scan_next

line_at_eof:
    lea rax, [rip + line_lens]
    mov rcx, r15
    sub rcx, r13
    mov qword ptr [rax + r14*8], rcx
    inc r14
    jmp done_parse

line_at_nl:
    lea rax, [rip + line_lens]
    mov rcx, r15
    sub rcx, r13
    mov qword ptr [rax + r14*8], rcx
    inc r14
    lea r13, [r15 + 1]
    cmp r14, 1000
    jl next_line

done_parse:
    mov qword ptr [rip + line_count], r14

sort_begin:
    mov r14, qword ptr [rip + line_count]
    cmp r14, 2
    jl output_start

    mov r13, 1

isort_outer:
    cmp r13, r14
    jge output_start

    lea rax, [rip + line_ptrs]
    mov r8, qword ptr [rax + r13*8]
    lea rax, [rip + line_lens]
    mov r9, qword ptr [rax + r13*8]

    mov r15, r13
    dec r15

isort_inner:
    test r15, r15
    js isort_place

    lea rax, [rip + line_ptrs]
    mov rdi, qword ptr [rax + r15*8]
    lea rax, [rip + line_lens]
    mov rsi, qword ptr [rax + r15*8]
    mov rdx, r8
    mov rcx, r9

    push r8
    push r9
    push r13
    push r14
    push r15
    call str_compare
    pop r15
    pop r14
    pop r13
    pop r9
    pop r8

    test rax, rax
    jle isort_place

    lea rax, [rip + line_ptrs]
    mov rbx, qword ptr [rax + r15*8]
    mov qword ptr [rax + r15*8 + 8], rbx

    lea rax, [rip + line_lens]
    mov rbx, qword ptr [rax + r15*8]
    mov qword ptr [rax + r15*8 + 8], rbx

    dec r15
    jmp isort_inner

isort_place:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r15*8 + 8], r8
    lea rax, [rip + line_lens]
    mov qword ptr [rax + r15*8 + 8], r9

    inc r13
    jmp isort_outer

output_start:
    mov r14, qword ptr [rip + line_count]
    xor r13, r13

write_loop:
    cmp r13, r14
    jge prog_exit

    lea rax, [rip + line_ptrs]
    mov r8, qword ptr [rax + r13*8]
    lea rax, [rip + line_lens]
    mov r9, qword ptr [rax + r13*8]

    test r9, r9
    jz write_nl_only

    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall

write_nl_only:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_byte]
    mov edx, 1
    syscall

    inc r13
    jmp write_loop

prog_exit:
    mov eax, 60
    xor edi, edi
    syscall
