.globl _start

.bss
.align 16
buffer:
    .zero 65536
out_buf:
    .zero 64

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 0x0A
    jne check_ws
    inc r12
check_ws:
    cmp al, 0x20
    je set_ws
    cmp al, 0x09
    jb set_nonws
    cmp al, 0x0D
    jbe set_ws
set_nonws:
    test r15, r15
    jnz process_loop
    inc r13
    mov r15d, 1
    jmp process_loop
set_ws:
    xor r15d, r15d
    jmp process_loop

done_reading:
    lea rdi, [rip + out_buf]
    mov rsi, r12
    call fmt_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r13
    call fmt_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r14
    call fmt_num
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + out_buf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fmt_num:
    test rsi, rsi
    jnz fmt_nonzero
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
fmt_nonzero:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    lea rcx, [rbp - 1]
    mov r8d, 10
    mov rax, rsi
fmt_loop:
    xor edx, edx
    div r8
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz fmt_loop
    lea r8, [rbp - 1]
fmt_copy:
    cmp rcx, r8
    jge fmt_done
    movzx eax, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    jmp fmt_copy
fmt_done:
    add rsp, 32
    pop rbp
    ret
