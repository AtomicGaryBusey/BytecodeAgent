.globl _start

.equ BUF_SIZE, 4096

.section .bss
.align 4
buf:    .zero BUF_SIZE
.align 4
outbuf: .zero 64

.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #BUF_SIZE
    svc #0
    cbz x0, done_reading
    adrp x1, buf
    add x1, x1, :lo12:buf

process_loop:
    cbz x0, read_loop
    ldrb w2, [x1], #1
    sub x0, x0, #1
    add x21, x21, #1
    cmp w2, #0x20
    beq byte_ws
    sub w3, w2, #0x09
    cmp w3, #4
    bls byte_ws
    cbz x22, new_word
    b process_loop

new_word:
    mov x22, #1
    add x20, x20, #1
    b process_loop

byte_ws:
    mov x22, #0
    cmp w2, #0x0A
    bne process_loop
    add x19, x19, #1
    b process_loop

done_reading:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x26, x25
    mov x0, x25
    mov x1, x19
    bl fmt_num
    mov x25, x0
    mov w3, #0x20
    strb w3, [x25], #1
    mov x0, x25
    mov x1, x20
    bl fmt_num
    mov x25, x0
    mov w3, #0x20
    strb w3, [x25], #1
    mov x0, x25
    mov x1, x21
    bl fmt_num
    mov x25, x0
    mov w3, #0x0A
    strb w3, [x25], #1
    sub x2, x25, x26
    mov x8, #64
    mov x0, #1
    mov x1, x26
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

fmt_num:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    cbz x1, fmt_zero
    add x3, sp, #16
    mov x2, #0

fmt_digit:
    cbz x1, fmt_rev
    mov x4, #10
    udiv x5, x1, x4
    msub x4, x5, x4, x1
    add x4, x4, #48
    strb w4, [x3, x2]
    add x2, x2, #1
    mov x1, x5
    b fmt_digit

fmt_rev:
    sub x2, x2, #1

fmt_copy:
    cmp x2, #0
    blt fmt_done
    ldrb w4, [x3, x2]
    strb w4, [x0], #1
    sub x2, x2, #1
    b fmt_copy

fmt_done:
    ldp x29, x30, [sp], #32
    ret

fmt_zero:
    mov w4, #48
    strb w4, [x0], #1
    ldp x29, x30, [sp], #32
    ret
