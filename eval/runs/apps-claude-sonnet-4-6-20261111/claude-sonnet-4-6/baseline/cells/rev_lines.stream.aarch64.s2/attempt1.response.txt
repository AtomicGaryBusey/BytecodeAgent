.globl _start

.bss
.align 4
input_buf:
    .zero 65536
output_buf:
    .zero 65538

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    mov x2, #65536
    cmp x20, x2
    blt read_loop

read_done:
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x21, #0

line_loop:
    cmp x21, x20
    bge exit_ok

    mov x22, x21

search_nl:
    cmp x22, x20
    bge after_search
    ldrb w0, [x19, x22]
    cmp w0, #0x0A
    beq after_search
    add x22, x22, #1
    b search_nl

after_search:
    sub x23, x22, x21
    cbz x23, write_nl_only
    sub x26, x22, #1
    mov x25, #0

rev_loop:
    ldrb w0, [x19, x26]
    strb w0, [x24, x25]
    add x25, x25, #1
    cmp x26, x21
    beq write_nl
    sub x26, x26, #1
    b rev_loop

write_nl_only:
    mov x25, #0

write_nl:
    mov w0, #0x0A
    strb w0, [x24, x25]
    add x25, x25, #1

    mov x8, #64
    mov x0, #1
    mov x1, x24
    mov x2, x25
    svc #0

    cmp x22, x20
    bge exit_ok
    add x21, x22, #1
    b line_loop

exit_ok:
    mov x8, #93
    mov x0, #0
    svc #0
