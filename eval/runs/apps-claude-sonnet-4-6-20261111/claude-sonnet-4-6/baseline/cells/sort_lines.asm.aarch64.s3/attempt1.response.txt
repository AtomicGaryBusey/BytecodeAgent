        .globl _start

        .section .rodata
newline:
        .byte   0x0A

        .section .bss
        .align  8
input_buf:
        .zero   32768
        .align  8
line_ptrs:
        .zero   8000
        .align  8
line_lens:
        .zero   8000

        .text

_start:
        adrp    x26, line_ptrs
        add     x26, x26, :lo12:line_ptrs
        adrp    x27, line_lens
        add     x27, x27, :lo12:line_lens
        adrp    x28, input_buf
        add     x28, x28, :lo12:input_buf

        mov     x19, #0
        mov     x21, #0

read_loop:
        mov     x0, #0
        add     x1, x28, x19
        mov     x2, #32768
        sub     x2, x2, x19
        cbz     x2, read_done
        mov     x8, #63
        svc     #0
        cbz     x0, read_done
        add     x19, x19, x0
        b       read_loop

read_done:
        cbz     x19, print_sorted

        mov     x20, #0
        mov     x22, x28

parse_loop:
        cmp     x20, x19
        bge     parse_final

        ldrb    w0, [x28, x20]
        add     x20, x20, #1
        cmp     w0, #0x0A
        bne     parse_loop

        add     x0, x28, x20
        sub     x0, x0, #1
        sub     x0, x0, x22

        str     x22, [x26, x21, lsl #3]
        str     x0,  [x27, x21, lsl #3]
        add     x21, x21, #1
        add     x22, x28, x20
        b       parse_loop

parse_final:
        add     x0, x28, x19
        sub     x0, x0, x22
        cbz     x0, do_sort

        str     x22, [x26, x21, lsl #3]
        str     x0,  [x27, x21, lsl #3]
        add     x21, x21, #1

do_sort:
        cmp     x21, #1
        ble     print_sorted

        mov     x20, #1

outer_loop:
        cmp     x20, x21
        bge     print_sorted

        ldr     x23, [x26, x20, lsl #3]
        ldr     x24, [x27, x20, lsl #3]
        sub     x25, x20, #1

inner_loop:
        cmp     x25, #0
        blt     do_insert

        ldr     x0, [x26, x25, lsl #3]
        ldr     x1, [x27, x25, lsl #3]
        mov     x2, x23
        mov     x3, x24
        bl      compare_lines

        cmp     x0, #0
        ble     do_insert

        add     x0, x25, #1
        ldr     x1, [x26, x25, lsl #3]
        str     x1, [x26, x0, lsl #3]
        ldr     x1, [x27, x25, lsl #3]
        str     x1, [x27, x0, lsl #3]

        sub     x25, x25, #1
        b       inner_loop

do_insert:
        add     x0, x25, #1
        str     x23, [x26, x0, lsl #3]
        str     x24, [x27, x0, lsl #3]
        add     x20, x20, #1
        b       outer_loop

print_sorted:
        cbz     x21, exit_prog
        mov     x20, #0

print_loop:
        cmp     x20, x21
        bge     exit_prog

        ldr     x1, [x26, x20, lsl #3]
        ldr     x2, [x27, x20, lsl #3]

        cbz     x2, write_nl

        mov     x0, #1
        mov     x8, #64
        svc     #0

write_nl:
        adrp    x1, newline
        add     x1, x1, :lo12:newline
        mov     x0, #1
        mov     x2, #1
        mov     x8, #64
        svc     #0

        add     x20, x20, #1
        b       print_loop

exit_prog:
        mov     x0, #0
        mov     x8, #93
        svc     #0

compare_lines:
        cmp     x1, x3
        csel    x4, x1, x3, lo
        mov     x5, #0

cmp_byte_loop:
        cmp     x5, x4
        bge     cmp_prefix

        ldrb    w6, [x0, x5]
        ldrb    w7, [x2, x5]
        cmp     w6, w7
        bne     cmp_diff

        add     x5, x5, #1
        b       cmp_byte_loop

cmp_diff:
        sub     w0, w6, w7
        sxtw    x0, w0
        ret

cmp_prefix:
        sub     x0, x1, x3
        ret
