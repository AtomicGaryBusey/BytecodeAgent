.globl _start

.section .bss
.align 4
input_buf:
    .zero 65536
output_buf:
    .zero 65538

.text
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, #0

read_loop:
    mov  x8, #63
    mov  x0, #0
    add  x1, x19, x20
    mov  x2, #65536
    sub  x2, x2, x20
    cbz  x2, done_read
    svc  #0
    cbz  x0, done_read
    add  x20, x20, x0
    b    read_loop

done_read:
    mov  x21, #0

process_lines:
    cmp  x21, x20
    beq  exit_program
    mov  x22, x21

find_newline:
    cmp  x21, x20
    beq  no_trailing_newline
    ldrb w3, [x19, x21]
    cmp  w3, #0x0A
    beq  found_newline
    add  x21, x21, #1
    b    find_newline

found_newline:
    sub  x23, x21, x22
    add  x21, x21, #1
    b    reverse_and_print

no_trailing_newline:
    sub  x23, x20, x22

reverse_and_print:
    adrp x24, output_buf
    add  x24, x24, :lo12:output_buf
    mov  x25, #0

reverse_loop:
    cmp  x25, x23
    beq  write_line
    sub  x26, x23, #1
    sub  x26, x26, x25
    add  x26, x26, x22
    ldrb w3, [x19, x26]
    strb w3, [x24, x25]
    add  x25, x25, #1
    b    reverse_loop

write_line:
    mov  w3, #0x0A
    strb w3, [x24, x23]
    mov  x8, #64
    mov  x0, #1
    mov  x1, x24
    add  x2, x23, #1
    svc  #0
    b    process_lines

exit_program:
    mov  x8, #93
    mov  x0, #0
    svc  #0
