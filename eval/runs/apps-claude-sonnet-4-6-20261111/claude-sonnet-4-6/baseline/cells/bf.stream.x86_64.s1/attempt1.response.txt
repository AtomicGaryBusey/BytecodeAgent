.globl _start

.bss
.align 16
stdin_buf:
    .zero 65536
.align 16
tape:
    .zero 30000
.align 16
out_buf:
    .zero 65536
.align 16
jmp_tbl:
    .zero 262144
.align 16
brk_stk:
    .zero 262144

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle read_done
    syscall
    test eax, eax
    jle read_done
    add r12d, eax
    jmp read_more

read_done:
    xor ecx, ecx
    lea rax, [rip + stdin_buf]

scan_bang:
    cmp ecx, r12d
    jge no_bang
    cmp byte ptr [rax + rcx], 0x21
    je found_bang
    inc ecx
    jmp scan_bang

found_bang:
    mov r13d, ecx
    mov r14d, ecx
    inc r14d
    jmp do_validate

no_bang:
    mov r13d, r12d
    mov r14d, r12d

do_validate:
    xor ecx, ecx
    xor ebx, ebx
    lea rsi, [rip + stdin_buf]
    lea rdi, [rip + brk_stk]
    lea rbp, [rip + jmp_tbl]

val_loop:
    cmp ecx, r13d
    jge val_done
    movzx edx, byte ptr [rsi + rcx]
    cmp dl, 0x5B
    je open_brk
    cmp dl, 0x5D
    je close_brk
    inc ecx
    jmp val_loop

open_brk:
    mov dword ptr [rdi + rbx*4], ecx
    inc ebx
    inc ecx
    jmp val_loop

close_brk:
    test ebx, ebx
    jle err_exit
    dec ebx
    mov edx, dword ptr [rdi + rbx*4]
    mov dword ptr [rbp + rdx*4], ecx
    mov dword ptr [rbp + rcx*4], edx
    inc ecx
    jmp val_loop

val_done:
    test ebx, ebx
    jnz err_exit

    lea rbx, [rip + stdin_buf]
    lea r9, [rip + tape]
    lea r10, [rip + jmp_tbl]
    lea r11, [rip + out_buf]
    xor r15d, r15d
    xor esi, esi
    xor ecx, ecx

exec_loop:
    cmp ecx, r13d
    jge exec_done
    movzx edx, byte ptr [rbx + rcx]
    cmp dl, 0x3E
    je cmd_gt
    cmp dl, 0x3C
    je cmd_lt
    cmp dl, 0x2B
    je cmd_plus
    cmp dl, 0x2D
    je cmd_minus
    cmp dl, 0x2E
    je cmd_dot
    cmp dl, 0x2C
    je cmd_comma
    cmp dl, 0x5B
    je cmd_lbr
    cmp dl, 0x5D
    je cmd_rbr
    inc ecx
    jmp exec_loop

cmd_gt:
    inc esi
    inc ecx
    jmp exec_loop

cmd_lt:
    dec esi
    inc ecx
    jmp exec_loop

cmd_plus:
    inc byte ptr [r9 + rsi]
    inc ecx
    jmp exec_loop

cmd_minus:
    dec byte ptr [r9 + rsi]
    inc ecx
    jmp exec_loop

cmd_dot:
    movzx eax, byte ptr [r9 + rsi]
    mov byte ptr [r11 + r15], al
    inc r15d
    inc ecx
    jmp exec_loop

cmd_comma:
    cmp r14d, r12d
    jge comma_zero
    movzx eax, byte ptr [rbx + r14]
    mov byte ptr [r9 + rsi], al
    inc r14d
    inc ecx
    jmp exec_loop

comma_zero:
    mov byte ptr [r9 + rsi], 0
    inc ecx
    jmp exec_loop

cmd_lbr:
    cmp byte ptr [r9 + rsi], 0
    jne lbr_enter
    mov ecx, dword ptr [r10 + rcx*4]
    inc ecx
    jmp exec_loop
lbr_enter:
    inc ecx
    jmp exec_loop

cmd_rbr:
    cmp byte ptr [r9 + rsi], 0
    je rbr_exit
    mov ecx, dword ptr [r10 + rcx*4]
    inc ecx
    jmp exec_loop
rbr_exit:
    inc ecx
    jmp exec_loop

exec_done:
    test r15d, r15d
    jz ok_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, r15d
    syscall

ok_exit:
    mov eax, 60
    xor edi, edi
    syscall

err_exit:
    mov eax, 60
    mov edi, 1
    syscall
