.globl _start

.bss
.align 16
stdin_buf:
    .zero 65536
.align 16
tape:
    .zero 30000
.align 16
output_buf:
    .zero 65536
.align 4
jump_table:
    .zero 262144
.align 4
bracket_stack:
    .zero 4000

.text
_start:
    xor r12d, r12d

.Lread_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    cmp r12d, 65536
    jl .Lread_more

.Lread_done:
    lea r13, [rip + stdin_buf]
    xor ecx, ecx

.Lfind_excl:
    cmp rcx, r12
    jge .Lno_excl
    cmp byte ptr [r13 + rcx], 0x21
    je .Lfound_excl
    inc rcx
    jmp .Lfind_excl

.Lfound_excl:
    mov r15, rcx
    lea r14, [rcx + 1]
    jmp .Ldo_validate

.Lno_excl:
    mov r15, r12
    mov r14, r12

.Ldo_validate:
    lea rbx, [rip + bracket_stack]
    xor r9d, r9d
    xor ecx, ecx
    lea rdx, [rip + jump_table]

.Lval_loop:
    cmp rcx, r15
    jge .Lval_done
    movzx eax, byte ptr [r13 + rcx]
    cmp al, 0x5B
    je .Lval_open
    cmp al, 0x5D
    je .Lval_close
    inc rcx
    jmp .Lval_loop

.Lval_open:
    cmp r9d, 1000
    jge .Lerr_exit
    mov dword ptr [rbx + r9*4], ecx
    inc r9d
    inc rcx
    jmp .Lval_loop

.Lval_close:
    test r9d, r9d
    jz .Lerr_exit
    dec r9d
    mov eax, dword ptr [rbx + r9*4]
    mov dword ptr [rdx + rax*4], ecx
    mov dword ptr [rdx + rcx*4], eax
    inc rcx
    jmp .Lval_loop

.Lval_done:
    test r9d, r9d
    jnz .Lerr_exit

    xor r10d, r10d
    xor r11d, r11d
    xor r8d, r8d
    lea rbp, [rip + tape]
    lea r9, [rip + output_buf]

.Lbf_loop:
    cmp r10, r15
    jge .Lbf_done
    movzx eax, byte ptr [r13 + r10]
    cmp al, 0x3E
    je .Lcmd_right
    cmp al, 0x3C
    je .Lcmd_left
    cmp al, 0x2B
    je .Lcmd_plus
    cmp al, 0x2D
    je .Lcmd_minus
    cmp al, 0x2E
    je .Lcmd_dot
    cmp al, 0x2C
    je .Lcmd_comma
    cmp al, 0x5B
    je .Lcmd_open
    cmp al, 0x5D
    je .Lcmd_close
    inc r10
    jmp .Lbf_loop

.Lcmd_right:
    inc r11
    inc r10
    jmp .Lbf_loop

.Lcmd_left:
    dec r11
    inc r10
    jmp .Lbf_loop

.Lcmd_plus:
    inc byte ptr [rbp + r11]
    inc r10
    jmp .Lbf_loop

.Lcmd_minus:
    dec byte ptr [rbp + r11]
    inc r10
    jmp .Lbf_loop

.Lcmd_dot:
    movzx eax, byte ptr [rbp + r11]
    mov byte ptr [r9 + r8], al
    inc r8
    inc r10
    jmp .Lbf_loop

.Lcmd_comma:
    cmp r14, r12
    jge .Lcomma_eof
    movzx eax, byte ptr [r13 + r14]
    inc r14
    mov byte ptr [rbp + r11], al
    inc r10
    jmp .Lbf_loop

.Lcomma_eof:
    mov byte ptr [rbp + r11], 0
    inc r10
    jmp .Lbf_loop

.Lcmd_open:
    cmp byte ptr [rbp + r11], 0
    jnz .Lopen_nonzero
    mov eax, dword ptr [rdx + r10*4]
    mov r10d, eax
    inc r10
    jmp .Lbf_loop

.Lopen_nonzero:
    inc r10
    jmp .Lbf_loop

.Lcmd_close:
    cmp byte ptr [rbp + r11], 0
    jz .Lclose_zero
    mov eax, dword ptr [rdx + r10*4]
    mov r10d, eax
    inc r10
    jmp .Lbf_loop

.Lclose_zero:
    inc r10
    jmp .Lbf_loop

.Lbf_done:
    test r8, r8
    jz .Lok_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    syscall

.Lok_exit:
    mov eax, 60
    xor edi, edi
    syscall

.Lerr_exit:
    mov eax, 60
    mov edi, 1
    syscall
