.globl _start

.bss
.balign 4
input_buf:     .zero 65536
.balign 4
tape:          .zero 30000
.balign 4
jump_table:    .zero 262144
.balign 4
bracket_stack: .zero 4000

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x28, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x28
    mov x2, #65536
    sub x2, x2, x28
    cbz x2, read_done
    svc #0
    cbz x0, read_done
    add x28, x28, x0
    b read_loop

read_done:
    mov x1, #0

find_sep:
    cmp x1, x28
    b.ge no_sep
    ldrb w2, [x19, x1]
    cmp w2, #0x21
    b.eq found_sep
    add x1, x1, #1
    b find_sep

no_sep:
    mov x20, x28
    mov x24, x28
    mov x25, x28
    b do_brackets

found_sep:
    mov x20, x1
    add x24, x1, #1
    mov x25, x28

do_brackets:
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x27, bracket_stack
    add x27, x27, :lo12:bracket_stack
    mov x0, #0
    mov x1, #0

scan_loop:
    cmp x1, x20
    b.ge scan_done
    ldrb w2, [x19, x1]
    cmp w2, #0x5B
    b.eq do_open
    cmp w2, #0x5D
    b.eq do_close
    add x1, x1, #1
    b scan_loop

do_open:
    str w1, [x27, x0, lsl #2]
    add x0, x0, #1
    add x1, x1, #1
    b scan_loop

do_close:
    cbz x0, exit_bad
    sub x0, x0, #1
    ldr w3, [x27, x0, lsl #2]
    add w4, w1, #1
    str w4, [x26, x3, lsl #2]
    add w4, w3, #1
    str w4, [x26, x1, lsl #2]
    add x1, x1, #1
    b scan_loop

scan_done:
    cbnz x0, exit_bad

    adrp x21, tape
    add x21, x21, :lo12:tape
    mov x22, #0
    mov x23, #0

exec_loop:
    cmp x23, x20
    b.ge exit_ok
    ldrb w0, [x19, x23]

    cmp w0, #0x3E
    b.eq cmd_gt
    cmp w0, #0x3C
    b.eq cmd_lt
    cmp w0, #0x2B
    b.eq cmd_plus
    cmp w0, #0x2D
    b.eq cmd_minus
    cmp w0, #0x2E
    b.eq cmd_dot
    cmp w0, #0x2C
    b.eq cmd_comma
    cmp w0, #0x5B
    b.eq cmd_lbr
    cmp w0, #0x5D
    b.eq cmd_rbr
    add x23, x23, #1
    b exec_loop

cmd_gt:
    add x22, x22, #1
    add x23, x23, #1
    b exec_loop

cmd_lt:
    sub x22, x22, #1
    add x23, x23, #1
    b exec_loop

cmd_plus:
    ldrb w0, [x21, x22]
    add w0, w0, #1
    strb w0, [x21, x22]
    add x23, x23, #1
    b exec_loop

cmd_minus:
    ldrb w0, [x21, x22]
    sub w0, w0, #1
    strb w0, [x21, x22]
    add x23, x23, #1
    b exec_loop

cmd_dot:
    add x1, x21, x22
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    add x23, x23, #1
    b exec_loop

cmd_comma:
    cmp x24, x25
    b.ge cmd_comma_eof
    ldrb w0, [x19, x24]
    strb w0, [x21, x22]
    add x24, x24, #1
    add x23, x23, #1
    b exec_loop

cmd_comma_eof:
    strb wzr, [x21, x22]
    add x23, x23, #1
    b exec_loop

cmd_lbr:
    ldrb w0, [x21, x22]
    cbnz w0, cmd_lbr_nz
    ldr w23, [x26, x23, lsl #2]
    b exec_loop
cmd_lbr_nz:
    add x23, x23, #1
    b exec_loop

cmd_rbr:
    ldrb w0, [x21, x22]
    cbz w0, cmd_rbr_z
    ldr w23, [x26, x23, lsl #2]
    b exec_loop
cmd_rbr_z:
    add x23, x23, #1
    b exec_loop

exit_ok:
    mov x8, #93
    mov x0, #0
    svc #0

exit_bad:
    mov x8, #93
    mov x0, #1
    svc #0
