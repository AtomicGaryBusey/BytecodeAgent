.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .zero 65536
.align 4
line_buf:
    .zero 128

.text

_start:
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits
    mov x19, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0

row_loop:
    cmp x20, x19
    bge done_rows
    sub x21, x19, x20
    cmp x21, #16
    ble 1f
    mov x21, #16
1:
    adrp x26, line_buf
    add x26, x26, :lo12:line_buf
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    add x23, x23, x20

    // 8 hex digit offset
    mov x0, x20
    mov x4, #7
offset_loop:
    lsl x3, x4, #2
    lsr x2, x0, x3
    and x2, x2, #0xf
    ldrb w3, [x25, x2]
    strb w3, [x26], #1
    cbz x4, offset_done
    sub x4, x4, #1
    b offset_loop
offset_done:

    // two spaces
    mov w3, #0x20
    strb w3, [x26], #1
    strb w3, [x26], #1

    // hex bytes loop i=0..15
    mov x24, #0
hex_loop:
    cmp x24, #16
    bge hex_done
    // extra space before byte 8
    cmp x24, #8
    bne 2f
    mov w3, #0x20
    strb w3, [x26], #1
2:
    cmp x24, x21
    bge hex_pad
    // print "XX "
    ldrb w0, [x23, x24]
    lsr w2, w0, #4
    ldrb w3, [x25, x2]
    strb w3, [x26], #1
    and w2, w0, #0xf
    ldrb w3, [x25, x2]
    strb w3, [x26], #1
    mov w3, #0x20
    strb w3, [x26], #1
    b hex_next
hex_pad:
    // print "   "
    mov w3, #0x20
    strb w3, [x26], #1
    strb w3, [x26], #1
    strb w3, [x26], #1
hex_next:
    add x24, x24, #1
    b hex_loop
hex_done:

    // " |"
    mov w3, #0x20
    strb w3, [x26], #1
    mov w3, #0x7c
    strb w3, [x26], #1

    // ASCII loop
    mov x24, #0
ascii_loop:
    cmp x24, x21
    bge ascii_done
    ldrb w0, [x23, x24]
    cmp w0, #0x20
    blt ascii_dot
    cmp w0, #0x7e
    bgt ascii_dot
    strb w0, [x26], #1
    b ascii_next
ascii_dot:
    mov w3, #0x2e
    strb w3, [x26], #1
ascii_next:
    add x24, x24, #1
    b ascii_loop
ascii_done:

    // "|\n"
    mov w3, #0x7c
    strb w3, [x26], #1
    mov w3, #0x0a
    strb w3, [x26], #1

    // write line
    mov x8, #64
    mov x0, #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    sub x2, x26, x1
    svc #0

    add x20, x20, #16
    b row_loop

done_rows:
    // print final offset
    adrp x26, line_buf
    add x26, x26, :lo12:line_buf
    mov x0, x19
    mov x4, #7
final_loop:
    lsl x3, x4, #2
    lsr x2, x0, x3
    and x2, x2, #0xf
    ldrb w3, [x25, x2]
    strb w3, [x26], #1
    cbz x4, final_done
    sub x4, x4, #1
    b final_loop
final_done:
    mov w3, #0x0a
    strb w3, [x26], #1

    mov x8, #64
    mov x0, #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    sub x2, x26, x1
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
