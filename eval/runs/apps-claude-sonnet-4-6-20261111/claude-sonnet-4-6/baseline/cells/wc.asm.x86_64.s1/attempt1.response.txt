.globl _start

.section .bss
    .align 16
buffer: .zero 65536
outbuf: .zero 64

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_input
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

byte_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 0x0A
    je byte_nl
    cmp al, 0x20
    je byte_ws
    cmp al, 0x09
    jb byte_word
    cmp al, 0x0D
    jbe byte_ws

byte_word:
    test r15, r15
    jnz byte_loop
    inc r13
    mov r15d, 1
    jmp byte_loop

byte_nl:
    inc r12
byte_ws:
    xor r15, r15
    jmp byte_loop

done_input:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call fmt_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call fmt_uint
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fmt_uint:
    push rbx
    sub rsp, 24
    test rax, rax
    jnz fu_nonzero
    mov byte ptr [rdi], 0x30
    inc rdi
    add rsp, 24
    pop rbx
    ret

fu_nonzero:
    lea rbx, [rsp + 20]
    xor ecx, ecx
    mov r8d, 10

fu_loop:
    xor rdx, rdx
    div r8
    add dl, 0x30
    dec rbx
    mov byte ptr [rbx], dl
    inc ecx
    test rax, rax
    jnz fu_loop

fu_copy:
    test ecx, ecx
    jz fu_end
    movzx eax, byte ptr [rbx]
    mov byte ptr [rdi], al
    inc rbx
    inc rdi
    dec ecx
    jmp fu_copy

fu_end:
    add rsp, 24
    pop rbx
    ret
