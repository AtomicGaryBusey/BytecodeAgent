.globl _start

.section .rodata
newline:
    .byte 0x0A

.bss
.align 16
input_buf:
    .zero 65536

.text
_start:
    lea r15, [rip + input_buf]
    xor r12d, r12d

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov eax, 0
    mov edi, 0
    mov rsi, r15
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d

process_lines:
    cmp r13, r12
    jge exit_prog
    mov r14, r13

find_newline:
    cmp r14, r12
    jge at_eof
    movzx eax, byte ptr [r15 + r14]
    cmp al, 0x0A
    je found_newline
    inc r14
    jmp find_newline

found_newline:
    mov rbx, r13
    mov rcx, r14
    dec rcx
    jmp do_reverse

at_eof:
    mov rbx, r13
    mov rcx, r12
    dec rcx

do_reverse:
reverse_loop:
    cmp rbx, rcx
    jge reverse_done
    movzx eax, byte ptr [r15 + rbx]
    movzx edx, byte ptr [r15 + rcx]
    mov byte ptr [r15 + rbx], dl
    mov byte ptr [r15 + rcx], al
    inc rbx
    dec rcx
    jmp reverse_loop

reverse_done:
    mov rdx, r14
    sub rdx, r13
    jz write_nl
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    add rsi, r13
    syscall

write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    cmp r14, r12
    jge exit_prog
    lea r13, [r14 + 1]
    jmp process_lines

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
