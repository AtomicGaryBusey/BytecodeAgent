.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 8
input_buf:
    .zero 49152
.align 8
output_buf:
    .zero 70000

.text
_start:
    xor rbx, rbx

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 49152
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buf]
    lea rdi, [rip + output_buf]
    lea r10, [rip + b64_table]
    mov rcx, rbx
    mov rbx, rdi
    xor r8d, r8d

encode_loop:
    cmp rcx, 3
    jl encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 8
    or al, byte ptr [rsi + 1]
    shl eax, 8
    or al, byte ptr [rsi + 2]
    add rsi, 3
    sub rcx, 3

    mov r9d, eax
    shr r9d, 18
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 1], r9b

    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 2], r9b

    mov r9d, eax
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 3], r9b

    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jl encode_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp encode_loop

encode_tail:
    test rcx, rcx
    jnz has_tail

    test r8d, r8d
    jz encode_done
    mov byte ptr [rdi], 10
    inc rdi
    jmp encode_done

has_tail:
    cmp rcx, 1
    je tail1

    movzx eax, byte ptr [rsi]
    shl eax, 8
    or al, byte ptr [rsi + 1]

    mov r9d, eax
    shr r9d, 10
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shr r9d, 4
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 1], r9b

    mov r9d, eax
    shl r9d, 2
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 2], r9b

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    jmp do_newline

tail1:
    movzx eax, byte ptr [rsi]

    mov r9d, eax
    shr r9d, 2
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi], r9b

    mov r9d, eax
    shl r9d, 4
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov byte ptr [rdi + 1], r9b

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4

do_newline:
    mov byte ptr [rdi], 10
    inc rdi

encode_done:
    mov rdx, rdi
    sub rdx, rbx
    test rdx, rdx
    jz exit_prog

    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall

exit_prog:
    mov rax, 60
    xor edi, edi
    syscall
