.globl _start

.section .rodata
nl:
    .byte 10

.section .bss
.align 8
inbuf:
    .zero 32768
.align 8
lptr:
    .zero 8000
.align 8
llen:
    .zero 8000

.text

_start:
    xor r12d, r12d

.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    jle .Lrdone
    syscall
    test rax, rax
    jle .Lrdone
    add r12, rax
    jmp .Lread

.Lrdone:
    lea rbx, [rip + inbuf]
    xor r13, r13
    xor r14, r14
    xor r15, r15

.Lparse:
    cmp r13, r12
    jge .Lpend
    movzx eax, byte ptr [rbx + r13]
    cmp al, 10
    jne .Lpnxt
    lea rcx, [rip + lptr]
    lea rax, [rbx + r15]
    mov qword ptr [rcx + r14*8], rax
    lea rcx, [rip + llen]
    mov rax, r13
    sub rax, r15
    mov qword ptr [rcx + r14*8], rax
    inc r14
    inc r13
    mov r15, r13
    jmp .Lparse

.Lpnxt:
    inc r13
    jmp .Lparse

.Lpend:
    cmp r15, r12
    jge .Lnofin
    lea rcx, [rip + lptr]
    lea rax, [rbx + r15]
    mov qword ptr [rcx + r14*8], rax
    lea rcx, [rip + llen]
    mov rax, r12
    sub rax, r15
    mov qword ptr [rcx + r14*8], rax
    inc r14

.Lnofin:
    mov r8, 1

.Lisort:
    cmp r8, r14
    jge .Lsorted
    mov r9, r8

.Liinner:
    test r9, r9
    jz .Librk

    lea rbx, [rip + lptr]
    lea rcx, [rip + llen]
    mov rdx, r9
    dec rdx

    mov rdi, qword ptr [rbx + rdx*8]
    mov rsi, qword ptr [rbx + r9*8]
    mov r10, qword ptr [rcx + rdx*8]
    mov r11, qword ptr [rcx + r9*8]

    mov rax, r10
    cmp rax, r11
    jbe .Lcmploop
    mov rax, r11

.Lcmploop:
    xor rbx, rbx

.Lcmpbyte:
    cmp rbx, rax
    jge .Lcmppfx
    movzx ecx, byte ptr [rdi + rbx]
    movzx edx, byte ptr [rsi + rbx]
    cmp cl, dl
    ja .Lswap
    jb .Librk
    inc rbx
    jmp .Lcmpbyte

.Lcmppfx:
    cmp r10, r11
    ja .Lswap

.Librk:
    inc r8
    jmp .Lisort

.Lswap:
    lea rbx, [rip + lptr]
    lea rcx, [rip + llen]
    mov rdx, r9
    dec rdx

    mov rdi, qword ptr [rbx + rdx*8]
    mov rsi, qword ptr [rbx + r9*8]
    mov qword ptr [rbx + rdx*8], rsi
    mov qword ptr [rbx + r9*8], rdi

    mov rdi, qword ptr [rcx + rdx*8]
    mov rsi, qword ptr [rcx + r9*8]
    mov qword ptr [rcx + rdx*8], rsi
    mov qword ptr [rcx + r9*8], rdi

    dec r9
    jmp .Liinner

.Lsorted:
    xor r13, r13

.Lout:
    cmp r13, r14
    jge .Lexit

    lea rbx, [rip + lptr]
    lea rcx, [rip + llen]
    mov rsi, qword ptr [rbx + r13*8]
    mov rdx, qword ptr [rcx + r13*8]

    test rdx, rdx
    jz .Loutnl

    mov eax, 1
    mov edi, 1
    syscall

.Loutnl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall

    inc r13
    jmp .Lout

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall
