.globl _start

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 64
tmpbuf:
    .zero 32

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 0x0A
    je is_newline
    cmp al, 0x09
    jl is_word_char
    cmp al, 0x0D
    jle is_whitespace
    cmp al, 0x20
    je is_whitespace
is_word_char:
    test r15, r15
    jnz process_loop
    inc r13
    mov r15, 1
    jmp process_loop
is_newline:
    inc r12
is_whitespace:
    xor r15, r15
    jmp process_loop

done:
    lea rbx, [rip + outbuf]
    mov rax, r12
    call format_num
    mov byte ptr [rbx], 0x20
    inc rbx
    mov rax, r13
    call format_num
    mov byte ptr [rbx], 0x20
    inc rbx
    mov rax, r14
    call format_num
    mov byte ptr [rbx], 0x0A
    inc rbx
    lea rsi, [rip + outbuf]
    sub rbx, rsi
    mov rax, 1
    mov rdi, 1
    mov rdx, rbx
    lea rsi, [rip + outbuf]
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

format_num:
    test rax, rax
    jnz 1f
    mov byte ptr [rbx], 0x30
    inc rbx
    ret
1:
    lea rdi, [rip + tmpbuf]
    xor rcx, rcx
2:
    test rax, rax
    jz 3f
    xor rdx, rdx
    mov r8, 10
    div r8
    add dl, 0x30
    mov byte ptr [rdi + rcx], dl
    inc rcx
    jmp 2b
3:
4:
    dec rcx
    movzx eax, byte ptr [rdi + rcx]
    mov byte ptr [rbx], al
    inc rbx
    test rcx, rcx
    jnz 4b
    ret
