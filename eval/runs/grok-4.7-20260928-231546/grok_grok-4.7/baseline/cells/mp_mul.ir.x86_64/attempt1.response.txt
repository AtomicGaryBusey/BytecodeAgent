export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero
@zero
	%zi =l phi @start 0, @zerost %zi1
	%zd =w ceql %zi, %n
	jnz %zd, @iloop, @zerost
@zerost
	%z8 =l mul %zi, 8
	%zp =l add %r, %z8
	storel 0, %zp
	%zi1 =l add %zi, 1
	jmp @zero
@iloop
	%i =l phi @zero 0, @jend %i1
	%id =w ceql %i, %na
	jnz %id, @trim, @jloop
@jloop
	%j =l phi @iloop 0, @jbody %j1
	%carry =l phi @iloop 0, @jbody %carry1
	%jd =w ceql %j, %nb
	jnz %jd, @jend, @jbody
@jbody
	%ia =l mul %i, 8
	%ap =l add %a, %ia
	%av =l loadl %ap
	%jb =l mul %j, 8
	%bp =l add %b, %jb
	%bv =l loadl %bp
	%alo =l and %av, 4294967295
	%ahi =l shr %av, 32
	%blo =l and %bv, 4294967295
	%bhi =l shr %bv, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%mid =l add %p1, %p2
	%midc =w cultl %mid, %p1
	%midlo =l shl %mid, 32
	%plo =l add %p0, %midlo
	%ploc =w cultl %plo, %p0
	%midhi =l shr %mid, 32
	%midcl =l extuw %midc
	%midcs =l shl %midcl, 32
	%plocl =l extuw %ploc
	%h0 =l add %p3, %midhi
	%h1 =l add %h0, %midcs
	%hi =l add %h1, %plocl
	%ij =l add %i, %j
	%ro =l mul %ij, 8
	%rp =l add %r, %ro
	%rv =l loadl %rp
	%s1 =l add %rv, %plo
	%c1 =w cultl %s1, %rv
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	storel %s2, %rp
	%c12 =w add %c1, %c2
	%c12l =l extuw %c12
	%carry1 =l add %hi, %c12l
	%j1 =l add %j, 1
	jmp @jloop
@jend
	%inb =l add %i, %nb
	%eo =l mul %inb, 8
	%ep =l add %r, %eo
	storel %carry, %ep
	%i1 =l add %i, 1
	jmp @iloop
@trim
	%k =l phi @iloop %n, @trimdec %k1
	%kz =w ceql %k, 0
	jnz %kz, @ret, @trimchk
@trimchk
	%km =l sub %k, 1
	%ko =l mul %km, 8
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%kvz =w ceql %kv, 0
	jnz %kvz, @trimdec, @ret
@trimdec
	%k1 =l sub %k, 1
	jmp @trim
@ret
	ret %k
}
