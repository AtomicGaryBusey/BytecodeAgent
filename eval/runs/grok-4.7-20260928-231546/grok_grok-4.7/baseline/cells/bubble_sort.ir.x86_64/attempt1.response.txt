export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @oinc %i1
	%ic =w csltl %i, %n
	jnz %ic, @obody, @end
@obody
	%n1 =l sub %n, 1
	%lim =l sub %n1, %i
	jmp @inner
@inner
	%j =l phi @obody 0, @jinc %j1
	%jc =w csltl %j, %lim
	jnz %jc, @ibody, @oinc
@ibody
	%off =l mul %j, 8
	%p =l add %a, %off
	%v =l loadl %p
	%jn =l add %j, 1
	%off2 =l mul %jn, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%gt =w csgtl %v, %v2
	jnz %gt, @swap, @jinc
@swap
	storel %v2, %p
	storel %v, %p2
	jmp @jinc
@jinc
	%j1 =l add %j, 1
	jmp @inner
@oinc
	%i1 =l add %i, 1
	jmp @outer
@end
	ret 0
}
