qsort_pairs:
cmp rsi, 1
jg .Lsort
xor eax, eax
ret
.Lsort:
push rbx
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
lea r15, [rsi - 1]
xor r13d, r13d
xor r14d, r14d
.Lloop:
cmp r14, r15
jge .Lpart_done
mov rax, r14
shl rax, 4
mov rcx, r15
shl rcx, 4
mov rdx, qword ptr [rbx + rax]
cmp rdx, qword ptr [rbx + rcx]
jl .Lless
jg .Lge
mov rdx, qword ptr [rbx + rax + 8]
cmp rdx, qword ptr [rbx + rcx + 8]
jg .Lge
.Lless:
cmp r13, r14
je .Lnoswap
mov rdx, r13
shl rdx, 4
mov rsi, qword ptr [rbx + rdx]
mov rdi, qword ptr [rbx + rax]
mov qword ptr [rbx + rdx], rdi
mov qword ptr [rbx + rax], rsi
mov rsi, qword ptr [rbx + rdx + 8]
mov rdi, qword ptr [rbx + rax + 8]
mov qword ptr [rbx + rdx + 8], rdi
mov qword ptr [rbx + rax + 8], rsi
.Lnoswap:
inc r13
.Lge:
inc r14
jmp .Lloop
.Lpart_done:
cmp r13, r15
je .Lnopfinal
mov rax, r13
shl rax, 4
mov rcx, r15
shl rcx, 4
mov rdx, qword ptr [rbx + rax]
mov rsi, qword ptr [rbx + rcx]
mov qword ptr [rbx + rax], rsi
mov qword ptr [rbx + rcx], rdx
mov rdx, qword ptr [rbx + rax + 8]
mov rsi, qword ptr [rbx + rcx + 8]
mov qword ptr [rbx + rax + 8], rsi
mov qword ptr [rbx + rcx + 8], rdx
.Lnopfinal:
cmp r13, 1
jle .Lleft_done
mov rdi, rbx
mov rsi, r13
call qsort_pairs
.Lleft_done:
mov rax, r12
sub rax, r13
dec rax
cmp rax, 1
jle .Lright_done
mov rdi, r13
inc rdi
shl rdi, 4
add rdi, rbx
mov rsi, rax
call qsort_pairs
.Lright_done:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
