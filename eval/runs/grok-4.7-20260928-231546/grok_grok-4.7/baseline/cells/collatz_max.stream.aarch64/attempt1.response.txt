collatz_max:
    mov x2, #0
    mov x1, #1
.Lnext:
    cmp x1, x0
    b.gt .Lret
    mov x3, x1
    mov x4, #0
.Lcollatz:
    cmp x3, #1
    b.eq .Lmax
    tbz x3, #0, .Leven
    add x3, x3, x3, lsl #1
    add x3, x3, #1
    add x4, x4, #1
    b .Lcollatz
.Leven:
    asr x3, x3, #1
    add x4, x4, #1
    b .Lcollatz
.Lmax:
    cmp x4, x2
    csel x2, x4, x2, gt
    add x1, x1, #1
    b .Lnext
.Lret:
    mov x0, x2
    ret
