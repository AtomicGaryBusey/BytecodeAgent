export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %lonew, @nless %lokeep
	%hi =l phi @start %n, @less %hikeep, @nless %hinew
	%cont =w csltl %lo, %hi
	jnz %cont, @body, @check
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%islt =w csltl %val, %key
	jnz %islt, @less, @nless
@less
	%lonew =l add %mid, 1
	%hikeep =l copy %hi
	jmp @loop
@nless
	%lokeep =l copy %lo
	%hinew =l copy %mid
	jmp @loop
@check
	%in =w csltl %lo, %n
	jnz %in, @cmp, @miss
@cmp
	%off2 =l mul %lo, 8
	%ptr2 =l add %a, %off2
	%val2 =l loadl %ptr2
	%eq =w ceql %val2, %key
	jnz %eq, @hit, @miss
@hit
	ret %lo
@miss
	%neg =l copy -1
	ret %neg
}
