packbits_encode:
    xor r8, r8
    xor r9, r9
    test rsi, rsi
    jle .Ldone
.Lmain:
    cmp r8, rsi
    jge .Ldone
    movzx eax, byte ptr [rdi + r8]
    lea r10, [r8 + 1]
    lea r11, [r8 + 128]
    cmp r11, rsi
    cmovg r11, rsi
.Lrun_i:
    cmp r10, r11
    jge .Lrun_i_done
    cmp al, byte ptr [rdi + r10]
    jne .Lrun_i_done
    inc r10
    jmp .Lrun_i
.Lrun_i_done:
    sub r10, r8
    cmp r10, 3
    jl .Ldo_literal
    mov rax, rcx
    sub rax, r9
    cmp rax, 2
    jl .Lfail
    mov eax, 257
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lmain
.Ldo_literal:
    mov r10, r8
.Llit_while:
    cmp r10, rsi
    jge .Llit_emit
    lea rax, [r8 + 128]
    cmp r10, rax
    jge .Llit_emit
    movzx eax, byte ptr [rdi + r10]
    lea r11, [r10 + 1]
    cmp r11, rsi
    jge .Llit_adv
    cmp al, byte ptr [rdi + r11]
    jne .Llit_adv
    inc r11
    cmp r11, rsi
    jge .Llit_adv
    cmp al, byte ptr [rdi + r11]
    jne .Llit_adv
    jmp .Llit_emit
.Llit_adv:
    inc r10
    jmp .Llit_while
.Llit_emit:
    mov r11, r10
    sub r11, r8
    mov rax, rcx
    sub rax, r9
    dec rax
    cmp rax, r11
    jl .Lfail
    mov eax, r11d
    dec eax
    mov byte ptr [rdx + r9], al
    lea rax, [rdx + r9 + 1]
    add r9, r11
    inc r9
.Lcopy:
    test r11, r11
    jz .Lmain
    movzx r10d, byte ptr [rdi + r8]
    mov byte ptr [rax], r10b
    inc rax
    inc r8
    dec r11
    jmp .Lcopy
.Lfail:
    mov rax, -1
    ret
.Ldone:
    mov rax, r9
    ret
