export function l $binary_search(l %a, l %n, l %key) {
@start
	%nle =w cslel %n, 0
	jnz %nle, @notfound, @init
@init
	%lo0 =l copy 0
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @init %lo0, @greater %lo_new, @less %lo_same
	%hi =l phi @init %hi0, @greater %hi_same, @less %hi_new
	%le =w cslel %lo, %hi
	jnz %le, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@found
	ret %mid
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @greater, @less
@greater
	%lo_new =l add %mid, 1
	%hi_same =l copy %hi
	jmp @loop
@less
	%lo_same =l copy %lo
	%hi_new =l sub %mid, 1
	jmp @loop
@notfound
	%none =l copy -1
	ret %none
}
