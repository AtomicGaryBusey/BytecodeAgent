bubble_sort:
cmp rsi, 2
jl .Ldone
xor r8, r8
.Louter:
lea rax, [rsi - 1]
cmp r8, rax
jge .Ldone
xor r9, r9
mov r10, rax
sub r10, r8
.Linner:
cmp r9, r10
jge .Lnext_i
mov rax, qword ptr [rdi + r9*8]
mov rcx, qword ptr [rdi + r9*8 + 8]
cmp rax, rcx
jle .Lnoswap
mov qword ptr [rdi + r9*8], rcx
mov qword ptr [rdi + r9*8 + 8], rax
.Lnoswap:
inc r9
jmp .Linner
.Lnext_i:
inc r8
jmp .Louter
.Ldone:
xor rax, rax
ret
