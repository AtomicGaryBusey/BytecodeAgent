mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov rbx, rsi
mov r12, rsi
imul r12, rbx
mov r13, r8
lea r14, [r8 + r12*8]
mov rbp, rcx
mov r15, rdx
xor eax, eax
.Lcopy_a:
cmp rax, r12
jge .Lcopy_a_done
mov r9, qword ptr [rdi + rax*8]
mov qword ptr [r13 + rax*8], r9
inc rax
jmp .Lcopy_a
.Lcopy_a_done:
xor eax, eax
.Lzero_out:
cmp rax, r12
jge .Lzero_out_done
mov qword ptr [rbp + rax*8], 0
inc rax
jmp .Lzero_out
.Lzero_out_done:
xor eax, eax
.Lset_ident:
cmp rax, rbx
jge .Lpow_loop
mov r9, rax
imul r9, rbx
add r9, rax
mov qword ptr [rbp + r9*8], 1
inc rax
jmp .Lset_ident
.Lpow_loop:
test r15, r15
jz .Lzero_tmp
test r15, 1
jz .Lskip_mul
xor esi, esi
.Lmul_i:
cmp rsi, rbx
jge .Lmul_i_done
xor edi, edi
.Lmul_j:
cmp rdi, rbx
jge .Lmul_j_done
xor eax, eax
xor ecx, ecx
.Lmul_t:
cmp rcx, rbx
jge .Lmul_t_done
mov r8, rsi
imul r8, rbx
add r8, rcx
mov r9, qword ptr [rbp + r8*8]
mov r8, rcx
imul r8, rbx
add r8, rdi
mov r10, qword ptr [r13 + r8*8]
imul r9, r10
add rax, r9
inc rcx
jmp .Lmul_t
.Lmul_t_done:
mov r8, rsi
imul r8, rbx
add r8, rdi
mov qword ptr [r14 + r8*8], rax
inc rdi
jmp .Lmul_j
.Lmul_j_done:
inc rsi
jmp .Lmul_i
.Lmul_i_done:
xor eax, eax
.Lcopy_out:
cmp rax, r12
jge .Lskip_mul
mov r8, qword ptr [r14 + rax*8]
mov qword ptr [rbp + rax*8], r8
inc rax
jmp .Lcopy_out
.Lskip_mul:
shr r15, 1
test r15, r15
jz .Lzero_tmp
xor esi, esi
.Lsq_i:
cmp rsi, rbx
jge .Lsq_i_done
xor edi, edi
.Lsq_j:
cmp rdi, rbx
jge .Lsq_j_done
xor eax, eax
xor ecx, ecx
.Lsq_t:
cmp rcx, rbx
jge .Lsq_t_done
mov r8, rsi
imul r8, rbx
add r8, rcx
mov r9, qword ptr [r13 + r8*8]
mov r8, rcx
imul r8, rbx
add r8, rdi
mov r10, qword ptr [r13 + r8*8]
imul r9, r10
add rax, r9
inc rcx
jmp .Lsq_t
.Lsq_t_done:
mov r8, rsi
imul r8, rbx
add r8, rdi
mov qword ptr [r14 + r8*8], rax
inc rdi
jmp .Lsq_j
.Lsq_j_done:
inc rsi
jmp .Lsq_i
.Lsq_i_done:
xor eax, eax
.Lcopy_base:
cmp rax, r12
jge .Lpow_loop
mov r8, qword ptr [r14 + rax*8]
mov qword ptr [r13 + rax*8], r8
inc rax
jmp .Lcopy_base
.Lzero_tmp:
mov rcx, r12
shl rcx, 1
xor eax, eax
.Lzt:
cmp rax, rcx
jge .Ldone
mov qword ptr [r13 + rax*8], 0
inc rax
jmp .Lzt
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
