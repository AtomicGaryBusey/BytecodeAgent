export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%gt1 =w csgtl %n, 1
	jnz %gt1, @outer, @done
@outer
	%i =l phi @start 0, @outer_cont %i_next
	%ilt =w csltl %i, %nm1
	jnz %ilt, @inner_init, @done
@inner_init
	%limit =l sub %nm1, %i
	jmp @inner
@inner
	%j =l phi @inner_init 0, @inner_cont %j_next
	%jlt =w csltl %j, %limit
	jnz %jlt, @body, @outer_cont
@body
	%j8 =l mul %j, 8
	%pj =l add %a, %j8
	%vj =l loadl %pj
	%jp1 =l add %j, 1
	%jp18 =l mul %jp1, 8
	%pjp1 =l add %a, %jp18
	%vjp1 =l loadl %pjp1
	%gt =w csgtl %vj, %vjp1
	jnz %gt, @swap, @inner_cont
@swap
	storel %vjp1, %pj
	storel %vj, %pjp1
	jmp @inner_cont
@inner_cont
	%j_next =l add %j, 1
	jmp @inner
@outer_cont
	%i_next =l add %i, 1
	jmp @outer
@done
	ret 0
}
