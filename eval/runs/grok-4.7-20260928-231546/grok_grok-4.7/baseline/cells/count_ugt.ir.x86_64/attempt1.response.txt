export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
