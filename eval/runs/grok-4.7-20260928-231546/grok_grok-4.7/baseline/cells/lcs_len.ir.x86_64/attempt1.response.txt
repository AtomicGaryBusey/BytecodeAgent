export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @clear
@clear
	%cj =l phi @start 0, @clear_step %cj1
	%cj_le =w cslel %cj, %n
	jnz %cj_le, @clear_step, @ihead
@clear_step
	%cshift =l shl %cj, 3
	%cptr =l add %row, %cshift
	storel 0, %cptr
	%cj1 =l add %cj, 1
	jmp @clear
@ihead
	%i =l phi @clear 0, @iinc %i1
	%i_lt =w csltl %i, %m
	jnz %i_lt, @ibody, @finish
@ibody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @jhead
@jhead
	%j =l phi @ibody 1, @jstore %j1
	%diag =l phi @ibody 0, @jstore %up
	%j_le =w cslel %j, %n
	jnz %j_le, @jbody, @iinc
@jbody
	%jshift =l shl %j, 3
	%jptr =l add %row, %jshift
	%up =l loadl %jptr
	%jm1 =l sub %j, 1
	%lshift =l shl %jm1, 3
	%lptr =l add %row, %lshift
	%left =l loadl %lptr
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @eq_yes, @eq_no
@eq_yes
	%sum =l add %diag, 1
	jmp @jstore
@eq_no
	%lt =w csltl %left, %up
	jnz %lt, @max_up, @max_left
@max_up
	jmp @jstore
@max_left
	jmp @jstore
@jstore
	%nv =l phi @eq_yes %sum, @max_up %up, @max_left %left
	storel %nv, %jptr
	%j1 =l add %j, 1
	jmp @jhead
@iinc
	%i1 =l add %i, 1
	jmp @ihead
@finish
	%nshift =l shl %n, 3
	%nptr =l add %row, %nshift
	%res =l loadl %nptr
	ret %res
}
