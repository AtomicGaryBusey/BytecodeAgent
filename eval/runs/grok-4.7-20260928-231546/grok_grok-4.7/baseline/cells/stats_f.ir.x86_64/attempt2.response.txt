export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %i0 =l copy 0
    %one =l copy 1
    %min0 =l shl %one, 63
    %max0 =l xor %min0, -1
    %sum0 =l copy 0
    %odd0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @cont %inext
    %min =l phi @start %min0, @cont %minn
    %max =l phi @start %max0, @cont %maxn
    %sum =l phi @start %sum0, @cont %sumn
    %odd =l phi @start %odd0, @cont %oddn
    %cmp =w csltl %i, %n
    jnz %cmp, @body, @done
@body
    %off =l mul %i, 8
    %ap =l add %a, %off
    %elem =l loadl %ap
    %y =l call %f(l %elem)
    %lt =w csltl %y, %min
    jnz %lt, @setmin, @keepmin
@setmin
    jmp @minjoin
@keepmin
    jmp @minjoin
@minjoin
    %minn =l phi @setmin %y, @keepmin %min
    %gt =w csgtl %y, %max
    jnz %gt, @setmax, @keepmax
@setmax
    jmp @maxjoin
@keepmax
    jmp @maxjoin
@maxjoin
    %maxn =l phi @setmax %y, @keepmax %max
    %sumn =l add %sum, %y
    %bit =l and %y, 1
    %isodd =w cnel %bit, 0
    jnz %isodd, @incodd, @keepodd
@incodd
    %oddinc =l add %odd, 1
    jmp @cont
@keepodd
    %oddkeep =l copy %odd
    jmp @cont
@cont
    %oddn =l phi @incodd %oddinc, @keepodd %oddkeep
    %inext =l add %i, 1
    jmp @loop
@done
    storel %min, %out
    %o1 =l add %out, 8
    storel %max, %o1
    %o2 =l add %out, 16
    storel %sum, %o2
    ret %odd
}
