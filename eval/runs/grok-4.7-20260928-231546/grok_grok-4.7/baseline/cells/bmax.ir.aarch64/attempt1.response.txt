export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @head
@head
	%i =l phi @start %i0, @merge %i1
	%max =l phi @start %max0, @merge %max1
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @take, @skip
@take
	%maxt =l copy %val
	jmp @merge
@skip
	%maxs =l copy %max
	jmp @merge
@merge
	%max1 =l phi @take %maxt, @skip %maxs
	%i1 =l add %i, 1
	jmp @head
@end
	ret %max
}
