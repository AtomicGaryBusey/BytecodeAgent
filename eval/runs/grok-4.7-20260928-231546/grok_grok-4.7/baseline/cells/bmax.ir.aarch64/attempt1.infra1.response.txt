export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	%cmp0 =w csltl %i0, %n
	jnz %cmp0, @loop, @done
@loop
	%i =l phi @start %i0, @latch %i1
	%max =l phi @start %max0, @latch %max1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @take, @skip
@take
	jmp @latch
@skip
	jmp @latch
@latch
	%max1 =l phi @take %val, @skip %max
	%i1 =l add %i, 1
	%cmp =w csltl %i1, %n
	jnz %cmp, @loop, @done
@done
	%res =l phi @start %max0, @latch %max1
	ret %res
}
