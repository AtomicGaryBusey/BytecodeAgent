export function l $collatz_max(l %n) {
@start
	jmp @kloop
@kloop
	%k =l phi @start 1, @nextk %knext
	%max =l phi @start 0, @nextk %maxnext
	%past =w csgtl %k, %n
	jnz %past, @end, @istart
@istart
	jmp @iloop
@iloop
	%x =l phi @istart %k, @stepped %xnext
	%steps =l phi @istart 0, @stepped %stepsnext
	%done =w ceql %x, 1
	jnz %done, @upd, @parity
@parity
	%mask =l and %x, 1
	%odd =w cnel %mask, 0
	jnz %odd, @odd, @even
@even
	%xe =l shr %x, 1
	jmp @stepped
@odd
	%xm =l mul %x, 3
	%xo =l add %xm, 1
	jmp @stepped
@stepped
	%xnext =l phi @even %xe, @odd %xo
	%stepsnext =l add %steps, 1
	jmp @iloop
@upd
	%gt =w csgtl %steps, %max
	jnz %gt, @take, @keep
@take
	jmp @nextk
@keep
	jmp @nextk
@nextk
	%maxnext =l phi @take %steps, @keep %max
	%knext =l add %k, 1
	jmp @kloop
@end
	ret %max
}
