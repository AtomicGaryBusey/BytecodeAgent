packbits_decode:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx rax, byte ptr [rdi + r8]
    inc r8
    cmp rax, -128
    je .Lloop
    test rax, rax
    js .Lrep
    lea r10, [rax + 1]
    lea rax, [r8 + r10]
    cmp rax, rsi
    ja .Lsrc
    lea rax, [r9 + r10]
    cmp rax, rcx
    ja .Llit_over
.Llit_cp:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jnz .Llit_cp
    jmp .Lloop
.Llit_over:
    mov r11, rcx
    sub r11, r9
    test r11, r11
    jz .Lcap
.Llit_ov:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Llit_ov
    jmp .Lcap
.Lrep:
    neg rax
    inc rax
    cmp r8, rsi
    jae .Lsrc
    mov r11b, byte ptr [rdi + r8]
    inc r8
    lea r10, [r9 + rax]
    cmp r10, rcx
    ja .Lrep_over
.Lrep_fill:
    mov byte ptr [rdx + r9], r11b
    inc r9
    dec rax
    jnz .Lrep_fill
    jmp .Lloop
.Lrep_over:
    mov r10, rcx
    sub r10, r9
    test r10, r10
    jz .Lcap
.Lrep_ov:
    mov byte ptr [rdx + r9], r11b
    inc r9
    dec r10
    jnz .Lrep_ov
.Lcap:
    mov rax, -1
    ret
.Lsrc:
    mov rax, -2
    ret
.Ldone:
    mov rax, r9
    ret
