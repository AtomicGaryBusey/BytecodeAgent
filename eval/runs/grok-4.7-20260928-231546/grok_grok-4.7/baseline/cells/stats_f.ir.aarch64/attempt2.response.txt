export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odds0 =l and %y0, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @maxj %inext
	%min =l phi @start %y0, @maxj %minn
	%max =l phi @start %y0, @maxj %maxn
	%sum =l phi @start %y0, @maxj %sumn
	%odds =l phi @start %odds0, @maxj %oddsn
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%sumn =l add %sum, %y
	%bit =l and %y, 1
	%oddsn =l add %odds, %bit
	%lt =w csltl %y, %min
	jnz %lt, @newmin, @keepmin
@newmin
	jmp @minj
@keepmin
	jmp @minj
@minj
	%minn =l phi @newmin %y, @keepmin %min
	%gt =w csgtl %y, %max
	jnz %gt, @newmax, @keepmax
@newmax
	jmp @maxj
@keepmax
	jmp @maxj
@maxj
	%maxn =l phi @newmax %y, @keepmax %max
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%p1 =l add %out, 8
	storel %max, %p1
	%p2 =l add %out, 16
	storel %sum, %p2
	ret %odds
}
