export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	%zero =l copy 0
	%k0 =w ceql %k, 0
	jnz %k0, @ident, @copy_init
@ident
	jmp @i_loop
@i_loop
	%i =l phi @ident 0, @i_next %i1
	%i_lt =w csltl %i, %n
	jnz %i_lt, @j_init, @zero_init
@j_init
	jmp @j_loop
@j_loop
	%j =l phi @j_init 0, @j_next %j1
	%j_lt =w csltl %j, %n
	jnz %j_lt, @j_body, @i_next
@j_body
	%eq =w ceql %i, %j
	%valf =l extuw %eq
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, 8
	%addr =l add %out, %off
	storel %valf, %addr
	jmp @j_next
@j_next
	%j1 =l add %j, 1
	jmp @j_loop
@i_next
	%i1 =l add %i, 1
	jmp @i_loop
@copy_init
	jmp @c_loop
@c_loop
	%ci =l phi @copy_init 0, @c_body %ci1
	%ci_lt =w csltl %ci, %nn
	jnz %ci_lt, @c_body, @p_init
@c_body
	%coff =l mul %ci, 8
	%caddr =l add %a, %coff
	%cval =l loadl %caddr
	%oaddr =l add %out, %coff
	storel %cval, %oaddr
	%ci1 =l add %ci, 1
	jmp @c_loop
@p_init
	jmp @p_loop
@p_loop
	%p =l phi @p_init 1, @p_inc %p1
	%p_lt =w csltl %p, %k
	jnz %p_lt, @mi_init, @zero_init
@mi_init
	jmp @mi_loop
@mi_loop
	%mi =l phi @mi_init 0, @mi_inc %mi1
	%mi_lt =w csltl %mi, %n
	jnz %mi_lt, @mj_init, @cb_init
@mj_init
	jmp @mj_loop
@mj_loop
	%mj =l phi @mj_init 0, @mj_inc %mj1
	%mj_lt =w csltl %mj, %n
	jnz %mj_lt, @mt_init, @mi_inc
@mt_init
	jmp @mt_loop
@mt_loop
	%mt =l phi @mt_init 0, @mt_body %mt1
	%sum =l phi @mt_init 0, @mt_body %sum1
	%mt_lt =w csltl %mt, %n
	jnz %mt_lt, @mt_body, @mt_store
@mt_body
	%row =l mul %mi, %n
	%oidx =l add %row, %mt
	%ooff =l mul %oidx, 8
	%oaddr =l add %out, %ooff
	%ov =l loadl %oaddr
	%acol =l mul %mt, %n
	%aidx =l add %acol, %mj
	%aoff =l mul %aidx, 8
	%aaddr =l add %a, %aoff
	%av =l loadl %aaddr
	%prod =l mul %ov, %av
	%sum1 =l add %sum, %prod
	%mt1 =l add %mt, 1
	jmp @mt_loop
@mt_store
	%row2 =l mul %mi, %n
	%sidx =l add %row2, %mj
	%soff =l mul %sidx, 8
	%saddr =l add %tmp, %soff
	storel %sum, %saddr
	jmp @mj_inc
@mj_inc
	%mj1 =l add %mj, 1
	jmp @mj_loop
@mi_inc
	%mi1 =l add %mi, 1
	jmp @mi_loop
@cb_init
	jmp @cb_loop
@cb_loop
	%cb =l phi @cb_init 0, @cb_body %cb1
	%cb_lt =w csltl %cb, %nn
	jnz %cb_lt, @cb_body, @p_inc
@cb_body
	%cboff =l mul %cb, 8
	%cbsrc =l add %tmp, %cboff
	%cbval =l loadl %cbsrc
	%cbdst =l add %out, %cboff
	storel %cbval, %cbdst
	%cb1 =l add %cb, 1
	jmp @cb_loop
@p_inc
	%p1 =l add %p, 1
	jmp @p_loop
@zero_init
	jmp @z_loop
@z_loop
	%zi =l phi @zero_init 0, @z_body %zi1
	%zi_lt =w csltl %zi, %nn2
	jnz %zi_lt, @z_body, @done
@z_body
	%zoff =l mul %zi, 8
	%zaddr =l add %tmp, %zoff
	storel %zero, %zaddr
	%zi1 =l add %zi, 1
	jmp @z_loop
@done
	ret %zero
}
