export function l $collatz_max(l %n) {
@start
	%maxp =l alloc8 8
	%kp =l alloc8 8
	%xp =l alloc8 8
	%sp =l alloc8 8
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%three =l copy 3
	storel %zero, %maxp
	storel %one, %kp
	jmp @kloop
@kloop
	%k =l loadl %kp
	%k_le =w cslel %k, %n
	jnz %k_le, @kbody, @done
@kbody
	storel %k, %xp
	storel %zero, %sp
	jmp @cloop
@cloop
	%x =l loadl %xp
	%x_is_1 =w ceql %x, %one
	jnz %x_is_1, @cupd, @cstep
@cstep
	%bit =l and %x, %one
	%is_odd =w cnel %bit, %zero
	jnz %is_odd, @odd, @even
@even
	%x2 =l div %x, %two
	storel %x2, %xp
	jmp @cinc
@odd
	%t3 =l mul %x, %three
	%x3 =l add %t3, %one
	storel %x3, %xp
	jmp @cinc
@cinc
	%s =l loadl %sp
	%s1 =l add %s, %one
	storel %s1, %sp
	jmp @cloop
@cupd
	%steps =l loadl %sp
	%maxv =l loadl %maxp
	%gt =w csgtl %steps, %maxv
	jnz %gt, @setmax, @knext
@setmax
	storel %steps, %maxp
	jmp @knext
@knext
	%kcur =l loadl %kp
	%kn =l add %kcur, %one
	storel %kn, %kp
	jmp @kloop
@done
	%result =l loadl %maxp
	ret %result
}
