export function l $reverse(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%j0 =l sub %n, %one
	jmp @loop
@loop
	%i =l phi @start %zero, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @done
@body
	%ioff =l mul %i, %eight
	%joff =l mul %j, %eight
	%ip =l add %a, %ioff
	%jp =l add %a, %joff
	%iv =l loadl %ip
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%inext =l add %i, %one
	%jnext =l sub %j, %one
	jmp @loop
@done
	ret %zero
}
