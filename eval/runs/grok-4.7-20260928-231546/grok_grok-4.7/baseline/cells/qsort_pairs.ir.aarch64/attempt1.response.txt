export function l $qsort_pairs(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret0, @dosort
@ret0
	ret 0
@dosort
	%nm1 =l sub %n, 1
	%pvoff =l mul %nm1, 16
	%pp =l add %a, %pvoff
	%pk =l loadl %pp
	%pp8 =l add %pp, 8
	%pv =l loadl %pp8
	jmp @loop
@loop
	%j =l phi @dosort 0, @adv %jnext
	%i =l phi @dosort 0, @adv %inext
	%jcmp =w csltl %j, %nm1
	jnz %jcmp, @body, @partend
@body
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jp8 =l add %jp, 8
	%jv =l loadl %jp8
	%klt =w csltl %jk, %pk
	jnz %klt, @doswap, @keyge
@keyge
	%keq =w ceql %jk, %pk
	jnz %keq, @keyeq, @noswap
@keyeq
	%vlt =w csltl %jv, %pv
	jnz %vlt, @doswap, @noswap
@doswap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ip8 =l add %ip, 8
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%iinc =l add %i, 1
	jmp @adv
@noswap
	jmp @adv
@adv
	%inext =l phi @doswap %iinc, @noswap %i
	%jnext =l add %j, 1
	jmp @loop
@partend
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ip28 =l add %ip2, 8
	%iv2 =l loadl %ip28
	storel %pk, %ip2
	storel %pv, %ip28
	storel %ik2, %pp
	storel %iv2, %pp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%rp =l add %a, %roff
	%rcnt =l sub %n, %ip1
	%r2 =l call $qsort_pairs(l %rp, l %rcnt)
	ret 0
}
