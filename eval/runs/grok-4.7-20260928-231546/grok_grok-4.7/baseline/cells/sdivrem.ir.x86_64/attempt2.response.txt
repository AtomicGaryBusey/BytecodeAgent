export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @bzero, @chkmin
@bzero
	storel %a, %rem
	%zq =l copy 0
	ret %zq
@chkmin
	%one =l copy 1
	%min =l shl %one, 63
	%ismin =w ceql %a, %min
	jnz %ismin, @chkb, @div
@chkb
	%mone =l copy -1
	%isn1 =w ceql %b, %mone
	jnz %isn1, @ovf, @div
@ovf
	%zr =l copy 0
	storel %zr, %rem
	ret %min
@div
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
