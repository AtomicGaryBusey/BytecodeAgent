export function l $fib(l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%lt2 =w csltl %n, %two
	jnz %lt2, @base, @loop
@base
	ret %n
@loop
	%a =l phi @start %zero, @loop %an
	%b =l phi @start %one, @loop %bn
	%i =l phi @start %one, @loop %inext
	%sum =l add %a, %b
	%an =l copy %b
	%bn =l copy %sum
	%inext =l add %i, %one
	%cmp =w csltl %inext, %n
	jnz %cmp, @loop, @done
@done
	ret %sum
}
