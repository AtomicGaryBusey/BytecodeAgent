crc32:
    mov eax, 0xffffffff
    test rsi, rsi
    jz .Ldone
.Lbyte:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit:
    shr eax, 1
    jnc .Lskip
    xor eax, 0xedb88320
.Lskip:
    dec ecx
    jnz .Lbit
    inc rdi
    dec rsi
    jnz .Lbyte
.Ldone:
    xor eax, 0xffffffff
    ret
