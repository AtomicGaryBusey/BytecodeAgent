export function l $reverse(l %a, l %n) {
@start
	%iz =l copy 0
	%jz =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %iz, @body %inext
	%j =l phi @start %jz, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @done
@body
	%ioff =l mul %i, 8
	%ip =l add %a, %ioff
	%iv =l loadl %ip
	%joff =l mul %j, 8
	%jp =l add %a, %joff
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
