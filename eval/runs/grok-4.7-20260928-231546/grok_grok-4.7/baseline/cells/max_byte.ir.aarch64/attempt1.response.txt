export function l $max_byte(l %p, l %n) {
@start
	%isz =w ceql %n, 0
	jnz %isz, @zero, @loop
@zero
	ret 0
@loop
	%i =l phi @start 0, @next %i1
	%max =l phi @start 0, @next %max1
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @take, @skip
@take
	%maxt =l copy %b
	jmp @next
@skip
	%maxs =l copy %max
	jmp @next
@next
	%max1 =l phi @take %maxt, @skip %maxs
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
