export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%s =l phi @start 0, @body %s1
	%i =l phi @start 0, @body %i1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%r =l call %f(l %v)
	%s1 =l add %s, %r
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %s
}
