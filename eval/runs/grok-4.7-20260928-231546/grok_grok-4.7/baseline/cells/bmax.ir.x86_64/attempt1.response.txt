export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%max =l phi @start %max0, @next %max1
	%i =l phi @start %i0, @next %i1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%max1 =l phi @update %val, @keep %max
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
