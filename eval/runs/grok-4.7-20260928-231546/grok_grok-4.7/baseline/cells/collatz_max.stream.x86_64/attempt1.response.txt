collatz_max:
xor rax, rax
mov rcx, 1
.Lnextk:
cmp rcx, rdi
jg .Lret
mov rdx, rcx
xor rsi, rsi
.Lstep:
cmp rdx, 1
je .Lupd
test rdx, 1
jnz .Lodd
shr rdx, 1
jmp .Linc
.Lodd:
lea rdx, [rdx + rdx*2 + 1]
.Linc:
inc rsi
jmp .Lstep
.Lupd:
cmp rsi, rax
jle .Lskip
mov rax, rsi
.Lskip:
inc rcx
jmp .Lnextk
.Lret:
ret
