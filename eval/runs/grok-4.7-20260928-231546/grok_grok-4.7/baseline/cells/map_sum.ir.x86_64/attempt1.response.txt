export function l $map_sum(l %a, l %n, l %f) {
@start
	%izero =l copy 0
	%szero =l copy 0
	jmp @head
@head
	%i =l phi @start %izero, @body %inext
	%sum =l phi @start %szero, @body %sumnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%elem =l loadl %ptr
	%mapped =l call %f(l %elem)
	%sumnext =l add %sum, %mapped
	%inext =l add %i, 1
	jmp @head
@done
	ret %sum
}
