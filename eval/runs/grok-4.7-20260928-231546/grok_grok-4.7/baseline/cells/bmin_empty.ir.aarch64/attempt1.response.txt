export function l $bmin_empty(l %a, l %n) {
@start
    %imax =l copy 9223372036854775807
    %izero =l copy 0
    %ione =l copy 1
    %ieight =l copy 8
    jmp @head
@head
    %i =l phi @start %izero, @latch %inext
    %m =l phi @start %imax, @latch %mnext
    %cmp =w csltl %i, %n
    jnz %cmp, @body, @end
@body
    %off =l mul %i, %ieight
    %ptr =l add %a, %off
    %v =l loadl %ptr
    %lt =w csltl %v, %m
    jnz %lt, @take, @skip
@take
    %mt =l copy %v
    jmp @latch
@skip
    %ms =l copy %m
    jmp @latch
@latch
    %mnext =l phi @take %mt, @skip %ms
    %inext =l add %i, %ione
    jmp @head
@end
    ret %m
}
