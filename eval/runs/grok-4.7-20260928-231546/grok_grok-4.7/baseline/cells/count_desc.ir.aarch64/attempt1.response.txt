export function l $count_desc(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @zero, @init
@zero
	%z =l copy 0
	ret %z
@init
	%nm1 =l sub %n, 1
	%bytes =l mul %nm1, 8
	%end =l add %a, %bytes
	jmp @head
@head
	%p =l phi @init %a, @body %pnext
	%cnt =l phi @init 0, @body %cntnext
	%cond =w cultl %p, %end
	jnz %cond, @body, @done
@body
	%v0 =l loadl %p
	%pnext =l add %p, 8
	%v1 =l loadl %pnext
	%gt =w csgtl %v0, %v1
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	jmp @head
@done
	ret %cnt
}
