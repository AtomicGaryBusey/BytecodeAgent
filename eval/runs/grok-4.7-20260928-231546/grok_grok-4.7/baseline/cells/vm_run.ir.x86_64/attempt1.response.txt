export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @main
@main
	%pc =l phi @start 0, @latch %pc_n
	%sp =l phi @start 0, @latch %sp_n
	%at_end =w ceql %pc, %len
	jnz %at_end, @finish, @fetch
@fetch
	%pc_off =l mul %pc, 8
	%op_ptr =l add %code, %pc_off
	%op =l loadl %op_ptr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @chk1
@chk1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @chk2
@chk2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @chk3
@chk3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @chk4
@chk4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @chk5
@chk5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @chk6
@chk6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @chk7
@chk7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @chk8
@chk8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @chk9
@chk9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @chk10
@chk10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @chk11
@chk11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_other
@op_push
	%p_pc1 =l add %pc, 1
	%p_off =l mul %p_pc1, 8
	%p_ptr =l add %code, %p_off
	%p_val =l loadl %p_ptr
	%p_soff =l mul %sp, 8
	%p_slot =l add %stack, %p_soff
	storel %p_val, %p_slot
	%pc_push =l add %pc, 2
	%sp_push =l add %sp, 1
	jmp @latch
@op_add
	%a_sp1 =l sub %sp, 1
	%a_sp2 =l sub %sp, 2
	%a_boff =l mul %a_sp1, 8
	%a_aoff =l mul %a_sp2, 8
	%a_bptr =l add %stack, %a_boff
	%a_aptr =l add %stack, %a_aoff
	%a_b =l loadl %a_bptr
	%a_a =l loadl %a_aptr
	%a_sum =l add %a_a, %a_b
	storel %a_sum, %a_aptr
	%pc_add =l add %pc, 1
	%sp_add =l copy %a_sp1
	jmp @latch
@op_sub
	%s_sp1 =l sub %sp, 1
	%s_sp2 =l sub %sp, 2
	%s_boff =l mul %s_sp1, 8
	%s_aoff =l mul %s_sp2, 8
	%s_bptr =l add %stack, %s_boff
	%s_aptr =l add %stack, %s_aoff
	%s_b =l loadl %s_bptr
	%s_a =l loadl %s_aptr
	%s_diff =l sub %s_a, %s_b
	storel %s_diff, %s_aptr
	%pc_sub =l add %pc, 1
	%sp_sub =l copy %s_sp1
	jmp @latch
@op_mul
	%m_sp1 =l sub %sp, 1
	%m_sp2 =l sub %sp, 2
	%m_boff =l mul %m_sp1, 8
	%m_aoff =l mul %m_sp2, 8
	%m_bptr =l add %stack, %m_boff
	%m_aptr =l add %stack, %m_aoff
	%m_b =l loadl %m_bptr
	%m_a =l loadl %m_aptr
	%m_prod =l mul %m_a, %m_b
	storel %m_prod, %m_aptr
	%pc_mul =l add %pc, 1
	%sp_mul =l copy %m_sp1
	jmp @latch
@op_dup
	%d_sp1 =l sub %sp, 1
	%d_off =l mul %d_sp1, 8
	%d_ptr =l add %stack, %d_off
	%d_val =l loadl %d_ptr
	%d_noff =l mul %sp, 8
	%d_nptr =l add %stack, %d_noff
	storel %d_val, %d_nptr
	%pc_dup =l add %pc, 1
	%sp_dup =l add %sp, 1
	jmp @latch
@op_swap
	%w_sp1 =l sub %sp, 1
	%w_sp2 =l sub %sp, 2
	%w_boff =l mul %w_sp1, 8
	%w_aoff =l mul %w_sp2, 8
	%w_bptr =l add %stack, %w_boff
	%w_aptr =l add %stack, %w_aoff
	%w_b =l loadl %w_bptr
	%w_a =l loadl %w_aptr
	storel %w_b, %w_aptr
	storel %w_a, %w_bptr
	%pc_swap =l add %pc, 1
	%sp_swap =l copy %sp
	jmp @latch
@op_drop
	%pc_drop =l add %pc, 1
	%sp_drop =l sub %sp, 1
	jmp @latch
@op_over
	%o_sp2 =l sub %sp, 2
	%o_off =l mul %o_sp2, 8
	%o_ptr =l add %stack, %o_off
	%o_val =l loadl %o_ptr
	%o_noff =l mul %sp, 8
	%o_nptr =l add %stack, %o_noff
	storel %o_val, %o_nptr
	%pc_over =l add %pc, 1
	%sp_over =l add %sp, 1
	jmp @latch
@op_lt
	%l_sp1 =l sub %sp, 1
	%l_sp2 =l sub %sp, 2
	%l_boff =l mul %l_sp1, 8
	%l_aoff =l mul %l_sp2, 8
	%l_bptr =l add %stack, %l_boff
	%l_aptr =l add %stack, %l_aoff
	%l_b =l loadl %l_bptr
	%l_a =l loadl %l_aptr
	%l_cmp =w csltl %l_a, %l_b
	%l_res =l extsw %l_cmp
	storel %l_res, %l_aptr
	%pc_lt =l add %pc, 1
	%sp_lt =l copy %l_sp1
	jmp @latch
@op_jz
	%z_sp1 =l sub %sp, 1
	%z_off =l mul %z_sp1, 8
	%z_ptr =l add %stack, %z_off
	%z_v =l loadl %z_ptr
	%z_pc1 =l add %pc, 1
	%z_toff =l mul %z_pc1, 8
	%z_tptr =l add %code, %z_toff
	%z_t =l loadl %z_tptr
	%z_is0 =w ceql %z_v, 0
	jnz %z_is0, @op_jz_yes, @op_jz_no
@op_jz_yes
	%pc_jzy =l copy %z_t
	%sp_jzy =l copy %z_sp1
	jmp @latch
@op_jz_no
	%pc_jzn =l add %pc, 2
	%sp_jzn =l copy %z_sp1
	jmp @latch
@op_jmp
	%j_pc1 =l add %pc, 1
	%j_off =l mul %j_pc1, 8
	%j_ptr =l add %code, %j_off
	%j_t =l loadl %j_ptr
	%pc_jmp =l copy %j_t
	%sp_jmp =l copy %sp
	jmp @latch
@op_other
	%pc_oth =l add %pc, 1
	%sp_oth =l copy %sp
	jmp @latch
@latch
	%pc_n =l phi @op_push %pc_push, @op_add %pc_add, @op_sub %pc_sub, @op_mul %pc_mul, @op_dup %pc_dup, @op_swap %pc_swap, @op_drop %pc_drop, @op_over %pc_over, @op_lt %pc_lt, @op_jz_yes %pc_jzy, @op_jz_no %pc_jzn, @op_jmp %pc_jmp, @op_other %pc_oth
	%sp_n =l phi @op_push %sp_push, @op_add %sp_add, @op_sub %sp_sub, @op_mul %sp_mul, @op_dup %sp_dup, @op_swap %sp_swap, @op_drop %sp_drop, @op_over %sp_over, @op_lt %sp_lt, @op_jz_yes %sp_jzy, @op_jz_no %sp_jzn, @op_jmp %sp_jmp, @op_other %sp_oth
	jmp @main
@finish
	%sp_fin =l copy %sp
	jmp @zloop
@zloop
	%zi =l phi @finish %sp_fin, @zbody %zi_next
	%zdone =w ceql %zi, %cap
	jnz %zdone, @ret, @zbody
@zbody
	%z_off =l mul %zi, 8
	%z_ptr =l add %stack, %z_off
	storel 0, %z_ptr
	%zi_next =l add %zi, 1
	jmp @zloop
@ret
	%empty =w ceql %sp, 0
	jnz %empty, @ret0, @ret_top
@ret0
	%ret0v =l copy 0
	ret %ret0v
@ret_top
	%top_i =l sub %sp, 1
	%top_off =l mul %top_i, 8
	%top_ptr =l add %stack, %top_off
	%topv =l loadl %top_ptr
	ret %topv
}
