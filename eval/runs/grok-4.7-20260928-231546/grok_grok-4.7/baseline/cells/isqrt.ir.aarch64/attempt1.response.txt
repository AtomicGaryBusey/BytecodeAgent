export function l $isqrt(l %n) {
@start
	%one =l copy 1
	%hi0 =l copy 4294967296
	jmp @loop
@loop
	%lo =l phi @start %one, @setlo %lonew, @sethi %losame
	%hi =l phi @start %hi0, @setlo %hisame, @sethi %hinew
	%le =w culel %lo, %hi
	jnz %le, @body, @done
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%gt =w cugtl %mid, %quot
	jnz %gt, @sethi, @setlo
@sethi
	%hinew =l sub %mid, %one
	%losame =l copy %lo
	jmp @loop
@setlo
	%lonew =l add %mid, %one
	%hisame =l copy %hi
	jmp @loop
@done
	ret %hi
}
