export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %countnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%saddr =l add %src, %off
	%daddr =l add %dst, %off
	%orig =l loadl %saddr
	%mapped =l call %f(l %orig)
	storel %mapped, %daddr
	%lt =w csltl %mapped, %orig
	%ltl =l extuw %lt
	%countnext =l add %count, %ltl
	%inext =l add %i, 1
	jmp @loop
@end
	ret %count
}
