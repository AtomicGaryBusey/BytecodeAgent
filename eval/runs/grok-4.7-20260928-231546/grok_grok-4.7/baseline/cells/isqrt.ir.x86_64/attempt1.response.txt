export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @setlo %lo_keep, @sethi %lo_next
	%hi =l phi @start 4294967295, @setlo %hi_keep, @sethi %hi_next
	%lt =w cultl %lo, %hi
	jnz %lt, @body, @done
@body
	%sum =l add %lo, %hi
	%sum1 =l add %sum, 1
	%mid =l udiv %sum1, 2
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	%lo_keep =l copy %mid
	%hi_keep =l copy %hi
	jmp @loop
@sethi
	%lo_next =l copy %lo
	%hi_next =l sub %mid, 1
	jmp @loop
@done
	ret %lo
}
