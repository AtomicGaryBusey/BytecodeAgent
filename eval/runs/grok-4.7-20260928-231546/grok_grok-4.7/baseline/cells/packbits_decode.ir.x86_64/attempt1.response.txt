export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %si =l copy 0
    %oi =l copy 0
    jmp @loop
@loop
    %si1 =l phi @start %si, @nopc %sin, @litd %sil, @repd %sir
    %oi1 =l phi @start %oi, @nopc %oin, @litd %oil, @repd %oir
    %more =w csltl %si1, %n
    jnz %more, @rhdr, @ok
@ok
    ret %oi1
@rhdr
    %hp =l add %src, %si1
    %h =w loadsb %hp
    %si2 =l add %si1, 1
    %isnop =w ceqw %h, -128
    jnz %isnop, @nopc, @chkneg
@nopc
    %sin =l copy %si2
    %oin =l copy %oi1
    jmp @loop
@chkneg
    %isneg =w csltw %h, 0
    jnz %isneg, @repeat, @literal
@literal
    %h1 =w add %h, 1
    %count =l extuw %h1
    %rem =l sub %n, %si2
    %bad =w csgtl %count, %rem
    jnz %bad, @esrc, @linit
@esrc
    %m2 =l copy -2
    ret %m2
@linit
    %i0 =l copy 0
    jmp @lcopy
@lcopy
    %i1 =l phi @linit %i0, @lwrite %i2
    %sic =l phi @linit %si2, @lwrite %sic2
    %oic =l phi @linit %oi1, @lwrite %oic2
    %ldone =w ceql %i1, %count
    jnz %ldone, @litd, @lone
@litd
    %sil =l copy %sic
    %oil =l copy %oic
    jmp @loop
@lone
    %lfull =w cslel %cap, %oic
    jnz %lfull, @ecap, @lwrite
@lwrite
    %sp =l add %src, %sic
    %b =w loadub %sp
    %dp =l add %out, %oic
    storeb %b, %dp
    %sic2 =l add %sic, 1
    %oic2 =l add %oic, 1
    %i2 =l add %i1, 1
    jmp @lcopy
@ecap
    %m1 =l copy -1
    ret %m1
@repeat
    %remr =l sub %n, %si2
    %rbad =w csltl %remr, 1
    jnz %rbad, @esrc, @rread
@rread
    %rsp =l add %src, %si2
    %rb =w loadub %rsp
    %sir0 =l add %si2, 1
    %one =w copy 1
    %cntw =w sub %one, %h
    %rcount =l extuw %cntw
    %j0 =l copy 0
    jmp @rcopy
@rcopy
    %j1 =l phi @rread %j0, @rwrite %j2
    %oir =l phi @rread %oi1, @rwrite %oir2
    %rdone =w ceql %j1, %rcount
    jnz %rdone, @repd, @rone
@repd
    %sir =l copy %sir0
    %oirb =l copy %oir
    jmp @loop
@rone
    %rfull =w cslel %cap, %oir
    jnz %rfull, @ecap, @rwrite
@rwrite
    %rdp =l add %out, %oir
    storeb %rb, %rdp
    %oir2 =l add %oir, 1
    %j2 =l add %j1, 1
    jmp @rcopy
}
