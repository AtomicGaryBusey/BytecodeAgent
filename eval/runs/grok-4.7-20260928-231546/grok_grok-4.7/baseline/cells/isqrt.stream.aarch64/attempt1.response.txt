    b.lo .Lret
    mov w2, #-1
    mov x1, #1
    cmp x0, x2
    csel x2, x0, x2, lo
.Lloop:
    cmp x1, x2
    b.hi .Lhi
    add x3, x1, x2
    lsr x3, x3, #1
    udiv x4, x0, x3
    cmp x3, x4
    b.hi .Lbig
    add x1, x3, #1
    b .Lloop

.Lhi:

.Lret:
