export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%max =l phi @start 0, @next %max1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%max1 =l phi @update %b, @keep %max
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
