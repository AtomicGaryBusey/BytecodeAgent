export function l $base64_encode(l %src, l %n, l %out) {
@start
	%np2 =l add %n, 2
	%ceil3 =l udiv %np2, 3
	%retv =l mul %ceil3, 4
	%rem =l urem %n, 3
	%full =l sub %n, %rem
	jmp @loop
@loop
	%i =l phi @start 0, @groupdone %inext
	%ps =l phi @start %src, @groupdone %psnext
	%po =l phi @start %out, @groupdone %ponext
	%more =w csltl %i, %full
	jnz %more, @group, @tail
@group
	%gp1 =l add %ps, 1
	%gp2 =l add %ps, 2
	%gb0 =l loadub %ps
	%gb1 =l loadub %gp1
	%gb2 =l loadub %gp2
	%gs0 =l shl %gb0, 16
	%gs1 =l shl %gb1, 8
	%go0 =l or %gs0, %gs1
	%gv =l or %go0, %gb2
	jmp @cloop
@tail
	%rz =w ceql %rem, 0
	jnz %rz, @ret, @rtest
@rtest
	%r1 =w ceql %rem, 1
	jnz %r1, @one, @two
@one
	%ob0 =l loadub %ps
	%ov =l shl %ob0, 16
	jmp @cloop
@two
	%tp1 =l add %ps, 1
	%tb0 =l loadub %ps
	%tb1 =l loadub %tp1
	%ts0 =l shl %tb0, 16
	%ts1 =l shl %tb1, 8
	%tv =l or %ts0, %ts1
	jmp @cloop
@cloop
	%k =l phi @group 0, @one 0, @two 0, @cloop %k1
	%sh =w phi @group 18, @one 18, @two 18, @cloop %sh1
	%pc =l phi @group %po, @one %po, @two %po, @cloop %pc1
	%cnt =l phi @group 4, @one 2, @two 3, @cloop %cntk
	%vv =l phi @group %gv, @one %ov, @two %tv, @cloop %vvk
	%mode =w phi @group 0, @one 1, @two 2, @cloop %modek
	%raw =l shr %vv, %sh
	%idx =l and %raw, 63
	%madd =l add %idx, 65
	%mge26 =w csgel %idx, 26
	%mge26l =l extuw %mge26
	%madd6 =l mul %mge26l, 6
	%mb =l add %madd, %madd6
	%mge52 =w csgel %idx, 52
	%mge52l =l extuw %mge52
	%msub75 =l mul %mge52l, -75
	%mc =l add %mb, %msub75
	%mge62 =w csgel %idx, 62
	%mge62l =l extuw %mge62
	%mis63 =w ceql %idx, 63
	%mis63l =l extuw %mis63
	%mp3 =l mul %mis63l, 3
	%mdelta =l add %mp3, -15
	%madj =l mul %mdelta, %mge62l
	%md =l add %mc, %madj
	%mch =w copy %md
	storeb %mch, %pc
	%k1 =l add %k, 1
	%sh1 =w sub %sh, 6
	%pc1 =l add %pc, 1
	%cntk =l copy %cnt
	%vvk =l copy %vv
	%modek =w copy %mode
	%klt =w csltl %k1, %cnt
	jnz %klt, @cloop, @after
@after
	%is0 =w ceqw %mode, 0
	jnz %is0, @groupdone, @padchk
@groupdone
	%inext =l add %i, 3
	%psnext =l add %ps, 3
	%ponext =l add %po, 4
	jmp @loop
@padchk
	%is1 =w ceqw %mode, 1
	jnz %is1, @pad2, @pad1
@pad2
	%eq =w copy 61
	%pp1 =l add %pc1, 1
	storeb %eq, %pc1
	storeb %eq, %pp1
	jmp @ret
@pad1
	%eq1 =w copy 61
	storeb %eq1, %pc1
	jmp @ret
@ret
	ret %retv
}
