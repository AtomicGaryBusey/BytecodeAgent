binary_search:
    mov x3, xzr
    sub x4, x1, #1
.Lloop:
    cmp x3, x4
    b.gt .Lnotfound
    add x5, x3, x4
    lsr x5, x5, #1
    ldr x6, [x0, x5, lsl #3]
    cmp x6, x2
    b.eq .Lfound
    b.lt .Lgo_right
    sub x4, x5, #1
    b .Lloop
.Lgo_right:
    add x3, x5, #1
    b .Lloop
.Lfound:
    mov x0, x5
    ret
.Lnotfound:
    mov x0, #-1
    ret
