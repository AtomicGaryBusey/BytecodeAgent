export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%si =l phi @start 0, @donop %si_n, @litdone %si_l, @repdone %si_r
	%oi =l phi @start 0, @donop %oi_n, @litdone %oi_l, @repdone %oi_r
	%more =w csltl %si, %n
	jnz %more, @hdr, @done
@done
	ret %oi
@hdr
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @donop, @sign
@donop
	%si_n =l copy %si1
	%oi_n =l copy %oi
	jmp @loop
@sign
	%isneg =w csltw %h, 0
	jnz %isneg, @rep, @lit
@lit
	%cw =w add %h, 1
	%cnt =l extuw %cw
	%rem =l sub %n, %si1
	%bad =w csgtl %cnt, %rem
	jnz %bad, @eos, @litenter
@litenter
	%k0 =l copy 0
	%o0 =l copy %oi
	jmp @ll
@ll
	%k =l phi @litenter %k0, @lw %k1
	%o =l phi @litenter %o0, @lw %o1
	%litend =w ceql %k, %cnt
	jnz %litend, @litdone, @lchk
@lchk
	%lfull =w ceql %o, %cap
	jnz %lfull, @eoc, @lw
@lw
	%p0 =l add %src, %si1
	%p =l add %p0, %k
	%b =w loadub %p
	%d =l add %out, %o
	storeb %b, %d
	%o1 =l add %o, 1
	%k1 =l add %k, 1
	jmp @ll
@litdone
	%si_l =l add %si1, %cnt
	%oi_l =l add %oi, %cnt
	jmp @loop
@rep
	%has =w csltl %si1, %n
	jnz %has, @rinit, @eos
@rinit
	%rw =w sub 1, %h
	%rc =l extuw %rw
	%sp =l add %src, %si1
	%vb =w loadub %sp
	%sirb =l add %si1, 1
	%j0 =l copy 0
	%q0 =l copy %oi
	jmp @rl
@rl
	%j =l phi @rinit %j0, @rwst %j1
	%q =l phi @rinit %q0, @rwst %q1
	%rend =w ceql %j, %rc
	jnz %rend, @repdone, @rchk
@rchk
	%rfull =w ceql %q, %cap
	jnz %rfull, @eoc, @rwst
@rwst
	%dp =l add %out, %q
	storeb %vb, %dp
	%q1 =l add %q, 1
	%j1 =l add %j, 1
	jmp @rl
@repdone
	%si_r =l copy %sirb
	%oi_r =l add %oi, %rc
	jmp @loop
@eos
	ret -2
@eoc
	ret -1
}
