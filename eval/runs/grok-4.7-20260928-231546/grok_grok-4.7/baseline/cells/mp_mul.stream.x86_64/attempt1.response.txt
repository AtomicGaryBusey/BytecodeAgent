mp_mul:
    test rsi, rsi
    jz .Lzero
    test rcx, rcx
    jz .Lzero
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rdx
    mov r14, r8
    mov r15, rsi
    mov rbx, rcx
    lea rdi, [r15 + rbx]
    xor eax, eax
    xor ecx, ecx
.Lclear:
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    cmp rcx, rdi
    jb .Lclear
    xor ebp, ebp
.Louter:
    mov rdx, qword ptr [r12 + rbp*8]
    lea rdi, [r14 + rbp*8]
    xor esi, esi
    xor ecx, ecx
.Linner:
    mulx r10, r9, qword ptr [r13 + rcx*8]
    add r9, qword ptr [rdi + rcx*8]
    adc r10, 0
    add r9, rsi
    adc r10, 0
    mov qword ptr [rdi + rcx*8], r9
    mov rsi, r10
    inc rcx
    cmp rcx, rbx
    jb .Linner
    mov qword ptr [rdi + rcx*8], rsi
    inc rbp
    cmp rbp, r15
    jb .Louter
    lea rax, [r15 + rbx]
.Ltrim:
    test rax, rax
    jz .Lrestore
    dec rax
    cmp qword ptr [r14 + rax*8], 0
    jz .Ltrim
    inc rax
.Lrestore:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lzero:
    add rsi, rcx
    xor eax, eax
    test rsi, rsi
    jz .Lzero_ret
.Lzero_loop:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rsi
    jb .Lzero_loop
    xor eax, eax
.Lzero_ret:
    ret
