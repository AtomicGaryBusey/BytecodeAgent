export function l $fill_down(l %a, l %n, l %start) {
@start
	%iz =l copy 0
	jmp @head
@head
	%i =l phi @start %iz, @body %inext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%val =l sub %start, %i
	storel %val, %p
	%inext =l add %i, 1
	jmp @head
@end
	ret %n
}
