export function l $qsort_pairs(l %a, l %n) {
@start
	%gt1 =w csgtl %n, 1
	jnz %gt1, @body, @ret0
@ret0
	ret 0
@body
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%paddr =l add %a, %poff
	%pk =l loadl %paddr
	%pvaddr =l add %paddr, 8
	%pv =l loadl %pvaddr
	jmp @loop
@loop
	%j =l phi @body 0, @cont %jnext
	%i =l phi @body 0, @cont %inext
	%jlt =w csltl %j, %nm1
	jnz %jlt, @check, @after
@check
	%j16 =l mul %j, 16
	%jaddr =l add %a, %j16
	%kj =l loadl %jaddr
	%jvaddr =l add %jaddr, 8
	%vj =l loadl %jvaddr
	%klt =w csltl %kj, %pk
	%keq =w ceql %kj, %pk
	%vle =w cslel %vj, %pv
	%eqle =w and %keq, %vle
	%le =w or %klt, %eqle
	jnz %le, @doswap, @noswap
@doswap
	%i16 =l mul %i, 16
	%iaddr =l add %a, %i16
	%ik =l loadl %iaddr
	%ivaddr =l add %iaddr, 8
	%iv =l loadl %ivaddr
	storel %kj, %iaddr
	storel %vj, %ivaddr
	storel %ik, %jaddr
	storel %iv, %jvaddr
	%i1 =l add %i, 1
	jmp @cont
@noswap
	%i0 =l copy %i
	jmp @cont
@cont
	%inext =l phi @doswap %i1, @noswap %i0
	%jnext =l add %j, 1
	jmp @loop
@after
	%i16b =l mul %i, 16
	%ia =l add %a, %i16b
	%ik2 =l loadl %ia
	%iva =l add %ia, 8
	%iv2 =l loadl %iva
	storel %pk, %ia
	storel %pv, %iva
	storel %ik2, %paddr
	storel %iv2, %pvaddr
	%left =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%rcount =l sub %n, %ip1
	%roff =l mul %ip1, 16
	%rp =l add %a, %roff
	%right =l call $qsort_pairs(l %rp, l %rcount)
	ret 0
}
