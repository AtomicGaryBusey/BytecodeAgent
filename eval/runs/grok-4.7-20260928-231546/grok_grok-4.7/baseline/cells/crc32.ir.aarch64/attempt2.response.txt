export function l $crc32(l %p, l %n) {
@start
	%crcinit =w copy -1
	%izero =l copy 0
	%poly =w copy -306674912
	jmp @loop
@loop
	%i =l phi @start %izero, @bitend %inext
	%crc =w phi @start %crcinit, @bitend %crcbyte
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%ptr =l add %p, %i
	%byte =w loadub %ptr
	%crc1 =w xor %crc, %byte
	%kzero =l copy 0
	jmp @bitloop
@bitloop
	%k =l phi @body %kzero, @bitafter %k3
	%c =w phi @body %crc1, @bitafter %cnext
	%kcmp =w csltl %k, 8
	jnz %kcmp, @bitbody, @bitend
@bitbody
	%lsb =w and %c, 1
	jnz %lsb, @xorpath, @shrpath
@xorpath
	%shrx =w shr %c, 1
	%cx =w xor %shrx, %poly
	jmp @bitafter
@shrpath
	%cs =w shr %c, 1
	jmp @bitafter
@bitafter
	%cnext =w phi @xorpath %cx, @shrpath %cs
	%k3 =l add %k, 1
	jmp @bitloop
@bitend
	%crcbyte =w copy %c
	%inext =l add %i, 1
	jmp @loop
@done
	%crcout =w xor %crc, -1
	%result =l extuw %crcout
	ret %result
}
