export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_init, @bwd_init
@fwd_init
	%i0 =l copy 0
	jmp @fwd_test
@fwd_test
	%i =l phi @fwd_init %i0, @fwd_body %i1
	%cont =w csltl %i, %n
	jnz %cont, @fwd_body, @done
@fwd_body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%da =l add %dst, %off
	%v =l loadl %sa
	storel %v, %da
	%i1 =l add %i, 1
	jmp @fwd_test
@bwd_init
	%j0 =l copy %n
	jmp @bwd_test
@bwd_test
	%j =l phi @bwd_init %j0, @bwd_body %j1
	%bcont =w csgtl %j, 0
	jnz %bcont, @bwd_body, @done
@bwd_body
	%j1 =l sub %j, 1
	%boff =l mul %j1, 8
	%bsa =l add %src, %boff
	%bda =l add %dst, %boff
	%bv =l loadl %bsa
	storel %bv, %bda
	jmp @bwd_test
@done
	%z =l copy 0
	ret %z
}
