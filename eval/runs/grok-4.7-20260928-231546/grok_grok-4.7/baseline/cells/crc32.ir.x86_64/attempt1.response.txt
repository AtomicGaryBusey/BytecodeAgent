export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%crc =w phi @start -1, @byte %crc8
	%i =l phi @start 0, @byte %i2
	%cmp =w csltl %i, %n
	jnz %cmp, @byte, @finish
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%c0 =w xor %crc, %b
	%bit0 =w and %c0, 1
	%mask0 =w neg %bit0
	%shr0 =w shr %c0, 1
	%pm0 =w and %mask0, 0xedb88320
	%c1 =w xor %shr0, %pm0
	%bit1 =w and %c1, 1
	%mask1 =w neg %bit1
	%shr1 =w shr %c1, 1
	%pm1 =w and %mask1, 0xedb88320
	%c2 =w xor %shr1, %pm1
	%bit2 =w and %c2, 1
	%mask2 =w neg %bit2
	%shr2 =w shr %c2, 1
	%pm2 =w and %mask2, 0xedb88320
	%c3 =w xor %shr2, %pm2
	%bit3 =w and %c3, 1
	%mask3 =w neg %bit3
	%shr3 =w shr %c3, 1
	%pm3 =w and %mask3, 0xedb88320
	%c4 =w xor %shr3, %pm3
	%bit4 =w and %c4, 1
	%mask4 =w neg %bit4
	%shr4 =w shr %c4, 1
	%pm4 =w and %mask4, 0xedb88320
	%c5 =w xor %shr4, %pm4
	%bit5 =w and %c5, 1
	%mask5 =w neg %bit5
	%shr5 =w shr %c5, 1
	%pm5 =w and %mask5, 0xedb88320
	%c6 =w xor %shr5, %pm5
	%bit6 =w and %c6, 1
	%mask6 =w neg %bit6
	%shr6 =w shr %c6, 1
	%pm6 =w and %mask6, 0xedb88320
	%c7 =w xor %shr6, %pm6
	%bit7 =w and %c7, 1
	%mask7 =w neg %bit7
	%shr7 =w shr %c7, 1
	%pm7 =w and %mask7, 0xedb88320
	%crc8 =w xor %shr7, %pm7
	%i2 =l add %i, 1
	jmp @outer
@finish
	%out =w xor %crc, -1
	%ret =l extuw %out
	ret %ret
}
