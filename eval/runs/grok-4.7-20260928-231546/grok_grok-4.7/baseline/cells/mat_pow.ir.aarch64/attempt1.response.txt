export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%kz =w ceql %k, 0
	jnz %kz, @ident, @copy_a
@ident
	%i0 =l copy 0
	jmp @ident_i
@ident_i
	%i =l phi @ident %i0, @ident_i_inc %i1
	%ilt =w csltl %i, %n
	jnz %ilt, @ident_j_init, @zero_init
@ident_j_init
	%j0 =l copy 0
	jmp @ident_j
@ident_j
	%j =l phi @ident_j_init %j0, @ident_j_body %j1
	%jlt =w csltl %j, %n
	jnz %jlt, @ident_j_body, @ident_i_inc
@ident_j_body
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, 8
	%addr =l add %out, %off
	%eq =w ceql %i, %j
	%val =l extuw %eq
	storel %val, %addr
	%j1 =l add %j, 1
	jmp @ident_j
@ident_i_inc
	%i1 =l add %i, 1
	jmp @ident_i
@copy_a
	%c0 =l copy 0
	jmp @copy_loop
@copy_loop
	%ci =l phi @copy_a %c0, @copy_body %c1
	%clt =w csltl %ci, %nn
	jnz %clt, @copy_body, @pow_init
@copy_body
	%coff =l mul %ci, 8
	%ca =l add %a, %coff
	%cv =l loadl %ca
	%co =l add %out, %coff
	storel %cv, %co
	%c1 =l add %ci, 1
	jmp @copy_loop
@pow_init
	%p0 =l copy 1
	jmp @pow_loop
@pow_loop
	%p =l phi @pow_init %p0, @pow_after %p1
	%plt =w csltl %p, %k
	jnz %plt, @mul_i_init, @zero_init
@mul_i_init
	%mi0 =l copy 0
	jmp @mul_i
@mul_i
	%mi =l phi @mul_i_init %mi0, @mul_i_inc %mi1
	%milt =w csltl %mi, %n
	jnz %milt, @mul_j_init, @copyback_init
@mul_j_init
	%mj0 =l copy 0
	jmp @mul_j
@mul_j
	%mj =l phi @mul_j_init %mj0, @mul_store %mj1
	%mjlt =w csltl %mj, %n
	jnz %mjlt, @mul_t_init, @mul_i_inc
@mul_t_init
	%mt0 =l copy 0
	%sum0 =l copy 0
	jmp @mul_t
@mul_t
	%mt =l phi @mul_t_init %mt0, @mul_t_body %mt1
	%sum =l phi @mul_t_init %sum0, @mul_t_body %sum1
	%mtlt =w csltl %mt, %n
	jnz %mtlt, @mul_t_body, @mul_store
@mul_t_body
	%row =l mul %mi, %n
	%oi =l add %row, %mt
	%oo =l mul %oi, 8
	%oa =l add %out, %oo
	%ov =l loadl %oa
	%tr =l mul %mt, %n
	%ai =l add %tr, %mj
	%ao =l mul %ai, 8
	%aa =l add %a, %ao
	%av =l loadl %aa
	%prod =l mul %ov, %av
	%sum1 =l add %sum, %prod
	%mt1 =l add %mt, 1
	jmp @mul_t
@mul_store
	%srow =l mul %mi, %n
	%si =l add %srow, %mj
	%so =l mul %si, 8
	%sa =l add %tmp, %so
	storel %sum, %sa
	%mj1 =l add %mj, 1
	jmp @mul_j
@mul_i_inc
	%mi1 =l add %mi, 1
	jmp @mul_i
@copyback_init
	%b0 =l copy 0
	jmp @copyback
@copyback
	%bi =l phi @copyback_init %b0, @copyback_body %b1
	%blt =w csltl %bi, %nn
	jnz %blt, @copyback_body, @pow_after
@copyback_body
	%bo =l mul %bi, 8
	%ba =l add %tmp, %bo
	%bv =l loadl %ba
	%bd =l add %out, %bo
	storel %bv, %bd
	%b1 =l add %bi, 1
	jmp @copyback
@pow_after
	%p1 =l add %p, 1
	jmp @pow_loop
@zero_init
	%tmpsz =l mul %nn, 2
	%z0 =l copy 0
	jmp @zero_loop
@zero_loop
	%zi =l phi @zero_init %z0, @zero_body %z1
	%zlt =w csltl %zi, %tmpsz
	jnz %zlt, @zero_body, @done
@zero_body
	%zo =l mul %zi, 8
	%za =l add %tmp, %zo
	%zz =l copy 0
	storel %zz, %za
	%z1 =l add %zi, 1
	jmp @zero_loop
@done
	%r =l copy 0
	ret %r
}
