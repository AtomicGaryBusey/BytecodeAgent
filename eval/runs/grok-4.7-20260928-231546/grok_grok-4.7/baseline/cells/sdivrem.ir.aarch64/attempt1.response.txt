export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @byzero, @chk
@byzero
	storel %a, %rem
	%z =l copy 0
	ret %z
@chk
	%n1 =w ceql %b, -1
	jnz %n1, @chka, @div
@chka

	%imin =l shl %one, 63
	%am =w ceql %a, %imin
	jnz %am, @ovf, @div
@ovf
	%zr =l copy 0
	storel %zr, %rem
	ret %imin
@div

	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
