memmove_i64:
xor rax, rax
test rdx, rdx
jz .Lret
cmp rdi, rsi
ja .Lbackward
xor rcx, rcx
.Lforward:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
cmp rcx, rdx
jb .Lforward
ret
.Lbackward:
lea rcx, [rdx - 1]
.Lbackloop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
dec rcx
jns .Lbackloop
.Lret:
ret
