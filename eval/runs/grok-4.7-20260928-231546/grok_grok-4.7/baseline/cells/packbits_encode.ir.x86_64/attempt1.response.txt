export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @rep_wr %i_rep, @cpy_done %i_lit
	%wri =l phi @start 0, @rep_wr %wri_rep, @cpy_done %wri_lit
	%atend =w ceql %i, %n
	jnz %atend, @success, @runi_init
@runi_init
	%ipa =l add %src, %i
	%ib0 =w loadub %ipa
	jmp @runi_loop
@runi_loop
	%ic =l phi @runi_init 0, @runi_inc %ic2
	%ilt128 =w csltl %ic, 128
	jnz %ilt128, @runi_ckn, @runi_outc
@runi_ckn
	%iidx =l add %i, %ic
	%iltn =w csltl %iidx, %n
	jnz %iltn, @runi_ckeq, @runi_outn
@runi_ckeq
	%iq =l add %src, %iidx
	%iv =w loadub %iq
	%ieq =w ceqw %iv, %ib0
	jnz %ieq, @runi_inc, @runi_outd
@runi_inc
	%ic2 =l add %ic, 1
	jmp @runi_loop
@runi_outc
	%ir1 =l copy %ic
	jmp @runi_join
@runi_outn
	%ir2 =l copy %ic
	jmp @runi_join
@runi_outd
	%ir3 =l copy %ic
	jmp @runi_join
@runi_join
	%runi =l phi @runi_outc %ir1, @runi_outn %ir2, @runi_outd %ir3
	%isrep =w csgel %runi, 3
	jnz %isrep, @rep_chk, @lit_head
@rep_chk
	%wri_rep =l add %wri, 2
	%repok =w cslel %wri_rep, %cap
	jnz %repok, @rep_wr, @fail
@rep_wr
	%hdrl =l sub 257, %runi
	%hdrw =w copy %hdrl
	%dp0 =l add %out, %wri
	storeb %hdrw, %dp0
	%sp0 =l add %src, %i
	%bv0 =w loadub %sp0
	%dp1 =l add %dp0, 1
	storeb %bv0, %dp1
	%i_rep =l add %i, %runi
	jmp @main
@lit_head
	%j =l phi @runi_join %i, @lit_adv %jnxt
	%jltn =w csltl %j, %n
	jnz %jltn, @lit_chklen, @lit_fromn
@lit_fromn
	%jf1 =l copy %j
	jmp @lit_emit
@lit_chklen
	%jrel =l sub %j, %i
	%jlt128 =w csltl %jrel, 128
	jnz %jlt128, @litj_init, @lit_fromlen
@lit_fromlen
	%jf2 =l copy %j
	jmp @lit_emit
@litj_init
	%jpa =l add %src, %j
	%jb0 =w loadub %jpa
	jmp @litj_loop
@litj_loop
	%jc =l phi @litj_init 0, @litj_inc %jc2
	%jclt128 =w csltl %jc, 128
	jnz %jclt128, @litj_ckn, @litj_outc
@litj_ckn
	%jidx =l add %j, %jc
	%jidxlt =w csltl %jidx, %n
	jnz %jidxlt, @litj_ckeq, @litj_outn
@litj_ckeq
	%jq =l add %src, %jidx
	%jv =w loadub %jq
	%jeq =w ceqw %jv, %jb0
	jnz %jeq, @litj_inc, @litj_outd
@litj_inc
	%jc2 =l add %jc, 1
	jmp @litj_loop
@litj_outc
	%jr1 =l copy %jc
	jmp @litj_join
@litj_outn
	%jr2 =l copy %jc
	jmp @litj_join
@litj_outd
	%jr3 =l copy %jc
	jmp @litj_join
@litj_join
	%runj =l phi @litj_outc %jr1, @litj_outn %jr2, @litj_outd %jr3
	%jlt3 =w csltl %runj, 3
	jnz %jlt3, @lit_adv, @lit_fromrun
@lit_adv
	%jnxt =l add %j, 1
	jmp @lit_head
@lit_fromrun
	%jf3 =l copy %j
	jmp @lit_emit
@lit_emit
	%jend =l phi @lit_fromn %jf1, @lit_fromlen %jf2, @lit_fromrun %jf3
	%len =l sub %jend, %i
	%need =l add %len, 1
	%wri_lit =l add %wri, %need
	%litok =w cslel %wri_lit, %cap
	jnz %litok, @lit_hdr, @fail
@lit_hdr
	%hm1 =l sub %len, 1
	%hm1w =w copy %hm1
	%hp =l add %out, %wri
	storeb %hm1w, %hp
	jmp @cpy_loop
@cpy_loop
	%k =l phi @lit_hdr 0, @cpy_body %k2
	%klt =w csltl %k, %len
	jnz %klt, @cpy_body, @cpy_done
@cpy_body
	%si =l add %i, %k
	%spb =l add %src, %si
	%sb =w loadub %spb
	%off =l add %wri, 1
	%off2 =l add %off, %k
	%dpb =l add %out, %off2
	storeb %sb, %dpb
	%k2 =l add %k, 1
	jmp @cpy_loop
@cpy_done
	%i_lit =l copy %jend
	jmp @main
@success
	ret %wri
@fail
	ret -1
}
