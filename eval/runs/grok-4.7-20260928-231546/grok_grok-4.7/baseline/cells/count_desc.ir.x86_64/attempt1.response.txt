export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 8
	%end =l add %a, %off
	jmp @loop
@loop
	%p =l phi @init %a, @body %pn
	%cnt =l phi @init 0, @body %cntn
	%done =w ceql %p, %end
	jnz %done, @out, @body
@body
	%x =l loadl %p
	%pn =l add %p, 8
	%y =l loadl %pn
	%gt =w csgtl %x, %y
	%gtl =l extuw %gt
	%cntn =l add %cnt, %gtl
	jmp @loop
@out
	ret %cnt
}
