export function l $count_ugt(l %a, l %n, l %t) {
@start
	%zero =l copy 0
	jmp @head
@head
	%i =l phi @start %zero, @body %inext
	%cnt =l phi @start %zero, @body %cntnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l add %i, 1
	jmp @head
@end
	ret %cnt
}
