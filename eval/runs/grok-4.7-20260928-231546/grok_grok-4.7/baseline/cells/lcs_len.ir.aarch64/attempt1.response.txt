export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	jmp @zcond
@zcond
	%zi =l phi @start %zero, @zbody %zinext
	%zdone =w csgtl %zi, %n
	jnz %zdone, @iloop, @zbody
@zbody
	%zoff =l shl %zi, 3
	%zp =l add %row, %zoff
	storel %zero, %zp
	%zinext =l add %zi, %one
	jmp @zcond
@iloop
	%i =l phi @zcond %zero, @iend %inext
	%idone =w csgel %i, %m
	jnz %idone, @done, @ibody
@ibody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @jcond
@jcond
	%j =l phi @ibody %one, @jstore %jnext
	%diag =l phi @ibody %zero, @jstore %diagout
	%jdone =w csgtl %j, %n
	jnz %jdone, @iend, @jbody
@jbody
	%joff =l shl %j, 3
	%jp =l add %row, %joff
	%up =l loadl %jp
	%jm1 =l sub %j, %one
	%jm1off =l shl %jm1, 3
	%jm1p =l add %row, %jm1off
	%left =l loadl %jm1p
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%sum =l add %diag, %one
	jmp @jstore
@nomatch
	%gt =w csgtl %left, %up
	jnz %gt, @takeleft, @takeup
@takeleft
	%nvl =l copy %left
	jmp @jstore
@takeup
	%nvu =l copy %up
	jmp @jstore
@jstore
	%newv =l phi @match %sum, @takeleft %nvl, @takeup %nvu
	storel %newv, %jp
	%diagout =l copy %up
	%jnext =l add %j, %one
	jmp @jcond
@iend
	%inext =l add %i, %one
	jmp @iloop
@done
	%noff =l shl %n, 3
	%np =l add %row, %noff
	%res =l loadl %np
	ret %res
}
