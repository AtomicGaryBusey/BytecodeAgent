export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %n =l add %na, %nb
    jmp @zero_head
@zero_head
    %zi =l phi @start 0, @zero_body %zi_next
    %zc =w csltl %zi, %n
    jnz %zc, @zero_body, @outer_head
@zero_body
    %zoff =l shl %zi, 3
    %zp =l add %r, %zoff
    storel 0, %zp
    %zi_next =l add %zi, 1
    jmp @zero_head
@outer_head
    %i =l phi @zero_head 0, @inner_done %i_next
    %ic =w csltl %i, %na
    jnz %ic, @outer_body, @to_scan
@to_scan
    %n_scan =l copy %n
    jmp @scan_head
@outer_body
    %ioff =l shl %i, 3
    %ap =l add %a, %ioff
    %ai =l loadl %ap
    jmp @inner_head
@inner_head
    %j =l phi @outer_body 0, @inner_body %j_next
    %carry =l phi @outer_body 0, @inner_body %carry_next
    %jc =w csltl %j, %nb
    jnz %jc, @inner_body, @inner_done
@inner_body
    %joff =l shl %j, 3
    %bp =l add %b, %joff
    %bj =l loadl %bp
    %a0 =l and %ai, 4294967295
    %a1 =l shr %ai, 32
    %b0 =l and %bj, 4294967295
    %b1 =l shr %bj, 32
    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1
    %mid =l add %p1, %p2
    %midovw =w cultl %mid, %p1
    %midov =l extuw %midovw
    %midsl =l shl %mid, 32
    %plo =l add %p0, %midsl
    %clow =w cultl %plo, %p0
    %clo =l extuw %clow
    %midsh =l shr %mid, 32
    %xtra =l shl %midov, 32
    %hi0 =l add %p3, %midsh
    %hi1 =l add %hi0, %xtra
    %prodhi =l add %hi1, %clo
    %idx =l add %i, %j
    %roff =l shl %idx, 3
    %rp =l add %r, %roff
    %rv =l loadl %rp
    %s1 =l add %rv, %plo
    %c1w =w cultl %s1, %rv
    %c1 =l extuw %c1w
    %s2 =l add %s1, %carry
    %c2w =w cultl %s2, %s1
    %c2 =l extuw %c2w
    storel %s2, %rp
    %c12 =l add %c1, %c2
    %carry_next =l add %prodhi, %c12
    %j_next =l add %j, 1
    jmp @inner_head
@inner_done
    %endi =l add %i, %nb
    %endoff =l shl %endi, 3
    %endp =l add %r, %endoff
    storel %carry, %endp
    %i_next =l add %i, 1
    jmp @outer_head
@scan_head
    %k =l phi @to_scan %n_scan, @scan_dec %k_next
    %kzero =w ceql %k, 0
    jnz %kzero, @ret_zero, @scan_dec
@scan_dec
    %k_next =l sub %k, 1
    %koff =l shl %k_next, 3
    %kp =l add %r, %koff
    %kv =l loadl %kp
    %knz =w cnel %kv, 0
    jnz %knz, @ret_found, @scan_head
@ret_found
    %kret =l add %k_next, 1
    ret %kret
@ret_zero
    ret 0
}
