export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @do %i1
	%cnt =l phi @start 0, @do %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @do, @end
@do
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%sv =l loadl %sp
	%rv =l call %f(l %sv)
	storel %rv, %dp
	%lt =w csltl %rv, %sv
	%ltl =l extuw %lt
	%cnt1 =l add %cnt, %ltl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
