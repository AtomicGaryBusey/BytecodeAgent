dot_i32_simd:
    pxor xmm2, xmm2
    mov rcx, rdx
    shr rcx, 2
    jz .Lfold
.Lvec:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    movdqa xmm3, xmm0
    movdqa xmm4, xmm1
    pmuldq xmm0, xmm1
    psrldq xmm3, 4
    psrldq xmm4, 4
    pmuldq xmm3, xmm4
    paddq xmm2, xmm0
    paddq xmm2, xmm3
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lvec
.Lfold:
    pextrq rax, xmm2, 0
    pextrq rcx, xmm2, 1
    add rax, rcx
    and rdx, 3
    jz .Ldone
.Lscalar:
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Lscalar
.Ldone:
    ret
