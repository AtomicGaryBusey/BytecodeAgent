export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%iz =l copy 0
	%wz =l copy 0
	jmp @loop
@loop
	%i =l phi @start %iz, @emit_rep_ok %i_r, @copy_done %i_l
	%wr =l phi @start %wz, @emit_rep_ok %wr_r, @copy_done %wr_l
	%atend =w csgel %i, %n
	jnz %atend, @ret_ok, @measure_run
@measure_run
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%k0 =l add %i, 1
	%cinit =w copy 1
	jmp @runloop
@runloop
	%k =l phi @measure_run %k0, @run_inc %k1
	%c =w phi @measure_run %cinit, @run_inc %c1
	%kend =w csgel %k, %n
	jnz %kend, @run_done, @run_checkcap
@run_checkcap
	%ccap =w csgew %c, 128
	jnz %ccap, @run_done, @run_checkbyte
@run_checkbyte
	%pk =l add %src, %k
	%bk =w loadub %pk
	%beq =w ceqw %bk, %b0
	jnz %beq, @run_inc, @run_done
@run_inc
	%k1 =l add %k, 1
	%c1 =w add %c, 1
	jmp @runloop
@run_done
	%isrep =w csgew %c, 3
	jnz %isrep, @emit_rep, @lit_init
@emit_rep
	%need_r =l add %wr, 2
	%nofit_r =w csgtl %need_r, %cap
	jnz %nofit_r, @ret_fail, @emit_rep_ok
@emit_rep_ok
	%hdr_r =w sub 257, %c
	%op_r =l add %out, %wr
	storeb %hdr_r, %op_r
	%op_r1 =l add %op_r, 1
	storeb %b0, %op_r1
	%cl =l extuw %c
	%i_r =l add %i, %cl
	%wr_r =l add %wr, 2
	jmp @loop
@lit_init
	%llen0 =w copy 0
	jmp @lit_scan
@lit_scan
	%j =l phi @lit_init %i, @lit_adv %j1
	%llen =w phi @lit_init %llen0, @lit_adv %llen1
	%jend =w csgel %j, %n
	jnz %jend, @lit_emit, @lit_checklen
@lit_checklen
	%jtoolong =w csgew %llen, 128
	jnz %jtoolong, @lit_emit, @lit_run_init
@lit_run_init
	%pj =l add %src, %j
	%bj =w loadub %pj
	%jk0 =l add %j, 1
	%jc0 =w copy 1
	jmp @lit_runloop
@lit_runloop
	%jk =l phi @lit_run_init %jk0, @lit_run_inc %jk1
	%jc =w phi @lit_run_init %jc0, @lit_run_inc %jc1
	%jkend =w csgel %jk, %n
	jnz %jkend, @lit_run_done, @lit_
