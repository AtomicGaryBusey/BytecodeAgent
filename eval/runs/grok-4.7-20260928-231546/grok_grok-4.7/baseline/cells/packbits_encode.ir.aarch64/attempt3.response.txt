export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @rep_do %i2, @copy_done %j_end
	%wr =l phi @start 0, @rep_do %wr2, @copy_done %wr3
	%in =w csltl %i, %n
	jnz %in, @run1_init, @done
@done
	ret %wr
@run1_init
	%pi =l add %src, %i
	%bi =w loadub %pi
	%k10 =l add %i, 1
	jmp @run1
@run1
	%c1 =w phi @run1_init 1, @run1_inc %c1n
	%k1 =l phi @run1_init %k10, @run1_inc %k1n
	%c1lt =w csltw %c1, 128
	jnz %c1lt, @run1_n, @run1_end
@run1_n
	%k1lt =w csltl %k1, %n
	jnz %k1lt, @run1_ld, @run1_end
@run1_ld
	%pk1 =l add %src, %k1
	%bk1 =w loadub %pk1
	%eq1 =w ceqw %bi, %bk1
	jnz %eq1, @run1_inc, @run1_end
@run1_inc
	%c1n =w add %c1, 1
	%k1n =l add %k1, 1
	jmp @run1
@run1_end
	%ge3 =w csgew %c1, 3
	jnz %ge3, @rep_chk, @lit_cond
@rep_chk
	%need2 =l add %wr, 2
	%ok2 =w cslel %need2, %cap
	jnz %ok2, @rep_do, @fail
@rep_do
	%hbase =w copy 257
	%hdr_r =w sub %hbase, %c1
	%op =l add %out, %wr
	storeb %hdr_r, %op
	%op1 =l add %op, 1
	storeb %bi, %op1
	%wr2 =l add %wr, 2
	%c1l =l extuw %c1
	%i2 =l add %i, %c1l
	jmp @loop
@lit_cond
	%j =l phi @run1_end %i, @lit_adv %jn
	%lj =w phi @run1_end 0, @lit_adv %ljn
	%jlt =w csltl %j, %n
	jnz %jlt, @lit_len, @lit_emit
@lit_len
	%ljlt =w csltw %lj, 128
	jnz %ljlt, @run2_init, @lit_emit
@run2_init
	%pj =l add %src, %j
	%bj =w loadub %pj
	%k20 =l add %j, 1
	jmp @run2
@run2
	%c2 =w phi @run2_init 1, @run2_inc %c2n
	%k2 =l phi @run2_init %k20, @run2_inc %k2n
	%c2lt =w csltw %c2, 128
	jnz %c2lt, @run2_n, @run2_end
@run2_n
	%k2lt =w csltl %k2, %n
	jnz %k2lt, @run2_ld, @run2_end
@run2_ld
	%pk2 =l add %src, %k2
	%bk2 =w loadub %pk2
	%eq2 =w ceqw %bj, %bk2
	jnz %eq2, @run2_inc, @run2_end
@run2_inc
	%c2n =w add %c2, 1
	%k2n =l add %k2, 1
	jmp @run2
@run2_end
	%lt3 =w csltw %c2, 3
	jnz %lt3, @lit_adv, @lit_emit
@lit_adv
	%jn =l add %j, 1
	%ljn =w add %lj, 1
	jmp @lit_cond
@lit_emit
	%ljl =l extuw %lj
	%need =l add %wr, %ljl
	%need1 =l add %need, 1
	%okl =w cslel %need1, %cap
	jnz %okl, @lit_hdr, @fail
@lit_hdr
	%hdr_l =w sub %lj, 1
	%obase =l add %out, %wr
	storeb %hdr_l, %obase
	%dst =l add %obase, 1
	jmp @copy
@copy
	%off =l phi @lit_hdr 0, @copy_step %offn
	%offlt =w csltl %off, %ljl
	jnz %offlt, @copy_b, @copy_done
@copy_b
	%soff =l add %i, %off
	%sp =l add %src, %soff
	%bv =w loadub %sp
	%dp =l add %dst, %off
	storeb %bv, %dp
	jmp @copy_step
@copy_step
	%offn =l add %off, 1
	jmp @copy
@copy_done
	%j_end =l copy %j
	%wr3 =l copy %need1
	jmp @loop
@fail
	%m1 =l copy -1
	ret %m1
}
