export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%ge =w csgel %i, %n
	jnz %ge, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @end, @next
@next
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %i
}
