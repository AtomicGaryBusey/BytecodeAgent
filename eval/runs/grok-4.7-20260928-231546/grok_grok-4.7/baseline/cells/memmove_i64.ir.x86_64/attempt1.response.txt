export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nzero =w ceql %n, 0
	jnz %nzero, @ret, @cmp
@cmp
	%before =w cultl %dst, %src
	jnz %before, @fwdinit, @bwdinit
@fwdinit
	%iz =l copy 0
	jmp @fwd
@fwd
	%i =l phi @fwdinit %iz, @fwdstep %inext
	%fdone =w ceql %i, %n
	jnz %fdone, @ret, @fwdstep
@fwdstep
	%fo =l mul %i, 8
	%fp =l add %src, %fo
	%tp =l add %dst, %fo
	%fv =l loadl %fp
	storel %fv, %tp
	%inext =l add %i, 1
	jmp @fwd
@bwdinit
	%one =l copy 1
	%jz =l sub %n, %one
	jmp @bwd
@bwd
	%j =l phi @bwdinit %jz, @bwdstep %jnext
	%bo =l mul %j, 8
	%bp =l add %src, %bo
	%dp =l add %dst, %bo
	%bv =l loadl %bp
	storel %bv, %dp
	%jdone =w ceql %j, 0
	jnz %jdone, @ret, @bwdstep
@bwdstep
	%jnext =l sub %j, 1
	jmp @bwd
@ret
	%rv =l copy 0
	ret %rv
}
