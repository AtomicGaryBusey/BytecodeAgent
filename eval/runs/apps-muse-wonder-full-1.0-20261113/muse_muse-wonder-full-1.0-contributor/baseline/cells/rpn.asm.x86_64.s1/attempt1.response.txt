.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 70000
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbx, [rip + input_buf]
    mov r13, rbx
    add r13, r12
outer_loop:
    cmp rbx, r13
    jae prog_exit
    mov r14, rbx
find_line_end:
    cmp r14, r13
    jae line_end_found
    cmp byte ptr [r14], 10
    je line_end_found
    inc r14
    jmp find_line_end
line_end_found:
    mov r15, r14
    cmp r14, r13
    jae line_setup
    inc r15
line_setup:
    xor r12d, r12d
token_loop:
skip_sep_loop:
    cmp rbx, r14
    jae line_done
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je sep_inc
    cmp al, 9
    je sep_inc
    jmp token_found
sep_inc:
    inc rbx
    jmp skip_sep_loop
token_found:
    mov r8, rbx
    mov r9, rbx
find_token_end:
    cmp r9, r14
    jae token_end_found
    movzx eax, byte ptr [r9]
    cmp al, 32
    je token_end_found
    cmp al, 9
    je token_end_found
    inc r9
    jmp find_token_end
token_end_found:
    mov rbx, r9
    mov rcx, r9
    sub rcx, r8
    cmp rcx, 1
    jne parse_literal
    movzx eax, byte ptr [r8]
    cmp al, 43
    je do_operator
    cmp al, 45
    je do_operator
    cmp al, 42
    je do_operator
    cmp al, 47
    je do_operator
    cmp al, 37
    je do_operator
    jmp parse_literal
do_operator:
    mov r10d, eax
    cmp r12, 2
    jb error_line
    lea rsi, [rip + eval_stack]
    mov r11, qword ptr [rsi + r12*8 - 8]
    mov rax, qword ptr [rsi + r12*8 - 16]
    cmp r10d, 43
    je op_add
    cmp r10d, 45
    je op_sub
    cmp r10d, 42
    je op_mul
    cmp r10d, 47
    je op_div
    jmp op_mod
op_add:
    add rax, r11
    jmp op_store
op_sub:
    sub rax, r11
    jmp op_store
op_mul:
    imul rax, r11
    jmp op_store
op_div:
    test r11, r11
    jz error_line
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne op_div_ok
    cmp r11, -1
    je error_line
op_div_ok:
    cqo
    idiv r11
    jmp op_store
op_mod:
    test r11, r11
    jz error_line
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne op_mod_ok
    cmp r11, -1
    je error_line
op_mod_ok:
    cqo
    idiv r11
    mov rax, rdx
    jmp op_store
op_store:
    mov qword ptr [rsi + r12*8 - 16], rax
    dec r12
    jmp token_loop
parse_literal:
    xor r10d, r10d
    mov rsi, r8
    cmp byte ptr [rsi], 45
    jne lit_no_sign
    mov r10d, 1
    inc rsi
lit_no_sign:
    mov rcx, r9
    sub rcx, rsi
    test rcx, rcx
    jz error_line
    cmp rcx, 18
    ja error_line
    xor eax, eax
lit_digit_loop:
    cmp rsi, r9
    jae lit_done
    movzx edx, byte ptr [rsi]
    cmp dl, 48
    jb error_line
    cmp dl, 57
    ja error_line
    imul rax, rax, 10
    sub edx, 48
    add rax, rdx
    inc rsi
    jmp lit_digit_loop
lit_done:
    test r10d, r10d
    jz lit_push
    neg rax
lit_push:
    lea rsi, [rip + eval_stack]
    mov qword ptr [rsi + r12*8], rax
    inc r12
    jmp token_loop
line_done:
    cmp r12, 1
    jne error_line
    lea rsi, [rip + eval_stack]
    mov rax, qword ptr [rsi]
    xor r8d, r8d
    test rax, rax
    jns fmt_pos
    mov r8d, 1
    neg rax
fmt_pos:
    lea rsi, [rip + out_buf]
    add rsi, 32
    mov r9, rsi
    dec rsi
    mov byte ptr [rsi], 10
    test rax, rax
    jnz fmt_loop
    dec rsi
    mov byte ptr [rsi], 48
    jmp fmt_sign
fmt_loop:
    test rax, rax
    jz fmt_sign
    xor edx, edx
    mov r10d, 10
    div r10
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    jmp fmt_loop
fmt_sign:
    test r8d, r8d
    jz fmt_write
    dec rsi
    mov byte ptr [rsi], 45
fmt_write:
    mov rdx, r9
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line
error_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 6
    syscall
    jmp next_line
next_line:
    mov rbx, r15
    jmp outer_loop
prog_exit:
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata
err_msg:
    .ascii "error"
    .byte 10
.bss
    .balign 8
input_buf:
    .zero 70000
    .balign 8
eval_stack:
    .zero 16384
    .balign 8
out_buf:
    .zero 32
