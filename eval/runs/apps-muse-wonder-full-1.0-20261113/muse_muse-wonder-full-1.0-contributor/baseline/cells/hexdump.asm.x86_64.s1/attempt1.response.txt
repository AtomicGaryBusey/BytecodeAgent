.globl _start
.text
_start:
    xor r12, r12
    lea rbx, [rip + input_buf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
row_loop:
    cmp r13, r12
    jae final_line
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe 1f
    mov eax, 16
1:
    mov r14, rax
    lea rdi, [rip + line_buf]
    mov eax, r13d
    call put_hex8
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    xor r15, r15
pos_loop:
    cmp r15, 16
    jae pos_done
    cmp r15, r14
    jae pos_blank
    mov rcx, r13
    add rcx, r15
    movzx eax, byte ptr [rbx + rcx]
    call put_hex2
    mov byte ptr [rdi], 32
    inc rdi
    jmp pos_next
pos_blank:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
pos_next:
    cmp r15, 7
    jne 1f
    mov byte ptr [rdi], 32
    inc rdi
1:
    inc r15
    jmp pos_loop
pos_done:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    xor r15, r15
ascii_loop:
    cmp r15, r14
    jae ascii_done
    mov rcx, r13
    add rcx, r15
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, 32
    jb ascii_dot
    cmp eax, 126
    ja ascii_dot
    mov byte ptr [rdi], al
    inc rdi
    jmp ascii_next
ascii_dot:
    mov byte ptr [rdi], 46
    inc rdi
ascii_next:
    inc r15
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    call write_all
    add r13, r14
    jmp row_loop
final_line:
    lea rdi, [rip + line_buf]
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    call write_all
    mov eax, 60
    xor edi, edi
    syscall
put_hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    lea r8, [rip + hexdigits]
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 1b
    ret
put_hex2:
    mov edx, eax
    shr edx, 4
    and edx, 15
    lea r8, [rip + hexdigits]
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    ret
write_all:
    cmp rdx, 0
    je write_all_done
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle write_all_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
write_all_done:
    ret
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 128
