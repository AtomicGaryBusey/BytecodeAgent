.globl _start
.text
_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r15, [rbx + r12]
    mov r14, rbx
    mov r13, rbx
scan_loop:
    cmp r13, r15
    jae handle_final
    cmp byte ptr [r13], 10
    je handle_line
    inc r13
    jmp scan_loop
handle_line:
    mov r8, r14
    lea r9, [r13 - 1]
rev_loop1:
    cmp r8, r9
    jae rev_done1
    mov al, byte ptr [r8]
    mov cl, byte ptr [r9]
    mov byte ptr [r8], cl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp rev_loop1
rev_done1:
    mov rsi, r14
    mov rdx, r13
    sub rdx, r14
    call write_all
    lea rsi, [rip + newline_byte]
    mov edx, 1
    call write_all
    inc r13
    mov r14, r13
    jmp scan_loop
handle_final:
    cmp r14, r15
    jae do_exit
    mov r8, r14
    lea r9, [r15 - 1]
rev_loop2:
    cmp r8, r9
    jae rev_done2
    mov al, byte ptr [r8]
    mov cl, byte ptr [r9]
    mov byte ptr [r8], cl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp rev_loop2
rev_done2:
    mov rsi, r14
    mov rdx, r15
    sub rdx, r14
    call write_all
    lea rsi, [rip + newline_byte]
    mov edx, 1
    call write_all
do_exit:
    xor edi, edi
    mov eax, 60
    syscall
    jmp do_exit
write_all:
    test rdx, rdx
    jz write_ret
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_ret
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_ret:
    ret
.section .rodata
newline_byte:
    .byte 10
.bss
.balign 16
buffer:
    .zero 65536
