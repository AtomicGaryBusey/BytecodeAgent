.globl _start
.text
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, rbx
    xor edi, edi
    mov edx, 4096
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    xor r14d, r14d
    lea r15, [rip + input_buf]
parse_loop:
    cmp r12, rbx
    jae parse_done
    mov r13, r12
scan_loop:
    cmp r12, rbx
    jae line_eof
    movzx eax, byte ptr [r15 + r12]
    cmp al, 10
    je line_nl
    inc r12
    jmp scan_loop
line_nl:
    call store_line
    inc r12
    jmp parse_loop
line_eof:
    call store_line
    jmp parse_loop
parse_done:
    lea r15, [rip + lines_arr]
    mov r12d, 1
sort_outer:
    cmp r12, r14
    jae sort_done
    mov rax, r12
    shl rax, 4
    add rax, r15
    mov rbp, qword ptr [rax]
    mov rbx, qword ptr [rax + 8]
    mov r13, r12
    dec r13
sort_inner:
    cmp r13, 0
    jl sort_insert
    mov rax, r13
    shl rax, 4
    add rax, r15
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    mov rdi, rbp
    mov rcx, rbx
    call compare_lines
    test eax, eax
    jle sort_insert
    mov rax, r13
    shl rax, 4
    add rax, r15
    mov r8, qword ptr [rax]
    mov r9, qword ptr [rax + 8]
    mov qword ptr [rax + 16], r8
    mov qword ptr [rax + 24], r9
    dec r13
    jmp sort_inner
sort_insert:
    mov rax, r13
    inc rax
    shl rax, 4
    add rax, r15
    mov qword ptr [rax], rbp
    mov qword ptr [rax + 8], rbx
    inc r12
    jmp sort_outer
sort_done:
    xor r12d, r12d
output_loop:
    cmp r12, r14
    jae output_done
    mov rax, r12
    shl rax, 4
    add rax, r15
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    test rdx, rdx
    jz write_nl
    mov eax, 1
    mov edi, 1
    syscall
write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc r12
    jmp output_loop
output_done:
    mov eax, 60
    xor edi, edi
    syscall
store_line:
    lea rax, [rip + lines_arr]
    mov rcx, r14
    shl rcx, 4
    add rax, rcx
    lea rcx, [r15 + r13]
    mov qword ptr [rax], rcx
    mov rcx, r12
    sub rcx, r13
    mov qword ptr [rax + 8], rcx
    inc r14
    ret
compare_lines:
    mov r8, rdx
    cmp r8, rcx
    cmova r8, rcx
    xor r9d, r9d
compare_qword_loop:
    mov rax, r8
    sub rax, r9
    cmp rax, 8
    jb compare_byte_loop
    mov rax, qword ptr [rsi + r9]
    mov r10, qword ptr [rdi + r9]
    cmp rax, r10
    jne compare_byte_loop
    add r9, 8
    jmp compare_qword_loop
compare_byte_loop:
    cmp r9, r8
    jae compare_lengths
    movzx eax, byte ptr [rsi + r9]
    movzx r10d, byte ptr [rdi + r9]
    cmp eax, r10d
    jb compare_less
    ja compare_greater
    inc r9
    jmp compare_byte_loop
compare_lengths:
    cmp rdx, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret
.section .rodata
nl:
    .byte 10
.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
lines_arr:
    .zero 16000
