.globl _start
.text
_start:
lea rbx, [rip + input_buf]
xor r12d, r12d
read_loop:
mov eax, 0
xor edi, edi
mov rsi, rbx
add rsi, r12
mov edx, 70000
sub rdx, r12
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
xor r13d, r13d
line_loop:
cmp r13, r12
jae done_exit
mov r14, r13
scan_line:
cmp r13, r12
jae scan_done_eof
movzx eax, byte ptr [rbx + r13]
cmp al, 10
je scan_done_nl
inc r13
jmp scan_line
scan_done_nl:
mov r15, r13
inc r13
jmp process_line
scan_done_eof:
mov r15, r13
process_line:
xor ebp, ebp
mov r8, r14
token_loop:
cmp r8, r15
jae after_tokens
movzx eax, byte ptr [rbx + r8]
cmp al, 32
je sep_skip
cmp al, 9
je sep_skip
jmp token_start
sep_skip:
inc r8
jmp token_loop
token_start:
mov r10, r8
token_scan:
cmp r10, r15
jae token_found
movzx eax, byte ptr [rbx + r10]
cmp al, 32
je token_found
cmp al, 9
je token_found
inc r10
jmp token_scan
token_found:
mov rcx, r10
sub rcx, r8
cmp rcx, 1
jne check_literal
movzx eax, byte ptr [rbx + r8]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator
jmp check_literal
do_operator:
mov edi, eax
cmp rbp, 2
jb error_output
lea rsi, [rip + eval_stack]
lea rsi, [rsi + rbp*8]
mov rcx, qword ptr [rsi - 8]
mov rax, qword ptr [rsi - 16]
cmp edi, 43
je op_add
cmp edi, 45
je op_sub
cmp edi, 42
je op_mul
cmp edi, 47
je op_div
jmp op_mod
op_add:
add rax, rcx
jmp op_done
op_sub:
sub rax, rcx
jmp op_done
op_mul:
imul rax, rcx
jmp op_done
op_div:
test rcx, rcx
jz error_output
cmp rcx, -1
jne op_div_ok
mov rdx, rax
add rdx, rdx
jnz op_div_ok
test rax, rax
js error_output
op_div_ok:
cqo
idiv rcx
jmp op_done
op_mod:
test rcx, rcx
jz error_output
cmp rcx, -1
jne op_mod_ok
mov rdx, rax
add rdx, rdx
jnz op_mod_ok
test rax, rax
js error_output
op_mod_ok:
cqo
idiv rcx
mov rax, rdx
jmp op_done
op_done:
mov qword ptr [rsi - 16], rax
dec rbp
mov r8, r10
jmp token_loop
check_literal:
xor edi, edi
mov rsi, r8
movzx eax, byte ptr [rbx + r8]
cmp al, 45
jne literal_range_check
mov edi, 1
inc rsi
literal_range_check:
mov rcx, r10
sub rcx, rsi
cmp rcx, 1
jb error_output
cmp rcx, 18
ja error_output
xor eax, eax
literal_loop:
cmp rsi, r10
jae literal_done
movzx edx, byte ptr [rbx + rsi]
cmp dl, 48
jb error_output
cmp dl, 57
ja error_output
sub edx, 48
imul rax, rax, 10
add rax, rdx
inc rsi
jmp literal_loop
literal_done:
test edi, edi
jz literal_push
neg rax
literal_push:
lea rsi, [rip + eval_stack]
mov qword ptr [rsi + rbp*8], rax
inc rbp
mov r8, r10
jmp token_loop
after_tokens:
cmp rbp, 1
jne error_output
lea rsi, [rip + eval_stack]
mov rax, qword ptr [rsi]
lea rsi, [rip + num_buf]
add rsi, 32
dec rsi
mov byte ptr [rsi], 10
xor edi, edi
test rax, rax
jns format_digits
mov edi, 1
neg rax
format_digits:
mov ecx, 10
format_digit_loop:
xor edx, edx
div rcx
add edx, 48
dec rsi
mov byte ptr [rsi], dl
test rax, rax
jnz format_digit_loop
test edi, edi
jz write_number
dec rsi
mov byte ptr [rsi], 45
write_number:
lea rdx, [rip + num_buf]
add rdx, 32
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
jmp line_loop
error_output:
mov eax, 1
mov edi, 1
lea rsi, [rip + error_msg]
mov edx, 6
syscall
jmp line_loop
done_exit:
mov eax, 60
xor edi, edi
syscall
jmp done_exit
.section .rodata
error_msg:
.ascii "error"
.byte 10
.bss
input_buf:
.zero 70000
eval_stack:
.zero 16384
num_buf:
.zero 32
