.globl _start
.text
_start:
lea rbp, [rip + buffer]
xor ebx, ebx
read_loop:
mov edx, 65536
sub rdx, rbx
jz read_done
xor edi, edi
lea rsi, [rbp + rbx]
xor eax, eax
syscall
test rax, rax
jz read_done
js read_done
add rbx, rax
jmp read_loop
read_done:
xor r12d, r12d
process_loop:
cmp r12, rbx
jae done_all
mov r13, r12
find_loop:
cmp r13, rbx
jae find_eof
cmp byte ptr [rbp + r13], 10
je find_nl
inc r13
jmp find_loop
find_nl:
lea r14, [r13 + 1]
jmp have_line
find_eof:
mov r14, r13
have_line:
mov r8, r12
lea r9, [r13 - 1]
rev_loop:
cmp r8, r9
jge rev_done
movzx eax, byte ptr [rbp + r8]
movzx ecx, byte ptr [rbp + r9]
mov byte ptr [rbp + r8], cl
mov byte ptr [rbp + r9], al
inc r8
dec r9
jmp rev_loop
rev_done:
mov rdx, r13
sub rdx, r12
test rdx, rdx
jz write_nl
mov eax, 1
mov edi, 1
lea rsi, [rbp + r12]
syscall
write_nl:
mov eax, 1
mov edi, 1
lea rsi, [rip + newline_byte]
mov edx, 1
syscall
mov r12, r14
jmp process_loop
done_all:
mov eax, 60
xor edi, edi
syscall
.section .rodata
newline_byte:
.byte 10
.bss
.align 16
buffer:
.zero 65536
