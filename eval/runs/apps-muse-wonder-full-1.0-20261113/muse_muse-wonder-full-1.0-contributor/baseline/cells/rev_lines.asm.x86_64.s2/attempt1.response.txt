.globl _start
.section .rodata
newline_byte:
.byte 10
.bss
.balign 16
buffer:
.zero 65536
.text
_start:
lea r12, [rip + buffer]
xor r13, r13
read_loop:
cmp r13, 65536
jae read_done
mov rax, 0
xor edi, edi
lea rsi, [r12 + r13]
mov rdx, 65536
sub rdx, r13
syscall
test rax, rax
jle read_done
add r13, rax
jmp read_loop
read_done:
xor r14, r14
line_loop:
cmp r14, r13
jae done
mov r15, r14
find_loop:
cmp r15, r13
jae line_found
mov al, byte ptr [r12 + r15]
cmp al, 10
je line_found
inc r15
jmp find_loop
line_found:
mov rsi, r14
cmp rsi, r15
jae rev_done
lea rdi, [r15 - 1]
rev_loop:
cmp rsi, rdi
jae rev_done
mov al, byte ptr [r12 + rsi]
mov cl, byte ptr [r12 + rdi]
mov byte ptr [r12 + rsi], cl
mov byte ptr [r12 + rdi], al
inc rsi
dec rdi
jmp rev_loop
rev_done:
mov rdx, r15
sub rdx, r14
test rdx, rdx
jz write_nl
mov rax, 1
mov rdi, 1
lea rsi, [r12 + r14]
syscall
write_nl:
mov rax, 1
mov rdi, 1
lea rsi, [rip + newline_byte]
mov rdx, 1
syscall
lea r14, [r15 + 1]
jmp line_loop
done:
mov rax, 60
xor edi, edi
syscall
