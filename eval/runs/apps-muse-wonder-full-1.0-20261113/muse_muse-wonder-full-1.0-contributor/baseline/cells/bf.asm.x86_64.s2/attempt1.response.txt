.globl _start
.text
_start:
lea rbx, [rip + input_buf]
lea r8, [rip + jump_table]
xor r12, r12
read_loop:
cmp r12, 65536
jae read_done
mov rax, 0
mov rdi, 0
lea rsi, [rbx + r12]
mov rdx, 65536
sub rdx, r12
syscall
cmp rax, 0
jle read_done
add r12, rax
jmp read_loop
read_done:
xor r9, r9
find_bang:
cmp r9, r12
jae no_bang
movzx eax, byte ptr [rbx + r9]
cmp al, 33
je found_bang
inc r9
jmp find_bang
found_bang:
mov r13, r9
lea r14, [r9 + 1]
jmp check_init
no_bang:
mov r13, r12
mov r14, r12
check_init:
mov rbp, rsp
xor r9, r9
check_loop:
cmp r9, r13
jae check_done
movzx eax, byte ptr [rbx + r9]
cmp al, 91
je check_open
cmp al, 93
je check_close
inc r9
jmp check_loop
check_open:
push r9
inc r9
jmp check_loop
check_close:
cmp rsp, rbp
je unbalanced
pop rcx
mov dword ptr [r8 + rcx*4], r9d
mov dword ptr [r8 + r9*4], ecx
inc r9
jmp check_loop
check_done:
cmp rsp, rbp
jne unbalanced
lea r15, [rip + tape]
xor r9, r9
exec_loop:
cmp r9, r13
jae exec_done
movzx eax, byte ptr [rbx + r9]
cmp al, 62
je cmd_gt
cmp al, 60
je cmd_lt
cmp al, 43
je cmd_plus
cmp al, 45
je cmd_minus
cmp al, 46
je cmd_dot
cmp al, 44
je cmd_comma
cmp al, 91
je cmd_lb
cmp al, 93
je cmd_rb
inc r9
jmp exec_loop
cmd_gt:
inc r15
inc r9
jmp exec_loop
cmd_lt:
dec r15
inc r9
jmp exec_loop
cmd_plus:
inc byte ptr [r15]
inc r9
jmp exec_loop
cmd_minus:
dec byte ptr [r15]
inc r9
jmp exec_loop
cmd_dot:
mov rax, 1
mov rdi, 1
mov rsi, r15
mov rdx, 1
syscall
inc r9
jmp exec_loop
cmd_comma:
cmp r14, r12
jae comma_eof
movzx eax, byte ptr [rbx + r14]
mov byte ptr [r15], al
inc r14
inc r9
jmp exec_loop
comma_eof:
mov byte ptr [r15], 0
inc r9
jmp exec_loop
cmd_lb:
cmp byte ptr [r15], 0
jne lb_next
mov r9d, dword ptr [r8 + r9*4]
lb_next:
inc r9
jmp exec_loop
cmd_rb:
cmp byte ptr [r15], 0
je rb_next
mov r9d, dword ptr [r8 + r9*4]
rb_next:
inc r9
jmp exec_loop
exec_done:
mov rax, 60
mov rdi, 0
syscall
unbalanced:
mov rax, 60
mov rdi, 1
syscall
.bss
.balign 8
input_buf:
.zero 65536
.balign 8
jump_table:
.zero 262144
.balign 8
tape:
.zero 30000
