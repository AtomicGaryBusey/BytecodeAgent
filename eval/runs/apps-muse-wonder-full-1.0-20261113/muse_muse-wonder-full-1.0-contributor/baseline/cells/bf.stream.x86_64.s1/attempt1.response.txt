.text
.globl _start
_start:
    xor r12, r12
    lea rbp, [rip + input_buf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbp + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    mov r13, r12
    mov r14, r12
    xor rcx, rcx
find_loop:
    cmp rcx, r12
    jae find_done
    cmp byte ptr [rbp + rcx], 33
    je find_found
    inc rcx
    jmp find_loop
find_found:
    mov r13, rcx
    lea r14, [rcx + 1]
find_done:
    lea r9, [rip + jumps]
    lea r10, [rip + bstack]
    xor rbx, rbx
    xor rcx, rcx
check_loop:
    cmp rcx, r13
    jae check_done
    movzx eax, byte ptr [rbp + rcx]
    cmp al, 91
    je is_open
    cmp al, 93
    je is_close
    inc rcx
    jmp check_loop
is_open:
    mov dword ptr [r10 + rbx*4], ecx
    inc rbx
    inc rcx
    jmp check_loop
is_close:
    test rbx, rbx
    jz fail_exit
    dec rbx
    mov eax, dword ptr [r10 + rbx*4]
    mov dword ptr [r9 + rax*4], ecx
    mov dword ptr [r9 + rcx*4], eax
    inc rcx
    jmp check_loop
check_done:
    test rbx, rbx
    jnz fail_exit
    xor rbx, rbx
    xor r15, r15
    lea r8, [rip + tape]
exec_loop:
    cmp rbx, r13
    jae success_exit
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 62
    je do_gt
    cmp al, 60
    je do_lt
    cmp al, 43
    je do_plus
    cmp al, 45
    je do_minus
    cmp al, 46
    je do_dot
    cmp al, 44
    je do_comma
    cmp al, 91
    je do_lb
    cmp al, 93
    je do_rb
    inc rbx
    jmp exec_loop
do_gt:
    inc r15
    inc rbx
    jmp exec_loop
do_lt:
    dec r15
    inc rbx
    jmp exec_loop
do_plus:
    inc byte ptr [r8 + r15]
    inc rbx
    jmp exec_loop
do_minus:
    dec byte ptr [r8 + r15]
    inc rbx
    jmp exec_loop
do_dot:
    mov rax, 1
    mov rdi, 1
    lea rsi, [r8 + r15]
    mov rdx, 1
    syscall
    inc rbx
    jmp exec_loop
do_comma:
    cmp r14, r12
    jae comma_eof
    movzx eax, byte ptr [rbp + r14]
    mov byte ptr [r8 + r15], al
    inc r14
    inc rbx
    jmp exec_loop
comma_eof:
    mov byte ptr [r8 + r15], 0
    inc rbx
    jmp exec_loop
do_lb:
    cmp byte ptr [r8 + r15], 0
    jne lb_next
    mov eax, dword ptr [r9 + rbx*4]
    lea rbx, [rax + 1]
    jmp exec_loop
lb_next:
    inc rbx
    jmp exec_loop
do_rb:
    cmp byte ptr [r8 + r15], 0
    je rb_next
    mov eax, dword ptr [r9 + rbx*4]
    lea rbx, [rax + 1]
    jmp exec_loop
rb_next:
    inc rbx
    jmp exec_loop
success_exit:
    mov rax, 60
    mov rdi, 0
    syscall
fail_exit:
    mov rax, 60
    mov rdi, 1
    syscall
.bss
.balign 16
input_buf:
    .zero 65536
tape:
    .zero 30000
jumps:
    .zero 262144
bstack:
    .zero 262144
