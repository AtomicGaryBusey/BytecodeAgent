.globl _start
.text
_start:
    xor r12, r12
    lea r13, [rip + inbuf]
read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r13 + r12]
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r14, r14
row_loop:
    cmp r14, r12
    jae done_rows
    mov rax, r12
    sub rax, r14
    cmp rax, 16
    jbe row_n_done
    mov rax, 16
row_n_done:
    mov r15, rax
    lea rbx, [rip + linebuf]
    mov ecx, 28
offset_loop:
    mov rax, r14
    shr rax, cl
    and eax, 15
    lea rdx, [rip + hexdigits]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rbx], al
    inc rbx
    sub ecx, 4
    jns offset_loop
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    add rbx, 2
    xor r8, r8
hex_loop:
    cmp r8, r15
    jae hex_pad
    mov rax, r14
    add rax, r8
    movzx eax, byte ptr [r13 + rax]
    mov edx, eax
    shr edx, 4
    and edx, 15
    lea rsi, [rip + hexdigits]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rbx], dl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx + 1], al
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp hex_after_slot
hex_pad:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
hex_after_slot:
    cmp r8, 7
    jne hex_no_extra
    mov byte ptr [rbx], 0x20
    inc rbx
hex_no_extra:
    inc r8
    cmp r8, 16
    jb hex_loop
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2
    xor r8, r8
ascii_loop:
    cmp r8, r15
    jae ascii_done
    mov rax, r14
    add rax, r8
    movzx eax, byte ptr [r13 + rax]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7e
    ja ascii_dot
    mov byte ptr [rbx], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rbx], 0x2e
ascii_next:
    inc rbx
    inc r8
    jmp ascii_loop
ascii_done:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + linebuf]
    mov rdx, rbx
    sub rdx, rsi
    syscall
    add r14, r15
    jmp row_loop
done_rows:
    lea rbx, [rip + linebuf]
    mov ecx, 28
final_loop:
    mov rax, r12
    shr rax, cl
    and eax, 15
    lea rdx, [rip + hexdigits]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rbx], al
    inc rbx
    sub ecx, 4
    jns final_loop
    mov byte ptr [rbx], 0x0a
    inc rbx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + linebuf]
    mov rdx, rbx
    sub rdx, rsi
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
