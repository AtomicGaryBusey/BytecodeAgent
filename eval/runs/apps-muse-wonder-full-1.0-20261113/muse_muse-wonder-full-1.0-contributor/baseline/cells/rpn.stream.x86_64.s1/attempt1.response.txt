.equ INPUT_CAP, 65552
.text
.globl _start
_start:
lea r12, [rip + input_buf]
read_loop:
lea rdx, [rip + input_buf]
add rdx, INPUT_CAP
sub rdx, r12
jz read_done
mov rax, 0
mov rdi, 0
mov rsi, r12
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
lea r13, [rip + input_buf]
line_loop:
cmp r13, r12
jae exit_program
mov r14, r13
find_newline:
cmp r14, r12
jae line_end_no_newline
movzx eax, byte ptr [r14]
cmp al, 10
je line_end_newline
inc r14
jmp find_newline
line_end_newline:
lea r15, [r14 + 1]
jmp eval_line
line_end_no_newline:
mov r15, r14
eval_line:
mov rbx, r13
xor rbp, rbp
token_loop:
cmp rbx, r14
jae end_tokens
movzx eax, byte ptr [rbx]
cmp al, 32
je skip_separator
cmp al, 9
je skip_separator
jmp token_start
skip_separator:
inc rbx
jmp token_loop
token_start:
mov rsi, rbx
scan_token:
cmp rbx, r14
jae token_scanned
movzx eax, byte ptr [rbx]
cmp al, 32
je token_scanned
cmp al, 9
je token_scanned
inc rbx
jmp scan_token
token_scanned:
mov rcx, rbx
sub rcx, rsi
cmp rcx, 1
jne check_number
movzx eax, byte ptr [rsi]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator
jmp check_number
do_operator:
mov r8d, eax
cmp rbp, 2
jb line_error
lea r9, [rip + eval_stack]
dec rbp
mov rcx, qword ptr [r9 + rbp*8]
dec rbp
mov rax, qword ptr [r9 + rbp*8]
cmp r8b, 43
je op_add
cmp r8b, 45
je op_sub
cmp r8b, 42
je op_mul
cmp r8b, 47
je op_div
jmp op_mod
op_add:
add rax, rcx
jmp push_result
op_sub:
sub rax, rcx
jmp push_result
op_mul:
imul rax, rcx
jmp push_result
op_div:
test rcx, rcx
jz line_error
cmp rcx, -1
jne do_div_quo
mov r10, 1
shl r10, 63
cmp rax, r10
je line_error
do_div_quo:
cqo
idiv rcx
jmp push_result
op_mod:
test rcx, rcx
jz line_error
cmp rcx, -1
jne do_div_rem
mov r10, 1
shl r10, 63
cmp rax, r10
je line_error
do_div_rem:
cqo
idiv rcx
mov rax, rdx
push_result:
lea r9, [rip + eval_stack]
mov qword ptr [r9 + rbp*8], rax
inc rbp
jmp token_loop
check_number:
xor r8d, r8d
mov rdi, rsi
movzx eax, byte ptr [rdi]
cmp al, 45
jne count_digits
mov r8d, 1
inc rdi
count_digits:
mov rcx, rbx
sub rcx, rdi
test rcx, rcx
jz line_error
cmp rcx, 18
ja line_error
xor rax, rax
parse_digits:
cmp rdi, rbx
jae parsed_digits
movzx edx, byte ptr [rdi]
sub edx, 48
cmp edx, 9
ja line_error
imul rax, rax, 10
add rax, rdx
inc rdi
jmp parse_digits
parsed_digits:
test r8d, r8d
jz push_number
neg rax
push_number:
lea r9, [rip + eval_stack]
mov qword ptr [r9 + rbp*8], rax
inc rbp
jmp token_loop
end_tokens:
cmp rbp, 1
jne line_error
lea r9, [rip + eval_stack]
mov rax, qword ptr [r9]
mov r9, rax
xor r8d, r8d
test rax, rax
jns magnitude_ok
mov r8d, 1
neg r9
magnitude_ok:
lea rsi, [rip + num_buf]
add rsi, 32
dec rsi
mov byte ptr [rsi], 10
digit_loop:
mov rax, r9
xor edx, edx
mov ecx, 10
div rcx
add dl, 48
dec rsi
mov byte ptr [rsi], dl
mov r9, rax
test rax, rax
jnz digit_loop
test r8d, r8d
jz print_write
dec rsi
mov byte ptr [rsi], 45
print_write:
lea rdx, [rip + num_buf]
add rdx, 32
sub rdx, rsi
mov rax, 1
mov rdi, 1
syscall
jmp next_line
line_error:
mov rax, 1
mov rdi, 1
lea rsi, [rip + error_msg]
mov edx, 6
syscall
next_line:
mov r13, r15
jmp line_loop
exit_program:
mov rax, 60
xor edi, edi
syscall
.section .rodata
error_msg:
.ascii "error"
.byte 10
.bss
.balign 8
input_buf:
.zero 65552
.balign 8
eval_stack:
.zero 262144
.balign 8
num_buf:
.zero 32
