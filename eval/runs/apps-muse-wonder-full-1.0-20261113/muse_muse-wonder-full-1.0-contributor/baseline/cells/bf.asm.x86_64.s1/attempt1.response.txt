.text
.globl _start
_start:
lea rbx, [rip + input_buf]
xor r12, r12
read_loop:
mov rax, 0
mov rdi, 0
lea rsi, [rbx + r12]
mov rdx, 65600
sub rdx, r12
syscall
cmp rax, 0
jle read_done
add r12, rax
jmp read_loop
read_done:
xor rcx, rcx
scan_loop:
cmp rcx, r12
jae no_bang
cmp byte ptr [rbx + rcx], 0x21
je found_bang
inc rcx
jmp scan_loop
found_bang:
mov r13, rcx
lea r14, [rcx + 1]
jmp validate_init
no_bang:
mov r13, r12
mov r14, r12
validate_init:
lea rdi, [rip + match_table]
xor rsi, rsi
xor r15, r15
validate_loop:
cmp rsi, r13
jae validate_done
movzx eax, byte ptr [rbx + rsi]
cmp al, 0x5b
je v_open
cmp al, 0x5d
je v_close
inc rsi
jmp validate_loop
v_open:
push rsi
inc r15
inc rsi
jmp validate_loop
v_close:
cmp r15, 0
je unbalanced
dec r15
pop rcx
mov dword ptr [rdi + rcx*4], esi
mov dword ptr [rdi + rsi*4], ecx
inc rsi
jmp validate_loop
validate_done:
cmp r15, 0
jne unbalanced
lea rdi, [rip + match_table]
lea rbp, [rip + tape_buf]
xor rsi, rsi
exec_loop:
cmp rsi, r13
jae exit_ok
movzx eax, byte ptr [rbx + rsi]
inc rsi
cmp al, 0x3e
je cmd_right
cmp al, 0x3c
je cmd_left
cmp al, 0x2b
je cmd_inc
cmp al, 0x2d
je cmd_dec
cmp al, 0x2e
je cmd_dot
cmp al, 0x2c
je cmd_comma
cmp al, 0x5b
je cmd_open
cmp al, 0x5d
je cmd_close
jmp exec_loop
cmd_right:
inc rbp
jmp exec_loop
cmd_left:
dec rbp
jmp exec_loop
cmd_inc:
inc byte ptr [rbp]
jmp exec_loop
cmd_dec:
dec byte ptr [rbp]
jmp exec_loop
cmd_dot:
push rsi
push rdi
mov rax, 1
mov rdi, 1
mov rsi, rbp
mov rdx, 1
syscall
pop rdi
pop rsi
jmp exec_loop
cmd_comma:
cmp r14, r12
jae comma_eof
movzx eax, byte ptr [rbx + r14]
mov byte ptr [rbp], al
inc r14
jmp exec_loop
comma_eof:
mov byte ptr [rbp], 0
jmp exec_loop
cmd_open:
cmp byte ptr [rbp], 0
jne exec_loop
mov rcx, rsi
dec rcx
mov eax, dword ptr [rdi + rcx*4]
mov rsi, rax
inc rsi
jmp exec_loop
cmd_close:
cmp byte ptr [rbp], 0
je exec_loop
mov rcx, rsi
dec rcx
mov eax, dword ptr [rdi + rcx*4]
mov rsi, rax
inc rsi
jmp exec_loop
unbalanced:
mov rax, 60
mov rdi, 1
syscall
jmp unbalanced
exit_ok:
mov rax, 60
xor rdi, rdi
syscall
jmp exit_ok
.bss
.balign 16
input_buf:
.zero 65600
tape_buf:
.zero 30000
match_table:
.zero 262400
