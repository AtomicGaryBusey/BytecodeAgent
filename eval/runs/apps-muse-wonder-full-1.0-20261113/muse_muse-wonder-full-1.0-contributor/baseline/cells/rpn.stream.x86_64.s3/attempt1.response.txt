.globl _start

token_loop:
skip_sep:

je skip_one
cmp al, 9
je skip_one
jmp token_start

jmp skip_sep

cmp rsi, r15
jae token_found

cmp al, 9
je token_found
inc rsi
jmp scan_token

movzx eax, byte ptr [r9]
cmp al, 43
je do_op
cmp al, 45
je do_op
cmp al, 42
je do_op
cmp al, 47
je do_op
cmp al, 37

mov rdx, r9
xor r10d, r10d
movzx eax, byte ptr [rdx]
cmp al, 45
jne digits_start
mov r10d, 1
inc rdx
digits_start:
mov r11, rsi
sub r11, rdx

jz line_error
cmp r11, 18

xor eax, eax

cmp rdx, rsi
jae digits_done
movzx ecx, byte ptr [rdx]
cmp cl, 48
jb line_error
cmp cl, 57
ja line_error
sub ecx, 48
imul rax, rax, 10
add rax, rcx
inc rdx
jmp digit_loop

test r10d, r10d
jz push_value
neg rax

lea rcx, [rip + eval_stack]
mov qword ptr [rcx + r8*8], rax
inc r8
jmp token_loop
do_op:
mov r10d, eax
cmp r8, 2
jb line_error
dec r8
lea rcx, [rip + eval_stack]
mov rdx, qword ptr [rcx + r8*8]
dec r8
mov rax, qword ptr [rcx + r8*8]
cmp r10d, 43
je op_add
cmp r10d, 45
je op_sub
cmp r10d, 42
je op_mul
cmp r10d, 47

jmp push_result
op_sub:
sub rax, rdx
jmp push_result
op_mul:
imul rax, rdx
jmp push_result
op_div:
test rdx, rdx
jz line_error
mov rcx, 1
shl rcx, 63
cmp rax, rcx
jne op_div_ok
cmp rdx, -1
je line_error

mov rcx, rdx
mov rdx, rax
sar rdx, 63
idiv rcx
jmp push_result
op_mod:
test rdx, rdx
jz line_error
mov rcx, 1
shl rcx, 63
cmp rax, rcx
jne op_mod_ok
cmp rdx, -1
je line_error
op_mod_ok:

mov rdx, rax
sar rdx, 63
idiv rcx

jmp push_result

lea rcx, [rip + eval_stack]
