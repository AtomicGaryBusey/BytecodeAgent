.equ BUFSZ, 131072
.text
.globl _start
_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d
read_loop:
    mov edx, BUFSZ
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    xor ecx, ecx
count_loop:
    cmp rcx, r12
    jae count_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 10
    jne check_ws
    inc r13
check_ws:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    ja is_word
is_ws:
    xor r15d, r15d
    jmp count_loop
is_word:
    test r15d, r15d
    jnz count_loop
    mov r15d, 1
    inc r14
    jmp count_loop
count_done:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
print_num:
    test rax, rax
    jnz print_num_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret
print_num_nonzero:
    xor ecx, ecx
print_num_div:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz print_num_div
print_num_out:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz print_num_out
    ret
.bss
.balign 16
buffer:
    .zero 131072
outbuf:
    .zero 64
