.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
read_loop:
    lea rdx, [rip + inbuf_end]
    sub rdx, rbx
    test rdx, rdx
    jle read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    lea rbx, [rip + inbuf]
    cmp rbx, r12
    jae exit_program
    lea r13, [rip + alphabet]
    lea r14, [rip + outbuf]
    xor r15d, r15d
encode_loop:
    cmp rbx, r12
    jae encode_done
    mov rax, r12
    sub rax, rbx
    cmp rax, 3
    jae group3
    cmp rax, 2
    je group2
    movzx eax, byte ptr [rbx]
    shl eax, 16
    add rbx, 1
    mov r8d, 2
    jmp emit_group
group2:
    movzx eax, byte ptr [rbx]
    shl eax, 16
    movzx ecx, byte ptr [rbx + 1]
    shl ecx, 8
    or eax, ecx
    add rbx, 2
    mov r8d, 3
    jmp emit_group
group3:
    movzx eax, byte ptr [rbx]
    shl eax, 16
    movzx ecx, byte ptr [rbx + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rbx + 2]
    or eax, ecx
    add rbx, 3
    mov r8d, 4
emit_group:
    mov r9d, 18
    xor r10d, r10d
emit_loop:
    cmp r10d, r8d
    jae emit_pad
    mov edx, eax
    mov ecx, r9d
    shr edx, cl
    and edx, 63
    mov dl, byte ptr [r13 + rdx]
    mov byte ptr [r14], dl
    jmp emit_next
emit_pad:
    mov byte ptr [r14], 61
emit_next:
    inc r14
    inc r10d
    sub r9d, 6
    cmp r10d, 4
    jb emit_loop
    add r15d, 4
    cmp r15d, 76
    jne encode_loop
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
    jmp encode_loop
encode_done:
    cmp r15d, 0
    je write_output
    mov byte ptr [r14], 10
    inc r14
write_output:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle exit_program
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_program:
    mov rax, 60
    xor edi, edi
    syscall
.section .rodata
.balign 8
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.bss
.balign 16
inbuf:
    .zero 65536
inbuf_end:
.balign 16
outbuf:
    .zero 70000
outbuf_end:
