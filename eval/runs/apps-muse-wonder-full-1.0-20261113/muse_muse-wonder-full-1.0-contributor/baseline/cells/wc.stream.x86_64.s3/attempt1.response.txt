.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jle done
    mov rbx, rax
    add r14, rax
    lea rsi, [rip + buffer]
    xor ecx, ecx
process_loop:
    cmp rcx, rbx
    jae read_loop
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je is_whitespace
    cmp al, 9
    jb not_whitespace
    cmp al, 13
    ja not_whitespace
is_whitespace:
    xor r15d, r15d
    jmp process_loop
not_whitespace:
    test r15d, r15d
    jnz process_loop
    mov r15d, 1
    inc r13
    jmp process_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call append_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call append_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call append_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
append_num:
    test rax, rax
    jnz append_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret
append_nonzero:
    xor ecx, ecx
append_div_loop:
    xor edx, edx
    mov r8d, 10
    div r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz append_div_loop
append_pop_loop:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz append_pop_loop
    ret
.bss
    .balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
