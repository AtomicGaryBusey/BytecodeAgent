.globl _start
.section .rodata
hexdigits:
.ascii "0123456789abcdef"
.bss
.balign 16
inbuf:
.zero 65536
outbuf:
.zero 128
.text
_start:
lea r12, [rip + inbuf]
xor r13, r13
read_loop:
mov rdx, 65536
sub rdx, r13
test rdx, rdx
jz done_read
mov rax, 0
xor rdi, rdi
lea rsi, [r12 + r13]
syscall
test rax, rax
jz done_read
js done_read
add r13, rax
jmp read_loop
done_read:
xor r14, r14
row_loop:
cmp r14, r13
jae final_line
mov rax, r13
sub rax, r14
cmp rax, 16
jbe have_n
mov rax, 16
have_n:
mov r15, rax
lea rbx, [rip + outbuf]
mov esi, 28
off_loop:
mov rax, r14
mov ecx, esi
shr rax, cl
and eax, 15
lea rdx, [rip + hexdigits]
mov al, byte ptr [rdx + rax]
mov byte ptr [rbx], al
inc rbx
sub esi, 4
jns off_loop
mov byte ptr [rbx], 32
inc rbx
mov byte ptr [rbx], 32
inc rbx
xor r8, r8
slot_loop:
cmp r8, r15
jae slot_pad
lea rax, [r12 + r14]
movzx eax, byte ptr [rax + r8]
mov edx, eax
shr eax, 4
and eax, 15
lea rcx, [rip + hexdigits]
mov al, byte ptr [rcx + rax]
mov byte ptr [rbx], al
inc rbx
mov eax, edx
and eax, 15
mov al, byte ptr [rcx + rax]
mov byte ptr [rbx], al
inc rbx
mov byte ptr [rbx], 32
inc rbx
jmp slot_after
slot_pad:
mov byte ptr [rbx], 32
mov byte ptr [rbx + 1], 32
mov byte ptr [rbx + 2], 32
add rbx, 3
slot_after:
cmp r8, 7
jne slot_no_extra
mov byte ptr [rbx], 32
inc rbx
slot_no_extra:
inc r8
cmp r8, 16
jb slot_loop
mov byte ptr [rbx], 32
inc rbx
mov byte ptr [rbx], 124
inc rbx
xor r8, r8
ascii_loop:
cmp r8, r15
jae ascii_done
lea rax, [r12 + r14]
movzx eax, byte ptr [rax + r8]
cmp eax, 32
jb ascii_dot
cmp eax, 126
ja ascii_dot
jmp ascii_store
ascii_dot:
mov eax, 46
ascii_store:
mov byte ptr [rbx], al
inc rbx
inc r8
jmp ascii_loop
ascii_done:
mov byte ptr [rbx], 124
inc rbx
mov byte ptr [rbx], 10
inc rbx
lea rsi, [rip + outbuf]
mov rdx, rbx
sub rdx, rsi
mov rax, 1
mov rdi, 1
syscall
add r14, r15
jmp row_loop
final_line:
lea rbx, [rip + outbuf]
mov esi, 28
final_loop:
mov rax, r13
mov ecx, esi
shr rax, cl
and eax, 15
lea rdx, [rip + hexdigits]
mov al, byte ptr [rdx + rax]
mov byte ptr [rbx], al
inc rbx
sub esi, 4
jns final_loop
mov byte ptr [rbx], 10
inc rbx
lea rsi, [rip + outbuf]
mov rdx, rbx
sub rdx, rsi
mov rax, 1
mov rdi, 1
syscall
mov rax, 60
xor rdi, rdi
syscall
