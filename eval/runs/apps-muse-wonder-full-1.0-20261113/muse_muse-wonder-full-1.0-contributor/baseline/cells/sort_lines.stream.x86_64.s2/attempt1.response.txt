.globl _start
_start:
xor ebx, ebx
read_loop:
cmp rbx, 32768
jae read_done
mov eax, 0
xor edi, edi
lea rsi, [rip + input_buf]
add rsi, rbx
mov rdx, 32768
sub rdx, rbx
syscall
cmp rax, 0
jle read_done
add rbx, rax
jmp read_loop
read_done:
xor r12d, r12d
xor ecx, ecx
xor edx, edx
lea r15, [rip + input_buf]
lea r13, [rip + line_ptrs]
lea r14, [rip + line_lens]
parse_loop:
cmp rcx, rbx
jae parse_finish
movzx eax, byte ptr [r15 + rcx]
cmp eax, 10
jne parse_next
mov rax, r15
add rax, rdx
mov qword ptr [r13 + r12*8], rax
mov rax, rcx
sub rax, rdx
mov qword ptr [r14 + r12*8], rax
inc r12
mov rdx, rcx
inc rdx
parse_next:
inc rcx
jmp parse_loop
parse_finish:
cmp rdx, rbx
jae parse_done
mov rax, r15
add rax, rdx
mov qword ptr [r13 + r12*8], rax
mov rax, rbx
sub rax, rdx
mov qword ptr [r14 + r12*8], rax
inc r12
parse_done:
xor ebx, ebx
sort_outer:
cmp r12, 2
jb sort_done
mov rax, r12
dec rax
cmp rbx, rax
jae sort_done
xor ebp, ebp
sort_inner:
mov rax, r12
dec rax
sub rax, rbx
cmp rbp, rax
jae sort_inner_done
mov rsi, qword ptr [r13 + rbp*8]
mov rdx, qword ptr [r14 + rbp*8]
mov rdi, qword ptr [r13 + rbp*8 + 8]
mov rcx, qword ptr [r14 + rbp*8 + 8]
call compare_lines
cmp rax, 0
jle sort_no_swap
mov rax, qword ptr [r13 + rbp*8]
mov rcx, qword ptr [r13 + rbp*8 + 8]
mov qword ptr [r13 + rbp*8], rcx
mov qword ptr [r13 + rbp*8 + 8], rax
mov rax, qword ptr [r14 + rbp*8]
mov rcx, qword ptr [r14 + rbp*8 + 8]
mov qword ptr [r14 + rbp*8], rcx
mov qword ptr [r14 + rbp*8 + 8], rax
sort_no_swap:
inc rbp
jmp sort_inner
sort_inner_done:
inc rbx
jmp sort_outer
sort_done:
xor ebx, ebx
out_loop:
cmp rbx, r12
jae out_done
mov rdx, qword ptr [r14 + rbx*8]
test rdx, rdx
jz out_newline
mov rsi, qword ptr [r13 + rbx*8]
mov eax, 1
mov edi, 1
syscall
out_newline:
mov eax, 1
mov edi, 1
lea rsi, [rip + newline_char]
mov edx, 1
syscall
inc rbx
jmp out_loop
out_done:
mov eax, 60
xor edi, edi
syscall
compare_lines:
mov r9, rdx
cmp r9, rcx
jbe cmp_chunk_check
mov r9, rcx
cmp_chunk_check:
cmp r9, 8
jb cmp_tail_check
cmp_chunk_loop:
mov rax, qword ptr [rsi]
mov r10, qword ptr [rdi]
cmp rax, r10
jne cmp_chunk_diff
add rsi, 8
add rdi, 8
sub r9, 8
cmp r9, 8
jae cmp_chunk_loop
jmp cmp_tail_check
cmp_chunk_diff:
bswap rax
bswap r10
cmp rax, r10
jb cmp_less
mov eax, 1
ret
cmp_tail_check:
test r9, r9
jz cmp_len
cmp_tail_loop:
movzx eax, byte ptr [rsi]
movzx r10d, byte ptr [rdi]
cmp eax, r10d
jb cmp_less
ja cmp_greater
inc rsi
inc rdi
dec r9
jnz cmp_tail_loop
cmp_len:
cmp rdx, rcx
jb cmp_less
ja cmp_greater
xor eax, eax
ret
cmp_less:
mov rax, -1
ret
cmp_greater:
mov eax, 1
ret
