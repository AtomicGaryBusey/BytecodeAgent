.globl _start
.text
_start:
    xor ebx, ebx
    xor ebp, ebp
    xor r12d, r12d
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jle done_reading
    add r12, rax
    mov r14, rax
    lea rsi, [rip + buffer]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_newline
    inc rbx
not_newline:
    mov edx, eax
    cmp dl, 32
    je is_space
    sub dl, 9
    cmp dl, 4
    jbe is_space
    test r13d, r13d
    jnz next_byte
    mov r13d, 1
    inc rbp
    jmp next_byte
is_space:
    xor r13d, r13d
next_byte:
    dec r14
    jnz byte_loop
    jmp read_loop
done_reading:
    lea rdi, [rip + outbuf]
    mov rax, rbx
    call append_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, rbp
    call append_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call append_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
append_num:
    sub rsp, 32
    lea rcx, [rsp + 32]
    mov r8, rcx
    mov r9, 10
    test rax, rax
    jnz digits_loop
    dec rcx
    mov byte ptr [rcx], 48
    jmp copy_loop
digits_loop:
    xor edx, edx
    div r9
    add dl, 48
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz digits_loop
copy_loop:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, r8
    jb copy_loop
    add rsp, 32
    ret
.bss
.balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
