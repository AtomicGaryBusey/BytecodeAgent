.globl _start
.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor rbx, rbx
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 4096
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    mov rbp, rax
    xor rcx, rcx
    lea rsi, [rip + buffer]
process_loop:
    cmp rcx, rbp
    jae read_loop
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je is_space
    cmp al, 9
    jb is_word
    cmp al, 13
    ja is_word
is_space:
    xor rbx, rbx
    jmp process_loop
is_word:
    test rbx, rbx
    jnz process_loop
    mov rbx, 1
    inc r13
    jmp process_loop
read_done:
    lea rbx, [rip + outbuf]
    mov rax, r12
    call write_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r13
    call write_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r14
    call write_num
    mov byte ptr [rbx], 10
    inc rbx
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
write_num:
    test rax, rax
    jnz 1f
    mov byte ptr [rbx], 48
    inc rbx
    ret
1:
    xor rcx, rcx
    mov r8, 10
2:
    xor rdx, rdx
    div r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz 2b
3:
    pop rdx
    mov byte ptr [rbx], dl
    inc rbx
    dec rcx
    jnz 3b
    ret

buffer:
    .zero 4096
outbuf:
