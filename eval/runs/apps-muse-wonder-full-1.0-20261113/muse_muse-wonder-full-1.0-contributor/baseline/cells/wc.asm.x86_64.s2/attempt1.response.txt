.globl _start
.text
_start:
xor ebx, ebx
xor r12d, r12d
xor r13d, r13d
xor r14d, r14d
read_loop:
mov rax, 0
mov rdi, 0
lea rsi, [rip + buffer]
mov rdx, 4096
syscall
test rax, rax
jz done
js done
add r13, rax
mov rcx, rax
lea rsi, [rip + buffer]
process_loop:
test rcx, rcx
jz read_loop
movzx eax, byte ptr [rsi]
inc rsi
dec rcx
cmp al, 10
jne not_newline
inc rbx
not_newline:
cmp al, 32
je is_whitespace
cmp al, 9
jb not_whitespace
cmp al, 13
ja not_whitespace
is_whitespace:
mov r14, 0
jmp process_loop
not_whitespace:
test r14, r14
jnz process_loop
mov r14, 1
inc r12
jmp process_loop
done:
lea rdi, [rip + outbuf]
mov rax, rbx
call append_decimal
mov byte ptr [rdi], 32
inc rdi
mov rax, r12
call append_decimal
mov byte ptr [rdi], 32
inc rdi
mov rax, r13
call append_decimal
mov byte ptr [rdi], 10
inc rdi
lea rsi, [rip + outbuf]
mov rdx, rdi
sub rdx, rsi
mov rax, 1
mov rdi, 1
syscall
mov rax, 60
xor edi, edi
syscall
append_decimal:
test rax, rax
jnz append_nonzero
mov byte ptr [rdi], 48
inc rdi
ret
append_nonzero:
sub rsp, 32
lea rcx, [rsp + 32]
mov r8, rcx
append_digit_loop:
xor edx, edx
mov r9, 10
div r9
dec rcx
add dl, 48
mov byte ptr [rcx], dl
test rax, rax
jnz append_digit_loop
append_copy_loop:
mov al, byte ptr [rcx]
mov byte ptr [rdi], al
inc rcx
inc rdi
cmp rcx, r8
jne append_copy_loop
add rsp, 32
ret
.bss
.balign 16
buffer:
.zero 4096
outbuf:
.zero 64
