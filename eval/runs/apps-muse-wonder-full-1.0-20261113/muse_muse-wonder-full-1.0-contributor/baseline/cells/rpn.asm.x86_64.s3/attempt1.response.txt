.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
read_loop:
    mov rax, 0
    xor edi, edi
    mov rsi, rbx
    mov edx, 8192
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    lea rbx, [rip + inbuf]
line_loop:
    cmp rbx, r12
    jae done
    mov r13, rbx
find_line_end:
    cmp r13, r12
    jae line_end_found
    cmp byte ptr [r13], 10
    je line_end_found
    inc r13
    jmp find_line_end
line_end_found:
    lea r15, [r13 + 1]
    cmp r13, r12
    jb next_ptr_ok
    mov r15, r13
next_ptr_ok:
    lea rbp, [rip + evalstack]
    xor r14d, r14d
eval_loop:
skip_loop:
    cmp rbx, r13
    jae eval_done
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_start
skip_one:
    inc rbx
    jmp skip_loop
token_start:
    mov rsi, rbx
scan_token:
    cmp rbx, r13
    jae token_scanned
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc rbx
    jmp scan_token
token_scanned:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne parse_multi
    movzx eax, byte ptr [rsi]
    cmp al, 43
    je do_op
    cmp al, 45
    je do_op
    cmp al, 42
    je do_op
    cmp al, 47
    je do_op
    cmp al, 37
    je do_op
    cmp al, 48
    jb line_error
    cmp al, 57
    ja line_error
    sub eax, 48
    cmp r14, 32768
    jae line_error
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp eval_loop
parse_multi:
    mov rdi, rsi
    mov r10, rbx
    xor r8d, r8d
    movzx eax, byte ptr [rdi]
    cmp al, 45
    jne parse_digits_setup
    mov r8d, 1
    inc rdi
parse_digits_setup:
    xor eax, eax
    xor r11d, r11d
digit_loop:
    cmp rdi, r10
    jae digits_done
    movzx ecx, byte ptr [rdi]
    cmp cl, 48
    jb line_error
    cmp cl, 57
    ja line_error
    sub ecx, 48
    imul rax, rax, 10
    add rax, rcx
    inc r11
    inc rdi
    jmp digit_loop
digits_done:
    test r11, r11
    jz line_error
    cmp r11, 18
    ja line_error
    test r8d, r8d
    jz push_value
    neg rax
push_value:
    cmp r14, 32768
    jae line_error
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp eval_loop
do_op:
    mov r10d, eax
    cmp r14, 2
    jb line_error
    mov rcx, qword ptr [rbp + r14*8 - 8]
    mov rax, qword ptr [rbp + r14*8 - 16]
    dec r14
    cmp r10b, 43
    je op_add
    cmp r10b, 45
    je op_sub
    cmp r10b, 42
    je op_mul
    cmp r10b, 47
    je op_divmod
    cmp r10b, 37
    je op_divmod
    jmp line_error
op_add:
    add rax, rcx
    jmp op_store
op_sub:
    sub rax, rcx
    jmp op_store
op_mul:
    imul rax, rcx
    jmp op_store
op_divmod:
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne do_idiv
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je line_error
do_idiv:
    cqo
    idiv rcx
    cmp r10b, 47
    je op_store
    mov rax, rdx
    jmp op_store
op_store:
    mov qword ptr [rbp + r14*8 - 8], rax
    jmp eval_loop
eval_done:
    cmp r14, 1
    jne line_error
    mov rax, qword ptr [rbp]
    xor r8d, r8d
    test rax, rax
    jns magnitude_done
    mov r8d, 1
    neg rax
magnitude_done:
    lea rdi, [rip + numbuf_end]
    dec rdi
    mov byte ptr [rdi], 10
format_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz format_loop
    test r8d, r8d
    jz format_done
    dec rdi
    mov byte ptr [rdi], 45
format_done:
    lea rdx, [rip + numbuf_end]
    sub rdx, rdi
    mov rsi, rdi
    mov rax, 1
    mov edi, 1
    syscall
    jmp next_line
line_error:
    mov rax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 6
    syscall
    jmp next_line
next_line:
    mov rbx, r15
    jmp line_loop
done:
    mov rax, 60
    xor edi, edi
    syscall
.section .rodata
err_msg:
    .ascii "error\n"
.bss
    .balign 16
inbuf:
    .zero 131072
    .balign 16
evalstack:
    .zero 262144
numbuf:
    .zero 32
numbuf_end:
