mov rax, 32768

jz read_done
mov rdx, rax
lea rsi, [rbx + r12]
xor edi, edi
xor eax, eax

test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
lea r15, [rip + entries]
xor r13d, r13d
xor r14d, r14d
xor ecx, ecx
parse_loop:
cmp rcx, r12
jae parse_done
movzx eax, byte ptr [rbx + rcx]
cmp eax, 10
jne parse_next
mov rax, rcx
sub rax, r14
shl rax, 32
or rax, r14
mov qword ptr [r15 + r13*8], rax
inc r13
inc rcx
mov r14, rcx
jmp parse_loop
parse_next:
inc rcx
jmp parse_loop
parse_done:
cmp r14, r12
jae parse_finished
mov rax, r12
sub rax, r14
shl rax, 32
or rax, r14
mov qword ptr [r15 + r13*8], rax
inc r13
parse_finished:
mov r12, r13
mov r13, 1
sort_outer:
cmp r13, r12
jae sort_done
mov r14, qword ptr [r15 + r13*8]
xor ebp, ebp
mov rdi, r13
search_loop:
cmp rbp, rdi
jae search_done
mov rax, rbp
add rax, rdi
shr rax, 1
mov rsi, qword ptr [r15 + rax*8]
mov rdx, r14
push rax
call compare
pop rcx
test rax, rax
jg search_hi
lea rbp, [rcx + 1]
jmp search_loop
search_hi:
mov rdi, rcx
jmp search_loop
search_done:
mov rax, r13
shift_loop:
cmp rax, rbp
jbe shift_done
mov rcx, qword ptr [r15 + rax*8 - 8]
mov qword ptr [r15 + rax*8], rcx
dec rax
jmp shift_loop
shift_done:
mov qword ptr [r15 + rbp*8], r14
inc r13
jmp sort_outer
sort_done:
xor r13d, r13d
out_loop:
cmp r13, r12
jae program_exit
mov rax, qword ptr [r15 + r13*8]
mov rdx, rax
shr rdx, 32
mov esi, eax
add rsi, rbx

mov rax, 60
xor edi, edi
syscall
compare:
mov eax, esi
mov rcx, rsi
shr rcx, 32
lea rsi, [rbx + rax]
mov eax, edx
mov r8, rdx
shr r8, 32
lea rdx, [rbx + rax]
mov r9, rcx
cmp r9, r8
jbe compare_min_done
mov r9, r8
compare_min_done:
xor r10d, r10d
compare_loop:
cmp r10, r9
jae compare_len
movzx eax, byte ptr [rsi + r10]
movzx r11d, byte ptr [rdx + r10]
cmp eax, r11d
jb compare_less
ja compare_greater
inc r10
jmp compare_loop
compare_len:
cmp rcx, r8
jb compare_less
ja compare_greater
xor eax, eax
ret
compare_less:
mov rax, -1
ret
compare_greater:
mov rax, 1
ret
write_all:
test rdx, rdx
jz write_all_done
write_all_loop:
mov rax, 1
mov rdi, 1
syscall
test rax, rax
jle write_all_done
add rsi, rax
sub rdx, rax
jnz write_all_loop
write_all_done:
ret
