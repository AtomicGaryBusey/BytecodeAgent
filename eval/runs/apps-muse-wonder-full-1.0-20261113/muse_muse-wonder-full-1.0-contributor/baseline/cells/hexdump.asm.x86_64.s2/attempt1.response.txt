.text
.globl _start
_start:
lea rbx, [rip + inbuf]
xor r12, r12
read_loop:
mov rdx, 70000
sub rdx, r12
lea rsi, [rbx + r12]
xor edi, edi
xor eax, eax
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
xor r13, r13
row_loop:
cmp r13, r12
jae final_line
mov rax, r12
sub rax, r13
cmp rax, 16
jbe n_ok
mov rax, 16
n_ok:
mov r14, rax
lea rdi, [rip + linebuf]
mov eax, r13d
call put_hex8
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
xor r15, r15
hex_loop:
cmp r15, 16
jae hex_done
cmp r15, r14
jae hex_blank
mov rax, r13
add rax, r15
movzx eax, byte ptr [rbx + rax]
call put_hex2
mov byte ptr [rdi], 32
inc rdi
jmp hex_after
hex_blank:
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
hex_after:
cmp r15, 7
jne hex_noextra
mov byte ptr [rdi], 32
inc rdi
hex_noextra:
inc r15
jmp hex_loop
hex_done:
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 124
inc rdi
xor r15, r15
ascii_loop:
cmp r15, r14
jae ascii_done
mov rax, r13
add rax, r15
movzx eax, byte ptr [rbx + rax]
cmp eax, 32
jb ascii_dot
cmp eax, 126
ja ascii_dot
mov byte ptr [rdi], al
jmp ascii_next
ascii_dot:
mov byte ptr [rdi], 46
ascii_next:
inc rdi
inc r15
jmp ascii_loop
ascii_done:
mov byte ptr [rdi], 124
inc rdi
mov byte ptr [rdi], 10
inc rdi
lea rsi, [rip + linebuf]
mov rdx, rdi
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
add r13, r14
jmp row_loop
final_line:
lea rdi, [rip + linebuf]
mov eax, r12d
call put_hex8
mov byte ptr [rdi], 10
inc rdi
lea rsi, [rip + linebuf]
mov rdx, rdi
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
mov eax, 60
xor edi, edi
syscall
put_hex8:
mov ecx, 28
put_hex8_loop:
mov edx, eax
shr edx, cl
and edx, 15
lea r8, [rip + hexdigits]
movzx edx, byte ptr [r8 + rdx]
mov byte ptr [rdi], dl
inc rdi
sub ecx, 4
jns put_hex8_loop
ret
put_hex2:
lea r8, [rip + hexdigits]
mov edx, eax
shr edx, 4
and edx, 15
movzx edx, byte ptr [r8 + rdx]
mov byte ptr [rdi], dl
inc rdi
mov edx, eax
and edx, 15
movzx edx, byte ptr [r8 + rdx]
mov byte ptr [rdi], dl
inc rdi
ret
.section .rodata
hexdigits:
.ascii "0123456789abcdef"
.bss
.balign 16
inbuf:
.zero 70000
linebuf:
.zero 128
