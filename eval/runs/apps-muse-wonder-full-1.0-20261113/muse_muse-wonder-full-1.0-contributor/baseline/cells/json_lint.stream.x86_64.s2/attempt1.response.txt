.globl _start
.text
_start:
    xor r12, r12
    lea r13, [rip + buffer]
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 8192
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call skip_ws
    call parse_value
    cmp qword ptr [rip + err_flag], 0
    jne print_error
    call skip_ws
    cmp rbx, r12
    jne top_extra
    jmp print_ok
top_extra:
    mov qword ptr [rip + err_pos], rbx
    jmp print_error
print_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, 9
    syscall
    mov rax, qword ptr [rip + err_pos]
    lea rcx, [rip + numbuf]
    add rcx, 32
    mov r8, rcx
    dec rcx
    mov byte ptr [rcx], 10
    test rax, rax
    jnz pe_convert
    dec rcx
    mov byte ptr [rcx], 48
    jmp pe_write
pe_convert:
    test rax, rax
    jz pe_write
    xor edx, edx
    mov r9, 10
    div r9
    add dl, 48
    dec rcx
    mov byte ptr [rcx], dl
    jmp pe_convert
pe_write:
    mov rsi, rcx
    mov rdx, r8
    sub rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
fail:
    mov qword ptr [rip + err_flag], 1
    mov qword ptr [rip + err_pos], rbx
    ret
skip_ws:
skip_ws_loop:
    cmp rbx, r12
    jae skip_ws_ret
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 32
    je skip_ws_inc
    cmp al, 9
    je skip_ws_inc
    cmp al, 10
    je skip_ws_inc
    cmp al, 13
    je skip_ws_inc
    ret
skip_ws_inc:
    inc rbx
    jmp skip_ws_loop
skip_ws_ret:
    ret
parse_value:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je pv_true
    cmp al, 102
    je pv_false
    cmp al, 110
    je pv_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
    jmp parse_number
pv_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp parse_literal
pv_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp parse_literal
pv_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    jmp parse_literal
parse_literal:
    xor ecx, ecx
pl_loop:
    cmp rcx, rdx
    jae pl_done
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, byte ptr [rsi + rcx]
    jne fail
    inc rbx
    inc rcx
    jmp pl_loop
pl_done:
    ret
parse_string:
    inc rbx
ps_loop:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 34
    je ps_done
    cmp al, 92
    je ps_escape
    cmp al, 32
    jb fail
    inc rbx
    jmp ps_loop
ps_done:
    inc rbx
    ret
ps_escape:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 34
    je ps_esc_ok
    cmp al, 92
    je ps_esc_ok
    cmp al, 47
    je ps_esc_ok
    cmp al, 98
    je ps_esc_ok
    cmp al, 102
    je ps_esc_ok
    cmp al, 110
    je ps_esc_ok
    cmp al, 114
    je ps_esc_ok
    cmp al, 116
    je ps_esc_ok
    cmp al, 117
    je ps_unicode
    jmp fail
ps_esc_ok:
    inc rbx
    jmp ps_loop
ps_unicode:
    inc rbx
    mov ecx, 4
ps_u_loop:
    test rcx, rcx
    je ps_loop
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb fail
    cmp al, 57
    jbe ps_u_ok
    cmp al, 65
    jb fail
    cmp al, 70
    jbe ps_u_ok
    cmp al, 97
    jb fail
    cmp al, 102
    ja fail
ps_u_ok:
    inc rbx
    dec rcx
    jmp ps_u_loop
parse_number:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 45
    jne pn_int
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
pn_int:
    cmp al, 48
    je pn_zero
pn_digits:
    cmp rbx, r12
    jae pn_frac
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb pn_frac
    cmp al, 57
    ja pn_frac
    inc rbx
    jmp pn_digits
pn_zero:
    inc rbx
pn_frac:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 46
    jne pn_exp
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
pn_frac_digits:
    cmp rbx, r12
    jae pn_exp
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb pn_exp
    cmp al, 57
    ja pn_exp
    inc rbx
    jmp pn_frac_digits
pn_exp:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 101
    je pn_exp_begin
    cmp al, 69
    jne pn_done
pn_exp_begin:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 43
    je pn_exp_sign
    cmp al, 45
    je pn_exp_sign
    jmp pn_exp_first
pn_exp_sign:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
pn_exp_first:
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
pn_exp_digits:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 48
    jb pn_done
    cmp al, 57
    ja pn_done
    inc rbx
    jmp pn_exp_digits
pn_done:
    ret
parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 93
    je pa_done_inc
pa_loop:
    call parse_value
    cmp qword ptr [rip + err_flag], 0
    jne pa_ret
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 44
    je pa_comma
    cmp al, 93
    je pa_done_inc
    jmp fail
pa_comma:
    inc rbx
    call skip_ws
    jmp pa_loop
pa_done_inc:
    inc rbx
    ret
pa_ret:
    ret
parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 125
    je po_done_inc
po_loop:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 34
    jne fail
    call parse_string
    cmp qword ptr [rip + err_flag], 0
    jne po_ret
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 58
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    cmp qword ptr [rip + err_flag], 0
    jne po_ret
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 44
    je po_comma
    cmp al, 125
    je po_done_inc
    jmp fail
po_comma:
    inc rbx
    call skip_ws
    jmp po_loop
po_done_inc:
    inc rbx
    ret
po_ret:
    ret
.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
.bss
    .align 16
buffer:
    .zero 131072
    .align 8
err_flag:
    .zero 8
err_pos:
    .zero 8
numbuf:
    .zero 32
