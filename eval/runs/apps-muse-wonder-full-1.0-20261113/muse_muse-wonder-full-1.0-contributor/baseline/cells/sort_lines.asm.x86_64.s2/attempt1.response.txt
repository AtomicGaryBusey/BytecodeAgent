.globl _start
.text
_start:
  lea rbx, [rip + buffer]
  xor r12d, r12d
read_loop:
  lea rsi, [rbx + r12]
  mov rdx, 65536
  sub rdx, r12
  test rdx, rdx
  jz read_done
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  xor r13d, r13d
  xor r14d, r14d
  xor r15d, r15d
parse_loop:
  cmp r15, r12
  jae parse_end
  movzx eax, byte ptr [rbx + r15]
  cmp al, 10
  jne parse_next
  lea rcx, [rbx + r14]
  lea rax, [rip + line_ptr]
  mov qword ptr [rax + r13*8], rcx
  mov rcx, r15
  sub rcx, r14
  lea rax, [rip + line_len]
  mov qword ptr [rax + r13*8], rcx
  inc r13
  mov r14, r15
  inc r14
parse_next:
  inc r15
  jmp parse_loop
parse_end:
  cmp r14, r12
  jae sort_start
  lea rcx, [rbx + r14]
  lea rax, [rip + line_ptr]
  mov qword ptr [rax + r13*8], rcx
  mov rcx, r12
  sub rcx, r14
  lea rax, [rip + line_len]
  mov qword ptr [rax + r13*8], rcx
  inc r13
sort_start:
  mov r12, 1
sort_outer:
  cmp r12, r13
  jae sort_done
  lea rax, [rip + line_ptr]
  mov r15, qword ptr [rax + r12*8]
  lea rax, [rip + line_len]
  mov rbp, qword ptr [rax + r12*8]
  mov r14, r12
  dec r14
sort_inner:
  cmp r14, 0
  jl sort_insert
  lea rax, [rip + line_ptr]
  mov rsi, qword ptr [rax + r14*8]
  lea rax, [rip + line_len]
  mov rdx, qword ptr [rax + r14*8]
  mov rdi, r15
  mov rcx, rbp
  call compare_lines
  test eax, eax
  jle sort_insert
  lea rax, [rip + line_ptr]
  mov rcx, qword ptr [rax + r14*8]
  mov qword ptr [rax + r14*8 + 8], rcx
  lea rax, [rip + line_len]
  mov rcx, qword ptr [rax + r14*8]
  mov qword ptr [rax + r14*8 + 8], rcx
  dec r14
  jmp sort_inner
sort_insert:
  lea rax, [rip + line_ptr]
  mov qword ptr [rax + r14*8 + 8], r15
  lea rax, [rip + line_len]
  mov qword ptr [rax + r14*8 + 8], rbp
  inc r12
  jmp sort_outer
sort_done:
  xor r12d, r12d
out_loop:
  cmp r12, r13
  jae done_all
  lea rax, [rip + line_len]
  mov rdx, qword ptr [rax + r12*8]
  test rdx, rdx
  jz out_newline
  lea rax, [rip + line_ptr]
  mov rsi, qword ptr [rax + r12*8]
  mov edi, 1
  mov eax, 1
  syscall
out_newline:
  mov edi, 1
  lea rsi, [rip + newline_char]
  mov edx, 1
  mov eax, 1
  syscall
  inc r12
  jmp out_loop
done_all:
  xor edi, edi
  mov eax, 60
  syscall
halt_loop:
  jmp halt_loop
compare_lines:
  mov r8, rdx
  cmp r8, rcx
  jbe cmp_min_ok
  mov r8, rcx
cmp_min_ok:
  xor r9d, r9d
cmp_loop:
  cmp r9, r8
  jae cmp_by_len
  movzx eax, byte ptr [rsi + r9]
  movzx r10d, byte ptr [rdi + r9]
  cmp eax, r10d
  jb cmp_less
  ja cmp_greater
  inc r9
  jmp cmp_loop
cmp_by_len:
  cmp rdx, rcx
  jb cmp_less
  ja cmp_greater
  xor eax, eax
  ret
cmp_less:
  mov eax, -1
  ret
cmp_greater:
  mov eax, 1
  ret
.section .rodata
newline_char:
  .byte 10
.bss
  .balign 8
buffer:
  .zero 65536
  .balign 8
line_ptr:
  .zero 262144
  .balign 8
line_len:
  .zero 262144
