.equ BUFSZ, 262144
.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
nl_msg:
    .ascii "\n"
.data
pos:
    .quad 0
inlen:
    .quad 0
.bss
    .balign 16
buf:
    .zero BUFSZ
numbuf:
    .zero 32
.text
.globl _start
_start:
    xor r12, r12
    lea r14, [rip + buf]
read_loop:
    mov rdx, BUFSZ
    sub rdx, r12
    cmp rdx, 0
    je read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r14 + r12]
    syscall
    cmp rax, 0
    je read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + inlen], r12
    xor eax, eax
    mov qword ptr [rip + pos], rax
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + inlen]
    jne fail
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    mov rax, 60
    xor edi, edi
    syscall
skip_ws:
skip_ws_loop:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae skip_ws_done
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 32
    je skip_ws_advance
    cmp al, 9
    je skip_ws_advance
    cmp al, 10
    je skip_ws_advance
    cmp al, 13
    je skip_ws_advance
    ret
skip_ws_advance:
    inc qword ptr [rip + pos]
    jmp skip_ws_loop
skip_ws_done:
    ret
expect_byte:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, bl
    jne fail
    inc qword ptr [rip + pos]
    ret
parse_string:
    inc qword ptr [rip + pos]
string_loop:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 34
    je string_done
    cmp al, 92
    je string_escape
    cmp al, 32
    jb fail
    inc qword ptr [rip + pos]
    jmp string_loop
string_done:
    inc qword ptr [rip + pos]
    ret
string_escape:
    inc qword ptr [rip + pos]
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 34
    je string_escape_done
    cmp al, 92
    je string_escape_done
    cmp al, 47
    je string_escape_done
    cmp al, 98
    je string_escape_done
    cmp al, 102
    je string_escape_done
    cmp al, 110
    je string_escape_done
    cmp al, 114
    je string_escape_done
    cmp al, 116
    je string_escape_done
    cmp al, 117
    je string_unicode
    jmp fail
string_escape_done:
    inc qword ptr [rip + pos]
    jmp string_loop
string_unicode:
    inc qword ptr [rip + pos]
    mov ebx, 4
string_unicode_loop:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 48
    jb fail
    cmp al, 57
    jbe string_unicode_ok
    cmp al, 65
    jb fail
    cmp al, 70
    jbe string_unicode_ok
    cmp al, 97
    jb fail
    cmp al, 102
    ja fail
string_unicode_ok:
    inc qword ptr [rip + pos]
    dec ebx
    jnz string_unicode_loop
    jmp string_loop
parse_value:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_value_string
    cmp al, 116
    je parse_true
    cmp al, 102
    je parse_false
    cmp al, 110
    je parse_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
    jmp parse_number
parse_value_string:
    call parse_string
    ret
parse_true:
    mov bl, 116
    call expect_byte
    mov bl, 114
    call expect_byte
    mov bl, 117
    call expect_byte
    mov bl, 101
    call expect_byte
    ret
parse_false:
    mov bl, 102
    call expect_byte
    mov bl, 97
    call expect_byte
    mov bl, 108
    call expect_byte
    mov bl, 115
    call expect_byte
    mov bl, 101
    call expect_byte
    ret
parse_null:
    mov bl, 110
    call expect_byte
    mov bl, 117
    call expect_byte
    mov bl, 108
    call expect_byte
    mov bl, 108
    call expect_byte
    ret
parse_array:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 93
    jne array_loop
    inc qword ptr [rip + pos]
    ret
array_loop:
    call parse_value
    call skip_ws
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 44
    je array_comma
    cmp al, 93
    jne fail
    inc qword ptr [rip + pos]
    ret
array_comma:
    inc qword ptr [rip + pos]
    call skip_ws
    jmp array_loop
parse_object:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]

    inc qword ptr [rip + pos]
    ret
object_loop:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail

    mov rdx, qword ptr [rip + pos]

    jae fail

    call skip_ws
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 44
    je object_comma
    cmp al, 125
    jne fail
    inc qword ptr [rip + pos]
    ret
object_comma:
    inc qword ptr [rip + pos]
    call skip_ws
    jmp object_loop
parse_number:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 45
    jne number_whole
    inc qword ptr [rip + pos]
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail

    mov rdx, qword ptr [rip + pos]

    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 48
    je number_whole_zero
    cmp al, 49
    jb fail
    cmp al, 57
    ja fail
    inc qword ptr [rip + pos]
    jmp number_whole_loop
number_whole_zero:
    inc qword ptr [rip + pos]
    jmp number_frac
number_whole_loop:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae number_frac
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 48
    jb number_frac
    cmp al, 57
    ja number_frac

number_frac:
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae number_done
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 46

    inc qword ptr [rip + pos]
    mov rdx, qword ptr [rip + pos]
    cmp rdx, qword ptr [rip + inlen]
    jae fail
    lea rcx, [rip + buf]
    movzx eax, byte ptr [rcx + rdx]
    cmp al, 48
    jb fail

number_frac_loop:

number_exp_take:
    inc qword ptr [rip + pos]

    jae fail

    movzx eax, byte ptr [rcx + rdx]

    cmp al, 45

number_exp_sign:

number_exp_require:
    mov rdx, qword ptr [rip + pos]

fail:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, 9
    syscall
    mov rax, qword ptr [rip + pos]
    lea rbx, [rip + numbuf + 32]
    cmp rax, 0
    jne convert_loop
    dec rbx
    mov byte ptr [rbx], 48
    jmp write_digits
convert_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, 48
    dec rbx
    mov byte ptr [rbx], dl
    cmp rax, 0
    jne convert_loop
write_digits:
    lea rcx, [rip + numbuf + 32]
    sub rcx, rbx
    mov rdx, rcx
    mov rsi, rbx
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl_msg]
    mov rdx, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
