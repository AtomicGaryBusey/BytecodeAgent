.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    adrp x25, hexdigits
    add x25, x25, :lo12:hexdigits
    mov x20, #0
read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
row_loop:
    cmp x21, x20
    b.hs rows_done
    sub x22, x20, x21
    cmp x22, #16
    b.lo row_n_ok
    mov x22, #16
row_n_ok:
    mov x23, x24
    mov x26, #28
off_digit_loop:
    lsr x27, x21, x26
    and x27, x27, #15
    ldrb w28, [x25, x27]
    strb w28, [x23], #1
    subs x26, x26, #4
    b.ge off_digit_loop
    mov w28, #32
    strb w28, [x23], #1
    strb w28, [x23], #1
    mov x26, #0
hex_slot_loop:
    cmp x26, #16
    b.hs hex_slots_done
    cmp x26, x22
    b.hs hex_slot_blank
    add x27, x21, x26
    ldrb w27, [x19, x27]
    lsr w28, w27, #4
    ldrb w29, [x25, x28]
    strb w29, [x23], #1
    and w28, w27, #15
    ldrb w29, [x25, x28]
    strb w29, [x23], #1
    mov w28, #32
    strb w28, [x23], #1
    b hex_slot_after
hex_slot_blank:
    mov w28, #32
    strb w28, [x23], #1
    strb w28, [x23], #1
    strb w28, [x23], #1
hex_slot_after:
    cmp x26, #7
    b.ne hex_slot_noextra
    mov w28, #32
    strb w28, [x23], #1
hex_slot_noextra:
    add x26, x26, #1
    b hex_slot_loop
hex_slots_done:
    mov w28, #32
    strb w28, [x23], #1
    mov w28, #124
    strb w28, [x23], #1
    mov x26, #0
ascii_loop:
    cmp x26, x22
    b.hs ascii_done
    add x27, x21, x26
    ldrb w27, [x19, x27]
    cmp w27, #32
    b.lo ascii_dot
    cmp w27, #126
    b.hi ascii_dot
    strb w27, [x23], #1
    b ascii_next
ascii_dot:
    mov w28, #46
    strb w28, [x23], #1
ascii_next:
    add x26, x26, #1
    b ascii_loop
ascii_done:
    mov w28, #124
    strb w28, [x23], #1
    mov w28, #10
    strb w28, [x23], #1
    mov x0, #1
    mov x1, x24
    sub x2, x23, x24
    mov x8, #64
    svc #0
    add x21, x21, x22
    b row_loop
rows_done:
    mov x23, x24
    mov x26, #28
final_digit_loop:
    lsr x27, x20, x26
    and x27, x27, #15
    ldrb w28, [x25, x27]
    strb w28, [x23], #1
    subs x26, x26, #4
    b.ge final_digit_loop
    mov w28, #10
    strb w28, [x23], #1
    mov x0, #1
    mov x1, x24
    sub x2, x23, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
hang:
    b hang
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 128
