.globl _start
.text
_start:
    xor ebx, ebx
    xor ebp, ebp
    xor r12d, r12d
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    lea rsi, [rip + buffer]
    mov rcx, rax
proc_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_nl
    inc rbx
not_nl:
    cmp al, 32
    je is_ws
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe is_ws
    test r13, r13
    jnz next_byte
    inc rbp
    mov r13d, 1
    jmp next_byte
is_ws:
    xor r13d, r13d
next_byte:
    dec rcx
    jnz proc_loop
    jmp read_loop
done_read:
    lea rdi, [rip + outbuf]
    mov rax, rbx
    call write_dec
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, rbp
    call write_dec
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call write_dec
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
write_dec:
    test rax, rax
    jnz wd_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret
wd_nonzero:
    sub rsp, 32
    lea r9, [rsp + 32]
    xor r8d, r8d
    mov ecx, 10
wd_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    inc r8
    test rax, rax
    jnz wd_loop
wd_copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    dec r8
    jnz wd_copy
    add rsp, 32
    ret
.bss
    .balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
