.globl _start
.text
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
read_loop:
    mov rsi, r12
    add rsi, r13
    mov rdx, 32768
    sub rdx, r13
    test rdx, rdx
    jz read_done
    mov rax, 0
    mov rdi, 0
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
    xor rbx, rbx
    xor rcx, rcx
parse_loop:
    cmp rcx, r13
    jae parse_after
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 10
    jne parse_next
    lea rax, [r12 + rbx]
    lea rdx, [rip + entries_a]
    mov r8, r14
    shl r8, 4
    add rdx, r8
    mov qword ptr [rdx], rax
    mov rax, rcx
    sub rax, rbx
    mov qword ptr [rdx + 8], rax
    inc r14
    lea rbx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_after:
    cmp rbx, r13
    jae sort_init
    lea rax, [r12 + rbx]
    lea rdx, [rip + entries_a]
    mov r8, r14
    shl r8, 4
    add rdx, r8
    mov qword ptr [rdx], rax
    mov rax, r13
    sub rax, rbx
    mov qword ptr [rdx + 8], rax
    inc r14
sort_init:
    mov qword ptr [rip + sort_n], r14
    lea rax, [rip + entries_a]
    mov qword ptr [rip + src_base], rax
    lea rax, [rip + entries_b]
    mov qword ptr [rip + dst_base], rax
    cmp r14, 1
    jbe sort_done
    mov qword ptr [rip + width_var], 1
width_loop:
    mov rax, qword ptr [rip + width_var]
    cmp rax, qword ptr [rip + sort_n]
    jae sort_done
    mov qword ptr [rip + lo_var], 0
lo_loop:
    mov rax, qword ptr [rip + lo_var]
    cmp rax, qword ptr [rip + sort_n]
    jae lo_done
    mov rax, qword ptr [rip + lo_var]
    add rax, qword ptr [rip + width_var]
    cmp rax, qword ptr [rip + sort_n]
    jbe 1f
    mov rax, qword ptr [rip + sort_n]
1:
    mov qword ptr [rip + mid_var], rax
    mov rax, qword ptr [rip + lo_var]
    mov rcx, qword ptr [rip + width_var]
    shl rcx, 1
    add rax, rcx
    cmp rax, qword ptr [rip + sort_n]
    jbe 1f
    mov rax, qword ptr [rip + sort_n]
1:
    mov qword ptr [rip + hi_var], rax
    mov rax, qword ptr [rip + lo_var]
    mov qword ptr [rip + i_var], rax
    mov qword ptr [rip + k_var], rax
    mov rax, qword ptr [rip + mid_var]
    mov qword ptr [rip + j_var], rax
merge_loop:
    mov rax, qword ptr [rip + i_var]
    cmp rax, qword ptr [rip + mid_var]
    jae tail_j_loop
    mov rax, qword ptr [rip + j_var]
    cmp rax, qword ptr [rip + hi_var]
    jae tail_i_loop
    mov rax, qword ptr [rip + src_base]
    mov rcx, qword ptr [rip + i_var]
    shl rcx, 4
    add rax, rcx
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov rax, qword ptr [rip + src_base]
    mov rcx, qword ptr [rip + j_var]
    shl rcx, 4
    add rax, rcx
    mov rdx, qword ptr [rax]
    mov rcx, qword ptr [rax + 8]
    call compare_lines
    cmp eax, 0
    jg copy_j_main
    mov rdi, qword ptr [rip + i_var]
    mov rsi, qword ptr [rip + k_var]
    call copy_entry
    inc qword ptr [rip + i_var]
    inc qword ptr [rip + k_var]
    jmp merge_loop
copy_j_main:
    mov rdi, qword ptr [rip + j_var]
    mov rsi, qword ptr [rip + k_var]
    call copy_entry
    inc qword ptr [rip + j_var]
    inc qword ptr [rip + k_var]
    jmp merge_loop
tail_i_loop:
    mov rax, qword ptr [rip + i_var]
    cmp rax, qword ptr [rip + mid_var]
    jae merge_done
    mov rdi, qword ptr [rip + i_var]
    mov rsi, qword ptr [rip + k_var]
    call copy_entry
    inc qword ptr [rip + i_var]
    inc qword ptr [rip + k_var]
    jmp tail_i_loop
tail_j_loop:
    mov rax, qword ptr [rip + j_var]
    cmp rax, qword ptr [rip + hi_var]
    jae merge_done
    mov rdi, qword ptr [rip + j_var]
    mov rsi, qword ptr [rip + k_var]
    call copy_entry
    inc qword ptr [rip + j_var]
    inc qword ptr [rip + k_var]
    jmp tail_j_loop
merge_done:
    mov rax, qword ptr [rip + width_var]
    shl rax, 1
    add qword ptr [rip + lo_var], rax
    jmp lo_loop
lo_done:
    mov rax, qword ptr [rip + src_base]
    mov rcx, qword ptr [rip + dst_base]
    mov qword ptr [rip + src_base], rcx
    mov qword ptr [rip + dst_base], rax
    mov rax, qword ptr [rip + width_var]
    shl rax, 1
    mov qword ptr [rip + width_var], rax
    jmp width_loop
sort_done:
    xor rbx, rbx
output_loop:
    cmp rbx, qword ptr [rip + sort_n]
    jae exit_prog
    mov rax, qword ptr [rip + src_base]
    mov rcx, rbx
    shl rcx, 4
    add rax, rcx
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    test rdx, rdx
    jz write_newline
    call write_all
write_newline:
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all
    inc rbx
    jmp output_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
compare_lines:
    xor r8, r8
cmp_loop:
    cmp r8, rsi
    jae cmp_a_done
    cmp r8, rcx
    jae cmp_gt
    movzx eax, byte ptr [rdi + r8]
    movzx r9d, byte ptr [rdx + r8]
    cmp eax, r9d
    jb cmp_lt
    ja cmp_gt
    inc r8
    jmp cmp_loop
cmp_a_done:
    cmp r8, rcx
    jae cmp_eq
cmp_lt:
    mov eax, -1
    ret
cmp_gt:
    mov eax, 1
    ret
cmp_eq:
    xor eax, eax
    ret
copy_entry:
    mov rax, qword ptr [rip + src_base]
    mov rcx, rdi
    shl rcx, 4
    add rax, rcx
    mov r8, qword ptr [rax]
    mov r9, qword ptr [rax + 8]
    mov rax, qword ptr [rip + dst_base]
    mov rcx, rsi
    shl rcx, 4
    add rax, rcx
    mov qword ptr [rax], r8
    mov qword ptr [rax + 8], r9
    ret
write_all:
    test rdx, rdx
    jz wa_done
wa_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle wa_done
    add rsi, rax
    sub rdx, rax
    jnz wa_loop
wa_done:
    ret
.section .rodata
newline_byte:
    .byte 10
.bss
.balign 8
input_buf:
    .zero 32768
.balign 8
entries_a:
    .zero 16000
.balign 8
entries_b:
    .zero 16000
.balign 8
src_base:
    .zero 8
dst_base:
    .zero 8
sort_n:
    .zero 8
width_var:
    .zero 8
lo_var:
    .zero 8
mid_var:
    .zero 8
hi_var:
    .zero 8
i_var:
    .zero 8
j_var:
    .zero 8
k_var:
    .zero 8
