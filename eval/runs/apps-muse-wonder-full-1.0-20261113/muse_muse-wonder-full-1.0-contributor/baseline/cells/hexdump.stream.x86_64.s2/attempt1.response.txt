.globl _start
.text
_start:
 lea rbx, [rip + input]
 xor r12d, r12d
read_loop:
 cmp r12, 65536
 jae done_read
 lea rsi, [rbx + r12]
 mov rdx, 65536
 sub rdx, r12
 xor edi, edi
 xor eax, eax
 syscall
 cmp rax, 0
 jle done_read
 add r12, rax
 jmp read_loop
done_read:
 xor r13d, r13d
row_loop:
 cmp r13, r12
 jae final_line
 mov rax, r12
 sub rax, r13
 cmp rax, 16
 jbe 1f
 mov eax, 16
1:
 mov r14, rax
 lea r15, [rip + linebuf]
 mov ecx, 28
offset_loop:
 mov rax, r13
 shr rax, cl
 and eax, 15
 mov rdx, rax
 lea rsi, [rip + hextab]
 mov dl, byte ptr [rsi + rdx]
 mov byte ptr [r15], dl
 inc r15
 sub ecx, 4
 jns offset_loop
 mov byte ptr [r15], 32
 inc r15
 mov byte ptr [r15], 32
 inc r15
 xor ebp, ebp
pos_loop:
 cmp rbp, r14
 jae pad_slot
 lea rax, [r13 + rbp]
 movzx eax, byte ptr [rbx + rax]
 mov edx, eax
 shr edx, 4
 and edx, 15
 lea rsi, [rip + hextab]
 mov dl, byte ptr [rsi + rdx]
 mov byte ptr [r15], dl
 inc r15
 and eax, 15
 mov edx, eax
 lea rsi, [rip + hextab]
 mov dl, byte ptr [rsi + rdx]
 mov byte ptr [r15], dl
 inc r15
 mov byte ptr [r15], 32
 inc r15
 jmp after_slot
pad_slot:
 mov byte ptr [r15], 32
 inc r15
 mov byte ptr [r15], 32
 inc r15
 mov byte ptr [r15], 32
 inc r15
after_slot:
 cmp rbp, 7
 jne no_extra
 mov byte ptr [r15], 32
 inc r15
no_extra:
 inc rbp
 cmp rbp, 16
 jb pos_loop
 mov byte ptr [r15], 32
 inc r15
 mov byte ptr [r15], 124
 inc r15
 xor ebp, ebp
ascii_loop:
 cmp rbp, r14
 jae ascii_done
 lea rax, [r13 + rbp]
 movzx eax, byte ptr [rbx + rax]
 cmp eax, 32
 jb dot
 cmp eax, 126
 ja dot
 mov byte ptr [r15], al
 jmp ascii_next
dot:
 mov byte ptr [r15], 46
ascii_next:
 inc r15
 inc rbp
 jmp ascii_loop
ascii_done:
 mov byte ptr [r15], 124
 inc r15
 mov byte ptr [r15], 10
 inc r15
 lea rsi, [rip + linebuf]
 mov rdx, r15
 sub rdx, rsi
 mov edi, 1
 mov eax, 1
 syscall
 add r13, r14
 jmp row_loop
final_line:
 lea r15, [rip + linebuf]
 mov ecx, 28
final_loop:
 mov rax, r12
 shr rax, cl
 and eax, 15
 mov rdx, rax
 lea rsi, [rip + hextab]
 mov dl, byte ptr [rsi + rdx]
 mov byte ptr [r15], dl
 inc r15
 sub ecx, 4
 jns final_loop
 mov byte ptr [r15], 10
 inc r15
 lea rsi, [rip + linebuf]
 mov rdx, r15
 sub rdx, rsi
 mov edi, 1
 mov eax, 1
 syscall
 xor edi, edi
 mov eax, 60
 syscall
.section .rodata
hextab:
 .ascii "0123456789abcdef"
.bss
 .balign 16
input:
 .zero 65536
linebuf:
 .zero 128
