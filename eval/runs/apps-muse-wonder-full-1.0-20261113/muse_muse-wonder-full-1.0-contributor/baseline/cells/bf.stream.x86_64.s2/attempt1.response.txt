.text
_start:
 lea rbx, [rip + input_buf]
 xor r12d, r12d
read_loop:
 cmp r12, 65536
 jae read_done
 xor eax, eax
 xor edi, edi
 lea rsi, [rbx + r12]
 mov rdx, 65536
 sub rdx, r12
 syscall
 test rax, rax
 jz read_done
 js read_done
 add r12, rax
 jmp read_loop
read_done:
 xor ecx, ecx
scan_loop:
 cmp rcx, r12
 jae scan_not_found
 cmp byte ptr [rbx + rcx], 33
 je scan_found
 inc rcx
 jmp scan_loop
scan_found:
 mov r14, r12
 mov r12, rcx
 lea r15, [rcx + 1]
 jmp filter_start
scan_not_found:
 mov r14, r12
 mov r15, r12
filter_start:
 lea r9, [rip + prog_buf]
 xor ecx, ecx
 xor r10d, r10d
filter_loop:
 cmp rcx, r12
 jae filter_done

 cmp al, 60

 je filter_keep
 cmp al, 46

 cmp al, 44

 cmp al, 91
 je filter_keep
 cmp al, 93
 je filter_keep
 jmp filter_loop
filter_keep:
 mov byte ptr [r9 + r10], al
 inc r10
 jmp filter_loop
filter_done:
 mov r12, r10
 lea r8, [rip + match_table]
 xor ecx, ecx
 xor edx, edx
bracket_loop:
 cmp rcx, r12
 jae bracket_done
 movzx eax, byte ptr [r9 + rcx]
 cmp al, 91
 je bracket_open
 cmp al, 93
 je bracket_close
 inc rcx
 jmp bracket_loop
bracket_open:
 push rcx
 inc rdx
 inc rcx
 jmp bracket_loop
bracket_close:
 test rdx, rdx
 jz exit_one
 dec rdx
 pop rax
 mov dword ptr [r8 + rcx*4], eax
 mov dword ptr [r8 + rax*4], ecx
 inc rcx
 jmp bracket_loop
bracket_done:
 test rdx, rdx
 jnz exit_one
 xor r13d, r13d
 lea rbp, [rip + tape]
interp_loop:
 cmp r13, r12
 jae exit_zero
 movzx eax, byte ptr [r9 + r13]
 inc r13
 cmp al, 62
 je do_gt
 cmp al, 60
 je do_lt
 cmp al, 43
 je do_plus
 cmp al, 45
 je do_minus
 cmp al, 46
 je do_dot
 cmp al, 44
 je do_comma
 cmp al, 91
 je do_open
 cmp al, 93
 je do_close
 jmp interp_loop
do_gt:
 inc rbp
 jmp interp_loop
do_lt:
 dec rbp
 jmp interp_loop
do_plus:
 inc byte ptr [rbp]
 jmp interp_loop
do_minus:

 jmp interp_loop
do_dot:

 mov edi, 1

do_comma:
 cmp r15, r14
 jae comma_eof
 mov al, byte ptr [rbx + r15]
 inc r15

 jmp interp_loop
comma_eof:
 mov byte ptr [rbp], 0

do_open:
 cmp byte ptr [rbp], 0
 jne interp_loop
 lea rcx, [r13 - 1]
 mov eax, dword ptr [r8 + rcx*4]
 lea r13, [rax + 1]
 jmp interp_loop
do_close:
 cmp byte ptr [rbp], 0
 je interp_loop
 lea rcx, [r13 - 1]
 mov eax, dword ptr [r8 + rcx*4]
 lea r13, [rax + 1]
 jmp interp_loop
exit_zero:
 mov eax, 60
 xor edi, edi
 syscall
 jmp exit_zero
exit_one:
 mov eax, 60
 mov edi, 1

 jmp exit_one
