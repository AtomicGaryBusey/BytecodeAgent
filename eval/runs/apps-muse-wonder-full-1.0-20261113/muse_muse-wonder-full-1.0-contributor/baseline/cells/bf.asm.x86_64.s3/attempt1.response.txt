.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rcx, rcx
find_loop:
    cmp rcx, r12
    jae find_none
    cmp byte ptr [rbx + rcx], 33
    je find_found
    inc rcx
    jmp find_loop
find_none:
    mov r13, r12
    mov r14, r12
    jmp filter_start
find_found:
    mov r13, rcx
    lea r14, [rcx + 1]
filter_start:
    xor rcx, rcx
    xor rdx, rdx
filter_loop:
    cmp rcx, r13
    jae filter_done
    mov al, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 62
    je f_gt
    cmp al, 60
    je f_lt
    cmp al, 43
    je f_plus
    cmp al, 45
    je f_minus
    cmp al, 46
    je f_dot
    cmp al, 44
    je f_comma
    cmp al, 91
    je f_open
    cmp al, 93
    je f_close
    jmp filter_loop
f_gt:
    mov al, 0
    jmp f_keep
f_lt:
    mov al, 1
    jmp f_keep
f_plus:
    mov al, 2
    jmp f_keep
f_minus:
    mov al, 3
    jmp f_keep
f_dot:
    mov al, 4
    jmp f_keep
f_comma:
    mov al, 5
    jmp f_keep
f_open:
    mov al, 6
    jmp f_keep
f_close:
    mov al, 7
    jmp f_keep
f_keep:
    mov byte ptr [rbx + rdx], al
    inc rdx
    jmp filter_loop
filter_done:
    mov r13, rdx
    lea r15, [rip + jumptab]
    xor rcx, rcx
    xor r8, r8
build_loop:
    cmp rcx, r13
    jae build_done
    mov al, byte ptr [rbx + rcx]
    cmp al, 6
    je b_open
    cmp al, 7
    je b_close
    inc rcx
    jmp build_loop
b_open:
    push rcx
    inc r8
    inc rcx
    jmp build_loop
b_close:
    test r8, r8
    jz fail_exit
    pop rdx
    dec r8
    lea r9, [rcx + 1]
    mov dword ptr [r15 + rdx*4], r9d
    lea r9, [rdx + 1]
    mov dword ptr [r15 + rcx*4], r9d
    inc rcx
    jmp build_loop
build_done:
    test r8, r8
    jnz fail_exit
    xor r8, r8
    lea r9, [rip + tape]
    lea r10, [rip + outbuf]
exec_loop:
    cmp r8, r13
    jae exec_done
    movzx eax, byte ptr [rbx + r8]
    inc r8
    cmp al, 4
    jb exec_low
    cmp al, 6
    jb exec_mid
    je do_open
do_close:
    cmp byte ptr [r9], 0
    je exec_loop
    mov r8d, dword ptr [r15 + r8*4 - 4]
    jmp exec_loop
do_open:
    cmp byte ptr [r9], 0
    jne exec_loop
    mov r8d, dword ptr [r15 + r8*4 - 4]
    jmp exec_loop
exec_mid:
    cmp al, 4
    jne do_comma
    mov al, byte ptr [r9]
    mov byte ptr [r10], al
    inc r10
    jmp exec_loop
do_comma:
    cmp r14, r12
    jae comma_eof
    mov al, byte ptr [rbx + r14]
    inc r14
    mov byte ptr [r9], al
    jmp exec_loop
comma_eof:
    mov byte ptr [r9], 0
    jmp exec_loop
exec_low:
    cmp al, 2
    jb exec_ptr
    je do_plus
    dec byte ptr [r9]
    jmp exec_loop
do_plus:
    inc byte ptr [r9]
    jmp exec_loop
exec_ptr:
    test al, al
    jne do_lt
    inc r9
    jmp exec_loop
do_lt:
    dec r9
    jmp exec_loop
exec_done:
    lea rsi, [rip + outbuf]
    mov rdx, r10
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_ok
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
    jmp exit_ok
fail_exit:
    mov rax, 60
    mov rdi, 1
    syscall
    jmp fail_exit
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
tape:
    .zero 30000
.balign 4
jumptab:
    .zero 262144
