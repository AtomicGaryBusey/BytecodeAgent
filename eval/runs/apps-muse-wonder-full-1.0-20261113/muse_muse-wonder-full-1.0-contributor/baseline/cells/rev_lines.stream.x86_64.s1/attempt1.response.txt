.globl _start
.text
_start:
    lea rbx, [rip + buffer]
    xor ebp, ebp
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + rbp]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add rbp, rax
    jmp read_loop
done_read:
    xor r12d, r12d
line_loop:
    cmp r12, rbp
    jge exit_prog
    mov r13, r12
find_loop:
    cmp r13, rbp
    jge found_end
    movzx eax, byte ptr [rbx + r13]
    cmp al, 10
    je found_end
    inc r13
    jmp find_loop
found_end:
    mov r14, r12
    mov r15, r13
    dec r15
rev_loop:
    cmp r14, r15
    jge rev_done
    mov al, byte ptr [rbx + r14]
    mov cl, byte ptr [rbx + r15]
    mov byte ptr [rbx + r14], cl
    mov byte ptr [rbx + r15], al
    inc r14
    dec r15
    jmp rev_loop
rev_done:
    cmp r13, rbp
    jl has_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r12]
    mov rdx, r13
    sub rdx, r12
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_str]
    mov edx, 1
    syscall
    mov r12, r13
    jmp line_loop
has_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r12]
    mov rdx, r13
    sub rdx, r12
    inc rdx
    syscall
    mov r12, r13
    inc r12
    jmp line_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata
newline_str:
    .byte 10
.bss
.balign 16
buffer:
    .zero 262144
