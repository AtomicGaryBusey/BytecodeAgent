export function l $shift_sat(l %x, l %n) {
@start
	%nw =w copy %n
	%ge64 =w csgel %n, 64
	jnz %ge64, @l_large, @l_small
@l_large
	%left_l =l copy 0
	jmp @do_right
@l_small
	%left_s =l shl %x, %nw
	jmp @do_right
@do_right
	%left =l phi @l_large %left_l, @l_small %left_s
	jnz %ge64, @r_large, @r_small
@r_large
	%c63 =w copy 63
	%right_l =l sar %x, %c63
	jmp @add
@r_small
	%right_s =l sar %x, %nw
	jmp @add
@add
	%right =l phi @r_large %right_l, @r_small %right_s
	%res =l add %left, %right
	ret %res
}
