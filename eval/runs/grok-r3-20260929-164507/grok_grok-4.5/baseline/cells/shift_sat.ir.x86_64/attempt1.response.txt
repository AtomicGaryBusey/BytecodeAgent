export function l $shift_sat(l %x, l %n) {
@start
	%nw =extrl %n
	%ge64 =csgel %n, 64
	jnz %ge64, @l_large, @l_small
@l_large
	%left_l =copy 0
	jmp @do_right
@l_small
	%left_s =shl l %x, %nw
	jmp @do_right
@do_right
	%left =phi @l_large %left_l, @l_small %left_s
	jnz %ge64, @r_large, @r_small
@r_large
	%c63 =copy 63
	%right_l =sar l %x, %c63
	jmp @add
@r_small
	%right_s =sar l %x, %nw
	jmp @add
@add
	%right =phi @r_large %right_l, @r_small %right_s
	%res =add l %left, %right
	ret %res
}
