bubble_sort:
mov x2, x0
mov x3, x1
cmp x3, #1
b.le .Lret0
mov x4, #0
.Louter:
sub x5, x3, x4
sub x5, x5, #1
mov x6, #0
.Linner:
ldr x7, [x2, x6, lsl #3]
add x8, x6, #1
ldr x9, [x2, x8, lsl #3]
cmp x7, x9
b.le .Lnoswap
str x9, [x2, x6, lsl #3]
str x7, [x2, x8, lsl #3]
.Lnoswap:
add x6, x6, #1
cmp x6, x5
b.lt .Linner
add x4, x4, #1
sub x5, x3, #1
cmp x4, x5
b.lt .Louter
.Lret0:
mov x0, #0
ret
