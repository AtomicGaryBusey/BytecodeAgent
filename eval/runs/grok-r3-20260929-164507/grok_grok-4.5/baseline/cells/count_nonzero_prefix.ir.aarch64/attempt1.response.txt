export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
	%nzero =w cnel %n, 0
	jnz %nzero, @loop, @retn
@loop
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%iszero =w ceql %val, 0
	jnz %iszero, @reti, @cont
@cont
	%i =l add %i, 1
	%lt =w csltl %i, %n
	jnz %lt, @loop, @retn
@reti
	ret %i
@retn
	ret %n
}
