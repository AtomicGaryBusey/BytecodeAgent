export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%min =l shl 1, 63
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @checkovf
@divzero
	storel %a, %rem
	ret 0
@checkovf
	%ismin =w ceql %a, %min
	%isneg1 =w ceql %b, -1
	%ovf =w and %ismin, %isneg1
	jnz %ovf, @overflow, @normal
@overflow
	storel 0, %rem
	ret %min
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
