export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%conv =l alloc8 8
	jmp @loop
@loop
	%i =l phi @start 0, @rep.emit %i.rep, @copy.done %i.lit
	%o =l phi @start 0, @rep.emit %o.rep, @copy.done %o.lit
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%pi =l add %src, %i
	%b =w loadub %pi
	jmp @run.loop
@run.loop
	%r =l phi @body 1, @run.inc %r.next
	%ir =l add %i, %r
	%ltn =w csltl %ir, %n
	jnz %ltn, @run.check128, @run.done
@run.check128
	%ltr =w csltl %r, 128
	jnz %ltr, @run.cmpb, @run.done
@run.cmpb
	%pir =l add %src, %ir
	%br =w loadub %pir
	%eq =w ceqw %b, %br
	jnz %eq, @run.inc, @run.done
@run.inc
	%r.next =l add %r, 1
	jmp @run.loop
@run.done
	%ge =w cslel 3, %r
	jnz %ge, @rep, @lit
@rep
	%need.rep =l add %o, 2
	%fits.rep =w cslel %need.rep, %cap
	jnz %fits.rep, @rep.emit, @fail
@rep.emit
	%hdrl.rep =l sub 257, %r
	storel %hdrl.rep, %conv
	%hdr.rep =w loadw %conv
	%po.rep0 =l add %out, %o
	storeb %hdr.rep, %po.rep0
	%o1.rep =l add %o, 1
	%po.rep1 =l add %out, %o1.rep
	storeb %b, %po.rep1
	%o.rep =l add %o, 2
	%i.rep =l add %i, %r
	jmp @loop
@lit
	jmp @lit.loop
@lit.loop
	%j =l phi @lit %i, @lit.inc %j.next
	%ltj =w csltl %j, %n
	jnz %ltj, @lit.checklen, @lit.done
@lit.checklen
	%diff =l sub %j, %i
	%lt128 =w csltl %diff, 128
	jnz %lt128, @lit.checkrun, @lit.done
@lit.checkrun
	%pj =l add %src, %j
	%bj =w loadub %pj
	jmp @rj.loop
@rj.loop
	%rj =l phi @lit.checkrun 1, @rj.inc %rj.next
	%jr =l add %j, %rj
	%ltjr =w csltl %jr, %n
	jnz %ltjr, @rj.check128, @rj.done
@rj.check128
	%ltrj =w csltl %rj, 128
	jnz %ltrj, @rj.cmp, @rj.done
@rj.cmp
	%pjr =l add %src, %jr
	%bjr =w loadub %pjr
	%eqj =w ceqw %bj, %bjr
	jnz %eqj, @rj.inc, @rj.done
@rj.inc
	%rj.next =l add %rj, 1
	jmp @rj.loop
@rj.done
	%gej =w cslel 3, %rj
	jnz %gej, @lit.done, @lit.inc
@lit.inc
	%j.next =l add %j, 1
	jmp @lit.loop
@lit.done
	%L =l sub %j, %i
	%need.lit =l add %L, 1
	%end.lit =l add %o, %need.lit
	%fits.lit =w cslel %end.lit, %cap
	jnz %fits.lit, @lit.emit, @fail
@lit.emit
	%hdrl.lit =l sub %L, 1
	storel %hdrl.lit, %conv
	%hdr.lit =w loadw %conv
	%po.lit =l add %out, %o
	storeb %hdr.lit, %po.lit
	%o.afterhdr =l add %o, 1
	jmp @copy.loop
@copy.loop
	%k =l phi @lit.emit 0, @copy.inc %k.next
	%o.cp =l phi @lit.emit %o.afterhdr, @copy.inc %o.next
	%ltk =w csltl %k, %L
	jnz %ltk, @copy.body, @copy.done
@copy.body
	%ik =l add %i, %k
	%pik =l add %src, %ik
	%bk =w loadub %pik
	%pok =l add %out, %o.cp
	storeb %bk, %pok
	jmp @copy.inc
@copy.inc
	%k.next =l add %k, 1
	%o.next =l add %o.cp, 1
	jmp @copy.loop
@copy.done
	%o.lit =l copy %o.cp
	%i.lit =l copy %j
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
