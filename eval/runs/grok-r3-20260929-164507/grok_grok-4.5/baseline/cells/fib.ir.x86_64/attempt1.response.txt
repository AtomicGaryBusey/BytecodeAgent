export function l $fib(l %n) {
@start
	%a0 =l copy 0
	%b0 =l copy 1
	%n0 =l copy %n
	jmp @loop
@loop
	%a =l phi @start %a0, @body %a1
	%b =l phi @start %b0, @body %b1
	%ncur =l phi @start %n0, @body %n1
	%cond =w csgtl %ncur, 0
	jnz %cond, @body, @end
@body
	%t =l add %a, %b
	%a1 =l copy %b
	%b1 =l copy %t
	%n1 =l sub %ncur, 1
	jmp @loop
@end
	ret %a
}
