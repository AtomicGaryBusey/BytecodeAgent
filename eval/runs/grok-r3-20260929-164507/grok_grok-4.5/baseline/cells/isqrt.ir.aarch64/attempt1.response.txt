export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @setlo %lo2, @sethi %lo
	%hi =l phi @start 4294967296, @setlo %hi, @sethi %hi2
	%lt =w cultl %lo, %hi
	jnz %lt, @body, @end
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	%lo2 =l add %mid, 1
	jmp @loop
@sethi
	%hi2 =l copy %mid
	jmp @loop
@end
	%res =l sub %lo, 1
	ret %res
}
