export function l $count_ugt(l %a, l %n, l %t) {
@start
	%c0 =l copy 0
	%i0 =l copy 0
	%z =w ceql %n, 0
	jnz %z, @end, @loop
@loop
	%c =l phi @start %c0, @aft %cn
	%i =l phi @start %i0, @aft %in
	%off =l mul %i, 8
	%ad =l add %a, %off
	%v =l loadl %ad
	%gt =w cugtl %v, %t
	jnz %gt, @yes, @no
@yes
	%cy =l add %c, 1
	jmp @aft
@no
	%cno =l copy %c
	jmp @aft
@aft
	%cn =l phi @yes %cy, @no %cno
	%in =l add %i, 1
	%more =w cultl %in, %n
	jnz %more, @loop, @end
@end
	%res =l phi @start %c0, @aft %cn
	ret %res
}
