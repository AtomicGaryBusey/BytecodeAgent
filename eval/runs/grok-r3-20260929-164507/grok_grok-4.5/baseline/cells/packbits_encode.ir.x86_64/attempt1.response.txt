export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @dorep %i2, @copydone %i_lit
	%o =l phi @start 0, @dorep %o2, @copydone %o_lit
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%pi =l add %src, %i
	%b0 =w loadub %pi
	jmp @runloop
@runloop
	%r =w phi @body 1, @runinc %rnext
	%rl =l extuw %r
	%ri =l add %i, %rl
	%cmprn =w csltl %ri, %n
	jnz %cmprn, @runcap, @runend
@runcap
	%cmprc =w csltw %r, 128
	jnz %cmprc, @runeq, @runend
@runeq
	%pri =l add %src, %ri
	%bi =w loadub %pri
	%eq =w ceqw %bi, %b0
	jnz %eq, @runinc, @runend
@runinc
	%rnext =w add %r, 1
	jmp @runloop
@runend
	%ge3 =w cslew 3, %r
	jnz %ge3, @emitrep, @emitlit
@emitrep
	%oplus2 =l add %o, 2
	%fit =w cslel %oplus2, %cap
	jnz %fit, @dorep, @nofit
@dorep
	%hdr =w sub 257, %r
	%po =l add %out, %o
	storeb %hdr, %po
	%o1 =l add %o, 1
	%po1 =l add %out, %o1
	storeb %b0, %po1
	%o2 =l add %o, 2
	%i2 =l add %i, %rl
	jmp @loop
@emitlit
	jmp @litloop
@litloop
	%L =w phi @emitlit 0, @litinc %Lnext
	%Ll =l extuw %L
	%j =l add %i, %Ll
	%cmjn =w csltl %j, %n
	jnz %cmjn, @chkL, @litend
@chkL
	%cmjl =w csltw %L, 128
	jnz %cmjl, @chkrun, @litend
@chkrun
	%pj =l add %src, %j
	%bj =w loadub %pj
	jmp @rjloop
@rjloop
	%rj =w phi @chkrun 1, @rjinc %rjnext
	%rjl =l extuw %rj
	%rjpos =l add %j, %rjl
	%cmrj =w csltl %rjpos, %n
	jnz %cmrj, @rjcap, @rjend
@rjcap
	%cmrc =w csltw %rj, 128
	jnz %cmrc, @rjeq, @rjend
@rjeq
	%prj =l add %src, %rjpos
	%brj =w loadub %prj
	%eqrj =w ceqw %brj, %bj
	jnz %eqrj, @rjinc, @rjend
@rjinc
	%rjnext =w add %rj, 1
	jmp @rjloop
@rjend
	%ge3rj =w cslew 3, %rj
	jnz %ge3rj, @litend, @litinc
@litinc
	%Lnext =w add %L, 1
	jmp @litloop
@litend
	%needw =w add %L, 1
	%need =l extuw %needw
	%oend =l add %o, %need
	%fitl =w cslel %oend, %cap
	jnz %fitl, @dolit, @nofit
@dolit
	%hdrl =w sub %L, 1
	%pol =l add %out, %o
	storeb %hdrl, %pol
	%o_ah =l add %o, 1
	jmp @copyloop
@copyloop
	%k =w phi @dolit 0, @copyinc %knext
	%cmpk =w csltw %k, %L
	jnz %cmpk, @docopy, @copydone
@docopy
	%kl =l extuw %k
	%spos =l add %i, %kl
	%ps =l add %src, %spos
	%byte =w loadub %ps
	%opos =l add %o_ah, %kl
	storeb %byte, %opos
	jmp @copyinc
@copyinc
	%knext =w add %k, 1
	jmp @copyloop
@copydone
	%i_lit =l add %i, %Ll
	%o_lit =l add %o, %need
	jmp @loop
@nofit
	ret -1
@done
	ret %o
}
