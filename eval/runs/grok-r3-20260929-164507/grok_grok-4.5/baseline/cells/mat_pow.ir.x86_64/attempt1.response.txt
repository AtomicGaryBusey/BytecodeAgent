export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %nn =l mul %n, %n
  %nn8 =l mul %nn, 8
  %tmp_prod =l add %tmp, %nn8
  jmp @id_i
@id_i
  %i =l phi @start 0, @id_i_end %i1
  %cmpi =w csltl %i, %n
  jnz %cmpi, @id_j_init, @id_done
@id_j_init
  jmp @id_j
@id_j
  %j =l phi @id_j_init 0, @id_j_end %j1
  %cmpj =w csltl %j, %n
  jnz %cmpj, @id_body, @id_i_end
@id_body
  %in =l mul %i, %n
  %idx =l add %in, %j
  %idx8 =l mul %idx, 8
  %ptr =l add %out, %idx8
  %eq =w ceql %i, %j
  %val =l extuw %eq
  storel %val, %ptr
  jmp @id_j_end
@id_j_end
  %j1 =l add %j, 1
  jmp @id_j
@id_i_end
  %i1 =l add %i, 1
  jmp @id_i
@id_done
  jmp @copy_i
@copy_i
  %ci =l phi @id_done 0, @copy_i_end %ci1
  %ccmpi =w csltl %ci, %n
  jnz %ccmpi, @copy_j_init, @copy_done
@copy_j_init
  jmp @copy_j
@copy_j
  %cj =l phi @copy_j_init 0, @copy_j_end %cj1
  %ccmpj =w csltl %cj, %n
  jnz %ccmpj, @copy_body, @copy_i_end
@copy_body
  %cin =l mul %ci, %n
  %cidx =l add %cin, %cj
  %cidx8 =l mul %cidx, 8
  %aptr =l add %a, %cidx8
  %cval =l loadl %aptr
  %tptr =l add %tmp, %cidx8
  storel %cval, %tptr
  jmp @copy_j_end
@copy_j_end
  %cj1 =l add %cj, 1
  jmp @copy_j
@copy_i_end
  %ci1 =l add %ci, 1
  jmp @copy_i
@copy_done
  jmp @pow_loop
@pow_loop
  %kk =l phi @copy_done %k, @pow_cont %kk_next
  %kgt0 =w csgtl %kk, 0
  jnz %kgt0, @pow_body, @pow_done
@pow_body
  %odd =l and %kk, 1
  %isodd =w cnel %odd, 0
  jnz %isodd, @do_mul_res, @after_mul_res
@do_mul_res
  jmp @mulres_i
@mulres_i
  %mi =l phi @do_mul_res 0, @mulres_i_end %mi1
  %mcmpi =w csltl %mi, %n
  jnz %mcmpi, @mulres_j_init, @mulres_done
@mulres_j_init
  jmp @mulres_j
@mulres_j
  %mj =l phi @mulres_j_init 0, @mulres_j_end %mj1
  %mcmpj =w csltl %mj, %n
  jnz %mcmpj, @mulres_t_init, @mulres_i_end
@mulres_t_init
  jmp @mulres_t
@mulres_t
  %mt =l phi @mulres_t_init 0, @mulres_body %mt1
  %sum =l phi @mulres_t_init 0, @mulres_body %sum1
  %mcmpt =w csltl %mt, %n
  jnz %mcmpt, @mulres_body, @mulres_store
@mulres_body
  %ain =l mul %mi, %n
  %aidx =l add %ain, %mt
  %aidx8 =l mul %aidx, 8
  %ap =l add %out, %aidx8
  %aval =l loadl %ap
  %bin =l mul %mt, %n
  %bidx =l add %bin, %mj
  %bidx8 =l mul %bidx, 8
  %bp =l add %tmp, %bidx8
  %bval =l loadl %bp
  %prod =l mul %aval, %bval
  %sum1 =l add %sum, %prod
  %mt1 =l add %mt, 1
  jmp @mulres_t
@mulres_store
  %cin =l mul %mi, %n
  %cidx =l add %cin, %mj
  %cidx8 =l mul %cidx, 8
  %cp =l add %tmp_prod, %cidx8
  storel %sum, %cp
  jmp @mulres_j_end
@mulres_j_end
  %mj1 =l add %mj, 1
  jmp @mulres_j
@mulres_i_end
  %mi1 =l add %mi, 1
  jmp @mulres_i
@mulres_done
  jmp @cpo_i
@cpo_i
  %pi =l phi @mulres_done 0, @cpo_i_end %pi1
  %pcmpi =w csltl %pi, %n
  jnz %pcmpi, @cpo_j_init, @after_mul_res
@cpo_j_init
  jmp @cpo_j
@cpo_j
  %pj =l phi @cpo_j_init 0, @cpo_j_end %pj1
  %pcmpj =w csltl %pj, %n
  jnz %pcmpj, @cpo_body, @cpo_i_end
@cpo_body
  %pin =l mul %pi, %n
  %pidx =l add %pin, %pj
  %pidx8 =l mul %pidx, 8
  %pp =l add %tmp_prod, %pidx8
  %pval =l loadl %pp
  %op =l add %out, %pidx8
  storel %pval, %op
  jmp @cpo_j_end
@cpo_j_end
  %pj1 =l add %pj, 1
  jmp @cpo_j
@cpo_i_end
  %pi1 =l add %pi, 1
  jmp @cpo_i
@after_mul_res
  jmp @mulbase_i
@mulbase_i
  %bi =l phi @after_mul_res 0, @mulbase_i_end %bi1
  %bcmpi =w csltl %bi, %n
  jnz %bcmpi, @mulbase_j_init, @mulbase_done
@mulbase_j_init
  jmp @mulbase_j
@mulbase_j
  %bj =l phi @mulbase_j_init 0, @mulbase_j_end %bj1
  %bcmpj =w csltl %bj, %n
  jnz %bcmpj, @mulbase_t_init, @mulbase_i_end
@mulbase_t_init
  jmp @mulbase_t
@mulbase_t
  %bt =l phi @mulbase_t_init 0, @mulbase_body %bt1
  %bsum =l phi @mulbase_t_init 0, @mulbase_body %bsum1
  %bcmpt =w csltl %bt, %n
  jnz %bcmpt, @mulbase_body, @mulbase_store
@mulbase_body
  %bain =l mul %bi, %n
  %baidx =l add %bain, %bt
  %baidx8 =l mul %baidx, 8
  %bap =l add %tmp, %baidx8
  %baval =l loadl %bap
  %bbin =l mul %bt, %n
  %bbidx =l add %bbin, %bj
  %bbidx8 =l mul %bbidx, 8
  %bbp =l add %tmp, %bbidx8
  %bbval =l loadl %bbp
  %bprod =l mul %baval, %bbval
  %bsum1 =l add %bsum, %bprod
  %bt1 =l add %bt, 1
  jmp @mulbase_t
@mulbase_store
  %bcin =l mul %bi, %n
  %bcidx =l add %bcin, %bj
  %bcidx8 =l mul %bcidx, 8
  %bcp =l add %tmp_prod, %bcidx8
  storel %bsum, %bcp
  jmp @mulbase_j_end
@mulbase_j_end
  %bj1 =l add %bj, 1
  jmp @mulbase_j
@mulbase_i_end
  %bi1 =l add %bi, 1
  jmp @mulbase_i
@mulbase_done
  jmp @cpb_i
@cpb_i
  %qi =l phi @mulbase_done 0, @cpb_i_end %qi1
  %qcmpi =w csltl %qi, %n
  jnz %qcmpi, @cpb_j_init, @pow_cont
@cpb_j_init
  jmp @cpb_j
@cpb_j
  %qj =l phi @cpb_j_init 0, @cpb_j_end %qj1
  %qcmpj =w csltl %qj, %n
  jnz %qcmpj, @cpb_body, @cpb_i_end
@cpb_body
  %qin =l mul %qi, %n
  %qidx =l add %qin, %qj
  %qidx8 =l mul %qidx, 8
  %qp =l add %tmp_prod, %qidx8
  %qval =l loadl %qp
  %tp =l add %tmp, %qidx8
  storel %qval, %tp
  jmp @cpb_j_end
@cpb_j_end
  %qj1 =l add %qj, 1
  jmp @cpb_j
@cpb_i_end
  %qi1 =l add %qi, 1
  jmp @cpb_i
@pow_cont
  %kk_next =l shr %kk, 1
  jmp @pow_loop
@pow_done
  %twonn =l mul %nn, 2
  jmp @zero_i
@zero_i
  %zi =l phi @pow_done 0, @zero_i_end %zi1
  %zcmp =w csltl %zi, %twonn
  jnz %zcmp, @zero_body, @finish
@zero_body
  %zidx8 =l mul %zi, 8
  %zp =l add %tmp, %zidx8
  storel 0, %zp
  jmp @zero_i_end
@zero_i_end
  %zi1 =l add %zi, 1
  jmp @zero_i
@finish
  ret 0
}
