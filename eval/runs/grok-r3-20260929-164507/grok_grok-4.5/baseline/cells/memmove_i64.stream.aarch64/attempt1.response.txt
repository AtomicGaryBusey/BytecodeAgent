memmove_i64:

cmp x0, x1
b.eq .Lret0
b.lo .Lforward
add x0, x0, x2, lsl #3
add x1, x1, x2, lsl #3
.Lbwd_loop:
ldr x3, [x1, #-8]!
str x3, [x0, #-8]!
subs x2, x2, #1
b.ne .Lbwd_loop
b .Lret0
.Lforward:
.Lfwd_loop:
ldr x3, [x1], #8
str x3, [x0], #8
subs x2, x2, #1
b.ne .Lfwd_loop
.Lret0:

ret
