packbits_decode:
push rbx
cld
mov r8, rdi
mov r9, rsi
mov r10, rdx
mov r11, rcx
mov rbx, rdx
.Lloop:
test r9, r9
jz .Ldone
movsx eax, byte ptr [r8]
add r8, 1
sub r9, 1
cmp eax, -128
je .Lloop
test eax, eax
jns .Lliteral
test r9, r9
jz .Lerr2
mov ecx, 1
sub ecx, eax
mov al, byte ptr [r8]
add r8, 1
sub r9, 1
cmp rcx, r11
ja .Lrep_ovf
mov rdx, rcx
mov rdi, r10
rep stosb
add r10, rdx
sub r11, rdx
jmp .Lloop
.Lrep_ovf:
mov rcx, r11
mov rdi, r10
rep stosb
jmp .Lerr1
.Lliteral:
mov ecx, eax
add ecx, 1
cmp r9, rcx
jb .Lerr2
cmp rcx, r11
ja .Llit_ovf
mov rdx, rcx
mov rsi, r8
mov rdi, r10
rep movsb
mov r8, rsi
mov r10, rdi
sub r9, rdx
sub r11, rdx
jmp .Lloop
.Llit_ovf:
mov rcx, r11
mov rsi, r8
mov rdi, r10
rep movsb
jmp .Lerr1
.Ldone:
mov rax, r10
sub rax, rbx
pop rbx
ret
.Lerr1:
mov rax, -1
pop rbx
ret
.Lerr2:
mov rax, -2
pop rbx
ret
