export function l $mulhi_u64(l %a, l %b) {
@start
	%a_lo =l and %a, 4294967295
	%a_hi =l shr %a, 32
	%b_lo =l and %b, 4294967295
	%b_hi =l shr %b, 32
	%p0 =l mul %a_lo, %b_lo
	%p1 =l mul %a_lo, %b_hi
	%p2 =l mul %a_hi, %b_lo
	%p3 =l mul %a_hi, %b_hi
	%p0_hi =l shr %p0, 32
	%p1_lo =l and %p1, 4294967295
	%p2_lo =l and %p2, 4294967295
	%cy1 =l add %p0_hi, %p1_lo
	%cy =l add %cy1, %p2_lo
	%cy_hi =l shr %cy, 32
	%p1_hi =l shr %p1, 32
	%p2_hi =l shr %p2, 32
	%res1 =l add %p3, %p1_hi
	%res2 =l add %res1, %p2_hi
	%res =l add %res2, %cy_hi
	ret %res
}
