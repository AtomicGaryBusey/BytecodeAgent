dot_i32_simd:
pxor xmm0, xmm0
mov rcx, rdx
cmp rcx, 4
jl .Lscalar_prep
.Lvector:
movdqu xmm1, xmmword ptr [rdi]
movdqu xmm2, xmmword ptr [rsi]
movdqa xmm3, xmm1
pmuldq xmm3, xmm2
pshufd xmm1, xmm1, 0x31
pshufd xmm2, xmm2, 0x31
pmuldq xmm1, xmm2
paddq xmm3, xmm1
paddq xmm0, xmm3
add rdi, 16
add rsi, 16
sub rcx, 4
cmp rcx, 4
jge .Lvector
.Lscalar_prep:
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
test rcx, rcx
jz .Ldone
.Lscalar_loop:
movsxd r8, dword ptr [rdi]
movsxd r9, dword ptr [rsi]
imul r8, r9
add rax, r8
add rdi, 4
add rsi, 4
dec rcx
jnz .Lscalar_loop
.Ldone:
ret
