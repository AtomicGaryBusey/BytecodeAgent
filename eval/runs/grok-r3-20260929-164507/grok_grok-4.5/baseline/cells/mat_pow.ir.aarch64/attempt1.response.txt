export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%i0 =l copy 0
	jmp @id_i
@id_i
	%i =l phi @start %i0, @id_i_incr %i1
	%cmp_i =w csltl %i, %n
	jnz %cmp_i, @id_j_start, @id_done
@id_j_start
	%j0 =l copy 0
	jmp @id_j
@id_j
	%j =l phi @id_j_start %j0, @id_set %j1
	%cmp_j =w csltl %j, %n
	jnz %cmp_j, @id_body, @id_i_incr
@id_body
	%cmp_eq =w ceql %i, %j
	jnz %cmp_eq, @set1, @set0
@set1
	%v1 =l copy 1
	jmp @id_set
@set0
	%v0 =l copy 0
	jmp @id_set
@id_set
	%val =l phi @set1 %v1, @set0 %v0
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, 8
	%ptr =l add %out, %off
	storel %val, %ptr
	%j1 =l add %j, 1
	jmp @id_j
@id_i_incr
	%i1 =l add %i, 1
	jmp @id_i
@id_done
	%is_zero =w ceql %k, 0
	jnz %is_zero, @clear, @pow_start
@pow_start
	%kk0 =l copy %k
	jmp @pow_loop
@pow_loop
	%kk =l phi @pow_start %kk0, @copy_done %kk1
	%cmp_kk =w csgtl %kk, 0
	jnz %cmp_kk, @do_mul, @clear
@do_mul
	%mi0 =l copy 0
	jmp @mul_i
@mul_i
	%mi =l phi @do_mul %mi0, @mul_i_incr %mi1
	%cmp_mi =w csltl %mi, %n
	jnz %cmp_mi, @mul_j_start, @mul_done
@mul_j_start
	%mj0 =l copy 0
	jmp @mul_j
@mul_j
	%mj =l phi @mul_j_start %mj0, @mul_store %mj1
	%cmp_mj =w csltl %mj, %n
	jnz %cmp_mj, @mul_t_start, @mul_i_incr
@mul_t_start
	%sum0 =l copy 0
	%mt0 =l copy 0
	jmp @mul_t
@mul_t
	%mt =l phi @mul_t_start %mt0, @mul_body %mt1
	%sum =l phi @mul_t_start %sum0, @mul_body %sum1
	%cmp_mt =w csltl %mt, %n
	jnz %cmp_mt, @mul_body, @mul_store
@mul_body
	%t1 =l mul %mi, %n
	%idx_x =l add %t1, %mt
	%off_x =l mul %idx_x, 8
	%ptr_x =l add %out, %off_x
	%x =l loadl %ptr_x
	%t2 =l mul %mt, %n
	%idx_y =l add %t2, %mj
	%off_y =l mul %idx_y, 8
	%ptr_y =l add %a, %off_y
	%y =l loadl %ptr_y
	%prod =l mul %x, %y
	%sum1 =l add %sum, %prod
	%mt1 =l add %mt, 1
	jmp @mul_t
@mul_store
	%t3 =l mul %mi, %n
	%idx_s =l add %t3, %mj
	%off_s =l mul %idx_s, 8
	%ptr_s =l add %tmp, %off_s
	storel %sum, %ptr_s
	%mj1 =l add %mj, 1
	jmp @mul_j
@mul_i_incr
	%mi1 =l add %mi, 1
	jmp @mul_i
@mul_done
	%ci0 =l copy 0
	jmp @copy_l
@copy_l
	%ci =l phi @mul_done %ci0, @copy_body %ci1
	%cmp_ci =w csltl %ci, %nn
	jnz %cmp_ci, @copy_body, @copy_done
@copy_body
	%off_c =l mul %ci, 8
	%ptr_t =l add %tmp, %off_c
	%val_c =l loadl %ptr_t
	%ptr_o =l add %out, %off_c
	storel %val_c, %ptr_o
	%ci1 =l add %ci, 1
	jmp @copy_l
@copy_done
	%kk1 =l sub %kk, 1
	jmp @pow_loop
@clear
	%tn0 =l copy 0
	%twonn =l mul %nn, 2
	jmp @zero_l
@zero_l
	%tn =l phi @clear %tn0, @zero_body %tn1
	%cmp_tn =w csltl %tn, %twonn
	jnz %cmp_tn, @zero_body, @zero_done
@zero_body
	%off_z =l mul %tn, 8
	%ptr_z =l add %tmp, %off_z
	storel 0, %ptr_z
	%tn1 =l add %tn, 1
	jmp @zero_l
@zero_done
	ret 0
}
