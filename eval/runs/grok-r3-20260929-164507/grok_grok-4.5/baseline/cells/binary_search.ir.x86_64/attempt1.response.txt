export function l $binary_search(l %a, l %n, l %key) {
@entry
	%low0 =l copy 0
	%high0 =l sub %n, 1
	jmp @loop
@loop
	%low =l phi @entry %low0, @go_high %low_h, @go_low %low_l
	%high =l phi @entry %high0, @go_high %high_h, @go_low %high_l
	%le =w cslel %low, %high
	jnz %le, @body, @fail
@body
	%sum =l add %low, %high
	%mid =l div %sum, 2
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @neq
@found
	ret %mid
@neq
	%lt =w csltl %val, %key
	jnz %lt, @go_high, @go_low
@go_high
	%low_h =l add %mid, 1
	%high_h =l copy %high
	jmp @loop
@go_low
	%low_l =l copy %low
	%high_l =l sub %mid, 1
	jmp @loop
@fail
	ret -1
}
