export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l copy 0
	jmp @outer_cond
@outer_cond
	%i =l phi @start %i0, @outer_inc %i1
	%nm1 =l sub %n, 1
	%oco =w csltl %i, %nm1
	jnz %oco, @outer_body, @end
@outer_body
	%j0 =l copy 0
	%nmi =l sub %n, %i
	%lim =l sub %nmi, 1
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body %j0, @inner_inc %j1
	%ico =w csltl %j, %lim
	jnz %ico, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%p =l add %a, %off
	%v =l loadl %p
	%jp1 =l add %j, 1
	%off1 =l mul %jp1, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%gt =w csgtl %v, %v1
	jnz %gt, @swap, @inner_inc
@swap
	storel %v1, %p
	storel %v, %p1
	jmp @inner_inc
@inner_inc
	%j1 =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i1 =l add %i, 1
	jmp @outer_cond
@end
	ret 0
}
