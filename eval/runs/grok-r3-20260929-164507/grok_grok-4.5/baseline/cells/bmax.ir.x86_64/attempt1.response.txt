export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @cont %i1
	%max =l phi @start %max0, @cont %max1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%val =l loadl %p
	%gt =w csgtl %val, %max
	jnz %gt, @upd, @nupd
@upd
	%mu =l copy %val
	jmp @cont
@nupd
	%mn =l copy %max
	jmp @cont
@cont
	%max1 =l phi @upd %mu, @nupd %mn
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
