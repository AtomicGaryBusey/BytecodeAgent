export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%min0 =l copy %y0
	%max0 =l copy %y0
	%sum0 =l copy %y0
	%t0 =l and %y0, 1
	%odd0 =w cnel %t0, 0
	jnz %odd0, @add1, @zero
@add1
	%cnt0 =l copy 1
	jmp @initdone
@zero
	%cnt0 =l copy 0
@initdone
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @initdone %i0, @next %inext
	%min =l phi @initdone %min0, @next %minn
	%max =l phi @initdone %max0, @next %maxn
	%sum =l phi @initdone %sum0, @next %sumn
	%cnt =l phi @initdone %cnt0, @next %cntn
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%idx =l shl %i, 3
	%ap =l add %a, %idx
	%ai =l loadl %ap
	%yi =l call %f(l %ai)
	%lt =w csltl %yi, %min
	jnz %lt, @umin, @kmin
@umin
	%minn =l copy %yi
	jmp @domax
@kmin
	%minn =l copy %min
@domax
	%gt =w csgtl %yi, %max
	jnz %gt, @umax, @kmax
@umax
	%maxn =l copy %yi
	jmp @dosum
@kmax
	%maxn =l copy %max
@dosum
	%sumn =l add %sum, %yi
	%ti =l and %yi, 1
	%oddi =w cnel %ti, 0
	jnz %oddi, @uodd, @kodd
@uodd
	%cntn =l add %cnt, 1
	jmp @next
@kodd
	%cntn =l copy %cnt
@next
	%inext =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
