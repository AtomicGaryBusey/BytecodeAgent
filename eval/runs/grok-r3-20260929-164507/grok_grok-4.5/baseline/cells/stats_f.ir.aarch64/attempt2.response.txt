export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%min0 =l copy %y0
	%max0 =l copy %y0
	%sum0 =l copy %y0
	%cnt0 =l and %y0, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @aftermax %inext
	%min =l phi @start %min0, @aftermax %minn
	%max =l phi @start %max0, @aftermax %maxn
	%sum =l phi @start %sum0, @aftermax %sumn
	%cnt =l phi @start %cnt0, @aftermax %cntn
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%idx =l shl %i, 3
	%ap =l add %a, %idx
	%ai =l loadl %ap
	%yi =l call %f(l %ai)
	%lt =w csltl %yi, %min
	jnz %lt, @umin, @kmin
@umin
	%min_u =l copy %yi
	jmp @aftermin
@kmin
	%min_k =l copy %min
	jmp @aftermin
@aftermin
	%minn =l phi @umin %min_u, @kmin %min_k
	%gt =w csgtl %yi, %max
	jnz %gt, @umax, @kmax
@umax
	%max_u =l copy %yi
	jmp @aftermax
@kmax
	%max_k =l copy %max
	jmp @aftermax
@aftermax
	%maxn =l phi @umax %max_u, @kmax %max_k
	%sumn =l add %sum, %yi
	%ti =l and %yi, 1
	%cntn =l add %cnt, %ti
	%inext =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
