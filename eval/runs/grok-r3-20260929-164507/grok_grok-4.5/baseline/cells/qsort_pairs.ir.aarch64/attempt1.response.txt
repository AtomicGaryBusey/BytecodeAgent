export function l $qsort_pairs(l %a, l %n) {
@start
  %le =w csltle %n, 1
  jnz %le, @done, @qs
@done
  ret 0
@qs
  %nm1 =l sub %n, 1
  %p2 =l mul %nm1, 2
  %po =l mul %p2, 8
  %ppk =l add %a, %po
  %pk =l loadl %ppk
  %ppv =l add %ppk, 8
  %pv =l loadl %ppv
  jmp @for_cond
@for_cond
  %i =l phi @qs 0, @for_inc %i_updated
  %j =l phi @qs 0, @for_inc %j_next
  %jlt =w csltl %j, %nm1
  jnz %jlt, @for_body, @part_end
@for_body
  %j2 =l mul %j, 2
  %jo =l mul %j2, 8
  %jpk =l add %a, %jo
  %jk =l loadl %jpk
  %jpv =l add %jpk, 8
  %jv =l loadl %jpv
  %klt =w csltl %jk, %pk
  jnz %klt, @do_swap, @check_eq
@check_eq
  %keq =w ceql %jk, %pk
  jnz %keq, @check_v, @no_swap
@check_v
  %vlt =w csltl %jv, %pv
  jnz %vlt, @do_swap, @no_swap
@do_swap
  %i2 =l mul %i, 2
  %io =l mul %i2, 8
  %ipk =l add %a, %io
  %ik =l loadl %ipk
  %ipv =l add %ipk, 8
  %iv =l loadl %ipv
  storel %jk, %ipk
  storel %jv, %ipv
  storel %ik, %jpk
  storel %iv, %jpv
  %i_from_swap =l add %i, 1
  jmp @for_inc
@no_swap
  %i_from_noswap =l copy %i
  jmp @for_inc
@for_inc
  %i_updated =l phi @do_swap %i_from_swap, @no_swap %i_from_noswap
  %j_next =l add %j, 1
  jmp @for_cond
@part_end
  %i2p =l mul %i, 2
  %iop =l mul %i2p, 8
  %ipkp =l add %a, %iop
  %ikp =l loadl %ipkp
  %ipvp =l add %ipkp, 8
  %ivp =l loadl %ipvp
  storel %pk, %ipkp
  storel %pv, %ipvp
  storel %ikp, %ppk
  storel %ivp, %ppv
  %rl =l call $qsort_pairs(l %a, l %i)
  %i1 =l add %i, 1
  %rn =l sub %n, %i1
  %ri2 =l mul %i1, 2
  %rio =l mul %ri2, 8
  %ra =l add %a, %rio
  %rr =l call $qsort_pairs(l %ra, l %rn)
  jmp @done
}
