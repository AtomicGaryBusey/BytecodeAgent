export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 0xffffffff
	%ahi =l shr %a, 32
	%blo =l and %b, 0xffffffff
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, 0xffffffff
	%p2lo =l and %p2, 0xffffffff
	%cy1 =l add %p0hi, %p1lo
	%cy =l add %cy1, %p2lo
	%cyhi =l shr %cy, 32
	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%hi1 =l add %p3, %p1hi
	%hi2 =l add %hi1, %p2hi
	%hi =l add %hi2, %cyhi
	ret %hi
}
