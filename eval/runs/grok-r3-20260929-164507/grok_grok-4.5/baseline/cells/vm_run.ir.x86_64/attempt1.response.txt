export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	jmp @loop
@loop
	%ge =w cgel %pc, %len
	jnz %ge, @halt, @fetch
@fetch
	%off =l mul %pc, 8
	%addr =l add %code, %off
	%op =l loadl %addr
	%ish =w ceql %op, 0
	jnz %ish, @halt, @c1
@c1
	%isp =w ceql %op, 1
	jnz %isp, @push, @c2
@push
	%p1 =l add %pc, 1
	%o1 =l mul %p1, 8
	%a1 =l add %code, %o1
	%v =l loadl %a1
	%s8 =l mul %sp, 8
	%sa =l add %stack, %s8
	storel %v, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop
@c2
	%isa =w ceql %op, 2
	jnz %isa, @add, @c3
@add
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%bp =l add %stack, %s8
	%b =l loadl %bp
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%ap =l add %stack, %s8
	%a =l loadl %ap
	%r =l add %a, %b
	storel %r, %ap
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c3
	%iss =w ceql %op, 3
	jnz %iss, @sub, @c4
@sub
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%bp =l add %stack, %s8
	%b =l loadl %bp
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%ap =l add %stack, %s8
	%a =l loadl %ap
	%r =l sub %a, %b
	storel %r, %ap
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c4
	%ism =w ceql %op, 4
	jnz %ism, @mul, @c5
@mul
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%bp =l add %stack, %s8
	%b =l loadl %bp
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%ap =l add %stack, %s8
	%a =l loadl %ap
	%r =l mul %a, %b
	storel %r, %ap
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c5
	%isd =w ceql %op, 5
	jnz %isd, @dup, @c6
@dup
	%tm =l sub %sp, 1
	%s8 =l mul %tm, 8
	%tp =l add %stack, %s8
	%t =l loadl %tp
	%s8 =l mul %sp, 8
	%sa =l add %stack, %s8
	storel %t, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c6
	%isw =w ceql %op, 6
	jnz %isw, @swap, @c7
@swap
	%i1 =l sub %sp, 1
	%o1 =l mul %i1, 8
	%p1 =l add %stack, %o1
	%v1 =l loadl %p1
	%i2 =l sub %sp, 2
	%o2 =l mul %i2, 8
	%p2 =l add %stack, %o2
	%v2 =l loadl %p2
	storel %v2, %p1
	storel %v1, %p2
	%pc =l add %pc, 1
	jmp @loop
@c7
	%isdr =w ceql %op, 7
	jnz %isdr, @drop, @c8
@drop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c8
	%iso =w ceql %op, 8
	jnz %iso, @over, @c9
@over
	%i2 =l sub %sp, 2
	%o2 =l mul %i2, 8
	%p2 =l add %stack, %o2
	%v2 =l loadl %p2
	%s8 =l mul %sp, 8
	%sa =l add %stack, %s8
	storel %v2, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c9
	%islt =w ceql %op, 9
	jnz %islt, @lt, @c10
@lt
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%bp =l add %stack, %s8
	%b =l loadl %bp
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%ap =l add %stack, %s8
	%a =l loadl %ap
	%cmp =w csltl %a, %b
	%r =l extuw %cmp
	storel %r, %ap
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@c10
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @c11
@jz
	%sp =l sub %sp, 1
	%s8 =l mul %sp, 8
	%vp =l add %stack, %s8
	%vv =l loadl %vp
	%p1 =l add %pc, 1
	%o1 =l mul %p1, 8
	%a1 =l add %code, %o1
	%t =l loadl %a1
	%z =w ceql %vv, 0
	jnz %z, @jz_take, @jz_skip
@jz_take
	%pc =l copy %t
	jmp @loop
@jz_skip
	%pc =l add %pc, 2
	jmp @loop
@c11
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jmp, @other
@jmp
	%p1 =l add %pc, 1
	%o1 =l mul %p1, 8
	%a1 =l add %code, %o1
	%t =l loadl %a1
	%pc =l copy %t
	jmp @loop
@other
	%pc =l add %pc, 1
	jmp @loop
@halt
	%i =l copy %sp
	jmp @clear
@clear
	%cd =w cgel %i, %cap
	jnz %cd, @done, @cl1
@cl1
	%o =l mul %i, 8
	%p =l add %stack, %o
	storel 0, %p
	%i =l add %i, 1
	jmp @clear
@done
	%em =w ceql %sp, 0
	jnz %em, @r0, @rt
@r0
	ret 0
@rt
	%ti =l sub %sp, 1
	%o =l mul %ti, 8
	%p =l add %stack, %o
	%tv =l loadl %p
	ret %tv
}
