export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 0xFFFFFFFF
	%i =l copy 0
	jmp @loop
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%j =w copy 0
	jmp @bitloop
@bitloop
	%jcmp =w csltw %j, 8
	jnz %jcmp, @bitbody, @bitdone
@bitbody
	%lsb =w and %crc, 1
	jnz %lsb, @odd, @even
@odd
	%crc =w shr %crc, 1
	%crc =w xor %crc, 0xEDB88320
	jmp @bitnext
@even
	%crc =w shr %crc, 1
	jmp @bitnext
@bitnext
	%j =w add %j, 1
	jmp @bitloop
@bitdone
	%i =l add %i, 1
	jmp @loop
@done
	%crc =w xor %crc, 0xFFFFFFFF
	%res =l extuw %crc
	ret %res
}
