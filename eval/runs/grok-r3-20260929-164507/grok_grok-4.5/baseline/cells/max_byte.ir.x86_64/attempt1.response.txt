export function l $max_byte(l %p, l %n) {
@start
	%nzero =w cnel %n, 0
	jnz %nzero, @loop, @ret0
@ret0
	ret 0
@loop
	%i =l phi @start 0, @merge %inext
	%max =l phi @start 0, @merge %maxnext
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%gt =w cultl %max, %b
	jnz %gt, @bigger, @notbigger
@bigger
	jmp @merge
@notbigger
	jmp @merge
@merge
	%maxnext =l phi @bigger %b, @notbigger %max
	%inext =l add %i, 1
	%more =w csltl %inext, %n
	jnz %more, @loop, @done
@done
	ret %maxnext
}
