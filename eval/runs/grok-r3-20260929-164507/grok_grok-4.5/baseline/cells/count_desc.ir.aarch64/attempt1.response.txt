export function l $count_desc(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret0, @init
@ret0
	ret 0
@init
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @cont %inext
	%cnt =l phi @init 0, @cont %cntnext
	%contcond =w csltl %i, %nm1
	jnz %contcond, @body, @done
@body
	%idx =l mul %i, 8
	%p =l add %a, %idx
	%val =l loadl %p
	%iplus =l add %i, 1
	%idx2 =l mul %iplus, 8
	%p2 =l add %a, %idx2
	%val2 =l loadl %p2
	%isgt =w csgtl %val, %val2
	jnz %isgt, @doinc, @noinc
@doinc
	%c1 =l add %cnt, 1
	jmp @mergec
@noinc
	jmp @mergec
@mergec
	%cntnext =l phi @doinc %c1, @noinc %cnt
	%inext =l add %i, 1
	jmp @cont
@cont
	jmp @loop
@done
	ret %cnt
}
