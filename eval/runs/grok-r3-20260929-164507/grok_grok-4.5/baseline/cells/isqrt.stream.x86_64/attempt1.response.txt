isqrt:
mov rax, 0
mov rdx, 1
shl rdx, 62
.Lsetbit:
cmp rdx, rdi
jbe .Lloop
shr rdx, 2
jmp .Lsetbit
.Lloop:
test rdx, rdx
jz .Lend
mov rcx, rax
add rcx, rdx
jc .Lnosub
cmp rdi, rcx
jb .Lnosub
sub rdi, rcx
shr rax, 1
add rax, rdx
jmp .Lshrbit
.Lnosub:
shr rax, 1
.Lshrbit:
shr rdx, 2
jmp .Lloop
.Lend:
ret
