export function l $reverse(l %a, l %n) {
@start
	%i0 =l copy 0
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%j =l phi @start %j0, @body %j1
	%cond =w csltl %i, %j
	jnz %cond, @body, @end
@body
	%offi =l mul %i, 8
	%pi =l add %a, %offi
	%vi =l loadl %pi
	%offj =l mul %j, 8
	%pj =l add %a, %offj
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@end
	ret 0
}
