collatz_max:
mov rax, 0
mov rcx, 1
.Louter:
cmp rcx, rdi
jg .Ldone
mov rdx, rcx
mov rsi, 0
.Linner:
cmp rdx, 1
je .Linner_done
test rdx, 1
jz .Leven
lea rdx, [rdx + rdx*2]
add rdx, 1
jmp .Linc
.Leven:
shr rdx, 1
.Linc:
inc rsi
jmp .Linner
.Linner_done:
cmp rsi, rax
jle .Lnext
mov rax, rsi
.Lnext:
inc rcx
jmp .Louter
.Ldone:
ret
