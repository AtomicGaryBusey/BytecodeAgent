export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %ipos =l copy 0
  %opos =l copy 0
  jmp @loop
@loop
  %cmp =w csltl %ipos, %n
  jnz %cmp, @haveheader, @done
@haveheader
  %hp =l add %src, %ipos
  %h =w loadsb %hp
  %ipos =l add %ipos, 1
  %isnop =w ceqw %h, -128
  jnz %isnop, @loop, @notnop
@notnop
  %islit =w csgew %h, 0
  jnz %islit, @literal, @repeat
@literal
  %countw =w add %h, 1
  %need =l extsw %countw
  %remain =l sub %n, %ipos
  %enough =w cslel %need, %remain
  jnz %enough, @litok, @err2
@litok
  %oremain =l sub %cap, %opos
  %oenough =w cslel %need, %oremain
  jnz %oenough, @litcopyfull, @litpartial
@litcopyfull
  %i =l copy 0
  jmp @litcopyloop
@litcopyloop
  %icmp =w csltl %i, %need
  jnz %icmp, @litcopyone, @litcopydone
@litcopyone
  %sp =l add %src, %ipos
  %spi =l add %sp, %i
  %b =w loadub %spi
  %op =l add %out, %opos
  %opi =l add %op, %i
  storeb %b, %opi
  %i =l add %i, 1
  jmp @litcopyloop
@litcopydone
  %ipos =l add %ipos, %need
  %opos =l add %opos, %need
  jmp @loop
@litpartial
  %ocount =l copy %oremain
  %i =l copy 0
  jmp @litpartloop
@litpartloop
  %icmp =w csltl %i, %ocount
  jnz %icmp, @litpartone, @litpartdone
@litpartone
  %sp =l add %src, %ipos
  %spi =l add %sp, %i
  %b =w loadub %spi
  %op =l add %out, %opos
  %opi =l add %op, %i
  storeb %b, %opi
  %i =l add %i, 1
  jmp @litpartloop
@litpartdone
  %rm1 =l copy -1
  ret %rm1
@repeat
  %countw =w sub 1, %h
  %remain =l sub %n, %ipos
  %enough =w cslel 1, %remain
  jnz %enough, @repok, @err2
@repok
  %bp =l add %src, %ipos
  %b =w loadub %bp
  %ipos =l add %ipos, 1
  %need =l extsw %countw
  %oremain =l sub %cap, %opos
  %oenough =w cslel %need, %oremain
  jnz %oenough, @repfull, @reppartial
@repfull
  %i =l copy 0
  jmp @reploop
@reploop
  %icmp =w csltl %i, %need
  jnz %icmp, @repone, @repdone
@repone
  %op =l add %out, %opos
  %opi =l add %op, %i
  storeb %b, %opi
  %i =l add %i, 1
  jmp @reploop
@repdone
  %opos =l add %opos, %need
  jmp @loop
@reppartial
  %ocount =l copy %oremain
  %i =l copy 0
  jmp @reppartloop
@reppartloop
  %icmp =w csltl %i, %ocount
  jnz %icmp, @reppartone, @reppartdone
@reppartone
  %op =l add %out, %opos
  %opi =l add %op, %i
  storeb %b, %opi
  %i =l add %i, 1
  jmp @reppartloop
@reppartdone
  %rm1 =l copy -1
  ret %rm1
@err2
  %rm2 =l copy -2
  ret %rm2
@done
  ret %opos
}
