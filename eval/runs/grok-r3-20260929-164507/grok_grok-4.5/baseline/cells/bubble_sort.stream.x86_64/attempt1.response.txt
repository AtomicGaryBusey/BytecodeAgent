bubble_sort:
cmp rsi, 1
jle .Ldone
mov r8, rsi
.Louter:
dec r8
xor rcx, rcx
.Linner:
mov rax, qword ptr [rdi + rcx*8]
mov rdx, qword ptr [rdi + rcx*8 + 8]
cmp rax, rdx
jle .Lnoswap
mov qword ptr [rdi + rcx*8], rdx
mov qword ptr [rdi + rcx*8 + 8], rax
.Lnoswap:
inc rcx
cmp rcx, r8
jl .Linner
cmp r8, 1
jg .Louter
.Ldone:
xor rax, rax
ret
