export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%i0 =l copy 1
	jmp @loop_i
@loop_i
	%max =l phi @start %max0, @next %max1
	%i =l phi @start %i0, @next %i1
	%cmpi =w csgtl %i, %n
	jnz %cmpi, @done, @body_i
@body_i
	%x0 =l copy %i
	%steps0 =l copy 0
	jmp @loop_c
@loop_c
	%x =l phi @body_i %x0, @do_inc %x1
	%steps =l phi @body_i %steps0, @do_inc %steps1
	%isone =w ceql %x, 1
	jnz %isone, @end_c, @step
@step
	%and =l and %x, 1
	%isodd =w ceql %and, 1
	jnz %isodd, @odd, @even
@even
	%x_even =l shr %x, 1
	jmp @do_inc
@odd
	%t =l mul %x, 3
	%x_odd =l add %t, 1
	jmp @do_inc
@do_inc
	%x1 =l phi @even %x_even, @odd %x_odd
	%steps1 =l add %steps, 1
	jmp @loop_c
@end_c
	%cmpm =w csgtl %steps, %max
	jnz %cmpm, @upd, @noup
@upd
	%maxu =l copy %steps
	jmp @next
@noup
	%maxn =l copy %max
	jmp @next
@next
	%max1 =l phi @upd %maxu, @noup %maxn
	%i1 =l add %i, 1
	jmp @loop_i
@done
	ret %max
}
