export function l $popcount(l %x) {
@start
	%a =l shr %x, 1
	%b =l and %a, 0x5555555555555555
	%c =l sub %x, %b
	%d =l and %c, 0x3333333333333333
	%e =l shr %c, 2
	%f =l and %e, 0x3333333333333333
	%g =l add %d, %f
	%h =l shr %g, 4
	%i =l add %g, %h
	%j =l and %i, 0x0f0f0f0f0f0f0f0f
	%k =l shr %j, 8
	%l =l add %j, %k
	%m =l shr %l, 16
	%n =l add %l, %m
	%o =l shr %n, 32
	%p =l add %n, %o
	%q =l and %p, 255
	ret %q
}
