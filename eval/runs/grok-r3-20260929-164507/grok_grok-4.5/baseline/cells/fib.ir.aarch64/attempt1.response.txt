export function l $fib(l %n) {
@start
	%a0 =l copy 0
	%b0 =l copy 1
	%i0 =l copy 0
	jmp @loop
@loop
	%a =l phi @start %a0, @body %anew
	%b =l phi @start %b0, @body %bnew
	%i =l phi @start %i0, @body %inew
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%tmp =l add %a, %b
	%anew =l copy %b
	%bnew =l copy %tmp
	%inew =l add %i, 1
	jmp @loop
@end
	ret %a
}
