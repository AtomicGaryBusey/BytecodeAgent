export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i =l copy 0
	%odd =l copy 0
	%sum =l copy 0
	%min =l copy 9223372036854775807
	%max =l copy -9223372036854775808
	jmp @loop
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%aptr =l add %a, %off
	%val =l loadl %aptr
	%y =l call %f(l %val)
	%sum =l add %sum, %y
	%cmpmin =w csltl %y, %min
	jnz %cmpmin, @upmin, @aftermin
@upmin
	%min =l copy %y
	jmp @aftermin
@aftermin
	%cmpmax =w csgtl %y, %max
	jnz %cmpmax, @upmax, @aftermax
@upmax
	%max =l copy %y
	jmp @aftermax
@aftermax
	%and =l and %y, 1
	%isodd =w cnel %and, 0
	jnz %isodd, @incodd, @afterodd
@incodd
	%odd =l add %odd, 1
	jmp @afterodd
@afterodd
	%i =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
