lcs_len:
push rbx
push r12
mov r9, rdi
mov r10, rsi
mov r11, rdx
mov rbx, rcx
mov rdi, r8
xor eax, eax
lea rcx, [rbx + 1]
rep stosq
xor esi, esi
test r10, r10
jz .Ldone
.Louter:
movzx edx, byte ptr [r9 + rsi]
xor eax, eax
mov ecx, 1
test rbx, rbx
jz .Linnerdone
.Linner:
mov r12, qword ptr [r8 + rcx*8]
cmp dl, byte ptr [r11 + rcx - 1]
jne .Lnomatch
inc rax
mov qword ptr [r8 + rcx*8], rax
jmp .Lnext
.Lnomatch:
mov rdi, qword ptr [r8 + rcx*8 - 8]
cmp r12, rdi
cmovge rdi, r12
mov qword ptr [r8 + rcx*8], rdi
.Lnext:
mov rax, r12
inc rcx
cmp rcx, rbx
jle .Linner
.Linnerdone:
inc rsi
cmp rsi, r10
jl .Louter
.Ldone:
mov rax, qword ptr [r8 + rbx*8]
pop r12
pop rbx
ret
