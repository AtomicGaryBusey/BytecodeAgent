export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @cond
@cond
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%idx =l mul %i, 8
	%sp =l add %src, %idx
	%val =l loadl %sp
	%res =l call %f(l %val)
	%dp =l add %dst, %idx
	storel %res, %dp
	%less =w csltl %res, %val
	%lessl =l extsw %less
	%cnt1 =l add %cnt, %lessl
	%i1 =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
