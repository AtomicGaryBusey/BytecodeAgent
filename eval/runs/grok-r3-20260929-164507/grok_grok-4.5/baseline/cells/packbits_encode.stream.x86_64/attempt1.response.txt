packbits_encode:
	push rbx
	push rbp
	mov r11, rcx
	xor r8, r8
	xor r9, r9
.Lmain_loop:
	cmp r8, rsi
	jge .Ldone
	movzx eax, byte ptr [rdi + r8]
	mov ecx, 1
.Lrun_loop:
	lea r10, [r8 + rcx]
	cmp r10, rsi
	jge .Lrun_done
	cmp ecx, 128
	jge .Lrun_done
	cmp al, byte ptr [rdi + r10]
	jne .Lrun_done
	inc ecx
	jmp .Lrun_loop
.Lrun_done:
	cmp ecx, 3
	jl .Lliteral
	lea r10, [r9 + 2]
	cmp r10, r11
	jg .Lfail
	mov eax, 257
	sub eax, ecx
	mov byte ptr [rdx + r9], al
	mov al, byte ptr [rdi + r8]
	mov byte ptr [rdx + r9 + 1], al
	add r9, 2
	add r8, rcx
	jmp .Lmain_loop
.Lliteral:
	mov rbp, r8
.Llit_find:
	cmp rbp, rsi
	jge .Llit_emit
	mov rax, rbp
	sub rax, r8
	cmp rax, 128
	jge .Llit_emit
	movzx eax, byte ptr [rdi + rbp]
	mov ecx, 1
.Lrunj_loop:
	cmp ecx, 3
	jge .Llit_emit
	lea r10, [rbp + rcx]
	cmp r10, rsi
	jge .Lrunj_lt3
	cmp al, byte ptr [rdi + r10]
	jne .Lrunj_lt3
	inc ecx
	jmp .Lrunj_loop
.Lrunj_lt3:
	inc rbp
	jmp .Llit_find
.Llit_emit:
	mov rcx, rbp
	sub rcx, r8
	lea rax, [r9 + rcx + 1]
	cmp rax, r11
	jg .Lfail
	lea eax, [ecx - 1]
	mov byte ptr [rdx + r9], al
	xor ebx, ebx
.Lcopy_loop:
	cmp rbx, rcx
	jge .Lcopy_done
	lea r10, [r8 + rbx]
	mov al, byte ptr [rdi + r10]
	lea r10, [r9 + rbx + 1]
	mov byte ptr [rdx + r10], al
	inc rbx
	jmp .Lcopy_loop
.Lcopy_done:
	add r9, rcx
	inc r9
	mov r8, rbp
	jmp .Lmain_loop
.Ldone:
	mov rax, r9
	pop rbp
	pop rbx
	ret
.Lfail:
	mov rax, -1
	pop rbp
	pop rbx
	ret
