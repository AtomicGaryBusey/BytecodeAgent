export function l $count_desc(l %a, l %n) {
@start
	%nlt2 =w csltl %n, 2
	jnz %nlt2, @ret0, @init
@ret0
	ret 0
@init
	%nm1 =l sub %n, 1
	jmp @header
@header
	%i =l phi @init 0, @body %i1
	%cnt =l phi @init 0, @body %cnt1
	%cont =w csltl %i, %nm1
	jnz %cont, @body, @done
@body
	%i1 =l add %i, 1
	%idx8 =l mul %i, 8
	%ptr =l add %a, %idx8
	%v1 =l loadl %ptr
	%idx8b =l mul %i1, 8
	%ptr2 =l add %a, %idx8b
	%v2 =l loadl %ptr2
	%isgt =w csgtl %v1, %v2
	%addend =l extsw %isgt
	%cnt1 =l add %cnt, %addend
	jmp @header
@done
	ret %cnt
}
