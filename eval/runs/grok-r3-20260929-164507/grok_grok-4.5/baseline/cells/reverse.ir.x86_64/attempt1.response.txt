export function l $reverse(l %a, l %n) {
@start
	%cmp =w csltl %n, 2
	jnz %cmp, @done, @loop_init
@loop_init
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @loop_init 0, @body %inext
	%j =l sub %nm1, %i
	%cmp2 =w csltl %i, %j
	jnz %cmp2, @body, @done
@body
	%oi =l mul %i, 8
	%ai =l add %a, %oi
	%vi =l loadl %ai
	%oj =l mul %j, 8
	%aj =l add %a, %oj
	%vj =l loadl %aj
	storel %vj, %ai
	storel %vi, %aj
	%inext =l add %i, 1
	jmp @loop
@done
	ret 0
}
