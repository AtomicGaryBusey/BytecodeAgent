export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc0 =l loadl %pcp
	%atend =w ceql %pc0, %len
	jnz %atend, @halt, @fetch
@fetch
	%pc08 =l mul %pc0, 8
	%opaddr =l add %code, %pc08
	%op =l loadl %opaddr
	%is0 =w ceql %op, 0
	jnz %is0, @halt, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @c2
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @c3
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @c4
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @c5
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @c6
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @c7
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @c8
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @c9
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @c10
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @c11
@c11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_other
@op_push
	%pcp1 =l loadl %pcp
	%pcp1n =l add %pcp1, 1
	%pcp1n8 =l mul %pcp1n, 8
	%vaddr =l add %code, %pcp1n8
	%v =l loadl %vaddr
	%spp1 =l loadl %spp
	%spp18 =l mul %spp1, 8
	%saddr1 =l add %stack, %spp18
	storel %v, %saddr1
	%spp1n =l add %spp1, 1
	storel %spp1n, %spp
	%pcp1n2 =l add %pcp1, 2
	storel %pcp1n2, %pcp
	jmp @loop
@op_add
	%spa =l loadl %spp
	%spa1 =l sub %spa, 1
	%spa18 =l mul %spa1, 8
	%baddra =l add %stack, %spa18
	%ba =l loadl %baddra
	%spa2 =l sub %spa, 2
	%spa28 =l mul %spa2, 8
	%aaddra =l add %stack, %spa28
	%aa =l loadl %aaddra
	%suma =l add %aa, %ba
	storel %suma, %aaddra
	%span =l sub %spa, 1
	storel %span, %spp
	%pca =l loadl %pcp
	%pcan =l add %pca, 1
	storel %pcan, %pcp
	jmp @loop
@op_sub
	%sps =l loadl %spp
	%sps1 =l sub %sps, 1
	%sps18 =l mul %sps1, 8
	%baddrs =l add %stack, %sps18
	%bs =l loadl %baddrs
	%sps2 =l sub %sps, 2
	%sps28 =l mul %sps2, 8
	%aaddrs =l add %stack, %sps28
	%as =l loadl %aaddrs
	%diffs =l sub %as, %bs
	storel %diffs, %aaddrs
	%spsn =l sub %sps, 1
	storel %spsn, %spp
	%pcs =l loadl %pcp
	%pcsn =l add %pcs, 1
	storel %pcsn, %pcp
	jmp @loop
@op_mul
	%spm =l loadl %spp
	%spm1 =l sub %spm, 1
	%spm18 =l mul %spm1, 8
	%baddrm =l add %stack, %spm18
	%bm =l loadl %baddrm
	%spm2 =l sub %spm, 2
	%spm28 =l mul %spm2, 8
	%aaddrm =l add %stack, %spm28
	%am =l loadl %aaddrm
	%prod =l mul %am, %bm
	storel %prod, %aaddrm
	%spmn =l sub %spm, 1
	storel %spmn, %spp
	%pcm =l loadl %pcp
	%pcmn =l add %pcm, 1
	storel %pcmn, %pcp
	jmp @loop
@op_dup
	%spd =l loadl %spp
	%spd1 =l sub %spd, 1
	%spd18 =l mul %spd1, 8
	%taddrd =l add %stack, %spd18
	%topd =l loadl %taddrd
	%spd8 =l mul %spd, 8
	%saddrd =l add %stack, %spd8
	storel %topd, %saddrd
	%spdn =l add %spd, 1
	storel %spdn, %spp
	%pcd =l loadl %pcp
	%pcdn =l add %pcd, 1
	storel %pcdn, %pcp
	jmp @loop
@op_swap
	%spw =l loadl %spp
	%spw1 =l sub %spw, 1
	%spw18 =l mul %spw1, 8
	%baddrw =l add %stack, %spw18
	%bw =l loadl %baddrw
	%spw2 =l sub %spw, 2
	%spw28 =l mul %spw2, 8
	%aaddrw =l add %stack, %spw28
	%aw =l loadl %aaddrw
	storel %bw, %aaddrw
	storel %aw, %baddrw
	%pcw =l loadl %pcp
	%pcwn =l add %pcw, 1
	storel %pcwn, %pcp
	jmp @loop
@op_drop
	%spr =l loadl %spp
	%sprn =l sub %spr, 1
	storel %sprn, %spp
	%pcr =l loadl %pcp
	%pcrn =l add %pcr, 1
	storel %pcrn, %pcp
	jmp @loop
@op_over
	%spo =l loadl %spp
	%spo2 =l sub %spo, 2
	%spo28 =l mul %spo2, 8
	%aaddro =l add %stack, %spo28
	%ao =l loadl %aaddro
	%spo8 =l mul %spo, 8
	%saddro =l add %stack, %spo8
	storel %ao, %saddro
	%spon =l add %spo, 1
	storel %spon, %spp
	%pco =l loadl %pcp
	%pcon =l add %pco, 1
	storel %pcon, %pcp
	jmp @loop
@op_lt
	%spl =l loadl %spp
	%spl1 =l sub %spl, 1
	%spl18 =l mul %spl1, 8
	%baddrl =l add %stack, %spl18
	%bl =l loadl %baddrl
	%spl2 =l sub %spl, 2
	%spl28 =l mul %spl2, 8
	%aaddrl =l add %stack, %spl28
	%al =l loadl %aaddrl
	%cmpl =w csltl %al, %bl
	%resl =l extuw %cmpl
	storel %resl, %aaddrl
	%spln =l sub %spl, 1
	storel %spln, %spp
	%pcl =l loadl %pcp
	%pcln =l add %pcl, 1
	storel %pcln, %pcp
	jmp @loop
@op_jz
	%spj =l loadl %spp
	%spjn =l sub %spj, 1
	storel %spjn, %spp
	%spj8 =l mul %spjn, 8
	%vaddrj =l add %stack, %spj8
	%vj =l loadl %vaddrj
	%pcj =l loadl %pcp
	%pcj1 =l add %pcj, 1
	%pcj18 =l mul %pcj1, 8
	%taddrj =l add %code, %pcj18
	%tj =l loadl %taddrj
	%iszj =w ceql %vj, 0
	jnz %iszj, @jz_yes, @jz_no
@jz_yes
	storel %tj, %pcp
	jmp @loop
@jz_no
	%pcj2 =l add %pcj, 2
	storel %pcj2, %pcp
	jmp @loop
@op_jmp
	%pcg =l loadl %pcp
	%pcg1 =l add %pcg, 1
	%pcg18 =l mul %pcg1, 8
	%taddrg =l add %code, %pcg18
	%tg =l loadl %taddrg
	storel %tg, %pcp
	jmp @loop
@op_other
	%pcx =l loadl %pcp
	%pcxn =l add %pcx, 1
	storel %pcxn, %pcp
	jmp @loop
@halt
	%i0 =l loadl %spp
	jmp @zeroloop
@zeroloop
	%i =l phi @halt %i0, @zerobody %inext
	%idone =w csgel %i, %cap
	jnz %idone, @ret, @zerobody
@zerobody
	%i8 =l mul %i, 8
	%zaddr =l add %stack, %i8
	storel 0, %zaddr
	%inext =l add %i, 1
	jmp @zeroloop
@ret
	%spf =l loadl %spp
	%isempty =w ceql %spf, 0
	jnz %isempty, @ret0, @rettop
@ret0
	ret 0
@rettop
	%spf1 =l sub %spf, 1
	%spf18 =l mul %spf1, 8
	%raddr =l add %stack, %spf18
	%rval =l loadl %raddr
	ret %rval
}
