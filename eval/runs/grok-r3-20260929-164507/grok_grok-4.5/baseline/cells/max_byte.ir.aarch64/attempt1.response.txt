export function l $max_byte(l %p, l %n) {
@start
	jmp @check
@check
	%i =l phi @start 0, @next %i1
	%max =l phi @start 0, @next %max1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @update, @next
@update
	jmp @next
@next
	%max1 =l phi @body %max, @update %b
	%i1 =l add %i, 1
	jmp @check
@done
	ret %max
}
