export function l $isqrt(l %n) {
@start
	%lo_init =l copy 0
	%hi_init =l shl 1, 32
	jmp @loop
@loop
	%lo =l phi @start %lo_init, @toobig %lo_keep, @ok %lo_new
	%hi =l phi @start %hi_init, @toobig %hi_new, @ok %hi_keep
	%diff =l sub %hi, %lo
	%cmp =w cultl %diff, 2
	jnz %cmp, @done, @body
@body
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%cmp2 =w cugtl %mid, %quot
	jnz %cmp2, @toobig, @ok
@toobig
	%lo_keep =l copy %lo
	%hi_new =l copy %mid
	jmp @loop
@ok
	%lo_new =l copy %mid
	%hi_keep =l copy %hi
	jmp @loop
@done
	ret %lo
}
