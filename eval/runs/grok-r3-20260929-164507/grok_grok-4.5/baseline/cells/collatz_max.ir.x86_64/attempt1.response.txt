export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%i =l copy 1
	jmp @loop_i
@loop_i
	%cmp =w csgtl %i, %n
	jnz %cmp, @done, @body_i
@body_i
	%x =l copy %i
	%steps =l copy 0
	jmp @loop_steps
@loop_steps
	%isone =w ceql %x, 1
	jnz %isone, @end_steps, @body_steps
@body_steps
	%bit =l and %x, 1
	%iseven =w ceql %bit, 0
	jnz %iseven, @even, @odd
@even
	%x =l div %x, 2
	jmp @incr
@odd
	%tx =l mul %x, 3
	%x =l add %tx, 1
	jmp @incr
@incr
	%steps =l add %steps, 1
	jmp @loop_steps
@end_steps
	%cmpmax =w csgtl %steps, %max
	jnz %cmpmax, @update, @next_i
@update
	%max =l copy %steps
	jmp @next_i
@next_i
	%i =l add %i, 1
	jmp @loop_i
@done
	ret %max
}
