export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @ret0, @cmp
@cmp
	%dstlt =w cultl %dst, %src
	jnz %dstlt, @forward, @ge
@ge
	%dsteq =w ceql %dst, %src
	jnz %dsteq, @ret0, @backward
@forward
	jmp @fwd_loop
@fwd_loop
	%i =l phi @forward 0, @fwd_body %inext
	%fdone =w csgel %i, %n
	jnz %fdone, @ret0, @fwd_body
@fwd_body
	%offf =l shl %i, 3
	%psf =l add %src, %offf
	%pdf =l add %dst, %offf
	%vf =l loadl %psf
	storel %vf, %pdf
	%inext =l add %i, 1
	jmp @fwd_loop
@backward
	jmp @bwd_loop
@bwd_loop
	%j =l phi @backward %n, @bwd_body %jnext
	%bdone =w ceql %j, 0
	jnz %bdone, @ret0, @bwd_body
@bwd_body
	%jnext =l sub %j, 1
	%offb =l shl %jnext, 3
	%psb =l add %src, %offb
	%pdb =l add %dst, %offb
	%vb =l loadl %psb
	storel %vb, %pdb
	jmp @bwd_loop
@ret0
	ret 0
}
