export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%si =l copy 0
	%oi =l copy 0
	jmp @loop
@loop
	%cmploop =w csltl %si, %n
	jnz %cmploop, @body, @done
@body
	%sptr =l add %src, %si
	%h =w loadsb %sptr
	%si =l add %si, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @checksign
@checksign
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%countw =w add %h, 1
	%count =l extsw %countw
	%sum =l add %si, %count
	%srcover =w csgtl %sum, %n
	jnz %srcover, @err2, @litinit
@litinit
	%i =l copy 0
	jmp @litloop
@litloop
	%cmpi =w csltl %i, %count
	jnz %cmpi, @litbody, @loop
@litbody
	%outover =w csgel %oi, %cap
	jnz %outover, @err1, @litdo
@litdo
	%sptr2 =l add %src, %si
	%byte =w loadub %sptr2
	%optr =l add %out, %oi
	storeb %byte, %optr
	%si =l add %si, 1
	%oi =l add %oi, 1
	%i =l add %i, 1
	jmp @litloop
@repeat
	%countw2 =w sub 1, %h
	%count2 =l extsw %countw2
	%srcover2 =w csgel %si, %n
	jnz %srcover2, @err2, @repget
@repget
	%sptr3 =l add %src, %si
	%b =w loadub %sptr3
	%si =l add %si, 1
	%j =l copy 0
	jmp @reploop
@reploop
	%cmpj =w csltl %j, %count2
	jnz %cmpj, @repbody, @loop
@repbody
	%outover2 =w csgel %oi, %cap
	jnz %outover2, @err1, @repdo
@repdo
	%optr2 =l add %out, %oi
	storeb %b, %optr2
	%oi =l add %oi, 1
	%j =l add %j, 1
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	ret %oi
}
