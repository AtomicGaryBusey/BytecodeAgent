export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @header
@header
	%i =l phi @start 0, @body %i1
	%c =l phi @start 0, @body %c1
	%more =w csltl %i, %n
	jnz %more, @body, @exit
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%ai =l loadl %ap
	%isgt =w cugtl %ai, %t
	%inc =l extuw %isgt
	%c1 =l add %c, %inc
	%i1 =l add %i, 1
	jmp @header
@exit
	ret %c
}
