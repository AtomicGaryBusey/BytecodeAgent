export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%jinit =l copy 0
	jmp @init_cond
@init_cond
	%j =l phi @start %jinit, @init_body %jnext
	%ok =w cslel %j, %n
	jnz %ok, @init_body, @init_end
@init_body
	%off =l mul %j, 8
	%p =l add %row, %off
	storel 0, %p
	%jnext =l add %j, 1
	jmp @init_cond
@init_end
	%iinit =l copy 0
	jmp @i_cond
@i_cond
	%i =l phi @init_end %iinit, @i_inc %inext
	%iok =w csltl %i, %m
	jnz %iok, @i_body, @i_end
@i_body
	%diaginit =l copy 0
	%jinit2 =l copy 1
	jmp @j_cond
@j_cond
	%j2 =l phi @i_body %jinit2, @j_inc %jnext2
	%diag =l phi @i_body %diaginit, @j_inc %diagnext
	%jok =w cslel %j2, %n
	jnz %jok, @j_body, @j_end
@j_body
	%joff =l mul %j2, 8
	%pj =l add %row, %joff
	%up =l loadl %pj
	%px =l add %x, %i
	%xi =w loadub %px
	%jm1 =l sub %j2, 1
	%py =l add %y, %jm1
	%yj =w loadub %py
	%eq =w ceqw %xi, %yj
	jnz %eq, @eq_yes, @eq_no
@eq_yes
	%new1 =l add %diag, 1
	jmp @set
@eq_no
	%jm1off =l mul %jm1, 8
	%pl =l add %row, %jm1off
	%left =l loadl %pl
	%gt =w csgtl %up, %left
	jnz %gt, @max_up, @max_left
@max_up
	%new2 =l copy %up
	jmp @max_end
@max_left
	%new3 =l copy %left
	jmp @max_end
@max_end
	%new4 =l phi @max_up %new2, @max_left %new3
	jmp @set
@set
	%new =l phi @eq_yes %new1, @max_end %new4
	storel %new, %pj
	jmp @j_inc
@j_inc
	%diagnext =l copy %up
	%jnext2 =l add %j2, 1
	jmp @j_cond
@j_end
	jmp @i_inc
@i_inc
	%inext =l add %i, 1
	jmp @i_cond
@i_end
	%noff =l mul %n, 8
	%pn =l add %row, %noff
	%res =l loadl %pn
	ret %res
}
