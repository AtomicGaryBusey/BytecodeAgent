export function l $binary_search(l %a, l %n, l %key) {
@start
	%isempty =w ceql %n, 0
	jnz %isempty, @empty, @init
@empty
	ret -1
@init
	%lo =l copy 0
	%hi =l sub %n, 1
	jmp @loop
@loop
	%lo1 =l phi @init %lo, @update %lonew
	%hi1 =l phi @init %hi, @update %hinew
	%ok =w cslel %lo1, %hi1
	jnz %ok, @body, @miss
@body
	%diff =l sub %hi1, %lo1
	%half =l shr %diff, 1
	%mid =l add %lo1, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @hit, @cmp
@hit
	ret %mid
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @raise, @lower
@raise
	%lonew =l add %mid, 1
	%hinew =l copy %hi1
	jmp @update
@lower
	%lonew =l copy %lo1
	%hinew =l sub %mid, 1
	jmp @update
@update
	jmp @loop
@miss
	ret -1
}
