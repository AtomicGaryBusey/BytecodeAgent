export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
	jmp @loop
@loop
	%crc =w phi @start %crc0, @bitdone %crc2
	%i =l phi @start %i0, @bitdone %i1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc, %b
	%j0 =w copy 0
	jmp @bitloop
@bitloop
	%crc2 =w phi @body %crc1, @bitnext %crc7
	%j =w phi @body %j0, @bitnext %j1
	%jcmp =w csltw %j, 8
	jnz %jcmp, @bitbody, @bitdone
@bitbody
	%bit =w and %crc2, 1
	jnz %bit, @odd, @even
@odd
	%tmp =w shr %crc2, 1
	%crc3 =w xor %tmp, 3988292384
	jmp @bitnext
@even
	%crc4 =w shr %crc2, 1
	jmp @bitnext
@bitnext
	%crc7 =w phi @odd %crc3, @even %crc4
	%j1 =w add %j, 1
	jmp @bitloop
@bitdone
	%i1 =l add %i, 1
	jmp @loop
@done
	%crc9 =w xor %crc, 4294967295
	%res =l extuw %crc9
	ret %res
}
