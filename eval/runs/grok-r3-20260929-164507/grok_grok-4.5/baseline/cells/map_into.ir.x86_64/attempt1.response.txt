export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @header
@header
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%dptr =l add %dst, %off
	%v =l loadl %sptr
	%r =l call %f(l %v)
	storel %r, %dptr
	%lt =w csltl %r, %v
	%add =l extsw %lt
	%cnt1 =l add %cnt, %add
	%i1 =l add %i, 1
	jmp @header
@end
	ret %cnt
}
