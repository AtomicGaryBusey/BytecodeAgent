export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zna =w ceql %na, 0
	jnz %zna, @ret0, @chk_nb
@chk_nb
	%znb =w ceql %nb, 0
	jnz %znb, @ret0, @do_zero
@ret0
	ret 0
@do_zero
	%nr =l add %na, %nb
	jmp @z_header
@z_header
	%zi =l phi @do_zero 0, @z_body %zi_next
	%zcmp =w csltl %zi, %nr
	jnz %zcmp, @z_body, @mul_init
@z_body
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi_next =l add %zi, 1
	jmp @z_header
@mul_init
	jmp @i_header
@i_header
	%i =l phi @mul_init 0, @i_next %i_next
	%icmp =w csltl %i, %na
	jnz %icmp, @i_body, @find_len
@i_body
	%aoff =l mul %i, 8
	%aaddr =l add %a, %aoff
	%ai =l loadl %aaddr
	jmp @j_header
@j_header
	%j =l phi @i_body 0, @j_next %j_next
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @j_body, @i_next
@j_body
	%boff =l mul %j, 8
	%baddr =l add %b, %boff
	%bj =l loadl %baddr
	%alo =l and %ai, 4294967295
	%ahi =l shr %ai, 32
	%blo =l and %bj, 4294967295
	%bhi =l shr %bj, 32
	%w0 =l mul %alo, %blo
	%w0hi =l shr %w0, 32
	%x1y0 =l mul %ahi, %blo
	%t =l add %x1y0, %w0hi
	%w1low =l and %t, 4294967295
	%w2 =l shr %t, 32
	%x0y1 =l mul %alo, %bhi
	%w1 =l add %x0y1, %w1low
	%w1hi =l shr %w1, 32
	%x1y1 =l mul %ahi, %bhi
	%phit =l add %x1y1, %w2
	%phi =l add %phit, %w1hi
	%w0lo =l and %w0, 4294967295
	%w1sh =l shl %w1, 32
	%plo =l add %w1sh, %w0lo
	%ij =l add %i, %j
	%off0 =l mul %ij, 8
	%addr0 =l add %r, %off0
	%r0 =l loadl %addr0
	%sum0 =l add %r0, %plo
	%c0 =w cultl %sum0, %r0
	storel %sum0, %addr0
	%ij1 =l add %ij, 1
	%off1 =l mul %ij1, 8
	%addr1 =l add %r, %off1
	%r1 =l loadl %addr1
	%c0l =l extuw %c0
	%sum1a =l add %r1, %phi
	%ca =w cultl %sum1a, %r1
	%sum1 =l add %sum1a, %c0l
	%cb =w cultl %sum1, %sum1a
	storel %sum1, %addr1
	%cal =l extuw %ca
	%cbl =l extuw %cb
	%carry =l add %cal, %cbl
	%k0 =l add %ij1, 1
	jmp @prop_header
@prop_header
	%carry_p =l phi @j_body %carry, @prop_body %carry_next
	%k =l phi @j_body %k0, @prop_body %k_next
	%pcmp =w cnel %carry_p, 0
	jnz %pcmp, @prop_body, @j_next
@prop_body
	%koff =l mul %k, 8
	%kaddr =l add %r, %koff
	%kr =l loadl %kaddr
	%ksum =l add %kr, %carry_p
	%kc =w cultl %ksum, %kr
	storel %ksum, %kaddr
	%carry_next =l extuw %kc
	%k_next =l add %k, 1
	jmp @prop_header
@j_next
	%j_next =l add %j, 1
	jmp @j_header
@i_next
	%i_next =l add %i, 1
	jmp @i_header
@find_len
	jmp @len_header
@len_header
	%len =l phi @find_len %nr, @len_check %lm1
	%lcmp =w cnel %len, 0
	jnz %lcmp, @len_check, @ret0
@len_check
	%lm1 =l sub %len, 1
	%loff =l mul %lm1, 8
	%laddr =l add %r, %loff
	%lv =l loadl %laddr
	%lzero =w ceql %lv, 0
	jnz %lzero, @len_header, @ret_len
@ret_len
	ret %len
}
