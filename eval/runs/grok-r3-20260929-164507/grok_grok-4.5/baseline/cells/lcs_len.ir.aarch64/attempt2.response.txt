export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
	jmp @init.cond
@init.cond
	%j =l phi @start %j0, @init.body %j1
	%cinit =w cslel %j, %n
	jnz %cinit, @init.body, @init.end
@init.body
	%off =l mul %j, 8
	%p =l add %row, %off
	storel 0, %p
	%j1 =l add %j, 1
	jmp @init.cond
@init.end
	%i0 =l copy 0
	jmp @outer.cond
@outer.cond
	%i =l phi @init.end %i0, @outer.inc %i1
	%couter =w csltl %i, %m
	jnz %couter, @outer.body, @end
@outer.body
	%diag0 =l copy 0
	%jj0 =l copy 1
	jmp @inner.cond
@inner.cond
	%jj =l phi @outer.body %jj0, @inner.inc %jj1
	%diag =l phi @outer.body %diag0, @inner.inc %diag1
	%cinner =w cslel %jj, %n
	jnz %cinner, @inner.body, @outer.inc
@inner.body
	%offj =l mul %jj, 8
	%pj =l add %row, %offj
	%up =l loadl %pj
	%px =l add %x, %i
	%xb =w loadub %px
	%jm1 =l sub %jj, 1
	%py =l add %y, %jm1
	%yb =w loadub %py
	%eq =w ceqw %xb, %yb
	jnz %eq, @eq.yes, @eq.no
@eq.yes
	%val.eq =l add %diag, 1
	jmp @update
@eq.no
	%offl =l mul %jm1, 8
	%pl =l add %row, %offl
	%left =l loadl %pl
	%gt =w csgtl %up, %left
	jnz %gt, @max.up, @max.left
@max.up
	%val.up =l copy %up
	jmp @update
@max.left
	%val.left =l copy %left
	jmp @update
@update
	%val =l phi @eq.yes %val.eq, @max.up %val.up, @max.left %val.left
	storel %val, %pj
	%diag1 =l copy %up
	jmp @inner.inc
@inner.inc
	%jj1 =l add %jj, 1
	jmp @inner.cond
@outer.inc
	%i1 =l add %i, 1
	jmp @outer.cond
@end
	%offn =l mul %n, 8
	%pn =l add %row, %offn
	%res =l loadl %pn
	ret %res
}
