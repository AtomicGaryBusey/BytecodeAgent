binary_search:
cmp rsi, 0
je .Lnotfound
xor rcx, rcx
mov r8, rsi
dec r8
.Lloop:
cmp rcx, r8
jg .Lnotfound
mov rax, rcx
add rax, r8
shr rax, 1
mov r9, qword ptr [rdi + rax*8]
cmp r9, rdx
je .Lfound
jl .Lless
mov r8, rax
dec r8
jmp .Lloop
.Lless:
lea rcx, [rax + 1]
jmp .Lloop
.Lfound:
ret
.Lnotfound:
mov rax, -1
ret
