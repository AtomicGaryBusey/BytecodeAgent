export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%max =l phi @start %max0, @cont %max1
	%i =l phi @start %i0, @cont %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @update, @cont
@update
	%maxu =l copy %val
	jmp @cont
@cont
	%max1 =l phi @body %max, @update %maxu
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
