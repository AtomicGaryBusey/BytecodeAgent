export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
	jmp @outer_cond
@outer_cond
	%ci =w csltl %i, %n
	jnz %ci, @outer_body, @done
@outer_body
	%j =l copy 0
	%t =l sub %n, %i
	%lim =l sub %t, 1
	jmp @inner_cond
@inner_cond
	%cj =w csltl %j, %lim
	jnz %cj, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%pj =l add %a, %off
	%vj =l loadl %pj
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%pj1 =l add %a, %off1
	%vj1 =l loadl %pj1
	%gt =w csgtl %vj, %vj1
	jnz %gt, @do_swap, @inner_inc
@do_swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @inner_inc
@inner_inc
	%j =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
