collatz_max:
mov x1, #0
mov x2, #1
.Louter:
cmp x2, x0
b.gt .Ldone
mov x3, x2
mov x4, #0
.Linner:
cmp x3, #1
b.eq .Linner_done
tst x3, #1
b.eq .Leven
add x3, x3, x3, lsl #1
add x3, x3, #1
b .Linc
.Leven:
lsr x3, x3, #1
.Linc:
add x4, x4, #1
b .Linner
.Linner_done:
cmp x4, x1
csel x1, x4, x1, gt
add x2, x2, #1
b .Louter
.Ldone:
mov x0, x1
ret
