export function l $fill_down(l %a, l %n, l %start) {
@start
	%i =l copy 0
	jmp @loop_header
@loop_header
	%cmp =w csltl %i, %n
	jnz %cmp, @loop_body, @exit
@loop_body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l sub %start, %i
	storel %val, %ptr
	%i =l add %i, 1
	jmp @loop_header
@exit
	ret %n
}
