export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp =w cugt l %dst, %src
	jnz %cmp, @backward, @forward
@forward
	%i =l phi @start 0, @fbody %inext
	%cmpi =w cslt l %i, %n
	jnz %cmpi, @fbody, @done
@fbody
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%val =l loadl %sptr
	%dptr =l add %dst, %off
	storel %val, %dptr
	%inext =l add %i, 1
	jmp @forward
@backward
	%i =l phi @start %n, @bbody %inext
	%cmpi =w csgt l %i, 0
	jnz %cmpi, @bbody, @done
@bbody
	%inext =l sub %i, 1
	%off =l mul %inext, 8
	%sptr =l add %src, %off
	%val =l loadl %sptr
	%dptr =l add %dst, %off
	storel %val, %dptr
	jmp @backward
@done
	ret 0
}
