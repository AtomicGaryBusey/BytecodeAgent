export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp =cugtl %dst, %src
	jnz %cmp, @backward, @forward
@forward
	%i =copy l 0
@floop
	%cmpi =csltl %i, %n
	jnz %cmpi, @fbody, @done
@fbody
	%off =mul l %i, 8
	%sptr =add l %src, %off
	%val =loadl %sptr
	%dptr =add l %dst, %off
	storel %val, %dptr
	%i =add l %i, 1
	jmp @floop
@backward
	%i =copy l %n
@bloop
	%cmpi =csgtl %i, 0
	jnz %cmpi, @bbody, @done
@bbody
	%i =sub l %i, 1
	%off =mul l %i, 8
	%sptr =add l %src, %off
	%val =loadl %sptr
	%dptr =add l %dst, %off
	storel %val, %dptr
	jmp @bloop
@done
	ret 0
}
