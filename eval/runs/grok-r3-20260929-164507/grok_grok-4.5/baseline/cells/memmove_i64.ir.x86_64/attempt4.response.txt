export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%cmp =w cugtl %dst, %src
	jnz %cmp, @backward, @forward
@forward
	%i =l phi @start 0, @fbody %inext
	%cmpi =w csltl %i, %n
	jnz %cmpi, @fbody, @done
@fbody
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%val =l loadl %sptr
	%dptr =l add %dst, %off
	storel %val, %dptr
	%inext =l add %i, 1
	jmp @forward
@backward
	%j =l phi @start %n, @bbody %jnext
	%cmpj =w csgtl %j, 0
	jnz %cmpj, @bbody, @done
@bbody
	%jnext =l sub %j, 1
	%off2 =l mul %jnext, 8
	%sptr2 =l add %src, %off2
	%val2 =l loadl %sptr2
	%dptr2 =l add %dst, %off2
	storel %val2, %dptr2
	jmp @backward
@done
	ret 0
}
