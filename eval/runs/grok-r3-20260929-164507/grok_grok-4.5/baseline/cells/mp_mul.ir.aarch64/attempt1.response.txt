export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%mask =l copy 4294967295
	jmp @zero_test
@zero_test
	%zi =l phi @start 0, @zero_body %zi_next
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zero_body, @zero_done
@zero_body
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi_next =l add %zi, 1
	jmp @zero_test
@zero_done
	%na0 =w ceql %na, 0
	jnz %na0, @find_sig, @check_nb
@check_nb
	%nb0 =w ceql %nb, 0
	jnz %nb0, @find_sig, @mul_init
@mul_init
	jmp @muloop_i_test
@muloop_i_test
	%i =l phi @mul_init 0, @muloop_i_next %i_next
	%icmp =w csltl %i, %na
	jnz %icmp, @muloop_i_body, @find_sig
@muloop_i_body
	%aoff =l mul %i, 8
	%aaddr =l add %a, %aoff
	%ai =l loadl %aaddr
	jmp @muloop_j_test
@muloop_j_test
	%j =l phi @muloop_i_body 0, @muloop_j_next %j_next
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @muloop_j_body, @muloop_i_next
@muloop_j_body
	%boff =l mul %j, 8
	%baddr =l add %b, %boff
	%bj =l loadl %baddr
	%alo =l and %ai, %mask
	%ahi =l shr %ai, 32
	%blo =l and %bj, %mask
	%bhi =l shr %bj, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0h =l shr %p0, 32
	%p1l =l and %p1, %mask
	%p2l =l and %p2, %mask
	%midt =l add %p0h, %p1l
	%mid =l add %midt, %p2l
	%midl =l and %mid, %mask
	%midh =l shr %mid, 32
	%p0l =l and %p0, %mask
	%midlsh =l shl %midl, 32
	%lo =l add %p0l, %midlsh
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%hit =l add %p3, %p1h
	%hit2 =l add %hit, %p2h
	%hi =l add %hit2, %midh
	%kbase =l add %i, %j
	%koff =l mul %kbase, 8
	%kaddr =l add %r, %koff
	%rk =l loadl %kaddr
	%sumlo =l add %rk, %lo
	%carry0 =w cultl %sumlo, %rk
	storel %sumlo, %kaddr
	%k1 =l add %kbase, 1
	%k1off =l mul %k1, 8
	%k1addr =l add %r, %k1off
	%rk1 =l loadl %k1addr
	%carry0l =l extuw %carry0
	%hic =l add %hi, %carry0l
	%sumhi =l add %rk1, %hic
	%carry1 =w cultl %sumhi, %rk1
	storel %sumhi, %k1addr
	jmp @prop_test
@prop_test
	%pk =l phi @muloop_j_body %k1, @prop_body %pk_next
	%pc =w phi @muloop_j_body %carry1, @prop_body %pc_next
	jnz %pc, @prop_body, @muloop_j_next
@prop_body
	%pk_next =l add %pk, 1
	%pkoff =l mul %pk_next, 8
	%pkaddr =l add %r, %pkoff
	%rpk =l loadl %pkaddr
	%pcl =l extuw %pc
	%psum =l add %rpk, %pcl
	%pc_next =w cultl %psum, %rpk
	storel %psum, %pkaddr
	jmp @prop_test
@muloop_j_next
	%j_next =l add %j, 1
	jmp @muloop_j_test
@muloop_i_next
	%i_next =l add %i, 1
	jmp @muloop_i_test
@find_sig
	%sk0 =l copy %n
	jmp @sig_test
@sig_test
	%sk =l phi @find_sig %sk0, @sig_body %sk_next
	%skpos =w csltl 0, %sk
	jnz %skpos, @sig_check, @sig_done
@sig_check
	%sk1 =l sub %sk, 1
	%skoff =l mul %sk1, 8
	%skaddr =l add %r, %skoff
	%skval =l loadl %skaddr
	%skz =w ceql %skval, 0
	jnz %skz, @sig_body, @sig_done
@sig_body
	%sk_next =l sub %sk, 1
	jmp @sig_test
@sig_done
	ret %sk
}
