export function l $qsort_pairs(l %a.arg, l %n.arg) {
@start
	jmp @header
@header
	%a =l phi @start %a.arg, @rec_left %right_a, @rec_right %a
	%n =l phi @start %n.arg, @rec_left %right_n, @rec_right %left_n
	%nle1 =w cslel %n, 1
	jnz %nle1, @ret0, @doit
@ret0
	ret 0
@doit
	%nm1 =l sub %n, 1
	%mid =l div %n, 2
	%mid_off =l mul %mid, 16
	%nm1_off =l mul %nm1, 16
	%addr_mid =l add %a, %mid_off
	%addr_nm1 =l add %a, %nm1_off
	%addr_midv =l add %addr_mid, 8
	%addr_nm1v =l add %addr_nm1, 8
	%mkey =l loadl %addr_mid
	%mval =l loadl %addr_midv
	%nkey =l loadl %addr_nm1
	%nval =l loadl %addr_nm1v
	storel %nkey, %addr_mid
	storel %nval, %addr_midv
	storel %mkey, %addr_nm1
	storel %mval, %addr_nm1v
	jmp @part_loop
@part_loop
	%i =l phi @doit 0, @after_cmp %i.new
	%j =l phi @doit 0, @after_cmp %j.new
	%jlt =w csltl %j, %nm1
	jnz %jlt, @part_body, @part_done
@part_body
	%j_off =l mul %j, 16
	%addr_j =l add %a, %j_off
	%addr_jv =l add %addr_j, 8
	%jkey =l loadl %addr_j
	%jval =l loadl %addr_jv
	%ltkey =w csltl %jkey, %mkey
	%eqkey =w ceql %jkey, %mkey
	%ltval =w csltl %jval, %mval
	%andlt =w and %eqkey, %ltval
	%isless =w or %ltkey, %andlt
	jnz %isless, @do_swap, @no_swap
@do_swap
	%i_off =l mul %i, 16
	%addr_i =l add %a, %i_off
	%addr_iv =l add %addr_i, 8
	%ikey =l loadl %addr_i
	%ival =l loadl %addr_iv
	storel %jkey, %addr_i
	storel %jval, %addr_iv
	storel %ikey, %addr_j
	storel %ival, %addr_jv
	%i.sw =l add %i, 1
	jmp @after_cmp
@no_swap
	%i.ns =l copy %i
	jmp @after_cmp
@after_cmp
	%i.new =l phi @do_swap %i.sw, @no_swap %i.ns
	%j.new =l add %j, 1
	jmp @part_loop
@part_done
	%i_off2 =l mul %i, 16
	%addr_i2 =l add %a, %i_off2
	%addr_iv2 =l add %addr_i2, 8
	%ikey2 =l loadl %addr_i2
	%ival2 =l loadl %addr_iv2
	storel %mkey, %addr_i2
	storel %mval, %addr_iv2
	storel %ikey2, %addr_nm1
	storel %ival2, %addr_nm1v
	%left_n =l copy %i
	%ip1 =l add %i, 1
	%right_n =l sub %n, %ip1
	%ip1_off =l mul %ip1, 16
	%right_a =l add %a, %ip1_off
	%left_lt_right =w csltl %left_n, %right_n
	jnz %left_lt_right, @rec_left, @rec_right
@rec_left
	%r1 =l call $qsort_pairs(l %a, l %left_n)
	jmp @header
@rec_right
	%r2 =l call $qsort_pairs(l %right_a, l %right_n)
	jmp @header
}
