qsort_pairs:
cmp rsi, 1
jle .Lret0
push rbp
mov rbp, rsp
push rbx
push r12
push r13
push r14
sub rsp, 16
mov rbx, rdi
mov r12, rsi
lea rax, [r12 - 1]
shl rax, 4
mov r8, qword ptr [rbx + rax]
mov r9, qword ptr [rbx + rax + 8]
mov qword ptr [rsp], r8
mov qword ptr [rsp + 8], r9
xor r13, r13
xor r14, r14
.Lpart_loop:
mov rax, r12
dec rax
cmp r14, rax
jge .Lpart_end
mov rax, r14
shl rax, 4
mov rcx, qword ptr [rbx + rax]
mov rdx, qword ptr [rbx + rax + 8]
cmp rcx, qword ptr [rsp]
jl .Ldo_swap
jg .Lnextj
cmp rdx, qword ptr [rsp + 8]
jg .Lnextj
.Ldo_swap:
mov rax, r13
shl rax, 4
mov r8, qword ptr [rbx + rax]
mov r9, qword ptr [rbx + rax + 8]
mov qword ptr [rbx + rax], rcx
mov qword ptr [rbx + rax + 8], rdx
mov rax, r14
shl rax, 4
mov qword ptr [rbx + rax], r8
mov qword ptr [rbx + rax + 8], r9
inc r13
.Lnextj:
inc r14
jmp .Lpart_loop
.Lpart_end:
mov rax, r13
shl rax, 4
mov rcx, r12
dec rcx
shl rcx, 4
mov r8, qword ptr [rbx + rax]
mov r9, qword ptr [rbx + rax + 8]
mov rsi, qword ptr [rbx + rcx]
mov rdi, qword ptr [rbx + rcx + 8]
mov qword ptr [rbx + rax], rsi
mov qword ptr [rbx + rax + 8], rdi
mov qword ptr [rbx + rcx], r8
mov qword ptr [rbx + rcx + 8], r9
mov rdi, rbx
mov rsi, r13
call qsort_pairs
lea rax, [r13 + 1]
cmp rax, r12
jge .Lno_right
mov rdi, rax
shl rdi, 4
add rdi, rbx
mov rsi, r12
sub rsi, rax
call qsort_pairs
.Lno_right:
add rsp, 16
pop r14
pop r13
pop r12
pop rbx
pop rbp
.Lret0:
xor rax, rax
ret
