export function l $bubble_sort(l %a, l %n) {
@start
	jmp @oloop
@oloop
	%i =l phi @start 0, @oinc %inext
	%ocmp =w csltl %i, %n
	jnz %ocmp, @obody, @end
@obody
	%nmi =l sub %n, %i
	%lim =l sub %nmi, 1
	jmp @iloop
@iloop
	%j =l phi @obody 0, @jinc %jnext
	%icmp =w csltl %j, %lim
	jnz %icmp, @ibody, @oinc
@ibody
	%off =l mul %j, 8
	%p =l add %a, %off
	%vj =l loadl %p
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%vj1 =l loadl %p1
	%gt =w csgtl %vj, %vj1
	jnz %gt, @swap, @jinc
@swap
	storel %vj1, %p
	storel %vj, %p1
	jmp @jinc
@jinc
	%jnext =l add %j, 1
	jmp @iloop
@oinc
	%inext =l add %i, 1
	jmp @oloop
@end
	ret 0
}
