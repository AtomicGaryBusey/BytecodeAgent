export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_cond
@init_cond
	%jinit =l phi @start 0, @init_body %jinit_n
	%init_gt =w csgtl %jinit, %n
	jnz %init_gt, @init_done, @init_body
@init_body
	%init_off =l mul %jinit, 8
	%init_ptr =l add %row, %init_off
	storel 0, %init_ptr
	%jinit_n =l add %jinit, 1
	jmp @init_cond
@init_done
	jmp @i_cond
@i_cond
	%i =l phi @init_done 0, @j_done %i_n
	%i_ge =w csgel %i, %m
	jnz %i_ge, @i_done, @i_body
@i_body
	jmp @j_cond
@j_cond
	%j =l phi @i_body 1, @store %j_n
	%diag =l phi @i_body 0, @store %diag_n
	%j_gt =w csgtl %j, %n
	jnz %j_gt, @j_done, @j_body
@j_body
	%j_off =l mul %j, 8
	%pj =l add %row, %j_off
	%up =l loadl %pj
	%px =l add %x, %i
	%xi =w loadub %px
	%jm1 =l sub %j, 1
	%py =l add %y, %jm1
	%yj =w loadub %py
	%eq =w ceqw %xi, %yj
	jnz %eq, @equal, @noteq
@equal
	%val_eq =l add %diag, 1
	jmp @store
@noteq
	%jm1_off =l mul %jm1, 8
	%pjm1 =l add %row, %jm1_off
	%left =l loadl %pjm1
	%up_gt =w csgtl %up, %left
	jnz %up_gt, @use_up, @use_left
@use_up
	%val_up =l copy %up
	jmp @store
@use_left
	%val_left =l copy %left
	jmp @store
@store
	%val =l phi @equal %val_eq, @use_up %val_up, @use_left %val_left
	storel %val, %pj
	%diag_n =l copy %up
	%j_n =l add %j, 1
	jmp @j_cond
@j_done
	%i_n =l add %i, 1
	jmp @i_cond
@i_done
	%n_off =l mul %n, 8
	%pn =l add %row, %n_off
	%res =l loadl %pn
	ret %res
}
