export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%idx =l mul %i, 8
	%p =l add %a, %idx
	%val =l loadl %p
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
