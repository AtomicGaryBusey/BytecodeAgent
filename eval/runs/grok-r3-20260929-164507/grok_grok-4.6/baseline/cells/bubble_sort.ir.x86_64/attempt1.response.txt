export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @innerend %inext
	%cmpi =w csltl %i, %n
	jnz %cmpi, @outerbody, @done
@outerbody
	%nmi =l sub %n, %i
	%lim =l sub %nmi, 1
	jmp @inner
@inner
	%j =l phi @outerbody 0, @incr %jnext
	%cmpj =w csltl %j, %lim
	jnz %cmpj, @innerbody, @innerend
@innerbody
	%off =l shl %j, 3
	%pj =l add %a, %off
	%vj =l loadl %pj
	%jp1 =l add %j, 1
	%off1 =l shl %jp1, 3
	%pj1 =l add %a, %off1
	%vj1 =l loadl %pj1
	%need =w csgtl %vj, %vj1
	jnz %need, @swap, @incr
@swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @incr
@incr
	%jnext =l add %j, 1
	jmp @inner
@innerend
	%inext =l add %i, 1
	jmp @outer
@done
	ret 0
}
