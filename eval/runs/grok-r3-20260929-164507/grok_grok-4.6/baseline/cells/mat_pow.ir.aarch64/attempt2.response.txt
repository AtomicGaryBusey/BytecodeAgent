export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%z =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%nn =l mul %n, %n
	%nn2 =l add %nn, %nn
	jmp @idi
@idi
	%i =l phi @start %z, @idin %i2
	%icmp =w csltl %i, %n
	jnz %icmp, @idj, @iddone
@idj
	%j =l phi @idi %z, @idbody %j2
	%jcmp =w csltl %j, %n
	jnz %jcmp, @idbody, @idin
@idbody
	%ieqj =w ceql %i, %j
	%ival =l extuw %ieqj
	%irow =l mul %i, %n
	%ij =l add %irow, %j
	%ijoff =l mul %ij, %eight
	%ijptr =l add %out, %ijoff
	storel %ival, %ijptr
	%j2 =l add %j, %one
	jmp @idj
@idin
	%i2 =l add %i, %one
	jmp @idi
@iddone
	%knz =w cnel %k, %z
	jnz %knz, @copya, @cleanup
@copya
	%ca =l phi @iddone %z, @copyab %ca2
	%cacmp =w csltl %ca, %nn
	jnz %cacmp, @copyab, @powlp
@copyab
	%caoff =l mul %ca, %eight
	%casrc =l add %a, %caoff
	%cadst =l add %out, %caoff
	%caval =l loadl %casrc
	storel %caval, %cadst
	%ca2 =l add %ca, %one
	jmp @copya
@powlp
	%p =l phi @copya %one, @copypd %p2
	%pcmp =w csltl %p, %k
	jnz %pcmp, @muli, @cleanup
@muli
	%mi =l phi @powlp %z, @mulin %mi2
	%micmp =w csltl %mi, %n
	jnz %micmp, @mulj, @copyp
@mulj
	%mj =l phi @muli %z, @mulst %mj2
	%mjcmp =w csltl %mj, %n
	jnz %mjcmp, @mult, @mulin
@mult
	%mt =l phi @mulj %z, @multb %mt2
	%ms =l phi @mulj %z, @multb %ms2
	%mtcmp =w csltl %mt, %n
	jnz %mtcmp, @multb, @mulst
@multb
	%arow =l mul %mi, %n
	%aiti =l add %arow, %mt
	%aito =l mul %aiti, %eight
	%aitp =l add %out, %aito
	%ait =l loadl %aitp
	%brow =l mul %mt, %n
	%btji =l add %brow, %mj
	%btjo =l mul %btji, %eight
	%btjp =l add %a, %btjo
	%btj =l loadl %btjp
	%mp =l mul %ait, %btj
	%ms2 =l add %ms, %mp
	%mt2 =l add %mt, %one
	jmp @mult
@mulst
	%crow =l mul %mi, %n
	%ciji =l add %crow, %mj
	%cijo =l mul %ciji, %eight
	%cijp =l add %tmp, %cijo
	storel %ms, %cijp
	%mj2 =l add %mj, %one
	jmp @mulj
@mulin
	%mi2 =l add %mi, %one
	jmp @muli
@copyp
	%cp =l phi @muli %z, @copypb %cp2
	%cpcmp =w csltl %cp, %nn
	jnz %cpcmp, @copypb, @copypd
@copypb
	%cpoff =l mul %cp, %eight
	%cpsrc =l add %tmp, %cpoff
	%cpdst =l add %out, %cpoff
	%cpval =l loadl %cpsrc
	storel %cpval, %cpdst
	%cp2 =l add %cp, %one
	jmp @copyp
@copypd
	%p2 =l add %p, %one
	jmp @powlp
@cleanup
	jmp @zlp
@zlp
	%zi =l phi @cleanup %z, @zlpb %zi2
	%zicmp =w csltl %zi, %nn2
	jnz %zicmp, @zlpb, @zdone
@zlpb
	%zoff =l mul %zi, %eight
	%zptr =l add %tmp, %zoff
	storel %z, %zptr
	%zi2 =l add %zi, %one
	jmp @zlp
@zdone
	ret 0
}
