export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%nn =l mul %n, %n
	%nn8 =l mul %nn, %eight
	%base =l copy %tmp
	%prod =l add %tmp, %nn8
	jmp @idi
@idi
	%i =l phi @start %zero, @idj_d %i_n
	%i_ok =w csltl %i, %n
	jnz %i_ok, @idj, @id_done
@idj
	%j =l phi @idi %zero, @idj_b %j_n
	%j_ok =w csltl %j, %n
	jnz %j_ok, @idj_b, @idj_d
@idj_b
	%eq =w ceql %i, %j
	%val =l extuw %eq
	%in =l mul %i, %n
	%idx =l add %in, %j
	%off =l mul %idx, %eight
	%ptr =l add %out, %off
	storel %val, %ptr
	%j_n =l add %j, %one
	jmp @idj
@idj_d
	%i_n =l add %i, %one
	jmp @idi
@id_done
	%k_nz =w cnel %k, 0
	jnz %k_nz, @cp1, @cleanup
@cp1
	%c1 =l phi @id_done %zero, @cp1_b %c1_n
	%c1_ok =w csltl %c1, %nn
	jnz %c1_ok, @cp1_b, @bits
@cp1_b
	%c1off =l mul %c1, %eight
	%c1pa =l add %a, %c1off
	%c1pb =l add %base, %c1off
	%c1v =l loadl %c1pa
	storel %c1v, %c1pb
	%c1_n =l add %c1, %one
	jmp @cp1
@bits
	%kk =l phi @cp1 %k, @after_sq %kk_n
	%kk_nz =w cnel %kk, 0
	jnz %kk_nz, @bit_b, @cleanup
@bit_b
	%lsb =l and %kk, %one
	%lsb_nz =w cnel %lsb, 0
	jnz %lsb_nz, @mul_res, @do_sq
@mul_res
	jmp @mul
@do_sq
	jmp @mul
@mul
	%ma =l phi @mul_res %out, @do_sq %base
	%is_res =l phi @mul_res %one, @do_sq %zero
	%mb =l copy %base
	%mc =l copy %prod
	jmp @muli
@muli
	%mi =l phi @mul %zero, @mulj_d %mi_n
	%mi_ok =w csltl %mi, %n
	jnz %mi_ok, @mulj, @mul_d
@mulj
	%mj =l phi @muli %zero, @mult_d %mj_n
	%mj_ok =w csltl %mj, %n
	jnz %mj_ok, @mult, @mulj_d
@mult
	%mt =l phi @mulj %zero, @mult_b %mt_n
	%ms =l phi @mulj %zero, @mult_b %ms_n
	%mt_ok =w csltl %mt, %n
	jnz %mt_ok, @mult_b, @mult_d
@mult_b
	%ain =l mul %mi, %n
	%ait_i =l add %ain, %mt
	%ait_o =l mul %ait_i, %eight
	%ait_p =l add %ma, %ait_o
	%ait =l loadl %ait_p
	%btn =l mul %mt, %n
	%btj_i =l add %btn, %mj
	%btj_o =l mul %btj_i, %eight
	%btj_p =l add %mb, %btj_o
	%btj =l loadl %btj_p
	%mp =l mul %ait, %btj
	%ms_n =l add %ms, %mp
	%mt_n =l add %mt, %one
	jmp @mult
@mult_d
	%cin =l mul %mi, %n
	%cij_i =l add %cin, %mj
	%cij_o =l mul %cij_i, %eight
	%cij_p =l add %mc, %cij_o
	storel %ms, %cij_p
	%mj_n =l add %mj, %one
	jmp @mulj
@mulj_d
	%mi_n =l add %mi, %one
	jmp @muli
@mul_d
	%isrw =w cnel %is_res, 0
	jnz %isrw, @cpdst_res, @cpdst_base
@cpdst_res
	jmp @cploop
@cpdst_base
	jmp @cploop
@cploop
	%cdst =l phi @cpdst_res %out, @cpdst_base %base, @cploop_b %cdst
	%ci =l phi @cpdst_res %zero, @cpdst_base %zero, @cploop_b %ci_n
	%ci_ok =w csltl %ci, %nn
	jnz %ci_ok, @cploop_b, @cploop_d
@cploop_b
	%coff =l mul %ci, %eight
	%cps =l add %prod, %coff
	%cpd =l add %cdst, %coff
	%cv =l loadl %cps
	storel %cv, %cpd
	%ci_n =l add %ci, %one
	jmp @cploop
@cploop_d
	%isrw2 =w cnel %is_res, 0
	jnz %isrw2, @do_sq, @after_sq
@after_sq
	%kk_n =l shr %kk, 1
	jmp @bits
@cleanup
	%twonn =l add %nn, %nn
	jmp @zloop
@zloop
	%zi =l phi @cleanup %zero, @zloop_b %zi_n
	%zi_ok =w csltl %zi, %twonn
	jnz %zi_ok, @zloop_b, @zdone
@zloop_b
	%zoff =l mul %zi, %eight
	%zp =l add %tmp, %zoff
	storel %zero, %zp
	%zi_n =l add %zi, %one
	jmp @zloop
@zdone
	ret 0
}
