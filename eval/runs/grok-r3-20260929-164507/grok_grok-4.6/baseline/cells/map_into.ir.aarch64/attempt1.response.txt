export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%val =l loadl %sptr
	%res =l call %f(l %val)
	%dptr =l add %dst, %off
	storel %res, %dptr
	%lt =w csltl %res, %val
	%inc =l extuw %lt
	%cntnext =l add %cnt, %inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
