export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%scratch =l add %tmp, %nn8
	%nn2 =l add %nn, %nn
	%id_i0 =l copy 0
	jmp @id_i
@id_i
	%id_i =l phi @start %id_i0, @id_inext %id_i1
	%id_ci =w csltl %id_i, %n
	jnz %id_ci, @id_jinit, @cp_init
@id_jinit
	%id_j0 =l copy 0
	jmp @id_j
@id_j
	%id_j =l phi @id_jinit %id_j0, @id_jbody %id_j1
	%id_cj =w csltl %id_j, %n
	jnz %id_cj, @id_jbody, @id_inext
@id_jbody
	%id_eq =w ceql %id_i, %id_j
	%id_val =l extuw %id_eq
	%id_in =l mul %id_i, %n
	%id_idx =l add %id_in, %id_j
	%id_off =l mul %id_idx, 8
	%id_ptr =l add %out, %id_off
	storel %id_val, %id_ptr
	%id_j1 =l add %id_j, 1
	jmp @id_j
@id_inext
	%id_i1 =l add %id_i, 1
	jmp @id_i
@cp_init
	%cp_i0 =l copy 0
	jmp @cp_loop
@cp_loop
	%cp_i =l phi @cp_init %cp_i0, @cp_body %cp_i1
	%cp_c =w csltl %cp_i, %nn
	jnz %cp_c, @cp_body, @pw_init
@cp_body
	%cp_off =l mul %cp_i, 8
	%cp_ap =l add %a, %cp_off
	%cp_v =l loadl %cp_ap
	%cp_tp =l add %tmp, %cp_off
	storel %cp_v, %cp_tp
	%cp_i1 =l add %cp_i, 1
	jmp @cp_loop
@pw_init
	jmp @pw_loop
@pw_loop
	%kcur =l phi @pw_init %k, @pw_next %knext
	%kpos =w csgtl %kcur, 0
	jnz %kpos, @pw_odd, @zr_init
@pw_odd
	%kbit =l and %kcur, 1
	%isodd =w cnel %kbit, 0
	jnz %isodd, @m1_iinit, @m2_iinit
@m1_iinit
	%m1_i0 =l copy 0
	jmp @m1_i
@m1_i
	%m1_i =l phi @m1_iinit %m1_i0, @m1_inext %m1_i1
	%m1_ci =w csltl %m1_i, %n
	jnz %m1_ci, @m1_jinit, @c1_init
@m1_jinit
	%m1_j0 =l copy 0
	jmp @m1_j
@m1_j
	%m1_j =l phi @m1_jinit %m1_j0, @m1_jnext %m1_j1
	%m1_cj =w csltl %m1_j, %n
	jnz %m1_cj, @m1_tinit, @m1_inext
@m1_tinit
	%m1_t0 =l copy 0
	%m1_s0 =l copy 0
	jmp @m1_t
@m1_t
	%m1_t =l phi @m1_tinit %m1_t0, @m1_tbody %m1_t1
	%m1_s =l phi @m1_tinit %m1_s0, @m1_tbody %m1_s1
	%m1_ct =w csltl %m1_t, %n
	jnz %m1_ct, @m1_tbody, @m1_store
@m1_tbody
	%m1_ain =l mul %m1_i, %n
	%m1_ai =l add %m1_ain, %m1_t
	%m1_ao =l mul %m1_ai, 8
	%m1_ap =l add %out, %m1_ao
	%m1_av =l loadl %m1_ap
	%m1_bin =l mul %m1_t, %n
	%m1_bi =l add %m1_bin, %m1_j
	%m1_bo =l mul %m1_bi, 8
	%m1_bp =l add %tmp, %m1_bo
	%m1_bv =l loadl %m1_bp
	%m1_p =l mul %m1_av, %m1_bv
	%m1_s1 =l add %m1_s, %m1_p
	%m1_t1 =l add %m1_t, 1
	jmp @m1_t
@m1_store
	%m1_cin =l mul %m1_i, %n
	%m1_cix =l add %m1_cin, %m1_j
	%m1_co =l mul %m1_cix, 8
	%m1_cp =l add %scratch, %m1_co
	storel %m1_s, %m1_cp
	jmp @m1_jnext
@m1_jnext
	%m1_j1 =l add %m1_j, 1
	jmp @m1_j
@m1_inext
	%m1_i1 =l add %m1_i, 1
	jmp @m1_i
@c1_init
	%c1_i0 =l copy 0
	jmp @c1_loop
@c1_loop
	%c1_i =l phi @c1_init %c1_i0, @c1_body %c1_i1
	%c1_c =w csltl %c1_i, %nn
	jnz %c1_c, @c1_body, @m2_iinit
@c1_body
	%c1_off =l mul %c1_i, 8
	%c1_sp =l add %scratch, %c1_off
	%c1_v =l loadl %c1_sp
	%c1_op =l add %out, %c1_off
	storel %c1_v, %c1_op
	%c1_i1 =l add %c1_i, 1
	jmp @c1_loop
@m2_iinit
	%m2_i0 =l copy 0
	jmp @m2_i
@m2_i
	%m2_i =l phi @m2_iinit %m2_i0, @m2_inext %m2_i1
	%m2_ci =w csltl %m2_i, %n
	jnz %m2_ci, @m2_jinit, @c2_init
@m2_jinit
	%m2_j0 =l copy 0
	jmp @m2_j
@m2_j
	%m2_j =l phi @m2_jinit %m2_j0, @m2_jnext %m2_j1
	%m2_cj =w csltl %m2_j, %n
	jnz %m2_cj, @m2_tinit, @m2_inext
@m2_tinit
	%m2_t0 =l copy 0
	%m2_s0 =l copy 0
	jmp @m2_t
@m2_t
	%m2_t =l phi @m2_tinit %m2_t0, @m2_tbody %m2_t1
	%m2_s =l phi @m2_tinit %m2_s0, @m2_tbody %m2_s1
	%m2_ct =w csltl %m2_t, %n
	jnz %m2_ct, @m2_tbody, @m2_store
@m2_tbody
	%m2_ain =l mul %m2_i, %n
	%m2_ai =l add %m2_ain, %m2_t
	%m2_ao =l mul %m2_ai, 8
	%m2_ap =l add %tmp, %m2_ao
	%m2_av =l loadl %m2_ap
	%m2_bin =l mul %m2_t, %n
	%m2_bi =l add %m2_bin, %m2_j
	%m2_bo =l mul %m2_bi, 8
	%m2_bp =l add %tmp, %m2_bo
	%m2_bv =l loadl %m2_bp
	%m2_p =l mul %m2_av, %m2_bv
	%m2_s1 =l add %m2_s, %m2_p
	%m2_t1 =l add %m2_t, 1
	jmp @m2_t
@m2_store
	%m2_cin =l mul %m2_i, %n
	%m2_cix =l add %m2_cin, %m2_j
	%m2_co =l mul %m2_cix, 8
	%m2_cp =l add %scratch, %m2_co
	storel %m2_s, %m2_cp
	jmp @m2_jnext
@m2_jnext
	%m2_j1 =l add %m2_j, 1
	jmp @m2_j
@m2_inext
	%m2_i1 =l add %m2_i, 1
	jmp @m2_i
@c2_init
	%c2_i0 =l copy 0
	jmp @c2_loop
@c2_loop
	%c2_i =l phi @c2_init %c2_i0, @c2_body %c2_i1
	%c2_c =w csltl %c2_i, %nn
	jnz %c2_c, @c2_body, @pw_next
@c2_body
	%c2_off =l mul %c2_i, 8
	%c2_sp =l add %scratch, %c2_off
	%c2_v =l loadl %c2_sp
	%c2_tp =l add %tmp, %c2_off
	storel %c2_v, %c2_tp
	%c2_i1 =l add %c2_i, 1
	jmp @c2_loop
@pw_next
	%knext =l shr %kcur, 1
	jmp @pw_loop
@zr_init
	%zr_i0 =l copy 0
	jmp @zr_loop
@zr_loop
	%zr_i =l phi @zr_init %zr_i0, @zr_body %zr_i1
	%zr_c =w csltl %zr_i, %nn2
	jnz %zr_c, @zr_body, @done
@zr_body
	%zr_off =l mul %zr_i, 8
	%zr_p =l add %tmp, %zr_off
	storel 0, %zr_p
	%zr_i1 =l add %zr_i, 1
	jmp @zr_loop
@done
	ret 0
}
