binary_search:
    xor r8, r8
    mov r9, rsi
    dec r9
.Lloop:
    cmp r8, r9
    jg .Lnotfound
    lea rcx, [r8 + r9]
    shr rcx, 1
    cmp qword ptr [rdi + rcx*8], rdx
    je .Lfound
    jl .Lgo_right
    lea r9, [rcx - 1]
    jmp .Lloop
.Lgo_right:
    lea r8, [rcx + 1]
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lnotfound:
    mov rax, -1
    ret
