export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%nis0 =w ceql %n, 0
	jnz %nis0, @ret0, @zloop
@ret0
	ret 0
@zloop
	%zi =l phi @start 0, @zbody %zinext
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zbody, @oloop
@zbody
	%zoff =l mul %zi, 8
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zinext =l add %zi, 1
	jmp @zloop
@oloop
	%i =l phi @zloop 0, @ocont %inext
	%icmp =w csltl %i, %na
	jnz %icmp, @obody, @lloop
@obody
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%ai =l loadl %aptr
	jmp @iloop
@iloop
	%j =l phi @obody 0, @ibody %jnext
	%c =l phi @obody 0, @ibody %cnext
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @ibody, @idone
@ibody
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bj =l loadl %bptr
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rptr =l add %r, %roff
	%rij =l loadl %rptr
	%u0 =l and %ai, 4294967295
	%u1 =l shr %ai, 32
	%v0 =l and %bj, 4294967295
	%v1 =l shr %bj, 32
	%p00 =l mul %u0, %v0
	%p01 =l mul %u0, %v1
	%p10 =l mul %u1, %v0
	%p11 =l mul %u1, %v1
	%ps =l add %p01, %p10
	%pcy =w cultl %ps, %p01
	%s0 =l and %ps, 4294967295
	%s1 =l shr %ps, 32
	%s0s =l shl %s0, 32
	%t =l add %p00, %s0s
	%tcy =w cultl %t, %p00
	%pcyl =l extuw %pcy
	%tcyl =l extuw %tcy
	%cys32 =l shl %pcyl, 32
	%h1 =l add %p11, %s1
	%h2 =l add %h1, %tcyl
	%hi =l add %h2, %cys32
	%sum1 =l add %t, %rij
	%cr1 =w cultl %sum1, %t
	%cr1l =l extuw %cr1
	%hi1 =l add %hi, %cr1l
	%sum2 =l add %sum1, %c
	%cr2 =w cultl %sum2, %sum1
	%cr2l =l extuw %cr2
	%cnext =l add %hi1, %cr2l
	storel %sum2, %rptr
	%jnext =l add %j, 1
	jmp @iloop
@idone
	%inb =l add %i, %nb
	%coff =l mul %inb, 8
	%cptr =l add %r, %coff
	storel %c, %cptr
	jmp @ocont
@ocont
	%inext =l add %i, 1
	jmp @oloop
@lloop
	%k =l phi @oloop %n, @lbody %knext
	%kgt0 =w csgtl %k, 0
	jnz %kgt0, @lchk, @ldone
@lchk
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kptr =l add %r, %koff
	%kval =l loadl %kptr
	%kis0 =w ceql %kval, 0
	jnz %kis0, @lbody, @ldone
@lbody
	%knext =l sub %k, 1
	jmp @lloop
@ldone
	ret %k
}
