export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%k0 =l copy 0
	jmp @init
@init
	%k =l phi @start %k0, @initbody %knext
	%cinit =w csgtl %k, %n
	jnz %cinit, @initdone, @initbody
@initbody
	%koff =l mul %k, 8
	%kptr =l add %row, %koff
	storel 0, %kptr
	%knext =l add %k, 1
	jmp @init
@initdone
	%i0 =l copy 0
	jmp @outer
@outer
	%i =l phi @initdone %i0, @innerdone %inext
	%ci =w csgel %i, %m
	jnz %ci, @end, @ibody
@ibody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%d0 =l copy 0
	%j0 =l copy 1
	jmp @inner
@inner
	%j =l phi @ibody %j0, @jupd %jnext
	%diag =l phi @ibody %d0, @jupd %dnext
	%cj =w csgtl %j, %n
	jnz %cj, @innerdone, @jbody
@jbody
	%joff =l mul %j, 8
	%jptr =l add %row, %joff
	%up =l loadl %jptr
	%jm1 =l sub %j, 1
	%loff =l mul %jm1, 8
	%lptr =l add %row, %loff
	%left =l loadl %lptr
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @eq, @neq
@eq
	%veq =l add %diag, 1
	jmp @set
@neq
	%gt =w csgtl %up, %left
	jnz %gt, @mup, @mlt
@mup
	jmp @set
@mlt
	jmp @set
@set
	%val =l phi @eq %veq, @mup %up, @mlt %left
	storel %val, %jptr
	jmp @jupd
@jupd
	%jnext =l add %j, 1
	%dnext =l copy %up
	jmp @inner
@innerdone
	%inext =l add %i, 1
	jmp @outer
@end
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%res =l loadl %nptr
	ret %res
}
