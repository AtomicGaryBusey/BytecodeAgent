export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%cont =w csltl %i, %nm1
	jnz %cont, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v0 =l loadl %p
	%p1 =l add %p, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	%inc =l extuw %gt
	%cntnext =l add %cnt, %inc
	%inext =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
