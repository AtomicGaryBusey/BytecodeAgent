export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @end
@body
	%offi =l mul %i, 8
	%pi =l add %a, %offi
	%vi =l loadl %pi
	%offj =l mul %j, 8
	%pj =l add %a, %offj
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@end
	ret 0
}
