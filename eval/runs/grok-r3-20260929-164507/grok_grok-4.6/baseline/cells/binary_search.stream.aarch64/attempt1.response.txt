binary_search:
    mov x3, #0
.Lloop:
    cmp x3, x1
    b.ge .Lnotfound
    sub x4, x1, x3
    add x4, x3, x4, lsr #1
    ldr x5, [x0, x4, lsl #3]
    cmp x5, x2
    b.eq .Lfound
    b.gt .Lgreater
    add x3, x4, #1
    b .Lloop
.Lgreater:
    mov x1, x4
    b .Lloop
.Lfound:
    mov x0, x4
    ret
.Lnotfound:
    mov x0, #-1
    ret
