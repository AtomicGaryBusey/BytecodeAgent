export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy -1
	%i0 =l copy 0
	%eq0 =w ceql %n, 0
	jnz %eq0, @finish, @byteloop
@byteloop
	%crc1 =w phi @start %crc0, @nextbyte %crc5
	%i1 =l phi @start %i0, @nextbyte %i2
	%addr =l add %p, %i1
	%byte =w loadub %addr
	%crc2 =w xor %crc1, %byte
	%bits0 =w copy 8
	jmp @bitloop
@bitloop
	%crc3 =w phi @byteloop %crc2, @bitdone %crc5
	%bits1 =w phi @byteloop %bits0, @bitdone %bits2
	%bit =w and %crc3, 1
	%crc4 =w shr %crc3, 1
	jnz %bit, @doxor, @bitdone
@doxor
	%crc4x =w xor %crc4, 0xEDB88320
	jmp @bitdone
@bitdone
	%crc5 =w phi @bitloop %crc4, @doxor %crc4x
	%bits2 =w sub %bits1, 1
	jnz %bits2, @bitloop, @nextbyte
@nextbyte
	%i2 =l add %i1, 1
	%more =w csltl %i2, %n
	jnz %more, @byteloop, @finish
@finish
	%crc6 =w phi @start %crc0, @nextbyte %crc5
	%crc7 =w xor %crc6, -1
	%res =l extuw %crc7
	ret %res
}
