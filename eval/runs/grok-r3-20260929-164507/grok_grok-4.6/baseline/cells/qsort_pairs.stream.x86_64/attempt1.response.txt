qsort_pairs:
    cmp rsi, 1
    jle .Lret0
    push rbp
    push rbx
    push r12
    push r13
    push r14
    mov r12, rdi
    mov r13, rsi
    lea rax, [r13 - 1]
    shl rax, 4
    mov rbx, qword ptr [r12 + rax]
    mov rbp, qword ptr [r12 + rax + 8]
    xor r14, r14
    xor r11, r11
.Lloop:
    lea rax, [r13 - 1]
    cmp r11, rax
    jge .Ldonepart
    mov rax, r11
    shl rax, 4
    mov rcx, qword ptr [r12 + rax]
    mov rdx, qword ptr [r12 + rax + 8]
    cmp rcx, rbx
    jl .Lswap
    jg .Lnoswap
    cmp rdx, rbp
    jge .Lnoswap
.Lswap:
    mov rsi, r14
    shl rsi, 4
    mov rdi, qword ptr [r12 + rsi]
    mov r8, qword ptr [r12 + rsi + 8]
    mov qword ptr [r12 + rax], rdi
    mov qword ptr [r12 + rax + 8], r8
    mov qword ptr [r12 + rsi], rcx
    mov qword ptr [r12 + rsi + 8], rdx
    inc r14
.Lnoswap:
    inc r11
    jmp .Lloop
.Ldonepart:
    lea rax, [r13 - 1]
    shl rax, 4
    mov rsi, r14
    shl rsi, 4
    mov rdi, qword ptr [r12 + rsi]
    mov r8, qword ptr [r12 + rsi + 8]
    mov qword ptr [r12 + rax], rdi
    mov qword ptr [r12 + rax + 8], r8
    mov qword ptr [r12 + rsi], rbx
    mov qword ptr [r12 + rsi + 8], rbp
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs
    lea rax, [r14 + 1]
    shl rax, 4
    lea rdi, [r12 + rax]
    mov rsi, r13
    sub rsi, r14
    dec rsi
    call qsort_pairs
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
.Lret0:
    xor eax, eax
    ret
