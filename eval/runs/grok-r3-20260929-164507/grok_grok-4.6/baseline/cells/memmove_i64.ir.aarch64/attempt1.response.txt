export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @done, @check
@check
	%dst_gt_src =w cugtl %dst, %src
	jnz %dst_gt_src, @bwdloop, @fwdloop
@fwdloop
	%i =l phi @check 0, @fwdbody %i1
	%i_lt_n =w csltl %i, %n
	jnz %i_lt_n, @fwdbody, @done
@fwdbody
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%dptr =l add %dst, %off
	%v =l loadl %sptr
	storel %v, %dptr
	%i1 =l add %i, 1
	jmp @fwdloop
@bwdloop
	%j =l phi @check %n, @bwdbody %j1
	%j_gt_0 =w csgtl %j, 0
	jnz %j_gt_0, @bwdbody, @done
@bwdbody
	%j1 =l sub %j, 1
	%off2 =l mul %j1, 8
	%sptr2 =l add %src, %off2
	%dptr2 =l add %dst, %off2
	%v2 =l loadl %sptr2
	storel %v2, %dptr2
	jmp @bwdloop
@done
	ret 0
}
