export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @goright %lo_r, @goleft %lo_l
	%hi =l phi @start %n, @goright %hi_r, @goleft %hi_l
	%cont =w csltl %lo, %hi
	jnz %cont, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%idx =l mul %mid, 8
	%addr =l add %a, %idx
	%val =l loadl %addr
	%iseq =w ceql %val, %key
	jnz %iseq, @found, @cmp
@cmp
	%islt =w csltl %val, %key
	jnz %islt, @goright, @goleft
@goright
	%lo_r =l add %mid, 1
	%hi_r =l copy %hi
	jmp @loop
@goleft
	%lo_l =l copy %lo
	%hi_l =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
