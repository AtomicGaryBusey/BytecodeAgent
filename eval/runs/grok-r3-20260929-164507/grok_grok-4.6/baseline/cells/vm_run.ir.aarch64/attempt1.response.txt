export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc_p =l alloc8 8
	%sp_p =l alloc8 8
	storel 0, %pc_p
	storel 0, %sp_p
	jmp @loop
@loop
	%pc =l loadl %pc_p
	%atend =w ceql %pc, %len
	jnz %atend, @halt, @fetch
@fetch
	%off =l mul %pc, 8
	%ip =l add %code, %off
	%op =l loadl %ip
	%ishalt =w ceql %op, 0
	jnz %ishalt, @halt, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @c2
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @c3
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @c4
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @c5
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @c6
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @c7
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @c8
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @c9
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @c10
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @c11
@c11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_other
@op_push
	%pc_p1 =l add %pc, 1
	%off_v =l mul %pc_p1, 8
	%vp =l add %code, %off_v
	%v =l loadl %vp
	%sp =l loadl %sp_p
	%off_s =l mul %sp, 8
	%stp =l add %stack, %off_s
	storel %v, %stp
	%spn =l add %sp, 1
	storel %spn, %sp_p
	%pcn =l add %pc, 2
	storel %pcn, %pc_p
	jmp @loop
@op_add
	%sp_a =l loadl %sp_p
	%sp_a1 =l sub %sp_a, 1
	%sp_a2 =l sub %sp_a, 2
	%off_b =l mul %sp_a1, 8
	%off_a =l mul %sp_a2, 8
	%ptr_b =l add %stack, %off_b
	%ptr_a =l add %stack, %off_a
	%b =l loadl %ptr_b
	%a =l loadl %ptr_a
	%sum =l add %a, %b
	storel %sum, %ptr_a
	storel %sp_a1, %sp_p
	%pc_a =l add %pc, 1
	storel %pc_a, %pc_p
	jmp @loop
@op_sub
	%sp_s =l loadl %sp_p
	%sp_s1 =l sub %sp_s, 1
	%sp_s2 =l sub %sp_s, 2
	%off_sb =l mul %sp_s1, 8
	%off_sa =l mul %sp_s2, 8
	%ptr_sb =l add %stack, %off_sb
	%ptr_sa =l add %stack, %off_sa
	%b_s =l loadl %ptr_sb
	%a_s =l loadl %ptr_sa
	%diff =l sub %a_s, %b_s
	storel %diff, %ptr_sa
	storel %sp_s1, %sp_p
	%pc_s =l add %pc, 1
	storel %pc_s, %pc_p
	jmp @loop
@op_mul
	%sp_m =l loadl %sp_p
	%sp_m1 =l sub %sp_m, 1
	%sp_m2 =l sub %sp_m, 2
	%off_mb =l mul %sp_m1, 8
	%off_ma =l mul %sp_m2, 8
	%ptr_mb =l add %stack, %off_mb
	%ptr_ma =l add %stack, %off_ma
	%b_m =l loadl %ptr_mb
	%a_m =l loadl %ptr_ma
	%prod =l mul %a_m, %b_m
	storel %prod, %ptr_ma
	storel %sp_m1, %sp_p
	%pc_m =l add %pc, 1
	storel %pc_m, %pc_p
	jmp @loop
@op_dup
	%sp_d =l loadl %sp_p
	%sp_d1 =l sub %sp_d, 1
	%off_d1 =l mul %sp_d1, 8
	%ptr_d1 =l add %stack, %off_d1
	%val_d =l loadl %ptr_d1
	%off_d =l mul %sp_d, 8
	%ptr_d =l add %stack, %off_d
	storel %val_d, %ptr_d
	%sp_dn =l add %sp_d, 1
	storel %sp_dn, %sp_p
	%pc_d =l add %pc, 1
	storel %pc_d, %pc_p
	jmp @loop
@op_swap
	%sp_sw =l loadl %sp_p
	%sp_sw1 =l sub %sp_sw, 1
	%sp_sw2 =l sub %sp_sw, 2
	%off_sw1 =l mul %sp_sw1, 8
	%off_sw2 =l mul %sp_sw2, 8
	%ptr_sw1 =l add %stack, %off_sw1
	%ptr_sw2 =l add %stack, %off_sw2
	%top_sw =l loadl %ptr_sw1
	%below_sw =l loadl %ptr_sw2
	storel %below_sw, %ptr_sw1
	storel %top_sw, %ptr_sw2
	%pc_sw =l add %pc, 1
	storel %pc_sw, %pc_p
	jmp @loop
@op_drop
	%sp_dr =l loadl %sp_p
	%sp_drn =l sub %sp_dr, 1
	storel %sp_drn, %sp_p
	%pc_dr =l add %pc, 1
	storel %pc_dr, %pc_p
	jmp @loop
@op_over
	%sp_o =l loadl %sp_p
	%sp_o2 =l sub %sp_o, 2
	%off_o2 =l mul %sp_o2, 8
	%ptr_o2 =l add %stack, %off_o2
	%val_o =l loadl %ptr_o2
	%off_o =l mul %sp_o, 8
	%ptr_o =l add %stack, %off_o
	storel %val_o, %ptr_o
	%sp_on =l add %sp_o, 1
	storel %sp_on, %sp_p
	%pc_o =l add %pc, 1
	storel %pc_o, %pc_p
	jmp @loop
@op_lt
	%sp_lt =l loadl %sp_p
	%sp_lt1 =l sub %sp_lt, 1
	%sp_lt2 =l sub %sp_lt, 2
	%off_ltb =l mul %sp_lt1, 8
	%off_lta =l mul %sp_lt2, 8
	%ptr_ltb =l add %stack, %off_ltb
	%ptr_lta =l add %stack, %off_lta
	%b_lt =l loadl %ptr_ltb
	%a_lt =l loadl %ptr_lta
	%cmp_lt =w csltl %a_lt, %b_lt
	%res_lt =l extuw %cmp_lt
	storel %res_lt, %ptr_lta
	storel %sp_lt1, %sp_p
	%pc_lt =l add %pc, 1
	storel %pc_lt, %pc_p
	jmp @loop
@op_jz
	%sp_jz =l loadl %sp_p
	%sp_jz1 =l sub %sp_jz, 1
	storel %sp_jz1, %sp_p
	%off_jz =l mul %sp_jz1, 8
	%ptr_jz =l add %stack, %off_jz
	%v_jz =l loadl %ptr_jz
	%z_jz =w ceql %v_jz, 0
	jnz %z_jz, @jz_yes, @jz_no
@jz_yes
	%pc_jz1 =l add %pc, 1
	%off_jzt =l mul %pc_jz1, 8
	%ptr_jzt =l add %code, %off_jzt
	%t_jz =l loadl %ptr_jzt
	storel %t_jz, %pc_p
	jmp @loop
@jz_no
	%pc_jz2 =l add %pc, 2
	storel %pc_jz2, %pc_p
	jmp @loop
@op_jmp
	%pc_j1 =l add %pc, 1
	%off_jt =l mul %pc_j1, 8
	%ptr_jt =l add %code, %off_jt
	%t_j =l loadl %ptr_jt
	storel %t_j, %pc_p
	jmp @loop
@op_other
	%pc_ot =l add %pc, 1
	storel %pc_ot, %pc_p
	jmp @loop
@halt
	%sp_h =l loadl %sp_p
	jmp @zchk
@zchk
	%zi =l phi @halt %sp_h, @zbody %zi_n
	%zdone =w csgel %zi, %cap
	jnz %zdone, @doret, @zbody
@zbody
	%zoff =l mul %zi, 8
	%zptr =l add %stack, %zoff
	storel 0, %zptr
	%zi_n =l add %zi, 1
	jmp @zchk
@doret
	%empty =w ceql %sp_h, 0
	jnz %empty, @ret0, @rettop
@ret0
	ret 0
@rettop
	%topi =l sub %sp_h, 1
	%topoff =l mul %topi, 8
	%topptr =l add %stack, %topoff
	%topv =l loadl %topptr
	ret %topv
}
