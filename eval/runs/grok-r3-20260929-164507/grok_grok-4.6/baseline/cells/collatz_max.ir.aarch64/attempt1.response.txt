export function l $collatz_max(l %n) {
@start
	jmp @loopk
@loopk
	%k =l phi @start 1, @nextk %knext
	%max =l phi @start 0, @nextk %max2
	%cmpk =w cslel %k, %n
	jnz %cmpk, @loopc, @done
@loopc
	%x =l phi @loopk %k, @even %xeven, @odd %xodd
	%steps =l phi @loopk 0, @even %steps1, @odd %steps1
	%is1 =w ceql %x, 1
	jnz %is1, @gotsteps, @contc
@contc
	%steps1 =l add %steps, 1
	%bit =l and %x, 1
	%isodd =w cnel %bit, 0
	jnz %isodd, @odd, @even
@even
	%xeven =l shr %x, 1
	jmp @loopc
@odd
	%tmp =l mul %x, 3
	%xodd =l add %tmp, 1
	jmp @loopc
@gotsteps
	%gt =w csgtl %steps, %max
	jnz %gt, @newmax, @oldmax
@newmax
	jmp @nextk
@oldmax
	jmp @nextk
@nextk
	%max2 =l phi @newmax %steps, @oldmax %max
	%knext =l add %k, 1
	jmp @loopk
@done
	ret %max
}
