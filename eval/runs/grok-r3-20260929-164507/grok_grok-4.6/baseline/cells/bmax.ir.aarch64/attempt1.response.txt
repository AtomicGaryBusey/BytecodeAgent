export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @cont %inext
	%max =l phi @start %max0, @cont %maxnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%idx =l shl %i, 3
	%ptr =l add %a, %idx
	%val =l loadl %ptr
	%gt =w csgtl %val, %max
	jnz %gt, @update, @cont
@update
	jmp @cont
@cont
	%maxnext =l phi @update %val, @body %max
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
