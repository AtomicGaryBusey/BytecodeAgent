export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%idx =l mul %i, 8
	%sp =l add %src, %idx
	%v =l loadl %sp
	%r =l call %f(l %v)
	%dp =l add %dst, %idx
	storel %r, %dp
	%lt =w csltl %r, %v
	%lt64 =l extuw %lt
	%cnt1 =l add %cnt, %lt64
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
