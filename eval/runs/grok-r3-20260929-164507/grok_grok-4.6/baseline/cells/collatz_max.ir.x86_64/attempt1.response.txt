export function l $collatz_max(l %n) {
@start
	jmp @kcond
@kcond
	%k =l phi @start 1, @kinc %knext
	%max =l phi @start 0, @kinc %maxnext
	%kgt =w csgtl %k, %n
	jnz %kgt, @done, @xcond
@xcond
	%x =l phi @kcond %k, @xinc %xnext
	%steps =l phi @kcond 0, @xinc %stepsnext
	%xeq1 =w ceql %x, 1
	jnz %xeq1, @xdone, @xbody
@xbody
	%and1 =l and %x, 1
	%isodd =w ceql %and1, 1
	jnz %isodd, @odd, @even
@even
	%xeven =l div %x, 2
	jmp @xinc
@odd
	%xmul =l mul %x, 3
	%xodd =l add %xmul, 1
	jmp @xinc
@xinc
	%xnext =l phi @even %xeven, @odd %xodd
	%stepsnext =l add %steps, 1
	jmp @xcond
@xdone
	%bigger =w csgtl %steps, %max
	jnz %bigger, @newmax, @kinc
@newmax
	jmp @kinc
@kinc
	%maxnext =l phi @newmax %steps, @xdone %max
	%knext =l add %k, 1
	jmp @kcond
@done
	ret %max
}
