export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%iszero =w ceql %b, 0
	jnz %iszero, @divzero, @chkoverflow
@divzero
	storel %a, %rem
	ret 0
@chkoverflow
	%min =l shl 1, 63
	%ismin =w ceql %a, %min
	jnz %ismin, @chkm1, @dodiv
@chkm1
	%isneg1 =w ceql %b, -1
	jnz %isneg1, @overflow, @dodiv
@overflow
	storel 0, %rem
	ret %min
@dodiv
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
