export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%op =l phi @start %out, @next %op2
	%ge =w csgel %i, %n
	jnz %ge, @done, @body
@body
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%gt1 =w csgtl %rem, 1
	jnz %gt1, @ld1, @nld1
@ld1
	%p1 =l add %p0, 1
	%b1x =w loadub %p1
	jmp @aft1
@nld1
	jmp @aft1
@aft1
	%b1 =w phi @ld1 %b1x, @nld1 0
	%gt2 =w csgtl %rem, 2
	jnz %gt2, @ld2, @nld2
@ld2
	%p2 =l add %p0, 2
	%b2x =w loadub %p2
	jmp @aft2
@nld2
	jmp @aft2
@aft2
	%b2 =w phi @ld2 %b2x, @nld2 0
	%sh0 =w shl %b0, 16
	%sh1 =w shl %b1, 8
	%or01 =w or %sh0, %sh1
	%triple =w or %or01, %b2
	jmp @chars
@chars
	%k =w phi @aft2 0, @gotc %k1
	%shift =w phi @aft2 18, @gotc %shift1
	%shv =w shr %triple, %shift
	%idx =w and %shv, 63
	%kl =l extuw %k
	%needpad =w csgtl %kl, %rem
	jnz %needpad, @pad, @enc
@pad
	%chpad =w copy 61
	jmp @gotc
@enc
	%lt26 =w csltw %idx, 26
	jnz %lt26, @eA, @nA
@eA
	%chA =w add %idx, 65
	jmp @gotc
@nA
	%lt52 =w csltw %idx, 52
	jnz %lt52, @ea, @na
@ea
	%cha =w add %idx, 71
	jmp @gotc
@na
	%lt62 =w csltw %idx, 62
	jnz %lt62, @e0, @n0
@e0
	%ch0 =w sub %idx, 4
	jmp @gotc
@n0
	%is62 =w ceqw %idx, 62
	jnz %is62, @eplus, @eslash
@eplus
	%chplus =w copy 43
	jmp @gotc
@eslash
	%chslash =w copy 47
	jmp @gotc
@gotc
	%ch =w phi @pad %chpad, @eA %chA, @ea %cha, @e0 %ch0, @eplus %chplus, @eslash %chslash
	%dp =l add %op, %kl
	storeb %ch, %dp
	%k1 =w add %k, 1
	%shift1 =w sub %shift, 6
	%kdone =w ceqw %k1, 4
	jnz %kdone, @next, @chars
@next
	%i2 =l add %i, 3
	%op2 =l add %op, 4
	jmp @loop
@done
	%retv =l sub %op, %out
	ret %retv
}
