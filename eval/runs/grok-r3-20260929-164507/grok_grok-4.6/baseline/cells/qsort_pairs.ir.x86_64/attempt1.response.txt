export function l $qsort_pairs(l %a, l %n) {
@entry
	%nle1 =w cslel %n, 1
	jnz %nle1, @done, @doit
@done
	ret 0
@doit
	%nm1 =l sub %n, 1
	%offp =l mul %nm1, 16
	%pptr =l add %a, %offp
	%pkey =l loadl %pptr
	%pptrv =l add %pptr, 8
	%pval =l loadl %pptrv
	jmp @loop
@loop
	%i =l phi @doit 0, @cont %i2
	%j =l phi @doit 0, @cont %j2
	%jcmp =w csltl %j, %nm1
	jnz %jcmp, @body, @endloop
@body
	%offj =l mul %j, 16
	%jptr =l add %a, %offj
	%jkey =l loadl %jptr
	%jptrv =l add %jptr, 8
	%jval =l loadl %jptrv
	%klt =w csltl %jkey, %pkey
	jnz %klt, @swap, @checkeq
@checkeq
	%keq =w ceql %jkey, %pkey
	jnz %keq, @checkval, @cont
@checkval
	%vle =w cslel %jval, %pval
	jnz %vle, @swap, @cont
@swap
	%offi =l mul %i, 16
	%iptr =l add %a, %offi
	%ikey =l loadl %iptr
	%iptrv =l add %iptr, 8
	%ival =l loadl %iptrv
	storel %jkey, %iptr
	storel %jval, %iptrv
	storel %ikey, %jptr
	storel %ival, %jptrv
	%i1 =l add %i, 1
	jmp @cont
@cont
	%i2 =l phi @swap %i1, @checkeq %i, @checkval %i
	%j2 =l add %j, 1
	jmp @loop
@endloop
	%offi2 =l mul %i, 16
	%iptr2 =l add %a, %offi2
	%ikey2 =l loadl %iptr2
	%iptrv2 =l add %iptr2, 8
	%ival2 =l loadl %iptrv2
	storel %pkey, %iptr2
	storel %pval, %iptrv2
	storel %ikey2, %pptr
	storel %ival2, %pptrv
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%offr =l mul %ip1, 16
	%ar =l add %a, %offr
	%nr =l sub %n, %ip1
	%r2 =l call $qsort_pairs(l %ar, l %nr)
	ret 0
}
