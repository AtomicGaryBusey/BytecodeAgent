export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%p =l phi @start %a, @body %p1
	%cnt =l phi @start %n, @body %cnt1
	%sum =l phi @start 0, @body %sum1
	%cond =w cnel %cnt, 0
	jnz %cond, @body, @done
@body
	%val =l loadl %p
	%res =l call %f(l %val)
	%sum1 =l add %sum, %res
	%p1 =l add %p, 8
	%cnt1 =l sub %cnt, 1
	jmp @loop
@done
	ret %sum
}
