push rbx
push rbp
push r12
push r13
push r14
push r15
mov r15, rdi
mov r12, rsi
mov rbx, rdx
mov r13, rcx
mov r14, r8
mov rbp, r12
imul rbp, r12
xor r8, r8
.Lident_i:
xor r9, r9
.Lident_j:
xor eax, eax
cmp r8, r9
sete al
mov rdx, r8
imul rdx, r12
add rdx, r9
mov qword ptr [r13 + rdx*8], rax
inc r9
cmp r9, r12
jl .Lident_j
inc r8
cmp r8, r12
jl .Lident_i
test rbx, rbx
jz .Lzero_tmp
xor ecx, ecx
.Lcopy_a:
mov rax, qword ptr [r15 + rcx*8]
mov qword ptr [r14 + rcx*8], rax
inc rcx
cmp rcx, rbp
jl .Lcopy_a
.Lpow_loop:
test rbx, rbx
jz .Lzero_tmp
test rbx, 1
jz .Lsquare
lea rdi, [r14 + rbp*8]
mov rsi, r13
mov rdx, r14
lea r15, [rip + .Lafter_mul1]
jmp .Lmatmul
.Lafter_mul1:
xor ecx, ecx
lea r11, [r14 + rbp*8]
.Lcopy_out:
mov rax, qword ptr [r11 + rcx*8]
mov qword ptr [r13 + rcx*8], rax
inc rcx
cmp rcx, rbp
jl .Lcopy_out
.Lsquare:
lea rdi, [r14 + rbp*8]
mov rsi, r14
mov rdx, r14
lea r15, [rip + .Lafter_mul2]
jmp .Lmatmul
.Lafter_mul2:
xor ecx, ecx
lea r11, [r14 + rbp*8]
.Lcopy_base:
mov rax, qword ptr [r11 + rcx*8]
mov qword ptr [r14 + rcx*8], rax
inc rcx
cmp rcx, rbp
jl .Lcopy_base
shr rbx, 1
jmp .Lpow_loop
.Lmatmul:
xor r8, r8
.Lmul_i:
xor r9, r9
.Lmul_j:
xor r10, r10
xor ecx, ecx
.Lmul_t:
mov r11, r8
imul r11, r12
add r11, rcx
mov rax, qword ptr [rsi + r11*8]
mov r11, rcx
imul r11, r12
add r11, r9
imul rax, qword ptr [rdx + r11*8]
add r10, rax
inc rcx
cmp rcx, r12
jl .Lmul_t
mov r11, r8
imul r11, r12
add r11, r9
mov qword ptr [rdi + r11*8], r10
inc r9
cmp r9, r12
jl .Lmul_j
inc r8
cmp r8, r12
jl .Lmul_i
jmp r15
.Lzero_tmp:
xor ecx, ecx
lea rdx, [rbp + rbp]
.Lzero:
mov qword ptr [r14 + rcx*8], 0
inc rcx
cmp rcx, rdx
jl .Lzero
xor eax, eax
