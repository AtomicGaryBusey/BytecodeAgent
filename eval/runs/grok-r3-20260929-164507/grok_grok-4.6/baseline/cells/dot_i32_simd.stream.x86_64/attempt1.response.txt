dot_i32_simd:
    xor rax, rax
    pxor xmm4, xmm4
    cmp rdx, 4
    jl .Ltail
.Lloop:
    movdqu xmm0, xmmword ptr [rdi]
    movdqu xmm1, xmmword ptr [rsi]
    pshufd xmm2, xmm0, 0xEE
    pshufd xmm3, xmm1, 0xEE
    pmovsxdq xmm0, xmm0
    pmovsxdq xmm1, xmm1
    pmovsxdq xmm2, xmm2
    pmovsxdq xmm3, xmm3
    pmuldq xmm0, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm0
    paddq xmm4, xmm2
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jge .Lloop
    movdqa xmm0, xmm4
    psrldq xmm0, 8
    paddq xmm4, xmm0
    movq rax, xmm4
.Ltail:
    test rdx, rdx
    jz .Ldone
.Lscalar:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    sub rdx, 1
    jnz .Lscalar
.Ldone:
    ret
