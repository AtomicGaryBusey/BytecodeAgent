export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @aftermax %i1
	%min =l phi @start 0x7fffffffffffffff, @aftermax %min1
	%max =l phi @start 0x8000000000000000, @aftermax %max1
	%sum =l phi @start 0, @aftermax %sum1
	%cnt =l phi @start 0, @aftermax %cnt1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%cnt1 =l add %cnt, %odd
	%i1 =l add %i, 1
	%lt =w csltl %y, %min
	jnz %lt, @ismin, @notmin
@ismin
	jmp @aftermin
@notmin
	jmp @aftermin
@aftermin
	%min1 =l phi @ismin %y, @notmin %min
	%gt =w csgtl %y, %max
	jnz %gt, @ismax, @notmax
@ismax
	jmp @aftermax
@notmax
	jmp @aftermax
@aftermax
	%max1 =l phi @ismax %y, @notmax %max
	jmp @loop
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
