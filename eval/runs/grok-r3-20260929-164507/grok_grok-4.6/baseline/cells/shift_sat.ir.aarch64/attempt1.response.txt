export function l $shift_sat(l %x, l %n) {
@start
	jmp @loop
@loop
	%left =l phi @start %x, @body %left2
	%right =l phi @start %x, @body %right2
	%i =l phi @start 0, @body %i2
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%left2 =l shl %left, 1
	%right2 =l sar %right, 1
	%i2 =l add %i, 1
	jmp @loop
@done
	%res =l add %left, %right
	ret %res
}
