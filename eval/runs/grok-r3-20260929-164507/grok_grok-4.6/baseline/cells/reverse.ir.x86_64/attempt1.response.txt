export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%j =l phi @start %j0, @body %j1
	%cmp =w csltl %i, %j
	jnz %cmp, @body, @done
@body
	%iscale =l mul %i, 8
	%pi =l add %a, %iscale
	%jscale =l mul %j, 8
	%pj =l add %a, %jscale
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@done
	ret 0
}
