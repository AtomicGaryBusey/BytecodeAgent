export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @do_repeat %i2, @copy_done %j
	%wr =l phi @start 0, @do_repeat %wr2, @copy_done %wr3
	%cmpi =w csltl %i, %n
	jnz %cmpi, @body, @done
@body
	%ba =l add %src, %i
	%b =w loadub %ba
	jmp @rloop
@rloop
	%r =w phi @body 1, @rinc %r1
	%rl128 =w csltw %r, 128
	jnz %rl128, @rckn, @rdone
@rckn
	%rel =l extuw %r
	%ix =l add %i, %rel
	%ixn =w csltl %ix, %n
	jnz %ixn, @rckb, @rdone
@rckb
	%ra =l add %src, %ix
	%rv =w loadub %ra
	%eq =w ceqw %rv, %b
	jnz %eq, @rinc, @rdone
@rinc
	%r1 =w add %r, 1
	jmp @rloop
@rdone
	%rge3 =w csgew %r, 3
	jnz %rge3, @erep, @lscan
@erep
	%nd2 =l add %wr, 2
	%ok2 =w cslel %nd2, %cap
	jnz %ok2, @do_repeat, @ovf
@do_repeat
	%rh =w sub 257, %r
	%op =l add %out, %wr
	storeb %rh, %op
	%op1 =l add %op, 1
	storeb %b, %op1
	%wr2 =l add %wr, 2
	%rl2 =l extuw %r
	%i2 =l add %i, %rl2
	jmp @loop
@lscan
	%j =l phi @rdone %i, @ladv %jn
	%L =w phi @rdone 0, @ladv %L1
	%jln =w csltl %j, %n
	jnz %jln, @lckL, @lemit
@lckL
	%Ll =w csltw %L, 128
	jnz %Ll, @lckr, @lemit
@lckr
	%ja =l add %src, %j
	%bj =w loadub %ja
	jmp @jrloop
@jrloop
	%rj =w phi @lckr 1, @jrinc %rj1
	%rjl =w csltw %rj, 128
	jnz %rjl, @jrckn, @jrdone
@jrckn
	%rjel =l extuw %rj
	%jix =l add %j, %rjel
	%jixn =w csltl %jix, %n
	jnz %jixn, @jrckb, @jrdone
@jrckb
	%jra =l add %src, %jix
	%jrv =w loadub %jra
	%jeq =w ceqw %jrv, %bj
	jnz %jeq, @jrinc, @jrdone
@jrinc
	%rj1 =w add %rj, 1
	jmp @jrloop
@jrdone
	%rjge3 =w csgew %rj, 3
	jnz %rjge3, @lemit, @ladv
@ladv
	%jn =l add %j, 1
	%L1 =w add %L, 1
	jmp @lscan
@lemit
	%Ll64 =l extuw %L
	%need =l add %Ll64, 1
	%tot =l add %wr, %need
	%okL =w cslel %tot, %cap
	jnz %okL, @do_lit, @ovf
@do_lit
	%lh =w sub %L, 1
	%lop =l add %out, %wr
	storeb %lh, %lop
	jmp @cpy
@cpy
	%k =w phi @do_lit 0, @cpyinc %k1
	%kL =w csltw %k, %L
	jnz %kL, @cpy1, @copy_done
@cpy1
	%kl =l extuw %k
	%sa0 =l add %i, %kl
	%sa =l add %src, %sa0
	%cb =w loadub %sa
	%da0 =l add %lop, 1
	%da =l add %da0, %kl
	storeb %cb, %da
	jmp @cpyinc
@cpyinc
	%k1 =w add %k, 1
	jmp @cpy
@copy_done
	%wr3 =l add %wr, %need
	jmp @loop
@ovf
	ret -1
@done
	ret %wr
}
