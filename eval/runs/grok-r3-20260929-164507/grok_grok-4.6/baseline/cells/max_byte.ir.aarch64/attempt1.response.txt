export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i1
	%max =l phi @start 0, @cont %max1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @update, @cont
@update
	jmp @cont
@cont
	%max1 =l phi @body %max, @update %b
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %max
}
