export function l $max_byte(l %p, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @retzero, @loop
@retzero
	ret 0
@loop
	%i =l phi @start 0, @merge %inext
	%max =l phi @start 0, @merge %maxnext
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%isgreater =w cugtl %b, %max
	jnz %isgreater, @newmax, @keepmax
@newmax
	jmp @merge
@keepmax
	jmp @merge
@merge
	%maxnext =l phi @newmax %b, @keepmax %max
	%inext =l add %i, 1
	%more =w csltl %inext, %n
	jnz %more, @loop, @done
@done
	ret %maxnext
}
