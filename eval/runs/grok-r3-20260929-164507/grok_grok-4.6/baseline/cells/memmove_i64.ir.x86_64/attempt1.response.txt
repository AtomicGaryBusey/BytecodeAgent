export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nbytes =l mul %n, 8
	%cmplt =w cultl %dst, %src
	jnz %cmplt, @fwd, @bwd
@fwd
	%fend =l add %src, %nbytes
	jmp @fwdloop
@fwdloop
	%fs =l phi @fwd %src, @fwdbody %fs1
	%fd =l phi @fwd %dst, @fwdbody %fd1
	%fdone =w ceql %fs, %fend
	jnz %fdone, @ret, @fwdbody
@fwdbody
	%fval =l loadl %fs
	storel %fval, %fd
	%fs1 =l add %fs, 8
	%fd1 =l add %fd, 8
	jmp @fwdloop
@bwd
	%bs =l add %src, %nbytes
	%bd =l add %dst, %nbytes
	jmp @bwdloop
@bwdloop
	%bs0 =l phi @bwd %bs, @bwdbody %bs1
	%bd0 =l phi @bwd %bd, @bwdbody %bd1
	%bdone =w ceql %bs0, %src
	jnz %bdone, @ret, @bwdbody
@bwdbody
	%bs1 =l sub %bs0, 8
	%bd1 =l sub %bd0, 8
	%bval =l loadl %bs1
	storel %bval, %bd1
	jmp @bwdloop
@ret
	ret 0
}
