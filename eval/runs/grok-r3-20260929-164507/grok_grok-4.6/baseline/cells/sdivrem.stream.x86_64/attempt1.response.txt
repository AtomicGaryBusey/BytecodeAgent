sdivrem:
    mov rcx, rdx
    test rsi, rsi
    jz .Ldiv0
    cmp rsi, -1
    je .Ldivneg1
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Ldivneg1:
    mov rax, rdi
    neg rax
    mov qword ptr [rcx], 0
    ret
.Ldiv0:
    xor eax, eax
    mov qword ptr [rcx], rdi
    ret
