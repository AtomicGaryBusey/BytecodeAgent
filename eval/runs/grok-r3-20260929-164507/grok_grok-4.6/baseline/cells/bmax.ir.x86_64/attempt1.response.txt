export function l $bmax(l %a, l %n) {
@start
	%init =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %ni
	%m =l phi @start %init, @next %nm
	%t =w csltl %i, %n
	jnz %t, @body, @end
@body
	%bytes =l mul %i, 8
	%p =l add %a, %bytes
	%e =l loadl %p
	%b =w csltl %m, %e
	jnz %b, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%nm =l phi @upd %e, @keep %m
	%ni =l add %i, 1
	jmp @loop
@end
	ret %m
}
