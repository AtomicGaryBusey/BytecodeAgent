export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @ift %mid, @iff %lo2
	%hi =l phi @start 4294967296, @ift %hi2, @iff %mid
	%loplus =l add %lo, 1
	%cont =w cultl %loplus, %hi
	jnz %cont, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @ift, @iff
@ift
	%hi2 =l copy %hi
	jmp @loop
@iff
	%lo2 =l copy %lo
	jmp @loop
@done
	ret %lo
}
