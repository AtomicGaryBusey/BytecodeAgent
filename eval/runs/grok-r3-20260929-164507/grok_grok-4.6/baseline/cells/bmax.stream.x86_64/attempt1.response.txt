bmax:
	mov rax, qword ptr [rdi]
	cmp rsi, 1
	je .Ldone
	cmp rsi, 4
	jb .Lsmall
	mov rdx, qword ptr [rdi + 8]
	mov r8, qword ptr [rdi + 16]
	mov r9, qword ptr [rdi + 24]
	mov rcx, 4
	lea r10, [rsi - 3]
	cmp rcx, r10
	jge .Ltail
.Lloop:
	mov r11, qword ptr [rdi + rcx*8]
	cmp r11, rax
	cmovg rax, r11
	mov r11, qword ptr [rdi + rcx*8 + 8]
	cmp r11, rdx
	cmovg rdx, r11
	mov r11, qword ptr [rdi + rcx*8 + 16]
	cmp r11, r8
	cmovg r8, r11
	mov r11, qword ptr [rdi + rcx*8 + 24]
	cmp r11, r9
	cmovg r9, r11
	add rcx, 4
	cmp rcx, r10
	jl .Lloop
.Ltail:
	cmp rcx, rsi
	jge .Lreduce
	mov r11, qword ptr [rdi + rcx*8]
	cmp r11, rax
	cmovg rax, r11
	inc rcx
	jmp .Ltail
.Lreduce:
	cmp rdx, rax
	cmovg rax, rdx
	cmp r8, rax
	cmovg rax, r8
	cmp r9, rax
	cmovg rax, r9
	ret
.Lsmall:
	mov rdx, qword ptr [rdi + 8]
	cmp rdx, rax
	cmovg rax, rdx
	cmp rsi, 3
	jb .Ldone
	mov rdx, qword ptr [rdi + 16]
	cmp rdx, rax
	cmovg rax, rdx
.Ldone:
	ret
