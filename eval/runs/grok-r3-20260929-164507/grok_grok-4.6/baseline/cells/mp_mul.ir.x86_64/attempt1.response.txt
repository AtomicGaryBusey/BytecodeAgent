export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%mask =l copy 4294967295
	jmp @zeroinit
@zeroinit
	%zi =l phi @start 0, @zerobody %zinext
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zerobody, @outer
@zerobody
	%zoff =l mul %zi, 8
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zinext =l add %zi, 1
	jmp @zeroinit
@outer
	%i =l phi @zeroinit 0, @innerdone %inext
	%icmp =w csltl %i, %na
	jnz %icmp, @outerbody, @trimloop
@outerbody
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%aval =l loadl %aptr
	jmp @inner
@inner
	%j =l phi @outerbody 0, @innerbody %jnext
	%carry =l phi @outerbody 0, @innerbody %hi4
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @innerbody, @innerdone
@innerbody
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bval =l loadl %bptr
	%xlo =l and %aval, %mask
	%xhi =l shr %aval, 32
	%ylo =l and %bval, %mask
	%yhi =l shr %bval, 32
	%p0 =l mul %xlo, %ylo
	%p1 =l mul %xlo, %yhi
	%p2 =l mul %xhi, %ylo
	%p3 =l mul %xhi, %yhi
	%p1shl =l shl %p1, 32
	%p1shr =l shr %p1, 32
	%lo1 =l add %p0, %p1shl
	%cw1 =w cultl %lo1, %p0
	%cl1 =l extuw %cw1
	%hi1 =l add %p1shr, %cl1
	%p2shl =l shl %p2, 32
	%p2shr =l shr %p2, 32
	%lo2 =l add %lo1, %p2shl
	%cw2 =w cultl %lo2, %lo1
	%cl2 =l extuw %cw2
	%hi2a =l add %hi1, %p2shr
	%hi2 =l add %hi2a, %cl2
	%prodhi =l add %hi2, %p3
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rptr =l add %r, %roff
	%rval =l loadl %rptr
	%lo3 =l add %lo2, %rval
	%cw3 =w cultl %lo3, %lo2
	%cl3 =l extuw %cw3
	%hi3 =l add %prodhi, %cl3
	%lo4 =l add %lo3, %carry
	%cw4 =w cultl %lo4, %lo3
	%cl4 =l extuw %cw4
	%hi4 =l add %hi3, %cl4
	storel %lo4, %rptr
	%jnext =l add %j, 1
	jmp @inner
@innerdone
	%inb =l add %i, %nb
	%coff =l mul %inb, 8
	%cptr =l add %r, %coff
	storel %carry, %cptr
	%inext =l add %i, 1
	jmp @outer
@trimloop
	%k =l phi @outer %n, @trimcheck %k1
	%k0 =w ceql %k, 0
	jnz %k0, @ret0, @trimcheck
@trimcheck
	%k1 =l sub %k, 1
	%koff =l mul %k1, 8
	%kptr =l add %r, %koff
	%kval =l loadl %kptr
	%kz =w ceql %kval, 0
	jnz %kz, @trimloop, @retk
@retk
	ret %k
@ret0
	ret 0
}
