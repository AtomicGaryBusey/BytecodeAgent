export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @nop %inew_nop, @lit_commit %inew_l, @rep_commit %inew_r
	%o =l phi @start 0, @nop %o_nop, @lit_commit %onew_l, @rep_commit %onew_r
	%hasmore =w csltl %i, %n
	jnz %hasmore, @havehdr, @done
@havehdr
	%hdrptr =l add %src, %i
	%h =w loadsb %hdrptr
	%remain =l sub %n, %i
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @notnop
@nop
	%inew_nop =l add %i, 1
	%o_nop =l copy %o
	jmp @loop
@notnop
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%h64l =l extsw %h
	%needl =l add %h64l, 2
	%oksrcl =w cslel %needl, %remain
	jnz %oksrcl, @lit_ok, @errsrc
@lit_ok
	%litlen =l add %h64l, 1
	%spacel =l sub %cap, %o
	%truncl =w csgtl %litlen, %spacel
	jnz %truncl, @lit_tr, @lit_fu
@lit_tr
	%lcopy_a =l copy %spacel
	%lflag_a =w copy 1
	jmp @lit_go
@lit_fu
	%lcopy_b =l copy %litlen
	%lflag_b =w copy 0
	jmp @lit_go
@lit_go
	%lcopy =l phi @lit_tr %lcopy_a, @lit_fu %lcopy_b
	%lflag =w phi @lit_tr %lflag_a, @lit_fu %lflag_b
	%lsrc0 =l add %i, 1
	%lsrcp0 =l add %src, %lsrc0
	%ldstp0 =l add %out, %o
	jmp @lcloop
@lcloop
	%lsrcp =l phi @lit_go %lsrcp0, @lcbody %lsrcp1
	%ldstp =l phi @lit_go %ldstp0, @lcbody %ldstp1
	%lcnt =l phi @lit_go %lcopy, @lcbody %lcnt1
	%lmore =w csgtl %lcnt, 0
	jnz %lmore, @lcbody, @lcdone
@lcbody
	%lb =w loadub %lsrcp
	storeb %lb, %ldstp
	%lsrcp1 =l add %lsrcp, 1
	%ldstp1 =l add %ldstp, 1
	%lcnt1 =l sub %lcnt, 1
	jmp @lcloop
@lcdone
	jnz %lflag, @errcap, @lit_commit
@lit_commit
	%inew_l =l add %i, %needl
	%onew_l =l add %o, %litlen
	jmp @loop
@repeat
	%oksrr =w cslel 2, %remain
	jnz %oksrr, @rep_ok, @errsrc
@rep_ok
	%h64r =l extsw %h
	%replen =l sub 1, %h64r
	%spacer =l sub %cap, %o
	%truncr =w csgtl %replen, %spacer
	%bpos =l add %i, 1
	%bptr =l add %src, %bpos
	%rb =w loadub %bptr
	jnz %truncr, @rep_tr, @rep_fu
@rep_tr
	%rcopy_a =l copy %spacer
	%rflag_a =w copy 1
	jmp @rep_go
@rep_fu
	%rcopy_b =l copy %replen
	%rflag_b =w copy 0
	jmp @rep_go
@rep_go
	%rcopy =l phi @rep_tr %rcopy_a, @rep_fu %rcopy_b
	%rflag =w phi @rep_tr %rflag_a, @rep_fu %rflag_b
	%rdstp0 =l add %out, %o
	jmp @rcloop
@rcloop
	%rdstp =l phi @rep_go %rdstp0, @rcbody %rdstp1
	%rcnt =l phi @rep_go %rcopy, @rcbody %rcnt1
	%rmore =w csgtl %rcnt, 0
	jnz %rmore, @rcbody, @rcdone
@rcbody
	storeb %rb, %rdstp
	%rdstp1 =l add %rdstp, 1
	%rcnt1 =l sub %rcnt, 1
	jmp @rcloop
@rcdone
	jnz %rflag, @errcap, @rep_commit
@rep_commit
	%inew_r =l add %i, 2
	%onew_r =l add %o, %replen
	jmp @loop
@errsrc
	ret -2
@errcap
	ret -1
@done
	ret %o
}
