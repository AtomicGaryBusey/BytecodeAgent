max_byte:
    xor eax, eax
    test rsi, rsi
    jz .Ldone
    cmp rsi, 16
    jb .Lscalar
    movdqu xmm0, xmmword ptr [rdi + rsi - 16]
    cmp rsi, 16
    jbe .Lreduce
.Lsimd:
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    cmp rsi, 16
    ja .Lsimd
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    pextrb eax, xmm0, 0
    ret
.Lscalar:
    movzx edx, byte ptr [rdi]
    cmp edx, eax
    cmova eax, edx
    inc rdi
    dec rsi
    jnz .Lscalar
.Ldone:
    ret
