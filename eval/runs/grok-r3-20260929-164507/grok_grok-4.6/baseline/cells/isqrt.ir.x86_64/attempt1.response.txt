export function l $isqrt(l %n) {
@start
	%bit0 =l shl 1, 62
	jmp @loop
@loop
	%num =l phi @start %n, @cont %num2
	%res =l phi @start 0, @cont %res2
	%bit =l phi @start %bit0, @cont %bit2
	%nz =w cnel %bit, 0
	jnz %nz, @body, @done
@body
	%sum =l add %res, %bit
	%ge =w cugel %num, %sum
	jnz %ge, @take, @skip
@take
	%numt =l sub %num, %sum
	%rh =l shr %res, 1
	%rest =l add %rh, %bit
	jmp @cont
@skip
	%ress =l shr %res, 1
	jmp @cont
@cont
	%num2 =l phi @take %numt, @skip %num
	%res2 =l phi @take %rest, @skip %ress
	%bit2 =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
