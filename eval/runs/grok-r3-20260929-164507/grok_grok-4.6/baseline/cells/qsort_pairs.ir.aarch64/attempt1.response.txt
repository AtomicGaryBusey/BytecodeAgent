export function l $qsort_pairs(l %a, l %n) {
@start
	%nlt2 =w csltl %n, 2
	jnz %nlt2, @ret0, @doit
@ret0
	ret 0
@doit
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%paddr =l add %a, %poff
	%pk =l loadl %paddr
	%pvaddr =l add %paddr, 8
	%pv =l loadl %pvaddr
	%i0 =l sub 0, 1
	%j0 =l copy 0
	jmp @loop
@loop
	%i =l phi @doit %i0, @next %i2
	%j =l phi @doit %j0, @next %j1
	%jlt =w csltl %j, %nm1
	jnz %jlt, @body, @after
@body
	%joff =l mul %j, 16
	%jaddr =l add %a, %joff
	%jk =l loadl %jaddr
	%jvaddr =l add %jaddr, 8
	%jv =l loadl %jvaddr
	%klt =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vlt =w csltl %jv, %pv
	%both =w and %keq, %vlt
	%less =w or %klt, %both
	jnz %less, @swap, @noswap
@swap
	%i1 =l add %i, 1
	%ioff =l mul %i1, 16
	%iaddr =l add %a, %ioff
	%ik =l loadl %iaddr
	%ivaddr =l add %iaddr, 8
	%iv =l loadl %ivaddr
	storel %jk, %iaddr
	storel %jv, %ivaddr
	storel %ik, %jaddr
	storel %iv, %jvaddr
	jmp @next
@noswap
	%i1ns =l copy %i
	jmp @next
@next
	%i2 =l phi @swap %i1, @noswap %i1ns
	%j1 =l add %j, 1
	jmp @loop
@after
	%ip =l add %i, 1
	%ipoff =l mul %ip, 16
	%ipaddr =l add %a, %ipoff
	%ipk =l loadl %ipaddr
	%ipvaddr =l add %ipaddr, 8
	%ipv =l loadl %ipvaddr
	storel %pk, %ipaddr
	storel %pv, %ipvaddr
	storel %ipk, %paddr
	storel %ipv, %pvaddr
	%r =l call $qsort_pairs(l %a, l %ip)
	%ridx =l add %ip, 1
	%roff =l mul %ridx, 16
	%ra =l add %a, %roff
	%rn =l sub %n, %ridx
	%rr =l call $qsort_pairs(l %ra, l %rn)
	ret 0
}
