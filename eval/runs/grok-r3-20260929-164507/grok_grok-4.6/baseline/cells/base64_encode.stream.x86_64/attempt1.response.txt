base64_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rdx
    mov r14, rsi
    xor r15, r15
    test r14, r14
    jz .Ldone
.Lloop:
    xor r9d, r9d
    movzx r8d, byte ptr [r12]
    shl r8d, 16
    inc r12
    dec r14
    jz .Lrem1
    movzx eax, byte ptr [r12]
    shl eax, 8
    or r8d, eax
    inc r12
    dec r14
    jz .Lrem2
    movzx eax, byte ptr [r12]
    or r8d, eax
    inc r12
    dec r14
    mov ecx, 4
    jmp .Lemit
.Lrem2:
    mov ecx, 3
    mov r9d, 1
    jmp .Lemit
.Lrem1:
    mov ecx, 2
    mov r9d, 2
.Lemit:
    mov eax, r8d
    shr eax, 18
    and eax, 63
    shl r8d, 6
    lea ebx, [rax + 'A']
    lea edi, [rax + 'a' - 26]
    cmp eax, 26
    cmovae ebx, edi
    lea edi, [rax + '0' - 52]
    cmp eax, 52
    cmovae ebx, edi
    mov edi, '+'
    cmp eax, 62
    cmovae ebx, edi
    mov edi, '/'
    cmp eax, 63
    cmove ebx, edi
    mov byte ptr [r13], bl
    inc r13
    dec ecx
    jnz .Lemit
    test r9d, r9d
    jz .Lnopad
.Lpad:
    mov byte ptr [r13], '='
    inc r13
    dec r9d
    jnz .Lpad
.Lnopad:
    add r15, 4
    test r14, r14
    jnz .Lloop
.Ldone:
    mov rax, r15
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
