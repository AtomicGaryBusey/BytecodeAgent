export function l $binary_search(l %a, l %n, l %key) {
@start
	%high0 =l sub %n, 1
	jmp @loop
@loop
	%low =l phi @start 0, @right %lowr, @left %lowl
	%high =l phi @start %high0, @right %highr, @left %highl
	%le =w cslel %low, %high
	jnz %le, @body, @notfound
@body
	%sum =l add %low, %high
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @noteq
@noteq
	%lt =w csltl %val, %key
	jnz %lt, @right, @left
@right
	%lowr =l add %mid, 1
	%highr =l copy %high
	jmp @loop
@left
	%lowl =l copy %low
	%highl =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
