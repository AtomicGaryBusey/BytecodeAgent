export function l $base64_encode(l %src, l %n, l %out) {
@start
	%idxbuf =l alloc8 32
	jmp @loop
@loop
	%i =l phi @start 0, @advance %inext
	%o =l phi @start 0, @advance %onext
	%rem =l sub %n, %i
	%hasmore =w csgtl %rem, 0
	jnz %hasmore, @process, @done
@process
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%ge2 =w csgel %rem, 2
	jnz %ge2, @loadb1, @zerob1
@loadb1
	%p1 =l add %p0, 1
	%b1tmp =l loadub %p1
	jmp @gotb1
@zerob1
	jmp @gotb1
@gotb1
	%b1 =l phi @loadb1 %b1tmp, @zerob1 0
	%ge3 =w csgel %rem, 3
	jnz %ge3, @loadb2, @zerob2
@loadb2
	%p2 =l add %p0, 2
	%b2tmp =l loadub %p2
	jmp @gotb2
@zerob2
	jmp @gotb2
@gotb2
	%b2 =l phi @loadb2 %b2tmp, @zerob2 0
	%sh0 =l shl %b0, 16
	%sh1 =l shl %b1, 8
	%vor =l or %sh0, %sh1
	%v =l or %vor, %b2
	%ix0s =l shr %v, 18
	%ix0 =l and %ix0s, 63
	%ix1s =l shr %v, 12
	%ix1 =l and %ix1s, 63
	%ix2s =l shr %v, 6
	%ix2 =l and %ix2s, 63
	%ix3 =l and %v, 63
	storel %ix0, %idxbuf
	%ip1 =l add %idxbuf, 8
	storel %ix1, %ip1
	%ip2 =l add %idxbuf, 16
	storel %ix2, %ip2
	%ip3 =l add %idxbuf, 24
	storel %ix3, %ip3
	jmp @convloop
@convloop
	%k =l phi @gotb2 0, @stchar %knext
	%klt4 =w csltl %k, 4
	jnz %klt4, @convone, @afterconv
@convone
	%ksh =l shl %k, 3
	%kp =l add %idxbuf, %ksh
	%ixl =l loadl %kp
	%ix =w copy %ixl
	%lt26 =w csltw %ix, 26
	jnz %lt26, @leta, @chk52
@leta
	%chA =w add %ix, 65
	jmp @stchar
@chk52
	%lt52 =w csltw %ix, 52
	jnz %lt52, @letlow, @chk62
@letlow
	%cha =w add %ix, 71
	jmp @stchar
@chk62
	%lt62 =w csltw %ix, 62
	jnz %lt62, @letdig, @plsl
@letdig
	%chd =w sub %ix, 4
	jmp @stchar
@plsl
	%is62 =w ceqw %ix, 62
	jnz %is62, @plus, @slash
@plus
	jmp @stchar
@slash
	jmp @stchar
@stchar
	%ch =w phi @leta %chA, @letlow %cha, @letdig %chd, @plus 43, @slash 47
	%base =l add %out, %o
	%dst =l add %base, %k
	storeb %ch, %dst
	%knext =l add %k, 1
	jmp @convloop
@afterconv
	%rge3 =w csgel %rem, 3
	jnz %rge3, @gs3, @gsrem
@gs3
	jmp @havegs
@gsrem
	jmp @havegs
@havegs
	%gsize =l phi @gs3 3, @gsrem %rem
	%is1 =w ceql %gsize, 1
	jnz %is1, @pad2, @chk2
@pad2
	%bpad =l add %out, %o
	%d2 =l add %bpad, 2
	storeb 61, %d2
	%d3 =l add %bpad, 3
	storeb 61, %d3
	jmp @advance
@chk2
	%is2 =w ceql %gsize, 2
	jnz %is2, @pad1, @advance
@pad1
	%bpad2 =l add %out, %o
	%d3b =l add %bpad2, 3
	storeb 61, %d3b
	jmp @advance
@advance
	%inext =l add %i, %gsize
	%onext =l add %o, 4
	jmp @loop
@done
	ret %o
}
