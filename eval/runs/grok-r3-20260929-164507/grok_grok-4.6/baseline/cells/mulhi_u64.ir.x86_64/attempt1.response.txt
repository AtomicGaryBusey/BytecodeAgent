export function l $mulhi_u64(l %a, l %b) {
@start
	%a_lo =l and %a, 4294967295
	%a_hi =l shr %a, 32
	%b_lo =l and %b, 4294967295
	%b_hi =l shr %b, 32
	%p0 =l mul %a_lo, %b_lo
	%p1 =l mul %a_lo, %b_hi
	%p2 =l mul %a_hi, %b_lo
	%p3 =l mul %a_hi, %b_hi
	%p0_hi =l shr %p0, 32
	%p1_lo =l and %p1, 4294967295
	%p1_hi =l shr %p1, 32
	%p2_lo =l and %p2, 4294967295
	%p2_hi =l shr %p2, 32
	%mid1 =l add %p0_hi, %p1_lo
	%mid =l add %mid1, %p2_lo
	%carry =l shr %mid, 32
	%h1 =l add %p3, %p1_hi
	%h2 =l add %h1, %p2_hi
	%high =l add %h2, %carry
	ret %high
}
