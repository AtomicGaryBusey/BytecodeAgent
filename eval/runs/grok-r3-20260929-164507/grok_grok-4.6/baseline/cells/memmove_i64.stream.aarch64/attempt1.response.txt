memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.hi .Lbackward
.Lfwd_loop:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    b.ne .Lfwd_loop
    b .Ldone
.Lbackward:
    add x0, x0, x2, lsl #3
    add x1, x1, x2, lsl #3
.Lback_loop:
    ldr x3, [x1, #-8]!
    str x3, [x0, #-8]!
    subs x2, x2, #1
    b.ne .Lback_loop
.Ldone:
    mov x0, #0
    ret
