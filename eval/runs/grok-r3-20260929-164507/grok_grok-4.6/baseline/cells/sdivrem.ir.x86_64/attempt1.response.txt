export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @bzero, @chkmin
@bzero
	storel %a, %rem
	ret 0
@chkmin
	%min =l shl 1, 63
	%ismin =w ceql %a, %min
	jnz %ismin, @chkm1, @normal
@chkm1
	%ism1 =w ceql %b, -1
	jnz %ism1, @overflow, @normal
@overflow
	storel 0, %rem
	ret %min
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
