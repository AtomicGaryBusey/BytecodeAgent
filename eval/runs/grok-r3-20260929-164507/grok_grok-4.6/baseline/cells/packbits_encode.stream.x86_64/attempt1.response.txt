packbits_encode:
    xor r8, r8
    xor r9, r9
.Louter:
    cmp r8, rsi
    jge .Ldone
    movzx eax, byte ptr [rdi + r8]
    mov r10, 1
    lea r11, [r8 + 1]
.Lrunloop:
    cmp r10, 128
    jge .Lrun_done
    cmp r11, rsi
    jge .Lrun_done
    cmp byte ptr [rdi + r11], al
    jne .Lrun_done
    inc r10
    inc r11
    jmp .Lrunloop
.Lrun_done:
    cmp r10, 3
    jl .Ldo_literal
    lea rax, [r9 + 2]
    cmp rax, rcx
    ja .Lfail
    mov eax, 1
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Louter
.Ldo_literal:
    mov r11, r8
.Llitloop:
    cmp r11, rsi
    jge .Llit_done
    mov rax, r11
    sub rax, r8
    cmp rax, 128
    jge .Llit_done
    lea rax, [r11 + 2]
    cmp rax, rsi
    jge .Llit_advance
    movzx eax, byte ptr [rdi + r11]
    cmp al, byte ptr [rdi + r11 + 1]
    jne .Llit_advance
    cmp al, byte ptr [rdi + r11 + 2]
    je .Llit_done
.Llit_advance:
    inc r11
    jmp .Llitloop
.Llit_done:
    mov r10, r11
    sub r10, r8
    lea rax, [r9 + 1]
    add rax, r10
    cmp rax, rcx
    ja .Lfail
    mov eax, r10d
    dec eax
    mov byte ptr [rdx + r9], al
    inc r9
.Lcopy:
    movzx eax, byte ptr [rdi + r8]
    inc r8
    mov byte ptr [rdx + r9], al
    inc r9
    dec r10
    jnz .Lcopy
    jmp .Louter
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
