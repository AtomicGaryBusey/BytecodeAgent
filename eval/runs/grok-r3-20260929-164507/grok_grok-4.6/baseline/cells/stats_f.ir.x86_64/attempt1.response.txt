export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @after %inext
	%min =l phi @start %y0, @after %minn
	%max =l phi @start %y0, @after %maxn
	%sum =l phi @start %y0, @after %sumn
	%odds =l phi @start %odd0, @after %oddsn
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @keepmin
@setmin
	jmp @domax
@keepmin
	jmp @domax
@domax
	%minn =l phi @setmin %y, @keepmin %min
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @keepmax
@setmax
	jmp @after
@keepmax
	jmp @after
@after
	%maxn =l phi @setmax %y, @keepmax %max
	%sumn =l add %sum, %y
	%ob =l and %y, 1
	%oddsn =l add %odds, %ob
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %odds
}
