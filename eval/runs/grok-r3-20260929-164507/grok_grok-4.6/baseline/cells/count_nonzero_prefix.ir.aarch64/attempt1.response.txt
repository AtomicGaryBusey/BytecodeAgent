export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @found, @cont
@cont
	%i1 =l add %i, 1
	jmp @loop
@found
	ret %i
@done
	ret %n
}
