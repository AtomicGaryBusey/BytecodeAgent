export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @c_p1sm1 %pc_a, @c_p1sp1 %pc_b, @c_p1s0 %pc_c, @push %pc_d, @jzn %pc_e, @jzt %pc_f, @jmp %pc_g
	%sp =l phi @start 0, @c_p1sm1 %sp_a, @c_p1sp1 %sp_b, @c_p1s0 %sp_c, @push %sp_d, @jzn %sp_e, @jzt %sp_f, @jmp %sp_g
	%end =w ceql %pc, %len
	jnz %end, @halt, @fetch
@fetch
	%pcsh =l mul %pc, 8
	%instp =l add %code, %pcsh
	%op =l loadl %instp
	%eq0 =w ceql %op, 0
	jnz %eq0, @halt, @c1
@c1
	%eq1 =w ceql %op, 1
	jnz %eq1, @push, @c2
@c2
	%eq2 =w ceql %op, 2
	jnz %eq2, @add, @c3
@c3
	%eq3 =w ceql %op, 3
	jnz %eq3, @sub, @c4
@c4
	%eq4 =w ceql %op, 4
	jnz %eq4, @mul, @c5
@c5
	%eq5 =w ceql %op, 5
	jnz %eq5, @dup, @c6
@c6
	%eq6 =w ceql %op, 6
	jnz %eq6, @swap, @c7
@c7
	%eq7 =w ceql %op, 7
	jnz %eq7, @drop, @c8
@c8
	%eq8 =w ceql %op, 8
	jnz %eq8, @over, @c9
@c9
	%eq9 =w ceql %op, 9
	jnz %eq9, @lt, @c10
@c10
	%eq10 =w ceql %op, 10
	jnz %eq10, @jz, @c11
@c11
	%eq11 =w ceql %op, 11
	jnz %eq11, @jmp, @oth
@push
	%pcp1 =l add %pc, 1
	%p1sh =l mul %pcp1, 8
	%vp =l add %code, %p1sh
	%v =l loadl %vp
	%spsh =l mul %sp, 8
	%slot =l add %stack, %spsh
	storel %v, %slot
	%pc_d =l add %pc, 2
	%sp_d =l add %sp, 1
	jmp @loop
@add
	%bidx =l sub %sp, 1
	%bsh =l mul %bidx, 8
	%bp =l add %stack, %bsh
	%b =l loadl %bp
	%aidx =l sub %sp, 2
	%ash =l mul %aidx, 8
	%ap =l add %stack, %ash
	%a =l loadl %ap
	%sum =l add %a, %b
	storel %sum, %ap
	jmp @c_p1sm1
@sub
	%sbidx =l sub %sp, 1
	%sbsh =l mul %sbidx, 8
	%sbp =l add %stack, %sbsh
	%sb =l loadl %sbp
	%saidx =l sub %sp, 2
	%sash =l mul %saidx, 8
	%sap =l add %stack, %sash
	%sa =l loadl %sap
	%diff =l sub %sa, %sb
	storel %diff, %sap
	jmp @c_p1sm1
@mul
	%mbidx =l sub %sp, 1
	%mbsh =l mul %mbidx, 8
	%mbp =l add %stack, %mbsh
	%mb =l loadl %mbp
	%maidx =l sub %sp, 2
	%mash =l mul %maidx, 8
	%map =l add %stack, %mash
	%ma =l loadl %map
	%prod =l mul %ma, %mb
	storel %prod, %map
	jmp @c_p1sm1
@dup
	%didx =l sub %sp, 1
	%dsh =l mul %didx, 8
	%dp =l add %stack, %dsh
	%dv =l loadl %dp
	%dnsh =l mul %sp, 8
	%dnp =l add %stack, %dnsh
	storel %dv, %dnp
	jmp @c_p1sp1
@swap
	%w1 =l sub %sp, 1
	%w2 =l sub %sp, 2
	%w1sh =l mul %w1, 8
	%w2sh =l mul %w2, 8
	%w1p =l add %stack, %w1sh
	%w2p =l add %stack, %w2sh
	%wt1 =l loadl %w1p
	%wt2 =l loadl %w2p
	storel %wt2, %w1p
	storel %wt1, %w2p
	jmp @c_p1s0
@drop
	jmp @c_p1sm1
@over
	%oidx =l sub %sp, 2
	%osh =l mul %oidx, 8
	%optr =l add %stack, %osh
	%ov =l loadl %optr
	%onsh =l mul %sp, 8
	%onp =l add %stack, %onsh
	storel %ov, %onp
	jmp @c_p1sp1
@lt
	%lbidx =l sub %sp, 1
	%lbsh =l mul %lbidx, 8
	%lbp =l add %stack, %lbsh
	%lb =l loadl %lbp
	%laidx =l sub %sp, 2
	%lash =l mul %laidx, 8
	%lap =l add %stack, %lash
	%la =l loadl %lap
	%lcmp =w csltl %la, %lb
	%lres =l extuw %lcmp
	storel %lres, %lap
	jmp @c_p1sm1
@jz
	%zvidx =l sub %sp, 1
	%zvsh =l mul %zvidx, 8
	%zvp =l add %stack, %zvsh
	%zv =l loadl %zvp
	%ztpc =l add %pc, 1
	%ztsh =l mul %ztpc, 8
	%ztp =l add %code, %ztsh
	%zt =l loadl %ztp
	%zis0 =w ceql %zv, 0
	jnz %zis0, @jzt, @jzn
@jzt
	%pc_f =l copy %zt
	%sp_f =l sub %sp, 1
	jmp @loop
@jzn
	%pc_e =l add %pc, 2
	%sp_e =l sub %sp, 1
	jmp @loop
@jmp
	%jtpc =l add %pc, 1
	%jtsh =l mul %jtpc, 8
	%jtp =l add %code, %jtsh
	%jt =l loadl %jtp
	%pc_g =l copy %jt
	%sp_g =l copy %sp
	jmp @loop
@oth
	jmp @c_p1s0
@c_p1sm1
	%pc_a =l add %pc, 1
	%sp_a =l sub %sp, 1
	jmp @loop
@c_p1sp1
	%pc_b =l add %pc, 1
	%sp_b =l add %sp, 1
	jmp @loop
@c_p1s0
	%pc_c =l add %pc, 1
	%sp_c =l copy %sp
	jmp @loop
@halt
	%hempty =w ceql %sp, 0
	jnz %hempty, @no_top, @has_top
@no_top
	jmp @dozero
@has_top
	%htidx =l sub %sp, 1
	%htsh =l mul %htidx, 8
	%htp =l add %stack, %htsh
	%htv =l loadl %htp
	jmp @dozero
@dozero
	%retval =l phi @no_top 0, @has_top %htv
	jmp @zloop
@zloop
	%zi =l phi @dozero %sp, @zstore %zinext
	%zmore =w csltl %zi, %cap
	jnz %zmore, @zstore, @ret
@zstore
	%zsh =l mul %zi, 8
	%zp =l add %stack, %zsh
	storel 0, %zp
	%zinext =l add %zi, 1
	jmp @zloop
@ret
	ret %retval
}
