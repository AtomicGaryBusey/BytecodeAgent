export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p00 =l mul %alo, %blo
	%p01 =l mul %alo, %bhi
	%p10 =l mul %ahi, %blo
	%p11 =l mul %ahi, %bhi
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p01hi =l shr %p01, 32
	%p10lo =l and %p10, 4294967295
	%p10hi =l shr %p10, 32
	%u =l add %p00hi, %p01lo
	%u2 =l add %u, %p10lo
	%c =l shr %u2, 32
	%h1 =l add %p11, %p01hi
	%h2 =l add %h1, %p10hi
	%h =l add %h2, %c
	ret %h
}
