export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @read_h %i1, @lit_loop %i_lit, @rep_loop %i2
	%o =l phi @start 0, @read_h %o, @lit_loop %o_lit, @rep_loop %o_rep
	%cmp =w csltl %i, %n
	jnz %cmp, @read_h, @done
@done
	ret %o
@read_h
	%src_i =l add %src, %i
	%h =w loadsb %src_i
	%i1 =l add %i, 1
	%is_noop =w ceqw %h, -128
	jnz %is_noop, @main, @not_noop
@not_noop
	%is_lit =w csgew %h, 0
	jnz %is_lit, @literal, @repeat
@literal
	%h_l =l extsw %h
	%count =l add %h_l, 1
	%needed =l add %i1, %count
	%src_ok =w cslel %needed, %n
	jnz %src_ok, @lit_loop, @err_src
@lit_loop
	%k =l phi @literal 0, @lit_write %k1
	%i_lit =l phi @literal %i1, @lit_write %i_lit1
	%o_lit =l phi @literal %o, @lit_write %o_lit1
	%k_lt =w csltl %k, %count
	jnz %k_lt, @lit_check_cap, @main
@lit_check_cap
	%o_ge =w cslel %cap, %o_lit
	jnz %o_ge, @err_cap, @lit_write
@lit_write
	%sp =l add %src, %i_lit
	%b =w loadub %sp
	%op =l add %out, %o_lit
	storeb %b, %op
	%i_lit1 =l add %i_lit, 1
	%o_lit1 =l add %o_lit, 1
	%k1 =l add %k, 1
	jmp @lit_loop
@repeat
	%src_need =w cslel %n, %i1
	jnz %src_need, @err_src, @rep_got
@rep_got
	%sp2 =l add %src, %i1
	%rb =w loadub %sp2
	%i2 =l add %i1, 1
	%h_lr =l extsw %h
	%rcount =l sub 1, %h_lr
	jmp @rep_loop
@rep_loop
	%rk =l phi @rep_got 0, @rep_write %rk1
	%o_rep =l phi @rep_got %o, @rep_write %o_rep1
	%rk_lt =w csltl %rk, %rcount
	jnz %rk_lt, @rep_check_cap, @main
@rep_check_cap
	%o_ge2 =w cslel %cap, %o_rep
	jnz %o_ge2, @err_cap, @rep_write
@rep_write
	%op2 =l add %out, %o_rep
	storeb %rb, %op2
	%o_rep1 =l add %o_rep, 1
	%rk1 =l add %rk, 1
	jmp @rep_loop
@err_src
	ret -2
@err_cap
	ret -1
}
