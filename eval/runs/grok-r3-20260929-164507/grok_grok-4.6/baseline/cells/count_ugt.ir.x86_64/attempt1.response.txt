export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%c =w phi @start 0, @body %cnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%idx =l mul %i, 8
	%ptr =l add %a, %idx
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%cnext =w add %c, %gt
	%inext =l add %i, 1
	jmp @loop
@end
	%res =l extuw %c
	ret %res
}
