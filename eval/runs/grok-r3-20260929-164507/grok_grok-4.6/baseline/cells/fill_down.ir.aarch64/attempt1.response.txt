export function l $fill_down(l %a, l %n, l %start) {
@start
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%val =l sub %start, %i
	%off =l mul %i, 8
	%ptr =l add %a, %off
	storel %val, %ptr
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %n
}
