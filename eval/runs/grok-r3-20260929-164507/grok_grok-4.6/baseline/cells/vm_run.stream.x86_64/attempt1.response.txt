vm_run:
    mov r8, rcx
    xor r9d, r9d
    mov r10, rdx
.Lloop:
    cmp r9, rsi
    jae .Lhalt
    mov rax, qword ptr [rdi + r9*8]
    test rax, rax
    jz .Lhalt
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
.Lnext:
    inc r9
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rdi + r9*8 + 8]
    mov qword ptr [r10], rax
    add r10, 8
    inc r9
    jmp .Lnext
.Ladd:
    mov rax, qword ptr [r10 - 8]
    add qword ptr [r10 - 16], rax
    sub r10, 8
    jmp .Lnext
.Lsub:
    mov rax, qword ptr [r10 - 8]
    sub qword ptr [r10 - 16], rax
    sub r10, 8
    jmp .Lnext
.Lmul:
    mov rax, qword ptr [r10 - 16]
    imul rax, qword ptr [r10 - 8]
    mov qword ptr [r10 - 16], rax
    sub r10, 8
    jmp .Lnext
.Ldup:
    mov rax, qword ptr [r10 - 8]
    mov qword ptr [r10], rax
    add r10, 8
    jmp .Lnext
.Lswap:
    mov rax, qword ptr [r10 - 8]
    mov rcx, qword ptr [r10 - 16]
    mov qword ptr [r10 - 16], rax
    mov qword ptr [r10 - 8], rcx
    jmp .Lnext
.Ldrop:
    sub r10, 8
    jmp .Lnext
.Lover:
    mov rax, qword ptr [r10 - 16]
    mov qword ptr [r10], rax
    add r10, 8
    jmp .Lnext
.Llt:
    xor ecx, ecx
    mov rax, qword ptr [r10 - 16]
    cmp rax, qword ptr [r10 - 8]
    setl cl
    mov qword ptr [r10 - 16], rcx
    sub r10, 8
    jmp .Lnext
.Ljz:
    sub r10, 8
    cmp qword ptr [r10], 0
    jne .Ljznz
    mov r9, qword ptr [rdi + r9*8 + 8]
    jmp .Lloop
.Ljznz:
    inc r9
    jmp .Lnext
.Ljmp:
    mov r9, qword ptr [rdi + r9*8 + 8]
    jmp .Lloop
.Lhalt:
    xor r11d, r11d
    cmp r10, rdx
    je .Lzero
    mov r11, qword ptr [r10 - 8]
.Lzero:
    mov rdi, r10
    lea rcx, [rdx + r8*8]
    sub rcx, r10
    shr rcx, 3
    xor eax, eax
    rep stosq
    mov rax, r11
    ret
