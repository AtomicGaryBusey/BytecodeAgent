export function l $crc32(l %p, l %n) {
@start
	jmp @byteloop
@byteloop
	%p1 =l phi @start %p, @nextbyte %p2
	%n1 =l phi @start %n, @nextbyte %n2
	%c1 =w phi @start -1, @nextbyte %c3
	%more =w cnel %n1, 0
	jnz %more, @dobyt, @done
@dobyt
	%byte =w loadub %p1
	%cx =w xor %c1, %byte
	jmp @bitloop
@bitloop
	%c2 =w phi @dobyt %cx, @dobit %c4
	%cnt =w phi @dobyt 8, @dobit %cnt1
	jnz %cnt, @dobit, @nextbyte
@dobit
	%lsb =w and %c2, 1
	%mask =w neg %lsb
	%poly =w and %mask, 3988292384
	%cshr =w shr %c2, 1
	%c4 =w xor %cshr, %poly
	%cnt1 =w sub %cnt, 1
	jmp @bitloop
@nextbyte
	%p2 =l add %p1, 1
	%n2 =l sub %n1, 1
	%c3 =w copy %c2
	jmp @byteloop
@done
	%fin =w xor %c1, -1
	%res =l extuw %fin
	ret %res
}
