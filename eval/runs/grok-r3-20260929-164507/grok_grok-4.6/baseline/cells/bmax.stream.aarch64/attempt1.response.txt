bmax:
    lsr x2, x1, #2
    cbz x2, .Lsmall
    ld1 {v0.2d, v1.2d}, [x0], #32
    subs x2, x2, #1
    b.eq .Lreduce4
.Lvec4:
    ld1 {v2.2d, v3.2d}, [x0], #32
    smax v0.2d, v0.2d, v2.2d
    smax v1.2d, v1.2d, v3.2d
    subs x2, x2, #1
    b.ne .Lvec4
.Lreduce4:
    smax v0.2d, v0.2d, v1.2d
    and x2, x1, #3
    smaxp v0.2d, v0.2d, v0.2d
    umov x3, v0.d[0]
    cbz x2, .Ldone
.Ltail:
    ldr x4, [x0], #8
    cmp x4, x3
    csel x3, x4, x3, gt
    subs x2, x2, #1
    b.ne .Ltail
.Ldone:
    mov x0, x3
    ret
.Lsmall:
    ldr x3, [x0], #8
    subs x1, x1, #1
    b.eq .Ldone
.Lsmall_loop:
    ldr x4, [x0], #8
    cmp x4, x3
    csel x3, x4, x3, gt
    subs x1, x1, #1
    b.ne .Lsmall_loop
    b .Ldone
