memmove_i64:
xor eax, eax
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jae .Lbackward
xor ecx, ecx
.Lfwd:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
cmp rcx, rdx
jb .Lfwd
ret
.Lbackward:
dec rdx
mov r8, qword ptr [rsi + rdx*8]
mov qword ptr [rdi + rdx*8], r8
jnz .Lbackward
.Ldone:
ret
