mp_mul:
  push rbx
  push r12
  push r13
  push r14
  push r15
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  mov rbx, r8
  lea r10, [r13 + r15]
  test r10, r10
  jz .Lret0
  xor eax, eax
  xor ecx, ecx
.Lzero:
  mov qword ptr [rbx + rcx*8], rax
  inc rcx
  cmp rcx, r10
  jne .Lzero
  xor r8, r8
  jmp .Louter_test
.Louter:
  mov rdx, qword ptr [r12 + r8*8]
  test rdx, rdx
  jz .Lnext_i
  lea rdi, [rbx + r8*8]
  xor r9, r9
  xor r11, r11
  test r15, r15
  jz .Lstore_carry
.Linner:
  mulx rcx, rax, qword ptr [r14 + r9*8]
  add rax, r11
  adc rcx, 0
  add rax, qword ptr [rdi + r9*8]
  adc rcx, 0
  mov qword ptr [rdi + r9*8], rax
  mov r11, rcx
  inc r9
  cmp r9, r15
  jne .Linner
.Lstore_carry:
  mov qword ptr [rdi + r15*8], r11
.Lnext_i:
  inc r8
.Louter_test:
  cmp r8, r13
  jb .Louter
  mov rax, r10
.Ltrim:
  cmp qword ptr [rbx + rax*8 - 8], 0
  jne .Lret
  dec rax
  jnz .Ltrim
.Lret0:
  xor eax, eax
.Lret:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbx
  ret
