collatz_max:
xor r9, r9
mov rcx, 1
.Louter:
mov rsi, rcx
xor rdx, rdx
cmp rsi, 1
je .Linner_done
.Linner:
test rsi, 1
jz .Leven
lea rsi, [rsi + rsi*2 + 1]
inc rdx
.Leven:
shr rsi, 1
inc rdx
cmp rsi, 1
jne .Linner
.Linner_done:
cmp rdx, r9
cmovg r9, rdx
inc rcx
cmp rcx, rdi
jle .Louter
mov rax, r9
ret
