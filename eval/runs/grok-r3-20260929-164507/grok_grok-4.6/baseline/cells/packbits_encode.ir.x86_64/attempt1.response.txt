export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @rep_write %i_rep, @copy_done %j
	%wr =l phi @start 0, @rep_write %wr2, @copy_done %wrn
	%cmp_i =w csltl %i, %n
	jnz %cmp_i, @have, @ret_wr
@ret_wr
	ret %wr
@have
	%sp =l add %src, %i
	%b =w loadub %sp
	jmp @run1
@run1
	%r =w phi @have 1, @run1inc %rinc
	%rxl =l extuw %r
	%ir =l add %i, %rxl
	%ge_n =w cslel %n, %ir
	jnz %ge_n, @run1done, @run1b
@run1b
	%ge128 =w csgew %r, 128
	jnz %ge128, @run1done, @run1c
@run1c
	%sp2 =l add %src, %ir
	%c =w loadub %sp2
	%neq =w cnew %c, %b
	jnz %neq, @run1done, @run1inc
@run1inc
	%rinc =w add %r, 1
	jmp @run1
@run1done
	%rge3 =w csgew %r, 3
	jnz %rge3, @repeat, @litscan
@repeat
	%wr2 =l add %wr, 2
	%nofit =w csgtl %wr2, %cap
	jnz %nofit, @fail, @rep_write
@fail
	ret -1
@rep_write
	%hdr =w sub 257, %r
	%op =l add %out, %wr
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %b, %op1
	%rxl2 =l extuw %r
	%i_rep =l add %i, %rxl2
	jmp @loop
@litscan
	%j =l phi @run1done %i, @litadv %jnext
	%llen =w phi @run1done 0, @litadv %llenext
	%j_ge_n =w cslel %n, %j
	jnz %j_ge_n, @emit_lit, @litb
@litb
	%llen_ge128 =w csgew %llen, 128
	jnz %llen_ge128, @emit_lit, @litc
@litc
	%jsp =l add %src, %j
	%bj =w loadub %jsp
	jmp @run2
@run2
	%rj =w phi @litc 1, @run2inc %rjinc
	%rjxl =l extuw %rj
	%jr =l add %j, %rjxl
	%jr_ge_n =w cslel %n, %jr
	jnz %jr_ge_n, @run2done, @run2b
@run2b
	%rj_ge128 =w csgew %rj, 128
	jnz %rj_ge128, @run2done, @run2c
@run2c
	%jsp2 =l add %src, %jr
	%cj =w loadub %jsp2
	%nneq =w cnew %cj, %bj
	jnz %nneq, @run2done, @run2inc
@run2inc
	%rjinc =w add %rj, 1
	jmp @run2
@run2done
	%rj_ge3 =w csgew %rj, 3
	jnz %rj_ge3, @emit_lit, @litadv
@litadv
	%jnext =l add %j, 1
	%llenext =w add %llen, 1
	jmp @litscan
@emit_lit
	%need0 =l extuw %llen
	%need =l add %need0, 1
	%wrn =l add %wr, %need
	%nofit2 =w csgtl %wrn, %cap
	jnz %nofit2, @fail, @lit_write
@lit_write
	%hdr2 =w sub %llen, 1
	%op2 =l add %out, %wr
	storeb %hdr2, %op2
	jmp @copy
@copy
	%k =w phi @lit_write 0, @copyinc %knext
	%k_ge_L =w csgew %k, %llen
	jnz %k_ge_L, @copy_done, @copyb
@copyb
	%kl =l extuw %k
	%si =l add %i, %kl
	%saddr =l add %src, %si
	%byte =w loadub %saddr
	%wr1 =l add %wr, 1
	%di =l add %wr1, %kl
	%daddr =l add %out, %di
	storeb %byte, %daddr
	jmp @copyinc
@copyinc
	%knext =w add %k, 1
	jmp @copy
@copy_done
	jmp @loop
}
