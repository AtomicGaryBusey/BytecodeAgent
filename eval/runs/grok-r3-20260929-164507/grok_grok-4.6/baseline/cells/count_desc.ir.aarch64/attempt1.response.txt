export function l $count_desc(l %a, l %n) {
@start
	%nlt2 =w csltl %n, 2
	jnz %nlt2, @ret0, @setup
@ret0
	ret 0
@setup
	%nm1 =l sub %n, 1
	jmp @header
@header
	%i =l phi @setup 0, @body %inext
	%cnt =l phi @setup 0, @body %cntnext
	%more =w csltl %i, %nm1
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%p2 =l add %p, 8
	%v2 =l loadl %p2
	%gt =w csgtl %v, %v2
	%inc =l extuw %gt
	%cntnext =l add %cnt, %inc
	%inext =l add %i, 1
	jmp @header
@done
	ret %cnt
}
