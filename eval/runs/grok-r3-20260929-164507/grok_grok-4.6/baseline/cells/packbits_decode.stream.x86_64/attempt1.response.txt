packbits_decode:
    lea r8, [rdi + rsi]
    lea r9, [rdx + rcx]
    mov r10, rdx
.Lloop:
    cmp rdi, r8
    jae .Ldone
    movsx eax, byte ptr [rdi]
    inc rdi
    cmp eax, -128
    je .Lloop
    test eax, eax
    jns .Lliteral
    cmp rdi, r8
    jae .Lerr_src
    movzx ecx, byte ptr [rdi]
    inc rdi
    neg eax
    inc eax
    mov rsi, r9
    sub rsi, rdx
    cmp rax, rsi
    ja .Ltrunc_rep
    movzx r11d, cl
    imul r11d, r11d, 0x01010101
    mov esi, r11d
    shl r11, 32
    or r11, rsi
.Lfill8:
    cmp eax, 8
    jb .Lfill1
    mov qword ptr [rdx], r11
    add rdx, 8
    sub eax, 8
    jmp .Lfill8
.Lfill1:
    test eax, eax
    jz .Lloop
.Lfill1_loop:
    mov byte ptr [rdx], r11b
    inc rdx
    dec eax
    jnz .Lfill1_loop
    jmp .Lloop
.Ltrunc_rep:
    test rsi, rsi
    jz .Lerr_cap
.Ltrunc_rep_loop:
    mov byte ptr [rdx], cl
    inc rdx
    dec rsi
    jnz .Ltrunc_rep_loop
    jmp .Lerr_cap
.Lliteral:
    inc eax
    mov rsi, r8
    sub rsi, rdi
    cmp rsi, rax
    jb .Lerr_src
    mov rsi, r9
    sub rsi, rdx
    cmp rax, rsi
    ja .Ltrunc_lit
.Lcopy8:
    cmp eax, 8
    jb .Lcopy1
    mov rsi, qword ptr [rdi]
    mov qword ptr [rdx], rsi
    add rdi, 8
    add rdx, 8
    sub eax, 8
    jmp .Lcopy8
.Lcopy1:
    test eax, eax
    jz .Lloop
.Lcopy1_loop:
    movzx ecx, byte ptr [rdi]
    inc rdi
    mov byte ptr [rdx], cl
    inc rdx
    dec eax
    jnz .Lcopy1_loop
    jmp .Lloop
.Ltrunc_lit:
    test rsi, rsi
    jz .Lerr_cap
.Ltrunc_lit8:
    cmp rsi, 8
    jb .Ltrunc_lit1
    mov r11, qword ptr [rdi]
    mov qword ptr [rdx], r11
    add rdi, 8
    add rdx, 8
    sub rsi, 8
    jmp .Ltrunc_lit8
.Ltrunc_lit1:
    test rsi, rsi
    jz .Lerr_cap
.Ltrunc_lit1_loop:
    movzx eax, byte ptr [rdi]
    inc rdi
    mov byte ptr [rdx], al
    inc rdx
    dec rsi
    jnz .Ltrunc_lit1_loop
    jmp .Lerr_cap
.Ldone:
    mov rax, rdx
    sub rax, r10
    ret
.Lerr_src:
    mov rax, -2
    ret
.Lerr_cap:
    mov rax, -1
    ret
