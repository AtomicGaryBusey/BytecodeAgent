export function l $reverse(l a, l n) {
    @entry:
        %cmp = csltl %n, 2
        jnz %cmp, @loop
        ret 0

    @loop:
        %a1 = addl %a, %n
        %a2 = subl %a1, 1
        %a3 = subl %a2, 1
        %a4 = addl %a3, 8
        %a5 = loadl %a
        %a6 = loadl %a4
        %a7 = storel %a6, %a
        %a8 = storel %a5, %a4
        %a9 = addl %a, 8
        %a10 = subl %n, 2
        %a11 = csltl %a10, 2
        jnz %a11, @loop
        ret 0
}
