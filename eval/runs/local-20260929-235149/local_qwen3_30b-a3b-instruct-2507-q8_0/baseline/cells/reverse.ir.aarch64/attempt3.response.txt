export function l $reverse(l a, l n) {
    @entry:
        %cmp = csltl %n, 2
        jnz %cmp, @loop
        ret 0

    @loop:
        %a1 = loadl %a
        %a2 = loadl %a
        %a3 = addl %a, 8
        %a4 = addl %a, %n
        %a5 = subl %a4, 1
        %a6 = subl %a5, 1
        %a7 = addl %a6, 8
        %a8 = loadl %a7
        %a9 = storel %a8, %a
        %a10 = storel %a2, %a3
        %a11 = addl %a, 8
        %a12 = subl %n, 2
        %a13 = addl %a11, 8
        %a14 = subl %a12, 1
        %a15 = csltl %a14, 2
        jnz %a15, @loop
        ret 0
}
