export function l $crc32(l p, l n) {
    @entry:
        %crc = l 0xFFFFFFFF
        %i = l 0
        %loop = w 0
        %cmp = w slt %i, %n
        jnz %cmp, @loop_body
        %result = l xor %crc, 0xFFFFFFFF
        ret %result

    @loop_body:
        %byte = w loadub %p
        %crc = w xor %crc, %byte
        %j = l 0
        %bitloop = w 0
        %bitcmp = w slt %j, 8
        jnz %bitcmp, @bitloop_body
        %i = l add %i, 1
        %p = l add %p, 1
        %cmp = w slt %i, %n
        jnz %cmp, @loop_body
        %result = l xor %crc, 0xFFFFFFFF
        ret %result

    @bitloop_body:
        %lsb = w and %crc, 1
        %shifted = w shr %crc, 1
        %new_crc = w xor %shifted, 0xEDB88320
        %crc = w csel %lsb, %new_crc, %shifted
        %j = l add %j, 1
        %bitcmp = w slt %j, 8
        jnz %bitcmp, @bitloop_body
        %i = l add %i, 1
        %p = l add %p, 1
        %cmp = w slt %i, %n
        jnz %cmp, @loop_body
        %result = l xor %crc, 0xFFFFFFFF
        ret %result
}
