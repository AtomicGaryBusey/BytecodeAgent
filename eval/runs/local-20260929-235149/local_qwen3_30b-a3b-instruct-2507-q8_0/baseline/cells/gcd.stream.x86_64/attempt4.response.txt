gcd:
    # Handle base cases
    cmp rdi, 0
    jne .not_zero_a
    cmp rsi, 0
    jne .not_zero_b
    # gcd(0, 0) = 0
    xor rax, rax
    ret
.not_zero_a:
    cmp rsi, 0
    jne .both_nonzero
    # gcd(a, 0) = a
    mov rax, rdi
    ret
.not_zero_b:
    # gcd(0, b) = b
    mov rax, rsi
    ret
.both_nonzero:
    # Euclid's algorithm: gcd(a, b) = gcd(b, a mod b)
    # Use rdi = a, rsi = b
    # We'll use rdx for remainder, rbx for temp storage
    mov rbx, rdi
    mov rdx, rsi
    .Lloop:
        # Compute rdx = rbx % rdx
        xor rax, rax
        mov rax, rbx
        div rdx
        # rax = quotient, rdx = remainder
        # If remainder is 0, we're done
        test rdx, rdx
        jz .done
        # Otherwise, set rbx = rdx, rdi = rax, and repeat
        mov rdi, rdx
        mov rbx, rax
        jmp .Lloop
.done:
    mov rax, rbx
    ret
