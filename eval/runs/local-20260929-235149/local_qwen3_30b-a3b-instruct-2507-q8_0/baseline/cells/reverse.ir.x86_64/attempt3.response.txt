export function l $reverse(l a, l n) {
    @start:
        %cmp = csltl l n, 2
        jnz %cmp, @exit
        %i = const l 0
        %j = subl l n, 1
        @loop:
            %i_lt_j = csltl l %i, %j
            jz %i_lt_j, @exit
            %a_i = loadl l a, %i
            %a_j = loadl l a, %j
            storel l a, %i, %a_j
            storel l a, %j, %a_i
            %i = addl l %i, 1
            %j = subl l %j, 1
            jmp @loop
        @exit:
            ret l 0
}
