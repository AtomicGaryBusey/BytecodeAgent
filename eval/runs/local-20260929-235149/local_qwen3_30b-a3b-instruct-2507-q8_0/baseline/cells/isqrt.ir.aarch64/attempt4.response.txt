export function l $isqrt(l n) {
    @entry:
        %r = constl 0
        %low = constl 0
        %high = constl 4294967295
        %mid = constl 0
        %mid2 = constl 0
        %done = constw 0

    @loop:
        %done = csltl %low, %high
        jnz %done, @loop_body
        ret %r

    @loop_body:
        %mid = addl %low, %high
        %mid = shrl %mid, 1
        %mid2 = mul %mid, %mid
        %cmp = csltl %mid2, %n
        jnz %cmp, @in_range
        %low = addl %mid, 1
        jmp @loop

    @in_range:
        %r = %mid
        %high = subl %mid, 1
        jmp @loop
}
