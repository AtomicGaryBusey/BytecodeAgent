export function l $bubble_sort(l a, l n) {
    @entry:
        %0 = loadl l [a]
        %1 = loadl l [n]
        %2 = ceql %1, 0l
        jz %2, @return
        %3 = subl %1, 1l
        %4 = movl %3
        %5 = movl 0l
        @loop_outer:
            %6 = ceql %5, %3
            jz %6, @return
            %7 = movl 0l
            @loop_inner:
                %8 = ceql %7, %4
                jz %8, @after_inner
                %9 = addl a, %7
                %10 = loadl l [%9]
                %11 = addl a, %7
                %12 = addl 1l, %7
                %13 = loadl l [%11]
                %14 = csltl %10, %13
                jz %14, @skip_swap
                %15 = addl a, %7
                %16 = addl a, %7
                %17 = addl 1l, %7
                %18 = loadl l [%16]
                %19 = loadl l [%15]
                %20 = storel l [%15, %18]
                %21 = storel l [%16, %19]
            @skip_swap:
                %22 = addl %7, 1l
                %7 = movl %22
                jmp @loop_inner
            @after_inner:
                %23 = addl %5, 1l
                %5 = movl %23
                %24 = subl %4, 1l
                %4 = movl %24
                jmp @loop_outer
        @return:
            %25 = movl 0l
            ret %25
}
