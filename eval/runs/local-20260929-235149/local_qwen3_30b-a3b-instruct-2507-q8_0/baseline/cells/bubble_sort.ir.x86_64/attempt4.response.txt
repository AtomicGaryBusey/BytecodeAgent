export function l $bubble_sort(l a, l n) {
    @entry:
        %0 = loadl l [n]
        %1 = ceql %0, 0l
        jz %1, @return
        %2 = subl %0, 1l
        %3 = movl %2
        %4 = movl 0l
        @loop_outer:
            %5 = ceql %4, %3
            jz %5, @return
            %6 = movl 0l
            @loop_inner:
                %7 = ceql %6, %3
                jz %7, @after_inner
                %8 = addl a, %6
                %9 = loadl l [%8]
                %10 = addl %6, 1l
                %11 = addl a, %10
                %12 = loadl l [%11]
                %13 = csltl %9, %12
                jz %13, @skip_swap
                %14 = addl a, %6
                %15 = addl a, %10
                %16 = loadl l [%15]
                %17 = loadl l [%14]
                %18 = storel l [%14, %16]
                %19 = storel l [%15, %17]
            @skip_swap:
                %20 = addl %6, 1l
                %6 = movl %20
                jmp @loop_inner
            @after_inner:
                %21 = addl %4, 1l
                %4 = movl %21
                %22 = subl %3, 1l
                %3 = movl %22
                jmp @loop_outer
        @return:
            %23 = movl 0l
            ret %23
}
