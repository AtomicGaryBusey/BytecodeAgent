export function l $count_nonzero_prefix(l a, l n) {
    @start:
        %cmp = ceql %n, 0b
        jz %cmp, @done
        %i = constl 0b
        %p = addl %a, %i
        %val = loadl %p
        %zero = constl 0b
        %is_zero = ceql %val, %zero
        jz %is_zero, @found_zero
    @loop:
        %i = addl %i, 1b
        %p = addl %p, 8b
        %val = loadl %p
        %is_zero = ceql %val, %zero
        jnz %is_zero, @loop
    @found_zero:
        %result = addl %i, 1b
        jmp @done
    @done:
        %result = %i
    @return:
        ret %result
}
