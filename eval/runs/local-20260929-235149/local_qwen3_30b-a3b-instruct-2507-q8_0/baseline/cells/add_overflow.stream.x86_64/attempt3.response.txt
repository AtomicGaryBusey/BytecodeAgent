add_overflow:
    mov rax, rdi
    add rax, rsi
    mov qword ptr [rdx], rax
    mov rax, rdi
    sar rax, 63
    mov rbx, rax
    mov rax, rsi
    sar rax, 63
    mov rcx, rax
    xor rbx, rcx
    test rbx, rbx
    jz .check_result_sign
    mov rax, 0
    ret
.check_result_sign:
    mov rax, rdi
    add rax, rsi
    sar rax, 63
    cmp rax, rbx
    jne .overflow
    mov rax, 0
    ret
.overflow:
    mov rax, 1
    ret
