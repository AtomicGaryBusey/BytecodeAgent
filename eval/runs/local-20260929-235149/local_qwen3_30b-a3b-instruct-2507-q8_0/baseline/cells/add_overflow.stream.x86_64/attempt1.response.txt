add_overflow:
    mov rax, rdi
    add rax, rsi
    mov qword ptr [rdx], rax
    mov rax, 0
    # Check for signed overflow: if signs of a and b are the same but differ from result, overflow occurred
    mov rbx, rdi
    mov rcx, rsi
    or rbx, rcx
    # If both a and b are non-negative, check if result is negative
    test rdi, rdi
    js .check_negative
    test rsi, rsi
    js .check_negative
    # Both non-negative: if result is negative, overflow
    test rax, rax
    js .overflow
    ret
.check_negative:
    test rdi, rdi
    jns .check_both_negative
    test rsi, rsi
    jns .check_both_negative
    # Both negative: if result is non-negative, overflow
    test rax, rax
    jns .overflow
    ret
.check_both_negative:
    # One positive, one negative: no overflow possible
    ret
.overflow:
    mov rax, 1
    ret
