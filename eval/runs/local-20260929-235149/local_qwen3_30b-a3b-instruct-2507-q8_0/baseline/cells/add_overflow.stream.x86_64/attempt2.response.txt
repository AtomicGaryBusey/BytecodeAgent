add_overflow:
    mov rax, rdi
    add rax, rsi
    mov qword ptr [rdx], rax
    mov rax, 0
    mov rbx, rdi
    mov rcx, rsi
    mov rdi, rbx
    mov rsi, rcx
    # Check for signed overflow: if signs of a and b are the same but differ from result, overflow occurred
    mov rax, rdi
    add rax, rsi
    mov qword ptr [rdx], rax
    # Extract sign bits
    mov rax, rdi
    sar rax, 63
    mov rdi, rax
    mov rax, rsi
    sar rax, 63
    mov rsi, rax
    # If both signs are 1 (negative) or both are 0 (positive), check if result sign differs
    xor rdi, rsi
    test rdi, rdi
    jz .check_result_sign
    # Different signs: no overflow
    ret
.check_result_sign:
    mov rax, rdi
    add rax, rsi
    mov rax, rax
    sar rax, 63
    cmp rax, rdi
    jne .overflow
    ret
.overflow:
    mov rax, 1
    ret
