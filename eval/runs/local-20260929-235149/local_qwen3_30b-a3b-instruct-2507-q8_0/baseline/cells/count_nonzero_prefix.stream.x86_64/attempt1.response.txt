count_nonzero_prefix:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    mov r12, rdi
    mov r13, rsi
    xor rax, rax
    mov rbx, 0
.Lloop:
    cmp rbx, r13
    jge .Ldone
    mov rdi, r12
    mov rsi, rbx
    mov rdi, qword ptr [rdi + rsi*8]
    test rdi, rdi
    jz .Ldone
    inc rax
    inc rbx
    jmp .Lloop
.Ldone:
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
