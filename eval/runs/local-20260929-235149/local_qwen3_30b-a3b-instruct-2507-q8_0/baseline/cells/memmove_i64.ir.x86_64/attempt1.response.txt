export function l $memmove_i64(l dst, l src, l n) {
    @start:
        %cmp = csltl l n, 0
        jnz %cmp, @done
        %cmp = ceql l n, 0
        jnz %cmp, @done
        %result = movl l 0
        ret l %result

    @done:
        %dst_ptr = movl l dst
        %src_ptr = movl l src
        %count = movl l n
        %zero = movl l 0

        %cmp = csltl l %count, 1
        jnz %cmp, @copy_one
        %cmp = ceql l %count, 1
        jnz %cmp, @copy_one

        %dst_end = addl l %dst_ptr, %count
        %src_end = addl l %src_ptr, %count
        %cmp = csltl l %src_ptr, %dst_ptr
        jnz %cmp, @copy_forward

    @copy_backward:
        %dst_ptr = subl l %dst_ptr, 1
        %src_ptr = subl l %src_ptr, 1
        %count = subl l %count, 1
        %val = loadl l %src_ptr
        storel l %val, %dst_ptr
        %cmp = csltl l %count, 1
        jnz %cmp, @copy_backward
        %result = movl l dst
        ret l %result

    @copy_forward:
        %cmp = csltl l %src_ptr, %dst_ptr
        jnz %cmp, @copy_backward
        %cmp = ceql l %count, 1
        jnz %cmp, @copy_one

    @copy_loop:
        %val = loadl l %src_ptr
        storel l %val, %dst_ptr
        %dst_ptr = addl l %dst_ptr, 1
        %src_ptr = addl l %src_ptr, 1
        %count = subl l %count, 1
        %cmp = csltl l %count, 1
        jnz %cmp, @copy_loop

        %result = movl l dst
        ret l %result

    @copy_one:
        %val = loadl l %src_ptr
        storel l %val, %dst_ptr
        %result = movl l dst
        ret l %result
}
