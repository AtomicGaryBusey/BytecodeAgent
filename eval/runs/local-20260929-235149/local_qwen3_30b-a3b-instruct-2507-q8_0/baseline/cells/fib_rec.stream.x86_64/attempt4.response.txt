fib_rec:
    push rbp
    mov rbp, rsp
    sub rsp, 16
    cmp rdi, 1
    jle .base_case
    mov rax, rdi
    dec rax
    mov rdi, rax
    call fib_rec
    mov qword ptr [rbp - 8], rax
    mov rax, rdi
    sub rax, 2
    mov rdi, rax
    call fib_rec
    add rax, qword ptr [rbp - 8]
    jmp .return
.base_case:
    mov rax, rdi
.return:
    add rsp, 16
    pop rbp
    ret
