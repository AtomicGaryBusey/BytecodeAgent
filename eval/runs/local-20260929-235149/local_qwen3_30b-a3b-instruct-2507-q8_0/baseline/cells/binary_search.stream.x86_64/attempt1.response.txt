binary_search:
    push rbp
    mov rbp, rsp
    sub rsp, 16
    mov qword ptr [rbp - 8], rdi
    mov qword ptr [rbp - 16], rsi
    mov rax, 0
    mov rdi, 0
    mov rsi, rdx
    mov rdx, 0
    mov rcx, rsi
    cmp rcx, 0
    je .Lnot_found
.Lloop:
    mov rax, rdi
    add rax, rsi
    shr rax, 1
    mov r10, qword ptr [rbp - 8]
    mov r11, qword ptr [rbp - 16]
    mov rax, qword ptr [r10 + rax*8]
    cmp rax, rdx
    jg .Lright
    jl .Lleft
    mov rax, rdi
    add rax, rsi
    shr rax, 1
    jmp .Lfound
.Lright:
    mov rdi, rax
    inc rdi
    jmp .Lloop
.Lleft:
    mov rsi, rax
    dec rsi
    jmp .Lloop
.Lfound:
    pop rbp
    ret
.Lnot_found:
    mov rax, -1
    pop rbp
    ret
