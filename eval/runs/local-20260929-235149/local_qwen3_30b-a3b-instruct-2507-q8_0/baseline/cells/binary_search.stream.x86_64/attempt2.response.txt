binary_search:
    mov rax, -1
    cmp rsi, 0
    je .Ldone
    mov rdi, 0
    mov rsi, rsi
    dec rsi
.Lloop:
    mov rax, rdi
    add rax, rsi
    shr rax, 1
    mov r10, qword ptr [rdi + rax*8]
    cmp r10, rdx
    jg .Lright
    jl .Lleft
    ret
.Lright:
    mov rdi, rax
    inc rdi
    jmp .Lloop
.Lleft:
    mov rsi, rax
    dec rsi
    jmp .Lloop
.Ldone:
    ret
