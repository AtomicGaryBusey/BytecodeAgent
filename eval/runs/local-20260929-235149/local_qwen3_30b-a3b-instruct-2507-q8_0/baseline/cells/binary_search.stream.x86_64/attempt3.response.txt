binary_search:
    mov rax, -1
    cmp rsi, 0
    je .Ldone
    mov rbx, 0
    mov rcx, rsi
    dec rcx
.Lloop:
    mov rax, rbx
    add rax, rcx
    shr rax, 1
    mov r10, qword ptr [rdi + rax*8]
    cmp r10, rdx
    jg .Lright
    jl .Lleft
    ret
.Lright:
    mov rax, rbx
    add rax, rcx
    shr rax, 1
    inc rax
    mov rbx, rax
    jmp .Lloop
.Lleft:
    mov rax, rbx
    add rax, rcx
    shr rax, 1
    dec rax
    mov rcx, rax
    jmp .Lloop
.Ldone:
    ret
