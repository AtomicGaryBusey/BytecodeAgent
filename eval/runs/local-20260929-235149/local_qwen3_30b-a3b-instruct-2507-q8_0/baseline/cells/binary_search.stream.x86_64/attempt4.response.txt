binary_search:
    mov rax, -1
    cmp rsi, 0
    je .Ldone
    mov rbx, 0
    mov rcx, rsi
    dec rcx
    mov r12, rbx
    mov r13, rcx
.Lloop:
    mov rax, r12
    add rax, r13
    shr rax, 1
    mov r10, qword ptr [rdi + rax*8]
    cmp r10, rdx
    jg .Lright
    jl .Lleft
    mov rax, rax
    ret
.Lright:
    mov rax, r12
    add rax, r13
    shr rax, 1
    inc rax
    mov r12, rax
    jmp .Lloop
.Lleft:
    mov rax, r12
    add rax, r13
    shr rax, 1
    dec rax
    mov r13, rax
    jmp .Lloop
.Ldone:
    ret
