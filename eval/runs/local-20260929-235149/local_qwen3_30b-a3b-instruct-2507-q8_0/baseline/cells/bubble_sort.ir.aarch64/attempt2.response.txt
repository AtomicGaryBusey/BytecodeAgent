export function l $bubble_sort(l a, l n) {
    @entry:
        %i = const l 0
        %n_minus_1 = subl l n, 1
        jz w %n_minus_1, @return

    @loop1:
        %j = const l 0
        %swapped = const w 1

    @loop2:
        %j_addr = addl l a, mul l %j, 8
        %j_plus_1_addr = addl l a, mul l addl l %j, 1, 8
        %j_val = loadl l %j_addr
        %j_plus_1_val = loadl l %j_plus_1_addr
        %cmp = csltl l %j_val, %j_plus_1_val
        jz w %cmp, @skip_swap
        %temp = loadl l %j_plus_1_addr
        storel l %j_val, %j_plus_1_addr
        storel l %temp, %j_addr
        %swapped = const w 1

    @skip_swap:
        %j = addl l %j, 1
        %j_lt_n_minus_1 = csltl l %j, %n_minus_1
        jnz w %j_lt_n_minus_1, @loop2

        %i = addl l %i, 1
        %i_lt_n_minus_1 = csltl l %i, %n_minus_1
        jnz w %i_lt_n_minus_1, @loop1

    @return:
        %ret = const l 0
        ret l %ret
}
