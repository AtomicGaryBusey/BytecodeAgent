map_sum:
    push rbp
    mov rbp, rsp
    sub rsp, 16
    mov rax, 0
    mov rcx, 0
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    jmp .Lloop_start
.Lloop:
    mov rax, qword ptr [r8 + rcx*8]
    mov rdi, rax
    call r10
    add qword ptr [rbp-8], rax
    inc rcx
.Lloop_start:
    cmp rcx, r9
    jl .Lloop
    mov rax, qword ptr [rbp-8]
    mov rsp, rbp
    pop rbp
    ret
