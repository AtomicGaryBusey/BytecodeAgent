map_sum:
    xor rax, rax
    mov rcx, 0
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    jmp .Lloop_start
.Lloop:
    mov rax, qword ptr [r8 + rcx*8]
    mov rdi, rax
    call r10
    add rax, qword ptr [rsp]
    inc rcx
.Lloop_start:
    cmp rcx, r9
    jl .Lloop
    ret
    .Lloop_end:
    mov qword ptr [rsp], rax
    ret
