export function l $mulhi_u64(l a, l b) {
    @start:
        %a_low = andl a, 4294967295
        %a_high = shrl a, 32
        %b_low = andl b, 4294967295
        %b_high = shrl b, 32
        %p00 = mul %a_low, %b_low
        %p01 = mul %a_low, %b_high
        %p10 = mul %a_high, %b_low
        %p11 = mul %a_high, %b_high
        %c1 = shrl %p00, 32
        %p00 = andl %p00, 4294967295
        %p01 = add %p01, %c1
        %c2 = shrl %p01, 32
        %p01 = andl %p01, 4294967295
        %p10 = add %p10, %c2
        %c3 = shrl %p10, 32
        %p10 = andl %p10, 4294967295
        %p11 = add %p11, %c3
        %result = add %p11, %p10
        ret %result
}
