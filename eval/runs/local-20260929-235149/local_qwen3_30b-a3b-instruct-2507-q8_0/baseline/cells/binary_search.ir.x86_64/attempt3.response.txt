export function l $binary_search(l a, l n, l key) {
    @start:
        %low = l 0
        %high = l n
        %mid = l 0
        %mid_val = l 0
        %cmp = w 0
    @loop:
        %cmp = csltl %low, %high
        jnz %cmp, @not_found
        %mid = addl %low, %high
        %mid = shrl %mid, 1
        %mid_val = loadl [a + mulil %mid, 8]
        %cmp = csltl %mid_val, %key
        jnz %cmp, @less
        %cmp = csgtl %mid_val, %key
        jnz %cmp, @greater
        ret %mid
    @less:
        %low = addl %mid, 1
        jmp @loop
    @greater:
        %high = subl %mid, 1
        jmp @loop
    @not_found:
        ret l -1
}
