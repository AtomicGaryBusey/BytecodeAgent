collatz_max:
    push rbp
    mov rbp, rsp
    sub rsp, 16
    mov qword ptr [rbp - 8], rdi
    mov rax, 0
    mov rdi, 1
    jmp .Lloop

.Lloop:
    mov rsi, rdi
    mov rcx, 0
    mov rax, 1
    jmp .Lcollatz

.Lcollatz:
    cmp rsi, 1
    je .Lcollatz_done
    test rsi, 1
    jz .Leven
    mov rax, rsi
    lea rax, [rax + rax*2 + 1]
    mov rsi, rax
    inc rcx
    jmp .Lcollatz
.Leven:
    shr rsi, 1
    inc rcx
    jmp .Lcollatz

.Lcollatz_done:
    mov rax, rcx
    cmp rax, qword ptr [rbp - 16]
    jle .Lnext
    mov qword ptr [rbp - 16], rax
.Lnext:
    inc rdi
    cmp rdi, qword ptr [rbp - 8]
    jle .Lloop

    mov rax, qword ptr [rbp - 16]
    mov rsp, rbp
    pop rbp
    ret
