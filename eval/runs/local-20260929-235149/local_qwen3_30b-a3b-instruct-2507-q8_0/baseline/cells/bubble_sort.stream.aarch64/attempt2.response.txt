bubble_sort:
    mov x19, x0          // save base pointer
    mov x20, x1          // save n
    cmp x20, #1          // if n <= 1, return 0
    b.le .Lreturn

.Louter_loop:
    mov x21, #0          // i = 0
    mov x22, #0          // swapped = 0

.Linner_loop:
    cmp x21, x20         // if i >= n, break
    b.ge .Lnext_outer

    mov x23, #0          // j = 0

.Lcompare:
    cmp x23, x20         // if j >= n, break
    b.ge .Lnext_outer

    add x24, x19, x23, lsl #3  // address of a[j]
    add x25, x19, x23, lsl #3, add #8  // address of a[j+1]
