bubble_sort:
    cmp x1, #1
    b.le .Lreturn
    mov x19, x0
    mov x20, x1
    mov x21, #0

.Louter_loop:
    mov x22, #0
    mov x23, #0

.Linner_loop:
    cmp x23, x20
    b.ge .Lnext_outer
    add x24, x19, x23, lsl #3
    add x25, x24, #8
    ldr x26, [x24]
    ldr x27, [x25]
    cmp x26, x27
    b.le .Lnext_j
    str x27, [x24]
    str x26, [x25]
    mov x22, #1

.Lnext_j:
    add x23, x23, #1
    b .Linner_loop

.Lnext_outer:
    cmp x22, #0
    b.eq .Lreturn
    add x21, x21, #1
    b .Louter_loop

.Lreturn:
    mov x0, #0
    ret
