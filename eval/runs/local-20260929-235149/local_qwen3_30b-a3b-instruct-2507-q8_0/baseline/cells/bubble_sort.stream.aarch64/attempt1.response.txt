bubble_sort:
    mov x19, x0          // save base pointer
    mov x20, x1          // save n
    mov x21, x2          // save loop counter i
    mov x22, x3          // save loop counter j
    mov x23, x4          // save temp for swap
    mov x24, x5          // save flag for swaps
    mov x25, x6          // save loop counter i (outer)
    mov x26, x7          // save loop counter j (inner)

    cmp x20, #1          // if n <= 1, return 0
    b.le .Lreturn

.Louter_loop:
    mov x25, #0          // i = 0
    mov x24, #0          // swapped = 0

.Linner_loop:
    cmp x25, x20         // if i >= n, break
    b.ge .Lnext_outer

    mov x26, #0          // j = 0

.Lcompare:
    cmp x26, x20         // if j >= n-1, break
    b.ge .Lnext_outer

    add x27, x19, x26, lsl #3  // address of a[j]
    add x28, x19, x26, lsl #3, add #8  // address of a[j+1]
