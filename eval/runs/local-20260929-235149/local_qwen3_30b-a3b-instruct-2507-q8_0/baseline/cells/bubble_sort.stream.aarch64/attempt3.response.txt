bubble_sort:
    mov x19, x0          // save base pointer
    mov x20, x1          // save n
    cmp x20, #1          // if n <= 1, return 0
    b.le .Lreturn

.Louter_loop:
    mov x21, #0          // i = 0
    mov x22, #0          // swapped = 0

.Linner_loop:
    cmp x21, x20         // if i >= n, break
    b.ge .Lnext_outer

    mov x23, #0          // j = 0

.Lcompare:
    cmp x23, x20         // if j >= n, break
    b.ge .Lnext_outer

    add x24, x19, x23, lsl #3  // address of a[j]
    add x25, x24, #8         // address of a[j+1]

    ldr x26, [x24]           // load a[j]
    ldr x27, [x25]           // load a[j+1]

    cmp x26, x27             // if a[j] <= a[j+1], no swap
    b.le .Lnext_j

    // swap a[j] and a[j+1]
    str x27, [x24]           // store a[j+1] to a[j]
    str x26, [x25]           // store a[j] to a[j+1]
    mov x22, #1              // swapped = 1

.Lnext_j:
    add x23, x23, #1         // j++
    b .Lcompare

.Lnext_outer:
    cmp x22, #0              // if no swaps, array is sorted
    b.eq .Lreturn

    add x21, x21, #1         // i++
    b .Linner_loop

.Lreturn:
    mov x0, #0               // return 0
    ret
