export function l $collatz_max(l n) {
    @start:
        %max = l 0
        %k = l 0
        %loop = w 1
    @loop:
        %cmp = csltl %k, %n
        jnz %cmp, @body
        ret %max
    @body:
        %k = addl %k, 1
        %x = l %k
        %steps = l 0
        %step_loop = w 1
    @step_loop:
        %cmp = ceql %x, 1
        jnz %cmp, @step_body
        %max = csltl %steps, %max
        jnz %max, @loop
        jmp @loop
    @step_body:
        %even = andl %x, 1
        jnz %even, @even_case
        %x = addl %x, %x
        %x = addl %x, 1
        %x = srl %x, 1
        jmp @step_loop
    @even_case:
        %x = srl %x, 1
        jmp @step_loop
}
