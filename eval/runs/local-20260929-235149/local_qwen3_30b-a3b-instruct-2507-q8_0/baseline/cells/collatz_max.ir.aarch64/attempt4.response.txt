export function l $collatz_max(l n) {
    @start:
        %max = l 0
        %k = l 1
        %loop = w 1
    @loop:
        %cmp = csltl %k, %n
        jnz %cmp, @body
        %cmp = ceql %k, %n
        jnz %cmp, @body
        ret %max
    @body:
        %x = l %k
        %steps = l 0
        %step_loop = w 1
    @step_loop:
        %cmp = ceql %x, 1
        jnz %cmp, @step_body
        %cmp = csltl %steps, %max
        jnz %cmp, @update_max
        jmp @next_k
    @step_body:
        %even = andl %x, 1
        jnz %even, @even_case
        %x = addl %x, %x
        %x = addl %x, 1
        %x = srl %x, 1
        jmp @step_loop
    @even_case:
        %x = srl %x, 1
        jmp @step_loop
    @update_max:
        %max = l %steps
        jmp @next_k
    @next_k:
        %k = addl %k, 1
        jmp @loop
}
