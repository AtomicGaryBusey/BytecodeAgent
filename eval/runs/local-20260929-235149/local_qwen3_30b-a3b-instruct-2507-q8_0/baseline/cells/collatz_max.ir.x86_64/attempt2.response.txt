export function l $collatz_max(l n) {
    @start:
        %max = l 0
        %k = l 1
        %loop = b $loop_body $loop_end
    @loop_body:
        %steps = l 0
        %x = l %k
        %step_loop = b $step_body $step_end
    @step_body:
        %x_eq_1 = ceql %x, 1
        jnz %x_eq_1, $step_end
        %x_even = csltl %x, 0
        jnz %x_even, $step_even
        %x = l %x * 3 + 1
        jmp $step_loop_end
    @step_even:
        %x = l %x / 2
        jmp $step_loop_end
    @step_loop_end:
        %steps = l %steps + 1
        %loop = b $step_body $step_end
    @step_end:
        %max_gt_steps = csltl %max, %steps
        jnz %max_gt_steps, $update_max
        jmp $update_max_end
    @update_max:
        %max = l %steps
    @update_max_end:
        %k_lt_n = csltl %k, %n
        jnz %k_lt_n, $loop_body
    @loop_end:
        %result = l %max
        ret %result
}
