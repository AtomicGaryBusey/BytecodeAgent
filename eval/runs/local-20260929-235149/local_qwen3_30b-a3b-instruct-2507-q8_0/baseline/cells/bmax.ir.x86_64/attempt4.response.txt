export function l $bmax(l a, l n) {
    @entry:
        %cmp = csltl %n, 1
        jnz %cmp, @panic
        %max = loadl %a
        %i = 1
        %loop = @loop
    @loop:
        %i_lt_n = csltl %i, %n
        jz %i_lt_n, @end
        %a_i = addl %a, mulil %i, 8
        %a_i_val = loadl %a_i
        %max_lt_a_i = csltl %max, %a_i_val
        jz %max_lt_a_i, @loop_next
        %max = %a_i_val
    @loop_next:
        %i = addl %i, 1
        jump @loop
    @end:
        return %max
    @panic:
        unreachable
}
