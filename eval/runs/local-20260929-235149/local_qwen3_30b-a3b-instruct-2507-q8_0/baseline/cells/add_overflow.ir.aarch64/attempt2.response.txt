export function l $add_overflow(l a, l b, l out) {
    @entry:
        %sum = addl a, b
        %overflow = addl a, b
        %sign_a = sracl a, 63
        %sign_b = sracl b, 63
        %sign_sum = sracl %sum, 63
        %not_same_sign = ceqwl %sign_a, %sign_b
        %diff_sign_sum = ceqwl %sign_a, %sign_sum
        %overflow_detected = andw %not_same_sign, %diff_sign_sum
        storel %sum, out
        ret %overflow_detected
}
