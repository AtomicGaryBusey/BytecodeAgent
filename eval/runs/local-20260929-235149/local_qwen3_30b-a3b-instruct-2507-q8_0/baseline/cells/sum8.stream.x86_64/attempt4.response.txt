    mov rax, qword ptr [rsp + 56]  # a
    mov rbx, qword ptr [rsp + 48]  # b
    mov rcx, qword ptr [rsp + 40]  # c
    mov rdx, qword ptr [rsp + 32]  # d
    mov rsi, qword ptr [rsp + 24]  # e
    mov rdi, qword ptr [rsp + 16]  # f
    mov r8, qword ptr [rsp + 8]    # g
    mov r9, qword ptr [rsp]        # h

    # Compute +1*a
    # rax = a

    # Compute -2*b
    mov r10, rbx
    shl r10, 1
    neg r10
    add rax, r10

    # Compute +3*c
    mov r10, rcx
    shl r10, 1
    add r10, rcx
    add rax, r10

    # Compute -4*d
    mov r10, rdx
    shl r10, 2
    neg r10
    add rax, r10

    # Compute +5*e
    mov r10, rsi
    shl r10, 2
    add r10, rsi
    add rax, r10

    # Compute -6*f
    mov r10, rdi
    shl r10, 2
    add r10, rdi
    shl r10, 1
    neg r10
    add rax, r10

    # Compute +7*g
    mov r10, r8
    shl r10, 3
    sub r10, r8
    add rax, r10

    # Compute -8*h
    mov r10, r9
    shl r10, 3
    neg r10
    add rax, r10

    ret
