export function l $fib(l n) {
    @entry:
        %cmp = csltl n, 2l
        jnz %cmp, @small
        %ret = movl n
        ret %ret

    @small:
        %a = movl 0l
        %b = movl 1l
        %i = movl 2l
        @loop:
            %cmp = csltl %i, n
            jz %cmp, @done
            %next = addl %a, %b
            %a = %b
            %b = %next
            %i = addl %i, 1l
            jmp @loop

    @done:
        ret %b
}
