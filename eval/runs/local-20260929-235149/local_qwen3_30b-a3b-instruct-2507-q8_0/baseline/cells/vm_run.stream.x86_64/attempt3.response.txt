vm_run:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    mov r12, rdi          # code
    mov r13, rsi          # len
    mov r14, rdx          # stack
    mov r15, rcx          # cap
    xor rax, rax          # pc = 0
    xor rbx, rbx          # sp = 0
    xor rdi, rdi          # d = 0

    # Initialize stack to zeros
    mov rdi, r15
    mov rsi, r14
    xor rdx, rdx
    mov rcx, r15
    shr rcx, 3
    jz .stack_init_done
    .stack_init_loop:
        mov qword ptr [rsi + rcx*8 - 8], rdx
        dec rcx
        jnz .stack_init_loop
    .stack_init_done:

    # Main loop
    .loop:
        cmp rax, r13
        jge .done
        mov rdi, qword ptr [r12 + rax*8]
        cmp rdi, 11
        jg .default
        cmp rdi, 0
        je .halt
        cmp rdi, 1
        je .push
        cmp rdi, 2
        je .add
        cmp rdi, 3
        je .sub
        cmp rdi, 4
        je .mul
        cmp rdi, 5
        je .dup
        cmp rdi, 6
        je .swap
        cmp rdi, 7
        je .drop
        cmp rdi, 8
        je .over
        cmp rdi, 9
        je .lt
        cmp rdi, 10
        je .jz
        cmp rdi, 11
        je .jmp
        jmp .default

    .halt:
        jmp .done

    .push:
        mov rdi, qword ptr [r12 + rax*8 + 8]
        mov qword ptr [r14 + rbx*8], rdi
        inc rbx
        add rax, 2
        jmp .loop

    .add:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov rsi, qword ptr [r14 + rbx*8 - 16]
        add rsi, rdi
        mov qword ptr [r14 + rbx*8 - 16], rsi
        dec rbx
        add rax, 1
        jmp .loop

    .sub:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov rsi, qword ptr [r14 + rbx*8 - 16]
        sub rsi, rdi
        mov qword ptr [r14 + rbx*8 - 16], rsi
        dec rbx
        add rax, 1
        jmp .loop

    .mul:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov rsi, qword ptr [r14 + rbx*8 - 16]
        imul rsi, rdi
        mov qword ptr [r14 + rbx*8 - 16], rsi
        dec rbx
        add rax, 1
        jmp .loop

    .dup:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov qword ptr [r14 + rbx*8], rdi
        inc rbx
        add rax, 1
        jmp .loop

    .swap:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov rsi, qword ptr [r14 + rbx*8 - 16]
        mov qword ptr [r14 + rbx*8 - 8], rsi
        mov qword ptr [r14 + rbx*8 - 16], rdi
        add rax, 1
        jmp .loop

    .drop:
        dec rbx
        add rax, 1
        jmp .loop

    .over:
        mov rdi, qword ptr [r14 + rbx*8 - 16]
        mov qword ptr [r14 + rbx*8], rdi
        inc rbx
        add rax, 1
        jmp .loop

    .lt:
        mov rdi, qword ptr [r14 + rbx*8 - 8]
        mov rsi, qword ptr [r14 + rbx*8 - 16]
        cmp rsi, rdi
        setl al
        movzx rax, al
        mov qword ptr [r14 + rbx*8 - 16], rax
        dec rbx
        add rax, 1
        jmp .loop

    .jz:
        mov rdi, qword ptr [r12 + rax*8 + 8]
        test rdi, rdi
        jz .jz_target
        add rax, 2
        jmp .loop
    .jz_target:
        mov rax, qword ptr [r12 + rax*8 + 8]
        jmp .loop

    .jmp:
        mov rax, qword ptr [r12 + rax*8 + 8]
        jmp .loop

    .default:
        add rax, 1
        jmp .loop

    .done:
        mov rax, rbx
        test rax, rax
        jz .return_zero
        mov rax, qword ptr [r14 + rbx*8 - 8]
        jmp .return

    .return_zero:
        xor rax, rax

    .return:
        # Zero out unused stack slots
        mov rcx, r15
        sub rcx, rbx
        mov rdi, r14
        xor rsi, rsi
        mov rdx, rcx
        shr rdx, 3
        jz .return_done
        .zero_loop:
            mov qword ptr [rdi + rcx*8 - 8], rsi
            dec rcx
            jnz .zero_loop
        .return_done:
        add rsp, 32
        pop rbp
        ret
