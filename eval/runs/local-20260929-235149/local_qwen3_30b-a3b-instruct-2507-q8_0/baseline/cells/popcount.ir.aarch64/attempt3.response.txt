export function l $popcount(l x) {
    @start:
        %acc = constl 0
        %mask = constl 1
        %count = constl 64
        @loop:
            %test = cneql x, constl 0
            jnz %test, @body
            ret %acc
        @body:
            %bit = andl x, %mask
            %new_acc = addl %acc, %bit
            %x_shifted = shrl x, constl 1
            %count_minus = subl %count, constl 1
            %count_minus_eq_0 = ceql %count_minus, constl 0
            jnz %count_minus_eq_0, @done
            %x = %x_shifted
            %acc = %new_acc
            %count = %count_minus
            jmp @loop
        @done:
            ret %acc
}
