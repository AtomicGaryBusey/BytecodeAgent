isqrt:
    mov x19, x0           // save n in x19
    mov x20, #0           // low = 0
    mov x21, #0x80000000  // high = 2^31, will be shifted up
    mov x22, #32          // bit position counter

.loop:
    cmp x22, #0           // if bit position < 0, done
    b.eq  .done

    lsr x23, x21, #1      // mid = (low + high) / 2, but we're building it bit by bit
    add x23, x20, x23     // candidate = low + mid

    mul x24, x23, x23     // candidate * candidate
    cmp x24, x19          // compare with n
    b.hs  .skip           // if candidate^2 > n, skip

    mov x20, x23          // low = candidate
    add x21, x21, x21     // high = high * 2
    b     .next

.skip:
    add x21, x21, x21     // high = high * 2
    sub x21, x21, #1      // high = high - 1

.next:
    sub x22, x22, #1      // decrement bit position
    b     .loop

.done:
    mov x0, x20           // return low
    ret
