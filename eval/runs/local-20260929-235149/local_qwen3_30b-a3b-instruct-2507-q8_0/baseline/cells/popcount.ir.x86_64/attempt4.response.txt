export function l $popcount(l x) {
    @entry:
        %acc = constl 0
        %mask = constl 1
        %count = constl 64
        %loop:
            %test = csltl %count, constl 0
            jnz %test, @exit
            %bit = andl %x, %mask
            %acc = addl %acc, %bit
            %x = shrl %x, constl 1
            %count = addl %count, constl -1
            jump @loop
        @exit:
            ret %acc
}
