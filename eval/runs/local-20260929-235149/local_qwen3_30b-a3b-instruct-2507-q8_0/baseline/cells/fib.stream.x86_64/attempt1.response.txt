fib:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    mov r12, 0
    mov r13, 1
    cmp rdi, 1
    jbe .Ldone
    mov rbx, 2
.Lloop:
    mov rax, r12
    add rax, r13
    mov r12, r13
    mov r13, rax
    inc rbx
    cmp rbx, rdi
    jne .Lloop
.Ldone:
    mov rax, r13
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
