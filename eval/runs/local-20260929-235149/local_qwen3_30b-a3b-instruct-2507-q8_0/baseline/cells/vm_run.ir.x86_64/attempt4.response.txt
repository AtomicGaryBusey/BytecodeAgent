export function l $vm_run(l code, l len, l stack, l cap) {
    @entry:
        %pc = 0
        %sp = 0
        %stack_ptr = stack
        %stack_end = stack + cap * 8
        %zero = 0
        %one = 1
        %two = 2
        %eight = 8
        %max_ops = 5000
        %ops = 0

    @loop:
        %pc_ge_len = csltl %pc, %len
        jz %pc_ge_len, @exit
        %ops_ge_max = csltl %ops, %max_ops
        jz %ops_ge_max, @exit
        %op = loadl l [code + %pc * 8]
        %pc_next = %pc + %two
        %op_is_halt = ceql %op, %zero
        jnz %op_is_halt, @case_1
        jmp @exit

    @case_1:
        %op_is_push = ceql %op, %one
        jz %op_is_push, @case_2
        %value = loadl l [code + %pc_next * 8]
        %sp_next = %sp + %one
        %stack_ptr_next = %stack_ptr + %eight
        storel l %value, [%stack_ptr_next - %eight]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_2:
        %op_is_add = ceql %op, %two
        jz %op_is_add, @case_3
        %b = loadl l [%stack_ptr - %eight]
        %a = loadl l [%stack_ptr - %eight * 2]
        %sum = %a + %b
        %sp_next = %sp - %one
        %stack_ptr_next = %stack_ptr - %eight
        storel l %sum, [%stack_ptr_next]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_3:
        %op_is_sub = ceql %op, %three
        jz %op_is_sub, @case_4
        %b = loadl l [%stack_ptr - %eight]
        %a = loadl l [%stack_ptr - %eight * 2]
        %diff = %a - %b
        %sp_next = %sp - %one
        %stack_ptr_next = %stack_ptr - %eight
        storel l %diff, [%stack_ptr_next]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_4:
        %op_is_mul = ceql %op, %four
        jz %op_is_mul, @case_5
        %b = loadl l [%stack_ptr - %eight]
        %a = loadl l [%stack_ptr - %eight * 2]
        %prod = %a * %b
        %sp_next = %sp - %one
        %stack_ptr_next = %stack_ptr - %eight
        storel l %prod, [%stack_ptr_next]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_5:
        %op_is_dup = ceql %op, %five
        jz %op_is_dup, @case_6
        %top = loadl l [%stack_ptr - %eight]
        %sp_next = %sp + %one
        %stack_ptr_next = %stack_ptr + %eight
        storel l %top, [%stack_ptr_next - %eight]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_6:
        %op_is_swap = ceql %op, %six
        jz %op_is_swap, @case_7
        %top = loadl l [%stack_ptr - %eight]
        %next = loadl l [%stack_ptr - %eight * 2]
        storel l %top, [%stack_ptr - %eight * 2]
        storel l %next, [%stack_ptr - %eight]
        %pc = %pc_next
        jmp @loop

    @case_7:
        %op_is_drop = ceql %op, %seven
        jz %op_is_drop, @case_8
        %sp_next = %sp - %one
        %stack_ptr_next = %stack_ptr - %eight
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_8:
        %op_is_over = ceql %op, %eight
        jz %op_is_over, @case_9
        %below = loadl l [%stack_ptr - %eight * 2]
        %sp_next = %sp + %one
        %stack_ptr_next = %stack_ptr + %eight
        storel l %below, [%stack_ptr_next - %eight]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_9:
        %op_is_lt = ceql %op, %nine
        jz %op_is_lt, @case_10
        %b = loadl l [%stack_ptr - %eight]
        %a = loadl l [%stack_ptr - %eight * 2]
        %a_lt_b = csltl %a, %b
        %sp_next = %sp - %one
        %stack_ptr_next = %stack_ptr - %eight
        storel l %a_lt_b, [%stack_ptr_next]
        %pc = %pc_next
        %sp = %sp_next
        %stack_ptr = %stack_ptr_next
        jmp @loop

    @case_10:
        %op_is_jz = ceql %op, %ten
        jz %op_is_jz, @case_11
        %value = loadl l [code + %pc_next * 8]
        %value_eq_zero = ceql %value, %zero
        jnz %value_eq_zero, @jz_else
        %pc = %value
        jmp @loop
    @jz_else:
        %pc = %pc_next
        jmp @loop

    @case_11:
        %op_is_jmp = ceql %op, %eleven
        jz %op_is_jmp, @default
        %target = loadl l [code + %pc_next * 8]
        %pc = %target
        jmp @loop

    @default:
        %pc = %pc_next
        jmp @loop

    @exit:
        %stack_empty = ceql %sp, %zero
        jz %stack_empty, @not_empty
        %result = %zero
        jmp @return

    @not_empty:
        %result = loadl l [%stack_ptr - %eight]

    @return:
        %sp_zeroed = %sp
        %stack_ptr_zeroed = %stack_ptr
        %zero_count = cap
        %zero_loop:
            %zero_count_eq_zero = ceql %zero_count, %zero
            jz %zero_count_eq_zero, @zero_done
            %stack_ptr_zeroed = %stack_ptr_zeroed - %eight
            storel l %zero, [%stack_ptr_zeroed]
            %zero_count = %zero_count - %one
            jmp @zero_loop

    @zero_done:
        ret %result
}
