@label
l %r =l loadl(l %n)
w %c =w ceql(l %r, l 0)
jnz(w %c, @label)
l %res =l const 0
ret(l %res)

@label
l %r =l loadl(l %n)
w %c =w ceql(l %r, l 1)
jnz(w %c, @label)
l %res =l const 1
ret(l %res)

@label
l %r =l loadl(l %n)
l %n1 =l sub(l %r, l 1)
l %n2 =l sub(l %r, l 2)
l %a =l call $fib_rec(l %n1)
l %b =l call $fib_rec(l %n2)
l %res =l add(l %a, l %b)
ret(l %res)
