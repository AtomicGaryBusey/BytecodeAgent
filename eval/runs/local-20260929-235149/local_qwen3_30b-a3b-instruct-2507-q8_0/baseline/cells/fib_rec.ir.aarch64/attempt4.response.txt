export function l $fib_rec(l %n) {
    @entry
    l %r = l loadl(l %n)
    w %c = w ceql(l %r, l 0)
    jnz(w %c, @base0)

    l %res = l const 0
    ret(l %res)

    @base0
    l %r = l loadl(l %n)
    w %c = w ceql(l %r, l 1)
    jnz(w %c, @base1)

    l %res = l const 1
    ret(l %res)

    @base1
    l %r = l loadl(l %n)
    l %n1 = l sub(l %r, l 1)
    l %n2 = l sub(l %r, l 2)
    l %a = l call $fib_rec(l %n1)
    l %b = l call $fib_rec(l %n2)
    l %res = l add(l %a, l %b)
    ret(l %res)
}
