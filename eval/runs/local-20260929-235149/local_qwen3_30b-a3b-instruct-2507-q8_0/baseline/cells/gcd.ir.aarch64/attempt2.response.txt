export function l $gcd(l a, l b) {
    @entry:
        %cmp = ceql b, 0w
        jnz %cmp, @base_case
        ret a

    @base_case:
        %cmp = ceql a, 0w
        jnz %cmp, @swap
        ret b

    @swap:
        %temp = movl b
        %b = movl a
        %a = movl %temp
        %div = sdivl a, b
        %mod = smodl a, b
        %a = movl b
        %b = movl %mod
        %cmp = ceql %b, 0w
        jnz %cmp, @entry
        ret %a
}
