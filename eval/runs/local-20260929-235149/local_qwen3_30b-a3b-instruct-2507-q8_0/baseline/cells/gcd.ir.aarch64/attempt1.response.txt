export function l $gcd(l a, l b) {
    @entry:
        %cmp = ceql a, 0w
        jnz %cmp, @base_case
        %cmp2 = ceql b, 0w
        jnz %cmp2, @base_case
        %result = movl a
        ret %result

    @base_case:
        %cmp = ceql b, 0w
        jnz %cmp, @swap
        %result = movl a
        ret %result

    @swap:
        %temp = movl b
        %b = movl a
        %a = movl %temp
        %div = sdivl a, b
        %mod = smodl a, b
        %a = movl b
        %b = movl %mod
        %cmp = ceql %b, 0w
        jnz %cmp, @entry
        %result = movl %a
        ret %result
}
