export function l $gcd(l a, l b) {
    @start:
        %cmp = ceql a, 0
        jnz %cmp, @base_case
        %cmp2 = ceql b, 0
        jnz %cmp2, @base_case
        %result = copy a
        ret %result

    @base_case:
        %cmp = ceql b, 0
        jnz %cmp, @loop
        ret a

    @loop:
        %r = modl a, b
        %a = copy b
        %b = copy %r
        %cmp = ceql %b, 0
        jnz %cmp, @loop
        ret %a
}
