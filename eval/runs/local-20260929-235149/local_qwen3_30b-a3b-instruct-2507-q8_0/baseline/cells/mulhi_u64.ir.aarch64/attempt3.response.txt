export function l $mulhi_u64(l a, l b) {
    @start:
    %a_low = andl a, 4294967295
    %a_high = shrl a, 32
    %b_low = andl b, 4294967295
    %b_high = shrl b, 32
    %p00 = mulw %a_low, %b_low
    %p01 = mulw %a_low, %b_high
    %p10 = mulw %a_high, %b_low
    %p11 = mulw %a_high, %b_high
    %c1 = addw %p01, %p10
    %c2 = addw %c1, %p00
    %c3 = shrl %c2, 32
    %c4 = addw %p11, %c3
    %c5 = shrl %c4, 32
    %c6 = addw %c5, %p11
    %result = shrl %c6, 32
    ret %result
}
