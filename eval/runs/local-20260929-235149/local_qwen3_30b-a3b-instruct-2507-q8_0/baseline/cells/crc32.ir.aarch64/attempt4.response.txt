export function l $crc32(l %p, l %n) {
    @entry:
        %crc =l const 0xFFFFFFFF
        %i =l const 0
        jnz %n, @loop_body

    @loop_body:
        %b =w loadub %p
        %crc =w xor %crc, %b
        %j =w const 0
        jnz %j, @bit_loop

    @bit_loop:
        %bit =w and %crc, 1
        jz %bit, @shift_only
        %crc =w xor %crc, 0xEDB88320
    @shift_only:
        %crc =w shr %crc, 1
        %j =w add %j, 1
        %j =w lt %j, 8
        jnz %j, @bit_loop

        %i =w add %i, 1
        %p =l add %p, 1
        %i =w lt %i, %n
        jnz %i, @loop_body

        %crc =w xor %crc, 0xFFFFFFFF
        %crc =l extuw %crc
        ret %crc
}
