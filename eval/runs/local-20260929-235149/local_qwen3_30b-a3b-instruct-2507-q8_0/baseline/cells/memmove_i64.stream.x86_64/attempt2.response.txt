memmove_i64:
    test rdx, rdx
    jz .done

    cmp rdi, rsi
    jge .copy_backward

    mov rcx, rdx
    mov rax, rdi
    mov rbx, rsi
    jmp .copy_forward_loop

.copy_backward:
    mov rcx, rdx
    lea rax, [rdi + rdx*8 - 8]
    lea rbx, [rsi + rdx*8 - 8]
    jmp .copy_backward_loop

.copy_forward_loop:
    test rcx, rcx
    jz .done
    mov rax, qword ptr [rbx]
    mov qword ptr [rax], rax
    add rax, 8
    add rbx, 8
    sub rcx, 1
    jmp .copy_forward_loop

.copy_backward_loop:
    test rcx, rcx
    jz .done
    mov rax, qword ptr [rbx]
    mov qword ptr [rax], rax
    sub rax, 8
    sub rbx, 8
    sub rcx, 1
    jmp .copy_backward_loop

.done:
    xor rax, rax
    ret
