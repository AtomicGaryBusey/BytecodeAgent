memmove_i64:
    # Check if n is zero
    test rdx, rdx
    jz .done

    # Check if dst and src overlap and determine direction
    cmp rdi, rsi
    jge .copy_backward

    # Forward copy: dst after src or no overlap
    mov rcx, rdx
    mov rax, rdi
    mov rbx, rsi
    jmp .copy_forward_loop

.copy_backward:
    # Backward copy: dst before src
    mov rcx, rdx
    lea rax, [rdi + rdx*8 - 8]
    lea rbx, [rsi + rdx*8 - 8]
    jmp .copy_backward_loop

.copy_forward_loop:
    mov rdi, rax
    mov rsi, rbx
    mov rdx, rcx
    test rdx, rdx
    jz .done
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    add rdi, 8
    add rsi, 8
    sub rcx, 1
    jmp .copy_forward_loop

.copy_backward_loop:
    mov rdi, rax
    mov rsi, rbx
    mov rdx, rcx
    test rdx, rdx
    jz .done
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    sub rdi, 8
    sub rsi, 8
    sub rcx, 1
    jmp .copy_backward_loop

.done:
    xor rax, rax
    ret
