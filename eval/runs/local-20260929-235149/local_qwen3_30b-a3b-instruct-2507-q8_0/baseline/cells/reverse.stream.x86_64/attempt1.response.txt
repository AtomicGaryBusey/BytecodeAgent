reverse:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    mov r12, rdi          # r12 = a
    mov r13, rsi          # r13 = n
    xor rbx, rbx          # rbx = i = 0
    cmp r13, 1
    jle .Ldone            # if n <= 1, nothing to do
.Lloop:
    mov rax, r13
    sub rax, rbx
    dec rax               # rax = n - 1 - i
    mov rcx, rbx
    mov rdx, rax
    mov rax, qword ptr [r12 + rcx*8]  # rax = a[i]
    mov rdi, qword ptr [r12 + rdx*8]  # rdi = a[n-1-i]
    mov qword ptr [r12 + rcx*8], rdi  # a[i] = a[n-1-i]
    mov qword ptr [r12 + rdx*8], rax  # a[n-1-i] = a[i]
    inc rbx               # i++
    cmp rbx, r13
    jge .Ldone
    shr r13, 1
    cmp rbx, r13
    jle .Lloop
.Ldone:
    xor rax, rax          # return 0
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
