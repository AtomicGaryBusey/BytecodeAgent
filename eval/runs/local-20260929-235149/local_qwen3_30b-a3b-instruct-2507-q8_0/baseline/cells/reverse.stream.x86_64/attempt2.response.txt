reverse:
    mov rax, rdi          # rax = a
    mov rcx, rsi          # rcx = n
    xor rbx, rbx          # rbx = i = 0
    cmp rcx, 1
    jle .Ldone            # if n <= 1, nothing to do
.Lloop:
    mov rdx, rcx
    sub rdx, rbx
    dec rdx               # rdx = n - 1 - i
    mov rdi, qword ptr [rax + rbx*8]  # rdi = a[i]
    mov rsi, qword ptr [rax + rdx*8]  # rsi = a[n-1-i]
    mov qword ptr [rax + rbx*8], rsi  # a[i] = a[n-1-i]
    mov qword ptr [rax + rdx*8], rdi  # a[n-1-i] = a[i]
    inc rbx               # i++
    cmp rbx, rcx
    jge .Ldone
    shr rcx, 1
    cmp rbx, rcx
    jle .Lloop
.Ldone:
    xor rax, rax          # return 0
    ret
