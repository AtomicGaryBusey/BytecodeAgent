reverse:
    mov rax, rdi
    mov rcx, rsi
    xor rbx, rbx
    cmp rcx, 1
    jle .Ldone
.Lloop:
    mov rdx, rcx
    sub rdx, rbx
    dec rdx
    mov rdi, qword ptr [rax + rbx*8]
    mov rsi, qword ptr [rax + rdx*8]
    mov qword ptr [rax + rbx*8], rsi
    mov qword ptr [rax + rdx*8], rdi
    inc rbx
    cmp rbx, rcx
    jge .Ldone
    shr rcx, 1
    cmp rbx, rcx
    jle .Lloop
.Ldone:
    xor rax, rax
    ret
