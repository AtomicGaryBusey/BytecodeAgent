@label $fib_rec
    %n = l arg
    %cmp0 = w csltl l %n, l 2
    jnz %cmp0, @label $fib_rec_base
    %cmp1 = w ceql l %n, l 1
    jnz %cmp1, @label $fib_rec_base
    %ret = l 1
    ret l %ret
@label $fib_rec_base
    %cmp2 = w ceql l %n, l 0
    jnz %cmp2, @label $fib_rec_rec
    %ret = l 0
    ret l %ret
@label $fib_rec_rec
    %n1 = l sub l %n, l 1
    %call1 = l call $fib_rec(l %n1)
    %n2 = l sub l %n, l 2
    %call2 = l call $fib_rec(l %n2)
    %sum = l add l %call1, l %call2
    ret l %sum
