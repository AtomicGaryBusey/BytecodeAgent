export function l $fib(l n) {
    @entry:
        %cmp = csltl n, 2
        jnz %cmp, @if1, @else1
    @if1:
        %res = ceql n, 0
        jnz %res, @return0, @return1
    @return0:
        ret 0
    @return1:
        ret 1
    @else1:
        %a = loadl 0
        %b = loadl 1
        %i = loadl 2
        %loop:
            %cmp2 = csltl %i, n
            jnz %cmp2, @loop_body, @return
        @loop_body:
            %next = addl %a, %b
            %a = %b
            %b = %next
            %i = addl %i, 1
            j @loop
        @return:
            ret %b
}
