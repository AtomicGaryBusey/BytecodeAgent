sum8:
    // x0 = a, x1 = b, x2 = c, x3 = d, x4 = e, x5 = f, x6 = g, x7 = h
    // Compute: a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h
    // Use x19 for intermediate values, preserve x20-x28

    mov x19, x0           // x19 = a
    mov x20, x1           // x20 = b
    mov x21, x2           // x21 = c
    mov x22, x3           // x22 = d
    mov x23, x4           // x23 = e
    mov x24, x5           // x24 = f
    mov x25, x6           // x25 = g
    mov x26, x7           // x26 = h

    // Compute -2*b
    lsl x20, x20, #1      // x20 = 2*b
    neg x20, x20          // x20 = -2*b

    // Compute 3*c
    add x21, x21, x21     // x21 = 2*c
    add x21, x21, x21     // x21 = 4*c
    sub x21, x21, x21     // x21 = 3*c (by subtracting c)
    // Actually: 3*c = 2*c + c
    add x21, x21, x21     // x21 = 2*c
    add x21, x21, x21     // x21 = 4*c
    sub x21, x21, x21     // x21 = 3*c (by subtracting c)

    // Better: 3*c = 2*c + c
    add x21, x21, x21     // x21 = 2*c
    add x21, x21, x21     // x21 = 3*c

    // Compute -4*d
    lsl x22, x22, #2      // x22 = 4*d
    neg x22, x22          // x22 = -4*d

    // Compute 5*e
    add x23, x23, x23     // x23 = 2*e
    add x23, x23, x23     // x23 = 4*e
    add x23, x23, x23     // x23 = 5*e

    // Compute -6*f
    add x24, x24, x24     // x24 = 2*f
    add x24, x24, x24     // x24 = 4*f
    add x24, x24, x24     // x24 = 6*f
    neg x24, x24          // x24 = -6*f

    // Compute 7*g
    add x25, x25, x25     // x25 = 2*g
    add x25, x25, x25     // x25 = 4*g
    add x25, x25, x25     // x25 = 8*g
    sub x25, x25, x25     // x25 = 7*g (by subtracting g)

    // Compute -8*h
    lsl x26, x26, #3      // x26 = 8*h
    neg x26, x26          // x26 = -8*h

    // Sum all terms: a + (-2*b) + (3*c) + (-4*d) + (5*e) + (-6*f) + (7*g) + (-8*h)
    add x19, x19, x20     // x19 = a - 2*b
    add x19, x19, x21     // x19 = a - 2*b + 3*c
    add x19, x19, x22     // x19 = a - 2*b + 3*c - 4*d
    add x19, x19, x23     // x19 = a - 2*b + 3*c - 4*d + 5*e
    add x19, x19, x24     // x19 = a - 2*b + 3*c - 4*d + 5*e - 6*f
    add x19, x19, x25     // x19 = a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g
    add x19, x19, x26     // x19 = a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h

    mov x0, x19           // x0 = result
    ret                   // return
