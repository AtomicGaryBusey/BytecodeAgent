dot_i32_simd:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, 0
    mov rbx, 0
    cmp r14, 4
    jl .Lscalar
    mov rax, r14
    and rax, 3
    sub r14, rax
    mov rdi, r12
    mov rsi, r13
    mov rdx, r14
    shr rdx, 2
    mov rax, 0
    pxor xmm0, xmm0
    pxor xmm1, xmm1
    pxor xmm2, xmm2
    pxor xmm3, xmm3
    .Lloop:
    movdqu xmm4, xmmword ptr [rdi + rbx*4]
    movdqu xmm5, xmmword ptr [rsi + rbx*4]
    pmovsxdq xmm4, xmm4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm0, xmm4
    movdqu xmm4, xmmword ptr [rdi + rbx*4 + 16]
    movdqu xmm5, xmmword ptr [rsi + rbx*4 + 16]
    pmovsxdq xmm4, xmm4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm1, xmm4
    movdqu xmm4, xmmword ptr [rdi + rbx*4 + 32]
    movdqu xmm5, xmmword ptr [rsi + rbx*4 + 32]
    pmovsxdq xmm4, xmm4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm2, xmm4
    movdqu xmm4, xmmword ptr [rdi + rbx*4 + 48]
    movdqu xmm5, xmmword ptr [rsi + rbx*4 + 48]
    pmovsxdq xmm4, xmm4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm3, xmm4
    add rbx, 16
    dec rdx
    jnz .Lloop
    movdqa xmm4, xmm0
    movdqa xmm5, xmm1
    movdqa xmm6, xmm2
    movdqa xmm7, xmm3
    punpckhqdq xmm0, xmm1
    punpckhqdq xmm2, xmm3
    paddq xmm0, xmm2
    paddq xmm4, xmm5
    paddq xmm6, xmm7
    paddq xmm0, xmm6
    movq rax, xmm0
    mov rdi, r12
    mov rsi, r13
    mov rdx, r14
    add rax, rbx
    mov r14, 0
    .Lscalar:
    cmp r14, 0
    je .Ldone
    mov rax, 0
    .Lscalar_loop:
    mov eax, dword ptr [rdi + r14*4]
    imul eax, dword ptr [rsi + r14*4]
    add rax, rdx
    inc r14
    cmp r14, r15
    jl .Lscalar_loop
    .Ldone:
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
