count_nonzero_prefix:
    mov x1, x0          // x1 = a
    mov x2, x1          // x2 = a (will be used as pointer)
    mov x3, xzr         // x3 = 0 (count)
    cbz x1, .Ldone      // if a is null, return 0
    cbz x2, .Ldone      // if n is 0, return 0
    mov x4, xzr         // x4 = 0 (index)
    mov x5, xzr         // x5 = 0 (zero flag)

.Lloop:
    ldr x6, [x2, x4, lsl #3]  // x6 = a[i]
    cbz x6, .Ldone          // if a[i] == 0, break
    add x4, x4, #1          // i++
    cmp x4, x1              // if i >= n, break
    b.lt .Lloop             // continue if i < n

.Ldone:
    mov x0, x4              // return count
    ret                     // return
