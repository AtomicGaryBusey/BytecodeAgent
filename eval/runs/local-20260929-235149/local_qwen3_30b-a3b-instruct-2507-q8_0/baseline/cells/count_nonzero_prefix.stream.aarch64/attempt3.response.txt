count_nonzero_prefix:
    cbz x1, .Ldone          // if n == 0, return 0
    mov x2, x0              // x2 = a
    mov x3, xzr             // x3 = count = 0
    mov x4, xzr             // x4 = index = 0

.Lloop:
    cmp x4, x1              // compare index with n
    b.ge .Ldone             // if index >= n, break
    ldr x5, [x2, x4, lsl #3] // load a[index]
    cbz x5, .Ldone          // if a[index] == 0, break
    add x3, x3, #1          // increment count
    add x4, x4, #1          // increment index
    b .Lloop                // continue

.Ldone:
    mov x0, x3              // return count
    ret                     // return
