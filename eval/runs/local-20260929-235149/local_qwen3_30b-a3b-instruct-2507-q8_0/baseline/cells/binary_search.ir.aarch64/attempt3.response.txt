export function l $binary_search(l a, l n, l key) {
    @entry:
        %low = constl 0
        %high = subl n, 1
        %result = constl -1
        jz %high, @exit
    @loop:
        %mid = addl %low, %high
        %mid = shrl %mid, 1
        %mid_val = loadl [a, mulil %mid, 8]
        %cmp = csltl %mid_val, key
        jnz %cmp, @low
        %cmp = ceql %mid_val, key
        jnz %cmp, @found
        jnz %cmp, @high
    @low:
        %low = addl %mid, 1
        jnz %high, @loop
    @high:
        %high = subl %mid, 1
        jnz %high, @loop
    @found:
        %result = %mid
    @exit:
        ret %result
}
