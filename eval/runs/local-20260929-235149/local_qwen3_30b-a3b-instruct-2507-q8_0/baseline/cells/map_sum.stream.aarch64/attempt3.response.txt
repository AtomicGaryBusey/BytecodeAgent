map_sum:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x19, x0          // save a
    mov x20, x1          // save n
    mov x21, x2          // save f
    mov x0, #0           // result = 0
    cmp x20, #0
    b.eq .Ldone
    mov x22, #0          // i = 0
.Lloop:
    ldr x23, [x19, x22, lsl #3]  // a[i]
    mov x1, x23          // arg for f
    blr x21              // r = f(a[i])
    add x0, x0, x0       // result += r
    add x22, x22, #1     // i++
    cmp x22, x20
    b.lt .Lloop
.Ldone:
    ldp x29, x30, [sp], #16
    ret
