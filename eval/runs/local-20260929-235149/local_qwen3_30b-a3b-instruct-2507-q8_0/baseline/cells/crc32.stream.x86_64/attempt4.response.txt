crc32:
    mov rax, 0xFFFFFFFF
    mov rdi, qword ptr [rsp + 16]
    mov rsi, qword ptr [rsp + 24]
    test rsi, rsi
    jz .Ldone
    mov rcx, 0
.Lloop:
    movzx rdx, byte ptr [rdi + rcx]
    xor rax, rdx
    mov rdx, 0xEDB88320
    mov r8, 8
.Lbitloop:
    test rax, 1
    jz .Lshift
    xor rax, rdx
.Lshift:
    shr rax, 1
    dec r8
    jnz .Lbitloop
    inc rcx
    cmp rcx, rsi
    jb .Lloop
.Ldone:
    xor rax, 0xFFFFFFFF
