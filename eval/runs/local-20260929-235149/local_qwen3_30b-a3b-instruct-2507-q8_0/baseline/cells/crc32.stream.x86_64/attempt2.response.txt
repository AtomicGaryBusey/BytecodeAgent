crc32:
    push rbp
    mov rbp, rsp
    sub rsp, 16
    xor eax, eax
    mov rdi, 0xFFFFFFFF
    mov rsi, rdi
    mov rdx, 0xEDB88320
    mov rdi, qword ptr [rbp + 16]
    mov rsi, qword ptr [rbp + 24]
    test rsi, rsi
    jz .Ldone
    mov rcx, 0
.Lloop:
    movzx rax, byte ptr [rdi + rcx]
    xor rdi, rax
    mov rax, 0
    mov r8, 8
.Lbitloop:
    test rdi, 1
    jz .Lshift
    xor rdi, rdx
.Lshift:
    shr rdi, 1
    dec r8
    jnz .Lbitloop
    inc rcx
    cmp rcx, rsi
    jb .Lloop
.Ldone:
    xor rdi, 0xFFFFFFFF
