export function l $memmove_i64(l dst, l src, l n) {
    @entry:
        %cmp = csltl n, 0
        jnz %cmp, @exit
        %cmp = csltl n, 1
        jnz %cmp, @copy_one
        ret 0

    @copy_one:
        %dst_val = loadl dst
        %src_val = loadl src
        storel %src_val, %dst_val
        ret 0

    @exit:
        ret 0
}
