export function l $count_nonzero_prefix(l a, l n) {
    @start:
        %cmp = ceql l n, l 0
        jnz %cmp, @zero_case
        ret l 0

    @zero_case:
        %i = l 0
        %loop:
            %addr = addl l a, mul l %i, l 8
            %val = loadl l %addr
            %is_zero = ceql l %val, l 0
            jnz %is_zero, @done
            %i = addl l %i, l 1
            %cmp = csltl l %i, l n
            jnz %cmp, @loop
        @done:
            ret l %i
}
