bubble_sort:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi        # a
    mov r13, rsi        # n
    mov r14, 0          # i
    mov r15, 0          # j
    cmp r13, 1
    jle .Ldone

.Louter_loop:
    mov r15, 0          # j = 0
    mov rbx, r13        # temp_n = n
    dec rbx
    cmp r15, rbx
    jge .Linner_done
    jmp .Linner_loop

.Linner_loop:
    mov rax, qword ptr [r12 + r15*8]
    mov rcx, qword ptr [r12 + r15*8 + 8]
    cmp rax, rcx
    jle .Lno_swap
    # swap
    mov qword ptr [r12 + r15*8], rcx
    mov qword ptr [r12 + r15*8 + 8], rax
.Lno_swap:
    inc r15
    mov rbx, r13
    dec rbx
    cmp r15, rbx
    jl .Linner_loop

.Linner_done:
    inc r14
    mov rbx, r13
    dec rbx
    cmp r14, rbx
    jl .Louter_loop

.Ldone:
    mov rax, 0
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
