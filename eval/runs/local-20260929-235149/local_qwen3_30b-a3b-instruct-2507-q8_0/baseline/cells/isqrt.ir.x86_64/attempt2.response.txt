export function l $isqrt(l n) {
    @start:
        %r = movl 0
        %low = movl 0
        %high = movl 4294967295
        %mid = movl 0
        %mid2 = movl 0
        %done = movw 0

    @loop:
        %done = ceql %low, %high
        jnz %done, @done

        %mid = addl %low, %high
        %mid = shrl %mid, 1
        %mid2 = mulw %mid, %mid
        %cmp = csltl %mid2, %n
        jnz %cmp, @low
        %r = movl %mid
        %low = addl %mid, 1
        jmp @loop

    @low:
        %high = subl %mid, 1
        jmp @loop

    @done:
        retl %r
}
