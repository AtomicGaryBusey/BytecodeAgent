memmove_i64:
    // Save callee-saved registers
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    mov x19, x0          // dst
    mov x20, x1          // src
    mov x21, x2          // n
    mov x22, x19         // dst_copy
    mov x23, x20         // src_copy
    mov x24, x21         // n_copy

    // Check if n is 0
    cbz x24, .Lreturn

    // Check if dst >= src (copy forward)
    cmp x19, x20
    bhs .Lforward

    // Copy backward (overlap: dst after src)
    mov x25, x21         // n_copy
    add x26, x20, x25, lsl #3  // src_end = src + n*8
    add x27, x19, x25, lsl #3  // dst_end = dst + n*8
    sub x26, x26, #8     // src_end - 8
    sub x27, x27, #8     // dst_end - 8

.Lbackward_loop:
    ldr x28, [x26, #-8]  // load src[i]
    str x28, [x27, #-8]  // store to dst[i]
    sub x26, x26, #8
    sub x27, x27, #8
    sub x25, x25, #1
    cbz x25, .Lreturn
    b .Lbackward_loop

.Lforward:
    // Copy forward (overlap: dst before src or no overlap)
    mov x25, x21         // n_copy
    mov x26, x19         // dst_ptr
    mov x27, x20         // src_ptr

.Lforward_loop:
    ldr x28, [x27], #8   // load src[i], post-increment
    str x28, [x26], #8   // store to dst[i], post-increment
    sub x25, x25, #1
    cbz x25, .Lreturn
    b .Lforward_loop

.Lreturn:
    // Restore callee-saved registers
    ldp x27, x28, [sp], #16
    ldp x25, x26, [sp], #16
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    mov x0, #0           // return 0
    ret
