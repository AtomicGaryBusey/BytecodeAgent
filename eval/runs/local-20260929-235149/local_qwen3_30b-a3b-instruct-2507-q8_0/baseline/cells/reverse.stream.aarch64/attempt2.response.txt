reverse:
    mov x2, x0          // x2 = a
    mov x3, x1          // x3 = n
    mov x4, #0          // x4 = i = 0
    cmp x3, #1          // if n <= 1, return 0
    b.le .Lreturn

.Lloop:
    sub x5, x3, #1      // j = n - 1
    sub x5, x5, x4      // j = n - 1 - i
    cmp x4, x5          // if i >= j, break
    b.ge .Lreturn

    ldr x6, [x2, x4, lsl #3]  // x6 = a[i]
    ldr x7, [x2, x5, lsl #3]  // x7 = a[j]
    str x7, [x2, x4, lsl #3]  // a[i] = a[j]
    str x6, [x2, x5, lsl #3]  // a[j] = a[i]
    add x4, x4, #1          // i++
    b .Lloop

.Lreturn:
    mov x0, #0              // return 0
    ret
