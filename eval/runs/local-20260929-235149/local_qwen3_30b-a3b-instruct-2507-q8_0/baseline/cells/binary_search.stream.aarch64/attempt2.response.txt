binary_search:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x19, x0          // a
    mov x20, x1          // n
    mov x21, x2          // key
    mov x22, #0          // low
    mov x23, x20         // high = n
    cbz x20, .Lnot_found // if n == 0, key not found

.Lloop:
    cmp x22, x23         // if low >= high, exit loop
    bge .Lnot_found
    add x24, x22, x23    // low + high
    add x24, x24, #1     // +1 to handle overflow in shift
    lsr x24, x24, #1     // mid = (low + high + 1) / 2, then shift right by 1
    ldr x25, [x19, x24, lsl #3] // a[mid]
    cmp x25, x21         // compare a[mid] with key
    bne .Lcheck          // if not equal, check direction
    mov x0, x24          // found: return mid
    ldp x29, x30, [sp], #16
    ret

.Lcheck:
    cmp x25, x21         // compare a[mid] with key again
    bge .Lhigh           // if a[mid] >= key, search left half
    mov x22, x24         // low = mid + 1
    add x22, x22, #1
    b .Lloop

.Lhigh:
    mov x23, x24         // high = mid
    b .Lloop

.Lnot_found:
    mov x0, #-1          // not found: return -1
    ldp x29, x30, [sp], #16
    ret
