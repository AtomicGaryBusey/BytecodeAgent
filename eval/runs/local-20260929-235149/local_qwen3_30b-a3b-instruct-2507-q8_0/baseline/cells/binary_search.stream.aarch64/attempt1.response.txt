binary_search:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x19, x0          // a
    mov x20, x1          // n
    mov x21, x2          // key
    mov x22, #0          // low
    mov x23, x20         // high = n
    cbz x20, .Lnot_found // if n == 0, key not found

.Lloop:
    cmp x22, x23         // if low >= high, exit loop
    bge .Lnot_found
    add x24, x22, x23    // low + high
    udiv x24, x24, #2    // mid = (low + high) / 2
