export function l $mulhi_u64(l a, l b)
{
    @label0:
        %a_hi = shrq(a, 32)
        %a_lo = andl(a, 0xffffffff)
        %b_hi = shrq(b, 32)
        %b_lo = andl(b, 0xffffffff)

        mul %a_hi %b_hi -> %hi_product1
        mul %a_hi %b_lo -> %mid_product1
        mul %a_lo %b_hi -> %mid_product2
        mul %a_lo %b_lo -> %lo_product

        shrq(%mid_product1, 32) -> %carry1
        addl(%hi_product1, %carry1) -> %sum1
        shrq(%sum1, 32) -> %carry2
        addl(%mid_product2, %carry1) -> %sum2
        addl(%lo_product, %sum2) -> %sum_lo

        addl(%sum1, %carry2) -> %result_hi
        ret %result_hi
}
