export function l $mulhi_u64(l a, l b)
{
    @label0:
        %a_hi = shrq a, 32
        %a_lo = andl a, 0xffffffff
        %b_hi = shrq b, 32
        %b_lo = andl b, 0xffffffff

        %hi_product1 = mull %a_hi, %b_hi
        %mid_product1 = mull %a_hi, %b_lo
        %mid_product2 = mull %a_lo, %b_hi
        %lo_product = mull %a_lo, %b_lo

        %carry1 = shrq %mid_product1, 32
        %sum1 = addl %hi_product1, %carry1
        %carry2 = shrq %sum1, 32
        %sum2 = addl %mid_product2, %carry1
        %sum_lo = addl %lo_product, %sum2

        %result_hi = addl %sum1, %carry2
        ret %result_hi
}
