reverse:
    cmp x1, #0          // Compare n to 0
    b.eq .Lend          // If n is zero, exit

.Lloop:
    mov x2, xzr        // i = 0 (scratch register)
    lsr x3, x1, #1     // half_n = n >> 1
    cbz x3, .Lend      // If n is odd and < 2, exit

.Lswap:
    add x4, x0, x2, lsl #6 // index_i = a + i
    sub x5, x1, x2     // idx_n_minus_1_minus_i = (n - 1) - i
    sub x5, x5, xzr    // Convert to negative for subtraction
    add x5, x0, x5, lsl #6 // address of a[n-1-i]
    ldr x6, [x4]       // Load value at a[i]
    ldr x7, [x5]       // Load value at a[n-1-i]
    str x7, [x4]       // Store a[n-1-i] to a[i]
    str x6, [x5]       // Store a[i] to a[n-1-i]

.Lnext:
    add x2, x2, #1     // i++
    cmp x2, x3         // Compare i with half_n
    b.ne .Lswap        // If not equal, loop

.Lend:
    mov x0, xzr        // Return 0 in x0 (success)
    ret                // Return
