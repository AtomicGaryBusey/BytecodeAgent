gcd:
    test rsi, rsi          # If b is zero, gcd(a, 0) = a
    jz  .end              

.loop:                     
    mov rax, rdi           # Set ax = a (a will be the result)
    cdq                     # Sign extend rax into rdx:rax for idiv by rbx later
                            # Since b is positive, this effectively sets edx to 0
    div rsi                # Divide a by b: quotient in rax, remainder in rdx

    mov rdi, rsi           # Set a = b (next step of Euclid's algorithm)
    mov rsi, rdx           # Set b = a % b (remainder from the division)
    test rdx, rdx          # If remainder is zero
    jnz .loop              
    
.end:
    ret                     # Return value in RAX
