gcd:
    test rsi, rsi          # If b is zero, gcd(a, 0) = a
    jz .end

.loop:
    mov rax, rdi           # Set ax = a (a will be the result)
    cqo                     # Sign extend rdx:rax for idiv by rbx later
                            # Since b is positive, this effectively sets edx to 0
    xor rcx, rcx           # Clear rcx for use as temporary storage
    mov rcx, rsi           # Store b in rcx for division operation

    shr rdi, 63            # Check if a is negative; set rdx accordingly
    sbb rdx, rdx           # Set rdx to -1 (all bits set) if a < 0 else clear it
    add rdx, 1             # If a was non-negative, make rdx = 1 for positive division

    xor rdx, rdx           # Clear rdx before division to ensure no sign extension issues
    idiv rcx               # Divide a by b: quotient in rax, remainder in rdx

    mov rdi, rsi           # Set a = b (next step of Euclid's algorithm)
    mov rsi, rdx           # Set b = a % b (remainder from the division)
    test rdx, rdx          # If remainder is zero
    jnz .loop              
    
.end:
    ret                     # Return value in RAX
