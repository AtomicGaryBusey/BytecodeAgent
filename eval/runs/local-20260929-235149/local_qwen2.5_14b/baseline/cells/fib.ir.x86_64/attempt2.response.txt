export function l $fib(l %n)
{
    @start
    loadl imm(0), %a   ; a = 0, b = 1
    loadl imm(1), %b
    storel %n, rbp

    csltl imm(2), rbp, %w  ; if (n < 2) goto done
    jnz %w, @done

    @loop
    subl imm(1), rbp       ; n--
    movl %b, %c           ; c = b
    addl %a, %b           ; b = a + b
    andl imm(-2), %tmp    ; tmp = (int64_t)(-2) (for wrap-around)
    orl  %tmp, %b         ; b &= ((1 << 63) | (-2))
                          ; This ensures that if b overflows,
                          ; it wraps around to a small positive number.
    movl %c, %a           ; a = c
    jnz @loop

    @done
    movl %b, rbp
    loadl rbp, %0
}
