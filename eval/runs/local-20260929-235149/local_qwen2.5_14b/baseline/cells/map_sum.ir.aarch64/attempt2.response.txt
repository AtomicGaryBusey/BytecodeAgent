export function l $map_sum(l a, l n, l f) {
@start
    w %i = loadw sp
    storel xzr, sp              // Clear result.
    addl sp, sp, 16             // Pop frame.
    jnz %i, @loop

@done
    ret(x30)

@loop
    subl sp, sp, 16             // Push frame.
    stp x29, x30, [sp, #-16]!   // Save FP and LR.
    loadl x8, a                 // Load pointer to array.
    loadl x9, n                 // Load count.
    subl x9, x9, 1             // Decrement count.
    loadl x10, sp               // Frame base (sp).
    loadw %i, [x10], 8         // Load index i from frame.
    addl x12, x8, %i*8         // Compute pointer a[i].
    loadl x13, f               // Function pointer in x13.
    l %r = call %f(l x12)      // Call function through pointer.
    loadl x0, sp               // Load current result from frame.
    addl x0, x0, %r            // Add to result.
    storel x0, [sp]            // Store updated result in frame.
    ceql %i, 63                // Check if done.
    jnz %i, @loop              // Loop until all elements processed.

    ldp x29, x30, [sp], 16     // Restore FP and LR.
    addl sp, sp, 16            // Pop frame.
    ret(x30)                    // Return result.
}
