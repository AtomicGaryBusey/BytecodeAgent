vm_run:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

.Lloop_start:
    ldr x4, [x0], #8
    cbz x4, .Lhalt
    cmp x4, #1
    b.eq .LPUSH
    cmp x4, #2
    b.eq .LADD
    cmp x4, #3
    b.eq .LSUB
    cmp x4, #4
    b.eq .LMUL
    cmp x4, #5
    b.eq .LDUP
    cmp x4, #6
    b.eq .LSWAP
    cmp x4, #7
    b.eq .LDROP
    cmp x4, #8
    b.eq .LOVER
    cmp x4, #9
    b.eq .LLT
    cmp x4, #10
    b.eq .LJZ
    cmp x4, #11
    b.eq .LJMP
    add x0, x0, #-8
    cbz x2, .Lfinit

.Lloop_end:
    ldr x4, [x0], #8
    cbnz x4, .Lloop_start
    mov x0, x1
    br x29

.Lhalt:
    cmp x3, #0
    beq .Lreturn_zero
    mov x0, x1
    br x29

.LPUSH:
    ldr x5, [x0]
    str x5, [x2], #8
    add x0, x0, #-8
    b .Lfinit

.LADD:
    ldr x5, [x2], #-8
    ldr x6, [x2], #-8
    add x5, x5, x6
    str x5, [x2]
    add x0, x0, #-16
    b .Lfinit

.LSUB:
    ldr x5, [x2], #-8
    ldr x6, [x2], #-8
    sub x5, x5, x6
    str x5, [x2]
    add x0, x0, #-16
    b .Lfinit

.LMUL:
    ldr x5, [x2], #-8
    ldr x6, [x2], #-8
    umulh x7, x5, x6
    str x7, [x2]
    add x0, x0, #-16
    b .Lfinit

.LDUP:
    ldr x5, [x2]
    str x5, [x2], #8
    add x0, x0, #-8
    b .Lfinit

.LSWAP:
    stp x5, x6, [sp, #-16]!
    ldr x5, [x2], #-8
    ldr x6, [x2]
    str x5, [x2]
    str x6, [x2], #-8
    ldp x5, x6, [sp], #16
    add x0, x0, #-8
    b .Lfinit

.LDROP:
    sub x2, x2, #8
    add x0, x0, #-8
    b .Lfinit

.LOVER:
    ldr x5, [x2], #-8
    str x5, [x2], #8
    str x5, [x2]
    add x0, x0, #-8
    b .Lfinit

.LLT:
    ldr x5, [x2], #-8
    ldr x6, [x2], #-8
    cmp x5, x6
    cset x5, lt
    str x5, [x2]
    add x0, x0, #-16
    b .Lfinit

.LJZ:
    ldr x4, [x0], #8
    ldr x5, [x2], #-8
    cbz x5, .Ljump_to_t
    add x0, x0, #-8
    b .Lfinit

.LJMP:
    mov x0, x4
    b .Lloop_end

.Lreturn_zero:
    mov x0, #0
    br x29

.Ljump_to_t:
    sub x0, x0, #16
    b .Lloop_start

.Lfinit:
    ldr x5, [x2]
    cbz x5, .Lfinit_return_value
    sub x2, x2, #8
    b .Lreturn_zero

.Lfinit_return_value:
    str x0, [sp], #16
    br x29
