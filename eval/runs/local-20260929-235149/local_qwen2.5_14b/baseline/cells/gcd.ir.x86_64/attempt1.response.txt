export function l $gcd(l a, l b)
{
    @1:
        csltl %w0, a, b      # Set %w0 to 1 if a < b else 0
        jnz   %w0, @2       # Jump to @2 if a < b

    @3:
        ceql %w1, b, 0      # Set %w1 to 1 if b == 0 else 0
        jnz   %w1, @4       # Jump to @4 if b == 0
        csltl %w2, a, b     # Compute remainder of a divided by b into w2

    @5:
        storel a, %r0       # Store a in r0
        loadl  b, a         # Load b into a
        loadl  %r0, b       # Load stored value from r0 back to b
        loadl  %w2, b       # Load remainder into b

    @6:
        j     @3            # Loop until b is 0

    @4:
        loadl a             # Return a (the gcd)
}

@2:
    storel a, %r0           # Swap a and b
    loadl  b, a
    loadl  %r0, b
    j      @1               # Continue from the start
