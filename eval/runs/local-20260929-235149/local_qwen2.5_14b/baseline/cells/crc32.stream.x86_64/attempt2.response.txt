crc32:
    push        rbp
    mov         rbp, rsp              # Setup stack frame
    sub         rsp, 8                # Align stack for calls

    xor         rcx, rcx              # Zero counter rcx
    mov         rbx, -1               # Initialize crc with 0xFFFFFFFF

loop_start:
    cmp         rsi, rcx              # Compare n and counter
    jge         loop_body             # If n >= counter, continue

done:
    xor         rax, rbx              # Final XOR to get CRC value
    pop         rbp                    # Restore stack frame
    ret                                # Return

loop_body:
    movzx       r8d, byte ptr [rdi + rcx]  # Load next byte into r8
    xor         rbx, r8                  # XOR crc with current byte
    lea         rdx, [rcx + 1]           # Increment counter by 1
    mov         rcx, rdx                 # Move new count to rcx

shift_loop_start:
    test        rbx, 1                   # Test if lowest bit is set
    jz          shift_loop_end           # If not set, skip XOR
    xor         rbx, -3729904560         # XOR with polynomial
