add_overflow:
    mov     rbx, rsi               # Save b in rbx
    xor     rcx, rcx               # Clear rcx for later use as counter or flag
    test    rdi, rdi               # Check if a is zero
    jz      .a_zero                # If a is zero, jump to handle zero addition
    mov     rax, rdi               # Move a into rax
    sbb     rcx, rcx               # Set flags: CF = 1 if b > a and different signs
    jz      .no_overflow           # Jump if no overflow (Z flag is set)
    add     rax, rbx               # Perform a + b in RAX, result may wrap
    mov     qword ptr [rdx], rax   # Store wrapped sum at *out
    or      rax, -1                # Set RAX to 0xFFFFFFFFFFFFFFFF indicating overflow
    ret                            # Return with overflow flag set

.a_zero:
    test    rbx, rbx               # Check if b is zero
    jz      .b_zero                # If b is also zero, both are zero and no operation needed
    xor     rax, rax               # Set RAX to 0 indicating no overflow
    mov     qword ptr [rdx], rbx   # Store value of b at *out if a is zero
    ret                            # Return with no overflow

.b_zero:
    xor     rax, rax               # Both a and b are zero, set RAX to 0 indicating no overflow
    mov     qword ptr [rdx], rdi   # Store value of a (which is zero) at *out if both a and b are zero
    ret                            # Return with no overflow

.no_overflow:
    add     rax, rbx               # Perform a + b in RAX directly as there's no risk of wrap
    mov     qword ptr [rdx], rax   # Store sum at *out
    xor     rax, rax               # Set RAX to 0 indicating no overflow
    ret                            # Return with no overflow
