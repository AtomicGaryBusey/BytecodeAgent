add_overflow:
    mov     rax, rdi               # Move a to rax for comparison with b
    cmp     qword ptr [rsi], rax   # Compare a and b
    js      .b_negative            # Jump if b is negative
    jrcxz  .no_addition            # Jump if a is zero (0 + any = no overflow)
    jc      .overflow_case          # Jump if carry flag set on comparison

.b_positive:
    sbb     rcx, rcx               # Set flags: CF = 1 if b > a and they have different signs
    xor     rcx, -1                # Invert flags to clear CF for positive addition check
    jz      .no_addition           # Jump if no overflow (Z flag is set)

.overflow_case:
    add     rdi, rsi               # Perform a + b in RDI, result may wrap
    mov     qword ptr [rdx], rdi   # Store wrapped sum at *out
    or      rax, -1                # Set RAX to 0xFFFFFFFFFFFFFFFF indicating overflow
    ret                            # Return with overflow flag set

.no_addition:
    xor     eax, eax               # Set EAX to 0 indicating no overflow
    add     rdi, rsi               # Perform a + b in RDI, no wrap expected
    mov     qword ptr [rdx], rdi   # Store sum at *out
    ret                            # Return with no overflow

.b_negative:
    jrcxz  .no_addition            # Jump if a is zero (0 + any = no overflow)
    jc      .overflow_case          # Jump if carry flag set on comparison for negative b
    sbb     rcx, rcx               # Set flags: CF = 1 if a < b and they have different signs
    xor     rcx, -1                # Invert flags to clear CF for negative addition check
    jz      .no_addition           # Jump if no overflow (Z flag is set)
