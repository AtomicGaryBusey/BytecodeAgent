add_overflow:
    mov     rax, rsi               # Move b to rax for comparison and subtraction
    sub     rax, rdi               # Calculate b - a in rax
    sbb     rcx, rcx               # Set flags: CF = 1 if (b < a), else CF = 0
    xor     rcx, -1                # Invert flags to set CF to !CF, ZF cleared
    not     rcx                     # RCX now has 0xFFFFFFFFFFFFFFFF for overflow case
    jz      .no_overflow           # Jump if no overflow (Z flag is set)
    add     rdi, rsi               # Perform a + b in RDI, result may wrap
    mov     qword ptr [rdx], rdi   # Store wrapped sum at *out
    or      rax, -1                # Set RAX to 0xFFFFFFFFFFFFFFFF indicating overflow
    ret                            # Return with overflow flag set

.no_overflow:
    add     rdi, rsi               # Perform a + b in RDI, no wrap expected
    mov     qword ptr [rdx], rdi   # Store sum at *out
    xor     eax, eax               # Set EAX to 0 indicating no overflow
    ret                            # Return with no overflow
