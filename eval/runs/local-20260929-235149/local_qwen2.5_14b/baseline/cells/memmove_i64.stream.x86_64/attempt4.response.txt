memmove_i64:
    test rdx, rdx
    jz .return_zero

    lea rbx, [rsi + rdx*8]
    lea r12, [rdi + rdx*8]

    cmp rdi, rsi
    jl copy_forward

copy_backward:
    mov rcx, -1
.copy_loop_backward:
    inc rcx
    neg rcx
    lea rax, [rbx + rcx * 8]
    mov qword ptr [r12 + rcx * 8], qword ptr [rax]
