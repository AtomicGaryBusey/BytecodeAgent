vm_run:
    mov     rbp, rsp              # Save RBP
    push    rbx                    # Save RBX
    sub     rsp, 48               # Allocate local variables on the stack

    lea     rcx, [rsp+24]         # Stack pointer (rsp) + space for up to 63 entries
    mov     rax, rdx              # Cap = RDX into RAX
    sub     rax, rcx              # Calculate initial stack capacity
    imul    rax, rax              # RAX * (RAX - 1) / 2
    shr     rax, 1                # Space for up to cap entries + 8 bytes padding

    mov     qword ptr [rsp], rdi  # Code pointer in local storage
    mov     qword ptr [rsp+8], rsi # Length in local storage
    lea     rbx, [rdx-1]          # Stack top position (initially -1)
    lea     rcx, [r8-1]           # Stack capacity minus one

.Lloop:
    mov     rax, qword ptr [qword ptr rsp]  # Load code[pc]
