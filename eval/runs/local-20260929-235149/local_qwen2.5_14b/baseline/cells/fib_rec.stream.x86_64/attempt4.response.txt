fib_rec:
    cmp rdi, 1              # If n <= 1, jump to base cases
    jle .Lbase_case

    dec rdi                 # n -= 1 for fib(n-1)
    push rbx                # Save callee-saved registers before call
    push rbp
    sub rsp, 8              # Align stack for the recursive call (64-bit ABI requires this)
    call fib_rec            # Call fib_rec(n-1), save result in rax (high part of sum)

    mov rbx, rax            # Save fib(n-1) in rbx for later use

    dec rdi                 # n -= 1 again for fib(n-2)
    push rbx                # Save rbx before recursive call to avoid clobbering
    sub rsp, 8              # Align stack for the recursive call (64-bit ABI requires this)
    call fib_rec            # Call fib_rec(n-2), add result to previous one in rax

    pop rbx                 # Restore rbx after second recursive call
    add rax, rbx            # Add results of two calls (fib(n-1) and fib(n-2))
    mov rbx, rsp            # Save current stack pointer for restoration before return

    add rsp, 16             # Clean up the alignment subtractions from above
    pop rbp                 # Restore rbp
    pop rbx                 # Restore rbx
    mov rsp, rbx            # Restore original stack pointer from saved value

    ret                     # Return with result in rax (sum of fib(n-1) and fib(n-2))

.Lbase_case:
    test rdi, rdi           # If n == 0, set eax to 0; otherwise, set it to 1
    cmove rbx, qword ptr [rip + .Lbase_cases]   # Load base cases into rbx

    xor eax, eax            # Clear eax for n == 0
    cmovne rax, rbx         # Correctly set rax based on condition

    ret                     # Return with result in rax

.Lbase_cases:
    .quad 0                 # Value for fib_rec(0)
