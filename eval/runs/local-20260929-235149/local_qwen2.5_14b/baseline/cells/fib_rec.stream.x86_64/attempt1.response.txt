fib_rec:
    cmp rdi, 0
    je .Lbase_case_0
    cmp rdi, 1
    jle .Lbase_case_1

    # n >= 2: compute fib(n-1) + fib(n-2)
    sub rdi, 1
    push rbp
    mov rbp, rsp
    push rbx
    call fib_rec            # Call fib_rec(n-1), save rax as high part of sum
    mov rbx, rax            # Save fib(n-1) in rbx for later use

    sub rdi, 1              # Decrement n by another 1 to get (n-2)
    push rbp                # Adjust stack again
    mov rbp, rsp
    push rbx
    call fib_rec            # Call fib_rec(n-2), add to previous result in rax

    pop rbx                 # Restore rbx
    pop rbp                 # Restore rbp
    add rax, rbx            # Add fib(n-1) and fib(n-2)

    pop rbx
    pop rbp
    ret                     # Return with sum in rax (result is already correct)

.Lbase_case_0:
    xor eax, eax            # fib_rec(0) = 0
    mov rsp, rbp
    pop rbp
    ret

.Lbase_case_1:
    mov eax, 1              # fib_rec(1) = 1
    mov rsp, rbp
    pop rbp
    ret
