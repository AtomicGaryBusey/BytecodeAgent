bubble_sort:
    cmp rsi, 0          # Compare n with 0
    jle .Lend           # If n <= 0, jump to end

.Louter_loop:
    mov rbx, rdi        # Copy array base address to rbx
    sub rsi, 1          # Decrement n by 1 for inner loop limit calculation
    xor rcx, rcx        # Initialize outer loop counter (rcx) to 0

.Linner_loop:
    lea rdx, [rbx + rcx]      # Load effective address of current element into rdx
    add rcx, 8                # Increment index by 8 bytes for next comparison
    cmp qword ptr [rdx], qword ptr [rdx + 8]         # Compare current and next elements
