bubble_sort:
    cmp rsi, 0          # Compare n with 0
    jle .Lend           # If n <= 0, jump to end

.Louter_loop:
    mov rbx, rdi        # Copy array base address to rbx
    sub rsi, 1          # Decrement n by 1 for inner loop limit calculation
    xor rcx, rcx        # Initialize outer loop counter (rcx) to 0

.Linner_loop:
    cmp rcx, rsi        # Compare current index with remaining elements count
    jge .Lnext_outer    # If all pairs are sorted, proceed to next iteration of outer loop

    lea rdx, [rbx + rcx]      # Load effective address of current element into rdx
    mov rax, qword ptr [rdx]  # Load current element to rax
    add rcx, 8                # Increment index by 8 bytes for next comparison
    mov rbx, qword ptr [rdx + 8]   # Load the next element into rbx
    cmp rax, rbx               # Compare current and next elements
    jle .Lnext_comparison      # If current <= next, skip swap

    mov qword ptr [rdx], rbx        # Store next element to current position
    mov qword ptr [rdx + 8], rax    # Move current element to the place of next

.Lnext_comparison:
    loop .Linner_loop             # Decrement rcx and continue inner loop

.Lnext_outer:
    add rsi, 1                    # Increment n by 1 for next outer iteration
    jle .Lend                     # If n <= 0, jump to end after fixing up
    jmp .Louter_loop              # Jump back to start of outer loop

.Lend:
    xor rax, rax                  # Return 0
    ret                           # Return
