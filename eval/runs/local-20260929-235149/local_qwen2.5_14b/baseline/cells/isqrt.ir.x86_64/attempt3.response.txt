export function l $isqrt(l n) {
    @start
    %high_bits = shr.l(n, 32)
    jz @low_32_bits

    %r = movl(0xffffffffffffffff)
    csltl %cmp, %high_bits, 16
    jnz @middle_bits

    @low_32_bits
    %r = movl(0)
    ceql %cmp, n, 0
    jz @return_r

    @middle_bits
    %midpoint = sll.l(%high_bits, 32)
    %shifted_n = shr.l(n, 16)
    csltl %cmp, %shifted_n, 8
    jnz @full_high

    %r = orl(%midpoint, 0xffff)
    ceql %cmp, n, 0
    jz @return_r

    @full_high
    %higher_mid = orl(%midpoint, 0xffffffff)
    csltl %cmp, higher_mid, 4
    jnz @upper_middle

    %r = orl(%midpoint, 0xff)
    ceql %cmp, n, 0
    jz @return_r

    @upper_middle
    %highest_bit = movl(1)
    csltl %cmp, higher_mid, 2
    jnz @lower_upper_middle

    %r = orl(%midpoint, 0xf)
    ceql %cmp, n, 0
    jz @return_r

    @lower_upper_middle
    csltl %cmp, higher_mid, 1
    jnz @middle_high_bit_set

    %r = orl(%midpoint, 0x3)
    ceql %cmp, n, 0
    jz @return_r

    @middle_high_bit_set
    csltl %cmp, higher_mid, 0
    jnz @lower_middle_high_bit_set

    %r = orl(%midpoint, 1)
    ceql %cmp, n, 0
    jz @return_r

    @lower_middle_high_bit_set
    %r = orl(%midpoint, 2)

    @return_r
    ret.l %r
}
