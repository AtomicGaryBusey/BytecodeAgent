export function l $crc32(l %p, l %n)
{
%mask =l 0xFFFFFFFF
%crc =l %mask
%index =l 0

loop_start:
%csltl %done, %index, %n
jnz .end

%byte =w loadub %p(%index)
%xor %crc, %byte
%byte =w %crc & 1
csltl %shift, %byte, 0
jnz .odd_shift
.even_shift:
%crc =l %crc >> 1
jmp .loop_end

.odd_shift:
%mask =l 0xEDB88320
%xor %crc, %mask
%crc =l %crc >> 1

.loop_end:
%index =l %index + 1
jmp loop_start

.end:
%mask =l 0xFFFFFFFF
%xor %crc, %mask
ret %crc
}
