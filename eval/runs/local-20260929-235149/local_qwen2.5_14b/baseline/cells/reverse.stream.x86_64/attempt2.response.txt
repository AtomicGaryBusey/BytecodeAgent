reverse:
    test rsi, rsi          # Check if n is zero
    jz .Ldone              # If zero, jump to done

    lea rcx, [rsi - 1]     # Calculate index for last element: rcx = n-1
    xor r8, r8             # Initialize loop counter: r8 = 0

.Lloop:
    cmp r8, rcx            # Check if loop counter is greater than or equal to (n-1)
    jge .Ldone             # If so, jump to done

    mov r9, qword ptr [rdi + r8 * 8]     # Load a[r8]
    mov rax, qword ptr [rdi + rcx * 8]   # Load a[rcx]
    mov qword ptr [rdi + rcx * 8], r9    # Store a[r8] to a[rcx]
    mov qword ptr [rdi + r8 * 8], rax    # Store a[rcx] to a[r8]

    inc r8                  # Increment loop counter
    dec rcx                 # Decrement index for last element

    jmp .Lloop              # Loop back

.Ldone:
    xor eax, eax            # Set return value to zero
    ret                     # Return
