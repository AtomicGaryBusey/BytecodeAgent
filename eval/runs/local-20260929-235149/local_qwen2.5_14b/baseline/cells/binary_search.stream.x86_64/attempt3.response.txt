binary_search:
    mov     rbp, rsp                   # Save stack pointer for return alignment
    test    rsi, rsi                   # Test if n is zero
    jle     .Lreturn_minus_1           # Return -1 if array is empty

    lea     rcx, [rsi-1]               # Calculate high index: n-1 in rcx
    xor     rdx, rdx                   # Low index starts at 0 (in rdx)
.Lloop:
    cmp     rdx, rcx                   # Check if low > high
    jg      .Lreturn_minus_1           # If so, key is not present

    mov     rbx, rdx                   # Save low in rbx to restore it after call
    add     rax, rdx                   # Calculate mid index: (low + high) / 2
    shr     rax, 31                    # Mid = (rbx+rcx)/2; shift right for divide by two
    add     rax, rcx                   # Complete mid calculation: rbx + (rcx-rbx)/2
    mov     rdi, qword ptr [rdi+rax*8] # Load a[mid]
    cmp     rsi, rdx                   # Check if low == high
    je      .Lcheck_key                # If so, go check key value

.Lcmp_mid:
    sub     rax, rbx                   # mid - low (in rax)
    test    rax, rax                   # If mid != low
    jne     .Lcheck_key                # Go check key value if not equal

.Lcheck_key:
    cmp     qword ptr [rdi], r8        # Compare a[mid] with key
    je      .Lfound                    # Found the index if equal
    jl      .Lset_high                 # If a[mid] < key, set high = mid - 1
    jmp     .Lset_low                  # Else (a[mid] > key), set low = mid + 1

.Lset_high:
    dec     rcx                        # Adjust high: mid - 1
    mov     rdx, rbx                   # Restore low index from rbx
    cmp     rcx, rdx                   # Check if new high is less than low
    jge     .Lloop                     # Continue searching

.Lset_low:
    inc     rbx                        # Adjust low: mid + 1
    mov     rdx, rbx                   # Restore low index from rbx
    cmp     rcx, rsi                   # Check if new low is greater than high
    jl      .Lloop                     # Continue searching

.Lreturn_minus_1:
    xor     eax, eax                   # Clear eax to zero for return value
    dec     eax                        # Set return value to -1
    jmp     .Ldone                     # Exit function

.Lfound:
    sub     rcx, rbx                   # Calculate index: mid - low
    mov     rax, rcx                   # Move calculated index into rax for return

.Ldone:
    leave                              # Restore stack pointer and cleanup
    ret                                # Return to caller
