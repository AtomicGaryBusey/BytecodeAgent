popcount:
    mov     x8, #0x5555555555555555          // Magic number for first step
    and     x9, x0, x8                       // Apply mask to get odd bits
    lsl     x10, x0, #1                      // Shift even bits into place
    orr     x0, x9, x10                      // Combine odd and even bits

    mov     x8, #0x3333333333333333          // Magic number for second step
    and     x9, x0, x8                       // Apply mask to get lower nibbles
    lsl     x10, x0, #2                      // Shift higher nibbles into place
    orr     x0, x9, x10                      // Combine lower and higher nibbles

    mov     x8, #0x0f0f0f0f0f0f0f0f          // Magic number for third step
    and     x9, x0, x8                       // Apply mask to get bytes
    lsl     x10, x0, #4                      // Shift next bytes into place
    orr     x0, x9, x10                      // Combine current and next bytes

    mov     x8, #0x00ff00ff00ff00ff          // Magic number for fourth step
    and     x9, x0, x8                       // Apply mask to get 2-byte chunks
    lsl     x10, x0, #8                      // Shift next 2-byte chunks into place
    orr     x0, x9, x10                      // Combine current and next 2-byte chunks

    mov     x8, #0x0000ffff0000ffff          // Magic number for fifth step
    and     x9, x0, x8                       // Apply mask to get 4-byte words
    lsl     x10, x0, #16                     // Shift next 4-byte words into place
    orr     x0, x9, x10                      // Combine current and next 4-byte words

    mov     x8, #0x000000ff000000ff          // Magic number for sixth step
    and     x9, x0, x8                       // Apply mask to get halfwords
    lsl     x10, x0, #24                     // Shift next halfwords into place
    orr     x0, x9, x10                      // Combine current and next halfwords

    mov     x8, #0x00000000ff000000          // Magic number for seventh step
    and     x9, x0, x8                       // Apply mask to get bytes
    lsl     x10, x0, #32                     // Shift next bytes into place
    orr     x0, x9, x10                      // Combine current and next bytes

    mov     x8, #0x0000000000ff0000          // Magic number for eighth step
    and     x9, x0, x8                       // Apply mask to get bytes
    lsl     x10, x0, #40                     // Shift next bytes into place
    orr     x0, x9, x10                      // Combine current and next bytes

    mov     x8, #0x000000000000ff00          // Magic number for ninth step
    and     x9, x0, x8                       // Apply mask to get byte
    lsl     x10, x0, #48                     // Shift next bytes into place
    orr     x0, x9, x10                      // Combine current and next byte

    mov     x8, #0x00000000000000ff          // Magic number for tenth step
    and     x9, x0, x8                       // Apply mask to get byte
    lsl     x10, x0, #56                     // Shift next bytes into place
    orr     x0, x9, x10                      // Combine current and next byte

    mov     x8, #0x7f                         // Magic number for final step
    and     x9, x0, x8                        // Apply mask to get top 7 bits
    asr     x10, x0, #56                     // Shift all but top bit out
    bic     x0, x10, x10, lsr #1             // Remove every second bit (popcount)
    add     x9, x9, x10                       // Add top 7 bits and remaining 1s
    mov     x10, x9, lsr #8                  // Shift right by 8 to remove high byte
