sum_to:
    cmp x0, #0                  // Compare input with zero
    b.le Lzero                  // Branch if n <= 0

Linit_loop:
    mov x1, #0                 // Initialize sum to 0
    mov x2, #1                 // Start counter from 1
Lloop:
    cbz x2, Lreturn            // Check for termination condition (x2 == 0)
    add x1, x1, x2             // Sum += x2
    add x2, x2, #1             // Increment counter by one
    b Lloop                     // Loop until x2 > n

Lzero:
    mov x0, #0                 // Set result to zero for n <= 0
    ret                        // Return from function

Lreturn:
    mov x0, x1                 // Move result to return register
    ret                        // Return from function
