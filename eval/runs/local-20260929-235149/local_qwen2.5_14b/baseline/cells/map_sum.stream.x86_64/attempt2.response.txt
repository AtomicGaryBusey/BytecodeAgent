map_sum:
    mov     rbx, rdi             # Save pointer to array in rbx
    test    rsi, rsi             # If n is 0
    jle     .Lreturn_zero        # Return immediately with zero

    xor     rcx, rcx             # Index i = 0
    call    rdx                  # Call f(a[0])
    add     r12, rax             # Store result in r12; use it as sum accumulator
    mov     r13, rbx             # Save base address of array

.Lloop:
    inc     rcx                  # Increment index i
    cmp     rcx, rsi             # Compare with n
    jge     .Lreturn_sum         # If done, return sum in rax

    lea     rdi, [r13 + rcx * 8]   # Calculate address of a[i]
    call    rdx                  # Call f(a[i])
    add     r12, rax             # Add result to sum
    jmp     .Lloop              # Loop until done

.Lreturn_sum:
    mov     rax, r12             # Return sum in rax
    ret

.Lreturn_zero:
    xor     eax, eax             # Zero the return value
    ret
