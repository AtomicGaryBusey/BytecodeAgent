map_sum:
    mov     r8, rdi              # Store a in r8
    test    rsi, rsi             # If n is 0
    jle     .Lreturn_zero        # Return immediately with zero

    xor     rcx, rcx             # Sum = f(a[0])
    call    rdx                  # Call f(a[0])
    add     rcx, rax             # Add result to sum in rcx

.Lloop:
    inc     rcx                  # Increment index
    cmp     rcx, rsi             # Compare with n
    jl      .Lcall_f            # If not done, call f(a[i])

.Lreturn_sum:
    mov     rax, rcx             # Return sum in rax
    ret

.Lcall_f:
    mov     rdi, qword ptr [r8 + rcx * 8]   # Load a[rcx]
    call    rdx                  # Call f(a[i])
    add     rcx, rax             # Add result to sum in rcx
    jmp     .Lloop              # Loop until done

.Lreturn_zero:
    xor     eax, eax             # Zero the return value
    ret
