export function l $binary_search(l a, l n, l key)
{
    @1:
        %lo = movl l 0
        %hi = subl n, l 1

    csltl %lo, %hi, w %done
    jnz %done, @2

    %mid = addl %lo, %hi
    sarl %mid, l 1

    %cmp = loadl a, %mid
    csltl key, %cmp, w %lt
    jnz %lt, @3
    csltl %cmp, key, w %gt
    jnz %gt, @4

    movq l -1, rax
    ret

    @2:
        ceql key, loadl a, l 0, w %eq
        jnz %eq, @5
        movq l -1, rax
        ret

    @3: // key < arr[mid]
        storel %mid, %hi
        jmp @1

    @4: // key > arr[mid]
        subl %mid, a, %base
        loadl %base, %lo
        jmp @1

    @5:
        movq %mid, rax
        ret
}
