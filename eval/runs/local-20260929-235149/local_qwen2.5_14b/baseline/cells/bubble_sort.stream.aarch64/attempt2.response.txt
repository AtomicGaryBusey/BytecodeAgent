bubble_sort:
    cmp x1, #0           // Compare n with 0.
    ble done             // If n <= 0, exit.
.Lloop:
    mov x2, xzr          // Initialize swap count to 0.
.Linner_loop:
    cmp x2, x1           // Check if a full pass is completed.
    bge done             // Exit if no swaps in the last pass.
    sub x3, x1, #1       // Calculate index n-1-i for comparison.
    cbz x3, .Lend_outer  // If i == n-1, end outer loop.
    lsl x4, x3, #3       // Shift left to get byte offset in array.
    add x5, x0, x4       // Calculate pointer a[i].
    ldr x6, [x5]         // Load a[i] into x6.
    add x7, x5, #8       // Calculate pointer a[i+1].
    ldr x8, [x7]         // Load a[i+1] into x8.
    cmp x6, x8           // Compare a[i] with a[i+1].
    bge .Lno_swap        // If a[i] <= a[i+1], no swap needed.
    str x8, [x5]         // Swap: store a[i+1] in a[i].
    str x6, [x7]         // Swap: store a[i] in a[i+1].
    add x2, x2, #1       // Increment swap count.
.Lno_swap:
    sub x3, x3, #1       // Decrement i.
    b .Linner_loop       // Continue inner loop.
.Lend_outer:
    mov x0, xzr          // Set return value to 0 (success).
done:
    ret                  // Return.
