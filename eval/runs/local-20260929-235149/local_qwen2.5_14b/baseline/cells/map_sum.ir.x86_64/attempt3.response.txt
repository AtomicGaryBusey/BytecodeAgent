export function l $map_sum(l %a, l %n, l %f) {
    @1:
        csltl %done, 0, %n       // if (n <= 0) return 0;
        jnz .Lend

        subq $8, rbp              // allocate stack frame
        movl -16(rbp), rbp        // save caller-saved

    @2:
        loadl %i, 0(%a)
        %r =l call %f(l %i)      // result of f(a[0])
        csltl %done, 8, rbp
        jnz .Lskip_init
        storel -8(rbp), %r       // sum += f(a[i])

    @3:
        loadl %j, 8(%a,%n,-8)
        %r =l call %f(l %j)
        addq -8(rbp), %r         // sum += f(a[n-1])
        storel -8(rbp), %r

    @4:
        subl $1, %n               // n--
        jnz .Lloop

    .Lskip_init:
        movl -8(rbp), %sum
    .Lend:
        addq $8, rbp              // deallocate stack frame
        movl -16(rbp), rbp        // restore caller-saved
        ret
}
