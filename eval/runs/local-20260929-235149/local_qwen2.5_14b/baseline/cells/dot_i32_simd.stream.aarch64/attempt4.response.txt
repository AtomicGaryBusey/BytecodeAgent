dot_i32_simd:
    stp x19, x20, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    lsr x8, x2, #2       // x8 = n / 4 (vector loop iterations)
    mov x9, xzr          // sum accumulator
.Lvec_loop:
    lsl x3, x8, #2
    add x3, x19, x3      // load address for a
    ldr q0, [x3]
    lsl x4, x8, #2
    add x4, x20, x4      // load address for b
    ldr q1, [x4]
    umlal v2.4s, v0.4s, v1.4s
