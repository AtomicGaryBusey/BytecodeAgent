dot_i32_simd:
    sub rsp, 8           # Reserve space for aligning stack to 16 bytes
    mov rax, rdx         # Initialize result in RAX
    test rdx, rdx        # Test if n is zero
    jle .Lreturn         # Return early for n == 0
    and rdx, -4          # Align count down to a multiple of 4 for vector loop
    sub rcx, rdx         # Remaining elements as scalar loop length

.Lvector_loop:
    movdqa xmm1, [rsi]   # Load vector from b
    pmovsxdq xmm0, [rdi] # Sign-extend and load a into XMM0
    pshufd xmm2, xmm1, 0x66 # Shuffle for vertical multiplication
    pmuldq xmm0, xmm2    # Vector multiply
    paddd xmm0, [rsp+8]  # Add vector sum into scalar accumulator (loaded earlier)
    movdqa [rsp], xmm0   # Store updated vector sum back
    add rsi, 16          # Increment b pointer by 4 elements (16 bytes)
    add rdi, 16          # Increment a pointer by 4 elements (16 bytes)
    sub rdx, 4           # Decrement count by 4
    jg .Lvector_loop     # Loop until n < 4

.Lscalar_loop:
    test rcx, rcc        # Check if scalar loop is needed
