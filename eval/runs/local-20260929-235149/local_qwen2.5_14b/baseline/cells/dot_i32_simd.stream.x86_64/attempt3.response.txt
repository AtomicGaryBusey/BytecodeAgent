dot_i32_simd:
    sub rsp, 8           # Reserve space for aligning stack to 16 bytes
    mov rax, rdx         # Initialize result in RAX
    test rdx, rdx        # Test if n is zero
    jle .Lreturn         # Return early for n == 0
    and rdx, -4          # Align count down to a multiple of 4 for vector loop
    sub rcx, rdx         # Remaining elements as scalar loop length

.Lvector_loop:
    movdqa xmm1, [rsi]   # Load vector from b
    pmovsxdq xmm0, [rdi] # Sign-extend and load a into XMM0
    pshufd xmm2, xmm1, 0x66 # Shuffle for vertical multiplication
    pmuldq xmm0, xmm2    # Vector multiply
    movdqa xmm3, xmm0    # Copy result to preserve input registers
    paddd xmm3, [rsp]    # Add vector sum into scalar accumulator (loaded earlier)
    movd [rsp], xmm3     # Store updated vector sum back
    add rsi, 16          # Increment b pointer by 4 elements (16 bytes)
    add rdi, 16          # Increment a pointer by 4 elements (16 bytes)
    sub rdx, 4           # Decrement count by 4
    jg .Lvector_loop     # Loop until n < 4

.Lscalar_loop:
    test rcx, rcx        # Check if scalar loop is needed
    jle .Laccumulate_sum # Skip scalar loop if remaining elements are zero
    mov eax, [rdi]       # Load a[i]
    imul eax, dword ptr [rsi] # Multiply by b[i], use int64_t wrap
    add rax, [rsp]       # Add to accumulated sum
    mov [rsp], rax       # Store updated scalar sum back
    add rsi, 4           # Increment b pointer by 1 element (4 bytes)
    add rdi, 4           # Increment a pointer by 1 element (4 bytes)
    dec rcx              # Decrement count by 1
    jge .Lscalar_loop    # Loop until scalar loop is done

.Laccumulate_sum:
    movd xmm0, [rsp]     # Load accumulated vector sum into XMM0
    psrad xmm0, 32       # Shift right to convert from int64 to int32
    pshufb xmm0, .Lshuf_mask   # Shuffle high quadword out of the way
