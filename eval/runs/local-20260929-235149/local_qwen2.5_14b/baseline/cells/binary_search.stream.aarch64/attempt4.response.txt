binary_search:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

.Lloop_start:
    cmp x1, #0
    ble .Lfound_neg_one

    cbz x1, .Lfound_neg_one

    add x5, x0, x1                      // Calculate end address of array a
    and x4, x1, #-2                     // mid = (left + right) / 2 rounded down to even
    lsl x6, x4, #31                     // Shift left by 31 bits to prepare for arithmetic shift right
    asr x6, x6, #30                     // Arithmetic shift right by 30 bits to effectively divide by 2

    add x6, x0, x6                     // mid = (left + mid)

    ldr x7, [x6]                        // Load a[mid]
    cmp x7, x2                          // Compare key with a[mid]

    b.eq .Lfound_index                  // If key == a[mid], return mid

    b.lt .Ladjust_right                 // If key < a[mid], adjust right
    b.gt .Ladjust_left                  // If key > a[mid], adjust left

.Ladjust_right:
    sub x1, x4                          // Adjust right = mid - 1
