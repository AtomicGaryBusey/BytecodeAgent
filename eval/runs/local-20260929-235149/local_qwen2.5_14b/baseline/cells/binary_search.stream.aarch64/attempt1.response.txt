binary_search:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

.Lloop_start:
    cmp x1, #0
    ble .Lfound_neg_one

    cbz x1, .Lfound_neg_one

    mov x3, x0
    mov x4, x1
    lsr x5, x4, #1
    sub x4, x4, x5                    // mid = (left + right) / 2

    cmp x2, xzr
    b.eq .Lfound_neg_one

    ldr x6, [x3, x4, lsl #3]          // Load a[mid]
    cmp x6, x2
    b.gt .Lsearch_left                // If key > a[mid], search left half
    cbz xzr, .Lreturn_mid             // If mid == 0 (key <= a[mid]), return mid

.Lsearch_right:
    sub x4, x1, x5                    // right = mid - 1
    b .Lloop_start                     // Continue searching in the right half

.Lsearch_left:
    add x3, x3, x5                    // left = mid + 1
    mov x1, x4                         // Update right to mid
    b .Lloop_start                      // Continue searching in the left half

.Lreturn_mid:
    mov x0, x4
    b .Lexit

.Lfound_neg_one:
    mov x0, #-1

.Lexit:
    ldp x29, x30, [sp], #16
    ret
