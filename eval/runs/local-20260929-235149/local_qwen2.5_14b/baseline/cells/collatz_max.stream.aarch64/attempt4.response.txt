    stp x19, x20, [sp, #-16]!
    mov x19, x0                // Save function arg n in x19
    mov x20, #0                // Initialize max_steps to 0

.Lloop_start:
    cbz x19, .Lend            // If n == 0, end loop and return result
    sub x25, x19, #1          // Calculate k = n - 1 (for decrementing)
    mov x30, x19              // Save loop counter in x30 to restore later

.Lcollatz_loop:
    lsl x24, x25, #0          // Check if x25 is even
    cbz x24, .Lodd            // If zero (even), go to odd case handler
    asr x26, x25, #1          // Divide by 2: x -> x/2
    add x31, x31, #1          // Increment step count in x31
