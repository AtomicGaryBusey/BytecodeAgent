collatz_max:
    stp x19, x20, [sp, #-16]!
    mov x19, x0                // Save function arg n in x19
    mov x20, #0                // Initialize max_steps to 0

.Lloop_start:
    cbz x19, .Lend            // If n == 0, end loop and return result
    sub x18, x19, #1          // Calculate k = n - 1 (for decrementing)
    mov x21, x18              // Save k in x21 for Collatz sequence

.Lcollatz_loop:
    lsl x22, x21, #0          // Check if x21 is even
    cbz x22, .Lodd            // If zero (even), go to odd case handler
    asr x21, x21, #1          // Divide by 2: x -> x/2
    bne .Lcollatz_loop        // Loop until result is 1

.Lodd:
    add x23, x21, x21         // Calculate 2*x21
    orr x21, x23, #1          // Add one and OR to get 3x+1: x -> 3x+1
    bne .Lcollatz_loop        // Loop until result is 1

.Lupdate_max:
    cmp x20, x25              // Compare current max_steps with steps count (in x25)
    ble .Lnoreturn            // If not greater, go to return without updating max_steps
    mov x20, x25              // Update max_steps with steps count

.Lnoreturn:
    subs x19, x19, #1         // Decrement n by 1 for next iteration (or end loop)
    bne .Lloop_start          // If not zero, start Collatz sequence for next k
.Lend:
    ldp x19, x20, [sp], #16   // Restore caller-saved registers and pop stack
    mov x0, x20               // Return max_steps in x0
    ret                       // Return
