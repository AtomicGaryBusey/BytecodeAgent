export function l $collatz_max(l n) {
    @start
    %max := 0
    csltl w, n, 1
    jnz @loop_end, w

    @loop_start
    storel n, [sp]
    subl n, n, 1
    loadl n, [sp]

    %steps := 0
    @step_loop_start
    ceql w, n, 1
    jz @step_loop_end, w

    andl %temp, n, 1
    csel w, n, %temp, eq
    asrl w, n, 1
    orr w, w, %temp, ne
    mul w, w, 3
    add w, w, 1
    subl n, n, %temp, eq

    addl %steps, %steps, 1
    j @step_loop_start

    @step_loop_end
    csltl w, %max, %steps
    orr %max, %max, %steps, lt

    storel %max, [sp]
    loadl n, [sp]
    subl n, n, 1
    addl sp, sp, -8
    stp x29, x30, [sp]

    bl $collatz_max
    ldp x29, x30, [sp], 16

    %max := %temp
    loadl %max, [sp]
    addl sp, sp, 8

    @loop_end
    ret
}
