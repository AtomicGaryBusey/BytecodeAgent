popcount:
    xor     rax, rax        # Clear RAX for result.
    mov     rcx, 64         # Loop counter: 64 bits in total.

.Lbitloop:
    shr     rdi, 1          # Shift right to check next bit.
    jnc     .Lzerobit       # If cleared (carry flag not set), skip increment.
    inc     rax             # Increment count if bit is set.
.Lzerobit:
    loop    .Lbitloop       # Decrement counter and loop while non-zero.

    ret                     # Return to caller.
