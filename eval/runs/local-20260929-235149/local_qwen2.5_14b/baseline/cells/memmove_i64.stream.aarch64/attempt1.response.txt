memmove_i64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #8*2        // Allocate local variables on the stack
    str x5, [sp, 8]
    str x7, [sp]

    cbz x2, .Ldone          // If n == 0, return

    cmp x1, x0              // dst < src (forward copy)
    b.ge .Lbackward_copy

.Lforward_copy:
    add x4, x30, #8*2      // End of forward copy: restore stack and return
    ldp x29, x30, [sp], #16
    ret

    cmp x7, #1              // n == 1
    b.eq .Lforward_copy_1

.Lforward_loop:
    str x3, [x0]
    add x0, x0, #8          // dst++
    ldr x3, [x5]
    sub x7, x7, #1          // n--
    cmp x7, #1
    b.ne .Lforward_loop

.Lforward_copy_1:
    str x3, [x0]            // copy last element
    b .Lforward_copy

.Lbackward_copy:
    cbz x2, .Ldone          // If n == 0, return

    cmp x1, x0              // src < dst (backward copy)
    b.lt .Lreverse_copy     // Reverse if needed to simplify copying in forward direction

    add x4, x30, #8*2      // End of backward copy: restore stack and return
    ldp x29, x30, [sp], #16
    ret

.Lbackward_loop:
    str x3, [x0]
    add x0, x0, #8          // dst++
    sub x5, x5, #8          // src--
    sub x7, x7, #1          // n--
    cmp x7, #1
    b.ne .Lbackward_loop

.Lforward_copy_1_backward:
    str x3, [x0]            // copy last element
    b .Lforward_copy

.Lreverse_copy:
    add x6, x5, x2, lsl #3  // tmp = src + n * 8
    cmp x7, #1              // n == 1
    b.eq .Lreverse_copy_1   // special case for n == 1

.Lreverse_loop:
    str x3, [x0]
    add x5, x6, #-8         // src = tmp - (i * 8)
    ldr x3, [x5]
    sub x0, x4, #8          // dst = end_of_copy
    sub x7, x7, #1          // n--
    cmp x7, #1
    b.ne .Lreverse_loop

.Lreverse_copy_1:
    str x3, [x4]            // copy last element
    b .Lforward_copy

.Ldone:
    mov x0, #0              // return 0
    ldp x29, x30, [sp], #16
    ret
