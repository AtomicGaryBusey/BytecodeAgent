memmove_i64:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    cbz x2, .Ldone          // If n == 0, return

.Lcheck_direction:
    cmp x1, x0              // Check if dst < src (forward copy)
    b.ge .Lforward_copy     // Branch if forward copy needed
                            // Otherwise continue with backward copy logic below

.Lbackward_copy:
    cbz x2, .Ldone          // If n == 0, return from here directly

    mov x4, x5              // Start address of src for reverse loop
    add x6, x5, x2, lsl #3  // tmp = src + n * 8 (end addr)
                            // Now x4 points to the start and x6 to the end

.Lreverse_loop:
    sub x7, x6, x4          // Calculate remaining elements
    cbz x7, .Lforward_copy_1_backward   // If only one element left

    str x3, [x0]
    add x0, x0, #8          // dst++
    sub x5, x5, #8          // src--
    b Lreverse_loop         // Continue until one element left

.Lforward_copy:
    cmp x1, x4              // Check if reverse is still needed
    b.lt .Lforward_copy     // Direct forward copy if so

.Lforward_loop:
    str x3, [x0]
    add x0, x0, #8          // dst++
    ldr x3, [x5]
    sub x2, x2, #1          // n--
    cbz x2, .Ldone
    b Lforward_loop         // Continue until done

.Lforward_copy_1_backward:
    str x3, [x0]            // copy last element if remaining

.Ldone:
    mov x0, #0              // return 0
    ldp x29, x30, [sp], #16
    ret
