export function l $popcount(l x) {
    %low = andl(x, -0x5555555555555555)
    @addcarry
    %high = andl(x, 0x5555555555555555)
    %sum1 = addl(%low, lsr(%high, 1))
    %low2 = andl(%sum1, -0x3333333333333333)
    @addcarry2
    %high2 = andl(%sum1, 0x3333333333333333)
    %sum2 = addl(%low2, lsr(%high2, 2))
    %low3 = andl(%sum2, -0x0f0f0f0f0f0f0f0f)
    @addcarry3
    %high3 = andl(%sum2, 0x0f0f0f0f0f0f0f0f)
    %sum3 = addl(%low3, lsr(%high3, 4))
    %low4 = andl(%sum3, -0x00ff00ff00ff00ff)
    @addcarry4
    %high4 = andl(%sum3, 0x00ff00ff00ff00ff)
    %sum4 = addl(%low4, lsr(%high4, 8))
    %a = andl(%sum4, -0x0000ffff0000ffff)
    @addcarry5
    %b = andl(%sum4, 0x0000ffff0000ffff)
    %sum5 = addl(%a, lsr(%b, 16))
    %c = andl(%sum5, -0x000000ff000000ff)
    @addcarry6
    %d = andl(%sum5, 0x000000ff000000ff)
    %sum6 = addl(%c, lsr(%d, 32))
    ret %sum6
}
