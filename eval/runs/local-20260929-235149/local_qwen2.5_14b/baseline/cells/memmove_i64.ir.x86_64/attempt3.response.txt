export function l $memmove_i64(l %dst, l %src, l %n) {
    @start:
        csltl %cmp_zero, %n, 0
        jnz .@done

        sub rsp, 8

    @loop_top:
        loadl %src_i, %r12, %src
        loadl %dst_i, %r13, %dst
        
        storel %src_i, %r14, %dst
        
        add %r12, 8, %rcx
        add %r14, 8, %rdi
        sub %n, 1, %rbx

        jz .@done

        dec %rbx
        csltl %cmp_end, %rbx, 0
        jnz .@loop_top
    
    @done:
        add rsp, 8
        storel $0, %rax
}
