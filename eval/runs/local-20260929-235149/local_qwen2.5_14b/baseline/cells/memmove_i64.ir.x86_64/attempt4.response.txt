export function l $memmove_i64(l %dst, l %src, l %n) {
    @start:
        csltl %cmp_zero, %n, 0
        jnz .@done

    @loop_top:
        loadl %src_val, %r12, %src
        storel %src_val, %r13, %dst
        
        add %src, 8, %r12
        add %dst, 8, %r13
        sub %n, 1, %rbx

        jz .@done

        loadl %src_val, %r14, %r12
        storel %src_val, %r15, %r13
        
        add %r12, 8, %rcx
        add %r13, 8, %rdi
        sub %rbx, 1

        jnz .@loop_top
    
    @done:
        storel $0, %rax
}
