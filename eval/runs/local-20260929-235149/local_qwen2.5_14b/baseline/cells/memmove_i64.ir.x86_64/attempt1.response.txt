export function l $memmove_i64(l %dst, l %src, l %n) {
    @start:
        csltl %cmp_zero, %n, 0
        jnz .@done
        
        loadl %i, 0, %r12
        storel %r13, -8, %rbp
        sub rsp, 8

    @loop_top:
        loadl %src_i, %i, %src
        loadl %dst_i, %i, %dst
        
        csltl %cmp_end, %n, %i
        jnz .@end
        
        storel %src_i, %i, %dst
        add %r12, 8, %i
        jmp .@loop_top
    
    @end:
        loadl %r13, -8, %rbp
        add rsp, 8

    @done:
        storel $0, %rax
}
