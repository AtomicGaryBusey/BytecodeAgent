export function l $crc32(l %p, l %n) {
    @start
    %crc =l 0xFFFFFFFF
    %i =l 0

    %cmp =l %i %n
    jnz %cmp @loop

    ret %crc

    @loop
    %byte =w loadub %p, %i
    %crc =w xor %crc, %byte

    incw %i
    csltl %k, %i, 0
    jnz %k @end_loop

    jmp @loop

    @end_loop
    %j =w 8w
    @inner_loop
    csltl %bit, %crc, 0
    jnz %bit @shift_xor
    %crc =w shr %crc, 1
    jmp @inner_loop_done

    @shift_xor
    %crc =w shr %crc, 1
    %crc =w xor %crc, 0xEDB88320

    @inner_loop_done
    decw %j
    csltl %k, %j, 0
    jnz %k @inner_loop

    %crc =l xor %crc, 0xFFFFFFFF
    ret %crc
}
