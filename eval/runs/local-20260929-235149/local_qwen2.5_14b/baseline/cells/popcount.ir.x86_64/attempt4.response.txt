export function l $popcount(l x) {
    @entry:
        %x = arg 0
        %count = constl 0
        %mask = constl -1

    @loop:
        csltl %bit, %mask, %x
        ceql %zero, %bit, %mask
        jnz %zero, @end

        addl %count, %count, %bit
        srlq %mask, %mask, 1
        jmp @loop

    @end:
        ret %count
}
