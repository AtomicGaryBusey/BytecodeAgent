export function l $fib_rec(l %x)
{
    @1
    loadl %n, %x
    csltl %is_zero, %n, 0
    jnz %is_zero, @2
    movl %r, 0
    ret

    @2
    ceql %cmp_one, %n, 1
    jnz %cmp_one, @3
    movl %r, 1
    ret

    @3
    subl %n_minus_1, %n, 1
    stp x29, x30, [sp, #-16]!
    %fib_rec_n_minus_1 =l call $fib_rec(l %n_minus_1)
    ldp x29, x30, [sp], #16

    subl %n_minus_2, %n, 2
    stp x29, x30, [sp, #-16]!
    %fib_rec_n_minus_2 =l call $fib_rec(l %n_minus_2)
    ldp x29, x30, [sp], #16

    addl %r, %fib_rec_n_minus_1, %fib_rec_n_minus_2
    ret
}
