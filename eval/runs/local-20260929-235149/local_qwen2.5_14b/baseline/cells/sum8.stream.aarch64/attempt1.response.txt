sum8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // a - 2*b + 3*c - 4*d
    lsl x5, x1, #1          // 2*b
    sub x0, x0, x5          // a - 2*b
    add x6, x2, x2          // 2*3c = 6c
    add x0, x0, x6, lsr #1  // (a-2b) + 3*c
    sub x7, x3, x3, lsl #2  // -4*d
    add x0, x0, x7

    // +5*e - 6*f
    movk x8, #0x1e          // constant for 5
    mul x9, x4, x8          // e*5
    sub x5, x5, x5, lsl #1  // -6*f
    add x0, x0, x9
    add x0, x0, x5

    // +7*g - 8*h
    movk x10, #0x24         // constant for 7
    mul x11, x6, x10        // g*7
    sub x7, x7, x7, lsl #3  // -8*h
    add x0, x0, x11
    add x0, x0, x7

    ldp x29, x30, [sp], #16
    ret
