isqrt:
    mov     rax, rdi            # Move input to rax for initial checks
    test    rdi, rdi            # Check if input is zero
    jz      .Ldone              # If input is zero, return immediately

    xor     rbx, rbx            # Initialize root bit position counter (RBX)
    mov     rcx, 64             # Start from the highest bit position (RCX)

.Lbitloop:
    cmp     rcx, 0              # Check if we have processed all bits
    jle     .Lcompute           # If done, go to compute phase

    shr     rax, 1              # Shift input right by one bit
    shl     rbx, 1              # Double the root bit position counter
    setc    al                  # Set AL to 0 or 1 based on carry flag (input odd/even)
    add     rbx, rax            # Add current bit of input to root bit position if odd

    dec     rcx                 # Decrement loop counter
    jmp     .Lbitloop           # Continue processing bits

.Lcompute:
    mov     rdx, 0              # Initialize RDX for division
    div     rbx                 # Divide input by accumulated bit positions (RBX)
    add     rbx, 1              # Add one to the root bit position counter

    test    rbx, 1              # Check if RBX is odd
    jz      .Ldone              # If even, done; else adjust result

    sub     rbx, rdx            # Adjust result based on division remainder
.Ldone:
    mov     rdi, rbx            # Move final result to RDI (for correctness checks)
    ret                         # Return result in RAX
