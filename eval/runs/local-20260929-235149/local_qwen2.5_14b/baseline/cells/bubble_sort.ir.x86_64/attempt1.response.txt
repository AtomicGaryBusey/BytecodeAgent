export function l $bubble_sort(l a, l n)
{
    %i = loadl(a)
    %j = subl(n, 1)

outer_loop:
    @label outer_loop

    csltl %j, 0, %done_outer

inner_loop:
    %k = loadl(%i)
    %cmp = addl(%k, 1)
    %next = loadl(%cmp)

    csgtq %next, %k, %swapped
    jnz %swapped, %swap_elements

    csltl %j, %i, %done_inner
    @label %done_inner

    addl $1, %i
    subl 1, %j
    jnz %j, inner_loop

    @label %done_outer

    xorq $0, %rax
    ret

%swap_elements:
    storel %next, %k
    storel %k, %cmp
    movl $1, %rax

    csltl %rax, 0, %continue_outer
    jnz %continue_outer, outer_loop

@label %continue_outer
}
