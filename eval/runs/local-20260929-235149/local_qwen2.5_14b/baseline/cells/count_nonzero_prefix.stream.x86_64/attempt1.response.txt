count_nonzero_prefix:
    xor     rcx, rcx                  # Initialize counter rcx to 0
.Lloop:
    cmp     rdx, rcx                  # Compare index rcx with n (rdx)
    jge     .Lend                     # If rcx >= n, return n in rax
    mov     rax, qword ptr [rdi + rcx * 8] # Load a[rcx]
    test    rax, rax                  # Check if element is zero
    jnz     .Linc                     # If nonzero, increment counter and continue
    jmp     .Lreturn                  # Return current index (rcx in rax)
.Linc:
    inc     rcx                        # Increment counter
    jmp     .Lloop                    # Loop to next element
.Lend:
    mov     rax, rdx                  # Move n into rax if all elements are nonzero
.Lreturn:
    ret                               # Return with result in rax
