count_nonzero_prefix:
    xor     rax, rax                  # Initialize return value to 0
    test    rsi, rsi                  # Check if n is 0
    jle     .Lreturn                  # If n <= 0, return immediately

.Lloop_start:
    cmp     rcx, rsi                  # Compare current index (rcx) with length (rsi)
    je      .Lend                     # If reached end of array, return index in rax
    mov     rbx, qword ptr [rdi + rcx * 8] # Load element a[rcx]
    test    rbx, rbx                  # Check if element is zero
    jz      .Lreturn                  # If found a zero element, return current index
    inc     rax                        # Increment nonzero count (rax)
    inc     rcx                        # Move to next index
    jmp     .Lloop_start              # Loop back to start

.Lend:
    mov     rdi, rbx                  # Restore callee-saved register if used
    ret                               # Return with n in rax if no zeros were found

.Lreturn:
    xor     rcx, rcx                  # Clear rcx
    jmp     .Lloop_start              # Loop back to start to reset loop index
