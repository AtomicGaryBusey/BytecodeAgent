function l $add_overflow(l %a, l %b, l %out) {
    @1:
        loadl %a, %rax
        loadl %b, %rbx
        addl %rbx, %rcx
        movl %rcx, %rdx
        notl %rbx
        addl %rbx, %rcx
        sbb $0, %rax
        jnz @2
        xorb %al, %al
        jmp @3
    @2:

    @3:

}
