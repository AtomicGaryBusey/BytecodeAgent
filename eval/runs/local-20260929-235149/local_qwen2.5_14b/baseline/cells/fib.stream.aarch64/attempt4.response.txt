fib:
    cmp x0, #0
    b.eq L.return_0
    cmp x0, #1
    b.le L.return_1

L.init_loop:
    mov x5, #0          // a = fib(0) = 0
    mov x6, #1          // b = fib(1) = 1
    sub x4, x0, #2      // n -= 2; loop until n <= 1

L.loop:
    add x7, x5, x6      // c = a + b
    mov x5, x6          // a = b
    mov x6, x7          // b = c
    sub x4, x4, #1      // n -= 1; loop until n <= 0

L.return_2:
    cmp x4, #0
    b.eq L.end_function
    mov x5, #1           // a = fib(1) = 1
    mov x6, #0           // b = fib(0) = 0
    sub x4, x0, #1       // n -= 1; loop until n <= 0

L.return_3:
    cmp x4, #0
    b.eq L.end_function
    mov x5, #0           // a = fib(0) = 0
    mov x6, #1           // b = fib(1) = 1
    sub x4, x0, #1       // n -= 1; loop until n <= 0

L.end_loop:
    cmp x4, #0
    b.eq L.return_value
    add x5, x6          // c = a + b
