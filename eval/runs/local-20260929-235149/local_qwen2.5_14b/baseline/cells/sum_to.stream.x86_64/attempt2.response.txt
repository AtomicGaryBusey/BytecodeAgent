sum_to:
    cmp rdi, 1       # Compare n with 1
    jle .Lzero      # Jump if less than or equal to zero
    
    lea rax, [rdi + 1]   # Load effective address for sum formula (n+1)
    shr rax, 2           # Right shift by 2 bits (divide by 4)
    mul rdi              # Multiply by n
    mov rcx, rdi         # Move n to rcx
    sar rcx, 31          # Sign extend n
    add rax, rcx         # Adjust for division rounding

.Lzero:
    xor eax, eax         # Return 0 if n <= 0
