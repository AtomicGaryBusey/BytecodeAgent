sum_to:
    cmp rdi, 1               # Compare n with 1
    jle .Lzero               # Jump if less than or equal to zero

    mov rcx, rdi             # Move n to rcx for calculation
    add rcx, 1               # Add 1 to n in rcx (n+1)
    shr rcx, 2               # Right shift by 2 bits (divide by 4)
    mul rdi                  # Multiply by original n
    sar rcx, 31              # Sign extend rcx for rounding adjustment
    add rax, rcx             # Adjust sum for proper rounding

.Lzero:
    xor eax, eax             # Return 0 if n <= 0
