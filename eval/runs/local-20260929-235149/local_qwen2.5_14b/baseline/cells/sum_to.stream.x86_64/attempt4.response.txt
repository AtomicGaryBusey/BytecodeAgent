sum_to:
    cmp rdi, 1               # Compare n with 1
    jle .Lzero               # Jump if less than or equal to zero

    xor rcx, rcx             # Clear rcx for loop counter
.Lloop:
    add rcx, 1               # Increment counter by 1
    sub rdi, 2               # Decrement n by 2
    jle .Lend                # Jump if less than or equal to zero (end condition)
    add rax, rcx             # Add current value of rcx to sum in rax
    add rax, rcx             # Double count due to decrementing n by 2
    jmp .Lloop               # Loop back

.Lend:
    shr rax, 1               # Right shift to get final result (divide by 2)
    
.Lzero:
    xor eax, eax             # Return 0 if n <= 0
