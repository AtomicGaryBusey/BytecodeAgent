bmax:
    cmp x1, #1          // Compare n with 1
    b.eq L.return_max   // If n == 1, return *a

    ldr x2, [x0], #8    // Load first element into x2 and increment a by 8
    mov x3, x2          // Initialize max value to the first element

L.loop:
    subs x1, x1, #1     // Decrement n by 1 (for termination)
    b.eq L.return_max   // If n == 0, return current max value

    ldr x4, [x0, #-8]   // Load next element into x4 without changing a
    cmp x3, x4          // Compare current max with next element
    b.hs L.continue     // If x3 >= x4, continue looping without changing x3

    mov x3, x4          // Update max value to x4 if it's larger

L.continue:
    b L.loop            // Loop again

L.return_max:
    mov x0, x3          // Move the maximum value into return register x0
    ret                 // Return
