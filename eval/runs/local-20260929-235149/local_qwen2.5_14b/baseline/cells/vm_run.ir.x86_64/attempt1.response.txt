export function l $vm_run(l code, l len, l stack, l cap) {
    @start
    %pc = loadl(code)
    jnz c1
        ret 0
    c1:
    %top = sub cap 1
    storel %top, stack
    %d = loadl(stack)
    ceql %d, 0
    jnz exit
    @loop
    %op = and %pc, 1
    jnz arith
        jnz control
            ret %d
    arith:
    srl %pc, 1
    %val = loadl(code)
    %a = loadl(stack)
    jnz add
        jnz sub
            jnz mul
                jnz dup
                    jnz swap
                        jnz drop
                            jnz over
                                ret %d
    add:
    %b = loadl(add_1, stack)
    %sum = add %a, %b
    storel %sum, (add_2, stack)
    jmp loop_end
    sub_1:
    %b = loadl(sub_1, stack)
    %diff = sub %a, %b
    storel %diff, (sub_2, stack)
    jmp loop_end
    mul_1:
    %b = loadl(mul_1, stack)
    %prod = mul %a, %b
    storel %prod, (mul_2, stack)
    jmp loop_end
    dup_1:
    storel %a, (dup_2, stack)
    jmp loop_end
    swap_1:
    %b = loadl(swap_1, stack)
    storel %b, (swap_2, stack)
    storel %a, (swap_3, stack)
    jmp loop_end
    drop_1:
    sub %top, 8
    jnz end_loop
        ret %d
    over_1:
    %c = loadl(over_1, stack)
    storel %c, (over_2, stack)
    storel %a, (over_3, stack)
    jmp loop_end
    @control
    srl %pc, 1
    jnz halt
        jnz push_
            jnz jumpz
                jnz jump
                    ret %d
    push_:
    storel %val, (push_1, stack)
    add %top, 8
    sub %pc, 2
    jmp loop_end
    jumpz_1:
    csltl %a, %val
    jnz loop_
        jmp control_end
    control_end:
    ret %d
    @loop_
    srl %pc, 32
    add %top, 8
    sub %pc, 2
    jmp loop_end
    jump_1:
    sub %pc, 2
    jmp loop_
    halt:
    ret %d
    @loop_end
    add %pc, 2
    jnz end_loop
        cmp %top, (end_loop-8)
        jlt @start
    end_loop:
}
