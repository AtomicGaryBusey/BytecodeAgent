export function l $vm_run(l code, l len, l stack, l cap) {
    @start
    %pc = loadl(code)
    csltl %pc, 0
    jnz exit
        ret 0
    exit:
    %top = sub cap 1
    storel %top, stack
    %d = loadl(stack)
    ceql %d, 0
    jnz end
        @loop
        %op = and %pc, 1
        jnz arith
            jmp control
        arith:
        srl %pc, 1
        %val = loadl(code)
        %a = loadl(stack)
        ceql %val, 1
        jnz push_
            ceql %val, 2
            jnz add_
                ceql %val, 3
                jnz sub_
                    ceql %val, 4
                    jnz mul_
                        ceql %val, 5
                        jnz dup_
                            ceql %val, 6
                            jnz swap_
                                ceql %val, 7
                                jnz drop_
                                    ceql %val, 8
                                    jnz over_
                                        ret %d
        push_:
        storel %val, (push_1, stack)
        add %top, 8
        sub %pc, 2
        jmp loop_end
        add_:
        %b = loadl(add_1, stack)
        %sum = add %a, %b
        storel %sum, (add_2, stack)
        jmp loop_end
        sub_:
        %b = loadl(sub_1, stack)
        %diff = sub %a, %b
        storel %diff, (sub_2, stack)
        jmp loop_end
        mul_:
        %b = loadl(mul_1, stack)
        %prod = mul %a, %b
        storel %prod, (mul_2, stack)
        jmp loop_end
        dup_:
        storel %a, (dup_2, stack)
        jmp loop_end
        swap_:
        %b = loadl(swap_1, stack)
        storel %b, (swap_2, stack)
        storel %a, (swap_3, stack)
        jmp loop_end
        drop_:
        sub %top, 8
        jnz end_loop
            ret %d
        over_:
        %c = loadl(over_1, stack)
        storel %c, (over_2, stack)
        storel %a, (over_3, stack)
        jmp loop_end
    @control
    srl %pc, 1
    ceql %val, 0
    jnz halt_
        ceql %val, 9
        jnz lt_
            ceql %val, 10
            jnz jumpz_
                ceql %val, 11
                jmp jump_
                    ret %d
    halt_:
    ret %d
    @lt_
    %b = loadl(code)
    csltl %a, %b
    storel $w, (lt_1, stack)
    add %top, 8
    sub %pc, 2
    jmp loop_end
    @jumpz_
    srl %pc, 32
    %val = loadl(code)
    csltl %d, %val
    jnz loop__
        jmp control_end
    control_end:
    ret %d
    @loop__
    add %top, 8
    sub %pc, 2
    jmp loop_end
    @jump_
    srl %pc, 32
    %t = loadl(code)
    sub %pc, 2
    jnz control_
        ret %d
    end_loop:
}
@end
