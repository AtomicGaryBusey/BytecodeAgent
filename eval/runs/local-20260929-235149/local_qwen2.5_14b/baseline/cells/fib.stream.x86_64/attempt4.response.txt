fib:
    cmp rdi, 1          # Compare n to 1
    jle .Lbase         # Jump if less or equal to the base case

    mov rcx, 0             # Initialize fib(0)
    mov rdx, 1             # Initialize fib(1)

.Lloop:
    add rax, rcx           # Compute next Fibonacci number (fib(n-2) + fib(n-1))
    xchg rcx, rdx          # Swap fib(n-2) and fib(n-1)
                            # Store the new value in rdx
    mov rax, rdx           # Move the current fib(n-1) to rax for next addition
    dec rdi                # Decrement n
    jg .Lloop              # Loop until n is 1

.Lbase:
    test rdi, rdi         # Check if n is zero
    sete al               # Set AL to 1 if equal (result is 0), else 0 (result is 1)
    movsx rax, al         # Sign-extend AL into RAX for the result

    ret                    # Return
