fib:
    cmp rdi, 1          # Compare n to 1
    jle .Lbase         # Jump if less or equal to the base case

.Lloop:
    mov rbx, qword ptr [rsp]   # Load fib(n-2) into rbx
    add rbx, qword ptr [rsp + 8] # Add fib(n-1) to it
    adc rax, 0                # Carry flag adds to RAX
    lea rbp, [rbp + 8]        # Update stack pointer for next iteration
    sub rsp, 8                # Reserve space for the new pair
    mov qword ptr [rsp], rbx  # Store fib(n-2) as fib(n)
    loop .Lloop               # Loop until n is 1

.Lbase:
    cmp rdi, 0              # Check if n is 0
    sete al                 # Set AL to 1 if equal to zero (result is 0), else 0 (result is 1)
    movsx rax, al           # Sign-extend AL into RAX for the result

    add rsp, 8              # Restore stack pointer
    ret                      # Return
