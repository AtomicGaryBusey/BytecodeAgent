bmax:
    cmp rsi, 1
    je .Lreturn_max
    lea rcx, [rdi + rsi * 8 - 8]
    mov rax, qword ptr [rcx]

.Lloop:
    sub rcx, 8
    jge .Lcheck_next
    jmp .Lreturn_max

.Lcheck_next:
    mov rdx, qword ptr [rcx]
    cmp rdx, rax
    jle .Lnext
    mov rax, rdx

.Lnext:
    loop .Lloop

.Lreturn_max:
    ret
