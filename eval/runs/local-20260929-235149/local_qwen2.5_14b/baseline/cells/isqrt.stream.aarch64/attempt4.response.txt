isqrt:
    stp     x19, x20, [sp, #-16]!
    mov     x19, #1            // Initial guess for sqrt(n) (r)
.Lloop:
    umulh   x2, x19, x19       // Calculate r*r
    cmp     x2, x0             // Compare r*r with n
    b.ge    .Lexit             // Exit if correct estimate reached
    add     x3, x0, x19        // Calculate numerator: n + r
    mov     x4, x19            // Prepare denominator for next step: (r * 2)
    lsl     x4, x4, #1         // Double r to get denominator: (r * 2)
    cbz     x4, .Lexit         // If denominator is zero, return current estimate
    udiv    x5, x3, x4         // Update guess for sqrt(n): (n + r) / (r * 2)
    fmov    d19, x5            // Move to floating point register for rounding
    cvt.s.u d20, d19           // Convert to float and back to round down
