isqrt:
    stp     x19, x20, [sp, #-16]!
    mov     x19, #1            // Initial guess for sqrt(n) (r)
.Lloop:
    umulh   x2, x19, x19       // Calculate r*r
    cmp     x2, x0             // Compare r*r with n
    b.ge    .Lexit             // Exit if correct estimate reached
    add     x3, x0, x19        // Calculate numerator: n + r
    umulh   x4, x19, #2        // Check denominator fits in 64 bits safely: (r * 2)
