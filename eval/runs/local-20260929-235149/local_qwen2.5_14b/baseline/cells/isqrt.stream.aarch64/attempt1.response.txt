    mov     x19, #0x80000000    // Initial guess for sqrt(n) (r)
.Lloop:

    orr     x3, x2, x2, lsl #1  // r - 1 | (r - 1) << 1 = n >> 30 | n >> 60 if n > 2^30

    cmp     x4, x19             // Check for overflow (r*r < 2^64)
    b.lo    .Lexit              // Exit if no risk of overflow
    orr     x3, x3, x19         // n >> 60 | r - 1: prevent underestimation when n > 2^30 but not far from it
.Lmid:

    add     x5, xzr, x19        // Initialize numerator with denominator for safe division
    umulh   x6, x4, x3          // Check if denominator fits in 64 bits safely: n >> 60 | r - 1
    b.eq    .Ldivide            // Proceed if no overflow risk
    add     x2, x19, #1         // Increase guess to ensure correct rounding up
    umulh   x3, x2, x2
    cmp     x3, x0              // Check new r*r against n for adjustment
    b.ge    .Lloop              // Continue search if too large
.Ldivide:
    lsr     x6, x0, #1          // Shift right by 1 to ensure division is exact (n + r) / (r * 2)
    add     x5, x0, x4          // Calculate numerator: n + r
    umulh   x7, x3, x3          // Check if denominator fits in 64 bits safely after shift
    b.eq    .Ldivide2           // Proceed if no overflow risk
    udiv    x5, x5, x3          // Calculate (n + r) / (r * 2)
    mov     x19, x5             // Update guess with result of division
    umulh   x4, x19, x19        // Check for overflow again

    b.ge    .Lexit              // Exit if correct estimate reached
.Ldivide2:

    mov     x19, x5             // Update guess with result
    umulh   x4, x19, x19        // Final check for correct estimate r*r <= n
    cmp     x4, x0              // Compare against original number n
.Lexit:
