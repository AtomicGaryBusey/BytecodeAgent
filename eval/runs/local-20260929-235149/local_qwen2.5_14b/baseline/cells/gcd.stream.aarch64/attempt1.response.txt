gcd:
    cbz x1, .Lreturn      // If b is zero, return a
.Lloop:
    tbz x0, #63, .Lsub   // Check if both are negative (only need this once)
    subs x2, x0, x1      // a = a - b
    cbz x1, .Lreturn     // If new b is zero, return a
    mov x0, x1           // Otherwise swap a and b
    mov x1, x2
    b .Lloop             // Loop back
.Lsub:
    negs x2, x0          // Invert sign of both if negative
    cbz x1, .Lreturn     // If new b is zero, return a (with inverted signs)
    mov x0, x1           // Otherwise swap a and b
    mov x1, x2
    negs x0, x0          // Un-invert the sign of the non-zero number
    negs x1, x1
    b .Lloop             // Loop back
.Lreturn:
    ret                   // Return
