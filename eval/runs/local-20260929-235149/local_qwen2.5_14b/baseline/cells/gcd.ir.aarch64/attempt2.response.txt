export function l $gcd(l a, l b)
{
    @entry
    %r0 = csltl w0, a, 0
    jnz %r0, .Lexit
    ceql %r1, b, 0
    jnz %r1, .Lreturn
    movl w1, a
    ret x30
.Lloop:
    bandl l2, a, b
    ceql %r3, l2, 0
    jnz %r3, .Lreturn
    shril l4, b, 1
    csltl w5, a, b
    jnz %r5, .Leven
    bandl l6, a, 1
    ceql %r7, l6, 0
    jnz %r7, .Lodd
    addl l8, a, b
    storel l8, [x29]
    ret x30
.Leven:
    loadl l9, [x29]
    loadl l10, [x28]
    ret x30
.Lodd:
    shril l11, a, 1
    csltl w12, b, 0
    jnz %r12, .Lloop
    ceql %r13, b, 0
    jnz %r13, .Lreturn
    movl w14, a
    ret x30
.Lreturn:
    csltl w15, a, b
    jnz %r15, .Lexit
    ceql %r16, b, 0
    jnz %r16, .Lreturn
    movl w17, a
    ret x30
.Lexit:
    csltl w18, b, 0
    jnz %r18, .Lreturn
    ceql %r19, a, 0
    jnz %r19, .Lreturn
    movl w20, b
    ret x30
}
