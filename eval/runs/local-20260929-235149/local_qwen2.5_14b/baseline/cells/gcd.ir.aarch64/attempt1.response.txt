export function l $gcd(l a, l b)
{
    @entry
    %r0 = csltl w0, a, b
    jnz %r0, .Lexit
    %r1 = movl w1, 0
.Lloop:
    %r2 = bandl l0, a, b
    ceql %r3, %r2, 0
    jnz %r3, .Lreturn
    storel l0, [x28]
    %r4 = shril l0, b, 1
    storel l0, [x29]
    %r5 = movl w6, a
    loadl l7, [x29]
    bandl l0, b, 1
    ceql %r6, l0, 0
    jnz %r6, .Lodd
.Leven:
    loadl l8, [x28]
    loadl l9, [x29]
    ret x30
.Lodd:
    bandl l0, a, 1
    ceql %r7, l0, 0
    jnz %r7, .Lloop
    addl l0, a, b
    storel l0, [x28]
    ret x30
.Lreturn:
    loadl l0, [x29]
    ret x30
.Lexit:
    movl w1, a
.Lreturn:
    %r8 = bandl l0, a, b
    ceql %r9, %r8, 0
    jnz %r9, .Lreturn
    loadl l0, [x29]
    ret x30
}
