export function l $gcd(l a, l b)
{
    @entry
    csltl w0, a, 63
    jnz %w0, .Lexit
    ceql %w1, b, 0
    jnz %w1, .Lreturn
    movl x0, a
    ret x30

.Lloop:
    bandl l2, a, b
    ceql %w3, l2, 0
    jnz %w3, .Lreturn
    shril l4, b, 1
    csltl w5, a, b
    jnz %w5, .Leven
    bandl l6, a, 1
    ceql %w7, l6, 0
    jnz %w7, .Lodd
    shril l8, a, 1
    csltl w9, b, 0
    jnz %w9, .Leven2
    movl x0, b
    ret x30

.Leven:
    movl x0, b
    ret x30

.Leven2:
    movl x0, a
    ret x30

.Lodd:
    shril l10, a, 1
    csltl w11, b, 0
    jnz %w11, .Lloop
    ceql %w12, b, 0
    jnz %w12, .Lreturn
    movl x0, a
    ret x30

.Lreturn:
    movl x0, b
    csltl w13, b, 63
    jnz %w13, .Lexit
    ceql %w14, a, 0
    jnz %w14, .Lexit
    movl x0, a
    ret x30

.Lexit:
    movl x0, b
    ret x30
}
