export function l $sum_to(l %n)
{
    @label1
        loadl   %zero, rax                   # Initialize sum to zero.
        csltl   w%positive, rax, %n         # Check if n > 0.
        jnz     w%positive, @label2          # If not, return immediately.

    @label2
        ceql    w%one, %n, l1               # Check for n == 1 case.
        jnz     w%one, @label3               # If n is one, return sum (1).

    @label4
        subl    %one, %n                     # Decrement n by 1.
        call    l $sum_to                    # Recursively calculate sum for n-1.
        addl    rax, rax                    # Double the result of recursion.
        incq    rax                          # Increment to correct sum (2*sum + 1).

    @label3
        ret                                  # Return the result.
}
