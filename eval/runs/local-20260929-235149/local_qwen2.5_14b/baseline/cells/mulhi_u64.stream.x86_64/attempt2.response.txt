mulhi_u64:
    mov     rbx, rdi          # Save a in rbx for later use
    shr     rsi, 32           # Shift b right to start high part in lower half
    mul     rbx               # Multiply full width: rax = low(a*b),	rdx = high(a*b)
    mov     rdx, rdi         # Move a back into rdx
    and     esi, 0xFFFFFFFF   # Mask out upper half of b
    add     rsi, rdx         # Add the masked part with shifted value to get full lower half of b
    mul     rbx              # Multiply again to get final high part in rdx
    mov     rax, rdx         # Move high part to rax for return
