mulhi_u64:
    mov     rdx, rdi          # Save a in rdx for later use
    mov     rbx, rsi          # Save b in rbx for later use
    shr     esi, 32           # Shift b right to start high part in lower half of rax
    mul     rdx               # Multiply full width: rdx:rax = low(a*b)
    mov     rcx, rax          # Save low part in rcx for later use
    and     ebx, 0xFFFFFFFF   # Mask out upper half of b
    shl     rbx, 32           # Shift back the masked b to align with a for next mul
    add     rbx, rdx         # Add the masked part with shifted value to get full lower half of b
    mov     rax, rcx          # Move low part (rcx) back to rax for second multiplication
    mul     rbx               # Multiply again: rdx:rax = high(a*b)
    mov     rax, rdx         # Move high part to rax for return
