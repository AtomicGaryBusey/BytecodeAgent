mulhi_u64:
    mov     rdx, rdi          # Move a into rdx for later use
    shr     rsi, 32           # Shift b right to start high part in lower half
    mul     rdx               # Multiply full width: rax = low(a*b),	rdx = high(a*b)
    mov     rbx, rdi          # Save a in rbx
    and     rsi, 0xFFFFFFFF   # Mask out upper half of b
