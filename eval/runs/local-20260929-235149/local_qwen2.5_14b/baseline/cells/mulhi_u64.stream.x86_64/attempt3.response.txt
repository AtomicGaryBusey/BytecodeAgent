mulhi_u64:
    mov     rdx, rdi          # Save a in rdx for later use
    shr     rsi, 32           # Shift b right to start high part in lower half of rax
    mul     rdx               # Multiply full width: rdx:rax = low(a*b)
    mov     rbx, rax          # Save low part in rbx for later use
    and     esi, 0xFFFFFFFF   # Mask out upper half of b
    shl     rsi, 32           # Shift back the masked b to align with a for next mul
    add     rsi, rdx         # Add the masked part with shifted value to get full lower half of b
    mov     rax, rbx          # Move low part (rbx) back to rax for second multiplication
    mul     esi               # Multiply again: rdx:rax = high(a*b)
    mov     rax, rdx         # Move high part to rax for return
