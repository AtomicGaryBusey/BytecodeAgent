export function l $isqrt(l n) {
@start
%high = sllq n, -32
%bits = orl %high, n
%max = li 0x100000000
%root = mov %max

@store = srlq %root, 1

@loop
%low = andl %bits, -2
%guess = orl %store, %low
%squarehi = muluh %guess, %guess
%squarelo = mullo %guess, %guess
%csltl %cmp, n, %squarelo
%ceql %cmpeq, %squarehi, 0
%andw %combined, %cmp, %cmpeq
%jnz @done, %combined

@store = mov %guess
%root = shr %store, 1
%high = shr %bits, 1
%max = shr %max, 2
%jz @loop, %max

@done
%result = mov %store
retl %result
}
