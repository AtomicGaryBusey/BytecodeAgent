export function l $fib_rec(l n) {
    @0:
        csltl w %cmp =l ==z n 0
        jnz   @2 %cmp
            movl rax 0
            ret
        @2:
        csltl w %cmp1 =l ==z n 1
        jnz   @4 %cmp1
            movl rax 1
            ret
        @4:
        subl rsp 8
        storeq n -8(%rsp)
        call $fib_rec(l(loadq -8(%rsp)))
        addl rsp 8
        decq n
        subl rsp 8
        storeq n -8(%rsp)
        call $fib_rec(l(loadq -8(%rsp)))
        addl rsp 8
        addq rax rdi
        ret
}
