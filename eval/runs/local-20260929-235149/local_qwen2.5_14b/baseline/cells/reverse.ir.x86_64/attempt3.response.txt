export function l $reverse(l %a, l %n) {
    @begin
    csltl %end %n 0
    jnz .Lend

.Lloop:
    movl %i %n
    subl %half %i 1
    brz .Lswap

.loadstore:
    loadl %val1 l (%a,%half,8)
    loadl %val2 l (%a,%i,8)
    storel l (%a,%half,8) %val2
    storel l (%a,%i,8) %val1
.Lswap:
    subl %n %n 1
    jnz .Lloop

.Lend:
    movl $0 l %retval
    ret
}
