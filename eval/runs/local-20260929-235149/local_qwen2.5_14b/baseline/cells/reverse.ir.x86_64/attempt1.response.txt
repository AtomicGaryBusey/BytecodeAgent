export function l $reverse(l %a, l %n) {
    @begin
    csltl %end %n 0
    jnz .Lend

.Lloop:
    csltl %half %n 2
    subl %idx %n %half
    brz .Lswap

.loadstore:
    loadl %val1 l (%a,%half,8)
    loadl %val2 l (%a,%idx,8)
    storel l (%a,%half,8) %val2
    storel l (%a,%idx,8) %val1
.Lswap:
    addl %half %half 1
    brnz .Lloop %half

.Lend:
    movl $0 l %retval
    ret
}
