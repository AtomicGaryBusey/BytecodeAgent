export function l $reverse(l %a, l %n) {
    @begin
    movl %end %n 1
    negl %end %end
    jnz .Lend

.Lloop:
    csltl %half %n 0
    shrq %i %n 1
    brz .Lswap

.loadstore:
    loadl %val1 l (%a,%i,8)
    loadl %val2 l (%a,%half,8)
    storel l (%a,%half,8) %val1
    storel l (%a,%i,8) %val2
.Lswap:
    addq %n %end 1
    jnz .Lloop

.Lend:
    movl $0 l %retval
    ret
}
