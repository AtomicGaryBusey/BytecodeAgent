crc32:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x18, #0xFFFFFFFF      // Initialize crc with 0xFFFFFFFF
    cbz x1, .Lend             // If n == 0, return immediately

.Lloop:
    ldrb w17, [x0], #1        // Load byte and increment pointer p
    eor x18, x18, x17         // crc ^= byte
    mov x2, #0                 // Initialize bit counter
.Lbitloop:
    tbz x18, #0, .Lshift      // If (crc & 1) == 0
    ldr w19, [x26]            // Load constant into register
    eor x18, x18, x19         // crc = (crc >> 1) ^ 0xEDB88320
.Lshift:
    asr x18, x18, #1          // crc >>= 1
    add x2, x2, #1            // Increment bit counter
    cmp x2, #8                // Check if all bits processed
    blt .Lbitloop             // Continue loop if not

.Lend:
    eor x0, x18, #0xFFFFFFFF  // Final xor with 0xFFFFFFFF
    ldp x29, x30, [sp], #16   // Restore stack pointer and return
    mov x26, #0xEDB88320      // Define the constant in a register
