export function l $memmove_i64(l dst, l src, l n)
{
  @start
    %i = loadl zero
    %end = addl n i1

  @loop
    csltl i end i0
    jnz i done

    storel loadl src i8 dst i8
    %dstp = addl dst i8
    %srcl = addl src i8
    %np = subl n i1
    storel np srcl np dstp

    %i = addl i l1
    jmp loop

  @done
    loadl zero x0
}
