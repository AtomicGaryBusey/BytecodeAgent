export function l $collatz_max(l n) {
    %max := 0;
    @loop:
        csltl w %done, n, 1;
        jnz w %done, @exit;

        %steps := 0;
        @inner_loop:
            ceql w %end, n, 1;
            jz w %end, @next_k;

            cltl w %even, n, -63;
            jz w %even, @multiply_three_and_add_one;

            sarl l n, n, 1;
            jmp @check_end;

        @multiply_three_and_add_one:
            imull l n, n, 3;
            addl l n, n, 1;

        @check_end:
        csltl w %end, n, 1;
        jz w %end, @inner_loop;

    @next_k:
        subq l n, n, 1;
        imull l rsi, rsi, rsi;
        addl l %max, %max, %steps;
        jmp @loop;

    @exit:
        movl l rax, %max;
}
