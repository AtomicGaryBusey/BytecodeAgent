count_nonzero_prefix:
    cmp x1, #0
    ble .Lreturn_zero
.Lloop:
    cbz x1, .Lreturn_index
    ldr x2, [x0], #8
    cmp x2, #0
    bne .Lnext_element
    add x0, x1, #-1
    ret
.Lnext_element:
    sub x1, x1, #1
    bne .Lloop
.Lreturn_index:
    mov x0, x1
    ret
.Lreturn_zero:
    mov x0, x1
    ret
