export function l $binary_search(l a, l n, l key)
{
    %lo = loadl a
    %hi = add n, -1
    @loop_start:
        csltl %diff, %lo, %hi
        jnz %diff, @exit
        %mid = srl %diff, 1
        storel %mid, x9
        %index = loadl a, %mid
        ceql %index, key, %eq
        jnz %eq, @found
        csgt key, %index, %lo_lt_mid
        jnz %lo_lt_mid, @move_hi_up
    @move_lo_up:
        %lo = add %lo, 1
        jmp @loop_start
    @move_hi_up:
        %hi = sub %hi, 1
        jmp @loop_start
    @found:
        storel x9, a
        ret
    @exit:
        mov -1, x0
        ret
}
