export function l $fib(l n)
{
    @entry
    %a = 0                 # a = 0
    %b = 1                 # b = 1
    csltl %i, n, n         # i = n - 1 (n > 0)
    jnz %i, @loop          # if i == 0: jump to exit
    @loop
    storel a, &b           # fib[n-2] = a
    csltl %t, b, a         # t = b + a (64-bit wrap)
    addl a, b              # a += b
    ceql %i, i, 1          # i = i - 1 (n > 0)
    jnz i, @exit           # if i == 0: jump to exit
    subl n, n, 1          # n -= 1
    jmp @loop              # loop
    @exit
    loadl b, &b            # return b
}
