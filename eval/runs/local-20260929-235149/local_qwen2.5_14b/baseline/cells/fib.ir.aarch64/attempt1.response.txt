export function l $fib(l n)
{
    @entry
    %a = loadl n, 0       # a = 0
    %b = loadl n, 1       # b = 1
    csltl %i, n, n        # i = n - 1 (n > 0)
    jnz %i, @loop         # if i == 0: jump to exit
    @loop
    storel a, n, 8       # fib[n-2] = a
    csltl %t, b, b        # t = b + a (64-bit wrap)
    addl a, b, a          # a += b
    jnz i, @exit          # if i == 0: jump to exit
    subl n, n, 1         # n -= 1
    ceql %i, i, n        # i = i - 1 (n > 0)
    jmp @loop             # loop
    @exit
    storel a, n, 8       # fib[n-2] = a
    loadl b, n            # return b
}
