collatz_max:
    mov     rbx, rsp                    # Save caller's stack pointer in RBX
    xor     r12, r12                    # Initialize max steps to 0 (using XOR for efficiency)
    test    rdi, rdi                    # Test if n is less than or equal to 0
    jle     .Lreturn                    # If so, return immediately

.Lloop_top:
    mov     rcx, rdi                    # Copy n to RCX as loop counter
    dec     rcx                          # Start loop with k = n - 1 (k starts at n and goes down)
    jz      .Lreturn                    # Exit if n is 1

.Lstep_loop:
    test    rcx, rcx                     # Test for k == 0 to exit
    jle     .Lupdate_max                 # If k <= max found so far, update max steps and go back to loop top

    mov     rax, rcx                    # Copy k to RAX for counting steps
.Lstep_loop_start:
    test    rax, 1                       # Check if current number is odd
    jz      .Ldivide_even                # If even, jump to divide by 2
    lea     rdx, [rax*4+1]              # Compute 3*RAX + 1 and store in RDX
    add     rax, rdx                    # Update RAX with new value (3*RAX + 1)
.Ldivide_even:
    shr     rax, 1                      # Divide by 2 using shift right
    jnz     .Lstep_loop_start           # Continue loop if result is not zero

    mov     rcx, rdi                    # Restore RCX to n for comparison
    cmp     rax, r12                     # Compare current step count to max found so far
    cmovg   r12, rax                   # Update max steps if higher
    dec     rdi                         # Decrement n for next iteration

.Lupdate_max:
    test    rcx, rcx                    # Check for loop exit condition
    jle     .Lloop_top                  # Go back to loop top if k > 0

.Lreturn:
    mov     rsp, rbx                    # Restore caller's stack pointer
    ret
