collatz_max:
    mov     rbp, rsp                    # Save rsp in a callee-saved reg.
    lea     rcx, [rdi+1]                # Start loop with k = n + 1 to avoid special case handling for k == 1

.Lloop_top:
    cmp     rcx, 2                      # Loop while (k > 1)
    jle     .Lreturn                    # When k <= 1, return.
    sub     rcx, 1                      # Decrement k in rcx
    mov     rax, rcx                    # Copy k to rax for counting steps

.Lstep_loop:
    test    rax, -2                     # Test if RAX is even (using -2 as mask for all bits except the lowest one)
    jz      .Ldivide_even               # If even, divide by 2
    lea     rdx, [rax*4+1]              # Compute 3*RAX + 1 and store in RDX
    add     rax, rdx                    # Update RAX to new value (3*RAX + 1)
    dec     rcx                         # Decrement counter (k)

.Ldivide_even:
    shr     rax, 1                      # Divide by 2 using shift right
    jnz     .Lstep_loop                 # If result is not zero, continue loop

    cmp     rcx, r12                     # Compare current step count to max found so far
    jle     .Lloop_top                   # If less or equal, go back to loop
    mov     r12, rcx                    # Update max steps if higher

.Lreturn:
    mov     rsp, rbp                    # Restore caller's stack pointer.
    ret
