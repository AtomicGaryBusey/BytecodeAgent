collatz_max:
    mov     rbx, rsp                    # Save caller's stack pointer in RBX
    xor     rax, rax                    # Initialize max steps to 0 (using XOR for efficiency)
    mov     r12, rdi                    # Store n in R12 for loop control

.Lloop_top:
    test    r12, r12                     # Test if n is less than or equal to 0
    jle     .Lreturn                     # If so, return immediately

    xor     rcx, rcx                      # Initialize step count for each k to 0
    mov     rdi, r12                      # Copy loop limit to RDI
.Linner_loop:
    test    rcx, rcx                       # Test if we are done with inner loop (k <= 0)
    jle     .Lupdate_max                   # If so, update max steps and prepare for next n

    mov     rax, rdi                      # Copy k to RAX for counting steps
.Lstep_loop_start:
    test    rax, 1                         # Check if current number is odd
    jz      .Leven_divide                  # If even, jump to divide by 2
    lea     rdx, [rax*4+1]                # Compute 3*RAX + 1 and store in RDX
    add     rax, rdx                      # Update RAX with new value (3*RAX + 1)
.Leven_divide:
    shr     rax, 1                        # Divide by 2 using shift right
    jnz     .Lstep_loop_start             # Continue loop if result is not zero

    inc     rcx                            # Increment step count for current k
    dec     rdi                            # Decrement k for next iteration
    test    rdi, rdi                      # Check if we are done with inner loop (k <= 0)
    jg      .Linner_loop                  # Continue inner loop

.Lupdate_max:
    cmp     rcx, rax                       # Compare current max steps to newly computed step count for k
    cmovg   rax, rcx                     # Update max steps if higher

    dec     r12                            # Decrement n for next iteration
    jnz     .Lloop_top                    # Continue outer loop

.Lreturn:
    mov     rsp, rbx                      # Restore caller's stack pointer
    ret
