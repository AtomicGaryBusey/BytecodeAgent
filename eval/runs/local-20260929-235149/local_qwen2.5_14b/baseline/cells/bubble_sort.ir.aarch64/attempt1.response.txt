export function l $bubble_sort(l x0, l x1)
{
    %i = 0;
    @loop_i:
        csltl w2, %i, x1;
        jnz w2, @done;

        %j = 0;
        @loop_j:
            csltl w3, %j, x1;
            jnz w3, @end_loop_j;

            %left = loadl (x0 + %j * 8);
            %right = loadl (x0 + (%j + 1) * 8);
            ceql w4, %left, %right;
            csltl w5, %left, %right;
            orr w6, w4, w5;
            jz w6, @skip_swap;

            storel x0 + %j * 8, %right;
            storel x0 + (%j + 1) * 8, %left;

        @skip_swap:
        add %j = %j + 1;
        jmp @loop_j;

    @end_loop_j:

    add %i = %i + 1;
    jmp @loop_i;

    @done:
    ret 0;
}
