export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%s1 =l add %p01, %p00hi
	%s1lo =l and %s1, 4294967295
	%s1hi =l shr %s1, 32
	%s2 =l add %p10, %s1lo
	%s2hi =l shr %s2, 32
	%t =l add %p11, %s1hi
	%hi =l add %t, %s2hi
	ret %hi
}
