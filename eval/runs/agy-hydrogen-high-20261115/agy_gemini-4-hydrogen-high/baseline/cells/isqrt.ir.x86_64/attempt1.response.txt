export function l $isqrt(l %n) {
@start
    jmp @loop
@loop
    %lo =l phi @start 0, @if_le %lo_next, @if_gt %lo
    %hi =l phi @start 4294967295, @if_le %hi, @if_gt %hi_next
    %ans =l phi @start 0, @if_le %mid, @if_gt %ans
    %cond =w culel %lo, %hi
    jnz %cond, @body, @done
@body
    %sum =l add %lo, %hi
    %mid =l shr %sum, 1
    %sq =l mul %mid, %mid
    %le =w culel %sq, %n
    jnz %le, @if_le, @if_gt
@if_le
    %lo_next =l add %mid, 1
    jmp @loop
@if_gt
    %hi_next =l sub %mid, 1
    jmp @loop
@done
    ret %ans
}
