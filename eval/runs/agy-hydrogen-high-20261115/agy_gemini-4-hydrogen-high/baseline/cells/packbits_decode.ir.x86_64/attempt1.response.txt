export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i_next
	%o =l phi @start 0, @next %o_next
	%has_hdr =w csltl %i, %n
	jnz %has_hdr, @read_hdr, @done
@read_hdr
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%i1 =l add %i, 1
	%is_nop =w ceqw %hw, -128
	jnz %is_nop, @next, @not_nop
@not_nop
	%is_lit =w csgew %hw, 0
	jnz %is_lit, @lit_check, @rep_check
@lit_check
	%hl =l extsw %hw
	%cnt =l add %hl, 1
	%end_i =l add %i1, %cnt
	%ok =w cslel %end_i, %n
	jnz %ok, @lit_loop_init, @err_trunc
@lit_loop_init
	jmp @lit_loop
@lit_loop
	%li =l phi @lit_loop_init %i1, @lit_write %li1
	%lo =l phi @lit_loop_init %o, @lit_write %lo1
	%rem =l phi @lit_loop_init %cnt, @lit_write %rem1
	%more =w csgtl %rem, 0
	jnz %more, @lit_check_cap, @next
@lit_check_cap
	%has_cap =w csltl %lo, %cap
	jnz %has_cap, @lit_write, @err_cap
@lit_write
	%sp =l add %src, %li
	%b =w loadub %sp
	%dp =l add %out, %lo
	storeb %b, %dp
	%li1 =l add %li, 1
	%lo1 =l add %lo, 1
	%rem1 =l sub %rem, 1
	jmp @lit_loop
@rep_check
	%hl_r =l extsw %hw
	%cnt_r =l sub 1, %hl_r
	%ok_r =w csltl %i1, %n
	jnz %ok_r, @rep_loop_init, @err_trunc
@rep_loop_init
	%rp =l add %src, %i1
	%rb =w loadub %rp
	%i2 =l add %i1, 1
	jmp @rep_loop
@rep_loop
	%ro =l phi @rep_loop_init %o, @rep_write %ro1
	%rrem =l phi @rep_loop_init %cnt_r, @rep_write %rrem1
	%rmore =w csgtl %rrem, 0
	jnz %rmore, @rep_check_cap, @next
@rep_check_cap
	%rhas_cap =w csltl %ro, %cap
	jnz %rhas_cap, @rep_write, @err_cap
@rep_write
	%rdp =l add %out, %ro
	storeb %rb, %rdp
	%ro1 =l add %ro, 1
	%rrem1 =l sub %rrem, 1
	jmp @rep_loop
@next
	%i_next =l phi @read_hdr %i1, @lit_loop %li, @rep_loop %i2
	%o_next =l phi @read_hdr %o, @lit_loop %lo, @rep_loop %ro
	jmp @loop
@done
	ret %o
@err_cap
	ret -1
@err_trunc
	ret -2
}
