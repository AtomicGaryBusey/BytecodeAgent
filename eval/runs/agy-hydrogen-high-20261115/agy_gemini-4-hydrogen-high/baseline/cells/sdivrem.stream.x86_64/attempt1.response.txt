sdivrem:
    mov rcx, rdx
    mov rax, rdi
    test rsi, rsi
    jz .Ldiv_zero
    cmp rsi, -1
    je .Ldiv_neg_one
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Ldiv_neg_one:
    neg rax
    xor edx, edx
    mov qword ptr [rcx], rdx
    ret
.Ldiv_zero:
    xor eax, eax
    mov qword ptr [rcx], rdi
    ret
