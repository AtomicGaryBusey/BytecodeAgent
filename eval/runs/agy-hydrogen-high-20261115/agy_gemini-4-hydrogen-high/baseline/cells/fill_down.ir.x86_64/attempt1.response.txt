export function l $fill_down(l %a, l %n, l %start) {
@start
	%cond0 =w csltl 0, %n
	jnz %cond0, @loop, @done
@loop
	%i =l phi @start 0, @loop %i1
	%ptr =l phi @start %a, @loop %ptr1
	%val =l phi @start %start, @loop %val1
	storel %val, %ptr
	%i1 =l add %i, 1
	%ptr1 =l add %ptr, 8
	%val1 =l sub %val, 1
	%cond1 =w csltl %i1, %n
	jnz %cond1, @loop, @done
@done
	ret %n
}
