@start

	%ret =l mul %groups, 4

@loop_hdr

	%b2 =w loadub %p2

	%v2 =w and %s2, 63
	%v3 =w and %val, 63

	%a0 =w add %v0, 65
	%a0_gt25 =w csgtw %v0, 25
	%a0_d1 =w mul %a0_gt25, 6
	%a0_1 =w add %a0, %a0_d1

	%a0_d2 =w mul %a0_gt51, 75

	%a0_3 =w sub %a0_2, %a0_d3

	%c0 =w add %a0_3, %a0_d4

	%a1 =w add %v1, 65

	%a1_d1 =w mul %a1_gt25, 6

	%a1_d2 =w mul %a1_gt51, 75

	%a1_d3 =w mul %a1_gt61, 15

	%a2_d1 =w mul %a2_gt25, 6
	%a2_1 =w add %a2, %a2_d1
	%a2_gt51 =w csgtw %v2, 51
	%a2_d2 =w mul %a2_gt51, 75
	%a2_2 =w sub %a2_1, %a2_d2
	%a2_gt61 =w csgtw %v2, 61
	%a2_d3 =w mul %a2_gt61, 15
	%a2_3 =w sub %a2_2, %a2_d3
	%a2_gt62 =w csgtw %v2, 62
	%a2_d4 =w mul %a2_gt62, 3
	%c2 =w add %a2_3, %a2_d4

	%a3 =w add %v3, 65
	%a3_gt25 =w csgtw %v3, 25
	%a3_d1 =w mul %a3_gt25, 6
	%a3_1 =w add %a3, %a3_d1
	%a3_gt51 =w csgtw %v3, 51
	%a3_d2 =w mul %a3_gt51, 75
	%a3_2 =w sub %a3_1, %a3_d2
	%a3_gt61 =w csgtw %v3, 61
	%a3_d3 =w mul %a3_gt61, 15
	%a3_3 =w sub %a3_2, %a3_d3
	%a3_gt62 =w csgtw %v3, 62
	%a3_d4 =w mul %a3_gt62, 3
	%c3 =w add %a3_3, %a3_d4

	storeb %c0, %po
	%po1 =l add %po, 1
	storeb %c1, %po1
	%po2 =l add %po, 2
	storeb %c2, %po2
	%po3 =l add %po, 3
	storeb %c3, %po3

	%ps_next =l add %ps, 3
	%po_next =l add %po, 4
	%rem_next =l sub %rem, 3
	jmp @loop_hdr
@tail
	%is0 =w ceql %rem, 0
	jnz %is0, @done, @tail_nz
@tail_nz
	%is1 =w ceql %rem, 1
	jnz %is1, @tail1, @tail2
@tail1
	%tb0 =w loadub %ps
	%tv0 =w shr %tb0, 2
	%tb0_low =w and %tb0, 3
	%tv1 =w shl %tb0_low, 4

	%u0 =w add %tv0, 65
	%u0_gt25 =w csgtw %tv0, 25
	%u0_d1 =w mul %u0_gt25, 6
	%u0_1 =w add %u0, %u0_d1

	%u0_d2 =w mul %u0_gt51, 75
	%u0_2 =w sub %u0_1, %u0_d2

	%u0_d3 =w mul %u0_gt61, 15
	%u0_3 =w sub %u0_2, %u0_d3

	%u1_d2 =w mul %u1_gt51, 75
	%u1_2 =w sub %u1_1, %u1_d2
	%u1_gt61 =w csgtw %tv1, 61
	%u1_d3 =w mul %u1_gt61, 15

	storeb %tc1, %tpo1

	storeb 61, %tpo2
	%tpo3 =l add %po, 3
	storeb 61, %tpo3
	jmp @done

	%w_p1 =l add %ps, 1
	%wb0 =w loadub %ps
	%wb1 =w loadub %w_p1
	%wv0 =w shr %wb0, 2
	%wb0_low =w and %wb0, 3
	%wb0_hi =w shl %wb0_low, 4

	%w0_d2 =w mul %w0_gt51, 75

	%w0_d3 =w mul %w0_gt61, 15

	%w0_d4 =w mul %w0_gt62, 3

	%w1_gt25 =w csgtw %wv1, 25
	%w1_d1 =w mul %w1_gt25, 6
	%w1_1 =w add %w1, %w1_d1
	%w1_gt51 =w csgtw %wv1, 51
	%w1_d2 =w mul %w1_gt51, 75
	%w1_2 =w sub %w1_1, %w1_d2
	%w1_gt61 =w csgtw %wv1, 61
	%w1_d3 =w mul %w1_gt61, 15
	%w1_3 =w sub %w1_2, %w1_d3
	%w1_gt62 =w csgtw %wv1, 62
	%w1_d4 =w mul %w1_gt62, 3
	%wc1 =w add %w1_3, %w1_d4

	%w2 =w add %wv2, 65
	%w2_gt25 =w csgtw %wv2, 25
	%w2_d1 =w mul %w2_gt25, 6
	%w2_1 =w add %w2, %w2_d1
	%w2_gt51 =w csgtw %wv2, 51
	%w2_d2 =w mul %w2_gt51, 75
	%w2_2 =w sub %w2_1, %w2_d2

	%w2_d4 =w mul %w2_gt62, 3

	storeb %wc1, %wpo1
	%wpo2 =l add %po, 2
	storeb %wc2, %wpo2
	%wpo3 =l add %po, 3
	storeb 61, %wpo3
	jmp @done

	ret %ret
}
