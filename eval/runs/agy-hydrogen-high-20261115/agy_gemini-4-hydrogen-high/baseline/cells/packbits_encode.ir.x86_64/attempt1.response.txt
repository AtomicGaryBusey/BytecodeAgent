export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @main_loop

@main_loop
    %i =l phi @start 0, @emit_repeat %i_rep, @emit_lit %i_lit
    %pos =l phi @start 0, @emit_repeat %pos_rep, @emit_lit %pos_lit
    %cond_main =w csltl %i, %n
    jnz %cond_main, @measure_run, @done

@measure_run
    %p_i =l add %src, %i
    %b =w loadub %p_i
    jmp @run_loop

@run_loop
    %r =w phi @measure_run 1, @run_next %r_next
    %cond_r128 =w csltw %r, 128
    jnz %cond_r128, @run_chk_n, @run_done

@run_chk_n
    %r_l =l extuw %r
    %idx =l add %i, %r_l
    %cond_rn =w csltl %idx, %n
    jnz %cond_rn, @run_chk_eq, @run_done

@run_chk_eq
    %p_idx =l add %src, %idx
    %b_cur =w loadub %p_idx
    %cond_eq =w ceqw %b_cur, %b
    jnz %cond_eq, @run_next, @run_done

@run_next
    %r_next =w add %r, 1
    jmp @run_loop

@run_done
    %is_rep =w csgew %r, 3
    jnz %is_rep, @do_repeat, @do_literal

@do_repeat
    %need_rep =l add %pos, 2
    %fit_rep =w cslel %need_rep, %cap
    jnz %fit_rep, @emit_repeat, @oom

@emit_repeat
    %hdr_rep =w sub 257, %r
    %p_out_rep0 =l add %out, %pos
    storeb %hdr_rep, %p_out_rep0
    %p_out_rep1 =l add %p_out_rep0, 1
    storeb %b, %p_out_rep1
    %r_final_l =l extuw %r
    %i_rep =l add %i, %r_final_l
    %pos_rep =l copy %need_rep
    jmp @main_loop

@do_literal
    jmp @lit_loop

@lit_loop
    %L =w phi @do_literal 1, @lit_next %L_next
    %cond_L128 =w csltw %L, 128
    jnz %cond_L128, @lit_chk_n, @lit_done

@lit_chk_n
    %L_l =l extuw %L
    %j =l add %i, %L_l
    %cond_jn =w csltl %j, %n
    jnz %cond_jn, @lit_chk_run3_bounds, @lit_done

@lit_chk_run3_bounds
    %j2 =l add %j, 2
    %cond_j2n =w csltl %j2, %n
    jnz %cond_j2n, @lit_chk_run3_eq1, @lit_next

@lit_chk_run3_eq1
    %p_j0 =l add %src, %j
    %c0 =w loadub %p_j0
    %p_j1 =l add %p_j0, 1
    %c1 =w loadub %p_j1
    %eq01 =w ceqw %c0, %c1
    jnz %eq01, @lit_chk_run3_eq2, @lit_next

@lit_chk_run3_eq2
    %p_j2 =l add %p_j0, 2
    %c2 =w loadub %p_j2
    %eq02 =w ceqw %c0, %c2
    jnz %eq02, @lit_done, @lit_next

@lit_next
    %L_next =w add %L, 1
    jmp @lit_loop

@lit_done
    %L_fin_l =l extuw %L
    %pkt_len =l add %L_fin_l, 1
    %need_lit =l add %pos, %pkt_len
    %fit_lit =w cslel %need_lit, %cap
    jnz %fit_lit, @lit_hdr, @oom

@lit_hdr
    %hdr_lit =w sub %L, 1
    %p_out_lit =l add %out, %pos
    storeb %hdr_lit, %p_out_lit
    %pos_hdr =l add %pos, 1
    %i_lit =l add %i, %L_fin_l
    jmp @copy_loop

@copy_loop
    %k =l phi @lit_hdr %i, @copy_step %k_next
    %p_pos =l phi @lit_hdr %pos_hdr, @copy_step %p_pos_next
    %cond_copy =w csltl %k, %i_lit
    jnz %cond_copy, @copy_step, @emit_lit

@copy_step
    %p_src_k =l add %src, %k
    %byte_k =w loadub %p_src_k
    %p_out_k =l add %out, %p_pos
    storeb %byte_k, %p_out_k
    %k_next =l add %k, 1
    %p_pos_next =l add %p_pos, 1
    jmp @copy_loop

@emit_lit
    %pos_lit =l copy %p_pos
    jmp @main_loop

@oom
    ret -1

@done
    ret %pos
}
