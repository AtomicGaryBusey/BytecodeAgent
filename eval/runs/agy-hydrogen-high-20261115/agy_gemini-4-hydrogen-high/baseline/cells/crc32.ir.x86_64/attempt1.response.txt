export function l $crc32(l %p, l %n) {
@start
	%has_bytes =w csgtl %n, 0
	jnz %has_bytes, @byte_loop, @done
@byte_loop
	%ptr =l phi @start %p, @byte_next %ptr_next
	%rem =l phi @start %n, @byte_next %rem_next
	%crc =w phi @start -1, @byte_next %c_next
	%b =w loadub %ptr
	%crc_xor =w xor %crc, %b
	jmp @bit_loop
@bit_loop
	%c =w phi @byte_loop %crc_xor, @bit_loop %c_next
	%i =w phi @byte_loop 8, @bit_loop %i_next
	%lsb =w and %c, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -306674912
	%c_shr =w shr %c, 1
	%c_next =w xor %c_shr, %poly
	%i_next =w sub %i, 1
	%bit_more =w cnew %i_next, 0
	jnz %bit_more, @bit_loop, @byte_next
@byte_next
	%ptr_next =l add %ptr, 1
	%rem_next =l sub %rem, 1
	%byte_more =w csgtl %rem_next, 0
	jnz %byte_more, @byte_loop, @done
@done
	%crc_final =w phi @start -1, @byte_next %c_next
	%res_w =w xor %crc_final, -1
	%res =l extuw %res_w
	ret %res
}
