export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @header
@header
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ps =l add %src, %off
	%x =l loadl %ps
	%y =l call %f(l %x)
	%pd =l add %dst, %off
	storel %y, %pd
	%lt =w csltl %y, %x
	%ltl =l extuw %lt
	%cnt1 =l add %cnt, %ltl
	%i1 =l add %i, 1
	jmp @header
@done
	ret %cnt
}
