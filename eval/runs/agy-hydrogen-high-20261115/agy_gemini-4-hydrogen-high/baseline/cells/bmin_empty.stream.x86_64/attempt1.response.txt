bmin_empty:
    test rsi, rsi
    jle .Lempty
    cmp rsi, 8
    jl .Llt8
    movdqu xmm2, xmmword ptr [rdi]
    movdqu xmm3, xmmword ptr [rdi + 16]
    movdqu xmm4, xmmword ptr [rdi + 32]
    movdqu xmm5, xmmword ptr [rdi + 48]
    lea rcx, [rdi + rsi*8]
    lea rdx, [rcx - 64]
    add rdi, 64
    cmp rdi, rdx
    jae .Ltail8
.Lloop8:
    movdqu xmm6, xmmword ptr [rdi]
    movdqu xmm7, xmmword ptr [rdi + 16]
    movdqu xmm8, xmmword ptr [rdi + 32]
    movdqu xmm9, xmmword ptr [rdi + 48]
    movdqa xmm0, xmm2
    pcmpgtq xmm0, xmm6
    pblendvb xmm2, xmm6, xmm0
    movdqa xmm0, xmm3
    pcmpgtq xmm0, xmm7
    pblendvb xmm3, xmm7, xmm0
    movdqa xmm0, xmm4
    pcmpgtq xmm0, xmm8
    pblendvb xmm4, xmm8, xmm0
    movdqa xmm0, xmm5
    pcmpgtq xmm0, xmm9
    pblendvb xmm5, xmm9, xmm0
    add rdi, 64
    cmp rdi, rdx
    jb .Lloop8
.Ltail8:
    movdqu xmm6, xmmword ptr [rdx]
    movdqu xmm7, xmmword ptr [rdx + 16]
    movdqu xmm8, xmmword ptr [rdx + 32]
    movdqu xmm9, xmmword ptr [rdx + 48]
    movdqa xmm0, xmm2
    pcmpgtq xmm0, xmm6
    pblendvb xmm2, xmm6, xmm0
    movdqa xmm0, xmm3
    pcmpgtq xmm0, xmm7
    pblendvb xmm3, xmm7, xmm0
    movdqa xmm0, xmm4
    pcmpgtq xmm0, xmm8
    pblendvb xmm4, xmm8, xmm0
    movdqa xmm0, xmm5
    pcmpgtq xmm0, xmm9
    pblendvb xmm5, xmm9, xmm0
.Lreduce4:
    movdqa xmm0, xmm2
    pcmpgtq xmm0, xmm4
    pblendvb xmm2, xmm4, xmm0
    movdqa xmm0, xmm3
    pcmpgtq xmm0, xmm5
    pblendvb xmm3, xmm5, xmm0
    movdqa xmm0, xmm2
    pcmpgtq xmm0, xmm3
    pblendvb xmm2, xmm3, xmm0
    movq rax, xmm2
    pextrq rdx, xmm2, 1
    cmp rdx, rax
    cmovl rax, rdx
    ret
.Llt8:
    cmp rsi, 4
    jl .Lsmall
    movdqu xmm2, xmmword ptr [rdi]
    movdqu xmm3, xmmword ptr [rdi + 16]
    movdqu xmm4, xmmword ptr [rdi + rsi*8 - 32]
    movdqu xmm5, xmmword ptr [rdi + rsi*8 - 16]
    jmp .Lreduce4
.Lsmall:
    mov rax, qword ptr [rdi]
    cmp rsi, 1
    je .Ldone
    mov rdx, qword ptr [rdi + 8]
    mov rcx, qword ptr [rdi + rsi*8 - 8]
    cmp rdx, rax
    cmovl rax, rdx
    cmp rcx, rax
    cmovl rax, rcx
.Ldone:
    ret
.Lempty:
    mov rax, 0x7fffffffffffffff
    ret
