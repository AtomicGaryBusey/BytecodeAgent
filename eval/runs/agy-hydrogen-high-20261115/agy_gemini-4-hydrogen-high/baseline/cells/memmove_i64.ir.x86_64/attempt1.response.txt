export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w cslel %n, 0
	jnz %empty, @done, @check_dir
@check_dir
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_init, @bwd_init
@fwd_init
	%fn =l mul %n, 8
	jmp @fwd_loop
@fwd_loop
	%foff =l phi @fwd_init 0, @fwd_body %foff_next
	%fcond =w csltl %foff, %fn
	jnz %fcond, @fwd_body, @done
@fwd_body
	%fs =l add %src, %foff
	%fd =l add %dst, %foff
	%fval =l loadl %fs
	storel %fval, %fd
	%foff_next =l add %foff, 8
	jmp @fwd_loop
@bwd_init
	%bn =l mul %n, 8
	jmp @bwd_loop
@bwd_loop
	%boff =l phi @bwd_init %bn, @bwd_body %boff_prev
	%bcond =w csgtl %boff, 0
	jnz %bcond, @bwd_body, @done
@bwd_body
	%boff_prev =l sub %boff, 8
	%bs =l add %src, %boff_prev
	%bd =l add %dst, %boff_prev
	%bval =l loadl %bs
	storel %bval, %bd
	jmp @bwd_loop
@done
	ret 0
}
