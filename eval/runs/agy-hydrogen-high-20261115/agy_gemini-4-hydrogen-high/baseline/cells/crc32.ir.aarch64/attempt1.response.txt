export function l $crc32(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @byte_done %i1
	%ptr =l phi @start %p, @byte_done %ptr1
	%crc =w phi @start 4294967295, @byte_done %c2
	%cond =w csltl %i, %n
	jnz %cond, @byte_start, @done
@byte_start
	%b =w loadub %ptr
	%crc0 =w xor %crc, %b
@bit_loop
	%c =w phi @byte_start %crc0, @bit_loop %c2
	%k =w phi @byte_start 8, @bit_loop %k1
	%bit =w and %c, 1
	%mask =w sub 0, %bit
	%poly =w and %mask, 3988292384
	%shr =w shr %c, 1
	%c2 =w xor %shr, %poly
	%k1 =w sub %k, 1
	jnz %k1, @bit_loop, @byte_done
@byte_done
	%i1 =l add %i, 1
	%ptr1 =l add %ptr, 1
	jmp @loop
@done
	%res32 =w xor %crc, 4294967295
	%res =l extuw %res32
	ret %res
}
