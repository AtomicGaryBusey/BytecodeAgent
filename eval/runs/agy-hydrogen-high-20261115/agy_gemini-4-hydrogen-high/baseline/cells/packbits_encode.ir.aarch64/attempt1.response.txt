export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @repeat_done %i_rep, @lit_copy_loop %i_lit
    %opos =l phi @entry 0, @repeat_done %opos_rep, @lit_copy_loop %opos_lit
    %cond_n =w csltl %i, %n
    jnz %cond_n, @measure_run, @done

@measure_run
    %p0 =l add %src, %i
    %b0 =w loadub %p0
    jmp @run_loop

@run_loop
    %r =w phi @measure_run 1, @run_next %r_inc
    %r_lt_128 =w csltw %r, 128
    jnz %r_lt_128, @run_chk_n, @run_end

@run_chk_n
    %r_l =l extsw %r
    %idx =l add %i, %r_l
    %idx_lt_n =w csltl %idx, %n
    jnz %idx_lt_n, @run_chk_eq, @run_end

@run_chk_eq
    %p_idx =l add %src, %idx
    %b_idx =w loadub %p_idx
    %b_eq =w ceqw %b_idx, %b0
    jnz %b_eq, @run_next, @run_end

@run_next
    %r_inc =w add %r, 1
    jmp @run_loop

@run_end
    %is_rep =w csgew %r, 3
    jnz %is_rep, @do_repeat, @lit_scan

@do_repeat
    %opos_rep =l add %opos, 2
    %fit_rep =w cslel %opos_rep, %cap
    jnz %fit_rep, @repeat_done, @fail

@repeat_done
    %hdr_rep =w sub 257, %r
    %out_p0 =l add %out, %opos
    storeb %hdr_rep, %out_p0
    %out_p1 =l add %out_p0, 1
    storeb %b0, %out_p1
    %r_end_l =l extsw %r
    %i_rep =l add %i, %r_end_l
    jmp @loop

@lit_scan
    %L =w phi @run_end 1, @lit_next %L_inc
    %L_lt_128 =w csltw %L, 128
    jnz %L_lt_128, @lit_chk_n, @lit_emit_chk

@lit_chk_n
    %L_l =l extsw %L
    %j =l add %i, %L_l
    %j_lt_n =w csltl %j, %n
    jnz %j_lt_n, @lit_chk_run3, @lit_emit_chk

@lit_chk_run3
    %j_plus_2 =l add %j, 2
    %has_3 =w csltl %j_plus_2, %n
    jnz %has_3, @lit_cmp_bytes, @lit_next

@lit_cmp_bytes
    %pj0 =l add %src, %j
    %bj0 =w loadub %pj0
    %pj1 =l add %pj0, 1
    %bj1 =w loadub %pj1
    %eq01 =w ceqw %bj0, %bj1
    jnz %eq01, @lit_cmp_2, @lit_next

@lit_cmp_2
    %pj2 =l add %pj0, 2
    %bj2 =w loadub %pj2
    %eq02 =w ceqw %bj0, %bj2
    jnz %eq02, @lit_emit_chk, @lit_next

@lit_next
    %L_inc =w add %L, 1
    jmp @lit_scan

@lit_emit_chk
    %L_fin_l =l extsw %L
    %opos_1 =l add %opos, 1
    %opos_lit =l add %opos_1, %L_fin_l
    %fit_lit =w cslel %opos_lit, %cap
    jnz %fit_lit, @lit_write_hdr, @fail

@lit_write_hdr
    %hdr_lit =w sub %L, 1
    %out_lhdr =l add %out, %opos
    storeb %hdr_lit, %out_lhdr
    %i_lit =l add %i, %L_fin_l
    jmp @lit_copy_loop

@lit_copy_loop
    %k =l phi @lit_write_hdr 0, @lit_copy_step %k_inc
    %k_lt_L =w csltl %k, %L_fin_l
    jnz %k_lt_L, @lit_copy_step, @loop

@lit_copy_step
    %src_k_addr =l add %i, %k
    %src_k_ptr =l add %src, %src_k_addr
    %byte_k =w loadub %src_k_ptr
    %out_k_addr =l add %opos_1, %k
    %out_k_ptr =l add %out, %out_k_addr
    storeb %byte_k, %out_k_ptr
    %k_inc =l add %k, 1
    jmp @lit_copy_loop

@fail
    ret -1

@done
    ret %opos
}
