export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @outer_cond

@outer_cond
    %i =l phi @start 0, @outer_next %i_next
    %j =l phi @start 0, @outer_next %j_next
    %more =w csltl %i, %n
    jnz %more, @outer_body, @done

@outer_body
    %rem =l sub %n, %i
    %is_ge3 =w csgel %rem, 3
    jnz %is_ge3, @case3, @case_lt3

@case3
    %p3_0 =l add %src, %i
    %p3_1 =l add %p3_0, 1
    %p3_2 =l add %p3_0, 2
    %b3_0 =w loadub %p3_0
    %b3_1 =w loadub %p3_1
    %b3_2 =w loadub %p3_2
    %s3_0 =w shl %b3_0, 16
    %s3_1 =w shl %b3_1, 8
    %t3_0 =w or %s3_0, %s3_1
    %val3 =w or %t3_0, %b3_2
    %i_next3 =l add %i, 3
    jmp @emit_init

@case_lt3
    %is_2 =w ceql %rem, 2
    jnz %is_2, @case2, @case1

@case2
    %p2_0 =l add %src, %i
    %p2_1 =l add %p2_0, 1
    %b2_0 =w loadub %p2_0
    %b2_1 =w loadub %p2_1
    %s2_0 =w shl %b2_0, 16
    %s2_1 =w shl %b2_1, 8
    %val2 =w or %s2_0, %s2_1
    %i_next2 =l add %i, 2
    jmp @emit_init

@case1
    %p1_0 =l add %src, %i
    %b1_0 =w loadub %p1_0
    %val1 =w shl %b1_0, 16
    %i_next1 =l add %i, 1
    jmp @emit_init

@emit_init
    %val =w phi @case3 %val3, @case2 %val2, @case1 %val1
    %k =w phi @case3 4, @case2 3, @case1 2
    %i_next =l phi @case3 %i_next3, @case2 %i_next2, @case1 %i_next1
    jmp @inner_cond

@inner_cond
    %slot =w phi @emit_init 0, @inner_store %slot_next
    %j_cur =l phi @emit_init %j, @inner_store %j_step
    %cont =w csltw %slot, 4
    jnz %cont, @inner_body, @outer_next

@inner_body
    %is_enc =w csltw %slot, %k
    jnz %is_enc, @inner_encode, @inner_pad

@inner_encode
    %sh1 =w sub 3, %slot
    %sh =w mul %sh1, 6
    %shifted =w shr %val, %sh
    %v =w and %shifted, 63
    %c0 =w add %v, 65
    %gt25 =w csgtw %v, 25
    %d1 =w mul %gt25, 6
    %c1 =w add %c0, %d1
    %gt51 =w csgtw %v, 51
    %d2 =w mul %gt51, 75
    %c2 =w sub %c1, %d2
    %gt61 =w csgtw %v, 61
    %d3 =w mul %gt61, 15
    %c3 =w sub %c2, %d3
    %gt62 =w csgtw %v, 62
    %d4 =w mul %gt62, 3
    %ch_enc =w add %c3, %d4
    jmp @inner_store

@inner_pad
    jmp @inner_store

@inner_store
    %ch =w phi @inner_encode %ch_enc, @inner_pad 61
    %out_ptr =l add %out, %j_cur
    storeb %ch, %out_ptr
    %slot_next =w add %slot, 1
    %j_step =l add %j_cur, 1
    jmp @inner_cond

@outer_next
    %j_next =l copy %j_cur
    jmp @outer_cond

@done
    ret %j
}
