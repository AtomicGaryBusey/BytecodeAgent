export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zero_loop

@zero_loop
	%zi =l phi @start 0, @zero_body %zi_next
	%zcond =w cslel %zi, %n
	jnz %zcond, @zero_body, @outer_loop

@zero_body
	%zoff =l mul %zi, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%zi_next =l add %zi, 1
	jmp @zero_loop

@outer_loop
	%i =l phi @zero_loop 0, @outer_next %i_next
	%ocond =w csltl %i, %m
	jnz %ocond, @outer_body, @done

@outer_body
	%px =l add %x, %i
	%xi =w loadub %px
	jmp @inner_loop

@inner_loop
	%j =l phi @outer_body 1, @inner_latch %j_next
	%diag =l phi @outer_body 0, @inner_latch %up
	%icond =w cslel %j, %n
	jnz %icond, @inner_body, @outer_next

@inner_body
	%joff =l mul %j, 8
	%pj =l add %row, %joff
	%up =l loadl %pj
	%jm1 =l sub %j, 1
	%py =l add %y, %jm1
	%yj =w loadub %py
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch

@match
	%val_m =l add %diag, 1
	jmp @inner_latch

@nomatch
	%jm1off =l mul %jm1, 8
	%pjm1 =l add %row, %jm1off
	%left =l loadl %pjm1
	%ge =w csgel %up, %left
	jnz %ge, @inner_latch, @take_left

@take_left
	jmp @inner_latch

@inner_latch
	%val =l phi @match %val_m, @nomatch %up, @take_left %left
	storel %val, %pj
	%j_next =l add %j, 1
	jmp @inner_loop

@outer_next
	%i_next =l add %i, 1
	jmp @outer_loop

@done
	%noff =l mul %n, 8
	%pn =l add %row, %noff
	%res =l loadl %pn
	ret %res
}
