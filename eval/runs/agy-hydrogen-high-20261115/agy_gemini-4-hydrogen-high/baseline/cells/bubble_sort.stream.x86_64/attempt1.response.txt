bubble_sort:
    cmp rsi, 1
    jle .Ldone
    dec rsi
.Louter:
    xor ecx, ecx
    xor edx, edx
.Linner:
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], r8
    mov edx, 1
.Lnoswap:
    inc rcx
    cmp rcx, rsi
    jl .Linner
    test edx, edx
    jz .Ldone
    dec rsi
    jnz .Louter
.Ldone:
    xor eax, eax
    ret
