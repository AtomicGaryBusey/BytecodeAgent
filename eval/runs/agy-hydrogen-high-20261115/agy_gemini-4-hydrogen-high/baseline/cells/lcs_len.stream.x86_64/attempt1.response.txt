lcs_len:
    test rcx, rcx
    jle .Ln_zero
    test rsi, rsi
    jg .Lm_pos
    mov rdx, -1
    mov rsi, -1
    jmp .Lreconstruct
.Ln_zero:
    mov qword ptr [r8], 0
    xor eax, eax
    ret
.Lm_pos:
    push rbx
    sub rsp, 128
    pxor xmm9, xmm9
    movdqa [rsp], xmm9
    movdqa [rsp + 16], xmm9
    movdqa [rsp + 32], xmm9
    movdqa [rsp + 48], xmm9
    cmp rcx, 64
    jle .Lzero_done
    movdqa [rsp + 64], xmm9
    movdqa [rsp + 80], xmm9
    movdqa [rsp + 96], xmm9
    movdqa [rsp + 112], xmm9
.Lzero_done:
    mov r9, rsp
    mov r10, rcx
    cmp r10, 16
    jl .Lcopy_tail
.Lcopy16:
    movdqu xmm0, [rdx]
    movdqa [r9], xmm0
    add rdx, 16
    add r9, 16
    sub r10, 16
    cmp r10, 16
    jge .Lcopy16
.Lcopy_tail:
    test r10, 8
    jz .Lcopy_no8
    mov rax, qword ptr [rdx]
    mov qword ptr [r9], rax
    add rdx, 8
    add r9, 8
.Lcopy_no8:
    test r10, 4
    jz .Lcopy_no4
    mov eax, dword ptr [rdx]
    mov dword ptr [r9], eax
    add rdx, 4
    add r9, 4
.Lcopy_no4:
    test r10, 2
    jz .Lcopy_no2
    movzx eax, word ptr [rdx]
    mov word ptr [r9], ax
    add rdx, 2
    add r9, 2
.Lcopy_no2:
    test r10, 1
    jz .Lcopy_done
    mov al, byte ptr [rdx]
    mov byte ptr [r9], al
.Lcopy_done:
    movdqa xmm0, [rsp]
    lea r9, [rdi + rsi]
    mov rdx, -1
    cmp rcx, 16
    jg .Ln_gt16
    add rsp, 128
.Lloop16:
    movzx eax, byte ptr [rdi]
    movd xmm8, eax
    pshufb xmm8, xmm9
    movdqa xmm10, xmm0
    pcmpeqb xmm10, xmm8
    pmovmskb eax, xmm10
    mov r10, rdx
    and r10, rax
    andn rax, rax, rdx
    add rdx, r10
    or rdx, rax
    inc rdi
    cmp rdi, r9
    jne .Lloop16
    pop rbx
    jmp .Lreconstruct
.Ln_gt16:
    movdqa xmm1, [rsp + 16]
    cmp rcx, 32
    jg .Ln_gt32
    add rsp, 128
.Lloop32:
    movzx eax, byte ptr [rdi]
    movd xmm8, eax
    pshufb xmm8, xmm9
    movdqa xmm10, xmm1
    pcmpeqb xmm10, xmm8
    pmovmskb r10d, xmm10
    shl r10, 16
    movdqa xmm10, xmm0
    pcmpeqb xmm10, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    mov rax, rdx
    and rax, r10
    andn r10, r10, rdx
    add rdx, rax
    or rdx, r10
    inc rdi
    cmp rdi, r9
    jne .Lloop32
    pop rbx
    jmp .Lreconstruct
.Ln_gt32:
    movdqa xmm2, [rsp + 32]
    movdqa xmm3, [rsp + 48]
    cmp rcx, 64
    jg .Ln_gt64
    add rsp, 128
.Lloop64:
    movzx eax, byte ptr [rdi]
    movd xmm8, eax
    pshufb xmm8, xmm9
    movdqa xmm10, xmm3
    pcmpeqb xmm10, xmm8
    pmovmskb r10d, xmm10
    shl r10, 16
    movdqa xmm10, xmm2
    pcmpeqb xmm10, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    shl r10, 16
    movdqa xmm10, xmm1
    pcmpeqb xmm10, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    shl r10, 16
    movdqa xmm10, xmm0
    pcmpeqb xmm10, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    mov rax, rdx
    and rax, r10
    andn r10, r10, rdx
    add rdx, rax
    or rdx, r10
    inc rdi
    cmp rdi, r9
    jne .Lloop64
    pop rbx
    jmp .Lreconstruct
.Ln_gt64:
    movdqa xmm4, [rsp + 64]
    movdqa xmm5, [rsp + 80]
    movdqa xmm6, [rsp + 96]
    movdqa xmm7, [rsp + 112]
    add rsp, 128
    mov rsi, -1
.Lloop128:
    movzx eax, byte ptr [rdi]
    movd xmm8, eax
    pshufb xmm8, xmm9
    movdqa xmm10, xmm3
    movdqa xmm11, xmm7
    pcmpeqb xmm10, xmm8
    pcmpeqb xmm11, xmm8
    pmovmskb r10d, xmm10
    pmovmskb r11d, xmm11
    shl r10, 16
    shl r11, 16
    movdqa xmm10, xmm2
    movdqa xmm11, xmm6
    pcmpeqb xmm10, xmm8
    pcmpeqb xmm11, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    pmovmskb eax, xmm11
    or r11, rax
    shl r10, 16
    shl r11, 16
    movdqa xmm10, xmm1
    movdqa xmm11, xmm5
    pcmpeqb xmm10, xmm8
    pcmpeqb xmm11, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    pmovmskb eax, xmm11
    or r11, rax
    shl r10, 16
    shl r11, 16
    movdqa xmm10, xmm0
    movdqa xmm11, xmm4
    pcmpeqb xmm10, xmm8
    pcmpeqb xmm11, xmm8
    pmovmskb eax, xmm10
    or r10, rax
    pmovmskb eax, xmm11
    or r11, rax
    mov rax, rdx
    and rax, r10
    andn r10, r10, rdx
    mov rbx, rsi
    and rbx, r11
    andn r11, r11, rsi
    add rdx, rax
    adc rsi, rbx
    or rdx, r10
    or rsi, r11
    inc rdi
    cmp rdi, r9
    jne .Lloop128
    pop rbx
.Lreconstruct:
    not rdx
    xor eax, eax
    mov qword ptr [r8], rax
    mov r9d, 64
    cmp rcx, r9
    cmovle r9, rcx
    mov r10d, 1
.Lrec1:
    shr rdx, 1
    adc rax, 0
    mov qword ptr [r8 + r10*8], rax
    inc r10
    cmp r10, r9
    jle .Lrec1
    cmp rcx, 64
    jle .Ldone
    not rsi
.Lrec2:
    shr rsi, 1
    adc rax, 0
    mov qword ptr [r8 + r10*8], rax
    inc r10
    cmp r10, rcx
    jle .Lrec2
.Ldone:
    ret
