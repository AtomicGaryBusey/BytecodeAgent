.Lsse_loop:
    movdqu xmm0, xmmword ptr [rdi]
    pshufb xmm0, xmm8
    movdqa xmm1, xmm0
    pand xmm0, xmm9
    pmulhuw xmm0, xmm10
    pand xmm1, xmm11
    pmullw xmm1, xmm12
    por xmm0, xmm1

    movdqa xmm1, xmm0
    psubusb xmm1, xmm13
    movdqa xmm2, xmm14
    pcmpgtb xmm2, xmm0
    pand xmm2, xmm15
    paddb xmm1, xmm2
    movdqa xmm3, xmm7
    pshufb xmm3, xmm1
    paddb xmm0, xmm3

    movdqu xmmword ptr [rdx], xmm0
    add rdi, 12
    add rdx, 16
    sub rsi, 12
    cmp rsi, 16
    jae .Lsse_loop

.Ltail_loop:
    test rsi, rsi
    jz .Ldone
    cmp rsi, 3
    jb .Lpartial

    movzx r10d, byte ptr [rdi]
    shl r10d, 24
    movzx eax, byte ptr [rdi + 1]
    shl eax, 16
    or r10d, eax
    movzx eax, byte ptr [rdi + 2]
    shl eax, 8
    or r10d, eax
    mov r9d, 4
    add rdi, 3
    sub rsi, 3
    jmp .Lemit_group

.Lpartial:
    movzx r10d, byte ptr [rdi]
    shl r10d, 24
    mov r9d, 2
    cmp rsi, 1
    je .Lpartial_set
    movzx eax, byte ptr [rdi + 1]
    shl eax, 16
    or r10d, eax
    mov r9d, 3
.Lpartial_set:
    xor esi, esi

.Lemit_group:
    mov ecx, 4
    sub ecx, r9d

.Lchar_loop:
    rol r10d, 6
    mov eax, r10d
    and eax, 63
    lea r8d, [rax + 65]
    cmp eax, 26
    jb .Lchar_store
    lea r8d, [rax + 71]
    cmp eax, 52
    jb .Lchar_store
    lea r8d, [rax - 4]
    cmp eax, 62
    jb .Lchar_store
    mov r8d, 43
    je .Lchar_store
    mov r8d, 47
.Lchar_store:
    mov byte ptr [rdx], r8b
    inc rdx
    dec r9d
    jnz .Lchar_loop

    test ecx, ecx
    jz .Ltail_loop
.Lpad_loop:
    mov byte ptr [rdx], 0x3d
    inc rdx
    dec ecx
    jnz .Lpad_loop

.Ldone:
    mov rax, rdx
    sub rax, r11
    ret
