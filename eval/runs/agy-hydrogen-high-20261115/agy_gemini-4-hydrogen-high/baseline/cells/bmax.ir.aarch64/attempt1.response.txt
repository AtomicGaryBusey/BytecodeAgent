export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l sub %n, 1
	%cond0 =w csgtl %i0, 0
	jnz %cond0, @loop, @exit0
@exit0
	ret %max0
@loop
	%ptr =l phi @start %a, @next %ptr1
	%rem =l phi @start %i0, @next %rem1
	%max =l phi @start %max0, @next %max1
	%ptr1 =l add %ptr, 8
	%val =l loadl %ptr1
	%gt =w csgtl %val, %max
	jnz %gt, @take, @skip
@take
	jmp @next
@skip
	jmp @next
@next
	%max1 =l phi @take %val, @skip %max
	%rem1 =l sub %rem, 1
	%more =w csgtl %rem1, 0
	jnz %more, @loop, @exit
@exit
	ret %max1
}
