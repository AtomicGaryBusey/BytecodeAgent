export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_cond
@init_cond
	%j0 =l phi @start 0, @init_body %j0_next
	%c0 =w cslel %j0, %n
	jnz %c0, @init_body, @outer_cond
@init_body
	%off0 =l mul %j0, 8
	%p0 =l add %row, %off0
	storel 0, %p0
	%j0_next =l add %j0, 1
	jmp @init_cond
@outer_cond
	%i =l phi @init_cond 0, @outer_next %i_next
	%ci =w csltl %i, %m
	jnz %ci, @outer_body, @done
@outer_body
	%xp =l add %x, %i
	%xi =w loadub %xp
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body 1, @inner_store %j_next
	%diag =l phi @outer_body 0, @inner_store %up
	%left =l phi @outer_body 0, @inner_store %val
	%cj =w cslel %j, %n
	jnz %cj, @inner_body, @outer_next
@inner_body
	%off =l mul %j, 8
	%rowp =l add %row, %off
	%up =l loadl %rowp
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%val_match =l add %diag, 1
	jmp @inner_store
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @take_up, @take_left
@take_up
	jmp @inner_store
@take_left
	jmp @inner_store
@inner_store
	%val =l phi @match %val_match, @take_up %up, @take_left %left
	storel %val, %rowp
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_next
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	%noff =l mul %n, 8
	%res_p =l add %row, %noff
	%res =l loadl %res_p
	ret %res
}
