crc32:
    mvn w2, wzr
    cmp x1, #0
    b.le .Ldone
    mov w3, #0x8320
    movk w3, #0xedb8, lsl #16
.Lbyte_loop:
    ldrb w4, [x0], #1
    eor w2, w2, w4
    mov w6, #8
.Lbit_loop:
    lsr w4, w2, #1
    eor w5, w4, w3
    tst w2, #1
    csel w2, w5, w4, ne
    subs w6, w6, #1
    b.ne .Lbit_loop
    subs x1, x1, #1
    b.ne .Lbyte_loop
.Ldone:
    mvn w0, w2
    ret
