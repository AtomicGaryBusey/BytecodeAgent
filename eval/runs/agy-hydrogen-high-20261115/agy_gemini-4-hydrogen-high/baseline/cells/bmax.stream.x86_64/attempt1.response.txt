bmax:
    mov rax, qword ptr [rdi]
    cmp rsi, 4
    jl .Ltail_init
    mov r8, qword ptr [rdi + 8]
    mov r9, qword ptr [rdi + 16]
    mov r10, qword ptr [rdi + 24]
    mov rcx, 4
    lea r11, [rsi - 3]
    cmp rcx, r11
    jge .Lreduce
.Lloop4:
    mov rdx, qword ptr [rdi + rcx*8]
    cmp rdx, rax
    cmovg rax, rdx
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rdx, r8
    cmovg r8, rdx
    mov rdx, qword ptr [rdi + rcx*8 + 16]
    cmp rdx, r9
    cmovg r9, rdx
    mov rdx, qword ptr [rdi + rcx*8 + 24]
    cmp rdx, r10
    cmovg r10, rdx
    add rcx, 4
    cmp rcx, r11
    jl .Lloop4
.Lreduce:
    cmp r8, rax
    cmovg rax, r8
    cmp r9, rax
    cmovg rax, r9
    cmp r10, rax
    cmovg rax, r10
    jmp .Ltail_cmp
.Ltail_init:
    mov rcx, 1
.Ltail_cmp:
    cmp rcx, rsi
    jge .Ldone
.Ltail:
    mov rdx, qword ptr [rdi + rcx*8]
    cmp rdx, rax
    cmovg rax, rdx
    inc rcx
    cmp rcx, rsi
    jl .Ltail
.Ldone:
    ret
