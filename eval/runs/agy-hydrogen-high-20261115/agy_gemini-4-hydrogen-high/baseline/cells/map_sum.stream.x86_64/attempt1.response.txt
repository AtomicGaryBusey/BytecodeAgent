map_sum:
    test rsi, rsi
    jle .Lzero
    push rbx
    push rbp
    push r12
    push r13
    sub rsp, 8
    mov rbx, rdi
    lea rbp, [rdi + rsi*8]
    mov r12, rdx
    xor r13d, r13d
.Lloop:
    mov rdi, qword ptr [rbx]
    call r12
    add r13, rax
    add rbx, 8
    cmp rbx, rbp
    jne .Lloop
    mov rax, r13
    add rsp, 8
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lzero:
    xor eax, eax
    ret
