export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc_p =l alloc8 8
	%d_p =l alloc8 8
	%i_p =l alloc8 8
	storel 0, %pc_p
	storel 0, %d_p
	jmp @loop
@loop
	%pc =l loadl %pc_p
	%in_bounds =w csltl %pc, %len
	jnz %in_bounds, @fetch, @halt
@fetch
	%pc_off =l shl %pc, 3
	%code_addr =l add %code, %pc_off
	%op =l loadl %code_addr
	%is0 =w ceql %op, 0
	jnz %is0, @halt, @chk1
@chk1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @chk2
@chk2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @chk3
@chk3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @chk4
@chk4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @chk5
@chk5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @chk6
@chk6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @chk7
@chk7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @chk8
@chk8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @chk9
@chk9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @chk10
@chk10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @chk11
@chk11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_nop
@op_nop
	%npc_nop =l add %pc, 1
	storel %npc_nop, %pc_p
	jmp @loop
@op_push
	%p1_addr =l add %code_addr, 8
	%val =l loadl %p1_addr
	%d1 =l loadl %d_p
	%s1_off =l shl %d1, 3
	%s1_addr =l add %stack, %s1_off
	storel %val, %s1_addr
	%nd1 =l add %d1, 1
	storel %nd1, %d_p
	%npc1 =l add %pc, 2
	storel %npc1, %pc_p
	jmp @loop
@op_add
	%d2 =l loadl %d_p
	%idx2_b =l sub %d2, 1
	%off2_b =l shl %idx2_b, 3
	%addr2_b =l add %stack, %off2_b
	%b2 =l loadl %addr2_b
	%addr2_a =l sub %addr2_b, 8
	%a2 =l loadl %addr2_a
	%r2 =l add %a2, %b2
	storel %r2, %addr2_a
	storel %idx2_b, %d_p
	%npc2 =l add %pc, 1
	storel %npc2, %pc_p
	jmp @loop
@op_sub
	%d3 =l loadl %d_p
	%idx3_b =l sub %d3, 1
	%off3_b =l shl %idx3_b, 3
	%addr3_b =l add %stack, %off3_b
	%b3 =l loadl %addr3_b
	%addr3_a =l sub %addr3_b, 8
	%a3 =l loadl %addr3_a
	%r3 =l sub %a3, %b3
	storel %r3, %addr3_a
	storel %idx3_b, %d_p
	%npc3 =l add %pc, 1
	storel %npc3, %pc_p
	jmp @loop
@op_mul
	%d4 =l loadl %d_p
	%idx4_b =l sub %d4, 1
	%off4_b =l shl %idx4_b, 3
	%addr4_b =l add %stack, %off4_b
	%b4 =l loadl %addr4_b
	%addr4_a =l sub %addr4_b, 8
	%a4 =l loadl %addr4_a
	%r4 =l mul %a4, %b4
	storel %r4, %addr4_a
	storel %idx4_b, %d_p
	%npc4 =l add %pc, 1
	storel %npc4, %pc_p
	jmp @loop
@op_dup
	%d5 =l loadl %d_p
	%off5 =l shl %d5, 3
	%dst5 =l add %stack, %off5
	%src5 =l sub %dst5, 8
	%v5 =l loadl %src5
	storel %v5, %dst5
	%nd5 =l add %d5, 1
	storel %nd5, %d_p
	%npc5 =l add %pc, 1
	storel %npc5, %pc_p
	jmp @loop
@op_swap
	%d6 =l loadl %d_p
	%idx6_b =l sub %d6, 1
	%off6_b =l shl %idx6_b, 3
	%addr6_b =l add %stack, %off6_b
	%b6 =l loadl %addr6_b
	%addr6_a =l sub %addr6_b, 8
	%a6 =l loadl %addr6_a
	storel %b6, %addr6_a
	storel %a6, %addr6_b
	%npc6 =l add %pc, 1
	storel %npc6, %pc_p
	jmp @loop
@op_drop
	%d7 =l loadl %d_p
	%nd7 =l sub %d7, 1
	storel %nd7, %d_p
	%npc7 =l add %pc, 1
	storel %npc7, %pc_p
	jmp @loop
@op_over
	%d8 =l loadl %d_p
	%off8 =l shl %d8, 3
	%dst8 =l add %stack, %off8
	%src8 =l sub %dst8, 16
	%v8 =l loadl %src8
	storel %v8, %dst8
	%nd8 =l add %d8, 1
	storel %nd8, %d_p
	%npc8 =l add %pc, 1
	storel %npc8, %pc_p
	jmp @loop
@op_lt
	%d9 =l loadl %d_p
	%idx9_b =l sub %d9, 1
	%off9_b =l shl %idx9_b, 3
	%addr9_b =l add %stack, %off9_b
	%b9 =l loadl %addr9_b
	%addr9_a =l sub %addr9_b, 8
	%a9 =l loadl %addr9_a
	%cmp9 =w csltl %a9, %b9
	%r9 =l extuw %cmp9
	storel %r9, %addr9_a
	storel %idx9_b, %d_p
	%npc9 =l add %pc, 1
	storel %npc9, %pc_p
	jmp @loop
@op_jz
	%d10 =l loadl %d_p
	%nd10 =l sub %d10, 1
	storel %nd10, %d_p
	%off10 =l shl %nd10, 3
	%addr10 =l add %stack, %off10
	%v10 =l loadl %addr10
	%z10 =w ceql %v10, 0
	jnz %z10, @op_jz_taken, @op_jz_not
@op_jz_taken
	%t10_addr =l add %code_addr, 8
	%t10 =l loadl %t10_addr
	storel %t10, %pc_p
	jmp @loop
@op_jz_not
	%npc10 =l add %pc, 2
	storel %npc10, %pc_p
	jmp @loop
@op_jmp
	%t11_addr =l add %code_addr, 8
	%t11 =l loadl %t11_addr
	storel %t11, %pc_p
	jmp @loop
@halt
	%hd =l loadl %d_p
	storel %hd, %i_p
	jmp @zloop
@zloop
	%zi =l loadl %i_p
	%zcond =w csltl %zi, %cap
	jnz %zcond, @zbody, @done
@zbody
	%zoff =l shl %zi, 3
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%nzi =l add %zi, 1
	storel %nzi, %i_p
	jmp @zloop
@done
	%fd =l loadl %d_p
	%has_top =w csgtl %fd, 0
	jnz %has_top, @ret_top, @ret_zero
@ret_top
	%top_idx =l sub %fd, 1
	%top_off =l shl %top_idx, 3
	%top_addr =l add %stack, %top_off
	%res =l loadl %top_addr
	ret %res
@ret_zero
	ret 0
}
