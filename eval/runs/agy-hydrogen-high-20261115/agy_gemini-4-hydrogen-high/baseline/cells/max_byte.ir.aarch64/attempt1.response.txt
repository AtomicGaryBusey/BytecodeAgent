export function l $max_byte(l %p, l %n) {
@start
	%has =w csgtl %n, 0
	jnz %has, @loop, @empty
@empty
	ret 0
@loop
	%i =l phi @start 0, @next %i1
	%max =l phi @start 0, @next %max1
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%gt =w csgtl %b, %max
	jnz %gt, @take_b, @keep_max
@take_b
	jmp @next
@keep_max
	jmp @next
@next
	%max1 =l phi @take_b %b, @keep_max %max
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @done
@done
	ret %max1
}
