count_ugt:
    xor eax, eax
    test rsi, rsi
    jz .Ldone
    cmp rsi, 4
    jb .Ltail_loop
    pcmpeqd xmm5, xmm5
    psllq xmm5, 63
    movq xmm4, rdx
    punpcklqdq xmm4, xmm4
    pxor xmm4, xmm5
    pxor xmm0, xmm0
    pxor xmm1, xmm1
    sub rsi, 4
.Lloop4:
    movdqu xmm2, xmmword ptr [rdi]
    movdqu xmm3, xmmword ptr [rdi + 16]
    pxor xmm2, xmm5
    pxor xmm3, xmm5
    pcmpgtq xmm2, xmm4
    pcmpgtq xmm3, xmm4
    psubq xmm0, xmm2
    psubq xmm1, xmm3
    add rdi, 32
    sub rsi, 4
    jae .Lloop4
    add rsi, 2
    js .Lno_pair
    movdqu xmm2, xmmword ptr [rdi]
    pxor xmm2, xmm5
    pcmpgtq xmm2, xmm4
    psubq xmm0, xmm2
    add rdi, 16
.Lno_pair:
    paddq xmm0, xmm1
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq rax, xmm0
    test esi, 1
    jz .Ldone
    cmp rdx, qword ptr [rdi]
    adc rax, 0
    ret
.Ltail_loop:
    cmp rdx, qword ptr [rdi]
    adc rax, 0
    add rdi, 8
    dec rsi
    jnz .Ltail_loop
.Ldone:
    ret
