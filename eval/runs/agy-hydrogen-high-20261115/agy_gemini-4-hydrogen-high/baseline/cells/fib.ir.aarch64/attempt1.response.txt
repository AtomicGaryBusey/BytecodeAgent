export function l $fib(l %n) {
@start
	%cond0 =w csgtl %n, 0
	jnz %cond0, @loop, @done_zero
@done_zero
	ret 0
@loop
	%i =l phi @start 0, @loop %i_next
	%a =l phi @start 0, @loop %b
	%b =l phi @start 1, @loop %sum
	%sum =l add %a, %b
	%i_next =l add %i, 1
	%cond =w csltl %i_next, %n
	jnz %cond, @loop, @done
@done
	ret %b
}
