export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%p =l phi @start %a, @next %p1
	%rem =l phi @start %n, @next %rem1
	%min =l phi @start 9223372036854775807, @next %min1
	%done =w cslel %rem, 0
	jnz %done, @exit, @body
@body
	%val =l loadl %p
	%lt =w csltl %val, %min
	jnz %lt, @take, @next
@take
	jmp @next
@next
	%min1 =l phi @body %min, @take %val
	%p1 =l add %p, 8
	%rem1 =l sub %rem, 1
	jmp @loop
@exit
	ret %min
}
