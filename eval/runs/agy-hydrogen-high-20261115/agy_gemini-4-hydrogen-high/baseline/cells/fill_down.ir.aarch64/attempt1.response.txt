export function l $fill_down(l %a, l %n, l %start) {
@start
	%c0 =w csltl %n, 1
	jnz %c0, @done, @loop
@loop
	%i =l phi @start 0, @loop %i1
	%p =l phi @start %a, @loop %p1
	%v =l phi @start %start, @loop %v1
	storel %v, %p
	%i1 =l add %i, 1
	%p1 =l add %p, 8
	%v1 =l sub %v, 1
	%cond =w csltl %i1, %n
	jnz %cond, @loop, @done
@done
	ret %n
}
