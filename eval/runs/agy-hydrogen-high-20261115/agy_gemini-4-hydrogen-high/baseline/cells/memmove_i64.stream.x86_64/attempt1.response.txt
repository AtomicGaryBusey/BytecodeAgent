memmove_i64:
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
.Lbackward_loop:
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov qword ptr [rdi + rdx*8], rax
    jnz .Lbackward_loop
    jmp .Ldone
.Lforward:
    xor ecx, ecx
.Lforward_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jb .Lforward_loop
.Ldone:
    xor eax, eax
    ret
