export function l $qsort_pairs(l %a, l %n) {
@start
    jmp @loop
@loop
    %base =l phi @start %a, @rec_right %base, @rec_left %right_ptr
    %cnt =l phi @start %n, @rec_right %len_L, @rec_left %len_R
    %le1 =w cslel %cnt, 1
    jnz %le1, @ret_zero, @part_init
@ret_zero
    ret 0
@part_init
    %cnt_m1 =l sub %cnt, 1
    %mid =l shr %cnt_m1, 1
    %mid_off =l shl %mid, 4
    %mid_p =l add %base, %mid_off
    %pk =l loadl %mid_p
    %mid_p8 =l add %mid_p, 8
    %pv =l loadl %mid_p8
    %end_off =l shl %cnt, 4
    %neg16 =l sub 0, 16
    jmp @outer_part
@outer_part
    %oi_start =l phi @part_init %neg16, @do_swap %oi
    %oj_start =l phi @part_init %end_off, @do_swap %oj
    jmp @inc_i
@inc_i
    %oi_prev =l phi @outer_part %oi_start, @inc_i_next %oi
    %oi =l add %oi_prev, 16
    %pi =l add %base, %oi
    %ki =l loadl %pi
    %ci_klt =w csltl %ki, %pk
    jnz %ci_klt, @inc_i_next, @chk_i_eq
@chk_i_eq
    %ci_keq =w ceql %ki, %pk
    jnz %ci_keq, @chk_i_v, @dec_j
@chk_i_v
    %pi8 =l add %pi, 8
    %vi =l loadl %pi8
    %ci_vlt =w csltl %vi, %pv
    jnz %ci_vlt, @inc_i_next, @dec_j
@inc_i_next
    jmp @inc_i
@dec_j
    %oj_prev =l phi @chk_i_eq %oj_start, @chk_i_v %oj_start, @dec_j_next %oj
    %oj =l sub %oj_prev, 16
    %pj =l add %base, %oj
    %kj =l loadl %pj
    %cj_kgt =w csgtl %kj, %pk
    jnz %cj_kgt, @dec_j_next, @chk_j_eq
@chk_j_eq
    %cj_keq =w ceql %kj, %pk
    jnz %cj_keq, @chk_j_v, @part_cmp
@chk_j_v
    %pj8 =l add %pj, 8
    %vj =l loadl %pj8
    %cj_vgt =w csgtl %vj, %pv
    jnz %cj_vgt, @dec_j_next, @part_cmp
@dec_j_next
    jmp @dec_j
@part_cmp
    %done_part =w csgel %oi, %oj
    jnz %done_part, @part_done, @do_swap
@do_swap
    %pi8_s =l add %pi, 8
    %vi_s =l loadl %pi8_s
    %pj8_s =l add %pj, 8
    %vj_s =l loadl %pj8_s
    storel %kj, %pi
    storel %vj_s, %pi8_s
    storel %ki, %pj
    storel %vi_s, %pj8_s
    jmp @outer_part
@part_done
    %j =l shr %oj, 4
    %len_L =l add %j, 1
    %right_off =l add %oj, 16
    %right_ptr =l add %base, %right_off
    %len_R =l sub %cnt, %len_L
    %L_lt_R =w csltl %len_L, %len_R
    jnz %L_lt_R, @rec_left, @rec_right
@rec_left
    %r1 =l call $qsort_pairs(l %base, l %len_L)
    jmp @loop
@rec_right
    %r2 =l call $qsort_pairs(l %right_ptr, l %len_R)
    jmp @loop
}
