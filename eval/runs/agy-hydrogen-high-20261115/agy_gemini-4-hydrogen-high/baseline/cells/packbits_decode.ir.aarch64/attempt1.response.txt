export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @noop %i1, @lit_done %i_lit, @rep_done %i_rep
	%o =l phi @start 0, @noop %o, @lit_done %o_lit, @rep_done %o_rep
	%has_src =w csltl %i, %n
	jnz %has_src, @read_hdr, @done
@done
	ret %o
@read_hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	%is_noop =w ceqw %h, -128
	jnz %is_noop, @noop, @not_noop
@noop
	jmp @loop
@not_noop
	%is_lit =w csgew %h, 0
	jnz %is_lit, @lit_check, @rep_check
@lit_check
	%hl =l extsw %h
	%len =l add %hl, 1
	%i_end =l add %i1, %len
	%bad_src =w csgtl %i_end, %n
	jnz %bad_src, @err_src, @lit_loop
@lit_loop
	%i_lit =l phi @lit_check %i1, @lit_write %i_lit_next
	%o_lit =l phi @lit_check %o, @lit_write %o_lit_next
	%more_lit =w csltl %i_lit, %i_end
	jnz %more_lit, @lit_cap_chk, @lit_done
@lit_cap_chk
	%lit_cap_ok =w csltl %o_lit, %cap
	jnz %lit_cap_ok, @lit_write, @err_cap
@lit_write
	%ps =l add %src, %i_lit
	%b =w loadub %ps
	%po =l add %out, %o_lit
	storeb %b, %po
	%i_lit_next =l add %i_lit, 1
	%o_lit_next =l add %o_lit, 1
	jmp @lit_loop
@lit_done
	jmp @loop
@rep_check
	%has_rep_src =w csltl %i1, %n
	jnz %has_rep_src, @rep_prep, @err_src
@rep_prep
	%pr =l add %src, %i1
	%rval =w loadub %pr
	%i_rep =l add %i1, 1
	%hr =l extsw %h
	%cnt =l sub 1, %hr
	jmp @rep_loop
@rep_loop
	%o_rep =l phi @rep_prep %o, @rep_write %o_rep_next
	%rem =l phi @rep_prep %cnt, @rep_write %rem_next
	%more_rep =w csgtl %rem, 0
	jnz %more_rep, @rep_cap_chk, @rep_done
@rep_cap_chk
	%rep_cap_ok =w csltl %o_rep, %cap
	jnz %rep_cap_ok, @rep_write, @err_cap
@rep_write
	%pro =l add %out, %o_rep
	storeb %rval, %pro
	%o_rep_next =l add %o_rep, 1
	%rem_next =l sub %rem, 1
	jmp @rep_loop
@rep_done
	jmp @loop
@err_src
	ret -2
@err_cap
	ret -1
}
