export function l $binary_search(l %a, l %n, l %key) {
@start
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start 0, @less %mid1, @greater %lo
	%hi =l phi @start %hi0, @less %hi, @greater %midm1
	%cond =w cslel %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @less, @greater
@less
	%mid1 =l add %mid, 1
	jmp @loop
@greater
	%midm1 =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
