export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%nr =l add %na, %nb
	%has_r =w cnel %nr, 0
	jnz %has_r, @zero_loop, @ret_zero

@zero_loop
	%zi =l phi @start 0, @zero_loop %zi_next
	%zoff =l shl %zi, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zi_next =l add %zi, 1
	%zcond =w csltl %zi_next, %nr
	jnz %zcond, @zero_loop, @after_zero

@after_zero
	%na_nz =w cnel %na, 0
	jnz %na_nz, @check_nb, @ret_zero

@check_nb
	%nb_nz =w cnel %nb, 0
	jnz %nb_nz, @outer_loop, @ret_zero

@outer_loop
	%i =l phi @check_nb 0, @outer_next %i_next
	%aoff =l shl %i, 3
	%aptr =l add %a, %aoff
	%ai =l loadl %aptr
	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	jmp @inner_loop

@inner_loop
	%j =l phi @outer_loop 0, @inner_loop %j_next
	%carry =l phi @outer_loop 0, @inner_loop %carry_next
	%boff =l shl %j, 3
	%bptr =l add %b, %boff
	%bj =l loadl %bptr
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%p00_hi =l shr %p00, 32
	%p00_lo =l and %p00, 4294967295
	%mid1 =l add %p01, %p00_hi
	%mid1_lo =l and %mid1, 4294967295
	%mid1_hi =l shr %mid1, 32
	%mid2 =l add %p10, %mid1_lo
	%mid2_lo =l and %mid2, 4294967295
	%mid2_hi =l shr %mid2, 32
	%mid2_sh =l shl %mid2_lo, 32
	%lo =l or %p00_lo, %mid2_sh
	%hi_t =l add %p11, %mid1_hi
	%hi =l add %hi_t, %mid2_hi
	%ij =l add %i, %j
	%roff =l shl %ij, 3
	%rptr =l add %r, %roff
	%cur =l loadl %rptr
	%s1 =l add %lo, %cur
	%c1w =w cultl %s1, %lo
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rptr
	%hc1 =l add %hi, %c1
	%carry_next =l add %hc1, %c2
	%j_next =l add %j, 1
	%jcond =w csltl %j_next, %nb
	jnz %jcond, @inner_loop, @outer_next

@outer_next
	%inb =l add %i, %nb
	%c_off =l shl %inb, 3
	%c_ptr =l add %r, %c_off
	storel %carry_next, %c_ptr
	%i_next =l add %i, 1
	%icond =w csltl %i_next, %na
	jnz %icond, @outer_loop, @trim_loop

@trim_loop
	%k =l phi @outer_next %nr, @trim_next %k1
	%k1 =l sub %k, 1
	%koff =l shl %k1, 3
	%kptr =l add %r, %koff
	%val =l loadl %kptr
	%val_nz =w cnel %val, 0
	jnz %val_nz, @ret_k, @trim_next

@trim_next
	%k1_nz =w cnel %k1, 0
	jnz %k1_nz, @trim_loop, @ret_zero

@ret_k
	ret %k

@ret_zero
	ret 0
}
