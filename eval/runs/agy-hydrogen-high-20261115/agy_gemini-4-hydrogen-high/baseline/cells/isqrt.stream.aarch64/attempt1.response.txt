isqrt:
    cbz x0, .Lret
    clz x1, x0
    bic x1, x1, #1
    mov x2, #0x4000000000000000
    lsr x2, x2, x1
    mov x1, #0
.Lloop:
    add x3, x1, x2
    lsr x1, x1, #1
    subs x4, x0, x3
    add x5, x1, x2
    csel x0, x4, x0, hs
    csel x1, x5, x1, hs
    lsr x2, x2, #2
    cbnz x2, .Lloop
    mov x0, x1
.Lret:
    ret
