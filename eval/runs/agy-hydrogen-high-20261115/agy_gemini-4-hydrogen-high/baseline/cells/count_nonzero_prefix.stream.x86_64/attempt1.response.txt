count_nonzero_prefix:
    xor eax, eax
    cmp rsi, 4
    jb .Ltail
    pxor xmm0, xmm0
    lea rdx, [rsi - 3]
.Lloop4:
    movdqu xmm1, xmmword ptr [rdi + rax*8]
    movdqu xmm2, xmmword ptr [rdi + rax*8 + 16]
    pcmpeqd xmm1, xmm0
    pcmpeqd xmm2, xmm0
    pshufd xmm3, xmm1, 0xb1
    pshufd xmm4, xmm2, 0xb1
    pand xmm1, xmm3
    pand xmm2, xmm4
    packssdw xmm1, xmm2
    pmovmskb ecx, xmm1
    test ecx, ecx
    jnz .Lfound
    add rax, 4
    cmp rax, rdx
    jb .Lloop4
.Ltail:
    cmp rax, rsi
    jae .Ldone
.Ltail_loop:
    cmp qword ptr [rdi + rax*8], 0
    je .Ldone
    inc rax
    cmp rax, rsi
    jb .Ltail_loop
.Ldone:
    ret
.Lfound:
    bsf ecx, ecx
    shr ecx, 2
    add rax, rcx
    ret
