dot_i32_simd:
    pxor xmm0, xmm0
    xor eax, eax
    xor ecx, ecx
    cmp rdx, 4
    jl .Ltail_check
    lea r8, [rdx - 3]
.Lsimd_loop:
    movdqu xmm1, xmmword ptr [rdi + rcx*4]
    movdqu xmm2, xmmword ptr [rsi + rcx*4]
    movdqa xmm3, xmm1
    pmuldq xmm3, xmm2
    psrldq xmm1, 4
    psrldq xmm2, 4
    pmuldq xmm1, xmm2
    paddq xmm0, xmm3
    paddq xmm0, xmm1
    add rcx, 4
    cmp rcx, r8
    jl .Lsimd_loop
    pshufd xmm1, xmm0, 0x4E
    paddq xmm0, xmm1
    movq rax, xmm0
.Ltail_check:
    cmp rcx, rdx
    jge .Ldone
.Ltail_loop:
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    cmp rcx, rdx
    jl .Ltail_loop
.Ldone:
    ret
