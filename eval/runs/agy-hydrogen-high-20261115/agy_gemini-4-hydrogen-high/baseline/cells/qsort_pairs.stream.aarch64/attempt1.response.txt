qsort_pairs:
    cmp x1, #1
    b.le .Lret0
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    mov x19, x0
    mov x20, x1
.Lpartition:
    sub x9, x20, #1
    add x13, x19, x9, lsl #4
    lsr x9, x9, #1
    add x9, x19, x9, lsl #4
    ldp x10, x11, [x9]
    mov x12, x19
.Lscan_i:
    ldp x14, x15, [x12]
    cmp x14, x10
    b.ne .Lcmp_i
    cmp x15, x11
.Lcmp_i:
    b.ge .Lscan_j
    add x12, x12, #16
    b .Lscan_i
.Lscan_j:
    ldp x2, x3, [x13]
    cmp x2, x10
    b.ne .Lcmp_j
    cmp x3, x11
.Lcmp_j:
    b.le .Lcheck_swap
    sub x13, x13, #16
    b .Lscan_j
.Lcheck_swap:
    cmp x12, x13
    b.hs .Lpart_done
    stp x2, x3, [x12], #16
    stp x14, x15, [x13], #-16
    b .Lscan_i
.Lpart_done:
    sub x2, x13, x19
    lsr x2, x2, #4
    add x2, x2, #1
    add x3, x13, #16
    sub x4, x20, x2
    cmp x2, x4
    b.gt .Lright_smaller
    mov x0, x19
    mov x1, x2
    mov x19, x3
    mov x20, x4
    b .Ldo_call
.Lright_smaller:
    mov x0, x3
    mov x1, x4
    mov x20, x2
.Ldo_call:
    bl qsort_pairs
    cmp x20, #1
    b.gt .Lpartition
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
.Lret0:
    mov x0, #0
    ret
