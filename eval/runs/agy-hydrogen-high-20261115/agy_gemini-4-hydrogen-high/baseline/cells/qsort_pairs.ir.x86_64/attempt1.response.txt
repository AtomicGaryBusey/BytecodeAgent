export function l $qsort_pairs(l %a0, l %n0) {
@start
	jmp @tail_loop
@tail_loop
	%a =l phi @start %a0, @tail_left %a, @tail_right %right_p
	%n =l phi @start %n0, @tail_left %n_left, @tail_right %n_right
	%done =w cslel %n, 1
	jnz %done, @ret_zero, @partition_init
@partition_init
	%mid =l shr %n, 1
	%mid_off =l shl %mid, 4
	%piv_p =l add %a, %mid_off
	%pk =l loadl %piv_p
	%piv_vp =l add %piv_p, 8
	%pv =l loadl %piv_vp
	%last_idx =l sub %n, 1
	%last_off =l shl %last_idx, 4
	%end_p =l add %a, %last_off
	jmp @loop
@loop
	%lt =l phi @partition_init %a, @do_lt %lt_next, @do_gt %lt, @do_eq %lt
	%i =l phi @partition_init %a, @do_lt %i_lt, @do_gt %i, @do_eq %i_eq
	%gt =l phi @partition_init %end_p, @do_lt %gt, @do_gt %gt_next, @do_eq %gt
	%cond =w culel %i, %gt
	jnz %cond, @loop_body, @loop_end
@loop_body
	%ik =l loadl %i
	%i_vp =l add %i, 8
	%iv =l loadl %i_vp
	%k_lt =w csltl %ik, %pk
	jnz %k_lt, @do_lt, @chk_k_gt
@chk_k_gt
	%k_gt =w csgtl %ik, %pk
	jnz %k_gt, @do_gt, @chk_v
@chk_v
	%v_lt =w csltl %iv, %pv
	jnz %v_lt, @do_lt, @chk_v_gt
@chk_v_gt
	%v_gt =w csgtl %iv, %pv
	jnz %v_gt, @do_gt, @do_eq
@do_lt
	%ltk =l loadl %lt
	%lt_vp =l add %lt, 8
	%ltv =l loadl %lt_vp
	storel %ik, %lt
	storel %iv, %lt_vp
	storel %ltk, %i
	storel %ltv, %i_vp
	%lt_next =l add %lt, 16
	%i_lt =l add %i, 16
	jmp @loop
@do_gt
	%gtk =l loadl %gt
	%gt_vp =l add %gt, 8
	%gtv =l loadl %gt_vp
	storel %ik, %gt
	storel %iv, %gt_vp
	storel %gtk, %i
	storel %gtv, %i_vp
	%gt_next =l sub %gt, 16
	jmp @loop
@do_eq
	%i_eq =l add %i, 16
	jmp @loop
@loop_end
	%left_bytes =l sub %lt, %a
	%n_left =l shr %left_bytes, 4
	%right_p =l add %gt, 16
	%right_bytes =l sub %end_p, %gt
	%n_right =l shr %right_bytes, 4
	%left_smaller =w cslel %n_left, %n_right
	jnz %left_smaller, @rec_left, @rec_right
@rec_left
	%need_left =w csgtl %n_left, 1
	jnz %need_left, @call_left, @tail_right
@call_left
	%unused1 =l call $qsort_pairs(l %a, l %n_left)
	jmp @tail_right
@tail_right
	jmp @tail_loop
@rec_right
	%need_right =w csgtl %n_right, 1
	jnz %need_right, @call_right, @tail_left
@call_right
	%unused2 =l call $qsort_pairs(l %right_p, l %n_right)
	jmp @tail_left
@tail_left
	jmp @tail_loop
@ret_zero
	ret 0
}
