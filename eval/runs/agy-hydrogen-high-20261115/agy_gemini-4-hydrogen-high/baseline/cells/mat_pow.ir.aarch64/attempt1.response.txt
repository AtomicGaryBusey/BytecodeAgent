export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %nn8 =l mul %nn, 8
    %prod =l add %tmp, %nn8
    %n_plus_1 =l add %n, 1
    %stride_diag =l mul %n_plus_1, 8
    %n8 =l mul %n, 8
    %twonn =l mul %nn, 2
    jmp @zero_out_loop

@zero_out_loop
    %zi =l phi @start 0, @zero_out_loop %zi_next
    %zoff =l mul %zi, 8
    %zptr =l add %out, %zoff
    storel 0, %zptr
    %zi_next =l add %zi, 1
    %zcond =w csltl %zi_next, %nn
    jnz %zcond, @zero_out_loop, @diag_loop

@diag_loop
    %di =l phi @zero_out_loop 0, @diag_loop %di_next
    %doff =l mul %di, %stride_diag
    %dptr =l add %out, %doff
    storel 1, %dptr
    %di_next =l add %di, 1
    %dcond =w csltl %di_next, %n
    jnz %dcond, @diag_loop, @check_k0

@check_k0
    %k0 =w ceql %k, 0
    jnz %k0, @zero_tmp_loop, @copy_a_loop

@copy_a_loop
    %ci =l phi @check_k0 0, @copy_a_loop %ci_next
    %coff =l mul %ci, 8
    %csrc =l add %a, %coff
    %cdst =l add %tmp, %coff
    %cval =l loadl %csrc
    storel %cval, %cdst
    %ci_next =l add %ci, 1
    %ccond =w csltl %ci_next, %nn
    jnz %ccond, @copy_a_loop, @pow_loop

@pow_loop
    %cur_k =l phi @copy_a_loop %k, @matmul_done %m_k
    %bit =l and %cur_k, 1
    %bitw =w cnel %bit, 0
    jnz %bitw, @do_mul_out, @after_mul_out

@do_mul_out
    jmp @matmul

@after_mul_out
    %ak =l phi @pow_loop %cur_k, @matmul_done %m_k
    %next_k =l shr %ak, 1
    %done =w ceql %next_k, 0
    jnz %done, @zero_tmp_loop, @do_sq_base

@do_sq_base
    jmp @matmul

@matmul
    %m_left =l phi @do_mul_out %out, @do_sq_base %tmp
    %m_mode =w phi @do_mul_out 0, @do_sq_base 1
    %m_k =l phi @do_mul_out %cur_k, @do_sq_base %next_k
    jmp @mm_i_loop

@mm_i_loop
    %mi =l phi @matmul 0, @mm_i_next %mi_next
    %mi_row_off =l mul %mi, %n8
    %left_row =l add %m_left, %mi_row_off
    %prod_row =l add %prod, %mi_row_off
    jmp @mm_j_loop

@mm_j_loop
    %mj =l phi @mm_i_loop 0, @mm_j_next %mj_next
    %mj_off =l mul %mj, 8
    %tmp_col =l add %tmp, %mj_off
    jmp @mm_t_loop

@mm_t_loop
    %mt =l phi @mm_j_loop 0, @mm_t_loop %mt_next
    %msum =l phi @mm_j_loop 0, @mm_t_loop %msum_next
    %mt_off =l mul %mt, 8
    %l_ptr =l add %left_row, %mt_off
    %l_val =l loadl %l_ptr
    %mt_row_off =l mul %mt, %n8
    %r_ptr =l add %tmp_col, %mt_row_off
    %r_val =l loadl %r_ptr
    %p_val =l mul %l_val, %r_val
    %msum_next =l add %msum, %p_val
    %mt_next =l add %mt, 1
    %t_cond =w csltl %mt_next, %n
    jnz %t_cond, @mm_t_loop, @mm_j_next

@mm_j_next
    %prod_ptr =l add %prod_row, %mj_off
    storel %msum_next, %prod_ptr
    %mj_next =l add %mj, 1
    %j_cond =w csltl %mj_next, %n
    jnz %j_cond, @mm_j_loop, @mm_i_next

@mm_i_next
    %mi_next =l add %mi, 1
    %i_cond =w csltl %mi_next, %n
    jnz %i_cond, @mm_i_loop, @mm_copy_loop

@mm_copy_loop
    %mci =l phi @mm_i_next 0, @mm_copy_loop %mci_next
    %mcoff =l mul %mci, 8
    %mc_src =l add %prod, %mcoff
    %mc_dst =l add %m_left, %mcoff
    %mc_val =l loadl %mc_src
    storel %mc_val, %mc_dst
    %mci_next =l add %mci, 1
    %mc_cond =w csltl %mci_next, %nn
    jnz %mc_cond, @mm_copy_loop, @matmul_done

@matmul_done
    jnz %m_mode, @pow_loop, @after_mul_out

@zero_tmp_loop
    %zti =l phi @check_k0 0, @after_mul_out 0, @zero_tmp_loop %zti_next
    %ztoff =l mul %zti, 8
    %ztptr =l add %tmp, %ztoff
    storel 0, %ztptr
    %zti_next =l add %zti, 1
    %ztcond =w csltl %zti_next, %twonn
    jnz %ztcond, @zero_tmp_loop, @done

@done
    ret 0
}
