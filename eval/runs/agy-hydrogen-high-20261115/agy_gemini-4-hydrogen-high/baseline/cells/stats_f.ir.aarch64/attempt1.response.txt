export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%has_more =w csltl 1, %n
	jnz %has_more, @loop, @done
@loop
	%i =l phi @start 1, @loop_next %i1
	%min =l phi @start %y0, @loop_next %min1
	%max =l phi @start %y0, @loop_next %max1
	%sum =l phi @start %y0, @loop_next %sum1
	%odd =l phi @start %odd0, @loop_next %odd1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%is_lt =w csltl %y, %min
	jnz %is_lt, @upd_min, @chk_max
@upd_min
	jmp @chk_max
@chk_max
	%min1 =l phi @loop %min, @upd_min %y
	%is_gt =w csgtl %y, %max
	jnz %is_gt, @upd_max, @loop_next
@upd_max
	jmp @loop_next
@loop_next
	%max1 =l phi @chk_max %max, @upd_max %y
	%sum1 =l add %sum, %y
	%bit =l and %y, 1
	%odd1 =l add %odd, %bit
	%i1 =l add %i, 1
	%cont =w csltl %i1, %n
	jnz %cont, @loop, @done
@done
	%fmin =l phi @start %y0, @loop_next %min1
	%fmax =l phi @start %y0, @loop_next %max1
	%fsum =l phi @start %y0, @loop_next %sum1
	%fodd =l phi @start %odd0, @loop_next %odd1
	storel %fmin, %out
	%out1 =l add %out, 8
	storel %fmax, %out1
	%out2 =l add %out, 16
	storel %fsum, %out2
	ret %fodd
}
