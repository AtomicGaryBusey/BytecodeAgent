count_desc:
    xor eax, eax
    cmp rsi, 2
    jl .Ldone
    movdqu xmm0, xmmword ptr [rdi]
    pxor xmm3, xmm3
    lea rcx, [rsi - 2]
    cmp rcx, 2
    jl .Ltail
.Lloop:
    add rdi, 16
    movdqu xmm2, xmmword ptr [rdi]
    movdqa xmm1, xmm2
    palignr xmm1, xmm0, 8
    pcmpgtq xmm0, xmm1
    psubq xmm3, xmm0
    movdqa xmm0, xmm2
    sub rcx, 2
    cmp rcx, 2
    jge .Lloop
.Ltail:
    movdqa xmm1, xmm3
    psrldq xmm1, 8
    paddq xmm3, xmm1
    movq rax, xmm3
    movq rdx, xmm0
    pextrq r8, xmm0, 1
    cmp rdx, r8
    setg r9b
    movzx r9d, r9b
    add rax, r9
    test rcx, rcx
    jz .Ldone
    mov rdx, qword ptr [rdi + 16]
    cmp r8, rdx
    setg r9b
    movzx r9d, r9b
    add rax, r9
.Ldone:
    ret
