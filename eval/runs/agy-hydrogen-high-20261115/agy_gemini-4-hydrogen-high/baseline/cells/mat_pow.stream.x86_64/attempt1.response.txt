mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r11, rsi
    imul r11, rsi
    lea r13, [rsi*8]
    mov r9, r11
    mov rax, rcx
.Lzero_out:
    mov qword ptr [rax], 0
    add rax, 8
    dec r9
    jnz .Lzero_out
    mov r9, rsi
    mov rax, rcx
    lea r10, [r13 + 8]
.Lset_diag:
    mov qword ptr [rax], 1
    add rax, r10
    dec r9
    jnz .Lset_diag
    test rdx, rdx
    jz .Ldone_pow
    mov r9, r11
    mov rax, rdi
    mov r10, r8
.Lcopy_a:
    mov rbx, qword ptr [rax]
    mov qword ptr [r10], rbx
    add rax, 8
    add r10, 8
    dec r9
    jnz .Lcopy_a
.Lpow_loop:
    test rdx, 1
    jz .Ldo_square_check
    mov r14, rcx
    jmp .Lmatmul
.Ldo_square_check:
    shr rdx, 1
    jz .Ldone_pow
    mov r14, r8
.Lmatmul:
    mov rax, r14
    lea rbp, [r8 + r11*8]
    mov rdi, rsi
.Louter_i:
    xor r9d, r9d
.Lmiddle_j:
    lea r12, [r8 + r9]
    xor ebx, ebx
    xor r10d, r10d
.Linner_t:
    mov r15, qword ptr [rax + r10]
    imul r15, qword ptr [r12]
    add rbx, r15
    add r12, r13
    add r10, 8
    cmp r10, r13
    jb .Linner_t
    mov qword ptr [rbp + r9], rbx
    add r9, 8
    cmp r9, r13
    jb .Lmiddle_j
    add rax, r13
    add rbp, r13
    dec rdi
    jnz .Louter_i
    lea rax, [r8 + r11*8]
    mov rbp, r14
    mov rdi, r11
.Lcopy_res:
    mov r15, qword ptr [rax]
    mov qword ptr [rbp], r15
    add rax, 8
    add rbp, 8
    dec rdi
    jnz .Lcopy_res
    cmp r14, rcx
    je .Ldo_square_check
    jmp .Lpow_loop
.Ldone_pow:
    lea r9, [r11 + r11]
    mov rax, r8
.Lzero_tmp:
    mov qword ptr [rax], 0
    add rax, 8
    dec r9
    jnz .Lzero_tmp
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
