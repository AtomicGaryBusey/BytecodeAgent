bubble_sort:
	cmp x1, #1
	b.le .Ldone
	sub x1, x1, #1
	add x9, x0, x1, lsl #3
.Louter:
	mov x10, x0
	mov w11, #0
.Linner:
	ldp x12, x13, [x10]
	cmp x12, x13
	b.le .Lnoswap
	stp x13, x12, [x10]
	mov w11, #1
.Lnoswap:
	add x10, x10, #8
	cmp x10, x9
	b.lo .Linner
	cbz w11, .Ldone
	sub x9, x9, #8
	cmp x9, x0
	b.hi .Louter
.Ldone:
	mov x0, #0
	ret
