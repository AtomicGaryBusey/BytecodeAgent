export function l $collatz_max(l %n) {
@start
	jmp @outer_hdr
@outer_hdr
	%k =l phi @start 1, @outer_inc %k_next
	%max =l phi @start 0, @outer_inc %max_next
	%cond_k =w cslel %k, %n
	jnz %cond_k, @inner_hdr, @done
@inner_hdr
	%x =l phi @outer_hdr %k, @even %x_even, @odd %x_odd
	%steps =l phi @outer_hdr 0, @even %steps_next, @odd %steps_next
	%cond_x =w cnel %x, 1
	jnz %cond_x, @inner_body, @outer_next
@inner_body
	%lsb =l and %x, 1
	%is_odd =w cnel %lsb, 0
	%steps_next =l add %steps, 1
	jnz %is_odd, @odd, @even
@even
	%x_even =l shr %x, 1
	jmp @inner_hdr
@odd
	%x3 =l mul %x, 3
	%x_odd =l add %x3, 1
	jmp @inner_hdr
@outer_next
	%is_gt =w csgtl %steps, %max
	jnz %is_gt, @update_max, @keep_max
@update_max
	jmp @outer_inc
@keep_max
	jmp @outer_inc
@outer_inc
	%max_next =l phi @update_max %steps, @keep_max %max
	%k_next =l add %k, 1
	jmp @outer_hdr
@done
	ret %max
}
