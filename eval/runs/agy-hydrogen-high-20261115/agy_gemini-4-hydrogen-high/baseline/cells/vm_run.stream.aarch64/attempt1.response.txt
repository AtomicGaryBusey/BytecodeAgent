vm_run:
    mov x4, #0
    mov x5, x2
    add x6, x2, x3, lsl #3
.Lloop:
    cmp x4, x1
    b.hs .Ldone
    ldr x9, [x0, x4, lsl #3]
    cbz x9, .Ldone
    cmp x9, #5
    b.hi .Lge6
    cmp x9, #1
    b.eq .Lop_push
    cmp x9, #2
    b.eq .Lop_add
    cmp x9, #3
    b.eq .Lop_sub
    cmp x9, #4
    b.eq .Lop_mul
    cmp x9, #5
    b.eq .Lop_dup
    b .Lnext1
.Lge6:
    cmp x9, #6
    b.eq .Lop_swap
    cmp x9, #7
    b.eq .Lop_drop
    cmp x9, #8
    b.eq .Lop_over
    cmp x9, #9
    b.eq .Lop_lt
    cmp x9, #10
    b.eq .Lop_jz
    cmp x9, #11
    b.eq .Lop_jmp
.Lnext1:
    add x4, x4, #1
    b .Lloop
.Lop_push:
    add x10, x4, #1
    ldr x10, [x0, x10, lsl #3]
    str x10, [x5], #8
    add x4, x4, #2
    b .Lloop
.Lop_add:
    ldp x10, x11, [x5, #-16]
    add x10, x10, x11
    b .Lbin_end
.Lop_sub:
    ldp x10, x11, [x5, #-16]
    sub x10, x10, x11
    b .Lbin_end
.Lop_mul:
    ldp x10, x11, [x5, #-16]
    mul x10, x10, x11
    b .Lbin_end
.Lop_lt:
    ldp x10, x11, [x5, #-16]
    cmp x10, x11
    cset x10, lt
.Lbin_end:
    str x10, [x5, #-16]
    sub x5, x5, #8
    add x4, x4, #1
    b .Lloop
.Lop_dup:
    ldr x10, [x5, #-8]
    str x10, [x5], #8
    add x4, x4, #1
    b .Lloop
.Lop_swap:
    ldp x10, x11, [x5, #-16]
    stp x11, x10, [x5, #-16]
    add x4, x4, #1
    b .Lloop
.Lop_drop:
    sub x5, x5, #8
    add x4, x4, #1
    b .Lloop
.Lop_over:
    ldr x10, [x5, #-16]
    str x10, [x5], #8
    add x4, x4, #1
    b .Lloop
.Lop_jz:
    ldr x10, [x5, #-8]!
    cbz x10, .Lop_jmp
    add x4, x4, #2
    b .Lloop
.Lop_jmp:
    add x10, x4, #1
    ldr x4, [x0, x10, lsl #3]
    b .Lloop
.Ldone:
    mov x0, #0
    cmp x5, x2
    b.eq .Lzero_loop
    ldr x0, [x5, #-8]
.Lzero_loop:
    cmp x5, x6
    b.hs .Lret
    str xzr, [x5], #8
    b .Lzero_loop
.Lret:
    ret
