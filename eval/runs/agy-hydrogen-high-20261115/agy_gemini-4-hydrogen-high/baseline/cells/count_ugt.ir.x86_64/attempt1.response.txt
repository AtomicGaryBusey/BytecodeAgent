export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cond =w csgtl %n, 0
	jnz %cond, @loop, @zero
@zero
	ret 0
@loop
	%i =l phi @start 0, @loop %i1
	%cnt =l phi @start 0, @loop %cnt1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @loop, @done
@done
	ret %cnt1
}
