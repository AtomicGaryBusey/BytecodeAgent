export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @cond
@cond
	%i =l phi @start 0, @loop %i_next
	%cnt =l phi @start 0, @loop %cnt_next
	%more =w csltl %i, %n
	jnz %more, @loop, @done
@loop
	%off =l mul %i, 8
	%psrc =l add %src, %off
	%x =l loadl %psrc
	%y =l call %f(l %x)
	%pdst =l add %dst, %off
	storel %y, %pdst
	%lt =w csltl %y, %x
	%lt64 =l extuw %lt
	%cnt_next =l add %cnt, %lt64
	%i_next =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
