export function l $collatz_max(l %n) {
@start
	jmp @outer_cond
@outer_cond
	%k =l phi @start 1, @outer_next %k_next
	%max =l phi @start 0, @outer_next %max_next
	%cond =w cslel %k, %n
	jnz %cond, @outer_body, @done
@outer_body
	jmp @inner_cond
@inner_cond
	%x =l phi @outer_body %k, @inner_next %x_next
	%steps =l phi @outer_body 0, @inner_next %steps_next
	%is_one =w ceql %x, 1
	jnz %is_one, @inner_done, @inner_body
@inner_body
	%odd =l and %x, 1
	%is_odd =w ceql %odd, 1
	jnz %is_odd, @do_odd, @do_even
@do_odd
	%t1 =l mul %x, 3
	%x_odd =l add %t1, 1
	jmp @inner_next
@do_even
	%x_even =l shr %x, 1
	jmp @inner_next
@inner_next
	%x_next =l phi @do_odd %x_odd, @do_even %x_even
	%steps_next =l add %steps, 1
	jmp @inner_cond
@inner_done
	%gt =w csgtl %steps, %max
	jnz %gt, @update_max, @outer_next
@update_max
	jmp @outer_next
@outer_next
	%max_next =l phi @inner_done %max, @update_max %steps
	%k_next =l add %k, 1
	jmp @outer_cond
@done
	ret %max
}
