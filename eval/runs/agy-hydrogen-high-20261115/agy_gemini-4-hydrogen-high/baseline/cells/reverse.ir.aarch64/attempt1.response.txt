export function l $reverse(l %a, l %n) {
@start
	%cond0 =w csgtl %n, 1
	jnz %cond0, @init, @done
@init
	%n1 =l sub %n, 1
	%off =l mul %n1, 8
	%right0 =l add %a, %off
	jmp @loop
@loop
	%left =l phi @init %a, @body %left1
	%right =l phi @init %right0, @body %right1
	%cond =w cultl %left, %right
	jnz %cond, @body, @done
@body
	%vl =l loadl %left
	%vr =l loadl %right
	storel %vr, %left
	storel %vl, %right
	%left1 =l add %left, 8
	%right1 =l sub %right, 8
	jmp @loop
@done
	ret 0
}
