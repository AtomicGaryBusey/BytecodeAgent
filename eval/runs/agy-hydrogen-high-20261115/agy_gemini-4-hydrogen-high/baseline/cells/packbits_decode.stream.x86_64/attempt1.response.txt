packbits_decode:
    cld
    lea r8, [rdi + rsi]
    lea r9, [rdx + rcx]
    mov r10, rdx
    mov rsi, rdi
    mov rdi, rdx
.Lloop:
    cmp rsi, r8
    jae .Ldone
    movsx rax, byte ptr [rsi]
    inc rsi
    test rax, rax
    js .Lneg
    inc rax
    lea rcx, [rsi + rax]
    cmp rcx, r8
    ja .Lerr_src
    lea rdx, [rdi + rax]
    cmp rdx, r9
    ja .Llit_overflow
    mov rcx, rax
    rep movsb
    jmp .Lloop
.Lneg:
    cmp rax, -128
    je .Lloop
    neg rax
    inc rax
    cmp rsi, r8
    jae .Lerr_src
    mov rcx, rax
    movzx eax, byte ptr [rsi]
    inc rsi
    lea rdx, [rdi + rcx]
    cmp rdx, r9
    ja .Lrep_overflow
    rep stosb
    jmp .Lloop
.Llit_overflow:
    mov rcx, r9
    sub rcx, rdi
    rep movsb
    mov rax, -1
    ret
.Lrep_overflow:
    mov rcx, r9
    sub rcx, rdi
    rep stosb
    mov rax, -1
    ret
.Lerr_src:
    mov rax, -2
    ret
.Ldone:
    mov rax, rdi
    sub rax, r10
    ret
