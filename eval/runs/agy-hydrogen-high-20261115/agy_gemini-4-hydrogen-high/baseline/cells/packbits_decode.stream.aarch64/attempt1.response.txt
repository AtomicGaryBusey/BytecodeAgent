packbits_decode:
    add x1, x0, x1
    mov x4, x2
    add x3, x2, x3
.Lloop:
    cmp x0, x1
    b.hs .Ldone
    ldrsb x5, [x0], #1
    cmp x5, #0
    b.lt .Lnot_lit
    add x5, x5, #1
    sub x6, x1, x0
    cmp x6, x5
    b.lo .Lerr_src
    sub x6, x3, x2
    cmp x6, x5
    b.lo .Llit_overflow
.Llit_copy:
    ldrb w7, [x0], #1
    strb w7, [x2], #1
    subs x5, x5, #1
    b.ne .Llit_copy
    b .Lloop
.Lnot_lit:
    cmn x5, #128
    b.eq .Lloop
    cmp x0, x1
    b.hs .Lerr_src
    ldrb w7, [x0], #1
    mov x6, #1
    sub x5, x6, x5
    sub x6, x3, x2
    cmp x6, x5
    b.lo .Lrep_overflow
.Lrep_fill:
    strb w7, [x2], #1
    subs x5, x5, #1
    b.ne .Lrep_fill
    b .Lloop
.Llit_overflow:
    cbz x6, .Lerr_cap
.Llit_ov_loop:
    ldrb w7, [x0], #1
    strb w7, [x2], #1
    subs x6, x6, #1
    b.ne .Llit_ov_loop
    b .Lerr_cap
.Lrep_overflow:
    cbz x6, .Lerr_cap
.Lrep_ov_loop:
    strb w7, [x2], #1
    subs x6, x6, #1
    b.ne .Lrep_ov_loop
.Lerr_cap:
    mov x0, #-1
    ret
.Lerr_src:
    mov x0, #-2
    ret
.Ldone:
    sub x0, x2, x4
    ret
