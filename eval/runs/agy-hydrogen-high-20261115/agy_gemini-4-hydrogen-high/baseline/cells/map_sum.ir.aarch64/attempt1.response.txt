export function l $map_sum(l %a, l %n, l %f) {
@start
	%c0 =w csgtl %n, 0
	jnz %c0, @loop, @empty
@empty
	ret 0
@loop
	%i =l phi @start 0, @loop %i1
	%p =l phi @start %a, @loop %p1
	%sum =l phi @start 0, @loop %sum1
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum1 =l add %sum, %r
	%p1 =l add %p, 8
	%i1 =l add %i, 1
	%c =w csltl %i1, %n
	jnz %c, @loop, @done
@done
	ret %sum1
}
