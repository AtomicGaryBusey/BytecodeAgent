max_byte:
    xor eax, eax
    cmp rsi, 16
    jb .Llt16
    movdqu xmm0, xmmword ptr [rdi]
    add rdi, 16
    sub rsi, 16
    jz .Lreduce16
.Lloop16:
    cmp rsi, 16
    jb .Llast16
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    jnz .Lloop16
    jmp .Lreduce16
.Llast16:
    movdqu xmm1, xmmword ptr [rdi + rsi - 16]
    pmaxub xmm0, xmm1
.Lreduce16:
    pshufd xmm1, xmm0, 0x4e
    pmaxub xmm0, xmm1
.Lreduce8:
    pshufd xmm1, xmm0, 0x01
    pmaxub xmm0, xmm1
.Lreduce4:
    pshuflw xmm1, xmm0, 0x01
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx edx, ah
    movzx eax, al
    cmp eax, edx
    cmovb eax, edx
    ret
.Llt16:
    test rsi, rsi
    jz .Ldone
    cmp rsi, 8
    jb .Llt8
    movq xmm0, qword ptr [rdi]
    movq xmm1, qword ptr [rdi + rsi - 8]
    pmaxub xmm0, xmm1
    jmp .Lreduce8
.Llt8:
    cmp rsi, 4
    jb .Llt4
    movd xmm0, dword ptr [rdi]
    movd xmm1, dword ptr [rdi + rsi - 4]
    pmaxub xmm0, xmm1
    jmp .Lreduce4
.Llt4:
    movzx eax, byte ptr [rdi]
    cmp rsi, 1
    je .Ldone
    movzx ecx, byte ptr [rdi + 1]
    cmp eax, ecx
    cmovb eax, ecx
    movzx ecx, byte ptr [rdi + rsi - 1]
    cmp eax, ecx
    cmovb eax, ecx
.Ldone:
    ret
