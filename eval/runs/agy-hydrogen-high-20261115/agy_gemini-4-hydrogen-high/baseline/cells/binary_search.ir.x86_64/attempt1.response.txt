export function l $binary_search(l %a, l %n, l %key) {
@start
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start 0, @less %mid1, @greater %lo
	%hi =l phi @start %hi0, @less %hi, @greater %midm1
	%cond =w cslel %lo, %hi
	jnz %cond, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%iseq =w ceql %val, %key
	jnz %iseq, @found, @cmp
@cmp
	%islt =w csltl %val, %key
	jnz %islt, @less, @greater
@less
	%mid1 =l add %mid, 1
	jmp @loop
@greater
	%midm1 =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
