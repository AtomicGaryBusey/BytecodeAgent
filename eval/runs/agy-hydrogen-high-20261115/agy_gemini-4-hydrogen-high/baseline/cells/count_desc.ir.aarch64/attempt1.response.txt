export function l $count_desc(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret0, @init
@ret0
	ret 0
@init
	%first =l loadl %a
	%p0 =l add %a, 8
	%rem0 =l sub %n, 1
	jmp @loop
@loop
	%prev =l phi @init %first, @loop %cur
	%p =l phi @init %p0, @loop %p1
	%rem =l phi @init %rem0, @loop %rem1
	%cnt =l phi @init 0, @loop %cnt1
	%cur =l loadl %p
	%is_desc =w csgtl %prev, %cur
	%inc =l extuw %is_desc
	%cnt1 =l add %cnt, %inc
	%p1 =l add %p, 8
	%rem1 =l sub %rem, 1
	%more =w cnel %rem1, 0
	jnz %more, @loop, @done
@done
	ret %cnt1
}
