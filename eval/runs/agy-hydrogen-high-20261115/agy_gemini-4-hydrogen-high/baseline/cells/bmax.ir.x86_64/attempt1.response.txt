export function l $bmax(l %a, l %n) {
@start
    %max0 =l loadl %a
    jmp @loop
@loop
    %i =l phi @start 1, @next %i1
    %max =l phi @start %max0, @next %max1
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %gt =w csgtl %val, %max
    jnz %gt, @update, @next
@update
    jmp @next
@next
    %max1 =l phi @body %max, @update %val
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %max
}
