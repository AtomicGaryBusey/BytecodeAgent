export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%idx =l phi @start 0, @step %next
	%cond =w csltl %idx, %n
	jnz %cond, @check, @done
@check
	%off =l mul %idx, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%is_zero =w ceql %val, 0
	jnz %is_zero, @done, @step
@step
	%next =l add %idx, 1
	jmp @loop
@done
	ret %idx
}
