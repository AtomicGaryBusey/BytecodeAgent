export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %lo1, @greater %lo
	%hi =l phi @start 4294967295, @less %hi, @greater %hi1
	%ans =l phi @start 0, @less %mid, @greater %ans
	%cond =w culel %lo, %hi
	jnz %cond, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @less, @greater
@less
	%lo1 =l add %mid, 1
	jmp @loop
@greater
	%hi1 =l sub %mid, 1
	jmp @loop
@done
	ret %ans
}
