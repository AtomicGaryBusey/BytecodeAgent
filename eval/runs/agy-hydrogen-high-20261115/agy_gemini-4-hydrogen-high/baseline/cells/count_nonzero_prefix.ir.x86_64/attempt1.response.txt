export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%c0 =w csgtl %n, 0
	jnz %c0, @loop, @ret_n
@loop
	%i =l phi @start 0, @next %i1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%iszero =w ceql %val, 0
	jnz %iszero, @ret_i, @next
@next
	%i1 =l add %i, 1
	%cont =w csltl %i1, %n
	jnz %cont, @loop, @ret_n
@ret_i
	ret %i
@ret_n
	ret %n
}
