export function l $bmin_empty(l %a, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @ret_empty, @init
@ret_empty
	ret 9223372036854775807
@init
	%first =l loadl %a
	%one =l copy 1
	jmp @loop
@loop
	%i =l phi @init %one, @step %i_next
	%min =l phi @init %first, @step %min_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%is_less =w csltl %val, %min
	jnz %is_less, @update, @step
@update
	jmp @step
@step

	%i_next =l add %i, 1
	jmp @loop
@done

}
