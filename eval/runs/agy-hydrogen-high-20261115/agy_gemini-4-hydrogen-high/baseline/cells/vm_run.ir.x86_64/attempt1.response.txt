export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	jmp @loop
@loop
	%pc =l phi @entry 0, @op_push %push_pc, @op_add %add_pc, @op_sub %sub_pc, @op_mul %mul_pc, @op_dup %dup_pc, @op_swap %swap_pc, @op_drop %drop_pc, @op_over %over_pc, @op_lt %lt_pc, @op_jz_taken %jzt_pc, @op_jz_not %jzn_pc, @op_jmp %jmp_pc, @op_other %oth_pc
	%sp =l phi @entry 0, @op_push %push_sp, @op_add %add_sp, @op_sub %sub_sp, @op_mul %mul_sp, @op_dup %dup_sp, @op_swap %sp, @op_drop %drop_sp, @op_over %over_sp, @op_lt %lt_sp, @op_jz_taken %jz_sp1, @op_jz_not %jz_sp1, @op_jmp %sp, @op_other %sp
	%loop_run =w csltl %pc, %len
	jnz %loop_run, @fetch, @done
@fetch
	%pc_off =l shl %pc, 3
	%pc_ptr =l add %code, %pc_off
	%op =l loadl %pc_ptr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @done, @check_lt6
@check_lt6
	%is_lt6 =w csltl %op, 6
	jnz %is_lt6, @d1, @d6
@d1
	%is_1 =w ceql %op, 1
	jnz %is_1, @op_push, @d2
@d2
	%is_2 =w ceql %op, 2
	jnz %is_2, @op_add, @d3
@d3
	%is_3 =w ceql %op, 3
	jnz %is_3, @op_sub, @d4
@d4
	%is_4 =w ceql %op, 4
	jnz %is_4, @op_mul, @d5
@d5
	%is_5 =w ceql %op, 5
	jnz %is_5, @op_dup, @op_other
@d6
	%is_6 =w ceql %op, 6
	jnz %is_6, @op_swap, @d7
@d7
	%is_7 =w ceql %op, 7
	jnz %is_7, @op_drop, @d8
@d8
	%is_8 =w ceql %op, 8
	jnz %is_8, @op_over, @d9
@d9
	%is_9 =w ceql %op, 9
	jnz %is_9, @op_lt, @d10
@d10
	%is_10 =w ceql %op, 10
	jnz %is_10, @op_jz, @d11
@d11
	%is_11 =w ceql %op, 11
	jnz %is_11, @op_jmp, @op_other
@op_push
	%push_argp =l add %pc_ptr, 8
	%push_v =l loadl %push_argp
	%push_off =l shl %sp, 3
	%push_dst =l add %stack, %push_off
	storel %push_v, %push_dst
	%push_sp =l add %sp, 1
	%push_pc =l add %pc, 2
	jmp @loop
@op_add
	%add_sp =l sub %sp, 1
	%add_boff =l shl %add_sp, 3
	%add_bp =l add %stack, %add_boff
	%add_b =l loadl %add_bp
	%add_ap =l sub %add_bp, 8
	%add_a =l loadl %add_ap
	%add_res =l add %add_a, %add_b
	storel %add_res, %add_ap
	%add_pc =l add %pc, 1
	jmp @loop
@op_sub
	%sub_sp =l sub %sp, 1
	%sub_boff =l shl %sub_sp, 3
	%sub_bp =l add %stack, %sub_boff
	%sub_b =l loadl %sub_bp
	%sub_ap =l sub %sub_bp, 8
	%sub_a =l loadl %sub_ap
	%sub_res =l sub %sub_a, %sub_b
	storel %sub_res, %sub_ap
	%sub_pc =l add %pc, 1
	jmp @loop
@op_mul
	%mul_sp =l sub %sp, 1
	%mul_boff =l shl %mul_sp, 3
	%mul_bp =l add %stack, %mul_boff
	%mul_b =l loadl %mul_bp
	%mul_ap =l sub %mul_bp, 8
	%mul_a =l loadl %mul_ap
	%mul_res =l mul %mul_a, %mul_b
	storel %mul_res, %mul_ap
	%mul_pc =l add %pc, 1
	jmp @loop
@op_dup
	%dup_off =l shl %sp, 3
	%dup_dst =l add %stack, %dup_off
	%dup_src =l sub %dup_dst, 8
	%dup_v =l loadl %dup_src
	storel %dup_v, %dup_dst
	%dup_sp =l add %sp, 1
	%dup_pc =l add %pc, 1
	jmp @loop
@op_swap
	%swap_sp1 =l sub %sp, 1
	%swap_boff =l shl %swap_sp1, 3
	%swap_bp =l add %stack, %swap_boff
	%swap_ap =l sub %swap_bp, 8
	%swap_b =l loadl %swap_bp
	%swap_a =l loadl %swap_ap
	storel %swap_b, %swap_ap
	storel %swap_a, %swap_bp
	%swap_pc =l add %pc, 1
	jmp @loop
@op_drop
	%drop_sp =l sub %sp, 1
	%drop_pc =l add %pc, 1
	jmp @loop
@op_over
	%over_off =l shl %sp, 3
	%over_dst =l add %stack, %over_off
	%over_src =l sub %over_dst, 16
	%over_v =l loadl %over_src
	storel %over_v, %over_dst
	%over_sp =l add %sp, 1
	%over_pc =l add %pc, 1
	jmp @loop
@op_lt
	%lt_sp =l sub %sp, 1
	%lt_boff =l shl %lt_sp, 3
	%lt_bp =l add %stack, %lt_boff
	%lt_b =l loadl %lt_bp
	%lt_ap =l sub %lt_bp, 8
	%lt_a =l loadl %lt_ap
	%lt_cmp =w csltl %lt_a, %lt_b
	%lt_res =l extuw %lt_cmp
	storel %lt_res, %lt_ap
	%lt_pc =l add %pc, 1
	jmp @loop
@op_jz
	%jz_sp1 =l sub %sp, 1
	%jz_off =l shl %jz_sp1, 3
	%jz_vp =l add %stack, %jz_off
	%jz_v =l loadl %jz_vp
	%jz_nz =w cnel %jz_v, 0
	jnz %jz_nz, @op_jz_not, @op_jz_taken
@op_jz_taken
	%jzt_argp =l add %pc_ptr, 8
	%jzt_pc =l loadl %jzt_argp
	jmp @loop
@op_jz_not
	%jzn_pc =l add %pc, 2
	jmp @loop
@op_jmp
	%jmp_argp =l add %pc_ptr, 8
	%jmp_pc =l loadl %jmp_argp
	jmp @loop
@op_other
	%oth_pc =l add %pc, 1
	jmp @loop
@done
	%sp_zero =w ceql %sp, 0
	jnz %sp_zero, @done_empty, @done_nonempty
@done_empty
	jmp @zero_init
@done_nonempty
	%top_off =l shl %sp, 3
	%top_end =l add %stack, %top_off
	%top_ptr =l sub %top_end, 8
	%top_val =l loadl %top_ptr
	jmp @zero_init
@zero_init
	%ret =l phi @done_empty 0, @done_nonempty %top_val
	jmp @zero_loop
@zero_loop
	%zi =l phi @zero_init %sp, @zero_body %zi_next
	%z_more =w csltl %zi, %cap
	jnz %z_more, @zero_body, @exit
@zero_body
	%z_off =l shl %zi, 3
	%z_ptr =l add %stack, %z_off
	storel 0, %z_ptr
	%zi_next =l add %zi, 1
	jmp @zero_loop
@exit
	ret %ret
}
