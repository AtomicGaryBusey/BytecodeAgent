    xor r8d, r8d
    mov r9, rdx
.Lloop:
    cmp r8, rsi
    jae .Lhalt
    mov rax, qword ptr [rdi + r8*8]
    inc r8
    cmp eax, 5
    je .Ldup
    ja .Lge6
    cmp eax, 2
    je .Ladd
    ja .L3_4
    test eax, eax
    jz .Lhalt
.Lpush:
    mov rax, qword ptr [rdi + r8*8]
    inc r8
    mov qword ptr [r9], rax
    add r9, 8
    jmp .Lloop
.L3_4:

.Lmul:

.Ladd:

    jmp .Lloop
.Lsub:

    sub qword ptr [r9 - 16], rax
    sub r9, 8
    jmp .Lloop
.Ldup:

    mov qword ptr [r9], rax
    add r9, 8
    jmp .Lloop
.Lge6:

    je .Llt
    ja .L10_11
    cmp eax, 7

.Lover:

    add r9, 8
    jmp .Lloop
.Lswap:

    mov r10, qword ptr [r9 - 16]

    mov qword ptr [r9 - 8], r10

.Ldrop:

    jmp .Lloop
.Llt:

    jmp .Lloop
.L10_11:

    jne .Ljmp
.Ljz:

    cmp qword ptr [r9], 0
    je .Ljmp
    inc r8
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8]
    jmp .Lloop
.Lhalt:

    cmp r9, rdx
    je .Lzero_init

.Lzero_init:

.Lzero_loop:

    jae .Ldone
    mov qword ptr [r9], 0
    add r9, 8
    jmp .Lzero_loop
.Ldone:
