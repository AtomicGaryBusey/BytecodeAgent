export function l $max_byte(l %p, l %n) {
@start
	%c0 =w cslel %n, 0
	jnz %c0, @ret0, @loop
@ret0
	ret 0
@loop
	%cur =l phi @start %p, @step %next
	%rem =l phi @start %n, @step %rem1
	%max =l phi @start 0, @step %newmax
	%b =l loadub %cur
	%gt =w csgtl %b, %max
	jnz %gt, @update, @step
@update
	jmp @step
@step
	%newmax =l phi @loop %max, @update %b
	%next =l add %cur, 1
	%rem1 =l sub %rem, 1
	%more =w csgtl %rem1, 0
	jnz %more, @loop, @done
@done
	ret %newmax
}
