export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%is_zero =w ceql %b, 0
	jnz %is_zero, @divzero, @check_neg1
@check_neg1
	%is_neg1 =w ceql %b, -1
	jnz %is_neg1, @check_min, @normal
@check_min
	%min =l shl 1, 63
	%is_min =w ceql %a, %min
	jnz %is_min, @overflow, @normal
@divzero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret %min
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
