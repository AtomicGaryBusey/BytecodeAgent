packbits_encode:
    test rsi, rsi
    jle .Lzero
    test rcx, rcx
    jl .Loverflow
    mov r8, rcx
    add rsi, rdi

.Lloop:
    lea r9, [rdi + 2]
    cmp r9, rsi
    jae .Lliteral
    movzx eax, byte ptr [rdi]
    cmp al, byte ptr [rdi + 1]
    jne .Lliteral
    cmp al, byte ptr [rdi + 2]
    jne .Lliteral

    cmp rcx, 2
    jl .Loverflow
    lea r9, [rdi + 128]
    cmp r9, rsi
    cmova r9, rsi
    lea r10, [rdi + 3]

.Lrep_loop:
    cmp r10, r9
    jae .Lrep_end
    cmp al, byte ptr [r10]
    jne .Lrep_end
    inc r10
    jmp .Lrep_loop

.Lrep_end:
    mov r11, r10
    sub r11, rdi
    mov r9d, 1
    sub r9d, r11d
    mov byte ptr [rdx], r9b
    mov byte ptr [rdx + 1], al
    add rdx, 2
    sub rcx, 2
    mov rdi, r10
    cmp rdi, rsi
    jb .Lloop
    jmp .Ldone

.Lliteral:
    lea r9, [rdi + 128]
    cmp r9, rsi
    cmova r9, rsi
    lea r10, [rdi + 1]

.Llit_loop:
    cmp r10, r9
    jae .Llit_end
    lea rax, [r10 + 2]
    cmp rax, rsi
    jae .Llit_fast_end
    movzx eax, byte ptr [r10]
    cmp al, byte ptr [r10 + 1]
    jne .Llit_next
    cmp al, byte ptr [r10 + 2]
    je .Llit_end

.Llit_next:
    inc r10
    jmp .Llit_loop

.Llit_fast_end:
    mov r10, r9

.Llit_end:
    mov r11, r10
    sub r11, rdi
    lea rax, [r11 + 1]
    cmp rcx, rax
    jl .Loverflow
    lea eax, [r11 - 1]
    mov byte ptr [rdx], al
    inc rdx
    sub rcx, r11
    dec rcx

    cmp r11, 8
    jb .Lcopy_bytes

.Lcopy8:
    mov rax, qword ptr [rdi]
    mov qword ptr [rdx], rax
    add rdi, 8
    add rdx, 8
    sub r11, 8
    cmp r11, 8
    jae .Lcopy8
    test r11, r11
    jz .Lcopy_done

.Lcopy_bytes:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec r11
    jnz .Lcopy_bytes

.Lcopy_done:
    cmp rdi, rsi
    jb .Lloop

.Ldone:
    mov rax, r8
    sub rax, rcx
    ret

.Lzero:
    xor eax, eax
    ret

.Loverflow:
    mov rax, -1
    ret
