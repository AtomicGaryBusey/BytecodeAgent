export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	jmp @id_i_header
@id_i_header
	%id_i =l phi @start 0, @id_i_latch %id_i_next
	%id_i_cond =w csltl %id_i, %n
	jnz %id_i_cond, @id_j_header, @pow_header
@id_j_header
	%id_j =l phi @id_i_header 0, @id_j_body %id_j_next
	%id_j_cond =w csltl %id_j, %n
	jnz %id_j_cond, @id_j_body, @id_i_latch
@id_j_body
	%id_idx =l mul %id_i, %n
	%id_idx2 =l add %id_idx, %id_j
	%id_off =l mul %id_idx2, 8
	%id_ptr =l add %out, %id_off
	%is_diag_w =w ceql %id_i, %id_j
	%is_diag =l extuw %is_diag_w
	storel %is_diag, %id_ptr
	%id_j_next =l add %id_j, 1
	jmp @id_j_header
@id_i_latch
	%id_i_next =l add %id_i, 1
	jmp @id_i_header
@pow_header
	%step =l phi @id_i_header 0, @copy_done %step_next
	%pow_cond =w csltl %step, %k
	jnz %pow_cond, @mul_i_header, @zero_header
@mul_i_header
	%mi =l phi @pow_header 0, @mul_i_latch %mi_next
	%mi_cond =w csltl %mi, %n
	jnz %mi_cond, @mul_j_header, @copy_header
@mul_j_header
	%mj =l phi @mul_i_header 0, @mul_j_latch %mj_next
	%mj_cond =w csltl %mj, %n
	jnz %mj_cond, @mul_t_header, @mul_i_latch
@mul_t_header
	%mt =l phi @mul_j_header 0, @mul_t_body %mt_next
	%acc =l phi @mul_j_header 0, @mul_t_body %acc_next
	%mt_cond =w csltl %mt, %n
	jnz %mt_cond, @mul_t_body, @mul_j_latch
@mul_t_body
	%out_idx =l mul %mi, %n
	%out_idx2 =l add %out_idx, %mt
	%out_off =l mul %out_idx2, 8
	%out_ptr =l add %out, %out_off
	%out_val =l loadl %out_ptr
	%a_idx =l mul %mt, %n
	%a_idx2 =l add %a_idx, %mj
	%a_off =l mul %a_idx2, 8
	%a_ptr =l add %a, %a_off
	%a_val =l loadl %a_ptr
	%prod =l mul %out_val, %a_val
	%acc_next =l add %acc, %prod
	%mt_next =l add %mt, 1
	jmp @mul_t_header
@mul_j_latch
	%tmp_idx =l mul %mi, %n
	%tmp_idx2 =l add %tmp_idx, %mj
	%tmp_off =l mul %tmp_idx2, 8
	%tmp_ptr =l add %tmp, %tmp_off
	storel %acc, %tmp_ptr
	%mj_next =l add %mj, 1
	jmp @mul_j_header
@mul_i_latch
	%mi_next =l add %mi, 1
	jmp @mul_i_header
@copy_header
	%cidx =l phi @mul_i_header 0, @copy_body %cidx_next
	%c_cond =w csltl %cidx, %nn
	jnz %c_cond, @copy_body, @copy_done
@copy_body
	%coff =l mul %cidx, 8
	%csrc =l add %tmp, %coff
	%cdst =l add %out, %coff
	%cval =l loadl %csrc
	storel %cval, %cdst
	%cidx_next =l add %cidx, 1
	jmp @copy_header
@copy_done
	%step_next =l add %step, 1
	jmp @pow_header
@zero_header
	%zidx =l phi @pow_header 0, @zero_body %zidx_next
	%z_cond =w csltl %zidx, %nn2
	jnz %z_cond, @zero_body, @done
@zero_body
	%zoff =l mul %zidx, 8
	%zptr =l add %tmp, %zoff
	storel 0, %zptr
	%zidx_next =l add %zidx, 1
	jmp @zero_header
@done
	ret 0
}
