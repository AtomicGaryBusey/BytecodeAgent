export function l $count_ugt(l %a, l %n, l %t) {
@start
	%has_elems =w csgtl %n, 0
	jnz %has_elems, @loop, @empty
@empty
	ret 0
@loop
	%i =l phi @start 0, @next %i1
	%cnt =l phi @start 0, @next %cnt1
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gt_l =l extuw %gt
	%cnt1 =l add %cnt, %gt_l
	%i1 =l add %i, 1
	%more =w csltl %i1, %n
	jnz %more, @next, @done
@next
	jmp @loop
@done
	ret %cnt1
}
