map_into:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    push rbx
    push rbp
    push r12
    push r13
    push r14
    mov rbp, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx
    xor ebx, ebx
.Lloop:
    mov rdi, qword ptr [r12]
    call r14
    mov qword ptr [rbp], rax
    cmp rax, qword ptr [r12]
    setl cl
    movzx ecx, cl
    add rbx, rcx
    add r12, 8
    add rbp, 8
    dec r13
    jnz .Lloop
    mov rax, rbx
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Ldone:
    ret
