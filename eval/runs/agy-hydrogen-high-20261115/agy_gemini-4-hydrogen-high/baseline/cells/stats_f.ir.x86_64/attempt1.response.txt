export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%p0 =l add %a, 8
	%rem0 =l sub %n, 1
	%c0 =w csgtl %rem0, 0
	jnz %c0, @loop, @done

@loop
	%p =l phi @start %p0, @loop_next %p_next
	%rem =l phi @start %rem0, @loop_next %rem_next
	%min =l phi @start %y0, @loop_next %min_next
	%max =l phi @start %y0, @loop_next %max_next
	%sum =l phi @start %y0, @loop_next %sum_next
	%odd =l phi @start %odd0, @loop_next %odd_next
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum_next =l add %sum, %y
	%is_odd =l and %y, 1
	%odd_next =l add %odd, %is_odd
	%lt =w csltl %y, %min
	jnz %lt, @upd_min, @check_max

@upd_min
	jmp @check_max

@check_max
	%min_next =l phi @loop %min, @upd_min %y
	%gt =w csgtl %y, %max
	jnz %gt, @upd_max, @loop_next

@upd_max
	jmp @loop_next

@loop_next
	%max_next =l phi @check_max %max, @upd_max %y
	%p_next =l add %p, 8
	%rem_next =l sub %rem, 1
	%c =w csgtl %rem_next, 0
	jnz %c, @loop, @done

@done
	%min_f =l phi @start %y0, @loop_next %min_next
	%max_f =l phi @start %y0, @loop_next %max_next
	%sum_f =l phi @start %y0, @loop_next %sum_next
	%odd_f =l phi @start %odd0, @loop_next %odd_next
	storel %min_f, %out
	%out1 =l add %out, 8
	storel %max_f, %out1
	%out2 =l add %out, 16
	storel %sum_f, %out2
	ret %odd_f
}
