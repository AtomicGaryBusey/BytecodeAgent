binary_search:
    mov x9, #0
    subs x10, x1, #1
    b.lt .Lnot_found
.Lloop:
    add x11, x9, x10
    lsr x11, x11, #1
    ldr x12, [x0, x11, lsl #3]
    cmp x12, x2
    b.eq .Lfound
    b.lt .Lgo_right
    sub x10, x11, #1
    cmp x9, x10
    b.le .Lloop
    b .Lnot_found
.Lgo_right:
    add x9, x11, #1
    cmp x9, x10
    b.le .Lloop
.Lnot_found:
    mov x0, #-1
    ret
.Lfound:
    mov x0, x11
    ret
