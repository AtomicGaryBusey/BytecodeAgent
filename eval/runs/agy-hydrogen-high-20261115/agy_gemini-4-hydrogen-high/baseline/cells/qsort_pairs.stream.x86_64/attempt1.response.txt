qsort_pairs:
    cmp rsi, 1
    jle .Ldone
    push rbx
    push rbp
    push r12
.Lloop:
    lea rcx, [rsi - 1]
    and rcx, -2
    mov rdx, qword ptr [rdi + rcx*8]
    mov r10, qword ptr [rdi + rcx*8 + 8]
    lea r8, [rdi - 16]
    mov r9, rsi
    shl r9, 4
    add r9, rdi
.Lpart_loop:
.Li_loop:
    add r8, 16
    mov rax, qword ptr [r8]
    cmp rax, rdx
    jl .Li_loop
    jg .Li_done
    mov rax, qword ptr [r8 + 8]
    cmp rax, r10
    jl .Li_loop
.Li_done:
.Lj_loop:
    sub r9, 16
    mov rax, qword ptr [r9]
    cmp rax, rdx
    jg .Lj_loop
    jl .Lj_done
    mov rax, qword ptr [r9 + 8]
    cmp rax, r10
    jg .Lj_loop
.Lj_done:
    cmp r8, r9
    jae .Lpart_done
    movdqu xmm0, xmmword ptr [r8]
    movdqu xmm1, xmmword ptr [r9]
    movdqu xmmword ptr [r8], xmm1
    movdqu xmmword ptr [r9], xmm0
    jmp .Lpart_loop
.Lpart_done:
    mov rcx, r9
    sub rcx, rdi
    shr rcx, 4
    add rcx, 1
    sub rsi, rcx
    add r9, 16
    cmp rcx, rsi
    jbe .Lleft_smaller
    mov rbx, rdi
    mov r12, rcx
    mov rdi, r9
    jmp .Ldo_call
.Lleft_smaller:
    mov rbx, r9
    mov r12, rsi
    mov rsi, rcx
.Ldo_call:
    call qsort_pairs
    mov rdi, rbx
    mov rsi, r12
    cmp rsi, 1
    jg .Lloop
    pop r12
    pop rbp
    pop rbx
.Ldone:
    xor eax, eax
    ret
