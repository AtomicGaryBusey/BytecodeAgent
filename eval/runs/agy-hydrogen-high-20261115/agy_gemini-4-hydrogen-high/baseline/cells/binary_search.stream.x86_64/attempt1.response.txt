binary_search:
    xor ecx, ecx
    lea r8, [rsi - 1]
.Lloop:
    cmp rcx, r8
    jg .Lnot_found
    lea r9, [rcx + r8]
    shr r9, 1
    mov r10, qword ptr [rdi + r9*8]
    cmp r10, rdx
    je .Lfound
    jl .Lless
    lea r8, [r9 - 1]
    jmp .Lloop
.Lless:
    lea rcx, [r9 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnot_found:
    mov rax, -1
    ret
