crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
    mov r8d, 0xEDB88320
.Lbyte_loop:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit_loop:
    shr eax, 1
    jnc .Lno_xor
    xor eax, r8d
.Lno_xor:
    dec ecx
    jnz .Lbit_loop
    inc rdi
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    not eax
    ret
