export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l sub %n, 1
	%c0 =w csgtl %i0, 0
	jnz %c0, @outer, @done
@outer
	%i =l phi @start %i0, @outer_latch %i1
	jmp @inner
@inner
	%j =l phi @outer 0, @inner_next %j1
	%c1 =w csltl %j, %i
	jnz %c1, @inner_body, @outer_latch
@inner_body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%cmp =w csgtl %v0, %v1
	jnz %cmp, @swap, @inner_next
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_next
@inner_next
	%j1 =l add %j, 1
	jmp @inner
@outer_latch
	%i1 =l sub %i, 1
	%c2 =w csgtl %i1, 0
	jnz %c2, @outer, @done
@done
	ret 0
}
