export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @cond
@cond
	%i =l phi @start 0, @body %i1
	%j =l phi @start %j0, @body %j1
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%off1 =l shl %i, 3
	%p1 =l add %a, %off1
	%off2 =l shl %j, 3
	%p2 =l add %a, %off2
	%v1 =l loadl %p1
	%v2 =l loadl %p2
	storel %v2, %p1
	storel %v1, %p2
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @cond
@done
	ret 0
}
