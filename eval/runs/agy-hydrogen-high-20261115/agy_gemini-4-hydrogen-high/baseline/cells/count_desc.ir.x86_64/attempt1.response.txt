export function l $count_desc(l %a, l %n) {
@start
	%c =w csgtl %n, 1
	jnz %c, @init, @ret0
@ret0
	ret 0
@init
	%prev0 =l loadl %a
	%p0 =l add %a, 8
	%rem0 =l sub %n, 1
	jmp @loop
@loop
	%p =l phi @init %p0, @loop %p1
	%prev =l phi @init %prev0, @loop %cur
	%rem =l phi @init %rem0, @loop %rem1
	%cnt =l phi @init 0, @loop %cnt1
	%cur =l loadl %p
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%p1 =l add %p, 8
	%rem1 =l sub %rem, 1
	%more =w csgtl %rem1, 0
	jnz %more, @loop, @done
@done
	ret %cnt1
}
