export function l $bubble_sort(l %a, l %n) {
@start
	%end0 =l sub %n, 1
	jmp @outer_cond
@outer_cond
	%end =l phi @start %end0, @outer_next %end1
	%oc =w csltl 0, %end
	jnz %oc, @inner_cond, @done
@inner_cond
	%j =l phi @outer_cond 0, @inner_next %j1
	%ic =w csltl %j, %end
	jnz %ic, @inner_body, @outer_next
@inner_body
	%off0 =l mul %j, 8
	%p0 =l add %a, %off0
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%swap =w csltl %v1, %v0
	jnz %swap, @do_swap, @inner_next
@do_swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_next
@inner_next
	%j1 =l add %j, 1
	jmp @inner_cond
@outer_next
	%end1 =l sub %end, 1
	jmp @outer_cond
@done
	ret 0
}
