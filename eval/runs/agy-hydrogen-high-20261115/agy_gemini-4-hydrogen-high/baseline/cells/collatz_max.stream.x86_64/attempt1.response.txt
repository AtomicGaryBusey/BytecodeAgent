collatz_max:
    xor eax, eax
    cmp rdi, 1
    jle .Ldone
    mov ecx, 2
.Louter:
    mov rdx, rcx
    xor esi, esi
.Linner:
    lea r8, [rdx + rdx*2 + 1]
    shr rdx, 1
    cmovc rdx, r8
    inc rsi
    cmp rdx, 1
    jne .Linner
    cmp rsi, rax
    cmovg rax, rsi
    inc rcx
    cmp rcx, rdi
    jle .Louter
.Ldone:
    ret
