export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%le0 =w cslel %n, 0
	jnz %le0, @done, @check_eq
@check_eq
	%eq =w ceql %dst, %src
	jnz %eq, @done, @check_dir
@check_dir
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_loop, @bwd_loop
@fwd_loop
	%i =l phi @check_dir 0, @fwd_body %i1
	%fwd_cond =w csltl %i, %n
	jnz %fwd_cond, @fwd_body, @done
@fwd_body
	%off_f =l mul %i, 8
	%s_f =l add %src, %off_f
	%d_f =l add %dst, %off_f
	%val_f =l loadl %s_f
	storel %val_f, %d_f
	%i1 =l add %i, 1
	jmp @fwd_loop
@bwd_loop
	%j =l phi @check_dir %n, @bwd_body %j1
	%bwd_cond =w csgtl %j, 0
	jnz %bwd_cond, @bwd_body, @done
@bwd_body
	%j1 =l sub %j, 1
	%off_b =l mul %j1, 8
	%s_b =l add %src, %off_b
	%d_b =l add %dst, %off_b
	%val_b =l loadl %s_b
	storel %val_b, %d_b
	jmp @bwd_loop
@done
	ret 0
}
