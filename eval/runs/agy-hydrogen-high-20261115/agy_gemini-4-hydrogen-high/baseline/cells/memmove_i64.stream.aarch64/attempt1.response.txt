memmove_i64:
    cmp x2, #0
    b.le .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.hi .Lbwd
.Lfwd:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    b.ne .Lfwd
    b .Ldone
.Lbwd:
    sub x2, x2, #1
    ldr x3, [x1, x2, lsl #3]
    str x3, [x0, x2, lsl #3]
    cbnz x2, .Lbwd
.Ldone:
    mov x0, #0
    ret
