export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%has_n =w csgtl %n, 0
	jnz %has_n, @zero_loop, @ret_zero

@zero_loop
	%zi =l phi @start 0, @zero_loop %zi_next
	%zoff =l shl %zi, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zi_next =l add %zi, 1
	%zcond =w csltl %zi_next, %n
	jnz %zcond, @zero_loop, @check_nonzero

@check_nonzero
	%na_nz =w csgtl %na, 0
	jnz %na_nz, @check_nb, @ret_zero

@check_nb
	%nb_nz =w csgtl %nb, 0
	jnz %nb_nz, @outer_loop, @ret_zero

@outer_loop
	%i =l phi @check_nb 0, @outer_next %i_next
	%aoff =l shl %i, 3
	%aptr =l add %a, %aoff
	%ai =l loadl %aptr
	%ai_lo =l and %ai, 4294967295
	%ai_hi =l shr %ai, 32
	jmp @inner_loop

@inner_loop
	%j =l phi @outer_loop 0, @inner_loop %j_next
	%carry =l phi @outer_loop 0, @inner_loop %carry_next
	%boff =l shl %j, 3
	%bptr =l add %b, %boff
	%bj =l loadl %bptr
	%bj_lo =l and %bj, 4294967295
	%bj_hi =l shr %bj, 32

	%p00 =l mul %ai_lo, %bj_lo
	%p01 =l mul %ai_lo, %bj_hi
	%p10 =l mul %ai_hi, %bj_lo
	%p11 =l mul %ai_hi, %bj_hi

	%p00_lo =l and %p00, 4294967295
	%p00_hi =l shr %p00, 32

	%t1 =l add %p00_hi, %p01
	%t2 =l add %t1, %p10
	%cmid_w =w cultl %t2, %t1
	%cmid =l extuw %cmid_w
	%cmid_hi =l shl %cmid, 32

	%t2_lo =l and %t2, 4294967295
	%t2_lo_sh =l shl %t2_lo, 32
	%lo =l add %t2_lo_sh, %p00_lo

	%t2_hi =l shr %t2, 32
	%hi1 =l add %p11, %t2_hi
	%hi =l add %hi1, %cmid_hi

	%ij =l add %i, %j
	%roff =l shl %ij, 3
	%rptr =l add %r, %roff
	%rij =l loadl %rptr

	%sum1 =l add %rij, %lo
	%c1_w =w cultl %sum1, %rij
	%c1 =l extuw %c1_w

	%sum2 =l add %sum1, %carry
	%c2_w =w cultl %sum2, %sum1
	%c2 =l extuw %c2_w

	storel %sum2, %rptr

	%hi_c1 =l add %hi, %c1
	%carry_next =l add %hi_c1, %c2

	%j_next =l add %j, 1
	%jcond =w csltl %j_next, %nb
	jnz %jcond, @inner_loop, @outer_next

@outer_next
	%inb =l add %i, %nb
	%rnext_off =l shl %inb, 3
	%rnext_ptr =l add %r, %rnext_off
	storel %carry_next, %rnext_ptr
	%i_next =l add %i, 1
	%icond =w csltl %i_next, %na
	jnz %icond, @outer_loop, @trim_loop

@trim_loop
	%k =l phi @outer_next %n, @trim_step %k_prev
	%k_pos =w csgtl %k, 0
	jnz %k_pos, @trim_step, @ret_k

@trim_step
	%k_prev =l sub %k, 1
	%koff =l shl %k_prev, 3
	%kptr =l add %r, %koff
	%kval =l loadl %kptr
	%kval_nz =w cnel %kval, 0
	jnz %kval_nz, @ret_k, @trim_loop

@ret_k
	ret %k

@ret_zero
	ret 0
}
