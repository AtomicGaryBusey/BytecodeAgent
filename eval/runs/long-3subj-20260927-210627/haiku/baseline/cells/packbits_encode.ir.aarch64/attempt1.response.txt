export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i = copy 0
    %out_pos = copy 0
    jmp @loop

@loop
    %i_lt_n = csltl %i, %n
    jnz %i_lt_n, @loop_body, @end

@loop_body
    %src_i = add %src, %i
    %byte_i =w loadub %src_i

    %run_len = copy 1
    %run_j = add %i, 1

@count_run
    %run_j_lt_n = csltl %run_j, %n
    %run_len_lt_128 = csltl %run_len, 128
    %can_count_run = and %run_j_lt_n, %run_len_lt_128
    jnz %can_count_run, @get_byte_run, @run_done

@get_byte_run
    %src_runj = add %src, %run_j
    %byte_runj =w loadub %src_runj
    %bytes_eq = ceql %byte_i, %byte_runj
    jnz %bytes_eq, @inc_run, @run_done

@inc_run
    %run_len = add %run_len, 1
    %run_j = add %run_j, 1
    jmp @count_run

@run_done
    %run_ge_3 = csltl 2, %run_len
    jnz %run_ge_3, @emit_repeat, @emit_literal_start

@emit_repeat
    %available = sub %cap, %out_pos
    %need = copy 2
    %fits = cslel %need, %available
    jnz %fits, @write_repeat, @return_neg1

@write_repeat
    %header_val = sub 257, %run_len
    %header =w trunc %header_val
    %out_addr_h = add %out, %out_pos
    storeb %header, %out_addr_h
    %out_pos = add %out_pos, 1
    %out_addr_d = add %out, %out_pos
    storeb %byte_i, %out_addr_d
    %out_pos = add %out_pos, 1
    %i = add %i, %run_len
    jmp @loop

@emit_literal_start
    %lit_start = copy %i
    %lit_len = copy 0
    %lit_j = copy %i
    jmp @lit_loop

@lit_loop
    %lit_j_lt_n = csltl %lit_j, %n
    %lit_len_lt_128 = csltl %lit_len, 128
    %can_continue_lit = and %lit_j_lt_n, %lit_len_lt_128
    jnz %can_continue_lit, @check_lit_run_len, @lit_done

@check_lit_run_len
    %src_litj = add %src, %lit_j
    %byte_litj =w loadub %src_litj
    %lit_run_len = copy 1
    %lit_run_j = add %lit_j, 1
    jmp @count_lit_run

@count_lit_run
    %lit_run_j_lt_n = csltl %lit_run_j, %n
    %lit_run_len_lt_128 = csltl %lit_run_len, 128
    %can_count_lit_run = and %lit_run_j_lt_n, %lit_run_len_lt_128
    jnz %can_count_lit_run, @get_byte_lit_run, @lit_run_done

@get_byte_lit_run
    %src_lit_runj = add %src, %lit_run_j
    %byte_lit_runj =w loadub %src_lit_runj
    %lit_bytes_eq = ceql %byte_litj, %byte_lit_runj
    jnz %lit_bytes_eq, @inc_lit_run, @lit_run_done

@inc_lit_run
    %lit_run_len = add %lit_run_len, 1
    %lit_run_j = add %lit_run_j, 1
    jmp @count_lit_run

@lit_run_done
    %lit_run_ge_3 = csltl 2, %lit_run_len
    jnz %lit_run_ge_3, @lit_done, @add_to_literal

@add_to_literal
    %lit_len = add %lit_len, 1
    %lit_j = add %lit_j, 1
    jmp @lit_loop

@lit_done
    %need_lit = add 1, %lit_len
    %avail_lit = sub %cap, %out_pos
    %fits_lit = cslel %need_lit, %avail_lit
    jnz %fits_lit, @write_literal, @return_neg1

@write_literal
    %header_lit_val = sub %lit_len, 1
    %header_lit =w trunc %header_lit_val
    %out_addr_hdr = add %out, %out_pos
    storeb %header_lit, %out_addr_hdr
    %out_pos = add %out_pos, 1
    %copy_idx = copy 0
    jmp @copy_loop

@copy_loop
    %can_copy = csltl %copy_idx, %lit_len
    jnz %can_copy, @do_copy, @copy_done

@do_copy
    %offset = add %lit_start, %copy_idx
    %src_addr = add %src, %offset
    %byte_copy =w loadub %src_addr
    %out_addr = add %out, %out_pos
    storeb %byte_copy, %out_addr
    %out_pos = add %out_pos, 1
    %copy_idx = add %copy_idx, 1
    jmp @copy_loop

@copy_done
    %i = add %lit_start, %lit_len
    jmp @loop

@return_neg1
    ret -1

@end
    ret %out_pos
}
