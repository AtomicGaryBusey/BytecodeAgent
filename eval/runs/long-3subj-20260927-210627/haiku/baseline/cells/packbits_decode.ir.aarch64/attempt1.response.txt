export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @packet_start
    
@packet_start
    %src_pos =l phi @entry 0, @noop_end %src_pos_final, @repeat_end %src_pos_final, @literal_end %src_pos_final
    %out_pos =l phi @entry 0, @noop_end %out_pos_final, @repeat_end %out_pos_final, @literal_end %out_pos_final
    
    %at_end =w ceql %src_pos, %n
    jnz %at_end, @return_result
    
    %hdr_ptr =l add %src, %src_pos
    %h =w loadsb %hdr_ptr
    %src_pos_new =l add %src_pos, 1
    
    %is_noop =w ceql %h, -128
    jnz %is_noop, @noop_end
    
    %is_literal =w csgel %h, 0
    jnz %is_literal, @literal_start
    
    %has_byte =w csltl %src_pos_new, %n
    jz %has_byte, @src_error
    
    %rbyte_ptr =l add %src, %src_pos_new
    %rbyte =w loadub %rbyte_ptr
    %h64 =l sext %h
    %repeat_count_init =l sub 1, %h64
    
    jmp @repeat_init
    
@repeat_init
    %rout_pos_init =l copy %out_pos
    jmp @repeat_loop
    
@repeat_loop
    %rout_pos =l phi @repeat_init %rout_pos_init, @repeat_continue %rout_pos_new
    %rcount =l phi @repeat_init %repeat_count_init, @repeat_continue %rcount_new
    
    %r_done =w ceql %rcount, 0
    jnz %r_done, @repeat_end
    
    %r_can_write =w csltl %rout_pos, %cap
    jz %r_can_write, @repeat_continue
    
    %r_out_ptr =l add %out, %rout_pos
    storeb %rbyte, %r_out_ptr
    
@repeat_continue
    %rout_pos_new =l add %rout_pos, 1
    %rcount_new =l sub %rcount, 1
    jmp @repeat_loop
    
@repeat_end
    %src_pos_final =l add %src_pos_new, 1
    %out_pos_final =l copy %rout_pos
    jmp @packet_start
    
@noop_end
    %src_pos_final =l copy %src_pos_new
    %out_pos_final =l copy %out_pos
    jmp @packet_start
    
@literal_start
    %h64_lit =l sext %h
    %lit_count_init =l add %h64_lit, 1
    
    %lit_end =l add %src_pos_new, %lit_count_init
    %no_data =w csgtl %lit_end, %n
    jnz %no_data, @src_error
    
    jmp @literal_init
    
@literal_init
    %lout_pos_init =l copy %out_pos
    %lsrc_pos_init =l copy %src_pos_new
    %lit_idx_init =l copy 0
    jmp @literal_loop
    
@literal_loop
    %lout_pos =l phi @literal_init %lout_pos_init, @literal_continue %lout_pos_new
    %lsrc_pos =l phi @literal_init %lsrc_pos_init, @literal_continue %lsrc_pos_new
    %lit_idx =l phi @literal_init %lit_idx_init, @literal_continue %lit_idx_new
    
    %l_done =w ceql %lit_idx, %lit_count_init
    jnz %l_done, @literal_end
    
    %lbyte_ptr =l add %src, %lsrc_pos
    %lbyte =w loadub %lbyte_ptr
    %lsrc_pos_new =l add %lsrc_pos, 1
    
    %l_can_write =w csltl %lout_pos, %cap
    jz %l_can_write, @literal_continue
    
    %l_out_ptr =l add %out, %lout_pos
    storeb %lbyte, %l_out_ptr
    
@literal_continue
    %lout_pos_new =l add %lout_pos, 1
    %lit_idx_new =l add %lit_idx, 1
    jmp @literal_loop
    
@literal_end
    %src_pos_final =l copy %lsrc_pos
    %out_pos_final =l copy %lout_pos
    jmp @packet_start
    
@return_result
    %overflow =w csgtl %out_pos, %cap
    jnz %overflow, @overflow_error
    
    ret %out_pos
    
@overflow_error
    ret -1
    
@src_error
    ret -2
}
