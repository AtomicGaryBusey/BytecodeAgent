export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @packet_loop

@packet_loop
    %src_pos =l phi @entry 0, @next_packet %next_src_pos
    %out_pos =l phi @entry 0, @next_packet %next_out_pos
    %at_end =w ceq %src_pos, %n
    jnz %at_end, @finish
    %hdr_addr =l add %src, %src_pos
    %h =w loadsb %hdr_addr
    %is_noop =w ceq %h, -128
    jnz %is_noop, @noop
    %is_lit =w csge %h, 0
    jnz %is_lit, @literal
    jmp @repeat

@noop
    %noop_src =l add %src_pos, 1
    %noop_out =l add %out_pos, 0
    jmp @next_packet

@repeat
    %rep_src_idx =l add %src_pos, 1
    %has_rep =w cslt %rep_src_idx, %n
    jz %has_rep, @error_src
    %rep_addr =l add %src, %rep_src_idx
    %rep_byte =w loadub %rep_addr
    %h_long =l extsw %h
    %rep_cnt =l sub 1, %h_long
    jmp @rep_write_loop

@rep_write_loop
    %rep_i =l phi @repeat 0, @rep_write_loop %rep_i_next
    %rep_out =l phi @repeat %out_pos, @rep_write_loop %rep_out_next
    %rep_done =w ceq %rep_i, %rep_cnt
    jnz %rep_done, @rep_end
    %rep_can_w =w cslt %rep_out, %cap
    jz %rep_can_w, @rep_skip_w
    %rep_addr_w =l add %out, %rep_out
    storeb %rep_byte, %rep_addr_w

@rep_skip_w
    %rep_out_next =l add %rep_out, 1
    %rep_i_next =l add %rep_i, 1
    jmp @rep_write_loop

@rep_end
    %rep_src_final =l add %src_pos, 2
    %rep_out_final =l add %rep_out, 0
    jmp @next_packet

@literal
    %h_long_l =l extsw %h
    %lit_cnt =l add %h_long_l, 1
    %lit_src_start =l add %src_pos, 1
    %lit_check =l add %lit_src_start, %lit_cnt
    %exceeds =w csgt %lit_check, %n
    jnz %exceeds, @error_src
    jmp @lit_write_loop

@lit_write_loop
    %lit_i =l phi @literal 0, @lit_write_loop %lit_i_next
    %lit_out =l phi @literal %out_pos, @lit_write_loop %lit_out_next
    %lit_done =w ceq %lit_i, %lit_cnt
    jnz %lit_done, @lit_end
    %lit_src =l add %lit_src_start, %lit_i
    %lit_addr =l add %src, %lit_src
    %lit_byte =w loadub %lit_addr
    %lit_can_w =w cslt %lit_out, %cap
    jz %lit_can_w, @lit_skip_w
    %lit_addr_w =l add %out, %lit_out
    storeb %lit_byte, %lit_addr_w

@lit_skip_w
    %lit_out_next =l add %lit_out, 1
    %lit_i_next =l add %lit_i, 1
    jmp @lit_write_loop

@lit_end
    %lit_final_src =l add %lit_src_start, %lit_cnt
    %lit_out_final =l add %lit_out, 0
    jmp @next_packet

@next_packet
    %next_src_pos =l phi @noop %noop_src, @rep_end %rep_src_final, @lit_end %lit_final_src
    %next_out_pos =l phi @noop %noop_out, @rep_end %rep_out_final, @lit_end %lit_out_final
    jmp @packet_loop

@finish
    %chk_ovf =w csgt %out_pos, %cap
    jnz %chk_ovf, @error_ovf
    ret %out_pos

@error_ovf
    ret -1

@error_src
    ret -2
}
