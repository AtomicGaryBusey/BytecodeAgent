export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    jmp @main_loop

@main_loop
    %pc = phi l @entry 0, @next_iter %pc_new
    %sp = phi l @entry 0, @next_iter %sp_new
    
    %cond = csltl %pc, %len
    jnz %cond, @load_instr
    jmp @cleanup_start

@load_instr
    %pc_bytes = mul %pc, 8
    %instr_ptr = add %code, %pc_bytes
    %instr = loadl %instr_ptr
    
    %is_0 = ceql %instr, 0
    jnz %is_0, @cleanup_start
    
    %is_1 = ceql %instr, 1
    jnz %is_1, @opcode_push
    
    %is_2 = ceql %instr, 2
    jnz %is_2, @opcode_add
    
    %is_3 = ceql %instr, 3
    jnz %is_3, @opcode_sub
    
    %is_4 = ceql %instr, 4
    jnz %is_4, @opcode_mul
    
    %is_5 = ceql %instr, 5
    jnz %is_5, @opcode_dup
    
    %is_6 = ceql %instr, 6
    jnz %is_6, @opcode_swap
    
    %is_7 = ceql %instr, 7
    jnz %is_7, @opcode_drop
    
    %is_8 = ceql %instr, 8
    jnz %is_8, @opcode_over
    
    %is_9 = ceql %instr, 9
    jnz %is_9, @opcode_lt
    
    %is_10 = ceql %instr, 10
    jnz %is_10, @opcode_jz
    
    %is_11 = ceql %instr, 11
    jnz %is_11, @opcode_jmp
    
    %pc_new = add %pc, 1
    %sp_new = %sp
    jmp @next_iter

@opcode_push
    %pc_plus_1 = add %pc, 1
    %idx = mul %pc_plus_1, 8
    %ptr = add %code, %idx
    %val = loadl %ptr
    
    %stack_idx = mul %sp, 8
    %stack_ptr = add %stack, %stack_idx
    storel %val, %stack_ptr
    
    %pc_new = add %pc, 2
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_add
    %sp_minus_1 = sub %sp, 1
    %idx_b = mul %sp_minus_1, 8
    %ptr_b = add %stack, %idx_b
    %b = loadl %ptr_b
    
    %sp_minus_2 = sub %sp, 2
    %idx_a = mul %sp_minus_2, 8
    %ptr_a = add %stack, %idx_a
    %a = loadl %ptr_a
    
    %sum = add %a, %b
    storel %sum, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_minus_1
    jmp @next_iter

@opcode_sub
    %sp_minus_1 = sub %sp, 1
    %idx_b = mul %sp_minus_1, 8
    %ptr_b = add %stack, %idx_b
    %b = loadl %ptr_b
    
    %sp_minus_2 = sub %sp, 2
    %idx_a = mul %sp_minus_2, 8
    %ptr_a = add %stack, %idx_a
    %a = loadl %ptr_a
    
    %diff = sub %a, %b
    storel %diff, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_minus_1
    jmp @next_iter

@opcode_mul
    %sp_minus_1 = sub %sp, 1
    %idx_b = mul %sp_minus_1, 8
    %ptr_b = add %stack, %idx_b
    %b = loadl %ptr_b
    
    %sp_minus_2 = sub %sp, 2
    %idx_a = mul %sp_minus_2, 8
    %ptr_a = add %stack, %idx_a
    %a = loadl %ptr_a
    
    %prod = mul %a, %b
    storel %prod, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_minus_1
    jmp @next_iter

@opcode_dup
    %sp_minus_1 = sub %sp, 1
    %idx = mul %sp_minus_1, 8
    %ptr = add %stack, %idx
    %val = loadl %ptr
    
    %stack_idx = mul %sp, 8
    %stack_ptr = add %stack, %stack_idx
    storel %val, %stack_ptr
    
    %pc_new = add %pc, 1
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_swap
    %sp_minus_1 = sub %sp, 1
    %sp_minus_2 = sub %sp, 2
    
    %idx_1 = mul %sp_minus_1, 8
    %ptr_1 = add %stack, %idx_1
    %val_1 = loadl %ptr_1
    
    %idx_2 = mul %sp_minus_2, 8
    %ptr_2 = add %stack, %idx_2
    %val_2 = loadl %ptr_2
    
    storel %val_2, %ptr_1
    storel %val_1, %ptr_2
    
    %pc_new = add %pc, 1
    %sp_new = %sp
    jmp @next_iter

@opcode_drop
    %pc_new = add %pc, 1
    %sp_new = sub %sp, 1
    jmp @next_iter

@opcode_over
    %sp_minus_2 = sub %sp, 2
    %idx = mul %sp_minus_2, 8
    %ptr = add %stack, %idx
    %val = loadl %ptr
    
    %stack_idx = mul %sp, 8
    %stack_ptr = add %stack, %stack_idx
    storel %val, %stack_ptr
    
    %pc_new = add %pc, 1
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_lt
    %sp_minus_1 = sub %sp, 1
    %idx_b = mul %sp_minus_1, 8
    %ptr_b = add %stack, %idx_b
    %b = loadl %ptr_b
    
    %sp_minus_2 = sub %sp, 2
    %idx_a = mul %sp_minus_2, 8
    %ptr_a = add %stack, %idx_a
    %a = loadl %ptr_a
    
    %cmp = csltl %a, %b
    %result = zext %cmp
    storel %result, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_minus_1
    jmp @next_iter

@opcode_jz
    %sp_minus_1 = sub %sp, 1
    %idx = mul %sp_minus_1, 8
    %ptr = add %stack, %idx
    %val = loadl %ptr
    
    %pc_plus_1 = add %pc, 1
    %idx_operand = mul %pc_plus_1, 8
    %ptr_operand = add %code, %idx_operand
    %target = loadl %ptr_operand
    
    %is_zero = ceql %val, 0
    jnz %is_zero, @jz_jump
    
    %pc_new = add %pc, 2
    %sp_new = %sp_minus_1
    jmp @next_iter

@jz_jump
    %pc_new = %target
    %sp_new = %sp_minus_1
    jmp @next_iter

@opcode_jmp
    %pc_plus_1 = add %pc, 1
    %idx_operand = mul %pc_plus_1, 8
    %ptr_operand = add %code, %idx_operand
    %target = loadl %ptr_operand
    
    %pc_new = %target
    %sp_new = %sp
    jmp @next_iter

@next_iter
    jmp @main_loop

@cleanup_start
    jmp @clear_loop

@clear_loop
    %clear_idx = phi l @cleanup_start %sp, @clear_next %clear_idx_next
    
    %cond = csltl %clear_idx, %cap
    jnz %cond, @clear_entry
    jmp @return_value

@clear_entry
    %idx_bytes = mul %clear_idx, 8
    %ptr = add %stack, %idx_bytes
    storel 0, %ptr
    jmp @clear_next

@clear_next
    %clear_idx_next = add %clear_idx, 1
    jmp @clear_loop

@return_value
    %sp_gt_0 = csltl 0, %sp
    jnz %sp_gt_0, @return_top
    ret 0

@return_top
    %top_idx = sub %sp, 1
    %top_idx_bytes = mul %top_idx, 8
    %top_ptr = add %stack, %top_idx_bytes
    %result = loadl %top_ptr
    ret %result
}
