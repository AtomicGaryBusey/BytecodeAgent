export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
    jmp @main_loop

@main_loop
    %pc = phi l 0, %pc_new
    %sp = phi l 0, %sp_new
    
    %pc_lt_len = cltl %pc, %len
    jnz %pc_lt_len, @load_instr
    jmp @cleanup_start

@load_instr
    %pc_bytes = mul %pc, 8
    %instr_ptr = add %code, %pc_bytes
    %instr = loadl %instr_ptr
    
    %is_halt = ceql %instr, 0
    jnz %is_halt, @cleanup_start
    
    %is_push = ceql %instr, 1
    jnz %is_push, @opcode_push
    
    %is_add = ceql %instr, 2
    jnz %is_add, @opcode_add
    
    %is_sub = ceql %instr, 3
    jnz %is_sub, @opcode_sub
    
    %is_mul = ceql %instr, 4
    jnz %is_mul, @opcode_mul
    
    %is_dup = ceql %instr, 5
    jnz %is_dup, @opcode_dup
    
    %is_swap = ceql %instr, 6
    jnz %is_swap, @opcode_swap
    
    %is_drop = ceql %instr, 7
    jnz %is_drop, @opcode_drop
    
    %is_over = ceql %instr, 8
    jnz %is_over, @opcode_over
    
    %is_lt = ceql %instr, 9
    jnz %is_lt, @opcode_lt
    
    %is_jz = ceql %instr, 10
    jnz %is_jz, @opcode_jz
    
    %is_jmp = ceql %instr, 11
    jnz %is_jmp, @opcode_jmp
    
    %pc_new = add %pc, 1
    %sp_new = %sp
    jmp @next_iter

@opcode_push
    %pc_plus_1 = add %pc, 1
    %operand_offset = mul %pc_plus_1, 8
    %operand_ptr = add %code, %operand_offset
    %operand = loadl %operand_ptr
    
    %stack_offset = mul %sp, 8
    %stack_ptr = add %stack, %stack_offset
    storel %operand, %stack_ptr
    
    %pc_new = add %pc, 2
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_add
    %sp_m1 = sub %sp, 1
    %sp_m2 = sub %sp, 2
    
    %off_b = mul %sp_m1, 8
    %ptr_b = add %stack, %off_b
    %b = loadl %ptr_b
    
    %off_a = mul %sp_m2, 8
    %ptr_a = add %stack, %off_a
    %a = loadl %ptr_a
    
    %sum = add %a, %b
    storel %sum, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_m1
    jmp @next_iter

@opcode_sub
    %sp_m1 = sub %sp, 1
    %sp_m2 = sub %sp, 2
    
    %off_b = mul %sp_m1, 8
    %ptr_b = add %stack, %off_b
    %b = loadl %ptr_b
    
    %off_a = mul %sp_m2, 8
    %ptr_a = add %stack, %off_a
    %a = loadl %ptr_a
    
    %diff = sub %a, %b
    storel %diff, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_m1
    jmp @next_iter

@opcode_mul
    %sp_m1 = sub %sp, 1
    %sp_m2 = sub %sp, 2
    
    %off_b = mul %sp_m1, 8
    %ptr_b = add %stack, %off_b
    %b = loadl %ptr_b
    
    %off_a = mul %sp_m2, 8
    %ptr_a = add %stack, %off_a
    %a = loadl %ptr_a
    
    %prod = mul %a, %b
    storel %prod, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_m1
    jmp @next_iter

@opcode_dup
    %sp_m1 = sub %sp, 1
    
    %off = mul %sp_m1, 8
    %ptr = add %stack, %off
    %val = loadl %ptr
    
    %new_off = mul %sp, 8
    %new_ptr = add %stack, %new_off
    storel %val, %new_ptr
    
    %pc_new = add %pc, 1
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_swap
    %sp_m1 = sub %sp, 1
    %sp_m2 = sub %sp, 2
    
    %off_1 = mul %sp_m1, 8
    %ptr_1 = add %stack, %off_1
    %val_1 = loadl %ptr_1
    
    %off_2 = mul %sp_m2, 8
    %ptr_2 = add %stack, %off_2
    %val_2 = loadl %ptr_2
    
    storel %val_2, %ptr_1
    storel %val_1, %ptr_2
    
    %pc_new = add %pc, 1
    %sp_new = %sp
    jmp @next_iter

@opcode_drop
    %pc_new = add %pc, 1
    %sp_new = sub %sp, 1
    jmp @next_iter

@opcode_over
    %sp_m2 = sub %sp, 2
    
    %off = mul %sp_m2, 8
    %ptr = add %stack, %off
    %val = loadl %ptr
    
    %new_off = mul %sp, 8
    %new_ptr = add %stack, %new_off
    storel %val, %new_ptr
    
    %pc_new = add %pc, 1
    %sp_new = add %sp, 1
    jmp @next_iter

@opcode_lt
    %sp_m1 = sub %sp, 1
    %sp_m2 = sub %sp, 2
    
    %off_b = mul %sp_m1, 8
    %ptr_b = add %stack, %off_b
    %b = loadl %ptr_b
    
    %off_a = mul %sp_m2, 8
    %ptr_a = add %stack, %off_a
    %a = loadl %ptr_a
    
    %cmp = cltl %a, %b
    storel %cmp, %ptr_a
    
    %pc_new = add %pc, 1
    %sp_new = %sp_m1
    jmp @next_iter

@opcode_jz
    %sp_m1 = sub %sp, 1
    
    %off = mul %sp_m1, 8
    %ptr = add %stack, %off
    %val = loadl %ptr
    
    %pc_plus_1 = add %pc, 1
    %operand_offset = mul %pc_plus_1, 8
    %operand_ptr = add %code, %operand_offset
    %target = loadl %operand_ptr
    
    %is_zero = ceql %val, 0
    jnz %is_zero, @jz_take_jump
    
    %pc_new = add %pc, 2
    %sp_new = %sp_m1
    jmp @next_iter

@jz_take_jump
    %pc_new = %target
    %sp_new = %sp_m1
    jmp @next_iter

@opcode_jmp
    %pc_plus_1 = add %pc, 1
    %operand_offset = mul %pc_plus_1, 8
    %operand_ptr = add %code, %operand_offset
    %target = loadl %operand_ptr
    
    %pc_new = %target
    %sp_new = %sp
    jmp @next_iter

@next_iter
    jmp @main_loop

@cleanup_start
    jmp @clear_loop

@clear_loop
    %idx = phi l %sp, %idx_next
    
    %idx_lt_cap = cltl %idx, %cap
    jnz %idx_lt_cap, @clear_entry
    jmp @return_value

@clear_entry
    %off = mul %idx, 8
    %ptr = add %stack, %off
    storel 0, %ptr
    jmp @clear_next

@clear_next
    %idx_next = add %idx, 1
    jmp @clear_loop

@return_value
    %sp_gt_0 = cltl 0, %sp
    jnz %sp_gt_0, @return_top
    ret 0

@return_top
    %top_idx = sub %sp, 1
    %top_offset = mul %top_idx, 8
    %top_ptr = add %stack, %top_offset
    %result = loadl %top_ptr
    ret %result
}
