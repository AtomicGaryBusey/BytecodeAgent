export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %i_ptr =l alloc8 8
    %j_ptr =l alloc8 8
    %diag_ptr =l alloc8 8
    %up_ptr =l alloc8 8
    storel 0, %j_ptr
    jmp @init_loop

@init_loop
    %j_val =l loadl %j_ptr
    %n_plus_1 =l add %n, 1
    %cond =w csltl %j_val, %n_plus_1
    jnz %cond, @init_body, @init_done

@init_body
    %j_val =l loadl %j_ptr
    %offset =l mul %j_val, 8
    %addr =l add %row, %offset
    storel 0, %addr
    %j_next =l add %j_val, 1
    storel %j_next, %j_ptr
    jmp @init_loop

@init_done
    storel 0, %i_ptr
    jmp @main_loop

@main_loop
    %i_val =l loadl %i_ptr
    %main_cond =w csltl %i_val, %m
    jnz %main_cond, @main_body, @main_done

@main_body
    storel 0, %diag_ptr
    storel 1, %j_ptr
    jmp @inner_loop

@inner_loop
    %j_val =l loadl %j_ptr
    %n_plus_1 =l add %n, 1
    %inner_cond =w csltl %j_val, %n_plus_1
    jnz %inner_cond, @inner_body, @inner_done

@inner_body
    %j_val =l loadl %j_ptr
    %offset =l mul %j_val, 8
    %addr_j =l add %row, %offset
    %up_val =l loadl %addr_j
    storel %up_val, %up_ptr

    %i_val =l loadl %i_ptr
    %x_addr =l add %x, %i_val
    %x_val =w loadub %x_addr

    %j_val =l loadl %j_ptr
    %j_minus_1 =l sub %j_val, 1
    %y_addr =l add %y, %j_minus_1
    %y_val =w loadub %y_addr

    %x_val_ext =l extuw %x_val
    %y_val_ext =l extuw %y_val
    %eq =w ceql %x_val_ext, %y_val_ext
    jnz %eq, @match, @nomatch

@match
    %j_val =l loadl %j_ptr
    %diag_val =l loadl %diag_ptr
    %new_val =l add %diag_val, 1
    %offset =l mul %j_val, 8
    %addr_j =l add %row, %offset
    storel %new_val, %addr_j
    %up_val =l loadl %up_ptr
    storel %up_val, %diag_ptr
    %j_next =l add %j_val, 1
    storel %j_next, %j_ptr
    jmp @inner_loop

@nomatch
    %j_val =l loadl %j_ptr
    %j_minus_1 =l sub %j_val, 1
    %offset_jm1 =l mul %j_minus_1, 8
    %addr_jm1 =l add %row, %offset_jm1
    %row_jm1 =l loadl %addr_jm1

    %up_val =l loadl %up_ptr
    %ge =w csgtl %up_val, %row_jm1
    jnz %ge, @use_up, @use_jm1

@use_up
    %j_val =l loadl %j_ptr
    %up_val =l loadl %up_ptr
    %offset =l mul %j_val, 8
    %addr_j =l add %row, %offset
    storel %up_val, %addr_j
    storel %up_val, %diag_ptr
    %j_next =l add %j_val, 1
    storel %j_next, %j_ptr
    jmp @inner_loop

@use_jm1
    %j_val =l loadl %j_ptr
    %j_minus_1 =l sub %j_val, 1
    %offset_jm1 =l mul %j_minus_1, 8
    %addr_jm1 =l add %row, %offset_jm1
    %row_jm1 =l loadl %addr_jm1
    %offset =l mul %j_val, 8
    %addr_j =l add %row, %offset
    storel %row_jm1, %addr_j
    %up_val =l loadl %up_ptr
    storel %up_val, %diag_ptr
    %j_next =l add %j_val, 1
    storel %j_next, %j_ptr
    jmp @inner_loop

@inner_done
    %i_val =l loadl %i_ptr
    %i_next =l add %i_val, 1
    storel %i_next, %i_ptr
    jmp @main_loop

@main_done
    %offset =l mul %n, 8
    %addr_n =l add %row, %offset
    %result =l loadl %addr_n
    ret %result
}
