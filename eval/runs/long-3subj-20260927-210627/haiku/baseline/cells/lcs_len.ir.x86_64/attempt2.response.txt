export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %i = alloc8 8
    %j = alloc8 8
    %diag = alloc8 8
    %up = alloc8 8
    storel 0, %j
    
@init_loop
    %j_val = loadl %j
    %n_plus_1 = add %n, 1
    %cond = csltl %j_val, %n_plus_1
    jnz %cond, @init_body, @init_done
    
@init_body
    %j_val = loadl %j
    %offset = mul %j_val, 8
    %addr = add %row, %offset
    storel 0, %addr
    %j_next = add %j_val, 1
    storel %j_next, %j
    jmp @init_loop
    
@init_done
    storel 0, %i
    
@main_loop
    %i_val = loadl %i
    %main_cond = csltl %i_val, %m
    jnz %main_cond, @main_body, @main_done
    
@main_body
    storel 0, %diag
    storel 1, %j
    jmp @inner_loop
    
@inner_loop
    %j_val = loadl %j
    %n_plus_1 = add %n, 1
    %inner_cond = csltl %j_val, %n_plus_1
    jnz %inner_cond, @inner_body, @inner_done
    
@inner_body
    %j_val = loadl %j
    %offset = mul %j_val, 8
    %addr_j = add %row, %offset
    %up_val = loadl %addr_j
    storel %up_val, %up
    
    %i_val = loadl %i
    %x_addr = add %x, %i_val
    %x_val =w loadub %x_addr
    %x_val_ext =l extuw %x_val
    
    %j_val = loadl %j
    %j_minus_1 = sub %j_val, 1
    %y_addr = add %y, %j_minus_1
    %y_val =w loadub %y_addr
    %y_val_ext =l extuw %y_val
    
    %eq = ceql %x_val_ext, %y_val_ext
    jnz %eq, @match, @nomatch
    
@match
    %j_val = loadl %j
    %diag_val = loadl %diag
    %new_val = add %diag_val, 1
    %offset = mul %j_val, 8
    %addr_j = add %row, %offset
    storel %new_val, %addr_j
    %up_val = loadl %up
    storel %up_val, %diag
    %j_next = add %j_val, 1

    %addr_jm1 = add %row, %offset_jm1
    %row_jm1 = loadl %addr_jm1
    
    %up_val = loadl %up
    %ge = csgtl %up_val, %row_jm1
    jnz %ge, @use_up, @use_jm1
    
@use_up
    %j_val = loadl %j
    %up_val = loadl %up
    %offset = mul %j_val, 8
    %addr_j = add %row, %offset
    storel %up_val, %addr_j
    storel %up_val, %diag
    %j_next = add %j_val, 1
    storel %j_next, %j
    jmp @inner_loop
    
@use_jm1
    %j_val = loadl %j
    %j_minus_1 = sub %j_val, 1
    %offset_jm1 = mul %j_minus_1, 8
    %addr_jm1 = add %row, %offset_jm1
    %row_jm1 = loadl %addr_jm1
    %offset = mul %j_val, 8
    %addr_j = add %row, %offset
    storel %row_jm1, %addr_j
    %up_val = loadl %up
    storel %up_val, %diag
    %j_next = add %j_val, 1
    storel %j_next, %j
    jmp @inner_loop
    
@inner_done
    %i_val = loadl %i
    %i_next = add %i_val, 1
    storel %i_next, %i
    jmp @main_loop
    
@main_done
    %offset = mul %n, 8
    %addr_n = add %row, %offset
    %result = loadl %addr_n
    ret %result
}
