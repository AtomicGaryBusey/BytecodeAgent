export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    jmp @init_start

@init_start
    %j = phi l 0 @entry, %j_next @init_loop
    %cond =w cslel %j %n
    jnz %cond @init_loop @i_loop_start

@init_loop
    %offset = mull %j 8
    %addr = addl %row %offset
    storel 0 %addr
    %j_next = addl %j 1
    jmp @init_start

@i_loop_start
    jmp @i_loop

@i_loop
    %i = phi l 0 @i_loop_start, %i_next @i_loop_next
    %i_cond =w csltl %i %m
    jnz %i_cond @j_loop_start @return

@j_loop_start
    jmp @j_loop

@j_loop
    %j2 = phi l 1 @j_loop_start, %j2_next @j_loop_next
    %diag = phi l 0 @j_loop_start, %up @j_loop_next
    %j_cond =w cslel %j2 %n
    jnz %j_cond @process @i_loop_next

@process
    %j2_8 = mull %j2 8
    %up_addr = addl %row %j2_8
    %up = loadl %up_addr
    
    %xi_addr = addl %x %i
    %xi =w loadub %xi_addr
    
    %jm1 = subl %j2 1
    %yjm1_addr = addl %y %jm1
    %yjm1 =w loadub %yjm1_addr
    
    %eq =w ceqw %xi %yjm1
    
    jnz %eq @match @nomatch

@match
    %new_val = addl %diag 1
    jmp @store

@nomatch
    %jm1_8 = mull %jm1 8
    %prev_addr = addl %row %jm1_8
    %prev_val = loadl %prev_addr
    
    %cmp =w csltl %up %prev_val
    jnz %cmp @take_prev @take_up

@take_prev
    jmp @store_pre

@take_up
    jmp @store_pre

@store_pre
    %new_val = phi l %prev_val @take_prev, %up @take_up
    jmp @store

@store
    %final_val = phi l %new_val @match, %new_val @store_pre
    %j2_8_2 = mull %j2 8
    %store_addr = addl %row %j2_8_2
    storel %final_val %store_addr
    jmp @j_loop_next

@j_loop_next
    %j2_next = addl %j2 1
    jmp @j_loop

@i_loop_next
    %i_next = addl %i 1
    jmp @i_loop

@return
    %n_8 = mull %n 8
    %result_addr = addl %row %n_8
    %result = loadl %result_addr
    ret l %result
}
