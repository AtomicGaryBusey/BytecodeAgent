export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    %zero = sub %n %n
    %one = add %zero 1
    jmp @init_loop_start

@init_loop_start
    %j_init = copy %zero
    jmp @init_loop

@init_loop
    %j_il = phi l %j_init @init_loop_start, %j_next_il @init_body
    %j_cond =w cslel %j_il %n
    jnz %j_cond @init_body @i_loop_start

@init_body
    %offset = mull %j_il 8
    %addr = addl %row %offset
    storel %zero %addr
    %j_next_il = addl %j_il %one
    jmp @init_loop

@i_loop_start
    %i_init = copy %zero
    jmp @i_loop

@i_loop
    %i_il = phi l %i_init @i_loop_start, %i_next @i_next
    %i_cond =w csltl %i_il %m
    jnz %i_cond @i_body @return

@i_body
    %diag_start = copy %zero
    %j_start = copy %one
    jmp @j_loop

@j_loop
    %j2 = phi l %j_start @i_body, %j2_next @j_body
    %diag = phi l %diag_start @i_body, %diag_next @j_body
    %j2_cond =w cslel %j2 %n
    jnz %j2_cond @j_body @i_next

@j_body
    %offset_j = mull %j2 8
    %addr_j = addl %row %offset_j
    %up = loadl %addr_j

    %addr_x = addl %x %i_il
    %x_val =w loadub %addr_x

    %j2m1 = subl %j2 %one
    %addr_y = addl %y %j2m1
    %y_val =w loadub %addr_y

    %eq_cond =w ceqw %x_val %y_val
    jnz %eq_cond @match @nomatch

@match
    %new_val_m = addl %diag %one
    jmp @store

@nomatch
    %offset_j2m1 = mull %j2m1 8
    %addr_j2m1 = addl %row %offset_j2m1
    %row_j2m1 = loadl %addr_j2m1

    %is_less =w csltl %row_j2m1 %up
    jnz %is_less @take_up @take_row_j2m1

@take_up
    jmp @store

@take_row_j2m1
    jmp @store

@store
    %new_val = phi l %new_val_m @match, %up @take_up, %row_j2m1 @take_row_j2m1

    %offset_store = mull %j2 8
    %addr_store = addl %row %offset_store
    storel %new_val %addr_store

    %diag_next = copy %up
    %j2_next = addl %j2 %one
    jmp @j_loop

@i_next
    %i_next = addl %i_il %one
    jmp @i_loop

@return
    %offset_n = mull %n 8
    %addr_n = addl %row %offset_n
    %result = loadl %addr_n
    ret l %result
}
