function l $map_index(l %idx) {
    @entry
    %lt26 =w cslel %idx, 25
    jnz %lt26 @upper
    %lt52 =w cslel %idx, 51
    jnz %lt52 @lower
    %lt62 =w cslel %idx, 61
    jnz %lt62 @digit
    %eq62 =w ceql %idx, 62
    jnz %eq62 @plus
    ret 47

    @upper
    %c =l add %idx, 65
    ret %c

    @lower
    %tmp =l sub %idx, 26
    %c =l add %tmp, 97
    ret %c

    @digit
    %tmp =l sub %idx, 52
    %c =l add %tmp, 48
    ret %c

    @plus
    ret 43
}
