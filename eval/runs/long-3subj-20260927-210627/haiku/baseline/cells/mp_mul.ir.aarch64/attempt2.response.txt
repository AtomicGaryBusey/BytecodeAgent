export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %na_zero =w ceql %na, 0
  jnz %na_zero @return_zero @check_nb
  
@check_nb
  %nb_zero =w ceql %nb, 0
  jnz %nb_zero @return_zero @init_r
  
@init_r
  %result_size =l add %na, %nb
  jmp @init_loop
  
@init_loop
  %i =l phi @init_r 0, @init_next %i_inc
  %i_done =w ceql %i, %result_size
  jnz %i_done @outer_loop @do_init
  
@do_init
  %addr =l mul %i, 8
  %ptr =l add %r, %addr
  storel 0, %ptr
  jmp @init_next
  
@init_next
  %i_inc =l add %i, 1
  jmp @init_loop
  
@outer_loop
  %i =l phi @init_loop 0, @outer_next %i_inc
  %i_done =w ceql %i, %na
  jnz %i_done @find_sig @inner_loop
  
@inner_loop
  %j =l phi @outer_loop 0, @inner_next %j_inc
  %j_done =w ceql %j, %nb
  jnz %j_done @outer_next @mul_limbs
  
@mul_limbs
  %a_off =l mul %i, 8
  %a_ptr =l add %a, %a_off
  %a_val =l loadl %a_ptr
  
  %b_off =l mul %j, 8
  %b_ptr =l add %b, %b_off
  %b_val =l loadl %b_ptr
  
  %a_lo =l and %a_val, 0xffffffff
  %a_hi =l shr %a_val, 32
  %b_lo =l and %b_val, 0xffffffff
  %b_hi =l shr %b_val, 32
  
  %p00 =l mul %a_lo, %b_lo
  %p01 =l mul %a_lo, %b_hi
  %p10 =l mul %a_hi, %b_lo
  %p11 =l mul %a_hi, %b_hi
  
  %pm =l add %p01, %p10
  %pm_c_w =w cultl %pm, %p01
  %pm_c =l zext %pm_c_w
  
  %pm_lo =l and %pm, 0xffffffff
  %pm_hi =l shr %pm, 32
  %pm_lo_sh =l shl %pm_lo, 32
  
  %lo =l add %p00, %pm_lo_sh
  %lo_c_w =w cultl %lo, %p00
  %lo_c =l zext %lo_c_w
  
  %hi_1 =l add %p11, %pm_hi
  %hi_2 =l add %hi_1, %pm_c
  %hi =l add %hi_2, %lo_c
  
  %ij =l add %i, %j
  
  %r_ij_off =l mul %ij, 8
  %r_ij_ptr =l add %r, %r_ij_off
  %r_ij =l loadl %r_ij_ptr
  %new_r_ij =l add %r_ij, %lo
  %c_ij_w =w cultl %new_r_ij, %r_ij
  %c_ij =l zext %c_ij_w
  storel %new_r_ij, %r_ij_ptr
  
  %r_ij1_off =l add %r_ij_off, 8
  %r_ij1_ptr =l add %r, %r_ij1_off
  %r_ij1 =l loadl %r_ij1_ptr
  %new_r_ij1_t =l add %r_ij1, %hi
  %new_r_ij1 =l add %new_r_ij1_t, %c_ij
  %c_ij1_w =w cultl %new_r_ij1, %r_ij1
  storel %new_r_ij1, %r_ij1_ptr
  
  jnz %c_ij1_w @carry_prop @inner_next
  
@carry_prop
  %k =l add %ij, 2
  jmp @carry_loop
  
@carry_loop
  %k_val =l phi @carry_prop %k, @carry_next %k_inc
  %k_done =w ceql %k_val, %result_size
  jnz %k_done @inner_next @do_carry
  
@do_carry
  %r_k_off =l mul %k_val, 8
  %r_k_ptr =l add %r, %r_k_off
  %r_k =l loadl %r_k_ptr
  %r_k_new =l add %r_k, 1
  storel %r_k_new, %r_k_ptr
  %r_k_zero =w ceql %r_k_new, 0
  jnz %r_k_zero @carry_next @inner_next
  
@carry_next
  %k_inc =l add %k_val, 1
  jmp @carry_loop
  
@inner_next
  %j_inc =l add %j, 1
  jmp @inner_loop
  
@outer_next
  %i_inc =l add %i, 1
  jmp @outer_loop
  
@find_sig
  %k_init =l add %result_size, -1
  jmp @find_loop
  
@find_loop
  %k =l phi @find_sig %k_init, @find_next %k_dec
  %k_neg =w csltl %k, 0
  jnz %k_neg @return_zero @check_k
  
@check_k
  %r_k_off =l mul %k, 8
  %r_k_ptr =l add %r, %r_k_off
  %r_k =l loadl %r_k_ptr
  %r_k_nz =w cnel %r_k, 0
  jnz %r_k_nz @return_k @find_next
  
@find_next
  %k_dec =l add %k, -1
  jmp @find_loop
  
@return_k
  %res =l add %k, 1
  ret %res
  
@return_zero
  ret 0
}
