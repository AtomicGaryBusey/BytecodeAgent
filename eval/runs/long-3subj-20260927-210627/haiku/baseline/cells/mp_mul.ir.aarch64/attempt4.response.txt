export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
  %na_zero =w ceql %na, 0
  jnz %na_zero, @return_zero, @check_nb

@check_nb
  %nb_zero =w ceql %nb, 0
  jnz %nb_zero, @return_zero, @init_r

@init_r
  %result_size =l add %na, %nb
  jmp @init_loop

@init_loop
  %i =l phi @init_r 0, @init_next %i_inc
  %i_done =w ceql %i, %result_size
  jnz %i_done, @outer_loop, @do_init

@do_init
  %addr =l mul %i, 8
  %ptr =l add %r, %addr
  storel 0, %ptr
  jmp @init_next

@init_next
  %i_inc =l add %i, 1
  jmp @init_loop

@outer_loop
  %i =l phi @init_loop 0, @outer_next %i_inc
  %i_done =w ceql %i, %na
  jnz %i_done, @find_sig, @inner_loop

@inner_loop
  %j =l phi @outer_loop 0, @inner_next %j_inc
  %j_done =w ceql %j, %nb
  jnz %j_done, @outer_next, @mul_limbs

@mul_limbs
  %a_off =l mul %i, 8
  %a_ptr =l add %a, %a_off
  %a_val =l loadl %a_ptr

  %b_off =l mul %j, 8
  %b_ptr =l add %b, %b_off
  %b_val =l loadl %b_ptr

  %a_lo =l and %a_val, 4294967295
  %a_hi =l shr %a_val, 32
  %b_lo =l and %b_val, 4294967295
  %b_hi =l shr %b_val, 32

  %p00 =l mul %a_lo, %b_lo
  %p01 =l mul %a_lo, %b_hi
  %p10 =l mul %a_hi, %b_lo
  %p11 =l mul %a_hi, %b_hi

  %pm =l add %p01, %p10
  %pm_c =w cultl %pm, %p01

  %pm_lo =l and %pm, 4294967295
  %pm_hi =l shr %pm, 32
  %pm_lo_sh =l shl %pm_lo, 32

  %lo =l add %p00, %pm_lo_sh
  %lo_c =w cultl %lo, %p00

  %hi_1 =l add %p11, %pm_hi
  jnz %pm_c, @add_pm_c, @check_lo_c

@add_pm_c
  %hi_2 =l add %hi_1, 1
  jmp @check_lo_c

@check_lo_c
  %hi_base =l phi @add_pm_c %hi_2, @mul_limbs %hi_1
  jnz %lo_c, @add_lo_c, @add_to_r

@add_lo_c
  %hi =l add %hi_base, 1
  jmp @add_to_r

@add_to_r
  %hi_final =l phi @add_lo_c %hi, @check_lo_c %hi_base

  %ij =l add %i, %j
  %r_ij_off =l mul %ij, 8
  %r_ij_ptr =l add %r, %r_ij_off
  %r_ij_old =l loadl %r_ij_ptr
  %r_ij_new =l add %r_ij_old, %lo
  %c_ij =w cultl %r_ij_new, %r_ij_old
  storel %r_ij_new, %r_ij_ptr

  %ij1 =l add %ij, 1
  %r_ij1_off =l mul %ij1, 8
  %r_ij1_ptr =l add %r, %r_ij1_off
  %r_ij1_old =l loadl %r_ij1_ptr
  %r_ij1_sum =l add %r_ij1_old, %hi_final
  %c_hi =w cultl %r_ij1_sum, %r_ij1_old

  jnz %c_ij, @add_ij_carry, @check_hi_only

@add_ij_carry
  %r_ij1_sum2 =l add %r_ij1_sum, 1
  %c_ij1 =w cultl %r_ij1_sum2, %r_ij1_sum
  storel %r_ij1_sum2, %r_ij1_ptr
  jnz %c_hi, @carry_prop, @check_ij_carry

@check_ij_carry
  jnz %c_ij1, @carry_prop, @inner_next

@check_hi_only
  storel %r_ij1_sum, %r_ij1_ptr
  jnz %c_hi, @carry_prop, @inner_next

@carry_prop
  %k_init =l add %ij1, 1
  jmp @carry_loop

@carry_loop
  %k =l phi @carry_prop %k_init, @carry_next %k_inc
  %k_done =w ceql %k, %result_size
  jnz %k_done, @inner_next, @do_carry

@do_carry
  %r_k_off =l mul %k, 8
  %r_k_ptr =l add %r, %r_k_off
  %r_k_old =l loadl %r_k_ptr
  %r_k_new =l add %r_k_old, 1
  %r_k_c =w cultl %r_k_new, %r_k_old
  storel %r_k_new, %r_k_ptr
  jnz %r_k_c, @carry_next, @inner_next

@carry_next
  %k_inc =l add %k, 1
  jmp @carry_loop

@inner_next
  %j_inc =l add %j, 1
  jmp @inner_loop

@outer_next
  %i_inc =l add %i, 1
  jmp @outer_loop

@find_sig
  %k_init =l add %result_size, -1
  jmp @find_loop

@find_loop
  %k =l phi @find_sig %k_init, @find_next %k_dec
  %k_neg =w csltl %k, 0
  jnz %k_neg, @return_zero, @check_k

@check_k
  %r_k_off =l mul %k, 8
  %r_k_ptr =l add %r, %r_k_off
  %r_k =l loadl %r_k_ptr
  %r_k_nz =w cnel %r_k, 0
  jnz %r_k_nz, @return_k, @find_next

@find_next
  %k_dec =l add %k, -1
  jmp @find_loop

@return_k
  %res =l add %k, 1
  ret %res

@return_zero
  ret 0
}
