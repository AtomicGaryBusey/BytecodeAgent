export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @mainloop(0, 0)

@mainloop(l %i, l %o)
    w %more = csltl %i, %n
    jnz %more, @start_runloop, @done

@start_runloop
    jmp @runloop(%i, 1, 1)

@runloop(l %i, l %j_off, l %len)
    l %j = add %i, %j_off
    w %has_next = csltl %j, %n
    jnz %has_next, @check_len_limit, @runend_done

@check_len_limit
    w %len_ok = csltl %len, 128
    jnz %len_ok, @check_match, @runend_done

@check_match
    w %b_i = loadub %src + %i
    w %b_j = loadub %src + %j
    w %match = ceql %b_i, %b_j
    jnz %match, @run_inc, @runend_done

@run_inc
    l %j_off_new = add %j_off, 1
    l %len_new = add %len, 1
    jmp @runloop(%i, %j_off_new, %len_new)

@runend_done
    w %is_run = csgtl %len, 2
    jnz %is_run, @do_run_packet, @do_lit_packet

@do_run_packet
    l %need = add %o, 2
    w %fits = csltl %need, %cap
    jnz %fits, @write_run, @error_ret

@write_run
    l %hdr = sub 257, %len
    w %hdr_b = truncl %hdr
    storeb %hdr_b, %out + %o
    w %b = loadub %src + %i
    l %o1 = add %o, 1
    storeb %b, %out + %o1
    l %i_new = add %i, %len
    l %o_new = add %o1, 1
    jmp @mainloop(%i_new, %o_new)

@do_lit_packet
    jmp @find_lit_loop(%i, 0, %i, %o)

@find_lit_loop(l %i, l %lit_len, l %j, l %o)
    w %j_ok = csltl %j, %n
    jnz %j_ok, @check_lit_len, @do_lit_write

@check_lit_len
    w %lit_len_ok = csltl %lit_len, 128
    jnz %lit_len_ok, @start_next_runloop, @do_lit_write

@start_next_runloop
    jmp @next_runloop(%j, 1, 1, %i, %lit_len, %o)

@next_runloop(l %j, l %k_off, l %next_len, l %i, l %lit_len, l %o)
    l %k = add %j, %k_off
    w %k_ok = csltl %k, %n
    jnz %k_ok, @check_next_len_limit, @next_run_end

@check_next_len_limit
    w %next_len_ok = csltl %next_len, 128
    jnz %next_len_ok, @check_next_match, @next_run_end

@check_next_match
    w %b_j = loadub %src + %j
    w %b_k = loadub %src + %k
    w %next_match = ceql %b_j, %b_k
    jnz %next_match, @next_run_inc, @next_run_end

@next_run_inc
    l %k_off_new = add %k_off, 1
    l %next_len_new = add %next_len, 1
    jmp @next_runloop(%j, %k_off_new, %next_len_new, %i, %lit_len, %o)

@next_run_end
    w %is_next_run = csgtl %next_len, 2
    jnz %is_next_run, @do_lit_write, @lit_add_byte

@lit_add_byte
    l %lit_len_new = add %lit_len, 1
    l %j_new = add %j, 1
    jmp @find_lit_loop(%i, %lit_len_new, %j_new, %o)

@do_lit_write
    l %lit_total = add %lit_len, 1
    l %need = add %o, %lit_total
    w %fits = csltl %need, %cap
    jnz %fits, @write_lit, @error_ret

@write_lit
    l %hdr = sub %lit_len, 1
    w %hdr_b = truncl %hdr
    storeb %hdr_b, %out + %o
    l %o1 = add %o, 1
    jmp @copy_lit_loop(%i, %o1, 0, %lit_len)

@copy_lit_loop(l %i, l %out_pos, l %copy_idx, l %lit_len)
    w %not_done = csltl %copy_idx, %lit_len
    jnz %not_done, @copy_lit_byte, @lit_done

@copy_lit_byte
    l %src_pos = add %i, %copy_idx
    w %b = loadub %src + %src_pos
    storeb %b, %out + %out_pos
    l %out_pos_new = add %out_pos, 1
    l %copy_idx_new = add %copy_idx, 1
    jmp @copy_lit_loop(%i, %out_pos_new, %copy_idx_new, %lit_len)

@lit_done
    l %i_new = add %i, %lit_len
    jmp @mainloop(%i_new, %out_pos)

@error_ret
    ret -1

@done
    ret %o
}
