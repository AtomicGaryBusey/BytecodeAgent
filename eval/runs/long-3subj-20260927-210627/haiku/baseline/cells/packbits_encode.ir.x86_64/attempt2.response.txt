export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    l %i0 = copy 0
    l %o0 = copy 0
    jmp @mainloop

@mainloop
    l %i = phi @entry %i0, @repwrite %i1, @copyend %i2
    l %o = phi @entry %o0, @repwrite %o2, @copyend %o3
    w %cond = csltl %i, %n
    jnz %cond, @check_run, @finish

@check_run
    l %j = copy %i
    l %rlen = copy 0
    jmp @runloop

@runloop
    w %jlt = csltl %j, %n
    jnz %jlt, @check_rlen, @runend

@check_rlen
    w %rlen_lt = csltl %rlen, 128
    jnz %rlen_lt, @check_byte, @runend

@check_byte
    w %b_i = loadub %src + %i
    w %b_j = loadub %src + %j
    w %eq = ceql %b_i, %b_j
    jnz %eq, @incrun, @runend

@incrun
    l %j = add %j, 1
    l %rlen = add %rlen, 1
    jmp @runloop

@runend
    w %isrepeat = csgtl %rlen, 2
    jnz %isrepeat, @emitrepeat, @emitliteral

@emitrepeat
    l %need = add %o, 2
    w %fits = csltl %need, %cap
    jnz %fits, @repwrite, @error

@repwrite
    l %hdr = sub 257, %rlen
    w %hb = truncl %hdr
    storeb %hb, %out + %o
    w %b_i = loadub %src + %i
    l %o_temp = add %o, 1
    storeb %b_i, %out + %o_temp
    l %i1 = add %i, %rlen
    l %o2 = add %o_temp, 1
    jmp @mainloop

@emitliteral
    l %llen = copy 0
    l %lj = copy %i
    jmp @litloop

@litloop
    w %ljlt = csltl %lj, %n
    jnz %ljlt, @check_llen, @litend

@check_llen
    w %llen_lt = csltl %llen, 128
    jnz %llen_lt, @check_nextrun, @litend

@check_nextrun
    l %k = copy %lj
    l %nrlen = copy 0
    jmp @nrunloop

@nrunloop
    w %klt = csltl %k, %n
    jnz %klt, @check_nrlen, @nrunend

@check_nrlen
    w %nrlen_lt = csltl %nrlen, 128
    jnz %nrlen_lt, @check_nextbyte, @nrunend

@check_nextbyte
    w %b_lj = loadub %src + %lj
    w %b_k = loadub %src + %k
    w %eq = ceql %b_lj, %b_k
    jnz %eq, @incnrun, @nrunend

@incnrun
    l %k = add %k, 1
    l %nrlen = add %nrlen, 1
    jmp @nrunloop

@nrunend
    w %isnextrepeat = csgtl %nrlen, 2
    jnz %isnextrepeat, @litend, @incllit

@incllit
    l %llen = add %llen, 1
    l %lj = add %lj, 1
    jmp @litloop

@litend
    l %total = add %llen, 1
    l %need = add %o, %total
    w %fits = csltl %need, %cap
    jnz %fits, @litwrite, @error

@litwrite
    l %hdr = sub %llen, 1
    w %hb = truncl %hdr
    storeb %hb, %out + %o
    l %op = add %o, 1
    l %idx = copy 0
    jmp @copyloop

@copyloop
    w %notdone = csltl %idx, %llen
    jnz %notdone, @copybyte, @copyend

@copybyte
    l %sp = add %i, %idx
    w %b = loadub %src + %sp
    storeb %b, %out + %op
    l %op = add %op, 1
    l %idx = add %idx, 1
    jmp @copyloop

@copyend
    l %i2 = add %i, %llen
    l %o3 = copy %op
    jmp @mainloop

@error
    ret -1

@finish
    ret %o
}
