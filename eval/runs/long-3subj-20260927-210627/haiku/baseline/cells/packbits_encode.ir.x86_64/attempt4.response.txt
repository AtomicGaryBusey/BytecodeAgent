export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @mainloop 0 0

@mainloop(l %i, l %o)
    %cond = csltl %i %n
    jnz %cond @process @done

@process
    jmp @calc_run %i 1 1

@calc_run(l %i, l %off, l %len)
    %j = add %i %off
    %jok = csltl %j %n
    jnz %jok @check_len @run_end
    
@check_len
    %lenok = csltl %len 128
    jnz %lenok @check_match @run_end
    
@check_match
    %addr_i = add %src %i
    %b_i =w loadub %addr_i
    %addr_j = add %src %j
    %b_j =w loadub %addr_j
    %match = ceql %b_i %b_j
    jnz %match @inc_run @run_end

@inc_run
    %off_new = add %off 1
    %len_new = add %len 1
    jmp @calc_run %i %off_new %len_new

@run_end
    %is_run = csgtl %len 2
    jnz %is_run @do_repeat @do_literal

@do_repeat
    %need = add %o 2
    %fits = csltl %need %cap
    jnz %fits @write_repeat @error_ret

@write_repeat
    %hdr = sub 257 %len
    %hdr_b =w truncl %hdr
    %out_addr = add %out %o
    storeb %hdr_b %out_addr
    %addr_i = add %src %i
    %b =w loadub %addr_i
    %o1 = add %o 1
    %out_addr2 = add %out %o1
    storeb %b %out_addr2
    %i_new = add %i %len
    %o_new = add %o1 1
    jmp @mainloop %i_new %o_new

@do_literal
    jmp @find_lit %i 0 %i %o

@find_lit(l %si, l %llen, l %j, l %o)
    %jok = csltl %j %n
    jnz %jok @check_llen @emit_lit

@check_llen
    %llenok = csltl %llen 128
    jnz %llenok @check_run @emit_lit

@check_run
    jmp @calc_run_j %j 1 1 %si %llen %o

@calc_run_j(l %j, l %off, l %len, l %si, l %llen, l %o)
    %k = add %j %off
    %kok = csltl %k %n
    jnz %kok @check_lenj @run_endj
    
@check_lenj
    %lenok = csltl %len 128
    jnz %lenok @check_matchj @run_endj
    
@check_matchj
    %addr_j = add %src %j
    %b_j =w loadub %addr_j
    %addr_k = add %src %k
    %b_k =w loadub %addr_k
    %match = ceql %b_j %b_k
    jnz %match @inc_runj @run_endj

@inc_runj
    %off_new = add %off 1
    %len_new = add %len 1
    jmp @calc_run_j %j %off_new %len_new %si %llen %o

@run_endj
    %is_run = csgtl %len 2
    jnz %is_run @emit_lit @add_byte

@add_byte
    %llen_new = add %llen 1
    %j_new = add %j 1
    jmp @find_lit %si %llen_new %j_new %o

@emit_lit
    %lit_total = add %llen 1
    %need = add %o %lit_total
    %fits = csltl %need %cap
    jnz %fits @write_lit @error_ret

@write_lit
    %hdr = sub %llen 1
    %hdr_b =w truncl %hdr
    %out_addr = add %out %o
    storeb %hdr_b %out_addr
    %o1 = add %o 1
    jmp @copy_lit %si %o1 0 %llen

@copy_lit(l %si, l %op, l %idx, l %llen)
    %cond = csltl %idx %llen
    jnz %cond @copy_byte @lit_done

@copy_byte
    %sp = add %si %idx
    %b =w loadub %sp
    %op_idx = add %op %idx
    %out_addr = add %out %op_idx
    storeb %b %out_addr
    %idx_new = add %idx 1
    jmp @copy_lit %si %op %idx_new %llen

@lit_done
    %i_new = add %si %llen
    %o_new = add %op %llen
    jmp @mainloop %i_new %o_new

@error_ret
    ret -1

@done
    ret %o
}
