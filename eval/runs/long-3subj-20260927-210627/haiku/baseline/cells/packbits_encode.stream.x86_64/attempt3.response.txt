packbits_encode:
    push r12
    push r13
    push r14
    push r15
    
    xor r8, r8          # i = 0 (input position)
    xor r9, r9          # out_pos = 0 (output position)
    
.Louter:
    cmp r8, rsi         # if i >= n
    jge .Ldone
    
    # Calculate max run length for this position
    mov r10, rsi        # r10 = n
    sub r10, r8         # r10 = n - i
    cmp r10, 128
    jle .Lmax_run_ok
    mov r10, 128        # cap at 128
.Lmax_run_ok:
    
    # Count consecutive bytes equal to src[i]
    movzx r11d, byte ptr [rdi + r8]
    mov r12, 1          # r12 = run_length = 1
    
.Lrun_count_loop:
    cmp r12, r10        # if run_length >= max_run
    jge .Lrun_count_done
    
    mov rax, r8
    add rax, r12
    movzx eax, byte ptr [rdi + rax]
    
    cmp al, r11b        # if src[i + run_length] != src[i]
    jne .Lrun_count_done
    
    inc r12             # run_length++
    jmp .Lrun_count_loop
    
.Lrun_count_done:
    cmp r12, 3          # if run_length < 3
    jl .Lemit_literal
    
    # Emit repeat packet
    mov rax, r9
    add rax, 2
    cmp rax, rcx
    jg .Lno_room
    
    mov al, r12b
    dec al
    neg al
    mov byte ptr [rdx + r9], al
    
    mov al, r11b
    mov byte ptr [rdx + r9 + 1], al
    
    add r9, 2
    add r8, r12
    jmp .Louter
    
.Lemit_literal:
    mov r13, r8         # j = i
    
.Llit_find_loop:
    cmp r13, rsi        # if j >= n
    jge .Llit_find_done
    
    mov r14, r13
    sub r14, r8         # r14 = j - i
    cmp r14, 128        # if j - i >= 128
    jge .Llit_find_done
    
    # Calculate run(j)
    mov r15, rsi
    sub r15, r13        # r15 = n - j
    cmp r15, 128
    jle .Llit_run_max_ok
    mov r15, 128
.Llit_run_max_ok:
    
    movzx r10d, byte ptr [rdi + r13]
    mov r11, 1
    
.Llit_run_count_loop:
    cmp r11, r15
    jge .Llit_run_count_done
    
    mov rax, r13
    add rax, r11
    movzx eax, byte ptr [rdi + rax]
    
    cmp al, r10b
    jne .Llit_run_count_done
    
    inc r11
    jmp .Llit_run_count_loop
    
.Llit_run_count_done:
    cmp r11, 3
    jl .Llit_next_byte
    jmp .Llit_find_done
    
.Llit_next_byte:
    inc r13
    jmp .Llit_find_loop
    
.Llit_find_done:
    mov r14, r13
    sub r14, r8         # r14 = literal length
    
    # Check if we have room for 1 + L bytes
    mov rax, r9
    add rax, 1
    add rax, r14
    cmp rax, rcx
    jg .Lno_room
    
    # Write header: L - 1
    mov al, r14b
    dec al
    mov byte ptr [rdx + r9], al
    inc r9
    
    # Prepare base pointers
    mov r11, rdi
    add r11, r8         # r11 = src + i
    mov r12, rdx
    add r12, r9         # r12 = out + out_pos
    
    # Copy L bytes
    xor r10, r10
.Llit_copy_loop:
    cmp r10, r14
    jge .Llit_copy_done
    
    mov al, byte ptr [r11 + r10]
    mov byte ptr [r12 + r10], al
    inc r10
    jmp .Llit_copy_loop
    
.Llit_copy_done:
    add r9, r14
    mov r8, r13
    jmp .Louter
    
.Ldone:
    mov rax, r9
    pop r15
    pop r14
    pop r13
    pop r12
    ret
    
.Lno_room:
    mov rax, -1
    pop r15
    pop r14
    pop r13
    pop r12
    ret
