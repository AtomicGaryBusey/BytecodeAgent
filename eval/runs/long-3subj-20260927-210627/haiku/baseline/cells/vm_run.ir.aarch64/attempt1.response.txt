export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%pc = copy 0
	%sp = copy 0
	jmp @loop

@loop
	%pc_eq_len = ceql %pc, %len
	jnz %pc_eq_len @halt
	
	%pc_offset = mull %pc, 8
	%code_addr = addl %code, %pc_offset
	%instr = loadl %code_addr
	
	%is_halt = ceql %instr, 0
	jnz %is_halt @halt
	
	%is_push = ceql %instr, 1
	jnz %is_push @push
	
	%is_add = ceql %instr, 2
	jnz %is_add @add
	
	%is_sub = ceql %instr, 3
	jnz %is_sub @sub
	
	%is_mul = ceql %instr, 4
	jnz %is_mul @mul
	
	%is_dup = ceql %instr, 5
	jnz %is_dup @dup
	
	%is_swap = ceql %instr, 6
	jnz %is_swap @swap
	
	%is_drop = ceql %instr, 7
	jnz %is_drop @drop
	
	%is_over = ceql %instr, 8
	jnz %is_over @over
	
	%is_lt = ceql %instr, 9
	jnz %is_lt @lt
	
	%is_jz = ceql %instr, 10
	jnz %is_jz @jz
	
	%is_jmp = ceql %instr, 11
	jnz %is_jmp @jmp
	
	%pc = addl %pc, 1
	jmp @loop

@push
	%pc_offset = mull %pc, 8
	%code_addr = addl %code, %pc_offset
	%code_addr_plus_8 = addl %code_addr, 8
	%val = loadl %code_addr_plus_8
	
	%sp_offset = mull %sp, 8
	%stack_addr = addl %stack, %sp_offset
	storel %val, %stack_addr
	
	%sp = addl %sp, 1
	%pc = addl %pc, 2
	jmp @loop

@add
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%b_addr = addl %stack, %sp_minus_1_offset
	%b = loadl %b_addr
	
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%a_addr = addl %stack, %sp_minus_2_offset
	%a = loadl %a_addr
	
	%result = addl %a, %b
	storel %result, %a_addr
	
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@sub
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%b_addr = addl %stack, %sp_minus_1_offset
	%b = loadl %b_addr
	
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%a_addr = addl %stack, %sp_minus_2_offset
	%a = loadl %a_addr
	
	%result = subl %a, %b
	storel %result, %a_addr
	
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@mul
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%b_addr = addl %stack, %sp_minus_1_offset
	%b = loadl %b_addr
	
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%a_addr = addl %stack, %sp_minus_2_offset
	%a = loadl %a_addr
	
	%result = mull %a, %b
	storel %result, %a_addr
	
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@dup
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%top_addr = addl %stack, %sp_minus_1_offset
	%top = loadl %top_addr
	
	%sp_offset = mull %sp, 8
	%new_top_addr = addl %stack, %sp_offset
	storel %top, %new_top_addr
	
	%sp = addl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@swap
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%b_addr = addl %stack, %sp_minus_1_offset
	%b = loadl %b_addr
	
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%a_addr = addl %stack, %sp_minus_2_offset
	%a = loadl %a_addr
	
	storel %b, %a_addr
	storel %a, %b_addr
	
	%pc = addl %pc, 1
	jmp @loop

@drop
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@over
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%below_top_addr = addl %stack, %sp_minus_2_offset
	%below_top = loadl %below_top_addr
	
	%sp_offset = mull %sp, 8
	%new_top_addr = addl %stack, %sp_offset
	storel %below_top, %new_top_addr
	
	%sp = addl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@lt
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%b_addr = addl %stack, %sp_minus_1_offset
	%b = loadl %b_addr
	
	%sp_minus_2 = subl %sp, 2
	%sp_minus_2_offset = mull %sp_minus_2, 8
	%a_addr = addl %stack, %sp_minus_2_offset
	%a = loadl %a_addr
	
	%cmp_w = csltl %a, %b
	%cmp = zextl %cmp_w
	storel %cmp, %a_addr
	
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @loop

@jz
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%v_addr = addl %stack, %sp_minus_1_offset
	%v = loadl %v_addr
	
	%sp = subl %sp, 1
	
	%t_offset = mull %pc, 8
	%t_offset_plus_8 = addl %t_offset, 8
	%t_addr = addl %code, %t_offset_plus_8
	%t = loadl %t_addr
	
	%is_zero = ceql %v, 0
	jnz %is_zero @jz_zero
	
	%pc = addl %pc, 2
	jmp @loop

@jz_zero
	%pc = copy %t
	jmp @loop

@jmp
	%t_offset = mull %pc, 8
	%t_offset_plus_8 = addl %t_offset, 8
	%t_addr = addl %code, %t_offset_plus_8
	%t = loadl %t_addr
	
	%pc = copy %t
	jmp @loop

@halt
	%is_empty = ceql %sp, 0
	jnz %is_empty @set_result_zero
	
	%sp_minus_1 = subl %sp, 1
	%sp_minus_1_offset = mull %sp_minus_1, 8
	%top_addr = addl %stack, %sp_minus_1_offset
	%result = loadl %top_addr
	
	%i = copy %sp
	jmp @clear_loop_cond

@set_result_zero
	%result = copy 0
	%i = copy 0
	jmp @clear_loop_cond

@clear_loop
	%i_offset = mull %i, 8
	%clear_addr = addl %stack, %i_offset
	storel 0, %clear_addr
	%i = addl %i, 1

@clear_loop_cond
	%is_done = ceql %i, %cap
	jnz %is_done @return_result
	jmp @clear_loop

@return_result
	ret %result
}
