export function l $qsort_pairs(l %a, l %n) {
@entry
    %cond =w csltl %n 2
    jnz %cond @ret0
    
    %pk =l loadl %a
    %pv_addr =l addl %a 8
    %pv =l loadl %pv_addr
    
    %i_init =l copy 1
    %p_init =l copy 0
    jmp @loop_header
    
@loop_header
    %i =l phi @entry %i_init, @loop_iterate %i_next
    %p =l phi @entry %p_init, @loop_iterate %p_after
    
    %test_i =w csltl %i %n
    jnz %test_i @loop_body
    jmp @partition_end
    
@loop_body
    %i2 =l mull %i 2
    %addr_i =l addl %a %i2
    %ki =l loadl %addr_i
    %vi_addr =l addl %addr_i 8
    %vi =l loadl %vi_addr
    
    %less_key =w csltl %ki %pk
    jnz %less_key @perform_swap
    
    %eq_key =w ceql %ki %pk
    jnz %eq_key @check_value
    
    jmp @no_swap
    
@check_value
    %less_val =w csltl %vi %pv
    jnz %less_val @perform_swap
    
    jmp @no_swap
    
@no_swap
    %p_after_1 =l copy %p
    jmp @loop_iterate
    
@perform_swap
    %pp1 =l addl %p 1
    %p2 =l mull %pp1 2
    %addr_p =l addl %a %p2
    %kp =l loadl %addr_p
    %vp_addr =l addl %addr_p 8
    %vp =l loadl %vp_addr
    
    storel %ki %addr_p
    storel %vi %vp_addr
    storel %kp %addr_i
    storel %vp %vi_addr
    
    %p_after_2 =l copy %pp1
    jmp @loop_iterate
    
@loop_iterate
    %p_after =l phi @no_swap %p_after_1, @perform_swap %p_after_2
    %i_next =l addl %i 1
    jmp @loop_header
    
@partition_end
    %p2_f =l mull %p 2
    %addr_p_f =l addl %a %p2_f
    %kp_f =l loadl %addr_p_f
    %vp_addr_f =l addl %addr_p_f 8
    %vp_f =l loadl %vp_addr_f
    
    storel %pk %addr_p_f
    storel %pv %vp_addr_f
    storel %kp_f %a
    %val_addr_0 =l addl %a 8
    storel %vp_f %val_addr_0
    
    %call_left =l call $qsort_pairs(l %a, l %p)
    
    %p_plus_1 =l addl %p 1
    %p_plus_1_2 =l mull %p_plus_1 2
    %addr_right =l addl %a %p_plus_1_2
    %n_right =l subl %n %p
    %n_right_m1 =l subl %n_right 1
    
    %call_right =l call $qsort_pairs(l %addr_right, l %n_right_m1)
    
    jmp @ret0
    
@ret0
    %result =l copy 0
    ret %result
}
