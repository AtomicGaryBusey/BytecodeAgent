export function l $qsort_pairs(l %a, l %n) {
@entry
    %base =w csltl %n, 2
    jnz %base @ret0
    
    %pk =l loadl %a
    %pva =l addl %a, 8
    %pv =l loadl %pva
    
    %i_init =l copy 1
    %p_init =l copy 0
    jmp @ploop_hdr
    
@ploop_hdr
    %i =l phi @entry %i_init, @ploop_next %i_new
    %p =l phi @entry %p_init, @ploop_next %p_new
    
    %ic =w csltl %i, %n
    jnz %ic @pbody
    jmp @pend
    
@pbody
    %io =l mull %i, 2
    %ia =l addl %a, %io
    %ik =l loadl %ia
    %iva =l addl %ia, 8
    %iv =l loadl %iva
    
    %kc =w csltl %ik, %pk
    jnz %kc @swap
    
    %keq =w ceql %ik, %pk
    jnz %keq @vchk
    jmp @ni
    
@vchk
    %vc =w csltl %iv, %pv
    jnz %vc @swap
    
@ni
    %in =l addl %i, 1
    %p_keep =l copy %p
    jmp @ploop_next
    
@swap
    %pp1 =l addl %p, 1
    %po =l mull %pp1, 2
    %pa =l addl %a, %po
    
    %pk2 =l loadl %pa
    %pva2 =l addl %pa, 8
    %pv2 =l loadl %pva2
    
    storel %ik, %pa
    storel %iv, %pva2
    
    storel %pk2, %ia
    storel %pv2, %iva
    
    %in2 =l addl %i, 1
    %p_new2 =l copy %pp1
    jmp @ploop_next
    
@ploop_next
    %i_new =l phi @ni %in, @swap %in2
    %p_new =l phi @ni %p_keep, @swap %p_new2
    jmp @ploop_hdr
    
@pend
    %po2 =l mull %p, 2
    %pa2 =l addl %a, %po2
    
    %pk3 =l loadl %pa2
    %pva3 =l addl %pa2, 8
    %pv3 =l loadl %pva3
    
    storel %pk, %pa2
    storel %pv, %pva3
    
    storel %pk3, %a
    %av =l addl %a, 8
    storel %pv3, %av
    
    %rl =l call $qsort_pairs(l %a, l %p)
    
    %pp2 =l addl %p, 1
    %ro =l mull %pp2, 2
    %ra =l addl %a, %ro
    %rn =l subl %n, %p
    %rn2 =l subl %rn, 1
    
    %rr =l call $qsort_pairs(l %ra, l %rn2)
    
@ret0
    %r =l copy 0
    ret %r
}
