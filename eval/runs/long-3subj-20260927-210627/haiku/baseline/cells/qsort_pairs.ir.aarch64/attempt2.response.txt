export function l $qsort_pairs(l %a, l %n) {
    %cond =w csltl %n 2
    jnz %cond @ret0
@partition
    %pk =l loadl %a
    %pva =l addl %a 8
    %pv =l loadl %pva
    
    jmp @ploop_hdr
    
@ploop_hdr
    %i =l phi @partition 1, @ploop_next %i_next
    %p =l phi @partition 0, @ploop_next %p_next
    
    %ic =w csltl %i %n
    jnz %ic @pbody
    jmp @pend
    
@pbody
    %io =l mull %i 2
    %ia =l addl %a %io
    %ik =l loadl %ia
    %iva =l addl %ia 8
    %iv =l loadl %iva
    
    %kc =w csltl %ik %pk
    jnz %kc @swap
    
    %keq =w ceql %ik %pk
    jnz %keq @check_value
    jmp @ni
    
@check_value
    %vc =w csltl %iv %pv
    jnz %vc @swap
    jmp @ni
    
@swap
    %pp1 =l addl %p 1
    %po =l mull %pp1 2
    %pa =l addl %a %po
    
    %pk2 =l loadl %pa
    %pva2 =l addl %pa 8
    %pv2 =l loadl %pva2
    
    storel %ik %pa
    storel %iv %pva2
    
    storel %pk2 %ia
    storel %pv2 %iva
    
    %p_new =l copy %pp1
    jmp @ploop_next
    
@ni
    %p_new =l copy %p
    jmp @ploop_next
    
@ploop_next
    %i_next =l addl %i 1
    %p_next =l phi @swap %p_new, @ni %p_new
    jmp @ploop_hdr
    
@pend
    %po2 =l mull %p 2
    %pa2 =l addl %a %po2
    
    %pk3 =l loadl %pa2
    %pva3 =l addl %pa2 8
    %pv3 =l loadl %pva3
    
    storel %pk %pa2
    storel %pv %pva3
    
    storel %pk3 %a
    %av =l addl %a 8
    storel %pv3 %av
    
    %rl =l call $qsort_pairs(l %a, l %p)
    
    %pp2 =l addl %p 1
    %ro =l mull %pp2 2
    %ra =l addl %a %ro
    %rn =l subl %n %p
    %rn2 =l subl %rn 1
    
    %rr =l call $qsort_pairs(l %ra, l %rn2)
    jmp @ret0
    
@ret0
    %r =l copy 0
    ret %r
}
