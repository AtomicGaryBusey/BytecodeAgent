export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
  %is_zero_a =w ceql %na, 0
  jnz %is_zero_a @zero_out @check_b
@check_b
  %is_zero_b =w ceql %nb, 0
  jnz %is_zero_b @zero_out @init_clear
@zero_out
  %total =l add %na, %nb
  %i =l copy 0
@clear_loop
  %cond =w csltl %i, %total
  jnz %cond @clear_body @multiply_outer
@clear_body
  %offset =l shl %i, 3
  %addr =l add %r, %offset
  storel 0, %addr
  %i =l add %i, 1
  jmp @clear_loop
@init_clear
  %total =l add %na, %nb
  %i =l copy 0
@init_loop
  %cond =w csltl %i, %total
  jnz %cond @init_body @multiply_outer
@init_body
  %offset =l shl %i, 3
  %addr =l add %r, %offset
  storel 0, %addr
  %i =l add %i, 1
  jmp @init_loop
@multiply_outer
  %i =l copy 0
  %total =l add %na, %nb
@outer_loop
  %i_cond =w csltl %i, %na
  jnz %i_cond @outer_body @find_result
@outer_body
  %i_offset =l shl %i, 3
  %ai_addr =l add %a, %i_offset
  %ai =l loadl %ai_addr
  %j =l copy 0
@j_loop
  %j_cond =w csltl %j, %nb
  jnz %j_cond @j_body @i_next
@j_body
  %j_offset =l shl %j, 3
  %bj_addr =l add %b, %j_offset
  %bj =l loadl %bj_addr
  %ai_lo =l and %ai, 0xffffffff
  %ai_hi =l shr %ai, 32
  %bj_lo =l and %bj, 0xffffffff
  %bj_hi =l shr %bj, 32
  %p00 =l mul %ai_lo, %bj_lo
  %p01 =l mul %ai_lo, %bj_hi
  %p10 =l mul %ai_hi, %bj_lo
  %p11 =l mul %ai_hi, %bj_hi
  %p00_lo =l and %p00, 0xffffffff
  %p00_hi =l shr %p00, 32
  %p01_lo =l and %p01, 0xffffffff
  %p01_hi =l shr %p01, 32
  %p10_lo =l and %p10, 0xffffffff
  %p10_hi =l shr %p10, 32
  %p11_lo =l and %p11, 0xffffffff
  %p11_hi =l shr %p11, 32
  %mid =l add %p00_hi, %p01_lo
  %mid =l add %mid, %p10_lo
  %mid_lo =l and %mid, 0xffffffff
  %mid_hi =l shr %mid, 32
  %prod_lo =l shl %mid_lo, 32
  %prod_lo =l or %prod_lo, %p00_lo
  %prod_hi =l add %p01_hi, %p10_hi
  %prod_hi =l add %prod_hi, %p11_lo
  %prod_hi =l add %prod_hi, %mid_hi
  %k =l add %i, %j
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %rk_new =l add %rk_old, %prod_lo
  storel %rk_new, %rk_addr
  %carry =w cultl %rk_new, %rk_old
  %k =l add %k, 1
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %prod_hi_carry =l add %prod_hi, %carry
  %rk_new =l add %rk_old, %prod_hi_carry
  storel %rk_new, %rk_addr
  %carry =w cultl %rk_new, %rk_old
  %k =l add %k, 1
@carry_loop
  jnz %carry @carry_add @j_next
@carry_add
  %k_cond =w csltl %k, %total
  jnz %k_cond @carry_cont @j_next
@carry_cont
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %rk_new =l add %rk_old, 1
  storel %rk_new, %rk_addr
  %carry =w cultl %rk_new, %rk_old
  %k =l add %k, 1
  jmp @carry_loop
@j_next
  %j =l add %j, 1
  jmp @j_loop
@i_next
  %i =l add %i, 1
  jmp @outer_loop
@find_result
  %count =l copy %total
@count_loop
  %count_cond =w csgtl %count, 0
  jnz %count_cond @count_body @ret_zero
@ret_zero
  ret 0
@count_body
  %count_m1 =l add %count, -1
  %count_offset =l shl %count_m1, 3
  %count_addr =l add %r, %count_offset
  %count_val =l loadl %count_addr
  %is_nonzero =w cnel %count_val, 0
  jnz %is_nonzero @ret_count @count_dec
@count_dec
  %count =l add %count, -1
  jmp @count_loop
@ret_count
  ret %count
}
