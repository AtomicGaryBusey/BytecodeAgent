export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
  %na_zero =w ceql %na, 0
  jnz %na_zero @zero_result
  %nb_zero =w ceql %nb, 0
  jnz %nb_zero @zero_result
  
  %total =l add %na, %nb
  
  %i =l copy 0
@clear_loop
  %cond =w csltl %i, %total
  jnz %cond @clear_body
  jmp @multiply_loop_outer
  
@clear_body
  %offset =l shl %i, 3
  %addr =l add %r, %offset
  storel 0, %addr
  %i =l add %i, 1
  jmp @clear_loop
  
@multiply_loop_outer
  %i =l copy 0
@i_loop
  %i_cond =w csltl %i, %na
  jnz %i_cond @i_body
  jmp @count_significant
  
@i_body
  %i_offset =l shl %i, 3
  %ai_addr =l add %a, %i_offset
  %ai =l loadl %ai_addr
  
  %j =l copy 0
@j_loop
  %j_cond =w csltl %j, %nb
  jnz %j_cond @j_body
  jmp @i_next
  
@j_body
  %j_offset =l shl %j, 3
  %bj_addr =l add %b, %j_offset
  %bj =l loadl %bj_addr
  
  %ai_lo =l and %ai, 0xffffffff
  %ai_hi =l shr %ai, 32
  %bj_lo =l and %bj, 0xffffffff
  %bj_hi =l shr %bj, 32
  
  %p0 =l mul %ai_lo, %bj_lo
  %p1 =l mul %ai_lo, %bj_hi
  %p2 =l mul %ai_hi, %bj_lo
  %p3 =l mul %ai_hi, %bj_hi
  
  %p0_lo =l and %p0, 0xffffffff
  %p0_hi =l shr %p0, 32
  
  %p1_lo =l and %p1, 0xffffffff
  %p1_hi =l shr %p1, 32
  
  %p2_lo =l and %p2, 0xffffffff
  %p2_hi =l shr %p2, 32
  
  %p3_lo =l and %p3, 0xffffffff
  %p3_hi =l shr %p3, 32
  
  %mid =l add %p0_hi, %p1_lo
  %mid =l add %mid, %p2_lo
  
  %mid_lo =l and %mid, 0xffffffff
  %mid_hi =l shr %mid, 32
  
  %prod_lo =l shl %mid_lo, 32
  %prod_lo =l or %p0_lo, %prod_lo
  
  %prod_hi =l add %p1_hi, %p2_hi
  %prod_hi =l add %prod_hi, %p3_lo
  %prod_hi =l add %prod_hi, %mid_hi
  
  %k =l add %i, %j
  
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %rk_new =l add %rk_old, %prod_lo
  storel %rk_new, %rk_addr
  %carry1 =w cultl %rk_new, %rk_old
  
  %k =l add %k, 1
  
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %prod_hi_carry =l add %prod_hi, %carry1
  %rk_new =l add %rk_old, %prod_hi_carry
  storel %rk_new, %rk_addr
  %carry2 =w cultl %rk_new, %rk_old
  
  %k =l add %k, 1
  
@carry_loop
  jnz %carry2 @carry_body
  jmp @j_next
  
@carry_body
  %k_cond =w csltl %k, %total
  jnz %k_cond @carry_add
  jmp @j_next
  
@carry_add
  %k_offset =l shl %k, 3
  %rk_addr =l add %r, %k_offset
  %rk_old =l loadl %rk_addr
  %rk_new =l add %rk_old, 1
  storel %rk_new, %rk_addr
  %carry_new =w cultl %rk_new, %rk_old
  %k =l add %k, 1
  %carry2 =w copy %carry_new
  jmp @carry_loop
  
@j_next
  %j =l add %j, 1
  jmp @j_loop
  
@i_next
  %i =l add %i, 1
  jmp @i_loop
  
@count_significant
  %count =l copy %total
  jmp @count_loop
  
@count_loop
  %count_cond =w csgtl %count, 0
  jnz %count_cond @count_body
  ret 0
  
@count_body
  %count_minus_1 =l add %count, -1
  %count_offset =l shl %count_minus_1, 3
  %count_addr =l add %r, %count_offset
  %count_val =l loadl %count_addr
  %is_nonzero =w cnel %count_val, 0
  jnz %is_nonzero @return_count
  
  %count =l add %count, -1
  jmp @count_loop
  
@return_count
  ret %count
  
@zero_result
  %total =l add %na, %nb
  %i =l copy 0
@zero_clear_loop
  %cond =w csltl %i, %total
  jnz %cond @zero_clear_body
  ret 0
  
@zero_clear_body
  %offset =l shl %i, 3
  %addr =l add %r, %offset
  storel 0, %addr
  %i =l add %i, 1
  jmp @zero_clear_loop
}
