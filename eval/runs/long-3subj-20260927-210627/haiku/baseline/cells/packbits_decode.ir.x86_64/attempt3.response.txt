export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    %zero_val =l sub %n, %n
    jmp @main_loop

@main_loop
    %pos =l phi @entry %zero_val, @noop %pos_1, @lit_done %pos_lit, @rep_done %pos_rep
    %out_pos =l phi @entry %zero_val, @noop %out_pos_1, @lit_done %out_pos_lit, @rep_done %out_pos_rep
    
    %cond =w csltl %pos, %n
    jnz %cond @read_header
    
    %overflow =w csgtl %out_pos, %cap
    jnz %overflow @return_overflow
    ret %out_pos

@return_overflow
    ret -1

@read_header
    %h_addr =l add %src, %pos
    %h =w loadsb %h_addr
    %pos_next =l add %pos, 1
    
    %is_neg =w csltw %h, 0
    jnz %is_neg @check_m128
    jmp @literal

@check_m128
    %is_m128 =w ceqw %h, -128
    jnz %is_m128 @noop
    jmp @repeat

@noop
    %pos_1 =l copy %pos_next
    %out_pos_1 =l copy %out_pos
    jmp @main_loop
    
@literal
    %count_w =w add %h, 1
    %count =l sextw %count_w
    
    %need_end =l add %pos_next, %count
    %has_src =w cslel %need_end, %n
    jnz %has_src @lit_copy
    ret -2

@lit_copy
    %lit_i =l copy %zero_val
    jmp @lit_loop

@lit_loop
    %cond =w csltl %lit_i, %count
    jnz %cond @lit_body
    jmp @lit_done

@lit_body
    %src_idx =l add %pos_next, %lit_i
    %src_addr =l add %src, %src_idx
    %byte =w loadub %src_addr
    
    %out_idx =l add %out_pos, %lit_i
    %can_write =w csltl %out_idx, %cap
    jnz %can_write @lit_write
    jmp @lit_next

@lit_write
    %out_addr =l add %out, %out_idx
    storeb %byte, %out_addr
    jmp @lit_next

@lit_next
    %lit_i =l add %lit_i, 1
    jmp @lit_loop

@lit_done
    %pos_lit =l add %pos_next, %count
    %out_pos_lit =l add %out_pos, %count
    jmp @main_loop

@repeat
    %count_w =w sub 1, %h
    %count =l sextw %count_w
    
    %need_end =l add %pos_next, 1
    %has_src =w cslel %need_end, %n
    jnz %has_src @rep_read_byte
    ret -2

@rep_read_byte
    %byte_addr =l add %src, %pos_next
    %byte =w loadub %byte_addr
    jmp @rep_loop_start

@rep_loop_start
    %rep_i =l copy %zero_val
    jmp @rep_loop

@rep_loop
    %cond =w csltl %rep_i, %count
    jnz %cond @rep_body
    jmp @rep_done

@rep_body
    %out_idx =l add %out_pos, %rep_i
    %can_write =w csltl %out_idx, %cap
    jnz %can_write @rep_write
    jmp @rep_next

@rep_write
    %out_addr =l add %out, %out_idx
    storeb %byte, %out_addr
    jmp @rep_next

@rep_next
    %rep_i =l add %rep_i, 1
    jmp @rep_loop

@rep_done
    %pos_rep =l add %pos_next, 1
    %out_pos_rep =l add %out_pos, %count
    jmp @main_loop
}
