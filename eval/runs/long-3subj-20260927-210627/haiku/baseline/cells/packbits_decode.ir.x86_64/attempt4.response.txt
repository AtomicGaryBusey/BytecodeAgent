export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    %pos_init =l copy 0
    %out_pos_init =l copy 0
    jmp @loop

@loop
    %pos =l phi @entry %pos_init, @skip_packet %pos_skip, @lit_end %pos_lit, @rep_end %pos_rep
    %out_pos =l phi @entry %out_pos_init, @skip_packet %out_pos_skip, @lit_end %out_pos_lit, @rep_end %out_pos_rep
    
    %cond_end =w csge %pos, %n
    jnz %cond_end @finish
    
    %h_ptr =l add %src, %pos
    %h =w loadsb %h_ptr
    
    %cond_neg =w cslt %h, 0
    jnz %cond_neg @check_m128
    jmp @handle_lit

@check_m128
    %cond_m128 =w ceq %h, -128
    jnz %cond_m128 @skip_packet
    jmp @handle_rep

@skip_packet
    %pos_skip =l add %pos, 1
    %out_pos_skip =l copy %out_pos
    jmp @loop

@handle_lit
    %lit_cnt_w =w add %h, 1
    %lit_cnt =l cast %lit_cnt_w
    
    %lit_end_pos =l add %pos, 1
    %lit_end_pos2 =l add %lit_end_pos, %lit_cnt
    
    %cond_src =w csle %lit_end_pos2, %n
    jnz %cond_src @lit_init
    ret -2

@lit_init
    %lit_i_init =l copy 0
    jmp @lit_body

@lit_body
    %lit_i =l phi @lit_init %lit_i_init, @lit_inc %lit_i_inc
    
    %cond_lit_end =w csge %lit_i, %lit_cnt
    jnz %cond_lit_end @lit_end
    
    %src_off =l add %pos, 1
    %src_off2 =l add %src_off, %lit_i
    %src_b =w loadub %src_off2
    
    %out_off =l add %out_pos, %lit_i
    
    %cond_cap =w cslt %out_off, %cap
    jnz %cond_cap @lit_wr
    jmp @lit_inc

@lit_wr
    %out_ptr =l add %out, %out_off
    storeb %src_b, %out_ptr
    jmp @lit_inc

@lit_inc
    %lit_i_inc =l add %lit_i, 1
    jmp @lit_body

@lit_end
    %pos_lit =l add %pos, 1
    %pos_lit2 =l add %pos_lit, %lit_cnt
    %out_pos_lit =l add %out_pos, %lit_cnt
    jmp @loop

@handle_rep
    %rep_cnt_w =w sub 1, %h
    %rep_cnt =l cast %rep_cnt_w
    
    %rep_end_check =l add %pos, 2
    
    %cond_rep_src =w csle %rep_end_check, %n
    jnz %cond_rep_src @rep_byte
    ret -2

@rep_byte
    %rep_ptr =l add %src, %pos
    %rep_ptr2 =l add %rep_ptr, 1
    %rep_b =w loadub %rep_ptr2
    jmp @rep_init

@rep_init
    %rep_i_init =l copy 0
    jmp @rep_body

@rep_body
    %rep_i =l phi @rep_init %rep_i_init, @rep_inc %rep_i_inc
    
    %cond_rep_end =w csge %rep_i, %rep_cnt
    jnz %cond_rep_end @rep_end
    
    %out_off_rep =l add %out_pos, %rep_i
    
    %cond_cap_rep =w cslt %out_off_rep, %cap
    jnz %cond_cap_rep @rep_wr
    jmp @rep_inc

@rep_wr
    %out_ptr_rep =l add %out, %out_off_rep
    storeb %rep_b, %out_ptr_rep
    jmp @rep_inc

@rep_inc
    %rep_i_inc =l add %rep_i, 1
    jmp @rep_body

@rep_end
    %pos_rep =l add %pos, 2
    %out_pos_rep =l add %out_pos, %rep_cnt
    jmp @loop

@finish
    %cond_overflow =w csgt %out_pos, %cap
    jnz %cond_overflow @ret_neg1
    ret %out_pos

@ret_neg1
    ret -1
}
