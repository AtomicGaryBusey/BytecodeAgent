export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @main_loop

@main_loop
    %pos =l phi @entry 0, @lit_continue %pos, @rep_continue %pos, @noop_continue %pos
    %out_pos =l phi @entry 0, @lit_continue %out_pos, @rep_continue %out_pos, @noop_continue %out_pos
    
    %cmp =w csltl %pos, %n
    jnz %cmp @read
    %overflow =w csltl %cap, %out_pos
    jnz %overflow @return_overflow
    ret %out_pos

@read
    %h_ptr =l add %src, %pos
    %h =w loadsb %h_ptr
    %h_l =l sextw %h
    %pos_new =l add %pos, 1
    
    %is_lit =w csgtw %h, -1
    jnz %is_lit @literal
    
    %is_noop =w ceqw %h, -128
    jnz %is_noop @noop_continue
    
    jmp @repeat

@noop_continue
    jmp @main_loop

@literal
    %count =l add %h_l, 1
    
    %need =l add %pos_new, %count
    %ok =w cslel %need, %n
    jnz %ok @lit_write_start
    ret -2

@lit_write_start
    jmp @lit_loop

@lit_loop
    %i =l phi @lit_write_start 0, @lit_next_byte %i_new
    
    %cond =w csltl %i, %count
    jnz %cond @lit_loop_body
    
    %pos =l add %pos_new, %count
    %out_pos =l add %out_pos, %count
    jmp @lit_continue

@lit_loop_body
    %src_off =l add %pos_new, %i
    %byte =w loadub %src_off
    
    %out_off =l add %out_pos, %i
    
    %in_bounds =w csltl %out_off, %cap
    jnz %in_bounds @lit_do_write
    jmp @lit_next_byte

@lit_do_write
    %out_addr =l add %out, %out_off
    storeb %byte, %out_addr

@lit_next_byte
    %i_new =l add %i, 1
    jmp @lit_loop

@lit_continue
    jmp @main_loop

@repeat
    %count =l sub 1, %h_l
    
    %need =l add %pos_new, 1
    %ok =w cslel %need, %n
    jnz %ok @rep_read
    ret -2

@rep_read
    %byte_addr =l add %src, %pos_new
    %byte =w loadub %byte_addr
    %pos_after_byte =l add %pos_new, 1
    
    jmp @rep_loop

@rep_loop
    %i =l phi @rep_read 0, @rep_next_byte %i_new
    
    %cond =w csltl %i, %count
    jnz %cond @rep_loop_body
    
    %pos =l copy %pos_after_byte
    %out_pos =l add %out_pos, %count
    jmp @rep_continue

@rep_loop_body
    %out_off =l add %out_pos, %i
    
    %in_bounds =w csltl %out_off, %cap
    jnz %in_bounds @rep_do_write
    jmp @rep_next_byte

@rep_do_write
    %out_addr =l add %out, %out_off
    storeb %byte, %out_addr

@rep_next_byte
    %i_new =l add %i, 1
    jmp @rep_loop

@rep_continue
    jmp @main_loop

@return_overflow
    ret -1
}
