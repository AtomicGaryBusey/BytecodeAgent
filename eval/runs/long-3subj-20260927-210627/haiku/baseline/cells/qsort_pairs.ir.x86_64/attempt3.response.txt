export function l $qsort_pairs(l %a, l %n) {
@entry
	%one =l const 1
	%zero =l const 0
	%size_pair =l const 16
	%size_value =l const 8
	
	%cond_n =w cslel %n, %one
	jnz %cond_n, @return_zero
	
	%piv_k =l loadl %a
	%piv_v_addr =l add l %a, %size_value
	%piv_v =l loadl %piv_v_addr
	
	jmp @partition_init

@partition_init
	jmp @partition_loop

@partition_loop
	%i_curr =l phi @partition_init %one, @partition_inc %i_next
	%p_curr =l phi @partition_init %zero, @partition_inc %p_next
	
	%cond_i =w csltl %i_curr, %n
	jz %cond_i, @final_swap
	
	%i_offset =l mul l %i_curr, %size_pair
	%i_key_addr =l add l %a, %i_offset
	%i_val_addr =l add l %i_key_addr, %size_value
	%i_key =l loadl %i_key_addr
	%i_val =l loadl %i_val_addr
	
	%cond1 =w csltl %i_key, %piv_k
	jnz %cond1, @do_swap
	
	%cond2 =w ceql %i_key, %piv_k
	jz %cond2, @no_swap
	
	%cond3 =w csltl %i_val, %piv_v
	jz %cond3, @no_swap

@do_swap
	%p_inc =l add l %p_curr, %one
	%p_offset =l mul l %p_inc, %size_pair
	%p_key_addr =l add l %a, %p_offset
	%p_val_addr =l add l %p_key_addr, %size_value
	%p_key =l loadl %p_key_addr
	%p_val =l loadl %p_val_addr
	
	storel %i_key, %p_key_addr
	storel %i_val, %p_val_addr
	storel %p_key, %i_key_addr
	storel %p_val, %i_val_addr
	
	jmp @partition_inc

@no_swap
	jmp @partition_inc

@partition_inc
	%p_next =l phi @do_swap %p_inc, @no_swap %p_curr
	%i_next =l add l %i_curr, %one
	jmp @partition_loop

@final_swap
	%p_offset_final =l mul l %p_curr, %size_pair
	%p_key_addr_final =l add l %a, %p_offset_final
	%p_val_addr_final =l add l %p_key_addr_final, %size_value
	%p_key_final =l loadl %p_key_addr_final
	%p_val_final =l loadl %p_val_addr_final
	
	storel %piv_k, %p_key_addr_final
	storel %piv_v, %p_val_addr_final
	storel %p_key_final, %a
	storel %p_val_final, %piv_v_addr
	
	%res_left =l call $qsort_pairs(l %a, l %p_curr)
	
	%p_inc_final =l add l %p_curr, %one
	%right_offset =l mul l %p_inc_final, %size_pair
	%right_addr =l add l %a, %right_offset
	%right_count =l sub l %n, %p_inc_final
	
	%res_right =l call $qsort_pairs(l %right_addr, l %right_count)
	
	jmp @return_zero

@return_zero
	ret 0
}
