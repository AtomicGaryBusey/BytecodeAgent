export function l $qsort_pairs(l %a, l %n) {
@entry
	%cond_n =w cslel %n, 1
	jnz %cond_n, @return_zero
	
	%piv_k =l loadl %a
	%piv_v_addr =l add %a, 8
	%piv_v =l loadl %piv_v_addr
	
	jmp @partition_init

@partition_init
	jmp @partition_loop

@partition_loop
	%i_curr =l phi @partition_init 1, @partition_inc %i_next
	%p_curr =l phi @partition_init 0, @partition_inc %p_next
	
	%cond_i =w csltl %i_curr, %n
	jz %cond_i, @final_swap
	
	%i_offset =l mul %i_curr, 16
	%i_key_addr =l add %a, %i_offset
	%i_val_addr =l add %i_key_addr, 8
	%i_key =l loadl %i_key_addr
	%i_val =l loadl %i_val_addr
	
	%cond1 =w csltl %i_key, %piv_k
	jnz %cond1, @do_swap
	
	%cond2 =w ceql %i_key, %piv_k
	jz %cond2, @no_swap
	
	%cond3 =w csltl %i_val, %piv_v
	jz %cond3, @no_swap

@do_swap
	%p_inc =l add %p_curr, 1
	%p_offset =l mul %p_inc, 16
	%p_key_addr =l add %a, %p_offset
	%p_val_addr =l add %p_key_addr, 8
	%p_key =l loadl %p_key_addr
	%p_val =l loadl %p_val_addr
	
	storel %i_key, %p_key_addr
	storel %i_val, %p_val_addr
	storel %p_key, %i_key_addr
	storel %p_val, %i_val_addr
	
	jmp @partition_inc

@no_swap
	jmp @partition_inc

@partition_inc
	%p_next =l phi @do_swap %p_inc, @no_swap %p_curr
	%i_next =l add %i_curr, 1
	jmp @partition_loop

@final_swap
	%p_offset_final =l mul %p_curr, 16
	%p_key_addr_final =l add %a, %p_offset_final
	%p_val_addr_final =l add %p_key_addr_final, 8
	%p_key_final =l loadl %p_key_addr_final
	%p_val_final =l loadl %p_val_addr_final
	
	storel %piv_k, %p_key_addr_final
	storel %piv_v, %p_val_addr_final
	storel %p_key_final, %a
	storel %p_val_final, %piv_v_addr
	
	%res_left =l call $qsort_pairs(l %a, l %p_curr)
	
	%p_inc_final =l add %p_curr, 1
	%right_offset =l mul %p_inc_final, 16
	%right_addr =l add %a, %right_offset
	%right_count =l sub %n, %p_inc_final
	
	%res_right =l call $qsort_pairs(l %right_addr, l %right_count)
	
	jmp @return_zero

@return_zero
	ret 0
}
