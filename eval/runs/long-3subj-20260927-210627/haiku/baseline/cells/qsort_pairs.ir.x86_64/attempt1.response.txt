export function l $qsort_pairs(l %a, l %n) {
@entry
	%n_le_1 =w csle %n, 1
	jnz %n_le_1, @done
	
	%piv_k =l loadl %a
	%piv_v_addr =l add l %a, 8
	%piv_v =l loadl %piv_v_addr
	
	%i0 =l const 1
	%p0 =l const 0
	
	jmp @loop_cond

@loop_cond
	%i =l phi @entry %i0, @loop_tail %i1
	%p =l phi @entry %p0, @loop_tail %p1
	
	%cond_loop =w cslt %i, %n
	jz %cond_loop, @swap_pivot
	
	%offset_i =l mul l %i, 16
	%addr_i_k =l add l %a, %offset_i
	%addr_i_v =l add l %addr_i_k, 8
	%key_i =l loadl %addr_i_k
	%val_i =l loadl %addr_i_v
	
	%cmp1 =w cslt %key_i, %piv_k
	jnz %cmp1, @need_swap
	%cmp2 =w ceq %key_i, %piv_k
	jz %cmp2, @no_need_swap
	%cmp3 =w cslt %val_i, %piv_v
	jz %cmp3, @no_need_swap
	
@need_swap
	%p_plus_1 =l add l %p, 1
	%offset_p_plus_1 =l mul l %p_plus_1, 16
	%addr_p_plus_1_k =l add l %a, %offset_p_plus_1
	%addr_p_plus_1_v =l add l %addr_p_plus_1_k, 8
	%key_p_plus_1 =l loadl %addr_p_plus_1_k
	%val_p_plus_1 =l loadl %addr_p_plus_1_v
	
	storel %key_i, %addr_p_plus_1_k
	storel %val_i, %addr_p_plus_1_v
	storel %key_p_plus_1, %addr_i_k
	storel %val_p_plus_1, %addr_i_v
	
	%p_swapped =l copy %p_plus_1
	jmp @loop_tail

@no_need_swap
	%p_swapped =l copy %p
	jmp @loop_tail

@loop_tail
	%p1 =l phi @need_swap %p_swapped, @no_need_swap %p_swapped
	%i1 =l add l %i, 1
	jmp @loop_cond

@swap_pivot
	%offset_p_final =l mul l %p, 16
	%addr_p_final_k =l add l %a, %offset_p_final
	%addr_p_final_v =l add l %addr_p_final_k, 8
	%key_p_final =l loadl %addr_p_final_k
	%val_p_final =l loadl %addr_p_final_v
	
	storel %piv_k, %addr_p_final_k
	storel %piv_v, %addr_p_final_v
	storel %key_p_final, %a
	storel %val_p_final, %piv_v_addr
	
	%res_left =l call $qsort_pairs(l %a, l %p)
	
	%p_plus_1_final =l add l %p, 1
	%offset_right =l mul l %p_plus_1_final, 16
	%addr_right =l add l %a, %offset_right
	%n_right =l sub l %n, %p
	%n_right_final =l sub l %n_right, 1
	
	%res_right =l call $qsort_pairs(l %addr_right, l %n_right_final)
	
	jmp @done

@done
	ret 0
}
