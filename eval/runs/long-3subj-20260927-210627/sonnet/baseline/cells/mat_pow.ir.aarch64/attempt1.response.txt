export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nsq =l mul %n, %n
	%scroff =l mul %nsq, 8
	%scratch =l add %tmp, %scroff
	%base =l copy %tmp
	jmp @copybase_top
@copybase_top
	%ib1 =l phi @start 0, @copybase_body %ib2
	%cmpb =w csltl %ib1, %nsq
	jnz %cmpb, @copybase_body, @initout_i_pre
@copybase_body
	%boff =l mul %ib1, 8
	%bsrc =l add %a, %boff
	%bdst =l add %base, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	%ib2 =l add %ib1, 1
	jmp @copybase_top
@initout_i_pre
	jmp @initout_i_top
@initout_i_top
	%ii1 =l phi @initout_i_pre 0, @initout_i_next %ii2
	%cmpi =w csltl %ii1, %n
	jnz %cmpi, @initout_j_pre, @mainloop_pre
@initout_j_pre
	jmp @initout_j_top
@initout_j_top
	%jj1 =l phi @initout_j_pre 0, @initout_j_next %jj2
	%cmpj =w csltl %jj1, %n
	jnz %cmpj, @initout_j_body, @initout_i_next
@initout_j_body
	%oidx1 =l mul %ii1, %n
	%oidx2 =l add %oidx1, %jj1
	%ooff =l mul %oidx2, 8
	%oaddr =l add %out, %ooff
	%eqij =w ceql %ii1, %jj1
	jnz %eqij, @initout_seteq, @initout_setne
@initout_seteq
	storel 1, %oaddr
	jmp @initout_j_next
@initout_setne
	storel 0, %oaddr
	jmp @initout_j_next
@initout_j_next
	%jj2 =l add %jj1, 1
	jmp @initout_j_top
@initout_i_next
	%ii2 =l add %ii1, 1
	jmp @initout_i_top
@mainloop_pre
	jmp @mainloop_top
@mainloop_top
	%kcur =l phi @mainloop_pre %k, @loopend %knext
	%kz =w ceql %kcur, 0
	jnz %kz, @donezero, @mainloop_body
@mainloop_body
	%kand =l and %kcur, 1
	%kandz =w ceql %kand, 0
	jnz %kandz, @skipmul1, @domul1_pre
@domul1_pre
	jmp @mul1_i_pre
@mul1_i_pre
	jmp @mul1_i_top
@mul1_i_top
	%m1i1 =l phi @mul1_i_pre 0, @mul1_i_next %m1i2
	%m1ic =w csltl %m1i1, %n
	jnz %m1ic, @mul1_j_pre, @copy1_pre
@mul1_j_pre
	jmp @mul1_j_top
@mul1_j_top
	%m1j1 =l phi @mul1_j_pre 0, @mul1_j_next %m1j2
	%m1jc =w csltl %m1j1, %n
	jnz %m1jc, @mul1_t_pre, @mul1_i_next
@mul1_t_pre
	jmp @mul1_t_top
@mul1_t_top
	%m1t1 =l phi @mul1_t_pre 0, @mul1_t_body %m1t2
	%m1s1 =l phi @mul1_t_pre 0, @mul1_t_body %m1s2
	%m1tc =w csltl %m1t1, %n
	jnz %m1tc, @mul1_t_body, @mul1_t_done
@mul1_t_body
	%m1poff1 =l mul %m1i1, %n
	%m1poff2 =l add %m1poff1, %m1t1
	%m1poff =l mul %m1poff2, 8
	%m1paddr =l add %out, %m1poff
	%m1pval =l loadl %m1paddr
	%m1qoff1 =l mul %m1t1, %n
	%m1qoff2 =l add %m1qoff1, %m1j1
	%m1qoff =l mul %m1qoff2, 8
	%m1qaddr =l add %base, %m1qoff
	%m1qval =l loadl %m1qaddr
	%m1prod =l mul %m1pval, %m1qval
	%m1s2 =l add %m1s1, %m1prod
	%m1t2 =l add %m1t1, 1
	jmp @mul1_t_top
@mul1_t_done
	%m1roff1 =l mul %m1i1, %n
	%m1roff2 =l add %m1roff1, %m1j1
	%m1roff =l mul %m1roff2, 8
	%m1raddr =l add %scratch, %m1roff
	storel %m1s1, %m1raddr
	jmp @mul1_j_next
@mul1_j_next
	%m1j2 =l add %m1j1, 1
	jmp @mul1_j_top
@mul1_i_next
	%m1i2 =l add %m1i1, 1
	jmp @mul1_i_top
@copy1_pre
	jmp @copy1_top
@copy1_top
	%c1i1 =l phi @copy1_pre 0, @copy1_body %c1i2
	%c1ic =w csltl %c1i1, %nsq
	jnz %c1ic, @copy1_body, @domul2_pre
@copy1_body
	%c1off =l mul %c1i1, 8
	%c1src =l add %scratch, %c1off
	%c1dst =l add %out, %c1off
	%c1val =l loadl %c1src
	storel %c1val, %c1dst
	%c1i2 =l add %c1i1, 1
	jmp @copy1_top
@skipmul1
	jmp @domul2_pre
@domul2_pre
	jmp @mul2_i_pre
@mul2_i_pre
	jmp @mul2_i_top
@mul2_i_top
	%m2i1 =l phi @mul2_i_pre 0, @mul2_i_next %m2i2
	%m2ic =w csltl %m2i1, %n
	jnz %m2ic, @mul2_j_pre, @copy2_pre
@mul2_j_pre
	jmp @mul2_j_top
@mul2_j_top
	%m2j1 =l phi @mul2_j_pre 0, @mul2_j_next %m2j2
	%m2jc =w csltl %m2j1, %n
	jnz %m2jc, @mul2_t_pre, @mul2_i_next
@mul2_t_pre
	jmp @mul2_t_top
@mul2_t_top
	%m2t1 =l phi @mul2_t_pre 0, @mul2_t_body %m2t2
	%m2s1 =l phi @mul2_t_pre 0, @mul2_t_body %m2s2
	%m2tc =w csltl %m2t1, %n
	jnz %m2tc, @mul2_t_body, @mul2_t_done
@mul2_t_body
	%m2poff1 =l mul %m2i1, %n
	%m2poff2 =l add %m2poff1, %m2t1
	%m2poff =l mul %m2poff2, 8
	%m2paddr =l add %base, %m2poff
	%m2pval =l loadl %m2paddr
	%m2qoff1 =l mul %m2t1, %n
	%m2qoff2 =l add %m2qoff1, %m2j1
	%m2qoff =l mul %m2qoff2, 8
	%m2qaddr =l add %base, %m2qoff
	%m2qval =l loadl %m2qaddr
	%m2prod =l mul %m2pval, %m2qval
	%m2s2 =l add %m2s1, %m2prod
	%m2t2 =l add %m2t1, 1
	jmp @mul2_t_top
@mul2_t_done
	%m2roff1 =l mul %m2i1, %n
	%m2roff2 =l add %m2roff1, %m2j1
	%m2roff =l mul %m2roff2, 8
	%m2raddr =l add %scratch, %m2roff
	storel %m2s1, %m2raddr
	jmp @mul2_j_next
@mul2_j_next
	%m2j2 =l add %m2j1, 1
	jmp @mul2_j_top
@mul2_i_next
	%m2i2 =l add %m2i1, 1
	jmp @mul2_i_top
@copy2_pre
	jmp @copy2_top
@copy2_top
	%c2i1 =l phi @copy2_pre 0, @copy2_body %c2i2
	%c2ic =w csltl %c2i1, %nsq
	jnz %c2ic, @copy2_body, @loopend
@copy2_body
	%c2off =l mul %c2i1, 8
	%c2src =l add %scratch, %c2off
	%c2dst =l add %base, %c2off
	%c2val =l loadl %c2src
	storel %c2val, %c2dst
	%c2i2 =l add %c2i1, 1
	jmp @copy2_top
@loopend
	%knext =l shr %kcur, 1
	jmp @mainloop_top
@donezero
	%nsq2 =l mul %nsq, 2
	jmp @zero_top
@zero_top
	%zi1 =l phi @donezero 0, @zero_body %zi2
	%zic =w csltl %zi1, %nsq2
	jnz %zic, @zero_body, @ret0
@zero_body
	%zoff =l mul %zi1, 8
	%zaddr =l add %tmp, %zoff
	storel 0, %zaddr
	%zi2 =l add %zi1, 1
	jmp @zero_top
@ret0
	ret 0
}
