export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %size =l mul %n, %n
    %bytesize =l mul %size, 8
    %tmp1 =l add %tmp, %bytesize
    %twosize =l mul %size, 2
    jmp @zloop
@zloop
    %zi =l phi @start 0, @zbody %zi2
    %zc =w csltl %zi, %twosize
    jnz %zc, @zbody, @zdone
@zbody
    %zoff =l mul %zi, 8
    %zp =l add %tmp, %zoff
    storel 0, %zp
    %zi2 =l add %zi, 1
    jmp @zloop
@zdone
    jmp @cloop
@cloop
    %ci =l phi @zdone 0, @cbody %ci2
    %cc =w csltl %ci, %size
    jnz %cc, @cbody, @cdone
@cbody
    %coff =l mul %ci, 8
    %cap =l add %a, %coff
    %cval =l loadl %cap
    %ctp =l add %tmp, %coff
    storel %cval, %ctp
    %ci2 =l add %ci, 1
    jmp @cloop
@cdone
    jmp @idi
@idi
    %ii =l phi @cdone 0, @idjdone %ii2
    %idic =w csltl %ii, %n
    jnz %idic, @idibody, @iddone
@idibody
    jmp @idj
@idj
    %jj =l phi @idibody 0, @idjbody %jj2
    %idjc =w csltl %jj, %n
    jnz %idjc, @idjbody, @idjdone
@idjbody
    %ieqj =w ceql %ii, %jj
    %idval =l extsw %ieqj
    %idxij =l mul %ii, %n
    %idxij2 =l add %idxij, %jj
    %idxoff =l mul %idxij2, 8
    %idp =l add %out, %idxoff
    storel %idval, %idp
    %jj2 =l add %jj, 1
    jmp @idj
@idjdone
    %ii2 =l add %ii, 1
    jmp @idi
@iddone
    jmp @mloop
@mloop
    %kc =l phi @iddone %k, @szdone %kc2
    %kne0 =w cnel %kc, 0
    jnz %kne0, @mbody, @mend
@mbody
    %kbit =l and %kc, 1
    %kbitnz =w cnel %kbit, 0
    jnz %kbitnz, @mulres, @msquare
@mulres
    jmp @mri
@mri
    %ri =l phi @mulres 0, @mrjdone %ri2
    %ric =w csltl %ri, %n
    jnz %ric, @mribody, @mridone
@mribody
    jmp @mrj
@mrj
    %rj =l phi @mribody 0, @mrtdone %rj2
    %rjc =w csltl %rj, %n
    jnz %rjc, @mrjbody, @mrjdone
@mrjbody
    jmp @mrt
@mrt
    %rt =l phi @mrjbody 0, @mrtbody %rt2
    %racc =l phi @mrjbody 0, @mrtbody %racc2
    %rtc =w csltl %rt, %n
    jnz %rtc, @mrtbody, @mrtdone
@mrtbody
    %rxidx =l mul %ri, %n
    %rxidx2 =l add %rxidx, %rt
    %rxoff =l mul %rxidx2, 8
    %rxp =l add %out, %rxoff
    %rxval =l loadl %rxp
    %ryidx =l mul %rt, %n
    %ryidx2 =l add %ryidx, %rj
    %ryoff =l mul %ryidx2, 8
    %ryp =l add %tmp, %ryoff
    %ryval =l loadl %ryp
    %rprod =l mul %rxval, %ryval
    %racc2 =l add %racc, %rprod
    %rt2 =l add %rt, 1
    jmp @mrt
@mrtdone
    %rdidx =l mul %ri, %n
    %rdidx2 =l add %rdidx, %rj
    %rdoff =l mul %rdidx2, 8
    %rdp =l add %tmp1, %rdoff
    storel %racc, %rdp
    %rj2 =l add %rj, 1
    jmp @mrj
@mrjdone
    %ri2 =l add %ri, 1
    jmp @mri
@mridone
    jmp @rcloop
@rcloop
    %rci =l phi @mridone 0, @rcbody %rci2
    %rcc =w csltl %rci, %size
    jnz %rcc, @rcbody, @rcdone
@rcbody
    %rcoff =l mul %rci, 8
    %rcsrc =l add %tmp1, %rcoff
    %rcval =l loadl %rcsrc
    %rcdst =l add %out, %rcoff
    storel %rcval, %rcdst
    %rci2 =l add %rci, 1
    jmp @rcloop
@rcdone
    jmp @rzloop
@rzloop
    %rzi =l phi @rcdone 0, @rzbody %rzi2
    %rzc =w csltl %rzi, %size
    jnz %rzc, @rzbody, @rzdone
@rzbody
    %rzoff =l mul %rzi, 8
    %rzp =l add %tmp1, %rzoff
    storel 0, %rzp
    %rzi2 =l add %rzi, 1
    jmp @rzloop
@rzdone
    jmp @msquare
@msquare
    jmp @sqi
@sqi
    %si =l phi @msquare 0, @sqjdone %si2
    %sic =w csltl %si, %n
    jnz %sic, @sqibody, @sqidone
@sqibody
    jmp @sqj
@sqj
    %sj =l phi @sqibody 0, @sqtdone %sj2
    %sjc =w csltl %sj, %n
    jnz %sjc, @sqjbody, @sqjdone
@sqjbody
    jmp @sqt
@sqt
    %st =l phi @sqjbody 0, @sqtbody %st2
    %sacc =l phi @sqjbody 0, @sqtbody %sacc2
    %stc =w csltl %st, %n
    jnz %stc, @sqtbody, @sqtdone
@sqtbody
    %sxidx =l mul %si, %n
    %sxidx2 =l add %sxidx, %st
    %sxoff =l mul %sxidx2, 8
    %sxp =l add %tmp, %sxoff
    %sxval =l loadl %sxp
    %syidx =l mul %st, %n
    %syidx2 =l add %syidx, %sj
    %syoff =l mul %syidx2, 8
    %syp =l add %tmp, %syoff
    %syval =l loadl %syp
    %sprod =l mul %sxval, %syval
    %sacc2 =l add %sacc, %sprod
    %st2 =l add %st, 1
    jmp @sqt
@sqtdone
    %sdidx =l mul %si, %n
    %sdidx2 =l add %sdidx, %sj
    %sdoff =l mul %sdidx2, 8
    %sdp =l add %tmp1, %sdoff
    storel %sacc, %sdp
    %sj2 =l add %sj, 1
    jmp @sqj
@sqjdone
    %si2 =l add %si, 1
    jmp @sqi
@sqidone
    jmp @scloop
@scloop
    %sci =l phi @sqidone 0, @scbody %sci2
    %scc =w csltl %sci, %size
    jnz %scc, @scbody, @scdone
@scbody
    %scoff =l mul %sci, 8
    %scsrc =l add %tmp1, %scoff
    %scval =l loadl %scsrc
    %scdst =l add %tmp, %scoff
    storel %scval, %scdst
    %sci2 =l add %sci, 1
    jmp @scloop
@scdone
    jmp @szloop
@szloop
    %szi =l phi @scdone 0, @szbody %szi2
    %szc =w csltl %szi, %size
    jnz %szc, @szbody, @szdone
@szbody
    %szoff =l mul %szi, 8
    %szp =l add %tmp1, %szoff
    storel 0, %szp
    %szi2 =l add %szi, 1
    jmp @szloop
@szdone
    %kc2 =l shr %kc, 1
    jmp @mloop
@mend
    jmp @fzloop
@fzloop
    %fzi =l phi @mend 0, @fzbody %fzi2
    %fzc =w csltl %fzi, %size
    jnz %fzc, @fzbody, @fzdone
@fzbody
    %fzoff =l mul %fzi, 8
    %fzp =l add %tmp, %fzoff
    storel 0, %fzp
    %fzi2 =l add %fzi, 1
    jmp @fzloop
@fzdone
    ret 0
}
