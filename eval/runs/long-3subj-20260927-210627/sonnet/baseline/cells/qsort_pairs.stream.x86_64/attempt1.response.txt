qsort_pairs:
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rbp
    sub rsp, 8

    mov r12, rdi
    mov r13, rsi

.Lloop:
    cmp r13, 2
    jl .Ldone

    mov r8, r13
    dec r8

    mov r10, r8
    shl r10, 4
    add r10, r12
    mov r14, [r10]
    mov r15, [r10+8]

    xor rax, rax
    xor rcx, rcx

.Lpart_loop:
    cmp rcx, r8
    jge .Lpart_done

    mov r10, rcx
    shl r10, 4
    add r10, r12

    mov r9, [r10]
    cmp r9, r14
    jl .Lswap_needed
    jg .Lno_swap
    mov r9, [r10+8]
    cmp r9, r15
    jl .Lswap_needed
    jmp .Lno_swap

.Lswap_needed:
    mov r11, rax
    shl r11, 4
    add r11, r12

    mov rsi, [r10]
    mov rdi, [r10+8]
    mov rdx, [r11]
    mov rbx, [r11+8]
    mov [r10], rdx
    mov [r10+8], rbx
    mov [r11], rsi
    mov [r11+8], rdi

    inc rax

.Lno_swap:
    inc rcx
    jmp .Lpart_loop

.Lpart_done:
    mov r10, r8
    shl r10, 4
    add r10, r12

    mov r11, rax
    shl r11, 4
    add r11, r12

    mov rsi, [r10]
    mov rdi, [r10+8]
    mov rdx, [r11]
    mov rbx, [r11+8]
    mov [r10], rdx
    mov [r10+8], rbx
    mov [r11], rsi
    mov [r11+8], rdi

    mov rbx, rax

    mov r15, r13
    sub r15, rbx
    dec r15

    mov r14, rbx
    inc r14
    shl r14, 4
    add r14, r12

    cmp rbx, r15
    jg .Lrecurse_right

    mov rdi, r12
    mov rsi, rbx
    call qsort_pairs
    mov r12, r14
    mov r13, r15
    jmp .Lloop

.Lrecurse_right:
    mov rdi, r14
    mov rsi, r15
    call qsort_pairs
    mov r13, rbx
    jmp .Lloop

.Ldone:
    xor eax, eax
    add rsp, 8
    pop rbp
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
