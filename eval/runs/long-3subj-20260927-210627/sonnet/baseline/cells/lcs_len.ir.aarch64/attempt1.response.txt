export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @zero_check
@zero_check
    %jz =l phi @start 0, @zero_body %jz1
    %contz =w cslel %jz, %n
    jnz %contz, @zero_body, @outer_check
@zero_body
    %zoff =l mul %jz, 8
    %zaddr =l add %row, %zoff
    storel 0, %zaddr
    %jz1 =l add %jz, 1
    jmp @zero_check
@outer_check
    %i =l phi @zero_check 0, @outer_next %i1
    %conti =w csltl %i, %m
    jnz %conti, @outer_body, @done
@outer_body
    %xaddr =l add %x, %i
    %xb =w loadub %xaddr
    jmp @inner_check
@inner_check
    %j =l phi @outer_body 1, @inner_next %jnext
    %diag =l phi @outer_body 0, @inner_next %diagnext
    %contj =w cslel %j, %n
    jnz %contj, @inner_body, @outer_next
@inner_body
    %roff =l mul %j, 8
    %raddr =l add %row, %roff
    %up =l loadl %raddr
    %jm1 =l sub %j, 1
    %yaddr =l add %y, %jm1
    %yb =w loadub %yaddr
    %eq =w ceqw %xb, %yb
    %poff =l mul %jm1, 8
    %paddr =l add %row, %poff
    %left =l loadl %paddr
    jnz %eq, @set_diag, @check_max
@set_diag
    %newval1 =l add %diag, 1
    storel %newval1, %raddr
    jmp @inner_next
@check_max
    %gtc =w csgtl %up, %left
    jnz %gtc, @use_up, @use_left
@use_up
    storel %up, %raddr
    jmp @inner_next
@use_left
    storel %left, %raddr
    jmp @inner_next
@inner_next
    %diagnext =l copy %up
    %jnext =l add %j, 1
    jmp @inner_check
@outer_next
    %i1 =l add %i, 1
    jmp @outer_check
@done
    %noff =l mul %n, 8
    %naddr =l add %row, %noff
    %res =l loadl %naddr
    ret %res
}
