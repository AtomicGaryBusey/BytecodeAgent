export function l $qsort_pairs(l %a, l %n) {
@start
    %pi =l alloc8 8
    %pj =l alloc8 8
    %lt2 =w csltl %n, 2
    jnz %lt2, @done, @setup
@setup
    %half =l div %n, 2
    %pivoff =l mul %half, 16
    %pivkptr =l add %a, %pivoff
    %pivk =l loadl %pivkptr
    %pivvoff =l add %pivoff, 8
    %pivvptr =l add %a, %pivvoff
    %pivv =l loadl %pivvptr
    %negone =l sub 0, 1
    storel %negone, %pi
    storel %n, %pj
    jmp @findi
@findi
    %ic =l loadl %pi
    %ic2 =l add %ic, 1
    storel %ic2, %pi
    %ioff =l mul %ic2, 16
    %ikptr =l add %a, %ioff
    %ik =l loadl %ikptr
    %ilt1 =w csltl %ik, %pivk
    jnz %ilt1, @findi, @findi_eq
@findi_eq
    %ieq =w ceql %ik, %pivk
    jnz %ieq, @findi_val, @findj
@findi_val
    %ivoff =l add %ioff, 8
    %ivptr =l add %a, %ivoff
    %iv =l loadl %ivptr
    %ivlt =w csltl %iv, %pivv
    jnz %ivlt, @findi, @findj
@findj
    %jc =l loadl %pj
    %jc2 =l sub %jc, 1
    storel %jc2, %pj
    %joff =l mul %jc2, 16
    %jkptr =l add %a, %joff
    %jk =l loadl %jkptr
    %jgt1 =w csltl %pivk, %jk
    jnz %jgt1, @findj, @findj_eq
@findj_eq
    %jeq =w ceql %jk, %pivk
    jnz %jeq, @findj_val, @checksplit
@findj_val
    %jvoff =l add %joff, 8
    %jvptr =l add %a, %jvoff
    %jv =l loadl %jvptr
    %jvgt =w csltl %pivv, %jv
    jnz %jvgt, @findj, @checksplit
@checksplit
    %ifinal =l loadl %pi
    %jfinal =l loadl %pj
    %ige =w csgel %ifinal, %jfinal
    jnz %ige, @split, @swap
@swap
    %ioff2 =l mul %ifinal, 16
    %ikptr2 =l add %a, %ioff2
    %ivptr2 =l add %ikptr2, 8
    %joff2 =l mul %jfinal, 16
    %jkptr2 =l add %a, %joff2
    %jvptr2 =l add %jkptr2, 8
    %tk =l loadl %ikptr2
    %tv =l loadl %ivptr2
    %tk2 =l loadl %jkptr2
    %tv2 =l loadl %jvptr2
    storel %tk2, %ikptr2
    storel %tv2, %ivptr2
    storel %tk, %jkptr2
    storel %tv, %jvptr2
    jmp @findi
@split
    %splitidx =l add %jfinal, 1
    %rsize =l sub %n, %splitidx
    %roff =l mul %splitidx, 16
    %rptr =l add %a, %roff
    %r1 =l call $qsort_pairs(l %a, l %splitidx)
    %r2 =l call $qsort_pairs(l %rptr, l %rsize)
    jmp @done
@done
    ret 0
}
