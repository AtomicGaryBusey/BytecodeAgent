export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zinit
@zinit
	%j =l phi @start 0, @zbody %jn
	%zcnd =w cslel %j, %n
	jnz %zcnd, @zbody, @outer_pre
@zbody
	%zoff =l mul %j, 8
	%zaddr =l add %row, %zoff
	storel 0, %zaddr
	%jn =l add %j, 1
	jmp @zinit
@outer_pre
	jmp @outer
@outer
	%i =l phi @outer_pre 0, @outer_end %in
	%ocnd =w csltl %i, %m
	jnz %ocnd, @outer_body, @done
@outer_body
	%xaddr =l add %x, %i
	%xi =w loadub %xaddr
	jmp @inner
@inner
	%diagv =l phi @outer_body 0, @store %diagn
	%jv =l phi @outer_body 1, @store %jnn
	%icnd2 =w cslel %jv, %n
	jnz %icnd2, @inner_body, @outer_end
@inner_body
	%rjoff =l mul %jv, 8
	%rjaddr =l add %row, %rjoff
	%jm1 =l sub %jv, 1
	%rjm1off =l mul %jm1, 8
	%rjm1addr =l add %row, %rjm1off
	%up =l loadl %rjaddr
	%prevrow =l loadl %rjm1addr
	%yaddr =l add %y, %jm1
	%yj =w loadub %yaddr
	%eqcnd =w ceql %xi, %yj
	%gtcnd =w csgtl %up, %prevrow
	jnz %gtcnd, @take_up, @take_prev
@take_up
	%mxu =l copy %up
	jmp @merge
@take_prev
	%mxp =l copy %prevrow
	jmp @merge
@merge
	%mxv =l phi @take_up %mxu, @take_prev %mxp
	%diagp1 =l add %diagv, 1
	jnz %eqcnd, @seteq, @setmax
@seteq
	%nv1 =l copy %diagp1
	jmp @store
@setmax
	%nv2 =l copy %mxv
	jmp @store
@store
	%newval =l phi @seteq %nv1, @setmax %nv2
	storel %newval, %rjaddr
	%diagn =l copy %up
	%jnn =l add %jv, 1
	jmp @inner
@outer_end
	%in =l add %i, 1
	jmp @outer
@done
	%noff =l mul %n, 8
	%naddr =l add %row, %noff
	%result =l loadl %naddr
	ret %result
}
