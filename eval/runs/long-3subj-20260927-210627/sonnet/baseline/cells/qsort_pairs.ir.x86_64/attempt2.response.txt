export function l $qsort_pairs(l %a0, l %n0) {
@start
	jmp @loop
@loop
	%av =l phi @start %a0, @done %a2
	%nv =l phi @start %n0, @done %n2
	%small =w csltl %nv, 2
	jnz %small, @ret0, @setup
@ret0
	%z =l copy 0
	ret %z
@setup
	%mid =l div %nv, 2
	%midoff =l mul %mid, 16
	%midp =l add %av, %midoff
	%pk =l loadl %midp
	%pvoff =l add %midp, 8
	%pv =l loadl %pvoff
	%lastidx =l sub %nv, 1
	%lastoff =l mul %lastidx, 16
	%j0 =l add %av, %lastoff
	%i0 =l copy %av
	jmp @pbody
@pbody
	%ic =l phi @setup %i0, @doswap %i2
	%jc =l phi @setup %j0, @doswap %j2
	jmp @scani
@scani
	%ii =l phi @pbody %ic, @advi %ii2
	%ik =l loadl %ii
	%ivoff =l add %ii, 8
	%iv =l loadl %ivoff
	%ltk =w csltl %ik, %pk
	%eqk1 =w ceql %ik, %pk
	%ltv =w csltl %iv, %pv
	%andi =w and %eqk1, %ltv
	%isless =w or %ltk, %andi
	jnz %isless, @advi, @scanj
@advi
	%ii2 =l add %ii, 16
	jmp @scani
@scanj
	%jj =l phi @scani %jc, @advj %jj2
	%jk =l loadl %jj
	%jvoff =l add %jj, 8
	%jv =l loadl %jvoff
	%gtk =w csgtl %jk, %pk
	%eqk2 =w ceql %jk, %pk
	%gtv =w csgtl %jv, %pv
	%andj =w and %eqk2, %gtv
	%isgreater =w or %gtk, %andj
	jnz %isgreater, @advj, @checkij
@advj
	%jj2 =l sub %jj, 16
	jmp @scanj
@checkij
	%le =w cslel %ii, %jj
	jnz %le, @doswap, @done
@doswap
	%ikk =l loadl %ii
	%ii8 =l add %ii, 8
	%ivv =l loadl %ii8
	%jkk =l loadl %jj
	%jj8 =l add %jj, 8
	%jvv =l loadl %jj8
	storel %jkk, %ii
	storel %jvv, %ii8
	storel %ikk, %jj
	storel %ivv, %jj8
	%i2 =l add %ii, 16
	%j2 =l sub %jj, 16
	jmp @pbody
@done
	%diff =l sub %ii, %av
	%leftCount =l div %diff, 16
	%rightCount =l sub %nv, %leftCount
	%a2 =l copy %ii
	%n2 =l copy %rightCount
	%dummy =l call $qsort_pairs(l %av, l %leftCount)
	jmp @loop
}
