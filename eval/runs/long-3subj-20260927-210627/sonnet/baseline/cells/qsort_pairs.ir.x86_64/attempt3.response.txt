export function l $qsort_pairs(l %a0, l %n0) {
@start
	jmp @loop
@loop
	%av =l phi @start %a0, @cont %av2
	%nv =l phi @start %n0, @cont %nv2
	%small =w csltl %nv, 2
	jnz %small, @ret0, @setup
@ret0
	ret 0
@setup
	%pk =l loadl %av
	%pvp =l add %av, 8
	%pv =l loadl %pvp
	%i0 =l sub %av, 16
	%noff =l mul %nv, 16
	%j0 =l add %av, %noff
	jmp @pbody
@pbody
	%ic =l phi @setup %i0, @doswap %ii3
	%jc =l phi @setup %j0, @doswap %jj3
	jmp @scani
@scani
	%iiprev =l phi @pbody %ic, @scani %ii2
	%ii2 =l add %iiprev, 16
	%ik =l loadl %ii2
	%ivp =l add %ii2, 8
	%iv =l loadl %ivp
	%ltk =w csltl %ik, %pk
	%eqk1 =w ceql %ik, %pk
	%ltv =w csltl %iv, %pv
	%andi =w and %eqk1, %ltv
	%isless =w or %ltk, %andi
	jnz %isless, @scani, @scanj
@scanj
	%jjprev =l phi @scani %jc, @scanj %jj2
	%jj2 =l sub %jjprev, 16
	%jk =l loadl %jj2
	%jvp =l add %jj2, 8
	%jv =l loadl %jvp
	%gtk =w csgtl %jk, %pk
	%eqk2 =w ceql %jk, %pk
	%gtv =w csgtl %jv, %pv
	%andj =w and %eqk2, %gtv
	%isgreater =w or %gtk, %andj
	jnz %isgreater, @scanj, @checkij
@checkij
	%crossed =w csgel %ii2, %jj2
	jnz %crossed, @done, @doswap
@doswap
	%ikk =l loadl %ii2
	%ii8 =l add %ii2, 8
	%ivv =l loadl %ii8
	%jkk =l loadl %jj2
	%jj8 =l add %jj2, 8
	%jvv =l loadl %jj8
	storel %jkk, %ii2
	storel %jvv, %ii8
	storel %ikk, %jj2
	storel %ivv, %jj8
	%ii3 =l copy %ii2
	%jj3 =l copy %jj2
	jmp @pbody
@done
	%diff =l sub %jj2, %av
	%leftCount =l div %diff, 16
	%leftCount1 =l add %leftCount, 1
	%rightPtr =l add %jj2, 16
	%rightCount =l sub %nv, %leftCount1
	%dummy =l call $qsort_pairs(l %av, l %leftCount1)
	jmp @cont
@cont
	%av2 =l copy %rightPtr
	%nv2 =l copy %rightCount
	jmp @loop
}
