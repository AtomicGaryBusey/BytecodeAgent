mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r9, rsi
    imul r9, rsi
    imul rax, r9, 8
    lea r10, [r8+rax]
    xor r11, r11
.Lzero_out:
    cmp r11, r9
    jge .Lout_zero_done
    mov qword ptr [rcx + r11*8], 0
    inc r11
    jmp .Lzero_out
.Lout_zero_done:
    xor r11, r11
.Ldiag:
    cmp r11, rsi
    jge .Ldiag_done
    mov r12, r11
    imul r12, rsi
    add r12, r11
    mov qword ptr [rcx + r12*8], 1
    inc r11
    jmp .Ldiag
.Ldiag_done:
    xor r11, r11
.Lcopy_a:
    cmp r11, r9
    jge .Lcopy_a_done
    mov r12, qword ptr [rdi + r11*8]
    mov qword ptr [r8 + r11*8], r12
    inc r11
    jmp .Lcopy_a
.Lcopy_a_done:
.Lloop:
    test rdx, rdx
    jz .Lloop_end
    test rdx, 1
    jz .Lskip_mult
    mov r11, rcx
    mov r12, r8
    call .Lmul
    xor r13, r13
.Lcopy_out:
    cmp r13, r9
    jge .Lcopy_out_done
    mov r14, qword ptr [r10 + r13*8]
    mov qword ptr [rcx + r13*8], r14
    inc r13
    jmp .Lcopy_out
.Lcopy_out_done:
.Lskip_mult:
    shr rdx, 1
    test rdx, rdx
    jz .Lloop_end
    mov r11, r8
    mov r12, r8
    call .Lmul
    xor r13, r13
.Lcopy_b:
    cmp r13, r9
    jge .Lcopy_b_done
    mov r14, qword ptr [r10 + r13*8]
    mov qword ptr [r8 + r13*8], r14
    inc r13
    jmp .Lcopy_b
.Lcopy_b_done:
    jmp .Lloop
.Lloop_end:
    mov r11, r9
    add r11, r9
    xor r13, r13
.Lzero_tmp:
    cmp r13, r11
    jge .Lzero_tmp_done
    mov qword ptr [r8 + r13*8], 0
    inc r13
    jmp .Lzero_tmp
.Lzero_tmp_done:
    mov rax, 0
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.Lmul:
    xor r13, r13
.Lmul_i:
    cmp r13, rsi
    jge .Lmul_iend
    mov r14, r13
    imul r14, rsi
    xor rbx, rbx
.Lmul_j:
    cmp rbx, rsi
    jge .Lmul_jend
    xor rax, rax
    xor rbp, rbp
.Lmul_t:
    cmp rbp, rsi
    jge .Lmul_tend
    lea rdi, [r14+rbp]
    mov r15, qword ptr [r11 + rdi*8]
    mov rdi, rbp
    imul rdi, rsi
    add rdi, rbx
    imul r15, qword ptr [r12 + rdi*8]
    add rax, r15
    inc rbp
    jmp .Lmul_t
.Lmul_tend:
    lea rdi, [r14+rbx]
    mov qword ptr [r10 + rdi*8], rax
    inc rbx
    jmp .Lmul_j
.Lmul_jend:
    inc r13
    jmp .Lmul_i
.Lmul_iend:
    ret
