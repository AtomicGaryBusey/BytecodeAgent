mat_pow:
    // x0=a, x1=n, x2=k, x3=out, x4=tmp
    mul x5, x1, x1          // nn = n*n
    add x6, x4, x5, lsl #3  // P = tmp + nn*8   (B is tmp itself, x4)

    // zero out matrix (nn elements)
    mov x8, #0
.Lzero_out:
    cmp x8, x5
    b.ge .Lzero_out_done
    str xzr, [x3, x8, lsl #3]
    add x8, x8, #1
    b .Lzero_out
.Lzero_out_done:

    // set diagonal of out to 1
    add x9, x1, #1           // n+1
    mov x8, #0                // i=0
.Ldiag_loop:
    cmp x8, x1
    b.ge .Ldiag_done
    mul x10, x8, x9           // idx = i*(n+1)
    mov x11, #1
    str x11, [x3, x10, lsl #3]
    add x8, x8, #1
    b .Ldiag_loop
.Ldiag_done:

    // copy a into B (tmp)
    mov x8, #0
.Lcopy_a_loop:
    cmp x8, x5
    b.ge .Lcopy_a_done
    ldr x9, [x0, x8, lsl #3]
    str x9, [x4, x8, lsl #3]
    add x8, x8, #1
    b .Lcopy_a_loop
.Lcopy_a_done:

.Lmain_loop:
    cbz x2, .Lmain_done
    and x7, x2, #1
    cbz x7, .Lskip_out_mul

    // P = out * B
    mov x8, #0                // i
.LoutB_i:
    cmp x8, x1
    b.ge .LoutB_i_done
    mul x13, x8, x1
    mov x9, #0                // j
.LoutB_j:
    cmp x9, x1
    b.ge .LoutB_j_done
    mov x11, #0                // sum
    mov x10, #0                // t
.LoutB_t:
    cmp x10, x1
    b.ge .LoutB_t_done
    add x15, x13, x10
    ldr x16, [x3, x15, lsl #3]     // out(i,t)
    mul x14, x10, x1
    add x14, x14, x9
    ldr x17, [x4, x14, lsl #3]     // B(t,j)
    mul x16, x16, x17
    add x11, x11, x16
    add x10, x10, #1
    b .LoutB_t
.LoutB_t_done:
    add x15, x13, x9
    str x11, [x6, x15, lsl #3]
    add x9, x9, #1
    b .LoutB_j
.LoutB_j_done:
    add x8, x8, #1
    b .LoutB_i
.LoutB_i_done:

    // copy P into out
    mov x8, #0
.Lcopy_P_out:
    cmp x8, x5
    b.ge .Lcopy_P_out_done
    ldr x9, [x6, x8, lsl #3]
    str x9, [x3, x8, lsl #3]
    add x8, x8, #1
    b .Lcopy_P_out
.Lcopy_P_out_done:

.Lskip_out_mul:

    // P = B * B
    mov x8, #0
.LBB_i:
    cmp x8, x1
    b.ge .LBB_i_done
    mul x13, x8, x1
    mov x9, #0
.LBB_j:
    cmp x9, x1
    b.ge .LBB_j_done
    mov x11, #0
    mov x10, #0
.LBB_t:
    cmp x10, x1
    b.ge .LBB_t_done
    add x15, x13, x10
    ldr x16, [x4, x15, lsl #3]     // B(i,t)
    mul x14, x10, x1
    add x14, x14, x9
    ldr x17, [x4, x14, lsl #3]     // B(t,j)
    mul x16, x16, x17
    add x11, x11, x16
    add x10, x10, #1
    b .LBB_t
.LBB_t_done:
    add x15, x13, x9
    str x11, [x6, x15, lsl #3]
    add x9, x9, #1
    b .LBB_j
.LBB_j_done:
    add x8, x8, #1
    b .LBB_i
.LBB_i_done:

    // copy P into B
    mov x8, #0
.Lcopy_P_B:
    cmp x8, x5
    b.ge .Lcopy_P_B_done
    ldr x9, [x6, x8, lsl #3]
    str x9, [x4, x8, lsl #3]
    add x8, x8, #1
    b .Lcopy_P_B
.Lcopy_P_B_done:

    lsr x2, x2, #1
    b .Lmain_loop

.Lmain_done:
    // zero tmp (2*nn elements)
    lsl x5, x5, #1        // total = 2*nn
    mov x8, #0
.Lzero_tmp:
    cmp x8, x5
    b.ge .Lzero_tmp_done
    str xzr, [x4, x8, lsl #3]
    add x8, x8, #1
    b .Lzero_tmp
.Lzero_tmp_done:

    mov x0, #0
    ret
