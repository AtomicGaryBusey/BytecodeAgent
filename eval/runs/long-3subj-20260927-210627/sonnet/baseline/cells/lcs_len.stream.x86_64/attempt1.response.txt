lcs_len:
    push rbx
    push r12

    # initialize row[0..n] = 0
    xor r10, r10
.Linit:
    cmp r10, rcx
    jg .Linit_done
    mov qword ptr [r8 + r10*8], 0
    inc r10
    jmp .Linit
.Linit_done:

    xor r9, r9
.Louter:
    cmp r9, rsi
    jge .Ldone
    movzx ebx, byte ptr [rdi + r9]
    xor r11, r11
    mov r10, 1
.Linner:
    cmp r10, rcx
    jg .Linner_done
    mov rax, [r8 + r10*8]
    movzx r12d, byte ptr [rdx + r10 - 1]
    cmp bl, r12b
    jne .Lnoteq
    lea r12, [r11 + 1]
    jmp .Lstore
.Lnoteq:
    mov r12, [r8 + r10*8 - 8]
    cmp rax, r12
    cmovge r12, rax
.Lstore:
    mov [r8 + r10*8], r12
    mov r11, rax
    inc r10
    jmp .Linner
.Linner_done:
    inc r9
    jmp .Louter
.Ldone:
    mov rax, [r8 + rcx*8]
    pop r12
    pop rbx
    ret
