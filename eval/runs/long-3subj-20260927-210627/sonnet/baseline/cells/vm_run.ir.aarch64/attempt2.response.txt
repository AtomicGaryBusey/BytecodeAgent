export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pcptr =l alloc8 8
    %spptr =l alloc8 8
    storel 0, %pcptr
    storel 0, %spptr
    jmp @loop
@loop
    %pc0 =l loadl %pcptr
    %sp0 =l loadl %spptr
    %atend =w ceql %pc0, %len
    jnz %atend, @done, @body
@body
    %codeoff =l mul %pc0, 8
    %codeaddr =l add %code, %codeoff
    %op =l loadl %codeaddr
    %eq0 =w ceql %op, 0
    jnz %eq0, @op_halt, @chk1
@chk1
    %eq1 =w ceql %op, 1
    jnz %eq1, @op_push, @chk2
@chk2
    %eq2 =w ceql %op, 2
    jnz %eq2, @op_add, @chk3
@chk3
    %eq3 =w ceql %op, 3
    jnz %eq3, @op_sub, @chk4
@chk4
    %eq4 =w ceql %op, 4
    jnz %eq4, @op_mul, @chk5
@chk5
    %eq5 =w ceql %op, 5
    jnz %eq5, @op_dup, @chk6
@chk6
    %eq6 =w ceql %op, 6
    jnz %eq6, @op_swap, @chk7
@chk7
    %eq7 =w ceql %op, 7
    jnz %eq7, @op_drop, @chk8
@chk8
    %eq8 =w ceql %op, 8
    jnz %eq8, @op_over, @chk9
@chk9
    %eq9 =w ceql %op, 9
    jnz %eq9, @op_lt, @chk10
@chk10
    %eq10 =w ceql %op, 10
    jnz %eq10, @op_jz, @chk11
@chk11
    %eq11 =w ceql %op, 11
    jnz %eq11, @op_jmp, @op_default
@op_halt
    jmp @done
@op_push
    %pushvaddr =l add %codeaddr, 8
    %pushv =l loadl %pushvaddr
    %pushsoff =l mul %sp0, 8
    %pushsaddr =l add %stack, %pushsoff
    storel %pushv, %pushsaddr
    %pushsp =l add %sp0, 1
    storel %pushsp, %spptr
    %pushpc =l add %pc0, 2
    storel %pushpc, %pcptr
    jmp @loop
@op_add
    %addsp1 =l sub %sp0, 1
    %addoff1 =l mul %addsp1, 8
    %addaddr1 =l add %stack, %addoff1
    %addb =l loadl %addaddr1
    %addsp2 =l sub %sp0, 2
    %addoff2 =l mul %addsp2, 8
    %addaddr2 =l add %stack, %addoff2
    %adda =l loadl %addaddr2
    %addres =l add %adda, %addb
    storel %addres, %addaddr2
    storel %addsp1, %spptr
    %addpc =l add %pc0, 1
    storel %addpc, %pcptr
    jmp @loop
@op_sub
    %subsp1 =l sub %sp0, 1
    %suboff1 =l mul %subsp1, 8
    %subaddr1 =l add %stack, %suboff1
    %subb =l loadl %subaddr1
    %subsp2 =l sub %sp0, 2
    %suboff2 =l mul %subsp2, 8
    %subaddr2 =l add %stack, %suboff2
    %suba =l loadl %subaddr2
    %subres =l sub %suba, %subb
    storel %subres, %subaddr2
    storel %subsp1, %spptr
    %subpc =l add %pc0, 1
    storel %subpc, %pcptr
    jmp @loop
@op_mul
    %mulsp1 =l sub %sp0, 1
    %muloff1 =l mul %mulsp1, 8
    %muladdr1 =l add %stack, %muloff1
    %mulb =l loadl %muladdr1
    %mulsp2 =l sub %sp0, 2
    %muloff2 =l mul %mulsp2, 8
    %muladdr2 =l add %stack, %muloff2
    %mula =l loadl %muladdr2
    %mulres =l mul %mula, %mulb
    storel %mulres, %muladdr2
    storel %mulsp1, %spptr
    %mulpc =l add %pc0, 1
    storel %mulpc, %pcptr
    jmp @loop
@op_dup
    %dupsp1 =l sub %sp0, 1
    %dupoff1 =l mul %dupsp1, 8
    %dupaddr1 =l add %stack, %dupoff1
    %dupv =l loadl %dupaddr1
    %dupoff2 =l mul %sp0, 8
    %dupaddr2 =l add %stack, %dupoff2
    storel %dupv, %dupaddr2
    %dupsp =l add %sp0, 1
    storel %dupsp, %spptr
    %duppc =l add %pc0, 1
    storel %duppc, %pcptr
    jmp @loop
@op_swap
    %swsp1 =l sub %sp0, 1
    %swoff1 =l mul %swsp1, 8
    %swaddr1 =l add %stack, %swoff1
    %swv1 =l loadl %swaddr1
    %swsp2 =l sub %sp0, 2
    %swoff2 =l mul %swsp2, 8
    %swaddr2 =l add %stack, %swoff2
    %swv2 =l loadl %swaddr2
    storel %swv2, %swaddr1
    storel %swv1, %swaddr2
    %swpc =l add %pc0, 1
    storel %swpc, %pcptr
    jmp @loop
@op_drop
    %dropsp =l sub %sp0, 1
    storel %dropsp, %spptr
    %droppc =l add %pc0, 1
    storel %droppc, %pcptr
    jmp @loop
@op_over
    %ovsp2 =l sub %sp0, 2
    %ovoff2 =l mul %ovsp2, 8
    %ovaddr2 =l add %stack, %ovoff2
    %ovv =l loadl %ovaddr2
    %ovoff3 =l mul %sp0, 8
    %ovaddr3 =l add %stack, %ovoff3
    storel %ovv, %ovaddr3
    %ovsp =l add %sp0, 1
    storel %ovsp, %spptr
    %ovpc =l add %pc0, 1
    storel %ovpc, %pcptr
    jmp @loop
@op_lt
    %ltsp1 =l sub %sp0, 1
    %ltoff1 =l mul %ltsp1, 8
    %ltaddr1 =l add %stack, %ltoff1
    %ltb =l loadl %ltaddr1
    %ltsp2 =l sub %sp0, 2
    %ltoff2 =l mul %ltsp2, 8
    %ltaddr2 =l add %stack, %ltoff2
    %lta =l loadl %ltaddr2
    %ltcmp =w csltl %lta, %ltb
    %ltcmpl =l extuw %ltcmp
    storel %ltcmpl, %ltaddr2
    storel %ltsp1, %spptr
    %ltpc =l add %pc0, 1
    storel %ltpc, %pcptr
    jmp @loop
@op_jz
    %jzsp1 =l sub %sp0, 1
    %jzoff1 =l mul %jzsp1, 8
    %jzaddr1 =l add %stack, %jzoff1
    %jzv =l loadl %jzaddr1
    storel %jzsp1, %spptr
    %jztaddr =l add %codeaddr, 8
    %jzt =l loadl %jztaddr
    %jziszero =w ceql %jzv, 0
    jnz %jziszero, @jz_true, @jz_false
@jz_true
    storel %jzt, %pcptr
    jmp @loop
@jz_false
    %jzpc =l add %pc0, 2
    storel %jzpc, %pcptr
    jmp @loop
@op_jmp
    %jmptaddr =l add %codeaddr, 8
    %jmpt =l loadl %jmptaddr
    storel %jmpt, %pcptr
    jmp @loop
@op_default
    %defpc =l add %pc0, 1
    storel %defpc, %pcptr
    jmp @loop
@done
    %fsp =l loadl %spptr
    %ziptr =l alloc8 8
    storel %fsp, %ziptr
    jmp @zloop
@zloop
    %zi =l loadl %ziptr
    %zdone =w ceql %zi, %cap
    jnz %zdone, @zend, @zbody
@zbody
    %zoff =l mul %zi, 8
    %zaddr =l add %stack, %zoff
    storel 0, %zaddr
    %zi2 =l add %zi, 1
    storel %zi2, %ziptr
    jmp @zloop
@zend
    %isempty =w ceql %fsp, 0
    jnz %isempty, @ret_zero, @ret_top
@ret_zero
    ret 0
@ret_top
    %topi =l sub %fsp, 1
    %topoff =l mul %topi, 8
    %topaddr =l add %stack, %topoff
    %result =l loadl %topaddr
    ret %result
}
