export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%zi0 =l copy 0
	jmp @zloop
@zloop
	%zi =l phi @start %zi0, @zbody %zi2
	%zc =w csltl %zi, %total
	jnz %zc, @zbody, @zdone
@zbody
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi2 =l add %zi, 1
	jmp @zloop
@zdone
	%oi0 =l copy 0
	jmp @outer
@outer
	%oi =l phi @zdone %oi0, @inner_done %oi2
	%ocnd =w csltl %oi, %na
	jnz %ocnd, @outer_body, @finish_mul
@outer_body
	%aioff =l mul %oi, 8
	%aip =l add %a, %aioff
	%ai =l loadl %aip
	%carry0 =l copy 0
	%oj0 =l copy 0
	jmp @inner
@inner
	%oj =l phi @outer_body %oj0, @inner_body %oj2
	%carry =l phi @outer_body %carry0, @inner_body %carry2
	%icnd =w csltl %oj, %nb
	jnz %icnd, @inner_body, @inner_done
@inner_body
	%bjoff =l mul %oj, 8
	%bjp =l add %b, %bjoff
	%bj =l loadl %bjp
	%alo =l and %ai, 4294967295
	%ahi =l shr %ai, 32
	%blo =l and %bj, 4294967295
	%bhi =l shr %bj, 32
	%t0 =l mul %alo, %blo
	%t1 =l mul %alo, %bhi
	%t2 =l mul %ahi, %blo
	%t3 =l mul %ahi, %bhi
	%t0hi =l shr %t0, 32
	%t0lo =l and %t0, 4294967295
	%t1hi =l shr %t1, 32
	%t1lo =l and %t1, 4294967295
	%t2hi =l shr %t2, 32
	%t2lo =l and %t2, 4294967295
	%mida =l add %t0hi, %t1lo
	%midb =l add %mida, %t2lo
	%midlo =l and %midb, 4294967295
	%midcarry =l shr %midb, 32
	%midshift =l shl %midlo, 32
	%lo =l add %t0lo, %midshift
	%hia =l add %t3, %t1hi
	%hib =l add %hia, %t2hi
	%hi =l add %hib, %midcarry
	%tt =l add %lo, %carry
	%c0w =w cultl %tt, %lo
	%c0 =l extuw %c0w
	%hi2 =l add %hi, %c0
	%ridx =l add %oi, %oj
	%roff =l mul %ridx, 8
	%rp =l add %r, %roff
	%rv =l loadl %rp
	%s =l add %rv, %tt
	%c1w =w cultl %s, %rv
	%c1 =l extuw %c1w
	storel %s, %rp
	%carry2 =l add %hi2, %c1
	%oj2 =l add %oj, 1
	jmp @inner
@inner_done
	%fidx =l add %oi, %nb
	%foff =l mul %fidx, 8
	%fp =l add %r, %foff
	storel %carry, %fp
	%oi2 =l add %oi, 1
	jmp @outer
@finish_mul
	%k0 =l copy %total
	jmp @kloop
@kloop
	%k =l phi @finish_mul %k0, @kbody %knext
	%kgt =w csgtl %k, 0
	jnz %kgt, @kcheck, @kdone
@kcheck
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @kbody, @kdone
@kbody
	%knext =l sub %k, 1
	jmp @kloop
@kdone
	ret %k
}
