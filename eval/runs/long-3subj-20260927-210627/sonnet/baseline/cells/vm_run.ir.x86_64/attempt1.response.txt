export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcslot =l alloc8 8
	%spslot =l alloc8 8
	%resslot =l alloc8 8
	%islot =l alloc8 8
	storel 0, %pcslot
	storel 0, %spslot
	jmp @loop
@loop
	%pc =l loadl %pcslot
	%c0 =w ceql %pc, %len
	jnz %c0, @done, @cont
@cont
	%pc1 =l loadl %pcslot
	%codeoff =l mul %pc1, 8
	%codeptr =l add %code, %codeoff
	%op =l loadl %codeptr
	%isHalt =w ceql %op, 0
	jnz %isHalt, @done, @c1
@c1
	%isPush =w ceql %op, 1
	jnz %isPush, @push, @c2
@c2
	%isAdd =w ceql %op, 2
	jnz %isAdd, @add, @c3
@c3
	%isSub =w ceql %op, 3
	jnz %isSub, @sub, @c4
@c4
	%isMul =w ceql %op, 4
	jnz %isMul, @mul, @c5
@c5
	%isDup =w ceql %op, 5
	jnz %isDup, @dup, @c6
@c6
	%isSwap =w ceql %op, 6
	jnz %isSwap, @swap, @c7
@c7
	%isDrop =w ceql %op, 7
	jnz %isDrop, @drop, @c8
@c8
	%isOver =w ceql %op, 8
	jnz %isOver, @over, @c9
@c9
	%isLt =w ceql %op, 9
	jnz %isLt, @lt, @c10
@c10
	%isJz =w ceql %op, 10
	jnz %isJz, @jz, @c11
@c11
	%isJmp =w ceql %op, 11
	jnz %isJmp, @jmp1, @other
@other
	%pco =l loadl %pcslot
	%pcn =l add %pco, 1
	storel %pcn, %pcslot
	jmp @loop
@push
	%pcp =l loadl %pcslot
	%codeoffp =l mul %pcp, 8
	%opoff =l add %codeoffp, 8
	%opptr =l add %code, %opoff
	%v =l loadl %opptr
	%spp =l loadl %spslot
	%soff =l mul %spp, 8
	%sptr =l add %stack, %soff
	storel %v, %sptr
	%spn =l add %spp, 1
	storel %spn, %spslot
	%pcn2 =l add %pcp, 2
	storel %pcn2, %pcslot
	jmp @loop
@add
	%spa =l loadl %spslot
	%sp1a =l sub %spa, 1
	%sp2a =l sub %spa, 2
	%off1a =l mul %sp1a, 8
	%off2a =l mul %sp2a, 8
	%ptr1a =l add %stack, %off1a
	%ptr2a =l add %stack, %off2a
	%ba =l loadl %ptr1a
	%aa =l loadl %ptr2a
	%ra =l add %aa, %ba
	storel %ra, %ptr2a
	storel %sp1a, %spslot
	%pca =l loadl %pcslot
	%pcna =l add %pca, 1
	storel %pcna, %pcslot
	jmp @loop
@sub
	%spb =l loadl %spslot
	%sp1b =l sub %spb, 1
	%sp2b =l sub %spb, 2
	%off1b =l mul %sp1b, 8
	%off2b =l mul %sp2b, 8
	%ptr1b =l add %stack, %off1b
	%ptr2b =l add %stack, %off2b
	%bb =l loadl %ptr1b
	%ab =l loadl %ptr2b
	%rb =l sub %ab, %bb
	storel %rb, %ptr2b
	storel %sp1b, %spslot
	%pcb =l loadl %pcslot
	%pcnb =l add %pcb, 1
	storel %pcnb, %pcslot
	jmp @loop
@mul
	%spm =l loadl %spslot
	%sp1m =l sub %spm, 1
	%sp2m =l sub %spm, 2
	%off1m =l mul %sp1m, 8
	%off2m =l mul %sp2m, 8
	%ptr1m =l add %stack, %off1m
	%ptr2m =l add %stack, %off2m
	%bm =l loadl %ptr1m
	%am =l loadl %ptr2m
	%rm =l mul %am, %bm
	storel %rm, %ptr2m
	storel %sp1m, %spslot
	%pcm =l loadl %pcslot
	%pcnm =l add %pcm, 1
	storel %pcnm, %pcslot
	jmp @loop
@dup
	%spd =l loadl %spslot
	%sp1d =l sub %spd, 1
	%off1d =l mul %sp1d, 8
	%ptr1d =l add %stack, %off1d
	%ad =l loadl %ptr1d
	%off2d =l mul %spd, 8
	%ptr2d =l add %stack, %off2d
	storel %ad, %ptr2d
	%spnd =l add %spd, 1
	storel %spnd, %spslot
	%pcd =l loadl %pcslot
	%pcnd =l add %pcd, 1
	storel %pcnd, %pcslot
	jmp @loop
@swap
	%sps =l loadl %spslot
	%sp1s =l sub %sps, 1
	%sp2s =l sub %sps, 2
	%off1s =l mul %sp1s, 8
	%off2s =l mul %sp2s, 8
	%ptr1s =l add %stack, %off1s
	%ptr2s =l add %stack, %off2s
	%bs =l loadl %ptr1s
	%aswp =l loadl %ptr2s
	storel %bs, %ptr2s
	storel %aswp, %ptr1s
	%pcs =l loadl %pcslot
	%pcns =l add %pcs, 1
	storel %pcns, %pcslot
	jmp @loop
@drop
	%spo =l loadl %spslot
	%spno =l sub %spo, 1
	storel %spno, %spslot
	%pco2 =l loadl %pcslot
	%pcno2 =l add %pco2, 1
	storel %pcno2, %pcslot
	jmp @loop
@over
	%spv =l loadl %spslot
	%sp2v =l sub %spv, 2
	%off2v =l mul %sp2v, 8
	%ptr2v =l add %stack, %off2v
	%av =l loadl %ptr2v
	%off3v =l mul %spv, 8
	%ptr3v =l add %stack, %off3v
	storel %av, %ptr3v
	%spnv =l add %spv, 1
	storel %spnv, %spslot
	%pcv =l loadl %pcslot
	%pcnv =l add %pcv, 1
	storel %pcnv, %pcslot
	jmp @loop
@lt
	%spl =l loadl %spslot
	%sp1l =l sub %spl, 1
	%sp2l =l sub %spl, 2
	%off1l =l mul %sp1l, 8
	%off2l =l mul %sp2l, 8
	%ptr1l =l add %stack, %off1l
	%ptr2l =l add %stack, %off2l
	%bl =l loadl %ptr1l
	%al =l loadl %ptr2l
	%cmpl =w csltl %al, %bl
	%rl =l extsw %cmpl
	storel %rl, %ptr2l
	storel %sp1l, %spslot
	%pcl =l loadl %pcslot
	%pcnl =l add %pcl, 1
	storel %pcnl, %pcslot
	jmp @loop
@jz
	%spj =l loadl %spslot
	%sp1j =l sub %spj, 1
	%off1j =l mul %sp1j, 8
	%ptr1j =l add %stack, %off1j
	%vj =l loadl %ptr1j
	storel %sp1j, %spslot
	%pcj =l loadl %pcslot
	%iszero =w ceql %vj, 0
	jnz %iszero, @jz_true, @jz_false
@jz_true
	%codeoffj =l mul %pcj, 8
	%opoffj =l add %codeoffj, 8
	%opptrj =l add %code, %opoffj
	%tj =l loadl %opptrj
	storel %tj, %pcslot
	jmp @loop
@jz_false
	%pcnj =l add %pcj, 2
	storel %pcnj, %pcslot
	jmp @loop
@jmp1
	%pcjm =l loadl %pcslot
	%codeoffjm =l mul %pcjm, 8
	%opoffjm =l add %codeoffjm, 8
	%opptrjm =l add %code, %opoffjm
	%tjm =l loadl %opptrjm
	storel %tjm, %pcslot
	jmp @loop
@done
	%spf =l loadl %spslot
	%iszero2 =w ceql %spf, 0
	jnz %iszero2, @res_zero, @res_top
@res_top
	%spm1 =l sub %spf, 1
	%offr =l mul %spm1, 8
	%ptrr =l add %stack, %offr
	%result1 =l loadl %ptrr
	storel %result1, %resslot
	jmp @fillinit
@res_zero
	storel 0, %resslot
	jmp @fillinit
@fillinit
	%spf2 =l loadl %spslot
	storel %spf2, %islot
	jmp @fillloop
@fillloop
	%i =l loadl %islot
	%fc =w csltl %i, %cap
	jnz %fc, @fillbody, @ret
@fillbody
	%ioff =l mul %i, 8
	%iptr =l add %stack, %ioff
	storel 0, %iptr
	%in =l add %i, 1
	storel %in, %islot
	jmp @fillloop
@ret
	%result =l loadl %resslot
	ret %result
}
