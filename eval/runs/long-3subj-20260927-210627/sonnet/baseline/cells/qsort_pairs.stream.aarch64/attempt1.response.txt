qsort_pairs:
    sub sp, sp, #64
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]
    stp x23, x24, [sp, #32]
    stp x29, x30, [sp, #48]
    add x29, sp, #48

    mov x19, x0
    mov x20, x1

.Lloop:
    cmp x20, #2
    blt .Ldone

    sub x13, x20, #1
    lsl x11, x13, #4
    add x11, x19, x11
    ldr x0, [x11]
    ldr x1, [x11, #8]

    mov x9, #0
    mov x10, #0

.Lpart_loop:
    cmp x10, x13
    bge .Lpart_done

    lsl x15, x10, #4
    add x15, x19, x15
    ldr x12, [x15]
    ldr x14, [x15, #8]

    cmp x12, x0
    blt .Lless
    bgt .Lnotless
    cmp x14, x1
    blt .Lless
    b .Lnotless

.Lless:
    lsl x11, x9, #4
    add x11, x19, x11
    ldr x2, [x11]
    ldr x3, [x11, #8]
    str x12, [x11]
    str x14, [x11, #8]
    str x2, [x15]
    str x3, [x15, #8]
    add x9, x9, #1

.Lnotless:
    add x10, x10, #1
    b .Lpart_loop

.Lpart_done:
    lsl x11, x9, #4
    add x11, x19, x11
    lsl x15, x13, #4
    add x15, x19, x15
    ldr x2, [x11]
    ldr x3, [x11, #8]
    ldr x4, [x15]
    ldr x5, [x15, #8]
    str x4, [x11]
    str x5, [x11, #8]
    str x2, [x15]
    str x3, [x15, #8]

    sub x14, x13, x9
    add x15, x9, #1
    lsl x15, x15, #4
    add x15, x19, x15

    cmp x9, x14
    bgt .Lright_smaller

.Lleft_le:
    mov x21, x15
    mov x22, x14
    mov x0, x19
    mov x1, x9
    bl qsort_pairs
    mov x19, x21
    mov x20, x22
    b .Lloop

.Lright_smaller:
    mov x23, x9
    mov x0, x15
    mov x1, x14
    bl qsort_pairs
    mov x20, x23
    b .Lloop

.Ldone:
    mov x0, #0
    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    ldp x23, x24, [sp, #32]
    ldp x29, x30, [sp, #48]
    add sp, sp, #64
    ret
