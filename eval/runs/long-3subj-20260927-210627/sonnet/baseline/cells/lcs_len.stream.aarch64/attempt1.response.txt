lcs_len:
    sub sp, sp, #64
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]
    stp x23, x24, [sp, #32]
    stp x25, x26, [sp, #48]

    mov x19, x0    // x
    mov x20, x1    // m
    mov x21, x2    // y
    mov x22, x3    // n
    mov x23, x4    // row

    mov x9, #0
.Lzero:
    cmp x9, x22
    b.gt .Lzero_done
    str xzr, [x23, x9, lsl #3]
    add x9, x9, #1
    b .Lzero
.Lzero_done:

    mov x24, #0    // i
.Louter:
    cmp x24, x20
    b.ge .Louter_done
    ldrb w9, [x19, x24]   // x[i]
    mov x10, xzr          // diag
    mov x25, #1           // j
.Linner:
    cmp x25, x22
    b.gt .Linner_done
    ldr x11, [x23, x25, lsl #3]   // up = row[j]
    sub x12, x25, #1
    ldrb w13, [x21, x12]          // y[j-1]
    ldr x14, [x23, x12, lsl #3]   // row[j-1]
    cmp w9, w13
    b.eq .Lmatch
    cmp x11, x14
    csel x15, x11, x14, ge
    b .Lstore
.Lmatch:
    add x15, x10, #1
.Lstore:
    str x15, [x23, x25, lsl #3]
    mov x10, x11
    add x25, x25, #1
    b .Linner
.Linner_done:
    add x24, x24, #1
    b .Louter
.Louter_done:

    ldr x0, [x23, x22, lsl #3]

    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    ldp x23, x24, [sp, #32]
    ldp x25, x26, [sp, #48]
    add sp, sp, #64
    ret
