packbits_decode:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx r10, byte ptr [rdi + r8]
    inc r8
    cmp r10, -128
    je .Lloop
    cmp r10, 0
    jl .Lrepeat
    inc r10
    mov r11, r8
    add r11, r10
    cmp r11, rsi
    ja .Lerr_src
.Lliteral_loop:
    test r10, r10
    jz .Lloop
    cmp r9, rcx
    jae .Lerr_cap
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jmp .Lliteral_loop
.Lrepeat:
    mov r11, 1
    sub r11, r10
    mov r10, r8
    inc r10
    cmp r10, rsi
    ja .Lerr_src
    mov al, byte ptr [rdi + r8]
    inc r8
    mov r10, r11
.Lrepeat_loop:
    test r10, r10
    jz .Lloop
    cmp r9, rcx
    jae .Lerr_cap
    mov byte ptr [rdx + r9], al
    inc r9
    dec r10
    jmp .Lrepeat_loop
.Ldone:
    mov rax, r9
    ret
.Lerr_src:
    mov rax, -2
    ret
.Lerr_cap:
    mov rax, -1
    ret
