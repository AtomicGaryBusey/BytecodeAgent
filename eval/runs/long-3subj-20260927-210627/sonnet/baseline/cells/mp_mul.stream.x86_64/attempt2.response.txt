mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r14, rsi
    mov r12, rdx
    mov r15, rcx
    mov r13, r8
    lea r11, [r14 + r15]
    xor rcx, rcx
.Lzero_loop:
    cmp rcx, r11
    jge .Lzero_done
    mov qword ptr [r13 + rcx*8], 0
    inc rcx
    jmp .Lzero_loop
.Lzero_done:
    test r14, r14
    jz .Lcount
    test r15, r15
    jz .Lcount
    xor rbp, rbp
.Louter:
    cmp rbp, r14
    jge .Lcount
    xor r10, r10
    xor r9, r9
.Linner:
    cmp r9, r15
    jge .Linner_done
    mov rax, qword ptr [rbx + rbp*8]
    mul qword ptr [r12 + r9*8]
    add rax, r10
    adc rdx, 0
    lea rcx, [rbp + r9]
    add rax, qword ptr [r13 + rcx*8]
    adc rdx, 0
    mov qword ptr [r13 + rcx*8], rax
    mov r10, rdx
    inc r9
    jmp .Linner
.Linner_done:
    lea rcx, [rbp + r15]
    mov qword ptr [r13 + rcx*8], r10
    inc rbp
    jmp .Louter
.Lcount:
    mov rax, r11
.Lcount_loop:
    test rax, rax
    jz .Lcount_done
    mov rcx, qword ptr [r13 + rax*8 - 8]
    test rcx, rcx
    jnz .Lcount_done
    dec rax
    jmp .Lcount_loop
.Lcount_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
