export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %pos =l phi @start 0, @nop %pos1, @litdone %need_end, @rdone %need_end2
    %opos =l phi @start 0, @nop %opos, @litdone %opos_l, @rdone %opos_r
    %pge =w csgel %pos, %n
    jnz %pge, @done, @readh
@readh
    %sp =l add %src, %pos
    %h =w loadsb %sp
    %pos1 =l add %pos, 1
    %isnop =w ceqw %h, -128
    jnz %isnop, @nop, @checklit
@nop
    jmp @loop
@checklit
    %islit =w csgew %h, 0
    jnz %islit, @literal, @repeat
@literal
    %hl =l extsw %h
    %count =l add %hl, 1
    %need_end =l add %pos1, %count
    %toofar =w csgtl %need_end, %n
    jnz %toofar, @err2, @litinit
@litinit
    %sp2 =l add %src, %pos1
    jmp @litloop
@litloop
    %i =l phi @litinit 0, @litbody %i2
    %opos_l =l phi @litinit %opos, @litbody %opos2
    %idone =w csgel %i, %count
    jnz %idone, @litdone, @litcheck
@litcheck
    %ofull =w csgel %opos_l, %cap
    jnz %ofull, @err1, @litbody
@litbody
    %sb =l add %sp2, %i
    %b =w loadub %sb
    %ob =l add %out, %opos_l
    storeb %b, %ob
    %i2 =l add %i, 1
    %opos2 =l add %opos_l, 1
    jmp @litloop
@litdone
    jmp @loop
@repeat
    %need_end2 =l add %pos1, 1
    %toofar2 =w csgtl %need_end2, %n
    jnz %toofar2, @err2, @repinit
@repinit
    %rb_addr =l add %src, %pos1
    %rb =w loadub %rb_addr
    %negh =w sub 0, %h
    %cnt =w add %negh, 1
    %cntl =l extsw %cnt
    jmp @rloop
@rloop
    %j =l phi @repinit 0, @rbody %j2
    %opos_r =l phi @repinit %opos, @rbody %opos_r2
    %jdone =w csgel %j, %cntl
    jnz %jdone, @rdone, @rcheck
@rcheck
    %rofull =w csgel %opos_r, %cap
    jnz %rofull, @err1, @rbody
@rbody
    %ob2 =l add %out, %opos_r
    storeb %rb, %ob2
    %j2 =l add %j, 1
    %opos_r2 =l add %opos_r, 1
    jmp @rloop
@rdone
    jmp @loop
@done
    ret %opos
@err1
    ret -1
@err2
    ret -2
}
