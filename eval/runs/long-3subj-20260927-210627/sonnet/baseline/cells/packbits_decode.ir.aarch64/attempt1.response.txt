export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %si =l phi @start 0, @cont %sic
    %oi =l phi @start 0, @cont %oic
    %inbound =w csltl %si, %n
    jnz %inbound, @body, @done
@body
    %p =l add %src, %si
    %h =w loadsb %p
    %si1 =l add %si, 1
    %isnop =w ceqw %h, -128
    jnz %isnop, @nop, @chk
@nop
    jmp @cont
@chk
    %islit =w csgew %h, 0
    jnz %islit, @lit, @rep
@lit
    %h64 =l extsw %h
    %count =l add %h64, 1
    %need =l add %si1, %count
    %over =w csgtl %need, %n
    jnz %over, @err2, @litloop
@litloop
    %i =l phi @lit 0, @litwrite %i2
    %si2 =l phi @lit %si1, @litwrite %si2n
    %oi2 =l phi @lit %oi, @litwrite %oi2n
    %donelit =w csgel %i, %count
    jnz %donelit, @litdone, @litcap
@litcap
    %full1 =w csgel %oi2, %cap
    jnz %full1, @err1, @litwrite
@litwrite
    %sp2 =l add %src, %si2
    %b1 =w loadub %sp2
    %op2 =l add %out, %oi2
    storeb %b1, %op2
    %si2n =l add %si2, 1
    %oi2n =l add %oi2, 1
    %i2 =l add %i, 1
    jmp @litloop
@litdone
    jmp @cont
@rep
    %h64r =l extsw %h
    %rep =l sub 1, %h64r
    %need2 =l add %si1, 1
    %over2 =w csgtl %need2, %n
    jnz %over2, @err2, @repget
@repget
    %bp =l add %src, %si1
    %byteval =w loadub %bp
    %si3 =l add %si1, 1
    jmp @reploop
@reploop
    %j =l phi @repget 0, @repwrite %j2
    %oi3 =l phi @repget %oi, @repwrite %oi3n
    %donerep =w csgel %j, %rep
    jnz %donerep, @repdone, @repcap
@repcap
    %full2 =w csgel %oi3, %cap
    jnz %full2, @err1, @repwrite
@repwrite
    %op3 =l add %out, %oi3
    storeb %byteval, %op3
    %oi3n =l add %oi3, 1
    %j2 =l add %j, 1
    jmp @reploop
@repdone
    jmp @cont
@cont
    %sic =l phi @nop %si1, @litdone %si2, @repdone %si3
    %oic =l phi @nop %oi, @litdone %oi2, @repdone %oi3
    jmp @loop
@done
    ret %oi
@err1
    ret -1
@err2
    ret -2
}
