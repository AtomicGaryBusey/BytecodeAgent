export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%oi =l copy 0
	jmp @loop
@loop
	%c1 =w csltl %i, %n
	jnz %c1, @body, @done
@body
	%pi =l add %src, %i
	%b0 =w loadub %pi
	%rlen =l copy 1
	%jj =l add %i, 1
	jmp @runloop
@runloop
	%rc1 =w csltl %rlen, 128
	jnz %rc1, @runchk1, @rundone
@runchk1
	%rc2 =w csltl %jj, %n
	jnz %rc2, @runcmp, @rundone
@runcmp
	%pj =l add %src, %jj
	%bj =w loadub %pj
	%rc3 =w ceqw %bj, %b0
	jnz %rc3, @runinc, @rundone
@runinc
	%rlen =l add %rlen, 1
	%jj =l add %jj, 1
	jmp @runloop
@rundone
	%islit =w csltl %rlen, 3
	jnz %islit, @literal, @repeat
@repeat
	%need =l add %oi, 2
	%fits =w csgel %cap, %need
	jnz %fits, @repwrite, @fail
@repwrite
	%hdr =l sub 257, %rlen
	%poi =l add %out, %oi
	storeb %hdr, %poi
	%poi2 =l add %poi, 1
	storeb %b0, %poi2
	%oi =l add %oi, 2
	%i =l add %i, %rlen
	jmp @loop
@literal
	%j =l copy %i
	jmp @litloop
@litloop
	%lc1 =w csltl %j, %n
	jnz %lc1, @litchk2, @litdone
@litchk2
	%ld =l sub %j, %i
	%lc2 =w csltl %ld, 128
	jnz %lc2, @litrun, @litdone
@litrun
	%pj2 =l add %src, %j
	%bj2 =w loadub %pj2
	%rlen2 =l copy 1
	%jj2 =l add %j, 1
	jmp @runloop2
@runloop2
	%r2c1 =w csltl %rlen2, 3
	jnz %r2c1, @runchk21, @rundone2
@runchk21
	%r2c2 =w csltl %jj2, %n
	jnz %r2c2, @runcmp2, @rundone2
@runcmp2
	%pj3 =l add %src, %jj2
	%bj3 =w loadub %pj3
	%r2c3 =w ceqw %bj3, %bj2
	jnz %r2c3, @runinc2, @rundone2
@runinc2
	%rlen2 =l add %rlen2, 1
	%jj2 =l add %jj2, 1
	jmp @runloop2
@rundone2
	%runlt3 =w csltl %rlen2, 3
	jnz %runlt3, @litcontinue, @litdone
@litcontinue
	%j =l add %j, 1
	jmp @litloop
@litdone
	%L =l sub %j, %i
	%need2 =l add %oi, 1
	%need3 =l add %need2, %L
	%fits2 =w csgel %cap, %need3
	jnz %fits2, @litwrite, @fail
@litwrite
	%hdr2 =l sub %L, 1
	%po =l add %out, %oi
	storeb %hdr2, %po
	%oi =l add %oi, 1
	%k =l copy 0
	jmp @copyloop
@copyloop
	%cc1 =w csltl %k, %L
	jnz %cc1, @copybody, @copydone
@copybody
	%sp2 =l add %i, %k
	%spp =l add %src, %sp2
	%cb =w loadub %spp
	%dp =l add %oi, %k
	%dpp =l add %out, %dp
	storeb %cb, %dpp
	%k =l add %k, 1
	jmp @copyloop
@copydone
	%oi =l add %oi, %L
	%i =l add %i, %L
	jmp @loop
@fail
	ret -1
@done
	ret %oi
}
