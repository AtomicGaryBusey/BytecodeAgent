packbits_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rdi
    mov r9, rsi
    mov r12, rdx
    mov r11, rcx
    xor r13, r13
    xor r14, r14

.Lmain_loop:
    cmp r13, r9
    jge .Ldone

    mov al, byte ptr [rbx + r13]
    mov r15, 1
    mov r8, r13
    inc r8
.Lrun_loop1:
    cmp r15, 128
    jae .Lrun_done1
    cmp r8, r9
    jge .Lrun_done1
    mov dl, byte ptr [rbx + r8]
    cmp dl, al
    jne .Lrun_done1
    inc r15
    inc r8
    jmp .Lrun_loop1
.Lrun_done1:

    cmp r15, 3
    jl .Lliteral

    mov rcx, r14
    add rcx, 2
    cmp rcx, r11
    jg .Lfail

    mov rcx, 257
    sub rcx, r15
    mov byte ptr [r12 + r14], cl
    mov byte ptr [r12 + r14 + 1], al
    add r14, 2
    add r13, r15
    jmp .Lmain_loop

.Lliteral:
    mov r8, r13
.Llit_scan:
    cmp r8, r9
    jge .Llit_scan_done
    mov rcx, r8
    sub rcx, r13
    cmp rcx, 128
    jge .Llit_scan_done

    mov al, byte ptr [rbx + r8]
    mov r15, 1
    mov rdx, r8
    inc rdx
.Lrun_loop2:
    cmp r15, 3
    jge .Lrun_done2
    cmp rdx, r9
    jge .Lrun_done2
    mov cl, byte ptr [rbx + rdx]
    cmp cl, al
    jne .Lrun_done2
    inc r15
    inc rdx
    jmp .Lrun_loop2
.Lrun_done2:
    cmp r15, 3
    jge .Llit_scan_done
    inc r8
    jmp .Llit_scan
.Llit_scan_done:

    mov r15, r8
    sub r15, r13

    mov rcx, r14
    add rcx, r15
    add rcx, 1
    cmp rcx, r11
    jg .Lfail

    mov rax, r15
    dec rax
    mov byte ptr [r12 + r14], al
    inc r14

    xor rax, rax
.Lcopy_loop:
    cmp rax, r15
    jge .Lcopy_done
    mov dl, byte ptr [rbx + r13 + rax]
