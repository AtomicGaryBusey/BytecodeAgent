packbits_encode:
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rdi      # src
    mov r9, rsi       # n
    mov r12, rdx      # out
    mov r11, rcx      # cap
    xor r13, r13      # i
    xor r14, r14      # o

.Lmain_loop:
    cmp r13, r9
    jge .Ldone

    lea r10, [rbx + r13]     # &src[i]
    mov al, byte ptr [r10]
    mov r15, 1               # run length
.Lrun_loop1:
    cmp r15, 128
    jae .Lrun_done1
    mov rcx, r13
    add rcx, r15
    cmp rcx, r9
    jge .Lrun_done1
    mov dl, byte ptr [r10 + r15]
    cmp dl, al
    jne .Lrun_done1
    inc r15
    jmp .Lrun_loop1
.Lrun_done1:

    cmp r15, 3
    jl .Lliteral

    mov rcx, r14
    add rcx, 2
    cmp rcx, r11
    jg .Lfail

    mov rcx, 257
    sub rcx, r15
    mov byte ptr [r12 + r14], cl
    mov byte ptr [r12 + r14 + 1], al
    add r14, 2
    add r13, r15
    jmp .Lmain_loop

.Lliteral:
    mov r8, r13              # j = i
.Llit_scan:
    cmp r8, r9
    jge .Llit_scan_done
    mov rcx, r8
    sub rcx, r13
    cmp rcx, 128
    jge .Llit_scan_done

    lea r10, [rbx + r8]
    mov al, byte ptr [r10]
    mov r15, 1
.Lrun_loop2:
    cmp r15, 3
    jge .Lrun_done2
    mov rcx, r8
    add rcx, r15
    cmp rcx, r9
    jge .Lrun_done2
    mov dl, byte ptr [r10 + r15]
    cmp dl, al
    jne .Lrun_done2
    inc r15
    jmp .Lrun_loop2
.Lrun_done2:
    cmp r15, 3
    jge .Llit_scan_done
    inc r8
    jmp .Llit_scan
.Llit_scan_done:

    mov r15, r8
    sub r15, r13             # L = j - i

    mov rcx, r14
    add rcx, r15
    inc rcx
    cmp rcx, r11
    jg .Lfail

    mov rax, r15
    dec rax
    mov byte ptr [r12 + r14], al
    inc r14

    lea r10, [rbx + r13]      # &src[i]
    lea rcx, [r12 + r14]      # &out[o]
    xor rax, rax
.Lcopy_loop:
    cmp rax, r15
    jge .Lcopy_done
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [rcx + rax], dl
    inc rax
    jmp .Lcopy_loop
.Lcopy_done:
    add r14, r15
    mov r13, r8
    jmp .Lmain_loop

.Lfail:
    mov rax, -1
    jmp .Lret

.Ldone:
    mov rax, r14

.Lret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
