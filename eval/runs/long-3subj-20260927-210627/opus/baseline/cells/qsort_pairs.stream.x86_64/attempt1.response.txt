qsort_pairs:
    push rbx
    push r12
    push r13
    mov rbx, rdi
    mov r12, rsi
.Lloop:
    cmp r12, 2
    jl .Ldone
    mov rax, r12
    shr rax, 1
    shl rax, 4
    lea rdi, [r12 - 1]
    shl rdi, 4
    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    mov r10, qword ptr [rbx + rdi]
    mov r11, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rax], r10
    mov qword ptr [rbx + rax + 8], r11
    mov qword ptr [rbx + rdi], r8
    mov qword ptr [rbx + rdi + 8], r9
    xor ecx, ecx
    xor edx, edx
.Lpart:
    cmp rdx, rdi
    jge .Lpend
    mov r10, qword ptr [rbx + rdx]
    mov r11, qword ptr [rbx + rdx + 8]
    cmp r10, r8
    jl .Lless
    jg .Lnext
    cmp r11, r9
    jge .Lnext
.Lless:
    mov rax, qword ptr [rbx + rcx]
    mov rsi, qword ptr [rbx + rcx + 8]
    mov qword ptr [rbx + rdx], rax
    mov qword ptr [rbx + rdx + 8], rsi
    mov qword ptr [rbx + rcx], r10
    mov qword ptr [rbx + rcx + 8], r11
    add rcx, 16
.Lnext:
    add rdx, 16
    jmp .Lpart
.Lpend:
    mov rax, qword ptr [rbx + rcx]
    mov rsi, qword ptr [rbx + rcx + 8]
    mov qword ptr [rbx + rdi], rax
    mov qword ptr [rbx + rdi + 8], rsi
    mov qword ptr [rbx + rcx], r8
    mov qword ptr [rbx + rcx + 8], r9
    mov r13, rcx
    mov rdi, rbx
    mov rsi, rcx
    shr rsi, 4
    call qsort_pairs
    lea rbx, [rbx + r13 + 16]
    shr r13, 4
    sub r12, r13
    dec r12
    jmp .Lloop
.Ldone:
    xor eax, eax
    pop r13
    pop r12
    pop rbx
    ret
