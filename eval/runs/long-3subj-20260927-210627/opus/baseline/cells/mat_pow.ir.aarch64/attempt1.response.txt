export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%idx =l copy 0
	jmp @idl
@idl
	%c =w csltl %idx, %nn
	jnz %c, @idb, @iddone
@idb
	%qi =l div %idx, %n
	%rj =l rem %idx, %n
	%eq =w ceql %qi, %rj
	%v =l extuw %eq
	%off =l mul %idx, 8
	%p =l add %out, %off
	storel %v, %p
	%idx =l add %idx, 1
	jmp @idl
@iddone
	%r =l copy 0
	jmp @powl
@powl
	%c2 =w csltl %r, %k
	jnz %c2, @powb, @zero
@powb
	%i =l copy 0
	jmp @il
@il
	%ci =w csltl %i, %n
	jnz %ci, @ib, @copy
@ib
	%j =l copy 0
	jmp @jl
@jl
	%cj =w csltl %j, %n
	jnz %cj, @jb, @inext
@jb
	%s =l copy 0
	%t =l copy 0
	jmp @tl
@tl
	%ct =w csltl %t, %n
	jnz %ct, @tb, @tdone
@tb
	%in =l mul %i, %n
	%x1 =l add %in, %t
	%x1o =l mul %x1, 8
	%px =l add %out, %x1o
	%xv =l loadl %px
	%tn =l mul %t, %n
	%y1 =l add %tn, %j
	%y1o =l mul %y1, 8
	%py =l add %a, %y1o
	%yv =l loadl %py
	%pr =l mul %xv, %yv
	%s =l add %s, %pr
	%t =l add %t, 1
	jmp @tl
@tdone
	%in2 =l mul %i, %n
	%z1 =l add %in2, %j
	%z1o =l mul %z1, 8
	%pz =l add %tmp, %z1o
	storel %s, %pz
	%j =l add %j, 1
	jmp @jl
@inext
	%i =l add %i, 1
	jmp @il
@copy
	%ci2 =l copy 0
	jmp @cl
@cl
	%cc =w csltl %ci2, %nn
	jnz %cc, @cb, @cdone
@cb
	%co =l mul %ci2, 8
	%ps =l add %tmp, %co
	%cv =l loadl %ps
	%pd =l add %out, %co
	storel %cv, %pd
	%ci2 =l add %ci2, 1
	jmp @cl
@cdone
	%r =l add %r, 1
	jmp @powl
@zero
	%tot =l mul %nn, 2
	%zi =l copy 0
	jmp @zl
@zl
	%cz =w csltl %zi, %tot
	jnz %cz, @zb, @end
@zb
	%zo =l mul %zi, 8
	%pzz =l add %tmp, %zo
	storel 0, %pzz
	%zi =l add %zi, 1
	jmp @zl
@end
	ret 0
}
