export function l $qsort_pairs(l %a0, l %n0) {
@start
	%a =l copy %a0
	%n =l copy %n0
@top
	%c =w cslel %n, 1
	jnz %c, @done, @work
@work
	%mid =l div %n, 2
	%om =l mul %mid, 16
	%pm =l add %a, %om
	%last =l sub %n, 1
	%ol =l mul %last, 16
	%pl =l add %a, %ol
	%k1 =l loadl %pm
	%pmv =l add %pm, 8
	%v1 =l loadl %pmv
	%k2 =l loadl %pl
	%plv =l add %pl, 8
	%v2 =l loadl %plv
	storel %k2, %pm
	storel %v2, %pmv
	storel %k1, %pl
	storel %v1, %plv
	%pk =l copy %k1
	%pv =l copy %v1
	%i =l copy 0
	%j =l copy 0
@loop
	%cj =w csltl %j, %last
	jnz %cj, @body, @endloop
@body
	%oj =l mul %j, 16
	%pj =l add %a, %oj
	%kj =l loadl %pj
	%pjv =l add %pj, 8
	%vj =l loadl %pjv
	%lt =w csltl %kj, %pk
	jnz %lt, @swap, @chk2
@chk2
	%eq =w ceql %kj, %pk
	jnz %eq, @chk3, @next
@chk3
	%lt2 =w csltl %vj, %pv
	jnz %lt2, @swap, @next
@swap
	%oi =l mul %i, 16
	%pi =l add %a, %oi
	%piv =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %piv
	storel %kj, %pi
	storel %vj, %piv
	storel %ki, %pj
	storel %vi, %pjv
	%i =l add %i, 1
@next
	%j =l add %j, 1
	jmp @loop
@endloop
	%oi2 =l mul %i, 16
	%pi2 =l add %a, %oi2
	%pi2v =l add %pi2, 8
	%ki2 =l loadl %pi2
	%vi2 =l loadl %pi2v
	storel %pk, %pi2
	storel %pv, %pi2v
	storel %ki2, %pl
	storel %vi2, %plv
	%r =l call $qsort_pairs(l %a, l %i)
	%i1 =l add %i, 1
	%off =l mul %i1, 16
	%a =l add %a, %off
	%n =l sub %n, %i1
	jmp @top
@done
	ret 0
}
