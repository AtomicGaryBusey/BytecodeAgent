export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%n1 =l add %n, 1
	%i =l copy 0
@init
	%c =w csltl %i, %nn
	jnz %c, @initbody, @pinit
@initbody
	%m =l rem %i, %n1
	%d =w ceql %m, 0
	%dl =l extuw %d
	%off =l mul %i, 8
	%p =l add %out, %off
	storel %dl, %p
	%i =l add %i, 1
	jmp @init
@pinit
	%it =l copy 0
@ploop
	%c2 =w csltl %it, %k
	jnz %c2, @mul, @clear
@mul
	%r =l copy 0
@rloop
	%c3 =w csltl %r, %n
	jnz %c3, @cinit, @copyinit
@cinit
	%col =l copy 0
@cloop
	%c4 =w csltl %col, %n
	jnz %c4, @cbody, @rnext
@cbody
	%s =l copy 0
	%t =l copy 0
@tloop
	%c5 =w csltl %t, %n
	jnz %c5, @tbody, @tdone
@tbody
	%rn =l mul %r, %n
	%xi =l add %rn, %t
	%xo =l mul %xi, 8
	%xp =l add %out, %xo
	%x =l loadl %xp
	%tn =l mul %t, %n
	%yi =l add %tn, %col
	%yo =l mul %yi, 8
	%yp =l add %a, %yo
	%y =l loadl %yp
	%xy =l mul %x, %y
	%s =l add %s, %xy
	%t =l add %t, 1
	jmp @tloop
@tdone
	%rn2 =l mul %r, %n
	%zi =l add %rn2, %col
	%zo =l mul %zi, 8
	%zp =l add %tmp, %zo
	storel %s, %zp
	%col =l add %col, 1
	jmp @cloop
@rnext
	%r =l add %r, 1
	jmp @rloop
@copyinit
	%j =l copy 0
@copyloop
	%c6 =w csltl %j, %nn
	jnz %c6, @copybody, @pnext
@copybody
	%jo =l mul %j, 8
	%sp =l add %tmp, %jo
	%v =l loadl %sp
	%dp =l add %out, %jo
	storel %v, %dp
	%j =l add %j, 1
	jmp @copyloop
@pnext
	%it =l add %it, 1
	jmp @ploop
@clear
	%tot =l mul %nn, 2
	%q =l copy 0
@clrloop
	%c7 =w csltl %q, %tot
	jnz %c7, @clrbody, @done
@clrbody
	%qo =l mul %q, 8
	%qp =l add %tmp, %qo
	storel 0, %qp
	%q =l add %q, 1
	jmp @clrloop
@done
	ret 0
}
