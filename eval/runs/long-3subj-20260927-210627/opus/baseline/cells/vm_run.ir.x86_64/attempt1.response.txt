export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
@loop
	%e =w ceql %pc, %len
	jnz %e, @done, @fetch
@fetch
	%o =l mul %pc, 8
	%p =l add %code, %o
	%op =l loadl %p
	%t1 =l sub %sp, 1
	%t1o =l mul %t1, 8
	%tp =l add %stack, %t1o
	%ap =l sub %tp, 8
	%np =l add %tp, 8
	%c0 =w ceql %op, 0
	jnz %c0, @done, @n1
@n1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @n2
@n2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @n3
@n3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @n4
@n4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @n5
@n5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @n6
@n6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @n7
@n7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @n8
@n8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @n9
@n9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @n10
@n10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @n11
@n11
	%c11 =w ceql %op, 11
	jnz %c11, @jmp, @other
@push
	%q1 =l add %p, 8
	%v1 =l loadl %q1
	storel %v1, %np
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%b2 =l loadl %tp
	%a2 =l loadl %ap
	%r2 =l add %a2, %b2
	storel %r2, %ap
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@sub
	%b3 =l loadl %tp
	%a3 =l loadl %ap
	%r3 =l sub %a3, %b3
	storel %r3, %ap
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@mul
	%b4 =l loadl %tp
	%a4 =l loadl %ap
	%r4 =l mul %a4, %b4
	storel %r4, %ap
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%b5 =l loadl %tp
	storel %b5, %np
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%b6 =l loadl %tp
	%a6 =l loadl %ap
	storel %a6, %tp
	storel %b6, %ap
	%pc =l add %pc, 1
	jmp @loop
@drop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%a8 =l loadl %ap
	storel %a8, %np
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@lt
	%b9 =l loadl %tp
	%a9 =l loadl %ap
	%w9 =w csltl %a9, %b9
	%r9 =l extuw %w9
	storel %r9, %ap
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%v10 =l loadl %tp
	%sp =l sub %sp, 1
	%z10 =w ceql %v10, 0
	jnz %z10, @jmp, @jzf
@jzf
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%q11 =l add %p, 8
	%pc =l loadl %q11
	jmp @loop
@other
	%pc =l add %pc, 1
	jmp @loop
@done
	%i =l copy %sp
@zl
	%zc =w csltl %i, %cap
	jnz %zc, @zb, @zend
@zb
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zl
@zend
	%res =l copy 0
	%emp =w ceql %sp, 0
	jnz %emp, @ret, @gettop
@gettop
	%g1 =l sub %sp, 1
	%go =l mul %g1, 8
	%gp =l add %stack, %go
	%res =l loadl %gp
@ret
	ret %res
}
