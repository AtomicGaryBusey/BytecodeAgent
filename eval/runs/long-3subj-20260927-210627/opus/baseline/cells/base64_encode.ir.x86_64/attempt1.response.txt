export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy %out
	%v =l copy 0
	%cnt =l copy 0
@loop
	%rem =l sub %n, %i
	%cf =w csgel %rem, 3
	jnz %cf, @full, @tail
@full
	%p =l add %src, %i
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%t0 =l or %s0, %s1
	%v =l or %t0, %b2
	%i =l add %i, 3
	%cnt =l copy 4
	jmp @emit
@tail
	%cz =w ceql %rem, 0
	jnz %cz, @done, @tail1
@tail1
	%tp =l add %src, %i
	%tb0 =l loadub %tp
	%v =l shl %tb0, 16
	%cnt =l copy 2
	%c2 =w ceql %rem, 2
	jnz %c2, @tail2, @emit
@tail2
	%tp1 =l add %tp, 1
	%tb1 =l loadub %tp1
	%ts1 =l shl %tb1, 8
	%v =l or %v, %ts1
	%cnt =l copy 3
	jmp @emit
@emit
	%k =l copy 0
	%sh =l copy 18
@eloop
	%xs =l shr %v, %sh
	%x =l and %xs, 63
	%ch =l add %x, 65
	%g1 =w csgel %x, 26
	%e1 =l extuw %g1
	%m1 =l mul %e1, 6
	%ch =l add %ch, %m1
	%g2 =w csgel %x, 52
	%e2 =l extuw %g2
	%m2 =l mul %e2, 75
	%ch =l sub %ch, %m2
	%g3 =w csgel %x, 62
	%e3 =l extuw %g3
	%m3 =l mul %e3, 15
	%ch =l sub %ch, %m3
	%g4 =w ceql %x, 63
	%e4 =l extuw %g4
	%m4 =l mul %e4, 3
	%ch =l add %ch, %m4
	storeb %ch, %o
	%o =l add %o, 1
	%k =l add %k, 1
	%sh =l sub %sh, 6
	%ck =w csltl %k, %cnt
	jnz %ck, @eloop, @eend
@eend
	%c4 =w ceql %cnt, 4
	jnz %c4, @loop, @pad
@pad
	%cp =w csltl %k, 4
	jnz %cp, @padw, @done
@padw
	storeb 61, %o
	%o =l add %o, 1
	%k =l add %k, 1
	jmp @pad
@done
	%r =l sub %o, %out
	ret %r
}
