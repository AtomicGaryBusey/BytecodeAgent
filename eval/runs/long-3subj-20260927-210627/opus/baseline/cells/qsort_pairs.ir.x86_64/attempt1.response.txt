export function l $qsort_pairs(l %a0, l %n0) {
@start
	%a =l copy %a0
	%n =l copy %n0
@loop
	%c =w csltl %n, 2
	jnz %c, @done, @part
@part
	%m =l div %n, 2
	%pm =l mul %m, 16
	%pm =l add %a, %pm
	%nl =l sub %n, 1
	%pl =l mul %nl, 16
	%pl =l add %a, %pl
	%pm8 =l add %pm, 8
	%pl8 =l add %pl, 8
	%k1 =l loadl %pm
	%v1 =l loadl %pm8
	%k2 =l loadl %pl
	%v2 =l loadl %pl8
	storel %k2, %pm
	storel %v2, %pm8
	storel %k1, %pl
	storel %v1, %pl8
	%pk =l copy %k1
	%pv =l copy %v1
	%i =l copy 0
	%j =l copy 0
@jloop
	%cj =w csltl %j, %nl
	jnz %cj, @body, @endp
@body
	%pj =l mul %j, 16
	%pj =l add %a, %pj
	%pj8 =l add %pj, 8
	%kj =l loadl %pj
	%vj =l loadl %pj8
	%lt =w csltl %kj, %pk
	jnz %lt, @doswap, @chk
@chk
	%eq =w ceql %kj, %pk
	jnz %eq, @chk2, @next
@chk2
	%lt2 =w csltl %vj, %pv
	jnz %lt2, @doswap, @next
@doswap
	%pi =l mul %i, 16
	%pi =l add %a, %pi
	%pi8 =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %pi8
	storel %kj, %pi
	storel %vj, %pi8
	storel %ki, %pj
	storel %vi, %pj8
	%i =l add %i, 1
@next
	%j =l add %j, 1
	jmp @jloop
@endp
	%qi =l mul %i, 16
	%qi =l add %a, %qi
	%qi8 =l add %qi, 8
	%ka =l loadl %qi
	%va =l loadl %qi8
	%kb =l loadl %pl
	%vb =l loadl %pl8
	storel %kb, %qi
	storel %vb, %qi8
	storel %ka, %pl
	storel %va, %pl8
	%ln =l copy %i
	%rn =l sub %nl, %i
	%ra =l add %qi, 16
	%cs =w csltl %ln, %rn
	jnz %cs, @recl, @recr
@recl
	%r1 =l call $qsort_pairs(l %a, l %ln)
	%a =l copy %ra
	%n =l copy %rn
	jmp @loop
@recr
	%r2 =l call $qsort_pairs(l %ra, l %rn)
	%n =l copy %ln
	jmp @loop
@done
	ret 0
}
