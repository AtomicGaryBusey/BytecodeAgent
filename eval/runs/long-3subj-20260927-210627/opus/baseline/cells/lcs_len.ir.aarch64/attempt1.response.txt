export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%k =l copy 0
@zloop
	%zc =w cslel %k, %n
	jnz %zc, @zbody, @zdone
@zbody
	%zo =l mul %k, 8
	%zp =l add %row, %zo
	storel 0, %zp
	%k =l add %k, 1
	jmp @zloop
@zdone
	%i =l copy 0
@iloop
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @done
@ibody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%diag =l copy 0
	%left =l copy 0
	%j =l copy 1
@jloop
	%jc =w cslel %j, %n
	jnz %jc, @jbody, @inext
@jbody
	%jo =l mul %j, 8
	%rp =l add %row, %jo
	%up =l loadl %rp
	%jm =l sub %j, 1
	%yp =l add %y, %jm
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%val =l add %diag, 1
	jmp @store
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @takeup, @takeleft
@takeup
	%val =l copy %up
	jmp @store
@takeleft
	%val =l copy %left
@store
	storel %val, %rp
	%left =l copy %val
	%diag =l copy %up
	%j =l add %j, 1
	jmp @jloop
@inext
	%i =l add %i, 1
	jmp @iloop
@done
	%no =l mul %n, 8
	%np =l add %row, %no
	%r =l loadl %np
	ret %r
}
