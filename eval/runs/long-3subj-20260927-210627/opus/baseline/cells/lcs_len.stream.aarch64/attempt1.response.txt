lcs_len:
    // x0=x, x1=m, x2=y, x3=n, x4=row
    mov x9, #0
.Lzero:
    str xzr, [x4, x9, lsl #3]
    add x9, x9, #1
    cmp x9, x3
    b.le .Lzero
    mov x10, #0                 // i
.Louter:
    cmp x10, x1
    b.ge .Ldone
    ldrb w11, [x0, x10]         // x[i]
    mov x12, #0                 // diag
    mov x13, #0                 // left = row[j-1], row[0]=0
    mov x9, #1                  // j
.Linner:
    cmp x9, x3
    b.gt .Lnexti
    ldr x14, [x4, x9, lsl #3]   // up
    sub x15, x9, #1
    ldrb w15, [x2, x15]         // y[j-1]
    cmp x14, x13
    csel x16, x14, x13, ge      // max(up,left)
    add x17, x12, #1
    cmp w11, w15
    csel x16, x17, x16, eq
    str x16, [x4, x9, lsl #3]
    mov x13, x16
    mov x12, x14
    add x9, x9, #1
    b .Linner
.Lnexti:
    add x10, x10, #1
    b .Louter
.Ldone:
    ldr x0, [x4, x3, lsl #3]
    ret
