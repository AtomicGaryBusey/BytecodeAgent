packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx eax, byte ptr [rdi + r8]
    cmp eax, -128
    je .Lnop
    test eax, eax
    js .Lrep
    lea r10, [r8 + 1]
    lea r11, [r10 + rax + 1]
    cmp r11, rsi
    jg .Lerr2
.Llit:
    cmp r10, r11
    jge .Llitdone
    cmp r9, rcx
    jge .Lerr1
    movzx eax, byte ptr [rdi + r10]
    mov byte ptr [rdx + r9], al
    inc r10
    inc r9
    jmp .Llit
.Llitdone:
    mov r8, r11
    jmp .Lloop
.Lrep:
    lea r10, [r8 + 2]
    cmp r10, rsi
    jg .Lerr2
    mov r11d, 1
    sub r11d, eax
    movzx eax, byte ptr [rdi + r8 + 1]
.Lrl:
    cmp r9, rcx
    jge .Lerr1
    mov byte ptr [rdx + r9], al
    inc r9
    dec r11d
    jnz .Lrl
    mov r8, r10
    jmp .Lloop
.Lnop:
    inc r8
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
.Lerr2:
    mov rax, -2
    ret
.Lerr1:
    mov rax, -1
    ret
