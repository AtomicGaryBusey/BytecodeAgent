lcs_len:
    push rbx
    push r12
    xor eax, eax
.Lzero:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Lzero
    xor r9d, r9d
.Louter:
    cmp r9, rsi
    jge .Ldone
    movzx r10d, byte ptr [rdi + r9]
    xor r11d, r11d
    xor ebx, ebx
    mov eax, 1
.Linner:
    cmp rax, rcx
    jg .Lnexti
    mov r12, qword ptr [r8 + rax*8]
    movzx edx, byte ptr [rdx + 0]
    jmp .Lcont
.Lcont:
    jmp .Lnexti
.Lnexti:
    inc r9
    jmp .Louter
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    pop r12
    pop rbx
    ret
