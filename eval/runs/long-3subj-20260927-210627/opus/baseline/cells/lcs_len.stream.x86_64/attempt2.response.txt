lcs_len:
    push rbx
    push r12
    push r13
    xor eax, eax
.Lzero:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Lzero
    xor r9d, r9d
.Louter:
    cmp r9, rsi
    jge .Ldone
    movzx r10d, byte ptr [rdi + r9]
    xor r11d, r11d
    mov eax, 1
.Linner:
    cmp rax, rcx
    jg .Lnexti
    mov r12, qword ptr [r8 + rax*8]
    movzx ebx, byte ptr [rdx + rax - 1]
    cmp ebx, r10d
    jne .Lnomatch
    lea r13, [r11 + 1]
    jmp .Lstore
.Lnomatch:
    mov r13, qword ptr [r8 + rax*8 - 8]
    cmp r13, r12
    cmovl r13, r12
.Lstore:
    mov qword ptr [r8 + rax*8], r13
    mov r11, r12
    inc rax
    jmp .Linner
.Lnexti:
    inc r9
    jmp .Louter
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    pop r13
    pop r12
    pop rbx
    ret
