export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i =l copy 0
    %o =l copy 0
@loop
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %p =l add %src, %i
    %b =w loadub %p
    %r =l copy 1
@rl
    %k =l add %i, %r
    %c1 =w csltl %k, %n
    jnz %c1, @rl2, @rend
@rl2
    %c2 =w csltl %r, 128
    jnz %c2, @rl3, @rend
@rl3
    %q =l add %src, %k
    %b2 =w loadub %q
    %c3 =w ceqw %b2, %b
    jnz %c3, @rl4, @rend
@rl4
    %r =l add %r, 1
    jmp @rl
@rend
    %c4 =w csgel %r, 3
    jnz %c4, @rep, @lit
@rep
    %t =l add %o, 2
    %c5 =w csgtl %t, %cap
    jnz %c5, @fail, @rep2
@rep2
    %h =l sub 257, %r
    %hw =w copy %h
    %po =l add %out, %o
    storeb %hw, %po
    %po1 =l add %po, 1
    storeb %b, %po1
    %o =l copy %t
    %i =l add %i, %r
    jmp @loop
@lit
    %j =l copy %i
@ll
    %c6 =w csltl %j, %n
    jnz %c6, @ll2, @lend
@ll2
    %d =l sub %j, %i
    %c7 =w csltl %d, 128
    jnz %c7, @ll3, @lend
@ll3
    %j2 =l add %j, 2
    %c8 =w csltl %j2, %n
    jnz %c8, @ll4, @lnext
@ll4
    %pj =l add %src, %j
    %x0 =w loadub %pj
    %pj1 =l add %pj, 1
    %x1 =w loadub %pj1
    %c9 =w ceqw %x0, %x1
    jnz %c9, @ll5, @lnext
@ll5
    %pj2 =l add %pj, 2
    %x2 =w loadub %pj2
    %c10 =w ceqw %x0, %x2
    jnz %c10, @lend, @lnext
@lnext
    %j =l add %j, 1
    jmp @ll
@lend
    %L =l sub %j, %i
    %t2 =l add %o, 1
    %t2 =l add %t2, %L
    %c11 =w csgtl %t2, %cap
    jnz %c11, @fail, @lh
@lh
    %hl =l sub %L, 1
    %hlw =w copy %hl
    %ph =l add %out, %o
    storeb %hlw, %ph
    %m =l copy 0
@cl
    %c12 =w csltl %m, %L
    jnz %c12, @cb, @cend
@cb
    %si =l add %i, %m
    %ps =l add %src, %si
    %v =w loadub %ps
    %di =l add %o, 1
    %di =l add %di, %m
    %pd =l add %out, %di
    storeb %v, %pd
    %m =l add %m, 1
    jmp @cl
@cend
    %o =l copy %t2
    %i =l copy %j
    jmp @loop
@fail
    ret -1
@done
    ret %o
}
