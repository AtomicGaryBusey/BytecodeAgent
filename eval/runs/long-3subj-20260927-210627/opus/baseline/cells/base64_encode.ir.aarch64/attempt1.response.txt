export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @grp, @done
@grp
	%rem =l sub %n, %i
	%p =l add %src, %i
	%b0 =l loadub %p
	%v =l shl %b0, 16
	%k =l copy 1
	%c1 =w csgtl %rem, 1
	jnz %c1, @l1, @emit
@l1
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v =l or %v, %s1
	%k =l copy 2
	%c2 =w csgtl %rem, 2
	jnz %c2, @l2, @emit
@l2
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%v =l or %v, %b2
	%k =l copy 3
	jmp @emit
@emit
	%j =l copy 0
@inner
	%cj =w csltl %j, 4
	jnz %cj, @body, @next
@body
	%q =l add %out, %o
	%cp =w csgtl %j, %k
	jnz %cp, @pad, @chr
@pad
	storeb 61, %q
	jmp @adv
@chr
	%sh =l mul %j, 6
	%sh2 =l sub 18, %sh
	%x =l shr %v, %sh2
	%x =l and %x, 63
	%ch =l add %x, 65
	%g1 =w csgel %x, 26
	%e1 =l extuw %g1
	%t1 =l mul %e1, 6
	%ch =l add %ch, %t1
	%g2 =w csgel %x, 52
	%e2 =l extuw %g2
	%t2 =l mul %e2, 75
	%ch =l sub %ch, %t2
	%g3 =w csgel %x, 62
	%e3 =l extuw %g3
	%t3 =l mul %e3, 15
	%ch =l sub %ch, %t3
	%g4 =w csgel %x, 63
	%e4 =l extuw %g4
	%t4 =l mul %e4, 3
	%ch =l add %ch, %t4
	%chw =w copy %ch
	storeb %chw, %q
	jmp @adv
@adv
	%j =l add %j, 1
	%o =l add %o, 1
	jmp @inner
@next
	%i =l add %i, 3
	jmp @loop
@done
	ret %o
}
