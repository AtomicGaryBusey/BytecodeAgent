export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%k =l copy 0
@zl
	%cz =w csltl %k, %n
	jnz %cz, @zb, @zd
@zb
	%zo =l mul %k, 8
	%zp =l add %r, %zo
	storel 0, %zp
	%k =l add %k, 1
	jmp @zl
@zd
	%i =l copy 0
@il
	%ci =w csltl %i, %na
	jnz %ci, @ib, @count
@ib
	%ao =l mul %i, 8
	%ap =l add %a, %ao
	%x =l loadl %ap
	%x0 =l and %x, 4294967295
	%x1 =l shr %x, 32
	%carry =l copy 0
	%j =l copy 0
	%rp =l add %r, %ao
	%bp =l copy %b
@jl
	%cj =w csltl %j, %nb
	jnz %cj, @jb, @je
@jb
	%y =l loadl %bp
	%y0 =l and %y, 4294967295
	%y1 =l shr %y, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%m0 =l shr %p00, 32
	%m1 =l and %p01, 4294967295
	%m2 =l and %p10, 4294967295
	%mid =l add %m0, %m1
	%mid =l add %mid, %m2
	%lo0 =l and %p00, 4294967295
	%ms =l shl %mid, 32
	%lo =l or %lo0, %ms
	%h1 =l shr %p01, 32
	%h2 =l shr %p10, 32
	%h3 =l shr %mid, 32
	%hi =l add %p11, %h1
	%hi =l add %hi, %h2
	%hi =l add %hi, %h3
	%t =l loadl %rp
	%t =l add %t, %lo
	%c1 =w cultl %t, %lo
	%t =l add %t, %carry
	%c2 =w cultl %t, %carry
	storel %t, %rp
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%carry =l add %hi, %c1l
	%carry =l add %carry, %c2l
	%rp =l add %rp, 8
	%bp =l add %bp, 8
	%j =l add %j, 1
	jmp @jl
@je
	%cn =w csgtl %nb, 0
	jnz %cn, @st, @inext
@st
	storel %carry, %rp
@inext
	%i =l add %i, 1
	jmp @il
@count
	%k =l copy %n
@cl
	%ck =w csgtl %k, 0
	jnz %ck, @cb, @done
@cb
	%km =l sub %k, 1
	%ko =l mul %km, 8
	%kp =l add %r, %ko
	%v =l loadl %kp
	%nz =w cnel %v, 0
	jnz %nz, @done, @cdec
@cdec
	%k =l copy %km
	jmp @cl
@done
	ret %k
}
