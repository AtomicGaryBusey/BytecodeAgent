export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%b =w loadub %p
	%r =l copy 1
@rl
	%k =l add %i, %r
	%c1 =w csltl %k, %n
	jnz %c1, @rl2, @rend
@rl2
	%c2 =w csltl %r, 128
	jnz %c2, @rl3, @rend
@rl3
	%q =l add %src, %k
	%x =w loadub %q
	%c3 =w ceqw %x, %b
	jnz %c3, @rl4, @rend
@rl4
	%r =l add %r, 1
	jmp @rl
@rend
	%c4 =w csgel %r, 3
	jnz %c4, @rep, @lit
@rep
	%t =l add %o, 2
	%c5 =w csgtl %t, %cap
	jnz %c5, @fail, @rep2
@rep2
	%rw =w copy %r
	%h =w sub 257, %rw
	%po =l add %out, %o
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %t
	%i =l add %i, %r
	jmp @loop
@lit
	%j =l copy %i
@ll
	%d1 =w csltl %j, %n
	jnz %d1, @ll2, @lend
@ll2
	%dj =l sub %j, %i
	%d2 =w csltl %dj, 128
	jnz %d2, @ll3, @lend
@ll3
	%j1 =l add %j, 1
	%d3 =w csltl %j1, %n
	jnz %d3, @ll4, @ladv
@ll4
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %src, %j1
	%bj1 =w loadub %pj1
	%d4 =w ceqw %bj, %bj1
	jnz %d4, @ll5, @ladv
@ll5
	%j2 =l add %j, 2
	%d5 =w csltl %j2, %n
	jnz %d5, @ll6, @ladv
@ll6
	%pj2 =l add %src, %j2
	%bj2 =w loadub %pj2
	%d6 =w ceqw %bj, %bj2
	jnz %d6, @lend, @ladv
@ladv
	%j =l add %j, 1
	jmp @ll
@lend
	%L =l sub %j, %i
	%t2 =l add %o, 1
	%t3 =l add %t2, %L
	%c6 =w csgtl %t3, %cap
	jnz %c6, @fail, @lw
@lw
	%Lw =w copy %L
	%hl =w sub %Lw, 1
	%ph =l add %out, %o
	storeb %hl, %ph
	%m =l copy 0
@cp
	%c7 =w csltl %m, %L
	jnz %c7, @cp2, @cpend
@cp2
	%si =l add %i, %m
	%ps =l add %src, %si
	%v =w loadub %ps
	%di =l add %t2, %m
	%pd =l add %out, %di
	storeb %v, %pd
	%m =l add %m, 1
	jmp @cp
@cpend
	%o =l copy %t3
	%i =l copy %j
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
