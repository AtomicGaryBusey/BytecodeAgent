vm_run:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jae .Ldone
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 11
    ja .Lnext1
    test rax, rax
    je .Ldone
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    jmp .Ljmp
.Lnext1:
    inc r8
    jmp .Lloop
.Lpush:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    mov r10, qword ptr [rdx + r9*8 - 8]
    dec r9
    add qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Lsub:
    mov r10, qword ptr [rdx + r9*8 - 8]
    dec r9
    sub qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Lmul:
    mov r10, qword ptr [rdx + r9*8 - 8]
    dec r9
    imul r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Ldup:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 8], r11
    mov qword ptr [rdx + r9*8 - 16], r10
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    mov r10, qword ptr [rdx + r9*8 - 8]
    dec r9
    xor r11d, r11d
    cmp qword ptr [rdx + r9*8 - 8], r10
    setl r11b
    mov qword ptr [rdx + r9*8 - 8], r11
    inc r8
    jmp .Lloop
.Ljz:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    test r10, r10
    jnz .Lskip2
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Lskip2:
    add r8, 2
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ldone:
    xor eax, eax
    test r9, r9
    jz .Lzero
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lzero:
    cmp r9, rcx
    jae .Lret
    mov qword ptr [rdx + r9*8], 0
    inc r9
    jmp .Lzero
.Lret:
    ret
