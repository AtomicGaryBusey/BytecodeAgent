packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Lmain:
    cmp r8, rsi
    jge .Ldone
    movzx eax, byte ptr [rdi + r8]
    lea r10, [r8 + 2]
    cmp r10, rsi
    jge .Llit
    cmp al, byte ptr [rdi + r8 + 1]
    jne .Llit
    cmp al, byte ptr [rdi + r8 + 2]
    jne .Llit
    lea r10, [r8 + 3]
    lea r11, [r8 + 128]
    cmp r11, rsi
    cmovg r11, rsi
.Lrun:
    cmp r10, r11
    jge .Lrunend
    cmp al, byte ptr [rdi + r10]
    jne .Lrunend
    inc r10
    jmp .Lrun
.Lrunend:
    lea r11, [r9 + 2]
    cmp r11, rcx
    jg .Lfail
    mov r11, r8
    sub r11, r10
    inc r11
    mov byte ptr [rdx + r9], r11b
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    mov r8, r10
    jmp .Lmain
.Llit:
    mov r10, r8
    lea r11, [r8 + 128]
    cmp r11, rsi
    cmovg r11, rsi
.Llitloop:
    cmp r10, r11
    jge .Llitend
    lea rax, [r10 + 2]
    cmp rax, rsi
    jge .Llitnext
    movzx eax, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llitnext
    cmp al, byte ptr [rdi + r10 + 2]
    jne .Llitnext
    jmp .Llitend
.Llitnext:
    inc r10
    jmp .Llitloop
.Llitend:
    mov rax, r10
    sub rax, r8
    lea r11, [r9 + rax + 1]
    cmp r11, rcx
    jg .Lfail
    lea r11d, [eax - 1]
    mov byte ptr [rdx + r9], r11b
    inc r9
.Lcopy:
    movzx r11d, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    cmp r8, r10
    jl .Lcopy
    jmp .Lmain
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
