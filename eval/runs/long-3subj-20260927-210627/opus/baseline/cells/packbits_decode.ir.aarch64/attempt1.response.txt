export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%more =w csltl %i, %n
	jnz %more, @header, @done
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%i =l add %i, 1
	%isnop =w ceql %h, -128
	jnz %isnop, @loop, @kind
@kind
	%islit =w csgel %h, 0
	jnz %islit, @lit, @rep
@lit
	%cnt =l add %h, 1
	%end =l add %i, %cnt
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @litloop
@litloop
	%lmore =w csltl %i, %end
	jnz %lmore, @litcheck, @loop
@litcheck
	%full =w csgel %o, %cap
	jnz %full, @err1, @litstore
@litstore
	%sp =l add %src, %i
	%c =w loadub %sp
	%op =l add %out, %o
	storeb %c, %op
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @litloop
@rep
	%rcnt =l sub 1, %h
	%need =l add %i, 1
	%rbad =w csgtl %need, %n
	jnz %rbad, @err2, @repread
@repread
	%rp =l add %src, %i
	%rc =w loadub %rp
	%i =l add %i, 1
	%k =l copy 0
@reploop
	%rmore =w csltl %k, %rcnt
	jnz %rmore, @repcheck, @loop
@repcheck
	%rfull =w csgel %o, %cap
	jnz %rfull, @err1, @repstore
@repstore
	%rop =l add %out, %o
	storeb %rc, %rop
	%o =l add %o, 1
	%k =l add %k, 1
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	ret %o
}
