export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%done =w csgel %i, %n
	jnz %done, @ret_ok, @hdr
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%hl =l extsw %h
	%i =l add %i, 1
	%isnop =w ceql %hl, -128
	jnz %isnop, @loop, @kind
@kind
	%islit =w csgel %hl, 0
	jnz %islit, @lit, @rep
@lit
	%cnt =l add %hl, 1
	%end =l add %i, %cnt
	%bad =w csgtl %end, %n
	jnz %bad, @ret_src, @litloop
@litloop
	%ld =w ceql %cnt, 0
	jnz %ld, @loop, @litbody
@litbody
	%full =w csgel %o, %cap
	jnz %full, @ret_cap, @litstore
@litstore
	%sp =l add %src, %i
	%c =w loadub %sp
	%op =l add %out, %o
	storeb %c, %op
	%o =l add %o, 1
	%i =l add %i, 1
	%cnt =l sub %cnt, 1
	jmp @litloop
@rep
	%bad2 =w csgel %i, %n
	jnz %bad2, @ret_src, @repinit
@repinit
	%sp2 =l add %src, %i
	%rc =w loadub %sp2
	%i =l add %i, 1
	%rcnt =l sub 1, %hl
@reploop
	%rd =w ceql %rcnt, 0
	jnz %rd, @loop, @repbody
@repbody
	%full2 =w csgel %o, %cap
	jnz %full2, @ret_cap, @repstore
@repstore
	%op2 =l add %out, %o
	storeb %rc, %op2
	%o =l add %o, 1
	%rcnt =l sub %rcnt, 1
	jmp @reploop
@ret_ok
	ret %o
@ret_src
	ret -2
@ret_cap
	ret -1
}
