export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@zl
	%c =w cslel %j, %n
	jnz %c, @zb, @zd
@zb
	%o =l mul %j, 8
	%p =l add %row, %o
	storel 0, %p
	%j =l add %j, 1
	jmp @zl
@zd
	%i =l copy 0
@il
	%c =w csltl %i, %m
	jnz %c, @ib, @done
@ib
	%xp =l add %x, %i
	%xi =w loadub %xp
	%diag =l copy 0
	%left =l copy 0
	%j =l copy 1
@jl
	%c =w cslel %j, %n
	jnz %c, @jb, @inext
@jb
	%o =l mul %j, 8
	%p =l add %row, %o
	%up =l loadl %p
	%yp =l add %y, %j
	%yp =l sub %yp, 1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%v =l add %diag, 1
	jmp @store
@nomatch
	%v =l copy %up
	%g =w csltl %up, %left
	jnz %g, @useleft, @store
@useleft
	%v =l copy %left
@store
	storel %v, %p
	%left =l copy %v
	%diag =l copy %up
	%j =l add %j, 1
	jmp @jl
@inext
	%i =l add %i, 1
	jmp @il
@done
	%o =l mul %n, 8
	%p =l add %row, %o
	%r =l loadl %p
	ret %r
}
