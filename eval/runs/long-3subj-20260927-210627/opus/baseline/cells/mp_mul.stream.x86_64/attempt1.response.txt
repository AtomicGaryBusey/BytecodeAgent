mp_mul:
    push rbx
    push r12
    push r13
    push r14
    mov r10, rdx
    lea r9, [rsi + rcx]
    xor eax, eax
    xor r11d, r11d
.Lz:
    cmp r11, r9
    jae .Lzd
    mov qword ptr [r8 + r11*8], rax
    inc r11
    jmp .Lz
.Lzd:
    test rcx, rcx
    jz .Lcount
    xor r11d, r11d
.Li:
    cmp r11, rsi
    jae .Lcount
    mov rbx, qword ptr [rdi + r11*8]
    lea r12, [r8 + r11*8]
    xor r13d, r13d
    xor r14d, r14d
.Lj:
    mov rax, rbx
    mul qword ptr [r10 + r13*8]
    add rax, qword ptr [r12 + r13*8]
    adc rdx, 0
    add rax, r14
    adc rdx, 0
    mov qword ptr [r12 + r13*8], rax
    mov r14, rdx
    inc r13
    cmp r13, rcx
    jb .Lj
    mov qword ptr [r12 + rcx*8], r14
    inc r11
    jmp .Li
.Lcount:
    mov rax, r9
.Lc:
    test rax, rax
    jz .Ldone
    cmp qword ptr [r8 + rax*8 - 8], 0
    jne .Ldone
    dec rax
    jmp .Lc
.Ldone:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
