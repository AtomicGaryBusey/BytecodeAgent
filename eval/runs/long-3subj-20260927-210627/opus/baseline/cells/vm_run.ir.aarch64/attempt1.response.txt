export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%e =w ceql %pc, %len
	jnz %e, @end, @fetch
@fetch
	%o1 =l mul %pc, 8
	%ia =l add %code, %o1
	%op =l loadl %ia
	%ib =l add %ia, 8
	%t1 =l mul %d, 8
	%sp =l add %stack, %t1
	%tp =l sub %sp, 8
	%ap =l sub %sp, 16
	%c0 =w ceql %op, 0
	jnz %c0, @end, @n1
@n1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @n2
@n2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @n3
@n3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @n4
@n4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @n5
@n5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @n6
@n6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @n7
@n7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @n8
@n8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @n9
@n9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @n10
@n10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @n11
@n11
	%c11 =w ceql %op, 11
	jnz %c11, @jmp, @next1
@push
	%v =l loadl %ib
	storel %v, %sp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%a2 =l loadl %ap
	%b2 =l loadl %tp
	%r2 =l add %a2, %b2
	storel %r2, %ap
	%d =l sub %d, 1
	jmp @next1
@sub
	%a3 =l loadl %ap
	%b3 =l loadl %tp
	%r3 =l sub %a3, %b3
	storel %r3, %ap
	%d =l sub %d, 1
	jmp @next1
@mul
	%a4 =l loadl %ap
	%b4 =l loadl %tp
	%r4 =l mul %a4, %b4
	storel %r4, %ap
	%d =l sub %d, 1
	jmp @next1
@dup
	%a5 =l loadl %tp
	storel %a5, %sp
	%d =l add %d, 1
	jmp @next1
@swap
	%a6 =l loadl %ap
	%b6 =l loadl %tp
	storel %b6, %ap
	storel %a6, %tp
	jmp @next1
@drop
	%d =l sub %d, 1
	jmp @next1
@over
	%a8 =l loadl %ap
	storel %a8, %sp
	%d =l add %d, 1
	jmp @next1
@lt
	%a9 =l loadl %ap
	%b9 =l loadl %tp
	%w9 =w csltl %a9, %b9
	%r9 =l extsw %w9
	storel %r9, %ap
	%d =l sub %d, 1
	jmp @next1
@jz
	%v10 =l loadl %tp
	%d =l sub %d, 1
	%z10 =w ceql %v10, 0
	jnz %z10, @jmp, @jzno
@jzno
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%pc =l loadl %ib
	jmp @loop
@next1
	%pc =l add %pc, 1
	jmp @loop
@end
	%i =l copy %d
@zloop
	%zc =w csltl %i, %cap
	jnz %zc, @zbody, @ret
@zbody
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@ret
	%emp =w ceql %d, 0
	jnz %emp, @ret0, @rettop
@ret0
	ret 0
@rettop
	%to =l mul %d, 8
	%tpp =l add %stack, %to
	%tpp2 =l sub %tpp, 8
	%res =l loadl %tpp2
	ret %res
}
