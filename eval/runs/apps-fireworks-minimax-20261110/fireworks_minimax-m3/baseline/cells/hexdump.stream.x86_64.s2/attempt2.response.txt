.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .space 65536
out_buf: .space 80

.section .text
.globl _start
_start:
    xor r15d, r15d
    lea r12, [rip + input_buf]

read_more:
    mov eax, 0
    mov edi, 0
    mov rsi, r12
    mov edx, 65536
    sub edx, r15d
    syscall
    cmp rax, 0
    jle read_done
    add r15d, eax
    add r12, rax
    cmp r15d, 65536
    jge read_done
    jmp read_more

read_done:
    xor r12d, r12d

row_loop:
    cmp r12d, r15d
    jge done_rows
    mov r13d, r15d
    sub r13d, r12d
    cmp r13d, 16
    jle n_set
    mov r13d, 16
n_set:
    mov rdi, r12
    call format_offset
    mov word ptr [out_buf + 8], 0x2020
    call format_byte_slots
    mov word ptr [out_buf + 59], 0x7C20
    call format_ascii
    lea rcx, [r13 + 61]
    mov word ptr [out_buf + rcx], 0x0A7C
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    lea rdx, [r13 + 63]
    syscall
    add r12d, 16
    jmp row_loop

done_rows:
    mov rdi, r15
    call format_offset
    mov byte ptr [out_buf + 8], 0x0A
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, 9
    syscall
    mov eax, 60
    xor edi, edi
    syscall

format_offset:
    lea rsi, [rip + hex_chars]
    mov rcx, 8
    mov rax, rdi
fmt_off_loop:
    dec ecx
    mov rdx, rax
    and edx, 0x0F
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [out_buf + rcx], dl
    shr rax, 4
    test ecx, ecx
    jnz fmt_off_loop
    ret

format_byte_slots:
    lea rsi, [rip + hex_chars]
    xor ecx, ecx
    mov r14d, 10
fbs_loop:
    cmp ecx, 16
    je fbs_done
    cmp ecx, r13d
    jae fbs_blank
    lea rax, [r12 + rcx]
    movzx eax, byte ptr [input_buf + rax]
    mov edx, eax
    shr al, 4
    and dl, 0x0F
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [out_buf + r14], al
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [out_buf + r14 + 1], dl
    mov byte ptr [out_buf + r14 + 2], ' '
    jmp fbs_after
fbs_blank:
    mov byte ptr [out_buf + r14], ' '
    mov byte ptr [out_buf + r14 + 1], ' '
    mov byte ptr [out_buf + r14 + 2], ' '
fbs_after:
    add r14d, 3
    cmp ecx, 7
    jne fbs_no_extra
    mov byte ptr [out_buf + r14], ' '
    inc r14d
fbs_no_extra:
    inc ecx
    jmp fbs_loop
fbs_done:
    ret

format_ascii:
    xor ecx, ecx
fas_loop:
    cmp ecx, r13d
    je fas_done
    lea rax, [r12 + rcx]
    movzx edx, byte ptr [input_buf + rax]
    cmp dl, 0x20
    jb fas_dot
    cmp dl, 0x7E
    ja fas_dot
    jmp fas_print
fas_dot:
    mov dl, '.'
fas_print:
    mov byte ptr [out_buf + 61 + rcx], dl
    inc ecx
    jmp fas_loop
fas_done:
    ret
