.section .rodata
hex_table: .ascii "0123456789abcdef"

.section .text
.globl _start

# hex_nibble: convert low nibble of al to ASCII hex digit.
# Input: al = value (upper bits ignored).
# Output: al = ASCII character.
# Clobbers: rax.
hex_nibble:
    and al, 0x0F
    movzx eax, byte ptr [rip + hex_table + rax]
