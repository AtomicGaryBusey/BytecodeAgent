.section .rodata

.section .bss
.align 4

.align 4

.align 4

.section .text
.globl _start

    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
1:  cmp x19, #65536
    b.ge 2f
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19

    svc #0
    cmp x0, #0

    b.ge 5f

    add x1, x22, x20

    add x21, x22, #1

    mov x8, #93

process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!

    mov x20, x1
    mov x21, x0
    mov x22, #0
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack

skip_ws:
    cmp x21, x20
    b.ge end_check
    ldrb w0, [x21]
    cmp w0, #0x20
    b.eq ws_inc
    cmp w0, #0x09
    b.ne token_begin

    add x21, x21, #1
    b skip_ws

token_begin:
    mov x25, x21
find_end:
    cmp x25, x20
    b.ge classify
    ldrb w0, [x25]
    cmp w0, #0x20
    b.eq classify
    cmp w0, #0x09
    b.eq classify
    add x25, x25, #1
    b find_end

classify:
    sub x26, x25, x21
    cmp x26, #1
    b.ne try_literal
    ldrb w0, [x21]
    cmp w0, #0x2b
    b.eq op_add
    cmp w0, #0x2d
    b.eq op_sub
    cmp w0, #0x2a
    b.eq op_mul
    cmp w0, #0x2f
    b.eq op_div
    cmp w0, #0x25
    b.eq op_mod
    b try_literal

try_literal:
    cbz x26, do_error
    mov x28, #0
    ldrb w0, [x21]
    cmp w0, #0x2d
    b.ne parse_setup
    mov x28, #1
parse_setup:
    mov x5, x21
    cbz x28, parse_start
    add x5, x5, #1
parse_start:
    mov x3, #0
    mov x4, #0
    b parse_loop

parse_loop:
    cmp x5, x20
    b.ge after_loop
    ldrb w0, [x5]
    cmp w0, #0x20
    b.eq after_loop
    cmp w0, #0x09
    b.eq after_loop
    cmp w0, #0x30
    b.lt do_error
    cmp w0, #0x39
    b.gt do_error
    sub w0, w0, #0x30
    mov w6, w0
    mov x7, #10
    mul x3, x3, x7
    add x3, x3, x6
    add x4, x4, #1
    add x5, x5, #1
    cmp x4, #18
    b.gt do_error
    b parse_loop

    cbz x4, do_error
    cbz x28, push_value
    neg x3, x3
    b push_value

    str x3, [x24, x22, lsl #3]
    add x22, x22, #1
    b next_token

op_add:
    cmp x22, #2
    b.lt do_error
    sub x22, x22, #1
    ldr x1, [x24, x22, lsl #3]
    sub x22, x22, #1
    ldr x0, [x24, x22, lsl #3]

    str x0, [x24, x22, lsl #3]

    b next_token

op_sub:
    cmp x22, #2
    b.lt do_error
    sub x22, x22, #1
    ldr x1, [x24, x22, lsl #3]
    sub x22, x22, #1
    ldr x0, [x24, x22, lsl #3]

    str x0, [x24, x22, lsl #3]
    add x22, x22, #1
    b next_token

    b.lt do_error
    sub x22, x22, #1
    ldr x1, [x24, x22, lsl #3]
    sub x22, x22, #1
    ldr x0, [x24, x22, lsl #3]

    b.lt do_error
    sub x22, x22, #1
    ldr x1, [x24, x22, lsl #3]

    cmp x22, #2
    b.lt do_error
    sub x22, x22, #1
    ldr x1, [x24, x22, lsl #3]
    cbz x1, do_error
    sub x22, x22, #1
    ldr x0, [x24, x22, lsl #3]
    mov x2, #1
    lsl x2, x2, #63
    cmp x0, x2
