.section .rodata
err_msg: .ascii "error\n"

.section .bss
.align 4
input_buf: .space 65536
.align 4
value_stack: .space 8192

.section .text
.globl _start

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
1:  mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq 2f
    add x19, x19, x0
    cmp x19, #65536
    b.lt 1b
2:  mov x21, #0
3:  cmp x21, x19
    b.ge 9f
    mov x22, x21
4:  cmp x22, x19
    b.ge 5f
    add x23, x20, x22
    ldrb w0, [x23]
    cmp w0, #0x0a
    b.eq 5f
    add x22, x22, #1
    b 4b
5:  add x0, x20, x21
    add x1, x20, x22
    bl process_line
    add x21, x22, #1
    b 3b
9:  mov x0, #0
    mov x8, #93
    svc #0

process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    stp x25, x26, [sp, #-16]!
    stp x27, x28, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, x0
    mov x22, #0
    mov x23, #0
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack
    b loop_check

skip_ws:
    cmp x21, x20
    b.ge token_classify
    ldrb w0, [x21]
    cmp w0, #0x20
    b.eq ws_step
    cmp w0, #0x09
    b.ne token_start
ws_step:
    add x21, x21, #1
    b skip_ws

token_start:
    mov x25, x21
find_end:
    cmp x25, x20
    b.ge token_classify
    ldrb w0, [x25]
    cmp w0, #0x20
    b.eq token_classify
    cmp w0, #0x09
    b.eq token_classify
    add x25, x25, #1
    b find_end

token_classify:
    cmp x21, x25
    b.ge loop_check
    sub x26, x25, x21
    cmp x26, #1
    b.ne length_gt_1
    ldrb w0, [x21]
    cmp w0, #0x2b
    b.eq op_add
    cmp w0, #0x2d
    b.eq op_sub
    cmp w0, #0x2a
    b.eq op_mul
    cmp w0, #0x2f
    b.eq op_div
    cmp w0, #0x25
    b.eq op_mod
    b parse_single_digit

length_gt_1:
    ldrb w0, [x21]
    cmp w0, #0x2d
    b.ne positive_number
    sub x26, x26, #1
    cmp x26, #0
    b.le set_error
    cmp x26, #18
    b.gt set_error
    mov x27, x21
    add x27, x27, #1
digit_loop_neg:
    cmp x27, x25
    b.ge parse_neg_num
    ldrb w0, [x27]
    cmp w0, #'0'
    b.lt set_error
    cmp w0, #'9'
    b.gt set_error
    add x27, x27, #1
    b digit_loop_neg

parse_neg_num:
    mov x0, x21
    mov x1, x25
    bl parse_signed_number
    str x0, [x24, x22, lsl #3]
    add x22, x22, #1
    b next_token

positive_number:
    cmp w0, #'0'
    b.lt set_error
    cmp w0, #'9'
    b.gt set_error
    cmp x26, #1
    b.lt set_error
    cmp x26, #18
    b.gt set_error
    mov x27, x21
digit_loop_pos:
    cmp x27, x25
    b.ge parse_pos_num
    ldrb w0, [x27]
    cmp w0, #'0'
    b.lt set_error
    cmp w0, #'9'
    b.gt set_error
    add x27, x27, #1
    b digit_loop_pos

parse_pos_num:
    mov x0, x21
    mov x1, x25
    bl parse_signed_number
    str x0, [x24, x22, lsl #3]
    add x22, x22, #1
    b next_token

parse_single_digit:
    ldrb w0, [x21]
    cmp w0, #'0'
    b.lt set_error
    cmp w0, #'9'
    b.gt set_error
    sub w0, w0, #'0'
    mov x0, w0, uxtw
