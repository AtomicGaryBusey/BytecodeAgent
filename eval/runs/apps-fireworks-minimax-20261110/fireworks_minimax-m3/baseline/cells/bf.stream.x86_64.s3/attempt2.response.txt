.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 4096

.section .bss
.align 16
inbuf: .zero 65536
.align 16
outbuf: .zero 4096
tape: .zero 30000
.align 4
match_tbl: .zero 262144
bracket_stack: .zero 4000

.section .text
.globl _start
_start:
    lea r15, [rip + inbuf]
    xor ecx, ecx
    mov edx, INBUF_SIZE
    mov rsi, r15
.read_loop:
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jz .read_done
    js .read_err
    add ecx, eax
    add rsi, rax
    sub edx, eax
    jnz .read_loop
.read_done:
    xor edi, edi
.find_excl:
    cmp edi, ecx
    je .no_excl
    mov al, byte ptr [r15 + rdi]
    cmp al, 0x21
    je .found_excl
    inc edi
    jmp .find_excl
.no_excl:
    mov r8d, ecx
    xor r9d, r9d
    jmp .split_done
.found_excl:
    mov r8d, edi
    mov r9, rcx
    sub r9, rdi
    dec r9
.split_done:
    lea r12, [rip + inbuf]
    lea rbp, [r12 + r8 + 1]

    lea r13, [rip + match_tbl]
    lea rdi, [rip + bracket_stack]
    xor ecx, ecx
    xor esi, esi
.scan_loop:
    cmp ecx, r8d
    je .scan_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    je .scan_open
    cmp al, 0x5D
    je .scan_close
    jmp .scan_next
.scan_open:
    mov dword ptr [rdi + rsi*4], ecx
    inc esi
    jmp .scan_next
.scan_close:
    test esi, esi
    jz .bracket_mismatch
    dec esi
    mov eax, dword ptr [rdi + rsi*4]
    mov dword ptr [r13 + rax*4], ecx
    mov dword ptr [r13 + rcx*4], eax
.scan_next:
    inc ecx
    jmp .scan_loop
.scan_done:
    test esi, esi
    jnz .bracket_mismatch

    xor ecx, ecx
    xor edx, edx
    lea r14, [rip + tape]
    xor ebx, ebx
    lea r15, [rip + outbuf]
    xor r10d, r10d
.run_loop:
    cmp ecx, r8d
    je .run_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x3E
    je .op_right
    cmp al, 0x3C
    je .op_left
    cmp al, 0x2B
    je .op_inc
    cmp al, 0x2D
    je .op_dec
    cmp al, 0x2E
    je .op_out
    cmp al, 0x2C
    je .op_in
    cmp al, 0x5B
    je .op_jmp_fwd
    cmp al, 0x5D
    je .op_jmp_back
    jmp .run_next
.op_right:
    inc edx
    jmp .run_next
.op_left:
    dec edx
    jmp .run_next
.op_inc:
    inc byte ptr [r14 + rdx]
    jmp .run_next
.op_dec:
    dec byte ptr [r14 + rdx]
    jmp .run_next
.op_out:
    mov al, byte ptr [r14 + rdx]
    mov byte ptr [r15 + r10], al
    inc r10d
    cmp r10d, OUTBUF_SIZE
    jne .run_next
    mov eax, SYS_WRITE
    mov rsi, r15
    mov edi, 1
    mov edx, OUTBUF_SIZE
    syscall
    xor r10d, r10d
    jmp .run_next
.op_in:
    cmp ebx, r9d
    jae .op_in_zero
    mov al, byte ptr [rbp + rbx]
    inc ebx
    jmp .op_in_store
.op_in_zero:
    xor eax, eax
.op_in_store:
    mov byte ptr [r14 + rdx], al
    jmp .run_next
.op_jmp_fwd:
    cmp byte ptr [r14 + rdx], 0
    jne .run_next
    mov eax, dword ptr [r13 + rcx*4]
    inc eax
    mov ecx, eax
    jmp .run_loop
.op_jmp_back:
    cmp byte ptr [r14 + rdx], 0
    je .run_next
    mov eax, dword ptr [r13 + rcx*4]
    inc eax
    mov ecx, eax
    jmp .run_loop
.run_next:
    inc ecx
    jmp .run_loop
.run_done:
    test r10d, r10d
    jz .run_exit
    mov eax, SYS_WRITE
    mov rsi, r15
    mov edi, 1
    mov edx, r10d
    syscall
.run_exit:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
.bracket_mismatch:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
.read_err:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
