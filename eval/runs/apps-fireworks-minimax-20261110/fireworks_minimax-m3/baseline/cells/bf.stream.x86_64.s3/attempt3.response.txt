.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 4096

.section .bss
.align 16
in_buf: .zero 65536
.align 16
out_buf: .zero 4096
tape: .zero 30000
.align 4
match_tbl: .zero 262144
.align 4
bracket_stack: .zero 4000

.section .text
.globl _start
_start:
    lea r15, [rip + in_buf]
    xor ecx, ecx
    mov edx, IN_BUF_SIZE
    mov rsi, r15

1:
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jz 2f
    js 9f
    add ecx, eax
    add rsi, rax
    sub edx, eax
    jnz 1b

2:
    xor edi, edi
3:
    cmp edi, ecx
    je 4f
    mov al, byte ptr [r15 + rdi]
    cmp al, 0x21
    je 5f
    inc edi
    jmp 3b

4:
    mov r8d, ecx
    xor r9d, r9d
    jmp 6f

5:
    mov r8d, edi
    mov r9, rcx
    sub r9, rdi
    dec r9

6:
    lea r12, [rip + in_buf]
    lea rbp, [r12 + r8 + 1]
    lea r13, [rip + match_tbl]
    lea rdi, [rip + bracket_stack]
    xor ecx, ecx
    xor esi, esi

7:
    cmp ecx, r8d
    je 8f
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    je 71f
    cmp al, 0x5D
    je 72f
    jmp 73f

71:
    mov dword ptr [rdi + rsi*4], ecx
    inc esi
    jmp 73f

72:
    test esi, esi
    jz 9f
    dec esi
    mov eax, dword ptr [rdi + rsi*4]
    mov dword ptr [r13 + rax*4], ecx
    mov dword ptr [r13 + rcx*4], eax

73:
    inc ecx
    jmp 7b

8:
    test esi, esi
    jnz 9f

    xor ecx, ecx
    xor edx, edx
    lea r14, [rip + tape]
    xor ebx, ebx
    lea r15, [rip + out_buf]
    xor r10d, r10d

10:
    cmp ecx, r8d
    je 20f
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x3E
    je 11f
    cmp al, 0x3C
    je 12f
    cmp al, 0x2B
    je 13f
    cmp al, 0x2D
    je 14f
    cmp al, 0x2E
    je 15f
    cmp al, 0x2C
    je 16f
    cmp al, 0x5B
    je 17f
    cmp al, 0x5D
    je 18f
    jmp 19f

11:
    inc edx
    jmp 19f

12:
    dec edx
    jmp 19f

13:
    inc byte ptr [r14 + rdx]
    jmp 19f

14:
    dec byte ptr [r14 + rdx]
    jmp 19f

15:
    mov al, byte ptr [r14 + rdx]
    mov byte ptr [r15 + r10], al
    inc r10d
    cmp r10d, OUT_BUF_SIZE
    jne 19f
    mov eax, SYS_WRITE
    mov rsi, r15
    mov edi, 1
    mov edx, OUT_BUF_SIZE
    syscall
    xor r10d, r10d
    jmp 19f

16:
    cmp ebx, r9d
    jae 161f
    mov al, byte ptr [rbp + rbx]
    inc ebx
    jmp 162f

161:
    xor eax, eax

162:
    mov byte ptr [r14 + rdx], al
    jmp 19f

17:
    cmp byte ptr [r14 + rdx], 0
    jne 19f
    mov eax, dword ptr [r13 + rcx*4]
    inc eax
    mov ecx, eax
    jmp 10b

18:
    cmp byte ptr [r14 + rdx], 0
    je 19f
    mov eax, dword ptr [r13 + rcx*4]
    inc eax
    mov ecx, eax
    jmp 10b

19:
    inc ecx
    jmp 10b

20:
    test r10d, r10d
    jz 30f
    mov eax, SYS_WRITE
    mov rsi, r15
    mov edi, 1
    mov edx, r10d
    syscall

30:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

9:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
