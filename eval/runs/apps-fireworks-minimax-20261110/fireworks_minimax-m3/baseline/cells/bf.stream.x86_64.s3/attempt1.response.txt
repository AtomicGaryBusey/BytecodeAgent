.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFFER_SIZE, 4096
.equ INBUF_SIZE, 65536

.section .bss
.align 16
inbuf: .zero 65536
outbuf: .zero 4096
tape: .zero 30000
match_tbl: .zero 262144
bracket_stack: .zero 4000

.section .text
.globl _start
_start:
    lea r15, [rip + inbuf]        # base of input buffer

    xor ecx, ecx                  # total bytes read
    mov edx, INBUF_SIZE
    mov rsi, r15
.read_loop:
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jz .read_done
    js .read_err
    add ecx, eax
    add rsi, rax
    sub edx, eax
    jnz .read_loop

.read_done:
    # search for first '!' (0x21)
    xor edx, edx
.find_excl:
    cmp edx, ecx
    je .no_excl
    mov al, byte ptr [r15 + rdx]
    cmp al, 0x21
    je .found_excl
    inc edx
    jmp .find_excl

.no_excl:
    mov r8, rcx                    # program_len = total
    xor r9d, r9d                   # input_len = 0
    jmp .split_done

.found_excl:
    mov r8, rdx                    # program_len = index of '!'
    mov r9, rcx
    sub r9, rdx
    dec r9                         # input_len = total - pos -1

.split_done:
    mov r12, r15                   # program base = input buffer
    lea rbp, [r15 + r8 + 1]        # input base (after program and '!')

    lea r13, [rip + match_tbl]     # match table base
    lea rdi, [rip + bracket_stack] # stack for bracket matching
    xor ecx, ecx                   # ip = 0
    xor esi, esi                   # depth = 0

.scan_loop:
    cmp ecx, r8
