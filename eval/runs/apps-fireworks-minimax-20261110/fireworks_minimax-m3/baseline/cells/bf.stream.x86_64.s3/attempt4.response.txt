.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 4096

.section .bss
.align 16
in_buf: .zero 65536
.align 16
out_buf: .zero 4096
.align 4
tape: .zero 30000
.align 4
match_tbl: .zero 262144
.align 4
bracket_stack: .zero 4000

.section .text
.globl _start
_start:
    lea r12, [rip + in_buf]
    lea r13, [rip + match_tbl]
    lea r14, [rip + tape]
    lea r15, [rip + out_buf]

    xor ecx, ecx
    mov edx, IN_BUF_SIZE
    mov rsi, r12
.Lread_loop:
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jz .Lread_done
    js .Lexit_err
    add ecx, eax
    add rsi, rax
    sub edx, eax
    jnz .Lread_loop
.Lread_done:
    xor edi, edi
.Lscan_loop:
    cmp edi, ecx
    jae .Lno_bang
    cmp byte ptr [r12 + rdi], 0x21
    je .Lfound_bang
    inc edi
    jmp .Lscan_loop
.Lno_bang:
    mov r8d, ecx
    xor r9d, r9d
    jmp .Lsplit_done
.Lfound_bang:
    mov r8d, edi
    mov r9d, ecx
    sub r9d, edi
    dec r9d
.Lsplit_done:
    lea rbp, [r12 + r8 + 1]

    lea rdi, [rip + bracket_stack]
    xor ecx, ecx
    xor esi, esi
.Lbracket_loop:
    cmp ecx, r8d
    jae .Lbracket_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    je .Lleft_bracket
    cmp al, 0x5D
    je .Lright_bracket
    jmp .Lbracket_next
.Lleft_bracket:
    cmp esi, 1000
    jae .Lexit_err
    mov dword ptr [rdi + rsi*4], ecx
    inc esi
    jmp .Lbracket_next
.Lright_bracket:
    test esi, esi
    jz .Lexit_err
    dec esi
    mov eax, dword ptr [rdi + rsi*4]
    mov edx, ecx
    mov dword ptr [r13 + rax*4], edx
    mov dword ptr [r13 + rdx*4], eax
    jmp .Lbracket_next
.Lbracket_next:
    inc ecx
    jmp .Lbracket_loop
.Lbracket_done:
    test esi, esi
    jnz .Lexit_err

    xor ecx, ecx
    xor r10d, r10d
    xor ebx, ebx
    xor r11d, r11d
.Lexec_loop:
    cmp ecx, r8d
    jae .Lexec_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x3E
    je .Lcmd_gt
    cmp al, 0x3C
    je .Lcmd_lt
    cmp al, 0x2B
    je .Lcmd_inc
    cmp al, 0x2D
    je .Lcmd_dec
    cmp al, 0x2E
    je .Lcmd_out
    cmp al, 0x2C
    je .Lcmd_in
    cmp al, 0x5B
    je .Lcmd_lbr
    cmp al, 0x5D
    je .Lcmd_rbr
    jmp .Lexec_next
.Lcmd_gt:
    inc r10d
    jmp .Lexec_next
.Lcmd_lt:
    dec r10d
    jmp .Lexec_next
.Lcmd_inc:
    inc byte ptr [r14 + r10]
    jmp .Lexec_next
.Lcmd_dec:
    dec byte ptr [r14 + r10]
    jmp .Lexec_next
.Lcmd_out:
    mov al, byte ptr [r14 + r10]
    mov byte ptr [r15 + r11], al
    inc r11d
    cmp r11d, OUT_BUF_SIZE
    jb .Lexec_next
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r15
    mov edx, OUT_BUF_SIZE
    syscall
    xor r11d, r11d
    jmp .Lexec_next
.Lcmd_in:
    cmp ebx, r9d
    jae .Lcmd_in_eof
    mov al, byte ptr [rbp + rbx]
    inc ebx
    jmp .Lcmd_in_store
.Lcmd_in_eof:
    xor eax, eax
.Lcmd_in_store:
    mov byte ptr [r14 + r10], al
    jmp .Lexec_next
.Lcmd_lbr:
    cmp byte ptr [r14 + r10], 0
    jne .Lexec_next
    mov ecx, dword ptr [r13 + rcx*4]
    inc ecx
    jmp .Lexec_loop
.Lcmd_rbr:
    cmp byte ptr [r14 + r10], 0
    je .Lexec_next
    mov ecx, dword ptr [r13 + rcx*4]
    inc ecx
    jmp .Lexec_loop
.Lexec_next:
    inc ecx
    jmp .Lexec_loop
.Lexec_done:
    test r11d, r11d
    jz .Lexit_ok
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r15
    mov edx, r11d
    syscall
.Lexit_ok:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
.Lexit_err:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
