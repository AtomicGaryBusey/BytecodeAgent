.text
.globl _start
_start:
    lea r15, [rip + buffer]
    xor r12d, r12d

.L_read:
    mov eax, 0
    mov edi, 0
    lea rsi, [r15 + r12]
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jz .L_read_done
    js .L_err_exit
    add r12d, eax
    cmp r12d, 65536
    je .L_read_done
    jmp .L_read

.L_read_done:
    mov r14d, r12d
    xor r12d, r12d

.L_next_line:
    cmp r12d, r14d
    jae .L_exit
    mov r13d, r12d

.L_find_nl:
    cmp r13d, r14d
    jae .L_process
    cmp byte ptr [r15 + r13], 0x0A
    je .L_process
    inc r13d
    jmp .L_find_nl

.L_process:
    cmp r12d, r13d
    je .L_write_nl

    mov esi, r13d
    dec esi
    mov ebx, r12d

.L_rev:
    cmp ebx, esi
    jae .L_write_line
    movzx eax, byte ptr [r15 + rbx]
    movzx edx, byte ptr [r15 + rsi]
    mov byte ptr [r15 + rsi], al
    mov byte ptr [r15 + rbx], dl
    inc ebx
    dec esi
    jmp .L_rev

.L_write_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + r12]
    mov edx, r13d
    sub edx, r12d
    syscall

.L_write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    cmp r13d, r14d
    jae .L_exit
    lea r12d, [r13d + 1]
    jmp .L_next_line

.L_exit:
    mov eax, 60
    xor edi, edi
    syscall

.L_err_exit:
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
newline: .byte 0x0A

.bss
.p2align 4
buffer: .space 65536
