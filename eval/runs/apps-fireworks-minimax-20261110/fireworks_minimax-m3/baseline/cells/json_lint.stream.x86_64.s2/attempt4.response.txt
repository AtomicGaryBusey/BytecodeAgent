.section .rodata
error_msg: .ascii "error at "
ok_msg:   .ascii "ok\n"
newline:  .ascii "\n"

.section .bss
.align 8
cur_off:      .space 8
end_off:      .space 8
error_off:    .space 8
input_buffer: .space 65536

.section .text
.globl _start

_start:
    lea rdi, [rip + input_buffer]
    mov rsi, rdi
    mov rdx, 65536
.read_loop:
    mov rax, 0
    mov rdi, 0
    syscall
    cmp rax, 0
    jle .read_done
    add rsi, rax
    sub rdx, rax
    jnz .read_loop
.read_done:
    sub rsi, rdi
    mov [rip + end_off], rsi
    mov qword ptr [rip + cur_off], 0
    call parse_document

parse_document:
    call parse_ws
    call parse_value
    call parse_ws
    mov rax, [rip + cur_off]
    cmp rax, [rip + end_off]
    jne error_set_and_exit
    jmp success

success:
    lea rsi, [rip + ok_msg]
    mov rdi, 1
    mov rdx, 3
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

error_set_and_exit:
    mov rax, [rip + cur_off]
    jmp set_error_off

set_error_off:
    mov [rip + error_off], rax
    jmp print_error

print_error:
    lea rsi, [rip + error_msg]
    mov rdi, 1
    mov rdx, 9
    mov rax, 1
    syscall
    mov rcx, [rip + error_off]
    call print_decimal
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_decimal:
    sub rsp, 32
    mov r10, rsp
    xor r8, r8
    mov r11, 10
.conv_loop:
    xor rdx, rdx
    mov rax, rcx
    div r11
    add dl, '0'
    mov byte ptr [r10 + 19 - r8], dl
