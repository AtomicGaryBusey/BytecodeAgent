.section .bss
buf: .space 65537

.section .text
.globl _start
_start:
    lea rbp, [rip + buf]
    xor r12, r12
    mov r13, rbp
    mov r14, 65537
read_loop:
    test r14, r14
    jz read_done
    mov rax, 0
    xor rdi, rdi
    mov rsi, r13
    mov rdx, r14
    syscall
    test rax, rax
    jz read_done
    js err
    add r12, rax
    add r13, rax
    sub r14, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit
    mov byte ptr [rbp + r12], 10
    xor r13, r13
line_loop:
    mov r14, r13
find_nl:
    cmp r14, r12
    jge final_line
    cmp byte ptr [rbp + r14], 10
    je line_found
    inc r14
    jmp find_nl
line_found:
    call reverse_and_write
    mov r13, r14
    inc r13
    jmp line_loop
final_line:
    cmp r13, r12
    jge exit
    call reverse_and_write
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
err:
    mov rax, 60
    mov rdi, 1
    syscall

reverse_and_write:
    mov r15, r13
    mov rcx, r14
    dec rcx
reverse_loop:
    cmp r15, rcx
    jge reversed
    movzx eax, byte ptr [rbp + r15]
    movzx edx, byte ptr [rbp + rcx]
    mov byte ptr [rbp + r15], dl
    mov byte ptr [rbp + rcx], al
    inc r15
    dec rcx
    jmp reverse_loop
reversed:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rbp + r13]
    mov rdx, r14
    sub rdx, r13
    inc rdx
    syscall
    mov r15, r13
    mov rcx, r14
    dec rcx
reverse_back_loop:
    cmp r15, rcx
    jge reversed_back
    movzx eax, byte ptr [rbp + r15]
    movzx edx, byte ptr [rbp + rcx]
    mov byte ptr [rbp + r15], dl
    mov byte ptr [rbp + rcx], al
    inc r15
    dec rcx
    jmp reverse_back_loop
reversed_back:
    ret
