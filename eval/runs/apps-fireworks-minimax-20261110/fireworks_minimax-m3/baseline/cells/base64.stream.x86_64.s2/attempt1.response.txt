.section .rodata
b64_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
input_buf: .space 49152
.balign 16
output_buf: .space 66400
.balign 16
group_buf: .space 4

.section .text
.globl _start

_start:
    xor r12, r12             # r12 = total input length

.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    test rdx, rdx
    jz .read_done
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    test r12, r12
    jz .exit_zero

    xor r13, r13             # r13 = output index
    xor r14, r14             # r14 = base64 chars in current line
    xor r15, r15             # r15 = input index i

.encode_loop:
    mov rcx, r12
    sub rcx, r15
    je .encode_done
    cmp rcx, 1
    je .one_byte
    cmp rcx, 2
    je .two_bytes
    jmp .three_bytes

.one_byte:
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + r15]
    mov edx, eax
    shr edx, 2
    and edx, 0x3F
    mov rcx, eax
