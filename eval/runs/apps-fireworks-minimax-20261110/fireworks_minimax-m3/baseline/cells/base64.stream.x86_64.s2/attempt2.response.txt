.section .rodata
b64_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
input_buf: .space 49152
.balign 16
output_buf: .space 66400

.section .text
.globl _start

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14

    lea r8, [rip + input_buf]
    lea r9, [rip + output_buf]
    lea r10, [rip + b64_table]

.read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r8
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    jz .read_done
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    test r12, r12
    jz .exit_zero

    xor r15, r15

.encode_loop:
    mov rax, r12
    sub rax, r15
    cmp rax, 3
    jb .handle_remainder

    movzx ebx, byte ptr [r8 + r15]
    movzx ecx, byte ptr [r8 + r15 + 1]
    movzx edx, byte ptr [r8 + r15 + 2]

    mov edi, ebx
    shr edi, 2
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip0
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip0:

    mov edi, ebx
    shl edi, 4
    mov esi, ecx
    shr esi, 4
    or edi, esi
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip1
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip1:

    mov edi, ecx
    shl edi, 2
    mov esi, edx
    shr esi, 6
    or edi, esi
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip2
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip2:

    mov edi, edx
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip3
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip3:

    add r15, 3
    jmp .encode_loop

.handle_remainder:
    mov rax, r12
    sub rax, r15
    cmp rax, 1
    je .one_byte
    cmp rax, 2
    je .two_bytes
    jmp .encode_done

.one_byte:
    movzx ebx, byte ptr [r8 + r15]

    mov edi, ebx
    shr edi, 2
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip1b
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip1b:

    mov edi, ebx
    shl edi, 4
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip1c
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip1c:

    mov byte ptr [r9 + r13], 0x3D
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip1d
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip1d:

    mov byte ptr [r9 + r13], 0x3D
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip1e
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip1e:

    jmp .encode_done

.two_bytes:
    movzx ebx, byte ptr [r8 + r15]
    movzx ecx, byte ptr [r8 + r15 + 1]

    mov edi, ebx
    shr edi, 2
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip2b
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip2b:

    mov edi, ebx
    shl edi, 4
    mov esi, ecx
    shr esi, 4
    or edi, esi
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip2c
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip2c:

    mov edi, ecx
    shl edi, 2
    and edi, 0x3F
    movzx edi, byte ptr [r10 + rdi]
    mov byte ptr [r9 + r13], dil
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip2d
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip2d:

    mov byte ptr [r9 + r13], 0x3D
    inc r13
    inc r14
    cmp r14, 76
    jne .Lskip2e
    mov byte ptr [r9 + r13], 0x0A
    inc r13
    xor r14, r14
.Lskip2e:

    jmp .encode_done

.encode_done:
    test r14, r14
    jz .skip_final_lf
    mov byte ptr [r9 + r13], 0x0A
    inc r13
.skip_final_lf:

    mov rax, 1
    mov rdi, 1
    mov rsi, r9
    mov rdx, r13
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

.exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
