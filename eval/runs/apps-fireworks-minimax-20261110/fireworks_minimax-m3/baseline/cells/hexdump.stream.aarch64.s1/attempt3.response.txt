.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
.align 4
input_buf: .space 65536
line_buf: .space 80

.section .text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x21, line_buf
    add x21, x21, :lo12:line_buf
    adrp x23, hex_chars
    add x23, x23, :lo12:hex_chars

    mov x19, xzr
    mov x4, x20

read_loop:
    mov x0, #0
    mov x1, x4
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cmp x2, #0
    b.le read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x4, x4, x0
    add x19, x19, x0
    b read_loop
read_done:

    mov x4, x20
    mov x5, xzr

row_loop:
    sub x6, x19, x5
    mov x7, #16
    cmp x6, x7
    csel x6, x7, x6, gt
    cmp x6, #0
    b.le rows_done

    mov x22, x21

    mov x9, x5
    mov x10, #28
1:  lsr x11, x9, x10
    and x11, x11, #0xf
    ldrb w12, [x23, x11]
    strb w12, [x22], #1
    sub x10, x10, #4
    cmp x10, #0
    b.ge 1b

    mov w12, #0x20
    strb w12, [x22], #1
    strb w12, [x22], #1

    mov x24, xzr
2:  cmp x24, x6
    b.lt 3f
    mov w12, #0x20
    strb w12, [x22], #1
    strb w12, [x22], #1
    strb w12, [x22], #1
    b 4f

3:  ldrb w11, [x4, x24]
    lsr x10, x11, #4
    ldrb w12, [x23, x10]
    strb w12, [x22], #1
    and x11, x11, #0xf
    ldrb w12, [x23, x11]
    strb w12, [x22], #1
    mov w12, #0x20
    strb w12, [x22], #1

4:  cmp x24, #7
    b.ne 5f
    mov w12, #0x20
    strb w12, [x22], #1
5:  add x24, x24, #1
    cmp x24, #16
    b.lt 2b

    mov w12, #0x20
    strb w12, [x22], #1
    mov w12, #0x7c
    strb w12, [x22], #1

    mov x24, xzr
6:  cmp x24, x6
    b.ge 7f
    ldrb w11, [x4, x24]
    cmp w11, #0x20
    b.lt 8f
    cmp w11, #0x7e
    b.gt 8f
    strb w11, [x22], #1
    b 9f
8:  mov w12, #0x2e
    strb w12, [x22], #1
9:  add x24, x24, #1
    b 6b
7:  mov w12, #0x7c
    strb w12, [x22], #1
    mov w12, #0x0a
    strb w12, [x22], #1

    sub x2, x22, x21
    mov x1, x21
    mov x0, #1
    mov x8, #64
    svc #0

    add x4, x4, x6
    add x5, x5, x6
    b row_loop

rows_done:
    mov x22, x21
    mov x9, x19
    mov x10, #28
10: lsr x11, x9, x10
    and x11, x11, #0xf
    ldrb w12, [x23, x11]
    strb w12, [x22], #1
    sub x10, x10, #4
    cmp x10, #0
    b.ge 10b

    mov w12, #0x0a
    strb w12, [x22], #1

    sub x2, x22, x21
    mov x1, x21
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
