.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
nl:
    .byte 0x0A

.section .bss
.align 16
input_buf:
    .space 49152
.align 16
output_buf:
    .space 65600

.section .text

_start:
    xor r12, r12

.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    syscall
    cmp rax, 0
    je .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    lea r13, [rip + input_buf]
    lea r14, [rip + output_buf]
    lea r10, [rip + alphabet]
    xor r15, r15

.encoding_loop:
    cmp r12, 3
    jb .handle_rem
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    movzx edx, byte ptr [r13 + 2]
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14], sil
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ecx
    shr edi, 4
    or esi, edi
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 1], sil
    mov esi, ecx
    and esi, 0xF
    shl esi, 2
    mov edi, edx
    shr edi, 6
    or esi, edi
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 2], sil
    mov esi, edx
    and esi, 0x3F
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 3], sil
    add r13, 3
    add r14, 4
    add r15, 4
    sub r12, 3
    jmp .encoding_loop

.handle_rem:
    cmp r12, 0
    je .encode_done
    cmp r12, 1
    je .rem1
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14], sil
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ecx
    shr edi, 4
    or esi, edi
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 1], sil
    mov esi, ecx
    and esi, 0xF
    shl esi, 2
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 2], sil
    mov byte ptr [r14 + 3], '='
    add r15, 4
    jmp .encode_done

.rem1:
    movzx eax, byte ptr [r13]
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14], sil
    mov esi, eax
    and esi, 3
    shl esi, 4
    movzx esi, byte ptr [r10 + rsi]
    mov byte ptr [r14 + 1], sil
    mov byte ptr [r14 + 2], '='
    mov byte ptr [r14 + 3], '='
    add r15, 4

.encode_done:
    lea r14, [rip + output_buf]
    xor r12, r12

.output_loop:
    mov rcx, r15
    sub rcx, r12
    jz .all_done
    cmp rcx, 76
    jbe .use_min
    mov rcx, 76

.use_min:
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, rcx
    syscall
    add r12, rax
    add r14, rax
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    jmp .output_loop

.all_done:
    mov rax, 60
    xor rdi, rdi
    syscall
