.equ BUFSIZE, 65536
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .rodata
hex_table:
    .byte 0x30, 0x31, 0x32, 0x33
    .byte 0x34, 0x35, 0x36, 0x37
    .byte 0x38, 0x39, 0x61, 0x62
    .byte 0x63, 0x64, 0x65, 0x66

.section .bss
.align 4
input_buf: .space BUFSIZE

.section .text
.globl _start

format_hex8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    adrp x2, hex_table
    add x2, x2, :lo12:hex_table
    mov x3, #0
.L_fh8_loop:
    mov x4, #7
    sub x4, x4, x3
    lsl x4, x4, #2
    lsr x5, x0, x4
    and x5, x5, #0xf
    ldrb w5, [x2, x5]
    strb w5, [x1, x3]
    add x3, x3, #1
    cmp x3, #8
    b.lt .L_fh8_loop
    ldp x29, x30, [sp], #16
    ret

format_hex2:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    adrp x2, hex_table
    add x2, x2, :lo12:hex_table
    lsr w3, w0, #4
    and w3, w3, #0xf
    ldrb w3, [x2, x3]
    strb w3, [x1, #0]
    and w3, w0, #0xf
    ldrb w3, [x2, x3]
    strb w3, [x1, #1]
    ldp x29, x30, [sp], #16
    ret

_start:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x19, #0

.L_read_loop:
    cmp x19, #BUFSIZE
    b.ge .L_read_done
    mov x0, #0
    add x1, x9, x19
    mov x2, #BUFSIZE
    sub x2, x2, x19
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .L_read_done
    add x19, x19, x0
    b .L_read_loop
.L_read_done:

    mov x21, x19
    mov x22, #0

    sub sp, sp, #96
    mov x23, sp

.L_rows_loop:
    cmp x21, #0
    b.le .L_rows_done

    mov x24, #16
    cmp x21, #16
    b.ge .L_chunk_ready
    mov x24, x21
.L_chunk_ready:

    mov x0, x22
    mov x1, x23
    bl format_hex8

    mov w0, #32
    strb w0, [x23, #8]
    strb w0, [x23, #9]

    mov x25, #0
.L_hex_col:
    cmp x25, #16
    b.ge .L_hex_done

    mov x26, #10
    add x26, x26, x25, lsl #1
    add x26, x26, x25
    cmp x25, #8
    b.lt .L_no_extra_off
    add x26, x26, #1
.L_no_extra_off:
    add x26, x26, x23

    cmp x25, x24
    b.ge .L_hex_empty

    ldrb w0, [x9, x25]
    mov x1, x26
    bl format_hex2
    mov w0, #32
    strb w0, [x26, #2]
    b .L_hex_cont
.L_hex_empty:
    mov w0, #32
    strb w0, [x26, #0]
    strb w0, [x26, #1]
    strb w0, [x26, #2]
.L_hex_cont:

    cmp x25, #7
    b.ne .L_skip_ext_sp
    mov w0, #32
    strb w0, [x23, #34]
.L_skip_ext_sp:

    add x25, x25, #1
    b .L_hex_col
.L_hex_done:

    mov w0, #32
    strb w0, [x23, #59]
    mov w0, #124
    strb w0, [x23, #60]

    mov x25, #0
.L_ascii_col:
    cmp x25, x24
    b.ge .L_ascii_done
    ldrb w0, [x9, x25]
    cmp w0, #32
    b.lt .L_ascii_dot
    cmp w0, #126
    b.gt .L_ascii_dot
    b .L_ascii_write
.L_ascii_dot:
    mov w0, #46
.L_ascii_write:
    add x26, x23, #61
    add x26, x26, x25
    strb w0, [x26]
    add x25, x25, #1
    b .L_ascii_col
.L_ascii_done:

    add x26, x23, #61
    add x26, x26, x24
    mov w0, #124
    strb w0, [x26]
    add x26, x26, #1
    mov w0, #10
    strb w0, [x26]

    mov x0, #1
    mov x1, x23
    add x2, x24, #63
    mov x8, #SYS_WRITE
    svc #0

    add x9, x9, x24
    add x22, x22, x24
    sub x21, x21, x24
    b .L_rows_loop

.L_rows_done:
    mov x0, x22
    mov x1, x23
    bl format_hex8
    mov w0, #10
    strb w0, [x23, #8]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
