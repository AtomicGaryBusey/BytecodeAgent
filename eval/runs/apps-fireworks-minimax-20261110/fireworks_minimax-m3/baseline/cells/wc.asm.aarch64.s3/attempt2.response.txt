.section .bss
.align 3
buf: .zero 65536
.align 3
state: .zero 8
.align 3
cnt_lines: .zero 8
.align 3
cnt_words: .zero 8
.align 3
cnt_bytes: .zero 8
.align 3
outbuf: .zero 32

.section .text
.globl _start

_start:
    adrp x19, cnt_lines
    add x19, x19, :lo12:cnt_lines
    adrp x20, cnt_words
    add x20, x20, :lo12:cnt_words
    adrp x21, cnt_bytes
    add x21, x21, :lo12:cnt_bytes
    adrp x22, state
    add x22, x22, :lo12:state

.Lread_again:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq .Lread_done
    b.lt .Lerror
    ldr x3, [x21]
    add x3, x3, x0
    str x3, [x21]
    mov x9, x1
    add x10, x9, x0
    ldrb w13, [x22]
.Lprocess_loop:
    cmp x9, x10
    b.ge .Lprocess_done
    ldrb w16, [x9]
    cmp w16, #0x20
    b.eq .Lis_ws
    cmp w16, #0x09
    b.eq .Lis_ws
    cmp w16, #0x0A
    b.eq .Lis_ws
    cmp w16, #0x0B
    b.eq .Lis_ws
    cmp w16, #0x0C
    b.eq .Lis_ws
    cmp w16, #0x0D
    b.eq .Lis_ws
.Lnon_ws:
    cbnz w13, .Lnext_byte
    mov w13, #1
    strb w13, [x22]
    b .Lnext_byte
.Lis_ws:
    cmp w16, #0x0A
    b.ne .Lupdate_word
    ldr x17, [x19]
    add x17, x17, #1
    str x17, [x19]
.Lupdate_word:
    cbz w13, .Lnext_byte
    ldr x17, [x20]
    add x17, x17, #1
    str x17, [x20]
    mov w13, #0
    strb w13, [x22]
.Lnext_byte:
    add x9, x9, #1
    b .Lprocess_loop
.Lprocess_done:
    b .Lread_again

.Lread_done:
    ldrb w13, [x22]
    cbz w13, .Lfinal
    ldr x17, [x20]
    add x17, x17, #1
    str x17, [x20]
.Lfinal:
    ldr x0, [x19]
    bl print_num
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov w12, #0x20
    strb w12, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ldr x0, [x20]
    bl print_num
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov w12, #0x20
    strb w12, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ldr x0, [x21]
    bl print_num
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov w12, #0x0A
    strb w12, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.Lerror:
    mov x0, #1
    mov x8, #94
    svc #0

print_num:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x9, x9, #31
    mov x10, #0
    mov x14, #10
    cbnz x0, .Ldiv
    mov w12, #0x30
    strb w12, [x9]
    mov x10, #1
    b .Lwrite
.Ldiv:
    udiv x11, x0, x14
    msub x12, x11, x14, x0
    add w12, w12, #0x30
    strb w12, [x9]
    sub x9, x9, #1
    add x10, x10, #1
    mov x0, x11
    cbnz x0, .Ldiv
    add x9, x9, #1
.Lwrite:
    mov x1, x9
    mov x0, #1
    mov x2, x10
    mov x8, #64
    svc #0
    ret
