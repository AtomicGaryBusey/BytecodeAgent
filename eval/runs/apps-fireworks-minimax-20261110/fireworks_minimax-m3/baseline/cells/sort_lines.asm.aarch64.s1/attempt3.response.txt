.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, line_off
    add x20, x20, :lo12:line_off
    adrp x21, line_len
    add x21, x21, :lo12:line_len

    // Read all input
    mov x9, #0
    mov x10, #32768
1:  cmp x9, x10
    b.ge 2f
    mov x0, #0
    add x1, x19, x9
    sub x2, x10, x9
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x9, x9, x0
    b 1b
2:

    // Parse lines
    mov x10, #0
    mov x11, #0
    mov x12, #0
3:  cmp x10, x9
    b.ge 4f
    ldrb w0, [x19, x10]
    cmp w0, #0x0a
    b.ne 5f
    sub w13, w10, w11
    str w13, [x21, x12, lsl #2]
    str w11, [x20, x12, lsl #2]
    add x12, x12, #1
    add x10, x10, #1
    mov x11, x10
    b 3b
5:  add x10, x10, #1
    b 3b
4:  cmp x11, x9
    b.ge 6f
    sub w13, w9, w11
    str w13, [x21, x12, lsl #2]
    str w11, [x20, x12, lsl #2]
    add x12, x12, #1
6:

    // Insertion sort
    mov x13, #1
8:  cmp x13, x12
    b.ge 7f
    ldr w15, [x20, x13, lsl #2]
    ldr w16, [x21, x13, lsl #2]
    mov x14, x13
9:  cmp x14, #0
    b.le 10f
    sub x17, x14, #1
    ldr w0, [x20, x17, lsl #2]
    ldr w1, [x21, x17, lsl #2]

    cmp w1, w16
    csel w3, w1, w16, lt
    mov x2, #0
12: add x6, x0, x2
    ldrb w4, [x19, x6]
    add x7, x15, x2
    ldrb w5, [x19, x7]
    cmp w4, w5
    b.ne 13f
    add x2, x2, #1
    cmp x2, x3
    b.lt 12b

    cmp w1, w16
    b.gt 14f
    b 10f
13: b.gt 14f
    b 10f
14: ldr w0, [x20, x17, lsl #2]
    ldr w1, [x21, x17, lsl #2]
    str w0, [x20, x14, lsl #2]
    str w1, [x21, x14, lsl #2]
    sub x14, x14, #1
    b 9b
10: str w15, [x20, x14, lsl #2]
    str w16, [x21, x14, lsl #2]
    add x13, x13, #1
    b 8b
7:

    // Print sorted lines
    mov x14, #0
15: cmp x14, x12
    b.ge 16f
    ldr w15, [x20, x14, lsl #2]
    ldr w16, [x21, x14, lsl #2]
    mov x0, #1
    add x1, x19, w15, uxtw
    mov x2, x16
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    mov x8, #64
    svc #0
    add x14, x14, #1
    b 15b
16: mov x0, #0
    mov x8, #93
    svc #0
    mov x8, #94
    svc #0

.section .rodata
.align 4
newline_byte:
    .byte 0x0a

.bss
.align 4
input_buf:
    .zero 32768
.align 4
line_off:
    .zero 4000
.align 4
line_len:
    .zero 4000
