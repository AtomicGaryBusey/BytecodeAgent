.equ CHAR_EXCL, 0x21
.equ MAX_INPUT, 65536
.equ BUF_SIZE, 4096
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 1000
.equ CMD_PLUS, 0x2B
.equ CMD_MINUS, 0x2D
.equ CMD_GT,   0x3E
.equ CMD_LT,   0x3C
.equ CMD_DOT,  0x2E
.equ CMD_COMMA,0x2C
.equ CMD_OPEN, 0x5B
.equ CMD_CLOSE,0x5D

.section .bss
.align 16
stdin_buf: .zero MAX_INPUT
bracket_match: .zero MAX_INPUT*4
tape: .zero TAPE_SIZE
output_buf: .zero BUF_SIZE
bracket_stack: .zero STACK_SIZE*4

.text
.globl _start
_start:
    xor r12, r12
.Lread_loop:
    mov rdi, 0
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    xor rax, rax
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    cmp r12, MAX_INPUT
    jb .Lread_loop
.Lread_done:
    mov r9, r12

    xor r13, r13
    lea rbx, [rip + stdin_buf]
.Lfind_excl:
    cmp r13, r9
    je .Lno_excl
    movzx eax, byte ptr [rbx + r13]
    cmp al, CHAR_EXCL
    je .Lfound_excl
    inc r13
    jmp .Lfind_excl
.Lfound_excl:
    mov r8, r13
    inc r13
    mov r10, r13
    jmp .Lafter_excl
.Lno_excl:
    mov r8, r9
    mov r10, r9
.Lafter_excl:
    mov r11, r9
    sub r11, r10
    lea r10, [rbx + r10]

    lea rcx, [rip + bracket_match]

    xor rdx, rdx
    xor r12, r12
.Lvalidate_loop:
    cmp r12, r8
    je .Lvalidate_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, CMD_OPEN
    je .Lhandle_open
    cmp al, CMD_CLOSE
    je .Lhandle_close
    jmp .Lvalidate_next
.Lhandle_open:
    mov dword ptr [rip + bracket_stack + rdx*4], r12
