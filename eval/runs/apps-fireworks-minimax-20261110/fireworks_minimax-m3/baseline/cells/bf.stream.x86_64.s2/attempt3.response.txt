.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 1000
.equ CMD_PLUS, 0x2B
.equ CMD_MINUS, 0x2D
.equ CMD_GT, 0x3E
.equ CMD_LT, 0x3C
.equ CMD_DOT, 0x2E
.equ CMD_COMMA, 0x2C
.equ CMD_OPEN, 0x5B
.equ CMD_CLOSE, 0x5D
.equ CHAR_EXCL, 0x21

.section .bss
.align 16
stdin_buf: .zero MAX_INPUT
bracket_match: .zero MAX_INPUT*4
bracket_stack: .zero STACK_SIZE*4
tape: .zero TAPE_SIZE

.text
.globl _start

_start:
    lea r9, [rip + stdin_buf]

    xor r12, r12
.Lread_loop:
    mov rdi, 0
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    xor rax, rax
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    cmp r12, MAX_INPUT
    jb .Lread_loop
.Lread_done:
    xor r8, r8
.Lscan_excl:
    cmp r8, r12
    je .Lno_excl
    movzx eax, byte ptr [r9 + r8]
    cmp al, CHAR_EXCL
    je .Lfound_excl
    inc r8
    jmp .Lscan_excl
.Lfound_excl:
    mov r11, r8
    inc r11
    jmp .Lafter_excl
.Lno_excl:
    mov r8, r12
    mov r11, r8
.Lafter_excl:
    mov rbx, r12
    sub rbx, r11
    lea r15, [r9 + r11]

    lea r10, [rip + bracket_match]
    lea rbp, [rip + bracket_stack]

    xor r12, r12
    xor rdx, rdx
.Lvalidate_loop:
    cmp r12, r8
    je .Lvalidate_end
    movzx eax, byte ptr [r9 + r12]
    cmp al, CMD_OPEN
    je .Lval_open
    cmp al, CMD_CLOSE
    je .Lval_close
    jmp .Lval_next
.Lval_open:
    cmp rdx, STACK_SIZE
    jae .Linvalid_exit
    mov dword ptr [rbp + rdx*4], r12
