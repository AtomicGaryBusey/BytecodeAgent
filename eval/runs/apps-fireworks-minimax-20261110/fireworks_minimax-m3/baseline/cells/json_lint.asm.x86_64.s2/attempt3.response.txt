.section .rodata
msg_ok: .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err_pre: .ascii "error at "
msg_err_pre_len = . - msg_err_pre
msg_newline: .ascii "\n"
msg_newline_len = . - msg_newline

.section .text
.globl _start
_start:
    sub rsp, 65568
    mov rbp, rsp
    xor r12, r12
    xor r14, r14
    xor r13, r13
.read_loop:
    mov rdi, 0
    lea rsi, [rbp + r14]
    mov rdx, 65536
    sub rdx, r14
    jz .read_done
    xor rax, rax
    syscall
    cmp rax, 0
    je .read_done
    js .read_error
    add r14, rax
    jmp .read_loop
.read_done:
    xor r12, r12
    call parse_document
    test rax, rax
    jz .handle_error
    call print_ok
    xor rdi, rdi
    mov rax, 60
    syscall
.handle_error:
    call print_error
    mov rdi, 1
    mov rax, 60
    syscall
.read_error:
    mov rdi, 1
    mov rax, 60
    syscall

parse_document:
    call skip_ws
    call parse_value
    test rax, rax
    jz .pd_fail
    call skip_ws
    mov rax, r12
    cmp rax, r14
    je .pd_ok
    jmp store_err
.pd_fail:
    xor rax, rax
    ret
.pd_ok:
    mov rax, 1
    ret

skip_ws:
.sw_loop:
    mov rax, r12
    cmp rax, r14
    jae .sw_done
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x20
    je .sw_inc
    cmp al, 0x09
    je .sw_inc
    cmp al, 0x0a
    je .sw_inc
    cmp al, 0x0d
    je .sw_inc
.sw_done:
    ret
.sw_inc:
    inc r12
    jmp .sw_loop

parse_value:
    mov rax, r12
    cmp rax, r14
    jae .pv_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    je parse_number
    cmp al, '1'
    je parse_number
    cmp al, '2'
    je parse_number
    cmp al, '3'
    je parse_number
    cmp al, '4'
    je parse_number
    cmp al, '5'
    je parse_number
    cmp al, '6'
    je parse_number
    cmp al, '7'
    je parse_number
    cmp al, '8'
    je parse_number
    cmp al, '9'
    je parse_number
.pv_err:
    jmp store_err

parse_object:
    inc r12
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .po_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '}'
    je .po_empty
    cmp al, '"'
    jne .po_err
.po_loop:
    call parse_string
    test rax, rax
    jz .po_ret_fail
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .po_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, ':'
    jne .po_err
    inc r12
    call skip_ws
    call parse_value
    test rax, rax
    jz .po_ret_fail
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .po_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '}'
    je .po_done
    cmp al, ','
    jne .po_err
    inc r12
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .po_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '"'
    jne .po_err
    jmp .po_loop
.po_empty:
    inc r12
    mov rax, 1
    ret
.po_done:
    inc r12
    mov rax, 1
    ret
.po_err:
    jmp store_err
.po_ret_fail:
    xor rax, rax
    ret

parse_array:
    inc r12
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .pa_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, ']'
    je .pa_empty
.pa_loop:
    call parse_value
    test rax, rax
    jz .pa_ret_fail
    call skip_ws
    mov rax, r12
    cmp rax, r14
    jae .pa_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, ']'
    je .pa_done
    cmp al, ','
    jne .pa_err
    inc r12
    call skip_ws
    jmp .pa_loop
.pa_empty:
    inc r12
    mov rax, 1
    ret
.pa_done:
    inc r12
    mov rax, 1
    ret
.pa_err:
    jmp store_err
.pa_ret_fail:
    xor rax, rax
    ret

parse_string:
    inc r12
.ps_loop:
    mov rax, r12
    cmp rax, r14
    jae .ps_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '"'
    je .ps_end
    cmp al, '\\'
    je .ps_escape
    cmp al, 0x20
    jb .ps_err
    inc r12
    jmp .ps_loop
.ps_escape:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .ps_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '"'
    je .ps_esc_end
    cmp al, '\\'
    je .ps_esc_end
    cmp al, '/'
    je .ps_esc_end
    cmp al, 'b'
    je .ps_esc_end
    cmp al, 'f'
    je .ps_esc_end
    cmp al, 'n'
    je .ps_esc_end
    cmp al, 'r'
    je .ps_esc_end
    cmp al, 't'
    je .ps_esc_end
    cmp al, 'u'
    je .ps_escape_u
    jmp store_err
.ps_esc_end:
    inc r12
    jmp .ps_loop
.ps_escape_u:
    inc r12
    call parse_hex4
    test rax, rax
    jz .ps_ret_fail
    jmp .ps_loop
.ps_end:
    inc r12
    mov rax, 1
    ret
.ps_err:
    jmp store_err
.ps_ret_fail:
    xor rax, rax
    ret

parse_hex4:
    mov rcx, 4
.ph_loop:
    mov rax, r12
    cmp rax, r14
    jae .ph_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .ph_err
    cmp al, '9'
    jbe .ph_ok
    cmp al, 'a'
    jb .ph_err
    cmp al, 'f'
    jbe .ph_ok
    cmp al, 'A'
    jb .ph_err
    cmp al, 'F'
    jbe .ph_ok
.ph_err:
    jmp store_err
.ph_ok:
    inc r12
    dec rcx
    jnz .ph_loop
    mov rax, 1
    ret

parse_number:
    movzx eax, byte ptr [rbp + r12]
    cmp al, '-'
    jne .pn_no_minus
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pn_err
    movzx eax, byte ptr [rbp + r12]
    jmp .pn_check_int
.pn_no_minus:
.pn_check_int:
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    cmp al, '0'
    je .pn_int_zero
    inc r12
.pn_int_nonzero_loop:
    mov rax, r12
    cmp rax, r14
    jae .pn_after_int
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_after_int
    cmp al, '9'
    ja .pn_after_int
    inc r12
    jmp .pn_int_nonzero_loop
.pn_int_zero:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pn_after_int
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_after_int
    cmp al, '9'
    ja .pn_after_int
    jmp .pn_err
.pn_after_int:
    mov rax, r12
    cmp rax, r14
    jae .pn_no_frac
    movzx eax, byte ptr [rbp + r12]
    cmp al, '.'
    jne .pn_no_frac
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pn_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    inc r12
.pn_frac_loop:
    mov rax, r12
    cmp rax, r14
    jae .pn_frac_done
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_frac_done
    cmp al, '9'
    ja .pn_frac_done
    inc r12
    jmp .pn_frac_loop
.pn_frac_done:
.pn_no_frac:
    mov rax, r12
    cmp rax, r14
    jae .pn_ok
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'e'
    je .pn_exp
    cmp al, 'E'
    je .pn_exp
    jmp .pn_ok
.pn_exp:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pn_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '+'
    je .pn_exp_sign
    cmp al, '-'
    je .pn_exp_sign
    jmp .pn_exp_check_digit
.pn_exp_sign:
    inc r12
.pn_exp_check_digit:
    mov rax, r12
    cmp rax, r14
    jae .pn_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_err
    cmp al, '9'
    ja .pn_err
    inc r12
.pn_exp_loop:
    mov rax, r12
    cmp rax, r14
    jae .pn_ok
    movzx eax, byte ptr [rbp + r12]
    cmp al, '0'
    jb .pn_ok
    cmp al, '9'
    ja .pn_ok
    inc r12
    jmp .pn_exp_loop
.pn_ok:
    mov rax, 1
    ret
.pn_err:
    jmp store_err

parse_true:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pt_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'r'
    jne .pt_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pt_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'u'
    jne .pt_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pt_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'e'
    jne .pt_err
    inc r12
    mov rax, 1
    ret
.pt_err:
    jmp store_err

parse_false:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pf_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'a'
    jne .pf_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pf_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'l'
    jne .pf_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pf_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 's'
    jne .pf_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pf_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'e'
    jne .pf_err
    inc r12
    mov rax, 1
    ret
.pf_err:
    jmp store_err

parse_null:
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pnl_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'u'
    jne .pnl_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pnl_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'l'
    jne .pnl_err
    inc r12
    mov rax, r12
    cmp rax, r14
    jae .pnl_err
    movzx eax, byte ptr [rbp + r12]
    cmp al, 'l'
    jne .pnl_err
    inc r12
    mov rax, 1
    ret
.pnl_err:
    jmp store_err

store_err:
    mov r13, r12
    xor rax, rax
    ret

print_ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_err_pre]
    mov rdx, msg_err_pre_len
    syscall
    mov rax, r13
    call print_uint
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_newline]
    mov rdx, msg_newline_len
    syscall
    ret

print_uint:
    mov r10, rax
    xor rcx, rcx
    test r10, r10
    jnz .pu_main
    lea r8, [rbp + 65536]
    mov byte ptr [r8], '0'
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    syscall
    ret
.pu_main:
    lea r8, [rbp + 65536]
    mov r11, 10
.pu_loop:
    xor rdx, rdx
    mov rax, r10
    div r11
    add rdx, '0'
    mov byte ptr [r8 + rcx], dl
    inc rcx
    mov r10, rax
    test r10, r10
    jnz .pu_loop
    xor r9, r9
    lea rsi, [rcx - 1]
.pu_rev:
    cmp r9, rsi
    jae .pu_write
    mov dl, [r8 + r9]
    mov al, [r8 + rsi]
    mov [r8 + r9], al
    mov [r8 + rsi], dl
    inc r9
    dec rsi
    jmp .pu_rev
.pu_write:
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, rcx
    syscall
    ret
