.equ BUFSIZE, 49152

.globl _start
.text

_start:
    xor r12, r12

.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, BUFSIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, BUFSIZE
    jl .read_loop
.read_done:

    lea r11, [rip + alphabet]
    lea rbx, [rip + outbuf]
    lea r13, [rip + inbuf]

    mov r14, rbx
    mov rcx, r12
    test rcx, rcx
    jz .no_input

    xor rdx, rdx
    mov rax, rcx
    mov rcx, 3
    div rcx
    mov r15, rax
    mov rbp, rdx

    xor r9, r9

.group_loop:
    cmp r9, r15
    je .after_groups

    movzx eax, byte ptr [r13]
    movzx esi, byte ptr [r13 + 1]
    movzx edx, byte ptr [r13 + 2]

    mov r8d, eax
    shr r8d, 2
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14], al

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, esi
    shr r10d, 4
    or r8d, r10d
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 1], al

    mov r8d, esi
    and r8d, 0xF
    shl r8d, 2
    mov r10d, edx
    shr r10d, 6
    or r8d, r10d
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 2], al

    mov r8d, edx
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 3], al

    add r13, 3
    add r14, 4
    inc r9
    jmp .group_loop

.after_groups:
    cmp rbp, 0
    je .no_input
    cmp rbp, 1
    je .rem1

.rem2:
    movzx eax, byte ptr [r13]
    movzx esi, byte ptr [r13 + 1]

    mov r8d, eax
    shr r8d, 2
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14], al

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, esi
    shr r10d, 4
    or r8d, r10d
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 1], al

    mov r8d, esi
    and r8d, 0xF
    shl r8d, 2
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 2], al

    mov byte ptr [r14 + 3], 0x3D
    add r14, 4
    jmp .no_input

.rem1:
    movzx eax, byte ptr [r13]

    mov r8d, eax
    shr r8d, 2
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14], al

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov al, byte ptr [r11 + r8]
    mov byte ptr [r14 + 1], al

    mov byte ptr [r14 + 2], 0x3D
    mov byte ptr [r14 + 3], 0x3D
    add r14, 4

.no_input:
    mov r12, r14
    sub r12, rbx

    xor r13, r13

.output_loop:
    cmp r13, r12
    je .exit

    mov r15, r12
    sub r15, r13
    cmp r15, 76
    jbe .count_set
    mov r15, 76
.count_set:
    lea rsi, [rip + outbuf]
    add rsi, r13
    lea rdi, [rip + line_buf]
    mov rcx, r15
    rep movsb

    mov byte ptr [rdi], 0x0A

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r15
    inc rdx
    syscall

    add r13, r15
    jmp .output_loop

.exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.balign 16
inbuf: .space 49152
.balign 16
outbuf: .space 65540
.balign 16
line_buf: .space 78

.section .rodata
.balign 16
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
