.equ BUFSIZE, 49152

.globl _start
.text

_start:
    xor r12, r12

.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, BUFSIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, BUFSIZE
    jl .read_loop
.read_done:

    lea r11, [rip + alphabet]
    lea r13, [rip + outbuf]

    test r12, r12
    jz .flush

    xor rdx, rdx
    mov rax, r12
    mov rcx, 3
    div rcx
    mov r15, rax
    mov rbp, rdx

    lea rsi, [rip + inbuf]
    xor r9, r9

.group_loop:
    cmp r9, r15
    je .after_groups

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]

    mov r8d, eax
    shr r8d, 2
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13], r10b

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 1], r10b

    mov r8d, ecx
    and r8d, 0xF
    shl r8d, 2
    mov r10d, edx
    shr r10d, 6
    or r8d, r10d
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 2], r10b

    mov r8d, edx
    and r8d, 0x3F
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 3], r10b

    add rsi, 3
    add r13, 4
    inc r9
    jmp .group_loop

.after_groups:
    test rbp, rbp
    jz .flush
    cmp rbp, 1
    je .rem1

.rem2:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]

    mov r8d, eax
    shr r8d, 2
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13], r10b

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 1], r10b

    mov r8d, ecx
    and r8d, 0xF
    shl r8d, 2
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 2], r10b

    mov byte ptr [r13 + 3], 0x3D
    add r13, 4
    jmp .flush

.rem1:
    movzx eax, byte ptr [rsi]

    mov r8d, eax
    shr r8d, 2
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13], r10b

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10b, byte ptr [r11 + r8]
    mov byte ptr [r13 + 1], r10b

    mov byte ptr [r13 + 2], 0x3D
    mov byte ptr [r13 + 3], 0x3D
    add r13, 4

.flush:
    lea r14, [rip + outbuf]
    mov r12, r13
    sub r12, r14

    test r12, r12
    jz .exit

    xor rbx, rbx

.output_loop:
    cmp rbx, r12
    je .exit

    mov r15, r12
    sub r15, rbx
    cmp r15, 76
    jbe .count_ok
    mov r15, 76
.count_ok:

    lea rdi, [rip + line_buf]
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rcx, r15
    rep movsb
    mov byte ptr [rdi], 0x0A

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r15
    inc rdx
    syscall

    add rbx, r15
    jmp .output_loop

.exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.balign 16
inbuf: .space 49152
.balign 16
outbuf: .space 65540
.balign 16
line_buf: .space 78

.section .rodata
.balign 16
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
