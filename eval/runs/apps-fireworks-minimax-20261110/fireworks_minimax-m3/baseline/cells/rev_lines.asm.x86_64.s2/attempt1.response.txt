.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
buf: .space 65536

.section .text
_start:
    # read all input into buf
    xor r12, r12                # total bytes read
    lea r13, [rip + buf]        # buffer base address
.read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    mov rsi, r13
    add rsi, r12                # ptr = buf + total
    mov rdx, 65536
    sub rdx, r12                # remaining space
    je .read_done               # buffer full
    syscall
    test rax, rax
    jz .read_done               # EOF
    add r12, rax
    jmp .read_loop
.read_done:
    # process lines
    xor r14, r14                # current line start offset
.line_loop:
    cmp r14, r12
    je .done

    # find next newline
    mov r15, r14
    mov rcx, r12
    sub rcx, r15                # bytes left to scan
.find_nl:
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x0A
    je .found_nl
    inc r15
    dec rcx
    jnz .find_nl
    # not found, r15 == total
    jmp .process_line
.found_nl:
.process_line:
    # line is from r14 to r15-1
    mov rdx, r15
    sub rdx, r14                # line length

    # reverse line in place if length >= 2
    cmp rdx, 2
    jl .skip_reverse
    lea rdi, [r13 + r14]        # start of line
    lea rsi, [r13 + r15 - 1]    # end of line
    mov rcx, rdx
    shr rcx, 1                  # half length
.reverse_loop:
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    dec rcx
    jnz .reverse_loop
.skip_reverse:

    # write reversed line if non-empty
    test rdx, rdx
    jz .write_nl
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    add rsi, r14
    syscall                     # write(1, buf+start, len)
.write_nl:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall                     # write(1, &nl, 1)

    # advance to next line
    cmp r15, r12
    je .done
    mov r14, r15
    inc r14
    jmp .line_loop

.done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
