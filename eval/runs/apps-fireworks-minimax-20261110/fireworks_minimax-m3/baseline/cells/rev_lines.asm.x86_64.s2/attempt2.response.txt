.globl _start

.section .rodata
nl: .byte 0x0A

.section .bss
.p2align 4
buf: .space 65536

.section .text
_start:
    xor r12, r12
    lea r13, [rip + buf]
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz .read_done
    syscall
    test rax, rax
    js .do_exit_err
    jz .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    xor r14, r14
.line_loop:
    cmp r14, r12
    je .exit0
    mov r15, r14
.find_eol:
    cmp r15, r12
    je .eol_done
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x0A
    je .eol_done
    inc r15
    jmp .find_eol
.eol_done:
    mov rcx, r15
    sub rcx, r14
    cmp rcx, 2
    jl .skip_rev
    lea rdi, [r13 + r14]
    lea rsi, [r13 + r15 - 1]
    mov r9, rcx
    shr r9, 1
.rev:
    movzx eax, byte ptr [rdi]
    movzx edx, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    dec r9
    jnz .rev
.skip_rev:
    test rcx, rcx
    jz .skip_wline
    mov rax, 1
    mov rdi, 1
    lea rsi, [r13 + r14]
    mov rdx, rcx
    syscall
.skip_wline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    cmp r15, r12
    je .exit0
    inc r15
    mov r14, r15
    jmp .line_loop

.exit0:
    mov rax, 60
    xor rdi, rdi
    syscall
.do_exit_err:
    mov rax, 60
    mov rdi, 1
    syscall
