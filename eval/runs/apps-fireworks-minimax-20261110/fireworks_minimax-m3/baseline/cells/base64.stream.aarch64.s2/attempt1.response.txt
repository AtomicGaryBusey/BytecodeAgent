.section .rodata
.align 4
base64_alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4
input_buf:
.space 49152
output_buf:
.space 66500

.section .text
.globl _start

_start:
// Read all input from stdin into input_buf
adrp x19, input_buf
add x19, x19, :lo12:input_buf
mov x24, #0
0:
mov x0, #0          // fd = 0
add x1, x19, x24   // buf = input_buf + input_len
mov x2, #49152
sub x2, x2, x24    // count = max - input_len
mov x8, #63         // sys_read
svc #0
cmp x0, #0
b.le 1f            // <=0: EOF or error
add x24, x24, x0
cmp x24, #49152
b.lt 0b
1:
// Setup for encoding
mov x20, x24       // input_remaining
adrp x25, output_buf
add x25, x25, :lo12:output_buf
mov x21, x25       // output_ptr
mov x22, #0        // line_pos
adrp x23, base64_alphabet
add x23, x23, :lo12:base64_alphabet
b 2f

// Encoding loop
2:
cmp x20, #0
b.le 7f           // done
cmp x20, #3
b.ge 3f           // full group
cmp x20, #1
b.eq 4f           // one byte
// two bytes
b 5f

3: // encode_full
ldrb w4, [x19], #1
ldrb w5, [x19], #1
ldrb w6, [x19], #1
sub x20, x20, #3
lsr w7, w4, #2
and w7, w7, #0x3F
lsl w8, w4, #4
lsr w9, w5, #4
orr w8, w8, w9
and w8, w8, #0x3F
lsl w9, w5, #2
lsr w10, w6, #6
orr w9, w9, w10
and w9, w9, #0x3F
and w10, w6, #0x3F
ldrb w0, [x23, w7, uxtw]
bl output_char
ldrb w0, [x23, w8, uxtw]
bl output_char
ldrb w0, [x23, w9, uxtw]
bl output_char
ldrb w0, [x23, w10, uxtw]
bl output_char
b 2b

4: // encode_one
ldrb w4, [x19], #1
sub x20, x20, #1
lsr w7, w4, #2
and w7, w7, #0x3F
lsl w8, w4, #4
and w8, w8, #0x3F
ldrb w0, [x23, w7, uxtw]
bl output_char
ldrb w0, [x23, w8, uxtw]
bl output_char
mov w0, #61
bl output_char
mov w0, #61
bl output_char
b 7f

5: // encode_two
ldrb w4, [x19], #1
ldrb w5, [x19], #1
sub x20, x20, #2
lsr w7, w4, #2
and w7, w7, #0x3F
lsl w8, w4, #4
lsr w9, w5, #4
orr w8, w8, w9
and w8, w8, #0x3F
and w9, w5, #0x0F
lsl w9, w9, #2
and w9, w9, #0x3F
ldrb w0, [x23, w7, uxtw]
bl output_char
ldrb w0, [x23, w8, uxtw]
bl output_char
ldrb w0, [x23, w9, uxtw]
bl output_char
mov w0, #61
bl output_char
b 7f

7: // encode_done
cbnz x22, 6f
b 8f
6: // write final newline if needed
mov w0, #10
strb w0, [x21]
add x21, x21, #1
8: // write output to stdout
mov x0, #1
mov x1, x25
sub x2, x21, x25
mov x8, #64
svc #0
mov x0, #0
mov x8, #93
svc #0

output_char:
// w0 = char to output
mov x9, x21
mov x10, x22
strb w0, [x9]
add x9, x9, #1
add x10, x10, #1
cmp x10, #76
b.ne 9f
mov w0, #10
strb w0, [x9]
add x9, x9, #1
mov x10, #0
9:
mov x21, x9
mov x22, x10
ret
