//
// rev_lines: read stdin, write each line with its bytes reversed.
//
// Linux/AArch64 static assembly. Reads up to 65536 bytes of input, processes
// each newline-terminated line in order, reverses the line's bytes in place
// in a .bss buffer, and writes the reversed line followed by one '\n'.
//

.section .bss
.p2align 4
buf:    .space 65537

.section .text
.globl _start
_start:
    // --- read whole input into buf ---
    adrp x20, buf
    add  x20, x20, :lo12:buf
    mov  x19, #0                  // total bytes in buffer
    mov  x21, x20                 // next read position
1:
    mov  x0, #0                   // fd = stdin
    mov  x1, x21
    mov  x2, #65536
    sub  x2, x2, x19              // remaining capacity
    cbz  x2, 2f                   // buffer entirely full
    mov  x8, #63                  // sys_read
    svc  #0
    cmp  x0, #0
    b.le 2f                       // EOF (0) or error (<0)
    add  x19, x19, x0
    add  x21, x21, x0
    b    1b
2:
    // --- emit each line in order (x11 = line start in buffer) ---
    mov  x11, #0
3:
    cmp  x11, x19
    b.ge 9f                       // no more bytes => done
    mov  x12, x11                 // scan pointer
4:
    cmp  x12, x19
    b.ge 7f                       // reached end with no newline
    ldrb w0, [x20, x12]
    cmp  w0, #0x0A
    b.eq 5f
    add  x12, x12, #1
    b    4b
5:
    // line is bytes [x11, x12); the newline byte itself is at x12
    cmp  x11, x12
    b.eq 6f                       // empty line: nothing to reverse
    mov  x13, x11
    sub  x14, x12, #1
10:
    cmp  x13, x14
    b.ge 6f
    ldrb w15, [x20, x13]
    ldrb w16, [x20, x14]
    strb w16, [x20, x13]
    strb w15, [x20, x14]
    add  x13, x13, #1
    sub  x14, x14, #1
    b    10b
6:
    // write reversed line + the newline byte at x12 in a single write
    mov  x0, #1                   // fd = stdout
    add  x1, x20, x11
    sub  x2, x12, x11
    add  x2, x2, #1
    mov  x8, #64                  // sys_write
    svc  #0
    add  x11, x12, #1             // next line starts after the newline
    b    3b
7:
    // final, unterminated line is [x11, x19)
    cmp  x11, x19
    b.eq 9f                       // nothing left to write
    mov  x13, x11
    sub  x14, x19, #1
11:
    cmp  x13, x14
    b.ge 12f
    ldrb w15, [x20, x13]
    ldrb w16, [x20, x14]
    strb w16, [x20, x13]
    strb w15, [x20, x14]
    add  x13, x13, #1
    sub  x14, x14, #1
    b    11b
12:
    // append one '\n' just past the input, then write final line + it
    mov  w15, #0x0A
    strb w15, [x20, x19]          // buf has 65537 bytes: room for this
    mov  x0, #1
    add  x1, x20, x11
    sub  x2, x19, x11
    add  x2, x2, #1
    mov  x8, #64
    svc  #0
9:
    mov  x0, #0                   // status 0
    mov  x8, #93                  // sys_exit
    svc  #0
