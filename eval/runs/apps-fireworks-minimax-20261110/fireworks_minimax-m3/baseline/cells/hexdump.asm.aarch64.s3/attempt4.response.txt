.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 65536

_start:
    adrp x10, buf
    add x10, x10, :lo12:buf
    adrp x11, rbuf
    add x11, x11, :lo12:rbuf
    mov x9, #0

.read_loop:
    mov x13, #INPUT_MAX
    sub x13, x13, x9
    cbz x13, .read_done
    mov x2, x13
    add x1, x10, x9
    mov x0, #0
    mov w8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .read_done
    add x9, x9, x0
    b .read_loop

.read_done:
    mov x20, #0

.row_loop:
    cmp x20, x9
    b.ge .rows_done
    sub x21, x9, x20
    cmp x21, #16
    b.le .min_skip
    mov x21, #16
.min_skip:
    mov w0, w20
    mov x1, x11
    bl write_hex8
    mov w3, #0x20
    add x4, x11, #8
    mov x2, #52
.fill_spaces:
    strb w3, [x4], #1
    sub x2, x2, #1
    cbnz x2, .fill_spaces
    mov w4, #'|'
    strb w4, [x11, #60]
    mov x24, #0
.hex_loop:
    cmp x24, x21
    b.ge .hex_done
    cmp x24, #8
    b.lt .lt8
    sub w5, w24, #8
    add w5, w5, w5, lsl #1
    add w5, w5, #35
    b .pos
.lt8:
    add w5, w24, w24, lsl #1
    add w5, w5, #10
.pos:
    add x6, x11, x5
    add x26, x10, x20
    add x26, x26, x24
    ldrb w0, [x26]
    mov x1, x6
    bl write_hex2
    add x24, x24, #1
    b .hex_loop

.hex_done:
    mov x24, #0
.ascii_loop:
    cmp x24, x21
    b.ge .ascii_done
    add x26, x10, x20
    add x26, x26, x24
    ldrb w5, [x26]
    cmp w5, #0x20
    blt .dot
    cmp w5, #0x7E
    bgt .dot
    b .nodot
.dot:
    mov w5, #'.'
.nodot:
    add x6, x11, #61
    add x6, x6, x24
    strb w5, [x6]
    add x24, x24, #1
    b .ascii_loop

.ascii_done:
    add x6, x11, #61
    add x6, x6, x21
    mov w5, #'|'
    strb w5, [x6]
    add x6, x6, #1
    mov w5, #'\n'
    strb w5, [x6]
    add x2, x21, #63
    mov x1, x11
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0
    add x20, x20, #16
    b .row_loop

.rows_done:
    mov w0, w9
    mov x1, x11
    bl write_hex8
    mov w3, #'\n'
    strb w3, [x11, #8]
    mov x2, #9
    mov x1, x11
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov w8, #SYS_EXIT
    svc #0

write_hex8:
    mov w4, w0
    mov w5, #28
    mov w6, #8
.hx8_loop:
    lsr w7, w4, w5
    cmp w7, #10
    blt .hx8_d
    add w7, w7, #0x57
    b .hx8_e
.hx8_d:
    add w7, w7, #0x30
.hx8_e:
    strb w7, [x1], #1
    sub w5, w5, #4
    sub w6, w6, #1
    cbnz w6, .hx8_loop
    ret

write_hex2:
    lsr w4, w0, #4
    and w4, w4, #0xf
    cmp w4, #10
    blt .hx2_d1
    add w4, w4, #0x57
    b .hx2_e1
.hx2_d1:
    add w4, w4, #0x30
.hx2_e1:
    strb w4, [x1], #1
    and w5, w0, #0xf
    cmp w5, #10
    blt .hx2_d2
    add w5, w5, #0x57
    b .hx2_e2
.hx2_d2:
    add w5, w5, #0x30
.hx2_e2:
    strb w5, [x1], #1
    ret

.section .bss
buf: .space 65536
rbuf: .space 80
