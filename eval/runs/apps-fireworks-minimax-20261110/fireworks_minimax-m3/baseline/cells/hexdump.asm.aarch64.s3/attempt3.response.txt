.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

_start:
    adrp x11, buf
    add x11, x11, :lo12:buf
    mov x9, #0

.Lread_loop:
    mov x13, #BUF_SIZE
    sub x13, x13, x9
    mov x14, #4096
    cmp x13, x14
    csel x14, x13, x14, lo
    cbz x13, .Lread_done
    mov x0, #0
    add x1, x11, x9
    mov x2, x14
    mov w8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .Lread_after
    add x9, x9, x0
    b .Lread_loop

.Lread_after:
    b.lt .Lread_error
.Lread_done:
    adrp x12, rbuf
    add x12, x12, :lo12:rbuf

    mov x20, #0
.Lrow_loop:
    cmp x20, x9
    b.ge .Lrows_done

    sub x21, x9, x20
    cmp x21, #16
    csel x22, x21, #16, lo

    mov x1, x12
    mov w0, w20
    bl write_hex8

    mov w3, #0x20
    strb w3, [x12, #8]
    strb w3, [x12, #9]

    add x23, x12, #10
    mov w2, #49
.Lhex_fill:
    strb w3, [x23], #1
    sub w2, w2, #1
    cbnz w2, .Lhex_fill

    mov x24, #0
.Lhex_write:
    cmp x24, x22
    b.ge .Lhex_after

    cmp x24, #8
    b.lt .Lj_lt8
    sub w5, w24, #8
    add w5, w5, w5, lsl #1
    add w5, w5, #35
    b .Lj_pos
.Lj_lt8:
    add w5, w24, w24, lsl #1
    add w5, w5, #10
.Lj_pos:
    add x1, x12, x5

    add x26, x11, x20
    add x26, x26, x24
    ldrb w4, [x26]
    mov w0, w4
    bl write_hex2

    add x24, x24, #1
    b .Lhex_write
.Lhex_after:

    mov w3, #0x20
    strb w3, [x12, #59]
    mov w4, #'|'
    strb w4, [x12, #60]

    mov x24, #0
.Lascii_loop:
    cmp x24, x22
    b.ge .Lascii_done

    add x26, x11, x20
    add x26, x26, x24
    ldrb w5, [x26]
    cmp w5, #0x20
    blt .Lascii_dot
    cmp w5, #0x7E
    bgt .Lascii_dot
    add x23, x12, #61
    add x23, x23, x24
    strb w5, [x23]
    b .Lascii_inc
.Lascii_dot:
    add x23, x12, #61
    add x23, x23, x24
    mov w6, #'.'
    strb w6, [x23]
.Lascii_inc:
    add x24, x24, #1
    b .Lascii_loop
.Lascii_done:

    add x23, x12, #61
    add x23, x23, x22
    mov w4, #'|'
    strb w4, [x23]
    add x23, x23, #1
    mov w6, #'\n'
    strb w6, [x23]

    add x2, x22, #63
    mov x1, x12
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0

    add x20, x20, #16
    b .Lrow_loop

.Lrows_done:
    mov x1, x12
    mov w0, w9
    bl write_hex8
    mov w3, #'\n'
    strb w3, [x12, #8]
    mov x2, #9
    mov x1, x12
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov w8, #SYS_EXIT
    svc #0

.Lread_error:
    mov x0, #1
    mov w8, #SYS_EXIT
    svc #0

write_hex8:
    mov w4, w0
    mov w5, #8
    mov w6, #28
.Lhx8_loop:
    lsr w7, w4, w6
    cmp w7, #10
    blt .Lhx8_d
    add w7, w7, #0x57
    b .Lhx8_e
.Lhx8_d:
    add w7, w7, #0x30
.Lhx8_e:
    strb w7, [x1], #1
    sub w6, w6, #4
    sub w5, w5, #1
    cbnz w5, .Lhx8_loop
    ret

write_hex2:
    lsr w4, w0, #4
    and w4, w4, #0xf
    cmp w4, #10
    blt .Lhx2_d1
    add w4, w4, #0x57
    b .Lhx2_e1
.Lhx2_d1:
    add w4, w4, #0x30
.Lhx2_e1:
    strb w4, [x1], #1

    and w5, w0, #0xf
    cmp w5, #10
    blt .Lhx2_d2
    add w5, w5, #0x57
    b .Lhx2_e2
.Lhx2_d2:
    add w5, w5, #0x30
.Lhx2_e2:
    strb w5, [x1], #1
    ret

.section .bss
buf: .space 65536
rbuf: .space 80
