.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // Read all input into buf
    adrp x12, buf
    add x12, x12, :lo12:buf
    mov x9, 0          // total bytes read

read_loop:
    mov x13, #0x1000
    lsl x13, x13, #4  // x13 = 65536 (buffer capacity)
    sub x13, x13, x9  // remaining bytes
    mov x14, #4096
    cmp x13, x14
    csel x14, x13, x14, lo // x14 = min(remaining, 4096)
    cbz x13, read_done
    mov x0, #0
    add x1, x12, x9
    mov x2, x14
    mov w8, #SYS_READ
    svc #0
    cbnz x0, 1f
    b read_done
1:
    tbnz x0, #63, read_error
    add x9, x9, x0
    b read_loop

read_error:
    mov x0, #1
    mov w8, #SYS_EXIT
    svc #0

read_done:
    // Allocate row buffer on stack (80 bytes)
    sub sp, sp, #80
    mov x28, sp         // base pointer for row
    mov x20, 0          // row offset i

row_loop:
    cmp x20, x9
    b.ge rows_done
    sub x21, x9, x20    // remaining bytes
    mov x22, #16
    cmp x21, x22
    csel x22, x21, x22, lo // n = min(remaining, 16)

    // 1. Write offset (8 hex digits)
    mov x1, x28
    mov w0, w20
    bl write_hex8

    // 2. Two spaces at offsets 8-9
    mov w3, #0x20
    strb w3, [x28, #8]
    strb w3, [x28, #9]

    // 3. Fill hex column indices 10-58 with spaces (49 bytes)
    add x23, x28, #10
    mov w2, #49
fill_hex_spaces:
    strb w3, [x23], #1
    sub w2, w2, #1
    cbnz w2, fill_hex_spaces

    // 4. Write hex digits for actual bytes
    mov x24, 0          // j
hex_loop:
    cmp x24, x22
    b.ge hex_done
    // Compute hex1 position
    cmp x24, #7
    ble low_hex
    // high group (j >= 8)
    mov w5, #10
    mov w6, w24
    lsl w6, w6, #1
    add w5, w5, w6
    add w5, w5, w24
    add w5, w5, #1
    b compute_hex
low_hex:
    mov w5, #10
    mov w6, w24
    lsl w6, w6, #1
    add w5, w5, w6
    add w5, w5, w24
compute_hex:
    add x1, x28, x5     // pointer to hex1
    // Load byte from buf[i+j]
    add x26, x12, x20
    add x26, x26, x24
    ldrb w4, [x26]
    mov w0, w4
    bl write_hex2
    add x24, x24, #1
    b hex_loop
hex_done:
    // 5. Space before first pipe and the pipe itself
    strb w3, [x28, #59]
    mov w4, #'|'
    strb w4, [x28, #60]

    // 6. ASCII characters (only n of them)
    add x23, x28, #61
    mov x24, 0
ascii_loop:
    cmp x24, x22
    b.ge ascii_done
    add x26, x12, x20
    add x26, x26, x24
    ldrb w5, [x26]
    cmp w5, #0x20
    blt ascii_dot
    cmp w5, #0x7E
    bgt ascii_dot
    strb w5, [x23], #1
    b ascii_inc
ascii_dot:
    mov w6, #'.'
    strb w6, [x23], #1
ascii_inc:
    add x24, x24, #1
    b ascii_loop
ascii_done:
    // 7. Pipe and newline
    mov w4, #'|'
    strb w4, [x23], #1
    mov w6, #'\n'
    strb w6, [x23]

    // Write the row: length = n + 63
    add x2, x22, #63
    mov x1, x28
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0

    add x20, x20, #16
    b row_loop

rows_done:
    // Print total length: 8 hex digits + newline
    mov x1, x28
    mov w0, w9
    bl write_hex8
    mov w3, #0x0A
    strb w3, [x28, #8]
    mov x1, x28
    mov x2, #9
    mov x0, #1
    mov w8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov w8, #SYS_EXIT
    svc #0

// Subroutines ----------------------------------------------------------

write_hex8:
    // w0 = 32-bit value, x1 = buffer (8 bytes). Writes digits and advances x1.
    mov w2, #8
    mov w3, #28
.Lwrite_hex8_loop:
    lsr w4, w0, w3
    and w4, w4, #0xf
    mov w0, w4
    bl conv_nibble
    strb w0, [x1], #1
    sub w3, w3, #4
    sub w2, w2, #1
    cbnz w2, .Lwrite_hex8_loop
    ret

write_hex2:
    // w0 = byte, x1 = buffer (2 bytes). Writes two hex digits and advances x1.
    mov w4, w0
    lsr w2, w4, #4
    and w2, w2, #0xf
    bl conv_nibble
    strb w0, [x1], #1
    and w3, w4, #0xf
    mov w0, w3
    bl conv_nibble
    strb w0, [x1], #1
    ret

conv_nibble:
    // w0 = nibble (0-15) => returns ASCII digit
    cmp w0, #10
    blt .Lconv_digit
    sub w0, w0, #10
    add w0, w0, #0x61   // 'a'
    ret
.Lconv_digit:
    add w0, w0, #0x30   // '0'
    ret

.section .bss
buf: .space 65536
