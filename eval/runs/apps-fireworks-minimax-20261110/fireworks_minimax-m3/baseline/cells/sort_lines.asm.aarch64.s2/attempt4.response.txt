.globl _start
.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000
.equ LINE_SIZE, 8

.text

_start:
    // Set up buffer base
    adrp x10, input_buf
    add x10, x10, :lo12:input_buf
    // x11 = total bytes read
    mov x11, #0
    // x12 = buffer capacity (BUF_SIZE)
    mov x12, #BUF_SIZE

1:  // read_loop
    sub x2, x12, x11
    cbz x2, 2f
    add x1, x10, x11
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x11, x11, x0
    b 1b
2:  // read_done

    // Parse lines into offset/length pairs
    adrp x15, lines
    add x15, x15, :lo12:lines
    // x9 = line_start offset (initially 0)
    mov x9, #0
    // x13 = current position (pos)
    mov x13, #0
    // x14 = line count
    mov x14, #0

3:  // parse_loop
    cmp x13, x11
    b.ge 4f
    add x0, x10, x13
    ldrb w1, [x0]
    cmp w1, #0x0A
    b.ne 5f
    // newline found: compute length = pos - line_start
    sub x2, x13, x9
    // store entry: offset at base+line_idx*8, length at +4
    lsl x3, x14, #3
    add x4, x15, x3
    str w9, [x4]
    str w2, [x4, #4]
    add x14, x14, #1
    // new line start = pos + 1 (skip newline)
    add x9, x13, #1
    // advance pos past newline
    add x13, x13, #1
    b 3b
5:  // not newline
    add x13, x13, #1
    b 3b
4:  // parse_done

    // Final line if input does not end with newline
    cmp x9, x11
    b.ge 6f
    sub x2, x11, x9
    lsl x3, x14, #3
    add x4, x15, x3
    str w9, [x4]
    str w2, [x4, #4]
    add x14, x14, #1
6:  // sort_start

    // Insertion sort
    cmp x14, #1
    b.le 7f
    mov x16, #1          // i = 1
8:  // outer_loop
    cmp x16, x14
    b.ge 7f
    mov x17, x16         // j = i
9:  // inner_loop
    cmp x17, #0
    b.le 10f
    // compute addresses of lines[j-1] and lines[j]
    sub x18, x17, #1
    lsl x19, x18, #3
    add x20, x15, x19    // addr of desc[j-1]
    lsl x21, x17, #3
    add x22, x15, x21    // addr of desc[j]

    // load offset and length for both lines
    ldr w23, [x20]       // offset1
    ldr w24, [x20, #4]   // length1
    ldr w25, [x22]       // offset2
    ldr w26, [x22, #4]   // length2

    // determine min_len = (len1 < len2) ? len1 : len2
    cmp w24, w26
    csel w27, w24, w26, lo

    // get pointers to the line starts in the buffer
    add x28, x10, x23    // p1 = buffer + offset1
    add x29, x10, x25    // p2 = buffer + offset2

    mov w30, #0          // k = 0
11: // cmp_loop
    cmp w30, w27
    b.ge 12f
    ldrb w31, [x28, w30]
    ldrb w32, [x29, w30]
    cmp w31, w32
    b.ne 13f
    add w30, w30, #1
    b 11b
13: // cmp_diff
    sub w33, w31, w32
    cmp w33, #0
    b.gt 14f
    b 10f
12: // cmp_len_diff
    sub w33, w24, w26
    cmp w33, #0
    b.gt 14f
    b 10f
14: // do_swap
    str w25, [x20]
    str w26, [x20, #4]
    str w23, [x22]
    str w24, [x22, #4]
    sub x17, x17, #1
    b 9b
10: // inner_break
    add x16, x16, #1
    b 8b
7:  // sort_done

    // Prepare newline character pointer
    adrp x6, newline
    add x6, x6, :lo12:newline
    mov x16, #0          // i = 0
15: // output_loop
    cmp x16, x14
    b.ge 16f
    lsl x3, x16, #3
    add x4, x15, x3
    ldr w23, [x4]        // offset
    ldr w24, [x4, #4]    // length
    // write line
    mov x0, #1
    add x1, x10, x23     // buffer pointer
    mov x2, x24
    mov x8, #64
    svc #0
    // write newline
    mov x0, #1
    mov x1, x6
    mov x2, #1
    mov x8, #64
    svc #0
    add x16, x16, #1
    b 15b
16: // exit
    mov x0, #0
    mov x8, #93
    svc #0
17: b 17b

.section .rodata
newline: .byte 0x0A

.section .bss
.p2align 3
input_buf: .space BUF_SIZE
.p2align 3
lines: .space MAX_LINES * LINE_SIZE
