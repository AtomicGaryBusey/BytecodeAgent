.globl _start
.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000
.equ DESC_SIZE, 16

.text

_start:
    // Load buffer descriptors
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0          // total bytes read
    mov x21, #BUF_SIZE

1:  // read_loop
    sub x2, x21, x20
    cbz x2, 2f
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:  // end_read

    // Parse lines into descriptors
    adrp x9, line_descs
    add x9, x9, :lo12:line_descs
    mov x21, #0          // pos
    mov x22, #0          // N (line count)
    mov x23, #0          // line_start offset

3:  // parse_loop
    cmp x21, x20
    b.ge 4f
    add x0, x19, x21
    ldrb w1, [x0]
    cmp w1, #0x0A
    b.ne 5f
    sub x2, x21, x23     // length
    lsl x3, x22, #4
    add x4, x9, x3       // descriptor address
    add x5, x19, x23     // pointer to line start
    str x5, [x4]
    str w2, [x4, #8]
    add x22, x22, #1
    add x23, x21, #1     // new line start
    add x21, x21, #1
    b 3b
5:  // not newline
    add x21, x21, #1
    b 3b
4:  // end_parse

    // Final line if any
    cmp x23, x20
    b.ge 6f
    sub x2, x20, x23
    lsl x3, x22, #4
    add x4, x9, x3
    add x5, x19, x23
    str x5, [x4]
    str w2, [x4, #8]
    add x22, x22, #1
6:  // done parsing

    // Insertion sort
    cmp x22, #1
    b.le 7f
    mov x23, #1          // i = 1
8:  // outer_loop
    cmp x23, x22
    b.ge 7f
    mov x24, x23         // j = i
9:  // inner_loop
    cmp x24, #0
    b.le 10f
    sub x25, x24, #1     // j-1
    lsl x13, x25, #4
    add x15, x9, x13     // addr of desc[j-1]
    lsl x14, x24, #4
    add x17, x9, x14     // addr of desc[j]

    ldr x2, [x15]        // ptr1
    ldr w3, [x15, #8]    // len1
    cmp w3, #0
    b.eq 13f             // if len1 == 0, swap it with len2 (which is >=1 or 0)
    ldr x4, [x17]        // ptr2
    ldr w5, [x17, #8]    // len2

    // Determine common length to compare
    cmp w3, w5
    csel w6, w3, w5, lo  // min_len
    mov w7, #0
11: // cmp_loop
    cmp w7, w6
    b.ge 12f
    ldrb w8, [x2, x7]
    ldrb w9, [x4, x7]
    cmp w8, w9
    b.ne 13f
    add w7, w7, #1
    b 11b
13: // diff or one exhausted
    sub w0, w8, w9
    cmp w0, #0
    b.gt 14f
12: // equal_prefix (only if we got here via b.ge from cmp_loop)
    sub w0, w3, w5
    cmp w0, #0
    b.gt 14f
    b 15f
14: // do_swap
    ldr x18, [x15]
    ldr w19, [x15, #8]
    ldr x20, [x17]
    ldr w21, [x17, #8]
    str x20, [x15]
    str w21, [x15, #8]
    str x18, [x17]
    str w19, [x17, #8]
    sub x24, x24, #1
    b 9b
15: // break
10: // end inner
    add x23, x23, #1
    b 8b
7:  // done sorting

    // Output lines
    adrp x25, newline
    add x25, x25, :lo12:newline
    mov x23, #0
16: // output_loop
    cmp x23, x22
    b.ge 17f
    lsl x3, x23, #4
    add x4, x9, x3
    ldr x1, [x4]         // pointer
    ldr w2, [x4, #8]     // length
    mov x0, #1
    mov x8, #64
    svc #0               // write line
    mov x1, x25
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0               // write newline
    add x23, x23, #1
    b 16b
17: // done output

    // Exit
    mov x0, #0
    mov x8, #93
    svc #0
18: b 18b

.section .rodata
newline: .byte 0x0A

.section .bss
.p2align 4
input_buf: .space BUF_SIZE
.p2align 4
line_descs: .space MAX_LINES * DESC_SIZE
