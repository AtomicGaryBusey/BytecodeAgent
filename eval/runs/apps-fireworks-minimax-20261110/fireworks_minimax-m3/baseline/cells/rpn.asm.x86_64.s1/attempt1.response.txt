.section .rodata
error_msg: .ascii "error\n"
error_len = . - error_msg
min_msg: .ascii "-9223372036854775808\n"
min_len = . - min_msg

.section .bss
input_buf: .space 65536
value_stack: .space 1024 * 8

.section .text
.globl _start
.intel_syntax noprefix

_start:
    # r12 = total bytes read
    xor r12d, r12d
    lea rbx, [rip + input_buf]

read_loop:
    mov rax, 0               # syscall read
    mov rdi, 0
    lea rsi, [rbx + r12]     # buffer offset
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 65536
    jl read_loop
read_done:
    xor r13d, r13d           # line start index = 0

line_loop:
    cmp r13, r12
    jge done_program
    mov r14d, r13d           # line end index
find_newline:
    cmp r14, r12
    jge end_line
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x0A
    je end_line
    inc r14
    jmp find_newline

end_line:
    # line from r13 (inclusive) to r14 (exclusive)
    # initialise per‑line state
    xor r15d, r15d           # value stack index = 0
    xor r8d, r8d             # error flag = 0
    mov rcx, r13             # current position in line

line_token_loop:
    cmp rcx, r14
    jge line_end

    # skip spaces and tabs
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x20
    je skip_ws
    cmp al, 0x09
    je skip_ws
    jmp parse_token_start

skip_ws:
    inc rcx
    jmp line_token_loop

parse_token_start:
    mov rdx, rcx             # token end pointer

scan_token:
    cmp rdx, r14
    jge token_end
    movzx eax, byte ptr [rbx + rdx]
    cmp al, 0x20
    je token_end
    cmp al, 0x09
    je token_end
    inc rdx
    jmp scan_token

token_end:
    # token lies in [rcx, rdx)
    mov rax, rdx
    sub rax, rcx             # token length
    cmp rax, 1
    jne treat_as_number

    # length 1: possible operator
    movzx eax, byte ptr [rbx + rcx]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    # not an operator -> it must be a single digit
    jmp treat_as_number

treat_as_number:
    xor r9d, r9d             # sign flag (0 = positive, 1 = negative)
    mov al, byte ptr [rbx + rcx]
    cmp al, '-'
    jne num_skip_sign
    inc rcx
    mov r9d, 1
num_skip_sign:
    cmp rcx, rdx
    jge error_set           # no digits after '-'
    mov al, byte ptr [rbx + rcx]
    cmp al, '0'
    jb error_set
    cmp al, '9'
    ja error_set
    xor r10, r10             # accumulator
    xor r11d, r11d           # digit count

parse_digits:
    mov al, byte ptr [rbx + rcx]
    cmp al, '0'
    jb error_set
    cmp al, '9'
    ja error_set
    imul r10, 10
    movzx eax, al
    sub al, '0'
    add r10, rax
    inc rcx
    inc r11d
    cmp rcx, rdx
    jl parse_digits

    cmp r11d, 18
    ja error_set

    test r9d, r9d
    jz push_number
    neg r10
push_number:
    mov [value_stack + r15*8], r10
    inc r15
    jmp token_done

op_add:
    cmp r15, 2
    jb error_set
    mov rax, [value_stack + (r15-1)*8]   # b
    mov rcx, [value_stack + (r15-2)*8]   # a
    add rcx, rax
    mov [value_stack + (r15-2)*8], rcx
    dec r15
    jmp token_done

op_sub:
    cmp r15, 2
    jb error_set
    mov rax, [value_stack + (r15-1)*8]   # b
    mov rcx, [value_stack + (r15-2)*8]   # a
    sub rcx, rax
    mov [value_stack + (r15-2)*8], rcx
    dec r15
    jmp token_done

op_mul:
    cmp r15, 2
    jb error_set
    mov rax, [value_stack + (r15-1)*8]   # b
    mov rcx, [value_stack + (r15-2)*8]   # a
    imul rcx, rax
    mov [value_stack + (r15-2)*8], rcx
    dec r15
    jmp token_done

op_div:
    push rdx
    mov rax, [value_stack + (r15-1)*8]   # b
    mov rcx, [value_stack + (r15-2)*8]   # a
    test rax, rax
    jz error_set
    cmp rax, 0xFFFFFFFFFFFFFFFF
    jne do_div
    cmp rcx, 0x8000000000000000
    je error_set
do_div:
    mov rax, rcx          # dividend a
    cqo
    idiv rcx              # divisor b
    mov [value_stack + (r15-2)*8], rax   # quotient
    dec r15
    pop rdx
    jmp token_done

op_mod:
    push rdx
    mov rax, [value_stack + (r15-1)*8]   # b
    mov rcx, [value_stack + (r15-2)*8]   # a
    test rax, rax
    jz error_set
    cmp rax, 0xFFFFFFFFFFFFFFFF
    jne do_mod
    cmp rcx, 0x8000000000000000
    je error_set
do_mod:
    mov rax, rcx          # dividend a
    cqo
    idiv rcx              # divisor b
    mov [value_stack + (r15-2)*8], rdx   # remainder
    dec r15
    pop rdx
    jmp token_done

error_set:
    mov r8d, 1
    mov rcx, r14
    jmp token_done

token_done:
    mov rcx, rdx
    jmp line_token_loop

line_end:
    cmp r8d, 0
    jne out_error
    cmp r15, 1
    jne out_error
    # value to print is the only item on stack
    mov rax, [value_stack]
    call print_number
    jmp line_continue

out_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall

line_continue:
    mov r13, r14
    inc r13
    jmp line_loop

done_program:
    mov rax, 60
    xor rdi, rdi
    syscall

# -------------------------------------------------------------
# print_number: convert signed 64‑bit integer in rax to decimal,
# output followed by a newline.
# -------------------------------------------------------------
print_number:
    push rbx
    cmp rax, 0x8000000000000000
    jne pn_not_min
    # special minimum value
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_msg]
    mov rdx, min_len
    syscall
    pop rbx
    ret

pn_not_min:
    sub rsp, 32
    mov rbx, rsp           # buffer start
    xor r9d, r9d           # sign flag (0 positive, 1 negative)
    test rax, rax
    jns pn_mag
    inc r9d
    neg rax
pn_mag:
    lea rcx, [rbx + 31]    # point to end of buffer
    xor r10d, r10d         # digit count
    cmp rax, 0
    jne pn_digit_loop
    # value is zero
    mov byte ptr [rcx], '0'
    inc r10d
    jmp pn_conv_done
pn_digit_loop:
    xor rdx, rdx
    mov r11, 10
    div r11
    add dl, '0'
    dec rcx
    mov [rcx], dl
    inc r10d
    test rax, rax
    jnz pn_digit_loop
pn_conv_done:
    mov rsi, rbx
    test r9d, r9d
    jz pn_copy
    mov byte ptr [rsi], '-'
    inc rsi
pn_copy:
    mov rdx, r10
    test rdx, rdx
    jz pn_newline
pn_copy_loop:
    mov al, [rcx]
    mov [rsi], al
    inc rcx
    inc rsi
    dec rdx
    jnz pn_copy_loop
pn_newline:
    mov byte ptr [rsi], 0x0A
    inc rsi
    mov rdx, rsi
    sub rdx, rbx
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall
    add rsp, 32
    pop rbx
    ret

# -------------------------------------------------------------
# print_error: output "error\n".
# -------------------------------------------------------------
print_error:
    push rbx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall
    pop rbx
    ret
