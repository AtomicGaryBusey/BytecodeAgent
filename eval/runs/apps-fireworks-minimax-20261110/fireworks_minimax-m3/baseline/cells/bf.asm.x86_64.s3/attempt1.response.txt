.equ BUF_SIZE, 65536
.equ TAPE_BYTES, 30000
.equ MATCH_BYTES, 131072
.equ STACK_BYTES, 2000

.section .bss
input_buffer:  .space BUF_SIZE
tape:          .space TAPE_BYTES
match_table:   .space MATCH_BYTES
bracket_stack: .space STACK_BYTES
bracket_stack_end:

.section .text
.globl _start

_start:
    lea r12, [rip + input_buffer]
    mov r13, BUF_SIZE
.L_read_loop:
    test r13, r13
    jz .L_read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz .L_read_done
    js .L_read_done
    add r12, rax
    sub r13, rax
    jmp .L_read_loop
.L_read_done:
    mov r14, r12

    lea rbx, [rip + input_buffer]
    mov rcx, rbx
.L_sep_loop:
    cmp rcx, r14
    jae .L_no_sep
    cmp byte ptr [rcx], 0x21
    je .L_have_sep
    inc rcx
    jmp .L_sep_loop
.L_no_sep:
    mov rcx, r14
.L_have_sep:
    mov r15, rcx

    mov r10, rbx
    lea r11, [rip + bracket_stack]
.L_scan_loop:
    cmp r10, r15
    jae .L_scan_done
    movzx eax, byte ptr [r10]
    cmp al, 0x5b
    je .L_scan_open
    cmp al, 0x5d
    je .L_scan_close
    jmp .L_scan_next
.L_scan_open:
    lea rdx, [rip + bracket_stack_end]
    cmp r11, rdx
    jae .L_unbalanced
    mov rsi, r10
    sub rsi, rbx
    mov word ptr [r11], si
    add r11, 2
    jmp .L_scan_next
.L_scan_close:
    lea rdx, [rip + bracket_stack]
    cmp r11, rdx
    je .L_unbalanced
    sub r11, 2
    movzx rcx, word ptr [r11]
    lea rdi, [rip + match_table]
    mov rsi, r10
    sub rsi, rbx
    mov word ptr [rdi + rcx*2], si
    mov word ptr [rdi + rsi*2], cx
.L_scan_next:
    inc r10
    jmp .L_scan_loop
.L_scan_done:
    lea rdx, [rip + bracket_stack]
    cmp r11, rdx
    jne .L_unbalanced

    mov r12, rbx
    lea r13, [rip + tape]
    mov rsi, r15
    inc rsi
    mov rbp, 5000000
.L_run_loop:
    cmp r12, r15
    jae .L_exit_ok
    dec rbp
    js .L_exit_ok
    movzx eax, byte ptr [r12]
    cmp al, 0x3e
    je .L_cmd_gt
    cmp al, 0x3c
    je .L_cmd_lt
    cmp al, 0x2b
    je .L_cmd_plus
    cmp al, 0x2d
    je .L_cmd_minus
    cmp al, 0x2e
    je .L_cmd_dot
    cmp al, 0x2c
    je .L_cmd_comma
    cmp al, 0x5b
    je .L_cmd_lbrack
    cmp al, 0x5d
    je .L_cmd_rbrack
    jmp .L_run_next
.L_cmd_gt:
    inc r13
    jmp .L_run_next
.L_cmd_lt:
    dec r13
    jmp .L_run_next
.L_cmd_plus:
    inc byte ptr [r13]
    jmp .L_run_next
.L_cmd_minus:
    dec byte ptr [r13]
    jmp .L_run_next
.L_cmd_dot:
    push rsi
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, 1
    syscall
    pop rsi
    jmp .L_run_next
.L_cmd_comma:
    cmp rsi, r14
    jae .L_comma_eof
    movzx eax, byte ptr [rsi]
    inc rsi
    jmp .L_comma_store
.L_comma_eof:
    xor eax, eax
.L_comma_store:
    mov byte ptr [r13], al
    jmp .L_run_next
.L_cmd_lbrack:
    cmp byte ptr [r13], 0
    jne .L_run_next
    mov rcx, r12
    sub rcx, rbx
    lea rdi, [rip + match_table]
    movzx rcx, word ptr [rdi + rcx*2]
    lea r12, [rbx + rcx + 1]
    jmp .L_run_loop
.L_cmd_rbrack:
    cmp byte ptr [r13], 0
    je .L_run_next
    mov rcx, r12
    sub rcx, rbx
    lea rdi, [rip + match_table]
    movzx rcx, word ptr [rdi + rcx*2]
    lea r12, [rbx + rcx + 1]
    jmp .L_run_loop
.L_run_next:
    inc r12
    jmp .L_run_loop
.L_exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
.L_unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
