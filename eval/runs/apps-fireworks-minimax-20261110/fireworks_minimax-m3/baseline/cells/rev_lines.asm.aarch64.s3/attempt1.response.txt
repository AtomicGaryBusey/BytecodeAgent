.globl _start
.text

_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf
    add  x9,  x19, #0x10000

    mov  x21, x19
read_loop:
    mov  x0,  #0
    mov  x1,  x21
    sub  x2,  x9, x21
    cbz  x2,  read_done
    mov  x8,  #63
    svc  #0
    cmp  x0,  #0
    b.lt read_err
    b.eq read_done
    add  x21, x21, x0
    b    read_loop
read_err:
    mov  x0,  #1
    mov  x8,  #93
    svc  #0
read_done:
    mov  x20, x21

    cmp  x20, x19
    b.eq do_exit

    mov  x24, x19
    mov  x23, x19
    b    scan_loop

scan_loop:
    cmp  x23, x20
    b.ge finalize
    ldrb w0, [x23]
    cmp  w0, #0x0A
    b.eq finalize
    add  x23, x23, #1
    b    scan_loop

finalize:
    sub  x25, x23, x24

    mov  x26, #0
rev_loop:
    cmp  x26, x25
    b.ge rev_done
    sub  x1,  x23, x26
    sub  x1,  x1, #1
    ldrb w2, [x1]
    strb w2, [x22, x26]
    add  x26, x26, #1
    b    rev_loop
rev_done:
    cmp  x25, #0
    b.eq skip_write
    mov  x0,  #1
    mov  x1,  x22
    mov  x2,  x25
    mov  x8,  #64
    svc  #0
skip_write:
    mov  x0,  #1
    adrp x1,  newline
    add  x1,  x1, :lo12:newline
    mov  x2,  #1
    mov  x8,  #64
    svc  #0

    cmp  x23, x20
    b.ge do_exit

    add  x24, x23, #1
    cmp  x24, x20
    b.ge do_exit
    mov  x23, x24
    b    scan_loop

do_exit:
    mov  x0, #0
    mov  x8, #93
    svc  #0

.section .rodata
newline: .byte 0x0A

.section .bss
.p2align 4
inbuf:  .space 0x10000
outbuf: .space 0x10000
