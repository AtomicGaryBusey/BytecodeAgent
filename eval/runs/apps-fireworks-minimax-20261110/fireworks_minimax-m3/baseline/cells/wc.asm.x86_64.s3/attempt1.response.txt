.globl _start
.section .bss
output_buf: .space 32
buffer:     .space 65536
.section .rodata
space:   .byte 0x20
newline: .byte 0x0a
.section .text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor rbx, rbx
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz .read_done
    add rbx, rax
    cmp rbx, 65536
    jae .read_done
    jmp .read_loop
.read_done:
    mov r12, rbx
    lea rsi, [rip + buffer]
    mov rcx, rbx
.scan_loop:
    test rcx, rcx
    jz .scan_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x0a
    jne .not_nl
    inc r13
.not_nl:
    cmp al, 0x20
    je .is_ws
    cmp al, 0x09
    je .is_ws
    cmp al, 0x0a
    je .is_ws
    cmp al, 0x0b
    je .is_ws
    cmp al, 0x0c
    je .is_ws
    cmp al, 0x0d
    je .is_ws
    test r15, r15
    jnz .next_byte
    inc r14
    mov r15, 1
    jmp .next_byte
.is_ws:
    xor r15, r15
.next_byte:
    inc rsi
    dec rcx
    jnz .scan_loop
.scan_done:
    mov rax, r13
    call print_num
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space]
    mov rdx, 1
    syscall
    mov rax, r14
    call print_num
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space]
    mov rdx, 1
    syscall
    mov rax, r12
    call print_num
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
print_num:
    lea rsi, [rip + output_buf]
    add rsi, 31
    xor ecx, ecx
    mov r10, 10
.pn_loop:
    xor edx, edx
    div r10
    add dl, 0x30
    mov byte ptr [rsi], dl
    dec rsi
    inc ecx
    test rax, rax
    jnz .pn_loop
    inc rsi
    mov rax, 1
    mov rdi, 1
    mov rdx, rcx
    syscall
    ret
