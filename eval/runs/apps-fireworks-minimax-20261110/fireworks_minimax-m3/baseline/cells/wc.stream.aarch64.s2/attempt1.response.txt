.section .rodata
.align 2
space_char: .byte 0x20
newline_char: .byte 0x0A

.section .bss
.align 4
buffer: .space 8192

.section .text
.globl _start

.set BUFSIZE, 8192

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x24, buffer
    add x24, x24, :lo12:buffer

read_loop:
    mov x0, #0
    mov x1, x24
    mov x2, #BUFSIZE
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    mov x23, x0
    add x21, x21, x23
    mov x25, #0

process_loop:
    cmp x25, x23
    b.ge process_done
    ldrb w26, [x24, x25]
    cmp w26, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    cmp w26, #0x20
    b.eq is_ws
    cmp w26, #0x09
    b.eq is_ws
    cmp w26, #0x0A
    b.eq is_ws
    cmp w26, #0x0B
    b.eq is_ws
    cmp w26, #0x0C
    b.eq is_ws
    cmp w26, #0x0D
    b.eq is_ws
    mov x22, #1
    b 2f
is_ws:
    cbz x22, 2f
    add x20, x20, #1
    mov x22, #0
2:
    add x25, x25, #1
    b process_loop

process_done:
    b read_loop

read_done:
    cbz x22, 3f
    add x20, x20, #1
3:
    mov x0, x19
    bl print_number
    adrp x1, space_char
    add x1, x1, :lo12:space_char
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, x20
    bl print_number
    adrp x1, space_char
    add x1, x1, :lo12:space_char
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, x21
    bl print_number
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_number:
    sub sp, sp, #32
    mov x9, x0
    mov x27, sp
    add x27, x27, #31
    mov x28, #0
    cbz x9, 1f
    mov x12, #10
2:
    udiv x10, x9, x12
    msub x11, x10, x12, x9
    add w11, w11, #48
    strb w11, [x27]
    sub x27, x27, #1
    add x28, x28, #1
    mov x9, x10
    cbnz x9, 2b
    add x1, x27, #1
    mov x2, x28
    mov x0, #1
    mov x8, #64
    svc #0
    add sp, sp, #32
    ret
1:
    mov w11, #48
    strb w11, [x27]
    mov x1, x27
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    add sp, sp, #32
    ret
