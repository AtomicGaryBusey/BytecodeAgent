// hexdump: hex + ASCII dump of stdin (like `hexdump -C`), with every row printed
// and a final offset line.  AArch64 Linux syscall ABI, no libc.

        .globl  _start

        .section .bss
        .align  4
buf_in:
        .space  65536
        .align  4
line_buf:
        .space  128

        .section .text

// ---------------------------------------------------------------------------
// hex_nibble: low nibble of w0 -> lowercase hex digit in w0
// ---------------------------------------------------------------------------
hex_nibble:
        and     w0, w0, #0xf
        cmp     w0, #9
        b.gt    hex_n_let
        add     w0, w0, #'0'
        ret
hex_n_let:
        sub     w0, w0, #10
        add     w0, w0, #'a'
        ret

// ---------------------------------------------------------------------------
// hex8_to_buf: byte w1 -> "xx " at buf x0, advance x0
// ---------------------------------------------------------------------------
hex8_to_buf:
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp
        mov     x9, x0
        mov     w10, w1

        lsr     w0, w10, #4
        bl      hex_nibble
        strb    w0, [x9], #1

        and     w0, w10, #0xf
        bl      hex_nibble
        strb    w0, [x9], #1

        mov     w0, #' '
        strb    w0, [x9], #1

        mov     x0, x9
        ldp     x29, x30, [sp], #16
        ret

// ---------------------------------------------------------------------------
// hex32_to_buf: value w1 -> 8 hex digits at buf x0, advance x0
// ---------------------------------------------------------------------------
hex32_to_buf:
        stp     x29, x30, [sp, #-16]!
        mov     x29, sp
        mov     x9, x0
        mov     w10, w1
        mov     x11, #8
hex32_lp:
        sub     x11, x11, #1
        lsr     w0, w10, x11, lsl #2
