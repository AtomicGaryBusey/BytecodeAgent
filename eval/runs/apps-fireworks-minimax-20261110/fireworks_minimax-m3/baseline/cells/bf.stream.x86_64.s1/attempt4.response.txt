.section .bss
.align 4
input_buf: .space 65536
.align 4
match_tbl: .space 262144
.align 4
bracket_stack: .space 4096
.align 4
tape: .space 30000
.align 4
output_byte: .space 8

.section .text
.globl _start

_start:
  lea r12, [rip + input_buf]
  xor r15, r15
.read_loop:
  cmp r15, 65536
  je .read_done
  xor rdi, rdi
  lea rsi, [r12 + r15]
  mov rdx, 65536
  sub rdx, r15
  xor rax, rax
  syscall
  cmp rax, 0
  jle .read_done
  add r15, rax
  jmp .read_loop
.read_done:

  xor rbx, rbx
  mov rcx, -1
.find_bang_loop:
  cmp rbx, r15
  je .find_bang_done
  cmp byte ptr [r12 + rbx], 0x21
  jne .find_bang_next
  mov rcx, rbx
  jmp .find_bang_done
.find_bang_next:
  inc rbx
  jmp .find_bang_loop
.find_bang_done:

  lea r13, [rip + tape]
  lea r14, [rip + match_tbl]
  cmp rcx, -1
  je .no_bang
  mov r9, rcx
  lea r10, [rcx + 1]
  mov r11, r15
  jmp .have_parts
.no_bang:
  mov r9, r15
  mov r10, r15
  mov r11, r15
.have_parts:

  xor rbx, rbx
  xor rdx, rdx
.val_loop:
  cmp rbx, r9
  je .val_done
  movzx eax, byte ptr [r12 + rbx]
  cmp al, '['
  je .val_open
  cmp al, ']'
  je .val_close
  jmp .val_next
.val_open:
  lea rsi, [rip + bracket_stack]
  mov dword ptr [rsi + rdx*4], ebx
  inc rdx
.val_next:
  inc rbx
  jmp .val_loop
.val_close:
  test rdx, rdx
  jz .invalid_exit
  dec rdx
  lea rsi, [rip + bracket_stack]
  mov ecx, dword ptr [rsi + rdx*4]
  mov dword ptr [r14 + rcx*4], ebx
  mov dword ptr [r14 + rbx*4], ecx
  jmp .val_next
.val_done:
  test rdx, rdx
  jnz .invalid_exit

.run_bf:
  xor rbx, rbx
  mov r15, 0
  mov r8, r10
.run_loop:
  cmp rbx, r9
  je .run_done
  movzx eax, byte ptr [r12 + rbx]
  cmp al, '>'
  je .cmd_gt
  cmp al, '<'
  je .cmd_lt
  cmp al, '+'
  je .cmd_plus
  cmp al, '-'
  je .cmd_minus
  cmp al, '.'
  je .cmd_dot
  cmp al, ','
  je .cmd_comma
  cmp al, '['
  je .cmd_open
  cmp al, ']'
  je .cmd_close
.cmd_advance:
  inc rbx
  jmp .run_loop
.cmd_gt:
  inc r15
  jmp .cmd_advance
.cmd_lt:
  dec r15
  jmp .cmd_advance
.cmd_plus:
  inc byte ptr [r13 + r15]
  jmp .cmd_advance
.cmd_minus:
  dec byte ptr [r13 + r15]
  jmp .cmd_advance
.cmd_dot:
  movzx eax, byte ptr [r13 + r15]
  mov byte ptr [rip + output_byte], al
  mov rdi, 1
  lea rsi, [rip + output_byte]
  mov rdx, 1
  mov rax, 1
  syscall
  jmp .cmd_advance
.cmd_comma:
  cmp r8, r11
  jae .cmd_comma_zero
  movzx eax, byte ptr [r12 + r8]
  mov byte ptr [r13 + r15], al
  inc r8
  jmp .cmd_advance
.cmd_comma_zero:
  mov byte ptr [r13 + r15], 0
  jmp .cmd_advance
.cmd_open:
  movzx eax, byte ptr [r13 + r15]
  test eax, eax
  jnz .cmd_advance
  mov eax, dword ptr [r14 + rbx*4]
  lea rbx, [rax + 1]
  jmp .run_loop
.cmd_close:
  movzx eax, byte ptr [r13 + r15]
  test eax, eax
  jz .cmd_advance
  mov eax, dword ptr [r14 + rbx*4]
  lea rbx, [rax + 1]
  jmp .run_loop
.run_done:
  xor rdi, rdi
  mov rax, 60
  syscall

.invalid_exit:
  mov rdi, 1
  mov rax, 60
  syscall
