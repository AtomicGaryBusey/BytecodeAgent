.globl _start
.text

_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    syscall
    mov r12, rax
    mov r13, 0
    lea r14, [rip + input_buffer]
    call skip_ws
    cmp r13, r12
    jae error_exit
    call parse_value
    call skip_ws
    cmp r13, r12
    je ok_exit

error_exit:
    lea rsi, [rip + err_msg]
    mov rdx, 9
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, r13
    call print_num
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall
    mov rdi, 1
    mov rax, 231
    syscall

ok_exit:
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    mov rax, 1
    mov rdi, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

print_num:
    sub rsp, 24
    lea r10, [rsp + 23]
    mov r11, 10
.pn_loop:
    xor rdx, rdx
    div r11
    add dl, '0'
    dec r10
    mov [r10], dl
    test rax, rax
    jnz .pn_loop
    mov rsi, r10
    lea rdx, [rsp + 23]
    sub rdx, r10
    mov rax, 1
    mov rdi, 1
    syscall
    add rsp, 24
    ret

skip_ws:
.sw_loop:
    cmp r13, r12
    jae .sw_done
    movzx eax, byte ptr [r14 + r13]
    cmp al, 0x20
    je .sw_advance
    cmp al, 0x09
    je .sw_advance
    cmp al, 0x0a
    je .sw_advance
    cmp al, 0x0d
    je .sw_advance
    jmp .sw_done
.sw_advance:
    inc r13
    jmp .sw_loop
.sw_done:
    ret

parse_value:
    cmp r13, r12
    jae .pv_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, '{'
    je .pv_object
    cmp al, '['
    je .pv_array
    cmp al, '"'
    je .pv_string
    cmp al, 't'
    je .pv_true
    cmp al, 'f'
    je .pv_false
    cmp al, 'n'
    je .pv_null
    cmp al, '-'
    je .pv_number
    cmp al, '0'
    je .pv_zero
    cmp al, '1'
    jb .pv_err
    cmp al, '9'
    ja .pv_err
.pv_zero:
.pv_number:
    jmp parse_number
.pv_object:
    jmp parse_object
.pv_array:
    jmp parse_array
.pv_string:
    jmp parse_string
.pv_true:
    jmp parse_true
.pv_false:
    jmp parse_false
.pv_null:
    jmp parse_null
.pv_err:
    jmp error_exit

parse_object:
    inc r13
    call skip_ws
    cmp r13, r12
    jae .po_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, '}'
    je .po_end
.po_member:
    call parse_string
    call skip_ws
    cmp r13, r12
    jae .po_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, ':'
    jne .po_err
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r12
    jae .po_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, ','
    je .po_comma
    cmp al, '}'
    je .po_end
    jmp .po_err
.po_comma:
    inc r13
    call skip_ws
    jmp .po_member
.po_end:
    inc r13
    ret
.po_err:
    jmp error_exit

parse_array:
    inc r13
    call skip_ws
    cmp r13, r12
    jae .pa_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, ']'
    je .pa_end
.pa_value:
    call parse_value
.pa_after:
    call skip_ws
    cmp r13, r12
    jae .pa_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, ','
    je .pa_comma
    cmp al, ']'
    je .pa_end
    jmp .pa_err
.pa_comma:
    inc r13
    call skip_ws
    jmp .pa_value
.pa_end:
    inc r13
    ret
.pa_err:
    jmp error_exit

parse_string:
    cmp r13, r12
    jae .ps_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, '"'
    jne .ps_err
    inc r13
.ps_loop:
    cmp r13, r12
    jae .ps_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, '"'
    je .ps_end
    cmp al, 0x5c
    je .ps_escape
    cmp al, 0x20
    jb .ps_err
    inc r13
    jmp .ps_loop
.ps_escape:
    inc r13
    cmp r13, r12
    jae .ps_err
    movzx eax, byte ptr [r14 + r13]
    cmp al, '"'
    je .ps_esc1
    cmp al, 0x5c
    je .ps_esc1
    cmp al, '/'
    je .ps_esc1
    cmp al, 'b'
    je .ps_esc1
    cmp al, 'f'
    je .ps_esc1
    cmp al, 'n'
    je .ps_esc1
    cmp al, 'r'
    je .ps_esc1
    cmp al, 't'
    je .ps_esc1
    cmp al, 'u'
    je .ps_
