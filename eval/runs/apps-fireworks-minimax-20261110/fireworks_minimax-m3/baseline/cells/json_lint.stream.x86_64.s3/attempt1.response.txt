.globl _start
.text

_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    syscall
    mov r12, rax
    mov r13, 0
    lea r14, [rip + input_buffer]
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r12
    je .ok
    mov rdi, r13
    call print_error
.ok:
    call print_ok

skip_ws:
.loop:
    cmp r13, r12
    jae .done
    movzx eax, byte ptr [r14 + r13]
    cmp al, 0x20
    je .ws
    cmp al, 0x09
    je .ws
    cmp al, 0x0a
    je .ws
    cmp al, 0x0d
    je .ws
    jmp .done
.ws:
    inc r13
    jmp .loop
.done:
    ret

parse_value:
    push r12
    push r13
    push r14
    push rbx
    cmp r13, r12
    jae .fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, '{'
    je .object
    cmp al, '['
    je .array
    cmp al, '"'
    je .string
    cmp al, 't'
    je .true
    cmp al, 'f'
    je .false
    cmp al, 'n'
    je .null
    cmp al, '-'
    je .number
    cmp al, '0'
    je .zero
    cmp al, '1'
    jb .fail
    cmp al, '9'
    ja .fail
    jmp .number
.object:
    call parse_object
    jmp .done
.array:
    call parse_array
    jmp .done
.string:
    call parse_string
    jmp .done
.true:
    call parse_true
    jmp .done
.false:
    call parse_false
    jmp .done
.null:
    call parse_null
    jmp .done
.zero:
.number:
    call parse_number
.done:
