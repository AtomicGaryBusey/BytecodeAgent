.text
.globl _start

_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    je done_read
    syscall
    test rax, rax
    jz done_read
    js read_error
    add r12, rax
    jmp read_loop

read_error:
    mov rax, 60
    mov rdi, 1
    syscall

done_read:
    xor r13, r13
    xor rbx, rbx
    xor r15b, r15b
    xor rcx, rcx

process_loop:
    cmp rcx, r12
    jae done_process
    movzx eax, byte ptr [rip + buffer + rcx]
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jbe is_ws

is_word:
    test r15b, r15b
    jnz next_byte
    mov r15b, 1
    jmp next_byte

is_ws:
    test r15b, r15b
    jz check_newline
    xor r15b, r15b
    inc rbx
check_newline:
    cmp al, 0x0A
    jne next_byte
    inc r13

next_byte:
    inc rcx
    jmp process_loop

done_process:
    test r15b, r15b
    jz after_count
    inc rbx
after_count:
    sub rsp, 80
    mov r14, rsp

    mov rax, r13
    call itoa_lr
    mov byte ptr [r14], 0x20
    inc r14

    mov rax, rbx
    call itoa_lr
    mov byte ptr [r14], 0x20
    inc r14

    mov rax, r12
    call itoa_lr
    mov byte ptr [r14], 0x0A
    inc r14

    mov rdx, r14
    sub rdx, rsp
    mov rsi, rsp
    mov rax, 1
    mov rdi, 1
    syscall

    add rsp, 80

    mov rax, 60
    xor rdi, rdi
    syscall

# Convert non-negative rax to decimal ASCII digits at *r14 (forward).
# Advances r14 past the last digit. Preserves the caller's rbx.
itoa_lr:
    push rbx
    test rax, rax
    jnz nz_case
    mov byte ptr [r14], 0x30
    inc r14
    pop rbx
    ret
nz_case:
    mov rbx, 10
    mov r8, r14
digits_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    mov byte ptr [r14], dl
    inc r14
    test rax, rax
    jnz digits_loop
    push r14
    dec r14
reverse_loop:
    cmp r8, r14
    jae reverse_done
    mov al, byte ptr [r8]
    mov bl, byte ptr [r14]
    mov byte ptr [r8], bl
    mov byte ptr [r14], al
    inc r8
    dec r14
    jmp reverse_loop
reverse_done:
    pop r14
    pop rbx
    ret

.section .bss
buffer: .space 65536
