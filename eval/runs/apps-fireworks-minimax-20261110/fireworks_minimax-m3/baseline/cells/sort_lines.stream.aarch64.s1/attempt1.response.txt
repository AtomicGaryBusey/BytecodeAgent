    add x4, x4, x0
    adr x5, input_buf
    add x5, x5, x2
    cmp w1, w3

    b 1b

2:  sub w0, w1, w3
    sxtw x0, w0
    b 4f

3:  sub w0, w7, w8
    sxtw x0, w0

4:  ret
