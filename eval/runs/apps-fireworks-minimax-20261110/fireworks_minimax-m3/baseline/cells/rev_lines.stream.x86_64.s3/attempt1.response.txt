.globl _start
_start:
    # Set up buffer pointer and read input
    lea r12, [rip + buffer]
    xor r13, r13                # total bytes read
    jmp read_check
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    mov rsi, r12
    add rsi, r13                # buf + total
    mov rdx, 65536
    sub rdx, r13                # remaining
    syscall
    test rax, rax
    jz read_done
    js read_error
    add r13, rax
read_check:
    cmp r13, 65536
    jb read_loop
read_done:
    # Scan input and process each line
    xor r14, r14                # start index of current line
    xor r15, r15                # current index
    jmp scan_check
scan_loop:
    cmp byte ptr [r12 + r15], 0x0A
    je line_found
    inc r15
scan_check:
    cmp r15, r13
    jb scan_loop
    jmp after_scan
line_found:
    # Length = end - start
    mov r9, r15
    sub r9, r14
    cmp r9, 1
    jbe 1f                      # skip reverse if length 0 or 1
    mov rcx, r14                # left index
    mov rdx, r15
    dec rdx                     # right inclusive index
2:  cmp rcx, rdx
    jae 1f
    mov al, [r12 + rcx]
    mov bl, [r12 + rdx]
    mov [r12 + rcx], bl
    mov [r12 + rdx], al
    inc rcx
    dec rdx
    jmp 2b
1:  # Write reversed line
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [r12 + r14]
    mov rdx, r9
    syscall
    # Write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    # Advance to next line
    mov r14, r15
    inc r14
    inc r15
    jmp scan_check
after_scan:
    # Process final line if input didn't end with newline
    cmp r14, r13
    jae exit_prog
    mov r9, r13
    sub r9, r14
    cmp r9, 1
    jbe 1f
    mov rcx, r14
    mov rdx, r13
    dec rdx
3:  cmp rcx, rdx
    jae 1f
    mov al, [r12 + rcx]
    mov bl, [r12 + rdx]
    mov [r12 + rcx], bl
    mov [r12 + rdx], al
    inc rcx
    dec rdx
    jmp 3b
1:  mov rax, 1
    mov rdi, 1
    lea rsi, [r12 + r14]
    mov rdx, r9
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
exit_prog:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
read_error:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
newline: .byte 0x0A

.section .bss
.align 16
buffer: .zero 65536
