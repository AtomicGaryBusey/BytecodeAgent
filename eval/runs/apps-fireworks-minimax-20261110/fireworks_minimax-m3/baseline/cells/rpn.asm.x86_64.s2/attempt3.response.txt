.section .rodata
error_msg: .ascii "error\n"
.equ error_len, . - error_msg

.section .bss
.p2align 4
buffer: .space 65536
.p2align 4
rpn_stack: .space 8192
.p2align 4
out_buf: .space 32

.section .text
.globl _start
_start:
    lea rbp, [rip + rpn_stack]
    xor r12, r12
    lea r15, [rip + buffer]
    xor r13, r13

.read_loop:
    mov rdi, 0
    lea rsi, [r15 + r13]
    mov rdx, 65536
    sub rdx, r13
    mov rax, 0
    syscall
    test rax, rax
    jz .Lread_done
    add r13, rax
    cmp r13, 65536
    je .Lread_done
    jmp .read_loop

.Lread_done:
    xor r14, r14
    cmp r14, r13
    jge .Lexit

.line_loop:
    cmp r14, r13
    jge .Lexit
    mov r9, r14

    lea rdi, [r15 + r14]
    mov rcx, r13
    sub rcx, r14
    mov al, 0x0A
    repne scasb
    jnz .Lno_newline

    mov rax, rdi
    sub rax, r15
    dec rax
    mov r8, rax
    dec r8
    lea r14, [rax + 1]
    jmp .Lprocess_line

.Lno_newline:
    mov r8, r13
    dec r8
    mov r14, r13

.Lprocess_line:
    xor r12, r12
    xor r11, r11
    mov r10, r9

.Ltoken_loop:
.Lskip_sep:
    cmp r10, r8
    jg .Lend_tokens
    movzx eax, byte ptr [r15 + r10]
    cmp al, 0x20
    je .Lsep_inc
    cmp al, 0x09
    je .Lsep_inc
    jmp .Ltoken_start
.Lsep_inc:
    inc r10
    jmp .Lskip_sep

.Ltoken_start:
    mov rcx, r10
.Lscan_token:
    cmp r10, r8
    jg .Ltoken_done
    movzx eax, byte ptr [r15 + r10]
    cmp al, 0x20
    je .Ltoken_done
    cmp al, 0x09
    je .Ltoken_done
    inc r10
    jmp .Lscan_token

.Ltoken_done:
    mov rdx, r10
    sub rdx, rcx
    cmp rdx, 1
    jne .Ltry_int
    movzx eax, byte ptr [r15 + rcx]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod
    jmp .Ltry_int

.Lop_add:
    cmp r12, 16
    jl .Lerror
    lea rsi, [rbp + r12 - 16]
    mov rcx, [rsi]
    lea rdi, [rbp + r12 - 8]
    mov rbx, [rdi]
    mov rax, rcx
    add rax, rbx
    mov [rsi], rax
    sub r12, 8
    jmp .Ltoken_loop

.Lop_sub:
    cmp r12, 16
    jl .Lerror
    lea rsi, [rbp + r12 - 16]
    mov rcx, [rsi]
    lea rdi, [rbp + r12 - 8]
    mov rbx, [rdi]
    mov rax, rcx
    sub rax, rbx
    mov [rsi], rax
    sub r12, 8
    jmp .Ltoken_loop

.Lop_mul:
    cmp r12, 16
    jl .Lerror
    lea rsi, [rbp + r12 - 16]
    mov rcx, [rsi]
    lea rdi, [rbp + r12 - 8]
    mov rbx, [rdi]
    mov rax, rcx
    imul rax, rbx
    mov [rsi], rax
    sub r12, 8
    jmp .Ltoken_loop

.Lop_div:
    cmp r12, 16
    jl .Lerror
    lea rsi, [rbp + r12 - 16]
    mov rcx, [rsi]
    lea rdi, [rbp + r12 - 8]
    mov rbx, [rdi]
    test rbx, rbx
    jz .Lerror
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne .Ldo_div
    cmp rbx, -1
    je .Lerror
.Ldo_div:
    mov rax, rcx
    mov rdx, 0
    test rax, rax
    jns .Ldiv_pos
    or rdx, -1
.Ldiv_pos:
    idiv rbx
    mov [rsi], rax
    sub r12, 8
    jmp .Ltoken_loop

.Lop_mod:
    cmp r12, 16
    jl .Lerror
    lea rsi, [rbp + r12 - 16]
    mov rcx, [rsi]
    lea rdi, [rbp + r12 - 8]
    mov rbx, [rdi]
    test rbx, rbx
    jz .Lerror
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne .Ldo_mod
    cmp rbx, -1
    je .Lerror
.Ldo_mod:
    mov rax, rcx
    mov rdx, 0
    test rax, rax
    jns .Lmod_pos
    or rdx, -1
.Lmod_pos:
    idiv rbx
    mov [rsi], rdx
    sub r12, 8
    jmp .Ltoken_loop

.Ltry_int:
    mov r9, rcx
    mov r8, rdx
    cmp r8, 19
    ja .Lerror
    xor rbx, rbx
    movzx eax, byte ptr [r15 + r9]
    cmp al, '-'
    jne .Lpos_int
    mov rbx, 1
    inc r9
    dec r8
    test r8, r8
    jz .Lerror
    cmp r8, 18
    ja .Lerror
    jmp .Lstart_digits
.Lpos_int:
    cmp al, '0'
    jb .Lerror
    cmp al, '9'
    ja .Lerror
    cmp r8, 18
    ja .Lerror
    mov r10, r8
    jmp .Lparse_digits
.Lstart_digits:
    mov r10, r8

.Lparse_digits:
    xor rax, rax
    test r10, r10
    jnz .Ldigit_loop
    jmp .Lpush_int

.Ldigit_loop:
    movzx rdx, byte ptr [r15 + r9]
    cmp dl, '0'
    jb .Lerror
    cmp dl, '9'
    ja .Lerror
    sub dl, '0'
    movzx rdx, dl
    mov rcx, 10
    mul rcx
    add rax, rdx
    inc r9
    dec r10
    jnz .Ldigit_loop

.Lpush_int:
    test rbx, rbx
    jz .Lstore_int
    neg rax
.Lstore_int:
    mov [rbp + r12], rax
    add r12, 8
    jmp .Ltoken_loop

.Lerror:
    mov r11, 1
    jmp .Lend_tokens

.Lend_tokens:
    test r11, r11
    jnz .Lprint_error
    cmp r12, 8
    jne .Lprint_error
    mov rax, [rbp + r12 - 8]
    lea r8, [rip + out_buf]
    test rax, rax
    js .Lprint_neg
    lea rdi, [r8 + 31]
    mov byte ptr [rdi], 0x0A
    dec rdi
    mov rcx, 10
.Lpos_conv:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov [rdi], dl
    dec rdi
    test rax, rax
    jnz .Lpos_conv
    lea rsi, [rdi + 1]
    mov rdx, r8
    add rdx, 31
    sub rdx, rsi
    inc rdx
    mov rdi, 1
    mov rax, 1
    syscall
    jmp .Lend_line

.Lprint_neg:
    lea rdi, [r8 + 31]
    mov byte ptr [rdi], 0x0A
    dec rdi
    mov byte ptr [rdi], '-'
    dec rdi
    mov r9, rax
    neg r9
    mov rax, r9
    mov rcx, 10
.Lneg_conv:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov [rdi], dl
    dec rdi
    test rax, rax
    jnz .Lneg_conv
    lea rsi, [rdi + 1]
    mov rdx, r8
    add rdx, 31
    sub rdx, rsi
    inc rdx
    mov rdi, 1
    mov rax, 1
    syscall
    jmp .Lend_line

.Lprint_error:
    lea rsi, [rip + error_msg]
    mov rdx, 6
    mov rdi, 1
    mov rax, 1
    syscall

.Lend_line:
    jmp .line_loop

.Lexit:
    mov rax, 60
    xor rdi, rdi
    syscall
