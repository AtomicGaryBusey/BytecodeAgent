# hexdump for x86-64 Linux

.text
.globl _start

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
input_buffer: .space 65536
line_buf: .space 80

.text
_start:
    xor r12, r12
    lea r15, [rip + input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r15 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    js error_exit
    je read_done
    add r12, rax
    jmp read_loop

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

read_done:
    xor rbx, rbx

row_loop:
    cmp rbx, r12
    jae rows_done

    mov rcx, r12
    sub rcx, rbx
    cmp rcx, 16
    jle row_len_ok
    mov rcx, 16
row_len_ok:
    mov r13, rcx

    lea r14, [rip + line_buf]

    mov rax, rbx
    mov rcx, 28
    mov r9, 8
digit_offset_loop:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov al, byte ptr [rip + hex_chars + rdx]
    mov [r14], al
    inc r14
    sub rcx, 4
    dec r9
    jnz digit_offset_loop

    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14

    xor r9, r9
    mov r10, rbx
byte_loop:
    cmp r9, r13
    jb print_byte
    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    jmp after_byte
print_byte:
    movzx r8d, byte ptr [r15 + r10]
    mov rdx, r8
    shr rdx, 4
    and rdx, 0xF
    mov al, byte ptr [rip + hex_chars + rdx]
    mov [r14], al
    inc r14
    mov rdx, r8
    and rdx, 0xF
    mov al, byte ptr [rip + hex_chars + rdx]
    mov [r14], al
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    inc r10
after_byte:
    cmp r9, 7
    jne no_extra_space
    mov byte ptr [r14], ' '
    inc r14
no_extra_space:
    inc r9
    cmp r9, 16
    jb byte_loop

    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], '|'
    inc r14

    mov rcx, r13
ascii_loop:
    test rcx, rcx
    jz ascii_done
    movzx eax, byte ptr [r15 + r10]
    cmp al, 0x20
    jb not_printable
    cmp al, 0x7E
    ja not_printable
    mov [r14], al
    jmp ascii_char_done
not_printable:
    mov byte ptr [r14], '.'
ascii_char_done:
    inc r14
    inc r10
    dec rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r14], '|'
    inc r14
    mov byte ptr [r14], 0x0a
    inc r14

    lea rsi, [rip + line_buf]
    mov rdx, r14
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    add rbx, 16
    jmp row_loop

rows_done:
    lea r14, [rip + line_buf]
    mov rax, r12
    mov rcx, 28
    mov r9, 8
final_digit_loop:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov al, byte ptr [rip + hex_chars + rdx]
    mov [r14], al
    inc r14
    sub rcx, 4
    dec r9
    jnz final_digit_loop
    mov byte ptr [r14], 0x0a
    inc r14

    lea rsi, [rip + line_buf]
    mov rdx, r14
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
