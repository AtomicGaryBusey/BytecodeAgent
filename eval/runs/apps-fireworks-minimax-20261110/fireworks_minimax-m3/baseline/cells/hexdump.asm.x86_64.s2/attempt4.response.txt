.text
.globl _start

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buffer: .space 65536
line_buf: .space 80

.text
_start:
    xor r12, r12
    lea r15, [rip + input_buffer]
    lea r11, [rip + hex_chars]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r15 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    js error_exit
    je read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_loop

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

read_done:
    xor rbx, rbx

row_loop:
    cmp rbx, r12
    jae rows_done

    mov rcx, r12
    sub rcx, rbx
    cmp rcx, 16
    jbe 1f
    mov rcx, 16
1:
    mov r13, rcx

    lea r14, [rip + line_buf]

    mov rax, rbx
    mov rcx, 28
    mov r9, 8
2:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov al, byte ptr [r11 + rdx]
    mov byte ptr [r14], al
    inc r14
    sub rcx, 4
    dec r9
    jnz 2b

    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14

    xor r9, r9
    mov r10, rbx
3:
    cmp r9, r13
    jb 4f

    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    jmp 5f

4:
    movzx r8d, byte ptr [r15 + r10]
    mov rdx, r8
    shr rdx, 4
    and rdx, 0xF
    mov al, byte ptr [r11 + rdx]
    mov byte ptr [r14], al
    inc r14
    mov rdx, r8
    and rdx, 0xF
    mov al, byte ptr [r11 + rdx]
    mov byte ptr [r14], al
    inc r14
    mov byte ptr [r14], ' '
    inc r14
    inc r10

5:
    cmp r9, 7
    jne 6f
    mov byte ptr [r14], ' '
    inc r14
6:
    inc r9
    cmp r9, 16
    jb 3b

    mov byte ptr [r14], ' '
    inc r14
    mov byte ptr [r14], '|'
    inc r14

    mov r10, rbx
    mov rcx, r13
7:
    test rcx, rcx
    jz 8f
    mov al, byte ptr [r15 + r10]
    cmp al, 0x20
    jb 9f
    cmp al, 0x7E
    jbe 10f
9:
    mov al, '.'
10:
    mov byte ptr [r14], al
    inc r14
    inc r10
    dec rcx
    jmp 7b

8:
    mov byte ptr [r14], '|'
    inc r14
    mov byte ptr [r14], 0x0a
    inc r14

    lea rsi, [rip + line_buf]
    mov rdx, r14
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    add rbx, 16
    jmp row_loop

rows_done:
    lea r14, [rip + line_buf]
    mov rax, r12
    mov rcx, 28
    mov r9, 8
11:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov al, byte ptr [r11 + rdx]
    mov byte ptr [r14], al
    inc r14
    sub rcx, 4
    dec r9
    jnz 11b
    mov byte ptr [r14], 0x0a
    inc r14

    lea rsi, [rip + line_buf]
    mov rdx, r14
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
