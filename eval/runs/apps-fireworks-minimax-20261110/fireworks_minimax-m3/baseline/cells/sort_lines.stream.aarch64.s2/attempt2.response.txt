.section .rodata
.align 2
nln:
    .byte 0x0A

.equ INPUT_MAX, 32768
.equ LINES_MAX, 1000

.section .bss
.align 2
input_buf:
    .skip INPUT_MAX
.align 3
line_starts:
    .skip LINES_MAX * 8
.align 3
line_ends:
    .skip LINES_MAX * 8

.section .text
.globl _start

_start:
    mov  x0, #0
    adrp x1, input_buf
    add  x1, x1, :lo12:input_buf
    mov  x2, #INPUT_MAX
    bl   read_all
    mov  x19, x0

    mov  x20, #0
    mov  x21, #0
    mov  x22, #0
    adrp x23, input_buf
    add  x23, x23, :lo12:input_buf
    adrp x24, line_starts
    add  x24, x24, :lo12:line_starts
    adrp x25, line_ends
    add  x25, x25, :lo12:line_ends

parse_loop:
    cmp  x22, x19
    b.ge parse_done
    ldrb w26, [x23, x22]
    cmp  w26, #0x0A
    b.ne parse_next
    str  x21, [x24, x20, lsl #3]
    str  x22, [x25, x20, lsl #3]
    add  x20, x20, #1
    add  x21, x22, #1
parse_next:
    add  x22, x22, #1
    b    parse_loop

parse_done:
    cmp  x21, x19
    b.ge sort_setup
    str  x21, [x24, x20, lsl #3]
    str  x19, [x25, x20, lsl #3]
    add  x20, x20, #1

sort_setup:
    mov  x26, x20

    mov  x21, #0
sort_outer:
    cmp  x21, x26
    b.ge sort_done
    mov  x27, x21
    add  x22, x21, #1
sort_inner:
    cmp  x22, x26
    b.ge sort_swap
    ldr  x0, [x24, x27, lsl #3]
    ldr  x1, [x25, x27, lsl #3]
    ldr  x2, [x24, x22, lsl #3]
    ldr  x3, [x25, x22, lsl #3]
    bl   line_compare
    cmp  x0, #0
    b.le 1f
    mov  x27, x22
1:  add  x22, x22, #1
    b    sort_inner

sort_swap:
    ldr  x0, [x24, x21, lsl #3]
    ldr  x1, [x24, x27, lsl #3]
    str  x1, [x24, x21, lsl #3]
    str  x0, [x24, x27, lsl #3]
    ldr  x0, [x25, x21, lsl #3]
    ldr  x1, [x25, x27, lsl #3]
    str  x1, [x25, x21, lsl #3]
    str  x0, [x25, x27, lsl #3]
    add  x21, x21, #1
    b    sort_outer

sort_done:
    mov  x21, #0
output_loop:
    cmp  x21, x26
    b.ge output_done
    ldr  x0, [x24, x21, lsl #3]
    ldr  x1, [x25, x21, lsl #3]
    sub  x1, x1, x0
    add  x0, x23, x0
    mov  x2, x1
    mov  x1, x0
    mov  x0, #1
    bl   write_all
    mov  x0, #1
    adrp x1, nln
    add  x1, x1, :lo12:nln
    mov  x2, #1
    bl   write_all
    add  x21, x21, #1
    b    output_loop

output_done:
    mov  x0, #0
    mov  x8, #93
    svc  #0

read_all:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    mov  x9, x1
    mov  x10, x2
    mov  x11, #0
1:  cmp  x11, x10
    b.ge 2f
    mov  x0, #0
    add  x1, x9, x11
    sub  x2, x10, x11
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le 2f
    add  x11, x11, x0
    b    1b
2:  mov  x0, x11
    ldp  x29, x30, [sp], #16
    ret

write_all:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    mov  x9, x0
    mov  x10, x1
    mov  x11, x2
    mov  x12, #0
3:  cmp  x12, x11
    b.ge 4f
    mov  x0, x9
    add  x1, x10, x12
    sub  x2, x11, x12
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le 4f
    add  x12, x12, x0
    b    3b
4:  ldp  x29, x30, [sp], #16
    ret

line_compare:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    adrp x4, input_buf
    add  x4, x4, :lo12:input_buf
lc_loop:
    cmp  x0, x1
    b.ge lc_a_end
    cmp  x2, x3
    b.ge lc_b_end
    ldrb w5, [x4, x0]
    ldrb w6, [x4, x2]
    cmp  w5, w6
    b.lo lc_neg
    b.hi lc_pos
    add  x0, x0, #1
    add  x2, x2, #1
    b    lc_loop

lc_a_end:
    cmp  x2, x3
    b.ge lc_zero
    b    lc_neg

lc_b_end:
    mov  x0, #1
    b    lc_done

lc_neg:
    mov  x0, #-1
    b    lc_done
lc_pos:
    mov  x0, #1
lc_zero:
    mov  x0, #0
lc_done:
    ldp  x29, x30, [sp], #16
    ret
