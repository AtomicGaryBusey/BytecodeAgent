.section .rodata
.align 2
nl:
    .byte 0x0A

.equ INPUT_MAX, 32768
.equ LINES_MAX, 1000

.section .bss
.align 4
input_buf:
    .skip INPUT_MAX
.align 3
.line_starts:
    .skip LINES_MAX * 8
.align 3
.line_ends:
    .skip LINES_MAX * 8

.section .text
.globl _start

_start:

    adrp x1, input_buf
    add  x1, x1, :lo12:input_buf
    mov  x2, #INPUT_MAX

    mov  x19, x0

    mov  x21, #0
    mov  x22, #0

parse_loop:

    b.ge parse_done
    ldrb w24, [x23, x22]
    cmp  w24, #0x0A
    b.ne parse_next
    adrp x25, .line_starts
    add  x25, x25, :lo12:.line_starts
    str  x21, [x25, x20, lsl #3]
    adrp x26, .line_ends
    add  x26, x26, :lo12:.line_ends
    str  x22, [x26, x20, lsl #3]
    add  x20, x20, #1
    add  x21, x22, #1
parse_next:

    b    parse_loop
parse_done:

    b.ge parse_skip
    adrp x25, .line_starts
    add  x25, x25, :lo12:.line_starts
    str  x21, [x25, x20, lsl #3]
    adrp x26, .line_ends
    add  x26, x26, :lo12:.line_ends
    str  x19, [x26, x20, lsl #3]
    add  x20, x20, #1
parse_skip:
    mov  x24, x20

    mov  x25, #0
    adrp x29, .line_starts
    add  x29, x29, :lo12:.line_starts
    adrp x30, .line_ends
    add  x30, x30, :lo12:.line_ends

sort_outer:
    sub  x0, x24, x25
    cmp  x0, #1
    b.le sort_done
    mov  x27, x25
    add  x28, x25, #1
sort_inner:
    cmp  x28, x24
    b.ge sort_swap
    ldr  x0, [x29, x27, lsl #3]
    ldr  x1, [x30, x27, lsl #3]
    ldr  x2, [x29, x28, lsl #3]
    ldr  x3, [x30, x28, lsl #3]
    bl   line_compare
    cmp  x0, #0
    b.le 1f
    mov  x27, x28
1:
    add  x28, x28, #1
    b    sort_inner
sort_swap:
    ldr  x0, [x29, x25, lsl #3]
    ldr  x1, [x29, x27, lsl #3]
    str  x1, [x29, x25, lsl #3]
    str  x0, [x29, x27, lsl #3]
    ldr  x0, [x30, x25, lsl #3]
    ldr  x1, [x30, x27, lsl #3]
    str  x1, [x30, x25, lsl #3]
    str  x0, [x30, x27, lsl #3]
    add  x25, x25, #1
    b    sort_outer
sort_done:
    mov  x25, #0
    adrp x23, input_buf
    add  x23, x23, :lo12:input_buf
output_loop:
    cmp  x25, x24
    b.ge output_done
    ldr  x0, [x29, x25, lsl #3]
    add  x1, x23, x0
    ldr  x2, [x30, x25, lsl #3]
    sub  x2, x2, x0
    mov  x0, #1
    bl   write
    mov  x0, #1
    adrp x1, nl
    add  x1, x1, :lo12:nl
    mov  x2, #1
    bl   write
    add  x25, x25, #1
    b    output_loop
output_done:
    mov  x0, #0
    mov  x8, #93
    svc  #0

read_all:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    mov  x9, x1
    mov  x10, x2
    mov  x11, #0

    b.ge 2f
    mov  x0, #0
    add  x1, x9, x11
    sub  x2, x10, x11
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le 2f
    add  x11, x11, x0
    b    1b
2:
    mov  x0, x11
    ldp  x29, x30, [sp], #16
    ret

    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    mov  x8, #64
    svc  #0
    ldp  x29, x30, [sp], #16

    mov  x29, sp
    mov  x4, x0
    mov  x5, x2
    adrp x6, input_buf

    cmp  x4, x1
    b.ge 3f
    cmp  x5, x3
    b.ge 4f
    ldrb w7, [x6, x4]
    ldrb w8, [x6, x5]
    cmp  w7, w8
    b.lt 5f
    b.gt 6f
    add  x4, x4, #1
    add  x5, x5, #1
    b    1b
3:
    cmp  x5, x3
    b.ge 7f
    mov  x0, #-1
    b    9f
4:
    mov  x0, #1
    b    9f
5:
    mov  x0, #-1
    b    9f
6:
    mov  x0, #1
    b    9f
7:
    mov  x0, #0
9:
    ldp  x29, x30, [sp], #16
    ret
