.section .rodata
.align 1
nl_sym:
.byte 0x0A

.section .bss
.align 4
buf:
.skip 65536

.section .text
.globl _start
_start:
    mov x3, #0
    adrp x4, buf
    add x4, x4, :lo12:buf
    adrp x10, nl_sym
    add x10, x10, :lo12:nl_sym

.L_read_loop:
    mov x0, #0
    add x1, x4, x3
    mov x2, #65536
    sub x2, x2, x3
    mov x8, #63
    svc #0
    cbz x0, .L_read_done
    add x3, x3, x0
    b .L_read_loop

.L_read_done:
    cbz x3, .L_exit

    mov x5, #0
    mov x6, #0

.L_process_loop:
    ldrb w7, [x4, x5]
    cmp w7, #10
    b.ne .L_continue

    sub x9, x5, #1
    mov x0, #1
    mov x2, #1
    mov x8, #64

.L_emit_reversed:
    cmp x9, x6
    b.lt .L_emit_nl
    add x1, x4, x9
    svc #0
    sub x9, x9, #1
    b .L_emit_reversed

.L_emit_nl:
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0

    add x6, x5, #1

.L_continue:

    cmp x5, x3
    b.lt .L_process_loop

    cmp x6, x3
    b.ge .L_exit

    sub x9, x3, #1
    mov x0, #1
    mov x2, #1
    mov x8, #64

.L_emit_final_reversed:

    b.lt .L_emit_final_nl
    add x1, x4, x9
    svc #0
    sub x9, x9, #1
    b .L_emit_final_reversed

.L_emit_final_nl:
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0

.L_exit:
    mov x0, #0
    mov x8, #93
    svc #0
