.section .rodata
ok_str: .ascii "ok\n"
ok_len = . - ok_str
err_prefix: .ascii "error at "
err_prefix_len = . - err_prefix
newline: .ascii "\n"

.section .bss
.align 16
buffer: .zero 65536
.align 8
num_buf: .zero 20

.text
.globl _start
.equ BUFSIZE, 65536

_start:
    lea rbx, [rip + buffer]
    xor r12, r12
.Lread_loop:
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, BUFSIZE - r12
    mov rax, 0
    syscall
    test rax, rax
    jz .Lread_done
    add r12, rax
    cmp r12, BUFSIZE
    jae .Lread_done
    jmp .Lread_loop
.Lread_done:
    lea r14, [rbx + r12]
    mov r13, rbx
    call parse_document
    jmp exit_ok

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall

exit_err:
    mov rax, 60
    mov rdi, 1
    syscall

parse_document:
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jne report_error
    call print_ok
    jmp exit_ok

skip_ws:
    jmp .Lws_check
.Lws_loop:
    inc r13
.Lws_check:
    cmp r13, r14
    jae .Lws_done
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je .Lws_loop
    cmp al, 0x09
    je .Lws_loop
    cmp al, 0x0A
    je .Lws_loop
    cmp al, 0x0D
    je .Lws_loop
.Lws_done:
    ret

parse_value:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe parse_number
    jmp report_error

parse_object:
    movzx eax, byte ptr [r13]
    cmp al, '{'
    jne report_error
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '}'
    je .Lobj_empty
    call parse_string
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, ':'
    jne report_error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
.Lobj_loop:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '}'
    je .Lobj_end
    cmp al, ','
    jne report_error
    inc r13
    call skip_ws
    call parse_string
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, ':'
    jne report_error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    jmp .Lobj_loop
.Lobj_empty:
    inc r13
    ret
.Lobj_end:
    inc r13
    ret

parse_array:
    movzx eax, byte ptr [r13]
    cmp al, '['
    jne report_error
    inc r13
    call skip_ws
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, ']'
    je .Larr_empty
    call parse_value
    call skip_ws
.Larr_loop:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, ']'
    je .Larr_end
    cmp al, ','
    jne report_error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    jmp .Larr_loop
.Larr_empty:
    inc r13
    ret
.Larr_end:
    inc r13
    ret

parse_string:
    movzx eax, byte ptr [r13]
    cmp al, '"'
    jne report_error
    inc r13
.Lstr_loop:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '"'
    je .Lstr_end
    cmp al, '\\'
    je .Lstr_escape
    cmp al, 0x20
    jb report_error
    inc r13
    jmp .Lstr_loop
.Lstr_escape:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '"'
    je .Lstr_esc_simple
    cmp al, '\\'
    je .Lstr_esc_simple
    cmp al, '/'
    je .Lstr_esc_simple
    cmp al, 'b'
    je .Lstr_esc_simple
    cmp al, 'f'
    je .Lstr_esc_simple
    cmp al, 'n'
    je .Lstr_esc_simple
    cmp al, 'r'
    je .Lstr_esc_simple
    cmp al, 't'
    je .Lstr_esc_simple
    cmp al, 'u'
    je .Lstr_esc_u
    jmp report_error
.Lstr_esc_simple:
    inc r13
    jmp .Lstr_loop
.Lstr_esc_u:
    inc r13
    mov rcx, 4
.Lstr_esc_u_loop:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe .Lstr_esc_u_ok
    cmp al, 'a'
    jb report_error
    cmp al, 'f'
    jbe .Lstr_esc_u_ok
    cmp al, 'A'
    jb report_error
    cmp al, 'F'
    jbe .Lstr_esc_u_ok
    jmp report_error
.Lstr_esc_u_ok:
    inc r13
    dec rcx
    jnz .Lstr_esc_u_loop
    jmp .Lstr_loop
.Lstr_end:
    inc r13
    ret

parse_true:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 't'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'r'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'u'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'e'
    jne report_error
    inc r13
    ret

parse_false:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'f'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'a'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'l'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 's'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'e'
    jne report_error
    inc r13
    ret

parse_null:
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'n'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'u'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'l'
    jne report_error
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, 'l'
    jne report_error
    inc r13
    ret

parse_number:
    movzx eax, byte ptr [r13]
    cmp al, '-'
    je .Lnum_sign
    jmp .Lnum_int_start
.Lnum_sign:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
.Lnum_int_start:
    cmp al, '0'
    je .Lnum_zero
    cmp al, '1'
    jb report_error
    cmp al, '9'
    ja report_error
    inc r13
.Lnum_int_loop:
    cmp r13, r14
    jae .Lnum_frac
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .Lnum_frac
    cmp al, '9'
    ja .Lnum_frac
    inc r13
    jmp .Lnum_int_loop
.Lnum_zero:
    inc r13
    jmp .Lnum_frac
.Lnum_frac:
    cmp r13, r14
    jae .Lnum_exp
    movzx eax, byte ptr [r13]
    cmp al, '.'
    jne .Lnum_exp
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
    inc r13
.Lnum_frac_loop:
    cmp r13, r14
    jae .Lnum_exp
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .Lnum_exp
    cmp al, '9'
    ja .Lnum_exp
    inc r13
    jmp .Lnum_frac_loop
.Lnum_exp:
    cmp r13, r14
    jae .Lnum_done
    movzx eax, byte ptr [r13]
    cmp al, 'e'
    je .Lnum_e
    cmp al, 'E'
    je .Lnum_e
    jmp .Lnum_done
.Lnum_e:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
    cmp al, '+'
    je .Lnum_e_sign
    cmp al, '-'
    je .Lnum_e_sign
    jmp .Lnum_e_check
.Lnum_e_sign:
    inc r13
    cmp r13, r14
    jae report_error
    movzx eax, byte ptr [r13]
.Lnum_e_check:
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
    inc r13
.Lnum_e_loop:
    cmp r13, r14
    jae .Lnum_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb .Lnum_done
    cmp al, '9'
    ja .Lnum_done
    inc r13
    jmp .Lnum_e_loop
.Lnum_done:
    ret

report_error:
    mov rax, r13
    sub rax, rbx
    call print_error
    jmp exit_err

print_ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_str]
    mov rdx, ok_len
    syscall
    ret

print_error:
    lea rbx, [rip + num_buf + 20]
    mov rcx, 0
    test rax, rax
    jnz .Lprint_num_loop
    dec rbx
    mov byte ptr [rbx], '0'
    mov rcx, 1
    jmp .Lprint_num_done
.Lprint_num_loop:
    mov rdx, 0
    mov r8, 10
    div r8
    add dl, '0'
    dec rbx
    mov byte ptr [rbx], dl
    inc rcx
    test rax, rax
    jnz .Lprint_num_loop
.Lprint_num_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, err_prefix_len
    syscall
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, rcx
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    ret
