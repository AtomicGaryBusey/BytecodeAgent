.globl _start

.section .rodata
newline_char:
    .byte 0x0A

.section .bss
.balign 16
input_buf:
    .zero 32769
.balign 16
line_starts:
    .zero 8000
.balign 16
line_lens:
    .zero 8000

.section .text
_start:
    # Stage 1: Read all of stdin into input_buf (at most 32768 bytes).
    xor r12, r12
    lea r13, [rip + input_buf]
.read_loop:
    cmp r12, 32768
    jge .read_done
    mov rdi, 0
    mov rsi, r13
    mov rdx, 32768
    sub rdx, r12
    mov rax, 0
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    add r13, rax
    jmp .read_loop
.read_done:

    # Stage 2: Parse stdin into lines, storing (ptr, len) for each.
    # A line is bytes up to (not including) the next '\n', or trailing
    # bytes after the last newline as a final line. Empty lines allowed.
    xor rcx, rcx
    xor rbx, rbx
    xor rbp, rbp
.parse_loop:
    cmp rbp, r12
    je .parse_done_no_newline
.scan_inner:
    cmp rbp, r12
    je .parse_done_no_newline
    movzx eax, byte ptr [input_buf + rbp]
    cmp al, 0x0A
    je .found_newline
    inc rbp
    jmp .scan_inner
.found_newline:
    mov rax, rbp
    sub rax, rbx
    mov r8, rcx
    shl r8, 3
    mov [line_lens + r8], rax
    lea rax, [rip + input_buf]
    add rax, rbx
    mov [line_starts + r8], rax
    inc rcx
    cmp rcx, 1000
    je .parse_done
    inc rbp
    mov rbx, rbp
    cmp rbp, r12
    je .parse_done_no_newline
    jmp .parse_loop
.parse_done_no_newline:
    mov rax, rbp
    sub rax, rbx
    test rax, rax
    jz .parse_done
    mov r10, rax
    lea rax, [rip + input_buf]
    add rax, rbx
    mov r8, rcx
    shl r8, 3
    mov [line_starts + r8], rax
    mov [line_lens + r8], r10
    inc rcx
.parse_done:
    mov r15, rcx

    # Stage 3: Sort the lines in place, ascending byte-wise (selection sort).
    # n <= 1000; O(n^2) is at most one million comparisons, well under budget.
    test r15, r15
    jz .sort_done
    mov r13, r15
    dec r13
    xor r12, r12
.sort_outer:
    cmp r12, r13
    jge .sort_done
    mov r14, r12
    lea rbx, [r12 + 1]
.sort_inner:
    cmp rbx, r15
    jge .sort_inner_done
    mov r8, r14
    shl r8, 3
    mov rdi, [line_starts + r8]
    mov rsi, [line_lens + r8]
    mov r11, rbx
    shl r11, 3
    mov rdx, [line_starts + r11]
    mov rcx, [line_lens + r11]
    # Inline compare A=(rdi,rsi) with B=(rdx,rcx).
    # If A > B, set m=j; otherwise leave m unchanged.
    mov rax, rsi
    cmp rax, rcx
    jle .cmp_min
    mov rax, rcx
.cmp_min:
    test rax, rax
    jz .cmp_pfx
    xor r8, r8
.cmp_byte_loop:
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp al, r10b
    jne .cmp_diff
    inc r8
    cmp r8, rax
    jl .cmp_byte_loop
.cmp_pfx:
    cmp rsi, rcx
    jg .update_m
    jmp .skip_update
.cmp_diff:
    ja .update_m
    jmp .skip_update
.update_m:
    mov r14, rbx
.skip_update:
    inc rbx
    jmp .sort_inner
.sort_inner_done:
    cmp r12, r14
    je .skip_swap
    mov r8, r12
    shl r8, 3
    mov r9, r14
    shl r9, 3
    mov rax, [line_starts + r8]
    mov rcx, [line_starts + r9]
    mov [line_starts + r8], rcx
    mov [line_starts + r9], rax
    mov rax, [line_lens + r8]
    mov rcx, [line_lens + r9]
    mov [line_lens + r8], rcx
    mov [line_lens + r9], rax
.skip_swap:
    inc r12
    jmp .sort_outer
.sort_done:

    # Stage 4: Write each sorted line followed by '\n' to stdout.
    xor r12, r12
.output_outer:
    cmp r12, r15
    jge .output_done
    mov r8, r12
    shl r8, 3
    mov rsi, [line_starts + r8]
    mov rdx, [line_lens + r8]
    mov rdi, 1
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r12
    jmp .output_outer
.output_done:
    xor rdi, rdi
    mov rax, 60
    syscall
