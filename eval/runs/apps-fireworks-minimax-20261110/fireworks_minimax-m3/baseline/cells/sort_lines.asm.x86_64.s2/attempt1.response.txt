.globl _start
.section .bss
.align 16
buffer: .space 32769
lines: .space 8000

.section .rodata
newline: .byte 0x0A

.text
_start:
    sub rsp, 16
    lea r12, [rip + buffer]
    lea r13, [rip + lines]

    xor rcx, rcx
    mov r15d, 32768
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [r12 + rcx]
    mov edx, r15d
    sub edx, ecx
    syscall
    test rax, rax
    jz read_done
    add rcx, rax
    jmp read_loop
read_done:

    xor rsi, rsi
    xor r14, r14
parse_loop:
    cmp rsi, rcx
    jae parse_done
    mov r8, rsi
    xor r9, r9
scan_loop:
    cmp rsi, rcx
    jae scan_end
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x0A
    je newline_found
    inc rsi
    jmp scan_loop
newline_found:
    mov r9, rsi
    sub r9, r8
    mov eax, r8d
    mov [r13 + r14*8], eax
    mov eax, r9d
    mov [r13 + r14*8 + 4], eax
    inc r14
    inc rsi
    jmp parse_loop
scan_end:
    mov r9, rcx
    sub r9, r8
    test r9, r9
    jz parse_done
    mov eax, r8d
    mov [r13 + r14*8], eax
    mov eax, r9d
    mov [r13 + r14*8 + 4], eax
    inc r14
parse_done:

    xor r15, r15
outer_loop:
    cmp r15, r14
    jge sort_done
    mov rbp, r15
    lea rax, [r15 + 1]
    mov [rsp], rax
inner_loop:
    mov rax, [rsp]
    cmp rax, r14
    jge inner_done
    mov rdi, rax
    mov rsi, rbp
    call compare_lines
    test rax, rax
    jge not_less
    mov rbp, rdi
not_less:
    mov rax, [rsp]
    inc rax
    mov [rsp], rax
    jmp inner_loop
inner_done:
    cmp rbp, r15
    je skip_swap
    mov rax, [r13 + r15*8]
    mov rdx, [r13 + rbp*8]
    mov [r13 + r15*8], rdx
    mov [r13 + rbp*8], rax
skip_swap:
    inc r15
    jmp outer_loop
sort_done:

    xor r15, r15
output_loop:
    cmp r15, r14
    jge output_done
    mov eax, [r13 + r15*8]
    mov r8d, eax
    mov eax, [r13 + r15*8 + 4]
    mov r9d, eax
    test r9, r9
    jz skip_content
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r8]
    mov edx, r9d
    syscall
skip_content:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r15
    jmp output_loop
output_done:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov eax, [r13 + rdi*8]
    mov r8d, eax
    mov eax, [r13 + rdi*8 + 4]
    mov r9d, eax
    mov eax, [r13 + rsi*8]
    mov r10d, eax
    mov eax, [r13 + rsi*8 + 4]
    mov r11d, eax
    mov rcx, r9
    cmp rcx, r11
    jle .min_len_ok
    mov rcx, r11
.min_len_ok:
    xor rdx, rdx
.compare_loop:
    cmp rdx, rcx
    je .compare_lengths
    mov rsi, r8
    add rsi, rdx
    movzx eax, byte ptr [r12 + rsi]
    mov rsi, r10
    add rsi, rdx
    movzx ebx, byte ptr [r12 + rsi]
    cmp al, bl
    jb .less
    ja .greater
    inc rdx
    jmp .compare_loop
.less:
    mov rax, -1
    ret
.greater:
    mov rax, 1
    ret
.compare_lengths:
    cmp r9, r11
    jb .less
    ja .greater
    xor rax, rax
    ret
