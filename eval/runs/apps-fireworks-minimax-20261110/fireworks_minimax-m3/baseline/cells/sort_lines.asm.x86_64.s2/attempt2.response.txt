.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
.balign 16
input_buf: .space 32769
.balign 16
line_arr: .space 8200

.equ MAX_INPUT, 32768

.text
_start:
    lea r12, [rip + input_buf]
    lea r13, [rip + line_arr]
    xor r14, r14

.Lread_input:
    mov eax, 0
    mov edi, 0
    lea rsi, [r12 + r14]
    mov edx, MAX_INPUT
    sub edx, r14d
    jz .Lread_done
    syscall
    test rax, rax
    js .Lread_done
    jz .Lread_done
    add r14, rax
    jmp .Lread_input
.Lread_done:

    xor rsi, rsi
    xor r15, r15

.Lparse_lines:
    cmp rsi, r14
    jae .Lparse_done
    mov r8, rsi
    jmp .Lscan_check
.Lscan_loop:
    inc rsi
.Lscan_check:
    cmp rsi, r14
    jae .Lscan_end
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x0A
    jne .Lscan_loop
    mov rcx, rsi
    sub rcx, r8
    mov [r13 + r15*8], r8d
    mov [r13 + r15*8 + 4], ecx
    inc r15
    inc rsi
    jmp .Lparse_lines
.Lscan_end:
    mov rcx, rsi
    sub rcx, r8
    jz .Lparse_done
    mov [r13 + r15*8], r8d
    mov [r13 + r15*8 + 4], ecx
    inc r15
.Lparse_done:

    mov rbx, 1
.Lsort_outer:
    cmp rbx, r15
    jge .Lsort_done
    mov rbp, rbx
.Lsort_inner:
    test rbp, rbp
    jz .Lsort_inner_done
    mov rdi, rbp
    dec rdi
    mov rsi, rbp
    call compare_lines
    test rax, rax
    jle .Lsort_inner_done
    mov rdi, rbp
    dec rdi
    mov rsi, rbp
    call swap_lines
    dec rbp
    jmp .Lsort_inner
.Lsort_inner_done:
    inc rbx
    jmp .Lsort_outer
.Lsort_done:

    xor rbx, rbx
.Loutput_loop:
    cmp rbx, r15
    jge .Loutput_done
    mov r8d, [r13 + rbx*8]
    mov r9d, [r13 + rbx*8 + 4]
    test r9, r9
    jz .Lskip_data
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r8]
    mov edx, r9d
    syscall
.Lskip_data:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp .Loutput_loop
.Loutput_done:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    push rbx
    mov r8d, [r13 + rdi*8]
    mov r10d, [r13 + rdi*8 + 4]
    mov r9d, [r13 + rsi*8]
    mov r11d, [r13 + rsi*8 + 4]
    mov rcx, r10
    cmp rcx, r11
    jle .Lmin_ok
    mov rcx, r11
.Lmin_ok:
    xor rdx, rdx
.Lcmp_bytes:
    cmp rdx, rcx
    je .Lend_bytes
    lea rax, [r8 + rdx]
    movzx ebx, byte ptr [r12 + rax]
    lea rax, [r9 + rdx]
    movzx eax, byte ptr [r12 + rax]
    cmp bl, al
    jb .Lless
    ja .Lgreater
    inc rdx
    jmp .Lcmp_bytes
.Lend_bytes:
    cmp r10, r11
    jb .Lless
    ja .Lgreater
    xor rax, rax
    pop rbx
    ret
.Lless:
    mov rax, -1
    pop rbx
    ret
.Lgreater:
    mov rax, 1
    pop rbx
    ret

swap_lines:
    mov rax, [r13 + rdi*8]
    mov rdx, [r13 + rsi*8]
    mov [r13 + rsi*8], rax
    mov [r13 + rdi*8], rdx
    ret
