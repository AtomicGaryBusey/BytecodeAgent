//
// wc - count lines, words, and bytes of standard input
//

.section .text

// print_dec: write decimal n at *buf, advancing *buf
// input:  x0 = value, x1 = buf pointer
// output: x0 = buf pointer past the digits
// clobbers x2-x7, x9 (caller-saved)
print_dec:
    cmp     x0, #0
    b.ne    .pd_normal
    mov     w2, #0x30
    strb    w2, [x1], #1
    mov     x0, x1
    ret
.pd_normal:
    sub     sp, sp, #32
    mov     x2, sp
    mov     x3, #0
    mov     x9, #10
.pd_loop:
    udiv    x4, x0, x9
    msub    x5, x4, x9, x0
    add     w5, w5, #0x30
    strb    w5, [x2, x3]
    add     x3, x3, #1
    mov     x0, x4
    cbnz    x0, .pd_loop
    mov     x4, #0
.pd_copy:
    sub     x6, x3, x4
    sub     x6, x6, #1
    ldrb    w7, [x2, x6]
    strb    w7, [x1], #1
    add     x4, x4, #1
    cmp     x4, x3
    b.lt    .pd_copy
    add     sp, sp, #32
    mov     x0, x1
    ret

.globl _start
_start:
    mov     x11, xzr
    mov     x12, xzr
    mov     x13, xzr
    mov     w14, wzr

    adrp    x9, read_buf
    add     x9, x9, :lo12:read_buf

.L_read:
    mov     x0, xzr
    mov     x1, x9
    mov     x2, #65536
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    .L_eof
    mov     x17, x0
    mov     x16, x9

.L_proc:
    cbz     x17, .L_read
    ldrb    w18, [x16], #1
    sub     x17, x17, #1
    add     x11, x11, #1

    cmp     w18, #0x0A
    b.ne    .L_no_nl
    add     x12, x12, #1
.L_no_nl:
    cmp     w18, #0x09
    b.lt    .L_not_ws
    cmp     w18, #0x0D
    b.le    .L_is_ws
    cmp     w18, #0x20
    b.ne    .L_not_ws
.L_is_ws:
    cbz     w14, .L_after_wc
    mov     w14, wzr
    b       .L_after_wc
.L_not_ws:
    cbnz    w14, .L_after_wc
    mov     w14, #1
    add     x13, x13, #1
.L_after_wc:
    b       .L_proc

.L_eof:
    adrp    x10, out_buf
    add     x10, x10, :lo12:out_buf
    mov     x15, x10

    mov     x0, x12
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x20
    strb    w2, [x15], #1

    mov     x0, x13
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x20
    strb    w2, [x15], #1

    mov     x0, x11
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x0A
    strb    w2, [x15], #1

    mov     x0, #1
    sub     x2, x15, x10
    mov     x1, x10
    mov     x8, #64
    svc     #0

    mov     x0, xzr
    mov     x8, #93
    svc     #0

.section .bss
.align 4
read_buf:
    .space  65536
.align 4
out_buf:
    .space  64
