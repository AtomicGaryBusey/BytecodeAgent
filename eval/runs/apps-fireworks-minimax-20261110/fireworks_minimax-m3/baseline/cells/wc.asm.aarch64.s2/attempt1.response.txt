//
// wc - count lines, words, and bytes of standard input
//

.section .text

// print_dec: write decimal n at *buf (forward)
// input:  x0 = n, x1 = buf
// output: x0 = buf past the digits
print_dec:
    cmp     x0, #0
    b.ne    .pd_normal
    mov     w2, #0x30
    strb    w2, [x1], #1
    mov     x0, x1
    ret
.pd_normal:
    sub     sp, sp, #32
    mov     x2, sp
    mov     x3, #0
.pd_loop:
    udiv    x4, x0, #10
    msub    x5, x4, #10, x0
    add     w5, w5, #0x30
    strb    w5, [x2, x3]
    add     x3, x3, #1
    mov     x0, x4
    cbnz    x0, .pd_loop
    mov     x4, #0
.pd_copy:
    sub     x6, x3, x4
    sub     x6, x6, #1
    ldrb    w7, [x2, x6]
    strb    w7, [x1], #1
    add     x4, x4, #1
    cmp     x4, x3
    b.lt    .pd_copy
    add     sp, sp, #32
    mov     x0, x1
    ret

.globl _start
_start:
    mov     x11, xzr                // bytes count
    mov     x12, xzr                // lines count
    mov     x13, xzr                // words count
    mov     w14, wzr                // in_word flag

    adrp    x9, read_buf
    add     x9, x9, :lo12:read_buf  // x9 = read buffer pointer

.L_read:
    mov     x0, xzr                 // fd = 0 (stdin)
    mov     x1, x9
    mov     x2, #65536
    mov     x8, #63                 // sys_read
    svc     #0
    cmp     x0, #0
    b.le    .L_eof
    mov     x17, x0                 // bytes remaining
    mov     x16, x9                 // current read pointer

.L_proc:
    cbz     x17, .L_read            // chunk processed, read more
    ldrb    w18, [x16], #1          // load byte, advance pointer
    sub     x17, x17, #1
    add     x11, x11, #1            // bytes++

    cmp     w18, #0x0A
    b.ne    .L_no_nl
    add     x12, x12, #1            // lines++
.L_no_nl:
    // whitespace check: (b == 0x09..0x0D) || (b == 0x20)
    cmp     w18, #0x09
    b.lt    .L_not_ws
    cmp     w18, #0x0D
    b.le    .L_is_ws
    cmp     w18, #0x20
    b.ne    .L_not_ws
.L_is_ws:
    cbz     w14, .L_after_wc        // not in word, no change
    mov     w14, wzr                // transition out of word
    b       .L_after_wc
.L_not_ws:
    cbnz    w14, .L_after_wc        // in word, no change
    mov     w14, #1                 // transition into word
    add     x13, x13, #1            // words++
.L_after_wc:
    b       .L_proc

.L_eof:
    adrp    x10, out_buf
    add     x10, x10, :lo12:out_buf
    mov     x15, x10                // x15 = current output position

    mov     x0, x12                 // lines
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x20               // ' '
    strb    w2, [x15], #1

    mov     x0, x13                 // words
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x20               // ' '
    strb    w2, [x15], #1

    mov     x0, x11                 // bytes
    mov     x1, x15
    bl      print_dec
    mov     x15, x0

    mov     w2, #0x0A               // '\n'
    strb    w2, [x15], #1

    mov     x0, #1                  // fd = 1 (stdout)
    sub     x2, x15, x10            // length to write
    mov     x1, x10
    mov     x8, #64                 // sys_write
    svc     #0

    mov     x0, xzr                 // status 0
    mov     x8, #93                 // sys_exit
    svc     #0

.section .bss
.align 4
read_buf:
    .space  65536
.align 4
out_buf:
    .space  64
