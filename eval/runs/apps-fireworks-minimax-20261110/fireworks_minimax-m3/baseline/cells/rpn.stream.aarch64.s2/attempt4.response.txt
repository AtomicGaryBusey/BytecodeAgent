.equ INPUT_SIZE, 65536
.equ STACK_SIZE, 8000
.equ OUT_BUF_SIZE, 32

.section .rodata
error_msg:  .ascii "error\n"
zero_msg:  .ascii "0\n"
minus_char: .ascii "-"
newline_char: .ascii "\n"

.section .bss
.align 3
input_buf: .space INPUT_SIZE
.align 3
stack: .space STACK_SIZE
.align 3
out_buf: .space OUT_BUF_SIZE

.text
.globl _start

//--------------------------------------------------------------
// Entry point: read entire stdin into input_buf, then parse lines.
//--------------------------------------------------------------
_start:
    // x24 = base of input_buf
    adrp x24, input_buf
    add  x24, x24, :lo12:input_buf

    // x23 = total bytes read so far
    mov  x23, #0

read_loop:
    // pointer = input_buf + total_len
    add  x1, x24, x23
    // max bytes to read
    mov  x2, #0x1, lsl #16    // 0x10000 = 65536
