.globl _start

// wc: count lines, words, and bytes of stdin.

.section .text
_start:
    mov x19, #0           // lines
    mov x20, #0           // words
    mov x21, #0           // bytes
    mov x22, #0           // in_word (0 or 1)

    adrp x23, buffer
    add x23, x23, :lo12:buffer

.Lread_loop:
    mov x0, #0            // fd = 0 (stdin)
    mov x1, x23           // buf
    movz x2, #1, lsl #16  // count = 65536
    mov x8, #63           // sys_read
    svc #0

    cmp x0, #0
    b.le .Lfinish         // 0 (EOF) or negative

    mov x24, x0           // n = bytes to process
    mov x25, x23          // ptr

.Lprocess_loop:
    ldrb w0, [x25], #1    // load byte, advance ptr
    add x21, x21, #1      // bytes++

    // Count newlines
    cmp w0, #0x0A
    b.ne .Lskip_newline
    add x19, x19, #1      // lines++
.Lskip_newline:

    // Is it whitespace? 0x09..0x0D or 0x20
    cmp w0, #0x20
    b.eq .Lis_space
    cmp w0, #0x09
    b.lt .Lnot_space
    cmp w0, #0x0D
    b.le .Lis_space

.Lnot_space:
    cbnz x22, .Lcontinue
    mov x22, #1
    add x20, x20, #1      // words++
    b .Lcontinue
.Lis_space:
    mov x22, #0
.Lcontinue:
    subs x24, x24, #1
    b.ne .Lprocess_loop
    b .Lread_loop

.Lfinish:
    // Build "<lines> <words> <bytes>\n" backwards in outbuf
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf
    mov x11, #10          // divisor constant
    mov x9, #63
    mov w10, #10          // '\n'
    strb w10, [x26, x9]
    sub x9, x9, #1

    // bytes (x21)
    mov x14, x9
    mov x0, x21
.Lcv_bytes:
    cbz x0, .Lcv_bytes_done
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add w13, w13, #48
    strb w13, [x26, x9]
    sub x9, x9, #1
    mov x0, x12
    b .Lcv_bytes
.Lcv_bytes_done:
    cmp x9, x14
    b.ne .Lskip_zero_bytes
    mov w10, #48          // '0'
    strb w10, [x26, x9]
    sub x9, x9, #1
.Lskip_zero_bytes:
    mov w10, #32          // ' '
    strb w10, [x26, x9]
    sub x9, x9, #1

    // words (x20)
    mov x14, x9
    mov x0, x20
.Lcv_words:
    cbz x0, .Lcv_words_done
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add w13, w13, #48
    strb w13, [x26, x9]
    sub x9, x9, #1
    mov x0, x12
    b .Lcv_words
.Lcv_words_done:
    cmp x9, x14
    b.ne .Lskip_zero_words
    mov w10, #48          // '0'
    strb w10, [x26, x9]
    sub x9, x9, #1
.Lskip_zero_words:
    mov w10, #32          // ' '
    strb w10, [x26, x9]
    sub x9, x9, #1

    // lines (x19)
    mov x14, x9
    mov x0, x19
.Lcv_lines:
    cbz x0, .Lcv_lines_done
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add w13, w13, #48
    strb w13, [x26, x9]
    sub x9, x9, #1
    mov x0, x12
    b .Lcv_lines
.Lcv_lines_done:
    cmp x9, x14
    b.ne .Lskip_zero_lines
    mov w10, #48          // '0'
    strb w10, [x26, x9]
    sub x9, x9, #1
.Lskip_zero_lines:

    // Write outbuf[x9+1 .. 63] to stdout
    add x1, x26, x9
    add x1, x1, #1
    mov x2, #63
    sub x2, x2, x9        // length = 63 - x9
    mov x0, #1            // fd = 1 (stdout)
    mov x8, #64           // sys_write
    svc #0

    mov x0, #0
    mov x8, #93           // sys_exit
    svc #0

.section .bss
.align 4
buffer: .zero 65536
.align 4
outbuf: .zero 64
