.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4
input_buf:
    .space 49152
.align 4
output_buf:
    .space 66500

.section .text
.globl _start

_start:
    // Read all input from stdin into input_buf
    mov     x15, #0
    mov     x16, #49152
    adrp    x17, input_buf
    add     x17, x17, :lo12:input_buf
.L_read_loop:
    cmp     x15, x16
    b.ge    .L_read_done
    add     x1, x17, x15
    sub     x2, x16, x15
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cbz     x0, .L_read_done
    add     x15, x15, x0
    b       .L_read_loop
.L_read_done:
    mov     x12, x15          // total number of bytes read
    cbz     x12, .L_exit      // empty input -> no output

    // Initialize pointers and line counter
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf
    adrp    x13, alphabet
    add     x13, x13, :lo12:alphabet
    adrp    x19, output_buf
    add     x19, x19, :lo12:output_buf
    mov     x10, x19          // output buffer pointer
    mov     w14, #0           // current line length (0‑75)

.L_loop3:
    cmp     x12, #3
    blt     .L_final

    // Load three bytes of input
    ldrb    w20, [x9]
    ldrb    w21, [x9, #1]
    ldrb    w22, [x9, #2]

    // --- character 0 ---
    lsr     w3, w20, #2
    mov     x3, w3
