.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4
input_buf:
    .space 49152
.align 4
output_buf:
    .space 66500

.section .text
.globl _start

_start:
    // Read all of stdin into input_buf
    mov     x15, #0
    mov     x16, #49152
    adrp    x17, input_buf
    add     x17, x17, :lo12:input_buf
.L_read_loop:
    cmp     x15, x16
    b.ge    .L_read_done
    add     x1, x17, x15
    sub     x2, x16, x15
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cbz     x0, .L_read_done
    add     x15, x15, x0
    b       .L_read_loop
.L_read_done:
    mov     x12, x15                  // total bytes read
    cbz     x12, .L_exit_clean        // empty input -> no output

    // Set up pointers
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf
    adrp    x13, alphabet
    add     x13, x13, :lo12:alphabet
    adrp    x19, output_buf
    add     x19, x19, :lo12:output_buf
    mov     x10, x19                  // current output position
    mov     w14, #0                   // column in current line (0..75)

.L_loop3:
    cmp     x12, #3
    blt     .L_final

    // Read three bytes b0, b1, b2
    ldrb    w20, [x9], #1
    ldrb    w21, [x9], #1
    ldrb    w22, [x9], #1
    sub     x12, x12, #3

    // c0 = b0 >> 2
    lsr     w3, w20, #2
    // c1 = ((b0 & 3) << 4) | (b1 >> 4)
    and     w4, w20, #3
    lsl     w4, w4, #4
    lsr     w5, w21, #4
    orr     w4, w4, w5
    // c2 = ((b1 & 15) << 2) | (b2 >> 6)
    and     w5, w21, #0x0F
    lsl     w5, w5, #2
    lsr     w6, w22, #6
    orr     w5, w5, w6
    // c3 = b2 & 63
    and     w6, w22, #0x3F

    // Look up characters in the alphabet
    ldrb    w0, [x13, x3]
    ldrb    w1, [x13, x4]
    ldrb    w2, [x13, x5]
    ldrb    w7, [x13, x6]

    // Write four output characters
    strb    w0, [x10], #1
    strb    w1, [x10], #1
    strb    w2, [x10], #1
    strb    w7, [x10], #1

    add     w14, w14, #4
    cmp     w14, #76
    b.ne    .L_loop3

    // Completed a 76-char line: emit newline
    mov     w0, #10
    strb    w0, [x10], #1
    mov     w14, #0
    b       .L_loop3

.L_final:
    cbz     x12, .L_check_nl
    cmp     x12, #1
    b.eq    .L_final1

    // Two bytes remaining
    ldrb    w20, [x9], #1
    ldrb    w21, [x9], #1

    lsr     w3, w20, #2
    and     w4, w20, #3
    lsl     w4, w4, #4
    lsr     w5, w21, #4
    orr     w4, w4, w5
    and     w5, w21, #0x0F
    lsl     w5, w5, #2

    ldrb    w0, [x13, x3]
    ldrb    w1, [x13, x4]
    ldrb    w2, [x13, x5]
    mov     w7, #61                   // '='

    strb    w0, [x10], #1
    strb    w1, [x10], #1
    strb    w2, [x10], #1
    strb    w7, [x10], #1

    add     w14, w14, #4
    b       .L_check_nl

.L_final1:
    // One byte remaining
    ldrb    w20, [x9], #1

    lsr     w3, w20, #2
    and     w4, w20, #3
    lsl     w4, w4, #4

    ldrb    w0, [x13, x3]
    ldrb    w2, [x13, x4]
    mov     w1, #61                   // '='
    mov     w7, #61                   // '='

    strb    w0, [x10], #1
    strb    w2, [x10], #1
    strb    w1, [x10], #1
    strb    w7, [x10], #1

    add     w14, w14, #4
    // fall through

.L_check_nl:
    cmp     w14, #76
    b.ne    .L_partial_nl
    mov     w0, #10
    strb    w0, [x10], #1
    b       .L_write

.L_partial_nl:
    cbnz    w14, .L_emit_nl
    b       .L_write
.L_emit_nl:
    mov     w0, #10
    strb    w0, [x10], #1

.L_write:
    sub     x2, x10, x19
    mov     x1, x19
    mov     x0, #1
    mov     x8, #64
    svc     #0

.L_exit_clean:
    mov     x0, #0
    mov     x8, #93
    svc     #0
