.section .rodata
hex_table: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .space 65536
.align 16
line_buf: .space 96

.text
.globl _start
_start:
    xor r13, r13

.Lread_loop:
    cmp r13, 65536
    jae .Lread_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buf + r13]
    mov rdx, 65536
    sub rdx, r13
    syscall
    cmp rax, 0
    jle .Lread_done
    add r13, rax
    jmp .Lread_loop

.Lread_done:
    xor r15, r15

.Lrow_loop:
    mov rax, r13
    sub rax, r15
    test rax, rax
    jle .Lprint_total
    cmp rax, 16
    jb .Lless16
    mov rax, 16
.Lless16:
    mov rcx, rax
    lea rsi, [line_buf]
    mov rax, r15
    call .Lprint_hex8
    mov word ptr [rsi], 0x2020
    add rsi, 2
    xor ebx, ebx
.Lslot_loop:
    cmp ebx, ecx
    jb .Lslot_real
    mov dword ptr [rsi], 0x20202020
    add rsi, 3
    jmp .Lslot_cont
.Lslot_real:
    movzx eax, byte ptr [input_buf + r15 + rbx]
    call .Lprint_hex2_space
.Lslot_cont:
    cmp ebx, 7
    jne .Lno_extra
    mov byte ptr [rsi], 0x20
    inc rsi
.Lno_extra:
    inc ebx
    cmp ebx, 16
    jb .Lslot_loop
    mov word ptr [rsi], 0x7c20
    add rsi, 2
    xor ebx, ebx
.Lascii_loop:
    cmp ebx, ecx
    jae .Lascii_done
    movzx eax, byte ptr [input_buf + r15 + rbx]
    cmp al, 0x20
    jb .Ldot
    cmp al, 0x7e
    jbe .Lnotdot
.Ldot:
    mov al, 0x2e
.Lnotdot:
    mov byte ptr [rsi], al
    inc rsi
    inc ebx
    jmp .Lascii_loop
.Lascii_done:
    mov word ptr [rsi], 0x0a7c
    add rsi, 2
    lea r8, [line_buf]
    mov rdx, rsi
    sub rdx, r8
    mov rsi, r8
    mov rax, 1
    mov rdi, 1
    syscall
    add r15, 16
    jmp .Lrow_loop

.Lprint_total:
    lea rsi, [line_buf]
    mov rax, r13
    call .Lprint_hex8
    mov byte ptr [rsi], 0x0a
    mov rax, 1
    mov rdi, 1
    lea rsi, [line_buf]
    mov rdx, 9
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

.Lprint_hex8:
    push rbx
    mov ebx, eax
    mov r8d, 8
.Lhex8_lp:
    rol ebx, 4
    mov eax, ebx
    and eax, 0x0f
    movzx eax, byte ptr [hex_table + rax]
    mov byte ptr [rsi], al
    inc rsi
    dec r8d
    jnz .Lhex8_lp
    pop rbx
    ret

.Lprint_hex2_space:
    push rbx
    mov ebx, eax
    rol ebx, 4
    mov eax, ebx
    and eax, 0x0f
    movzx eax, byte ptr [hex_table + rax]
    mov byte ptr [rsi], al
    inc rsi
    rol ebx, 4
    mov eax, ebx
    and eax, 0x0f
    movzx eax, byte ptr [hex_table + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    pop rbx
    ret
