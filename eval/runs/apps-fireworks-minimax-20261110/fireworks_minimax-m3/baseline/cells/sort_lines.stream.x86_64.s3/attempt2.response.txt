.section .rodata
newline: .byte 10

.section .bss
.align 16
buffer: .space 32768
descriptors: .space 4000

.section .text
.globl _start

_start:
    lea r13, [rip + buffer]
    xor r15, r15

.read_loop:
    mov edx, 32768
    sub edx, r15d
    jz .read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r15]
    syscall
    test rax, rax
    jz .read_done
    add r15, rax
    jmp .read_loop

.read_done:
    lea r12, [rip + descriptors]
    xor r14, r14
    xor r9, r9

.parse_loop:
    cmp r9, r15
    jae .parse_done
    mov ebx, r9d
    mov r10, r9
.find_newline:
    cmp r10, r15
    jae .no_newline
    movzx eax, byte ptr [r13 + r10]
    cmp al, 10
    je .newline_found
    inc r10
    jmp .find_newline

.newline_found:
    mov eax, r10d
    sub eax, ebx
    mov edi, r14d
    shl edi, 2
    mov word ptr [r12 + rdi], bx
    mov word ptr [r12 + rdi + 2], ax
    inc r14
    mov r9, r10
    inc r9
    jmp .parse_loop

.no_newline:
    mov eax, r15d
    sub eax, ebx
    mov edi, r14d
    shl edi, 2
    mov word ptr [r12 + rdi], bx
    mov word ptr [r12 + rdi + 2], ax
    inc r14

.parse_done:
    cmp r14, 2
    jl .sort_done
    mov r9, 1
.sort_outer:
    cmp r9, r14
    jae .sort_done
    mov edi, r9d
    shl edi, 2
    movzx ebx, word ptr [r12 + rdi]
    movzx ebp, word ptr [r12 + rdi + 2]
    mov r10, r9
    dec r10
    jmp .sort_cond

.sort_cond:
    test r10, r10
    js .sort_store
    mov edi, r10d
    shl edi, 2
    movzx edx, word ptr [r12 + rdi]
    movzx ecx, word ptr [r12 + rdi + 2]
    push r10
    push r9
    mov rdi, rbx
    mov esi, ebp
    call compare_lines
    pop r9
    pop r10
    test eax, eax
    jl .sort_shift
    jmp .sort_store

.sort_shift:
    mov edi, r10d
    shl edi, 2
    movzx eax, word ptr [r12 + rdi]
    movzx ecx, word ptr [r12 + rdi + 2]
    mov r11, r10
    inc r11
    mov edx, r11d
    shl edx, 2
    mov word ptr [r12 + rdx], ax
    mov word ptr [r12 + rdx + 2], cx
    dec r10
    jmp .sort_cond

.sort_store:
    mov eax, r10d
    inc eax
    shl eax, 2
    mov word ptr [r12 + rax], bx
    mov word ptr [r12 + rax + 2], bp
    inc r9
    jmp .sort_outer

.sort_done:
    xor r9, r9

.output_loop:
    cmp r9, r14
    je .exit_program
    mov edi, r9d
    shl edi, 2
    movzx ebx, word ptr [r12 + rdi]
    movzx ecx, word ptr [r12 + rdi + 2]
    mov eax, 1
    mov edi, 1
    lea rsi, [r13 + rbx]
    mov edx, ecx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r9
    jmp .output_loop

.exit_program:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    push rbx
    push rbp
    push r12
    mov rax, rsi
    cmp rax, rcx
    jle .cl_min_done
    mov rax, rcx
.cl_min_done:
    lea r9, [r13 + rdi]
    lea r10, [r13 + rdx]
    xor r11, r11
.cl_loop:
    cmp r11, rax
    je .cl_eq
    movzx eax, byte ptr [r9 + r11]
    mov edi, eax
    movzx eax, byte ptr [r10 + r11]
    cmp edi, eax
    jne .cl_diff
    inc r11
    jmp .cl_loop
.cl_diff:
    sub edi, eax
.cl_ret:
    pop r12
    pop rbp
    pop rbx
    mov eax, edi
    ret
.cl_eq:
    sub esi, ecx
    mov edi, esi
    jmp .cl_ret
