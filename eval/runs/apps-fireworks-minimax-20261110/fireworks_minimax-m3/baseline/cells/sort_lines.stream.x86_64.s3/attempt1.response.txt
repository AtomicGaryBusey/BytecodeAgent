.section .rodata
newline: .byte 10

.section .bss
.align 16
buffer: .space 32768
descriptors: .space 4000

.section .text
.globl _start

_start:
    lea r13, [rip + buffer]          # buffer base
    xor ebp, ebp                      # total_read = 0

read_loop:
    mov rdx, 32768
    sub rdx, rbp                      # remaining = 32768 - total_read
    cmp rdx, 0
    je read_done
    mov rax, 0                        # syscall read
    mov rdi, 0                        # fd = stdin
    lea rsi, [r13 + rbp]              # buf = buffer + total_read
    syscall
    cmp rax, 0
    je read_done
    add rbp, rax                      # total_read += bytes_read
    jmp read_loop

read_done:
    xor r15d, r15d                    # i = 0 (scan index)
    xor r14d, r14d                    # n = 0 (line count)
    lea r12, [rip + descriptors]      # descriptors base

parse_loop:
    cmp r15, rbp                      # if i >= total_read
    jae parse_done
    mov rbx, r15                      # start = i
    mov rcx, r15                      # j = i
find_newline:
    cmp rcx, rbp
    je no_newline
    movzx eax, byte ptr [r13 + rcx]
    cmp al, 10
    je newline_found
    inc rcx
    jmp find_newline

newline_found:
    mov rdx, rcx
    sub rdx, rbx                      # length = j - start
    mov rax, r14
    shl rax, 2                        # offset = n * 4
    mov word ptr [r12 + rax], bx      # store offset
    mov word ptr [r12 + rax + 2], dx  # store length
    inc r14
    lea r15, [rcx + 1]                # i = j + 1
    jmp parse_loop

no_newline:
    mov rdx, rbp
    sub rdx, rbx                      # length = total_read - start
    mov rax, r14
    shl rax, 2
    mov word ptr [r12 + rax], bx
    mov word ptr [r12 + rax + 2], dx
    inc r14
    jmp parse_done

parse_done:
    # Insertion sort
    mov r15, 1                        # i = 1
    cmp r14, 2
    jl sort_done                      # skip if n < 2

sort_outer:
    cmp r15, r14
    jae sort_done
    # load key = descriptors[i]
    mov rax, r15
    shl rax, 2
    movzx ebx, word ptr [r12 + rax]       # key.offset
    movzx ecx, word ptr [r12 + rax + 2]   # key.length
    lea rbp, [r15 - 1]                    # j = i-1
    jmp sort_cond

sort_cond:
    cmp rbp, 0
    jl sort_store
    # load arr[j] into edx (offset) and r8d (length)
    mov rax, rbp
    shl rax, 2
    add rax, r12
    movzx edx, word ptr [rax]
    movzx r8d, word ptr [rax + 2]
    # call compare_lines(key_offset=rbx, key_len=rcx, arr_offset=rdx, arr_len=r8)
    push rbx
    push rcx
    mov rdi, rbx
    mov rsi, rcx
    mov rcx, r8
    call compare_lines
    pop rcx
    pop rbx
    cmp eax, 0
    jl sort_shift
    jmp sort_store

sort_shift:
    # shift arr[j] down to arr[j+1]
    mov rax, rbp
    shl rax, 2
    add rax, r12
    movzx edi, word ptr [rax]
    movzx esi, word ptr [rax + 2]
    mov rdx, rbp
    inc rdx
    shl rdx, 2
    add rdx, r12
    mov word ptr [rdx], di
    mov word ptr [rdx + 2], si
    dec rbp
    jmp sort_cond

sort_store:
    # store key at arr[j+1]
    mov rax, rbp
    inc rax
    shl rax, 2
    add rax, r12
    mov word ptr [rax], bx
    mov word ptr [rax + 2], cx
    inc r15
    jmp sort_outer

sort_done:
    xor r15d, r15d                    # i = 0 for output

output_loop:
    cmp r15, r14
    je exit_program
    mov rax, r15
    shl rax, 2
    movzx ebx, word ptr [r12 + rax]       # offset
    movzx ecx, word ptr [r12 + rax + 2]   # length
    # write line bytes
    mov rax, 1                        # sys_write
    mov rdi, 1                        # fd = stdout
    lea rsi, [r13 + rbx]
    mov rdx, rcx
    syscall
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r15
    jmp output_loop

exit_program:
    mov rax, 60                       # sys_exit
    xor rdi, rdi                      # status = 0
    syscall

compare_lines:
    # r13 is buffer base (global)
    # arguments: rdi = offset1, rsi = length1, rdx = offset2, rcx = length2
    # returns: eax = signed difference (negative if line1 < line2)
    mov rax, rsi
    cmp rax, rcx
    jle .min_set
    mov rax, rcx
.min_set:
    xor r8, r8
.compare_loop:
    cmp r8, rax
    je .compare_lengths
    movzx eax, byte ptr [r13 + rdi + r8]
