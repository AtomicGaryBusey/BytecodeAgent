.globl _start

.section .rodata
error_msg:
    .ascii "error\n"
error_len = . - error_msg

.section .bss
.balign 16
input_buffer:
    .space 65536
.balign 8
rpn_stack:
    .space 8192
.balign 8
num_buf:
    .space 32

.text
_start:
    xor r12, r12
.Lread_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle .Linput_done
    add r12, rax
    cmp r12, 65536
    jae .Linput_done
    jmp .Lread_loop
.Linput_done:
    xor r13, r13
.Lprocess_lines:
    cmp r13, r12
    je .Lall_done
    mov r14, r13
.Lfind_eol:
    cmp r14, r12
    je .Lline_done
    mov al, byte ptr [rip + input_buffer + r14]
