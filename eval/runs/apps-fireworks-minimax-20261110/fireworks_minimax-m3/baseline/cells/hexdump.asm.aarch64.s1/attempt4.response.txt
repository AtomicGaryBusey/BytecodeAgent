.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .bss
buf: .space 65536
row: .space 80

.section .text

_start:
    adrp x9, buf
    add  x9, x9, :lo12:buf
    mov  x10, xzr

.L_read_loop:
    cmp x10, #65536
    b.ge .L_input_done
    mov x0, #0
    add x1, x9, x10
    mov x2, #65536
    sub x2, x2, x10
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .L_input_done
    add x10, x10, x0
    b .L_read_loop

.L_input_done:
    mov x12, xzr

.L_row_loop:
    cmp x12, x10
    b.ge .L_total_line
    sub x18, x10, x12
    mov x17, x18
    cmp x17, #16
    b.le .L_n_done
    mov x17, #16
.L_n_done:
    adrp x16, row
    add  x16, x16, :lo12:row
    mov x19, x12
    mov x20, xzr
.L_off_digits:
    lsr w21, w19, #28
    and w21, w21, #0xf
    add w0, w21, #'0'
    cmp w0, #'9'
    b.le .L_off_nibble_ok
    add w0, w0, #0x27
.L_off_nibble_ok:
    strb w0, [x16, x20]
    lsl x19, x19, #4
    add x20, x20, #1
    cmp x20, #8
    b.lt .L_off_digits
    mov w0, #' '
    strb w0, [x16, #8]
    strb w0, [x16, #9]
    mov x20, xzr
.L_slots_loop:
    cmp x20, #16
    b.ge .L_slots_done
    add x21, x20, x20, lsl #1
    add x21, x21, #10
    cmp x20, #8
    b.lt .L_slot_no_extra
    add x21, x21, #1
.L_slot_no_extra:
    cmp x20, x17
    b.ge .L_slot_empty
    add x23, x9, x12
    ldrb w24, [x23, x20]
    mov w0, w24
    mov x1, x16
    mov x2, x21
    bl print_hex_byte
    b .L_slot_continue
.L_slot_empty:
    mov w0, #' '
    strb w0, [x16, x21]
    add x22, x21, #1
    strb w0, [x16, x22]
    add x22, x22, #1
    strb w0, [x16, x22]
.L_slot_continue:
    add x20, x20, #1
    b .L_slots_loop
.L_slots_done:
    mov w0, #' '
    strb w0, [x16, #59]
    mov w0, #'|'
    strb w0, [x16, #60]
    mov x20, xzr
.L_chars_loop:
    cmp x20, x17
    b.ge .L_chars_done
    add x23, x9, x12
    ldrb w24, [x23, x20]
    cmp w24, #0x20
    b.lt .L_chars_dot
    cmp w24, #0x7e
    b.le .L_chars_store
.L_chars_dot:
    mov w24, #'.'
.L_chars_store:
    add x22, x20, #61
    strb w24, [x16, x22]
    add x20, x20, #1
    b .L_chars_loop
.L_chars_done:
    add x22, x17, #61
    mov w0, #'|'
    strb w0, [x16, x22]
    add x22, x22, #1
    mov w0, #'\n'
    strb w0, [x16, x22]
    add x2, x17, #63
    mov x0, #1
    mov x1, x16
    mov x8, #SYS_WRITE
    svc #0
    add x12, x12, #16
    b .L_row_loop
.L_total_line:
    adrp x16, row
    add  x16, x16, :lo12:row
    mov x19, x10
    mov x20, xzr
.L_total_digits:
    lsr w21, w19, #28
    and w21, w21, #0xf
    add w0, w21, #'0'
    cmp w0, #'9'
    b.le .L_total_nibble_ok
    add w0, w0, #0x27
.L_total_nibble_ok:
    strb w0, [x16, x20]
    lsl x19, x19, #4
    add x20, x20, #1
    cmp x20, #8
    b.lt .L_total_digits
    mov w0, #'\n'
    strb w0, [x16, #8]
    mov x0, #1
    mov x1, x16
    mov x2, #9
    mov x8, #SYS_WRITE
    svc #0
    mov x0, xzr
    mov x8, #SYS_EXIT
    svc #0

print_hex_byte:
    lsr w3, w0, #4
    and w3, w3, #0xf
    add w3, w3, #'0'
    cmp w3, #'9'
    b.le .L_high_ok
    add w3, w3, #0x27
.L_high_ok:
    strb w3, [x1, x2]
    add x2, x2, #1
    and w3, w0, #0xf
    add w3, w3, #'0'
    cmp w3, #'9'
    b.le .L_low_ok
    add w3, w3, #0x27
.L_low_ok:
    strb w3, [x1, x2]
    add x2, x2, #1
    mov w3, #' '
    strb w3, [x1, x2]
    ret
