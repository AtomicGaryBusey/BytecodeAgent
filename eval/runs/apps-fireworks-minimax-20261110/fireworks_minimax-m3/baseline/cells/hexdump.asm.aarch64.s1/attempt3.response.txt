.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .bss
buf: .skip 65536
row: .skip 80

.section .text

_start:
    adrp x9, buf
    add x9, x9, :lo12:buf
    mov x10, xzr

.L_read_loop:
    cmp x10, #65536
    b.ge .L_input_done
    mov x0, #0
    add x1, x9, x10
    mov x2, #65536
    sub x2, x2, x10
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le .L_input_done
    add x10, x10, x0
    b .L_read_loop

.L_input_done:
    mov x12, xzr

.L_row_loop:
    cmp x12, x10
    b.ge .L_final_offset
    sub x18, x10, x12
    mov x17, #16
    cmp x18, #16
    csel x17, x18, x17, lt

    adrp x16, row
    add x16, x16, :lo12:row

    mov w19, w12
    mov w20, xzr

.L_off_digits:
    lsr w21, w19, #28
    and w21, w21, #0xf
    mov x1, x16
    mov w2, w20
    mov w0, w21
    bl print_nibble
    lsl w19, w19, #4
    add w20, w20, #1
    cmp w20, #8
    b.lt .L_off_digits

    mov w21, #' '
    strb w21, [x16, #8]
    strb w21, [x16, #9]

    mov w20, xzr

.L_slots_loop:
    cmp w20, #16
    b.ge .L_slots_done

    add w21, w20, w20, lsl #1
    add w21, w21, #10
    cmp w20, #8
    b.lt .L_slot_idx_done
    add w21, w21, #1
.L_slot_idx_done:
    mov w22, w21

    cmp w20, w17
    b.ge .L_slot_empty

    add x23, x9, x12
    ldrb w24, [x23, x20]
    mov x1, x16
    mov w2, w22
    mov w0, w24
    bl print_hex_byte
    mov w24, #' '
    add w25, w22, #2
    strb w24, [x16, x25]
    b .L_slot_continue

.L_slot_empty:
    mov w24, #' '
    strb w24, [x16, w22]
    add w25, w22, #1
    strb w24, [x16, w25]
    add w25, w25, #1
    strb w24, [x16, w25]

.L_slot_continue:
    add w20, w20, #1
    b .L_slots_loop

.L_slots_done:
    mov w24, #' '
    strb w24, [x16, #59]
    mov w24, #'|'
    strb w24, [x16, #60]

    mov w20, xzr

.L_chars_loop:
    cmp w20, w17
    b.ge .L_chars_done
    add x23, x9, x12
    ldrb w24, [x23, x20]
    cmp w24, #0x20
    b.lt .L_chars_dot
    cmp w24, #0x7e
    b.gt .L_chars_dot
    b .L_chars_store

.L_chars_dot:
    mov w24, #'.'
.L_chars_store:
    add w25, w20, #61
    strb w24, [x16, x25]
    add w20, w20, #1
    b .L_chars_loop

.L_chars_done:
    add w25, w17, #61
    mov w24, #'|'
    strb w24, [x16, x25]
    add w25, w25, #1
    mov w24, #'\n'
    strb w24, [x16, x25]

    add x2, x17, #63
    mov x0, #1
    mov x1, x16
    mov x8, #SYS_WRITE
    svc #0

    add x12, x12, #16
    b .L_row_loop

.L_final_offset:
    mov w19, w10
    mov w20, xzr

.L_off_final:
    lsr w21, w19, #28
    and w21, w21, #0xf
    mov x1, x16
    mov w2, w20
    mov w0, w21
    bl print_nibble
    lsl w19, w19, #4
    add w20, w20, #1
    cmp w20, #8
    b.lt .L_off_final

    mov w21, #'\n'
    strb w21, [x16, #8]

    mov x0, #1
    mov x1, x16
    mov x2, #9
    mov x8, #SYS_WRITE
    svc #0

    mov x0, xzr
    mov x8, #SYS_EXIT
    svc #0

print_nibble:
    add w0, w0, #0x30
    cmp w0, #0x3a
    b.lt 1f
    add w0, w0, #0x27
1:
    strb w0, [x1, w2]
    ret

print_hex_byte:
    lsr w3, w0, #4
    and w3, w3, #0xf
    add w3, w3, #0x30
    cmp w3, #0x3a
    b.lt 2f
    add w3, w3, #0x27
2:
    strb w3, [x1, w2]
    and w3, w0, #0xf
    add w3, w3, #0x30
    cmp w3, #0x3a
    b.lt 3f
    add w3, w3, #0x27
3:
    add w2, w2, #1
    strb w3, [x1, w2]
    ret
