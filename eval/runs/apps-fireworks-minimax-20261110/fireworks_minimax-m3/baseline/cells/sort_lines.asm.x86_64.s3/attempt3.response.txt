.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .rodata
.align 16
newline:
    .byte 0x0a

.section .bss
.align 16
buf:
    .space MAX_INPUT
.align 16
lines:
    .space MAX_LINES * 8
.align 16
key_temp:
    .space 8

.section .text
_start:
    lea r12, [rip + buf]
    mov rbp, r12
    mov r14d, MAX_INPUT

.Lread_loop:
    mov eax, SYS_READ
    xor edi, edi
    mov rsi, rbp
    mov edx, r14d
    syscall
    test rax, rax
    js .Lexit_err
    jz .Lread_done
    add rbp, rax
    sub r14d, eax
    jnz .Lread_loop

.Lread_done:
    lea r13, [rip + buf]
    xor ebx, ebx
    lea r15, [rip + lines]

.Lparse_loop:
    cmp r13, rbp
    jae .Lparse_done
    mov rdi, r13

.Lfind_nl:
    cmp r13, rbp
    jae .Lno_nl
    cmp byte ptr [r13], 0x0a
    je .Lhave_nl
    inc r13
    jmp .Lfind_nl

.Lhave_nl:
    mov rax, rdi
    sub rax, r12
    mov rsi, r13
    sub rsi, rdi
    lea rcx, [r15 + rbx*8]
    mov dword ptr [rcx], eax
    mov dword ptr [rcx + 4], esi
    inc r13
    inc ebx
    cmp ebx, MAX_LINES
    jge .Lparse_done
    jmp .Lparse_loop

.Lno_nl:
    mov rax, rdi
    sub rax, r12
    sub r13, rdi
    lea rcx, [r15 + rbx*8]
    mov dword ptr [rcx], eax
    mov dword ptr [rcx + 4], r13d
    inc ebx

.Lparse_done:
    cmp ebx, 2
    jl .Lsort_done

    mov r14d, 1

.Louter_loop:
    cmp r14d, ebx
    jge .Lsort_done

    mov rax, r14
    lea rdi, [r15 + rax*8]
    lea rsi, [rip + key_temp]
    mov ecx, dword ptr [rdi]
    mov dword ptr [rsi], ecx
    mov ecx, dword ptr [rdi + 4]
    mov dword ptr [rsi + 4], ecx

    mov r13d, r14d

.Linner_loop:
    test r13d, r13d
    jz .Linsert

    mov rax, r13
    dec rax
    lea rdi, [r15 + rax*8]
    lea rsi, [rip + key_temp]

    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    call compare
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx

    test eax, eax
    jle .Linsert

    mov rax, r13
    dec rax
    lea rdi, [r15 + rax*8]
    lea rsi, [r15 + r13*8]
    mov ecx, dword ptr [rdi]
    mov dword ptr [rsi], ecx
    mov ecx, dword ptr [rdi + 4]
    mov dword ptr [rsi + 4], ecx

    dec r13d
    jmp .Linner_loop

.Linsert:
    mov rax, r13
    lea rdi, [r15 + rax*8]
    lea rsi, [rip + key_temp]
    mov ecx, dword ptr [rsi]
    mov dword ptr [rdi], ecx
    mov ecx, dword ptr [rsi + 4]
    mov dword ptr [rdi + 4], ecx

    inc r14d
    jmp .Louter_loop

.Lsort_done:
    xor r14d, r14d

.Loutput_loop:
    cmp r14d, ebx
    jge .Lexit_ok

    lea rdi, [r15 + r14*8]
    mov eax, dword ptr [rdi]
    mov ecx, dword ptr [rdi + 4]

    lea rsi, [rip + buf]
    add rsi, rax

    mov eax, SYS_WRITE
    mov edi, STDOUT
    mov rdx, rcx
    syscall

    lea rsi, [rip + newline]
    mov eax, SYS_WRITE
    mov edi, STDOUT
    mov edx, 1
    syscall

    inc r14d
    jmp .Loutput_loop

.Lexit_ok:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

.Lexit_err:
    mov eax, SYS_EXIT
    mov edi, 1
    syscall

compare:
    push rbx
    push rbp
    push r12
    push r13

    mov r12d, dword ptr [rdi]
    mov r13d, dword ptr [rdi + 4]
    mov ebx, dword ptr [rsi]
    mov ebp, dword ptr [rsi + 4]

    lea rdi, [rip + buf]
    lea rsi, [rdi + r12]
    add rdi, rbx

    mov eax, r13d
    cmp eax, ebp
    jle .Lmin_set
    mov eax, ebp
.Lmin_set:
    test eax, eax
    jz .Lcmp_len

.Lcmp_loop:
    movzx ecx, byte ptr [rsi]
    movzx edx, byte ptr [rdi]
    cmp ecx, edx
    jne .Lcmp_diff
    inc rsi
    inc rdi
    dec eax
    jnz .Lcmp_loop

.Lcmp_len:
    mov eax, r13d
    sub eax, ebp
    jmp .Lcmp_done

.Lcmp_diff:
    sub ecx, edx
    mov eax, ecx

.Lcmp_done:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
