// base64: Base64-encode standard input using RFC 4648 standard alphabet.
// Writes to stdout in lines of exactly 76 chars + '\n'.
// Empty input produces no output. Exit status 0.

_start:
    adrp x11, input_buf
    add  x11, x11, :lo12:input_buf
    mov  x10, #0

read_loop:
    cmp  x10, #49152
    b.ge done_reading
    mov  x0,  #0
    add  x1,  x11, x10
    mov  x2,  #49152
    sub  x2,  x2,  x10
    mov  x8,  #63
    svc  #0
    cmp  x0,  #0
    b.le done_reading
    add  x10, x10, x0
    b    read_loop

done_reading:
    cmp  x10, #0
    b.eq do_exit

    adrp x16, alphabet
    add  x16, x16, :lo12:alphabet
    adrp x17, output_buf
    add  x17, x17, :lo12:output_buf

    mov  x12, #0
    mov  x13, #0
    b    encode_loop

encode_loop:
    sub  x14, x10, x12
    cmp  x14, #3
    b.lo handle_partial

    ldrb w0, [x11, x12]
    ldrb w1, [x11, x12, #1]
