.section .text
.globl _start

_start:
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    mov rdx, 49152
    xor r9, r9
.read_more:
    mov rax, 0
    lea rsi, [rip + input_buf]
    add rsi, r9
    mov rdx, 49152
    sub rdx, r9
    syscall
    test rax, rax
    js .bad_exit
    jz .read_done
    add r9, rax
    cmp r9, 49152
    jb .read_more
.read_done:
    test r9, r9
    jz .no_output

    lea r12, [rip + output_buf]
    lea r14, [rip + input_buf]
    lea r15, [rip + alphabet]
    xor r13, r13
    xor rbx, rbx
    xor rbp, rbp

.enc_loop:
    lea rax, [rbx + 3]
    cmp rax, r9
    ja .enc_rem

    movzx ecx, byte ptr [r14 + rbx]
    movzx edx, byte ptr [r14 + rbx + 1]
    movzx esi, byte ptr [r14 + rbx + 2]

    mov edi, ecx
    shr edi, 2
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13], al

    mov edi, ecx
    shl edi, 4
    mov eax, edx
    shr eax, 4
    or edi, eax
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 1], al

    mov edi, edx
    shl edi, 2
    mov eax, esi
    shr eax, 6
    or edi, eax
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 2], al

    mov edi, esi
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 3], al

    add r13, 4
    add rbx, 3
    add rbp, 4
    cmp rbp, 76
    je .wrap
    jmp .enc_loop

.wrap:
    mov byte ptr [r12 + r13], 0x0a
    inc r13
    xor rbp, rbp
    jmp .enc_loop

.enc_rem:
    mov rax, r9
    sub rax, rbx
    test rax, rax
    jz .maybe_final_nl
    cmp rax, 1
    je .rem1
    jmp .rem2

.rem1:
    movzx ecx, byte ptr [r14 + rbx]
    mov edi, ecx
    shr edi, 2
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13], al

    mov edi, ecx
    shl edi, 4
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 1], al

    mov byte ptr [r12 + r13 + 2], 0x3d
    mov byte ptr [r12 + r13 + 3], 0x3d
    add r13, 4
    add rbp, 4
    cmp rbp, 76
    je .rem_wrap
    jmp .maybe_final_nl

.rem2:
    movzx ecx, byte ptr [r14 + rbx]
    movzx edx, byte ptr [r14 + rbx + 1]
    mov edi, ecx
    shr edi, 2
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13], al

    mov edi, ecx
    shl edi, 4
    mov eax, edx
    shr eax, 4
    or edi, eax
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 1], al

    mov edi, edx
    shl edi, 2
    and edi, 0x3F
    movzx eax, byte ptr [r15 + rdi]
    mov byte ptr [r12 + r13 + 2], al

    mov byte ptr [r12 + r13 + 3], 0x3d
    add r13, 4
    add rbp, 4
    cmp rbp, 76
    je .rem_wrap
    jmp .maybe_final_nl

.rem_wrap:
    mov byte ptr [r12 + r13], 0x0a
    inc r13
    xor rbp, rbp

.maybe_final_nl:
    test rbp, rbp
    jz .do_write
    mov byte ptr [r12 + r13], 0x0a
    inc r13

.do_write:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r13
    syscall

.no_output:
    mov rax, 60
    xor rdi, rdi
    syscall

.bad_exit:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
input_buf:
.zero 49152
.balign 16
output_buf:
.zero 66560
