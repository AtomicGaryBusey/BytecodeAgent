.section .rodata
error_msg: .ascii "error\n"
error_len = . - error_msg

min_str: .ascii "-9223372036854775808\n"
min_len = . - min_str

zero_str: .ascii "0\n"
zero_len = . - zero_str

.section .bss
.align 16
input_buf: .space 65536

.align 8
stack: .zero 256 * 8

.align 8
stack_depth: .zero 8

.align 1
error_flag: .zero 1

.align 8
out_buf: .space 32

.section .text
.globl _start
.intel_syntax noprefix

_start:
    # r12 = total bytes read
    xor r12, r12
.read_loop:
    mov rax, 0              # sys_read
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz .eof_input
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.eof_input:
    # r12 = total length, r13 used for later (rename)
    mov r13, r12               # total length
    test r13, r13
    jz .exit_program           # empty input: no lines

    lea rbp, [rip + input_buf] # base pointer for line scanning
    xor r12, r12               # current scanning offset

.parse_loop:
    mov r15, r12               # start of line
.find_nl:
    cmp r12, r13
    jae .last_line
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x0A
    je .found_nl
    inc r12
    jmp .find_nl
.found_nl:
    # r12 points to newline; evaluate line [r15, r12)
    push rbp
    push r12
    mov rdi, r15
    mov rsi, r12
    call evaluate_line
    pop r12
    pop rbp
    inc r12                     # skip newline
    cmp r12, r13
    jae .exit_program
    jmp .parse_loop
.last_line:
    # final line (may end without newline)
    push rbp
    push r12
    mov rdi, r15
    mov rsi, r13
    call evaluate_line
    pop r12
    pop rbp
.exit_program:
    mov rax, 60                # sys_exit
    xor rdi, rdi
    syscall

###############################################################################
# evaluate_line: rdi = start offset, rsi = end offset (exclusive)
# Uses rbp as base pointer to input buffer.
###############################################################################
evaluate_line:
    # set up base pointer
    lea rbp, [rip + input_buf]
    mov r12, rdi               # current offset
    mov r13, rsi               # end offset

    # reset stack and error flag
    mov qword ptr [stack_depth], 0
    mov byte ptr [error_flag], 0

    jmp .token_loop_start

.token_loop_start:
    cmp r12, r13
    jae .end_eval

    # skip separators (space 0x20, tab 0x09)
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x20
    je .skip_sep
    cmp al, 0x09
    je .skip_sep

    # token start
    mov r14, r12
    inc r12
.scan_tok:
    cmp r12, r13
    jae .end_tok
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x20
    je .end_tok
    cmp al, 0x09
    je .end_tok
    inc r12
    jmp .scan_tok
.end_tok:
    # token length = r12 - r14
    mov r15, r12
    sub r15, r14
    # rsi = token address, rdx = length
    lea rsi, [rbp + r14]
    mov rdx, r15
    call process_token
    cmp byte ptr [error_flag], 0
    jne .end_eval
    jmp .token_loop_start

.skip_sep:
    inc r12
    jmp .token_loop_start

.end_eval:
    cmp byte ptr [error_flag], 0
    jne .print_err

    cmp qword ptr [stack_depth], 1
    jne .print_err

    # success: output result
    mov rax, [stack]
    call print_number
    ret

.print_err:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall
    ret

###############################################################################
# process_token: rsi = token bytes, rdx = length
###############################################################################
process_token:
    mov rcx, rdx

    # check for single-character operator
    cmp rcx, 1
    jne .try_literal
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .handle_operator
    cmp al, '-'
    je .handle_operator
    cmp al, '*'
    je .handle_operator
    cmp al, '/'
    je .handle_operator
    cmp al, '%'
    je .handle_operator

.try_literal:
    # check optional leading '-'
    xor r8, r8                # sign: 0 = positive, 1 = negative
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .no_sign
    mov r8, 1
    add rsi, 1
    dec rcx
.no_sign:
    # rcx = number of digits, must be 1..18
    cmp rcx, 0
    je .invalid_literal
    cmp rcx, 18
    ja .invalid_literal

    xor r9, r9                # accumulator (magnitude)
    xor r10, r10              # temporary
.parse_digit:
    test rcx, rcx
    jz .digits_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .invalid_literal
    cmp al, '9'
    ja .invalid_literal
    sub al, '0'
    movzx r10d, al
    imul r9, r9, 10
    add r9, r10
    inc rsi
    dec rcx
    jmp .parse_digit

.digits_done:
    # apply sign
    test r8, r8
    jz .value_ready
    mov rax, r9
    neg rax
    jmp .value_ready

.value_ready:
    # push value onto stack
    mov r11, qword ptr [stack_depth]
    mov qword ptr [stack + r11*8], rax
    inc qword ptr [stack_depth]
    ret

.invalid_literal:
    mov byte ptr [error_flag], 1
    ret

.handle_operator:
    # operator character is in al
    mov r11, qword ptr [stack_depth]
    cmp r11, 2
    jb .op_underflow

    # pop b (top) and a (second)
    mov r12, r11
    dec r12
    mov rbx, qword ptr [stack + r12*8]    # b
    mov r12, r11
    sub r12, 2
    mov rax, qword ptr [stack + r12*8]    # a

    sub qword ptr [stack_depth], 2

    cmp al, '+'
    je .op_add
    cmp al, '-'
    je .op_sub
    cmp al, '*'
    je .op_mul
    cmp al, '/'
    je .op_div
    cmp al, '%'
    je .op_mod
    jmp .op_underflow

.op_add:
    add rax, rbx
    jmp .op_done
.op_sub:
    sub rax, rbx
    jmp .op_done
.op_mul:
    imul rax, rbx
    jmp .op_done

.op_div:
    test rbx, rbx
    jz .op_div_by_zero
    cmp rbx, -1
    jne .op_div_normal
    cmp rax, 0x8000000000000000
    je .op_div_overflow
.op_div_normal:
    cqo
    idiv rbx
    jmp .op_done

.op_div_by_zero:
    mov byte ptr [error_flag], 1
    ret

.op_div_overflow:
    mov byte ptr [error_flag], 1
    ret

.op_mod:
    test rbx, rbx
    jz .op_mod_by_zero
    cmp rbx, -1
    jne .op_mod_normal
    cmp rax, 0x8000000000000000
    je .op_mod_overflow
.op_mod_normal:
    cqo
    idiv rbx
    mov rax, rdx
    jmp .op_done

.op_mod_by_zero:
    mov byte ptr [error_flag], 1
    ret

.op_mod_overflow:
    mov byte ptr [error_flag], 1
    ret

.op_underflow:
    mov byte ptr [error_flag], 1
    ret

.op_done:
    # push result
    mov r11, qword ptr [stack_depth]
    mov qword ptr [stack + r11*8], rax
    inc qword ptr [stack_depth]
    ret

###############################################################################
# print_number: rax = value to print (signed 64-bit)
# Prints decimal representation followed by newline.
###############################################################################
print_number:
    test rax, rax
    jz .print_zero
    cmp rax, 0x8000000000000000
    je .print_min

    xor r12, r12                # sign flag (0=positive,1=negative)
    mov r13, rax                # magnitude (positive)
    test r13, r13
    jns .pos_num
    mov r12, 1
    neg r13
.pos_num:
    # r13 = magnitude > 0
    lea rbx, [rip + out_buf]
    add rbx, 24                 # points past max digit area
    mov r14, rbx
.digit_loop:
    xor rdx, rdx
    mov rax, r13
    mov rcx, 10
    div rcx                     # quotient rax, remainder rdx
    add dl, '0'
    dec r14
    mov byte ptr [r14], dl
    mov r13, rax
    test r13, r13
    jnz .digit_loop

    # sign if needed
    test r12, r12
    jz .no_sign_extend
    dec r14
    mov byte ptr [r14], '-'
.no_sign_extend:
    # length = rbx - r14
    mov rcx, rbx
    sub rcx, r14
    # append newline
    lea rdi, [r14 + rcx]
    mov byte ptr [rdi], 0x0A
    inc rcx
    # write
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, rcx
    syscall
    ret

.print_zero:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + zero_str]
    mov rdx, zero_len
    syscall
    ret

.print_min:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_str]
    mov rdx, min_len
    syscall
    ret
