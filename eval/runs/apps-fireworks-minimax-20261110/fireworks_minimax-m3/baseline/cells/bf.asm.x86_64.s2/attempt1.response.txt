.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ JMP_TABLE_SIZE, MAX_INPUT * 2
.equ BUF_SIZE, 4096

.section .bss
input_buf:     .space MAX_INPUT
.align 16
tape:          .space TAPE_SIZE
.align 16
jmp_table:     .space JMP_TABLE_SIZE
.align 16

.section .text
.globl _start
_start:
    # Phase 1: read all of stdin into input_buf
    xor r15, r15
    mov r14, MAX_INPUT
read_loop:
    test r14, r14
    jz read_done
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, r14
    syscall
    cmp rax, 0
    jle read_done
    add r15, rax
    sub r14, rax
    jmp read_loop

read_done:
    # r15 = total bytes read
    # Phase 2: find first '!' separator
    xor r12, r12
    lea r13, [rip + input_buf]
bang_loop:
    cmp r12, r15
    jge no_bang
    mov al, byte ptr [r13 + r12]
    cmp al, '!'
    je bang_found
    inc r12
    jmp bang_loop

bang_found:
    mov r10, r12
    lea r11, [r12 + 1]
    jmp phase3

no_bang:
    mov r10, r15
    mov r11, r15

phase3:
    # Bracket validation + jump table build
    sub rsp, 2048
    xor r12, r12
    xor rbx, rbx
    lea r14, [rip + jmp_table]
scan_loop:
    cmp rbx, r10
    jge scan_end
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '['
    je handle_open
    cmp al, ']'
    je handle_close
    jmp scan_advance

handle_open:
    cmp r12, 1000
    jge bracket_error
    mov word ptr [rsp + r12*2], bx
    inc r12
    jmp scan_advance

handle_close:
    test r12, r12
    jz bracket_error
    dec r12
    mov dx, word ptr [rsp + r12*2]
    mov word ptr [r14 + rbx*2], dx
    mov word ptr [r14 + rdx*2], bx
    jmp scan_advance

scan_advance:
    inc rbx
    jmp scan_loop

scan_end:
    test r12, r12
    jnz bracket_error
    add rsp, 2048
    jmp exec

bracket_error:
    add rsp, 2048
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall

exec:
    lea r12, [rip + input_buf]
    lea r13, [rip + tape]
    lea r14, [rip + jmp_table]
    mov rax, r15
    sub rax, r11
    mov r8, rax
    lea r15, [r12 + r11]
    sub rsp, BUF_SIZE
    xor rbp, rbp
    xor rbx, rbx
    xor rdi, rdi
    xor r9, r9

exec_loop:
    cmp rbx, r10
    jge exec_end
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '>'
    je do_right
    cmp al, '<'
    je do_left
    cmp al, '+'
    je do_inc
    cmp al, '-'
    je do_dec
    cmp al, '.'
    je do_out
    cmp al, ','
    je do_in
    cmp al, '['
    je do_open
    cmp al, ']'
    je do_close
    inc rbx
    jmp exec_loop

do_right:
    inc rdi
    jmp exec_advance

do_left:
    dec rdi
    jmp exec_advance

do_inc:
    inc byte ptr [r13 + rdi]
    jmp exec_advance

do_dec:
    dec byte ptr [r13 + rdi]
    jmp exec_advance

do_out:
    mov al, byte ptr [r13 + rdi]
    mov byte ptr [rsp + rbp], al
    inc rbp
    cmp rbp, BUF_SIZE
    je flush_out
    jmp exec_advance

do_in:
    cmp r9, r8
    jge do_in_zero
    movzx eax, byte ptr [r15 + r9]
    mov byte ptr [r13 + rdi], al
    inc r9
    jmp exec_advance

do_in_zero:
    mov byte ptr [r13 + rdi], 0
    jmp exec_advance

do_open:
    cmp byte ptr [r13 + rdi], 0
    jne exec_advance
    movzx ebx, word ptr [r14 + rbx*2]
    inc rbx
    jmp exec_loop

do_close:
    cmp byte ptr [r13 + rdi], 0
    je exec_advance
    movzx ebx, word ptr [r14 + rbx*2]
    inc rbx
    jmp exec_loop

exec_advance:
    inc rbx
    jmp exec_loop

flush_out:
    mov rax, SYS_WRITE
    mov rdi, 1
    mov rsi, rsp
    mov rdx, BUF_SIZE
    syscall
    xor rbp, rbp
    jmp exec_advance

exec_end:
    test rbp, rbp
    jz clean_exit
    mov rax, SYS_WRITE
    mov rdi, 1
    mov rsi, rsp
    mov rdx, rbp
    syscall

clean_exit:
    add rsp, BUF_SIZE
    mov rax, SYS_EXIT
    mov rdi, 0
    syscall
