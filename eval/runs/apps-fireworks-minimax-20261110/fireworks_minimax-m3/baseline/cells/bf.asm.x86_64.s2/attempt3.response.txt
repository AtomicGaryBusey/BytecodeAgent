```asm
.globl _start

.section .bss
buf: .zero 65536
tape: .zero 30000
bracket_match: .zero 262144
stack: .zero 4000
out_buf: .zero 4096

.section .text
_start:
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    xor eax, eax
    syscall
    mov r12, rax

.read_loop:
    cmp r12, 65536
    jae .read_done
    mov rdx, 65536
    sub rdx, r12
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    xor eax, eax
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:

    xor ecx, ecx
    lea rdi, [rip + buf]
.find_sep:
    cmp ecx, r12
    jae .no_sep
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x21
    je .found_sep
    inc ecx
    jmp .find_sep
.no_sep:
    mov r13, r12
    mov r14, r12
    mov r15, r12
    jmp .sep_done
.found_sep:
    mov r13, rcx
    lea r14, [rcx + 1]
    mov r15, r12
.sep_done:

    xor r9d, r9d
    xor r8d, r8d
    lea rdi, [rip + buf]
.check_loop:
    cmp r8d, r13d
    jae .check_end
    movzx eax, byte ptr [rdi + r8]
    cmp al, 0x5b
    je .open_bracket
    cmp al, 0x5d
    je .close_bracket
    inc r8d
    jmp .check_loop
.open_bracket:
    cmp r9d, 1000
    jae .unbalanced
    lea rsi, [rip + stack]
    mov dword ptr [rsi + r9*4], r8d
    inc r9d
    inc r8d
    jmp .check_loop
.close_bracket:
    test r9d, r9d
    jz .unbalanced
    dec r9d
    lea rsi, [rip + stack]
    mov edx, dword ptr [rsi + r9*4]
    lea rsi, [rip + bracket_match]
    mov dword ptr [rsi + rdx*4], r8d
    mov dword ptr [rsi + r8*4], edx
    inc r8d
    jmp .check_loop
.check_end:
    test r9d, r9d
    jnz .unbalanced
    jmp .execute
.unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.execute:
    xor ebx, ebx
    xor r12d, r12d
    xor ecx, ecx
    xor r10d, r10d

.exec_loop:
    cmp ecx, r13d
    jae .exec_end

    lea rdi, [rip + buf]
    movzx eax, byte ptr [rdi + rcx]

    cmp al, 0x3e
    je .cmd_gt
    cmp al, 0x3c
    je .cmd_lt
    cmp al, 0x2b
    je .cmd_plus
    cmp al, 0x2d
    je .cmd_minus
    cmp al, 0x2e
    je .cmd_dot
    cmp al, 0x2c
    je .cmd_comma
    cmp al, 0x5b
    je .cmd_lbra
    cmp al, 0x5d
    je .cmd_rbra

    inc ecx
    jmp .exec_loop

.cmd_gt:
    inc ebx
    inc ecx
    jmp .exec_loop
.cmd_lt:
    dec ebx
    inc ecx
    jmp .exec_loop
.cmd_plus:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    inc al
    mov byte ptr [rdi + rbx], al
    inc ecx
    jmp .exec_loop
.cmd_minus:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    dec al
    mov byte ptr [rdi + rbx], al
    inc ecx
    jmp .exec_loop
.cmd_dot:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + r10], al
    inc r10d
    cmp r10d, 4096
    jb .cmd_dot_done
    push rcx
    push rbx
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, r10d
    mov eax, 1
    syscall
    pop rbx
    pop rcx
    xor r10d, r10d
.cmd_dot_done:
    inc ecx
    jmp .exec_loop

.cmd_comma:
    mov rax, r15
    sub rax, r14
    cmp r12, rax
    jae .input_exhausted
    mov rax, r14
    add rax, r12
    lea rdi, [rip + buf]
    movzx eax, byte ptr [rdi + rax]
    inc r12
    jmp .input_done
.input_exhausted:
    xor eax, eax
.input_done:
    lea rdi, [rip + tape]
    mov byte ptr [rdi + rbx], al
    inc ecx
    jmp .exec_loop

.cmd_lbra:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jnz .skip_lbra
    lea rdi, [rip + bracket_match]
    mov ecx, dword ptr [rdi + rcx*4]
    inc ecx
    jmp .exec_loop
.skip_lbra:
    inc ecx
    jmp .exec_loop

.cmd_rbra:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jz .skip_rbra
    lea rdi, [rip + bracket_match]
    mov ecx, dword ptr [rdi + rcx*4]
    inc ecx
    jmp .exec_loop
.skip_rbra:
    inc ecx
    jmp .exec_loop

.exec_end:
    test r10d, r10d
    jz .exit_ok
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, r10d
    mov eax, 1
    syscall
.exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
```
