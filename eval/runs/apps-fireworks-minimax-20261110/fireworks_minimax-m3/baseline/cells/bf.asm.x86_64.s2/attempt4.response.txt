.globl _start

.section .bss
buf: .zero 65536
tape: .zero 30000
bracket_match: .zero 131072
bf_stack: .zero 2000
out_buf: .zero 65536

.section .text
_start:
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jle .read_done
    mov r12, rax

.read_loop:
    cmp r12, 65536
    jae .read_done
    mov rdx, 65536
    sub rdx, r12
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    xor eax, eax
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:

    xor rcx, rcx
    lea rdi, [rip + buf]
.find_sep:
    cmp rcx, r12
    jae .no_sep
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x21
    je .found_sep
    inc rcx
    jmp .find_sep
.no_sep:
    mov r13, r12
    mov r14, r12
    mov r15, r12
    jmp .sep_done
.found_sep:
    mov r13, rcx
    lea r14, [rcx + 1]
    mov r15, r12
.sep_done:

    xor r8, r8
    xor r9, r9
    lea rdi, [rip + buf]
.check_loop:
    cmp r8, r13
    jae .check_end
    movzx eax, byte ptr [rdi + r8]
    cmp al, 0x5b
    je .open_bracket
    cmp al, 0x5d
    je .close_bracket
    inc r8
    jmp .check_loop
.open_bracket:
    cmp r9, 1000
    jae .unbalanced
    lea rsi, [rip + bf_stack]
    mov word ptr [rsi + r9*2], r8w
    inc r9
    inc r8
    jmp .check_loop
.close_bracket:
    test r9, r9
    jz .unbalanced
    dec r9
    lea rsi, [rip + bf_stack]
    movzx edx, word ptr [rsi + r9*2]
    lea rsi, [rip + bracket_match]
    mov word ptr [rsi + rdx*2], r8w
    mov word ptr [rsi + r8*2], dx
    inc r8
    jmp .check_loop
.check_end:
    test r9, r9
    jnz .unbalanced
    jmp .balanced
.unbalanced:
    mov edi, 1
    mov eax, 60
    syscall
.balanced:

    xor ebx, ebx
    xor rcx, rcx
    xor r12, r12
    xor r10, r10

.exec_loop:
    cmp rcx, r13
    jae .exec_end

    lea rdi, [rip + buf]
    movzx eax, byte ptr [rdi + rcx]

    cmp al, 0x3e
    je .cmd_gt
    cmp al, 0x3c
    je .cmd_lt
    cmp al, 0x2b
    je .cmd_plus
    cmp al, 0x2d
    je .cmd_minus
    cmp al, 0x2e
    je .cmd_dot
    cmp al, 0x2c
    je .cmd_comma
    cmp al, 0x5b
    je .cmd_lbra
    cmp al, 0x5d
    je .cmd_rbra

    inc rcx
    jmp .exec_loop

.cmd_gt:
    inc ebx
    inc rcx
    jmp .exec_loop
.cmd_lt:
    dec ebx
    inc rcx
    jmp .exec_loop
.cmd_plus:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    inc al
    mov byte ptr [rdi + rbx], al
    inc rcx
    jmp .exec_loop
.cmd_minus:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    dec al
    mov byte ptr [rdi + rbx], al
    inc rcx
    jmp .exec_loop
.cmd_dot:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + r10], al
    inc r10
    inc rcx
    cmp r10, 65536
    jb .exec_loop
    push rcx
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, 65536
    mov eax, 1
    syscall
    pop rcx
    xor r10, r10
    jmp .exec_loop

.cmd_comma:
    mov rax, r15
    sub rax, r14
    cmp r12, rax
    jae .input_exhausted
    mov rax, r14
    add rax, r12
    lea rdi, [rip + buf]
    movzx eax, byte ptr [rdi + rax]
    inc r12
    jmp .input_done
.input_exhausted:
    xor eax, eax
.input_done:
    lea rdi, [rip + tape]
    mov byte ptr [rdi + rbx], al
    inc rcx
    jmp .exec_loop

.cmd_lbra:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jnz .skip_lbra
    lea rdi, [rip + bracket_match]
    movzx ecx, word ptr [rdi + rcx*2]
    inc ecx
    jmp .exec_loop
.skip_lbra:
    inc rcx
    jmp .exec_loop

.cmd_rbra:
    lea rdi, [rip + tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jz .skip_rbra
    lea rdi, [rip + bracket_match]
    movzx ecx, word ptr [rdi + rcx*2]
    inc ecx
    jmp .exec_loop
.skip_rbra:
    inc rcx
    jmp .exec_loop

.exec_end:
    test r10, r10
    jz .exit_ok
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, r10d
    mov eax, 1
    syscall
.exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
