.section .text
.globl _start
_start:
    xor r12, r12
    lea rbx, [rip + input_buf]
    lea r8, [rip + hex_chars]
.read_loop:
    mov edx, 65536
    sub rdx, r12
    jle .read_done
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    xor r13, r13
.row_loop:
    cmp r13, r12
    jae .row_done
    mov rcx, r12
    sub rcx, r13
    cmp rcx, 16
    jbe .set_n
    mov rcx, 16
.set_n:
    mov r15, rcx
    lea r10, [rip + outbuf]
    xor r11, r11
    mov rax, r13
    mov rcx, 8
.format_off:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10 + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz .format_off
    add r11, 8
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11
    xor r9, r9
.hex_loop:
    cmp r9, 15
    ja .hex_done
    cmp r9, r15
    jae .skip_hex
    mov rsi, rbx
    add rsi, r13
    add rsi, r9
    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 4
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [r10 + r11], dl
    inc r11
    mov ecx, eax
    and ecx, 0xf
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [r10 + r11], dl
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11
    cmp r9, 7
    jne .after_extra1
    mov byte ptr [r10 + r11], 0x20
    inc r11
.after_extra1:
    inc r9
    jmp .hex_loop
.skip_hex:
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11
    cmp r9, 7
    jne .after_extra2
    mov byte ptr [r10 + r11], 0x20
    inc r11
.after_extra2:
    inc r9
    jmp .hex_loop
.hex_done:
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], '|'
    inc r11
    xor r9, r9
.ascii_loop:
    cmp r9, r15
    jae .ascii_done
    mov rsi, rbx
    add rsi, r13
    add rsi, r9
    movzx eax, byte ptr [rsi]
    cmp al, 0x7e
    ja .nonprint
    cmp al, 0x20
    jb .nonprint
    jmp .write_char
.nonprint:
    mov al, '.'
.write_char:
    mov byte ptr [r10 + r11], al
    inc r11
    inc r9
    jmp .ascii_loop
.ascii_done:
    mov byte ptr [r10 + r11], '|'
    inc r11
    mov byte ptr [r10 + r11], 0x0a
    inc r11
    mov edi, 1
    mov rsi, r10
    mov rdx, r11
    mov eax, 1
    syscall
    add r13, 16
    jmp .row_loop
.row_done:
    lea r10, [rip + outbuf]
    xor r11, r11
    mov rax, r12
    mov rcx, 8
.format_final:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10 + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz .format_final
    add r11, 8
    mov byte ptr [r10 + r11], 0x0a
    add r11, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r11
    mov eax, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .space 65536
.align 16
outbuf: .space 80
