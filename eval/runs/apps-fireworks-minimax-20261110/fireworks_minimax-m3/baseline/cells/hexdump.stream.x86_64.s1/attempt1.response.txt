.section .text
.globl _start

_start:
    lea rbx, [rip + buffer]
    lea r14, [rip + hex_chars]
    xor r12, r12

.read_loop:
    mov rdi, rbx
    add rdi, r12
    mov rsi, rdi
    mov rdx, 65536
    sub rdx, r12
    jle .read_done
    xor rax, rax
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    xor r13, r13

.row_loop:
    cmp r13, r12
    jae .final_offset

    mov rcx, r12
    sub rcx, r13
    cmp rcx, 16
    jbe 1f
    mov rcx, 16
1:
    mov r15, rcx

    lea r10, [rip + outbuf]

    mov rax, r13
    mov r8, 7
.format_off:
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r14 + rdx]
    mov byte ptr [r10 + r8], dl
    shr rax, 4
    dec r8
    jns .format_off
    add r10, 8

    mov word ptr [r10], 0x2020
    add r10, 2

    xor r9, r9
.fmt_first:
    cmp r9, r15
    jb .fmt_f_byte
    mov dword ptr [r10], 0x20202020
    jmp .fmt_f_done
.fmt_f_byte:
    movzx eax, byte ptr [rbx + r13 + r9]
