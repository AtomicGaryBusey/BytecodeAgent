.section .text
.globl _start

_start:
    # input buffer pointer
    lea rbx, [rip + input_buf]
    # hex table pointer
    lea r8, [rip + hex_chars]
    # total length = 0
    xor r12, r12

.read_loop:
    # remaining bytes in buffer
    mov edx, 65536
    sub rdx, r12
    jle .read_done
    # buffer address = rbx + r12
    lea rsi, [rbx + r12]
    xor edi, edi          # fd = 0 (stdin)
    xor eax, eax          # syscall read
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    xor r13, r13          # current row offset

.row_loop:
    cmp r13, r12
    jae .row_done

    # n = min(16, r12 - r13)
    mov rcx, r12
    sub rcx, r13
    cmp rcx, 16
    jbe .set_n
    mov rcx, 16
.set_n:
    mov r15, rcx          # n

    # prepare output buffer
    lea r10, [rip + outbuf]
    xor r11, r11          # current index

    # format offset (8 hex digits) into outbuf[0..7]
    mov rax, r13
    mov rcx, 8
.format_off:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10 + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz .format_off

    # two spaces after offset
    add r11, 8
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11

    # hex column: positions 10.. (16 * 3 chars, plus extra after 7th)
    xor r9, r9            # i = 0
.hex_loop:
    cmp r9, 15
    ja .hex_done

    cmp r9, r15
    jae .skip_hex

    # present byte
    mov rsi, rbx
    add rsi, r13
    add rsi, r9
    movzx edx, byte ptr [rsi]

    # high nibble
    mov rcx, rdx
    shr cl, 4
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [r10 + r11], dl
    inc r11

    # low nibble
    mov rcx, rdx
    and rcx, 0xf
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [r10 + r11], dl
    inc r11

    # space after byte
    mov byte ptr [r10 + r11], 0x20
    inc r11

    # extra space after position 7
    cmp r9, 7
    jne .after_extra1
    mov byte ptr [r10 + r11], 0x20
    inc r11
.after_extra1:
    inc r9
    jmp .hex_loop

.skip_hex:
    # missing byte: three spaces
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], 0x20
    inc r11

    cmp r9, 7
    jne .after_extra2
    mov byte ptr [r10 + r11], 0x20
    inc r11
.after_extra2:
    inc r9
    jmp .hex_loop

.hex_done:
    # space + '|' before ascii column
    mov byte ptr [r10 + r11], 0x20
    inc r11
    mov byte ptr [r10 + r11], '|'
    inc r11

    # ascii column: i = 0..n-1
    xor r9, r9
.ascii_loop:
    cmp r9, r15
    jae .ascii_done

    mov rsi, rbx
    add rsi, r13
    add rsi, r9
    movzx edx, byte ptr [rsi]

    cmp edx, 0x20
    jb .nonprint
    cmp edx, 0x7E
    ja .nonprint
    jmp .write_char
.nonprint:
    mov dl, '.'
.write_char:
    mov byte ptr [r10 + r11], dl
    inc r11
    inc r9
    jmp .ascii_loop

.ascii_done:
    # closing '|'
    mov byte ptr [r10 + r11], '|'
    inc r11
    # newline
    mov byte ptr [r10 + r11], 0x0a
    inc r11

    # write the line
    mov edi, 1
    mov rsi, r10
    mov rdx, r11
    mov eax, 1
    syscall

    # next row
    add r13, 16
    jmp .row_loop

.row_done:
    # print total length offset line
    lea r10, [rip + outbuf]
    xor r11, r11
    mov rax, r12
    mov rcx, 8
.format_final:
    mov rdx, rax
    and rdx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10 + rcx - 1], dl
    shr rax, 4
    dec rcx
    jnz .format_final
    mov byte ptr [r10 + 8], 0x0a
    mov r11, 9

    mov edi, 1
    mov rsi, r10
    mov rdx, r11
    mov eax, 1
    syscall

    # exit
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_chars: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .space 65536
.align 16
outbuf: .space 96
