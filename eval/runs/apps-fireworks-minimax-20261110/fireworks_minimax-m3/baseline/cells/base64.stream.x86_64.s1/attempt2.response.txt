.globl _start

.section .rodata
.align 16
alphabet:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:
	.zero 49152
outbuf:
	.zero 70000

.text
_start:
	# Read all input
	xor r12, r12
	mov r13, 49152
.L_read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov rdx, r13
	sub rdx, r12
	syscall
	test rax, rax
	jle .L_read_done
	add r12, rax
	cmp r12, r13
	jb .L_read_loop
.L_read_done:
	mov r14, r12
	test r14, r14
	jz .L_exit

	# r10 = output position, r11 = line char count, r12 = input index
	xor r10, r10
	xor r11, r11
	xor r12, r12
	lea r9, [rip + alphabet]

.L_triple_loop:
	mov rax, r14
	sub rax, r12
	cmp rax, 3
	jb .L_remainder

	lea rdi, [rip + inbuf]
	movzx eax, byte ptr [rdi + r12]
	movzx ecx, byte ptr [rdi + r12 + 1]
	movzx edx, byte ptr [rdi + r12 + 2]
	shl eax, 16
	or eax, ecx
	shl eax, 8
	or eax, edx
	lea r8, [rip + outbuf]

	# char 0
	mov ebx, eax
	shr ebx, 18
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 2f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
2:

	# char 1
	mov ebx, eax
	shr ebx, 12
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 3f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
3:

	# char 2
	mov ebx, eax
	shr ebx, 6
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 4f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
4:

	# char 3
	mov ebx, eax
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 5f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
5:

	add r12, 3
	jmp .L_triple_loop

.L_remainder:
	mov rax, r14
	sub rax, r12
	jz .L_finalize
	cmp rax, 1
	je .L_one_byte

	# 2 bytes remaining
	lea rdi, [rip + inbuf]
	movzx eax, byte ptr [rdi + r12]
	movzx ecx, byte ptr [rdi + r12 + 1]
	shl eax, 16
	or eax, ecx
	shl eax, 8
	lea r8, [rip + outbuf]

	# char 0
	mov ebx, eax
	shr ebx, 18
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 6f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
6:

	# char 1
	mov ebx, eax
	shr ebx, 12
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 7f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
7:

	# char 2
	mov ebx, eax
	shr ebx, 6
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 8f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
8:

	# '='
	mov byte ptr [r8 + r10], 61
	inc r10
	inc r11
	cmp r11, 76
	jne .L_finalize
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
	jmp .L_finalize

.L_one_byte:
	lea rdi, [rip + inbuf]
	movzx eax, byte ptr [rdi + r12]
	shl eax, 16
	lea r8, [rip + outbuf]

	# char 0
	mov ebx, eax
	shr ebx, 18
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 10f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
10:

	# char 1
	mov ebx, eax
	shr ebx, 12
	and ebx, 0x3F
	movzx ebx, byte ptr [r9 + rbx]
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	cmp r11, 76
	jne 11f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
11:

	# '==' pad
	mov byte ptr [r8 + r10], 61
	inc r10
	inc r11
	cmp r11, 76
	jne 12f
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
12:
	mov byte ptr [r8 + r10], 61
	inc r10
	inc r11
	cmp r11, 76
	jne .L_finalize
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11

.L_finalize:
	test r11, r11
	jz .L_write_all
	lea r8, [rip + outbuf]
	mov byte ptr [r8 + r10], 10
	inc r10

.L_write_all:
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + outbuf]
	mov rdx, r10
	syscall

.L_exit:
	mov rax, 60
	xor rdi, rdi
	syscall
