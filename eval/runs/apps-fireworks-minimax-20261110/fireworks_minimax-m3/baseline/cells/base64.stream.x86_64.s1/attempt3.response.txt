.globl _start

.section .rodata
.align 16
alphabet:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:
	.zero 49152
outbuf:
	.zero 70000

.text
emit_char:
	cmp r11, 76
	jne .L_ec_skip_nl
	mov byte ptr [r8 + r10], 10
	inc r10
	xor r11, r11
.L_ec_skip_nl:
	mov byte ptr [r8 + r10], bl
	inc r10
	inc r11
	ret

_start:
	xor r12, r12
	mov r13, 49152
.L_rd_loop:
	xor eax, eax
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov rdx, r13
	sub rdx, r12
	syscall
	test rax, rax
	jle .L_rd_done
	add r12, rax
	cmp r12, r13
	jb .L_rd_loop
.L_rd_done:
	mov r15, r12
	test r15, r15
	jz .L_exit

	xor r10, r10
	xor r11, r11
	lea r8, [rip + outbuf]
	lea r9, [rip + alphabet]
	lea r13, [rip + inbuf]
	xor r12, r12

.L_main_loop:
	cmp r12, r15
	jae .L_encode_done
	mov rcx, r15
	sub rcx, r12
	cmp rcx, 3
	jb .L_encode_tail

	movzx eax, byte ptr [r13 + r12]
	movzx ecx, byte ptr [r13 + r12 + 1]
	movzx edx, byte ptr [r13 + r12 + 2]

	mov esi, eax
	shr esi, 2
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, eax
	and esi, 3
	shl esi, 4
	mov ebp, ecx
	shr ebp, 4
	or esi, ebp
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, ecx
	and esi, 0xF
	shl esi, 2
	mov ebp, edx
	shr ebp, 6
	or esi, ebp
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, edx
	and esi, 0x3F
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	add r12, 3
	jmp .L_main_loop

.L_encode_tail:
	mov rcx, r15
	sub rcx, r12
	jz .L_encode_done
	cmp rcx, 1
	je .L_tail_1

	movzx eax, byte ptr [r13 + r12]
	movzx ecx, byte ptr [r13 + r12 + 1]

	mov esi, eax
	shr esi, 2
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, eax
	and esi, 3
	shl esi, 4
	mov ebp, ecx
	shr ebp, 4
	or esi, ebp
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, ecx
	and esi, 0xF
	shl esi, 2
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov bl, 0x3D
	call emit_char
	jmp .L_encode_done

.L_tail_1:
	movzx eax, byte ptr [r13 + r12]

	mov esi, eax
	shr esi, 2
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov esi, eax
	and esi, 3
	shl esi, 4
	movzx ebx, byte ptr [r9 + rsi]
	call emit_char

	mov bl, 0x3D
	call emit_char
	call emit_char

.L_encode_done:
	cmp r11, 0
	je .L_skip_final_nl
	mov bl, 10
	call emit_char
.L_skip_final_nl:

	mov eax, 1
	mov edi, 1
	lea rsi, [rip + outbuf]
	mov edx, r10d
	syscall

.L_exit:
	mov eax, 60
	xor edi, edi
	syscall
