.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .bss
.align 4
input_buf:
    .skip 65536
.align 3
val_stack:
    .skip 8192
.align 3
num_buf:
    .skip 32

.section .rodata
err_msg:
    .ascii "error\n"
err_len = . - err_msg
nl_msg:
    .ascii "\n"
.align 3
min_str:
    .ascii "-9223372036854775808"
min_len = . - min_str

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x24, #65536

.Lread_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x24, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x20, x20, x0
    cmp x20, x24
    b.lt .Lread_loop

.Lread_done:
    add x21, x19, x20
    mov x22, x19

.Lline_loop:
    cmp x22, x21
    b.ge .Lall_done
    mov x23, x22

.Lfind_nl:
    cmp x23, x21
    b.ge .Lno_nl
    ldrb w0, [x23]
    cmp w0, #10
    b.eq .Lfound_nl
    add x23, x23, #1
    b .Lfind_nl

.Lfound_nl:
    mov x0, x22
    mov x1, x23
    bl process_line
    add x22, x23, #1
    b .Lline_loop

.Lno_nl:
    cmp x22, x21
    b.eq .Lall_done
    mov x0, x22
    mov x1, x21
    bl process_line

.Lall_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0


process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!

    mov x19, x0
    mov x20, x1
    mov x21, #0
    adrp x22, val_stack
    add x22, x22, :lo12:val_stack
    mov x9, x19

.Lpl_loop:
.Lpl_skip:
    cmp x9, x20
    b.ge .Lpl_end
    ldrb w0, [x9]
    cmp w0, #0x20
    b.eq .Lpl_ws
    cmp w0, #0x09
    b.eq .Lpl_ws
    mov x10, x9

.Lpl_tok_loop:
    cmp x9, x20
    b.ge .Lpl_tok_end
    ldrb w0, [x9]
    cmp w0, #0x20
    b.eq .Lpl_tok_end
    cmp w0, #0x09
    b.eq .Lpl_tok_end
    add x9, x9, #1
    b .Lpl_tok_loop

.Lpl_tok_end:
    mov x12, x9
    mov x0, x10
    mov x1, x9
    bl classify
    mov x9, x12
    cmp w0, #0
    b.eq .Lpl_number
    cmp w0, #1
    b.eq .Lpl_operator
    b .Lpl_err

.Lpl_ws:
    add x9, x9, #1
    b .Lpl_skip

.Lpl_number:
    lsl x11, x21, #3
    str x2, [x22, x11]
    add x21, x21, #1
    b .Lpl_loop

.Lpl_operator:
    cmp x21, #2
    b.lt .Lpl_err
    sub x21, x21, #1
    lsl x11, x21, #3
    ldr x4, [x22, x11]
    sub x21, x21, #1
    lsl x11, x21, #3
    ldr x5, [x22, x11]
    cmp w3, #'+'
    b.eq .Lpl_add
    cmp w3, #'-'
    b.eq .Lpl_sub
    cmp w3, #'*'
    b.eq .Lpl_mul
    cmp w3, #'/'
    b.eq .Lpl_div
    b .Lpl_mod

.Lpl_add:
    add x2, x5, x4
    b .Lpl_push
.Lpl_sub:
    sub x2, x5, x4
    b .Lpl_push
.Lpl_mul:
    mul x2, x5, x4
    b .Lpl_push

.Lpl_div:
    cbnz x4, .Lpl_div2
    b .Lpl_err
.Lpl_div2:
    movz x14, #0x8000, lsl #48
    cmp x5, x14
    b.ne .Lpl_div3
    cmn x4, #1
    b.eq .Lpl_err
.Lpl_div3:
    sdiv x2, x5, x4
    b .Lpl_push

.Lpl_mod:
    cbnz x4, .Lpl_mod2
    b .Lpl_err
.Lpl_mod2:
    movz x14, #0x8000, lsl #48
    cmp x5, x14
    b.ne .Lpl_mod3
    cmn x4, #1
    b.eq .Lpl_err
.Lpl_mod3:
    sdiv x13, x5, x4
    msub x2, x13, x4, x5
    b .Lpl_push

.Lpl_push:
    lsl x11, x21, #3
    str x2, [x22, x11]
    add x21, x21, #1
    b .Lpl_loop

.Lpl_end:
    cmp x21, #1
    b.ne .Lpl_err
    sub x21, x21, #1
    ldr x0, [x22, x21, lsl #3]
    bl print_int
    b .Lpl_ret

.Lpl_err:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #err_len
    mov x8, #SYS_write
    svc #0

.Lpl_ret:
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret


classify:
    sub x9, x1, x0
    cbz x9, .Lcl_err
    ldrb w10, [x0]
    cmp w10, #'+'
    b.eq .Lcl_plus_op
    cmp w10, #'-'
    b.eq .Lcl_minus_check
    cmp w10, #'*'
    b.eq .Lcl_star_op
    cmp w10, #'/'
    b.eq .Lcl_slash_op
    cmp w10, #'%'
    b.eq .Lcl_percent_op
    cmp w10, #'0'
    b.lt .Lcl_err
    cmp w10, #'9'
    b.gt .Lcl_err
    mov x16, xzr
    mov x11, x0
    b .Lcl_digit_len

.Lcl_minus_check:
    cmp x9, #1
    b.eq .Lcl_minus_op
    mov x16, #1
    mov x11, x0
    add x11, x11, #1
    sub x9, x9, #1

.Lcl_digit_len:
    cmp x9, #0
    b.eq .Lcl_err
    cmp x9, #19
    b.ge .Lcl_err
    mov x14, xzr
    mov x15, #10

.Lcl_digit_loop:
    cmp x11, x1
    b.eq .Lcl_digit_done
    ldrb w13, [x11]
    cmp w13, #'0'
    b.lt .Lcl_err
    cmp w13, #'9'
    b.gt .Lcl_err
    mul x14, x14, x15
    and w13, w13, #0xf
    add x14, x14, x13
    add x11, x11, #1
    b .Lcl_digit_loop

.Lcl_digit_done:
    cbz x16, .Lcl_positive
    neg x14, x14

.Lcl_positive:
    mov x2, x14
    mov w0, #0
    ret

.Lcl_plus_op:
    cmp x9, #1
    b.ne .Lcl_err
    mov w3, #'+'
    mov w0, #1
    ret

.Lcl_minus_op:
    mov w3, #'-'
    mov w0, #1
    ret

.Lcl_star_op:
    cmp x9, #1
    b.ne .Lcl_err
    mov w3, #'*'
    mov w0, #1
    ret

.Lcl_slash_op:
    cmp x9, #1
    b.ne .Lcl_err
    mov w3, #'/'
    mov w0, #1
    ret

.Lcl_percent_op:
    cmp x9, #1
    b.ne .Lcl_err
    mov w3, #'%'
    mov w0, #1
    ret

.Lcl_err:
    mov w0, #2
    ret


print_int:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    movz x14, #0x8000, lsl #48
    cmp x0, x14
    b.ne .Lpi_not_min

    mov x0, #1
    adrp x1, min_str
    add x1, x1, :lo12:min_str
    mov x2, #min_len
    mov x8, #SYS_write
    svc #0
    b .Lpi_newline

.Lpi_not_min:
    mov x3, xzr
    cbz x0, .Lpi_zero

    cmp x0, #0
    b.gt .Lpi_doit
    mov x3, #1
    neg x0, x0
    b .Lpi_doit

.Lpi_zero:
    adrp x4, num_buf
    add x4, x4, :lo12:num_buf
    add x5, x4, #19
    sub x5, x5, #1
    mov w8, #'0'
    strb w8, [x5]
    b .Lpi_sign

.Lpi_doit:
    adrp x4, num_buf
    add x4, x4, :lo12:num_buf
    add x5, x4, #19
    mov x6, #10

.Lpi_dloop:
    udiv x7, x0, x6
    msub x8, x7, x6, x0
    add w8, w8, #'0'
    sub x5, x5, #1
    strb w8, [x5]
    mov x0, x7
    cbnz x0, .Lpi_dloop

.Lpi_sign:
    cbz x3, .Lpi_no_sign
    sub x5, x5, #1
    mov w8, #'-'
    strb w8, [x5]

.Lpi_no_sign:
    add x4, x4, #19
    sub x2, x4, x5

    mov x0, #1
    mov x1, x5
    mov x8, #SYS_write
    svc #0

.Lpi_newline:
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    ldp x29, x30, [sp], #16
    ret
