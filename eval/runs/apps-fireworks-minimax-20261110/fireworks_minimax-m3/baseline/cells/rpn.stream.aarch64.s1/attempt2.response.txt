.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .bss
.align 4
input_buf:
    .skip 65536
.align 3
stack:
    .skip 8192
.align 3
num_buf:
    .skip 32

.section .rodata
.align 3
min_str:
    .ascii "-9223372036854775808"
min_len = . - min_str
err_msg:
    .ascii "error\n"
err_len = . - err_msg
nl_str:
    .ascii "\n"

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, xzr
    mov x21, #65536
.Lrd_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le .Lrd_done
    add x20, x20, x0
    cmp x20, x21
    b.lt .Lrd_loop
.Lrd_done:
    add x21, x19, x20
    mov x22, x19
.Lll_loop:
    cmp x22, x21
    b.ge .Lall_done
    mov x23, x22
.Lfn_loop:
    cmp x23, x21
    b.ge .Lno_nl
    ldrb w0, [x23]
    cmp w0, #10
    b.eq .Lfound_nl
    add x23, x23, #1
    b .Lfn_loop
.Lfound_nl:
    mov x0, x22
    mov x1, x23
    bl process_line
    add x22, x23, #1
    b .Lll_loop
.Lno_nl:
    cmp x22, x21
    b.eq .Lall_done
    mov x0, x22
    mov x1, x21
    bl process_line
.Lall_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0


process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!

    mov x19, x0
    mov x20, x1
    mov x21, xzr
    adrp x22, stack
    add x22, x22, :lo12:stack
    mov x9, x19

.Lpl_loop:
    cmp x9, x20
    b.eq .Lpl_end

.Lpl_skip:
    cmp x9, x20
    b.eq .Lpl_end
    ldrb w0, [x9]
    cmp w0, #0x20
    b.eq .Lpl_skip1
    cmp w0, #0x09
    b.eq .Lpl_skip1
    b .Lpl_tok_start
.Lpl_skip1:
    add x9, x9, #1
    b .Lpl_skip

.Lpl_tok_start:
    mov x10, x9
.Lpl_tok_end_loop:
    cmp x9, x20
    b.eq .Lpl_tok_end
    ldrb w0, [x9]
    cmp w0, #0x20
    b.eq .Lpl_tok_end
    cmp w0, #0x09
    b.eq .Lpl_tok_end
    add x9, x9, #1
    b .Lpl_tok_end_loop
.Lpl_tok_end:
    mov x0, x10
    mov x1, x9
    bl classify
    cmp w0, #0
    b.eq .Lpl_number
    cmp w0, #1
    b.eq .Lpl_operator
    b .Lpl_error

.Lpl_number:
    lsl x11, x21, #3
    str x2, [x22, x11]
    add x21, x21, #1
    b .Lpl_loop

.Lpl_operator:
    cmp x21, #2
    b.lt .Lpl_error
    sub x21, x21, #1
    lsl x11, x21, #3
    ldr x4, [x22, x11]
    sub x21, x21, #1
    lsl x11, x21, #3
    ldr x5, [x22, x11]

    cmp w3, #'+'
    b.eq .Lpl_add
    cmp w3, #'-'
    b.eq .Lpl_sub
    cmp w3, #'*'
    b.eq .Lpl_mul
    cmp w3, #'/'
    b.eq .Lpl_div
    b .Lpl_mod

.Lpl_add:
    add x2, x5, x4
    b .Lpl_push
.Lpl_sub:
    sub x2, x5, x4
    b .Lpl_push
.Lpl_mul:
    mul x2, x5, x4
    b .Lpl_push
.Lpl_div:
    cbnz x4, .Lpl_div2
    b .Lpl_error
.Lpl_div2:
    movz x12, #0x8000, lsl #48
    cmp x5, x12
    b.ne .Lpl_div_ok
    cmn x4, #1
    b.eq .Lpl_error
.Lpl_div_ok:
    sdiv x2, x5, x4
    b .Lpl_push
.Lpl_mod:
    cbnz x4, .Lpl_mod2
    b .Lpl_error
.Lpl_mod2:
    movz x12, #0x8000, lsl #48
    cmp x5, x12
    b.ne .Lpl_mod_ok
    cmn x4, #1
    b.eq .Lpl_error
.Lpl_mod_ok:
    sdiv x13, x5, x4
    msub x2, x13, x4, x5
.Lpl_push:
    lsl x11, x21, #3
    str x2, [x22, x11]
    add x21, x21, #1
    b .Lpl_loop

.Lpl_end:
    cmp x21, #1
    b.ne .Lpl_error
    sub x21, x21, #1
    ldr x0, [x22, x21, lsl #3]
    bl print_int
    b .Lpl_exit

.Lpl_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #err_len
    mov x8, #SYS_write
    svc #0

.Lpl_exit:
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret


classify:
    sub x9, x1, x0
    cbz x9, .Lcl_err

    ldrb w10, [x0]
    cmp w10, #'+'
    b.eq .Lcl_plus
    cmp w10, #'-'
    b.eq .Lcl_minus
    cmp w10, #'*'
    b.eq .Lcl_star
    cmp w10, #'/'
    b.eq .Lcl_slash
    cmp w10, #'%'
    b.eq .Lcl_percent
    cmp w10, #'0'
    b.lt .Lcl_err
    cmp w10, #'9'
    b.gt .Lcl_err
    b .Lcl_digit

.Lcl_plus:
    cmp x9, #1
    b.ne .Lcl_err
    mov w3, #'+'
    mov w0, #1
    ret

.Lcl_minus:
    cmp x9, #1
    b.ne .Lcl_minus_num
    mov w3, #'-'
    mov w0, #1
    ret
.Lcl_minus_num:
    cmp x9, #20
    b.ge .Lcl_err
    add x11, x0, #1
    mov x12, xzr
    mov x13, xzr
.Lcl_neg_loop:
    cmp x11, x1
    b.eq .Lcl_neg_done
    ldrb w14, [x11]
    cmp w14, #'0'
    b.lt .Lcl_err
    cmp w14, #'9'
    b.gt .Lcl_err
    mov x15, #10
    mul x12, x12, x15
    sub x14, w14, #'0'
