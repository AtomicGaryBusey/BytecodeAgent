```asm
.globl _start

.section .rodata
.balign 16
whitespace_table:
    .zero 9
    .byte 1, 1, 1, 1, 1
    .zero 18
    .byte 1
    .zero 223
space_byte:
    .byte 0x20
newline_byte:
    .byte 0x0A

.section .bss
.balign 8
byte_count:
    .zero 8
line_count:
    .zero 8
word_count:
    .zero 8
in_word:
    .zero 1
buffer:
    .zero 4096

.section .text

_start:
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 4096
    syscall
    test rax, rax
    jle .done_reading

    mov rcx, rax
    xor r15, r15

.process_byte:
    test rcx, rcx
    jz .read_loop

    lea rax, [rip + buffer]
    movzx ebx, byte ptr [rax + r15]
    inc qword ptr [rip + byte_count]

    cmp bl, 0x0A
    jne .not_newline
    inc qword ptr [rip + line_count]
.not_newline:

    lea rax, [rip + whitespace_table]
    movzx edx, byte ptr [rax + rbx]
    test dl, dl
    jz .not_whitespace

    mov byte ptr [rip + in_word], 0
    jmp .continue_loop
.not_whitespace:
    cmp byte ptr [rip + in_word], 0
    jne .continue_loop
    inc qword ptr [rip + word_count]
    mov byte ptr [rip + in_word], 1
.continue_loop:

    inc r15
    dec rcx
    jmp .process_byte

.done_reading:
    mov rdi, [rip + line_count]
    call print_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_byte]
    mov rdx, 1
    syscall

    mov rdi, [rip + word_count]
    call print_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_byte]
    mov rdx, 1
    syscall

    mov rdi, [rip + byte_count]
    call print_num

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

print_num:
    push rbx
    sub rsp, 24
    mov rax, rdi
    lea rcx, [rsp + 23]

.pn_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    add rdx, 0x30
    mov byte ptr [rcx], dl
    dec rcx
    test rax, rax
    jnz .pn_loop

.pn_write:
    inc rcx
    lea rdx, [rsp + 24]
    sub rdx, rcx

    mov rax, 1
    mov rdi, 1
    mov rsi, rcx
    syscall

    add rsp, 24
    pop rbx
    ret
```
