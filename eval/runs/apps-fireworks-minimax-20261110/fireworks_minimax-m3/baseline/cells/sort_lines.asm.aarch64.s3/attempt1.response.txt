.section .text
.globl _start

_start:
	// Read all of stdin into input_buf
	adrp	x10, input_buf
	add	x10, x10, :lo12:input_buf
	mov	x11, #0
.Lread_loop:
	cmp	x11, #32768
	b.ge	.Lread_done
	mov	x0, #0
	mov	x1, x10
	mov	x2, #32768
	sub	x2, x2, x11
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	.Lread_done
	add	x11, x11, x0
	add	x10, x10, x0
	b	.Lread_loop

.Lread_done:
	// Parse lines
	mov	x12, #0
	mov	x13, #0
	adrp	x14, input_buf
	add	x14, x14, :lo12:input_buf
.Lparse_loop:
	cmp	x13, x11
	b.ge	.Lparse_done
	mov	x15, x13
.Lscan:
	cmp	x15, x11
	b.ge	.Lfinal_line
	ldrb	w16, [x14, x15]
	cmp	w16, #0x0a
	b.eq	.Lfound_nl
	add	x15, x15, #1
	b	.Lscan

.Lfound_nl:
	sub	x16, x15, x13
	adrp	x17, lines
	add	x17, x17, :lo12:lines
	add	x18, x17, x12, lsl #3
	str	w13, [x18]
	str	w16, [x18, #4]
	add	x12, x12, #1
	add	x13, x15, #1
	b	.Lparse_loop

.Lfinal_line:
	cmp	x13, x11
	b.ge	.Lparse_done
	sub	x16, x11, x13
	adrp	x17, lines
	add	x17, x17, :lo12:lines
	add	x18, x17, x12, lsl #3
	str	w13, [x18]
	str	w16, [x18, #4]
	add	x12, x12, #1
.Lparse_done:
	mov	x19, x12

	// Insertion sort
	cmp	x19, #1
	b.le	.Lsort_done
	mov	x20, #1
.Lsort_outer:
	cmp	x20, x19
	b.ge	.Lsort_done

	adrp	x17, lines
	add	x17, x17, :lo12:lines
	lsl	x21, x20, #3
	add	x21, x17, x21
	ldr	w23, [x21]
	ldr	w24, [x21, #4]

	sub	x25, x20, #1
.Lsort_inner:
	cmp	x25, #0
	b.lt	.Linsert

	lsl	x22, x25, #3
	add	x22, x17, x22
	ldr	w26, [x22]
	ldr	w27, [x22, #4]

	cmp	w27, w24
	b.eq	.Llen_eq
	b.lo	.Llen_lt

.Llen_gt:
	cmp	w24, #0
	b.eq	.Lshift
	mov	x28, #0
.Lcj_gt:
	cmp	x28, w24
	b.ge	.Lshift
	add	x29, x26, x28
	ldrb	w30, [x14, x29]
	add	x0, x23, x28
	ldrb	w0, [x14, x0]
	cmp	w30, w0
	b.hi	.Lshift
	b.lo	.Lnoshift
	add	x28, x28, #1
	b	.Lcj_gt

.Llen_lt:
	cmp	w27, #0
	b.eq	.Lnoshift
	mov	x28, #0
.Lcj_lt:
	cmp	x28, w27
	b.ge	.Lnoshift
	add	x29, x26, x28
	ldrb	w30, [x14, x29]
	add	x0, x23, x28
	ldrb	w0, [x14, x0]
	cmp	w30, w0
	b.hi	.Lshift
	b.lo	.Lnoshift
	add	x28, x28, #1
	b	.Lcj_lt

.Llen_eq:
	cmp	w27, #0
	b.eq	.Lnoshift
	mov	x28, #0
.Lcej:
	cmp	x28, w27
	b.ge	.Lnoshift
	add	x29, x26, x28
	ldrb	w30, [x14, x29]
	add	x0, x23, x28
	ldrb	w0, [x14, x0]
	cmp	w30, w0
	b.hi	.Lshift
	b.lo	.Lnoshift
	add	x28, x28, #1
	b	.Lcej

.Lshift:
	add	x29, x25, #1
	lsl	x22, x29, #3
	add	x22, x17, x22
	str	w26, [x22]
	str	w27, [x22, #4]
	sub	x25, x25, #1
	b	.Lsort_inner

.Lnoshift:
	b	.Linsert

.Linsert:
	add	x29, x25, #1
	lsl	x22, x29, #3
	add	x22, x17, x22
	str	w23, [x22]
	str	w24, [x22, #4]
	add	x20, x20, #1
	b	.Lsort_outer

.Lsort_done:
	// Output sorted lines
	mov	x20, #0
.Lout_loop:
	cmp	x20, x19
	b.ge	.Lexit

	adrp	x17, lines
	add	x17, x17, :lo12:lines
	lsl	x21, x20, #3
	add	x21, x17, x21
	ldr	w22, [x21]
	ldr	w23, [x21, #4]

	mov	x0, #1
	adrp	x24, input_buf
	add	x24, x24, :lo12:input_buf
	add	x1, x24, x22, uxtw
	mov	x2, x23
	mov	x8, #64
	svc	#0

	mov	x0, #1
	adrp	x1, newline
	add	x1, x1, :lo12:newline
	mov	x2, #1
	mov	x8, #64
	svc	#0

	add	x20, x20, #1
	b	.Lout_loop

.Lexit:
	mov	x0, #0
	mov	x8, #93
	svc	#0

.section .bss
.balign 4
input_buf:	.space 32768
.balign 4
lines:		.space 8000

.section .rodata
newline:	.byte 0x0a
