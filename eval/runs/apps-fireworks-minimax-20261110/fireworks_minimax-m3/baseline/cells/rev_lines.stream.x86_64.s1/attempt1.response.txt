.text
.globl _start

_start:
    xor r14, r14

.L_read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, 65536
    sub rdx, r14
    jz .L_read_done
    syscall
    test rax, rax
    jle .L_read_done
    add r14, rax
    jmp .L_read_loop
.L_read_done:

    xor r12, r12

.L_process_lines:
    cmp r12, r14
    jae .L_exit

    lea r10, [rip + input_buffer]
    lea r11, [rip + output_buffer]

    mov r13, r12
.L_find_newline:
    cmp r13, r14
    je .L_got_line
    movzx eax, byte ptr [r10 + r13]
    cmp al, 0x0a
    je .L_got_line
    inc r13
    jmp .L_find_newline
.L_got_line:
    mov rcx, r13
    sub rcx, r12
    test rcx, rcx
    jz .L_write_nl

    mov r8, r13
    xor r9, r9
.L_rev_loop:
    cmp r9, rcx
    jge .L_rev_done
    dec r8
    movzx eax, byte ptr [r10 + r8]
    mov byte ptr [r11 + r9], al
    inc r9
    jmp .L_rev_loop
.L_rev_done:

    mov rax, 1
    mov rdi, 1
    mov rsi, r11
    mov rdx, rcx
    syscall

.L_write_nl:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall

    cmp r13, r14
    jae .L_set_end
    mov r12, r13
    inc r12
    jmp .L_process_lines
.L_set_end:
    mov r12, r14
    jmp .L_process_lines

.L_exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.balign 16
input_buffer: .space 65536
output_buffer: .space 65536

.section .rodata
newline_char: .byte 0x0a
