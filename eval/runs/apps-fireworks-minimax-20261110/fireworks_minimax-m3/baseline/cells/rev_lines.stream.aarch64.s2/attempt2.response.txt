//
// rev_lines for AArch64 Linux
// Self-contained assembly: no libc, no C runtime.
// Reads stdin, prints each line byte-reversed followed by newline.
//

.globl _start

.section .bss
inbuf:  .space 65536           // input buffer
outbuf: .space 65537           // one reversed line incl. newline

.section .text

// _start -- entry point. x0..x30 = 0; never returns; ends via exit.
_start:
    // Phase 1: read all of stdin into inbuf.
    adrp    x16, inbuf
    add     x16, x16, :lo12:inbuf
    mov     x1, x16
    mov     x10, #0                         // total bytes read

.Lread_loop:
    mov     x0, #0                          // fd = 0 (stdin)
    mov     x2, #65536
    sub     x2, x2, x10                     // remaining buffer space
    mov     x8, #63                         // sys_read
    svc     #0
    cmp     x0, #0
    b.eq    .Lread_done                     // EOF
    add     x10, x10, x0
    add     x1, x1, x0
    b       .Lread_loop

.Lread_done:
    // x10 = total bytes; x16 = &inbuf

    // Phase 2: process each line.
    adrp    x17, outbuf
    add     x17, x17, :lo12:outbuf
    mov     x11, #0                         // line start offset

.Lline_loop:
    cmp     x11, x10
    b.ge    .Lexit                          // no more data

    mov     x12, x11                        // scan position
    mov     x13, #0                         // 1 if newline found
.Lscan_loop:
    cmp     x12, x10
    b.ge    .Lscan_done
    ldrb    w14, [x16, x12]
    cmp     w14, #0x0A
    b.eq    .Lscan_newline
    add     x12, x12, #1
    b       .Lscan_loop
.Lscan_newline:
    mov     x13, #1
.Lscan_done:
    // x12 = end-of-line position; x13 = 1 if newline, 0 if EOF.
    mov     x18, x12                        // save end position

    // Reverse bytes [x11, x18) into outbuf.
    mov     x19, #0                         // output write index
.Lrev_loop:
    cmp     x12, x11
    b.eq    .Lrev_done
    sub     x12, x12, #1
    ldrb    w14, [x16, x12]
    strb    w14, [x17, x19]
    add     x19, x19, #1
    b       .Lrev_loop
.Lrev_done:
    mov     w14, #0x0A
    strb    w14, [x17, x19]                 // append newline
    add     x19, x19, #1

    mov     x0, #1                          // fd = 1 (stdout)
    mov     x1, x17
    mov     x2, x19
    mov     x8, #64                         // sys_write
    svc     #0

    cbnz    x13, .Lline_next                // newline found: continue
    // No newline: final line, we're done.

.Lexit:
    mov     x0, #0
    mov     x8, #93                         // sys_exit
    svc     #0

.Lline_next:
    add     x11, x18, #1                    // skip the newline
    b       .Lline_loop
