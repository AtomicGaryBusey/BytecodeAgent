.section .rodata
.p2align 4
ws_table:
    .zero 9
    .byte 1, 1, 1, 1, 1
    .zero 242

space_str:
    .byte 0x20

newline_str:
    .byte 0x0a

.section .bss
.p2align 4
buffer:
    .space 65536
.p2align 4
numbuf:
    .space 32

.section .text
.globl _start

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    mov rcx, rax
    xor rbx, rbx

byte_loop:
    cmp rbx, rcx
    jae done_byte_loop

    movzx eax, byte ptr [rsi + rbx]
    inc r12

    lea rdi, [rip + ws_table]
    movzx edi, byte ptr [rdi + rax]
    test edi, edi
    jz .not_ws

    xor r15, r15
    cmp al, 0x0a
    jne .next_byte
    inc r13
    jmp .next_byte

.not_ws:
    test r15, r15
    jnz .next_byte
    inc r14
    mov r15, 1

.next_byte:
    inc rbx
    jmp byte_loop

done_byte_loop:
    jmp read_loop

done_reading:
    mov rdi, r13
    call write_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space_str]
    mov edx, 1
    syscall

    mov rdi, r14
    call write_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space_str]
    mov edx, 1
    syscall

    mov rdi, r12
    call write_uint

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_str]
    mov edx, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

write_uint:
    lea rsi, [rip + numbuf + 31]
    mov rax, rdi
    mov ecx, 10
    xor r8, r8
.wu_loop:
    xor edx, edx
    div rcx
    add edx, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    inc r8
    test rax, rax
    jnz .wu_loop

    mov eax, 1
    mov edi, 1
    mov rdx, r8
    syscall
    ret
