.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_line:
    .zero 128

.section .text
.globl _start

_start:
    xor r12, r12

.read_input:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz .input_done
    add r12, rax
    cmp r12, 65536
    je .input_done
    jmp .read_input

.input_done:
    lea rbx, [rip + hex_digits]
    lea r15, [rip + input_buf]
    xor r14, r14
    test r12, r12
    jz .print_total

.row_loop:
    mov rcx, r12
    sub rcx, r14
    test rcx, rcx
    jz .print_total
    cmp rcx, 16
    jbe .got_n
    mov rcx, 16
.got_n:
    mov r10, rcx

    lea r13, [rip + output_line]
    mov rdi, r13

    mov rax, r14
    mov rcx, 28
    mov r9, 8
.row_offset:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub cl, 4
    dec r9
    jnz .row_offset

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi

    xor r9, r9
.hex_col:
    cmp r9, 15
    jg .hex_done
    cmp r9, r10
    jge .empty_hex

    mov r11, r14
    add r11, r9
    movzx edx, byte ptr [r15 + r11]

    mov rsi, rdx
    shr rsi, 4
    mov dl, byte ptr [rbx + rsi]
    mov byte ptr [rdi], dl
    inc rdi
    mov rsi, rdx
    and rsi, 0xF
    mov dl, byte ptr [rbx + rsi]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp .after_hex

.empty_hex:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi

.after_hex:
    cmp r9, 7
    jne .no_extra_space
    mov byte ptr [rdi], 0x20
    inc rdi
.no_extra_space:
    inc r9
    jmp .hex_col

.hex_done:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi

    xor r9, r9
.ascii_col:
    cmp r9, r10
    je .ascii_done

    mov r11, r14
    add r11, r9
    movzx edx, byte ptr [r15 + r11]

    cmp dl, 0x20
    jb .ascii_dot
    cmp dl, 0x7e
    ja .ascii_dot
    jmp .ascii_put
.ascii_dot:
    mov dl, 0x2e
.ascii_put:
    mov byte ptr [rdi], dl
    inc rdi
    inc r9
    jmp .ascii_col

.ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rax, 1
    mov rsi, r13
    mov rdx, rdi
    sub rdx, r13
    mov rdi, 1
    syscall

    add r14, r10
    jmp .row_loop

.print_total:
    lea r13, [rip + output_line]
    mov rdi, r13

    mov rax, r12
    mov rcx, 28
    mov r9, 8
.total_offset:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub cl, 4
    dec r9
    jnz .total_offset

    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rax, 1
    mov rsi, r13
    mov rdx, rdi
    sub rdx, r13
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
