.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf: .zero 65536
.align 16
output_line: .zero 128

.section .text
.globl _start

_start:
    # read all input into input_buf (max 65536 bytes)
    xor r12, r12            # total bytes read
    mov r13, 65536          # buffer capacity

.read_loop:
    mov rax, 0              # sys_read
    mov rdi, 0              # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r12            # point to next free byte
    mov rdx, r13
    sub rdx, r12            # remaining bytes
    syscall
    test rax, rax
    jz .read_done           # EOF
    add r12, rax
    cmp r12, r13
    je .read_done           # buffer full
    jmp .read_loop

.read_done:
    # prepare for output
    lea rbx, [rip + hex_digits]
    xor r14, r14            # current offset = 0

    test r12, r12
    jz .final_total         # empty input

.row_loop:
    # compute remaining bytes
    mov rcx, r12
    sub rcx, r14
    test rcx, rcx
    jz .final_total         # no more rows

    # n = min(remaining, 16)
    cmp rcx, 16
    jbe .set_n
    mov rcx, 16
.set_n:
    mov r10, rcx            # n

    # start filling line buffer
    lea r15, [rip + output_line]
    mov rdi, r15            # fill pointer

    # write offset (8 hex digits)
    mov rax, r14            # offset value
    mov rcx, 28             # shift amount for first nibble
    mov r9, 8
.offset_loop_row:
        mov rdx, rax
        shr rdx, cl
        and rdx, 0xF
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi], dl
        inc rdi
        sub cl, 4
        dec r9
        jnz .offset_loop_row

    # two spaces after offset
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

    # hex column loop i = 0..15
    xor r9, r9              # i = 0
.hex_loop:
        cmp r9, 15
        jg .hex_done

        cmp r9, r10          # i < n ?
        jge .empty_slot

        # load byte from buffer
        mov r11, r14
        add r11, r9          # offset + i
        lea rsi, [rip + input_buf]
        movzx edx, byte ptr [rsi + r11]

        # high nibble
        mov rsi, rdx
        shr rsi, 4
        and rsi, 0xF
        mov al, byte ptr [rbx + rsi]
        mov byte ptr [rdi], al
        inc rdi

        # low nibble
        mov rsi, rdx
        and rsi, 0xF
        mov al, byte ptr [rbx + rsi]
        mov byte ptr [rdi], al
        inc rdi

        # space after digit
        mov byte ptr [rdi], ' '
        inc rdi

        jmp .after_slot

.empty_slot:
        mov byte ptr [rdi], ' '
        inc rdi
        mov byte ptr [rdi], ' '
        inc rdi
        mov byte ptr [rdi], ' '
        inc rdi
.after_slot:

        # extra space after slot 7
        cmp r9, 7
        jne .skip_extra
        mov byte ptr [rdi], ' '
        inc rdi
.skip_extra:

        inc r9
        jmp .hex_loop

.hex_done:

    # ascii column: space, '|', chars, '|', newline
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor r9, r9              # i = 0
.ascii_loop:
        cmp r9, r10
        je .ascii_done

        mov r11, r14
        add r11, r9
        lea rsi, [rip + input_buf]
        movzx edx, byte ptr [rsi + r11]

        cmp dl, 0x20
        jb .print_dot
        cmp dl, 0x7E
        ja .print_dot
        # printable
        mov byte ptr [rdi], dl
        jmp .after_char
.print_dot:
        mov byte ptr [rdi], '.'
.after_char:
        inc rdi
        inc r9
        jmp .ascii_loop

.ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    # write line
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, rdi
    sub rdx, r15
    syscall

    # advance offset
    add r14, r10
    jmp .row_loop

.final_total:
    # print total length (8 hex digits) then newline
    lea r15, [rip + output_line]
    mov rdi, r15

    mov rax, r12            # total bytes
    mov rcx, 28
    mov r9, 8
.final_offset_loop:
        mov rdx, rax
        shr rdx, cl
        and rdx, 0xF
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi], dl
        inc rdi
        sub cl, 4
        dec r9
        jnz .final_offset_loop

    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, rdi
    sub rdx, r15
    syscall

    # exit
    mov rax, 60
    xor rdi, rdi
    syscall
