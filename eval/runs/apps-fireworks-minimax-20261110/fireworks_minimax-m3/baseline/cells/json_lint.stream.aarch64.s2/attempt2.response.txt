.section .rodata
.align 2
err_prefix: .ascii "error at "
ok_msg: .ascii "ok\n"
newline_msg: .ascii "\n"

.section .bss
.align 4
input_buf: .space 65536
print_buf: .space 16
error_flag: .space 4
error_offset: .space 4

.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    mov x22, #0
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
read_loop:
    mov x2, #65536
    sub x2, x2, x22
    add x1, x23, x22
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x22, x22, x0
    cmp x22, #65536
    b.ge read_done
    b read_loop
read_done:
    mov x19, x23
    add x21, x19, x22
    mov x20, x19
    mov x0, x20
    bl parse_document
    b .

skip_ws:
    b skip_ws_check
skip_ws_loop:
    add x0, x0, #1
skip_ws_check:
    cmp x0, x21
    b.ge skip_ws_done
    ldrb w2, [x0]
    cmp w2, #0x20
    b.eq skip_ws_loop
    cmp w2, #0x09
    b.eq skip_ws_loop
    cmp w2, #0x0A
    b.eq skip_ws_loop
    cmp w2, #0x0D
    b.eq skip_ws_loop
skip_ws_done:
    ret

report_error:
    adrp x10, error_flag
    add x10, x10, :lo12:error_flag
    ldr w11, [x10]
    cbnz w11, 1f
    adrp x11, error_offset
    add x11, x11, :lo12:error_offset
    str w9, [x11]
    mov w12, #1
    str w12, [x10]
1:
    ret

print_number:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    adrp x1, print_buf
    add x1, x1, :lo12:print_buf
    mov x2, #0
    cbnz x0, 1f
    mov w3, #0x30
    strb w3, [x1]
    mov x2, #1
    b 2f
1:
    mov x4, x0
digit_loop:
    cbz x4, 2f
    udiv x5, x4, #10
