.section .bss
.align 16
buf:    .space 32768
.align 16
lines:  .space 8000
.align 16
idx:    .space 4000
.align 16
tmp:    .space 4000

.section .rodata
nl_byte:
    .byte 0x0a

.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .text
.globl _start

_start:
    # Phase 1: read all of stdin into buf
    xor r12d, r12d
.read_a:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, BUF_SIZE
    sub edx, r12d
    syscall
    cmp rax, 0
    jle .read_b
    add r12, rax
    cmp r12d, BUF_SIZE
    jb .read_a
.read_b:
    test r12d, r12d
    jz .exit_main

    # Phase 2: parse input into lines
    lea rbx, [rip + lines]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
.parse_a:
    cmp r13d, r12d
    jae .parse_d
    cmp r15d, MAX_LINES
    jae .parse_d
    movzx eax, byte ptr [buf + r13]
    cmp al, 0x0a
    jne .parse_c
    mov dword ptr [rbx + r15*8], r14d
    mov eax, r13d
    sub eax, r14d
    mov dword ptr [rbx + r15*8 + 4], eax
    inc r15d
    inc r13d
    mov r14d, r13d
    jmp .parse_a
.parse_c:
    inc r13d
    jmp .parse_a
.parse_d:
    cmp r14d, r12d
    jae .parse_e
    cmp r15d, MAX_LINES
    jae .parse_e
    mov dword ptr [rbx + r15*8], r14d
    mov eax, r12d
    sub eax, r14d
    mov dword ptr [rbx + r15*8 + 4], eax
    inc r15d
.parse_e:
    mov r12d, r15d

    # Phase 3: bottom-up merge sort of idx[0..r12d)
    xor ecx, ecx
.init_a:
    cmp ecx, r12d
    jae .init_b
    mov dword ptr [idx + rcx*4], ecx
    inc ecx
    jmp .init_a
.init_b:
    mov r13d, 1
.sort_a:
    cmp r13d, r12d
    jae .sort_g
    xor r14d, r14d
.sort_b:
    cmp r14d, r12d
    jae .sort_f
    mov ecx, r14d
    add ecx, r13d
    mov edx, r14d
    add edx, r13d
    add edx, r13d
    cmp edx, r12d
    jbe .sort_c
    mov edx, r12d
.sort_c:
    cmp ecx, edx
    jae .sort_e
    mov edi, r14d
    mov esi, ecx
    call merge
.sort_e:
    add r14d, r13d
    add r14d, r13d
    jmp .sort_b
.sort_f:
    shl r13d, 1
    jmp .sort_a
.sort_g:

    # Phase 4: write sorted lines to stdout
    xor ecx, ecx
.output_a:
    cmp ecx, r12d
    jae .exit_main
    mov eax, dword ptr [idx + rcx*4]
    mov edx, dword ptr [lines + eax*8 + 4]
    test edx, edx
    jz .output_b
    mov r8d, dword ptr [lines + eax*8]
    lea rsi, [rip + buf]
    add rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
.output_b:
    lea rsi, [rip + nl_byte]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    inc ecx
    jmp .output_a

.exit_main:
    mov eax, 60
    xor edi, edi
    syscall

# merge: merge idx[lo..mid) with idx[mid..hi) into tmp, copy back
merge:
    push r12
    push r13
    mov r8d, edi
    mov r9d, esi
    xor r10d, r10d
.m_a:
    cmp r8d, esi
    jae .m_c
    cmp r9d, edx
    jae .m_d
    push rdi
    push rsi
    push rdx
    push r8
    push r9
    push r10
    mov edi, dword ptr [idx + r8*4]
    mov esi, dword ptr [idx + r9*4]
    call line_cmp
    pop r10
    pop r9
    pop r8
    pop rdx
    pop rsi
    pop rdi
    test eax, eax
    jle .m_b
    mov eax, dword ptr [idx + r9*4]
    mov dword ptr [tmp + r10*4], eax
    inc r9d
    inc r10d
    jmp .m_a
.m_b:
    mov eax, dword ptr [idx + r8*4]
    mov dword ptr [tmp + r10*4], eax
    inc r8d
    inc r10d
    jmp .m_a
.m_c:
    cmp r9d, edx
    jae .m_e
    jmp .m_g
.m_d:
.m_d_loop:
    cmp r8d, esi
    jae .m_e
    mov eax, dword ptr [idx + r8*4]
    mov dword ptr [tmp + r10*4], eax
    inc r8d
    inc r10d
    jmp .m_d_loop
.m_e:
    push rdi
    xor r11d, r11d
    mov r8d, edi
.m_e_loop:
    cmp r11d, r10d
    jae .m_f
    mov eax, dword ptr [tmp + r11*4]
    mov dword ptr [idx + r8*4], eax
    inc r8d
    inc r11d
    jmp .m_e_loop
.m_f:
    pop rdi
    pop r13
    pop r12
    ret
.m_g:
.m_g_loop:
    cmp r9d, edx
    jae .m_e
    mov eax, dword ptr [idx + r9*4]
    mov dword ptr [tmp + r10*4], eax
    inc r9d
    inc r10d
    jmp .m_g_loop

# line_cmp: compare lines i_a=edi vs i_b=esi, return eax (-1, 0, 1)
line_cmp:
    push r12
    push r13
    push r14
    cmp edi, esi
    jne .lc_ne
    xor eax, eax
    jmp .lc_done
.lc_ne:
    mov r8d, dword ptr [lines + rdi*8]
    mov r9d, dword ptr [lines + rdi*8 + 4]
    mov r10d, dword ptr [lines + rsi*8]
    mov r11d, dword ptr [lines + rsi*8 + 4]
    mov ecx, r9d
    cmp ecx, r11d
    jle .lc_min
    mov ecx, r11d
.lc_min:
    test ecx, ecx
    jnz .lc_bytes
    cmp r9d, r11d
    jb .lc_neg
    ja .lc_pos
    xor eax, eax
    jmp .lc_done
.lc_bytes:
    lea r12, [rip + buf]
    add r12, r8
    lea r13, [rip + buf]
    add r13, r10
    xor r14d, r14d
.lc_loop:
    movzx eax, byte ptr [r12 + r14]
    movzx edx, byte ptr [r13 + r14]
    cmp eax, edx
    jb .lc_neg
    ja .lc_pos
    inc r14d
    cmp r14d, ecx
    jb .lc_loop
    cmp r9d, r11d
    jb .lc_neg
    ja .lc_pos
    xor eax, eax
    jmp .lc_done
.lc_neg:
    mov eax, -1
    jmp .lc_done
.lc_pos:
    mov eax, 1
.lc_done:
    pop r14
    pop r13
    pop r12
    ret
