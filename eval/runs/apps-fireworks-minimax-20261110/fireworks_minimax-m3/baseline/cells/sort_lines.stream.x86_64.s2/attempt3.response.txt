.section .rodata
newline_byte: .byte 0x0a

.text
.globl _start
_start:
    sub rsp, 0xA100
    xor r12d, r12d
.read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, rsp
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    cmp r12d, 32768
    jb .read_loop
.read_done:
    test r12, r12
    jz .exit_ok
    lea rbx, [rsp + 0x8000]
    mov r13, rsp
    xor r14, r14
    xor r15, r15
    xor ebp, ebp
.parse_loop:
    cmp r15, r12
    jae .parse_done
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x0a
    jne .parse_next
    mov dword ptr [rbx + r14*8], ebp
    mov eax, r15d
    sub eax, ebp
    mov dword ptr [rbx + r14*8 + 4], eax
    inc r14
    inc r15
    mov ebp, r15d
    jmp .parse_loop
.parse_next:
    inc r15
    jmp .parse_loop
.parse_done:
    cmp ebp, r12
