.section .rodata
.align 1
newline_byte:
    .byte 0x0a

.section .bss
.align 16
input_buf:
    .zero 32768
.align 16
line_table:
    .zero 8000

.text
.globl _start
_start:
    xor r12, r12
.read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    cmp r12, 32768
    jb .read_loop
.read_done:
    lea rbx, [rip + line_table]
    lea rax, [rip + input_buf]
    xor r13, r13
    xor r14, r14
    xor r15, r15
.parse_loop:
    cmp r15, r12
    jae .parse_done
    movzx ecx, byte ptr [rax + r15]
    cmp ecx, 0x0a
    jne .parse_next
    mov ecx, r15d
    sub ecx, r14d
    mov dword ptr [rbx + r13*8 + 0], r14d
    mov dword ptr [rbx + r13*8 + 4], ecx
    inc r13
    inc r15
    mov r14, r15
    jmp .parse_loop
.parse_next:
    inc r15
    jmp .parse_loop
.parse_done:
    cmp r14, r12
    jae .do_sort
    mov ecx, r12d
    sub ecx, r14d
    mov dword ptr [rbx + r13*8 + 0], r14d
    mov dword ptr [rbx + r13*8 + 4], ecx
    inc r13
.do_sort:
    xor r14, r14
.sort_outer:
    cmp r14, r13
    jae .do_output
    mov r15, r14
    mov r10, r14
    inc r10
.sort_inner:
    cmp r10, r13
    jae .sort_swap
    mov edi, dword ptr [rbx + r10*8 + 0]
    mov esi, dword ptr [rbx + r10*8 + 4]
    mov edx, dword ptr [rbx + r15*8 + 0]
    mov ecx, dword ptr [rbx + r15*8 + 4]
    call compare_lines
    test eax, eax
    jge .sort_no_update
    mov r15, r10
.sort_no_update:
    inc r10
    jmp .sort_inner
.sort_swap:
    mov eax, dword ptr [rbx + r14*8 + 0]
    mov ecx, dword ptr [rbx + r14*8 + 4]
    mov edx, dword ptr [rbx + r15*8 + 0]
    mov esi, dword ptr [rbx + r15*8 + 4]
    mov dword ptr [rbx + r14*8 + 0], edx
    mov dword ptr [rbx + r14*8 + 4], esi
    mov dword ptr [rbx + r15*8 + 0], eax
    mov dword ptr [rbx + r15*8 + 4], ecx
    inc r14
    jmp .sort_outer
.do_output:
    xor r14, r14
.out_loop:
    cmp r14, r13
    jae .exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + input_buf]
    add rsi, dword ptr [rbx + r14*8 + 0]
