.section .rodata
newline: .byte 0x0a

.section .bss
.balign 16
input_buf: .space 32768
.balign 16
line_table: .space 8000

.text
.globl _start
_start:
    xor ecx, ecx
.read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rcx
    mov edx, 32768
    sub edx, ecx
    syscall
    test rax, rax
    jle .read_done
    add rcx, rax
    cmp ecx, 32768
    jb .read_loop
.read_done:
    mov r12, rcx

    lea rbx, [rip + line_table]
    lea r13, [rip + input_buf]
    xor r14, r14
    xor r15, r15
    xor ebp, ebp
.parse_loop:
    cmp r15, r12
    jae .parse_done
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x0a
    jne .parse_next
    mov dword ptr [rbx + r14*8], ebp
    mov eax, r15d
    sub eax, ebp
    mov dword ptr [rbx + r14*8 + 4], eax
    inc r14
    inc r15
    mov ebp, r15d
    jmp .parse_loop
.parse_next:
    inc r15
    jmp .parse_loop
.parse_done:
    cmp rbp, r12
    jae .do_sort
    mov dword ptr [rbx + r14*8], ebp
    mov eax, r12d
    sub eax, ebp
    mov dword ptr [rbx + r14*8 + 4], eax
    inc r14

.do_sort:
    xor r15, r15
.sort_outer:
    cmp r15, r14
    jae .do_output
    mov rbp, r15
    mov r13, r15
    inc r13
.sort_inner:
    cmp r13, r14
    jae .sort_swap
    mov edi, dword ptr [rbx + r13*8]
    mov esi, dword ptr [rbx + r13*8 + 4]
    mov edx, dword ptr [rbx + rbp*8]
    mov ecx, dword ptr [rbx + rbp*8 + 4]
    call compare_lines
    test eax, eax
    jge .sort_no_update
    mov rbp, r13
.sort_no_update:
    inc r13
    jmp .sort_inner
.sort_swap:
    cmp rbp, r15
    je .sort_next_i
    mov eax, dword ptr [rbx + r15*8]
    mov ecx, dword ptr [rbx + r15*8 + 4]
    mov edx, dword ptr [rbx + rbp*8]
    mov esi, dword ptr [rbx + rbp*8 + 4]
    mov dword ptr [rbx + r15*8], edx
    mov dword ptr [rbx + r15*8 + 4], esi
    mov dword ptr [rbx + rbp*8], eax
    mov dword ptr [rbx + rbp*8 + 4], ecx
.sort_next_i:
    inc r15
    jmp .sort_outer

.do_output:
    xor r15, r15
.out_loop:
    cmp r15, r14
    jae .exit_ok
    mov eax, dword ptr [rbx + r15*8]
    lea rsi, [rip + input_buf]
    add rsi, rax
    mov edx, dword ptr [rbx + r15*8 + 4]
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r15
    jmp .out_loop

.exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx

    mov r8, r13
    cmp r8, r15
    jbe .min_set
    mov r8, r15
.min_set:
    lea r9, [rip + input_buf]
    add r9, r12
    lea r10, [rip + input_buf]
    add r10, r14
    xor rbx, rbx
.cmp_loop:
    cmp rbx, r8
    jae .prefix_done
    movzx esi, byte ptr [r9 + rbx]
    movzx eax, byte ptr [r10 + rbx]
    cmp sil, al
    je .next_byte
    sub esi, eax
    mov eax, esi
    jmp .cmp_done
.next_byte:
    inc rbx
    jmp .cmp_loop
.prefix_done:
    mov eax, r13d
    sub eax, r15d
.cmp_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
