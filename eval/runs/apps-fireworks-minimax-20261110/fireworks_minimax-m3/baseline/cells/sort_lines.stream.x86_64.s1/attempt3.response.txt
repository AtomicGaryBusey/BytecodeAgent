.section .rodata
newline: .byte 10

.bss
.balign 16
input_buf: .space 32768
.balign 16
entries:   .space 16000

.text
.globl _start
_start:
    # read stdin into input_buf
    lea rbx, [rip + input_buf]
    xor r12, r12                # bytes read
.read_loop:
    mov eax, 0                  # sys_read
    mov edi, 0                  # fd = stdin
    lea rsi, [rbx + r12]
    mov edx, 32768
    sub edx, r12
