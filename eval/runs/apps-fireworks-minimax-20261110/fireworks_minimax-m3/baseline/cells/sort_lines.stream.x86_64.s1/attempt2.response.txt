.bss
.balign 16
input_buf:    .space 32768
.balign 16
entries:      .space 16000
.balign 16
output_buf:   .space 34816

.text
.globl _start

# Sort lines from stdin: byte-wise ascending, duplicates preserved, each line ends with \n.
_start:
    # Read stdin into input_buf. r12 = total bytes read.
    lea rbx, [rip + input_buf]
    xor r12, r12
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, 32768
    jb .read_loop
.read_done:
    # Parse lines into entries[]. r8=line_start, r9=count, r10=pos, rdi=write ptr.
    lea rdi, [rip + entries]
    xor r8, r8
    xor r9, r9
    xor r10, r10
.parse_loop:
    cmp r10, r12
    je .parse_tail
    cmp byte ptr [input_buf + r10], 10
    jne .parse_continue
    lea rcx, [rip + input_buf]
    add rcx, r8
    mov [rdi], rcx
    mov rax, r10
    sub rax, r8
    mov [rdi + 8], rax
    add rdi, 16
    inc r9
    mov r8, r10
    inc r8
.parse_continue:
    inc r10
    jmp .parse_loop
.parse_tail:
    mov rax, r12
    sub rax, r8
    jz .sort_start
    lea rcx, [rip + input_buf]
    add rcx, r8
    mov [rdi], rcx
    mov [rdi + 8], rax
    inc r9
.sort_start:
    # Insertion sort. r12=i, r13=key_ptr, r14=key_len, r15=j. rbp/rbx hold entries[j].
    cmp r9, 1
    jbe .sort_done
    mov r12, 1
.sort_outer:
    cmp r12, r9
    jae .sort_done
    mov r13, [entries + r12*16]
