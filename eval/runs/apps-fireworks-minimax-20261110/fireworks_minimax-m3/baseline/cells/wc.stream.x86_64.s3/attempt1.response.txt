.globl _start
.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_more:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    test rax, rax
    jz print
    js error_exit
    mov r8, rax
    add r12, r8
    xor rcx, rcx
    mov rbx, r8

process_byte:
    cmp rcx, rbx
    jge read_more
    movzx eax, byte ptr [rsi + rcx]
    lea rdx, [rip + ws_table]
    movzx eax, byte ptr [rdx + rax]
    test eax, eax
    jz not_ws
    xor r15, r15
    cmp eax, 2
    jne advance
    inc r13
    jmp advance

not_ws:
    test r15, r15
    jnz advance
    inc r14
    mov r15b, 1

advance:
    inc rcx
    jmp process_byte

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

print:
    lea rsi, [rip + out_buf]
    mov r11, rsi
    xor r10, r10
    mov rax, r13
    call itoa
    add r10, rcx
    add rsi, rcx
    mov byte ptr [rsi], 0x20
    inc r10
    inc rsi
    mov rax, r14
    call itoa
    add r10, rcx
    add rsi, rcx
    mov byte ptr [rsi], 0x20
    inc r10
    inc rsi
    mov rax, r12
    call itoa
    add r10, rcx
    add rsi, rcx
    mov byte ptr [rsi], 0x0a
    inc r10
    mov rax, 1
    mov rdi, 1
    mov rsi, r11
    mov rdx, r10
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

itoa:
    push rbx
    mov rbx, rsi
    mov r11, 10
    xor rcx, rcx

iloop:
    test rax, rax
    jz irev_check
    xor rdx, rdx
    div r11
    add rdx, 0x30
    mov byte ptr [rbx + rcx], dl
    inc rcx
    jmp iloop

irev_check:
    test rcx, rcx
    jnz irev
    mov byte ptr [rbx], 0x30
    mov rcx, 1
    jmp irex

irev:
    mov rdi, rbx
    lea rsi, [rbx + rcx - 1]

iloop2:
    cmp rdi, rsi
    jae irev_done
    mov dl, byte ptr [rdi]
    mov r8b, byte ptr [rsi]
    mov byte ptr [rdi], r8b
    mov byte ptr [rsi], dl
    inc rdi
    dec rsi
    jmp iloop2

irev_done:
    mov rsi, rbx

irex:
    pop rbx
    ret

.section .rodata
.balign 16
ws_table:
    .zero 9
    .byte 1
    .byte 2
    .byte 1
    .byte 1
    .byte 1
    .zero 18
    .byte 1
    .zero 223

.section .bss
.balign 16
buf: .space 65536
.balign 16
out_buf: .space 32
