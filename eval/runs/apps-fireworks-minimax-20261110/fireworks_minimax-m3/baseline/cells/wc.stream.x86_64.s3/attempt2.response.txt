.globl _start
.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    test rax, rax
    jz print
    js error
    mov r8, rax
    add r12, r8
    xor rcx, rcx
    mov rbx, r8

process_byte:
    cmp rcx, rbx
    jge read_loop
    movzx eax, byte ptr [rsi + rcx]
    lea rdx, [rip + ws_table]
    movzx eax, byte ptr [rdx + rax]
    test eax, eax
    jz not_ws
    xor r15, r15
    cmp eax, 2
    jne advance
    inc r13
    jmp advance

not_ws:
    test r15, r15
    jnz advance
    inc r14
    mov r15b, 1

advance:
    inc rcx
    jmp process_byte

error:
    mov rax, 60
    mov rdi, 1
    syscall

print:
    lea rsi, [rip + out_buf]
    mov r11, rsi
    xor r10, r10

    mov rax, r13
    call convert
    add r10, rcx
    mov byte ptr [rsi], 0x20
    inc rsi
    inc r10

    mov rax, r14
    call convert
    add r10, rcx
    mov byte ptr [rsi], 0x20
    inc rsi
    inc r10

    mov rax, r12
    call convert
    add r10, rcx
    mov byte ptr [rsi], 0x0a
    inc r10

    mov rax, 1
    mov rdi, 1
    mov rsi, r11
    mov rdx, r10
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

convert:
    push r11
    push rbx
    push r12
    mov r11, 10
    xor rcx, rcx
    test rax, rax
    jnz conv_loop
    mov byte ptr [rsi], 0x30
    inc rsi
    mov rcx, 1
    jmp conv_done
conv_loop:
    xor rdx, rdx
    div r11
    add rdx, 0x30
    push rdx
    inc rcx
    test rax, rax
    jnz conv_loop
conv_write:
    pop rdx
    mov byte ptr [rsi], dl
    inc rsi
    dec rcx
    jnz conv_write
conv_done:
    pop r12
    pop rbx
    pop r11
    ret

.section .rodata
.balign 16
ws_table:
    .zero 9
    .byte 1, 2, 1, 1, 1
    .zero 18
    .byte 1
    .zero 223

.section .bss
.balign 16
buf: .space 65536
.balign 16
out_buf: .space 32
