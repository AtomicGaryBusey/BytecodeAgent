.globl _start
.text

_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r12d, r12d

read_chunk:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz print_it
    js error_exit

    mov r13, rax
    add r8, rax
    xor ecx, ecx
    jmp scan0

scan_byte:
    inc rcx
scan0:
    cmp rcx, r13
    jge read_chunk

    movzx eax, byte ptr [rsi + rcx]

    cmp al, 0x0a
    jne not_nl
    inc r9
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0a
    je is_ws
    cmp al, 0x0b
    je is_ws
    cmp al, 0x0c
    je is_ws
    cmp al, 0x0d
    je is_ws
    jmp is_nws

is_ws:
    xor r12, r12
    jmp scan_byte

is_nws:
    test r12, r12
    jnz scan_byte
    inc r10
    mov r12b, 1
    jmp scan_byte

print_it:
    lea rdi, [rip + out]
    mov rsi, r9
    call print_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r10
    call print_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r8
    call print_uint
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + out]
    mov rdx, rdi
    sub rdx, rsi

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out]
    syscall

    mov eax, 60
    xor edi, edi
    syscall

error_exit:
    mov eax, 60
    mov edi, 1
    syscall

print_uint:
    test rsi, rsi
    jnz pu_nz
    mov byte ptr [rdi], 0x30
    inc rdi
    ret

pu_nz:
    xor ebx, ebx
    mov ecx, 10

pu_div:
    mov rax, rsi
    xor edx, edx
    div rcx
    add dl, 0x30
    push rdx
    inc ebx
    mov rsi, rax
    test rax, rax
    jnz pu_div

pu_write:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ebx
    jnz pu_write
    ret

.section .bss
.balign 16
buf: .space 65536
.balign 16
out: .space 32
