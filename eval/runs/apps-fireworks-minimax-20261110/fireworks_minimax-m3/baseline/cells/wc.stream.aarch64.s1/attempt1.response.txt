//
// wc - count lines, words, bytes of standard input (AArch64 Linux).
//

.section .bss
.balign 16
buffer: .space 65536

.text
.globl _start
_start:
    mov x19, #0                 // total bytes
    mov x20, #0                 // newlines
    mov x21, #0                 // words
    mov x22, #0                 // in_word state (0 or 1)
    mov x23, #0                 // bytes already in buffer
    adrp x24, buffer
    add x24, x24, :lo12:buffer

.L_read_loop:
    mov x2, #65536
    sub x2, x2, x23             // remaining capacity
    cbz x2, .L_read_done        // safety

    mov x0, #0                  // fd = 0 (stdin)
    add x1, x24, x23
    mov x8, #63                 // sys_read
    svc #0

    cmp x0, #0
    b.le .L_read_done           // 0 = EOF, <0 = error

    mov x25, x0                 // bytes read this call
    mov x0, x25
    add x1, x24, x23
    bl process_chunk

    add x23, x23, x25
    add x19, x19, x25
    b .L_read_loop

.L_read_done:
    mov x0, x20
    mov w1, #0x20               // ' '
    bl print_num
    mov x0, x21
    mov w1, #0x20
    bl print_num
    mov x0, x19
    mov w1, #0x0A               // '\n'
    bl print_num

    mov x0, #0
    mov x8, #93                 // sys_exit
    svc #0

//
// process_chunk(x0 = len, x1 = buf)
// Increments x20 (lines) and x21 (words); updates x22 (in_word).
//
process_chunk:
    cbz x0, .L_pc_done
.L_pc_loop:
    ldrb w9, [x1], #1

    cmp w9, #0x0A
    b.ne .L_pc_skip_nl
    add x20, x20, #1
.L_pc_skip_nl:

    // is_ws = ((b - 9) in [0..4]) || (b == 0x20)
    sub w10, w9, #0x09
    cmp w10, #4
    b.ls .L_pc_is_ws
    cmp w9, #0x20
    b.ne .L_pc_is_word

.L_pc_is_ws:
    cbz x22, .L_pc_continue
    mov x22, #0
    b .L_pc_continue

.L_pc_is_word:
    cbnz x22, .L_pc_continue
    add x21, x21, #1
    mov x22, #1

.L_pc_continue:
    sub x0, x0, #1
    cbnz x0, .L_pc_loop
.L_pc_done:
    ret

//
// print_num(x0 = n, w1 = trailing byte; 0 for none)
// Emits "<n>[trailing]" to stdout.
//
print_num:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32

    mov x9, sp
    mov x10, x0

    cbnz x10, .L_pn_nz
    mov w11, #0x30
    strb w11, [x9]
    mov x12, #1
    b .L_pn_ac

.L_pn_nz:
    mov x12, #0
.L_pn_cd:
    udiv x5, x10, #10
