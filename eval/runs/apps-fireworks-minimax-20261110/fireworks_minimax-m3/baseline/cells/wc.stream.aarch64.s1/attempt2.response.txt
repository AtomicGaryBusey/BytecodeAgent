//
// wc - count lines, words, bytes of standard input (AArch64 Linux).
//

.section .bss
.balign 16
buffer:
    .space 65536

.text
.globl _start
_start:
    mov     x19, #0            // total bytes
    mov     x20, #0            // newline counter (lines)
    mov     x21, #0            // word counter
    mov     x22, #0            // in_word state (0 outside, 1 inside)
    mov     x23, #0            // current offset in buffer

    adrp    x24, buffer
    add     x24, x24, :lo12:buffer

.L_read_loop:
    mov     x2, #65536
    sub     x2, x2, x23        // remaining capacity
    cbz     x2, .L_read_done

    mov     x0, #0             // fd = 0 (stdin)
    add     x1, x24, x23
    mov     x8, #63            // sys_read
    svc     #0

    cmp     x0, #0
    b.le    .L_read_done       // 0 = EOF, <0 = error

    mov     x25, x0            // bytes read this call
    mov     x26, x25
    add     x27, x24, x23

.L_proc_loop:
    cbz     x26, .L_proc_done
    ldrb    w28, [x27], #1

    cmp     w28, #0x0A
    b.ne    1f
    add     x20, x20, #1
1:

    // Whitespace if byte is in [9,13] or equal to 32.
    sub     w29, w28, #9
    cmp     w29, #4
    b.ls    2f
    cmp     w28, #32
    b.ne    3f
2:
    cbz     x22, 4f
    mov     x22, #0
    b       4f
3:
    cbnz    x22, 4f
    add     x21, x21, #1
    mov     x22, #1
4:

    sub     x26, x26, #1
    b       .L_proc_loop

.L_proc_done:
    add     x19, x19, x25
    add     x23, x23, x25
    b       .L_read_loop

.L_read_done:
    mov     x0, x20
    mov     w1, #0x20
    bl      print_num

    mov     x0, x21
    mov     w1, #0x20
    bl      print_num

    mov     x0, x19
    mov     w1, #0x0A
    bl      print_num

    mov     x0, #0
    mov     x8, #93            // sys_exit
    svc     #0

// print_num(x0 = value, w1 = trailing byte; 0 for none)
print_num:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp
    sub     sp, sp, #32

    mov     x10, x0
    mov     w11, w1

    add     x12, sp, #20       // one past end of digit area
    cbnz    x10, 1f
    sub     x12, x12, #1
    mov     w13, #0x30
    strb    w13, [x12]
    b       2f
1:
    mov     x14, #10
3:
    udiv    x15, x10, x14
    msub    x16, x15, x14, x10
    add     w16, w16, #0x30
    sub     x12, x12, #1
    strb    w16, [x12]
    mov     x10, x15
    cbnz    x10, 3b
2:
    add     x1, sp, #20
    sub     x2, x1, x12
    mov     x1, x12
    mov     x0, #1
    mov     x8, #64
    svc     #0

    cbz     w11, 4f
    add     x1, sp, #30
    strb    w11, [x1]
    mov     x0, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0
4:
    add     sp, sp, #32
    ldp     x29, x30, [sp], #16
    ret
