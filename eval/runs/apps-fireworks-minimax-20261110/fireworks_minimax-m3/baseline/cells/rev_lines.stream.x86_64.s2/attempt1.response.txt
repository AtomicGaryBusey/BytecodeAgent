.globl _start
.text
_start:
    xor r9, r9
.L_read_loop:
    mov rcx, 65536
    sub rcx, r9
    jle .L_read_done
    lea rsi, [rip + buffer]
    add rsi, r9
    mov rdx, rcx
    mov rdi, 0
    mov rax, 0
    syscall
    cmp rax, 0
    je .L_read_done
    add r9, rax
    jmp .L_read_loop
.L_read_done:
    lea r10, [rip + buffer]
    xor r12, r12
.L_line_loop:
    cmp r12, r9
    jge .L_done_all
    mov r13, r12
.L_scan:
    cmp r13, r9
    jge .L_no_newline
    movzx eax, byte ptr [r10 + r13]
    cmp al, 0x0A
    je .L_found_newline
    inc r13
    jmp .L_scan
.L_found_newline:
    mov r14, r13
    dec r14
    mov r15, r12
.L_write_rev_loop:
    cmp r14, r15
    jl .L_after_rev_write
    lea rsi, [r10 + r14]
    mov rdi, 1
    mov rax, 1
    mov rdx, 1
    syscall
    dec r14
    jmp .L_write_rev_loop
.L_after_rev_write:
    lea rsi, [rip + newline_char]
    mov rdi, 1
    mov rax, 1
    mov rdx, 1
    syscall
    inc r13
    mov r12, r13
    jmp .L_line_loop
.L_no_newline:
    cmp r12, r9
    jge .L_done_all
    mov r14, r9
    dec r14
    mov r15, r12
.L_write_rev_final:
    cmp r14, r15
    jl .L_after_rev_final
    lea rsi, [r10 + r14]
    mov rdi, 1
    mov rax, 1
    mov rdx, 1
    syscall
    dec r14
    jmp .L_write_rev_final
.L_after_rev_final:
    lea rsi, [rip + newline_char]
    mov rdi, 1
    mov rax, 1
    mov rdx, 1
    syscall
    jmp .L_done_all
.L_done_all:
    xor rdi, rdi
    mov rax, 60
    syscall

.section .rodata
newline_char: .byte 0x0A

.section .bss
.align 16
buffer: .space 65536
