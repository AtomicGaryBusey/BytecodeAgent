.section .rodata
hex_low:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf:
    .zero 65536
linebuf:
    .zero 80

.section .text
.globl _start
_start:
    xor r13d, r13d
    lea r12, [rip + input_buf]
.read_loop:
    mov rax, 65536
    sub rax, r13
    jz .read_done
    mov rdi, 0
    lea rsi, [r12 + r13]
    mov rdx, rax
    mov rax, 0
    syscall
    test rax, rax
    jle .read_done
    add r13, rax
    jmp .read_loop
.read_done:
    lea rbx, [rip + hex_low]
    xor r14d, r14d
    lea r15, [rip + linebuf]
.row_loop:
    cmp r14, r13
    jge .end_rows
    mov rax, r13
    sub rax, r14
    cmp rax, 16
    jbe .row_n_small
    mov rax, 16
.row_n_small:
    mov r10, rax
    mov rsi, r15
    mov r8, r14
    xor r9d, r9d
.offset_loop:
    mov ecx, r9d
    mov eax, 7
    sub eax, ecx
    shl eax, 2
    mov cl, al
    mov eax, r8d
    shr eax, cl
    and eax, 0xF
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rsi], al
    inc rsi
    inc r9d
    cmp r9d, 8
    jl .offset_loop
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    xor r9d, r9d
.hexpos_loop:
    mov r11d, r9d
    cmp r11d, r10d
    jge .print_blank
    mov r11, r12
    add r11, r14
    add r11, r9
    movzx eax, byte ptr [r11]
    mov edx, eax
    shr dl, 4
    and edx, 0xF
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    mov edx, eax
    and edx, 0xF
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    jmp .after_pos
.print_blank:
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
.after_pos:
    cmp r9d, 7
    jne .skip_extra
    mov byte ptr [rsi], ' '
    inc rsi
.skip_extra:
    inc r9d
    cmp r9d, 16
    jl .hexpos_loop
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], '|'
    inc rsi
    xor r9d, r9d
.ascii_loop:
    cmp r9d, r10d
    jge .ascii_end
    mov r11, r12
    add r11, r14
    add r11, r9
    movzx eax, byte ptr [r11]
    cmp al, 0x20
    jb .print_dot_ascii
    cmp al, 0x7E
    ja .print_dot_ascii
    mov byte ptr [rsi], al
    jmp .after_ascii
.print_dot_ascii:
    mov byte ptr [rsi], '.'
.after_ascii:
    inc rsi
    inc r9d
    jmp .ascii_loop
.ascii_end:
    mov byte ptr [rsi], '|'
    inc rsi
    mov byte ptr [rsi], 0x0A
    inc rsi
    mov rdx, rsi
    sub rdx, r15
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    syscall
    add r14, r10
    jmp .row_loop
.end_rows:
    mov rsi, r15
    mov r8, r13
    xor r9d, r9d
.total_offset_loop:
    mov ecx, r9d
    mov eax, 7
    sub eax, ecx
    shl eax, 2
    mov cl, al
    mov eax, r8d
    shr eax, cl
    and eax, 0xF
    movzx eax, byte ptr [rbx + rax]
    mov byte ptr [rsi], al
    inc rsi
    inc r9d
    cmp r9d, 8
    jl .total_offset_loop
    mov byte ptr [rsi], 0x0A
    inc rsi
    mov rdx, rsi
    sub rdx, r15
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    syscall
    mov rax, 231
    xor rdi, rdi
    syscall
