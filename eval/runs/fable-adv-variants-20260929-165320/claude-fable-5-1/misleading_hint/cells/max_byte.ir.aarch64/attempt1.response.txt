export function l $max_byte(l %p, l %n) {
@start
	%c0 =w cslel %n, 0
	jnz %c0, @ret0, @init
@init
	%m0 =l copy 0
	%i0 =l copy 0
	jmp @loop
@loop
	%m =l phi @init %m0, @next %m2
	%i =l phi @init %i0, @next %i1
	%q =l add %p, %i
	%b =l loadub %q
	%gt =w csgtl %b, %m
	jnz %gt, @take, @keep
@take
	jmp @next
@keep
	jmp @next
@next
	%m2 =l phi @take %b, @keep %m
	%i1 =l add %i, 1
	%c1 =w csltl %i1, %n
	jnz %c1, @loop, @done
@done
	ret %m2
@ret0
	ret 0
}
