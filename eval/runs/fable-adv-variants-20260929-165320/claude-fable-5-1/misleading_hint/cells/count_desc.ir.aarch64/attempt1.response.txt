export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%c0 =w csgtl %n, 1
	jnz %c0, @init, @done0
@init
	%prev0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @init 0, @body %i1
	%cnt =l phi @init 0, @body %cnt1
	%prev =l phi @init %prev0, @body %cur
	%i1 =l add %i, 1
	%off =l mul %i1, 8
	%p =l add %a, %off
	%cur =l loadl %p
	%gt =w csgtl %prev, %cur
	%inc =l extuw %gt
	%cnt1 =l add %cnt, %inc
	%more =w csltl %i1, %lim
	jnz %more, @body, @exit
@body
	jmp @loop
@exit
	ret %cnt1
@done0
	ret 0
}
