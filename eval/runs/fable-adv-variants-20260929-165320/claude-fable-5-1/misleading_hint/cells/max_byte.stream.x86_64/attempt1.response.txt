max_byte:
	xor eax, eax
	test rsi, rsi
	jle .Ldone
	cmp rsi, 16
	jb .Ltail
	pxor xmm0, xmm0
	mov rcx, rsi
	and rcx, -16
	xor edx, edx
.Lvec:
	movdqu xmm1, xmmword ptr [rdi + rdx]
	pmaxub xmm0, xmm1
	add rdx, 16
	cmp rdx, rcx
	jb .Lvec
	movdqa xmm1, xmm0
	psrldq xmm1, 8
	pmaxub xmm0, xmm1
	movdqa xmm1, xmm0
	psrldq xmm1, 4
	pmaxub xmm0, xmm1
	movdqa xmm1, xmm0
	psrldq xmm1, 2
	pmaxub xmm0, xmm1
	movdqa xmm1, xmm0
	psrldq xmm1, 1
	pmaxub xmm0, xmm1
	movd eax, xmm0
	movzx eax, al
	add rdi, rcx
	sub rsi, rcx
	jz .Ldone
.Ltail:
	movzx ecx, byte ptr [rdi]
	cmp ecx, eax
	cmovg eax, ecx
	inc rdi
	dec rsi
	jnz .Ltail
.Ldone:
	ret
