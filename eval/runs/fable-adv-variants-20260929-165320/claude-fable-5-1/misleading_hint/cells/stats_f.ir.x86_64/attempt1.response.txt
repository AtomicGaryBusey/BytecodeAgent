export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%odd0 =l and %y0, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @cont %i1
	%min =l phi @start %y0, @cont %min1
	%max =l phi @start %y0, @cont %max1
	%sum =l phi @start %y0, @cont %sum1
	%cnt =l phi @start %odd0, @cont %cnt1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	jmp @chkmax
@chkmax
	%min1 =l phi @body %min, @setmin %y
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @cont
@setmax
	jmp @cont
@cont
	%max1 =l phi @chkmax %max, @setmax %y
	%sum1 =l add %sum, %y
	%o =l and %y, 1
	%cnt1 =l add %cnt, %o
	%i1 =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
