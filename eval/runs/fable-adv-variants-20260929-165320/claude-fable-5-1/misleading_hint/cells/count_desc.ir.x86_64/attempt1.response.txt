export function l $count_desc(l %a, l %n) {
@start
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret0, @init
@init
	%m =l sub %n, 1
	%prev =l loadl %a
	%i =l copy 1
	%cnt =l copy 0
	jmp @loop
@loop
	%ip =l phi @init %i, @body %i2
	%cp =l phi @init %cnt, @body %cnt2
	%pv =l phi @init %prev, @body %cur
	%done =w csgel %ip, %n
	jnz %done, @out, @body
@body
	%off =l mul %ip, 8
	%p =l add %a, %off
	%cur =l loadl %p
	%gt =w csgtl %pv, %cur
	%gtl =l extuw %gt
	%cnt2 =l add %cp, %gtl
	%i2 =l add %ip, 1
	jmp @loop
@out
	ret %cp
@ret0
	ret 0
}
