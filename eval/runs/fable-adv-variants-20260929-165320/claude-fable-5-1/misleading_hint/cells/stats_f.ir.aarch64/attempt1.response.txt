export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%p1 =l add %a, 8
	%end =l mul %n, 8
	%end =l add %a, %end
@loop
	%p =l phi @start %p1, @body %pn
	%min =l phi @start %y0, @body %minn
	%max =l phi @start %y0, @body %maxn
	%sum =l phi @start %y0, @body %sumn
	%cnt =l phi @start %odd0, @body %cntn
	%c =w cultl %p, %end
	jnz %c, @body, @done
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%lt2 =l extuw %lt
	%d1 =l sub %min, %y
	%m1 =l mul %d1, %lt2
	%minn =l sub %min, %m1
	%gt =w csgtl %y, %max
	%gt2 =l extuw %gt
	%d2 =l sub %y, %max
	%m2 =l mul %d2, %gt2
	%maxn =l add %max, %m2
	%sumn =l add %sum, %y
	%o =l and %y, 1
	%cntn =l add %cnt, %o
	%pn =l add %p, 8
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
