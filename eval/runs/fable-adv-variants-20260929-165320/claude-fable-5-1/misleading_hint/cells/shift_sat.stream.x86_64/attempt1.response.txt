shift_sat:
	mov rcx, rsi
	mov rdx, 63
	cmp rcx, rdx
	cmova rcx, rdx          # clamp shift count to 63
	mov rax, rdi
	sar rax, cl             # x >> n (arithmetic), correct for all n via clamp
	mov rdx, rdi
	shl rdx, cl             # x << min(n,63)
	xor r8d, r8d
	cmp rsi, 63
	cmova rdx, r8           # n >= 64: left shift result is 0
	add rax, rdx
	ret
