export function l $max_byte(l %p, l %n) {
@start
	%z =w csltl %n, 1
	jnz %z, @ret0, @loop
@loop
	%i =l phi @start 0, @body %i2
	%m =l phi @start 0, @body %m2
	%q =l add %p, %i
	%b =l loadub %q
	%c =w csltl %m, %b
	jnz %c, @take, @keep
@take
	%m3 =l copy %b
	jmp @body
@keep
	%m4 =l copy %m
	jmp @body
@body
	%m2 =l phi @take %m3, @keep %m4
	%i2 =l add %i, 1
	%d =w csltl %i2, %n
	jnz %d, @loop, @done
@done
	ret %m2
@ret0
	ret 0
}
