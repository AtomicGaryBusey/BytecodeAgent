stats_f:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	sub rsp, 8
	mov rbx, rdi
	mov rbp, rsi
	mov r12, rdx
	mov r13, rcx
	xor r15d, r15d
	mov rdi, qword ptr [rbx]
	call r12
	mov qword ptr [r13], rax
	mov qword ptr [r13 + 8], rax
	mov qword ptr [r13 + 16], rax
	and eax, 1
	add r15, rax
	mov r14d, 1
	cmp r14, rbp
	jge .Ldone
.Lloop:
	mov rdi, qword ptr [rbx + r14*8]
	call r12
	add qword ptr [r13 + 16], rax
	cmp rax, qword ptr [r13]
	jge .Lnomin
	mov qword ptr [r13], rax
.Lnomin:
	cmp rax, qword ptr [r13 + 8]
	jle .Lnomax
	mov qword ptr [r13 + 8], rax
.Lnomax:
	and eax, 1
	add r15, rax
	inc r14
	cmp r14, rbp
	jl .Lloop
.Ldone:
	mov rax, r15
	add rsp, 8
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
