export function l $shift_sat(l %x, l %n) {
@start
	%ge =w csgel %n, 64
	%gel =l extuw %ge
	%mask =l sub 0, %gel
	%nmask =l xor %mask, -1
	%shl =l shl %x, %n
	%shl2 =l and %shl, %nmask
	%a =l and %n, %nmask
	%b =l and %mask, 63
	%nsar =l or %a, %b
	%sar =l sar %x, %nsar
	%r =l add %shl2, %sar
	ret %r
}
