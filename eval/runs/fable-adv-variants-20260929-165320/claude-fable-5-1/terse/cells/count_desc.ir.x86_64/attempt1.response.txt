export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
	%c =l copy 0
	%ok =w csgtl %m, 0
	jnz %ok, @loop, @done
@loop
	%x =l loadl %a
	%a =l add %a, 8
	%y =l loadl %a
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%c =l add %c, %gl
	%m =l sub %m, 1
	%ok =w csgtl %m, 0
	jnz %ok, @loop, @done
@done
	ret %c
}
