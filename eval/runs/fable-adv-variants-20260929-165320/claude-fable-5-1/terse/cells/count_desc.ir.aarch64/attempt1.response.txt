export function l $count_desc(l %a, l %n) {
@start
	%s =l mul %n, 8
	%e =l add %a, %s
	%end =l sub %e, 8
	%c =w csltl %a, %end
	jnz %c, @loop, @done
@loop
	%p =l phi @start %a, @loop %q
	%cnt =l phi @start 0, @loop %cnt2
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%cnt2 =l add %cnt, %gl
	%c2 =w csltl %q, %end
	jnz %c2, @loop, @done
@done
	%r =l phi @start 0, @loop %cnt2
	ret %r
}
