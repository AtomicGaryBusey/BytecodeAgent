export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%v0 =l loadl %a
	%y0 =l call %f(l %v0)
	%c0 =l and %y0, 1
	%e =l mul %n, 8
	%end =l add %a, %e
	%p0 =l add %a, 8
@loop
	%p =l phi @start %p0, @body %p1
	%mn =l phi @start %y0, @body %mn1
	%mx =l phi @start %y0, @body %mx1
	%s =l phi @start %y0, @body %s1
	%c =l phi @start %c0, @body %c1
	%done =w ceql %p, %end
	jnz %done, @exit, @body
@body
	%v =l loadl %p
	%y =l call %f(l %v)
	%lt =w csltl %y, %mn
	%ltl =l extsw %lt
	%mk1 =l neg %ltl
	%d1 =l sub %mn, %y
	%m1 =l and %d1, %mk1
	%mn1 =l sub %mn, %m1
	%gt =w csgtl %y, %mx
	%gtl =l extsw %gt
	%mk2 =l neg %gtl
	%d2 =l sub %y, %mx
	%m2 =l and %d2, %mk2
	%mx1 =l add %mx, %m2
	%s1 =l add %s, %y
	%o =l and %y, 1
	%c1 =l add %c, %o
	%p1 =l add %p, 8
	jmp @loop
@exit
	storel %mn, %out
	%o1 =l add %out, 8
	storel %mx, %o1
	%o2 =l add %out, 16
	storel %s, %o2
	ret %c
}
