export function l $count_ugt(l %a, l %n, l %t) {
@start
	%end =l mul %n, 8
	%end =l add %a, %end
@loop
	%p =l phi @start %a, @body %p1
	%c =l phi @start 0, @body %c1
	%done =w ceql %p, %end
	jnz %done, @ret, @body
@body
	%v =l loadl %p
	%g =w cugtl %v, %t
	%ge =l extuw %g
	%c1 =l add %c, %ge
	%p1 =l add %p, 8
	jmp @loop
@ret
	ret %c
}
