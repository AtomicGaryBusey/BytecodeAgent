export function l $fill_down(l %a, l %n, l %start) {
@start
	%end =l mul %n, 8
	%end =l add %a, %end
	jmp @cond
@cond
	%p =l phi @start %a, @loop %p1
	%v =l phi @start %start, @loop %v1
	%c =w cnel %p, %end
	jnz %c, @loop, @done
@loop
	storel %v, %p
	%p1 =l add %p, 8
	%v1 =l sub %v, 1
	jmp @cond
@done
	ret %n
}
