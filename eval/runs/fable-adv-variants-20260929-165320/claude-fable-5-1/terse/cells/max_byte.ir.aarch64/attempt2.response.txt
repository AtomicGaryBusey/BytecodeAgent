export function l $max_byte(l %p, l %n) {
@start
	%end =l add %p, %n
@loop
	%m =l phi @start 0, @body %m, @upd %b
	%q =l phi @start %p, @body %q2, @upd %q2
	%c =w cultl %q, %end
	jnz %c, @body, @done
@body
	%b =l loadub %q
	%q2 =l add %q, 1
	%g =w cultl %m, %b
	jnz %g, @upd, @loop
@upd
	jmp @loop
@done
	ret %m
}
