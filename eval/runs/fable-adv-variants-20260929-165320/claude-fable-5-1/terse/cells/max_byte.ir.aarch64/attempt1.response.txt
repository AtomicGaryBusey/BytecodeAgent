export function l $max_byte(l %p, l %n) {
@start
	%end =l add %p, %n
	%m0 =l copy 0
@loop
	%m =l phi @start %m0, @body %m2
	%q =l phi @start %p, @body %q2
	%c =w cultl %q, %end
	jnz %c, @body, @done
@body
	%b =l loadub %q
	%g =w cultl %m, %b
	%d =l sub %b, %m
	%dm =l mul %d, %g
	%m2 =l add %m, %dm
	%q2 =l add %q, 1
	jmp @loop
@done
	ret %m
}
