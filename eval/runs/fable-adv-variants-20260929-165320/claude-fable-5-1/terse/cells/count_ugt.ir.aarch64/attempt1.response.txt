export function l $count_ugt(l %a, l %n, l %t) {
@start
	%end =l mul %n, 8
	%end =l add %a, %end
@loop
	%p =l phi @start %a, @body %p2
	%c =l phi @start 0, @body %c2
	%more =w cultl %p, %end
	jnz %more, @body, @done
@body
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gt =l extuw %gt
	%c2 =l add %c, %gt
	%p2 =l add %p, 8
	jmp @loop
@done
	ret %c
}
