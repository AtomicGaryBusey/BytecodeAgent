export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%end =l mul %n, 8
	%end =l add %src, %end
	%cnt =l copy 0
	%s =l copy %src
	%d =l copy %dst
@loop
	%c =w cnel %s, %end
	jnz %c, @body, @done
@body
	%x =l loadl %s
	%r =l call %f(l %x)
	storel %r, %d
	%lt =w csltl %r, %x
	%ltl =l extuw %lt
	%cnt =l add %cnt, %ltl
	%s =l add %s, 8
	%d =l add %d, 8
	jmp @loop
@done
	ret %cnt
}
