export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @cont %inext
	%max =l phi @start %first, @cont %nextmax
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csltl %max, %val
	jnz %gt, @upd, @skip
@upd
	jmp @cont
@skip
	jmp @cont
@cont
	%nextmax =l phi @upd %val, @skip %max
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
