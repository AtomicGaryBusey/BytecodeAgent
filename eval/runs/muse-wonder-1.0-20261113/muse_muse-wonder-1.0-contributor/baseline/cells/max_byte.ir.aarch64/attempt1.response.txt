export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @inc %inext
	%max =l phi @start 0, @inc %maxcur
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %max
	jnz %gt, @update, @inc
@update
	jmp @inc
@inc
	%maxcur =l phi @body %max, @update %b
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
