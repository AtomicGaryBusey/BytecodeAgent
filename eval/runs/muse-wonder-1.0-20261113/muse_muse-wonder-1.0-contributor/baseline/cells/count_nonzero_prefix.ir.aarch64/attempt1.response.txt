export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @start %zero, @inc %inext
	%cond =w csltl %i, %n
	jnz %cond, @check, @done
@check
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @inc, @done
@inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
