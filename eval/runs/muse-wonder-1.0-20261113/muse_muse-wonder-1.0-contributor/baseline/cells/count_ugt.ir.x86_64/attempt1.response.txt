export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%cnt =l phi @start 0, @body %cnt_next
	%i =l phi @start 0, @body %i_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	%gtl =l extuw %gt
	%cnt_next =l add %cnt, %gtl
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
