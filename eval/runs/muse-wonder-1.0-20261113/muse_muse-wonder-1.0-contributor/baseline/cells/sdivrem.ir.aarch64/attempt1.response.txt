export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bzero =w ceql %b, 0
	jnz %bzero, @bzero, @checkneg1
@checkneg1
	%bneg1 =w ceql %b, -1
	jnz %bneg1, @neg1, @normal
@bzero
	storel %a, %rem
	ret 0
@neg1
	%qneg =l sub 0, %a
	%zero =l copy 0
	storel %zero, %rem
	ret %qneg
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
