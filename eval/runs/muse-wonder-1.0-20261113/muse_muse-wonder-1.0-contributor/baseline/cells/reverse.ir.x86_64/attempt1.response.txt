export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @end
@body
	%ioff =l mul %i, 8
	%iptr =l add %a, %ioff
	%joff =l mul %j, 8
	%jptr =l add %a, %joff
	%ival =l loadl %iptr
	%jval =l loadl %jptr
	storel %jval, %iptr
	storel %ival, %jptr
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@end
	ret 0
}
