export function l $shift_sat(l %x, l %n) {
@start
	jmp @loop
@loop
	%left =l phi @start %x, @loopbody %left2
	%right =l phi @start %x, @loopbody %right2
	%cnt =l phi @start %n, @loopbody %cnt2
	%done =w ceql %cnt, 0
	jnz %done, @end, @loopbody
@loopbody
	%left2 =l shl %left, 1
	%right2 =l sar %right, 1
	%cnt2 =l sub %cnt, 1
	jmp @loop
@end
	%res =l add %left, %right
	ret %res
}
