export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @outer_next %inext
	%crc =w phi @start -1, @outer_next %crcdone
	%more =w csltl %i, %n
	jnz %more, @body, @finish
@body
	%ptr =l add %p, %i
	%byte =w loadub %ptr
	%crc0 =w xor %crc, %byte
	jmp @inner
@inner
	%k =l phi @body 0, @inner_next %knext
	%crci =w phi @body %crc0, @inner_next %crcnext
	%kmore =w csltl %k, 8
	jnz %kmore, @innerbody, @outer_next
@innerbody
	%bit =w and %crci, 1
	jnz %bit, @poly, @nopoly
@poly
	%shiftedp =w shr %crci, 1
	%crcp =w xor %shiftedp, -306674912
	jmp @inner_next
@nopoly
	%crcnp =w shr %crci, 1
	jmp @inner_next
@inner_next
	%crcnext =w phi @poly %crcp, @nopoly %crcnp
	%knext =l add %k, 1
	jmp @inner
@outer_next
	%crcdone =w copy %crci
	%inext =l add %i, 1
	jmp @outer
@finish
	%final =w xor %crc, -1
	%res =l extuw %final
	ret %res
}
