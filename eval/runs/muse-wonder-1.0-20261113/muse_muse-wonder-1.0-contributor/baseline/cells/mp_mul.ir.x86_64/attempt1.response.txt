export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	jmp @init
@init
	%ii =l phi @start 0, @initbody %iinext
	%inotdone =w csltl %ii, %total
	jnz %inotdone, @initbody, @outer
@initbody
	%iioff =l mul %ii, 8
	%iiptr =l add %r, %iioff
	storel 0, %iiptr
	%iinext =l add %ii, 1
	jmp @init
@outer
	%i =l phi @init 0, @storecarry %inext
	%onotdone =w csltl %i, %na
	jnz %onotdone, @inner, @trim
@inner
	%j =l phi @outer 0, @innerbody %jnext
	%carry =l phi @outer 0, @innerbody %carrynext
	%jnotdone =w csltl %j, %nb
	jnz %jnotdone, @innerbody, @storecarry
@innerbody
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bv =l loadl %bptr
	%x0 =l and %av, 4294967295
	%x1 =l shr %av, 32
	%y0 =l and %bv, 4294967295
	%y1 =l shr %bv, 32
	%p0 =l mul %x0, %y0
	%p1 =l mul %x0, %y1
	%p2 =l mul %x1, %y0
	%p3 =l mul %x1, %y1
	%p0h =l shr %p0, 32
	%p1l =l and %p1, 4294967295
	%p2l =l and %p2, 4294967295
	%tmid1 =l add %p0h, %p1l
	%tmid =l add %tmid1, %p2l
	%d1 =l and %tmid, 4294967295
	%cmid =l shr %tmid, 32
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%p3l =l and %p3, 4294967295
	%th1 =l add %p1h, %p2h
	%th2 =l add %th1, %p3l
	%th =l add %th2, %cmid
	%d2 =l and %th, 4294967295
	%chigh =l shr %th, 32
	%p3h =l shr %p3, 32
	%d3 =l add %p3h, %chigh
	%d0 =l and %p0, 4294967295
	%d1s =l shl %d1, 32
	%pl =l add %d0, %d1s
	%d3s =l shl %d3, 32
	%ph =l add %d2, %d3s
	%ijidx =l add %i, %j
	%ijoff =l mul %ijidx, 8
	%ijptr =l add %r, %ijoff
	%rk =l loadl %ijptr
	%s1 =l add %pl, %rk
	%c1w =w cultl %s1, %pl
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	storel %s2, %ijptr
	%c1 =l extuw %c1w
	%c2 =l extuw %c2w
	%carrytmp =l add %ph, %c1
	%carrynext =l add %carrytmp, %c2
	%jnext =l add %j, 1
	jmp @inner
@storecarry
	%scidx =l add %i, %nb
	%scoff =l mul %scidx, 8
	%scptr =l add %r, %scoff
	storel %carry, %scptr
	%inext =l add %i, 1
	jmp @outer
@trim
	%k =l phi @outer %total, @trimbody %knext
	%kpos =w csgtl %k, 0
	jnz %kpos, @trimcheck, @done
@trimcheck
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kptr =l add %r, %koff
	%kval =l loadl %kptr
	%kiszero =w ceql %kval, 0
	jnz %kiszero, @trimbody, @done
@trimbody
	%knext =l copy %km1
	jmp @trim
@done
	ret %k
}
