export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%sum =l phi @start 0, @body %sumnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%r =l call %f(l %x)
	%sumnext =l add %sum, %r
	%inext =l add %i, 1
	jmp @loop
@done
	ret %sum
}
