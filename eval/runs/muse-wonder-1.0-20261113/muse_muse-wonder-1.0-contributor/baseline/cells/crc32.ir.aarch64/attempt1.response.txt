export function l $crc32(l %p, l %n) {
@start
	%poly =w copy -306674912
	jmp @loop
@loop
	%crc =w phi @start -1, @body %crcnext
	%i =l phi @start 0, @body %inext
	%has =w csltl %i, %n
	jnz %has, @body, @done
@body
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crcb =w xor %crc, %b
	%bit0 =w and %crcb, 1
	%mask0 =w sub 0, %bit0
	%shr0 =w shr %crcb, 1
	%pm0 =w and %poly, %mask0
	%c0 =w xor %shr0, %pm0
	%bit1 =w and %c0, 1
	%mask1 =w sub 0, %bit1
	%shr1 =w shr %c0, 1
	%pm1 =w and %poly, %mask1
	%c1 =w xor %shr1, %pm1
	%bit2 =w and %c1, 1
	%mask2 =w sub 0, %bit2
	%shr2 =w shr %c1, 1
	%pm2 =w and %poly, %mask2
	%c2 =w xor %shr2, %pm2
	%bit3 =w and %c2, 1
	%mask3 =w sub 0, %bit3
	%shr3 =w shr %c2, 1
	%pm3 =w and %poly, %mask3
	%c3 =w xor %shr3, %pm3
	%bit4 =w and %c3, 1
	%mask4 =w sub 0, %bit4
	%shr4 =w shr %c3, 1
	%pm4 =w and %poly, %mask4
	%c4 =w xor %shr4, %pm4
	%bit5 =w and %c4, 1
	%mask5 =w sub 0, %bit5
	%shr5 =w shr %c4, 1
	%pm5 =w and %poly, %mask5
	%c5 =w xor %shr5, %pm5
	%bit6 =w and %c5, 1
	%mask6 =w sub 0, %bit6
	%shr6 =w shr %c5, 1
	%pm6 =w and %poly, %mask6
	%c6 =w xor %shr6, %pm6
	%bit7 =w and %c6, 1
	%mask7 =w sub 0, %bit7
	%shr7 =w shr %c6, 1
	%pm7 =w and %poly, %mask7
	%crcnext =w xor %shr7, %pm7
	%inext =l add %i, 1
	jmp @loop
@done
	%fin =w xor %crc, -1
	%res =l extuw %fin
	ret %res
}
