export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @low %lo_next, @high %lo
	%hi =l phi @start %n, @low %hi, @high %hi_next
	%cont =w csltl %lo, %hi
	jnz %cont, @mid, @notfound
@mid
	%sum =l add %lo, %hi
	%midv =l div %sum, 2
	%off =l shl %midv, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @compare
@found
	ret %midv
@compare
	%lt =w csltl %val, %key
	jnz %lt, @low, @high
@low
	%lo_next =l add %midv, 1
	jmp @loop
@high
	%hi_next =l copy %midv
	jmp @loop
@notfound
	ret -1
}
