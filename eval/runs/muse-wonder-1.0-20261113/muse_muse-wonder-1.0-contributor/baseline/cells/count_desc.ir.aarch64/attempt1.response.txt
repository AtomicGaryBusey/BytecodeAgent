export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%cond =w csltl %i, %lim
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%pnext =l add %p, 8
	%y =l loadl %pnext
	%gt =w csgtl %x, %y
	%inc =l extuw %gt
	%cntnext =l add %cnt, %inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
