export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%x =l loadl %srcp
	%r =l call %f(l %x)
	%dstp =l add %dst, %off
	storel %r, %dstp
	%lt =w csltl %r, %x
	%ltl =l extsw %lt
	%cntnext =l add %cnt, %ltl
	%inext =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
