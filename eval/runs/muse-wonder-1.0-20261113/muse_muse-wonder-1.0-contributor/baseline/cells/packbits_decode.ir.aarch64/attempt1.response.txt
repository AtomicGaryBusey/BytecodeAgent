export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%psi =l alloc8 8
	%poi =l alloc8 8
	%pi =l alloc8 8
	%pcnt =l alloc8 8
	%pov =l alloc8 8
	%pbv =l alloc4 4
	storel 0, %psi
	storel 0, %poi
	storel 0, %pi
	storel 0, %pcnt
	storel 0, %pov
	jmp @loop
@loop
	%si0 =l loadl %psi
	%has0 =w csltl %si0, %n
	jnz %has0, @hdr, @fin
@fin
	%oi0 =l loadl %poi
	ret %oi0
@hdr
	%si1 =l loadl %psi
	%ph0 =l add %src, %si1
	%h0 =w loadsb %ph0
	%isnop0 =w ceqw %h0, -128
	jnz %isnop0, @nop, @notnop
@nop
	%si2 =l loadl %psi
	%si3 =l add %si2, 1
	storel %si3, %psi
	jmp @loop
@notnop
	%isneg0 =w csltw %h0, 0
	jnz %isneg0, @repchk, @litchk
@litchk
	%hl0 =l extsw %h0
	%cnt0 =l add %hl0, 1
	%si4 =l loadl %psi
	%hdrend0 =l add %si4, 1
	%srcend0 =l add %hdrend0, %cnt0
	%sok0 =w cslel %srcend0, %n
	jnz %sok0, @litcap, @errsrc
@litcap
	%oi1 =l loadl %poi
	%outend0 =l add %oi1, %cnt0
	%cok0 =w cslel %outend0, %cap
	jnz %cok0, @litnorm, @littrunc
@litnorm
	storel %cnt0, %pcnt
	storel 0, %pi
	storel 0, %pov
	jmp @litloop
@littrunc
	%oi2 =l loadl %poi
	%avail0 =l sub %cap, %oi2
	storel %avail0, %pcnt
	storel 0, %pi
	storel 1, %pov
	jmp @litloop
@litloop
	%i0 =l loadl %pi
	%cc0 =l loadl %pcnt
	%more0 =w csltl %i0, %cc0
	jnz %more0, @litbody, @litdone
@litbody
	%i1 =l loadl %pi
	%si5 =l loadl %psi
	%sidx0 =l add %si5, 1
	%sidx1 =l add %sidx0, %i1
	%ps0 =l add %src, %sidx1
	%bt0 =w loadub %ps0
	%oi3 =l loadl %poi
	%didx0 =l add %oi3, %i1
	%pd0 =l add %out, %didx0
	storeb %bt0, %pd0
	%i2 =l add %i1, 1
	storel %i2, %pi
	jmp @litloop
@litdone
	%ov0 =l loadl %pov
	%isov0 =w cnel %ov0, 0
	jnz %isov0, @errcap, @litadv
@litadv
	%si6 =l loadl %psi
	%cc1 =l loadl %pcnt
	%si7 =l add %si6, 1
	%si8 =l add %si7, %cc1
	storel %si8, %psi
	%oi4 =l loadl %poi
	%oi5 =l add %oi4, %cc1
	storel %oi5, %poi
	jmp @loop
@repchk
	%si9 =l loadl %psi
	%need0 =l add %si9, 2
	%sok1 =w cslel %need0, %n
	jnz %sok1, @repcap, @errsrc
@repcap
	%hl1 =l extsw %h0
	%cnt1 =l sub 1, %hl1
	%oi6 =l loadl %poi
	%outend1 =l add %oi6, %cnt1
	%cok1 =w cslel %outend1, %cap
	jnz %cok1, @repnorm, @reptrunc
@repnorm
	storel %cnt1, %pcnt
	storel 0, %pi
	storel 0, %pov
	%si10 =l loadl %psi
	%ridx0 =l add %si10, 1
	%pr0 =l add %src, %ridx0
	%rb0 =w loadub %pr0
	storeb %rb0, %pbv
	jmp @reploop
@reptrunc
	%oi7 =l loadl %poi
	%avail1 =l sub %cap, %oi7
	storel %avail1, %pcnt
	storel 0, %pi
	storel 1, %pov
	%si11 =l loadl %psi
	%ridx1 =l add %si11, 1
	%pr1 =l add %src, %ridx1
	%rb1 =w loadub %pr1
	storeb %rb1, %pbv
	jmp @reploop
@reploop
	%i3 =l loadl %pi
	%cc2 =l loadl %pcnt
	%more1 =w csltl %i3, %cc2
	jnz %more1, @repbody, @repdone
@repbody
	%i4 =l loadl %pi
	%rb2 =w loadub %pbv
	%oi8 =l loadl %poi
	%didx1 =l add %oi8, %i4
	%pd1 =l add %out, %didx1
	storeb %rb2, %pd1
	%i5 =l add %i4, 1
	storel %i5, %pi
	jmp @reploop
@repdone
	%ov1 =l loadl %pov
	%isov1 =w cnel %ov1, 0
	jnz %isov1, @errcap, @repadv
@repadv
	%si12 =l loadl %psi
	%si13 =l add %si12, 2
	storel %si13, %psi
	%cc3 =l loadl %pcnt
	%oi9 =l loadl %poi
	%oi10 =l add %oi9, %cc3
	storel %oi10, %poi
	jmp @loop
@errsrc
	%m2 =l copy -2
	ret %m2
@errcap
	%m1 =l copy -1
	ret %m1
}
