export function l $fib_rec(l %n) {
@start
	%is0 =w ceql %n, 0
	jnz %is0, @base0, @check1
@base0
	ret 0
@check1
	%is1 =w ceql %n, 1
	jnz %is1, @base1, @rec
@base1
	ret 1
@rec
	%nm1 =l sub %n, 1
	%a =l call $fib_rec(l %nm1)
	%nm2 =l sub %n, 2
	%b =l call $fib_rec(l %nm2)
	%sum =l add %a, %b
	ret %sum
}
