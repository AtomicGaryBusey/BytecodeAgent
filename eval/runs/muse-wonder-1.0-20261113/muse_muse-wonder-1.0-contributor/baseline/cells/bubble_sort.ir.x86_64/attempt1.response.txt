export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer_cond
@outer_cond
	%i =l phi @start 0, @outer_next %i_next
	%oc =w csltl %i, %n
	jnz %oc, @inner_cond, @done
@inner_cond
	%j =l phi @outer_cond 0, @inner_next %j_next
	%n1 =l sub %n, 1
	%lim =l sub %n1, %i
	%ic =w csltl %j, %lim
	jnz %ic, @inner_body, @outer_next
@inner_body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%off1 =l add %off, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%cmp =w csgtl %v0, %v1
	jnz %cmp, @swap, @inner_next
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_next
@inner_next
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_next
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
