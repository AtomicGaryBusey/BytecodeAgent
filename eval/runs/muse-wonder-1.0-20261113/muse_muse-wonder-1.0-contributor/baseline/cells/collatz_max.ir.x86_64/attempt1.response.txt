export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @outernext %knext
	%max =l phi @start 0, @outernext %maxnext
	%c1 =w cslel %k, %n
	jnz %c1, @inner, @done
@inner
	%x =l phi @outer %k, @innercont %xnext
	%steps =l phi @outer 0, @innercont %stepsnext
	%c2 =w ceql %x, 1
	jnz %c2, @update, @step
@step
	%low =l and %x, 1
	%c3 =w ceql %low, 1
	jnz %c3, @odd, @even
@even
	%xe =l div %x, 2
	jmp @innercont
@odd
	%xo =l mul %x, 3
	%xo1 =l add %xo, 1
	jmp @innercont
@innercont
	%xnext =l phi @even %xe, @odd %xo1
	%stepsnext =l add %steps, 1
	jmp @inner
@update
	%c4 =w csgtl %steps, %max
	jnz %c4, @newmax, @keepmax
@newmax
	jmp @outernext
@keepmax
	jmp @outernext
@outernext
	%maxnext =l phi @newmax %steps, @keepmax %max
	%knext =l add %k, 1
	jmp @outer
@done
	ret %max
}
