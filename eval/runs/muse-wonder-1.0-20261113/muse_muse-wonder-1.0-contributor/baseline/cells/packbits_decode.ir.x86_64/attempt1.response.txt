export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l copy 0
	%one =l copy 1
	%mone =l copy -1
	%mtwo =l copy -2
	%m128 =l copy -128
	%pi =l alloc8 8
	%po =l alloc8 8
	%pk =l alloc8 8
	%plen =l alloc8 8
	%pfill =l alloc8 8
	%pb =l alloc4 4
	storel %zero, %pi
	storel %zero, %po
	jmp @header
@header
	%ih =l loadl %pi
	%has =w csltl %ih, %n
	jnz %has, @havehdr, @done
@done
	%odone =l loadl %po
	ret %odone
@havehdr
	%phdr =l add %src, %ih
	%hb =w loadsb %phdr
	%h =l extsw %hb
	%inext =l add %ih, %one
	storel %inext, %pi
	%isnoop =w ceql %h, %m128
	jnz %isnoop, @header, @dispatch
@dispatch
	%isneg =w csltl %h, %zero
	jnz %isneg, @repeat, @literal
@literal
	%lenlit =l add %h, %one
	storel %lenlit, %plen
	%iaftl =l loadl %pi
	%reml =l sub %n, %iaftl
	%badl =w csgtl %lenlit, %reml
	jnz %badl, @err2, @litcap
@litcap
	%oll =l loadl %po
	%lenll =l loadl %plen
	%endl =l add %oll, %lenll
	%overl =w csgtl %endl, %cap
	jnz %overl, @litoverinit, @litnorminit
@litnorminit
	storel %zero, %pk
	jmp @litnormloop
@litnormloop
	%knl =l loadl %pk
	%lennl =l loadl %plen
	%morenl =w csltl %knl, %lennl
	jnz %morenl, @litnormbody, @litnormdone
@litnormbody
	%knb =l loadl %pk
	%inb =l loadl %pi
	%sposnb =l add %inb, %knb
	%psrcnb =l add %src, %sposnb
	%bytenb =w loadub %psrcnb
	%onb =l loadl %po
	%oposnb =l add %onb, %knb
	%poutnb =l add %out, %oposnb
	storeb %bytenb, %poutnb
	%knextnb =l add %knb, %one
	storel %knextnb, %pk
	jmp @litnormloop
@litnormdone
	%ind =l loadl %pi
	%lend =l loadl %plen
	%inewd =l add %ind, %lend
	storel %inewd, %pi
	%ond =l loadl %po
	%onewd =l add %ond, %lend
	storel %onewd, %po
	jmp @header
@litoverinit
	%ooi =l loadl %po
	%filloi =l sub %cap, %ooi
	storel %filloi, %pfill
	storel %zero, %pk
	jmp @litoverloop
@litoverloop
	%kol =l loadl %pk
	%fillol =l loadl %pfill
	%moreol =w csltl %kol, %fillol
	jnz %moreol, @litoverbody, @err1
@litoverbody
	%kob =l loadl %pk
	%iob =l loadl %pi
	%sposob =l add %iob, %kob
	%psrcob =l add %src, %sposob
	%byteob =w loadub %psrcob
	%oob =l loadl %po
	%oposob =l add %oob, %kob
	%poutob =l add %out, %oposob
	storeb %byteob, %poutob
	%knextob =l add %kob, %one
	storel %knextob, %pk
	jmp @litoverloop
@repeat
	%cntr =l sub %one, %h
	storel %cntr, %plen
	%iaftr =l loadl %pi
	%remr =l sub %n, %iaftr
	%badr =w csltl %remr, %one
	jnz %badr, @err2, @repload
@repload
	%irb =l loadl %pi
	%prb =l add %src, %irb
	%byterb =w loadub %prb
	storeb %byterb, %pb
	%irbnext =l add %irb, %one
	storel %irbnext, %pi
	%orb =l loadl %po
	%cntrb =l loadl %plen
	%endrb =l add %orb, %cntrb
	%overrb =w csgtl %endrb, %cap
	jnz %overrb, @repoverinit, @repnorminit
@repnorminit
	storel %zero, %pk
	jmp @repnormloop
@repnormloop
	%krnl =l loadl %pk
	%cntrnl =l loadl %plen
	%morernl =w csltl %krnl, %cntrnl
	jnz %morernl, @repnormbody, @repnormdone
@repnormbody
	%krnb =l loadl %pk
	%brnb =w loadub %pb
	%ornb =l loadl %po
	%oposrnb =l add %ornb, %krnb
	%poutrnb =l add %out, %oposrnb
	storeb %brnb, %poutrnb
	%knextrnb =l add %krnb, %one
	storel %knextrnb, %pk
	jmp @repnormloop
@repnormdone
	%ornd =l loadl %po
	%cntrnd =l loadl %plen
	%onewrd =l add %ornd, %cntrnd
	storel %onewrd, %po
	jmp @header
@repoverinit
	%oroi =l loadl %po
	%fillroi =l sub %cap, %oroi
	storel %fillroi, %pfill
	storel %zero, %pk
	jmp @repoverloop
@repoverloop
	%krol =l loadl %pk
	%fillrol =l loadl %pfill
	%morerol =w csltl %krol, %fillrol
	jnz %morerol, @repoverbody, @err1
@repoverbody
	%krob =l loadl %pk
	%brob =w loadub %pb
	%orob =l loadl %po
	%oposrob =l add %orob, %krob
	%poutrob =l add %out, %oposrob
	storeb %brob, %poutrob
	%knextrob =l add %krob, %one
	storel %knextrob, %pk
	jmp @repoverloop
@err1
	ret %mone
@err2
	ret %mtwo
}
