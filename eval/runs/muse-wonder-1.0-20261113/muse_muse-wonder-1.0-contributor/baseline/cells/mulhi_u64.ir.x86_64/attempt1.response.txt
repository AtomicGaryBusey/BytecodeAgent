export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, 4294967295
	%p1hi =l shr %p1, 32
	%p2lo =l and %p2, 4294967295
	%p2hi =l shr %p2, 32
	%p3lo =l and %p3, 4294967295
	%p3hi =l shr %p3, 32
	%mid1 =l add %p0hi, %p1lo
	%mid2 =l add %mid1, %p2lo
	%carry1 =l shr %mid2, 32
	%sum1 =l add %p1hi, %p2hi
	%sum2 =l add %sum1, %p3lo
	%sum3 =l add %sum2, %carry1
	%r2 =l and %sum3, 4294967295
	%carry2 =l shr %sum3, 32
	%r3 =l add %p3hi, %carry2
	%r3shl =l shl %r3, 32
	%hi =l add %r3shl, %r2
	ret %hi
}
