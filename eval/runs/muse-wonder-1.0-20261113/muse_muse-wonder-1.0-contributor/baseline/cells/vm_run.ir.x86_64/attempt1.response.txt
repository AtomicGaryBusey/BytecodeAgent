export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pca =l alloc8 8
	%spa =l alloc8 8
	%idxa =l alloc8 8
	storel 0, %pca
	storel 0, %spa
	storel 0, %idxa
	jmp @loop
@loop
	%pc0 =l loadl %pca
	%sp0 =l loadl %spa
	%is_end =w csgel %pc0, %len
	jnz %is_end, @finish, @fetch
@fetch
	%op_off =l mul %pc0, 8
	%op_ptr =l add %code, %op_off
	%op =l loadl %op_ptr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @finish, @d_push
@d_push
	%c_push =w ceql %op, 1
	jnz %c_push, @op_push, @d_add
@d_add
	%c_add =w ceql %op, 2
	jnz %c_add, @op_add, @d_sub
@d_sub
	%c_sub =w ceql %op, 3
	jnz %c_sub, @op_sub, @d_mul
@d_mul
	%c_mul =w ceql %op, 4
	jnz %c_mul, @op_mul, @d_dup
@d_dup
	%c_dup =w ceql %op, 5
	jnz %c_dup, @op_dup, @d_swap
@d_swap
	%c_swap =w ceql %op, 6
	jnz %c_swap, @op_swap, @d_drop
@d_drop
	%c_drop =w ceql %op, 7
	jnz %c_drop, @op_drop, @d_over
@d_over
	%c_over =w ceql %op, 8
	jnz %c_over, @op_over, @d_lt
@d_lt
	%c_lt =w ceql %op, 9
	jnz %c_lt, @op_lt, @d_jz
@d_jz
	%c_jz =w ceql %op, 10
	jnz %c_jz, @op_jz, @d_jmp
@d_jmp
	%c_jmp =w ceql %op, 11
	jnz %c_jmp, @op_jmp, @op_default
@op_push
	%p_argidx =l add %pc0, 1
	%p_argoff =l mul %p_argidx, 8
	%p_argptr =l add %code, %p_argoff
	%p_val =l loadl %p_argptr
	%p_dstoff =l mul %sp0, 8
	%p_dstptr =l add %stack, %p_dstoff
	storel %p_val, %p_dstptr
	%p_sp =l add %sp0, 1
	storel %p_sp, %spa
	%p_pc =l add %pc0, 2
	storel %p_pc, %pca
	jmp @loop
@op_add
	%a_bidx =l sub %sp0, 1
	%a_boff =l mul %a_bidx, 8
	%a_bptr =l add %stack, %a_boff
	%a_bval =l loadl %a_bptr
	%a_aidx =l sub %sp0, 2
	%a_aoff =l mul %a_aidx, 8
	%a_aptr =l add %stack, %a_aoff
	%a_aval =l loadl %a_aptr
	%a_res =l add %a_aval, %a_bval
	storel %a_res, %a_aptr
	%a_sp =l sub %sp0, 1
	storel %a_sp, %spa
	%a_pc =l add %pc0, 1
	storel %a_pc, %pca
	jmp @loop
@op_sub
	%sb_bidx =l sub %sp0, 1
	%sb_boff =l mul %sb_bidx, 8
	%sb_bptr =l add %stack, %sb_boff
	%sb_bval =l loadl %sb_bptr
	%sb_aidx =l sub %sp0, 2
	%sb_aoff =l mul %sb_aidx, 8
	%sb_aptr =l add %stack, %sb_aoff
	%sb_aval =l loadl %sb_aptr
	%sb_res =l sub %sb_aval, %sb_bval
	storel %sb_res, %sb_aptr
	%sb_sp =l sub %sp0, 1
	storel %sb_sp, %spa
	%sb_pc =l add %pc0, 1
	storel %sb_pc, %pca
	jmp @loop
@op_mul
	%mu_bidx =l sub %sp0, 1
	%mu_boff =l mul %mu_bidx, 8
	%mu_bptr =l add %stack, %mu_boff
	%mu_bval =l loadl %mu_bptr
	%mu_aidx =l sub %sp0, 2
	%mu_aoff =l mul %mu_aidx, 8
	%mu_aptr =l add %stack, %mu_aoff
	%mu_aval =l loadl %mu_aptr
	%mu_res =l mul %mu_aval, %mu_bval
	storel %mu_res, %mu_aptr
	%mu_sp =l sub %sp0, 1
	storel %mu_sp, %spa
	%mu_pc =l add %pc0, 1
	storel %mu_pc, %pca
	jmp @loop
@op_dup
	%du_tidx =l sub %sp0, 1
	%du_toff =l mul %du_tidx, 8
	%du_tptr =l add %stack, %du_toff
	%du_val =l loadl %du_tptr
	%du_dstff =l mul %sp0, 8
	%du_dstptr =l add %stack, %du_dstff
	storel %du_val, %du_dstptr
	%du_sp =l add %sp0, 1
	storel %du_sp, %spa
	%du_pc =l add %pc0, 1
	storel %du_pc, %pca
	jmp @loop
@op_swap
	%sw_bidx =l sub %sp0, 1
	%sw_boff =l mul %sw_bidx, 8
	%sw_bptr =l add %stack, %sw_boff
	%sw_bval =l loadl %sw_bptr
	%sw_aidx =l sub %sp0, 2
	%sw_aoff =l mul %sw_aidx, 8
	%sw_aptr =l add %stack, %sw_aoff
	%sw_aval =l loadl %sw_aptr
	storel %sw_bval, %sw_aptr
	storel %sw_aval, %sw_bptr
	%sw_pc =l add %pc0, 1
	storel %sw_pc, %pca
	jmp @loop
@op_drop
	%dr_sp =l sub %sp0, 1
	storel %dr_sp, %spa
	%dr_pc =l add %pc0, 1
	storel %dr_pc, %pca
	jmp @loop
@op_over
	%ov_sidx =l sub %sp0, 2
	%ov_soff =l mul %ov_sidx, 8
	%ov_sptr =l add %stack, %ov_soff
	%ov_val =l loadl %ov_sptr
	%ov_dstff =l mul %sp0, 8
	%ov_dstptr =l add %stack, %ov_dstff
	storel %ov_val, %ov_dstptr
	%ov_sp =l add %sp0, 1
	storel %ov_sp, %spa
	%ov_pc =l add %pc0, 1
	storel %ov_pc, %pca
	jmp @loop
@op_lt
	%lt_bidx =l sub %sp0, 1
	%lt_boff =l mul %lt_bidx, 8
	%lt_bptr =l add %stack, %lt_boff
	%lt_bval =l loadl %lt_bptr
	%lt_aidx =l sub %sp0, 2
	%lt_aoff =l mul %lt_aidx, 8
	%lt_aptr =l add %stack, %lt_aoff
	%lt_aval =l loadl %lt_aptr
	%lt_cmp =w csltl %lt_aval, %lt_bval
	%lt_res =l extsw %lt_cmp
	storel %lt_res, %lt_aptr
	%lt_sp =l sub %sp0, 1
	storel %lt_sp, %spa
	%lt_pc =l add %pc0, 1
	storel %lt_pc, %pca
	jmp @loop
@op_jz
	%jz_tidx =l add %pc0, 1
	%jz_toff =l mul %jz_tidx, 8
	%jz_tptr =l add %code, %jz_toff
	%jz_tval =l loadl %jz_tptr
	%jz_vidx =l sub %sp0, 1
	%jz_voff =l mul %jz_vidx, 8
	%jz_vptr =l add %stack, %jz_voff
	%jz_vval =l loadl %jz_vptr
	%jz_sp =l sub %sp0, 1
	storel %jz_sp, %spa
	%jz_iszero =w ceql %jz_vval, 0
	jnz %jz_iszero, @jz_taken, @jz_not
@jz_taken
	storel %jz_tval, %pca
	jmp @loop
@jz_not
	%jzn_pc =l add %pc0, 2
	storel %jzn_pc, %pca
	jmp @loop
@op_jmp
	%jm_tidx =l add %pc0, 1
	%jm_toff =l mul %jm_tidx, 8
	%jm_tptr =l add %code, %jm_toff
	%jm_tval =l loadl %jm_tptr
	storel %jm_tval, %pca
	jmp @loop
@op_default
	%df_pc =l add %pc0, 1
	storel %df_pc, %pca
	jmp @loop
@finish
	storel %sp0, %idxa
	jmp @zero_loop
@zero_loop
	%z_i =l loadl %idxa
	%z_done =w csgel %z_i, %cap
	jnz %z_done, @ret_dispatch, @zero_body
@zero_body
	%zb_off =l mul %z_i, 8
	%zb_ptr =l add %stack, %zb_off
	storel 0, %zb_ptr
	%zb_i =l add %z_i, 1
	storel %zb_i, %idxa
	jmp @zero_loop
@ret_dispatch
	%r_empty =w ceql %sp0, 0
	jnz %r_empty, @ret_zero, @ret_top
@ret_zero
	ret 0
@ret_top
	%rt_idx =l sub %sp0, 1
	%rt_off =l mul %rt_idx, 8
	%rt_ptr =l add %stack, %rt_off
	%rt_val =l loadl %rt_ptr
	ret %rt_val
}
