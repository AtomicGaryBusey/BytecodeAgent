export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @update %lo_next
	%hi =l phi @start 4294967296, @update %hi_next
	%diff =l sub %hi, %lo
	%cond =w cugtl %diff, 1
	jnz %cond, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%q =l udiv %n, %mid
	%le =w culel %mid, %q
	jnz %le, @setlo, @sethi
@setlo
	jmp @update
@sethi
	jmp @update
@update
	%lo_next =l phi @setlo %mid, @sethi %lo
	%hi_next =l phi @setlo %hi, @sethi %mid
	jmp @loop
@done
	ret %lo
}
