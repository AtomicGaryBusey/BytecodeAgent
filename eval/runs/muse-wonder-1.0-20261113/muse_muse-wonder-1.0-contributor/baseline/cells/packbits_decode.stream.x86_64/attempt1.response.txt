packbits_decode:
xor r8d, r8d
xor r9d, r9d
.Lloop:
cmp r8, rsi
jae .Ldone
movsx rax, byte ptr [rdi + r8]
inc r8
test rax, rax
jns .Lliteral
cmp rax, -128
je .Lloop
mov r10, 1
sub r10, rax
cmp r8, rsi
jae .Lerr_src
movzx r11d, byte ptr [rdi + r8]
inc r8
lea rax, [r9 + r10]
cmp rax, rcx
jg .Lrep_overflow
.Lrep_copy:
test r10, r10
je .Lloop
mov byte ptr [rdx + r9], r11b
inc r9
dec r10
jmp .Lrep_copy
.Lrep_overflow:
mov r10, rcx
sub r10, r9
.Lrep_over_copy:
test r10, r10
je .Lerr_cap
mov byte ptr [rdx + r9], r11b
inc r9
dec r10
jmp .Lrep_over_copy
.Lliteral:
lea r10, [rax + 1]
lea rax, [r8 + r10]
cmp rax, rsi
jg .Lerr_src
lea rax, [r9 + r10]
cmp rax, rcx
jg .Llit_overflow
.Llit_copy:
test r10, r10
je .Lloop
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r10
jmp .Llit_copy
.Llit_overflow:
mov r10, rcx
sub r10, r9
.Llit_over_copy:
test r10, r10
je .Lerr_cap
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r10
jmp .Llit_over_copy
.Ldone:
mov rax, r9
ret
.Lerr_src:
mov rax, -2
ret
.Lerr_cap:
mov rax, -1
ret
