export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%min_init =l copy 9223372036854775807
	%tmax =l copy -9223372036854775807
	%max_init =l sub %tmax, 1
	jmp @loop
@loop
	%i =l phi @start 0, @next %i_next
	%min =l phi @start %min_init, @next %min_next
	%max =l phi @start %max_init, @next %max_next
	%sum =l phi @start 0, @next %sum_next
	%cnt =l phi @start 0, @next %cnt_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%ltmin =w csltl %y, %min
	jnz %ltmin, @min_y, @min_old
@min_y
	jmp @min_done
@min_old
	jmp @min_done
@min_done
	%min_next =l phi @min_y %y, @min_old %min
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @max_y, @max_old
@max_y
	jmp @max_done
@max_old
	jmp @max_done
@max_done
	%max_next =l phi @max_y %y, @max_old %max
	jmp @next
@next
	%sum_next =l add %sum, %y
	%bit =l and %y, 1
	%isodd =w cnel %bit, 0
	%isodd_l =l extsw %isodd
	%cnt_next =l add %cnt, %isodd_l
	%i_next =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
