export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%fp =l alloc8 64
	%spi =l add %fp, 0
	%spj =l add %fp, 8
	%spcarry =l add %fp, 16
	%spk =l add %fp, 24
	%zero =l copy 0
	storel %zero, %spi
	jmp @zcond
@zcond
	%zidx =l loadl %spi
	%zc =w csltl %zidx, %total
	jnz %zc, @zbody, @zdone
@zbody
	%zoff =l mul %zidx, 8
	%zptr =l add %r, %zoff
	storel %zero, %zptr
	%znext =l add %zidx, 1
	storel %znext, %spi
	jmp @zcond
@zdone
	storel %zero, %spi
	jmp @ocond
@ocond
	%oi =l loadl %spi
	%oc =w csltl %oi, %na
	jnz %oc, @oenter, @odone
@oenter
	storel %zero, %spj
	storel %zero, %spcarry
	jmp @icond
@icond
	%ij =l loadl %spj
	%ic =w csltl %ij, %nb
	jnz %ic, @ibody, @idone
@ibody
	%ibi =l loadl %spi
	%ibj =l loadl %spj
	%ibcarry =l loadl %spcarry
	%aoff =l mul %ibi, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	%boff =l mul %ibj, 8
	%bptr =l add %b, %boff
	%bv =l loadl %bptr
	%axlo =l and %av, 4294967295
	%axhi =l shr %av, 32
	%bylo =l and %bv, 4294967295
	%byhi =l shr %bv, 32
	%p0 =l mul %axlo, %bylo
	%p1 =l mul %axlo, %byhi
	%p2 =l mul %axhi, %bylo
	%p3 =l mul %axhi, %byhi
	%p1s =l shl %p1, 32
	%p2s =l shl %p2, 32
	%sa =l add %p0, %p1s
	%c1w =w cultl %sa, %p0
	%c1 =l extuw %c1w
	%lo =l add %sa, %p2s
	%c2w =w cultl %lo, %sa
	%c2 =l extuw %c2w
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%ha =l add %p3, %p1h
	%hi =l add %ha, %p2h
	%hi2 =l add %hi, %c1
	%hif =l add %hi2, %c2
	%k =l add %ibi, %ibj
	%koff =l mul %k, 8
	%kptr =l add %r, %koff
	%rval =l loadl %kptr
	%s1 =l add %lo, %rval
	%caw =w cultl %s1, %lo
	%ca =l extuw %caw
	%low =l add %s1, %ibcarry
	%cbw =w cultl %low, %s1
	%cb =l extuw %cbw
	%nc1 =l add %hif, %ca
	%ncarry =l add %nc1, %cb
	storel %low, %kptr
	storel %ncarry, %spcarry
	%jnext =l add %ibj, 1
	storel %jnext, %spj
	jmp @icond
@idone
	%idi =l loadl %spi
	%idcarry =l loadl %spcarry
	%idpos =l add %idi, %nb
	%idposoff =l mul %idpos, 8
	%idposptr =l add %r, %idposoff
	%idold =l loadl %idposptr
	%idnew =l add %idold, %idcarry
	storel %idnew, %idposptr
	%idnext =l add %idi, 1
	storel %idnext, %spi
	jmp @ocond
@odone
	storel %total, %spk
	jmp @scond
@scond
	%sk1 =l loadl %spk
	%siszero =w ceql %sk1, %zero
	jnz %siszero, @sdone, @scheck
@scheck
	%sks =l loadl %spk
	%skm1 =l sub %sks, 1
	%skoff =l mul %skm1, 8
	%skptr =l add %r, %skoff
	%srv =l loadl %skptr
	%sis0 =w ceql %srv, %zero
	jnz %sis0, @sdec, @sdone
@sdec
	%skd =l loadl %spk
	%sknew =l sub %skd, 1
	storel %sknew, %spk
	jmp @scond
@sdone
	%retk =l loadl %spk
	ret %retk
}
