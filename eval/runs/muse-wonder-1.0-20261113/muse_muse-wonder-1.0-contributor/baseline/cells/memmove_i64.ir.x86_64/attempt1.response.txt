export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%back =w cultl %src, %dst
	%empty =w cslel %n, 0
	%last =l sub %n, 1
	jnz %empty, @end, @dispatch
@dispatch
	jnz %back, @bloop, @floop
@floop
	%fi =l phi @dispatch 0, @fbody %fi1
	%fcond =w csltl %fi, %n
	jnz %fcond, @fbody, @end
@fbody
	%foff =l mul %fi, 8
	%fs =l add %src, %foff
	%fd =l add %dst, %foff
	%fv =l loadl %fs
	storel %fv, %fd
	%fi1 =l add %fi, 1
	jmp @floop
@bloop
	%bi =l phi @dispatch %last, @bbody %bi1
	%bcond =w csgel %bi, 0
	jnz %bcond, @bbody, @end
@bbody
	%boff =l mul %bi, 8
	%bs =l add %src, %boff
	%bd =l add %dst, %boff
	%bv =l loadl %bs
	storel %bv, %bd
	%bi1 =l sub %bi, 1
	jmp @bloop
@end
	ret 0
}
