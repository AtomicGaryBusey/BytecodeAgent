bubble_sort:
    cmp rsi, 1
    jle .Ldone
    xor edx, edx
.Louter:
    mov rax, rsi
    dec rax
    cmp rdx, rax
    jge .Ldone
    xor ecx, ecx
.Linner:
    mov rax, rsi
    dec rax
    sub rax, rdx
    cmp rcx, rax
    jge .Lnext
    mov rax, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rcx*8 + 8]
    cmp rax, r8
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rcx*8 + 8], rax
.Lnoswap:
    inc rcx
    jmp .Linner
.Lnext:
    inc rdx
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
