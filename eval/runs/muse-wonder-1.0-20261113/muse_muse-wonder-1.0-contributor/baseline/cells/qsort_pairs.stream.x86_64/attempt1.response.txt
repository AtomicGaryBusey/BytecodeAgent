qsort_pairs:
push rbx
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
cmp r12, 2
jl .Ldone
mov rcx, r12
shr rcx, 1
mov rax, rcx
shl rax, 4
add rax, rbx
mov rdx, r12
dec rdx
shl rdx, 4
add rdx, rbx
mov rcx, qword ptr [rax]
mov r8, qword ptr [rdx]
mov qword ptr [rax], r8
mov qword ptr [rdx], rcx
mov rcx, qword ptr [rax + 8]
mov r8, qword ptr [rdx + 8]
mov qword ptr [rax + 8], r8
mov qword ptr [rdx + 8], rcx
mov r13, qword ptr [rdx]
mov r14, qword ptr [rdx + 8]
mov r15, 0
mov rcx, 0
.Lpart_loop:
mov rax, r12
dec rax
cmp rcx, rax
jge .Lpart_done
mov rax, rcx
shl rax, 4
add rax, rbx
cmp qword ptr [rax], r13
jl .Lless
jg .Lnotless
cmp qword ptr [rax + 8], r14
jl .Lless
jmp .Lnotless
.Lless:
mov rdx, r15
shl rdx, 4
add rdx, rbx
mov r8, qword ptr [rdx]
mov r9, qword ptr [rax]
mov qword ptr [rdx], r9
mov qword ptr [rax], r8
mov r8, qword ptr [rdx + 8]
mov r9, qword ptr [rax + 8]
mov qword ptr [rdx + 8], r9
mov qword ptr [rax + 8], r8
inc r15
.Lnotless:
inc rcx
jmp .Lpart_loop
.Lpart_done:
mov rax, r12
dec rax
shl rax, 4
add rax, rbx
mov rdx, r15
shl rdx, 4
add rdx, rbx
mov r8, qword ptr [rdx]
mov r9, qword ptr [rax]
mov qword ptr [rdx], r9
mov qword ptr [rax], r8
mov r8, qword ptr [rdx + 8]
mov r9, qword ptr [rax + 8]
mov qword ptr [rdx + 8], r9
mov qword ptr [rax + 8], r8
mov r13, r15
mov r14, rdx
add r14, 16
mov r15, r12
sub r15, r13
dec r15
mov rdi, rbx
mov rsi, r13
call qsort_pairs
mov rdi, r14
mov rsi, r15
call qsort_pairs
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
