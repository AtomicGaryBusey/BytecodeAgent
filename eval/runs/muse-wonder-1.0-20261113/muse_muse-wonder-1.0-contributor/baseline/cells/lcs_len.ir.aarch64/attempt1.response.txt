export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%np1 =l add %n, 1
	jmp @init_cond
@init_cond
	%ji =l phi @start 0, @init_body %ji_next
	%init_c =w csltl %ji, %np1
	jnz %init_c, @init_body, @init_done
@init_body
	%off_i =l mul %ji, 8
	%p_i =l add %row, %off_i
	storel 0, %p_i
	%ji_next =l add %ji, 1
	jmp @init_cond
@init_done
	jmp @outer_cond
@outer_cond
	%i =l phi @init_done 0, @outer_next %i_next
	%outer_c =w csltl %i, %m
	jnz %outer_c, @inner_setup, @done
@inner_setup
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner_cond
@inner_cond
	%j =l phi @inner_setup 1, @store %j_next
	%diag =l phi @inner_setup 0, @store %up
	%inner_c =w csltl %j, %np1
	jnz %inner_c, @inner_body, @outer_next
@inner_body
	%offj =l mul %j, 8
	%pj =l add %row, %offj
	%up =l loadl %pj
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	%offjm1 =l mul %jm1, 8
	%pjm1 =l add %row, %offjm1
	%left =l loadl %pjm1
	%j_next =l add %j, 1
	jnz %eq, @take_diag, @take_max
@take_diag
	%val_d =l add %diag, 1
	jmp @store
@take_max
	%max_cmp =w csltl %up, %left
	jnz %max_cmp, @max_left, @max_up
@max_up
	jmp @store
@max_left
	jmp @store
@store
	%val =l phi @take_diag %val_d, @max_up %up, @max_left %left
	storel %val, %pj
	jmp @inner_cond
@outer_next
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	%offn =l mul %n, 8
	%pn =l add %row, %offn
	%res =l loadl %pn
	ret %res
}
