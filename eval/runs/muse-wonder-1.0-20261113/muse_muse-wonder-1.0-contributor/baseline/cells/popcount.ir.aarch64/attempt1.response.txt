export function l $popcount(l %x) {
@start
	jmp @loop
@loop
	%cur =l phi @start %x, @body %cur_next
	%count =l phi @start 0, @body %count_next
	%i =l phi @start 0, @body %i_next
	%cond =w csltl %i, 64
	jnz %cond, @body, @end
@body
	%bit =l and %cur, 1
	%count_next =l add %count, %bit
	%cur_next =l shr %cur, 1
	%i_next =l add %i, 1
	jmp @loop
@end
	ret %count
}
