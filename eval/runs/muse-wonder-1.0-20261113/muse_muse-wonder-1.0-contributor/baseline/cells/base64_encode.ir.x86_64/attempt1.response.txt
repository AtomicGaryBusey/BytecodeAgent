export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @full %finext, @tail1 %n, @tail2 %n
	%o =l phi @start 0, @full %fonext, @tail1 %t1onext, @tail2 %t2onext
	%has =w csltl %i, %n
	jnz %has, @body, @end
@body
	%rem =l sub %n, %i
	%isfull =w csgel %rem, 3
	jnz %isfull, @full, @tailcheck
@end
	ret %o
@full
	%fsrc0 =l add %src, %i
	%fb0l =l loadub %fsrc0
	%fsrc1 =l add %fsrc0, 1
	%fb1l =l loadub %fsrc1
	%fsrc2 =l add %fsrc0, 2
	%fb2l =l loadub %fsrc2
	%fb0 =w copy %fb0l
	%fb1 =w copy %fb1l
	%fb2 =w copy %fb2l
	%fout0 =l add %out, %o
	%fout1 =l add %fout0, 1
	%fout2 =l add %fout0, 2
	%fout3 =l add %fout0, 3
	%fv0 =w shr %fb0, 2
	%mf0ge26 =w csgew %fv0, 26
	%mf0ge52 =w csgew %fv0, 52
	%mf0ge62 =w csgew %fv0, 62
	%mf0base =w add %fv0, 65
	%mf0a1 =w mul %mf0ge26, 6
	%mf0c1 =w add %mf0base, %mf0a1
	%mf0a2 =w mul %mf0ge52, 75
	%mf0c2 =w sub %mf0c1, %mf0a2
	%mf0vm =w sub %fv0, 62
	%mf0sd =w mul %mf0vm, 4
	%mf0sp =w add %mf0sd, 43
	%mf0not =w xor %mf0ge62, 1
	%mf0lc =w mul %mf0c2, %mf0not
	%mf0sc =w mul %mf0sp, %mf0ge62
	%mf0ch =w add %mf0lc, %mf0sc
	storeb %mf0ch, %fout0
	%fv1a =w and %fb0, 3
	%fv1b =w shl %fv1a, 4
	%fv1c =w shr %fb1, 4
	%fv1 =w or %fv1b, %fv1c
	%mf1ge26 =w csgew %fv1, 26
	%mf1ge52 =w csgew %fv1, 52
	%mf1ge62 =w csgew %fv1, 62
	%mf1base =w add %fv1, 65
	%mf1a1 =w mul %mf1ge26, 6
	%mf1c1 =w add %mf1base, %mf1a1
	%mf1a2 =w mul %mf1ge52, 75
	%mf1c2 =w sub %mf1c1, %mf1a2
	%mf1vm =w sub %fv1, 62
	%mf1sd =w mul %mf1vm, 4
	%mf1sp =w add %mf1sd, 43
	%mf1not =w xor %mf1ge62, 1
	%mf1lc =w mul %mf1c2, %mf1not
	%mf1sc =w mul %mf1sp, %mf1ge62
	%mf1ch =w add %mf1lc, %mf1sc
	storeb %mf1ch, %fout1
	%fv2a =w and %fb1, 15
	%fv2b =w shl %fv2a, 2
	%fv2c =w shr %fb2, 6
	%fv2 =w or %fv2b, %fv2c
	%mf2ge26 =w csgew %fv2, 26
	%mf2ge52 =w csgew %fv2, 52
	%mf2ge62 =w csgew %fv2, 62
	%mf2base =w add %fv2, 65
	%mf2a1 =w mul %mf2ge26, 6
	%mf2c1 =w add %mf2base, %mf2a1
	%mf2a2 =w mul %mf2ge52, 75
	%mf2c2 =w sub %mf2c1, %mf2a2
	%mf2vm =w sub %fv2, 62
	%mf2sd =w mul %mf2vm, 4
	%mf2sp =w add %mf2sd, 43
	%mf2not =w xor %mf2ge62, 1
	%mf2lc =w mul %mf2c2, %mf2not
	%mf2sc =w mul %mf2sp, %mf2ge62
	%mf2ch =w add %mf2lc, %mf2sc
	storeb %mf2ch, %fout2
	%fv3 =w and %fb2, 63
	%mf3ge26 =w csgew %fv3, 26
	%mf3ge52 =w csgew %fv3, 52
	%mf3ge62 =w csgew %fv3, 62
	%mf3base =w add %fv3, 65
	%mf3a1 =w mul %mf3ge26, 6
	%mf3c1 =w add %mf3base, %mf3a1
	%mf3a2 =w mul %mf3ge52, 75
	%mf3c2 =w sub %mf3c1, %mf3a2
	%mf3vm =w sub %fv3, 62
	%mf3sd =w mul %mf3vm, 4
	%mf3sp =w add %mf3sd, 43
	%mf3not =w xor %mf3ge62, 1
	%mf3lc =w mul %mf3c2, %mf3not
	%mf3sc =w mul %mf3sp, %mf3ge62
	%mf3ch =w add %mf3lc, %mf3sc
	storeb %mf3ch, %fout3
	%finext =l add %i, 3
	%fonext =l add %o, 4
	jmp @loop
@tailcheck
	%isone =w ceql %rem, 1
	jnz %isone, @tail1, @tail2
@tail1
	%t1src0 =l add %src, %i
	%t1b0l =l loadub %t1src0
	%t1b0 =w copy %t1b0l
	%t1out0 =l add %out, %o
	%t1out1 =l add %t1out0, 1
	%t1out2 =l add %t1out0, 2
	%t1out3 =l add %t1out0, 3
	%t1v0 =w shr %t1b0, 2
	%m10ge26 =w csgew %t1v0, 26
	%m10ge52 =w csgew %t1v0, 52
	%m10ge62 =w csgew %t1v0, 62
	%m10base =w add %t1v0, 65
	%m10a1 =w mul %m10ge26, 6
	%m10c1 =w add %m10base, %m10a1
	%m10a2 =w mul %m10ge52, 75
	%m10c2 =w sub %m10c1, %m10a2
	%m10vm =w sub %t1v0, 62
	%m10sd =w mul %m10vm, 4
	%m10sp =w add %m10sd, 43
	%m10not =w xor %m10ge62, 1
	%m10lc =w mul %m10c2, %m10not
	%m10sc =w mul %m10sp, %m10ge62
	%m10ch =w add %m10lc, %m10sc
	storeb %m10ch, %t1out0
	%t1v1a =w and %t1b0, 3
	%t1v1b =w shl %t1v1a, 4
	%m11ge26 =w csgew %t1v1b, 26
	%m11ge52 =w csgew %t1v1b, 52
	%m11ge62 =w csgew %t1v1b, 62
	%m11base =w add %t1v1b, 65
	%m11a1 =w mul %m11ge26, 6
	%m11c1 =w add %m11base, %m11a1
	%m11a2 =w mul %m11ge52, 75
	%m11c2 =w sub %m11c1, %m11a2
	%m11vm =w sub %t1v1b, 62
	%m11sd =w mul %m11vm, 4
	%m11sp =w add %m11sd, 43
	%m11not =w xor %m11ge62, 1
	%m11lc =w mul %m11c2, %m11not
	%m11sc =w mul %m11sp, %m11ge62
	%m11ch =w add %m11lc, %m11sc
	storeb %m11ch, %t1out1
	%t1eq =w copy 61
	storeb %t1eq, %t1out2
	storeb %t1eq, %t1out3
	%t1onext =l add %o, 4
	jmp @loop
@tail2
	%t2src0 =l add %src, %i
	%t2b0l =l loadub %t2src0
	%t2src1 =l add %t2src0, 1
	%t2b1l =l loadub %t2src1
	%t2b0 =w copy %t2b0l
	%t2b1 =w copy %t2b1l
	%t2out0 =l add %out, %o
	%t2out1 =l add %t2out0, 1
	%t2out2 =l add %t2out0, 2
	%t2out3 =l add %t2out0, 3
	%t2v0 =w shr %t2b0, 2
	%m20ge26 =w csgew %t2v0, 26
	%m20ge52 =w csgew %t2v0, 52
	%m20ge62 =w csgew %t2v0, 62
	%m20base =w add %t2v0, 65
	%m20a1 =w mul %m20ge26, 6
	%m20c1 =w add %m20base, %m20a1
	%m20a2 =w mul %m20ge52, 75
	%m20c2 =w sub %m20c1, %m20a2
	%m20vm =w sub %t2v0, 62
	%m20sd =w mul %m20vm, 4
	%m20sp =w add %m20sd, 43
	%m20not =w xor %m20ge62, 1
	%m20lc =w mul %m20c2, %m20not
	%m20sc =w mul %m20sp, %m20ge62
	%m20ch =w add %m20lc, %m20sc
	storeb %m20ch, %t2out0
	%t2v1a =w and %t2b0, 3
	%t2v1b =w shl %t2v1a, 4
	%t2v1c =w shr %t2b1, 4
	%t2v1 =w or %t2v1b, %t2v1c
	%m21ge26 =w csgew %t2v1, 26
	%m21ge52 =w csgew %t2v1, 52
	%m21ge62 =w csgew %t2v1, 62
	%m21base =w add %t2v1, 65
	%m21a1 =w mul %m21ge26, 6
	%m21c1 =w add %m21base, %m21a1
	%m21a2 =w mul %m21ge52, 75
	%m21c2 =w sub %m21c1, %m21a2
	%m21vm =w sub %t2v1, 62
	%m21sd =w mul %m21vm, 4
	%m21sp =w add %m21sd, 43
	%m21not =w xor %m21ge62, 1
	%m21lc =w mul %m21c2, %m21not
	%m21sc =w mul %m21sp, %m21ge62
	%m21ch =w add %m21lc, %m21sc
	storeb %m21ch, %t2out1
	%t2v2a =w and %t2b1, 15
	%t2v2b =w shl %t2v2a, 2
	%m22ge26 =w csgew %t2v2b, 26
	%m22ge52 =w csgew %t2v2b, 52
	%m22ge62 =w csgew %t2v2b, 62
	%m22base =w add %t2v2b, 65
	%m22a1 =w mul %m22ge26, 6
	%m22c1 =w add %m22base, %m22a1
	%m22a2 =w mul %m22ge52, 75
	%m22c2 =w sub %m22c1, %m22a2
	%m22vm =w sub %t2v2b, 62
	%m22sd =w mul %m22vm, 4
	%m22sp =w add %m22sd, 43
	%m22not =w xor %m22ge62, 1
	%m22lc =w mul %m22c2, %m22not
	%m22sc =w mul %m22sp, %m22ge62
	%m22ch =w add %m22lc, %m22sc
	storeb %m22ch, %t2out2
	%t2eq =w copy 61
	storeb %t2eq, %t2out3
	%t2onext =l add %o, 4
	jmp @loop
}
