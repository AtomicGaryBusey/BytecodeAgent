export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @low %lo_next, @high %lo
	%hi =l phi @start %n, @low %hi, @high %hi_next
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %val, %key
	jnz %lt, @low, @high
@low
	%lo_next =l add %mid, 1
	jmp @loop
@high
	%hi_next =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
