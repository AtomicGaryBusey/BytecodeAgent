export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @outer_inc %i_next
	%outer_lim =l sub %n, 1
	%oc =w csltl %i, %outer_lim
	jnz %oc, @inner, @done
@inner
	%j =l phi @outer 0, @inner_inc %j_next
	%inner_lim =l sub %outer_lim, %i
	%ic =w csltl %j, %inner_lim
	jnz %ic, @body, @outer_inc
@body
	%off =l mul %j, 8
	%ptr =l add %a, %off
	%ptr1 =l add %ptr, 8
	%v =l loadl %ptr
	%v1 =l loadl %ptr1
	%gt =w csltl %v1, %v
	jnz %gt, @swap, @inner_inc
@swap
	storel %v1, %ptr
	storel %v, %ptr1
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner
@outer_inc
	%i_next =l add %i, 1
	jmp @outer
@done
	ret 0
}
