export function l $count_desc(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cntnext
	%limit =l sub %n, 1
	%cond =w csltl %i, %limit
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%ptr2 =l add %ptr, 8
	%y =l loadl %ptr2
	%gt =w csltl %y, %x
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	%inext =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
