export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%cur =l phi @start %n, @next %nnext
	%res =l phi @start 0, @next %resnext
	%bit =l phi @start 4611686018427387904, @next %bitnext
	%done =w ceql %bit, 0
	jnz %done, @end, @body
@body
	%sum =l add %res, %bit
	%ge =w cugel %cur, %sum
	jnz %ge, @taken, @else
@taken
	%nt =l sub %cur, %sum
	%rht =l shr %res, 1
	%rt =l add %rht, %bit
	jmp @next
@else
	%rhe =l shr %res, 1
	jmp @next
@next
	%nnext =l phi @taken %nt, @else %cur
	%resnext =l phi @taken %rt, @else %rhe
	%bitnext =l shr %bit, 2
	jmp @loop
@end
	ret %res
}
