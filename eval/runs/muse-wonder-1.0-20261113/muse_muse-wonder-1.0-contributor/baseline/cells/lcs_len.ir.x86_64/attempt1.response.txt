export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_head
@init_head
	%j =l phi @start 0, @init_body %jnext0
	%icond =w cslel %j, %n
	jnz %icond, @init_body, @outer_head
@init_body
	%ioff =l mul %j, 8
	%ip =l add %row, %ioff
	storel 0, %ip
	%jnext0 =l add %j, 1
	jmp @init_head
@outer_head
	%i =l phi @init_head 0, @outer_next %inext
	%ocond =w csltl %i, %m
	jnz %ocond, @inner_head, @done
@inner_head
	%jj =l phi @outer_head 1, @inner_tail %jnext
	%diag =l phi @outer_head 0, @inner_tail %diagnext
	%jcond =w cslel %jj, %n
	jnz %jcond, @inner_body, @outer_next
@inner_body
	%joff =l mul %jj, 8
	%pj =l add %row, %joff
	%up =l loadl %pj
	%jm1 =l sub %jj, 1
	%jmoff =l mul %jm1, 8
	%pjm =l add %row, %jmoff
	%left =l loadl %pjm
	%px =l add %x, %i
	%xb =w loadub %px
	%py =l add %y, %jm1
	%yb =w loadub %py
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%mval =l add %diag, 1
	storel %mval, %pj
	jmp @inner_tail
@nomatch
	%lt =w csltl %up, %left
	jnz %lt, @maxleft, @maxup
@maxleft
	jmp @havemax
@maxup
	jmp @havemax
@havemax
	%maxv =l phi @maxleft %left, @maxup %up
	storel %maxv, %pj
	jmp @inner_tail
@inner_tail
	%diagnext =l copy %up
	%jnext =l add %jj, 1
	jmp @inner_head
@outer_next
	%inext =l add %i, 1
	jmp @outer_head
@done
	%noff =l mul %n, 8
	%pn =l add %row, %noff
	%res =l loadl %pn
	ret %res
}
