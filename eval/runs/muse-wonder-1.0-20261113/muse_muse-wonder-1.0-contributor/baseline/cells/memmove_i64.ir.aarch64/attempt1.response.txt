export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nonpos =w csltl %n, 1
	jnz %nonpos, @done, @noteq
@noteq
	%eq =w ceql %dst, %src
	jnz %eq, @done, @dir
@dir
	%forward =w cultl %dst, %src
	jnz %forward, @fwd_loop, @bwd_loop
@fwd_loop
	%fi =l phi @dir 0, @fwd_body %fnext
	%fcond =w csltl %fi, %n
	jnz %fcond, @fwd_body, @done
@fwd_body
	%foff =l mul %fi, 8
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fv =l loadl %fsp
	storel %fv, %fdp
	%fnext =l add %fi, 1
	jmp @fwd_loop
@bwd_loop
	%bi =l phi @dir %n, @bwd_body %bprev
	%bzero =w ceql %bi, 0
	jnz %bzero, @done, @bwd_body
@bwd_body
	%bprev =l sub %bi, 1
	%boff =l mul %bprev, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	jmp @bwd_loop
@done
	ret 0
}
