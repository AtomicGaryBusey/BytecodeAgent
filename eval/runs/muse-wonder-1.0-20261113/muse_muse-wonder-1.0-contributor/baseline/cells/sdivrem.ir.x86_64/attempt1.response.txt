export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%is_zero =w ceql %b, 0
	jnz %is_zero, @zero, @nonzero
@zero
	storel %a, %rem
	ret 0
@nonzero
	%is_neg_one =w ceql %b, -1
	jnz %is_neg_one, @neg_one, @normal
@neg_one
	%q_neg =l sub 0, %a
	%r_zero =l copy 0
	storel %r_zero, %rem
	ret %q_neg
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
