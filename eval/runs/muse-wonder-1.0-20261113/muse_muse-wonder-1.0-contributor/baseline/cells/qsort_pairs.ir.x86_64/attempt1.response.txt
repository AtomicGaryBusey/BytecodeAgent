export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @init
@done
	ret 0
@init
	%nm1 =l sub %n, 1
	%lastoff =l mul %nm1, 16
	%last =l add %a, %lastoff
	%pkey =l loadl %last
	%last1 =l add %last, 8
	%pval =l loadl %last1
	%izero =l copy 0
	%jzero =l copy 0
	jmp @scan
@scan
	%i =l phi @init %izero, @merge %icur
	%j =l phi @init %jzero, @merge %jnext
	%jlt =w csltl %j, %nm1
	jnz %jlt, @body, @finish
@body
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%jkey =l loadl %jptr
	%jptr1 =l add %jptr, 8
	%jval =l loadl %jptr1
	%klt =w csltl %jkey, %pkey
	%keq =w ceql %jkey, %pkey
	%vlt =w csltl %jval, %pval
	%andv =w and %keq, %vlt
	%less =w or %klt, %andv
	jnz %less, @swap, @noswap
@swap
	%ioff =l mul %i, 16
	%iptr =l add %a, %ioff
	%ikey =l loadl %iptr
	%iptr1 =l add %iptr, 8
	%ival =l loadl %iptr1
	storel %jkey, %iptr
	storel %jval, %iptr1
	storel %ikey, %jptr
	storel %ival, %jptr1
	%iinc =l add %i, 1
	jmp @merge
@noswap
	jmp @merge
@merge
	%icur =l phi @swap %iinc, @noswap %i
	%jnext =l add %j, 1
	jmp @scan
@finish
	%foff =l mul %i, 16
	%fptr =l add %a, %foff
	%fkey =l loadl %fptr
	%fptr1 =l add %fptr, 8
	%fval =l loadl %fptr1
	storel %pkey, %fptr
	storel %pval, %fptr1
	storel %fkey, %last
	storel %fval, %last1
	%p =l copy %a
	%m =l copy %i
	%r =l call $qsort_pairs(l %p, l %m)
	%ri =l add %i, 1
	%roff =l mul %ri, 16
	%p2 =l add %a, %roff
	%m2 =l sub %nm1, %i
	%r2 =l call $qsort_pairs(l %p2, l %m2)
	ret 0
}
