export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%pvnn =l alloc8 8
	%pviter =l alloc8 8
	%pvi =l alloc8 8
	%pvj =l alloc8 8
	%pvt =l alloc8 8
	%pvidx =l alloc8 8
	%pvsum =l alloc8 8
	%nn0 =l mul %n, %n
	storel %nn0, %pvnn
	storel 0, %pvidx
	jmp @copycond
@copycond
	%ccidx =l loadl %pvidx
	%ccnn =l loadl %pvnn
	%ccc =w csltl %ccidx, %ccnn
	jnz %ccc, @copybody, @copydone
@copybody
	%cbidx =l loadl %pvidx
	%cboff =l mul %cbidx, 8
	%cbpa =l add %a, %cboff
	%cbv =l loadl %cbpa
	%cbpt =l add %tmp, %cboff
	storel %cbv, %cbpt
	%cbidx2 =l add %cbidx, 1
	storel %cbidx2, %pvidx
	jmp @copycond
@copydone
	storel 0, %pvi
	jmp @idicond
@idicond
	%idici =l loadl %pvi
	%idicc =w csltl %idici, %n
	jnz %idicc, @idibody, @ididone
@idibody
	storel 0, %pvj
	jmp @idjcond
@idjcond
	%idjcj =l loadl %pvj
	%idjcc =w csltl %idjcj, %n
	jnz %idjcc, @idjbody, @idjdone
@idjbody
	%idjbi =l loadl %pvi
	%idjbj =l loadl %pvj
	%idjbbase =l mul %idjbi, %n
	%idjbidx =l add %idjbbase, %idjbj
	%idjboff =l mul %idjbidx, 8
	%idjbp =l add %out, %idjboff
	%idjbeq =w ceql %idjbi, %idjbj
	%idjbv =l extuw %idjbeq
	storel %idjbv, %idjbp
	%idjbj2 =l add %idjbj, 1
	storel %idjbj2, %pvj
	jmp @idjcond
@idjdone
	%idjdi =l loadl %pvi
	%idjdi2 =l add %idjdi, 1
	storel %idjdi2, %pvi
	jmp @idicond
@ididone
	storel 0, %pviter
	jmp @maincond
@maincond
	%mcit =l loadl %pviter
	%mcc =w csltl %mcit, %k
	jnz %mcc, @mainbody, @maindone
@mainbody
	storel 0, %pvi
	jmp @micond
@micond
	%mici =l loadl %pvi
	%micc =w csltl %mici, %n
	jnz %micc, @mibody, @midone
@mibody
	storel 0, %pvj
	jmp @mjcond
@mjcond
	%mjcj =l loadl %pvj
	%mjcc =w csltl %mjcj, %n
	jnz %mjcc, @mjbody, @mjdone
@mjbody
	storel 0, %pvsum
	storel 0, %pvt
	jmp @mtcond
@mtcond
	%mtct =l loadl %pvt
	%mtcc =w csltl %mtct, %n
	jnz %mtcc, @mtbody, @mtdone
@mtbody
	%mtbi =l loadl %pvi
	%mtbj =l loadl %pvj
	%mtbt =l loadl %pvt
	%mtbibase =l mul %mtbi, %n
	%mtbia =l add %mtbibase, %mtbt
	%mtboffa =l mul %mtbia, 8
	%mtbpa =l add %out, %mtboffa
	%mtbva =l loadl %mtbpa
	%mtbtbase =l mul %mtbt, %n
	%mtbib =l add %mtbtbase, %mtbj
	%mtboffb =l mul %mtbib, 8
	%mtbpb =l add %tmp, %mtboffb
	%mtbvb =l loadl %mtbpb
	%mtbprod =l mul %mtbva, %mtbvb
	%mtbsum =l loadl %pvsum
	%mtbsum2 =l add %mtbsum, %mtbprod
	storel %mtbsum2, %pvsum
	%mtbt2 =l add %mtbt, 1
	storel %mtbt2, %pvt
	jmp @mtcond
@mtdone
	%mtdi =l loadl %pvi
	%mtdj =l loadl %pvj
	%mtdbase =l mul %mtdi, %n
	%mtdidx =l add %mtdbase, %mtdj
	%mtdnn =l loadl %pvnn
	%mtddidx =l add %mtdnn, %mtdidx
	%mtdoff =l mul %mtddidx, 8
	%mtdp =l add %tmp, %mtdoff
	%mtdsum =l loadl %pvsum
	storel %mtdsum, %mtdp
	%mtdj2 =l add %mtdj, 1
	storel %mtdj2, %pvj
	jmp @mjcond
@mjdone
	%mjdi =l loadl %pvi
	%mjdi2 =l add %mjdi, 1
	storel %mjdi2, %pvi
	jmp @micond
@midone
	storel 0, %pvidx
	jmp @bkcond
@bkcond
	%bkcidx =l loadl %pvidx
	%bkcnn =l loadl %pvnn
	%bkcc =w csltl %bkcidx, %bkcnn
	jnz %bkcc, @bkbody, @bkdone
@bkbody
	%bkbidx =l loadl %pvidx
	%bkbnn =l loadl %pvnn
	%bkbsidx =l add %bkbnn, %bkbidx
	%bkboffs =l mul %bkbsidx, 8
	%bkbps =l add %tmp, %bkboffs
	%bkbv =l loadl %bkbps
	%bkboffd =l mul %bkbidx, 8
	%bkbpd =l add %out, %bkboffd
	storel %bkbv, %bkbpd
	%bkbidx2 =l add %bkbidx, 1
	storel %bkbidx2, %pvidx
	jmp @bkcond
@bkdone
	%bkdit =l loadl %pviter
	%bkdit2 =l add %bkdit, 1
	storel %bkdit2, %pviter
	jmp @maincond
@maindone
	storel 0, %pvidx
	jmp @zcond
@zcond
	%zcidx =l loadl %pvidx
	%zcnn =l loadl %pvnn
	%zclim =l mul %zcnn, 2
	%zcc =w csltl %zcidx, %zclim
	jnz %zcc, @zbody, @zdone
@zbody
	%zbidx =l loadl %pvidx
	%zboff =l mul %zbidx, 8
	%zbp =l add %tmp, %zboff
	storel 0, %zbp
	%zbidx2 =l add %zbidx, 1
	storel %zbidx2, %pvidx
	jmp @zcond
@zdone
	ret 0
}
