dot_i32_simd:
pxor xmm0, xmm0
xor ecx, ecx
mov r8, rdx
and r8, -4
cmp rcx, r8
jae .Lvec_done
.Lvec_loop:
movdqu xmm1, xmmword ptr [rdi + rcx*4]
movdqu xmm2, xmmword ptr [rsi + rcx*4]
pmovsxdq xmm3, xmm1
pmovsxdq xmm4, xmm2
pmuldq xmm3, xmm4
paddq xmm0, xmm3
psrldq xmm1, 8
psrldq xmm2, 8
pmovsxdq xmm3, xmm1
pmovsxdq xmm4, xmm2
pmuldq xmm3, xmm4
paddq xmm0, xmm3
add rcx, 4
cmp rcx, r8
jb .Lvec_loop
.Lvec_done:
movq rax, xmm0
psrldq xmm0, 8
movq r8, xmm0
add rax, r8
cmp rcx, rdx
jae .Ldone
.Lscalar_loop:
movsxd r8, dword ptr [rdi + rcx*4]
movsxd r9, dword ptr [rsi + rcx*4]
imul r8, r9
add rax, r8
inc rcx
cmp rcx, rdx
jb .Lscalar_loop
.Ldone:
ret
