export function l $collatz_max(l %n) {
@start
	jmp @kloop
@kloop
	%k =l phi @start 1, @knext %k_next
	%max =l phi @start 0, @knext %max_next
	%kcond =w cslel %k, %n
	jnz %kcond, @kbody, @done
@kbody
	jmp @xloop
@xloop
	%x =l phi @kbody %k, @xnext %x_next
	%steps =l phi @kbody 0, @xnext %steps_next
	%xcond =w cnel %x, 1
	jnz %xcond, @xbody, @xdone
@xbody
	%parity =l and %x, 1
	%isodd =w cnel %parity, 0
	jnz %isodd, @odd, @even
@even
	%x_even =l div %x, 2
	jmp @xnext
@odd
	%x_odd1 =l mul %x, 3
	%x_odd =l add %x_odd1, 1
	jmp @xnext
@xnext
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @xloop
@xdone
	%gt =w csgtl %steps, %max
	jnz %gt, @upd, @noupd
@upd
	jmp @knext
@noupd
	jmp @knext
@knext
	%max_next =l phi @upd %steps, @noupd %max
	%k_next =l add %k, 1
	jmp @kloop
@done
	ret %max
}
