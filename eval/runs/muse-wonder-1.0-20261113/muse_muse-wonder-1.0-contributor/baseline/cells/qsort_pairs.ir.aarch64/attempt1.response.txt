export function l $qsort_pairs(l %a0, l %n0) {
@start
	jmp @loop_cond
@loop_cond
	%a =l phi @start %a0, @loop_next %a_next
	%n =l phi @start %n0, @loop_next %n_next
	%is_big =w csgtl %n, 1
	jnz %is_big, @partition_init, @done
@done
	ret 0
@partition_init
	%last_idx =l sub %n, 1
	%last_off =l mul %last_idx, 16
	%last_ptr =l add %a, %last_off
	%pk =l loadl %last_ptr
	%last_ptr1 =l add %last_ptr, 8
	%pv =l loadl %last_ptr1
	jmp @scan
@scan
	%i =l phi @partition_init 0, @scan_next %i_next
	%j =l phi @partition_init 0, @scan_next %j_next
	%scan_cond =w csltl %j, %last_idx
	jnz %scan_cond, @scan_body, @scan_done
@scan_body
	%j_off =l mul %j, 16
	%j_ptr =l add %a, %j_off
	%jk =l loadl %j_ptr
	%j_ptr1 =l add %j_ptr, 8
	%jv =l loadl %j_ptr1
	%klt =w csltl %jk, %pk
	jnz %klt, @less, @key_not_less
@key_not_less
	%keq =w ceql %jk, %pk
	jnz %keq, @check_v, @no_swap
@check_v
	%vlt =w csltl %jv, %pv
	jnz %vlt, @less, @no_swap
@less
	%i_off =l mul %i, 16
	%i_ptr =l add %a, %i_off
	%ik =l loadl %i_ptr
	%i_ptr1 =l add %i_ptr, 8
	%iv =l loadl %i_ptr1
	storel %jk, %i_ptr
	storel %jv, %i_ptr1
	storel %ik, %j_ptr
	storel %iv, %j_ptr1
	%i_inc =l add %i, 1
	jmp @scan_next
@no_swap
	jmp @scan_next
@scan_next
	%i_next =l phi @less %i_inc, @no_swap %i
	%j_next =l add %j, 1
	jmp @scan
@scan_done
	%i_off2 =l mul %i, 16
	%i_ptr2 =l add %a, %i_off2
	%ik2 =l loadl %i_ptr2
	%i_ptr12 =l add %i_ptr2, 8
	%iv2 =l loadl %i_ptr12
	storel %pk, %i_ptr2
	storel %pv, %i_ptr12
	storel %ik2, %last_ptr
	storel %iv2, %last_ptr1
	%rn_tmp =l sub %n, %i
	%right_n =l sub %rn_tmp, 1
	%right_idx =l add %i, 1
	%right_off =l mul %right_idx, 16
	%right_ptr =l add %a, %right_off
	%left_smaller =w csltl %i, %right_n
	jnz %left_smaller, @recurse_left, @recurse_right
@recurse_left
	%r1 =l call $qsort_pairs(l %a, l %i)
	jmp @loop_next
@recurse_right
	%r2 =l call $qsort_pairs(l %right_ptr, l %right_n)
	jmp @loop_next
@loop_next
	%a_next =l phi @recurse_left %right_ptr, @recurse_right %a
	%n_next =l phi @recurse_left %right_n, @recurse_right %i
	jmp @loop_cond
}
