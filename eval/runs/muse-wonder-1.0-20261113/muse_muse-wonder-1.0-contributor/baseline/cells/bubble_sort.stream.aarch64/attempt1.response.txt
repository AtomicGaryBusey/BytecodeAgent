sub x9, x1, #1
mov x10, xzr
.Louter:
cmp x10, x9
b.ge .Lret
sub x11, x9, x10
mov x12, xzr
.Linner:
cmp x12, x11
b.ge .Lnext
ldr x13, [x0, x12, lsl #3]
add x14, x12, #1
ldr x15, [x0, x14, lsl #3]
cmp x13, x15
b.le .Lnoswap
str x15, [x0, x12, lsl #3]
str x13, [x0, x14, lsl #3]
.Lnoswap:
add x12, x12, #1
b .Linner
.Lnext:
add x10, x10, #1
b .Louter
.Lret:
mov x0, #0
ret
