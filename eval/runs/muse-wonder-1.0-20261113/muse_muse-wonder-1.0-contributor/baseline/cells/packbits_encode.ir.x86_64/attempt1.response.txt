export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	%pj =l alloc8 8
	%pr =l alloc8 8
	%pt =l alloc8 8
	%pk =l alloc8 8
	%pl =l alloc8 8
	storel 0, %pi
	storel 0, %po
	jmp @outer_cond
@outer_cond
	%oc_i =l loadl %pi
	%oc_c =w csltl %oc_i, %n
	jnz %oc_c, @run_i_init, @done
@done
	%done_v =l loadl %po
	ret %done_v
@fail
	%fail_v =l copy -1
	ret %fail_v
@run_i_init
	%rii_i =l loadl %pi
	storel 1, %pr
	%rii_t =l add %rii_i, 1
	storel %rii_t, %pt
	jmp @run_i_cond
@run_i_cond
	%ric_t =l loadl %pt
	%ric_c1 =w csltl %ric_t, %n
	jnz %ric_c1, @run_i_c2, @run_i_done
@run_i_c2
	%ri2_r =l loadl %pr
	%ri2_c =w csltl %ri2_r, 128
	jnz %ri2_c, @run_i_c3, @run_i_done
@run_i_c3
	%ri3_i =l loadl %pi
	%ri3_t =l loadl %pt
	%ri3_pi =l add %src, %ri3_i
	%ri3_bi =w loadub %ri3_pi
	%ri3_pt =l add %src, %ri3_t
	%ri3_bt =w loadub %ri3_pt
	%ri3_c =w ceqw %ri3_bi, %ri3_bt
	jnz %ri3_c, @run_i_body, @run_i_done
@run_i_body
	%rib_r =l loadl %pr
	%rib_r2 =l add %rib_r, 1
	storel %rib_r2, %pr
	%rib_t =l loadl %pt
	%rib_t2 =l add %rib_t, 1
	storel %rib_t2, %pt
	jmp @run_i_cond
@run_i_done
	%rid_r =l loadl %pr
	%rid_c =w csgel %rid_r, 3
	jnz %rid_c, @repeat_check, @literal_init
@repeat_check
	%rpc_o =l loadl %po
	%rpc_need =l add %rpc_o, 2
	%rpc_c =w cslel %rpc_need, %cap
	jnz %rpc_c, @repeat_write, @fail
@repeat_write
	%rpw_o =l loadl %po
	%rpw_i =l loadl %pi
	%rpw_r =l loadl %pr
	%rpw_rw =w loadsw %pr
	%rpw_hdr =w sub 257, %rpw_rw
	%rpw_p0 =l add %out, %rpw_o
	storeb %rpw_hdr, %rpw_p0
	%rpw_p1 =l add %rpw_p0, 1
	%rpw_ps =l add %src, %rpw_i
	%rpw_b =w loadub %rpw_ps
	storeb %rpw_b, %rpw_p1
	%rpw_on =l add %rpw_o, 2
	storel %rpw_on, %po
	%rpw_in =l add %rpw_i, %rpw_r
	storel %rpw_in, %pi
	jmp @outer_cond
@literal_init
	%li_i =l loadl %pi
	storel %li_i, %pj
	jmp @lit_cond1
@lit_cond1
	%lc1_j =l loadl %pj
	%lc1_c =w csltl %lc1_j, %n
	jnz %lc1_c, @lit_cond2, @lit_done
@lit_cond2
	%lc2_j =l loadl %pj
	%lc2_i =l loadl %pi
	%lc2_len =l sub %lc2_j, %lc2_i
	%lc2_c =w csltl %lc2_len, 128
	jnz %lc2_c, @run_j_init, @lit_done
@run_j_init
	%rji_j =l loadl %pj
	storel 1, %pr
	%rji_t =l add %rji_j, 1
	storel %rji_t, %pt
	jmp @run_j_cond
@run_j_cond
	%rjc_t =l loadl %pt
	%rjc_c =w csltl %rjc_t, %n
	jnz %rjc_c, @run_j_c2, @run_j_done
@run_j_c2
	%rj2_r =l loadl %pr
	%rj2_c =w csltl %rj2_r, 128
	jnz %rj2_c, @run_j_c3, @run_j_done
@run_j_c3
	%rj3_j =l loadl %pj
	%rj3_t =l loadl %pt
	%rj3_pj =l add %src, %rj3_j
	%rj3_bj =w loadub %rj3_pj
	%rj3_pt =l add %src, %rj3_t
	%rj3_bt =w loadub %rj3_pt
	%rj3_c =w ceqw %rj3_bj, %rj3_bt
	jnz %rj3_c, @run_j_body, @run_j_done
@run_j_body
	%rjb_r =l loadl %pr
	%rjb_r2 =l add %rjb_r, 1
	storel %rjb_r2, %pr
	%rjb_t =l loadl %pt
	%rjb_t2 =l add %rjb_t, 1
	storel %rjb_t2, %pt
	jmp @run_j_cond
@run_j_done
	%rjd_r =l loadl %pr
	%rjd_c =w csltl %rjd_r, 3
	jnz %rjd_c, @lit_advance, @lit_done
@lit_advance
	%la_j =l loadl %pj
	%la_j2 =l add %la_j, 1
	storel %la_j2, %pj
	jmp @lit_cond1
@lit_done
	%ld_j =l loadl %pj
	%ld_i =l loadl %pi
	%ld_L =l sub %ld_j, %ld_i
	storel %ld_L, %pl
	%ld_o =l loadl %po
	%ld_n1 =l add %ld_o, 1
	%ld_need =l add %ld_n1, %ld_L
	%ld_c =w cslel %ld_need, %cap
	jnz %ld_c, @lit_write, @fail
@lit_write
	%lw_o =l loadl %po
	%lw_p0 =l add %out, %lw_o
	%lw_Lw =w loadsw %pl
	%lw_hdr =w sub %lw_Lw, 1
	storeb %lw_hdr, %lw_p0
	storel 0, %pk
	jmp @copy_cond
@copy_cond
	%cc_k =l loadl %pk
	%cc_L =l loadl %pl
	%cc_c =w csltl %cc_k, %cc_L
	jnz %cc_c, @copy_body, @copy_done
@copy_body
	%cb_k =l loadl %pk
	%cb_i =l loadl %pi
	%cb_si =l add %cb_i, %cb_k
	%cb_ps =l add %src, %cb_si
	%cb_b =w loadub %cb_ps
	%cb_o =l loadl %po
	%cb_d1 =l add %cb_o, 1
	%cb_di =l add %cb_d1, %cb_k
	%cb_pd =l add %out, %cb_di
	storeb %cb_b, %cb_pd
	%cb_k2 =l add %cb_k, 1
	storel %cb_k2, %pk
	jmp @copy_cond
@copy_done
	%cd_o =l loadl %po
	%cd_L =l loadl %pl
	%cd_o1 =l add %cd_o, 1
	%cd_on =l add %cd_o1, %cd_L
	storel %cd_on, %po
	%cd_j =l loadl %pj
	storel %cd_j, %pi
	jmp @outer_cond
}
