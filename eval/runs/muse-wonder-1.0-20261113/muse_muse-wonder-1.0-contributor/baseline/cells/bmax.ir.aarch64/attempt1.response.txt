export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	%one =l copy 1
	jmp @loop
@loop
	%i =l phi @start %one, @incr %inext
	%max =l phi @start %first, @incr %curmax
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w csltl %max, %val
	jnz %gt, @update, @keep
@update
	jmp @incr
@keep
	jmp @incr
@incr
	%curmax =l phi @update %val, @keep %max
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
