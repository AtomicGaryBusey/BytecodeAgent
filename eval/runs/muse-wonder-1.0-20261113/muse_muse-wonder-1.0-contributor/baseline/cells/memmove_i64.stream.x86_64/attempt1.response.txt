memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
    mov rcx, rdx
.Lback:
    dec rcx
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    test rcx, rcx
    jne .Lback
    ret
.Lforward:
    xor ecx, ecx
.Lfwd_loop:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jb .Lfwd_loop
.Ldone:
    ret
