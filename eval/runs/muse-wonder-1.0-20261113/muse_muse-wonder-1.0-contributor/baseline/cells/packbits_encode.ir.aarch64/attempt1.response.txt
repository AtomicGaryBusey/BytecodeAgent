export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @incr %i_next
	%op =l phi @start 0, @incr %op_next
	%notdone =w csltl %i, %n
	jnz %notdone, @run_init, @done
@done
	ret %op
@run_init
	%p_i =l add %src, %i
	%b_i =w loadub %p_i
	jmp @run_loop
@run_loop
	%r =l phi @run_init 1, @run_inc %r_inc
	%r_lt128 =w csltl %r, 128
	jnz %r_lt128, @run_chk_idx, @run_done
@run_chk_idx
	%ipr =l add %i, %r
	%idx_lt_n =w csltl %ipr, %n
	jnz %idx_lt_n, @run_chk_byte, @run_done
@run_chk_byte
	%p_r =l add %src, %ipr
	%b_r =w loadub %p_r
	%eq_r =w ceqw %b_r, %b_i
	jnz %eq_r, @run_inc, @run_done
@run_inc
	%r_inc =l add %r, 1
	jmp @run_loop
@run_done
	%is_rep =w csgel %r, 3
	jnz %is_rep, @repeat, @literal_init
@repeat
	%op_p2 =l add %op, 2
	%no_fit_r =w csgtl %op_p2, %cap
	jnz %no_fit_r, @fail, @repeat_store
@repeat_store
	%rw =w copy %r
	%hdr_r =w sub 257, %rw
	%o_ptr =l add %out, %op
	storeb %hdr_r, %o_ptr
	%o_ptr1 =l add %o_ptr, 1
	storeb %b_i, %o_ptr1
	%i_next_r =l add %i, %r
	%op_next_r =l add %op, 2
	jmp @incr
@literal_init
	jmp @lit_loop
@lit_loop
	%j =l phi @literal_init %i, @lit_inc %j_inc
	%j_lt_n =w csltl %j, %n
	jnz %j_lt_n, @lit_chk_len, @lit_done
@lit_chk_len
	%j_m_i =l sub %j, %i
	%len_lt128 =w csltl %j_m_i, 128
	jnz %len_lt128, @runj_init, @lit_done
@runj_init
	%p_j =l add %src, %j
	%b_j =w loadub %p_j
	jmp @runj_loop
@runj_loop
	%rj =l phi @runj_init 1, @runj_inc %rj_inc
	%rj_lt128 =w csltl %rj, 128
	jnz %rj_lt128, @runj_chk_idx, @runj_done
@runj_chk_idx
	%jprj =l add %j, %rj
	%idxj_lt_n =w csltl %jprj, %n
	jnz %idxj_lt_n, @runj_chk_byte, @runj_done
@runj_chk_byte
	%p_rj =l add %src, %jprj
	%b_rj =w loadub %p_rj
	%eq_j =w ceqw %b_rj, %b_j
	jnz %eq_j, @runj_inc, @runj_done
@runj_inc
	%rj_inc =l add %rj, 1
	jmp @runj_loop
@runj_done
	%rj_lt3 =w csltl %rj, 3
	jnz %rj_lt3, @lit_inc, @lit_done
@lit_inc
	%j_inc =l add %j, 1
	jmp @lit_loop
@lit_done
	%L =l sub %j, %i
	%L_p1 =l add %L, 1
	%op_p_L1 =l add %op, %L_p1
	%no_fit_l =w csgtl %op_p_L1, %cap
	jnz %no_fit_l, @fail, @lit_store_hdr
@lit_store_hdr
	%Lw =w copy %L
	%hdr_l =w sub %Lw, 1
	%o_ptr_l =l add %out, %op
	storeb %hdr_l, %o_ptr_l
	%dst_base =l add %o_ptr_l, 1
	jmp @copy_loop
@copy_loop
	%k =l phi @lit_store_hdr 0, @copy_inc %k_inc
	%k_lt_L =w csltl %k, %L
	jnz %k_lt_L, @copy_body, @copy_done
@copy_body
	%s_idx =l add %i, %k
	%s_ptr =l add %src, %s_idx
	%b_c =w loadub %s_ptr
	%d_ptr =l add %dst_base, %k
	storeb %b_c, %d_ptr
	jmp @copy_inc
@copy_inc
	%k_inc =l add %k, 1
	jmp @copy_loop
@copy_done
	%op_next_l =l add %op, %L_p1
	jmp @incr
@incr
	%i_next =l phi @repeat_store %i_next_r, @copy_done %j
	%op_next =l phi @repeat_store %op_next_r, @copy_done %op_next_l
	jmp @loop
@fail
	%neg1 =l sub 0, 1
	ret %neg1
}
