push r13

je .Ldone
xor r8d, r8d
.Lmul_i:

jae .Lmul_copy
xor r9d, r9d
.Lmul_j:

add rcx, r9

add rax, r9

inc r9
jmp .Lmul_j

.Lmul_copy:

.Lmul_next_iter:

jmp .Lpow_loop

pop r13
