export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	storel 0, %ip
	jmp @loop
@loop
	%lpc =l loadl %pcp
	%lend =w ceql %lpc, %len
	jnz %lend, @finish, @exec
@exec
	%epc =l loadl %pcp
	%ed =l loadl %dp
	%eoff =l mul %epc, 8
	%eptr =l add %code, %eoff
	%eop =l loadl %eptr
	%c0 =w ceql %eop, 0
	jnz %c0, @finish, @chk1
@chk1
	%c1 =w ceql %eop, 1
	jnz %c1, @op_push, @chk2
@chk2
	%c2 =w ceql %eop, 2
	jnz %c2, @op_add, @chk3
@chk3
	%c3 =w ceql %eop, 3
	jnz %c3, @op_sub, @chk4
@chk4
	%c4 =w ceql %eop, 4
	jnz %c4, @op_mul, @chk5
@chk5
	%c5 =w ceql %eop, 5
	jnz %c5, @op_dup, @chk6
@chk6
	%c6 =w ceql %eop, 6
	jnz %c6, @op_swap, @chk7
@chk7
	%c7 =w ceql %eop, 7
	jnz %c7, @op_drop, @chk8
@chk8
	%c8 =w ceql %eop, 8
	jnz %c8, @op_over, @chk9
@chk9
	%c9 =w ceql %eop, 9
	jnz %c9, @op_lt, @chk10
@chk10
	%c10 =w ceql %eop, 10
	jnz %c10, @op_jz, @chk11
@chk11
	%c11 =w ceql %eop, 11
	jnz %c11, @op_jmp, @op_default
@op_push
	%p_idx =l add %epc, 1
	%p_ooff =l mul %p_idx, 8
	%p_optr =l add %code, %p_ooff
	%p_val =l loadl %p_optr
	%p_soff =l mul %ed, 8
	%p_sptr =l add %stack, %p_soff
	storel %p_val, %p_sptr
	%p_nd =l add %ed, 1
	storel %p_nd, %dp
	%p_npc =l add %epc, 2
	storel %p_npc, %pcp
	jmp @loop
@op_add
	%a_m1 =l sub %ed, 1
	%a_m2 =l sub %ed, 2
	%a_boff =l mul %a_m1, 8
	%a_bptr =l add %stack, %a_boff
	%a_b =l loadl %a_bptr
	%a_aoff =l mul %a_m2, 8
	%a_aptr =l add %stack, %a_aoff
	%a_a =l loadl %a_aptr
	%a_r =l add %a_a, %a_b
	storel %a_r, %a_aptr
	storel %a_m1, %dp
	%a_npc =l add %epc, 1
	storel %a_npc, %pcp
	jmp @loop
@op_sub
	%s_m1 =l sub %ed, 1
	%s_m2 =l sub %ed, 2
	%s_boff =l mul %s_m1, 8
	%s_bptr =l add %stack, %s_boff
	%s_b =l loadl %s_bptr
	%s_aoff =l mul %s_m2, 8
	%s_aptr =l add %stack, %s_aoff
	%s_a =l loadl %s_aptr
	%s_r =l sub %s_a, %s_b
	storel %s_r, %s_aptr
	storel %s_m1, %dp
	%s_npc =l add %epc, 1
	storel %s_npc, %pcp
	jmp @loop
@op_mul
	%m_m1 =l sub %ed, 1
	%m_m2 =l sub %ed, 2
	%m_boff =l mul %m_m1, 8
	%m_bptr =l add %stack, %m_boff
	%m_b =l loadl %m_bptr
	%m_aoff =l mul %m_m2, 8
	%m_aptr =l add %stack, %m_aoff
	%m_a =l loadl %m_aptr
	%m_r =l mul %m_a, %m_b
	storel %m_r, %m_aptr
	storel %m_m1, %dp
	%m_npc =l add %epc, 1
	storel %m_npc, %pcp
	jmp @loop
@op_dup
	%d_m1 =l sub %ed, 1
	%d_toff =l mul %d_m1, 8
	%d_tptr =l add %stack, %d_toff
	%d_val =l loadl %d_tptr
	%d_soff =l mul %ed, 8
	%d_sptr =l add %stack, %d_soff
	storel %d_val, %d_sptr
	%d_nd =l add %ed, 1
	storel %d_nd, %dp
	%d_npc =l add %epc, 1
	storel %d_npc, %pcp
	jmp @loop
@op_swap
	%w_m1 =l sub %ed, 1
	%w_m2 =l sub %ed, 2
	%w_boff =l mul %w_m1, 8
	%w_bptr =l add %stack, %w_boff
	%w_b =l loadl %w_bptr
	%w_aoff =l mul %w_m2, 8
	%w_aptr =l add %stack, %w_aoff
	%w_a =l loadl %w_aptr
	storel %w_b, %w_aptr
	storel %w_a, %w_bptr
	%w_npc =l add %epc, 1
	storel %w_npc, %pcp
	jmp @loop
@op_drop
	%dr_nd =l sub %ed, 1
	storel %dr_nd, %dp
	%dr_npc =l add %epc, 1
	storel %dr_npc, %pcp
	jmp @loop
@op_over
	%o_m2 =l sub %ed, 2
	%o_aoff =l mul %o_m2, 8
	%o_aptr =l add %stack, %o_aoff
	%o_val =l loadl %o_aptr
	%o_soff =l mul %ed, 8
	%o_sptr =l add %stack, %o_soff
	storel %o_val, %o_sptr
	%o_nd =l add %ed, 1
	storel %o_nd, %dp
	%o_npc =l add %epc, 1
	storel %o_npc, %pcp
	jmp @loop
@op_lt
	%t_m1 =l sub %ed, 1
	%t_m2 =l sub %ed, 2
	%t_boff =l mul %t_m1, 8
	%t_bptr =l add %stack, %t_boff
	%t_b =l loadl %t_bptr
	%t_aoff =l mul %t_m2, 8
	%t_aptr =l add %stack, %t_aoff
	%t_a =l loadl %t_aptr
	%t_cmp =w csltl %t_a, %t_b
	%t_r =l extsw %t_cmp
	storel %t_r, %t_aptr
	storel %t_m1, %dp
	%t_npc =l add %epc, 1
	storel %t_npc, %pcp
	jmp @loop
@op_jz
	%z_idx =l add %epc, 1
	%z_ooff =l mul %z_idx, 8
	%z_optr =l add %code, %z_ooff
	%z_tgt =l loadl %z_optr
	%z_m1 =l sub %ed, 1
	%z_toff =l mul %z_m1, 8
	%z_tptr =l add %stack, %z_toff
	%z_val =l loadl %z_tptr
	storel %z_m1, %dp
	%z_is0 =w ceql %z_val, 0
	jnz %z_is0, @jz_taken, @jz_next
@jz_taken
	storel %z_tgt, %pcp
	jmp @loop
@jz_next
	%z_npc =l add %epc, 2
	storel %z_npc, %pcp
	jmp @loop
@op_jmp
	%j_idx =l add %epc, 1
	%j_ooff =l mul %j_idx, 8
	%j_optr =l add %code, %j_ooff
	%j_tgt =l loadl %j_optr
	storel %j_tgt, %pcp
	jmp @loop
@op_default
	%df_npc =l add %epc, 1
	storel %df_npc, %pcp
	jmp @loop
@finish
	%f_d =l loadl %dp
	storel %f_d, %ip
	jmp @clear
@clear
	%cl_i =l loadl %ip
	%cl_has =w csltl %cl_i, %cap
	jnz %cl_has, @clear_body, @checkret
@clear_body
	%cl_off =l mul %cl_i, 8
	%cl_ptr =l add %stack, %cl_off
	storel 0, %cl_ptr
	%cl_ni =l add %cl_i, 1
	storel %cl_ni, %ip
	jmp @clear
@checkret
	%r_d =l loadl %dp
	%r_z =w ceql %r_d, 0
	jnz %r_z, @retzero, @rettop
@retzero
	ret 0
@rettop
	%r_m1 =l sub %r_d, 1
	%r_off =l mul %r_m1, 8
	%r_ptr =l add %stack, %r_off
	%r_v =l loadl %r_ptr
	ret %r_v
}
