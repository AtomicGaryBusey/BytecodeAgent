export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%initmin =l copy 9223372036854775807
	%negmax =l copy -9223372036854775807
	%initmax =l sub %negmax, 1
	jmp @loop
@loop
	%i =l phi @start 0, @inc %inext
	%min =l phi @start %initmin, @inc %min2
	%max =l phi @start %initmax, @inc %max2
	%sum =l phi @start 0, @inc %sum2
	%cnt =l phi @start 0, @inc %cnt2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%sum2 =l add %sum, %y
	%oddbit =l and %y, 1
	%isodd =w cnel %oddbit, 0
	%oddl =l extsw %isodd
	%cnt2 =l add %cnt, %oddl
	%ltmin =w csltl %y, %min
	jnz %ltmin, @setmin, @keepmin
@setmin
	jmp @aftermin
@keepmin
	jmp @aftermin
@aftermin
	%min2 =l phi @setmin %y, @keepmin %min
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @setmax, @keepmax
@setmax
	jmp @aftermax
@keepmax
	jmp @aftermax
@aftermax
	%max2 =l phi @setmax %y, @keepmax %max
	jmp @inc
@inc
	%inext =l add %i, 1
	jmp @loop
@end
	%p1 =l add %out, 8
	%p2 =l add %out, 16
	storel %min, %out
	storel %max, %p1
	storel %sum, %p2
	ret %cnt
}
