packbits_encode:
push r12
push r13
push r14
xor r12d, r12d
xor r13d, r13d
.Louter:
cmp r12, rsi
jge .Ldone
movzx r10d, byte ptr [rdi + r12]
lea r8, [r12 + 1]
mov r11d, 1
.Lrun1:
cmp r8, rsi
jge .Lrun1_done
cmp r11, 128
jge .Lrun1_done
cmp byte ptr [rdi + r8], r10b
jne .Lrun1_done
inc r8
inc r11
jmp .Lrun1
.Lrun1_done:
cmp r11, 3
jae .Lrepeat
mov r14, r12
.Llit_loop:
cmp r14, rsi
jge .Llit_done
mov rax, r14
sub rax, r12
cmp rax, 128
jge .Llit_done
movzx r10d, byte ptr [rdi + r14]
lea r8, [r14 + 1]
mov r11d, 1
.Lrun2:
cmp r8, rsi
jge .Lrun2_done
cmp r11, 128
jge .Lrun2_done
cmp byte ptr [rdi + r8], r10b
jne .Lrun2_done
inc r8
inc r11
jmp .Lrun2
.Lrun2_done:
cmp r11, 3
jae .Llit_done
inc r14
jmp .Llit_loop
.Llit_done:
mov rax, r14
sub rax, r12
mov r9, rcx
sub r9, r13
lea r10, [rax + 1]
cmp r10, r9
jg .Lfail
mov r9, rax
dec r9
mov byte ptr [rdx + r13], r9b
inc r13
mov r8, r12
.Lcopy:
cmp r8, r14
jge .Lcopy_done
mov r10b, byte ptr [rdi + r8]
mov byte ptr [rdx + r13], r10b
inc r13
inc r8
jmp .Lcopy
.Lcopy_done:
mov r12, r14
jmp .Louter
.Lrepeat:
mov r9, rcx
sub r9, r13
cmp r9, 2
jl .Lfail
mov eax, 257
sub rax, r11
mov byte ptr [rdx + r13], al
inc r13
mov byte ptr [rdx + r13], r10b
inc r13
add r12, r11
jmp .Louter
.Ldone:
mov rax, r13
pop r14
pop r13
pop r12
ret
.Lfail:
mov rax, -1
pop r14
pop r13
pop r12
ret
