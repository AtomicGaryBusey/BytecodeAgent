export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nnb =l mul %nn, 8
	%tnn =l mul %nn, 2
	%prod =l add %tmp, %nnb
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%pe =l alloc8 8
	%pidx =l alloc8 8
	%psum =l alloc8 8
	storel 0, %pidx
	jmp @zocond
@zocond
	%z1 =l loadl %pidx
	%z2 =w csltl %z1, %nn
	jnz %z2, @zobody, @zodone
@zobody
	%z3 =l loadl %pidx
	%z4 =l mul %z3, 8
	%z5 =l add %out, %z4
	storel 0, %z5
	%z6 =l loadl %pidx
	%z7 =l add %z6, 1
	storel %z7, %pidx
	jmp @zocond
@zodone
	storel 0, %pi
	jmp @dgcond
@dgcond
	%d1 =l loadl %pi
	%d2 =w csltl %d1, %n
	jnz %d2, @dgbody, @dgdone
@dgbody
	%d3 =l loadl %pi
	%d4 =l mul %d3, %n
	%d5 =l add %d4, %d3
	%d6 =l mul %d5, 8
	%d7 =l add %out, %d6
	storel 1, %d7
	%d8 =l loadl %pi
	%d9 =l add %d8, 1
	storel %d9, %pi
	jmp @dgcond
@dgdone
	storel 0, %pidx
	jmp @cpcond
@cpcond
	%c1 =l loadl %pidx
	%c2 =w csltl %c1, %nn
	jnz %c2, @cpbody, @cpdone
@cpbody
	%c3 =l loadl %pidx
	%c4 =l mul %c3, 8
	%c5 =l add %a, %c4
	%c6 =l loadl %c5
	%c7 =l add %tmp, %c4
	storel %c6, %c7
	%c8 =l loadl %pidx
	%c9 =l add %c8, 1
	storel %c9, %pidx
	jmp @cpcond
@cpdone
	storel 0, %pe
	jmp @pwcond
@pwcond
	%w1 =l loadl %pe
	%w2 =w csltl %w1, %k
	jnz %w2, @pwbody, @pwdone
@pwbody
	storel 0, %pi
	jmp @micond
@micond
	%i1 =l loadl %pi
	%i2 =w csltl %i1, %n
	jnz %i2, @mibody, @midone
@mibody
	storel 0, %pj
	jmp @mjcond
@mjcond
	%j1 =l loadl %pj
	%j2 =w csltl %j1, %n
	jnz %j2, @mjbody, @mjdone
@mjbody
	storel 0, %psum
	storel 0, %pt
	jmp @mtcond
@mtcond
	%t1 =l loadl %pt
	%t2 =w csltl %t1, %n
	jnz %t2, @mtbody, @mtdone
@mtbody
	%t3 =l loadl %pi
	%t4 =l mul %t3, %n
	%t5 =l loadl %pt
	%t6 =l add %t4, %t5
	%t7 =l mul %t6, 8
	%t8 =l add %out, %t7
	%t9 =l loadl %t8
	%t10 =l loadl %pt
	%t11 =l mul %t10, %n
	%t12 =l loadl %pj
	%t13 =l add %t11, %t12
	%t14 =l mul %t13, 8
	%t15 =l add %tmp, %t14
	%t16 =l loadl %t15
	%t17 =l mul %t9, %t16
	%t18 =l loadl %psum
	%t19 =l add %t18, %t17
	storel %t19, %psum
	%t20 =l loadl %pt
	%t21 =l add %t20, 1
	storel %t21, %pt
	jmp @mtcond
@mtdone
	%m1 =l loadl %pi
	%m2 =l mul %m1, %n
	%m3 =l loadl %pj
	%m4 =l add %m2, %m3
	%m5 =l mul %m4, 8
	%m6 =l add %prod, %m5
	%m7 =l loadl %psum
	storel %m7, %m6
	%m8 =l loadl %pj
	%m9 =l add %m8, 1
	storel %m9, %pj
	jmp @mjcond
@mjdone
	%i3 =l loadl %pi
	%i4 =l add %i3, 1
	storel %i4, %pi
	jmp @micond
@midone
	storel 0, %pidx
	jmp @pccond
@pccond
	%p1 =l loadl %pidx
	%p2 =w csltl %p1, %nn
	jnz %p2, @pcbody, @pcdone
@pcbody
	%p3 =l loadl %pidx
	%p4 =l mul %p3, 8
	%p5 =l add %prod, %p4
	%p6 =l loadl %p5
	%p7 =l add %out, %p4
	storel %p6, %p7
	%p8 =l loadl %pidx
	%p9 =l add %p8, 1
	storel %p9, %pidx
	jmp @pccond
@pcdone
	%w3 =l loadl %pe
	%w4 =l add %w3, 1
	storel %w4, %pe
	jmp @pwcond
@pwdone
	storel 0, %pidx
	jmp @ztcond
@ztcond
	%s1 =l loadl %pidx
	%s2 =w csltl %s1, %tnn
	jnz %s2, @ztbody, @ztdone
@ztbody
	%s3 =l loadl %pidx
	%s4 =l mul %s3, 8
	%s5 =l add %tmp, %s4
	storel 0, %s5
	%s6 =l loadl %pidx
	%s7 =l add %s6, 1
	storel %s7, %pidx
	jmp @ztcond
@ztdone
	ret 0
}
