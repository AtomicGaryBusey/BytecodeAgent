export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%max =l phi @start 0, @next %maxnext
	%has =w csltl %i, %n
	jnz %has, @body, @end
@body
	%ptr =l add %p, %i
	%b =l loadub %ptr
	%gt =w csltl %max, %b
	jnz %gt, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%maxnext =l phi @update %b, @keep %max
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
