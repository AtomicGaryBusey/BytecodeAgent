

export function l $binary_search(l a, l n, l key) {
  @start:
    movl #0, %1
    subll %2, %n, #1
    jmp @loop
  @loop:
    csltl %3, %2, %1
    jnz %3, @not_found
    addl %4, %1, %2
    divl %3, %4, #2
    imul l %5, %3, #8
    addl %6, %a, %5
    loadl %7, %6
    ceql %8, %7, %key
    jnz %8, @found
@check_less:
    csltl %8, %7, %key
    jnz %8, @low_update
    jmp @high_dec
@low_update:
    addl %1, %3, #1
    jmp @loop
@high_dec:
    subll %2, %3, #1
    jmp @loop
@found:
    movl %0, %3
    jmp @done
@not_found:
    movl %0, #-1
    jmp @done
@done:
    retl %0
}
