

export function l $binary_search(l a, l n, l key) {
  @start:
    movl %1, #0
    ceql %2, n, %1
    jnz %2, @set_result_minus1
    movl %2, #0
    subll %3, n, #1
    jmp @loop
  @set_result_minus1:
    movl %0, #-1
    jmp @done
  @loop:
    csltl %10, %3, %2
    jnz %10, @not_found
    addl %5, %2, %3
    divl %4, %5, #2
    imul l %6, %4, #8
    addl %6, a, %6
    loadl %7, %6
    ceql %11, %7, key
    jnz %11, @found
    csltl %10, %7, key
    jnz %10, @increase_low
    subll %3, %4, #1
    jmp @loop
  @increase_low:
    addl %2, %4, #1
    jmp @loop
  @found:
    movl %0, %4
    jmp @done
  @not_found:
    movl %0, #-1
    jmp @done
  @done:
}
