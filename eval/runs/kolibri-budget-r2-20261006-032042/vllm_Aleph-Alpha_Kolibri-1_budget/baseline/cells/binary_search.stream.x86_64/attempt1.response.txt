

binary_search:
mov r8, 0
mov r9, rsi
dec r9
.Lloop:
cmp r8, r9
jle .Lmid
jmp .not_found
.Lmid:
mov rax, r8
add rax, r9
shr rax, 1
mov r10, qword ptr [rdi + rax*8]
cmp r10, rdx
je .found
jl .Lleft
jg .Lright
.found:
jmp .done
.Lleft:
lea r8, [rax + 1]
jmp .Lloop
.Lright:
lea r9, [rax - 1]
jmp .Lloop
.not_found:
mov rax, -1
jmp .done
.done:
ret
