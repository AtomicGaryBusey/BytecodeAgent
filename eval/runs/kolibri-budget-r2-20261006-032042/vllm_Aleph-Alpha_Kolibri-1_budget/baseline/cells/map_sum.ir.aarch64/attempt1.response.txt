

export function l $map_sum(l a, l n, l f) {
  @map_sum:
    %ptr =l a
    %count =l n
    %sum =l 0l
    jmp @check

  @check:
    %eq =w ceql %count, 0l
    jnz %eq, @done

  @body:
    %elem =l loadl [%ptr]
    %fval =l call %f(l %elem)
    %sum =l addl %sum, %fval
    %ptr =l addl %ptr, 8l
    %count =l subtl %count, 1l
    jmp @check

  @done:
    %sum
}
