

export function l $map_sum(l p, l c, l g) {
  @map_sum:
    %sum =l 0l
    %ptr =l p
    %count =l c
    %func =l g
    jmp @check

  @check:
    %eq =w ceql %count, 0l
    jnz %eq, @done

  @body:
    %elem =l loadl [%ptr]
    %fval =l call %func(l %elem)
    %sum =l addl %sum, %fval
    %ptr =l addl %ptr, 8l
    %count =l subtl %count, 1l
    jmp @check

  @done:
    %sum
}
