

export function l $memmove_i64( l dst, l src, l n ) {
@start:
%counter = n
%le = cltle dst, src
jnz @forward_init
%n8 = mull %counter, 8
%ptr_dst = addl dst, %n8
%ptr_src = addl src, %n8
jmp @b_loop
@forward_init:
%ptr_dst = dst
%ptr_src = src
jmp @f_loop
@f_loop:
%le_f = cltle %counter, 0
jnz @f_done
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%ptr_src = addl %ptr_src, 8
%ptr_dst = addl %ptr_dst, 8
%counter = subll %counter, 1
jmp @f_loop
@f_done:
jmp @done
@b_loop:
%le_b = cltle %counter, 0
jnz @b_done
%ptr_dst = subll %ptr_dst, 8
%ptr_src = subll %ptr_src, 8
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%counter = subll %counter, 1
jmp @b_loop
@b_done:
jmp @done
@done:
%ret = 0
}
