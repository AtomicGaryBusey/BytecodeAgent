

export function l $memmove_i64(l %a, l %b, l %c)
@start:
%lt = csltl %a, %b
jnz @forward
jmp @backward
@forward:
%ptr_src = %b
%ptr_dst = %a
%cnt = %c
jmp @f_loop
@backward:
%n8 = mull %c, 8
%ptr_dst = addl %a, %n8
%ptr_dst = subll %ptr_dst, 8
%ptr_src = addl %b, %n8
%ptr_src = subll %ptr_src, 8
%cnt = %c
jmp @b_loop
@f_loop:
%cond = csltl 0, %cnt
jnz @f_copy
jmp @f_done
@f_copy:
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%ptr_src = addl %ptr_src, 8
%ptr_dst = addl %ptr_dst, 8
%cnt = subll %cnt, 1
jmp @f_loop
@f_done:
jmp @done
@b_loop:
%cond = csltl 0, %cnt
jnz @b_copy
jmp @b_done
@b_copy:
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%ptr_src = subll %ptr_src, 8
%ptr_dst = subll %ptr_dst, 8
%cnt = subll %cnt, 1
jmp @b_loop
@b_done:
jmp @done
@done:
ret 0
