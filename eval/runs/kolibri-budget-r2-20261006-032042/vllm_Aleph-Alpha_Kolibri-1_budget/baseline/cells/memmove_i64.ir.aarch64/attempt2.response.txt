

export function l $memmove_i64( l %a, l %b, l %c ) {
@start:
%lt1 = csltl %c, 1
jnz @done
%lt = csltl %a, %b
jnz @forward
%lt_rev = csltl %b, %a
jnz @backward
jmp @done
@forward:
%ptr_dst = %a
%ptr_src = %b
%cnt = %c
jmp @f_loop
@backward:
%n8 = mull %c, 8
%ptr_dst = addl %a, %n8
%ptr_dst = subll %ptr_dst, 8
%ptr_src = addl %b, %n8
%ptr_src = subll %ptr_src, 8
%cnt = %c
jmp @b_loop
@f_loop:
%eq_zero = ceql %cnt, 0
jnz @f_done
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%ptr_src = addl %ptr_src, 8
%ptr_dst = addl %ptr_dst, 8
%cnt = subll %cnt, 1
jmp @f_loop
@f_done:
jmp @done
@b_loop:
%eq_zero = ceql %cnt, 0
jnz @b_done
%val = loadl %ptr_src
%store = storel %ptr_dst, %val
%ptr_src = subll %ptr_src, 8
%ptr_dst = subll %ptr_dst, 8
%cnt = subll %cnt, 1
jmp @b_loop
@b_done:
jmp @done
@done:
ret 0
}
