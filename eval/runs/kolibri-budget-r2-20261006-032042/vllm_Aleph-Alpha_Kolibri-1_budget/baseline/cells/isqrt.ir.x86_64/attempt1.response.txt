

export function l $isqrt(l $n)
@entry
    %lo = 1
    %hi = $n
    jmp @loop
@loop
    %lt = csltl %hi, %lo
    jnz @done
    %diff = %hi - %lo
    %half = %diff / 2
    %mid = %lo + %half
    %div = $n / %mid
    %gt = csltl %div, %mid
    jnz @midtoobig
    %lo = %mid + 1
    jmp @loop
@midtoobig
    %hi = %mid - 1
    jmp @loop
@done
    ret %hi
