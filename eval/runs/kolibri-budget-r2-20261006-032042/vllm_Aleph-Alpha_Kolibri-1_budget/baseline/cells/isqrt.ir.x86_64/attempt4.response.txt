

export function l $isqrt(l %arg) l {
@entry
    %lo = $0
    %hi = $4294967295
    jmp @loop
@loop
    %lt = cltlu %hi, %lo
    jnz @done
    %sum = %lo + %hi
    %mid = %sum / $2
    %mid_eq_0 = ceql %mid, $0
    jnz @mid_not_zero
    %lo = $1
    jmp @loop
@mid_not_zero
    %quot = %arg / %mid
    %lt_quot_mid = cltlu %quot, %mid
    jnz @big
    %lo = %mid + $1
    jmp @loop
@big
    %hi = %mid - $1
    jmp @loop
@done
    ret %hi
}
