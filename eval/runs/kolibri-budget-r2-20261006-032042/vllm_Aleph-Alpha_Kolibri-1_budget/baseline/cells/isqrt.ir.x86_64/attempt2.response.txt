

export function l $isqrt(l $arg)
@entry
    %lo = 0
    %hi = 4294967295
    jmp @loop
@loop
    %lt = csltl %hi, %lo
    jnz @done
    %sum = %lo + %hi
    %mid = %sum / 2
    %sq = %mid * %mid
    %gt = csltl $arg, %sq
    jnz @big
    %lo = %mid + 1
    jmp @loop
@big
    %hi = %mid - 1
    jmp @loop
@done
    ret %hi
