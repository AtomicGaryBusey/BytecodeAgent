

export function l $collatz_max(l %n)
{
  @entry
    l %max = 0
    l %k = 1
    jmp @outer_loop
  @outer_loop
    %lt_nk = csltl %n, %k
    jnz %lt_nk, @done
    l %x = %k
    l %steps = 0
    jmp @inner_loop
  @inner_loop
    %eq_one = ceql %x, 1
    jnz %eq_one, @inner_done
    %parity = andl %x, 1
    %is_even = ceql %parity, 0
    jnz %is_even, @even_case
  @odd_case
    l %temp = addl %x, %x
    l %temp = addl %temp, %x
    l %temp = addl %temp, 1
    l %x = %temp
    jmp @inc_steps
  @even_case
    l %x = shrl %x, 1
    jmp @inc_steps
  @inc_steps
    l %steps = addl %steps, 1
    jmp @inner_loop
  @inner_done
    %lt_max = csltl %max, %steps
    jnz %lt_max, @set_max
    jmp @increment_k
  @set_max
    l %max = %steps
    jmp @increment_k
  @increment_k
    l %k = addl %k, 1
    jmp @outer_loop
  @done
    ret l %max
}
