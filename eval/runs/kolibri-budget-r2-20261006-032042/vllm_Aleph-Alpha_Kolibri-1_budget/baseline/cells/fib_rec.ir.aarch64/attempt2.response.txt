

export function l $fib_rec(l %n) {
@entry:
  %one =l 1
  %two =l 2
  %lt2 =w csltl(l %n, l %two)
  jnz %lt2, @base
  %n1 =l subl(l %n, l %one)
  %n2 =l subl(l %n, l %two)
  %a =l call $fib_rec(l %n1)
  %b =l call $fib_rec(l %n2)
  %res =l addl(l %a, l %b)
  jmp @exit
@base:
  %res =l %n
@exit:
  ret %res
}
