

memmove_i64:
mov rax, rdi
sub rax, rsi
jl .Lforward
.Lbackward:
mov rcx, rdx
test rcx, rcx
jz .Ldone
lea r8, [rsi + rdx*8 - 8]
lea r9, [rdi + rdx*8 - 8]
.Lcopy_bwd:
mov rax, qword ptr [r8]
mov qword ptr [r9], rax
sub r8, 8
sub r9, 8
dec rcx
jnz .Lcopy_bwd
mov rax, 0
jmp .Ldone
.Lforward:
mov rcx, rdx
test rcx, rcx
jz .Ldone
mov r8, rsi
mov r9, rdi
.Lcopy_fwd:
mov rax, qword ptr [r8]
mov qword ptr [r9], rax
add r8, 8
add r9, 8
dec rcx
jnz .Lcopy_fwd
mov rax, 0
.Ldone:
ret
