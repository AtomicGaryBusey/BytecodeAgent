

memmove_i64:
    test rdx, rdx
    jz .done
    cmp rdi, rsi
    je .done
    jl .forward
.backward:
    mov rcx, rdx
    add rdi, rdx
    dec rdi
    add rsi, rdx
    dec rsi
.bloop:
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    dec rsi
    dec rdi
    dec rcx
    jnz .bloop
    jmp .done
.forward:
    mov rcx, rdx
.floop:
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    inc rsi
    inc rdi
    dec rcx
    jnz .floop
    jmp .done
.done:
    xor rax, rax
    ret
