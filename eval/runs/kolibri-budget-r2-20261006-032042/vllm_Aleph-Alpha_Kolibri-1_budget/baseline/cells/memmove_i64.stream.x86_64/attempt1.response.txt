

memmove_i64:
    test rdx, rdx
    jz .done
    cmp rdi, rsi
    je .done
    jl .forward
.forward:
    mov rcx, rdx
.floop:
    mov rax, [rsi]
    mov [rdi], rax
    inc rsi
    inc rdi
    dec rcx
    jnz .floop
    jmp .done
.backward:
    mov r10, rdx
    dec r10
    shl r10, 3
    add rdi, r10
    add rsi, r10
    mov rcx, rdx
.bloop:
    mov rax, [rsi]
    mov [rdi], rax
    dec rsi
    dec rdi
    dec rcx
    jnz .bloop
    jmp .done
.done:
    xor rax, rax
    ret
