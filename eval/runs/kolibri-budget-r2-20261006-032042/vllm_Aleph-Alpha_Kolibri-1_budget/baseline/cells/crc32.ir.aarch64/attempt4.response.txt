

export function l $crc32(l %p, l %n)
@start
    %crc =w 0xFFFFFFFF
    %zero_l =l 0
    %one_l =l 1
    %eight =w 8
    %poly =w 0xEDB88320
    jmp @outer_loop
@outer_loop
    %gt0 =w csltl %zero_l, %n
    jnz @outer_body, @done
@outer_body
    %b =w loadub %p
    %crc =w xor %crc, %b
    %j =w 0
    jmp @inner_loop
@inner_loop
    %lt8 =w csltl %j, %eight
    jnz @inner_body, @inner_done
@inner_body
    %temp =w shr %crc, 1
    %lsb =w and %crc, 1
    %mask =w and %lsb, %poly
    %crc =w xor %temp, %mask
    %j =w add %j, 1
    jmp @inner_loop
@inner_done
    %p =l addl %p, %one_l
    %n =l subl %n, %one_l
    jmp @outer_loop
@done
    %crc_final =w xor %crc, 0xFFFFFFFF
    %res =l extuw %crc_final
    ret %res
