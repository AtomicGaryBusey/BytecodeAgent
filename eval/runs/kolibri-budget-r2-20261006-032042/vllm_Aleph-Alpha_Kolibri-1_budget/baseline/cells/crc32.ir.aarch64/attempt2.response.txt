

export function l $crc32(l %p, l %n)
@start {
    %c =w 0xFFFFFFFF
    %i =w %n
    %j =w 0
    %zero =w 0
    %eight =w 8
    %poly =w 0xEDB88320
    jmp @outer_loop
}
@outer_loop {
    %gt0 =w csltl %zero, %i
    jnz @outer_body, @done
}
@outer_body {
    %b =w loadub %p
    %c =w xor %c, %b
    %j =w 0
    jmp @inner_loop
}
@inner_loop {
    %lt8 =w csltl %j, %eight
    jnz @inner_body, @inner_done
}
@inner_body {
    %temp =w shr %c, 1
    %lsb =w and %c, 1
    %mask =w and %lsb, %poly
    %c =w xor %temp, %mask
    %j =w add %j, 1
    jmp @inner_loop
}
@inner_done {
    %p =l addl %p, 1
    %i =w sub %i, 1
    jmp @outer_loop
}
@done {
    %c_final =w xor %c, 0xFFFFFFFF
    %res =l extuw %c_final
    ret %res
}
