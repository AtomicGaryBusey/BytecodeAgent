

export function l $fib_rec(l $n)
  @fib_rec:
    %zero =l 0
    %one =l 1
    %eq0 =l ceql $n, %zero
    jnz %eq0, skip_zero
    retl %zero
  @skip_zero:
    %eq1 =l ceql $n, %one
    jnz %eq1, skip_one
    retl %one
  @skip_one:
    %n1 =l subl $n, 1
    %n2 =l subl $n, 2
    %r1 =l call $fib_rec(l %n1)
    %r2 =l call $fib_rec(l %n2)
    %result =l addl %r1, %r2
    retl %result
