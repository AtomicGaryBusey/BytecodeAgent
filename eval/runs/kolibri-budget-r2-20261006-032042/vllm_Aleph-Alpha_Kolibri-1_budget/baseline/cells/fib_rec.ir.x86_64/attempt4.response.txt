

export function l $fib_rec(l %n):l
  @fib_rec:
    %one =l 1
    %lt1 =w csltl %n, %one
    jnz %lt1, return_zero
    %lt2 =w csltl %one, %n
    jnz %lt2, recursive
    retl %one
  @return_zero:
    retl %n
  @recursive:
    %n1 =l subl %n, 1
    %n2 =l subl %n, 2
    %r1 =l call $fib_rec(l %n1)
    %r2 =l call $fib_rec(l %n2)
    %result =l addl %r1, %r2
    retl %result
