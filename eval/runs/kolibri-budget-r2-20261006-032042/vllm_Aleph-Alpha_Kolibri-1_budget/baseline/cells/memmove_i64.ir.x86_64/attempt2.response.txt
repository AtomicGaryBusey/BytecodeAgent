

export function l $memmove_i64(l $dst, l $src, l $n)
@memmove_i64:
    %result = 0
    %cmp1 = csltl $dst, $src
    %cmp2 = csltl $src, $dst
    jnz %cmp1, @forward
    jnz %cmp2, @backward
    ret %result
@forward:
    %src_ptr = $src
    %dst_ptr = $dst
    %counter = $n
    jnz %counter, @loop_f
    ret %result
@loop_f:
    %val = loadl [%src_ptr]
    storel [%dst_ptr], %val
    %src_ptr = addl %src_ptr, 8
    %dst_ptr = addl %dst_ptr, 8
    %counter = subl %counter, 1
    jnz %counter, @loop_f
    ret %result
@backward:
    %src_ptr = $src
    %dst_ptr = $dst
    %pre_counter = $n
    jnz %pre_counter, @pre_b
    %counter = $n
    jnz %counter, @loop_b
    ret %result
@pre_b:
    %src_ptr = addl %src_ptr, 8
    %dst_ptr = addl %dst_ptr, 8
    %pre_counter = subl %pre_counter, 1
    jnz %pre_counter, @pre_b
    %counter = $n
    jnz %counter, @loop_b
    ret %result
@loop_b:
    %src_ptr = subl %src_ptr, 8
    %dst_ptr = subl %dst_ptr, 8
    %val = loadl [%src_ptr]
    storel [%dst_ptr], %val
    %counter = subl %counter, 1
    jnz %counter, @loop_b
    ret %result
