

export function l memmove(l $dst, l $src, l $n)
@memmove:
    %result = 0
    %cmp = csltl $dst, $src
    jnz %cmp, @backward
    jmp @forward
@backward:
    %src_ptr = $src
    %dst_ptr = $dst
    %pre_counter = $n
    jnz %pre_counter, @pre_b
    %counter = $n
    jmp @loop_b_check
@pre_b:
    %src_ptr = addl %src_ptr, 8
    %dst_ptr = addl %dst_ptr, 8
    %pre_counter = subl %pre_counter, 1
    jnz %pre_counter, @pre_b
    %counter = $n
    jmp @loop_b_check
@loop_b_check:
    jnz %counter, @loop_b_body
    ret %result
@loop_b_body:
    %src_ptr = subl %src_ptr, 8
    %dst_ptr = subl %dst_ptr, 8
    %val = loadl [%src_ptr]
    storel [%dst_ptr], %val
    %counter = subl %counter, 1
    jmp @loop_b_check
@forward:
    %src_ptr = $src
    %dst_ptr = $dst
    %counter = $n
    jmp @loop_f_check
@loop_f_check:
    jnz %counter, @loop_f_body
    ret %result
@loop_f_body:
    %val = loadl [%src_ptr]
    storel [%dst_ptr], %val
    %src_ptr = addl %src_ptr, 8
    %dst_ptr = addl %dst_ptr, 8
    %counter = subl %counter, 1
    jmp @loop_f_check
