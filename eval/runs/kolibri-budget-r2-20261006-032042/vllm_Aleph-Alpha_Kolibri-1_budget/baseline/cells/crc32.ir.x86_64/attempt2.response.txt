

export function l $crc32(l $p, l $n) {
  @init
  %crc =w 4294967295
  %ptr =l $p
  %cnt =l $n
  jmp @outer_loop
  @outer_loop
  %cnt_zero =w ceql %cnt, 0
  jnz @after_loop
  %byte =w loadub %ptr
  %crc =w xor %crc, %byte
  %inner_cnt =w 8
  @inner_loop
  %inner_zero =w ceql %inner_cnt, 0
  jnz @after_inner
  %lsb =w and %crc, 1
  %mask =w and %lsb, 3988292384
  %crc =w shr %crc, 1
  %crc =w xor %crc, %mask
  %inner_cnt =w sub %inner_cnt, 1
  jmp @inner_loop
  @after_inner
  %ptr =l add %ptr, 1
  %cnt =l sub %cnt, 1
  jmp @outer_loop
  @after_loop
  %result =w xor %crc, 4294967295
  extuw %result
}
