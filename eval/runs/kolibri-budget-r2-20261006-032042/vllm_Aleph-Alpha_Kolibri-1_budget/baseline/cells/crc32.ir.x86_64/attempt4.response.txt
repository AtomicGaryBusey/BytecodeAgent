

export function l $crc32(l p, l n) {
  @init
  %ptr = l %p
  %cnt = l %n
  %crc = w 4294967295
  jmp @outer_loop
  @outer_loop
  %cnt_zero = w ceql %cnt, 0
  jnz @after_loop
  %byte = w loadub %ptr
  %crc = w xor %crc, %byte
  %inner_cnt = w 8
  jmp @inner_loop
  @inner_loop
  %inner_zero = w ceql %inner_cnt, 0
  jnz @after_inner
  %lsb = w and %crc, 1
  %poly = w 3988292384
  %eq_lsb = w ceql %lsb, 1
  jnz @lsb_one
  %crc = w shr %crc, 1
  jmp @inner_dec
  @lsb_one
  %crc = w shr %crc, 1
  %crc = w xor %crc, %poly
  @inner_dec
  %inner_cnt = w sub %inner_cnt, 1
  jmp @inner_loop
  @after_inner
  %ptr = l add %ptr, 1
  %cnt = l sub %cnt, 1
  jmp @outer_loop
  @after_loop
  %result = w xor %crc, 4294967295
  extuw %result
  retl %result
}
