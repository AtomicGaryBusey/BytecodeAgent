

export function l $crc32(l $p, l $n) {
  @init
  %ptr =l $p
  %cnt =l $n
  %crc =w 4294967295
  jmp @outer_loop
  @outer_loop
  %cnt_zero =w ceql %cnt, 0
  jnz @outer_end
  %byte =w loadub %ptr
  %crc =w xor %crc, %byte
  %shift_inner =w 8
  jmp @shift_loop
  @shift_loop
  %shift_inner_zero =w ceql %shift_inner, 0
  jnz @shift_end
  %lsb =w and %crc, 1
  %mask =w and %lsb, 3988292384
  %crc =w shr %crc, 1
  %crc =w xor %crc, %mask
  %shift_inner =w sub %shift_inner, 1
  jmp @shift_loop
  @shift_end
  %ptr =l add %ptr, 1
  %cnt =l sub %cnt, 1
  jmp @outer_loop
  @outer_end
  %result =w xor %crc, 4294967295
  %ret =l extuw %result
  %ret
}
