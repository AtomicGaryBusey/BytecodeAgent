

export function l $mulhi_u64(l a, l b) {
  @start
  %al = andl %a 0xFFFFFFFF
  %ah = shrl %a 32
  %bl = andl %b 0xFFFFFFFF
  %bh = shrl %b 32
  %p0 = mul %al %bl
  %p1 = mul %ah %bl
  %p2 = mul %al %bh
  %p3 = mul %ah %bh
  %p0_hi = shrl %p0 32
  %p1_lo = andl %p1 0xFFFFFFFF
  %p2_lo = andl %p2 0xFFFFFFFF
  %p1_hi = shrl %p1 32
  %p2_hi = shrl %p2 32
  %sum_lo = addl %p1_lo %p2_lo
  %sum_lo2 = addl %sum_lo %p0_hi
  %carry_lo = shrl %sum_lo2 32
  %mid_high = addl %p1_hi %p2_hi
  %mid_high2 = addl %mid_high %carry_lo
  %high = addl %p3 %mid_high2
  retl %high
}
