

export function l $mulhi_u64(l a, l b) {
  @start
  %lo_a = andl a 0xFFFFFFFF
  %hi_a = shrl a 32
  %lo_b = andl b 0xFFFFFFFF
  %hi_b = shrl b 32
  %t1 = mul %lo_a %lo_b
  %t2 = mul %hi_a %lo_b
  %t3 = mul %lo_a %hi_b
  %t4 = mul %hi_a %hi_b
  %shifted_t1 = shrl %t1 32
  %sum = addl %t2 %t3
  %sum2 = addl %sum %shifted_t1
  %shifted_sum2 = shrl %sum2 32
  %high = addl %t4 %shifted_sum2
  retl %high
}
