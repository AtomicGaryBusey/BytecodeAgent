

export function l $binary_search(l a, l n, l key)
@start:
    %lo = 0
    %hi = subl n, 1
    jmp @loop
@loop:
    csltl %hi, %lo
    jnz @exit
    %sum = addl %lo, %hi
    %mid = cdivl %sum, 2
    %val = loadl a, %mid
    ceql %val, key
    jnz @found
    csltl %val, key
    jnz @lo_inc
    %hi = subl %hi, 1
    jmp @loop
@lo_inc:
    %lo = addl %lo, 1
    jmp @loop
@found:
    ret %mid
@exit:
    ret -1
