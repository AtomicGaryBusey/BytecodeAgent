

export function l $binary_search(l %a, l %n, l %key)
{
@start:
    %lo = 0
    %one = 1
    %neg1 = subl %lo, %one
    %two = 2
    %hi = subl %n, %one
    jmp @loop
@loop:
    csltl %hi, %lo
    jnz @exit
    %sum = addl %lo, %hi
    %mid = cdivl %sum, %two
    %val = loadl %a, %mid
    ceql %val, %key
    jnz @found
    csltl %val, %key
    jnz @lo_inc
    %hi = subl %mid, %one
    jmp @loop
@lo_inc:
    %lo = addl %mid, %one
    jmp @loop
@found:
    ret %mid
@exit:
    ret %neg1
}
