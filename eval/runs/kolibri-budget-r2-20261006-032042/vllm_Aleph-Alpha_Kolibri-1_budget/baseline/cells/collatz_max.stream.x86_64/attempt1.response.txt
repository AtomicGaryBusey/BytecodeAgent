

collatz_max:
mov r10, rdi
xor rax, rax
.Louter_loop:
test r10, r10
jle .Louter_done
mov rdx, r10
xor rsi, rsi
.Linner_loop:
cmp rdx, 1
je .Linner_done
test rdx, 1
jz .Linner_even
mov r8, rdx
imul r8, 3
add r8, 1
mov rdx, r8
jmp .Linner_inc
.Linner_even:
shr rdx, 1
.Linner_inc:
inc rsi
jmp .Linner_loop
.Linner_done:
cmp rsi, rax
jle .Ldecrement
mov rax, rsi
.Ldecrement:
dec r10
jmp .Louter_loop
.Louter_done:
cld
ret
