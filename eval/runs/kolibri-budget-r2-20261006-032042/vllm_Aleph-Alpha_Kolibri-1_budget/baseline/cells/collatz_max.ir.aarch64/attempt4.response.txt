

export function collatz_max(l) l {
    @start:
        movl $1, %k
        movl $0, %max
        jmp @outer_loop
    @outer_loop:
        csgtl %done, %k, %arg0
        jnz %done, @done
        movl %k, %val
        movl $0, %steps
        jmp @inner_loop
    @inner_loop:
        ceql %val, $1, %done_val
        jnz %done_val, @inner_done
        andl $1, %val, %lsb
        ceql %lsb, $0, %iszero
        jnz %iszero, @even_branch
        imull $3, %val, %val
        addl $1, %val, %val
        jmp @inner_next
    @even_branch:
        shrl $1, %val, %val
        jmp @inner_next
    @inner_next:
        addl $1, %steps, %steps
        jmp @inner_loop
    @inner_done:
        csltl %islt, %max, %steps
        jnz %islt, @update_max
        jmp @increment_k
    @update_max:
        movl %steps, %max
        jmp @increment_k
    @increment_k:
        addl $1, %k, %k
        jmp @outer_loop
    @done:
        movl %max, %x0
        ret
}
