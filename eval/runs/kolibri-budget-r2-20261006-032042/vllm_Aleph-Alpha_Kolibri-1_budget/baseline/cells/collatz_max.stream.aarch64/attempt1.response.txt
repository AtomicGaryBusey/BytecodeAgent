

collatz_max:
    mov x5, #1
    mov x4, #0
.Louter:
    mov x1, x5
    mov x2, #0
.Linner:
    cmp x1, #1
    b.eq .Linner_end
    tbnz x1, #0, .Lodd
    lsr x1, x1, #1
    b .Linner_inc
.Lodd:
    lsl x3, x1, #1
    add x3, x3, x1
    add x3, x3, #1
    mov x1, x3
.Linner_inc:
    add x2, x2, #1
    b .Linner
.Linner_end:
    cmp x2, x4
    b.le .Lincrement
    mov x4, x2
.Lincrement:
    add x5, x5, #1
    cmp x5, x0
    b.le .Louter
    mov x0, x4
    ret
