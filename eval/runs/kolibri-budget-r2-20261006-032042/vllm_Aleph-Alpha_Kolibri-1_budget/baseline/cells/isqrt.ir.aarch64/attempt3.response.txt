

export function l $isqrt(l n)
@init
%lo = 0
%hi = 4294967295
@loop_start
%less = csltl %hi %lo
jnz %less @exit
%sum = addl %lo %hi
%mid = divl %sum 2
%zero = ceql %mid 0
jnz %zero @mid_zero
%quot = divl %n %mid
%lt = csltl %quot %mid
jnz %lt @set_hi
%lo = addl %mid 1
jmp @loop_start
@mid_zero
%lo = addl %mid 1
jmp @loop_start
@set_hi
%hi = subl %mid 1
jmp @loop_start
@exit
%hi
