

export function l $isqrt(l)
@init
%lo = 0
%hi = 4294967295
@loop_start
%less = csltl %lo %hi
jnz %less @loop_body
jmp @exit
@loop_body
%sum = addl %lo %hi
%sum1 = addl %sum 1
%mid = divl %sum1 2
%quot = divl %n %mid
%lt = csltl %quot %mid
jnz %lt @set_hi
%lo = %mid
jmp @loop_start
@set_hi
%hi = subl %mid 1
jmp @loop_start
@exit
%lo
