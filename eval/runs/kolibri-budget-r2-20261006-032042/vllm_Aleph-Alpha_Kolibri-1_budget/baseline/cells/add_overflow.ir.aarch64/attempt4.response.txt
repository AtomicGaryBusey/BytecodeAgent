

export function l $add_overflow(l %a, l %b, l %out)
@start
%sum = addl %a %b
storel %sum %out
%neg_a = csltl %a 0
%neg_b = csltl %b 0
%neg_sum = csltl %sum 0
%overflow = 0l
jnz %neg_a, @a_neg
@a_nonneg
jnz %neg_b, @b_neg
@both_nonneg
jnz %neg_sum, @overflow_pos
jmp @done
@overflow_pos
%overflow = 1l
jmp @done
@b_neg
jmp @done
@a_neg
jnz %neg_b, @both_neg
jmp @done
@both_neg
jnz %neg_sum, @done
%overflow = 1l
jmp @done
@done
retl %overflow
