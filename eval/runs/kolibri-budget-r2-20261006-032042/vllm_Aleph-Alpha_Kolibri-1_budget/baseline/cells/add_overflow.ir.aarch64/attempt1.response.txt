

export function l $add_overflow(l a, l b, l out)
@start
%sum = addl a b
storel %sum, out
%overflow = 0
%a_lt0 = csltl a 0
%b_lt0 = csltl b 0
%sum_lt0 = csltl %sum 0
jnz %a_lt0, @a_negative
@a_nonneg
jnz %b_lt0, @a_nonneg_b_neg
@a_nonneg_b_nonneg
jnz %sum_lt0, @set_overflow_pos
jmp @done
@set_overflow_pos
%overflow = 1
jmp @done
@a_nonneg_b_neg
jmp @done
@a_negative
jnz %b_lt0, @a_negative_b_nonneg
@a_negative_b_neg
jnz %sum_lt0, @a_negative_b_neg_sum_neg
%overflow = 1
jmp @done
@a_negative_b_neg_sum_neg
jmp @done
@a_negative_b_nonneg
jmp @done
@done
retl %overflow
