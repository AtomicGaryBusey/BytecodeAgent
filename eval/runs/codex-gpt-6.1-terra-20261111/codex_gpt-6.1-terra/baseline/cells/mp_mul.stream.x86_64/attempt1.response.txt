push r13
push r14

.Lzero:

jae .Lzero_done
mov qword ptr [r8 + r11*8], 0

xor r11d, r11d
test rcx, rcx
jz .Lcount
.Louter:
cmp r11, rsi
jae .Lcount
mov r12, qword ptr [rdi + r11*8]
lea r13, [r8 + r11*8]
xor r14d, r14d
xor ebx, ebx

adc rdx, 0

mov rax, r10
.Ltrim:

jz .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0

pop r12
