export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inc
	%min =l phi @start 9223372036854775807, @next %newmin
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%addr =l add %a, %offset
	%value =l loadl %addr
	%less =w csltl %value, %min
	jnz %less, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%newmin =l phi @update %value, @keep %min
	%inc =l add %i, 1
	jmp @loop
@done
	ret %min
}
