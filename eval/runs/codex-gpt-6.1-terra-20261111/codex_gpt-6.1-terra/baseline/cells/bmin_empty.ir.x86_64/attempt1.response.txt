export function l $bmin_empty(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%min =l phi @start 9223372036854775807, @next %min1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%less =w csltl %value, %min
	jnz %less, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%min1 =l phi @update %value, @keep %min
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %min
}
