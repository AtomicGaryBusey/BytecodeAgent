export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer

@outer
	%i =l phi @start 0, @repeat_write %repeat_next, @literal_done %lit_end
	%used =l phi @start 0, @repeat_write %repeat_used, @literal_done %lit_used
	%finished =w csgel %i, %n
	jnz %finished, @success, @run_init

@run_init
	%run_addr =l add %src, %i
	%run_byte =w loadub %run_addr
	jmp @run_loop

@run_loop
	%r =l phi @run_init 1, @run_extend %r_next
	%r_full =w cegel %r, 128
	jnz %r_full, @run_done, @run_bounds

@run_bounds
	%run_pos =l add %i, %r
	%run_end =w csgel %run_pos, %n
	jnz %run_end, @run_done, @run_check

@run_check
	%run_ptr =l add %src, %run_pos
	%run_value =w loadub %run_ptr
	%run_same =w ceqw %run_value, %run_byte
	jnz %run_same, @run_extend, @run_done

@run_extend
	%r_next =l add %r, 1
	jmp @run_loop

@run_done
	%is_repeat =w csgel %r, 3
	jnz %is_repeat, @repeat_check, @literal_init

@repeat_check
	%repeat_used =l add %used, 2
	%repeat_fits =w cslel %repeat_used, %cap
	jnz %repeat_fits, @repeat_write, @failure

@repeat_write
	%repeat_header =l sub 257, %r
	%repeat_out =l add %out, %used
	storeb %repeat_header, %repeat_out
	%repeat_byte_out =l add %repeat_out, 1
	storeb %run_byte, %repeat_byte_out
	%repeat_next =l add %i, %r
	jmp @outer

@literal_init
	jmp @literal_scan

@literal_scan
	%j =l phi @literal_init %i, @literal_advance %j_next
	%lit_len =l sub %j, %i
	%lit_full =w cegel %lit_len, 128
	jnz %lit_full, @literal_end, @literal_bounds

@literal_bounds
	%lit_at_end =w csgel %j, %n
	jnz %lit_at_end, @literal_end, @literal_lookahead

@literal_lookahead
	%remaining =l sub %n, %j
	%has_three =w csgel %remaining, 3
	jnz %has_three, @literal_check_three, @literal_advance

@literal_check_three
	%lit_ptr =l add %src, %j
	%lit_b0 =w loadub %lit_ptr
	%lit_ptr1 =l add %lit_ptr, 1
	%lit_b1 =w loadub %lit_ptr1
	%lit_eq1 =w ceqw %lit_b0, %lit_b1
	jnz %lit_eq1, @literal_check_last, @literal_advance

@literal_check_last
	%lit_ptr2 =l add %lit_ptr, 2
	%lit_b2 =w loadub %lit_ptr2
	%lit_eq2 =w ceqw %lit_b0, %lit_b2
	jnz %lit_eq2, @literal_end, @literal_advance

@literal_advance
	%j_next =l add %j, 1
	jmp @literal_scan

@literal_end
	%lit_end =l copy %j
	%packet_size =l add %lit_len, 1
	%lit_used =l add %used, %packet_size
	%lit_fits =w cslel %lit_used, %cap
	jnz %lit_fits, @literal_write, @failure

@literal_write
	%lit_header =l sub %lit_len, 1
	%lit_out =l add %out, %used
	storeb %lit_header, %lit_out
	%lit_data =l add %lit_out, 1
	jmp @literal_copy

@literal_copy
	%k =l phi @literal_write 0, @literal_copy_byte %k_next
	%copy_done =w cegel %k, %lit_len
	jnz %copy_done, @literal_done, @literal_copy_byte

@literal_copy_byte
	%copy_pos =l add %i, %k
	%copy_src =l add %src, %copy_pos
	%copy_byte =w loadub %copy_src
	%copy_out =l add %lit_data, %k
	storeb %copy_byte, %copy_out
	%k_next =l add %k, 1
	jmp @literal_copy

@literal_done
	jmp @outer

@success
	ret %used

@failure
	ret -1
}
