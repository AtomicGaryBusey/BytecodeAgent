export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @next %npc
	%sp =l phi @start 0, @next %nsp
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch
@fetch
	%off =l mul %pc, 8
	%addr =l add %code, %off
	%op =l loadl %addr
	%halt =w ceql %op, 0
	jnz %halt, @finish, @dispatch1
@dispatch1
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @dispatch2
@dispatch2
	%is_add =w ceql %op, 2
	%is_sub =w ceql %op, 3
	%is_mul =w ceql %op, 4
	%is_lt =w ceql %op, 9
	%arith1 =w or %is_add, %is_sub
	%arith2 =w or %is_mul, %is_lt
	%arith =w or %arith1, %arith2
	jnz %arith, @binary, @dispatch3
@dispatch3
	%is_dup =w ceql %op, 5
	jnz %is_dup, @dup, @dispatch4
@dispatch4
	%is_swap =w ceql %op, 6
	jnz %is_swap, @swap, @dispatch5
@dispatch5
	%is_drop =w ceql %op, 7
	jnz %is_drop, @drop, @dispatch6
@dispatch6
	%is_over =w ceql %op, 8
	jnz %is_over, @over, @dispatch7
@dispatch7
	%is_jz =w ceql %op, 10
	jnz %is_jz, @jz, @jmpop
@push
	%operand_addr =l add %addr, 8
	%operand =l loadl %operand_addr
	%push_off =l mul %sp, 8
	%push_addr =l add %stack, %push_off
	storel %operand, %push_addr
	%push_sp =l add %sp, 1
	%push_pc =l add %pc, 2
	jmp @next
@binary
	%bi =l sub %sp, 1
	%ai =l sub %sp, 2
	%boff =l mul %bi, 8
	%aoff =l mul %ai, 8
	%baddr =l add %stack, %boff
	%aaddr =l add %stack, %aoff
	%b =l loadl %baddr
	%a =l loadl %aaddr
	jnz %is_add, @add, @binary2
@binary2
	jnz %is_sub, @sub, @binary3
@binary3
	jnz %is_mul, @mul, @lt
@add
	%addval =l add %a, %b
	jmp @binary_done
@sub
	%subval =l sub %a, %b
	jmp @binary_done
@mul
	%mulval =l mul %a, %b
	jmp @binary_done
@lt
	%ltw =w csltl %a, %b
	%ltval =l extuw %ltw
	jmp @binary_done
@binary_done
	%result =l phi @add %addval, @sub %subval, @mul %mulval, @lt %ltval
	storel %result, %aaddr
	%binary_pc =l add %pc, 1
	jmp @next
@dup
	%dup_i =l sub %sp, 1
	%dup_off =l mul %dup_i, 8
	%dup_src =l add %stack, %dup_off
	%dup_val =l loadl %dup_src
	%dup_dst_off =l mul %sp, 8
	%dup_dst =l add %stack, %dup_dst_off
	storel %dup_val, %dup_dst
	%dup_sp =l add %sp, 1
	%dup_pc =l add %pc, 1
	jmp @next
@swap
	%swap_i =l sub %sp, 1
	%swap_j =l sub %sp, 2
	%swap_ioff =l mul %swap_i, 8
	%swap_joff =l mul %swap_j, 8
	%swap_ip =l add %stack, %swap_ioff
	%swap_jp =l add %stack, %swap_joff
	%swap_iv =l loadl %swap_ip
	%swap_jv =l loadl %swap_jp
	storel %swap_iv, %swap_jp
	storel %swap_jv, %swap_ip
	%swap_pc =l add %pc, 1
	jmp @next
@drop
	%drop_sp =l sub %sp, 1
	%drop_pc =l add %pc, 1
	jmp @next
@over
	%over_i =l sub %sp, 2
	%over_off =l mul %over_i, 8
	%over_src =l add %stack, %over_off
	%over_val =l loadl %over_src
	%over_dst_off =l mul %sp, 8
	%over_dst =l add %stack, %over_dst_off
	storel %over_val, %over_dst
	%over_sp =l add %sp, 1
	%over_pc =l add %pc, 1
	jmp @next
@jz
	%jz_sp =l sub %sp, 1
	%jz_off =l mul %jz_sp, 8
	%jz_addr =l add %stack, %jz_off
	%jz_val =l loadl %jz_addr
	%jz_operand_addr =l add %addr, 8
	%jz_target =l loadl %jz_operand_addr
	%jz_zero =w ceql %jz_val, 0
	jnz %jz_zero, @jz_taken, @jz_fall
@jz_taken
	jmp @next
@jz_fall
	%jz_fall_pc =l add %pc, 2
	jmp @next
@jmpop
	%jmp_addr =l add %addr, 8
	%jmp_target =l loadl %jmp_addr
	jmp @next
@next
	%npc =l phi @push %push_pc, @binary_done %binary_pc, @dup %dup_pc, @swap %swap_pc, @drop %drop_pc, @over %over_pc, @jz_taken %jz_target, @jz_fall %jz_fall_pc, @jmpop %jmp_target
	%nsp =l phi @push %push_sp, @binary_done %bi, @dup %dup_sp, @swap %sp, @drop %drop_sp, @over %over_sp, @jz_taken %jz_sp, @jz_fall %jz_sp, @jmpop %sp
	jmp @loop
@finish
	%empty =w ceql %sp, 0
	jnz %empty, @empty_result, @top_result
@empty_result
	jmp @clear_start
@top_result
	%top_i =l sub %sp, 1
	%top_off =l mul %top_i, 8
	%top_addr =l add %stack, %top_off
	%top_val =l loadl %top_addr
	jmp @clear_start
@clear_start
	%ret =l phi @empty_result 0, @top_result %top_val
	jmp @clear_loop
@clear_loop
	%i =l phi @clear_start %sp, @clear_body %i_next
	%clear_done =w ceql %i, %cap
	jnz %clear_done, @return, @clear_body
@clear_body
	%clear_off =l mul %i, 8
	%clear_addr =l add %stack, %clear_off
	storel 0, %clear_addr
	%i_next =l add %i, 1
	jmp @clear_loop
@return
	ret %ret
}
