export function l $fib(l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %nexti
    %a =l phi @start 0, @body %b
    %b =l phi @start 1, @body %next
    %done =w ceql %i, %n
    jnz %done, @end, @body
@body
    %next =l add %a, %b
    %nexti =l add %i, 1
    jmp @loop
@end
    ret %a
}
