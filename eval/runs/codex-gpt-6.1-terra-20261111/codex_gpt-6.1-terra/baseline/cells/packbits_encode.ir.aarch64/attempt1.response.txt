export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%used =l copy 0
	jmp @outer
@outer
	%more =w csltl %i, %n
	jnz %more, @measure_start, @done
@measure_start
	%ip =l add %src, %i
	%byte =w loadub %ip
	%run =l copy 1
	jmp @measure_check
@measure_check
	%runroom =w csltl %run, 128
	jnz %runroom, @measure_bounds, @measure_done
@measure_bounds
	%pos =l add %i, %run
	%inbounds =w csltl %pos, %n
	jnz %inbounds, @measure_load, @measure_done
@measure_load
	%rp =l add %src, %pos
	%rb =w loadub %rp
	%same =w ceqw %rb, %byte
	jnz %same, @measure_inc, @measure_done
@measure_inc
	%run =l add %run, 1
	jmp @measure_check
@measure_done
	%repeat =w csgel %run, 3
	jnz %repeat, @repeat_check, @literal_start
@repeat_check
	%repeatend =l add %used, 2
	%repeatfits =w cslel %repeatend, %cap
	jnz %repeatfits, @repeat_write, @fail
@repeat_write
	%header =l sub 257, %run
	%oh =l add %out, %used
	storeb %header, %oh
	%ob =l add %oh, 1
	storeb %byte, %ob
	%used =l copy %repeatend
	%i =l add %i, %run
	jmp @outer
@literal_start
	%j =l copy %i
	jmp @literal_check
@literal_check
	%litlen =l sub %j, %i
	%litroom =w csltl %litlen, 128
	jnz %litroom, @literal_bounds, @literal_emit
@literal_bounds
	%litin =w csltl %j, %n
	jnz %litin, @literal_probe, @literal_emit
@literal_probe
	%j1 =l add %j, 1
	%j2 =l add %j, 2
	%hasthree =w csltl %j2, %n
	jnz %hasthree, @literal_compare, @literal_advance
@literal_compare
	%p0 =l add %src, %j
	%p1 =l add %src, %j1
	%p2 =l add %src, %j2
	%b0 =w loadub %p0
	%b1 =w loadub %p1
	%b2 =w loadub %p2
	%eq01 =w ceqw %b0, %b1
	%eq12 =w ceqw %b1, %b2
	%triple =w and %eq01, %eq12
	jnz %triple, @literal_emit, @literal_advance
@literal_advance
	%j =l add %j, 1
	jmp @literal_check
@literal_emit
	%length =l sub %j, %i
	%packetsize =l add %length, 1
	%litend =l add %used, %packetsize
	%litfits =w cslel %litend, %cap
	jnz %litfits, @literal_write, @fail
@literal_write
	%lh =l sub %length, 1
	%lhp =l add %out, %used
	storeb %lh, %lhp
	%copyidx =l copy 0
	jmp @copy_check
@copy_check
	%copyleft =w csltl %copyidx, %length
	jnz %copyleft, @copy_byte, @copy_done
@copy_byte
	%si =l add %i, %copyidx
	%sptr =l add %src, %si
	%cb =w loadub %sptr
	%di =l add %used, 1
	%di2 =l add %di, %copyidx
	%dptr =l add %out, %di2
	storeb %cb, %dptr
	%copyidx =l add %copyidx, 1
	jmp @copy_check
@copy_done
	%used =l copy %litend
	%i =l copy %j
	jmp @outer
@done
	ret %used
@fail
	ret -1
}
