export function l $fill_down(l %a, l %n, l %start) {
@entry
	%empty = ceql %n, 0
	jnz %empty, @done, @loop
@loop
	%i =l phi @entry 0, @loop %next
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%v =l sub %start, %i
	storel %v, %p
	%next =l add %i, 1
	%more =w csltl %next, %n
	jnz %more, @loop, @done
@done
	ret %n
}
