export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %next, @upper %lo
	%hi =l phi @start %n, @lower %hi, @upper %mid
	%has =w csltl %lo, %hi
	jnz %has, @search, @missing
@search
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @lower, @upper
@lower
	%next =l add %mid, 1
	jmp @loop
@upper
	jmp @loop
@found
	ret %mid
@missing
	ret -1
}
