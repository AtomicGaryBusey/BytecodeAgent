export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop

@loop
	%i =l phi @start 1, @body %next
	%min =l phi @start %y0, @body %newmin
	%max =l phi @start %y0, @body %newmax
	%sum =l phi @start %y0, @body %newsum
	%count =l phi @start %odd0, @body %newcount
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	jnz %ismin, @setmin, @keepmin

@setmin
	jmp @minend

@keepmin
	jmp @minend

@minend
	%newmin =l phi @setmin %y, @keepmin %min
	%ismax =w csgtl %y, %max
	jnz %ismax, @setmax, @keepmax

@setmax
	jmp @maxend

@keepmax
	jmp @maxend

@maxend
	%newmax =l phi @setmax %y, @keepmax %max
	%newsum =l add %sum, %y
	%odd =l and %y, 1
	%newcount =l add %count, %odd
	%next =l add %i, 1
	jmp @bodyend

@bodyend
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
