export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @direction
@direction
	%forward =w cultl %dst, %src
	jnz %forward, @finit, @binit
@finit
	jmp @floop
@floop
	%i =l phi @finit 0, @floop %inext
	%offset =l mul %i, 8
	%sp =l add %src, %offset
	%dp =l add %dst, %offset
	%value =l loadl %sp
	storel %value, %dp
	%inext =l add %i, 1
	%more =w cultl %inext, %n
	jnz %more, @floop, @done
@binit
	jmp @bloop
@bloop
	%remaining =l phi @binit %n, @bloop %index
	%index =l sub %remaining, 1
	%boffset =l mul %index, 8
	%bsp =l add %src, %boffset
	%bdp =l add %dst, %boffset
	%bvalue =l loadl %bsp
	storel %bvalue, %bdp
	%bmore =w cnel %index, 0
	jnz %bmore, @bloop, @done
@done
	ret 0
}
