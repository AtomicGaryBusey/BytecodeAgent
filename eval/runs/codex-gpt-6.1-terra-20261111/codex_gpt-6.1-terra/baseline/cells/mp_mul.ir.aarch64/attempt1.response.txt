export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero

@zero
	%zi =l phi @start 0, @zero_body %znext
	%ztest =w cultl %zi, %n
	jnz %ztest, @zero_body, @zero_done

@zero_body
	%zoff =l shl %zi, 3
	%zp =l add %r, %zoff
	storel 0, %zp
	%znext =l add %zi, 1
	jmp @zero

@zero_done
	%az =w ceql %na, 0
	%bz =w ceql %nb, 0
	%empty =w or %az, %bz
	jnz %empty, @scan_start, @outer

@outer
	%i =l phi @zero_done 0, @outer_next %inext
	%aoff =l shl %i, 3
	%ap =l add %a, %aoff
	%av =l loadl %ap
	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	jmp @inner

@inner
	%j =l phi @outer 0, @inner %jnext
	%carry =l phi @outer 0, @inner %hi2
	%boff =l shl %j, 3
	%bp =l add %b, %boff
	%bv =l loadl %bp
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%p00 =l mul %a0, %b0
	%p10 =l mul %a1, %b0
	%p01 =l mul %a0, %b1
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%u =l add %p10, %p00hi
	%ulo =l and %u, 4294967295
	%uhi =l shr %u, 32
	%v =l add %p01, %ulo
	%vhi =l shr %v, 32
	%vlo =l shl %v, 32
	%p00lo =l and %p00, 4294967295
	%lo =l or %vlo, %p00lo
	%ht =l add %p11, %uhi
	%hi =l add %ht, %vhi
	%idx =l add %i, %j
	%roff =l shl %idx, 3
	%rp =l add %r, %roff
	%old =l loadl %rp
	%s1 =l add %lo, %old
	%c1 =w cultl %s1, %lo
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	storel %s2, %rp
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%hi1 =l add %hi, %c1l
	%hi2 =l add %hi1, %c2l
	%jnext =l add %j, 1
	%morej =w cultl %jnext, %nb
	jnz %morej, @inner, @inner_done

@inner_done
	%lastidx =l add %i, %nb
	%lastoff =l shl %lastidx, 3
	%lastp =l add %r, %lastoff
	storel %hi2, %lastp
	jmp @outer_next

@outer_next
	%inext =l add %i, 1
	%morei =w cultl %inext, %na
	jnz %morei, @outer, @scan_start

@scan_start
	jmp @scan

@scan
	%k =l phi @scan_start %n, @scan_zero %prev
	%has =w cnel %k, 0
	jnz %has, @scan_load, @done_zero

@scan_load
	%prev =l sub %k, 1
	%soff =l shl %prev, 3
	%sptr =l add %r, %soff
	%val =l loadl %sptr
	%nonzero =w cnel %val, 0
	jnz %nonzero, @done, @scan_zero

@scan_zero
	jmp @scan

@done
	ret %k

@done_zero
	ret 0
}
