push rbp
push r12
push r13
push r14
push r15

.Linit_row:
xor r13d, r13d
.Linit_col:
xor eax, eax
cmp r12, r13
sete al
mov r10, r12
imul r10, rsi
add r10, r13
mov qword ptr [rcx + r10*8], rax

inc r12

jb .Linit_row

xor r12d, r12d
.Lrow:
mov rbx, r12
imul rbx, rsi
xor r13d, r13d
.Lcol:
xor r15d, r15d
xor r14d, r14d
.Linner:
mov rax, rbx
add rax, r14
mov r11, qword ptr [rcx + rax*8]

imul rax, rsi
add rax, r13

jb .Lcol

jb .Lrow

mov qword ptr [rcx + r10*8], rax
inc r10
cmp r10, r9

xor r10d, r10d
