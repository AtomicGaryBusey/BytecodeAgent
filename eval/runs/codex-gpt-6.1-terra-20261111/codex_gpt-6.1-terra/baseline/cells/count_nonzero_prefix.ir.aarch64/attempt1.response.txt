export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inc
	%inbounds =w csltl %i, %n
	jnz %inbounds, @load, @done
@load
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%nonzero =w cnel %value, 0
	jnz %nonzero, @next, @done
@next
	%inc =l add %i, 1
	jmp @loop
@done
	ret %i
}
