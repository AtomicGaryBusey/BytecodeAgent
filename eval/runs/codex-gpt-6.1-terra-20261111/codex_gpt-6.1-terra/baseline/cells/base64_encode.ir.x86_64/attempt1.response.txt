export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @advance %inext
	%o =l phi @start 0, @advance %onext
	%more =w csltl %i, %n
	jnz %more, @load0, @done
@load0
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%i1 =l add %i, 1
	%has1 =w csltl %i1, %n
	jnz %has1, @load1, @no1
@load1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	jmp @check2
@no1
	jmp @check2
@check2
	%x1 =l phi @load1 %b1, @no1 0
	%i2 =l add %i, 2
	%has2 =w csltl %i2, %n
	jnz %has2, @load2, @no2
@load2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	jmp @prepare
@no2
	jmp @prepare
@prepare
	%x2 =l phi @load2 %b2, @no2 0
	%v0 =l shr %b0, 2
	%t1 =l and %b0, 3
	%t2 =l shl %t1, 4
	%t3 =l shr %x1, 4
	%v1 =l or %t2, %t3
	%t4 =l and %x1, 15
	%t5 =l shl %t4, 2
	%t6 =l shr %x2, 6
	%v2 =l or %t5, %t6
	%v3 =l and %x2, 63
	jmp @char
@char
	%k =l phi @prepare 0, @char_next %knext
	%v =l phi @prepare %v0, @char_next %vnext
	%upper =w csltl %v, 26
	jnz %upper, @map_upper, @check_lower
@map_upper
	%cu =l add %v, 65
	jmp @mapped
@check_lower
	%lower =w csltl %v, 52
	jnz %lower, @map_lower, @check_digit
@map_lower
	%cl =l add %v, 71
	jmp @mapped
@check_digit
	%digit =w csltl %v, 62
	jnz %digit, @map_digit, @check_plus
@map_digit
	%cd =l sub %v, 4
	jmp @mapped
@check_plus
	%plus =w ceql %v, 62
	jnz %plus, @map_plus, @map_slash
@map_plus
	jmp @mapped
@map_slash
	jmp @mapped
@mapped
	%c =l phi @map_upper %cu, @map_lower %cl, @map_digit %cd, @map_plus 43, @map_slash 47
	%is2 =w ceql %k, 2
	%missing1 =w ceqw %has1, 0
	%pad2 =w and %is2, %missing1
	%is3 =w ceql %k, 3
	%missing2 =w ceqw %has2, 0
	%pad3 =w and %is3, %missing2
	%pad =w or %pad2, %pad3
	jnz %pad, @write_pad, @write_char
@write_pad
	jmp @write
@write_char
	jmp @write
@write
	%ch =l phi @write_pad 61, @write_char %c
	%pos =l add %o, %k
	%dst =l add %out, %pos
	storeb %ch, %dst
	%knext =l add %k, 1
	%finished =w ceql %knext, 4
	jnz %finished, @advance, @char_next
@char_next
	%next1 =w ceql %knext, 1
	jnz %next1, @choose1, @choose_later
@choose1
	jmp @chosen
@choose_later
	%next2 =w ceql %knext, 2
	jnz %next2, @choose2, @choose3
@choose2
	jmp @chosen
@choose3
	jmp @chosen
@chosen
	%vnext =l phi @choose1 %v1, @choose2 %v2, @choose3 %v3
	jmp @char
@advance
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @group
@done
	ret %o
}
