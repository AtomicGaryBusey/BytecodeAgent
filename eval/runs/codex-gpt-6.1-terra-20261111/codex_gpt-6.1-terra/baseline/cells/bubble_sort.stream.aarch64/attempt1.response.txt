bubble_sort:
cmp x1, #1
b.le .Ldone
sub x2, x1, #1
.Louter:
mov x3, #0
.Linner:
ldr x4, [x0, x3, lsl #3]
add x5, x3, #1
ldr x6, [x0, x5, lsl #3]
cmp x4, x6
b.le .Lnext
str x6, [x0, x3, lsl #3]
str x4, [x0, x5, lsl #3]
.Lnext:
mov x3, x5
cmp x3, x2
b.lt .Linner
subs x2, x2, #1
b.ne .Louter
.Ldone:
mov x0, #0
ret
