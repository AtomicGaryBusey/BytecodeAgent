export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @setup

@setup
	%last =l sub %n, 1
	%pivot_off =l mul %last, 16
	%pivot_ptr =l add %a, %pivot_off
	%pivot_val_ptr =l add %pivot_ptr, 8
	%pivot_key =l loadl %pivot_ptr
	%pivot_val =l loadl %pivot_val_ptr
	jmp @scan

@scan
	%i =l phi @setup 0, @next %i_next
	%j =l phi @setup 0, @next %j_next
	%more =w csltl %j, %last
	jnz %more, @compare, @partition

@compare
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jvp =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jvp
	%key_less =w csltl %jk, %pivot_key
	%key_equal =w ceql %jk, %pivot_key
	%val_less =w csltl %jv, %pivot_val
	%equal_less =w and %key_equal, %val_less
	%less =w or %key_less, %equal_less
	jnz %less, @swap, @skip

@swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ivp =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%inc_i =l add %i, 1
	jmp @next

@skip
	jmp @next

@next
	%i_next =l phi @swap %inc_i, @skip %i
	%j_next =l add %j, 1
	jmp @scan

@partition
	%final_off =l mul %i, 16
	%final_ptr =l add %a, %final_off
	%final_val_ptr =l add %final_ptr, 8
	%final_key =l loadl %final_ptr
	%final_val =l loadl %final_val_ptr
	storel %pivot_key, %final_ptr
	storel %pivot_val, %final_val_ptr
	storel %final_key, %pivot_ptr
	storel %final_val, %pivot_val_ptr
	%left_result =l call $qsort_pairs(l %a, l %i)
	%right_ptr =l add %final_ptr, 16
	%right_n =l sub %last, %i
	%right_result =l call $qsort_pairs(l %right_ptr, l %right_n)
	jmp @done

@done
	ret 0
}
