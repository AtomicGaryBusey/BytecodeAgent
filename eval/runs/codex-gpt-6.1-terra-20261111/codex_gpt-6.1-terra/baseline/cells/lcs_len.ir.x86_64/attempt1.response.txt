export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%k =l phi @start 0, @initbody %knext
	%initcond =w cslel %k, %n
	jnz %initcond, @initbody, @outer
@initbody
	%koff =l mul %k, 8
	%kp =l add %row, %koff
	storel 0, %kp
	%knext =l add %k, 1
	jmp @init
@outer
	%i =l phi @init 0, @outerend %inext
	%outercond =w csltl %i, %m
	jnz %outercond, @outerbody, @done
@outerbody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%j =l phi @outerbody 1, @innerend %jnext
	%diag =l phi @outerbody 0, @innerend %up
	%innercond =w cslel %j, %n
	jnz %innercond, @innerbody, @outerend
@innerbody
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%off =l mul %j, 8
	%rp =l add %row, %off
	%up =l loadl %rp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch
@match
	%matched =l add %diag, 1
	jmp @innerend
@mismatch
	%leftp =l sub %rp, 8
	%left =l loadl %leftp
	%greater =w csgtl %up, %left
	jnz %greater, @useup, @useleft
@useup
	jmp @innerend
@useleft
	jmp @innerend
@innerend
	%value =l phi @match %matched, @useup %up, @useleft %left
	storel %value, %rp
	%jnext =l add %j, 1
	jmp @inner
@outerend
	%inext =l add %i, 1
	jmp @outer
@done
	%resultoff =l mul %n, 8
	%resultp =l add %row, %resultoff
	%result =l loadl %resultp
	ret %result
}
