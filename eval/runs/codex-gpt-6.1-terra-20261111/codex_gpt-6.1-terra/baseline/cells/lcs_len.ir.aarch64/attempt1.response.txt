export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init

@init
	%k =l phi @start 0, @init_body %knext
	%initcond =w cslel %k, %n
	jnz %initcond, @init_body, @outer

@init_body
	%koff =l mul %k, 8
	%kp =l add %row, %koff
	storel 0, %kp
	%knext =l add %k, 1
	jmp @init

@outer
	%i =l phi @init 0, @outer_next %inext
	%outercond =w csltl %i, %m
	jnz %outercond, @outer_body, @done

@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outer_body 1, @update %jnext
	%diag =l phi @outer_body 0, @update %up
	%innercond =w cslel %j, %n
	jnz %innercond, @inner_body, @outer_next

@inner_body
	%joff =l mul %j, 8
	%jp =l add %row, %joff
	%up =l loadl %jp
	%ym1 =l sub %j, 1
	%yp =l add %y, %ym1
	%yb =w loadub %yp
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @nomatch

@match
	%matched =l add %diag, 1
	jmp @update

@nomatch
	%leftp =l sub %jp, 8
	%left =l loadl %leftp
	%upgreater =w csgtl %up, %left
	jnz %upgreater, @use_up, @use_left

@use_up
	jmp @update

@use_left
	jmp @update

@update
	%value =l phi @match %matched, @use_up %up, @use_left %left
	storel %value, %jp
	%jnext =l add %j, 1
	jmp @inner

@outer_next
	%inext =l add %i, 1
	jmp @outer

@done
	%noff =l mul %n, 8
	%np =l add %row, %noff
	%result =l loadl %np
	ret %result
}
