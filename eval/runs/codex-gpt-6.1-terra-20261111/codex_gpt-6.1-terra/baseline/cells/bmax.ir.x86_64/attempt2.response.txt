export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @advance %inext
	%max =l phi @start %first, @advance %chosen
	%more =w csltl %i, %n
	jnz %more, @next, @done
@next
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w csgtl %value, %max
	jnz %greater, @update, @keep
@update
	jmp @advance
@keep
	jmp @advance
@advance
	%chosen =l phi @update %value, @keep %max
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
