export function l $crc32(l %p, l %n) {
@start
	jmp @byte
@byte
	%ptr =l phi @start %p, @next %ptrnext
	%left =l phi @start %n, @next %leftnext
	%crc =w phi @start 4294967295, @next %bitcrc
	%more =w cnel %left, 0
	jnz %more, @load, @done
@load
	%b =w loadub %ptr
	%initial =w xor %crc, %b
	jmp @bit
@bit
	%c =w phi @load %initial, @bit %bitcrc
	%i =w phi @load 0, @bit %inext
	%low =w and %c, 1
	%mask =w sub 0, %low
	%poly =w and %mask, 3988292384
	%shift =w shr %c, 1
	%bitcrc =w xor %shift, %poly
	%inext =w add %i, 1
	%again =w csltw %inext, 8
	jnz %again, @bit, @next
@next
	%ptrnext =l add %ptr, 1
	%leftnext =l sub %left, 1
	jmp @byte
@done
	%final =w xor %crc, 4294967295
	%result =l extuw %final
	ret %result
}
