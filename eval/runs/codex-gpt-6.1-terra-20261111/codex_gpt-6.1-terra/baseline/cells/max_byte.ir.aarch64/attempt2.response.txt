export function l $max_byte(l %p, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @merge %next
    %max =l phi @start 0, @merge %newmax
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %greater =w cugtl %b, %max
    jnz %greater, @update, @keep
@update
    jmp @merge
@keep
    jmp @merge
@merge
    %newmax =l phi @update %b, @keep %max
    %next =l add %i, 1
    jmp @loop
@done
    ret %max
}
