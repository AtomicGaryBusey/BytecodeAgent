export function l $max_byte(l %p, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %next
    %max =l phi @start 0, @body %newmax
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %greater =w cugtl %b, %max
    %newmax =l select %greater, %b, %max
    %next =l add %i, 1
    jmp @loop
@done
    ret %max
}
