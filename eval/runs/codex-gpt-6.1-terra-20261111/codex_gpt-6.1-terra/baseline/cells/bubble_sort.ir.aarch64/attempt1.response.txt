export function l $bubble_sort(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @outer

@outer
	%end =l phi @start %n, @nextpass %newend
	%limit =l sub %end, 1
	jmp @inner

@inner
	%i =l phi @outer 0, @advance %next
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csgtl %x, %y
	jnz %swap, @doswap, @advance

@doswap
	storel %y, %p
	storel %x, %q
	jmp @advance

@advance
	%next =l add %i, 1
	%more =w csltl %next, %limit
	jnz %more, @inner, @nextpass

@nextpass
	%newend =l sub %end, 1
	%again =w csgtl %newend, 1
	jnz %again, @outer, @done

@done
	ret 0
}
