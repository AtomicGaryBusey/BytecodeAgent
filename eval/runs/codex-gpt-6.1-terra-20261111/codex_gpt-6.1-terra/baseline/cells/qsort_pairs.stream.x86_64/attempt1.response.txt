qsort_pairs:
push rbx
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
cmp r13, 1
jle .Ldone
mov rax, r13
shr rax, 1
shl rax, 4
mov r14, qword ptr [r12 + rax]
mov r15, qword ptr [r12 + rax + 8]
xor ebx, ebx
xor ecx, ecx
mov r8, r13
.Lpartition:
cmp rcx, r8
jge .Lpartition_done
mov rax, rcx
shl rax, 4
lea rdx, [r12 + rax]
mov r9, qword ptr [rdx]
mov r10, qword ptr [rdx + 8]
cmp r9, r14
jl .Lless
jg .Lgreater
cmp r10, r15
jl .Lless
jg .Lgreater
inc rcx
jmp .Lpartition
.Lless:
mov rax, rbx
shl rax, 4
lea r11, [r12 + rax]
mov rax, qword ptr [r11]
mov rsi, qword ptr [r11 + 8]
mov qword ptr [r11], r9
mov qword ptr [r11 + 8], r10
mov qword ptr [rdx], rax
mov qword ptr [rdx + 8], rsi
inc rbx
inc rcx
jmp .Lpartition
.Lgreater:
dec r8
mov rax, r8
shl rax, 4
lea r11, [r12 + rax]
mov rax, qword ptr [r11]
mov rsi, qword ptr [r11 + 8]
mov qword ptr [r11], r9
mov qword ptr [r11 + 8], r10
mov qword ptr [rdx], rax
mov qword ptr [rdx + 8], rsi
jmp .Lpartition
.Lpartition_done:
mov r14, r8
cmp rbx, 1
jle .Lright
mov rdi, r12
mov rsi, rbx
call qsort_pairs
.Lright:
sub r13, r14
cmp r13, 1
jle .Ldone
shl r14, 4
lea rdi, [r12 + r14]
mov rsi, r13
call qsort_pairs
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
