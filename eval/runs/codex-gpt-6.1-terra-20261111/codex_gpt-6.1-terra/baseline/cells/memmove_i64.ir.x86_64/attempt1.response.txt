export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @choose

@choose
	%forward =w cultl %dst, %src
	jnz %forward, @fwd, @back

@fwd
	%i =l phi @choose 0, @fwd_loop %inext
	%foff =l mul %i, 8
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fv =l loadl %fsrc
	storel %fv, %fdst
	jmp @fwd_loop

@fwd_loop
	%inext =l add %i, 1
	%fmore =w cultl %inext, %n
	jnz %fmore, @fwd, @done

@back
	%j =l phi @choose %n, @back_loop %jnext
	%jnext =l sub %j, 1
	%boff =l mul %jnext, 8
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bv =l loadl %bsrc
	storel %bv, %bdst
	jmp @back_loop

@back_loop
	%bmore =w cnel %jnext, 0
	jnz %bmore, @back, @done

@done
	ret 0
}
