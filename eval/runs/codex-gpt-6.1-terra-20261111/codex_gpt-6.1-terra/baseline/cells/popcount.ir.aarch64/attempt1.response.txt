export function l $popcount(l %x) {
@start
	%a =l copy %x
	%n =l copy 0
	jmp @loop
@loop
	%v =l phi @start %a, @body %next
	%c =l phi @start %n, @body %inc
	%nonzero =w cnel %v, 0
	jnz %nonzero, @body, @done
@body
	%minus =l sub %v, 1
	%next =l and %v, %minus
	%inc =l add %c, 1
	jmp @loop
@done
	ret %c
}
