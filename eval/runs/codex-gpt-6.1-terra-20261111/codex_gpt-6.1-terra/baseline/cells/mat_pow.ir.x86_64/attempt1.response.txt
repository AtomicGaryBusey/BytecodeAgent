export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%count =l mul %n, %n
	%bytes =l mul %count, 8
	%product =l add %tmp, %bytes
	jmp @init

@init
	%i =l phi @start 0, @init_next %inext
	%doneinit =w ceql %i, %count
	jnz %doneinit, @power_check, @init_body

@init_body
	%row =l div %i, %n
	%col =l rem %i, %n
	%diag =w ceql %row, %col
	%value =l extuw %diag
	%off =l mul %i, 8
	%op =l add %out, %off
	storel %value, %op
	jmp @init_next

@init_next
	%inext =l add %i, 1
	jmp @init

@power_check
	%iter =l phi @init %k, @copy_done %iter_next
	%finished =w ceql %iter, 0
	jnz %finished, @clear_start, @cell_start

@cell_start
	jmp @cell_check

@cell_check
	%cell =l phi @cell_start 0, @cell_next %cell_next_value
	%cells_done =w ceql %cell, %count
	jnz %cells_done, @copy_start, @cell_body

@cell_body
	%r =l div %cell, %n
	%c =l rem %cell, %n
	%rowbase =l mul %r, %n
	jmp @dot_check

@dot_check
	%t =l phi @cell_body 0, @dot_body %tnext
	%sum =l phi @cell_body 0, @dot_body %sum_next
	%dot_done =w ceql %t, %n
	jnz %dot_done, @dot_store, @dot_body

@dot_body
	%oi =l add %rowbase, %t
	%oo =l mul %oi, 8
	%oa =l add %out, %oo
	%x =l loadl %oa
	%arow =l mul %t, %n
	%ai =l add %arow, %c
	%ao =l mul %ai, 8
	%aa =l add %a, %ao
	%y =l loadl %aa
	%term =l mul %x, %y
	%sum_next =l add %sum, %term
	%tnext =l add %t, 1
	jmp @dot_check

@dot_store
	%poff =l mul %cell, 8
	%pp =l add %product, %poff
	storel %sum, %pp
	jmp @cell_next

@cell_next
	%cell_next_value =l add %cell, 1
	jmp @cell_check

@copy_start
	jmp @copy_check

@copy_check
	%ci =l phi @copy_start 0, @copy_body %ci_next
	%copy_finished =w ceql %ci, %count
	jnz %copy_finished, @copy_done, @copy_body

@copy_body
	%coff =l mul %ci, 8
	%cp =l add %product, %coff
	%cv =l loadl %cp
	%cop =l add %out, %coff
	storel %cv, %cop
	%ci_next =l add %ci, 1
	jmp @copy_check

@copy_done
	%iter_next =l sub %iter, 1
	jmp @power_check

@clear_start
	%total =l mul %count, 2
	jmp @clear_check

@clear_check
	%zi =l phi @clear_start 0, @clear_body %zi_next
	%clear_done =w ceql %zi, %total
	jnz %clear_done, @return, @clear_body

@clear_body
	%zoff =l mul %zi, 8
	%zp =l add %tmp, %zoff
	storel 0, %zp
	%zi_next =l add %zi, 1
	jmp @clear_check

@return
	ret 0
}
