export function l $count_desc(l %a, l %n) {
@start
	%small = csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%first = loadl %a
	%end = sub %n, 1
	jmp @loop
@loop
	%i = phi @init 0, @loop %next
	%prev = phi @init %first, @loop %cur
	%count = phi @init 0, @loop %newcount
	%next = add %i, 1
	%offset = mul %next, 8
	%ptr = add %a, %offset
	%cur = loadl %ptr
	%desc = csgtl %prev, %cur
	%inc = extuw %desc
	%newcount = add %count, %inc
	%more = csltl %next, %end
	jnz %more, @loop, @done
@done
	ret %newcount
}
