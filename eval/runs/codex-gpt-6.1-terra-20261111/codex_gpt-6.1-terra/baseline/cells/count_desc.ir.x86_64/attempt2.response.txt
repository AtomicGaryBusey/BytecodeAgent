export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%first =l loadl %a
	%end =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @loop %next
	%prev =l phi @init %first, @loop %cur
	%count =l phi @init 0, @loop %newcount
	%next =l add %i, 1
	%offset =l mul %next, 8
	%ptr =l add %a, %offset
	%cur =l loadl %ptr
	%desc =w csgtl %prev, %cur
	%inc =l extuw %desc
	%newcount =l add %count, %inc
	%more =w csltl %next, %end
	jnz %more, @loop, @done
@done
	ret %newcount
}
