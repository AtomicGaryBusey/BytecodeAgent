export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%count =l mul %n, %n
	%bytes =l mul %count, 8
	%prod =l add %tmp, %bytes
	jmp @init

@init
	%i =l phi @start 0, @initbody %inext
	%imore =w csltl %i, %count
	jnz %imore, @initbody, @power

@initbody
	%row =l div %i, %n
	%col =l rem %i, %n
	%diag =w ceql %row, %col
	%value =l extuw %diag
	%offset =l mul %i, 8
	%optr =l add %out, %offset
	storel %value, %optr
	%inext =l add %i, 1
	jmp @init

@power
	%step =l phi @init %k, @copydone %stepnext
	%hasstep =w csgtl %step, 0
	jnz %hasstep, @rows, @clear

@rows
	%r =l phi @power 0, @rowdone %rnext
	%rmore =w csltl %r, %n
	jnz %rmore, @cols, @copy

@cols
	%c =l phi @rows 0, @coldone %cnext
	%cmore =w csltl %c, %n
	jnz %cmore, @dot, @rowdone

@dot
	%t =l phi @cols 0, @dotbody %tnext
	%sum =l phi @cols 0, @dotbody %sumplus
	%tmore =w csltl %t, %n
	jnz %tmore, @dotbody, @dotdone

@dotbody
	%rbase =l mul %r, %n
	%leftidx =l add %rbase, %t
	%leftoff =l mul %leftidx, 8
	%leftptr =l add %out, %leftoff
	%left =l loadl %leftptr
	%tbase =l mul %t, %n
	%rightidx =l add %tbase, %c
	%rightoff =l mul %rightidx, 8
	%rightptr =l add %a, %rightoff
	%right =l loadl %rightptr
	%term =l mul %left, %right
	%sumplus =l add %sum, %term
	%tnext =l add %t, 1
	jmp @dot

@dotdone
	%outbase =l mul %r, %n
	%outidx =l add %outbase, %c
	%outoff =l mul %outidx, 8
	%prodptr =l add %prod, %outoff
	storel %sum, %prodptr
	jmp @coldone

@coldone
	%cnext =l add %c, 1
	jmp @cols

@rowdone
	%rnext =l add %r, 1
	jmp @rows

@copy
	%j =l phi @rows 0, @copybody %jnext
	%jmore =w csltl %j, %count
	jnz %jmore, @copybody, @copydone

@copybody
	%joff =l mul %j, 8
	%src =l add %prod, %joff
	%dst =l add %out, %joff
	%v =l loadl %src
	storel %v, %dst
	%jnext =l add %j, 1
	jmp @copy

@copydone
	%stepnext =l sub %step, 1
	jmp @power

@clear
	%z =l phi @power 0, @clearbody %znext
	%twocount =l mul %count, 2
	%zmore =w csltl %z, %twocount
	jnz %zmore, @clearbody, @done

@clearbody
	%zoff =l mul %z, 8
	%zptr =l add %tmp, %zoff
	storel 0, %zptr
	%znext =l add %z, 1
	jmp @clear

@done
	ret 0
}
