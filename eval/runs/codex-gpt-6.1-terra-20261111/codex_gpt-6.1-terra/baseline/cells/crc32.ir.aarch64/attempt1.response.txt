export function l $crc32(l %p, l %n) {
@start
	jmp @bytes
@bytes
	%ptr =l phi @start %p, @next %ptr1
	%left =l phi @start %n, @next %left1
	%crc =w phi @start 4294967295, @next %cnext
	%more =w cnel %left, 0
	jnz %more, @byte, @done
@byte
	%b =w loadub %ptr
	%init =w xor %crc, %b
	jmp @bits
@bits
	%c =w phi @byte %init, @bits %cnext
	%i =w phi @byte 8, @bits %i1
	%low =w and %c, 1
	%mask =w sub 0, %low
	%poly =w and %mask, 3988292384
	%shift =w shr %c, 1
	%cnext =w xor %shift, %poly
	%i1 =w sub %i, 1
	%again =w cnew %i1, 0
	jnz %again, @bits, @next
@next
	%ptr1 =l add %ptr, 1
	%left1 =l sub %left, 1
	jmp @bytes
@done
	%final =w xor %crc, 4294967295
	%result =l extuw %final
	ret %result
}
