export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @merge %next
	%max =l phi @start %first, @merge %newmax
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%greater =w csgtl %value, %max
	jnz %greater, @update, @keep
@update
	jmp @merge
@keep
	jmp @merge
@merge
	%newmax =l phi @update %value, @keep %max
	%next =l add %i, 1
	jmp @loop
@done
	ret %max
}
