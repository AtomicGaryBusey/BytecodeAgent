export function l $isqrt(l %n) {
@start
	jnz 1, @loop, @loop
@loop
	%lo =l phi @start 0, @next %newlo
	%hi =l phi @start 4294967296, @next %newhi
	%diff =l sub %hi, %lo
	%done =w ceql %diff, 1
	jnz %done, @end, @test
@test
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%square =l mul %mid, %mid
	%fits =w culel %square, %n
	jnz %fits, @lower, @upper
@lower
	jmp @next
@upper
	jmp @next
@next
	%newlo =l phi @lower %mid, @upper %lo
	%newhi =l phi @lower %hi, @upper %mid
	jmp @loop
@end
	ret %lo
}
