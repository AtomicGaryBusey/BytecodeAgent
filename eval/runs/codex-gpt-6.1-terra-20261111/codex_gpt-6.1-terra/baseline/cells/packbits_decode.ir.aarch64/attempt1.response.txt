export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet

@packet
	%i =l phi @start 0, @noop %next, @lit_done %lit_end, @rep_done %rep_end
	%o =l phi @start 0, @noop %o, @lit_done %lit_o_end, @rep_done %rep_o_end
	%finished =w ceql %i, %n
	jnz %finished, @success, @header

@header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%next =l add %i, 1
	%nonnegative =w csgesw %h, 0
	jnz %nonnegative, @literal, @negative

@negative
	%is_noop =w ceqw %h, -128
	jnz %is_noop, @noop, @repeat

@noop
	jmp @packet

@literal
	%hw =l extsw %h
	%lit_count =l add %hw, 1
	%lit_remain =l sub %n, %next
	%lit_short =w csltl %lit_remain, %lit_count
	jnz %lit_short, @source_error, @lit_setup

@lit_setup
	%lit_end =l add %next, %lit_count
	%lit_o_end =l add %o, %lit_count
	jmp @lit_loop

@lit_loop
	%li =l phi @lit_setup %next, @lit_copy %li_next
	%lo =l phi @lit_setup %o, @lit_copy %lo_next
	%lit_complete =w ceql %li, %lit_end
	jnz %lit_complete, @lit_done, @lit_space

@lit_space
	%lit_full =w ceql %lo, %cap
	jnz %lit_full, @capacity_error, @lit_copy

@lit_copy
	%lp =l add %src, %li
	%lb =w loadub %lp
	%lop =l add %out, %lo
	storeb %lb, %lop
	%li_next =l add %li, 1
	%lo_next =l add %lo, 1
	jmp @lit_loop

@lit_done
	jmp @packet

@repeat
	%rep_missing =w ceql %next, %n
	jnz %rep_missing, @source_error, @rep_setup

@rep_setup
	%rh =l extsw %h
	%rep_count =l sub 1, %rh
	%rep_end =l add %next, 1
	%rep_o_end =l add %o, %rep_count
	%rp =l add %src, %next
	%rb =w loadub %rp
	jmp @rep_loop

@rep_loop
	%ro =l phi @rep_setup %o, @rep_copy %ro_next
	%rep_complete =w ceql %ro, %rep_o_end
	jnz %rep_complete, @rep_done, @rep_space

@rep_space
	%rep_full =w ceql %ro, %cap
	jnz %rep_full, @capacity_error, @rep_copy

@rep_copy
	%rop =l add %out, %ro
	storeb %rb, %rop
	%ro_next =l add %ro, 1
	jmp @rep_loop

@rep_done
	jmp @packet

@success
	ret %o

@source_error
	ret -2

@capacity_error
	ret -1
}
