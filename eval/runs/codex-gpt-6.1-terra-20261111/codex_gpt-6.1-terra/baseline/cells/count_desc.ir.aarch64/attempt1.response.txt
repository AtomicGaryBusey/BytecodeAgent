export function l $count_desc(l %a, l %n) {
@start
	%enough =w cslel 2, %n
	jnz %enough, @init, @empty
@empty
	ret 0
@init
	%first =l loadl %a
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @body %nexti
	%count =l phi @init 0, @body %nextcount
	%prev =l phi @init %first, @body %cur
	%more =w csltl %i, %limit
	jnz %more, @body, @done
@body
	%nexti =l add %i, 1
	%offset =l mul %nexti, 8
	%addr =l add %a, %offset
	%cur =l loadl %addr
	%descending =w csgtl %prev, %cur
	%inc =l extuw %descending
	%nextcount =l add %count, %inc
	jmp @loop
@done
	ret %count
}
