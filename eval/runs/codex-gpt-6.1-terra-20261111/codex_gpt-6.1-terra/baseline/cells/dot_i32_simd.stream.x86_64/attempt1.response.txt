dot_i32_simd:
pxor xmm0, xmm0
cmp rdx, 4
jl .Lreduce
.Lvector:
movdqu xmm1, xmmword ptr [rdi]
movdqu xmm2, xmmword ptr [rsi]
movdqa xmm3, xmm1
movdqa xmm4, xmm2
pmuldq xmm1, xmm2
psrldq xmm3, 4
psrldq xmm4, 4
pmuldq xmm3, xmm4
paddq xmm0, xmm1
paddq xmm0, xmm3
add rdi, 16
add rsi, 16
sub rdx, 4
cmp rdx, 4
jge .Lvector
.Lreduce:
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
test rdx, rdx
jz .Ldone
.Lscalar:
movsxd rcx, dword ptr [rdi]
movsxd r8, dword ptr [rsi]
imul rcx, r8
add rax, rcx
add rdi, 4
add rsi, 4
dec rdx
jnz .Lscalar
.Ldone:
ret
