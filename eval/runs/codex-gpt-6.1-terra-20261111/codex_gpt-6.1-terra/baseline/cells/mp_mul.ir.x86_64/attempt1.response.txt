export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero

@zero
	%zi =l phi @start 0, @zero_body %zi_next
	%ztest =w cultl %zi, %n
	jnz %ztest, @zero_body, @outer

@zero_body
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zi_next =l add %zi, 1
	jmp @zero

@outer
	%i =l phi @zero 0, @outer_next %i_next
	%itest =w cultl %i, %na
	jnz %itest, @outer_body, @scan

@outer_body
	%aoff =l mul %i, 8
	%ap =l add %a, %aoff
	%av =l loadl %ap
	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	jmp @inner

@inner
	%j =l phi @outer_body 0, @inner_body %j_next
	%carry =l phi @outer_body 0, @inner_body %carry_next
	%jtest =w cultl %j, %nb
	jnz %jtest, @inner_body, @inner_done

@inner_body
	%boff =l mul %j, 8
	%bp =l add %b, %boff
	%bv =l loadl %bp
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p10lo =l and %p10, 4294967295
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%midlo =l shl %mid, 32
	%p00lo =l and %p00, 4294967295
	%lo =l or %midlo, %p00lo
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%midhi =l shr %mid, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%hi =l add %hi1, %midhi
	%idx =l add %i, %j
	%roff =l mul %idx, 8
	%rp =l add %r, %roff
	%old =l loadl %rp
	%s0 =l add %lo, %old
	%c0w =w cultl %s0, %lo
	%c0 =l extuw %c0w
	%s1 =l add %s0, %carry
	%c1w =w cultl %s1, %s0
	%c1 =l extuw %c1w
	storel %s1, %rp
	%h2 =l add %hi, %c0
	%carry_next =l add %h2, %c1
	%j_next =l add %j, 1
	jmp @inner

@inner_done
	%has_b =w cnel %nb, 0
	jnz %has_b, @store_carry, @outer_next

@store_carry
	%endidx =l add %i, %nb
	%endoff =l mul %endidx, 8
	%endp =l add %r, %endoff
	storel %carry, %endp
	jmp @outer_next

@outer_next
	%i_next =l add %i, 1
	jmp @outer

@scan
	%k =l phi @outer %n, @scan_zero %k_prev
	%nonempty =w cnel %k, 0
	jnz %nonempty, @scan_load, @done_zero

@scan_load
	%k_prev =l sub %k, 1
	%koff =l mul %k_prev, 8
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%knz =w cnel %kv, 0
	jnz %knz, @done_nonzero, @scan_zero

@scan_zero
	jmp @scan

@done_zero
	ret 0

@done_nonzero
	ret %k
}
