export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %next
	%count =l phi @start 0, @body %sum
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%value =l loadl %addr
	%gt =w cugtl %value, %t
	%inc =l extuw %gt
	%sum =l add %count, %inc
	%next =l add %i, 1
	jmp @loop
@done
	ret %count
}
