export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %next, @greater %lo
	%hi =l phi @start %n, @less %hi, @greater %mid
	%active =w csltl %lo, %hi
	jnz %active, @read, @missing
@read
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%offset =l shl %mid, 3
	%addr =l add %a, %offset
	%value =l loadl %addr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%isless =w csltl %value, %key
	jnz %isless, @less, @greater
@less
	%next =l add %mid, 1
	jmp @loop
@greater
	jmp @loop
@found
	ret %mid
@missing
	ret -1
}
