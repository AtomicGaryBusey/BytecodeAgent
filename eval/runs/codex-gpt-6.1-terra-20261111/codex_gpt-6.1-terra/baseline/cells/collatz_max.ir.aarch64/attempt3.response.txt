export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %knext
	%max =l phi @start 0, @next %newmax
	jmp @collatz
@collatz
	%x =l phi @outer %k, @merge %xnext
	%count =l phi @outer 0, @merge %countnext
	%done =w ceql %x, 1
	jnz %done, @finish, @step
@step
	%odd =l and %x, 1
	%isodd =w cnel %odd, 0
	jnz %isodd, @oddstep, @evenstep
@oddstep
	%triple =l mul %x, 3
	%oddnext =l add %triple, 1
	jmp @merge
@evenstep
	%evennext =l shr %x, 1
	jmp @merge
@merge
	%xnext =l phi @oddstep %oddnext, @evenstep %evennext
	%countnext =l add %count, 1
	jmp @collatz
@finish
	%greater =w csgtl %count, %max
	jnz %greater, @update, @keep
@update
	jmp @maxmerge
@keep
	jmp @maxmerge
@maxmerge
	%newmax =l phi @update %count, @keep %max
	%last =w ceql %k, %n
	jnz %last, @return, @next
@next
	%knext =l add %k, 1
	jmp @outer
@return
	ret %newmax
}
