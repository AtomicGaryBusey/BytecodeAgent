export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k = phi @start 1, @next %knext
	%max = phi @start 0, @next %newmax
	jmp @collatz
@collatz
	%x = phi @outer %k, @step %xnext
	%count = phi @outer 0, @step %countnext
	%done =l ceql %x, 1
	jnz %done, @finish, @step
@step
	%odd =l and %x, 1
	%isodd =w cnel %odd, 0
	%triple =l mul %x, 3
	%oddnext =l add %triple, 1
	%evennext =l shr %x, 1
	%xnext =l select %isodd, %oddnext, %evennext
	%countnext =l add %count, 1
	jmp @collatz
@finish
	%greater =w csgtl %count, %max
	%newmax =l select %greater, %count, %max
	%last =w ceql %k, %n
	jnz %last, @return, @next
@next
	%knext =l add %k, 1
	jmp @outer
@return
	ret %newmax
}
