export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %knext
	%max =l phi @start 0, @next %newmax
	jmp @inner
@inner
	%x =l phi @outer %k, @step %xnext
	%count =l phi @outer 0, @step %countnext
	%done =w ceql %x, 1
	jnz %done, @next, @step
@step
	%odd =l and %x, 1
	%isodd =w cnel %odd, 0
	%half =l div %x, 2
	%triple =l mul %x, 3
	%oddnext =l add %triple, 1
	%xnext =l phi @step %half, @step %oddnext
	%countnext =l add %count, 1
	jmp @inner
@next
	%greater =w csgtl %count, %max
	jnz %greater, @update, @keep
@update
	jmp @merge
@keep
	jmp @merge
@merge
	%newmax =l phi @update %count, @keep %max
	%last =w ceql %k, %n
	%knext =l add %k, 1
	jnz %last, @end, @outer
@end
	ret %newmax
}
