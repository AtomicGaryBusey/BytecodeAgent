export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet
@packet
	%si =l phi @start 0, @noop %afterh, @litdone %litend, @repdone %repend
	%oi =l phi @start 0, @noop %oi, @litdone %litout, @repdone %repout
	%finished =w ceql %si, %n
	jnz %finished, @success, @header
@header
	%hp =l add %src, %si
	%h =w loadsb %hp
	%afterh =l add %si, 1
	%isnoop =w ceqw %h, -128
	jnz %isnoop, @noop, @classify
@noop
	jmp @packet
@classify
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%hl =l extsw %h
	%litlen =l add %hl, 1
	%litend =l add %afterh, %litlen
	%litbad =w csgtl %litend, %n
	jnz %litbad, @badsource, @litcheck
@litcheck
	%litout =l add %oi, %litlen
	%litoverflow =w csgtl %litout, %cap
	%litavail =l sub %cap, %oi
	%litcount =l copy %litlen
	jnz %litoverflow, @litshort, @litsetup
@litshort
	jmp @litsetup
@litsetup
	%lc =l phi @litcheck %litcount, @litshort %litavail
	jmp @litloop
@litloop
	%li =l phi @litsetup 0, @litbody %linext
	%litfinished =w ceql %li, %lc
	jnz %litfinished, @litresult, @litbody
@litbody
	%lso =l add %afterh, %li
	%lsp =l add %src, %lso
	%byte =w loadub %lsp
	%loo =l add %oi, %li
	%lop =l add %out, %loo
	storeb %byte, %lop
	%linext =l add %li, 1
	jmp @litloop
@litresult
	jnz %litoverflow, @overflow, @litdone
@litdone
	jmp @packet
@repeat
	%repend =l add %afterh, 1
	%repbad =w csgtl %repend, %n
	jnz %repbad, @badsource, @repcheck
@repcheck
	%rh =l extsw %h
	%replen =l sub 1, %rh
	%repout =l add %oi, %replen
	%repoverflow =w csgtl %repout, %cap
	%repavail =l sub %cap, %oi
	%rsp =l add %src, %afterh
	%repbyte =w loadub %rsp
	jnz %repoverflow, @repshort, @repsetup
@repshort
	jmp @repsetup
@repsetup
	%rc =l phi @repcheck %replen, @repshort %repavail
	jmp @reploop
@reploop
	%ri =l phi @repsetup 0, @repbody %rinext
	%repfinished =w ceql %ri, %rc
	jnz %repfinished, @represult, @repbody
@repbody
	%roo =l add %oi, %ri
	%rop =l add %out, %roo
	storeb %repbyte, %rop
	%rinext =l add %ri, 1
	jmp @reploop
@represult
	jnz %repoverflow, @overflow, @repdone
@repdone
	jmp @packet
@success
	ret %oi
@badsource
	ret -2
@overflow
	ret -1
}
