export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @setup

@setup
	%mid =l shr %n, 1
	%midoff =l shl %mid, 4
	%midptr =l add %a, %midoff
	%pk =l loadl %midptr
	%midvalptr =l add %midptr, 8
	%pv =l loadl %midvalptr
	jmp @loop

@loop
	%i =l phi @setup 0, @less %inext, @equal %inext, @greater %i
	%lt =l phi @setup 0, @less %ltnext, @equal %lt, @greater %lt
	%gt =l phi @setup %n, @less %gt, @equal %gt, @greater %gtnext
	%more =w csltl %i, %gt
	jnz %more, @compare, @recurse

@compare
	%ioff =l shl %i, 4
	%iptr =l add %a, %ioff
	%ivptr =l add %iptr, 8
	%ik =l loadl %iptr
	%iv =l loadl %ivptr
	%kless =w csltl %ik, %pk
	%keq =w ceql %ik, %pk
	%vless =w csltl %iv, %pv
	%eqvless =w and %keq, %vless
	%isless =w or %kless, %eqvless
	jnz %isless, @less, @checkgreater

@checkgreater
	%kgreater =w csgtl %ik, %pk
	%vgreater =w csgtl %iv, %pv
	%eqvgreater =w and %keq, %vgreater
	%isgreater =w or %kgreater, %eqvgreater
	jnz %isgreater, @greater, @equal

@less
	%ltoff =l shl %lt, 4
	%ltptr =l add %a, %ltoff
	%ltvptr =l add %ltptr, 8
	%ltk =l loadl %ltptr
	%ltv =l loadl %ltvptr
	storel %ik, %ltptr
	storel %iv, %ltvptr
	storel %ltk, %iptr
	storel %ltv, %ivptr
	%ltnext =l add %lt, 1
	%inext =l add %i, 1
	jmp @loop

@equal
	%inext =l add %i, 1
	jmp @loop

@greater
	%gtnext =l sub %gt, 1
	%gtoff =l shl %gtnext, 4
	%gtptr =l add %a, %gtoff
	%gtvptr =l add %gtptr, 8
	%gtk =l loadl %gtptr
	%gtv =l loadl %gtvptr
	storel %gtk, %iptr
	storel %gtv, %ivptr
	storel %ik, %gtptr
	storel %iv, %gtvptr
	jmp @loop

@recurse
	%left =l call $qsort_pairs(l %a, l %lt)
	%rightoff =l shl %gt, 4
	%rightptr =l add %a, %rightoff
	%rightn =l sub %n, %gt
	%right =l call $qsort_pairs(l %rightptr, l %rightn)
	jmp @done

@done
	ret 0
}
