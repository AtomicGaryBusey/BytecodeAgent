export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop

@loop
	%i =l phi @start 1, @bodyend %inext
	%min =l phi @start %y0, @bodyend %minnext
	%max =l phi @start %y0, @bodyend %maxnext
	%sum =l phi @start %y0, @bodyend %sumnext
	%count =l phi @start %odd0, @bodyend %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	jnz %ismin, @newmin, @oldmin

@newmin
	jmp @minmerge

@oldmin
	jmp @minmerge

@minmerge
	%minnext =l phi @newmin %y, @oldmin %min
	%ismax =w csgtl %y, %max
	jnz %ismax, @newmax, @oldmax

@newmax
	jmp @bodyend

@oldmax
	jmp @bodyend

@bodyend
	%maxnext =l phi @newmax %y, @oldmax %max
	%sumnext =l add %sum, %y
	%odd =l and %y, 1
	%countnext =l add %count, %odd
	%inext =l add %i, 1
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
