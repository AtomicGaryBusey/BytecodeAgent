export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @nextgroup %inext
	%o =l phi @start 0, @nextgroup %onext
	%more =w csltl %i, %n
	jnz %more, @load0, @done
@load0
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%remaining =l sub %n, %i
	%has1 =w csgtl %remaining, 1
	jnz %has1, @load1, @no1
@load1
	%p1 =l add %p0, 1
	%b1load =l loadub %p1
	jmp @after1
@no1
	jmp @after1
@after1
	%b1 =l phi @load1 %b1load, @no1 0
	%has2 =w csgtl %remaining, 2
	jnz %has2, @load2, @no2
@load2
	%p2 =l add %p0, 2
	%b2load =l loadub %p2
	jmp @after2
@no2
	jmp @after2
@after2
	%b2 =l phi @load2 %b2load, @no2 0
	%v0 =l shl %b0, 16
	%v1 =l shl %b1, 8
	%v01 =l or %v0, %v1
	%v =l or %v01, %b2
	jmp @char
@char
	%j =l phi @after2 0, @charend %jnext
	%is2 =w ceql %j, 2
	%is3 =w ceql %j, 3
	%one =w ceql %remaining, 1
	%two =w ceql %remaining, 2
	%pad2 =w and %is2, %one
	%short =w or %one, %two
	%pad3 =w and %is3, %short
	%pad =w or %pad2, %pad3
	jnz %pad, @padding, @index
@padding
	jmp @emit
@index
	%shift0 =l sub 3, %j
	%shift =l mul %shift0, 6
	%shifted =l shr %v, %shift
	%idx =l and %shifted, 63
	%upper =w csltl %idx, 26
	jnz %upper, @mapupper, @checklower
@mapupper
	%cu =l add %idx, 65
	jmp @emit
@checklower
	%lower =w csltl %idx, 52
	jnz %lower, @maplower, @checkdigit
@maplower
	%cl =l add %idx, 71
	jmp @emit
@checkdigit
	%digit =w csltl %idx, 62
	jnz %digit, @mapdigit, @mapspecial
@mapdigit
	%cd =l sub %idx, 4
	jmp @emit
@mapspecial
	%is62 =w ceql %idx, 62
	jnz %is62, @mapplus, @mapslash
@mapplus
	jmp @emit
@mapslash
	jmp @emit
@emit
	%c =l phi @padding 61, @mapupper %cu, @maplower %cl, @mapdigit %cd, @mapplus 43, @mapslash 47
	%opos =l add %o, %j
	%op =l add %out, %opos
	storeb %c, %op
	jmp @charend
@charend
	%jnext =l add %j, 1
	%charhasmore =w csltl %jnext, 4
	jnz %charhasmore, @char, @nextgroup
@nextgroup
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @group
@done
	ret %o
}
