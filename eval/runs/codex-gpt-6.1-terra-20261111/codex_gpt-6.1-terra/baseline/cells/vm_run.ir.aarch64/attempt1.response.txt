export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @next1 %pc1, @next2 %pc2, @jump %target, @jznext %jzpc, @jztaken %jztarget
	%depth =l phi @start 0, @next1 %d1, @next2 %d2, @jump %depth, @jznext %jzdepth, @jztaken %jzdepth
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch
@fetch
	%coff =l mul %pc, 8
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%halt =w ceql %op, 0
	jnz %halt, @finish, @dispatch
@dispatch
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @checkbin
@checkbin
	%ge2 =w csgel %op, 2
	%le4 =w cslel %op, 4
	%arith =w and %ge2, %le4
	%islt =w ceql %op, 9
	%isbin =w or %arith, %islt
	jnz %isbin, @binary, @checkdup
@checkdup
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @checkswap
@checkswap
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @checkdrop
@checkdrop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @checkover
@checkover
	%isover =w ceql %op, 8
	jnz %isover, @over, @checkjz
@checkjz
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @checkjmp
@checkjmp
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jump, @unknown
@push
	%pushoperandaddr =l add %caddr, 8
	%pushvalue =l loadl %pushoperandaddr
	%pushoff =l mul %depth, 8
	%pushaddr =l add %stack, %pushoff
	storel %pushvalue, %pushaddr
	%pushdepth =l add %depth, 1
	jmp @next2
@binary
	%bi =l sub %depth, 1
	%ai =l sub %depth, 2
	%boff =l mul %bi, 8
	%aoff =l mul %ai, 8
	%baddr =l add %stack, %boff
	%aaddr =l add %stack, %aoff
	%b =l loadl %baddr
	%a =l loadl %aaddr
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @checksub
@add
	%sum =l add %a, %b
	jmp @binstore
@checksub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @checkmul
@sub
	%diff =l sub %a, %b
	jmp @binstore
@checkmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @lt
@mul
	%product =l mul %a, %b
	jmp @binstore
@lt
	%less =w csltl %a, %b
	%less64 =l extuw %less
	jmp @binstore
@binstore
	%result =l phi @add %sum, @sub %diff, @mul %product, @lt %less64
	storel %result, %aaddr
	jmp @next1
@dup
	%dupi =l sub %depth, 1
	%dupoff =l mul %dupi, 8
	%dupsrc =l add %stack, %dupoff
	%dupvalue =l loadl %dupsrc
	%dupdstoff =l mul %depth, 8
	%dupdst =l add %stack, %dupdstoff
	storel %dupvalue, %dupdst
	%dupdepth =l add %depth, 1
	jmp @next1
@swap
	%swapi =l sub %depth, 1
	%swapj =l sub %depth, 2
	%swapioff =l mul %swapi, 8
	%swapjoff =l mul %swapj, 8
	%swapip =l add %stack, %swapioff
	%swapjp =l add %stack, %swapjoff
	%swapiv =l loadl %swapip
	%swapjv =l loadl %swapjp
	storel %swapiv, %swapjp
	storel %swapjv, %swapip
	jmp @next1
@drop
	%dropdepth =l sub %depth, 1
	jmp @next1
@over
	%overi =l sub %depth, 2
	%overoff =l mul %overi, 8
	%oversrc =l add %stack, %overoff
	%overvalue =l loadl %oversrc
	%overdstoff =l mul %depth, 8
	%overdst =l add %stack, %overdstoff
	storel %overvalue, %overdst
	%overdepth =l add %depth, 1
	jmp @next1
@jz
	%jzdepth =l sub %depth, 1
	%jzoff =l mul %jzdepth, 8
	%jzaddr =l add %stack, %jzoff
	%jzvalue =l loadl %jzaddr
	%jzoperandaddr =l add %caddr, 8
	%jztarget =l loadl %jzoperandaddr
	%iszero =w ceql %jzvalue, 0
	jnz %iszero, @jztaken, @jznext
@jztaken
	jmp @loop
@jznext
	%jzpc =l add %pc, 2
	jmp @loop
@jump
	%jmpoperandaddr =l add %caddr, 8
	%target =l loadl %jmpoperandaddr
	jmp @loop
@unknown
	jmp @next1
@next1
	%d1 =l phi @binstore %bi, @dup %dupdepth, @swap %depth, @drop %dropdepth, @over %overdepth, @unknown %depth
	%pc1 =l add %pc, 1
	jmp @loop
@next2
	%d2 =l copy %pushdepth
	%pc2 =l add %pc, 2
	jmp @loop
@finish
	%empty =w ceql %depth, 0
	jnz %empty, @emptyret, @topret
@emptyret
	jmp @clear
@topret
	%topi =l sub %depth, 1
	%topoff =l mul %topi, 8
	%topaddr =l add %stack, %topoff
	%top =l loadl %topaddr
	jmp @clear
@clear
	%retval =l phi @emptyret 0, @topret %top, @clearbody %retval
	%i =l phi @emptyret %depth, @topret %depth, @clearbody %inext
	%done =w ceql %i, %cap
	jnz %done, @return, @clearbody
@clearbody
	%clearoff =l mul %i, 8
	%clearaddr =l add %stack, %clearoff
	storel 0, %clearaddr
	%inext =l add %i, 1
	jmp @clear
@return
	ret %retval
}
