packbits_encode:
    cld
    xor r8d, r8d
    xor r9d, r9d
.Lpacket:
    cmp r8, rsi
    jae .Ldone
    movzx eax, byte ptr [rdi + r8]
    mov r10d, 1
.Lrun:
    cmp r10, 128
    jae .Lrun_done
    lea r11, [r8 + r10]
    cmp r11, rsi
    jae .Lrun_done
    cmp al, byte ptr [rdi + r11]
    jne .Lrun_done
    inc r10
    jmp .Lrun
.Lrun_done:
    cmp r10, 3
    jb .Lliteral_start
    mov r11, rcx
    sub r11, r9
    cmp r11, 2
    jb .Lfail
    mov r11d, 1
    sub r11d, r10d
    mov byte ptr [rdx + r9], r11b
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lpacket
.Lliteral_start:
    xor r10d, r10d
.Lliteral_scan:
    cmp r10, 128
    jae .Lliteral_emit
    lea r11, [r8 + r10]
    cmp r11, rsi
    jae .Lliteral_emit
    lea rax, [r11 + 2]
    cmp rax, rsi
    jae .Lliteral_advance
    mov al, byte ptr [rdi + r11]
    cmp al, byte ptr [rdi + r11 + 1]
    jne .Lliteral_advance
    cmp al, byte ptr [rdi + r11 + 2]
    je .Lliteral_emit
.Lliteral_advance:
    inc r10
    jmp .Lliteral_scan
.Lliteral_emit:
    mov r11, rcx
    sub r11, r9
    lea rax, [r10 + 1]
    cmp r11, rax
    jb .Lfail
    lea eax, [r10 - 1]
    mov byte ptr [rdx + r9], al
    inc r9
    xor r11d, r11d
.Lliteral_copy:
    cmp r11, r10
    jae .Lliteral_done
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    inc r11
    jmp .Lliteral_copy
.Lliteral_done:
    jmp .Lpacket
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
