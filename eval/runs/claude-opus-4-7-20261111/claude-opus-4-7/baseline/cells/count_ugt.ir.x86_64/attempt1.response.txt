export function l $count_ugt(l %a, l %n, l %t) {
@start
	%count =l copy 0
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l mul %i, 8
	%addr2 =l add %a, %addr
	%v =l loadl %addr2
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%count =l add %count, %gtl
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
