export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop_head
@loop_head
    %i =l phi @start 0, @after_packet %i_next
    %w =l phi @start 0, @after_packet %w_next
    %at_end =w csgel %i, %n
    jnz %at_end, @return_w, @compute_run_init
@compute_run_init
    %pi =l add %src, %i
    %b0 =w loadub %pi
    %j_init =l add %i, 1
    jmp @run_loop_head
@run_loop_head
    %r =l phi @compute_run_init 1, @run_continue %r_inc
    %j =l phi @compute_run_init %j_init, @run_continue %j_inc
    %r_at_cap =w csgel %r, 128
    jnz %r_at_cap, @run_done, @run_check_n
@run_check_n
    %j_ge_n =w csgel %j, %n
    jnz %j_ge_n, @run_done, @run_check_eq
@run_check_eq
    %pj =l add %src, %j
    %bj =w loadub %pj
    %neq =w cnew %bj, %b0
    jnz %neq, @run_done, @run_continue
@run_continue
    %r_inc =l add %r, 1
    %j_inc =l add %j, 1
    jmp @run_loop_head
@run_done
    %run_ge_3 =w csgel %r, 3
    jnz %run_ge_3, @check_repeat_fit, @lit_scan_head
@check_repeat_fit
    %rem_r =l sub %cap, %w
    %fits_r =w csgel %rem_r, 2
    jnz %fits_r, @do_repeat, @return_fail
@do_repeat
    %hdr_r =l sub 257, %r
    %po_r =l add %out, %w
    storeb %hdr_r, %po_r
    %po_r2 =l add %po_r, 1
    storeb %b0, %po_r2
    %w_next_r =l add %w, 2
    %i_next_r =l add %i, %r
    jmp @after_packet
@lit_scan_head
    %jl =l phi @run_done %i, @lit_cont %jl_next
    %jl_ge_n =w csgel %jl, %n
    jnz %jl_ge_n, @lit_done, @lit_check_len
@lit_check_len
    %lit_len_s =l sub %jl, %i
    %len_ge_128 =w csgel %lit_len_s, 128
    jnz %len_ge_128, @lit_done, @lit_check_run
@lit_check_run
    %jl_p2 =l add %jl, 2
    %has_3 =w csltl %jl_p2, %n
    jnz %has_3, @lit_check_eq1, @lit_cont
@lit_check_eq1
    %pjl =l add %src, %jl
    %b_jl =w loadub %pjl
    %pjl_1 =l add %pjl, 1
    %b_jl1 =w loadub %pjl_1
    %eq1 =w ceqw %b_jl, %b_jl1
    jnz %eq1, @lit_check_eq2, @lit_cont
@lit_check_eq2
    %pjl_2 =l add %pjl, 2
    %b_jl2 =w loadub %pjl_2
    %eq2 =w ceqw %b_jl, %b_jl2
    jnz %eq2, @lit_done, @lit_cont
@lit_cont
    %jl_next =l add %jl, 1
    jmp @lit_scan_head
@lit_done
    %L =l sub %jl, %i
    %needed =l add %L, 1
    %rem_l =l sub %cap, %w
    %fits_l =w csgel %rem_l, %needed
    jnz %fits_l, @do_literal, @return_fail
@do_literal
    %hdr_l =l sub %L, 1
    %po_l =l add %out, %w
    storeb %hdr_l, %po_l
    %w_after_hdr =l add %w, 1
    jmp @copy_head
@copy_head
    %k =l phi @do_literal 0, @copy_body %k_next
    %k_lt_L =w csltl %k, %L
    jnz %k_lt_L, @copy_body, @copy_done
@copy_body
    %src_pos =l add %src, %i
    %src_k_pos =l add %src_pos, %k
    %byte_k =w loadub %src_k_pos
    %w_pos =l add %w_after_hdr, %k
    %out_k_pos =l add %out, %w_pos
    storeb %byte_k, %out_k_pos
    %k_next =l add %k, 1
    jmp @copy_head
@copy_done
    %i_next_l =l copy %jl
    %w_next_l =l add %w_after_hdr, %L
    jmp @after_packet
@after_packet
    %i_next =l phi @do_repeat %i_next_r, @copy_done %i_next_l
    %w_next =l phi @do_repeat %w_next_r, @copy_done %w_next_l
    jmp @loop_head
@return_fail
    %neg1 =l copy -1
    ret %neg1
@return_w
    ret %w
}
