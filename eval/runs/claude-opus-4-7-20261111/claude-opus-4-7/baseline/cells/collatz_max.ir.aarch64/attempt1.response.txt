export function l $collatz_max(l %n) {
@start
	jmp @loop_k_head
@loop_k_head
	%k =l phi @start 1, @next %k_next
	%max =l phi @start 0, @next %max_next
	%cond_k =w csgtl %k, %n
	jnz %cond_k, @end, @body_k
@body_k
	jmp @loop_c_head
@loop_c_head
	%x =l phi @body_k %k, @even %x_even, @odd %x_odd
	%steps =l phi @body_k 0, @even %steps_e, @odd %steps_o
	%done =w ceql %x, 1
	jnz %done, @check_max, @step
@step
	%rem =l and %x, 1
	%remnz =w cnel %rem, 0
	jnz %remnz, @odd, @even
@even
	%x_even =l div %x, 2
	%steps_e =l add %steps, 1
	jmp @loop_c_head
@odd
	%x3 =l mul %x, 3
	%x_odd =l add %x3, 1
	%steps_o =l add %steps, 1
	jmp @loop_c_head
@check_max
	%gt =w csgtl %steps, %max
	jnz %gt, @update, @no_update
@update
	jmp @next
@no_update
	jmp @next
@next
	%max_next =l phi @update %steps, @no_update %max
	%k_next =l add %k, 1
	jmp @loop_k_head
@end
	ret %max
}
