    xor eax, eax
    mov rcx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov rdx, rcx
    xor r8d, r8d
.Linner:
    cmp rdx, 1
    je .Lcheck
    test dl, 1
    jnz .Lodd
    shr rdx, 1
    inc r8
    jmp .Linner
.Lodd:
    lea rdx, [rdx + rdx*2]
    inc rdx
    inc r8
    jmp .Linner
.Lcheck:
    cmp r8, rax
    jle .Lskip
    mov rax, r8
.Lskip:
    inc rcx
    jmp .Louter
.Ldone:
    ret
