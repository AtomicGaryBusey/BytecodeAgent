export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	%resp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	storel 0, %resp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%sp =l loadl %spp
	%atend =w ceql %pc, %len
	jnz %atend, @done, @cont
@cont
	%pcoff =l mul %pc, 8
	%codeaddr =l add %code, %pcoff
	%op =l loadl %codeaddr
	%is0 =w ceql %op, 0
	jnz %is0, @done, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @opush, @c2
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @oadd, @c3
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @osub, @c4
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @omul, @c5
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @odup, @c6
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @oswap, @c7
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @odrop, @c8
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @oover, @c9
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @olt, @c10
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @ojz, @ojmp
@opush
	%pcp1 =l add %pc, 1
	%offp1 =l mul %pcp1, 8
	%addrp1 =l add %code, %offp1
	%vp =l loadl %addrp1
	%soffp =l mul %sp, 8
	%saddrp =l add %stack, %soffp
	storel %vp, %saddrp
	%nsp =l add %sp, 1
	storel %nsp, %spp
	%npc =l add %pc, 2
	storel %npc, %pcp
	jmp @loop
@oadd
	%spm1a =l sub %sp, 1
	%spm2a =l sub %sp, 2
	%boffa =l mul %spm1a, 8
	%baddra =l add %stack, %boffa
	%ba =l loadl %baddra
	%aoffa =l mul %spm2a, 8
	%aaddra =l add %stack, %aoffa
	%aa =l loadl %aaddra
	%ra =l add %aa, %ba
	storel %ra, %aaddra
	storel %spm1a, %spp
	%npca =l add %pc, 1
	storel %npca, %pcp
	jmp @loop
@osub
	%spm1s =l sub %sp, 1
	%spm2s =l sub %sp, 2
	%boffs =l mul %spm1s, 8
	%baddrs =l add %stack, %boffs
	%bs =l loadl %baddrs
	%aoffs =l mul %spm2s, 8
	%aaddrs =l add %stack, %aoffs
	%as =l loadl %aaddrs
	%rs =l sub %as, %bs
	storel %rs, %aaddrs
	storel %spm1s, %spp
	%npcs =l add %pc, 1
	storel %npcs, %pcp
	jmp @loop
@omul
	%spm1m =l sub %sp, 1
	%spm2m =l sub %sp, 2
	%boffm =l mul %spm1m, 8
	%baddrm =l add %stack, %boffm
	%bm =l loadl %baddrm
	%aoffm =l mul %spm2m, 8
	%aaddrm =l add %stack, %aoffm
	%am =l loadl %aaddrm
	%rm =l mul %am, %bm
	storel %rm, %aaddrm
	storel %spm1m, %spp
	%npcm =l add %pc, 1
	storel %npcm, %pcp
	jmp @loop
@odup
	%spm1d =l sub %sp, 1
	%toffd =l mul %spm1d, 8
	%taddrd =l add %stack, %toffd
	%tvd =l loadl %taddrd
	%noffd =l mul %sp, 8
	%naddrd =l add %stack, %noffd
	storel %tvd, %naddrd
	%nspd =l add %sp, 1
	storel %nspd, %spp
	%npcd =l add %pc, 1
	storel %npcd, %pcp
	jmp @loop
@oswap
	%spm1w =l sub %sp, 1
	%spm2w =l sub %sp, 2
	%toffw =l mul %spm1w, 8
	%taddrw =l add %stack, %toffw
	%soffw =l mul %spm2w, 8
	%saddrw =l add %stack, %soffw
	%tvw =l loadl %taddrw
	%svw =l loadl %saddrw
	storel %svw, %taddrw
	storel %tvw, %saddrw
	%npcw =l add %pc, 1
	storel %npcw, %pcp
	jmp @loop
@odrop
	%spm1r =l sub %sp, 1
	storel %spm1r, %spp
	%npcr =l add %pc, 1
	storel %npcr, %pcp
	jmp @loop
@oover
	%spm2o =l sub %sp, 2
	%toffo =l mul %spm2o, 8
	%taddro =l add %stack, %toffo
	%tvo =l loadl %taddro
	%noffo =l mul %sp, 8
	%naddro =l add %stack, %noffo
	storel %tvo, %naddro
	%nspo =l add %sp, 1
	storel %nspo, %spp
	%npco =l add %pc, 1
	storel %npco, %pcp
	jmp @loop
@olt
	%spm1l =l sub %sp, 1
	%spm2l =l sub %sp, 2
	%boffl =l mul %spm1l, 8
	%baddrl =l add %stack, %boffl
	%bl =l loadl %baddrl
	%aoffl =l mul %spm2l, 8
	%aaddrl =l add %stack, %aoffl
	%al =l loadl %aaddrl
	%cmpl =w csltl %al, %bl
	%cmplL =l extuw %cmpl
	storel %cmplL, %aaddrl
	storel %spm1l, %spp
	%npcl =l add %pc, 1
	storel %npcl, %pcp
	jmp @loop
@ojz
	%spm1j =l sub %sp, 1
	%voffj =l mul %spm1j, 8
	%vaddrj =l add %stack, %voffj
	%vj =l loadl %vaddrj
	storel %spm1j, %spp
	%iszj =w ceql %vj, 0
	jnz %iszj, @jztake, @jzfall
@jztake
	%pc1j =l add %pc, 1
	%off1j =l mul %pc1j, 8
	%addr1j =l add %code, %off1j
	%tj =l loadl %addr1j
	storel %tj, %pcp
	jmp @loop
@jzfall
	%pc2j =l add %pc, 2
	storel %pc2j, %pcp
	jmp @loop
@ojmp
	%pc1jm =l add %pc, 1
	%off1jm =l mul %pc1jm, 8
	%addr1jm =l add %code, %off1jm
	%tjm =l loadl %addr1jm
	storel %tjm, %pcp
	jmp @loop
@done
	%fsp =l loadl %spp
	%hasv =w csgtl %fsp, 0
	jnz %hasv, @hasval, @zeroinit
@hasval
	%rspm =l sub %fsp, 1
	%roff =l mul %rspm, 8
	%raddr =l add %stack, %roff
	%rv =l loadl %raddr
	storel %rv, %resp
	jmp @zeroinit
@zeroinit
	storel %fsp, %ip
	jmp @zloop
@zloop
	%iz =l loadl %ip
	%cmpz =w csltl %iz, %cap
	jnz %cmpz, @zbody, @zend
@zbody
	%zoff =l mul %iz, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%iz1 =l add %iz, 1
	storel %iz1, %ip
	jmp @zloop
@zend
	%finres =l loadl %resp
	ret %finres
}
