export function l $crc32(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @loopend %inext
	%crc =w phi @start 4294967295, @loopend %crcfinal
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crcb =w xor %crc, %b
	jmp @inner
@inner
	%j =w phi @body 0, @after %jnext
	%cj =w phi @body %crcb, @after %cnext
	%icond =w csltw %j, 8
	jnz %icond, @inbody, @loopend
@inbody
	%bit =w and %cj, 1
	%shifted =w shr %cj, 1
	jnz %bit, @xorblk, @noxor
@xorblk
	%cx =w xor %shifted, 3988292384
	jmp @after
@noxor
	jmp @after
@after
	%cnext =w phi @xorblk %cx, @noxor %shifted
	%jnext =w add %j, 1
	jmp @inner
@loopend
	%crcfinal =w copy %cj
	%inext =l add %i, 1
	jmp @loop
@end
	%crcf =w xor %crc, 4294967295
	%ret =l extuw %crcf
	ret %ret
}
