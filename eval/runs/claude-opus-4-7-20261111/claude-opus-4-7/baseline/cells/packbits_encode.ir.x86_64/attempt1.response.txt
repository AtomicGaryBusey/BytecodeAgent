export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%sloti =l alloc8 8
	%sloto =l alloc8 8
	%slotk =l alloc8 8
	%slotr =l alloc8 8
	%slotj =l alloc8 8
	storel 0, %sloti
	storel 0, %sloto
	jmp @loop
@loop
	%i0 =l loadl %sloti
	%done =w csgel %i0, %n
	jnz %done, @endok, @do_run
@do_run
	%srci =l add %src, %i0
	%b =w loadub %srci
	storel %i0, %slotk
	storel 0, %slotr
	jmp @rloop
@rloop
	%k0 =l loadl %slotk
	%r0 =l loadl %slotr
	%rc1 =w csgel %k0, %n
	jnz %rc1, @rend, @rchk2
@rchk2
	%rc2 =w csgel %r0, 128
	jnz %rc2, @rend, @rchk3
@rchk3
	%srck =l add %src, %k0
	%bk =w loadub %srck
	%rc3 =w cnew %bk, %b
	jnz %rc3, @rend, @rinc
@rinc
	%k1 =l add %k0, 1
	%r1 =l add %r0, 1
	storel %k1, %slotk
	storel %r1, %slotr
	jmp @rloop
@rend
	%rfinal =l loadl %slotr
	%is_rep =w csgel %rfinal, 3
	jnz %is_rep, @do_repeat, @do_literal
@do_repeat
	%o0 =l loadl %sloto
	%o_after =l add %o0, 2
	%rfits =w csgtl %o_after, %cap
	jnz %rfits, @fail, @rep_write
@rep_write
	%hdrl =l sub 257, %rfinal
	%hdr =w copy %hdrl
	%outp =l add %out, %o0
	storeb %hdr, %outp
	%outp1 =l add %outp, 1
	storeb %b, %outp1
	storel %o_after, %sloto
	%inew =l add %i0, %rfinal
	storel %inew, %sloti
	jmp @loop
@do_literal
	storel %i0, %slotj
	jmp @lloop
@lloop
	%j0 =l loadl %slotj
	%lc1 =w csgel %j0, %n
	jnz %lc1, @lend, @lchk2
@lchk2
	%jmi =l sub %j0, %i0
	%lc2 =w csgel %jmi, 128
	jnz %lc2, @lend, @lchk3
@lchk3

	%lc3 =w csltl %jp2, %n
	jnz %lc3, @lchkeq, @lcontinue
@lchkeq

	%pj1 =l add %pj, 1
	%pj2 =l add %pj, 2
	%sj =w loadub %pj
	%sj1 =w loadub %pj1
	%sj2 =w loadub %pj2
	%eq1 =w ceqw %sj, %sj1
	%eq2 =w ceqw %sj1, %sj2
	%eqboth =w and %eq1, %eq2
	jnz %eqboth, @lend, @lcontinue
@lcontinue
	%j1 =l add %j0, 1
	storel %j1, %slotj
	jmp @lloop
@lend
	%jfinal =l loadl %slotj
	%L =l sub %jfinal, %i0
	%o0l =l loadl %sloto
	%need =l add %L, 1
	%oafl =l add %o0l, %need
	%lfits =w csgtl %oafl, %cap
	jnz %lfits, @fail, @lit_write
@lit_write
	%hdrll =l sub %L, 1
	%hdrlw =w copy %hdrll
	%outpl =l add %out, %o0l
	storeb %hdrlw, %outpl
	%dst0 =l add %outpl, 1
	%src0 =l add %src, %i0
	storel 0, %slotk
	jmp @copyloop
@copyloop
	%cnt =l loadl %slotk
	%cdone =w csgel %cnt, %L
	jnz %cdone, @copyend, @copystep
@copystep
	%srcp =l add %src0, %cnt
	%dstp =l add %dst0, %cnt
	%byte =w loadub %srcp
	storeb %byte, %dstp
	%cnt1 =l add %cnt, 1
	storel %cnt1, %slotk
	jmp @copyloop
@copyend
	storel %oafl, %sloto
	storel %jfinal, %sloti
	jmp @loop
@endok
	%ofinal =l loadl %sloto
	ret %ofinal
@fail
	ret -1
}
