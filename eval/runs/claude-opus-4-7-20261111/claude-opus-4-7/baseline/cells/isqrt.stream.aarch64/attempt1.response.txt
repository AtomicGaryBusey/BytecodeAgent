isqrt:
    mov     x1, #0                 // res = 0
    mov     x2, #1
    lsl     x2, x2, #62            // bit = 1 << 62
.Ladjust:
    cmp     x2, x0
    b.ls    .Lloop
    lsr     x2, x2, #2
    b       .Ladjust
.Lloop:
    cbz     x2, .Ldone
    add     x3, x1, x2             // res + bit
    cmp     x0, x3
    b.lo    .Lelse
    sub     x0, x0, x3             // n -= res + bit
    lsr     x1, x1, #1
    add     x1, x1, x2             // res = (res >> 1) + bit
    lsr     x2, x2, #2
    b       .Lloop
.Lelse:
    lsr     x1, x1, #1
    lsr     x2, x2, #2
    b       .Lloop
.Ldone:
    mov     x0, x1
    ret
