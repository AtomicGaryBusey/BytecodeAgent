export function l $bubble_sort(l %a, l %n) {
@start
    jmp @outer_cond
@outer_cond
    %nn =l phi @start %n, @outer_next %nn2
    %c1 =w csgtl %nn, 1
    jnz %c1, @outer_body, @end
@outer_body
    %limit =l sub %nn, 1
    jmp @inner_cond
@inner_cond
    %i =l phi @outer_body 0, @inner_next %i2
    %c2 =w csltl %i, %limit
    jnz %c2, @inner_body, @outer_next
@inner_body
    %off =l mul %i, 8
    %p1 =l add %off, %a
    %p2 =l add %p1, 8
    %v1 =l loadl %p1
    %v2 =l loadl %p2
    %c3 =w csgtl %v1, %v2
    jnz %c3, @swap, @inner_next
@swap
    storel %v2, %p1
    storel %v1, %p2
    jmp @inner_next
@inner_next
    %i2 =l add %i, 1
    jmp @inner_cond
@outer_next
    %nn2 =l sub %nn, 1
    jmp @outer_cond
@end
    ret 0
}
