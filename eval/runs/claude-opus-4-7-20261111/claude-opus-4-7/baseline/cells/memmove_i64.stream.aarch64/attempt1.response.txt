memmove_i64:
    cmp x2, #0
    b.le .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.hi .Lbackward
.Lforward:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    b.ne .Lforward
    b .Ldone
.Lbackward:
    add x1, x1, x2, lsl #3
    add x0, x0, x2, lsl #3
.Lbloop:
    ldr x3, [x1, #-8]!
    str x3, [x0, #-8]!
    subs x2, x2, #1
    b.ne .Lbloop
.Ldone:
    mov x0, #0
    ret
