export function l $collatz_max(l %n) {
@start
	%pmax =l alloc8 8
	%pk =l alloc8 8
	%px =l alloc8 8
	%pst =l alloc8 8
	storel 0, %pmax
	storel 1, %pk
	jmp @loop
@loop
	%k =l loadl %pk
	%cmp =w csgtl %k, %n
	jnz %cmp, @end, @body
@body
	storel %k, %px
	storel 0, %pst
	jmp @inner
@inner
	%x =l loadl %px
	%done =w ceql %x, 1
	jnz %done, @update, @step
@step
	%even =l and %x, 1
	%iseven =w ceql %even, 0
	jnz %iseven, @half, @triple
@half
	%xh =l shr %x, 1
	storel %xh, %px
	jmp @incstep
@triple
	%t =l mul %x, 3
	%xt =l add %t, 1
	storel %xt, %px
	jmp @incstep
@incstep
	%s =l loadl %pst
	%s2 =l add %s, 1
	storel %s2, %pst
	jmp @inner
@update
	%steps =l loadl %pst
	%max =l loadl %pmax
	%gt =w csgtl %steps, %max
	jnz %gt, @setmax, @next
@setmax
	storel %steps, %pmax
	jmp @next
@next
	%k2 =l loadl %pk
	%k3 =l add %k2, 1
	storel %k3, %pk
	jmp @loop
@end
	%result =l loadl %pmax
	ret %result
}
