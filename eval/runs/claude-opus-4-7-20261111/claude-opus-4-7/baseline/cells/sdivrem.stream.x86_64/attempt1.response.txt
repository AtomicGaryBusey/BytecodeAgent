sdivrem:
    mov rcx, rdx
    test rsi, rsi
    je .Lbzero
    cmp rsi, -1
    je .Lbminus1
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Lbzero:
    mov qword ptr [rcx], rdi
    xor eax, eax
    ret
.Lbminus1:
    mov qword ptr [rcx], 0
    mov rax, rdi
    neg rax
    ret
