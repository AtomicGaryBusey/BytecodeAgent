export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%tz =w ceql %total, 0
	jnz %tz, @ret0, @zloop_start
@ret0
	ret 0
@zloop_start
	jmp @zloop
@zloop
	%zi =l phi @zloop_start 0, @zbody %zi_next
	%zcmp =w csltl %zi, %total
	jnz %zcmp, @zbody, @outer_start
@zbody
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi_next =l add %zi, 1
	jmp @zloop
@outer_start
	jmp @outer
@outer

	%oicmp =w csltl %oi, %na
	jnz %oicmp, @inner_start, @find_start
@inner_start
	jmp @inner
@inner

	%ojcmp =w csltl %oj, %nb
	jnz %ojcmp, @mul_body, @outer_inc
@mul_body

	%ai_addr =l add %a, %ai_off
	%ai =l loadl %ai_addr
	%bj_off =l mul %oj, 8
	%bj_addr =l add %b, %bj_off
	%bj =l loadl %bj_addr

	%al =l and %ai, %mask
	%ah =l shr %ai, 32
	%bl =l and %bj, %mask
	%bh =l shr %bj, 32
	%p00 =l mul %al, %bl
	%p01 =l mul %al, %bh
	%p10 =l mul %ah, %bl
	%p11 =l mul %ah, %bh
	%p00_lo =l and %p00, %mask
	%p00_hi =l shr %p00, 32
	%p01_lo =l and %p01, %mask
	%p01_hi =l shr %p01, 32
	%p10_lo =l and %p10, %mask
	%p10_hi =l shr %p10, 32
	%t1a =l add %p00_hi, %p01_lo
	%t1 =l add %t1a, %p10_lo
	%t1_lo =l and %t1, %mask
	%t1_hi =l shr %t1, 32
	%lo_shift =l shl %t1_lo, 32
	%prod_lo =l or %lo_shift, %p00_lo

	%hi_b =l add %hi_a, %p10_hi
	%prod_hi =l add %hi_b, %t1_hi

	%new_rij =l add %rij, %prod_lo

	storel %new_rij, %rij_addr
	%c1l =l extuw %c1w
	%carry_init =l add %prod_hi, %c1l
	%k_init =l add %ij, 1
	jmp @cloop
@cloop
	%ck =l phi @mul_body %k_init, @cbody %ck_next
	%cv =l phi @mul_body %carry_init, @cbody %cv_next
	%cvz =w ceql %cv, 0
	jnz %cvz, @cloop_done, @cbody
@cbody
	%ckoff =l mul %ck, 8
	%rk_addr =l add %r, %ckoff
	%rkv =l loadl %rk_addr
	%new_rkv =l add %rkv, %cv
	%c2w =w cultl %new_rkv, %rkv
	%cv_next =l extuw %c2w
	storel %new_rkv, %rk_addr
	%ck_next =l add %ck, 1
	jmp @cloop
@cloop_done
	%oj_next =l add %oj, 1
	jmp @inner
@outer_inc
	%oi_next =l add %oi, 1
	jmp @outer
@find_start
	jmp @find_loop
@find_loop
	%sk =l phi @find_start %total, @find_dec %sk_dec
	%skz =w ceql %sk, 0
	jnz %skz, @ret_sk, @find_check
@find_check

	%skoff =l mul %skm1, 8
	%skaddr =l add %r, %skoff
	%skv =l loadl %skaddr
	%skvz =w ceql %skv, 0
	jnz %skvz, @find_dec, @ret_sk
@find_dec
	%sk_dec =l sub %sk, 1
	jmp @find_loop
@ret_sk
	ret %sk
}
