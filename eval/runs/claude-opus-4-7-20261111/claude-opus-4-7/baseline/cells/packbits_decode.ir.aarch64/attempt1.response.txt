export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %pi =l alloc8 8
    %po =l alloc8 8
    %pj =l alloc8 8
    storel 0, %pi
    storel 0, %po
    jmp @loop
@loop
    %i =l loadl %pi
    %o =l loadl %po
    %done =w csgel %i, %n
    jnz %done, @ret_ok, @read_header
@read_header
    %hp =l add %src, %i
    %h =w loadsb %hp
    %i1 =l add %i, 1
    storel %i1, %pi
    %is_nop =w ceqw %h, -128
    jnz %is_nop, @loop, @check_lit
@check_lit
    %is_lit =w csgew %h, 0
    jnz %is_lit, @literal, @repeat
@literal
    %count =w add %h, 1
    %cnt =l extsw %count
    %need =l add %i1, %cnt
    %ok =w cslel %need, %n
    jnz %ok, @lit_init, @ret_err_src
@lit_init
    storel 0, %pj
    jmp @lit_loop
@lit_loop
    %j =l loadl %pj
    %done2 =w csgel %j, %cnt
    jnz %done2, @loop, @lit_check_cap
@lit_check_cap
    %o2 =l loadl %po
    %ocheck =w csgel %o2, %cap
    jnz %ocheck, @ret_err_cap, @lit_write
@lit_write
    %i2 =l loadl %pi
    %sp2 =l add %src, %i2
    %b =w loadub %sp2
    %op2 =l add %out, %o2
    storeb %b, %op2
    %i3 =l add %i2, 1
    storel %i3, %pi
    %o3 =l add %o2, 1
    storel %o3, %po
    %j1 =l add %j, 1
    storel %j1, %pj
    jmp @lit_loop
@repeat
    %count2 =w sub 1, %h
    %cnt2 =l extsw %count2
    %need2 =l add %i1, 1
    %ok2 =w cslel %need2, %n
    jnz %ok2, @rep_read, @ret_err_src
@rep_read
    %sp3 =l add %src, %i1
    %rb =w loadub %sp3
    %i4 =l add %i1, 1
    storel %i4, %pi
    storel 0, %pj
    jmp @rep_loop
@rep_loop
    %j2 =l loadl %pj
    %done3 =w csgel %j2, %cnt2
    jnz %done3, @loop, @rep_check_cap
@rep_check_cap
    %o4 =l loadl %po
    %ocheck2 =w csgel %o4, %cap
    jnz %ocheck2, @ret_err_cap, @rep_write
@rep_write
    %op3 =l add %out, %o4
    storeb %rb, %op3
    %o5 =l add %o4, 1
    storel %o5, %po
    %j3 =l add %j2, 1
    storel %j3, %pj
    jmp @rep_loop
@ret_ok
    ret %o
@ret_err_src
    ret -2
@ret_err_cap
    ret -1
}
