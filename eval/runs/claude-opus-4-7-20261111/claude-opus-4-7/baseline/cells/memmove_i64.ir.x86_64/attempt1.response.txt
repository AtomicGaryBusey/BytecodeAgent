export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %dir =w cultl %dst, %src
    jnz %dir, @finit, @binit
@finit
    jmp @floop
@floop
    %i =l phi @finit 0, @fbody %inext
    %fcond =w csltl %i, %n
    jnz %fcond, @fbody, @done
@fbody
    %off =l mul %i, 8
    %sp =l add %src, %off
    %dp =l add %dst, %off
    %v =l loadl %sp
    storel %v, %dp
    %inext =l add %i, 1
    jmp @floop
@binit
    %jinit =l sub %n, 1
    jmp @bloop
@bloop
    %j =l phi @binit %jinit, @bbody %jnext
    %bcond =w csgel %j, 0
    jnz %bcond, @bbody, @done
@bbody
    %off2 =l mul %j, 8
    %sp2 =l add %src, %off2
    %dp2 =l add %dst, %off2
    %v2 =l loadl %sp2
    storel %v2, %dp2
    %jnext =l sub %j, 1
    jmp @bloop
@done
    ret 0
}
