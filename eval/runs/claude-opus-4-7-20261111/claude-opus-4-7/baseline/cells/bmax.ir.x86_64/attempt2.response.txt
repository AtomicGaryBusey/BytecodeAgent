export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	jmp @loop
@loop
	%best =l phi @start %first, @upd %v, @keep %bk
	%i =l phi @start 1, @upd %ni1, @keep %ni2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %best
	jnz %gt, @upd, @keep
@upd
	%ni1 =l add %i, 1
	jmp @loop
@keep
	%bk =l copy %best
	%ni2 =l add %i, 1
	jmp @loop
@end
	ret %best
}
