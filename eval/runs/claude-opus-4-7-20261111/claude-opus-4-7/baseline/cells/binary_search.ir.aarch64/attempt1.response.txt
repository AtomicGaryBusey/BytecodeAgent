export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @goright %lor, @goleft %lol
	%hi =l phi @start %n, @goright %hir, @goleft %hil
	%cmp =w csltl %lo, %hi
	jnz %cmp, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%offset =l shl %mid, 3
	%addr =l add %a, %offset
	%val =l loadl %addr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cont
@cont
	%lt =w csltl %val, %key
	jnz %lt, @goright, @goleft
@goright
	%lor =l add %mid, 1
	%hir =l copy %hi
	jmp @loop
@goleft
	%lol =l copy %lo
	%hil =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
