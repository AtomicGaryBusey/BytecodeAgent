export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @mainloop
@mainloop
    %i =l phi @start 0, @after_lit %i_al, @after_rep %i_ar, @after_nop %i_an
    %w =l phi @start 0, @after_lit %w_al, @after_rep %w_ar, @after_nop %w_an
    %cond =w csltl %i, %n
    jnz %cond, @readhdr, @done
@readhdr
    %hptr =l add %src, %i
    %h =w loadsb %hptr
    %ih =l add %i, 1
    %islit =w csgew %h, 0
    jnz %islit, @literal, @notlit
@notlit
    %isnop =w ceqw %h, -128
    jnz %isnop, @after_nop, @repeat
@after_nop
    %i_an =l copy %ih
    %w_an =l copy %w
    jmp @mainloop
@literal
    %count =w add %h, 1
    %countl =l extsw %count
    %rem =l sub %n, %ih
    %enough =w csgel %rem, %countl
    jnz %enough, @litstart, @err2
@litstart
    %litend =l add %ih, %countl
    jmp @litloop
@litloop
    %il =l phi @litstart %ih, @litbody %il_next
    %wl =l phi @litstart %w, @litbody %wl_next
    %cond_l =w csltl %il, %litend
    jnz %cond_l, @litcheck, @after_lit
@litcheck
    %capok =w csltl %wl, %cap
    jnz %capok, @litbody, @err1
@litbody
    %sp =l add %src, %il
    %b =w loadub %sp
    %op =l add %out, %wl
    storeb %b, %op
    %il_next =l add %il, 1
    %wl_next =l add %wl, 1
    jmp @litloop
@after_lit
    %i_al =l copy %il
    %w_al =l copy %wl
    jmp @mainloop
@repeat
    %rcount =w sub 1, %h
    %rcountl =l extsw %rcount
    %rem2 =l sub %n, %ih
    %enough2 =w csgel %rem2, 1
    jnz %enough2, @reprd, @err2
@reprd
    %sp2 =l add %src, %ih
    %rb =w loadub %sp2
    %i_rb =l add %ih, 1
    %repend =l add %w, %rcountl
    jmp @reploop
@reploop
    %wr =l phi @reprd %w, @repbody %wr_next
    %cond_r =w csltl %wr, %repend
    jnz %cond_r, @repcheck, @after_rep
@repcheck
    %capok2 =w csltl %wr, %cap
    jnz %capok2, @repbody, @err1
@repbody
    %op2 =l add %out, %wr
    storeb %rb, %op2
    %wr_next =l add %wr, 1
    jmp @reploop
@after_rep
    %i_ar =l copy %i_rb
    %w_ar =l copy %wr
    jmp @mainloop
@err1
    %n1 =l copy -1
    ret %n1
@err2
    %n2 =l copy -2
    ret %n2
@done
    ret %w
}
