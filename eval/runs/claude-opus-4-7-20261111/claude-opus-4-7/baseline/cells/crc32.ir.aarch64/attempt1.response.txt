export function l $crc32(l %p, l %n) {
@start
	jmp @byteloop
@byteloop
	%crc =w phi @start -1, @bitend %crcnew
	%i =l phi @start 0, @bitend %inext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc0 =w xor %crc, %b
	jmp @bitloop
@bitloop
	%crc1 =w phi @body %crc0, @bitstep %crc2
	%j =w phi @body 0, @bitstep %jnext
	%bcond =w csltw %j, 8
	jnz %bcond, @bitstep, @bitend
@bitstep
	%lsb =w and %crc1, 1
	%negl =w sub 0, %lsb
	%mv =w and %negl, 3988292384
	%shifted =w shr %crc1, 1
	%crc2 =w xor %shifted, %mv
	%jnext =w add %j, 1
	jmp @bitloop
@bitend
	%crcnew =w copy %crc1
	%inext =l add %i, 1
	jmp @byteloop
@end
	%crcfinal =w xor %crc, -1
	%r =l extuw %crcfinal
	ret %r
}
