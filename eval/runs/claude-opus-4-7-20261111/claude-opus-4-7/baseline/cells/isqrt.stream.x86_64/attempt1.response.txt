isqrt:
    xor rax, rax
    mov rcx, 1
    shl rcx, 62
.Lloop:
    test rcx, rcx
    jz .Ldone
    mov rdx, rax
    add rdx, rcx
    cmp rdi, rdx
    jb .Lskip
    sub rdi, rdx
    shr rax, 1
    add rax, rcx
    shr rcx, 2
    jmp .Lloop
.Lskip:
    shr rax, 1
    shr rcx, 2
    jmp .Lloop
.Ldone:
    ret
