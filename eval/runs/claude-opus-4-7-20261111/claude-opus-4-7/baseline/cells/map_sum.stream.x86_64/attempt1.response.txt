map_sum:
    push rbx
    push r12
    push r13
    push r14
    sub rsp, 8
    xor ebx, ebx
    test rsi, rsi
    jle .Ldone
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
.Lloop:
    mov rdi, qword ptr [r12]
    call r14
    add rbx, rax
    add r12, 8
    dec r13
    jnz .Lloop
.Ldone:
    mov rax, rbx
    add rsp, 8
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
