    cmp rsi, 2
    jl .Ldone
    mov rcx, rsi
    dec rcx
    xor edx, edx
.Lloop:
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]

    setg r10b

    cmp rdx, rcx

.Ldone:
