export function l $shift_sat(l %x, l %n) {
@start
	%big =w csgel %n, 64
	jnz %big, @bigshift, @smallshift
@smallshift
	%ls =l shl %x, %n
	%rs =l sar %x, %n
	jmp @join
@bigshift
	%lb =l copy 0
	%neg =w csltl %x, 0
	%negl =l extsw %neg
	%rb =l sub 0, %negl
	jmp @join
@join
	%left =l phi @smallshift %ls, @bigshift %lb
	%right =l phi @smallshift %rs, @bigshift %rb
	%sum =l add %left, %right
	ret %sum
}
