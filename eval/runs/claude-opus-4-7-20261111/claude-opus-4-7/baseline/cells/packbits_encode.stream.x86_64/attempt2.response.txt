packbits_encode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r13, rsi
    mov rbp, rdx
    mov r15, rcx
    xor r12, r12
    xor r14, r14
.Lmain:
    cmp r12, r13
    jge .Ldone
    mov al, byte ptr [rbx + r12]
    mov r8, r12
    mov r9, r12
    add r9, 128
    cmp r9, r13
    jle .Lmaxok
    mov r9, r13
.Lmaxok:
.Lruncount:
    cmp r8, r9
    jge .Lrundone
    cmp byte ptr [rbx + r8], al
    jne .Lrundone
    inc r8
    jmp .Lruncount
.Lrundone:
    mov rcx, r8
    sub rcx, r12
    cmp rcx, 3
    jl .Lliteral
    mov r9, r14
    add r9, 2
    cmp r9, r15
    jg .Lfail
    mov r10d, 257
    sub r10d, ecx
    mov byte ptr [rbp + r14], r10b
    mov byte ptr [rbp + r14 + 1], al
    add r14, 2
    add r12, rcx
    jmp .Lmain
.Lliteral:
    mov r8, r12
.Llitloop:
    cmp r8, r13
    jge .Llitdone
    mov r9, r8
    sub r9, r12
    cmp r9, 128
    jge .Llitdone
    mov r9, r8
    add r9, 2
    cmp r9, r13
    jge .Llitadv
    mov r10b, byte ptr [rbx + r8]
    cmp r10b, byte ptr [rbx + r8 + 1]
    jne .Llitadv
    cmp r10b, byte ptr [rbx + r8 + 2]
    jne .Llitadv
    jmp .Llitdone
.Llitadv:
    inc r8
    jmp .Llitloop
.Llitdone:
    mov rcx, r8
    sub rcx, r12
    mov r9, r14
    add r9, rcx
    add r9, 1
    cmp r9, r15
    jg .Lfail
    mov r10, rcx
    sub r10, 1
    mov byte ptr [rbp + r14], r10b
    add r14, 1
    lea rdi, [rbx + r12]
    lea rsi, [rbp + r14]
    xor r9, r9
.Lcopyloop:
    cmp r9, rcx
    jge .Lcopydone
    mov r10b, byte ptr [rdi + r9]
    mov byte ptr [rsi + r9], r10b
    inc r9
    jmp .Lcopyloop
.Lcopydone:
    add r14, rcx
    add r12, rcx
    jmp .Lmain
.Ldone:
    mov rax, r14
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lfail:
    mov rax, -1
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
