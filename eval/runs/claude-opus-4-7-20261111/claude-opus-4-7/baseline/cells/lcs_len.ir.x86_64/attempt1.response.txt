export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_check
@init_check
	%j1 =l phi @start 0, @init_body %j1n
	%cond1 =w csgtl %j1, %n
	jnz %cond1, @init_done, @init_body
@init_body
	%off1 =l mul %j1, 8
	%addr1 =l add %row, %off1
	storel 0, %addr1
	%j1n =l add %j1, 1
	jmp @init_check
@init_done
	jmp @outer_check
@outer_check
	%i =l phi @init_done 0, @inner_done %in
	%c1 =w csgel %i, %m
	jnz %c1, @end, @outer_body
@outer_body
	%xi_addr =l add %x, %i
	%xi =w loadub %xi_addr
	jmp @inner_check
@inner_check
	%j =l phi @outer_body 1, @next_j %jn
	%diag =l phi @outer_body 0, @next_j %up
	%c2 =w csgtl %j, %n
	jnz %c2, @inner_done, @inner_body
@inner_body
	%joff =l mul %j, 8
	%rj_addr =l add %row, %joff
	%up =l loadl %rj_addr
	%jm1 =l sub %j, 1
	%yjm1_addr =l add %y, %jm1
	%yjm1 =w loadub %yjm1_addr
	%eq =w ceqw %xi, %yjm1
	jnz %eq, @match, @nomatch
@match
	%nv =l add %diag, 1
	storel %nv, %rj_addr
	jmp @next_j
@nomatch
	%rjm1_off =l sub %joff, 8
	%rjm1_addr =l add %row, %rjm1_off
	%left =l loadl %rjm1_addr
	%gt =w csgtl %up, %left
	jnz %gt, @use_up, @use_left
@use_up
	storel %up, %rj_addr
	jmp @next_j
@use_left
	storel %left, %rj_addr
	jmp @next_j
@next_j
	%jn =l add %j, 1
	jmp @inner_check
@inner_done
	%in =l add %i, 1
	jmp @outer_check
@end
	%noff =l mul %n, 8
	%naddr =l add %row, %noff
	%result =l loadl %naddr
	ret %result
}
