export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%cond =w csgtl %nm1, 0
	jnz %cond, @loop, @end
@loop
	%i =l phi @start 0, @loop %inext
	%count =l phi @start 0, @loop %cnext
	%off =l mul %i, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	%incl =l extuw %gt
	%cnext =l add %count, %incl
	%inext =l add %i, 1
	%lc =w csltl %inext, %nm1
	jnz %lc, @loop, @end
@end
	%result =l phi @start 0, @loop %cnext
	ret %result
}
