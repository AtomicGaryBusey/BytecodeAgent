export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %inext
	%max =l phi @start 0, @cont %maxnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%cmp =w cugtl %b, %max
	jnz %cmp, @update, @nokeep
@update
	jmp @cont
@nokeep
	jmp @cont
@cont
	%maxnext =l phi @update %b, @nokeep %max
	%inext =l add %i, 1
	jmp @loop
@end
	ret %max
}
