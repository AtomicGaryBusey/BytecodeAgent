export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	jmp @outer_cond
@outer_cond
	%i =l phi @start 0, @outer_inc %i2
	%c1 =w csltl %i, %nm1
	jnz %c1, @outer_body, @done
@outer_body
	%lim =l sub %nm1, %i
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body 0, @inner_inc %j2
	%c2 =w csltl %j, %lim
	jnz %c2, @inner_body, @outer_inc
@inner_body
	%off =l mul %j, 8
	%p1 =l add %a, %off
	%p2 =l add %p1, 8
	%v1 =l loadl %p1
	%v2 =l loadl %p2
	%c3 =w csgtl %v1, %v2
	jnz %c3, @swap, @inner_inc
@swap
	storel %v2, %p1
	storel %v1, %p2
	jmp @inner_inc
@inner_inc
	%j2 =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i2 =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
