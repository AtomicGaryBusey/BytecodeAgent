bubble_sort:
    mov x0, #0
    cmp x1, #2
    b.lt .Ldone
    mov x2, #0
.Louter:
    sub x3, x1, #1
    sub x3, x3, x2
    mov x4, #0
    cmp x3, #0
    b.le .Ldone
.Linner:
    cmp x4, x3
    b.ge .Lnext
    ldr x5, [x0, x4, lsl #3]
    add x6, x4, #1
    ldr x7, [x0, x6, lsl #3]
    cmp x5, x7
    b.le .Lskip
    str x7, [x0, x4, lsl #3]
    str x5, [x0, x6, lsl #3]
.Lskip:
    add x4, x4, #1
    b .Linner
.Lnext:
    add x2, x2, #1
    cmp x2, x1
    b.lt .Louter
.Ldone:
    mov x0, #0
    ret
