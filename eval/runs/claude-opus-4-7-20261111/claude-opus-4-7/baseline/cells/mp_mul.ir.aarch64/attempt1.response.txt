export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %total =l add %na, %nb
    %iszero =w ceql %total, 0
    jnz %iszero, @ret0, @init
@ret0
    ret 0
@init
    jmp @zloop
@zloop
    %zi =l phi @init 0, @zbody %zi1
    %zdone =w cslel %total, %zi
    jnz %zdone, @mulinit, @zbody
@zbody
    %zp =l mul %zi, 8
    %zaddr =l add %r, %zp
    storel 0, %zaddr
    %zi1 =l add %zi, 1
    jmp @zloop
@mulinit
    jmp @outer
@outer
    %i =l phi @mulinit 0, @oend %i1
    %odone =w cslel %na, %i
    jnz %odone, @findk, @obody
@obody
    %ioff =l mul %i, 8
    %aaddr =l add %a, %ioff
    %ai =l loadl %aaddr
    jmp @inner
@inner
    %j =l phi @obody 0, @ibody %j1
    %carry =l phi @obody 0, @ibody %ncarry
    %idone =w cslel %nb, %j
    jnz %idone, @oend, @ibody
@ibody
    %joff =l mul %j, 8
    %baddr =l add %b, %joff
    %bj =l loadl %baddr
    %aL =l and %ai, 4294967295
    %aH =l shr %ai, 32
    %bL =l and %bj, 4294967295
    %bH =l shr %bj, 32
    %p00 =l mul %aL, %bL
    %p01 =l mul %aL, %bH
    %p10 =l mul %aH, %bL
    %p11 =l mul %aH, %bH
    %p01lo =l shl %p01, 32
    %p01hi =l shr %p01, 32
    %p10lo =l shl %p10, 32
    %p10hi =l shr %p10, 32
    %x =l add %p00, %p01lo
    %c1w =w cultl %x, %p00
    %c1 =l extuw %c1w
    %y =l add %x, %p10lo
    %c2w =w cultl %y, %x
    %c2 =l extuw %c2w
    %lo =l copy %y
    %hi0 =l add %p11, %p01hi
    %hi1 =l add %hi0, %p10hi
    %hi2 =l add %hi1, %c1
    %hi =l add %hi2, %c2
    %ij =l add %i, %j
    %ijoff =l mul %ij, 8
    %raddr =l add %r, %ijoff
    %rij =l loadl %raddr
    %s1 =l add %rij, %lo
    %cs1w =w cultl %s1, %rij
    %cs1 =l extuw %cs1w
    %s2 =l add %s1, %carry
    %cs2w =w cultl %s2, %s1
    %cs2 =l extuw %cs2w
    storel %s2, %raddr
    %ncarry0 =l add %hi, %cs1
    %ncarry =l add %ncarry0, %cs2
    %j1 =l add %j, 1
    jmp @inner
@oend
    %ij2 =l add %i, %nb
    %ij2off =l mul %ij2, 8
    %raddr2 =l add %r, %ij2off
    storel %carry, %raddr2
    %i1 =l add %i, 1
    jmp @outer
@findk
    jmp @kloop
@kloop
    %k =l phi @findk %total, @kdec %km1
    %kiszero =w ceql %k, 0
    jnz %kiszero, @kret, @kbody
@kbody
    %km1 =l sub %k, 1
    %kmoff =l mul %km1, 8
    %kaddr =l add %r, %kmoff
    %kv =l loadl %kaddr
    %kvz =w ceql %kv, 0
    jnz %kvz, @kdec, @kret
@kdec
    jmp @kloop
@kret
    ret %k
}
