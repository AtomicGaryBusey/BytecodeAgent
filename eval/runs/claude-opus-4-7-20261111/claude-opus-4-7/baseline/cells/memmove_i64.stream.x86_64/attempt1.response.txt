memmove_i64:
  test rdx, rdx
  jz .Ldone
  cmp rdi, rsi
  je .Ldone
  jb .Lforward
  mov rcx, rdx
.Lbackloop:
  mov rax, qword ptr [rsi + rcx*8 - 8]
  mov qword ptr [rdi + rcx*8 - 8], rax
  dec rcx
  jnz .Lbackloop
  jmp .Ldone
.Lforward:
  xor rcx, rcx
.Lfwdloop:
  mov rax, qword ptr [rsi + rcx*8]
  mov qword ptr [rdi + rcx*8], rax
  inc rcx
  cmp rcx, rdx
  jb .Lfwdloop
.Ldone:
  xor rax, rax
  ret
