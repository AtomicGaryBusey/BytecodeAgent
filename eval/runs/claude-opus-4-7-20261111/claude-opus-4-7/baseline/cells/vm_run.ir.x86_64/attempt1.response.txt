export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	%resp =l alloc8 8
	%izp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	storel 0, %resp
	jmp @loop
@loop
	%pc0 =l loadl %pcp
	%cond0 =w csltl %pc0, %len
	jnz %cond0, @body, @done
@body
	%pc1 =l loadl %pcp
	%ipa1 =l mul %pc1, 8
	%ipa2 =l add %ipa1, %code
	%op =l loadl %ipa2
	%cc0 =w ceql %op, 0
	jnz %cc0, @done, @chk1
@chk1
	%cc1 =w ceql %op, 1
	jnz %cc1, @op_push, @chk2
@chk2
	%cc2 =w ceql %op, 2
	jnz %cc2, @op_add, @chk3
@chk3
	%cc3 =w ceql %op, 3
	jnz %cc3, @op_sub, @chk4
@chk4
	%cc4 =w ceql %op, 4
	jnz %cc4, @op_mul, @chk5
@chk5
	%cc5 =w ceql %op, 5
	jnz %cc5, @op_dup, @chk6
@chk6
	%cc6 =w ceql %op, 6
	jnz %cc6, @op_swap, @chk7
@chk7
	%cc7 =w ceql %op, 7
	jnz %cc7, @op_drop, @chk8
@chk8
	%cc8 =w ceql %op, 8
	jnz %cc8, @op_over, @chk9
@chk9
	%cc9 =w ceql %op, 9
	jnz %cc9, @op_lt, @chk10
@chk10
	%cc10 =w ceql %op, 10
	jnz %cc10, @op_jz, @chk11
@chk11
	jmp @op_jmp
@op_push
	%ppc =l loadl %pcp
	%pp1 =l add %ppc, 1
	%paa1 =l mul %pp1, 8
	%paa2 =l add %paa1, %code
	%pv =l loadl %paa2
	%psp =l loadl %spp
	%pss1 =l mul %psp, 8
	%pss2 =l add %pss1, %stack
	storel %pv, %pss2
	%pspn =l add %psp, 1
	storel %pspn, %spp
	%ppcn =l add %ppc, 2
	storel %ppcn, %pcp
	jmp @loop
@op_add
	%asp =l loadl %spp
	%asp1 =l sub %asp, 1
	%aba1 =l mul %asp1, 8
	%aba2 =l add %aba1, %stack
	%ab =l loadl %aba2
	%asp2 =l sub %asp1, 1
	%aaa1 =l mul %asp2, 8
	%aaa2 =l add %aaa1, %stack
	%aa =l loadl %aaa2
	%ar =l add %aa, %ab
	storel %ar, %aaa2
	storel %asp1, %spp
	%apc =l loadl %pcp
	%apcn =l add %apc, 1
	storel %apcn, %pcp
	jmp @loop
@op_sub
	%ssp =l loadl %spp
	%ssp1 =l sub %ssp, 1
	%sba1 =l mul %ssp1, 8
	%sba2 =l add %sba1, %stack
	%sb =l loadl %sba2
	%ssp2 =l sub %ssp1, 1
	%saa1 =l mul %ssp2, 8
	%saa2 =l add %saa1, %stack
	%sa =l loadl %saa2
	%sr =l sub %sa, %sb
	storel %sr, %saa2
	storel %ssp1, %spp
	%spc =l loadl %pcp
	%spcn =l add %spc, 1
	storel %spcn, %pcp
	jmp @loop
@op_mul
	%msp =l loadl %spp
	%msp1 =l sub %msp, 1
	%mba1 =l mul %msp1, 8
	%mba2 =l add %mba1, %stack
	%mb =l loadl %mba2
	%msp2 =l sub %msp1, 1
	%maa1 =l mul %msp2, 8
	%maa2 =l add %maa1, %stack
	%ma =l loadl %maa2
	%mr =l mul %ma, %mb
	storel %mr, %maa2
	storel %msp1, %spp
	%mpc =l loadl %pcp
	%mpcn =l add %mpc, 1
	storel %mpcn, %pcp
	jmp @loop
@op_dup
	%dsp =l loadl %spp
	%dspm1 =l sub %dsp, 1
	%daa1 =l mul %dspm1, 8
	%daa2 =l add %daa1, %stack
	%dv =l loadl %daa2
	%dsa1 =l mul %dsp, 8
	%dsa2 =l add %dsa1, %stack
	storel %dv, %dsa2
	%dspn =l add %dsp, 1
	storel %dspn, %spp
	%dpc =l loadl %pcp
	%dpcn =l add %dpc, 1
	storel %dpcn, %pcp
	jmp @loop
@op_swap
	%wsp =l loadl %spp
	%wspm1 =l sub %wsp, 1
	%waa1 =l mul %wspm1, 8
	%waa2 =l add %waa1, %stack
	%wspm2 =l sub %wsp, 2
	%wba1 =l mul %wspm2, 8
	%wba2 =l add %wba1, %stack
	%wva =l loadl %waa2
	%wvb =l loadl %wba2
	storel %wvb, %waa2
	storel %wva, %wba2
	%wpc =l loadl %pcp
	%wpcn =l add %wpc, 1
	storel %wpcn, %pcp
	jmp @loop
@op_drop
	%ksp =l loadl %spp
	%kspn =l sub %ksp, 1
	storel %kspn, %spp
	%kpc =l loadl %pcp
	%kpcn =l add %kpc, 1
	storel %kpcn, %pcp
	jmp @loop
@op_over
	%osp =l loadl %spp
	%ospm2 =l sub %osp, 2
	%oba1 =l mul %ospm2, 8
	%oba2 =l add %oba1, %stack
	%ov =l loadl %oba2
	%osa1 =l mul %osp, 8
	%osa2 =l add %osa1, %stack
	storel %ov, %osa2
	%ospn =l add %osp, 1
	storel %ospn, %spp
	%opc =l loadl %pcp
	%opcn =l add %opc, 1
	storel %opcn, %pcp
	jmp @loop
@op_lt
	%lsp =l loadl %spp
	%lsp1 =l sub %lsp, 1
	%lba1 =l mul %lsp1, 8
	%lba2 =l add %lba1, %stack
	%lb =l loadl %lba2
	%lsp2 =l sub %lsp1, 1
	%laa1 =l mul %lsp2, 8
	%laa2 =l add %laa1, %stack
	%la =l loadl %laa2
	%ltw =w csltl %la, %lb
	%lrl =l extuw %ltw
	storel %lrl, %laa2
	storel %lsp1, %spp
	%lpc =l loadl %pcp
	%lpcn =l add %lpc, 1
	storel %lpcn, %pcp
	jmp @loop
@op_jz
	%jsp =l loadl %spp
	%jspn =l sub %jsp, 1
	%jva1 =l mul %jspn, 8
	%jva2 =l add %jva1, %stack
	%jv =l loadl %jva2
	storel %jspn, %spp
	%jiz =w ceql %jv, 0
	jnz %jiz, @jz_yes, @jz_no
@jz_yes
	%jypc =l loadl %pcp
	%jyp1 =l add %jypc, 1
	%jya1 =l mul %jyp1, 8
	%jya2 =l add %jya1, %code
	%jyt =l loadl %jya2
	storel %jyt, %pcp
	jmp @loop
@jz_no
	%jnpc =l loadl %pcp
	%jnpcn =l add %jnpc, 2
	storel %jnpcn, %pcp
	jmp @loop
@op_jmp
	%mjpc =l loadl %pcp
	%mjp1 =l add %mjpc, 1
	%mja1 =l mul %mjp1, 8
	%mja2 =l add %mja1, %code
	%mjt =l loadl %mja2
	storel %mjt, %pcp
	jmp @loop
@done
	%fsp =l loadl %spp
	storel %fsp, %izp
	%fspz =w ceql %fsp, 0
	jnz %fspz, @zero_loop, @result_val
@result_val
	%frm =l sub %fsp, 1
	%fra1 =l mul %frm, 8
	%fra2 =l add %fra1, %stack
	%fr =l loadl %fra2
	storel %fr, %resp
	jmp @zero_loop
@zero_loop
	%iz =l loadl %izp
	%zc =w csltl %iz, %cap
	jnz %zc, @zero_do, @ret
@zero_do
	%za1 =l mul %iz, 8
	%za2 =l add %za1, %stack
	storel 0, %za2
	%izn =l add %iz, 1
	storel %izn, %izp
	jmp @zero_loop
@ret
	%res =l loadl %resp
	ret %res
}
