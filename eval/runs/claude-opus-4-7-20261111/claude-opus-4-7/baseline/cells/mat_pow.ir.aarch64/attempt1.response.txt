export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %nnb =l mul %nn, 8
    %tmp2 =l add %tmp, %nnb
    jmp @init_test
@init_test
    %ii =l phi @start 0, @init_body %ii_n
    %c_i =w csltl %ii, %nn
    jnz %c_i, @init_body, @init_done
@init_body
    %off1 =l mul %ii, 8
    %pa1 =l add %a, %off1
    %pout1 =l add %out, %off1
    %ptmp1 =l add %tmp, %off1
    %ptmp21 =l add %tmp2, %off1
    %va1 =l loadl %pa1
    storel 0, %pout1
    storel %va1, %ptmp1
    storel 0, %ptmp21
    %ii_n =l add %ii, 1
    jmp @init_test
@init_done
    jmp @id_test
@id_test
    %di =l phi @init_done 0, @id_body %di_n
    %c_d =w csltl %di, %n
    jnz %c_d, @id_body, @id_done
@id_body
    %dia =l mul %di, %n
    %diab =l add %dia, %di
    %doff =l mul %diab, 8
    %dp =l add %out, %doff
    storel 1, %dp
    %di_n =l add %di, 1
    jmp @id_test
@id_done
    jmp @loop_head
@loop_head
    %kk =l phi @id_done %k, @after_square %kk_half
    %c_k0 =w ceql %kk, 0
    jnz %c_k0, @final, @loop_body
@loop_body
    %kbit =l and %kk, 1
    %c_bit =w ceql %kbit, 0
    jnz %c_bit, @after_mul, @m1_i_test
@m1_i_test
    %m1i =l phi @loop_body 0, @m1_i_cont %m1i_n
    %c_m1i =w csltl %m1i, %n
    jnz %c_m1i, @m1_i_body, @m1_i_done
@m1_i_body
    jmp @m1_j_test
@m1_j_test
    %m1j =l phi @m1_i_body 0, @m1_j_store %m1j_n
    %c_m1j =w csltl %m1j, %n
    jnz %c_m1j, @m1_j_body, @m1_i_cont
@m1_j_body
    jmp @m1_t_test
@m1_t_test
    %m1t =l phi @m1_j_body 0, @m1_t_body %m1t_n
    %m1s =l phi @m1_j_body 0, @m1_t_body %m1s_n
    %c_m1t =w csltl %m1t, %n
    jnz %c_m1t, @m1_t_body, @m1_j_store
@m1_t_body
    %m1_ia =l mul %m1i, %n
    %m1_iat =l add %m1_ia, %m1t
    %m1_offA =l mul %m1_iat, 8
    %m1_pA =l add %out, %m1_offA
    %m1_vA =l loadl %m1_pA
    %m1_tb =l mul %m1t, %n
    %m1_tbj =l add %m1_tb, %m1j
    %m1_offB =l mul %m1_tbj, 8
    %m1_pB =l add %tmp, %m1_offB
    %m1_vB =l loadl %m1_pB
    %m1_prod =l mul %m1_vA, %m1_vB
    %m1s_n =l add %m1s, %m1_prod
    %m1t_n =l add %m1t, 1
    jmp @m1_t_test
@m1_j_store
    %m1_ic =l mul %m1i, %n
    %m1_icj =l add %m1_ic, %m1j
    %m1_offC =l mul %m1_icj, 8
    %m1_pC =l add %tmp2, %m1_offC
    storel %m1s, %m1_pC
    %m1j_n =l add %m1j, 1
    jmp @m1_j_test
@m1_i_cont
    %m1i_n =l add %m1i, 1
    jmp @m1_i_test
@m1_i_done
    jmp @m1_copy_test
@m1_copy_test
    %mc1 =l phi @m1_i_done 0, @m1_copy_body %mc1_n
    %c_mc1 =w csltl %mc1, %nn
    jnz %c_mc1, @m1_copy_body, @after_mul
@m1_copy_body
    %mc1_off =l mul %mc1, 8
    %mc1_pt2 =l add %tmp2, %mc1_off
    %mc1_pout =l add %out, %mc1_off
    %mc1_v =l loadl %mc1_pt2
    storel %mc1_v, %mc1_pout
    storel 0, %mc1_pt2
    %mc1_n =l add %mc1, 1
    jmp @m1_copy_test
@after_mul
    %kk_half =l shr %kk, 1
    %c_kh0 =w ceql %kk_half, 0
    jnz %c_kh0, @final, @m2_i_test
@m2_i_test
    %m2i =l phi @after_mul 0, @m2_i_cont %m2i_n
    %c_m2i =w csltl %m2i, %n
    jnz %c_m2i, @m2_i_body, @m2_i_done
@m2_i_body
    jmp @m2_j_test
@m2_j_test
    %m2j =l phi @m2_i_body 0, @m2_j_store %m2j_n
    %c_m2j =w csltl %m2j, %n
    jnz %c_m2j, @m2_j_body, @m2_i_cont
@m2_j_body
    jmp @m2_t_test
@m2_t_test
    %m2t =l phi @m2_j_body 0, @m2_t_body %m2t_n
    %m2s =l phi @m2_j_body 0, @m2_t_body %m2s_n
    %c_m2t =w csltl %m2t, %n
    jnz %c_m2t, @m2_t_body, @m2_j_store
@m2_t_body
    %m2_ia =l mul %m2i, %n
    %m2_iat =l add %m2_ia, %m2t
    %m2_offA =l mul %m2_iat, 8
    %m2_pA =l add %tmp, %m2_offA
    %m2_vA =l loadl %m2_pA
    %m2_tb =l mul %m2t, %n
    %m2_tbj =l add %m2_tb, %m2j
    %m2_offB =l mul %m2_tbj, 8
    %m2_pB =l add %tmp, %m2_offB
    %m2_vB =l loadl %m2_pB
    %m2_prod =l mul %m2_vA, %m2_vB
    %m2s_n =l add %m2s, %m2_prod
    %m2t_n =l add %m2t, 1
    jmp @m2_t_test
@m2_j_store
    %m2_ic =l mul %m2i, %n
    %m2_icj =l add %m2_ic, %m2j
    %m2_offC =l mul %m2_icj, 8
    %m2_pC =l add %tmp2, %m2_offC
    storel %m2s, %m2_pC
    %m2j_n =l add %m2j, 1
    jmp @m2_j_test
@m2_i_cont
    %m2i_n =l add %m2i, 1
    jmp @m2_i_test
@m2_i_done
    jmp @m2_copy_test
@m2_copy_test
    %mc2 =l phi @m2_i_done 0, @m2_copy_body %mc2_n
    %c_mc2 =w csltl %mc2, %nn
    jnz %c_mc2, @m2_copy_body, @after_square
@m2_copy_body
    %mc2_off =l mul %mc2, 8
    %mc2_pt2 =l add %tmp2, %mc2_off
    %mc2_ptmp =l add %tmp, %mc2_off
    %mc2_v =l loadl %mc2_pt2
    storel %mc2_v, %mc2_ptmp
    storel 0, %mc2_pt2
    %mc2_n =l add %mc2, 1
    jmp @m2_copy_test
@after_square
    jmp @loop_head
@final
    jmp @fz_test
@fz_test
    %fz =l phi @final 0, @fz_body %fz_n
    %c_fz =w csltl %fz, %nn
    jnz %c_fz, @fz_body, @fz_done
@fz_body
    %fz_off =l mul %fz, 8
    %fz_p =l add %tmp, %fz_off
    storel 0, %fz_p
    %fz_n =l add %fz, 1
    jmp @fz_test
@fz_done
    ret 0
}
