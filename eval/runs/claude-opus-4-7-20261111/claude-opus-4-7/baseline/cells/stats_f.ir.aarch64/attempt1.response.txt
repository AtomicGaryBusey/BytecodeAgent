export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %y0v =l loadl %a
    %y =l call %f(l %y0v)
    %odd0 =l and %y, 1
    jmp @loop
@loop
    %i =l phi @start 1, @cont %inext
    %min =l phi @start %y, @cont %minfinal
    %max =l phi @start %y, @cont %maxfinal
    %sum =l phi @start %y, @cont %sumnew
    %odd =l phi @start %odd0, @cont %oddnew
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %off =l mul %i, 8
    %ap =l add %a, %off
    %x =l loadl %ap
    %yi =l call %f(l %x)
    %lt =w csltl %yi, %min
    jnz %lt, @setmin, @checkmax
@setmin
    jmp @checkmax
@checkmax
    %mintmp =l phi @body %min, @setmin %yi
    %gt =w csgtl %yi, %max
    jnz %gt, @setmax, @aftermax
@setmax
    jmp @aftermax
@aftermax
    %maxtmp =l phi @checkmax %max, @setmax %yi
    jmp @cont
@cont
    %minfinal =l copy %mintmp
    %maxfinal =l copy %maxtmp
    %sumnew =l add %sum, %yi
    %bit =l and %yi, 1
    %oddnew =l add %odd, %bit
    %inext =l add %i, 1
    jmp @loop
@end
    storel %min, %out
    %o1 =l add %out, 8
    storel %max, %o1
    %o2 =l add %out, 16
    storel %sum, %o2
    ret %odd
}
