export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @aftermax %ni
	%sum =l phi @start %y0, @aftermax %nsum
	%cnt =l phi @start %odd0, @aftermax %ncnt
	%min =l phi @start %y0, @aftermax %nmin
	%max =l phi @start %y0, @aftermax %nmax
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %off, %a
	%x =l loadl %addr
	%y =l call %f(l %x)
	%nsum =l add %sum, %y
	%odd =l and %y, 1
	%ncnt =l add %cnt, %odd
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @nomin
@setmin
	jmp @aftermin
@nomin
	jmp @aftermin
@aftermin
	%nmin =l phi @setmin %y, @nomin %min
	%gt =w csltl %max, %y
	jnz %gt, @setmax, @nomax
@setmax
	jmp @aftermax
@nomax
	jmp @aftermax
@aftermax
	%nmax =l phi @setmax %y, @nomax %max
	%ni =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
