qsort_pairs:
    cmp rsi, 1
    jle .Lret0
    push rbx
    push rbp
    push r12
    mov rbx, rdi
    mov rbp, rsi
    mov rax, rbp
    shl rax, 4
    sub rax, 16
    mov r10, qword ptr [rbx + rax]
    mov r11, qword ptr [rbx + rax + 8]
    xor r8, r8
    xor r9, r9
.Lpart_loop:
    cmp r9, rax
    jge .Lpart_done
    mov rcx, qword ptr [rbx + r9]
    mov rdx, qword ptr [rbx + r9 + 8]
    cmp rcx, r10
    jl .Lpart_less
    jg .Lpart_next
    cmp rdx, r11
    jge .Lpart_next
.Lpart_less:
    mov rdi, qword ptr [rbx + r8]
    mov rsi, qword ptr [rbx + r8 + 8]
    mov qword ptr [rbx + r8], rcx
    mov qword ptr [rbx + r8 + 8], rdx
    mov qword ptr [rbx + r9], rdi
    mov qword ptr [rbx + r9 + 8], rsi
    add r8, 16
.Lpart_next:
    add r9, 16
    jmp .Lpart_loop
.Lpart_done:
    mov rdi, qword ptr [rbx + r8]
    mov rsi, qword ptr [rbx + r8 + 8]
    mov qword ptr [rbx + r8], r10
    mov qword ptr [rbx + r8 + 8], r11
    mov qword ptr [rbx + rax], rdi
    mov qword ptr [rbx + rax + 8], rsi
    shr r8, 4
    mov r12, r8
    mov rdi, rbx
    mov rsi, r12
    call qsort_pairs
    mov rdi, r12
    shl rdi, 4
    add rdi, 16
    add rdi, rbx
    mov rsi, rbp
    sub rsi, r12
    sub rsi, 1
    call qsort_pairs
    xor rax, rax
    pop r12
    pop rbp
    pop rbx
    ret
.Lret0:
    xor rax, rax
    ret
