export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @ret, @work
@work
	%mid =l shr %n, 1
	%last =l sub %n, 1
	%midoff =l mul %mid, 16
	%pmid =l add %a, %midoff
	%pmidv =l add %pmid, 8
	%lastoff =l mul %last, 16
	%plast =l add %a, %lastoff
	%plastv =l add %plast, 8
	%t1 =l loadl %pmid
	%t2 =l loadl %pmidv
	%t3 =l loadl %plast
	%t4 =l loadl %plastv
	storel %t3, %pmid
	storel %t4, %pmidv
	storel %t1, %plast
	storel %t2, %plastv
	%pk =l loadl %plast
	%pv =l loadl %plastv
	jmp @loop
@loop
	%i =l phi @work -1, @nextiter %inew
	%j =l phi @work 0, @nextiter %jinc
	%cond =w csltl %j, %last
	jnz %cond, @body, @loopend
@body
	%joff =l mul %j, 16
	%pj =l add %a, %joff
	%pjv =l add %pj, 8
	%jk =l loadl %pj
	%jv =l loadl %pjv
	%klt =w csltl %jk, %pk
	jnz %klt, @doswap, @chkeq
@chkeq
	%keq =w ceql %jk, %pk
	jnz %keq, @chkv, @noswap
@chkv
	%vle =w cslel %jv, %pv
	jnz %vle, @doswap, @noswap
@doswap
	%iinc =l add %i, 1
	%ioff =l mul %iinc, 16
	%pi =l add %a, %ioff
	%piv =l add %pi, 8
	%sk =l loadl %pi
	%sv =l loadl %piv
	storel %jk, %pi
	storel %jv, %piv
	storel %sk, %pj
	storel %sv, %pjv
	jmp @nextiter
@noswap
	jmp @nextiter
@nextiter
	%inew =l phi @doswap %iinc, @noswap %i
	%jinc =l add %j, 1
	jmp @loop
@loopend
	%p =l add %i, 1
	%poff =l mul %p, 16
	%pp =l add %a, %poff
	%ppv =l add %pp, 8
	%sk2 =l loadl %pp
	%sv2 =l loadl %ppv
	storel %pk, %pp
	storel %pv, %ppv
	storel %sk2, %plast
	storel %sv2, %plastv
	%rightp =l add %p, 1
	%rightoff =l mul %rightp, 16
	%rightaddr =l add %a, %rightoff
	%rightn =l sub %last, %p
	%r1 =l call $qsort_pairs(l %a, l %p)
	%r2 =l call $qsort_pairs(l %rightaddr, l %rightn)
	jmp @ret
@ret
	ret 0
}
