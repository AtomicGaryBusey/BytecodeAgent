export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop_check
@loop_check
	%i =l phi @start 0, @loop_tail %i_new
	%o =l phi @start 0, @loop_tail %o_new
	%done =w csgel %i, %n
	jnz %done, @finish, @loop_body
@loop_body
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%has2 =w csgel %rem, 2
	jnz %has2, @load1, @skip1
@load1
	%p1 =l add %p0, 1
	%b1v =l loadub %p1
	jmp @after1
@skip1
	jmp @after1
@after1
	%b1 =l phi @load1 %b1v, @skip1 0
	%has3 =w csgel %rem, 3
	jnz %has3, @load2, @skip2
@load2
	%p2 =l add %p0, 2
	%b2v =l loadub %p2
	jmp @after2
@skip2
	jmp @after2
@after2
	%b2 =l phi @load2 %b2v, @skip2 0
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%u01 =l or %s0, %s1
	%val =l or %u01, %b2
	%sh0 =l shr %val, 18
	%v0 =l and %sh0, 63
	%c0a =w csgel %v0, 26
	%c0al =l extuw %c0a
	%c0am =l mul %c0al, 6
	%c0b =w csgel %v0, 52
	%c0bl =l extuw %c0b
	%c0bm =l mul %c0bl, 75
	%c0c =w csgel %v0, 62
	%c0cl =l extuw %c0c
	%c0cm =l mul %c0cl, 15
	%c0d =w csgel %v0, 63
	%c0dl =l extuw %c0d
	%c0dm =l mul %c0dl, 3
	%c0t1 =l add %v0, 65
	%c0t2 =l add %c0t1, %c0am
	%c0t3 =l sub %c0t2, %c0bm
	%c0t4 =l sub %c0t3, %c0cm
	%c0 =l add %c0t4, %c0dm
	%sh1 =l shr %val, 12
	%v1 =l and %sh1, 63
	%c1a =w csgel %v1, 26
	%c1al =l extuw %c1a
	%c1am =l mul %c1al, 6
	%c1b =w csgel %v1, 52
	%c1bl =l extuw %c1b
	%c1bm =l mul %c1bl, 75
	%c1c =w csgel %v1, 62
	%c1cl =l extuw %c1c
	%c1cm =l mul %c1cl, 15
	%c1d =w csgel %v1, 63
	%c1dl =l extuw %c1d
	%c1dm =l mul %c1dl, 3
	%c1t1 =l add %v1, 65
	%c1t2 =l add %c1t1, %c1am
	%c1t3 =l sub %c1t2, %c1bm
	%c1t4 =l sub %c1t3, %c1cm
	%c1 =l add %c1t4, %c1dm
	%sh2 =l shr %val, 6
	%v2 =l and %sh2, 63
	%c2a =w csgel %v2, 26
	%c2al =l extuw %c2a
	%c2am =l mul %c2al, 6
	%c2b =w csgel %v2, 52
	%c2bl =l extuw %c2b
	%c2bm =l mul %c2bl, 75
	%c2c =w csgel %v2, 62
	%c2cl =l extuw %c2c
	%c2cm =l mul %c2cl, 15
	%c2d =w csgel %v2, 63
	%c2dl =l extuw %c2d
	%c2dm =l mul %c2dl, 3
	%c2t1 =l add %v2, 65
	%c2t2 =l add %c2t1, %c2am
	%c2t3 =l sub %c2t2, %c2bm
	%c2t4 =l sub %c2t3, %c2cm
	%c2m =l add %c2t4, %c2dm
	%v3 =l and %val, 63
	%c3a =w csgel %v3, 26
	%c3al =l extuw %c3a
	%c3am =l mul %c3al, 6
	%c3b =w csgel %v3, 52
	%c3bl =l extuw %c3b
	%c3bm =l mul %c3bl, 75
	%c3c =w csgel %v3, 62
	%c3cl =l extuw %c3c
	%c3cm =l mul %c3cl, 15
	%c3d =w csgel %v3, 63
	%c3dl =l extuw %c3d
	%c3dm =l mul %c3dl, 3
	%c3t1 =l add %v3, 65
	%c3t2 =l add %c3t1, %c3am
	%c3t3 =l sub %c3t2, %c3bm
	%c3t4 =l sub %c3t3, %c3cm
	%c3m =l add %c3t4, %c3dm
	%rem_eq1 =w ceql %rem, 1
	jnz %rem_eq1, @c2_eq, @c2_ok
@c2_eq
	jmp @after_c2
@c2_ok
	jmp @after_c2
@after_c2
	%c2f =l phi @c2_eq 61, @c2_ok %c2m
	%rem_lt3 =w csltl %rem, 3
	jnz %rem_lt3, @c3_eq, @c3_ok
@c3_eq
	jmp @after_c3
@c3_ok
	jmp @after_c3
@after_c3
	%c3f =l phi @c3_eq 61, @c3_ok %c3m
	%op0 =l add %out, %o
	storeb %c0, %op0
	%op1 =l add %op0, 1
	storeb %c1, %op1
	%op2 =l add %op0, 2
	storeb %c2f, %op2
	%op3 =l add %op0, 3
	storeb %c3f, %op3
	jmp @loop_tail
@loop_tail
	%i_new =l add %i, 3
	%o_new =l add %o, 4
	jmp @loop_check
@finish
	ret %o
}
