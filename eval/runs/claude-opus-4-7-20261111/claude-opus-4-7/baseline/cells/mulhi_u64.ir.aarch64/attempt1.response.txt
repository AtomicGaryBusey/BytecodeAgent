export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%aL =l and %a, %mask
	%aH =l shr %a, 32
	%bL =l and %b, %mask
	%bH =l shr %b, 32
	%ll =l mul %aL, %bL
	%lh =l mul %aL, %bH
	%hl =l mul %aH, %bL
	%hh =l mul %aH, %bH
	%llhi =l shr %ll, 32
	%lhlo =l and %lh, %mask
	%hllo =l and %hl, %mask
	%m1 =l add %llhi, %lhlo
	%mid =l add %m1, %hllo
	%lhhi =l shr %lh, 32
	%hlhi =l shr %hl, 32
	%midhi =l shr %mid, 32
	%r1 =l add %hh, %lhhi
	%r2 =l add %r1, %hlhi
	%r =l add %r2, %midhi
	ret %r
}
