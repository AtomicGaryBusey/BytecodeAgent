export function l $isqrt(l %n0) {
@start
	jmp @loop
@loop
	%n =l phi @start %n0, @take %nt, @skip %ns
	%res =l phi @start 0, @take %rt, @skip %rs
	%bit =l phi @start 4611686018427387904, @take %bt, @skip %bs
	%cond =w cnel %bit, 0
	jnz %cond, @body, @end
@body
	%sum =l add %res, %bit
	%cmp =w cugel %n, %sum
	jnz %cmp, @take, @skip
@take
	%nt =l sub %n, %sum
	%rh =l shr %res, 1
	%rt =l add %rh, %bit
	%bt =l shr %bit, 2
	jmp @loop
@skip
	%ns =l copy %n
	%rs =l shr %res, 1
	%bs =l shr %bit, 2
	jmp @loop
@end
	ret %res
}
