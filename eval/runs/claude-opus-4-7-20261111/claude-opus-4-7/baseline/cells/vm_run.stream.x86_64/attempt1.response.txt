vm_run:
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jge .Ldone
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 0
    je .Ldone
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc r8
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    mov rax, qword ptr [rdx + r9*8 - 8]
    add qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Lsub:
    mov rax, qword ptr [rdx + r9*8 - 8]
    sub qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Lmul:
    mov rax, qword ptr [rdx + r9*8 - 8]
    imul rax, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 8], r10
    mov qword ptr [rdx + r9*8 - 16], rax
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov rax, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov r10, qword ptr [rdx + r9*8 - 16]
    xor r11d, r11d
    cmp r10, rax
    setl r11b
    mov qword ptr [rdx + r9*8 - 16], r11
    dec r9
    inc r8
    jmp .Lloop
.Ljz:
    dec r9
    mov rax, qword ptr [rdx + r9*8]
    test rax, rax
    jnz .Ljz_skip
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ljz_skip:
    add r8, 2
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ldone:
    xor rax, rax
    test r9, r9
    jz .Lzero_tail
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lzero_tail:
    mov r10, r9
.Lzero_loop:
    cmp r10, rcx
    jge .Lret
    mov qword ptr [rdx + r10*8], 0
    inc r10
    jmp .Lzero_loop
.Lret:
    ret
