    movi v0.2d, #0
    mov x3, #0
    cmp x2, #4
    b.lt .Ltail
.Lloop:
    ld1 {v1.4s}, [x0], #16
    ld1 {v2.4s}, [x1], #16
    smull v3.2d, v1.2s, v2.2s
    smlal2 v3.2d, v1.4s, v2.4s
    add v0.2d, v0.2d, v3.2d
    sub x2, x2, #4
    cmp x2, #4
    b.ge .Lloop
    addp d0, v0.2d
    fmov x3, d0
.Ltail:
    cbz x2, .Ldone
.Ltail_loop:
    ldr w4, [x0], #4
    ldr w5, [x1], #4
    smaddl x3, w4, w5, x3
    subs x2, x2, #1
    b.ne .Ltail_loop
.Ldone:
    mov x0, x3
    ret
