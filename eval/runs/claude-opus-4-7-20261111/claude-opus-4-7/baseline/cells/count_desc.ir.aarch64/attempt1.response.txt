export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%cmp =w csgtl %nm1, 0
	jnz %cmp, @loop, @end
@loop
	%i =l phi @start 0, @loop %inext
	%count =l phi @start 0, @loop %cnext
	%off =l mul %i, 8
	%p1 =l add %a, %off
	%v1 =l loadl %p1
	%p2 =l add %p1, 8
	%v2 =l loadl %p2
	%gt =w csgtl %v1, %v2
	%gtl =l extuw %gt
	%cnext =l add %count, %gtl
	%inext =l add %i, 1
	%cont =w csltl %inext, %nm1
	jnz %cont, @loop, @end
@end
	%result =l phi @start 0, @loop %cnext
	ret %result
}
