export function l $qsort_pairs(l %a, l %n) {
@start
    %c0 =w csltl %n, 2
    jnz %c0, @done, @work
@done
    ret 0
@work
    %mid =l div %n, 2
    %last =l sub %n, 1
    %midoff =l mul %mid, 16
    %midptr =l add %a, %midoff
    %lastoff =l mul %last, 16
    %lastptr =l add %a, %lastoff
    %mk =l loadl %midptr
    %mva =l add %midptr, 8
    %mv =l loadl %mva
    %lk =l loadl %lastptr
    %lva =l add %lastptr, 8
    %lv =l loadl %lva
    storel %lk, %midptr
    storel %lv, %mva
    storel %mk, %lastptr
    storel %mv, %lva
    %pkey =l loadl %lastptr
    %pval =l loadl %lva
    jmp @loop
@loop
    %i =l phi @work 0, @next %inext
    %j =l phi @work 0, @next %jnext
    %cj =w csltl %j, %last
    jnz %cj, @body, @endloop
@body
    %joff =l mul %j, 16
    %jptr =l add %a, %joff
    %jkey =l loadl %jptr
    %jva =l add %jptr, 8
    %jval =l loadl %jva
    %ltkey =w csltl %jkey, %pkey
    %eqkey =w ceql %jkey, %pkey
    %ltval =w csltl %jval, %pval
    %eqlt =w and %eqkey, %ltval
    %swap =w or %ltkey, %eqlt
    jnz %swap, @doswap, @next
@doswap
    %ioff =l mul %i, 16
    %iptr =l add %a, %ioff
    %ik =l loadl %iptr
    %iva =l add %iptr, 8
    %iv =l loadl %iva
    storel %jkey, %iptr
    storel %jval, %iva
    storel %ik, %jptr
    storel %iv, %jva
    %iinc =l add %i, 1
    jmp @next
@next
    %inext =l phi @doswap %iinc, @body %i
    %jnext =l add %j, 1
    jmp @loop
@endloop
    %ioff2 =l mul %i, 16
    %iptr2 =l add %a, %ioff2
    %ik2 =l loadl %iptr2
    %iva2 =l add %iptr2, 8
    %iv2 =l loadl %iva2
    storel %pkey, %iptr2
    storel %pval, %iva2
    storel %ik2, %lastptr
    storel %iv2, %lva
    %r1 =l call $qsort_pairs(l %a, l %i)
    %rightn =l sub %last, %i
    %roff =l add %ioff2, 16
    %rptr =l add %a, %roff
    %r2 =l call $qsort_pairs(l %rptr, l %rightn)
    ret 0
}
