export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @less %newlo, @greater %lo
	%hi =l phi @start %n, @less %hi, @greater %newhi
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @cont
@cont
	%lt =w csltl %val, %key
	jnz %lt, @less, @greater
@less
	%newlo =l add %mid, 1
	jmp @loop
@greater
	%newhi =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
