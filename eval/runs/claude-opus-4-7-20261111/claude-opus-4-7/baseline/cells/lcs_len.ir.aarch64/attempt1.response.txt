export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init_check
@init_check
	%j =l phi @start 0, @init_body %jn
	%cmp =w cslel %j, %n
	jnz %cmp, @init_body, @init_done
@init_body
	%off =l mul %j, 8
	%ptr =l add %row, %off
	storel 0, %ptr
	%jn =l add %j, 1
	jmp @init_check
@init_done
	jmp @outer_check
@outer_check
	%i =l phi @init_done 0, @outer_end %in
	%cmp2 =w csltl %i, %m
	jnz %cmp2, @outer_body, @outer_done
@outer_body
	%xp =l add %x, %i
	%xi =w loadub %xp
	jmp @inner_check
@inner_check
	%j2 =l phi @outer_body 1, @inner_next %j2n
	%diag =l phi @outer_body 0, @inner_next %up
	%cmp3 =w cslel %j2, %n
	jnz %cmp3, @inner_body, @inner_done
@inner_body
	%rowj_off =l mul %j2, 8
	%rowj_ptr =l add %row, %rowj_off
	%up =l loadl %rowj_ptr
	%jm1 =l sub %j2, 1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @no_match
@match
	%new =l add %diag, 1
	storel %new, %rowj_ptr
	jmp @inner_next
@no_match
	%rowjm1_off =l mul %jm1, 8
	%rowjm1_ptr =l add %row, %rowjm1_off
	%left =l loadl %rowjm1_ptr
	%cmp4 =w csgtl %up, %left
	jnz %cmp4, @use_up, @use_left
@use_up
	storel %up, %rowj_ptr
	jmp @inner_next
@use_left
	storel %left, %rowj_ptr
	jmp @inner_next
@inner_next
	%j2n =l add %j2, 1
	jmp @inner_check
@inner_done
	jmp @outer_end
@outer_end
	%in =l add %i, 1
	jmp @outer_check
@outer_done
	%result_off =l mul %n, 8
	%result_ptr =l add %row, %result_off
	%result =l loadl %result_ptr
	ret %result
}
