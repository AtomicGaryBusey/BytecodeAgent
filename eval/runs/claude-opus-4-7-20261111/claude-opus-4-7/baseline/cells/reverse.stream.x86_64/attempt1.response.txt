reverse:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    lea rcx, [rdi + rsi*8 - 8]
    mov rdx, rdi
.Lloop:
    cmp rdx, rcx
    jge .Ldone
    mov r8, qword ptr [rdx]
    mov r9, qword ptr [rcx]
    mov qword ptr [rdx], r9
    mov qword ptr [rcx], r8
    add rdx, 8
    sub rcx, 8
    jmp .Lloop
.Ldone:
    ret
