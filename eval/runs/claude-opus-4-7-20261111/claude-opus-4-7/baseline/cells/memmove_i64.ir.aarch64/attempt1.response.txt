export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %cmp =w cugtl %dst, %src
    jnz %cmp, @backward, @forward
@forward
    %i =l copy 0
    jmp @floop
@floop
    %fd =w csgel %i, %n
    jnz %fd, @end, @fbody
@fbody
    %off =l mul %i, 8
    %sp =l add %src, %off
    %dp =l add %dst, %off
    %v =l loadl %sp
    storel %v, %dp
    %i =l add %i, 1
    jmp @floop
@backward
    %j =l sub %n, 1
    jmp @bloop
@bloop
    %bd =w csltl %j, 0
    jnz %bd, @end, @bbody
@bbody
    %boff =l mul %j, 8
    %bsp =l add %src, %boff
    %bdp =l add %dst, %boff
    %bv =l loadl %bsp
    storel %bv, %bdp
    %j =l sub %j, 1
    jmp @bloop
@end
    ret 0
}
