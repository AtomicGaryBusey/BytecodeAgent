export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%cmp =w csgtl %b, %max
	jnz %cmp, @update, @cont
@update
	%max =l copy %b
	jmp @cont
@cont
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
