binary_search:
  xor eax, eax
.Lloop:
  cmp rax, rsi
  jge .Lnotfound
  lea r8, [rax + rsi]
  shr r8, 1
  mov r9, qword ptr [rdi + r8*8]
  cmp r9, rdx
  je .Lfound
  jl .Lless
  mov rsi, r8
  jmp .Lloop
.Lless:
  lea rax, [r8 + 1]
  jmp .Lloop
.Lfound:
  mov rax, r8
  ret
.Lnotfound:
  mov rax, -1
  ret
