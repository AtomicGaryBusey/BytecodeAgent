bubble_sort:
    xor rax, rax
    cmp rsi, 1
    jle .Ldone
    mov rcx, rsi
.Louter:
    dec rcx
    jz .Ldone
    xor rdx, rdx
.Linner:
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r8, r9
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], r8
.Lnoswap:
    inc rdx
    cmp rdx, rcx
    jl .Linner
    jmp .Louter
.Ldone:
    xor rax, rax
    ret
