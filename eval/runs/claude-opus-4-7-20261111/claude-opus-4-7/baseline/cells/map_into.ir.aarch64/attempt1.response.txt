export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
@loop
	%cond =l csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	storel %y, %dp
	%lt =w csltl %y, %x
	%lte =l extuw %lt
	%count =l add %count, %lte
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
