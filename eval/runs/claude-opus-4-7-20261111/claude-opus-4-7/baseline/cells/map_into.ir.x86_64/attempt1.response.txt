export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %ni
	%count =l phi @start 0, @body %ncount
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	storel %y, %dp
	%lt =w csltl %y, %x
	%ltl =l extuw %lt
	%ncount =l add %count, %ltl
	%ni =l add %i, 1
	jmp @loop
@end
	ret %count
}
