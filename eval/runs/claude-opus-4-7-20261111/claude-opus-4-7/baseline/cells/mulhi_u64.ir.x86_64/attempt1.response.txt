export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%aL =l and %a, %mask
	%aH =l shr %a, 32
	%bL =l and %b, %mask
	%bH =l shr %b, 32
	%ll =l mul %aL, %bL
	%lh =l mul %aL, %bH
	%hl =l mul %aH, %bL
	%hh =l mul %aH, %bH
	%llH =l shr %ll, 32
	%lhL =l and %lh, %mask
	%hlL =l and %hl, %mask
	%lhH =l shr %lh, 32
	%hlH =l shr %hl, 32
	%m1 =l add %llH, %lhL
	%mid =l add %m1, %hlL
	%midH =l shr %mid, 32
	%h1 =l add %hh, %lhH
	%h2 =l add %h1, %hlH
	%high =l add %h2, %midH
	ret %high
}
