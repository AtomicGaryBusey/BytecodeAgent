dot_i32_simd:
    xor eax, eax
    xor ecx, ecx
    pxor xmm2, xmm2
    cmp rdx, 4
    jb .Lreduce
.Lvector:
    movdqu xmm0, xmmword ptr [rdi + rcx*4]
    movdqu xmm1, xmmword ptr [rsi + rcx*4]
    movdqa xmm3, xmm0
    pmuldq xmm3, xmm1
    psrldq xmm0, 4
    psrldq xmm1, 4
    pmuldq xmm0, xmm1
    paddq xmm3, xmm0
    paddq xmm2, xmm3
    add rcx, 4
    lea r8, [rcx + 4]
    cmp r8, rdx
    jbe .Lvector
.Lreduce:
    movq rax, xmm2
    psrldq xmm2, 8
    movq r8, xmm2
    add rax, r8
.Ltail:
    cmp rcx, rdx
    jae .Ldone
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    jmp .Ltail
.Ldone:
    ret
