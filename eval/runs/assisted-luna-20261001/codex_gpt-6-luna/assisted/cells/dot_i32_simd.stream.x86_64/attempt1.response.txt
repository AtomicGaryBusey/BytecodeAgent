.intel_syntax noprefix
dot_i32_simd:
    xor eax, eax
    xor ecx, ecx
    mov r8, rdx
    cmp r8, 4
    jb .Lscalar
.Lvector:
    movdqu xmm0, xmmword ptr [rdi + rcx*4]
    movdqu xmm1, xmmword ptr [rsi + rcx*4]
    movdqa xmm2, xmm0
    pmovsxdq xmm2, xmm2
    movdqa xmm3, xmm1
    pmovsxdq xmm3, xmm3
    pmuldq xmm2, xmm3
    movdqa xmm4, xmm0
    psrldq xmm4, 4
    pmovsxdq xmm4, xmm4
    movdqa xmm5, xmm1
    psrldq xmm5, 4
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm2, xmm4
    movdqa xmm4, xmm0
    psrldq xmm4, 8
    pmovsxdq xmm4, xmm4
    movdqa xmm5, xmm1
    psrldq xmm5, 8
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm2, xmm4
    movdqa xmm4, xmm0
    psrldq xmm4, 12
    pmovsxdq xmm4, xmm4
    movdqa xmm5, xmm1
    psrldq xmm5, 12
    pmovsxdq xmm5, xmm5
    pmuldq xmm4, xmm5
    paddq xmm2, xmm4
    movq r9, xmm2
    psrldq xmm2, 8
    movq r10, xmm2
    add r9, r10
    add rax, r9
    add rcx, 4
    lea r9, [rcx + 4]
    cmp r9, r8
    jbe .Lvector
.Lscalar:
    cmp rcx, r8
    jae .Ldone
.Ltail:
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r10, dword ptr [rsi + rcx*4]
    imul r9, r10
    add rax, r9
    inc rcx
    cmp rcx, r8
    jb .Ltail
.Ldone:
    ret
