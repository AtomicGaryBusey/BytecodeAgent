export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%backward =w cugtl %dst, %src
	jnz %backward, @back, @forward
@forward
	%i =l copy 0
@floop
	%more =w csltl %i, %n
	jnz %more, @fbody, @done
@fbody
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%value =l loadl %sp
	storel %value, %dp
	%i =l add %i, 1
	jmp @floop
@back
	%i =l copy %n
@bloop
	%more =w csgtl %i, 0
	jnz %more, @bbody, @done
@bbody
	%i =l sub %i, 1
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%value =l loadl %sp
	storel %value, %dp
	jmp @bloop
@done
	%zero =l copy 0
	ret %zero
}
