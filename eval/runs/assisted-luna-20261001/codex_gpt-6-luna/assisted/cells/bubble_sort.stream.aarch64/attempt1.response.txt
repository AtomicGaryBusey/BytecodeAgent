bubble_sort:
cmp x1, #1
b.ls .Ldone
sub x2, x1, #1
.Louter:
mov x3, #0
.Linner:
ldr x4, [x0, x3, lsl #3]
add x5, x3, #1
ldr x6, [x0, x5, lsl #3]
cmp x4, x6
b.le .Lnoswap
str x6, [x0, x3, lsl #3]
str x4, [x0, x5, lsl #3]
.Lnoswap:
add x3, x3, #1
cmp x3, x2
b.lo .Linner
sub x2, x2, #1
cbnz x2, .Louter
.Ldone:
mov x0, #0
ret
