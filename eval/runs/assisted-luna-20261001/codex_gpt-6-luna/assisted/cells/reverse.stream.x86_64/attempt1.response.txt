reverse:
    test rsi, rsi
    jle .Ldone
    xor ecx, ecx
    dec rsi
.Lloop:
    cmp rcx, rsi
    jge .Ldone
    mov rdx, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rsi*8]
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rsi*8], rdx
    inc rcx
    dec rsi
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
