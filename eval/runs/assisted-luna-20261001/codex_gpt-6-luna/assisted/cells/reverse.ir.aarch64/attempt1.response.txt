export function l $reverse(l %a, l %n) {
@start
	%left =l copy 0
	%right =l sub %n, 1
	jmp @loop
@loop
	%more =w csltl %left, %right
	jnz %more, @body, @done
@body
	%leftoff =l mul %left, 8
	%leftptr =l add %a, %leftoff
	%leftval =l loadl %leftptr
	%rightoff =l mul %right, 8
	%rightptr =l add %a, %rightoff
	%rightval =l loadl %rightptr
	storel %rightval, %leftptr
	storel %leftval, %rightptr
	%left =l add %left, 1
	%right =l sub %right, 1
	jmp @loop
@done
	%zero =l copy 0
	ret %zero
}
