export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%nonempty =w csgtl %n, 0
	jnz %nonempty, @check, @done
@check
	%dstafter =w cugtl %dst, %src
	jnz %dstafter, @overlapcheck, @forwardinit
@overlapcheck
	%delta =l sub %dst, %src
	%bytes =l mul %n, 8
	%overlap =w cultl %delta, %bytes
	jnz %overlap, @backwardinit, @forwardinit
@forwardinit
	%i =l copy 0
	jmp @forward
@forward
	%more =w csltl %i, %n
	jnz %more, @forwardbody, @done
@forwardbody
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i =l add %i, 1
	jmp @forward
@backwardinit
	%i =l copy %n
	jmp @backward
@backward
	%moreback =w csgtl %i, 0
	jnz %moreback, @backwardbody, @done
@backwardbody
	%i =l sub %i, 1
	%offback =l mul %i, 8
	%spback =l add %src, %offback
	%dpback =l add %dst, %offback
	%vback =l loadl %spback
	storel %vback, %dpback
	jmp @backward
@done
	%zero =l copy 0
	ret %zero
}
