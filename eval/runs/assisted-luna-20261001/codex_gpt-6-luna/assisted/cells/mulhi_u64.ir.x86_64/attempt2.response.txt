export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, %mask
	%p10lo =l and %p10, %mask
	%t0 =l add %p00hi, %p01lo
	%t1 =l add %t0, %p10lo
	%carry =l shr %t1, 32
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%u0 =l add %p11, %p01hi
	%u1 =l add %u0, %p10hi
	%result =l add %u1, %carry
	ret %result
}
