export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%mid0 =l add %p01, %p00
	%carry0 =l shr %mid0, 32
	%mid1 =l add %mid0, %p10
	%carry1 =l shr %mid1, 32
	%upper =l add %p11, %a1
	%upper =l add %upper, %b1
	%upper =l sub %upper, %a1
	%upper =l sub %upper, %b1
	%upper =l add %upper, %carry0
	%upper =l add %upper, %carry1
	%result =l shr %mid1, 32
	%result =l add %upper, %result
	ret %result
}
