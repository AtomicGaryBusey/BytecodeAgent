export function l $fib_rec(l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @checkone
@zero
	%z =l copy 0
	ret %z
@checkone
	%isone =w ceql %n, 1
	jnz %isone, @one, @recurse
@one
	%o =l copy 1
	ret %o
@recurse
	%n1 =l sub %n, 1
	%a =l call $fib_rec(l %n1)
	%n2 =l sub %n, 2
	%b =l call $fib_rec(l %n2)
	%sum =l add %a, %b
	ret %sum
}
