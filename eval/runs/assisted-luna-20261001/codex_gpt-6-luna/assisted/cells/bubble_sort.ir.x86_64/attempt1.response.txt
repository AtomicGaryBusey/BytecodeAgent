export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%limit =l sub %n, 1
	%more =w csltl %i, %limit
	jnz %more, @inner_init, @done
@inner_init
	%j =l copy 0
	%end =l sub %limit, %i
	jmp @inner
@inner
	%jmore =w csltl %j, %end
	jnz %jmore, @compare, @next_outer
@compare
	%off =l mul %j, 8
	%p =l add %a, %off
	%pnext =l add %p, 8
	%x =l loadl %p
	%y =l loadl %pnext
	%swap =w csgtl %x, %y
	jnz %swap, @do_swap, @advance
@do_swap
	storel %y, %p
	storel %x, %pnext
	jmp @advance
@advance
	%j =l add %j, 1
	jmp @inner
@next_outer
	%i =l add %i, 1
	jmp @outer
@done
	%zero =l copy 0
	ret %zero
}
