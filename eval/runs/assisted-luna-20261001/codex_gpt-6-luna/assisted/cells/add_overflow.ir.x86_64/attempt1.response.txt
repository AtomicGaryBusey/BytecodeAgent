export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%signa =w csltl %a, 0
	%signb =w csltl %b, 0
	%signsum =w csltl %sum, 0
	%same =w ceqw %signa, %signb
	%changed =w cnew %signsum, %signa
	%overflow =w and %same, %changed
	%result =l extsw %overflow
	ret %result
}
