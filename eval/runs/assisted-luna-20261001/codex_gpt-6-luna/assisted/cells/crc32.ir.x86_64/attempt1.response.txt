export function l $crc32(l %p, l %n) {
@start
	%c =w copy -1
	%i =l copy 0
@byte
	%more =w csltl %i, %n
	jnz %more, @bytebody, @done
@bytebody
	%addr =l add %p, %i
	%b =w loadub %addr
	%c =w xor %c, %b
	%j =w copy 0
@bit
	%bitsmore =w csltw %j, 8
	jnz %bitsmore, @bitbody, @nextbyte
@bitbody
	%lsb =w and %c, 1
	jnz %lsb, @bitone, @bitzero
@bitone
	%shift1 =w shr %c, 1
	%c =w xor %shift1, -306674912
	jmp @bitnext
@bitzero
	%c =w shr %c, 1
@bitnext
	%j =w add %j, 1
	jmp @bit
@nextbyte
	%i =l add %i, 1
	jmp @byte
@done
	%result =w xor %c, -1
	%out =l extuw %result
	ret %out
}
