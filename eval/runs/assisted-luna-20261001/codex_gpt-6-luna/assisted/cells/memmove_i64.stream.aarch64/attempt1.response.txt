memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.ls .Lforward
    add x3, x1, x2, lsl #3
    cmp x0, x3
    b.hs .Lforward
    add x0, x0, x2, lsl #3
    add x1, x1, x2, lsl #3
.Lbackward:
    ldr x4, [x1, #-8]!
    str x4, [x0, #-8]!
    subs x2, x2, #1
    b.ne .Lbackward
    b .Ldone
.Lforward:
    cbz x2, .Ldone
.Lforward_loop:
    ldr x4, [x1], #8
    str x4, [x0], #8
    subs x2, x2, #1
    b.ne .Lforward_loop
.Ldone:
    mov x0, #0
    ret
