memmove_i64:
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    jbe .Lforward
    mov r8, rdi
    sub r8, rsi
    mov rcx, rdx
    shl rcx, 3
    cmp r8, rcx
    jae .Lforward
.Lbackward:
    dec rdx
    mov r8, qword ptr [rsi + rdx*8]
    mov qword ptr [rdi + rdx*8], r8
    test rdx, rdx
    jnz .Lbackward
    ret
.Lforward:
    xor ecx, ecx
.Lforward_loop:
    cmp rcx, rdx
    jae .Ldone
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    jmp .Lforward_loop
.Ldone:
    ret
