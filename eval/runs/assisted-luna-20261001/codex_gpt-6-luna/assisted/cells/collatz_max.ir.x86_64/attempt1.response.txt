export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%best =l copy 0
	jmp @outer
@outer
	%more =w csle l %k, %n
	jnz %more, @begin, @done
@begin
	%x =l copy %k
	%steps =l copy 0
	jmp @inner
@inner
	%finished =w ceql %x, 1
	jnz %finished, @check, @step
@step
	%odd =l and %x, 1
	%iseven =w ceql %odd, 0
	jnz %iseven, @halve, @triple
@halve
	%x =l shr %x, 1
	jmp @count
@triple
	%x =l add %x, %x
	%x =l add %x, %k
	%x =l sub %x, %k
	%x =l sub %x, %k
	%x =l add %x, 1
	jmp @count
@count
	%steps =l add %steps, 1
	jmp @inner
@check
	%better =w csgtl %steps, %best
	jnz %better, @update, @next
@update
	%best =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
