export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967295
	jmp @loop
@loop
	%done =w ceql %lo, %hi
	jnz %done, @exit, @step
@step
	%diff =l sub %hi, %lo
	%diff1 =l add %diff, 1
	%half =l shr %diff1, 1
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%fits =w culel %mid, %quot
	jnz %fits, @raise, @lower
@raise
	%lo =l copy %mid
	jmp @loop
@lower
	%hi =l sub %mid, 1
	jmp @loop
@exit
	ret %lo
}
