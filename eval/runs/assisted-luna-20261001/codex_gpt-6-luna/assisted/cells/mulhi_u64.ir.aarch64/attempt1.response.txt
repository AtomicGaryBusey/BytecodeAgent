export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p01lo =l and %p01, %mask
	%p10lo =l and %p10, %mask
	%mid0 =l shr %p00, 32
	%mid1 =l add %mid0, %p01lo
	%mid2 =l add %mid1, %p10lo
	%carry =l shr %mid2, 32
	%hi0 =l shr %p01, 32
	%hi1 =l shr %p10, 32
	%hi2 =l add %p11, %hi0
	%hi3 =l add %hi2, %hi1
	%hi4 =l add %hi3, %carry
	ret %hi4
}
