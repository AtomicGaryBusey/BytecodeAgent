export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%azero =l copy 0
	%aneg =w csltl %a, %azero
	%bneg =w csltl %b, %azero
	%sneg =w csltl %sum, %azero
	%same =w ceqw %aneg, %bneg
	%diff =w cnew %aneg, %sneg
	%overflow =w and %same, %diff
	jnz %overflow, @yes, @no
@yes
	%one =l copy 1
	ret %one
@no
	%zero =l copy 0
	ret %zero
}
