export function l $fib(l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%iszero =w ceql %n, %zero
	jnz %iszero, @donezero, @init
@donezero
	ret %zero
@init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 1
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%next =l add %a, %b
	%a =l copy %b
	%b =l copy %next
	%i =l add %i, 1
	jmp @loop
@done
	ret %b
}
