export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%more =w csltl %lo, %hi
	jnz %more, @body, @miss
@body
	%sum =l add %lo, %hi
	%mid =l udiv %sum, 2
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%less =w csltl %val, %key
	jnz %less, @right, @check
@right
	%lo =l add %mid, 1
	jmp @loop
@check
	%greater =w csgtl %val, %key
	jnz %greater, @left, @found
@left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@miss
	%minus =l copy -1
	ret %minus
}
