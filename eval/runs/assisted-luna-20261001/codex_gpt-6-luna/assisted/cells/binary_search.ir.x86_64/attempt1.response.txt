export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%active =w csltl %lo, %hi
	jnz %active, @search, @missing
@search
	%sum =l add %lo, %hi
	%mid =l udiv %sum, 2
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%value =l loadl %ptr
	%less =w csltl %value, %key
	jnz %less, @right, @check
@right
	%next =l add %mid, 1
	%lo =l copy %next
	jmp @loop
@check
	%greater =w csgtl %value, %key
	jnz %greater, @left, @found
@left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@missing
	%minusone =l copy -1
	ret %minusone
}
