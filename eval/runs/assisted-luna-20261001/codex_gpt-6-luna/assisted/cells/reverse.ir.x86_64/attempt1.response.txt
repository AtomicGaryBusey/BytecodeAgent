export function l $reverse(l %a, l %n) {
@start
	%left =l copy 0
	%right =l sub %n, 1
	jmp @loop
@loop
	%more =w csltl %left, %right
	jnz %more, @body, @done
@body
	%loff =l mul %left, 8
	%laddr =l add %a, %loff
	%roff =l mul %right, 8
	%raddr =l add %a, %roff
	%lv =l loadl %laddr
	%rv =l loadl %raddr
	storel %rv, %laddr
	storel %lv, %raddr
	%left =l add %left, 1
	%right =l sub %right, 1
	jmp @loop
@done
	ret 0
}
