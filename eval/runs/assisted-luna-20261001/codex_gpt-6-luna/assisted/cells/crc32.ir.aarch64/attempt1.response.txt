export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
@outer
	%more =w csltl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%j =w copy 0
@inner
	%bits =w csltw %j, 8
	jnz %bits, @bit, @next
@bit
	%low =w and %crc, 1
	jnz %low, @xorpoly, @shift
@xorpoly
	%shifted1 =w shr %crc, 1
	%crc =w xor %shifted1, -306674912
	jmp @bitdone
@shift
	%crc =w shr %crc, 1
@bitdone
	%j =w add %j, 1
	jmp @inner
@next
	%i =l add %i, 1
	jmp @outer
@done
	%result =w xor %crc, -1
	%out =l extuw %result
	ret %out
}
