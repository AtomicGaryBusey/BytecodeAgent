crc32:
    mov eax, 0xffffffff
    test rsi, rsi
    jz .Ldone
.Lbyte:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit:
    mov edx, eax
    and edx, 1
    shr eax, 1
    test edx, edx
    jz .Lnext
    xor eax, 0xedb88320
.Lnext:
    dec ecx
    jnz .Lbit
    inc rdi
    dec rsi
    jnz .Lbyte
.Ldone:
    not eax
    ret
