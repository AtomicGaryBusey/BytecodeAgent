bubble_sort:
    xor eax, eax
    cmp rsi, 1
    jle .Ldone
    dec rsi
.Louter:
    xor ecx, ecx
    mov rdx, rsi
.Linner:
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lnoswap
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], r8
.Lnoswap:
    inc rcx
    cmp rcx, rdx
    jl .Linner
    dec rsi
    jg .Louter
.Ldone:
    xor eax, eax
    ret
