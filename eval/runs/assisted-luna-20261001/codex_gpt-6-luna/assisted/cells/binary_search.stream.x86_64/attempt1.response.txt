binary_search:
mov rcx, rsi
mov r8, rdx
xor r9d, r9d
.Lloop:
cmp r9, rcx
jge .Lnot_found
mov r10, rcx
sub r10, r9
shr r10, 1
add r10, r9
mov r11, qword ptr [rdi + r10*8]
cmp r11, r8
je .Lfound
jl .Lright
mov rcx, r10
jmp .Lloop
.Lright:
lea r9, [r10 + 1]
jmp .Lloop
.Lfound:
mov rax, r10
ret
.Lnot_found:
mov rax, -1
ret
