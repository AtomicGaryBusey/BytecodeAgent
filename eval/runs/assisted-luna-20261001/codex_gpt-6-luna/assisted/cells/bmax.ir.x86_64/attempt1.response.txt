export function l $bmax(l %a, l %n) {
@start
	%best =l loadl %a
	%i =l copy 1
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%greater =w csgtl %v, %best
	jnz %greater, @update, @next
@update
	%best =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %best
}
