export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967296
	jmp @loop
@loop
	%gap =l sub %hi, %lo
	%done =w culel %gap, 1
	jnz %done, @exit, @search
@search
	%sum =l add %lo, %hi
	%mid =l udiv %sum, 2
	%q =l udiv %n, %mid
	%fits =w culel %mid, %q
	jnz %fits, @raise, @lower
@raise
	%lo =l copy %mid
	jmp @loop
@lower
	%hi =l copy %mid
	jmp @loop
@exit
	ret %lo
}
