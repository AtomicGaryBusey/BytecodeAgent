export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%more =w csltl %i, %n
	jnz %more, @inner_start, @done
@inner_start
	%j =l copy 0
	%limit0 =l sub %n, %i
	%limit =l sub %limit0, 1
@inner
	%in_range =w csltl %j, %limit
	jnz %in_range, @compare, @next_pass
@compare
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%j1 =l add %j, 1
	%off1 =l mul %j1, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%swap =w csgtl %v0, %v1
	jnz %swap, @do_swap, @no_swap
@do_swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @no_swap
@no_swap
	%j =l add %j, 1
	jmp @inner
@next_pass
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
