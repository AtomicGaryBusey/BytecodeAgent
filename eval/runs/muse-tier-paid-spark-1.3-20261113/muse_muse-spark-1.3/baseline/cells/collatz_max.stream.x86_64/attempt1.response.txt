collatz_max:
xor eax, eax
mov ecx, 1
.Louter:
cmp rcx, rdi
jg .Ldone
mov rdx, rcx
xor esi, esi
.Linner:
cmp rdx, 1
je .Linner_done
test rdx, 1
jnz .Lodd
shr rdx, 1
inc rsi
jmp .Linner
.Lodd:
lea rdx, [rdx + rdx*2 + 1]
inc rsi
jmp .Linner
.Linner_done:
cmp rsi, rax
cmova rax, rsi
inc rcx
jmp .Louter
.Ldone:
ret
