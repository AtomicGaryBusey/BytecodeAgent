export function l $count_desc(l %a, l %n) {
@start
  %c =w csltl %n, 2
  jnz %c, @ret0, @pre
@ret0
  %z =l copy 0
  ret %z
@pre
  %limit =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @pre 0, @body %i_next
  %cnt =l phi @pre 0, @body %cnt_next
  %t =w csltl %i, %limit
  jnz %t, @body, @done
@body
  %off =l mul %i, 8
  %p0 =l add %a, %off
  %v0 =l loadl %p0
  %p1 =l add %p0, 8
  %v1 =l loadl %p1
  %gt =w csgtl %v0, %v1
  %inc =l extuw %gt
  %cnt_next =l add %cnt, %inc
  %i_next =l add %i, 1
  jmp @loop
@done
  ret %cnt
}
