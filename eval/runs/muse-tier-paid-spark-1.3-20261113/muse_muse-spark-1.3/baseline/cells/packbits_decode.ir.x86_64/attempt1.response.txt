export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
%i0 =l copy 0
%o0 =l copy 0
jmp @loop
@loop
%i =l phi @start %i0, @nop %i_nop, @lit_done %i_lit_next, @rep_done %i_rep_next
%o =l phi @start %o0, @nop %o_nop, @lit_done %o_lit_final, @rep_done %o_rep_final
%eof =w csgel %i, %n
jnz %eof, @ret, @hdr
@ret
ret %o
@hdr
%hp =l add %src, %i
%hb =w loadsb %hp
%h =l extsw %hb
%is_nop =w ceql %h, -128
jnz %is_nop, @nop, @not_nop
@not_nop
%is_lit =w csgel %h, 0
jnz %is_lit, @lit, @rep
@nop
%i_nop =l add %i, 1
%o_nop =l copy %o
jmp @loop
@lit
%h_plus1 =l add %h, 1
%need =l add %h, 2
%rem =l sub %n, %i
%lit_short =w csltl %rem, %need
jnz %lit_short, @err_src, @lit_ok
@lit_ok
%j0 =l copy 0
%o_lit0 =l copy %o
jmp @lit_loop
@lit_loop
%j =l phi @lit_ok %j0, @lit_body %j_next
%o_lit =l phi @lit_ok %o_lit0, @lit_body %o_lit_next
%lit_done_c =w ceql %j, %h_plus1
jnz %lit_done_c, @lit_done, @lit_cap
@lit_cap
%lit_full =w csgel %o_lit, %cap
jnz %lit_full, @err_cap, @lit_body
@lit_body
%lit_src_off =l add %i, 1
%lit_src_off2 =l add %lit_src_off, %j
%lit_src_ptr =l add %src, %lit_src_off2
%lit_b =w loadub %lit_src_ptr
%lit_dst_ptr =l add %out, %o_lit
storeb %lit_b, %lit_dst_ptr
%j_next =l add %j, 1
%o_lit_next =l add %o_lit, 1
jmp @lit_loop
@lit_done
%i_lit_next =l add %i, %need
%o_lit_final =l copy %o_lit
jmp @loop
@rep
%rem_rep =l sub %n, %i
%rep_short =w csltl %rem_rep, 2
jnz %rep_short, @err_src, @rep_ok
@rep_ok
%one =l copy 1
%count =l sub %one, %h
%rep_val_off =l add %i, 1
%rep_val_ptr =l add %src, %rep_val_off
%val =w loadub %rep_val_ptr
%k0 =l copy 0
%o_rep0 =l copy %o
jmp @rep_loop
@rep_loop
%k =l phi @rep_ok %k0, @rep_body %k_next
%o_rep =l phi @rep_ok %o_rep0, @rep_body %o_rep_next
%rep_done_c =w ceql %k, %count
jnz %rep_done_c, @rep_done, @rep_cap
@rep_cap
%rep_full =w csgel %o_rep, %cap
jnz %rep_full, @err_cap, @rep_body
@rep_body
%rep_dst_ptr =l add %out, %o_rep
storeb %val, %rep_dst_ptr
%k_next =l add %k, 1
%o_rep_next =l add %o_rep, 1
jmp @rep_loop
@rep_done
%i_rep_next =l add %i, 2
%o_rep_final =l copy %o_rep
jmp @loop
@err_cap
%r_cap =l copy -1
ret %r_cap
@err_src
%r_src =l copy -2
ret %r_src
}
