export function l $crc32(l %p, l %n) {
@start
  %crc0 =w copy -1
  %i0 =l copy 0
  %poly =w copy -306674912
  %eight =w copy 8
  jmp @outer_cond
@outer_cond
  %i =l phi @start %i0, @outer_inc %i_next
  %crc_outer =w phi @start %crc0, @outer_inc %crc_after
  %c1 =w csltl %i, %n
  jnz %c1, @outer_body, @done
@outer_body
  %addr =l add %p, %i
  %b =w loadub %addr
  %crc_xor =w xor %crc_outer, %b
  %k0 =w copy 0
  jmp @inner_cond
@inner_cond
  %k =w phi @outer_body %k0, @inner_merge %k_next
  %crc_inner =w phi @outer_body %crc_xor, @inner_merge %crc_step
  %c2 =w csltw %k, %eight
  jnz %c2, @inner_test, @outer_inc
@inner_test
  %bit =w and %crc_inner, 1
  jnz %bit, @inner_odd, @inner_even
@inner_odd
  %shr1 =w shr %crc_inner, 1
  %xor1 =w xor %shr1, %poly
  jmp @inner_merge
@inner_even
  %shr2 =w shr %crc_inner, 1
  jmp @inner_merge
@inner_merge
  %crc_step =w phi @inner_odd %xor1, @inner_even %shr2
  %k_next =w add %k, 1
  jmp @inner_cond
@outer_inc
  %i_next =l add %i, 1
  %crc_after =w copy %crc_inner
  jmp @outer_cond
@done
  %res_w =w xor %crc_outer, -1
  %res =l extuw %res_w
  ret %res
}
