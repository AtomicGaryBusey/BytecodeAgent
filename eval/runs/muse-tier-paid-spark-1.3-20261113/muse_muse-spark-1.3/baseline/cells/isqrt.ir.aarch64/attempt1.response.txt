export function l $isqrt(l %n) {
@start
  %zero =l copy 0
  %one =l copy 1
  %two32 =l shl %one, 32
  %max =l sub %two32, 1
  jmp @loop
@loop
  %lo =l phi @start %zero, @cont %lo_cont
  %hi =l phi @start %max, @cont %hi_cont
  %ans =l phi @start %zero, @cont %ans_cont
  %cond =w culel %lo, %hi
  jnz %cond, @body, @end
@body
  %sum =l add %lo, %hi
  %mid =l shr %sum, 1
  %sq =l mul %mid, %mid
  %cmp =w culel %sq, %n
  jnz %cmp, @less, @greater
@less
  %lo_less =l add %mid, 1
  %hi_less =l copy %hi
  %ans_less =l copy %mid
  jmp @cont
@greater
  %lo_greater =l copy %lo
  %hi_greater =l sub %mid, 1
  %ans_greater =l copy %ans
  jmp @cont
@cont
  %lo_cont =l phi @less %lo_less, @greater %lo_greater
  %hi_cont =l phi @less %hi_less, @greater %hi_greater
  %ans_cont =l phi @less %ans_less, @greater %ans_greater
  jmp @loop
@end
  ret %ans
}
