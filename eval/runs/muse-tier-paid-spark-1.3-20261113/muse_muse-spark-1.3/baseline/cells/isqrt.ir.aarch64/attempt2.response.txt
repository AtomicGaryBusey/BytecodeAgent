export function l $isqrt(l %n) {
@start
  %lo0 = copy 0
  %hi0 = copy 4294967295
  %ans0 = copy 0
  jmp @loop
@loop
  %lo = phi @start %lo0, @less %lo1, @greater %lo2
  %hi = phi @start %hi0, @less %hi1, @greater %hi2
  %ans = phi @start %ans0, @less %ans1, @greater %ans2
  %cond = culel %lo, %hi
  jnz %cond, @body, @done
@body
  %sum = addl %lo, %hi
  %mid = udivl %sum, 2
  %sq = mull %mid, %mid
  %le = culel %sq, %n
  jnz %le, @less, @greater
@less
  %lo1 = addl %mid, 1
  %hi1 = copy %hi
  %ans1 = copy %mid
  jmp @loop
@greater
  %lo2 = copy %lo
  %hi2 = subl %mid, 1
  %ans2 = copy %ans
  jmp @loop
@done
  ret %ans
}
