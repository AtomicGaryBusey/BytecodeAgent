export function l $isqrt(l %n) {
@start
  %lo0 =l copy 0
  %hi0 =l copy 4294967295
  %ans0 =l copy 0
  jmp @loop
@loop
  %lo =l phi @start %lo0, @less %lo1, @greater %lo2
  %hi =l phi @start %hi0, @less %hi1, @greater %hi2
  %ans =l phi @start %ans0, @less %ans1, @greater %ans2
  %cond =w culel %lo, %hi
  jnz %cond, @body, @done
@body
  %sum =l add %lo, %hi
  %mid =l udiv %sum, 2
  %sq =l mul %mid, %mid
  %le =w culel %sq, %n
  jnz %le, @less, @greater
@less
  %lo1 =l add %mid, 1
  %hi1 =l copy %hi
  %ans1 =l copy %mid
  jmp @loop
@greater
  %lo2 =l copy %lo
  %hi2 =l sub %mid, 1
  %ans2 =l copy %ans
  jmp @loop
@done
  ret %ans
}
