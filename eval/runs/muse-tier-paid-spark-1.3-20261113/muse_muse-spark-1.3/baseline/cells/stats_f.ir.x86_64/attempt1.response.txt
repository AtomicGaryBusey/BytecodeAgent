export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%iptr =l alloc8 8
	%minptr =l alloc8 8
	%maxptr =l alloc8 8
	%sumptr =l alloc8 8
	%cntptr =l alloc8 8
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	storel %y0, %minptr
	storel %y0, %maxptr
	storel %y0, %sumptr
	%t0 =l and %y0, 1
	%c0 =w cnel %t0, 0
	jnz %c0, @odd0, @even0
@odd0
	%one0 =l copy 1
	storel %one0, %cntptr
	%i1a =l copy 1
	storel %i1a, %iptr
	jmp @loop
@even0
	%zero0 =l copy 0
	storel %zero0, %cntptr
	%i1b =l copy 1
	storel %i1b, %iptr
	jmp @loop
@loop
	%i =l loadl %iptr
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%pa =l add %a, %off
	%av =l loadl %pa
	%y =l call %f(l %av)
	%curmin =l loadl %minptr
	%islt =w csltl %y, %curmin
	jnz %islt, @updatemin, @checkmax
@updatemin
	storel %y, %minptr
	jmp @checkmax
@checkmax
	%curmax =l loadl %maxptr
	%isgt =w csgtl %y, %curmax
	jnz %isgt, @updatemax, @updatesum
@updatemax
	storel %y, %maxptr
	jmp @updatesum
@updatesum
	%cursum =l loadl %sumptr
	%newsum =l add %cursum, %y
	storel %newsum, %sumptr
	%tb =l and %y, 1
	%oddb =w cnel %tb, 0
	jnz %oddb, @incodd, @noinc
@incodd
	%curcnt1 =l loadl %cntptr
	%newcnt1 =l add %curcnt1, 1
	storel %newcnt1, %cntptr
	jmp @incdone
@noinc
	jmp @incdone
@incdone
	%curi =l loadl %iptr
	%newi =l add %curi, 1
	storel %newi, %iptr
	jmp @loop
@done
	%fmin =l loadl %minptr
	%fmax =l loadl %maxptr
	%fsum =l loadl %sumptr
	%fcnt =l loadl %cntptr
	storel %fmin, %out
	%out1 =l add %out, 8
	storel %fmax, %out1
	%out2 =l add %out, 16
	storel %fsum, %out2
	ret %fcnt
}
