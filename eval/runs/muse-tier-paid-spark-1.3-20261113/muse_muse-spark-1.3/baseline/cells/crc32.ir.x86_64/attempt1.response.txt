export function l $crc32(l %p, l %n) {
@start
  %crc0 =w copy -1
  %i0 =l copy 0
  jmp @outer_cond
@outer_cond
  %i =l phi @start %i0, @outer_next %i_next
  %crc_outer =w phi @start %crc0, @outer_next %crc_next
  %cond =w csltl %i, %n
  jnz %cond, @outer_body, @done
@outer_body
  %addr =l add %p, %i
  %b =w loadub %addr
  %crc_xor =w xor %crc_outer, %b
  %bit0 =w copy 0
  jmp @inner_cond
@inner_cond
  %bit =w phi @outer_body %bit0, @inner_body %bit_next
  %crc_cur =w phi @outer_body %crc_xor, @inner_body %crc_new
  %cond_inner =w csltw %bit, 8
  jnz %cond_inner, @inner_body, @outer_next
@inner_body
  %lsb =w and %crc_cur, 1
  %shifted =w shr %crc_cur, 1
  %neg =w sub 0, %lsb
  %masked =w and %neg, -306674912
  %crc_new =w xor %shifted, %masked
  %bit_next =w add %bit, 1
  jmp @inner_cond
@outer_next
  %crc_next =w copy %crc_cur
  %i_next =l add %i, 1
  jmp @outer_cond
@done
  %crc_final =w xor %crc_outer, -1
  %res =l extuw %crc_final
  ret %res
}
