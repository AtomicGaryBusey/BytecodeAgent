export function l $max_byte(l %p, l %n) {
@start
  %c =w ceql %n, 0
  jnz %c, @ret0, @loop
@ret0
  %r0 =l copy 0
  ret %r0
@loop
  %i =l phi @start 0, @join %i2
  %m =l phi @start 0, @join %m2
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %addr =l add %p, %i
  %b =l loadub %addr
  %gt =w cugtl %b, %m
  jnz %gt, @upd, @keep
@upd
  jmp @join
@keep
  jmp @join
@join
  %m2 =l phi @upd %b, @keep %m
  %i2 =l add %i, 1
  jmp @loop
@done
  ret %m
}
