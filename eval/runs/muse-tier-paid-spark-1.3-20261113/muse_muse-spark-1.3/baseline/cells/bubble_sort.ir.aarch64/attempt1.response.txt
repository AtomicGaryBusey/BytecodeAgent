export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer_cond
@outer_cond
	%i =l phi @start 0, @outer_next %i_next
	%c0 =w csltl %i, %n
	jnz %c0, @outer_body, @done
@outer_body
	%t0 =l sub %n, %i
	%limit =l sub %t0, 1
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body 0, @inner_next %j_next
	%c1 =w csltl %j, %limit
	jnz %c1, @inner_body, @outer_next
@inner_body
	%j8 =l mul %j, 8
	%addr0 =l add %a, %j8
	%v0 =l loadl %addr0
	%addr1 =l add %addr0, 8
	%v1 =l loadl %addr1
	%c2 =w csgtl %v0, %v1
	jnz %c2, @do_swap, @inner_next
@do_swap
	storel %v1, %addr0
	storel %v0, %addr1
	jmp @inner_next
@inner_next
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_next
	%i_next =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
