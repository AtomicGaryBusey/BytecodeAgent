export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %zero =l copy 0
    %one =l copy 1
    %t32 =l shl %one, 32
    %mask =l sub %t32, 1
    %n =l add %na, %nb
    jmp @zero_head
@zero_head
    %k0 =l phi @start %zero, @zero_body %k1
    %cz =w csltl %k0, %n
    jnz %cz, @zero_body, @outer_head
@zero_body
    %zoff =l shl %k0, 3
    %zaddr =l add %r, %zoff
    storel %zero, %zaddr
    %k1 =l add %k0, 1
    jmp @zero_head
@outer_head
    %i0 =l phi @zero_head %zero, @outer_latch %i1
    %co =w csltl %i0, %na
    jnz %co, @outer_body, @scan_head
@outer_body
    %aoff =l shl %i0, 3
    %aaddr =l add %a, %aoff
    %ai =l loadl %aaddr
    jmp @inner_head
@inner_head
    %j0 =l phi @outer_body %zero, @inner_body %j1
    %carry0 =l phi @outer_body %zero, @inner_body %carry1
    %ci =w csltl %j0, %nb
    jnz %ci, @inner_body, @outer_latch
@inner_body
    %boff =l shl %j0, 3
    %baddr =l add %b, %boff
    %bj =l loadl %baddr
    %ij =l add %i0, %j0
    %roff =l shl %ij, 3
    %raddr =l add %r, %roff
    %rold =l loadl %raddr
    %a0 =l and %ai, %mask
    %a1 =l shr %ai, 32
    %b0 =l and %bj, %mask
    %b1 =l shr %bj, 32
    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1
    %mid =l add %p1, %p2
    %cmid =w cultl %mid, %p1
    %midsh =l shl %mid, 32
    %prodlo =l add %p0, %midsh
    %clo =w cultl %prodlo, %p0
    %midhi =l shr %mid, 32
    %cmidl =l extuw %cmid
    %cmidsh =l shl %cmidl, 32
    %clol =l extuw %clo
    %hit1 =l add %p3, %midhi
    %hit2 =l add %hit1, %cmidsh
    %prodhi =l add %hit2, %clol
    %t1 =l add %prodlo, %rold
    %c1 =w cultl %t1, %prodlo
    %t2 =l add %t1, %carry0
    %c2 =w cultl %t2, %t1
    %c1l =l extuw %c1
    %c2l =l extuw %c2
    %th1 =l add %prodhi, %c1l
    %carry1 =l add %th1, %c2l
    storel %t2, %raddr
    %j1 =l add %j0, 1
    jmp @inner_head
@outer_latch
    %lidx =l add %i0, %nb
    %loff =l shl %lidx, 3
    %laddr =l add %r, %loff
    storel %carry0, %laddr
    %i1 =l add %i0, 1
    jmp @outer_head
@scan_head
    %sk0 =l phi @outer_head %n, @scan_body %sk1
    %csn =w cnel %sk0, 0
    jnz %csn, @scan_check, @done
@scan_check
    %skm1 =l sub %sk0, 1
    %soff =l shl %skm1, 3
    %saddr =l add %r, %soff
    %sval =l loadl %saddr
    %csv =w ceql %sval, 0
    jnz %csv, @scan_body, @done
@scan_body
    %sk1 =l copy %skm1
    jmp @scan_head
@done
    ret %sk0
}
