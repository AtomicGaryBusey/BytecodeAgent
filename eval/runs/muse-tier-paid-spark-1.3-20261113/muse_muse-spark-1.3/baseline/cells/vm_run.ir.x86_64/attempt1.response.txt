export function l $vm_run(l %code, l %len, l %stk, l %cap) {
@start
  %pcptr =l alloc8 8
  %dptr =l alloc8 8
  %iptr =l alloc8 8
  %z0 =l copy 0
  storel %z0, %pcptr
  storel %z0, %dptr
  jmp @loop
@loop
  %pc_loop =l loadl %pcptr
  %c_loop =w csltl %pc_loop, %len
  jnz %c_loop, @dispatch, @cleanup
@dispatch
  %pc_dis =l loadl %pcptr
  %off_dis =l mul %pc_dis, 8
  %addr_dis =l add %code, %off_dis
  %op_dis =l loadl %addr_dis
  %is0 =w ceql %op_dis, 0
  jnz %is0, @do_halt, @c1
@c1
  %pc_c1 =l loadl %pcptr
  %off_c1 =l mul %pc_c1, 8
  %addr_c1 =l add %code, %off_c1
  %op_c1 =l loadl %addr_c1
  %is1 =w ceql %op_c1, 1
  jnz %is1, @do_push, @c2
@c2
  %pc_c2 =l loadl %pcptr
  %off_c2 =l mul %pc_c2, 8
  %addr_c2 =l add %code, %off_c2
  %op_c2 =l loadl %addr_c2
  %is2 =w ceql %op_c2, 2
  jnz %is2, @do_add, @c3
@c3
  %pc_c3 =l loadl %pcptr
  %off_c3 =l mul %pc_c3, 8
  %addr_c3 =l add %code, %off_c3
  %op_c3 =l loadl %addr_c3
  %is3 =w ceql %op_c3, 3
  jnz %is3, @do_sub, @c4
@c4
  %pc_c4 =l loadl %pcptr
  %off_c4 =l mul %pc_c4, 8
  %addr_c4 =l add %code, %off_c4
  %op_c4 =l loadl %addr_c4
  %is4 =w ceql %op_c4, 4
  jnz %is4, @do_mul, @c5
@c5
  %pc_c5 =l loadl %pcptr
  %off_c5 =l mul %pc_c5, 8
  %addr_c5 =l add %code, %off_c5
  %op_c5 =l loadl %addr_c5
  %is5 =w ceql %op_c5, 5
  jnz %is5, @do_dup, @c6
@c6
  %pc_c6 =l loadl %pcptr
  %off_c6 =l mul %pc_c6, 8
  %addr_c6 =l add %code, %off_c6
  %op_c6 =l loadl %addr_c6
  %is6 =w ceql %op_c6, 6
  jnz %is6, @do_swap, @c7
@c7
  %pc_c7 =l loadl %pcptr
  %off_c7 =l mul %pc_c7, 8
  %addr_c7 =l add %code, %off_c7
  %op_c7 =l loadl %addr_c7
  %is7 =w ceql %op_c7, 7
  jnz %is7, @do_drop, @c8
@c8
  %pc_c8 =l loadl %pcptr
  %off_c8 =l mul %pc_c8, 8
  %addr_c8 =l add %code, %off_c8
  %op_c8 =l loadl %addr_c8
  %is8 =w ceql %op_c8, 8
  jnz %is8, @do_over, @c9
@c9
  %pc_c9 =l loadl %pcptr
  %off_c9 =l mul %pc_c9, 8
  %addr_c9 =l add %code, %off_c9
  %op_c9 =l loadl %addr_c9
  %is9 =w ceql %op_c9, 9
  jnz %is9, @do_lt, @c10
@c10
  %pc_c10 =l loadl %pcptr
  %off_c10 =l mul %pc_c10, 8
  %addr_c10 =l add %code, %off_c10
  %op_c10 =l loadl %addr_c10
  %is10 =w ceql %op_c10, 10
  jnz %is10, @do_jz, @c11
@c11
  %pc_c11 =l loadl %pcptr
  %off_c11 =l mul %pc_c11, 8
  %addr_c11 =l add %code, %off_c11
  %op_c11 =l loadl %addr_c11
  %is11 =w ceql %op_c11, 11
  jnz %is11, @do_jmp, @do_default
@do_halt
  jmp @cleanup
@do_push
  %pc_push =l loadl %pcptr
  %d_push =l loadl %dptr
  %off_push =l mul %pc_push, 8
  %addr_push =l add %code, %off_push
  %taddr_push =l add %addr_push, 8
  %v_push =l loadl %taddr_push
  %soff_push =l mul %d_push, 8
  %saddr_push =l add %stk, %soff_push
  storel %v_push, %saddr_push
  %dnew_push =l add %d_push, 1
  storel %dnew_push, %dptr
  %pcnew_push =l add %pc_push, 2
  storel %pcnew_push, %pcptr
  jmp @loop
@do_add
  %pc_add =l loadl %pcptr
  %d_add =l loadl %dptr
  %d1_add =l sub %d_add, 1
  %offb_add =l mul %d1_add, 8
  %addrb_add =l add %stk, %offb_add
  %b_add =l loadl %addrb_add
  %d2_add =l sub %d_add, 2
  %offa_add =l mul %d2_add, 8
  %addra_add =l add %stk, %offa_add
  %a_add =l loadl %addra_add
  %sum_add =l add %a_add, %b_add
  storel %sum_add, %addra_add
  storel %d1_add, %dptr
  %pcnew_add =l add %pc_add, 1
  storel %pcnew_add, %pcptr
  jmp @loop
@do_sub
  %pc_sub =l loadl %pcptr
  %d_sub =l loadl %dptr
  %d1_sub =l sub %d_sub, 1
  %offb_sub =l mul %d1_sub, 8
  %addrb_sub =l add %stk, %offb_sub
  %b_sub =l loadl %addrb_sub
  %d2_sub =l sub %d_sub, 2
  %offa_sub =l mul %d2_sub, 8
  %addra_sub =l add %stk, %offa_sub
  %a_sub =l loadl %addra_sub
  %dif_sub =l sub %a_sub, %b_sub
  storel %dif_sub, %addra_sub
  storel %d1_sub, %dptr
  %pcnew_sub =l add %pc_sub, 1
  storel %pcnew_sub, %pcptr
  jmp @loop
@do_mul
  %pc_mul =l loadl %pcptr
  %d_mul =l loadl %dptr
  %d1_mul =l sub %d_mul, 1
  %offb_mul =l mul %d1_mul, 8
  %addrb_mul =l add %stk, %offb_mul
  %b_mul =l loadl %addrb_mul
  %d2_mul =l sub %d_mul, 2
  %offa_mul =l mul %d2_mul, 8
  %addra_mul =l add %stk, %offa_mul
  %a_mul =l loadl %addra_mul
  %prod_mul =l mul %a_mul, %b_mul
  storel %prod_mul, %addra_mul
  storel %d1_mul, %dptr
  %pcnew_mul =l add %pc_mul, 1
  storel %pcnew_mul, %pcptr
  jmp @loop
@do_dup
  %pc_dup =l loadl %pcptr
  %d_dup =l loadl %dptr
  %d1_dup =l sub %d_dup, 1
  %off_dup =l mul %d1_dup, 8
  %addr_dup =l add %stk, %off_dup
  %a_dup =l loadl %addr_dup
  %off2_dup =l mul %d_dup, 8
  %addr2_dup =l add %stk, %off2_dup
  storel %a_dup, %addr2_dup
  %dnew_dup =l add %d_dup, 1
  storel %dnew_dup, %dptr
  %pcnew_dup =l add %pc_dup, 1
  storel %pcnew_dup, %pcptr
  jmp @loop
@do_swap
  %pc_swap =l loadl %pcptr
  %d_swap =l loadl %dptr
  %d1_swap =l sub %d_swap, 1
  %offb_swap =l mul %d1_swap, 8
  %addrb_swap =l add %stk, %offb_swap
  %b_swap =l loadl %addrb_swap
  %d2_swap =l sub %d_swap, 2
  %offa_swap =l mul %d2_swap, 8
  %addra_swap =l add %stk, %offa_swap
  %a_swap =l loadl %addra_swap
  storel %b_swap, %addra_swap
  storel %a_swap, %addrb_swap
  %pcnew_swap =l add %pc_swap, 1
  storel %pcnew_swap, %pcptr
  jmp @loop
@do_drop
  %pc_drop =l loadl %pcptr
  %d_drop =l loadl %dptr
  %dnew_drop =l sub %d_drop, 1
  storel %dnew_drop, %dptr
  %pcnew_drop =l add %pc_drop, 1
  storel %pcnew_drop, %pcptr
  jmp @loop
@do_over
  %pc_over =l loadl %pcptr
  %d_over =l loadl %dptr
  %d2_over =l sub %d_over, 2
  %off_over =l mul %d2_over, 8
  %addr_over =l add %stk, %off_over
  %a_over =l loadl %addr_over
  %off2_over =l mul %d_over, 8
  %addr2_over =l add %stk, %off2_over
  storel %a_over, %addr2_over
  %dnew_over =l add %d_over, 1
  storel %dnew_over, %dptr
  %pcnew_over =l add %pc_over, 1
  storel %pcnew_over, %pcptr
  jmp @loop
@do_lt
  %pc_lt =l loadl %pcptr
  %d_lt =l loadl %dptr
  %d1_lt =l sub %d_lt, 1
  %offb_lt =l mul %d1_lt, 8
  %addrb_lt =l add %stk, %offb_lt
  %b_lt =l loadl %addrb_lt
  %d2_lt =l sub %d_lt, 2
  %offa_lt =l mul %d2_lt, 8
  %addra_lt =l add %stk, %offa_lt
  %a_lt =l loadl %addra_lt
  %c_lt =w csltl %a_lt, %b_lt
  %r_lt =l extsw %c_lt
  storel %r_lt, %addra_lt
  storel %d1_lt, %dptr
  %pcnew_lt =l add %pc_lt, 1
  storel %pcnew_lt, %pcptr
  jmp @loop
@do_jz
  %pc_jz =l loadl %pcptr
  %d_jz =l loadl %dptr
  %d1_jz =l sub %d_jz, 1
  %off_jz =l mul %d1_jz, 8
  %addr_jz =l add %stk, %off_jz
  %v_jz =l loadl %addr_jz
  storel %d1_jz, %dptr
  %iszero_jz =w ceql %v_jz, 0
  jnz %iszero_jz, @jz_zero, @jz_nz
@jz_zero
  %pc_jz0 =l loadl %pcptr
  %off_jz0 =l mul %pc_jz0, 8
  %addr_jz0 =l add %code, %off_jz0
  %taddr_jz0 =l add %addr_jz0, 8
  %t_jz0 =l loadl %taddr_jz0
  storel %t_jz0, %pcptr
  jmp @loop
@jz_nz
  %pc_jz1 =l loadl %pcptr
  %pcnew_jz1 =l add %pc_jz1, 2
  storel %pcnew_jz1, %pcptr
  jmp @loop
@do_jmp
  %pc_jmp =l loadl %pcptr
  %off_jmp =l mul %pc_jmp, 8
  %addr_jmp =l add %code, %off_jmp
  %taddr_jmp =l add %addr_jmp, 8
  %t_jmp =l loadl %taddr_jmp
  storel %t_jmp, %pcptr
  jmp @loop
@do_default
  %pc_def =l loadl %pcptr
  %pcnew_def =l add %pc_def, 1
  storel %pcnew_def, %pcptr
  jmp @loop
@cleanup
  %d_cln =l loadl %dptr
  storel %d_cln, %iptr
  jmp @zloop
@zloop
  %i_zl =l loadl %iptr
  %c_zl =w csltl %i_zl, %cap
  jnz %c_zl, @zbody, @retblk
@zbody
  %i_zb =l loadl %iptr
  %off_zb =l mul %i_zb, 8
  %addr_zb =l add %stk, %off_zb
  %z_zb =l copy 0
  storel %z_zb, %addr_zb
  %inew_zb =l add %i_zb, 1
  storel %inew_zb, %iptr
  jmp @zloop
@retblk
  %d_ret =l loadl %dptr
  %isempty_ret =w ceql %d_ret, 0
  jnz %isempty_ret, @retempty, @rettop
@retempty
  %r_empty =l copy 0
  ret %r_empty
@rettop
  %d_top =l loadl %dptr
  %dm1_top =l sub %d_top, 1
  %off_top =l mul %dm1_top, 8
  %addr_top =l add %stk, %off_top
  %v_top =l loadl %addr_top
  ret %v_top
}
