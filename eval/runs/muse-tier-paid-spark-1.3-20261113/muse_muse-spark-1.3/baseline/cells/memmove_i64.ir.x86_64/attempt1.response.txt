export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c0 =w cslel %n, 0
	jnz %c0, @ret, @cmp
@cmp
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_head, @bwd_head
@fwd_head
	%fi =l phi @cmp 0, @fwd_body %fi_next
	%fcond =w csltl %fi, %n
	jnz %fcond, @fwd_body, @ret
@fwd_body
	%foff =l mul %fi, 8
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fval =l loadl %fsrc
	storel %fval, %fdst
	%fi_next =l add %fi, 1
	jmp @fwd_head
@bwd_head
	%bi =l phi @cmp %n, @bwd_body %bi_next
	%bcond =w csgtl %bi, 0
	jnz %bcond, @bwd_body, @ret
@bwd_body
	%bi_dec =l sub %bi, 1
	%boff =l mul %bi_dec, 8
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	%bi_next =l copy %bi_dec
	jmp @bwd_head
@ret
	ret 0
}
