export function l $qsort_pairs(l %a0, l %n0) {
@start
%small =w cslel %n0, 1
jnz %small, @ret0, @prep
@ret0
ret 0
@prep
%nminus1 =l sub %n0, 1
%mid =l shr %n0, 1
%mid_off =l mul %mid, 16
%mid_ptr =l add %a0, %mid_off
%last_off =l mul %nminus1, 16
%last_ptr =l add %a0, %last_off
%mk =l loadl %mid_ptr
%mid_ptr8 =l add %mid_ptr, 8
%mv =l loadl %mid_ptr8
%lk =l loadl %last_ptr
%last_ptr8 =l add %last_ptr, 8
%lv =l loadl %last_ptr8
storel %lk, %mid_ptr
storel %lv, %mid_ptr8
storel %mk, %last_ptr
storel %mv, %last_ptr8
%zero =l copy 0
jmp @loop
@loop
%i_cur =l phi @prep %zero, @next %i_next
%j_cur =l phi @prep %zero, @next %j_next
%cond =w csltl %j_cur, %nminus1
jnz %cond, @body, @done
@body
%j_off =l mul %j_cur, 16
%j_ptr =l add %a0, %j_off
%jk =l loadl %j_ptr
%j_ptr8 =l add %j_ptr, 8
%jv =l loadl %j_ptr8
%lt_key =w csltl %jk, %mk
%eq_key =w ceql %jk, %mk
%lt_val =w csltl %jv, %mv
%eq_and_lt =w and %eq_key, %lt_val
%is_less =w or %lt_key, %eq_and_lt
jnz %is_less, @do_swap, @no_swap
@do_swap
%i_off =l mul %i_cur, 16
%i_ptr =l add %a0, %i_off
%i_ptr8 =l add %i_ptr, 8
%ik =l loadl %i_ptr
%iv =l loadl %i_ptr8
storel %jk, %i_ptr
storel %jv, %i_ptr8
storel %ik, %j_ptr
storel %iv, %j_ptr8
%i_inc =l add %i_cur, 1
jmp @next
@no_swap
jmp @next
@next
%i_next =l phi @do_swap %i_inc, @no_swap %i_cur
%j_next =l add %j_cur, 1
jmp @loop
@done
%fi_off =l mul %i_cur, 16
%fi_ptr =l add %a0, %fi_off
%fi_ptr8 =l add %fi_ptr, 8
%fik =l loadl %fi_ptr
%fiv =l loadl %fi_ptr8
%flk =l loadl %last_ptr
%flv =l loadl %last_ptr8
storel %flk, %fi_ptr
storel %flv, %fi_ptr8
storel %fik, %last_ptr
storel %fiv, %last_ptr8
%right_idx =l add %i_cur, 1
%right_off =l mul %right_idx, 16
%right_ptr =l add %a0, %right_off
%right_size =l sub %n0, %right_idx
%r1 =l call $qsort_pairs(l %a0, l %i_cur)
%r2 =l call $qsort_pairs(l %right_ptr, l %right_size)
ret 0
}
