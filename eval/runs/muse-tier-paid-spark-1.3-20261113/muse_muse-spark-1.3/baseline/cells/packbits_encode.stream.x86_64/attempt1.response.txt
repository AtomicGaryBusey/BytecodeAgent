packbits_encode:
xor r8, r8
xor r9, r9
.Louter:
cmp r8, rsi
jge .Ldone
mov r10, 1
mov r11b, byte ptr [rdi + r8]
.Lrun:
cmp r10, 128
jge .Lrun_done
mov rax, r8
add rax, r10
cmp rax, rsi
jge .Lrun_done
mov al, byte ptr [rdi + rax]
cmp al, r11b
jne .Lrun_done
inc r10
jmp .Lrun
.Lrun_done:
cmp r10, 3
jge .Lrepeat
mov r11, r8
.Llit:
cmp r11, rsi
jge .Llit_done
mov rax, r11
sub rax, r8
cmp rax, 128
jge .Llit_done
mov rax, rsi
sub rax, r11
cmp rax, 3
jl .Llit_next
mov al, byte ptr [rdi + r11]
cmp al, byte ptr [rdi + r11 + 1]
jne .Llit_next
cmp al, byte ptr [rdi + r11 + 2]
je .Llit_done
.Llit_next:
inc r11
jmp .Llit
.Llit_done:
mov r10, r11
sub r10, r8
mov rax, r9
add rax, r10
add rax, 1
cmp rax, rcx
ja .Lfail
mov rax, r10
dec rax
mov byte ptr [rdx + r9], al
xor rax, rax
.Lcopy:
cmp rax, r10
jge .Lcopy_done
add rax, r8
mov r11b, byte ptr [rdi + rax]
sub rax, r8
add rax, r9
mov byte ptr [rdx + rax + 1], r11b
sub rax, r9
inc rax
jmp .Lcopy
.Lcopy_done:
add r8, r10
add r9, r10
inc r9
jmp .Louter
.Lrepeat:
mov rax, r9
add rax, 2
cmp rax, rcx
ja .Lfail
mov rax, 257
sub rax, r10
mov byte ptr [rdx + r9], al
mov al, r11b
mov byte ptr [rdx + r9 + 1], al
add r9, 2
add r8, r10
jmp .Louter
.Ldone:
mov rax, r9
ret
.Lfail:
mov rax, -1
ret
