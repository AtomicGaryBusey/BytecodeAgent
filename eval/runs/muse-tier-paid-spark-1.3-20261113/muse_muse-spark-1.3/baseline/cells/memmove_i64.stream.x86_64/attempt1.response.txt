memmove_i64:
test rdx, rdx
jle .Ldone
cmp rdi, rsi
jb .Lfwd
je .Ldone
mov rcx, rdi
sub rcx, rsi
mov r8, rdx
shl r8, 3
cmp rcx, r8
jb .Lbwd
.Lfwd:
xor ecx, ecx
.Lfwd_loop:
cmp rcx, rdx
jae .Ldone
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
jmp .Lfwd_loop
.Lbwd:
mov rcx, rdx
dec rcx
.Lbwd_loop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
dec rcx
jns .Lbwd_loop
.Ldone:
xor eax, eax
ret
