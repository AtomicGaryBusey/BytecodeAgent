export function l $mp_mul(l %a0, l %na0, l %b0, l %nb0, l %r0) {
@start
  %zero =l copy 0
  %all1 =l sub %zero, 1
  %mask =l shr %all1, 32
  %n =l add %na0, %nb0
  jmp @zero_cond
@zero_cond
  %zi =l phi @start %zero, @zero_body %zi_next
  %zcond =w cultl %zi, %n
  jnz %zcond, @zero_body, @outer_cond
@zero_body
  %zoff =l mul %zi, 8
  %zaddr =l add %r0, %zoff
  storel %zero, %zaddr
  %zi_next =l add %zi, 1
  jmp @zero_cond
@outer_cond
  %i =l phi @zero_cond %zero, @outer_latch %i_next
  %ocond =w cultl %i, %na0
  jnz %ocond, @inner_cond, @scan_cond
@inner_cond
  %j =l phi @outer_cond %zero, @inner_latch %j_next
  %icond =w cultl %j, %nb0
  jnz %icond, @mul_body, @outer_latch
@mul_body
  %aoff =l mul %i, 8
  %aptr =l add %a0, %aoff
  %x =l loadl %aptr
  %boff =l mul %j, 8
  %bptr =l add %b0, %boff
  %y =l loadl %bptr
  %x0 =l and %x, %mask
  %x1 =l shr %x, 32
  %y0 =l and %y, %mask
  %y1 =l shr %y, 32
  %p0 =l mul %x0, %y0
  %p1 =l mul %x0, %y1
  %p2 =l mul %x1, %y0
  %p3 =l mul %x1, %y1
  %mid =l add %p1, %p2
  %cmid_w =w cultl %mid, %p1
  %cmid =l extuw %cmid_w
  %mid_shl =l shl %mid, 32
  %mid_shr =l shr %mid, 32
  %lo =l add %p0, %mid_shl
  %c0_w =w cultl %lo, %p0
  %c0 =l extuw %c0_w
  %hi1 =l add %p3, %mid_shr
  %hi2 =l add %hi1, %c0
  %cmid_shl =l shl %cmid, 32
  %hi =l add %hi2, %cmid_shl
  %k =l add %i, %j
  %koff =l mul %k, 8
  %kaddr =l add %r0, %koff
  %rk =l loadl %kaddr
  %s0 =l add %rk, %lo
  %cc0_w =w cultl %s0, %rk
  %cc0 =l extuw %cc0_w
  storel %s0, %kaddr
  %k1 =l add %k, 1
  %k1off =l mul %k1, 8
  %k1addr =l add %r0, %k1off
  %rk1 =l loadl %k1addr
  %s1 =l add %rk1, %hi
  %cc1_w =w cultl %s1, %rk1
  %cc1 =l extuw %cc1_w
  %s2 =l add %s1, %cc0
  %cc2_w =w cultl %s2, %s1
  %cc2 =l extuw %cc2_w
  storel %s2, %k1addr
  %carry0 =l add %cc1, %cc2
  %idx0 =l add %k, 2
  jmp @prop_cond
@prop_cond
  %pidx =l phi @mul_body %idx0, @prop_body %pidx_next
  %pcarry =l phi @mul_body %carry0, @prop_body %pcarry_next
  %pneed1 =w cnel %pcarry, %zero
  %pneed2 =w cultl %pidx, %n
  %pneed =w and %pneed1, %pneed2
  jnz %pneed, @prop_body, @inner_latch
@prop_body
  %poff =l mul %pidx, 8
  %paddr =l add %r0, %poff
  %pr =l loadl %paddr
  %ps =l add %pr, %pcarry
  %pc_w =w cultl %ps, %pr
  %pcarry_next =l extuw %pc_w
  storel %ps, %paddr
  %pidx_next =l add %pidx, 1
  jmp @prop_cond
@inner_latch
  %j_next =l add %j, 1
  jmp @inner_cond
@outer_latch
  %i_next =l add %i, 1
  jmp @outer_cond
@scan_cond
  %sk =l phi @outer_cond %n, @scan_next %sk_next
  %sis0 =w ceql %sk, %zero
  jnz %sis0, @scan_ret0, @scan_load
@scan_ret0
  ret %zero
@scan_load
  %km1a =l sub %sk, 1
  %koffa =l mul %km1a, 8
  %kaddra =l add %r0, %koffa
  %kv =l loadl %kaddra
  %kisnz =w cnel %kv, %zero
  jnz %kisnz, @scan_ret, @scan_next
@scan_ret
  ret %sk
@scan_next
  %sk_next =l copy %km1a
  jmp @scan_cond
}
