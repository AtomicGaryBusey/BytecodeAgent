export function l $count_desc(l %a, l %n) {
@start
  %small =w csltl %n, 2
  jnz %small, @ret0, @preloop
@ret0
  ret 0
@preloop
  %limit =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @preloop 0, @body %inext
  %cnt =l phi @preloop 0, @body %cntnext
  %cond =w csltl %i, %limit
  jnz %cond, @body, @done
@body
  %off =l mul %i, 8
  %ptr0 =l add %a, %off
  %v0 =l loadl %ptr0
  %ptr1 =l add %ptr0, 8
  %v1 =l loadl %ptr1
  %gt =w csgtl %v0, %v1
  %gtl =l extuw %gt
  %cntnext =l add %cnt, %gtl
  %inext =l add %i, 1
  jmp @loop
@done
  ret %cnt
}
