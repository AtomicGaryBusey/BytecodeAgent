export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
  %a0 =l loadl %a
  %y0 =l call %f(l %a0)
  %c0 =l and %y0, 1
  %i0 =l copy 1
  jmp @loop
@loop
  %i =l phi @start %i0, @body %i_next
  %mn =l phi @start %y0, @body %mn_next
  %mx =l phi @start %y0, @body %mx_next
  %sm =l phi @start %y0, @body %sm_next
  %ct =l phi @start %c0, @body %ct_next
  %cc =w csltl %i, %n
  jnz %cc, @body, @done
@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %av =l loadl %ptr
  %y =l call %f(l %av)
  %sm_next =l add %sm, %y
  %odd =l and %y, 1
  %ct_next =l add %ct, %odd
  %i_next =l add %i, 1
  %lt =w csltl %y, %mn
  %lt_e =l extsw %lt
  %zero =l copy 0
  %mask_lo =l sub %zero, %lt_e
  %diff_lo =l sub %y, %mn
  %adj_lo =l and %diff_lo, %mask_lo
  %mn_next =l add %mn, %adj_lo
  %gt =w csltl %mx, %y
  %gt_e =l extsw %gt
  %mask_hi =l sub %zero, %gt_e
  %diff_hi =l sub %y, %mx
  %adj_hi =l and %diff_hi, %mask_hi
  %mx_next =l add %mx, %adj_hi
  jmp @loop
@done
  %out1 =l add %out, 8
  %out2 =l add %out, 16
  storel %mn, %out
  storel %mx, %out1
  storel %sm, %out2
  ret %ct
}
