export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %si_p =l alloc8 8
  %di_p =l alloc8 8
  %k_p =l alloc8 8
  %zero_init =l copy 0
  storel %zero_init, %si_p
  storel %zero_init, %di_p
  jmp @loop
@loop
  %si_loop =l loadl %si_p
  %loop_cond =w csltl %si_loop, %n
  jnz %loop_cond, @header, @done
@done
  %di_done =l loadl %di_p
  ret %di_done
@header
  %hdr_addr =l add %src, %si_loop
  %h_w =w loadsb %hdr_addr
  %h =l extsw %h_w
  %neg128 =l copy -128
  %is_nop =w ceql %h, %neg128
  jnz %is_nop, @nop, @check_lit
@nop
  %one_nop =l copy 1
  %si_nop_new =l add %si_loop, %one_nop
  storel %si_nop_new, %si_p
  jmp @loop
@check_lit
  %zero_check =l copy 0
  %is_neg =w csltl %h, %zero_check
  jnz %is_neg, @repeat, @literal
@literal
  %one_lit_len =l copy 1
  %len_lit =l add %h, %one_lit_len
  %two_lit_need =l copy 2
  %need_lit =l add %h, %two_lit_need
  %rem_lit =l sub %n, %si_loop
  %lit_srcfail =w csltl %rem_lit, %need_lit
  jnz %lit_srcfail, @srcfail, @literal_capcheck
@srcfail
  %neg2 =l copy -2
  ret %neg2
@literal_capcheck
  %di_lit =l loadl %di_p
  %di_plus_len =l add %di_lit, %len_lit
  %lit_overflow =w csltl %cap, %di_plus_len
  jnz %lit_overflow, @literal_trunc, @literal_full
@literal_full
  %zero_lit_full =l copy 0
  storel %zero_lit_full, %k_p
  jmp @literal_full_loop
@literal_full_loop
  %k_lit_full =l loadl %k_p
  %lit_full_cond =w csltl %k_lit_full, %len_lit
  jnz %lit_full_cond, @literal_full_body, @literal_full_done
@literal_full_body
  %src_off_lit =l add %si_loop, %k_lit_full
  %one_lit_body =l copy 1
  %src_off_lit1 =l add %src_off_lit, %one_lit_body
  %src_addr_lit =l add %src, %src_off_lit1
  %dst_off_lit =l add %di_lit, %k_lit_full
  %dst_addr_lit =l add %out, %dst_off_lit
  %byte_lit =w loadub %src_addr_lit
  storeb %byte_lit, %dst_addr_lit
  %k_next_lit_full =l add %k_lit_full, %one_lit_body
  storel %k_next_lit_full, %k_p
  jmp @literal_full_loop
@literal_full_done
  %si_lit_new =l add %si_loop, %need_lit
  storel %si_lit_new, %si_p
  %di_lit_new =l add %di_lit, %len_lit
  storel %di_lit_new, %di_p
  jmp @loop
@literal_trunc
  %fill_lit =l sub %cap, %di_lit
  %zero_lit_trunc =l copy 0
  storel %zero_lit_trunc, %k_p
  jmp @literal_trunc_loop
@literal_trunc_loop
  %k_lit_trunc =l loadl %k_p
  %lit_trunc_cond =w csltl %k_lit_trunc, %fill_lit
  jnz %lit_trunc_cond, @literal_trunc_body, @capfail
@literal_trunc_body
  %src_off_lit_tr =l add %si_loop, %k_lit_trunc
  %one_lit_tr =l copy 1
  %src_off_lit_tr1 =l add %src_off_lit_tr, %one_lit_tr
  %src_addr_lit_tr =l add %src, %src_off_lit_tr1
  %dst_off_lit_tr =l add %di_lit, %k_lit_trunc
  %dst_addr_lit_tr =l add %out, %dst_off_lit_tr
  %byte_lit_tr =w loadub %src_addr_lit_tr
  storeb %byte_lit_tr, %dst_addr_lit_tr
  %k_next_lit_tr =l add %k_lit_trunc, %one_lit_tr
  storel %k_next_lit_tr, %k_p
  jmp @literal_trunc_loop
@capfail
  %neg1 =l copy -1
  ret %neg1
@repeat
  %one_rep =l copy 1
  %len_rep =l sub %one_rep, %h
  %rem_rep =l sub %n, %si_loop
  %two_rep =l copy 2
  %rep_srcfail =w csltl %rem_rep, %two_rep
  jnz %rep_srcfail, @srcfail, @repeat_capcheck
@repeat_capcheck
  %di_rep =l loadl %di_p
  %di_plus_rep =l add %di_rep, %len_rep
  %rep_overflow =w csltl %cap, %di_plus_rep
  jnz %rep_overflow, @repeat_trunc, @repeat_full
@repeat_full
  %si_plus1_rep =l add %si_loop, %one_rep
  %rep_src_addr =l add %src, %si_plus1_rep
  %rep_val_full =w loadub %rep_src_addr
  %zero_rep_full =l copy 0
  storel %zero_rep_full, %k_p
  jmp @repeat_full_loop
@repeat_full_loop
  %k_rep_full =l loadl %k_p
  %rep_full_cond =w csltl %k_rep_full, %len_rep
  jnz %rep_full_cond, @repeat_full_body, @repeat_full_done
@repeat_full_body
  %dst_off_rep =l add %di_rep, %k_rep_full
  %dst_addr_rep =l add %out, %dst_off_rep
  storeb %rep_val_full, %dst_addr_rep
  %k_next_rep_full =l add %k_rep_full, %one_rep
  storel %k_next_rep_full, %k_p
  jmp @repeat_full_loop
@repeat_full_done
  %si_rep_new =l add %si_loop, %two_rep
  storel %si_rep_new, %si_p
  %di_rep_new =l add %di_rep, %len_rep
  storel %di_rep_new, %di_p
  jmp @loop
@repeat_trunc
  %fill_rep =l sub %cap, %di_rep
  %si_plus1_trunc =l add %si_loop, %one_rep
  %rep_src_addr_trunc =l add %src, %si_plus1_trunc
  %rep_val_trunc =w loadub %rep_src_addr_trunc
  %zero_rep_trunc =l copy 0
  storel %zero_rep_trunc, %k_p
  jmp @repeat_trunc_loop
@repeat_trunc_loop
  %k_rep_trunc =l loadl %k_p
  %rep_trunc_cond =w csltl %k_rep_trunc, %fill_rep
  jnz %rep_trunc_cond, @repeat_trunc_body, @capfail
@repeat_trunc_body
  %dst_off_rep_tr =l add %di_rep, %k_rep_trunc
  %dst_addr_rep_tr =l add %out, %dst_off_rep_tr
  storeb %rep_val_trunc, %dst_addr_rep_tr
  %k_next_rep_tr =l add %k_rep_trunc, %one_rep
  storel %k_next_rep_tr, %k_p
  jmp @repeat_trunc_loop
}
