export function l $qsort_pairs(l %a, l %n) {
@start
  %c0 =w cslel %n, 1
  jnz %c0, @base, @partition
@base
  %rz0 =l copy 0
  ret %rz0
@partition
  %zero =l copy 0
  %mid =l sar %n, 1
  %mid16 =l mul %mid, 16
  %pivot_ptr =l add %a, %mid16
  %pivot_ptr1 =l add %pivot_ptr, 8
  %pk0 =l loadl %pivot_ptr
  %pv0 =l loadl %pivot_ptr1
  %nm1 =l sub %n, 1
  %last_off =l mul %nm1, 16
  %last_ptr =l add %a, %last_off
  %last_ptr1 =l add %last_ptr, 8
  %lk0 =l loadl %last_ptr
  %lv0 =l loadl %last_ptr1
  storel %lk0, %pivot_ptr
  storel %lv0, %pivot_ptr1
  storel %pk0, %last_ptr
  storel %pv0, %last_ptr1
  jmp @loop
@loop
  %store =l phi @partition %zero, @next %store_next
  %i =l phi @partition %zero, @next %i_next
  %cond =w csltl %i, %nm1
  jnz %cond, @body, @after
@body
  %off =l mul %i, 16
  %ptr =l add %a, %off
  %ptr1 =l add %ptr, 8
  %k =l loadl %ptr
  %v =l loadl %ptr1
  %lt =w csltl %k, %pk0
  %eq =w ceql %k, %pk0
  %vlt =w csltl %v, %pv0
  %eqv =w and %eq, %vlt
  %less =w or %lt, %eqv
  jnz %less, @doswap, @noswap
@doswap
  %soff =l mul %store, 16
  %sptr =l add %a, %soff
  %sptr1 =l add %sptr, 8
  %sk =l loadl %sptr
  %sv =l loadl %sptr1
  storel %sk, %ptr
  storel %sv, %ptr1
  storel %k, %sptr
  storel %v, %sptr1
  %store_inc =l add %store, 1
  jmp @next
@noswap
  %store_same =l copy %store
  jmp @next
@next
  %store_next =l phi @doswap %store_inc, @noswap %store_same
  %i_next =l add %i, 1
  jmp @loop
@after
  %s_off_f =l mul %store, 16
  %s_ptr_f =l add %a, %s_off_f
  %s_ptr_f1 =l add %s_ptr_f, 8
  %skf =l loadl %s_ptr_f
  %svf =l loadl %s_ptr_f1
  %last_off2 =l mul %nm1, 16
  %last_ptr2 =l add %a, %last_off2
  %last_ptr2_1 =l add %last_ptr2, 8
  %lkf =l loadl %last_ptr2
  %lvf =l loadl %last_ptr2_1
  storel %lkf, %s_ptr_f
  storel %lvf, %s_ptr_f1
  storel %skf, %last_ptr2
  storel %svf, %last_ptr2_1
  %tmp1 =l sub %n, %store
  %right_size =l sub %tmp1, 1
  %store_p1 =l add %store, 1
  %right_off =l mul %store_p1, 16
  %right_ptr =l add %a, %right_off
  %r1 =l call $qsort_pairs(l %a, l %store)
  %r2 =l call $qsort_pairs(l %right_ptr, l %right_size)
  %rsum =l add %r1, %r2
  ret %rsum
}
