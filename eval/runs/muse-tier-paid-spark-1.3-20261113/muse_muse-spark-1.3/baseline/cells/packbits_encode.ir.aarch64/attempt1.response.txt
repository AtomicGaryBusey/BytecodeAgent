export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
  %i0 =l copy 0
  %o0 =l copy 0
  jmp @outer_cond
@outer_cond
  %i =l phi @start %i0, @repeat_write %i_rep_next, @lit_next %i_lit_next
  %o =l phi @start %o0, @repeat_write %o_rep_next, @lit_next %o_lit_next
  %c_outer =w csltl %i, %n
  jnz %c_outer, @run_init, @done
@done
  ret %o
@run_init
  %p_i =l add %src, %i
  %b0 =w loadub %p_i
  %run0 =l copy 1
  jmp @run_cond
@run_cond
  %run =l phi @run_init %run0, @run_inc %run_next
  %c_run1 =w csltl %run, 128
  jnz %c_run1, @run_check2, @run_done
@run_check2
  %tmp =l add %i, %run
  %c_run2 =w csltl %tmp, %n
  jnz %c_run2, @run_check3, @run_done
@run_check3
  %p_tmp =l add %src, %tmp
  %b_tmp =w loadub %p_tmp
  %c_eq =w ceqw %b_tmp, %b0
  jnz %c_eq, @run_inc, @run_done
@run_inc
  %run_next =l add %run, 1
  jmp @run_cond
@run_done
  %c_lt3 =w csltl %run, 3
  jnz %c_lt3, @lit_init, @repeat_check
@repeat_check
  %need_rep =l add %o, 2
  %c_rep_nofit =w csltl %cap, %need_rep
  jnz %c_rep_nofit, @fail, @repeat_write
@fail
  %neg =l copy -1
  ret %neg
@repeat_write
  %run_w =w copy %run
  %c257 =w copy 257
  %hdr_rep =w sub %c257, %run_w
  %out_p =l add %out, %o
  storeb %hdr_rep, %out_p
  %out_p1 =l add %out_p, 1
  storeb %b0, %out_p1
  %o_rep_next =l add %o, 2
  %i_rep_next =l add %i, %run
  jmp @outer_cond
@lit_init
  %j0 =l copy %i
  jmp @lit_cond
@lit_cond
  %j =l phi @lit_init %j0, @lit_adv %j_next
  %c_lit1 =w csltl %j, %n
  jnz %c_lit1, @lit_check2, @lit_done
@lit_check2
  %diff =l sub %j, %i
  %c_lit2 =w csltl %diff, 128
  jnz %c_lit2, @lit_run_init, @lit_done
@lit_run_init
  %p_j =l add %src, %j
  %bj0 =w loadub %p_j
  %r2_0 =l copy 1
  jmp @lit_run_cond
@lit_run_cond
  %r2 =l phi @lit_run_init %r2_0, @lit_run_inc %r2_next
  %c_r2_1 =w csltl %r2, 128
  jnz %c_r2_1, @lit_run_check2, @lit_run_done
@lit_run_check2
  %tmp2 =l add %j, %r2
  %c_r2_2 =w csltl %tmp2, %n
  jnz %c_r2_2, @lit_run_check3, @lit_run_done
@lit_run_check3
  %p_tmp2 =l add %src, %tmp2
  %b_tmp2 =w loadub %p_tmp2
  %c_eq2 =w ceqw %b_tmp2, %bj0
  jnz %c_eq2, @lit_run_inc, @lit_run_done
@lit_run_inc
  %r2_next =l add %r2, 1
  jmp @lit_run_cond
@lit_run_done
  %c_r2_lt3 =w csltl %r2, 3
  jnz %c_r2_lt3, @lit_adv, @lit_done
@lit_adv
  %j_next =l add %j, 1
  jmp @lit_cond
@lit_done
  %llen =l sub %j, %i
  %need_lit_tmp =l add %o, 1
  %need_lit =l add %need_lit_tmp, %llen
  %c_lit_nofit =w csltl %cap, %need_lit
  jnz %c_lit_nofit, @fail, @lit_write_init
@lit_write_init
  %llenw =w copy %llen
  %hdr_lit =w sub %llenw, 1
  %out_p_lit =l add %out, %o
  storeb %hdr_lit, %out_p_lit
  %k0 =l copy 0
  jmp @lit_copy_cond
@lit_copy_cond
  %k =l phi @lit_write_init %k0, @lit_copy_body %k_next
  %c_copy =w csltl %k, %llen
  jnz %c_copy, @lit_copy_body, @lit_next
@lit_copy_body
  %src_off =l add %i, %k
  %src_p_k =l add %src, %src_off
  %b_k =w loadub %src_p_k
  %dst_off1 =l add %o, 1
  %dst_off =l add %dst_off1, %k
  %dst_p_k =l add %out, %dst_off
  storeb %b_k, %dst_p_k
  %k_next =l add %k, 1
  jmp @lit_copy_cond
@lit_next
  %o_lit_next_tmp =l add %o, 1
  %o_lit_next =l add %o_lit_next_tmp, %llen
  %i_lit_next =l copy %j
  jmp @outer_cond
}
