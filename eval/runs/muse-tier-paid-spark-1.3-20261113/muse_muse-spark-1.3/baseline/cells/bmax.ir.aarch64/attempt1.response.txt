export function l $bmax(l %a, l %n) {
@start
  %m0 =l loadl %a
  jmp @loop
@loop
  %i =l phi @start 1, @cont %inext
  %m =l phi @start %m0, @cont %mnext
  %c =w csltl %i, %n
  jnz %c, @body, @done
@body
  %off =l mul %i, 8
  %p =l add %a, %off
  %v =l loadl %p
  %g =w csgtl %v, %m
  jnz %g, @greater, @less
@greater
  jmp @cont
@less
  jmp @cont
@cont
  %mnext =l phi @greater %v, @less %m
  %inext =l add %i, 1
  jmp @loop
@done
  ret %m
}
