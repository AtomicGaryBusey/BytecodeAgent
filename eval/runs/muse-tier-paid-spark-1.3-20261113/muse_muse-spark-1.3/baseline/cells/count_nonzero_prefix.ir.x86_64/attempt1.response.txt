export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @inc %inext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%iszero =w ceql %v, 0
	jnz %iszero, @done, @inc
@inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
