export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%cnt =l phi @start 0, @body %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%soff =l mul %i, 8
	%saddr =l add %src, %soff
	%x =l loadl %saddr
	%r =l call %f(l %x)
	%doff =l mul %i, 8
	%daddr =l add %dst, %doff
	storel %r, %daddr
	%lt =w csltl %r, %x
	%ltl =l extuw %lt
	%cnt2 =l add %cnt, %ltl
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
