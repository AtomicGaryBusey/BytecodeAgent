export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%n_minus1 =l sub %n, %one
	%c_start =w cslel %n, %one
	jnz %c_start, @ret, @outer_head
@outer_head
	%i =l phi @start %zero, @outer_latch %i_next
	%outer_cond =w csltl %i, %n_minus1
	jnz %outer_cond, @inner_setup, @ret
@inner_setup
	%inner_limit =l sub %n_minus1, %i
	jmp @inner_head
@inner_head
	%j =l phi @inner_setup %zero, @inner_latch %j_next
	%inner_cond =w csltl %j, %inner_limit
	jnz %inner_cond, @inner_body, @outer_latch
@inner_body
	%off_j =l mul %j, %eight
	%addr_j =l add %a, %off_j
	%val_j =l loadl %addr_j
	%j1 =l add %j, %one
	%off_j1 =l mul %j1, %eight
	%addr_j1 =l add %a, %off_j1
	%val_j1 =l loadl %addr_j1
	%swap_cond =w csgtl %val_j, %val_j1
	jnz %swap_cond, @swap, @noswap
@swap
	storel %val_j1, %addr_j
	storel %val_j, %addr_j1
	jmp @inner_latch
@noswap
	jmp @inner_latch
@inner_latch
	%j_next =l add %j, %one
	jmp @inner_head
@outer_latch
	%i_next =l add %i, %one
	jmp @outer_head
@ret
	ret %zero
}
