export function l $packbits_encode(l %src0, l %n0, l %out0, l %cap0) {
@start
  %slot_i =l alloc8 8
  %slot_o =l alloc8 8
  %slot_j =l alloc8 8
  %slot_k =l alloc8 8
  %zero_init_i =l copy 0
  storel %zero_init_i, %slot_i
  %zero_init_o =l copy 0
  storel %zero_init_o, %slot_o
  jmp @outer_check
@outer_check
  %i_outer =l loadl %slot_i
  %c_outer =w csltl %i_outer, %n0
  jnz %c_outer, @outer_body, @outer_end
@outer_end
  %o_end =l loadl %slot_o
  ret %o_end
@outer_body
  %i_ob =l loadl %slot_i
  %i_ob_p2 =l add %i_ob, 2
  %c_ob_has3 =w csltl %i_ob_p2, %n0
  jnz %c_ob_has3, @check_bytes, @literal_init
@check_bytes
  %i_cb =l loadl %slot_i
  %p_cb0 =l add %src0, %i_cb
  %b_cb0 =w loadub %p_cb0
  %i_cb_p1 =l add %i_cb, 1
  %p_cb1 =l add %src0, %i_cb_p1
  %b_cb1 =w loadub %p_cb1
  %eq_cb01 =w ceqw %b_cb0, %b_cb1
  jnz %eq_cb01, @check_bytes2, @literal_init
@check_bytes2
  %i_cb_p2 =l add %i_cb, 2
  %p_cb2 =l add %src0, %i_cb_p2
  %b_cb2 =w loadub %p_cb2
  %eq_cb12 =w ceqw %b_cb1, %b_cb2
  jnz %eq_cb12, @repeat_init, @literal_init
@repeat_init
  %i_ri =l loadl %slot_i
  storel %i_ri, %slot_k
  jmp @rep_check
@rep_check
  %k_rc =l loadl %slot_k
  %i_rc =l loadl %slot_i
  %c_rc_a =w csltl %k_rc, %n0
  %d_rc =l sub %k_rc, %i_rc
  %c_rc_b =w csltl %d_rc, 128
  %c_rc_bounds =w and %c_rc_a, %c_rc_b
  %c_rc_bounds_l =l extuw %c_rc_bounds
  %off_rc =l mul %d_rc, %c_rc_bounds_l
  %k_safe_rc =l add %i_rc, %off_rc
  %p_safe_rc =l add %src0, %k_safe_rc
  %b_safe_rc =w loadub %p_safe_rc
  %p_i_rc =l add %src0, %i_rc
  %b_i_rc =w loadub %p_i_rc
  %c_rc_eq =w ceqw %b_safe_rc, %b_i_rc
  %c_rc_cont =w and %c_rc_eq, %c_rc_bounds
  jnz %c_rc_cont, @rep_next, @rep_done
@rep_next
  %k_rn =l loadl %slot_k
  %k_rn_next =l add %k_rn, 1
  storel %k_rn_next, %slot_k
  jmp @rep_check
@rep_done
  %k_rd =l loadl %slot_k
  %i_rd =l loadl %slot_i
  %r_rd =l sub %k_rd, %i_rd
  %o_rd =l loadl %slot_o
  %o_rd_p2 =l add %o_rd, 2
  %c_rd_fit =w cslel %o_rd_p2, %cap0
  jnz %c_rd_fit, @rep_emit, @ret_neg1
@rep_emit
  %c257_re =w copy 257
  %r_re_w =w copy %r_rd
  %hdr_re_w =w sub %c257_re, %r_re_w
  %out_o_re =l add %out0, %o_rd
  storeb %hdr_re_w, %out_o_re
  %o_rd_p1 =l add %o_rd, 1
  %out_o1_re =l add %out0, %o_rd_p1
  %p_i_re =l add %src0, %i_rd
  %b_re =w loadub %p_i_re
  storeb %b_re, %out_o1_re
  %o_rd_new =l add %o_rd, 2
  storel %o_rd_new, %slot_o
  %i_rd_new =l add %i_rd, %r_rd
  storel %i_rd_new, %slot_i
  jmp @outer_check
@ret_neg1
  %neg1 =l copy -1
  ret %neg1
@literal_init
  %i_li =l loadl %slot_i
  storel %i_li, %slot_j
  jmp @lit_check1
@lit_check1
  %j_lc1 =l loadl %slot_j
  %c_lc1 =w csltl %j_lc1, %n0
  jnz %c_lc1, @lit_check2, @lit_done
@lit_check2
  %i_lc2 =l loadl %slot_i
  %d_lc2 =l sub %j_lc1, %i_lc2
  %c_lc2 =w csltl %d_lc2, 128
  jnz %c_lc2, @lit_check3, @lit_done
@lit_check3
  %j_lc1_p2 =l add %j_lc1, 2
  %c_lc3_has3 =w csltl %j_lc1_p2, %n0
  jnz %c_lc3_has3, @lit_check_bytes, @lit_next
@lit_check_bytes
  %p_lb0 =l add %src0, %j_lc1
  %b_lb0 =w loadub %p_lb0
  %j_lc1_p1 =l add %j_lc1, 1
  %p_lb1 =l add %src0, %j_lc1_p1
  %b_lb1 =w loadub %p_lb1
  %eq_lb01 =w ceqw %b_lb0, %b_lb1
  jnz %eq_lb01, @lit_check_bytes2, @lit_next
@lit_check_bytes2
  %j_lc1_p2_b =l add %j_lc1, 2
  %p_lb2 =l add %src0, %j_lc1_p2_b
  %b_lb2 =w loadub %p_lb2
  %eq_lb12 =w ceqw %b_lb1, %b_lb2
  jnz %eq_lb12, @lit_done, @lit_next
@lit_next
  %j_ln =l loadl %slot_j
  %j_ln_next =l add %j_ln, 1
  storel %j_ln_next, %slot_j
  jmp @lit_check1
@lit_done
  %j_ld =l loadl %slot_j
  %i_ld =l loadl %slot_i
  %L_ld =l sub %j_ld, %i_ld
  %o_ld =l loadl %slot_o
  %o_ld_p1 =l add %o_ld, 1
  %o_ld_p1_L =l add %o_ld_p1, %L_ld
  %c_ld_fit =w cslel %o_ld_p1_L, %cap0
  jnz %c_ld_fit, @lit_emit, @ret_neg1
@lit_emit
  %L_le_w =w copy %L_ld
  %c1_le =w copy 1
  %hdr_le_w =w sub %L_le_w, %c1_le
  %out_o_le =l add %out0, %o_ld
  storeb %hdr_le_w, %out_o_le
  %zero_le =l copy 0
  storel %zero_le, %slot_k
  jmp @copy_check
@copy_check
  %k_cc =l loadl %slot_k
  %c_cc =w csltl %k_cc, %L_ld
  jnz %c_cc, @copy_body, @copy_done
@copy_body
  %i_ld_k =l add %i_ld, %k_cc
  %p_src_cb =l add %src0, %i_ld_k
  %b_cb_copy =w loadub %p_src_cb
  %o_ld_p1_k =l add %o_ld_p1, %k_cc
  %p_dst_cb =l add %out0, %o_ld_p1_k
  storeb %b_cb_copy, %p_dst_cb
  %k_cc_next =l add %k_cc, 1
  storel %k_cc_next, %slot_k
  jmp @copy_check
@copy_done
  storel %o_ld_p1_L, %slot_o
  storel %j_ld, %slot_i
  jmp @outer_check
}
