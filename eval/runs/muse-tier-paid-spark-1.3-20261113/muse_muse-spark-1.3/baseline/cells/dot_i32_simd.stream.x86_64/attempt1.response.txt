dot_i32_simd:
pxor xmm0, xmm0
mov rcx, rdx
shr rcx, 2
test rcx, rcx
jz .Lreduce
.Lvloop:
movdqu xmm1, xmmword ptr [rdi]
movdqu xmm2, xmmword ptr [rsi]
movdqa xmm3, xmm1
pmuldq xmm1, xmm2
paddq xmm0, xmm1
pshufd xmm3, xmm3, 177
pshufd xmm2, xmm2, 177
pmuldq xmm3, xmm2
paddq xmm0, xmm3
add rdi, 16
add rsi, 16
dec rcx
jnz .Lvloop
.Lreduce:
movdqa xmm1, xmm0
psrldq xmm1, 8
paddq xmm0, xmm1
movq rax, xmm0
mov rcx, rdx
and rcx, 3
jz .Ldone
.Lscalar:
movsxd r8, dword ptr [rdi]
movsxd r9, dword ptr [rsi]
imul r8, r9
add rax, r8
add rdi, 4
add rsi, 4
dec rcx
jnz .Lscalar
.Ldone:
ret
