binary_search:
mov rcx, 0
mov r8, rsi
.Lloop:
cmp rcx, r8
jge .Ldone
mov rax, r8
sub rax, rcx
shr rax, 1
add rax, rcx
mov r9, qword ptr [rdi + rax*8]
cmp r9, rdx
jl .Lless
mov r8, rax
jmp .Lloop
.Lless:
lea rcx, [rax + 1]
jmp .Lloop
.Ldone:
cmp rcx, rsi
jge .Lnotfound
mov rax, qword ptr [rdi + rcx*8]
cmp rax, rdx
je .Lfound
.Lnotfound:
mov rax, -1
ret
.Lfound:
mov rax, rcx
ret
