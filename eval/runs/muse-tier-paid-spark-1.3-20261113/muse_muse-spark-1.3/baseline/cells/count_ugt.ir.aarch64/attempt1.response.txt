export function l $count_ugt(l %a, l %n, l %t) {
@start
	%z =l copy 0
	jmp @loop
@loop
	%i =l phi @start %z, @yes %i1, @no %i1
	%c =l phi @start %z, @yes %c1, @no %c
	%done =w csltl %i, %n
	jnz %done, @body, @end
@body
	%o =l mul %i, 8
	%p =l add %a, %o
	%v =l loadl %p
	%g =w cugtl %v, %t
	%i1 =l add %i, 1
	jnz %g, @yes, @no
@yes
	%c1 =l add %c, 1
	jmp @loop
@no
	jmp @loop
@end
	ret %c
}
