export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zero_head
@zero_head
	%zi =l phi @start 0, @zero_body %zi_next
	%zc =w cslel %zi, %n
	jnz %zc, @zero_body, @outer_head
@zero_body
	%zoff =l mul %zi, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%zi_next =l add %zi, 1
	jmp @zero_head
@outer_head
	%i =l phi @zero_head 0, @outer_latch %i_next
	%oc =w csltl %i, %m
	jnz %oc, @inner_head, @done
@inner_head
	%j =l phi @outer_head 1, @inner_store %j_next
	%diag =l phi @outer_head 0, @inner_store %up
	%ic =w cslel %j, %n
	jnz %ic, @inner_body, @outer_latch
@inner_body
	%joff =l mul %j, 8
	%row_j_ptr =l add %row, %joff
	%up =l loadl %row_j_ptr
	%jm1 =l sub %j, 1
	%yptr =l add %y, %jm1
	%yv =w loadub %yptr
	%xptr =l add %x, %i
	%xv =w loadub %xptr
	%eq =w ceqw %xv, %yv
	jnz %eq, @do_match, @do_nomatch
@do_match
	%match_val =l add %diag, 1
	jmp @inner_store
@do_nomatch
	%jm1off =l mul %jm1, 8
	%row_jm1_ptr =l add %row, %jm1off
	%prev =l loadl %row_jm1_ptr
	%lt =w csltl %up, %prev
	jnz %lt, @take_prev, @take_up
@take_prev
	jmp @inner_store
@take_up
	jmp @inner_store
@inner_store
	%new =l phi @do_match %match_val, @take_prev %prev, @take_up %up
	storel %new, %row_j_ptr
	%j_next =l add %j, 1
	jmp @inner_head
@outer_latch
	%i_next =l add %i, 1
	jmp @outer_head
@done
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%res =l loadl %nptr
	ret %res
}
