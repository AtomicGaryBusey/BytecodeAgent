export function l $shift_sat(l %x, l %n) {
@start
%zero =l copy 0
jmp @head
@head
%left =l phi @start %x, @body %left1
%right =l phi @start %x, @body %right1
%i =l phi @start %zero, @body %i1
%c =w csltl %i, %n
jnz %c, @body, @done
@body
%left1 =l shl %left, 1
%right1 =l sar %right, 1
%i1 =l add %i, 1
jmp @head
@done
%r =l add %left, %right
ret %r
}
