export function l $fib(l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%a =l phi @start 0, @body %anext
	%b =l phi @start 1, @body %bnext
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%next =l add %a, %b
	%anext =l copy %b
	%bnext =l copy %next
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %a
}
