export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%oi =l mul %i, 8
	%ai =l add %a, %oi
	%oj =l mul %j, 8
	%aj =l add %a, %oj
	%vi =l loadl %ai
	%vj =l loadl %aj
	storel %vj, %ai
	storel %vi, %aj
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
