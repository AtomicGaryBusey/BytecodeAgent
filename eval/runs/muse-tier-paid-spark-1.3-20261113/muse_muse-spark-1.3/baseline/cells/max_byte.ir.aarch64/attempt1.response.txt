export function l $max_byte(l %p, l %n) {
@start
	%t0 =w cslel %n, 0
	jnz %t0, @ret0, @init
@ret0
	ret 0
@init
	%m0 =l loadub %p
	jmp @loop
@loop
	%i =l phi @init 1, @cont %inext
	%m =l phi @init %m0, @cont %mnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %m
	jnz %gt, @greater, @less
@greater
	jmp @cont
@less
	jmp @cont
@cont
	%mnext =l phi @greater %b, @less %m
	%inext =l add %i, 1
	jmp @loop
@done
	ret %m
}
