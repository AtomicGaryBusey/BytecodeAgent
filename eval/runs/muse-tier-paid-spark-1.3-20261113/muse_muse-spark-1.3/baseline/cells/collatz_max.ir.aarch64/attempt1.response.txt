export function l $collatz_max(l %n) {
@start
	jmp @outer_cond
@outer_cond
	%k =l phi @start 1, @outer_next %k_next
	%best =l phi @start 0, @outer_next %best_next
	%cond_outer =w cslel %k, %n
	jnz %cond_outer, @outer_body, @outer_done
@outer_body
	%x_init =l copy %k
	jmp @inner_cond
@inner_cond
	%x =l phi @outer_body %x_init, @inner_merge %x_next
	%steps =l phi @outer_body 0, @inner_merge %steps_next
	%cond_inner =w cnel %x, 1
	jnz %cond_inner, @inner_body, @inner_done
@inner_body
	%t =l and %x, 1
	%is_odd =w cnel %t, 0
	jnz %is_odd, @odd, @even
@even
	%x_even =l udiv %x, 2
	jmp @inner_merge
@odd
	%m =l mul %x, 3
	%x_odd =l add %m, 1
	jmp @inner_merge
@inner_merge
	%x_next =l phi @even %x_even, @odd %x_odd
	%steps_next =l add %steps, 1
	jmp @inner_cond
@inner_done
	%is_better =w csgtl %steps, %best
	jnz %is_better, @take_steps, @take_best
@take_steps
	%best_from_steps =l copy %steps
	jmp @outer_next
@take_best
	%best_from_best =l copy %best
	jmp @outer_next
@outer_next
	%best_next =l phi @take_steps %best_from_steps, @take_best %best_from_best
	%k_next =l add %k, 1
	jmp @outer_cond
@outer_done
	ret %best
}
