export function l $count_nonzero_prefix(l %a, l %n) {
@start
  %zero =l copy 0
  jmp @loop
@loop
  %i =l phi @start %zero, @next %inext
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %off =l mul %i, 8
  %addr =l add %a, %off
  %val =l loadl %addr
  %iszero =w ceql %val, 0
  jnz %iszero, @done, @next
@next
  %inext =l add %i, 1
  jmp @loop
@done
  %res =l phi @loop %i, @body %i
  ret %res
}
