export function l $bmin_empty(l %a, l %n) {
@start
  %is_empty =w ceql %n, 0
  jnz %is_empty, @empty, @head_init
@empty
  ret 9223372036854775807
@head_init
  %m0 =l loadl %a
  %one =l copy 1
  jmp @loop
@loop
  %i =l phi @head_init %one, @tail %i_next
  %m =l phi @head_init %m0, @tail %m_tail
  %cont =w csltl %i, %n
  jnz %cont, @body, @exit
@body
  %off =l mul %i, 8
  %addr =l add %a, %off
  %v =l loadl %addr
  %isless =w csltl %v, %m
  jnz %isless, @less, @nless
@less
  jmp @tail
@nless
  jmp @tail
@tail
  %m_tail =l phi @less %v, @nless %m
  %i_next =l add %i, 1
  jmp @loop
@exit
  ret %m
}
