export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %zero =l copy 0
  %n2 =l mul %n, %n
  %n2x8 =l mul %n2, 8
  %tmp1base =l add %tmp, %n2x8
  %tmptotal =l mul %n2, 2
  jmp @id_loop
@id_loop
  %id_idx =l phi @start 0, @id_body %id_next
  %id_cond =w csltl %id_idx, %n2
  jnz %id_cond, @id_body, @copy_loop
@id_body
  %id_i =l sdiv %id_idx, %n
  %id_j =l srem %id_idx, %n
  %id_eq =w ceql %id_i, %id_j
  %id_val =l extuw %id_eq
  %id_off =l mul %id_idx, 8
  %id_addr =l add %out, %id_off
  storel %id_val, %id_addr
  %id_next =l add %id_idx, 1
  jmp @id_loop
@copy_loop
  %cp_idx =l phi @id_loop 0, @copy_body %cp_next
  %cp_cond =w csltl %cp_idx, %n2
  jnz %cp_cond, @copy_body, @pow_check
@copy_body
  %cp_off =l mul %cp_idx, 8
  %cp_src =l add %a, %cp_off
  %cp_dst =l add %tmp, %cp_off
  %cp_val =l loadl %cp_src
  storel %cp_val, %cp_dst
  %cp_next =l add %cp_idx, 1
  jmp @copy_loop
@pow_check
  %p =l phi @copy_loop 0, @pow_inc %p_next
  %p_cond =w csltl %p, %k
  jnz %p_cond, @mm_outer, @zero_loop
@mm_outer
  %mm_idx =l phi @pow_check 0, @mm_store %mm_next
  %mm_cond =w csltl %mm_idx, %n2
  jnz %mm_cond, @t_loop, @copyback_loop
@t_loop
  %t =l phi @mm_outer 0, @t_body %t_next
  %sum =l phi @mm_outer 0, @t_body %sum_next
  %t_cond =w csltl %t, %n
  jnz %t_cond, @t_body, @mm_store
@t_body
  %mm_i =l sdiv %mm_idx, %n
  %mm_j =l srem %mm_idx, %n
  %in_row =l mul %mm_i, %n
  %a_idx =l add %in_row, %t
  %a_off =l mul %a_idx, 8
  %a_addr =l add %out, %a_off
  %x =l loadl %a_addr
  %b_row =l mul %t, %n
  %b_idx =l add %b_row, %mm_j
  %b_off =l mul %b_idx, 8
  %b_addr =l add %tmp, %b_off
  %y =l loadl %b_addr
  %prod =l mul %x, %y
  %sum_next =l add %sum, %prod
  %t_next =l add %t, 1
  jmp @t_loop
@mm_store
  %st_off =l mul %mm_idx, 8
  %st_addr =l add %tmp1base, %st_off
  storel %sum, %st_addr
  %mm_next =l add %mm_idx, 1
  jmp @mm_outer
@copyback_loop
  %cb_idx =l phi @mm_outer 0, @copyback_body %cb_next
  %cb_cond =w csltl %cb_idx, %n2
  jnz %cb_cond, @copyback_body, @pow_inc
@copyback_body
  %cb_off =l mul %cb_idx, 8
  %cb_src =l add %tmp1base, %cb_off
  %cb_dst =l add %out, %cb_off
  %cb_val =l loadl %cb_src
  storel %cb_val, %cb_dst
  %cb_next =l add %cb_idx, 1
  jmp @copyback_loop
@pow_inc
  %p_next =l add %p, 1
  jmp @pow_check
@zero_loop
  %z_idx =l phi @pow_check 0, @zero_body %z_next
  %z_cond =w csltl %z_idx, %tmptotal
  jnz %z_cond, @zero_body, @done
@zero_body
  %z_off =l mul %z_idx, 8
  %z_addr =l add %tmp, %z_off
  storel %zero, %z_addr
  %z_next =l add %z_idx, 1
  jmp @zero_loop
@done
  ret %zero
}
