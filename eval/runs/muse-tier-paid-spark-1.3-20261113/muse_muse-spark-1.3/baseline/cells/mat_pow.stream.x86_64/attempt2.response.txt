mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rcx
mov r15, r8
mov rbx, rdx
mov rax, r13
imul rax, r13
mov rbp, rax
xor r10, r10
.Li_out:
cmp r10, r13
jge .Lout_done
xor r11, r11
.Lj_out:
cmp r11, r13
jge .Li_out_next
mov rax, r10
imul rax, r13
add rax, r11
cmp r10, r11
jne .Lout_zero
mov rdx, 1
jmp .Lout_store
.Lout_zero:
xor rdx, rdx
.Lout_store:
mov qword ptr [r14 + rax*8], rdx
inc r11
jmp .Lj_out
.Li_out_next:
inc r10
jmp .Li_out
.Lout_done:
test rbx, rbx
jz .Lzero_tmp
xor rax, rax
.Lcopy:
cmp rax, rbp
jge .Lcopy_done
mov rdx, qword ptr [r12 + rax*8]
mov qword ptr [r15 + rax*8], rdx
inc rax
jmp .Lcopy
.Lcopy_done:
.Lpow_loop:
test rbx, rbx
jz .Lzero_tmp
lea rsi, [r15 + rbp*8]
xor r10, r10
.Li:
cmp r10, r13
jge .Lcopy_back
xor r11, r11
.Lj:
cmp r11, r13
jge .Li_next
xor r8, r8
xor r9, r9
.Lt:
cmp r9, r13
jge .Lstore_c
mov rax, r10
imul rax, r13
add rax, r9
mov rax, qword ptr [r14 + rax*8]
mov rdx, r9
imul rdx, r13
add rdx, r11
mov rdx, qword ptr [r15 + rdx*8]
imul rax, rdx
add r8, rax
inc r9
jmp .Lt
.Lstore_c:
mov rax, r10
imul rax, r13
add rax, r11
mov qword ptr [rsi + rax*8], r8
inc r11
jmp .Lj
.Li_next:
inc r10
jmp .Li
.Lcopy_back:
xor rax, rax
.Lcb:
cmp rax, rbp
jge .Lcb_done
mov rdx, qword ptr [rsi + rax*8]
mov qword ptr [r14 + rax*8], rdx
inc rax
jmp .Lcb
.Lcb_done:
dec rbx
jmp .Lpow_loop
.Lzero_tmp:
mov rax, rbp
add rax, rbp
xor rdx, rdx
.Lz:
cmp rdx, rax
jge .Ldone
mov qword ptr [r15 + rdx*8], 0
inc rdx
jmp .Lz
.Ldone:
xor rax, rax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
