export function l $collatz_max(l %n) {
@start
	%maxp =l alloc8 8
	%kp =l alloc8 8
	%xp =l alloc8 8
	%stepsp =l alloc8 8
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	%three =l copy 3
	storel %zero, %maxp
	storel %one, %kp
	jmp @outer_cond
@outer_cond
	%k0 =l loadl %kp
	%oc =w cslel %k0, %n
	jnz %oc, @outer_body, @outer_done
@outer_body
	%k_init =l loadl %kp
	storel %k_init, %xp
	storel %zero, %stepsp
	jmp @inner_cond
@inner_cond
	%x_cond =l loadl %xp
	%ic =w cnel %x_cond, %one
	jnz %ic, @inner_body, @inner_done
@inner_body
	%xb =l loadl %xp
	%odd_bit =l and %xb, %one
	%is_even =w ceql %odd_bit, %zero
	jnz %is_even, @even, @odd
@even
	%xe =l loadl %xp
	%xe2 =l div %xe, %two
	storel %xe2, %xp
	jmp @inner_next
@odd
	%xo =l loadl %xp
	%xo3 =l mul %xo, %three
	%xo4 =l add %xo3, %one
	storel %xo4, %xp
	jmp @inner_next
@inner_next
	%s =l loadl %stepsp
	%s1 =l add %s, %one
	storel %s1, %stepsp
	jmp @inner_cond
@inner_done
	%steps =l loadl %stepsp
	%max =l loadl %maxp
	%gt =w csgtl %steps, %max
	jnz %gt, @update_max, @no_update
@update_max
	%steps2 =l loadl %stepsp
	storel %steps2, %maxp
	jmp @outer_next
@no_update
	jmp @outer_next
@outer_next
	%kold =l loadl %kp
	%knew =l add %kold, %one
	storel %knew, %kp
	jmp @outer_cond
@outer_done
	%res =l loadl %maxp
	ret %res
}
