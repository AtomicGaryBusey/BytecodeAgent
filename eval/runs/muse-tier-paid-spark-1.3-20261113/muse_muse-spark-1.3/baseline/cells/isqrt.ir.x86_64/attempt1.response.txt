export function l $isqrt(l %n) {
@start
  %zero =l copy 0
  %max =l copy 4294967295
  jmp @loop
@loop
  %lo =l phi @start %zero, @then %lo_t, @else %lo_e
  %hi =l phi @start %max, @then %hi_t, @else %hi_e
  %ans =l phi @start %zero, @then %ans_t, @else %ans_e
  %c =w culel %lo, %hi
  jnz %c, @body, @done
@body
  %sum =l add %lo, %hi
  %mid =l shr %sum, 1
  %sq =l mul %mid, %mid
  %le =w culel %sq, %n
  jnz %le, @then, @else
@then
  %lo_t =l add %mid, 1
  %hi_t =l copy %hi
  %ans_t =l copy %mid
  jmp @loop
@else
  %hi_e =l sub %mid, 1
  %lo_e =l copy %lo
  %ans_e =l copy %ans
  jmp @loop
@done
  ret %ans
}
