export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%c0 =l copy 0
	%c1 =l copy 1
	%neg1 =l sub %c0, %c1
	%is0 =w ceql %b, 0
	jnz %is0, @div0, @chk
@div0
	storel %a, %rem
	ret %c0
@chk
	%isneg1 =w ceql %b, %neg1
	jnz %isneg1, @chk2, @div
@chk2
	%nega =l sub %c0, %a
	%eq =w ceql %a, %nega
	%nonz =w cnel %a, 0
	%ismin =w and %eq, %nonz
	jnz %ismin, @ovf, @div
@ovf
	storel %c0, %rem
	ret %a
@div
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
