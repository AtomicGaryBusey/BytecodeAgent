export function l $count_ugt(l %a, l %n, l %t) {
@start
  %i0 =l copy 0
  %c0 =l copy 0
  jmp @check
@check
  %i =l phi @start %i0, @join %i2
  %c =l phi @start %c0, @join %c2
  %cond =w csltl %i, %n
  jnz %cond, @body, @done
@body
  %off =l mul %i, 8
  %ptr =l add %a, %off
  %v =l loadl %ptr
  %gt =w cugtl %v, %t
  jnz %gt, @yes, @no
@yes
  %c1 =l add %c, 1
  jmp @join
@no
  jmp @join
@join
  %c2 =l phi @yes %c1, @no %c
  %i2 =l add %i, 1
  jmp @check
@done
  ret %c
}
