export function l $reverse(l %a, l %n) {
@start
  %zero =l copy 0
  %j0 =l sub %n, 1
  jmp @loop
@loop
  %i =l phi @start %zero, @body %i_next
  %j =l phi @start %j0, @body %j_next
  %c =w csltl %i, %j
  jnz %c, @body, @done
@body
  %io =l mul %i, 8
  %pa =l add %a, %io
  %va =l loadl %pa
  %jo =l mul %j, 8
  %pb =l add %a, %jo
  %vb =l loadl %pb
  storel %vb, %pa
  storel %va, %pb
  %i_next =l add %i, 1
  %j_next =l sub %j, 1
  jmp @loop
@done
  %r =l copy 0
  ret %r
}
