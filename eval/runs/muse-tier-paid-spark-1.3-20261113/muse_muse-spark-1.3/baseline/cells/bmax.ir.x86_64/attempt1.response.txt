export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
	jmp @check
@check
	%i =l phi @start %i0, @next %i1
	%m =l phi @start %m0, @next %m1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%g =w csgtl %v, %m
	jnz %g, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%m1 =l phi @update %v, @keep %m
	%i1 =l add %i, 1
	jmp @check
@done
	ret %m
}
