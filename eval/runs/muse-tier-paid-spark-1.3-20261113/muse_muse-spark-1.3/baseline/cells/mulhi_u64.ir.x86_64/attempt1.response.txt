export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a1, %b0
	%p2 =l mul %a0, %b1
	%p3 =l mul %a1, %b1
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, 4294967295
	%p1hi =l shr %p1, 32
	%p2lo =l and %p2, 4294967295
	%p2hi =l shr %p2, 32
	%p3lo =l and %p3, 4294967295
	%p3hi =l shr %p3, 32
	%c1a =l add %p0hi, %p1lo
	%c1 =l add %c1a, %p2lo
	%c1carry =l shr %c1, 32
	%c2a =l add %p1hi, %p2hi
	%c2b =l add %c2a, %p3lo
	%c2 =l add %c2b, %c1carry
	%c2lo =l and %c2, 4294967295
	%c2carry =l shr %c2, 32
	%c3 =l add %p3hi, %c2carry
	%hi1 =l mul %c3, 4294967296
	%hi =l add %hi1, %c2lo
	ret %hi
}
