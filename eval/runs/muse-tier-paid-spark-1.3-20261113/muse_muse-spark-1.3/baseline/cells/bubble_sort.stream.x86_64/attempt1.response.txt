bubble_sort:
cmp rsi, 1
jle .Ldone
mov rcx, rsi
dec rcx
.Louter:
mov rdx, 0
.Linner:
cmp rdx, rcx
jge .Linner_done
mov r8, qword ptr [rdi + rdx*8]
mov r9, qword ptr [rdi + rdx*8 + 8]
cmp r8, r9
jle .Lnoswap
mov qword ptr [rdi + rdx*8], r9
mov qword ptr [rdi + rdx*8 + 8], r8
.Lnoswap:
inc rdx
jmp .Linner
.Linner_done:
dec rcx
jnz .Louter
.Ldone:
mov rax, 0
ret
