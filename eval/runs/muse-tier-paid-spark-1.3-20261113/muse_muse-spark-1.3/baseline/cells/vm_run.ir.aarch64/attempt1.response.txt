export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
%pcslot =l alloc8 8
%dslot =l alloc8 8
%islot =l alloc8 8
%k0 =l copy 0
%k1 =l copy 1
%k2 =l copy 2
%k3 =l copy 3
%k4 =l copy 4
%k5 =l copy 5
%k6 =l copy 6
%k7 =l copy 7
%k8 =l copy 8
%k9 =l copy 9
%k10 =l copy 10
%k11 =l copy 11
storel %k0, %pcslot
storel %k0, %dslot
jmp @loop
@loop
%pc =l loadl %pcslot
%depth =l loadl %dslot
%atend =w ceql %pc, %len
jnz %atend, @finish, @fetch
@fetch
%pc8 =l mul %pc, %k8
%opaddr =l add %code, %pc8
%op =l loadl %opaddr
%is0 =w ceql %op, %k0
jnz %is0, @do_halt, @chk1
@chk1
%is1 =w ceql %op, %k1
jnz %is1, @do_push, @chk2
@chk2
%is2 =w ceql %op, %k2
jnz %is2, @do_add, @chk3
@chk3
%is3 =w ceql %op, %k3
jnz %is3, @do_sub, @chk4
@chk4
%is4 =w ceql %op, %k4
jnz %is4, @do_mul, @chk5
@chk5
%is5 =w ceql %op, %k5
jnz %is5, @do_dup, @chk6
@chk6
%is6 =w ceql %op, %k6
jnz %is6, @do_swap, @chk7
@chk7
%is7 =w ceql %op, %k7
jnz %is7, @do_drop, @chk8
@chk8
%is8 =w ceql %op, %k8
jnz %is8, @do_over, @chk9
@chk9
%is9 =w ceql %op, %k9
jnz %is9, @do_lt, @chk10
@chk10
%is10 =w ceql %op, %k10
jnz %is10, @do_jz, @chk11
@chk11
%is11 =w ceql %op, %k11
jnz %is11, @do_jmp, @do_default
@do_halt
jmp @finish
@do_push
%push_pc1 =l add %pc, %k1
%push_off =l mul %push_pc1, %k8
%push_caddr =l add %code, %push_off
%push_val =l loadl %push_caddr
%push_doff =l mul %depth, %k8
%push_saddr =l add %stack, %push_doff
storel %push_val, %push_saddr
%push_nd =l add %depth, %k1
storel %push_nd, %dslot
%push_npc =l add %pc, %k2
storel %push_npc, %pcslot
jmp @loop
@do_add
%add_dm1 =l sub %depth, %k1
%add_dm1o =l mul %add_dm1, %k8
%add_baddr =l add %stack, %add_dm1o
%add_b =l loadl %add_baddr
%add_dm2 =l sub %depth, %k2
%add_dm2o =l mul %add_dm2, %k8
%add_aaddr =l add %stack, %add_dm2o
%add_a =l loadl %add_aaddr
%add_r =l add %add_a, %add_b
storel %add_r, %add_aaddr
%add_nd =l sub %depth, %k1
storel %add_nd, %dslot
%add_npc =l add %pc, %k1
storel %add_npc, %pcslot
jmp @loop
@do_sub
%sub_dm1 =l sub %depth, %k1
%sub_dm1o =l mul %sub_dm1, %k8
%sub_baddr =l add %stack, %sub_dm1o
%sub_b =l loadl %sub_baddr
%sub_dm2 =l sub %depth, %k2
%sub_dm2o =l mul %sub_dm2, %k8
%sub_aaddr =l add %stack, %sub_dm2o
%sub_a =l loadl %sub_aaddr
%sub_r =l sub %sub_a, %sub_b
storel %sub_r, %sub_aaddr
%sub_nd =l sub %depth, %k1
storel %sub_nd, %dslot
%sub_npc =l add %pc, %k1
storel %sub_npc, %pcslot
jmp @loop
@do_mul
%mul_dm1 =l sub %depth, %k1
%mul_dm1o =l mul %mul_dm1, %k8
%mul_baddr =l add %stack, %mul_dm1o
%mul_b =l loadl %mul_baddr
%mul_dm2 =l sub %depth, %k2
%mul_dm2o =l mul %mul_dm2, %k8
%mul_aaddr =l add %stack, %mul_dm2o
%mul_a =l loadl %mul_aaddr
%mul_r =l mul %mul_a, %mul_b
storel %mul_r, %mul_aaddr
%mul_nd =l sub %depth, %k1
storel %mul_nd, %dslot
%mul_npc =l add %pc, %k1
storel %mul_npc, %pcslot
jmp @loop
@do_dup
%dup_dm1 =l sub %depth, %k1
%dup_dm1o =l mul %dup_dm1, %k8
%dup_taddr =l add %stack, %dup_dm1o
%dup_v =l loadl %dup_taddr
%dup_doff =l mul %depth, %k8
%dup_saddr =l add %stack, %dup_doff
storel %dup_v, %dup_saddr
%dup_nd =l add %depth, %k1
storel %dup_nd, %dslot
%dup_npc =l add %pc, %k1
storel %dup_npc, %pcslot
jmp @loop
@do_swap
%swap_dm1 =l sub %depth, %k1
%swap_dm1o =l mul %swap_dm1, %k8
%swap_topaddr =l add %stack, %swap_dm1o
%swap_b =l loadl %swap_topaddr
%swap_dm2 =l sub %depth, %k2
%swap_dm2o =l mul %swap_dm2, %k8
%swap_secaddr =l add %stack, %swap_dm2o
%swap_a =l loadl %swap_secaddr
storel %swap_b, %swap_secaddr
storel %swap_a, %swap_topaddr
%swap_npc =l add %pc, %k1
storel %swap_npc, %pcslot
jmp @loop
@do_drop
%drop_nd =l sub %depth, %k1
storel %drop_nd, %dslot
%drop_npc =l add %pc, %k1
storel %drop_npc, %pcslot
jmp @loop
@do_over
%over_dm2 =l sub %depth, %k2
%over_dm2o =l mul %over_dm2, %k8
%over_saddr =l add %stack, %over_dm2o
%over_v =l loadl %over_saddr
%over_doff =l mul %depth, %k8
%over_taddr =l add %stack, %over_doff
storel %over_v, %over_taddr
%over_nd =l add %depth, %k1
storel %over_nd, %dslot
%over_npc =l add %pc, %k1
storel %over_npc, %pcslot
jmp @loop
@do_lt
%lt_dm1 =l sub %depth, %k1
%lt_dm1o =l mul %lt_dm1, %k8
%lt_baddr =l add %stack, %lt_dm1o
%lt_b =l loadl %lt_baddr
%lt_dm2 =l sub %depth, %k2
%lt_dm2o =l mul %lt_dm2, %k8
%lt_aaddr =l add %stack, %lt_dm2o
%lt_a =l loadl %lt_aaddr
%lt_c =w csltl %lt_a, %lt_b
%lt_r =l extuw %lt_c
storel %lt_r, %lt_aaddr
%lt_nd =l sub %depth, %k1
storel %lt_nd, %dslot
%lt_npc =l add %pc, %k1
storel %lt_npc, %pcslot
jmp @loop
@do_jz
%jz_pc1 =l add %pc, %k1
%jz_off =l mul %jz_pc1, %k8
%jz_caddr =l add %code, %jz_off
%jz_t =l loadl %jz_caddr
%jz_dm1 =l sub %depth, %k1
%jz_dm1o =l mul %jz_dm1, %k8
%jz_saddr =l add %stack, %jz_dm1o
%jz_v =l loadl %jz_saddr
storel %jz_dm1, %dslot
%jz_isz =w ceql %jz_v, %k0
jnz %jz_isz, @jz_zero, @jz_nzero
@jz_zero
storel %jz_t, %pcslot
jmp @loop
@jz_nzero
%jzn_npc =l add %pc, %k2
storel %jzn_npc, %pcslot
jmp @loop
@do_jmp
%jmp_pc1 =l add %pc, %k1
%jmp_off =l mul %jmp_pc1, %k8
%jmp_caddr =l add %code, %jmp_off
%jmp_t =l loadl %jmp_caddr
storel %jmp_t, %pcslot
jmp @loop
@do_default
%def_npc =l add %pc, %k1
storel %def_npc, %pcslot
jmp @loop
@finish
%fdepth =l loadl %dslot
storel %fdepth, %islot
jmp @zloop
@zloop
%zi =l loadl %islot
%zcond =w csltl %zi, %cap
jnz %zcond, @zbody, @zdone
@zbody
%zb_off =l mul %zi, %k8
%zb_addr =l add %stack, %zb_off
storel %k0, %zb_addr
%zb_ni =l add %zi, %k1
storel %zb_ni, %islot
jmp @zloop
@zdone
%zd_empty =w ceql %fdepth, %k0
jnz %zd_empty, @ret0, @retnz
@ret0
ret %k0
@retnz
%ret_dm1 =l sub %fdepth, %k1
%ret_off =l mul %ret_dm1, %k8
%ret_addr =l add %stack, %ret_off
%ret_v =l loadl %ret_addr
ret %ret_v
}
