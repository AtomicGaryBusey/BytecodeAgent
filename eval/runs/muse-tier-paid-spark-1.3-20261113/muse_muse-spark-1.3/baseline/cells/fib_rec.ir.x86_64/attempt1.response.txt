export function l $fib_rec(l %n) {
@start
  %two =l copy 2
  %c =w csltl %n, %two
  jnz %c, @base, @rec
@base
  ret %n
@rec
  %one =l copy 1
  %nm1 =l sub %n, %one
  %r1 =l call $fib_rec(l %nm1)
  %two2 =l copy 2
  %nm2 =l sub %n, %two2
  %r2 =l call $fib_rec(l %nm2)
  %res =l add %r1, %r2
  ret %res
}
