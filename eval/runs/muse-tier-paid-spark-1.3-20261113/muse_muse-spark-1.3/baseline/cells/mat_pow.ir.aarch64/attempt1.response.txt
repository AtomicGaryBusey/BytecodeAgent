export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
  %zero =l sub %n, %n
  %one =l udiv %n, %n
  %two =l add %one, %one
  %four =l add %two, %two
  %eight =l add %four, %four
  %nn =l mul %n, %n
  %nn8 =l mul %nn, %eight
  %prod =l add %tmp, %nn8
  %tmp_total =l add %nn, %nn
  jmp @zout_head
@zout_head
  %p0 =l phi @start %zero, @zout_body %p0_next
  %c_p0 =w csltl %p0, %nn
  jnz %c_p0, @zout_body, @diag_head
@zout_body
  %off_p0 =l mul %p0, %eight
  %addr_p0 =l add %out, %off_p0
  storel %zero, %addr_p0
  %p0_next =l add %p0, %one
  jmp @zout_head
@diag_head
  %d =l phi @zout_head %zero, @diag_body %d_next
  %c_d =w csltl %d, %n
  jnz %c_d, @diag_body, @kcheck
@diag_body
  %dn =l mul %d, %n
  %dnd =l add %dn, %d
  %off_d =l mul %dnd, %eight
  %addr_d =l add %out, %off_d
  storel %one, %addr_d
  %d_next =l add %d, %one
  jmp @diag_head
@kcheck
  %is_k0 =w ceql %k, %zero
  jnz %is_k0, @ztmp_head, @copy_head
@copy_head
  %p1 =l phi @kcheck %zero, @copy_body %p1_next
  %c_p1 =w csltl %p1, %nn
  jnz %c_p1, @copy_body, @iter_head
@copy_body
  %off_p1a =l mul %p1, %eight
  %addr_p1a =l add %a, %off_p1a
  %v_p1 =l loadl %addr_p1a
  %addr_p1t =l add %tmp, %off_p1a
  storel %v_p1, %addr_p1t
  %p1_next =l add %p1, %one
  jmp @copy_head
@iter_head
  %iter =l phi @copy_head %zero, @iter_next %iter_next
  %c_iter =w csltl %iter, %k
  jnz %c_iter, @p_head, @ztmp_head
@p_head
  %p =l phi @iter_head %zero, @t_done %p_next
  %c_p =w csltl %p, %nn
  jnz %c_p, @p_body, @copyback_head
@p_body
  %i =l udiv %p, %n
  %j =l urem %p, %n
  jmp @t_head
@t_head
  %t =l phi @p_body %zero, @t_body %t_next
  %sum =l phi @p_body %zero, @t_body %sum_next
  %c_t =w csltl %t, %n
  jnz %c_t, @t_body, @t_done
@t_body
  %inm =l mul %i, %n
  %inmt =l add %inm, %t
  %off_int =l mul %inmt, %eight
  %addr_out =l add %out, %off_int
  %v_out =l loadl %addr_out
  %tn =l mul %t, %n
  %tnj =l add %tn, %j
  %off_tnj =l mul %tnj, %eight
  %addr_base =l add %tmp, %off_tnj
  %v_base =l loadl %addr_base
  %prod_ij =l mul %v_out, %v_base
  %sum_next =l add %sum, %prod_ij
  %t_next =l add %t, %one
  jmp @t_head
@t_done
  %off_p =l mul %p, %eight
  %addr_prod =l add %prod, %off_p
  storel %sum, %addr_prod
  %p_next =l add %p, %one
  jmp @p_head
@copyback_head
  %q =l phi @p_head %zero, @copyback_body %q_next
  %c_q =w csltl %q, %nn
  jnz %c_q, @copyback_body, @iter_next
@copyback_body
  %off_q =l mul %q, %eight
  %addr_qprod =l add %prod, %off_q
  %v_q =l loadl %addr_qprod
  %addr_qout =l add %out, %off_q
  storel %v_q, %addr_qout
  %q_next =l add %q, %one
  jmp @copyback_head
@iter_next
  %iter_next =l add %iter, %one
  jmp @iter_head
@ztmp_head
  %r =l phi @kcheck %zero, @iter_head %zero, @ztmp_body %r_next
  %c_r =w csltl %r, %tmp_total
  jnz %c_r, @ztmp_body, @done
@ztmp_body
  %off_r =l mul %r, %eight
  %addr_r =l add %tmp, %off_r
  storel %zero, %addr_r
  %r_next =l add %r, %one
  jmp @ztmp_head
@done
  ret %zero
}
