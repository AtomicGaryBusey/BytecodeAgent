export function l $base64_encode(l %src, l %n, l %out) {
@start

  %one =l copy 1
  %two =l copy 2
  %three =l copy 3
  %four =l copy 4
  %w1 =w copy 1
  %w3 =w copy 3
  %w4 =w copy 4
  %w15 =w copy 15
  %w26 =w copy 26
  %w43 =w copy 43
  %w47 =w copy 47
  %w52 =w copy 52
  %w62 =w copy 62
  %w63 =w copy 63

  %n_plus2 =l add %n, %two

  %retlen =l mul %groups_total, %four

  %src_cur =l phi @start %src, @loop_body %src_next
  %out_cur =l phi @start %out, @loop_body %out_next

  jnz %cond, @loop_body, @rem_check
@loop_body

  %src_p2 =l add %src_cur, %two

  %r1e1_p5 =w mul %r1e1_emd, %w47

  %r1_out_p2 =l add %out_cur, %two
  %r1_out_p3 =l add %out_cur, %three
  storeb %r1e0_enc, %out_cur
  storeb %r1e1_enc, %r1_out_p1
  storeb %weq, %r1_out_p2
  storeb %weq, %r1_out_p3
  jmp @done
@rem2
  %r2_src_p1 =l add %src_cur, %one
  %r2_b0 =w loadub %src_cur
  %r2_b1 =w loadub %r2_src_p1
  %r2_i0 =w shr %r2_b0, 2
  %r2_tmp0 =w and %r2_b0, %w3

  %r2e0_lt52 =w csltw %r2_i0, %w52

  %r2e0_cmb =w sub %r2e0_lt62, %r2e0_lt52

  %r2e0_enc =w add %r2e0_s0123, %r2e0_p5
  %r2e1_lt26 =w csltw %r2_i1, %w26
