xor r9, r9
.Lloop:

jge .Ldone
movsx r10, byte ptr [rdi + r8]

je .Lnop
cmp r10, 0
jge .Lliteral
mov rax, rsi

cmp rax, 2

mov r11, 1
sub r11, r10

sub rax, r9

jg .Lrep_trunc

dec r11
jnz .Lrep_loop
add r8, 2

.Lrep_trunc:

sub r11, r9

.Lrep_trunc_loop:

sub rax, r8

sub rax, r9

inc r9
dec r10

jz .Lcaperr
inc r8

mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r11
jnz .Llit_trunc_loop
jmp .Lcaperr
.Lnop:

jmp .Lloop
.Ldone:

ret
.Lsrcerr:

.Lcaperr:
