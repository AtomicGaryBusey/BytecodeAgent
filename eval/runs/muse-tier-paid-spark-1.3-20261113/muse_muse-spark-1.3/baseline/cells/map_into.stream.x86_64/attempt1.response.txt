push r12
push r13
push r14
push r15
mov r12, rcx
mov r13, rdi
mov r14, rsi
mov r15, rdx
xor ebx, ebx
sub rsp, 16
test r15, r15
jz .Ldone
.Lloop:
mov rdi, qword ptr [r14]
mov qword ptr [rsp], rdi
call r12
mov rcx, qword ptr [rsp]
mov qword ptr [r13], rax
cmp rax, rcx
jge .Lskip
add rbx, 1
.Lskip:
add r13, 8
add r14, 8
sub r15, 1
jne .Lloop
.Ldone:
add rsp, 16
mov rax, rbx
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
