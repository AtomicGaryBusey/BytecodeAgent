export function l $memmove_i64(l %dst, l %src, l %n) {
@start
  %c_empty =w csltl %n, 1
  jnz %c_empty, @done, @dir1
@dir1
  %c_dst_lt_src =w csltl %dst, %src
  jnz %c_dst_lt_src, @fwd_init, @dir2
@dir2
  %nbytes =l mul %n, 8
  %src_end =l add %src, %nbytes
  %c_overlap =w csltl %dst, %src_end
  jnz %c_overlap, @bwd_init, @fwd_init
@fwd_init
  %fwd_zero =l copy 0
  jmp @fwd_loop
@fwd_loop
  %fi =l phi @fwd_init %fwd_zero, @fwd_body %fi_next
  %c_fwd =w csltl %fi, %n
  jnz %c_fwd, @fwd_body, @done
@fwd_body
  %fi_bytes =l mul %fi, 8
  %fwd_src_ptr =l add %src, %fi_bytes
  %fwd_dst_ptr =l add %dst, %fi_bytes
  %fwd_val =l loadl %fwd_src_ptr
  storel %fwd_val, %fwd_dst_ptr
  %fi_next =l add %fi, 1
  jmp @fwd_loop
@bwd_init
  jmp @bwd_loop
@bwd_loop
  %bi =l phi @bwd_init %n, @bwd_body %bi_next
  %c_bwd =w csltl %bi, 1
  jnz %c_bwd, @done, @bwd_body
@bwd_body
  %bi_dec =l sub %bi, 1
  %bi_bytes =l mul %bi_dec, 8
  %bwd_src_ptr =l add %src, %bi_bytes
  %bwd_dst_ptr =l add %dst, %bi_bytes
  %bwd_val =l loadl %bwd_src_ptr
  storel %bwd_val, %bwd_dst_ptr
  %bi_next =l copy %bi_dec
  jmp @bwd_loop
@done
  %ret0 =l copy 0
  ret %ret0
}
