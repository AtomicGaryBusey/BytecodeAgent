export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
  %zero =l copy 0
  %one =l copy 1
  %eight =l copy 8
  %n1 =l add %n, %one
  jmp @zhead
@zhead
  %zi =l phi @start %zero, @zbody %zi_next
  %zc =w csltl %zi, %n1
  jnz %zc, @zbody, @outer_pre
@zbody
  %zoff =l mul %zi, %eight
  %zaddr =l add %row, %zoff
  storel %zero, %zaddr
  %zi_next =l add %zi, %one
  jmp @zhead
@outer_pre
  jmp @ohead
@ohead
  %i =l phi @outer_pre %zero, @o_latch %i_next
  %oc =w csltl %i, %m
  jnz %oc, @inner_pre, @done
@inner_pre
  jmp @ihead
@ihead
  %j =l phi @inner_pre %one, @join %j_next
  %diag =l phi @inner_pre %zero, @join %diag_next
  %ic =w csltl %j, %n1
  jnz %ic, @ibody, @o_latch
@ibody
  %joff =l mul %j, %eight
  %jaddr =l add %row, %joff
  %up =l loadl %jaddr
  %xaddr =l add %x, %i
  %xb =w loadub %xaddr
  %jm1 =l sub %j, %one
  %yaddr =l add %y, %jm1
  %yb =w loadub %yaddr
  %eq =w ceqw %xb, %yb
  jnz %eq, @then, @else
@then
  %then_val =l add %diag, %one
  jmp @join
@else
  %jm1e =l sub %j, %one
  %pmoff =l mul %jm1e, %eight
  %pmaddr =l add %row, %pmoff
  %prev =l loadl %pmaddr
  %lt =w csltl %up, %prev
  jnz %lt, @max_prev, @max_up
@max_prev
  %max_prev_v =l copy %prev
  jmp @join
@max_up
  %max_up_v =l copy %up
  jmp @join
@join
  %new_val =l phi @then %then_val, @max_prev %max_prev_v, @max_up %max_up_v
  %joff2 =l mul %j, %eight
  %jaddr2 =l add %row, %joff2
  storel %new_val, %jaddr2
  %j_next =l add %j, %one
  %diag_next =l copy %up
  jmp @ihead
@o_latch
  %i_next =l add %i, %one
  jmp @ohead
@done
  %noff =l mul %n, %eight
  %naddr =l add %row, %noff
  %res =l loadl %naddr
  ret %res
}
