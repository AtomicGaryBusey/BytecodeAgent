export function l $sum_to(l %n) {
@start
    %n_pos = cslel %n, 0
    jnz %n_pos, @ret_zero, @loop
@loop
    %sum =l phi @start 0, @loop_body %sum_new
    %i =l phi @start 0, @loop_body %i_next
    %i_next =l add %i, 1
    %sum_new =l add %sum, %i_next
    %done =w cslel %n, %i_next
    jnz %done, @ret_sum, @loop_body
@loop_body
    jmp @loop
@ret_sum
    ret %sum_new
@ret_zero
    ret 0
}
