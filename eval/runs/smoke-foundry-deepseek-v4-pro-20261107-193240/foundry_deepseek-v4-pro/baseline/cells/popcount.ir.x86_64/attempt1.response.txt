export function l $popcount(l %x) {
@start
	%tx =l copy %x
	%odd_mask =l copy 0x5555555555555555
	%even_mask =l copy 0xaaaaaaaaaaaaaaaa
	%m2 =l copy 0x3333333333333333
	%m4 =l copy 0x0f0f0f0f0f0f0f0f
	%m8 =l copy 0x00ff00ff00ff00ff
	%m16 =l copy 0x0000ffff0000ffff
	%m32 =l copy 0x00000000ffffffff

	%shift1 =l shr %tx, 1
	%and_even =l and %tx, %even_mask
	%and_odd =l and %shift1, %odd_mask
	%bits2 =l add %and_even, %and_odd

	%shift2 =l shr %bits2, 2
	%and2a =l and %bits2, %m2
	%and2b =l and %shift2, %m2
	%bits4 =l add %and2a, %and2b

	%shift4 =l shr %bits4, 4
	%and4a =l and %bits4, %m4
	%and4b =l and %shift4, %m4
	%bits8 =l add %and4a, %and4b

	%shift8 =l shr %bits8, 8
	%and8a =l and %bits8, %m8
	%and8b =l and %shift8, %m8
	%bits16 =l add %and8a, %and8b

	%shift16 =l shr %bits16, 16
	%and16a =l and %bits16, %m16
	%and16b =l and %shift16, %m16
	%bits32 =l add %and16a, %and16b

	%shift32 =l shr %bits32, 32
	%and32a =l and %bits32, %m32
	%and32b =l and %shift32, %m32
	%bits64 =l add %and32a, %and32b

	ret %bits64
}
