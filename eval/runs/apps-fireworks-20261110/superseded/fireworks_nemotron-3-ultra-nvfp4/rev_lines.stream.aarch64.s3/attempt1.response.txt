.section .text
.globl _start
_start:
    // read all input into buffer
    mov x19, 0                      // total_read = 0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf   // x20 = buffer base
    mov x21, 65536                  // max input size
    // load address of newline byte
    adrp x30, newline_byte
    add x30, x30, :lo12:newline_byte

read_loop:
    sub x22, x21, x19               // remaining space
    cbz x22, read_done              // buffer full (should not happen per limits)
    mov x8, 63                      // sys_read
    mov x0, 0                       // fd = stdin
    add x1, x20, x19                // buf = buffer + total_read
    mov x2, x22                     // count = remaining
    svc #0
    cbz x0, read_done               // EOF
    add x19, x19, x0                // total_read += bytes_read
    b read_loop

read_done:
    cbz x19, exit_prog              // no input
    mov x23, 0                      // current index i
    mov x24, 0                      // line_start

process_loop:
    cmp x23, x19
    b.ge process_after_loop
    ldrb w25, [x20, x23]            // load byte
    cmp w25, 10                     // '\n'
    b.ne not_newline

    // reverse line from line_start to i-1
    mov x26, x24                    // left
    sub x27, x23, 1                 // right
    cmp x26, x27
    b.ge reverse_done
reverse_loop:
    ldrb w28, [x20, x26]
    ldrb w29, [x20, x27]
    strb w29, [x20, x26]
    strb w28, [x20, x27]
    add x26, x26, 1
    sub x27, x27, 1
    cmp x26, x27
    b.lt reverse_loop
reverse_done:
    // write reversed line
    mov x8, 64                      // sys_write
    mov x0, 1                       // fd = stdout
    add x1, x20, x24                // buf = buffer + line_start
    sub x2, x23, x24                // count = line length
    svc #0
    // write newline
    mov x8, 64
    mov x0, 1
    mov x1, x30                     // address of newline byte
    mov x2, 1
    svc #0
    add x24, x23, 1                 // line_start = i + 1

not_newline:
    add x23, x23, 1
    b process_loop

process_after_loop:
    cmp x24, x19
    b.ge exit_prog                  // no final partial line

    // reverse final line (from line_start to end)
    mov x26, x24
    sub x27, x19, 1
    cmp x26, x27
    b.ge final_reverse_done
final_reverse_loop:
    ldrb w28, [x20, x26]
    ldrb w29, [x20, x27]
    strb w29, [x20, x26]
    strb w28, [x20, x27]
    add x26, x26, 1
    sub x27, x27, 1
    cmp x26, x27
    b.lt final_reverse_loop
final_reverse_done:
    // write final line
    mov x8, 64
    mov x0, 1
    add x1, x20, x24
    sub x2, x19, x24
    svc #0
    // write newline
    mov x8, 64
    mov x0, 1
    mov x1, x30
    mov x2, 1
    svc #0

exit_prog:
    mov x8, 93                      // sys_exit
    mov x0, 0
    svc #0

.section .rodata
newline_byte:
    .byte 10

.section .bss
    .align 16
