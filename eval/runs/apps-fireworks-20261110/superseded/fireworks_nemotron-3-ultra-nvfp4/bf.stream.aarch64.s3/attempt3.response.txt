.section .bss
.align 4
tape:       .space 30000
prog_buf:   .space 65536
input_buf:  .space 65536
bracket_map:.space 262144

.section .rodata
.align 3
instr_limit: .quad 5000000

.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

read_all:
    mov x19, xzr
    mov x20, #65536
    adr x21, prog_buf
1:
    mov x8, #SYS_READ
    mov x0, #0
    add x1, x21, x19
    sub x2, x20, x19
    svc #0
    cbz x0, 2f
    add x19, x19, x0
    cmp x19, x20
    b.lt 1b
2:
    mov x0, x19
    ret

split_program:
    mov x2, xzr
    mov x3, xzr
    adr x4, prog_buf
1:
    cmp x3, x0
    b.ge 2f
    ldrb w5, [x4, x3]
    cmp w5, #0x21
    b.eq 3f
    add x3, x3, #1
    b 1b
2:
    mov x1, x0
    mov x0, xzr
    mov x2, x0
    ret
3:
    mov x0, x3
    sub x1, x22, x3
    sub x1, x1, #1
    add x2, x3, #1
    ret

validate_brackets:
    mov x9, xzr
    mov x10, #1000
    adr x11, bracket_map
    mov x12, xzr
1:
    cmp x12, x0
    b.ge 2f
    ldrb w3, [x1, x12]
    cmp w3, #'['
    b.eq 3f
    cmp w3, #']'
    b.eq 4f
    add x12, x12, #1
    b 1b
3:
    cmp x9, x10
    b.ge error
    str w12, [x11, x9, lsl #2]
    add x9, x9, #1
    add x12, x12, #1
    b 1b
4:
    cbz x9, error
    sub x9, x9, #1
    ldr w4, [x11, x9, lsl #2]
    str w12, [x11, x4, lsl #2]
    str w4, [x11, x12, lsl #2]
    add x12, x12, #1
    b 1b
2:
    cbnz x9, error
    mov x0, xzr
    ret
error:
    mov x0, #1
    ret

interpret:
    adr x19, tape
    mov x20, xzr
    mov x23, xzr
    mov x26, xzr
    adr x27, bracket_map
    ldr x28, =instr_limit
