// base64 encoder - AArch64 Linux, static, no libc
// reads stdin fully, encodes to standard base64, wraps at 76 chars

    .section .rodata
    .balign 4
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .bss
    .balign 16
inbuf:
    .zero 65536
    .balign 16
outbuf:
    .zero 131072

    .text
    .globl _start

_start:
    // ---- read all of stdin into inbuf ----
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                     // total bytes read
read_loop:
    movz x2, #1, lsl #16            // 65536 capacity
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0                      // fd 0
    add x1, x19, x20
    mov x8, #63                     // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    // ---- set up encoding state ----
    add x23, x19, x20               // end of input
    adrp x24, table
    add x24, x24, :lo12:table
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, #0                     // current line length

enc_loop:
    sub x25, x23, x19               // remaining bytes
    cmp x25, #3
    b.lo enc_tail

    ldrb w0, [x19]
    ldrb w1, [x19, #1]
    ldrb w2, [x19, #2]
    add x19, x19, #3

    lsr w3, w0, #2                  // c0
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5                  // c1
    and w6, w1, #15
    lsl w6, w6, #2
    lsr w7, w2, #6
    orr w6, w6, w7                  // c2
    and w7, w2, #63                 // c3

    ldrb w3, [x24, w3, uxtw]
    ldrb w4, [x24, w4, uxtw]
    ldrb w6, [x24, w6, uxtw]
    ldrb w7, [x24, w7, uxtw]

    strb w3, [x21]
    strb w4, [x21, #1]
    strb w6, [x21, #2]
    strb w7, [x21, #3]
    add x21, x21, #4
    add x22, x22, #4

    cmp x22, #76
    b.ne enc_loop
    mov w3, #10
    strb w3, [x21]
    add x21, x21, #1
    mov x22, #0
    b enc_loop

enc_tail:
    cbz x25, enc_finish
    ldrb w0, [x19]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4

    ldrb w3, [x24, w3, uxtw]
    strb w3, [x21]
    add x21, x21, #1
    add x22, x22, #1

    cmp x25, #1
    b.eq enc_tail1

    // two bytes remain
    ldrb w1, [x19, #1]
    lsr w5, w1, #4
    orr w4, w4, w5
    ldrb w4, [x24, w4, uxtw]
    and w6, w1, #15
    lsl w6, w6, #2
    ldrb w6, [x24, w6, uxtw]
    mov w7, #61                     // '='
    strb w4, [x21]
    strb w6, [x21, #1]
    strb w7, [x21, #2]
    add x21, x21, #3
    add x22, x22, #3
    b enc_finish

enc_tail1:
    // one byte remains
    ldrb w4, [x24, w4, uxtw]
    mov w7, #61                     // '='
    strb w4, [x21]
    strb w7, [x21, #1]
    strb w7, [x21, #2]
    add x21, x21, #3
    add x22, x22, #3

enc_finish:
    cbz x22, enc_write
    mov w3, #10
    strb w3, [x21]
    add x21, x21, #1

enc_write:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1
    cbz x2, enc_exit
write_loop:
    mov x0, #1                      // fd 1
    mov x8, #64                     // write
    svc #0
    cmp x0, #0
    b.le enc_exit
    add x1, x1, x0
    subs x2, x2, x0
    b.ne write_loop

enc_exit:
    mov x0, #0
    mov x8, #93                     // exit
    svc #0
