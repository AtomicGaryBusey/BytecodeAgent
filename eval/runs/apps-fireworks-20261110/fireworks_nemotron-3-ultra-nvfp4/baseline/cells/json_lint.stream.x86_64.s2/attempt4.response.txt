.section .rodata
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg

.section .data
error_offset: .quad -1

.section .bss
input_buffer: .zero 65536
input_len: .zero 8

.section .text
.globl _start

_start:
    # Read all input
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    mov rax, 0
    syscall
    test rax, rax
    js read_error
    mov qword ptr [rip + input_len], rax
    test rax, rax
    jz empty_input

    # Initialize parser state
    mov qword ptr [rip + error_offset], -1
    
    # Load buffer and length into registers for fast access
    lea r12, [rip + input_buffer]
    mov r13, qword ptr [rip + input_len]
    xor rcx, rcx        # rcx = current offset
    
    # Skip leading whitespace
    call skip_whitespace
    
    # Parse value
    call parse_value
    
    # Check for error
    cmp qword ptr [rip + error_offset], -1
    jne print_error
    
    # Skip trailing whitespace
    call skip_whitespace
    
    # Check if we consumed all input
    cmp rcx, r13
    jne print_error_trailing
    
    # Success
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

read_error:
    mov rdi, 1
    mov rax, 60
    syscall

empty_input:
    mov qword ptr [rip + error_offset], 0
    jmp print_error

print_error_trailing:
    mov qword ptr [rip + error_offset], rcx

print_error:
    # Convert error_offset to decimal string and print
    mov rax, qword ptr [rip + error_offset]
    # Use stack for number buffer
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 0x0A
    dec rdi
    test rax, rax
    js print_error_neg
    jz print_error_zero
print_error_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz print_error_loop
    jmp print_error_done
print_error_zero:
    mov byte ptr [rdi], '0'
    dec rdi
print_error_done:
    inc rdi
    # Print "error at "
    mov rdx, error_prefix_len
    lea rsi, [rip + error_prefix]
    mov rdi, 1
    mov rax, 1
    syscall
    # Print number + newline
    lea rsi, [rdi]
    mov rdx, 32
    sub rdx, rdi
    add rsp, 32
    mov rdi, 1
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_error_neg:
    mov byte ptr [rdi], '0'
    dec rdi
    jmp print_error_done

# ============================================================
# Helper functions
# ============================================================

# skip_whitespace: rcx = offset, r12 = buffer, r13 = len, updates rcx
skip_whitespace:
.skip_loop:
    cmp rcx, r13
    jae .skip_done
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x20
    je .skip_next
    cmp al, 0x09
    je .skip_next
    cmp al, 0x0A
    je .skip_next
    cmp al, 0x0D
    je .skip_next
    ret
.skip_next:
    inc rcx
    jmp .skip_loop
.skip_done:
    ret

# set_error: sets error_offset to rcx if not already set
set_error:
    cmp qword ptr [rip + error_offset], -1
    jne set_error_done
    mov qword ptr [rip + error_offset], rcx
set_error_done:
    ret

# check_eof: returns with carry set if at EOF
check_eof:
    cmp rcx, r13
    jb check_eof_not
    stc
    ret
check_eof_not:
    clc
    ret

# set_error_eof: set error at EOF
set_error_eof:
    call set_error
    ret

# parse_value: rcx = offset, returns with rcx updated
parse_value:
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    je parse_number
    cmp al, '1'
    je parse_number
    cmp al, '2'
    je parse_number
    cmp al, '3'
    je parse_number
    cmp al, '4'
    je parse_number
    cmp al, '5'
    je parse_number
    cmp al, '6'
    je parse_number
    cmp al, '7'
    je parse_number
    cmp al, '8'
    je parse_number
    cmp al, '9'
    je parse_number
    jmp set_error

# parse_object: rcx at '{', returns after '}'
parse_object:
    inc rcx                 # consume '{'
    call skip_whitespace
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '}'
    je parse_object_empty
    # Parse members
parse_object_member_loop:
    call parse_string
    cmp qword ptr [rip + error_offset], -1
    jne parse_object_done
    call skip_whitespace
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, ':'
    jne set_error
    inc rcx                 # consume ':'
    call skip_whitespace
    call parse_value
    cmp qword ptr [rip + error_offset], -1
    jne parse_object_done
    call skip_whitespace
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '}'
    je parse_object_end
    cmp al, ','
    jne set_error
    inc rcx                 # consume ','
    call skip_whitespace
    jmp parse_object_member_loop
parse_object_empty:
    inc rcx                 # consume '}'
    ret
parse_object_end:
    inc rcx                 # consume '}'
    ret
parse_object_done:
    ret

# parse_array: rcx at '[', returns after ']'
parse_array:
    inc rcx                 # consume '['
    call skip_whitespace
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, ']'
    je parse_array_empty
    # Parse elements
parse_array_element_loop:
    call parse_value
    cmp qword ptr [rip + error_offset], -1
    jne parse_array_done
    call skip_whitespace
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, ']'
    je parse_array_end
    cmp al, ','
    jne set_error
    inc rcx                 # consume ','
    call skip_whitespace
    jmp parse_array_element_loop
parse_array_empty:
    inc rcx                 # consume ']'
    ret
parse_array_end:
    inc rcx                 # consume ']'
    ret
parse_array_done:
    ret

# parse_string: rcx at '"', returns after closing '"'
parse_string:
    inc rcx                 # consume opening '"'
parse_string_loop:
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '"'
    je parse_string_end
    cmp al, '\\'
    je parse_string_escape
    cmp al, 0x1F
    jbe set_error           # control chars 0x00-0x1F are errors
    inc rcx
    jmp parse_string_loop
parse_string_escape:
    inc rcx                 # consume '\'
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '"'
    je parse_string_escape_ok
    cmp al, '\\'
    je parse_string_escape_ok
    cmp al, '/'
    je parse_string_escape_ok
    cmp al, 'b'
    je parse_string_escape_ok
    cmp al, 'f'
    je parse_string_escape_ok
    cmp al, 'n'
    je parse_string_escape_ok
    cmp al, 'r'
    je parse_string_escape_ok
    cmp al, 't'
    je parse_string_escape_ok
    cmp al, 'u'
    je parse_string_unicode
    jmp set_error
parse_string_escape_ok:
    inc rcx
    jmp parse_string_loop
parse_string_unicode:
    inc rcx                 # consume 'u'
    mov r8, 4               # 4 hex digits
parse_string_unicode_loop:
    call check_eof
    jc set_error_eof
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '0'
    jb set_error
    cmp al, '9'
    jbe parse_string_unicode_ok
    cmp al, 'A'
    jb set_error
    cmp al, 'F'
    jbe parse_string_unicode_ok
    cmp al, 'a'
    jb set_error
    cmp al, 'f'
    ja set_error
parse_string_unicode_ok:
    inc rcx
    dec r8
    jnz parse_string_unicode_loop
    jmp parse_string_loop
parse_string_end:
    inc rcx                 # consume closing '"'
    ret

# parse_true: rcx at 't'
parse_true:
    call check_eof
    jc set_error_eof
    add rcx, 4
    cmp rcx, r13
    ja parse_true_error
    mov eax, dword ptr [r12 + rcx - 4]
    cmp eax, 'eurt'  # 'true' in little-endian
