.section .rodata
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
newline: .byte 0x0A

.section .data
error_offset: .quad -1

.section .bss
input_buffer: .zero 65536
input_len: .zero 8
parse_stack: .zero 8000  # 1000 entries * 8 bytes
stack_top: .zero 8

.section .text
.globl _start

_start:
    # Read all input
    mov rdi, 0          # fd = stdin
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    mov rax, 0          # sys_read
    syscall
    test rax, rax
    js read_error
    mov qword ptr [rip + input_len], rax
    test rax, rax
    jz empty_input

    # Initialize parser state
    mov qword ptr [rip + stack_top], 0
    mov qword ptr [rip + error_offset], -1
    
    # Skip leading whitespace
    xor rcx, rcx        # rcx = current offset
    call skip_whitespace
    
    # Parse value
    call parse_value
    
    # Check for error
    cmp qword ptr [rip + error_offset], -1
    jne print_error
    
    # Skip trailing whitespace
    call skip_whitespace
    
    # Check if we consumed all input
    cmp rcx, qword ptr [rip + input_len]
    jne print_error_trailing
    
    # Success
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

read_error:
    mov rdi, 1
    mov rax, 60
    syscall

empty_input:
    # Empty input is error at 0
    mov qword ptr [rip + error_offset], 0
    jmp print_error

print_error_trailing:
    # Error at current position (trailing garbage)
    mov qword ptr [rip + error_offset], rcx

print_error:
    # Convert error_offset to decimal string and print
    mov rax, qword ptr [rip + error_offset]
    lea rdi, [rip + num_buffer + 20]
    mov byte ptr [rdi], 0x0A
    dec rdi
    test rax, rax
    js print_error_neg
    jz print_error_zero
print_error_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz print_error_loop
    jmp print_error_done
print_error_zero:
    mov byte ptr [rdi], '0'
    dec rdi
print_error_done:
    inc rdi
    # Print "error at "
    mov rdx, error_prefix_len
    lea rsi, [rip + error_prefix]
    mov rdi, 1
    mov rax, 1
    syscall
    # Print number + newline
    lea rsi, [rdi]
    mov rdx, 20
    sub rdx, rdi
    add rdx, 1  # include newline
    mov rdi, 1
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_error_neg:
    # Should not happen, but handle anyway
    mov byte ptr [rdi], '0'
    dec rdi
    jmp print_error_done

# ============================================================
# Helper functions
# ============================================================

# skip_whitespace: rcx = offset, updates rcx
skip_whitespace:
    mov rsi, qword ptr [rip + input_len]
    lea rdi, [rip + input_buffer]
.skip_loop:
    cmp rcx, rsi
    jae .skip_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x20
    je .skip_next
    cmp al, 0x09
    je .skip_next
    cmp al, 0x0A
    je .skip_next
    cmp al, 0x0D
    je .skip_next
    ret
.skip_next:
    inc rcx
    jmp .skip_loop
.skip_done:
    ret

# parse_value: rcx = offset, returns with rcx updated, sets error_offset on error
parse_value:
    cmp qword ptr [rip + error_offset], -1
    jne .error_already
    cmp rcx, qword ptr [rip + input_len]
    jae .error_eof
    movzx eax, byte ptr [rip + input_buffer + rcx]
