.section .bss

.section .text
_start:
    xor r12, r12        # lines
    xor r13, r13        # words
    xor r14, r14        # bytes
    xor r15, r15        # in_word flag

read_loop:
    mov rax, 0          # sys_read
    mov rdi, 0          # fd 0
    lea rsi, [rip + buffer]
    mov rdx, 8192
    syscall
    test rax, rax
    jz output_results
    add r14, rax        # increment byte count
    mov rcx, rax        # number of bytes read
    lea rsi, [rip + buffer]
    call process_buffer
    jmp read_loop

process_buffer:
    test rcx, rcx
    jz .done
.loop:
    movzx r8, byte ptr [rsi]
    cmp r8, 0x0A
    jne .check_ws
    inc r12
.check_ws:
    movzx r9, byte ptr [rip + is_whitespace + r8]
