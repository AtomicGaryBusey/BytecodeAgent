.section .text
.globl _start

_start:
    // buffer address in x19
    adrp    x19, buffer
    add     x19, x19, :lo12:buffer
    // total_len = 0 (x20)
    mov     x20, 0
    // remaining = 65536 (x21)
    mov     x21, 65536

read_loop:
    // read(fd=0, buf=buffer+total_len, count=remaining)
    mov     x8, 63          // SYS_READ
    mov     x0, 0           // stdin
    add     x1, x19, x20    // buf
    mov     x2, x21         // count
    svc     #0
    // x0 = bytes read
    cbz     x0, read_done   // EOF
    add     x20, x20, x0    // total_len += bytes_read
    sub     x21, x21, x0    // remaining -= bytes_read
    cbnz    x21, read_loop  // continue if space left

read_done:
    // if total_len == 0, exit
    cbz     x20, exit_prog

    // ptr = buffer_start (x21), end = buffer_start + total_len (x22)
    mov     x21, x19
    add     x22, x19, x20

process_loop:
    cmp     x21, x22
    b.ge    exit_prog       // done

    // find newline starting at x21
    mov     x23, x21        // scan pointer
find_newline:
    cmp     x23, x22
    b.ge    no_newline_found
    ldrb    w24, [x23]
    add     x23, x23, 1
    cmp     w24, 10         // '\n'
    b.ne    find_newline
    // newline found at x23-1
    sub     x23, x23, 1
    b       line_found

no_newline_found:
    mov     x23, x22        // line_end = buffer_end

line_found:
    // line_start = x21, line_end = x23
    sub     x24, x23, x21   // line_len
    // reverse in place if line_len > 1
    cmp     x24, 1
    b.le    skip_reverse
    mov     x25, x21        // left
    sub     x26, x23, 1     // right
reverse_loop:
    cmp     x25, x26
    b.ge    skip_reverse
    ldrb    w27, [x25]
    ldrb    w28, [x26]
    strb    w28, [x25]
    strb    w27, [x26]
    add     x25, x25, 1
    sub     x26, x26, 1
    b       reverse_loop

skip_reverse:
    // write reversed line if line_len > 0
    cbz     x24, write_newline
    mov     x8, 64          // SYS_WRITE
    mov     x0, 1           // stdout
    mov     x1, x21         // buf
    mov     x2, x24         // count
    svc     #0

write_newline:
    // write newline
    adrp    x1, newline
    add     x1, x1, :lo12:newline
    mov     x8, 64
    mov     x0, 1
    mov     x2, 1
    svc     #0

    // advance ptr
    cmp     x23, x22
    b.eq    advance_to_end
    add     x21, x23, 1     // skip newline
    b       process_loop

advance_to_end:
    mov     x21, x22
    b       process_loop

exit_prog:
    mov     x8, 93          // SYS_EXIT
    mov     x0, 0
    svc     #0

.section .rodata
newline:
    .byte   10

.section .bss
.align  12
buffer:
    .space  65536
