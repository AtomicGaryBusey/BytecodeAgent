.globl _start

.section .bss
.align 8
input_buffer: .zero 65536
rpn_stack: .zero 8192

.section .rodata
error_msg: .ascii "error\n"
error_len = . - error_msg
min_int_str: .ascii "-9223372036854775808\n"
min_int_len = . - min_int_str
max_pos: .quad 9223372036854775807
max_neg_mag: .quad 9223372036854775808
minus_char: .byte '-'
num_buffer: .zero 24
num_buffer_end:

.section .text
_start:
    lea rdi, [rip + input_buffer]
    mov rsi, 65536
    call read_all
    lea rdi, [rip + input_buffer]
    mov rsi, rax
    call process_lines
    xor rdi, rdi
    mov rax, 60
    syscall

read_all:
    mov rbx, 0
    mov r12, rdi
    mov r13, rsi
.read_loop:
    test r13, r13
    jz .read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz .read_done
    add rbx, rax
    add r12, rax
    sub r13, rax
    jmp .read_loop
.read_done:
    mov rax, rbx
    ret

process_lines:
    mov r12, rdi
    mov r13, rsi
    add r13, r12
.line_loop:
    cmp r12, r13
    jae .line_done
    mov rax, r12
.find_nl:
    cmp rax, r13
    jae .no_nl
    cmp byte ptr [rax], 10
    je .found_nl
    inc rax
    jmp .find_nl
.no_nl:
    mov rdi, r12
    mov rsi, r13
    call eval_line
    jmp .line_done
.found_nl:
    mov rdi, r12
    mov rsi, rax
    push r12
    push r13
    call eval_line
    pop r13
    pop r12
    lea r12, [rax + 1]
    jmp .line_loop
.line_done:
    ret

eval_line:
    mov r12, rdi
    mov r13, rsi
    lea r14, [rip + rpn_stack]
    mov r15, r14
.token_loop:
    cmp r12, r13
    jae .token_done
.skip_spaces:
    cmp r12, r13
    jae .token_done
    mov al, byte ptr [r12]
    cmp al, ' '
    je .skip_next
    cmp al, '\t'
    je .skip_next
    jmp .token_start
.skip_next:
    inc r12
    jmp .skip_spaces
.token_start:
    mov rdi, r12
.find_token_end:
    cmp r12, r13
    jae .token_end_found
    mov al, byte ptr [r12]
    cmp al, ' '
    je .token_end_found
    cmp al, '\t'
    je .token_end_found
    inc r12
    jmp .find_token_end
.token_end_found:
    mov rsi, r12
    mov rcx, rsi
    sub rcx, rdi
    cmp rcx, 1
    jne .parse_integer
    mov al, byte ptr [rdi]
    cmp al, '+'
    je .op_add
    cmp al, '-'
    je .op_sub
    cmp al, '*'
    je .op_mul
    cmp al, '/'
    je .op_div
    cmp al, '%'
    je .op_mod
    jmp .parse_integer

.op_add:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    add rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_sub:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    sub rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_mul:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    imul rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_div:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    test rbx, rbx
    jz .error
    mov rcx, qword ptr [rip + max_neg_mag]
    cmp rax, rcx
    jne .div_ok
    cmp rbx, -1
    je .error
.div_ok:
    cqo
    idiv rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_mod:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    test rbx, rbx
    jz .error
    mov rcx, qword ptr [rip + max_neg_mag]
    cmp rax, rcx
    jne .mod_ok
    cmp rbx, -1
    je .error
.mod_ok:
    cqo
    idiv rbx
    mov qword ptr [r15], rdx
    add r15, 8
    jmp .token_loop

.parse_integer:
    mov rax, rdi
    mov rbx, 0
    mov rcx, 0
    mov rdx, 0
    mov r8, 10
    mov r9, rsi
.parse_sign:
    cmp rax, r9
    jae .parse_digits
    mov cl, byte ptr [rax]
    cmp cl, '-'
    jne .parse_digits
    mov rbx, 1
    inc rax
    inc rcx
    jmp .parse_digits
.parse_digits:
    cmp rax, r9
    jae .check_digits
    mov cl, byte ptr [rax]
    cmp cl, '0'
    jb .check_digits
    cmp cl, '9'
    ja .check_digits
    inc rax
    inc rcx
    sub cl, '0'
    movzx r10, cl
    mov rax, rdx
    mul r8
    jc .int_error
    add rax, r10
    jc .int_error
    mov rdx, rax
    cmp rcx, 19
    ja .int_error
    jmp .parse_digits
.check_digits:
    test rcx, rcx
    jz .int_error
    cmp rcx, 19
    ja .int_error
    test rbx, rbx
    jz .check_positive
    mov rax, qword ptr [rip + max_neg_mag]
    cmp rdx, rax
    ja .int_error
    neg rdx
    jmp .push_value
.check_positive:
    mov rax, qword ptr [rip + max_pos]
    cmp rdx, rax
    ja .int_error
.push_value:
    mov qword ptr [r15], rdx
    add r15, 8
    mov r12, rsi
    jmp .token_loop

.int_error:
    jmp .error

.token_done:
    mov rax, r15
    sub rax, r14
    cmp rax, 8
    jne .error
    mov rdi, qword ptr [r14]
    call print_int
    ret

.error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall
    ret

print_int:
    test rdi, rdi
    jns .positive
    mov rax, qword ptr [rip + max_neg_mag]
    cmp rdi, rax
    jne .neg_not_min
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, min_int_len
    syscall
    ret
.neg_not_min:
    push rdi
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + minus_char]
    mov rdx, 1
    syscall
    pop rdi
    neg rdi
.positive:
    lea rsi, [rip + num_buffer + 23]
    mov byte ptr [rsi], 10
    dec rsi
    mov rax, rdi
    mov rcx, 10
.convert_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz .convert_loop
    inc rsi
    mov rdi, 1
    lea rax, [rip + num_buffer + 24]
    sub rax, rsi
    mov rdx, rax
    mov rax, 1
    syscall
    ret
