.globl _start

.section .bss
.align 8
input_buffer: .zero 65536
rpn_stack: .zero 8192

.section .rodata
error_msg: .ascii "error\n"
error_len = . - error_msg
min_int_str: .ascii "-9223372036854775808\n"
min_int_len = . - min_int_str
max_pos: .quad 9223372036854775807
max_neg_mag: .quad 9223372036854775808

.section .text
_start:
    # Read all input
    lea rdi, [rip + input_buffer]
    mov rsi, 65536
    call read_all
    # Process lines
    lea rdi, [rip + input_buffer]
    mov rsi, rax
    call process_lines
    # Exit
    xor rdi, rdi
    mov rax, 60
    syscall

# read_all: read from stdin until EOF or buffer full
# rdi = buffer, rsi = max bytes
# returns total bytes read in rax
read_all:
    mov rbx, 0          # total read
    mov r12, rdi        # current pointer
    mov r13, rsi        # remaining
.read_loop:
    test r13, r13
    jz .read_done
    mov rax, 0          # sys_read
    mov rdi, 0          # stdin
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz .read_done
    add rbx, rax
    add r12, rax
    sub r13, rax
    jmp .read_loop
.read_done:
    mov rax, rbx
    ret

# process_lines: iterate over lines in buffer
# rdi = buffer start, rsi = length
process_lines:
    mov r12, rdi        # current position
    mov r13, rsi
    add r13, r12        # end pointer
.line_loop:
    cmp r12, r13
    jae .line_done
    # Find newline
    mov rax, r12
.find_nl:
    cmp rax, r13
    jae .no_nl
    cmp byte ptr [rax], 10
    je .found_nl
    inc rax
    jmp .find_nl
.no_nl:
    # Last line without newline
    mov rdi, r12
    mov rsi, r13
    call eval_line
    jmp .line_done
.found_nl:
    mov rdi, r12
    mov rsi, rax
    push r12
    push r13
    call eval_line
    pop r13
    pop r12
    lea r12, [rax + 1]
    jmp .line_loop
.line_done:
    ret

# eval_line: evaluate one RPN line
# rdi = line start, rsi = line end (exclusive)
eval_line:
    mov r12, rdi        # current position in line
    mov r13, rsi        # end of line
    lea r14, [rip + rpn_stack]  # stack base
    mov r15, r14        # stack top (next free slot)
.token_loop:
    cmp r12, r13
    jae .token_done
.skip_spaces:
    cmp r12, r13
    jae .token_done
    mov al, byte ptr [r12]
    cmp al, ' '
    je .skip_next
    cmp al, '\t'
    je .skip_next
    jmp .token_start
.skip_next:
    inc r12
    jmp .skip_spaces
.token_start:
    mov rdi, r12        # token start
.find_token_end:
    cmp r12, r13
    jae .token_end_found
    mov al, byte ptr [r12]
    cmp al, ' '
    je .token_end_found
    cmp al, '\t'
    je .token_end_found
    inc r12
    jmp .find_token_end
.token_end_found:
    mov rsi, r12        # token end
    mov rcx, rsi
    sub rcx, rdi        # token length in rcx
    cmp rcx, 1
    jne .parse_integer
    mov al, byte ptr [rdi]
    cmp al, '+'
    je .op_add
    cmp al, '-'
    je .op_sub
    cmp al, '*'
    je .op_mul
    cmp al, '/'
    je .op_div
    cmp al, '%'
    je .op_mod
    jmp .parse_integer

.op_add:
    # Need at least 2 elements on stack
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]  # b (top)
    sub r15, 8
    mov rax, qword ptr [r15]  # a
    add rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_sub:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    sub rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_mul:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    imul rax, rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_div:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    test rbx, rbx
    jz .error
    # Check for INT64_MIN / -1
    mov rcx, qword ptr [rip + max_neg_mag]
    cmp rax, rcx
    jne .div_ok
    cmp rbx, -1
    je .error
.div_ok:
    cqo
    idiv rbx
    mov qword ptr [r15], rax
    add r15, 8
    jmp .token_loop

.op_mod:
    mov rax, r15
    sub rax, r14
    cmp rax, 16
    jb .error
    sub r15, 8
    mov rbx, qword ptr [r15]
    sub r15, 8
    mov rax, qword ptr [r15]
    test rbx, rbx
    jz .error
    mov rcx, qword ptr [rip + max_neg_mag]
    cmp rax, rcx
    jne .mod_ok
    cmp rbx, -1
    je .error
.mod_ok:
    cqo
    idiv rbx
    mov qword ptr [r15], rdx
    add r15, 8
    jmp .token_loop

.parse_integer:
    # Parse integer literal: optional '-' followed by 1-18 digits
    # rdi = token start, rsi = token end, rcx = length
    mov rax, rdi
    mov rbx, 0          # sign: 0 = positive, 1 = negative
    mov rcx, 0          # digit count
    mov rdx, 0          # accumulated value (unsigned magnitude)
    mov r8, 10
    mov r9, rsi         # token end
.parse_sign:
    cmp rax, r9
    jae .parse_digits
    mov cl, byte ptr [rax]
    cmp cl, '-'
    jne .parse_digits
    mov rbx, 1
    inc rax
    inc rcx
    jmp .parse_digits
.parse_digits:
    cmp rax, r9
    jae .check_digits
    mov cl, byte ptr [rax]
    cmp cl, '0'
    jb .check_digits
    cmp cl, '9'
    ja .check_digits
    # Valid digit
    inc rax
    inc rcx
    sub cl, '0'
    movzx r10, cl
    # Check overflow: rdx * 10 + r10
    mov rax, rdx
    mul r8
    jc .int_error       # overflow in mul (result > 2^64-1)
    add rax, r10
    jc .int_error       # overflow in add
    mov rdx, rax
    # Check digit count <= 18
    cmp rcx, 19
    ja .int_error
    jmp .parse_digits
.check_digits:
    test rcx, rcx
    jz .int_error       # no digits
    cmp rcx, 19
    ja .int_error       # more than 18 digits
    # Check value bounds
    test rbx, rbx
    jz .check_positive
    # negative: max magnitude is 9223372036854775808
    mov rax, qword ptr [rip + max_neg_mag]
    cmp rdx, rax
    ja .int_error
    # Push negative value
    neg rdx
    jmp .push_value
.check_positive:
    # positive: max is 9223372036854775807
    mov rax, qword ptr [rip + max_pos]
    cmp rdx, rax
    ja .int_error
.push_value:
    mov qword ptr [r15], rdx
    add r15, 8
    mov r12, rsi        # continue from token end
    jmp .token_loop

.int_error:
    jmp .error

.token_done:
    # End of line tokens, check stack has exactly one value
    mov rax, r15
    sub rax, r14
    cmp rax, 8
    jne .error
    # Print the value
    mov rdi, qword ptr [r14]
    call print_int
    ret

.error:
    # Print "error\n"
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall
    ret

# print_int: print signed 64-bit integer in rdi followed by newline
print_int:
    test rdi, rdi
    jns .positive
    # Negative: check for INT64_MIN
    mov rax, qword ptr [rip + max_neg_mag]
    cmp rdi, rax
    jne .neg_not_min
    # Special case: -9223372036854775808
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, min_int_len
    syscall
    ret
.neg_not_min:
    # Print '-' then positive magnitude
    push rdi
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + minus_char]
    mov rdx, 1
    syscall
    pop rdi
    neg rdi
.positive:
    # Convert to decimal string
    lea rsi, [rip + num_buffer_end]
    mov byte ptr [rsi], 10  # newline
    dec rsi
    mov rax, rdi
    mov rcx, 10
.convert_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz .convert_loop
    inc rsi
    # Write the string
    mov rdi, 1
    mov rdx, rsi
    sub rdx, [rip + num_buffer_end]  # wrong, need length
    # Calculate length: num_buffer_end - rsi + 1 (for newline)
    mov rdx, qword ptr [rip + num_buffer_end]
    sub rdx, rsi
    inc rdx
    mov rax, 1
    syscall
    ret

.section .rodata
minus_char: .byte '-'
num_buffer_end: .skip 0
