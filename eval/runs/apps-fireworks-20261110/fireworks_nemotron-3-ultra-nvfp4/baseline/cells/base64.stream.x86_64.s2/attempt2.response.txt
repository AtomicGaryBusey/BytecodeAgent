.globl _start

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buffer:   .zero 4096
output_buffer:  .zero 4096
.align 8
input_len:      .zero 8
input_offset:   .zero 8
output_offset:  .zero 8
col_count:      .zero 8
leftover_len:   .zero 1
leftover_byte1: .zero 1
leftover_byte2: .zero 1

.section .text
_start:
    # Zero all state variables
    xor eax, eax
    mov qword ptr [rip + input_len], rax
    mov qword ptr [rip + input_offset], rax
    mov qword ptr [rip + output_offset], rax
    mov qword ptr [rip + col_count], rax
    mov byte ptr [rip + leftover_len], al
    # Base64 table address in rbp
    lea rbp, [rip + base64_table]

read_loop:
    # Load state into registers
    mov r8, qword ptr [rip + input_offset]
    mov r9, qword ptr [rip + input_len]
    lea r8, [rip + input_buffer + r8]        # input_ptr
