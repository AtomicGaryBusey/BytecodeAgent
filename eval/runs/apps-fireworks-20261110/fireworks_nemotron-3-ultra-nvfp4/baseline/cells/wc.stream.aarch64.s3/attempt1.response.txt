.globl _start

.section .text
_start:
    // Initialize counters
    mov x19, #0          // lines
    mov x20, #0          // words
    mov x21, #0          // bytes
    mov x22, #0          // in_word flag

    // Buffer address
    adrp x23, buffer
    add x23, x23, :lo12:buffer

read_loop:
    // sys_read(0, buffer, 4096)
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0

    cbz x0, process_done     // EOF
    mov x25, x0              // bytes read
    add x21, x21, x25        // total bytes

    // Process buffer
    mov x24, x23
    add x24, x24, x25        // end of valid data

process_buffer_loop:
    cmp x23, x24
    b.ge read_loop           // buffer exhausted, read more

    ldrb w5, [x23], #1       // load byte and advance pointer

    // Check for newline (0x0A)
    cmp w5, #0x0A
    b.ne not_newline
    add x19, x19, #1
not_newline:

    // Whitespace check: space (0x20), tab (0x09), newline (0x0A),
    // vertical tab (0x0B), form feed (0x0C), carriage return (0x0D)
    cmp w5, #0x20
    b.eq whitespace
    cmp w5, #0x0D
    b.gt not_whitespace      // > 0x0D and not space -> not whitespace
    cmp w5, #0x09
    b.lt not_whitespace      // < 0x09 -> not whitespace
    // 0x09 .. 0x0D are all whitespace
    b whitespace

not_whitespace:
    // Non-whitespace byte
    cbnz x22, continue_word  // already in a word
    mov x22, #1              // enter word
    add x20, x20, #1         // increment word count
continue_word:
    b process_buffer_loop

whitespace:
    cbz x22, process_buffer_loop
    mov x22, #0              // leave word
    b process_buffer_loop

process_done:
    // Prepare output buffer on stack (64 bytes)
    sub sp, sp, #64
    mov x26, sp              // start of buffer
    add x27, sp, #64         // end of buffer (exclusive)

    // Write bytes (x21) backwards
    mov x0, x21
    mov x1, x27
    bl utoa_back
    mov x27, x0

    // Write space
    sub x27, x27, #1
    mov w5, #' '
    strb w5, [x27]

    // Write words (x20)
    mov x0, x20
    mov x1, x27
    bl utoa_back
    mov x27, x0

    // Write space
    sub x27, x27, #1
    mov w5, #' '
    strb w5, [x27]

    // Write lines (x19)
    mov x0, x19
    mov x1, x27
    bl utoa_back
    mov x27, x0

    // Write newline
    sub x27, x27, #1
    mov w5, #0x0A
    strb w5, [x27]

    // Output the string
    sub x2, x26, x27         // length = end - start
    mov x0, #1               // stdout
    mov x1, x27
    mov x8, #64              // sys_write
    svc #0

    // Exit
    mov x0, #0
    mov x8, #93              // sys_exit
    svc #0

// utoa_back: convert unsigned 64-bit integer in x0 to decimal string
//   x1 = pointer to end of buffer (write backwards)
// Returns new buffer pointer in x0.
utoa_back:
    mov x2, #10
1:
    udiv x3, x0, x2
    msub x4, x3, x2, x0      // x4 = x0 % 10
    add x4, x4, #'0'
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x3
    cbnz x0, 1b
    mov x0, x1
    ret

.section .bss
.align 12
buffer:
    .zero 4096
