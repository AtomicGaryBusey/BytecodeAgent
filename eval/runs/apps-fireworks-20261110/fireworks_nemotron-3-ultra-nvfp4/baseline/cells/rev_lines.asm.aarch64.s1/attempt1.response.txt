.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

.section .text
.globl _start
_start:
    // Get addresses of buffers
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x22, rev_buf
    add  x22, x22, :lo12:rev_buf

    // Read all input into input_buf
    mov x20, #0              // total_read = 0
    mov x21, #BUF_SIZE       // remaining = BUF_SIZE
    mov x23, x19             // buf_ptr = input_buf
read_loop:
    mov x8, #SYS_READ
    mov x0, #0               // stdin
    mov x1, x23
    mov x2, x21
    svc #0
    cbz x0, read_done        // EOF
    add x23, x23, x0         // advance buf_ptr
    sub x21, x21, x0         // remaining -= bytes_read
    add x20, x20, x0         // total_read += bytes_read
    cbnz x21, read_loop      // continue if buffer not full
read_done:
    cbz x20, exit_zero       // no input -> exit

    // Prepare for line processing
    mov x21, x19
    add x21, x21, x20        // end = input_buf + total_read
    mov x23, x19             // src = start of current line

process_loop:
    cmp x23, x21
    b.ge exit_zero           // no more lines

    // Scan for newline
    mov x24, x23             // scan = src
scan_loop:
    cmp x24, x21
    b.ge no_newline          // reached end without newline
    ldrb w25, [x24], #1      // load byte and advance scan
    cmp w25, #0x0A
    b.ne scan_loop
    // Found newline at x24-1
    sub x24, x24, #1         // x24 points to newline
    b process_line

no_newline:
    mov x24, x21             // line_end = end

process_line:
    // line_len = x24 - x23
    sub x25, x24, x23
    cbz x25, write_newline   // zero-length line: just write newline

    // Reverse line into rev_buf
    mov x26, #0              // index
reverse_loop:
    cmp x26, x25
    b.ge reverse_done
    // src_offset = line_len - 1 - index
    sub x27, x25, x26
    sub x27, x27, #1
    ldrb w28, [x23, x27]
    strb w28, [x22, x26]
    add x26, x26, #1
    b reverse_loop

reverse_done:
    // Write reversed line
    mov x8, #SYS_WRITE
    mov x0, #1               // stdout
    mov x1, x22
    mov x2, x25
    svc #0

write_newline:
    // Write newline (0x0A) from stack
    sub sp, sp, #16
    mov w26, #0x0A
    strb w26, [sp]
    mov x8, #SYS_WRITE
    mov x0, #1
    mov x1, sp
    mov x2, #1
    svc #0
    add sp, sp, #16

    // If we processed the last line (no terminating newline), exit
    cmp x24, x21
    b.eq exit_zero

    // Otherwise, skip the newline and continue
    add x23, x24, #1
    b process_loop

exit_zero:
    mov x8, #SYS_EXIT
    mov x0, #0
    svc #0

.section .bss
.align 4
input_buf: .space BUF_SIZE
rev_buf:   .space BUF_SIZE
