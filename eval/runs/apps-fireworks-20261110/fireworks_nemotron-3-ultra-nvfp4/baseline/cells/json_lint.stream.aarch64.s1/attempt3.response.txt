.section .bss
.balign 16
buffer: .space 65536
num_buf: .space 16

.section .rodata
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
error_suffix: .ascii "\n"
error_suffix_len = . - error_suffix
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
ten_const: .word 10

.section .text
.globl _start

_start:
    // Initialize globals
    adrp x19, buffer
    add  x19, x19, :lo12:buffer
    mov  x20, #0          // length
    mov  x21, #0          // index
    mov  x22, #0          // error_offset
    mov  x23, #0          // error_flag

    // Read all input
read_loop:
    mov  x0, #0                     // fd = 0
    add  x1, x19, x20               // buf + offset
    mov  x2, #65536
    sub  x2, x2, x20                // remaining space
    mov  x8, #63                    // sys_read
    svc  #0
    cbz  x0, read_done              // EOF
    add  x20, x20, x0               // increase length
    cmp  x20, #65536
    b.le read_loop
    // input exceeds limit (should not happen per spec)
    b    read_loop

read_done:
    // Parse JSON document
    bl   parse_document
    cbnz x23, output_error          // error_flag set

    // Success: write "ok\n"
    mov  x0, #1                     // stdout
    adrp x1, ok_msg
    add  x1, x1, :lo12:ok_msg
    mov  x2, #ok_len
    mov  x8, #64                    // sys_write
    svc  #0
    mov  x0, #0                     // exit status 0
    b    exit_prog

output_error:
    // Write error message
    bl   write_error
    mov  x0, #1                     // exit status 1

exit_prog:
    mov  x8, #93                    // sys_exit
    svc  #0

// ============================================================
// Parsing functions
// ============================================================

// parse_document: returns 1 on success, 0 on error (sets error_flag)
parse_document:
    stp  x29, x30, [sp, #-16]!
    bl   skip_ws
    cbnz x23, parse_doc_err
    bl   parse_value
    cbz  x0, parse_doc_err
    bl   skip_ws
    cbnz x23, parse_doc_err
    cmp  x21, x20
    b.ne parse_doc_err
    mov  x0, #1
    b    parse_doc_ret
parse_doc_err:
    mov  x0, #0
parse_doc_ret:
    ldp  x29, x30, [sp], #16
    ret

// skip_ws: advance x21 over whitespace (space, tab, newline, carriage return)
skip_ws:
skip_ws_loop:
    cmp  x21, x20
    b.hs skip_ws_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x20
    b.eq skip_ws_next
    cmp  w0, #0x09
    b.eq skip_ws_next
    cmp  w0, #0x0A
    b.eq skip_ws_next
    cmp  w0, #0x0D
    b.eq skip_ws_next
    b    skip_ws_done
skip_ws_next:
    add  x21, x21, #1
    b    skip_ws_loop
skip_ws_done:
    ret

// parse_value: dispatch based on first non-whitespace byte
parse_value:
    stp  x29, x30, [sp, #-16]!
    bl   skip_ws
    cbnz x23, parse_val_err
    cmp  x21, x20
    b.hs parse_val_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x7B      // '{'
    b.eq parse_val_obj
    cmp  w0, #0x5B      // '['
    b.eq parse_val_arr
    cmp  w0, #0x22      // '"'
    b.eq parse_val_str
    cmp  w0, #0x74      // 't'
    b.eq parse_val_true
    cmp  w0, #0x66      // 'f'
    b.eq parse_val_false
    cmp  w0, #0x6E      // 'n'
    b.eq parse_val_null
    cmp  w0, #0x2D      // '-'
    b.eq parse_val_num
    cmp  w0, #0x30      // '0'
    b.lt parse_val_err
    cmp  w0, #0x39
    b.le parse_val_num
    b    parse_val_err

parse_val_obj:
    bl   parse_object
    b    parse_val_ret
parse_val_arr:
    bl   parse_array
    b    parse_val_ret
parse_val_str:
    bl   parse_string
    b    parse_val_ret
parse_val_true:
    bl   parse_true
    b    parse_val_ret
parse_val_false:
    bl   parse_false
    b    parse_val_ret
parse_val_null:
    bl   parse_null
    b    parse_val_ret
parse_val_num:
    bl   parse_number
    b    parse_val_ret

parse_val_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_val_err

parse_val_err:
    mov  x0, #0
parse_val_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_object
parse_object:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_obj_err
    cmp  x21, x20
    b.hs parse_obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x7B
    b.ne parse_obj_err
    add  x21, x21, #1
    bl   skip_ws
    cbnz x23, parse_obj_err
    cmp  x21, x20
    b.hs parse_obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x7D      // '}'
    b.ne parse_obj_member
    add  x21, x21, #1
    mov  x0, #1
    b    parse_obj_ret

parse_obj_member:
    bl   parse_string
    cbz  x0, parse_obj_err
    bl   skip_ws
    cbnz x23, parse_obj_err
    cmp  x21, x20
    b.hs parse_obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x3A      // ':'
    b.ne parse_obj_err
    add  x21, x21, #1
    bl   skip_ws
    cbnz x23, parse_obj_err
    bl   parse_value
    cbz  x0, parse_obj_err

parse_obj_loop:
    bl   skip_ws
    cbnz x23, parse_obj_err
    cmp  x21, x20
    b.hs parse_obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x2C      // ','
    b.eq parse_obj_comma
    cmp  w0, #0x7D      // '}'
    b.eq parse_obj_close
    b    parse_obj_err

parse_obj_comma:
    add  x21, x21, #1
    bl   skip_ws
    cbnz x23, parse_obj_err
    b    parse_obj_member

parse_obj_close:
    add  x21, x21, #1
    mov  x0, #1
    b    parse_obj_ret

parse_obj_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_obj_err

parse_obj_err:
    mov  x0, #0
parse_obj_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_array
parse_array:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_arr_err
    cmp  x21, x20
    b.hs parse_arr_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x5B      // '['
    b.ne parse_arr_err
    add  x21, x21, #1
    bl   skip_ws
    cbnz x23, parse_arr_err
    cmp  x21, x20
    b.hs parse_arr_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x5D      // ']'
    b.ne parse_arr_value
    add  x21, x21, #1
    mov  x0, #1
    b    parse_arr_ret

parse_arr_value:
    bl   parse_value
    cbz  x0, parse_arr_err
    bl   skip_ws
    cbnz x23, parse_arr_err
    cmp  x21, x20
    b.hs parse_arr_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x2C      // ','
    b.eq parse_arr_comma
    cmp  w0, #0x5D      // ']'
    b.eq parse_arr_close
    b    parse_arr_err

parse_arr_comma:
    add  x21, x21, #1
    bl   skip_ws
    cbnz x23, parse_arr_err
    b    parse_arr_value

parse_arr_close:
    add  x21, x21, #1
    mov  x0, #1
    b    parse_arr_ret

parse_arr_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_arr_err

parse_arr_err:
    mov  x0, #0
parse_arr_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_string
parse_string:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_str_err
    cmp  x21, x20
    b.hs parse_str_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x22      // '"'
    b.ne parse_str_err
    add  x21, x21, #1

parse_str_loop:
    cmp  x21, x20
    b.hs parse_str_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x22      // closing quote
    b.eq parse_str_close
    cmp  w0, #0x5C      // backslash
    b.eq parse_str_escape
    cmp  w0, #0x20
    b.lt parse_str_err  // control character
    add  x21, x21, #1
    b    parse_str_loop

parse_str_escape:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_str_eof
    ldrb w0, [x19, x21]
    // valid escapes: " \ / b f n r t u
    cmp  w0, #0x22
    b.eq parse_str_esc_ok
    cmp  w0, #0x5C
    b.eq parse_str_esc_ok
    cmp  w0, #0x2F
    b.eq parse_str_esc_ok
    cmp  w0, #0x62
    b.eq parse_str_esc_ok
    cmp  w0, #0x66
    b.eq parse_str_esc_ok
    cmp  w0, #0x6E
    b.eq parse_str_esc_ok
    cmp  w0, #0x72
    b.eq parse_str_esc_ok
    cmp  w0, #0x74
    b.eq parse_str_esc_ok
    cmp  w0, #0x75
    b.eq parse_str_unicode
    b    parse_str_err

parse_str_esc_ok:
    add  x21, x21, #1
    b    parse_str_loop

parse_str_unicode:
    add  x21, x21, #1
    mov  x1, #4
parse_str_uni_loop:
    cmp  x21, x20
    b.hs parse_str_eof
    ldrb w0, [x19, x21]
    // hex digit check
    cmp  w0, #0x30
    b.lt parse_str_err
    cmp  w0, #0x39
    b.le parse_str_uni_next
    cmp  w0, #0x41
    b.lt parse_str_err
    cmp  w0, #0x46
    b.le parse_str_uni_next
    cmp  w0, #0x61
    b.lt parse_str_err
    cmp  w0, #0x66
    b.gt parse_str_err
parse_str_uni_next:
    add  x21, x21, #1
    subs x1, x1, #1
    b.ne parse_str_uni_loop
    b    parse_str_loop

parse_str_close:
    add  x21, x21, #1
    mov  x0, #1
    b    parse_str_ret

parse_str_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_str_err

parse_str_err:
    mov  x0, #0
parse_str_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_number
parse_number:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_num_err
    // optional '-'
    cmp  x21, x20
    b.hs parse_num_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x2D
    b.ne parse_num_int
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_num_eof
    ldrb w0, [x19, x21]

parse_num_int:
    cmp  w0, #0x30
    b.eq parse_num_zero
    cmp  w0, #0x31
    b.lt parse_num_err
    cmp  w0, #0x39
    b.gt parse_num_err
    // digit 1-9
    add  x21, x21, #1
    b    parse_num_int_digits

parse_num_zero:
    add  x21, x21, #1
    b    parse_num_after_int

parse_num_int_digits:
    cmp  x21, x20
    b.hs parse_num_after_int
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.lt parse_num_after_int
    cmp  w0, #0x39
    b.gt parse_num_after_int
    add  x21, x21, #1
    b    parse_num_int_digits

parse_num_after_int:
    cmp  x21, x20
    b.hs parse_num_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x2E      // '.'
    b.eq parse_num_frac
    cmp  w0, #0x65      // 'e'
    b.eq parse_num_exp
    cmp  w0, #0x45      // 'E'
    b.eq parse_num_exp
    b    parse_num_done

parse_num_frac:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_num_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.lt parse_num_err
    cmp  w0, #0x39
    b.gt parse_num_err
    add  x21, x21, #1
    b    parse_num_frac_digits

parse_num_frac_digits:
    cmp  x21, x20
    b.hs parse_num_after_frac
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.lt parse_num_after_frac
    cmp  w0, #0x39
    b.gt parse_num_after_frac
    add  x21, x21, #1
    b    parse_num_frac_digits

parse_num_after_frac:
    cmp  x21, x20
    b.hs parse_num_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x65
    b.eq parse_num_exp
    cmp  w0, #0x45
    b.eq parse_num_exp
    b    parse_num_done

parse_num_exp:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_num_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x2B      // '+'
    b.eq parse_num_exp_sign
    cmp  w0, #0x2D      // '-'
    b.eq parse_num_exp_sign
    b    parse_num_exp_digits_check

parse_num_exp_sign:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_num_eof

parse_num_exp_digits_check:
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.lt parse_num_err
    cmp  w0, #0x39
    b.gt parse_num_err
    add  x21, x21, #1
    b    parse_num_exp_digits

parse_num_exp_digits:
    cmp  x21, x20
    b.hs parse_num_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.lt parse_num_done
    cmp  w0, #0x39
    b.gt parse_num_done
    add  x21, x21, #1
    b    parse_num_exp_digits

parse_num_done:
    mov  x0, #1
    b    parse_num_ret

parse_num_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_num_err

parse_num_err:
    mov  x0, #0
parse_num_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_true
parse_true:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_true_err
    // 't','r','u','e'
    cmp  x21, x20
    b.hs parse_true_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x74
    b.ne parse_true_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_true_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x72
    b.ne parse_true_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_true_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x75
    b.ne parse_true_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_true_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x65
    b.ne parse_true_err
    add  x21, x21, #1
    mov  x0, #1
    b    parse_true_ret

parse_true_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_true_err

parse_true_err:
    mov  x0, #0
parse_true_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_false
parse_false:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_false_err
    // 'f','a','l','s','e'
    cmp  x21, x20
    b.hs parse_false_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x66
    b.ne parse_false_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_false_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x61
    b.ne parse_false_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_false_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x6C
    b.ne parse_false_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_false_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x73
    b.ne parse_false_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_false_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x65
    b.ne parse_false_err
    add  x21, x21, #1
    mov  x0, #1
    b    parse_false_ret

parse_false_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_false_err

parse_false_err:
    mov  x0, #0
parse_false_ret:
    ldp  x29, x30, [sp], #16
    ret

// parse_null
parse_null:
    stp  x29, x30, [sp, #-16]!
    cbnz x23, parse_null_err
    // 'n','u','l','l'
    cmp  x21, x20
    b.hs parse_null_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x6E
    b.ne parse_null_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_null_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x75
    b.ne parse_null_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_null_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x6C
    b.ne parse_null_err
    add  x21, x21, #1
    cmp  x21, x20
    b.hs parse_null_eof
    ldrb w0, [x19, x21]
    cmp  w0, #0x6C
    b.ne parse_null_err
    add  x21, x21, #1
    mov  x0, #1
    b    parse_null_ret

parse_null_eof:
    mov  x23, #1
    mov  x22, x21
    b    parse_null_err

parse_null_err:
    mov  x0, #0
parse_null_ret:
    ldp  x29, x30, [sp], #16
    ret

// ============================================================
// Output functions
// ============================================================

// write_error: prints "error at N\n" where N is in x22
write_error:
    stp  x29, x30, [sp, #-16]!
    // write prefix
    mov  x0, #1
    adrp x1, error_prefix
    add  x1, x1, :lo12:error_prefix
    mov  x2, #error_prefix_len
    mov  x8, #64
    svc  #0

    // convert x22 to decimal in num_buf (backwards)
    adrp x3, num_buf
    add  x3, x3, :lo12:num_buf
    add  x4, x3, #15      // end of buffer
    mov  x5, x22
    cbz  x5, write_zero

write_digits:
    cmp  x5, #0
    beq  write_digits_done
    mov  x6, x5
    adrp x7, ten_const
    ldr  w7, [x7, :lo12:ten_const]
    udiv x5, x5, x7
    msub x6, x5, x7, x6   // x6 = remainder
    add  w6, w6, #0x30
    strb w6, [x4], #-1
    b    write_digits

write_zero:
    mov  w6, #0x30
    strb w6, [x4], #-1

write_digits_done:
    // x4 points to one before first digit
    add  x1, x4, #1
    sub  x2, x3, x4
    add  x2, x2, #15
    mov  x0, #1
    mov  x8, #64
    svc  #0

    // write suffix
    mov  x0, #1
    adrp x1, error_suffix
    add  x1, x1, :lo12:error_suffix
    mov  x2, #error_suffix_len
    mov  x8, #64
    svc  #0

    ldp  x29, x30, [sp], #16
    ret
