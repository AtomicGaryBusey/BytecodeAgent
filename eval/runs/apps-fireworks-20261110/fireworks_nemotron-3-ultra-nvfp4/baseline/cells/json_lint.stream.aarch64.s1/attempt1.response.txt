.section .bss
.balign 16
buffer: .space 65536
num_buf: .space 16

.section .rodata
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
error_suffix: .ascii "\n"
error_suffix_len = . - error_suffix
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg

.section .text
.globl _start

_start:
    // x19 = buffer base, x20 = buf_len, x21 = idx, x22 = error_idx
    adrp x19, buffer
    add  x19, x19, :lo12:buffer
    mov  x20, #0
    mov  x21, #0
    mov  x22, #0

    // read all input
read_loop:
    mov  x0, #0                     // fd = 0
    add  x1, x19, x20               // buf + offset
    sub  x2, #65536, x20            // remaining space
