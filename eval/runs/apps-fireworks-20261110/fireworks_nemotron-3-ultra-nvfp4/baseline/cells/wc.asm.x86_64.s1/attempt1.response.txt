.section .bss
.align 16
read_buf: .zero 8192
out_buf: .zero 128

.section .text
.equ READ_BUF_SIZE, 8192
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor rbx, rbx
    lea r15, [rip + out_buf]

read_loop:
    mov rdi, 0
    lea rsi, [rip + read_buf]
    mov rdx, READ_BUF_SIZE
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + read_buf]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    je handle_newline
    movzx eax, byte ptr [rip + ws_table + rax]
    test al, al
    jnz handle_whitespace
    test rbx, rbx
    jnz next_byte
    inc r13
    mov rbx, 1
    jmp next_byte
handle_newline:
    inc r12
handle_whitespace:
    mov rbx, 0
next_byte:
    dec rcx
    jnz process_loop
    jmp read_loop

read_done:
    mov rdi, r12
    call append_number
    mov byte ptr [r15], ' '
    inc r15
    mov rdi, r13
    call append_number
    mov byte ptr [r15], ' '
    inc r15
    mov rdi, r14
    call append_number
    mov byte ptr [r15], 0x0A
    inc r15

    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

append_number:
    sub rsp, 32
    mov rsi, rsp
    mov rcx, 0
    mov r8, rdi
    test r8, r8
    jnz .Ldiv_loop
    mov byte ptr [rsi], '0'
    inc rcx
    jmp .Lcopy_digits
.Ldiv_loop:
    mov rax, r8
    mov rdi, 10
    xor rdx, rdx
    div rdi
    mov r8, rax
    add dl, '0'
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test r8, r8
    jnz .Ldiv_loop
.Lcopy_digits:
    lea rdi, [rsi + rcx - 1]
    mov rsi, r15
.Lcopy_loop:
    mov al, byte ptr [rdi]
    mov byte ptr [rsi], al
    inc rsi
    dec rdi
    cmp rdi, rsp
    jge .Lcopy_loop
    add r15, rcx
    add rsp, 32
    ret
