.section .bss
.align 4096
input_buf:    .space 65536
.align 8
input_len:    .zero 8
parse_pos:    .zero 8
error_pos:    .zero 8
num_buf:      .space 32

.section .data
msg_ok:           .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err_prefix:   .ascii "error at "
msg_err_prefix_len = . - msg_err_prefix
msg_nl:           .ascii "\n"
msg_nl_len = . - msg_nl

.section .text
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.equ CHAR_SPACE,  0x20
.equ CHAR_TAB,    0x09
.equ CHAR_NL,     0x0A
.equ CHAR_CR,     0x0D
.equ CHAR_LBRACE, 0x7B
.equ CHAR_RBRACE, 0x7D
.equ CHAR_LBRACK, 0x5B
.equ CHAR_RBRACK, 0x5D
.equ CHAR_COLON,  0x3A
.equ CHAR_COMMA,  0x2C
.equ CHAR_QUOTE,  0x22
.equ CHAR_BSLASH, 0x5C
.equ CHAR_MINUS,  0x2D
.equ CHAR_DOT,    0x2E
.equ CHAR_0,      0x30
.equ CHAR_1,      0x31
.equ CHAR_9,      0x39
.equ CHAR_a,      0x61
.equ CHAR_f,      0x66
.equ CHAR_A,      0x41
.equ CHAR_F,      0x46
.equ CHAR_e,      0x65
.equ CHAR_E,      0x45
.equ CHAR_PLUS,   0x2B
.equ CHAR_t,      0x74
.equ CHAR_r,      0x72
.equ CHAR_u,      0x75
.equ CHAR_n,      0x6E
.equ CHAR_b,      0x62
.equ CHAR_l,      0x6C
.equ CHAR_s,      0x73
.equ CHAR_d,      0x64
.equ CHAR_SLASH,  0x2F

_start:
    mov qword ptr [rip + input_len], 0
    mov qword ptr [rip + parse_pos], 0
    mov qword ptr [rip + error_pos], -1

    mov r12, 0
.read_loop:
    cmp r12, 65536
    jae .read_done
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    jmp .read_loop

.read_done:
    mov qword ptr [rip + input_len], r12
    call parse_document
    test rax, rax
    jz .print_error
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    mov rax, SYS_EXIT
    mov rdi, 0
    syscall

.print_error:
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_err_prefix]
    mov rdx, msg_err_prefix_len
    syscall

    mov rax, qword ptr [rip + error_pos]
    lea rdi, [rip + num_buf + 31]
    mov byte ptr [rdi], 0x0A
    mov rcx, 10
    test rax, rax
    js .write_num_done
.convert_loop:
    dec rdi
    xor rdx, rdx
    div rcx
    add dl, CHAR_0
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .convert_loop
.write_num_done:
    lea rsi, [rdi]
    lea rdx, [rip + num_buf + 32]
    sub rdx, rsi
    mov rax, SYS_WRITE
    mov rdi, 1
    syscall

    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_nl]
    mov rdx, msg_nl_len
    syscall

    mov rax, SYS_EXIT
    mov rdi, 1
    syscall

parse_document:
    call skip_ws
    cmp qword ptr [rip + error_pos], -1
    jne .err_ret
    call parse_value
    cmp qword ptr [rip + error_pos], -1
    jne .err_ret
    call skip_ws
    cmp qword ptr [rip + error_pos], -1
    jne .err_ret
    mov rax, qword ptr [rip + parse_pos]
    cmp rax, qword ptr [rip + input_len]
    je .ok_ret
    mov qword ptr [rip + error_pos], rax
.err_ret:
    xor rax, rax
    ret
.ok_ret:
    mov rax, 1
    ret

skip_ws:
    mov rax, qword ptr [rip + parse_pos]
    mov rbx, qword ptr [rip + input_len]
    cmp rax, rbx
    jae .ws_done
.ws_loop:
    movzx ecx, byte ptr [rip + input_buf + rax]
