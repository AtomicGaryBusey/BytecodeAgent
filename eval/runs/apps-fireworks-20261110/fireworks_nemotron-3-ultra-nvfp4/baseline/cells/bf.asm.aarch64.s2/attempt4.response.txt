.section .bss
.align 12
stdin_buf:
    .zero 65536
tape:
    .zero 30000
bracket_stack:
    .zero 4000
jump_table:
    .zero 262144

.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94

.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NESTING, 1000

_start:
    // Load addresses of .bss symbols
    adrp x27, stdin_buf
    add  x27, x27, :lo12:stdin_buf

    adrp x26, bracket_stack
    add  x26, x26, :lo12:bracket_stack

    adrp x25, jump_table
    add  x25, x25, :lo12:jump_table

    // Read all stdin into stdin_buf
    mov  x19, #0                      // total bytes read
    mov  x20, x27                     // buffer pointer
    movz x21, #0x1, lsl #16           // MAX_INPUT = 65536

read_loop:
    cmp  x21, #0
    beq  read_done
    mov  x8, #SYS_READ
    mov  x0, #0
    mov  x1, x20
    mov  x2, x21
    svc  #0
    cbz  x0, read_done
    add  x19, x19, x0
    add  x20, x20, x0
    sub  x21, x21, x0
    b    read_loop

read_done:
    // x19 = total bytes read, x27 = stdin_buf, x25 = jump_table, x26 = bracket_stack
    // Find first '!' (0x21)
    mov  x20, x27
    mov  x21, x19
    mov  x22, #0

find_bang:
    cmp  x22, x21
    bge  no_bang
    ldrb w23, [x20, x22]
    cmp  w23, #0x21
    beq  found_bang
    add  x22, x22, #1
    b    find_bang

found_bang:
    mov  x23, x22                     // program_len
    add  x24, x22, #1                 // input_start offset
    sub  x21, x19, x24                // input_len (reuse x21)
    b    validate_brackets

no_bang:
    mov  x23, x19                     // program_len
    mov  x24, #0                      // input_start offset
    mov  x21, #0                      // input_len

validate_brackets:
    // Save input_start (x24) and input_len (x21) for later interpretation
    mov  x28, x24
    mov  x29, x21

    // Registers for validation:
    // x20 = program pointer (stdin_buf)
    // x21 = program_len (x23)
    // x22 = current index
    // x25 = jump_table base (preserved)
    // x26 = bracket_stack pointer (current)
    // x30 = bracket_stack base
    mov  x20, x27
    mov  x21, x23
    mov  x22, #0
    mov  x30, x26                     // save bracket_stack base

validate_loop:
    cmp  x22, x21
    bge  validate_done
    ldrb w24, [x20, x22]
    cmp  w24, #'['
    bne  check_close
    // Push current index onto stack
    str  w22, [x26], #4
    // Check stack overflow (max 4000 bytes = 1000 entries)
    sub  x24, x26, x30
    cmp  x24, #4000
    bhi  bracket_error
    b    validate_next

check_close:
    cmp  w24, #']'
    bne  validate_next
    // Pop from stack
    cmp  x26, x30
    beq  bracket_error                // unmatched ']'
    sub  x26, x26, #4
    ldr  w24, [x26]                   // matching '[' index
    // Set jump_table for both brackets
    str  w22, [x25, x24, lsl #2]      // jump_table[open] = close
    str  w24, [x25, x22, lsl #2]      // jump_table[close] = open
    b    validate_next

validate_next:
    add  x22, x22, #1
    b    validate_loop

validate_done:
    // Check if any unmatched '[' left
    cmp  x26, x30
    bne  bracket_error
    // Validation successful, proceed to interpret
    b    interpret

bracket_error:
    mov  x8, #SYS_EXIT
    mov  x0, #1
    svc  #0

interpret:
    // Load tape address
    adrp x26, tape
    add  x26, x26, :lo12:tape

    // Registers for interpretation:
    // x19 = tape pointer (0..29999)
    // x20 = program counter (index)
    // x21 = program length
    // x22 = input index
    // x23 = input length
    // x24 = input start offset
    // x25 = jump_table base
    // x26 = tape base
    // x27 = stdin_buf base
    // x28 = input_start offset (saved)
    // x29 = input_len (saved)
    mov  x19, #0
    mov  x20, #0
    mov  x21, x23                     // program_len
    mov  x22, #0                      // input index
    mov  x23, x29                     // input_len
    // x24 already input_start offset (x28)
    // x25 jump_table base
    // x26 tape base
    // x27 stdin_buf base

interpret_loop:
    cmp  x20, x21
    bge  interpret_done
    ldrb w24, [x27, x20]              // current instruction

    cmp  w24, #'>'
    bne  check_lt
    add  x19, x19, #1
    b    inc_pc

check_lt:
    cmp  w24, #'<'
    bne  check_plus
    sub  x19, x19, #1
    b    inc_pc

check_plus:
    cmp  w24, #'+'
    bne  check_minus
    ldrb w30, [x26, x19]
    add  w30, w30, #1
    strb w30, [x26, x19]
    b    inc_pc

check_minus:
    cmp  w24, #'-'
    bne  check_dot
    ldrb w30, [x26, x19]
    sub  w30, w30, #1
    strb w30, [x26, x19]
    b    inc_pc

check_dot:
    cmp  w24, #'.'
    bne  check_comma
    ldrb w30, [x26, x19]
    // Write byte to stdout
    sub  sp, sp, #16
    strb w30, [sp]
    mov  x8, #SYS_WRITE
    mov  x0, #1
    mov  x1, sp
    mov  x2, #1
    svc  #0
    add  sp, sp, #16
    b    inc_pc

check_comma:
    cmp  w24, #','
    bne  check_lbracket
    cmp  x22, x23
    bge  input_exhausted
    // Input available: stdin_buf[input_start + x22]
    add  x30, x27, x28                // stdin_buf + input_start
    ldrb w30, [x30, x22]
    add  x22, x22, #1
    strb w30, [x26, x19]
    b    inc_pc

input_exhausted:
    strb wzr, [x26, x19]
    b    inc_pc

check_lbracket:
    cmp  w24, #'['
    bne  check_rbracket
    ldrb w30, [x26, x19]
    cbnz w30, inc_pc
    // Jump to matching ']' + 1
    ldr  w30, [x25, x20, lsl #2]
    add  x20, x30, #1
    b    interpret_loop

check_rbracket:
    cmp  w24, #']'
    bne  inc_pc
    ldrb w30, [x26, x19]
    cbz  w30, inc_pc
    // Jump back to matching '[' + 1
    ldr  w30, [x25, x20, lsl #2]
    add  x20, x30, #1
    b    interpret_loop

inc_pc:
    add  x20, x20, #1
    b    interpret_loop

interpret_done:
    mov  x8, #SYS_EXIT
    mov  x0, #0
    svc  #0
