.global _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .bss
.balign 4096
input_buffer: .space 65536
output_buffer: .space 256

.section .text
_start:
    // Read all input into input_buffer
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer   // x19 = input_buffer base
    mov x20, x19                       // x20 = current write pointer
    mov x21, #65536                    // x21 = remaining capacity
    mov x22, #0                        // x22 = total bytes read

read_loop:
    mov x8, #63                        // read
    mov x0, #0                         // stdin
    mov x1, x20
    mov x2, x21
    svc #0
    cbz x0, read_done                  // EOF
    add x22, x22, x0                   // total += nread
    add x20, x20, x0                   // advance write pointer
    sub x21, x21, x0                   // reduce remaining
    cbnz x21, read_loop                // continue if space left

read_done:
    // Process input in 16-byte rows
    mov x20, #0                        // x20 = offset of current row
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits     // x23 = hex_digits table

process_loop:
    cmp x20, x22
    b.ge print_final_offset            // all bytes processed

    // Row length = min(16, total - offset)
    sub x21, x22, x20
    cmp x21, #16
    mov x24, #16
    csel x21, x21, x24, hi             // x21 = row_len (1..16)

    // Format and print one row
    // x19 = buffer base, x20 = offset, x21 = row_len, x23 = hex_digits
    bl format_row

    add x20, x20, #16
    b process_loop

print_final_offset:
    // Print total length as 8 hex digits + newline
    // x22 = total bytes, x23 = hex_digits
    bl print_offset_newline

    // Exit
    mov x8, #93
    mov x0, #0
    svc #0

// format_row: format one 16-byte row and write to stdout
// x19 = input_buffer base
// x20 = row offset (0, 16, 32, ...)
// x21 = row_len (1..16)
// x23 = hex_digits address
format_row:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer  // x24 = output_buffer base
    mov x25, x24                       // x25 = current output pointer

    // 1. Offset: 8 hex digits + two spaces
    mov x26, #28                       // shift for nibble 7
    mov x27, #8                        // 8 nibbles
offset_loop:
    lsr x28, x20, x26
    and x28, x28, #0xf
    ldrb w28, [x23, x28]
    strb w28, [x25], #1
    subs x26, x26, #4
    subs x27, x27, #1
    b.ne offset_loop

    // Two spaces
    mov w28, #0x2020
    strh w28, [x25], #2

    // 2. Hex bytes: 16 positions
    mov x26, #0                        // i = 0
hex_byte_loop:
    cmp x26, #16
    b.ge hex_byte_done

    cmp x26, x21                       // i < row_len ?
    b.ge hex_byte_blank

    // Real byte
    add x27, x19, x20
    add x27, x27, x26
    ldrb w27, [x27]

    // High nibble
    lsr w28, w27, #4
    ldrb w28, [x23, x28]
    strb w28, [x25], #1
    // Low nibble
    and w28, w27, #0xf
    ldrb w28, [x23, x28]
    strb w28, [x25], #1
    // Space
    mov w28, #' '
    strb w28, [x25], #1
    b hex_byte_after_content

hex_byte_blank:
    // Three spaces
    mov w28, #' '
    strb w28, [x25], #1
    strb w28, [x25], #1
    strb w28, [x25], #1

hex_byte_after_content:
    // Extra space after position 7
    cmp x26, #7
    b.ne hex_byte_noextra
    mov w28, #' '
    strb w28, [x25], #1
hex_byte_noextra:
    add x26, x26, #1
    b hex_byte_loop

hex_byte_done:
    // 3. " |" + ASCII + "|" + newline
    mov w26, #0x7c20                   // " |" (little-endian: space, '|')
    strh w26, [x25], #2

    // ASCII column
    mov x26, #0
ascii_loop:
    cmp x26, x21
    b.ge ascii_done

    add x27, x19, x20
    add x27, x27, x26
    ldrb w27, [x27]

    // Printable 0x20..0x7e ?
    cmp w27, #0x20
    b.lt ascii_dot
    cmp w27, #0x7e
    b.gt ascii_dot
    strb w27, [x25], #1
    b ascii_next

ascii_dot:
    mov w27, #'.'
    strb w27, [x25], #1

ascii_next:
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w26, #0x0a7c                   // "|\n"
    strh w26, [x25], #2

    // Write output buffer
    sub x2, x25, x24                   // length
    mov x8, #64                        // write
    mov x0, #1                         // stdout
    mov x1, x24
    svc #0

    ldp x29, x30, [sp], #16
    ret

// print_offset_newline: print x22 as 8 hex digits + newline
// x22 = value, x23 = hex_digits
print_offset_newline:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, x24

    mov x26, #28
    mov x27, #8
final_offset_loop:
    lsr x28, x22, x26
    and x28, x28, #0xf
    ldrb w28, [x23, x28]
    strb w28, [x25], #1
    subs x26, x26, #4
    subs x27, x27, #1
    b.ne final_offset_loop

    mov w28, #'\n'
    strb w28, [x25], #1

    sub x2, x25, x24
    mov x8, #64
    mov x0, #1
    mov x1, x24
    svc #0

    ldp x29, x30, [sp], #16
    ret
