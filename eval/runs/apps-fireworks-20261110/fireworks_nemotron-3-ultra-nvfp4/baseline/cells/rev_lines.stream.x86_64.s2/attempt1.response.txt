.section .bss
.align 4096
input_buf:
    .zero 65536
output_buf:
    .zero 66000

.section .text
.global _start
_start:
    # Read all input into input_buf
    lea rbx, [rip + input_buf]
    xor r14, r14                 # total_read = 0
    mov r15, 65536               # max_size
read_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # stdin
    lea rsi, [rbx + r14]
    mov rdx, r15
    sub rdx, r14                 # remaining bytes
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    cmp r14, r15
    jne read_loop
read_done:
    lea rbp, [rbx + r14]         # rbp = input_end
    lea r12, [rip + output_buf]  # r12 = output_buf base
    mov r13, r12                 # r13 = output_ptr
    mov r14, rbx                 # r14 = line_start
process_outer:
    cmp r14, rbp
    jae process_done             # no more lines
    mov r15, r14                 # r15 = scan_ptr
find_newline:
    cmp r15, rbp
    je no_newline_found
    mov cl, byte ptr [r15]
    inc r15
    cmp cl, 0x0A
    jne find_newline
    # Found newline at r15-1
    lea rsi, [r15 - 1]           # line_end_exclusive
    mov rcx, rsi
    sub rcx, r14                 # length
    test rcx, rcx
    jz write_newline_after_found
    lea rsi, [r15 - 2]           # last byte of line
    mov rdi, r13
reverse_loop_found:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    dec rcx
    jnz reverse_loop_found
    mov r13, rdi
write_newline_after_found:
    mov byte ptr [r13], 0x0A
    inc r13
    mov r14, r15
    jmp process_outer
no_newline_found:
    # Final line without terminating newline
    mov rsi, rbp
    mov rcx, rsi
    sub rcx, r14
    test rcx, rcx
    jz write_newline_after_no_newline
    lea rsi, [rbp - 1]
    mov rdi, r13
reverse_loop_no_newline:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    dec rcx
    jnz reverse_loop_no_newline
    mov r13, rdi
write_newline_after_no_newline:
    mov byte ptr [r13], 0x0A
    inc r13
    jmp process_done
process_done:
    mov rax, 1                   # sys_write
    mov rdi, 1                   # stdout
    mov rsi, r12
    mov rdx, r13
    sub rdx, r12                 # output length
    syscall
    mov rax, 60                  # sys_exit
    xor rdi, rdi
    syscall
