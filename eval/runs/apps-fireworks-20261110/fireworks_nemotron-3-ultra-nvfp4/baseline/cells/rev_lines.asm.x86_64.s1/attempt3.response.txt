.section .rodata
newline: .byte 10

.section .text
.globl _start

_start:
    # Allocate 65536-byte buffer on the stack (1 MiB available, 65536 is multiple of 16)
    sub rsp, 65536
    mov rbx, rsp            # rbx = buffer base
    xor rbp, rbp            # total_read = 0
    mov r14, 65536          # max size

read_loop:
    mov rdi, 0              # stdin
    lea rsi, [rbx + rbp]
    mov rdx, r14
    sub rdx, rbp            # remaining space
    test rdx, rdx
    jz read_done
    mov rax, 0              # sys_read
    syscall
    test rax, rax
    jz read_done            # EOF
    js read_error           # error (negative)
    add rbp, rax            # total_read += bytes read
    cmp rbp, r14
    jne read_loop

read_done:
    # rbp = total length
    mov r12, 0              # pos
    mov r13, 0              # line_start
    mov r14, rbp            # len
    test r14, r14
    jz exit_success         # empty input

process_loop:
    cmp r12, r14
    jae process_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 10              # newline?
    jne not_newline

    # Found newline at r12, line is [r13, r12)
    mov rdi, r13
    mov rsi, r12
    call reverse_buffer     # reverse in place

    # Write reversed line
    mov rdi, 1              # stdout
    lea rsi, [rbx + r13]
    mov rdx, r12
    sub rdx, r13            # length
    test rdx, rdx
    jz write_newline
    call write_all

write_newline:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all

    inc r12
    mov r13, r12
    jmp process_loop

not_newline:
    inc r12
    jmp process_loop

process_done:
    # Final line (if any) from r13 to r14
    cmp r13, r14
    jae exit_success
    mov rdi, r13
    mov rsi, r14
    call reverse_buffer

    mov rdi, 1
    lea rsi, [rbx + r13]
    mov rdx, r14
    sub rdx, r13
    call write_all

    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_all

exit_success:
    xor rdi, rdi
    mov rax, 60             # sys_exit
    syscall

read_error:
    mov rdi, 1
    mov rax, 60
    syscall

# reverse_buffer(rdi=start, rsi=end_exclusive)
# Reverses bytes in [rdi, rsi-1] in place.
reverse_buffer:
    cmp rdi, rsi
    jae 1f
    dec rsi                 # make inclusive end
2:
    cmp rdi, rsi
    jae 1f
    mov al, byte ptr [rdi]
    mov ah, byte ptr [rsi]
    mov byte ptr [rdi], ah
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp 2b
1:
    ret

# write_all(rdi=fd, rsi=buf, rdx=len)
# Writes exactly len bytes, looping on partial writes.
write_all:
    mov r15, rdx            # remaining
    test r15, r15
    jz 1f
2:
    mov rax, 1              # sys_write
    syscall
    test rax, rax
    js 3f
    sub r15, rax
    jz 1f
    add rsi, rax
    jmp 2b
1:
    ret
3:
    mov rdi, 1
    mov rax, 60
    syscall
