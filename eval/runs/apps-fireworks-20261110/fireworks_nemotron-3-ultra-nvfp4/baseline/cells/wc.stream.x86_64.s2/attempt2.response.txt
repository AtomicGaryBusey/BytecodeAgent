.globl _start

.section .bss
.align 16
buffer: .zero 8192
output_buffer: .zero 64

.section .text
_start:
    # Initialize counters
    xor r12, r12          # lines
    xor r13, r13          # words
    xor r14, r14          # bytes
    xor r15, r15          # in_word flag (0 = not in word, 1 = in word)

read_loop:
    # sys_read(fd=0, buf, count=8192)
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 8192
    syscall

    test rax, rax
    jz process_done

    add r14, rax          # update byte count

    mov rcx, rax          # number of bytes read
    lea rsi, [rip + buffer]

process_byte:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10            # '\n'
    je newline
    cmp al, 32            # ' '
    je whitespace
    cmp al, 9             # '\t'
    je whitespace
    cmp al, 11            # '\v'
    je whitespace
    cmp al, 12            # '\f'
    je whitespace
    cmp al, 13            # '\r'
    je whitespace

    # Non-whitespace byte
    test r15, r15
    jnz next_byte
    inc r13               # new word
    mov r15, 1
    jmp next_byte

newline:
    inc r12               # count line
    # fall through to whitespace handling

whitespace:
    mov r15, 0            # not in word

next_byte:
    dec rcx
    jnz process_byte
    jmp read_loop

process_done:
    # Prepare output: "<lines> <words> <bytes>\n"
    lea rdi, [rip + output_buffer]
    mov r10, rdi          # save start address

    mov rax, r12
    call convert_and_copy
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r13
    call convert_and_copy
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r14
    call convert_and_copy
    mov byte ptr [rdi], 10
    inc rdi

    # Write output
    mov rdx, rdi
    sub rdx, r10          # length = current - start
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    syscall

    # Exit
    mov rax, 60
    xor rdi, rdi
    syscall

# convert_and_copy:
#   Input: rax = number, rdi = output buffer pointer
#   Output: rdi = updated pointer (after copied digits)
#   Clobbers: rax, rcx, rdx, rsi, r8, r9, r10, r11, memory on stack
convert_and_copy:
    sub rsp, 24           # allocate 24 bytes on stack (enough for 20 digits + null)
    mov rcx, rsp
    add rcx, 23
    mov byte ptr [rcx], 0

    test rax, rax
    jnz .convert_loop

    dec rcx
    mov byte ptr [rcx], '0'
    jmp .copy

.convert_loop:
    xor rdx, rdx
    mov r8, 10
    div r8
    dec rcx
    add dl, '0'
    mov byte ptr [rcx], dl
    test rax, rax
    jnz .convert_loop

.copy:
    mov rsi, rcx
    mov rcx, rsp
    add rcx, 23
    sub rcx, rsi          # rcx = length
    rep movsb             # copies from rsi to rdi, updates rdi
    add rsp, 24
    ret
