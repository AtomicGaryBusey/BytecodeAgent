.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 12
input_buffer:
    .zero 49152
final_output:
    .zero 67000

.section .text
.globl _start

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 49152

_start:
    // Read all input into input_buffer
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0                     // total_read

read_loop:
    mov x0, #STDIN
    add x1, x19, x20
    mov x2, #MAX_INPUT
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done               // read_done if 0 bytes read
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    b.ne read_loop

read_done:
    // x20 = input_len
    // Set up for encoding
    adrp x21, final_output
    add x21, x21, :lo12:final_output
    mov x22, #0                     // final_len
    mov x23, #0                     // line_pos
    adrp x24, b64_table
    add x24, x24, :lo12:b64_table
    mov x25, #0                     // input index i

encode_loop:
    cmp x25, x20
    b.ge encode_done
    sub x26, x20, x25               // remaining bytes
    cmp x26, #3
    b.ge encode_3bytes
    cmp x26, #2
    b.eq encode_2bytes
    // else 1 byte
    b encode_1byte

encode_3bytes:
    // Load 3 bytes using base + index
    add x27, x19, x25               // pointer to current byte
    ldrb w27, [x27]                 // b0
    ldrb w28, [x27, #1]             // b1
    ldrb w29, [x27, #2]             // b2
    // Compute 4 base64 chars
    // char0 = b0 >> 2
    ubfx w30, w27, #2, #6
    // char1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and w26, w27, #0x3
    lsl w26, w26, #4
    ubfx w27, w28, #4, #4
    orr w26, w26, w27
    // char2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and w27, w28, #0xF
    lsl w27, w27, #2
    ubfx w28, w29, #6, #2
    orr w27, w27, w28
    // char3 = b2 & 0x3F
    and w28, w29, #0x3F
    // Map to alphabet and store
    ldrb w30, [x24, w30, uxtw #0]
    ldrb w26, [x24, w26, uxtw #0]
    ldrb w27, [x24, w27, uxtw #0]
    ldrb w28, [x24, w28, uxtw #0]
    // Store 4 chars, handling line breaks
    b store_4chars

encode_2bytes:
    add x27, x19, x25
    ldrb w27, [x27]                 // b0
    ldrb w28, [x27, #1]             // b1
    // char0 = b0 >> 2
    ubfx w30, w27, #2, #6
    // char1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and w26, w27, #0x3
    lsl w26, w26, #4
    ubfx w27, w28, #4, #4
    orr w26, w26, w27
    // char2 = (b1 & 0xF) << 2
    and w27, w28, #0xF
    lsl w27, w27, #2
    // char3 = padding '='
    mov w28, #0x3D                  // '='
    // Map first three
    ldrb w30, [x24, w30, uxtw #0]
    ldrb w26, [x24, w26, uxtw #0]
    ldrb w27, [x24, w27, uxtw #0]
    b store_3chars_pad

encode_1byte:
    add x27, x19, x25
    ldrb w27, [x27]                 // b0
    // char0 = b0 >> 2
    ubfx w30, w27, #2, #6
    // char1 = (b0 & 0x3) << 4
    and w26, w27, #0x3
    lsl w26, w26, #4
    // char2 = padding '='
    mov w27, #0x3D
    // char3 = padding '='
    mov w28, #0x3D
    ldrb w30, [x24, w30, uxtw #0]
    ldrb w26, [x24, w26, uxtw #0]
    b store_2chars_pad

store_4chars:
    // Store 4 chars, insert newline if line_pos == 76
    // We'll store one by one, checking line_pos each time
    // char0
    strb w30, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne store_char1
    // insert newline
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
store_char1:
    strb w26, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne store_char2
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
store_char2:
    strb w27, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne store_char3
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
store_char3:
    strb w28, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne after_store_4
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
after_store_4:
    add x25, x25, #3
    b encode_loop

store_3chars_pad:
    // Store 3 chars + padding, with line breaks
    strb w30, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp1
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp1:
    strb w26, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp2
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp2:
    strb w27, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp3
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp3:
    strb w28, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne after_store_3pad
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
after_store_3pad:
    add x25, x25, #2
    b encode_loop

store_2chars_pad:
    strb w30, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp4
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp4:
    strb w26, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp5
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp5:
    strb w27, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne sp6
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
sp6:
    strb w28, [x21, x22]
    add x22, x22, #1
    add x23, x23, #1
    cmp x23, #76
    b.ne after_store_2pad
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1
    mov x23, #0
after_store_2pad:
    add x25, x25, #1
    b encode_loop

encode_done:
    // If line_pos > 0, we need to ensure newline at end? Spec: last line ends with newline.
    // Our loop already adds newline when line_pos reaches 76. But if final line not full, we must add newline.
    // However, spec: "Every line, including the last, ends with one newline 0x0A."
    // So if line_pos != 0, we need to add a newline.
    cbz x23, write_output
    mov w30, #0x0A
    strb w30, [x21, x22]
    add x22, x22, #1

write_output:
    // Write final_output to stdout
    cbz x22, exit_success
    mov x0, #STDOUT
    mov x1, x21
    mov x2, x22
    mov x8, #SYS_WRITE
    svc #0
    // Assume write succeeds fully (since small)

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    // Should not return
    b exit_success
