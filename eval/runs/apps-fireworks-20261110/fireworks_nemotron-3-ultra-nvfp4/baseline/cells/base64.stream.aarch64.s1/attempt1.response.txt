.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 12
input_buffer:
    .zero 49152
final_output:
    .zero 67000

.section .text
.globl _start
.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 49152

_start:
    // Read all input into input_buffer
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0                     // total_read

1:  // read_loop
    mov x0, #STDIN
    add x1, x19, x20
    mov x2, #MAX_INPUT
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, 2f                      // read_done
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    b.ne 1b

2:  // read_done: x20 = input_len
    // Set up for encoding
    adrp x21, final_output
    add x21, x21, :lo12:final_output
    mov x22, #0                     // final_len
    mov x23, #0                     // line_pos
    adrp x24, b64_table
    add x24, x24, :lo12:b64_table
    mov x19, #0                     // input index i

3:  // encode_loop
    cmp x19, x20
    b.ge 4f                         // encode_done
    sub x25, x20, x19               // remaining bytes
    cmp x25, #3
    b.ge 5f                         // encode_3bytes
    cmp x25, #2
    b.eq 6f                         // encode_2bytes

    // encode_1byte
    ldrb w26, [x19]
    ubfx w27, w26, #2, #6
    ldrb w27, [x24, w27]
