.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 12
input_buffer:
    .zero 49152
final_output:
    .zero 67000

.section .text
.globl _start

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 49152

_start:
    // Read all input into input_buffer
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0                     // total_read

read_loop:
    mov x0, #STDIN
    add x1, x19, x20
    mov x2, #MAX_INPUT
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done               // read_done if 0 bytes read
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    b.ne read_loop

read_done:
    // x20 = input_len
    // Set up for encoding
    adrp x21, final_output
    add x21, x21, :lo12:final_output
    mov x22, #0                     // final_len
    mov x23, #0                     // line_pos
    adrp x24, b64_table
    add x24, x24, :lo12:b64_table
    mov x25, #0                     // input index i

encode_loop:
    cmp x25, x20
    b.ge encode_done
    sub x26, x20, x25               // remaining bytes
    cmp x26, #3
    b.ge encode_3bytes
    cmp x26, #2
    b.eq encode_2bytes
    // else 1 byte
    b encode_1byte

encode_3bytes:
    // Load 3 bytes
    ldrb w27, [x19, x25]            // b0
    ldrb w28, [x19, x25, #1]        // b1
