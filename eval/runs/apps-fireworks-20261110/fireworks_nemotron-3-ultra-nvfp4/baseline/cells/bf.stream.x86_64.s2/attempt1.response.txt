.globl _start
.text
_start:
    # Read all stdin into input_buffer
    xor rbx, rbx
    lea r13, [rip + input_buffer]
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, 65536
    jl read_loop
read_done:
    # Scan for '!' (0x21)
    mov rcx, rbx
    xor rdi, rdi
    mov rsi, r13
scan_loop:
    cmp rdi, rcx
    jge no_bang
    cmp byte ptr [rsi + rdi], 0x21
    je found_bang
    inc rdi
    jmp scan_loop
found_bang:
    mov r12, rdi                  # program_len
    lea r14, [rsi + rdi + 1]      # input_base
    mov r15, rcx
    sub r15, rdi
    dec r15                       # input_len
    jmp bracket_match
no_bang:
    mov r12, rcx                  # program_len
    xor r14, r14                  # input_base = 0
    xor r15, r15                  # input_len = 0
bracket_match:
    # r12 = program_len, r13 = program_base, r14 = input_base, r15 = input_len
    lea r8, [rip + jump_table]
    lea r9, [rip + bracket_stack]
    xor r10, r10                  # stack_ptr
    xor rdi, rdi                  # i
bracket_loop:
    cmp rdi, r12
    jge bracket_done
    movzx eax, byte ptr [r13 + rdi]
    cmp al, '['
    je bracket_open
    cmp al, ']'
    je bracket_close
    jmp bracket_next
bracket_open:
    cmp r10, 1000
    jge bracket_error
    mov dword ptr [r9 + r10*4], edi
    inc r10
    jmp bracket_next
bracket_close:
    cmp r10, 0
    je bracket_error
    dec r10
    mov esi, dword ptr [r9 + r10*4]
    mov dword ptr [r8 + rsi*4], edi
    mov dword ptr [r8 + rdi*4], esi
    jmp bracket_next
bracket_next:
    inc rdi
    jmp bracket_loop
bracket_done:
    cmp r10, 0
    jne bracket_error
    jmp execute
bracket_error:
    mov rax, 60
    mov rdi, 1
    syscall
execute:
    # Set up execution registers
    mov rbx, r12                  # program_len
    mov rsi, r13                  # program_base
    mov r10, r14                  # input_base
    mov r12, r15                  # input_len
    mov r15, r8                   # jump_table_base
    lea r8, [rip + tape]          # tape_base
    xor rdi, rdi                  # ip = 0
    xor r9, r9                    # tape_ptr = 0
    xor r11, r11                  # input_index = 0
    lea r13, [rip + output_buffer] # output_base
    xor r14, r14                  # output_index = 0
exec_loop:
    cmp rdi, rbx
    jge exec_done
    movzx eax, byte ptr [rsi + rdi]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_lbracket
    cmp al, ']'
    je cmd_rbracket
    jmp cmd_next
cmd_gt:
    inc r9
    jmp cmd_next
cmd_lt:
    dec r9
    jmp cmd_next
cmd_plus:
    inc byte ptr [r8 + r9]
    jmp cmd_next
cmd_minus:
    dec byte ptr [r8 + r9]
    jmp cmd_next
cmd_dot:
    mov al, byte ptr [r8 + r9]
    mov byte ptr [r13 + r14], al
    inc r14
    jmp cmd_next
cmd_comma:
    cmp r11, r12
    jge comma_exhausted
    mov al, byte ptr [r10 + r11]
    mov byte ptr [r8 + r9], al
    inc r11
    jmp cmd_next
comma_exhausted:
    mov byte ptr [r8 + r9], 0
    jmp cmd_next
cmd_lbracket:
    cmp byte ptr [r8 + r9], 0
    jne cmd_next
    mov edi, dword ptr [r15 + rdi*4]
    jmp cmd_next
cmd_rbracket:
    cmp byte ptr [r8 + r9], 0
    je cmd_next
    mov edi, dword ptr [r15 + rdi*4]
    jmp cmd_next
cmd_next:
    inc rdi
    jmp exec_loop
exec_done:
    # Write output
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, r14
    syscall
    # Exit with status 0
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.align 16
input_buffer: .zero 65536
jump_table: .zero 262144
bracket_stack: .zero 4000
tape: .zero 30000
output_buffer: .zero 65536
