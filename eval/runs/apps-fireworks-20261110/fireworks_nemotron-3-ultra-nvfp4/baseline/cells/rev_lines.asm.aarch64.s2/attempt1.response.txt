.section .bss
.align 12
input_buf:
    .zero 65536
output_buf:
    .zero 65537

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x25, output_buf
    add  x25, x25, :lo12:output_buf

    mov x20, #0              // total_len = 0
    mov x21, #65536          // max_size

read_loop:
    mov x0, #0               // fd = 0
    add x1, x19, x20         // buf = input_buf + total_len
    sub x2, x21, x20         // count = max_size - total_len
    mov x8, #63              // read
    svc #0
    cbz x0, read_done        // EOF
    add x20, x20, x0         // total_len += bytes_read
    cmp x20, x21
    b.eq read_done           // buffer full
    b read_loop

read_done:
    mov x21, #0              // i = 0

line_loop:
    cmp x21, x20
    b.ge exit_program        // i >= total_len -> done

    mov x22, x21             // line_start = i

find_newline:
    cmp x21, x20
    b.ge newline_not_found
    ldrb w24, [x19, x21]     // load byte
    cmp w24, #0x0A
    b.eq newline_found
    add x21, x21, #1
    b find_newline

newline_found:
    sub x23, x21, x22        // line_len = i - line_start
    add x21, x21, #1         // skip newline
    b process_line

newline_not_found:
    sub x23, x21, x22        // line_len = total_len - line_start
    b process_line

process_line:
    mov x26, #0              // output index
    cbz x23, write_line      // empty line -> just write newline

reverse_loop:
    add x27, x22, x23        // line_start + line_len
    sub x27, x27, #1         // -1
    sub x27, x27, x26        // -output_index
    ldrb w24, [x19, x27]     // load from end of line
    strb w24, [x25, x26]     // store to output_buf
    add x26, x26, #1
    cmp x26, x23
    b.lt reverse_loop

write_line:
    mov w24, #0x0A
    strb w24, [x25, x23]     // append newline
    // write(output_buf, line_len+1)
    mov x0, #1               // fd = 1
    mov x1, x25
    add x2, x23, #1
    mov x28, x2              // remaining
    mov x29, x1              // current pointer

write_loop:
    mov x8, #64              // write
    svc #0
    cbz x0, write_error
    cmp x0, #0
    b.le write_error
    subs x28, x28, x0
    beq write_done
    add x29, x29, x0
    mov x1, x29
    mov x2, x28
    b write_loop

write_done:
    b line_loop

write_error:
    mov x0, #1
    mov x8, #93
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_program
