.section .rodata
newline: .byte 0x0A

.section .bss
.align 16
input_buffer: .zero 32768
line_ptrs: .zero 8000
line_lens: .zero 8000

.section .text
.globl _start

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

_start:
    # Read all input
    xor r12, r12                # total_read = 0
    lea r13, [rip + input_buffer]
    mov r14, BUF_SIZE

.read_loop:
    mov rax, SYS_READ
    mov rdi, STDIN
    mov rsi, r13
    mov rdx, r14
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add r13, rax
    sub r14, rax
    jnz .read_loop

.read_done:
    # Parse lines
    xor r15, r15                # nlines = 0
    test r12, r12
    jz .parse_done

    xor rsi, rsi                # current index
    xor rdi, rdi                # line count
    xor r8, r8                  # start of current line

.parse_scan:
    cmp rsi, r12
    jae .parse_end_buffer
    mov al, byte ptr [input_buffer + rsi]
    cmp al, 0x0A
    je .parse_newline
    inc rsi
    jmp .parse_scan

.parse_newline:
    mov r9, rsi
    sub r9, r8                  # length
    lea r10, [rip + input_buffer]
    add r10, r8
    mov qword ptr [line_ptrs + rdi*8], r10
    mov qword ptr [line_lens + rdi*8], r9
    inc rdi
    inc rsi
    mov r8, rsi
    jmp .parse_scan

.parse_end_buffer:
    cmp r8, r12
    jae .parse_done
    mov r9, r12
    sub r9, r8
    lea r10, [rip + input_buffer]
    add r10, r8
    mov qword ptr [line_ptrs + rdi*8], r10
    mov qword ptr [line_lens + rdi*8], r9
    inc rdi

.parse_done:
    mov r15, rdi                # nlines

    # Sort lines with insertion sort
    cmp r15, 1
    jbe .sort_done

    mov rbx, 1                  # i = 1

.outer_loop:
    mov r8, qword ptr [line_ptrs + rbx*8]
    mov r9, qword ptr [line_lens + rbx*8]
    mov rcx, rbx
    dec rcx                     # j = i-1

.inner_loop:
    # Compare line at rcx with saved line (r8, r9)
    mov rsi, qword ptr [line_ptrs + rcx*8]
    mov rdi, qword ptr [line_lens + rcx*8]
    xor r11, r11                # byte index

.cmp_loop:
    cmp r11, rdi
    jae .cmp_len_check1
    cmp r11, r9
    jae .cmp_len_check2
    mov al, byte ptr [rsi + r11]
    mov bl, byte ptr [r8 + r11]
    cmp al, bl
    jne .cmp_diff
    inc r11
    jmp .cmp_loop

.cmp_diff:
    cmp al, bl
    ja .move_line               # line[rcx] > saved
    jb .inner_done              # line[rcx] < saved

.cmp_len_check1:
    # line[rcx] exhausted
    cmp r11, r9
    jae .equal_lines            # both exhausted -> equal
    # line[rcx] shorter -> line[rcx] < saved
    jmp .inner_done

.cmp_len_check2:
    # saved line exhausted, line[rcx] longer -> line[rcx] > saved
    jmp .move_line

.equal_lines:
    jmp .inner_done

.move_line:
    # Move line at rcx to rcx+1
    mov r10, qword ptr [line_ptrs + rcx*8]
    mov r11, qword ptr [line_lens + rcx*8]
    mov qword ptr [line_ptrs + rcx*8 + 8], r10
    mov qword ptr [line_lens + rcx*8 + 8], r11
    dec rcx
    jns .inner_loop
    # rcx < 0, fall through to insert at 0

.inner_done:
    # Insert saved line at rcx+1
    mov qword ptr [line_ptrs + rcx*8 + 8], r8
    mov qword ptr [line_lens + rcx*8 + 8], r9

    inc rbx
    cmp rbx, r15
    jb .outer_loop

.sort_done:
    # Output lines
    xor rbx, rbx

.output_loop:
    cmp rbx, r15
    jae .output_done
    mov rsi, qword ptr [line_ptrs + rbx*8]
    mov rdx, qword ptr [line_lens + rbx*8]
    test rdx, rdx
    jz .write_newline
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    syscall

.write_newline:
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc rbx
    jmp .output_loop

.output_done:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
