.section .rodata
error_msg: .ascii "error\n"
.equ error_len, . - error_msg
min_int_str: .ascii "-9223372036854775808\n"
.equ min_int_len, . - min_int_str

.section .bss
.align 8
stack_array: .zero 8192

.section .text
.globl _start

_start:
    sub rsp, 65536
    mov r12, rsp
    call read_all
    mov r13, rax
    call process_lines
    add rsp, 65536
    mov rax, 60
    xor rdi, rdi
    syscall

read_all:
    push rbx
    push r12
    xor rbx, rbx
    mov r14, 65536
.read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jz .read_done
    js .read_error
    add rbx, rax
    cmp rbx, r14
    jne .read_loop
.read_done:
    mov rax, rbx
    pop r12
    pop rbx
    ret
.read_error:
    mov rax, 60
    mov rdi, 1
    syscall

process_lines:
    push r12
    push r13
    push r14
    push r15
    mov r14, r12
    add r14, r13
.line_loop:
    cmp r12, r14
    jae .done
    mov r15, r12
    xor r13, r13
.scan_loop:
    cmp r15, r14
    jae .no_newline
    cmp byte ptr [r15], 0x0A
    je .found_newline
    inc r15
    inc r13
    jmp .scan_loop
.found_newline:
    mov rdi, r12
    mov rsi, r13
    call eval_line
    lea r12, [r
