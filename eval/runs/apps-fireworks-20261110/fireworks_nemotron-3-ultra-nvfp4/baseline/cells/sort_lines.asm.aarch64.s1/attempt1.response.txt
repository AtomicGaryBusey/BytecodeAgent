.section .text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

_start:
    // Read all input
    bl read_all

    // Parse lines
    bl parse_lines

    // Sort lines
    ldr x0, =line_count
    ldr x0, [x0]
    cbz x0, write_output
    mov x1, #0
    mov x2, x0
    bl merge_sort

write_output:
    bl write_all

    // Exit
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// read_all: reads stdin into input_buf, returns total bytes in x0
read_all:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0          // total read
    mov x21, #MAX_INPUT  // remaining

1:
    mov x0, #0           // fd = stdin
    add x1, x19, x20     // buf + total
    mov x2, x21          // count = remaining
    mov x8, #SYS_READ
    svc #0
    cbz x0, 2f           // EOF
    add x20, x20, x0     // total += nread
    sub x21, x21, x0     // remaining -= nread
    cbnz x21, 1b         // continue if buffer not full

2:
    adrp x1, input_len
    add x1, x1, :lo12:input_len
    str x20, [x1]
    mov x0, x20
    ldp x29, x30, [sp], #16
    ret

// parse_lines: splits input_buf into lines, fills line_ptrs/line_lens/line_count
parse_lines:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x19, input_len
    add x19, x19, :lo12:input_len
    ldr x19, [x19]       // total bytes

    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

    adrp x21, line_ptrs
    add x21, x21, :lo12:line_ptrs

    adrp x22, line_lens
    add x22, x22, :lo12:line_lens

    mov x23, #0          // line count
    mov x24, #0          // current index
    cbz x19, 9f

1:
    mov x25, x24         // start of line

2:
    cmp x24, x19
    b.ge 3f              // reached end
    ldrb w26, [x20, x24]
    cmp w26, #10         // '\n'
    b.eq 4f
    add x24, x24, #1
    b 2b

3:  // end of input without newline
    sub x26, x24, x25    // line length
    cbz x26, 9f          // empty remainder after final newline -> not a line
    add x27, x20, x25    // line pointer
    str x27, [x21, x23, lsl #3]
    str x26, [x22, x23, lsl #3]
    add x23, x23, #1
    b 9f

4:  // found newline
    sub x26, x24, x25    // line length (may be 0)
    add x27, x20, x25
    str x27, [x21, x23, lsl #3]
    str x26, [x22, x23, lsl #3]
    add x23, x23, #1
    add x24, x24, #1     // skip newline
    cmp x24, x19
    b.lt 1b

9:
    adrp x0, line_count
    add x0, x0, :lo12:line_count
    str x23, [x0]
    mov x0, x23
    ldp x29, x30, [sp], #16
    ret

// merge_sort(lo, hi): sorts lines[lo..hi-1]
merge_sort:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]

    mov x19, x0  // lo
    mov x20, x1  // hi
    sub x21, x20, x19
    cmp x21, #1
    b.le 1f

    add x21, x19, x20
    lsr x21, x21, #1  // mid

    mov x0, x19
    mov x1, x21
    bl merge_sort

    mov x0, x21
    mov x1, x20
    bl merge_sort

    mov x0, x19
    mov x1, x21
    mov x2, x20
    bl merge

1:
    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret

// merge(lo, mid, hi): merges two sorted halves using temp arrays
merge:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #80
    stp x19, x20, [sp, #0]
    stp x21, x22, [sp, #16]
    stp x23, x24, [sp, #32]
    stp x25, x26, [sp, #48]
    stp x27, x28, [sp, #64]

    mov x19, x0  // lo
    mov x20, x1  // mid
    mov x21, x2  // hi

    adrp x22, line_ptrs
    add x22, x22, :lo12:line_ptrs
    adrp x23, line_lens
    add x23, x23, :lo12:line_lens
    adrp x24, temp_ptrs
    add x24, x24, :lo12:temp_ptrs
    adrp x25, temp_lens
    add x25, x25, :lo12:temp_lens

    sub x26, x20, x19  // n_left
    cbz x26, 12f

    // Copy left half to temp
    mov x27, #0
2:
    add x28, x19, x27
    ldr x28, [x22, x28, lsl #3]
    str x28, [x24, x27, lsl #3]
    ldr x28, [x23, x19, lsl #3]
    add x28, x19, x27
    ldr x28, [x23, x28, lsl #3]
    str x28, [x25, x27, lsl #3]
    add x27, x27, #1
    cmp x27, x26
    b.lt 2b

    // Merge back
    mov x27, #0        // i (temp index)
    mov x28, x20       // j (right index)
    mov x30, x19       // k (output index)  -- use x30 as k (saved on stack)

3:
    cmp x27, x26
    b.ge 4f            // left exhausted
    cmp x28, x21
    b.ge 5f            // right exhausted

    // Compare temp[i] vs line[j]
    ldr x3, [x24, x27, lsl #3]   // temp ptr
    ldr x4, [x25, x27, lsl #3]   // temp len
    ldr x5, [x22, x28, lsl #3]   // line ptr
    ldr x6, [x23, x28, lsl #3]   // line len

    // Compare bytes
    mov x7, x4
    cmp x7, x6
    csel x7, x7, x6, hi          // x7 = min(len1, len2)
    cbz x7, 6f

    mov x8, #0
7:
    ldrb w9, [x3, x8]
    ldrb w10, [x5, x8]
    cmp w9, w10
    b.ne 8f
    add x8, x8, #1
    cmp x8, x7
    b.lt 7b

6:  // all min_len bytes equal
    sub x11, x4, x6
    b 9f

8:  // bytes differ
    sub w11, w9, w10
    sxtw x11, w11

9:
    cmp x11, #0
    b.le 10f

    // Take from right
    ldr x12, [x22, x28, lsl #3]
    ldr x13, [x23, x28, lsl #3]
    str x12, [x22, x30, lsl #3]
    str x13, [x23, x30, lsl #3]
    add x28, x28, #1
    add x30, x30, #1
    b 3b

10:
    // Take from left (temp)
    ldr x12, [x24, x27, lsl #3]
    ldr x13, [x25, x27, lsl #3]
    str x12, [x22, x30, lsl #3]
    str x13, [x23, x30, lsl #3]
    add x27, x27, #1
    add x30, x30, #1
    b 3b

4:  // left exhausted, right already in place
    b 12f

5:  // right exhausted, copy remaining left
11:
    cmp x27, x26
    b.ge 12f
    ldr x12, [x24, x27, lsl #3]
    ldr x13, [x25, x27, lsl #3]
    str x12, [x22, x30, lsl #3]
    str x13, [x23, x30, lsl #3]
    add x27, x27, #1
    add x30, x30, #1
    b 11b

12:
    ldp x19, x20, [sp, #0]
    ldp x21, x22, [sp, #16]
    ldp x23, x24, [sp, #32]
    ldp x25, x26, [sp, #48]
    ldp x27, x28, [sp, #64]
    add sp, sp, #80
    ldp x29, x30, [sp], #16
    ret

// write_all: writes sorted lines to stdout with newlines
write_all:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x19, line_count
    add x19, x19, :lo12:line_count
    ldr x19, [x19]       // n

    adrp x20, line_ptrs
    add x20, x20, :lo12:line_ptrs
    adrp x21, line_lens
    add x21, x21, :lo12:line_lens

    adrp x22, newline_char
    add x22, x22, :lo12:newline_char

    mov x23, #0
1:
    cmp x23, x19
    b.ge 2f

    ldr x1, [x20, x23, lsl #3]  // ptr
    ldr x2, [x21, x23, lsl #3]  // len
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    mov x0, #1
    mov x1, x22
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0

    add x23, x23, #1
    b 1b

2:
    ldp x29, x30, [sp], #16
    ret

.section .rodata
newline_char:
    .byte 10

.section .bss
.align 3
input_buf:
    .space MAX_INPUT
input_len:
    .space 8
line_ptrs:
    .space MAX_LINES * 8
line_lens:
    .space MAX_LINES * 8
line_count:
    .space 8
temp_ptrs:
    .space MAX_LINES * 8
temp_lens:
    .space MAX_LINES * 8
