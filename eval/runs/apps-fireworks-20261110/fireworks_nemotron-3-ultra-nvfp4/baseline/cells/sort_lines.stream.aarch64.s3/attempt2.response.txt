.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

// ------------------------------------------------------------
// Compare two lines.
// Input:  x0 = ptr1, x1 = len1, x2 = ptr2, x3 = len2
// Output: x0 = -1 (line1 < line2), 0 (equal), 1 (line1 > line2)
// Clobbers: x4-x7
// ------------------------------------------------------------
compare_lines:
    mov     x4, 0                  // index
    mov     x5, x1
    cmp     x5, x3
    csel    x5, x5, x3, hi         // x5 = min(len1, len2)
    cbz     x5, .Lcompare_lengths  // if min_len == 0, skip byte loop

.Lcompare_byte_loop:
    ldrb    w6, [x0, x4]
    ldrb    w7, [x2, x4]
    cmp     w6, w7
    b.ne    .Lcompare_byte_diff
    add     x4, x4, 1
    cmp     x4, x5
    b.lt    .Lcompare_byte_loop

.Lcompare_lengths:
    cmp     x1, x3
    b.lt    .Lcompare_less
    b.gt    .Lcompare_greater
    mov     x0, 0
    ret

.Lcompare_byte_diff:
    // unsigned comparison of the differing bytes
    cmp     w6, w7
    b.lo    .Lcompare_less
    b       .Lcompare_greater

.Lcompare_less:
    mov     x0, -1
    ret

.Lcompare_greater:
    mov     x0, 1
    ret


// ------------------------------------------------------------
// Entry point
// ------------------------------------------------------------
_start:
    // ---- Read all input into input_buf ----
    adrp    x20, input_buf
    add     x20, x20, :lo12:input_buf
    mov     x19, 0                  // total_read = 0
    mov     x21, MAX_INPUT          // remaining

.Lread_loop:
    mov     x0, STDIN
    add     x1, x20, x19
    mov     x2, x21
    mov     x8, SYS_READ
    svc     0
    cbz     x0, .Lread_done
    add     x19, x19, x0            // total_read += ret
    sub     x21, x21, x0            // remaining -= ret
    cbnz    x21, .Lread_loop

.Lread_done:
    // ---- Parse lines ----
    adrp    x21, line_ptrs
    add     x21, x21, :lo12:line_ptrs
    adrp    x22, line_lens
    add     x22, x22, :lo12:line_lens
    mov     x23, 0                  // line_count = 0
    mov     x24, 0                  // current offset in buffer
    cmp     x19, 0
    beq     .Lparse_done

.Lparse_outer:
    cmp     x24, x19
    b.ge    .Lparse_done
    mov     x25, x24                // line start offset

.Lparse_inner:
    cmp     x24, x19
    b.ge    .Lparse_inner_end
    ldrb    w27, [x20, x24]
    add     x24, x24, 1
    cmp     w27, 0x0A
    b.ne    .Lparse_inner
    // found newline
    sub     x26, x24, x25
    sub     x26, x26, 1             // length without newline
    b       .Lparse_store_line

.Lparse_inner_end:
    sub     x26, x24, x25           // length up to end

.Lparse_store_line:
    add     x28, x20, x25           // line start address
    str     x28, [x21, x23, lsl #3]
    str     x26, [x22, x23, lsl #3]
    add     x23, x23, 1
    b       .Lparse_outer

.Lparse_done:
    // x23 = line_count (n)
    // x21 = line_ptrs, x22 = line_lens
    // ---- Prepare for mergesort ----
    mov     x19, x23                // n = line_count
    // load temp buffers
    adrp    x20, temp_ptrs
    add     x20, x20, :lo12:temp_ptrs
    adrp    x21, temp_lens
    add     x21, x21, :lo12:temp_lens
    // Now: x20 = temp_ptrs, x21 = temp_lens
    // We'll use x22 = src_ptrs, x23 = src_lens, x24 = dst_ptrs, x25 = dst_lens
    // Initially src = line_ptrs/line_lens, dst = temp_ptrs/temp_lens
    // But we have line_ptrs in x21? Wait, we overwrote x21 with temp_ptrs.
    // Let's reload line_ptrs and line_lens into x22, x23.
    adrp    x22, line_ptrs
    add     x22, x22, :lo12:line_ptrs
    adrp    x23, line_lens
    add     x23, x23, :lo12:line_lens
    // x24 = temp_ptrs (x20), x25 = temp_lens (x21)
    mov     x24, x20
    mov     x25, x21

    // width in x26
    mov     x26, 1

.Lwidth_loop:
    cmp     x26, x19
    b.ge    .Lwidth_done

    // i in x27
    mov     x27, 0

.Li_loop:
    cmp     x27, x19
    b.ge    .Li_loop_done

    // left = x27, mid = left + width
    add     x28, x27, x26           // mid
    cmp     x28, x19
    b.ge    .Lcopy_remaining        // only one run left, copy it

    // right = min(mid + width, n)
    add     x29, x28, x26           // mid + width
    cmp     x29, x19
    csel    x29, x29, x19, hi       // right

    // Merge runs [x27, x28) and [x28, x29) from src (x22,x23) to dst (x24,x25) at position x27.
    // We'll use:
    //   x0 = i (left index), x1 = j (mid index), x2 = k (dest index)
    mov     x0, x27                 // i
    mov     x1, x28                 // j
    mov     x2, x27                 // k

.Lmerge_loop:
    cmp     x0, x28
    b.ge    .Lmerge_i_done
    cmp     x1, x29
    b.ge    .Lmerge_j_done

    // Load line i
    ldr     x3, [x22, x0, lsl #3]
    ldr     x4, [x23, x0, lsl #3]
    // Load line j
    ldr     x5, [x22, x1, lsl #3]
    ldr     x6, [x23, x1, lsl #3]
    // Compare
    mov     x7, x0                  // save i
    mov     x8, x1                  // save j
    mov     x9, x2                  // save k
    bl      compare_lines
    // x0 = result
    mov     x0, x7                  // restore i
    mov     x1, x8                  // restore j
    mov     x2, x9                  // restore k
    cmp     x0, 0
    b.le    .Ltake_i                // <= 0 means line_i <= line_j

    // Take j
    str     x5, [x24, x2, lsl #3]
    str     x6, [x25, x2, lsl #3]
    add     x1, x1, 1
    b       .Lmerge_next

.Ltake_i:
    str     x3, [x24, x2, lsl #3]
    str     x4, [x25, x2, lsl #3]
    add     x0, x0, 1

.Lmerge_next:
    add     x2, x2, 1
    b       .Lmerge_loop

.Lmerge_i_done:
    // Copy remaining j
    cmp     x1, x29
    b.ge    .Lmerge_done
.Lcopy_j_loop:
    ldr     x3, [x22, x1, lsl #3]
    ldr     x4, [x23, x1, lsl #3]
    str     x3, [x24, x2, lsl #3]
    str     x4, [x25, x2, lsl #3]
    add     x1, x1, 1
    add     x2, x2, 1
    cmp     x1, x29
    b.lt    .Lcopy_j_loop
    b       .Lmerge_done

.Lmerge_j_done:
    // Copy remaining i
    cmp     x0, x28
    b.ge    .Lmerge_done
.Lcopy_i_loop:
    ldr     x3, [x22, x0, lsl #3]
    ldr     x4, [x23, x0, lsl #3]
    str     x3, [x24, x2, lsl #3]
    str     x4, [x25, x2, lsl #3]
    add     x0, x0, 1
    add     x2, x2, 1
    cmp     x0, x28
    b.lt    .Lcopy_i_loop

.Lmerge_done:
    // Next pair
    mov     x27, x29
    b       .Li_loop

.Lcopy_remaining:
    // Copy the rest of the run from src to dst
    // x27 is start, x19 is n
.Lcopy_rem_loop:
    cmp     x27, x19
    b.ge    .Li_loop_done
    ldr     x3, [x22, x27, lsl #3]
    ldr     x4, [x23, x27, lsl #3]
    str     x3, [x24, x27, lsl #3]
    str     x4, [x25, x27, lsl #3]
    add     x27, x27, 1
    b       .Lcopy_rem_loop

.Li_loop_done:
    // Swap src and dst for next pass
    // x22 <-> x24, x23 <-> x25
    mov     x0, x22
    mov     x22, x24
    mov     x24, x0
    mov     x0, x23
    mov     x23, x25
    mov     x25, x0

    // Double width
    add     x26, x26, x26
    b       .Lwidth_loop

.Lwidth_done:
    // Sorted data is now in src (x22, x23)
    // ---- Write output ----
    mov     x27, 0                  // index

.Loutput_loop:
    cmp     x27, x19
    b.ge    .Loutput_done
    ldr     x0, [x22, x27, lsl #3]  // ptr
    ldr     x1, [x23, x27, lsl #3]  // len
    // Write line
    mov     x2, x1
    mov     x0, STDOUT
    mov     x8, SYS_WRITE
    svc     0
    // Write newline
    adr     x1, newline_char
    mov     x2, 1
    mov     x0, STDOUT
    mov     x8, SYS_WRITE
    svc     0
    add     x27, x27, 1
    b       .Loutput_loop

.Loutput_done:
    mov     x0, 0
    mov     x8, SYS_EXIT
    svc     0


// ------------------------------------------------------------
// Data sections
// ------------------------------------------------------------
.section .rodata
newline_char:
    .byte   0x0A

.section .bss
.align 12
input_buf:
    .space  MAX_INPUT
line_ptrs:
    .space  MAX_LINES * 8
line_lens:
    .space  MAX_LINES * 8
temp_ptrs:
    .space  MAX_LINES * 8
temp_lens:
    .space  MAX_LINES * 8
