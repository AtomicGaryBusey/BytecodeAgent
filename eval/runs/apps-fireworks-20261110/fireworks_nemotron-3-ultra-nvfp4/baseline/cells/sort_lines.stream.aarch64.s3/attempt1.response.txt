.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

_start:
    // ---- Read all input into input_buf ----
    adrp x20, input_buf
    add  x20, x20, :lo12:input_buf
    mov  x19, 0                  // total_read = 0
    mov  x21, MAX_INPUT          // remaining

read_loop:
    mov  x0, STDIN
    add  x1, x20, x19
    mov  x2, x21
    mov  x8, SYS_READ
    svc  0
    cbz  x0, read_done
    add  x19, x19, x0            // total_read += ret
    sub  x21, x21, x0            // remaining -= ret
    cbnz x21, read_loop

read_done:
    // ---- Parse lines ----
    adrp x21, line_ptrs
    add  x21, x21, :lo12:line_ptrs
    adrp x22, line_lens
    add  x22, x22, :lo12:line_lens
    mov  x23, 0                  // line_count = 0
    mov  x24, 0                  // current offset in buffer
    cmp  x19, 0
    beq  parse_done

parse_outer:
    cmp  x24, x19
    b.ge parse_done
    mov  x25, x24                // line start offset

parse_inner:
    cmp  x24, x19
    b.ge parse_inner_end
    ldrb w27, [x20, x24]
    add  x24, x24, 1
    cmp  w27, 0x0A
    b.ne parse_inner
    // found newline
    sub  x26, x24, x25
    sub  x26, x26, 1             // length without newline
    b    parse_store_line

parse_inner_end:
    sub  x26, x24, x25           // length up to end

parse_store_line:
    add  x28, x20, x25           // line start address
    str  x28, [x21, x23, lsl #3]
    str  x26, [x22, x23, lsl #3]
    add  x23, x23, 1
    b    parse_outer

parse_done:
    // x23 = line_count (n)
    // x21 = line_ptrs, x22 = line_lens
    // ---- Prepare for mergesort ----
    mov  x19, x23                // n = line_count
    // load temp buffers
    adrp x20, temp_ptrs
    add  x20, x20, :lo12:temp_ptrs
    adrp x21, temp_lens
    add  x21, x21, :lo12:temp_lens
    mov  x22, x21? Wait x21 is temp_lens, but we need line_ptrs in x22, line_lens in x23.
