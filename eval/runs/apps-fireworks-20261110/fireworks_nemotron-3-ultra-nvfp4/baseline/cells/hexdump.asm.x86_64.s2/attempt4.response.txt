.globl _start

.text
_start:
    xor r12, r12                    # total_bytes = 0

.read_loop:
    mov rdx, 65536
    sub rdx, r12                    # remaining space
    jz .read_done
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdi, 0                      # fd = 0 (stdin)
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz .read_done
    add r12, rax                    # total_bytes += bytes_read
    jmp .read_loop

.read_done:
    mov [rip + total_bytes], r12    # store total_bytes

    test r12, r12
    jz .print_total                 # empty input

    mov r13, r12                    # total_bytes
    xor r14, r14                    # offset = 0

.chunk_loop:
    mov rax, r13
    sub rax, r14                    # remaining bytes
    cmp rax, 16
    jbe .last_chunk
    mov r15, 16                     # n = 16
    jmp .format_write
.last_chunk:
    mov r15, rax                    # n = remaining (1..15)
.format_write:
    mov rdi, r14                    # offset
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov rdx, r15                    # n
    call format_line

    # write formatted line to stdout
    mov rdi, 1                      # fd = 1 (stdout)
    lea rsi, [rip + line_buffer]
    mov rdx, rax                    # line length
    mov rax, 1                      # sys_write
    syscall

    add r14, r15                    # advance offset
    cmp r14, r13
    jb .chunk_loop

.print_total:
    # print total_bytes as 8 hex digits + newline
    mov rax, [rip + total_bytes]
    lea rdi, [rip + line_buffer]    # destination buffer
    lea rsi, [rip + hex_table]
    mov rcx, 8
.print_total_loop:
    mov rbx, rax
    shr rbx, 28
    and rbx, 0xF
    mov bl, byte ptr [rsi + rbx]
    mov byte ptr [rdi], bl
    inc rdi
    shl rax, 4
    dec rcx
    jnz .print_total_loop
    mov byte ptr [rdi], 0x0A        # newline
    inc rdi
    # write total line
    lea rsi, [rip + line_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    # exit
    mov rax, 60                     # sys_exit
    xor rdi, rdi
    syscall

# format_line subroutine
# Input: rdi = offset, rsi = src pointer, rdx = n (1..16)
# Output: rax = line length, line stored in line_buffer
format_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    lea r11, [rip + line_buffer]   # start of line buffer
    mov r12, r11                   # current write pointer
    mov r13, rsi                   # src
    mov r14, rdx                   # n
    mov r15, rdi                   # offset

    # Write offset as 8 hex digits
    mov rax, r15
    mov rcx, 8
.write_offset_loop:
    mov rbx, rax
    shr rbx, 28
    and rbx, 0xF
    mov bl, byte ptr [rip + hex_table + rbx]
    mov byte ptr [r12], bl
    inc r12
    shl rax, 4
    dec rcx
    jnz .write_offset_loop

    # Two spaces after offset
    mov word ptr [r12], 0x2020
    add r12, 2

    # Hex bytes for 16 positions
    xor rbx, rbx
.hex_byte_loop:
    cmp rbx, 16
    jae .hex_done
    cmp rbx, r14
    jb .print_hex_byte
    # print three spaces for missing bytes
    mov byte ptr [r12], 0x20
    mov byte ptr [r12+1], 0x20
    mov byte ptr [r12+2], 0x20
    add r12, 3
    jmp .check_extra_space
.print_hex_byte:
    mov al, byte ptr [r13 + rbx]
    # high nibble
    movzx eax, al
    shr eax, 4
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r12], al
    # low nibble
    movzx eax, byte ptr [r13 + rbx]
    and al, 0xF
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r12+1], al
    # space after byte
    mov byte ptr [r12+2], 0x20
    add r12, 3
.check_extra_space:
    cmp rbx, 7
    jne .no_extra
    mov byte ptr [r12], 0x20
    inc r12
.no_extra:
    inc rbx
    jmp .hex_byte_loop
.hex_done:

    # Space before '|'
    mov byte ptr [r12], 0x20
    inc r12
    mov byte ptr [r12], 0x7C       # '|'
    inc r12

    # ASCII column
    xor rbx, rbx
.ascii_loop:
    cmp rbx, r14
    jae .ascii_done
    mov al, byte ptr [r13 + rbx]
    cmp al, 0x20
    jb .ascii_dot
    cmp al, 0x7E
    ja .ascii_dot
    mov byte ptr [r12], al
    inc r12
    jmp .ascii_next
.ascii_dot:
    mov byte ptr [r12], 0x2E       # '.'
    inc r12
.ascii_next:
    inc rbx
    jmp .ascii_loop
.ascii_done:
    mov byte ptr [r12], 0x7C       # '|'
    inc r12
    mov byte ptr [r12], 0x0A       # '\n'
    inc r12

    # Compute line length
    mov rax, r12
    sub rax, r11

    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.section .rodata
hex_table: .ascii "0123456789abcdef"

.section .bss
.align 16
total_bytes: .zero 8
input_buffer: .zero 65536
line_buffer: .zero 256
