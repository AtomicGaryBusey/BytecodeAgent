.section .bss
.align 12
buffer:     .space 65536
conv_buf:   .space 24
out_buf:    .space 64

.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // Initialize counters
    mov x19, #0          // lines
    mov x20, #0          // words
    mov x21, #0          // bytes
    mov x22, #0          // in_word flag (0 = not in word, 1 = in word)

    // Buffer address
    adrp x26, buffer
    add x26, x26, :lo12:buffer

    // Output buffer address
    adrp x27, out_buf
    add x27, x27, :lo12:out_buf
    mov x1, x27          // current output pointer (x1 used by write_number)

.Lread_loop:
    mov x0, #0           // stdin
    ldr x2, =65536       // max read size
    mov x8, #SYS_READ
    svc #0

    cbz x0, .Lread_done  // EOF

    add x21, x21, x0     // byte count

    mov x23, x26         // start of data
    add x24, x26, x0     // end of data

.Lprocess_loop:
    cmp x23, x24
    b.eq .Lread_loop

    ldrb w25, [x23], #1  // load byte and advance pointer

    // Check for newline (0x0A)
    cmp w25, #0x0A
    b.ne .Lcheck_ws
    add x19, x19, #1     // increment line count

.Lcheck_ws:
    // Check for whitespace: space(0x20), tab(0x09), newline(0x0A), vt(0x0B), ff(0x0C), cr(0x0D)
    cmp w25, #0x20
    b.eq .Lwhitespace
    cmp w25, #0x09
    b.eq .Lwhitespace
    cmp w25, #0x0A
    b.eq .Lwhitespace
    cmp w25, #0x0B
    b.eq .Lwhitespace
    cmp w25, #0x0C
    b.eq .Lwhitespace
    cmp w25, #0x0D
    b.eq .Lwhitespace

    // Non-whitespace byte
    cbz x22, .Lstart_word
    b .Lprocess_loop

.Lstart_word:
    mov x22, #1
    add x20, x20, #1     // increment word count
    b .Lprocess_loop

.Lwhitespace:
    cbnz x22, .Lend_word
    b .Lprocess_loop

.Lend_word:
    mov x22, #0
    b .Lprocess_loop

.Lread_done:
    // Build output string: lines words bytes\n
    // x1 points to start of output buffer (x27)
    mov x0, x19
    bl write_number

    mov w2, #0x20        // space
    strb w2, [x1], #1

    mov x0, x20
    bl write_number

    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x21
    bl write_number

    mov w2, #0x0A        // newline
    strb w2, [x1], #1

    // Write output
    sub x2, x1, x27      // length
    mov x0, #1           // stdout
    mov x1, x27
    mov x8, #SYS_WRITE
    svc #0

    // Exit
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// write_number: converts x0 (64-bit unsigned) to decimal ASCII, writes to buffer at x1, updates x1.
// Uses conv_buf (24 bytes) as temporary storage for digits (max 20 digits for 64-bit).
// Registers used: x0 (value), x1 (output ptr), x2, x3, x4, x5, x6, x7.
write_number:
    adrp x5, conv_buf
    add x5, x5, :lo12:conv_buf
    add x5, x5, #23          // point to last byte of conv_buf
    mov x4, #0               // digit count

    cbz x0, .Lwrite_zero

.Lwrite_loop:
    mov x2, #10
    udiv x3, x0, x2
    msub x6, x3, x2, x0      // x6 = x0 % 10
    add x6, x6, #0x30        // '0'
    strb w6, [x5], #-1       // store digit backwards
    add x4, x4, #1
    mov x0, x3
    cbnz x0, .Lwrite_loop

    add x5, x5, #1           // point to first digit
    cbz x4, .Lwrite_done     // should not happen

.Lcopy_loop:
    ldrb w6, [x5], #1
    strb w6, [x1], #1
    subs x4, x4, #1
    b.ne .Lcopy_loop

.Lwrite_done:
    ret

.Lwrite_zero:
    mov w6, #0x30
    strb w6, [x1], #1
    ret
