.section .rodata
newline: .byte 10

.section .bss
.align 4096
input_buffer: .zero 65536

.section .text
.globl _start
_start:
    # Read all input into input_buffer (max 65536 bytes)
    mov r13, input_buffer          # buffer base
    xor r12, r12                   # total_read = 0
    mov r14, 65536                 # remaining = 65536

.read_loop:
    test r14, r14
    jz .read_done
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd = stdin
    lea rsi, [r13 + r12]           # buf = buffer + total_read
    mov rdx, r14                   # count = remaining
    syscall
    test rax, rax
    jz .read_done
    add r12, rax                   # total_read += bytes read
    sub r14, rax                   # remaining -= bytes read
    jmp .read_loop

.read_done:
    test r12, r12
    jz .exit                       # empty input -> nothing to do

    mov r14, r13
    add r14, r12                   # r14 = buffer_end (exclusive)
    mov r15, r13                   # r15 = current line start

.process_loop:
    cmp r15, r14
    jae .exit                      # reached end of buffer

    mov rsi, r15                   # scan for newline

.scan_loop:
    cmp rsi, r14
    jae .no_newline                # no newline found in remainder
    mov al, byte ptr [rsi]
    cmp al, 10
    je .found_newline
    inc rsi
    jmp .scan_loop

.found_newline:
    mov rbx, rsi                   # save newline pointer (rbx)
    # reverse line in place: [r15, rbx-1]
    mov rsi, r15
    mov rdi, rbx
    dec rdi                        # rdi = last byte of line

.rev_loop:
    cmp rsi, rdi
    jae .rev_done
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rdi]
    mov byte ptr [rsi], bl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp .rev_loop

.rev_done:
    # write reversed line + newline (from r15 to rbx inclusive)
    mov rax, 1                     # sys_write
    mov rdi, 1                     # fd = stdout
    mov rsi, r15
    mov rdx, rbx
    sub rdx, r15
    inc rdx                        # length = rbx - r15 + 1
    syscall

    # advance to next line
    mov r15, rbx
    inc r15
    jmp .process_loop

.no_newline:
    # last line without terminating newline: reverse [r15, r14-1]
    mov rsi, r15
    mov rdi, r14
    dec rdi

.rev_loop2:
    cmp rsi, rdi
    jae .rev_done2
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rdi]
    mov byte ptr [rsi], bl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp .rev_loop2

.rev_done2:
    # write reversed line
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, r14
    sub rdx, r15                   # length = r14 - r15
    syscall

    # write final newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

.exit:
    mov rax, 60                    # sys_exit
    xor rdi, rdi                   # status = 0
    syscall
