.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94

.section .text
.globl _start

_start:
    // Read all stdin into input_buf (max 65536 bytes)
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0                  // total_read
    mov x21, #65536              // max_read

read_loop:
    sub x2, x21, x20             // remaining space
    cbz x2, read_done
    mov x0, #0                   // fd = 0
    add x1, x19, x20             // buf = input_buf + total_read
    bl sys_read
    cbz x0, read_done            // EOF
    add x20, x20, x0             // total_read += nread
    b read_loop

read_done:
    // x20 = total input length
    // Find first '!' (0x21)
    mov x22, #0                  // index
    mov x23, #0                  // found flag (0 = not found)
find_loop:
    cmp x22, x20
    b.ge find_done               // reached end without '!'
    ldrb w10, [x19, x22]
    cmp w10, #0x21
    b.eq found_bang
    add x22, x22, #1
    b find_loop

found_bang:
    // x22 = index of '!'
    mov x21, x22                 // program_len
    add x24, x19, x22            // address of '!'
    add x24, x24, #1             // program_input_start
    sub x23, x20, x22            // total - index
    sub x23, x23, #1             // program_input_len
    b validate_brackets

find_done:
    // no '!' found
    mov x21, x20                 // program_len = total_read
    mov x24, xzr                 // program_input_start = 0
    mov x23, #0                  // program_input_len = 0

validate_brackets:
    // bracket_stack in .bss
    adrp x10, bracket_stack
    add x10, x10, :lo12:bracket_stack
    mov x11, #0                  // stack pointer (count)
    // bracket_match array
    adrp x12, bracket_match
    add x12, x12, :lo12:bracket_match
    mov x13, #0                  // i

validate_loop:
    cmp x13, x21
    b.ge validate_done
    ldrb w14, [x19, x13]
    cmp w14, #'['
    b.eq push_bracket
    cmp w14, #']'
    b.eq pop_bracket
    add x13, x13, #1
    b validate_loop

push_bracket:
    str w13, [x10, x11, lsl #2]
    add x11, x11, #1
    add x13, x13, #1
    b validate_loop

pop_bracket:
    cbz x11, bracket_error
    sub x11, x11, #1
    ldr w15, [x10, x11, lsl #2]   // matching [
    str w13, [x12, x15, lsl #2]   // bracket_match[matching [] = ]
    str w15, [x12, x13, lsl #2]   // bracket_match[] = matching [
    add x13, x13, #1
    b validate_loop

validate_done:
    cbnz x11, bracket_error
    b execute

bracket_error:
    mov x0, #1
    bl sys_exit

execute:
    // tape base
    adrp x14, tape
    add x14, x14, :lo12:tape
    mov x15, #0                  // tape_ptr
    mov x16, #0                  // ip
    mov x17, #0                  // input_idx
    mov x18, #0                  // output_idx
    // output_buf base
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    // x19 = input_buf (program + input)
    // x21 = program_len
    // x23 = program_input_len
    // x24 = program_input_start
    // x12 = bracket_match base
    mov x10, #0                  // command counter

execute_loop:
    // optional command limit (5,000,000)
    cmp x10, #5000000
