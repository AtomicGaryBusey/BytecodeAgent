.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 49152
.equ LINE_LEN, 76

_start:
    // Read all input into input_buffer
    adr     x20, input_buffer
    mov     x21, #MAX_INPUT
    mov     x19, #0                 // total_read

read_loop:
    mov     x8, #SYS_READ
    mov     x0, #0                  // stdin
    add     x1, x20, x19            // buf = input_buffer + total_read
    sub     x2, x21, x19            // count = MAX_INPUT - total_read
    svc     #0
    cbz     x0, read_done           // EOF
    add     x19, x19, x0            // total_read += bytes_read
    cmp     x19, x21
    b.ne    read_loop               // continue if buffer not full

read_done:
    // Prepare for encoding
    adr     x22, output_buffer
    mov     x23, #0                 // output_index
    mov     x24, #0                 // line_pos
    mov     x25, x20                // input_ptr
    mov     x26, x19                // input_remaining
    adr     x27, b64_table          // base64 alphabet table

    // Main encoding loop: process 3 bytes at a time
encode_loop:
    cmp     x26, #3
    b.lt    encode_remainder

    // Load 3 bytes
    ldrb    w0, [x25], #1
    ldrb    w1, [x25], #1
    ldrb    w2, [x25], #1
    sub     x26, x26, #3

    // Compute 4 sextets
    ubfx    w3, w0, #2, #6          // s0 = w0 >> 2
    and     w4, w0, #0x3
    lsl     w4, w4, #4
    ubfx    w5, w1, #4, #4          // w1 >> 4
    orr     w4, w4, w5              // s1
    and     w5, w1, #0xF
    lsl     w5, w5, #2
    ubfx    w6, w2, #6, #2          // w2 >> 6
    orr     w5, w5, w6              // s2
    and     w6, w2, #0x3F           // s3

    // Output s0
    ldrb    w7, [x27, x3]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    1f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
1:
    // Output s1
    ldrb    w7, [x27, x4]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    2f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
2:
    // Output s2
    ldrb    w7, [x27, x5]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    3f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
3:
    // Output s3
    ldrb    w7, [x27, x6]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    4f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
4:
    b       encode_loop

// Handle remaining 1 or 2 bytes
encode_remainder:
    cbz     x26, encode_done
    cmp     x26, #1
    b.eq    remainder_1

    // Two bytes remaining
    ldrb    w0, [x25], #1
    ldrb    w1, [x25], #1

    ubfx    w3, w0, #2, #6          // s0
    and     w4, w0, #0x3
    lsl     w4, w4, #4
    ubfx    w5, w1, #4, #4
    orr     w4, w4, w5              // s1
    and     w5, w1, #0xF
    lsl     w5, w5, #2              // s2

    // Output s0
    ldrb    w7, [x27, x3]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    5f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
5:
    // Output s1
    ldrb    w7, [x27, x4]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    6f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
6:
    // Output s2
    ldrb    w7, [x27, x5]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    7f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
7:
    // Output padding '='
    mov     w7, #0x3D
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    encode_done
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
    b       encode_done

remainder_1:
    // One byte remaining
    ldrb    w0, [x25]

    ubfx    w3, w0, #2, #6          // s0
    and     w4, w0, #0x3
    lsl     w4, w4, #4              // s1

    // Output s0
    ldrb    w7, [x27, x3]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    8f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
8:
    // Output s1
    ldrb    w7, [x27, x4]
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    9f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
9:
    // Output padding '=='
    mov     w7, #0x3D
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    10f
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0
10:
    strb    w7, [x22, x23]
    add     x23, x23, #1
    add     x24, x24, #1
    cmp     x24, #LINE_LEN
    b.ne    encode_done
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1
    mov     x24, #0

encode_done:
    // If line_pos != 0, output final newline
    cbz     x24, write_output
    mov     w7, #0x0A
    strb    w7, [x22, x23]
    add     x23, x23, #1

write_output:
    // Write output_buffer to stdout
    mov     x8, #SYS_WRITE
    mov     x0, #1                  // stdout
    mov     x1, x22                 // output_buffer
    mov     x2, x23                 // length
    svc     #0

    // Exit
    mov     x8, #SYS_EXIT
    mov     x0, #0
    svc     #0

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buffer:
    .space MAX_INPUT
output_buffer:
    .space 66400
