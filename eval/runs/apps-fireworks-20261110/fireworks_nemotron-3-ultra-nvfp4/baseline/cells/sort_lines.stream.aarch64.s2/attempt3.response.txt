.globl _start

.equ STDIN_FD, 0
.equ STDOUT_FD, 1
.equ STDERR_FD, 2

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94

.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ LINE_DESC_SIZE, 16   // 8 bytes pointer + 8 bytes length

.section .bss
.align 12
input_buffer:
    .space MAX_INPUT
line_descriptors:
    .space MAX_LINES * LINE_DESC_SIZE
line_count:
    .space 8

.section .text
.align 4

// _start: entry point
_start:
    // Read all input into input_buffer
    bl read_all_input
    // x0 = total bytes read
    cbz x0, exit_success   // empty input -> nothing to do

    // Parse lines
    mov x1, x0             // x1 = total bytes
    bl parse_lines
    // x0 = number of lines
    cbz x0, exit_success   // no lines

    // Sort lines
    bl sort_lines

    // Write sorted lines
    bl write_sorted_lines

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// read_all_input: reads from stdin until EOF into input_buffer
// Returns total bytes read in x0
read_all_input:
    // Use x19 as buffer pointer, x20 as total read, x21 as remaining capacity
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #MAX_INPUT

read_loop:
    // Check if buffer full
    cmp x21, #0
    beq read_done

    // read(fd=0, buf=x19, count=x21)
    mov x0, #STDIN_FD
    mov x1, x19
    mov x2, x21
    mov x8, #SYS_READ
    svc #0

    // x0 = bytes read (or -errno)
    cbz x0, read_done      // EOF
    // Check for error (negative)
    cmp x0, #0
    blt read_error

    // Update pointers
    add x19, x19, x0
    add x20, x20, x0
    sub x21, x21, x0
    b read_loop

read_done:
    mov x0, x20
    ret

read_error:
    // Exit with error status 1
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

// parse_lines: parse input_buffer into line_descriptors
// x1 = total bytes in buffer
// Returns number of lines in x0
parse_lines:
    // x19 = current pointer in buffer
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    // x20 = end of buffer
    add x20, x19, x1
    // x21 = line_descriptors array pointer
    adrp x21, line_descriptors
    add x21, x21, :lo12:line_descriptors
    // x22 = line count
    mov x22, #0
    // x23 = start of current line
    mov x23, x19

parse_loop:
    cmp x19, x20
    b.ge parse_check_last_line

    // Load byte
    ldrb w24, [x19]
    cmp w24, #0x0A   // '\n'
    b.ne parse_next_char

    // Found newline: line from x23 to x19 (exclusive)
    // Length = x19 - x23
    sub x24, x19, x23
    // Store descriptor: pointer = x23, length = x24
    str x23, [x21]
    str x24, [x21, #8]
    add x21, x21, #LINE_DESC_SIZE
    add x22, x22, #1

    // Next line starts after newline
    add x23, x19, #1

parse_next_char:
    add x19, x19, #1
    b parse_loop

parse_check_last_line:
    // If buffer doesn't end with newline, last line from x23 to x20
    cmp x23, x20
    b.ge parse_done
    sub x24, x20, x23
    str x23, [x21]
    str x24, [x21, #8]
    add x22, x22, #1

parse_done:
    // Store line count
    adrp x25, line_count
    add x25, x25, :lo12:line_count
    str x22, [x25]
    mov x0, x22
    ret

// sort_lines: sort line_descriptors array using quicksort
// Uses line_count from memory
sort_lines:
    // Load line count
    adrp x19, line_count
    add x19, x19, :lo12:line_count
    ldr x20, [x19]   // n
    cbz x20, sort_done
    // Array pointer
    adrp x21, line_descriptors
    add x21, x21, :lo12:line_descriptors
    // Call quicksort(x21, 0, n-1)
    mov x0, x21
    mov x1, #0
    sub x2, x20, #1
    bl quicksort
sort_done:
    ret

// quicksort: sorts descriptors in [x0] from index x1 to x2 inclusive
// x0 = base pointer, x1 = lo, x2 = hi
quicksort:
    // Stack frame: save x19-x22, x30
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!

    mov x19, x0   // base
    mov x20, x1   // lo
    mov x21, x2   // hi

    // if lo >= hi return
    cmp x20, x21
    b.ge quicksort_return

    // Partition
    mov x0, x19
    mov x1, x20
    mov x2, x21
    bl partition
    // x0 = pivot index

    // Recurse on left: lo to pivot-1
    mov x22, x0   // save pivot
    mov x0, x19
    mov x1, x20
    sub x2, x22, #1
    bl quicksort

    // Recurse on right: pivot+1 to hi
    mov x0, x19
    mov x1, x22
    add x1, x1, #1
    mov x2, x21
    bl quicksort

quicksort_return:
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

// partition: Lomuto partition scheme
// x0 = base, x1 = lo, x2 = hi
// Returns pivot index in x0
partition:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x19, x0   // base
    mov x20, x1   // lo
    mov x21, x2   // hi

    // Choose pivot as hi element
    // pivot descriptor at index hi
    lsl x22, x21, #4   // hi * 16
    add x22, x19, x22
    ldr x23, [x22]     // pivot pointer
    ldr x24, [x22, #8] // pivot length

    // i = lo - 1
    sub x22, x20, #1   // x22 = i

    // Loop j from lo to hi-1
    mov x25, x20       // j = lo
partition_loop:
    cmp x25, x21
    b.ge partition_done_loop

    // Compare element j with pivot
    lsl x26, x25, #4
    add x26, x19, x26
    ldr x27, [x26]     // element pointer
    ldr x28, [x26, #8] // element length

    // Compare lines: memcmp(element, pivot)
    mov x0, x27
    mov x1, x23
    mov x2, x28
    mov x3, x24
    bl memcmp_lines
    // x0 = result: <0, 0, >0

    cmp x0, #0
    b.ge partition_no_swap

    // Swap element j with element i+1
    add x22, x22, #1   // i++
    // Swap descriptors at indices i and j
    // If i != j, swap
    cmp x22, x25
    b.eq partition_no_swap

    // Swap using caller-saved registers x9, x10 for j's descriptor
    // Reload descriptor at i
    lsl x26, x22, #4
    add x26, x19, x26
    ldr x27, [x26]
    ldr x28, [x26, #8]
    // Reload descriptor at j
    lsl x29, x25, #4
    add x29, x19, x29
    ldr x9, [x29]
    ldr x10, [x29, #8]
    // Store j's descriptor at i
    str x9, [x26]
    str x10, [x26, #8]
    // Store i's descriptor at j
    str x27, [x29]
    str x28, [x29, #8]

partition_no_swap:
    add x25, x25, #1
    b partition_loop

partition_done_loop:
    // Swap pivot (hi) with i+1
    add x22, x22, #1   // final pivot position
    // If pivot position != hi, swap
    cmp x22, x21
    b.eq partition_return

    // Reload descriptor at pivot position (i+1)
    lsl x26, x22, #4
    add x26, x19, x26
    ldr x27, [x26]
    ldr x28, [x26, #8]
    // Reload pivot descriptor (hi)
    lsl x29, x21, #4
    add x29, x19, x29
    ldr x9, [x29]
    ldr x10, [x29, #8]
    // Swap
    str x9, [x26]
    str x10, [x26, #8]
    str x27, [x29]
    str x28, [x29, #8]

partition_return:
    mov x0, x22
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

// memcmp_lines: compare two lines
// x0 = ptr1, x1 = ptr2, x2 = len1, x3 = len2
// Returns: x0 <0, 0, >0 (like memcmp)
memcmp_lines:
    // Use x4 as min_len, x5 as index
    cmp x2, x3
    csel x4, x2, x3, lt   // min_len = min(len1, len2)
    mov x5, #0
    cbz x4, memcmp_len_check

memcmp_loop:
    ldrb w6, [x0, x5]
    ldrb w7, [x1, x5]
    cmp w6, w7
    b.ne memcmp_diff
    add x5, x5, #1
    cmp x5, x4
    b.lt memcmp_loop

memcmp_len_check:
    // All min_len bytes equal, compare lengths
    sub x0, x2, x3
    ret

memcmp_diff:
    // Return difference as signed byte? But we need signed result for comparison.
    // Since bytes are unsigned, we can subtract and sign-extend.
    sub w0, w6, w7
    sxtw x0, w0
    ret

// write_sorted_lines: write all lines in sorted order to stdout
write_sorted_lines:
    // Load line count
    adrp x19, line_count
    add x19, x19, :lo12:line_count
    ldr x20, [x19]   // n
    cbz x20, write_done

    // Array pointer
    adrp x21, line_descriptors
    add x21, x21, :lo12:line_descriptors

    mov x22, #0      // index
write_loop:
    cmp x22, x20
    b.ge write_done

    // Get descriptor
    lsl x23, x22, #4
    add x23, x21, x23
    ldr x1, [x23]     // pointer
    ldr x2, [x23, #8] // length

    // Write line
    mov x0, #STDOUT_FD
    mov x8, #SYS_WRITE
    svc #0

    // Write newline
    adrp x24, newline_char
    add x24, x24, :lo12:newline_char
    mov x0, #STDOUT_FD
    mov x1, x24
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0

    add x22, x22, #1
    b write_loop

write_done:
    ret

.section .rodata
newline_char:
    .byte 0x0A
