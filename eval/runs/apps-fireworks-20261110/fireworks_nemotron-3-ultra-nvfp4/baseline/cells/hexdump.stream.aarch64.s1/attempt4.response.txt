.section .rodata
hex_digits: .ascii "0123456789abcdef"
.align 3

.section .bss
.align 12
input_buffer: .space 65536
output_buffer: .space 500000

.section .text
.globl _start
_start:
    // Setup buffer addresses
    adrp x19, input_buffer
    add  x19, x19, :lo12:input_buffer
    adrp x20, output_buffer
    add  x20, x20, :lo12:output_buffer
    adrp x21, hex_digits
    add  x21, x21, :lo12:hex_digits

    // Read all input
    mov x22, 0          // total_read
read_loop:
    mov x25, 65536
    sub x25, x25, x22
    cbz x25, read_done
    mov x8, 63          // read
    mov x0, 0           // stdin
    add x1, x19, x22
    mov x2, x25
    svc #0
    cbz x0, read_done
    add x22, x22, x0
    b read_loop

read_done:
    // Process input
    mov x23, 0          // offset
    mov x24, 0          // output_pos
process_loop:
    cmp x23, x22
    b.ge process_done

    // row_len = min(16, total_read - offset)
    sub x25, x22, x23
    mov x26, 16
    cmp x25, 16
    csel x25, x26, x25, gt   // x25 = row_len

    add x27, x20, x24        // current output pointer

    // Write offset (8 hex digits)
    mov x28, 8
    mov x29, x23
1:
    lsr x30, x29, 60
    and x30, x30, 0xF
    ldrb w30, [x21, x30]
    strb w30, [x27], #1
    lsl x29, x29, 4
    subs x28, x28, 1
    b.ne 1b

    // Two spaces
    mov w30, 0x2020
    strh w30, [x27], #2

    // Hex bytes (16 positions)
    mov x26, 0
2:
    cmp x26, 16
    b.ge 3f

    cmp x26, x25
    b.ge 4f

    // i < row_len: load byte and print hex
    add x29, x19, x23
    add x29, x29, x26
    ldrb w28, [x29]

    lsr w29, w28, 4
    ldrb w29, [x21, x29]
    strb w29, [x27], #1

    and w29, w28, 0xF
    ldrb w29, [x21, x29]
    strb w29, [x27], #1

    mov w29, ' '
    strb w29, [x27], #1
    b 5f

4:
    // print three spaces
    mov w29, ' '
    strb w29, [x27], #1
    strb w29, [x27], #1
    strb w29, [x27], #1

5:
    // extra space after 7th byte (i==7)
    cmp x26, 7
    b.ne 6f
    mov w29, ' '
    strb w29, [x27], #1

6:
    add x26, x26, 1
    b 2b

3:
    // Print " |"
    mov w29, 0x7C20   // ' ' then '|'
    strh w29, [x27], #2

    // ASCII representation
    mov x26, 0
7:
    cmp x26, x25
    b.ge 8f
    add x29, x19, x23
    add x29, x29, x26
    ldrb w28, [x29]
    cmp w28, 0x20
    b.lt 9f
    cmp w28, 0x7E
    b.gt 9f
    // printable
    strb w28, [x27], #1
    b 10f
9:
    // non-printable
    mov w28, '.'
    strb w28, [x27], #1
10:
    add x26, x26, 1
    b 7b

8:
    // Print "|" and newline
    mov w28, '|'
    strb w28, [x27], #1
    mov w28, '\n'
    strb w28, [x27], #1

    // Update offset and output_pos
    add x23, x23, x25
    sub x24, x27, x20
    b process_loop

process_done:
    // Write final offset (8 hex digits + newline)
    add x27, x20, x24
    mov x28, 8
    mov x29, x22
11:
    lsr x30, x29, 60
    and x30, x30, 0xF
    ldrb w30, [x21, x30]
    strb w30, [x27], #1
    lsl x29, x29, 4
    subs x28, x28, 1
    b.ne 11b
    mov w28, '\n'
    strb w28, [x27], #1

    // Calculate total output length
    sub x24, x27, x20

    // Write output to stdout
    mov x8, 64          // write
    mov x0, 1           // stdout
    mov x1, x20
    mov x2, x24
    svc #0

    // Exit
    mov x8, 93          // exit
    mov x0, 0
    svc #0
