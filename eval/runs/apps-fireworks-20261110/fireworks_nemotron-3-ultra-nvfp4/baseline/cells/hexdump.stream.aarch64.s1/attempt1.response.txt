.section .rodata
hex_digits: .ascii "0123456789abcdef"
.align 3

.section .bss
.align 12
input_buffer: .space 65536
output_buffer: .space 400000

.section .text
.globl _start
_start:
    // Initialize buffer pointers
    adrp x19, input_buffer
    add  x19, x19, :lo12:input_buffer
    adrp x20, output_buffer
    add  x20, x20, :lo12:output_buffer
    adrp x23, hex_digits
    add  x23, x23, :lo12:hex_digits

    mov x21, 0          // total_read
    mov x22, 0          // output_pos

    // ----- Read all input -----
read_loop:
    mov x0, 65536
    sub x0, x0, x21          // remaining space
    cbz x0, read_done        // buffer full (should not happen per spec)
    mov x8, 63               // read
    mov x0, 0                // stdin
    add x1, x19, x21         // buffer + total_read
    mov x2, x0               // count = remaining
    svc #0
    cbz x0, read_done        // EOF
    add x21, x21, x0         // total_read += bytes_read
    b read_loop

read_done:
    // ----- Process input buffer -----
    mov x24, 0               // offset
process_loop:
    cmp x24, x21
    b.ge process_done

    // row_len = min(16, total_read - offset)
    sub x25, x21, x24
    mov x26, 16
    cmp x25, 16
    csel x25, x26, x25, gt   // x25 = row_len

    add x27, x20, x22        // current output pointer

    // Write offset (8 hex digits)
    mov x28, 8
    mov x29, x24
1:
    lsr x30, x29, 60
    and x30, x30, 0xF
    ldrb w30, [x23, x30]
    strb w30, [x27], #1
    lsl x29, x29, 4
    subs x28, x28, 1
    b.ne 1b

    // Two spaces
    mov w30, 0x2020
    strh w30, [x27], #2

    // Hex bytes (16 positions)
    mov x26, 0
2:
    cmp x26, 16
    b.ge 3f

    cmp x26, x25
    b.ge 2f_skip
