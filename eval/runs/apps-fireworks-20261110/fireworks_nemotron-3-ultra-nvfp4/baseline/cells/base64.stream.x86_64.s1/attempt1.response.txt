.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAX_INPUT, 49152
.equ MAX_OUTPUT, 70000

.section .text
_start:
    # Read all input into input_buf
    xor r12, r12                # total_read = 0
    lea r14, [rip + input_buf]  # current input pointer

.read_loop:
    mov rdi, 0                  # stdin
    mov rsi, r14
    mov rdx, MAX_INPUT
    sub rdx, r12
    jz .read_done
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz .read_done
    add r12, rax                # total_read += bytes read
    add r14, rax
    cmp r12, MAX_INPUT
    jne .read_loop

.read_done:
    test r12, r12
    jz .write_output            # empty input -> no output

    # Encode input_buf to output_buf with line breaks
    lea r14, [rip + input_buf]  # input pointer
    lea r15, [rip + output_buf] # output pointer
    mov r13, 0                  # output length
    mov r8, 0                   # chars in current line (0-76)
    lea rbx, [rip + b64_table]  # base64 alphabet
    mov r9, r12                 # remaining input bytes

.encode_loop:
    cmp r9, 3
    jb .encode_final

    # Load 3 bytes
    movzx eax, byte ptr [r14]
    movzx edx, byte ptr [r14+1]
    movzx ecx, byte ptr [r14+2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx

    # First base64 char
    mov r10d, eax
    shr r10d, 18
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c1_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c1_done:

    # Second base64 char
    mov r10d, eax
    shr r10d, 12
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c2_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c2_done:

    # Third base64 char
    mov r10d, eax
    shr r10d, 6
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c3_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c3_done:

    # Fourth base64 char
    mov r10d, eax
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c4_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c4_done:

    add r14, 3
    sub r9, 3
    jmp .encode_loop

.encode_final:
    cmp r9, 0
    je .encode_done

    # 1 or 2 bytes left
    movzx eax, byte ptr [r14]
    mov r10d, eax
    shl r10d, 16
    cmp r9, 2
    jne .one_byte

    # Two bytes
    movzx edx, byte ptr [r14+1]
    shl edx, 8
    or r10d, edx
    mov eax, r10d

    # Char 1
    mov r10d, eax
    shr r10d, 18
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c1_done2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c1_done2:

    # Char 2
    mov r10d, eax
    shr r10d, 12
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c2_done2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c2_done2:

    # Char 3
    mov r10d, eax
    shr r10d, 6
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c3_done2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c3_done2:

    # Char 4 = '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c4_done2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c4_done2:
    jmp .encode_done

.one_byte:
    # One byte
    mov eax, r10d

    # Char 1
    mov r10d, eax
    shr r10d, 18
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c1_done1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c1_done1:

    # Char 2
    mov r10d, eax
    shr r10d, 12
    and r10d, 0x3F
    mov r10b, byte ptr [rbx + r10]
    mov byte ptr [r15], r10b
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c2_done1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c2_done1:

    # Char 3 = '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c3_done1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c3_done1:

    # Char 4 = '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc r8
    cmp r8, 76
    jne .c4_done1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor r8, r8
.c4_done1:

.encode_done:
    # Final newline if last line not empty
    test r8, r8
    jz .write_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13

.write_output:
    test r13, r13
    jz .exit
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r13
    mov rax, SYS_WRITE
    syscall
    cmp rax, r13
    jne .write_partial
    jmp .exit

.write_partial:
    sub r13, rax
    add rsi, rax
    mov rdx, r13
    mov rax, SYS_WRITE
    syscall
    cmp rax, r13
    jne .write_partial

.exit:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 4096
input_buf:
    .zero MAX_INPUT
output_buf:
    .zero MAX_OUTPUT
