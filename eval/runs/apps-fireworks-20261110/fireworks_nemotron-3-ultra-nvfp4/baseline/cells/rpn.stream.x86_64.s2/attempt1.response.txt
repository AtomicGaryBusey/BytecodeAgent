.text
.globl _start

# System call numbers
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

# Constants
.equ MAX_INPUT, 65536
.equ MAX_STACK, 512

# Entry point
_start:
    call read_all
    call process_input
    mov edi, 0
    call sys_exit

# Read all input into input_buffer
read_all:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov qword ptr [rip + input_len], 0
    lea r12, [rip + input_buffer]
.read_loop:
    mov rax, qword ptr [rip + input_len]
    cmp rax, MAX_INPUT
    jae .read_done
    mov rdi, 0
    lea rsi, [r12 + rax]
    mov rdx, MAX_INPUT
    sub rdx, rax
    call sys_read
    test rax, rax
    jz .read_done
    add qword ptr [rip + input_len], rax
    jmp .read_loop
.read_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# Process input buffer line by line
process_input:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rsi, qword ptr [rip + input_buffer]
    mov rcx, qword ptr [rip + input_len]
    test rcx, rcx
    jz .done
.line_loop:
    # Find newline
    mov rdi, rsi
    mov r8, rcx
    xor r9, r9
.find_nl:
    test r8, r8
    jz .no_nl
    movzx eax, byte ptr [rdi + r9]
    cmp al, 10
    je .found_nl
    inc r9
    dec r8
    jmp .find_nl
.found_nl:
    mov rdx, r9
    call process_line
    inc r9
    add rsi, r9
    sub rcx, r9
    test rcx, rcx
    jnz .line_loop
    jmp .done
.no_nl:
    mov rdx, rcx
    call process_line
.done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# Process a single line (rsi = start, rdx = length)
process_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    mov r13, rdx
    mov qword ptr [rip + eval_stack_top], 0
    mov byte ptr [rip + error_flag], 0
.token_loop:
    test r13, r13
    jz .after_tokens
    movzx eax, byte ptr [r12]
    cmp al, ' '
    je .skip_sep
    cmp al, '\t'
    je .skip_sep
    jmp .token_start
.skip_sep:
    inc r12
    dec r13
    jmp .token_loop
.token_start:
    mov rdi, r12
    mov rcx, r13
    xor rsi, rsi
.find_token_end:
    test rcx, rcx
    jz .token_end_found
    movzx eax, byte ptr [rdi + rsi]
    cmp al, ' '
    je .token_end_found
    cmp al, '\t'
    je .token_end_found
    inc rsi
    dec rcx
    jmp .find_token_end
.token_end_found:
    add r12, rsi
    sub r13, rsi
    call process_token
    cmp byte ptr [rip + error_flag], 0
    jne .line_error
    jmp .token_loop
.after_tokens:
    mov rax, qword ptr [rip + eval_stack_top]
    cmp rax, 1
    jne .line_error
    mov rdi, qword ptr [rip + eval_stack]
    mov rdi, qword ptr [rdi]
    call write_number
    jmp .line_done
.line_error:
    call write_error
.line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

# Process a token (rdi = start, rsi = length)
process_token:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    cmp r13, 1
    jne .parse_number
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je .op_add
    cmp al, '-'
    je .op_sub
    cmp al, '*'
    je .op_mul
    cmp al, '/'
    je .op_div
    cmp al, '%'
    je .op_mod
.parse_number:
    xor r14, r14
    mov r15, 0
    movzx eax, byte ptr [r12]
    cmp al, '-'
    jne .check_digits
    mov r14, 1
    mov r15, 1
    dec r13
    jz .number_error
.check_digits:
    cmp r13, 0
    je .number_error
    cmp r13, 18
    ja .number_error
    mov rcx, r13
    lea rdi, [r12 + r15]
.digit_loop:
    movzx eax, byte ptr [rdi]
    cmp al, '0'
    jb .number_error
    cmp al, '9'
    ja .number_error
    inc rdi
    dec rcx
    jnz .digit_loop
    xor rax, rax
    mov rcx, r13
    lea rdi, [r12 + r15]
.parse_loop:
    movzx rdx, byte ptr [rdi]
    sub edx, '0'
    imul rax, rax, 10
    add rax, rdx
    inc rdi
    dec rcx
    jnz .parse_loop
    test r14, r14
    jz .push_number
    neg rax
.push_number:
    call push_one
    jmp .token_done
.op_add:
    call do_add
    jmp .token_done
.op_sub:
    call do_sub
    jmp .token_done
.op_mul:
    call do_mul
    jmp .token_done
.op_div:
    call do_div
    jmp .token_done
.op_mod:
    call do_mod
    jmp .token_done
.number_error:
    mov byte ptr [rip + error_flag], 1
.token_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# Stack operations
push_one:
    mov rcx, qword ptr [rip + eval_stack_top]
    cmp rcx, MAX_STACK
    jae .overflow
    mov rdx, qword ptr [rip + eval_stack]
    mov qword ptr [rdx + rcx*8], rax
    inc qword ptr [rip + eval_stack_top]
    ret
.overflow:
    mov byte ptr [rip + error_flag], 1
    ret

pop_two:
    mov rcx, qword ptr [rip + eval_stack_top]
    cmp rcx, 2
    jb .underflow
    dec rcx
    mov rax, qword ptr [rip + eval_stack + rcx*8]
