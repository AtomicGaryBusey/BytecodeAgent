.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 16
input_buffer: .zero MAX_INPUT
line_ptrs: .zero MAX_LINES * 8
line_lens: .zero MAX_LINES * 4
output_buffer: .zero MAX_INPUT + MAX_LINES
num_lines: .zero 8
total_read: .zero 8

.section .text
.globl _start

_start:
    # Read all input
    xor r10, r10                    # total_read = 0
    lea rsi, [rip + input_buffer]
    mov rdx, MAX_INPUT
read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    syscall
    test rax, rax
    jz read_done
    add r10, rax
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov [rip + total_read], r10

    # Parse lines
    lea rsi, [rip + input_buffer]
    mov rdi, r10
    call parse_lines

    # Sort lines
    mov rdi, rax
    call insertion_sort

    # Write output
    call write_output

    # Exit
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

parse_lines:
    # rsi = buffer, rdi = total_bytes
    # returns number of lines in rax
    mov r8, rsi
    lea r9, [rsi + rdi]             # end pointer
    xor r10, r10                    # line count
    lea r14, [rip + line_ptrs]
    lea r15, [rip + line_lens]
parse_loop:
    cmp r8, r9
    jae parse_done
    mov r11, r8                     # line_start
find_newline:
    cmp byte ptr [r8], 0x0A
    je found_newline
    inc r8
    cmp r8, r9
    jne find_newline
    # No newline found, line goes to end
    mov rdx, r8
    sub rdx, r11
    mov [r14 + r10*8], r11
    mov [r15 + r10*4], edx
    inc r10
    jmp parse_done
found_newline:
    mov rdx, r8
    sub rdx, r11
    mov [r14 + r10*8], r11
    mov [r15 + r10*4], edx
    inc r10
    inc r8                          # skip newline
    jmp parse_loop
parse_done:
    mov [rip + num_lines], r10
    mov rax, r10
    ret

compare_idx:
    # rdi = idx1, rsi = idx2
    # returns -1, 0, 1 in rax
    lea r8, [rip + line_ptrs]
    lea r9, [rip + line_lens]
    mov r10, [r8 + rdi*8]
    mov r11d, [r9 + rdi*4]
    mov r12, [r8 + rsi*8]
    mov r13d, [r9 + rsi*4]

    mov eax, r11d
    cmp eax, r13d
    cmovg eax, r13d                 # eax = min(len1, len2)
    test eax, eax
    jz compare_lengths
    xor ecx, ecx
byte_loop:
    movzx edx, byte ptr [r10 + rcx]
    movzx edi, byte ptr [r12 + rcx]
    cmp dl, dil
    jne byte_diff
    inc rcx
    cmp ecx, eax
    jne byte_loop
compare_lengths:
    cmp r11d, r13d
    jl less
    jg greater
    xor eax, eax
    ret
less:
    mov eax, -1
    ret
greater:
    mov eax, 1
    ret
byte_diff:
    cmp dl, dil
    jl less
    jg greater
    xor eax, eax
    ret

insertion_sort:
    # rdi = num_lines
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r14, rdi                    # save num_lines
    lea r12, [rip + line_ptrs]
    lea r13, [rip + line_lens]
    mov r15, 1                      # i = 1
    cmp r14, 1
    jle sort_done
outer_loop:
    mov rbx, r15                    # j = i
inner_loop:
    test rbx, rbx
    jz inner_done
    mov rdi, rbx
    dec rdi                         # j-1
    mov rsi, rbx
    call compare_idx
    cmp rax, 1
    jle inner_done
    # Swap line_ptrs
    mov r8, [r12 + rdi*8]
    mov r9, [r12 + rsi*8]
    mov [r12 + rdi*8], r9
    mov [r12 + rsi*8], r8
    # Swap line_lens
    mov r8d, [r13 + rdi*4]
    mov r9d, [r13 + rsi*4]
    mov [r13 + rdi*4], r9d
    mov [r13 + rsi*4], r8d
    dec rbx
    jmp inner_loop
inner_done:
    inc r15
    cmp r15, r14
    jne outer_loop
sort_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

write_output:
    mov r10, [rip + num_lines]
    test r10, r10
    jz write_done
    lea r11, [rip + output_buffer]
    xor r12, r12                    # out_pos
    xor r13, r13                    # line index
    lea r14, [rip + line_ptrs]
    lea r15, [rip + line_lens]
write_loop:
    mov r8, [r14 + r13*8]
    mov r9d, [r15 + r13*4]
    # Copy line to output buffer
    mov rdi, r11
    add rdi, r12
    mov rsi, r8
    mov ecx, r9d
    rep movsb
    add r12, r9
    # Add newline
    mov byte ptr [r11 + r12], 0x0A
    inc r12
    inc r13
    cmp r13, r10
    jne write_loop
    # Write all
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r12
    call write_all
write_done:
    ret

write_all:
    # rdi = fd, rsi = buf, rdx = count
    push rdi
    push rsi
    push rdx
write_loop:
    mov rax, SYS_WRITE
    syscall
    test rax, rax
    jle write_error
    sub rdx, rax
    jz write_done
    add rsi, rax
    jmp write_loop
write_done:
    pop rdx
    pop rsi
    pop rdi
    ret
write_error:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall
