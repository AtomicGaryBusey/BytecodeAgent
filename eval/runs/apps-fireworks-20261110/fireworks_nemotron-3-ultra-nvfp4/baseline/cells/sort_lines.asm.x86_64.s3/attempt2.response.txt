.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 16
input_buffer: .zero MAX_INPUT
line_ptrs: .zero MAX_LINES * 8
line_lens: .zero MAX_LINES * 4
output_buffer: .zero MAX_INPUT + MAX_LINES
num_lines: .zero 8
total_read: .zero 8

.section .text
.globl _start

_start:
    # Read all input
    xor r10, r10                    # total_read = 0
    lea rsi, [rip + input_buffer]
    mov rdx, MAX_INPUT
.read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    syscall
    test rax, rax
    jz .read_done
    add r10, rax
    add rsi, rax
    sub rdx, rax
    jnz .read_loop
.read_done:
    mov [rip + total_read], r10

    # Parse lines
    lea rsi, [rip + input_buffer]
    mov rdi, r10
    call parse_lines

    # Sort lines
    mov rdi, rax
    call insertion_sort

    # Write output
    call write_output

    # Exit
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

parse_lines:
    # rsi = buffer, rdi = total_bytes
    # returns number of lines in rax
    mov r8, rsi
    lea r9, [rsi + rdi]             # end pointer
    xor r10, r10                    # line count
.loop:
    cmp r8, r9
    jae .done
    mov r11, r8                     # line_start
.find_newline:
    cmp byte ptr [r8], 0x0A
    je .found
    inc r8
    cmp r8, r9
    jne .find_newline
    # No newline found, line goes to end
    mov rdx, r8
    sub rdx, r11
    mov [rip + line_ptrs + r10*8], r11
    mov [rip + line_lens + r10*4], edx
    inc r10
    jmp .done
.found:
    mov rdx, r8
    sub rdx, r11
    mov [rip + line_ptrs + r10*8], r11
    mov [rip + line_lens + r10*4], edx
    inc r10
    inc r8                          # skip newline
    jmp .loop
.done:
    mov [rip + num_lines], r10
    mov rax, r10
    ret

compare_idx:
    # rdi = idx1, rsi = idx2
    # returns -1, 0, 1 in rax
    mov r8, [rip + line_ptrs + rdi*8]
    mov r9d, [rip + line_lens + rdi*4]
    mov r10, [rip + line_ptrs + rsi*8]
    mov r11d, [rip + line_lens + rsi*4]

    mov eax, r9d
    cmp eax, r11d
    cmovg eax, r11d                 # eax = min(len1, len2)
    test eax, eax
    jz .compare_lengths
    xor ecx, ecx
.byte_loop:
    movzx edx, byte ptr [r8 + rcx]
    movzx edi, byte ptr [r10 + rcx]
    cmp dl, dil
    jne .byte_diff
    inc rcx
    cmp ecx, eax
    jne .byte_loop
.compare_lengths:
    cmp r9d, r11d
    jl .less
    jg .greater
    xor eax, eax
    ret
.less:
    mov eax, -1
    ret
.greater:
    mov eax, 1
    ret
.byte_diff:
    cmp dl, dil
    jl .less
    jg .greater
    xor eax, eax
    ret

insertion_sort:
    # rdi = num_lines
    push rbx
    push r12
    push r13
    push r14
    mov r14, rdi                    # save num_lines
    mov r12, 1                      # i = 1
    cmp r14, 1
    jle .done
.outer_loop:
    mov r13, r12                    # j = i
.inner_loop:
    test r13, r13
    jz .inner_done
    mov rdi, r13
    dec rdi                         # j-1
    mov rsi, r13
    call compare_idx
    cmp rax, 1
    jle .inner_done
    # Swap line_ptrs
    mov r8, [rip + line_ptrs + rdi*8]
    mov r9, [rip + line_ptrs + rsi*8]
    mov [rip + line_ptrs + rdi*8], r9
    mov [rip + line_ptrs + rsi*8], r8
    # Swap line_lens
    mov r8d, [rip + line_lens + rdi*4]
    mov r9d, [rip + line_lens + rsi*4]
    mov [rip + line_lens + rdi*4], r9d
    mov [rip + line_lens + rsi*4], r8d
    dec r13
    jmp .inner_loop
.inner_done:
    inc r12
    cmp r12, r14
    jne .outer_loop
.done:
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

write_output:
    mov r10, [rip + num_lines]
    test r10, r10
    jz .done
    lea r11, [rip + output_buffer]
    xor r12, r12                    # out_pos
    xor r13, r13                    # line index
.loop:
    mov r8, [rip + line_ptrs + r13*8]
    mov r9d, [rip + line_lens + r13*4]
    # Copy line to output buffer
    mov rdi, r11
    add rdi, r12
    mov rsi, r8
    mov ecx, r9d
    rep movsb
    add r12, r9
    # Add newline
    mov byte ptr [r11 + r12], 0x0A
    inc r12
    inc r13
    cmp r13, r10
    jne .loop
    # Write all
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r12
    call write_all
.done:
    ret

write_all:
    # rdi = fd, rsi = buf, rdx = count
    push rdi
    push rsi
    push rdx
.write_loop:
    mov rax, SYS_WRITE
    syscall
    test rax, rax
    jle .write_error
    sub rdx, rax
    jz .write_done
    add rsi, rax
    jmp .write_loop
.write_done:
    pop rdx
    pop rsi
    pop rdi
    ret
.write_error:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall
