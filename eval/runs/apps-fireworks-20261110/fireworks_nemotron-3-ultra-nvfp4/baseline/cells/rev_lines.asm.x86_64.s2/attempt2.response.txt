.globl _start

.section .bss
.align 4096
input_buf:
    .zero 65536
output_buf:
    .zero 131072

.section .text
_start:
    # Read all input into input_buf
    mov rdi, 0                  # fd = 0 (stdin)
    lea rsi, [rip + input_buf]  # buffer pointer
    mov rdx, 65536              # max bytes to read
    xor r12, r12                # total_read = 0

read_loop:
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done                # EOF
    add r12, rax                # total_read += bytes read
    add rsi, rax                # advance buffer pointer
    sub rdx, rax                # decrease remaining space
    jnz read_loop               # continue if buffer not full

read_done:
    test r12, r12
    jz exit_program             # empty input -> nothing to output

    # Process input buffer: reverse each line and write to output_buf
    lea rsi, [rip + input_buf]  # current input pointer
    lea rdi, [rip + output_buf] # current output pointer
    lea r14, [rip + output_buf] # output buffer base for length calc
    mov rcx, r12                # remaining input bytes

process_loop:
    test rcx, rcx
    jz write_output             # no more input

    mov r9, rsi                 # line start
    xor r10, r10                # line length (including newline if found)

find_newline:
    test rcx, rcx
    jz line_end_no_newline      # reached end of input without newline
    mov al, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 0x0A                # newline?
    jne find_newline

    # Found newline: line length is r10 - 1 (exclude newline)
    dec r10
    jmp reverse_copy

line_end_no_newline:
    # No newline: line length is r10 (all remaining bytes)
    # r10 already holds the length

reverse_copy:
    # Copy line in reverse from r9 to rdi
    # Source end = r9 + r10 - 1
    lea r11, [r9 + r10 - 1]
    mov r13, r10                # copy counter

copy_loop:
    test r13, r13
    jz copy_done
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    dec r11
    dec r13
    jmp copy_loop

copy_done:
    # Append newline to output
    mov byte ptr [rdi], 0x0A
    inc rdi
    jmp process_loop

write_output:
    # Write output buffer to stdout
    mov rdx, rdi
    sub rdx, r14                # output length = current_ptr - base_ptr
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = 1 (stdout)
    lea rsi, [rip + output_buf]
    syscall

exit_program:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status = 0
    syscall
