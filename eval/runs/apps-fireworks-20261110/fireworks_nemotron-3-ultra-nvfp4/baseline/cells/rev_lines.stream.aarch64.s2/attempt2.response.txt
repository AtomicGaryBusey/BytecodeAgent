.section .rodata
newline_char:
    .byte 0x0A

.section .bss
.align 12
buffer:
    .space 65536

.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 65536

_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0

read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done

    mov x0, #STDIN
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0

    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x22, x19, x20
    mov x23, x19

process_loop:
    cmp x23, x22
    b.ge process_done

    mov x24, x23

find_newline:
    cmp x24, x22
    b.ge no_newline_found
    ldrb w25, [x24]
    cmp w25, #0x0A
    b.eq newline_found
    add x24, x24, #1
    b find_newline

newline_found:
    mov x0, x23
    mov x1, x24
    bl reverse_mem

    mov x0, #STDOUT
    mov x1, x23
    sub x2, x24, x23
    add x2, x2, #1
    bl write_all

    add x23, x24, #1
    b process_loop

no_newline_found:
    cmp x23, x22
    b.eq process_done

    mov x0, x23
    mov x1, x22
    bl reverse_mem

    mov x0, #STDOUT
    mov x1, x23
    sub x2, x22, x23
    bl write_all

    mov x0, #STDOUT
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x2, #1
    bl write_all

process_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

reverse_mem:
    cmp x0, x1
    b.ge reverse_done
    sub x1, x1, #1
reverse_loop:
    cmp x0, x1
    b.ge reverse_done
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w2, [x1]
    strb w3, [x0]
    add x0, x0, #1
    sub x1, x1, #1
    b reverse_loop
reverse_done:
    ret

write_all:
    mov x4, x2
    mov x5, x1
    mov x6, x0
write_loop:
    cbz x4, write_done
    mov x0, x6
    mov x1, x5
    mov x2, x4
    mov x8, #SYS_WRITE
    svc #0
    sub x4, x4, x0
    add x5, x5, x0
    b write_loop
write_done:
    ret
