.section .text
.globl _start

_start:
    // Initialize counters
    mov x19, #0          // lines
    mov x20, #0          // words
    mov x21, #0          // bytes
    mov x22, #0          // in_word flag (0 = not in word, 1 = in word)

read_loop:
    // read(0, buffer, 4096)
    mov x8, #63          // sys_read
    mov x0, #0           // fd = stdin
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov x2, #4096
    svc #0               // x0 = bytes read

    cbz x0, process_done // EOF

    add x21, x21, x0     // bytes += count

    // Process buffer
    mov x3, #0           // index
process_loop:
    cmp x3, x0
    b.ge read_loop       // processed all bytes in this chunk

    ldrb w4, [x1, x3]    // load byte

    // Count newlines (0x0A)
    cmp w4, #0x0A
    b.ne not_newline
    add x19, x19, #1
not_newline:

    // Check for whitespace: space(0x20), tab(0x09), newline(0x0A), vtab(0x0B), ff(0x0C), cr(0x0D)
    mov x5, #0           // whitespace flag
    cmp w4, #0x20
    cset x6, eq
    orr x5, x5, x6
    cmp w4, #0x09
    cset x6, eq
    orr x5, x5, x6
    cmp w4, #0x0A
    cset x6, eq
    orr x5, x5, x6
    cmp w4, #0x0B
    cset x6, eq
    orr x5, x5, x6
    cmp w4, #0x0C
    cset x6, eq
    orr x5, x5, x6
    cmp w4, #0x0D
    cset x6, eq
    orr x5, x5, x6

    // If whitespace
    cbnz x5, check_word_end
    // Not whitespace
    cbnz x22, continue_processing
    mov x22, #1          // enter word
    b continue_processing

check_word_end:
    cbz x22, continue_processing
    add x20, x20, #1     // word ends
    mov x22, #0

continue_processing:
    add x3, x3, #1
    b process_loop

process_done:
    // If still in a word at EOF, count it
    cbz x22, output
    add x20, x20, #1

output:
    // Print lines
    mov x0, x19
    bl print_num
    // Print space
    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    // Print words
    mov x0, x20
    bl print_num
    // Print space
    mov x0, #1
    adrp x1, space_str
    add x1, x1, :lo12:space_str
    mov x2, #1
    mov x8, #64
    svc #0

    // Print bytes
    mov x0, x21
    bl print_num
    // Print newline
    mov x0, #1
    adrp x1, newline_str
    add x1, x1, :lo12:newline_str
    mov x2, #1
    mov x8, #64
    svc #0

    // Exit
    mov x8, #93
    mov x0, #0
    svc #0

// print_num: prints unsigned 64-bit integer in x0 to stdout (fd=1)
// Uses a 32-byte stack buffer for digits.
print_num:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #16           // 32 bytes total for buffer
    mov x3, sp                // buffer pointer

    cbz x0, print_zero

    mov x4, #0                // digit count
gen_digits:
    mov x5, #10
    udiv x6, x0, x5           // x6 = x0 / 10
    msub x7, x6, x5, x0       // x7 = x0 % 10
    add w7, w7, #0x30         // '0'
    strb w7, [x3, x4]
    add x4, x4, #1
    mov x0, x6
    cbnz x0, gen_digits

    // Reverse buffer in place
    mov x5, #0
    sub x6, x4, #1
reverse_loop:
    cmp x5, x6
    b.ge reverse_done
    ldrb w7, [x3, x5]
    ldrb w8, [x3, x6]
    strb w8, [x3, x5]
    strb w7, [x3, x6]
    add x5, x5, #1
    sub x6, x6, #1
    b reverse_loop
reverse_done:

    // Write buffer
    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #64
    svc #0
    b print_num_done

print_zero:
    mov w7, #0x30
    strb w7, [x3]
    mov x0, #1
    mov x1, x3
    mov x2, #1
    mov x8, #64
    svc #0

print_num_done:
    add sp, sp, #16
    ldp x29, x30, [sp], #16
    ret

.section .rodata
space_str:
    .byte 0x20
newline_str:
    .byte 0x0A

.section .bss
.align 12
buffer:
    .space 4096
