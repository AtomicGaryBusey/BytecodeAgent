.section .data
newline_char: .byte 10

.section .bss
.align 8
input_len: .zero 8
.align 4096
input_buf: .zero 65536
temp_buf: .zero 65536

.section .text
.globl _start

_start:
    # read all input into input_buf
    xor r12, r12                # total bytes read
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    mov rdi, 0                  # stdin
read_loop:
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov [rip + input_len], r12

    # process lines
    xor rsi, rsi                # current index
    xor rdi, rdi                # line start
process_loop:
    cmp rsi, r12
    jge process_end
    mov al, byte ptr [rip + input_buf + rsi]
