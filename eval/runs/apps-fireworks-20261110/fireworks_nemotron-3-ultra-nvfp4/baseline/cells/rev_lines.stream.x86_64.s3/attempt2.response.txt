.section .data
newline_char: .byte 10

.section .bss
.align 4096
input_buf: .zero 65536
temp_buf: .zero 65536

.section .text
.globl _start

_start:
    lea rbx, [rip + input_buf]
    lea rbp, [rip + temp_buf]
    xor r12, r12                # total bytes read

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jne read_loop

read_done:
    xor r13, r13                # current index
    xor r14, r14                # line start index

process_loop:
    cmp r13, r12
    jge check_last_line
    mov al, byte ptr [rbx + r13]
    cmp al, 10
    je newline_found
    inc r13
    jmp process_loop

newline_found:
    call reverse_and_write
    inc r13
    mov r14, r13
    jmp process_loop

check_last_line:
    cmp r14, r12
    jge exit_program
    mov r13, r12
    call reverse_and_write

exit_program:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

reverse_and_write:
    # r14 = start offset, r13 = end offset (exclusive)
    # rbx = input_buf, rbp = temp_buf
    mov rcx, r13
    sub rcx, r14                # line length
    test rcx, rcx
    jz write_newline_only

    lea rsi, [rbx + r13 - 1]    # last byte of line
    mov rdi, rbp                # start of temp_buf

reverse_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    dec rcx
    jnz reverse_loop

    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    mov rsi, rbp
    mov rdx, r13
    sub rdx, r14
    syscall

write_newline_only:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    ret
