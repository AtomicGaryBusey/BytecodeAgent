.section .text
.globl _start

_start:
    xor r12, r12          # line_count
    xor r13, r13          # word_count
    xor r14, r14          # byte_count
    xor r15, r15          # in_word flag (0 = not in word, 1 = in word)

read_loop:
    mov rax, 0            # sys_read
    mov rdi, 0            # stdin
    lea rsi, [rip + read_buf]
    mov rdx, 4096
    syscall
    test rax, rax
    jz read_done
    add r14, rax          # byte_count += bytes_read
    mov rcx, rax          # length
    lea rsi, [rip + read_buf]
    call process_buffer
    jmp read_loop

read_done:
    test r15, r15
    jz no_final_word
    inc r13               # count final word if input ended in a word
no_final_word:

    lea rdi, [rip + output_buf]
    mov r10, rdi          # save start of output buffer
    mov rsi, r12
    call write_number
    mov byte ptr [rdi], ' '
    inc rdi
    mov rsi, r13
    call write_number
    mov byte ptr [rdi], ' '
    inc rdi
    mov rsi, r14
    call write_number
    mov byte ptr [rdi], 0x0A
    inc rdi

    mov rax, 1            # sys_write
    mov rdi, 1            # stdout
    mov rsi, r10
    mov rdx, rdi
    sub rdx, r10
    syscall

    mov rax, 60           # sys_exit
    xor rdi, rdi
    syscall

# process_buffer(rsi=buffer, rcx=length)
# updates r12 (lines), r13 (words), r15 (in_word)
process_buffer:
    test rcx, rcx
    jz .done
.loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A          # newline
    je .is_newline
    cmp al, 0x20          # space
    je .is_whitespace
    cmp al, 0x09          # tab
    je .is_whitespace
    cmp al, 0x0B          # vertical tab
    je .is_whitespace
    cmp al, 0x0C          # form feed
    je .is_whitespace
    cmp al, 0x0D          # carriage return
    je .is_whitespace
    # non-whitespace
    mov r15, 1
    jmp .next
.is_newline:
    inc r12
.is_whitespace:
    test r15, r15
    jz .next
    inc r13
    xor r15, r15
.next:
    dec rcx
    jnz .loop
.done:
    ret

# write_number(rdi=output_ptr, rsi=number) -> rdi = new output_ptr
write_number:
    test rsi, rsi
    jnz .non_zero
    mov byte ptr [rdi], '0'
    inc rdi
    ret
.non_zero:
    mov rax, rsi
    xor rcx, rcx          # digit count
    mov r10, 10
    lea r8, [rip + digit_buf]  # base address of digit buffer
.div_loop:
    xor rdx, rdx
    div r10               # rax = quotient, rdx = remainder
    add dl, '0'
    mov byte ptr [r8 + rcx], dl
    inc rcx
    test rax, rax
    jnz .div_loop
    # copy digits in reverse order
    lea r9, [r8 + rcx - 1]
.copy_loop:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    dec r9
    dec rcx
    jnz .copy_loop
    ret

.section .bss
.align 16
read_buf:   .zero 4096
output_buf: .zero 100
digit_buf:  .zero 24
