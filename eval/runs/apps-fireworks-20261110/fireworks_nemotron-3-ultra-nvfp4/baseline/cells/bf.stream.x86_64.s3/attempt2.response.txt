.globl _start

.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_OUTPUT, 65536
.equ MAX_NESTING, 1000
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.section .bss
.align 4096
input_buffer: .zero MAX_INPUT
bracket_map: .zero MAX_INPUT * 4
tape: .zero TAPE_SIZE
output_buffer: .zero MAX_OUTPUT
stack: .zero MAX_NESTING * 4

.section .text
_start:
    # Read all stdin into input_buffer
    xor r12, r12            # total_read = 0
    lea rsi, [rip + input_buffer]
    mov rdx, MAX_INPUT
1:
    mov rax, SYS_READ
    xor rdi, rdi            # fd = 0
    syscall
    test rax, rax
    js read_error           # negative = error (should not happen)
    jz read_done            # zero = EOF
    add r12, rax            # total_read += rax
    add rsi, rax
    sub rdx, rax
    jnz 1b
read_done:
    # r12 = total_read
    # Find first '!' (0x21)
    lea rdi, [rip + input_buffer]
    mov rcx, r12
    xor rax, rax
    mov al, 0x21
    repne scasb
    jne no_bang
    # found at position (rdi - input_buffer - 1)
    lea rax, [rip + input_buffer]
    lea r13, [rdi - 1]
    sub r13, rax            # r13 = program_length
    jmp have_bang
no_bang:
    mov r13, r12            # program_length = total_read
have_bang:
    # r13 = program_length
    # input_start = program_length + 1 if bang found, else total_read
    # input_length = total_read - input_start
    cmp r13, r12
    je no_input
    lea r14, [r13 + 1]      # input_start
    mov r15, r12
    sub r15, r14            # input_length
    jmp input_ready
no_input:
    mov r14, r12            # input_start = total_read
    xor r15, r15            # input_length = 0
input_ready:
    # Bracket matching validation and mapping
    # Use stack (in .bss) to store positions of '['
    # bracket_map: for each program index, store matching index (int32)
    lea rsi, [rip + input_buffer]   # program start
    lea rdi, [rip + stack]          # stack pointer
    xor rbp, rbp                    # stack depth = 0
    xor rcx, rcx                    # ip = 0
bracket_loop:
    cmp rcx, r13
    jge bracket_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, '['
    je bracket_open
    cmp al, ']'
    je bracket_close
    inc rcx
    jmp bracket_loop
bracket_open:
    # push current index onto stack
    cmp rbp, MAX_NESTING
    jge bracket_error
    mov dword ptr [rdi + rbp*4], ecx
    inc rbp
    inc rcx
    jmp bracket_loop
bracket_close:
    # pop from stack
    test rbp, rbp
    jz bracket_error
    dec rbp
    mov edx, dword ptr [rdi + rbp*4]   # matching '['
    # store mapping both ways
    mov dword ptr [rip + bracket_map + rcx*4], edx   # ] -> [
