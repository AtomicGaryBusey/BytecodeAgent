.globl _start

.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_OUTPUT, 65536
.equ MAX_NESTING, 1000
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.section .bss
.align 4096
input_buffer: .zero MAX_INPUT
bracket_map: .zero MAX_INPUT * 4
tape: .zero TAPE_SIZE
output_buffer: .zero MAX_OUTPUT
stack: .zero MAX_NESTING * 4

.section .text
_start:
    # Read all stdin into input_buffer
    xor r12, r12            # total_read = 0
    lea rsi, [rip + input_buffer]
    mov rdx, MAX_INPUT
1:
    mov rax, SYS_READ
    xor rdi, rdi            # fd = 0
    syscall
    test rax, rax
    js read_error           # negative = error
    jz read_done            # zero = EOF
    add r12, rax            # total_read += rax
    add rsi, rax
    sub rdx, rax
    jnz 1b
read_done:
    # r12 = total_read
    # Find first '!' (0x21)
    lea rdi, [rip + input_buffer]
    mov rcx, r12
    xor rax, rax
    mov al, 0x21
    repne scasb
    jne no_bang
    # found at position (rdi - input_buffer - 1)
    lea r13, [rdi - 1 - input_buffer]   # program_length
