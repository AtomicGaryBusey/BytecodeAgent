.section .text
.globl _start

_start:
    // Read all input into input_buffer
    adrp    x19, input_buffer
    add     x19, x19, :lo12:input_buffer   // x19 = input_buffer base
    mov     x20, 0                         // total_read = 0
    mov     x21, 49152                     // max_input = 49152

read_loop:
    sub     x22, x21, x20                  // remaining = max_input - total_read
    cbz     x22, read_done
    mov     x8, 63                         // sys_read
    mov     x0, 0                          // fd = stdin
    add     x1, x19, x20                   // buf = input_buffer + total_read
    mov     x2, x22                        // count = remaining
    svc     #0
    cbz     x0, read_done                  // EOF
    add     x20, x20, x0                   // total_read += bytes_read
    b       read_loop

read_done:
    // x20 = input length
    // Set up registers for encoding
    adrp    x26, input_buffer
    add     x26, x26, :lo12:input_buffer   // x26 = input_buffer base
    adrp    x21, output_buffer
    add     x21, x21, :lo12:output_buffer  // x21 = output_buffer base (current output pointer)
    adrp    x27, b64_table
    add     x27, x27, :lo12:b64_table      // x27 = base64 table base
    mov     x22, 0                         // line_len = 0
    mov     x23, 0                         // input_idx = 0
    mov     x24, x20                       // input_len = total_read

encode_loop:
    sub     x25, x24, x23                  // remaining bytes
    cmp     x25, 3
    b.ge    encode_triplet
    cbz     x25, encode_finish
    cmp     x25, 1
    b.eq    encode_one_byte
    // x25 == 2
    b       encode_two_bytes

encode_triplet:
    add     x28, x26, x23                  // x28 = &input[input_idx]
    ldrb    w0, [x28]                      // byte 0
    ldrb    w1, [x28, #1]                  // byte 1
    ldrb    w2, [x28, #2]                  // byte 2

    lsr     w3, w0, 2                      // idx0
    and     w4, w0, 3
    lsl     w4, w4, 4
    lsr     w5, w1, 4
    orr     w4, w4, w5                     // idx1
    and     w5, w1, 0xF
    lsl     w5, w5, 2
    lsr     w6, w2, 6
    orr     w5, w5, w6                     // idx2
    and     w6, w2, 0x3F                   // idx3

    // Convert indices to 64-bit for table lookup
    mov     x3, x3
    mov     x4, x4
    mov     x5, x5
    mov     x6, x6

    ldrb    w7, [x27, x3]
    ldrb    w8, [x27, x4]
    ldrb    w9, [x27, x5]
    ldrb    w10, [x27, x6]

    strb    w7, [x21], #1
    strb    w8, [x21], #1
    strb    w9, [x21], #1
    strb    w10, [x21], #1

    add     x22, x22, 4                    // line_len += 4
    cmp     x22, 76
    b.ne    encode_triplet_done
    mov     w7, 0x0A                       // newline
    strb    w7, [x21], #1
    mov     x22, 0                         // line_len = 0

encode_triplet_done:
    add     x23, x23, 3
    b       encode_loop

encode_one_byte:
    ldrb    w0, [x26, x23]
    lsr     w3, w0, 2
    and     w4, w0, 3
    lsl     w4, w4, 4

    mov     x3, x3
    mov     x4, x4

    ldrb    w7, [x27, x3]
    ldrb    w8, [x27, x4]
    strb    w7, [x21], #1
    strb    w8, [x21], #1
    mov     w7, '='
    strb    w7, [x21], #1
    strb    w7, [x21], #1

    add     x22, x22, 4
    cmp     x22, 76
    b.ne    encode_finish
    mov     w7, 0x0A
    strb    w7, [x21], #1
    mov     x22, 0
    b       encode_finish

encode_two_bytes:
    add     x28, x26, x23                  // x28 = &input[input_idx]
    ldrb    w0, [x28]                      // byte 0
    ldrb    w1, [x28, #1]                  // byte 1
    lsr     w3, w0, 2
    and     w4, w0, 3
    lsl     w4, w4, 4
    lsr     w5, w1, 4
    orr     w4, w4, w5
    and     w5, w1, 0xF
    lsl     w5, w5, 2

    mov     x3, x3
    mov     x4, x4
    mov     x5, x5

    ldrb    w7, [x27, x3]
    ldrb    w8, [x27, x4]
    ldrb    w9, [x27, x5]
    strb    w7, [x21], #1
    strb    w8, [x21], #1
    strb    w9, [x21], #1
    mov     w7, '='
    strb    w7, [x21], #1

    add     x22, x22, 4
    cmp     x22, 76
    b.ne    encode_finish
    mov     w7, 0x0A
    strb    w7, [x21], #1
    mov     x22, 0

encode_finish:
    cbz     x22, write_output
    mov     w7, 0x0A
    strb    w7, [x21], #1

write_output:
    adrp    x0, output_buffer
    add     x0, x0, :lo12:output_buffer    // x0 = output_buffer base
    sub     x2, x21, x0                    // x2 = output length
    cbz     x2, exit_prog
    mov     x1, x0                         // x1 = buffer
    mov     x0, 1                          // fd = stdout
    mov     x8, 64                         // sys_write
    svc     #0

exit_prog:
    mov     x8, 93                         // sys_exit
    mov     x0, 0                          // status = 0
    svc     #0

.section .rodata
.align 3
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 12
input_buffer:
    .zero 49152
.align 12
output_buffer:
    .zero 66560
