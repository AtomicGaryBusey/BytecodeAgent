.section .bss
.align 16
buffer: .zero 65536
.align 8
input_len: .zero 8
.align 8
parse_pos: .zero 8
.align 8
error_pos: .zero 8
.align 8
nesting_depth: .zero 8
.align 8
state_stack: .zero 8000

.section .rodata
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
newline: .ascii "\n"

.section .text
.globl _start

_start:
    # Read all input into buffer
    mov qword ptr [rip + input_len], 0
    lea rdi, [rip + buffer]

read_loop:
    mov rax, 0          # sys_read
    mov rdi, 0          # stdin
    mov rsi, rdi        # buffer pointer (current position)
    mov rdx, 65536
    sub rdx, [rip + input_len]
    syscall
    test rax, rax
    jz read_done
    add qword ptr [rip + input_len], rax
    add rdi, rax
    cmp qword ptr [rip + input_len], 65536
    jne read_loop

read_done:
    # Initialize parser state
    mov qword ptr [rip + parse_pos], 0
    mov qword ptr [rip + error_pos], 0
    mov qword ptr [rip + nesting_depth], 0
    
    # Skip leading whitespace
    call skip_whitespace
    
    # Check if empty input (only whitespace)
    mov rax, [rip + parse_pos]
    cmp rax, [rip + input_len]
    je error_eof
    
    # Parse value
    call parse_value
    
    # Check for error during parsing
    cmp qword ptr [rip + error_pos], 0
    jne report_error
    
    # Skip trailing whitespace
    call skip_whitespace
    
    # Check if we consumed all input
    mov rax, [rip + parse_pos]
    cmp rax, [rip + input_len]
    jne error_extra
    
    # Success
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    syscall
    mov rax, 60         # sys_exit
    xor rdi, rdi        # exit code 0
    syscall

# Skip whitespace (space, tab, newline, carriage return)
skip_whitespace:
    mov rax, [rip + parse_pos]
.skip_loop:
    cmp rax, [rip + input_len]
    je .skip_done
    movzx ecx, byte ptr [rip + buffer + rax]
