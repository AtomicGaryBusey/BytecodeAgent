.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .bss
.align 16
input_buffer: .zero 65536

.section .text
.globl _start

sys_read:
    mov rax, 0
    syscall
    ret

sys_write:
    mov rax, 1
    syscall
    ret

sys_exit:
    mov rax, 60
    syscall

print_line:
    # rdi = offset, rsi = data pointer, rdx = n (1..16)
    sub rsp, 128
    mov r8, rsp
    mov r9, rdi
    mov r10, rsi
    mov r11, rdx
    # write offset as 8 hex digits
    mov r12, r8
    mov eax, r9d
    mov ecx, 28
    lea r15, [rip + hex_digits]
.offset_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r12], dl
    inc r12
    sub ecx, 4
    jns .offset_loop
    # two spaces
    mov word ptr [r12], 0x2020
    add r12, 2
    # hex column
    xor r13, r13
.hex_loop:
    cmp r13, r11
    jge .hex_blank
    movzx eax, byte ptr [r10 + r13]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r12], dl
    mov edx, eax
    and edx, 0xF
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r12+1], dl
    jmp .hex_write_space
.hex_blank:
    mov word ptr [r12], 0x2020
.hex_write_space:
    mov byte ptr [r12+2], 0x20
    cmp r13, 7
    je .hex_extra
    add r12, 3
    jmp .hex_next
.hex_extra:
    mov byte ptr [r12+3], 0x20
    add r12, 4
.hex_next:
    inc r13
    cmp r13, 16
    jne .hex_loop
    # separator
    mov byte ptr [r12], 0x20
    mov byte ptr [r12+1], '|'
    lea r14, [r12+2]
    xor r13, r13
.ascii_loop:
    cmp r13, r11
    jge .ascii_done
    movzx eax, byte ptr [r10 + r13]
    cmp al, 0x20
    jb .ascii_dot
    cmp al, 0x7E
    ja .ascii_dot
    mov byte ptr [r14 + r13], al
    jmp .ascii_next
.ascii_dot:
    mov byte ptr [r14 + r13], '.'
.ascii_next:
    inc r13
    jmp .ascii_loop
.ascii_done:
    mov byte ptr [r14 + r13], '|'
    mov byte ptr [r14 + r13 + 1], 0x0A
    mov rdx, r11
    add rdx, 63
    mov rdi, 1
    mov rsi, r8
    call sys_write
    add rsp, 128
    ret

_start:
    xor rbx, rbx
.read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    call sys_read
    test rax, rax
    jz .read_done
    add rbx, rax
    cmp rbx, 65536
    jne .read_loop
.read_done:
    xor r13, r13
.process_loop:
    cmp r13, rbx
    jge .process_done
    mov rdi, r13
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov rdx, rbx
    sub rdx, r13
    cmp rdx, 16
    jbe .call_print
    mov rdx, 16
.call_print:
    push r13
    call print_line
    pop r13
    add r13, 16
    jmp .process_loop
.process_done:
    sub rsp, 16
    mov r8, rsp
    mov rax, rbx
    mov ecx, 28
    lea r15, [rip + hex_digits]
.final_hex_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r8], dl
    inc r8
    sub ecx, 4
    jns .final_hex_loop
    mov byte ptr [r8], 0x0A
    mov rdi, 1
    mov rsi, rsp
    mov rdx, 9
    call sys_write
    add rsp, 16
    xor rdi, rdi
    call sys_exit
