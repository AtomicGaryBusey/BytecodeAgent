.globl _start

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INT64_MIN, -9223372036854775808
.equ MAX_INPUT, 65536
.equ STACK_DEPTH, 256

.section .text

_start:
    // Read all input into buffer
    mov x19, 0                      // total bytes read
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer

read_loop:
    mov x0, STDIN
    add x1, x20, x19
    mov x2, MAX_INPUT
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    // Process lines
    mov x21, 0                      // current offset
    mov x22, x19                    // total size

process_line_loop:
    cmp x21, x22
    b.ge exit_program

    // Find end of line
    mov x23, x21
find_newline:
    cmp x23, x22
    b.ge line_end_no_newline
    ldrb w24, [x20, x23]
    add x23, x23, 1
    cmp w24, 0x0A
    b.ne find_newline

    // Line from x21 to x23-1 (newline at x23-1)
    mov x0, x20
    mov x1, x21
    sub x2, x23, 1                  // exclude newline
    bl evaluate_line
    bl write_output

    mov x21, x23
    b process_line_loop

line_end_no_newline:
    // Last line without newline
    mov x0, x20
    mov x1, x21
    mov x2, x22
    bl evaluate_line
    bl write_output

exit_program:
    mov x0, 0
    mov x8, SYS_EXIT
    svc #0

// evaluate_line(x0=buf, x1=start, x2=end) -> x0=result, x1=error(0/1)
// Uses only caller-saved registers x0-x18.
evaluate_line:
    mov x3, x1          // current offset
    mov x4, 0           // stack depth
    adrp x5, rpn_stack
    add x5, x5, :lo12:rpn_stack
    mov x15, 0          // error flag

eval_loop:
    cmp x3, x2
    b.ge eval_done

skip_spaces:
    cmp x3, x2
    b.ge eval_done
    ldrb w9, [x0, x3]
    cmp w9, 0x20
    b.eq skip_next
    cmp w9, 0x09
    b.ne token_start
skip_next:
    add x3, x3, 1
    b skip_spaces

token_start:
    mov x6, x3
scan_token:
    cmp x3, x2
    b.ge token_end
    ldrb w9, [x0, x3]
    cmp w9, 0x20
    b.eq token_end
    cmp w9, 0x09
    b.eq token_end
    add x3, x3, 1
    b scan_token

token_end:
    mov x7, x3
    sub x8, x7, x6      // token length
    cbz x8, eval_loop   // empty token (should not happen)

    // Check for operator (length 1)
    cmp x8, 1
    b.ne parse_integer
    ldrb w9, [x0, x6]
    cmp w9, '+'
    b.eq op_add
    cmp w9, '-'
    b.eq op_sub
    cmp w9, '*'
    b.eq op_mul
    cmp w9, '/'
    b.eq op_div
    cmp w9, '%'
    b.eq op_mod
    b parse_integer

op_add:
op_sub:
op_mul:
op_div:
op_mod:
    cmp x4, 2
    b.lt eval_error
    sub x4, x4, 1
    ldr x13, [x5, x4, lsl #3]   // b
    sub x4, x4, 1
    ldr x12, [x5, x4, lsl #3]   // a

    cmp w9, '+'
    b.ne check_sub
    add x14, x12, x13
    b push_result
check_sub:
    cmp w9, '-'
    b.ne check_mul
    sub x14, x12, x13
    b push_result
check_mul:
    cmp w9, '*'
    b.ne check_div
    mul x14, x12, x13
    b push_result
check_div:
    cmp w9, '/'
    b.ne check_mod
    cbz x13, eval_error
    mov x10, INT64_MIN
    cmp x12, x10
    b.ne do_div
    cmp x13, -1
    b.eq eval_error
do_div:
    sdiv x14, x12, x13
    b push_result
check_mod:
    cbz x13, eval_error
    mov x10, INT64_MIN
    cmp x12, x10
    b.ne do_mod
    cmp x13, -1
    b.eq eval_error
do_mod:
    sdiv x10, x12, x13
    msub x14, x10, x13, x12
    b push_result

push_result:
    str x14, [x5, x4, lsl #3]
    add x4, x4, 1
    b eval_loop

parse_integer:
    mov x16, 1          // sign
    mov x10, x6         // token start
    mov x11, x8         // token length
    ldrb w9, [x0, x10]
    cmp w9, '-'
    b.ne check_digits
    mov x16, -1
    add x10, x10, 1
    sub x11, x11, 1
    cbz x11, eval_error // just '-'

check_digits:
    cmp x11, 18
    b.gt eval_error
    cbz x11, eval_error
    mov x14, 0
    mov x17, 10         // constant 10
parse_digit_loop:
    cbz x11, parse_done
    ldrb w9, [x0, x10]
    sub w9, w9, '0'
    cmp w9, 9
    b.gt eval_error     // not a digit (unsigned compare works because w9 is 0..255-'0')
    mul x14, x14, x17
    add x14, x14, x9
    add x10, x10, 1
    sub x11, x11, 1
    b parse_digit_loop

parse_done:
    mul x14, x14, x16   // apply sign
    str x14, [x5, x4, lsl #3]
    add x4, x4, 1
    b eval_loop

eval_error:
    mov x15, 1
    b eval_done

eval_done:
    cmp x15, 1
    b.eq return_error
    cmp x4, 1
    b.ne return_error
    ldr x0, [x5, 0]
    mov x1, 0
    ret

return_error:
    mov x0, 0
    mov x1, 1
    ret

// write_output(x0=value, x1=error_flag)
write_output:
    cbz x1, write_number
    // Write "error\n"
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, 6
    mov x0, STDOUT
    mov x8, SYS_WRITE
    svc #0
    ret

write_number:
    // Convert x0 to decimal string and write
    bl print_int64
    ret

// print_int64(x0=value) -> writes to stdout
print_int64:
    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer
    mov x2, 0                   // digit count
    cbz x0, write_zero
    mov x3, x0
    cmp x3, 0
    b.ge positive
    // negative
    mov x4, INT64_MIN
    cmp x3, x4
    b.ne not_min
    // Special case INT64_MIN
    adrp x5, int64_min_str
    add x5, x5, :lo12:int64_min_str
    mov x2, 19
    b write_string
not_min:
    neg x3, x3
    mov x4, 1                   // negative flag
    b convert
positive:
    mov x3, x0
    mov x4, 0
convert:
    mov x5, 10
convert_loop:
    cbz x3, convert_done
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add x7, x7, '0'
    strb w7, [x1, x2]
    add x2, x2, 1
    mov x3, x6
    b convert_loop
convert_done:
    cmp x4, 1
    b.ne reverse
    mov x6, '-'
    strb w6, [x1, x2]
    add x2, x2, 1
reverse:
    // Reverse the whole buffer (digits + optional sign at end)
    mov x5, 0
    sub x6, x2, 1
reverse_loop:
    cmp x5, x6
    b.ge write_string
    ldrb w7, [x1, x5]
    ldrb w8, [x1, x6]
    strb w8, [x1, x5]
    strb w7, [x1, x6]
    add x5, x5, 1
    sub x6, x6, 1
    b reverse_loop
write_string:
    // Add newline
    mov x6, 0x0A
    strb w6, [x1, x2]
    add x2, x2, 1
    mov x0, STDOUT
    mov x8, SYS_WRITE
    svc #0
    ret

write_zero:
    mov x6, '0'
    strb w6, [x1, 0]
    mov x6, 0x0A
    strb w6, [x1, 1]
    mov x0, STDOUT
    mov x2, 2
    mov x8, SYS_WRITE
    svc #0
    ret

.section .rodata
error_msg:
    .ascii "error\n"
int64_min_str:
    .ascii "9223372036854775808"

.section .bss
.align 3
rpn_stack:
    .zero STACK_DEPTH * 8

output_buffer:
    .zero 32

input_buffer:
    .zero MAX_INPUT
