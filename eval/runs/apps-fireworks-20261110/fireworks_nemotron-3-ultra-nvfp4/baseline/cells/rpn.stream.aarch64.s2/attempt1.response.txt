.globl _start

.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INT64_MIN, -9223372036854775808

.section .text
_start:
    // Read all input into buffer
    mov x19, 0                      // total bytes read
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer

read_loop:
    mov x0, STDIN
    add x1, x20, x19
    mov x2, 65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    // Process lines
    mov x21, 0                      // current offset
    mov x22, x19                    // total size

process_line_loop:
    cmp x21, x22
    b.ge exit_program

    // Find end of line
    mov x23, x21
find_newline:
    cmp x23, x22
    b.ge line_end_no_newline
    ldrb w24, [x20, x23]
    add x23, x23, 1
    cmp w24, 0x0A
    b.ne find_newline

    // Line from x21 to x23-1 (newline at x23-1)
    mov x0, x20
    mov x1, x21
    mov x2, x23
    bl evaluate_line
    bl write_output

    mov x21, x23
    b process_line_loop

line_end_no_newline:
    // Last line without newline
    mov x0, x20
    mov x1, x21
    mov x2, x22
    bl evaluate_line
    bl write_output

exit_program:
    mov x0, 0
    mov x8, SYS_EXIT
    svc #0

// evaluate_line(x0=buf, x1=start, x2=end) -> x0=result, x1=error(0/1)
// Uses only caller-saved registers x0-x18.
evaluate_line:
    mov x3, x1          // current offset
    mov x4, 0           // stack depth
    adrp x5, rpn_stack
    add x5, x5, :lo12:rpn_stack
    mov x15, 0          // error flag

eval_loop:
    cmp x3, x2
    b.ge eval_done

skip_spaces:
    cmp x3, x2
    b.ge eval_done
    ldrb w9, [x0, x3]
    cmp w9, 0x20
    b.eq skip_next
    cmp w9, 0x09
    b.ne token_start
skip_next:
    add x3, x3, 1
    b skip_spaces

token_start:
    mov x6, x3
scan_token:
    cmp x3, x2
    b.ge token_end
    ldrb w9, [x0, x3]
    cmp w9, 0x20
    b.eq token_end
    cmp w9, 0x09
    b.eq token_end
    add x3, x3, 1
    b scan_token

token_end:
    mov x7, x3
    sub x8, x7, x6
    cbz x8, eval_loop

    // Check for operator (length 1)
    cmp x8, 1
    b.ne parse_integer
    ldrb w9, [x0, x6]
    cmp w9, '+'
    b.eq op_add
    cmp w9, '-'
    b.eq op_sub
    cmp w9, '*'
    b.eq op_mul
    cmp w9, '/'
    b.eq op_div
    cmp w9, '%'
    b.eq op_mod
    b parse_integer

op_add:
op_sub:
op_mul:
op_div:
op_mod:
    cmp x4, 2
    b.lt eval_error
    sub x4, x4, 1
    ldr x13, [x5, x4, lsl #3]   // b
    sub x4, x4, 1
    ldr x12, [x5, x4, lsl #3]   // a

    cmp w9, '+'
    b.ne check_sub
    add x14, x12, x13
    b push_result
check_sub:
    cmp w9, '-'
    b.ne check_mul
    sub x14, x12, x13
    b push_result
check_mul:
    cmp w9, '*'
    b.ne check_div
    mul x14, x12, x13
    b push_result
check_div:
    cmp w9, '/'
    b.ne check_mod
    cbz x13, eval_error
    mov x10, INT64_MIN
    cmp x12, x10
    b.ne do_div
    cmp x13, -1
    b.eq eval_error
do_div:
    sdiv x14, x12, x13
    b push_result
check_mod:
    cbz x13, eval_error
    mov x10, INT64_MIN
    cmp x12, x10
    b.ne do_mod
    cmp x13, -1
    b.eq eval_error
do_mod:
    sdiv x10, x12, x13
    msub x14, x10, x13, x12
    b push_result

push_result:
    str x14, [x5, x4, lsl #3]
    add x4, x4, 1
    b eval_loop

parse_integer:
    mov x16, 1          // sign
    mov x10, x6
    ldrb w9, [x0, x10]
    cmp w9, '-'
    b.ne parse_digits
    mov x16, -1
    add x10, x10, 1
    sub x8, x8, 1
    cbz x8, eval_error

parse_digits:
    cmp x8, 18
    b.gt eval_error
    cbz x8, eval_error
    mov x14, 0
parse_digit_loop:
    cbz x8, parse_done
    ldrb w9, [x0, x10]
    sub w9, w9, '0'
    cmp w9, 9
    b.gt eval_error
    mul x14, x14, 10
