.section .rodata
base64_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4096
input_buf: .space 49152
output_buf: .space 66500

.section .text
.globl _start
_start:
    # read all input
    xor rbx, rbx                # total_read = 0
.Lread_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = 0
    lea rsi, [input_buf + rbx]
    mov rdx, 49152
    sub rdx, rbx
    syscall
    test rax, rax
    jz .Lread_done
    add rbx, rax
    cmp rbx, 49152
    jne .Lread_loop
.Lread_done:

    # base64 encode
    mov rsi, input_buf
    mov rdi, output_buf
    xor rcx, rcx                # line_char_count = 0
    mov r8, rbx                 # remaining bytes
.Lfull_loop:
    cmp r8, 3
    jb .Lremainder
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx r9d, byte ptr [rsi+2]
    add rsi, 3
    sub r8, 3

    # idx0
    mov edx, eax
    shr edx, 2
    mov al, byte ptr [base64_table + rdx]
    call put_char

    # idx1
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov r10d, ebx
    shr r10d, 4
    or edx, r10d
    mov al, byte ptr [base64_table + rdx]
    call put_char

    # idx2
    mov edx, ebx
    and edx, 0xF
    shl edx, 2
    mov r10d, r9d
    shr r10d, 6
    or edx, r10d
    mov al, byte ptr [base64_table + rdx]
    call put_char

    # idx3
    mov edx, r9d
    and edx, 0x3F
    mov al, byte ptr [base64_table + rdx]
    call put_char

    jmp .Lfull_loop

.Lremainder:
    test r8, r8
    jz .Lfinish
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]

    # idx0
    mov edx, eax
    shr edx, 2
    mov al, byte ptr [base64_table + rdx]
    call put_char

    # idx1
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov r10d, ebx
    shr r10d, 4
    or edx, r10d
    mov al, byte ptr [base64_table + rdx]
    call put_char

    cmp r8, 1
    je .Lone_byte

    # two bytes remainder: idx2 and padding '='
    mov edx, ebx
    and edx, 0xF
    shl edx, 2
    mov al, byte ptr [base64_table + rdx]
    call put_char
    mov al, '='
    call put_char
    jmp .Lfinish

.Lone_byte:
    # one byte remainder: two padding '='
    mov al, '='
    call put_char
    mov al, '='
    call put_char

.Lfinish:
    test rcx, rcx
    jz .Lwrite
    mov byte ptr [rdi], 0x0A
    inc rdi

.Lwrite:
    mov rsi, output_buf
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz .Lexit
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = 1
    syscall

.Lexit:
    mov rax, 60                 # sys_exit
    xor rdi, rdi
    syscall

put_char:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    cmp rcx, 76
    jne .Lput_ret
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rcx, rcx
.Lput_ret:
    ret
