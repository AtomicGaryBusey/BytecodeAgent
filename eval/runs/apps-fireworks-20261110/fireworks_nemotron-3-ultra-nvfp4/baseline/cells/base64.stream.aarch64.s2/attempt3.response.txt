.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ READ_SIZE, 4096
.equ OUT_BUF_SIZE, 8192

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 8
input_buffer: .space READ_SIZE
output_buffer: .space OUT_BUF_SIZE

.section .text
.globl _start
_start:
    // Load addresses of static data
    adrp x19, output_buffer
    add x19, x19, :lo12:output_buffer
    adrp x20, output_buffer
    add x20, x20, :lo12:output_buffer
    add x20, x20, #OUT_BUF_SIZE        // x20 = output_buffer_end
    adrp x21, base64_table
    add x21, x21, :lo12:base64_table
    adrp x28, input_buffer
    add x28, x28, :lo12:input_buffer

    // Initialize state registers
    sub x22, x20, x19                  // buffer_limit = OUT_BUF_SIZE
    mov x23, #0                        // output_pos = 0
    mov x24, #0                        // line_pos = 0
    mov w25, #0                        // leftover_count = 0
    // w26, w27 for leftover bytes (uninitialized)

read_loop:
    // read from stdin
    mov x8, #SYS_READ
    mov x0, #STDIN
    mov x1, x28                        // input_buffer
    mov x2, #READ_SIZE
    svc #0
    cbz x0, end_of_input               // EOF

    // x0 = bytes read
    mov x1, x28                        // current pointer
    add x2, x1, x0                     // end pointer

    // Handle leftovers
    cbnz w25, have_leftovers
    b process_full_groups

have_leftovers:
    cmp w25, #1
    b.eq have_one_leftover
    // two leftovers
    cmp x1, x2
    b.hs not_enough_for_three_2
    ldrb w0, [x1], #1
    mov w1, w26
    mov w2, w27
    bl encode_three
    b process_full_groups

have_one_leftover:
    cmp x1, x2
    b.lo not_enough_for_three_1
    ldrb w1, [x1], #1
