.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ READ_SIZE, 4096
.equ OUT_BUF_SIZE, 8192

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 8
input_buffer: .space READ_SIZE
output_buffer: .space OUT_BUF_SIZE

.section .text
.globl _start

// flush_output: write output_buffer to stdout, reset output_pos
// Clobbers: x0-x8, x30
flush_output:
    cbz x23, 1f                 // if output_pos == 0, nothing to do
    mov x8, #SYS_WRITE
    mov x0, #STDOUT
    mov x1, x19                 // output_buffer base
    mov x2, x23                 // output_pos
    svc #0
    mov x23, #0                 // reset output_pos
1:
    ret

// emit_newline: output a newline, handle buffer full
// Clobbers: x0-x8, x30
emit_newline:
    cmp x23, x22                // output_pos == buffer_limit?
    b.ne 1f
    bl flush_output
1:
    mov w0, #0x0A               // '\n'
    strb w0, [x19, x23]
    add x23, x23, #1
    mov x24, #0                 // line_pos = 0
    ret

// emit_char(w0): output a character, handle line length and buffer full
// Clobbers: x0-x8, x30
emit_char:
    cmp x24, #76                // line_pos == 76?
    b.ne 1f
    bl emit_newline
1:
    cmp x23, x22                // output_pos == buffer_limit?
    b.ne 2f
    bl flush_output
2:
    strb w0, [x19, x23]
    add x23, x23, #1
    add x24, x24, #1
    ret

// encode_three(w0,w1,w2): encode three bytes, call emit_char four times
// w0,w1,w2 are the three input bytes (0-255)
// Clobbers: x0-x8, x30
encode_three:
    // byte0 = w0, byte1 = w1, byte2 = w2
    // char0 = byte0 >> 2
    ubfx w3, w0, #2, #6         // w3 = (w0 >> 2) & 0x3F
    ldrb w3, [x21, w3]
