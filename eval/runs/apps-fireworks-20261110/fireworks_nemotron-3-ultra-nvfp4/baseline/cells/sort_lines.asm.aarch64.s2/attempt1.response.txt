.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94
.equ STDIN, 0
.equ STDOUT, 1
.equ STDERR, 2
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 8
input_buf: .space MAX_INPUT
line_ptrs: .space MAX_LINES * 8
line_lens: .space MAX_LINES * 4
output_buf: .space MAX_INPUT + MAX_LINES
num_lines: .space 8
total_read: .space 8

.section .text
.globl _start
_start:
    // Read all input
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #MAX_INPUT
1:
    mov x0, #STDIN
    add x1, x19, x20
    mov x2, x21
    mov x8, #SYS_READ
    svc #0
    cbz x0, 2f
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, 1b
2:
    // Parse lines
    adrp x22, line_ptrs
    add x22, x22, :lo12:line_ptrs
    adrp x23, line_lens
    add x23, x23, :lo12:line_lens
    mov x24, #0
    mov x25, #0
    mov x26, #0
3:
    cmp x26, x20
    b.ge 4f
    ldrb w27, [x19, x26]
    cmp w27, #0x0A
    b.ne 5f
    sub x27, x26, x25
    cmp x24, #MAX_LINES
    b.ge 6f
    add x28, x19, x25
    str x28, [x22, x24, lsl #3]
    str w27, [x23, x24, lsl #2]
    add x24, x24, #1
6:
    add x25, x26, #1
5:
    add x26, x26, #1
    b 3b
4:
    cmp x25, x20
    b.ge 7f
    sub x27, x20, x25
    cmp x24, #MAX_LINES
    b.ge 7f
    add x28, x19, x25
    str x28, [x22, x24, lsl #3]
    str w27, [x23, x24, lsl #2]
    add x24, x24, #1
7:
    // Sort lines (insertion sort)
    cmp x24, #1
    b.le 9f
    mov x25, #1
8:
    cmp x25, x24
    b.ge 9f
    ldr x26, [x22, x25, lsl #3]
    ldr w27, [x23, x25, lsl #2]
    sub x28, x25, #1
10:
    cmp x28, #0
    b.lt 11f
    ldr x0, [x22, x28, lsl #3]
    ldr w1, [x23, x28, lsl #2]
    mov x2, x26
    mov x3, x27
    bl compare_lines
    cmp x0, #0
    b.le 11f
    ldr x0, [x22, x28, lsl #3]
    ldr w1, [x23, x28, lsl #2]
    add x4, x28, #1
    str x0, [x22, x4, lsl #3]
    str w1, [x23, x4, lsl #2]
    sub x28, x28, #1
    b 10b
11:
    add x4, x28, #1
    str x26, [x22, x4, lsl #3]
    str w27, [x23, x4, lsl #2]
    add x25, x25, #1
    b 8b
9:
    // Build output buffer
    adrp x19, output_buf
    add x19, x19, :lo12:output_buf
    mov x20, #0
    mov x21, #0
12:
    cmp x21, x24
    b.ge 13f
    ldr x0, [x22, x21, lsl #3]
    ldr w5, [x23, x21, lsl #2]
    cbz w5, 14f
    mov x2, x0
    add x3, x19, x20
    mov w4, w5
15:
    ldrb w6, [x2], #1
    strb w6, [x3], #1
    subs w4, w4, #1
    b.ne 15b
    add x20, x20, x5
14:
    mov w6, #0x0A
    strb w6, [x19, x20]
    add x20, x20, #1
    add x21, x21, #1
    b 12b
13:
    // Write output
    mov x0, #STDOUT
    mov x1, x19
    mov x2, x20
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, x20
    b.eq 16f
    add x19, x19, x0
    sub x20, x20, x0
    b 13b
16:
    // Exit
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

compare_lines:
    mov x4, x0
    mov x5, x1
    mov x6, x2
    mov x7, x3
    cmp x5, x7
    csel x8, x5, x7, lo
    cbz x8, .Lcmp_len
.Lcmp_bytes:
    ldrb w9, [x4], #1
    ldrb w10, [x6], #1
    cmp w9, w10
    b.ne .Lcmp_byte_diff
    subs x8, x8, #1
    b.ne .Lcmp_bytes
.Lcmp_len:
    cmp x5, x7
    cset x0, gt
    cset x1, lt
    sub x0, x0, x1
    ret
.Lcmp_byte_diff:
    cmp w9, w10
    cset x0, gt
    cset x1, lt
    sub x0, x0, x1
    ret
