.section .rodata
newline_char: .byte 10

.section .text
.global _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ BUFSIZE, 65536

_start:
    sub rsp, BUFSIZE
    mov r12, rsp          # buffer start
    xor r13, r13          # bytes read so far
    mov r14, BUFSIZE

read_loop:
    mov rax, SYS_READ
    mov rdi, STDIN
    lea rsi, [r12 + r13]
    mov rdx, r14
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    cmp r13, r14
    jne read_loop

read_done:
    mov rsi, r12          # current position
    lea r15, [r12 + r13]  # end of buffer (exclusive)

process_loop:
    cmp rsi, r15
    jae process_done

    mov r8, rsi           # line start

find_newline:
    cmp rsi, r15
    jae no_newline_found
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne find_newline

    # found newline at rsi-1
    dec rsi               # rsi points to newline
    mov r9, r8            # reverse start
    mov r10, rsi
    dec r10               # reverse end (before newline)
    call reverse_in_place

    mov rdx, rsi
    sub rdx, r8
    inc rdx               # length including newline
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    mov rsi, r8
    syscall

    inc rsi               # move past newline
    jmp process_loop

no_newline_found:
    # r8 is line start, r15 is buffer end
    mov r9, r8
    mov r10, r15
    dec r10               # last byte
    call reverse_in_place

    mov rdx, r15
    sub rdx, r8           # line length
    test rdx, rdx
    jz write_final_newline
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    mov rsi, r8
    syscall

write_final_newline:
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall

process_done:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

reverse_in_place:
    # r9 = start, r10 = end (inclusive)
    cmp r9, r10
    jge reverse_done
reverse_loop:
    mov r11b, byte ptr [r9]
    mov al, byte ptr [r10]
    mov byte ptr [r9], al
    mov byte ptr [r10], r11b
    inc r9
    dec r10
    cmp r9, r10
    jl reverse_loop
reverse_done:
    ret
