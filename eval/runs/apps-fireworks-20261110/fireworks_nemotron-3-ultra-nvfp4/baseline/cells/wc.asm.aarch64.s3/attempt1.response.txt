.section .text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // Initialize counters
    mov x19, #0      // lines
    mov x20, #0      // words
    mov x21, #0      // bytes
    mov x22, #0      // in_word flag

    b read_loop

read_loop:
    // read from stdin
    mov x0, #0       // fd = 0
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov x2, #4096    // buffer size
    mov x8, #SYS_READ
    svc #0
    // x0 = bytes read
    cbz x0, read_done
    add x21, x21, x0 // byte_count += bytes_read

    // process buffer
    mov x3, x1       // current pointer
    add x4, x1, x0   // end pointer

process_loop:
    cmp x3, x4
    b.eq read_loop   // processed all bytes, read next chunk

    ldrb w5, [x3], #1

    // Check for newline (0x0A)
    cmp w5, #0x0A
    b.eq handle_newline

    // Check for other whitespace: space (0x20), tab (0x09), vt (0x0B), ff (0x0C), cr (0x0D)
    cmp w5, #0x20
    b.eq handle_whitespace
    cmp w5, #0x09
    b.eq handle_whitespace
    cmp w5, #0x0B
    b.eq handle_whitespace
    cmp w5, #0x0C
    b.eq handle_whitespace
    cmp w5, #0x0D
    b.eq handle_whitespace

    // Not whitespace
    mov x22, #1
    b process_loop

handle_whitespace:
    cbnz x22, end_word
    b process_loop

end_word:
    add x20, x20, #1
    mov x22, #0
    b process_loop

handle_newline:
    add x19, x19, #1
    cbnz x22, end_word_nl
    b process_loop

end_word_nl:
    add x20, x20, #1
    mov x22, #0
    b process_loop

read_done:
    // EOF reached, check if we were in a word
    cbz x22, output_results
    add x20, x20, #1

output_results:
    // Prepare output buffer
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf

    // Append line count
    mov x0, x19
    mov x1, x23
    bl append_number
    mov x23, x0
    // space
    mov w0, #0x20
    strb w0, [x23], #1

    // Append word count
    mov x0, x20
    mov x1, x23
    bl append_number
    mov x23, x0
    // space
    mov w0, #0x20
    strb w0, [x23], #1

    // Append byte count
    mov x0, x21
    mov x1, x23
    bl append_number
    mov x23, x0
    // newline
    mov w0, #0x0A
    strb w0, [x23], #1

    // Write output
    mov x0, #1       // stdout
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x23, x1  // length
    mov x8, #SYS_WRITE
    svc #0

    // Exit
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// append_number: x0 = number, x1 = dest pointer, returns new dest pointer in x0
append_number:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32   // temp buffer (32 bytes)

    mov x2, sp
    add x2, x2, #32   // x2 = end of temp buffer
    mov x3, #10

    cbz x0, write_zero

convert_loop:
    udiv x4, x0, x3
    msub x5, x4, x3, x0   // remainder
    add x5, x5, #0x30     // '0'
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    cbnz x0, convert_loop

    // x2 points to start of digits, length = (sp+32) - x2
    mov x4, sp
    add x4, x4, #32
    sub x4, x4, x2   // length
    b copy_digits

write_zero:
    mov w5, #0x30
    sub x2, x2, #1
    strb w5, [x2]
    mov x4, #1

copy_digits:
    cbz x4, copy_done
copy_loop:
    ldrb w5, [x2], #1
    strb w5, [x1], #1
    subs x4, x4, #1
    b.ne copy_loop

copy_done:
    mov x0, x1   // return new dest pointer
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret

.section .bss
.align 12
buffer:
    .space 4096

.align 4
output_buf:
    .space 64
