.section .bss
.align 16
input_buf: .space 65536
output_buf: .space 65536

.section .rodata
newline_byte: .byte 0x0A

.section .text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    adrp x24, newline_byte
    add x24, x24, :lo12:newline_byte

    mov x20, 0

read_loop:
    mov x8, SYS_READ
    mov x0, 0
    add x1, x19, x20
    mov x2, BUF_SIZE
    sub x2, x2, x20
    svc 0
    cbz x0, read_done
    add x20, x20, x0
    cmp x20, BUF_SIZE
    b.ne read_loop

read_done:
    cbz x20, exit_success

    add x22, x19, x20
    mov x21, x19

process_loop:
    cmp x21, x22
    b.hs exit_success

    mov x25, x21
find_newline:
    cmp x25, x22
    b.hs no_newline_found
    ldrb w26, [x25]
    cmp w26, 0x0A
    b.eq newline_found
    add x25, x25, 1
    b find_newline

newline_found:
    sub x26, x25, x21
    cbz x26, write_newline_only
    mov x27, x23
    add x28, x21, x26
    sub x28, x28, 1
    mov x29, x26
reverse_loop:
    ldrb w30, [x28]
    strb w30, [x27], 1
    sub x28, x28, 1
    subs x29, x29, 1
    b.ne reverse_loop
    mov x8, SYS_WRITE
    mov x0, 1
    mov x1, x23
    mov x2, x26
    svc 0

write_newline_only:
    mov x8, SYS_WRITE
    mov x0, 1
    mov x1, x24
    mov x2, 1
    svc 0
    add x21, x25, 1
    b process_loop

no_newline_found:
    sub x26, x22, x21
    cbz x26, exit_success
    mov x27, x23
    add x28, x21, x26
    sub x28, x28, 1
    mov x29, x26
reverse_loop2:
    ldrb w30, [x28]
    strb w30, [x27], 1
    sub x28, x28, 1
    subs x29, x29, 1
    b.ne reverse_loop2
    mov x8, SYS_WRITE
    mov x0, 1
    mov x1, x23
    mov x2, x26
    svc 0
    mov x8, SYS_WRITE
    mov x0, 1
    mov x1, x24
    mov x2, 1
    svc 0

exit_success:
    mov x8, SYS_EXIT
    mov x0, 0
    svc 0
