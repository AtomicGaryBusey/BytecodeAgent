.section .bss
.align 12
input_buf:  .zero 65536
output_buf: .zero 65537

.section .text
.globl _start

_start:
    // Load buffer addresses
    adr x19, input_buf
    adr x20, output_buf

    // Read all input into input_buf
    mov x21, 0               // total_read = 0
    mov x22, 65536           // max_read

read_loop:
    mov x0, 0                // fd = stdin
    add x1, x19, x21         // buf = input_buf + total_read
    sub x2, x22, x21         // count = max_read - total_read
    mov x8, 63               // sys_read
    svc 0
    cbz x0, read_done        // EOF if 0
    add x21, x21, x0         // total_read += bytes_read
    cmp x21, x22
    b.lt read_loop           // continue if buffer not full

read_done:
    // Process input buffer: reverse each line, write to output_buf
    mov x23, x19             // p = start of input
    add x24, x19, x21        // limit = start + total_read
    mov x25, x20             // out_ptr = output_buf

process_loop:
    cmp x23, x24
    b.ge write_output        // done if p >= limit

    mov x26, x23             // line_start = p

find_newline:
    cmp x23, x24
    b.ge newline_not_found   // reached end without newline
    ldrb w27, [x23]          // load byte
    add x23, x23, 1          // advance p
    cmp w27, 10              // '\n'?
    b.ne find_newline

    // Found newline at x23-1; line is [x26, x23-1)
    sub x27, x23, x26        // x27 = (x23 - x26) - 1 = line_len
    sub x27, x27, 1
    cbz x27, add_newline     // zero-length line

    // Reverse copy: src = line_start + line_len - 1
    add x28, x26, x27
    sub x28, x28, 1
1:
    ldrb w29, [x28]
    strb w29, [x25], 1
    sub x28, x28, 1
    subs x27, x27, 1
    b.ne 1b

add_newline:
    mov w29, 10
    strb w29, [x25], 1
    b process_loop

newline_not_found:
    // Last line (no terminating newline): [x26, x24)
    sub x27, x24, x26        // line_len
    cbz x27, write_output    // empty remainder -> not a line
    add x28, x26, x27
    sub x28, x28, 1
2:
    ldrb w29, [x28]
    strb w29, [x25], 1
    sub x28, x28, 1
    subs x27, x27, 1
    b.ne 2b
    mov w29, 10
    strb w29, [x25], 1

write_output:
    sub x2, x25, x20         // output length
    cbz x2, exit_program     // nothing to write

    mov x0, 1                // fd = stdout
    mov x1, x20              // buf = output_buf

write_loop:
    mov x8, 64               // sys_write
    svc 0
    cmp x0, 0
    b.le exit_error          // error
    sub x2, x2, x0
    cbz x2, exit_program
    add x1, x1, x0
    b write_loop

exit_program:
    mov x0, 0
    mov x8, 93               // sys_exit
    svc 0
    b exit_program           // should not return

exit_error:
    mov x0, 1
    mov x8, 93
    svc 0
    b exit_error
