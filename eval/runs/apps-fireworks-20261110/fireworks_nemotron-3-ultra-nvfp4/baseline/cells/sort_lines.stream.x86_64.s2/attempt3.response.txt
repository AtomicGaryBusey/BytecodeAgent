.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 4096
input_buffer:
    .zero MAX_INPUT
line_array:
    .zero MAX_LINES * 16

.section .rodata
newline_byte:
    .byte 10

.section .text

_start:
    # ----- Read all input -----
    lea r15, [rip + input_buffer]   # r15 = buffer base
    xor r12, r12                    # r12 = total bytes read

read_loop:
    mov rax, SYS_READ
    mov rdi, STDIN
    lea rsi, [r15 + r12]
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jne read_loop

read_done:
    # ----- Parse lines -----
    test r12, r12
    jz no_lines

    xor r13, r13                    # r13 = line count
    lea r14, [rip + line_array]     # r14 = line_array base (write pointer)
    mov rsi, r15                    # current position
    mov rdi, r15                    # start of current line
    mov rcx, r12                    # remaining bytes

parse_loop:
    test rcx, rcx
    jz parse_done
    cmp byte ptr [rsi], 10
    je found_newline
    inc rsi
    dec rcx
    jmp parse_loop

found_newline:
    mov rax, rsi
    sub rax, rdi                    # rax = line length
    mov qword ptr [r14], rdi
    mov qword ptr [r14 + 8], rax
    add r14, 16
    inc r13
    inc rsi
    dec rcx
    mov rdi, rsi
    jmp parse_loop

parse_done:
    lea rax, [r15 + r12]            # buffer end
    cmp rdi, rax
    jae no_final_line
    mov rcx, rax
    sub rcx, rdi
    mov qword ptr [r14], rdi
    mov qword ptr [r14 + 8], rcx
    inc r13

no_final_line:
no_lines:
    # ----- Sort lines (insertion sort) -----
    test r13, r13
    jz write_output

    lea r14, [rip + line_array]     # base of line array
    mov rax, r13
    shl rax, 4                      # multiply by 16
    lea r12, [r14 + rax]            # r12 = end pointer (base + count*16)
    lea rdi, [r14 + 16]             # rdi = pointer to element i (starting at i=1)

outer_loop:
    cmp rdi, r12
    jge write_output

    mov r8, qword ptr [rdi]         # current line pointer
    mov r9, qword ptr [rdi + 8]     # current line length
    lea rsi, [rdi - 16]             # rsi = pointer to previous element

inner_loop:
    cmp rsi, r14
    jl insert_done

    mov r10, qword ptr [rsi]        # line1 pointer
    mov r11, qword ptr [rsi + 8]    # line1 length
    xor rax, rax
    xor rcx, rcx

1:  cmp rcx, r11
    jge 2f
    cmp rcx, r9
    jge 3f
    movzx rdx, byte ptr [r10 + rcx]
    cmp dl, byte ptr [r8 + rcx]
    jl 4f
    jg 5f
    inc rcx
    jmp 1b

2:  cmp rcx, r9
    jge 6f
    mov rax, -1
    jmp 7f

3:  mov rax, 1
    jmp 7f

4:  mov rax, -1
    jmp 7f

5:  mov rax, 1
    jmp 7f

6:  xor rax, rax

7:  cmp rax, 0
    jle insert_done

    # move line1 one position forward
    mov qword ptr [rsi + 16], r10
    mov qword ptr [rsi + 24], r11
    sub rsi, 16
    jmp inner_loop

insert_done:
    mov qword ptr [rsi + 16], r8
    mov qword ptr [rsi + 24], r9
    add rdi, 16
    jmp outer_loop

write_output:
    mov rbx, r14                    # rbx = current line struct pointer

write_loop:
    cmp rbx, r12
    jge exit_program

    mov rsi, qword ptr [rbx]        # line data pointer
    mov rdx, qword ptr [rbx + 8]    # line length
    call write_all

    # write newline
    lea rsi, [rip + newline_byte]
    mov rdx, 1
    call write_all

    add rbx, 16
    jmp write_loop

write_all:
    # rsi = buffer, rdx = count
    # preserves rbx, r12, r13, r14, r15
    push rdi
    push rax
    push rcx
    push r11
1:  test rdx, rdx
    jz 2f
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    syscall
    cmp rax, -1
    je 2f
    sub rdx, rax
    add rsi, rax
    jmp 1b
2:  pop r11
    pop rcx
    pop rax
    pop rdi
    ret

exit_program:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
