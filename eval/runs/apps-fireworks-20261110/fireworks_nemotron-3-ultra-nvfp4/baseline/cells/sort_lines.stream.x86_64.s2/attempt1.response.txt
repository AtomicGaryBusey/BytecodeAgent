.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 4096
input_buffer:
    .zero MAX_INPUT
line_array:
    .zero MAX_LINES * 16
sort_stack:
    .zero MAX_LINES * 16
pivot_storage:
    .zero 16

.section .rodata
newline_byte:
    .byte 10

.section .text

_start:
    # ----- Read all input -----
    lea r15, [rip + input_buffer]   # r15 = buffer base
    xor r12, r12                    # r12 = total bytes read

read_loop:
    mov rax, SYS_READ
    mov rdi, STDIN
    lea rsi, [r15 + r12]
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jne read_loop

read_done:
    # ----- Parse lines -----
    test r12, r12
    jz no_lines

    xor r13, r13            # r13 = line count
    lea r14, [rip + line_array]  # r14 = line_array base
    mov rsi, r15            # current position
    mov rdi, r15            # start of current line
    mov rcx, r12            # remaining bytes

parse_loop:
    test rcx, rcx
    jz parse_done
    cmp byte ptr [rsi], 10
    je found_newline
    inc rsi
    dec rcx
    jmp parse_loop

found_newline:
    mov rax, rsi
    sub rax, rdi            # rax = line length
    mov [r14], rdi
    mov [r14 + 8], rax
    add r14, 16
    inc r13
    inc rsi
    dec rcx
    mov rdi, rsi
    jmp parse_loop

parse_done:
    lea rax, [r15 + r12]    # buffer end
    cmp rdi, rax
    jae no_final_line
    mov rcx, rax
    sub rcx, rdi
    mov [r14], rdi
    mov [r14 + 8], rcx
    inc r13

no_final_line:
no_lines:
    # ----- Sort lines -----
    test r13, r13
    jz write_output

    # Initialize sort stack
    lea r15, [rip + sort_stack]
    mov qword [r15], 0
