.section .bss
.align 12
input_buf:
    .zero 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 65536
.equ READ_CHUNK, 4096

_start:
    // Initialize input buffer pointer
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, 0                 // total bytes read
    mov  x21, BUF_SIZE          // buffer capacity
    mov  x23, READ_CHUNK        // read chunk size in register

read_loop:
    // Calculate remaining space
    sub  x22, x21, x20
    cmp  x22, x23
    csel x22, x22, x23, hi      // x22 = min(remaining, READ_CHUNK)
    cbz  x22, read_done         // buffer full

    // read(syscall 63): x0=fd, x1=buf, x2=count
    mov  x0, STDIN
    add  x1, x19, x20           // buffer + offset
    mov  x8, SYS_READ
    svc  #0

    cbz  x0, read_done          // EOF (0 bytes read)
    add  x20, x20, x0           // total += bytes read
    b    read_loop

read_done:
    // x20 = total bytes read
    // Process buffer in 16-byte rows
    mov  x21, 0                 // current offset / index

process_loop:
    cmp  x21, x20
    b.ge print_final_offset

    // Print offset (8 hex digits)
    mov  x0, x21
    bl   print_hex8
    // Print two spaces
    mov  x0, ' '
    bl   putchar
    mov  x0, ' '
    bl   putchar

    // Print hex bytes (16 positions)
    mov  x22, 0                 // i = 0
hex_byte_loop:
    cmp  x22, 16
    b.ge hex_done

    add  x24, x21, x22          // buffer index = offset + i
    cmp  x24, x20
    b.ge print_hex_space        // beyond data, print spaces

    // Load byte
    ldrb w24, [x19, x24]
    // Print as two hex digits
    mov  x0, x24
    and  x0, x0, 0xFF
    bl   print_hex2
    b    hex_next

print_hex_space:
    // Print three spaces
    mov  x0, ' '
    bl   putchar
    mov  x0, ' '
    bl   putchar
    mov  x0, ' '
    bl   putchar

hex_next:
    add  x22, x22, 1
    cmp  x22, 8
    b.ne hex_byte_loop
    // After 8th byte, print extra space
    mov  x0, ' '
    bl   putchar
    b    hex_byte_loop

hex_done:
    // Print space, '|'
    mov  x0, ' '
    bl   putchar
    mov  x0, '|'
    bl   putchar

    // Print ASCII column
    mov  x22, 0
ascii_loop:
    cmp  x22, 16
    b.ge ascii_done
    add  x24, x21, x22
    cmp  x24, x20
    b.ge ascii_done

    ldrb w24, [x19, x24]
    cmp  w24, 0x20
    b.lt print_dot
    cmp  w24, 0x7E
    b.gt print_dot
    // Printable
    mov  x0, x24
    bl   putchar
    b    ascii_next

print_dot:
    mov  x0, '.'
    bl   putchar

ascii_next:
    add  x22, x22, 1
    b    ascii_loop

ascii_done:
    // Print '|' and newline
    mov  x0, '|'
    bl   putchar
    mov  x0, '\n'
    bl   putchar

    // Next row
    add  x21, x21, 16
    b    process_loop

print_final_offset:
    // Print total bytes as 8 hex digits + newline
    mov  x0, x20
    bl   print_hex8
    mov  x0, '\n'
    bl   putchar

    // Exit
    mov  x0, 0
    mov  x8, SYS_EXIT
    svc  #0

// print_hex8: print x0 as 8 lowercase hex digits
// x0 = value (assumed to fit in 32 bits)
print_hex8:
    // Print 8 nibbles from high to low
    mov  x1, 28
hex8_loop:
    lsr  x2, x0, x1
    and  x2, x2, 0xF
    adrp x3, hex_digits
    add  x3, x3, :lo12:hex_digits
    ldrb w2, [x3, x2]
    mov  x0, x2
    bl   putchar
    subs x1, x1, 4
    b.pl hex8_loop
    ret

// print_hex2: print x0 (low 8 bits) as 2 lowercase hex digits
print_hex2:
    // High nibble
    lsr  x1, x0, 4
    and  x1, x1, 0xF
    adrp x2, hex_digits
    add  x2, x2, :lo12:hex_digits
    ldrb w1, [x2, x1]
    mov  x0, x1
    bl   putchar
    // Low nibble
    and  x1, x0, 0xF
    adrp x2, hex_digits
    add  x2, x2, :lo12:hex_digits
    ldrb w1, [x2, x1]
    mov  x0, x1
    bl   putchar
    ret

// putchar: write single character in x0 (low 8 bits) to stdout
// Uses a small stack buffer
putchar:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp
    strb w0, [sp, #-1]!
    mov  x0, STDOUT
    mov  x1, sp
    mov  x2, 1
    mov  x8, SYS_WRITE
    svc  #0
    ldrb w0, [sp], #1
    ldp  x29, x30, [sp], #16
    ret
