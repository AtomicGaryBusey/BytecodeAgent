.section .bss
.align 12
input_buf:
    .zero 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 65536
.equ READ_CHUNK, 4096

_start:
    // Initialize input buffer pointer
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, 0                 // total bytes read
    mov  x21, BUF_SIZE          // buffer capacity

read_loop:
    // Calculate remaining space
    sub  x22, x21, x20
    cmp  x22, READ_CHUNK
    csel x22, x22, READ_CHUNK, hi   // x22 = min(remaining, READ_CHUNK)
