.section .bss
.align 12
input_buf:
    .zero 65536

.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 65536
.equ READ_CHUNK, 4096

_start:
    // Read all input into input_buf
    mov x19, input_buf          // current buffer position
