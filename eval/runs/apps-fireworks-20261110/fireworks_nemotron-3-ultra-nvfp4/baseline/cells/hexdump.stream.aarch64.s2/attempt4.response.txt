.section .bss
.align 12
input_buf:
    .zero 65536
output_buf:
    .zero 983040

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
.global _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 983040
.equ READ_CHUNK, 4096

_start:
    // Initialize pointers
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add  x21, x21, :lo12:output_buf
    mov  x22, x21                     // output_ptr
    mov  x23, OUT_BUF_SIZE
    add  x23, x21, x23                // output_limit

    mov  x20, 0                       // total_read
    mov  x24, READ_CHUNK              // read chunk size

read_loop:
    // Calculate remaining space in input buffer
    sub  x25, IN_BUF_SIZE, x20
