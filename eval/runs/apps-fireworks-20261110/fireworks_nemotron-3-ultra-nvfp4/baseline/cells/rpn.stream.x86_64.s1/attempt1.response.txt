.section .rodata
max_div_10: .quad 922337203685477580
int64_min: .quad -9223372036854775808
error_msg: .ascii "error\n"
error_len = . - error_msg

.section .bss
.align 8
input_buf: .zero 65536
stack: .zero 1024 * 8
outbuf: .zero 32

.section .text
.globl _start

_start:
    lea r13, [rip + stack]          # r13 = stack base (global)
    mov rdi, 0                      # stdin
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    xor r12, r12                    # total_read

read_loop:
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65536
    jge read_done
    lea rsi, [rip + input_buf + r12]
