.section .rodata
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
err_prefix: .ascii "error at "
err_prefix_len = . - err_prefix
newline: .ascii "\n"
newline_len = . - newline

.section .bss
.align 12
input_buf: .space 65536
input_len: .space 8

.section .text
.globl _start

// System call numbers
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ SYS_EXIT_GROUP, 94

// Parser state offsets on stack
.equ STACK_FRAME_SIZE, 16
.equ MAX_DEPTH, 1000
.equ STATE_STACK_SIZE, MAX_DEPTH * 8

// State values
.equ STATE_START, 0
.equ STATE_VALUE, 1
.equ STATE_OBJECT_KEY, 2
.equ STATE_OBJECT_COLON, 3
.equ STATE_OBJECT_VALUE, 4
.equ STATE_OBJECT_COMMA_OR_END, 5
.equ STATE_ARRAY_VALUE, 6
.equ STATE_ARRAY_COMMA_OR_END, 7
.equ STATE_STRING, 8
.equ STATE_STRING_ESCAPE, 9
.equ STATE_STRING_UNICODE, 10
.equ STATE_NUMBER, 11
.equ STATE_NUMBER_FRAC, 12
.equ STATE_NUMBER_EXP, 13
.equ STATE_NUMBER_EXP_SIGN, 14
.equ STATE_NUMBER_EXP_DIGITS, 15
.equ STATE_TRUE, 16
.equ STATE_FALSE, 17
.equ STATE_NULL, 18
.equ STATE_DONE, 19

// Character constants
.equ CHAR_SPACE, 0x20
.equ CHAR_TAB, 0x09
.equ CHAR_NL, 0x0A
.equ CHAR_CR, 0x0D
.equ CHAR_LBRACE, 0x7B  // {
.equ CHAR_RBRACE, 0x7D  // }
.equ CHAR_LBRACKET, 0x5B // [
.equ CHAR_RBRACKET, 0x5D // ]
.equ CHAR_COLON, 0x3A   // :
.equ CHAR_COMMA, 0x2C   // ,
.equ CHAR_QUOTE, 0x22   // "
.equ CHAR_BACKSLASH, 0x5C // \
.equ CHAR_MINUS, 0x2D   // -
.equ CHAR_DOT, 0x2E     // .
.equ CHAR_E, 0x65       // e
.equ CHAR_UPPER_E, 0x45 // E
.equ CHAR_PLUS, 0x2B    // +
.equ CHAR_0, 0x30
.equ CHAR_9, 0x39
.equ CHAR_a, 0x61
.equ CHAR_f, 0x66
.equ CHAR_A, 0x41
.equ CHAR_F, 0x46
.equ CHAR_b, 0x62
.equ CHAR_f_char, 0x66
.equ CHAR_n, 0x6E
.equ CHAR_r, 0x72
.equ CHAR_t, 0x74
.equ CHAR_u, 0x75
.equ CHAR_SLASH, 0x2F   // /

_start:
    // Initialize stack pointer for our use (already 16-byte aligned)
    // Read all input into buffer
    mov x19, xzr        // total bytes read = 0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf  // buffer pointer
    
read_loop:
    // Check if buffer full (65536 bytes)
    cmp x19, #65536
    b.ge read_done
    
    // read(0, buf + x19, 65536 - x19)
    mov x8, #SYS_READ
    mov x0, #0
    add x1, x20, x19
    sub x2, xzr, x19
    add x2, x2, #65536
    svc #0
    
    // Check result
    cmp x0, #0
    b.le read_done      // 0 = EOF, negative = error (treat as EOF)
    
    add x19, x19, x0    // increment total read
    b read_loop

read_done:
    // Store input length
    adrp x0, input_len
    add x0, x0, :lo12:input_len
    str x19, [x0]
    
    // Parse JSON
    // x19 = input length
    // x20 = input buffer
    // Use stack for parser state
    // We'll use x21 as current position (offset)
    // x22 as current state
    // x23 as stack pointer for parser state stack
    
    mov x21, xzr        // current offset = 0
    mov x22, #STATE_START
    
    // Allocate state stack on stack
    sub sp, sp, #STATE_STACK_SIZE
