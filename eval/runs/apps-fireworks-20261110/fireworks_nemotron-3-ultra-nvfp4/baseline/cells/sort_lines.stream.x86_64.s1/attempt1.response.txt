.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ LINE_STRIDE, 16

.section .rodata
newline: .byte 0x0A

.section .bss
.align 16
input_buf: .zero MAX_INPUT
line_array: .zero 16000
input_len: .zero 8
line_count: .zero 8

.text
.global _start
_start:
    xor r12, r12          # input_len
    xor r13, r13          # line_count

read_loop:
    mov rax, MAX_INPUT
    sub rax, r12
    test rax, rax
    jz read_done
    mov rdi, STDIN
    lea rsi, [input_buf + r12]
    mov rdx, rax
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov qword ptr [input_len], r12

    # parse lines
    lea r14, [line_array]
    xor r15, r15          # start offset
    xor rbx, rbx          # i
parse_loop:
    cmp rbx, r12
    jae parse_done
    movzx eax, byte ptr [input_buf + rbx]
    cmp al, 0x0A
    jne not_newline
    # newline found
    mov rax, rbx
    sub rax, r15          # length
    lea rdx, [input_buf + r15]
    mov qword ptr [r14 + r13*LINE_STRIDE], rdx
