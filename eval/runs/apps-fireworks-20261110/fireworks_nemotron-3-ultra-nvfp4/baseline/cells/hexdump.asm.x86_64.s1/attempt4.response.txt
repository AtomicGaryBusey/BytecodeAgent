.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAX_INPUT, 65536

.section .bss
.align 16
input_buffer: .zero MAX_INPUT
output_buffer: .zero 80

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .text
.globl _start

_start:
    xor r12, r12                # total bytes read
read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [input_buffer + r12]
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jne read_loop
read_done:
    xor r14, r14                # offset
process_loop:
    cmp r14, r12
    jge process_done
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jle n_ok
    mov r15, 16
n_ok:
    call generate_line
    mov rdi, 1
    lea rsi, [output_buffer]
    mov rdx, rax
    call write_all
    add r14, 16
    jmp process_loop
process_done:
    call print_final_offset
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

generate_line:
    lea rdi, [output_buffer]
    mov rbx, r14
    mov rsi, 7
    lea r8, [hex_digits]
    lea r9, [input_buffer]
offset_loop:
    mov rax, rbx
    mov cl, sil
    shl cl, 2
    shr rax, cl
    and eax, 0xF
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    jns offset_loop
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor rsi, rsi
    lea r10, [r9 + r14]         # base address of current row
hex_loop:
    cmp rsi, r15
    jl hex_byte
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
    jmp hex_next
hex_byte:
    movzx eax, byte ptr [r10 + rsi]
    mov ecx, eax
    shr ecx, 4
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], cl
    and eax, 0xF
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi+1], al
    mov byte ptr [rdi+2], ' '
    add rdi, 3
hex_next:
    cmp rsi, 7
    jne no_extra_space
    mov byte ptr [rdi], ' '
    inc rdi
no_extra_space:
    inc rsi
    cmp rsi, 16
    jne hex_loop
    mov word ptr [rdi], 0x7C20
    add rdi, 2
    xor rsi, rsi
ascii_loop:
    cmp rsi, r15
    jge ascii_done
    movzx eax, byte ptr [r10 + rsi]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7E
    jg ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rdi], '.'
ascii_next:
    inc rdi
    inc rsi
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rax, [output_buffer]
    sub rdi, rax
    mov rax, rdi
    ret

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    xor rbx, rbx
write_loop:
    mov rax, SYS_WRITE
    mov rdi, r8
    lea rsi, [r9 + rbx]
    mov rdx, r10
    sub rdx, rbx
    syscall
    test rax, rax
    jz write_error
    add rbx, rax
    cmp rbx, r10
    jne write_loop
    ret
write_error:
    mov rax, SYS_EXIT
    mov rdi, 1
    syscall

print_final_offset:
    lea rdi, [output_buffer]
    mov rbx, r12
    mov rsi, 7
    lea r8, [hex_digits]
final_loop:
    mov rax, rbx
    mov cl, sil
    shl cl, 2
    shr rax, cl
    and eax, 0xF
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    jns final_loop
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, 9
    lea rsi, [output_buffer]
    mov rdi, 1
    call write_all
    ret
