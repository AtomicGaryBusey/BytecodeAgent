.section .rodata
hex_digits: .ascii "0123456789abcdef"

.section .bss
.align 12
input_buffer: .space 65536
output_buffer: .space 524288
output_buffer_end = output_buffer + 524288

.section .text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536
.equ READ_CHUNK, 4096

_start:
    // Initialize state
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, 0                      // total bytes read

read_loop:
    mov x0, STDIN
    add x1, x19, x20                // buffer + offset
    mov x2, READ_CHUNK
    mov x8, SYS_READ
    svc #0
    cbz x0, read_done               // EOF
    add x20, x20, x0                // total += bytes read
    cmp x20, MAX_INPUT
    b.le read_loop
    // Input larger than MAX_INPUT, truncate (should not happen per spec)
    b read_done

read_done:
    // Process buffer
    mov x21, 0                      // current offset
    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    mov x23, x22                    // output write pointer
    adrp x24, output_buffer_end
    add x24, x24, :lo12:output_buffer_end

process_loop:
    cmp x21, x20
    b.ge process_done
    bl format_line
    add x21, x21, 16
    b process_loop

process_done:
    // Final offset line
    bl format_final_offset
    // Write all output
    mov x0, STDOUT
    mov x1, x22
    sub x2, x23, x22
    mov x8, SYS_WRITE
    svc #0
    // Exit
    mov x0, 0
    mov x8, SYS_EXIT
    svc #0

// format_line: format 16 bytes at input_buffer+x21 into output at x23
// x19=input_base, x20=total, x21=offset, x23=out_ptr, x24=out_end
// Clobbers: x0-x10, x22, x30 (saved)
format_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // n = min(16, total - offset)
    sub x0, x20, x21
    cmp x0, 16
    csel x0, x0, 16, hi             // x0 = n (1-16)

    // Write offset (8 hex digits)
    mov x1, x21
    bl write_hex8

    // Two spaces
    mov x1, ' '
    strb w1, [x23], #1
    strb w1, [x23], #1

    // Hex bytes (16 positions)
    mov x1, 0                       // i
1:  cmp x1, 16
    b.ge 2f
    cmp x1, x0
    b.ge 3f                         // missing byte
    // Present byte
    ldrb w2, [x19, x21, x1]
    bl write_hex_byte
    b 4f
3:  // Missing byte: three spaces
    mov x2, ' '
    strb w2, [x23], #1
    strb w2, [x23], #1
    strb w2, [x23], #1
4:  // Extra space after 7th byte (i==7)
    cmp x1, 7
    b.ne 5f
    mov x2, ' '
    strb w2, [x23], #1
5:  add x1, x1, 1
    b 1b

2:  // Separator: space, '|'
    mov x1, ' '
    strb w1, [x23], #1
    mov x1, '|'
    strb w1, [x23], #1

    // ASCII column (n bytes)
    mov x1, 0
6:  cmp x1, x0
    b.ge 7f
    ldrb w2, [x19, x21, x1]
    cmp w2, 0x20
    b.lt 8f
    cmp w2, 0x7E
    b.gt 8f
    strb w2, [x23], #1
    b 9f
8:  mov w2, '.'
    strb w2, [x23], #1
9:  add x1, x1, 1
    b 6b

7:  // Terminator: '|', newline
    mov x1, '|'
    strb w1, [x23], #1
    mov x1, '\n'
    strb w1, [x23], #1

    ldp x29, x30, [sp], #16
    ret

// write_hex8: x1 = 32-bit value, write 8 hex digits at x23, update x23
// Clobbers: x0-x5
write_hex8:
    mov x2, 28
    mov x3, 8
1:  lsr x4, x1, x2
    and x4, x4, 0xF
    adrp x5, hex_digits
    add x5, x5, :lo12:hex_digits
    ldrb w4, [x5, x4]
    strb w4, [x23], #1
    subs x2, x2, 4
    subs x3, x3, 1
    b.ne 1b
    ret

// write_hex_byte: x2 = byte, write 2 hex digits + space at x23
// Clobbers: x3-x5
write_hex_byte:
    // High nibble
    lsr x3, x2, 4
    and x3, x3, 0xF
    adrp x4, hex_digits
    add x4, x4, :lo12:hex_digits
    ldrb w5, [x4, x3]
    strb w5, [x23], #1
    // Low nibble
    and x3, x2, 0xF
    ldrb w5, [x4, x3]
    strb w5, [x23], #1
    // Space
    mov x3, ' '
    strb w3, [x23], #1
    ret

// format_final_offset: x20 = total, write 8 hex digits + newline at x23
format_final_offset:
    stp x29, x30, [sp, #-16]!
    mov x1, x20
    bl write_hex8
    mov x1, '\n'
    strb w1, [x23], #1
    ldp x29, x30, [sp], #16
    ret
