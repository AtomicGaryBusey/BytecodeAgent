.section .text
.global _start

_start:
    // Initialize counters
    mov x19, 0          // lines
    mov x20, 0          // words
    mov x21, 0          // bytes
    mov x22, 0          // in_word (0 = false, 1 = true)

    // Get buffer address
    adrp x23, buffer
    add x23, x23, :lo12:buffer

.Lread_loop:
    // read(0, buffer, 65536)
    mov x0, 0
    mov x1, x23
    mov x2, 65536
    mov x8, 63
    svc #0

    // x0 = bytes read
    cbz x0, .Lread_done

    mov x24, x0         // count
    mov x25, 0          // index

.Lprocess_loop:
    ldrb w26, [x23, x25]
    add x21, x21, 1     // bytes++

    // Check for newline
    cmp w26, 0x0A
    b.ne .Lnot_newline
    add x19, x19, 1     // lines++
.Lnot_newline:

    // Check for whitespace (space, tab, newline, vt, ff, cr)
    cmp w26, 0x20
    b.eq .Lwhitespace
    cmp w26, 0x09
    b.eq .Lwhitespace
    cmp w26, 0x0A
    b.eq .Lwhitespace
    cmp w26, 0x0B
    b.eq .Lwhitespace
    cmp w26, 0x0C
    b.eq .Lwhitespace
    cmp w26, 0x0D
    b.eq .Lwhitespace

    // Non-whitespace
    cbz x22, .Lstart_word
    b .Lcontinue

.Lstart_word:
    mov x22, 1
    b .Lcontinue

.Lwhitespace:
    cbnz x22, .Lend_word
    b .Lcontinue

.Lend_word:
    add x20, x20, 1
    mov x22, 0

.Lcontinue:
    add x25, x25, 1
    cmp x25, x24
    b.lt .Lprocess_loop
    b .Lread_loop

.Lread_done:
    // Count final word if input ended while in a word
    cbz x22, .Loutput
    add x20, x20, 1

.Loutput:
    // Set up stack buffers
    sub sp, sp, 128
    mov x28, sp         // save base pointer
    mov x23, sp         // output buffer pointer
    add x24, sp, 64     // temporary conversion buffer

    // Convert and copy lines (x19)
    mov x0, x19
    mov x1, x24
    bl itoa
    mov x25, x0         // length
    mov x26, x1         // source
    cbz x25, .Lafter_copy1
.Lcopy1:
    ldrb w27, [x26], 1
    strb w27, [x23], 1
    subs x25, x25, 1
    b.ne .Lcopy1
.Lafter_copy1:
    mov w27, ' '
    strb w27, [x23], 1

    // Convert and copy words (x20)
    mov x0, x20
    mov x1, x24
    bl itoa
    mov x25, x0
    mov x26, x1
    cbz x25, .Lafter_copy2
.Lcopy2:
    ldrb w27, [x26], 1
    strb w27, [x23], 1
    subs x25, x25, 1
    b.ne .Lcopy2
.Lafter_copy2:
    mov w27, ' '
    strb w27, [x23], 1

    // Convert and copy bytes (x21)
    mov x0, x21
    mov x1, x24
    bl itoa
    mov x25, x0
    mov x26, x1
    cbz x25, .Lafter_copy3
.Lcopy3:
    ldrb w27, [x26], 1
    strb w27, [x23], 1
    subs x25, x25, 1
    b.ne .Lcopy3
.Lafter_copy3:
    mov w27, '\n'
    strb w27, [x23], 1

    // Write output
    mov x0, 1
    mov x1, x28
    sub x2, x23, x28
    mov x8, 64
    svc #0

    // Exit
    mov x0, 0
    mov x8, 93
    svc #0

// itoa: convert unsigned 64-bit integer in x0 to decimal string at buffer x1.
// Returns length in x0, pointer to string in x1.
// Uses registers x2-x7, preserves x19-x28.
itoa:
    mov x2, x1          // save buffer start
    add x2, x2, 31      // point to last byte (we'll write backwards)
    mov x3, 10
    mov x4, 0           // digit count
    cbz x0, .Lzero
.Lloop:
    udiv x5, x0, x3
    msub x6, x5, x3, x0 // x6 = x0 % 10
    add x6, x6, '0'
    strb w6, [x2], -1   // store and decrement
    add x4, x4, 1
    mov x0, x5
    cbnz x0, .Lloop
    b .Ldone
.Lzero:
    mov w6, '0'
    strb w6, [x2], -1
    add x4, x4, 1
.Ldone:
    // x2 now points to one before the first digit
    add x1, x2, 1       // start of string
    mov x0, x4          // length
    ret

.section .bss
.align 12
buffer: .space 65536
