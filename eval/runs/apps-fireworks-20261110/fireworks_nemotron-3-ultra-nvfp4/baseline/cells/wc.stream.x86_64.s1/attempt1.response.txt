.section .text
.globl _start

_start:
    lea r14, [rip + buffer]
    xor rbx, rbx        # lines
    xor rbp, rbp        # words
    xor r12, r12        # bytes
    xor r13, r13        # in_word

.read_loop:
    mov rax, 0          # sys_read
    mov rdi, 0          # stdin
    mov rsi, r14        # buffer
    mov rdx, 4096       # count
    syscall
    test rax, rax
    jz .read_done
    add r12, rax        # total bytes
    mov rcx, rax        # loop counter
    xor rsi, rsi        # index

.process_byte:
    movzx eax, byte ptr [r14 + rsi]
    cmp al, 0x0A
    jne .check_ws
    inc rbx             # newline -> line++

.check_ws:
    cmp al, 0x09
    je .whitespace
    cmp al, 0x0B
    je .whitespace
    cmp al, 0x0C
    je .whitespace
    cmp al, 0x0D
    je .whitespace
    cmp al, 0x20
    je .whitespace

    # not whitespace
    test r13, r13
    jnz .next_byte
    mov r13, 1
    inc rbp             # new word
    jmp .next_byte

.whitespace:
    test r13, r13
    jz .next_byte
    mov r13, 0

.next_byte:
    inc rsi
    dec rcx
    jnz .process_byte
    jmp .read_loop

.read_done:
    lea rdi, [rip + out_buf]

    # lines
    mov rdx, rbx
    call append_number
    mov byte ptr [rdi], ' '
    inc rdi

    # words
    mov rdx, rbp
    call append_number
    mov byte ptr [rdi], ' '
    inc rdi

    # bytes
    mov rdx, r12
    call append_number
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + out_buf]
    sub rdi, rsi        # length
    mov rdx, rdi
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    syscall

    mov rax, 60         # sys_exit
    xor rdi, rdi
    syscall

append_number:
    # rdx = number, rdi = output pointer
    test rdx, rdx
    jnz .convert
    mov byte ptr [rdi], '0'
    inc rdi
    ret

.convert:
    lea rsi, [rip + tmp_buf + 31]
    mov rax, rdx
    mov rcx, 10
.div_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .div_loop

    lea rcx, [rip + tmp_buf + 31]
    sub rcx, rsi        # length
    rep movsb
    ret

.section .bss
.align 4096
buffer: .zero 4096
out_buf: .zero 100
tmp_buf: .zero 32
