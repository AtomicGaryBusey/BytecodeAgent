.section .rodata
base64_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4096
input_buf: .zero 49152
output_buf: .zero 66400

.section .text
.globl _start

_start:
    # Read all input
    mov r12, 0
    lea r13, [rip + input_buf]
    mov r14, 49152

read_loop:
    mov rdx, r14
    sub rdx, r12
    test rdx, rdx
    jz read_done
    lea rsi, [r13 + r12]
    mov rdi, 0
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    # Setup encoding
    lea rbp, [rip + output_buf]
    mov rdi, rbp
    mov rsi, r13
    mov rcx, r12
    xor r14, r14
    lea rbx, [rip + base64_table]

encode_loop:
    cmp rcx, 3
    jge encode_3bytes
    cmp rcx, 2
    je encode_2bytes
    # rcx == 1
    jmp encode_1byte

encode_3bytes:
    mov r8b, [rsi]
    mov r9b, [rsi+1]
    mov r10b, [rsi+2]

    # sextet0
    movzx eax, r8b
    shr eax, 2
    mov al, [rbx+rax]
    call putchar

    # sextet1
    movzx eax, r8b
    and eax, 3
    shl eax, 4
    movzx edx, r9b
    shr edx, 4
    or eax, edx
    mov al, [rbx+rax]
    call putchar

    # sextet2
    movzx eax, r9b
    and eax, 0xF
    shl eax, 2
    movzx edx, r10b
    shr edx, 6
    or eax, edx
    mov al, [rbx+rax]
    call putchar

    # sextet3
    movzx eax, r10b
    and eax, 0x3F
    mov al, [rbx+rax]
    call putchar

    add rsi, 3
    sub rcx, 3
    test rcx, rcx
    jnz encode_loop
    jmp encode_finish

encode_2bytes:
    mov r8b, [rsi]
    mov r9b, [rsi+1]

    # sextet0
    movzx eax, r8b
    shr eax, 2
    mov al, [rbx+rax]
    call putchar

    # sextet1
    movzx eax, r8b
    and eax, 3
    shl eax, 4
    movzx edx, r9b
    shr edx, 4
    or eax, edx
    mov al, [rbx+rax]
    call putchar

    # sextet2
    movzx eax, r9b
    and eax, 0xF
    shl eax, 2
    mov al, [rbx+rax]
    call putchar

    # padding
    mov al, '='
    call putchar

    jmp encode_finish

encode_1byte:
    mov r8b, [rsi]

    # sextet0
    movzx eax, r8b
    shr eax, 2
    mov al, [rbx+rax]
    call putchar

    # sextet1
    movzx eax, r8b
    and eax, 3
    shl eax, 4
    mov al, [rbx+rax]
    call putchar

    # padding
    mov al, '='
    call putchar
    mov al, '='
    call putchar

    jmp encode_finish

encode_finish:
    test r14, r14
    jz write_output
    mov al, 10
    call putchar

write_output:
    mov rdx, rdi
    sub rdx, rbp
    test rdx, rdx
    jz exit_success
    mov rsi, rbp

write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_loop

exit_success:
    mov rax, 60
    mov rdi, 0
    syscall

putchar:
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne putchar_ret
    mov byte [rdi], 10
    inc rdi
    xor r14, r14
putchar_ret:
    ret
