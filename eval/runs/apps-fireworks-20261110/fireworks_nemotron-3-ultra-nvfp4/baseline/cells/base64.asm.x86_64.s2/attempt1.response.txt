.globl _start

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
input_buf:
    .zero 49152
    .align 16
output_buf:
    .zero 70000

.section .text
_start:
    # Read all input into input_buf
    xor r12, r12                # total_read = 0
    lea r14, [rip + input_buf]  # r14 = input_buf base

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [r14 + r12]        # buffer + total_read
    mov rdx, 49152
    sub rdx, r12                # remaining space
    syscall
    test rax, rax
    jz read_done
    add r12, rax                # total_read += bytes read
    cmp r12, 49152
    jne read_loop

read_done:
    # Encode base64
    lea rsi, [r14]              # input pointer
    lea rdi, [r14 + r12]        # input end
    lea r13, [rip + output_buf] # r13 = output_buf base
    mov rdx, r13                # output pointer
    xor rcx, rcx                # line length = 0
    lea rbx, [rip + base64_table]

encode_loop:
    cmp rsi, rdi
    je encode_done
    mov r15, rdi
    sub r15, rsi                # remaining bytes
    cmp r15, 3
    jl handle_short

    # Process 3 bytes
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi+1]
    movzx r10d, byte ptr [rsi+2]
    add rsi, 3

    # index0 = byte0 >> 2
    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call output_char

    # index1 = ((byte0 & 3) << 4) | (byte1 >> 4)
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov r11d, r9d
    shr r11d, 4
    or eax, r11d
    mov al, byte ptr [rbx + rax]
    call output_char

    # index2 = ((byte1 & 15) << 2) | (byte2 >> 6)
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov r11d, r10d
    shr r11d, 6
    or eax, r11d
    mov al, byte ptr [rbx + rax]
    call output_char

    # index3 = byte2 & 63
    mov eax, r10d
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call output_char

    jmp encode_loop

handle_short:
    cmp r15, 2
    je two_bytes
    cmp r15, 1
    jne encode_done

    # One byte left
    movzx r8d, byte ptr [rsi]
    add rsi, 1

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call output_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbx + rax]
    call output_char

    mov al, '='
    call output_char
    call output_char
    jmp encode_done

two_bytes:
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi+1]
    add rsi, 2

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call output_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov r11d, r9d
    shr r11d, 4
    or eax, r11d
    mov al, byte ptr [rbx + rax]
    call output_char

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rbx + rax]
    call output_char

    mov al, '='
    call output_char
    jmp encode_done

encode_done:
    cmp rcx, 0
    je write_output
    mov byte ptr [rdx], 0x0A
    inc rdx

write_output:
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    mov rsi, r13                # output_buf
    mov rdx, rdx
    sub rdx, r13                # length
    syscall

    mov rax, 60                 # sys_exit
    xor rdi, rdi
    syscall

# output_char(al): writes character in al to output buffer, handling line breaks.
# Uses rdx (output pointer), rcx (current line length). Preserves other registers.
output_char:
    cmp rcx, 76
    jne 1f
    mov byte ptr [rdx], 0x0A
    inc rdx
    xor rcx, rcx
1:
    mov byte ptr [rdx], al
    inc rdx
    inc rcx
    ret
