.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 66400

.section .bss
.align 12
input_buf:
    .space INPUT_MAX
output_buf:
    .space OUTPUT_MAX

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
.globl _start
_start:
    // Registers:
    // x19 = input_len
    // x20 = output_len (total)
    // x21 = line_len (0-75)
    // x22 = input_buf base
    // x23 = output_buf base
    // x24 = base64_table base
    // x25 = input index
    // x26 = output index
    // x27 = line_len (current line)
    // x28 = temp

    mov x19, #0          // input_len = 0
    mov x20, #0          // output_len = 0
    mov x21, #0          // line_len = 0

    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

    // ---------- Read loop ----------
read_loop:
    mov x0, #0                     // stdin
    add x1, x22, x19               // buf = input_buf + input_len
    mov x3, #INPUT_MAX
    sub x2, x3, x19                // count = INPUT_MAX - input_len
    cbz x2, read_done              // buffer full (should not happen)
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done              // EOF
    add x19, x19, x0               // input_len += bytes_read
    b read_loop

read_done:
    cbz x19, exit_success          // empty input -> nothing to do

    // ---------- Encode ----------
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    adrp x24, base64_table
    add x24, x24, :lo12:base64_table

    mov x25, #0                    // input index
    mov x26, #0                    // output index
    mov x27, #0                    // line_len

encode_loop:
    sub x28, x19, x25              // remaining input bytes
    cmp x28, #3
    b.lt encode_last               // less than 3 bytes left

    // Load 3 bytes
    ldrb w0, [x22, x25]
    ldrb w1, [x22, x25, #1]
