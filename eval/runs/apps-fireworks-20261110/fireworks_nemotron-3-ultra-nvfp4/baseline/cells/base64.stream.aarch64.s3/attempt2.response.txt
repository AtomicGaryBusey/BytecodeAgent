.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 66400

.section .bss
.align 12
input_buf:
    .space INPUT_MAX
output_buf:
    .space OUTPUT_MAX

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
.globl _start
_start:
    // Registers:
    // x19 = input_len
    // x20 = output_len (total bytes in output_buf)
    // x21 = line_len (0-75)
    // x22 = input_buf base
    // x23 = output_buf base
    // x24 = base64_table base
    // x25 = input index (i)
    // x26 = output index (o)
    // x27 = temp (base64 char index)
    // x28 = temp (address, etc.)

    mov x19, #0          // input_len = 0
    mov x20, #0          // output_len = 0
    mov x21, #0          // line_len = 0

    adrp x22, input_buf
    add x22, x22, :lo12:input_buf

    // ---------- Read loop ----------
read_loop:
    mov x0, #0                     // stdin
    add x1, x22, x19               // buf = input_buf + input_len
    mov x3, #INPUT_MAX
    sub x2, x3, x19                // count = INPUT_MAX - input_len
    cbz x2, read_done              // buffer full (should not happen per limits)
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done              // EOF
    add x19, x19, x0               // input_len += bytes_read
    b read_loop

read_done:
    cbz x19, exit_success          // empty input -> nothing to do

    // ---------- Encode ----------
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    adrp x24, base64_table
    add x24, x24, :lo12:base64_table

    mov x25, #0                    // input index i = 0
    mov x26, #0                    // output index o = 0
    // x21 already 0 (line_len)

encode_loop:
    sub x28, x19, x25              // remaining = input_len - i
    cmp x28, #3
    b.lt encode_last               // less than 3 bytes left

    // Load 3 bytes: compute base address once
    add x28, x22, x25              // x28 = input_buf + i
    ldrb w0, [x28]                 // b0
    ldrb w1, [x28, #1]             // b1
    ldrb w2, [x28, #2]             // b2

    // Encode 3 bytes -> 4 base64 chars
    // out0 = (b0 >> 2) & 0x3F
    ubfx w3, w0, #2, #6            // w3 = (b0 >> 2) & 0x3F
    // out1 = ((b0 & 0x03) << 4) | ((b1 >> 4) & 0x0F)
    and w4, w0, #3                 // b0 & 3
    lsl w4, w4, #4                 // << 4
    ubfx w5, w1, #4, #4            // (b1 >> 4) & 0xF
    orr w4, w4, w5                 // out1
    // out2 = ((b1 & 0x0F) << 2) | ((b2 >> 6) & 0x03)
    and w5, w1, #0xF               // b1 & 0xF
    lsl w5, w5, #2                 // << 2
    ubfx w6, w2, #6, #2            // (b2 >> 6) & 3
    orr w5, w5, w6                 // out2
    // out3 = b2 & 0x3F
    and w6, w2, #0x3F              // out3

    // Map to base64 chars and store
    // char0
    ldrb w7, [x24, w3, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 1f
    mov w7, #0x0A                  // newline
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
1:
    // char1
    ldrb w7, [x24, w4, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 2f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
2:
    // char2
    ldrb w7, [x24, w5, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 3f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
3:
    // char3
    ldrb w7, [x24, w6, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 4f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
4:
    add x25, x25, #3               // i += 3
    b encode_loop

encode_last:
    // Handle remaining 1 or 2 bytes
    cmp x28, #0
    beq encode_finish              // no remaining bytes (should not happen here)

    cmp x28, #1
    b.eq encode_1byte
    // must be 2 bytes
    // Load 2 bytes
    add x28, x22, x25
    ldrb w0, [x28]                 // b0
    ldrb w1, [x28, #1]             // b1

    // out0 = (b0 >> 2) & 0x3F
    ubfx w3, w0, #2, #6
    // out1 = ((b0 & 3) << 4) | ((b1 >> 4) & 0xF)
    and w4, w0, #3
    lsl w4, w4, #4
    ubfx w5, w1, #4, #4
    orr w4, w4, w5
    // out2 = ((b1 & 0xF) << 2) & 0x3F
    and w5, w1, #0xF
    lsl w5, w5, #2
    // out3 = padding (64? but we use '=' char, index not in table)
    // We'll handle padding separately

    // Output char0
    ldrb w7, [x24, w3, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 5f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
5:
    // Output char1
    ldrb w7, [x24, w4, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 6f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
6:
    // Output char2
    ldrb w7, [x24, w5, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 7f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
7:
    // Output padding '='
    mov w7, #0x3D                  // '='
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne encode_finish
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
    b encode_finish

encode_1byte:
    // Load 1 byte
    add x28, x22, x25
    ldrb w0, [x28]                 // b0

    // out0 = (b0 >> 2) & 0x3F
    ubfx w3, w0, #2, #6
    // out1 = ((b0 & 3) << 4) & 0x3F
    and w4, w0, #3
    lsl w4, w4, #4
    // out2, out3 = padding

    // Output char0
    ldrb w7, [x24, w3, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 8f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
8:
    // Output char1
    ldrb w7, [x24, w4, uxtw #0]
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 9f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
9:
    // Output padding '='
    mov w7, #0x3D
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne 10f
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0
10:
    // Output padding '='
    mov w7, #0x3D
    strb w7, [x23, x26]
    add x26, x26, #1
    add x21, x21, #1
    cmp x21, #76
    b.ne encode_finish
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1
    mov x21, #0

encode_finish:
    // If line_len > 0, need final newline
    cbz x21, write_output
    mov w7, #0x0A
    strb w7, [x23, x26]
    add x26, x26, #1

write_output:
    // x26 = output_len
    mov x0, #1                     // stdout
    mov x1, x23                    // output_buf
    mov x2, x26                    // count
    mov x8, #SYS_WRITE
    svc #0
    // ignore write result, assume success

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    // should not return
    b exit_success
