.section .rodata
hex_table: .ascii "0123456789abcdef"

.section .bss
.align 12
input_buf: .space 65536
output_buf: .space 80

.section .text
.globl _start
_start:
    // Read all input into input_buf
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0                  // total_bytes = 0

read_loop:
    mov x3, #0x10000             // 65536
    sub x3, x3, x19              // remaining space
    mov x2, #0x1000              // 4096
    cmp x2, x3
    csel x2, x2, x3, hi          // x2 = min(4096, remaining)
    cbz x2, read_done

    mov x0, #0                   // fd = 0
    add x1, x20, x19             // buf = input_buf + total_bytes
    mov x8, #63                  // read
    svc #0
    cbz x0, read_done
    add x19, x19, x0             // total_bytes += bytes read
    b read_loop

read_done:
    // Process lines
    mov x22, #0                  // current offset = 0
    adrp x10, hex_table
    add x10, x10, :lo12:hex_table

process_loop:
    cmp x22, x19
    b.ge process_done

    sub x23, x19, x22            // bytes left
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, hi       // x23 = n (bytes in this line)

    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x27, x21                 // output pointer

    // 1. Offset as 8 hex digits
    mov x25, x22
    mov x24, #8
offset_loop:
    lsr w26, w25, #28
    ldrb w26, [x10, x26]
    strb w26, [x27], #1
    lsl w25, w25, #4
    subs x24, x24, #1
    b.ne offset_loop

    // two spaces
    mov w26, #' '
    strb w26, [x27], #1
    strb w26, [x27], #1

    // 2. Hex bytes: for i = 0..15
    mov x24, #0
hex_loop:
    cmp x24, #16
    b.ge hex_done
    cmp x24, x23
    b.lt print_hex_byte
    // print three spaces
    mov w26, #' '
    strb w26, [x27], #1
    strb w26, [x27], #1
    strb w26, [x27], #1
    b hex_next

print_hex_byte:
    add x26, x20, x22
    add x26, x26, x24
    ldrb w26, [x26]
    // high nibble
    ubfx w28, w26, #4, #4
    ldrb w28, [x10, x28]
    strb w28, [x27], #1
    // low nibble
    ubfx w28, w26, #0, #4
    ldrb w28, [x10, x28]
    strb w28, [x27], #1
    // space
    mov w28, #' '
    strb w28, [x27], #1

hex_next:
    // extra space after 7th byte (i == 7)
    cmp x24, #7
    b.ne no_extra_space
    mov w28, #' '
    strb w28, [x27], #1
no_extra_space:
    add x24, x24, #1
    b hex_loop

hex_done:
    // 3. Space, '|'
    mov w26, #' '
    strb w26, [x27], #1
    mov w26, #'|'
    strb w26, [x27], #1

    // 4. ASCII chars for n bytes
    mov x24, #0
ascii_loop:
    cmp x24, x23
    b.ge ascii_done
    add x26, x20, x22
    add x26, x26, x24
    ldrb w26, [x26]
    // if 0x20 <= w26 <= 0x7E print w26 else '.'
    cmp w26, #0x20
    b.lt print_dot
    cmp w26, #0x7E
    b.gt print_dot
    strb w26, [x27], #1
    b ascii_next
print_dot:
    mov w26, #'.'
    strb w26, [x27], #1
ascii_next:
    add x24, x24, #1
    b ascii_loop

ascii_done:
    // '|', newline
    mov w26, #'|'
    strb w26, [x27], #1
    mov w26, #'\n'
    strb w26, [x27], #1

    // write output line
    sub x2, x27, x21           // length
    mov x0, #1                 // fd = 1
    mov x1, x21                // buf
    mov x8, #64                // write
    svc #0

    // advance offset
    add x22, x22, x23
    b process_loop

process_done:
    // print final offset (total bytes) as 8 hex digits + newline
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x27, x21
    mov x25, x19
    mov x24, #8
final_offset_loop:
    lsr w26, w25, #28
    ldrb w26, [x10, x26]
    strb w26, [x27], #1
    lsl w25, w25, #4
    subs x24, x24, #1
    b.ne final_offset_loop
    mov w26, #'\n'
    strb w26, [x27], #1

    sub x2, x27, x21
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

    // exit
    mov x0, #0
    mov x8, #93
    svc #0
