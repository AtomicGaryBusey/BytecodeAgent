.globl _start

.section .rodata
ok_str:       .ascii "ok\n"
ok_len = . - ok_str
err_prefix:   .ascii "error at "
err_prefix_len = . - err_prefix
newline:      .ascii "\n"
newline_len = . - newline

.section .bss
.align 3
buffer:       .space 65536

.section .text
.align 4

// write_str(x1=buf, x2=len) -> writes to fd=1
write_str:
    mov x0, 1
    mov x8, 64
    svc #0
    ret

// write_decimal(x0=value) -> writes decimal representation to fd=1
write_decimal:
    stp x29, x30, [sp, #-16]!
    sub sp, sp, 32              // 32 bytes for digits
    mov x9, sp
    add x10, x9, 32             // end pointer
    cmp x0, 0
    b.ne 1f
    sub x10, x10, 1
    mov w11, '0'
    strb w11, [x10]
    b 2f
1:  // convert loop
    mov x11, 10
    udiv x12, x0, x11
    msub x13, x12, x11, x0      // remainder
    add w13, w13, '0'
    sub x10, x10, 1
    strb w13, [x10]
    mov x0, x12
    cbnz x0, 1b
2:
    mov x1, x10
    sub x2, x9, x10
    neg x2, x2                  // length = end - start
    bl write_str
    add sp, sp, 32
    ldp x29, x30, [sp], #16
    ret

// print_ok
print_ok:
    adr x1, ok_str
    mov x2, ok_len
    b write_str

// print_error
print_error:
    stp x29, x30, [sp, #-16]!
    adr x1, err_prefix
    mov x2, err_prefix_len
    bl write_str
    mov x0, x22
    bl write_decimal
    adr x1, newline
    mov x2, newline_len
    bl write_str
    ldp x29, x30, [sp], #16
    ret

// parse_ws: skip whitespace, returns 0 in x0
parse_ws:
1:  cmp x19, x21
    b.ge 2f
    ldrb w9, [x20, x19]
    cmp w9, 0x20
    b.eq 3f
    cmp w9, 0x09
    b.eq 3f
    cmp w9, 0x0A
    b.eq 3f
    cmp w9, 0x0D
    b.eq 3f
2:  mov x0, 0
    ret
3:  add x19, x19, 1
    b 1b

// parse_true
parse_true:
    cmp x19, x21
    b.ge parse_true_err
    add x9, x19, 4
    cmp x9, x21
    b.gt parse_true_err
    ldrb w9, [x20, x19]
    cmp w9, 't'
    b.ne parse_true_err
    ldrb w9, [x20, x19, 1]
    cmp w9, 'r'
    b.ne parse_true_err
    ldrb w9, [x20, x19, 2]
    cmp w9, 'u'
    b.ne parse_true_err
    ldrb w9, [x20, x19, 3]
    cmp w9, 'e'
    b.ne parse_true_err
    add x19, x19, 4
    mov x0, 0
    ret
parse_true_err:
    mov x22, x19
    mov x0, -1
    ret

// parse_false
parse_false:
    cmp x19, x21
    b.ge parse_false_err
    add x9, x19, 5
    cmp x9, x21
    b.gt parse_false_err
    ldrb w9, [x20, x19]
    cmp w9, 'f'
    b.ne parse_false_err
    ldrb w9, [x20, x19, 1]
    cmp w9, 'a'
    b.ne parse_false_err
    ldrb w9, [x20, x19, 2]
    cmp w9, 'l'
    b.ne parse_false_err
    ldrb w9, [x20, x19, 3]
    cmp w9, 's'
    b.ne parse_false_err
    ldrb w9, [x20, x19, 4]
    cmp w9, 'e'
    b.ne parse_false_err
    add x19, x19, 5
    mov x0, 0
    ret
parse_false_err:
    mov x22, x19
    mov x0, -1
    ret

// parse_null
parse_null:
    cmp x19, x21
    b.ge parse_null_err
    add x9, x19, 4
    cmp x9, x21
    b.gt parse_null_err
    ldrb w9, [x20, x19]
    cmp w9, 'n'
    b.ne parse_null_err
    ldrb w9, [x20, x19, 1]
    cmp w9, 'u'
    b.ne parse_null_err
    ldrb w9, [x20, x19, 2]
    cmp w9, 'l'
    b.ne parse_null_err
    ldrb w9, [x20, x19, 3]
    cmp w9, 'l'
    b.ne parse_null_err
    add x19, x19, 4
    mov x0, 0
    ret
parse_null_err:
    mov x22, x19
    mov x0, -1
    ret

// parse_string
parse_string:
    stp x29, x30, [sp, #-16]!
    add x19, x19, 1             // skip opening quote
1:  cmp x19, x21
    b.ge parse_str_eof
    ldrb w9, [x20, x19]
    cmp w9, '"'
    b.eq parse_str_end
    cmp w9, '\\'
    b.eq parse_str_escape
    cmp w9, 0x20
    b.lt parse_str_ctrl
    add x19, x19, 1
    b 1b

parse_str_escape:
    add x19, x19, 1
    cmp x19, x21
    b.ge parse_str_eof
    ldrb w9, [x20, x19]
    // check allowed escapes
    cmp w9, '"'
    b.eq parse_str_esc_ok
    cmp w9, '\\'
    b.eq parse_str_esc_ok
    cmp w9, '/'
    b.eq parse_str_esc_ok
    cmp w9, 'b'
    b.eq parse_str_esc_ok
    cmp w9, 'f'
    b.eq parse_str_esc_ok
    cmp w9, 'n'
    b.eq parse_str_esc_ok
    cmp w9, 'r'
    b.eq parse_str_esc_ok
    cmp w9, 't'
    b.eq parse_str_esc_ok
    cmp w9, 'u'
    b.eq parse_str_unicode
    b parse_str_err

parse_str_unicode:
    mov x9, 4
1:  add x19, x19, 1
    cmp x19, x21
    b.ge parse_str_eof
    ldrb w10, [x20, x19]
    // hex digit check
    cmp w10, '0'
    b.lt parse_str_err
    cmp w10, '9'
    b.le 2f
    cmp w10, 'A'
    b.lt parse_str_err
    cmp w10, 'F'
    b.le 2f
    cmp w10, 'a'
    b.lt parse_str_err
    cmp w10, 'f'
    b.gt parse_str_err
2:  subs x9, x9, 1
    b.ne 1b
    add x19, x19, 1
    b 1b

parse_str_esc_ok:
    add x19, x19, 1
    b 1b

parse_str_end:
    add x19, x19, 1
    mov x0, 0
    b parse_str_exit

parse_str_eof:
    mov x22, x19
    mov x0, -1
    b parse_str_exit

parse_str_ctrl:
    mov x22, x19
    mov x0, -1
    b parse_str_exit

parse_str_err:
    mov x22, x19
    mov x0, -1
    b parse_str_exit

parse_str_exit:
    ldp x29, x30, [sp], #16
    ret

// parse_number
parse_number:
    stp x29, x30, [sp, #-16]!
    // optional minus
    ldrb w9, [x20, x19]
    cmp w9, '-'
    b.ne parse_num_int
    add x19, x19, 1
    cmp x19, x21
    b.ge parse_num_err
    ldrb w9, [x20, x19]

parse_num_int:
    cmp w9, '0'
    b.eq parse_num_int_zero
    cmp w9, '1'
    b.lt parse_num_err
    cmp w9, '9'
    b.gt parse_num_err
    // digit 1-9
    add x19, x19, 1
    b parse_num_int_digits

parse_num_int_zero:
    add x19, x19, 1

parse_num_int_digits:
    cmp x19, x21
    b.ge parse_num_after_int
    ldrb w9, [x20, x19]
    cmp w9, '0'
    b.lt parse_num_after_int
    cmp w9, '9'
    b.gt parse_num_after_int
    add x19, x19, 1
    b parse_num_int_digits

parse_num_after_int:
    // fraction?
    cmp x19, x21
    b.ge parse_num_after_frac
    ldrb w9, [x20, x19]
    cmp w9, '.'
    b.ne parse_num_after_frac
    add x19, x19, 1
    cmp x19, x21
    b.ge parse_num_err
    ldrb w9, [x20, x19]
    cmp w9, '0'
    b.lt parse_num_err
    cmp w9, '9'
    b.gt parse_num_err
    add x19, x19, 1
    b parse_num_frac_digits

parse_num_frac_digits:
    cmp x19, x21
    b.ge parse_num_after_frac
    ldrb w9, [x20, x19]
    cmp w9, '0'
    b.lt parse_num_after_frac
    cmp w9, '9'
    b.gt parse_num_after_frac
    add x19, x19, 1
    b parse_num_frac_digits

parse_num_after_frac:
    // exponent?
    cmp x19, x21
    b.ge parse_num_done
    ldrb w9, [x20, x19]
    cmp w9, 'e'
    b.eq parse_num_exp
    cmp w9, 'E'
    b.ne parse_num_done

parse_num_exp:
    add x19, x19, 1
    cmp x19, x21
    b.ge parse_num_err
    ldrb w9, [x20, x19]
    cmp w9, '+'
    b.eq parse_num_exp_sign
    cmp w9, '-'
    b.eq parse_num_exp_sign
    b parse_num_exp_digit

parse_num_exp_sign:
    add x19, x19, 1
    cmp x19, x21
    b.ge parse_num_err
    ldrb w9, [x20, x19]

parse_num_exp_digit:
    cmp w9, '0'
    b.lt parse_num_err
    cmp w9, '9'
    b.gt parse_num_err
    add x19, x19, 1

parse_num_exp_digits:
    cmp x19, x21
    b.ge parse_num_done
    ldrb w9, [x20, x19]
    cmp w9, '0'
    b.lt parse_num_done
    cmp w9, '9'
    b.gt parse_num_done
    add x19, x19, 1
    b parse_num_exp_digits

parse_num_done:
    mov x0, 0
    b parse_num_exit

parse_num_err:
    mov x22, x19
    mov x0, -1
    b parse_num_exit

parse_num_exit:
    ldp x29, x30, [sp], #16
    ret

// parse_value
parse_value:
    stp x29, x30, [sp, #-16]!
    bl parse_ws
    cbnz x0, parse_val_err
    cmp x19, x21
    b.ge parse_val_eof
    ldrb w9, [x20, x19]
    cbz w9, parse_val_err       // shouldn't happen

    cmp w9, '{'
    b.eq parse_val_obj
    cmp w9, '['
    b.eq parse_val_arr
    cmp w9, '"'
    b.eq parse_val_str
    cmp w9, 't'
    b.eq parse_val_true
    cmp w9, 'f'
    b.eq parse_val_false
    cmp w9, 'n'
    b.eq parse_val_null
    cmp w9, '-'
    b.eq parse_val_num
    cmp w9, '0'
    b.eq parse_val_num
    cmp w9, '1'
    b.lt parse_val_err
    cmp w9, '9'
    b.gt parse_val_err
    // digit 1-9
    b parse_val_num

parse_val_obj:
    bl parse_object
    b parse_val_done
parse_val_arr:
    bl parse_array
    b parse_val_done
parse_val_str:
    bl parse_string
    b parse_val_done
parse_val_true:
    bl parse_true
    b parse_val_done
parse_val_false:
    bl parse_false
    b parse_val_done
parse_val_null:
    bl parse_null
    b parse_val_done
parse_val_num:
    bl parse_number
    b parse_val_done

parse_val_eof:
    mov x22, x19
    mov x0, -1
    b parse_val_exit
parse_val_err:
    mov x22, x19
    mov x0, -1
    b parse_val_exit
parse_val_done:
    // x0 already set
parse_val_exit:
    ldp x29, x30, [sp], #16
    ret

// parse_object
parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, 1             // skip '{'
    bl parse_ws
    cbnz x0, parse_obj_err
    cmp x19, x21
    b.ge parse_obj_eof
    ldrb w9, [x20, x19]
    cmp w9, '}'
    b.eq parse_obj_empty

parse_obj_member_loop:
    bl parse_string
    cbnz x0, parse_obj_err
    bl parse_ws
    cbnz x0, parse_obj_err
    cmp x19, x21
    b.ge parse_obj_eof
    ldrb w9, [x20, x19]
    cmp w9, ':'
    b.ne parse_obj_err_char
    add x19, x19, 1
    bl parse_ws
    cbnz x0, parse_obj_err
    bl parse_value
    cbnz x0, parse_obj_err
    bl parse_ws
    cbnz x0, parse_obj_err
    cmp x19, x21
    b.ge parse_obj_eof
    ldrb w9, [x20, x19]
    cmp w9, '}'
    b.eq parse_obj_end
    cmp w9, ','
    b.ne parse_obj_err_char
    add x19, x19, 1
    bl parse_ws
    cbnz x0, parse_obj_err
    b parse_obj_member_loop

parse_obj_empty:
    add x19, x19, 1
    mov x0, 0
    b parse_obj_exit
parse_obj_end:
    add x19, x19, 1
    mov x0, 0
    b parse_obj_exit

parse_obj_eof:
    mov x22, x19
    mov x0, -1
    b parse_obj_exit
parse_obj_err_char:
    mov x22, x19
    mov x0, -1
    b parse_obj_exit
parse_obj_err:
    // error already set by subcall
    b parse_obj_exit

parse_obj_exit:
    ldp x29, x30, [sp], #16
    ret

// parse_array
parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, 1             // skip '['
    bl parse_ws
    cbnz x0, parse_arr_err
    cmp x19, x21
    b.ge parse_arr_eof
    ldrb w9, [x20, x19]
    cmp w9, ']'
    b.eq parse_arr_empty

parse_arr_value_loop:
    bl parse_value
    cbnz x0, parse_arr_err
    bl parse_ws
    cbnz x0, parse_arr_err
    cmp x19, x21
    b.ge parse_arr_eof
    ldrb w9, [x20, x19]
    cmp w9, ']'
    b.eq parse_arr_end
    cmp w9, ','
    b.ne parse_arr_err_char
    add x19, x19, 1
    bl parse_ws
    cbnz x0, parse_arr_err
    b parse_arr_value_loop

parse_arr_empty:
    add x19, x19, 1
    mov x0, 0
    b parse_arr_exit
parse_arr_end:
    add x19, x19, 1
    mov x0, 0
    b parse_arr_exit

parse_arr_eof:
    mov x22, x19
    mov x0, -1
    b parse_arr_exit
parse_arr_err_char:
    mov x22, x19
    mov x0, -1
    b parse_arr_exit
parse_arr_err:
    b parse_arr_exit

parse_arr_exit:
    ldp x29, x30, [sp], #16
    ret

// parse_document: ws value ws end
parse_document:
    stp x29, x30, [sp, #-16]!
    bl parse_ws
    cbnz x0, parse_doc_err
    bl parse_value
    cbnz x0, parse_doc_err
    bl parse_ws
    cbnz x0, parse_doc_err
    cmp x19, x21
    b.ne parse_doc_extra
    mov x0, 0
    b parse_doc_exit
parse_doc_extra:
    mov x22, x19
    mov x0, -1
    b parse_doc_exit
parse_doc_err:
    b parse_doc_exit
parse_doc_exit:
    ldp x29, x30, [sp], #16
    ret

// _start
_start:
    // set up globals
    adrp x20, buffer
    add x20, x20, :lo12:buffer
    mov x19, 0          // pos
    mov x21, 0          // length
    mov x23, 0          // depth (unused)

read_loop:
    mov x0, 0           // stdin
    add x1, x20, x21
    mov x2, 65536
    sub x2, x2, x21
    mov x8, 63
    svc #0
    cmp x0, 0
    b.le read_done
    add x21, x21, x0
    cmp x21, 65536
    b.lt read_loop

read_done:
    cbz x21, empty_input
    bl parse_document
    cbnz x0, parse_failed
    bl print_ok
    mov x0, 0
    b exit_prog

empty_input:
    mov x22, 0
    b parse_failed

parse_failed:
    bl print_error
    mov x0, 1

exit_prog:
    mov x8, 93
    svc #0
