.section .rodata
space: .byte ' '
newline: .byte 0x0A

.section .bss
.equ BUFSIZE, 4096
buffer: .zero BUFSIZE

.section .text
.globl _start

# write_buf: write buffer at rsi, length rdx to stdout (fd=1)
write_buf:
    mov rax, 1          # sys_write
    mov rdi, 1          # fd = stdout
    syscall
    ret

# write_number: convert unsigned 64-bit integer in rdi to decimal and write to stdout
write_number:
    sub rsp, 32         # allocate 32-byte buffer on stack (16-byte aligned)
    lea rsi, [rsp + 32] # point to end of buffer
    test rdi, rdi
    jz .zero
.loop:
    mov rax, rdi
    xor rdx, rdx
    mov rcx, 10
    div rcx             # rax = quotient, rdx = remainder
    add dl, '0'
    dec rsi
    mov [rsi], dl
    mov rdi, rax
    test rax, rax
    jnz .loop
    jmp .write
.zero:
    dec rsi
    mov byte ptr [rsi], '0'
.write:
    lea rdx, [rsp + 32]
    sub rdx, rsi        # length
    call write_buf
    add rsp, 32
    ret

_start:
    # Initialize counters
    xor r12, r12        # lines
    xor r13, r13        # words
    xor r14, r14        # bytes
    xor r15, r15        # in_word (0 = false, 1 = true)

.read_loop:
    mov rax, 0          # sys_read
    mov rdi, 0          # fd = stdin
    lea rsi, [rip + buffer]
    mov rdx, BUFSIZE
    syscall
    test rax, rax
    jz .done_reading    # EOF

    add r14, rax        # add to byte count

    # Process buffer
    mov rcx, rax        # byte count
.process_loop:
    lodsb               # load byte at rsi into al, increment rsi
    cmp al, 0x0A        # newline
    je .newline
    cmp al, 0x09        # tab
    je .whitespace
    cmp al, 0x0B        # vertical tab
    je .whitespace
    cmp al, 0x0C        # form feed
    je .whitespace
    cmp al, 0x0D        # carriage return
    je .whitespace
    cmp al, 0x20        # space
    je .whitespace

    # Non-whitespace byte
    test r15, r15
    jnz .continue
    mov r15, 1
    inc r13             # new word
    jmp .continue

.newline:
    inc r12             # line count
    test r15, r15
    jz .continue
    mov r15, 0
    jmp .continue

.whitespace:
    test r15, r15
    jz .continue
    mov r15, 0

.continue:
    dec rcx
    jnz .process_loop

    jmp .read_loop

.done_reading:
    # Output lines, words, bytes
    mov rdi, r12
    call write_number
    lea rsi, [rip + space]
    mov rdx, 1
    call write_buf

    mov rdi, r13
    call write_number
    lea rsi, [rip + space]
    mov rdx, 1
    call write_buf

    mov rdi, r14
    call write_number
    lea rsi, [rip + newline]
    mov rdx, 1
    call write_buf

    # Exit
    mov rax, 60         # sys_exit
    xor rdi, rdi        # status 0
    syscall
