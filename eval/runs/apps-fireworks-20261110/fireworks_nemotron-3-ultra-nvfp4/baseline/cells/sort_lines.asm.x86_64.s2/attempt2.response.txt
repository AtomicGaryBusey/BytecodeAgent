    jz .exit_success
    xor r15, r15                    # index = 0
    lea r13, [rip + line_array]
.output_loop:
    mov rsi, qword ptr [r13 + r15*16]      # line.start
    mov rdx, qword ptr [r13 + r15*16 + 8]  # line.len
    # write line
    mov rax, SYS_WRITE
    mov rdi, 1
    syscall
    # write newline
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    inc r15
    cmp r15, r14
    jne .output_loop

.exit_success:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
