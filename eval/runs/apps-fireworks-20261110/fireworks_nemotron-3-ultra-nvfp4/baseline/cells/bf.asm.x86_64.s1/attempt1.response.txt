.section .bss
.align 4096
stdin_buf:
    .zero 65536
bracket_match:
    .zero 65536*4
tape:
    .zero 30000
output_buf:
    .zero 65536
bracket_stack:
    .zero 1000*4

.section .text
.globl _start
_start:
    # read all stdin into stdin_buf
    lea r12, [stdin_buf]   # buffer base
    xor r13, r13           # total_read
    mov r14, 65536         # max size
.read_loop:
    mov rax, 0             # sys_read
    mov rdi, 0             # fd=0
    lea rsi, [r12 + r13]   # current position
    mov rdx, r14
    sub rdx, r13           # remaining
    syscall
    test rax, rax
    jz .read_done
    add r13, rax
    cmp r13, r14
    jne .read_loop
.read_done:
    # r13 = total bytes read
    # find first '!' (0x21)
    lea rdi, [stdin_buf]
    mov rcx, r13
    xor rax, rax
    mov al, 0x21
    repne scasb
    jnz .no_bang
    # found '!'
    lea rsi, [rdi - 1]
    sub rsi, stdin_buf
    mov r15, rsi           # prog_len
    mov r14, rcx           # input_len
    lea r8, [rdi]          # input_start
    jmp .have_split
.no_bang:
    mov r15, r13           # prog_len
    xor r14, r14           # input_len
    lea r8, [stdin_buf]    # input_start (unused)
.have_split:
    # r15 = prog_len, r14 = input_len, r8 = input_start
    # bracket validation
    lea rbp, [bracket_stack]
    xor rbx, rbx           # stack depth
    xor r12, r12           # ip
.bracket_loop:
    cmp r12, r15
    jge .bracket_done
    movzx eax, byte ptr [stdin_buf + r12]
    cmp al, '['
    je .push_bracket
    cmp al, ']'
    je .pop_bracket
    inc r12
    jmp .bracket_loop
.push_bracket:
    cmp rbx, 1000
    jge .bracket_error
    mov dword ptr [bracket_stack + rbx*4], r12d
    inc rbx
    inc r12
    jmp .bracket_loop
.pop_bracket:
    test rbx, rbx
    jz .bracket_error
    dec rbx
    mov edx, dword ptr [bracket_stack + rbx*4]
    mov dword ptr [bracket_match + r12*4], edx
    mov dword ptr [bracket_match + rdx*4], r12d
    inc r12
    jmp .bracket_loop
.bracket_done:
    test rbx, rbx
    jnz .bracket_error
    jmp .interpret
.bracket_error:
    mov rax, 60
    mov rdi, 1
    syscall

.interpret:
    xor rbx, rbx           # dp
    xor r12, r12           # ip
    xor r13, r13           # inp
    xor rsi, rsi           # out_ptr
    # r15 = prog_len, r14 = input_len, r8 = input_start
.interpret_loop:
    cmp r12, r15
    jge .interpret_done
    movzx eax, byte ptr [stdin_buf + r12]
    cmp al, '>'
    je .cmd_right
    cmp al, '<'
    je .cmd_left
    cmp al, '+'
    je .cmd_inc
    cmp al, '-'
    je .cmd_dec
    cmp al, '.'
    je .cmd_out
    cmp al, ','
    je .cmd_in
    cmp al, '['
    je .cmd_jz
    cmp al, ']'
    je .cmd_jnz
    inc r12
    jmp .interpret_loop

.cmd_right:
    inc rbx
    inc r12
    jmp .interpret_loop

.cmd_left:
    dec rbx
    inc r12
    jmp .interpret_loop

.cmd_inc:
    inc byte ptr [tape + rbx]
    inc r12
    jmp .interpret_loop

.cmd_dec:
    dec byte ptr [tape + rbx]
    inc r12
    jmp .interpret_loop

.cmd_out:
    mov al, byte ptr [tape + rbx]
    mov byte ptr [output_buf + rsi], al
    inc rsi
    inc r12
    jmp .interpret_loop

.cmd_in:
    cmp r13, r14
    jge .cmd_in_eof
    mov al, byte ptr [r8 + r13]
    mov byte ptr [tape + rbx], al
    inc r13
    inc r12
    jmp .interpret_loop
.cmd_in_eof:
    mov byte ptr [tape + rbx], 0
    inc r12
    jmp .interpret_loop

.cmd_jz:
    cmp byte ptr [tape + rbx], 0
    jne .cmd_jz_not_zero
    mov edx, dword ptr [bracket_match + r12*4]
    lea r12, [rdx + 1]
    jmp .interpret_loop
.cmd_jz_not_zero:
    inc r12
    jmp .interpret_loop

.cmd_jnz:
    cmp byte ptr [tape + rbx], 0
    je .cmd_jnz_zero
    mov edx, dword ptr [bracket_match + r12*4]
    lea r12, [rdx + 1]
    jmp .interpret_loop
.cmd_jnz_zero:
    inc r12
    jmp .interpret_loop

.interpret_done:
    test rsi, rsi
    jz .exit_success
    mov rdx, rsi
    lea rsi, [output_buf]
    mov rax, 1
    mov rdi, 1
    syscall
.exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
