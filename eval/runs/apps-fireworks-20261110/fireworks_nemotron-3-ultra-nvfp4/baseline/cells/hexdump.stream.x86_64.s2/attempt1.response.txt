.section .rodata
hex_table: .ascii "0123456789abcdef"
zero_line: .ascii "00000000\n"
zero_line_len = . - zero_line

.section .bss
.align 16
input_buffer: .zero 65536
input_len: .zero 8

.section .text
.globl _start
_start:
    # Stack: reserve 128 bytes for line buffer (16-byte aligned)
    sub rsp, 128
    mov r15, rsp                # r15 = line buffer

    # Read all input into input_buffer
    mov r14, input_buffer       # base
    xor r12, r12                # total bytes read = 0
    mov r13, 65536              # remaining capacity

.read_loop:
    mov rdi, 0                  # stdin
    lea rsi, [r14 + r12]        # current position
    mov rdx, r13                # remaining
    xor rax, rax                # sys_read = 0
    syscall
    test rax, rax
    jz .read_done               # EOF
    add r12, rax                # update total
    sub r13, rax                # update remaining
    jnz .read_loop              # continue if buffer not full (though input <= 65536)

.read_done:
    mov qword ptr [rip + input_len], r12
    test r12, r12
    jz .print_zero              # empty input

    xor r13, r13                # current offset = 0

.line_loop:
    cmp r13, r12
    jae .all_lines_done

    # n = min(16, r12 - r13)
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    cmova rax, 16               # rax = n (1..16)
