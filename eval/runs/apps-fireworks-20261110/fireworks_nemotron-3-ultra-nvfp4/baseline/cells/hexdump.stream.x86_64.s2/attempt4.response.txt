.section .rodata
hex_table: .ascii "0123456789abcdef"
zero_line: .ascii "00000000\n"
zero_line_len = . - zero_line

.section .bss
.align 16
input_buffer: .zero 65536
input_len: .zero 8

.section .text
.globl _start
_start:
    sub rsp, 128
    mov r15, rsp                # r15 = line buffer (128 bytes)
    lea r10, [rip + hex_table]  # r10 = hex_table address
    lea r14, [rip + input_buffer] # r14 = input_buffer address
    xor r12, r12                # total bytes read = 0
    mov r13, 65536              # remaining capacity

.read_loop:
    mov rdi, 0                  # stdin
    lea rsi, [r14 + r12]        # current position
    mov rdx, r13                # remaining
    xor rax, rax                # sys_read = 0
    syscall
    test rax, rax
    jz .read_done               # EOF
    add r12, rax                # update total
    sub r13, rax                # update remaining
    jnz .read_loop

.read_done:
    mov qword ptr [rip + input_len], r12
    test r12, r12
    jz .print_zero              # empty input

    xor r13, r13                # current offset = 0

.line_loop:
    cmp r13, r12
    jae .all_lines_done

    # n = min(16, r12 - r13)
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe .n_ok
    mov rax, 16
.n_ok:
    mov r8b, al                 # r8b = n (1..16)

    # Build line in buffer at r15
    mov rsi, r15
    mov edi, r13d               # offset (32 bits)
    mov rcx, 8
.offset_loop:
    rol edi, 4
    mov eax, edi
    and eax, 0xF
    mov bl, byte ptr [r10 + rax]
    mov byte ptr [rsi], bl
    inc rsi
    dec rcx
    jnz .offset_loop
    # two spaces
    mov word ptr [rsi], 0x2020
    add rsi, 2

    # Hex bytes
    lea rdi, [r14 + r13]        # pointer to input bytes
    xor r9b, r9b                # i = 0
.hex_loop:
    cmp r9b, r8b
    jae .hex_padding
    # print byte as two hex digits + space
    movzx eax, byte ptr [rdi + r9]
    mov ecx, eax
    shr ecx, 4
    mov bl, byte ptr [r10 + rcx]
    mov byte ptr [rsi], bl
    inc rsi
    mov ecx, eax
    and ecx, 0xF
    mov bl, byte ptr [r10 + rcx]
    mov byte ptr [rsi], bl
    inc rsi
    mov byte ptr [rsi], 32
    inc rsi
    jmp .hex_next
.hex_padding:
    mov word ptr [rsi], 0x2020
    mov byte ptr [rsi+2], 32
    add rsi, 3
.hex_next:
    cmp r9b, 7
    jne .no_extra_space
    mov byte ptr [rsi], 32
    inc rsi
.no_extra_space:
    inc r9b
    cmp r9b, 16
    jb .hex_loop

    # ASCII column
    mov byte ptr [rsi], 32
    inc rsi
    mov byte ptr [rsi], '|'
    inc rsi
    xor r9b, r9b
.ascii_loop:
    cmp r9b, r8b
    jae .ascii_done
    movzx eax, byte ptr [rdi + r9]
    cmp al, 0x20
    jb .ascii_dot
    cmp al, 0x7E
    ja .ascii_dot
    mov byte ptr [rsi], al
    jmp .ascii_next
.ascii_dot:
    mov byte ptr [rsi], '.'
.ascii_next:
    inc rsi
    inc r9b
    jmp .ascii_loop
.ascii_done:
    mov byte ptr [rsi], '|'
    inc rsi
    mov byte ptr [rsi], 10
    inc rsi

    # Write line
    mov rdx, rsi
    sub rdx, r15
    mov rdi, 1
    mov rsi, r15
    mov rax, 1
    syscall

    # Next line
    add r13, r8
    jmp .line_loop

.all_lines_done:
    # Print final offset (r12)
    mov rsi, r15
    mov edi, r12d
    mov rcx, 8
.offset_loop2:
    rol edi, 4
    mov eax, edi
    and eax, 0xF
    mov bl, byte ptr [r10 + rax]
    mov byte ptr [rsi], bl
    inc rsi
    dec rcx
    jnz .offset_loop2
    mov byte ptr [rsi], 10
    inc rsi
    mov rdx, rsi
    sub rdx, r15
    mov rdi, 1
    mov rsi, r15
    mov rax, 1
    syscall

    xor rdi, rdi
    mov rax, 60
    syscall

.print_zero:
    mov rdi, 1
    lea rsi, [rip + zero_line]
    mov rdx, zero_line_len
    mov rax, 1
    syscall
    xor rdi, rdi
    mov rax, 60
    syscall
