.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
.align 4096
input_buffer: .zero 32768
line_ptrs: .zero 8000
line_lens: .zero 4000
input_len: .zero 8
line_count: .zero 4

.section .text
_start:
    # ---- read all input ----
    mov r12, 0
    lea r13, [rip + input_buffer]
    mov r14, 32768
.read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, r14
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add r13, rax
    sub r14, rax
    jnz .read_loop
.read_done:
    mov qword ptr [rip + input_len], r12

    # ---- parse lines ----
    lea rsi, [rip + input_buffer]
    mov rcx, r12
    lea rdi, [rip + input_buffer]
    mov r15d, 0
.parse_loop:
    test rcx, rcx
    jz .parse_after_loop
    cmp byte ptr [rsi], 0x0A
    je .newline_found
    inc rsi
    dec rcx
    jmp .parse_loop
.newline_found:
    mov rax, rsi
    sub rax, rdi
    lea rbx, [rip + line_ptrs]
    mov qword ptr [rbx + r15*8], rdi
    lea rbx, [rip + line_lens]
    mov dword ptr [rbx + r15*4], eax
    inc r15d
    inc rsi
    dec rcx
    mov rdi, rsi
    jmp .parse_loop
.parse_after_loop:
    cmp rdi, rsi
    jge .parse_done
    mov rax, rsi
    sub rax, rdi
    lea rbx, [rip + line_ptrs]
    mov qword ptr [rbx + r15*8], rdi
    lea rbx, [rip + line_lens]
    mov dword ptr [rbx + r15*4], eax
    inc r15d
.parse_done:
    mov dword ptr [rip + line_count], r15d

    # ---- insertion sort ----
    mov r12d, 1
.sort_outer:
    cmp r12d, dword ptr [rip + line_count]
    jge .sort_done
    mov r13d, r12d
.sort_inner:
    test r13d, r13d
    jz .sort_inner_done
    lea rdi, [rip + line_ptrs]
    mov r8, qword ptr [rdi + (r13-1)*8]
    mov r9, qword ptr [rdi + r13*8]
    lea rdi, [rip + line_lens]
    mov r10d, dword ptr [rdi + (r13-1)*4]
    mov r11d, dword ptr [rdi + r13*4]
    call compare_lines
    cmp rax, 1
    jne .sort_inner_done
    lea rdi, [rip + line_ptrs]
    mov rax, qword ptr [rdi + (r13-1)*8]
    mov rcx, qword ptr [rdi + r13*8]
    mov qword ptr [rdi + (r13-1)*8], rcx
    mov qword ptr [rdi + r13*8], rax
    lea rdi, [rip + line_lens]
    mov eax, dword ptr [rdi + (r13-1)*4]
    mov ecx, dword ptr [rdi + r13*4]
    mov dword ptr [rdi + (r13-1)*4], ecx
    mov dword ptr [rdi + r13*4], eax
    dec r13d
    jmp .sort_inner
.sort_inner_done:
    inc r12d
    jmp .sort_outer
.sort_done:

    # ---- output sorted lines ----
    mov r12d, 0
.output_loop:
    cmp r12d, dword ptr [rip + line_count]
    jge .output_done
    lea rdi, [rip + line_ptrs]
    mov rsi, qword ptr [rdi + r12*8]
    lea rdi, [rip + line_lens]
    mov edx, dword ptr [rdi + r12*4]
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r12d
    jmp .output_loop
.output_done:

    mov rax, 60
    xor rdi, rdi
    syscall

compare_lines:
    mov eax, r10d
    cmp eax, r11d
    cmovg eax, r11d
    test eax, eax
    jz .compare_lengths
    mov ecx, eax
    mov rsi, r8
    mov rdi, r9
.compare_loop:
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rdi]
    cmp al, bl
    jne .compare_diff
    inc rsi
    inc rdi
    dec ecx
    jnz .compare_loop
.compare_lengths:
    mov eax, r10d
    cmp eax, r11d
    je .compare_equal
    jg .compare_greater
    mov eax, -1
    ret
.compare_diff:
    ja .compare_greater
    mov eax, -1
    ret
.compare_greater:
    mov eax, 1
    ret
.compare_equal:
    xor eax, eax
    ret
