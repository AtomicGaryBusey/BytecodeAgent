.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
.align 4096
input_buffer: .zero 32768
line_ptrs: .zero 8000      # 1000 * 8
line_lens: .zero 4000      # 1000 * 4
input_len: .zero 8
line_count: .zero 4

.section .text
_start:
    # ---- read all input ----
    mov r12, 0                  # total bytes read
    mov r13, input_buffer       # current buffer pointer
    mov r14, 32768              # remaining space
.read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    mov rsi, r13
    mov rdx, r14
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add r13, rax
    sub r14, rax
    jnz .read_loop
.read_done:
    mov qword ptr [rip + input_len], r12

    # ---- parse lines ----
    mov rsi, input_buffer
    mov rcx, r12                # input length
    mov rdi, input_buffer       # line start
    mov r15d, 0                 # line count
.parse_loop:
    test rcx, rcx
    jz .parse_after_loop
    cmp byte ptr [rsi], 0x0A
    je .newline_found
    inc rsi
    dec rcx
    jmp .parse_loop
.newline_found:
    mov rax, rsi
    sub rax, rdi                # line length
    mov rbx, line_ptrs
    mov qword ptr [rbx + r15*8], rdi
    mov rbx, line_lens
    mov dword ptr [rbx + r15*4], eax
    inc r15d
    inc rsi
    dec rcx
    mov rdi, rsi
    jmp .parse_loop
.parse_after_loop:
    cmp rdi, rsi
    jge .parse_done
    mov rax, rsi
    sub rax, rdi
    mov rbx, line_ptrs
    mov qword ptr [rbx + r15*8], rdi
    mov rbx, line_lens
    mov dword ptr [rbx + r15*4], eax
    inc r15d
.parse_done:
    mov dword ptr [rip + line_count], r15d

    # ---- insertion sort ----
    mov r12d, 1                 # i
.sort_outer:
    cmp r12d, dword ptr [rip + line_count]
    jge .sort_done
    mov r13d, r12d              # j = i
.sort_inner:
    test r13d, r13d
    jz .sort_inner_done
    # load line[j-1] and line[j]
    mov rdi, line_ptrs
    mov r8, qword ptr [rdi + (r13-1)*8]
    mov r9, qword ptr [rdi + r13*8]
    mov rdi, line_lens
    mov r10d, dword ptr [rdi + (r13-1)*4]
    mov r11d, dword ptr [rdi + r13*4]
    call compare_lines
    cmp rax, 1
    jne .sort_inner_done
    # swap pointers
    mov rdi, line_ptrs
    mov rax, qword ptr [rdi + (r13-1)*8]
    mov rcx, qword ptr [rdi + r13*8]
    mov qword ptr [rdi + (r13-1)*8], rcx
    mov qword ptr [rdi + r13*8], rax
    # swap lengths
    mov rdi, line_lens
    mov eax, dword ptr [rdi + (r13-1)*4]
    mov ecx, dword ptr [rdi + r13*4]
    mov dword ptr [rdi + (r13-1)*4], ecx
    mov dword ptr [rdi + r13*4], eax
    dec r13d
    jmp .sort_inner
.sort_inner_done:
    inc r12d
    jmp .sort_outer
.sort_done:

    # ---- output sorted lines ----
    mov r12d, 0
.output_loop:
    cmp r12d, dword ptr [rip + line_count]
    jge .output_done
    mov rdi, line_ptrs
    mov rsi, qword ptr [rdi + r12*8]
    mov rdi, line_lens
    mov edx, dword ptr [rdi + r12*4]
    # write line
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    syscall
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r12d
    jmp .output_loop
.output_done:

    # ---- exit ----
    mov rax, 60                 # sys_exit
    xor rdi, rdi
    syscall

# compare_lines:
#   r8 = ptr1, r9 = ptr2, r10d = len1, r11d = len2
#   returns rax = 1 if line1 > line2, 0 if equal, -1 if line1 < line2
compare_lines:
    mov eax, r10d
    cmp eax, r11d
    cmovg eax, r11d             # eax = min(len1, len2)
    test eax, eax
    jz .compare_lengths
    mov ecx, eax
    mov rsi, r8
    mov rdi, r9
.compare_loop:
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rdi]
    cmp al, bl
    jne .compare_diff
    inc rsi
    inc rdi
    dec ecx
    jnz .compare_loop
.compare_lengths:
    mov eax, r10d
    cmp eax, r11d
    je .compare_equal
    jg .compare_greater
    mov eax, -1
    ret
.compare_diff:
    ja .compare_greater
    mov eax, -1
    ret
.compare_greater:
    mov eax, 1
    ret
.compare_equal:
    xor eax, eax
    ret
