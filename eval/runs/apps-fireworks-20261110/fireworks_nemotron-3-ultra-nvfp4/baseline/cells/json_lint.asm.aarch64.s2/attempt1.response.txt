.section .text
.globl _start

_start:
    // Initialize global pointers
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x20, input_len
    add  x20, x20, :lo12:input_len
    adrp x21, error_pos
    add  x21, x21, :lo12:error_pos

    // error_pos = -1
    mov  x0, #-1
    str  x0, [x21]

    // Read all stdin into buffer (max 65536 bytes)
    mov  x22, 0                  // total bytes read
    movz x23, #1, lsl 16         // capacity = 65536

read_loop:
    mov  x0, 0                   // fd = stdin
    add  x1, x19, x22            // buffer + total
    sub  x2, x23, x22            // remaining space
    mov  x8, 63                  // read syscall
    svc  #0
    cbz  x0, read_done           // EOF
    add  x22, x22, x0
    b    read_loop

read_done:
    str  x22, [x20]              // store input length

    // Parse JSON document
    mov  x0, 0
    bl   parse_document

    // Check result
    ldr  x0, [x21]
    cmp  x0, #-1
    b.ne error_case

    // Valid JSON
    bl   print_ok
    mov  x0, 0
    b    exit_prog

error_case:
    bl   print_error
    mov  x0, 1

exit_prog:
    mov  x8, 93                  // exit syscall
    svc  #0

// ------------------------------------------------------------
// parse_document: entry at x0=0, returns final position in x0
// ------------------------------------------------------------
parse_document:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp

    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne parse_doc_ret

    bl   skip_ws
    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne parse_doc_ret

    bl   parse_value
    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne parse_doc_ret

    bl   skip_ws
    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne parse_doc_ret

    ldr  x10, [x20]
    cmp  x0, x10
    b.eq parse_doc_ret

    // Extra non-whitespace bytes
    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne parse_doc_ret
    str  x0, [x21]

parse_doc_ret:
    ldp  x29, x30, [sp], #16
    ret

// ------------------------------------------------------------
// skip_ws: x0 = position, returns new position in x0
// ------------------------------------------------------------
skip_ws:
    stp  x29, x30, [sp, #-16]!
    mov  x29, sp

skip_ws_loop:
    ldr  x9, [x21]
    cmp  x9, #-1
    b.ne skip_ws_ret

    ldr  x10, [x20]
    cmp  x0, x10
    b.ge skip_ws_ret

    ldrb w11, [x19, x0]
    cmp  w11
