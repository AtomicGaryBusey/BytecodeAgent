.section .bss
.align 12
buffer:         .space 65536
input_len:      .quad 0
error_offset:   .quad 0
parse_index:    .quad 0
output_buf:     .space 32

.section .rodata
ok_str:         .ascii "ok\n"
error_prefix:   .ascii "error at "
newline:        .ascii "\n"

.section .text
.globl _start

_start:
    // Initialize globals in callee-saved registers
    adrp x19, buffer
    add  x19, x19, :lo12:buffer
    mov  x20, 0           // input_len
    mov  x21, 0           // parse_index
    mov  x22, -1          // error_offset (sentinel -1)

    // Read all input
read_loop:
    mov  x0, 65536
    sub  x0, x0, x20
    cbz  x0, read_done
    add  x1, x19, x20
    mov  x2, x0
    mov  x8, 63           // read
    svc  #0
    cbz  x0, read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    bl   parse_document

    // Check for error
    cmp  x22, -1
    b.eq print_ok

print_error:
    // Write "error at "
    adrp x0, error_prefix
    add  x0, x0, :lo12:error_prefix
    mov  x1, 9
    mov  x8, 64
    svc  #0

    // Convert error_offset (x22) to decimal
    mov  x0, x22
    adrp x1, output_buf
    add  x1, x1, :lo12:output_buf
    bl   utoa
    // x0 = length
    mov  x2, x0
    adrp x0, output_buf
    add  x0, x0, :lo12:output_buf
    mov  x1, x0
    mov  x8, 64
    svc  #0

    // Write newline
    adrp x0, newline
    add  x0, x0, :lo12:newline
    mov  x1, 1
    mov  x8, 64
    svc  #0

    mov  x0, 1
    b    exit_prog

print_ok:
    adrp x0, ok_str
    add  x0, x0, :lo12:ok_str
    mov  x1, 4
    mov  x8, 64
    svc  #0
    mov  x0, 0

exit_prog:
    mov  x8, 93
    svc  #0

// ============================================================
// Parser functions
// ============================================================
parse_document:
    bl   parse_ws
    bl   parse_value
    bl   parse_ws
    cmp  x21, x20
    b.ne doc_trailing_error
    ret

doc_trailing_error:
    cmp  x22, -1
    b.ne 1f
    mov  x22, x21
1:  ret

parse_ws:
1:  cmp  x21, x20
    b.ge 2f
    ldrb w0, [x19, x21]
    cmp  w0, 0x20
    b.eq 3f
    cmp  w0, 0x09
    b.eq 3f
    cmp  w0, 0x0A
    b.eq 3f
    cmp  w0, 0x0D
    b.eq 3f
2:  ret
3:  add  x21, x21, 1
    b    1b

parse_value:
    cmp  x22, -1
    b.ne val_err
    cmp  x21, x20
    b.ge val_eof
    ldrb w0, [x19, x21]
    cmp  w0, '{'
    b.eq parse_object
    cmp  w0, '['
    b.eq parse_array
    cmp  w0, '"'
    b.eq parse_string
    cmp  w0, 't'
    b.eq parse_true
    cmp  w0, 'f'
    b.eq parse_false
    cmp  w0, 'n'
    b.eq parse_null
    cmp  w0, '-'
    b.eq parse_number
    cmp  w0, '0'
    b.lt val_error
    cmp  w0, '9'
    b.gt val_error
    b    parse_number

val_eof:
    cmp  x22, -1
    b.ne val_err
    mov  x22, x20
val_err:
    ret

val_error:
    cmp  x22, -1
    b.ne val_err
    mov  x22, x21
    ret

parse_object:
    add  x21, x21, 1          // consume '{'
    bl   parse_ws
    cmp  x21, x20
    b.ge obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, '}'
    b.ne obj_parse_member
    add  x21, x21, 1
    ret

obj_parse_member:
    bl   parse_member
    cmp  x22, -1
    b.ne obj_err
    bl   parse_ws
obj_loop:
    cmp  x21, x20
    b.ge obj_eof
    ldrb w0, [x19, x21]
    cmp  w0, ','
    b.ne obj_check_close
    add  x21, x21, 1
    bl   parse_ws
    bl   parse_member
    cmp  x22, -1
    b.ne obj_err
    bl   parse_ws
    b    obj_loop

obj_check_close:
    cmp  w0, '}'
    b.ne obj_error
    add  x21, x21, 1
    ret

obj_error:
    cmp  x22, -1
    b.ne obj_err
    mov  x22, x21
obj_err:
    ret

obj_eof:
    cmp  x22, -1
    b.ne obj_err
    mov  x22, x20
    ret

parse_member:
    bl   parse_string
    cmp  x22, -1
    b.ne mem_err
    bl   parse_ws
    cmp  x21, x20
    b.ge mem_eof
    ldrb w0, [x19, x21]
    cmp  w0, ':'
    b.ne mem_error
    add  x21, x21, 1
    bl   parse_ws
    bl   parse_value
mem_err:
    ret
mem_eof:
    cmp  x22, -1
    b.ne mem_err
    mov  x22, x20
    ret
mem_error:
    cmp  x22, -1
    b.ne mem_err
    mov  x22, x21
    ret

parse_array:
    add  x21, x21, 1          // consume '['
    bl   parse_ws
    cmp  x21, x20
    b.ge arr_eof
    ldrb w0, [x19, x21]
    cmp  w0, ']'
    b.ne arr_parse_value
    add  x21, x21, 1
    ret

arr_parse_value:
    bl   parse_value
    cmp  x22, -1
    b.ne arr_err
    bl   parse_ws
arr_loop:
    cmp  x21, x20
    b.ge arr_eof
    ldrb w0, [x19, x21]
    cmp  w0, ','
    b.ne arr_check_close
    add  x21, x21, 1
    bl   parse_ws
    bl   parse_value
    cmp  x22, -1
    b.ne arr_err
    bl   parse_ws
    b    arr_loop

arr_check_close:
    cmp  w0, ']'
    b.ne arr_error
    add  x21, x21, 1
    ret

arr_error:
    cmp  x22, -1
    b.ne arr_err
    mov  x22, x21
arr_err:
    ret

arr_eof:
    cmp  x22, -1
    b.ne arr_err
    mov  x22, x20
    ret

parse_string:
    add  x21, x21, 1          // consume '"'
str_loop:
    cmp  x21, x20
    b.ge str_eof
    ldrb w0, [x19, x21]
    cmp  w0, '"'
    b.eq str_end
    cmp  w0, '\\'
    b.eq str_escape
    cmp  w0, 0x20
    b.lt str_error
    add  x21, x21, 1
    b    str_loop

str_end:
    add  x21, x21, 1
    ret

str_escape:
    add  x21, x21, 1
    cmp  x21, x20
    b.ge str_eof
    ldrb w0, [x19, x21]
    // Check for simple escapes
    cmp  w0, '"'
    b.eq esc_simple
    cmp  w0, '\\'
    b.eq esc_simple
    cmp  w0, '/'
    b.eq esc_simple
    cmp  w0, 'b'
    b.eq esc_simple
    cmp  w0, 'f'
    b.eq esc_simple
    cmp  w0, 'n'
    b.eq esc_simple
    cmp  w0, 'r'
    b.eq esc_simple
    cmp  w0, 't'
    b.eq esc_simple
    cmp  w0, 'u'
    b.eq esc_unicode
    b    str_error

esc_simple:
    add  x21, x21, 1
    b    str_loop

esc_unicode:
    add  x21, x21, 1
    // Need 4 hex digits
    add  x0, x21, 4
    cmp  x0, x20
    b.gt str_eof
    mov  x0, 4
esc_hex_loop:
    ldrb w1, [x19, x21]
    cmp  w1, '0'
    b.lt str_error
    cmp  w1, '9'
    b.le esc_hex_ok
    cmp  w1, 'a'
    b.lt esc_check_upper
    cmp  w1, 'f'
    b.le esc_hex_ok
    b    str_error
esc_check_upper:
    cmp  w1, 'A'
    b.lt str_error
    cmp  w1, 'F'
    b.gt str_error
esc_hex_ok:
    add  x21, x21, 1
    subs x0, x0, 1
    b.ne esc_hex_loop
    b    str_loop

str_error:
    cmp  x22, -1
    b.ne str_err
    mov  x22, x21
str_err:
    ret

str_eof:
    cmp  x22, -1
    b.ne str_err
    mov  x22, x20
    ret

parse_true:
    add  x0, x21, 4
    cmp  x0, x20
    b.gt lit_eof
    ldrb w0, [x19, x21]
    cmp  w0, 't'
    b.ne lit_error
    ldrb w0, [x19, x21, 1]
    cmp  w0, 'r'
    b.ne lit_error
    ldrb w0, [x19, x21, 2]
    cmp  w0, 'u'
    b.ne lit_error
    ldrb w0, [x19, x21, 3]
    cmp  w0, 'e'
    b.ne lit_error
    add  x21, x21, 4
    ret

parse_false:
    add  x0, x21, 5
    cmp  x0, x20
    b.gt lit_eof
    ldrb w0, [x19, x21]
    cmp  w0, 'f'
    b.ne lit_error
    ldrb w0, [x19, x21, 1]
    cmp  w0, 'a'
    b.ne lit_error
    ldrb w0, [x19, x21, 2]
    cmp  w0, 'l'
    b.ne lit_error
    ldrb w0, [x19, x21, 3]
    cmp  w0, 's'
    b.ne lit_error
    ldrb w0, [x19, x21, 4]
    cmp  w0, 'e'
    b.ne lit_error
    add  x21, x21, 5
    ret

parse_null:
    add  x0, x21, 4
    cmp  x0, x20
    b.gt lit_eof
    ldrb w0, [x19, x21]
    cmp  w0, 'n'
    b.ne lit_error
    ldrb w0, [x19, x21, 1]
    cmp  w0, 'u'
    b.ne lit_error
    ldrb w0, [x19, x21, 2]
    cmp  w0, 'l'
    b.ne lit_error
    ldrb w0, [x19, x21, 3]
    cmp  w0, 'l'
    b.ne lit_error
    add  x21, x21, 4
    ret

lit_error:
    cmp  x22, -1
    b.ne lit_err
    mov  x22, x21
lit_err:
    ret

lit_eof:
    cmp  x22, -1
    b.ne lit_err
    mov  x22, x20
    ret

parse_number:
    // Optional '-'
    cmp  x21, x20
    b.ge num_eof
    ldrb w0, [x19, x21]
    cmp  w0, '-'
    b.ne num_int
    add  x21, x21, 1

num_int:
    cmp  x21, x20
    b.ge num_eof
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.eq num_int_zero
    cmp  w0, '1'
    b.lt num_error
    cmp  w0, '9'
    b.gt num_error
    // Digit 1-9
    add  x21, x21, 1
    // More digits
1:  cmp  x21, x20
    b.ge num_after_int
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.lt num_after_int
    cmp  w0, '9'
    b.gt num_after_int
    add  x21, x21, 1
    b    1b

num_int_zero:
    add  x21, x21, 1

num_after_int:
    // Optional fraction
    cmp  x21, x20
    b.ge num_after_frac
    ldrb w0, [x19, x21]
    cmp  w0, '.'
    b.ne num_after_frac
    add  x21, x21, 1
    cmp  x21, x20
    b.ge num_eof
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.lt num_error
    cmp  w0, '9'
    b.gt num_error
    // At least one digit
1:  add  x21, x21, 1
    cmp  x21, x20
    b.ge num_after_frac
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.lt num_after_frac
    cmp  w0, '9'
    b.gt num_after_frac
    b    1b

num_after_frac:
    // Optional exponent
    cmp  x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp  w0, 'e'
    b.eq num_exp
    cmp  w0, 'E'
    b.ne num_done
num_exp:
    add  x21, x21, 1
    cmp  x21, x20
    b.ge num_eof
    ldrb w0, [x19, x21]
    cmp  w0, '+'
    b.eq num_exp_sign
    cmp  w0, '-'
    b.eq num_exp_sign
    cmp  w0, '0'
    b.lt num_error
    cmp  w0, '9'
    b.gt num_error
    b    num_exp_digits

num_exp_sign:
    add  x21, x21, 1
    cmp  x21, x20
    b.ge num_eof
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.lt num_error
    cmp  w0, '9'
    b.gt num_error

num_exp_digits:
1:  add  x21, x21, 1
    cmp  x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp  w0, '0'
    b.lt num_done
    cmp  w0, '9'
    b.gt num_done
    b    1b

num_done:
    ret

num_error:
    cmp  x22, -1
    b.ne num_err
    mov  x22, x21
num_err:
    ret

num_eof:
    cmp  x22, -1
    b.ne num_err
    mov  x22, x20
    ret

// ============================================================
// utoa: convert unsigned 64-bit integer in x0 to decimal string at x1
// Returns length in x0.
// Uses stack for temporary digit storage (max 20 digits).
// ============================================================
utoa:
    mov  x4, x1           // output buffer
    mov  x5, x0           // value
    cmp  x5, 0
    b.ne 1f
    mov  w0, '0'
    strb w0, [x4]
    mov  x0, 1
    ret

1:  sub  sp, sp, 32       // allocate 32 bytes on stack (aligned)
    mov  x6, sp           // digit buffer
    mov  x7, 0            // digit count
    mov  x8, 10

2:  udiv x9, x5, x8       // quotient
    msub x10, x9, x8, x5  // remainder
    add  w10, w10, '0'
    strb w10, [x6, x7]
    add  x7, x7, 1
    mov  x5, x9
    cbnz x5, 2b

    // Now reverse digits into output buffer
    mov  x5, x4           // output pointer
    sub  x7, x7, 1        // last digit index
3:  ldrb w8, [x6, x7]
    strb w8, [x5], 1
    subs x7, x7, 1
    b.ge 3b

    add  sp, sp, 32
    sub  x0, x5, x4       // length
    ret
