.global _start

.section .bss
.align 12
input_buffer: .space 65536
tape: .space 30000
output_buffer: .space 65536
bracket_stack: .space 4000
jump_table: .space 262144

.section .text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ STDERR, 2

_start:
    // Initialize registers
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0                     // total_bytes_read = 0
    mov x21, #0                     // program_length = 0
    mov x22, #0                     // program_input_start = 0
    mov x23, #0                     // program_input_length = 0
    adrp x24, tape
    add x24, x24, :lo12:tape        // tape_ptr = tape base
    mov x25, #0                     // ip = 0
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table  // jump_table base
    adrp x27, output_buffer
    add x27, x27, :lo12:output_buffer // output_buffer base
    mov x28, x27                    // output_ptr = output_buffer base

    // Read all input into input_buffer
1:  // read loop
    mov x0, #STDIN
    mov x1, x19
    add x1, x1, x20                 // buf = input_buffer + total_bytes_read
    mov x2, #65536
    sub x2, x2, x20                 // count = 65536 - total_bytes_read
    cmp x2, #0
    beq 2f                          // buffer full
    mov x8, #SYS_READ
    svc #0
    cbz x0, 2f                      // EOF
    add x20, x20, x0                // total_bytes_read += bytes_read
    b 1b

2:  // Find first '!' (0x21)
    mov x21, #0                     // program_length = 0
    mov x0, x19                     // input_buffer base
3:  cmp x21, x20
    bge 4f                          // reached end of input
    ldrb w1, [x0, x21]
    cmp w1, #0x21
    beq 5f                          // found '!'
    add x21, x21, #1
    b 3b

4:  // No '!' found
    mov x22, x20                    // program_input_start = total_bytes_read
    mov x23, #0                     // program_input_length = 0
    b 6f

5:  // Found '!' at index x21
    add x22, x21, #1                // program_input_start = program_length + 1
    sub x23, x20, x22               // program_input_length = total - program_input_start

6:  // Bracket matching and jump table construction
    adrp x0, bracket_stack
    add x0, x0, :lo12:bracket_stack // x0 = bracket_stack base
    mov x1, #0                      // stack pointer (index in 32-bit words)
    mov x2, #0                      // i = 0
7:  cmp x2, x21
    bge 8f                          // done scanning program
    ldrb w3, [x19, x2]              // program byte
    cmp w3, #'['                    // 0x5B
    bne 9f
    // push i onto stack
    str w2, [x0, x1, lsl #2]
    add x1, x1, #1
    b 10f
9:  cmp w3, #']'                    // 0x5D
    bne 10f
    // pop matching [
    cbz x1, bracket_error           // stack empty -> error
    sub x1, x1, #1
    ldr w4, [x0, x1, lsl #2]        // w4 = matching [
    // set jump_table for both brackets
    str w2, [x26, x4, lsl #2]       // jump_table[matching [] = i (]
    str w4, [x26, x2, lsl #2]       // jump_table[i] = matching [
10: add x2, x2, #1
    b 7b

8:  // Check stack empty
    cbnz x1, bracket_error
    b interpret_start

bracket_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

interpret_start:
    // Initialize interpreter state
    // tape_ptr already at tape base (x24)
    // ip = 0 (x25)
    // program_input_ptr = input_buffer + program_input_start
    add x29, x19, x22               // x29 = program_input_ptr
    add x30, x19, x20               // x30 = program_input_end (input_buffer + total_bytes_read)
    // output_ptr already at output_buffer base (x28)
    // output_buffer end = output_buffer + 65536
    adrp x9, output_buffer
    add x9, x9, :lo12:output_buffer
    add x9, x9, #65536              // x9 = output_buffer_end

interpret_loop:
    cmp x25, x21
    bge interpret_end               // ip >= program_length -> done
    ldrb w0, [x19, x25]             // current command
    // Switch on command
    cbz w0, interpret_next          // shouldn't happen, but skip null
    cmp w0, #'>'                    // 0x3E
    beq cmd_right
    cmp w0, #'<'                    // 0x3C
    beq cmd_left
    cmp w0, #'+'                    // 0x2B
    beq cmd_inc
    cmp w0, #'-'                    // 0x2D
    beq cmd_dec
    cmp w0, #'.'                    // 0x2E
    beq cmd_out
    cmp w0, #','                    // 0x2C
    beq cmd_in
    cmp w0, #'['                    // 0x5B
    beq cmd_jz
    cmp w0, #']'                    // 0x5D
    beq cmd_jnz
    // ignore other characters
    b interpret_next

cmd_right:
    add x24, x24, #1
    b interpret_next

cmd_left:
    sub x24, x24, #1
    b interpret_next

cmd_inc:
    ldrb w0, [x24]
    add w0, w0, #1
    strb w0, [x24]
    b interpret_next

cmd_dec:
    ldrb w0, [x24]
    sub w0, w0, #1
    strb w0, [x24]
    b interpret_next

cmd_out:
    ldrb w0, [x24]
    strb w0, [x28]
    add x28, x28, #1
    cmp x28, x9
    bne interpret_next
    // output buffer full, flush
    bl flush_output
    b interpret_next

cmd_in:
    cmp x29, x30
    bge in_eof
    ldrb w0, [x29], #1
    strb w0, [x24]
    b interpret_next
in_eof:
    strb wzr, [x24]
    b interpret_next

cmd_jz:
    ldrb w0, [x24]
    cbnz w0, interpret_next
    // jump to matching ]
    ldr w0, [x26, x25, lsl #2]
    mov x25, x0
    b interpret_next

cmd_jnz:
    ldrb w0, [x24]
    cbz w0, interpret_next
    // jump to matching [
    ldr w0, [x26, x25, lsl #2]
    mov x25, x0
    b interpret_next

interpret_next:
    add x25, x25, #1
    b interpret_loop

interpret_end:
    // Flush remaining output
    cmp x28, x27
    beq exit_success
    bl flush_output

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// flush_output: write output_buffer to stdout
// uses x27 (output_buffer base), x28 (output_ptr), x9 (output_buffer_end)
// preserves x19-x28, x29, x30? We'll save/restore used registers.
flush_output:
    stp x29, x30, [sp, #-16]!
    mov x0, #STDOUT
    mov x1, x27
    sub x2, x28, x27                // length
    mov x8, #SYS_WRITE
    svc #0
    // reset output_ptr
    mov x28, x27
    ldp x29, x30, [sp], #16
    ret
```
