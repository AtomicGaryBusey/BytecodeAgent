.globl _start

.text
_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d

read_loop:
    mov edx, 65536
    sub rdx, r12
    jz read_done
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor ecx, ecx

scan_loop:
    cmp rcx, r12
    jae scan_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 0x0A
    jne 1f
    inc r8
1:
    cmp al, 0x20
    je ws
    cmp al, 0x09
    je ws
    cmp al, 0x0A
    je ws
    cmp al, 0x0B
    je ws
    cmp al, 0x0C
    je ws
    cmp al, 0x0D
    je ws
    test r10, r10
    jnz scan_loop
    inc r9
    mov r10d, 1
    jmp scan_loop
ws:
    xor r10d, r10d
    jmp scan_loop

scan_done:
    mov rdi, r8
    call print_num
    call print_space
    mov rdi, r9
    call print_num
    call print_space
    mov rdi, r12
    call print_num
    call print_newline

    xor edi, edi
    mov eax, 60
    syscall

print_num:
    lea rsi, [rip + numbuf + 31]
    mov rax, rdi
    mov r10d, 10
2:
    xor edx, edx
    div r10
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 2b
    lea rdx, [rip + numbuf + 31]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

print_space:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + space_char]
    mov edx, 1
    syscall
    ret

print_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_char]
    mov edx, 1
    syscall
    ret

.section .rodata
space_char:   .ascii " "
newline_char: .ascii "\n"

.section .bss
.align 16
buffer:  .zero 65536
.align 16
numbuf:  .zero 32
