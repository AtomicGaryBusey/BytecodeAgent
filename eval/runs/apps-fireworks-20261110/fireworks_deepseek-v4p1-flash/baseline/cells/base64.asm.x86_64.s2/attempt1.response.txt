.section .text
.globl _start
_start:
    # read all input
    xor r12, r12
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    test r12, r12
    jz .exit_zero

    lea r13, [rip + outbuf]
    xor r14, r14
.chunk_loop:
    mov rax, r12
    sub rax, r14
    mov rbx, rax
    cmp rbx, 57
    jbe .chunk_ok
    mov rbx, 57
.chunk_ok:
    lea rsi, [rip + inbuf]
    add rsi, r14
    mov rdi, r13
    mov rcx, rbx
.encode_loop:
    cmp rcx, 3
    jb .less_than_3
    # 3 bytes
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]
    movzx r8d, byte ptr [rsi+2]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, edx
    shr r11d, 4
    or r10d, r11d
    mov r11d, edx
    and r11d, 15
    shl r11d, 2
    mov r15d, r8d
    shr r15d, 6
    or r11d, r15d
    mov r15d, r8d
    and r15d, 63
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r9]
    mov byte ptr [rdi], al
    movzx eax, byte ptr [rdx + r10]
    mov byte ptr [rdi+1], al
    movzx eax, byte ptr [rdx + r11]
    mov byte ptr [rdi+2], al
    movzx eax, byte ptr [rdx + r15]
    mov byte ptr [rdi+3], al
    add rsi, 3
    add rdi, 4
    sub rcx, 3
    jmp .encode_loop
.less_than_3:
    cmp rcx, 2
    je .two_bytes
    cmp rcx, 1
    je .one_byte
    jmp .chunk_done
.two_bytes:
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi+1]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, edx
    shr r11d, 4
    or r10d, r11d
    mov r11d, edx
    and r11d, 15
    shl r11d, 2
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r9]
    mov byte ptr [rdi], al
    movzx eax, byte ptr [rdx + r10]
    mov byte ptr [rdi+1], al
    movzx eax, byte ptr [rdx + r11]
    mov byte ptr [rdi+2], al
    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    xor rcx, rcx
    jmp .chunk_done
.one_byte:
    movzx eax, byte ptr [rsi]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r9]
    mov byte ptr [rdi], al
    movzx eax, byte ptr [rdx + r10]
    mov byte ptr [rdi+1], al
    mov byte ptr [rdi+2], 0x3D
    mov byte ptr [rdi+3], 0x3D
    add rdi, 4
    xor rcx, rcx
    jmp .chunk_done
.chunk_done:
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov r13, rdi
    add r14, rbx
    cmp r14, r12
    jb .chunk_loop

    # write output
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
.write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz .write_loop

.exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000
