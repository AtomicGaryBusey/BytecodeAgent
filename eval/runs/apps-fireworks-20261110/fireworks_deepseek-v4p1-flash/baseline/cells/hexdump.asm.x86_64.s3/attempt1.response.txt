.globl _start
.text
_start:
    # read loop
    xor r12d, r12d          # total = 0
.read_loop:
    mov eax, 0              # read
    xor edi, edi            # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12            # buf + total
    mov edx, 65536
    sub edx, r12d           # count = 65536 - total
    syscall
    test rax, rax
    jle .read_done          # if ret <= 0, break
    add r12, rax
    jmp .read_loop

.read_done:
    # process rows
    xor r13d, r13d          # off = 0
    test r12, r12
    jz .after_rows          # if total == 0, skip rows

.row_loop:
    # n = min(16, total - off)
    mov rax, r12
    sub rax, r13            # remaining = total - off
    cmp rax, 16
    jbe .n_ok
    mov rax, 16
.n_ok:
    mov r14, rax            # r14 = n

    # build row in rowbuf
    lea rdi, [rip + rowbuf]

    # offset: 8 hex digits of off (r13)
    mov r8d, 7
.offset_loop:
    mov rcx, r8
    shl rcx, 2              # shift = i*4
    mov rax, r13
    shr rax, cl             # rax = off >> shift
    and eax, 0xF
    lea rdx, [rip + hexdigits]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r8d
    jns .offset_loop

    # two spaces
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    add rdi, 2

    # hex bytes: for i = 0..15
    xor r8d, r8d            # i = 0
.hex_loop:
    cmp r8, 16
    jae .hex_done
    cmp r8, r14
    jae .print_spaces
    # print byte at inbuf[off + i]
    mov rax, r13
    add rax, r8
    lea rdx, [rip + inbuf]
    movzx eax, byte ptr [rdx + rax]
    # high nibble
    mov ecx, eax
    shr ecx, 4
    lea rdx, [rip + hexdigits]
    mov cl, byte ptr [rdx + rcx]
    mov byte ptr [rdi], cl
    inc rdi
    # low nibble
    and eax, 0xF
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rdi], al
    inc rdi
    # space
    mov byte ptr [rdi], ' '
    inc rdi
    jmp .after_byte
.print_spaces:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
.after_byte:
    # if i == 7, extra space
    cmp r8, 7
    jne .no_extra
    mov byte ptr [rdi], ' '
    inc rdi
.no_extra:
    inc r8
    jmp .hex_loop
.hex_done:

    # space, '|'
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], '|'
    add rdi, 2

    # ascii: for i = 0..n-1
    xor r8d, r8d
.ascii_loop:
    cmp r8, r14
    jae .ascii_done
    mov rax, r13
    add rax, r8
    lea rdx, [rip + inbuf]
    movzx eax, byte ptr [rdx + rax]
    cmp al, 0x20
    jb .dot
    cmp al, 0x7E
    ja .dot
    jmp .store_char
.dot:
    mov al, '.'
.store_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp .ascii_loop
.ascii_done:

    # '|', '\n'
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi+1], 10
    add rdi, 2

    # write row
    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    # advance off
    add r13, r14
    cmp r13, r12
    jb .row_loop

.after_rows:
    # print final offset
    lea rdi, [rip + rowbuf]
    mov r8d, 7
.final_loop:
    mov rcx, r8
    shl rcx, 2
    mov rax, r12
    shr rax, cl
    and eax, 0xF
    lea rdx, [rip + hexdigits]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r8d
    jns .final_loop
    mov byte ptr [rdi], 10
    inc rdi
    # write
    lea rsi, [rip + rowbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    # exit(0)
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
inbuf:
    .zero 65536
rowbuf:
    .zero 128
