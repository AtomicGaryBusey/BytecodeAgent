.text
.globl _start

_start:
    # read all input into inbuf
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    js exit
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines
    lea r14, [rip + inbuf]
    lea r15, [r14 + r12]
    mov rbx, r14
    xor r13, r13
parse_loop:
    cmp rbx, r15
    jae parse_done
    mov rax, rbx
    sub rax, r14
    lea rcx, [rip + line_off]
    mov [rcx + r13*4], eax
    mov rsi, rbx
scan_loop:
    cmp rsi, r15
    jae no_newline
    cmp byte ptr [rsi], 10
    je found_newline
    inc rsi
    jmp scan_loop
found_newline:
    mov rdx, rsi
    sub rdx, rbx
    lea rcx, [rip + line_len]
    mov [rcx + r13*4], edx
    inc r13
    lea rbx, [rsi + 1]
    jmp parse_loop
no_newline:
    mov rdx, r15
    sub rdx, rbx
    lea rcx, [rip + line_len]
    mov [rcx + r13*4], edx
    inc r13
    mov rbx, r15
    jmp parse_loop
parse_done:
    # sort lines using insertion sort
    cmp r13, 1
    jbe sort_done
    mov rbx, 1
sort_outer:
    cmp rbx, r13
    jae sort_done
    lea rcx, [rip + line_off]
    mov r12d, [rcx + rbx*4]
    lea rcx, [rip + line_len]
    mov r15d, [rcx + rbx*4]
    mov rbp, rbx
sort_inner:
    test rbp, rbp
    jz sort_insert
    lea rcx, [rip + line_off]
    mov r10d, [rcx + rbp*4 - 4]
    lea rcx, [rip + line_len]
    mov r11d, [rcx + rbp*4 - 4]
    lea rdi, [r14 + r10]
    lea rsi, [r14 + r12]
    mov edx, r11d
    mov ecx, r15d
    call compare_lines
    test eax, eax
    jle sort_insert
    lea rcx, [rip + line_off]
    mov r10d, [rcx + rbp*4 - 4]
    mov [rcx + rbp*4], r10d
    lea rcx, [rip + line_len]
    mov r11d, [rcx + rbp*4 - 4]
    mov [rcx + rbp*4], r11d
    dec rbp
    jmp sort_inner
sort_insert:
    lea rcx, [rip + line_off]
    mov [rcx + rbp*4], r12d
    lea rcx, [rip + line_len]
    mov [rcx + rbp*4], r15d
    inc rbx
    jmp sort_outer
sort_done:
    # build output buffer
    lea rdi, [rip + outbuf]
    xor r8, r8
output_loop:
    cmp r8, r13
    jae output_done
    lea rcx, [rip + line_off]
    mov r10d, [rcx + r8*4]
    lea rcx, [rip + line_len]
    mov r11d, [rcx + r8*4]
    lea rsi, [r14 + r10]
    mov rcx, r11
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r8
    jmp output_loop
output_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit
    add rsi, rax
    sub rdx, rax
    jnz write_loop
exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    # rdi = ptr_a, rsi = ptr_b, edx = len_a, ecx = len_b
    # returns eax = -1 if a < b, 0 if equal, 1 if a > b
    mov r8d, edx
    cmp r8d, ecx
    cmova r8d, ecx          # r8d = min(len_a, len_b)
    test r8d, r8d
    jz .length_cmp
    # swap rsi and rdi so that rsi = ptr_a, rdi = ptr_b
    mov r10, rdi
    mov rdi, rsi
    mov rsi, r10
    mov rcx, r8
    repe cmpsb
    jne .diff
.length_cmp:
    cmp edx, ecx
    jb .less
    ja .greater
    xor eax, eax
    ret
.diff:
    movzx r10d, byte ptr [rsi - 1]
    movzx r11d, byte ptr [rdi - 1]
    cmp r10d, r11d
    jb .less
    ja .greater
    xor eax, eax
    ret
.less:
    mov eax, -1
    ret
.greater:
    mov eax, 1
    ret

.section .bss
.align 16
inbuf: .zero 32768
outbuf: .zero 40000
line_off: .zero 4000
line_len: .zero 4000
