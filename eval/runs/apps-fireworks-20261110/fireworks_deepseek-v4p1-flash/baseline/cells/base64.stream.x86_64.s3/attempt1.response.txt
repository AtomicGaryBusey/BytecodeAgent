.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000

.section .text
_start:
    # Read all input
    xor r12, r12          # r12 = total bytes read
read_loop:
    mov rax, 0            # sys_read
    mov rdi, 0            # fd 0
    lea rsi, [rip+inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    # Encode
    xor r13, r13          # r13 = output index
    xor r14, r14          # r14 = column
    xor r15, r15          # r15 = input index
    test r12, r12
    jz finish

encode_loop:
    mov rax, r12
    sub rax, r15
    cmp rax, 3
    jb tail
    # Load 3 bytes
    lea rsi, [rip+inbuf]
    add rsi, r15
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]
    shl eax, 16
    shl ebx, 8
    or eax, ebx
    or eax, ecx
    # Convert to base64
    lea rdi, [rip+alphabet]
    mov edx, eax
    shr edx, 18
    and edx, 63
    movzx r8d, byte ptr [rdi+rdx]
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx r9d, byte ptr [rdi+rdx]
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx r10d, byte ptr [rdi+rdx]
    mov edx, eax
    and edx, 63
    movzx r11d, byte ptr [rdi+rdx]
    # Store
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], r8b
    mov byte ptr [rdi+1], r9b
    mov byte ptr [rdi+2], r10b
    mov byte ptr [rdi+3], r11b
    add r13, 4
    add r15, 3
    add r14, 4
    cmp r14, 76
    jne encode_loop
    # Newline
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], 0x0A
    inc r13
    xor r14, r14
    jmp encode_loop

tail:
    test rax, rax
    jz finish
    cmp rax, 1
    je tail1
    # 2 bytes remaining
    lea rsi, [rip+inbuf]
    add rsi, r15
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    shl eax, 16
    shl ebx, 8
    or eax, ebx
    lea rdi, [rip+alphabet]
    mov edx, eax
    shr edx, 18
    and edx, 63
    movzx r8d, byte ptr [rdi+rdx]
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx r9d, byte ptr [rdi+rdx]
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx r10d, byte ptr [rdi+rdx]
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], r8b
    mov byte ptr [rdi+1], r9b
    mov byte ptr [rdi+2], r10b
    mov byte ptr [rdi+3], 0x3D
    add r13, 4
    add r15, 2
    add r14, 4
    cmp r14, 76
    jne finish
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], 0x0A
    inc r13
    xor r14, r14
    jmp finish

tail1:
    lea rsi, [rip+inbuf]
    add rsi, r15
    movzx eax, byte ptr [rsi]
    shl eax, 16
    lea rdi, [rip+alphabet]
    mov edx, eax
    shr edx, 18
    and edx, 63
    movzx r8d, byte ptr [rdi+rdx]
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx r9d, byte ptr [rdi+rdx]
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], r8b
    mov byte ptr [rdi+1], r9b
    mov byte ptr [rdi+2], 0x3D
    mov byte ptr [rdi+3], 0x3D
    add r13, 4
    add r15, 1
    add r14, 4
    cmp r14, 76
    jne finish
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], 0x0A
    inc r13
    xor r14, r14
    jmp finish

finish:
    test r13, r13
    jz exit_ok
    test r14, r14
    jz write_out
    lea rdi, [rip+outbuf]
    add rdi, r13
    mov byte ptr [rdi], 0x0A
    inc r13
write_out:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip+outbuf]
    mov rdx, r13
    syscall
exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
