    .globl _start
    .text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    adrp x25, hexdig
    add x25, x25, :lo12:hexdig
    mov x20, #0
    movz x27, #1, lsl #16

read_loop:
    sub x2, x27, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs dump_done
    sub x23, x20, x21
    cmp x23, #16
    b.ls row_short
    mov x23, #16
row_short:
    mov x24, x22
    mov x0, x21
    bl emit_hex8
    mov w10, #32
    strb w10, [x24], #1
    strb w10, [x24], #1
    add x26, x19, x21
    mov x9, #0

slot_loop:
    cmp x9, #16
    b.hs slots_done
    cmp x9, x23
    b.hs slot_blank
    ldrb w0, [x26, x9]
    bl emit_hex2
    mov w10, #32
    strb w10, [x24], #1
    b slot_extra
slot_blank:
    mov w10, #32
    strb w10, [x24], #1
    strb w10, [x24], #1
    strb w10, [x24], #1
slot_extra:
    cmp x9, #7
    b.ne slot_next
    mov w10, #32
    strb w10, [x24], #1
slot_next:
    add x9, x9, #1
    b slot_loop

slots_done:
    mov w10, #32
    strb w10, [x24], #1
    mov w10, #124
    strb w10, [x24], #1
    mov x9, #0

char_loop:
    cmp x9, x23
    b.hs chars_done
    ldrb w10, [x26, x9]
    sub w11, w10, #32
    cmp w11, #94
    b.hi char_dot
    b char_store
char_dot:
    mov w10, #46
char_store:
    strb w10, [x24], #1
    add x9, x9, #1
    b char_loop

chars_done:
    mov w10, #124
    strb w10, [x24], #1
    mov w10, #10
    strb w10, [x24], #1
    mov x0, #1
    mov x1, x22
    sub x2, x24, x22
    mov x8, #64
    svc #0
    add x21, x21, x23
    b row_loop

dump_done:
    mov x24, x22
    mov x0, x20
    bl emit_hex8
    mov w10, #10
    strb w10, [x24], #1
    mov x0, #1
    mov x1, x22
    sub x2, x24, x22
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

emit_hex8:
    mov x9, #28
eh8_loop:
    lsr x10, x0, x9
    and x10, x10, #15
    ldrb w10, [x25, x10]
    strb w10, [x24], #1
    subs x9, x9, #4
    b.ge eh8_loop
    ret

emit_hex2:
    lsr w10, w0, #4
    ldrb w10, [x25, x10]
    strb w10, [x24], #1
    and w10, w0, #15
    ldrb w10, [x25, x10]
    strb w10, [x24], #1
    ret

    .section .rodata
hexdig:
    .ascii "0123456789abcdef"

    .bss
    .align 4
inbuf:
    .zero 65536
    .align 4
outbuf:
    .zero 256
