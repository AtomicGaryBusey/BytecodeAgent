```asm
// sort_lines: read lines from stdin, sort ascending bytewise, output.
// AArch64 Linux, no libc.

.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 65536
.equ MAX_LINES, 1000

.section .bss
.align 12
inbuf:
    .zero INBUF_SIZE
.align 12
outbuf:
    .zero OUTBUF_SIZE
.align 4
line_off:
    .zero MAX_LINES * 4
line_len:
    .zero MAX_LINES * 4

.text
.globl _start

_start:
    // Read all input into inbuf.
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x0, #0                  // fd 0
    add x1, x19, x20            // buf + total
    mov x2, #INBUF_SIZE
    sub x2, x2, x20             // count = INBUF_SIZE - total
    mov x8, #63                 // read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:

    // Parse lines into line_off/line_len.
    adrp x21, line_off
    add x21, x21, :lo12:line_off
    adrp x22, line_len
    add x22, x22, :lo12:line_len
    mov x23, #0                 // line count
    mov x24, #0                 // current position
parse_loop:
    cmp x24, x20
    b.hs parse_done
    mov x25, x24                // search index for newline
find_nl:
    cmp x25, x20
    b.hs no_nl
    ldrb w3, [x19, x25]
    cmp w3, #10
    b.eq found_nl
    add x25, x25, #1
    b find_nl
no_nl:
    mov x25, x20
found_nl:
    sub x26, x25, x24           // length = newline_pos - start
    lsl x27, x23, #2
    str w24, [x21, x27]         // line_off[count] = start
    str w26, [x22, x27]         // line_len[count] = length
    add x23, x23, #1
    add x24, x25, #1            // next start = newline_pos + 1
    b parse_loop
parse_done:

    // Insertion sort lines by bytewise comparison.
    cmp x23, #2
    b.lo sort_done
    mov x4, #1                  // i = 1
sort_outer:
    cmp x4, x23
    b.hs sort_done
    lsl x5, x4, #2
    ldr w6, [x21, x5]           // key_off
    ldr w7, [x22, x5]           // key_len
    mov x8, x4                  // j = i
sort_inner:
    cbz x8, sort_insert
    sub x9, x8, #1
    lsl x10, x9, #2
    ldr w11, [x21, x10]         // off[j-1]
    ldr w12, [x22, x10]         // len[j-1]
    // Compare (w11,w12) vs (w6,w7).
    cmp w12, w7
    csel w13, w12, w7, lo       // min length
    add x14, x19, x11           // ptr to line[j-1]
    add x15, x19, x6            // ptr to key
    mov x16, #0                 // byte index
cmp_loop:
    cmp x16, x13
    b.hs cmp_equal
    ldrb w17, [x14, x16]
    ldrb w18, [x15, x16]
    cmp w17, w18
    b.ne cmp_diff
    add x16, x16, #1
    b cmp_loop
cmp_equal:
    cmp w12, w7
    b.lo line1_less
    b.hi line1_greater
    b sort_insert               // equal
cmp_diff:
    b.lo line1_less
    b.hi line1_greater
line1_less:
    b sort_insert
line1_greater:
    lsl x10, x8, #2
    str w11, [x21, x10]         // off[j] = off[j-1]
    str w12, [x22, x10]         // len[j] = len[j-1]
    sub x8, x8, #1
    b sort_inner
sort_insert:
    lsl x10, x8, #2
    str w6, [x21, x10]          // off[j] = key_off
    str w7, [x22, x10]          // len[j] = key_len
    add x4, x4, #1
    b sort_outer
sort_done:

    // Build output: each line followed by newline.
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x25, #0                 // output position
    mov x4, #0                  // i = 0
build_loop:
    cmp x4, x23
    b.hs build_done
    lsl x5, x4, #2
    ldr w6, [x21, x5]           // off
    ldr w7, [x22, x5]           // len
    add x8, x19, x6             // src = inbuf + off
    add x9, x24, x25            // dst = outbuf + outpos
    mov x10, #0
copy_loop:
    cmp x10, x7
    b.hs copy_done
    ldrb w11, [x8, x10]
    strb w11, [x9, x10]
    add x10, x10, #1
    b copy_loop
copy_done:
    add x25, x25, x7
    mov w11, #10                // newline
    strb w11, [x24, x25]
    add x25, x25, #1
    add x4, x4, #1
    b build_loop
build_done:

    // Write output to stdout.
    mov x0, #1                  // fd 1
    mov x1, x24                 // outbuf
    mov x2, x25                 // length
    mov x8, #64                 // write
    svc #0

    // Exit with status 0.
    mov x0, #0
    mov x8, #93                 // exit
    svc #0
```
