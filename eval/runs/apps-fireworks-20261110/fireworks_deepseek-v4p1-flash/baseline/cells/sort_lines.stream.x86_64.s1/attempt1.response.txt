.section .rodata
newline:
    .byte 0x0A

.section .bss
.align 16
input_buffer:
    .zero 32768
.align 16
lines:
    .zero 16000
.align 16
temp:
    .zero 16000

.section .text
.globl _start
_start:
    # read all input
    xor r12, r12                # total bytes read
.read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    # r12 = total bytes read
    lea rbx, [rip + input_buffer]   # buffer base
    xor r13, r13                # line_count = 0
    xor r14, r14                # i = 0
    xor r15, r15                # start = 0
.parse_loop:
    cmp r14, r12
    jae .parse_end
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x0A
    jne .parse_next
    # found newline
    mov rdi, r15
    add rdi, rbx                # ptr = buffer + start
    mov rsi, r14
    sub rsi, r15                # len = i - start
    mov rcx, r13
    shl rcx, 4
    lea rdx, [rip + lines]
    add rdx, rcx
    mov [rdx], rdi
    mov [rdx+8], rsi
    inc r13
    lea r15, [r14+1]
.parse_next:
    inc r14
    jmp .parse_loop
.parse_end:
    cmp r15, r12
    jae .parse_done
    # final line
    mov rdi, r15
    add rdi, rbx
    mov rsi, r12
    sub rsi, r15
    mov rcx, r13
    shl rcx, 4
    lea rdx, [rip + lines]
    add rdx, rcx
    mov [rdx], rdi
    mov [rdx+8], rsi
    inc r13
.parse_done:
    mov r12, r13                # n = line_count
    lea r14, [rip + lines]      # src = lines
    lea r15, [rip + temp]       # dst = temp
    mov r13, 1                  # width = 1
    cmp r12, 1
    jbe .sort_done

.merge_pass_loop:
    cmp r13, r12
    jae .sort_done
    xor rbx, rbx                # i = 0
.merge_run_loop:
    cmp rbx, r12
    jae .merge_pass_end
    # left = i = rbx
    # mid = min(i + width, n)
    mov r8, rbx
    add r8, r13
    cmp r8, r12
    jbe .mid_ok
    mov r8, r12
.mid_ok:
    # right = min(i + 2*width, n)
    mov r9, rbx
    lea r10, [r13 + r13]
    add r9, r10
    cmp r9, r12
    jbe .right_ok
    mov r9, r12
.right_ok:
    mov rbp, r9                 # save right in rbp

    # p1 = src + left*16
    mov rax, rbx
    shl rax, 4
    lea r10, [r14 + rax]        # p1
    # end1 = src + mid*16
    mov rax, r8
    shl rax, 4
    lea r11, [r14 + rax]        # end1
    # p2 = src + mid*16
    mov rax, r8
    shl rax, 4
    lea rcx, [r14 + rax]        # p2
    # end2 = src + right*16
    mov rax, r9
    shl rax, 4
    lea rdx, [r14 + rax]        # end2
    # out = dst + left*16
    mov rax, rbx
    shl rax, 4
    lea rsi, [r15 + rax]        # out

.merge_loop:
    cmp r10, r11
    jae .copy_rest2
    cmp rcx, rdx
    jae .copy_rest1
    # compare lines at r10 and rcx
    mov rax, [r10]              # ptr1
    mov r8, [r10+8]             # len1
    mov rdi, [rcx]              # ptr2
    mov r9, [rcx+8]             # len2
    push rbx
.cmp_loop:
    test r8, r8
    jz .cmp_len1_end
    test r9, r9
    jz .cmp_len2_end
    mov bl, byte ptr [rax]
    cmp bl, byte ptr [rdi]
    jne .cmp_diff
    inc rax
    inc rdi
    dec r8
    dec r9
    jmp .cmp_loop
.cmp_diff:
    jb .take_p1
    jmp .take_p2
.cmp_len1_end:
    jmp .take_p1
.cmp_len2_end:
    jmp .take_p2

.take_p1:
    pop rbx
    mov rax, [r10]
    mov r8, [r10+8]
    mov [rsi], rax
    mov [rsi+8], r8
    add r10, 16
    add rsi, 16
    jmp .merge_loop

.take_p2:
    pop rbx
    mov rax, [rcx]
    mov r8, [rcx+8]
    mov [rsi], rax
    mov [rsi+8], r8
    add rcx, 16
    add rsi, 16
    jmp .merge_loop

.copy_rest1:
    cmp r10, r11
    jae .merge_run_done
    mov rax, [r10]
    mov r8, [r10+8]
    mov [rsi], rax
    mov [rsi+8], r8
    add r10, 16
    add rsi, 16
    jmp .copy_rest1

.copy_rest2:
    cmp rcx, rdx
    jae .merge_run_done
    mov rax, [rcx]
    mov r8, [rcx+8]
    mov [rsi], rax
    mov [rsi+8], r8
    add rcx, 16
    add rsi, 16
    jmp .copy_rest2

.merge_run_done:
    mov rbx, rbp                # i = right
    jmp .merge_run_loop

.merge_pass_end:
    # swap src and dst
    mov rax, r14
    mov r14, r15
    mov r15, rax
    shl r13, 1
    jmp .merge_pass_loop

.sort_done:
    test r12, r12
    jz .exit
    xor rbx, rbx                # i = 0
.output_loop:
    cmp rbx, r12
    jae .exit
    mov rax, rbx
    shl rax, 4
    lea r10, [r14 + rax]
    mov rsi, [r10]              # ptr
    mov rdx, [r10+8]            # len
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    syscall
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc rbx
    jmp .output_loop

.exit:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
