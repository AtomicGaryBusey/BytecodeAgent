    .globl _start
    .text
_start:
    // read input
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x0, #0                  // fd 0
    add x1, x19, x20            // buf + total
    mov x2, #32768
    sub x2, x2, x20             // remaining space
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #32768
    b.lt read_loop
read_done:
    adrp x0, total_bytes
    add x0, x0, :lo12:total_bytes
    str w20, [x0]

    // parse lines
    adrp x21, start_off
    add x21, x21, :lo12:start_off
    adrp x22, line_len
    add x22, x22, :lo12:line_len
    mov x23, #0                 // line count
    mov x24, #0                 // i = 0
parse_loop:
    cmp x24, x20
    b.ge parse_done
    mov x25, x24                // start
advance:
    cmp x24, x20
    b.ge line_end
    ldrb w0, [x19, x24]
    cmp w0, #10                 // '\n'
    b.eq line_end
    add x24, x24, #1
    b advance
line_end:
    sub x26, x24, x25           // length
    str w25, [x21, x23, lsl #2]
    str w26, [x22, x23, lsl #2]
    add x23, x23, #1
    cmp x24, x20
    b.ge parse_done
    add x24, x24, #1            // skip newline
    b parse_loop
parse_done:
    adrp x0, line_count
    add x0, x0, :lo12:line_count
    str w23, [x0]

    // insertion sort
    mov x24, #1                 // i
sort_outer:
    cmp x24, x23
    b.ge sort_done
    mov x25, x24                // j
sort_inner:
    cmp x25, #0
    b.le sort_next
    sub x26, x25, #1            // j-1
    ldr w0, [x21, x26, lsl #2]  // start j-1
    ldr w1, [x22, x26, lsl #2]  // len j-1
    ldr w2, [x21, x25, lsl #2]  // start j
    ldr w3, [x22, x25, lsl #2]  // len j
    add x4, x19, x0             // ptr1
    add x5, x19, x2             // ptr2
    cmp w1, w3
    csel w6, w1, w3, lo         // min len
    mov x7, #0                  // k
cmp_loop:
    cmp x7, x6
    b.ge cmp_prefix
    ldrb w8, [x4, x7]
    ldrb w9, [x5, x7]
    cmp w8, w9
    b.ne cmp_diff
    add x7, x7, #1
    b cmp_loop
cmp_diff:
    b.lo sort_next              // w8 < w9
    b.hi swap_lines             // w8 > w9
cmp_prefix:
    cmp w1, w3
    b.lo sort_next
    b.hi swap_lines
    b sort_next
swap_lines:
    ldr w0, [x21, x26, lsl #2]
    ldr w1, [x22, x26, lsl #2]
    ldr w2, [x21, x25, lsl #2]
    ldr w3, [x22, x25, lsl #2]
    str w2, [x21, x26, lsl #2]
    str w3, [x22, x26, lsl #2]
    str w0, [x21, x25, lsl #2]
    str w1, [x22, x25, lsl #2]
    sub x25, x25, #1
    b sort_inner
sort_next:
    add x24, x24, #1
    b sort_outer
sort_done:

    // write output
    mov x24, #0
write_loop:
    cmp x24, x23
    b.ge write_done
    ldr w0, [x21, x24, lsl #2]  // start
    ldr w2, [x22, x24, lsl #2]  // len
    add x1, x19, x0             // ptr
    mov x0, #1                  // fd 1
    mov x8, #64                 // write
    svc #0
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x24, x24, #1
    b write_loop
write_done:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

    .section .rodata
newline:
    .byte 10

    .bss
    .balign 4
inbuf:      .zero 32768
    .balign 4
start_off:  .zero 4000
    .balign 4
line_len:   .zero 4000
    .balign 4
line_count: .zero 4
    .balign 4
total_bytes: .zero 4
