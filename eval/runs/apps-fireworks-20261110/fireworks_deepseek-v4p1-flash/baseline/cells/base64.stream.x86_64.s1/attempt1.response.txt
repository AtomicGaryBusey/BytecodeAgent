```asm
.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buf:
    .zero 65536
output_buf:
    .zero 131072

.section .text
.globl _start
_start:
    # read all input
    xor r12, r12
read_loop:
    mov eax, 0          # read
    mov edi, 0          # fd 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop

done_read:
    # r12 = input length
    # encode
    lea rsi, [rip + input_buf]
    lea rdi, [rip + output_buf]
    mov rcx, r12
    xor r8d, r8d        # column
    lea r10, [rip + alphabet]

    test rcx, rcx
    jz end_encode

encode_loop:
    cmp rcx, 3
    jb remainder

    # load 3 bytes
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi+1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi+2]
    or eax, edx

    add rsi, 3
    sub rcx, 3

    # first char
    mov r9d, eax
    shr r9d, 18
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    # second char
    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    # third char
    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    # fourth char
    and eax, 63
    movzx eax, byte ptr [r10 + rax]
    mov [rdi], al
    inc rdi

    add r8d, 4
    cmp r8d, 76
    jne encode_loop

    # newline
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r8d, r8d
    jmp encode_loop

remainder:
    test rcx, rcx
    jz end_encode

    cmp rcx, 1
    je rem1

    # rem2
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi+1]
    shl edx, 8
    or eax, edx

    mov r9d, eax
    shr r9d, 18
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    mov byte ptr [rdi], '='
    inc rdi

    add r8d, 4
    cmp r8d, 76
    jne end_encode
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r8d, r8d
    jmp end_encode

rem1:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov r9d, eax
    shr r9d, 18
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    movzx r9d, byte ptr [r10 + r9]
    mov [rdi], r9b
    inc rdi

    mov byte ptr [rdi], '='
    inc rdi
    mov byte ptr [rdi], '='
    inc rdi

    add r8d, 4
    cmp r8d, 76
    jne end_encode
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r8d, r8d

end_encode:
    # check if any output
    lea rax, [rip + output_buf]
    cmp rdi, rax
    je no_output
    test r8d, r8d
    jz no_output
    mov byte ptr [rdi], 0x0A
    inc rdi

no_output:
    # write output
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_program

write_loop:
    mov eax, 1          # write
    mov edi, 1          # fd 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_program:
    mov eax, 60         # exit
    xor edi, edi
    syscall
```
