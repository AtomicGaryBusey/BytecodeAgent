.section .text
.globl _start

.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ LINE_STRUCT_SIZE, 16
.equ OUTPUT_BUF_SIZE, MAX_INPUT + MAX_LINES + 16

_start:
    xor r12, r12
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jl read_loop
read_done:
    xor r13, r13
    xor rbx, rbx
    test r12, r12
    jz no_lines

parse_loop:
    mov r14, rbx
find_nl:
    cmp rbx, r12
    jge end_of_input_no_nl
    lea rcx, [rip + input_buf]
    mov al, byte ptr [rcx + rbx]
    cmp al, 10
    je found_nl
    inc rbx
    jmp find_nl

found_nl:
    mov rax, rbx
    sub rax, r14
    lea rcx, [rip + lines]
    mov rdx, r13
    shl rdx, 4
    add rcx, rdx
    lea rdx, [rip + input_buf]
    add rdx, r14
    mov [rcx], rdx
    mov [rcx + 8], rax
    inc r13
    inc rbx
    cmp rbx, r12
    jl parse_loop
    jmp parse_done

end_of_input_no_nl:
    mov rax, r12
    sub rax, r14
    test rax, rax
    jz parse_done
    lea rcx, [rip + lines]
    mov rdx, r13
    shl rdx, 4
    add rcx, rdx
    lea rdx, [rip + input_buf]
    add rdx, r14
    mov [rcx], rdx
    mov [rcx + 8], rax
    inc r13

parse_done:
no_lines:
    cmp r13, 2
    jl sort_done
    mov r15, 1
outer_loop:
    cmp r15, r13
    jge sort_done
    lea rcx, [rip + lines]
    mov rdx, r15
    shl rdx, 4
    add rcx, rdx
    mov r8, [rcx]
    mov r9, [rcx + 8]
    mov r10, r15
inner_loop:
    test r10, r10
    jz insert_key
    lea rcx, [rip + lines]
    mov rdx, r10
    dec rdx
    shl rdx, 4
    add rcx, rdx
    mov rdi, [rcx]
    mov rsi, [rcx + 8]
    mov rdx, r8
    mov rcx, r9
    call line_cmp
    test eax, eax
    jle insert_key
    lea rcx, [rip + lines]
    mov rdx, r10
    shl rdx, 4
    add rcx, rdx
    mov rdi, [rcx - 16]
    mov rsi, [rcx - 8]
    mov [rcx], rdi
    mov [rcx + 8], rsi
    dec r10
    jmp inner_loop
insert_key:
    lea rcx, [rip + lines]
    mov rdx, r10
    shl rdx, 4
    add rcx, rdx
    mov [rcx], r8
    mov [rcx + 8], r9
    inc r15
    jmp outer_loop
sort_done:

    lea r14, [rip + output_buf]
    xor r15, r15
output_loop:
    cmp r15, r13
    jge output_done
    lea rcx, [rip + lines]
    mov rdx, r15
    shl rdx, 4
    add rcx, rdx
    mov rsi, [rcx]
    mov rcx, [rcx + 8]
    test rcx, rcx
    jz add_newline
    mov rdi, r14
    rep movsb
    mov r14, rdi
add_newline:
    mov byte ptr [r14], 10
    inc r14
    inc r15
    jmp output_loop
output_done:
    lea rsi, [rip + output_buf]
    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

line_cmp:
    push r8
    push r9
    push r10
    push r11
    mov r8, rsi
    cmp r8, rcx
    cmovg r8, rcx
    mov r9, rdi
    mov r10, rsi
    mov r11, rcx
    mov rsi, r9
    mov rdi, rdx
    mov rcx, r8
    repe cmpsb
    test rcx, rcx
    jz .all_equal
    jb .less
    mov eax, 1
    pop r11
    pop r10
    pop r9
    pop r8
    ret
.less:
    mov eax, -1
    pop r11
    pop r10
    pop r9
    pop r8
    ret
.all_equal:
    cmp r10, r11
    je .equal
    jb .less
    mov eax, 1
    pop r11
    pop r10
    pop r9
    pop r8
    ret
.equal:
    xor eax, eax
    pop r11
    pop r10
    pop r9
    pop r8
    ret

.section .bss
.align 16
input_buf:    .zero MAX_INPUT
lines:        .zero MAX_LINES * LINE_STRUCT_SIZE
output_buf:   .zero OUTPUT_BUF_SIZE
