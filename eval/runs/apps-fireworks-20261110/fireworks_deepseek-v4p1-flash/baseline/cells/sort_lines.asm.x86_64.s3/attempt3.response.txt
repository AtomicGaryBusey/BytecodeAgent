.section .text
.globl _start

.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ OUTPUT_BUF_SIZE, MAX_INPUT + MAX_LINES + 16

_start:
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, MAX_INPUT
    jb read_loop
read_done:
    lea r14, [rip + input_buf]
    xor r13, r13
    xor rbx, rbx
    test r12, r12
    jz parse_done

parse_loop:
    mov rcx, rbx
scan_nl:
    cmp rcx, r12
    jae eof_line
    cmp byte ptr [r14 + rcx], 10
    je found_nl
    inc rcx
    jmp scan_nl

found_nl:
    lea rdx, [rip + line_ptrs]
    lea rax, [r14 + rbx]
    mov [rdx + r13*8], rax
    lea rdx, [rip + line_lens]
    mov rax, rcx
    sub rax, rbx
    mov [rdx + r13*8], rax
    inc r13
    lea rbx, [rcx + 1]
    cmp rbx, r12
    jb parse_loop
    jmp parse_done

eof_line:
    cmp rbx, r12
    jae parse_done
    lea rdx, [rip + line_ptrs]
    lea rax, [r14 + rbx]
    mov [rdx + r13*8], rax
    lea rdx, [rip + line_lens]
    mov rax, r12
    sub rax, rbx
    mov [rdx + r13*8], rax
    inc r13

parse_done:
    cmp r13, 2
    jl sort_done
    lea rbx, [rip + line_ptrs]
    lea rbp, [rip + line_lens]
    mov r15, 1
sort_outer:
    cmp r15, r13
    jge sort_done
    mov r11, [rbx + r15*8]
    mov r12, [rbp + r15*8]
    mov r10, r15
sort_inner:
    test r10, r10
    jz sort_place
    lea r9, [r10 - 1]
    mov rdi, [rbx + r9*8]
    mov rsi, [rbp + r9*8]
    mov rdx, r11
    mov rcx, r12
    call cmp_lines
    test eax, eax
    jle sort_place
    lea r9, [r10 - 1]
    mov rax, [rbx + r9*8]
    mov [rbx + r10*8], rax
    mov rax, [rbp + r9*8]
    mov [rbp + r10*8], rax
    dec r10
    jmp sort_inner
sort_place:
    mov [rbx + r10*8], r11
    mov [rbp + r10*8], r12
    inc r15
    jmp sort_outer
sort_done:
    lea rbx, [rip + line_ptrs]
    lea rbp, [rip + line_lens]
    lea r14, [rip + output_buf]
    xor r15, r15
out_loop:
    cmp r15, r13
    jge out_done
    mov rsi, [rbx + r15*8]
    mov rcx, [rbp + r15*8]
    test rcx, rcx
    jz out_nl
    mov rdi, r14
    rep movsb
    mov r14, rdi
out_nl:
    mov byte ptr [r14], 10
    inc r14
    inc r15
    jmp out_loop
out_done:
    lea rsi, [rip + output_buf]
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9d, r9d
cl_loop:
    cmp r9, r8
    jae cl_prefix
    mov al, byte ptr [rdi + r9]
    cmp al, byte ptr [rdx + r9]
    jb cl_less
    ja cl_greater
    inc r9
    jmp cl_loop
cl_prefix:
    cmp rsi, rcx
    jb cl_less
    ja cl_greater
    xor eax, eax
    ret
cl_less:
    mov eax, -1
    ret
cl_greater:
    mov eax, 1
    ret

.section .bss
.align 16
input_buf:   .zero MAX_INPUT
line_ptrs:   .zero MAX_LINES * 8
line_lens:   .zero MAX_LINES * 8
output_buf:  .zero OUTPUT_BUF_SIZE
