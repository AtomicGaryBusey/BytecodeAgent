.equ OUTBUF_SIZE, 8192

.section .rodata
err_msg:
    .ascii "error\n"
nlchar:
    .byte 10

.section .bss
.align 16
inbuf:
    .zero 65536
.align 16
outbuf:
    .zero 8192
.align 16
stk:
    .zero 16384
.align 16
numtmp:
    .zero 32
.align 8
outpos:
    .zero 8

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [rip + inbuf]
    add r13, rbx

line_loop:
    cmp r12, r13
    jae done_all
    mov rcx, r12
find_nl:
    cmp rcx, r13
    jae no_nl
    cmp byte ptr [rcx], 10
    je found_nl
    inc rcx
    jmp find_nl
found_nl:
    mov r14, rcx
    lea r15, [rcx + 1]
    jmp process_line
no_nl:
    mov r14, r13
    mov r15, r13

process_line:
    xor ebx, ebx
    mov rbp, r12

tok_loop:
    cmp rbp, r14
    jae tok_end
    movzx eax, byte ptr [rbp]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    jmp tok_start
skip_sep:
    inc rbp
    jmp tok_loop

tok_start:
    mov r8, rbp
scan_tok:
    inc rbp
    cmp rbp, r14
    jae tok_done
    movzx eax, byte ptr [rbp]
    cmp al, 32
    je tok_done
    cmp al, 9
    je tok_done
    jmp scan_tok
tok_done:
    mov r9, rbp
    sub r9, r8
    cmp r9, 1
    jne try_literal
    movzx eax, byte ptr [r8]
    cmp al, 43
    je do_add
    cmp al, 45
    je do_sub
    cmp al, 42
    je do_mul
    cmp al, 47
    je do_div
    cmp al, 37
    je do_mod
try_literal:
    movzx eax, byte ptr [r8]
    cmp al, 45
    jne lit_pos
    cmp r9, 2
    jb err
    cmp r9, 19
    ja err
    lea rsi, [r8 + 1]
    mov rcx, r9
    dec rcx
    xor r10d, r10d
lit_neg_loop:
    movzx eax, byte ptr [rsi]
    sub eax, 48
    cmp eax, 9
    ja err
    lea r10, [r10 + r10*4]
    lea r10, [rax + r10*2]
    inc rsi
    dec rcx
    jnz lit_neg_loop
    neg r10
    jmp push_val
lit_pos:
    cmp r9, 18
    ja err
    mov rsi, r8
    mov rcx, r9
    xor r10d, r10d
lit_pos_loop:
    movzx eax, byte ptr [rsi]
    sub eax, 48
    cmp eax, 9
    ja err
    lea r10, [r10 + r10*4]
    lea r10, [rax + r10*2]
    inc rsi
    dec rcx
    jnz lit_pos_loop
push_val:
    cmp rbx, 2048
    jae err
    lea rsi, [rip + stk]
    mov [rsi + rbx*8], r10
    inc rbx
    jmp tok_loop

do_add:
    cmp rbx, 2
    jb err
    lea rsi, [rip + stk]
    mov rax, [rsi + rbx*8 - 16]
    add rax, [rsi + rbx*8 - 8]
    mov [rsi + rbx*8 - 16], rax
    dec rbx
    jmp tok_loop

do_sub:
    cmp rbx, 2
    jb err
    lea rsi, [rip + stk]
    mov rax, [rsi + rbx*8 - 16]
    sub rax, [rsi + rbx*8 - 8]
    mov [rsi + rbx*8 - 16], rax
    dec rbx
    jmp tok_loop

do_mul:
    cmp rbx, 2
    jb err
    lea rsi, [rip + stk]
    mov rax, [rsi + rbx*8 - 16]
    imul rax, [rsi + rbx*8 - 8]
    mov [rsi + rbx*8 - 16], rax
    dec rbx
    jmp tok_loop

do_div:
    cmp rbx, 2
    jb err
    lea rsi, [rip + stk]
    mov rax, [rsi + rbx*8 - 16]
    mov rcx, [rsi + rbx*8 - 8]
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne 1f
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je err
1:
    cqo
    idiv rcx
    mov [rsi + rbx*8 - 16], rax
    dec rbx
    jmp tok_loop

do_mod:
    cmp rbx, 2
    jb err
    lea rsi, [rip + stk]
    mov rax, [rsi + rbx*8 - 16]
    mov rcx, [rsi + rbx*8 - 8]
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne 1f
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je err
1:
    cqo
    idiv rcx
    mov [rsi + rbx*8 - 16], rdx
    dec rbx
    jmp tok_loop

err:
    lea rsi, [rip + err_msg]
    mov edx, 6
    call out_bytes
    jmp next_line

tok_end:
    cmp rbx, 1
    jne err
    lea rsi, [rip + stk]
    mov rax, [rsi]
    call emit_value
    jmp next_line

next_line:
    mov r12, r15
    jmp line_loop

done_all:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

emit_value:
    lea rdi, [rip + numtmp + 32]
    mov r8, rax
    xor r9d, r9d
    test rax, rax
    jns 1f
    mov r9d, 1
    neg r8
1:
    mov rax, r8
    mov r10, 10
2:
    xor edx, edx
    div r10
    add dl, 48
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz 2b
    test r9d, r9d
    jz 3f
    dec rdi
    mov byte ptr [rdi], 45
3:
    lea rdx, [rip + numtmp + 32]
    sub rdx, rdi
    mov rsi, rdi
    call out_bytes
    lea rsi, [rip + nlchar]
    mov edx, 1
    call out_bytes
    ret

out_bytes:
    mov rax, [rip + outpos]
    add rax, rdx
    cmp rax, OUTBUF_SIZE
    jbe 1f
    call flush
1:
    mov rcx, rdx
    test rcx, rcx
    jz 3f
    mov rax, [rip + outpos]
    lea rdi, [rip + outbuf]
    add rdi, rax
2:
    mov al, [rsi]
    mov [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz 2b
3:
    add [rip + outpos], rdx
    ret

flush:
    push rsi
    push rdx
    push rdi
    push r8
    push r9
    mov r9, [rip + outpos]
    test r9, r9
    jz 9f
    lea r8, [rip + outbuf]
4:
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jle 9f
    add r8, rax
    sub r9, rax
    jnz 4b
9:
    mov qword ptr [rip + outpos], 0
    pop r9
    pop r8
    pop rdi
    pop rdx
    pop rsi
    ret
