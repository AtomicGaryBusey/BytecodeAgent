    sub x2, x20, x1

// print_dec: x0 = value, x1 = buffer ptr (advanced past digits)

    mov x3, #10
    cbnz x0, pd_loop
    mov w4, #0x30

    add w5, w5, #0x30
    strb w5, [x1], #1
    mov x0, x4

    mov x5, x3

    ldrb w6, [x4]
    ldrb w7, [x5]
    strb w7, [x4], #1
    strb w6, [x5], #-1

    ret

// skip_ws: advance x19 over space/tab/lf/cr

    cmp x19, x20
    b.hs sw_done
    ldrb w9, [x19]
    cmp w9, #0x20
    b.eq sw_next
    cmp w9, #0x09
    b.eq sw_next
    cmp w9, #0x0A
    b.eq sw_next
    cmp w9, #0x0D
    b.eq sw_next
sw_done:
    ret
sw_next:
    add x19, x19, #1
    b skip_ws

// fail: record error position, return 0

    mov w0, #0
    ret

// parse_value: dispatch on the byte at x19
parse_value:
    cmp x19, x20
    b.hs fail
    ldrb w9, [x19]
    cmp w9, #0x22
    b.eq parse_string
    cmp w9, #0x7B
    b.eq parse_object
    cmp w9, #0x5B
    b.eq parse_array
    cmp w9, #0x74
    b.eq parse_true
    cmp w9, #0x66
    b.eq parse_false
    cmp w9, #0x6E
    b.eq parse_null
    cmp w9, #0x2D
    b.eq parse_number
    sub w10, w9, #0x30
    cmp w10, #10
    b.lo parse_number
    b fail

// match_lit: x0 = literal address, w1 = length; x19 points at first char
// success: x19 += length, w0 = 1
// failure: x21 = mismatch position, w0 = 0

    mov x10, x19
    mov w11, w1
1:  cmp x10, x20
    b.hs 2f
    ldrb w12, [x10]
    ldrb w13, [x9], #1
    cmp w12, w13
    b.ne 2f

    subs w11, w11, #1
    b.ne 1b
    mov x19, x10
    mov w0, #1
    ret
2:  mov x21, x10
    mov w0, #0
    ret

    add x0, x0, :lo12:lit_true

    b match_lit

parse_false:

    add x0, x0, :lo12:lit_false

parse_null:
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null

    b match_lit

    cmp w9, #0x5C
    b.eq ps_esc_ok
    cmp w9, #0x2F
    b.eq ps_esc_ok
    cmp w9, #0x62
    b.eq ps_esc_ok
    cmp w9, #0x66
    b.eq ps_esc_ok
    cmp w9, #0x6E
    b.eq ps_esc_ok
    cmp w9, #0x72
    b.eq ps_esc_ok
    cmp w9, #0x74
    b.eq ps_esc_ok
    cmp w9, #0x75

    b.ne ps_hexloop

    ret
