```asm
    .globl _start

    .equ SYS_read, 63
    .equ SYS_write, 64
    .equ SYS_exit, 93
    .equ INBUF_SIZE, 65536
    .equ OUTBUF_SIZE, 65537

    .bss
    .align 4
inbuf:
    .zero INBUF_SIZE
outbuf:
    .zero OUTBUF_SIZE

    .text
_start:
    mov x19, #0
read_loop:
    mov x2, #INBUF_SIZE
    sub x2, x2, x19
    cbz x2, read_done
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    add x21, x20, x19
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf

process_loop:
    cmp x20, x21
    b.hs process_done

    mov x23, x20
find_nl:
    cmp x23, x21
    b.hs no_newline
    ldrb w24, [x23]
    cmp w24, #10
    b.eq found_nl
    add x23, x23, #1
    b find_nl

no_newline:
    mov x23, x21

found_nl:
    sub x24, x23, #1
rev_copy:
    cmp x24, x20
    b.lt rev_done
    ldrb w25, [x24]
    strb w25, [x22], #1
    sub x24, x24, #1
    b rev_copy

rev_done:
    mov w25, #10
    strb w25, [x22], #1

    cmp x23, x21
    b.hs process_done
    add x20, x23, #1
    b process_loop

process_done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x22, x1
    cbz x2, skip_write
    mov x0, #1
    mov x8, #SYS_write
    svc #0

skip_write:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
```
