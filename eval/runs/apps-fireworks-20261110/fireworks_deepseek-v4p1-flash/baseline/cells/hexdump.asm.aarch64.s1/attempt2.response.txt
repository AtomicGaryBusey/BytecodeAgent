.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    mov x21, x25
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0
    adrp x28, hex_chars
    add x28, x28, :lo12:hex_chars

row_loop:
    cmp x22, x20
    b.ge rows_done
    sub x23, x20, x22
    cmp x23, #16
    b.le 1f
    mov x23, #16
1:
    mov w3, w22
    mov x4, #8
2:
    ubfx w5, w3, #28, #4
    ldrb w6, [x28, x5]
    strb w6, [x21], #1
    lsl w3, w3, #4
    subs x4, x4, #1
    b.ne 2b
    mov w6, #' '
    strb w6, [x21], #1
    strb w6, [x21], #1

    mov x24, #0
3:
    cmp x24, #16
    b.ge 4f
    cmp x24, x23
    b.ge 5f
    add x7, x19, x22
    ldrb w8, [x7, x24]
    lsr w9, w8, #4
    ldrb w10, [x28, x9]
    strb w10, [x21], #1
    and w9, w8, #0xF
    ldrb w10, [x28, x9]
    strb w10, [x21], #1
    mov w10, #' '
    strb w10, [x21], #1
    b 6f
5:
    mov w10, #' '
    strb w10, [x21], #1
    strb w10, [x21], #1
    strb w10, [x21], #1
6:
    cmp x24, #7
    b.ne 7f
    mov w10, #' '
    strb w10, [x21], #1
7:
    add x24, x24, #1
    b 3b
4:
    mov w10, #' '
    strb w10, [x21], #1
    mov w10, #'|'
    strb w10, [x21], #1

    mov x24, #0
8:
    cmp x24, x23
    b.ge 9f
    add x7, x19, x22
    ldrb w8, [x7, x24]
    cmp w8, #0x20
    b.lt 10f
    cmp w8, #0x7E
    b.gt 10f
    strb w8, [x21], #1
    b 11f
10:
    mov w10, #'.'
    strb w10, [x21], #1
11:
    add x24, x24, #1
    b 8b
9:
    mov w10, #'|'
    strb w10, [x21], #1
    mov w10, #'\n'
    strb w10, [x21], #1
    add x22, x22, #16
    b row_loop

rows_done:
    mov w3, w20
    mov x4, #8
12:
    ubfx w5, w3, #28, #4
    ldrb w6, [x28, x5]
    strb w6, [x21], #1
    lsl w3, w3, #4
    subs x4, x4, #1
    b.ne 12b
    mov w10, #'\n'
    strb w10, [x21], #1

    mov x0, #1
    mov x1, x25
    sub x2, x21, x25
write_loop:
    cbz x2, write_done
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le write_done
    add x1, x1, x0
    subs x2, x2, x0
    b.gt write_loop
write_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 330000
