.globl _start
.text
_start:
    lea r14, [rip + inbuf]
    xor r12, r12
read_loop:
    mov rsi, r14
    add rsi, r12
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + outbuf]
    xor rbx, rbx
    xor rcx, rcx
scan_loop:
    cmp rcx, r12
    jae end_of_input
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 0x0A
    je found_newline
    inc rcx
    jmp scan_loop
found_newline:
    mov rdx, rcx
    dec rdx
reverse_loop:
    cmp rdx, rbx
    jl line_done
    movzx eax, byte ptr [r14 + rdx]
    mov byte ptr [r13], al
    inc r13
    dec rdx
    jmp reverse_loop
line_done:
    mov byte ptr [r13], 0x0A
    inc r13
    lea rbx, [rcx + 1]
    inc rcx
    jmp scan_loop
end_of_input:
    cmp rbx, r12
    jae output
    mov rdx, r12
    dec rdx
reverse_final:
    cmp rdx, rbx
    jl final_done
    movzx eax, byte ptr [r14 + rdx]
    mov byte ptr [r13], al
    inc r13
    dec rdx
    jmp reverse_final
final_done:
    mov byte ptr [r13], 0x0A
    inc r13
output:
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
    test rdx, rdx
    jz exit
    mov rdi, 1
write_loop:
    mov rax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop
exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
