.globl _start
.text
_start:
    xor r12d, r12d
    lea rbp, [rip + inbuf]
.read_loop:
    cmp r12, 65536
    jae .read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rbp + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    lea r15, [rip + outbuf]
    xor r13d, r13d
    xor r14d, r14d
.line_loop:
    cmp r13, r12
    jae .write_output
    mov rbx, r13
.find_nl:
    cmp rbx, r12
    jae .no_newline
    cmp byte ptr [rbp + rbx], 10
    je .found_newline
    inc rbx
    jmp .find_nl
.found_newline:
    mov rdx, rbx
    dec rdx
.rev_loop:
    cmp rdx, r13
    jb .rev_done
    mov al, byte ptr [rbp + rdx]
    mov byte ptr [r15 + r14], al
    inc r14
    dec rdx
    jmp .rev_loop
.rev_done:
    mov byte ptr [r15 + r14], 10
    inc r14
    lea r13, [rbx + 1]
    jmp .line_loop
.no_newline:
    mov rdx, r12
    dec rdx
.rev_final:
    cmp rdx, r13
    jb .rev_final_done
    mov al, byte ptr [rbp + rdx]
    mov byte ptr [r15 + r14], al
    inc r14
    dec rdx
    jmp .rev_final
.rev_final_done:
    mov byte ptr [r15 + r14], 10
    inc r14
.write_output:
    mov rsi, r15
    mov rdx, r14
.write_loop:
    test rdx, rdx
    jz .exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .exit
    add rsi, rax
    sub rdx, rax
    jmp .write_loop
.exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
