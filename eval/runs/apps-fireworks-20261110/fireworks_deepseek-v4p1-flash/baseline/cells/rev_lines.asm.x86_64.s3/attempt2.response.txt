.globl _start
.text
_start:
    mov qword ptr [rip + inlen], 0
.read_loop:
    mov r12, qword ptr [rip + inlen]
    cmp r12, 65536
    jae .read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add qword ptr [rip + inlen], rax
    jmp .read_loop
.read_done:
    lea r9, [rip + inbuf]
    lea r15, [rip + outbuf]
    mov r12, qword ptr [rip + inlen]
    xor r13d, r13d
    xor r14d, r14d
.line_loop:
    cmp r13, r12
    jae .write_output
    mov rbx, r13
.find_nl:
    cmp rbx, r12
    jae .no_newline
    cmp byte ptr [r9 + rbx], 10
    je .found_newline
    inc rbx
    jmp .find_nl
.found_newline:
    mov rdx, rbx
.rev_loop:
    cmp rdx, r13
    jbe .rev_done
    dec rdx
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [r15 + r14], al
    inc r14
    jmp .rev_loop
.rev_done:
    mov byte ptr [r15 + r14], 10
    inc r14
    lea r13, [rbx + 1]
    jmp .line_loop
.no_newline:
    mov rdx, r12
.rev_final:
    cmp rdx, r13
    jbe .rev_final_done
    dec rdx
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [r15 + r14], al
    inc r14
    jmp .rev_final
.rev_final_done:
    mov byte ptr [r15 + r14], 10
    inc r14
.write_output:
    mov qword ptr [rip + outlen], r14
    mov qword ptr [rip + woff], 0
.write_loop:
    mov rbx, qword ptr [rip + woff]
    mov r14, qword ptr [rip + outlen]
    cmp rbx, r14
    jae .exit
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .exit
    add qword ptr [rip + woff], rax
    jmp .write_loop
.exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072
inlen:
    .zero 8
outlen:
    .zero 8
woff:
    .zero 8
