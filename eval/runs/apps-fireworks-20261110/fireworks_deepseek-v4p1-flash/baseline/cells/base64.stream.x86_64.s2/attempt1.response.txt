.globl _start

.section .bss
.align 16
inbuf:
    .zero 65536
encbuf:
    .zero 66000

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline:
    .byte 10

.text
_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + encbuf]
    mov rax, r12
    xor edx, edx
    mov r9, 3
    div r9
    mov r10, rax
    mov r11, rdx
    lea rbx, [rip + b64tab]
    test r10, r10
    jz enc_rem
enc_triple_loop:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]
    mov r8d, eax
    shr r8d, 2
    mov r9b, byte ptr [rbx + r8]
    mov byte ptr [rdi], r9b
    and eax, 3
    shl eax, 4
    mov r8d, ecx
    shr r8d, 4
    or eax, r8d
    mov r9b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r9b
    and ecx, 15
    shl ecx, 2
    mov r8d, edx
    shr r8d, 6
    or ecx, r8d
    mov r9b, byte ptr [rbx + rcx]
    mov byte ptr [rdi+2], r9b
    and edx, 63
    mov r9b, byte ptr [rbx + rdx]
    mov byte ptr [rdi+3], r9b
    add rsi, 3
    add rdi, 4
    dec r10
    jnz enc_triple_loop
enc_rem:
    test r11, r11
    jz enc_done
    cmp r11, 1
    je enc_rem1
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi+1]
    mov r8d, eax
    shr r8d, 2
    mov r9b, byte ptr [rbx + r8]
    mov byte ptr [rdi], r9b
    and eax, 3
    shl eax, 4
    mov r8d, ecx
    shr r8d, 4
    or eax, r8d
    mov r9b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r9b
    and ecx, 15
    shl ecx, 2
    mov r9b, byte ptr [rbx + rcx]
    mov byte ptr [rdi+2], r9b
    mov byte ptr [rdi+3], 61
    add rdi, 4
    jmp enc_done
enc_rem1:
    movzx eax, byte ptr [rsi]
    mov r8d, eax
    shr r8d, 2
    mov r9b, byte ptr [rbx + r8]
    mov byte ptr [rdi], r9b
    and eax, 3
    shl eax, 4
    mov r9b, byte ptr [rbx + rax]
    mov byte ptr [rdi+1], r9b
    mov byte ptr [rdi+2], 61
    mov byte ptr [rdi+3], 61
    add rdi, 4
enc_done:
    lea rax, [rip + encbuf]
    mov r12, rdi
    sub r12, rax
    lea rbx, [rip + encbuf]
    xor r13, r13
write_loop:
    cmp r13, r12
    jae write_done
    mov rdx, r12
    sub rdx, r13
    cmp rdx, 76
    jbe wl_chunk_ok
    mov rdx, 76
wl_chunk_ok:
    lea rsi, [rbx + r13]
    mov rdi, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    add r13, 76
    jmp write_loop
write_done:
    xor edi, edi
    mov rax, 60
    syscall
