.text
.globl _start
_start:
    # Read all of stdin into inbuf
    xor r12, r12
.read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    js .error_exit
    jz .done_reading
    add r12, rax
    jmp .read_loop

.done_reading:
    lea r15, [rip + hexdigits]
    xor r13, r13

.row_loop:
    cmp r13, r12
    jae .after_rows
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe .n_ok
    mov r14, 16
.n_ok:
    lea rsi, [rip + inbuf]
    add rsi, r13
    lea rdi, [rip + outbuf]

    # Offset: 8 hex digits
    mov eax, r13d
    mov rcx, 8
.offset_hex_loop:
    rol eax, 4
    mov edx, eax
    and edx, 0x0F
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec rcx
    jnz .offset_hex_loop

    # Two spaces
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    add rdi, 2

    # Hex byte columns
    xor rcx, rcx
.byte_loop:
    cmp rcx, 16
    jae .bytes_done
    cmp rcx, r14
    jae .pad_three

    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    and eax, 0x0F
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    mov dl, byte ptr [r15 + rax]
    mov byte ptr [rdi+1], dl
    add rdi, 2
    mov byte ptr [rdi], ' '
    inc rdi
    jmp .next_byte

.pad_three:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3

.next_byte:
    cmp rcx, 7
    jne .no_extra
    mov byte ptr [rdi], ' '
    inc rdi
.no_extra:
    inc rcx
    jmp .byte_loop

.bytes_done:
    # Space and opening pipe
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], '|'
    add rdi, 2

    # ASCII column
    xor rcx, rcx
.ascii_loop:
    cmp rcx, r14
    jae .ascii_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb .dot
    cmp al, 0x7E
    ja .dot
    mov byte ptr [rdi], al
    inc rdi
    jmp .ascii_next
.dot:
    mov byte ptr [rdi], '.'
    inc rdi
.ascii_next:
    inc rcx
    jmp .ascii_loop

.ascii_done:
    # Closing pipe and newline
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi+1], 0x0A
    add rdi, 2

    # Write the line
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r13, 16
    jmp .row_loop

.after_rows:
    # Final offset line
    lea rdi, [rip + outbuf]
    mov eax, r12d
    mov rcx, 8
.total_hex_loop:
    rol eax, 4
    mov edx, eax
    and edx, 0x0F
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec rcx
    jnz .total_hex_loop
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    # Exit with status 0
    mov eax, 60
    xor edi, edi
    syscall

.error_exit:
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 128
