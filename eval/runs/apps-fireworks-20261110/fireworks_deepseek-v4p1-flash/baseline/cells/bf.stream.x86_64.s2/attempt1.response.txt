.globl _start

.section .text
_start:
    # Read all stdin into inbuf
    lea rbx, [rip + inbuf]
    xor r8, r8
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r8]
    mov rdx, 65536
    sub rdx, r8
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r8, rax
    cmp r8, 65536
    jb read_loop
read_done:
    # r8 = total input length
    mov r14, rbx          # r14 = inbuf base (program base)
    mov rcx, r8           # rcx = total length
    xor r9, r9            # index
find_bang:
    cmp r9, rcx
    jae no_bang
    mov al, byte ptr [r14 + r9]
    cmp al, 0x21
    je found_bang
    inc r9
    jmp find_bang
found_bang:
    mov r8, r9            # proglen = index of '!'
    lea rbx, [r14 + r9 + 1] # input base
    mov rbp, rcx
    sub rbp, r9
    sub rbp, 1            # input length
    jmp bracket_check
no_bang:
    mov r8, rcx           # proglen = total length
    lea rbx, [r14 + rcx]  # input base = end
    xor rbp, rbp          # input length = 0

bracket_check:
    lea r9, [rip + match]
    lea r10, [rip + bstack]
    xor rsi, rsi          # i = 0
    xor rdi, rdi          # sp = 0
bracket_loop:
    cmp rsi, r8
    jae bracket_done
    mov al, byte ptr [r14 + rsi]
    cmp al, '['
    je do_push
    cmp al, ']'
    je do_pop
    inc rsi
    jmp bracket_loop
do_push:
    mov dword ptr [r10 + rdi*4], esi
    inc rdi
    inc rsi
    jmp bracket_loop
do_pop:
    test rdi, rdi
    jz bracket_fail
    dec rdi
    mov eax, dword ptr [r10 + rdi*4]
    mov dword ptr [r9 + rax*4], esi
    mov dword ptr [r9 + rsi*4], eax
    inc rsi
    jmp bracket_loop
bracket_done:
    test rdi, rdi
    jnz bracket_fail
    jmp execute
bracket_fail:
    mov rax, 60
    mov rdi, 1
    syscall

execute:
    lea r12, [rip + tape]
    mov r13, r12
    xor r15, r15          # pc = 0
    xor r10, r10          # input pos = 0
execute_loop:
    cmp r15, r8
    jae execute_done
    movzx eax, byte ptr [r14 + r15]
    cmp al, '>'
    je do_gt
    cmp al, '<'
    je do_lt
    cmp al, '+'
    je do_plus
    cmp al, '-'
    je do_minus
    cmp al, '.'
    je do_dot
    cmp al, ','
    je do_comma
    cmp al, '['
    je do_lb
    cmp al, ']'
    je do_rb
    inc r15
    jmp execute_loop

do_gt:
    inc r13
    inc r15
    jmp execute_loop
do_lt:
    dec r13
    inc r15
    jmp execute_loop
do_plus:
    inc byte ptr [r13]
    inc r15
    jmp execute_loop
do_minus:
    dec byte ptr [r13]
    inc r15
    jmp execute_loop
do_dot:
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, 1
    syscall
    inc r15
    jmp execute_loop
do_comma:
    cmp r10, rbp
    jae do_comma_zero
    mov al, byte ptr [rbx + r10]
    inc r10
    mov byte ptr [r13], al
    inc r15
    jmp execute_loop
do_comma_zero:
    mov byte ptr [r13], 0
    inc r15
    jmp execute_loop
do_lb:
    cmp byte ptr [r13], 0
    jne lb_nonzero
    mov eax, dword ptr [r9 + r15*4]
    lea r15, [rax + 1]
    jmp execute_loop
lb_nonzero:
    inc r15
    jmp execute_loop
do_rb:
    cmp byte ptr [r13], 0
    je rb_zero
    mov eax, dword ptr [r9 + r15*4]
    lea r15, [rax + 1]
    jmp execute_loop
rb_zero:
    inc r15
    jmp execute_loop

execute_done:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.balign 16
inbuf:
    .zero 65536
.balign 16
match:
    .zero 262144
.balign 16
tape:
    .zero 30000
.balign 16
bstack:
    .zero 4000
