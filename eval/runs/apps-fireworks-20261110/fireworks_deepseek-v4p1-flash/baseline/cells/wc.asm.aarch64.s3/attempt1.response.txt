```asm
    .globl _start
    .text
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    adrp x23, buf
    add x23, x23, :lo12:buf
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, done_reading
    mov x25, x0
    mov x26, x23

process_loop:
    ldrb w27, [x26], #1
    add x21, x21, #1
    cmp w27, #10
    b.eq newline
    cmp w27, #0x20
    b.eq is_ws
    sub w28, w27, #0x09
    cmp w28, #4
    b.ls is_ws
    cbnz x22, skip
    add x20, x20, #1
    mov x22, #1
    b skip
newline:
    add x19, x19, #1
    mov x22, #0
    b skip
is_ws:
    mov x22, #0
skip:
    subs x25, x25, #1
    b.ne process_loop
    b read_loop

done_reading:
    mov x0, x19
    bl print_num
    adrp x1, space_char
    add x1, x1, :lo12:space_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, x20
    bl print_num
    adrp x1, space_char
    add x1, x1, :lo12:space_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, x21
    bl print_num
    adrp x1, newline_char
    add x1, x1, :lo12:newline_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_num:
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x2, x1, #31
    mov w3, #10
    mov x4, x0
    mov x5, #0
    cbnz x4, 1f
    mov w6, #0x30
    strb w6, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret
1:
    udiv x6, x4, x3
    msub x7, x6, x3, x4
    add w7, w7, #0x30
    strb w7, [x2, #-1]!
    add x5, x5, #1
    mov x4, x6
    cbnz x4, 1b
    mov x0, #1
    mov x1, x2
    mov x2, x5
    mov x8, #64
    svc #0
    ret

    .section .rodata
space_char:
    .ascii " "
newline_char:
    .ascii "\n"

    .bss
    .align 4
buf:
    .zero 4096
numbuf:
    .zero 32
```
