.globl _start

.section .text
_start:
    lea rbx, [rip + buffer]
    xor r12, r12

read_loop:
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov rax, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13
    xor r14, r14

scan_loop:
    cmp r14, r12
    jae scan_done
    mov al, byte ptr [rbx + r14]
    cmp al, 10
    jne scan_next
    mov rdi, r13
    mov rsi, r14
    call reverse_range
    mov rdx, r14
    sub rdx, r13
    lea rsi, [rbx + r13]
    mov rdi, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    lea r13, [r14 + 1]

scan_next:
    inc r14
    jmp scan_loop

scan_done:
    cmp r13, r12
    jae exit_prog
    mov rdi, r13
    mov rsi, r12
    call reverse_range
    mov rdx, r12
    sub rdx, r13
    lea rsi, [rbx + r13]
    mov rdi, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall

reverse_range:
    mov rax, rsi
    sub rax, rdi
    cmp rax, 2
    jb reverse_done
    lea rsi, [rsi - 1]
reverse_loop:
    cmp rdi, rsi
    jae reverse_done
    mov al, byte ptr [rbx + rdi]
    mov dl, byte ptr [rbx + rsi]
    mov byte ptr [rbx + rdi], dl
    mov byte ptr [rbx + rsi], al
    inc rdi
    dec rsi
    jmp reverse_loop
reverse_done:
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
buffer:
    .zero 65536
