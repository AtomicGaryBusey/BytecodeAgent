.globl _start
.text
_start:
    lea rbp, [rip + input_buf]
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, rbp
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    lea r13, [rip + out_buf]
row_loop:
    cmp rbx, r12
    jae done_rows
    lea r10, [rbp + rbx]
    mov r14, r12
    sub r14, rbx
    cmp r14, 16
    jbe n_ok
    mov r14, 16
n_ok:
    mov eax, ebx
    call write_hex8
    mov word ptr [r13], 0x2020
    add r13, 2
    xor r15d, r15d
hex_loop:
    cmp r15, r14
    jae pad_spaces
    movzx eax, byte ptr [r10 + r15]
    mov edx, eax
    shr edx, 4
    and eax, 0xF
    cmp dl, 10
    jb d1
    add dl, 'a' - 10
    jmp s1
d1:
    add dl, '0'
s1:
    mov [r13], dl
    cmp al, 10
    jb d2
    add al, 'a' - 10
    jmp s2
d2:
    add al, '0'
s2:
    mov [r13+1], al
    add r13, 2
    mov byte ptr [r13], ' '
    inc r13
    jmp next_i
pad_spaces:
    mov word ptr [r13], 0x2020
    mov byte ptr [r13+2], ' '
    add r13, 3
next_i:
    cmp r15, 7
    jne no_extra
    mov byte ptr [r13], ' '
    inc r13
no_extra:
    inc r15
    cmp r15, 16
    jb hex_loop
    mov byte ptr [r13], ' '
    inc r13
    mov byte ptr [r13], '|'
    inc r13
    xor r15d, r15d
char_loop:
    cmp r15, r14
    jae char_done
    movzx eax, byte ptr [r10 + r15]
    cmp al, 0x20
    jb not_print
    cmp al, 0x7E
    ja not_print
    jmp store_char
not_print:
    mov al, '.'
store_char:
    mov [r13], al
    inc r13
    inc r15
    jmp char_loop
char_done:
    mov byte ptr [r13], '|'
    inc r13
    mov byte ptr [r13], 0x0A
    inc r13
    add rbx, 16
    jmp row_loop
done_rows:
    mov eax, r12d
    call write_hex8
    mov byte ptr [r13], 0x0A
    inc r13
    lea rsi, [rip + out_buf]
    mov rdx, r13
    sub rdx, rsi
    mov edi, 1
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

write_hex8:
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    cmp edx, 10
    jb 2f
    add edx, 'a' - 10
    jmp 3f
2:
    add edx, '0'
3:
    mov [r13], dl
    inc r13
    sub ecx, 4
    jns 1b
    ret

.section .bss
.align 16
input_buf:
    .zero 65536
.align 16
out_buf:
    .zero 330000
