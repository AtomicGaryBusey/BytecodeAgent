.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
inbuf:
    .zero 65536
encbuf:
    .zero 70000
outbuf:
    .zero 70000

.text
_start:
    # read all input
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_ok

    # encode
    xor r13, r13          # i
    xor r14, r14          # j
    lea rbx, [rip + inbuf]
    lea rbp, [rip + encbuf]
    lea r15, [rip + alphabet]

encode_loop:
    lea rax, [r13 + 3]
    cmp rax, r12
    ja encode_remainder

    movzx eax, byte ptr [rbx + r13]
    movzx ecx, byte ptr [rbx + r13 + 1]
    movzx edx, byte ptr [rbx + r13 + 2]

    # c0
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r15 + rsi]
    mov byte ptr [rbp + r14], sil
    inc r14

    # c1
    and eax, 3
    shl eax, 4
    mov edi, ecx
    shr edi, 4
    or eax, edi
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rbp + r14], al
    inc r14

    # c2
    and ecx, 15
    shl ecx, 2
    mov edi, edx
    shr edi, 6
    or ecx, edi
    movzx ecx, byte ptr [r15 + rcx]
    mov byte ptr [rbp + r14], cl
    inc r14

    # c3
    and edx, 63
    movzx edx, byte ptr [r15 + rdx]
    mov byte ptr [rbp + r14], dl
    inc r14

    add r13, 3
    jmp encode_loop

encode_remainder:
    mov rax, r12
    sub rax, r13
    test rax, rax
    jz encode_done
    cmp rax, 1
    je rem1

    # rem == 2
    movzx eax, byte ptr [rbx + r13]
    movzx ecx, byte ptr [rbx + r13 + 1]

    # c0
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r15 + rsi]
    mov byte ptr [rbp + r14], sil
    inc r14

    # c1
    and eax, 3
    shl eax, 4
    mov edi, ecx
    shr edi, 4
    or eax, edi
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rbp + r14], al
    inc r14

    # c2
    and ecx, 15
    shl ecx, 2
    movzx ecx, byte ptr [r15 + rcx]
    mov byte ptr [rbp + r14], cl
    inc r14

    # '='
    mov byte ptr [rbp + r14], 61
    inc r14
    jmp encode_done

rem1:
    movzx eax, byte ptr [rbx + r13]

    # c0
    mov esi, eax
    shr esi, 2
    movzx esi, byte ptr [r15 + rsi]
    mov byte ptr [rbp + r14], sil
    inc r14

    # c1
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rbp + r14], al
    inc r14

    # '=='
    mov byte ptr [rbp + r14], 61
    inc r14
    mov byte ptr [rbp + r14], 61
    inc r14

encode_done:
    # r14 = enc_len
    xor r13, r13          # out index
    lea rbx, [rip + outbuf]
    lea rbp, [rip + encbuf]
    xor r15, r15          # k
    xor r12, r12          # column

newline_loop:
    cmp r15, r14
    jae newline_done
    mov al, byte ptr [rbp + r15]
    mov byte ptr [rbx + r13], al
    inc r13
    inc r15
    inc r12
    cmp r12, 76
    jne newline_loop
    mov byte ptr [rbx + r13], 10
    inc r13
    xor r12, r12
    jmp newline_loop

newline_done:
    test r12, r12
    jz write_out
    mov byte ptr [rbx + r13], 10
    inc r13

write_out:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
