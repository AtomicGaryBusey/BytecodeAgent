// rev_lines: reverse each line of stdin
.section .bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 65537

.section .text
.globl _start
_start:
    mov x19, #0
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x24, x19
    b.ge scan_done
    ldrb w25, [x20, x24]
    cmp w25, #10
    b.ne scan_next
    mov x26, x24
reverse_loop1:
    cmp x26, x23
    b.le reverse_done1
    sub x26, x26, #1
    ldrb w27, [x20, x26]
    strb w27, [x21, x22]
    add x22, x22, #1
    b reverse_loop1
reverse_done1:
    mov w27, #10
    strb w27, [x21, x22]
    add x22, x22, #1
    add x23, x24, #1
scan_next:
    add x24, x24, #1
    b scan_loop

scan_done:
    cmp x23, x19
    b.ge output_done
    mov x26, x19
reverse_loop2:
    cmp x26, x23
    b.le reverse_done2
    sub x26, x26, #1
    ldrb w27, [x20, x26]
    strb w27, [x21, x22]
    add x22, x22, #1
    b reverse_loop2
reverse_done2:
    mov w27, #10
    strb w27, [x21, x22]
    add x22, x22, #1

output_done:
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
