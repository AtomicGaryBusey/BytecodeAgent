.globl _start
.text
_start:
    // initialize jump table
    adrp x0, jumptab
    add x0, x0, :lo12:jumptab
    adrp x1, cmd_next
    add x1, x1, :lo12:cmd_next
    mov x2, #256
fill_loop:
    str x1, [x0], #8
    subs x2, x2, #1
    b.ne fill_loop
    // set special entries
    adrp x0, jumptab
    add x0, x0, :lo12:jumptab
    adrp x1, cmd_gt
    add x1, x1, :lo12:cmd_gt
    str x1, [x0, #496]
    adrp x1, cmd_lt
    add x1, x1, :lo12:cmd_lt
    str x1, [x0, #480]
    adrp x1, cmd_plus
    add x1, x1, :lo12:cmd_plus
    str x1, [x0, #344]
    adrp x1, cmd_minus
    add x1, x1, :lo12:cmd_minus
    str x1, [x0, #360]
    adrp x1, cmd_dot
    add x1, x1, :lo12:cmd_dot
    str x1, [x0, #368]
    adrp x1, cmd_comma
    add x1, x1, :lo12:cmd_comma
    str x1, [x0, #352]
    adrp x1, cmd_lb
    add x1, x1, :lo12:cmd_lb
    str x1, [x0, #728]
    adrp x1, cmd_rb
    add x1, x1, :lo12:cmd_rb
    str x1, [x0, #744]
    // read all stdin
    mov x19, #0          // total
read_loop:
    mov x0, #0           // fd 0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63          // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop
read_done:
    // find '!'
    adrp x25, inbuf
    add x25, x25, :lo12:inbuf
    mov x20, #0
find_bang:
    cmp x20, x19
    b.ge no_bang
    ldrb w24, [x25, x20]
    cmp w24, #0x21
    b.eq found_bang
    add x20, x20, #1
    b find_bang
found_bang:
    mov x23, x20        // prog_len
    add x26, x25, x20
    add x26, x26, #1    // input base
    sub x27, x19, x20
    sub x27, x27, #1    // input_len
    b setup_done
no_bang:
    mov x23, x19        // prog_len
    add x26, x25, x19   // input base = end
    mov x27, #0         // input_len = 0
setup_done:
    // bracket check
    mov x21, #0         // stack pointer
    mov x20, #0         // i
balance_loop:
    cmp x20, x23
    b.ge balance_done
    ldrb w24, [x25, x20]
    cmp w24, #'['
    b.eq push_bracket
    cmp w24, #']'
    b.eq pop_bracket
    add x20, x20, #1
    b balance_loop
push_bracket:
    adrp x0, bstack
    add x0, x0, :lo12:bstack
    str w20, [x0, x21, lsl #2]
    add x21, x21, #1
    add x20, x20, #1
    b balance_loop
pop_bracket:
    cbz x21, unbalanced
    sub x21, x21, #1
    adrp x0, bstack
    add x0, x0, :lo12:bstack
    ldr w1, [x0, x21, lsl #2]   // j
    adrp x2, match
    add x2, x2, :lo12:match
    strh w1, [x2, x20, lsl #1]  // match[i] = j
    strh w20, [x2, x1, lsl #1]  // match[j] = i
    add x20, x20, #1
    b balance_loop
balance_done:
    cbnz x21, unbalanced
    b run_program
unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0
run_program:
    adrp x28, match
    add x28, x28, :lo12:match
    adrp x29, tape
    add x29, x29, :lo12:tape
    adrp x30, outbuf
    add x30, x30, :lo12:outbuf
    adrp x18, jumptab
    add x18, x18, :lo12:jumptab
    mov x19, #0          // ip
    mov x20, #0          // dp
    mov x21, #0          // outlen
    mov x22, #0          // input_pos
run_loop:
    cmp x19, x23
    b.ge run_done
    ldrb w24, [x25, x19]
    ldr x0, [x18, x24, lsl #3]
    br x0
cmd_gt:
    add x20, x20, #1
    b cmd_next
cmd_lt:
    sub x20, x20, #1
    b cmd_next
cmd_plus:
    ldrb w24, [x29, x20]
    add w24, w24, #1
    strb w24, [x29, x20]
    b cmd_next
cmd_minus:
    ldrb w24, [x29, x20]
    sub w24, w24, #1
    strb w24, [x29, x20]
    b cmd_next
cmd_dot:
    ldrb w24, [x29, x20]
    strb w24, [x30, x21]
    add x21, x21, #1
    b cmd_next
cmd_comma:
    cmp x22, x27
    b.ge comma_zero
    ldrb w24, [x26, x22]
    add x22, x22, #1
    b comma_store
comma_zero:
    mov w24, #0
comma_store:
    strb w24, [x29, x20]
    b cmd_next
cmd_lb:
    ldrb w24, [x29, x20]
    cbnz w24, cmd_next
    ldrh w24, [x28, x19, lsl #1]
    mov x19, x24
    b cmd_next
cmd_rb:
    ldrb w24, [x29, x20]
    cbz w24, cmd_next
    ldrh w24, [x28, x19, lsl #1]
    mov x19, x24
    b cmd_next
cmd_next:
    add x19, x19, #1
    b run_loop
run_done:
    // write output
    cmp x21, #0
    b.eq exit0
    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x2, x21
    mov x8, #64
    svc #0
exit0:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 4
inbuf: .zero 65536
.align 4
tape: .zero 30000
.align 4
outbuf: .zero 65536
.align 4
match: .zero 131072
.align 4
bstack: .zero 4096
.align 3
jumptab: .zero 2048
