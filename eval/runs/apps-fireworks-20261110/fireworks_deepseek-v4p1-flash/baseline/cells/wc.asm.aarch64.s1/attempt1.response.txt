```asm
    .globl _start

    .equ READ_BUF_SIZE, 8192
    .equ OUT_BUF_SIZE, 128

    .text
_start:
    mov x19, #0          // bytes
    mov x20, #0          // lines
    mov x21, #0          // words
    mov x22, #0          // in_word

read_loop:
    mov x0, #0           // fd = stdin
    adrp x1, readbuf
    add x1, x1, :lo12:readbuf
    mov x2, #READ_BUF_SIZE
    mov x8, #63          // read
    svc #0

    cmp x0, #0
    b.eq read_done

    mov x2, x0           // counter
    adrp x3, readbuf
    add x3, x3, :lo12:readbuf

process_loop:
    cbz x2, read_loop
    ldrb w4, [x3], #1
    add x19, x19, #1

    cmp w4, #10
    b.eq is_newline

    cmp w4, #0x20
    b.eq is_ws
    cmp w4, #0x09
    b.eq is_ws
    cmp w4, #0x0B
    b.eq is_ws
    cmp w4, #0x0C
    b.eq is_ws
    cmp w4, #0x0D
    b.eq is_ws

    cbz x22, start_word
    b next_byte

is_newline:
    add x20, x20, #1
    mov x22, #0
    b next_byte

is_ws:
    mov x22, #0
    b next_byte

start_word:
    add x21, x21, #1
    mov x22, #1

next_byte:
    sub x2, x2, #1
    b process_loop

read_done:
    adrp x27, outbuf
    add x27, x27, :lo12:outbuf
    add x27, x27, #OUT_BUF_SIZE

    sub x27, x27, #1
    mov w0, #10
    strb w0, [x27]

    mov x0, x19
    bl print_num

    sub x27, x27, #1
    mov w0, #32
    strb w0, [x27]

    mov x0, x21
    bl print_num

    sub x27, x27, #1
    mov w0, #32
    strb w0, [x27]

    mov x0, x20
    bl print_num

    adrp x0, outbuf
    add x0, x0, :lo12:outbuf
    add x0, x0, #OUT_BUF_SIZE
    sub x2, x0, x27

    mov x0, #1
    mov x1, x27
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_num:
    mov x2, #10
1:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add w4, w4, #48
    sub x27, x27, #1
    strb w4, [x27]
    mov x0, x3
    cbnz x0, 1b
    ret

    .bss
    .balign 4
readbuf:
    .skip READ_BUF_SIZE
    .balign 4
outbuf:
    .skip OUT_BUF_SIZE
```
