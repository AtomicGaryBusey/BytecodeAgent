.section .text
.globl _start

_start:
    lea rbx, [rip + buffer]

    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    lea rsi, [rbx + r12]
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:

    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor rcx, rcx

scan_loop:
    cmp rcx, r12
    jae scan_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 10
    jne 1f
    inc r13
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jb 3f
    cmp al, 13
    jbe 2f
    jmp 3f
2:
    xor r15, r15
    jmp 4f
3:
    test r15, r15
    jnz 4f
    inc r14
    mov r15, 1
4:
    inc rcx
    jmp scan_loop
scan_done:

    lea rdi, [rip + outbuf + 32]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r12
    call utoa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r14
    call utoa
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call utoa

    lea rsi, [rip + outbuf + 32]
    sub rsi, rdi
    mov rdx, rsi
    mov rsi, rdi
    mov rdi, 1
    mov rax, 1
    syscall

    xor rdi, rdi
    mov rax, 60
    syscall

utoa:
    mov rcx, 10
utoa_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz utoa_loop
    ret

.section .bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 32
