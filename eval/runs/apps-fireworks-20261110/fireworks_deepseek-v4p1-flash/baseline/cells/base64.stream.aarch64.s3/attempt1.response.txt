// base64 encoder for AArch64 Linux
// Reads stdin, writes base64 encoded output to stdout.

.text
.globl _start

_start:
    // Read all input into inbuf
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0                 // total bytes read

read_loop:
    mov x0, #0                  // fd = stdin
    add x1, x20, x19            // buf = inbuf + total
    mov x2, #49152
    sub x2, x2, x19             // count = 49152 - total
    cbz x2, read_done
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le read_done              // EOF or error
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, exit_program       // empty input -> exit

    // Setup alphabet address
    adrp x23, alphabet
    add x23, x23, :lo12:alphabet

    // Setup output buffer
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x21, x24                // outptr
    mov w22, #0                 // column

    // x20 is already inbuf base
    // x19 is remaining bytes

    cmp x19, #3
    b.lo tail

loop3:
    ldrb w3, [x20], #1
    ldrb w4, [x20], #1
    ldrb w5, [x20], #1
    lsl w3, w3, #16
    lsl w4, w4, #8
    orr w3, w3, w4
    orr w3, w3, w5

    mov w4, #18
    mov w5, #4
1:
    lsr w6, w3, w4
    and w6, w6, #63
    ldrb w6, [x23, w6]
