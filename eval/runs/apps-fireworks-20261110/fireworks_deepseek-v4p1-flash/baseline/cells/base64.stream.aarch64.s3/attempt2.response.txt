// base64 encoder for AArch64 Linux
// Reads all of stdin, writes base64 (RFC 4648) to stdout, 76-char lines.

.text
.globl _start

_start:
    // Read all input into inbuf
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0                 // total bytes read

read_loop:
    mov x0, #0                  // fd = stdin
    add x1, x20, x19            // buf = inbuf + total
    mov x2, #49152
    sub x2, x2, x19             // count = 49152 - total
    cbz x2, read_done
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le read_done              // EOF or error
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, exit0              // empty input -> exit with no output

    // Set up pointers
    mov x9, x20                 // input pointer
    add x10, x20, x19           // input end
    adrp x23, alphabet
    add x23, x23, :lo12:alphabet
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x11, x24                // output pointer
    mov x12, #0                 // column counter (0..75)

enc_loop:
    sub x13, x10, x9
    cmp x13, #3
    b.lo enc_tail

    ldrb w3, [x9], #1
    ldrb w4, [x9], #1
    ldrb w5, [x9], #1
    lsl w3, w3, #16
    lsl w4, w4, #8
    orr w3, w3, w4
    orr w3, w3, w5

    lsr w6, w3, #18
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    lsr w6, w3, #12
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    lsr w6, w3, #6
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    and w6, w3, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    add x12, x12, #4
    cmp x12, #76
    b.ne enc_loop
    mov w6, #10                 // '\n'
    strb w6, [x11], #1
    mov x12, #0
    b enc_loop

enc_tail:
    cmp x13, #1
    b.eq tail1
    cmp x13, #2
    b.eq tail2
    b finish

tail1:
    ldrb w3, [x9]
    lsl w3, w3, #16

    lsr w6, w3, #18
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    lsr w6, w3, #12
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    mov w6, #61                 // '='
    strb w6, [x11], #1
    strb w6, [x11], #1
    add x12, x12, #4
    b tail_nl

tail2:
    ldrb w3, [x9], #1
    ldrb w4, [x9], #1
    lsl w3, w3, #16
    lsl w4, w4, #8
    orr w3, w3, w4

    lsr w6, w3, #18
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    lsr w6, w3, #12
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    lsr w6, w3, #6
    and w6, w6, #63
    ldrb w6, [x23, x6]
    strb w6, [x11], #1

    mov w6, #61                 // '='
    strb w6, [x11], #1
    add x12, x12, #4

tail_nl:
    cmp x12, #76
    b.ne tail_done
    mov w6, #10                 // '\n'
    strb w6, [x11], #1
    mov x12, #0

tail_done:
    cbz x12, finish
    mov w6, #10                 // '\n'
    strb w6, [x11], #1

finish:
    sub x2, x11, x24            // output length
    mov x1, x24
write_loop:
    cbz x2, exit0
    mov x0, #1                  // fd = stdout
    mov x8, #64                 // write
    svc #0
    cmp x0, #0
    b.le exit0
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

exit0:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

.section .rodata
.align 4
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
inbuf:
    .zero 49152
.align 4
outbuf:
    .zero 70000
