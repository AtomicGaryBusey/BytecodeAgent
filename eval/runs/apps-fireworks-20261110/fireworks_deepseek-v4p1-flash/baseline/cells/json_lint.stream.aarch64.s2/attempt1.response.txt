```asm
.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_msg:
    .ascii "error at "
err_len = . - err_msg
newline:
    .ascii "\n"
nl_len = . - newline

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
out_buf:
    .zero 32

.text
.globl _start
_start:
    // read input
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0          // length
.Lread_loop:
    mov x2, #65536
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x20, x20, x0
    b .Lread_loop
.Lread_done:
    mov x21, x19         // cur = base
    bl parse_document
    cmp x0, #0
    b.ne .Lfail
    // success
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #ok_len
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
.Lfail:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #err_len
    mov x8, #64
    svc #0
    mov x0, x22
    bl print_uint
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #nl_len
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip_ws: advances x21 over whitespace
skip_ws:
    add x2, x19, x20   // end
.Lws_loop:
    cmp x21, x2
    b.hs .Lws_done
    ldrb w3, [x21]
    cmp w3, #0x20
    b.eq .Lws_adv
    cmp w3, #0x09
    b.eq .Lws_adv
    cmp w3, #0x0A
    b.eq .Lws_adv
    cmp w3, #0x0D
    b.eq .Lws_adv
    ret
.Lws_adv:
    add x21, x21, #1
    b .Lws_loop
.Lws_done:
    ret

// parse_document
parse_document:
    stp x29, x30, [sp, #-16]!
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Ldoc_fail
    bl parse_value
    cmp x0, #0
    b.ne .Ldoc_fail
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.ne .Ldoc_fail
    mov x0, #0
    b .Ldoc_ret
.Ldoc_fail:
    sub x22, x21, x19
    mov x0, #1
.Ldoc_ret:
    ldp x29, x30, [sp], #16
    ret

// parse_value
parse_value:
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lpv_fail
    ldrb w3, [x21]
    cmp w3, #'{'
    b.eq parse_object
    cmp w3, #'['
    b.eq parse_array
    cmp w3, #'"'
    b.eq parse_string
    cmp w3, #'t'
    b.eq parse_true
    cmp w3, #'f'
    b.eq parse_false
    cmp w3, #'n'
    b.eq parse_null
    cmp w3, #'-'
    b.eq parse_number
    cmp w3, #'0'
    b.lo .Lpv_fail
    cmp w3, #'9'
    b.hi .Lpv_fail
    b parse_number
.Lpv_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_true
parse_true:
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lpt_fail
    ldrb w3, [x21]
    cmp w3, #'t'
    b.ne .Lpt_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpt_fail
    ldrb w3, [x21]
    cmp w3, #'r'
    b.ne .Lpt_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpt_fail
    ldrb w3, [x21]
    cmp w3, #'u'
    b.ne .Lpt_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpt_fail
    ldrb w3, [x21]
    cmp w3, #'e'
    b.ne .Lpt_fail
    add x21, x21, #1
    mov x0, #0
    ret
.Lpt_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_false
parse_false:
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lpf_fail
    ldrb w3, [x21]
    cmp w3, #'f'
    b.ne .Lpf_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpf_fail
    ldrb w3, [x21]
    cmp w3, #'a'
    b.ne .Lpf_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpf_fail
    ldrb w3, [x21]
    cmp w3, #'l'
    b.ne .Lpf_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpf_fail
    ldrb w3, [x21]
    cmp w3, #'s'
    b.ne .Lpf_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpf_fail
    ldrb w3, [x21]
    cmp w3, #'e'
    b.ne .Lpf_fail
    add x21, x21, #1
    mov x0, #0
    ret
.Lpf_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_null
parse_null:
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lpn_fail
    ldrb w3, [x21]
    cmp w3, #'n'
    b.ne .Lpn_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpn_fail
    ldrb w3, [x21]
    cmp w3, #'u'
    b.ne .Lpn_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpn_fail
    ldrb w3, [x21]
    cmp w3, #'l'
    b.ne .Lpn_fail
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lpn_fail
    ldrb w3, [x21]
    cmp w3, #'l'
    b.ne .Lpn_fail
    add x21, x21, #1
    mov x0, #0
    ret
.Lpn_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_number
parse_number:
    add x2, x19, x20   // end
    // optional '-'
    cmp x21, x2
    b.hs .Lnum_fail
    ldrb w3, [x21]
    cmp w3, #'-'
    b.ne .Lnum_int
    add x21, x21, #1
.Lnum_int:
    cmp x21, x2
    b.hs .Lnum_fail
    ldrb w3, [x21]
    cmp w3, #'0'
    b.ne .Lnum_nonzero
    add x21, x21, #1
    b .Lnum_frac
.Lnum_nonzero:
    cmp w3, #'1'
    b.lo .Lnum_fail
    cmp w3, #'9'
    b.hi .Lnum_fail
    add x21, x21, #1
.Lnum_digits:
    cmp x21, x2
    b.hs .Lnum_frac
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lnum_frac
    cmp w3, #'9'
    b.hi .Lnum_frac
    add x21, x21, #1
    b .Lnum_digits
.Lnum_frac:
    cmp x21, x2
    b.hs .Lnum_exp
    ldrb w3, [x21]
    cmp w3, #'.'
    b.ne .Lnum_exp
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lnum_fail
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lnum_fail
    cmp w3, #'9'
    b.hi .Lnum_fail
    add x21, x21, #1
.Lnum_frac_digits:
    cmp x21, x2
    b.hs .Lnum_exp
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lnum_exp
    cmp w3, #'9'
    b.hi .Lnum_exp
    add x21, x21, #1
    b .Lnum_frac_digits
.Lnum_exp:
    cmp x21, x2
    b.hs .Lnum_ok
    ldrb w3, [x21]
    cmp w3, #'e'
    b.eq .Lnum_exp_start
    cmp w3, #'E'
    b.ne .Lnum_ok
.Lnum_exp_start:
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lnum_exp_digits
    ldrb w3, [x21]
    cmp w3, #'+'
    b.eq .Lnum_exp_sign
    cmp w3, #'-'
    b.ne .Lnum_exp_digits
.Lnum_exp_sign:
    add x21, x21, #1
.Lnum_exp_digits:
    cmp x21, x2
    b.hs .Lnum_fail
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lnum_fail
    cmp w3, #'9'
    b.hi .Lnum_fail
    add x21, x21, #1
.Lnum_exp_digits_loop:
    cmp x21, x2
    b.hs .Lnum_ok
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lnum_ok
    cmp w3, #'9'
    b.hi .Lnum_ok
    add x21, x21, #1
    b .Lnum_exp_digits_loop
.Lnum_ok:
    mov x0, #0
    ret
.Lnum_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_string
parse_string:
    add x2, x19, x20   // end
    cmp x21, x2
    b.hs .Lstr_fail
    ldrb w3, [x21]
    cmp w3, #'"'
    b.ne .Lstr_fail
    add x21, x21, #1
.Lstr_loop:
    cmp x21, x2
    b.hs .Lstr_fail
    ldrb w3, [x21]
    cmp w3, #'"'
    b.eq .Lstr_end
    cmp w3, #'\\'
    b.eq .Lstr_escape
    cmp w3, #0x20
    b.lo .Lstr_fail
    add x21, x21, #1
    b .Lstr_loop
.Lstr_end:
    add x21, x21, #1
    mov x0, #0
    ret
.Lstr_escape:
    add x21, x21, #1
    cmp x21, x2
    b.hs .Lstr_fail
    ldrb w3, [x21]
    cmp w3, #'"'
    b.eq .Lstr_esc_ok
    cmp w3, #'\\'
    b.eq .Lstr_esc_ok
    cmp w3, #'/'
    b.eq .Lstr_esc_ok
    cmp w3, #'b'
    b.eq .Lstr_esc_ok
    cmp w3, #'f'
    b.eq .Lstr_esc_ok
    cmp w3, #'n'
    b.eq .Lstr_esc_ok
    cmp w3, #'r'
    b.eq .Lstr_esc_ok
    cmp w3, #'t'
    b.eq .Lstr_esc_ok
    cmp w3, #'u'
    b.eq .Lstr_unicode
    b .Lstr_fail
.Lstr_esc_ok:
    add x21, x21, #1
    b .Lstr_loop
.Lstr_unicode:
    add x21, x21, #1   // consume 'u'
    mov w4, #4
.Lstr_hex_loop:
    cmp x21, x2
    b.hs .Lstr_fail
    ldrb w3, [x21]
    cmp w3, #'0'
    b.lo .Lstr_hex_alpha
    cmp w3, #'9'
    b.hi .Lstr_hex_alpha
    b .Lstr_hex_ok
.Lstr_hex_alpha:
    cmp w3, #'a'
    b.lo .Lstr_hex_upper
    cmp w3, #'f'
    b.hi .Lstr_hex_upper
    b .Lstr_hex_ok
.Lstr_hex_upper:
    cmp w3, #'A'
    b.lo .Lstr_fail
    cmp w3, #'F'
    b.hi .Lstr_fail
.Lstr_hex_ok:
    add x21, x21, #1
    subs w4, w4, #1
    b.ne .Lstr_hex_loop
    b .Lstr_loop
.Lstr_fail:
    sub x22, x21, x19
    mov x0, #1
    ret

// parse_array
parse_array:
    stp x29, x30, [sp, #-16]!
    add x2, x19, x20
    cmp x21, x2
    b.hs .Larr_fail
    ldrb w3, [x21]
    cmp w3, #'['
    b.ne .Larr_fail
    add x21, x21, #1
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Larr_fail
    ldrb w3, [x21]
    cmp w3, #']'
    b.ne .Larr_parse_val
    add x21, x21, #1
    mov x0, #0
    b .Larr_ret
.Larr_parse_val:
    bl parse_value
    cmp x0, #0
    b.ne .Larr_fail
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Larr_fail
    ldrb w3, [x21]
    cmp w3, #','
    b.eq .Larr_comma
    cmp w3, #']'
    b.eq .Larr_close
    b .Larr_fail
.Larr_comma:
    add x21, x21, #1
    bl skip_ws
    b .Larr_parse_val
.Larr_close:
    add x21, x21, #1
    mov x0, #0
    b .Larr_ret
.Larr_fail:
    sub x22, x21, x19
    mov x0, #1
.Larr_ret:
    ldp x29, x30, [sp], #16
    ret

// parse_object
parse_object:
    stp x29, x30, [sp, #-16]!
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lobj_fail
    ldrb w3, [x21]
    cmp w3, #'{'
    b.ne .Lobj_fail
    add x21, x21, #1
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lobj_fail
    ldrb w3, [x21]
    cmp w3, #'}'
    b.ne .Lobj_member
    add x21, x21, #1
    mov x0, #0
    b .Lobj_ret
.Lobj_member:
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lobj_fail
    ldrb w3, [x21]
    cmp w3, #'"'
    b.ne .Lobj_fail
    bl parse_string
    cmp x0, #0
    b.ne .Lobj_fail
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lobj_fail
    ldrb w3, [x21]
    cmp w3, #':'
    b.ne .Lobj_fail
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cmp x0, #0
    b.ne .Lobj_fail
    bl skip_ws
    add x2, x19, x20
    cmp x21, x2
    b.hs .Lobj_fail
    ldrb w3, [x21]
    cmp w3, #','
    b.eq .Lobj_comma
    cmp w3, #'}'
    b.eq .Lobj_close
    b .Lobj_fail
.Lobj_comma:
    add x21, x21, #1
    bl skip_ws
    b .Lobj_member
.Lobj_close:
    add x21, x21, #1
    mov x0, #0
    b .Lobj_ret
.Lobj_fail:
    sub x22, x21, x19
    mov x0, #1
.Lobj_ret:
    ldp x29, x30, [sp], #16
    ret

// print_uint: prints unsigned integer in x0 to stdout
print_uint:
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    add x1, x1, #32
    mov x5, x1
    mov x2, #10
.Lpu_loop:
    cbz x0, .Lpu_done
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add x4, x4, #'0'
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x3
    b .Lpu_loop
.Lpu_done:
    cmp x1, x5
    b.ne .Lpu_write
    sub x1, x1, #1
    mov w4, #'0'
    strb w4, [x1]
.Lpu_write:
    mov x0, #1
    mov x2, x5
    sub x2, x2, x1
    mov x8, #64
    svc #0
    ret
```
