.globl _start

.section .rodata
newline:
    .byte 10

.section .bss
.align 4
input_buf:
    .zero 65536
lines:
    .zero 8192

.text
_start:
    // Get base addresses
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x22, lines
    add x22, x22, :lo12:lines

    // Read all input
    mov x20, #0          // total bytes read
read_loop:
    mov x0, #0           // fd 0
    add x1, x19, x20     // buffer + total
    mov x2, #65536
    sub x2, x2, x20      // remaining space
    mov x8, #63          // read
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt read_error
    add x20, x20, x0
    b read_loop
read_error:
    mov x0, #1
    mov x8, #93
    svc #0
read_done:

    // Parse lines
    mov x21, #0          // line count
    mov x23, #0          // current offset
parse_loop:
    cmp x23, x20
    b.ge parse_done
    mov x24, x23         // start of line
scan_loop:
    cmp x23, x20
    b.ge scan_end_no_newline
    ldrb w25, [x19, x23]
    cmp w25, #10
    b.eq scan_found_newline
    add x23, x23, #1
    b scan_loop
scan_found_newline:
    sub x25, x23, x24    // length
    lsl x26, x21, #3
    add x26, x22, x26
    str w24, [x26, #0]
    str w25, [x26, #4]
    add x21, x21, #1
    add x23, x23, #1
    b parse_loop
scan_end_no_newline:
    sub x25, x20, x24
    lsl x26, x21, #3
    add x26, x22, x26
    str w24, [x26, #0]
    str w25, [x26, #4]
    add x21, x21, #1
    mov x23, x20
    b parse_loop
parse_done:

    // Insertion sort
    mov x26, #1          // i = 1
sort_outer:
    cmp x26, x21
    b.ge sort_done
    lsl x27, x26, #3
    add x27, x22, x27
    ldr w28, [x27, #0]   // key_offset
    ldr w29, [x27, #4]   // key_len
    sub x24, x26, #1     // j = i-1
sort_inner:
    cmp x24, #0
    b.lt sort_inner_done
    lsl x27, x24, #3
    add x27, x22, x27
    ldr w0, [x27, #0]    // off_j
    ldr w1, [x27, #4]    // len_j
    mov w2, w28          // key_offset
    mov w3, w29          // key_len
    bl cmp_lines
    cmp x0, #0
    b.le sort_inner_done
    // shift lines[j] to lines[j+1]
    lsl x25, x24, #3
    add x25, x22, x25
    ldr w0, [x25, #0]
    ldr w1, [x25, #4]
    add x25, x25, #8
    str w0, [x25, #0]
    str w1, [x25, #4]
    sub x24, x24, #1
    b sort_inner
sort_inner_done:
    add x25, x24, #1
    lsl x25, x25, #3
    add x25, x22, x25
    str w28, [x25, #0]
    str w29, [x25, #4]
    add x26, x26, #1
    b sort_outer
sort_done:

    // Output
    mov x26, #0
out_loop:
    cmp x26, x21
    b.ge out_done
    lsl x27, x26, #3
    add x27, x22, x27
    ldr w24, [x27, #0]   // offset
    ldr w25, [x27, #4]   // len
    // write line
    mov x0, #1
    add x1, x19, x24
    mov x2, x25
    mov x8, #64
    svc #0
    // write newline
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x26, x26, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #93
    svc #0

// Compare two lines
// x0 = off_a, x1 = len_a, x2 = off_b, x3 = len_b
// returns x0 = -1, 0, 1
cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, lo
    add x5, x19, x0
    add x6, x19, x2
    mov x7, #0
cmp_loop:
    cmp x7, x4
    b.ge cmp_lengths
    ldrb w8, [x5, x7]
    ldrb w9, [x6, x7]
    cmp w8, w9
    b.lo cmp_less
    b.hi cmp_greater
    add x7, x7, #1
    b cmp_loop
cmp_lengths:
    cmp x1, x3
    b.lo cmp_less
    b.hi cmp_greater
    mov x0, #0
    ret
cmp_less:
    mov x0, #-1
    ret
cmp_greater:
    mov x0, #1
    ret
