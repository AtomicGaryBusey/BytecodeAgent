.section .bss
.align 16
buffer:         .space 65536
line_offsets:   .space 4000
line_lengths:   .space 4000
outbuf:         .space 65536

.section .text
.globl _start
_start:
    # read input
    xor r12, r12            # total_bytes = 0
read_loop:
    mov rax, 0              # read
    mov rdi, 0              # stdin
    lea rsi, [buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines
    xor r13, r13            # line_count = 0
    xor r14, r14            # ptr = 0
parse_loop:
    cmp r14, r12
    jae parse_done
    mov r15, r14            # start = ptr
find_nl_loop:
    cmp r14, r12
    jae find_nl_done
    mov al, byte ptr [buffer + r14]
    cmp al, 0x0A
    je find_nl_done
    inc r14
    jmp find_nl_loop
find_nl_done:
    # length = r14 - r15
    mov rax, r14
    sub rax, r15
    # store offset and length
    lea rcx, [line_offsets]
    mov dword ptr [rcx + r13*4], r15d
    lea rcx, [line_lengths]
    mov dword ptr [rcx + r13*4], eax
    inc r13
    # skip newline if present
    cmp r14, r12
    jae parse_loop
    mov al, byte ptr [buffer + r14]
    cmp al, 0x0A
    jne parse_loop
    inc r14
    jmp parse_loop
parse_done:
    # sort lines (insertion sort)
    mov r14, 1              # i = 1
sort_outer:
    cmp r14, r13
    jae sort_done
    # load key
    lea rax, [line_offsets]
    mov ebx, dword ptr [rax + r14*4]    # key_off
    lea rax, [line_lengths]
    mov ebp, dword ptr [rax + r14*4]    # key_len
    mov r15, r14
    dec r15                 # j = i - 1
sort_inner:
    cmp r15, 0
    jl sort_inner_done
    # load current element at j
    lea rax, [line_offsets]
    mov r8d, dword ptr [rax + r15*4]    # off1
    lea rax, [line_lengths]
    mov r9d, dword ptr [rax + r15*4]    # len1
    # compute min_len = min(len1, key_len)
    mov r10d, r9d
    cmp r10d, ebp
    jbe min_ok
    mov r10d, ebp
min_ok:
    # compare bytes
    xor r11d, r11d          # k = 0
cmp_loop:
    cmp r11d, r10d
    jae cmp_equal
    lea rax, [buffer]
    mov cl, byte ptr [rax + r8 + r11]
