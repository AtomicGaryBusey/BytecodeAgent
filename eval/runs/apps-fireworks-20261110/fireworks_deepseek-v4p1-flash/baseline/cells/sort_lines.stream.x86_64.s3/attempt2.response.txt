.section .bss
.align 16
buffer:         .space 65536
line_offsets:   .space 4000
line_lengths:   .space 4000
outbuf:         .space 65536

.section .text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor r14, r14
parse_loop:
    cmp r14, r12
    jae parse_done
    mov r15, r14
find_nl_loop:
    cmp r14, r12
    jae find_nl_done
    mov al, byte ptr [buffer + r14]
    cmp al, 0x0A
    je find_nl_done
    inc r14
    jmp find_nl_loop
find_nl_done:
    mov rax, r14
    sub rax, r15
    mov dword ptr [line_offsets + r13*4], r15d
    mov dword ptr [line_lengths + r13*4], eax
    inc r13
    cmp r14, r12
    jae parse_loop
    mov al, byte ptr [buffer + r14]
    cmp al, 0x0A
    jne parse_loop
    inc r14
    jmp parse_loop
parse_done:
    mov r14, 1
sort_outer:
    cmp r14, r13
    jae sort_done
    mov ebx, dword ptr [line_offsets + r14*4]
    mov ebp, dword ptr [line_lengths + r14*4]
    mov r15, r14
    dec r15
sort_inner:
    cmp r15, 0
    jl sort_inner_done
    mov r8d, dword ptr [line_offsets + r15*4]
    mov r9d, dword ptr [line_lengths + r15*4]
    mov r10d, r9d
    cmp r10d, ebp
    jbe min_ok
    mov r10d, ebp
min_ok:
    xor r11d, r11d
cmp_8_loop:
    lea eax, [r11 + 8]
    cmp eax, r10d
    ja cmp_tail
    mov eax, r8d
    add eax, r11d
    mov rdi, qword ptr [buffer + rax]
    mov eax, ebx
    add eax, r11d
    mov rsi, qword ptr [buffer + rax]
    cmp rdi, rsi
    jne cmp_diff_8
    add r11d, 8
    jmp cmp_8_loop
cmp_diff_8:
    xor rdi, rsi
    tzcnt rcx, rdi
    shr rcx, 3
    mov eax, r8d
    add eax, r11d
    add eax, ecx
    movzx edx, byte ptr [buffer + rax]
    mov eax, ebx
    add eax, r11d
    add eax, ecx
    movzx ecx, byte ptr [buffer + rax]
    cmp edx, ecx
    ja shift_element
    jmp sort_inner_done
cmp_tail:
    cmp r11d, r10d
    jae cmp_equal
    mov eax, r8d
    add eax, r11d
    movzx edx, byte ptr [buffer + rax]
    mov eax, ebx
    add eax, r11d
    movzx ecx, byte ptr [buffer + rax]
    cmp edx, ecx
    jne cmp_tail_diff
    inc r11d
    jmp cmp_tail
cmp_tail_diff:
    ja shift_element
    jmp sort_inner_done
cmp_equal:
    cmp r9d, ebp
    ja shift_element
    jmp sort_inner_done
shift_element:
    mov eax, dword ptr [line_offsets + r15*4]
    lea rcx, [r15 + 1]
    mov dword ptr [line_offsets + rcx*4], eax
    mov eax, dword ptr [line_lengths + r15*4]
    mov dword ptr [line_lengths + rcx*4], eax
    dec r15
    jmp sort_inner
sort_inner_done:
    inc r15
    mov dword ptr [line_offsets + r15*4], ebx
    mov dword ptr [line_lengths + r15*4], ebp
    inc r14
    jmp sort_outer
sort_done:
    xor r12, r12
    xor r14, r14
output_loop:
    cmp r14, r13
    jae output_done
    mov r8d, dword ptr [line_offsets + r14*4]
    mov r9d, dword ptr [line_lengths + r14*4]
    xor r11d, r11d
copy_loop:
    cmp r11d, r9d
    jae copy_done
    mov eax, r8d
    add eax, r11d
    movzx ecx, byte ptr [buffer + rax]
    mov byte ptr [outbuf + r12], cl
    inc r12
    inc r11d
    jmp copy_loop
copy_done:
    mov byte ptr [outbuf + r12], 0x0A
    inc r12
    inc r14
    jmp output_loop
output_done:
    test r12, r12
    jz exit_prog
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r12
    syscall
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
