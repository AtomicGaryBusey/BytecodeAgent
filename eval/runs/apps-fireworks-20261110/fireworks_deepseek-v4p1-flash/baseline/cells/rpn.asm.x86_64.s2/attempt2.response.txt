.globl _start

.section .text
_start:
    xor rbx, rbx
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop
read_done:
    lea rsi, [rip + input_buf]
    mov rdi, rsi
    add rdi, rbx
line_loop:
    cmp rsi, rdi
    jae done
    mov r8, rsi
scan_line:
    cmp rsi, rdi
    jae line_end_no_newline
    cmp byte ptr [rsi], 0x0A
    je found_newline
    inc rsi
    jmp scan_line
found_newline:
    mov r9, rsi
    inc rsi
    jmp process_line
line_end_no_newline:
    mov r9, rdi
process_line:
    xor r15, r15
    mov r10, r8
skip_sep:
    cmp r10, r9
    jae end_tokens
    mov al, byte ptr [r10]
    cmp al, 0x20
    je skip_one
    cmp al, 0x09
    je skip_one
    jmp token_start
skip_one:
    inc r10
    jmp skip_sep
token_start:
    mov r11, r10
scan_token:
    inc r10
    cmp r10, r9
    jae token_end
    mov al, byte ptr [r10]
    cmp al, 0x20
    je token_end
    cmp al, 0x09
    je token_end
    jmp scan_token
token_end:
    mov rax, r10
    sub rax, r11
    cmp rax, 1
    je check_operator
    jmp check_integer
check_operator:
    mov al, byte ptr [r11]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp check_integer
check_integer:
    mov al, byte ptr [r11]
    xor r12, r12
    cmp al, '-'
    jne int_start
    mov r12, 1
    inc r11
    cmp r11, r10
    jae error_line
int_start:
    xor rax, rax
    mov r13, r11
    xor r14, r14
parse_digit:
    cmp r13, r10
    jae int_done
    movzx ecx, byte ptr [r13]
    sub ecx, '0'
    cmp ecx, 9
    ja error_line
    imul rax, rax, 10
    add rax, rcx
    inc r14
    cmp r14, 18
    ja error_line
    inc r13
    jmp parse_digit
int_done:
    cmp r14, 0
    je error_line
    test r12, r12
    jz push_value
    neg rax
push_value:
    lea rbx, [rip + stack_arr]
    mov qword ptr [rbx + r15*8], rax
    inc r15
    jmp next_token
op_add:
    cmp r15, 2
    jb error_line
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx + r15*8 - 16]
    add rax, qword ptr [rbx + r15*8 - 8]
    dec r15
    mov qword ptr [rbx + r15*8 - 8], rax
    jmp next_token
op_sub:
    cmp r15, 2
    jb error_line
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx + r15*8 - 16]
    sub rax, qword ptr [rbx + r15*8 - 8]
    dec r15
    mov qword ptr [rbx + r15*8 - 8], rax
    jmp next_token
op_mul:
    cmp r15, 2
    jb error_line
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx + r15*8 - 16]
    imul rax, qword ptr [rbx + r15*8 - 8]
    dec r15
    mov qword ptr [rbx + r15*8 - 8], rax
    jmp next_token
op_div:
    cmp r15, 2
    jb error_line
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx + r15*8 - 16]
    mov rcx, qword ptr [rbx + r15*8 - 8]
    test rcx, rcx
    jz error_line
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_div
    cmp rcx, -1
    je error_line
do_div:
    cqo
    idiv rcx
    dec r15
    mov qword ptr [rbx + r15*8 - 8], rax
    jmp next_token
op_mod:
    cmp r15, 2
    jb error_line
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx + r15*8 - 16]
    mov rcx, qword ptr [rbx + r15*8 - 8]
    test rcx, rcx
    jz error_line
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_mod
    cmp rcx, -1
    je error_line
do_mod:
    cqo
    idiv rcx
    dec r15
    mov qword ptr [rbx + r15*8 - 8], rdx
    jmp next_token
next_token:
    jmp skip_sep
end_tokens:
    cmp r15, 1
    jne error_line
    push rsi
    push rdi
    lea rbx, [rip + stack_arr]
    mov rax, qword ptr [rbx]
    call print_int
    pop rdi
    pop rsi
    jmp line_loop
error_line:
    mov r14, rsi
    mov r13, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_error]
    mov edx, 6
    syscall
    mov rsi, r14
    mov rdi, r13
    jmp line_loop
print_int:
    lea rsi, [rip + line_out]
    mov rdi, rsi
    test rax, rax
    jns .positive
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
.positive:
    lea rcx, [rip + line_out + 32]
    mov r8, rcx
    mov rbx, 10
.convert_loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz .convert_loop
.copy_loop:
    cmp rcx, r8
    jae .done_copy
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    jmp .copy_loop
.done_copy:
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + line_out]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret
done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
msg_error:
    .ascii "error\n"

.section .bss
.align 8
input_buf:
    .zero 65536
stack_arr:
    .zero 8192
line_out:
    .zero 32
