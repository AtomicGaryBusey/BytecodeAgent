.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jz .done_read
    lea rbx, [rip + buffer]
    lea rbp, [rbx + rax]

process_loop:
    cmp rbx, rbp
    jae read_loop
    movzx eax, byte ptr [rbx]
    inc rbx
    inc r14
    cmp al, 10
    jne .not_nl
    inc r12
.not_nl:
    cmp al, 0x20
    je .ws
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe .ws
    test r15d, r15d
    jnz process_loop
    inc r13
    mov r15d, 1
    jmp process_loop
.ws:
    xor r15d, r15d
    jmp process_loop

.done_read:
    mov rdi, r12
    call print_u64
    call print_space
    mov rdi, r13
    call print_u64
    call print_space
    mov rdi, r14
    call print_u64
    call print_newline
    xor edi, edi
    mov eax, 60
    syscall

print_u64:
    sub rsp, 32
    lea rsi, [rsp + 32]
    mov rax, rdi
    mov ecx, 10
.loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .loop
    mov rdx, rsp
    add rdx, 32
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    add rsp, 32
    ret

print_space:
    mov edi, 1
    lea rsi, [rip + space_char]
    mov edx, 1
    mov eax, 1
    syscall
    ret

print_newline:
    mov edi, 1
    lea rsi, [rip + newline_char]
    mov edx, 1
    mov eax, 1
    syscall
    ret

.section .rodata
space_char: .byte 0x20
newline_char: .byte 0x0A

.bss
.align 16
buffer: .zero 65536
