    .text
    .globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz done
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb not_ws
    cmp al, 13
    jbe is_ws
not_ws:
    test r15, r15
    jnz next_char
    inc r13
    mov r15, 1
    jmp next_char
is_ws:
    xor r15, r15
next_char:
    jmp process_loop

done:
    lea rdi, [rip + out_buf]
    mov rax, r12
    call format_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call format_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call format_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

format_num:
    push rbx
    sub rsp, 32
    lea rsi, [rsp + 31]
    mov rbx, 10
    test rax, rax
    jnz convert
    dec rsi
    mov byte ptr [rsi], 48
    jmp copy
convert:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert
copy:
    lea rcx, [rsp + 32]
    sub rcx, rsi
    rep movsb
    add rsp, 32
    pop rbx
    ret

    .section .bss
    .align 16
buffer:
    .zero 4096
out_buf:
    .zero 64
