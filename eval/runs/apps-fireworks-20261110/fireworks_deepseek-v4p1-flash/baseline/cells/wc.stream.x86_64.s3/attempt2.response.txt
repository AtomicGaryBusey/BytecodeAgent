    .text
    .globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne .not_nl
    inc r12
.not_nl:
    cmp al, 9
    jb .not_ws
    cmp al, 13
    jbe .is_ws
    cmp al, 32
    je .is_ws
.not_ws:
    test r15d, r15d
    jnz process_loop
    inc r13
    mov r15d, 1
    jmp process_loop
.is_ws:
    xor r15d, r15d
    jmp process_loop

read_done:
    lea rdi, [rip + out_buf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_num:
    push rbx
    push rsi
    push rcx
    push rdx
    sub rsp, 32
    lea rsi, [rsp + 32]
    mov rbx, 10
    test rax, rax
    jnz .digits
    dec rsi
    mov byte ptr [rsi], 48
    jmp .emit
.digits:
    xor edx, edx
    div rbx
    add edx, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .digits
.emit:
    lea rcx, [rsp + 32]
    sub rcx, rsi
.emit_loop:
    test rcx, rcx
    jz .done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp .emit_loop
.done:
    add rsp, 32
    pop rdx
    pop rcx
    pop rsi
    pop rbx
    ret

    .section .bss
    .align 16
buffer:
    .zero 65536
out_buf:
    .zero 64
