    .section .bss
    .align 4
inbuf:
    .zero 65536
outbuf:
    .zero 65537

    .text
    .globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf

    mov x21, #0
read_loop:
    mov x0, #0
    add x1, x19, x21
    movz x2, #1, lsl #16
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop
read_done:

    mov x22, #0
    mov x24, #0
    mov x25, #0
process_loop:
    cmp x25, x21
    b.hs process_end
    ldrb w2, [x19, x25]
    cmp w2, #10
    b.eq found_newline
    add x25, x25, #1
    b process_loop

found_newline:
    sub x5, x25, x24
    mov x4, x22
    cbz x5, skip_rev
    sub x3, x25, #1
rev_loop:
    ldrb w6, [x19, x3]
    strb w6, [x26, x4]
    sub x3, x3, #1
    add x4, x4, #1
    subs x5, x5, #1
    b.ne rev_loop
skip_rev:
    mov x22, x4
    mov w6, #10
    strb w6, [x26, x22]
    add x22, x22, #1
    add x24, x25, #1
    mov x25, x24
    b process_loop

process_end:
    cmp x24, x21
    b.hs done_output
    sub x5, x21, x24
    mov x4, x22
    sub x3, x21, #1
rev_final:
    ldrb w6, [x19, x3]
    strb w6, [x26, x4]
    sub x3, x3, #1
    add x4, x4, #1
    subs x5, x5, #1
    b.ne rev_final
    mov x22, x4
    mov w6, #10
    strb w6, [x26, x22]
    add x22, x22, #1

done_output:
    mov x0, #1
    mov x1, x26
    mov x2, x22
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
