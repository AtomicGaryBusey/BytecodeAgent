```assembly
.globl _start

.section .text
_start:
    # Read all stdin into inbuf
    xor r14, r14             # total = 0
read_loop:
    mov eax, 0               # sys_read
    xor edi, edi             # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r14
    mov rdx, 65536
    sub rdx, r14
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    cmp r14, 65536
    jb read_loop
read_done:
    # r14 = total length
    lea rbx, [rip + inbuf]
    xor rcx, rcx
find_bang:
    cmp rcx, r14
    jae no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r13, rcx             # prog_len
    lea r15, [rip + inbuf]
    add r15, rcx
    inc r15                  # input_start
    mov r12, r14
    sub r12, rcx
    dec r12                  # input_len
    jmp bracket_check
no_bang:
    mov r13, r14             # prog_len = total
    lea r15, [rip + inbuf]
    add r15, r14             # input_start = inbuf + total
    xor r12, r12             # input_len = 0
bracket_check:
    # rbx already = inbuf base
    xor r8, r8               # i = 0
    lea r9, [rip + bstack]
bracket_loop:
    cmp r8, r13
    jae bracket_done
    movzx eax, byte ptr [rbx + r8]
    cmp al, '['
    je push_bracket
    cmp al, ']'
    je pop_bracket
    inc r8
    jmp bracket_loop
push_bracket:
    mov qword ptr [r9], r8
    add r9, 8
    inc r8
    jmp bracket_loop
pop_bracket:
    lea r10, [rip + bstack]
    cmp r9, r10
    je bracket_error
    sub r9, 8
    mov r10, qword ptr [r9]
    lea r11, [rip + jump]
    mov eax, r8d
    inc eax
    mov dword ptr [r11 + r10*4], eax
    mov eax, r10d
    inc eax
    mov dword ptr [r11 + r8*4], eax
    inc r8
    jmp bracket_loop
bracket_done:
    lea r10, [rip + bstack]
    cmp r9, r10
    jne bracket_error
    jmp run_init

bracket_error:
    mov eax, 60
    mov edi, 1
    syscall

run_init:
    lea rbx, [rip + tape]
    lea r14, [rip + inbuf]
    lea rsi, [rip + outbuf]
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor rbp, rbp
run_loop:
    cmp r8, r13
    jae run_done
    movzx eax, byte ptr [r14 + r8]
    cmp al, '>'
    je op_gt
    cmp al, '<'
    je op_lt
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '.'
    je op_dot
    cmp al, ','
    je op_comma
    cmp al, '['
    je op_lb
    cmp al, ']'
    je op_rb
    inc r8
    jmp run_loop
op_gt:
    inc r9
    inc r8
    jmp run_loop
op_lt:
    dec r9
    inc r8
    jmp run_loop
op_plus:
    inc byte ptr [rbx + r9]
    inc r8
    jmp run_loop
op_minus:
    dec byte ptr [rbx + r9]
    inc r8
    jmp run_loop
op_dot:
    mov al, byte ptr [rbx + r9]
    mov byte ptr [rsi + rbp], al
    inc rbp
    inc r8
    jmp run_loop
op_comma:
    cmp r10, r12
    jae comma_eof
    mov al, byte ptr [r15 + r10]
    inc r10
    jmp comma_store
comma_eof:
    xor eax, eax
comma_store:
    mov byte ptr [rbx + r9], al
    inc r8
    jmp run_loop
op_lb:
    cmp byte ptr [rbx + r9], 0
    jne lb_not_zero
    lea rdx, [rip + jump]
    mov eax, dword ptr [rdx + r8*4]
    mov r8, rax
    jmp run_loop
lb_not_zero:
    inc r8
    jmp run_loop
op_rb:
    cmp byte ptr [rbx + r9], 0
    je rb_zero
    lea rdx, [rip + jump]
    mov eax, dword ptr [rdx + r8*4]
    mov r8, rax
    jmp run_loop
rb_zero:
    inc r8
    jmp run_loop
run_done:
    test rbp, rbp
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rdx, rbp
    syscall
exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 16
inbuf:   .zero 65536
tape:    .zero 30000
outbuf:  .zero 65536
.align 16
jump:    .zero 262144
.align 16
bstack:  .zero 8000
```
