.text
.globl _start
_start:
    // initialize counters
    mov x19, #0        // lines
    mov x20, #0        // words
    mov x21, #0        // bytes
    mov x22, #0        // in_word

    // load buffer address
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0         // fd 0
    mov x1, x23        // buf
    mov x2, #65536     // count
    mov x8, #63        // read
    svc #0
    cmp x0, #0
    b.le read_done

    add x21, x21, x0   // bytes += n
    mov x24, #0        // index = 0

process_loop:
    cmp x24, x0
    b.ge read_loop

    ldrb w25, [x23, x24]
    add x24, x24, #1

    // check newline
    cmp w25, #0x0A
    b.ne not_newline
    add x19, x19, #1
not_newline:

    // check whitespace
    cmp w25, #0x20
    b.eq is_space
    cmp w25, #0x09
    b.lo is_word
    cmp w25, #0x0D
    b.hi is_word

is_space:
    mov x22, #0
    b process_next

is_word:
    cbnz x22, process_next
    add x20, x20, #1
    mov x22, #1

process_next:
    b process_loop

read_done:
    // format output
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x25, x1        // save start

    mov x0, x19
    bl print_num

    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x20
    bl print_num

    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x21
    bl print_num

    mov w2, #0x0A
    strb w2, [x1], #1

    sub x2, x1, x25
    mov x1, x25
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// print_num: x0 = value, x1 = dest pointer
print_num:
    sub sp, sp, #32
    mov x2, sp
    add x3, x2, #32
    mov x4, #10
    mov x5, x0
    cbnz x5, convert_loop

    mov w6, #0x30
    strb w6, [x1], #1
    add sp, sp, #32
    ret

convert_loop:
    udiv x6, x5, x4
    msub x7, x6, x4, x5
    add x7, x7, #0x30
    sub x3, x3, #1
    strb w7, [x3]
    mov x5, x6
    cbnz x5, convert_loop

    add x8, x2, #32
copy_loop:
    cmp x3, x8
    b.ge copy_done
    ldrb w6, [x3], #1
    strb w6, [x1], #1
    b copy_loop

copy_done:
    add sp, sp, #32
    ret

.section .bss
.align 4
buf:
.zero 65536
outbuf:
.zero 64
