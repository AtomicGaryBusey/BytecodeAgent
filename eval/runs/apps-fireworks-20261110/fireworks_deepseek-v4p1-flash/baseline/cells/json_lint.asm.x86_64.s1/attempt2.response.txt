.text
.globl _start

_start:
    lea r13, [rip + buf]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz done_reading
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle done_reading
    add r12, rax
    jmp read_loop
done_reading:
    lea r14, [r13 + r12]
    mov rbx, r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r14
    jne error_at_rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp rbx, r14
    jae .L_skip_ws_ret
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je .L_skip_ws_advance
    cmp al, 0x09
    je .L_skip_ws_advance
    cmp al, 0x0A
    je .L_skip_ws_advance
    cmp al, 0x0D
    je .L_skip_ws_advance
.L_skip_ws_ret:
    ret
.L_skip_ws_advance:
    inc rbx
    jmp skip_ws

parse_value:
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error_at_rbx
    cmp al, '9'
    jbe parse_number
    jmp error_at_rbx

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r14
    jae error_at_rbx
    cmp byte ptr [rbx], '}'
    je .L_obj_empty
.L_obj_member_loop:
    call parse_string
    call skip_ws
    cmp rbx, r14
    jae error_at_rbx
    cmp byte ptr [rbx], ':'
    jne error_at_rbx
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je .L_obj_next
    cmp al, '}'
    je .L_obj_close
    jmp error_at_rbx
.L_obj_next:
    inc rbx
    call skip_ws
    jmp .L_obj_member_loop
.L_obj_empty:
    inc rbx
    ret
.L_obj_close:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r14
    jae error_at_rbx
    cmp byte ptr [rbx], ']'
    je .L_arr_empty
.L_arr_value_loop:
    call parse_value
    call skip_ws
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je .L_arr_next
    cmp al, ']'
    je .L_arr_close
    jmp error_at_rbx
.L_arr_next:
    inc rbx
    call skip_ws
    jmp .L_arr_value_loop
.L_arr_empty:
    inc rbx
    ret
.L_arr_close:
    inc rbx
    ret

parse_string:
    cmp rbx, r14
    jae error_at_rbx
    cmp byte ptr [rbx], '"'
    jne error_at_rbx
    inc rbx
.L_str_loop:
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je .L_str_end
    cmp al, 0x5C
    je .L_str_escape
    cmp al, 0x20
    jb error_at_rbx
    inc rbx
    jmp .L_str_loop
.L_str_escape:
    inc rbx
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je .L_str_esc_adv
    cmp al, 0x5C
    je .L_str_esc_adv
    cmp al, '/'
    je .L_str_esc_adv
    cmp al, 'b'
    je .L_str_esc_adv
    cmp al, 'f'
    je .L_str_esc_adv
    cmp al, 'n'
    je .L_str_esc_adv
    cmp al, 'r'
    je .L_str_esc_adv
    cmp al, 't'
    je .L_str_esc_adv
    cmp al, 'u'
    je .L_str_unicode
    jmp error_at_rbx
.L_str_esc_adv:
    inc rbx
    jmp .L_str_loop
.L_str_unicode:
    inc rbx
    mov ecx, 4
.L_str_hex_loop:
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb .L_str_not_hex
    cmp al, '9'
    jbe .L_str_hex_ok
    cmp al, 'A'
    jb .L_str_not_hex
    cmp al, 'F'
    jbe .L_str_hex_ok
    cmp al, 'a'
    jb .L_str_not_hex
    cmp al, 'f'
    jbe .L_str_hex_ok
.L_str_not_hex:
    jmp error_at_rbx
.L_str_hex_ok:
    inc rbx
    dec ecx
    jnz .L_str_hex_loop
    jmp .L_str_loop
.L_str_end:
    inc rbx
    ret

parse_number:
    cmp byte ptr [rbx], '-'
    jne .L_num_int_part
    inc rbx
    cmp rbx, r14
    jae error_at_rbx
.L_num_int_part:
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    je .L_num_zero
    cmp al, '1'
    jb error_at_rbx
    cmp al, '9'
    ja error_at_rbx
.L_num_digits:
    inc rbx
    cmp rbx, r14
    jae .L_num_frac_check
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb .L_num_frac_check
    cmp al, '9'
    ja .L_num_frac_check
    jmp .L_num_digits
.L_num_zero:
    inc rbx
    cmp rbx, r14
    jae .L_num_frac_check
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb .L_num_frac_check
    cmp al, '9'
    jbe error_at_rbx
.L_num_frac_check:
    cmp rbx, r14
    jae .L_num_exp_check
    cmp byte ptr [rbx], '.'
    jne .L_num_exp_check
    inc rbx
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb error_at_rbx
    cmp al, '9'
    ja error_at_rbx
.L_num_frac_digits:
    inc rbx
    cmp rbx, r14
    jae .L_num_exp_check
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb .L_num_exp_check
    cmp al, '9'
    ja .L_num_exp_check
    jmp .L_num_frac_digits
.L_num_exp_check:
    cmp rbx, r14
    jae .L_num_done
    movzx eax, byte ptr [rbx]
    cmp al, 'e'
    je .L_num_exp
    cmp al, 'E'
    je .L_num_exp
    ret
.L_num_exp:
    inc rbx
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je .L_num_exp_sign
    cmp al, '-'
    je .L_num_exp_sign
    jmp .L_num_exp_digit
.L_num_exp_sign:
    inc rbx
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
.L_num_exp_digit:
    cmp al, '0'
    jb error_at_rbx
    cmp al, '9'
    ja error_at_rbx
.L_num_exp_digits:
    inc rbx
    cmp rbx, r14
    jae .L_num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb .L_num_done
    cmp al, '9'
    ja .L_num_done
    jmp .L_num_exp_digits
.L_num_done:
    ret

parse_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp parse_literal
parse_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp parse_literal
parse_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp parse_literal

parse_literal:
    test ecx, ecx
    jz .L_lit_done
    cmp rbx, r14
    jae error_at_rbx
    movzx eax, byte ptr [rbx]
    cmp al, byte ptr [rsi]
    jne error_at_rbx
    inc rbx
    inc rsi
    dec ecx
    jmp parse_literal
.L_lit_done:
    ret

error_at_rbx:
    mov rax, rbx
    sub rax, r13
    lea rdi, [rip + num_buf + 20]
    mov rsi, rdi
    mov ecx, 10
.L_err_convert:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz .L_err_convert
    mov r8, rdi
    mov r9, rsi
    sub r9, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_nl]
    mov edx, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
msg_nl: .ascii "\n"
str_true: .ascii "true"
str_false: .ascii "false"
str_null: .ascii "null"

.section .bss
.align 16
buf: .zero 65536
num_buf: .zero 32
