.text
.globl _start
_start:
    // Get buffer address
    adrp x19, buf
    add x19, x19, :lo12:buf

    // Read loop
    mov x20, #0          // total bytes read
read_loop:
    mov x0, #0           // fd = 0 (stdin)
    add x1, x19, x20     // buf + offset
    mov x2, #65536
    sub x2, x2, x20      // remaining space
    cbz x2, read_done    // buffer full (should not happen for input <= 65536)
    mov x8, #63          // read
    svc #0
    cbz x0, read_done    // end of input
    add x20, x20, x0
    b read_loop

read_done:
    // x20 = total bytes
    // Count lines and words
    mov x21, #0          // lines
    mov x22, #0          // words
    mov x23, #0          // in_word flag (0 = outside word)
    mov x24, #0          // index

count_loop:
    cmp x24, x20
    b.hs count_done
    ldrb w0, [x19, x24]
    add x24, x24, #1

    // Check for newline
    cmp w0, #0x0A
    b.ne 1f
    add x21, x21, #1
1:
    // Check for whitespace: space, tab, newline, vtab, form feed, carriage return
    cmp w0, #0x20
    b.eq is_space
    cmp w0, #0x09
    b.eq is_space
    cmp w0, #0x0A
    b.eq is_space
    cmp w0, #0x0B
    b.eq is_space
    cmp w0, #0x0C
    b.eq is_space
    cmp w0, #0x0D
    b.eq is_space

    // Not whitespace
    cbz x23, new_word
    b count_loop

is_space:
    mov x23, #0
    b count_loop

new_word:
    add x22, x22, #1
    mov x23, #1
    b count_loop

count_done:
    // Build output string
    adrp x19, outbuf
    add x19, x19, :lo12:outbuf
    mov x1, x19          // x1 = output pointer

    // Print lines
    mov x0, x21
    bl print_num
    mov w2, #' '
    strb w2, [x1], #1

    // Print words
    mov x0, x22
    bl print_num
    mov w2, #' '
    strb w2, [x1], #1

    // Print bytes
    mov x0, x20
    bl print_num
    mov w2, #'\n'
    strb w2, [x1], #1

    // Write output
    mov x0, #1           // fd = 1 (stdout)
    mov x2, x1
    sub x2, x2, x19      // length
    mov x1, x19          // buffer start
    mov x8, #64          // write
    svc #0

    // Exit
    mov x0, #0
    mov x8, #93          // exit
    svc #0

// print_num: convert unsigned integer in x0 to decimal, append to buffer at x1
// x0 = number, x1 = output pointer, returns updated x1
print_num:
    cbz x0, print_zero
    sub sp, sp, #32
    mov x2, sp
    mov x3, #0           // digit count
1:
    cbz x0, 2f
    mov x4, #10
    udiv x5, x0, x4
    msub x6, x5, x4, x0   // x6 = x0 - x5*10
    add x6, x6, #'0'
    strb w6, [x2, x3]
    add x3, x3, #1
    mov x0, x5
    b 1b
2:
    cbz x3, 3f
    sub x3, x3, #1
    ldrb w6, [x2, x3]
    strb w6, [x1], #1
    b 2b
3:
    add sp, sp, #32
    ret

print_zero:
    mov w2, #'0'
    strb w2, [x1], #1
    ret

.section .bss
.align 4
buf:
    .zero 65536
outbuf:
    .zero 64
