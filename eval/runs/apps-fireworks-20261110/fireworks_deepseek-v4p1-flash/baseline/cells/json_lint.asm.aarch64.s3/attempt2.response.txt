.globl _start

.section .rodata
ok_msg:    .ascii "ok\n"
err_msg:   .ascii "error at "
nl_msg:    .ascii "\n"
lit_true:  .ascii "true"
lit_false: .ascii "false"
lit_null:  .ascii "null"

.section .bss
.align 4
inbuf:   .zero 65536
numbuf:  .zero 16

.section .text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
.Lread_loop:
    mov x0, #0
    add x1, x19, x20
    movz w2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x20, x20, x0
    b .Lread_loop
.Lread_done:
    mov x21, #0
    bl skip_ws
    bl parse_value
    cbnz x0, .Lerror
    bl skip_ws
    cmp x21, x20
    b.ne .Lerror
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.Lerror:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #16
    mov x0, x21
    mov x5, #10
    cbnz x0, .Ldig_loop
    sub x1, x1, #1
    mov w6, #48
    strb w6, [x1]
    b .Ldig_done
.Ldig_loop:
    udiv x7, x0, x5
    msub x9, x7, x5, x0
    add x9, x9, #48
    sub x1, x1, #1
    strb w9, [x1]
    mov x0, x7
    cbnz x0, .Ldig_loop
.Ldig_done:
    adrp x2, numbuf
    add x2, x2, :lo12:numbuf
    add x2, x2, #16
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip_ws: advance x21 past spaces/tabs/newlines/CR.
// w0 = first non-ws byte, or -1 at end of input.
skip_ws:
    cmp x21, x20
    b.hs .Lws_eof
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq .Lws_adv
    cmp w0, #0x09
    b.eq .Lws_adv
    cmp w0, #0x0A
    b.eq .Lws_adv
    cmp w0, #0x0D
    b.eq .Lws_adv
    ret
.Lws_adv:
    add x21, x21, #1
    b skip_ws
.Lws_eof:
    mov w0, #-1
    ret

// parse_value: x0 = 0 ok, 1 error. x21 = error offset on failure.
parse_value:
    cmp x21, x20
    b.hs .Lpv_err
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    cmp w0, #'-'
    b.eq parse_number
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls parse_number
.Lpv_err:
    mov x0, #1
    ret

parse_string:
    add x21, x21, #1
.Lstr_loop:
    cmp x21, x20
    b.hs .Lstr_err
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq .Lstr_end
    cmp w0, #0x5C
    b.eq .Lstr_esc
    cmp w0, #0x20
    b.lo .Lstr_err
    add x21, x21, #1
    b .Lstr_loop
.Lstr_esc:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lstr_err
    ldrb w0, [x19, x21]
    cmp w0, #'u'
    b.eq .Lstr_uni
    cmp w0, #0x22
    b.eq .Lstr_esc_ok
    cmp w0, #0x5C
    b.eq .Lstr_esc_ok
    cmp w0, #'/'
    b.eq .Lstr_esc_ok
    cmp w0, #'b'
    b.eq .Lstr_esc_ok
    cmp w0, #'f'
    b.eq .Lstr_esc_ok
    cmp w0, #'n'
    b.eq .Lstr_esc_ok
    cmp w0, #'r'
    b.eq .Lstr_esc_ok
    cmp w0, #'t'
    b.eq .Lstr_esc_ok
    b .Lstr_err
.Lstr_esc_ok:
    add x21, x21, #1
    b .Lstr_loop
.Lstr_uni:
    add x21, x21, #1
    mov w9, #4
.Lstr_uni_loop:
    cmp x21, x20
    b.hs .Lstr_err
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls .Lstr_uni_ok
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi .Lstr_err
.Lstr_uni_ok:
    add x21, x21, #1
    subs w9, w9, #1
    b.ne .Lstr_uni_loop
    b .Lstr_loop
.Lstr_end:
    add x21, x21, #1
    mov x0, #0
    ret
.Lstr_err:
    mov x0, #1
    ret

parse_number:
    cmp x21, x20
    b.hs .Lnum_err
    ldrb w0, [x19, x21]
    cmp w0, #'-'
    b.ne .Lnum_int
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_err
    ldrb w0, [x19, x21]
.Lnum_int:
    cmp w0, #'0'
    b.eq .Lnum_zero
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi .Lnum_err
    add x21, x21, #1
.Lnum_digits:
    cmp x21, x20
    b.hs .Lnum_after_int
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi .Lnum_after_int
    add x21, x21, #1
    b .Lnum_digits
.Lnum_zero:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_after_int
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls .Lnum_err
    b .Lnum_after_int
.Lnum_after_int:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'.'
    b.eq .Lnum_frac
    b .Lnum_exp_check
.Lnum_frac:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_err
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi .Lnum_err
    add x21, x21, #1
.Lnum_frac_loop:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi .Lnum_exp_check
    add x21, x21, #1
    b .Lnum_frac_loop
.Lnum_exp_check:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.eq .Lnum_exp
    cmp w0, #'E'
    b.eq .Lnum_exp
    b .Lnum_done
.Lnum_exp:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_err
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq .Lnum_exp_sign
    cmp w0, #'-'
    b.ne .Lnum_exp_digits
.Lnum_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_err
    ldrb w0, [x19, x21]
.Lnum_exp_digits:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi .Lnum_err
    add x21, x21, #1
.Lnum_exp_loop:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi .Lnum_done
    add x21, x21, #1
    b .Lnum_exp_loop
.Lnum_done:
    mov x0, #0
    ret
.Lnum_err:
    mov x0, #1
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp w0, #']'
    b.eq .Larr_empty
.Larr_elem:
    bl parse_value
    cbnz x0, .Larr_err
    bl skip_ws
    cmp w0, #','
    b.eq .Larr_comma
    cmp w0, #']'
    b.eq .Larr_end
    b .Larr_err
.Larr_comma:
    add x21, x21, #1
    bl skip_ws
    b .Larr_elem
.Larr_empty:
    add x21, x21, #1
    mov x0, #0
    b .Larr_ret
.Larr_end:
    add x21, x21, #1
    mov x0, #0
    b .Larr_ret
.Larr_err:
    mov x0, #1
.Larr_ret:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp w0, #'}'
    b.eq .Lobj_empty
.Lobj_member:
    cmp w0, #0x22
    b.ne .Lobj_err
    bl parse_string
    cbnz x0, .Lobj_err
    bl skip_ws
    cmp w0, #':'
    b.ne .Lobj_err
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cbnz x0, .Lobj_err
    bl skip_ws
    cmp w0, #','
    b.eq .Lobj_comma
    cmp w0, #'}'
    b.eq .Lobj_end
    b .Lobj_err
.Lobj_comma:
    add x21, x21, #1
    bl skip_ws
    b .Lobj_member
.Lobj_empty:
    add x21, x21, #1
    mov x0, #0
    b .Lobj_ret
.Lobj_end:
    add x21, x21, #1
    mov x0, #0
    b .Lobj_ret
.Lobj_err:
    mov x0, #1
.Lobj_ret:
    ldp x29, x30, [sp], #16
    ret

parse_true:
    adrp x10, lit_true
    add x10, x10, :lo12:lit_true
    mov w9, #4
    b parse_lit
parse_false:
    adrp x10, lit_false
    add x10, x10, :lo12:lit_false
    mov w9, #5
    b parse_lit
parse_null:
    adrp x10, lit_null
    add x10, x10, :lo12:lit_null
    mov w9, #4
    b parse_lit

parse_lit:
    cmp x21, x20
    b.hs .Llit_err
    ldrb w0, [x19, x21]
    ldrb w1, [x10], #1
    cmp w0, w1
    b.ne .Llit_err
    add x21, x21, #1
    subs w9, w9, #1
    b.ne parse_lit
    mov x0, #0
    ret
.Llit_err:
    mov x0, #1
    ret
