```asm
.globl _start

.section .rodata
ok_str:     .ascii "ok\n"
lit_true:   .ascii "true"
lit_false:  .ascii "false"
lit_null:   .ascii "null"
err_str:    .ascii "error at "

.section .bss
.align 4
inbuf:      .zero 65536
outbuf:     .zero 32

.section .text
_start:
    // read all input
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0          // total read
1:
    mov x0, #0           // fd 0
    add x1, x19, x20
    movz x2, #1, lsl #16 // 65536
    sub x2, x2, x20
    mov x8, #63          // read
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
    mov x21, #0          // index = 0
    bl skip_ws
    bl parse_value
    cbnz x0, .Lerror
    bl skip_ws
    cmp x21, x20
    b.ne .Lerror
    // success
    mov x0, #1
    adrp x1, ok_str
    add x1, x1, :lo12:ok_str
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.Lerror:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    adrp x2, err_str
    add x2, x2, :lo12:err_str
    mov x3, #9
    mov x4, #0
3:
    ldrb w5, [x2, x4]
    strb w5, [x1, x4]
    add x4, x4, #1
    cmp x4, x3
    b.lo 3b
    // convert x21 to decimal
    mov x0, x21
    mov x5, #10
    mov x6, #0
    cbz x0, 4f
5:
    udiv x7, x0, x5
    msub x8, x7, x5, x0
    add x8, x8, #'0'
    strb w8, [x1, x4]
    add x4, x4, #1
    add x6, x6, #1
    mov x0, x7
    cbnz x0, 5b
    b 6f
4:
    mov w8, #'0'
    strb w8, [x1, x4]
    add x4, x4, #1
    mov x6, #1
6:
    // reverse digits from index 9 to x4-1
    mov x7, #9
    sub x8, x4, #1
7:
    cmp x7, x8
    b.ge 8f
    ldrb w9, [x1, x7]
    ldrb w10, [x1, x8]
    strb w10, [x1, x7]
    strb w9, [x1, x8]
    add x7, x7, #1
    sub x8, x8, #1
    b 7b
8:
    mov w9, #'\n'
    strb w9, [x1, x4]
    add x4, x4, #1
    mov x0, #1
    // x1 already outbuf
    mov x2, x4
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip_ws: skips whitespace, returns current byte in w0 or -1 if EOF
skip_ws:
1:
    cmp x21, x20
    b.hs 2f
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq 3f
    cmp w0, #0x09
    b.eq 3f
    cmp w0, #0x0A
    b.eq 3f
    cmp w0, #0x0D
    b.eq 3f
    ret
3:
    add x21, x21, #1
    b 1b
2:
    mov w0, #-1
    ret

// parse_value: parses a value, returns 0 on success, 1 on error
parse_value:
    cmp x21, x20
    b.hs .Lerr_val
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq parse_string
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    cmp w0, #'-'
    b.eq parse_number
    cmp w0, #'0'
    b.lo .Lerr_val
    cmp w0, #'9'
    b.hi .Lerr_val
    b parse_number
.Lerr_val:
    mov x0, #1
    ret

// parse_string
parse_string:
    add x21, x21, #1
1:
    cmp x21, x20
    b.hs .Lerr_str
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq .Lstr_end
    cmp w0, #'\\'
    b.eq .Lescape
    cmp w0, #0x20
    b.lo .Lerr_str
    add x21, x21, #1
    b 1b
.Lescape:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lerr_str
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq .Lesc_ok
    cmp w0, #'\\'
    b.eq .Lesc_ok
    cmp w0, #'/'
    b.eq .Lesc_ok
    cmp w0, #'b'
    b.eq .Lesc_ok
    cmp w0, #'f'
    b.eq .Lesc_ok
    cmp w0, #'n'
    b.eq .Lesc_ok
    cmp w0, #'r'
    b.eq .Lesc_ok
    cmp w0, #'t'
    b.eq .Lesc_ok
    cmp w0, #'u'
    b.eq .Lesc_unicode
    b .Lerr_str
.Lesc_ok:
    add x21, x21, #1
    b 1b
.Lesc_unicode:
    add x21, x21, #1
    mov x9, #4
2:
    cmp x21, x20
    b.hs .Lerr_str
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo 3f
    cmp w0, #'9'
    b.hi 3f
    b 4f
3:
    cmp w0, #'a'
    b.lo 5f
    cmp w0, #'f'
    b.hi 5f
    b 4f
5:
    cmp w0, #'A'
    b.lo .Lerr_str
    cmp w0, #'F'
    b.hi .Lerr_str
4:
    add x21, x21, #1
    subs x9, x9, #1
    b.ne 2b
    b 1b
.Lstr_end:
    add x21, x21, #1
    mov x0, #0
    ret
.Lerr_str:
    mov x0, #1
    ret

// parse_number
parse_number:
    cmp x21, x20
    b.hs .Lerr_num
    ldrb w0, [x19, x21]
    cmp w0, #'-'
    b.ne .Lnum_no_minus
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lerr_num
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lerr_num
    cmp w0, #'9'
    b.hi .Lerr_num
.Lnum_no_minus:
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.ne .Lnum_digit1_9
    // leading zero
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lnum_check_frac
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lnum_check_frac
    cmp w0, #'9'
    b.hi .Lnum_check_frac
    b .Lerr_num
.Lnum_digit1_9:
    cmp w0, #'1'
    b.lo .Lerr_num
    cmp w0, #'9'
    b.hi .Lerr_num
    add x21, x21, #1
.Lnum_digits_loop:
    cmp x21, x20
    b.hs .Lnum_check_frac
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lnum_check_frac
    cmp w0, #'9'
    b.hi .Lnum_check_frac
    add x21, x21, #1
    b .Lnum_digits_loop
.Lnum_check_frac:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'.'
    b.ne .Lnum_check_exp
    // frac
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lerr_num
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lerr_num
    cmp w0, #'9'
    b.hi .Lerr_num
    add x21, x21, #1
.Lnum_frac_loop:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lnum_check_exp
    cmp w0, #'9'
    b.hi .Lnum_check_exp
    add x21, x21, #1
    b .Lnum_frac_loop
.Lnum_check_exp:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.eq .Lnum_exp
    cmp w0, #'E'
    b.ne .Lnum_done
.Lnum_exp:
    add x21, x21, #1
    cmp x21, x20
    b.hs .Lerr_num
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq .Lnum_exp_sign
    cmp w0, #'-'
    b.ne .Lnum_exp_digit
.Lnum_exp_sign:
    add x21, x21, #1
.Lnum_exp_digit:
    cmp x21, x20
    b.hs .Lerr_num
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lerr_num
    cmp w0, #'9'
    b.hi .Lerr_num
    add x21, x21, #1
.Lnum_exp_loop:
    cmp x21, x20
    b.hs .Lnum_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lo .Lnum_done
    cmp w0, #'9'
    b.hi .Lnum_done
    add x21, x21, #1
    b .Lnum_exp_loop
.Lnum_done:
    mov x0, #0
    ret
.Lerr_num:
    mov x0, #1
    ret

// parse_array
parse_array:
    add x21, x21, #1
    bl skip_ws
    cmp w0, #']'
    b.eq .Larr_empty
    bl parse_value
    cbnz x0, .Lerr_arr
.Larr_loop:
    bl skip_ws
    cmp w0, #','
    b.eq .Larr_comma
    cmp w0, #']'
    b.eq .Larr_end
    b .Lerr_arr
.Larr_comma:
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cbnz x0, .Lerr_arr
    b .Larr_loop
.Larr_end:
    add x21, x21, #1
    mov x0, #0
    ret
.Larr_empty:
    add x21, x21, #1
    mov x0, #0
    ret
.Lerr_arr:
    mov x0, #1
    ret

// parse_object
parse_object:
    add x21, x21, #1
    bl skip_ws
    cmp w0, #'}'
    b.eq .Lobj_empty
.Lobj_member:
    cmp w0, #'"'
    b.ne .Lerr_obj
    bl parse_string
    cbnz x0, .Lerr_obj
    bl skip_ws
    cmp w0, #':'
    b.ne .Lerr_obj
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cbnz x0, .Lerr_obj
.Lobj_loop:
    bl skip_ws
    cmp w0, #','
    b.eq .Lobj_comma
    cmp w0, #'}'
    b.eq .Lobj_end
    b .Lerr_obj
.Lobj_comma:
    add x21, x21, #1
    bl skip_ws
    cmp w0, #'"'
    b.ne .Lerr_obj
    bl parse_string
    cbnz x0, .Lerr_obj
    bl skip_ws
    cmp w0, #':'
    b.ne .Lerr_obj
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cbnz x0, .Lerr_obj
    b .Lobj_loop
.Lobj_end:
    add x21, x21, #1
    mov x0, #0
    ret
.Lobj_empty:
    add x21, x21, #1
    mov x0, #0
    ret
.Lerr_obj:
    mov x0, #1
    ret

// parse_true
parse_true:
    mov x9, #4
    adrp x10, lit_true
    add x10, x10, :lo12:lit_true
1:
    cmp x21, x20
    b.hs .Lerr_true
    ldrb w0, [x19, x21]
    ldrb w1, [x10], #1
    cmp w0, w1
    b.ne .Lerr_true
    add x21, x21, #1
    subs x9, x9, #1
    b.ne 1b
    mov x0, #0
    ret
.Lerr_true:
    mov x0, #1
    ret

// parse_false
parse_false:
    mov x9, #5
    adrp x10, lit_false
    add x10, x10, :lo12:lit_false
1:
    cmp x21, x20
    b.hs .Lerr_false
    ldrb w0, [x19, x21]
    ldrb w1, [x10], #1
    cmp w0, w1
    b.ne .Lerr_false
    add x21, x21, #1
    subs x9, x9, #1
    b.ne 1b
    mov x0, #0
    ret
.Lerr_false:
    mov x0, #1
    ret

// parse_null
parse_null:
    mov x9, #4
    adrp x10, lit_null
    add x10, x10, :lo12:lit_null
1:
    cmp x21, x20
    b.hs .Lerr_null
    ldrb w0, [x19, x21]
    ldrb w1, [x10], #1
    cmp w0, w1
    b.ne .Lerr_null
    add x21, x21, #1
    subs x9, x9, #1
    b.ne 1b
    mov x0, #0
    ret
.Lerr_null:
    mov x0, #1
    ret
```
