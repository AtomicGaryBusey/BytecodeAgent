_start:
    lea r12, [rip + buffer]
    xor r13, r13
.read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    add rsi, r13
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz .read_done
    add r13, rax
    cmp r13, 65536
    jb .read_loop
.read_done:
    mov rbx, r13
    xor rcx, rcx
.find_bang:
    cmp rcx, rbx
    jae .no_bang
    cmp byte ptr [r12 + rcx], '!'
    je .found_bang
    inc rcx
    jmp .find_bang

    mov r13, rcx
    lea rax, [r12 + rcx + 1]
    mov qword ptr [rip + input_ptr], rax
    mov rax, rbx
    sub rax, rcx
    dec rax
    mov qword ptr [rip + input_len], rax
    jmp .after_find
.no_bang:
    mov r13, rbx
    lea rax, [r12 + rbx]
    mov qword ptr [rip + input_ptr], rax
    mov qword ptr [rip + input_len], 0
.after_find:
    lea r9, [rip + jump_table]
    xor r14, r14
    xor r10, r10
.bcheck_loop:
    cmp r14, r13
    jae .bcheck_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, '['
    je .bcheck_open
    cmp al, ']'
    je .bcheck_close
    inc r14
    jmp .bcheck_loop
.bcheck_open:
    push r14
    inc r10
    inc r14
    jmp .bcheck_loop
.bcheck_close:
    test r10, r10
    jz .bcheck_error
    dec r10
    pop rax
    mov ecx, r14d
    add ecx, 1
    mov dword ptr [r9 + rax*4], ecx
    mov ecx, eax
    add ecx, 1
    mov dword ptr [r9 + r14*4], ecx
    inc r14
    jmp .bcheck_loop
.bcheck_done:
    test r10, r10
    jnz .bcheck_error
    lea r8, [rip + tape]
    xor rbx, rbx
    xor r14, r14
    mov r15, qword ptr [rip + input_ptr]
    mov rbp, qword ptr [rip + input_len]
    jmp .loop

.bcheck_error:
    mov rax, 60
    mov rdi, 1
    syscall

.loop:
    cmp r14, r13
    jae .exit_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, '>'
    je .cmd_gt
    cmp al, '<'
    je .cmd_lt
    cmp al, '+'
    je .cmd_plus
    cmp al, '-'
    je .cmd_minus
    cmp al, '.'
    je .cmd_dot
    cmp al, ','
    je .cmd_comma
    cmp al, '['
    je .cmd_lb
    cmp al, ']'
    je .cmd_rb
    inc r14
    jmp .loop

.cmd_gt:
    inc rbx
    inc r14
    jmp .loop
.cmd_lt:
    dec rbx
    inc r14
    jmp .loop
.cmd_plus:
    inc byte ptr [r8 + rbx]
    inc r14
    jmp .loop
.cmd_minus:
    dec byte ptr [r8 + rbx]
    inc r14
    jmp .loop
.cmd_dot:

    mov rdi, 1
    lea rsi, [r8 + rbx]
    mov rdx, 1
    syscall
    inc r14
    jmp .loop
.cmd_comma:
    test rbp, rbp
    jz .comma_zero
    mov al, byte ptr [r15]
    mov byte ptr [r8 + rbx], al
    inc r15
    dec rbp
    jmp .comma_done
.comma_zero:

    mov byte ptr [r8 + rbx], al
.comma_done:
    inc r14
    jmp .loop
.cmd_lb:
    mov al, byte ptr [r8 + rbx]
    test al, al
    jz .lb_zero
    inc r14
    jmp .loop
.lb_zero:

.cmd_rb:
    mov al, byte ptr [r8 + rbx]
    test al, al
    jz .rb_zero
    mov r14d, dword ptr [r9 + r14*4]
    jmp .loop
.rb_zero:
    inc r14
    jmp .loop

.exit_ok:

    xor rdi, rdi
    syscall
