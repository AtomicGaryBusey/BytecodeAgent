.text
.globl _start
_start:
    lea r12, [rip + input_buf]
    lea r15, [rip + eval_stack]
    xor r14, r14
    xor r13, r13
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r14]
    mov rdx, 65536
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r13, r13
line_loop:
    cmp r13, r14
    jae done_all
    mov r9, r13
find_newline:
    cmp r9, r14
    jae line_found
    cmp byte ptr [r12 + r9], 10
    je line_found
    inc r9
    jmp find_newline
line_found:
    mov r8, r13
    xor ebx, ebx
token_loop:
skip_sep:
    cmp r8, r9
    jae line_end_tokens
    mov al, [r12 + r8]
    cmp al, ' '
    je skip_sep_inc
    cmp al, 9
    je skip_sep_inc
    jmp token_start
skip_sep_inc:
    inc r8
    jmp skip_sep
token_start:
    mov r10, r8
scan_token:
    inc r8
    cmp r8, r9
    jae token_end
    mov al, [r12 + r8]
    cmp al, ' '
    je token_end
    cmp al, 9
    je token_end
    jmp scan_token
token_end:
    mov rcx, r8
    sub rcx, r10
    cmp rcx, 1
    je single_char
    mov al, [r12 + r10]
    cmp al, '-'
    jne parse_positive
    cmp rcx, 19
    ja invalid_token
    lea rsi, [r12 + r10 + 1]
    lea rdi, [r12 + r8]
    xor edx, edx
parse_neg_loop:
    cmp rsi, rdi
    jae parse_neg_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    sub eax, '0'
    imul rdx, rdx, 10
    add rdx, rax
    inc rsi
    jmp parse_neg_loop
parse_neg_done:
    neg rdx
    mov rax, rdx
    jmp push_value
parse_positive:
    cmp rcx, 18
    ja invalid_token
    lea rsi, [r12 + r10]
    lea rdi, [r12 + r8]
    xor edx, edx
parse_pos_loop:
    cmp rsi, rdi
    jae parse_pos_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    sub eax, '0'
    imul rdx, rdx, 10
    add rdx, rax
    inc rsi
    jmp parse_pos_loop
parse_pos_done:
    mov rax, rdx
    jmp push_value
single_char:
    mov al, [r12 + r10]
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    movzx eax, byte ptr [r12 + r10]
    sub eax, '0'
    jmp push_value
push_value:
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp token_loop
op_plus:
    cmp rbx, 2
    jb invalid_token
    dec rbx
    mov rcx, qword ptr [r15 + rbx*8]
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    add rax, rcx
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp token_loop
op_minus:
    cmp rbx, 2
    jb invalid_token
    dec rbx
    mov rcx, qword ptr [r15 + rbx*8]
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    sub rax, rcx
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp token_loop
op_mul:
    cmp rbx, 2
    jb invalid_token
    dec rbx
    mov rcx, qword ptr [r15 + rbx*8]
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    imul rax, rcx
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp token_loop
op_div:
    cmp rbx, 2
    jb invalid_token
    dec rbx
    mov rcx, qword ptr [r15 + rbx*8]
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    test rcx, rcx
    jz invalid_token
    cmp rcx, -1
    jne do_div
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je invalid_token
do_div:
    cqo
    idiv rcx
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp token_loop
op_mod:
    cmp rbx, 2
    jb invalid_token
    dec rbx
    mov rcx, qword ptr [r15 + rbx*8]
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    test rcx, rcx
    jz invalid_token
    cmp rcx, -1
    jne do_mod
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je invalid_token
do_mod:
    cqo
    idiv rcx
    mov qword ptr [r15 + rbx*8], rdx
    inc rbx
    jmp token_loop
invalid_token:
    jmp error_line
line_end_tokens:
    cmp rbx, 1
    jne error_line
    dec rbx
    mov rax, qword ptr [r15 + rbx*8]
    lea rsi, [rip + out_buf + 31]
    mov byte ptr [rsi], 10
    test rax, rax
    jns print_positive
    mov r10, 1
    neg rax
    jmp print_convert
print_positive:
    xor r10, r10
print_convert:
    mov rcx, 10
convert_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_loop
    test r10, r10
    jz print_write
    dec rsi
    mov byte ptr [rsi], '-'
print_write:
    lea rdx, [rip + out_buf + 32]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp next_line
error_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 6
    syscall
    jmp next_line
next_line:
    cmp r9, r14
    jae set_end
    lea r13, [r9 + 1]
    jmp line_loop
set_end:
    mov r13, r14
    jmp line_loop
done_all:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
error_msg:
    .ascii "error\n"

.bss
input_buf:
    .zero 65536
    .align 8
eval_stack:
    .zero 8192
out_buf:
    .zero 32
