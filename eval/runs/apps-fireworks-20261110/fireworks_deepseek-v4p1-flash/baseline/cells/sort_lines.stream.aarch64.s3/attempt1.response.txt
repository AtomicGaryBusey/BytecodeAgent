    .globl _start

    .text
_start:
    // set up pointers
    adrp x19, buf
    add  x19, x19, :lo12:buf
    adrp x25, line_off
    add  x25, x25, :lo12:line_off
    adrp x26, line_len
    add  x26, x26, :lo12:line_len

    // read all of stdin
    mov  x20, #0
read_loop:
    mov  x0, #0
    add  x1, x19, x20
    movz x2, #1, lsl #16          // 65536
    sub  x2, x2, x20
    mov  x8, #63                  // read
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop
read_done:

    // parse into lines
    mov  x21, #0                  // number of lines
    mov  x22, #0                  // current offset
parse_loop:
    cmp  x22, x20
    b.ge parse_done
    mov  x23, x22                 // line start
parse_scan:
    cmp  x22, x20
    b.ge parse_eol_final
    ldrb w24, [x19, x22]
    cmp  w24, #10
    b.eq parse_eol
    add  x22, x22, #1
    b    parse_scan
parse_eol:
    sub  x24, x22, x23
    bl   record_line
    add  x22, x22, #1
    b    parse_loop
parse_eol_final:
    sub  x24, x22, x23
    bl   record_line
parse_done:

    // insertion sort
    mov  x27, #1
sort_outer:
    cmp  x27, x21
    b.ge sort_done
    mov  x28, x27
sort_inner:
    cmp  x28, #0
    b.le sort_next
    sub  x0, x28, #1
    mov  x1, x28
    bl   cmp_lines
    cmp  w0, #0
    b.le sort_next
    // swap entries at x28-1 and x28
    sub  x9, x28, #1
    ldr  w10, [x25, x9, lsl #2]
    ldr  w11, [x25, x28, lsl #2]
    str  w11, [x25, x9, lsl #2]
    str  w10, [x25, x28, lsl #2]
    ldr  w10, [x26, x9, lsl #2]
    ldr  w11, [x26, x28, lsl #2]
    str  w11, [x26, x9, lsl #2]
    str  w10, [x26, x28, lsl #2]
    sub  x28, x28, #1
    b    sort_inner
sort_next:
    add  x27, x27, #1
    b    sort_outer
sort_done:

    // build output
    adrp x24, outbuf
    add  x24, x24, :lo12:outbuf
    mov  x27, #0
    mov  x29, #0                  // output length
out_loop:
    cmp  x27, x21
    b.ge out_done
    ldr  w10, [x25, x27, lsl #2]
    ldr  w11, [x26, x27, lsl #2]
    add  x12, x19, x10            // source
    add  x13, x24, x29            // destination
    mov  x14, #0
out_copy:
    cmp  x14, x11
    b.ge out_copy_done
    ldrb w15, [x12, x14]
    strb w15, [x13, x14]
    add  x14, x14, #1
    b    out_copy
out_copy_done:
    add  x29, x29, x11
    add  x13, x24, x29
    mov  w15, #10
    strb w15, [x13]
    add  x29, x29, #1
    add  x27, x27, #1
    b    out_loop
out_done:
    mov  x0, #1
    mov  x1, x24
    mov  x2, x29
    mov  x8, #64                  // write
    svc  #0
    mov  x0, #0
    mov  x8, #94                  // exit_group
    svc  #0

// record a line: w23 = start offset, w24 = length
record_line:
    cmp  x21, #1000
    b.ge record_skip
    str  w23, [x25, x21, lsl #2]
    str  w24, [x26, x21, lsl #2]
    add  x21, x21, #1
record_skip:
    ret

// compare lines x0 and x1, return w0 < 0, 0, > 0
cmp_lines:
    ldr  w3, [x25, x0, lsl #2]
    ldr  w4, [x25, x1, lsl #2]
    ldr  w6, [x26, x0, lsl #2]
    ldr  w7, [x26, x1, lsl #2]
    add  x3, x19, x3
    add  x4, x19, x4
    cmp  w6, w7
    csel w8, w6, w7, lo           // min length
    mov  x9, #0
cmp_loop:
    cmp  w9, w8
    b.ge cmp_prefix
    ldrb w10, [x3, x9]
    ldrb w11, [x4, x9]
    cmp  w10, w11
    b.ne cmp_diff
    add  w9, w9, #1
    b    cmp_loop
cmp_diff:
    sub  w0, w10, w11
    ret
cmp_prefix:
    sub  w0, w6, w7
    ret

    .bss
    .balign 16
buf:
    .zero 65536
    .balign 16
line_off:
    .zero 4000
    .balign 16
line_len:
    .zero 4000
    .balign 16
outbuf:
    .zero 70000
