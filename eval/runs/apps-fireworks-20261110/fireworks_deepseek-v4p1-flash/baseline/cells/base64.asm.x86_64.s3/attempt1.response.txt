.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
    .zero 49152
output_buf:
    .zero 131072

.text
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    test rdx, rdx
    jz done_reading
    syscall
    test rax, rax
    jle done_reading
    add r12, rax
    jmp read_loop
done_reading:
    lea r8, [rip + input_buf]
    lea r9, [r8 + r12]
    lea r10, [rip + output_buf]
    xor r11d, r11d
    lea r12, [rip + b64_table]

encode_loop:
    mov rax, r9
    sub rax, r8
    cmp rax, 3
    jb tail
    movzx esi, byte ptr [r8]
    movzx edi, byte ptr [r8 + 1]
    movzx ebx, byte ptr [r8 + 2]
    add r8, 3

    mov eax, esi
    shr eax, 2
    call emit_b64

    mov eax, esi
    and eax, 3
    shl eax, 4
    mov edx, edi
    shr edx, 4
    or eax, edx
    call emit_b64

    mov eax, edi
    and eax, 15
    shl eax, 2
    mov edx, ebx
    shr edx, 6
    or eax, edx
    call emit_b64

    mov eax, ebx
    and eax, 63
    call emit_b64

    jmp encode_loop

tail:
    cmp eax, 1
    je tail1
    cmp eax, 2
    je tail2
    jmp finalize

tail1:
    movzx esi, byte ptr [r8]
    mov eax, esi
    shr eax, 2
    call emit_b64
    mov eax, esi
    and eax, 3
    shl eax, 4
    call emit_b64
    mov eax, 61
    call putc
    mov eax, 61
    call putc
    jmp finalize

tail2:
    movzx esi, byte ptr [r8]
    movzx edi, byte ptr [r8 + 1]
    mov eax, esi
    shr eax, 2
    call emit_b64
    mov eax, esi
    and eax, 3
    shl eax, 4
    mov edx, edi
    shr edx, 4
    or eax, edx
    call emit_b64
    mov eax, edi
    and eax, 15
    shl eax, 2
    call emit_b64
    mov eax, 61
    call putc
    jmp finalize

finalize:
    test r11d, r11d
    jz no_final_nl
    mov byte ptr [r10], 10
    inc r10
no_final_nl:
    lea rsi, [rip + output_buf]
    mov rdx, r10
    sub rdx, rsi
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

emit_b64:
    movzx eax, byte ptr [r12 + rax]
putc:
    mov byte ptr [r10], al
    inc r10
    inc r11d
    cmp r11d, 76
    jne 1f
    mov byte ptr [r10], 10
    inc r10
    xor r11d, r11d
1:
    ret
