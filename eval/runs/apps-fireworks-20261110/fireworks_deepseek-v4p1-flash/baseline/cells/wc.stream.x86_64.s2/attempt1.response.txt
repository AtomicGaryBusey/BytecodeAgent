.section .bss
.balign 16
buffer:
    .zero 65536
.balign 16
outbuf:
    .zero 64

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz done_reading
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax
process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne .not_newline
    inc r12
.not_newline:
    cmp al, 32
    je .is_space
    cmp al, 9
    je .is_space
    cmp al, 10
    je .is_space
    cmp al, 11
    je .is_space
    cmp al, 12
    je .is_space
    cmp al, 13
    je .is_space
    test r15, r15
    jnz process_loop
    inc r13
    mov r15, 1
    jmp process_loop
.is_space:
    xor r15d, r15d
    jmp process_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_u64
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_u64
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_u64
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_u64:
    xor ecx, ecx
    mov r9, 10
    test rax, rax
    jnz .convert_loop
    mov byte ptr [rdi], 48
    inc rdi
    ret
.convert_loop:
    xor edx, edx
    div r9
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz .convert_loop
.copy_digits:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz .copy_digits
    ret
