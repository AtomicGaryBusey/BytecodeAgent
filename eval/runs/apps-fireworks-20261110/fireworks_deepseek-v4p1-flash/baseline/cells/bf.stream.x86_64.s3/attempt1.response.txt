.globl _start
.text
_start:
    # read all stdin
    xor ebx, ebx          # offset = 0
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop
read_done:
    # rbx = total bytes
    # find first '!'
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [inbuf + rcx], 33
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov r9, rbx           # L = total
    lea r14, [rip + inbuf]
    add r14, rbx          # input ptr = end
    mov r15, r14          # input end = end
    jmp after_split
found_bang:
    mov r9, rcx           # L = bang index
    lea r14, [rip + inbuf]
    add r14, rcx
    inc r14               # input start = bang+1
    lea r15, [rip + inbuf]
    add r15, rbx          # input end = total
after_split:
    lea rbx, [rip + inbuf]  # program base
    lea rbp, [rip + tape]
    lea r8, [rip + jump]
    # bracket matching
    xor r10d, r10d        # stack count
    xor r11d, r11d        # i
scan_loop:
    cmp r11, r9
    jae scan_done
    movzx eax, byte ptr [rbx + r11]
    cmp al, 91            # '['
    jne .Lscan_not_open
    mov dword ptr [bstack + r10*4], r11d
    inc r10
    jmp .Lscan_next
.Lscan_not_open:
    cmp al, 93            # ']'
    jne .Lscan_next
    test r10, r10
    jz error
    dec r10
    mov eax, dword ptr [bstack + r10*4]
    mov dword ptr [r8 + r11*4], eax
    mov dword ptr [r8 + rax*4], r11d
.Lscan_next:
    inc r11
    jmp scan_loop
scan_done:
    test r10, r10
    jnz error
    # interpret
    xor r13d, r13d        # pc = 0
    xor r12d, r12d        # data ptr = 0
    xor r10d, r10d        # outlen = 0
interp_loop:
    cmp r13, r9
    jae interp_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 62            # '>'
    je .Lgt
    cmp al, 60            # '<'
    je .Llt
    cmp al, 43            # '+'
    je .Lplus
    cmp al, 45            # '-'
    je .Lminus
    cmp al, 46            # '.'
    je .Ldot
    cmp al, 44            # ','
    je .Lcomma
    cmp al, 91            # '['
    je .Lopen
    cmp al, 93            # ']'
    je .Lclose
    jmp .Lnext
.Lgt:
    inc r12
    jmp .Lnext
.Llt:
    dec r12
    jmp .Lnext
.Lplus:
    inc byte ptr [rbp + r12]
    jmp .Lnext
.Lminus:
    dec byte ptr [rbp + r12]
    jmp .Lnext
.Ldot:
    mov al, byte ptr [rbp + r12]
    mov byte ptr [outbuf + r10], al
    inc r10
    cmp r10, 65536
    jb .Lnext
    call flush_out
    jmp .Lnext
.Lcomma:
    cmp r14, r15
    jae .Lcomma_zero
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbp + r12], al
    jmp .Lnext
.Lcomma_zero:
    mov byte ptr [rbp + r12], 0
    jmp .Lnext
.Lopen:
    cmp byte ptr [rbp + r12], 0
    jne .Lnext
    mov r13d, dword ptr [r8 + r13*4]
    jmp .Lnext
.Lclose:
    cmp byte ptr [rbp + r12], 0
    je .Lnext
    mov r13d, dword ptr [r8 + r13*4]
    jmp .Lnext
.Lnext:
    inc r13
    jmp interp_loop
interp_done:
    call flush_out
    xor edi, edi
    mov eax, 60
    syscall
error:
    mov edi, 1
    mov eax, 60
    syscall

flush_out:
    test r10, r10
    jz .Lflush_ret
    mov rdx, r10
    lea rsi, [rip + outbuf]
    mov edi, 1
    mov eax, 1
    syscall
    xor r10d, r10d
.Lflush_ret:
    ret

.bss
.align 16
inbuf:   .zero 65536
tape:    .zero 30000
jump:    .zero 262144
bstack:  .zero 262144
outbuf:  .zero 65536
