```asm
    .globl _start
    .text
_start:
    // x20 = inbuf, x21 = outbuf
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf

    // Read all of standard input into inbuf.
    mov x19, #0                 // total bytes read
read_loop:
    mov x0, #0                  // fd = 0
    mov x1, x20
    add x1, x1, x19             // buf + offset
    mov x2, #65536
    sub x2, x2, x19             // remaining space
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le read_done              // 0 or error -> stop reading
    add x19, x19, x0
    b read_loop
read_done:

    // Process input and build output in outbuf.
    mov x22, #0                 // output length
    mov x23, #0                 // current input index

main_loop:
    cmp x23, x19
    b.ge done_processing        // no more input

    mov x24, x23
find_nl:
    cmp x24, x19
    b.ge no_nl_found            // no newline before end of input
    ldrb w25, [x20, x24]
    cmp w25, #0x0A
    b.eq nl_found
    add x24, x24, #1
    b find_nl

nl_found:
    // Line is [x23, x24).  Reverse it into outbuf.
    sub x26, x24, x23           // line length
    mov x27, #0                 // i = 0
reverse_loop:
    cmp x27, x26
    b.ge reverse_done
    sub x28, x24, #1
    sub x28, x28, x27           // index of byte to copy
    ldrb w29, [x20, x28]
    add x30, x21, x22
    add x30, x30, x27           // outbuf + outlen + i
    strb w29, [x30]
    add x27, x27, #1
    b reverse_loop
reverse_done:
    add x22, x22, x26           // advance output length
    add x30, x21, x22
    mov w29, #0x0A
    strb w29, [x30]             // append newline
    add x22, x22, #1
    add x23, x24, #1            // skip the newline
    b main_loop

no_nl_found:
    // Final line is [x23, x19).  Reverse it into outbuf.
    sub x26, x19, x23           // line length
    mov x27, #0
reverse_final_loop:
    cmp x27, x26
    b.ge reverse_final_done
    sub x28, x19, #1
    sub x28, x28, x27
    ldrb w29, [x20, x28]
    add x30, x21, x22
    add x30, x30, x27
    strb w29, [x30]
    add x27, x27, #1
    b reverse_final_loop
reverse_final_done:
    add x22, x22, x26
    add x30, x21, x22
    mov w29, #0x0A
    strb w29, [x30]             // append newline
    add x22, x22, #1
    mov x23, x19                // mark all input consumed

done_processing:
    // Write the entire output buffer to stdout.
    mov x1, x21
    mov x2, x22
write_loop:
    mov x0, #1                  // fd = 1
    mov x8, #64                 // write
    svc #0
    cmp x0, #0
    b.le write_done             // error or nothing written
    sub x2, x2, x0              // remaining bytes
    add x1, x1, x0              // advance buffer pointer
    cbnz x2, write_loop
write_done:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

    .section .bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
```
