    .text
    .globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, x20
    mov x21, #0
read_loop:
    mov x0, #0
    mov x1, x19
    mov x2, #65536
    sub x2, x2, x21
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    add x21, x21, x0
    b read_loop
read_done:
    mov x22, x20
    add x23, x20, x21
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
process_loop:
    cmp x22, x23
    b.hs process_done
    mov x25, x22
find_newline:
    cmp x25, x23
    b.hs found_end
    ldrb w26, [x25]
    cmp w26, #0x0A
    b.eq found_newline
    add x25, x25, #1
    b find_newline
found_newline:
    sub x26, x25, #1
    b reverse_loop
found_end:
    sub x26, x23, #1
reverse_loop:
    cmp x26, x22
    b.lo reverse_done
    ldrb w27, [x26]
    strb w27, [x24]
    add x24, x24, #1
    sub x26, x26, #1
    b reverse_loop
reverse_done:

    strb w27, [x24]
    add x24, x24, #1
    cmp x25, x23
    b.hs process_done
    add x22, x25, #1
    b process_loop
process_done:

    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x24, x1
    cbz x2, exit
    mov x8, #64
    svc #0

    mov x8, #93
    svc #0
