.globl _start

.section .text
_start:
    xor r12, r12
    lea rsi, [rip + input_buf]
read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 131072
    sub rdx, r12
    jz read_done
    mov rax, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea rbx, [rip + input_buf]
    lea r15, [rbx + r12]
    mov r14, rbx
    mov r13, rbx
    xor r12, r12
parse_loop:
    cmp r13, r15
    jae parse_done
    mov al, byte ptr [r13]
    cmp al, 0x0A
    jne not_newline
    mov rax, r13
    sub rax, r14
    mov rcx, r12
    shl rcx, 4
    lea rdi, [rip + line_array]
    add rdi, rcx
    mov [rdi], r14
    mov [rdi+8], rax
    inc r12
    lea r14, [r13+1]
not_newline:
    inc r13
    jmp parse_loop
parse_done:
    cmp r14, r15
    jae no_final
    mov rax, r15
    sub rax, r14
    mov rcx, r12
    shl rcx, 4
    lea rdi, [rip + line_array]
    add rdi, rcx
    mov [rdi], r14
    mov [rdi+8], rax
    inc r12
no_final:

    cmp r12, 2
    jb sort_done
    mov r13, 1
outer_loop:
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + line_array]
    add rdi, rax
    mov r15, [rdi]
    mov rbx, [rdi+8]
    mov r14, r13
    dec r14
inner_loop:
    cmp r14, 0
    jl inner_done
    mov rax, r14
    shl rax, 4
    lea rsi, [rip + line_array]
    add rsi, rax
    mov r10, [rsi]
    mov r11, [rsi+8]
    mov rdi, r10
    mov rsi, r11
    mov rdx, r15
    mov rcx, rbx
    call compare
    test eax, eax
    jle inner_done
    mov rax, r14
    shl rax, 4
    lea rdi, [rip + line_array]
    add rdi, rax
    lea rsi, [rdi+16]
    mov rax, [rdi]
    mov [rsi], rax
    mov rax, [rdi+8]
    mov [rsi+8], rax
    dec r14
    jmp inner_loop
inner_done:
    inc r14
    mov rax, r14
    shl rax, 4
    lea rdi, [rip + line_array]
    add rdi, rax
    mov [rdi], r15
    mov [rdi+8], rbx
    inc r13
    cmp r13, r12
    jb outer_loop
sort_done:

    xor r13, r13
output_loop:
    cmp r13, r12
    jae exit
    mov rax, r13
    shl rax, 4
    lea rbx, [rip + line_array]
    add rbx, rax
    mov r14, [rbx]
    mov r15, [rbx+8]
    mov rdi, 1
    mov rsi, r14
    mov rdx, r15
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r13
    jmp output_loop
exit:
    mov rax, 60
    xor rdi, rdi
    syscall

compare:
    mov r8, rsi
    mov r9, rcx
    cmp rsi, rcx
    cmova rsi, rcx
    mov rcx, rsi
    mov rsi, rdi
    mov rdi, rdx
    test rcx, rcx
    jz len_cmp
    cld
    repe cmpsb
    jne diff
len_cmp:
    cmp r8, r9
    jb less
    ja greater
    xor eax, eax
    ret
diff:
    jb less
greater:
    mov eax, 1
    ret
less:
    mov eax, -1
    ret

.section .rodata
newline:
    .byte 0x0A

.section .bss
.align 16
input_buf:
    .zero 131072
line_array:
    .zero 16000
