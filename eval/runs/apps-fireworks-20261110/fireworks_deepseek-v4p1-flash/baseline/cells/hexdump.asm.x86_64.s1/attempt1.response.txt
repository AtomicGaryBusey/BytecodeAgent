.globl _start

.section .text
_start:
    # read input
    xor r8, r8                 # r8 = total bytes read
read_loop:
    cmp r8, 65536
    jae read_done
    mov rax, 0                 # syscall read
    xor rdi, rdi               # fd 0
    lea rsi, [rip + buf]
    add rsi, r8                # buf + len
    mov rdx, 65536
    sub rdx, r8                # count = 65536 - len
    syscall
    test rax, rax
    jle read_done
    add r8, rax
    jmp read_loop

read_done:
    # generate output
    lea rdi, [rip + outbuf]    # rdi = output pointer
    lea rsi, [rip + buf]       # rsi = input buffer
    xor r9, r9                 # r9 = offset
    lea r10, [rip + hex_table] # r10 = hex table

row_loop:
    cmp r9, r8
    jae done_rows
    # compute n = min(16, r8 - r9)
    mov rbx, r8
    sub rbx, r9
    cmp rbx, 16
    jbe n_ok
    mov rbx, 16
n_ok:
    # write offset (r9) as 8 hex digits
    mov rdx, r9
    mov rax, rdx
    shr rax, 28
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 24
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 20
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 16
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 12
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 8
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 4
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    # two spaces
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    add rdi, 2

    # byte loop i = 0..15
    xor rcx, rcx
byte_loop:
    cmp rcx, 16
    jae byte_loop_end
    cmp rcx, rbx
    jae write_spaces
    # i < n: output hex
    mov rdx, r9
    add rdx, rcx
    movzx eax, byte ptr [rsi + rdx]
    mov r11, rax
    shr eax, 4
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov eax, r11d
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    jmp byte_loop_next
write_spaces:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
byte_loop_next:
    cmp rcx, 7
    jne no_extra_space
    mov byte ptr [rdi], ' '
    inc rdi
no_extra_space:
    inc rcx
    jmp byte_loop
byte_loop_end:
    # one space, '|'
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], '|'
    add rdi, 2
    # ascii loop
    xor rcx, rcx
ascii_loop:
    cmp rcx, rbx
    jae ascii_loop_end
    mov rdx, r9
    add rdx, rcx
    movzx eax, byte ptr [rsi + rdx]
    cmp al, 0x20
    jb write_dot
    cmp al, 0x7e
    ja write_dot
    mov [rdi], al
    jmp ascii_next
write_dot:
    mov byte ptr [rdi], '.'
ascii_next:
    inc rdi
    inc rcx
    jmp ascii_loop
ascii_loop_end:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi+1], 10
    add rdi, 2
    add r9, 16
    jmp row_loop

done_rows:
    # final offset
    mov rdx, r8
    mov rax, rdx
    shr rax, 28
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 24
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 20
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 16
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 12
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 8
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    shr rax, 4
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov rax, rdx
    and eax, 0xF
    mov al, [r10 + rax]
    mov [rdi], al
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    # write output
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    # exit(0)
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 400000
