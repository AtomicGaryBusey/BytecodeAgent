.globl _start

.equ BUF_SIZE, 65536
.equ OUT_SIZE, 65537

.text
_start:
    # Read all of stdin into inbuf
    xor r12d, r12d                  # r12 = total bytes read
read_loop:
    mov eax, 0                      # read
    xor edi, edi                    # fd = 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, BUF_SIZE
    sub rdx, r12
    jz read_done                    # buffer full
    syscall
    test rax, rax
    jle read_done                   # EOF (or error)
    add r12, rax
    jmp read_loop

read_done:
    # r14 = start of current line, r15 = output cursor, rbx = end of input
    lea r14, [rip + inbuf]
    lea r15, [rip + outbuf]
    lea rbx, [rip + inbuf]
    add rbx, r12

scan:
    cmp r14, rbx
    jae finish
    mov rcx, r14
find_nl:
    cmp rcx, rbx
    jae no_nl
    cmp byte ptr [rcx], 10
    je found_nl
    inc rcx
    jmp find_nl

found_nl:
    # reverse bytes in [r14, rcx) into outbuf
    mov rdx, rcx
    dec rdx
rev1:
    cmp rdx, r14
    jb rev1_done
    mov al, byte ptr [rdx]
    mov byte ptr [r15], al
    inc r15
    dec rdx
    jmp rev1
rev1_done:
    mov byte ptr [r15], 10
    inc r15
    lea r14, [rcx + 1]
    jmp scan

no_nl:
    # final line without trailing newline: reverse [r14, rbx)
    mov rdx, rbx
    dec rdx
rev2:
    cmp rdx, r14
    jb rev2_done
    mov al, byte ptr [rdx]
    mov byte ptr [r15], al
    inc r15
    dec rdx
    jmp rev2
rev2_done:
    mov byte ptr [r15], 10
    inc r15

finish:
    # write outbuf, length = r15 - outbuf
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    jz exit_ok
    mov edi, 1
    mov eax, 1
    syscall

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.align 16
inbuf:
    .zero BUF_SIZE
outbuf:
    .zero OUT_SIZE
