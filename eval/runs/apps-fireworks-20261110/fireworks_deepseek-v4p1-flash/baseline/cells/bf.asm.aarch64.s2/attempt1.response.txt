```asm
.globl _start

.text
_start:
    // Read all stdin into inbuf
    adrp x0, inbuf
    add x0, x0, :lo12:inbuf
    mov x19, x0                 // current write pointer
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    movz x2, #1, lsl #16
    add x1, x1, x2              // end of buffer
    mov x20, x1                 // end pointer

read_loop:
    cmp x19, x20
    b.ge read_done
    mov x0, #0                  // fd 0
    mov x1, x19                 // buf
    subs x2, x20, x19           // count
    b.le read_done
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x23, x19                // x23 = end of input data
    adrp x21, inbuf
    add x21, x21, :lo12:inbuf
    mov x22, x21                // scan pointer
scan_bang:
    cmp x22, x23
    b.ge no_bang
    ldrb w0, [x22]
    cmp w0, #0x21
    b.eq found_bang
    add x22, x22, #1
    b scan_bang

no_bang:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf   // program_start
    mov x19, x23                // program_end = end of input
    mov x24, x23                // input_start = end
    mov x25, x23                // input_end = end
    b bracket_check

found_bang:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf   // program_start
    mov x19, x22                // program_end = address of '!'
    add x24, x22, #1            // input_start = after '!'
    mov x25, x23                // input_end = end of input
    b bracket_check

bracket_check:
    mov x9, x20                 // program_start
    mov x10, x19                // program_end
    mov x13, #0                 // offset i = 0
    mov x12, #0                 // stack top index
    adrp x11, jumptab
    add x11, x11, :lo12:jumptab
    adrp x14, bstack
    add x14, x14, :lo12:bstack

check_loop:
    cmp x9, x10
    b.ge check_done
    ldrb w0, [x9], #1
    cmp w0, #0x5B               // '['
    b.eq check_push
    cmp w0, #0x5D               // ']'
    b.eq check_pop
    add x13, x13, #1
    b check_loop

check_push:
    str w13, [x14, x12, lsl #2]
    add x12, x12, #1
    add x13, x13, #1
    b check_loop

check_pop:
    cbz x12, unbalanced
    sub x12, x12, #1
    ldr w15, [x14, x12, lsl #2]
    strh w13, [x11, x15, lsl #1]
    strh w15, [x11, x13, lsl #1]
    add x13, x13, #1
    b check_loop

check_done:
    cbnz x12, unbalanced
    b run_program

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0

run_program:
    adrp x22, tape
    add x22, x22, :lo12:tape
    mov x23, x22                // tape pointer = base
    mov x26, x24                // input pointer = input_start
    adrp x27, outbuf
    add x27, x27, :lo12:outbuf
    mov x28, x27                // outptr = outbuf base
    mov x21, x20                // pc = program_start

exec_loop:
    cmp x21, x19
    b.ge exec_done
    ldrb w0, [x21], #1
    cmp w0, #0x3E               // '>'
    b.eq cmd_gt
    cmp w0, #0x3C               // '<'
    b.eq cmd_lt
    cmp w0, #0x2B               // '+'
    b.eq cmd_plus
    cmp w0, #0x2D               // '-'
    b.eq cmd_minus
    cmp w0, #0x2E               // '.'
    b.eq cmd_dot
    cmp w0, #0x2C               // ','
    b.eq cmd_comma
    cmp w0, #0x5B               // '['
    b.eq cmd_lb
    cmp w0, #0x5D               // ']'
    b.eq cmd_rb
    b exec_loop

cmd_gt:
    add x23, x23, #1
    b exec_loop
cmd_lt:
    sub x23, x23, #1
    b exec_loop
cmd_plus:
    ldrb w1, [x23]
    add w1, w1, #1
    strb w1, [x23]
    b exec_loop
cmd_minus:
    ldrb w1, [x23]
    sub w1, w1, #1
    strb w1, [x23]
    b exec_loop
cmd_dot:
    ldrb w1, [x23]
    strb w1, [x28], #1
    b exec_loop
cmd_comma:
    cmp x26, x25
    b.ge comma_zero
    ldrb w1, [x26], #1
    strb w1, [x23]
    b exec_loop
comma_zero:
    mov w1, #0
    strb w1, [x23]
    b exec_loop

cmd_lb:
    ldrb w1, [x23]
    cbnz w1, exec_loop
    sub x2, x21, x20
    sub x2, x2, #1
    adrp x3, jumptab
    add x3, x3, :lo12:jumptab
    ldrh w4, [x3, x2, lsl #1]
    add x21, x20, x4
    add x21, x21, #1
    b exec_loop

cmd_rb:
    ldrb w1, [x23]
    cbz w1, exec_loop
    sub x2, x21, x20
    sub x2, x2, #1
    adrp x3, jumptab
    add x3, x3, :lo12:jumptab
    ldrh w4, [x3, x2, lsl #1]
    add x21, x20, x4
    add x21, x21, #1
    b exec_loop

exec_done:
    subs x2, x28, x27
    b.le exit_ok
    mov x0, #1                  // fd 1
    mov x1, x27                 // buf
    mov x8, #64                 // write
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.align 4
inbuf:      .zero 65536
outbuf:     .zero 65536
tape:       .zero 30000
jumptab:    .zero 131072
bstack:     .zero 4096
```
