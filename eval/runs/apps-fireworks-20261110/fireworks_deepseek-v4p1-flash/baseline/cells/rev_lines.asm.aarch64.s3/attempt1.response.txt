    .globl _start
    .text
_start:
    // Read all input into inbuf
    mov x19, #0                 // total bytes read
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf

read_loop:
    mov x0, #0                  // fd = 0
    add x1, x22, x19            // buf = inbuf + total
    mov x2, #65536
    sub x2, x2, x19             // count = 65536 - total
    cbz x2, read_done
    mov x8, #63                 // read
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x20, #0                 // i = 0
    mov x21, #0                 // out offset = 0

process_loop:
    cmp x20, x19
    b.hs process_done

    mov x25, x20                // s = i

find_nl:
    cmp x20, x19
    b.hs found_end_of_input
    ldrb w0, [x22, x20]
    cmp w0, #10
    b.eq found_newline
    add x20, x20, #1
    b find_nl

found_newline:
    mov x24, x20                // end = i
    add x20, x20, #1            // i++
    b reverse_line

found_end_of_input:
    mov x24, x19                // end = total
    mov x20, x19                // i = total

reverse_line:
    sub x1, x24, #1             // k = end - 1
rev_loop:
    cmp x1, x25
    b.lt rev_done
    ldrb w0, [x22, x1]
    strb w0, [x23, x21]
    add x21, x21, #1
    sub x1, x1, #1
    b rev_loop

rev_done:
    mov w0, #10
    strb w0, [x23, x21]
    add x21, x21, #1
    b process_loop

process_done:
    mov x20, #0                 // write offset
write_loop:
    cbz x21, exit
    mov x0, #1                  // fd = 1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, x20             // buf = outbuf + write offset
    mov x2, x21                 // count = remaining
    mov x8, #64                 // write
    svc #0
    add x20, x20, x0
    sub x21, x21, x0
    b write_loop

exit:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

    .section .bss
    .align 4
inbuf:
    .zero 65536
outbuf:
    .zero 65552
