// rpn: evaluate one reverse Polish notation expression per input line.
    .globl _start

    .text
_start:
    // Read all of stdin into inbuf.
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    add     x21, x19, x20
    mov     x22, x19
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf

line_loop:
    cmp     x22, x21
    b.hs    process_done
    mov     x24, x22
find_nl:
    cmp     x24, x21
    b.hs    fl_end
    ldrb    w0, [x24]
    cmp     w0, #10
    b.eq    fl_nl
    add     x24, x24, #1
    b       find_nl
fl_nl:
    mov     x25, x24
    add     x26, x24, #1
    b       do_line
fl_end:
    mov     x25, x24
    mov     x26, x21

do_line:
    adrp    x29, vstack
    add     x29, x29, :lo12:vstack
    mov     x28, x29
    mov     x9, x22
tok_loop:
    cmp     x9, x25
    b.hs    tok_done
    ldrb    w0, [x9]
    cmp     w0, #32
    b.eq    skip_sep
    cmp     w0, #9
    b.eq    skip_sep
    b       tok_start
skip_sep:
    add     x9, x9, #1
    b       tok_loop
tok_start:
    mov     x10, x9
tok_scan:
    add     x9, x9, #1
    cmp     x9, x25
    b.hs    tok_end
    ldrb    w0, [x9]
    cmp     w0, #32
    b.eq    tok_end
    cmp     w0, #9
    b.eq    tok_end
    b       tok_scan
tok_end:
    mov     x11, x9
    sub     x12, x11, x10
    cmp     x12, #1
    b.ne    try_number
    ldrb    w0, [x10]
    cmp     w0, #43
    b.eq    op_add
    cmp     w0, #45
    b.eq    op_sub
    cmp     w0, #42
    b.eq    op_mul
    cmp     w0, #47
    b.eq    op_div
    cmp     w0, #37
    b.eq    op_mod
    b       try_number

op_add:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #16
    b.lo    error_out
    ldr     x1, [x28, #-8]
    ldr     x2, [x28, #-16]
    add     x2, x2, x1
    sub     x28, x28, #8
    str     x2, [x28, #-8]
    b       tok_loop

op_sub:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #16
    b.lo    error_out
    ldr     x1, [x28, #-8]
    ldr     x2, [x28, #-16]
    sub     x2, x2, x1
    sub     x28, x28, #8
    str     x2, [x28, #-8]
    b       tok_loop

op_mul:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #16
    b.lo    error_out
    ldr     x1, [x28, #-8]
    ldr     x2, [x28, #-16]
    mul     x2, x2, x1
    sub     x28, x28, #8
    str     x2, [x28, #-8]
    b       tok_loop

op_div:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #16
    b.lo    error_out
    ldr     x1, [x28, #-8]
    ldr     x2, [x28, #-16]
    cbz     x1, error_out
    mov     x3, #1
    lsl     x3, x3, #63
    cmp     x2, x3
    b.ne    div_go
    cmn     x1, #1
    b.eq    error_out
div_go:
    sdiv    x4, x2, x1
    sub     x28, x28, #8
    str     x4, [x28, #-8]
    b       tok_loop

op_mod:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #16
    b.lo    error_out
    ldr     x1, [x28, #-8]
    ldr     x2, [x28, #-16]
    cbz     x1, error_out
    mov     x3, #1
    lsl     x3, x3, #63
    cmp     x2, x3
    b.ne    mod_go
    cmn     x1, #1
    b.eq    error_out
mod_go:
    sdiv    x4, x2, x1
    msub    x4, x4, x1, x2
    sub     x28, x28, #8
    str     x4, [x28, #-8]
    b       tok_loop

try_number:
    mov     x13, x10
    mov     x14, #0
    ldrb    w0, [x13]
    cmp     w0, #45
    b.ne    tn_digits
    mov     x14, #1
    add     x13, x13, #1
tn_digits:
    sub     x15, x11, x13
    cmp     x15, #1
    b.lo    error_out
    cmp     x15, #18
    b.hi    error_out
    mov     x0, #0
tn_loop:
    cmp     x13, x11
    b.hs    tn_done
    ldrb    w1, [x13], #1
    sub     w1, w1, #48
    cmp     w1, #9
    b.hi    error_out
    mov     x2, #10
    mul     x0, x0, x2
    add     x0, x0, x1
    b       tn_loop
tn_done:
    cbz     x14, tn_push
    neg     x0, x0
tn_push:
    str     x0, [x28], #8
    b       tok_loop

tok_done:
    mov     x0, x28
    sub     x0, x0, x29
    cmp     x0, #8
    b.ne    error_out
    ldr     x0, [x28, #-8]
    bl      print_i64
    b       line_next

error_out:
    bl      print_error

line_next:
    mov     x22, x26
    b       line_loop

process_done:
    adrp    x19, outbuf
    add     x19, x19, :lo12:outbuf
    mov     x21, x23
    sub     x21, x21, x19
    mov     x20, x19
wloop:
    cbz     x21, exit_ok
    mov     x0, #1
    mov     x1, x20
    mov     x2, x21
    mov     x8, #64
    svc     #0
    add     x20, x20, x0
    sub     x21, x21, x0
    b       wloop
exit_ok:
    mov     x0, #0
    mov     x8, #93
    svc     #0

print_i64:
    mov     x1, x23
    cmp     x0, #0
    b.ge    pi_pos
    mov     w2, #45
    strb    w2, [x1], #1
    neg     x0, x0
pi_pos:
    sub     sp, sp, #32
    mov     x2, sp
    add     x2, x2, #24
    mov     x3, x2
    mov     x4, #10
pi_loop:
    udiv    x5, x0, x4
    msub    x6, x5, x4, x0
    add     w6, w6, #48
    sub     x3, x3, #1
    strb    w6, [x3]
    mov     x0, x5
    cbnz    x0, pi_loop
pi_copy:
    cmp     x3, x2
    b.hs    pi_done
    ldrb    w6, [x3], #1
    strb    w6, [x1], #1
    b       pi_copy
pi_done:
    mov     w6, #10
    strb    w6, [x1], #1
    mov     x23, x1
    add     sp, sp, #32
    ret

print_error:
    mov     w0, #101
    strb    w0, [x23], #1
    mov     w0, #114
    strb    w0, [x23], #1
    strb    w0, [x23], #1
    mov     w0, #111
    strb    w0, [x23], #1
    mov     w0, #114
    strb    w0, [x23], #1
    mov     w0, #10
    strb    w0, [x23], #1
    ret

    .bss
    .balign 8
inbuf:
    .zero 65536
    .balign 8
vstack:
    .zero 32768
    .balign 8
outbuf:
    .zero 524288
