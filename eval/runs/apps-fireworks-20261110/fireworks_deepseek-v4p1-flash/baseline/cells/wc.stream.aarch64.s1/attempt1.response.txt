.globl _start

.section .bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    // read loop
    mov x19, #0          // total bytes read
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
read_loop:
    mov x0, #0           // fd 0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63          // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop
read_done:
    // count lines and words
    mov x21, #0          // lines
    mov x22, #0          // words
    mov x23, #0          // in_word flag
    mov x24, #0          // index
    cmp x19, #0
    b.eq count_done
count_loop:
    ldrb w25, [x20, x24]
    cmp w25, #10
    b.ne 1f
    add x21, x21, #1
1:
    cmp w25, #0x20
    b.eq is_ws
    cmp w25, #0x09
    b.eq is_ws
    cmp w25, #0x0A
    b.eq is_ws
    cmp w25, #0x0B
    b.eq is_ws
    cmp w25, #0x0C
    b.eq is_ws
    cmp w25, #0x0D
    b.eq is_ws
    // not whitespace
    cmp x23, #0
    b.ne 2f
    mov x23, #1
    add x22, x22, #1
2:
    b 3f
is_ws:
    mov x23, #0
3:
    add x24, x24, #1
    cmp x24, x19
    b.lo count_loop
count_done:
    // build output
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf
    mov x26, x20
    // lines
    mov x0, x21
    mov x1, x26
    bl utoa
    add x26, x26, x2
    mov w0, #0x20
    strb w0, [x26]
    add x26, x26, #1
    // words
    mov x0, x22
    mov x1, x26
    bl utoa
    add x26, x26, x2
    mov w0, #0x20
    strb w0, [x26]
    add x26, x26, #1
    // bytes
    mov x0, x19
    mov x1, x26
    bl utoa
    add x26, x26, x2
    mov w0, #10
    strb w0, [x26]
    add x26, x26, #1
    // write
    mov x0, #1
    mov x1, x20
    sub x2, x26, x20
    mov x8, #64
    svc #0
    // exit
    mov x0, #0
    mov x8, #93
    svc #0

utoa:
    sub sp, sp, #32
    mov x3, x0
    mov x4, x1
    mov x5, #0
    cbnz x3, 1f
    mov w6, #48
    strb w6, [x4]
    mov x2, #1
    add sp, sp, #32
    ret
1:
    mov x6, #10
2:
    udiv x7, x3, x6
    msub x8, x7, x6, x3
    add x8, x8, #48
    strb w8, [sp, x5]
    add x5, x5, #1
    mov x3, x7
    cbnz x3, 2b
    mov x9, #0
3:
    sub x10, x5, x9
    sub x10, x10, #1
    ldrb w11, [sp, x10]
    strb w11, [x4, x9]
    add x9, x9, #1
    cmp x9, x5
    b.lo 3b
    mov x2, x5
    add sp, sp, #32
    ret
