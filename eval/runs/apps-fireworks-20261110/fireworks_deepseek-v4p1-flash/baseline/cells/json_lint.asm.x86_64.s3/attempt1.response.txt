```asm
.globl _start

.section .text
_start:
    # read input
    lea rbx, [rip + input_buf]
    xor r12, r12
.L_start_read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    jz .L_start_read_done
    syscall
    cmp rax, 0
    jle .L_start_read_done
    add r12, rax
    jmp .L_start_read_loop
.L_start_read_done:
    mov [rip + input_len], r12
    lea rax, [rip + input_buf]
    mov [rip + cur_ptr], rax
    add rax, r12
    mov [rip + end_ptr], rax

    call parse_value
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_start_valid
    jmp report_error

.L_start_valid:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    mov eax, 60
    mov edi, 0
    syscall

skip_ws:
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
.L_skip_ws_loop:
    cmp rsi, rdi
    jae .L_skip_ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je .L_skip_ws_advance
    cmp al, 0x09
    je .L_skip_ws_advance
    cmp al, 0x0A
    je .L_skip_ws_advance
    cmp al, 0x0D
    je .L_skip_ws_advance
    jmp .L_skip_ws_done
.L_skip_ws_advance:
    inc rsi
    jmp .L_skip_ws_loop
.L_skip_ws_done:
    mov [rip + cur_ptr], rsi
    ret

parse_value:
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_value_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb .L_parse_value_check_other
    cmp al, '9'
    jbe parse_number
.L_parse_value_check_other:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp report_error
.L_parse_value_error_end:
    jmp report_error

parse_object:
    inc rsi
    mov [rip + cur_ptr], rsi
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_object_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '}'
    je .L_parse_object_empty
.L_parse_object_member_loop:
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_object_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    jne report_error
    call parse_string
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_object_error_end
    movzx eax, byte ptr [rsi]
    cmp al, ':'
    jne report_error
    inc rsi
    mov [rip + cur_ptr], rsi
    call parse_value
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_object_error_end
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je .L_parse_object_next_member
    cmp al, '}'
    je .L_parse_object_close
    jmp report_error
.L_parse_object_next_member:
    inc rsi
    mov [rip + cur_ptr], rsi
    call skip_ws
    jmp .L_parse_object_member_loop
.L_parse_object_empty:
    inc rsi
    mov [rip + cur_ptr], rsi
    ret
.L_parse_object_close:
    inc rsi
    mov [rip + cur_ptr], rsi
    ret
.L_parse_object_error_end:
    jmp report_error

parse_array:
    inc rsi
    mov [rip + cur_ptr], rsi
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_array_error_end
    movzx eax, byte ptr [rsi]
    cmp al, ']'
    je .L_parse_array_empty
.L_parse_array_value_loop:
    call parse_value
    call skip_ws
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_array_error_end
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je .L_parse_array_next_value
    cmp al, ']'
    je .L_parse_array_close
    jmp report_error
.L_parse_array_next_value:
    inc rsi
    mov [rip + cur_ptr], rsi
    call skip_ws
    jmp .L_parse_array_value_loop
.L_parse_array_empty:
    inc rsi
    mov [rip + cur_ptr], rsi
    ret
.L_parse_array_close:
    inc rsi
    mov [rip + cur_ptr], rsi
    ret
.L_parse_array_error_end:
    jmp report_error

parse_string:
    inc rsi
    mov [rip + cur_ptr], rsi
.L_parse_string_loop:
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_string_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je .L_parse_string_close
    cmp al, '\\'
    je .L_parse_string_escape
    cmp al, 0x20
    jb report_error
    inc rsi
    mov [rip + cur_ptr], rsi
    jmp .L_parse_string_loop
.L_parse_string_escape:
    inc rsi
    mov [rip + cur_ptr], rsi
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_string_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je .L_parse_string_valid_escape
    cmp al, '\\'
    je .L_parse_string_valid_escape
    cmp al, '/'
    je .L_parse_string_valid_escape
    cmp al, 'b'
    je .L_parse_string_valid_escape
    cmp al, 'f'
    je .L_parse_string_valid_escape
    cmp al, 'n'
    je .L_parse_string_valid_escape
    cmp al, 'r'
    je .L_parse_string_valid_escape
    cmp al, 't'
    je .L_parse_string_valid_escape
    cmp al, 'u'
    je .L_parse_string_unicode_escape
    jmp report_error
.L_parse_string_valid_escape:
    inc rsi
    mov [rip + cur_ptr], rsi
    jmp .L_parse_string_loop
.L_parse_string_unicode_escape:
    inc rsi
    mov [rip + cur_ptr], rsi
    mov ecx, 4
.L_parse_string_hex_loop:
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    cmp rsi, rdi
    jae .L_parse_string_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .L_parse_string_not_hex
    cmp al, '9'
    jbe .L_parse_string_is_hex
    cmp al, 'a'
    jb .L_parse_string_not_hex
    cmp al, 'f'
    jbe .L_parse_string_is_hex
    cmp al, 'A'
    jb .L_parse_string_not_hex
    cmp al, 'F'
    jbe .L_parse_string_is_hex
.L_parse_string_not_hex:
    jmp report_error
.L_parse_string_is_hex:
    inc rsi
    mov [rip + cur_ptr], rsi
    dec ecx
    jnz .L_parse_string_hex_loop
    jmp .L_parse_string_loop
.L_parse_string_close:
    inc rsi
    mov [rip + cur_ptr], rsi
    ret
.L_parse_string_error_end:
    jmp report_error

parse_number:
    mov rsi, [rip + cur_ptr]
    mov rdi, [rip + end_ptr]
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .L_parse_number_int_part
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_error_end
.L_parse_number_int_part:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je .L_parse_number_zero
    cmp al, '1'
    jb .L_parse_number_error_cur
    cmp al, '9'
    jbe .L_parse_number_digits
    jmp .L_parse_number_error_cur
.L_parse_number_zero:
    inc rsi
    mov [rip + cur_ptr], rsi
    jmp .L_parse_number_frac
.L_parse_number_digits:
    inc rsi
    mov [rip + cur_ptr], rsi
.L_parse_number_digit_loop:
    cmp rsi, rdi
    jae .L_parse_number_frac
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .L_parse_number_frac
    cmp al, '9'
    ja .L_parse_number_frac
    inc rsi
    mov [rip + cur_ptr], rsi
    jmp .L_parse_number_digit_loop
.L_parse_number_frac:
    cmp rsi, rdi
    jae .L_parse_number_exp
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne .L_parse_number_exp
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .L_parse_number_error_cur
    cmp al, '9'
    ja .L_parse_number_error_cur
.L_parse_number_frac_loop:
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_exp
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .L_parse_number_exp
    cmp al, '9'
    ja .L_parse_number_exp
    jmp .L_parse_number_frac_loop
.L_parse_number_exp:
    cmp rsi, rdi
    jae .L_parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, 'e'
    je .L_parse_number_exp_start
    cmp al, 'E'
    je .L_parse_number_exp_start
    jmp .L_parse_number_done
.L_parse_number_exp_start:
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_error_end
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .L_parse_number_exp_sign
    cmp al, '-'
    je .L_parse_number_exp_sign
    jmp .L_parse_number_exp_digit
.L_parse_number_exp_sign:
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_error_end
    movzx eax, byte ptr [rsi]
.L_parse_number_exp_digit:
    cmp al, '0'
    jb .L_parse_number_error_cur
    cmp al, '9'
    ja .L_parse_number_error_cur
.L_parse_number_exp_loop:
    inc rsi
    mov [rip + cur_ptr], rsi
    cmp rsi, rdi
    jae .L_parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .L_parse_number_done
    cmp al, '9'
    ja .L_parse_number_done
    jmp .L_parse_number_exp_loop
.L_parse_number_done:
    ret
.L_parse_number_error_cur:
    jmp report_error
.L_parse_number_error_end:
    jmp report_error

parse_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp expect_literal
parse_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp expect_literal
parse_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp expect_literal

expect_literal:
    mov rdi, [rip + cur_ptr]
    mov r8, [rip + end_ptr]
    xor edx, edx
.L_expect_literal_loop:
    cmp edx, ecx
    jae .L_expect_literal_success
    cmp rdi, r8
    jae .L_expect_literal_error_end
    movzx eax, byte ptr [rdi]
    cmp al, byte ptr [rsi + rdx]
    jne .L_expect_literal_error_cur
    inc rdi
    inc edx
    jmp .L_expect_literal_loop
.L_expect_literal_success:
    mov [rip + cur_ptr], rdi
    ret
.L_expect_literal_error_cur:
    mov [rip + cur_ptr], rdi
    jmp report_error
.L_expect_literal_error_end:
    mov [rip + cur_ptr], rdi
    jmp report_error

report_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_error_at]
    mov rdx, msg_error_at_len
    syscall
    mov rax, [rip + cur_ptr]
    lea rbx, [rip + input_buf]
    sub rax, rbx
    call print_decimal
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_newline]
    mov rdx, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

print_decimal:
    push rbx
    push rcx
    push rdx
    push rsi
    push rdi
    sub rsp, 32
    mov rbx, rsp
    mov rcx, 0
    mov rsi, 10
    test rax, rax
    jnz .L_print_decimal_loop
    mov byte ptr [rbx], '0'
    mov rcx, 1
    jmp .L_print_decimal_write
.L_print_decimal_loop:
    xor rdx, rdx
    div rsi
    add dl, '0'
    mov [rbx + rcx], dl
    inc rcx
    test rax, rax
    jnz .L_print_decimal_loop
    mov rsi, rbx
    lea rdi, [rbx + rcx - 1]
.L_print_decimal_reverse_loop:
    cmp rsi, rdi
    jae .L_print_decimal_write
    mov al, [rsi]
    mov dl, [rdi]
    mov [rsi], dl
    mov [rdi], al
    inc rsi
    dec rdi
    jmp .L_print_decimal_reverse_loop
.L_print_decimal_write:
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, rcx
    syscall
    add rsp, 32
    pop rdi
    pop rsi
    pop rdx
    pop rcx
    pop rbx
    ret

.section .rodata
str_true: .ascii "true"
str_false: .ascii "false"
str_null: .ascii "null"
msg_ok: .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_error_at: .ascii "error at "
msg_error_at_len = . - msg_error_at
msg_newline: .ascii "\n"

.section .bss
.align 8
input_buf: .zero 65536
input_len: .quad 0
cur_ptr: .quad 0
end_ptr: .quad 0
```
