.globl _start

.equ C_GT, 0x3e
.equ C_LT, 0x3c
.equ C_PLUS, 0x2b
.equ C_MINUS, 0x2d
.equ C_DOT, 0x2e
.equ C_COMMA, 0x2c
.equ C_LB, 0x5b
.equ C_RB, 0x5d
.equ C_BANG, 0x21

.text
_start:
    // Set up input buffer base
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf

    // Read all stdin into input_buf
    mov x19, #0                 // offset
    movz x20, #1, lsl #16       // max size = 65536
read_loop:
    mov x0, #0                  // fd 0
    add x1, x21, x19            // buf = input_buf + offset
    sub x2, x20, x19            // count = max - offset
    mov x8, #63                 // SYS_read
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    b read_loop
read_done:
    // x19 = total_len

    // Find first '!'
    mov x20, #0                 // index i
find_bang:
    cmp x20, x19
    b.ge no_bang
    ldrb w22, [x21, x20]
    cmp w22, #C_BANG
    b.eq found_bang
    add x20, x20, #1
    b find_bang

no_bang:
    mov x22, x19                // program_len = total_len
    mov x23, x19                // input_start = total_len
    mov x24, #0                 // input_len = 0
    b balance_check

found_bang:
    mov x22, x20                // program_len = i
    add x23, x20, #1            // input_start = i+1
    sub x24, x19, x23           // input_len = total_len - input_start
    b balance_check

balance_check:
    adrp x27, jump_table
    add x27, x27, :lo12:jump_table
    adrp x28, bracket_stack
    add x28, x28, :lo12:bracket_stack
    mov x25, #0                 // i = 0
    mov x26, #0                 // stack pointer = 0
balance_loop:
    cmp x25, x22
    b.ge balance_done
    ldrb w0, [x21, x25]
    cmp w0, #C_LB
    b.eq push_bracket
    cmp w0, #C_RB
    b.eq pop_bracket
    b balance_next
push_bracket:
    str w25, [x28, x26, lsl #2]
    add x26, x26, #1
    b balance_next
pop_bracket:
    cbz x26, unbalanced
    sub x26, x26, #1
    ldr w0, [x28, x26, lsl #2]
    str w25, [x27, x0, lsl #2]
    str w0, [x27, x25, lsl #2]
    b balance_next
balance_next:
    add x25, x25, #1
    b balance_loop
balance_done:
    cbnz x26, unbalanced
    b run_program

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0

run_program:
    adrp x20, tape
    add x20, x20, :lo12:tape
    adrp x29, output_buf
    add x29, x29, :lo12:output_buf
    add x30, x21, x23           // input data base
    mov x25, #0                 // program_counter = 0
    mov x26, #0                 // tape_ptr = 0
    mov x28, #0                 // input_ptr = 0
    mov x19, #0                 // out_len = 0

exec_loop:
    cmp x25, x22
    b.ge exec_done
    ldrb w0, [x21, x25]
    cmp w0, #C_GT
    b.eq cmd_gt
    cmp w0, #C_LT
    b.eq cmd_lt
    cmp w0, #C_PLUS
    b.eq cmd_plus
    cmp w0, #C_MINUS
    b.eq cmd_minus
    cmp w0, #C_DOT
    b.eq cmd_dot
    cmp w0, #C_COMMA
    b.eq cmd_comma
    cmp w0, #C_LB
    b.eq cmd_lb
    cmp w0, #C_RB
    b.eq cmd_rb
    b exec_next

cmd_gt:
    add x26, x26, #1
    b exec_next
cmd_lt:
    sub x26, x26, #1
    b exec_next
cmd_plus:
    ldrb w1, [x20, x26]
    add w1, w1, #1
    strb w1, [x20, x26]
    b exec_next
cmd_minus:
    ldrb w1, [x20, x26]
    sub w1, w1, #1
    strb w1, [x20, x26]
    b exec_next
cmd_dot:
    ldrb w1, [x20, x26]
    strb w1, [x29, x19]
    add x19, x19, #1
    b exec_next
cmd_comma:
    cmp x28, x24
    b.ge comma_zero
    ldrb w1, [x30, x28]
    add x28, x28, #1
    strb w1, [x20, x26]
    b exec_next
comma_zero:
    strb wzr, [x20, x26]
    b exec_next
cmd_lb:
    ldrb w1, [x20, x26]
    cbnz w1, exec_next
    ldr w25, [x27, x25, lsl #2]
    b exec_next
cmd_rb:
    ldrb w1, [x20, x26]
    cbz w1, exec_next
    ldr w25, [x27, x25, lsl #2]
    b exec_next

exec_next:
    add x25, x25, #1
    b exec_loop

exec_done:
    cbz x19, exit_zero
    mov x0, #1
    mov x1, x29
    mov x2, x19
    mov x8, #64
    svc #0
exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.align 4
input_buf:
    .zero 65536
.align 4
tape:
    .zero 30000
.align 4
jump_table:
    .zero 262144
.align 4
bracket_stack:
    .zero 4096
.align 4
output_buf:
    .zero 65536
</final_code>
