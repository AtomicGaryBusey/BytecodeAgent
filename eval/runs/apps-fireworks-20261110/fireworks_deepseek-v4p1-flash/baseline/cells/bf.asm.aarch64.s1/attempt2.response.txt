.globl _start

.equ SYS_READ,  63
.equ SYS_WRITE, 64
.equ SYS_EXIT,  93

.equ C_GT,    0x3e
.equ C_LT,    0x3c
.equ C_PLUS,  0x2b
.equ C_MINUS, 0x2d
.equ C_DOT,   0x2e
.equ C_COMMA, 0x2c
.equ C_LB,    0x5b
.equ C_RB,    0x5d
.equ C_BANG,  0x21

.text
_start:
    // read all of stdin into input_buf
    adrp x21, input_buf
    add  x21, x21, :lo12:input_buf
    mov  x19, #0
read_loop:
    mov  x0, #0
    add  x1, x21, x19
    movz x2, #1, lsl #16
    sub  x2, x2, x19
    mov  x8, #SYS_READ
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x19, x19, x0
    b    read_loop
read_done:
    // locate the first '!' byte
    mov  x20, #0
find_bang:
    cmp  x20, x19
    b.ge no_bang
    ldrb w0, [x21, x20]
    cmp  w0, #C_BANG
    b.eq found_bang
    add  x20, x20, #1
    b    find_bang
no_bang:
    mov  x22, x19
    mov  x23, x19
    mov  x24, #0
    b    scan_start
found_bang:
    mov  x22, x20
    add  x23, x20, #1
    sub  x24, x19, x23
scan_start:
    // build bracket jump table, checking balance
    adrp x27, jump_table
    add  x27, x27, :lo12:jump_table
    adrp x28, bracket_stack
    add  x28, x28, :lo12:bracket_stack
    mov  x25, #0
    mov  x26, #0
scan_loop:
    cmp  x25, x22
    b.ge scan_done
    ldrb w0, [x21, x25]
    cmp  w0, #C_LB
    b.eq scan_push
    cmp  w0, #C_RB
    b.eq scan_pop
    b    scan_next
scan_push:
    str  w25, [x28, x26, lsl #2]
    add  x26, x26, #1
    b    scan_next
scan_pop:
    cbz  x26, unbalanced
    sub  x26, x26, #1
    ldr  w0, [x28, x26, lsl #2]
    str  w25, [x27, x0, lsl #2]
    str  w0,  [x27, x25, lsl #2]
    b    scan_next
scan_next:
    add  x25, x25, #1
    b    scan_loop
scan_done:
    cbnz x26, unbalanced
    b    run_start
unbalanced:
    mov  x0, #1
    mov  x8, #SYS_EXIT
    svc  #0

run_start:
    adrp x20, tape
    add  x20, x20, :lo12:tape
    adrp x29, out_buf
    add  x29, x29, :lo12:out_buf
    add  x18, x21, x23
    mov  x25, #0
    mov  x26, #0
    mov  x28, #0
    mov  x19, #0
exec_loop:
    cmp  x25, x22
    b.ge exec_done
    ldrb w0, [x21, x25]
    cmp  w0, #C_GT
    b.eq op_gt
    cmp  w0, #C_LT
    b.eq op_lt
    cmp  w0, #C_PLUS
    b.eq op_plus
    cmp  w0, #C_MINUS
    b.eq op_minus
    cmp  w0, #C_DOT
    b.eq op_dot
    cmp  w0, #C_COMMA
    b.eq op_comma
    cmp  w0, #C_LB
    b.eq op_lb
    cmp  w0, #C_RB
    b.eq op_rb
    b    exec_next
op_gt:
    add  x26, x26, #1
    b    exec_next
op_lt:
    sub  x26, x26, #1
    b    exec_next
op_plus:
    ldrb w1, [x20, x26]
    add  w1, w1, #1
    strb w1, [x20, x26]
    b    exec_next
op_minus:
    ldrb w1, [x20, x26]
    sub  w1, w1, #1
    strb w1, [x20, x26]
    b    exec_next
op_dot:
    ldrb w1, [x20, x26]
    strb w1, [x29, x19]
    add  x19, x19, #1
    b    exec_next
op_comma:
    cmp  x28, x24
    b.ge op_comma_zero
    ldrb w1, [x18, x28]
    add  x28, x28, #1
    strb w1, [x20, x26]
    b    exec_next
op_comma_zero:
    strb wzr, [x20, x26]
    b    exec_next
op_lb:
    ldrb w1, [x20, x26]
    cbnz w1, exec_next
    ldr  w25, [x27, x25, lsl #2]
    b    exec_next
op_rb:
    ldrb w1, [x20, x26]
    cbz  w1, exec_next
    ldr  w25, [x27, x25, lsl #2]
    b    exec_next
exec_next:
    add  x25, x25, #1
    b    exec_loop
exec_done:
    cbz  x19, all_done
    mov  x0, #1
    mov  x1, x29
    mov  x2, x19
    mov  x8, #SYS_WRITE
    svc  #0
all_done:
    mov  x0, #0
    mov  x8, #SYS_EXIT
    svc  #0

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
tape:
    .zero 30000
.balign 16
jump_table:
    .zero 262144
.balign 16
bracket_stack:
    .zero 262144
.balign 16
out_buf:
    .zero 65536
