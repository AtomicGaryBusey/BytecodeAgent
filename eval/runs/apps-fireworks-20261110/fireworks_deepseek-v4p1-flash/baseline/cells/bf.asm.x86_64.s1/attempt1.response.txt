.text
.globl _start
_start:
    xor r12, r12
.read_loop:
    cmp r12, 65536
    jae .read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    lea r11, [rip + input_buffer]
    xor rcx, rcx
    mov r8, r12
    mov r9, r12
.find_bang:
    cmp rcx, r12
    jae .no_bang
    cmp byte ptr [r11 + rcx], 0x21
    je .found_bang
    inc rcx
    jmp .find_bang
.found_bang:
    mov r8, rcx
    lea r9, [rcx + 1]
.no_bang:
    lea rdi, [rip + jump_table]
    lea rsi, [rip + bracket_stack]
    xor rcx, rcx
    xor r10, r10
.check_loop:
    cmp rcx, r8
    jae .check_done
    mov al, byte ptr [r11 + rcx]
    cmp al, '['
    je .open_bracket
    cmp al, ']'
    je .close_bracket
    inc rcx
    jmp .check_loop
.open_bracket:
    mov dword ptr [rsi + r10*4], ecx
    inc r10
    inc rcx
    jmp .check_loop
.close_bracket:
    test r10, r10
    jz .unbalanced
    dec r10
    mov edx, dword ptr [rsi + r10*4]
    lea eax, [rcx + 1]
    mov dword ptr [rdi + rdx*4], eax
    lea eax, [rdx + 1]
    mov dword ptr [rdi + rcx*4], eax
    inc rcx
    jmp .check_loop
.check_done:
    test r10, r10
    jnz .unbalanced
    mov r10, r12
    lea r15, [rip + input_buffer]
    lea r14, [rip + tape]
    xor r13, r13
    xor r12, r12
    mov rbx, r9
    lea rbp, [rip + jump_table]
.run_loop:
    cmp r12, r8
    jae .program_done
    movzx eax, byte ptr [r15 + r12]
    inc r12
    cmp al, '>'
    je .cmd_right
    cmp al, '<'
    je .cmd_left
    cmp al, '+'
    je .cmd_inc
    cmp al, '-'
    je .cmd_dec
    cmp al, '.'
    je .cmd_out
    cmp al, ','
    je .cmd_in
    cmp al, '['
    je .cmd_open
    cmp al, ']'
    je .cmd_close
    jmp .run_loop
.cmd_right:
    inc r13
    jmp .run_loop
.cmd_left:
    dec r13
    jmp .run_loop
.cmd_inc:
    inc byte ptr [r14 + r13]
    jmp .run_loop
.cmd_dec:
    dec byte ptr [r14 + r13]
    jmp .run_loop
.cmd_out:
    mov rax, 1
    mov rdi, 1
    lea rsi, [r14 + r13]
    mov rdx, 1
    syscall
    jmp .run_loop
.cmd_in:
    cmp rbx, r10
    jae .input_exhausted
    movzx eax, byte ptr [r15 + rbx]
    inc rbx
    jmp .store_in
.input_exhausted:
    xor eax, eax
.store_in:
    mov byte ptr [r14 + r13], al
    jmp .run_loop
.cmd_open:
    cmp byte ptr [r14 + r13], 0
    jne .run_loop
    mov eax, dword ptr [rbp + r12*4 - 4]
    mov r12, rax
    jmp .run_loop
.cmd_close:
    cmp byte ptr [r14 + r13], 0
    je .run_loop
    mov eax, dword ptr [rbp + r12*4 - 4]
    mov r12, rax
    jmp .run_loop
.program_done:
    mov rax, 60
    xor rdi, rdi
    syscall
.unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
tape:
    .zero 30000
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
