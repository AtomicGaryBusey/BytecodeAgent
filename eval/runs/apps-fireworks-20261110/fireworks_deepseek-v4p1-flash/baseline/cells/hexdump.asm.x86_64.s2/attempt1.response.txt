.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 256

.text
.globl _start
_start:
    lea rbx, [rip + hexdigits]
    xor r12, r12        # total bytes read

read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0          # sys_read
    xor edi, edi        # fd = 0
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13        # offset = 0
    lea r15, [rip + buffer]   # row start pointer

process_loop:
    cmp r13, r12
    jae print_total

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe .n_ok
    mov r14, 16
.n_ok:

    lea rdi, [rip + outbuf]

    # print offset
    mov eax, r13d
    mov ecx, 28
.print_offset:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns .print_offset

    mov byte ptr [rdi], 32
    mov byte ptr [rdi+1], 32
    add rdi, 2

    # hex bytes
    xor ecx, ecx        # i = 0
.hex_loop:
    cmp ecx, 16
    jge .hex_done
    cmp ecx, r14d
    jge .print_spaces

    movzx eax, byte ptr [r15 + rcx]
    mov edx, eax
    shr edx, 4
    and edx, 0xF
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 0xF
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    jmp .next_hex

.print_spaces:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi+1], 32
    mov byte ptr [rdi+2], 32
    add rdi, 3

.next_hex:
    cmp ecx, 7
    jne .no_extra
    mov byte ptr [rdi], 32
    inc rdi
.no_extra:
    inc ecx
    jmp .hex_loop

.hex_done:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124   # '|'
    inc rdi

    # characters
    xor ecx, ecx
.char_loop:
    cmp ecx, r14d
    jge .char_done
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 32
    jb .print_dot
    cmp al, 126
    ja .print_dot
    jmp .print_char
.print_dot:
    mov al, 46    # '.'
.print_char:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp .char_loop

.char_done:
    mov byte ptr [rdi], 124   # '|'
    inc rdi
    mov byte ptr [rdi], 10    # newline
    inc rdi

    # write row
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r13, r14
    add r15, r14
    jmp process_loop

print_total:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    mov ecx, 28
.print_total_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns .print_total_loop
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
