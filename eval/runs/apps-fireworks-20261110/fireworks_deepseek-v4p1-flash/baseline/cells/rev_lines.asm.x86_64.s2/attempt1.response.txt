.globl _start

.section .rodata
newline:
    .byte 10

.section .bss
    .align 16
buf:
    .zero 65536

.section .text
_start:
    xor r12, r12
    lea rbx, [rip + buf]

read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    xor r13, r13
    cmp r13, r12
    jae exit_ok

process_loop:
    mov rcx, r13
find_nl:
    cmp rcx, r12
    jae no_nl_found
    cmp byte ptr [rbx + rcx], 10
    je nl_found
    inc rcx
    jmp find_nl

no_nl_found:
    mov r14, r12
    mov r15, r12
    jmp process_line

nl_found:
    mov r14, rcx
    lea r15, [rcx + 1]

process_line:
    cmp r13, r14
    je reverse_done
    mov rdi, r13
    lea rsi, [r14 - 1]
reverse_loop:
    cmp rdi, rsi
    jae reverse_done
    mov al, byte ptr [rbx + rdi]
    mov dl, byte ptr [rbx + rsi]
    mov byte ptr [rbx + rdi], dl
    mov byte ptr [rbx + rsi], al
    inc rdi
    dec rsi
    jmp reverse_loop

reverse_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r13]
    mov rdx, r14
    sub rdx, r13
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov r13, r15
    cmp r13, r12
    jb process_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
